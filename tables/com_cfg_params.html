Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE COM_CFG_PARAMS (
    CMCP_FROM_CMC_ID    D_INTEGER NOT NULL,
    CMCP_PARAM_NAME     D_STR_CODE NOT NULL,
    CMCP_ORDER          D_INTEGER DEFAULT 0 NOT NULL,
    CMCP_CAPTION        D_STR_255 NOT NULL,
    CMCP_DESCRIPTION    D_STR_255 NOT NULL,
    CMCP_EDIT_KIND      D_STR_CODE DEFAULT 'text' NOT NULL,
    CMCP_MAX_LENGTH     D_INTEGER DEFAULT 0 NOT NULL,
    CMCP_DEFAULT_VALUE  D_STR_CODE NOT NULL,
    CMCP_DEFAULT_TEXT   D_STR_CODE NOT NULL,
    CMCP_READ_ONLY      D_SMALLINT DEFAULT 0 NOT NULL,
    CMCP_IS_HIDDEN      D_SMALLINT DEFAULT 0 NOT NULL,
    CMCP_COMBO_LIST     D_STR_8100 NOT NULL
);

ALTER TABLE COM_CFG_PARAMS ADD CONSTRAINT PK_COM_CFG_PARAMS PRIMARY KEY (CMCP_FROM_CMC_ID, CMCP_PARAM_NAME);
ALTER TABLE COM_CFG_PARAMS ADD CONSTRAINT FK_COM_CFG_PARAMS_CMC FOREIGN KEY (CMCP_FROM_CMC_ID) REFERENCES COM_CFG (CMC_ID) ON DELETE CASCADE ON UPDATE CASCADE;
