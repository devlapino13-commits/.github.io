Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE OKT_IVR_CALLS (
    OIC_ID             D_ID NOT NULL,
    OIC_FROM_OIS_ID    D_ID,
    OIC_CALL_DATETIME  D_DATETIME,
    OIC_CALL_DATE      D_DATE,
    OIC_CALL_RESULT    D_INTEGER,
    OIC_DURATION       D_INTEGER DEFAULT 0
);

ALTER TABLE OKT_IVR_CALLS ADD CONSTRAINT PK_OKT_IVR_CALLS PRIMARY KEY (OIC_ID);
ALTER TABLE OKT_IVR_CALLS ADD CONSTRAINT FK_OKT_IVR_CALLS_1 FOREIGN KEY (OIC_FROM_OIS_ID) REFERENCES OKT_IVR_CALLS_SVOD (OIS_ID) ON UPDATE CASCADE;
ALTER TABLE OKT_IVR_CALLS ADD CONSTRAINT FK_OKT_IVR_CALLS_3 FOREIGN KEY (OIC_CALL_RESULT) REFERENCES REFDATA (RD_ID) ON UPDATE CASCADE;
CREATE INDEX OKT_IVR_CALLS_IDX1 ON OKT_IVR_CALLS (OIC_CALL_DATE);
