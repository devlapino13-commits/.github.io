    IM_CONTROLLER_1_ID             D_ID,
    IM_CONTROLLER_1_SIGN_DATETIME  D_DATETIME,

    IM_CONTROLLER_2_SIGN_DATETIME  D_DATETIME,
    IM_STATUS                      D_SMALLINT DEFAULT 0 NOT NULL,
    IM_THROUGH_WHOM                D_STR_255 NOT NULL,
    IM_AMOUNT_PAYMENT              CURR_TYPE NOT NULL,
    IM_TYPE                        D_INTEGER DEFAULT 0 NOT NULL
);
