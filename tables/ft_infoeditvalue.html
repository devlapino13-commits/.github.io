CREATE TABLE FT_INFOEDITVALUE (
    FIE_ID           D_ID NOT NULL,
    FIE_FROM_C_ID    D_ID NOT NULL,
    FIE_FROM_FIF_ID  D_ID NOT NULL,
    FIE_DATE         D_DATE NOT NULL,
    FIE_EDIT_VALUE   D_STR_LONG,

    FIE_DATETIME     D_DATETIME
);

ALTER TABLE FT_INFOEDITVALUE ADD CONSTRAINT PK_FT_INFOEDITVALUE PRIMARY KEY (FIE_ID);
ALTER TABLE FT_INFOEDITVALUE ADD CONSTRAINT FK_FT_INFOEDITVALUE_C_ID FOREIGN KEY (FIE_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE

  USING DESCENDING INDEX FK_FT_INFOEDITVALUE_FIF;
