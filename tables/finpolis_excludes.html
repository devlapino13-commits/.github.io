Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE FINPOLIS_EXCLUDES (
    FPE_ID               D_INTEGER NOT NULL,
    FPE_FROM_C_ID        D_ID NOT NULL,
    FPE_NAMEU            D_STR_255 NOT NULL,
    FPE_GR               D_DATE NOT NULL,
    FPE_REASON           D_STR_255 NOT NULL,
    FPE_CREATE_DATETIME  D_DATETIME default current_timestamp NOT NULL,
    FPE_CREATE_UN_ID     D_ID NOT NULL,
    FPE_CHANGE_DATETIME  D_DATETIME default current_timestamp NOT NULL,
    FPE_CHANGE_UN_ID     D_ID NOT NULL,
    FPE_IS_DELETE        D_SMALLINT default 0 NOT NULL
);

ALTER TABLE FINPOLIS_EXCLUDES ADD CONSTRAINT PK_FINPOLIS_EXCLUDES PRIMARY KEY (FPE_ID);
ALTER TABLE FINPOLIS_EXCLUDES ADD CONSTRAINT FK_FINPOLIS_EXCLUDES_1 FOREIGN KEY (FPE_CREATE_UN_ID) REFERENCES USERNAME (UN_ID);
ALTER TABLE FINPOLIS_EXCLUDES ADD CONSTRAINT FK_FINPOLIS_EXCLUDES_2 FOREIGN KEY (FPE_CHANGE_UN_ID) REFERENCES USERNAME (UN_ID);
CREATE INDEX FINPOLIS_EXCLUDES_C_ID ON FINPOLIS_EXCLUDES (FPE_FROM_C_ID);
