Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE FT_INFOFIELD (
    FIF_ID               D_ID NOT NULL,
    FIF_FROM_FIB_ID      D_ID NOT NULL,
    FIF_FIELDNAME        D_STR_CODE NOT NULL,
    FIF_DATE             D_DATE,
    FIF_CAPTION          D_STR_100 NOT NULL,
    FIF_BLANK_CAPTION    D_STR_2048 NOT NULL,
    FIF_ORDER_NO         D_INTEGER DEFAULT 0 NOT NULL,
    FIF_IS_HIDDEN        D_SMALLINT DEFAULT 0,
    FIF_IS_DELETE        D_SMALLINT DEFAULT 0 NOT NULL,
    FIF_FIELD_KIND       D_STR_CODE,
    FIF_NON_CLIENT_DATA  D_SMALLINT DEFAULT 0
);

ALTER TABLE FT_INFOFIELD ADD CONSTRAINT PK_FT_INFOFIELD PRIMARY KEY (FIF_ID);
ALTER TABLE FT_INFOFIELD ADD CONSTRAINT FK_FT_INFOFIELD_FIB FOREIGN KEY (FIF_FROM_FIB_ID) REFERENCES FT_INFOBLANK (FIB_ID) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE UNIQUE INDEX FT_INFOFIELD_FIB_FN ON FT_INFOFIELD (FIF_FROM_FIB_ID, FIF_FIELDNAME);
