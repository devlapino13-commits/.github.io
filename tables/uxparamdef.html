Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE UXPARAMDEF (
    UPD_UFP_CODE   D_STR_100 NOT NULL,
    UPD_UXA_CODE   D_STR_CODE,
    UPD_DEF_VALUE  D_STR_100 NOT NULL,
    UPD_LOCKED     D_BOOL NOT NULL
);

ALTER TABLE UXPARAMDEF ADD CONSTRAINT UNQ1_UXPARAMDEF UNIQUE (UPD_UFP_CODE, UPD_UXA_CODE);
ALTER TABLE UXPARAMDEF ADD CONSTRAINT FK_UXPARAMDEF_UFP FOREIGN KEY (UPD_UFP_CODE) REFERENCES UXFRPARAM (UFP_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE UXPARAMDEF ADD CONSTRAINT FK_UXPARAMDEF_UXA FOREIGN KEY (UPD_UXA_CODE) REFERENCES UXALIAS (UXA_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
