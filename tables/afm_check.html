Показать DDL (CREATE TABLE) 
CREATE TABLE AFM_CHECK (
    AC_ID           D_ID GENERATED BY DEFAULT AS IDENTITY,
    AC_FROM_C_ID    D_ID NOT NULL,
    AC_FROM_MA_ID   D_ID NOT NULL,
    AC_FROM_UN_ID   D_ID NOT NULL,
    AC_DATE         D_DATE default current_date,
    AC_IS_ACTUAL    D_SMALLINT default 1,
    AC_CATEGORY_1   D_SMALLINT default 1,
    AC_CATEGORY_2   D_SMALLINT default 1,
    AC_CATEGORY_3   D_SMALLINT default 1,
    AC_CATEGORY_4   D_SMALLINT default 1,
    AC_CATEGORY_5   D_SMALLINT default 1,
    AC_CATEGORY_6   D_SMALLINT default 1,
    AC_CATEGORY_7   D_SMALLINT default 1,
    AC_CATEGORY_8   D_SMALLINT default 1,
    AC_CATEGORY_9   D_SMALLINT default 1,
    AC_CATEGORY_10  D_SMALLINT default 1,
    AC_CATEGORY_11  D_SMALLINT default 1,
    AC_CATEGORY_12  D_SMALLINT default 1,
    AC_CATEGORY_13  D_SMALLINT default 1,
    AC_CATEGORY_14  D_SMALLINT default 1,
    AC_CATEGORY_15  D_SMALLINT default 1
);

ALTER TABLE AFM_CHECK ADD PRIMARY KEY (AC_ID);
ALTER TABLE AFM_CHECK ADD CONSTRAINT FK_AC_C FOREIGN KEY (AC_FROM_C_ID) REFERENCES CLIENTS (C_ID);
ALTER TABLE AFM_CHECK ADD CONSTRAINT FK_AC_UN FOREIGN KEY (AC_FROM_UN_ID) REFERENCES USERNAME (UN_ID);
CREATE INDEX IDX_AC_DATE ON AFM_CHECK (AC_DATE);
CREATE INDEX IDX_AC_IS_ACTUAL ON AFM_CHECK (AC_IS_ACTUAL);
