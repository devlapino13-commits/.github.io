Показать DDL (CREATE TABLE) 
CREATE TABLE CC_WEBSITE_LO_SMS (
    WLS_ID           INTEGER NOT NULL,
    WLS_FROM_CK_ID   INTEGER,
    WLS_LO_ID        ID_TYPE,
    WLS_COUNTER      INTEGER,
    WLS_SMS_TXT      STR_255,
    WSL_FROM_WMA_ID  D_INTEGER
);

ALTER TABLE CC_WEBSITE_LO_SMS ADD CONSTRAINT PK_CC_WEBSITE_LO_SMS PRIMARY KEY (WLS_ID);
ALTER TABLE CC_WEBSITE_LO_SMS ADD CONSTRAINT FK_CC_WEBSITE_LO_SMS_1 FOREIGN KEY (WLS_FROM_CK_ID) REFERENCES CC_CHECK_KNCLIENT (CK_ID) ON UPDATE CASCADE;
ALTER TABLE CC_WEBSITE_LO_SMS ADD CONSTRAINT FK_CC_WEBSITE_LO_SMS_2 FOREIGN KEY (WLS_LO_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE CC_WEBSITE_LO_SMS ADD CONSTRAINT FK_CC_WEBSITE_LO_SMS_3 FOREIGN KEY (WSL_FROM_WMA_ID) REFERENCES CC_WEBSITE_MINIAPP (WMA_ID) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE DESCENDING INDEX CC_WEBSITE_LO_SMS_IDX1 ON CC_WEBSITE_LO_SMS (WLS_LO_ID, WLS_COUNTER);
