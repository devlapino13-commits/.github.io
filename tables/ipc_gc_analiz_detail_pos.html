Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE IPC_GC_ANALIZ_DETAIL_POS (
    IGDP_ID                         D_ID NOT NULL,
    IGDP_FROM_IGA_ID                D_INTEGER NOT NULL,
    IGDP_REC_NO                     D_INTEGER,
    IGDP_OBJECT_KIND                D_INTEGER,
    IGDP_OBJECT_ID                  D_ID,
    IGDP_FROM_C_ID                  D_ID,
    IGDP_FROM_AG_ID                 D_INTEGER,
    IGDP_TERMINAL_ID                D_INTEGER,
    IGDP_TYPE                       D_SMALLINT,
    IGDP_FROM_IB_ID                 D_INTEGER,
    IGDP_IS_OUR_CARD                D_BOOL,
    IGDP_CARD_NO                    D_STR_20,
    IGDP_TRANS_NO                   D_STR_10,
    IGDP_TRANS_DATE                 D_DATE,
    IGDP_CURRENCY                   D_STR_LANG,
    IGDP_AMOUNT                     D_CURRENCY,
    IGDP_TSP_COMMIS_PERC            D_PERCENT,
    IGDP_TSP_COMMIS_AMOUNT          D_CURRENCY,
    IGDP_IS_ERROR                   D_BOOL,
    IGDP_ERROR_ID                   D_INTEGER,
    IGDP_IPC_EMITENT_COMMIS_PERC    D_PERCENT,
    IGDP_IPC_EMITENT_COMMIS_AMOUNT  D_CURRENCY,
    IGDP_INET_ACQUIRING             D_SMALLINT DEFAULT 0 NOT NULL
);

ALTER TABLE IPC_GC_ANALIZ_DETAIL_POS ADD CONSTRAINT CHK1_IPC_GC_ANALIZ_DETAIL_POS CHECK (igdp_type between 1 and 2);
ALTER TABLE IPC_GC_ANALIZ_DETAIL_POS ADD CONSTRAINT PK_IPC_GC_ANALIZ_DETAIL_POS PRIMARY KEY (IGDP_ID);
ALTER TABLE IPC_GC_ANALIZ_DETAIL_POS ADD CONSTRAINT FK_IGDP_AG_ID FOREIGN KEY (IGDP_FROM_AG_ID) REFERENCES AGREEMENTS (AG_ID) ON UPDATE CASCADE;
ALTER TABLE IPC_GC_ANALIZ_DETAIL_POS ADD CONSTRAINT FK_IGDP_IGA_ID FOREIGN KEY (IGDP_FROM_IGA_ID) REFERENCES IPC_GC_ANALIZ (IGA_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE IPC_GC_ANALIZ_DETAIL_POS ADD CONSTRAINT FK_IGDP_POS_IB_ID FOREIGN KEY (IGDP_FROM_IB_ID) REFERENCES IPC_BANKS (IB_ID) ON UPDATE CASCADE;
ALTER TABLE IPC_GC_ANALIZ_DETAIL_POS ADD CONSTRAINT FK_IGDP_RD_ID FOREIGN KEY (IGDP_ERROR_ID) REFERENCES REFDATA (RD_ID) ON UPDATE CASCADE;
ALTER TABLE IPC_GC_ANALIZ_DETAIL_POS ADD CONSTRAINT FK_IGDP_TRM_ID FOREIGN KEY (IGDP_TERMINAL_ID) REFERENCES IPC_TERMINALS (TRM_ID) ON UPDATE CASCADE;
