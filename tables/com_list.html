Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE COM_LIST (
    CML_ID           D_ID NOT NULL,
    CML_FROM_CMC_ID  D_INTEGER DEFAULT 0 NOT NULL,
    CML_DATE         D_DATE NOT NULL,
    CML_FROM_REG_ID  D_INTEGER NOT NULL,
    CML_INIT_C_ID    D_ID NOT NULL,
    CML_OBJECT_KIND  D_INTEGER NOT NULL,
    CML_OBJECT_ID    D_ID NOT NULL,
    CML_ADD_PARAMS   D_STR_LONG,
    CML_LAST_CMA_ID  D_ID,
    CML_LAST_CMP_ID  D_INTEGER,
    CML_LAST_STATUS  D_SMALLINT,
    CML_IS_CLOSED    D_SMALLINT DEFAULT 0 NOT NULL,
    CML_FROM_UN_ID   D_ID NOT NULL,
    CML_FROM_KCP_ID  D_ID
);

ALTER TABLE COM_LIST ADD CONSTRAINT PK_COM_LIST PRIMARY KEY (CML_ID);
ALTER TABLE COM_LIST ADD CONSTRAINT FK_COM_LIST_CMA FOREIGN KEY (CML_LAST_CMA_ID) REFERENCES COM_ACTION (CMA_ID) ON DELETE SET NULL ON UPDATE CASCADE;
ALTER TABLE COM_LIST ADD CONSTRAINT FK_COM_LIST_CMC FOREIGN KEY (CML_FROM_CMC_ID) REFERENCES COM_CFG (CMC_ID) ON UPDATE CASCADE;
ALTER TABLE COM_LIST ADD CONSTRAINT FK_COM_LIST_CMP FOREIGN KEY (CML_LAST_CMP_ID) REFERENCES COM_PACKET (CMP_ID) ON UPDATE CASCADE;
ALTER TABLE COM_LIST ADD CONSTRAINT FK_COM_LIST_C_ID FOREIGN KEY (CML_INIT_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE COM_LIST ADD CONSTRAINT FK_COM_LIST_KCP FOREIGN KEY (CML_FROM_KCP_ID) REFERENCES KNCLIENT_PDN (KCP_ID) ON UPDATE CASCADE;
ALTER TABLE COM_LIST ADD CONSTRAINT FK_COM_LIST_REG FOREIGN KEY (CML_FROM_REG_ID) REFERENCES REGION (REG_UNICODE) ON UPDATE CASCADE;
ALTER TABLE COM_LIST ADD CONSTRAINT FK_COM_LIST_UN FOREIGN KEY (CML_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
CREATE INDEX COM_LIST_OBJ ON COM_LIST (CML_OBJECT_KIND, CML_OBJECT_ID);
