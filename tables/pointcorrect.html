Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE POINTCORRECT (
    PTO_FROM_C_ID   D_ID NOT NULL,
    PTO_NUMBER      D_INTEGER NOT NULL,
    PTO_MEMO        D_STR_LONG NOT NULL,
    PTO_FROM_UN_ID  D_ID NOT NULL,
    PTO_DATETIME    D_DATETIME NOT NULL
);

ALTER TABLE POINTCORRECT ADD CONSTRAINT PK_POINTCORRECT PRIMARY KEY (PTO_FROM_C_ID, PTO_NUMBER);
ALTER TABLE POINTCORRECT ADD CONSTRAINT FK_POINTCORRECT_C FOREIGN KEY (PTO_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE POINTCORRECT ADD CONSTRAINT FK_POINTCORRECT_PTC FOREIGN KEY (PTO_FROM_C_ID, PTO_NUMBER) REFERENCES POINTCALC (PTC_FROM_C_ID, PTC_NUMBER) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE POINTCORRECT ADD CONSTRAINT FK_POINTCORRECT_UN FOREIGN KEY (PTO_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE
  USING INDEX FK_POINTCORRECT_1;
