Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE BKI_BUFFER (
    BKB_ID           D_ID NOT NULL,
    BKB_FROM_BKK_ID  D_ID NOT NULL,
    BKB_OBJECT_KIND  D_INTEGER NOT NULL,
    BKB_OBJECT_ID    D_ID NOT NULL,
    BKB_DATE         D_DATE NOT NULL,
    BKB_TAG          D_SMALLINT NOT NULL,
    BKB_TYPE         D_SMALLINT DEFAULT 0 NOT NULL
);

ALTER TABLE BKI_BUFFER ADD CONSTRAINT PK_BKI_BUFFER PRIMARY KEY (BKB_ID);
ALTER TABLE BKI_BUFFER ADD CONSTRAINT FK_BKI_BUFFER_1 FOREIGN KEY (BKB_FROM_BKK_ID) REFERENCES BKI_KIND (BKK_ID);
CREATE INDEX BKI_BUFFER_IDX1 ON BKI_BUFFER (BKB_OBJECT_KIND, BKB_OBJECT_ID);
CREATE INDEX BKI_BUFFER_IDX2 ON BKI_BUFFER (BKB_DATE);
CREATE DESCENDING INDEX BKI_BUFFER_IDX_TYPE ON BKI_BUFFER (BKB_TYPE);
