Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE UXFRITEM (
    UFI_CODE             D_STR_100 NOT NULL,
    UFI_PARENT_CODE      D_STR_100,
    UFI_UX_CODE          D_STR_CODE NOT NULL,
    UFI_UFR_CODE         D_STR_CODE,
    UFI_UFP_CODE         D_STR_100,
    UFI_KIND             D_INTEGER NOT NULL,
    UFI_CONTROL          D_STR_CODE NOT NULL,
    UFI_POSITION         D_INTEGER NOT NULL,
    UFI_CAPTION          D_STR_CODE NOT NULL,
    UFI_CAPTION_ALIGN    D_INTEGER NOT NULL,
    UFI_HINT             D_STR_LONG NOT NULL,
    UFI_HIDDEN           D_BOOL NOT NULL,
    UFI_READONLY         D_BOOL NOT NULL,
    UFI_GROUP_DIRECTION  D_INTEGER NOT NULL,
    UFI_H_ALIGN          D_INTEGER NOT NULL,
    UFI_V_ALIGN          D_INTEGER NOT NULL,
    UFI_WIDTH            D_INTEGER NOT NULL,
    UFI_HEIGHT           D_INTEGER NOT NULL,
    UFI_ADD_PARAMS       D_STR_CODE NOT NULL
);

ALTER TABLE UXFRITEM ADD CONSTRAINT PK_UXFRITEM PRIMARY KEY (UFI_CODE);
ALTER TABLE UXFRITEM ADD CONSTRAINT FK_UXFRITEM_PARENT FOREIGN KEY (UFI_PARENT_CODE) REFERENCES UXFRITEM (UFI_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE UXFRITEM ADD CONSTRAINT FK_UXFRITEM_UFP_CODE FOREIGN KEY (UFI_UFP_CODE) REFERENCES UXFRPARAM (UFP_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE UXFRITEM ADD CONSTRAINT FK_UXFRITEM_UFR_CODE FOREIGN KEY (UFI_UFR_CODE) REFERENCES UXFRAME (UFR_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE UXFRITEM ADD CONSTRAINT FK_UXFRITEM_UX_CODE FOREIGN KEY (UFI_UX_CODE) REFERENCES UXCODE (UX_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE INDEX UXFRITEM_POSITION ON UXFRITEM (UFI_POSITION);
