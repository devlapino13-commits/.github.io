Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE RESTRICT_CRED_RULES (
    RCR_ID          INTEGER NOT NULL,
    RCR_FROM_C_ID   ID_TYPE NOT NULL,
    RCR_FROM_RD_ID  INTEGER NOT NULL,
    RCR_DATE_START  DATE NOT NULL,
    RCR_DATE_END    DATE NOT NULL,
    RCR_IS_ACTIVE   INTEGER default 1 NOT NULL,
    RCR_VALUE       DECIMAL(12,2),
    RCR_NOTE        STR_255
);

ALTER TABLE RESTRICT_CRED_RULES ADD CONSTRAINT PK_RESTRICT_CRED_RULES PRIMARY KEY (RCR_ID);
ALTER TABLE RESTRICT_CRED_RULES ADD CONSTRAINT FK_RESTRICT_CRED_RULES_1 FOREIGN KEY (RCR_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE RESTRICT_CRED_RULES ADD CONSTRAINT FK_RESTRICT_CRED_RULES_2 FOREIGN KEY (RCR_FROM_RD_ID) REFERENCES REFDATA (RD_ID) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE UNIQUE INDEX IDX_RESTRICT_CRED_RULES ON RESTRICT_CRED_RULES (RCR_FROM_C_ID, RCR_FROM_RD_ID);
