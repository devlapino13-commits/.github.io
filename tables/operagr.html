Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE OPERAGR (
    OGR_ID            D_ID NOT NULL,
    OGR_FROM_OL_ID    D_ID NOT NULL,
    OGR_FROM_AG_ID    D_INTEGER NOT NULL,
    OGR_FROM_RD_ID    D_INTEGER NOT NULL,
    OGR_SIGN          D_SMALLINT NOT NULL,
    OGR_AMOUNT        D_CURRENCY NOT NULL,
    OGR_TAX_FL        D_CURRENCY NOT NULL,
    OGR_OBJECT_KIND   D_INTEGER,
    OGR_OBJECT_ID     D_ID,
    OGR_OL_DATE       D_DATE NOT NULL,
    OGR_OL_IS_ACTIVE  D_SMALLINT DEFAULT 1 NOT NULL
);

ALTER TABLE OPERAGR ADD CONSTRAINT CHK1_OPERAGR CHECK (ogr_sign in (-1, 1));
ALTER TABLE OPERAGR ADD CONSTRAINT PK_OPERAGR PRIMARY KEY (OGR_ID);
ALTER TABLE OPERAGR ADD CONSTRAINT FK_OPERAGR_AG FOREIGN KEY (OGR_FROM_AG_ID) REFERENCES AGREEMENTS (AG_ID) ON UPDATE CASCADE;
ALTER TABLE OPERAGR ADD CONSTRAINT FK_OPERAGR_RD FOREIGN KEY (OGR_FROM_RD_ID) REFERENCES REFDATA (RD_ID) ON UPDATE CASCADE;
CREATE INDEX AFK_OPERAGR_OL_ID ON OPERAGR (OGR_FROM_OL_ID);
CREATE INDEX OPERAGR_OBJ ON OPERAGR (OGR_OBJECT_KIND, OGR_OBJECT_ID);
CREATE INDEX OPERAGR_OL_DATE ON OPERAGR (OGR_OL_DATE);
