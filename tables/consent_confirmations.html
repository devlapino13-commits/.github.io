Показать DDL (CREATE TABLE) 
CREATE TABLE CONSENT_CONFIRMATIONS (
    CC_ID               D_INTEGER NOT NULL,
    CC_FROM_C_ID        D_ID NOT NULL,
    CC_CREATE_DATETIME  D_DATETIME NOT NULL,
    CC_STATUS           D_SMALLINT NOT NULL,
    CC_STATUS_DATETIME  D_DATETIME NOT NULL,
    CC_FROM_MCL_ID      D_ID,
    CC_FROM_UN_ID       D_ID
);

ALTER TABLE CONSENT_CONFIRMATIONS ADD CONSTRAINT PK_CONSENT_CONFIRMATIONS PRIMARY KEY (CC_ID);
ALTER TABLE CONSENT_CONFIRMATIONS ADD CONSTRAINT FK_CONSENT_CONFIRMATIONS_C_ID FOREIGN KEY (CC_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE CONSENT_CONFIRMATIONS ADD CONSTRAINT FK_CONSENT_CONFIRMATIONS_MCL_ID FOREIGN KEY (CC_FROM_MCL_ID) REFERENCES MC_LOGIN (MCL_ID) ON UPDATE CASCADE;
ALTER TABLE CONSENT_CONFIRMATIONS ADD CONSTRAINT FK_CONSENT_CONFIRMATIONS_UN_ID FOREIGN KEY (CC_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
