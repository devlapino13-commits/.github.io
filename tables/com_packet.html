Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE COM_PACKET (
    CMP_ID           D_INTEGER NOT NULL,
    CMP_STR_NO       D_STR_CODE NOT NULL,
    CMP_DATE         D_DATE NOT NULL,
    CMP_FROM_REG_ID  D_INTEGER NOT NULL,
    CMP_FROM_CMK_ID  D_INTEGER DEFAULT 0 NOT NULL,
    CMP_FROM_CMD_ID  D_INTEGER DEFAULT 0 NOT NULL,
    CMP_STATUS       D_SMALLINT DEFAULT 0 NOT NULL,
    CMP_IS_IPF       D_SMALLINT DEFAULT 0,
    CMP_MEMO         D_STR_LONG
);

ALTER TABLE COM_PACKET ADD CONSTRAINT CHK1_COM_PACKET_STATUS CHECK (cmp_status in (0, 1));
ALTER TABLE COM_PACKET ADD CONSTRAINT PK_COM_PACKET PRIMARY KEY (CMP_ID);
ALTER TABLE COM_PACKET ADD CONSTRAINT FK_COM_PACKET_CMD FOREIGN KEY (CMP_FROM_CMD_ID) REFERENCES COM_DECISION (CMD_ID) ON UPDATE CASCADE;
ALTER TABLE COM_PACKET ADD CONSTRAINT FK_COM_PACKET_CMK FOREIGN KEY (CMP_FROM_CMK_ID) REFERENCES COM_KIND (CMK_ID) ON UPDATE CASCADE;
ALTER TABLE COM_PACKET ADD CONSTRAINT FK_COM_PACKET_REG FOREIGN KEY (CMP_FROM_REG_ID) REFERENCES REGION (REG_UNICODE) ON UPDATE CASCADE;
CREATE UNIQUE INDEX COM_PACKET_STR_NO ON COM_PACKET (CMP_STR_NO);
