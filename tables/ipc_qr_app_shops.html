Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE IPC_QR_APP_SHOPS (
    IQS_ID           D_ID NOT NULL,
    IQS_NO           D_INTEGER NOT NULL,
    IQS_FROM_IQA_ID  D_ID NOT NULL,
    IQS_FROM_CS_ID   D_ID NOT NULL,
    IQS_TCP_DATA     D_STR_LONG NOT NULL,
    IQS_SHOP_NAME    D_STR_255 NOT NULL,
    IQS_HAS_QR       D_BOOL DEFAULT 0 NOT NULL
);

ALTER TABLE IPC_QR_APP_SHOPS ADD CONSTRAINT PK_IPC_QR_APP_SHOPS PRIMARY KEY (IQS_ID);
ALTER TABLE IPC_QR_APP_SHOPS ADD CONSTRAINT FK_IPC_QR_APP_SHOPS_1 FOREIGN KEY (IQS_FROM_IQA_ID) REFERENCES IPC_QR_APP (IQA_ID) ON UPDATE CASCADE;
ALTER TABLE IPC_QR_APP_SHOPS ADD CONSTRAINT FK_IPC_QR_APP_SHOPS_2 FOREIGN KEY (IQS_FROM_CS_ID) REFERENCES CLIENTSHOPS (CS_ID) ON UPDATE CASCADE;
CREATE UNIQUE INDEX IPC_QR_APP_SHOPS_NO ON IPC_QR_APP_SHOPS (IQS_FROM_IQA_ID, IQS_NO);
