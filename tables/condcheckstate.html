CREATE TABLE CONDCHECKSTATE (
    CCS_ID               D_INTEGER NOT NULL,
    CCS_OBJECT_KIND      D_INTEGER NOT NULL,
    CCS_OBJECT_ID        D_ID NOT NULL,
    CCS_CHECK_KIND       D_INTEGER NOT NULL,
