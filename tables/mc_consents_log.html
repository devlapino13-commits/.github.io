Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE MC_CONSENTS_LOG (
    MCOL_ID           ID_TYPE NOT NULL,
    MCOL_FROM_MCL_ID  ID_TYPE NOT NULL,
    MCOL_FROM_C_ID    ID_TYPE,
    MCOL_DATETIME     D_DATETIME NOT NULL
);

ALTER TABLE MC_CONSENTS_LOG ADD CONSTRAINT PK_MC_CONSENTS_LOG PRIMARY KEY (MCOL_ID);
ALTER TABLE MC_CONSENTS_LOG ADD CONSTRAINT FK_MC_CONSENTS_LOG_1 FOREIGN KEY (MCOL_FROM_MCL_ID) REFERENCES MC_LOGIN (MCL_ID) ON UPDATE CASCADE;
ALTER TABLE MC_CONSENTS_LOG ADD CONSTRAINT FK_MC_CONSENTS_LOG_2 FOREIGN KEY (MCOL_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
CREATE INDEX MC_CONSENTS_LOG_IDX1 ON MC_CONSENTS_LOG (MCOL_FROM_MCL_ID, MCOL_DATETIME);
