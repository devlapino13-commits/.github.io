CREATE TABLE EXTSYS_TOKEN (
    ET_ID               ID_TYPE NOT NULL,
    ET_FROM_ESM_ID      ID_TYPE,
    ET_FROM_C_ID        ID_TYPE,
    ET_CODE             VARCHAR(10),
    ET_TOKEN            STR_100,
    ET_CREATE_DATETIME  TIMESTAMP,
    ET_TOKEN_DATETIME   TIMESTAMP,
    ET_IS_ACTUAL        SMALLINT DEFAULT 0 NOT NULL,
    ET_CARDNUMBER       STR_100,
    ET_TOKEN_ESM_ID     ID_TYPE
);

ALTER TABLE EXTSYS_TOKEN ADD CONSTRAINT PK_EXTSYS_TOKEN PRIMARY KEY (ET_ID);
ALTER TABLE EXTSYS_TOKEN ADD CONSTRAINT FK_EXTSYS_TOKEN_1 FOREIGN KEY (ET_FROM_ESM_ID) REFERENCES EXTSYSMAIN (ESM_ID) ON UPDATE CASCADE;
ALTER TABLE EXTSYS_TOKEN ADD CONSTRAINT FK_EXTSYS_TOKEN_2 FOREIGN KEY (ET_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
CREATE INDEX AFK_EXTSYS_TOKEN_ESM_ID ON EXTSYS_TOKEN (ET_TOKEN_ESM_ID);
