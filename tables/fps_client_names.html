Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE FPS_CLIENT_NAMES (
    FCN_ID             D_INTEGER NOT NULL,
    FCN_FROM_FCB_ID    D_INTEGER NOT NULL,
    FCN_NAME           D_STR_100 NOT NULL,
    FCN_STATUS         D_SMALLINT NOT NULL,

ALTER TABLE FPS_CLIENT_NAMES ADD CONSTRAINT FK_FPS_CLIENT_NAMES_FCB FOREIGN KEY (FCN_FROM_FCB_ID) REFERENCES FPS_CLIENT_BANKS (FCB_ID) ON UPDATE CASCADE;
