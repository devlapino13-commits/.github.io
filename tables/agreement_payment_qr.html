CREATE TABLE AGREEMENT_PAYMENT_QR (
    APQ_ID               D_INTEGER NOT NULL,
    APQ_FROM_AG_ID       D_INTEGER NOT NULL,
    APQ_FROM_UN_ID       D_ID NOT NULL,
    APQ_STATUS           D_SMALLINT NOT NULL,
    APQ_IS_LAST          D_BOOL NOT NULL,
    APQ_QR_IMG_BLS_ID    D_ID,
    APQ_URL              D_STR_LONG,
    APQ_STATUS_DATETIME  D_DATETIME NOT NULL,
    APQ_DATETIME         D_DATETIME NOT NULL,
    APQ_ERROR_TEXT       D_STR_255,
    APQ_URL_BY_HANDS     D_BOOL DEFAULT 0 NOT NULL,
    APQ_URL_DATA         D_STR_8100
);
