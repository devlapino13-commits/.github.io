Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE CARDAKTMAIN (
    CAM_ID              D_INTEGER NOT NULL,
    CAM_DATE            D_DATE DEFAULT current_date NOT NULL,
    CAM_CARD_LIST       D_STR_8100 NOT NULL,
    CAM_CARD_COUNT      D_INTEGER NOT NULL,
    CAM_SRC_CPL_ID      D_INTEGER NOT NULL,
    CAM_DST_CPL_ID      D_INTEGER NOT NULL,
    CAM_DST_PIN_CPL_ID  D_INTEGER,
    CAM_STATUS          D_SMALLINT DEFAULT 0 NOT NULL,
    CAM_FROM_UN_ID      D_ID NOT NULL,
    CAM_DATETIME        D_DATETIME DEFAULT current_timestamp NOT NULL,
    CAM_FROM_OL_ID      D_ID,
    CAM_OL_DATETIME     D_DATETIME
);

ALTER TABLE CARDAKTMAIN ADD CONSTRAINT PK_CARDAKTMAIN PRIMARY KEY (CAM_ID);
ALTER TABLE CARDAKTMAIN ADD CONSTRAINT FK_CARDAKTMAIN_DST_CPL FOREIGN KEY (CAM_DST_CPL_ID) REFERENCES CARDPLACE (CPL_ID) ON UPDATE CASCADE;
ALTER TABLE CARDAKTMAIN ADD CONSTRAINT FK_CARDAKTMAIN_OL_ID FOREIGN KEY (CAM_FROM_OL_ID) REFERENCES OPERLIST (OL_ID) ON UPDATE CASCADE;
ALTER TABLE CARDAKTMAIN ADD CONSTRAINT FK_CARDAKTMAIN_PIN_CPL_ID FOREIGN KEY (CAM_DST_PIN_CPL_ID) REFERENCES CARDPLACE (CPL_ID) ON UPDATE CASCADE;
ALTER TABLE CARDAKTMAIN ADD CONSTRAINT FK_CARDAKTMAIN_SRC_CPL FOREIGN KEY (CAM_SRC_CPL_ID) REFERENCES CARDPLACE (CPL_ID) ON UPDATE CASCADE;
ALTER TABLE CARDAKTMAIN ADD CONSTRAINT FK_CARDAKTMAIN_UN_ID FOREIGN KEY (CAM_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
CREATE DESCENDING INDEX CARDAKTMAIN_IDX1 ON CARDAKTMAIN (CAM_OL_DATETIME);
CREATE INDEX CARDAKTMAIN_IDX2 ON CARDAKTMAIN (CAM_STATUS);
