Показать DDL (CREATE TABLE) 
CREATE TABLE CARDSYNC_ACTION (
    CSA_ID             D_INTEGER NOT NULL,
    CSA_FROM_CS_ID     D_INTEGER NOT NULL,
    CSA_STATUS         D_SMALLINT DEFAULT 0 NOT NULL,
    CSA_DATETIME       D_DATETIME NOT NULL,
    CSA_EXEC_DATETIME  D_DATETIME,
    CSA_TRY_QTY        D_INTEGER DEFAULT 0 NOT NULL
);

ALTER TABLE CARDSYNC_ACTION ADD CONSTRAINT PK_CARDSYNC_ACTION PRIMARY KEY (CSA_ID);
ALTER TABLE CARDSYNC_ACTION ADD CONSTRAINT FK_CARDSYNC_ACTION_1 FOREIGN KEY (CSA_FROM_CS_ID) REFERENCES CARDSYNC (CS_ID) ON UPDATE CASCADE;
