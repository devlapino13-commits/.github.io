Показать DDL (CREATE TABLE) 
CREATE TABLE SA_CONFIG (
    SAC_ID                 INTEGER NOT NULL,
    SAC_CODE               VARCHAR(20) NOT NULL,
    SAC_NAME               VARCHAR(100) NOT NULL,
    SAC_SHORT_NAME         VARCHAR(100),
    SAC_SIGN               SMALLINT DEFAULT 1,
    SAC_FORMULA            VARCHAR(2048),
    SAC_PRIORITY_ORDER     INTEGER DEFAULT 0,
    SAC_IS_VISIBLE         SMALLINT DEFAULT 1,
    SAC_SHOW_IN_START      SMALLINT DEFAULT 1,
    SAC_BASE_FOR_VOCATION  SMALLINT DEFAULT 0,
    SAC_ROUND              INTEGER DEFAULT -2,
    SAC_KIND               INTEGER DEFAULT 0,
    SAC_DT_BALANCE         VARCHAR(20),
    SAC_DT_NUMBER          INTEGER,
    SAC_KT_BALANCE         VARCHAR(20),
    SAC_KT_NUMBER          INTEGER,
    SAC_DETAIL             VARCHAR(255),
    SAC_SHOW_IN_OPERBOOK   INTEGER DEFAULT 1,
    SAC_IS_MINUS_STORNO    INTEGER DEFAULT 0,
    SAC_IS_CREDIT          INTEGER DEFAULT 0
);

ALTER TABLE SA_CONFIG ADD CONSTRAINT UNQ1_SA_CONFIG UNIQUE (SAC_CODE);
ALTER TABLE SA_CONFIG ADD CONSTRAINT PK_SA_CONFIG PRIMARY KEY (SAC_ID);
ALTER TABLE SA_CONFIG ADD CONSTRAINT FK_SA_CONFIG_1 FOREIGN KEY (SAC_DT_BALANCE) REFERENCES BALGROUP (BG_ID) ON UPDATE CASCADE;
ALTER TABLE SA_CONFIG ADD CONSTRAINT FK_SA_CONFIG_2 FOREIGN KEY (SAC_KT_BALANCE) REFERENCES BALGROUP (BG_ID) ON UPDATE CASCADE;
CREATE INDEX SA_CONFIG_KIND ON SA_CONFIG (SAC_KIND);
CREATE INDEX SA_CONFIG_VOC_BASE ON SA_CONFIG (SAC_BASE_FOR_VOCATION);
