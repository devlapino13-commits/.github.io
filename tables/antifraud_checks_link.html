CREATE TABLE ANTIFRAUD_CHECKS_LINK (
    AFCL_ID            D_ID GENERATED BY DEFAULT AS IDENTITY,
    AFCL_FROM_C_ID     D_ID NOT NULL,
    AFCL_OBJECT_KIND   D_INTEGER,
    AFCL_OBJECT_ID     D_ID,
    AFCL_CHECK_TYPE    D_SMALLINT NOT NULL,
    AFCL_CHECK_STAGE   D_SMALLINT NOT NULL,
    AFCL_FROM_AC_ID    D_ID,
    AFCL_CHECK_RESULT  D_SMALLINT NOT NULL,
    AFCL_DATETIME      D_DATETIME DEFAULT current_timestamp /* D_DATETIME = TIMESTAMP */ /* D_DATETIME = TIMESTAMP */ /* D_DATETIME = TIMESTAMP */ /* D_DATETIME = TIMESTAMP */
);

ALTER TABLE ANTIFRAUD_CHECKS_LINK ADD CONSTRAINT PK_ANTIFRAUD_CHECKS_LINK PRIMARY KEY (AFCL_ID);
ALTER TABLE ANTIFRAUD_CHECKS_LINK ADD CONSTRAINT FK_ANTIFRAUD_CHECKS_LINK_1 FOREIGN KEY (AFCL_FROM_AC_ID) REFERENCES ANTIFRAUD_CHECK (AC_ID) ON UPDATE CASCADE;
