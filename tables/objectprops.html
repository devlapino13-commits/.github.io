CREATE TABLE OBJECTPROPS (
    OPR_ID             ID_TYPE NOT NULL,
    OPR_OBJECT_ID      ID_TYPE NOT NULL,

CREATE INDEX OBJECTPROPS_SUBJECT ON OBJECTPROPS (OPR_SUBJECT_KIND, OPR_SUBJECT_ID);
