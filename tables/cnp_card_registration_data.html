CREATE TABLE CNP_CARD_REGISTRATION_DATA (
    CCRD_ID             D_ID GENERATED BY DEFAULT AS IDENTITY,
    CCRD_FROM_C_ID      D_ID NOT NULL,
    CCRD_CREATE_DATE    D_DATE DEFAULT current_date NOT NULL,
