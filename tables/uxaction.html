Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE UXACTION (
    UA_UX_CODE          D_STR_CODE NOT NULL,
    UA_CODE             D_STR_100 NOT NULL,
    UA_UXA_CODE         D_STR_100 NOT NULL,
    UA_NAME             D_STR_100 NOT NULL,
    UA_HINT             D_STR_255,
    UA_DISABLE_ON_EDIT  D_BOOL DEFAULT 0 NOT NULL,
    UA_READONLY         D_BOOL DEFAULT 0 NOT NULL,
    UA_SAFE_MODE        D_BOOL DEFAULT 0 NOT NULL,
    UA_INACTIVE         D_BOOL DEFAULT 0 NOT NULL,
    UA_MENU_INDEX       D_STR_20,
    UA_USR_ID_FIELD     D_STR_CODE,
    UA_USR_TEXT_FIELD   D_STR_CODE
);

ALTER TABLE UXACTION ADD CONSTRAINT PK_UXACTION PRIMARY KEY (UA_CODE);
ALTER TABLE UXACTION ADD CONSTRAINT FK_UXACTION_UXA FOREIGN KEY (UA_UXA_CODE) REFERENCES UXALIAS (UXA_CODE) ON UPDATE CASCADE;
ALTER TABLE UXACTION ADD CONSTRAINT FK_UXACTION_UX_CODE FOREIGN KEY (UA_UX_CODE) REFERENCES UXCODE (UX_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE INDEX UXACTION_MENU_INDEX ON UXACTION (UA_MENU_INDEX);
