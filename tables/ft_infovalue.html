Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE FT_INFOVALUE (
    FIV_ID           D_ID NOT NULL,
    FIV_FROM_C_ID    D_ID NOT NULL,
    FIV_FROM_FIF_ID  D_ID NOT NULL,
    FIV_DATE_BEGIN   D_DATE NOT NULL,
    FIV_DATE_END     D_DATE NOT NULL,
    FIV_VALUE        D_STR_LONG
);

ALTER TABLE FT_INFOVALUE ADD CONSTRAINT PK_FT_INFOVALUE PRIMARY KEY (FIV_ID);
ALTER TABLE FT_INFOVALUE ADD CONSTRAINT FK_FT_INFOVALUE_C_ID FOREIGN KEY (FIV_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE
  USING DESCENDING INDEX FK_FT_INFOVALUE_C_ID;
ALTER TABLE FT_INFOVALUE ADD CONSTRAINT FK_FT_INFOVALUE_FIF FOREIGN KEY (FIV_FROM_FIF_ID) REFERENCES FT_INFOFIELD (FIF_ID) ON UPDATE CASCADE
  USING DESCENDING INDEX FK_FT_INFOVALUE_FIF;
CREATE DESCENDING INDEX FT_INFOVALUE_DATE_BEGIN ON FT_INFOVALUE (FIV_DATE_BEGIN);
CREATE DESCENDING INDEX FT_INFOVALUE_DATE_END ON FT_INFOVALUE (FIV_DATE_END);
