Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE IPC_TERMINAL_OWNERSHIP (
    ITO_ID           D_INTEGER NOT NULL,
    ITO_TRM_ID       D_INTEGER NOT NULL,
    ITO_OBJECT_KIND  D_INTEGER NOT NULL,
    ITO_OBJECT_ID    D_ID NOT NULL,
    ITO_DATE_BEGIN   D_DATE DEFAULT current_date NOT NULL,
    ITO_DATE_END     D_DATE DEFAULT '31.12.9999' NOT NULL,
    ITO_DATETIME     D_DATETIME DEFAULT current_timestamp NOT NULL,
    ITO_FROM_UN_ID   D_ID NOT NULL,
    ITO_STATUS       D_SMALLINT DEFAULT 0 NOT NULL
);

ALTER TABLE IPC_TERMINAL_OWNERSHIP ADD CONSTRAINT CHK1_IPC_TERMINAL_OWNERSHIP CHECK (ito_date_end >= ito_date_begin);
ALTER TABLE IPC_TERMINAL_OWNERSHIP ADD CONSTRAINT CHK2_IPC_TERMINAL_OWNERSHIP CHECK (ito_status between -1 and 3);
ALTER TABLE IPC_TERMINAL_OWNERSHIP ADD CONSTRAINT PK_IPC_TERMINAL_OWNERSHIP PRIMARY KEY (ITO_ID);
ALTER TABLE IPC_TERMINAL_OWNERSHIP ADD CONSTRAINT FK_IPC_TERMINAL_OWNER_TRM_ID FOREIGN KEY (ITO_TRM_ID) REFERENCES IPC_TERMINALS (TRM_ID);
ALTER TABLE IPC_TERMINAL_OWNERSHIP ADD CONSTRAINT FK_IPC_TERMINAL_OWNER_UN_ID FOREIGN KEY (ITO_FROM_UN_ID) REFERENCES USERNAME (UN_ID);
CREATE INDEX IDX_IPC_TERMINAL_OWNER_DATE ON IPC_TERMINAL_OWNERSHIP (ITO_TRM_ID, ITO_DATE_BEGIN, ITO_DATE_END);
