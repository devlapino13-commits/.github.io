Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE IPC_GC_COMMISSION (
    IGC_ID                  D_ID NOT NULL,
    IGC_FROM_IGA_ID         D_INTEGER NOT NULL,
    IGC_FROM_C_ID           D_ID NOT NULL,
    IGC_FROM_AG_ID          D_INTEGER NOT NULL,
    IGC_COMMIS_CARDS_MB     D_PERCENT DEFAULT 0 NOT NULL,
    IGC_COMMIS_CARDS_OTHER  D_PERCENT DEFAULT 0 NOT NULL,
    IGC_AMOUNT              D_CURRENCY DEFAULT 0 NOT NULL,
    IGC_AMOUNT_CARDS_MB     D_CURRENCY DEFAULT 0 NOT NULL,
    IGC_AMOUNT_CARDS_OTHER  D_CURRENCY DEFAULT 0 NOT NULL,
    IGC_FROM_OL_ID          D_ID
);

ALTER TABLE IPC_GC_COMMISSION ADD CONSTRAINT PK_IPC_GC_COMMISSION PRIMARY KEY (IGC_ID);
ALTER TABLE IPC_GC_COMMISSION ADD CONSTRAINT FK_IPC_GC_COMMISSION_AG_ID FOREIGN KEY (IGC_FROM_AG_ID) REFERENCES AGREEMENTS (AG_ID) ON UPDATE CASCADE;
ALTER TABLE IPC_GC_COMMISSION ADD CONSTRAINT FK_IPC_GC_COMMISSION_C_ID FOREIGN KEY (IGC_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE IPC_GC_COMMISSION ADD CONSTRAINT FK_IPC_GC_COMMISSION_IGA_ID FOREIGN KEY (IGC_FROM_IGA_ID) REFERENCES IPC_GC_ANALIZ (IGA_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE IPC_GC_COMMISSION ADD CONSTRAINT FK_IPC_GC_COMMISSION_OL_ID FOREIGN KEY (IGC_FROM_OL_ID) REFERENCES OPERLIST (OL_ID) ON UPDATE CASCADE;
