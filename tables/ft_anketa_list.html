Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE FT_ANKETA_LIST (
    FAL_ID               D_ID NOT NULL,
    FAL_FROM_C_ID        D_ID NOT NULL,
    FAL_DATE             D_DATE NOT NULL,
    FAL_STATE            D_SMALLINT NOT NULL,
    FAL_NEED_CHECK       D_BOOL DEFAULT 0 NOT NULL,
    FAL_CREATE_UN_ID     D_ID NOT NULL,
    FAL_CREATE_DATETIME  D_DATETIME NOT NULL,
    FAL_FROM_FAK_ID      D_INTEGER
);

ALTER TABLE FT_ANKETA_LIST ADD CONSTRAINT PK_FT_ANKETA_LIST PRIMARY KEY (FAL_ID);
ALTER TABLE FT_ANKETA_LIST ADD CONSTRAINT FK_FT_ANKETA_LIST_C_ID FOREIGN KEY (FAL_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE
  USING DESCENDING INDEX FK_FT_ANKETA_LIST_C_ID;
ALTER TABLE FT_ANKETA_LIST ADD CONSTRAINT FK_FT_ANKETA_LIST_UN_ID FOREIGN KEY (FAL_CREATE_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
CREATE DESCENDING INDEX FT_ANKETA_LIST_DATE ON FT_ANKETA_LIST (FAL_DATE);
