Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE REFDATA (
    RD_ID         INTEGER NOT NULL,
    RD_PARENT_ID  INTEGER,
    RD_CODE       VARCHAR(255) NOT NULL,
    RD_MAIN_CODE  VARCHAR(255) NOT NULL,
    RD_NAME       VARCHAR(255),
    RD_MEMO       VARCHAR(255),
    RD_ORDERNO    INTEGER,
    RD_IS_ACTIVE  INTEGER DEFAULT 1,
    RD_IS_LOCKED  INTEGER DEFAULT 0,
    RD_STR_ID     VARCHAR(100),
    RD_TAG        INTEGER DEFAULT 0
);

ALTER TABLE REFDATA ADD CONSTRAINT PK_REFDATA PRIMARY KEY (RD_ID)
USING DESCENDING INDEX PK_REFDATA;
ALTER TABLE REFDATA ADD CONSTRAINT FK_REFDATA_1 FOREIGN KEY (RD_PARENT_ID) REFERENCES REFDATA (RD_ID) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE UNIQUE INDEX REFDATA_CODE ON REFDATA (RD_CODE);
CREATE INDEX REFDATA_ID_STR ON REFDATA (RD_MAIN_CODE, RD_STR_ID);
CREATE INDEX REFDATA_MAIN_CODE ON REFDATA (RD_MAIN_CODE);
CREATE INDEX REFDATA_NAME ON REFDATA (RD_MAIN_CODE, RD_NAME);
CREATE INDEX REFDATA_RD_MEMO ON REFDATA (RD_MEMO);
