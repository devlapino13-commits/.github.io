Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE APPBLOBS (
    AB_LOCAL_PATH   D_STR_255 NOT NULL,
    AB_NAME         D_STR_CODE NOT NULL,
    AB_PLATFORM     D_INTEGER NOT NULL,
    AB_FILE_MODIFY  D_DATETIME NOT NULL,
    AB_VER_MAJOR    D_INTEGER NOT NULL,
    AB_VER_MINOR    D_INTEGER NOT NULL,
    AB_VER_RELEASE  D_INTEGER DEFAULT 0 NOT NULL,
    AB_VER_BUILD    D_INTEGER DEFAULT 0 NOT NULL,
    AB_VER_STR      COMPUTED BY ((AB_VER_MAJOR || '.' || AB_VER_MINOR || '.' || AB_VER_RELEASE || '.' || AB_VER_BUILD)),
    AB_BLOB         D_BLOB_BINARY,
    AB_FROM_UN_ID   D_ID NOT NULL,
    AB_DATETIME     D_DATETIME NOT NULL
);

ALTER TABLE APPBLOBS ADD CONSTRAINT APPBLOBS_C_PLATFORM CHECK (AB_PLATFORM in (0, 1, 2, 3));
ALTER TABLE APPBLOBS ADD CONSTRAINT APPBLOBS_P_UNIQUE_NAME PRIMARY KEY (AB_PLATFORM, AB_LOCAL_PATH, AB_NAME)
USING INDEX APPBLOBS_I_UNIQUE_NAME;
CREATE INDEX APPBLOBS_I_FILE_MODIFY ON APPBLOBS (AB_FILE_MODIFY);
CREATE INDEX APPBLOBS_I_PLATFORM ON APPBLOBS (AB_PLATFORM);
