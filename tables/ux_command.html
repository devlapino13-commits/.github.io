Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE UX$COMMAND (
    CMD_UUID            D_UUID NOT NULL,
    CMD_OPEN_KIND       D_SMALLINT DEFAULT 0 NOT NULL,
    CMD_SOURCE_ROWS     D_SMALLINT DEFAULT 1 NOT NULL,
    CMD_SOURCE_REFRESH  D_SMALLINT DEFAULT 1 NOT NULL,
    CMD_MESSAGE_FIELD   D_STR_LONG NOT NULL,
    CMD_COMMIT_DIALOG   D_BOOL DEFAULT 1 NOT NULL
);

ALTER TABLE UX$COMMAND ADD CONSTRAINT PK_UX$COMMAND PRIMARY KEY (CMD_UUID);
ALTER TABLE UX$COMMAND ADD CONSTRAINT FK_UX$COMMAND_UFM FOREIGN KEY (CMD_UUID) REFERENCES UX$FORM (UFM_UUID) ON DELETE CASCADE ON UPDATE CASCADE
  USING INDEX FK_UX$COMMAND_CODE;
