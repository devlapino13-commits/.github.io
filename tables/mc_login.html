Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE MC_LOGIN (
    MCL_ID                   ID_TYPE NOT NULL,
    MCL_LOGIN                D_STR_CODE,
    MCL_FROM_C_ID            ID_TYPE,
    MCL_IS_DELETE            D_SMALLINT,
    MCL_DATETIME             D_DATETIME,
    MCL_PASS_HASH            D_STR_LONG,
    MCL_GOOGLE_ID            D_STR_255,
    MCL_LANG_CODE            D_STR_LANG,
    MCL_MESSAGE_LAST_READ    D_DATETIME,
    MCL_OS                   D_SMALLINT,
    MCL_APP_VERSION          D_STR_10,
    MCL_LAST_LOGIN_DATETIME  D_DATETIME,
    MCL_DEV_ID               D_UUID,
    MCL_BIRTH_DATE           D_DATE,
    MCL_USER_CODE            D_INTEGER,
    MCL_PREV_LOGIN_DATETIME  D_DATETIME,
    MCL_LAST_UPDATE_NOTIFY   D_DATETIME,
    MCL_PROJECT              D_INTEGER default 1 NOT NULL,
    MCL_FROM_REG_ID          D_INTEGER
);

ALTER TABLE MC_LOGIN ADD CONSTRAINT PK_MC_LOGIN PRIMARY KEY (MCL_ID);
ALTER TABLE MC_LOGIN ADD CONSTRAINT FK_MCL_REG_ID FOREIGN KEY (MCL_FROM_REG_ID) REFERENCES REGION (REG_UNICODE) ON UPDATE CASCADE;
ALTER TABLE MC_LOGIN ADD CONSTRAINT FK_MC_LOGIN_C_ID FOREIGN KEY (MCL_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
CREATE INDEX IDX_MC_LOGIN_MAIN ON MC_LOGIN (MCL_LOGIN, MCL_IS_DELETE);
CREATE INDEX MC_LOGIN_GOOGLE_ID ON MC_LOGIN (MCL_GOOGLE_ID);
