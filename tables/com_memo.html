Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE COM_MEMO (
    CMM_ID           D_INTEGER NOT NULL,
    CMM_FROM_CMA_ID  D_ID NOT NULL,
    CMM_MEMO_KIND    D_SMALLINT NOT NULL,
    CMM_MEMO         D_STR_LONG,
    CMM_IS_DELETE    D_SMALLINT DEFAULT 0 NOT NULL,
    CMM_FROM_UN_ID   D_ID NOT NULL,
    CMM_DATETIME     D_DATETIME
);

ALTER TABLE COM_MEMO ADD CONSTRAINT PK_COM_MEMO PRIMARY KEY (CMM_ID)
USING DESCENDING INDEX PK_COM_MEMO;
ALTER TABLE COM_MEMO ADD CONSTRAINT FK_COM_MEMO_CMA FOREIGN KEY (CMM_FROM_CMA_ID) REFERENCES COM_ACTION (CMA_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE COM_MEMO ADD CONSTRAINT FK_COM_MEMO_UN FOREIGN KEY (CMM_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
