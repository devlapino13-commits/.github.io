Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE IPC_FILE_ANALIZ (
    IFA_ID                 D_INTEGER NOT NULL,
    IFA_FILE_NAME          D_STR_100 NOT NULL,
    IFA_FILE_TYPE          D_SMALLINT NOT NULL,
    IFA_DATE               D_DATE NOT NULL,
    IFA_FROM_BLS_ID        D_ID NOT NULL,
    IFA_IS_DETAIL_PROBLEM  D_BOOL DEFAULT 0 NOT NULL,
    IFA_STATUS             D_SMALLINT DEFAULT 0 NOT NULL,
    IFA_FROM_UN_ID         D_ID NOT NULL,
    IFA_DATETIME           D_DATETIME DEFAULT current_timestamp NOT NULL,
    IFA_FROM_OL_ID         D_ID
);

ALTER TABLE IPC_FILE_ANALIZ ADD CONSTRAINT CHK2_IPC_FILE_ANALIZ CHECK (ifa_file_type in (1,2,3));
ALTER TABLE IPC_FILE_ANALIZ ADD CONSTRAINT CHK1_IPC_FILE_ANALIZ CHECK (ifa_status between -1 and 5);
ALTER TABLE IPC_FILE_ANALIZ ADD CONSTRAINT PK_IPC_FILE_ANALIZ PRIMARY KEY (IFA_ID);
ALTER TABLE IPC_FILE_ANALIZ ADD CONSTRAINT FK_IPC_FILE_ANALIZ_UN_ID FOREIGN KEY (IFA_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
CREATE UNIQUE INDEX IPC_FILE_ANALIZ_FILENAME ON IPC_FILE_ANALIZ (IFA_FILE_NAME);
CREATE INDEX IPC_FILE_ANALIZ_IDX_DATE ON IPC_FILE_ANALIZ (IFA_DATE);
