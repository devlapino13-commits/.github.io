Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE CARDEMBOMAIN (
    CEM_ID            D_INTEGER NOT NULL,
    CEM_DATE          D_DATE DEFAULT current_date NOT NULL,
    CEM_IS_PINCOVER   D_BOOL DEFAULT 0 NOT NULL,
    CEM_APP_COUNT     D_INTEGER NOT NULL,
    CEM_READY_COUNT   D_INTEGER DEFAULT 0 NOT NULL,
    CEM_FROM_CKD_ID   D_INTEGER NOT NULL,
    CEM_FROM_CES_ID   D_INTEGER NOT NULL,
    CEM_FILENAME      D_STR_100,
    CEM_FROM_CAMB_ID  D_INTEGER,
    CEM_FROM_CAM_ID   D_INTEGER,
    CEM_FROM_OL_ID    D_ID,
    CEM_FROM_UN_ID    D_ID NOT NULL
);

ALTER TABLE CARDEMBOMAIN ADD CONSTRAINT CHK1_CARDEMBOMAIN CHECK (cem_app_count > 0);
ALTER TABLE CARDEMBOMAIN ADD CONSTRAINT PK_CARDEMBOMAIN PRIMARY KEY (CEM_ID);
ALTER TABLE CARDEMBOMAIN ADD CONSTRAINT FK_CARDEMBOMAIN_CAMB_ID FOREIGN KEY (CEM_FROM_CAMB_ID) REFERENCES CARDAKTMAIN (CAM_ID) ON UPDATE CASCADE;
ALTER TABLE CARDEMBOMAIN ADD CONSTRAINT FK_CARDEMBOMAIN_CAM_ID FOREIGN KEY (CEM_FROM_CAM_ID) REFERENCES CARDAKTMAIN (CAM_ID) ON UPDATE CASCADE;
ALTER TABLE CARDEMBOMAIN ADD CONSTRAINT FK_CARDEMBOMAIN_CES_ID FOREIGN KEY (CEM_FROM_CES_ID) REFERENCES CARDEMBOSTATUS (CES_ID) ON UPDATE CASCADE;
ALTER TABLE CARDEMBOMAIN ADD CONSTRAINT FK_CARDEMBOMAIN_CKD_ID FOREIGN KEY (CEM_FROM_CKD_ID) REFERENCES CARDKIND (CKD_ID) ON UPDATE CASCADE;
ALTER TABLE CARDEMBOMAIN ADD CONSTRAINT FK_CARDEMBOMAIN_OL_ID FOREIGN KEY (CEM_FROM_OL_ID) REFERENCES OPERLIST (OL_ID) ON UPDATE CASCADE;
ALTER TABLE CARDEMBOMAIN ADD CONSTRAINT FK_CARDEMBOMAIN_UN_ID FOREIGN KEY (CEM_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
