CREATE TABLE REPFIELDS (
    RF_FROM_RL_ID  INTEGER NOT NULL,
    RF_NUMBER      INTEGER NOT NULL,
    RF_FIELD_NAME  VARCHAR(50) NOT NULL,
    RF_NAME_RUS    VARCHAR(50),
    RF_HINT_RUS    VARCHAR(256),
    RF_NAME_ENG    VARCHAR(50),
    RF_HINT_ENG    VARCHAR(256),
    RF_TYPE        SMALLINT DEFAULT 0 NOT NULL,
    RF_WIDTH       SMALLINT DEFAULT 64,
    RF_CAN_GROUP   SMALLINT DEFAULT 0,
    RF_AGREGATE    SMALLINT DEFAULT 0,
    RF_LOOKUP_SQL  VARCHAR(256)
);
