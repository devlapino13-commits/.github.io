CREATE GLOBAL TEMPORARY TABLE UXIMPORT (
    UIM_ID           D_INTEGER NOT NULL,
    UIM_PARENT_ID    D_INTEGER,
    UIM_CODE         D_STR_100 NOT NULL,
    UIM_NAME         D_STR_100 NOT NULL,
    UIM_KIND         D_STR_CODE NOT NULL,
    UIM_TABLE        D_STR_CODE,
    UIM_NEW_ITEM     D_BOOL DEFAULT 0 NOT NULL,
    UIM_NEW_CHILD    D_BOOL DEFAULT 0 NOT NULL,
    UIM_CAN_SELECT   D_BOOL DEFAULT 0 NOT NULL,
    UIM_SELECT_KIND  D_SMALLINT DEFAULT 0 NOT NULL
) ON COMMIT PRESERVE ROWS;
