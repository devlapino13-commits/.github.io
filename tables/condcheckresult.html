Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE CONDCHECKRESULT (
    CCR_ID           D_INTEGER NOT NULL,
    CCR_FROM_CCS_ID  D_INTEGER NOT NULL,
    CCR_FROM_CDM_ID  D_INTEGER,
    CCR_INFO         D_STR_255,
    CCR_MESSAGE      D_STR_LONG,
    CCR_IS_CRITICAL  D_SMALLINT NOT NULL,
    CCR_FROM_UN_ID   D_ID NOT NULL,
    CCR_DATETIME     D_DATETIME
);

ALTER TABLE CONDCHECKRESULT ADD CONSTRAINT PK_CONDCHECKRESULT PRIMARY KEY (CCR_ID);
ALTER TABLE CONDCHECKRESULT ADD CONSTRAINT FK_CONDCHECKRESULT_CCS FOREIGN KEY (CCR_FROM_CCS_ID) REFERENCES CONDCHECKSTATE (CCS_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE CONDCHECKRESULT ADD CONSTRAINT FK_CONDCHECKRESULT_CDM FOREIGN KEY (CCR_FROM_CDM_ID) REFERENCES CONDITIONMAIN (CDM_ID) ON UPDATE CASCADE;
ALTER TABLE CONDCHECKRESULT ADD CONSTRAINT FK_CONDCHECKRESULT_UN FOREIGN KEY (CCR_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
