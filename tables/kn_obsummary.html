    KOBS_BAL_DT_VAL  COMPUTED BY (iif(kobs_acc_type = 0 or (kobs_acc_type = 2 and kobs_bal_val >= 0), kobs_bal_val, 0)),
    KOBS_BAL_DT_TAX  COMPUTED BY (iif(kobs_acc_type = 0 or (kobs_acc_type = 2 and kobs_bal_tax >= 0), kobs_bal_tax, 0)),
    KOBS_BAL_KT_SUM  COMPUTED BY (iif(kobs_acc_type = 1 or (kobs_acc_type = 2 and kobs_bal_sum < 0), -kobs_bal_sum, 0)),
    KOBS_BAL_KT_VAL  COMPUTED BY (iif(kobs_acc_type = 1 or (kobs_acc_type = 2 and kobs_bal_val < 0), -kobs_bal_val, 0)),
    KOBS_BAL_KT_TAX  COMPUTED BY (iif(kobs_acc_type = 1 or (kobs_acc_type = 2 and kobs_bal_tax < 0), -kobs_bal_tax, 0))
);
