Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE IPC_TERMINALS (
    TRM_ID          D_INTEGER NOT NULL,
    TRM_TYPE        D_SMALLINT NOT NULL,
    TRM_ACTUAL      D_BOOL DEFAULT 1 NOT NULL,
    TRM_DATE_ANNUL  D_DATETIME,
    TRM_IPC_ID      D_STR_20 NOT NULL,
    TRM_FROM_ML_ID  D_ID NOT NULL,
    TRM_INVENT_STR  D_STR_CODE NOT NULL,
    TRM_SERIAL_NO   D_STR_255,
    TRM_SIMCARD_NO  D_STR_20,
    TRM_FROM_UN_ID  D_ID
);

ALTER TABLE IPC_TERMINALS ADD CONSTRAINT CHK1_IPC_TERMINALS CHECK (trm_type in (1,2));
ALTER TABLE IPC_TERMINALS ADD CONSTRAINT PK_IPC_TERMINALS PRIMARY KEY (TRM_ID);
ALTER TABLE IPC_TERMINALS ADD CONSTRAINT FK_IPC_TERMINALS_ML_ID FOREIGN KEY (TRM_FROM_ML_ID) REFERENCES MATLIST (ML_ID) ON UPDATE CASCADE;
ALTER TABLE IPC_TERMINALS ADD CONSTRAINT FK_IPC_TERMINALS_UN_ID FOREIGN KEY (TRM_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
CREATE UNIQUE INDEX IPC_TERMINALS_IDX1 ON IPC_TERMINALS (TRM_IPC_ID);
CREATE INDEX IPC_TERMINALS_INVENT_STR ON IPC_TERMINALS (TRM_INVENT_STR);
