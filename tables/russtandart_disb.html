CREATE TABLE RUSSTANDART_DISB (
    RSD_ID          INTEGER NOT NULL,
    RSD_FROM_KC_ID  ID_TYPE,
    RSD_FROM_C_ID   ID_TYPE NOT NULL,
    RSD_ACCOUNT     STR_100 NOT NULL,
    RSD_CARD_ID     STR_100 NOT NULL,
    RSD_STATUS      SMALLINT DEFAULT 0 NOT NULL,
    RSD_QUERY       BLOB SUB_TYPE TEXT SEGMENT SIZE 80,

);

ALTER TABLE RUSSTANDART_DISB ADD CONSTRAINT PK_RUSSTANDART_DISB PRIMARY KEY (RSD_ID);
ALTER TABLE RUSSTANDART_DISB ADD CONSTRAINT FK_RUSSTANDART_DISB_1 FOREIGN KEY (RSD_FROM_KC_ID) REFERENCES KNCLIENT (KC_ID) ON UPDATE CASCADE;
ALTER TABLE RUSSTANDART_DISB ADD CONSTRAINT FK_RUSSTANDART_DISB_2 FOREIGN KEY (RSD_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
