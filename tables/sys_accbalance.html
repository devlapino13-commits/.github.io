SYS_ACCBALANCE

Системный счета, системный план счетов. / В этой таблице также указывается правило формирования лицевого счета. / Допустимые символы при формировании лицевого счета: / B - Балансовый счет (обычно 5 символов: BBBBB) / V - Валюта (обычно 3 символа: VVV) / R - Код офиса (обычно 4 символа: RRRR) / X - Уникальный код объекта (займ, статьи расхода и т.д., обычно до 8 символов) / N - Порядковый номер лицевого счета, обычно до 8 символов. Используется когда нет прикрепленного объекта. / Кроме этого можно использовать цифры или точки как константы, они не будут заменены на другие значения.

Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE SYS_ACCBALANCE (
    SAB_ID            D_ID NOT NULL,
    SAB_PARENT_ID     D_ID,
    SAB_CODE          D_STR_CODE NOT NULL,
    SAB_NAME          D_STR_255,
    SAB_NAME_MASK     D_STR_255,
    SAB_ACC_MASK      D_STR_CODE,
    SAB_SHORT_MASK    D_STR_CODE,
    SAB_OBJECT_KIND   D_INTEGER,
    SAB_FULL_CODE     D_STR_255 NOT NULL,
    SAB_INVERT_CODE   D_STR_255 NOT NULL,
    SAB_IS_FIXED      SMALLINT,
    SAB_FIXED_AB_ID   D_ID,
    SAB_FIXED_INX_ID  D_ID,
    SAB_CREATE_KIND   D_INTEGER,
    SAB_EXPORT_1C     D_SMALLINT DEFAULT 0,
    SAB_TAX_KIND      D_SMALLINT
);

ALTER TABLE SYS_ACCBALANCE ADD CONSTRAINT PK_SYS_ACCBALANCE PRIMARY KEY (SAB_ID);
ALTER TABLE SYS_ACCBALANCE ADD CONSTRAINT FK_SYS_ACCBALANCE_AB FOREIGN KEY (SAB_FIXED_AB_ID) REFERENCES REF_ACCBALANCE (AB_ID) ON DELETE SET NULL ON UPDATE CASCADE;
ALTER TABLE SYS_ACCBALANCE ADD CONSTRAINT FK_SYS_ACCBALANCE_INX FOREIGN KEY (SAB_FIXED_INX_ID) REFERENCES REF_INCOME_EXPENSE (INX_ID) ON DELETE SET NULL ON UPDATE CASCADE;
ALTER TABLE SYS_ACCBALANCE ADD CONSTRAINT FK_SYS_ACCBALANCE_PARENT FOREIGN KEY (SAB_PARENT_ID) REFERENCES SYS_ACCBALANCE (SAB_ID) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE UNIQUE INDEX SYS_ACCBALANCE_FULL_CODE ON SYS_ACCBALANCE (SAB_FULL_CODE);
CREATE UNIQUE INDEX SYS_ACCBALANCE_INVERT_CODE ON SYS_ACCBALANCE (SAB_INVERT_CODE);
