CREATE TABLE KN_BALSUMMARY (
    KBS_ID           D_ID NOT NULL,
    KBS_AB_ID        ID_TYPE NOT NULL,
    KBS_DATE         DATE NOT NULL,
    KBS_DATE_END     DATE NOT NULL,
    KBS_ACC_TYPE     D_SMALLINT DEFAULT 0 NOT NULL,
    KBS_IS_EXTENDED  D_SMALLINT,
    KBS_DT_SUM       CURR_TYPE,
    KBS_DT_VAL       CURR_TYPE,
    KBS_DT_TAX       CURR_TYPE,
    KBS_KT_SUM       CURR_TYPE,
    KBS_KT_VAL       CURR_TYPE,
    KBS_KT_TAX       CURR_TYPE,
    KBS_BAL_SUM      CURR_TYPE,
    KBS_BAL_VAL      CURR_TYPE,
    KBS_BAL_TAX      CURR_TYPE,
    KBS_BAL_DT_SUM   COMPUTED BY (iif(kbs_acc_type = 0 or (kbs_acc_type = 2 and kbs_bal_sum >= 0), kbs_bal_sum, 0)),
    KBS_BAL_DT_VAL   COMPUTED BY (iif(kbs_acc_type = 0 or (kbs_acc_type = 2 and kbs_bal_val >= 0), kbs_bal_val, 0)),
    KBS_BAL_DT_TAX   COMPUTED BY (iif(kbs_acc_type = 0 or (kbs_acc_type = 2 and kbs_bal_tax >= 0), kbs_bal_tax, 0)),
    KBS_BAL_KT_SUM   COMPUTED BY (iif(kbs_acc_type = 1 or (kbs_acc_type = 2 and kbs_bal_sum < 0), -kbs_bal_sum, 0)),
    KBS_BAL_KT_VAL   COMPUTED BY (iif(kbs_acc_type = 1 or (kbs_acc_type = 2 and kbs_bal_val < 0), -kbs_bal_val, 0)),
    KBS_BAL_KT_TAX   COMPUTED BY (iif(kbs_acc_type = 1 or (kbs_acc_type = 2 and kbs_bal_tax < 0), -kbs_bal_tax, 0))
);
