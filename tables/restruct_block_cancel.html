Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE RESTRUCT_BLOCK_CANCEL (
    RBC_ID                         D_ID GENERATED BY DEFAULT AS IDENTITY,
    RBC_FROM_K_ID                  D_ID NOT NULL,
    RBC_FROM_OEN_ID                D_ID NOT NULL,
    RBC_DATE_BEGIN                 D_DATE NOT NULL,
    RBC_DATE_END                   D_DATE NOT NULL,
    RBC_MOVED_LAW_TO_ORIV          D_BOOL DEFAULT 0,
    RBC_MOVE_LAW_TO_ORIV_DATETIME  D_DATETIME,
    RBC_MOVE_LAW_TO_ORIV_ERROR     D_STR_255,
    RBC_MOVED_ORIV_TO_LAW          D_BOOL DEFAULT 0,
    RBC_MOVE_ORIV_TO_LAW_DATETIME  D_DATETIME,
    RBC_MOVE_ORIV_TO_LAW_ERROR     D_STR_255
);

ALTER TABLE RESTRUCT_BLOCK_CANCEL ADD CONSTRAINT PK_RESTRUCT_BLOCK_CANCEL PRIMARY KEY (RBC_ID);
ALTER TABLE RESTRUCT_BLOCK_CANCEL ADD CONSTRAINT FK_RESTRUCT_BLOCK_CANCEL_1 FOREIGN KEY (RBC_FROM_OEN_ID) REFERENCES OPEREXTRA_EXECUTION_NOTE (OEN_ID) ON UPDATE CASCADE;
ALTER TABLE RESTRUCT_BLOCK_CANCEL ADD CONSTRAINT FK_RESTRUCT_BLOCK_CANCEL_2 FOREIGN KEY (RBC_FROM_K_ID) REFERENCES KONTRACT (K_ID) ON UPDATE CASCADE;
CREATE INDEX RESTRUCT_BLOCK_CANCEL_IDX1 ON RESTRUCT_BLOCK_CANCEL (RBC_DATE_BEGIN, RBC_DATE_END);
CREATE INDEX RESTRUCT_BLOCK_CANCEL_IDX2 ON RESTRUCT_BLOCK_CANCEL (RBC_DATE_END);
