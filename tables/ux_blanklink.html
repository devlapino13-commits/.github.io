CREATE TABLE UX$BLANKLINK (
    UBL_ID           D_INTEGER NOT NULL,
    UBL_FROM_UBC_ID  D_INTEGER NOT NULL,
    UBL_BLANK_ID     D_INTEGER NOT NULL,
    UBL_REQUIRED     D_BOOL DEFAULT 0 NOT NULL,

ALTER TABLE UX$BLANKLINK ADD CONSTRAINT PK_UX$BLANKLINK PRIMARY KEY (UBL_ID);
ALTER TABLE UX$BLANKLINK ADD CONSTRAINT FK_UX$BLANKLINK_BLN FOREIGN KEY (UBL_BLANK_ID) REFERENCES BLANKNAME (BLN_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE UX$BLANKLINK ADD CONSTRAINT FK_UX$BLANKLINK_UBC FOREIGN KEY (UBL_FROM_UBC_ID) REFERENCES UX$BLANKCODE (UBC_ID) ON DELETE CASCADE ON UPDATE CASCADE;
