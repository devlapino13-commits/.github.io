CREATE TABLE AZ_WEBSITE (
    AW_ID              D_ID NOT NULL,
    AW_DATE            D_DATE,
    AW_DATETIME        D_DATETIME,
    AW_CL_FIO          D_STR_100,
    AW_PHONE           D_STR_20,
    AW_BIRTH_DATE      D_DATE,
    AW_LANGUAGE        D_STR_10,
    AW_STATUS          D_SMALLINT,
    AW_FROM_REG_ID     D_INTEGER,
    AW_FROM_UN_ID      ID_TYPE,
    AW_BLOCK_DATETIME  D_DATETIME
);

ALTER TABLE AZ_WEBSITE ADD CONSTRAINT FK_AZ_WEBSITE_2 FOREIGN KEY (AW_FROM_UN_ID) REFERENCES USERNAME (UN_ID);
