Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE SMEVPASS_BLOCKCLIENTS (
    SBC_ID                D_ID NOT NULL,
    SBC_FROM_C_ID         D_ID NOT NULL,
    SBC_BLOCK_DATETIME    D_DATETIME NOT NULL,
    SBC_UNBLOCK_DATETIME  D_DATETIME,
    SBC_UNBLOCK_UN_ID     D_ID,
    SBC_FROM_MA_ID        D_INTEGER
);

ALTER TABLE SMEVPASS_BLOCKCLIENTS ADD CONSTRAINT PK_SMEVPASS_BLOCKCLIENTS PRIMARY KEY (SBC_ID);
ALTER TABLE SMEVPASS_BLOCKCLIENTS ADD CONSTRAINT FK_SMEVPASS_BLOCKCLIENTS_C_ID FOREIGN KEY (SBC_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE SMEVPASS_BLOCKCLIENTS ADD CONSTRAINT FK_SMEVPASS_BLOCKCLIENTS_UN_ID FOREIGN KEY (SBC_UNBLOCK_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
CREATE DESCENDING INDEX SMEVPASS_BLOCKCLIENTS_IDX_BLDT ON SMEVPASS_BLOCKCLIENTS (SBC_BLOCK_DATETIME);
CREATE DESCENDING INDEX SMEVPASS_BLOCKCLIENTS_IDX_UBLDT ON SMEVPASS_BLOCKCLIENTS (SBC_UNBLOCK_DATETIME);
