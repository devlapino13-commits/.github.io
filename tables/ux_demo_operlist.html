Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE UX$DEMO_OPERLIST (
    OPL_ID           D_INTEGER NOT NULL,
    OPL_TYPE         D_SMALLINT NOT NULL,
    OPL_DATE         D_DATE NOT NULL,
    OPL_DOC_NO       D_STR_20,
    OPL_FROM_REG_ID  D_INTEGER NOT NULL,
    OPL_SRC_CLI_ID   D_INTEGER,
    OPL_SRC_PLC_ID   D_INTEGER,
    OPL_TRG_CLI_ID   D_INTEGER,
    OPL_TRG_PLC_ID   D_INTEGER,
    OPL_STATE        D_SMALLINT DEFAULT 0 NOT NULL
);

ALTER TABLE UX$DEMO_OPERLIST ADD CONSTRAINT PK_UX$DEMO_OPERLIST PRIMARY KEY (OPL_ID);
ALTER TABLE UX$DEMO_OPERLIST ADD CONSTRAINT FK_UX$DEMO_OPERLIST_SRC_CLI FOREIGN KEY (OPL_SRC_CLI_ID) REFERENCES UX$DEMO_CLIENT (CLI_ID) ON UPDATE CASCADE;
ALTER TABLE UX$DEMO_OPERLIST ADD CONSTRAINT FK_UX$DEMO_OPERLIST_SRC_PLC FOREIGN KEY (OPL_SRC_PLC_ID) REFERENCES UX$DEMO_PLACE (PLC_ID) ON UPDATE CASCADE;
ALTER TABLE UX$DEMO_OPERLIST ADD CONSTRAINT FK_UX$DEMO_OPERLIST_TRG_CLI FOREIGN KEY (OPL_TRG_CLI_ID) REFERENCES UX$DEMO_CLIENT (CLI_ID) ON UPDATE CASCADE;
ALTER TABLE UX$DEMO_OPERLIST ADD CONSTRAINT FK_UX$DEMO_OPERLIST_TRG_PLC FOREIGN KEY (OPL_TRG_PLC_ID) REFERENCES UX$DEMO_PLACE (PLC_ID) ON UPDATE CASCADE;
CREATE DESCENDING INDEX UX$DEMO_OPERLIST_DATE ON UX$DEMO_OPERLIST (OPL_DATE);
CREATE INDEX UX$DEMO_OPERLIST_REG ON UX$DEMO_OPERLIST (OPL_FROM_REG_ID);
