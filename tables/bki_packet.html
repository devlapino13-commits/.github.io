Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE BKI_PACKET (
    BKP_ID             D_ID NOT NULL,
    BKP_CODE           D_STR_100 NOT NULL,
    BKP_FROM_BKK_ID    D_ID NOT NULL,
    BKP_DATE           D_DATE NOT NULL,
    BKP_NUMBER         D_INTEGER NOT NULL,
    BKP_FROM_BKT_ID    D_ID NOT NULL,
    BKP_STATUS         D_SMALLINT DEFAULT 0 NOT NULL,
    BKP_DATETIME       D_DATETIME NOT NULL,
    BKP_RESULT_STATUS  D_STR_100,
    BKP_SEND_DATETIME  D_DATETIME
);

ALTER TABLE BKI_PACKET ADD CONSTRAINT PK_BKI_PACKET PRIMARY KEY (BKP_ID);
ALTER TABLE BKI_PACKET ADD CONSTRAINT FK_BKI_PACKET_1 FOREIGN KEY (BKP_FROM_BKK_ID) REFERENCES BKI_KIND (BKK_ID);
ALTER TABLE BKI_PACKET ADD CONSTRAINT FK_BKI_PACKET_2 FOREIGN KEY (BKP_FROM_BKT_ID) REFERENCES BKI_PACKET_TYPE (BKT_ID);
CREATE UNIQUE INDEX BKI_PACKET_CODE ON BKI_PACKET (BKP_CODE);
CREATE DESCENDING INDEX BKI_PACKET_IDX1 ON BKI_PACKET (BKP_DATE);
CREATE INDEX BKI_PACKET_STATUS ON BKI_PACKET (BKP_STATUS);
