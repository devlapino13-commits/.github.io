Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE DOCRECOGFIELD (
    DRF_ID                   INTEGER NOT NULL,
    DRF_FROM_DRM_ID          INTEGER,
    DRF_CODE                 D_STR_255,
    DRF_API_CODE             D_STR_255,
    DRF_NAME                 D_STR_255,
    DRF_INACTIVE             INTEGER DEFAULT 0,
    DRF_NOT_NULL             INTEGER DEFAULT 0,
    DRF_IS_EXTRA             INTEGER DEFAULT 0,
    DRF_DATA_TYPE            INTEGER DEFAULT 0,
    DRF_TAG                  SMALLINT,
    DRF_POSITION             INTEGER DEFAULT 0,
    DRF_INVISIBLE_ON_DOC     INTEGER DEFAULT 0,
    DRF_INVISIBLE_ON_CLIENT  INTEGER DEFAULT 0,
    DRF_EDIT_MASK            D_STR_CODE default '' NOT NULL,
    DRF_EDIT_MASK_HINT       D_STR_CODE default '' NOT NULL
);

ALTER TABLE DOCRECOGFIELD ADD CONSTRAINT PK_DOCRECOGFIELD PRIMARY KEY (DRF_ID);
ALTER TABLE DOCRECOGFIELD ADD CONSTRAINT FK_DOCRECOGFIELD_DRM FOREIGN KEY (DRF_FROM_DRM_ID) REFERENCES DOCRECOGMAIN (DRM_ID) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE UNIQUE INDEX DOCRECOGFIELD_CODE ON DOCRECOGFIELD (DRF_FROM_DRM_ID, DRF_CODE);
CREATE UNIQUE INDEX DOCRECOGFIELD_POS ON DOCRECOGFIELD (DRF_FROM_DRM_ID, DRF_POSITION);
