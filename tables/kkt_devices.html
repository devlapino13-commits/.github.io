Показать DDL (CREATE TABLE) 
CREATE TABLE KKT_DEVICES (
    KKD_ID                       D_INTEGER NOT NULL,
    KKD_FROM_REG_ID              D_INTEGER NOT NULL,
    KKD_CODE                     D_STR_CODE NOT NULL,
    KKD_DATE_BEGIN               D_DATE NOT NULL,
    KKD_DATE_END                 D_DATE DEFAULT '31.12.2999' NOT NULL,
    KKD_NOTSENT_COUNT            D_INTEGER,
    KKD_NOTSENT_DATETIME         D_DATETIME,
    KKD_NOTSENT_UPDATE_DATETIME  D_DATETIME
);

ALTER TABLE KKT_DEVICES ADD CONSTRAINT PK_KKT_DEVICES PRIMARY KEY (KKD_ID);
ALTER TABLE KKT_DEVICES ADD CONSTRAINT FK_KKT_DEVICES_REG_ID FOREIGN KEY (KKD_FROM_REG_ID) REFERENCES REGION (REG_UNICODE) ON UPDATE CASCADE;
CREATE UNIQUE INDEX KKT_DEVICES_CODE ON KKT_DEVICES (KKD_CODE);
