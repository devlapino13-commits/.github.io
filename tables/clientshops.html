CREATE TABLE CLIENTSHOPS (
    CS_ID             D_ID NOT NULL,
    CS_FROM_C_ID      D_ID NOT NULL,
    CS_NAME           D_STR_255 NOT NULL,
    CS_NAME_LAT       D_STR_255 NOT NULL,
    CS_FROM_ADL_ID    D_ID,
    CS_LATITUDE       D_GEO,
    CS_LONGITUDE      D_GEO,
    CS_CONTACT_FIO    D_STR_100,
    CS_PHONE          D_STR_20 NOT NULL,
    CS_HOURS          D_STR_20,
    CS_WEEKEND_HOURS  D_STR_20,
    CS_DATETIME       D_DATETIME DEFAULT current_timestamp NOT NULL,
    CS_FROM_UN_ID     D_ID NOT NULL,
    CS_QR_DATA        D_STR_2048,
    CS_IPC_ID         D_ID,
    CS_IS_DELETE      D_BOOL DEFAULT 0 NOT NULL,
    CS_FROM_RD_ID     D_INTEGER,
    CS_FROM_AL_ID     D_STR_20,
    CS_FROM_REG_ID    D_INTEGER,
    CS_DISCOUNT       D_PERCENT
);
