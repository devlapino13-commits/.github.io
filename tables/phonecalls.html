CREATE TABLE PHONECALLS (
    PNC_ID              D_ID NOT NULL,
    PNC_SRC_PHONE_ID    D_INTEGER NOT NULL,
    PNC_DST_PHONE_ID    D_INTEGER NOT NULL,
    PNC_SRC_C_ID        D_ID NOT NULL,
    PNC_DST_C_ID        D_ID NOT NULL,
    PNC_DIRECTION       D_SMALLINT NOT NULL,
    PNC_OBJECT_KIND     D_INTEGER,
    PNC_OBJECT_ID       D_ID,
    PNC_PRV_PHONE_ID    D_INTEGER,
    PNC_CALL_STATUS_ID  D_INTEGER DEFAULT 0 NOT NULL,
    PNC_START           D_DATETIME,
    PNC_FINISH          D_DATETIME,
    PNC_DURATION        D_INTEGER DEFAULT 0 NOT NULL,
    PNC_FILENAME        D_STR_255,
    PNC_CALL_RESULT_ID  D_INTEGER DEFAULT 0 NOT NULL,
    PNC_IS_DELINQ       D_BOOL DEFAULT 0 NOT NULL,
    PNC_IS_CONTPERSON   D_BOOL DEFAULT 0 NOT NULL,
    PNC_SOURCE          D_INTEGER DEFAULT 0
);
