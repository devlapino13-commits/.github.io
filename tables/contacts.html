Показать DDL (CREATE TABLE) 
CREATE TABLE CONTACTS (
    CC_ID                INTEGER DEFAULT -1 NOT NULL,
    CC_FROM_C_ID         ID_TYPE,
    CC_NAME              STR_100,
    CC_STATUS            STR_100,
    CC_ADDRESS           STR_255,
    CC_CELL_NO           STR_255,
    CC_CHECK_RESULT      INTEGER DEFAULT 0,
    CC_IN_PROCESS        INTEGER DEFAULT 0,
    CC_TRY_KOL           INTEGER DEFAULT 0,
    CC_CALL_USER         STR_255,
    CC_TIME_BEGIN        TIMESTAMP,
    CC_TIME_END          TIMESTAMP,
    CC_FROM_UN_ID        ID_TYPE,
    CC_CHECK_MEMO        STR_255,
    CC_TYPE              INTEGER DEFAULT 0,
    CC_CLIENT_OK         SMALLINT DEFAULT 0,
    CC_NAME_OK           SMALLINT DEFAULT 0,
    CC_STATUS_OK         SMALLINT DEFAULT 0,
    CC_ADDRESS_OK        SMALLINT DEFAULT 0,
    CC_CREATE_DATE       TIMESTAMP,
    CC_CHECKED_DATETIME  TIMESTAMP,
    CC_EDIT_DATE         TIMESTAMP,
    CC_OPER_INFO         STR_255,
    CC_LANGUAGE          INTEGER DEFAULT 0,
    CC_FROM_FL_ID        ID_TYPE,
    CC_FROM_ADL_ID       ID_TYPE,
    CC_SHORTNAME         STR_100,
    CC_DATE              DATE,
    CC_TOTAL_RES         INTEGER DEFAULT 0,
    CC_CHECKED_DATE      DATE,
    CC_IS_OLD_CONTACT    SMALLINT DEFAULT 0,
    CC_BLOCKED_USER      VARCHAR(255),
    CC_BIRTH_DATE        DATE,
    CC_CELL_NO_ADD       STR_100,
    CC_SVZ_COMMENTS      STR_255,
    CC_VERIF_IGNORE      SMALLINT DEFAULT 0,
    CC_CALL_BACK         TIMESTAMP,
    CC_NAME_TMP          D_STR_100,
    CC_IS_DELETE         D_SMALLINT
);
