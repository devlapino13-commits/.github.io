    OA_ID                     D_INTEGER NOT NULL,

    OA_FROM_C_ID              D_ID NOT NULL,
    OA_DATE                   D_DATE NOT NULL,
    OA_DATETIME               D_DATETIME NOT NULL,
    OA_FROM_EXA_ID            D_INTEGER,
    OA_JSON                   D_STR_8100 NOT NULL,
    OA_STATUS                 D_SMALLINT NOT NULL,
    OA_STATUS_DATETIME        D_DATETIME NOT NULL,
    OA_FROM_KBM_ID            D_ID,
    OA_LAST_CLIENT_OFFICE_ID  D_INTEGER,
    OA_APPROVE_DATETIME       D_DATETIME,
    OA_LAST_LO_ID             D_ID,
    OA_LO_DENIED_DATETIME     D_DATETIME,
    OA_CARD_TYPE              D_SMALLINT,
    OA_MBULAK_CARD_ID         D_INTEGER,
    OA_PAN_MASKED             D_STR_20,
    OA_PAN_ENCRYPTED          D_STR_LONG,
    OA_FROM_K_ID              D_ID,
    OA_IS_FIRED_LO            D_SMALLINT
);
