CREATE TABLE MC_TEST_ACCESS (
    MTA_ID            D_INTEGER NOT NULL,
    MTA_PHONE_NUMBER  D_STR_CODE NOT NULL,
    MTA_PIN_CODE      D_STR_CODE NOT NULL,
    MTA_FROM_C_ID     D_ID NOT NULL,
