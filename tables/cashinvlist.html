Показать DDL (CREATE TABLE) 
CREATE TABLE CASHINVLIST (
    CIL_ID            D_INTEGER NOT NULL,
    CIL_PARENT_ID     D_INTEGER,
    CIL_FROM_CSP_ID   D_INTEGER NOT NULL,
    CIL_DATE          D_DATE NOT NULL,
    CIL_OPER_KIND     D_INTEGER DEFAULT 0 NOT NULL,
    CIL_OFFICE_ID     D_INTEGER NOT NULL,
    CIL_CASHIER_ID    D_ID,
    CIL_INSPECTOR_ID  D_ID,
    CIL_NEED_BONDS    D_BOOL DEFAULT 0 NOT NULL,
    CIL_INV_AMOUNT    D_CURRENCY DEFAULT 0 NOT NULL,
    CIL_BOND_AMOUNT   D_CURRENCY NOT NULL,
    CIL_FROM_BLS_ID   D_ID,
    CIL_DATETIME      D_DATETIME NOT NULL,
    CIL_FROM_UN_ID    D_ID NOT NULL,
    CIL_FROM_EDL_ID   D_ID,
    CIL_IS_DELETE     D_BOOL DEFAULT 0 NOT NULL,
    CIL_STATE         D_SMALLINT DEFAULT 0 NOT NULL,
    CIL_IS_CLOSED     D_BOOL DEFAULT 0 NOT NULL
);

ALTER TABLE CASHINVLIST ADD CONSTRAINT PK_CASHINVLIST PRIMARY KEY (CIL_ID)
USING DESCENDING INDEX PK_CASHINVLIST;

  USING DESCENDING INDEX FK_CASHINVLIST_BLS;
ALTER TABLE CASHINVLIST ADD CONSTRAINT FK_CASHINVLIST_CASHIER FOREIGN KEY (CIL_CASHIER_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE CASHINVLIST ADD CONSTRAINT FK_CASHINVLIST_CSP FOREIGN KEY (CIL_FROM_CSP_ID) REFERENCES CASHPLACE (CSP_ID) ON UPDATE CASCADE;
ALTER TABLE CASHINVLIST ADD CONSTRAINT FK_CASHINVLIST_EDL FOREIGN KEY (CIL_FROM_EDL_ID) REFERENCES EDOC_LIST (EDL_ID) ON UPDATE CASCADE;
ALTER TABLE CASHINVLIST ADD CONSTRAINT FK_CASHINVLIST_INSPECTOR FOREIGN KEY (CIL_INSPECTOR_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE CASHINVLIST ADD CONSTRAINT FK_CASHINVLIST_OFFICE FOREIGN KEY (CIL_OFFICE_ID) REFERENCES REGION (REG_UNICODE) ON UPDATE CASCADE;
ALTER TABLE CASHINVLIST ADD CONSTRAINT FK_CASHINVLIST_PARENT FOREIGN KEY (CIL_PARENT_ID) REFERENCES CASHINVLIST (CIL_ID) ON DELETE CASCADE ON UPDATE CASCADE
  USING DESCENDING INDEX FK_CASHINVLIST_PARENT;
