CREATE TABLE KONTREFUND_DISB (
    KRD_ID                D_ID NOT NULL,
    KRD_FROM_K_ID         D_ID NOT NULL,
    KRD_AMOUNT            D_CURRENCY NOT NULL,
    KRD_FROM_OL_ID        D_ID,
    KRD_STATUS            D_SMALLINT DEFAULT 0 NOT NULL,
    KRD_CREATE_UN_ID      D_ID NOT NULL,
    KRD_CREATE_DATETIME   D_DATETIME,
    KRD_CONFIRM_UN_ID     D_ID,
    KRD_CONFIRM_DATETIME  D_DATETIME
);

ALTER TABLE KONTREFUND_DISB ADD CONSTRAINT PK_KONTREFUND_DISB PRIMARY KEY (KRD_ID);
ALTER TABLE KONTREFUND_DISB ADD CONSTRAINT FK_KONTREFUND_DISB_K_ID FOREIGN KEY (KRD_FROM_K_ID) REFERENCES KONTRACT (K_ID) ON UPDATE CASCADE;
ALTER TABLE KONTREFUND_DISB ADD CONSTRAINT FK_KONTREFUND_DISB_OL FOREIGN KEY (KRD_FROM_OL_ID) REFERENCES OPERLIST (OL_ID) ON UPDATE CASCADE;
ALTER TABLE KONTREFUND_DISB ADD CONSTRAINT FK_KONTREFUND_DISB_UN_CONFIRM FOREIGN KEY (KRD_CONFIRM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
ALTER TABLE KONTREFUND_DISB ADD CONSTRAINT FK_KONTREFUND_DISB_UN_CREATE FOREIGN KEY (KRD_CREATE_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
