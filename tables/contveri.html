Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE CONTVERI (
    CVR_ID           D_INTEGER NOT NULL,
    CVR_FROM_CDA_ID  D_ID NOT NULL,
    CVR_HASH_PIN     D_STR_CODE,
    CVR_STATUS       D_SMALLINT DEFAULT 0 NOT NULL,
    CVR_SIGN_KIND    D_INTEGER DEFAULT 0,
    CVR_FROM_UN_ID   D_ID,
    CVR_DATETIME     D_DATETIME
);

ALTER TABLE CONTVERI ADD CONSTRAINT CHK1_CONTVERI_STATUS CHECK (cvr_status in (-1, 0, 1));
ALTER TABLE CONTVERI ADD CONSTRAINT PK_CONTVERI PRIMARY KEY (CVR_ID);
ALTER TABLE CONTVERI ADD CONSTRAINT FK_CONTVERI_CDA FOREIGN KEY (CVR_FROM_CDA_ID) REFERENCES CONTDATA (CDA_ID) ON DELETE CASCADE ON UPDATE CASCADE
  USING DESCENDING INDEX FK_CONTVERI_CDS;
ALTER TABLE CONTVERI ADD CONSTRAINT FK_CONTVERI_UN FOREIGN KEY (CVR_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
