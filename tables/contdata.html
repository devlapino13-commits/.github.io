Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE CONTDATA (
    CDA_ID           D_ID NOT NULL,
    CDA_FROM_C_ID    D_ID NOT NULL,
    CDA_CONT_KIND    D_INTEGER NOT NULL,
    CDA_FROM_CLT_ID  D_INTEGER NOT NULL,
    CDA_DATE         D_DATE NOT NULL,
    CDA_STATUS       D_SMALLINT DEFAULT 0 NOT NULL,
    CDA_IS_DELETE    D_SMALLINT DEFAULT 0 NOT NULL,
    CDA_IS_MAIN      D_SMALLINT DEFAULT 0 NOT NULL,
    CDA_FROM_UN_ID   D_ID
);

ALTER TABLE CONTDATA ADD CONSTRAINT CHK1_CONTDATA_STATUS CHECK (cda_status in (0, 1, 2));
ALTER TABLE CONTDATA ADD CONSTRAINT CHK1_CONTDATA_DELETE CHECK (cda_is_delete in (0, 1));
ALTER TABLE CONTDATA ADD CONSTRAINT PK_CONTDATA PRIMARY KEY (CDA_ID)
USING DESCENDING INDEX PK_CONTDATA;
ALTER TABLE CONTDATA ADD CONSTRAINT FK_CONTDATA_CLT_ID FOREIGN KEY (CDA_FROM_CLT_ID) REFERENCES CONTLIST (CLT_ID) ON UPDATE CASCADE;
ALTER TABLE CONTDATA ADD CONSTRAINT FK_CONTDATA_C_ID FOREIGN KEY (CDA_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE
  USING DESCENDING INDEX FK_CONTDATA_C_ID;
ALTER TABLE CONTDATA ADD CONSTRAINT FK_CONTDATA_UN FOREIGN KEY (CDA_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
CREATE INDEX CONTDATA_CONT_KIND ON CONTDATA (CDA_CONT_KIND);
CREATE INDEX CONTDATA_DATE ON CONTDATA (CDA_DATE);
