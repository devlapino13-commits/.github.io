Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE UX$FRDATA (
    UFD_UFR_UUID  D_UUID NOT NULL,
    UFD_KIND      D_INTEGER NOT NULL,
    UFD_US_UUID   D_UUID
);

ALTER TABLE UX$FRDATA ADD CONSTRAINT CHK1_UX$FRDATA_KIND CHECK (ufd_kind in (1, 2, 3));
ALTER TABLE UX$FRDATA ADD CONSTRAINT PK_UX$FRDATA PRIMARY KEY (UFD_UFR_UUID, UFD_KIND);
ALTER TABLE UX$FRDATA ADD CONSTRAINT FK_UX$FRDATA_UFR_UUID FOREIGN KEY (UFD_UFR_UUID) REFERENCES UX$FRAME (UFR_UUID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE UX$FRDATA ADD CONSTRAINT FK_UX$FRDATA_US_UUID FOREIGN KEY (UFD_US_UUID) REFERENCES UX$SOURCE (US_UUID) ON UPDATE CASCADE;
