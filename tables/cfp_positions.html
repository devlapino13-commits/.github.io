Показать DDL (CREATE TABLE) 
CREATE TABLE CFP_POSITIONS (
    CP_ID                D_INTEGER NOT NULL,
    CP_PARENET_ID        D_INTEGER,
    CP_NAME_RU           D_STR_100,
    CP_NAME_KG           D_STR_100,
    CP_TYPE              D_INTEGER,
    CP_IS_DELETE         D_BOOL DEFAULT 0,
    CP_FROM_UN_ID        D_ID,
    CP_DATETIME          D_DATETIME,
    CP_FULL_NAME_RU      D_STR_255,
    CP_FULL_NAME_KG      D_STR_255,
    CP_INCOME_PREFIX_RU  D_STR_255,
    CP_INCOME_PREFIX_KG  D_STR_255
);

ALTER TABLE CFP_POSITIONS ADD CONSTRAINT PK_CFP_POSITIONS PRIMARY KEY (CP_ID);
ALTER TABLE CFP_POSITIONS ADD CONSTRAINT FK_CFP_POSITIONS_1 FOREIGN KEY (CP_PARENET_ID) REFERENCES CFP_POSITIONS (CP_ID) ON UPDATE CASCADE;
ALTER TABLE CFP_POSITIONS ADD CONSTRAINT FK_CFP_POSITIONS_2 FOREIGN KEY (CP_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
