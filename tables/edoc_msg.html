Показать DDL (CREATE TABLE) 
CREATE TABLE EDOC_MSG (
    EDM_FROM_EDS_ID  D_ID NOT NULL,
    EDM_FROM_MSG_ID  D_INTEGER NOT NULL,
    EDM_FROM_CLT_ID  D_INTEGER,
    EDM_FROM_CDA_ID  D_ID,
    EDM_HASH_PIN     D_STR_CODE,
    EDM_STATUS       D_SMALLINT

ALTER TABLE EDOC_MSG ADD CONSTRAINT CHK1_EDOC_MSG_STATUS CHECK (edm_status in (-1, 0, 1));
ALTER TABLE EDOC_MSG ADD CONSTRAINT PK_EDOC_MSG PRIMARY KEY (EDM_FROM_EDS_ID, EDM_FROM_MSG_ID)
USING DESCENDING INDEX PK_EDOC_MSG;
ALTER TABLE EDOC_MSG ADD CONSTRAINT FK_EDOC_MSG_CDA FOREIGN KEY (EDM_FROM_CDA_ID) REFERENCES CONTDATA (CDA_ID) ON UPDATE CASCADE;
ALTER TABLE EDOC_MSG ADD CONSTRAINT FK_EDOC_MSG_CLT FOREIGN KEY (EDM_FROM_CLT_ID) REFERENCES CONTLIST (CLT_ID) ON UPDATE CASCADE;
ALTER TABLE EDOC_MSG ADD CONSTRAINT FK_EDOC_MSG_EDS FOREIGN KEY (EDM_FROM_EDS_ID) REFERENCES EDOC_SIGN (EDS_ID) ON DELETE CASCADE ON UPDATE CASCADE
  USING DESCENDING INDEX FK_EDOC_MSG_EDS;
