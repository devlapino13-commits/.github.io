Определение

Показать DDL (CREATE TABLE) 
CREATE GLOBAL TEMPORARY TABLE UXDEPEND (
    UDP_CODE          D_STR_255 NOT NULL,
    UDP_PARENT_CODE   D_STR_255,
    UDP_OWNER_CODE    D_STR_100 NOT NULL,
    UDP_UX_CODE       D_STR_100,
    UDP_UX_KIND       D_STR_CODE,
    UDP_POSITION      D_INTEGER NOT NULL,
    UDP_CAPTION       D_STR_255,
    UDP_IMAGE_ID      D_INTEGER DEFAULT -1 NOT NULL,
    UDP_LOCK_DELETE   D_BOOL DEFAULT 0 NOT NULL,
    UDP_CHILD_LOCKED  D_BOOL DEFAULT 0 NOT NULL
) ON COMMIT PRESERVE ROWS;

ALTER TABLE UXDEPEND ADD CONSTRAINT PK_UXDEPEND PRIMARY KEY (UDP_CODE);
ALTER TABLE UXDEPEND ADD CONSTRAINT FK_UXDEPEND_PARENT FOREIGN KEY (UDP_PARENT_CODE) REFERENCES UXDEPEND (UDP_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE INDEX UXDEPEND_OWNER ON UXDEPEND (UDP_OWNER_CODE);
CREATE INDEX UXDEPEND_UX ON UXDEPEND (UDP_UX_CODE);
