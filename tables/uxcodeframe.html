Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE UXCODEFRAME (
    UCF_UX_CODE          D_STR_CODE NOT NULL,
    UCF_UFR_CODE         D_STR_CODE NOT NULL,
    UCF_MASTER_UFR_CODE  D_STR_CODE,
    UCF_IS_MAINFRAME     D_BOOL NOT NULL,
    UCF_MASTER_FIELD     D_STR_CODE NOT NULL,
    UCF_DETAIL_FIELD     D_STR_CODE NOT NULL
);

ALTER TABLE UXCODEFRAME ADD CONSTRAINT PK_UXCODEFRAME PRIMARY KEY (UCF_UX_CODE, UCF_UFR_CODE);
ALTER TABLE UXCODEFRAME ADD CONSTRAINT FK_UXCODEFRAME_FR_CODE FOREIGN KEY (UCF_UFR_CODE) REFERENCES UXFRAME (UFR_CODE) ON UPDATE CASCADE;
ALTER TABLE UXCODEFRAME ADD CONSTRAINT FK_UXCODEFRAME_MASTER_UFR_CODE FOREIGN KEY (UCF_MASTER_UFR_CODE) REFERENCES UXFRAME (UFR_CODE) ON UPDATE CASCADE;
ALTER TABLE UXCODEFRAME ADD CONSTRAINT FK_UXCODEFRAME_UX_CODE FOREIGN KEY (UCF_UX_CODE) REFERENCES UXCODE (UX_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
