Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE EDOC_CFG (
    EDC_ID                  D_INTEGER NOT NULL,
    EDC_CODE                D_STR_CODE NOT NULL,
    EDC_NAME                D_STR_100 NOT NULL,
    EDC_OBJECT_KIND         D_INTEGER NOT NULL,
    EDC_DATE_BEGIN          D_DATE NOT NULL,
    EDC_DATE_END            D_DATE,
    EDC_STATE               D_SMALLINT DEFAULT 0,
    EDC_EMAIL_ON_DRAFT      D_STR_255,
    EDC_DEFAULT_MANAGER_ID  D_ID,
    EDC_SMS_MASK            D_STR_LONG,
    EDC_FROM_REG_ID         D_INTEGER,
    EDC_PROCESS_KIND        D_SMALLINT DEFAULT 1 NOT NULL,
    EDC_USE_LAST_CONTDATA   D_SMALLINT DEFAULT 0 NOT NULL,
    EDC_OBJECT_UNIQ         D_SMALLINT DEFAULT 0 NOT NULL,
    EDC_NEED_PIN            D_BOOL DEFAULT 1 NOT NULL,
    EDC_PUSH_TITLE          D_STR_100 DEFAULT '' NOT NULL,
    EDC_PUSH_BODY           D_STR_255 DEFAULT '' NOT NULL,
    EDC_MSG_KIND            D_SMALLINT DEFAULT 0 NOT NULL,
    EDC_PERS_INFO           D_STR_255,
    EDC_PERSTEMP_KIND       D_SMALLINT DEFAULT 0 NOT NULL
);

ALTER TABLE EDOC_CFG ADD CONSTRAINT CHK1_EDOC_CFG_STATE CHECK (edc_state in (0, 1, 2, 3));
ALTER TABLE EDOC_CFG ADD CONSTRAINT PK_EDOC_CFG PRIMARY KEY (EDC_ID);
ALTER TABLE EDOC_CFG ADD CONSTRAINT FK_EDOC_CFG_MANAGER FOREIGN KEY (EDC_DEFAULT_MANAGER_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE EDOC_CFG ADD CONSTRAINT FK_EDOC_CFG_REG_ID FOREIGN KEY (EDC_FROM_REG_ID) REFERENCES REGION (REG_UNICODE) ON UPDATE CASCADE;
CREATE UNIQUE INDEX EDOC_CFG_CODE ON EDOC_CFG (EDC_CODE);
