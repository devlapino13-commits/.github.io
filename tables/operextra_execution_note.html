Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE OPEREXTRA_EXECUTION_NOTE (
    OEN_ID                      D_ID NOT NULL,
    OEN_FROM_OL_ID_CANCEL       D_ID,
    OEN_FROM_OX_ID_CANCEL       D_INTEGER,
    OEN_FROM_RD_ID              D_INTEGER,
    OEN_OBJECT_KIND             D_INTEGER,
    OEN_OBJECT_ID               D_ID,
    OEN_FROM_OL_ID_RESTORATION  D_ID,
    OEN_FROM_OX_ID_RESTORATION  D_INTEGER
);

ALTER TABLE OPEREXTRA_EXECUTION_NOTE ADD CONSTRAINT PK_OPEREXTRA_EXECUTION_NOTE PRIMARY KEY (OEN_ID);
ALTER TABLE OPEREXTRA_EXECUTION_NOTE ADD CONSTRAINT FK_OPEREXTRA_EXECUTION_NOTE_1 FOREIGN KEY (OEN_FROM_OL_ID_CANCEL, OEN_FROM_OX_ID_CANCEL) REFERENCES OPEREXTRA (OX_FROM_OL_ID, OX_ID) ON UPDATE CASCADE
  USING INDEX FK_OPEREXTRA_EXECUTION_NOTE_OX_CANCEL;
ALTER TABLE OPEREXTRA_EXECUTION_NOTE ADD CONSTRAINT FK_OPEREXTRA_EXECUTION_NOTE_2 FOREIGN KEY (OEN_FROM_OL_ID_RESTORATION, OEN_FROM_OX_ID_RESTORATION) REFERENCES OPEREXTRA (OX_FROM_OL_ID, OX_ID) ON UPDATE CASCADE
  USING INDEX FK_OPEREXTRA_EXECUTION_NOTE_RESTORATION;
ALTER TABLE OPEREXTRA_EXECUTION_NOTE ADD CONSTRAINT FK_OPEREXTRA_EXECUTION_NOTE_RD_ID FOREIGN KEY (OEN_FROM_RD_ID) REFERENCES REFDATA (RD_ID) ON UPDATE CASCADE;
CREATE INDEX OPEREXTRA_EXECUTION_NOTE_OBJ ON OPEREXTRA_EXECUTION_NOTE (OEN_OBJECT_KIND, OEN_OBJECT_ID);
