Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE BKI_MANUAL_MAIN (
    BMM_ID           D_ID NOT NULL,
    BMM_FROM_BKK_ID  D_ID NOT NULL,
    BMM_FROM_BKT_ID  D_ID,
    BMM_STATUS       D_INTEGER DEFAULT 0 NOT NULL,
    BMM_MEMO         D_STR_255,
    BMM_FROM_BKP_ID  D_ID
);

ALTER TABLE BKI_MANUAL_MAIN ADD CONSTRAINT PK_BKI_MANUAL_MAIN PRIMARY KEY (BMM_ID);
ALTER TABLE BKI_MANUAL_MAIN ADD CONSTRAINT FK_BKI_MANUAL_MAIN_BKK FOREIGN KEY (BMM_FROM_BKK_ID) REFERENCES BKI_KIND (BKK_ID) ON UPDATE CASCADE
  USING INDEX FK_BKI_MANUAL_MAIN_1;
ALTER TABLE BKI_MANUAL_MAIN ADD CONSTRAINT FK_BKI_MANUAL_MAIN_BKP FOREIGN KEY (BMM_FROM_BKP_ID) REFERENCES BKI_PACKET (BKP_ID) ON UPDATE CASCADE;
ALTER TABLE BKI_MANUAL_MAIN ADD CONSTRAINT FK_BKI_MANUAL_MAIN_BKT FOREIGN KEY (BMM_FROM_BKT_ID) REFERENCES BKI_PACKET_TYPE (BKT_ID) ON UPDATE CASCADE
  USING INDEX FK_BKI_MANUAL_MAIN_2;
CREATE INDEX BKI_MANUAL_MAIN_IDX1 ON BKI_MANUAL_MAIN (BMM_STATUS);
