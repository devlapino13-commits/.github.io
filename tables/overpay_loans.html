Показать DDL (CREATE TABLE) 
CREATE TABLE OVERPAY_LOANS (
    OPL_ID            D_ID NOT NULL,
    OPL_FROM_OPM_ID   D_ID NOT NULL,
    OPL_FROM_K_ID     D_ID NOT NULL,
    OPL_TYPE          D_SMALLINT NOT NULL,
    OPL_SUM           D_CURRENCY DEFAULT 0 NOT NULL,
    OPL_DATE          D_DATE,
    OPL_RES_SUM       D_CURRENCY DEFAULT 0 NOT NULL,
    OPL_OVERDUE_DAYS  D_INTEGER DEFAULT 0 NOT NULL,
    OPL_NO            D_SMALLINT

ALTER TABLE OVERPAY_LOANS ADD CONSTRAINT FK_OVERPAY_LOANS_K FOREIGN KEY (OPL_FROM_K_ID) REFERENCES KONTRACT (K_ID) ON UPDATE CASCADE;
