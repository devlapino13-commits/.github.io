CREATE TABLE MINIAPP_VERIFY (
    MV_ID                 INTEGER NOT NULL,
    MV_FROM_MA_ID         INTEGER NOT NULL,
    MV_FROM_C_ID          ID_TYPE NOT NULL,
    MV_FROM_RD_ID         INTEGER NOT NULL,
    MV_CONTACT_VALUE      STR_20 NOT NULL,
    MV_CONTACT_IS_VERIFY  INTEGER DEFAULT 1 NOT NULL,
    MV_SENDSMS_UN_ID      ID_TYPE,
    MV_SENDSMS_DATE       TIMESTAMP,
    MV_SENDSMS_COUNT      INTEGER DEFAULT 0 NOT NULL,
    MV_CONFSMS_UN_ID      ID_TYPE,
    MV_CONFSMS_DATE       TIMESTAMP,
    MV_PINCODE_HASH       STR_100,
    MV_IS_DELETE          INTEGER DEFAULT 0 NOT NULL
);

ALTER TABLE MINIAPP_VERIFY ADD CONSTRAINT PK_MINIAPP_VERIFY PRIMARY KEY (MV_ID);

ALTER TABLE MINIAPP_VERIFY ADD CONSTRAINT FK_MINIAPP_VERIFY_C_ID FOREIGN KEY (MV_FROM_C_ID) REFERENCES CLIENTS (C_ID);
ALTER TABLE MINIAPP_VERIFY ADD CONSTRAINT FK_MINIAPP_VERIFY_SENDSMS_UN_ID FOREIGN KEY (MV_SENDSMS_UN_ID) REFERENCES USERNAME (UN_ID);
