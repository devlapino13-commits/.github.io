Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE IPC_GC_ANALIZ (
    IGA_ID             D_INTEGER NOT NULL,
    IGA_FROM_IFA_ID    D_INTEGER NOT NULL,
    IGA_DATE_BEGIN     D_DATE NOT NULL,
    IGA_DATE_END       D_DATE NOT NULL,
    IGA_IS_QR_PROBLEM  D_BOOL DEFAULT 0 NOT NULL,
    IGA_GC_QR_SUMM     D_CURRENCY DEFAULT 0 NOT NULL,
    IGA_DB_QR_SUMM     D_CURRENCY DEFAULT 0 NOT NULL,
    IGA_FROM_UN_ID     D_ID
);

ALTER TABLE IPC_GC_ANALIZ ADD CONSTRAINT PK_IPC_GC_ANALIZ PRIMARY KEY (IGA_ID);
ALTER TABLE IPC_GC_ANALIZ ADD CONSTRAINT FK_IPC_GC_ANALIZ_IFA_ID FOREIGN KEY (IGA_FROM_IFA_ID) REFERENCES IPC_FILE_ANALIZ (IFA_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE IPC_GC_ANALIZ ADD CONSTRAINT FK_IPC_GC_ANALIZ_UN_ID FOREIGN KEY (IGA_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
