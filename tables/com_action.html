Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE COM_ACTION (
    CMA_ID           D_ID NOT NULL,
    CMA_FROM_CML_ID  D_ID NOT NULL,
    CMA_STATUS       D_INTEGER NOT NULL,
    CMA_FROM_CMP_ID  D_INTEGER,
    CMA_FROM_UN_ID   D_ID NOT NULL,
    CMA_DATETIME     D_DATETIME
);

ALTER TABLE COM_ACTION ADD CONSTRAINT PK_COM_ACTION PRIMARY KEY (CMA_ID)
USING DESCENDING INDEX PK_COM_ACTION;
ALTER TABLE COM_ACTION ADD CONSTRAINT FK_COM_ACTION_CML FOREIGN KEY (CMA_FROM_CML_ID) REFERENCES COM_LIST (CML_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE COM_ACTION ADD CONSTRAINT FK_COM_ACTION_CMP FOREIGN KEY (CMA_FROM_CMP_ID) REFERENCES COM_PACKET (CMP_ID) ON UPDATE CASCADE;
ALTER TABLE COM_ACTION ADD CONSTRAINT FK_COM_ACTION_UN FOREIGN KEY (CMA_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
