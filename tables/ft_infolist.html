Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE FT_INFOLIST (
    FIL_ID             D_ID NOT NULL,
    FIL_FROM_FIB_ID    D_ID NOT NULL,
    FIL_FROM_C_ID      D_ID NOT NULL,
    FIL_DATE_OPEN      D_DATE NOT NULL,
    FIL_DATE_CLOSE     D_DATE,
    FIL_IS_CLOSED      D_SMALLINT DEFAULT 0 NOT NULL,
    FIL_MEMO           D_STR_LONG,
    FIL_BLOCK_STATE    D_SMALLINT DEFAULT 0 NOT NULL,
    FIL_LAST_UN_ID     D_ID,
    FIL_SCHEDULE_DATE  D_DATE,
    FIL_LAST_FIC_ID    D_ID,
    FIL_FROM_REG_ID    D_INTEGER
);

ALTER TABLE FT_INFOLIST ADD CONSTRAINT PK_FT_INFOLIST PRIMARY KEY (FIL_ID)
USING DESCENDING INDEX PK_FT_INFOLIST;
ALTER TABLE FT_INFOLIST ADD CONSTRAINT FK_FT_INFOLIST_C_ID FOREIGN KEY (FIL_FROM_C_ID) REFERENCES CLIENTS (C_ID) ON UPDATE CASCADE;
ALTER TABLE FT_INFOLIST ADD CONSTRAINT FK_FT_INFOLIST_FIB FOREIGN KEY (FIL_FROM_FIB_ID) REFERENCES FT_INFOBLANK (FIB_ID) ON UPDATE CASCADE;
ALTER TABLE FT_INFOLIST ADD CONSTRAINT FK_FT_INFOLIST_FIC FOREIGN KEY (FIL_LAST_FIC_ID) REFERENCES FT_INFOCHECK (FIC_ID) ON DELETE SET NULL ON UPDATE CASCADE;
ALTER TABLE FT_INFOLIST ADD CONSTRAINT FK_FT_INFOLIST_REG FOREIGN KEY (FIL_FROM_REG_ID) REFERENCES REGION (REG_UNICODE) ON UPDATE CASCADE;
ALTER TABLE FT_INFOLIST ADD CONSTRAINT FK_FT_INFOLIST_UN FOREIGN KEY (FIL_LAST_UN_ID) REFERENCES USERNAME (UN_ID) ON UPDATE CASCADE;
CREATE DESCENDING INDEX FT_INFOLIST_SH_DATE ON FT_INFOLIST (FIL_SCHEDULE_DATE);
