Показать DDL (CREATE TABLE) 
CREATE TABLE MENULIST (
    ML_ID           INTEGER DEFAULT 0 NOT NULL,
    ML_PARENT_ID    INTEGER,
    ML_CODE         VARCHAR(100) NOT NULL,
    ML_PARENT_CODE  VARCHAR(100),
    ML_CAPTION      VARCHAR(255),
    ML_DESCRIPTION  VARCHAR(1024),
    ML_INDEX        D_INTEGER DEFAULT -1,
    ML_POSITION     D_INTEGER DEFAULT -1 NOT NULL,
    ML_IS_REPORT    INTEGER DEFAULT 0,
    ML_SECURE_KIND  INTEGER DEFAULT 0,
    ML_LEVEL        INTEGER DEFAULT 0,
    ML_HAS_CHILD    INTEGER DEFAULT 0,
    ML_KIND         D_SMALLINT DEFAULT 0 NOT NULL,
    ML_UA_CODE      D_STR_100
);

ALTER TABLE MENULIST ADD CONSTRAINT PK_MENULIST PRIMARY KEY (ML_ID);
ALTER TABLE MENULIST ADD CONSTRAINT FK_MENULIST_1 FOREIGN KEY (ML_PARENT_ID) REFERENCES MENULIST (ML_ID) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE MENULIST ADD CONSTRAINT FK_MENULIST_UA_CODE FOREIGN KEY (ML_UA_CODE) REFERENCES UXACTION (UA_CODE) ON UPDATE CASCADE;
CREATE UNIQUE INDEX MENULIST_CODE ON MENULIST (ML_CODE);
