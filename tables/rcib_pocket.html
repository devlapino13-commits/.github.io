CREATE TABLE RCIB_POCKET (
    RCP_ID                 ID_TYPE NOT NULL,
    RCP_DATE_BEGIN         DATE,
    RCP_DATE               DATE,
    RCP_VERSION            INTEGER,
    RCP_FILENAME           VARCHAR(255),
    RCP_SENDED             SMALLINT DEFAULT 0,
    RCP_CREATE_DATETIME    TIMESTAMP,
    RCP_SENDED_DATETIME    TIMESTAMP,
    RCP_CREATE_DATE        DATE,
    RCP_ERR_FILE_DATETIME  TIMESTAMP,
    RCP_TYPE               SMALLINT DEFAULT 1,
    RCP_HAS_ERR_FILE       SMALLINT DEFAULT 0,
    RCP_BASE_RCP_ID        ID_TYPE,
    RCP_IS_KREDIT          SMALLINT
);

ALTER TABLE RCIB_POCKET ADD CONSTRAINT PK_RCIB_POCKET PRIMARY KEY (RCP_ID)
USING DESCENDING INDEX PK_RCIB_POCKET;
ALTER TABLE RCIB_POCKET ADD CONSTRAINT FK_RCIB_POCKET_1 FOREIGN KEY (RCP_BASE_RCP_ID) REFERENCES RCIB_POCKET (RCP_ID) ON DELETE CASCADE ON UPDATE CASCADE
  USING DESCENDING INDEX FK_RCIB_POCKET_1;
CREATE DESCENDING INDEX RCIB_POCKET_IDX1 ON RCIB_POCKET (RCP_DATE, RCP_VERSION);
