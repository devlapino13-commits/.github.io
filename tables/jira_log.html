Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE JIRA_LOG (
    JRL_ID             D_INTEGER NOT NULL,
    JRL_ISSUE_ID       D_INTEGER NOT NULL,
    JRL_OLD_STATE      D_SMALLINT NOT NULL,
    JRL_NEW_STATE      D_SMALLINT NOT NULL,
    JRL_DEV_ID         D_ID,
    JRL_KPI_TIMEZONE   D_SMALLINT DEFAULT 0 NOT NULL,
    JRL_OLD_DATETIME   D_DATETIME NOT NULL,
    JRL_NEW_DATETIME   D_DATETIME NOT NULL,
    JRL_ORIG_DATETIME  D_STR_CODE NOT NULL
);

ALTER TABLE JIRA_LOG ADD CONSTRAINT PK_JIRA_LOG PRIMARY KEY (JRL_ID)
USING DESCENDING INDEX PK_JIRA_LOG;
ALTER TABLE JIRA_LOG ADD CONSTRAINT FK_JIRA_LOG_DEV FOREIGN KEY (JRL_DEV_ID) REFERENCES JIRA_DEV (JRD_ID) ON UPDATE CASCADE;
ALTER TABLE JIRA_LOG ADD CONSTRAINT FK_JIRA_LOG_ISSUE FOREIGN KEY (JRL_ISSUE_ID) REFERENCES JIRA_ISSUES (JRI_ISSUE_ID) ON DELETE CASCADE ON UPDATE CASCADE
  USING DESCENDING INDEX FK_JIRA_LOG_ISSUE;
CREATE DESCENDING INDEX JIRA_LOG_DATETIME ON JIRA_LOG (JRL_NEW_DATETIME);
CREATE DESCENDING INDEX JIRA_LOG_OLD_DATETIME ON JIRA_LOG (JRL_OLD_DATETIME);
