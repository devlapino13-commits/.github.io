Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE BKI_ERRORS (
    BKE_ID              D_ID NOT NULL,
    BKE_ERROR_CODE      D_STR_CODE,
    BKE_ERROR_ID        D_ID NOT NULL,
    BKE_DATE            D_DATE NOT NULL,
    BKE_FIXED           D_BOOL DEFAULT 0 NOT NULL,
    BKE_FIXED_UN_ID     D_ID,
    BKE_FIXED_DATETIME  D_DATETIME,
    BKE_FROM_BEV_ID     D_ID NOT NULL
);

ALTER TABLE BKI_ERRORS ADD CONSTRAINT PK_BKI_ERRORS PRIMARY KEY (BKE_ID);
ALTER TABLE BKI_ERRORS ADD CONSTRAINT FK_BKI_ERRORS_MML FOREIGN KEY (BKE_ERROR_ID) REFERENCES MEMOLIST (MML_ID) ON UPDATE CASCADE
  USING INDEX FK_BKI_ERRORS_2;
CREATE INDEX AFK_BKI_ERRORS_BEV_ID ON BKI_ERRORS (BKE_FROM_BEV_ID);
CREATE DESCENDING INDEX BKI_ERRORS_IDX1 ON BKI_ERRORS (BKE_DATE);
