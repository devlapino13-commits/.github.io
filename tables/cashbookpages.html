Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE CASHBOOKPAGES (
    CBP_DATE            DATE NOT NULL,
    CBP_FROM_REG_ID     INTEGER NOT NULL,
    CBP_NUMBER          INTEGER NOT NULL,
    CBP_DATETIME        TIMESTAMP,
    CBP_FIRST_PAGE_NUM  INTEGER,
    CBP_LAST_PAGE_NUM   INTEGER,
    CBP_SALDO_BEGIN     CURR_TYPE,
    CBP_SALDO_END       CURR_TYPE,
    CBP_FROM_UN_ID      ID_TYPE,
    CBP_IS_ACTUAL       SMALLINT,
    CBP_CHANGED_DATE    DATE
);

ALTER TABLE CASHBOOKPAGES ADD CONSTRAINT PK_CASHBOOKPAGES PRIMARY KEY (CBP_DATE, CBP_FROM_REG_ID, CBP_NUMBER);
ALTER TABLE CASHBOOKPAGES ADD CONSTRAINT FK_CASHBOOKPAGES_1 FOREIGN KEY (CBP_FROM_REG_ID) REFERENCES REGION (REG_UNICODE) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE CASHBOOKPAGES ADD CONSTRAINT FK_CASHBOOKPAGES_2 FOREIGN KEY (CBP_FROM_UN_ID) REFERENCES USERNAME (UN_ID) ON DELETE SET NULL ON UPDATE SET NULL;
