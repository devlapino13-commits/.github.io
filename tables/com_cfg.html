Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE COM_CFG (
    CMC_ID                 D_INTEGER NOT NULL,
    CMC_CODE               D_STR_CODE NOT NULL,
    CMC_NAME               D_STR_100 NOT NULL,
    CMC_COM_KIND           D_INTEGER NOT NULL,
    CMC_OBJECT_KIND        D_INTEGER NOT NULL,
    CMC_FROM_EDC_ID        D_INTEGER,
    CMC_STATE              D_SMALLINT DEFAULT 0 NOT NULL,
    CMC_SUPPORT_EMAIL      D_STR_LONG,
    CMC_STAFF_EMAIL        D_STR_LONG,
    CMC_STAFF_NOTIFY_KIND  D_SMALLINT DEFAULT 0 NOT NULL,
    CMC_AUTO_APPROVE       D_SMALLINT DEFAULT 0 NOT NULL
);

ALTER TABLE COM_CFG ADD CONSTRAINT CHK1_COM_CFG_STATE CHECK (cmc_state in (0, 1, 2, 3));
ALTER TABLE COM_CFG ADD CONSTRAINT PK_COM_CFG PRIMARY KEY (CMC_ID);
ALTER TABLE COM_CFG ADD CONSTRAINT FK_COM_CFG_EDOC FOREIGN KEY (CMC_FROM_EDC_ID) REFERENCES EDOC_CFG (EDC_ID) ON UPDATE CASCADE;
