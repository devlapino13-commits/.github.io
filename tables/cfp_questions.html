    CQ_ID           D_INTEGER NOT NULL,
    CQ_CODE_NAME    D_STR_100 NOT NULL,
    CQ_NAME_RU      D_STR_255 NOT NULL,
    CQ_NAME_KG      D_STR_255,
    CQ_DEFAULT_VAL  D_INTEGER DEFAULT 0 NOT NULL,
    CQ_MIN_VAL      D_INTEGER DEFAULT 0 NOT NULL,
    CQ_MAX_VAL      D_INTEGER DEFAULT 0 NOT NULL,
    CQ_IS_DELETE    D_BOOL DEFAULT 0 NOT NULL,

    CQ_DATETIME     D_DATETIME NOT NULL,
    CQ_BASIC_INCOM  D_BOOL DEFAULT 0 NOT NULL
