    EXA_DONE              D_INTEGER DEFAULT 0 NOT NULL,

    EXA_FROM_EXC_ID       D_INTEGER,

    EXA_WAIT_STATUS       D_SMALLINT,
    EXA_FROM_MA_ID        D_INTEGER,
    EXA_FROM_K_ID         D_ID,
    EXA_FROM_EDL_ID       D_ID,
    EXA_DISB_DATE         D_DATE,
    EXA_FROM_PR_ID        D_INTEGER,
    EXA_AMOUNT            D_CURRENCY,
    EXA_PERIOD            D_INTEGER,
    EXA_PERCENT           D_PERCENT,
    EXA_REPAY_DAY         D_INTEGER,
    EXA_LO_ID             D_ID,
    EXA_FROM_REG_ID       D_INTEGER,
    EXA_TRY_COUNT         D_INTEGER,
    EXA_FROM_UN_ID        D_ID,
    EXA_DATETIME          D_DATETIME,
    EXA_RESHIPPING_COUNT  D_INTEGER
);
