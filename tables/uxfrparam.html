Определение

Показать DDL (CREATE TABLE) 
CREATE TABLE UXFRPARAM (
    UFP_CODE         D_STR_100 NOT NULL,
    UFP_PARENT_CODE  D_STR_100,
    UFP_UFR_CODE     D_STR_CODE NOT NULL,
    UFP_ORIGIN_NAME  D_STR_CODE NOT NULL,
    UFP_NAME         D_STR_CODE NOT NULL,
    UFP_PK_FIELD     D_SMALLINT NOT NULL,
    UFP_IS_PARAM     D_BOOL DEFAULT 0 NOT NULL,
    UFP_INDEX        D_INTEGER NOT NULL,
    UFP_EDITOR       D_STR_CODE,
    UFP_IN_USE       D_BOOL DEFAULT 0 NOT NULL,
    UFP_REQUIRED     D_BOOL DEFAULT 0 NOT NULL,
    UFP_ACCESS_KIND  D_SMALLINT DEFAULT 0 NOT NULL,
    UFP_IS_DELETE    D_BOOL DEFAULT 0 NOT NULL,
    UFP_STATE        D_SMALLINT DEFAULT 0 NOT NULL
);

ALTER TABLE UXFRPARAM ADD CONSTRAINT PK_UXFRPARAM PRIMARY KEY (UFP_CODE);
ALTER TABLE UXFRPARAM ADD CONSTRAINT FK_UXFRPARAM_FR_CODE FOREIGN KEY (UFP_UFR_CODE) REFERENCES UXFRAME (UFR_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
ALTER TABLE UXFRPARAM ADD CONSTRAINT FK_UXFRPARAM_PARENT_CODE FOREIGN KEY (UFP_PARENT_CODE) REFERENCES UXFRPARAM (UFP_CODE) ON DELETE CASCADE ON UPDATE CASCADE;
CREATE INDEX UXFRPARAM_PARAM ON UXFRPARAM (UFP_UFR_CODE, UFP_ORIGIN_NAME);
