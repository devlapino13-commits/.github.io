CREATE PROCEDURE HP_CORRECT_SH_DATES_TEMP(
    KK_ID DECIMAL(18,0))
RETURNS (
    SH_DATE DATE,
    PR_IS_360 INTEGER,
    PR_PERCENT_FOR_DAYS INTEGER,
    K_PERCENT_PER_DAY DECIMAL(18,7),
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_DIFF_DATE INTEGER,
    SH_ID INTEGER)
AS
begin
  sh_perc = 0;
  sh_diff_date = 0;
  sh_id = 0;
  for select ok_ol_date, pr_is_360, pr_percent_for_days, k_percent_per_day, sum(ok_os)
      from operkred, kontract, product
      where ok_ol_is_active = 1 and
            ok_type = 8 and
            ok_from_k_id = :kk_id and
            ok_from_k_id = k_id and
            k_from_pr_id = pr_id and
            ok_from_c_id = 0
      group by ok_ol_date, pr_is_360, pr_percent_for_days, k_percent_per_day
      into :sh_date, :pr_is_360, :pr_percent_for_days, :k_percent_per_day, :sh_os
  do
  begin
    suspend;
  end
  sh_perc = 0;
  for select ok_sh_date, pr_is_360, pr_percent_for_days, k_percent_per_day, sum(-1 * ok_sh_os), sum(ok_sh_perc),
             sh_diff_date, sh_id
      from operkred, kontract, product, schedule
      where ok_ol_is_active = 1 and
            ok_type = 1 and
            ok_from_k_id = :kk_id and
            ok_from_k_id = k_id and
            k_from_pr_id = pr_id and
            ok_sh_id = sh_id and
            sh_from_k_id = :kk_id and
            ok_from_c_id = 0
      group by ok_sh_date, pr_is_360, pr_percent_for_days, k_percent_per_day, sh_diff_date, sh_id
      into :sh_date, :pr_is_360, :pr_percent_for_days, :k_percent_per_day, :sh_os, :sh_perc, :sh_diff_date, :sh_id
  do
  begin
    suspend;
  end
end
