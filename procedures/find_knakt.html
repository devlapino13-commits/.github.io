CREATE PROCEDURE FIND_KNAKT(
    REG_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_FOLDER_NO VARCHAR(100))
RETURNS (
    KA_ID DECIMAL(18,0),
    KA_DATE DATE,
    KA_FROM_REG_ID INTEGER,
    KA_AKT_NO VARCHAR(20),
    KA_MEMO VARCHAR(2048),
    IS_DELETE INTEGER,
    REG_NAME VARCHAR(100),
    IS_DELETE_NAME VARCHAR(20))
AS
begin
  if (rep_folder_no <> '') then
  begin
    for select distinct ka_id, ka_date, ka_from_reg_id, ka_akt_no, ka_memo, knaktlist.is_delete as is_delete, reg_name
        from knakt, knaktlist, region_list(:reg_id, 1, -1, 1), region
        where kal_from_ka_id = ka_id and
              (ka_from_reg_id = id_list or
              kal_new_reg_id = id_list) and
              ka_id <> 0 and
              kal_new_reg_id = reg_unicode and
              (ka_date between :rep_date_begin and :rep_date_end) and
              kal_new_kf_name starting with :rep_folder_no
        order by ka_date, ka_akt_no
        into :ka_id, :ka_date, :ka_from_reg_id, :ka_akt_no, :ka_memo, :is_delete, :reg_name
    do
    begin
      if (is_delete = 1) then
        is_delete_name = 'Аннулирован';
      else
        is_delete_name = '';
      suspend;
    end
  end
  else
  begin
    for select distinct ka_id, ka_date, ka_from_reg_id, ka_akt_no, ka_memo, knaktlist.is_delete as is_delete, reg_name
        from knakt, knaktlist, region_list(:reg_id, 1, -1, 1), region
        where kal_from_ka_id = ka_id and
              (ka_from_reg_id = id_list or
              kal_new_reg_id = id_list) and
              ka_id <> 0 and
              kal_new_reg_id = reg_unicode and
              (ka_date between :rep_date_begin and :rep_date_end)
        order by ka_date, ka_akt_no
        into :ka_id, :ka_date, :ka_from_reg_id, :ka_akt_no, :ka_memo, :is_delete, :reg_name
    do
    begin
      if (is_delete = 1) then
        is_delete_name = 'Аннулирован';
      else
        is_delete_name = '';
      suspend;
    end
  end
end
