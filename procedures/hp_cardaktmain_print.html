CREATE PROCEDURE HP_CARDAKTMAIN_PRINT(
    CAM_ID D_INTEGER)
RETURNS (
    CAM_DATE D_DATE,
    CAM_SIGN_DATE D_DATE,
    CAM_CARD_LIST D_STR_8100,
    CAM_CARD_COUNT D_INTEGER,
    CARD_RANGE D_STR_8100,
    PIN_COUNT D_INTEGER,
    PIN_RANGE D_STR_8100,
    IS_PINCOVER D_BOOL,
    REGS_NAME D_STR_CODE,
    CS_ID D_ID,
    CS_NAME_FIO D_STR_100,
    CS_LOGIN D_STR_CODE,
    CS_INN D_STR_20,
    CS_PASS_NO D_STR_CODE,
    CS_PASS_DATE D_DATE,
    CS_PASS_WHO D_STR_255,
    CS_ADDR D_STR_255,
    REGD_NAME D_STR_CODE,
    CD_ID D_ID,
    CD_NAME_FIO D_STR_100,
    CD_LOGIN D_STR_CODE,
    CD_INN D_STR_20,
    CD_PASS_NO D_STR_CODE,
    CD_PASS_DATE D_DATE,
    CD_PASS_WHO D_STR_255,
    CD_ADDR D_STR_255,
    CD_PIN_ID D_ID,
    CD_PIN_NAME_FIO D_STR_100,
    CDP_LOGIN D_STR_CODE,
    CD_PIN_INN D_STR_20,
    CD_PIN_PASS_NO D_STR_CODE,
    CD_PIN_PASS_DATE D_DATE,
    CD_PIN_PASS_WHO D_STR_255,
    CD_PIN_ADDR D_STR_255,
    COMPANY_NAME D_STR_255,
    COMPANYF_NAME D_STR_255,
    CAM_STATUS D_SMALLINT)
AS
begin
  company_name = f_company_name('short', 'rus');
  companyf_name = f_company_name('full', 'rus');

  for
    select
      cam_date,
      cam_ol_datetime,
      cam_card_list,
      cam_card_count,
      cam_status,
      regs_name,
      cs_id,
      cs_name_fio,
      regd_name,
      cd_id,
      cd_name_fio,
      cd_pin_id,
      cd_pin_name_fio
    from hp_cardaktmain_view(:cam_id)
  into
    :cam_date,
    :cam_sign_date,
    :cam_card_list,
    :cam_card_count,
    :cam_status,
    :regs_name,
    :cs_id,
    :cs_name_fio,
    :regd_name,
    :cd_id,
    :cd_name_fio,
    :cd_pin_id,
    :cd_pin_name_fio
  do
  begin
    -- от кого
    select
      c_inn,
      c_pass_no,
      c_pass_date,
      c_pass_who
    from clients
    where c_id = :cs_id
    into
      :cs_inn,
      :cs_pass_no,
      :cs_pass_date,
      :cs_pass_who;
    cs_login = (
      select first 1
        un_name
      from username
      where un_from_c_id = :cs_id);
    cs_addr = (
      select
        adl_address_str
      from addrclient
        inner join addrlist
          on adl_id = adc_from_adl_id
      where adc_from_c_id = :cs_id
        and adc_address_type = 1);

    -- кому карты
    select
      c_inn,
      c_pass_no,
      c_pass_date,
      c_pass_who
    from clients
    where c_id = :cd_id
    into
      :cd_inn,
      :cd_pass_no,
      :cd_pass_date,
      :cd_pass_who;
    if (cam_status = 1) then
      cd_login = (
        select first 1
          un_name
        from username
        where un_from_c_id = :cd_id);
    cd_addr = (
      select
        adl_address_str
      from addrclient
        inner join addrlist
          on adl_id = adc_from_adl_id
      where adc_from_c_id = :cd_id
        and adc_address_type = 1);

    -- кому ПИН-конверты
    if (cd_pin_id > 0) then
    begin
      select
        c_inn,
        c_pass_no,
        c_pass_date,
        c_pass_who
      from clients
      where c_id = :cd_pin_id
      into
        :cd_pin_inn,
        :cd_pin_pass_no,
        :cd_pin_pass_date,
        :cd_pin_pass_who;
      if (cam_status = 1) then
        cdp_login = (
          select first 1
            un_name
          from username
          where un_from_c_id = :cd_pin_id);
      cd_pin_addr = (
        select
          adl_address_str
        from addrclient
          inner join addrlist
            on adl_id = adc_from_adl_id
        where adc_from_c_id = :cd_pin_id
          and adc_address_type = 1);
    end

    select
      list(iif(first_card_id = last_card_id, first_card_id, first_card_id || '-' || last_card_id), ', ')
    from hp_cardrange_jsontolist(:cam_card_list)
    into
      :card_range;

    pin_count = 0;
    select
      list(iif(first_card_id = last_card_id, first_card_id, first_card_id || '-' || last_card_id), ', '),
      sum(card_count)
    from hp_cardrange_jsontolist(:cam_card_list)
    where is_pincover = 1
    into
      :pin_range,
      :pin_count;
    is_pincover = iif(pin_count > 0, 1, 0);

    suspend;
  end
end
