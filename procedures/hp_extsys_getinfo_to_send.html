CREATE PROCEDURE HP_EXTSYS_GETINFO_TO_SEND(
    ESM_ID D_ID)
RETURNS (
    C_INN D_STR_20,
    C_PHONE D_STR_100,
    C_NAME1 D_STR_100,
    C_NAME2 D_STR_100,
    C_NAME3 D_STR_100,
    PERMIT_DATE_END D_DATE,
    C_BIRTH_DATE D_DATE,
    C_ADDRESS_REG D_STR_255,
    C_ADDRESS_FACT D_STR_255,
    C_PASS_NO D_STR_CODE,
    C_PASS_WHO D_STR_255,
    C_BLOB_PASS1 D_BLOB_BINARY,
    C_BLOB_PASS2 D_BLOB_BINARY,
    C_BLOB_PHOTO D_BLOB_BINARY,
    C_PASS_DATE D_DATE,
    ESM_FULL_SOURCE D_BLOB_TEXT)
AS
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable esm_action d_str_20;
declare variable esm_date d_date;
declare variable c_blob_pass1_id d_id;
declare variable c_blob_pass2_id d_id;
declare variable c_blob_photo_id d_id;
begin
  -- возвращает данные в рамках автоматизации получения справки о разаработной плате для клиентов-бюджетников (RFCKG-3163)

  select esm_object_id,
         esm_date,
         esm_full_source,
         esm_action
  from extsysmain
  where esm_id = :esm_id
  into c_id,
       esm_date,
       esm_full_source,
       esm_action;

  if (coalesce(esm_action, '') <> 'send_code') then
  begin
    select c_inn,
           c_name_fio,
           f_cont_info('clients', :c_id, 'cell_main', 'ok'),
           c_birth_date,
           c_pass_no,
           c_pass_who,
           c_blob_pass1_id,
           c_blob_pass2_id,
           c_blob_photo_id,
           c_pass_date
    from clients
    where c_id = :c_id
    into c_inn,
         c_name_fio,
         c_phone,
         c_birth_date,
         c_pass_no,
         c_pass_who,
         c_blob_pass1_id,
         c_blob_pass2_id,
         c_blob_photo_id,
         c_pass_date;

    select txt1,
           txt2,
           txt3
    from hp_fio_clear(:c_name_fio)
    into c_name1,
         c_name2,
         c_name3;

    select res_blob
    from fm_blob_get(:c_blob_pass1_id)
    into c_blob_pass1;

    select res_blob
    from fm_blob_get(:c_blob_pass2_id)
    into c_blob_pass2;

    select res_blob
    from fm_blob_get(:c_blob_photo_id)
    into c_blob_photo;

    select adl_address_str
    from addrclient
    inner join addrlist on adl_id = adc_from_adl_id
    where adc_from_c_id = :c_id and
          adc_address_type = 1
    into c_address_reg;

    select adl_address_str
    from addrclient
    inner join addrlist on adl_id = adc_from_adl_id
    where adc_from_c_id = :c_id and
          adc_address_type = 2
    into c_address_fact;
    permit_date_end = dateadd(month, 6, :esm_date); -- дата окончания разрешения
  end

  suspend;
end
