CREATE PROCEDURE HP_ANTIFROD_GET_CALC(
    K_ID D_ID)
RETURNS (
    JSO D_INTEGER)
AS
declare variable criteria_code d_str_code;
declare variable acr_code d_str_code;
declare variable antifrod_res d_bool;
declare variable percent d_percent;
declare variable result d_bool;
declare variable afc_id d_id;
declare variable acr_id d_id;
declare variable res_text d_str_255;
declare variable jso_cat d_integer;
declare variable jso_cri d_integer;
declare variable jso_data d_integer;
declare variable jso_data_tmp d_integer;
declare variable jsa_criteria_arr d_integer;
declare variable jsa_cat_arr d_integer;
declare variable jsa_cri_arr d_integer;
declare variable res_add_jso d_integer;
declare variable criteria d_str_255;
declare variable critical_level d_smallint;
declare variable calc_datetime d_datetime;
begin
  jso = json.obj_create();
  jsa_criteria_arr = json.arr_create(jso, 'criteria_arr');
  jso_data = json.obj_create(jso, 'criteria_data');
  jsa_cat_arr = json.arr_create(jso, 'cat_arr');

  for select afc_id
  from antifrod_categories
  into afc_id
  do
  begin
    jso_cat = json.obj_create(jso, afc_id);
    json.arr_set_s(jsa_cat_arr, -1, afc_id);
    jsa_cri_arr = json.arr_create(jso_cat, 'cri_arr');

    for select acr_id, acr_percent, acr_code
    from antifrod_criteria
    where acr_from_afc_id = :afc_id
    into acr_id, percent, acr_code
    do
    begin
      json.arr_set_s(jsa_cri_arr, -1, acr_id);
      jso_cri = json.obj_create(jso_cat, acr_id);
      json.obj_set_s(jso_cri, 'antifrod_res', iif(acr_code containing '#reverse#', 0, 1));
      json.obj_set_s(jso_cri, 'criteria_res', iif(acr_code containing '#reverse#', 0, percent));
    end
  end

  for select criteria, criteria_code, critical_level, antifrod_res, res_text, res_add_jso, calc_datetime
  from hp_get_antifrod_data(:k_id)
  into criteria, criteria_code, critical_level, antifrod_res, res_text, res_add_jso, calc_datetime
  do
  begin
    if (criteria_code is null) then
      exception e_message res_text;

    if (json.contains(jso, 'calc_datetime') = 0) then
      json.obj_set_s(jso, 'calc_datetime', f_datetimetostr(calc_datetime, 'dd.mm.yyyy hh:nn:ss'));

    json.arr_set_s(jsa_criteria_arr, -1, criteria_code);
    jso_data_tmp = json.obj_create(jso_data, criteria_code);
    json.obj_set_s(jso_data_tmp, 'criteria', criteria);
    json.obj_set_s(jso_data_tmp, 'critical_level', critical_level);
    json.obj_set_s(jso_data_tmp, 'res_text', res_text);

    if (res_add_jso is not null) then
      json.obj_set_obj(jso_data_tmp, 'add', res_add_jso);

    for select afc_id, acr_id, acr_percent, acr_code
    from antifrod_criteria, antifrod_categories
    where afc_id = acr_from_afc_id
      and acr_code starting with :criteria_code || '#'
    into afc_id, acr_id, percent, acr_code
    do
    begin
      jso_cat = json.obj_get_obj(jso, afc_id);
      jso_cri = json.obj_get_obj(jso_cat, acr_id);

      if (acr_code containing '#reverse#') then
        result = 1 - antifrod_res;
      else
        result = antifrod_res;

      percent = percent * result;

      json.obj_set_s(jso_cri, 'criteria_code', criteria_code);
      json.obj_set_s(jso_cri, 'antifrod_res', antifrod_res);
      json.obj_set_s(jso_cri, 'criteria_res', percent);

      if (:acr_code containing '#critical#' and result > 0) then
        json.obj_set_s(jso_cat, 'is_critical', 1);
    end
  end

  execute procedure hp_antifrod_calc_total(jso);

  suspend;
end
