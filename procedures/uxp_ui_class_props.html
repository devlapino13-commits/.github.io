CREATE PROCEDURE UXP$UI_CLASS_PROPS(
    PROC_NAME D_STR_CODE)
RETURNS (
    CLASS_STR VARCHAR(32000) CHARACTER SET WIN1251,
    CREATE_STR VARCHAR(32000) CHARACTER SET WIN1251)
AS
declare variable field_name d_str_code;
declare variable prop_name d_str_code;
declare variable prop_type d_str_code;
declare variable char13 d_str_sym;
declare variable as_prop_type d_str_code;
declare variable public_str varchar(32000) character set win1251;
begin
  char13 = ascii_char(13);
  class_str = 'private' || char13;
  public_str = 'public' || char13;
  create_str = 'Query.Open;' || char13;

  for select field_name, prop_name, prop_type
  from uxp$ui_class_props_tmp(:proc_name)
  into :field_name, :prop_name, :prop_type
  do
  begin
    as_prop_type = prop_type;
    if (as_prop_type = 'Boolean') then
      as_prop_type = 'Integer = 1';
    else
    if (as_prop_type = 'TGUID') then
      as_prop_type = 'Guid';

    class_str = class_str || 'F' || prop_name || ' : '  || prop_type || ';' || char13;
    public_str = public_str || 'property ' || prop_name || ' : '  || prop_type || ' read ' || 'F' || prop_name || ' write F' || prop_name || ';' || char13;
    create_str = create_str || prop_name || ' := Query.FieldByName(''' || field_name || ''').as' || as_prop_type || ';' || char13;
  end

  class_str = class_str || char13 || public_str || ' end;';
  create_str = create_str || 'Query.Close;' || char13;

  suspend;
end
