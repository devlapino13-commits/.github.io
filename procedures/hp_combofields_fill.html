CREATE PROCEDURE HP_COMBOFIELDS_FILL(
    CF_ID INTEGER,
    CF_TABLE VARCHAR(100),
    CF_FIELD VARCHAR(100),
    CF_SP_PARAM VARCHAR(255),
    CF_MASTER_FIELD VARCHAR(100) = '')
AS
declare variable sql_text varchar(1024);
begin
  delete from combolist
  where cl_kind = :cf_id;
  if (cf_master_field is null) then
    cf_master_field = '';

  if (exists(select *
             from combofields
             where cf_kind = :cf_id and
                   cf_is_ready = 1)) then
  begin
    if (cf_table = 'matlist' and cf_field = 'ml_model') then
    begin
      sql_text = 'insert into combolist (cl_kind, cl_text) ' || 'select ' || :cf_id || ', ml_model from matlist group by 1, 2 order by count(*) desc rows 100';
    end else
    if (cf_table = 'clients' and cf_field = 'c_birth_city') then
    begin
      -- Для места рождения берем только часто встречающиеся
      sql_text =
        'insert into combolist (cl_kind, cl_text, cl_category) ' ||
        'select '|| :cf_id ||',  cast(trim(replace(replace(c_birth_city, ''  '', ''''), x''c2a0'', '' '')) as d_str_100), c_birth_country from clients where coalesce(' || cf_field ||
        ', '''') <> '''' and coalesce(' || cf_master_field || ', '''') <> '''' group by 1, 2, 3 having count(*) > 10';
    end else
    begin
        if (cf_master_field = '') then
          sql_text = 'insert into combolist (cl_kind, cl_text) ' || 'select distinct ' || :cf_id || ', trim(' || :cf_field || ') ' || 'from ' || :cf_table || ' ' || 'where coalesce(trim(' || :cf_field || '),'''')<>''''';
        else
          sql_text = 'insert into combolist (cl_kind, cl_text, cl_category) ' || 'select distinct ' || :cf_id || ', trim(' || :cf_field || '), trim(' || :cf_master_field || ') ' || 'from ' || :cf_table || ' ' || 'where coalesce(trim(' || :cf_field || '),'''')<>'''' and coalesce(trim(' || :cf_master_field || '), '''')<>''''';
    end
    execute statement :sql_text;
  end
end
