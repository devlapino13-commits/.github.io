CREATE PROCEDURE HP_INSTALLMENT_GOODS_ACC(
    REG_ID D_INTEGER)
RETURNS (
    AL_ID D_STR_20,
    AL_ACCOUNT VARCHAR(32),
    AL_NAME D_STR_255)
AS
begin
  for select al.al_id, al.al_account, al.al_name
      from acc_list al
      where al.al_from_bg_id = '21199' and
            al.al_from_reg_id = :reg_id
      into :al_id, :al_account, :al_name
  do
  begin
    suspend;
  end
end
