CREATE PROCEDURE HP_EXPORT_1C_OPERBOOK(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    OPER_DATE D_DATE,
    ACC_DT D_STR_20,
    ACC_KT D_STR_20,
    SUMMA_EKV D_CURRENCY,
    SUMMA_VAL D_CURRENCY,
    OPER_MEMO D_STR_LONG,
    ACC_DT_NAME D_STR_255,
    ACC_KT_NAME D_STR_255,
    ACC_DT_REG D_INTEGER,
    ACC_KT_REG D_INTEGER)
AS
declare variable PTEXT D_STR_100;
declare variable PLEN D_INTEGER;
declare variable P D_INTEGER;
declare variable DB_ID D_INTEGER;
declare variable ACC_DT_4 D_STR_10;
declare variable ACC_DT_7 D_STR_10;
declare variable ACC_KT_4 D_STR_10;
declare variable ACC_KT_7 D_STR_10;
begin
  db_id = f_get_db_id();

  ptext = 'договора займа';
  plen = char_length(ptext);

  for select ob_ol_date, ob_account_dt, ob_account_kt, sum(ob_summa), sum(ob_summa_val)
  from operbook
  where ob_ol_date between :rep_date_begin and :rep_date_end and
        ob_ol_is_active = 1
  group by 1, 2, 3
  into oper_date, acc_dt, acc_kt, summa_ekv, summa_val
  do
  begin
    acc_dt_name = '';
    acc_kt_name = '';
    acc_dt_reg = null;
    acc_kt_reg = null;
    oper_memo = '';

    select al_name, al_from_reg_id
    from acc_list
    where al_id = :acc_dt
    into acc_dt_name, acc_dt_reg;

    select al_name, al_from_reg_id
    from acc_list
    where al_id = :acc_kt
    into acc_kt_name, acc_kt_reg;

    select ob_detail
    from operbook
    where ob_ol_date between :rep_date_begin and :rep_date_end and
          ob_ol_is_active = 1 and
          ob_account_dt = :acc_dt and
          ob_account_kt = :acc_kt and
          coalesce(ob_detail, '') <> ''
    rows 1
    into oper_memo;

    if (coalesce(oper_memo, '') <> '') then
    begin
      p = position(ptext, oper_memo);
      if (p > 0) then
        oper_memo = trim(left(oper_memo, p + plen));
    end

    if (db_id = 39801) then
    begin
      if (exists(select 1 from export_exceptions_1c ee
                 where ee.ee_oper_memo = trim(:oper_memo))) then
         continue;

      acc_dt_4 = substring(:acc_dt from 8 for 4);
      acc_dt_7 = substring(:acc_dt from 8 for 7);
      acc_kt_4 = substring(:acc_kt from 8 for 4);
      acc_kt_7 = substring(:acc_kt from 8 for 7);

      if (exists(select 1 from export_exceptions_1c ee
                 where (:acc_dt_7 = ee.ee_dt and :acc_kt_4 = ee.ee_kt)
                    or (:acc_dt_7 = ee.ee_dt and :acc_kt_7 = ee.ee_kt)
                    or (:acc_dt_4 = ee.ee_dt and :acc_kt_4 = ee.ee_kt)
                    or (:acc_dt_4 = ee.ee_dt and :acc_kt_7 = ee.ee_kt))) then
         continue;
    end
    suspend;
  end
end
