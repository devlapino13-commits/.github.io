CREATE PROCEDURE HP_ATTACH_FILE_TO_SIGN_DOC(
    CCM_ID D_INTEGER,
    UN_ID D_ID,
    COMMAND D_STR_CODE = 'attach')
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable CCM_ATTACH_FILE_COUNT D_INTEGER;
declare variable CONST_ATTACH_TRY D_INTEGER;
declare variable action_str d_str_100;
begin
  -- RFCKG-4550
  const_attach_try = 1; -- Максимальное кол-во прикреплений файла после подписания документа
  res_text = '';

  action_str = case :command
                  when 'attach' then 'прикреплении'
                  when 'disattach' then 'откреплении'
                  else ''
               end;

  if (not exists(select ccm_attach_file_count
                 from cashcheckmain ccm
                 where ccm.ccm_id = :ccm_id)) then
    exception e_default(f_format('Ошибка при %3 файла — документ не найден в cashcheckmain' || ascii_char(13) || ascii_char(10)|| 'ccm_id = %1' || ascii_char(13) || ascii_char(10)|| 'un_id = %1',
                                 :ccm_id,
                                 :un_id,
                                 :action_str));

  select ccm_attach_file_count
  from cashcheckmain ccm
  where ccm.ccm_id = :ccm_id
  into ccm_attach_file_count;

  if (ccm_attach_file_count is null) then
    res_text = 'Количество прикреплений после подписания документа неизвестно';


  if (ccm_attach_file_count >= const_attach_try and action_str = 'attach') then
    res_text = f_format('Максимальное кол-во прикреплений файла после подписания документа %1,' || ascii_char(13) || ascii_char(10)|| ' количество прикреплений после подписания документа %2', const_attach_try, ccm_attach_file_count);

  if (res_text = '') then
  begin
    if (command = 'attach') then
    begin
      update cashcheckmain ccm
         set ccm.ccm_attach_file_count = :ccm_attach_file_count + 1
      where ccm.ccm_id = :ccm_id;
    end
    if (command = 'disattach' and ccm_attach_file_count > 0) then
    begin
      update cashcheckmain ccm
         set ccm.ccm_attach_file_count = :ccm_attach_file_count - 1
      where ccm.ccm_id = :ccm_id;

      res_text = 'Не удалось прикрепить файл';
    end
  end

  suspend;
end
