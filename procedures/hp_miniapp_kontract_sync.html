CREATE PROCEDURE HP_MINIAPP_KONTRACT_SYNC(
    SRC_OBJ_KIND INTEGER,
    SRC_OBJ_ID DECIMAL(18,0))
AS
declare variable dst_obj_id decimal(18,0);
declare variable disb_date date;
declare variable kf_id decimal(18,0);
declare variable kolich_repay integer;
declare variable percent decimal(18,7);
declare variable kolich_client integer;
declare variable amount decimal(18,4) = 0;
declare variable kommis_perc decimal(18,7);
declare variable c_id decimal(18,0);
declare variable loan_target integer;
declare variable is_cashier integer;
declare variable business_subtype integer;
declare variable kommis_lgota decimal(18,4) = 0;
declare variable is_poruch integer;
declare variable pr_id integer;
declare variable repay_day integer;
declare variable un_id decimal(18,0);
declare variable ma_number integer;
declare variable last_app_date date;
declare variable last_un_id decimal(18,0);
declare variable kc_id decimal(18,0);
declare variable cc_id integer;
declare variable cc_date date;
declare variable cc_checked_date date;
declare variable cc_check_result integer;
declare variable calc_date date;
declare variable refund integer;
declare variable add_info varchar(255);
declare variable ma_prepayment_summa curr_type;
declare variable add_month d_smallint;
declare variable disb_kind d_integer;
declare variable is_online d_smallint;
begin
  if (src_obj_kind = 9) then
  begin
    select k_from_ma_app_id
    from kontract
    where k_id = :src_obj_id
    into dst_obj_id;
    if (dst_obj_id is null) then
      exit;

    execute procedure hp_check_miniapp_kontract_diff(:src_obj_id, :dst_obj_id, 'k->m');

    select k_disb_date, k_from_kf_id, k_kolich_repay, k_percent,
      k_kolich_client_approved, k_amount_approved, k_kommis_perc, k_from_pr_id,
      k_repay_day, k_user_id, coalesce(k_refund, 0), k_add_month,
      coalesce(k_add_info, ''), k_disb_kind, k_is_online
    from kontract
    where k_id = :src_obj_id
    into disb_date, kf_id, kolich_repay, percent, kolich_client, amount,
      kommis_perc, :pr_id, repay_day, :un_id, refund, add_month, add_info,
      disb_kind, is_online;

    -- Kontract -> Miniapp
    select ma_number, ma_date_app, ma_from_un_id
    from miniapp
    where ma_id = :dst_obj_id and
          ma_date_end >= '31.12.2999'
    into :ma_number, last_app_date, last_un_id;

    if (ma_number is not null) then
    begin
      if (exists(select * from miniapp
          where
            ma_id = :dst_obj_id and
            ma_number = :ma_number and
            (ma_date_oper is distinct from :disb_date or
            ma_from_kf_id is distinct from :kf_id or
            ma_kolich_repay is distinct from :kolich_repay or
            ma_percent is distinct from :percent or
            ma_kolich_client is distinct from :kolich_client or
            ma_summa_minus is distinct from :amount or
            ma_repay_day is distinct from :repay_day or
            ma_from_pr_id is distinct from :pr_id or
            ma_kommis_perc is distinct from :kommis_perc or
            ma_refund is distinct from :refund or
            ma_add_month is distinct from :add_month or
            ma_add_info is distinct from :add_info or
            ma_disb_kind is distinct from :disb_kind or
            ma_is_online is distinct from :is_online
            ))) then
      begin
        if (last_app_date <> current_date or last_un_id <> un_id) then
        begin

          repay_day = (select ma_repay_day from miniapp where ma_id = :dst_obj_id and ma_number = :ma_number);

          insert into miniapp (
            ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id,
            ma_kolich_repay, ma_percent, ma_kolich_client, ma_memo, ma_is_active, ma_summa_minus,
            ma_from_un_id, ma_office_id, ma_currency_id, ma_no, ma_percent_category, ma_manager_id,
            ma_plan_disb_time, ma_expert_id, ma_repay_day, ma_source_system, ma_kommis_perc,
            ma_refund, ma_add_month, ma_add_info, ma_disb_kind, ma_is_online)
          select
            ma_id, -1, current_date, :disb_date, :kf_id, :pr_id, :kolich_repay,
            :percent, :kolich_client, ma_memo, 1, :amount, :un_id, ma_office_id,
            ma_currency_id, ma_no, :percent, ma_manager_id, ma_plan_disb_time,
            ma_expert_id, :repay_day, ma_source_system, :kommis_perc, :refund,
            :add_month, :add_info, :disb_kind, :is_online
          from miniapp
          where ma_id = :dst_obj_id and
                ma_number = :ma_number;
        end
        else
        begin
          update miniapp
          set ma_date_oper = :disb_date,
              ma_from_kf_id = :kf_id,
              ma_kolich_repay = :kolich_repay,
              ma_percent = :percent,
              ma_kolich_client = :kolich_client,
              ma_summa_minus = :amount,
              ma_from_pr_id = :pr_id,
              ma_kommis_perc = :kommis_perc,
              ma_refund = :refund,
              ma_add_month = :add_month, 
              ma_add_info = :add_info,
              ma_disb_kind = :disb_kind,
              ma_is_online = :is_online
          where ma_id = :dst_obj_id and
                ma_number = :ma_number;
        end
      end
    end

    for select kc_from_c_id, kc_amount_approved, kc_loan_target,
      iif(kc_status = 2, 1, 0), kc_busuness_subtype, kc_kommis_lgota, 0
    from knclient
    where kc_from_k_id = :src_obj_id
    union
    select z_from_c_id, 0.0, 0, 0, 0, 0, 1
    from zalog
    where z_type = 0 and
          z_from_k_id = :src_obj_id
    into :c_id, :amount, loan_target, is_cashier, :business_subtype, kommis_lgota, is_poruch
    do
    begin
      update or insert into miniappclient (
        mac_from_ma_id, mac_from_c_id, mac_amount, mac_is_poruch, mac_loan_target,
        mac_is_cashier, mac_business_subtype, mac_kommis_lgota)
      values (:dst_obj_id, :c_id, :amount, :is_poruch, :loan_target, :is_cashier,
        :business_subtype, :kommis_lgota)
      matching (mac_from_ma_id, mac_from_c_id);
    end

    for select mac_from_c_id
        from miniappclient
        where mac_from_ma_id = :dst_obj_id and
        mac_is_poruch <> -1  --лицо дающее согласие указывается на уровне заявки и не редактируется на уровне контракта(после нажатия кнопки сохранить удалялся лицо дающее согласие по истиссте в кг (при добавлении работы))
        into :c_id
    do
    begin
      if (not(exists(select *
                     from knclient
                     where kc_from_k_id = :src_obj_id and
                           kc_from_c_id = :c_id))) then
      begin
        if (not(exists(select *
                       from zalog
                       where z_from_k_id = :src_obj_id and
                             z_from_c_id = :c_id and
                             z_type = 0))) then
        begin
          delete from miniappclient
          where mac_from_ma_id = :dst_obj_id and
                mac_from_c_id = :c_id;
        end
      end
    end

    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :dst_obj_id and
              kc_amount_approved > 0
        into :c_id
    do
      execute procedure hp_clientstate_set(:c_id);

    update zalogauto
    set za_from_ma_id = :dst_obj_id
    where za_from_k_id = :src_obj_id;

    exit;
  end

  if (src_obj_kind = 11) then
  begin
    -- Если заявка не параллельный займ, только тогда сбрасываем контакнных лиц
    if (not exists(select *
    from miniapp ma, product
    where ma_from_pr_id = pr_id and
          pr_code containing '.par.' and
          ma_id = :src_obj_id and
          ma_date_end >= '31.12.2999')) then
    begin
      for select mac_from_c_id
          from miniappclient
          where mac_from_ma_id = :src_obj_id
          into :c_id
      do
      begin
        for select cont.cc_id, cont.cc_date, cont.cc_checked_date, cont.cc_check_result
            from contacts cont
            where cont.cc_from_c_id = :c_id and
                  cont.cc_is_old_contact = 0
            into :cc_id, :cc_date, :cc_checked_date, :cc_check_result
        do
        begin
          calc_date = cc_date;
          -- если контакт не обзвонен, за рассчетную дату беру дату создания контакта в БД.
          if ((cc_checked_date is not null) and
              (cc_check_result > 0)) then
            calc_date = cc_checked_date;
  
          if (datediff(day, :calc_date, current_date) > 30) then -- если разность между расчетной датой и текущей  превышает 30 дней
          begin
            update contacts cont
            set cont.cc_is_old_contact = 1
            where cont.cc_id = :cc_id;
          end
        end
      end
    end

    select k_id
    from kontract
    where k_from_ma_app_id = :src_obj_id
    order by k_disb_date desc
    rows 1
    into dst_obj_id;
    if (dst_obj_id is null) then
      exit;

    execute procedure hp_check_miniapp_kontract_diff(:dst_obj_id, :src_obj_id, 'm->k');

    -- Miniapp -> Kontract

    select ma_date_oper, ma_from_kf_id, ma_kolich_repay, ma_percent,
      ma_kolich_client, ma_summa_minus, ma_kommis_perc, ma_from_pr_id,
      ma_repay_day, coalesce(ma_refund, 0), ma_add_month, ma_add_info,
      ma_prepayment_summa, ma_disb_kind, ma_is_online
    from miniapp ma
    where ma_id = :src_obj_id and
          ma_date_end >= '31.12.2999'
    into disb_date, kf_id, kolich_repay, percent, kolich_client, amount,
      kommis_perc, pr_id, repay_day, refund, add_month, add_info,
      ma_prepayment_summa, disb_kind, is_online;

    update kontract
    set k_disb_date = :disb_date,
        k_kontract_date = f_get_kontract_date(:dst_obj_id, :disb_date),
        k_from_kf_id = :kf_id,
        k_kolich_repay = :kolich_repay,
        k_percent = :percent,
        k_kolich_client_approved = :kolich_client,
        k_kolich_client_applicate = :kolich_client,
        k_amount_approved = :amount,
        k_amount_applicate = :amount,
        k_kommis_perc = :kommis_perc,
        k_from_pr_id = :pr_id,
        k_repay_day = :repay_day,
        k_refund = :refund,
        k_add_month = :add_month,
        k_add_info = :add_info,
        k_prepayment_summa = :ma_prepayment_summa,
        k_disb_kind = :disb_kind,
        k_is_online = :is_online
    where k_id = :dst_obj_id and
          (k_disb_date is distinct from :disb_date or
          k_kontract_date is distinct from f_get_kontract_date(:dst_obj_id, :disb_date) or
          k_from_kf_id is distinct from :kf_id or
          k_kolich_repay is distinct from :kolich_repay or
          k_percent is distinct from :percent or
          k_kolich_client_approved is distinct from :kolich_client or
          k_amount_approved is distinct from :amount or
          k_amount_applicate is distinct from :amount or
          k_kommis_perc is distinct from :kommis_perc or
          k_refund is distinct from :refund or
          k_add_month is distinct from :add_month or
          k_add_info is distinct from :add_info or
          k_prepayment_summa is distinct from :ma_prepayment_summa or
          k_disb_kind is distinct from :disb_kind or
          k_is_online is distinct from :is_online or
          k_from_pr_id is distinct from :pr_id);

    for select mac_from_c_id, mac_amount, mac_loan_target, mac_is_cashier,
      mac_business_subtype, mac_kommis_lgota, mac_is_poruch
    from miniappclient
    where mac_from_ma_id = :src_obj_id
    into c_id, amount, loan_target, is_cashier, business_subtype, kommis_lgota,
      is_poruch
    do
    begin
      if (is_poruch in (-1, -2)) then
        continue;
      if (kolich_client = 1) then
        is_cashier = 1;
      if (is_poruch = 0) then
      begin
        kc_id = null;
        select kc_id
        from knclient
        where kc_from_k_id = :dst_obj_id and
              kc_from_c_id = :c_id and
              kc_amount_approved > 0 rows 1
        into :kc_id;

        if (kc_id is not null) then
        begin
          update knclient
          set kc_amount_approved = :amount,
              kc_loan_target = :loan_target,
              kc_status = iif(:is_cashier = 1, 2, 0),
              kc_busuness_subtype = :business_subtype,
              kc_kommis_lgota = :kommis_lgota
          where kc_id = :kc_id;
        end
        else
        begin
          -- Сохранение
          update or insert into knclient (
            kc_from_k_id, kc_from_c_id, kc_amount_approved, kc_amount_applicate,
            kc_loan_target, kc_status, kc_busuness_subtype, kc_kommis_lgota)
          values (:dst_obj_id, :c_id, :amount, :amount, :loan_target,
            iif(:is_cashier = 1, 2, 0), :business_subtype, :kommis_lgota)
          matching (kc_from_k_id, kc_from_c_id);
        end
      end
      else
      begin
        update or insert into zalog (z_from_k_id, z_from_c_id, z_type)
        values (:dst_obj_id, :c_id, 0)
        matching (z_from_k_id, z_from_c_id, z_type);
      end
    end

    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :dst_obj_id
        union
        select z_from_c_id
        from zalog
        where z_from_k_id = :dst_obj_id and
              z_type = 0
        into :c_id
    do
    begin
      if (not(exists(select *
                     from miniappclient
                     where mac_from_ma_id = :src_obj_id and
                           mac_from_c_id = :c_id))) then
      begin
        delete from knclient
        where kc_from_k_id = :dst_obj_id and
              kc_from_c_id = :c_id;
      end
    end
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :dst_obj_id and
              kc_amount_approved > 0
        into :c_id
    do
      execute procedure hp_clientstate_set(:c_id);

    update zalogauto
    set za_from_k_id = :dst_obj_id
    where za_from_ma_id = :src_obj_id;

    exit;
  end
end
