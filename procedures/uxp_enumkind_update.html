CREATE PROCEDURE UXP$ENUMKIND_UPDATE(
    OLD_EK_CODE D_STR_CODE,
    NEW_EK_CODE D_STR_CODE,
    NEW_EK_NAME D_STR_CODE,
    NEW_ERL_LOOKUP D_SMALLINT)
RETURNS (
    EK_CODE D_STR_CODE,
    EK_NAME D_STR_CODE,
    ERL_LOOKUP D_SMALLINT,
    ERL_UUID_CODE D_STR_CODE)
AS
declare variable need_insert d_bool;
begin
  if (new_erl_lookup is null) then
    new_erl_lookup = 0;

  need_insert = 1;
  if (coalesce(old_ek_code, '') <> '') then
  begin
    update enum_kind
    set ek_code = :new_ek_code,
        ek_name = :new_ek_name
    where ek_code = :old_ek_code
    returning ek_code, ek_name
    into ek_code, ek_name;

    if (row_count = 1) then
      need_insert = 0;
  end

  if (need_insert = 1) then
  begin
    insert into enum_kind (ek_code, ek_name)
    values (:new_ek_code, :new_ek_name)
    returning ek_code, ek_name
    into ek_code, ek_name;
  end

  update ux$ereflist
  set erl_lookup = :new_erl_lookup
  where erl_code = :new_ek_code
  returning uuid_to_char(erl_uuid), erl_lookup
  into erl_uuid_code, erl_lookup;

  suspend;
end
