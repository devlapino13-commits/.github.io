CREATE PROCEDURE AAREP_MINIAPP_ONE(
    REP_REGION_ID INTEGER,
    REP_MA_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    MA_FROM_REG_ID INTEGER,
    MA_ID INTEGER,
    MA_NUMBER INTEGER,
    MA_DATE_APP DATE,
    MA_DATE_OPER DATE,
    MA_FROM_KF_ID DECIMAL(18,0),
    MA_FROM_PR_ID INTEGER,
    MA_KOLICH_REPAY INTEGER,
    MA_PERCENT DECIMAL(18,7),
    MA_KOLICH_CLIENT INTEGER,
    MA_MEMO VARCHAR(255),
    MA_IS_ACTIVE INTEGER,
    MA_IS_AUTO INTEGER,
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_DATE_SAVE TIMESTAMP,
    MA_FROM_UN_ID DECIMAL(18,0),
    MA_REG_NAME VARCHAR(100),
    MA_PR_NAME VARCHAR(50),
    MA_USER_NAME VARCHAR(30),
    MA_LOAN_OFFICER VARCHAR(100),
    MA_KF_NAME VARCHAR(100),
    MA_OFFICE_ID INTEGER,
    MA_CURRENCY_ID VARCHAR(3),
    MA_NO INTEGER,
    MA_PERCENT_CATEGORY DECIMAL(18,7),
    MA_REFUND INTEGER)
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  ma_from_reg_id = 0;
  for select ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay, ma_percent,
             ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save, ma_from_un_id,
             ma_office_id, ma_currency_id, ma_no, ma_percent_category, reg_name, pr_name, un_name, c_name_fio,
             kal_new_kf_name, ma_refund
      from miniapp, product, region, knaktlist, username, clients
      where ma_id = :rep_ma_id and
            ma_from_pr_id = pr_id and
            ma_office_id = reg_unicode and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            kal_from_kf_id = ma_from_kf_id and
            un_id = ma_from_un_id and
            knaktlist.is_delete = 0 and
            kal_new_kf_officer = c_id
--      and ma_date_app<=:rep_date_end
--      and ma_date_end>=:rep_date_end
      order by ma_id desc, ma_number desc
      into :ma_id, :ma_number, :ma_date_app, :ma_date_oper, :ma_from_kf_id, :ma_from_pr_id, :ma_kolich_repay,
           :ma_percent, :ma_kolich_client, :ma_memo, :ma_is_active, :ma_is_auto, :ma_summa_minus, :ma_date_save,
           :ma_from_un_id, :ma_office_id, :ma_currency_id, :ma_no, :ma_percent_category, :ma_reg_name, :ma_pr_name,
           :ma_user_name, :ma_loan_officer, :ma_kf_name, :ma_refund

  do
  begin
    suspend;
  end
end
