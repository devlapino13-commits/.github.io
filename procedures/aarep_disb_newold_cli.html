CREATE PROCEDURE AAREP_DISB_NEWOLD_CLI(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    DISB_YEAR INTEGER,
    DISB_MONTH INTEGER,
    AVG_PERIOD DECIMAL(4,2),
    DISB_KOLICH INTEGER,
    DISB_AMOUNT CURR_TYPE,
    DISB_KOLICH_NEW_CL INTEGER,
    DISB_AMOUNT_NEW_CL CURR_TYPE,
    DISB_KOLICH_OLD_CL INTEGER,
    DISB_AMOUNT_OLD_CL CURR_TYPE,
    DATE_BEGIN DATE,
    DATE_END DATE,
    ACTIVE_KOLICH_CL INTEGER,
    ACTIVE_KOLICH_CL_NEW INTEGER,
    ACTIVE_KOLICH_CL_OLD INTEGER,
    AA_BALANCE CURR_TYPE,
    AA_BALANCE_NEW CURR_TYPE,
    AA_BALANCE_OLD CURR_TYPE)
AS
declare variable ok_os decimal(18,4);
declare variable k_id id_type;
declare variable k_amount_approved curr_type;
declare variable ok_os_new curr_type;
declare variable ok_os_old curr_type;
declare variable cm_is_client_mode integer;
begin
-- knclient_fix
  select cm_is_client_mode
  from company
  into cm_is_client_mode;

  if (cm_is_client_mode = 0) then
  begin

    aa_balance = 0;
    aa_balance_new = 0;
    aa_balance_old = 0;
    for with tabl1
        as (select extract(year from ks_date_begin) as disb_year, extract(month from ks_date_begin) as disb_month,
                   kc_amount_approved * k_index_curr_rate as disb_amount_cl, k_period, kc_circle
            from kontstate, kontract, knclient, knaktlist, region_list(:rep_region_id, 1, 3, 1)
            where ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  ks_date_begin <= :rep_date_end and
                  ks_from_k_id = k_id and
                  k_from_kf_id = kal_from_kf_id and
                  knaktlist.is_delete = 0 and
                  :rep_date_end between kal_date and kal_date_end and
                  kal_new_reg_id = id_list and
                  ks_from_k_id = kc_from_k_id and
                  kc_amount_approved > 0)
        select disb_year, disb_month, avg(k_period), count(k_period), sum(disb_amount_cl), count(
               case
                 when kc_circle <= 1 then k_period
               end),
               sum(
               case
                 when kc_circle <= 1 then disb_amount_cl
               end),
               count(
               case
                 when kc_circle > 1 then k_period
               end),
               sum(
               case
                 when kc_circle > 1 then disb_amount_cl
               end)
        from tabl1
        group by disb_year, disb_month
        into disb_year, disb_month, avg_period, disb_kolich, disb_amount, disb_kolich_new_cl, disb_amount_new_cl,
             disb_kolich_old_cl, disb_amount_old_cl
    do
    begin
      date_begin = '01.' || disb_month || '.' || disb_year;
      date_end = date_begin + 32;
      date_end = '01.' || extract(month from :date_end) || '.' || extract(year from :date_end);
      date_end = date_end - 1;

      ok_os_new = null;
      ok_os_old = null;
      select sum(
             case
               when kc_circle <= 1 then cast(kc_amount_approved as double precision) / k_amount_approved * ok_os
             end),
             sum(
             case
               when kc_circle > 1 then cast(kc_amount_approved as double precision) / k_amount_approved * ok_os
             end)
      from kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1), operkred, knclient
      where k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = id_list and
            :date_end between kal_date and kal_date_end and
            ok_from_k_id = k_id and
            ok_ol_is_active = 1 and
            ok_ol_date between :date_begin and :date_end and
            ok_type in (4, 5, 6, 8) and
            ok_os <> 0 and
            kc_from_k_id = k_id and
            kc_amount_approved > 0 and
            k_amount_approved <> 0 and
            ok_from_c_id = 0
      into ok_os_new, ok_os_old;

      if (ok_os_new is null) then
        ok_os_new = 0;
      if (ok_os_old is null) then
        ok_os_old = 0;
      aa_balance_new = aa_balance_new + ok_os_new;
      aa_balance_old = aa_balance_old + ok_os_old;
      aa_balance = aa_balance_new + aa_balance_old;

      active_kolich_cl = null;
      active_kolich_cl_new = null;
      active_kolich_cl_old = null;
      select count(kc_from_c_id), count(
             case
               when kc_circle <= 1 then kc_from_c_id
             end),
             count(
             case
               when kc_circle > 1 then kc_from_c_id
             end)
      from kontstate, knclient, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1)
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = id_list and
            :date_end between kal_date and kal_date_end
      into active_kolich_cl, active_kolich_cl_new, active_kolich_cl_old;
      if (active_kolich_cl is null) then
        active_kolich_cl = 0;
      if (active_kolich_cl_new is null) then
        active_kolich_cl_new = 0;
      if (active_kolich_cl_old is null) then
        active_kolich_cl_old = 0;

      suspend;
    end
    exit;
  end

  aa_balance = 0;
  aa_balance_new = 0;
  aa_balance_old = 0;
  for with tabl1
      as (select extract(year from ks_date_begin) as disb_year, extract(month from ks_date_begin) as disb_month,
                 kc_amount_approved * k_index_curr_rate as disb_amount_cl, k_period, kc_circle
          from kontstate, kontract, knclient, knaktlist, region_list(:rep_region_id, 1, 3, 1)
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin <= :rep_date_end and
                ks_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                :rep_date_end between kal_date and kal_date_end and
                kal_new_reg_id = id_list and
                ks_from_k_id = kc_from_k_id and
                kc_amount_approved > 0)
      select disb_year, disb_month, avg(k_period), count(k_period), sum(disb_amount_cl), count(
             case
               when kc_circle <= 1 then k_period
             end),
             sum(
             case
               when kc_circle <= 1 then disb_amount_cl
             end),
             count(
             case
               when kc_circle > 1 then k_period
             end),
             sum(
             case
               when kc_circle > 1 then disb_amount_cl
             end)
      from tabl1
      group by disb_year, disb_month
      into disb_year, disb_month, avg_period, disb_kolich, disb_amount, disb_kolich_new_cl, disb_amount_new_cl,
           disb_kolich_old_cl, disb_amount_old_cl
  do
  begin
    date_begin = '01.' || disb_month || '.' || disb_year;
    date_end = date_begin + 32;
    date_end = '01.' || extract(month from :date_end) || '.' || extract(year from :date_end);
    date_end = date_end - 1;

    ok_os_new = null;
    ok_os_old = null;
    select sum(
           case
             when kc_circle <= 1 then ok_os
           end),
           sum(
           case
             when kc_circle > 1 then ok_os
           end)
    from kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1), operkred, knclient
    where k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_new_reg_id = id_list and
          :date_end between kal_date and kal_date_end and
          ok_from_k_id = k_id and
          ok_ol_is_active = 1 and
          ok_ol_date between :date_begin and :date_end and
          ok_type in (4, 5, 6, 8) and
          ok_os <> 0 and
          kc_from_k_id = k_id and
          kc_amount_approved > 0 and
          k_amount_approved <> 0 and
          ok_from_c_id = kc_from_c_id
    into ok_os_new, ok_os_old;

    if (ok_os_new is null) then
      ok_os_new = 0;
    if (ok_os_old is null) then
      ok_os_old = 0;
    aa_balance_new = aa_balance_new + ok_os_new;
    aa_balance_old = aa_balance_old + ok_os_old;
    aa_balance = aa_balance_new + aa_balance_old;

    active_kolich_cl = null;
    active_kolich_cl_new = null;
    active_kolich_cl_old = null;
    select count(kc_from_c_id), count(
           case
             when kc_circle <= 1 then kc_from_c_id
           end),
           count(
           case
             when kc_circle > 1 then kc_from_c_id
           end)
    from kontstate, knclient, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1)
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_new_reg_id = id_list and
          :date_end between kal_date and kal_date_end
    into active_kolich_cl, active_kolich_cl_new, active_kolich_cl_old;
    if (active_kolich_cl is null) then
      active_kolich_cl = 0;
    if (active_kolich_cl_new is null) then
      active_kolich_cl_new = 0;
    if (active_kolich_cl_old is null) then
      active_kolich_cl_old = 0;

    suspend;
  end
end
