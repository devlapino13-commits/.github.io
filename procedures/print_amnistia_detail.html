CREATE PROCEDURE PRINT_AMNISTIA_DETAIL(
    AAREP_K_ID ID_TYPE)
RETURNS (
    KD_K_NEW ID_TYPE,
    K_KONTRACT_NO VARCHAR(50),
    K_AMOUNT_APPROVED CURR_TYPE,
    S_DATE DATE,
    S_NUMBER INTEGER,
    S_OS CURR_TYPE,
    S_PERC CURR_TYPE,
    P_OS CURR_TYPE,
    P_PERC CURR_TYPE,
    P_FINE CURR_TYPE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_OS_PERC CURR_TYPE,
    K_DISB_DATE DATE,
    K_KREDLINE_END DATE,
    IS_PAY_DIFF INTEGER,
    S_CALC_FINE CURR_TYPE,
    ZAD_OS_PERC_FINE CURR_TYPE)
AS
declare variable rep_date_end date;
declare variable s_perc_calc curr_type;
begin
  rep_date_end = current_date;
  for select distinct kc_from_k_id
      from kontstate ks, knclient kc
      where ks_from_k_id starting left(:aarep_k_id, 17) and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0
      into kd_k_new
  do
  begin
    select k_kontract_no, k_amount_approved, k_disb_date, k_kredline_end
    from kontract
    where k_id = :kd_k_new
    into k_kontract_no, k_amount_approved, k_disb_date, k_kredline_end;

    for select s_date, s_number, sum(s_os) s_os, sum(s_perc) s_perc, sum(p_os) p_os, sum(p_perc) p_perc,
               sum(p_fine) p_fine, sum(zad_os) zad_os, sum(zad_perc) zad_perc, sum(is_pay_diff) is_pay_diff
        from (select
                     case
                       when ok.ok_type = 4 then ok_ol_date
                       else :k_disb_date
                     end as s_date,
                     case
                       when ok.ok_type = 4 then ok_sh_id
                       else 0
                     end as s_number,

                     ok_sh_os as s_os, ok_sh_perc as s_perc, ok_os as p_os, ok_perc as p_perc, ok_fine as p_fine,
                     ok_sh_os + ok_os as zad_os, ok_sh_perc + ok_perc as zad_perc,
                     iif((ok_sh_date <> ok.ok_ol_date and ok.ok_type = 4), 1, 0) as is_pay_diff
              from operkred ok
              where ok_from_k_id = :kd_k_new and
                    ok_ol_is_active = 1 and
                    ok_type < 5 and
                    ok_ol_date <= :rep_date_end and
                    ok_from_c_id = 0
              union all
              select
                     case
                       when oko.oko_type = 4 then oko.oko_ol_date
                       else :k_disb_date
                     end as s_date,
                     case
                       when oko.oko_type = 4 then 99
                       else 0
                     end as s_number,

                     0 as s_os, 0 as s_perc, oko.oko_os as p_os, oko.oko_perc as p_perc, 0 as p_fine,
                     oko_sh_os + oko_os as zad_os, oko_sh_perc + oko_perc as zad_perc, 1 as is_pay_diff
              from operkroff oko
              where oko.oko_from_k_id = :kd_k_new and
                    oko.oko_ol_is_active = 1 and
                    oko.oko_type = 4 and
                    oko.oko_ol_date <= :rep_date_end) a
        group by s_date, s_number
        having sum(s_os) <> 0 or sum(s_perc) <> 0 or sum(p_os) <> 0 or sum(p_perc) <> 0 or sum(p_fine) <> 0 or sum(zad_os) <> 0 or sum(zad_perc) <> 0
        order by s_date, s_number
        into s_date, s_number, s_os, s_perc, p_os, p_perc, p_fine, zad_os, zad_perc, is_pay_diff
    do
    begin
      s_perc_calc = 0;
      s_calc_fine = 0;

      if (s_perc <> 0) then
      begin

        select sum(sh_calc_perc - zad_perc), sum(sh_calc_fine)
        from aarep_make_zad_list(:kd_k_new, :rep_date_end, 1)
--        where SH_DATE is not null
        into s_perc_calc, s_calc_fine;

        s_perc_calc = coalesce(s_perc_calc, 0);
        s_calc_fine = coalesce(s_calc_fine, 0);

        s_perc = s_perc + s_perc_calc;

      end

      zad_perc = s_perc + p_perc;

      zad_os_perc = zad_os + zad_perc;
      zad_os_perc_fine = zad_os + zad_perc + s_calc_fine;

      suspend;
    end
  end
end
