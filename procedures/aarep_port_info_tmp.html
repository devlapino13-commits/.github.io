CREATE PROCEDURE AAREP_PORT_INFO_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_REGION_KIND INTEGER)
RETURNS (
    REG_NAME VARCHAR(100),
    REP_BALANCE_ALL DECIMAL(18,4),
    REP_BALANCE_LAST DECIMAL(18,4),
    REP_BALANCE7_LAST DECIMAL(18,4),
    REP_BALANCE30_LAST DECIMAL(18,4),
    REP_BALANCE180_LAST DECIMAL(18,4),
    REP_PERCENT_SUM DECIMAL(18,4),
    REP_BALANCE1_180_LAST DECIMAL(18,4),
    REP_NONSECURED_BALANCE DECIMAL(18,4),
    REP_LOAN_COUNT INTEGER,
    REP_UNSECURED_LOAN_COUNT INTEGER,
    REP_JUST_CL_END INTEGER,
    REP_UNSECURED_JUST_CL_END INTEGER,
    REP_BALANCE30_REST_LAST DECIMAL(18,4),
    REP_IPF_BALANCE ID_TYPE,
    REP_IPF_LOAN_COUNT INTEGER,
    REP_IPF_JUST_CL_END INTEGER,
    REP_SPS_CUSTOM PERC_TYPE,
    REP_SPS_TRADIT_CRED PERC_TYPE,
    REP_SPS_IPF PERC_TYPE,
    REP_SPS_IPF_SIMPLE PERC_TYPE,
    REP_IS_IPF INTEGER,
    REP_IS_TRADIT INTEGER,
    K_KONTRACT_NO STR_100)
AS
declare variable reg_id integer;
declare variable curr_rate decimal(18,4) = 0;
declare variable reg_id_prior integer;
declare variable reg_id_tmp integer;
declare variable reg_report_level integer;
declare variable oks_delinq_date date;
declare variable rep_is_indiv_nonsecured integer;
declare variable k_id decimal(18,0);
declare variable c_id_tmp decimal(18,0);
declare variable dd_disb_date date;
declare variable dd_close_tmp date;
declare variable chet_kol_end integer;
declare variable db_id integer;
declare variable is_restruct smallint;
declare variable pr_code varchar(100);
declare variable rep_sps_simple perc_type;
declare variable rep_sps_nbkr perc_type;
begin

  :db_id = f_get_db_id();

  reg_id_prior = 0;
-- расчет кредитного партфеля на конец
  for select k_disb_date, oks_from_k_id, oks_region_id,
             iif(((pr_set_rate_in_kont = 1) and (cr_rate < k_index_curr_rate)), k_index_curr_rate, cr_rate) as curr_rate,
             oks_balance as rep_balance_all, oks_delinq_date,
--  iif (pr.pr_code containing '.ipf.',  kd_percent_per_day * 3.6000, kd_percent_per_day * 3.6500) as rep_percent_sum, -- было pr_code containing '.murabaha.' заменен от 02.04.2018 улан
             iif(pr.pr_code containing '.ipf.', kd_percent_per_day * iif(pr.pr_percent_for_days = 30, 3.6000, 3.6500), kd_percent_per_day * 3.6500) as rep_percent_sum,
             iif(pr_loan_type = 1 and pr_is_employer = 0 and pr_is_secured = 0, 1, 0), k_kolich_client_approved,
             pr_code, kont.k_kontract_no
      from oksummary oks, kontract kont, kontdata, opercurr, product pr, region_list(:rep_region_id, 1, -1, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_from_k_id = k_id and
            k_id = kd_from_k_id and
            kd_ol_is_delete = 0
--  and oks_region_id in (1010100, 1040100, 1060400, 1050100)
            and
            :rep_date_end between kd_ol_date and kd_date_end and
            :rep_date_end between cr_date and cr_date_end and
            cr_from_cu_id = k_index_curr_code and
            k_from_pr_id = pr_id and
            oks_close_state = 0 and
            oks.oks_region_id = id_list and
            ((:db_id = 81001 and
            :rep_date_end - oks_delinq_date <= 360) or
            :db_id <> 81001)
      into :dd_disb_date, :k_id, :reg_id, :curr_rate, rep_balance_all, oks_delinq_date, rep_percent_sum,
           :rep_is_indiv_nonsecured, :rep_loan_count, :pr_code, :k_kontract_no
  do
  begin
    if (curr_rate <> 1) then
      rep_balance_all = rep_balance_all * curr_rate;
    rep_percent_sum = rep_percent_sum * rep_balance_all;

    if (rep_is_indiv_nonsecured = 1) then
    begin
      rep_nonsecured_balance = rep_balance_all;
      rep_unsecured_loan_count = rep_loan_count;
    end
    else
    begin
      rep_nonsecured_balance = 0;
      rep_unsecured_loan_count = 0;
    end

    rep_sps_simple = 0;
    rep_sps_nbkr = 0;

    rep_is_ipf = 0; -- признак ипф кредита
    rep_is_tradit = 0; -- признак традиц. кредита

    rep_sps_custom = 0; -- спс вне зависимости от типа кредита (по формуле нбкр)
    rep_sps_tradit_cred = 0; -- спс по традиционным кредитам
    rep_sps_ipf = 0; -- спс по ипф (по формуле нбкр)
    rep_sps_ipf_simple = 0; -- спс по ипф (простой метод расчета)

    if (db_id in (39801, 41701)) then
    begin
      select rep_sps_simple, rep_sps_nbkr
      from get_kont_sps(:k_id, :rep_date_end, 0)
      into :rep_sps_simple, :rep_sps_nbkr;
    end

    rep_sps_simple = coalesce(rep_sps_simple, 0);
    rep_sps_nbkr = coalesce(rep_sps_nbkr, 0);

    rep_sps_custom = coalesce(rep_sps_nbkr, 0);

    if (pr_code containing '.ipf.') then -- было pr_code containing '.murabaha.' изменен от 02.04.2018 улан
    begin
      rep_ipf_balance = rep_balance_all;
      rep_ipf_loan_count = rep_loan_count;

      rep_sps_tradit_cred = 0;
      rep_sps_ipf = rep_sps_nbkr;
      rep_sps_ipf_simple = rep_sps_simple;

      rep_is_ipf = 1;
    end
    else
    begin
      rep_ipf_balance = 0;
      rep_ipf_loan_count = 0;

      rep_sps_tradit_cred = rep_sps_nbkr;
      rep_sps_ipf = 0;
      rep_sps_ipf_simple = 0;

      rep_is_tradit = 1;
    end

    if (oks_delinq_date <= :rep_date_end) then
      rep_balance_last = rep_balance_all;
    else
      rep_balance_last = 0;

    if (oks_delinq_date < :rep_date_end - 7) then
      rep_balance7_last = rep_balance_all;
    else
      rep_balance7_last = 0;

    if (oks_delinq_date < :rep_date_end - 30) then
    begin
      rep_balance30_last = rep_balance_all;
      rep_balance30_rest_last = rep_balance_all;
    end
    else
    begin
      rep_balance30_last = 0;
      rep_balance30_rest_last = 0;
      select is_restruct
      from aarep_get_restructuring(:k_id, :rep_date_end, :db_id)
      into is_restruct;

      if (is_restruct = 1) then
      begin
        rep_balance30_rest_last = rep_balance_all;
      end
    end

    if (oks_delinq_date < :rep_date_end - 180) then
      rep_balance180_last = rep_balance_all;
    else
      rep_balance180_last = 0;

    if ((oks_delinq_date <= :rep_date_end) and
        (oks_delinq_date >= :rep_date_end - 180)) then
      rep_balance1_180_last = rep_balance_all;
    else
      rep_balance1_180_last = 0;

    if (reg_id <> reg_id_prior) then
    begin
      reg_id_tmp = :reg_id;
      reg_report_level = 0;
      while ((reg_report_level <> :rep_region_kind) and (reg_id_tmp is not null)) do
      begin
        select reg_name, parent, reg_report_level
        from region
        where reg_unicode = :reg_id_tmp
        into :reg_name, :reg_id_tmp, reg_report_level;
      end
    end

    rep_just_cl_end = 0;
    rep_unsecured_just_cl_end = 0;
    rep_ipf_just_cl_end = 0;
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :c_id_tmp
    do
    begin
      chet_kol_end = 1;

      for select kc_date_close_os
          from knclient
          where ((kc_date_disb between '01.01.1900' and (:dd_disb_date - 1)) or
                (kc_date_disb = :dd_disb_date and
                kc_from_k_id < :k_id)) and
                kc_date_close_os between :dd_disb_date and '31.12.9999' and
                kc_from_c_id = :c_id_tmp
          into :dd_close_tmp
      do
      begin
        if (dd_close_tmp >= :rep_date_end) then
        begin
          chet_kol_end = 0;
        end
      end
      rep_just_cl_end = rep_just_cl_end + chet_kol_end;
      if (rep_is_indiv_nonsecured = 1) then
        rep_unsecured_just_cl_end = rep_unsecured_just_cl_end + chet_kol_end;
      if (pr_code containing '.ipf.') then
        rep_ipf_just_cl_end = rep_ipf_just_cl_end + chet_kol_end; -- было pr_code containing '.murabaha.' изменен от 02.04.2018 улан
    end

    suspend;
  end
end
