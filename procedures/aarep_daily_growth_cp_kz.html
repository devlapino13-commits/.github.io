CREATE PROCEDURE AAREP_DAILY_GROWTH_CP_KZ(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    OFFICE_ID ID_TYPE,
    OFFICE_NAME VARCHAR(50),
    GROWTH_PORTFOLIO_FOR_DAY CURR_TYPE,
    GROWTH_CL_FOR_DAY D_INTEGER,
    GROWTH_PAR30_FOR_DAY CURR_TYPE,
    GROWTH_PAR1_FOR_DAY CURR_TYPE,
    GROWTH_ISSUED_FOR_DAY D_INTEGER,
    GROWTH_PORTFOLIO_FOR_BM CURR_TYPE,
    GROWTH_CL_FOR_BM D_INTEGER,
    GROWTH_PAR30_FOR_BM CURR_TYPE,
    GROWTH_PAR1_FOR_BM CURR_TYPE,
    GROWTH_ISSUED_FOR_BM D_INTEGER,
    QUANTITY_WORKERS D_INTEGER,
    BM_BALANCE CURR_TYPE,
    YD_BALANCE CURR_TYPE,
    TD_BALANCE CURR_TYPE,
    BM_PAR30 CURR_TYPE,
    YD_PAR30 CURR_TYPE,
    TD_PAR30 CURR_TYPE,
    BM_PAR1 CURR_TYPE,
    YD_PAR1 CURR_TYPE,
    TD_PAR1 CURR_TYPE,
    BM_KL D_INTEGER,
    YD_KL D_INTEGER,
    TD_KL INTEGER,
    YD_ISSUED_CNT D_INTEGER,
    BM_ISSUED_CNT D_INTEGER,
    TD_ISSUED_CNT D_INTEGER)
AS
declare variable T_K_ID ID_TYPE;
declare variable K_DISB_DATE date;
declare variable TD_KL_TEMP integer;
declare variable Y_K_ID ID_TYPE;
declare variable K_DISB_DATE_Y date;
declare variable YD_KL_TEMP integer;
declare variable BM_K_ID ID_TYPE;
declare variable K_DISB_DATE_BM date;
declare variable BM_KL_TEMP integer;
declare variable REP_DATE date;
declare variable ISS_K_ID ID_TYPE;
declare variable ISS_CNT_TEMP integer;
begin
--Беру все активные офисы
  for select reg_unicode, reg_name
      from region reg, region_list(:rep_region_id, 1, -1, 1)
      where reg.is_delete = 0 and
            reg_status = 3 and
            reg.reg_date_close is null and
            reg.reg_date_open <= :rep_date_end and
            parent <> 1120000 --Убираем агентов
            and
            (reg_unicode = id_list)
      order by reg_name
      into office_id, office_name
  do
  begin
--Dobavil -1 po prosbe Bahtiyara Tolbaeva 11 apr 16
    rep_date = cast(('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end)) as date) - 1;

    bm_balance = 0;
    bm_par30 = 0;
    bm_par1 = 0;
    select sum(oks_balance),
           sum(iif(:rep_date - oks_delinq_date > 30, oks_balance, 0)),
           sum(iif(:rep_date - oks_delinq_date > 0 , oks_balance, 0))
    from oksummary oks
    where oks_close_state = 0 and
          :rep_date between oks_date_begin and oks_date_end and
          oks_region_id = :office_id
    into bm_balance, bm_par30, bm_par1;
    if (bm_balance is null) then bm_balance = 0;
    if (bm_par30 is null) then bm_par30 = 0;
    if (bm_par1 is null) then bm_par1 = 0;

--Беру портфель и ПАР30 и ПАР1 на предыдущую дату
    yd_balance = 0;
    yd_par30 = 0;
    yd_par1 = 0;
    select sum(oks_balance),
           sum(iif(cast(:rep_date_end - 1 as date) - oks_delinq_date > 30, oks_balance, 0)),
           sum(iif(cast(:rep_date_end - 1 as date) - oks_delinq_date > 0, oks_balance, 0))
    from oksummary oks
    where oks_close_state = 0 and
          :rep_date_end - 1 between oks_date_begin and oks_date_end and
          oks_region_id = :office_id
    into yd_balance, yd_par30, yd_par1;
    if (yd_balance is null) then yd_balance = 0;
    if (yd_par30 is null) then yd_par30 = 0;
    if (yd_par1 is null) then yd_par1 = 0;

--Беру портфель и ПАР30 и ПАР1 на дату отчета
    td_balance = 0;
    td_par30 = 0;
    td_par1 = 0;
    select sum(oks_balance),
           sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 30, oks_balance, 0)),
           sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 0, oks_balance, 0))
    from oksummary oks
    where oks_close_state = 0 and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :office_id
    into td_balance, td_par30, td_par1;
    if (td_balance is null) then td_balance = 0;
    if (td_par30 is null) then td_par30 = 0;
    if (td_par1 is null) then td_par1 = 0;

    td_kl = 0;
--Количество клиентов офиса на дату отчета
    for select distinct k_id, k_disb_date
        from kontstate ks, kontract kt, knaktlist kal
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end and
              ks_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              kal.is_delete = 0 and
              :rep_date_end between kal_date and kal_date_end and
              kal_new_reg_id = :office_id
        into t_k_id, k_disb_date
    do
    begin
      td_kl_temp = 0;
      for select res_just_client_count
          from hp_get_just_client_in_loan(:rep_date_end, :t_k_id, :k_disb_date)
          into td_kl_temp
      do
      begin
        if (td_kl_temp is null) then td_kl_temp = 0;
        td_kl = td_kl + td_kl_temp;
      end
    end

    yd_kl = 0;
--Количество клиентов офиса на предыдущую дату
    for select distinct k_id, k_disb_date
        from kontstate ks, kontract kt, knaktlist kal
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end - 1 between ks_date_begin and ks_date_end and
              ks_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              kal.is_delete = 0 and
              :rep_date_end - 1 between kal_date and kal_date_end and
              kal_new_reg_id = :office_id
        into y_k_id, k_disb_date_y
    do
    begin
      yd_kl_temp = 0;
      for select res_just_client_count
          from hp_get_just_client_in_loan(:rep_date_end - 1, :y_k_id, :k_disb_date_y)
          into yd_kl_temp
      do
      begin
        if (yd_kl_temp is null) then yd_kl_temp = 0;
        yd_kl = yd_kl + yd_kl_temp;
      end
    end

    bm_kl = 0;
--Количество клиентов офиса на начала месяца
    for select distinct k_id, k_disb_date
        from kontstate ks, kontract kt, knaktlist kal
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date between ks_date_begin and ks_date_end and
              ks_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              kal.is_delete = 0 and
              :rep_date between kal_date and kal_date_end and
              kal_new_reg_id = :office_id
        into bm_k_id, k_disb_date_bm
    do
    begin
      bm_kl_temp = 0;
      for select res_just_client_count
          from hp_get_just_client_in_loan(:rep_date, :bm_k_id, :k_disb_date_bm)
          into bm_kl_temp
      do
      begin
        if (bm_kl_temp is null) then bm_kl_temp = 0;
        bm_kl = bm_kl + bm_kl_temp;
      end
    end

    -- Выданные кредиты на предыдущую дату
    yd_issued_cnt = 0;
    for select count(*) 
        from kontstate ks
        where ks.ks_ol_is_delete = 0
          and ks.ks_state = 1
          and ks.ks_date_begin = :rep_date_end - 1
          and ks.ks_region_id = :office_id
        into yd_issued_cnt
    do begin end
    if (yd_issued_cnt is null) then yd_issued_cnt = 0;

    -- Выданные кредиты с начала месяца по дату отчета
    bm_issued_cnt = 0;
    for select count(*)
        from kontstate ks
        where ks.ks_ol_is_delete = 0
          and ks.ks_state = 1
          and ks.ks_date_begin between
              cast(('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end)) as date)
              and :rep_date_end
          and ks.ks_region_id = :office_id
        into bm_issued_cnt
    do begin end
    if (bm_issued_cnt is null) then bm_issued_cnt = 0;

    -- Выданные кредиты на дату отчёта
    td_issued_cnt = 0;
    for select count(*)
        from kontstate ks
        where ks.ks_ol_is_delete = 0
          and ks.ks_state = 1
          and ks.ks_date_begin = :rep_date_end
          and ks.ks_region_id = :office_id
        into td_issued_cnt
    do begin end
    if (td_issued_cnt is null) then td_issued_cnt = 0;

    select count(distinct hrp.ps_from_c_id) from hr_pstatus hrp
    where hrp.ps_from_reg_id = :office_id
      and :rep_date_end between hrp.ps_date_begin and hrp.ps_date_end
      and hrp.ps_from_jl_id = 7
      and hrp.ps_status = 1
      and hrp.ps_is_working = 1
    into :quantity_workers;

    growth_issued_for_day = td_issued_cnt; --Выданные кредиты за один день
    growth_issued_for_bm = bm_issued_cnt;  --Выданные кредиты с начала месяца
    growth_portfolio_for_day = td_balance - yd_balance; --Рост портфеля за один день
    growth_par30_for_day = td_par30 - yd_par30; --Рост ПАР30 за один день
    growth_par1_for_day = td_par1 - yd_par1; --Рост ПАР7 за один день
    growth_cl_for_day = td_kl - yd_kl; --Рост клиентов за один день
    growth_portfolio_for_bm = td_balance - bm_balance; --Рост портфеля с начала месяца
    growth_par30_for_bm = td_par30 - bm_par30; --Рост ПАР30 с начала месяца
    growth_par1_for_bm = td_par1 - bm_par1; --Рост ПАР7 с начала месяца
    growth_cl_for_bm = td_kl - bm_kl; --Рост клиентов с начала месяца
    
    if (growth_issued_for_day < 0) then growth_issued_for_day = 0;
    if (growth_portfolio_for_day + growth_cl_for_day + growth_par30_for_day + growth_par1_for_day + growth_portfolio_for_bm + growth_cl_for_bm + growth_par30_for_bm + growth_par1_for_bm + bm_balance + yd_balance + td_balance + bm_par30 + yd_par30 + td_par30 + bm_par1 + yd_par1 + td_par1 + bm_kl + yd_kl + td_kl <> 0) then
    suspend;
  end
end
