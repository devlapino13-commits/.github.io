CREATE PROCEDURE HP_OFFICE_OPEN_CLOSE_DATES
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REG_OPEN_DATE DATE,
    REG_CLOSE_DATE DATE)
AS
declare variable dd date;
begin
  for select reg_unicode, reg_code, reg_name
      from region
      where reg_status = 3
      into :reg_id, :reg_code, :reg_name
  do
  begin
    reg_open_date = null;
    reg_close_date = null;
    select min(ol_date_oper_day)
    from operlist
    where ol_from_reg_id = :reg_id and
          is_delete = 0
    into :dd;
    reg_open_date = :dd;

    select min(ps_date_begin)
    from hr_pstatus
    where ps_status = 1 and
          ps_from_reg_id = :reg_id
    into :dd;
    if (dd < reg_open_date) then
      reg_open_date = :dd;

    if (not(exists(select *
                   from oksummary
                   where oks_region_id = :reg_id and
                         oks_date_begin <= current_date and
                         oks_date_end >= current_date and
                         oks_close_state = 0 and
                         oks_balance <> 0))) then -- Нет активных кредитов, нужно проверить дату операций
    begin
      select max(ps_date_begin)
      from hr_pstatus
      where ps_from_reg_id = :reg_id and
            ps_status = 1
      into :dd;
      reg_close_date = :dd;
    end
    update region
    set reg_date_open = :reg_open_date,
        reg_date_close = :reg_close_date
    where reg_unicode = :reg_id;
    suspend;
  end
end
