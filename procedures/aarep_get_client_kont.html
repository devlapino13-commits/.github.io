CREATE PROCEDURE AAREP_GET_CLIENT_KONT(
    ESM_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0))
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(100),
    K_DISB_DATE DATE,
    K_PRODUCT_NAME VARCHAR(100),
    K_STATE VARCHAR(20),
    K_SELECTED INTEGER,
    KF_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    K_AMOUNT DECIMAL(18,4),
    KC_AMOUNT DECIMAL(18,4))
AS
declare variable is_found integer;
declare variable found_k_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable rep_date date;
declare variable orig_c_id decimal(18,0);
declare variable ks_state integer;
begin
  select esm_from_k_id, esm_date, esm_object_id
  from extsysmain
  where esm_id = :esm_id
  into :found_k_id, :rep_date, :orig_c_id;

  k_state = 'Активный';
  is_found = 0;
  for select kc_from_k_id, k_kontract_no, k_disb_date, pr_common_name, k_from_kf_id, k_amount_approved,
             kc_amount_approved
      from knclient, kontstate, kontract, product
      where ks_from_k_id = kc_from_k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            ks_state = 1 and
            ks_date_begin <= :rep_date and
            ks_date_end >= :rep_date and
            ks_ol_is_delete = 0 and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id
      order by k_disb_date
      into :k_id, k_kontract_no, k_disb_date, k_product_name, kf_id, k_amount, kc_amount
  do
  begin
    kf_name = '';
    reg_name = '';

    k_selected = 0;
    if (found_k_id = k_id) then
    begin
      is_found = 1;
      k_selected = 1;
    end
    select kal_new_kf_name, reg_code || ' ' || reg_name
    from knaktlist, region
    where kal_date <= :rep_date and
          kal_date_end >= :rep_date and
          knaktlist.is_delete = 0 and
          kal_from_kf_id = :kf_id and
          kal_new_reg_id = reg_unicode
    into :kf_name, reg_name;
    suspend;
  end
  if (is_found = 0) then
  begin
    k_selected = 1;
    k_state = 'xxx';
    for select k_id, k_kontract_no, k_disb_date, pr_common_name, k_from_kf_id, k_amount_approved, ks_state
        from kontract, product, kontstate
        where k_id = :found_k_id and
              k_from_pr_id = pr_id and
              ks_from_k_id = k_id and
              ks_ol_is_delete = 0 and
              ks_date_begin <= :rep_date and
              ks_date_end >= :rep_date
        into :k_id, k_kontract_no, k_disb_date, k_product_name, :kf_id, k_amount, ks_state
    do
    begin
      if (ks_state = 0) then
        k_state = 'Заявка';
      if (ks_state = 1) then
        k_state = 'Активный';
      if (ks_state = 2) then
        k_state = 'Закрыт';

      kc_amount = 0;
      select kc_amount_approved
      from knclient
      where kc_from_k_id = :k_id and
            kc_from_c_id = :orig_c_id and
            kc_amount_approved > 0
      into :kc_amount;

      kf_name = '';
      reg_name = '';
      select kal_new_kf_name, reg_code || ' ' || reg_name
      from knaktlist, region
      where kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = :kf_id and
            kal_new_reg_id = reg_unicode
      into :kf_name, reg_name;
      suspend;
    end
  end
end
