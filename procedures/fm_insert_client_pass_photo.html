CREATE PROCEDURE FM_INSERT_CLIENT_PASS_PHOTO(
    BLS_FILENAME TYPE OF COLUMN BLOBSTORE.BLS_FILENAME,
    BLS_CAPTION TYPE OF COLUMN BLOBSTORE.BLS_CAPTION,
    BLS_BLOB TYPE OF COLUMN BLOBSTORE.BLS_BLOB,
    BLS_BLOB_SIZE TYPE OF COLUMN BLOBSTORE.BLS_BLOB_SIZE,
    BLS_FROM_UN_ID TYPE OF COLUMN BLOBSTORE.BLS_FROM_UN_ID,
    C_ID TYPE OF COLUMN CLIENTS.C_ID,
    PASS_TYPE SMALLINT)
RETURNS (
    ERR_TEXT VARCHAR(100))
AS
declare variable bls_id type of column blobstore.bls_id;
declare variable c_verify_status type of column clients.c_verify_status;
declare variable blf_id type of column blobfiles.blf_id;
begin
  bls_id = null;
  err_text = '';
  select c_verify_status, c_blob_photo_id
  from clients
  where c_id = :c_id
  into c_verify_status, bls_id;
  if (c_verify_status <> 2) then
  begin
    execute procedure hp_insert_into_blobstore (:bls_filename, :bls_caption, :bls_blob, :bls_from_un_id, 0, 1) -- Фиксируем как загрузка из FsMobile
    returning_values :bls_id;

-- Затем сохраненное фото закрепляется клиенту
    if (pass_type = 1) then
      update clients
      set c_blob_pass1_id = :bls_id
      where c_id = :c_id;
    else
    if (pass_type = 2) then
      update clients
      set c_blob_pass2_id = :bls_id
      where c_id = :c_id;

  end
  else
  begin
    err_text = 'Клиент уже верифицирован. Обновление скана-паспорта недоступно.';
  end
  suspend;
end
