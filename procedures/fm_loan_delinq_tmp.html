CREATE PROCEDURE FM_LOAN_DELINQ_TMP(
    REP_DATE D_DATE,
    REG_ID D_INTEGER,
    LO_ID D_ID,
    MAX_DELINQ D_INTEGER)
RETURNS (
    K_ID D_ID,
    KF_ID D_ID,
    K_KONTRACT_NO D_STR_100,
    RES_DELINQ_DAY D_INTEGER)
AS
begin
    if (reg_id = -1) then
    begin
        for select oks_from_k_id, k_from_kf_id, k_kontract_no, datediff(day, oks_delinq_date, :rep_date)
            from oksummary, kontract
            where :rep_date between oks_date_begin + 0 and oks_date_end and
                  oks_delinq_date < :rep_date and
                  oks_balance > 0 and
                  oks_kc_id = :lo_id and
                  oks_from_k_id = k_id and
                  :rep_date - oks_delinq_date <= :max_delinq and
                  oks_close_state = 0
            order by oks_delinq_date desc
            into :k_id, :kf_id, k_kontract_no, res_delinq_day
        do
            suspend;
    end
    else 
    begin
        for select oks_from_k_id, k_from_kf_id, k_kontract_no, datediff(day, oks_delinq_date, :rep_date)
            from oksummary, kontract
            where :rep_date between oks_date_begin + 0 and oks_date_end and
                  oks_delinq_date < :rep_date and
                  oks_balance > 0 and
                  oks_region_id = :reg_id and
                  oks_from_k_id = k_id and
                  :rep_date - oks_delinq_date <= :max_delinq and
                  oks_close_state = 0
            order by oks_delinq_date desc
            into :k_id, :kf_id, k_kontract_no, res_delinq_day
        do
            suspend;
    end
end
