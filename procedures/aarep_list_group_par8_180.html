CREATE PROCEDURE AAREP_LIST_GROUP_PAR8_180(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    KF_OFFICER_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    REP_ACTIVE_BALANCE CURR_TYPE,
    DELINQ_BALANCE8_180 CURR_TYPE,
    DELINQ_DAY INTEGER,
    DATE_ADOPTION DATE,
    DELINQ_DAY_TRANSFER INTEGER,
    IS_OWN INTEGER)
AS
declare variable oks_region_id integer;
declare variable oks_kc_id id_type;
declare variable oks_from_k_id id_type;
declare variable region_id_prior integer;
declare variable kc_id_prior id_type;
begin
  region_id_prior = 0;
  kc_id_prior = 0;
  for select oks_region_id, oks_kc_id, oks_from_k_id, oks_balance, delinq_balance8_180, delinq_day, k_kontract_no,
             date_adoption, delinq_day_transfer, is_own
      from aarep_list_group_par8_180_tmp(:rep_region_id, :rep_date_end)
      order by oks_region_id, oks_kc_id
      into :oks_region_id, :oks_kc_id, :oks_from_k_id, :rep_active_balance, :delinq_balance8_180, :delinq_day,
           :k_kontract_no, :date_adoption, :delinq_day_transfer, :is_own
  do
  begin
    if (region_id_prior <> oks_region_id) then
    begin
      region_id_prior = oks_region_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :oks_region_id and
            r1.parent = r2.reg_unicode
      into reg_name, branch_name;
    end
    if (kc_id_prior <> oks_kc_id) then
    begin
      select c_name_fio
      from clients
      where c_id = :oks_kc_id
      into kf_officer_name;
    end
    suspend;
  end
end
