CREATE PROCEDURE HP_RESCHEDULE_FLOAT_LIST(
    REP_DATE D_DATE NOT NULL)
RETURNS (
    K_ID D_ID,
    OKS_DELINQ_DATE D_DATE,
    NEW_LEVEL D_INTEGER)
AS
declare variable delinq_date d_date;
declare variable perc_level d_integer;
declare variable kd_ol_date d_date;
declare variable k_disb_date d_date;
declare variable pr_code str_100;
declare variable db_id d_integer;
begin
  delinq_date = f_date_work_nday(rep_date, -3);
  db_id = f_get_db_id();

  for select oks_from_k_id, oks_delinq_date, k_disb_date, pr_code
  from oksummary, kontract, product
  where oks_from_k_id = k_id and
        k_from_pr_id = pr_id and
        pr_percent_kind = 1 and -- Плавающая ставка процента
        oks_date_begin <= (:rep_date - 1) and
        oks_date_end >= (:rep_date - 1) and
        oks_delinq_date <= (:rep_date - 1) and
        oks_close_state = 0 and
        coalesce(k_stop_perc, :rep_date + 1) > :rep_date
  into k_id, oks_delinq_date, k_disb_date, pr_code
  do
  begin
    if (oks_delinq_date > :delinq_date) then
      if (db_id <> 81001 or k_disb_date < cast('08.09.2021' as date)) then
        Continue;

    if (exists(select * from kontstate
    where ks_from_k_id = :k_id and         
          ks_state = 2 and
          ks_ol_is_delete = 0 and
          ks_date_begin >= :rep_date)) then
      Continue;

    new_level = 1;
    perc_level = 0;
    select kd_ol_date, kd_float_perc_level
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_is_delete = 0 and
          kd_ol_date <= kd_date_end
    order by kd_ol_date desc
    rows 1
    into kd_ol_date, perc_level;

    if (perc_level > 10) then -- Для того чтобы запретить повышение делайте perc_level=999
      Continue;

    if (db_id = 81001) then
    begin
        if (perc_level = 3 or (k_disb_date < '01.11.2020' and perc_level = 2)
            or (pr_code containing '.reliable_emp.' and perc_level >= 1)) then
          Continue;
    end
    else if (db_id in (39801, 41701)) then
    begin
        if (perc_level = 2) then
            continue;
    end
    else if (db_id = 94401) then
    begin
        if (perc_level = 2) then
            continue;
    end

    if (db_id = 81001) then
    begin
        if (kd_ol_date = rep_date) then
        begin
          if (k_disb_date < '01.11.2020') then
          begin
            if ((perc_level = 1 and f_date_work_nday(oks_delinq_date, 3) <= rep_date) or
                (perc_level = 2 and f_date_work_nday(oks_delinq_date, 13) <= rep_date)) then
              Continue;
          end else
          if (k_disb_date < '08.09.2021') then
          begin
            if ((perc_level = 1 and f_date_work_nday(oks_delinq_date, 3) <= rep_date) or
                (perc_level = 2 and f_date_work_nday(oks_delinq_date, 8) <= rep_date) or
                (perc_level = 3 and f_date_work_nday(oks_delinq_date, 17) <= rep_date)) then
              Continue;
          end else
            if (perc_level = 1) then
              Continue;
        end
    end
    else if (db_id in (39801, 41701)) then
    begin
        if (kd_ol_date = rep_date) then
        begin
            if ((perc_level = 1 and f_date_work_nday(oks_delinq_date, 3) <= rep_date) or
                (perc_level = 2 and f_date_work_nday(oks_delinq_date, 8) <= rep_date)) then
            Continue;
        end
    else if (db_id = 94401) then       --плавающая процентная ставка для АЗ
    begin
        if (kd_ol_date = rep_date) then
        begin
            if ((perc_level = 1 and f_date_work_nday(oks_delinq_date, 3) <= rep_date) or
                (perc_level = 2 and f_date_work_nday(oks_delinq_date, 8) <= rep_date)) then
            Continue;
        end
    end
    end

    if (db_id = 81001) then
    begin
        if (k_disb_date < '01.11.2020') then
        begin
          if (perc_level = 1) then
          begin
            if (f_date_work_nday(oks_delinq_date, 13) > rep_date) then
              Continue;
      
            new_level = 2;
          end
        end else
        begin
          if (perc_level = 1) then
          begin
            if (f_date_work_nday(oks_delinq_date, 8) > rep_date) then
              Continue;
      
            new_level = 2;
          end
          if (perc_level = 2) then
          begin
            if (f_date_work_nday(oks_delinq_date, 17) > rep_date) then
              Continue;
      
            new_level = 3;
          end
        end
    end
    else if (db_id in (39801, 41701)) then
    begin
        if (perc_level = 1) then
        begin
            if (f_date_work_nday(oks_delinq_date, 8) > rep_date) then
              Continue;
      
            new_level = 2;
        end
    end
    else if (db_id = 94401) then
    begin
        if (perc_level = 1) then
        begin
            if (f_date_work_nday(oks_delinq_date, 8) > rep_date) then
              Continue;
      
            new_level = 2;
        end
    end
    suspend;
  end
end
