CREATE PROCEDURE GEPS_AGREEMENT_CARD_SENT
AS
declare variable NUM_PROCESSING D_INTEGER;
declare variable NUM_ADDED D_INTEGER;
declare variable RES_TEXT D_STR_255;
declare variable KC_ID D_ID;
declare variable K_ID D_ID;
declare variable GS_ID D_ID;
declare variable EDC_ID D_INTEGER;
declare variable EDC_CODE D_STR_100;
declare variable PR_CODE D_STR_100;
begin
   if (extract(hour from current_timestamp) < 9 or
     extract(hour from current_timestamp) > 21) then
   exit;

  if (f_get_db_id() not in (39801, 41701)) then
    exit;

  num_processing = (
    select count(*)
    from edoc_blank, edoc_list
    where edb_from_edl_id = edl_id
      and edb_status = 0
      and edl_status <> -1);

  num_added = 0;

  if (num_processing < 50) then
  begin
    for
      select first 250
          gs.gs_id,
          gs.gs_from_kc_id,
          gs.gs_k_id
      from geps_agreement_card gs
      where gs.gs_mark = 0
    into :gs_id ,:kc_id,:k_id
    do
    begin
         edc_id = -1;
    
         select pr.pr_code
         from kontract  kt
         inner join product pr on pr.pr_id =kt.k_from_pr_id
         where kt.k_id=:k_id
         into :pr_code;
    
          if (pr_code containing '.consultation.') then
              edc_id =35;
          else
              edc_id =34;
    
         update geps_agreement_card  gs
         set gs.gs_from_edc_id = :edc_id
         where gs.gs_id =:gs_id;

         res_text = '';

         if (edc_id <> -1) then
         begin
             select edc.edc_code
             from edoc_cfg edc
             where edc.edc_id = :edc_id
             into :edc_code;

             select res_text
             from hp_edoc_ins(:edc_code, '8', :KC_ID, 1000000000000)
             into :res_text;
         end

         update geps_agreement_card  gs
         set gs.gs_mark = iif(:res_text = '', 1, -1)
         where gs.gs_from_kc_id = :kc_id;

         num_added = num_added + 1;
    end
    if (num_added = 0) then
     update fsrobot_sh_main set fsm_is_active = 0 where fsm_id = 79;
 end
end
