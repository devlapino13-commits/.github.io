CREATE PROCEDURE HP_KN_OKSUMMARY_INSERT(
    KC_ID D_ID NOT NULL,
    REP_DATE D_DATE NOT NULL,
    NEED_RECALC D_SMALLINT = 1)
AS
declare variable oks_date_begin d_date;
declare variable oks_close_state d_smallint;
declare variable dd_min d_date;
declare variable tmp_date d_date;
declare variable oks_from_k_id d_id;
begin
  -- Вставка новой даты на kn_oksummary и добавление недостающих концов месяца
  if (f_get_db_id() <> 81001) then
    exit;

  if (need_recalc is null) then
    need_recalc = 1;

  dd_min = cast('31.12.9999' as date);

  select oks_from_k_id, oks_date_begin, oks_close_state
  from kn_oksummary
  where oks_from_kc_id = :kc_id and
        oks_date_begin <= :rep_date
  order by oks_from_kc_id desc, oks_date_begin desc
  rows 1
  into oks_from_k_id, oks_date_begin, oks_close_state;

  if (oks_from_k_id is null) then
  begin
    select kc_from_k_id
    from knclient
    where kc_id = :kc_id
    into oks_from_k_id;
  end

  -- восстанавливаем концы предыдущих месяцев если займ активный либо есть долг по госпошлине
  if (oks_date_begin < (rep_date - extract(day from rep_date)) and oks_close_state = 0) then
  begin
    tmp_date = f_date_month_end(oks_date_begin);
    if (tmp_date = oks_date_begin) then
      tmp_date = f_date_month_end(tmp_date + 1);

    while (tmp_date < rep_date) do
    begin
      if (dd_min > tmp_date) then
        dd_min = tmp_date;

      insert into kn_oksummary (oks_from_kc_id, oks_date_begin, oks_from_k_id, oks_need_recalc)
      values (:kc_id, :tmp_date, :oks_from_k_id, 0);

      tmp_date = f_date_month_end(tmp_date + 1);
    end
  end

  -- вставляем на переданную дату
  if (oks_date_begin is null or oks_date_begin <> rep_date) then
  begin
    if (dd_min > rep_date) then
      dd_min = rep_date;

    insert into kn_oksummary (oks_from_kc_id, oks_date_begin, oks_from_k_id, oks_need_recalc)
    values (:kc_id, :rep_date, :oks_from_k_id, 0);
  end

  -- вставляем на конец месяца переданной даты
  tmp_date = f_date_month_end(rep_date);

  if (not exists(select *
                 from kn_oksummary
                 where oks_from_kc_id = :kc_id and
                       oks_date_begin = :tmp_date)) then
  begin
    if (dd_min > tmp_date) then
      dd_min = tmp_date;

    insert into kn_oksummary (oks_from_kc_id, oks_date_begin, oks_from_k_id, oks_need_recalc)
    values (:kc_id, :tmp_date, :oks_from_k_id, 0);
  end

  if (need_recalc = 1 and dd_min = cast('31.12.9999' as date)) then
    dd_min = rep_date;

  if (need_recalc = 1 and dd_min <> cast('31.12.9999' as date)) then
    execute procedure hp_kn_oksummary_recalc(:kc_id, :dd_min);
end
