CREATE PROCEDURE OPERSAL_SVERKA
RETURNS (
    OL_ID DECIMAL(18,0),
    O_DATE DATE,
    C_ID DECIMAL(18,0),
    SAC_ID INTEGER,
    OS_VALUE DECIMAL(18,4),
    OL_MEMO VARCHAR(255),
    C_NAME_FIO VARCHAR(100),
    SAC_NAME VARCHAR(100),
    OL_OPER_TYPE INTEGER,
    OL_OPER_SUBTYPE INTEGER)
AS
begin
  for with tab
      as (select ol_id, ol_date as o_date, os_from_c_id as c_id, os_from_sac_id as sac_id, sum(os_value) as os_value
          from opersalary, operlist
          where os_from_ol_id = ol_id and
                operlist.is_delete = 0
          group by ol_id, ol_date, os_from_c_id, os_from_sac_id
          having sum(os_value) <> 0
          union
          select osm_from_ol_id as ol_id, osm_ol_date as o_date, osm_from_c_id as c_id, osd_from_sac_id as sac_id,
                 -sum(osd_value) as os_value
          from opersaldetail, opersalmain
          where osd_from_osm_id = osm_id and
                osm_ol_is_active = 1
          group by osm_from_ol_id, osm_ol_date, osm_from_c_id, osd_from_sac_id
          having sum(osd_value) <> 0)
      select ol_id, o_date, c_id, sac_id, sum(os_value)
      from tab
      group by ol_id, o_date, c_id, sac_id
      having sum(os_value) <> 0
      into :ol_id, :o_date, c_id, sac_id, os_value
  do
  begin
    ol_memo = '';
    c_name_fio = '';
    sac_name = '';

    select ol_oper_type, ol_oper_subtype, ol_memo
    from operlist
    where ol_id = :ol_id
    into :ol_oper_type, :ol_oper_subtype, :ol_memo;

    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    select sac_name
    from sa_config
    where sac_id = :sac_id
    into :sac_name;

    suspend;
  end
end
