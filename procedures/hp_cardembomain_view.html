CREATE PROCEDURE HP_CARDEMBOMAIN_VIEW
RETURNS (
    CEM_ID D_INTEGER,
    CEM_DATE D_DATE,
    CEM_IS_PINCOVER D_BOOL,
    CEM_APP_COUNT D_INTEGER,
    CEM_READY_COUNT D_INTEGER,
    CEM_FROM_CKD_ID D_INTEGER,
    CKD_NAME D_STR_CODE,
    CKD_CODE D_STR_CODE,
    CKD_IS_OWN D_BOOL,
    CEM_FROM_CES_ID D_INTEGER,
    CES_NAME D_STR_CODE,
    CES_CODE D_STR_CODE,
    CEM_FILENAME D_STR_100,
    CEM_FROM_CAMB_ID D_INTEGER,
    CEM_FROM_CAM_ID D_INTEGER,
    CEM_FROM_UN_ID D_ID,
    CEM_INFO D_STR_100,
    UN_FIO D_STR_100,
    IS_REGERROR D_BOOL)
AS
begin
  for
    select
      c.cem_id,
      c.cem_date,
      c.cem_is_pincover, 
      c.cem_app_count,
      c.cem_ready_count,
      c.cem_from_ckd_id,
      d.ckd_name,
      d.ckd_code,
      d.ckd_is_own,
      c.cem_from_ces_id,
      s.ces_name,
      s.ces_code,
      c.cem_filename,
      c.cem_from_camb_id,
      c.cem_from_cam_id,
      c.cem_from_un_id,
      u.un_fio
    from cardembomain as c
      inner join cardkind as d
        on d.ckd_id = c.cem_from_ckd_id
      inner join cardembostatus as s
        on s.ces_id = c.cem_from_ces_id
      inner join username as u
        on u.un_id = c.cem_from_un_id
  into
    :cem_id,
    :cem_date,
    :cem_is_pincover,
    :cem_app_count,
    :cem_ready_count,
    :cem_from_ckd_id,
    :ckd_name,
    :ckd_code,
    :ckd_is_own,
    :cem_from_ces_id,
    :ces_name,
    :ces_code,
    :cem_filename,
    :cem_from_camb_id,
    :cem_from_cam_id,
    :cem_from_un_id,
    :un_fio
  do
  begin
    cem_info = 'Заявка №' || cem_id || ' от ' || f_datetostr(cem_date) ||
      ' на выпуск ' || cem_app_count || ' карт, типа "' || ckd_name || '"';

    if (cem_from_ces_id = 1 and cem_app_count <> cem_ready_count and
      exists(
        select * from cardlist
        where cal_from_cem_id = :cem_id and cal_status < 0))
    then is_regerror = 1;
      else is_regerror = 0;

    suspend;
  end
end
