CREATE PROCEDURE CM_RU_0147_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kal_new_kf_officer id_type;
declare variable k_applicate_date date;
declare variable k_disb_date date;
declare variable c_date_start date;
declare variable c_date_end date;
declare variable iter integer;
begin
  for select kal.kal_new_kf_officer, k_applicate_date, k.k_disb_date
      from kontract k
      inner join product pr on pr.pr_id = k.k_from_pr_id
      inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
      where k.k_id = :k_id and
            pr.pr_code like '%.ZALOG_CAR.%' and --Ловим продукт типа автозайм
            kal.is_delete = 0 and
            current_date between kal.kal_date and kal.kal_date_end
      into :kal_new_kf_officer, :k_applicate_date, :k_disb_date
  do
  begin
    if (:k_applicate_date < '08.07.2019') then
      exit;
    select min(b.c_date_end)
    from car_disb_svz_list b -- проверяем сотрудника в списке
    where b.c_id = :kal_new_kf_officer
    and cdsl_pr_code = '.zalog_car.'
    and :k_disb_date between b.c_date_start and b.c_date_end  rows 1
    into :c_date_end;
    if (c_date_end is null) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Выдача запрещена,ограничение на выдачу авто-займов  по списку ВиПС!';
      suspend;
      exit;
    end
    if (:k_disb_date > :c_date_end) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Выдача запрещена, ограничение на выдачу авто-займов  по дате! (по списку ВиПС)';
      suspend;
      exit;
    end
    if (c_date_end = '31.08.2019') then
    begin
      select count(*)
      from kontstate ks
      inner join kontract k on k.k_id = ks.ks_from_k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id
      where ks.ks_kc_id = :kal_new_kf_officer and
            ks.ks_state = 1 and
            ks.ks_date_begin between '23.07.2019' and :c_date_end and
            ks.ks_ol_is_delete = 0 and
            pr.pr_code like '%.ZALOG_CAR.%'  --Ловим продукт типа автозайм
      into :iter;
      if (:iter >= 3) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Выдача запрещена,превышение порога по общему количеству выданных авто-займов (по списку ВиПС)';
        suspend;
      end
    end
  end
end
