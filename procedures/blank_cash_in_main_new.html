CREATE PROCEDURE BLANK_CASH_IN_MAIN_NEW(
    OL DECIMAL(18,0))
RETURNS (
    KOB_SUMMA D_CURRENCY,
    ACC_KT D_STR_CODE,
    ACC_DT D_STR_CODE,
    BL_DATE D_DATE,
    BL_DOC_NO D_INTEGER,
    BL_MEMO D_STR_LONG,
    BL_CLIENT_NAME D_STR_255,
    BL_OFFICE_FULL_NAME D_STR_255,
    BL_OFFICE_BANK_NAME D_STR_255,
    BL_IS_DELETE D_SMALLINT,
    BL_CORR_ACC D_STR_255,
    BL_COMPANY D_STR_255,
    BL_OFFICE_NAME D_STR_100,
    BL_BRANCH_FULL_NAME D_STR_255,
    BL_DOC_NO_STR D_STR_20,
    REG_CODE D_STR_20,
    BL_DOC_CLIENT_NAME D_STR_255,
    BL_UN_FIO D_STR_100,
    BL_MANAGER_FIO D_STR_100,
    BL_TICKET_NO D_STR_CODE,
    KOB_SUMMA4 D_CURRENCY,
    KOB_SUMMA3 D_CURRENCY,
    KOB_SUMMA2 D_CURRENCY,
    KOB_SUMMA1 D_CURRENCY,
    ACC_KT4 D_STR_CODE,
    ACC_KT3 D_STR_CODE,
    ACC_KT2 D_STR_CODE,
    ACC_KT1 D_STR_CODE)
AS
declare variable office d_str_255;
declare variable company_name d_str_255;
declare variable bl_bank_name d_str_255;
declare variable bl_bank_id d_str_10;
declare variable sub_no d_integer;
declare variable ol_manager_c_id id_type;
declare variable ol_parent_id id_type;
declare variable kc_id id_type;
declare variable st_staff d_integer;
declare variable st_client d_integer;
declare variable i d_smallint;
begin
  -- № чека ПВН операции 05.08.2021 ЕАБ
  select pvn.po_ticket_no
  from pvnopers pvn
  where pvn.po_from_ol_id = :ol rows 1
  into :bl_ticket_no;

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */

  select reg_code, un_fio, ol_manager_c_id, ol_parent_id
  from operlist, region, username
  where ol_id = :ol and
        ol_from_reg_id = reg_unicode and
        ol_user_id = un_id
  into :reg_code, :bl_un_fio, ol_manager_c_id, :ol_parent_id;

  bl_manager_fio = '';
  if ((ol_manager_c_id is null) and
      (ol_parent_id is not null)) then
  begin
    select ol_manager_c_id
    from operlist
    where ol_id = :ol_parent_id
    into :ol_manager_c_id;
  end

  if (ol_manager_c_id is not null) then
  begin
    select c_name_fio
    from clients
    where c_id = :ol_manager_c_id
    into :bl_manager_fio;
  end

  begin

    for select ol_date, ol_doc_no, ol_memo, cm_name, reg_name, b_name, br_from_b_id, operlist.is_delete, b_corr_acc,
               reg_kont_company, ol_doc_client_name
        from company, region
        left join(bankregion left join banks on (br_from_b_id = b_id)) on (br_from_reg_id = reg_unicode and
              br_is_main = 1), operlist
        where ol_id = :ol and
              reg_unicode = ol_from_reg_id rows 1
        into :bl_date, :bl_doc_no, :bl_memo, :company_name, :office, :bl_bank_name, :bl_bank_id, :bl_is_delete,
             :bl_corr_acc, :bl_branch_full_name, :bl_doc_client_name
    do
    begin
      if (coalesce(bl_doc_client_name, '') = '') then
      begin
        select c_name_fio
        from operlist, clients
        where ol_document_client = c_id and
              ol_id = :ol
        into bl_doc_client_name;
      end
      bl_doc_client_name = coalesce(bl_doc_client_name, '');
      bl_corr_acc = replace(substring(bl_corr_acc from 8 for 5), '-', '.');

      bl_office_bank_name = bl_bank_id || ' ' || bl_bank_name;
      bl_office_full_name = company_name || ' ' || office;
      bl_company = company_name;
      bl_office_name = :office;

      kob_summa = 0;
      kob_summa1 = 0;
      kob_summa2 = 0;
      kob_summa3 = 0;
      kob_summa4 = 0;

      acc_kt = null;
      acc_kt1 = null;
      acc_kt2 = null;
      acc_kt3 = null;
      acc_kt4 = null;
      sub_no = 0;
      for select kob_from_kc_id, dt.acc_code
          from kn_operbook kob
          inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
          inner join sys_accbalance sab on sab.sab_id = dt.acc_from_sab_id
          where kob_from_ol_id = :ol and
                sab.sab_full_code = 'Деньги.Нал.Касса'
          group by kob_from_kc_id, dt.acc_code
          into :kc_id, :acc_dt
      do
      begin
        i = 0;
        sub_no = sub_no + 1;
        bl_doc_no_str = reg_code || '-' || right('000000' || bl_doc_no, 6) || '/' || sub_no;
        for select kt.acc_code, kob.kob_summa
            from kn_operbook kob
            inner join ref_account dt on dt.acc_id = kob.kob_acc_dt
            inner join ref_account kt on kt.acc_id = kob.kob_acc_kt
            inner join sys_accbalance sab on sab.sab_id = dt.acc_from_sab_id
            where kob_from_ol_id = :ol and
                  sab.sab_full_code = 'Деньги.Нал.Касса' and
                  (kob_from_kc_id = :kc_id or :kc_id is null)
            into :acc_kt, :kob_summa
        do
        begin
          if (i = 0) then
          begin
            kob_summa1 = kob_summa;
            acc_kt1 = acc_kt;
          end
          else
          if (i = 1) then
          begin
            kob_summa2 = kob_summa;
            acc_kt2 = acc_kt;
          end
          else
          if (i = 2) then
          begin
            kob_summa3 = kob_summa;
            acc_kt3 = acc_kt;
          end
          else
          if (i = 3) then
          begin
            kob_summa4 = kob_summa;
            acc_kt4 = acc_kt;
          end
          bl_client_name = bl_doc_client_name;

          select c_name_fio
          from clients c, knclient kn
          where c.c_id = kn.kc_from_c_id and
                kn.kc_id = :kc_id
          into bl_client_name;

          if (bl_client_name <> bl_doc_client_name) then
            bl_client_name = bl_doc_client_name || ' за ' || bl_client_name;
          i = i + 1;

        end
        kob_summa = kob_summa1 + kob_summa2 + kob_summa3 + kob_summa4;

        suspend;
      end

    end
  end
end
