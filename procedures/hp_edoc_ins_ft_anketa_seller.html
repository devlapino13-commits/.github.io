CREATE PROCEDURE HP_EDOC_INS_FT_ANKETA_SELLER(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
begin
  -- Процедура создает анкету ПФТД для продавцов
  -- Готовит бланки и отправляет СМС.
  res_text = '';
  res_is_error = 0;

  if (res_text = '') then
  if (not exists(
    select *
    from clientdocs
      inner join passkind
        on pk_id = cds_from_pk_id
      inner join passdoctype
        on pdt_id = pk_from_pdt_id
          and pdt_code in ('license', 'work_patent', 'certificate_ip' , 'certificate_yur')
    where cds_from_c_id = :c_id
      and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)))
  then res_text = 'Не прикреплен хотя бы один из документов предпринимателя';

  if (res_text = '') then
    select
      edl_id,
      res_text
    from hp_edoc_insert('ft.anketa.seller', f_objkind('clients', 'id'), :c_id, :un_id)
    into
      :edl_id,
      :res_text;

  if (res_text <> '') then
   res_is_error = 1;

  suspend;
end
