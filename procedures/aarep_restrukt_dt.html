CREATE PROCEDURE AAREP_RESTRUKT_DT(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_ID D_STR_64,
    REG_ID D_INTEGER,
    RESTRUCT_DATE D_DATE,
    OL_DATE D_DATE,
    KF_NAME D_STR_100,
    REG_NAME D_STR_100,
    PARENT_NAME D_STR_100,
    K_KREDLINE_END D_DATE,
    K_DATE_END D_DATE,
    KOL_PROLOG D_INTEGER,
    OKS_DELINQ_DAYS D_INTEGER,
    REP_ACTIVE_BALANCE D_CURRENCY,
    RT_KIND D_STR_100,
    K_KONTRACT_NO D_STR_20,
    PR_NAME D_STR_100,
    PR_COMMON_NAME D_STR_100,
    REP_ACTIVE_BALANCE_NEW D_CURRENCY,
    K_KOLICH_CLIENT_APPROVED D_INTEGER,
    LLC_CURRENT_AMOUNT D_CURRENCY,
    LLC_CURRENT_PERCENT D_PERCENT,
    K_AMOUNT_APPROVED D_CURRENCY,
    REG_STAT_REP_REGION D_STR_255,
    RESTRUCT_KOL D_INTEGER,
    MAX_DATE_GRAPH D_DATE,
    OKS_DELINQ_DATE D_DATE,
    K_PERIOD D_CURRENCY,
    K_FACT_DISB_DATE D_DATE,
    K_PERCENT D_PERCENT,
    PR_CODE D_STR_100,
    CMC_NAME D_STR_100,
    REASON_NAME D_STR_255,
    RESTRUCT_DATE_LAST D_DATE,
    K_DISB_KINDD D_STR_100,
    K_BY_RATING D_STR_64,
    UUID D_STR_CODE,
    PRE_RESTRUCT_DAYS D_INTEGER,
    NEXT_PAY_DATE D_DATE,
    POST_RESTRUCT_DAYS D_INTEGER)
AS
declare variable CURR varchar(3);
declare variable DEF_CURR varchar(3);
declare variable CURR_RATE D_CURRENCY;
declare variable K_INDEX_CURR_RATE D_CURRENCY;
declare variable PR_SET_RATE_IN_KONT D_INTEGER;
declare variable IS_RESTRUCT D_SMALLINT;
declare variable DB_ID D_INTEGER;
declare variable REG_ID_PRIOR D_INTEGER;
declare variable MAX_DATE_REST D_DATE;
declare variable REASON_ID D_INTEGER;
declare variable K_DISB_KIND D_INTEGER;
declare variable PRE_DELINQ_DATE D_DATE;
begin
  k_disb_kindd = '';
  select cm_default_currency
  from company
  into def_curr;
  :db_id = f_get_db_id();
  reg_id_prior = 0;

  for select oks_from_k_id, oks_region_id, oks_balance
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where oks_region_id = id_list and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0
      into k_id, reg_id, rep_active_balance -- портфель на дату отчета
  do
  begin
    select is_restruct, restruct_memo, restruct_date, restruct_kol, restruct_date_last
    from aarep_get_res_for_oper(:k_id, :rep_date_end, :db_id)
    into is_restruct, rt_kind, restruct_date, restruct_kol, restruct_date_last;

    ol_date = restruct_date;
    if (is_restruct = 1) then
    begin

      select kal_new_kf_name, k_kredline_end, k_kontract_no, k_kolich_client_approved, k_currency, k_index_curr_rate,
             pr_name, pr_common_name, pr_set_rate_in_kont, k_amount_approved, k_period, k_fact_disb_date, k_percent,
             pr_code, k_disb_kind, iif(k_by_rating = 1, 'По кредитному рейтингу', 'Стандартная выдача')
      from kontract, knaktlist, product
      where k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :restruct_date between kal_date and kal_date_end and
            k_from_pr_id = pr_id
      into kf_name, k_kredline_end, k_kontract_no, k_kolich_client_approved, curr, k_index_curr_rate, pr_name,
           pr_common_name, pr_set_rate_in_kont, k_amount_approved, k_period, k_fact_disb_date, k_percent, pr_code, k_disb_kind, k_by_rating;

      if (:db_id in (39801, 41701)) then
      begin
        k_disb_kindd = case :k_disb_kind
                         when 6 then 'На карту Мбулак'
                         when 5 then 'На карту клиента'
                         else 'Касса'
                       end;
      end

      if (reg_id_prior <> reg_id) then
      begin
        reg_id_prior = reg_id;

        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r1.reg_stat_rep_region
        from region r1, region r2
        where r1.reg_unicode = :reg_id and
              r1.parent = r2.reg_unicode
        into reg_name, parent_name, reg_stat_rep_region;
      end

      --    баланс на дату реструктуризации
      rep_active_balance_new = null;
      oks_delinq_date = null;
      select oks_balance, oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            :restruct_date between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into rep_active_balance_new, oks_delinq_date;
      if (rep_active_balance_new is null) then
        rep_active_balance_new = 0;
      oks_delinq_days = restruct_date - oks_delinq_date;
      if (oks_delinq_days < 0) then
        oks_delinq_days = 0;

      select max(iif(ok_type = 1, ok_sh_date, cast('01.01.1900' as date))),
             max(iif(ok_type = 2, ok_sh_date, cast('01.01.1900' as date)))
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date_end and
            ok_from_k_id = :k_id and
            ok_type in (1, 2) and
            ok_from_c_id = 0
      into max_date_graph, max_date_rest;

      if (max_date_graph < max_date_rest) then
      begin
        --        is_prolonged = 1;
        k_date_end = max_date_rest;
      end
      else
      begin
        --        is_prolonged = 0;
        k_date_end = max_date_graph;
      end

      kol_prolog = datediff(month, max_date_graph, k_date_end);

      curr_rate = 1;
      if (curr <> def_curr) then
      begin
        select cr_rate
        from opercurr
        where cr_from_cu_id = :curr and
              cr_date <= :rep_date_end and
              cr_date_end >= :rep_date_end rows 1
        into :curr_rate;
        if ((curr_rate is null) or (curr_rate = 0)) then
          curr_rate = 1;
        if (pr_set_rate_in_kont = 1) then
        begin
          if (curr_rate < k_index_curr_rate) then
            curr_rate = :k_index_curr_rate;
        end
        rep_active_balance_new = rep_active_balance_new * curr_rate;
        rep_active_balance = rep_active_balance * :curr_rate;
        rep_active_balance_new = cast(rep_active_balance_new as decimal(15,2));
        rep_active_balance = cast(rep_active_balance as decimal(15,2));
      end
      --  Вычисление РППУ контракта
      llc_current_amount = null;
      llc_current_percent = null;
      select (cast(lk_balance as double precision) * cast(lk_percent as double precision)), lk_percent
      from llc_main, llc_kont
      where lm_is_signed = 1 and
            lk_from_lm_id = lm_id
            --      and lk_is_signed = 1
            and
            :rep_date_end between lk_date and lk_date_end and
            lk_from_k_id = :k_id rows 1
      into llc_current_amount, llc_current_percent;

      llc_current_amount = llc_current_amount / 100;

      -- добавление поля cmc_name по задаче на jiro 1920
      cmc_name = '';
      reason_name = '';
      reason_id = 0;

      select cmc.cmc_name, f_json_str_get_value(cml.cml_add_params, 'reason', '0')
      from com_list cml, com_cfg cmc , com_action ca, com_operlist col
      where cml_from_cmc_id = cmc.cmc_id
      and cml.cml_object_kind = 9
      and cml.cml_object_id = :K_ID
      and cml.cml_last_status = 5
      and cml.cml_date = :restruct_date
      and ca.cma_from_cml_id = cml.cml_id
      and col.cmo_from_cma_id = ca.cma_id
      and col.cmo_ol_is_delete = 0
      and cmc.cmc_code = 'rest_lgota'
      rows 1
      into :cmc_name, :reason_id;

      if (reason_id > 0) then
        reason_name = (select rd_name from refdata where rd_id = :reason_id);

      pre_restruct_days = 0;
      pre_delinq_date = null;
      select oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            (:restruct_date - 1) between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into :pre_delinq_date;
      
      if (pre_delinq_date is not null) then
      begin
        pre_restruct_days = (:restruct_date - 1) - pre_delinq_date;
        if (pre_restruct_days < 0) then
          pre_restruct_days = 0;
      end

      next_pay_date = null;
      select min(sh_date)
      from schedule
      where sh_from_k_id = :k_id and
            sh_date > :restruct_date
      into :next_pay_date;

      post_restruct_days = 0;
      select coalesce(oks_delinq_day, 0)
      from oksummary
      where oks_from_k_id = :k_id and
            :rep_date_end between oks_date_begin and oks_date_end
      into :post_restruct_days;

      if (db_id <> 81001) then
        suspend;

      uuid = null;
      if (db_id = 81001) then
      begin
        for select f_uuid_cbr(f_uuid('knclient', kc_id))
            from knclient
            where kc_from_k_id = :k_id
            and kc_amount_approved > 0
            into uuid
        do
        begin
          suspend;
        end
      end
    end
  end
end
