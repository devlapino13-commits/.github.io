CREATE PROCEDURE HP_GET_INSURANCE_PARAMS(
    K_ID D_ID)
RETURNS (
    C_ID D_ID,
    K_SCORE D_INTEGER,
    PTS_DATE_BEGIN D_DATE,
    PTS_DATE_END D_DATE,
    TARIF PERC_TYPE,
    COMMISSION_AMOUNT D_CURRENCY,
    MBULAK_SHARE D_CURRENCY,
    POLICY_COMPANY_SHARE D_CURRENCY,
    TAX D_CURRENCY,
    AG_ID D_INTEGER,
    EDC_CODE D_STR_CODE)
AS
declare variable VAL D_CURRENCY;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable K_DISB_DATE D_DATE;
declare variable K_KREDLINE_END D_DATE;
declare variable N_VALUE D_INTEGER;
declare variable PR_CODE D_STR_100;
declare variable TEMP_RES D_CURRENCY;
declare variable AG_COMMIS_CARDS_OTHER D_CURRENCY;
declare variable IA_ID D_ID;
declare variable MBULAK_BASE D_CURRENCY;
begin
    select
        k.k_amount_approved, -- сумма страхования (в контексте страхования конечно же)
        k.k_kredline_end - k.k_disb_date,
        pr.pr_code,
        k.k_disb_date,
        k.k_kredline_end,
        kc.kc_from_c_id,
        coalesce(f_point(kc.kc_from_c_id, k.k_disb_date, iif(pr.pr_code containing '.ipf.', 2, 0)), 0)
    from kontract k
    inner join knclient kc on kc.kc_from_k_id = k.k_id
        and kc.kc_amount_approved > 0
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :k_id
    into
        :k_amount_approved,
        :n_value,
        :pr_code,
        :k_disb_date,
        :k_kredline_end,
        :c_id,
        :k_score;

    select first 1 pts.pts_date_begin, pts.pts_date_end
    from pointsaldo pts
    where pts.pts_from_c_id = :c_id
        and pts.pts_date_begin <= :k_disb_date
        and pts.pts_type = iif(:pr_code containing '.ipf.', 2, 0)
    order by pts.pts_date_begin desc
    into :pts_date_begin, :pts_date_end;

    if (pr_code containing '.ignore_insurance.') then
        exit;

    if (n_value <= 365) then
        n_value = 365;

    
    -- 1. Устанавливаем тариф в зависимости от продукта и баллов клиента
    if (pr_code containing '.ipf.') then -- RFCKG-5348
    begin
        tarif =
            case
                when :k_score between 0 and 19 then 19.178082 -- 7%
                when :k_score between 20 and 30 and :k_kredline_end <= dateadd(month, 6, :k_disb_date) then 10.958904 -- 4%
                when :k_score between 20 and 30 then 16.438356 -- 6%
                when :k_kredline_end <= dateadd(month, 6, :k_disb_date) then 8.219178 -- 3%
                else 13.698630 -- 5%
            end;
    end

    else if (pr_code containing '.sputnik.' or pr_code containing '.insur_base.') then
    begin
        -- Тариф по продуктам Спутник и Спутник удобный
        tarif =
            case
                when :k_score between 0 and 19 then 13.699000 -- 5%
                when :k_score between 20 and 41 then 12.329000 -- 4.5%
                else 10.958904 -- 4%
            end;
    end

    else if (pr_code not containing '.consultation.' and pr_code containing '.standart.' and pr_code containing '.ind.') then
    begin
        -- Тариф по продуктам Стандарт
        tarif =
            case
                when :k_score between 0 and 41 then 16.438000  -- 6%
                else 13.699000 -- 5%
            end;
    end
    else
        tarif = iif(f_get_const('new_loan_conditions') = 1, 5.479452, 5);

    if (pr_code containing '.standart.' and
        pr_code containing '.point_ignore.' and
        pr_code containing '.insurance.') then
    begin
        tarif = 6.849315; -- 2.5%
        n_value = 730;
    end

    -- 2. Рассчитываем параметры страхования
    -- 2.1. Определяем страховую компанию по тарифу
    if (pr_code containing '.ipf.') then
    begin
        -- СК Кыргызстан ИПФ - только для ИПФ продуктов по умолчанию
        edc_code = 'insurance.kyrgyz.ipf';
        ag_id = 7570;

    end
    else
    begin
        -- Переводим значения по типу 16.438000 в процентную ставку -> 6%
        val = tarif / 2.739;

        -- округляем по принципу: 4.5 -> 4.5; 4.01 -> 4.0; 4.99 -> 5
        val = iif(abs(val - trunc(val)) starting 0.5, round(val, 1), round(val));

        select first 1 ia.ia_id
        from insurance_agreement ia
        where :val between ia.ia_tarif_min and ia.ia_tarif_max
            and current_date between ia.ia_date_begin and ia.ia_date_end
            and ia.ia_is_active = 1
        order by ia.ia_id desc
        into :ia_id;
    
        select hp.ag_id, hp.edc_code
        from hp_insurance_get_random_company(:ia_id) hp
        into :ag_id, :edc_code;

    end


    -- 2.2. Определяем сумму комиссии
    temp_res = k_amount_approved * tarif;
    temp_res = temp_res * n_value;

    if (pr_code containing '.ipf.') then
    begin
        -- RFCKG-5348
        temp_res = round(temp_res / 100000, 2);
        tax = round(temp_res * 0.02, 2); -- НСП
        commission_amount = temp_res + tax;
    end
    else
    begin
        temp_res = temp_res * 1.02 / 100000;
        commission_amount = temp_res;
        tax = commission_amount / 102 * 2; -- НСП
    end


    -- 2.3. Определяем вспомогательную переменную
    if (pr_code containing '.ipf.') then
        -- RFCKG-5348
        ag_commis_cards_other =
            case
                when tarif <= 13.698630 then 90
                when tarif <= 16.438356 then 92
                else 93
            end;

    else if (ag_id = 7414) then
        -- RFCKG-4937 ЗАО Страховая компания "Кыргызстан"
        ag_commis_cards_other =
            case
                when tarif <= 13.699000 then 90
                when tarif <= 16.438356 then 92
                when tarif > 16.438356 then 93
                else 93
            end;

    else if (ag_id = 7130) then  -- ЗАО СК "Арсеналъ-Кыргызстан"
    begin
      ag_commis_cards_other = case
        when tarif <= 0.821918  then 30
        when tarif <= 1.095890  then 40
        when tarif <= 1.369863  then 50
        when tarif <= 1.643836  then 55
        when tarif <= 1.917808  then 60
        when tarif <= 2.191781  then 70
        when tarif <= 2.465753  then 75
        when tarif <= 13.424657 then 92
        else 93
      end;
    end

    else
        -- Все другие случаи
        select ag.ag_commis_cards_other
        from agreements ag
        where ag.ag_id = :ag_id
        into ag_commis_cards_other;    


    -- 2.4. Определяем долю Мбулак (RFCKG-5348)
    if (pr_code containing '.ipf.') then
        mbulak_base = (commission_amount - tax) * 0.90;
    else
        mbulak_base = commission_amount - tax;

    -- RFCKG-4803
    mbulak_share = mbulak_base * (ag_commis_cards_other / 100);


    -- 2.5. Определяем долю сраховой компании
    policy_company_share = commission_amount - mbulak_share;

    suspend;
end
