CREATE PROCEDURE HP_INS_UPD_INTERN(
    I_ID D_INTEGER,
    C_ID D_ID,
    REG_ID D_INTEGER,
    UN_ID D_ID,
    JL_ID D_INTEGER,
    IS_DELETE D_SMALLINT,
    DATE_BEGIN D_DATE = null,
    DATE_END D_DATE = null)
RETURNS (
    RES_TEXT D_STR_100,
    IS_ERR D_SMALLINT)
AS
declare variable date_begin_old d_date;
declare variable need_cc boolean;
declare variable msg_cc d_str_100;
declare variable fp_id d_integer;
declare variable c_name_fio d_str_100;
declare variable c_birth_date d_date;
declare variable c_pass_no d_str_code;
declare variable terr_categ_id d_integer;

begin
  is_err = 1;
  res_text = 'Произошла ошибка!';

  -- проверка RFCKG-4895
  need_cc = false;
  msg_cc = '';
  for
    select
      fp_id,
      c_name_fio,
      c_birth_date,
      c_pass_no,
      terr_categ_id
    from hp_check_client_fio_ft_fuzzy(:c_id, '', '', '', null, '', 1)
      where c_comment_id > 0
  into
    :fp_id,
    :c_name_fio,
    :c_birth_date,
    :c_pass_no,
    :terr_categ_id
  do
  begin
    need_cc = true;
    msg_cc = 'Совпадение по спискам ПФПД/ЛПД. Обратитесь в Службу комплаенс контроля!';
  end

  if (i_id = 0) then
  begin
    select is_err, res_text
    from hp_intern_ins_check(:c_id)
    into :is_err, :res_text;

    if (is_err = 0) then
    begin
      insert into interns (i_from_c_id, i_from_reg_id, i_registred_by, i_from_jl_id)
      values (:c_id, :reg_id, :un_id, :jl_id)
      returning i_id
      into :i_id;

      if (i_id > 0 and
          i_id is not null) then
      begin
        if (need_cc) then
        begin
          insert into fp_verify (
            fpv_ma_id,
            fpv_c_id,
            fpv_fp_id,
            fpv_fio,
            fpv_birthdate,
            fpv_pass_no,
            fpv_terr_categ_id)
          values (
            -:i_id,
            :c_id,
            :fp_id,
            :c_name_fio,
            :c_birth_date,
            :c_pass_no,
            :terr_categ_id);
          res_text = msg_cc;
        end else
          res_text = 'Стажер успешно зарегистрирован.';

        is_err = 0;
        execute procedure hp_intern_send_email(:i_id, 'registration');
        suspend;
        exit;
      end
    end
    else
    begin
      suspend;
      exit;
    end
  end
  else
  begin
    if (:is_delete = 1) then
    begin
      update interns
      set i_is_delete = :is_delete
      where (i_id = :i_id);

      res_text = 'Стажер успешно удален.';
      is_err = 0;
      suspend;
      exit;
    end
    else
    begin
      select i.i_date_begin
      from interns i
      where i.i_id = :i_id
      into :date_begin_old;

      update interns
      set i_from_c_id = :c_id,
          i_from_reg_id = :reg_id,
          i_registred_by = :un_id,
          i_registred_datetime = current_timestamp,
          i_is_delete = :is_delete,
          i_from_jl_id = :jl_id,
          i_date_begin = :date_begin,
          i_date_end = :date_end
      where (i_id = :i_id);

      if (need_cc and not exists(
        select *
        from fp_verify
        where fpv_ma_id = -:i_id
          and fpv_c_id = :c_id
          and fpv_is_confirm = 0)) then
      begin
        update or insert into fp_verify (
          fpv_ma_id,
          fpv_c_id,
          fpv_fp_id,
          fpv_fio,
          fpv_birthdate,
          fpv_pass_no,
          fpv_terr_categ_id,
          fpv_is_confirm,
          fpv_un_id,
          fpv_confirm_date,
          fpv_reason,
          mark_del)
        values (
          -:i_id,
          :c_id,
          :fp_id,
          :c_name_fio,
          :c_birth_date,
          :c_pass_no,
          :terr_categ_id,
          0,
          null,
          null,
          null,
          0)
         matching(fpv_ma_id);
        res_text = msg_cc;
      end else
        res_text = 'Информация о стажере была обновлена.';

      is_err = 0;
      if (:date_begin_old is null and :date_begin is not null) then
      begin
        execute procedure hp_intern_send_email(:i_id, 'hr_first_step');

        select is_err
        from hp_intern_acc(:i_id)
        into :is_err;
      end
      if (is_err <> 0) then
        res_text = 'Произошла ошибка при регистрации договора на оказание услуг.';
      suspend;
      exit;
    end
  end
end
