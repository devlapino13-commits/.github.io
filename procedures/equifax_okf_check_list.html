CREATE PROCEDURE EQUIFAX_OKF_CHECK_LIST(
    IS_ACTIVE D_SMALLINT = 0)
RETURNS (
    K_ID D_ID,
    SUM_TOTAL D_CURRENCY,
    CRED_ACTIVE D_INTEGER,
    KS_STATE D_INTEGER,
    EO_LAST_UPDATE D_DATE)
AS
declare variable REP_DATE_END D_DATE;
declare variable DO_SUSPEND D_INTEGER;
declare variable BASE_K_ID D_ID;
begin
    rep_date_end = current_date;

    -- беру все кредиты с активным долгом по внесистемке на сейчас, выданные после 01.07.2014
    for select okf.oko_from_k_id,
           sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc) as sum_total
    from operkroff okf
    inner join kontstate ks on ks.ks_from_k_id = okf.oko_from_k_id
    where okf.oko_ol_is_active = 1
    and okf.oko_ol_date <= :rep_date_end
    and ks.ks_state = 1
    and ks.ks_date_begin >= '01.07.2014'
    and ks.ks_ol_is_delete = 0
    group by okf.oko_from_k_id
    having sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc) > 0
    order by  okf.oko_from_k_id
    into :k_id, :sum_total
    do
    begin
        base_k_id = null;

        select kont.k_base_k_id
        from kontract kont
        where kont.k_id = :k_id
        into :base_k_id;

        if (base_k_id is null) then  -- инетресуют только кредиты не затронутые амнистией
        begin
            cred_active = 1;
    
            select t.eo_cred_active, t.eo_last_update
            from equifax_okf t
            where t.eo_from_k_id = :k_id
            order by t.eo_cred_active
            rows 1
            into :cred_active, :eo_last_update;
    
            select ks.ks_state
            from kontstate ks
            where ks.ks_ol_is_delete = 0
            and current_date between ks.ks_date_begin and ks.ks_date_end
            and ks.ks_from_k_id = :k_id
            into :ks_state;
    
            if (is_active = 0) then --интересуют только закрыте кредиты
            begin
                if (cred_active = 0) then
                    suspend;
            end
            else
            begin
                if (cred_active <> 0) then
                    suspend;    
            end
        end
    end
end
