CREATE PROCEDURE AAREP_RATING_DELINQ_ANALYSI(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(60),
    PARENT_NAME VARCHAR(60),
    OKRUG_NAME VARCHAR(60),
    KF_OFFICER_NAME VARCHAR(100),
    AMOUNT_ACTIVE_BEGIN CURR_TYPE,
    AMOUNT_ACTIVE_END CURR_TYPE,
    OS_REPAY CURR_TYPE,
    OS_WOFF CURR_TYPE,
    RATING CURR_TYPE)
AS
declare variable constanta_date date;
declare variable kal_new_reg_id integer;
declare variable kal_new_kf_officer id_type;
begin
  constanta_date = '31.12.2012';
  for select kal_new_reg_id, kal_new_kf_officer, sum(amount_active_begin), sum(amount_active_end), sum(os_repay),
             sum(os_woff)
      from aarep_rating_delinq_analysi_tmp(:rep_date_begin, :rep_date_end, :constanta_date), kontract, knaktlist
      where ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :constanta_date between kal_date and kal_date_end
      group by kal_new_reg_id, kal_new_kf_officer
      into :kal_new_reg_id, :kal_new_kf_officer, :amount_active_begin, :amount_active_end, :os_repay, :os_woff
  do
  begin
    reg_name = 0;
    parent_name = 0;
    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
    from region r1, region r2, region r3
    where r1.reg_unicode = :kal_new_reg_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into reg_name, parent_name, okrug_name;

    select c_name_fio
    from clients
    where c_id = :kal_new_kf_officer
    into kf_officer_name;

    rating = (:amount_active_begin - :amount_active_end - :os_woff) / :amount_active_begin * 100;
    rating = round(rating, 2);
    suspend;
  end
end
