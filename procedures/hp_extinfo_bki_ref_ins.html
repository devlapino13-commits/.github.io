CREATE PROCEDURE HP_EXTINFO_BKI_REF_INS(
    UN_ID D_ID NOT NULL,
    K_ID D_ID NOT NULL,
    IS_KI D_BOOL = 0)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_BOOL,
    SSP_IS_SEND D_BOOL,
    KI_IS_SEND D_BOOL)
AS
declare variable ma_id d_integer;
declare variable c_id d_id;
declare variable k_kontract_no d_str_20;
begin
  -- отправка запросов ССП/КИ при пересчете ПДН после 60 дней после выдачи (RFCRU-4309)
  -- используется в форме "ПДН - расчет в связи с неисполнением цели по рефинансированию"

  res_text = '';
  res_is_error = 0;
  ssp_is_send = 0;
  ki_is_send = 0;
  k_kontract_no = '';

  select k_from_ma_app_id,
         k_kontract_no
  from kontract
  where k_id = :k_id
  into ma_id,
       k_kontract_no;

  for select mac_from_c_id
      from miniappclient
      where mac_amount > 0 and
            mac_from_ma_id = :ma_id
      into c_id
  do
  begin
    if (exists(select *
               from extinfolist
               inner join extinfocfg on eic_id = eil_from_eic_id
               where eic_code = 'bki_ssp_40' and -- запрос ССП
                     eil_status = 2 and -- запрос успешный
                     eil_date >= current_date - 1 and -- дата запроса сегодня или вчера
                     eil_from_c_id = :c_id)) then
    begin
      if (res_text <> '') then
        res_text = res_text || ascii_char(10);
      res_text = res_text || k_kontract_no || ': запрос ССП не отправлен, т.к. есть свежий успешный запрос';
      res_is_error = 1;
    end
    else if (is_ki = 0) then -- была нажата кнопка "Отправить запрос ССП"
    begin
      if (not exists(select *
                     from pdn_ref_recalclist
                     where prr_k_id = :k_id and
                           prr_ref_ext_cnt_active > 0)) then
      begin
        if (res_text <> '') then
          res_text = res_text || ascii_char(10);
        res_text = res_text || k_kontract_no || ': запрос ССП не отправлен, т.к. отсутствуют активные внешние рефы';
        res_is_error = 1;
      end
    end

    if (res_is_error = 0) then
    begin
      begin
        execute procedure hp_extinfo_insert_bki_cfg('bki_ssp_40', c_id, f_objkind('miniapp', 'id'), ma_id, 1, un_id); -- отправка запроса ССП
        ssp_is_send = 1;

        when any do
        begin
          res_text = 'При отправке запроса ССП произошла ошибка (hp_extinfo_bki_ref_ins)';
          res_is_error = 1;
          suspend;
          exit;
        end
      end
    end
  end

  if (is_ki = 1) then -- была нажата кнопка "Отправить запрос ССП/КИ"
  begin
    begin
      if (exists(select *
                 from extinfolist
                 inner join extinfocfg on eic_id = eil_from_eic_id
                 where eic_code = 'equifax_40' and -- запрос КИ
                       eil_status = 2 and -- запрос успешный
                       eil_date >= current_date - 1 and -- дата запроса сегодня или вчера
                       eil_from_c_id = :c_id)) then
      begin
        res_text = res_text || ascii_char(10) || k_kontract_no || ': запрос КИ не отправлен, т.к. есть свежий успешный запрос';
        res_is_error = 1;
        suspend;
        exit;
      end
      else
      begin
        execute procedure hp_extinfo_insert('equifax_40', f_objkind('kontract', 'id'), k_id, 1, un_id); -- отправка запроса КИ
        ki_is_send = 1;
      end
    end

    when any do
    begin
      res_text = 'При отправке запроса КИ произошла ошибка (hp_extinfo_bki_ref_ins)';
      res_is_error = 1;
      suspend;
      exit;
    end
  end

  suspend;
end
