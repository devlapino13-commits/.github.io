CREATE PROCEDURE PDN_INFO_ACTION(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID)
RETURNS (
    KC_ID D_ID,
    C_ID D_ID,
    K_ID D_ID,
    ACL_ID D_ID,
    SURVEY_JSON D_STR_8100)
AS
declare variable ma_date d_date;
begin
    survey_json = '{}';

    if (:obj_kind not in ('miniappclient_income')) then
    begin
        suspend;
        exit;
    end

    select kc.kc_id,
           mac.mac_from_c_id,
           ma.ma_date_app,
           k.k_id
    from miniappclient mac
    join miniapp ma on ma.ma_id = mac.mac_from_ma_id
    join kontract k on k.k_from_ma_app_id = mac.mac_from_ma_id
    join knclient kc on kc.kc_from_k_id = k.k_id
                    and kc.kc_from_c_id = mac.mac_from_c_id
    where mac.mac_id = :obj_id
      and ma.ma_date_end = '31.12.9999'
      and ma.ma_is_active = 1
    into :kc_id,
         :c_id,
         :ma_date,
         :k_id;

    select acl_id
    from hp_actionlist_kontract_get(:k_id)
    into :acl_id;

    select ki.ki_survey
    from kk_income ki
    where ki.ki_from_k_id = :k_id
      and ki.ki_from_c_id = :c_id
    into :survey_json;

    if (coalesce(survey_json, '{}') = '{}') then
    begin
        select maci.maci_survey
        from miniappclient_income maci
        where maci.maci_from_mac_id = :obj_id
        into :survey_json;
    end


    if (coalesce(:survey_json, '{}') = '{}') then
    begin
        select ki.ki_survey
        from kk_income ki
        join knclient kc on kc.kc_from_k_id = ki.ki_from_k_id and ki.ki_from_c_id = kc.kc_from_c_id
        join knclient_pdn kcp on kcp.kc_id = kc.kc_id
        where kc.kc_amount_approved > 0
          and kc.kc_from_c_id = :c_id
          and kc.kc_from_k_id <> :k_id
          and kcp.kc_date between f_date_month_begin(:ma_date) and f_date_month_end(:ma_date)
          and coalesce(ki.ki_survey, '{}') <> '{}'
        order by kcp.kc_date
        rows 1
        into :survey_json;
    end

    survey_json = coalesce(:survey_json, '{}');
    suspend;
end
