CREATE PROCEDURE CM_RU_0066_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable KF_CLIENT_PASS_NO varchar(30);
declare variable KC_FROM_C_ID ID_TYPE;
declare variable C_CITIZEN integer;
declare variable RD_CODE varchar(255);
declare variable CDS_DOC_DATE_BEGIN date;
declare variable CDS_DOC_DATE_END date;
declare variable K CURR_TYPE;
declare variable PDT_CODE varchar(100);
begin
  res_error_is_critical = 1;
  kf_disb_date = null;
  select k_disb_date
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id and
        pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.|.LOYAL.|.blago.|.uspeh.|.RELIABLE.|.RELIABLE_CLIENT_PLUS.|.PAR.LOYAL.)%'
  into kf_disb_date;
  if (kf_disb_date is null) then
    exit;
  for select c_id, c_name_fio, c_pass_no, kc_amount_approved, c_citizen
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into kc_from_c_id, res_info_text, kf_client_pass_no, kc_amount, c_citizen
  do
  begin
    select rd_code
    from refdata
    where rd_id = :c_citizen
    into rd_code;
    if (rd_code in ('Citizen.KG', 'Citizen.TJ', 'Citizen.UZ', 'Citizen.KZ')) then
    begin
      cds_doc_date_begin = null;
      cds_doc_date_end = null;
      select cds_doc_date_begin, cds_doc_date_end
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :kc_from_c_id and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            cds_is_delete = 0 and
            pdt_code = 'magration_card' and
            coalesce(cds_doc_blob_1, 0) > 0 and
            (:kf_disb_date between cds_doc_date_begin and cds_doc_date_end or
            cds_doc_date_end >= '15.03.20')
      order by cds_doc_date_end desc
      rows 1
      into cds_doc_date_begin, cds_doc_date_end;
      if (cds_doc_date_begin is null) then
      begin
        --      если миграционной карты изначально не было то тогда необходимо ВНЖ или РВП
        if (not exists(select *
                       from clientdocs, passkind, passdoctype
                       where cds_from_c_id = :kc_from_c_id and
                             cds_from_pk_id = pk_id and
                             pk_from_pdt_id = pdt_id and
                             cds_is_delete = 0 and
                             :kf_disb_date between cds_doc_date_begin and cds_doc_date_end and
                             coalesce(cds_doc_blob_1, 0) > 0 and
                             pdt_code in ('registration_rvp', 'registration_vnj'))) then
        begin
          res_error_text = 'Если не было миграционной карты то необходимо РВП или ВНЖ';
          suspend;
        end
      end
      else
      begin
        if (not exists(select *
                       from clientdocs, passkind, passdoctype
                       where cds_from_c_id = :kc_from_c_id and
                             cds_from_pk_id = pk_id and
                             pk_from_pdt_id = pdt_id and
                             cds_is_delete = 0 and
                             :kf_disb_date between cds_doc_date_begin and cds_doc_date_end and
                             coalesce(cds_doc_blob_1, 0) > 0 and
                             pdt_code containing ('registration_'))) then
        begin
          res_error_text = 'Необходимо прикрепить временную регистрацию!';
          suspend;
        end
        cds_doc_date_end = coalesce(cds_doc_date_end, cast('01.01.1990' as date));
        if (:kf_disb_date - cds_doc_date_begin > 90 and
            cds_doc_date_end <= :kf_disb_date) then
        begin
          --        если миграционка просроченная и не продлен то проверяется на ниже следующие документы
          k = 0;
          for select pdt_code
              from clientdocs, passkind, passdoctype
              where cds_from_c_id = :kc_from_c_id and
                    cds_from_pk_id = pk_id and
                    pk_from_pdt_id = pdt_id and
                    cds_is_delete = 0 and
                    coalesce(cds_doc_blob_1, 0) > 0 and
                    (:kf_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, cast('01.01.1900' as date)) --
                    or
                    (:kf_disb_date >= cds_doc_date_begin and
                    pdt_need_date_end = 0)) and
                    (pdt_code in ('registration_vnj', -- Вид на жительство (ВНЖ)
                              'registration_rvp', -- Разрешение на временное проживание (РВП)
                              'work_patent', 'kvit_patent', 'work_permit', -- РНР - разрешение на работу
                              'work_permit_WTO', -- РНР членов ВТО
                              'application_citizenship_rassian_lang_speaker', -- Заявление о гражданстве носителя русского языка
                              'request_VKS', -- Ходотайство работодателя на привлечение ВКС
                              'work_permit_application_extension_VKS', -- Заявление работодателя о продлении РНР для ВКС
                              'job_certificate',
                                  -- Справка с места работы
                                  'application_rezidence_permit_vnj' -- Заявление о выдаче ВНЖ
                                  ) or
                    (:rd_code in ('Citizen.KG', 'Citizen.KZ') and
                    pdt_code containing 'job_agreement'))
              into pdt_code
          do
          begin
            if (pdt_code in ('work_patent', 'kvit_patent')) then
              k = k + 0.5;
            else
              k = k + 1;
          end
          if (k < 1) then
          begin
            res_error_text = 'Cрок пребывания в РФ по миграц. карте более 90 дней, требуется документ о продлении пребывания в РФ';
            if (rd_code in ('Citizen.KG', 'Citizen.KZ') and
                cds_doc_date_end between '15.03.2020' and '15.03.2021') then -- добавлено условие по запросу от Зары
              res_error_is_critical = 0;
            if (rd_code in ('Citizen.UZ', 'Citizen.TJ') and
                cds_doc_date_end between '15.03.2020' and '15.06.2021') then -- добавлено условие по запросу от Зары
              res_error_is_critical = 0;
            suspend;
          end
        end
      end
    end
  end
end
