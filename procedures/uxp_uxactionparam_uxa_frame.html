CREATE PROCEDURE UXP$UXACTIONPARAM_UXA_FRAME(
    UX_UUID D_UUID,
    FR_UUID D_UUID,
    PARAM_KIND D_SMALLINT,
    UX_UUID_CODE D_STR_CODE = null,
    FR_UUID_CODE D_STR_CODE = null)
RETURNS (
    UAP_UA_UUID D_UUID,
    UAP_UXA_UFP_UUID D_UUID,
    UAP_UXA_UFP_CODE D_STR_CODE,
    UAP_OWNER_UFP_UUID D_UUID,
    UAP_OWNER_UFP_CODE D_STR_CODE,
    UAP_KIND D_SMALLINT,
    UAP_DEFAULT D_STR_255,
    UAP_USE_DEFAULT D_SMALLINT,
    UAP_UXA_UFP_CAPTION D_STR_255,
    UAP_LOCKED D_BOOL,
    RES_EDITOR D_STR_CODE,
    RES_STATE D_INTEGER,
    RES_READONLY D_BOOL,
    RES_UXA_UUID D_UUID,
    RES_CATEGORY_1 D_STR_255,
    UF_DEFAULT D_STR_100,
    UAP_EXEC_PARAM_ID D_SMALLINT)
AS
declare variable uf_name d_str_100;
declare variable uf_pk_field d_smallint;
declare variable ufp_is_output d_bool;
declare variable uf_field_name d_str_code;
declare variable ux_kind d_str_code;
declare variable uxa_uuid d_uuid;
declare variable ua_uuid d_uuid;
declare variable uf_is_param d_bool;
declare variable ufp_lookup d_uuid;
declare variable ufp_lookup_code d_str_code;
declare variable us_code d_str_code;
declare variable prv_lookup d_uuid;
declare variable ufr_detail_field d_str_code;
declare variable frame_us_uuid d_uuid;
declare variable us_autoincfield d_str_code;
declare variable us_uuid d_uuid;
declare variable uxa_kind d_smallint;
declare variable res_source_field d_str_code;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code, 0);

  if (coalesce(fr_uuid, '') = '') then
    fr_uuid = uxf$uniq(fr_uuid_code);

  uxa_kind = 0;

  if (ux_uuid is not null) then
  begin
    ux_kind = uxf$code(ux_uuid, 'kind');
  
    if (ux_kind = 'uxaction') then
    begin
      ua_uuid = ux_uuid;

      select uxa_uuid, uxa_kind
      from uxp$uxalias_info(:ua_uuid)
      into uxa_uuid, uxa_kind;
    end
    else
    if (ux_kind = 'uxalias') then
      uxa_uuid = ux_uuid;
  end

  uap_ua_uuid = ua_uuid;
  res_state = 0;

  res_uxa_uuid = uxa_uuid;

  select us_uuid, us_autoincfield
  from ux$frame, ux$source
  where ufr_uuid = :fr_uuid and
        ufr_us_uuid = us_uuid
  into frame_us_uuid, us_autoincfield;

  prv_lookup = null;
  for select ufp_uuid, uf_name, uf_field_name, ufp_editor, uf_pk_field, uf_is_param, ufp_parent, uf_default,
             ufr_detail_field, us_uuid,
             iif(uf_is_param = 1 or
                (ufp_parent is null and uf_pk_field <> 0), 0, 1)
  from ux$frparam, ux$field, ux$source, ux$frame
  where ufr_uuid = :fr_uuid and
        ufp_ufr_uuid = ufr_uuid and
        ufp_uf_uuid = uf_uuid and
        uf_owner = us_uuid
  order by uf_is_param desc, uf_position
  into uap_uxa_ufp_uuid, uf_name, uf_field_name, res_editor, uf_pk_field, uf_is_param, ufp_lookup, uf_default,
       ufr_detail_field, us_uuid, ufp_is_output
  do
  begin
    if (param_kind = 0 and ufp_is_output <> 0) then
      Continue;

    if (param_kind = 1 and ufp_is_output <> 1) then
      Continue;

    if (param_kind = 0 and -- Входящие параметры
        uxa_kind = 2 and   -- Действие на создание
        frame_us_uuid = us_uuid and
        uf_field_name = us_autoincfield) then
      Continue;

    res_category_1 = '';
    if (ufp_lookup is not null) then
    begin
      if (uf_is_param = 1) then
      begin
        if (prv_lookup is distinct from ufp_lookup) then
        begin
          ufp_lookup_code = '';
          select uf_name
          from ux$frparam, ux$field
          where ufp_uuid = :ufp_lookup and
                ufp_uf_uuid = uf_uuid
          into ufp_lookup_code;

          us_code = '';
          select us_code
          from ux$frlookup, ux$source
          where ufl_ufp_uuid = :ufp_lookup and
                ufl_us_uuid = us_uuid
          into :us_code;
        end
        res_category_1 = 'Выборка ' || Upper(ufp_lookup_code) || ', параметры ' || upper(us_code);
      end
    end else
    begin
      if (uf_is_param = 1) then
        res_category_1 = ' Параметры';
      else
        res_category_1 = ' Поля';
    end

    prv_lookup = ufp_lookup;
    uap_uxa_ufp_code = uuid_to_char(uap_uxa_ufp_uuid);

    uap_kind = ufp_is_output;

    -- если в форме один фрейм, не показываем источник чтобы текст был покороче
    uap_uxa_ufp_caption = uf_field_name;

    if (uf_pk_field <> 0) then
      uap_uxa_ufp_caption = uap_uxa_ufp_caption || ' (pk' || uf_pk_field || ')';

    if (uap_uxa_ufp_caption not containing uf_name) then
      uap_uxa_ufp_caption = uap_uxa_ufp_caption || '   - ' || uf_name;

    select res_def_value, res_owner_ufp_uuid, res_owner_ufp_code, res_source_field,
           res_use_default, res_state, res_locked, res_readonly, res_exec_param_id
    from uxp$uxfrparam_value(:uap_uxa_ufp_uuid, :uxa_uuid, :ua_uuid)
    into :uap_default, uap_owner_ufp_uuid, uap_owner_ufp_code, res_source_field,
         :uap_use_default, :res_state, :uap_locked, :res_readonly, uap_exec_param_id;

    if (uap_owner_ufp_uuid is null and res_source_field <> '') then
      uap_owner_ufp_code = res_source_field;

    if (ufr_detail_field = uf_field_name) then
    begin
      uap_locked = 1;
      res_readonly = 1;
      res_state = -5;
    end
    suspend;
  end
end
