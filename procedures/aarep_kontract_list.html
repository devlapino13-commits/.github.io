      select max(sh_date), max(iif(:k_kolich_lgota + 1 = sh_id, sh_os + sh_perc, 0))
      from schedule
      where sh_from_k_id = :k_id
      into k_kont_last_date2, sum_month;

      select cl1.c_id
      from knclient kc1
      inner join clients cl1 on cl1.c_id = kc1.kc_from_c_id
      where kc1.kc_from_k_id = :k_id rows 1
      into :c_id;

      cl_points_date = case report_id
                         when 2 then k_disb_date
                         when 3 then report_date_end
                         when 4 then rep_date_closed
                       end;

      if (db_id = 81001 and report_id = 10) then
        cl_points = f_point(:cl_cashier_id, :report_date_end); -- RFCRU-2326
      else if (db_id = 81001 and report_id = 2) then
        cl_points = f_point_get(9, k_id, cl_cashier_id, cl_points_date); -- RFCRU-5380
      else
        cl_points = f_point(:c_id, :cl_points_date);

      select coalesce(kd.kd_percent, 0)
      from kontdata kd
      where kd.kd_from_k_id = :k_id and
            kd.kd_ol_date <= :report_date_end
      order by kd_ol_date desc
      rows 1
      into :kd_percent;

      refinance_type = ''; -- RFCRU-4160
      if (db_id = 81001 and k_refund = 'да') then
      begin
        select count(*)
        from kontrefinance
        where kr_from_ma_id = :ma_id and
              kr_from_k_id is not null
        into cnt_internal;

        select count(*)
        from kontrefinance
        where kr_from_ma_id = :ma_id and
              kr_from_k_id is null
        into  cnt_external;

        refinance_type = case
                            when cnt_internal > 0 and cnt_external > 0 then 'Внутреннее/Внешнее'
                            when cnt_internal > 0 then 'Внутреннее'
                            when cnt_external > 0 then 'Внешнее'
                            else ''
                         end;
      end
      client_phone = f_cont_info('clients', :c_id, 'cell_main', 'ok');

      row_counter = row_counter + 1;
      row_no = row_counter;

      suspend;
    end
  end ----end of 'first for"
