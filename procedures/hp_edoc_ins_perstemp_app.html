CREATE PROCEDURE HP_EDOC_INS_PERSTEMP_APP(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_BOOL)
AS
declare variable pte_edit_kind d_smallint;
declare variable asp_status d_str_code;
declare variable edc_name d_str_100;
begin
  res_is_error = 0;
  edl_id = 0;

  select pte_edit_kind
  from perstemp
  where pte_from_c_id = :c_id
  into pte_edit_kind;

  asp_status = f_edoc_status('asp.agreement', c_id, current_date);
  if (asp_status <> 'signed') then
  begin
    select edl_id, res_text
    from hp_edoc_ins_te_asp_agreement(:c_id, :un_id)
    into :edl_id, :res_text;
  end else
  begin
    if (pte_edit_kind = 1) then
    begin
      select edl_id, res_text
      from hp_edoc_ins('pdata.application', 0, :c_id, :un_id)
      into :edl_id, :res_text;
    end else
    begin
      select edl_id, res_text
      from hp_edoc_ins('pdata.soglasheniye', 0, :c_id, :un_id)
      into :edl_id, :res_text;
    end
  end

  if (res_text <> '') then
    res_is_error = 1;
  else
  if (edl_id <> 0) then
  begin
    select edc_name
    from edoc_list, edoc_cfg
    where edl_id = :edl_id and
          edl_from_edc_id = edc_id
    into edc_name;

    res_text = 'Электронный документ успешно создан: ' || coalesce(edc_name, '');
  end

  suspend;
end
