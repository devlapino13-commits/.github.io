CREATE PROCEDURE HP_ANKETA_INCOME_OUTCOME_WRITE(
    K_ID D_ID,
    MA_ID D_INTEGER,
    ENTER_CA_DATE D_DATE,
    COMENT_TEXT D_STR_255,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_255,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable mac_id d_integer;
declare variable caf_kpd_err d_str_255;
declare variable ca_id d_integer;
declare variable ca_date d_date;
begin
  --команад выполнить для модуля анкеты анализ доходов/расходов. RFCKG-3378
  res_info_text = '';
  res_error_text = '';



  --Получаем клиентов контракта
  for select kc.kc_from_c_id
  from knclient kc
  inner join clients c on c.c_id = kc.kc_from_c_id
  where kc.kc_from_k_id = :k_id
  into c_id
  do
  begin
    --Получаем клиента в заявке
    select mac.mac_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id
      and  mac.mac_from_c_id = :c_id
    into mac_id;

    select caf.caf_kpd_err
    from cfp_access_form caf
    where caf.caf_from_c_id = :c_id
      and caf.caf_mac_id = :mac_id
    into  caf_kpd_err;

    select ca.ca_id, ca.ca_date
    from cfp_anketa ca
    where ca.ca_object_kind = f_objkind('miniappclient', 'id')
      and ca.ca_object_id = :mac_id
      and ca.ca_from_c_id = :c_id
    into ca_id, ca_date;

    --Если дата анкеты которая пришла с формы отличается от даты в БД то меняем дату в таблице CFP_ANKETA
    if(enter_ca_date <> ca_date and coalesce(enter_ca_date, '') <> '' and ca_id is not null) then
    begin
      update cfp_anketa ca
        set ca.ca_date = :enter_ca_date,
            ca.ca_from_un_id = :un_id,
            ca.ca_datetime = current_timestamp --Также обнавляем время сохранения
      where ca.ca_id = :ca_id;
    end

    coment_text = iif (coalesce(:coment_text, '') = '', coalesce(:caf_kpd_err, ''), :coment_text);

    update or insert into cfp_access_form (caf_kpd_err, caf_datetime, caf_from_c_id, caf_mac_id, caf_from_ca_id)
    values (:coment_text, current_timestamp,:c_id, :mac_id, :ca_id)
    matching(caf_from_c_id, caf_mac_id, caf_from_ca_id);
    res_info_text = 'Готово!';
    suspend;
  end
end
