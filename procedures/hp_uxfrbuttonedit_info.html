CREATE PROCEDURE HP_UXFRBUTTONEDIT_INFO(
    UFP_CODE D_STR_CODE,
    UXA_CODE D_STR_CODE)
RETURNS (
    RES_UXA_CODE D_STR_CODE,
    RES_UXA_NAME D_STR_100,
    RES_FRAME_ID_FIELD D_STR_CODE,
    RES_FRAME_TEXT_FIELD D_STR_CODE,
    RES_UA_NAME D_STR_100)
AS
declare variable fr_code d_str_code;
begin
  select ufp_ufr_code, ufp_origin_name
  from uxfrparam
  where ufp_code = :ufp_code
  into fr_code, res_frame_id_field;

  select uxa_name
  from uxalias
  where uxa_code = :uxa_code
  into res_uxa_name;

  res_frame_text_field = '';
  res_uxa_code = '';
  res_ua_name = '';

  select ua_uxa_code, ua_usr_text_field
  from uxfrparam, uxaction
  where ufp_code = :ufp_code and
        ufp_ufr_code = ua_ux_code and
        ufp_origin_name = ua_usr_id_field
  into res_uxa_code, res_frame_text_field;

  if (res_uxa_code = uxa_code) then
  begin
    select ua_name
    from uxaction
    where ua_ux_code = :fr_code and
          ua_uxa_code = :res_uxa_code and
          ua_usr_id_field = :res_frame_id_field
    into res_ua_name;
  end else
  begin
    res_frame_text_field = '';
    res_uxa_code = uxa_code;
    res_ua_name = res_uxa_name;
  end

  suspend;
end
