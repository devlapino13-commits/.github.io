CREATE PROCEDURE AAREP_KONTRACT_FINE_TO_OS_NBKR(
    REP_DATE_END D_DATE)
RETURNS (
    K_ID D_ID,
    K_FACT_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    REGION_NAME D_STR_64,
    K_KONTRACT_NO D_STR_20,
    TOTAL_FINE D_CURRENCY,
    PAYED_FINE D_CURRENCY,
    SH_CALC_FINE D_CURRENCY,
    PERC_OF_AMOUNT_APPROVED D_CURRENCY)
AS
declare variable KS_STATE D_INTEGER;
begin
    -- Отчет отразит все кредиты где размер неустойки (штрафов, пени) более 10% от суммы кредита.
    -- Максимальный размер неустойки (штрафов, пени), начисленной за весь период действия кредита, не должен превышать 10 процентов от суммы выданного кредита.
    -- (RFCKG-3232)
    if (f_get_const('active_fine_accrual') = 0) then
        exit;

    for select
              ks.ks_from_k_id,
              ks.ks_date_begin,    --дата выдачи займа,
              k.k_amount_approved, --Сумма займа
              reg.reg_name,        --Офис
              k.k_kontract_no,      --Номер договора
              ks_state
    from kontstate ks
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join region reg on reg.reg_unicode = ks.ks_region_id
    where ks.ks_ol_is_delete = 0 -- говорит о том что операция не удалена
      and ks.ks_state = 1 -- говорит о том, что кредит активный
      and :rep_date_end between ks.ks_date_begin and ks.ks_date_end
    union all
    select
              ks.ks_from_k_id,
              k.k_fact_disb_date,  --дата выдачи займа,
              k.k_amount_approved, --Сумма займа
              reg.reg_name,        --Офис
              k.k_kontract_no,      --Номер договора
              ks_state
    from kontstate ks
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join region reg on reg.reg_unicode = ks.ks_region_id
    where ks.ks_ol_is_delete = 0 -- говорит о том что операция не удалена
      and ks.ks_state = 2 -- говорит о том, что заем закрыт
      and ks.ks_date_begin between :rep_date_end -7 and :rep_date_end     --кр. закрытые 7 дней
    into  k_id,
          k_fact_disb_date,
          k_amount_approved,
          region_name,
          k_kontract_no,
          ks_state
    do
    begin
        payed_fine = 0;
        sh_calc_fine = 0;
        total_fine = 0;
        perc_of_amount_approved = 0;

        select -sum(ok.ok_fine)  --получение погаш. пени
        from operkred ok
        where ok.ok_from_k_id = :k_id
          and ok.ok_type = 4
          and ok.ok_ol_is_active = 1 -- операция активна
        into :payed_fine;

        payed_fine = coalesce(payed_fine, 0);

        if (ks_state = 1) then
        begin
            select sum(sh_calc_fine) -- накопленная пеня на дату
            from aarep_make_zad_list(:k_id, :rep_date_end, 0)
            into sh_calc_fine;
    
            sh_calc_fine = coalesce(sh_calc_fine, 0);
        end

        total_fine = sh_calc_fine + payed_fine;
        perc_of_amount_approved = (k_amount_approved / 100) * 10;

        if(total_fine > perc_of_amount_approved) then
            suspend;
    end
end
