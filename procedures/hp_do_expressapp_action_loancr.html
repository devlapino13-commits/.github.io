CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_LOANCR(
    EXA_ID D_INTEGER)
RETURNS (
    RES_STATUS D_SMALLINT)
AS
declare variable exa_from_ma_id d_integer;
declare variable exa_from_un_id id_type;
declare variable k_id id_type;
declare variable k_status d_smallint;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable exa_date d_date;
declare variable eic_id d_integer;
begin
  res_status = 0;
  exa_from_ma_id = null;
  select exa_from_ma_id, exa_from_k_id, exa_from_un_id, exa_date
  from expressapp
  where exa_id = :exa_id
  into exa_from_ma_id, k_id, exa_from_un_id, exa_date;

  if (res_status = 0 and k_id is not null and k_status <= -1) then
    res_status = -1;

  if (res_status = 0 and exa_from_ma_id is null) then
    res_status = -1;

  if (res_status = 0 and k_id is null) then
  begin
    k_id = null;
    select k_id, k_status from kontract
    where k_from_ma_app_id = :exa_from_ma_id
    into k_id, k_status;

    if (k_id is not null and k_status >= -1) then
    begin
      if (k_status = -1) then res_status = -1;
    end else
    begin
      select k_id from make_new_kontract(:exa_from_ma_id, 0, :exa_from_un_id)
      into :k_id;
    end

    if (k_id is not null) then
    begin
      update expressapp
      set exa_from_k_id = :k_id,
          exa_from_edl_id = null  -- обнуляем, чтобы в следующих обработках не мешала
      where exa_id = :exa_id;
    end
  end

  -- После создания договора, если это тестовая база то фиксируем что ответы от внешних источников получены
  if (res_status = 0 and
      k_id is not null and
      not exists(select * from hp_db_is_actual)) then
  begin
    for select f_objkind('miniapp', 'id'), :exa_from_ma_id
    from rdb$database
    union
    select f_objkind('kontract', 'id'), :k_id
    from rdb$database
    union
    select f_objkind('clients', 'id'), kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    union
    select f_objkind('clients', 'id'), mac_from_c_id
    from miniappclient
    where mac_from_ma_id = :exa_from_ma_id and
          mac_amount > 0
    into obj_kind, obj_id
    do
    begin
      -- Получаем все конфигурации, по которым создавался запрос
      for select distinct eil_from_eic_id
      from extinfolist
      where eil_date >= :exa_date and
            eil_object_kind = :obj_kind and
            eil_object_id = :obj_id
      into eic_id
      do
      begin
        -- Если это тестовая база, фиксируем что ответы получены
        update extinfolist
        set eil_status = 2
        where eil_date >= :exa_date and
              eil_object_kind = :obj_kind and
              eil_object_id = :obj_id and
              eil_from_eic_id = :eic_id and
              eil_status <> 2;
      end
    end
  end


  suspend;
end
