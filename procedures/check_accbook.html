CREATE PROCEDURE CHECK_ACCBOOK(
    DT VARCHAR(20),
    KT VARCHAR(20))
RETURNS (
    CAN_SAVE SMALLINT)
AS
begin
  can_save = 0;
  for select distinct ab_from_a_id
      from accbook
      where (:dt between ab_dt_begin and ab_dt_end || '9') and
            (:kt between ab_kt_begin and ab_kt_end || '9') and
            ab_is_delete = 0 rows 1
      into :can_save
  do
  begin
    can_save = 1;
  end
  suspend;
end
