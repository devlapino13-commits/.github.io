CREATE PROCEDURE HP_UXFIELD_GET_FK(
    TABLE_NAME D_STR_CODE,
    FIELD_NAME D_STR_CODE)
RETURNS (
    FK_TABLE D_STR_CODE,
    FK_FIELD D_STR_CODE)
AS
begin
  table_name = upper(trim(table_name));
  field_name = upper(trim(field_name));

  fk_table = '';
  fk_field = '';
  select trim(lower(c.rdb$relation_name)), trim(lower(d.rdb$field_name))
  from rdb$ref_constraints b,
       rdb$relation_constraints a,
       rdb$relation_constraints c,
       rdb$index_segments d,
       rdb$index_segments e,
       rdb$indices i
  where (a.rdb$constraint_type = 'FOREIGN KEY') and
        (a.rdb$constraint_name = b.rdb$constraint_name) and
        (b.rdb$const_name_uq = c.rdb$constraint_name) and
        (c.rdb$index_name = d.rdb$index_name) and
        (a.rdb$index_name = e.rdb$index_name) and
        (a.rdb$index_name = i.rdb$index_name) and
        (a.rdb$relation_name = :table_name) and
        (e.rdb$field_name = :field_name)
  rows 1
  into :fk_table, :fk_field;

  if (fk_table is null) then
    fk_table = '';
  if (fk_field is null) then
    fk_field = '';

  suspend;
end
