CREATE PROCEDURE AAREP_INDIV_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(20),
    K_INDEX_CURR_CODE VARCHAR(3),
    K_DISB_DATE DATE,
    KC_AMOUNT_APPROVED CURR_TYPE,
    K_KREDLINE_END DATE,
    PR_NAME VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    SROK DECIMAL(15,1),
    K_KONT_BALANCE CURR_TYPE)
AS
declare variable cr_rate curr_type;
declare variable k_id id_type;
declare variable k_index_curr_rate curr_type;
declare variable doly curr_type;
begin
  select cr_rate
  from opercurr
  where :rep_date_end between cr_date and cr_date_end and
        cr_from_cu_id = '840' rows 1
  into :cr_rate;

  for select k_id, reg_code || ' ' || reg_name, parent_name, k_kontract_no, k_index_curr_code, k_index_curr_rate,
             k_disb_date, kc_amount_approved * k_index_curr_rate, k_kredline_end, pr_name, c_name_fio,
             cast(datediff(day, k_disb_date, :rep_date_end) as double precision) / 30,
             kc_amount_approved / k_amount_approved as doly
      from kontstate, kontract, product, knaktlist, region, region_list(:rep_region_id, 1, 3, 1), knclient, clients
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            k_kolich_client_approved <= 2 and
            k_from_kf_id = kal_from_kf_id and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = id_list and
            reg_unicode = id_list and
            kc_from_k_id = k_id and
            kc_from_c_id = c_id and
            kc_amount_approved > 0 and
            cast(datediff(day, k_disb_date, :rep_date_end) as decimal(15,1)) / 30 > 5
      into k_id, reg_name, parent_name, k_kontract_no, k_index_curr_code, k_index_curr_rate, k_disb_date,
           kc_amount_approved, k_kredline_end, pr_name, c_name_fio, srok, doly
  do
  begin
    k_kont_balance = 0;
    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :k_kont_balance;
    if (k_kont_balance < 0) then
      k_kont_balance = 0;

    if (doly < 1) then
      k_kont_balance = k_kont_balance * doly;

    if (k_index_curr_code = '840') then
    begin
      if (k_index_curr_rate > :cr_rate) then
        k_kont_balance = :k_kont_balance * :k_index_curr_rate;
      else
        k_kont_balance = :k_kont_balance * :cr_rate;
    end

    suspend;
  end
end
