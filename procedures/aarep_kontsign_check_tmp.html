CREATE PROCEDURE AAREP_KONTSIGN_CHECK_TMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    KC_USER_ID DECIMAL(18,0),
    KC_USER_FIO VARCHAR(100),
    KC_KONTRACT_NO VARCHAR(20),
    KC_CIRCLE INTEGER,
    KC_DISB_DATE DATE,
    KC_SIGN_TIME TIMESTAMP,
    KC_DISB_TIME TIMESTAMP,
    KC_REGION VARCHAR(100))
AS
declare variable k_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select k_id, k_kontract_no, k_circle, ks_date_begin, ol_datetime, reg_code || ' ' || reg_name
      from kontract, knaktlist, kontstate, operlist, region, region_list(:report_region_id, 1, -1, 1)
      where k_from_kf_id = kal_from_kf_id and
            kal_date <= ol_date and
            kal_date_end >= ol_date and
            knaktlist.is_delete = 0 and
            operlist.is_delete = 0 and
            k_id = ks_from_k_id and
            ks_state = 1 and
            ks_date_begin between :report_date_begin and :report_date_end and
            ks_ol_is_delete = 0 and
            ks_from_ol_id = ol_id and
            ol_from_reg_id = reg_unicode and
            (ol_from_reg_id = id_list)
      into :k_id, :kc_kontract_no, :kc_circle, :kc_disb_date, :kc_disb_time, :kc_region
  do
  begin
    kc_user_id = null;
    kc_sign_time = null;
    kc_user_fio = null;

    select kl_user_id, kl_datetime, c_name_fio
    from kontlog, username, clients
    where kl_from_k_id = :k_id and
          kl_user_id = un_id and
          un_from_c_id = c_id and
          kl_from_k_id = :k_id and
          kl_state = 1
    order by kl_datetime
    rows 1
    into :kc_user_id, :kc_sign_time, kc_user_fio;

    suspend;
  end
end
