CREATE PROCEDURE DELIQ_TEMP_ONE(
    KONT_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4))
AS
begin
  /* Procedure Text */
  for select ok_sh_date, sum(ok_sh_os + ok_os), sum(ok_sh_perc + ok_perc)
      from operkred
      where ok_from_c_id = 0 and
            ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date and
            ok_type <= 7 and
            ok_type > 0 and
            ok_from_k_id = :kont_id and
            ok_sh_date <= :rep_date and
            ok_sh_id > 0 and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
      group by ok_from_k_id, ok_sh_id, ok_sh_date
      having sum(ok_sh_os + ok_os) > 0 or sum(ok_sh_perc + ok_perc) > 0
      order by ok_sh_date
      rows 1
      into :sh_date, :sh_os, :sh_perc
  do
    suspend;
end
