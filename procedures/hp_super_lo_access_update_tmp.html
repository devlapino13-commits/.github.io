CREATE PROCEDURE HP_SUPER_LO_ACCESS_UPDATE_TMP
AS
declare variable ASLT_OFFICE_ID D_INTEGER;
declare variable ASLT_LO_ID D_ID;
declare variable ASLT_OFFICE_MANAGER_ID D_ID;
declare variable ASLT_REGION_NAME D_STR_100;
declare variable ASLT_OFFICE_NAME D_STR_100;
declare variable ASLT_OFFICE_PAR30_360 D_PERCENT;
declare variable ASLT_OFFICE_MANAGER_NAME D_STR_100;
declare variable ASLT_OFFICE_MANAGER_EXPERIENCE D_SMALLINT;
declare variable ASLT_LO_NAME D_STR_100;
declare variable ASLT_SUPER_CATEGORY D_SMALLINT;
declare variable ASLT_PAR30_360 D_PERCENT;
declare variable ASLT_EXPERIENCE D_SMALLINT;
declare variable ASLT_EARNINGS D_INTEGER;
declare variable ASLT_BALANCE D_CURRENCY;
declare variable ASLT_DELIQ30 D_CURRENCY;
declare variable ASLT_DELIQ360 D_CURRENCY;
declare variable ASLT_BALANCE_WITHOUT360 D_CURRENCY;
declare variable ASLT_DELIQ30_60 D_CURRENCY;
declare variable TMP_OFFICE_ID D_INTEGER;
declare variable C_NAME_FIO D_STR_100;
declare variable C_EXPERIENCE D_SMALLINT;
declare variable AAREP_BONUS_TOTAL D_CURRENCY;
declare variable OFFICE_DELIQ_BALANCE D_CURRENCY;
declare variable OFFICE_ACTIVE_BALANCE D_CURRENCY;
declare variable REP_REGION_ID D_INTEGER;
declare variable REP_DATE_BEGIN D_DATE;
declare variable REP_DATE_END D_DATE;
begin
  -- готовит список Супер КС для hp_super_lo_access_update, которая запускается роботом ежемесячно 5-го числа (RFCAZ-427)

  if (f_get_db_id() <> 94401) then
    exit;

  rep_region_id = 1000000; -- Azerbaycan
  rep_date_end = current_date - extract(day from current_date); -- конец предыдущего месяца
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1; -- начало предыдущего месяца

  delete from analitica_super_lo_tmp;

  insert into analitica_super_lo_tmp (aslt_office_id, aslt_lo_id, aslt_date_begin, aslt_date_end, aslt_region_name, aslt_office_name, aslt_lo_name, aslt_balance, aslt_deliq30, aslt_deliq360)
  select aa_curr_office_id, aa_curr_kc_id, :rep_date_begin, :rep_date_end, min(aarep_name_1), min(aarep_name_2), min(aarep_name_3), sum(aarep_active_balance_e), sum(aarep_deliq_balance30), sum(aarep_deliq_balance360)
  from aarep_analitica_portfolio(:rep_region_id, :rep_date_begin, :rep_date_end, -1, 0, null)
  group by aa_curr_office_id, aa_curr_kc_id
  having sum(aarep_active_balance_e) > 0;

  update analitica_super_lo_tmp
  set aslt_balance_without360 = aslt_balance - aslt_deliq360, -- КП чистый от просрочек 360+
      aslt_deliq30_60 = aslt_deliq30 - aslt_deliq360; -- Баланс кредитов с просрочкой 30-360

  update analitica_super_lo_tmp
  set aslt_par30_360 = 0
  where aslt_deliq30_60 = 0 and
        aslt_balance_without360 = 0;

  update analitica_super_lo_tmp
  set aslt_par30_360 = aslt_deliq30_60 / aslt_balance_without360 * 100 -- ПАР 30-360 КС в процентах
  where aslt_balance_without360 > 0;

  for select distinct aslt_office_id
      from analitica_super_lo_tmp
      into tmp_office_id
  do
  begin
    office_deliq_balance = null;
    office_active_balance = null;
    select sum(aslt_deliq30_60), sum(aslt_balance_without360)
    from analitica_super_lo_tmp
    where aslt_office_id = :tmp_office_id
    into office_deliq_balance, office_active_balance;

    if (coalesce(office_active_balance, 0) = 0) then
      aslt_office_par30_360 = 0;
    else
      aslt_office_par30_360 = office_deliq_balance / office_active_balance * 100; -- ПАР 30-360 офиса в процентах

    update analitica_super_lo_tmp
    set aslt_office_par30_360 = :aslt_office_par30_360
    where aslt_office_id = :tmp_office_id;

    aslt_office_manager_id = null;
    select ps.ps_from_c_id
    from hr_pstatus ps
    inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind
    where ps.ps_from_reg_id = :tmp_office_id and
          ps.ps_status = 1 and
          ps.ps_is_working = 1 and
          :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
          jk.jk_system_code = 'JobSystem.Region_KM' and
          jl.jl_name containing 'filialının meneceri'
    into aslt_office_manager_id;

    c_name_fio = null;
    select c_name_fio
    from clients
    where c_id = :aslt_office_manager_id
    into c_name_fio;

    c_experience = null;
    select c_stage
    from aarep_work_experience(:rep_date_end, :aslt_office_manager_id)
    into c_experience;

    update analitica_super_lo_tmp
    set aslt_office_manager_id = :aslt_office_manager_id,
        aslt_office_manager_name = :c_name_fio,
        aslt_office_manager_experience = :c_experience
    where aslt_office_id = :tmp_office_id;
  end

  for select distinct aslt_lo_id
      from analitica_super_lo_tmp
      where aslt_lo_id <> coalesce(aslt_office_manager_id, 0) -- менеджера офисов не входят в список Супер КС
      into aslt_lo_id
  do
  begin
    if (not exists(select *
                   from hr_pstatus
                   where ps_from_c_id = :aslt_lo_id and
                         ps_status = 1 and
                         ps_is_working = 1 and
                         :rep_date_end between ps_date_begin and ps_date_end)) then
      continue; -- КС не работает (уволен) на момент обновления списка Супер КС

    if (exists(select *
               from hr_pstatus ps
               inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
               inner join jobkind jk on jk.jk_id = jl.jl_kind
               where ps.ps_from_c_id = :aslt_lo_id and
                     ps.ps_status = 1 and
                     ps.ps_is_working = 1 and
                     :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
                     jk.jk_system_code = 'JobSystem.Region_KM')) then -- региональные менеджера не входят в список Супер КС
      continue;

    c_experience = null;
    select c_stage
    from aarep_work_experience(:rep_date_end, :aslt_lo_id)
    into c_experience;

    update analitica_super_lo_tmp
    set aslt_experience = :c_experience
    where aslt_lo_id = :aslt_lo_id;
  end

  -- формирование списка 2-й категории Супер КС
  for select aslt_office_id, aslt_lo_id
      from analitica_super_lo_tmp
      where aslt_office_id not in (1050900) and -- исключенные офисы (RFCAZ-563)
            aslt_office_par30_360 < 1 and -- ПАР офиса менее 1
            aslt_experience >= 6 and -- стаж КС 6+ месяцев
            aslt_par30_360 < 1 -- ПАР КС менее 1
      into aslt_office_id, aslt_lo_id
  do
  begin
    update analitica_super_lo_tmp
    set aslt_super_category = 2
    where aslt_office_id = :aslt_office_id and
          aslt_lo_id = :aslt_lo_id;
  end

  for select aslt_office_id, aslt_lo_id
      from analitica_super_lo_tmp
      where aslt_super_category is null
      into aslt_office_id, aslt_lo_id
  do
  begin
    aarep_bonus_total = null;
    select aarep_bonus_total
    from cb_az_lo_201602_one_lo_2_new(:aslt_office_id, :aslt_lo_id, :rep_date_end, 0)
    into aarep_bonus_total;

    update analitica_super_lo_tmp
    set aslt_earnings = :aarep_bonus_total
    where aslt_office_id = :aslt_office_id and
          aslt_lo_id = :aslt_lo_id;
  end

  -- формирование списка 1-й категории Супер КС
  for select aslt_office_id, aslt_lo_id
      from analitica_super_lo_tmp
      where aslt_super_category is null and -- из тех, кто не вошел во 2-ую категорию
            aslt_experience >= 6 and -- стаж КС 6+ месяцев
            aslt_par30_360 < 0.7 and -- ПАР КС менее 0,7
            aslt_earnings > 1000 -- зароботок более 1000 RFCAZ-618
      into aslt_office_id, aslt_lo_id
  do
  begin
    update analitica_super_lo_tmp
    set aslt_super_category = 1
    where aslt_office_id = :aslt_office_id and
          aslt_lo_id = :aslt_lo_id;
  end
end
