CREATE PROCEDURE HP_OVERPAYMENT_VIEW(
    DTBEG D_DATE = null,
    DTEND D_DATE = null,
    REG_ID D_ID = null)
RETURNS (
    RES_ID D_INTEGER,
    RES_OL_ID D_ID,
    RES_K_ID D_ID,
    RES_K_NO D_STR_20,
    RES_K_DISB_DATE D_DATE,
    RES_KF_NAME D_STR_100,
    RES_REG_ID D_INTEGER,
    RES_REG_NAME D_STR_CODE,
    RES_PROD_IS_IPF D_BOOL,
    RES_IS_GROUP D_BOOL,
    RES_IS_CARD_MBULAK D_BOOL,
    RES_TYPE D_STR_CODE,
    RES_C_ID D_ID,
    RES_C_NAME_FIO D_STR_100,
    RES_PHONE D_STR_20,
    RES_PHONEH D_STR_20,
    RES_EDL_ID D_ID,
    RES_CHECK_ESM_ID D_ID,
    RES_PAY_ESM_ID D_ID,
    RES_DATE_SIGN D_DATETIME,
    RES_DATE D_DATE,
    RES_TR_DATETIME D_DATETIME,
    RES_AMOUNT D_CURRENCY,
    RES_STATUS D_INTEGER)
AS
declare variable reg_lst d_str_8100;
begin
  dtbeg = coalesce(dtbeg, current_date);
  dtend = coalesce(dtend, current_date);
  reg_id = coalesce(reg_id, 1000000);

  reg_lst = '';
  select
    list(id_list, ';')
  from region_list(:reg_id,1,3,1)
  into
    :reg_lst;

  for
    select
      ovp_id,
      ovp_from_ol_id,
      k_id,
      k_kontract_no,
      k_disb_date,
      kal_new_kf_name,
      kal_new_reg_id,
      ovp_prod_is_ipf,
      ovp_is_group,
      ovp_is_card_mbulak,
      ovp_from_c_id,
      coalesce(ovp_phone, ''),
      ovp_edl_id,
      ovp_check_esm_id,
      ovp_pay_esm_id,
      ovp_date,
      ovp_tr_datetime,
      ovp_amount,
      ovp_status
    from overpayment
      inner join operlist
        on ol_id = ovp_from_ol_id
          and operlist.is_delete = 0
      inner join kontract
        on k_id = ovp_from_k_id
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and :dtend between kal_date and kal_date_end
          and knaktlist.is_delete = 0
      left join edoc_list
        on edl_id = ovp_edl_id
    where ovp_date between :dtbeg and :dtend
  into
    :res_id,
    :res_ol_id,
    :res_k_id,
    :res_k_no,
    :res_k_disb_date,
    :res_kf_name,
    :res_reg_id,
    :res_prod_is_ipf,
    :res_is_group,
    :res_is_card_mbulak,
    :res_c_id,
    :res_phone,
    :res_edl_id,
    :res_check_esm_id,
    :res_pay_esm_id,
    :res_date,
    :res_tr_datetime,
    :res_amount,
    :res_status
  do
  begin
    if (reg_lst containing res_reg_id)
    then
      res_reg_name = (select reg_name from region where reg_unicode = :res_reg_id);
    else
      continue;

    res_c_name_fio = (select c_name_fio from clients where c_id = :res_c_id);
    res_phoneh = iif(res_phone = '', '', left(res_phone, 6) || '****' || right(res_phone, 2));

    res_type = trim(iif(res_is_group = 1, 'групповой', 'индивидуальный')) || ', ' ||
      trim(iif(res_prod_is_ipf = 1, 'ипф', 'классика'));

    if (res_edl_id > 0)
    then
      res_date_sign = (
        select first 1 eds_status_datetime
        from edoc_sign
        where eds_from_edl_id = :res_edl_id
          and eds_status = 1
        order by eds_id desc);
    else
      res_date_sign = null;

    suspend;
  end
end
