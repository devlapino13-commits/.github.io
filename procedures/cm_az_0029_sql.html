CREATE PROCEDURE CM_AZ_0029_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable pr_id id_type;
declare variable pr_code str_100;
declare variable c_id id_type;
declare variable c_verify_status integer;
begin
  select k_disb_date, k_from_pr_id, pr_code
  from kontract k, product pr
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into kf_disb_date, pr_id, pr_code;
  for select kc_from_c_id, kc_amount_approved
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into c_id, kc_amount
  do
  begin
    select c_name_fio, c_verify_status
    from clients c
    where c_id = :c_id
    into res_info_text, c_verify_status;
    --блокировки на уровне клиента!
    --условия по возрасту >>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>
    if (c_verify_status <> 2) then
    begin
      /*комментарий: “для отправки контракта на рассмотрение на кредитный комитет статус
    верификации клиента должен быть “успешно верифицирован”.”
    сообщение: “muqavil?nin kredit komit?sin? gond?rilm?si ucun must?rinin verifikasiya
    statusu “успешно верифицирован” olmag? z?ruridir.”
        */
      res_error_text = 'Müştərinin verifikasiya statusu “Успешно верифицирован” olmalıdır.';
      res_error_is_critical = 2;
      suspend;
    end
  end
  if (pr_code containing '.president_decree.') then
  begin
--  проверка поручителя
    for
    select z_from_c_id from zalog
    where z_from_k_id = :k_id
    and z_type = 0
    into c_id
    do
    begin
      select c_name_fio, c_verify_status
      from clients c
      where c_id = :c_id
      into res_info_text, c_verify_status;
      if (c_verify_status <> 2) then
      begin
        res_error_text = 'Zaminin verifikasiya statusu “Успешно верифицирован” olmalıdır.';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end 
end
