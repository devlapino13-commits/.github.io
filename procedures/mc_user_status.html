CREATE PROCEDURE MC_USER_STATUS(
    MCL_ID D_ID)
RETURNS (
    TYPE_INT D_INTEGER,
    CLIENT_NAME D_STR_255,
    MAGAZINE_NAME D_STR_255)
AS
declare variable c_id id_type;
declare variable db_is_actual d_smallint;
declare variable mcl_login d_str_code;
begin
  if (f_get_db_id() not in (39801, 41701)) then
    exit;

  db_is_actual = 0;

  if (exists(select *
             from hp_db_is_actual)) then
    db_is_actual = 1;

  select mcl_from_c_id, mcl_login
  from mc_login
  where mcl_id = :mcl_id
  into :c_id, :mcl_login;

  select c_name_fio
  from clients
  where c_id = :c_id
  into :client_name;

  --> <<временное решение на вермя тестирования по товарам в рассрочку
  if ((db_is_actual = 0 and
      mcl_login = '996777777777')) then -- 1090101387630
  begin
    magazine_name = 'Агент';
    client_name = 'Клиент';
    type_int = 1;
    suspend;
    exit;
  end --<

  if (c_id is null) then
  begin
    type_int = 0; --новый клиент
    suspend;
    exit;
  end
  else
  begin
    select cs.cs_name
    from clientshops cs
    inner join mc_login mcl on mcl.mcl_login = cs.cs_phone
    where cs.cs_from_c_id = :c_id and
          mcl.mcl_id = :mcl_id
    into :magazine_name;

    if (magazine_name is not null) then
    begin
      type_int = 1;
      magazine_name = 'Контрагент';
      client_name = 'Клиент';
    end
    else
      type_int = 0;

    suspend;
  end
end
