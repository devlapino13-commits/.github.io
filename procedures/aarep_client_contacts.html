CREATE PROCEDURE AAREP_CLIENT_CONTACTS(
    C_ID DECIMAL(18,0))
RETURNS (
    CC_ID INTEGER,
    CC_FROM_C_ID DECIMAL(18,0),
    CC_NAME VARCHAR(100),
    CC_RELATION VARCHAR(100),
    CC_CELL_NO VARCHAR(255),
    CC_TYPE INTEGER,
    CC_BIRTH_DATE DATE,
    CC_LANGUAGE_ID INTEGER,
    CC_LANGUAGE VARCHAR(100),
    CC_CAN_EDIT INTEGER,
    CC_MEMO VARCHAR(100))
AS
declare variable cc_in_process integer;
declare variable cc_check_result integer;
begin
  for select cc_id, cc_from_c_id, cc_name, cc_status, cc_cell_no, cc_check_result, cc_in_process, cc_type, cc_language,
             cc_birth_date
      from contacts
      where cc_from_c_id = :c_id
      into :cc_id, :cc_from_c_id, :cc_name, :cc_relation, :cc_cell_no, :cc_check_result, :cc_in_process, :cc_type,
           :cc_language_id, :cc_birth_date
  do
  begin
    cc_memo = '';
    cc_can_edit = 0;
    if (cc_check_result = 0 and
        cc_in_process = 0) then
      cc_can_edit = 1; -- Еще можно редактировать

    cc_language = '';
    if (cc_language_id is not null) then
    begin
      select rd_name
      from refdata
      where rd_id = :cc_language_id
      into cc_language;
    end
    suspend;
  end
end
