CREATE PROCEDURE FM_SAVE_IE_BANK_DETAILS(
    OBJ_KIND D_STR_CODE NOT NULL,
    OBJ_ID D_STR_CODE NOT NULL,
    ACCOUNT_NUMBER D_STR_CODE NOT NULL,
    BIC D_STR_10 NOT NULL)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_100)
AS
begin
    -- RFCRU-5314
    -- Окно открывается через f_uri_open -> save_ie_bank_details
    -- Данное окно нужно для внесения банковских реквизитов

    if (f_objkind(:obj_kind, 'code') <> 'clients') then
    begin
        res_code = -1;
        res_text = 'Что-то пошло не так, просим обратиться в службу технической поддержки';
        suspend;
        exit;
    end

    if (char_length(:bic) <> 9) then
    begin
        res_code = -1;
        res_text = 'БИК должен состоять из 9 цифр';
        suspend;
        exit;
    end

    if (not exists
        (
            select 1
            from banks b
            where b.b_id = :bic
                and b.is_delete = 0
                and b.b_corr_acc is not null
        )
    ) then
    begin
        res_code = -1;
        res_text = 'Банк с данным БИК не найден';
        suspend;
        exit;
    end

    if (char_length(:account_number) <> 20) then
    begin
        res_code = -1;
        res_text = 'Номер счета должен состоять из 20 цифр';
        suspend;
        exit;
    end

    if (:account_number not starting '40802810') then
    begin
        res_code = -1;
        res_text = 'Номер счета должен начинаться на 40802810';
        suspend;
        exit;
    end

    if (exists
        (
            select 1
            from bankclient bc
            inner join banks b on bc.bc_from_b_id = b.b_id
                and b.is_delete = 0
                and b.b_corr_acc is not null
            where bc.bc_from_c_id = :obj_id
                and bc.bc_account_no = :account_number
                and bc.bc_is_main = 1
                and bc.bc_is_active = 1
        )
    ) then
    begin
        -- Если сотрудники добавляют один и тот же расчетный счет
        -- То просто выводим успех
        res_code = 1;
        res_text = 'Расчетный счет был изменен';
        suspend;
        exit;
    end



    -- Все проверки пройдены
    res_code = 1;
    res_text = 'Расчетный счет был изменен';

    execute procedure hp_straider_bankdetail_set(:obj_id, :bic, :account_number);
    suspend;
end
