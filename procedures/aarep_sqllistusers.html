CREATE PROCEDURE AAREP_SQLLISTUSERS(
    SL_ID INTEGER)
RETURNS (
    UN_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    EXEC_COUNT INTEGER,
    UN_FIO VARCHAR(100),
    SHARED_USER VARCHAR(30))
AS
declare variable un_name varchar(30);
begin
  for select slu_from_un_id, un1.un_name, un1.un_from_c_id, slu_execute_count, un2.un_name
      from sqllistusers, username un1, username un2
      where slu_from_sl_id = :sl_id and
            slu_from_un_id = un1.un_id and
            slu_shared_un_id = un2.un_id
      into :un_id, :un_name, :c_id, :exec_count, shared_user
  do
  begin
    if (exists(select *
               from hp_un_is_active(:un_id)
               where un_is_active = 1)) then
    begin
      un_fio = '';
      if (c_id is not null) then
      begin
        select (select txt2 || ' ' || txt1
                from hp_fio_clear(c_name_fio))
        from clients
        where c_id = :c_id
        into :un_fio;
      end
      else
        un_fio = un_name;
      suspend;
    end
  end

end
