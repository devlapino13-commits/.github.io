CREATE PROCEDURE HP_PRINT_INCOME_STATEMENT(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    CLIENT_FIO D_STR_100,
    C_PASS_NO D_STR_100,
    C_INN D_STR_100,
    MONTHLY_INCOME DECIMAL(15,2),
    EXTRA_AMOUNT DECIMAL(15,2),
    QR_CODE D_STR_LONG,
    EDL_BLANK_DATE D_DATE)
AS
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable ma_id d_integer;
declare variable mac_id d_integer;
declare variable kl_date d_date;
declare variable rep_date d_date;
declare variable obj_name d_str_20;
declare variable obj_id d_id;
declare variable edl_id d_id;
begin
    edl_blank_date = null;
    select max(kl.kl_datetime)
    from kontlog kl
    where kl.kl_from_k_id = :k_id and
          kl.kl_state = 1
    into kl_date;
 
    select kncl.kc_id, kont.k_from_ma_app_id, cl.c_id, cl.c_name_fio, cl.c_pass_no, cl.c_inn
    from knclient kncl
    inner join kontract kont on kont.k_id = kncl.kc_from_k_id
    inner join clients cl on cl.c_id = kncl.kc_from_c_id
    where kont.k_id = :k_id and
          kont.k_by_rating = 0 and
          kncl.kc_amount_approved > 0
    into kc_id, ma_id, c_id, client_fio, c_pass_no, c_inn;
 
    qr_code = null;
    if (coalesce(add_params, '') <> '') then
    begin
        edl_id = f_json_str_get_value(add_params, 'edl_id', null);
        qr_code = f_edoc_qr_by_edl(edl_id, c_id);
        select edl_blank_date
        from edoc_list
        where edl_id = :edl_id
        into edl_blank_date;
    end
 
    select macl.mac_id
    from miniappclient macl
    left join refdata rd on rd.rd_id = macl.mac_loan_target
    where macl.mac_from_ma_id = :ma_id and
          macl.mac_from_c_id = :c_id and
          rd.rd_code = 'Loan_Target.Consumer'
    into mac_id;
 
    obj_name = 'MiniAppClient';
    obj_id = mac_id;
    rep_date = current_date;
    if (not exists(select *
                   from actionlist
                   where acl_object_kind = f_objkind(:obj_name, 'id') and
                         acl_object_id = :obj_id and
                         acl_from_rd_id = f_get_rd_id('anketa.solvency'))) then
    begin
        obj_name = 'Clients';
        obj_id = c_id;
        rep_date = coalesce(kl_date, current_date);
    end
 
    if (exists(select *
               from cfp_anketa ca
               where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
                     ca.ca_object_id = :mac_id and
                     ca.ca_from_c_id = :c_id)) then
    begin
        select ca.ca_avr_income
        from cfp_anketa ca
        where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
              ca.ca_object_id = :mac_id and
              ca.ca_from_c_id = :c_id
        into monthly_income;
 
        select extra_amount
        from get_cfp_kpd(:mac_id)
        into extra_amount;
    end
    else
    begin
        select monthly_income, extra_amount
        from hp_get_kpd(:kc_id, :rep_date)
        into monthly_income, extra_amount;
    end
 
    suspend;
end
