CREATE PROCEDURE HP_GET_CLIENT_ADDRESS(
    C_ID ID_TYPE,
    ADC_ADDRESS_TYPE INTEGER)
RETURNS (
    PL_ID ID_TYPE,
    COUNTRY VARCHAR(100),
    ADL_POST_INDEX INTEGER,
    PARENT ID_TYPE,
    REGION_NAME VARCHAR(100),
    DISTRICT VARCHAR(100),
    CITY VARCHAR(100),
    LOCALITY VARCHAR(100),
    STREET_PLT VARCHAR(255),
    STREET VARCHAR(100),
    ADL_DEPART_NO D_STR_20,
    ADL_BUILDING_NO D_STR_20,
    ADL_SECTION_NO D_STR_20,
    ADL_FLAT_NO D_STR_20,
    PLT_CAN_SELECT INTEGER,
    ADL_STREET_ID ID_TYPE,
    ADL_ADDRESS_STR D_STR_255)
AS
declare variable pl_name varchar(100);
declare variable plt_name varchar(255);
begin
  /* Получаем полный адрес по C_ID */
  for select adc_address_type, adl_from_pl_id, adl_post_index, adl_depart_no, adl_building_no, adl_section_no,
             adl_flat_no, adl_street_id, adl_address_str
      from addrclient, addrlist, places
      where adc_from_c_id = :c_id and
            adc_from_adl_id = adl_id and
            adc_address_type = :adc_address_type and
            adl_from_pl_id = pl_id
      into adc_address_type, pl_id, adl_post_index, adl_depart_no, adl_building_no, adl_section_no, adl_flat_no,
           adl_street_id, adl_address_str
  do
  begin
    if (adl_street_id is not null) then
      pl_id = adl_street_id;
    country = null;
    region_name = null;
    district = null;
    city = null;
    locality = null;
    street = null;

    for with recursive tabl1
        as (select pl_id, parent, pl_name, pl_status
            from places
            where pl_id = :pl_id
            union all
            select pl.pl_id, pl.parent, pl.pl_name, pl.pl_status
            from places pl
            join tabl1 on (tabl1.parent = pl.pl_id))
        select parent, pl_name, plt_name, pt.plt_can_select
        from tabl1, placetype pt
        where pl_status = plt_id
        into parent, pl_name, plt_name, plt_can_select
    do
    begin
      if (coalesce(parent, '') = '') then
        country = pl_name;
      if (plt_name = 'Область') then
        :region_name = pl_name;
      if (plt_name = 'Район') then
        district = pl_name;
      if (plt_name = 'Город') then
        city = pl_name;
      if (plt_name in ('Населенный пункт', 'Деревня')) then
        locality = pl_name;
      if (plt_name = 'Улица' or plt_can_select = 2) then
      begin
        street = pl_name;
        street_plt = plt_name;
      end
    end

    suspend;
  end
end
