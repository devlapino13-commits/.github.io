CREATE PROCEDURE HP_LIST_QUERY(
    REG_ID INTEGER,
    DATE_BEGIN DATE,
    DATE_END DATE)
RETURNS (
    OLR_ID D_ID,
    OL_ID D_ID,
    N D_INTEGER,
    UN_FIO D_STR_100,
    OLR_APPR_RD_ID D_INTEGER,
    OLR_REG_DATETIME TIMESTAMP,
    OLR_REG_STATE D_SMALLINT,
    REG_NAME D_STR_100)
AS
begin
  n = 0;
  for select op.olr_id, op.olr_from_ol_id, u.un_fio, op.olr_reg_datetime, op.olr_reg_state,
              olr_appr_rd_id, r.reg_name
      from operlistappr op, username u, region_list(:reg_id, 1, 3, 1), region r
      where op.olr_from_reg_id = id_list and
            u.un_id = op.olr_reg_user and
            op.olr_appr_status = 0 and
            r.reg_unicode = op.olr_from_reg_id and
            cast(op.olr_reg_datetime as date) between :date_begin and :date_end
            order by olr_id
      into :olr_id, :ol_id, un_fio, :olr_reg_datetime, :olr_reg_state, :olr_appr_rd_id, :reg_name
  do
  begin
    n = n + 1;
    suspend;
  end
end
