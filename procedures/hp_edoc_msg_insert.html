CREATE PROCEDURE HP_EDOC_MSG_INSERT(
    EDL_ID D_ID,
    FORCE_EDS_ID D_ID = null,
    FORCE_SMS D_SMALLINT = 0)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable msg_datetime d_datetime;
declare variable eds_id d_id;
declare variable edm_status d_smallint;
declare variable edl_status d_smallint;
declare variable c_id d_id;
declare variable edc_state d_smallint;
declare variable edl_blank_status d_smallint;
declare variable retry_time d_smallint;
declare variable db_id d_id;
begin
  -- Отправляет сообщение всем, кто не подписал при параметре EDS_ID = null
  -- Иначе отправляет только указанному подписанту
  db_id = f_get_db_id();
  res_text = '';
  retry_time = iif(db_id in (39801, 41701), 5, 3);

  select edc_state
  from edoc_cfg, edoc_list
  where edl_id = :edl_id and
        edl_from_edc_id = edc_id
  into edc_state;

  if (edc_state = 0) then
    exit;

  if (edc_state in (1, 2)) then
  begin
    select res_text
    from hp_edoc_msg_send_draft(:edl_id)
    into res_text;

    if (res_text = '' and edc_state = 2) then
      execute procedure hp_edoc_msg_send_replica(:edl_id);

    suspend;
    exit;
  end

  -- Боевой режим для обычной рассылки СМС
  for select eds_id, eds_from_c_id, edl_status, edl_blank_status
  from edoc_sign, edoc_list
  where edl_id = :edl_id and
        eds_from_edl_id = edl_id and
        eds_from_c_id is not null and
        ((eds_status = 0 and :force_eds_id is null) or -- не подписанные записи
         (eds_id = :force_eds_id)) -- или явно указанный подписант
  into eds_id, c_id, edl_status, edl_blank_status
  do
  begin
    if (res_text = '' and edl_status = 0 and edl_blank_status < 0) then
      res_text = 'Ошибка при формировании эл. документа. Ожидайте 15 минут, затем обратитесь в тех. поддержку';

    if (res_text = '' and edl_status = 0 and edl_blank_status = 0) then
      res_text = 'Электронный документ формируется. Подождите 1 минуту';

    if (res_text = '' and edl_status = 1 and force_eds_id is null) then
      res_text = 'Электронный документ уже подписан';

    if (res_text = '' and edl_status = -1) then
      res_text = 'Электронный документ аннулирован';

    if (res_text = '') then
    begin
      msg_datetime = null;
      select edm_status, msg_datetime
      from edoc_msg, fsrobot_msgbox
      where edm_from_eds_id = :eds_id and
            edm_from_msg_id = msg_id and
            msg_object_kind = f_objkind('clients', 'id') and
            msg_object_id + 0 = :c_id -- +0 чтобы не использовать индекс по msg_object_id
      order by msg_datetime desc
      rows 1
      into edm_status, msg_datetime;

      if (edm_status = 1 and force_eds_id is null) then
        res_text = 'Подтверждение уже получено';
      else
      if (msg_datetime is not null and datediff(minute, msg_datetime, current_timestamp) < :retry_time) then
        res_text = 'Со времени предыдущей отправки прошло менее ' || :retry_time || ' минут.';
      else if (db_id not in (39801, 41701) and exists(
        select edm_from_eds_id
        from edoc_msg
          left join fsrobot_msgbox on msg_id = edm_from_msg_id
        where edm_from_eds_id = :eds_id
          and datediff(minute, msg_datetime, current_timestamp) < 60
        group by edm_from_eds_id
        having count(*) > 2
        )) then
          res_text = 'Нельзя отправлять более 3-х сообщений (PUSH, SMS) в течение часа';
    end
  end

  if (res_text = '') then
  begin
    for select eds_id
    from edoc_sign
    where eds_from_edl_id = :edl_id and
          eds_from_c_id is not null and
          ((eds_status = 0 and :force_eds_id is null) or -- не подписанные записи
           (eds_id = :force_eds_id)) -- или явно указанный подписант
    into eds_id
    do
    begin
      if (res_text = '') then
      begin
        select res_text
        from hp_edoc_sms_text(:eds_id, null, :force_sms)
        into res_text;
      end
    end
  end

  if (res_text = '') then
    execute procedure hp_edoc_msg_send_replica(:edl_id);

  suspend;
end
