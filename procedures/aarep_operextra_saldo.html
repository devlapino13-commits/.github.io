CREATE PROCEDURE AAREP_OPEREXTRA_SALDO(
    REP_OBJ_KIND VARCHAR(50),
    REP_OBJ_ID DECIMAL(18,0),
    REP_RD_ID INTEGER,
    REP_STR_ID VARCHAR(100),
    REP_DATE DATE,
    REP_SHOW_ZERO INTEGER)
RETURNS (
    RES_OBJECT_KIND INTEGER,
    RES_OBJECT_ID DECIMAL(18,0),
    RES_LAST_DATE DATE,
    RES_AMOUNT_SALDO DECIMAL(18,4),
    RES_TOTAL_PLUS DECIMAL(18,4),
    RES_TOTAL_MINUS DECIMAL(18,4),
    RES_TOTAL_WRITEOFF DECIMAL(18,4),
    RES_RD_ID INTEGER,
    RES_RD_NAME VARCHAR(255),
    RES_DETAIL VARCHAR(1024))
AS
begin
-- Процедура возвращается существующие обязательства по объекту
-- ID обязательства передается в параметре Rep_RD_ID, если 0 - значит все
-- Тип обязательства передается в параметре Rep_Str_ID, если "" - значит все
-- Тип обязательства могут быть "LoanRepay", "Zalog", "Poruch" и т.д.

--  HP_DO_SHOW_REPAYMENT  -- Показывает задолженность
--  HP_DO_REPAYMENT       -- Распределяет погашение

  res_object_kind = f_objkind(rep_obj_kind, 'id');

  if (rep_date is null) then
    rep_date = '31.12.2999';
  res_object_id = rep_obj_id;
  if (rep_str_id is null) then
    rep_str_id = '';

-- Перечень показателей для учета
  for select rd_id, rd_name
      from refdata
      where rd_main_code = 'OperExtraKind' and
            ((:rep_rd_id <> 0 and
            rd_id = :rep_rd_id) or
            (:rep_rd_id = 0)) and
            ((:rep_str_id <> '' and
            rd_str_id = :rep_str_id) or
            (rd_str_id = :rep_str_id)) and
            rd_code <> rd_main_code
      into :res_rd_id, res_rd_name
  do
  begin
    res_amount_saldo = 0;
    res_total_plus = 0;
    res_total_minus = 0;
    res_total_writeoff = 0;
    res_detail = '';
    res_last_date = null;
    select sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff), sum(ox_amount_plus), sum(ox_amount_minus),
           sum(ox_amount_writeoff), max(ox_ol_date)
    from operextra
    where ox_ol_is_active = 1 and
          ox_ol_date <= maxvalue(:rep_date, current_date) and -- если задним числом, показывает остаток на сегодняшний день
          ox_object_kind = :res_object_kind and
          ox_object_id = :res_object_id and
          ox_from_rd_id = :res_rd_id
    into :res_amount_saldo, :res_total_plus, :res_total_minus, :res_total_writeoff, :res_last_date;
    if (res_amount_saldo is null) then
      res_amount_saldo = 0;
    if (res_total_plus is null) then
      res_total_plus = 0;
    if (res_total_minus is null) then
      res_total_minus = 0;
    if (res_total_writeoff is null) then
      res_total_writeoff = 0;

    if (rep_show_zero = 1 or (rep_show_zero = 0 and
        res_amount_saldo <> 0)) then
    begin
      select ox_detail
      from operextra
      where ox_object_kind = :res_object_kind and
            ox_object_id = :res_object_id and
            ox_from_rd_id = :res_rd_id and
            ox_ol_is_active = 1 and
            ox_amount_plus > 0
      order by ox_from_ol_id desc, ox_id desc
      rows 1
      into :res_detail;
      if (res_detail is null) then
        res_detail = '';

      suspend;
    end
  end
end
