CREATE PROCEDURE MILLICARD_CREDIT_ADD(
    ESM_ID ID_TYPE,
    CREDITID STR_100,
    ERROR_CODE STR_100,
    FULL_SOURCE D_BLOB_TEXT)
AS
declare variable ERROR_TEXT STR_255;
declare variable ESM_PROJECT varchar(20);
declare variable ESM_ACTION STR_100;
declare variable ESM_STATUS smallint;
declare variable RD_ID D_STR_20;
begin
  if (esm_id is null) then
    exit;

  select em.esm_project, em.esm_action, em.esm_status
  from extsysmain em
  where em.esm_id = :esm_id
  into :esm_project, :esm_action, :esm_status;

  if (esm_project not in ('millicard_disb', 'millicard_biz_disb')) then
    exception e_default 'Project not related with Millicard';

  if (esm_action <> 'addCredit') then
    exception e_default 'Action is not an addCredit';

  if (esm_status <> 0) then
    exception e_default 'ESM_STATUS <> 0';

  delete from extsysanswer ea
  where ea.esa_from_esm_id = :esm_id;

  error_code = trim(coalesce(error_code, ''));
  creditid = trim(coalesce(creditid, ''));
  full_source = coalesce(full_source, '');
  error_text = '';

  if ((error_code = '') and
      (creditid = '')) then
    error_code = 'Ошибка не указана, а creditId не передан';

  if (error_code <> '') then
  begin
    select rd.rd_name
    from refdata rd
    where rd.rd_main_code = 'MillicardErrorsApi' and
          rd.rd_code = :error_code
    into :error_text;

    if (:error_code = 'E010002') then
    begin
        update const_var
        set cv_value = '1'
        where cv_code = 'mbf_empty_balance';
    end

    if (error_text = '') then
      error_text = error_code;

    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'error: ' || :error_text);

    update extsysmain em
    set em.esm_status = -1,
        em.esm_full_source = :full_source
    where em.esm_id = :esm_id;
  end
  else
  begin
    update extsysmain em
    set em.esm_status = 1,
        em.esm_trans_id = :creditid,
        em.esm_full_source = :full_source
    where em.esm_id = :esm_id;

    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'creditId: ' || :creditid);

    -- созданию задачу на утверждение кредита
    insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                            esm_full_source, esm_amount, esm_from_k_id, esm_account, esm_trans_id)
    select :esm_project, 'checkCredit', current_date, esm_object_kind, esm_object_id, 0, esm_full_source,
           esm_amount, esm_from_k_id, em.esm_account, :creditid
    from extsysmain em
    where em.esm_id = :esm_id;
  end
end
