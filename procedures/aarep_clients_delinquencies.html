CREATE PROCEDURE AAREP_CLIENTS_DELINQUENCIES(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_PERIOD D_INTEGER,
    K_FACT_DISB_DATE D_DATE,
    DELINQ_DAY D_INTEGER,
    KC_AMOUNT_APPLICATE CURR_TYPE,
    OKS_BALANCE CURR_TYPE,
    OKS_DELINQ_OS D_CURRENCY,
    OKS_DELINQ_PERC D_CURRENCY,
    NAME_OFFICER D_STR_100,
    KF_NAME_FIO D_STR_100,
    UN_FIO D_STR_100,
    CELL_NO D_STR_20,
    ADDRESS_JUR D_STR_255,
    ADDRESS_FACT D_STR_255,
    C_INN D_STR_20,
    SUMMA_OD CURR_TYPE,
    SUMMA_VOZN CURR_TYPE,
    SUMMA_PENYA CURR_TYPE,
    SUMMA_TOTAL CURR_TYPE,
    SUMMA_ISPOLN CURR_TYPE,
    SUMMA_GOSPOSHL CURR_TYPE)
AS
declare variable K_ID D_ID;
declare variable REG_ID D_ID;
declare variable K_FACT_DISB_LO_ID D_ID;
declare variable C_ID D_ID;
declare variable KF_ID D_ID;
declare variable ADC_ADDRESS_TYPE D_INTEGER;
declare variable ADDRESS_STR D_STR_255;
declare variable K_APPROVED_MANAGER D_ID;
declare variable V_ZAD_OS D_CURRENCY;
declare variable V_CALC_PERC D_CURRENCY;
declare variable V_CALC_FINE D_CURRENCY;
declare variable DB_ID D_INTEGER;
begin

  db_id = f_get_db_id();

  k_id = 0;
  k_kontract_no = 0;
  k_period = 0;
  k_fact_disb_lo_id = 0;
  k_approved_manager = 0;
  kf_id = 0;
  oks_balance = 0;
  oks_delinq_os = 0;
  oks_delinq_perc = 0;
  delinq_day = 0;
  reg_id = 0;
  for select k.k_id,
             k.k_kontract_no,
             k.k_period,
             k.k_fact_disb_date,
             k.k_fact_disb_lo_id,
             k.k_approved_manager,
             k.k_from_kf_id,
             oks.oks_balance,
             oks.oks_delinq_os,
             oks.oks_delinq_perc,
             :rep_date_end - oks.oks_delinq_date,
             t.id_list
      from oksummary oks
      inner join kontract k on k.k_id = oks.oks_from_k_id
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
      inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = oks.oks_region_id
      where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_close_state = 0 and
            :rep_date_end - oks.oks_delinq_date > 0 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
            (:db_id = 39801 or kal.kal_new_kf_officer = 1000000000100) and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into :k_id,
           :k_kontract_no,--Номер договора
           :k_period,--Срок кредита
           :k_fact_disb_date,--Дата выдачи
           :k_fact_disb_lo_id,--Выдавший СПФ ID
           :k_approved_manager,--Утвердивший сотрудник
           :kf_id, :oks_balance,--баланс кредита
           :oks_delinq_os,--Задол. ОС
           :oks_delinq_perc,--Задол.%%
           :delinq_day,--кол. прос.дней
           :reg_id
  do
  begin
    c_name_fio = '';
    c_id = 0;
    kc_amount_applicate = 0;
    for select c.c_name_fio,
               c.c_id,
               kc.kc_amount_applicate
        from knclient kc
        inner join clients c on c.c_id = kc.kc_from_c_id
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into :c_name_fio,--ФИО клиента
             :c_id,
             :kc_amount_applicate
    do
    begin
      :un_fio = '';
      select cli.c_name_fio
      from clients cli
      where cli.c_id = :k_approved_manager
      into :un_fio;--Утвердивший сотрудник

      reg_name = '';
      select reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_name;--Офис

      name_officer = '';
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :k_fact_disb_lo_id
      into :name_officer;--Выдавший СПФ

      kf_name_fio = '';
      select cn.c_name_fio
      from knaktlist kal1
      inner join clients cn on cn.c_id = kal1.kal_prior_lo_id
      where kal1.kal_from_kf_id = :kf_id and
            :rep_date_end between kal1.kal_date and kal1.kal_date_end
      into :kf_name_fio;--СПФ на контроле

      address_jur = '';
      address_fact = '';
      for select adcl.adc_address_type,
                 trim(coalesce(adl.adl_address_str, ''))
          from addrclient adcl
          inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
          where adcl.adc_from_c_id = :c_id and
                adcl.adc_address_type in (1, 2)
          into :adc_address_type,
               :address_str
      do
      begin
        if (adc_address_type = 1) then
          address_jur = :address_str;
        if (adc_address_type = 2) then
          address_fact = :address_str;
      end
      address_jur = coalesce(address_jur, '');
      address_fact = coalesce(address_fact, '');

      cell_no = 0;
      cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');

      -- RFCKZ-748
      if(db_id = 39801) then
      begin 
        -- ИИН заёмщика
        c_inn = '';
        select c.c_inn
        from clients c
        where c.c_id = :c_id
        into :c_inn;

        -- суммы из aarep_make_zad_list
        summa_od = 0;
        summa_vozn = 0;
        summa_penya = 0;
        for select zad_os, sh_calc_perc, sh_calc_fine
            from aarep_make_zad_list(:k_id, :rep_date_end, 1)
            into :v_zad_os, :v_calc_perc, :v_calc_fine
        do
        begin
          summa_od    = coalesce(:summa_od, 0)    + coalesce(:v_zad_os, 0);
          summa_vozn  = coalesce(:summa_vozn, 0)  + coalesce(:v_calc_perc, 0);
          summa_penya = coalesce(:summa_penya, 0) + coalesce(:v_calc_fine, 0);
        end
        summa_total = coalesce(:summa_od, 0) + coalesce(:summa_vozn, 0) + coalesce(:summa_penya, 0);

        -- исполнительная надпись (rd_id = 9348)
        summa_ispoln = 0;
        select coalesce(sum(ox.ox_amount_plus), 0)
        from operextra ox
        where ox.ox_object_kind = 9 and
              ox.ox_object_id = :k_id and
              ox.ox_from_rd_id = 9348 and
              ox.ox_ol_is_active = 1
        into :summa_ispoln;

        -- госпошлина (rd_id = 1559)
        summa_gosposhl = 0;
        select coalesce(sum(ox.ox_amount_plus), 0)
        from operextra ox
        where ox.ox_object_kind = 9 and
              ox.ox_object_id = :k_id and
              ox.ox_from_rd_id = 1559 and
              ox.ox_ol_is_active = 1
        into :summa_gosposhl;

      end
      suspend;
    end
  end
end
