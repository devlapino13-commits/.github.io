CREATE PROCEDURE HP_ANTIFRAUD_CLIENT_STATUS(
    C_ID D_ID)
RETURNS (
    AC_ID D_ID,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER,
    CLIENT_WAS_FOUND D_SMALLINT)
AS
declare variable C_NAME_FIO D_STR_255;
declare variable IS_WORKING D_BOOL;
declare variable C_INN D_STR_20;
declare variable AC_DATE D_DATETIME;
declare variable AC_STATUS D_STR_CODE;
begin
    -- Изначально ставим -1, что будет значить, что определить был ли найден клиент не удалось
    client_was_found = -1;

    -- Получаем ФИО клиента
    select c.c_name_fio, c.c_inn 
    from clients c 
    where c.c_id = :c_id
    into :c_name_fio, :c_inn;
    
    -- Проверяем статус API антифрода
    select is_working from antifraud_api_working_status
    into :is_working;
    
    -- Если ИНН пустой, выходим без ошибок
    if (coalesce(:c_inn, '') = '') then
    begin
        res_error_is_critical = 0;
        res_error_text = '';
        suspend;
        exit;
    end
    
    select first 1
        ac.ac_id,
        ac.ac_date,
        f_enum('antifraud_status', ac.ac_status, 'code')
    from antifraud_check ac
    where ac.ac_inn = :c_inn
    order by ac.ac_id desc
    into
        :ac_id, 
        :ac_date,
        :ac_status;
    
    -- 2.2. Безусловный стоп-фактор если найден в списках (не зависит от статуса API)
    if (ac_status = 'was_found') then
    begin
        client_was_found = 1;
        res_error_is_critical = 2; -- Критическая ошибка, блокировка
        res_error_text = 'Клиент ' || :c_name_fio || ' найден в государственных списках. К сожалению, вы не можете далее обслуживать данного клиента.';
        suspend;
        exit;
    end
    
    -- Если API выключен - выходим без ошибок
    if (coalesce(is_working, 0) = 0) then
    begin
        res_error_is_critical = 0;
        res_error_text = '';
        suspend;
        exit;
    end
    
    -- API включен - используем новую логику проверки
    
    -- 2.1. Проверка не выполнена сегодня
    if (ac_date is null or ac_date <> current_date) then
    begin
        res_error_is_critical = 2; -- Критическая ошибка, блокировка
        res_error_text = 'Клиент ' || :c_name_fio || ' не проверен на предмет фрода';
        suspend;
        exit;
    end
    
    -- 2.3. Ошибка при проверке
    if (ac_status in ('server_error', 'client_error', 'format_error')) then
    begin
        res_error_is_critical = 2; -- Критическая ошибка, блокировка
        res_error_text = 'При попытке проверить клиента ' || :c_name_fio || ' в гос. списках возникла ошибка. Просим связаться с разработчиками.';
        suspend;
        exit;
    end
    
    -- Проверка в процессе
    if (ac_status = 'in_process') then
    begin
        res_error_is_critical = 2; -- Критическая ошибка, блокировка
        res_error_text = 'Проверка антифрода клиента ' || :c_name_fio || ' еще не завершена. Дождитесь результатов проверки.';
        suspend;
        exit;
    end

    if (ac_status = 'not_found') then
    begin
        client_was_found = 0;
        suspend;
        exit;
    end
end
