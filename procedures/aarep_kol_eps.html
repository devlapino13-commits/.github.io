CREATE PROCEDURE AAREP_KOL_EPS(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    ACC_AB_CODE D_STR_CODE,
    ACC_CODE D_STR_CODE,
    REG_NAME D_STR_CODE,
    M1 D_INTEGER,
    M2 D_INTEGER,
    M3 D_INTEGER,
    M4 D_INTEGER,
    M5 D_INTEGER,
    M6 D_INTEGER,
    M7 D_INTEGER,
    M11 D_INTEGER,
    M9 D_INTEGER,
    M8 D_INTEGER,
    M12 D_INTEGER,
    M10 D_INTEGER,
    ACC_ADD_CODE D_STR_CODE)
AS
begin
  for select acd.acc_ab_code, acd.acc_code, r.reg_name, ack.acc_add_code,

             sum(
             case
               when extract(month from kob.kob_ol_date) = 1 then 1
               else 0
             end) m1,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 2 then 1
               else 0
             end) m2,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 3 then 1
               else 0
             end) m3,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 4 then 1
               else 0
             end) m4,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 5 then 1
               else 0
             end) m5,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 6 then 1
               else 0
             end) m6,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 7 then 1
               else 0
             end) m7,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 8 then 1
               else 0
             end) m8,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 9 then 1
               else 0
             end) m9,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 10 then 1
               else 0
             end) m10,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 11 then 1
               else 0
             end) m11,
             sum(
             case
               when extract(month from kob.kob_ol_date) = 12 then 1
               else 0
             end) m12

      from kn_operbook kob
      inner join ref_account acd on acd.acc_id = kob.kob_acc_dt
      inner join ref_account ack on ack.acc_id = kob.kob_acc_kt
      inner join kn_obdetail knob on knob.obd_id = kob.kob_detail_id
      inner join region r on r.reg_unicode = ack.acc_from_reg_id
      where kob.kob_ol_date between :rep_date_begin and :rep_date_end and
            kob_ol_is_active = 1 and
            acd.acc_ab_code in ('20202', '20501', '20803', '47423') and
            ack.acc_add_code in ('48809 (10)', '49209 (10)')
      group by acd.acc_ab_code, acd.acc_code, r.reg_name, ack.acc_add_code
      order by 1, 2, 3, 4
      into acc_ab_code, acc_code, reg_name,acc_add_code, m1, m2, m3, m4, m5, m6, m7, m8, m9, m10, m11, m12
  do
    suspend;
end
