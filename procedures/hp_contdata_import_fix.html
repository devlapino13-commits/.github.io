CREATE PROCEDURE HP_CONTDATA_IMPORT_FIX(
    C_ID D_ID)
AS
declare variable ovs_value d_str_100;
declare variable un_datetime d_datetime;
declare variable cda_id d_id;
declare variable rd_id d_integer;
declare variable rd_code d_str_100;
declare variable cont_kind d_str_code;
declare variable cont_kind_id d_integer;
declare variable cv_password d_str_100;
declare variable cv_status d_smallint;
declare variable pin d_str_code;
declare variable cda_is_main d_smallint;
declare variable sign_kind d_str_code;
declare variable cvr_status d_integer;
declare variable un_id d_id;
declare variable clt_id d_integer;
declare variable cda_tmp d_id;
begin
  for select cda_id
  from contdata
  where cda_from_c_id = :c_id
  into cda_id
  do
  begin
    delete from contveri
    where cvr_from_cda_id = :cda_id;
  end

  for select distinct cv_from_rd_id, trim(cv_contact_value), rd_code
  from contverify, refdata
  where cv_object_kind = 0 and
        cv_object_id = :c_id and
        cv_from_rd_id = rd_id and
        cv_status >= 0
  into rd_id, ovs_value, rd_code
  do
  begin
    if (ovs_value is null) then
      ovs_value = '';
  
    cont_kind = replace(rd_code, 'ContactInfo.', '');
    if (cont_kind in ('cell_main', 'cell_add')) then
      cont_kind = 'cell';

    cont_kind_id = f_enum('cont_kind', cont_kind, 'id');

    if (rd_code = 'ContactInfo.Cell_Main')
    then cda_is_main = 1;
    else cda_is_main = 0;

    ovs_value = trim(ovs_value);

    cda_id = null;
    clt_id = f_cont_id(:ovs_value, 'null');
    if (coalesce(clt_id, 0) <> 0) then
    begin
      select cda_id
      from contdata
      where cda_from_c_id = :c_id and
            cda_from_clt_id = :clt_id and
            cda_cont_kind = :cont_kind_id and
            cda_is_main = :cda_is_main
      into cda_id;

      when any do
      begin
        select cda_id
        from contdata
        where cda_from_c_id = :c_id and
              cda_from_clt_id = :clt_id and
              cda_cont_kind = :cont_kind_id and
              cda_is_main = :cda_is_main
        order by cda_is_delete, cda_date desc
        rows 1
        into cda_id;

        if (cda_id is not null) then
        begin
          for select cda_id
          from contdata
          where cda_from_c_id = :c_id and
                cda_from_clt_id = :clt_id and
                cda_cont_kind = :cont_kind_id and
                cda_is_main = :cda_is_main and
                cda_id <> :cda_id
          into cda_tmp
          do
          begin
            update edoc_msg
            set edm_from_cda_id = :cda_id
            where edm_from_cda_id = :cda_tmp;
          end

          delete from contdata
          where cda_from_c_id = :c_id and
                cda_from_clt_id = :clt_id and
                cda_cont_kind = :cont_kind_id and
                cda_is_main = :cda_is_main and
                cda_id <> :cda_id;
        end
      end
    end

    if (cda_id is not null) then
    begin
      for select cv_password, cv_status, coalesce(cv_verify_un_id, cv_from_un_id),
                 coalesce(cv_verify_datetime, cv_datetime)
          from contverify
          where cv_object_kind = 0 and
                cv_object_id = :c_id and
                cv_from_rd_id = :rd_id and
                cv_status >= 0 and
                trim(cv_contact_value) = :ovs_value
          order by cv_id
          into cv_password, cv_status, un_id, un_datetime
      do
      begin
        sign_kind = '';
        sign_kind = case cv_status
          when 0 then 'none'
          when 1 then 'finstart'
          when 2 then 'paper'
          when 3 then 'sms'
        end;

        if (cv_status <> 0) then cv_status = 1;

        pin = cv_password;

        insert into contveri (cvr_from_cda_id, cvr_hash_pin, cvr_status, cvr_sign_kind, cvr_from_un_id, cvr_datetime)
        values (:cda_id, :pin, :cv_status, f_enum('edoc_sign_kind', :sign_kind, 'id'), :un_id, :un_datetime);
      end
    end
  end
end
