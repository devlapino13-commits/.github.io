CREATE PROCEDURE HP_KN_OBSUMMARY_LIST(
    KC_ID ID_TYPE,
    REP_DATE DATE,
    K_ID ID_TYPE = null)
RETURNS (
    ACC_CODE D_STR_CODE,
    KOBS_DATE DATE,
    SAB_FULL_CODE D_STR_CODE,
    KOBS_DT_SUM CURR_TYPE,
    KOBS_DT_TAX CURR_TYPE,
    KOBS_KT_SUM CURR_TYPE,
    KOBS_KT_TAX CURR_TYPE,
    KOBS_BAL_SUM CURR_TYPE,
    KOBS_BAL_TAX CURR_TYPE,
    LAST_DATE DATE)
AS
declare variable acc_id id_type;
declare variable tmp_kc_id id_type;
begin
  if (kc_id = 0) then
    kc_id = null;

  if (kc_id is not null and
      k_id is null) then
  begin
    select kc_from_k_id
    from knclient
    where kc_id = :kc_id
    into :k_id;
  end
  if (kc_id is null and
      k_id is null) then
    exit;

  for select kc_id
      from knclient
      where kc_from_k_id = :k_id and
            (:kc_id is null or
            kc_id = :kc_id) and
            kc_amount_approved > 0
      into :tmp_kc_id
  do
  begin
    for select distinct kobs_acc_id, sab_full_code
        from kn_obsummary
        inner join ref_account on (kobs_acc_id = acc_id)
        left join sys_accbalance on (acc_from_sab_id = sab_id)
        where acc_object_kind = 8 and
              acc_object_id = :tmp_kc_id
        order by acc_code
        into :acc_id, sab_full_code
    do
    begin
      acc_code = f_fmt_acc(acc_id);

      if (rep_date is null) then
      begin
        for select kobs_date, kobs_dt_sum, kobs_dt_tax, kobs_kt_sum, kobs_kt_tax, kobs_bal_sum, kobs_bal_tax
            from kn_obsummary
            where kobs_acc_id = :acc_id
            order by kobs_acc_id, kobs_date
            into kobs_date, kobs_dt_sum, kobs_dt_tax, kobs_kt_sum, kobs_kt_tax, kobs_bal_sum, kobs_bal_tax
        do
        begin
          last_date = kobs_date;
          suspend;
        end
      end
      else
      begin
        for select kobs_date, kobs_dt_sum, kobs_dt_tax, kobs_kt_sum, kobs_kt_tax, kobs_bal_sum, kobs_bal_tax
            from kn_obsummary
            where kobs_acc_id = :acc_id and
                  kobs_date <= :rep_date
            order by kobs_acc_id desc, kobs_date desc
            rows 1
            into kobs_date, kobs_dt_sum, kobs_dt_tax, kobs_kt_sum, kobs_kt_tax, kobs_bal_sum, kobs_bal_tax
        do
        begin
          last_date = kobs_date;
          if (kobs_date < rep_date) then
          begin
            kobs_date = rep_date;
            kobs_dt_sum = 0;
            kobs_dt_tax = 0;
            kobs_kt_sum = 0;
            kobs_kt_tax = 0;
          end
          suspend;
        end
      end
    end
  end
end
