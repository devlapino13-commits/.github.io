CREATE PROCEDURE AAREP_STAFFTEAM_COUNT(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    RES_TEAM_ID INTEGER,
    RES_TEAM_NAME VARCHAR(100),
    RES_MEMBER_COUNT INTEGER,
    RES_TEAM_LEADER INTEGER)
AS
begin
  for select res_team_id, res_team_name, sum(iif(res_lo_id <> 0, 1, 0)), max(iif(res_lo_id <> 0, res_team_leader, 0))
      from aarep_staffteam_list_tmp(:rep_region_id, :rep_date_end)
      group by res_team_id, res_team_name
      into :res_team_id, :res_team_name, :res_member_count, res_team_leader
  do
  begin
    suspend;
  end
end
