CREATE PROCEDURE HP_GET_HTML_CLIENT_INFO(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
RETURNS (
    RES_HTML D_STR_8100)
AS
declare variable C_ID D_ID;
declare variable STATUS_MILITARY D_SMALLINT;
declare variable STATUS_TEXT_MILITARY D_STR_255;
declare variable QUERY_DATE_MILITARY D_DATE;
declare variable ANSWER_MILITARY D_DATE;
declare variable C_SEX D_SMALLINT;
declare variable AGE D_INTEGER;
declare variable LINK_MILITARY D_STR_LONG;
declare variable IS_MILITARY D_ID;
declare variable NEGATIV_KIB_STATUS D_SMALLINT;
declare variable IS_GAMBLER D_BOOL;
declare variable IS_OVERDUE D_BOOL;
declare variable IS_BANKROT D_BOOL;
declare variable NEGATIV_KIB_LINK D_STR_LONG;
declare variable NEGATIV_KIB_TXT D_STR_2048;
declare variable MA_ID D_ID;
declare variable CHECKME_REQUEST D_SMALLINT;
declare variable CHECKME_RISK_LEVEL D_STR_100;
declare variable CHECKME_STATUS D_SMALLINT;
declare variable CHECKME_RES_TEXT D_STR_255;
declare variable CHECKME_QUERY_DATE D_DATE;
declare variable CHECKME_ANSWER_DATE D_DATE;
declare variable CHECKME_LINK D_STR_LONG;
declare variable CHECKME_COLOR D_STR_CODE;
declare variable STOP_CREDIT_RES_HTML D_STR_2048;
declare variable IS_OVERDUE_90 D_SMALLINT;
declare variable IS_ASP D_SMALLINT;
declare variable IS_RESTRUCTURED D_SMALLINT;
declare variable RESTRUCT_KZ D_SMALLINT;
declare variable QUERY_COUNT_90 D_INTEGER;
begin
  res_html = '';
  link_military = '';
  is_military = 1;

---------------------------------CHECKME_INFO-----------------------------------
  if (obj_kind = f_objkind('miniapp', 'id')) then
    select mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :obj_id
      and mac.mac_amount > 0
    into c_id;

  if (obj_kind = f_objkind('kontract', 'id')) then
    select kc.kc_from_c_id
    from knclient kc
    where kc.kc_from_k_id = :obj_id
      and kc.kc_amount_approved > 0
    into c_id;


  if (f_get_const('afm_is_active') = '0') then
  begin
    select first 1
      service_status,
      res_text,
      query_date,
      answer_date
    from hp_checkme_status(:c_id)
    where service_code starting 'KFM'
    order by service_status asc
    into
      :checkme_status,
      :checkme_res_text,
      :checkme_query_date,
      :checkme_answer_date;
    
    select first 1 eil.eil_status
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
      and eic.eic_code = 'checkme_info'
    where eil.eil_from_c_id = :c_id
      and eil.eil_date = current_date
    order by eil.eil_id desc
    into checkme_request;
    
    if (checkme_request < 0) then
      :checkme_link = iif(checkme_status not in (1, 2), f_uri_sql('Отправить повторно', 'checkme_info_request', 'clients', :c_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', ''), '');
    else
      :checkme_link = '';
    
    if (checkme_status > 0) then
      :checkme_risk_level = f_format('Уровень риска: %1', iif(checkme_status = 2, 'Низкий', 'Высокий'));
    else
      :checkme_risk_level = 'Запрос в ПОД/ФТ ещё не был отправлен';
    
    :checkme_color = iif(checkme_status = 2, 'black', 'red');
    
    :res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="20%"><col width="40%"><col width="40%">');
    :res_html = blob_append(res_html,
        f_html_table_row('ПОД/ФТ', f_format('<span style="color: %1;">%2</span> %3', :checkme_color, :checkme_res_text, :checkme_link), :checkme_risk_level, -1, -1, -1, -1, -1, 3, ''));
  end
  else
  begin
    :checkme_link = '';

    if (obj_kind = f_objkind('miniapp', 'id')) then
      ma_id = :obj_id;

    if (obj_kind = f_objkind('kontract', 'id')) then
      select k.k_from_ma_app_id
      from kontract k
      where k.k_id = :obj_id
      into :ma_id;

    select
      status,
      res_text,
      answer_date
    from hp_afm_get_status('afm', :c_id, :ma_id)
    into
      :checkme_status,
      :checkme_res_text,
      :checkme_answer_date;

    :checkme_color = iif(checkme_status in (-1, 0, 1), 'black', 'red');

    if (:checkme_status < -1) then
      :checkme_link = f_uri_sql
        (
            'Отправить повторный запрос в ПОД/ФТ',
            'start_afm_check',
            'miniapp',
            :ma_id,
            'none',
            'Переотправить запрос в АФМ?',
            '',
            0,
            0,
            '',
            '',
            'dialog',
            :c_id
        );

    if (:checkme_status = 0) then
      :checkme_link = '';

    if (:checkme_status = 1) then
    begin
      :checkme_link = '';

      select status
      from hp_afm_get_status('pdl', :c_id, :ma_id)
      into :checkme_status;

      :checkme_color = 'black';
    end

    :checkme_risk_level = iif(checkme_status in (-1, 0), :checkme_res_text, f_format('Уровень риска: %1', iif(checkme_status = 1, 'Низкий', 'Высокий')));
    :res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="20%"><col width="40%"><col width="40%">');
    :res_html = blob_append(res_html,
        f_html_table_row('ПОД/ФТ', f_format('<span style="color: %1;">%2</span>', :checkme_color, iif(checkme_status in (-1, 0), 'В обработке', :checkme_res_text)), :checkme_risk_level,-1, -1, -1, -1, -1, 3, ''));
    :res_html = blob_append(res_html, '</table>');

    if (:checkme_link <> '') then
      :res_html = blob_append(res_html, '<div style="margin-top: 5px;">' || :checkme_link || '</div>');
  end
---------------------------------ПРИЗЫВНИК, ПРОСРОЧКА, ИГРОМАН, БАНКРОТ,--------------------------------------
  if (obj_kind = 9) then
  begin

    for select kc_from_c_id
    from knclient
    where kc_from_k_id = :obj_id
    and kc_amount_applicate > 0
    into :c_id
    do
    begin
       select status,
              query_date,
              answer
       from hp_military_query_status(:c_id)   --ПРИЗЫВНИК
       into :status_military,
            :query_date_military,
            :answer_military;

      select nkh.nkh_is_gambler,
             nkh.nkh_is_overdue,
             nkh.nkh_is_bankrot,
             nkh.nkh_is_overdue_90,
             nkh.nkh_is_restructured,
             eil.eil_status
      from negativ_kib_history nkh
      inner join extinfolist eil on eil.eil_id = nkh.nkh_from_eil_id   -- ПРОСРОЧКА, ИГРОМАН, БАНКРОТ, RFCKG-4215
      where nkh.nkh_from_c_id = :c_id
        and nkh.nkh_last = 1
      order by nkh.nkh_id desc
      rows 1
      into is_gambler,
           is_overdue,
           is_bankrot,
           is_overdue_90,
           is_restructured,
           negativ_kib_status;

      negativ_kib_link = f_uri_sql('Отправить повторно', 'negativ_kib_request', 'clients', :c_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', '');

    end

  end

  if (obj_kind = 11) then
  begin
    for select mac_from_c_id
    from miniappclient
    where mac_from_ma_id = :obj_id and
          mac_amount > 0
    into c_id
    do
    begin
      select  status,
              query_date,
              answer
       from hp_military_query_status(:c_id)      --ПРИЗЫВНИК
       into :status_military,
            :query_date_military,
            :answer_military;


       select nkh.nkh_is_gambler,
              nkh.nkh_is_overdue,
              nkh.nkh_is_bankrot,
              nkh.nkh_is_overdue_90,
              nkh.nkh_is_restructured,
              eil.eil_status,
              nkh.nkh_query_count_90
      from negativ_kib_history nkh
      inner join extinfolist eil on eil.eil_id = nkh.nkh_from_eil_id   -- ПРОСРОЧКА, ИГРОМАН, БАНКРОТ,
      where nkh.nkh_from_c_id = :c_id
        and nkh.nkh_last = 1
      order by nkh.nkh_id desc
      rows 1
      into is_gambler,
           is_overdue,
           is_bankrot,
           is_overdue_90,
           is_restructured,
           negativ_kib_status,
           query_count_90;

      negativ_kib_link = ' '|| f_uri_sql('Отправить повторно', 'negativ_kib_request', 'clients', :c_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', '');
    end
  end

  is_asp = f_is_asp_client(:c_id);

  if(coalesce(negativ_kib_status, -777) = -777) then
  begin
    negativ_kib_status = -777;
    is_gambler = 0;
    is_overdue = 0;
    is_bankrot = 0;
    is_overdue_90 = 0;
    is_restructured = 0;
  end

-- проверка реструктуризации --RFCKZ-582
  if (coalesce(is_restructured, 0) = 0) then
  begin
    restruct_kz = 0;
    select is_restructured
    from hp_is_restructured_kz(:c_id, 12)
    into :restruct_kz;

  if (coalesce(restruct_kz, 0) = 1) then
    is_restructured = 1;
  end

  negativ_kib_txt = (case negativ_kib_status
                          when 0 then 'Выполняется проверка'
                          when -1 then 'Выполняется проверка'
                          when -777 then 'Кредитный отчет еще не получен'
                          else 'Во время проверки произошла ошибка<br>' || negativ_kib_link
                        end);
  if (exists(select 1
             from extinfolist eil
             inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
             where eic.eic_code = 'pkb_available_reports' and
                   eil.eil_from_c_id = :c_id and
                   eil.eil_status = 2
             )) then
  begin
    if ((select priority_id
         from hp_get_priority_rep_id(:obj_kind, :obj_id)) = -1) then   -- RFCKZ-214

      negativ_kib_txt = 'Нет';
  end

  select c.c_sex,
         f_age(c.c_birth_date, current_date)
  from clients c
  where c.c_id = :c_id
  into c_sex, age;

  if (age not between 18 and 27 -- клиент младше 18 ИЛИ старше 27
      or (c_sex <> 1) ) then    -- клиент не мужчина
  begin
    is_military = 0;
  end

  status_text_military = (case status_military
                   when 0 then 'Уволен со срочной воинской службы <br>Дата увольнения: %1'
                   when 1 then 'Призван на срочную воинскую службу <br>Дата призыва: %1'
                   when 3 then '<span style="color: red;">Ошибка при получении статуса клиента</span>'
                   when -1 then '<span style="color: red;">Ошибка при получении статуса клиента</span>'
                   when -2 then '<span style="color: red;">Ошибка при получении статуса клиента</span>'
                   when -3 then 'Запрос не отправлен'
                   when -4 then 'Запрос отправлен.'
                   else ''
                 end);

  if(current_date <> query_date_military or
      coalesce(query_date_military, '') = '' or
      status_military in (-2)) then --Если текущий день отличается от даты проверки, у пользователя есть возможность запросить повторную проверку ИЛИ ответ на предыдущий запрос ошибка (-2)
    link_military = f_uri_sql(iif(coalesce(query_date_military, '') <> '', 'Проверить повторно', 'Проверить'), 'military_status', 'clients', :c_id, 'none', 'Подтверждаете', '', 0, 0, '', '', 'dialog', '');

  if (is_military = 1) then
  begin
    if (status_military in (-1, -2)) then
    begin
      res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="40%"><col width="20%">');
      res_html = blob_append(res_html, f_html_table_row('Призывник',
                                                        f_format(status_text_military, f_datetostr(answer_military)),
                                                        link_military,
                                                        -1, -1, -1, -1, -1, 3, ''));
      res_html = blob_append(res_html, '</table>');
    end
    else
    begin
      res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
      res_html = blob_append(res_html, f_html_table_row('Призывник',
                                                        f_format(status_text_military, f_datetostr(answer_military)),
                                                        '',
                                                        '',
                                                        -1, -1, -1, -1, 2, ''));
      res_html = blob_append(res_html, '</table>');
    end
  end

  res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');

  res_html = blob_append(res_html, f_html_table_row('Реструктуризация за последние 12 месяцев',  --RFCKZ-582
                                                     f_format('<span style="color: %1;">%2</span>',
                                                                iif(coalesce(is_restructured, 0) = 1, 'red', 'black'),
                                                                iif(coalesce(is_restructured, 0) = 1, 'Да, выдача невозможна', 'Нет')),
                                                      '', '', -1, -1, -1, -1, 2, ''));

  res_html = blob_append(res_html, f_html_table_row('Просроченная задолженность свыше 30 дней по банк. займам и/или свыше 1 дня по микрокредитам',
                                                      f_format('<span style="color: %1;">%2</span>',
                                                                 iif(is_overdue = 1, 'red', 'black'),
                                                                 iif(negativ_kib_status = 2, iif(is_overdue = 1, 'Да Выдача невозможна', 'Нет'), negativ_kib_txt)),
                                                      '', '', -1, -1, -1, -1, 2, ''));

  res_html = blob_append(res_html, f_html_table_row('Банкрот',
                                                      f_format('<span style="color: %1;">%2</span>',
                                                                 iif(is_bankrot = 1, 'red', 'black'),
                                                                 iif(negativ_kib_status = 2, iif(is_bankrot = 1, 'Да Выдача невозможна', 'Нет'), negativ_kib_txt)),
                                                      '', '', -1, -1, -1, -1, 2, ''));


  res_html = blob_append(res_html, f_html_table_row('Игроман',
                                                      f_format('<span style="color: %1;">%2</span>',
                                                                 iif(is_gambler = 1, 'orange', 'black'),
                                                                 iif(negativ_kib_status = 2, iif(is_gambler = 1, 'Да, предельное значение КДН было изменено <br>', 'Нет'), negativ_kib_txt)),
                                                      '', '', -1, -1, -1, -1, 2, ''));

  res_html = blob_append(res_html, f_html_table_row('Получатель АСП',
                                                      f_format('<span style="color: %1;">%2</span>',
                                                                 iif(is_asp = 1, 'orange', 'black'),
                                                                 iif(is_asp = 1, 'Да, предельное значение КДН было изменено на 0.25 <br>', 'Нет')),
                                                      '', '', -1, -1, -1, -1, 2, ''));

  res_html = blob_append(res_html, f_html_table_row('Просроченная задолженность свыше 90 дней',
                                                      f_format('<span style="color: %1;">%2</span>',
                                                                 iif(is_overdue_90 = 1, 'orange', 'black'),
                                                                 iif(negativ_kib_status = 2, iif(is_overdue_90 = 1, 'Да, предельное значение КДН было изменено на 0.25 <br>', 'Нет'), negativ_kib_txt)),
                                                      '', '', -1, -1, -1, -1, 2, ''));

res_html = blob_append(res_html, f_html_table_row('Количество запросов отчета КИБ за последние 90 дней ≥ 40',
                                                     f_format('<span style="color: %1;">%2</span>',
                                                                 iif(negativ_kib_status = 2, iif(coalesce(query_count_90, 0) >= 40, 'red', 'black'), 'black'),
                                                                 iif(negativ_kib_status = 2,
                                                                 iif(coalesce(query_count_90, 0) >= 40, 'Да, выдача невозможна', 'Нет'), 'Не проверено')),
                                                      '', '', -1, -1, -1, -1, 2, ''));

  res_html = blob_append(res_html, '</table>');

    -- RFCKZ-23
  select res_html
  from hp_get_html_stop_credit_info(:obj_kind, :obj_id)
  into :stop_credit_res_html;

  res_html = blob_append(res_html, stop_credit_res_html);

  -- RFCKZ-759, RFCKZ-996
  ma_id = null;

  if (obj_kind = f_objkind('miniapp', 'id')) then
    ma_id = :obj_id;

  if (obj_kind = f_objkind('kontract', 'id')) then
    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :obj_id
    into :ma_id;

  res_html = blob_append(res_html, (select res_html
                                    from hp_get_html_susn_status(:c_id, :ma_id, 1)));

  suspend;
end
