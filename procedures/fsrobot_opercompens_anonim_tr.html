CREATE PROCEDURE FSROBOT_OPERCOMPENS_ANONIM_TR(
    REP_DATE D_DATE)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable k_id d_id;
begin
    for select distinct k_id
    from fsrobot_opercompens_tmp(:rep_date)
    into :k_id
    do
    begin
        err_text = '';
        execute statement('execute procedure fsrobot_opercompens('''||f_datetostr(:rep_date)||''', ' || :k_id || ')')
            with autonomous transaction;
        when any do
        begin
          err_text = 'execute procedure fsrobot_opercompens('''||f_datetostr(:rep_date)||''', ' || :k_id || '); commit; --' || case gdscode
                                                                                                   when 335544345 then 'Lock conflict on no wait transaction'
                                                                                                   when 335544451 then 'Update conflicts with concurrent update'
                                                                                                   when 335544451 then 'Update conflicts with concurrent update'
                                                                                                   when 335544336 then 'Deadlock'
                                                                                                   else 'gdscode=' || gdscode
                                                                                                 end;
          suspend;
        end
    end
end
