CREATE PROCEDURE HP_CLOSEMONTH_1C_BANK
RETURNS (
    PCB_ID D_INTEGER,
    PCB_DATE D_DATE,
    PCB_DOC_NO D_STR_CODE,
    PCB_DETAIL D_STR_255,
    PCB_SIGN D_SMALLINT,
    PCB_ACC_DT_CODE D_STR_CODE,
    PCB_ACC_KT_CODE D_STR_CODE,
    PCB_AMOUNT_SUM D_CURRENCY,
    PCB_AMOUNT_VAL D_CURRENCY,
    PCB_AMOUNT_TAX_DT D_CURRENCY,
    PCB_AMOUNT_TAX_KT D_CURRENCY,
    PCB_FROM_CU_ID D_STR_LANG,
    PCB_MEMO D_STR_LONG,
    PCB_ACC_DT_ID D_ID,
    PCB_ACC_KT_ID D_ID,
    PCB_FROM_OL_ID D_STR_CODE,
    PCB_LINK_KIND D_SMALLINT,
    PCB_OL_CREATED D_SMALLINT)
AS
declare variable tmp_str_1 d_str_code;
declare variable tmp_str_2 d_str_code;
begin
  for select pcb_id, pcb_date, pcb_doc_no, pcb_detail, pcb_sign, pcb_acc_dt_code, pcb_acc_kt_code, pcb_amount_sum,
             pcb_amount_val, pcb_amount_tax_dt, pcb_amount_tax_kt, pcb_from_cu_id, pcb_memo, pcb_acc_dt_id,
             pcb_acc_kt_id, pcb_from_ol_id, pcb_link_kind, pcb_ol_created
      from packet_1c_imp_bank
      order by pcb_date, pcb_id
      into :pcb_id, :pcb_date, :pcb_doc_no, :pcb_detail, :pcb_sign, :pcb_acc_dt_code, :pcb_acc_kt_code, :pcb_amount_sum,
           :pcb_amount_val, :pcb_amount_tax_dt, :pcb_amount_tax_kt, :pcb_from_cu_id, :pcb_memo, :pcb_acc_dt_id,
           :pcb_acc_kt_id, :pcb_from_ol_id, :pcb_link_kind, :pcb_ol_created
  do
  begin
    if (pcb_acc_dt_id is not null) then
    begin
      tmp_str_1 = '';
      tmp_str_2 = '';
      select f_fmt_acc(acc_id), acc_code
      from ref_account
      where acc_id = :pcb_acc_dt_id
      into tmp_str_1, tmp_str_2;

      if (tmp_str_2 = pcb_acc_dt_code) then
        pcb_acc_dt_code = tmp_str_1;
    end

    if (pcb_acc_kt_id is not null) then
    begin
      tmp_str_1 = '';
      tmp_str_2 = '';
      select f_fmt_acc(acc_id), acc_code
      from ref_account
      where acc_id = :pcb_acc_kt_id
      into tmp_str_1, tmp_str_2;

      if (tmp_str_2 = pcb_acc_kt_code) then
        pcb_acc_kt_code = tmp_str_1;
    end

    suspend;
  end
end
