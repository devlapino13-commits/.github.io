CREATE PROCEDURE CB24_RFC1321_LOAN_APP_BKI(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    APPLICATION_ID D_INTEGER,
    APPLICATION_UID D_STR_CODE,
    APPLICATION_DATE D_DATE,
    APPLICATION_SUM D_CURRENCY,
    APPLICATION_TYPE D_STR_20,
    APPLICATION_STATUS D_STR_20,
    APPLICATION_DECISION D_DATE,
    APPLICATION_REJECT_REASON D_STR_255,
    SUBJECT_TYPE D_STR_20,
    SUBJECT_PRIVATE_FIO D_STR_100,
    SUBJECT_PRIVATE_BIRTH_DATE D_DATE,
    SUBJECT_PRIVATE_COUNTRY_CODE D_STR_20,
    SUBJECT_PRIVATE_DOC_SERIAL D_STR_20,
    SUBJECT_PRIVATE_DOC_NUMBER D_STR_20,
    RES_KONTRACT_NO D_STR_20,
    RES_C_ID D_ID,
    RES_KC_ID D_ID,
    RES_DISB_DATE D_DATE,
    RES_DOC_TYPE D_STR_100,
    RES_DOC_NO D_STR_CODE,
    RES_DOC_ID D_ID,
    BKI_SEND_DATE D_DATE,
    SIGN_DATE D_DATE,
    BKI_ERROR_TXT D_STR_LONG)
AS
declare variable new_app_event_id d_integer;
declare variable app_kind d_integer;
declare variable kont_kind d_integer;
begin
  new_app_event_id = f_enum('bki_event', '1.1_new_app', 'id');
  app_kind = f_objkind('miniapp', 'id');
  kont_kind = f_objkind('kontract', 'id');

  for select application_id, application_uid, application_date, application_sum, application_type, application_status,
             application_decision, application_reject_reason, subject_type, subject_private_fio,
             subject_private_birth_date, subject_private_country_code, subject_private_doc_serial,
             subject_private_doc_number, res_kontract_no, res_c_id, res_kc_id, res_disb_date, res_doc_type, res_doc_no,
             res_doc_id, res_sign_date
      from cb24_rfc1321_loan_applications(:rep_region_id, :rep_date_begin, :rep_date_end)
      into :application_id, :application_uid, :application_date, :application_sum, :application_type, :application_status,
           :application_decision, :application_reject_reason, :subject_type, :subject_private_fio,
           :subject_private_birth_date, :subject_private_country_code, :subject_private_doc_serial,
           :subject_private_doc_number, :res_kontract_no, :res_c_id, :res_kc_id, :res_disb_date, :res_doc_type,
           :res_doc_no, :res_doc_id, :sign_date
  do
  begin
    bki_send_date = null;
    bki_error_txt = null;

   with t_objects(object_kind, object_id) as(
        select :app_kind, :application_id
        from rdb$database
      union
        select :kont_kind, k_id
        from kontract k
        where k_from_ma_app_id = :application_id
    )
    select min(bkp_send_datetime)
    from t_objects, bki_packet, bki_loans, bki_clients, bki_events
    where bkl_from_bkp_id = bkp_id
      and bkc_from_bkl_id = bkl_id
      and bev_from_bkc_id = bkc_id
      and bev_event_id = :new_app_event_id
      and bev_status = 3
      and bkl_object_kind = object_kind
      and bkl_object_id = object_id
    into bki_send_date;

    if (bki_send_date is null) then
      with t_objects(object_kind, object_id) as(
          select :app_kind, :application_id
          from rdb$database
        union
          select :kont_kind, k_id
          from kontract k
          where k_from_ma_app_id = :application_id
      )
      select bkp_send_datetime, f_memo_text(bke_error_id)
      from t_objects, bki_packet, bki_loans, bki_clients, bki_events, bki_errors
      where bkl_from_bkp_id = bkp_id
        and bkc_from_bkl_id = bkl_id
        and bev_from_bkc_id = bkc_id
        and bke_from_bev_id = bev_id
        and bev_event_id = :new_app_event_id
        and bkl_object_kind = object_kind
        and bkl_object_id = object_id
      order by 1
      rows 1
      into bki_send_date, bki_error_txt;

    suspend;
  end
end
