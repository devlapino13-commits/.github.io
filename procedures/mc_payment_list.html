CREATE PROCEDURE MC_PAYMENT_LIST(
    MCL_ID D_ID,
    K_ID D_ID)
RETURNS (
    PAYMENT_DATE D_DATE,
    AMOUNT D_STR_100,
    IS_ACTIVE D_SMALLINT,
    IS_DELINQ D_SMALLINT)
AS
begin
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  if (coalesce(k_id, 0) <= 0) then
    exception e_default 'parametr "k_id" has invalid value';

  for select payment_date, amount, is_active, is_delinq
  from mc_payment_list_tmp(:mcl_id, :k_id)
  order by payment_date
  into :payment_date, amount, is_active, is_delinq
  do
  begin
    suspend;
  end
end
