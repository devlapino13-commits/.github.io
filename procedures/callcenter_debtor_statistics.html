CREATE PROCEDURE CALLCENTER_DEBTOR_STATISTICS(
    DATE_BEGIN TIMESTAMP,
    DATE_END TIMESTAMP)
RETURNS (
    OPER_FIO VARCHAR(255),
    POSITIVE INTEGER,
    NEGATIVE INTEGER,
    NOTCALLED INTEGER,
    TOTAL_SUMM INTEGER)
AS
declare variable d1 date;
declare variable d2 date;
declare variable ccdr_result smallint;
begin
  /*
    CREATED BY: IGUSEV 29.06.2012

    PURPOSE: Статистика для колл-центра, по проекту "Звонки должникам"

  */

  d1 = cast(date_begin as date);
  d2 = cast(date_end as date);

  for select cdr.ccdr_oper_fio
      from callcenter_debtors_result cdr
      where cdr.ccdr_call_datetime >= :d1 and
            cdr.ccdr_call_datetime < :d2 + 1 and
            cdr.ccdr_is_auto = 0
      group by cdr.ccdr_oper_fio
      order by cdr.ccdr_oper_fio
      into :oper_fio
  do
  begin
    negative = 0;
    positive = 0;
    notcalled = 0;
    total_summ = 0;
    for select cdr.ccdr_result
        from callcenter_debtors_result cdr
        where cdr.ccdr_call_datetime >= :d1 and
              cdr.ccdr_call_datetime < :d2 + 1 and
              cdr.ccdr_is_auto = 0 and
              cdr.ccdr_oper_fio = :oper_fio
        order by cdr.ccdr_result
        into :ccdr_result
    do
    begin
      if (ccdr_result = 0) then
        negative = negative + 1;
      if (ccdr_result = 1) then
        positive = positive + 1;
      if (ccdr_result = -1) then
        notcalled = notcalled + 1;
      total_summ = total_summ + 1;
    end
    suspend;
  end
end
