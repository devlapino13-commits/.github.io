CREATE PROCEDURE HP_EXPRESSAPP_INSERT(
    EXP_ID D_INTEGER,
    C_ID D_ID,
    EXA_AMOUNT D_CURRENCY,
    EXA_PERIOD D_INTEGER,
    EXA_DISB_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    EXA_ID D_INTEGER)
AS
declare variable point d_integer;
declare variable exa_percent d_percent;
declare variable base_k_id d_id;
declare variable exa_lo_id d_id;
declare variable pr_id d_integer;
declare variable exa_repay_day d_smallint;
declare variable exa_from_reg_id d_integer;
begin
  if (exa_amount > 100000) then
    exception e_message('На данный момент по продукту Индивидуальный ишеним экспресс можно оформить только 100000 сом.');

  pr_id = 440; -- продукт для экспресса
  exa_repay_day = 0;
  point = f_point(:c_id, current_date);

  select
    exp.exp_from_reg_id
  from expresspkg exp
  where exp.exp_id = :exp_id
  into
    :exa_from_reg_id;

  exa_percent =
    case
      when point <= 3 then 36
      when point between 4 and 6 then 35
      when point between 7 and 9 then 33
      when point between 10 and 12 then 31
      when point between 13 and 15 then 30
      when point between 16 and 19 then 28
      when point between 20 and 24 then 26
      when point between 25 and 30 then 25
      when point between 31 and 36 then 24
      when point between 37 and 42 then 23
      when point between 43 and 48 then 22
      when point >= 49 then 21
    end;

  select
    ks_from_k_id
  from knclient
    inner join kontstate
      on ks_from_k_id = kc_from_k_id
        and :exa_disb_date between ks_date_begin and ks_date_end
        and ks_state = 2
        and ks_ol_is_delete = 0
  where kc_from_c_id = :c_id
    and kc_amount_approved > 0
  order by ks_date_begin desc
  rows 1
  into
    :base_k_id;

  select
    kal_new_kf_officer
  from kontract
    inner join knaktlist
      on kal_from_kf_id = k_from_kf_id
        and current_date between kal_date and kal_date_end
        and is_delete = 0
  where k_id = :base_k_id
  into
    :exa_lo_id;

  update or insert into expressapp (
    exa_date, exa_from_c_id, exa_base_k_id, exa_from_exc_id, exa_final_status,
    exa_wait_status, exa_from_ma_id, exa_from_k_id, exa_from_edl_id, exa_disb_date,
    exa_from_pr_id, exa_amount, exa_period, exa_percent, exa_repay_day, exa_lo_id,
    exa_from_reg_id, exa_try_count, exa_from_un_id, exa_datetime, exa_from_exp_id, exa_group_id)
  values (
    null, :c_id, :base_k_id, 0, 0, 0, null, null, null, null, :pr_id, :exa_amount,
    :exa_period, :exa_percent, :exa_repay_day, :exa_lo_id, :exa_from_reg_id, 0, :un_id,
    current_timestamp, :exp_id, 0)
  matching (exa_from_c_id, exa_base_k_id)
  returning exa_id
  into :exa_id;

  suspend;
end
