CREATE PROCEDURE PRINT_AZ_INN_ENTITY(
    K_ID D_ID)
RETURNS (
    VOENINFO_VOEN D_ID,
    NATURALPERSON_LEGAL_ADDRESS D_STR_255,
    VOENINFO_CREATEDATE D_DATE,
    VOENINFO_LASTUPDATEDATE D_DATE)
AS
declare variable C_ID D_ID;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_STR_8100;
begin
  voeninfo_voen = 0;
  naturalperson_legal_address = '';
  voeninfo_createdate = null;
  voeninfo_lastupdatedate = null;
  -- На момент написания в АЗ нет групповых кредитов
  select kc.kc_from_c_id
  from knclient kc
  where kc.kc_from_k_id = :k_id
  into :c_id;

  select eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
        eic.eic_code = 'INN_entity'
  where eil.eil_from_c_id = :c_id and
        eil.eil_status = 2
  order by eil.eil_id desc
  rows 1
  into :bls_id;

  if (:bls_id is not null) then
  begin
    select cast(blx_blob as d_blob_text)
    from hp_get_blobex_blob(:bls_id)
    into :json_text;

    voeninfo_voen = f_json_str_get_value(:json_text, 'VOENInfo.VOEN', 0);
    naturalperson_legal_address = f_json_str_get_value(:json_text, 'NaturalPerson.LegalAddress', '');
    voeninfo_createdate = f_json_str_get_value(:json_text, 'VOENInfo.CreateDate', null);
    -- iif нужен так как если LastUpdateDate пуст выводит ошибку конвертации string в date
    voeninfo_lastupdatedate = iif(f_json_str_get_value(:json_text, 'VOENInfo.LastUpdateDate', null) = '', null, f_json_str_get_value(:json_text, 'VOENInfo.LastUpdateDate', null));
  end
  suspend;
end
