CREATE PROCEDURE AAREP_ALL_KONTRACTS_FOR_PERIOD(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    FIO_LIST VARCHAR(300),
    K_KONTRACT_NO D_STR_20,
    REGION_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    KONTRACT_TYPE D_STR_100,
    OFFICER_NAME D_STR_100,
    KAL_DATE D_DATE,
    STATUS D_STR_100,
    DATE_WRITEOFF D_DATE,
    LAL_NAME D_STR_100,
    HAVE_REPAY D_STR_10,
    REPAY_DATE D_DATE,
    REPAY_SUM CURR_TYPE,
    DEBT_REMAIN CURR_TYPE)
AS
declare variable REG_ID ID_TYPE;
declare variable K_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable OKO_OL_ID ID_TYPE;
declare variable OK_OL_ID ID_TYPE;
declare variable OKS_CLOSE_STATE integer;
declare variable OLD_REG_ID integer;
declare variable K_ID_OLD ID_TYPE;
declare variable OL_ID D_ID;
declare variable HAS_REPAY D_SMALLINT;
declare variable COUNTER D_INTEGER;
declare variable C_NAME_FIO D_STR_255;
declare variable RES_TEXT D_STR_20;
begin

  old_reg_id = 0;
  k_id_old = 0;

  for select oks.oks_from_k_id,
             oks.oks_close_state,
             oks.oks_region_id
      from oksummary oks
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(oks.oks_region_id as integer)
      where :rep_date_end between oks.oks_date_begin and oks.oks_date_end
      order by oks.oks_region_id
      into :k_id,
           :oks_close_state,
           :reg_id
  do
  begin
    status = '';
    if (old_reg_id <> reg_id) then
    begin

      office_name = null;
      region_name = null;

      select rg.reg_code || ' ' || rg.reg_name, rg2.reg_name
      from region rg
      inner join region rg2 on rg2.reg_unicode = rg.parent
      where rg.reg_unicode = :reg_id and
            rg2.reg_unicode > 0
      into :office_name,
           :region_name;

      old_reg_id = reg_id;
    end
    if (k_id_old <> k_id) then
    begin

      kontract_type = null;
      officer_name = null;
      kal_date = null;

      select iif(kt.k_kolich_client_approved = 1, 'индивидуальный', 'групповой'),
                 cl.c_name_fio,
                 kal.kal_date,
                 kt.k_kontract_no
      from kontract kt
      inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id
      inner join clients cl on cl.c_id = kal.kal_new_kf_officer
      where kt.k_id = :k_id and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into :kontract_type,
           :officer_name,
           :kal_date,
           :k_kontract_no;

      k_id_old = k_id;
    end
    counter = 0;
    fio_list = '';

    for select kc.kc_from_c_id
        from knclient kc
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into :c_id

    do
    begin
      c_name_fio = null;

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into c_name_fio;

      counter = counter + 1;
      if (fio_list <> '') then
        fio_list = fio_list || ' ; ' || :c_name_fio;
      else
        fio_list = :c_name_fio;
    end

    status = '';
    date_writeoff = null;
    lal_name = null;
    if (oks_close_state = 1) then
    begin
      status = 'спец. учёт';

      select ok.ok_ol_date
      from operkred ok
      where ok.ok_from_k_id = :k_id and
            ok.ok_type = 5 and
            ok.ok_ol_is_active = 1
      order by ok.ok_ol_date
      rows 1
      into :date_writeoff;

      select lal.lal_name
      from lawmain lam
      inner join lawkredit lak on lak.lak_from_lam_id = lam.lam_id
      inner join lawlist lal on lal.lal_id = lam.lam_from_lal_id
      where lak.lak_from_k_id = :k_id and
            lak.lak_from_c_id = :c_id and
            lam.lam_is_active = 1
      order by lam.lam_date_begin desc
      rows 1
      into :lal_name;

      debt_remain = null;

      select sum(oko_sh_os + oko_sh_perc + oko_sh_other + oko_os + oko_perc)
      from operkroff opf
      where opf.oko_from_k_id = :k_id and
            opf.oko_ol_date <= :rep_date_end and
            opf.oko_ol_is_active = 1
      into :debt_remain;

      debt_remain = coalesce(debt_remain, 0);
      repay_date = null;
      repay_sum = 0;
      have_repay = 'нет';
      has_repay = 0;


      for select distinct oko.oko_ol_date,
            -sum(coalesce(oko.oko_os, 0) + coalesce(oko.oko_perc, 0))
          from operkroff oko
          where oko.oko_from_k_id = :k_id and
                oko.oko_ol_date between :rep_date_begin and :rep_date_end and
                oko.oko_type = 4 and
                oko.oko_ol_is_active = 1
          group by oko.oko_ol_date
          into :repay_date,
               :repay_sum
      do
      begin
        repay_sum = coalesce(repay_sum, 0);
        --наличие оплаты
        if (repay_sum > 0) then
        begin
          have_repay = 'да';
          has_repay = 1;
        end
        suspend;
      end
      if (has_repay = 0) then
      suspend;
    end
    if (oks_close_state = 0) then
    begin

      status = 'активный';
      debt_remain = null;

      select sum(zad_os + sh_calc_perc + sh_calc_fine)
      from aarep_make_zad_list(:k_id, :rep_date_end, 1)
      into :debt_remain;

      debt_remain = coalesce(debt_remain, 0);
      repay_date = null;
      repay_sum = 0;
      has_repay = 0;
      have_repay = 'нет';

      for select distinct ok.ok_ol_date,
                          -sum(coalesce(ok.ok_os, 0) + coalesce(ok.ok_perc, 0) + coalesce(ok.ok_fine, 0) + coalesce(ok.ok_tax, 0))
          from operkred ok
          where ok.ok_from_k_id = :k_id and
                ok.ok_ol_is_active = 1 and
                ok.ok_ol_date between :rep_date_begin and :rep_date_end and
                ok.ok_type = 4
          group by ok.ok_ol_date
          into :repay_date,
               :repay_sum
      do
      begin

        repay_sum = coalesce(repay_sum, 0);
        --наличие оплаты
        if (repay_sum > 0) then
        begin
          have_repay = 'да';
          has_repay = 1;
        end
        suspend;
      end
      if (has_repay = 0) then
      suspend;
    end
  end

  status = '';
  k_id_old = 0;
  old_reg_id = 0;

  for select ks.ks_from_k_id, ks.ks_region_id
      from kontstate ks
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = (ks.ks_region_id)
      where ks.ks_state = 2 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_ol_is_delete = 0 and
            not exists(select *
                       from oksummary okf
                       where okf.oks_from_k_id = ks.ks_from_k_id and
                             :rep_date_end between okf.oks_date_begin and okf.oks_date_end and
                             okf.oks_close_state = 1)
      into :k_id,
           :reg_id

  do
  begin
    status = 'закрытый';
    if (old_reg_id <> reg_id) then
    begin
      office_name = null;
      region_name = null;

      select rg.reg_code || ' ' || rg.reg_name, rg2.reg_name
      from region rg
      inner join region rg2 on rg2.reg_unicode = rg.parent
      where rg.reg_unicode = :reg_id and
            rg2.reg_unicode > 0
      into :office_name,
           :region_name;

      old_reg_id = reg_id;
    end
    if (k_id_old <> k_id) then
    begin

      k_kontract_no = null;
      kontract_type = null;
      officer_name = null;
      kal_date = null;
      lal_name = null;
      res_text = '';

      select iif(kt.k_kolich_client_approved = 1, 'индивидуальный', 'групповой'), cl.c_name_fio,
             kal.kal_date, kt.k_kontract_no
      from kontract kt
      inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id
      inner join clients cl on cl.c_id = kal.kal_new_kf_officer
      where kt.k_id = :k_id and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into :kontract_type,
           :officer_name,
           :kal_date,
           :k_kontract_no;

      k_id_old = k_id;
    end
    counter = 0;
    fio_list = '';

    for select kc.kc_from_c_id
        from knclient kc
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into :c_id
    do
    begin

      c_name_fio = null;

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into c_name_fio;

      counter = counter + 1;
      if (fio_list <> '') then
        fio_list = fio_list || ' ; ' || :c_name_fio;
      else
        fio_list = :c_name_fio;
    end
    date_writeoff = null;

    select ok.ok_ol_date
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_type = 5 and
          ok.ok_ol_is_active = 1
    order by ok.ok_ol_date
    rows 1
    into :date_writeoff;
    select lal.lal_name
    from lawmain lam
    inner join lawkredit lak on lak.lak_from_lam_id = lam.lam_id
    inner join lawlist lal on lal.lal_id = lam.lam_from_lal_id
    where lak.lak_from_k_id = :k_id and
          lak.lak_from_c_id = :c_id and
          lam.lam_is_active = 1
    order by lam.lam_date_begin desc
    rows 1
    into :lal_name;

    debt_remain = 0;
    repay_date = null;
    repay_sum = 0;
    has_repay = 0;
    have_repay = 'нет';

    for select distinct ok.ok_ol_date,
          -sum(coalesce(ok.ok_os, 0) + coalesce(ok.ok_perc, 0) + coalesce(ok.ok_fine, 0))
        from operkred ok
        where ok.ok_from_k_id = :k_id and
              ok.ok_ol_is_active = 1 and
              ok.ok_ol_date between :rep_date_begin and :rep_date_end and
              ok.ok_type = 4
        group by ok.ok_ol_date
        into :repay_date,
             :repay_sum
    do
    begin
      repay_sum = coalesce(repay_sum, 0);
      --наличие оплаты
      if (repay_sum > 0) then
      begin
        have_repay = 'да';
        has_repay = 1;
      end
      suspend;
    end
    if (has_repay = 0) then
    suspend;
  end
end
