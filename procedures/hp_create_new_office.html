CREATE PROCEDURE HP_CREATE_NEW_OFFICE(
    ORIG_OFFICE_ID INTEGER,
    NEW_OFFICE_ID INTEGER,
    NEW_OFFICE_NAME VARCHAR(100),
    NEW_OFFICE_CODE VARCHAR(20),
    ORIG_FIND_STR VARCHAR(100),
    NEW_STR VARCHAR(100))
AS
begin
  insert into region (reg_unicode, parent, lb, rb, reg_status, is_delete, reg_code, reg_name, reg_phone,
                      reg_detail_kolich, reg_okpo, reg_inn, reg_regist_no, reg_company_name, reg_company_boss_titul,
                      reg_company_boss_name, reg_company_account_titul, reg_company_account_name, reg_kont_company,
                      reg_kib_code, reg_head_office, reg_populate_kind, reg_repay_on_weekend, reg_mountain_koef,
                      reg_loronostro_head, reg_sfond, reg_stat_rep_region, reg_sfond_min_zp, reg_report_level,
                      reg_address_legal, reg_date_open)
  select :new_office_id, parent, 0, 0, reg_status, is_delete, :new_office_code, :new_office_name, '', 0, reg_okpo,
         reg_inn, reg_regist_no, reg_company_name, reg_company_boss_titul, reg_company_boss_name,
         reg_company_account_titul, reg_company_account_name, reg_kont_company, 0, reg_head_office, reg_populate_kind,
         reg_repay_on_weekend, reg_mountain_koef, reg_loronostro_head, reg_sfond, reg_stat_rep_region, reg_sfond_min_zp,
         reg_report_level, reg_address_legal, current_date
  from region
  where reg_unicode = :orig_office_id;

  -- RFCKG-4782
  insert into regmountkoefs (rmk_from_reg_id, rmk_date_begin, rmk_date_end, rmk_mount_koef)
  select :new_office_id, current_date, cast('31.12.9999' as d_date), coalesce(reg_mountain_koef, 1)
  from region
  where reg_unicode = :new_office_id;

  insert into acc_list (al_id, al_from_reg_id, al_date, al_from_bg_id, al_currency, al_number, al_type, al_name,
                        al_memo, al_date_close, al_is_active, is_delete, al_from_bl_id)
  select :new_office_code || right(al_id, 14), :new_office_id, al_date, al_from_bg_id, al_currency, al_number, al_type,
         replace(al_name, :orig_find_str, :new_str), al_memo, al_date_close, al_is_active, is_delete, al_from_bl_id
  from acc_list
  where al_from_reg_id = :orig_office_id;

  insert into loronostro (ln_region_loro, ln_region_nostro, ln_account, ln_currency)
  select :new_office_id, ln_region_nostro, :new_office_code || right(ln_account, 14), ln_currency
  from loronostro
  where ln_region_loro = :orig_office_id;

  /*
-- отключено, больше не будет таблицы StandAcc, 18.01.2014 г., Sanjari
  insert into standacc (sa_from_reg_id, sa_from_pr_id, sa_from_f_id, sa_account_type, sa_account)
  select :new_office_id, sa_from_pr_id, sa_from_f_id, sa_account_type, :NEW_OFFICE_CODE||Right(sa_account,14)
  from standacc
  where sa_from_reg_id=:orig_office_id
  and sa_account is not null;

  insert into standacc (sa_from_reg_id, sa_from_pr_id, sa_from_f_id, sa_account_type, sa_account)
  select :new_office_id, sa_from_pr_id, sa_from_f_id, sa_account_type, null
  from standacc
  where sa_from_reg_id=:orig_office_id
  and sa_account is null;
  */

  insert into jobtree (jt_from_reg_id, jt_from_dep_id, jt_from_jl_id, jt_jl_parent_admin, jt_jl_parent_function,
                       jt_date_begin, jt_kolich)
  select :new_office_id, jt_from_dep_id, jt_from_jl_id, jt_jl_parent_admin, jt_jl_parent_function, jt_date_begin,
         jt_kolich
  from jobtree
  where jt_from_reg_id = :orig_office_id;

end
