CREATE PROCEDURE OKT_GROUPS_FOR_BLOCK(
    REP_PROJECT STR_100,
    REP_DATE DATE)
RETURNS (
    CC_PROJECT STR_100,
    K_KONTRACT_NO STR_100)
AS
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();

  if (rep_project = 'OKTELL.Projects.Sirena') then
  begin
    for select distinct 'Сирена 1', k_kontract_no
        from okt_sirena_list(:rep_date, 'OKTELL.Projects.Sirena1', '', '', '', '', '', 21, 60, 0, 0, 0, 0, :db_id)
        where called_oper_str = ''
        into :cc_project, k_kontract_no
    do
    begin
      suspend;
    end
  end
  else if (rep_project = 'OKTELL.Projects.Pantera') then
  begin
    for select distinct 'Пантера 1', k_kontract_no
        from okt_pantera_list(:rep_date, 'OKTELL.Projects.Pantera', '', '', '', '', 1, 9999, 0, 0, 0, 0, :db_id)
        where called_oper_str = ''
        into :cc_project, k_kontract_no
    do
    begin
      suspend;
    end
  end
  else if (rep_project = 'OKTELL.Projects.Ks_finance') then
  begin
    for select distinct 'КС Финанс 1', k_kontract_no
        from okt_pantera_list(:rep_date, 'OKTELL.Projects.Ks_finance_1', '', '', '', '', 1, 9999, 0, 0, 0, 0, :db_id)
        where called_oper_str = ''
        union
        select distinct 'КС Финанс 2', k_kontract_no
        from okt_pantera_list(:rep_date, 'OKTELL.Projects.Ks_finance_2', '', '', '', '', 1, 9999, 0, 0, 0, 0, :db_id)
        where called_oper_str = ''
        into :cc_project, k_kontract_no
    do
    begin
      suspend;
    end
  end
  else if (rep_project = 'OKTELL.PROJECTS.INVESTIGATION_AND_RECOVERY') then
  begin
    for select distinct 'ДРПВ', k_kontract_no
        from okt_pantera_list(:rep_date, 'OKTELL.PROJECTS.INVESTIGATION_AND_RECOVERY', '', '', '', '', 1, 9999, 0, 0, 0, 0, :db_id)
        where called_oper_str = ''
        into :cc_project, k_kontract_no
    do
    begin
      suspend;
    end
  end
end
