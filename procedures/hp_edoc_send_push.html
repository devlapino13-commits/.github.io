CREATE PROCEDURE HP_EDOC_SEND_PUSH(
    EDL_ID D_ID)
AS
declare variable un_google_id d_str_255;
declare variable msg_text d_str_long;
declare variable staff_id d_id;
declare variable client_list d_str_long;
begin
  -- отправляем уведомление сотруднику об успешном подписании эл.документа вручную

  un_google_id = '';
  msg_text = '';

  if (not exists(
  select * from edoc_list, edoc_sign
  where edl_id = :edl_id and
        edl_id = eds_from_edl_id and
        eds_from_c_id is null and
        eds_status = 1 and
        edl_status = 1)) then Exit;

  select edc_name || ' ' || f_objname(edl_object_kind, edl_object_id), un_google_id, un_from_c_id
  from edoc_list, username, edoc_cfg
  where edl_id = :edl_id and
        edl_from_edc_id = edc_id and
        edl_create_un_id = un_id and
        edc_state <> 0 and
        coalesce(un_google_id, '') <> ''
  into :msg_text, un_google_id, staff_id;

  if (un_google_id = '-') then
    un_google_id = '';

  if (coalesce(msg_text, '') <> '' and
      un_google_id <> '') then
  begin
    select list(c_name_fio)
    from edoc_sign, clients
    where eds_from_edl_id = :edl_id and
          eds_from_c_id = c_id
    into client_list;

    if (coalesce(client_list, '') <> '') then
    begin
      msg_text = msg_text || ' (' || client_list || ')';

      select res_text
      from fm_json_data(:msg_text, 'Электронный документ подписан', 'mailmessages_push', 'fsrobot_msgbox', null, null, null, null)
      into msg_text;

      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                  msg_command_kind, msg_object_kind, msg_object_id)
      values (2, :un_google_id, 'Электронный документ подписан', :msg_text, 1, 1000000000000,
              'edoc_sign_to_staff', f_objkind('clients', 'id'), :staff_id);
    end
  end
end
