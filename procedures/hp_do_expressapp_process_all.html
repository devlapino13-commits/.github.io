CREATE PROCEDURE HP_DO_EXPRESSAPP_PROCESS_ALL(
    GROUP_ID D_INTEGER)
AS
declare variable DATETIME_START D_DATETIME;
declare variable DATETIME_FINISH D_DATETIME;
declare variable IS_FOUND D_BOOL;
declare variable SECONDS D_INTEGER;
declare variable EXA_ID D_INTEGER;
declare variable WAS_RESTARTED D_INTEGER;
declare variable DB_ID D_INTEGER;
begin
  datetime_start = dateadd(hour, 9, cast(current_date as timestamp));    -- начало обработки
  datetime_finish = dateadd(hour, 19, cast(current_date as timestamp));   -- конец обработки
  if (group_id = 31) then datetime_finish = dateadd(hour, 22, cast(current_date as timestamp));

  was_restarted = 0;
  db_id = f_get_db_id();

  seconds = 0;
  while (seconds <= 30) do
  begin
    -- Выбираем заявки в офисах у кого местное время между 09:00 м 19:00
    is_found = 0;
    for select exa_id
    from expressapp, region, product
    where exa_group_id = :group_id and
          exa_from_exc_id > 0 and
          exa_final_status in (0, 1) and
          exa_from_reg_id = reg_unicode and
          exa_done = 0 and
          dateadd(hour, iif(:db_id in (39801, 41701), 0 , reg_timezone - 3), current_timestamp) between :datetime_start and :datetime_finish and
          iif(:db_id in (39801, 41701), exa_date <= current_date, true) and
          exa_from_pr_id = pr_id and
          iif(:db_id in (39801, 41701), pr_code not containing '.installment_goods.' and
                              pr_code not containing '.sugarmill.' and
                              pr_code not containing '.online_loan.' or
                              pr_code containing '.tms.', true)
    order by exa_id
    rows 1
    into exa_id
    do
    begin
      is_found = 1;

      -- отмечаем что начали обработку
      in autonomous transaction do
      begin
        update expressapp
        set exa_done = 1
        where exa_id = :exa_id;
      end
  
      in autonomous transaction do
      begin
        execute procedure hp_do_expressapp_process(:exa_id, 0); -- обработки заявки
  
        -- фиксируем завершения обработки
        update expressapp
        set exa_done = 2
        where exa_id = :exa_id;
      end
    end

    -- Если выборки ничего не дала, повторяем выборку с начала списка
    if (is_found = 0) then
    begin
      if (was_restarted = 1) then
        exit;

      in autonomous transaction do
      begin
        update expressapp
        set exa_done = 0
        where exa_group_id = :group_id and
              exa_done <> 0;
        was_restarted = 1;
      end

      if (row_count = 0) then
        exit;
    end
    seconds = extract(second from cast('NOW' as timestamp));
  end
end
