CREATE PROCEDURE HP_DOCRECOG_CREATE_DRM(
    PK_ID D_INTEGER)
AS
declare variable drm_api_code d_str_255;
declare variable drm_id d_integer;
declare variable cnt d_integer;

declare procedure insert_field(drm_id d_integer, drf_code d_str_255,
    drf_name d_str_255, drf_data_type d_integer)
  as
  begin
    insert into docrecogfield (drf_from_drm_id, drf_code, drf_name,
    drf_not_null, drf_is_extra, drf_data_type)
    values (:drm_id, :drf_code, :drf_name, 1, 1, :drf_data_type);
  end

begin
  -- Обрабатываем только те, которые активные и уже не привязаны
  for select pk_id, pk_name,
  pdt_need_doc_no, pdt_need_date_begin, pdt_need_date_end, pdt_need_disb_who,
  pdt_need_for_company,
  pdt_need_territory,
  coalesce(pk_caption_for_memo_1, '') memo_1,
  coalesce(pk_caption_for_memo_2, '') memo_2,
  coalesce(pk_caption_for_memo_3, '') memo_3,
  coalesce(pk_caption_for_memo_4, '') memo_4,
  coalesce(pk_caption_for_memo_5, '') memo_5,
  coalesce(pk_passport_issuers, '') ref_name
  from passkind, passdoctype
  where pk_from_pdt_id = pdt_id
    and pk_from_drm_id is null
    and pk_id > 0
    and pk_id = :pk_id
  as cursor doc do
  begin
    drm_api_code = '';

    insert into docrecogmain (drm_api_code, drm_name, drm_inactive,
      drm_is_manual, drm_recog_2, drm_main_fio_kind, drm_alt_fio_kind,
      drm_show_fields_staff, drm_show_fields_client)
    values (
      :drm_api_code, -- drm_api_code
      doc.pk_name,  -- drm_name
      0,             -- drm_inactive
      iif(:drm_api_code = '', 1, 0), -- drm_is_manual, если код есть, то 0 иначе 1
      0,      -- drm_recog_2, мы не знаем, нужно проверить все документы, у которых есть коды
      'orig', -- :drm_main_fio_kind, оставить как есть
      0,      -- drm_alt_fio_kind, не заполнять
      1,      -- drm_show_fields_staff, показываем
      1       -- drm_show_fields_client, показываем
      )
    returning drm_id
    into drm_id;

    if (doc.pdt_need_doc_no = 1) then
      execute procedure insert_field(drm_id, 'doc_no', '№ документа', 0);

    if (doc.pdt_need_date_begin = 1) then
      execute procedure insert_field(drm_id, 'date_begin', 'Дата выдачи', 1);

    if (doc.pdt_need_date_end = 1) then
      execute procedure insert_field(drm_id, 'date_end', 'Действует до', 1);

    if (doc.pdt_need_disb_who = 1) then
      if (doc.ref_name <> '') then
      begin
        execute procedure insert_field(drm_id, 'authority', 'Кем выдан', 5);
        execute procedure insert_field(drm_id, 'authority_code', 'Код органа', 0);
      end
      else
        execute procedure insert_field(drm_id, 'authority', 'Кем выдан', 0);

    if (doc.pdt_need_for_company = 1) then
      execute procedure insert_field(drm_id, 'for_company', 'Для компании', 0);

    if (doc.pdt_need_territory = 1) then
      execute procedure insert_field(drm_id, 'territory', 'Территория действия', 0);

    if (doc.memo_1 <> '') then
      execute procedure insert_field(drm_id, 'memo_1', doc.memo_1, 0);

    if (doc.memo_2 <> '') then
      execute procedure insert_field(drm_id, 'memo_2', doc.memo_2, 0);

    if (doc.memo_3 <> '') then
      execute procedure insert_field(drm_id, 'memo_3', doc.memo_3, 0);

    if (doc.memo_4 <> '') then
      execute procedure insert_field(drm_id, 'memo_4', doc.memo_4, 0);

    if (doc.memo_5 <> '') then
      execute procedure insert_field(drm_id, 'memo_5', doc.memo_5, 0);

    select count(*) cnt from docrecogfield
    where drf_from_drm_id = :drm_id
    into cnt;

    -- Если нужных полей нет, отключаем показ полей для документа
    if (cnt = 0) then
      update docrecogmain set
      drm_show_fields_staff = 0,
      drm_show_fields_client = 0
      where drm_id = :drm_id;

    -- Создаем привязку
    update passkind
    set pk_from_drm_id = :drm_id
    where pk_id = doc.pk_id;

  end
end
