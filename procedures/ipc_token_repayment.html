CREATE PROCEDURE IPC_TOKEN_REPAYMENT(
    C_ID D_ID,
    K_ID D_ID,
    TOKEN_ID D_STR_100,
    ACTION_CODE D_STR_20,
    REQUEST_TXT D_STR_8100,
    ANSWER_TXT D_STR_8100)
RETURNS (
    ERROR_TXT D_STR_255)
AS
declare variable ESM_AMOUNT D_CURRENCY;
declare variable ET_ID D_ID;
declare variable ESM_ID D_ID;
declare variable STATE D_INTEGER;
declare variable ETR_ID D_ID;
declare variable ERR_MESSAGE D_STR_255;
declare variable TRANS_ID D_STR_CODE;
declare variable OLD_STATE D_INTEGER;
declare variable DB_IS_ACTUAL D_SMALLINT;
begin
  error_txt = '';

  db_is_actual = 0;
  if (exists(select *
             from hp_db_is_actual)) then
    db_is_actual = 1;

  if (c_id is null) then
      error_txt = 'Поле c_id не заполнено';
  if (k_id is null) then
      error_txt = 'Поле k_id не заполнено';
  if (coalesce(token_id, '') = '') then
      error_txt = 'Поле token_id не заполнено';
  if (coalesce(action_code, '') = '') then
      error_txt = 'Поле action_code не заполнено';
  if ((coalesce(answer_txt, '') = '') and (action_code <> 'check_payment')) then
      error_txt = 'Поле answer_txt не заполнено';

  if (answer_txt <> '') then
  begin
      trans_id = f_json_str_get_value(:answer_txt, 'transaction_id', '');
      if ((coalesce(trans_id, '') = '') and (action_code <> 'check_payment')) then
          error_txt = 'transaction_id не передан';
  end

  if (error_txt <> '') then
  begin
    suspend;
    exit;
  end

  if (action_code = 'check_payment') then
  begin
    if (not exists(select *
                   from extsys_token et
                   inner join extsys_token_add eta on eta.eta_from_et_id = et.et_id
                   where et.et_from_c_id = :c_id and
                         et.et_is_actual = 1 and
                         eta.eta_token_id = :token_id)) then
    begin
      error_txt = 'Не найден актуальный токен';
      suspend;
      exit;
    end

    if (exists (select *
                from oksummary oks
                where current_date between oks.oks_date_begin and oks.oks_date_end and
                      oks.oks_from_k_id = :k_id and
                      oks.oks_close_state not in (0, 1))) then
    begin
      error_txt = 'Кредит не активный/без активного долга';
      suspend;
      exit;
    end

    if (db_is_actual = 1) then
    begin
        if (exists(select *
                   from extsys_token_repayment etr
                   inner join extsys_token et on et.et_id = etr.etr_from_et_id
                   where et.et_from_c_id = :c_id
                   and etr.etr_is_finished = 0)) then
        begin
          error_txt = 'Имеется платеж в процессе ч/з подвязанную карту';
          suspend;
          exit;
        end
    
        if (exists(
            select esm_from_k_id
              from extsysmain
              where esm_object_kind = 0 and
                    esm_object_id = :c_id and
                    esm_action = 'pay' and
                    esm_status = 0)) then
        begin
          error_txt = 'Имеется необработанный платеж ч/з платежную систему';
          suspend;
          exit;
        end
    end
  end
  else if ((action_code = 'create_transaction') or (action_code = 'confirm_transaction'))  then
  begin
    state = f_json_str_get_value(:answer_txt, 'state', null);
    esm_amount = f_json_str_get_value(:answer_txt, 'amount', 0);

    select et.et_id
    from extsys_token et
    inner join extsys_token_add eta on eta.eta_from_et_id = et.et_id
    where et.et_from_c_id = :c_id and
         et.et_is_actual = 1 and
         eta.eta_token_id = :token_id
    into :et_id;

    if (state is not null) then
    begin
       select etr.etr_id,
              etr.etr_state
       from extsys_token_repayment etr
       where etr.etr_transaction_id = :trans_id
       into :etr_id,
            :old_state;

       if (old_state not in (0, 1)) then
       begin
            error_txt = 'Опрерация, с текущим transaction_id ранее была передана с конечным статусом';
            suspend;
            exit;
       end

       if (etr_id is null) then
       begin
           insert into extsys_token_repayment (etr_from_et_id, etr_transaction_id, etr_state, etr_from_k_id, etr_amount)
           values (:et_id, :trans_id, :state, :k_id, :esm_amount)
           returning etr_id
           into :etr_id;
       end   

       if (state = 2) then
       begin
          insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id,
                                  esm_amount, esm_from_k_id, esm_full_source)
          values ('ipc_token_payment', 'pay', current_date,  0, :c_id,
                   :esm_amount, :k_id, 'request: ['||:request_txt||']; answer: ['||:answer_txt||']')
          returning esm_id
          into :esm_id;
       end

       err_message = f_json_str_get_value(:answer_txt, 'message', '');


       update extsys_token_repayment etr
       set etr.etr_error_msg = :err_message,
           etr.etr_state = :state,
           etr.etr_from_esm_id = :esm_id
       where etr.etr_id = :etr_id;

       insert into extsys_token_repayment_query (etj_from_etr_id, etj_request, etj_answer)
       values (:etr_id, :request_txt, :answer_txt);
    end
  end
  suspend;
end
