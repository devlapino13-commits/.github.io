CREATE PROCEDURE HP_CLI_CHECK_ADDRESS_EMAIL(
    C_ID ID_TYPE,
    ADL_ID ID_TYPE)
AS
declare variable address_str str_255;
declare variable ma_id integer;
declare variable msg_str d_blob_text;
declare variable c_name_fio str_100;
declare variable c_birth_date date;
declare variable kf_id id_type;
declare variable kf_name str_100;
declare variable adl_comment str_255;
declare variable adc_memo str_255;
begin
-- если это не новый клиент, тогда выход.
  if (exists(select *
             from kontstate ks
             inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
             where ks.ks_ol_is_delete = 0 and
                   ks.ks_state = 1 and
                   kncl.kc_amount_approved > 0 and
                   kncl.kc_from_c_id = :c_id)) then
    exit;

-- igusev Письмо от Бахтияра от 08.02.2019 "О приостановлении запросов на верификацию на ногах"
  if (extract(weekday from current_date) in (0, 6)) then -- 0 - воскоресенье, 6 - суббота
    exit;

  adl_id = null;

  if (adl_id is null) then
  begin
    select adcl.adc_from_adl_id, adcl.adc_memo
    from addrclient adcl
    where adcl.adc_from_c_id = :c_id and
          adcl.adc_address_type = 5 -- фактический адрес на родине
    into :adl_id, :adc_memo;
  end

  if (adl_id is null) then
    exit;

  address_str = '';

  select adl.adl_address_str, adl.adl_comment
  from addrlist adl
  where adl_id = :adl_id and
        left(adl_from_pl_id, 3) = '417' and
        right(adl.adl_id, 9) <> '000000000'
  into :address_str, :adl_comment;

  address_str = coalesce(address_str, '');

  if (address_str = '') then
    exit;

-- только для офисов: курский (москва) & киевский (москва)
-- беру активные заявки клиента
  for select ma_id, ma.ma_from_kf_id
      from miniappclient
      inner join miniapp ma on ma_id = mac_from_ma_id and
            ma_date_end = '31.12.9999' and
            ma.ma_is_active = 1
      inner join knaktlist kal on kal_from_kf_id = ma_from_kf_id and
            kal.is_delete = 0
      where mac_from_c_id = :c_id and
            mac_amount > 0 and
            kal_new_reg_id in (1010800, 1012400)
      into :ma_id, :kf_id
  do
  begin
-- если по заявке договор не аннулирован и не утвержден
    if (not(exists(select kont.k_id
                   from kontract kont
                   where kont.k_from_ma_app_id = :ma_id and
                         kont.k_status in (1, -2)))) then
    begin
      if (not exists(select *
                     from contacts_for_check cfc
                     where --cfc.cfc_from_adl_id = :adl_id
                           cfc.cfc_address_str = :address_str and
                           cfc_from_c_id = :c_id

          )) then
      begin
        --execute procedure fsvisit_action_create('Clients', :c_id, 'mc_address_verify');  #mc_address_verify

        /*
        select cl.c_name_fio, cl.c_birth_date
        from clients cl
        where cl.c_id = :c_id
        into :c_name_fio, :c_birth_date;

        select kal.kal_new_kf_name
        from knaktlist kal
        where kal.kal_from_kf_id = :kf_id and
              current_date between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0
        into :kf_name;

        adl_comment = coalesce(adl_comment, '');
        adc_memo = coalesce(adc_memo, '');

        -----
        --MSG_STR BLOB  D_BLOB_TEXT

' || iif((:adl_comment <> '' or adc_memo <> ''), 'Ориентир: ' || :adl_comment || iif(:adl_comment <> '' and adc_memo <> '', ', ', '') || :adc_memo || '. ' || '', '') -- added by Aleksei on 22-02-2019
        || 'Данные Верификации прошу направить по почте: underwriting@molbulak.com';

        execute procedure hp_send_quick_message('email', 'kg.mnm@mbf.kg',
            '"Верификация на ногах"; группа: ' || kf_name, :msg_str, null, null, 1000000000000);

        insert into contacts_for_check (cfk_from_ma_id, cfc_from_c_id, cfc_from_adl_id, cfc_address_str)
        values (:ma_id, :c_id, :adl_id, :address_str);
