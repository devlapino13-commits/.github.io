CREATE PROCEDURE AAREP_ADVANCE_SALDO_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_ACCOUNT VARCHAR(20),
    REPORT_KIND INTEGER)
RETURNS (
    ADV_ACCOUNT VARCHAR(20),
    ADV_ACCOUNT_NAME VARCHAR(255),
    ADV_CLIENT_ID DECIMAL(18,0),
    ADV_SUMMA DECIMAL(18,4))
AS
declare variable reg_id integer;
begin
  if (:report_kind = 0) then
  begin
    for select al_id, al_name, id_list
        from acc_list, region_list(:report_region_id, 1, 3, 1)
        where al_from_bg_id = :report_account and
              al_from_reg_id = id_list
        into :adv_account, adv_account_name, :reg_id
    do
    begin
      for select ob_client_dt, sum(ob_summa)
          from operbook
          where ob_ol_is_active = 1 and
                ob_account_dt = :adv_account and
                ob_ol_reg_id = :reg_id and
                ob_ol_date between '01.01.1900' and :report_date_end
          group by ob_client_dt
          into :adv_client_id, adv_summa
      do
      begin
        suspend;
      end
      for select ob_client_kt, sum(ob_summa)
          from operbook
          where ob_ol_is_active = 1 and
                ob_account_kt = :adv_account and
                ob_ol_reg_id = :reg_id and
                ob_ol_date between '01.01.1900' and :report_date_end
          group by ob_client_kt
          into :adv_client_id, adv_summa
      do
      begin
        suspend;
      end
    end
  end
  if (:report_kind = 1) then
  begin
    for select ob_account_dt, ob_client_dt, sum(ob_summa)
        from operbook, region_list(:report_region_id, 1, 3, 1)
        where ob_ol_is_active = 1 and
              ob_account_dt = :report_account and
              ob_ol_date between '01.01.1900' and :report_date_end and
              ob_ol_reg_id = id_list
        group by ob_account_dt, ob_client_dt
        into :adv_account, adv_client_id, adv_summa
    do
    begin
      select al_name
      from acc_list
      where al_id = :adv_account
      into :adv_account_name;
      suspend;
    end
    for select ob_account_kt, ob_client_kt, sum(ob_summa)
        from operbook, region_list(:report_region_id, 1, 3, 1)
        where ob_ol_is_active = 1 and
              ob_account_kt = :report_account and
              ob_ol_date between '01.01.1900' and :report_date_end and
              ob_ol_reg_id = id_list
        group by ob_account_kt, ob_client_kt
        into :adv_account, adv_client_id, adv_summa
    do
    begin
      select al_name
      from acc_list
      where al_id = :adv_account
      into :adv_account_name;
      suspend;
    end
  end
  if (:report_kind = 2) then
  begin
    for select ob_account_dt, ob_client_dt, sum(ob_summa)
        from operbook, region_list(:report_region_id, 1, 3, 1)
        where ob_ol_is_active = 1 and
              ob_client_dt = :report_account and
              ob_ol_date between '01.01.1900' and :report_date_end and
              ob_ol_reg_id = id_list
        group by ob_account_dt, ob_client_dt
        into :adv_account, adv_client_id, adv_summa
    do
    begin
      select al_name
      from acc_list
      where al_id = :adv_account
      into :adv_account_name;
      suspend;
    end
    for select ob_account_kt, ob_client_kt, sum(ob_summa)
        from operbook, region_list(:report_region_id, 1, 3, 1)
        where ob_ol_is_active = 1 and
              ob_client_kt = :report_account and
              ob_ol_date between '01.01.1900' and :report_date_end and
              ob_ol_reg_id = id_list
        group by ob_account_kt, ob_client_kt
        into :adv_account, adv_client_id, adv_summa
    do
    begin
      select al_name
      from acc_list
      where al_id = :adv_account
      into :adv_account_name;
      suspend;
    end
  end
end
