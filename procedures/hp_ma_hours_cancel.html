CREATE PROCEDURE HP_MA_HOURS_CANCEL
AS
declare variable RD_ID integer;
declare variable UN_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable MA_ID integer;
declare variable KF_ID ID_TYPE;
declare variable KF_NAME STR_100;
declare variable MFP_DATE_SAVE timestamp;
declare variable MINUTES_DIFF integer;
declare variable FIRST_WARNING_MINUT integer;
declare variable PENALTY_MINUT integer;
declare variable SECOND_WARNING_MINUT integer;
declare variable SAVE_PENALTY smallint;
begin
  if (current_date < '01.08.2018') then
  begin
    penalty_minut = 1440; -- 24 часа

-- Первое уведомление за 12 часов до начисления штрафа ------------------>
    first_warning_minut = 720; -- 720/60 = 12 (т.к. первое уведомление за 12 часов, т.е. с момента оформления заявки прошло 12 часов)

-- Второе уведомление за 4 часа до начисления штрафа -------------------->
    second_warning_minut = 1200; -- 1200/60 = 20 , т.е.  с момента оформления заявки прошло 20 часов
  end
  else
  begin
    penalty_minut = 4320; -- 4320/12 - это 72 часа

-- Первое уведомление за 24 часа до начисления штрафа ------------------>
    first_warning_minut = 1440; -- = 24 часа, т.е. с момента оформления заявки прошло 24 часа

-- Второе уведомление за 48 часа до начисления штрафа -------------------->
    second_warning_minut = 2880; -- 2880/60 = 48, т.е. с момента оформления заявки прошло 48 часов
  end

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'LOPenaltyReason.Miniapp_24_Hours'
  into :rd_id;

  un_id = 1000000000000;

  for select hps.ps_from_c_id
      from hr_pstatus hps
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where current_date between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            jk.jk_system_code = 'JobSystem.LO_Standart'
      into :c_id
  do
  begin
-- Первое уведомление за XXX часов (в зависимости от даты) до начисления штрафа ------------------>
    for select mfp.mfp_date_save, mfp.mfp_ma_id, mfp.mfp_from_kf_id,
               datediff(minute, mfp_date_save, current_timestamp)
        from miniapp_for_penalty mfp
        where mfp.mfp_state = 0 and
              mfp.mfp_current_lo_id = :c_id and
              coalesce(mfp.mfp_inform_status, 0) = 0 -- < важно!
        into :mfp_date_save, :ma_id, :kf_id, minutes_diff
    do
    begin
      if (minutes_diff between :first_warning_minut and :penalty_minut - 1) then
      begin
        execute procedure hp_miniapp_for_penalty_update('miniapp', ma_id); -- контрольная обработка

        if (exists(select *
                   from miniapp_for_penalty mfp
                   where mfp.mfp_ma_id = :ma_id and
                         mfp.mfp_state = 0)) then
        begin
          update miniapp_for_penalty mfp
          set mfp.mfp_inform_status = iif(:minutes_diff between :first_warning_minut and :second_warning_minut - 1, 1, 2)
          where mfp.mfp_ma_id = :ma_id;

          select kal.kal_new_kf_name
          from knaktlist kal
          where kal.kal_from_kf_id = :kf_id and
                current_date between kal.kal_date and kal.kal_date_end and
                kal.is_delete = 0 rows 1
          into :kf_name;

          minutes_diff = penalty_minut - minutes_diff;

          execute procedure fsrobot_insert_event('domailing', 'push_svz_ma_notification',
              :c_id || ';' || 0 || ';' || :kf_name || ';24_hours_inform;' || minutes_diff, :un_id);
        end
      end
    end
-------------------------------------------------------------------------<
-- Второе уведомлениеза XXX часов (в зависимости от даты) до начисления штрафа -------------------->
    for select mfp.mfp_date_save, mfp.mfp_ma_id, mfp.mfp_from_kf_id,
               datediff(minute, mfp_date_save, current_timestamp)
        from miniapp_for_penalty mfp
        where mfp.mfp_state = 0 and
              mfp.mfp_current_lo_id = :c_id and
              coalesce(mfp.mfp_inform_status, 1) = 1 -- < важно!
        into :mfp_date_save, :ma_id, :kf_id, :minutes_diff
    do
    begin
      if (minutes_diff between :second_warning_minut and :penalty_minut - 1) then
      begin
        execute procedure hp_miniapp_for_penalty_update('miniapp', ma_id); -- контрольная обработка

        if (exists(select *
                   from miniapp_for_penalty mfp
                   where mfp.mfp_ma_id = :ma_id and
                         mfp.mfp_state = 0)) then
        begin
          update miniapp_for_penalty mfp
          set mfp.mfp_inform_status = 2
          where mfp.mfp_ma_id = :ma_id;

          select kal.kal_new_kf_name
          from knaktlist kal
          where kal.kal_from_kf_id = :kf_id and
                current_date between kal.kal_date and kal.kal_date_end and
                kal.is_delete = 0 rows 1
          into :kf_name;

          minutes_diff = penalty_minut - minutes_diff;

          execute procedure fsrobot_insert_event('domailing', 'push_svz_ma_notification',
              :c_id || ';' || 0 || ';' || :kf_name || ';24_hours_inform;' || minutes_diff, :un_id);
        end
      end
    end
-------------------------------------------------------------------------<

    for select mfp.mfp_date_save, mfp.mfp_ma_id, mfp.mfp_from_kf_id
        from miniapp_for_penalty mfp
        where mfp.mfp_state = 0 and
              mfp.mfp_current_lo_id = :c_id and
              datediff(minute, mfp_date_save, current_timestamp) > :penalty_minut
        into :mfp_date_save, :ma_id, :kf_id
    do
    begin
      execute procedure hp_miniapp_for_penalty_update('miniapp', ma_id); -- контрольная обработка

      if (exists(select *
                 from miniapp_for_penalty mfp
                 where mfp.mfp_ma_id = :ma_id and
                       mfp.mfp_state = 0)) then
      begin
        save_penalty = 1;

-- до 1 августа, я знал что заявка аннулируется и miniapp_for_penalty  update-ится, после 1.08 этого происходить не будет.
        if (current_date >= '01.08.2018') then
        begin
          if (exists(select *
                     from lo_penalty lp
                     where lp.lp_from_ma_id = :ma_id)) then
            save_penalty = 0;
        end
        -- убираем из штрафа продукт Доверия
        if (save_penalty = 1) then
        begin
          if (exists(select * from miniapp, product
          where ma_id = :ma_id
          and ma_is_active = 1
          and ma_date_end = '31.12.9999'
          and ma_from_pr_id = pr_id
          and pr_code containing '.express_online.')) then save_penalty = 0;
        end

        if (save_penalty = 1) then
        begin
          insert into lo_penalty (lp_from_c_id, lp_from_ma_id, lp_days, lp_ma_date, lp_amount, lp_reason)
          values (:c_id, :ma_id, 0, null, 1000, :rd_id);

          select kal.kal_new_kf_name
          from knaktlist kal
          where kal.kal_from_kf_id = :kf_id and
                current_date between kal.kal_date and kal.kal_date_end and
                kal.is_delete = 0 rows 1
          into :kf_name;

          execute procedure fsrobot_insert_event('domailing', 'push_svz_ma_notification',
              :c_id || ';' || 0 || ';' || :kf_name || ';24_hours;0', :un_id);

          if (current_date < '01.08.2018') then -- с 1 августа я перестаю аннулировать заявки после начисления штрафов, согласно пиьма от Зары
          begin
            execute procedure hp_miniapp_status_change(:ma_id, :un_id, 0);

            insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id)
            values (10, :un_id, null, 0, :ma_id);

            update kontract kont
            set kont.k_status = -1
            where kont.k_status = 0 and
                  kont.k_from_ma_app_id = :ma_id;
          end
        end
      end
    end
  end
end
