CREATE PROCEDURE HP_EXPRESS_EDOC_CANCEL
AS
declare variable obj_kind d_integer;
declare variable obj_id d_integer;
begin
  -- Процедура для аннулирования предложений по экспрессу
  -- Если клиент не принял предложение в течении 3х дней, то аннулируем предложение и останавливаем процесс по клиенту.
  obj_kind = f_objkind('expressapp', 'id');

  for select edl.edl_object_id
      from edoc_list edl
      where edl.edl_object_kind = :obj_kind and
            edl.edl_status = 0 and
            datediff(day, edl.edl_blank_date, current_date) > 3 and
            edl.edl_blank_date > '15.04.2022' -- Не беру в учёт первый запуск экспресса до 15.04.2022
      into obj_id
  do
  begin
    execute procedure hp_edoc_cancel('loan.express', :obj_kind, :obj_id);

    update expressapp exa
    set exa.exa_final_status = -1
    where exa.exa_id = :obj_id;
  end
end
