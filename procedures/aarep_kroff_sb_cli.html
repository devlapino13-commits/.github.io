CREATE PROCEDURE AAREP_KROFF_SB_CLI(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SBA_FROM_K_ID DECIMAL(18,0),
    SBA_KF_NAME VARCHAR(100),
    SBA_KONTRACT_NO VARCHAR(20),
    SBA_DISB_DATE DATE,
    SBA_DATE_BEGIN DATE,
    SBA_REGION_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    SBA_LO_FIO VARCHAR(100),
    SBL_TYPE_STR VARCHAR(20),
    SBL_NAME VARCHAR(255),
    SBL_TEAM_NAME VARCHAR(255),
    AAREP_CLIENTS_KOL INTEGER,
    AAREP_CASHIER_FIO VARCHAR(255),
    AAREP_LEADER_FIO VARCHAR(255),
    OKO_PAYED_PERC DECIMAL(18,4),
    OKO_PAYED_OS DECIMAL(18,4),
    OKO_BALANCE_TOTAL DECIMAL(18,4),
    OKO_BALANCE_PERC DECIMAL(18,4),
    OKO_BALANCE_OTHER DECIMAL(18,4),
    OKO_BALANCE_OS DECIMAL(18,4),
    OKO_PAYED_OTHER DECIMAL(15,2),
    OKO_PAYED_OS_PER DECIMAL(15,2),
    OKO_PAYED_PERC_PER DECIMAL(15,2),
    OKO_PAYED_OTHER_PER DECIMAL(18,4),
    OKO_PAYED_TOTAL_PER DECIMAL(18,4),
    OKO_PAYED_TOTAL DECIMAL(18,4),
    SBA_AMOUNT_APPROVED DECIMAL(18,4),
    SBA_CLIENT_NAME VARCHAR(100),
    SBA_CL_BIRTH_DATE DATE,
    SBA_CL_TEL_NO VARCHAR(20),
    C_ADDRESS_YUR VARCHAR(255),
    C_ADDRESS_FACT VARCHAR(255))
AS
declare variable kc_from_c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable dolya double precision;
declare variable oko_payed_perc1 curr_type;
declare variable oko_payed_os1 curr_type;
declare variable oko_balance_total1 curr_type;
declare variable oko_balance_perc1 curr_type;
declare variable oko_balance_other1 curr_type;
declare variable oko_balance_os1 curr_type;
declare variable oko_payed_other1 curr_type;
declare variable oko_payed_os_per1 curr_type;
declare variable oko_payed_perc_per1 curr_type;
declare variable oko_payed_other_per1 curr_type;
declare variable oko_payed_total_per1 curr_type;
declare variable oko_payed_total1 curr_type;
begin
-- knclient_fix

  for select sba_from_k_id, sba_kf_name, sba_kontract_no, sba_disb_date, sba_date_begin, sba_region_name,
             aa_curr_branch_name, sba_lo_fio, sbl_type_str, sbl_name, sbl_team_name, aarep_clients_kol,
             aarep_cashier_fio, aarep_leader_fio, oko_payed_perc, oko_payed_os, oko_balance_total, oko_balance_perc,
             oko_balance_other, oko_balance_os, oko_payed_other, oko_payed_os_per, oko_payed_perc_per,
             oko_payed_other_per, oko_payed_total_per, oko_payed_total, sba_amount_approved
      from aarep_kroff_sb(:rep_region_id, :rep_date_begin, :rep_date_end)
      into :sba_from_k_id, :sba_kf_name, :sba_kontract_no, :sba_disb_date, :sba_date_begin, :sba_region_name,
           :aa_curr_branch_name, :sba_lo_fio, :sbl_type_str, :sbl_name, :sbl_team_name, :aarep_clients_kol,
           :aarep_cashier_fio, :aarep_leader_fio, :oko_payed_perc, :oko_payed_os, :oko_balance_total, :oko_balance_perc,
           :oko_balance_other, :oko_balance_os, :oko_payed_other, :oko_payed_os_per, :oko_payed_perc_per,
           :oko_payed_other_per, :oko_payed_total_per, :oko_payed_total, :sba_amount_approved
  do
  begin
    oko_payed_perc1 = oko_payed_perc;
    oko_payed_os1 = oko_payed_os;
    oko_balance_total1 = oko_balance_total;
    oko_balance_perc1 = oko_balance_perc;
    oko_balance_other1 = oko_balance_other;
    oko_balance_os1 = oko_balance_os;
    oko_payed_other1 = oko_payed_other;
    oko_payed_os_per1 = oko_payed_os_per;
    oko_payed_perc_per1 = oko_payed_perc_per;
    oko_payed_other_per1 = oko_payed_other_per;
    oko_payed_total_per1 = oko_payed_total_per;
    oko_payed_total1 = oko_payed_total;

    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :sba_from_k_id and
              kc_amount_approved > 0
--     and KC_CLOSED_OL_ID is null -- не будет включать тех которые закрыли свою долю
        into kc_from_c_id, kc_amount_approved
    do
    begin
      dolya = kc_amount_approved / sba_amount_approved;
      aarep_clients_kol = 1;

      oko_payed_perc = dolya * oko_payed_perc1;
      oko_payed_os = dolya * oko_payed_os1;
      oko_balance_total = dolya * oko_balance_total1;
      oko_balance_perc = dolya * oko_balance_perc1;
      oko_balance_other = dolya * oko_balance_other1;
      oko_balance_os = dolya * oko_balance_os1;
      oko_payed_other = dolya * oko_payed_other1;
      oko_payed_os_per = dolya * oko_payed_os_per1;
      oko_payed_perc_per = dolya * oko_payed_perc_per1;
      oko_payed_other_per = dolya * oko_payed_other_per1;
      oko_payed_total_per = dolya * oko_payed_total_per1;
      oko_payed_total = dolya * oko_payed_total1;

      select c_name_fio, c_birth_date,
             (select cont_value
              from hp_contdata_get_cell_main(0, :kc_from_c_id))
      from clients
      where c_id = :kc_from_c_id
      into sba_client_name, sba_cl_birth_date, sba_cl_tel_no;

      c_address_yur = null;
      c_address_fact = null;

      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :kc_from_c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      into :c_address_yur;
      if (c_address_yur is null) then
        c_address_yur = 'Не указан';

      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :kc_from_c_id and
            adc_address_type = 2 and
            adc_from_adl_id = adl_id
      into :c_address_fact;
      if (c_address_fact is null) then
        c_address_fact = 'Не указан';

      suspend;
    end
  end
end
