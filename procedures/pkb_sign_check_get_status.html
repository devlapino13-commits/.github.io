CREATE PROCEDURE PKB_SIGN_CHECK_GET_STATUS(
    CSC_ID D_ID)
RETURNS (
    STATUS D_SMALLINT,
    STATUS_STR D_STR_CODE,
    CODE D_STR_20,
    F_MESSAGE D_STR_255,
    REASON D_STR_255,
    IIN D_STR_20,
    ORG_NAME D_STR_100,
    SIGN_DATE D_STR_CODE,
    FIRST_NAME D_STR_20,
    LAST_NAME D_STR_20,
    MIDDLE_NAME D_STR_20,
    F_ORDER D_STR_20)
AS
declare variable esm_id d_id;
declare variable json_str d_blob_text;
declare variable array_size integer;
declare variable i integer;
begin
    --  УСПЕШНЫЙ КЕЙС
    --    json_str = '{
    --        "result": {
    --            "data": {
    --                "code": 7,
    --                "message": "Отчет подлинный. Подпись соответствует контексту.",
    --                "data": [
    --                    {
    --                        "iin": "920418400902",
    --                        "org_name": "LLP First Credit Bureau",
    --                        "sign_date": "2025-07-09T18:37:11+05:00",
    --                        "first_name": "АЙЖАН",
    --                        "last_name": "ЕСИРКЕПОВА",
    --                        "middle_name": "КОМЕКБАЙКЫЗЫ",
    --                        "order": 1
    --                    }
    --                ]
    --            }
    --        }
    --    }';
    
    --НЕУДАЧНЫЙ КЕЙС
    --    json_str = '{
    --        "error": {
    --            "code": -610,
    --            "message": "Некорректный тип файла или в документе отсутствует подпись.",
    --            "reason": "ERRCODE_SERVER_RESPONDED_WITH_FATAL_ERROR"
    --        }
    --    }';

    select
        csc_status,
        csc_from_esm_id
    from ces_sign_check
    where
        csc_id = :csc_id
        and csc_project = 1
    into
        :status,
        :esm_id;

    status_str =
        case
            when status = 0 then 'запрос в обработке, ожидайте результат ...'
            when status = 1 then 'запрос успешно обработан'
            when status = -1 then 'ошибка обработки запроса'
        end;

    select
        esa_message
    from extsysanswer
    where esa_from_esm_id = :esm_id
    into
        :json_str;

    code = '';
    f_message = '';
    iin = '';
    org_name = '';
    sign_date = '';
    first_name = '';
    last_name = '';
    middle_name = '';
    f_order = '';
    reason = '';

    if (json_str is not null) then
    begin
        code = f_json_blob_get_value(:json_str, 'error.code', '');
        f_message = f_json_blob_get_value(:json_str, 'error.message', '');
        reason = f_json_blob_get_value(:json_str, 'error.reason', '');

        if (reason = '') then
        begin
            code = f_json_blob_get_value(:json_str, 'result.data.code', '');
            f_message = f_json_blob_get_value(:json_str, 'result.data.message', '');

            iin = f_json_str_get_value(:json_str, 'result.data.data[0].iin', '');
            org_name = f_json_str_get_value(:json_str, 'result.data.data[0].org_name', '');
            sign_date = f_json_str_get_value(:json_str, 'result.data.data[0].sign_date', '');
            first_name = f_json_str_get_value(:json_str, 'result.data.data[0].first_name', '');
            last_name = f_json_str_get_value(:json_str, 'result.data.data[0].last_name', '');
            middle_name = f_json_str_get_value(:json_str, 'result.data.data[0].middle_name', '');
            f_order = f_json_str_get_value(:json_str, 'result.data.data[0].order', '');
        end
    end
    suspend;
end
