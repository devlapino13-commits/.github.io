CREATE PROCEDURE CM_AZ_0033_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable PR_CODE D_STR_100;
declare variable PR_CODE_LAST D_STR_100;
declare variable C_ID D_ID;
declare variable K_OLD_OL_DATE D_DATE;
declare variable K_OLD_DELINQ_DATE D_DATE;
declare variable K_OL_DEL_DAY_TMP integer;
declare variable SUM_DELINQ_DAYS integer;
declare variable MAX_DELINQ_DAYS integer;
declare variable K_ACTIV integer;
declare variable K_ID_LAST ID_TYPE;
declare variable K_DISB_DATE_LAST date;
declare variable KC_AMOUNT_LAST CURR_TYPE;
declare variable K_LAST_BALANCE_TMP CURR_TYPE;
declare variable KC_LAST_BALANCE CURR_TYPE;
begin
  for select KC_FROM_C_ID, KC_AMOUNT_APPROVED, K_DISB_DATE, PR_CODE
      from KNCLIENT KC, KONTRACT, PRODUCT
      where KC_FROM_K_ID = :K_ID and
            KC_AMOUNT_APPROVED > 0 and
            KC_FROM_K_ID = K_ID and
            K_FROM_PR_ID = PR_ID
      into C_ID, KC_AMOUNT, KF_DISB_DATE, PR_CODE
  do
  begin
    KC_AMOUNT_LAST = null;
    K_ID_LAST = null;
    for select KC_AMOUNT_APPROVED, PR_CODE, K_ID, K_DISB_DATE
        from KNCLIENT KC, KONTSTATE, KONTRACT, PRODUCT
        where KC_FROM_C_ID = :C_ID and
              KC_FROM_K_ID = K_ID and
              KS_OL_IS_DELETE = 0 and
              KS_STATE = 1 and
              :KF_DISB_DATE between KS_DATE_BEGIN and KS_DATE_END and
              KS_FROM_K_ID = K_ID and
              KS_FROM_K_ID <> :K_ID and
              K_FROM_PR_ID = PR_ID
        order by K_DISB_DATE desc
        into KC_AMOUNT_LAST, PR_CODE_LAST, K_ID_LAST, K_DISB_DATE_LAST
    do
    begin
      K_ACTIV = coalesce(K_ACTIV, 0) + 1;
      --e.       На одного клиента не может быть более 2 (двух) кредитов
      if (K_ACTIV > 1) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Bir müştəri üzrə 2 (iki) aktiv kreditdən çox ola bilməz';
        suspend;
      end
      --После выдачи основного кредита еще 3 месяца не прошло.
      if ((KF_DISB_DATE - K_DISB_DATE_LAST) / 30 < 3) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Əsas kreditin buraxılış tarixindən 3 ay keçməyib';
        suspend;
      end
      for select max(iif(OKS_DATE_END >= current_date, current_date, OKS_DATE_END + 1)) OL_DATE, OKS_DELINQ_DATE,
                 OKS_BALANCE
          from OKSUMMARY
          where oks_close_state = 0 and
                OKS_FROM_K_ID = :K_ID_LAST and
                OKS_DELINQ_DATE <= :KF_DISB_DATE and
                OKS_DELINQ_DATE <= OKS_DATE_END
          group by OKS_DELINQ_DATE, OKS_BALANCE
          into K_OLD_OL_DATE, K_OLD_DELINQ_DATE, :K_LAST_BALANCE_TMP
      do
      begin
        K_OL_DEL_DAY_TMP = coalesce(K_OLD_OL_DATE - K_OLD_DELINQ_DATE, 0);
        if (coalesce(MAX_DELINQ_DAYS, 0) < K_OL_DEL_DAY_TMP) then
          MAX_DELINQ_DAYS = K_OL_DEL_DAY_TMP;
        SUM_DELINQ_DAYS = coalesce(SUM_DELINQ_DAYS, 0) + K_OL_DEL_DAY_TMP;
        if (K_OLD_OL_DATE = current_date) then
          KC_LAST_BALANCE = K_LAST_BALANCE_TMP;
      end
      MAX_DELINQ_DAYS = coalesce(MAX_DELINQ_DAYS, 0);
      SUM_DELINQ_DAYS = coalesce(SUM_DELINQ_DAYS, 0);
      if (SUM_DELINQ_DAYS > 0) then
      begin
        --
        res_error_text = 'Aktiv kreditlər üzrə gecikmə günlərin say';
        res_error_is_critical = 1;
      end
      --Общий баланс основного и параллельного кредита превышает 3000 манатов
      if (KC_LAST_BALANCE + KC_AMOUNT > 3000.00) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Əsas və parallel kreditin balansı 3000 AZN-dən artıqdır';
        suspend;
      end
    end
  end
end
