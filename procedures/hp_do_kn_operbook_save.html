CREATE PROCEDURE HP_DO_KN_OPERBOOK_SAVE(
    OL_ID D_ID,
    LOAN_OL_ID D_ID,
    KC_ID D_ID,
    OPER_KIND D_STR_CODE,
    REG_ID D_INTEGER,
    OL_DATE D_DATE,
    OL_IS_ACTIVE D_INTEGER)
AS
declare variable dt_acc_id id_type;
declare variable kt_acc_id id_type;
declare variable loan_acc_group d_str_code;
declare variable amount curr_type;
declare variable amount_os curr_type;
declare variable amount_perc curr_type;
declare variable amount_fine curr_type;
declare variable kt_sys_acc d_str_code;
declare variable ob_detail d_str_255;
declare variable dt_sys_acc d_str_code;
declare variable amount_kommis curr_type;
declare variable is_adv integer;
declare variable is_closed integer;
declare variable amount_adv_os curr_type;
declare variable loan_in_reestr integer;
declare variable amount_tax curr_type;
declare variable reserv_kind integer;
declare variable tmp_amount curr_type;
declare variable non_resident smallint;
declare variable client_type smallint;
declare variable loan_reg_id integer;
declare variable amount_plus curr_type;
declare variable amount_minus curr_type;
declare variable reserv_kind_prv d_integer;
declare variable acc_id d_id;
declare variable amount_pereplat d_currency;
declare variable alt_acc_id d_id;
declare variable alt_acc_name d_str_code;
declare variable alt_sys_acc d_str_code;
declare variable orig_ol_id d_id;
declare variable orig_ol_date d_date;
declare variable is_reversed d_smallint;
declare variable new_ol_id d_id;
declare variable k_id d_id;
declare variable cl_count d_integer;
declare variable zalog_cost d_currency;
declare variable rl_ol_is_delete d_smallint;
declare variable sab_full_code d_str_code;
begin
-- процедура сохранения проводок по новому плану счетов
-- для ранее сохраненной операции

  -- Не формировать проводки ЕПС кроме РФ даже если учет идет в KN_OKSUMMARY (KZ)
  if (f_get_db_id() <> 81001) then
    exit;

-- 101010011087900101
-- 1010100001467149

  if (coalesce(oper_kind, '') = '') then
    exception e_default 'Oper_kind is empty';

  if (oper_kind not in ('disb', 'kommis', 'repay', 'pereplat', 'operkroff', 'operextra', 'operextraoff', 'money', 'writeoff', 'operkroffoff')) then
    exception e_default 'Incorrect oper kind';

  is_adv = 0;
  is_closed = 0;

  is_reversed = 0;
  for select rl_old_ol_id, rl_new_ol_id, rl_date, rl_ol_is_delete
  from reversallist
  where rl_new_ol_id in (:ol_id, :loan_ol_id)
  into :orig_ol_id, :new_ol_id, :orig_ol_date, rl_ol_is_delete
  do
  begin
    delete from kn_operbook
    where kob_from_ol_id = :new_ol_id;

    insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa,
                             kob_summa_val, kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id,
                             kob_ol_is_active, kob_detail_id, kob_client_dt, kob_client_kt, kob_from_kc_id)
    select :new_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, -kob_summa, -kob_summa_val,
           -kob_summa_tax_dt, -kob_summa_tax_kt, :orig_ol_date, kob_ol_reg_id, 1-:rl_ol_is_delete, kob_detail_id,
           kob_client_dt, kob_client_kt, kob_from_kc_id
    from kn_operbook
    where kob_from_ol_id = :orig_ol_id;

    is_reversed = 1;
  end
  if (is_reversed = 1) then Exit;

-- проверка наличия счетов
  if (ol_date is null or ol_is_active is null) then
  begin
    select ol_date, 1 - is_delete
    from operlist
    where ol_id = :ol_id
    into ol_date, :ol_is_active;
  end

  select opl_from_acc_id, sab_full_code
  from operloan, ref_account, sys_accbalance
  where opl_from_ol_id = :ol_id and
        opl_from_acc_id = acc_id and
        acc_from_sab_id = sab_id
  into acc_id, sab_full_code;

  if (acc_id is null) then
    acc_id = 0;

  select non_resident, client_type
  from hp_get_kn_client_resident(:kc_id, :ol_date)
  into non_resident, client_type;

  loan_acc_group = 'Микрозаймы'; -- в будущем нужно прикрепить название раздела системного счета в таблицу продуктов.

  if (oper_kind = 'disb') then
  begin
-- сумма выдач в разрезе договоров участников
    for select sum(iif(ok_type = 8, ok_os, 0.0)), sum(iif(ok_type = 1, ok_sh_perc, 0.0))
        from kn_operkred
        where ok_from_ol_id = :loan_ol_id and
              ok_type in (1, 8) and
              ok_from_kc_id = :kc_id
        into amount, tmp_amount
    do
    begin
      if (amount is null) then
        amount = 0;
      if (tmp_amount is null) then
        tmp_amount = 0;
      if (amount <> 0) then
      begin
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, 'ОС.Баланс', :kc_id, :non_resident, :client_type, :ol_date)
        into dt_acc_id;

        kt_acc_id = acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, 'Выдача микрозайма');

        if (exists(select *
                   from kontract, knclient
                   where k_reserv_kind = 2 -- по обеспеченным займам
                         and
                         kc_id = :kc_id and
                         k_id = kc_from_k_id)) then
        begin
          cl_count = 0;
          select kc_from_k_id from knclient
          where kc_id = :kc_id
          into :k_id;

          select count(*)
          from knclient
          where kc_from_k_id = :k_id
          and kc_amount_approved > 0
          into cl_count;

          amount = amount + tmp_amount;
          if (amount <> 0 and cl_count > 1) then -- групповые займы
          begin
            select acc_id
            from get_ref_account_knclient(:loan_acc_group, 'Поручительство.ВнеБаланс', :kc_id, :non_resident, :client_type, :ol_date)
            into dt_acc_id;

            select acc_id
            from get_ref_account_office('Внебаланс', '99999', :reg_id, '', -1, :non_resident, :client_type, :ol_date)
            into kt_acc_id;

            execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
                :ol_date, :reg_id, :ol_is_active,
                'Поручительства по обеспеченному займу за балансом');
          end

          zalog_cost = 0;
          select sum(z_cost_likvid)
          from zalog
          where z_from_k_id = :k_id and
                coalesce(z_cost_likvid, 0) <> 0
          into zalog_cost;
          if (zalog_cost is null) then zalog_cost = 0;

          if (zalog_cost <> 0) then
          begin
            amount = zalog_cost;
            select acc_id
            from get_ref_account_office('Внебаланс', '99998', :reg_id, '', -1, :non_resident, :client_type, :ol_date)
            into dt_acc_id;

            select acc_id
            from get_ref_account_knclient(:loan_acc_group, 'Залог.ВнеБаланс', :kc_id, :non_resident, :client_type, :ol_date)
            into kt_acc_id;

            execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
                :ol_date, :reg_id, :ol_is_active,
                'Залоговая стоимость по займу за балансом');
          end
        end
      end
    end
  end

  if (oper_kind = 'kommis') then
  begin
-- сумма доп.процента в разрезе договоров участников
    for select sum(-ok_kommis)
        from kn_operkred
        where ok_from_ol_id = :loan_ol_id and
              ok_type = 4 and
              ok_from_kc_id = :kc_id
        into amount
    do
    begin
      if (amount is null) then
        amount = 0;
      if (amount <> 0) then
      begin
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, 'ДопПроцент.Погашено', :kc_id, :non_resident, :client_type, :ol_date)
        into :kt_acc_id;

        dt_acc_id = acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, 'Погашение доп процента');
      end
    end
  end

  if (oper_kind in ('repay', 'writeoff')) then
  begin
-- погашения по займу
    is_adv = 0;
    is_closed = 0;
    if (exists(select *
               from kn_operkred
               where ok_from_ol_id = :loan_ol_id and
                     ok_type in (4, 5) and
                     ok_from_kc_id = :kc_id and
                     ok_ol_date < ok_sh_date and
                     ok_os <> 0)) then
      is_adv = 1;

    if (exists(select *
               from kontstate, knclient
               where ks_from_ol_id = :loan_ol_id and
                     ks_from_k_id = kc_from_k_id and
                     kc_id = :kc_id and
                     ks_state = 2)) then
      is_closed = 1;
  end

  if (oper_kind = 'repay') then
  begin
-- погашения по займу
    for select ok_ol_reg_id, sum(iif(ok_type = 4, -ok_os, 0)),
               sum(iif(:is_adv = 1 and :is_closed = 1 and ok_ol_date < ok_sh_date and ok_type = 4, -ok_os, 0)),
               sum(-ok_perc), sum(-ok_fine), sum(-ok_kommis_adv), sum(iif(ok_type = 9, -ok_os, 0))
        from kn_operkred
        where ok_from_ol_id = :loan_ol_id and
              ok_from_kc_id = :kc_id and
              ok_type in (4, 9)
        group by ok_ol_reg_id
        into :loan_reg_id, amount_os, amount_adv_os, amount_perc, amount_fine, amount_kommis, amount_pereplat
    do
    begin
      if (amount_os is null) then
        amount_os = 0;
      if (amount_adv_os is null) then
        amount_adv_os = 0;
      if (amount_perc is null) then
        amount_perc = 0;
      if (amount_fine is null) then
        amount_fine = 0;
      if (amount_kommis is null) then
        amount_kommis = 0;
      if (amount_adv_os <> 0) then
        amount_os = amount_os - amount_adv_os;
      if (amount_pereplat is null) then
        amount_pereplat = 0;

      if (amount_perc < 0) then
      begin
        if (amount_adv_os >= -amount_perc) then
        begin
          amount_adv_os = amount_adv_os + amount_perc;
          amount_os = amount_os - amount_perc;
        end
      end

      dt_acc_id = acc_id;

      alt_acc_name = ''; -- название показателя, за счет которого погашаются другие показатели
      if (coalesce(acc_id, 0) = 0 and
           amount_os + amount_adv_os + amount_perc + amount_fine + amount_pereplat + amount_kommis = 0 and
          (amount_os <> 0 or amount_perc <> 0 or amount_fine <> 0 or amount_pereplat <> 0 or amount_kommis <> 0)) then
      begin
        if (amount_os + amount_adv_os < 0) then
          alt_acc_name = 'os';
        else
        if (amount_perc < 0) then
          alt_acc_name = 'perc';
        else
        if (amount_fine < 0) then
          alt_acc_name = 'fine';
        else
        if (amount_pereplat < 0) then
          alt_acc_name = 'pereplat';
        else
        if (amount_kommis < 0) then
          alt_acc_name = 'kommis';
      end

      if (alt_acc_name = 'os' and
          (amount_perc < 0 or amount_fine < 0 or amount_pereplat < 0 or amount_kommis < 0)) then
        alt_acc_name = '';

      if (alt_acc_name = 'perc' and
          (amount_os + amount_adv_os < 0 or amount_fine < 0 or amount_pereplat < 0 or amount_kommis < 0)) then
        alt_acc_name = '';

      if (alt_acc_name = 'fine' and
          (amount_os + amount_adv_os < 0 or amount_perc < 0 or amount_pereplat < 0 or amount_kommis < 0)) then
        alt_acc_name = '';

      if (alt_acc_name = 'pereplat' and
          (amount_os + amount_adv_os < 0 or amount_perc < 0 or amount_fine < 0 or amount_kommis < 0)) then
        alt_acc_name = '';

      if (alt_acc_name = 'kommis' and
          (amount_os + amount_adv_os < 0 or amount_perc < 0 or amount_fine < 0 or amount_pereplat < 0)) then
        alt_acc_name = '';

      alt_acc_id = null;
      if (alt_acc_name <> '') then
      begin
        if (alt_acc_name = 'os') then
          alt_sys_acc = 'ОС.Баланс';
        if (alt_acc_name = 'perc') then
          alt_sys_acc = 'Процент.Погашено';
        if (alt_acc_name = 'fine') then
          alt_sys_acc = 'Пеня.Погашено';
        if (alt_acc_name = 'pereplat') then
          alt_sys_acc = 'Переплата.Погашено';
        if (alt_acc_name = 'kommis') then
          alt_sys_acc = 'КомиссияЗаДосроч.Погашено';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :alt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into alt_acc_id;
      end

      if (amount_os <> 0 and
          alt_acc_name <> 'os') then
      begin
        dt_acc_id = iif(alt_acc_name = '', acc_id, alt_acc_id);

        amount = amount_os;
        kt_sys_acc = 'ОС.Баланс';
        ob_detail = 'Погашение основного долга';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);
      end

      if (amount_adv_os <> 0 and
          alt_acc_name = '') then
      begin
        amount = amount_adv_os;
        kt_sys_acc = 'ОС.Баланс';
        ob_detail = 'Погашение основного долга';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into kt_acc_id;

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, 'Реализация.Погашено', :kc_id, :non_resident, :client_type, :ol_date)
        into dt_acc_id;

        ob_detail = 'Досрочное погашение основного долга';
        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);

        kt_acc_id = dt_acc_id;
        dt_acc_id = acc_id;

        amount = amount_adv_os;
        ob_detail = 'Погашение досрочно погашен. основного долга';
        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);
      end

      if (amount_perc <> 0 and
          alt_acc_name <> 'perc') then
      begin
        dt_acc_id = iif(alt_acc_name = '', acc_id, alt_acc_id);

        amount = amount_perc;
        kt_sys_acc = 'Процент.Погашено';
        ob_detail = 'Погашение процента';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);

        amount = amount_perc;
        dt_acc_id = kt_acc_id;
        ob_detail = 'Списание доначисленного процента';
        kt_sys_acc = 'Процент.Начислено';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);
      end

      if (amount_fine <> 0 and
          alt_acc_name <> 'fine') then
      begin
        dt_acc_id = iif(alt_acc_name = '', acc_id, alt_acc_id);

        amount = amount_fine;
        kt_sys_acc = 'Пеня.Погашено';
        ob_detail = 'Погашение пени';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into :kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);

        dt_acc_id = kt_acc_id;
        kt_sys_acc = 'Пеня.Доход';
        ob_detail = 'Начисление пени';

        select acc_id
        from get_ref_account_office(:loan_acc_group, :kt_sys_acc, :loan_reg_id, '', -1, :non_resident, :client_type, :ol_date)
        into kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);
      end

      if (amount_kommis <> 0 and
          alt_acc_name <> 'kommis') then
      begin
        dt_acc_id = iif(alt_acc_name = '', acc_id, alt_acc_id);

        amount = amount_kommis;
        kt_sys_acc = 'КомиссияЗаДосроч.Погашено';
        ob_detail = 'Погашение комиссии за досрочное полное погашение';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);

        dt_sys_acc = 'КомиссияЗаДосроч.Начислено';
        kt_sys_acc = 'КомиссияЗаДосроч.Доход';
        ob_detail = 'Начисление комиссии за досрочное погашение погашение';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :dt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into :dt_acc_id;

        select acc_id
        from get_ref_account_office(:loan_acc_group, :kt_sys_acc, :loan_reg_id, '', -1, :non_resident, :client_type, :ol_date)
        into :kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);

        dt_sys_acc = 'КомиссияЗаДосроч.Погашено';
        kt_acc_id = dt_acc_id;

        ob_detail = 'Списание доначисленного комиссии за досрочку на счет расчетов по комиссии';
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :dt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into dt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);
      end

      if (amount_pereplat <> 0 and
          alt_acc_name <> 'pereplat') then
      begin
        dt_acc_id = iif(alt_acc_name = '', acc_id, alt_acc_id);

        amount = amount_pereplat;
        kt_sys_acc = 'Переплата.Погашено';
        ob_detail = 'Переплата по займу';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into :kt_acc_id;

        execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, :amount, 0, :amount,
            :ol_date, :reg_id, :ol_is_active, :ob_detail);
      end

-- если погашались за счет процента, сторнируем ранее начисленные
      if (amount_perc < 0 and
          alt_acc_name = 'perc') then
      begin
        amount = amount_perc;
        ob_detail = 'Списание доначисленного процента';
        dt_sys_acc = 'Процент.Погашено';
        kt_sys_acc = 'Процент.Начислено';

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :dt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into dt_acc_id;

        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into kt_acc_id;
