CREATE PROCEDURE FT_START_RELATION_CALC(
    INFO_DATE D_DATE,
    C_ID D_ID)
RETURNS (
    START_RELATION_DATE D_DATE)
AS
declare variable FIL_DATE_OPEN D_DATE;
declare variable FIL_DATE_CLOSE D_DATE;
declare variable FIL_DATE_OPEN_OLD D_DATE;
begin
    start_relation_date = null;
    fil_date_open_old = null;

    for select fil.fil_date_open,
               fil.fil_date_close
    from ft_infolist fil
    where fil.fil_from_c_id = :c_id
    and fil.fil_date_open <= :info_date
    order by fil.fil_date_open  desc
    into :fil_date_open, :fil_date_close
    do
    begin
        if (start_relation_date is null) then
          start_relation_date = fil_date_open;

        if (fil_date_open_old is null) then
        begin
            fil_date_open_old = fil_date_open;
            continue;
        end

       if (fil_date_close + 1 = fil_date_open_old) then
       begin
            start_relation_date = fil_date_open;
            fil_date_open_old = fil_date_open;
            continue;
       end
       else
       begin
            start_relation_date = fil_date_open_old;
            suspend;
            exit;
       end
    end
    if (start_relation_date is not null) then
    begin
        suspend;
        exit;
    end
end
