CREATE PROCEDURE HP_FINCERT_GET_RESULT(
    C_ID D_ID,
    CHECK_DATE_TIME D_DATETIME)
RETURNS (
    RES D_SMALLINT,
    RES_TYPE D_STR_20,
    FF_CREATE_DATETIME D_DATETIME,
    BLOCK_DATE D_DATE,
    RES_CODE D_STR_CODE)
AS
declare variable c_pass_no d_str_code;
declare variable c_inn d_str_20;
declare variable phone_number d_str_20;
declare variable snils d_str_code;
begin
  -- Процедура возвращает результаты проверки по фидам ФинЦЕРТ по клиенту на момент времени
  -- RFCRU-3955
  select c_pass_no, c_inn, f_cont_info('clients', c_id, 'cell_main', 'ok')
  from clients
  where c_id = :c_id
  into c_pass_no, c_inn, phone_number;

  res_type = 'Паспорт';
  res_code = 'passport_hash';
  res = 0;
  ff_create_datetime = null;
  block_date = null;

  if (coalesce(c_pass_no, '') <> '') then
    select coalesce(res_status, 0), ff_create_datetime, res_date
    from hp_fincert_passport_check(:c_pass_no, :check_date_time)
    into res, ff_create_datetime, block_date;

  suspend;

  res_type = 'ИНН';
  res_code = 'inn';
  res = 0;
  ff_create_datetime = null;
  block_date = null;

  if (coalesce(c_inn, '') <> '') then
    select coalesce(res_status, 0), ff_create_datetime, res_date
    from hp_fincert_inn_check(:c_inn, :check_date_time)
    into res, ff_create_datetime, block_date;

  suspend;

  -- RFCRU-4205
  res_type = 'СНИЛС';
  res_code = 'snils';
  res = 0;
  ff_create_datetime = null;
  block_date = null;

  select cds_doc_no
  from clientdocs, passkind, passdoctype
  where cds_from_c_id = :c_id
    and cds_from_pk_id = pk_id
    and pk_from_pdt_id = pdt_id
    and pdt_code = 'snils'
    and cds_is_delete = 0
  order by cds_from_c_id desc, cds_doc_date_begin desc
  rows 1
  into snils;

  if (coalesce(snils, '') <> '') then
    select coalesce(res_status, 0), ff_create_datetime, res_date
    from hp_fincert_snils_check(:snils, :check_date_time)
    into res, ff_create_datetime, block_date;

  suspend;

  -- RFCRU-4164
  res_type = 'Номер телефона';
  res_code = 'phone_number';
  res = 0;
  ff_create_datetime = null;
  block_date = null;

  if (coalesce(phone_number, '') <> '') then
    select coalesce(res_status, 0), ff_create_datetime, res_date, res_code
    from hp_fincert_phone_check(:phone_number, :check_date_time)
    into res, ff_create_datetime, block_date, res_code;

  suspend;
end
