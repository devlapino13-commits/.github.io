CREATE PROCEDURE HP_CREATE_MENU_LIST
RETURNS (
    ML_ID INTEGER,
    ML_PARENT_ID INTEGER,
    ML_CODE VARCHAR(100),
    ML_CAPTION VARCHAR(255),
    ML_DESCRIPTION VARCHAR(1024))
AS
begin
  for select rl_id, parent, rl_name_rus, rl_hint_rus
      from reportlist
      where rl_id > 0
      into ml_id, ml_parent_id, ml_caption, ml_description
  do
  begin
    if (not(exists(select *
                   from menulist
                   where ml_id = :ml_id))) then
    begin
      if (ml_parent_id = 0) then
        ml_parent_id = null;
      if (ml_parent_id is not null) then
      begin
        if (exists(select *
                   from menulist
                   where ml_id = :ml_parent_id)) then
        begin
          insert into menulist (ml_id, ml_parent_id, ml_code, ml_caption, ml_description, ml_is_report, ml_secure_kind)
          values (:ml_id, :ml_parent_id, 'Report_' || :ml_id, :ml_caption, :ml_description, 1, 0);
          suspend;
        end
        else
        begin
          ml_description = 'ТРЕБУЕТСЯ ПОВТОРНЫЙ ЗАПУСК';
          suspend;
        end
      end
      else
      begin
        insert into menulist (ml_id, ml_parent_id, ml_code, ml_caption, ml_description, ml_is_report, ml_secure_kind)
        values (:ml_id, :ml_parent_id, 'Report_' || :ml_id, :ml_caption, :ml_description, 1, 0);
        suspend;
      end
    end
  end
end
