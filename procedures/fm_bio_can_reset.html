CREATE PROCEDURE FM_BIO_CAN_RESET(
    UN_NAME VARCHAR(30))
RETURNS (
    ERR_TEXT D_STR_255,
    IS_ALLOWED D_SMALLINT)
AS
declare variable C_CITIZEN D_INTEGER;
declare variable C_ID D_ID;
declare variable JOB_KIND D_STR_CODE;
begin
    ----------RFCKG-3931---------
    err_text = '';
    is_allowed = 0;

    select err_text
    from fm_user_can_reset_pwd(:un_name)
    into :err_text;

    if (err_text <> '') then
    begin
        suspend;
        exit;
    end

    select first 1
        c.c_citizen,
        c.c_id
    from username un
    inner join clients c on c.c_id = un.un_from_c_id
    where un.un_name =:un_name
    into
        :c_citizen,
        :c_id;

    if (:c_citizen <> f_get_rd_id('Citizen.KG')) then
    begin
        is_allowed = 2;
        suspend;
        exit;
    end

    if (exists
        (
            select 1
            from hr_pstatus ps
            inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
            inner join jobkind jk on jl.jl_kind = jk.jk_id
                and jk.jk_system_code not in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.Region_KM')
            where current_date between ps.ps_date_begin and ps.ps_date_end
                and ps.ps_from_c_id = :c_id
                and ps_is_working = 1
        )
    ) then
    begin
        is_allowed = 2;
        suspend;
        exit;
    end

    if (exists
        (
            select 1
            from extinfolist eil
            inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
            inner join extinfolistbls eib on eib.eib_from_eil_id = eil.eil_id
            where eil.eil_from_c_id = :c_id and
                 eic.eic_code = 'Tunduk' and
                 eil.eil_status = 2
        )
    ) then
    begin
        is_allowed = 1;
        suspend;
        exit;
    end

    err_text =  'Необходимо добавить фото с ГРС, обратитесь в отдел кадров';
    suspend;
end
