CREATE PROCEDURE UXP$UXMAINMENU_LIST_TMP(
    UN_ID D_ID,
    APP_NAME D_STR_CODE)
RETURNS (
    UML_ML_ID D_INTEGER,
    UML_UA_CODE D_STR_100,
    UA_IMAGE_LINKED D_BOOL)
AS
declare variable ua_uxa_uuid d_uuid;
declare variable is_fs_app d_bool;
declare variable uml_ua_uuid d_uuid;
declare variable uml_fs_disabled d_bool;
declare variable uml_web_disabled d_bool;
begin
  is_fs_app = 0;
  if (app_name in ('FinStart', 'Finance')) then
    is_fs_app = 1;

  for select uml_ml_id, uml_ua_uuid, ua_uxa_uuid, iif(ua_img_uuid is not null, 1, 0),
             uml_fs_disabled, uml_web_disabled
  from ux$menulist, ux$action
  where uml_ua_uuid = ua_uuid and
       (:is_fs_app = 0 or uml_fs_disabled = 0)
  into uml_ml_id, uml_ua_uuid, ua_uxa_uuid, ua_image_linked,
       uml_fs_disabled, uml_web_disabled
  do
  begin
    uml_ua_code = '';

    if (is_fs_app = 1 and uml_fs_disabled = 1) then
      Continue;

    if (app_name in ('WebStart', 'FsOnline') and uml_web_disabled = 1) then
      Continue;

    if (uxf$access(un_id, ua_uxa_uuid, app_name) = 1) then
    begin
      uml_ua_code = uuid_to_char(uml_ua_uuid);
      suspend;

      uml_ua_code = '';
      while (uml_ml_id is not null) do
      begin
        for select ml_parent_id
        from menulist
        where ml_id = :uml_ml_id
        into uml_ml_id
        do
        begin
          if (uml_ml_id is not null) then
            suspend;
        end
      end
    end
  end
end
