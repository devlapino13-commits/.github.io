CREATE PROCEDURE AAREP_CELL_OPERS(
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER,
    REPORT_DATE DATE,
    REPORT_PLACE INTEGER)
RETURNS (
    CB_CELL_NO VARCHAR(20),
    CB_AMOUNT DECIMAL(18,4),
    CAL_FROM_C_ID DECIMAL(18,0),
    CAL_STATE INTEGER,
    CAL_IS_CONTACT INTEGER,
    CAL_IS_CORPORATE INTEGER,
    CAL_HAS_LIMIT INTEGER,
    CAL_FROM_COP_ID INTEGER,
    CAL_FIO VARCHAR(100),
    CAL_REGION_ID INTEGER,
    CAL_BRANCH_ID INTEGER,
    CAL_FROM_JL_ID INTEGER,
    CAL_FROM_JL_NAME VARCHAR(100),
    CAL_REGION_NAME VARCHAR(100),
    CAL_LIMIT_AMOUNT DECIMAL(18,4),
    CB_AMOUNT_WITH_LIMIT DECIMAL(18,4),
    CAL_IS_WORKING INTEGER,
    CAL_HEAD_OFFICE INTEGER,
    CLK_LIMIT_KIND VARCHAR(100),
    CLK_LIMIT_AMOUNT DECIMAL(18,4))
AS
declare variable cvip_cell_no varchar(20);
begin
  for select cbl_cell_no, sum(cbl_amount)
      from cellbilloper, cellbill
      where cbl_bill_no = cb_bill_no and
            cb_year = :report_year and
            cb_month = :report_month
      group by cbl_cell_no
      having sum(cbl_amount) <> 0
      into :cb_cell_no, :cb_amount
  do
  begin
    cb_amount_with_limit = 0;
    cal_limit_amount = 0;
    cal_from_c_id = null;
    cal_state = -2;
    cal_is_contact = 0;
    cal_is_corporate = 0;
    cal_has_limit = 0;
    cal_from_cop_id = -2;
    cal_fio = '';
    cal_region_id = :report_place;
    cal_branch_id = :report_place;
    cal_from_jl_id = 0;
    cal_from_jl_name = '';
    cal_region_name = '';
    cvip_cell_no = '';
    cal_is_working = 0;

    select cal_from_c_id, cal_state, cal_is_contact, cal_is_corporate, cal_has_limit, cal_from_cop_id, cal_fio,
           cal_region_id, cal_branch_id, cal_from_jl_id, cal_from_jl_name, cal_region_name, cal_is_working,
           reg_head_office, clk_limit_kind, clk_limit_amount
    from aarep_cellaktlist_one(:cb_cell_no, :report_date), region
    where cal_region_id = reg_unicode
    into :cal_from_c_id, :cal_state, :cal_is_contact, :cal_is_corporate, :cal_has_limit, :cal_from_cop_id, :cal_fio,
         :cal_region_id, :cal_branch_id, :cal_from_jl_id, :cal_from_jl_name, :cal_region_name, :cal_is_working,
         :cal_head_office, :clk_limit_kind, :clk_limit_amount;

    select cvip_cell_no
    from cellvip
    where cvip_cell_no = :cb_cell_no and
          cvip_date_begin <= :report_date and
          cvip_date_end >= :report_date rows 1
    into :cvip_cell_no;
    if (cvip_cell_no is null) then
      cvip_cell_no = '';
    if (cal_region_name is null) then
      cal_region_name = '';
    if (cal_region_name = '') then
    begin
      select reg_name, reg_head_office
      from region
      where reg_unicode = :cal_region_id
      into :cal_region_name, cal_head_office;
    end
    if (cvip_cell_no <> '') then
    begin
      cal_limit_amount = cb_amount;
      cal_state = 2;
      cb_amount_with_limit = cb_amount;
      cb_amount = 0;
    end
    else
    begin
      if (cal_from_c_id is null) then
      begin
        cb_amount_with_limit = cb_amount;
        cal_limit_amount = cb_amount;
        cb_amount = 0;
      end
      else
      begin
        if (cal_has_limit = 1) then
        begin
          cb_amount_with_limit = cb_amount;

          -- S.K.
          -- cal_limit_amount = cb_amount;
          cal_limit_amount = (
            select sum(cjl_amount)
            from celljoblimit
            where cjl_from_jl_id = :cal_from_jl_id and
              cjl_date <= :report_date);

          if (cal_limit_amount is null) then
          begin
            exception e_message(f_format('У %1 отсутствует лимит. Поставьте лимит по должности "%2"', :cal_fio, :cal_from_jl_name));     -- RFCKG-4684
          end

          cb_amount = 0;
        end
        else
        begin
          if (cal_has_limit > 1) then
          begin
            cb_amount_with_limit = cb_amount;
            cal_limit_amount = clk_limit_amount;
            cb_amount = 0;
          end
        end
      end
    end
    cal_branch_id = cal_head_office;
    suspend;
  end
end
