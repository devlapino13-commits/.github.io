CREATE PROCEDURE AAREP_NOTICE_OF_ATTORNEY(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    HR_REG_NAME VARCHAR(100),
    EMPLOYEE_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    DOV_REG_NAME VARCHAR(100),
    DC_NAME VARCHAR(100),
    DL_DATE_BEGIN DATE,
    DL_DATE_END DATE,
    NEXT_DATE_DL DATE,
    DL_MEMO VARCHAR(30))
AS
declare variable reg_dov_id integer;
declare variable employee_id id_type;
begin

  for select reg_hr.reg_code || ' ' || reg_hr.reg_name hr_reg_name, c_name_fio, jl_name,
             reg_dov.reg_code || ' ' || reg_dov.reg_name dov_reg_name, dc_name, dl_date_begin,
             coalesce(dl_date_end_fact, dl_date_end) as dl_date_end, c_id, reg_dov.reg_unicode
      from dovlist, dovclients, hr_pstatus, joblist, clients, region reg_dov, region reg_hr, dovcateg, region_list(:rep_region_id, 1, -1, 1)
      where :rep_date_end between dl_date_begin and dl_date_end and
            dl_status = 1 and
            dl_id = dvc_from_dl_id and
            dvc_from_c_id = ps_from_c_id and
            ps_is_working = 1 and
            ps_status = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_jl_id = jl_id and
            ps_from_c_id = c_id and
            dvc_from_reg_id = reg_dov.reg_unicode and
            ps_from_reg_id = reg_hr.reg_unicode and
            dl_from_dc_id = dc_id and
            dc_id in (5, 7, 1, 4)
--    and dvc_from_reg_id = id_list
            and
            ps_from_reg_id = id_list
--and dl_date_end between :rep_date_end - 7 and :rep_date_end + 7
            and
            coalesce(dl_date_end_fact, dl_date_end) between :rep_date_end - 7 and :rep_date_end + 7
      into hr_reg_name, employee_fio, jl_name, dov_reg_name, dc_name, dl_date_begin, dl_date_end, employee_id,
           reg_dov_id
  do
  begin
    next_date_dl = null;
    select dl_date_begin
    from dovlist, dovclients
    where :dl_date_end + 1 between dl_date_begin and coalesce(dl_date_end_fact, dl_date_end) and
          dl_status = 1 and
          dl_id = dvc_from_dl_id and
          dvc_from_c_id = :employee_id and
          dvc_from_reg_id = :reg_dov_id and
          dl_from_dc_id in (5, 7, 1, 4) rows 1
    into next_date_dl;

    dl_memo = null;
    if (next_date_dl is not null) then
      dl_memo = 'Имеется новая доверенность';

    suspend;
  end
end
