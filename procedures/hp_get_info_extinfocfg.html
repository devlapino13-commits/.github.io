CREATE PROCEDURE HP_GET_INFO_EXTINFOCFG(
    C_ID D_ID,
    REP_DATE D_DATE,
    EIC_CODE D_STR_100)
RETURNS (
    IS_ERROR D_SMALLINT,
    REZ_TEXT D_STR_100)
AS
declare variable ssp_bls_id d_id;
declare variable json_full blob sub_type 0 segment size 80;
begin
  -- ССП
  is_error = 0;
  rez_text = '';
  ssp_bls_id = null;
  select eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id and
        eil_from_eic_id = eic_id and
        eic_code = :eic_code and
        eil_status = 2 and
        eil_date between(:rep_date - eic_info_actual_days) and :rep_date
  order by eil_date desc, eil_id
  rows 1
  into ssp_bls_id;

  if (ssp_bls_id is null) then
    rez_text = 'Ошибка получения КБКИ: не получен отчет СП';
  else
  begin
    select blx_blob
    from hp_get_blobex_blob(:ssp_bls_id)
    into json_full;

    if (json_full is null) then
      rez_text = 'Ошибка получения КБКИ: json пустой';

  end
  if (rez_text <> '') then
    is_error = 1;

  suspend;
end
