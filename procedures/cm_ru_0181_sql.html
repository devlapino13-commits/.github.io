CREATE PROCEDURE CM_RU_0181_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id id_type;
declare variable k_disb_date date;
declare variable c_name_fio d_str_100;
begin
  for select kc_from_c_id, k_disb_date, c_name_fio
  from kontract
  inner join knclient on kc_from_k_id = k_id
  inner join product on k_from_pr_id = pr_id
  inner join clients on kc_from_c_id = c_id
  where k_id = :k_id and
        kc_amount_approved > 0 and
        k_disb_kind = 5 and
        k_disb_date >= '15.01.2021' and
        pr_code not containing '.express_online.'
  into c_id, k_disb_date, c_name_fio
  do
  begin
    if (not exists(select *
                   from clientdocs
                   where cds_from_c_id = :c_id and
                         cds_is_delete = 0 and
                         :k_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.9999') and
                         cds_from_pk_id = 338 and
                         (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0))) then
    begin
      res_error_is_critical = 1;
      res_error_text = c_name_fio || ': Не добавлен документ:  «Информация о банковской карте» либо просрочена дата выдачи займа.';
      suspend;
    end
  end
end
