CREATE PROCEDURE HP_CLIENT_REL_STAFF(
    C_ID ID_TYPE)
RETURNS (
    IS_RELATIVE D_BOOL)
AS
declare variable ma_id d_integer;
declare variable staff_c_id id_type;
begin
  -- RFCKG-1513
  -- Входящий параметр c_id - лицо, дающее согласие на кредит.
  -- Если у c_id есть активная заявка, при этом он является родственником действующего сотрудника МБулак,
  -- то is_relative = 1, иначе 0

  is_relative = 0;

  select ma_id
  from miniapp
  inner join miniappclient on ma_id = mac_from_ma_id
  where ma_is_active = 1 and
        ma_date_end = '31.12.9999' and
        ma_date_oper >= (current_date - 30) and
        mac_from_c_id = :c_id and
        mac_amount = 0 and
        mac_is_poruch = -1
  order by ma_number desc rows 1
  into ma_id; -- активная заявка лица, дающего согласие

  -- если есть действующий сотрудник, который есть в списке родственных связей заданного c_id
  if (ma_id is not null and
      exists(select ps_from_c_id
             from hr_pstatus
             inner join hp_client_relation_tmp(:c_id, 4) on rel_other_c_id = ps_from_c_id
             where ps_status = 1 and
                   ps_is_working = 1)) then
    is_relative = 1; -- то они родственники

  suspend;
end
