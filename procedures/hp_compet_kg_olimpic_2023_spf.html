CREATE PROCEDURE HP_COMPET_KG_OLIMPIC_2023_SPF(
    CPP_ID INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable RES_DATE_BEGIN date;
declare variable RES_DATE_END date;
declare variable REP_PERIOD integer;
declare variable STAGE integer;
declare variable RD_ID integer;
declare variable RES_PERIOD_ID integer;
declare variable RES_TEAM_ID integer;
declare variable RES_LO_ID D_ID;
declare variable CPD_POINTS_1 decimal(18,2) = 0;
declare variable CPD_POINTS_2 decimal(18,2) = 0;
declare variable CPD_POINTS_3 decimal(18,2) = 0;
declare variable CPD_POINTS_4 decimal(18,2) = 0;
declare variable CPD_POINTS_5 decimal(18,2) = 0;
declare variable CPD_POINTS_6 decimal(18,2) = 0;
declare variable CPD_POINTS_7 decimal(18,2) = 0;
declare variable CPD_POINTS_8 decimal(18,2) = 0;
declare variable CPD_POINTS_9 decimal(18,2) = 0;
declare variable CPD_POINTS_ALL decimal(18,2) = 0;
declare variable CPR_FINAL_POINTS decimal(18,2) = 0;
declare variable CPR_ID integer;
declare variable CPR_GROUP_NAME varchar(255);
declare variable CPD_COMMENT blob sub_type 1 segment size 160;
declare variable IS_OTPUSK varchar(100);
declare variable SUBJECT_ID integer;
declare variable OVS_DATE_BEGIN date;
declare variable KOL_DAY_WORKED integer;
declare variable SUPER_POKAZATEL decimal(18,2);
declare variable LO_DIFF_WITHOUT_DELINQ decimal(18,2);
declare variable AVG_DIFF_WITHOUT_DELINQ decimal(18,2);
declare variable SP_RANK decimal(18,2);
declare variable LO_DISB_AMOUNT decimal(18,2);
declare variable LO_DISB_NEW_AMOUNT decimal(18,2);
declare variable TEAM_AVG_DISB_AMOUNT_RANK decimal(18,2);
declare variable LO_DISB_1_15 decimal(18,2);
declare variable TEAM_QTY decimal(18,2);
declare variable CPR_ADDITIONAL_POINTS decimal(18,2);
declare variable TEAM_AVG_DISB_AMOUNT decimal(18,2);
declare variable TEAM_PAR30_AVG decimal(18,2);
declare variable TEAM_PAR30_AVG_RANK decimal(18,2);
declare variable TEAM_DISB_NEW_AMOUNT_AVG decimal(18,2);
declare variable TEAM_DISB_NEW_AMOUNT_AVG_RANK decimal(18,2);
declare variable TEAM_BALANCE_AVG decimal(18,2);
declare variable TEAM_BALANCE_AVG_RANK decimal(18,2);
declare variable TEAM_DIFF_CLIENT_AVG decimal(18,2);
declare variable TEAM_DIFF_CLIENT_AVG_RANK decimal(18,2);
declare variable TEAM_DISB_LOAN_AVG decimal(18,2);
declare variable TEAM_DISB_LOAN_AVG_RANK decimal(18,2);
declare variable CPR1_ID D_INTEGER;
declare variable NEW_CPR_FINAL_ORDERNO integer;
declare variable SUPER_POKAZATEL_TXT D_STR_255;
declare variable KOEF_RETURN decimal(15,2);
declare variable AVG_KOEF_RETURN_TMP decimal(15,2);
declare variable AVG_KOEF_RETURN decimal(15,2);
declare variable TEAM_DISB_NEW_AMOUNT decimal(15,2);
declare variable TEAM_DISB_NEW_AMOUNT_RANK integer;
declare variable AD_ACTIVE_BALANCE decimal(15,2);
declare variable LO_DISB_LOAN decimal(15,2);
declare variable CPR_FINAL_ORDERNO integer;
declare variable AVG_KOEF_RETURN_RANK integer;
declare variable TEAM_ID integer;
declare variable CPD_ADD_DATA_1 decimal(15,2);
declare variable CPD_ADD_DATA_2 decimal(15,2);
declare variable SECOND_DECADE D_SMALLINT;
declare variable KOL_UVOL D_INTEGER;
begin
  -->> Удаляем существующие расчеты
  delete from competresult
  where (cpr_from_cpp_id = :cpp_id);

  /*   Расчет ОИ ВИПС 2023
   Stage:      0 - период (неделя)
               1 - Промежуточные итоги
               2 - Итоги
               3 - Финальные результаты  */

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end;

  rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end
  if (res_date_begin = cast('31.01.2023' as date)) then
  begin
    res_date_begin = '08.02.2023';
    res_date_end = '08.02.2023';

    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID команды и ID СВЗ
    for select res_field_1, sum(ad_active_balance)
        from hp_analitica_report('kont_lo_teamname_id', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, :ad_active_balance
    do
    begin
      if (exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1)) then
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, '', -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end
    end
    suspend;
    exit;

  end -- Формируем в памяти данные по компании
  -- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
  execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0, 'olimpic')
      returning_values :res_period_id;

  if (res_period_id is null) then
  begin
    res_text = 'Не удалось расчитать данные за период';
    suspend;
    exit;
  end

  select rd_id
  from refdata
  where rd_code = 'objectprops.staffteam'
  into subject_id;
  if (extract(day from :res_date_end) = 15) then
    second_decade = 1;
  else
    second_decade = 0;
  lo_disb_new_amount = 0;
  if (stage = 0) then
  begin
    -- Группируем данные из памяти по ID команды и ID СВЗ
    for select rd_name cpr_group_name, res_field_1, res_field_2, --
               ad_disb_new_amount, ad_disb_amount
        from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
        inner join refdata rd on rd_id = res_field_1 and
              rd_main_code = 'StaffTeamList' and
              rd_tag = :rd_id and
              rd.rd_is_active = 1
        where ad_active_balance > 0.0 and -- Берем тех у кого имеется портфель на конец периода
              exists(select *
                     from hr_pstatus ps
                     inner join joblist jl on ps_from_jl_id = jl_id
                     inner join jobkind jk on jl_kind = jk_id and
                           jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner', 'JobSystem.LO_Senior')
                     where ps_from_c_id = res_field_2 and
                           :res_date_end between ps_date_begin and ps_date_end and
                           ps_status = 1 and
                           ps_is_working = 1)
        into :cpr_group_name, :res_team_id, :res_lo_id, --
             :lo_disb_new_amount, :lo_disb_1_15
    do
    begin
      cpd_comment = '';
      if (second_decade = 0) then
        cpd_comment = 'Сумма выдач СПФ новым клиентам: <b>' || lo_disb_new_amount || '</b>';

      --    вычисляем дату вступление в команду
      ovs_date_begin = null;
      select ovs_date_begin
      from objectprops op --
      inner join objectvalues ov on --
            opr_id = ovs_from_opr_id and
            :res_date_end between ovs_date_begin and ovs_date_end --
      where opr_object_kind = 0 and
            opr_object_id = :res_lo_id and
            opr_subject_kind = -1 and
            opr_subject_id = :subject_id
      into ovs_date_begin;
      if (ovs_date_begin is not null) then
        kol_day_worked = res_date_end - ovs_date_begin + 1;
      else
        kol_day_worked = 0;

      is_otpusk = '';
      if (exists(select *
                 from hr_voctravel hv
                 inner join hr_orders ho on hrv_from_ho_id = ho_id
                 inner join sa_worktype sw on ho_order_type = sawt_id and
                       sawt_base_type in (4, 5) and
                       -- отпуск, декретный отпуск
                       sawt_id <> 20 and
                       sawt_hours_real >= 0
                 where hrv_from_c_id = :res_lo_id and
                       hrv_status = 1 and
                       :res_date_end between hrv_date_begin and hrv_date_end_fact and
                       iif(hrv_date_end_fact < :res_date_end, :res_date_end, hrv_date_end_fact) - iif(hrv_date_begin < :res_date_begin, :res_date_begin, hrv_date_begin) > 15)) then
      begin
        res_team_id = 0;
        is_otpusk = 'Декректный отпуск';
      end

      cpd_points_1 = 0.0;
      cpd_points_2 = 0.0;
      cpd_points_3 = 0.0;
      cpd_points_4 = 0.0;
      cpd_points_5 = 0.0;
      cpd_points_6 = 0.0;
      cpd_points_7 = 0.0;
      cpd_points_8 = 0.0;
      cpd_points_9 = 0.0;
      cpd_points_all = 0.0;

      cpr_group_name = coalesce(cpr_group_name, '');

      if (res_team_id > 0 and
          not exists(select *
                     from competresult
                     where cpr_from_cpp_id = :cpp_id and
                           cpr_object_kind = -1 and
                           cpr_object_id = :res_team_id)) then -- если такой строки нет, создаем новую
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment, cpr_calced_points, cpr_additional_points,
                                  cpr_final_points, cpr_order_in_group, cpr_final_orderno)
        values (:cpp_id, :cpr_group_name, -1, :res_team_id, 0, :un_id, '', 0, 0, 0, 0, 0)
        returning cpr_id
        into :cpr_id;
      end
      if (second_decade = 1) then
      begin
        lo_disb_1_15 = coalesce(lo_disb_1_15, 0);
        cpd_comment = cpd_comment || '<br>Сумма выдач с 1-15 СПФ: <b>' || lo_disb_1_15 || '</b>';
      end
      else
        lo_disb_1_15 = 0;
      if (kol_day_worked >= 5 and
          res_team_id > 0) then
      begin
        insert into competdetails (cpd_id, cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, --
                                   cpd_points_1, --
                                   cpd_points_2, --
                                   cpd_points_3, --
                                   cpd_points_4, --
                                   cpd_points_5, --
                                   cpd_points_6, --
                                   cpd_points_7, --
                                   cpd_points_8, --
                                   cpd_points_9, --
                                   cpd_comment, --
                                   cpd_staff_kolich, -- кол-во проработанных дней в данной команде
                                   cpd_add_data_1, --
                                   cpd_add_data_2, --
                                   cpd_add_data_3, --
                                   cpd_add_data_4, --
                                   cpd_add_data_5, --lo_disb_new_amount
                                   cpd_add_data_6, --lo_disb_1_15
                                   cpd_add_data_7, --
                                   cpd_add_data_8, --
                                   cpd_add_data_9 --
                                   )
        values (null, :cpr_id, 0, :res_lo_id, --
                :cpd_points_all, --
                :cpd_points_1, --
                :cpd_points_2, --
                :cpd_points_3, --
                :cpd_points_4, --
                :cpd_points_5, --
                :cpd_points_6, --
                :cpd_points_7, --
                :cpd_points_8, --
                :cpd_points_9, --
                :cpd_comment, :kol_day_worked, --
                0, 0, 0, 0, --
                :lo_disb_new_amount, :lo_disb_1_15, 0, 0, 0);
      end
      else
        cpd_comment = '';
    end
    if (:second_decade = 1) then
    begin
      select count(distinct cpr.cpr_id)
      from competresult cpr
      where cpr.cpr_from_cpp_id = :cpp_id
      into team_qty;

      for select cpr.cpr_id, avg(cpd_add_data_6) team_avg_disb_amount,
                 rank() over(order by avg(cpd_add_data_6) desc) team_avg_disb_amount_rank
          from competresult cpr
          inner join competdetails cd on cpr.cpr_id = cd.cpd_from_cpr_id and
                cd.cpd_staff_kolich >= 1
          where cpr.cpr_from_cpp_id = :cpp_id
          group by 1
          into cpr_id, team_avg_disb_amount, team_avg_disb_amount_rank
      do
      begin
        cpr_additional_points = 0;
        cpd_comment = '';
        if (team_avg_disb_amount_rank <= 15) then
          cpr_additional_points = 5;
        if (:team_qty - team_avg_disb_amount_rank < 10) then
          cpr_additional_points = -2;
        cpd_comment = cpd_comment || '<br>Ср. сумма выдач команды: <b>' || coalesce(team_avg_disb_amount, 0) || ';</b><br>    Ранг:' || team_avg_disb_amount_rank;
        cpd_comment = cpd_comment || '<br>    Доп.балл за выдачи: <b>' || cpr_additional_points || ';';

        update competresult
        set cpr_additional_points = coalesce(cpr_additional_points, 0) + :cpr_additional_points
        where cpr_from_cpp_id = :cpp_id and
              cpr_id = :cpr_id;

        update competdetails cd
        set cpd_comment = cpd_comment || :cpd_comment
        where cpd_from_cpr_id = :cpr_id;
      end
    end
    else
    begin
      for select cpr.cpr_id, avg(cpd_add_data_5) team_disb_new_amount,
                 rank() over(order by avg(cpd_add_data_5) desc) team_disb_new_amount_rank
          from competresult cpr
          inner join competdetails cpd on cpd.cpd_from_cpr_id = cpr.cpr_id and
                cpd.cpd_staff_kolich >= 1
          where cpr.cpr_from_cpp_id = :cpp_id
          group by 1
          into cpr_id, team_disb_new_amount, team_disb_new_amount_rank
      do
      begin
        cpr_additional_points = 0.0;
        cpd_comment = '';
        cpd_comment = cpd_comment || '<br>Средняя сумма выдач новым клиентам: <b>' || team_disb_new_amount || ';</b><br>    Ранг:' || team_disb_new_amount_rank;
        if (team_disb_new_amount_rank <= 5) then
          cpr_additional_points = 6 - :team_disb_new_amount_rank;
        cpd_comment = cpd_comment || '<br>    Доп.балл за выдачи новым клиентам: <b>' || cpr_additional_points || ';';

        update competresult
        set cpr_additional_points = coalesce(cpr_additional_points, 0) + :cpr_additional_points
        where cpr_from_cpp_id = :cpp_id and
              cpr_id = :cpr_id;

        update competdetails cd
        set cpd_comment = cpd_comment || :cpd_comment
        where cpd_from_cpr_id = :cpr_id;
      end
    end
  end

  if (stage = 1) then
  begin
    -- Группируем данные из памяти по ID команды и ID СВЗ
    for select rd_name cpr_group_name, --
               res_field_1, --
               res_field_2, --
               (ad_active_balance - ad_delinq_balance_end1) - (ad_begin_active_balance - ad_delinq_balance_begin1) as lo_diff_without_delinq, -- 2.    Рост портфеля без просрочек
               ad_disb_amount as lo_disb_amount, --
               ad_disb_loan as lo_disb_loan, --
               ad_return_os_koef as koef_return
        from hp_analitica_report('kont_lo_teamname_id, lo_id, own_portfolio, reg_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
        inner join refdata rd on rd_id = res_field_1 and
              rd_main_code = 'StaffTeamList' and
              rd_tag = :rd_id
        where res_field_3 = 'Собственный' and
              res_field_2 > 1000000000010 and
              res_field_1 > 0 and
              ad_active_balance > 0.0 and -- Берем тех у кого имеется портфель на конец периода
              exists(select *
                     from hr_pstatus ps
                     inner join joblist jl on ps_from_jl_id = jl_id
                     inner join jobkind jk on jl_kind = jk_id and
                           jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner', 'JobSystem.LO_Senior')
                     where ps_from_c_id = res_field_2 and
                           :res_date_end between ps_date_begin and ps_date_end and
                           ps_status = 1 and
                           ps_is_working = 1 and
                           ps.ps_from_reg_id = res_field_4)
        into :cpr_group_name, --
             :res_team_id, --
             :res_lo_id, --
             :lo_diff_without_delinq, --
             lo_disb_amount, --
             lo_disb_loan, --
             koef_return
    do
    begin

      lo_disb_new_amount = 0;
      select sum(kc.kc_amount_approved)
      from kontstate ks
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
            kc.kc_circle = 1 and
            kc.kc_amount_approved > 0
      where ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_date_begin between(:res_date_begin + 15) and :res_date_end and
            ks.ks_kc_id = :res_lo_id
      into lo_disb_new_amount;
      --    вычисляем дату вступление в команду
      ovs_date_begin = null;
      select ovs_date_begin
      from objectprops op --
      inner join objectvalues ov on --
            opr_id = ovs_from_opr_id and
            :res_date_end between ovs_date_begin and ovs_date_end --
      where opr_object_kind = 0 and
            opr_object_id = :res_lo_id and
            opr_subject_kind = -1 and
            opr_subject_id = :subject_id
      into ovs_date_begin;
      if (ovs_date_begin is not null) then
        kol_day_worked = res_date_end - ovs_date_begin + 1;
      else
        kol_day_worked = 0;

      is_otpusk = '';
      if (exists(select *
                 from hr_voctravel hv
                 inner join hr_orders ho on hrv_from_ho_id = ho_id
                 inner join sa_worktype sw on ho_order_type = sawt_id and
                       sawt_base_type in (4, 5) and
                       -- отпуск, декретный отпуск
                       sawt_id <> 20 and
                       sawt_hours_real >= 0
                 where hrv_from_c_id = :res_lo_id and
                       hrv_status = 1 and
                       :res_date_end between hrv_date_begin and hrv_date_end_fact and
                       iif(hrv_date_end_fact < :res_date_end, :res_date_end, hrv_date_end_fact) - iif(hrv_date_begin < :res_date_begin, :res_date_begin, hrv_date_begin) > 15)) then
      begin
        res_team_id = 0;
        is_otpusk = 'Декректный отпуск';
      end

      cpd_points_1 = 0.0;
      cpd_points_2 = 0.0;
      cpd_points_3 = 0.0;
      cpd_points_4 = 0.0;
      cpd_points_5 = 0.0;
      cpd_points_6 = 0.0;
      cpd_points_7 = 0.0;
      cpd_points_8 = 0.0;
      cpd_points_9 = 0.0;
      cpd_points_all = 0.0;

      cpr_group_name = coalesce(cpr_group_name, '');

      if (res_team_id > 0 and
          not exists(select *
                     from competresult
                     where cpr_from_cpp_id = :cpp_id and
                           cpr_object_kind = -1 and
                           cpr_object_id = :res_team_id)) then -- если такой строки нет, создаем новую
      begin
        cpr_final_points = 0;
        select sum(cpr_final_points)
        from competresult cr
        inner join competperiod cp on cpp_id = cpr_from_cpp_id and
              cpp_stage = 0 and
              cpp_date_end between :res_date_begin and :res_date_end
        where cr.cpr_object_kind = -1 and
              cpr_object_id = :res_team_id
        into cpr_final_points;
        cpr_final_points = coalesce(cpr_final_points, 0);

        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment, cpr_calced_points, cpr_additional_points,
                                  cpr_final_points, cpr_order_in_group, cpr_final_orderno)
        values (:cpp_id, :cpr_group_name, -1, :res_team_id, 0, :un_id, '', :cpr_final_points, 0, 0, 0, 0)
        returning cpr_id
        into :cpr_id;
      end
      cpd_comment = '';
      if (is_otpusk <> '') then
        cpd_comment = cpd_comment || 'СПФ не участвует в периоде. ' || is_otpusk || '<br>';

      cpd_comment = cpd_comment || 'Дата вступления в команду: <b>' || f_datetostr(ovs_date_begin) || '</b>';
      cpd_comment = cpd_comment || '<br><br><b>Показатели СПФ за период:</b>';
      --  cpd_comment = cpd_comment || '<br>1.Баланс ПАР1 СПФ: <b>' || round(lo_par1, 2) || '</b>';
      --  cpd_comment = cpd_comment || '<br>    1.1 Изменения баланс ПАР1 СПФ: <b>' || round(lo_par1_diff, 2) || '</b>';
      cpd_comment = cpd_comment || '<br>2.Рост портфеля без просрочек СПФ: <b>' || lo_diff_without_delinq || '</b>';

      select super_pokazatel, super_pokazatel_txt
      from hp_compet_kg_2021_superp(:res_lo_id, null, :rep_period, :res_date_begin, :res_date_end)
      into :super_pokazatel, super_pokazatel_txt;

      if (kol_day_worked >= 15 and
          res_team_id > 0) then
      begin
        insert into competdetails (cpd_id, cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, --
                                   cpd_points_1, --
                                   cpd_points_2, --
                                   cpd_points_3, --
                                   cpd_points_4, --
                                   cpd_points_5, --
                                   cpd_points_6, --
                                   cpd_points_7, --
                                   cpd_points_8, --
                                   cpd_points_9, --
                                   cpd_comment, --
                                   cpd_staff_kolich, -- кол-во проработанных дней в данной команде
                                   cpd_add_data_1, --lo_diff_without_delinq
                                   cpd_add_data_2, --koef_return
                                   cpd_add_data_3, --Кол-во уволенных ВИПС
                                   cpd_add_data_4, --super_pokazatel
                                   cpd_add_data_5, --lo_disb_new_amount
                                   cpd_add_data_6, -- lo_disb_amount
                                   cpd_add_data_7, --
                                   cpd_add_data_8, --lo_disb_1_15
                                   cpd_add_data_9 --
                                   )
        values (null, :cpr_id, 0, :res_lo_id, :cpd_points_all, --
                :cpd_points_1, --
                :cpd_points_2, --
                :cpd_points_3, --
                :cpd_points_4, --
                :cpd_points_5, --
                :cpd_points_6, --
                :cpd_points_7, --
                :cpd_points_8, --
                :cpd_points_9, --
                :cpd_comment, --
                :kol_day_worked, --
                :lo_diff_without_delinq, --
                :koef_return, --
                0, --
                :super_pokazatel, --
                :lo_disb_new_amount, --
                :lo_disb_amount, --
                0, 0, 0);
      end
    end
    team_qty = 0;
    select count(distinct cpr_id)
    from competresult
    where cpr_from_cpp_id = :cpp_id and
          cpr_object_id > 0
    into team_qty;

    for select cpr_id, avg(cpd_add_data_1) avg_diff_without_delinq, round(avg(cpd_add_data_2), 2) avg_koef_return
        from competresult cr
        inner join competdetails cp on cpd_from_cpr_id = cpr_id and
              cpd_staff_kolich >= 15 and
              cpr_object_id <> 0
        where cpr_from_cpp_id = :cpp_id and
              cpr_object_id > 0 and
              cpr_object_kind = -1
        group by 1
        into cpr_id, avg_diff_without_delinq, avg_koef_return
    do
    begin
      cpd_comment = '';
      --    1.    Рост портфеля без просрочек (средний показатель)
      cpd_points_1 = case
                       when avg_diff_without_delinq > 200000.00 then 15
                       when avg_diff_without_delinq > 150000.00 then 10
                       when avg_diff_without_delinq > 70000.00 then 4
                       when avg_diff_without_delinq > 20000.00 then 4
                       when avg_diff_without_delinq >= -20000.00 then 0
                       when avg_diff_without_delinq >= -50000.0 then -1
                       when avg_diff_without_delinq < -50000.0 then -7
                       else 0
                     end;

      cpd_points_2 = case
                       when avg_koef_return >= 99.5 then 15
                       when avg_koef_return >= 99 then 10
                       when avg_koef_return >= 98.5 then 5
                       when avg_koef_return >= 98 then 1
                       when avg_koef_return >= 97.5 then -2
                       when avg_koef_return >= 97 then -5
                       when avg_koef_return < 97 then -7
                       else 0
                     end;

      cpd_comment = cpd_comment || '<br><br><b>Показатели команды:</b><br>';
      cpd_comment = cpd_comment || '<br>Рост портфеля без просрочек (средний показатель): <b>' || :avg_diff_without_delinq || '.</b><br>    Балл: <b>' || :cpd_points_1 || '</b>.';
      cpd_comment = cpd_comment || '<br>Возвратность(средний показатель) :<b>' || :avg_koef_return || '.</b><br>    Балл за возвратность: <b>' || :cpd_points_2 || '</b>.';

      update competdetails
      set cpd_points_1 = :cpd_points_1,
          cpd_points_2 = :cpd_points_2,
          cpd_comment = cpd_comment || :cpd_comment
      where cpd_from_cpr_id = :cpr_id and
            cpd_staff_kolich >= 5;
    end

     kol_uvol = 0;
    for select count(res_is_working), res_team_id
        from aarep_staffteam_list(1000000, :res_date_end)
        where res_is_working = 0
        group by res_team_id, res_lo_id
        into kol_uvol, team_id
    do
    begin
      cpr_id = 0;
      select cpr.cpr_id
      from competresult cpr
      where cpr.cpr_from_cpp_id = :cpp_id and
            cpr.cpr_object_kind = -1 and
            cpr.cpr_object_id = :team_id
      into cpr_id;
      if (coalesce(kol_uvol, 0) > 0) then
        update competdetails cpd
        set cpd.cpd_points_4 = (-7 * :kol_uvol)
        where cpd.cpd_object_kind = 0 and
              cpd.cpd_from_cpr_id = :cpr_id;
    end
    --Супер-показатель
    for with t_rank
        as (select cpr_id, avg(cpd_add_data_4) sp_avg
            from competresult cr
            inner join competdetails cp on cpd_from_cpr_id = cpr_id and
                  cpd_staff_kolich <> 2
            where cpr_from_cpp_id = :cpp_id and
                  cpr_object_id <> 0
            group by 1
            order by 2),
        t_result
        as (
        --
        select distinct cpr_id, sp_avg, --
                        rank() over(order by sp_avg) sp_rank --Когда вырастающий
        from t_rank
        where :rep_period in (202301)
        union
        select distinct cpr_id, sp_avg, --
                        rank() over(order by sp_avg desc) sp_rank --Когда нужно в обратном порядке начислять баллы!
        from t_rank
        where :rep_period in (202302)
              --
        )
        select cpr_id, sp_avg, sp_rank
        from t_result
        order by 3
        into cpr_id, super_pokazatel, sp_rank
    do
    begin
      cpd_points_3 = 51.0 - sp_rank;
      cpd_comment = '<br>Супер-показатель: ' || round(:super_pokazatel, 2) || '.<br>    Ранг: ' || cast(:sp_rank as smallint) || '.<br>    Балл: ' || cpd_points_3;

      update competdetails
      set cpd_points_3 = :cpd_points_3,
          cpd_comment = cpd_comment || :cpd_comment
      where cpd_from_cpr_id = :cpr_id;
    end --Superpokazatel konec
  end
  if (stage = 2) then -- если уровень этапа
  begin
    for with t_team_info
        as (select res_field_1 as res_team_id,
                   sum(ad_delinq_balance_end2) / sum(ad_active_balance) / count(res_field_2) team_par30_avg,
                   avg(ad_active_balance) team_balance_avg, --
                   avg(ad_disb_amount) team_disb_new_amount_avg
            from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
            inner join refdata rd on rd_main_code = 'StaffTeamList' and
                  rd_id = res_field_1 and
                  rd_tag = :rd_id
            where res_field_1 > 0 and
                  res_field_2 > 1000000000010 and
                  ad_active_balance > 0.0
            group by 1)
        select res_team_id, --
               team_par30_avg, rank() over(order by team_par30_avg) team_par30_avg_rank, --
               team_balance_avg, rank() over(order by team_balance_avg) team_balance_avg_rank, --
               team_disb_new_amount_avg,
               rank() over(order by team_disb_new_amount_avg desc) team_disb_new_amount_avg_rank
        from t_team_info
        into res_team_id, --
             team_par30_avg, team_par30_avg_rank, --
             team_balance_avg, team_balance_avg_rank, --
             team_disb_new_amount_avg, team_disb_new_amount_avg_rank --
    do
    begin
      cpd_comment = '';
      cpr_additional_points = 0;

      /*    Команда с наименьшим ПАР30 в процентах    7 -->>  */
      if (team_par30_avg_rank = 1) then
      begin
        cpr_additional_points = cpr_additional_points + 7;
        cpd_comment = 'Команда с наименьшим ПАР30 в процентах ' || team_par30_avg || '. Доп.балл 7; ';
      end

      /*    Команда с наибольшим средним портфелем на 1 ВИПС    7 -->>  */
      if (team_balance_avg_rank = 1) then
      begin
        cpr_additional_points = cpr_additional_points + 7;
        cpd_comment = cpd_comment || 'Команда с наибольшим средним портфелем на 1 ВИПС ' || team_balance_avg || '. Доп.балл 7; ';
      end

      /*    Команда с наибольшими суммарными выдачами на 1 ВИПС за этап    7 -->>  */
      if (team_disb_new_amount_avg_rank = 1) then
      begin
        cpr_additional_points = cpr_additional_points + 7;
        cpd_comment = cpd_comment || 'Команда с наибольшими суммарными выдачами на 1 ВИПС за этап ' || team_disb_new_amount_avg || '. Доп.балл 7; ';
      end

      cpr_group_name = '';
      cpr_final_points = 0;
      select cpr_group_name, sum(cpr_final_points)
      from competresult cr
      inner join competperiod cp on cpp_id = cpr_from_cpp_id and
            cpp_stage = 1 and
            cpp_date_end between :res_date_begin and :res_date_end
      where cr.cpr_object_kind = -1 and
            cr.cpr_object_id = :res_team_id
      group by 1
      into cpr_group_name, cpr_final_points;
      cpr_final_points = coalesce(cpr_final_points, 0);

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id, cpr_additional_points, cpr_comment, cpr_final_points)
      values (:cpp_id, :cpr_group_name, -1, :res_team_id, :cpr_final_points, 0, :un_id, :cpr_additional_points,
              left(:cpd_comment, 255), :cpr_final_points + :cpr_additional_points)
      returning cpr_id
      into :cpr_id;
    end
  end

  if (stage = 3) then -- если уровень этапа
  begin
    for with t_team_info
        as (select res_field_1 as res_team_id,
                   sum(ad_delinq_balance_end2) / sum(ad_active_balance) / count(res_field_2) team_par30_avg,
                   avg(ad_diff_client) team_diff_client_avg, --
                   avg(ad_disb_loan) team_disb_loan_avg
            from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
            inner join refdata rd on rd_main_code = 'StaffTeamList' and
                  rd_id = res_field_1 and
                  rd_tag = :rd_id
            where res_field_1 > 0 and
                  res_field_2 > 1000000000010 and
                  ad_active_balance > 0.0
            group by 1)
        select res_team_id, --
               team_par30_avg, rank() over(order by team_par30_avg) team_par30_avg_rank, --
               team_diff_client_avg, rank() over(order by team_diff_client_avg) team_diff_client_avg_rank, --
               team_disb_loan_avg, rank() over(order by team_disb_loan_avg desc) team_disb_loan_avg_rank
        from t_team_info
        into res_team_id, --
             team_par30_avg, team_par30_avg_rank, --
             team_diff_client_avg, team_diff_client_avg_rank, --
             team_disb_loan_avg, team_disb_loan_avg_rank --
    do
    begin
      cpd_comment = '';
      cpr_additional_points = 0;
      cpr_group_name = '';
      cpr_final_points = 0;
      select cpr_group_name, sum(cpr_final_points)
      from competresult cr
      inner join competperiod cp on cpp_id = cpr_from_cpp_id and
            cpp_stage = 2 and
            cpp_date_end between :res_date_begin and :res_date_end
      where cr.cpr_object_kind = -1 and
            cpr_object_id = :res_team_id
      group by 1
      into cpr_group_name, cpr_final_points;
      cpr_final_points = coalesce(cpr_final_points, 0);

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id, cpr_additional_points, cpr_comment, cpr_final_points)
      values (:cpp_id, :cpr_group_name, -1, :res_team_id, :cpr_final_points, 0, :un_id, :cpr_additional_points,
              left(:cpd_comment, 255), :cpr_final_points + :cpr_additional_points)
      returning cpr_id
      into :cpr_id;
    end
  end

  -- Пересчитываем итоги по командам
  for select distinct cpr_id, --
                      cpd_points_1 + cpd_points_2 + cpd_points_3 + cpd_points_4 + cpd_points_5 + --
                      cpd_points_6 + cpd_points_7 + cpd_points_8 + cpd_points_9 as sum_points
      from competresult cr
      inner join competdetails cd on cr.cpr_id = cd.cpd_from_cpr_id
      where cpr_from_cpp_id = :cpp_id
      into :cpr_id, :cpd_points_all
  do
  begin
    update competdetails
    set cpd_points_all = :cpd_points_all
    where cpd_from_cpr_id = :cpr_id;

    update competresult
    set cpr_calced_points = :cpd_points_all,
        cpr_final_points = :cpd_points_all + cpr_additional_points
    where (cpr_from_cpp_id = :cpp_id and
          cpr_id = :cpr_id);
  end

  execute procedure hp_competition_recalc_order(:cpp_id);

  /*    5.4.    При равенстве всех итоговых баллов победитель определяется с учетом показателей в абсолютных значениях:
    первый критерий по количеству выдач на одного члена команды за отчетный период;
  в случае равенства по первому критерию, то расчет осуществляется
    по росту количества активных клиентов на одного члена команды. */
  team_disb_loan_avg = 0;
  team_diff_client_avg = 0;
  for with t_cpr
      as (select cr.cpr_id, --
                 cr.cpr_final_orderno, --
                 avg(ad_disb_loan) team_disb_loan_avg, --Ср.выдачи на одного СВЗ
                 avg(ad_diff_client) team_diff_client_avg --Ср.росту количества активных клиентов
          from competresult cr
          inner join hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart') on res_field_1 = cr.cpr_object_id and
                res_field_1 > 0 and
                res_field_2 > 1000000000010 and
                ad_active_balance > 0.0
          inner join refdata rd on rd_main_code = 'StaffTeamList' and
                rd_id = res_field_1 and
                rd_tag = :rd_id
          where cr.cpr_from_cpp_id = :cpp_id and
                cpr_object_kind = -1 and
                cpr_object_id > 0
          group by 1, 2)

      select cpr_id, cpr_final_orderno, team_disb_loan_avg, team_diff_client_avg,
             rank() over(order by cpr_final_orderno, team_disb_loan_avg desc, team_diff_client_avg desc) new_cpr_final_orderno
      from t_cpr
      into cpr_id, cpr_final_orderno, team_disb_loan_avg, team_diff_client_avg, new_cpr_final_orderno
  do
  begin
    cpd_comment = 'Ср.выдачи: ' || round(:team_disb_loan_avg) || '; Ср.прирост актив.кл.: ' || round(:team_diff_client_avg) || '; ';

    update competresult
    set cpr_final_orderno = :new_cpr_final_orderno,
        cpr_comment = cpr_comment || :cpd_comment
    where cpr_from_cpp_id = :cpp_id and
          cpr_id = :cpr_id;
  end
  if (stage = 1) then --  Команде, победившей по итогам отчетного месяца
  begin
    for select cpr.cpr_id
        from competresult cpr
        where cpr.cpr_from_cpp_id = :cpp_id and
              cpr.cpr_final_orderno = 1
        into :cpr1_id
    do
    begin
      cpr_additional_points = 1;
      cpd_comment = '<br>Балл за 1 место: ' || :cpr_additional_points;

      update competdetails cpd
      set cpd.cpd_comment = cpd_comment || :cpd_comment
      where cpd.cpd_from_cpr_id = :cpr1_id;

      update competresult cpr
      set cpr.cpr_additional_points = cpr.cpr_additional_points + :cpr_additional_points
      where cpr.cpr_id = :cpr1_id;
    end
  end
  suspend;
  when any do
  begin
    exception e_message 'internal error';
  end
