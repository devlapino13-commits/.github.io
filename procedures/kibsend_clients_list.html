CREATE PROCEDURE KIBSEND_CLIENTS_LIST(
    KP_ID INTEGER,
    K_ID DECIMAL(18,0),
    GARANT INTEGER)
RETURNS (
    UNIQ_CRED_NUM DECIMAL(18,0),
    FNAME VARCHAR(255),
    LNAME VARCHAR(255),
    PATRONYMIC VARCHAR(255),
    SEX_CODE VARCHAR(255),
    BIRTHD DATE,
    BIRTHPLACE VARCHAR(255),
    IDCARDT_ID INTEGER,
    PASSN VARCHAR(255),
    PASSPLACE VARCHAR(255),
    ADDR VARCHAR(255),
    ADDRFACT VARCHAR(255),
    PATENT VARCHAR(255),
    SOCC VARCHAR(255),
    STATCOM VARCHAR(255),
    C_ID DECIMAL(18,0),
    UNIQ_K_ID DECIMAL(18,0),
    GUARANTSTATUS_CODE VARCHAR(255),
    IS_GARANT INTEGER,
    POCKET_ID INTEGER,
    COUNTRYID VARCHAR(255),
    OWNAMOUNT DECIMAL(18,4),
    OWN_CLOSE_DATE DATE)
AS
declare variable c_name_fio varchar(100);
declare variable c_sex integer;
declare variable kk integer;
declare variable is_urgent integer;
declare variable original_id decimal(18,0);
declare variable kc_closed_ol_id decimal(18,0);
begin
  /* CHANGED BY IGUSEV 04.06.2012
     Chenges:
     Были добавлены входящие параметры:
          KP_ID - ID кредитного пакета
          K_ID - ID контракта
          GARANT  0 - заемщики; 1 - поручители
     Это было сделано для ускорения выборки данных

     **************
     CHANGED BY IGUSEV  14.06.2012
     Chenges:
     Был добавлен исходящий параметр OwnAmount - собственная сумма кредита

     **************
     CHANGED BY IGUSEV  30.07.2012
     Chenges: В случае, если клиент взял кредит под псевдонимом Client подставляю его реальные данные
     MEMO: связб по полю c_original_id = c_id
  */
  is_urgent = 0;
  countryid = 'kgz';
  for select kkr_uniq_cred_num, c_name_fio, c_sex, c_birth_date, 'Кыргызстан', c_doc_type, c_pass_no,
             c_pass_who, '', c_inn, '', c_id, kkr_from_k_id, kkc_status, kkr_from_kp_id,
             coalesce(kc_amount_approved, 0), /* сумма кредита выданная клиенту*/
             cl.c_original_id, /* реальный id клинта, если используется псевдоним Clients */
             kncl.kc_closed_ol_id
      from clients cl
      inner join kibclients on c_id = kkc_from_c_id
      inner join kibkredit on kkr_from_k_id = kkc_from_k_id and
            kkr_from_kp_id = kkc_from_kp_id
      inner join kibpocket on kkr_from_kp_id = kp_id
      left join knclient kncl on kncl.kc_from_c_id = kkc_from_c_id and
            kncl.kc_from_k_id = kkr_from_k_id
      where kkr_transid = 0 and
            ((kp_check_status = 1 and
            :is_urgent = 1) or
            (:is_urgent = 0)) and
            kp_id = :kp_id and
            kkr_from_k_id = :k_id and
            kkc_status = :garant and
            (((coalesce(kncl.kc_amount_approved, 0) > 0) and
            (:garant = 0)) or
            ((coalesce(kncl.kc_amount_approved, 0) = 0) and
            (:garant = 1)))
      order by kkr_from_kp_id, kkr_from_k_id, c_id
      into :uniq_cred_num, :c_name_fio, :c_sex, :birthd, :birthplace, :idcardt_id, :passn, :passplace, :patent, :socc,
           :statcom, :c_id, :uniq_k_id, :is_garant, :pocket_id, :ownamount, :original_id, :kc_closed_ol_id
  do
  begin
    addr = '';
    addrfact = '';
    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :c_id and
          adc_from_adl_id = adl_id and
          adc_address_type = 1
    into :addr;

    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :c_id and
          adc_from_adl_id = adl_id and
          adc_address_type = 2
    into :addrfact;
    if ((addrfact is null) or (addrfact = '')) then
      addrfact = addr;

    own_close_date = null;
    if (kc_closed_ol_id is not null) then
    begin
      select ol.ol_date
      from operlist ol
      where ol.ol_id = :kc_closed_ol_id and
            ol.is_delete = 0
      into :own_close_date;
    end

    if ((upper(c_name_fio) like '%CLIENT%') and
        (original_id is not null)) then
    begin
      select c_name_fio, c_sex, c_birth_date, c_doc_type, c_pass_no, c_pass_who, c_inn, c_id
      from clients cl
      where cl.c_id = :original_id
      into :c_name_fio, :c_sex, :birthd, :idcardt_id, :passn, :passplace, :socc, :c_id;
    end
    /* ------------ Т.к было выявлено что не всегда C_Full_Address заполнен, вставил этот код. Взят из тригера CADDRESS After Update*/
    addr = '';
    addrfact = '';
    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :c_id and
          adc_from_adl_id = adl_id and
          adc_address_type = 1
    into :addr;

    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :c_id and
          adc_from_adl_id = adl_id and
          adc_address_type = 2
    into :addrfact;
    if ((addrfact is null) or (addrfact = '')) then
      addrfact = addr;

    /*---------------------------------------------------------------------------------------*/

    if (is_garant = 0) then
      guarantstatus_code = '';
    if (is_garant = 1) then
      guarantstatus_code = '004';
    fname = '';
    lname = '';
    patronymic = '';
    kk = position(' ', :c_name_fio) - 1;

    c_name_fio = replace(c_name_fio, 'ң', 'н');
    c_name_fio = replace(c_name_fio, '�©', 'о');
    c_name_fio = replace(c_name_fio, 'ү', 'у');

    if (kk > 0) then
    begin
      lname = substring(:c_name_fio from 1 for kk);
      c_name_fio = substring(:c_name_fio from kk + 2 for 255);
      kk = position(' ', c_name_fio) - 1;
      if (kk > 0) then
      begin
        fname = substring(:c_name_fio from 1 for kk);
        c_name_fio = substring(:c_name_fio from kk + 2 for 255);
        patronymic = :c_name_fio;
      end
      else
        fname = :c_name_fio;
    end
    if (c_sex is null) then
      sex_code = 'Н';
    if (c_sex = 0) then
      sex_code = 'Ж';
    if (c_sex = 1) then
      sex_code = 'М';
    fname = trim(fname);
    lname = trim(lname);
    patronymic = trim(patronymic);
    suspend;
  end
end
