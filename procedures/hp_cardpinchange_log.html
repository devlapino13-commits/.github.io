CREATE PROCEDURE HP_CARDPINCHANGE_LOG(
    REGION_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    CLIENT_FIO D_STR_100,
    CARD_ID D_INTEGER,
    CPH_DATE D_DATE,
    CARD_REG_NAME D_STR_CODE,
    REG_NAME D_STR_CODE,
    STATUS D_STR_CODE,
    UN_FIO D_STR_100,
    ERR_TEXT D_STR_LONG,
    CPH_ID D_INTEGER)
AS
begin
  for
    select
      c_name_fio,
      cph_card_id,
      cph_date,
      r2.reg_name,
      r1.reg_name,
      s.name,
      un_fio,
      cph_err_text,
      cph_id
    from cardpinchange
      inner join clients
        on c_id = cph_from_c_id
      inner join region as r1
        on r1.reg_unicode = cph_region_id
      inner join region as r2
        on r2.reg_unicode = cph_card_region_id
      inner join username un
        on un_id = cph_from_un_id
      inner join hp_enum_list('cardpinchange_status') as s
        on s.id = cph_status
    where cph_date between :date_begin and :date_end
      and exists(
        select *
        from region_list(:region_id, 1, 3, 1)
        where cph_region_id = id_list)
  into
    :client_fio,
    :card_id,
    :cph_date,
    :card_reg_name,
    :reg_name,
    :status,
    :un_fio,
    :err_text,
    :cph_id
  do
  begin
    suspend;
  end
end
