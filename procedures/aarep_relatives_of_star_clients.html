CREATE PROCEDURE AAREP_RELATIVES_OF_STAR_CLIENTS(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    CC_CHECKED_DATE DATE,
    OFFICE_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(50),
    CC_FIO VARCHAR(100),
    CC_TEL_NO VARCHAR(255),
    CL_REL_STATUS VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    CC_ADDRESS VARCHAR(255),
    CC_LANG VARCHAR(100),
    TIMEZONE INTEGER,
    K_DISB_DATE DATE)
AS
begin
  for select reg_timezone, reg_name, k_kontract_no, k_disb_date, kc_from_c_id
      from kontstate, kontract, knclient, region
      where ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_from_k_id = k_id and
            k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            k_last_reg_id = reg_unicode
      order by reg_timezone, reg_name, k_kontract_no, k_disb_date, kc_from_c_id
      into timezone, office_name, k_kontract_no, k_disb_date, c_id
  do
  begin
--все контакты которые были автоматически проверифицированы за период
    for select cont.cc_name, cont.cc_cell_no, cont.cc_status, coalesce(adl_address_str, cc_address), rd_name,
               cont.cc_checked_date
        from contacts cont
        inner join refdata on (rd_id = cc_language and
              rd_main_code = 'Language')
        inner join addrlist on adl_id = cont.cc_from_adl_id
        where cc_from_c_id = :c_id and
              cont.cc_call_user = 'igusev' and
              :k_disb_date between cc_checked_date and cc_checked_date + 7 and
              exists(select *
                     from hp_is_close_relative(cc_status)
                     where result = 1)
        order by cc_from_c_id, cc_checked_date
        into :cc_fio, cc_tel_no, cl_rel_status, cc_address, cc_lang, cc_checked_date
    do
    begin
      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name_fio;

      suspend;
    end
  end
end
