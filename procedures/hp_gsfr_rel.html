CREATE PROCEDURE HP_GSFR_REL
RETURNS (
    GRAL_ID D_INTEGER,
    CLIENT_FIO D_STR_100,
    REL_FIO D_STR_100,
    GRAL_DATE D_DATE,
    UN_NAME D_STR_CODE)
AS
declare variable gral_from_c_id d_id;
declare variable gral_other_c_id d_id;
declare variable gral_from_un_id d_id;
begin
  -- возвращает список аннулированных родственных связей клиентов ГСФР (RFCKG-2614)

  for select gral_id,
             gral_from_c_id,
             gral_other_c_id,
             gral_date,
             gral_from_un_id
      from gsfr_rel_annul_list
      where gral_is_delete = 0
      order by gral_datetime desc
      into gral_id,
           gral_from_c_id,
           gral_other_c_id,
           gral_date,
           gral_from_un_id
  do
  begin
    client_fio = null;
    rel_fio = null;
    un_name = null;

    select c_name_fio
    from clients
    where c_id = :gral_from_c_id
    into client_fio;

    select c_name_fio
    from clients
    where c_id = :gral_other_c_id
    into rel_fio;

    select lower(un_name)
    from username
    where un_id = :gral_from_un_id
    into un_name;

    suspend;
  end
end
