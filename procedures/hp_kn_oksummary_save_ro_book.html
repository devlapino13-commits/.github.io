CREATE PROCEDURE HP_KN_OKSUMMARY_SAVE_RO_BOOK(
    REP_DATE_END D_DATE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable reg_id d_integer;
declare variable dt_ecl_vbs d_str_20;
declare variable kt_ecl_vbs d_str_20;
declare variable sum_ecl_vbs d_currency;
declare variable dt_delinq_os d_str_20;
declare variable kt_delinq_os d_str_20;
declare variable sum_delinq_os d_currency;
declare variable dt_delinq_perc d_str_20;
declare variable kt_delinq_perc d_str_20;
declare variable sum_delinq_perc d_currency;
declare variable ol_id d_id;
declare variable prv_reg_id d_integer;
declare variable un_id d_id;
declare variable ob_currency d_str_10;
declare variable oper_count d_integer;
declare variable book_count d_integer;
declare variable pat_name d_str_255;
declare variable dt_calc_perc d_str_20;
declare variable kt_calc_perc d_str_20;
declare variable sum_calc_perc d_currency;
begin
  res_is_error = 0;
  res_text = '';

  ob_currency = f_get_def_currency();
  un_id = uxf$user();

  oper_count = 0;
  book_count = 0;

  for select reg_id, pat_name,
             dt_ecl_vbs, kt_ecl_vbs, sum_ecl_vbs,
             dt_delinq_os, kt_delinq_os, sum_delinq_os,
             dt_delinq_perc, kt_delinq_perc, sum_delinq_perc,
             dt_calc_perc, kt_calc_perc, sum_calc_perc
  from hp_kn_oksummary_make_ro_book(:rep_date_end)
  where sum_ecl_vbs <> 0 or
        sum_delinq_os <> 0 or
        sum_delinq_perc <> 0 or
        sum_calc_perc <> 0
  order by reg_id, rec_id
  into reg_id, pat_name,
       dt_ecl_vbs, kt_ecl_vbs, sum_ecl_vbs,
       dt_delinq_os, kt_delinq_os, sum_delinq_os,
       dt_delinq_perc, kt_delinq_perc, sum_delinq_perc,
       dt_calc_perc, kt_calc_perc, sum_calc_perc
  do
  begin
    if (sum_ecl_vbs = 0 and sum_delinq_os = 0 and sum_delinq_perc = 0 and sum_calc_perc = 0) then
      Continue;

    if (coalesce(pat_name, '') = '') then
      pat_name = 'Займы';

    if (prv_reg_id is distinct from reg_id) then
    begin
      prv_reg_id = reg_id;

      insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_memo, ol_user_id,
                            ol_oper_subtype, ol_date_oper_day)
      values (:reg_id, 2, :rep_date_end, 'Начисление процентов и РО, перенос просрочек', :un_id, 11, :rep_date_end)
      returning (ol_id)
      into ol_id;

      oper_count = oper_count + 1;
    end

    if (sum_ecl_vbs <> 0) then
    begin
      book_count = book_count + 1;

      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_curr_unit,
                            ob_account_dt, ob_account_kt,
                            ob_summa_val, ob_summa,
                            ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, :ob_currency, 1, 1,
              :dt_ecl_vbs, :kt_ecl_vbs,
              :sum_ecl_vbs, :sum_ecl_vbs,
              'Корректировка РО: ' || :pat_name, :rep_date_end, 1, :reg_id);
    end

    if (sum_calc_perc <> 0) then
    begin
      book_count = book_count + 1;

      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_curr_unit,
                            ob_account_dt, ob_account_kt,
                            ob_summa_val, ob_summa,
                            ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, :ob_currency, 1, 1,
              :dt_calc_perc, :kt_calc_perc,
              :sum_calc_perc, :sum_calc_perc,
              'Начисление процентов: ' || :pat_name, :rep_date_end, 1, :reg_id);
    end

    if (sum_delinq_os <> 0) then
    begin
      book_count = book_count + 1;

      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_curr_unit,
                            ob_account_dt, ob_account_kt,
                            ob_summa_val, ob_summa,
                            ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, :ob_currency, 1, 1,
              :dt_delinq_os, :kt_delinq_os,
              :sum_delinq_os, :sum_delinq_os,
              'Перенос просроченного долга: ' || :pat_name, :rep_date_end, 1, :reg_id);
    end

    if (sum_delinq_perc <> 0) then
    begin
      book_count = book_count + 1;

      insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_curr_unit,
                            ob_account_dt, ob_account_kt,
                            ob_summa_val, ob_summa,
                            ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, :ob_currency, 1, 1,
              :dt_delinq_perc, :kt_delinq_perc,
              :sum_delinq_perc, :sum_delinq_perc,
              'Перенос просроченных процентов: ' || :pat_name, :rep_date_end, 1, :reg_id);
    end
  end

  if (book_count > 0) then
    res_text = 'Бух.проводки сформированы: ' || book_count;
  else
  begin
    res_is_error = 1;
    res_text = 'Нет проводок для формирования';
  end

  suspend;
end
