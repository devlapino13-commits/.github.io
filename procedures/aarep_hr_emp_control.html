CREATE PROCEDURE AAREP_HR_EMP_CONTROL(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME D_STR_CODE,
    C_NAME_FIO D_STR_100,
    INFO_LAST_WORK DATE,
    EQ_COMPANY D_STR_100,
    EQ_DATE_BEGIN D_DATE,
    EMP_KONTRACT_BEGIN D_DATE,
    EMP_KONTRACT_END D_DATE,
    CDS_DATE_MEMO D_STR_100)
AS
declare variable er_cancelled integer;
declare variable c_id d_id;
declare variable cds_doc_date_end d_date;
begin
  --2.       Добавить отчет по иностранным сотрудникам, двух видов:
  --ћ         Отчет по разрешениям на работу
  --ћ         Отчет по регистрациям
  --  select ST_USER_CLIENT_ACCESS, ST_USER_STAFF_ACCESS from GET_STORED_PROC_OPTIONS into :ST_CLIENT, :ST_STAFF;
  for select cl.c_name_fio, ps_from_c_id,r1.reg_code || ' ' || r1.reg_name
      from hr_pstatus h
      inner join hr_orders on ps_from_ho_id = ho_id
      inner join region_list(:rep_region_id, 1, 3, 1) on ps_from_reg_id = id_list
      inner join clients cl on cl.c_id = h.ps_from_c_id
      inner join joblist on ps_from_jl_id = jl_id
      inner join refdata rd on cl.c_citizen = rd.rd_id
      inner join region r1 on id_list = r1.reg_unicode
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            rd.rd_main_code = 'citizen' and
            rd.rd_code not in ('Citizen.RU', 'Citizen.KG', 'Citizen.KZ', 'Citizen.AM', 'Citizen.BY')

      into :c_name_fio, c_id,reg_name
  do
  begin
    info_last_work = null;
    --Дата последнего найма. Письмо от Патоновой Аиды. Добавил Музаффар>>>>>>>>>>>>>
    info_last_work = null;
    select max(ps_date_begin)
    from hr_pstatus
    inner join hr_orders on ps_from_ho_id = ho_id
    where ps_status = 1 and
          ps_from_c_id = :c_id and
          ho_order_type = 1 and
          ho_status = 1
    into :info_last_work;

    -- дата патента или разрешения на работу

    eq_date_begin = null;
    eq_company = null;
    select eq_date_begin, eq_company
    from empquota
    where eq_from_c_id = :c_id and
          :rep_date_end between eq_date_begin and eq_date_end
    order by eq_date_end desc
    rows 1
    into :eq_date_begin, :eq_company;
    if (eq_company not containing 'патент') then
      continue;

    /* select cds.cds_doc_date_begin, pk.pk_name, cds_doc_date_end
    from clientdocs cds, passkind pk, passdoctype pdt
    where cds_from_c_id = :c_id and
          cds_from_pk_id = pk.pk_id and
          pk.pk_from_pdt_id = pdt.pdt_id and
          pdt.pdt_code in ('work_patent') and
          cds.cds_is_delete = 0 and
          current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
    order by cds.cds_doc_date_end desc
    rows 1 --в случае нескольких действующих находим самый новый
    into :cds_doc_date_begin, pk_name, cds_doc_date_end;  */

    -- дата трудового договора
    emp_kontract_begin = null;
    emp_kontract_end = null;
    select cds.cds_doc_date_begin, cds_doc_date_end
    from clientdocs cds, passkind pk, passdoctype pdt
    where cds_from_c_id = :c_id and
          cds_from_pk_id = pk.pk_id and
          pk.pk_from_pdt_id = pdt.pdt_id and
          ((pdt.pdt_code = 'job_agreement') or
          (pdt.pdt_code = 'job_agreement_2')) and
          cds.cds_is_delete = 0 and
          current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
    order by cds.cds_doc_date_end desc
    rows 1 --в случае нескольких действующих находим самый новый
    into :emp_kontract_begin, emp_kontract_end;

    cds_date_memo = '';

    if (eq_date_begin is not null) then
    begin
      cds_doc_date_end = dateadd(month, 2, :eq_date_begin);

      if (coalesce(cds_doc_date_end, rep_date_end) <= :rep_date_end) then
      begin
        cds_date_memo = 'Срок сдачи ТД истек!';
      end

      if (cast((:cds_doc_date_end - :rep_date_end) as integer) between 1 and 7) then
      begin
        cds_date_memo = 'До окончания срока действия осталось меньше недели!';
      end
    end

      suspend;
  end

end
