CREATE PROCEDURE HP_FINEREFUND_ONE(
    FR_ID D_INTEGER)
AS
declare variable K_KONTRACT_NO varchar(20);
declare variable PAL_MAIN_ACC varchar(20);
declare variable ACC_DT1 varchar(20);
declare variable ACC_KT varchar(20);
declare variable ACC_DT varchar(20);
declare variable K_KOLICH_REPAY integer;
declare variable NEW_OL_ID1 D_ID;
declare variable K_KREDLINE_END date;
declare variable REP_DATE D_DATE;
declare variable OL_DATE D_DATE;
declare variable OK_FINE D_CURRENCY;
declare variable SH_ID D_INTEGER;
declare variable SH_DATE D_DATE;
declare variable K_LAST_REG_ID integer;
declare variable NEW_OL_ID D_ID;
declare variable K_ID D_ID;
declare variable C_ID D_ID;
begin
  for select fr.fr_from_k_id, fr.fr_from_c_id, cast(em.esm_ext_datetime as date)
  from finerefund fr
  inner join extsysmain em on em.esm_id = fr.fr_from_esm_id
  where fr.fr_id = :fr_id
  and fr.fr_status = 0
  into :k_id, :c_id, :rep_date
  do
  begin
      sh_id = null;
      sh_date = null;
      ol_date = null;
      ok_fine = null;
      for select ok_sh_id, ok_sh_date, max(ok_ol_date), sum(ok_fine)
          from operkred ok
          inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
          where ok_from_k_id = :k_id and
                ok_ol_is_active = 1 and
                ok_type = 4 and
                ok_ol_date <= :rep_date and
                ok.ok_sh_date >= '19.03.2020' and
                ol.ol_document_client = :c_id
          group by ok_sh_id, ok_sh_date
          having sum(ok_fine) < 0
          into sh_id, sh_date, ol_date, ok_fine
      do
      begin
        select pal_main_acc || '/' || pal_main_acc_no, k_kontract_no, k_last_reg_id, k_kolich_repay, k_kredline_end
        from kontract, product, prodacclist
        where k_id = :k_id and
              k_from_pr_id = pr_id and
              pr_from_pat_id = pal_from_pat_id and
              pal_from_at_id = 5
        into pal_main_acc, k_kontract_no, k_last_reg_id, k_kolich_repay, k_kredline_end;
    
        acc_kt = (select al_id
                  from acc_list
                  where al_from_reg_id = :k_last_reg_id and
                        al_currency = 417 and
                        al_account = '21199/1');
    
        if (coalesce(acc_kt, '') = '') then
          exception e_message 'счет кт не определен';
    
        acc_dt = (select al_id
                  from acc_list
                  where al_from_reg_id = :k_last_reg_id and
                        al_currency = 417 and
                        al_account = '10003/1003');
        if (coalesce(acc_dt, '') = '') then
            exception e_message 'счет дт не определен';
    
        acc_dt1 = (select al_id
                   from acc_list
                   where al_account = :pal_main_acc and
                         al_from_reg_id = :k_last_reg_id);
    
        if (coalesce(acc_dt1, '') = '') then
          exception e_message 'счет дт не определен';
    
        --создаем операцию
        select kv
        from get_kv(:k_last_reg_id, 5)
        into :new_ol_id;
    
        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                              ol_user_id, ol_doc_client_name)
        values (:new_ol_id, :k_last_reg_id, 1, :rep_date, 8,
                'корректировка по распоряжению №08/1 от 02.08.2021', 0, 0,
                (select un_id
                from username
                where un_name = 'fdjaparova'),
                'корректировка по распоряжению №08/1 от 02.08.2021');
        --ok_type = 4 погашение
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                              ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
        values (:new_ol_id, -1, :k_id, :sh_id, :sh_date, 4, 0, 0, -:ok_fine, 1, 0, 0, 0, 0, 0, :ol_date, :k_last_reg_id, 1,
                0);
    --    --ok_type = 9 переплата
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                              ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
        values (:new_ol_id, -1, :k_id, :k_kolich_repay, :rep_date, 9, :ok_fine, 0, 0, 1, 0, 0, 0, 0, 0, :ol_date,
                :k_last_reg_id, 1, 0);
    
        -- бух проводки
        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                              ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                              ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
                              ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt)
        values (:new_ol_id, -1, '417', 1, null, :acc_dt1, -:ok_fine, -:ok_fine, null, null, null, :acc_kt, null,
                f_format('возврат переплаты на баланс по кр.договору: %1', :k_kontract_no),
                0, 0, null, :ol_date, 1, :k_last_reg_id, :pal_main_acc, '21199/1');
    
         --создаем операцию
        select kv
        from get_kv(:k_last_reg_id, 5)
        into :new_ol_id1;
    
        insert into operlist (ol_id, ol_parent_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                              ol_user_id, ol_doc_client_name)
        values (:new_ol_id1, :new_ol_id,  :k_last_reg_id, 1, :rep_date, 8,
                'возврат по распоряжению №08/1 от 02.08.2021', 0, 0,
                (select un_id
                from username
                where un_name = 'fdjaparova'),
                'возврат по распоряжению №08/1 от 02.08.2021');
        --ok_type = 9 погашение
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                              ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
        values (:new_ol_id1, -1, :k_id, :k_kolich_repay, :rep_date, 9, -:ok_fine, 0, 0, 1, 0, 0, 0, 0, 0, :ol_date, :k_last_reg_id, 1,
                0);
    
        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                              ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                              ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
                              ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt)
        values (:new_ol_id1, -1, '417', 1, null, :acc_kt, -:ok_fine, -:ok_fine, null, null, null, :acc_dt, null,
                f_format('пополнение баланса по кр.договору: %1', :k_kontract_no), 0, 0, null,
                :ol_date, 1, :k_last_reg_id, '21199/1', '10003/1003');
    
        --suspend;
      end

      update finerefund fr set fr.fr_status = 1
      where  fr.fr_id = :fr_id;
   end
end
