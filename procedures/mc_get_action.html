CREATE PROCEDURE MC_GET_ACTION(
    CL_ACTION_CODE STR_100)
RETURNS (
    ACTION_ID ID_TYPE,
    CREATE_DATETIME TIMESTAMP,
    CONTACT_FIO STR_100,
    ADDRESS STR_255,
    ACTION_TYPE STR_100,
    ACTION_INFO STR_255,
    PHONE_NUMBER STR_100,
    CLIENT_FIO STR_100,
    OFFICE_CITY STR_100)
AS
declare variable rd_id integer;
declare variable object_kind integer;
declare variable object_id id_type;
begin
  if (cl_action_code <> 'mc_address_verify') then
    exit;

  select rd.rd_id
  from refdata rd
  where rd.rd_code = :cl_action_code
  into :rd_id;

  for select ao.aco_id, ao.aco_object_kind, ao.aco_object_id, al.acl_datetime
      from actionorder ao
      inner join actionlist al on ao.aco_id = al.acl_id
      where ao.aco_from_rd_id = :rd_id and
            ao.aco_status = 0
      into :action_id, :object_kind, :object_id, :create_datetime
  do
  begin
    if (object_kind = 0) then
    begin
      contact_fio = '';

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :object_id
      into :client_fio;

      phone_number = f_cont_info('clients', object_id, 'cell_main', 'ok');

      address = '';

      select adl.adl_address_str
      from addrclient adcl
      inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
      where adcl.adc_from_c_id = :object_id and
            adcl.adc_address_type = 5 -- фактический адрес на родине
      into :address;

      action_type = 'client_address_verify';
      action_info = 'Вам нужно осуществить визит-осмотр клиента [' || :client_fio || ']';
    end
    else
    begin
      address = '';

      select cont.cc_name, adl.adl_address_str, coalesce(cont.cc_cell_no, ''), cl.c_name_fio
      from contacts cont
      left join addrlist adl on adl.adl_id = cont.cc_from_adl_id
      inner join clients cl on cl.c_id = cont.cc_from_c_id
      where cont.cc_id = :object_id
      into :contact_fio, :address, :phone_number, :client_fio;

      action_type = 'contact_address_verify';
      action_info = 'Вам нужно осуществить визит-осмотр контактного лица [' || :contact_fio || ']';
    end

    office_city = 'г. Москва';

    suspend;
  end
end
