CREATE PROCEDURE PRINT_ZALOG(
    AAREP_K_ID DECIMAL(18,0),
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    LINE_NO INTEGER,
    Z_NUMBER INTEGER,
    Z_NAME VARCHAR(100),
    Z_ADDRESS VARCHAR(255),
    Z_C_ADDRESS VARCHAR(255),
    Z_DOCUMENT VARCHAR(255),
    Z_COST_MARKET DECIMAL(18,4),
    Z_COST_LIKVID DECIMAL(18,4),
    Z_COST_KONTRACT DECIMAL(18,4),
    Z_MEMO1 VARCHAR(1024),
    Z_MEMO2 VARCHAR(2048),
    Z_PHONE VARCHAR(100),
    Z_NAME_FIO VARCHAR(255),
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_TYPE INTEGER,
    CA_ADDRESS VARCHAR(255),
    Z_SUBTYPE VARCHAR(50),
    KC_PATENT_NO VARCHAR(50),
    KC_PATENT_DATE DATE,
    Z_BIRTH_DATE DATE,
    Z_PASS_WHO_CODE VARCHAR(20),
    Z_C_ALTERNAT_FIO VARCHAR(100),
    Z_C_EMAIL VARCHAR(100),
    C_BIRTH_CITY VARCHAR(100),
    CLIENT_INN VARCHAR(30),
    CLIENT_SNILS VARCHAR(30),
    CLIENT_NAME_OLD VARCHAR(255),
    CLIENT_NAME_WITH_LAT VARCHAR(255),
    CLIENT_PASS_LIST_OLD D_STR_8100,
    QR_CODE D_STR_LONG,
    EDL_ID D_ID,
    C_INN STR_20)
AS
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable Z_FROM_C_ID decimal(18,0);
declare variable KK integer;
declare variable C_ALTERNAT_FIO varchar(100);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  line_no = 0;

  for select z_id, z_name, z_document, z_cost_market, z_cost_likvid, z_cost_kontract, z_memo1, z_memo2, z_phone,
             z_name_fio, c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_type, adl_address_str,
             zs_name, kc_patent_no, kc_patent_date, z_from_c_id, c_birth_date, c.c_pass_who_code, c_name_eng, c_birth_city, trim(iif(c_citizen = 13, '', c_name_eng)),
             (select cont_value
              from hp_contdata_get_email_main(0, c_id)), c_inn
      from knclient, addrlist, zalog z
      left join clients c on (z_from_c_id = c_id)
      left join zalogsubtype on (z_subtype = zs_id)
      where z_from_k_id = :aarep_k_id and
            z_type = 1 and
            kc_from_k_id = :aarep_k_id and
            adl_id = z_from_adl_id
      order by z_id desc
      rows 1
      into :z_number, :z_name, :z_document, :z_cost_market, :z_cost_likvid, :z_cost_kontract, :z_memo1, :z_memo2,
           :z_phone, :z_name_fio, :c_name_fio, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_type,
           :ca_address, :z_subtype, :kc_patent_no, :kc_patent_date, :z_from_c_id, z_birth_date, z_pass_who_code, :z_c_alternat_fio,
            :c_birth_city, :c_alternat_fio, :z_c_email, :c_inn


  do
  begin
    z_address = ca_address;
    line_no = line_no + 1;
    if (z_name_fio is null) then
      z_name_fio = :c_name_fio;

    client_name_with_lat = z_name_fio;

    if (c_alternat_fio <> '') then
        client_name_with_lat = client_name_with_lat || ' (' || c_alternat_fio || ') ';

    select cds_doc_no
    from clientdocs
    where cds_from_c_id = :z_from_c_id
    and cds_from_pk_id = 61
    and cds_is_delete = 0
    order by cds_doc_date_begin desc 
    rows 1
    into client_snils;
    
    select cds_doc_no
    from clientdocs
    where cds_from_c_id = :z_from_c_id
    and cds_from_pk_id = 62
    and cds_is_delete = 0
    order by cds_doc_date_begin desc 
    rows 1
    into client_inn;

    select list(distinct(drv_final_value))
    from docrecogmain
    inner join docrecoglist on drl_from_drm_id = drm_id
    inner join docrecogfield on drf_from_drm_id = drm_id
    inner join docrecogvalue on drv_from_drf_id = drf_id and drv_from_drl_id = drl_id
    where
    drm_code = 'rus.cert.name_change' and
    drf_code = 'client_name_old' and
    drl_from_c_id = :z_from_c_id
    into client_name_old;

    if (client_name_old is null) then
        client_name_old = '';

    select
    left(list(distinct(cds_doc_no || ' ' ||
                lpad(extract(day from cds_doc_date_begin), 2, '0') || '.' ||
                lpad(extract(month from cds_doc_date_begin), 2, '0') || '.' ||
                extract(year from cds_doc_date_begin)  || ' ' || cds_doc_issued_by || ' ' || f_docrecog_value(drl_id, 'authority_code', ''))), 8100)
    from hp_clientdocs_pass_list(:z_from_c_id, 1)
    where cds_from_pk_id in (1,6,8,11,12,15,17,34,35,36,42,296,330)
    into client_pass_list_old;

    select adl_address_str
    from addrclient ac, addrlist al
    where ac.adc_from_adl_id = al.adl_id and
          ac.adc_address_type = 1 and
          ac.adc_from_c_id = :z_from_c_id
    into z_c_address; --Адрес залогодателя

    if (not((z_c_alternat_fio containing ('uulu')) or (z_c_alternat_fio containing ('kyzy')))) then
    begin
      kk = position(' ', :z_c_alternat_fio);
      if (kk > 0) then
      begin
        kk = position(' ', z_c_alternat_fio, kk + 1);
        if (kk > 0) then
          z_c_alternat_fio = substring(:z_c_alternat_fio from 1 for kk - 1);
      end
    end

    -- QR-код электронной подписи клиента. Текст пустой если не подписан
    if (coalesce(add_params, '') <> '') then
    begin
      edl_id = f_json_str_get_value(add_params, 'edl_id', null);
      qr_code = f_edoc_qr_by_edl(:edl_id, :z_from_c_id);
    end

    suspend;
  end
  if (line_no = 0) then suspend;
end
