CREATE PROCEDURE HP_EDOC_WEB_GET_CL_INFO(
    CL_UNIQ_ACC D_INTEGER,
    CL_BIRTHDATE D_DATE,
    CL_LANG D_STR_LANG)
RETURNS (
    ERR_NO D_SMALLINT,
    ERR_TEXT D_STR_255,
    MNT_SUBSCRIBER_ID D_ID,
    C_NAME_FIO D_STR_100,
    MNT_DESCRIPTION D_STR_255,
    MNT_ID D_ID,
    SUGGESTED_AMOUNT D_STR_20,
    KONTRACT_INFO D_STR_255,
    K_ID D_ID,
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID)
AS
declare variable cur_name d_str_20;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable pr_category_str d_str_code;
declare variable pr_category d_smallint;
declare variable has_loans d_smallint;
declare variable db_id d_integer;
declare variable can_show_loan integer;
declare variable dolg_vu d_currency;
declare variable k_curr_other d_currency;
declare variable is_client_bankrupt d_bool;
begin
  -- ВАЖНО! SUGGESTED_AMOUNT должен быть строкового типа, иначе фронт ломается RFCKG-1725

  err_no = 0;
  has_loans = 0;
  err_text = '';
  mnt_description = null;
  mnt_id = null;
  c_name_fio = null;
  db_id = f_get_db_id();
  is_client_bankrupt = 0;

  -- Проверка что uniq_id клиента передали
  if (err_no = 0 and
      cl_uniq_acc is null) then
  begin
    err_no = 1;
    err_text = f_lang(cl_lang, 'Лицевой счет не указан');
  end

  -- Проверка что дата рождения клиента указана
  if (err_no = 0 and
      cl_birthdate is null) then
  begin
    err_no = 2;
    err_text = f_lang(cl_lang, 'Дата рождения не указана');
  end

  if (err_no <> 0) then
  begin
    suspend;
    exit;
  end

  select c_id
  from hp_check_clientuniq_acc_no(:cl_birthdate, :cl_uniq_acc)
  into mnt_subscriber_id;

  if (mnt_subscriber_id = -1) then
  begin
    err_no = 3;
    err_text = f_lang(cl_lang, 'Клиент не найден, обратитесь в офис компании');
    suspend;
    exit;
  end

  mnt_description = '';
  select short_fio
  from clients, hp_fio_clear(c_name_fio)
  where c_id = :mnt_subscriber_id
  into c_name_fio;

  cur_name = (select cm_currency_name
              from company);

  if (db_id = 81001) then
    cur_name = left(cur_name, 3) || '.'; -- рубль -> руб.

  -- RFCRU-5274: Если клиент банкрот со статусом "Реализация имущества завершена", то рекомендуемую сумму платежа выводим "0*"
  if (db_id = 81001 and
      exists(select *
             from bankrupt brt, hp_get_kontur_info(brt.brt_from_eil_id) ki
             where brt.brt_from_c_id = :mnt_subscriber_id and
                   current_date between brt.brt_date_begin and brt.brt_date_end and
                   ki.status = 'Реализация имущества завершена')
  ) then
    is_client_bankrupt = 1;

  obj_kind = 'kontract';
  for select oks_from_k_id
  from oksummary oks
  inner join knclient kc on oks.oks_from_k_id = kc.kc_from_k_id
    and kc.kc_from_c_id = :mnt_subscriber_id
    and kc.kc_amount_approved > 0
  where oks.oks_close_state = 0
    and current_date between oks.oks_date_begin and oks.oks_date_end
  union
  select oko.oko_from_k_id
  from operkroff oko
  where oko.oko_from_c_id = :mnt_subscriber_id
    and oko.oko_ol_is_active = 1
  group by oko.oko_from_k_id
  having sum(oko.oko_os + oko.oko_perc + oko.oko_sh_os + oko.oko_sh_perc) > 0
  into k_id
  do
  begin
    if (db_id = 81001) then
    begin
        select iif(pr.pr_code containing '.oivo_pack.', f_get_const('moneta_oivo_acc'), f_get_const('moneta_acc'))
        from kontract kont
        inner join product pr on pr.pr_id = kont.k_from_pr_id
        where kont.k_id = :k_id
        into :mnt_id;
    end
    else
        mnt_id = f_get_const('moneta_acc');

    can_show_loan = 1;
    suggested_amount = 0;

    -- Убеждаемся, что отсутствуют необработанные pay-записи по договору или клиенту
    if (exists(select esm_from_k_id
               from extsysmain
               where esm_object_kind = 0 and
                     esm_object_id = :mnt_subscriber_id and
                     esm_action = 'pay' and
                     esm_status = 0 and
                     coalesce(esm_from_k_id, :k_id) = :k_id)) then
      can_show_loan = 0;

    -- Если по договору есть оплаты поздней датой, не даем платить по нему.
    -- Например вечером через кассу платят завтрашним днем. Если после этого
    -- принять через платежную систему, сядет текущей датой и все сломается

    if (can_show_loan = 1 and
        exists(select *
               from operkred
               where ok_ol_date > current_date and
                     ok_from_k_id = :k_id and
                     ok_ol_is_active = 1)) then
      can_show_loan = 0;

    if (can_show_loan = 1) then
    begin
      select round(recom_pay_sum, 2)
      from get_recommended_pay_sum(:k_id, current_date)
      into suggested_amount;

      if (db_id = 81001 and is_client_bankrupt = 1) then
        suggested_amount = 0; -- Для банкротов рекомендуемая сумма платежа должна быть 0

      if (db_id = 39801) then  -- Для Каспи и Процессинг в рекомендуемую сумму платежа необходимо включать начисленные суммы госпошлины и расходов по исполнительной надписи
      begin
        k_curr_other = (select coalesce(sum(res_amount_saldo), 0)
                        from hp_repay_extra_saldo(:k_id, 0, current_date));

        if (k_curr_other > 0) then
          suggested_amount = cast(suggested_amount as d_currency) + k_curr_other;
      end 
    end

    if (can_show_loan = 1 and
        cast(suggested_amount as d_currency) >= 0) then
    begin
      has_loans = 1;
      select k_kontract_no, k_disb_date, pr_category
      from kontract, product
      where k_id = :k_id and
            k_from_pr_id = pr_id
      into k_kontract_no, k_disb_date, pr_category;

      pr_category_str = f_format('(%1)', f_lang(:cl_lang, trim(iif(pr_category = 0, 'Основной', 'Параллельный'))));

      -- Без перевода, т.к. текст для Монеты
      mnt_description = f_format('Погашение займа по договору №%1, %2, без НДС', k_kontract_no, c_name_fio);

      if (exists(select *
                 from operkroff oko
                 where oko.oko_from_k_id = :k_id) and
          db_id in (39801, 41701)) then
      begin
        select (sh_os + sh_perc + sh_other)
        from aarep_operkroff_client(:mnt_subscriber_id)
        where k_id = :k_id
        into dolg_vu;
        dolg_vu = coalesce(dolg_vu, 0);
        suggested_amount = round(dolg_vu, 2);
        kontract_info = f_format(f_lang(cl_lang, 'Договор №%1 от %2 %3. Рекомендуемая сумма %4 %5'),
            k_kontract_no, f_datetostr(k_disb_date), pr_category_str, suggested_amount, cur_name);

        kontract_info = replace(kontract_info, 'Рекомендуемая сумма', 'Ваша задолженность составляет');
      end
      else
      begin
        kontract_info = f_format(f_lang(cl_lang, 'Договор №%1 от %2 %3. Рекомендуемая сумма %4 %5'),
            k_kontract_no, f_datetostr(k_disb_date), pr_category_str, iif (db_id = 81001 and is_client_bankrupt = 1, '0*', suggested_amount), cur_name);

        if (db_id = 81001 and is_client_bankrupt = 1) then
          kontract_info = kontract_info || ' (*Вы освобождены  от исполнения обязательств в связи с вашим банкротством)';
      end

      obj_id = k_id;

      suspend;
    end
  end

  if (db_id = 41701) then -- отображение консультационных услуг (RFCKG-4903)
  begin
    obj_kind = 'knclient';
    for select oks_from_k_id
    from oksummary oks
    inner join kontract k on k.k_id = oks.oks_from_k_id
    inner join knclient kc on oks.oks_from_k_id = kc.kc_from_k_id
      and kc.kc_from_c_id = :mnt_subscriber_id
      and kc.kc_amount_approved > 0
    where oks.oks_close_state = 0
      and current_date between oks.oks_date_begin and oks.oks_date_end
      and coalesce(k_cons_kind, 0) > 0 -- конс. услуги в рассрочку
    union
    select oko.oko_from_k_id
    from operkroff oko
    inner join kontract k on k.k_id = oko.oko_from_k_id
    where oko.oko_from_c_id = :mnt_subscriber_id
      and oko.oko_ol_is_active = 1
      and coalesce(k_cons_kind, 0) > 0 -- конс. услуги в рассрочку
    group by oko.oko_from_k_id
    having sum(oko.oko_os + oko.oko_perc + oko.oko_sh_os + oko.oko_sh_perc) > 0
    into k_id
    do
    begin
      suggested_amount = null;
      mnt_description = null;
      kontract_info = null;

      select kc_id,
             amount_recom,
             k_info || '. ' || text_recom
      from hp_consult_installment_amount(:k_id)
      into obj_id,
           suggested_amount,
           kontract_info;

      if (coalesce(suggested_amount, 0) > 0) then
      begin
        kontract_info = 'Консультационные услуги №' || kontract_info;
        mnt_description = 'Оплата консультационных услуг, ' || c_name_fio;
        suspend;
      end
    end
  end

  if (has_loans = 0) then
  begin
    -- Нет активных кредитов, но прямо мы об этом не говорим
    err_no = 4;
    err_text = f_lang(cl_lang, 'Произошла ошибка. Обратитесь в офис компании');
    suspend;
    exit;
  end
end
