    C_PASS_NO VARCHAR(30),

declare variable C_ID D_ID;
declare variable CREATOR D_ID;
declare variable OBJECT_KIND D_INTEGER;
declare variable EDL_ID D_ID;
declare variable C_HIRED_DATE date;
begin
  ag_commis_cards_mb = 0;
  ag_commis_cards_other = 0;
  c_id = 0;
  object_kind = 0;
  edl_id = null;
  if (object_id <= 1000000000000) then
  begin
    select ag.ag_object_id, coalesce(ag.ag_commis_cards_mb, 0), coalesce(ag.ag_commis_cards_other, 0)
    from agreements ag
    where ag.ag_id = :object_id
    into c_id, ag_commis_cards_mb, ag_commis_cards_other;
    :object_id = :c_id;

    object_kind = f_objkind('agreements', 'id');
  end

  select first_time, ank_filled_first_time_txt, ank_update_data_txt, c_name_fio, res_non_resident, c_birth_date,
         c_birth_city, c_nation_str, cl_sex, c_citizen_str, c_family_str, c_pass_no, c_pass_date, c_pass_who,
         c_pass_date_expire, c_inn, c_address_yur, c_address_fact, last_cell_main, c_email, c_is_public,
         c_is_benefic_owner, registr_type, registr_doc_no, registr_date_begin, registr_date_end, registr_issued_by,
         registr_memo_1, registr_memo_2, pant_type, pant_doc_no, pant_date_begin, pant_date_end, pant_issued_by,
         pant_memo_1, pant_memo_2, mac_loan_target_original, c_pft_risk_koef, c_pft_risk_memo, print_date, create_date,
         lo_fio, lo_id, lo_jl_name, male_txt, female_txt, ank_high_risk_txt, ank_low_risk_txt, c_org_type_long,
         c_name_full, c_alternat_fio, c_form_status_short, c_okpo, c_datetime, qr_code, mac_loan_target,
         kc_loan_target_original, c_authorized_capital, c_branches, c_doc_inform, c_goals, c_pass_who_code, reg_name,
         pl_name, pant_infolic1, pant_infolic2, pant_infolic3, kml_bank_name, kml_bank_no, kml_bank_bik, kml_polis_type,
         kml_polis_doc_no, kml_polis_date_begin, kml_polis_date_end, kml_polis_issued_by, hr_reg_name, ver_fio,
         sign_date, ksr_kont_titul, doc_no_legal, doc_date_legal, doc_issued_by_legal, pk_name , pant_memo3
  from print_client_cli(:object_id, :add_params)
  into first_time, ank_filled_first_time_txt, ank_update_data_txt, c_name_fio, res_non_resident, c_birth_date,
       c_birth_city, c_nation_str, cl_sex, c_citizen_str, c_family_str, c_pass_no, c_pass_date, c_pass_who,
       c_pass_date_expire, c_inn, c_address_yur, c_address_fact, last_cell_main, c_email, c_is_public,
       c_is_benefic_owner, registr_type, registr_doc_no, registr_date_begin, registr_date_end, registr_issued_by,
       registr_memo_1, registr_memo_2, pant_type, pant_doc_no, pant_date_begin, pant_date_end, pant_issued_by,
       pant_memo_1, pant_memo_2, mac_loan_target_original, c_pft_risk_koef, c_pft_risk_memo, print_date, create_date,
       lo_fio, lo_id, lo_jl_name, male_txt, female_txt, ank_high_risk_txt, ank_low_risk_txt, c_org_type_long,
       c_name_full, c_alternat_fio, c_form_status_short, c_okpo, c_datetime, qr_code, mac_loan_target,
       kc_loan_target_original, c_authorized_capital, c_branches, c_doc_inform, c_goals, c_pass_who_code, reg_name,
       pl_name, pant_infolic1, pant_infolic2, pant_infolic3, kml_bank_name, kml_bank_no, kml_bank_bik, kml_polis_type,
       kml_polis_doc_no, kml_polis_date_begin, kml_polis_date_end, kml_polis_issued_by, hr_reg_name, ver_fio, sign_date,
       ksr_kont_titul, doc_no_legal, doc_date_legal, doc_issued_by_legal, pk_name ,pant_memo3;

  select c.c_type, cast(c_pass_last_change as date), c.c_memo, c.c_hired_date
  from clients c
  where c.c_id = :object_id
  into c_type, last_date, c_memo, c_hired_date;

  -- BOARD-5325
  -- Если последнее изменение данных клиента были сделаны верификатором (пользователь без кадрового приказа) отображать информацию до верификатора
  with q as (
    select cll_from_un_id, cll_datetime
    from clientlog cll
    where cll.cll_from_c_id = :object_id
    union all
    select c_from_un_id, cl.c_datetime
    from clients cl
    where cl.c_id = :object_id)
        select un_fio, cll_from_un_id
        from q
        inner join username un on un.un_id = cll_from_un_id
        left join hr_pstatus ps on ps.ps_from_c_id = un.un_from_c_id
        where ps.ps_id is not null and
              ps.ps_is_working = 1 and
              ps.ps_is_on_staff = 1 and
              cast(cll_datetime as date) between ps.ps_date_begin and ps.ps_date_end
        order by cll_datetime desc
        rows 1
        into creator_fio, creator;
  -- BOARD-5325 КОНЕЦ

  if (last_date is null) then
    last_date = c_hired_date;

  select cb_ben_name, rd_name
  from hp_client_benefic(:object_id)
  where cb_is_signer = 1 rows 1
  into :cb_ben_name_yur, :rd_name;

  --edl_id = f_json_str_get_value(:add_params, 'edl_id', null);
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    edoc_number = edl_id;
  end

  if (edl_id is not null) then
  begin
    -- ФИО отправившего ссылку, дата подписания
    select c_id, c_name_fio, cast(eds_status_datetime as date), edl_create_datetime, cast(edl_create_datetime as date),
           un_name, cast(edl.edl_create_datetime as date)
    from edoc_list edl, edoc_sign, username, clients
    where edl_id = :edl_id and
          eds_from_edl_id = edl_id and
          eds_from_c_id = :object_id and
          edl_create_un_id = un_id and
          un_from_c_id = c_id
    into un_c_id, edoc_create_un_fio, date_sign, edoc_create_datetime, edoc_create_date, edoc_create_un_name, edoc_date;

    pant_infolic1 = '';
    pant_infolic2 = '';
    pant_infolic3 = '';
    for select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by,
               cd.cds_memo_1, cd.cds_memo_2
        from clientdocs cd
        inner join passkind pk on cd.cds_from_pk_id = pk_id
        inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
        where cds_from_c_id = :c_id and
              :edoc_create_date between cds_doc_date_begin and cds_doc_date_end and
              pdt.pdt_code in ('work_patent', 'License')
        order by cds_doc_date_end desc
        rows 1
        into :pant_type, :pant_doc_no, :pant_date_begin, :pant_date_end, :pant_issued_by, :pant_memo_1, :pant_memo_2
    do
    begin
      pant_infolic1 = pant_infolic1 || iif(pant_infolic1 = '', '', ascii_char(10)) || pant_doc_no || ', ' || pant_type || ', ' || f_datetostr(pant_date_begin) || ', ' || pant_issued_by || ', ' || f_datetostr(pant_date_end);
      pant_infolic2 = pant_infolic2 || iif(pant_infolic2 = '', '', ascii_char(10)) || :pant_memo_1;
      pant_infolic3 = pant_infolic3 || iif(pant_infolic3 = '', '', ascii_char(10)) || 'добровольного патента, №' || pant_doc_no || ' от ' || f_datetostr(pant_date_begin) || ' года';
    end
   date_sign_next_year = cast(extract(day from :date_sign)||'.'||extract(month from :date_sign )||'.'||(extract(year from :date_sign ) +1) as date);
  end

  last_year_date = '';
  last_year_date = '31.12.' || extract(year from :edoc_create_date);
  suspend;
end
