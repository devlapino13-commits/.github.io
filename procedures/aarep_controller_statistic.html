CREATE PROCEDURE AAREP_CONTROLLER_STATISTIC(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    CONTR_TYPE SMALLINT)
RETURNS (
    VISITOR STR_100,
    CONFIRMED INTEGER,
    NOTCONFIRMED INTEGER,
    REFUSED INTEGER,
    ADDRESSINCORRECT INTEGER,
    INFOCONFIRMED INTEGER,
    CLIENTDIDNOTWORK INTEGER,
    PLANSLEAVECITY INTEGER,
    CHANGEJOB INTEGER,
    CLIENTRESIGNED INTEGER,
    CLIENTFIRED INTEGER,
    NEGATIVEREPUTATION INTEGER,
    ADDRESSINCOMPLETE INTEGER,
    ADDRESSNOTEXISTS INTEGER,
    LODISCRETION INTEGER,
    CHECKQTY INTEGER,
    OFFICE_NAME STR_100)
AS
declare variable c_id id_type;
declare variable rd_id integer;
begin
  /*
    CONTR_TYPE  = 1 - СВО
                  2 - Контролер
    */
  select rd.rd_id
  from refdata rd
  where rd.rd_code = (case :contr_type
                        when 1 then 'ControllerType.Specialist_Visitor'
                        when 2 then 'ControllerType.Controller'
                      end)
  into :rd_id;

  for select visitor, c_id, sum(confirmed) confirmed, sum(notconfirmed) notconfirmed, sum(refused) refused,

             sum(addressincorrect) addressincorrect, sum(infoconfirmed) infoconfirmed,
             sum(clientdidnotwork) clientdidnotwork,

             sum(plansleavecity) plansleavecity, sum(changejob) changejob, sum(clientresigned) clientresigned,

             sum(clientfired) clientfired, sum(negativereputation) negativereputation,
             sum(addressincomplete) addressincomplete,

             sum(addressnotexists) addressnotexists, sum(lodiscretion) lodiscretion,
             sum(confirmed + notconfirmed + addressincorrect + infoconfirmed + clientdidnotwork + plansleavecity + changejob + clientresigned + clientfired + negativereputation + addressincomplete + addressnotexists + lodiscretion) as checkqty
      from (select cl.c_name_fio as visitor, cl.c_id,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.Confirmed', 1, 0) as confirmed,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.NotConfirmed', 1, 0) as notconfirmed,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.Refused', 1, 0) as refused,

                   iif(rd_res.rd_code = 'ControllerCheckStatus.AddressIncorrect', 1, 0) as addressincorrect,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.InfoConfirmed', 1, 0) as infoconfirmed,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.ClientDidNotWork', 1, 0) as clientdidnotwork,

                   iif(rd_res.rd_code = 'ControllerCheckStatus.PlansLeaveCity', 1, 0) as plansleavecity,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.ChangeJob', 1, 0) as changejob,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.ClientResigned', 1, 0) as clientresigned,

                   iif(rd_res.rd_code = 'ControllerCheckStatus.ClientFired', 1, 0) as clientfired,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.NegativeReputation', 1, 0) as negativereputation,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.AddressIncomplete', 1, 0) as addressincomplete,

                   iif(rd_res.rd_code = 'ControllerCheckStatus.AddressNotExists', 1, 0) as addressnotexists,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.LODiscretion', 1, 0) as lodiscretion
            from controller cont
            inner join refdata rd_res on rd_res.rd_id = cont.contr_check_result
            inner join username un on un.un_id = cont.contr_from_un_id
            inner join clients cl on cl.c_id = un.un_from_c_id
            where cont.contr_checked_date is not null and
                  cont.contr_rd_type = :rd_id and
                  cont.contr_is_delete in (0, 2) and
                  cont.contr_checked_date between :rep_date_begin and :rep_date_end) a
      group by visitor, c_id
      into :visitor, :c_id, :confirmed, :notconfirmed, :refused,

           :addressincorrect, :infoconfirmed, :clientdidnotwork,

           :plansleavecity, :changejob, :clientresigned,

           :clientfired, :negativereputation, :addressincomplete,

           :addressnotexists, :lodiscretion, :checkqty
  do
  begin
    select reg.reg_name
    from hr_pstatus hps
    inner join region reg on reg.reg_unicode = hps.ps_from_reg_id
    where current_date between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          hps.ps_from_c_id = :c_id
    into :office_name;

    suspend;
  end
end
