CREATE PROCEDURE FSROBOT_GET_KIND(
    MAIL_TO VARCHAR(1024),
    MAIL_SUBJECT VARCHAR(1024))
RETURNS (
    FSR_KIND INTEGER)
AS
declare variable tmp_str varchar(1024);
declare variable c_email varchar(255);
declare variable n integer;
begin
  tmp_str = mail_to;
  while (tmp_str <> '') do
  begin
    c_email = '';
    n = position(';', tmp_str);
    if (n > 0) then
    begin
      c_email = substring(tmp_str from 1 for n - 1);
      tmp_str = substring(tmp_str from n + 1);
    end
    else
    begin
      c_email = tmp_str;
      tmp_str = '';
    end

    if (c_email <> '') then
    begin
      if (exists(select *
                 from refdata
                 where rd_main_code = 'mailgroup' and
                       rd_str_id = :c_email)) then
      begin
        fsr_kind = 3;
        suspend;
        exit;
      end
    end
  end

  select fsr_kind
  from fsrobot
  where :mail_subject starting fsr_text rows 1
  into :fsr_kind;

  if (fsr_kind is null) then
    fsr_kind = -1;
  suspend;
end
