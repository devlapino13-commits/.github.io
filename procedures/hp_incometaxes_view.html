CREATE PROCEDURE HP_INCOMETAXES_VIEW(
    REGP_ID D_INTEGER,
    DT_BEG D_DATE,
    DT_END D_DATE,
    ACCDT D_STR_20,
    ACCKT D_STR_20,
    TAX_CODE D_INTEGER,
    DESCR D_STR_LONG,
    UN_ID D_ID,
    IS_KEEP D_BOOL)
RETURNS (
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    AMOUNT_COMM D_CURRENCY,
    AMOUNT_NDS D_CURRENCY,
    AMOUNT_NSP D_CURRENCY,
    AMOUNT_INCOME D_CURRENCY)
AS
declare variable ol_id d_id;
declare variable ol_parent_id d_id;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable num d_integer;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable total_line d_integer;
declare variable res_line_no d_integer;
begin
  for
    select
      id_list
    from region_list(:regp_id, 1, 3, 1)
  into
    :reg_id
  do
  begin
    acc_dt = (
      select
        al_id
      from acc_list
      where al_from_reg_id = :reg_id
        and al_currency = '417'
        and al_account = :accdt);
    if (acc_dt is not null) then
    begin
      amount_comm = (
        select kt_val - dt_val
        from hp_al_obor(:acc_dt, :dt_beg, :dt_end));

      -- ненулевое кредитовое сальдо за период
      if (amount_comm > 0) then
      begin
        reg_name = (select reg_name from region where reg_unicode = :reg_id);
        amount_nds = 0;
        amount_nsp = 0;
        if (tax_code = 100) then
        begin
          amount_nds = round(12 * amount_comm / 114, 2);
          amount_nsp = round(2 * amount_comm / 114, 2);
        end else
        if (tax_code <> 0) then
          amount_nsp = round(2 * amount_comm / 102, 2);
        amount_income = amount_comm - amount_nds - amount_nsp;

        if (is_keep = 1) then
        begin
          num = (select num from last_doc_number(current_date, 9, :reg_id));
          insert into operlist (
            ol_from_reg_id,
            ol_doc_no,
            ol_oper_type,
            ol_oper_subtype,
            ol_date,
            ol_doc_kind,
            ol_memo,
            ol_user_id)
          values (
            :reg_id,
            :num,
            2,
            23,
            :dt_end,
            9,
            'Признание доходов',
            :un_id)
          returning
            ol_id
          into
            :ol_id;
          ol_parent_id = ol_id;

          -- проводка по НДС
          if (amount_nds > 0) then
          begin
            acc_kt = (
              select
                al_id
              from acc_list
              where al_from_reg_id = 1090100
                and al_currency = '417'
                and al_account = '21131/25');
            for
              select
                res_reg_id,
                res_acc_dt,
                res_acc_kt,
                total_line,
                res_line_no
              from aarep_loronostro_link(:reg_id, :acc_dt, :acc_kt)
            into
              :res_reg_id,
              :res_acc_dt,
              :res_acc_kt,
              :total_line,
              :res_line_no
            do
            begin
              if (res_line_no > 1) then
                insert into operlist (
                  ol_parent_id,
                  ol_from_reg_id,
                  ol_doc_no,
                  ol_oper_type,
                  ol_oper_subtype,
                  ol_date,
                  ol_doc_kind,
                  ol_memo,
                  ol_user_id)
                values (
                  :ol_parent_id,
                  :res_reg_id,
                  :num,
                  2,
                  23,
                  :dt_end,
                  9,
                  'Признание доходов',
                  :un_id)
                returning
                  ol_id
                into
                  :ol_id;

              insert into operbook (
                ob_from_ol_id,
                ob_currency,
                ob_curr_rate,
                ob_account_dt,
                ob_account_kt,
                ob_detail,
                ob_summa,
                ob_summa_val)
              values (
                :ol_id,
                '417',
                1,
                :res_acc_dt,
                :res_acc_kt,
                'НДС к оплате: ' || :descr,
                :amount_nds,
                :amount_nds);
            end
          end

          -- проводка по НСП
          if (amount_nsp > 0) then
          begin
            acc_kt = (
              select
                al_id
              from acc_list
              where al_from_reg_id = :reg_id
                and al_currency = '417'
                and al_account = '21131/21');

            insert into operbook (
              ob_from_ol_id,
              ob_currency,
              ob_curr_rate,
              ob_account_dt,
              ob_account_kt,
              ob_detail,
              ob_summa,
              ob_summa_val)
            values (
              :ol_parent_id,
              '417',
              1,
              :acc_dt,
              :acc_kt,
              'НСП к оплате: ' || :descr,
              :amount_nsp,
              :amount_nsp);
          end

          -- проводка по признанию дохода
          if (amount_income > 0) then
          begin
            acc_kt = (
              select
                al_id
              from acc_list
              where al_from_reg_id = :reg_id
                and al_currency = '417'
                and al_account = :acckt);

            insert into operbook (
              ob_from_ol_id,
              ob_currency,
              ob_curr_rate,
              ob_account_dt,
              ob_account_kt,
              ob_detail,
              ob_summa,
              ob_summa_val)
            values (
              :ol_parent_id,
              '417',
              1,
              :acc_dt,
              :acc_kt,
              :descr,
              :amount_income,
              :amount_income);
          end
        end else suspend;
      end
    end
  end
end
