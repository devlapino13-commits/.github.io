CREATE PROCEDURE HP_EDOC_BLANK_PREVIEW(
    EDL_ID D_ID,
    EDB_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255,
    BLOB_LIST D_STR_8100)
AS
declare variable bls_item d_str_long;
declare variable edf_print_name d_str_100;
declare variable orig_bls_id d_id;
declare variable scan_bls_id d_id;
declare variable res_edb_id d_id;
declare variable pass_bls_id d_id;
declare variable photo_bls_id d_id;
declare variable c_name_fio d_str_100;
declare variable photo_change_bls_id d_id;
declare variable edbs_status d_smallint;
declare variable edbs_status_datetime d_date;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  res_is_error = 1;
  res_text = '';
  blob_list = '';

  if (edl_id is null) then
    edl_id = 0;
  if (edb_id is null) then
    edb_id = 0;

  if (edb_id <> 0) then
  begin
    edl_id = 0;
    select edb_from_edl_id
    from edoc_blank
    where edb_id = :edb_id
    into edl_id;
  end

  if (edl_id = 0) then
    res_text = f_format('Электронный документ не найден, EDL_ID %1, EDB_ID %2', edl_id, edb_id);

  if (res_text = '') then
  begin
    for select edb_id, edf_print_name, edb_from_bls_id, edbs_from_bls_id
        from edoc_blank
        inner join edoc_frx on (edb_from_edf_id = edf_id)
        left join edoc_blank_scan on (edbs_from_edb_id = edb_id)
        where edb_from_edl_id = :edl_id and
              (:edb_id = 0 or
              edb_id = :edb_id)
        into res_edb_id, edf_print_name, orig_bls_id, scan_bls_id
    do
    begin
      if (scan_bls_id is not null) then
      begin
        bls_item = '{"caption": "%1", "blobs":[%2, %3]}';
        if (orig_bls_id is not null) then
        begin
          bls_item = f_format(bls_item, edf_print_name, f_image_item('blobstore', scan_bls_id, 'Скан', un_id, 0, 1, 1), f_image_item('blobstore', orig_bls_id, 'Оригинал', un_id, 0, 1, 1));
        end
        else
        begin
          bls_item = f_format(bls_item, edf_print_name, f_image_item('blobstore', scan_bls_id, 'Скан', un_id, 0, 1, 1), f_image_item('edoc_blank', res_edb_id, 'Оригинал', un_id, 0, 1, 1));
        end
      end
      else
      begin
        bls_item = '{"caption": "%1", "blobs":[%2]}';
        if (orig_bls_id is not null) then
        begin
          bls_item = f_format(bls_item, edf_print_name, f_image_item('blobstore', orig_bls_id, 'Оригинал', un_id, 0, 1, 1));
        end
        else
        begin
          bls_item = f_format(bls_item, edf_print_name, f_image_item('edoc_blank', res_edb_id, 'Оригинал', un_id, 0, 1, 1));
        end
      end
      if (blob_list <> '') then
        blob_list = blob_list || ',';

      blob_list = blob_list || bls_item;
    end

    for select c_name_fio, c_blob_pass1_id, c_blob_photo_id
        from edoc_sign, clients
        where eds_from_edl_id = :edl_id and
              eds_from_c_id = c_id and
              (c_blob_pass1_id is not null or
              c_blob_photo_id is not null)
        into c_name_fio, pass_bls_id, photo_bls_id
    do
    begin
      c_name_fio = replace(c_name_fio, '"', '\"');
      if (pass_bls_id is not null and
          photo_bls_id is not null) then
      begin
        bls_item = '{"caption": "%1", "blobs":[%2, %3]}';
        bls_item = f_format(bls_item, c_name_fio, f_image_item('blobstore', pass_bls_id, 'Паспорт', un_id, 0, 1, 1), f_image_item('blobstore', photo_bls_id, 'Фото', un_id, 0, 1, 1));
      end
      if (pass_bls_id is not null and
          photo_bls_id is null) then
      begin
        bls_item = '{"caption": "%1", "blobs":[%2]}';
        bls_item = f_format(bls_item, c_name_fio, f_image_item('blobstore', pass_bls_id, 'Паспорт', un_id, 0, 1, 1));
      end
      if (pass_bls_id is null and
          photo_bls_id is not null) then
      begin
        bls_item = '{"caption": "%1", "blobs":[%2]}';
        bls_item = f_format(bls_item, c_name_fio, f_image_item('blobstore', photo_bls_id, 'Фото', un_id, 0, 1, 1));
      end

      if (blob_list <> '') then
        blob_list = blob_list || ',';

      blob_list = blob_list || bls_item;
    end

    if (f_get_db_id() = 81001) then
    begin
      select edbs.edbs_status, edbs.edbs_status_datetime
      from edoc_blank edb
      inner join edoc_blank_scan edbs on edbs.edbs_from_edb_id = edb.edb_id
      where edb.edb_id = :edb_id
      into :edbs_status, edbs_status_datetime;

      if (edbs_status in (0, 1, -1)) then
      begin
        select cd.cds_doc_blob_1
        from edoc_sign es
        inner join clientdocs cd on cd.cds_from_c_id = es.eds_from_c_id
        where es.eds_from_edl_id = :edl_id and
              cd.cds_from_pk_id = 355 and
              cds_is_delete = 0 and
              cd.cds_verify_status in (-3, 1) and
              current_date between cd.cds_doc_date_begin and cd.cds_doc_date_end
        order by cd.cds_datetime desc
        rows 1
        into :photo_change_bls_id;
      end
      else
      if (edbs_status = 2) then
      begin
        select cd.cds_doc_blob_1
        from edoc_sign es
        inner join clientdocs cd on cd.cds_from_c_id = es.eds_from_c_id
        where es.eds_from_edl_id = :edl_id and
              cd.cds_from_pk_id = 355 and
              cds_is_delete = 0 and
              cd.cds_verify_status in (2) and
              :edbs_status_datetime between cd.cds_doc_date_begin and cd.cds_doc_date_end
        order by cd.cds_datetime desc
        rows 1
        into :photo_change_bls_id;
      end
      if (photo_change_bls_id > 0) then
      begin
      bls_item = '{"caption": "%1", "blobs":[%2]}';
      bls_item = f_format(bls_item, 'Фото для смены номера', f_image_item('blobstore', photo_change_bls_id, 'Фото для смены номера', un_id, 0, 1, 1));
      if (blob_list <> '') then
        blob_list = blob_list || ',';
      blob_list = blob_list || bls_item;
     end
    end
    if (blob_list <> '') then
      blob_list = '[' || blob_list || ']';
  end

  if (blob_list = '') then
    res_text = 'Нет готовых бланков для просмотра. Они должны быть подписаны или отправлены на ручное подписание';

  if (res_text = '') then
    res_is_error = 0;

  if (db_id in (39801, 41701)) then
  begin
    if (not exists(
      select *
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_print_edoc')
      where has_access = 1))
    then blob_list = replace(blob_list, '"can_print":1', '"can_print":0');
  end

  suspend;
end
