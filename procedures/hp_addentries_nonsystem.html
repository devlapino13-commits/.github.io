CREATE PROCEDURE HP_ADDENTRIES_NONSYSTEM(
    OL_ID D_ID,
    REG_ID D_INTEGER,
    CU D_STR_LANG,
    ACC_FULL D_STR_20,
    ACC_SHORT D_STR_20)
AS
declare variable k_id d_id;
declare variable kont_no d_str_20;
declare variable os d_currency;
declare variable perc d_currency;
declare variable other d_currency;
declare variable is_ipf boolean;
declare variable detail d_str_long;
declare variable acc1 d_str_20;
declare variable acc2 d_str_20;
declare variable acc_dt1 d_str_20;
declare variable acc_dt2 d_str_20;
declare variable acc_dt3 d_str_20;
declare variable acc_kt1 d_str_20;
declare variable acc_kt2 d_str_20;
declare variable acc_kt3 d_str_20;
begin
  acc1 = '61099/3';
  acc2 = '61099/10';
  if (acc_short not in (acc1, acc2)) then
    exit;

  select
    k_id,
    k_kontract_no,
    abs(oko_os),
    abs(oko_perc),
    abs(oko_other)
  from operkroff
    inner join kontract
      on k_id = oko_from_k_id
  where oko_from_ol_id = :ol_id
    and oko_ol_is_active = 1
    and oko_type = 4
    and (oko_os + oko_perc + oko_other) <> 0
  into
    :k_id,
    :kont_no,
    :os,
    :perc,
    :other;

  if (k_id > 0) then
  begin
    acc_dt1 = acc_full;
    is_ipf = (acc_short = acc2);

    if (os > 0) then
    begin
      acc_kt1 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
          and al_account = iif(:is_ipf, '80433/1', '80421/1'));
      acc_dt2 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
          and al_account = iif(:is_ipf, '95002/2', '95002/1'));
      acc_kt2 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
          and al_account = iif(:is_ipf, '95001/2', '95001/1'));
      detail = f_format('Погашение сумм на внесистемном учете по ОС №%1', kont_no);
      insert into operbook (
        ob_from_ol_id,
        ob_account_dt,
        ob_account_kt,
        ob_currency,
        ob_curr_rate,
        ob_summa_val,
        ob_summa,
        ob_detail)
      values (
        :ol_id,
        :acc_dt1,
        :acc_kt1,
        :cu,
        1,
        :os,
        :os,
        :detail);
      insert into operbook (
        ob_from_ol_id,
        ob_account_dt,
        ob_account_kt,
        ob_currency,
        ob_curr_rate,
        ob_summa_val,
        ob_summa,
        ob_detail)
      values (
        :ol_id,
        :acc_dt2,
        :acc_kt2,
        :cu,
        1,
        :os,
        :os,
        :detail);
    end
    
    if (perc > 0) then
    begin
      acc_kt1 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
          and al_account = iif(:is_ipf, '80433/2', '80431/2'));
      acc_dt2 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
          and al_account = iif(:is_ipf, '95004/2', '95004/1'));
      acc_kt2 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
        and al_account = iif(:is_ipf, '95003/2', '95003/1'));
      detail = f_format('Погашение сумм на внесистемном учете по % №%1', kont_no);
      insert into operbook (
        ob_from_ol_id,
        ob_account_dt,
        ob_account_kt,
        ob_currency,
        ob_curr_rate,
        ob_summa_val,
        ob_summa,
        ob_detail)
      values (
        :ol_id,
        :acc_dt1,
        :acc_kt1,
        :cu,
        1,
        :perc,
        :perc,
        :detail);
      insert into operbook (
        ob_from_ol_id,
        ob_account_dt,
        ob_account_kt,
        ob_currency,
        ob_curr_rate,
        ob_summa_val,
        ob_summa,
        ob_detail)
      values (
        :ol_id,
        :acc_dt2,
        :acc_kt2,
        :cu,
        1,
        :perc,
        :perc,
        :detail);
    end

    if (other > 0 and is_ipf) then
    begin
      acc_kt1 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
          and al_account = '61102/2');
      acc_dt2 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
          and al_account = '95004/2');
      acc_kt2 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
        and al_account = '95003/2');
      acc_dt3 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
          and al_account = '80611/101');
      acc_kt3 = (
        select al_id from acc_list
        where al_from_reg_id = :reg_id and al_currency = :cu
        and al_account = '21131/21');
      detail = f_format('Погашение сумм на внесистемном учете по % №%1', kont_no);
      insert into operbook (
        ob_from_ol_id,
        ob_account_dt,
        ob_account_kt,
        ob_currency,
        ob_curr_rate,
        ob_summa_val,
        ob_summa,
        ob_detail)
      values (
        :ol_id,
        :acc_dt1,
        :acc_kt1,
        :cu,
        1,
        :other,
        :other,
        :detail);
      insert into operbook (
        ob_from_ol_id,
        ob_account_dt,
        ob_account_kt,
        ob_currency,
        ob_curr_rate,
        ob_summa_val,
        ob_summa,
        ob_detail)
      values (
        :ol_id,
        :acc_dt2,
        :acc_kt2,
        :cu,
        1,
        :other,
        :other,
        :detail);
      detail = f_format('НСП Погашение сумм на внесистемном учете по % №%1', kont_no);
      insert into operbook (
        ob_from_ol_id,
        ob_account_dt,
        ob_account_kt,
        ob_currency,
        ob_curr_rate,
        ob_summa_val,
        ob_summa,
        ob_detail)
      values (
        :ol_id,
        :acc_dt3,
        :acc_kt3,
        :cu,
        1,
        round(:other * 0.02, 2),
        round(:other * 0.02, 2),
        :detail);
    end
  end
end
