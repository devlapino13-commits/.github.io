CREATE PROCEDURE AAREP_OBJECT_CONTINFO_TMP(
    REP_LIST_KIND VARCHAR(100),
    REP_OBJECT_ID DECIMAL(18,0))
RETURNS (
    RES_KIND INTEGER,
    RES_ID DECIMAL(18,0),
    RES_FIELD_1 VARCHAR(255),
    RES_FIELD_2 VARCHAR(255),
    RES_FIELD_3 VARCHAR(255))
AS
begin
  res_field_1 = '';
  res_field_2 = '';
  res_field_3 = '';
  if (rep_list_kind = '0' or rep_list_kind = 'Clients') then
  begin
-- Физ.лицо
    res_kind = 0;
    res_id = rep_object_id;
    suspend;
    exit;
  end
  if (rep_list_kind = '9' or rep_list_kind = 'Kontract') then
  begin
-- Клиенты в кредите
    res_kind = 0;
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :rep_object_id and
              kc_amount_approved > 0
        into :res_id
    do
    begin
      suspend;
    end
  end
  if (rep_list_kind = 'Staff') then
  begin
-- Сотрудники
-- в Rep_object_id передается ID офиса
    res_kind = 0;
    for select ps_from_c_id, reg_code || ' ' || reg_name, jl_name, rd_name
        from region_list(:rep_object_id, 1, 3, 1), hr_pstatus, region, joblist, jobkind, refdata
        where ps_from_reg_id = id_list and
              ps_date_end >= '31.12.2999' and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_reg_id = reg_unicode and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_id = rd_id
        into :res_id, res_field_1, res_field_2, res_field_3
    do
    begin
      suspend;
    end
  end
end
