CREATE PROCEDURE HP_SUD_DO_ACTION(
    ID D_ID,
    KIND D_BOOL,
    ACTION_CODE D_STR_CODE,
    LAST_LOG_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TXT D_STR_255)
AS
declare variable smr_last_sul_id d_integer;
declare variable obj_kind d_integer;
declare variable obj_name d_str_code;
declare variable smm_from_un_id d_id;
begin
  if (id is null) then
    exit;

  res_txt = 'ok';
  -- Выполняет стандартные действия (меню Действие) над судебным запросом
  if (kind = 0) then
  begin
    select smr_last_sul_id, sul_object_kind
    from sud_mainreq, sud_mainlog
    where smr_id = :id
      and smr_last_sul_id = sul_id
    into smr_last_sul_id, obj_kind;
  end else
  if (kind = 1) then
  begin
    select slc_last_sul_id, sul_object_kind
    from sud_maincollect, sud_loanclient, sud_mainlog
    where smc_id = :id
      and slc_id = smc_from_slc_id
      and slc_last_sul_id = sul_id
    into smr_last_sul_id, obj_kind;
  end else
  begin
    suspend;
    exception e_default 'Unknown Kind = ' || coalesce(kind,  '<NULL>');
  end

  if (last_log_id <> 0 and coalesce(smr_last_sul_id, 0) <> last_log_id) then
  begin
    res_txt = f_format('Статус строки изменился, сначала обновите список. Id: %1', id);
    suspend;
    exit;
  end

  obj_name = f_objkind(obj_kind, 'name');

  if (action_code similar to '[-]?[[:digit:]]+') then
  begin
    if (action_code < 0) then
    begin
      select smm_from_un_id
      from sud_mainmemo
      where smm_from_sul_id = :smr_last_sul_id
      order by smm_datetime desc, smm_id desc
      rows 1
      into smm_from_un_id;

      if (coalesce(smm_from_un_id, 0) <> un_id) then
      begin
        res_txt = 'Сначала добавьте комментарий от своего имени.';
        suspend;
        exit;
      end
    end

    if (obj_name = 'sud_mainreq') then
    begin
      update sud_mainreq
      set smr_status = :action_code,
          smr_from_un_id = :un_id
      where smr_id = :id
        and smr_status <> :action_code;
    end else
    if (obj_name = 'sud_maindocs') then
    begin
      update sud_maindocs
      set smd_status = :action_code,
          smd_from_un_id = :un_id
      where smd_id = :id
        and smd_status <> :action_code;
    end else
    if (obj_name = 'sud_mainpay') then
    begin
      update sud_mainpay
      set smp_status = :action_code,
          smp_from_un_id = :un_id
      where smp_id = :id
        and smp_status <> :action_code;
    end else
    if (obj_name = 'sud_mainlist') then
    begin
      update sud_mainlist
      set sml_status = :action_code,
          sml_from_un_id = :un_id
      where sml_id = :id
        and sml_status <> :action_code;
    end else
    if (obj_name = 'sud_maincollect') then
    begin
      update sud_maincollect
      set smc_status = :action_code,
          smc_from_un_id = :un_id
      where smc_id = :id
        and smc_status <> :action_code;
    end else
      exception e_default f_format('Unknown object update: %1', obj_name);

    suspend;
    exit;
  end

  exception e_default f_format('Unknown action. Id: %1, Action: %2', id, action_code);
  suspend;
end
