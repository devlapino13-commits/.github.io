CREATE PROCEDURE HP_WEBCAM_PHOTOVER_RESULT(
    JSON_TXT D_BLOB_TEXT)
RETURNS (
    RESULT_CODE D_INTEGER,
    ERR_DESC D_STR_255,
    ERR_ACTION D_STR_255,
    LIVENESS_RESULT D_INTEGER,
    LIVENESS_DESC D_STR_100,
    LIVENESS_SCORE DECIMAL(6,2),
    DISGUISE_SCORE DECIMAL(6,2),
    SIMILARITY DECIMAL(6,2),
    HINT D_STR_255,
    RESULT_TOTAL D_SMALLINT)
AS
begin
  json_txt = replace(json_txt, '%', '');
  result_code = f_json_str_get_value(json_txt, 'result_code', 0);

  if (result_code = 0) then
  begin
    err_desc = left(f_json_str_get_value(json_txt, 'err_desc', ''), 255);
    err_action = left(f_json_str_get_value(json_txt, 'err_action', ''), 255);
    liveness_result = f_json_str_get_value(json_txt, 'info.liveness_result', -1);
    liveness_desc =
      case liveness_result
        when 0 then 'Лицо подлинное'
        when 1 then 'Лицо спорное, необходимо переснять'
        when 2 then 'Лицо замаскированное, необходимо переснять'
        when 3 then 'Лицо поддельное, попробуйте переснять'
        when 4 then 'Фото содержит несколько лиц, необходимо переснять'
        when 5 then 'Фото слишком маленькое лицо < 64 pics, необходимо переснять'
        when 6 then 'Фото сделано не как селфи, необходимо переснять'
        when -1 then 'Ошибка обработки'
      else '' end;
    liveness_score = f_json_str_get_value(json_txt, 'info.liveness_score', 0);
    disguise_score = f_json_str_get_value(json_txt, 'info.disguise_score', 0);
    similarity = f_json_str_get_value(json_txt, 'info.similarity', 0);
    hint = f_json_str_get_value(json_txt, 'info.hint', '');
    result_total = iif((liveness_result = 0) and (similarity >= 98), 1, 0);
    suspend;
  end

  when any do
  begin
    result_code = 0;
    err_desc = 'Ошибка обработки';
    err_action = 'Повторите через некоторое время';

    liveness_result = result_code;
    liveness_desc = err_desc;
    hint = err_action;

    liveness_score = 0;
    disguise_score = 0;
    similarity = 0;
    hint = err_action;

    result_total = 0;
    suspend;
  end
end
