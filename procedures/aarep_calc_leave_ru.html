CREATE PROCEDURE AAREP_CALC_LEAVE_RU(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_DISB_KC_FIO VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_BALANCE30 DECIMAL(18,4),
    PS_IS_ON_STAFF_STR VARCHAR(100),
    PS_IS_WORKING_STR VARCHAR(100),
    AA_PAR30 DECIMAL(18,4),
    JL_NAME VARCHAR(100))
AS
declare variable ps_is_on_staff integer;
declare variable ps_is_working integer;
declare variable ps_from_reg_id integer;
declare variable kc_prior_id decimal(18,0);
declare variable aa_disb_kc_id decimal(18,0);
begin
  kc_prior_id = 0;
  for select aa_disb_kc_id, sum(aa_amount_active_end), -- Портфель по категории А
             sum(aa_deliq_balance30) -- Просроченный портфель по категории А

      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, null, null, 0, 0, 0, 0, 0, 0)
      where aa_loan_kind_new in ('Категория A', 'Категория B') and
            dd_disb_date >= '01.06.2012'

      group by aa_disb_kc_id
      into :aa_disb_kc_id, :aa_amount_active_end, :aa_deliq_balance30

  do
  begin
    aa_par30 = 0;
    if (:aa_amount_active_end <> 0) then
      aa_par30 = cast(:aa_deliq_balance30 / :aa_amount_active_end * 100 as decimal(15,1));
    if (aa_par30 >= 7) then
    begin
-- Параметры кредитного служащего
      select c_name_fio, jl_name, ps_is_on_staff, ps_is_working, ps_from_reg_id
      from hr_pstatus, joblist, clients
      where ps_from_c_id = :aa_disb_kc_id and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_jl_id = jl_id and
            ps_from_c_id = c_id
      order by ps_is_working desc
      rows 1
      into :aa_disb_kc_fio, :jl_name, :ps_is_on_staff, :ps_is_working, :ps_from_reg_id;

      if (ps_is_on_staff is not null) then
      begin
        if (ps_is_on_staff = 1) then
          ps_is_on_staff_str = 'Штатный';
        else
          ps_is_on_staff_str = 'Внештатный';
        if (ps_is_working = 0) then
        begin
          ps_is_on_staff_str = ps_is_on_staff_str || ', ' || 'уволен';
          ps_is_working_str = 'Уволен';
        end
        else
        if (ps_is_working = 1) then
          ps_is_working_str = 'Работает';
      end
      else
        ps_is_on_staff_str = 'Приказ не оформлен';

      aa_curr_office_name = 'Не указан';
      aa_curr_branch_name = 'Не указан';
      if (ps_from_reg_id is not null) then
      begin
        select aa_curr_office_name, aa_curr_level2_name, aa_curr_level3_name
        from aarep_get_office_categ(:ps_from_reg_id, :rep_date_end)
        into :aa_curr_office_name, :aa_curr_branch_name, aa_curr_level_name;
      end

      if (ps_is_working = 1) then
        suspend;
    end
  end

end
