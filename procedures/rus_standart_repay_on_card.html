CREATE PROCEDURE RUS_STANDART_REPAY_ON_CARD
RETURNS (
    TARGET STR_100,
    OPERATION STR_100,
    ESM_ID ID_TYPE,
    IDCARD VARCHAR(100),
    AMOUNT VARCHAR(100),
    TRANSACTIONID ID_TYPE)
AS
declare variable esm_idcard str_100;
declare variable c_id id_type;
declare variable psi_id id_type;
begin
  target = 'giftdeposit';
  operation = 'repay';

  for select em.esm_id, em.esm_account, em.esm_amount, em.esm_trans_id, em.esm_object_id
      from extsysmain em
      where em.esm_project = 'rus_standart' and
            em.esm_action = 'CardRepay' and
            em.esm_status = 0
      into :esm_id, :esm_idcard, :amount, :transactionid, :c_id
  do
  begin
    for select psi_id
        from hp_paysys_get_cl_info(:c_id, 6) -- 6 - Русский Стандарт
        order by psm_date desc
        rows 1
        into :psi_id
    do
    begin
      select pys.psi_code
      from paysys_item pys
      where pys.psi_id = :psi_id
      into :idcard;
    end

    if (esm_idcard = idcard) then
      suspend;
  end
end
