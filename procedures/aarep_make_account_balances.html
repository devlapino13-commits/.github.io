CREATE PROCEDURE AAREP_MAKE_ACCOUNT_BALANCES(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    POB_AMOUNT DECIMAL(18,4),
    POB_REGION_NAME VARCHAR(100),
    POB_REGION_ID INTEGER,
    POB_BRANCH_ID INTEGER,
    POB_AMOUNT_BALANCE DECIMAL(18,4),
    POB_AMOUNT_DIFF DECIMAL(18,4),
    POB_ACC_DT VARCHAR(20),
    POB_ACC_KT VARCHAR(20),
    POB_BOOK_MEMO VARCHAR(1024))
AS
declare variable report_period integer;
begin
  report_period = extract(year from :report_date_end) || extract(month from :report_date_end);
-- Остаток портфеля
  pob_book_memo = 'Переклассификация сальдо счета 10931 и 10941';
  for with tabl1
      as (select al_from_reg_id, al_currency, al_from_bg_id, al_account, sum(oks_balance) oks_balance,
                 iif(pr_loan_type = 0, 'Групп.', 'Индив') loan_type,
                 case pr_product_kind
                   when 1 then 'Групповой'
                   when 2 then 'Параллельный'
                   when 3 then 'Индивидуальный'
                   when 4 then 'Для сотрудников'
                 end product_kind_str,
                 pr_product_kind
          from oksummary, kontract, standacc, acc_list, product
          where :report_date_end between oks_date_begin and oks_date_end and
                oks_balance > 0 and
                oks_from_k_id = k_id and
                k_from_pr_id = sa_from_pr_id and
                sa_account_type = 1 and
                sa_account = al_id and
                sa_from_reg_id = oks_region_id and
                k_from_pr_id = pr_id and
                oks_close_state = 0
          group by al_from_reg_id, al_currency, al_from_bg_id, al_account, al_name, loan_type, product_kind_str, pr_product_kind),
      tabl2
      as (select al_from_reg_id, al_currency, al_account, oks_balance, loan_type, product_kind_str,
                 iif(pr_product_kind in (1, 2), '10931/1', iif(pr_product_kind = 3 or (pr_product_kind = 4 and al_currency = '417'), '10941/1', iif(pr_product_kind = 4 and al_currency <> '417', '10941/10', ''))) new_account
          from tabl1)
      select al_from_reg_id, reg_code || ' ' || reg_name,
             (select al.al_id new_al_id
              from acc_list al
              where al.al_account = t2.new_account and
                    al.al_currency = t2.al_currency and
                    al.al_from_reg_id = t2.al_from_reg_id) account_dt,
             (select al.al_id new_al_id
              from acc_list al
              where al.al_account = t2.al_account and
                    al.al_currency = t2.al_currency and
                    al.al_from_reg_id = t2.al_from_reg_id) account_kt, oks_balance
      from tabl2 t2, region
      where t2.al_account <> t2.new_account and
            al_from_reg_id = reg_unicode
      into pob_region_id, pob_region_name, pob_acc_dt, pob_acc_kt, pob_amount
  do
  begin
    pob_branch_id = pob_region_id;
    pob_amount_balance = 0;
    pob_amount_diff = pob_amount - pob_amount_balance;
    suspend;
  end

-- Накопленные проценты
  pob_book_memo = 'Переклассификация сальдо счета 11437';
  for with tabl1
      as (select al_from_reg_id, al_currency, al_from_bg_id, al_account, sum(kcp_perc_saldo) kcp_perc_saldo,
                 iif(pr_loan_type = 0, 'Групп.', 'Индив') loan_type,
                 case pr_product_kind
                   when 1 then 'Групповой'
                   when 2 then 'Параллельный'
                   when 3 then 'Индивидуальный'
                   when 4 then 'Для сотрудников'
                 end product_kind_str
--       ,pr_name
                 ,
                 pr_product_kind
          from kontcalcperc, kontract, standacc, acc_list, product
          where kcp_period = :report_period and
                kcp_perc_saldo > 0 and
                kcp_from_k_id = k_id and
                k_from_pr_id = sa_from_pr_id and
                sa_account_type = 2 and
                sa_account = al_id and
                sa_from_reg_id = kcp_from_reg_id and
                k_from_pr_id = pr_id
          group by al_from_reg_id, al_currency, al_from_bg_id, al_account, al_name, loan_type, product_kind_str
--         ,pr_name
              , pr_product_kind),
      tabl2
      as (select al_from_reg_id, al_currency, al_account, kcp_perc_saldo, loan_type, product_kind_str,
                 iif(pr_product_kind in (1, 2), '11437/1', iif(pr_product_kind = 3 or (pr_product_kind = 4 and al_currency = '417'), '11437/2', iif(pr_product_kind = 4 and al_currency <> '417', '11437/10', ''))) new_account
          from tabl1)
      select al_from_reg_id, reg_code || ' ' || reg_name,
             (select al.al_id new_al_id
              from acc_list al
              where al.al_account = t2.new_account and
                    al.al_currency = t2.al_currency and
                    al.al_from_reg_id = t2.al_from_reg_id) account_dt,
             (select al.al_id new_al_id
              from acc_list al
              where al.al_account = t2.al_account and
                    al.al_currency = t2.al_currency and
                    al.al_from_reg_id = t2.al_from_reg_id) account_kt, kcp_perc_saldo
      from tabl2 t2, region
      where t2.al_account <> t2.new_account and
            al_from_reg_id = reg_unicode
      into pob_region_id, pob_region_name, pob_acc_dt, pob_acc_kt, pob_amount
  do
  begin
    pob_branch_id = pob_region_id;
    pob_amount_balance = 0;
    pob_amount_diff = pob_amount - pob_amount_balance;
    suspend;
  end
--  Полученный ДБП
  pob_book_memo = 'Переклассификация сальдо счета 21141';
  for with tabl1
      as (select al_from_reg_id, al_currency, al_from_bg_id, al_account, sum(kcp_kommis_saldo) kcp_kommis_saldo,
                 iif(pr_loan_type = 0, 'Групп.', 'Индив') loan_type,
                 case pr_product_kind
                   when 1 then 'Групповой'
                   when 2 then 'Параллельный'
                   when 3 then 'Индивидуальный'
                   when 4 then 'Для сотрудников'
                 end product_kind_str
--       ,pr_name
                 ,
                 pr_product_kind
          from kontcalcperc, kontract, standacc, acc_list, product
          where kcp_period = :report_period and
                kcp_kommis_saldo <> 0 and
                kcp_from_k_id = k_id and
                k_from_pr_id = sa_from_pr_id and
                sa_account_type = 7 and
                sa_account = al_id and
                sa_from_reg_id = kcp_from_reg_id and
                k_from_pr_id = pr_id
          group by al_from_reg_id, al_currency, al_from_bg_id, al_account, al_name, loan_type, product_kind_str
--         ,pr_name
              , pr_product_kind),
      tabl2
      as (select al_from_reg_id, al_currency, al_account, kcp_kommis_saldo, loan_type, product_kind_str,
                 iif(pr_product_kind in (1, 2), '21141/101', iif(pr_product_kind = 3 or (pr_product_kind = 4 and al_currency = '417'), '21141/102', iif(pr_product_kind = 4 and al_currency <> '417', '21141/102', ''))) new_account
          from tabl1)
      select al_from_reg_id, reg_code || ' ' || reg_name,
             (select al.al_id new_al_id
              from acc_list al
              where al.al_account = t2.new_account and
                    al.al_currency = t2.al_currency and
                    al.al_from_reg_id = t2.al_from_reg_id) account_dt,
             (select al.al_id new_al_id
              from acc_list al
              where al.al_account = t2.al_account and
                    al.al_currency = t2.al_currency and
                    al.al_from_reg_id = t2.al_from_reg_id) account_kt, kcp_kommis_saldo
      from tabl2 t2, region
      where t2.al_account <> t2.new_account and
            al_from_reg_id = reg_unicode
      into pob_region_id, pob_region_name, pob_acc_dt, pob_acc_kt, pob_amount
  do
  begin
    pob_branch_id = pob_region_id;
    pob_amount_balance = 0;
    pob_amount_diff = pob_amount - pob_amount_balance;
    suspend;
  end
end
