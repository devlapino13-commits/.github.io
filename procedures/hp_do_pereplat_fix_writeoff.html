CREATE PROCEDURE HP_DO_PEREPLAT_FIX_WRITEOFF(
    K_ID ID_TYPE,
    AMOUNT CURR_TYPE)
AS
declare variable ol_date date;
declare variable tmp_amount curr_type;
declare variable kont_str d_str_code;
declare variable reg_id integer;
declare variable pereplat_ol_id id_type;
declare variable acc_income d_str_code;
declare variable acc_pereplat d_str_code;
declare variable k_kontract_no d_str_code;
declare variable ol_id id_type;
begin
  ol_date = '31.12.2017';

  select k_kontract_no, k_kontract_no || ' от ' || f_datetostr(k_kontract_date)
  from kontract
  where k_id = :k_id
  into k_kontract_no, :kont_str;

  if (kont_str is null) then
    exception e_default 'Kontract not found: ' || coalesce(k_id, '<null>');

  select sum(-ok_os)
  from operkred
  where ok_ol_is_active = 1 and
        ok_ol_date <= :ol_date and
        ok_from_k_id = :k_id and
        ok_type = 9
  into tmp_amount;
  if (tmp_amount is null) then
    tmp_amount = 0;
  if (tmp_amount <> amount) then
    exception e_default 'Different input amount: ' || amount || ', saldo amount: ' || tmp_amount;

  select ok_from_ol_id
  from operkred
  where ok_from_k_id = :k_id and
        ok_type = 9 and
        ok_os < 0 and
        ok_ol_is_active = 1 and
        ok_ol_date < :ol_date
  order by ok_ol_date desc, ok_from_ol_id desc
  rows 1
  into :pereplat_ol_id;

  if (k_id <> 101010004981200101) then
  begin
    select ob_ol_reg_id, ob_account_kt
    from operbook
    where ob_from_ol_id = :pereplat_ol_id and
          (ob_account_kt containing '76-06' or
          ob_account_kt containing '76-04') rows 1
    into :reg_id, :acc_pereplat;

    if (acc_pereplat is null) then
      exception e_default 'No pereplat operbook in ol_id: ' || pereplat_ol_id;
  end

  insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_user_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified,
                        ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype)
  values (:reg_id, 2, :ol_date, 0, '', 9,
          'Признание доходом суммы переплаты по договору ' || :kont_str || 'по истечению срока',
          1, 0, 1000000000001, null, null, 0, :reg_id, null, null, 11)
  returning ol_id
  into :ol_id;

  if (k_id <> 101010004981200101) then
  begin
    acc_income = left(acc_pereplat, 7);
    acc_income = acc_income || '91-01000004';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_account_dt, ob_summa_val, ob_summa,
                          ob_account_kt, ob_detail)
    values (:ol_id, 1, '810', 1, :acc_pereplat, :amount, :amount, :acc_income, 'Переплата на доход');
  end

  execute procedure hp_do_pereplat_ol_id_link(:ol_id, :k_kontract_no, 1, '00-00', :amount);
end
