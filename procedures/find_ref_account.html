CREATE PROCEDURE FIND_REF_ACCOUNT(
    REG_ID INTEGER,
    FIND_AB_CODE D_STR_CODE,
    FIND_TEXT D_STR_255)
RETURNS (
    ACC_ID ID_TYPE,
    ACC_CODE D_STR_CODE,
    ACC_DOT_CODE D_STR_CODE,
    ACC_NAME D_STR_255,
    ACC_AB_ID ID_TYPE,
    ACC_AB_NAME D_STR_LONG,
    ACC_REG_NAME D_STR_255,
    ACC_MEMO D_STR_2048)
AS
declare variable FIND_AB_ID ID_TYPE;
declare variable INX_ID ID_TYPE;
declare variable ACC_OBJECT_KIND integer;
declare variable ACC_OBJECT_ID D_STR_CODE;
declare variable ACC_NUMBER D_STR_CODE;
declare variable ACC_MEMO_TMP D_STR_LONG;
declare variable RES_COUNT integer;
declare variable MAX_RES_COUNT integer;
declare variable NEED_SUSPEND integer;
declare variable INX_NAME D_STR_2048;
declare variable LOAN_NAME D_STR_2048;
begin
  if (reg_id is null) then
    reg_id = 0;
  if (find_ab_code is null) then
    find_ab_code = '';
  if (find_text is null) then
    find_text = '';
  find_ab_code = trim(find_ab_code);
  find_text = trim(find_text);

  if (find_ab_code = '' and
      find_text = '') then
    exit;

  res_count = 0;
  max_res_count = 500;

  if (find_ab_code <> '') then
  begin
    select ab_id
    from ref_accbalance
    where ab_code = :find_ab_code
    into :find_ab_id;

    for select ab_id, ab_code || ' ' || ab_name
        from ref_accbalance
        where (ab_id = :find_ab_id or
              ab_parent_id = :find_ab_id)
        into :acc_ab_id, :acc_ab_name
    do
    begin
      for select acc_id, acc_code, acc_name, ab_code || ' ' || ab_name, reg_code || ' ' || reg_name, acc_from_inx_id,
                 acc_object_kind, acc_object_id, acc_number, acc_memo
          from ref_account
          inner join ref_accbalance on (acc_from_ab_id = ab_id)
          left join region on (acc_from_reg_id = reg_unicode)
          left join ref_income_expense on (acc_from_inx_id = inx_id)
          where (:reg_id = 0 or
                acc_from_reg_id = :reg_id) and
                acc_from_ab_id = :acc_ab_id
          into acc_id, acc_code, acc_name, acc_ab_name, acc_reg_name, inx_id, acc_object_kind, acc_object_id,
               acc_number, acc_memo_tmp
      do
      begin
        acc_dot_code = acc_code;

        if (find_text = '') then
          need_suspend = 1;
        else
          need_suspend = 0;

        inx_name = '';
        loan_name = '';
        if (inx_id is not null) then
        begin
          select main.inx_code || ' ' || parent.inx_name || ', ' || main.inx_name
          from ref_income_expense main, ref_income_expense parent
          where main.inx_parent_id = parent.inx_id and
                main.inx_id = :inx_id
          into :inx_name;
        end

        if (acc_object_kind is not null and
            coalesce(acc_object_id, 0) <> 0) then
        begin
          if (acc_object_kind = 0) then
          begin
            select c_name_fio
            from clients
            where c_id = :acc_object_id
            into :loan_name;
          end
          if (acc_object_kind = 8) then
          begin
            select k_kontract_no || ', ' || c_name_fio
            from knclient, kontract, clients
            where kc_id = :acc_object_id and
                  kc_from_c_id = c_id and
                  kc_from_k_id = k_id
            into :loan_name;
          end
        end

        acc_memo = '';
        if (trim(coalesce(acc_memo_tmp, '')) <> '') then
          acc_memo = acc_memo_tmp;
        if (trim(coalesce(inx_name, '')) <> '') then
          acc_memo = inx_name;
        if (trim(coalesce(loan_name, '')) <> '') then
          acc_memo = loan_name;

        if (need_suspend = 0 and
            acc_code containing find_text) then
          need_suspend = 1;
        if (need_suspend = 0 and
            acc_name containing find_text) then
          need_suspend = 1;
        if (need_suspend = 0 and
            acc_memo_tmp containing find_text) then
          need_suspend = 1;
        if (need_suspend = 0 and
            acc_memo containing find_text) then
          need_suspend = 1;
        if (need_suspend = 0 and
            acc_reg_name containing find_text) then
          need_suspend = 1;
        if (need_suspend = 0 and
            acc_number = find_text) then
          need_suspend = 1;
        if (need_suspend = 0 and
            acc_object_id = find_text) then
          need_suspend = 1;

        if (need_suspend = 1) then
        begin
          res_count = res_count + 1;

          acc_dot_code = f_fmt_acc(acc_id);

          suspend;

          if (res_count >= max_res_count) then
            exit;
        end
      end
    end
  end
  else
  begin
    if (find_text similar to '[-]?[[:DIGIT:]]+') then
    begin
      if (char_length(:find_text) <= 10) then
      begin
        for select acc_id, acc_code, acc_name, ab_code || ' ' || ab_name, reg_code || ' ' || reg_name, acc_from_inx_id,
                   acc_object_kind, acc_object_id, acc_number, acc_memo
            from ref_account
            inner join ref_accbalance on (acc_from_ab_id = ab_id)
            left join region on (acc_from_reg_id = reg_unicode)
            left join ref_income_expense on (acc_from_inx_id = inx_id)
            where (:reg_id = 0 or
                  acc_from_reg_id = :reg_id) and
                  (acc_number = :find_text or
                  (acc_object_kind in (0, 8) and
                  acc_object_id = :find_text))
            into acc_id, acc_code, acc_name, acc_ab_name, acc_reg_name, inx_id, acc_object_kind, acc_object_id,
                 acc_number, acc_memo_tmp
        do
        begin
          need_suspend = 1;
          acc_dot_code = acc_code;

          inx_name = '';
          loan_name = '';
          if (inx_id is not null) then
          begin
            select main.inx_code || ' ' || parent.inx_name || ', ' || main.inx_name
            from ref_income_expense main, ref_income_expense parent
            where main.inx_parent_id = parent.inx_id and
                  main.inx_id = :inx_id
            into :inx_name;
          end

          if (acc_object_kind is not null and
              coalesce(acc_object_id, 0) <> 0) then
          begin
            if (acc_object_kind = 0) then
            begin
              select c_name_fio
              from clients
              where c_id = :acc_object_id
              into :loan_name;
            end
            if (acc_object_kind = 8) then
            begin
              select k_kontract_no || ', ' || c_name_fio
              from knclient, kontract, clients
              where kc_id = :acc_object_id and
                    kc_from_c_id = c_id and
                    kc_from_k_id = k_id
              into :loan_name;
            end
          end

          acc_memo = '';
          if (trim(coalesce(acc_memo_tmp, '')) <> '') then
            acc_memo = acc_memo_tmp;
          if (trim(coalesce(inx_name, '')) <> '') then
            acc_memo = inx_name;
          if (trim(coalesce(loan_name, '')) <> '') then
            acc_memo = loan_name;

          res_count = res_count + 1;

          acc_dot_code = f_fmt_acc(acc_id);

          suspend;

          if (res_count >= max_res_count) then
            exit;
        end
      end
      else
      begin
        for select acc_id, acc_code, acc_name, ab_code || ' ' || ab_name, reg_code || ' ' || reg_name, acc_from_inx_id,
                   acc_object_kind, acc_object_id, acc_number, acc_memo
            from ref_account
            inner join ref_accbalance on (acc_from_ab_id = ab_id)
            left join region on (acc_from_reg_id = reg_unicode)
            left join ref_income_expense on (acc_from_inx_id = inx_id)
            where acc_code = :find_text
            into acc_id, acc_code, acc_name, acc_ab_name, acc_reg_name, inx_id, acc_object_kind, acc_object_id,
                 acc_number, acc_memo_tmp
        do
        begin
          need_suspend = 1;
          acc_dot_code = acc_code;

          inx_name = '';
          loan_name = '';
          if (inx_id is not null) then
          begin
            select main.inx_code || ' ' || parent.inx_name || ', ' || main.inx_name
            from ref_income_expense main, ref_income_expense parent
            where main.inx_parent_id = parent.inx_id and
                  main.inx_id = :inx_id
            into :inx_name;
          end

          if (acc_object_kind is not null and
              coalesce(acc_object_id, 0) <> 0) then
          begin
            if (acc_object_kind = 0) then
            begin
              select c_name_fio
              from clients
              where c_id = :acc_object_id
              into :loan_name;
            end
            if (acc_object_kind = 8) then
            begin
              select k_kontract_no || ', ' || c_name_fio
              from knclient, kontract, clients
              where kc_id = :acc_object_id and
                    kc_from_c_id = c_id and
                    kc_from_k_id = k_id
              into :loan_name;
            end
          end

          acc_memo = '';
          if (trim(coalesce(acc_memo_tmp, '')) <> '') then
            acc_memo = acc_memo_tmp;
          if (trim(coalesce(inx_name, '')) <> '') then
            acc_memo = inx_name;
          if (trim(coalesce(loan_name, '')) <> '') then
            acc_memo = loan_name;

          res_count = res_count + 1;

          acc_dot_code = f_fmt_acc(acc_id);

          suspend;

          if (res_count >= max_res_count) then
            exit;
        end
      end

    end
  end
end
