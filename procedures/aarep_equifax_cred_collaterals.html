CREATE PROCEDURE AAREP_EQUIFAX_CRED_COLLATERALS(
    JSA D_INTEGER)
RETURNS (
    COLLATERAL_ID D_STR_255,
    ITEM_TYPE_STR D_STR_255,
    COLLATERAL_SUM D_CURRENCY,
    COLLATERAL_DATE D_DATE,
    DATE_ASSESSMENT D_DATE,
    FACT_END_DATE D_DATE)
AS
declare variable i d_integer;
  declare variable cnt d_integer;
  declare variable jso_collateral d_integer;

declare function getcatval(kind d_str_code, code d_str_code)
  returns d_str_255
  as
  begin
    return coalesce((select bct_name from bki_catalog where bct_kind = :kind and bct_code = :code), 'Неопределен');
  end

declare function my_strtonum (
    num_str d_str_code)
  returns d_currency
  as
  begin
    return replace(num_str, ',', '.');
    when any do
      return 0;
  end

declare function my_strtodate (
    date_str d_str_code)
  returns d_date
  as
  begin
    return date_str;
    when any do
      return null;
  end

begin
  if (coalesce(jsa, 0) = 0) then
    exit;

  i = 0;
  cnt = json.count_get(jsa);

  while (i < cnt) do
  begin
    jso_collateral = coalesce(json.arr_get_obj(jsa, i), json.obj_create());

    collateral_id = json.obj_get_val(jso_collateral, 'id');
    item_type_str = getcatval('bki_collateral_item_type', json.obj_get_val(jso_collateral, 'itemType'));
    collateral_sum = my_strtonum(json.obj_get_val(jso_collateral, 'sum', 0));
    collateral_date = my_strtodate(json.obj_get_val(jso_collateral, 'date'));
    date_assessment = my_strtodate(json.obj_get_val(jso_collateral, 'dateAssessment'));
    fact_end_date = my_strtodate(json.obj_get_val(jso_collateral, 'factEndDate'));

    suspend;
    i = i + 1;
  end 
end
