CREATE PROCEDURE FM_ASP_SEND_SMS(
    UN_ID D_ID,
    C_ID D_ID,
    SIGNER_C_ID D_STR_LONG)
RETURNS (
    RES_TEXT D_BLOB_TEXT,
    NEED_REFRESH D_SMALLINT)
AS
declare variable db_id integer;
declare variable tmp_int integer;
declare variable lang d_str_lang;
declare variable tmp_id id_type;
declare variable adl_address_str varchar(255);
declare variable c_pass_date_expire date;
declare variable hr_list d_str_long;
declare variable employee_region d_integer;
declare variable has_access d_bool;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable edc_id d_integer;
declare variable msg_kind d_smallint;
begin
    db_id = f_get_db_id();
    lang = f_get_lang_code(un_id, null);
    need_refresh = 0;
    res_text = '';

    if (db_id in (39801, 41701)) then
    begin
       -- Проверка срока действия паспорта
      c_pass_date_expire = null;
      select cl.c_pass_date_expire
      from clients cl
      where cl.c_id = :c_id
      into  c_pass_date_expire;

      if (res_text = '' and c_pass_date_expire < current_date) then
        res_text = f_lang(lang, 'Паспорт просрочен, необходима замена паспорта');

      -- Проверка наличия адреса
      if (res_text = '') then
      begin
        adl_address_str = null;
        select adl_address_str
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_from_c_id = :c_id
        and adc.adc_address_type = 1
        into adl_address_str;
    
        if (coalesce(adl_address_str, 'Не указан') = 'Не указан') then
          res_text = f_lang(lang, 'Юр.адрес не заполнен, сначала заполните адрес');
      end

      if (res_text = '') then
      begin
        -- Для отправки договора требуется верификация
        tmp_int = 0;
        select c_verify_status from clients
        where c_id = :c_id
        into tmp_int;
        if (tmp_int not in (2, -1)) then
          res_text = f_lang(lang, 'Клиент должен быть верифицирован');
      end

      -- Указание подписанта обязательно
      if (res_text = '' and not exists(select * from clients
      where c_id = :signer_c_id)) then
        res_text = f_lang(lang, 'Подписант должен быть указан');

      -- Блок на отправку кода сотрудникам
      if (res_text = '' and
          exists(select *
                 from hr_pstatus ps
                 where ps.ps_from_c_id = :c_id and
                       ps.ps_is_working = 1 and
                       ps.ps_status = 1 and
                       current_date between ps.ps_date_begin and ps.ps_date_end)) then
      begin
        select distinct list(un.un_id)
        from jobkind jk
        inner join joblist jl on jl.jl_kind = jk.jk_id
        inner join hr_pstatus ps on ps.ps_from_jl_id = jl.jl_id and
                            ps.ps_is_working = 1 and
                            ps.ps_status = 1 and
                            current_date between ps.ps_date_begin and ps.ps_date_end
        inner join username un on un.un_from_c_id = ps.ps_from_c_id
        where jk.jk_system_code = 'JobSystem.HR_Department'
        into hr_list;

        if (hr_list not containing un_id) then
          res_text = 'Отправить код сотрудникам могут только отдел кадров';
      end

      -- Блок на отравку кода кадровикам сотрудникам не вверенного сотрудника
      if (res_text = '' and
          exists(select *
                 from hr_pstatus ps
                 where ps.ps_from_c_id = :c_id and
                       ps.ps_is_working = 1 and
                       ps.ps_status = 1 and
                       current_date between ps.ps_date_begin and ps.ps_date_end)) then
      begin
        select ps.ps_from_reg_id
        from hr_pstatus ps
        where ps.ps_from_c_id = :c_id and
              ps.ps_is_working = 1 and
              ps.ps_status = 1 and
              current_date between ps.ps_date_begin and ps.ps_date_end
        into :employee_region;

        select iif(id_list is not null, 1, 0)
        from permissions pm
        inner join region_list(pm.pm_subject_id, 1,3,1) on id_list = :employee_region
        where current_date between pm.pm_date_begin and pm.pm_date_end and
                                   pm.pm_object_id = :un_id and
                                   pm.pm_is_delete = 0 and
                                   pm.pm_subject_kind = 4
        into :has_access;

        if (has_access is null or has_access = 0) then
          res_text = 'Отправить смс можно только сотрудникам ввереного региона';
      end 
    end

  if (res_text = '') then
  begin
    select edl_id, res_text
    from hp_edoc_ins_asp_agreement(:c_id, :un_id)
    into tmp_id, res_text;

    if (res_text = '') then
    begin
      if (db_id in (39801, 41701)) then
      begin
        -- В КГ указываем подписанта, выбранного пользователем
        update edoc_list
        set edl_manager_c_id = :signer_c_id
        where edl_id = :tmp_id;
      end
  
      need_refresh = 1;
      res_text = f_lang(lang, 'СМС успешно отправлено');

      select edl.edl_object_kind, edl.edl_object_id, edl.edl_from_edc_id
      from edoc_list edl
      where edl.edl_id = :tmp_id
      into obj_kind, obj_id, edc_id;

      select msg_kind
      from hp_edoc_get_msg_kind(:obj_kind, :obj_id, :edc_id)
      into msg_kind;

      if (msg_kind = 4) then
      begin
        res_text = 'Документ успешно сформирован';
      end
    end
  end
  suspend;
end
