declare function need_exclude(kc_id d_id, uid d_str_code)
returns boolean
as
begin
return exists(select 1
from knclient, kontract, kontrefinance
where kc_id = :kc_id
and k_id = kc_from_k_id
and kr_from_ma_id = k_from_ma_app_id
and kr_loan_uuid = :uid);
end

begin

  --rep_period = extract(year from cast(:rep_date_end as date)) || right('0' || extract(month from cast(:rep_date_end as date)), 2);
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  --k_id_prior = 0;
  k_base_k_id_old = 0;
  pr_id_old = 0;
  --lk_deliq_days_old = 0;

  select sab_id
  from sys_accbalance
  --  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  where sab_full_code = 'Микрозаймы.Внебаланс.Поручительство'
  into sab_id;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id_zalog_cost;

  reg_id_prior = 0;
  reg_name = '';
  for select oks_from_k_id, oks_balance, :rep_date_end - oks_delinq_date, k_kontract_no, k_amount_approved, k_disb_date,
             k_kolich_client_approved, coalesce(k_is_devided, 0), k_base_k_id, k_from_pr_id, k_stop_perc,
             cast(kont.k_percent_effect_fifd as decimal(15,3)) as cred_full_cost, k_fact_disb_online, oks_region_id,
             oks_delinq_date
      from oksummary oks, kontract kont
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks.oks_from_k_id = kont.k_id and
            oks_close_state = 0 --
            --and k_kontract_no in ('0119-740909-06')
            --and oks_from_k_id=101010123233800301
      order by k_base_k_id, k_id
          -- rows 100
      into k_id, oks_balance, delinq_day_tmp, k_kontract_no, k_amount_approved, k_disb_date, k_kolich_client_approved,
           k_is_devided, k_base_k_id, pr_id, k_stop_perc, cred_full_cost, k_fact_disb_online, reg_id, oks_delinq_date
  do
  begin

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end

    u_cli = '';
    select sbl_name
    from sb_akt, sb_list
    where sba_from_k_id = :k_id and
          :rep_date_end between sba_date_begin and sba_date_end and
          sba_status = 2 and
          sba_is_delete = 0 and
          sba_from_sbl_id = sbl_id rows 1
    into u_cli;
    if (u_cli <> 'ук') then
      u_cli = '';

    kl_date = null;
    select max(kl_datetime)
    from kontlog
    where kl_from_k_id = :k_id and
          kl_state = 1 and
          cast(kl_datetime as date) <= :k_disb_date
    into kl_date;

    z_cost_market = 0;
    z_cost_likvid = 0;

    select sum(coalesce(zal.z_cost_market, 0))
    from zalog zal
    where zal.z_from_k_id = :k_id and
          :rep_date_end between coalesce(z_date_begin, '01.01.1990') and coalesce(z_date_end, '31.12.9999')
    into :z_cost_market;

    z_cost_market = coalesce(z_cost_market, 0);

    lk_balance = null;
    lk_calcperc = null;
    lk_percent = null;

    lk_percent_tmp = null;
    lk_deliq_days_tmp = null;

    if (pr_id <> pr_id_old) then
    begin
      select pr_code, pr_loan_type, pr_product_kind, pr_name
      from product
      where pr_id = :pr_id
      into pr_code, pr_loan_type, pr_product_kind, pr_name;

      pr_id_old = pr_id;
    end

    if (pr_code containing '.selfemployed.') then
    begin
      client_type = 'СЗ';
    end
    else
    begin
      select kc_client_type, coalesce(kc_is_msp, 0)
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 rows 1
      into kc_client_type, kc_is_msp;

      if (kc_client_type = 1) then
      begin
        client_type = 'ЮЛ';
      end
      else
      if (kc_client_type = 2) then
      begin
        if (kc_is_msp = 1) then
          client_type = 'ИП-МСП';
        else
          client_type = 'ИП-неМСП';
      end
      else
      begin
        client_type = 'ФЛ';
      end
    end

    if (delinq_day_tmp < 0) then
      delinq_day_tmp = 0;

    if (k_is_devided = 1) then
    begin
      if (k_base_k_id <> k_base_k_id_old) then
      begin
        delinq_day = 0;
        is_restruct = 0;
        restruct_date_last = null;
        delinq_day_rest_date = null;
        is_secured = 0;

        lk_deliq_days = null;

        select res_deliq_days, res_is_restruct, res_sh_close_date, restruct_date_last, is_secured, delinq_day_rest_date,
               lk_deliq_days
        from hp_devided_kont_result(:rep_date_end, :k_base_k_id)
        into :delinq_day, :is_restruct, max_sh_date, restruct_date_last, is_secured, delinq_day_rest_date,
             lk_deliq_days;

        k_base_k_id_old = k_base_k_id;
      end
      if (k_id <> k_base_k_id) then
      begin
        select kont.k_kontract_no
        from kontract kont
        where kont.k_id = :k_base_k_id rows 1
        into :k_kontract_no;
      end
    end
    else
    begin
      delinq_day = delinq_day_tmp;
      lk_deliq_days = null;

      select is_restruct, restruct_date_last
      from aarep_get_restructuring(:k_id, :rep_date_end, 81001)
      into :is_restruct, :restruct_date_last;

      max_sh_date = null;

      select max(sh_date)
      from schedule
      where sh_from_k_id = :k_id
      into max_sh_date;

      is_secured = null;
      select is_secured
      from aarep_get_secured_loan(:pr_code, :pr_loan_type, :pr_product_kind, :pr_id, :k_disb_date, :k_kolich_client_approved, :k_id)
      into is_secured;

      delinq_day_rest_date = null;
      if (is_restruct = 1) then
      begin
        restruct_date_last = restruct_date_last - 1;
        select :restruct_date_last - oks_delinq_date
        from oksummary
        where oks_from_k_id = :k_id and
              :restruct_date_last between oks_date_begin and oks_date_end and
              oks_close_state = 0 rows 1
        into delinq_day_rest_date;
