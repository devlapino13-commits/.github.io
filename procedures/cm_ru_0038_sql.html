CREATE PROCEDURE CM_RU_0038_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable kc_from_c_id id_type;
declare variable acl_id id_type;
declare variable acl_status integer;
declare variable acl_datetime timestamp;
declare variable pr_code varchar(100);
begin
if (k_id = 101010023980900901) then exit;
  select k_kontract_no, k_disb_date, k_amount_approved, pr_code 
  from kontract, product
  where k_id=:k_id
  and k_from_pr_id = pr_id
  into :res_info_text, kf_disb_date, kc_amount, pr_code;
  if (pr_code not containing '.EMP.') then  
  begin
    -- Отказан  -2
    -- На доработку  -1
    -- В ожидании  0
    -- Принят на обработку  1
    -- Подтвежден  2
    
    select ACL_ID, ACL_STATUS, ACL_DATETIME
    from HP_ACTIONLIST_KONTRACT_GET(:K_ID)
    into :ACL_ID, :ACL_STATUS, ACL_DATETIME;
    
    KF_CLIENT_PASS_NO='';
    
    res_error_text='';
    res_error_is_critical=0;
    
    if (acl_status=-3) then
    begin
      res_error_text='Данные еще не отправлены на КК';
      res_error_is_critical=2;
      Suspend;
    end
    if (acl_status=-2) then
    begin
      res_error_text='Отказано кредитным комитетом';
      res_error_is_critical=2;
      Suspend;
    end
    if (acl_status=-1) then
    begin
      res_error_text='Кредитным комитетом отправлено на доработку';
      res_error_is_critical=2;
      Suspend;
    end
    if (acl_status=0 or acl_status=1) then
    begin
      res_error_text=' Заявка находится на рассмотрении кредитного комитета…';
      res_error_is_critical=2;
      Suspend;
    end
  end
end
