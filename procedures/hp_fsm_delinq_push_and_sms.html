CREATE PROCEDURE HP_FSM_DELINQ_PUSH_AND_SMS
AS
declare variable k_id d_id;
declare variable edl_id d_id;
declare variable res_text d_str_255;
begin
    if (f_get_db_id() <> 39801) then
    exit;

    for select k.k_id
    from oksummary oks
    join kontract k on oks.oks_from_k_id = k.k_id
    where oks.oks_close_state = 0
          and current_date between oks.oks_date_begin and oks.oks_date_end
          and current_date - oks.oks_delinq_date >= 9
          and not exists(select 1
                         from edoc_list edl
                         join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
                         where edc.edc_code = 'loan.delinq.notif'
                               and edl.edl_blank_status = 1
                               and edl.edl_status <> -1
                               and edl.edl_object_kind = 9
                               and edl.edl_object_id = k.k_id)
    into :k_id
    do
    begin
        select edl_id,
               res_text
        from hp_edoc_ins('loan.delinq.notif', f_objkind('kontract', 'id'), :k_id, 1000000000000)
        into :edl_id,
             :res_text;
    end
end
