CREATE PROCEDURE AAREP_KONTRACT_LIST_OP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    K_ID D_ID,
    ET_CARDNUMBER D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    KF_OFFICER_NAME D_STR_100,
    CL_CASHIER_NAME D_STR_100,
    UN_FIO_APPROVED D_STR_100,
    PR_NAME D_STR_CODE,
    K_IS_ONLINE D_STR_20,
    FPS_NAME_FIO D_STR_100,
    ET_CARDNUMBER_ACTUAL D_STR_CODE,
    K_REFUND D_STR_10)
AS
declare variable k_disb_kind d_integer; /* ET_CARDNUMBER_ACTUAL */
declare variable c_id bigint; /* ID клиента */
begin
  for select kon.k_id, kon.k_kontract_no, kon.k_disb_date, kon.k_amount_approved, pr.pr_name, c2.c_id, c2.c_name_fio,
             c1.c_name_fio, c3.c_name_fio, kon.k_disb_kind, iif(k_refund = 1, 'да', 'нет')
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 2) ak
      left join kontract kon on ak.aarep_k_id = kon.k_id
      left join product pr on kon.k_from_pr_id = pr.pr_id
      left join knaktlist akt on (ak.aarep_kf_id = akt.kal_from_kf_id and
            akt.kal_date <= ak.aarep_date_action and
            akt.kal_date_end >= ak.aarep_date_action and
            akt.is_delete = 0)
      left join clients c1 on akt.kal_new_kf_officer = c1.c_id
      left join clients c2 on kon.k_clients_cashier_id = c2.c_id
      left join clients c3 on kon.k_approved_manager = c3.c_id
      into k_id, k_kontract_no, k_disb_date, k_amount_approved, pr_name, c_id, cl_cashier_name, kf_officer_name,
           un_fio_approved, k_disb_kind, k_refund
  do
  begin
    et_cardnumber = '';
    et_cardnumber_actual = '';
    fps_name_fio = '';
    k_is_online = 'Касса';
    if (k_disb_kind = 5) then
    begin
      k_is_online = 'Карта';

      select et_cardnumber
      from extsysmain
        left join extsys_token on esm_account = et_token
      where esm_project = 'moneta_disb'
        and esm_action = 'carddisb'
        and esm_status = 1
        and esm_from_k_id = :k_id
      into et_cardnumber;
    end
    if (k_disb_kind = 7) then
    begin
      k_is_online = 'СБП';

      select fcn_name
      from knclient
        left join fps_loan_banks on flb_from_kc_id = kc_id
        left join fps_client_names on fcn_from_fcb_id = fbl_from_fcb_id
      where kc_from_k_id = :k_id and
        fcn_status = 2
      order by fcn_id desc
      rows 1
      into fps_name_fio;
    end
    if (k_disb_kind = 8) then
      k_is_online = 'Расчетный счет';

    -- Последнюю закрепленную карту вытаскиваем вне зависимости от типа выдачи для Панковой Ю.
    select et_cardnumber
    from extsys_token
    where et_is_actual = 1 and
          et_from_c_id = :c_id
    into et_cardnumber_actual;

    suspend;
  end
end
