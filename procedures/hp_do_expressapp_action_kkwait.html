CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_KKWAIT(
    EXA_ID D_INTEGER)
RETURNS (
    WAIT_STATUS D_SMALLINT)
AS
declare variable k_id id_type;
declare variable exa_date d_date;
declare variable rd_id d_integer;
declare variable acl_status d_smallint;
declare variable k_status d_smallint;
begin
  -- Проверка КК
  wait_status = -1;

  select exa_from_k_id, exa_date
  from expressapp
  where exa_id = :exa_id
  into k_id, exa_date;

  select k_status
  from kontract kont
  where k_id = :k_id
  into :k_status;

  if (k_status in (-2, -3)) then
    wait_status = 1; --   Заявка еще не переведена в договор';

  if (k_status = -1) then
    wait_status = -1; --  Договор аннулирован, обратитесь к КОБД';

  if (k_status = 1) then
    wait_status = 0; --   Договор уже утвержден, значит считаем что КК одобрил

  if (k_status = 0) then
  begin
   /* select rd_id
    from refdata
    where rd_code = 'ActionGroup.Check_Loan_Data'
    into rd_id; */

    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id;

    select acl_status
    from actionlist
    where acl_object_kind = 9 and
          acl_object_id = :k_id and
          acl_from_rd_id = :rd_id and
          acl_date >= :exa_date
    order by acl_id desc
    rows 1
    into acl_status;

    --Отмена заполнения -3
    --Отказан  -2
    --На доработку  -1
    --В ожидании  0
    --Принят на обработку  1
    --Подтвежден  2

    if (acl_status is null or acl_status in (0, 1)) then
      wait_status = 2; -- до этого было 1

    if (acl_status = -1) then
      wait_status = 1; -- Доработка, уведомляем ВИПС

    if (acl_status = 2) then
      wait_status = 0;

    if (acl_status in (-3, -2)) then
    begin
      wait_status = -1;

      update expressapp
      set exa_final_status = -1,
          exa_wait_status = -1
      where exa_id = :exa_id;
    end
  end

  suspend;
end
