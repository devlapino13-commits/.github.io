CREATE PROCEDURE UXP$UXMAINMENU_CFG_CREATE
AS
declare variable mainmenu_code d_str_code;
declare variable us_uuid d_uuid;
declare variable fr_uuid d_uuid;
begin
  mainmenu_code = 'mainmenu';

  select us_uuid
  from ux$source
  where us_code = :mainmenu_code
  into us_uuid;

  if (us_uuid is null) then
    execute procedure uxp$uxsource_create(:mainmenu_code, '', 1, 1, 'Главное меню приложения');

  select us_uuid
  from ux$source
  where us_code = :mainmenu_code
  into us_uuid;

  if (not exists(select * from ux$form where ufm_uuid = uxf$mainmenu())) then
  begin
    select res_fr_uuid
    from uxp$uxform_create('uxform_one', :us_uuid, '', 'Главное меню', :mainmenu_code)
    into fr_uuid;
  end
end
