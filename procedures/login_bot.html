CREATE PROCEDURE LOGIN_BOT(
    CELL_NO D_STR_100)
RETURNS (
    CAN_LOGIN D_BOOL,
    C_NAME D_STR_100)
AS
declare variable c_id id_type;
begin
  can_login = 0;

  select c_name_fio,
         c_id
  from hp_find_client_by_continfo(null, :cell_no)
  inner join hr_pstatus ps on ps.ps_from_c_id = c_id
  where ps.ps_is_working = 1 and
        current_date between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_status = 1
  rows 1
  into c_name, c_id;

  if (c_name is not null) then
      can_login = 1;

  suspend;
end
