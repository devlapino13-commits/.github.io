CREATE PROCEDURE AAREP_OFFICE_KM_TMP(
    REP_REGION_ID INTEGER)
RETURNS (
    REG_ID INTEGER,
    REG_DATE DATE,
    REG_KM_KOLICH INTEGER)
AS
declare variable dd1 date;
declare variable dd2 date;
begin
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into :reg_id
  do
  begin
    for select distinct ps_date_begin, ps_date_end
        from hr_pstatus, joblist, jobkind
        where ps_from_reg_id = :reg_id and
              ps_from_jl_id = jl_id and
              ps_is_working = 1 and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.Region_KM'
        into :dd1, :dd2
    do
    begin
      reg_date = :dd1;
      reg_km_kolich = 1;
      suspend;
      reg_date = :dd2;
      reg_km_kolich = -1;
      suspend;
    end
  end
end
