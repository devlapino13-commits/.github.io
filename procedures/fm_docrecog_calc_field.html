CREATE PROCEDURE FM_DOCRECOG_CALC_FIELD(
    DRL_ID D_ID,
    IS_DRAFT D_BOOL)
AS
declare variable DRM_ID D_INTEGER;
declare variable DRF_ID D_INTEGER;
declare variable TMP_DRF_ID D_INTEGER;
declare variable RECOG_FIO D_STR_255;
declare variable FINAL_FIO D_STR_255;
declare variable RECOG_VALUE D_STR_255;
declare variable FINAL_VALUE D_STR_255;
declare variable ABC_TO D_STR_CODE;
declare variable ABC_FROM D_STR_CODE;
declare variable DRM_ENG_FIO_KIND D_INTEGER;
declare variable DB_ID D_INTEGER;
declare variable IS_ACCEPTED D_BOOL;
declare variable DRV_IS_ACCEPTED D_BOOL;
declare variable DRV_IS_ACCEPTED_CALC D_BOOL;
declare variable NEW_RECOG_VALUE D_STR_255;
declare variable NEW_FINAL_VALUE D_STR_255;
declare variable RECOG_FIO_ENG D_STR_255;
declare variable FINAL_FIO_ENG D_STR_255;
declare variable TMP_ABC_FROM D_STR_10;
declare variable CREATE_CLIENT_MODE D_BOOL = 0;
declare variable TMP_RECOG_VALUE D_STR_255;
declare variable TMP_FINAL_VALUE D_STR_255;
declare variable LEN D_SMALLINT;
declare variable N D_SMALLINT;
declare variable PK_ID D_INTEGER;
declare variable TMP_RECOG_DATE D_DATE;
declare variable TMP_FINAL_DATE D_DATE;
declare variable COUNTRY_CODE D_STR_255;
declare variable DOC_IS_PASSPORT D_BOOL;
declare variable BIRTH_COUNTRY_CODE D_STR_255;
declare variable COUNTRY_SYM D_STR_10;
declare variable DRM_CODE D_STR_CODE;
begin
  -- процедура вычисляет и заполняет авто-вычисляемые поля
  -- например, полное ФИО формируется из полей фамилия, имя и отчетство

  db_id = f_get_db_id();

  select drl_from_drm_id, drm_main_fio_kind, drm_alt_fio_kind, iif(drl_from_c_id is null, 1, 0),
    drl_from_pk_id, drm_code
  from docrecoglist, docrecogmain
  where drl_id = :drl_id and
        drl_from_drm_id = drm_id
  into :drm_id, abc_from, drm_eng_fio_kind, create_client_mode, pk_id, drm_code;
  if (drm_id is null) then
    exit;

  if (abc_from is null) then
    abc_from = 'orig';
  if (drm_eng_fio_kind is null) then
    drm_eng_fio_kind = 0;

  select rd_code, pdt_is_main_doc
  from passkind, refdata, passdoctype
  where pk_id = :pk_id and
        pk_country_id = rd_id and
        pk_from_pdt_id = pdt_id
  into :country_code, doc_is_passport;

  country_sym = right(country_code, 2);

  abc_to = 'orig';
  if (db_id = 81001 or (db_id in (39801, 41701) and
      abc_from <> 'cyr')) then
    abc_to = 'cyr';
  else
    abc_from = 'orig'; -- abc_from с какого алфавита переводить

  -- формирование полного ФИО, перевод основного ФИО на нужный алфавит при необходимости
  drf_id = null;
  recog_fio = '';
  final_fio = '';
  drv_is_accepted = 1;

  -- обрабатываем вычисляемое поле, если:
  -- режим "is_draft", где только отсканировали и хотим показать как результат.
  -- Берем все поля, а не только действующие.

  -- режим НЕ "is_draft", то берем только неактивные поля,
  -- так как активные уже подтверждаются, возможно и отредактировано пользователем

  select drf_id
  from docrecogfield
  where drf_code = 'calc_name_full' and
        drf_inactive = 0 and
        drf_is_extra = 1 and
        drf_from_drm_id = :drm_id and
        (:is_draft = 1 or
        (:create_client_mode = 0 and
        drf_invisible_on_doc = 1) or -- если создается клиент, включаем невидимые в этом режиме
        (:create_client_mode = 1 and
        drf_invisible_on_client = 1)) -- если создается документ, включаем невидимые в этом режиме
  into drf_id;

  if (drf_id is not null) then
  begin
    tmp_drf_id = null;
    for select drf_id
        from docrecogfield
        where drf_from_drm_id = :drm_id and
              drf_inactive = 0 and
              drf_code in ('name', 'surname', 'patronymic')
        order by case drf_code
                   when 'surname' then 1
                   when 'name' then 2
                   else 3
                 end
        into :tmp_drf_id
    do
    begin
      recog_value = '';
      final_value = '';

      select coalesce(drv_recog_value, ''), coalesce(drv_final_value, ''),
             minvalue(drv_is_accepted, drv_is_accepted_calc)
      from docrecogvalue
      where drv_from_drl_id = :drl_id and
            drv_from_drf_id = :tmp_drf_id
      into recog_value, final_value, is_accepted;
      if (is_accepted is null) then
        is_accepted = 1;

      if (abc_from in ('lat', 'eng')) then
      begin
        recog_value = replace(recog_value, '0', 'O'); -- цифра 0 на lat O
        final_value = replace(final_value, '0', 'O'); -- цифра 0 на lat O
      end
      else
      begin
        recog_value = replace(recog_value, '0', 'О'); -- цифра 0 на cyr О
        final_value = replace(final_value, '0', 'О'); -- цифра 0 на cyr О
        recog_value = replace(recog_value, '3', 'З'); -- цифра 3 на cyr З
        final_value = replace(final_value, '3', 'З'); -- цифра 3 на cyr З
      end
      -- если одно из полей распознано некачественно, полное ФИО тоже отметим как некачественное
      if (is_accepted < drv_is_accepted) then
        drv_is_accepted = is_accepted;

      if (recog_value <> '') then
        recog_fio = recog_fio || ' ' || recog_value;
      if (final_value <> '') then
        final_fio = final_fio || ' ' || final_value;

      recog_value = trim(recog_value);
      final_value = trim(final_value);

      new_recog_value = upper(f_translit_fio(abc_from, abc_to, recog_value, country_sym));
      new_final_value = upper( f_translit_fio(abc_from, abc_to, final_value, country_sym));

      update docrecogvalue d
      set d.drv_is_accepted = :drv_is_accepted,
          d.drv_recog_value = :new_recog_value,
          d.drv_final_value = :new_final_value
      where d.drv_from_drl_id = :drl_id and
          d.drv_from_drf_id = :tmp_drf_id;
    end

    recog_fio = trim(recog_fio);
    final_fio = trim(final_fio);

    new_recog_value = f_translit_fio(abc_from, abc_to, recog_fio, country_sym);
    new_final_value = f_translit_fio(abc_from, abc_to, final_fio, country_sym);

    -- если был перевод на другой алфавит, вычисляемое качество делаем 0 (Сомнительным)
    drv_is_accepted_calc = drv_is_accepted;
    if (new_recog_value <> recog_fio or new_final_value <> final_fio) then
      drv_is_accepted_calc = 0;

    update or insert into docrecogvalue (drv_from_drl_id, drv_from_drf_id, drv_is_accepted, drv_recog_value,
                                         drv_final_value, drv_is_accepted_calc)
    values (:drl_id, :drf_id, :drv_is_accepted, :new_recog_value, :new_final_value, :drv_is_accepted_calc)
    matching (drv_from_drl_id, drv_from_drf_id);
  end

  -- формирование английского ФИО, перевод основного ФИО на ENG алфавит при необходимости
  drf_id = null;
  recog_fio_eng = '';
  final_fio_eng = '';

  -- обрабатываем вычисляемое поле, если:
  -- режим "is_draft", где только отсканировали и хотим показать как результат.
  -- Берем все поля, а не только действующие.

  -- режим НЕ "is_draft", то берем только неактивные поля,
  -- так как активные уже подтверждаются, возможно и отредактировано пользователем
  if (drm_eng_fio_kind <> 0) then
  begin
    select drf_id
    from docrecogfield
    where drf_code = 'calc_name_full_eng' and
          drf_inactive = 0 and
          drf_is_extra = 1 and
          drf_from_drm_id = :drm_id and
          (:is_draft = 1 or
          (:create_client_mode = 0 and
          drf_invisible_on_doc = 1) or -- если создается клиент, включаем невидимые в этом режиме
          (:create_client_mode = 1 and
          drf_invisible_on_client = 1)) -- если создается документ, включаем невидимые в этом режиме
    into drf_id;

    if (drf_id is not null) then
    begin
      -- drm_eng_fio_kind: Как заполнять английское ФИО:
      -- 0 - Не заполнять
      -- 1 - Заполнять из _ENG полей
      -- 2 - Переводить из оригинального ФИО
      -- 3 - Копировать оригинальное ФИО

      if (drm_eng_fio_kind = 1) then -- Заполнять из _ENG полей
      begin
        tmp_drf_id = null;
        drv_is_accepted = 1;
        for select drf_id
            from docrecogfield
            where drf_from_drm_id = :drm_id and
                  drf_inactive = 0 and
                  drf_code in ('name_eng', 'surname_eng', 'patronymic_eng')
            order by case drf_code
                       when 'surname_eng' then 1
                       when 'name_eng' then 2
                       else 3
                     end
            into :tmp_drf_id
        do
        begin
          recog_value = '';
          final_value = '';

          select coalesce(drv_recog_value, ''), coalesce(drv_final_value, ''),
                 minvalue(drv_is_accepted, drv_is_accepted_calc)
          from docrecogvalue
          where drv_from_drl_id = :drl_id and
                drv_from_drf_id = :tmp_drf_id
          into recog_value, final_value, is_accepted;
          if (is_accepted is null) then
            is_accepted = 1;

          recog_value = replace(recog_value, '0', 'O'); -- цифра 0 на lat O
          final_value = replace(final_value, '0', 'O'); -- цифра 0 на lat O

          -- если одно из полей распознано некачественно, полное ФИО тоже отметим как некачественное
          if (is_accepted < drv_is_accepted) then
            drv_is_accepted = is_accepted;

          if (recog_value <> '') then
            recog_fio_eng = recog_fio_eng || ' ' || recog_value;
          if (final_value <> '') then
            final_fio_eng = final_fio_eng || ' ' || final_value;
        end

        recog_value = trim(recog_value);
        final_value = trim(final_value);
      end

      if (drm_eng_fio_kind = 2) then -- Переводить из оригинального ФИО
      begin
        tmp_abc_from = abc_from;
        if (tmp_abc_from = 'orig') then
        begin
          if (db_id = 94401) then
            tmp_abc_from = 'lat';
          else
            tmp_abc_from = 'cyr';
        end
        recog_fio_eng = f_translit_fio(tmp_abc_from, 'eng', recog_fio, country_sym);
        final_fio_eng = f_translit_fio(tmp_abc_from, 'eng', final_fio, country_sym);
      end

      if (drm_eng_fio_kind = 3) then -- Копировать оригинальное ФИО
      begin
        recog_fio_eng = recog_fio;
        final_fio_eng = final_fio;
      end

      recog_fio_eng = trim(recog_fio_eng);
      final_fio_eng = trim(final_fio_eng);

      drv_is_accepted_calc = drv_is_accepted;

      if (drm_eng_fio_kind = 2) then
        drv_is_accepted_calc = 0;

      update or insert into docrecogvalue (drv_from_drl_id, drv_from_drf_id, drv_is_accepted, drv_recog_value,
                                           drv_final_value, drv_is_accepted_calc)
      values (:drl_id, :drf_id, :drv_is_accepted, :recog_fio_eng, :final_fio_eng, :drv_is_accepted_calc)
      matching (drv_from_drl_id, drv_from_drf_id);
    end
  end

  -- формирование альтернативного ФИО, на национальном языке, берем без конвертации из name, surname, patronymic
  drf_id = null;
  recog_fio = '';
  final_fio = '';
  drv_is_accepted = 1;

  -- обрабатываем вычисляемое поле, если:
  -- режим "is_draft", где только отсканировали и хотим показать как результат.
  -- Берем все поля, а не только действующие.

  -- режим НЕ "is_draft", то берем только неактивные поля,
  -- так как активные уже подтверждаются, возможно и отредактировано пользователем
  select drf_id
  from docrecogfield
  where drf_code = 'calc_name_full_alt' and
        drf_inactive = 0 and
        drf_is_extra = 1 and
        drf_from_drm_id = :drm_id and
        (:is_draft = 1 or
        (:create_client_mode = 0 and
        drf_invisible_on_doc = 1) or -- если создается клиент, включаем невидимые в этом режиме
        (:create_client_mode = 1 and
        drf_invisible_on_client = 1)) -- если создается документ, включаем невидимые в этом режиме
  into drf_id;

  if (drf_id is not null) then
  begin
    tmp_drf_id = null;
    for select drf_id
        from docrecogfield
        where drf_from_drm_id = :drm_id and
              drf_inactive = 0 and
              drf_code in ('name', 'surname', 'patronymic')
        order by case drf_code
                   when 'surname' then 1
                   when 'name' then 2
                   else 3
                 end
        into :tmp_drf_id
    do
    begin
      recog_value = '';
      final_value = '';

      select coalesce(drv_recog_value, ''), coalesce(drv_final_value, ''),
             minvalue(drv_is_accepted, drv_is_accepted_calc)
      from docrecogvalue
      where drv_from_drl_id = :drl_id and
            drv_from_drf_id = :tmp_drf_id
      into recog_value, final_value, is_accepted;
      if (is_accepted is null) then
        is_accepted = 1;

      -- В результате - латиница, поэтому
      recog_value = replace(recog_value, '0', 'O'); -- цифра 0 на lat O
      final_value = replace(final_value, '0', 'O'); -- цифра 0 на lat O

      -- если одно из полей распознано некачественно, полное ФИО тоже отметим как некачественное
      if (is_accepted < drv_is_accepted) then
        drv_is_accepted = is_accepted;

      if (recog_value <> '') then
        recog_fio = recog_fio || ' ' || recog_value;
      if (final_value <> '') then
        final_fio = final_fio || ' ' || final_value;
    end

    new_recog_value = trim(recog_fio);
    new_final_value = trim(final_fio);
    drv_is_accepted_calc = drv_is_accepted;

    update or insert into docrecogvalue (drv_from_drl_id, drv_from_drf_id, drv_is_accepted, drv_recog_value,
                                         drv_final_value, drv_is_accepted_calc)
    values (:drl_id, :drf_id, :drv_is_accepted, :new_recog_value, :new_final_value, :drv_is_accepted_calc)
    matching (drv_from_drl_id, drv_from_drf_id);
  end

  -- формирование пола клиента (0 = женщина, 1 - мужчина)
  drf_id = null;

  select drf_id
  from docrecogfield
  where drf_code = 'calc_gender' and
        drf_inactive = 0 and
        drf_is_extra = 1 and
        drf_from_drm_id = :drm_id and
        (:is_draft = 1 or
        (:create_client_mode = 0 and
        drf_invisible_on_doc = 1) or -- если создается клиент, включаем невидимые в этом режиме
        (:create_client_mode = 1 and
        drf_invisible_on_client = 1)) -- если создается документ, включаем невидимые в этом режиме
  into drf_id;

  if (drf_id is not null) then
  begin
    tmp_drf_id = null;
    drv_is_accepted = 1;
    tmp_recog_value = '0';
    tmp_final_value = '0';

    for select drf_id
        from docrecogfield
        where drf_from_drm_id = :drm_id and
              drf_inactive = 0 and
              drf_code = 'gender'
        into :tmp_drf_id
    do
    begin
      recog_value = '';
      final_value = '';

      select coalesce(drv_recog_value, ''), coalesce(drv_final_value, ''),
             minvalue(drv_is_accepted, drv_is_accepted_calc)
      from docrecogvalue
      where drv_from_drl_id = :drl_id and
            drv_from_drf_id = :tmp_drf_id
      into recog_value, final_value, is_accepted;
      if (is_accepted is null) then
        is_accepted = 1;

      -- если одно из полей распознано некачественно, полное ФИО тоже отметим как некачественное
      if (is_accepted < drv_is_accepted) then
        drv_is_accepted = is_accepted;

      if (recog_value in ('M', 'Э') or (recog_value starting 'М')) then
        tmp_recog_value = '1';
      else
        tmp_recog_value = '0';

      if (final_value in ('M', 'Э') or (final_value starting 'М')) then
        tmp_final_value = '1';
      else
        tmp_final_value = '0';
    end

    drv_is_accepted_calc = drv_is_accepted;
    update or insert into docrecogvalue (drv_from_drl_id, drv_from_drf_id, drv_is_accepted, drv_recog_value,
                                         drv_final_value, drv_is_accepted_calc)
    values (:drl_id, :drf_id, :drv_is_accepted, :tmp_recog_value, :tmp_final_value, :drv_is_accepted_calc)
    matching (drv_from_drl_id, drv_from_drf_id);
  end

  -- формирование полного номера документа (серия + номер)
  drf_id = null;

  select drf_id
  from docrecogfield
  where drf_code = 'doc_no' and
        drf_inactive = 0 and
        drf_is_extra = 1 and
        drf_from_drm_id = :drm_id and
        (:is_draft = 1 or
        (:create_client_mode = 0 and
        drf_invisible_on_doc = 1) or -- если создается клиент, включаем невидимые в этом режиме
        (:create_client_mode = 1 and
        drf_invisible_on_client = 1)) -- если создается документ, включаем невидимые в этом режиме
  into drf_id;

  if (drf_id is not null) then
  begin
    tmp_drf_id = null;
    drv_is_accepted = 1;
    tmp_recog_value = '';
    tmp_final_value = '';

    for select drf_id
        from docrecogfield
        where drf_from_drm_id = :drm_id and
              drf_inactive = 0 and
              drf_code in ('series', 'number')
        order by case drf_code
                   when 'series' then 1
                   else 2
                 end
        into :tmp_drf_id
    do
    begin
      recog_value = '';
      final_value = '';

      select coalesce(drv_recog_value, ''), coalesce(drv_final_value, ''),
             minvalue(drv_is_accepted, drv_is_accepted_calc)
      from docrecogvalue
      where drv_from_drl_id = :drl_id and
            drv_from_drf_id = :tmp_drf_id
      into recog_value, final_value, is_accepted;
      if (is_accepted is null) then
        is_accepted = 1;

      -- если одно из полей распознано некачественно, результат - некачественное
      if (is_accepted < drv_is_accepted) then
        drv_is_accepted = is_accepted;

      if (tmp_recog_value = '') then
        tmp_recog_value = trim(recog_value);
      else
        tmp_recog_value = tmp_recog_value || ' ' || trim (recog_value);

      if (tmp_final_value = '') then
        tmp_final_value = trim(final_value);
      else
        tmp_final_value = tmp_final_value || ' ' || trim (final_value);
    end

    drv_is_accepted_calc = drv_is_accepted;
    update or insert into docrecogvalue (drv_from_drl_id, drv_from_drf_id, drv_is_accepted, drv_recog_value,
                                         drv_final_value, drv_is_accepted_calc)
    values (:drl_id, :drf_id, :drv_is_accepted, :tmp_recog_value, :tmp_final_value, :drv_is_accepted_calc)
    matching (drv_from_drl_id, drv_from_drf_id);
  end

  -- формирование кода органа выдавщего паспорт, если поле добавлено вручную
  drf_id = null;
  tmp_drf_id = null;
  drv_is_accepted = 1;
  tmp_recog_value = '';
  tmp_final_value = '';

  select drf_id
  from docrecogfield
  where drf_code = 'authority_code' and
        drf_inactive = 0 and
        drf_is_extra = 1 and
        drf_from_drm_id = :drm_id and
        (:is_draft = 1 or
        (:create_client_mode = 0 and
        drf_invisible_on_doc = 1) or -- если создается клиент, включаем невидимые в этом режиме
        (:create_client_mode = 1 and
        drf_invisible_on_client = 1)) -- если создается документ, включаем невидимые в этом режиме
  into drf_id;

  if (drf_id is not null) then
  begin
    select drf_id
    from docrecogfield
    where drf_code = 'authority' and
          drf_inactive = 0 and
          drf_from_drm_id = :drm_id
    into tmp_drf_id;

    if (tmp_drf_id is not null) then
    begin
      select coalesce(drv_recog_value, ''), coalesce(drv_final_value, ''), drv_is_accepted, drv_is_accepted_calc
      from docrecogvalue
      where drv_from_drl_id = :drl_id and
            drv_from_drf_id = :tmp_drf_id
      into recog_value, final_value, drv_is_accepted, drv_is_accepted_calc;
