CREATE PROCEDURE CM_RU_0130_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable c_id id_type;
declare variable pr_code varchar(100);
declare variable kc_balance curr_type;
declare variable kc_balance_par curr_type;
declare variable kol_kred integer;
declare variable k_refund smallint;
declare variable kol_kred_par integer;
begin
  k_refund = 0;
  for select kc_from_c_id, kc_amount_approved, pr_code, k_refund
      from knclient, kontract, product
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_code not containing '.zalog_car.' and
            pr_code not containing '.education.'
      into c_id, kc_amount, pr_code, k_refund
  do
  begin
    k_refund = coalesce(k_refund, 0);
    kc_balance = 0;
    kc_balance_par = 0;
    kol_kred = 0;
    kol_kred_par = 0;
    select sum(cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance),
           sum(iif(pr_code containing '.par.', cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance, 0)),
           count(*), sum(iif(pr_code containing '.par.', 1, 0))
    from knclient, oksummary, kontract, product
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = oks_from_k_id and
          kc_from_k_id = k_id and
          :kf_disb_date between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0 and
          k_from_pr_id = pr_id
    into kc_balance, kc_balance_par, kol_kred, kol_kred_par;
    kc_balance = coalesce(kc_balance, 0);
    kc_balance_par = coalesce(kc_balance_par, 0);
    kol_kred = coalesce(kol_kred, 0);
    kol_kred_par = coalesce(kol_kred_par, 0);
    if (k_refund = 1) then
    begin
      /*
      if (kc_amount + kc_balance_par > 90000) then
      begin
        if (pr_code containing '.reliable_client.' or pr_code containing '.reliable_client_plus.') then
          exit;
        res_error_text = 'Баланс активных кредитов не должно превышать 90 тыс.';
        res_error_is_critical = 2;
        suspend;
      end
      */
      if (kol_kred_par > 1) then
      begin
        res_error_text = 'Количество активных кредитов не должно превышать более 2х';
        res_error_is_critical = 2;
        suspend;
      end
    end
    else
    begin
      /*
      if (kc_amount + kc_balance > 90000) then
      begin
        if (pr_code containing '.reliable_client.' or pr_code containing '.reliable_client_plus.') then
          exit;
        res_error_text = 'Баланс активных кредитов не должно превышать 90 тыс.';
        res_error_is_critical = 2;
        suspend;
      end
      */
      if (kol_kred > 1) then
      begin
        res_error_text = 'Количество активных кредитов не должно превышать более 2х';
        res_error_is_critical = 2;
        suspend;
      end
      if (pr_code not similar to '%(.par.|.express.)%' and
          kol_kred - kol_kred_par >= 1) then
      begin
        res_error_text = '2 и более основных кредитов не должно быть!';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end
end
