CREATE PROCEDURE HP_KNAKT_AUTO_TRANSFER_RU(
    REP_LO_ID DECIMAL(18,0),
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    RES_TEXT VARCHAR(255),
    IS_ERROR SMALLINT)
AS
declare variable categ_1_delinq_days integer;
declare variable categ_2_delinq_days integer;
declare variable staff_stage integer;
declare variable staff_par_day integer;
declare variable staff_par_koef decimal(18,7);
declare variable rep_delinq_date date;
declare variable lo_id decimal(18,0);
declare variable lo_balance decimal(18,4) = 0;
declare variable lo_delinq_balance decimal(18,4) = 0;
declare variable lo_loan_count integer;
declare variable lo_par decimal(18,7);
declare variable lo_last_bonus decimal(18,4) = 0;
declare variable lo_stage_date date;
declare variable lo_stage integer;
declare variable prior_date_begin date;
declare variable prior_date_end date;
declare variable id integer;
declare variable order_no integer;
declare variable max_lo integer;
declare variable max_folder integer;
declare variable transfer_count integer;
declare variable lo_language integer;
declare variable tkf_kf_id id_type;
declare variable c_nation integer;
declare variable col_cli integer;
declare variable avg_balance curr_type;
declare variable cl_lang_no integer;
declare variable lo_language_str varchar(100);
declare variable rd_code varchar(255);
declare variable tkf_language_str varchar(10);
declare variable k integer;
begin
-- Процедура автоматически распределяет кредитные папки сотрудника другим специалистам
  res_text = '';
-- если res_text<>'' and res_akt_id=0 значит есть ошибка
-- если res_text<>'' and res_akt_id>0 значит все ок, в интерфейсе отображаем res_text
  is_error = 1;
  if (rep_lo_id is null or rep_region_id is null or rep_date is null) then
  begin
    res_text = 'Не все входящие параметры указаны';
    suspend;
    exit;
  end
  prior_date_end = rep_date - extract(day from rep_date);
  prior_date_begin = prior_date_end - extract(day from prior_date_end) + 1;

-- Настройки для РФ согласно приказа от августа 2016 г.
  categ_1_delinq_days = 15; -- Кол-во просроченных дней, ниже которых распределяется в первую очередь
  categ_2_delinq_days = 30; -- Кол-во просроченных дней, ниже которых распределяется на втором этапе

  staff_stage = 2; -- Стаж специалиста, допускаемые для приема папок
  staff_par_day = 30; -- Дни просрочки, ниже которых участвуют в расчете PAR
  staff_par_koef = 8.0; -- PAR сотрудника, ниже которых допускается для приема папок
  rep_delinq_date = dateadd(day, -30, rep_date);

  delete from temp_knakt_folder; -- Очищаем временные таблицы
  delete from temp_knakt_lo; -- Очищаем временные таблицы

-- Заполняем список папок
  insert into temp_knakt_folder (tkf_kf_id, tkf_balance, tkf_loan_count, tkf_delinq_days)
  select k_from_kf_id, sum(oks_balance), sum(k_kolich_client_approved), max(datediff(day, oks_delinq_date, :rep_date)) -- если просроченные дни считается то тут не попадает просрочки так как ниже имеется услвия
  from oksummary, kontract
  where oks_date_begin <= :rep_date and
        oks_date_end >= :rep_date and
        oks_balance > 0 and
        oks_kc_id = :rep_lo_id and
        oks_region_id = :rep_region_id and
        oks_from_k_id = k_id and
        oks_delinq_date >= :rep_delinq_date -- данная условия специально берет кредиты без просрочек
        and oks_close_state = 0
  group by k_from_kf_id
  order by right(k_from_kf_id, 2) -- для случайной сортировки
  ;

-- Заполняем список сотрудников
  for with tab
      as (select oks_kc_id as lo_id, oks_balance as lo_balance,
                 iif(oks_delinq_date < :rep_delinq_date, oks_balance, 0.0) as lo_delinq_balance,
                 k_kolich_client_approved as lo_loan_count
          from oksummary, kontract
          where oks_date_begin <= :rep_date and
                oks_date_end >= :rep_date and
                oks_balance > 0 and
                oks_kc_id <> :rep_lo_id and
                oks_kc_id > 1000000000004 and
                oks_region_id = :rep_region_id and
                oks_from_k_id = k_id and
                oks_close_state = 0
          union
          select ps_from_c_id as lo_id, 0.0 as lo_balance, 0.0 as lo_delinq_balance, 0 as lo_loan_count
          from hr_pstatus, joblist, jobkind
          where ps_from_c_id <> :rep_lo_id and
                ps_from_reg_id = :rep_region_id and
                ps_date_end >= :rep_date and
                ps_date_begin <= :rep_date and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner'))

      select lo_id, sum(lo_balance), sum(lo_delinq_balance), sum(lo_loan_count)
      from tab
      group by lo_id
      into :lo_id, lo_balance, lo_delinq_balance, lo_loan_count
  do
  begin
-- Проверка на наличие приказа
    if (exists(select *
               from hr_pstatus, joblist, jobkind
               where ps_from_c_id = :lo_id and
                     ps_from_reg_id = :rep_region_id and
                     ps_date_end >= :rep_date and
                     ps_date_begin <= :rep_date and
                     ps_status = 1 and
                     ps_is_working = 1 and
                     ps_from_jl_id = jl_id and
                     jl_kind = jk_id and
                     jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner'))) then
    begin
-- Чтоб не был в декретном отпуске
      if (not(exists(select *
                     from hr_voctravel, hr_orders, sa_worktype
                     where hrv_from_ho_id = ho_id and
                           ho_order_type = sawt_id and
                           hrv_from_c_id = :lo_id and
                           sawt_base_type = 5 -- декретный отпуск
                           and
                           sawt_id <> 20 -- кроме выхода из декрета
                           and
                           hrv_date_begin <= :rep_date and
                           hrv_date_end_fact >= :rep_date))) then
      begin
        lo_par = 0.0;
        if (lo_balance > 0) then
          lo_par = round((cast(lo_delinq_balance as double precision) / cast(lo_balance as double precision)) * 100.0, 2);

-- Расчет стажа
        lo_stage_date = null;
        lo_stage = 0;
        /*
        select min(ps_date_begin)
        from hr_pstatus, joblist, jobkind
        where ps_from_c_id=:lo_id
        and ps_status=1
        and ps_is_working=1
        and ps_from_jl_id=jl_id
        and jl_kind=jk_id
        and jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner')
        into :lo_stage_date;
        if (lo_stage_date is not null) then
        begin
          lo_stage=datediff(month, lo_stage_date, rep_date+1);
          if (extract(day from lo_stage_date)>extract(day from (rep_date+1)) and lo_stage>0)
          then lo_stage=lo_stage-1;
        end
        */

        select lo_stage_month
        from hp_get_lo_stage(:lo_id, :rep_date)
        into lo_stage;

-- Расчет бонуса
        lo_last_bonus = 0.0;
        select aarep_bonus_total_fact
        from cb_ru_lo_201410_one_lo(:rep_region_id, :lo_id, :prior_date_begin, :prior_date_end, 0)
        into :lo_last_bonus;
        if (lo_last_bonus is null) then
          lo_last_bonus = 0;

--      Язык для обращения СВЗ
        lo_language_str = null;
        for select cl_lang_no
            from clanguage
            where cl_mark_speak = 5 and
                  cl_from_c_id = :lo_id
            order by cl_is_native desc
            into cl_lang_no
        do
        begin
          rd_code = '';
          select rd_code
          from refdata
          where rd_id = :cl_lang_no
          into rd_code;

          lo_language_str = case rd_code
                              when 'Language.KG' then coalesce(lo_language_str || ';', '') || 'KG'
                              when 'Language.UZ' then coalesce(lo_language_str || ';', '') || 'UZ'
                              when 'Language.TJ' then coalesce(lo_language_str || ';', '') || 'TJ'
                              else lo_language_str
                            end;
        end
        lo_language_str = coalesce(lo_language_str || ';', '') || 'RU';

-- Вставка записи в список специалистов
        insert into temp_knakt_lo (tkl_lo_id, tkl_par, tkl_balance, tkl_loan_count, tkl_last_bonus, tkl_stage,
                                   /*tkl_lo_language,*/ tkl_lo_language_str)
        values (:lo_id, :lo_par, :lo_balance, :lo_loan_count, :lo_last_bonus, :lo_stage,
                /*:lo_language, */ :lo_language_str);
      end
    end
  end
  if (not(exists(select *
                 from temp_knakt_lo))) then
  begin
    res_text = 'Нет действующих специалистов, которые могли бы принят папки.';
    suspend;
    exit;
  end

-- Заполняем язык для обращения по группе
  for select tkf_kf_id
      from temp_knakt_folder
      into tkf_kf_id
  do
  begin
    c_nation = null;
    with tabl1
    as (select distinct c_id, c_nation
        from kontract, kontstate, knclient, clients
        where k_from_kf_id = :tkf_kf_id and
              k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date between ks_date_begin and ks_date_end and
              ks_from_k_id = kc_from_k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = c_id)
    select c_nation, count(*)
    from tabl1
    group by c_nation
    order by 2 desc
    rows 1
    into c_nation, col_cli;

    c_nation = coalesce(c_nation, 0);

    rd_code = '';
    select rd_code
    from refdata
    where rd_id = :c_nation
    into rd_code;

    tkf_language_str = case rd_code
                         when 'Nation.KG' then 'KG'
                         when 'Nation.UZ' then 'UZ'
                         when 'Nation.TJ' then 'TJ'
                         else 'RU'
                       end;

    update temp_knakt_folder
    set tkf_language_str = :tkf_language_str
    where tkf_kf_id = :tkf_kf_id;
  end

-- Временные таблицы готовы, приступаем к авто-распределению
-- У распределенной папки будет заполнено поле TKF_LO_ID
-- В распределении участвует папки где Order_No>0

  update temp_knakt_folder
  set tkf_order_no = 0;
  update temp_knakt_lo
  set tkl_order_no = 0;

-- Определяем список специалистов у кого стаж более 2 месяцев, Par30<8% и не получает бонус
  order_no = 0;
  max_lo = 0;
  max_folder = 0;
  for select tkl_id
      from temp_knakt_lo
      where tkl_stage >= 6 -- было 3
            and
            tkl_par < 8.0 and
            tkl_last_bonus <= 0.0
      order by tkl_last_bonus, tkl_balance
      into :id
  do
  begin
    order_no = order_no + 1;
    update temp_knakt_lo
    set tkl_order_no = :order_no
    where tkl_id = :id;
  end
  max_lo = order_no;
  if (max_lo > 0) then -- если есть такие сотрудники, определяем папки для передачи
  begin
    order_no = 0;

-- Папки где просрочки <=15 дней
    for select tkf_id
        from temp_knakt_folder
        where tkf_delinq_days <= :categ_1_delinq_days and
              tkf_lo_id is null
        order by tkf_balance desc
        into :id
    do
    begin
      order_no = order_no + 1;
      update temp_knakt_folder
      set tkf_order_no = :order_no
      where tkf_id = :id;
    end
    k = 0;
-- продолжаем пока есть нераспределенные
    while (exists(select * from temp_knakt_folder where tkf_lo_id is null and tkf_order_no > 0) and k <= order_no) do
    begin
      k = k + 1; -- чтобы не зациклился, если в языках КС не имеется язык группы то уже будет зацикливаться
      for select tkl_lo_id, tkl_lo_language_str
          from temp_knakt_lo
          where tkl_order_no > 0
          order by tkl_order_no
          into :lo_id, :lo_language_str
      do
      begin
        for select tkf_id
            from temp_knakt_folder
            where tkf_lo_id is null and
                  tkf_order_no > 0 and
                  :lo_language_str containing tkf_language_str
            order by tkf_order_no
            rows 9
            into :id
        do
        begin
          update temp_knakt_folder
          set tkf_lo_id = :lo_id
          where tkf_id = :id;
        end
      end
    end

-- Распределаем где просрочка>15 дней
    update temp_knakt_folder
    set tkf_order_no = 0;

    order_no = 0;
    for select tkf_id
        from temp_knakt_folder
        where tkf_delinq_days > :categ_1_delinq_days and
              tkf_delinq_days <= :categ_2_delinq_days and
              tkf_lo_id is null
        order by tkf_balance desc
        into :id
    do
    begin
      order_no = order_no + 1;
      update temp_knakt_folder
      set tkf_order_no = :order_no
      where tkf_id = :id;
    end

-- Список специалистов и количество уже принятых папок

    for select tkl_id, tkl_lo_id, count(*), avg(tkf_balance)
        from temp_knakt_lo, temp_knakt_folder
        where tkl_order_no > 0 and
              tkf_lo_id = tkl_lo_id
        group by tkl_id, tkl_lo_id
        having count(*) > 0
        into :id, :lo_id, :transfer_count, :avg_balance
    do
    begin
      transfer_count = trunc(transfer_count / 9, 0); -- распределяем в пропорции 1х9
      if (transfer_count > 0) then
      begin
        update temp_knakt_lo
        set tkl_avg_balance = :avg_balance,
            tkl_transfer_count = :transfer_count
        where tkl_id = :id;
      end
    end

    while (exists(select * from temp_knakt_lo where tkl_transfer_count > 0)) do
    begin
      for select tkl_id, tkl_lo_id, tkl_transfer_count, tkl_avg_balance, tkl_lo_language_str
          from temp_knakt_lo
          where tkl_transfer_count > 0
          order by tkl_order_no
          into :id, :lo_id, :transfer_count, :avg_balance, :lo_language_str
      do
      begin
--        update temp_knakt_lo set tkl_transfer_count = :transfer_count - 1
--        where tkl_id = :id;
        k = 0;
        for select tkf_id
            from temp_knakt_folder
            where tkf_lo_id is null and
                  tkf_order_no > 0 and
                  :lo_language_str containing tkf_language_str and
                  abs(:avg_balance - tkf_balance) < 15000.0
            order by abs(:avg_balance - tkf_balance), -- наиболее близкий баланс к среднему значению
                tkf_order_no
            rows 1
            into :id
        do
        begin
          update temp_knakt_folder
          set tkf_lo_id = :lo_id
          where tkf_id = :id;

          update temp_knakt_lo
          set tkl_transfer_count = :transfer_count - 1
          where tkl_id = :id;
          k = 1;
        end

        for select tkf_id
            from temp_knakt_folder
            where tkf_lo_id is null and
                  tkf_order_no > 0 and
                  :lo_language_str containing tkf_language_str and
                  abs(:avg_balance - tkf_balance) between 15000.0 and 30000.0 and
                  :k = 0
            order by abs(:avg_balance - tkf_balance), -- наиболее близкий баланс к среднему значению
                tkf_order_no
            rows 1
            into :id
        do
        begin
          update temp_knakt_folder
          set tkf_lo_id = :lo_id
          where tkf_id = :id;

          update temp_knakt_lo
          set tkl_transfer_count = :transfer_count - 2
          where tkl_id = :id;
          k = 1;
        end

        if (k = 0) then
        begin
          update temp_knakt_lo
          set tkl_transfer_count = :transfer_count - 1
          where tkl_id = :id;
        end

      end
    end

    /*
    for
    select tkl_lo_id,
           tkl_order_no,
           count(*),
           avg(tkf_balance), -- вычисляем средний баланс
           tkl_lo_language_str
    from temp_knakt_lo, temp_knakt_folder
    where tkl_order_no>0
    and tkf_lo_id=tkl_lo_id
    group by tkl_lo_id,
             tkl_order_no,
             tkl_lo_language_str
    having count(*)>0
    order by tkl_order_no
    into :lo_id,
         :order_no,
         :transfer_count,
         :avg_balance,
         :lo_language_str
    do
    begin
      transfer_count = trunc(transfer_count/9, 0); -- распределяем в пропорции 1х9

      if (transfer_count>0) then
      begin

        for
        select tkf_id
        from temp_knakt_folder
        where tkf_lo_id is null
        and tkf_order_no > 0
        and :lo_language_str containing tkf_language_str
        and abs(:avg_balance - tkf_balance) < 15000.0
        order by abs(:avg_balance - tkf_balance),  -- наиболее близкий баланс к среднему значению
                 tkf_order_no
        rows :transfer_count
        into :id
        do
        begin
          update temp_knakt_folder
          set tkf_lo_id=:lo_id
          where tkf_id=:id;
        end

      end
    end
    */
  end -- Конец распределения 1-этапа (Просрочки <=15 дней)

  /*
  if (exists(select * from temp_knakt_folder where tkf_lo_id is null)) then -- Если остались папки для распределения
  begin
-- Второй этап, распределяем оставщиеся папки
    update temp_knakt_folder set tkf_order_no=0;
    update temp_knakt_lo set tkl_order_no=0;
  
-- Сортируем специалистов по кредитному портфель, самые большие идут первыми
    order_no=0;
    max_lo=0;
    max_folder=0;
    for select tkl_id from temp_knakt_lo
    order by tkl_balance desc
    into :id
    do
    begin
      order_no=order_no+1;
      update temp_knakt_lo
      set tkl_order_no=:order_no
      where tkl_id=:id;
    end
    max_lo=order_no;

-- распределяем все оставщиеся папки
    order_no=0;
    for
    select tkf_id,
           tkf_language_str
    from temp_knakt_folder
    where tkf_lo_id is null
    order by tkf_balance
    into :id,
         :tkf_language_str
    do
    begin
      lo_id=null;
      order_no=order_no+1;
      if (order_no>max_lo) then order_no=1;

      select tkl_lo_id from temp_knakt_lo
      where tkl_order_no = :order_no
      and tkl_lo_language_str containing :tkf_language_str
      into :lo_id;

      update temp_knakt_folder
      set tkf_lo_id=:lo_id
      where tkf_id=:id;
    end
  end
  */

  order_no = 0;
  select count(*)
  from temp_knakt_folder
  where tkf_lo_id is null
  into order_no;

  k = 0;
-- продолжаем пока есть нераспределенные
  while (exists(select * from temp_knakt_folder where tkf_lo_id is null) and k < order_no) do
  begin
    k = k + 1; -- чтобы не зациклился, если в языках КС не имеется язык группы то уже будет зацикливаться
    for select tkl_lo_id, tkl_lo_language_str
        from temp_knakt_lo
        where tkl_order_no > 0
        order by tkl_par
        into :lo_id, :lo_language_str
    do
    begin
      for select tkf_id
          from temp_knakt_folder
          where tkf_lo_id is null and
                :lo_language_str containing tkf_language_str rows 1
          into :id
      do
      begin
        update temp_knakt_folder
        set tkf_lo_id = :lo_id
        where tkf_id = :id;
      end
    end
  end

  if (exists(select *
             from temp_knakt_folder
             where tkf_lo_id is null)) then -- Если остались папки для распределения
  begin
    res_text = 'Не все папки удалось распределить.';
  end

  if (exists(select *
             from temp_knakt_folder
             where tkf_lo_id is not null)) then
  begin
    is_error = 0;
  end

  suspend;
end
