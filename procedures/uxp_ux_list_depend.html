CREATE PROCEDURE UXP$UX_LIST_DEPEND
RETURNS (
    RES_ID D_STR_255,
    RES_PARENT_ID D_STR_255,
    RES_KIND D_STR_CODE,
    RES_IMAGE D_SMALLINT,
    RES_NAME D_STR_100,
    RES_FCODE D_STR_100,
    RES_MEMO D_STR_LONG,
    RES_KIND_CAPTION D_STR_CODE,
    RES_SOURCE_UUID D_UUID,
    RES_DATETIME D_DATETIME,
    RES_WARNING_STATE D_SMALLINT,
    RES_UUID D_UUID,
    RES_UUID_STR D_STR_CODE,
    RES_SCODE D_STR_CODE,
    RES_CAPTION D_STR_100,
    RES_SOURCE_CODE D_STR_CODE,
    RES_READONLY D_BOOL)
AS
declare variable udp_child_locked d_bool;
declare variable udp_lock_delete d_bool;
declare variable udp_caption d_str_255;
begin
  res_memo = '';
  for select udp_code, udp_parent_code, udp_ux_code, udp_ux_kind, udp_image_id,
             udp_lock_delete, udp_child_locked, udp_caption, udp_uuid
  from ux$depend
  into :res_id, :res_parent_id, :res_fcode, :res_kind, :res_image,
       :udp_lock_delete, :udp_child_locked, udp_caption, res_uuid
  do
  begin
    res_uuid_str = uuid_to_char(res_uuid);

    res_caption = udp_caption;
    res_name = udp_caption;
    res_kind_caption = '';
    res_source_uuid = null;
    res_datetime = null;
    res_scode = '';

    res_kind_caption = uxf$kind(res_kind, 'caption');

    if (coalesce(res_uuid, '') <> '') then
      res_scode = uxf$code(res_uuid, 'code');

    res_readonly = 0;

    if (res_kind = 'uxframe') then
    begin
      select ufr_us_uuid,
             iif(ufm_uuid = ux_release_uuid or (ux_draft_user <> current_user and ufm_user <> current_user), 1, 0)
      from ux$frame, ux$form, ux$code
      where ufr_uuid = :res_uuid and
            ufr_owner = ufm_uuid and
            ufm_ux_uuid = ux_uuid
      into res_source_uuid, res_readonly;
    end else
    if (res_kind = 'uxsource') then
    begin
      select trim(iif(us_is_table = 1, 'table', 'procedure'))
      from ux$source
      where us_uuid = :res_uuid
      into res_kind;

      if (res_kind = 'table') then
      begin
        res_kind_caption = 'Таблица';
        res_image = 21;
      end
      else
      begin
        res_kind_caption = 'Процедура';
        res_image = 75;
      end

      res_source_uuid = res_uuid;
    end
    else
    if (res_kind = 'uxaction') then
    begin
      select iif(ufm_uuid = ux_release_uuid or (ux_draft_user <> current_user and ufm_user <> current_user), 1, 0)
      from ux$action, ux$form, ux$code
      where ua_uuid = :res_uuid and
            ua_owner = ufm_uuid and
            ufm_ux_uuid = ux_uuid
      into res_readonly;

      if (row_count = 0) then
      begin
        select iif(ufm_uuid = ux_release_uuid or (ux_draft_user <> current_user and ufm_user <> current_user), 1, 0)
        from ux$action, ux$frame, ux$form, ux$code
        where ua_uuid = :res_uuid and
              ua_owner = ufr_uuid and
              ufm_uuid = ufr_owner and
              ufm_ux_uuid = ux_uuid
        into res_readonly;
      end
    end else
    begin
      select ux_us_uuid, uu_datetime
      from ux$code, ux$uniq
      where ux_uuid = :res_uuid and
            ux_uuid = uu_id
      into res_source_uuid, res_datetime;
  
      if (row_count = 0) then
      begin
        select ux_us_uuid, uu_datetime,
               iif(ufm_uuid = ux_release_uuid or (ux_draft_user <> current_user and ufm_user <> current_user), 1, 0)
        from ux$code, ux$form, ux$uniq
        where ufm_uuid = :res_uuid and
              ux_uuid = ufm_ux_uuid and
              ux_uuid = ufm_uuid
        into res_source_uuid, res_datetime, res_readonly;
      end
    end

    res_warning_state = 0;

    if (udp_lock_delete = 1) then
      res_warning_state = 1;
    else
    if (udp_child_locked = 1) then
      res_warning_state = 2;

    res_source_code = uxf$code(res_source_uuid, 'code');

    suspend;
  end
end
