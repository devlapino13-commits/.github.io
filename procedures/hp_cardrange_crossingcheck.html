CREATE PROCEDURE HP_CARDRANGE_CROSSINGCHECK(
    CARD_LIST D_STR_8100,
    FCARD_ID D_INTEGER = null,
    LCARD_ID D_INTEGER = null)
RETURNS (
    IS_PINCOVER D_BOOL,
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
begin
  err_code = 0;
  err_text = '';

  if (fcard_id > 0 and lcard_id > 0) then
  begin
    if (exists(
      select *
      from hp_cardrange_jsontolist(:card_list)
      where
        (:fcard_id between first_card_id and last_card_id) or
        (:lcard_id between first_card_id and last_card_id) or
        (first_card_id between :fcard_id and :lcard_id) or
        (last_card_id between :fcard_id and :lcard_id)))
    then err_code = 1;
  end
  else
  begin
    if (exists(
      select *
      from hp_cardrange_jsontolist(:card_list) as a,
           hp_cardrange_jsontolist(:card_list) as b
      where (a.id <> b.id) and (
        a.first_card_id between b.first_card_id and b.last_card_id or
        a.first_card_id between b.first_card_id and b.last_card_id or
        b.first_card_id between a.first_card_id and a.last_card_id or
        b.first_card_id between a.first_card_id and a.last_card_id)))
    then err_code = 1;
  end

  if (err_code = 1) then
    err_text = 'Некорректный ввод. Обнаружено пересечение диапазонов.';

  is_pincover = coalesce((
    select
      cem_is_pincover
    from cardlist
      inner join cardembomain
        on cem_id = cal_from_cem_id
    where cal_card_id = :fcard_id), 0);

  suspend;
end
