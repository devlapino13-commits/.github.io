CREATE PROCEDURE CM_RU_0121_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kf_disb_date d_date;
declare variable c_id id_type;
declare variable k_from_kf_id id_type;
declare variable re_k_id id_type;
declare variable re_kolich_repay integer;
declare variable k_from_pr_id integer;
declare variable pr_code varchar(100);
declare variable re_k_period integer;
declare variable count_repay integer;
declare variable k_refund d_smallint;
declare variable add_pr_code d_str_100;
declare variable pr_code_one d_str_100;
begin
  for select k_from_kf_id, k_disb_date, k_from_pr_id, pr_code, k.k_clients_cashier_id, k_refund, pr_code
      from kontract k, product
      where k_id = :k_id and
            coalesce(k_refund, 0) = 1 and
            k_from_pr_id = pr_id
      into k_from_kf_id, kf_disb_date, k_from_pr_id, pr_code, c_id, k_refund, add_pr_code
  do
  begin
    if (pr_code containing '.summer_plus.') then -- Рефинансирование для Лето Плюс разрешено, условия контролируются сотрудниками
      exit;
    if (pr_code containing '.comfortable.') then -- Остальное все тоже самое
      exit;
    if (pr_code containing '.refinance.') then -- Продукт для рефинансирования только внешних заемов
      exit;
    if (pr_code containing '.refinance_oivo.') then -- Продукт для рефинансирования только внешних заемов
      exit;
    if (pr_code containing '.refinance_plus.') then
      exit;
    if (pr_code containing '.b_refinance.') then
      exit;

    if (pr_code not containing '.blago.' and
        pr_code not containing '.uspeh.' and
        pr_code not containing '.shans.' and
        pr_code not containing '.status.' and
        pr_code not containing '.reliable_client.' and
        pr_code not containing '.reliable_client_plus.'
        and not(k_refund = 1 and kf_disb_date >= '01.03.2022' and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.|.zalog.|.selfemployed.|.business.)%')

        ) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Выберите правильный продукт для перекредитование(«Благо» или «Успех» или «Шанс» или «Статус»).';
      suspend;
      exit;
    end

    pr_code_one = null;
    pr_code_one = case
                     when pr_code containing '.shans.' then '.shans.'
                     when pr_code containing '.express.' then '.express.'
                     when pr_code containing '.ZALOG_CAR.' then '.ZALOG_CAR.'
                     when pr_code containing '.education.' then '.education.'
                     when pr_code containing '.reliable_client.' then '.reliable_client.'
                     when pr_code containing '.LOYAL.' then '.LOYAL.'
                     when pr_code containing '.reliable_emp.' then '.reliable_emp.'
                     when pr_code containing '.exclusive.' then '.exclusive.'
                     when pr_code containing '.zalog.' then '.zalog.'
                     when pr_code containing '.selfemployed.' then '.selfemployed.'
                     when pr_code containing '.business.' then '.business.'
                    end;
    pr_code_one = trim(pr_code_one);

    --  проверка активного кредита
    --  по какому продукту берут то по такому долны перекредитоваться или могут другие продукты выбирать?
    --  из группового не могут ведь перекредитоваться в индив?
    re_k_id = null;
    re_kolich_repay = null;
    re_k_period = 0;
    select k_id, k_kolich_repay, k.k_period, pr_code
    from kontract k, product, kontstate, knclient
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
--          pr_code similar to '%(.blago.|.uspeh.|.barakat|.family|.shans.|.status.|.reliable_client.|.reliable_client_plus.)%' and
          k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          :kf_disb_date between ks_date_begin and ks_date_end and
          ks_date_begin < :kf_disb_date and
          pr_code containing :pr_code_one
    order by ks_date_begin desc
    rows 1
    into re_k_id, re_kolich_repay, re_k_period, pr_code;

--    if (re_k_id is null) then
    if (pr_code not similar to '%(.blago.|.uspeh.|.barakat|.family|.shans.|.status.|.reliable_client.|.reliable_client_plus.)%' and
        not(k_refund = 1 and kf_disb_date >= '01.03.2022' and (pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.|.comfortable.)%' or add_pr_code containing '.zalog.'))
    ) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Не имеется активный заем для перекредитование!';
      if (k_id = 101010033242500401) then res_error_is_critical = 0;
      if (k_id <> 101010063343400601) then
      suspend;
      exit;
    end
--  добавлено по письмоу Зары от 7.9.21, на рефинансирование преднадлежать продукты Благо, Успех, Надежный клиент и Надежный клиент +
    if (pr_code not similar to '%(.blago.|.uspeh.|.reliable_client.|.reliable_client_plus.)%'and
        not(k_refund = 1 and kf_disb_date >= '01.03.2022' and (pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.|.comfortable.)%' or add_pr_code containing '.zalog.'))) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Для перекредитование у акт. заема должны быть один из след. продуктов: Благо, Успех, Надежный клитент, Надежный кл. + ';
      suspend;
      exit;
    end

    --bkoshmatov приказ 162-11-20 от 19.11.2021 условие перекредитования
    if (re_k_period < 5 and
        not(k_refund = 1 and kf_disb_date >= '01.03.2022' and (pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.|.comfortable.)%' or add_pr_code containing '.zalog.'))) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Активный заем был выдан на срок до 5 месяцев!';
      suspend;
    end
    count_repay = 0;
    with tab
    as (select ok_sh_id, ok_sh_date
        from operkred ok
        where ok.ok_from_k_id = :re_k_id and
              ok.ok_ol_is_active = 1 and
              ok_type < 7
        group by ok_sh_id, ok_sh_date
        having sum(ok.ok_os + ok.ok_sh_os) <> 0 or sum(ok.ok_perc + ok.ok_sh_perc) <> 0)
    select count(ok_sh_id)
    from tab
    into :count_repay; --количество оставшихся погашений
    count_repay = coalesce(count_repay, 0);
    --bkoshmatov приказ 162-11-20 от 19.11.2021 условие перекредитования
    if (count_repay > 4 and
        not(k_refund = 1 and kf_disb_date >= '01.03.2022' and (pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.|.comfortable.)%' or add_pr_code containing '.zalog.'))
    ) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'По активному займу до конца срока осталось более 4 платежей';
      suspend;
    end
   /* re_kolich_repay = coalesce(re_kolich_repay, 0);
    count_legitim_pay = 0;
    for select ok_sh_id, -sum(ok.ok_os + ok.ok_perc + ok.ok_fine)
        from operkred ok
        where ok.ok_from_k_id = :re_k_id and
              ok.ok_ol_is_active = 1 and
              ok.ok_type = 4 and
              ok.ok_sh_date - ok.ok_ol_date <= 10
        group by ok_sh_id
        into :ok_sh_id, repay
    do
    begin
      if (exists(select *
                 from operkred ok
                 where ok.ok_from_k_id = :re_k_id and
                       ok.ok_ol_is_active = 1 and
                       ok.ok_type = 1 and
                       ok.ok_sh_id = :ok_sh_id and
                       ok.ok_sh_os + ok.ok_sh_perc <= :repay)) then
        count_legitim_pay = count_legitim_pay + 1;
    end
    --bkoshmatov приказ 162-11-20 от 19.11.2021 условие перекредитования
    /*3)    по активному займу полностью оплачено не менее 4 платежей по графику и до конца срока осталось не более 4 платежей.
Полностью оплаченными признаются платежи, внесенные не ранее чем за 10 дней до даты платежа по графику. 
*/
  /*  if (count_legitim_pay < 4) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'По активному займу еще не было 4 погашений!';
      suspend;
    end    */
  end
end
