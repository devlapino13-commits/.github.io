CREATE PROCEDURE IPC_QR_PAYMENTS_OL_CREATE
AS
declare variable iqp_id d_integer;
begin
    for select iqp.iqp_id
    from ipc_qr_payments iqp
    where iqp.iqp_state = 0
    and iqp.iqp_from_cs_id > 0
    into :iqp_id
    do
    begin
       execute procedure ipc_qr_payments_ol_create_one(:iqp_id);
    end
end
