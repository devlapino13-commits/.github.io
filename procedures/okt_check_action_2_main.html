CREATE PROCEDURE OKT_CHECK_ACTION_2_MAIN(
    PROJECT STR_100,
    OBJ_ID ID_TYPE,
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE,
    SUBJECT_KIND STR_100 = 'ContactInfo.Cell_Main')
RETURNS (
    IS_ACTIVE D_SMALLINT)
AS
declare variable K_ID ID_TYPE;
declare variable SUBJECT_ID D_INTEGER;
begin
  /*
    OKTELL
    процедура проверяет, активна задача или нет.   (
    Данная проверка предназначена для проектов подобных Сирене
    IS_ACTIVE (1-активна; 0-не активна)
  */

  is_active = 1;

  select rd.rd_id
  from refdata rd
  where rd.rd_code = :subject_kind
  into :subject_id;

  select pantk.pant_from_k_id
  from pantera_kontracts pantk
  where pantk.pant_id = :obj_id and
        pantk.pant_date_end = '30.12.9999'
  into :k_id;

  -- проверяю по игнор-листу
  select can_call
  from okt_can_call(null, :object_kind, :object_id, null, null, :subject_id, null)
  into :is_active;

  -- првоеряю по дате след. звонка
  is_active = coalesce(is_active, 0);

  if (is_active = 1) then
  begin
    select can_call
    from hp_can_call_check(:k_id, :object_kind, :object_id, null)
    into is_active;
  end

  is_active = coalesce(is_active, 0);

  suspend;
end
