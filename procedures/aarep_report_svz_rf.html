CREATE PROCEDURE AAREP_REPORT_SVZ_RF(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    KOL_KRED INTEGER,
    BALANCE CURR_TYPE,
    KOL_KRED_DELINQ INTEGER,
    BALANCE_DELINQ CURR_TYPE,
    KOL_KRED_DELINQ7 INTEGER,
    BALANCE_DELINQ7 CURR_TYPE,
    KOL_KRED_DELINQ30 INTEGER,
    BALANCE_DELINQ30 CURR_TYPE,
    KOL_KRED_NOT_OWN INTEGER,
    BALANCE_NOT_OWN CURR_TYPE,
    KOL_KRED_IS_OWN INTEGER,
    BALANCE_IS_OWN CURR_TYPE,
    BALANCE_DELINQ_IS_OWN DECIMAL(18,4),
    BALANCE_DELINQ7_IS_OWN DECIMAL(18,4),
    BALANCE_DELINQ30_IS_OWN DECIMAL(18,4),
    BALANCE_DELINQ_NOT_OWN DECIMAL(18,4),
    BALANCE_DELINQ7_NOT_OWN DECIMAL(18,4),
    BALANCE_DELINQ30_NOT_OWN DECIMAL(18,4),
    CB_PERSONAL_STAGE_MONTH INTEGER,
    BALANCE_DAY7_AGO CURR_TYPE,
    BALANCE_DELINQ_DAY7_AGO CURR_TYPE,
    BALANCE_DELINQ7_DAY7_AGO CURR_TYPE,
    BALANCE_DELINQ30_DAY7_AGO CURR_TYPE,
    BALANCE_NOT_OWN_DAY7_AGO CURR_TYPE,
    BALANCE_DELINQ_NOT_OWN_DAY7_AGO CURR_TYPE,
    BALANCE_DELINQ7_NOT_OWN_DAY7_AG CURR_TYPE,
    BALANCE_DELINQ30_NOT_OWN_DAY7_A CURR_TYPE,
    BALANCE_IS_OWN_DAY7_AGO CURR_TYPE,
    BALANCE_DELINQ_IS_OWN_DAY7_AGO CURR_TYPE,
    BALANCE_DELINQ7_IS_OWN_DAY7_AGO CURR_TYPE,
    BALANCE_DELINQ30_IS_OWN_DAY7_AG CURR_TYPE,
    BALANCE_DAY30_AGO CURR_TYPE,
    BALANCE_DELINQ_DAY30_AGO CURR_TYPE,
    BALANCE_DELINQ7_DAY30_AGO CURR_TYPE,
    BALANCE_DELINQ30_DAY30_AGO CURR_TYPE,
    BALANCE_NOT_OWN_DAY30_AGO CURR_TYPE,
    BALANCE_DELINQ_NOT_OWN_DAY30_AG CURR_TYPE,
    BALANCE_DELINQ7_NOT_OWN_DAY30_A CURR_TYPE,
    BALANCE_DELINQ30_NOT_OWN_DAY30_ CURR_TYPE,
    BALANCE_IS_OWN_DAY30_AGO CURR_TYPE,
    BALANCE_DELINQ_IS_OWN_DAY30_AGO CURR_TYPE,
    BALANCE_DELINQ7_IS_OWN_DAY30_AG CURR_TYPE,
    BALANCE_DELINQ30_IS_OWN_DAY30_A CURR_TYPE,
    OFFICER_ID DECIMAL(18,0),
    IS_OFFICER SMALLINT)
AS
declare variable reg_id_prior integer;
declare variable reg_id integer;
declare variable cb_personal_stage date;
declare variable rep_date_end7 date;
declare variable rep_date_end30 date;
begin
  reg_id_prior = 0;
  for select oks_region_id, oks_kc_id, sum(k_kolich_client_approved), sum(oks_balance),
             sum(iif(:rep_date_end >= oks_delinq_date, k_kolich_client_approved, 0)),
             sum(iif(:rep_date_end >= oks_delinq_date, oks_balance, 0)),
             sum(iif(:rep_date_end - oks_delinq_date > 7, k_kolich_client_approved, 0)),
             sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0)),
             sum(iif(:rep_date_end - oks_delinq_date > 30, k_kolich_client_approved, 0)),
             sum(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0)),
--       ѕереданные кредиты
             sum(iif(ks_kc_id <> oks_kc_id, k_kolich_client_approved, 0)),
             sum(iif(ks_kc_id <> oks_kc_id, oks_balance, 0)),
             sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end >= oks_delinq_date, oks_balance, 0)),
             sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end - oks_delinq_date > 7, oks_balance, 0)),
             sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end - oks_delinq_date > 30, oks_balance, 0)),
--       —обственные кредиты
             sum(iif(ks_kc_id = oks_kc_id, k_kolich_client_approved, 0)),
             sum(iif(ks_kc_id = oks_kc_id, oks_balance, 0)),
             sum(iif(ks_kc_id = oks_kc_id and :rep_date_end >= oks_delinq_date, oks_balance, 0)),
             sum(iif(ks_kc_id = oks_kc_id and :rep_date_end - oks_delinq_date > 7, oks_balance, 0)),
             sum(iif(ks_kc_id = oks_kc_id and :rep_date_end - oks_delinq_date > 30, oks_balance, 0))
      from oksummary, kontract, kontstate, region_list(:rep_region_id, 1, 3, 1)
      where oks_region_id = id_list and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0
--  and oks_kc_id > 1000000000003
            and
            :rep_date_end - oks_delinq_date <= 360 and
            oks_from_k_id = k_id and
            oks_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end
            and oks_close_state = 0
      group by 1, 2
      into reg_id, officer_id, kol_kred, balance, kol_kred_delinq, balance_delinq, kol_kred_delinq7, balance_delinq7,
           kol_kred_delinq30, balance_delinq30,

           kol_kred_not_own, balance_not_own, balance_delinq_not_own, balance_delinq7_not_own, balance_delinq30_not_own,

           kol_kred_is_own, balance_is_own, balance_delinq_is_own, balance_delinq7_is_own, balance_delinq30_is_own

  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end

    select c_name_fio
    from clients
    where c_id = :officer_id
    into officer_name;

--    ƒата поступлени¤ на работу
    cb_personal_stage = null;
    /*select ps_date_begin
    from hr_pstatus
    where ps_from_c_id = :officer_id and
          ps_date_begin is not null and
          ps_status = 1
    order by ps_date_begin
    rows 1
    into :cb_personal_stage;

    if (cb_personal_stage is null) then
      cb_personal_stage = :rep_date_end;
    cb_personal_stage_month = datediff(day, :cb_personal_stage, :rep_date_end) / 30;  */


    select lo_stage_month
    from hp_get_lo_stage(:officer_id, :rep_date_end)
    into:cb_personal_stage_month;

--  ƒл¤ расчета динамики вытаскиваетс¤ портфель и просроченные дний
--  ƒл¤ динамки 7 дней
    balance_day7_ago = 0;
    balance_delinq_day7_ago = 0;
    balance_delinq7_day7_ago = 0;
    balance_delinq30_day7_ago = 0;
    balance_not_own_day7_ago = 0;
    balance_delinq_not_own_day7_ago = 0;
    balance_delinq7_not_own_day7_ag = 0;
    balance_delinq30_not_own_day7_a = 0;
    balance_is_own_day7_ago = 0;
    balance_delinq_is_own_day7_ago = 0;
    balance_delinq7_is_own_day7_ago = 0;
    balance_delinq30_is_own_day7_ag = 0;

    rep_date_end7 = rep_date_end - 7;
    select sum(oks_balance), sum(iif(:rep_date_end7 >= oks_delinq_date, oks_balance, 0)),
           sum(iif(:rep_date_end7 - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(:rep_date_end7 - oks_delinq_date > 30, oks_balance, 0)),
--         ѕереданные кредиты
           sum(iif(ks_kc_id <> oks_kc_id, oks_balance, 0)),
           sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end7 >= oks_delinq_date, oks_balance, 0)),
           sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end7 - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end7 - oks_delinq_date > 30, oks_balance, 0)),
--         —обственные кредиты
           sum(iif(ks_kc_id = oks_kc_id, oks_balance, 0)),
           sum(iif(ks_kc_id = oks_kc_id and :rep_date_end7 >= oks_delinq_date, oks_balance, 0)),
           sum(iif(ks_kc_id = oks_kc_id and :rep_date_end7 - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(ks_kc_id = oks_kc_id and :rep_date_end7 - oks_delinq_date > 30, oks_balance, 0))
    from oksummary, kontract, kontstate
    where oks_region_id = :reg_id and
          :rep_date_end7 between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          :rep_date_end7 - oks_delinq_date <= 360 and
          oks_kc_id = :officer_id and
          oks_from_k_id = k_id and
          oks_from_k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end7 between ks_date_begin and ks_date_end
          and oks_close_state = 0
    into balance_day7_ago, balance_delinq_day7_ago, balance_delinq7_day7_ago, balance_delinq30_day7_ago,
         balance_not_own_day7_ago, balance_delinq_not_own_day7_ago, balance_delinq7_not_own_day7_ag,
         balance_delinq30_not_own_day7_a, balance_is_own_day7_ago, balance_delinq_is_own_day7_ago,
         balance_delinq7_is_own_day7_ago, balance_delinq30_is_own_day7_ag;

    balance_day7_ago = coalesce(balance_day7_ago, 0);
    balance_delinq_day7_ago = coalesce(balance_delinq_day7_ago, 0);
    balance_delinq7_day7_ago = coalesce(balance_delinq7_day7_ago, 0);
    balance_delinq30_day7_ago = coalesce(balance_delinq30_day7_ago, 0);
    balance_not_own_day7_ago = coalesce(balance_not_own_day7_ago, 0);
    balance_delinq_not_own_day7_ago = coalesce(balance_delinq_not_own_day7_ago, 0);
    balance_delinq7_not_own_day7_ag = coalesce(balance_delinq7_not_own_day7_ag, 0);
    balance_delinq30_not_own_day7_a = coalesce(balance_delinq30_not_own_day7_a, 0);
    balance_is_own_day7_ago = coalesce(balance_is_own_day7_ago, 0);
    balance_delinq_is_own_day7_ago = coalesce(balance_delinq_is_own_day7_ago, 0);
    balance_delinq7_is_own_day7_ago = coalesce(balance_delinq7_is_own_day7_ago, 0);
    balance_delinq30_is_own_day7_ag = coalesce(balance_delinq30_is_own_day7_ag, 0);

--  ƒл¤ динамики 30 дней

    balance_day30_ago = 0;
    balance_delinq_day30_ago = 0;
    balance_delinq7_day30_ago = 0;
    balance_delinq30_day30_ago = 0;
    balance_not_own_day30_ago = 0;
    balance_delinq_not_own_day30_ag = 0;
    balance_delinq7_not_own_day30_a = 0;
    balance_delinq30_not_own_day30_ = 0;
    balance_is_own_day30_ago = 0;
    balance_delinq_is_own_day30_ago = 0;
    balance_delinq7_is_own_day30_ag = 0;
    balance_delinq30_is_own_day30_a = 0;

    rep_date_end30 = rep_date_end - 30;
    select sum(oks_balance), sum(iif(:rep_date_end30 >= oks_delinq_date, oks_balance, 0)),
           sum(iif(:rep_date_end30 - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(:rep_date_end30 - oks_delinq_date > 30, oks_balance, 0)),
--         ѕереданные кредиты
           sum(iif(ks_kc_id <> oks_kc_id, oks_balance, 0)),
           sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end30 >= oks_delinq_date, oks_balance, 0)),
           sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end30 - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(ks_kc_id <> oks_kc_id and :rep_date_end30 - oks_delinq_date > 30, oks_balance, 0)),
--         —обственные кредиты
           sum(iif(ks_kc_id = oks_kc_id, oks_balance, 0)),
           sum(iif(ks_kc_id = oks_kc_id and :rep_date_end30 >= oks_delinq_date, oks_balance, 0)),
           sum(iif(ks_kc_id = oks_kc_id and :rep_date_end30 - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(ks_kc_id = oks_kc_id and :rep_date_end30 - oks_delinq_date > 30, oks_balance, 0))
    from oksummary, kontract, kontstate
    where oks_region_id = :reg_id and
          :rep_date_end30 between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          :rep_date_end30 - oks_delinq_date <= 360 and
          oks_kc_id = :officer_id and
          oks_from_k_id = k_id and
          oks_from_k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end30 between ks_date_begin and ks_date_end
          and oks_close_state = 0
    into balance_day30_ago, balance_delinq_day30_ago, balance_delinq7_day30_ago, balance_delinq30_day30_ago,
         balance_not_own_day30_ago, balance_delinq_not_own_day30_ag, balance_delinq7_not_own_day30_a,
         balance_delinq30_not_own_day30_, balance_is_own_day30_ago, balance_delinq_is_own_day30_ago,
         balance_delinq7_is_own_day30_ag, balance_delinq30_is_own_day30_a;

    balance_day30_ago = coalesce(balance_day30_ago, 0);
    balance_delinq_day30_ago = coalesce(balance_delinq_day30_ago, 0);
    balance_delinq7_day30_ago = coalesce(balance_delinq7_day30_ago, 0);
    balance_delinq30_day30_ago = coalesce(balance_delinq30_day30_ago, 0);
    balance_not_own_day30_ago = coalesce(balance_not_own_day30_ago, 0);
    balance_delinq_not_own_day30_ag = coalesce(balance_delinq_not_own_day30_ag, 0);
    balance_delinq7_not_own_day30_a = coalesce(balance_delinq7_not_own_day30_a, 0);
    balance_delinq30_not_own_day30_ = coalesce(balance_delinq30_not_own_day30_, 0);
    balance_is_own_day30_ago = coalesce(balance_is_own_day30_ago, 0);
    balance_delinq_is_own_day30_ago = coalesce(balance_delinq_is_own_day30_ago, 0);
    balance_delinq7_is_own_day30_ag = coalesce(balance_delinq7_is_own_day30_ag, 0);
    balance_delinq30_is_own_day30_a = coalesce(balance_delinq30_is_own_day30_a, 0);

    if (officer_id > 1000000000004) then
      is_officer = 1;
    else
      is_officer = 0;

    suspend;
  end
end
