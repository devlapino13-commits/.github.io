CREATE PROCEDURE HP_KLADR_GET_RES_ONCE(
    PARENT_PKI_ID D_ID = null,
    KLADR_NAME_IN D_STR_LONG = null,
    PL_NAME_IN D_STR_LONG = null)
RETURNS (
    KLADR_NAME D_STR_LONG,
    KLADR_STATUS D_STR_CODE,
    PL_NAME D_STR_LONG,
    KLADR_CODE D_STR_CODE,
    PL_CODE D_STR_CODE,
    KLADR_OKATO D_STR_CODE,
    PL_OKATO D_STR_CODE,
    STATUS D_STR_CODE,
    COMM D_STR_LONG)
AS
declare variable pki_id d_id;
declare variable state d_integer;
begin
  -- процедура выводит результаты одноразовой полной сверки КЛАДР и БД
  for select pki_id, pki_name || ' ' || pki_type, kst_pl_name || ' ' || kst_pl_type, pki_code,
      substring(kst_pl_code from 4), pki_okatd, pl_add_code, kst_status, kst_comm
  from places_kladr_import
  left join kladr_sync_tmp on kst_from_pki_id = pki_id + 0
  left join places on pl_id = kst_from_pl_id
  where ((:parent_pki_id is null and pki_parent is null) or pki_parent = :parent_pki_id)
  into pki_id, kladr_name, pl_name, kladr_code, pl_code, kladr_okato, pl_okato, status, comm
  do
  begin
    kladr_name = coalesce(kladr_name_in || ' / ', '') || kladr_name;
    pl_name = coalesce(pl_name_in || ' / ', '') || pl_name;

    state = right(kladr_code, 2);
    kladr_status = case
                     when state = 0 then 'Актуальный'
                     when state < 51 then 'Переименован'
                     when state = 51 then 'Переподчинен'
                     when state < 99 then 'Резерв'
                     else 'Не существует'
                   end;

    suspend;

    for select kladr_name, kladr_status, pl_name, kladr_code, pl_code, kladr_okato, pl_okato, status, comm
    from hp_kladr_get_res_once(:pki_id, :kladr_name, :pl_name)
    into :kladr_name, :kladr_status, :pl_name, :kladr_code, :pl_code, :kladr_okato, :pl_okato, :status, :comm
    do
      suspend;
  end

  if (parent_pki_id is not null) then
    exit;

  kladr_name = null;
  kladr_code = null;
  kladr_okato = null;
  kladr_status = null;
  status = 'Не найден в КЛАДР';

  for with recursive t1 (lvl, id, pl_parent, name, code, add_code) as (
    select 0, pl_id, parent, pl_name || ' ' || plt_short_name, substring(pl_code from 4), pl_add_code
  from places, placetype
  where pl_status = plt_id
    and plt_can_select < 2
    and parent = 810000000001
  union all
  select lvl + 1, pl_id, parent, name || ' / ' || pl_name || ' ' || plt_short_name, substring(pl_code from 4), pl_add_code
  from t1, places, placetype
  where parent = id
    and pl_status = plt_id
    and plt_can_select < 2
  )
  select name, code, add_code
  from t1, kladr_sync_tmp
  where kst_from_pl_id = id
    and kst_from_pki_id is null
  into pl_name, pl_code, pl_okato
  do
  begin
    comm = '';

    with recursive t1 (lvl, parent, name) as(
      select 0, pki_parent, pki_name || ' ' || pki_type
      from places_kladr_import
      where pki_okatd = :pl_okato
    union all
      select lvl + 1, pki_parent, pki_name || ' ' || pki_type || '/' || name
      from t1, places_kladr_import
      where pki_id = parent
    )
    select 'Совпадение по ОКАТО с ' || name
    from t1
    order by lvl desc
    rows 1
    into comm;

    suspend;
  end
end
