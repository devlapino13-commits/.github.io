CREATE PROCEDURE FM_SEND_TO_PKB(
    MA_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable C_FIO D_STR_100;
declare variable C_INN D_STR_20;
declare variable PASS_NO D_STR_20;
declare variable EIC_ID D_ID;
declare variable C_ID D_ID;
declare variable EIC_HAS_API D_SMALLINT;
declare variable EIC_INFO_ACTUAL_DAYS D_INTEGER;
declare variable EIC_CODE D_STR_20;
declare variable HAS_CREDIT_HISTORY D_SMALLINT;
begin
  res_text = '';

  select mac.mac_from_c_id
  from miniappclient mac
  where mac.mac_from_ma_id = :ma_id and
        mac.mac_amount > 0
  into :c_id;

  if (f_get_db_id() <> 39801) then
  begin
    suspend;
    exit;
  end

  if (not exists(select 1
                 from extinfolist eil
                 where eil.eil_from_c_id = :c_id and
                       eil.eil_object_kind = 11 and
                       eil.eil_object_id = :ma_id and
                       eil.eil_status = 2 and
                       eil.eil_from_eic_id = 11))
    then
    begin
      res_text = f_lang('kz', 'ЗАПРОС НЕ ОТПРАВЛЕН! Доступные отчеты еще не получены по клиенту. Повторите попытку через 3 минуты');
      suspend;
      exit;
    end

  select priority_id
  from hp_get_priority_rep_id(f_objkind('miniapp', 'id'), :ma_id)
  into :has_credit_history;

  if (has_credit_history = -1) then
  begin
    res_text = f_lang('kz', 'У клиента отсутствует кредитная история. Проходите дальше по процессу');
    suspend;
    exit;
  end 

  select eic_id,
         eic_has_api,
         eic_info_actual_days,
         eic_code
  from extinfocfg
  where eic_code = 'pkb_cib_report'
  into :eic_id,
       :eic_has_api,
       :eic_info_actual_days,
       :eic_code;

  if (eic_id is null) then
    exception e_default 'ExtinfoCfg not found: ' || coalesce(:eic_code, '<NULL>');

  select c_inn,
         c_pass_no,
         c_name_fio
  from clients
  where c_id = :c_id
  into c_inn,
       pass_no,
       c_fio;

  if (f_edoc_status('pers_data_consent', :ma_id,  current_date) <> 'signed') then
  begin
    res_text = f_lang('kz', 'Согласие на обработку перс. данных не подписан');
    suspend;
    exit;
  end

  if ((char_length(coalesce(c_inn, '')) <> 12) or
      (coalesce(pass_no, '') = '')) then
  begin
    res_text = f_lang('kz', 'Проверьте паспортные данные у ') || :c_fio;
    suspend;
    exit;
  end


  insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json, eil_from_eip_id, eil_from_un_id)
  values (:c_id, :eic_id, current_date, f_objkind('miniapp', 'id'), :ma_id, :eic_has_api, null, :un_id);

  res_text = f_lang('kz', 'Данные отправлены в ПКБ для проверки');
  suspend;
end
