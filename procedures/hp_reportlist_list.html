CREATE PROCEDURE HP_REPORTLIST_LIST
RETURNS (
    RL_ID INTEGER,
    PARENT INTEGER,
    LB INTEGER,
    RB INTEGER,
    RL_NAME_RUS VARCHAR(255),
    RL_HINT_RUS VARCHAR(1000),
    RL_NAME_ENG VARCHAR(255),
    RL_HINT_ENG VARCHAR(1000),
    RL_TYPE SMALLINT,
    RL_HOTKEY VARCHAR(20),
    RL_DATA_RUS BLOB SUB_TYPE BINARY SEGMENT SIZE 256,
    RL_DATA_ENG BLOB SUB_TYPE BINARY SEGMENT SIZE 256,
    RL_IS_READY SMALLINT,
    RL_USER_ID DECIMAL(18,0),
    RL_DATE_KIND SMALLINT,
    RL_REPORT_KIND INTEGER,
    RL_SQL VARCHAR(1024),
    RL_FILTER BLOB SUB_TYPE BINARY SEGMENT SIZE 256,
    RL_ORDER_NO INTEGER,
    RL_DATETIME TIMESTAMP,
    RES_READONLY D_BOOL,
    RES_UX_MEMO D_STR_255)
AS
declare variable ux_uuid d_uuid;
declare variable mm_uuid d_uuid;
declare variable uxa_uuid d_uuid;
begin
  mm_uuid = uxf$mainmenu();

  for select rl_id,
             parent,
             lb,
             rb,
             rl_name_rus,
             rl_hint_rus,
             rl_name_eng,
             rl_hint_eng,
             rl_type,
             rl_hotkey,
             rl_data_rus,
             rl_data_eng,
             rl_is_ready,
             rl_user_id,
             rl_date_kind,
             rl_report_kind,
             rl_sql,
             rl_filter,
             rl_order_no,
             rl_datetime,
             rl_ux_imported
      from reportlist
      into :rl_id,
           :parent,
           :lb,
           :rb,
           :rl_name_rus,
           :rl_hint_rus,
           :rl_name_eng,
           :rl_hint_eng,
           :rl_type,
           :rl_hotkey,
           :rl_data_rus,
           :rl_data_eng,
           :rl_is_ready,
           :rl_user_id,
           :rl_date_kind,
           :rl_report_kind,
           :rl_sql,
           :rl_filter,
           :rl_order_no,
           :rl_datetime,
           :res_readonly
  do
  begin
    res_ux_memo = '';

    if (res_readonly = 1) then
    begin
      for select ufm_ux_uuid
      from ux$report, ux$form
      where urt_import_kind = 1 and
            urt_import_id = :rl_id and
            urt_uuid = ufm_uuid
      order by urt_import_datetime desc
      rows 1
      into :ux_uuid
      do
      begin
        select uxa_uuid, 'Существует копия для FinStart2: ' || uxa_fcode || ',  ' || uxa_name
        from ux$alias
        where uxa_owner = :ux_uuid
        rows 1
        into uxa_uuid, res_ux_memo;

        if (exists(select ua_uuid
        from ux$action, ux$menulist
        where ua_owner = :mm_uuid and
              ua_uxa_uuid = :uxa_uuid and
              uml_ua_uuid = ua_uuid and
              uml_fs_disabled = 1)) then
          res_readonly = 0;
      end
    end

    suspend;
  end
end
