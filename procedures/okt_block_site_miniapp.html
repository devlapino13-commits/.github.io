CREATE PROCEDURE OKT_BLOCK_SITE_MINIAPP(
    OPERATOR STR_100,
    ESM_ID ID_TYPE,
    ACTION_TYPE SMALLINT,
    INPUT_CK_ID INTEGER)
RETURNS (
    MESSAGE_TEXT STR_100,
    RETURN_CK_ID INTEGER,
    BLANK_TYPE SMALLINT)
AS
declare variable blocked_operator str_100;
declare variable ck_status integer;
declare variable parent_id integer;
begin
  /*
  ACTION_TYPE = 1 - Заблокировать
  ACTION_TYPE = 2 - Разблокировать
  ACTION_TYPE = 3 - Разблокировать всех, у кого блокировока висит 30 минут и более
  */

  operator = coalesce(operator, '');
  message_text = '';
  return_ck_id = 0;

  if (operator <> '') then
  begin
    if (action_type = 1) then -- если блокировка
    begin
-- полчаю последний результат звонка
      select chk.ck_blocked_oper, chk.ck_id, chk.ck_status
      from cc_check_knclient chk
      where chk.ck_type = 6 -- заявки с сайта
            and
            chk.ck_from_esm_id = :esm_id and
            chk.ck_status <> -1 -- не дозвонилися, не считаю
      order by chk.ck_id desc
      rows 1
      into :blocked_operator, return_ck_id, :ck_status;

      blocked_operator = coalesce(blocked_operator, '');
      return_ck_id = coalesce(return_ck_id, 0);
      ck_status = coalesce(ck_status, 0);

      if (ck_status in (-2, 2)) then
      begin
        message_text = 'Контакт уже обработан. Блокировка отменена';
        return_ck_id = 0;
      end

      if ((blocked_operator <> operator) and
          (blocked_operator <> '')) then
      begin
        message_text = 'Контакт заблокирован ' || blocked_operator;
        return_ck_id = 0;
      end

      parent_id = null;

      if (ck_status = 0) then
        blank_type = 1; -- если клиенту еще не звонили, открываю форму первого диалога
      else
      if (ck_status = 1) then
      begin
        blank_type = 2; -- если клиенту ранее звонили открываю форму второго диалога
        parent_id = return_ck_id;
      end

      if ((message_text = '') and
          (ck_status in (0, 1))) then
      begin
        if (blocked_operator = '') then
        begin
          insert into cc_check_knclient (ck_type, ck_blocked_datetime, ck_blocked_oper, ck_from_esm_id, ck_status,
                                         ck_parent_id)
          values (6, current_timestamp, :operator, :esm_id, 0, :parent_id)
          returning ck_id
          into :return_ck_id;
        end
      end
    end
    else
    if (action_type = 2) then
    begin
      delete from cc_check_knclient chk
      where chk.ck_type = 6 -- заявки с сайта
            and
            chk.ck_blocked_oper = :operator and
            chk.ck_checked_date is null and
            chk.ck_id = :input_ck_id and
            chk.ck_status not in (-2, 2);
    end
    else
    if (action_type = 3) then
    begin
      delete from cc_check_knclient chk
      where chk.ck_type = 6 -- заявки с сайта
            and
            chk.ck_checked_date is null and
            chk.ck_status = 0 -- не обработанная
            and
            ((current_timestamp - chk.ck_blocked_datetime) > 0.041666667); -- если разница между текущей датой/временем и временем блокировки больше 1 часа
    end
  end
  else
    message_text = 'Оператор не определен';

  suspend;
end
