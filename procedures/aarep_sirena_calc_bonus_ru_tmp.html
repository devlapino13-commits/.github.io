CREATE PROCEDURE AAREP_SIRENA_CALC_BONUS_RU_TMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_TXT STR_100,
    CC_PROJECT STR_100,
    OPER_CALLED STR_100,
    PAYED_SUM_ALL_VIPS CURR_TYPE,
    PAYED_SUM_ALL_SIRENA CURR_TYPE,
    PAYED_SUM_ALL CURR_TYPE,
    PAYED_QTY INTEGER,
    CONTRACTS_BEGIN INTEGER,
    CONTRACTS_RECEIPTS INTEGER,
    CONTRACTS_CUR_UNSUCCESSFUL INTEGER,
    CONTRACTS_RECALLED INTEGER,
    EFFICIENCY PERC_TYPE,
    BONUS_FOLDER_SVZ CURR_TYPE,
    BONUS_FOLDER_SIRENA CURR_TYPE,
    BONUS_FOR_PERFORMANCE CURR_TYPE,
    BONUS_TO_ACCRUAL CURR_TYPE)
AS
declare variable oper_called_tmp str_100;
declare variable rep_project str_100;
declare variable k_calc varchar(2);
declare variable payed_sum curr_type;
declare variable project str_100;
declare variable cc_project_salary curr_type;
declare variable perc_recovery_svz perc_type;
declare variable perc_recovery_sirena perc_type;
declare variable for_efficiency curr_type;
declare variable recovery_plan curr_type;
declare variable pant_id decimal(18,0);
declare variable pant_from_k_id decimal(18,0);
declare variable k_id_tmp decimal(18,0);
declare variable pant_from_num_id integer;
declare variable pant_create_date_time timestamp;
declare variable pant_last_call_date date;
declare variable pant_cl_call_qty integer;
declare variable pant_cont_call_qty integer;
declare variable pant_date_begin date;
declare variable pant_date_end date;
declare variable ol_date_begin date;
declare variable ol_date_end date;
declare variable pant_project_type smallint;
declare variable pant_rd_project_type integer;
declare variable pant_from_un_id decimal(18,0);
declare variable pant_is_dismiss smallint;
begin
  /* Процедура создана для расчета бонусов по КЦ Сирена!
  Дата начало разработки: 11-июня 2018-года
  Разработчик: Рахманов Музаффар */

  cc_project_salary = 9000;
  perc_recovery_svz = 0; --Взыскание папок СВЗ
  perc_recovery_sirena = 0; --Взыскание папок Сирена 1
  recovery_plan = 0;

  rep_txt = 'Расчет бонусов за период с ' || --Дата начало(далее внизу)
  (select sval
   from datetostr(:rep_date_begin)) --
  || ' по ' || --Дата конца(далее внизу)
  (select sval
   from datetostr(:rep_date_end));

  rep_project = 'OKTELL.PROJECTS.SIRENA';
  for --
      with t1
      as (select distinct rd.rd_name, pant_id, pant_from_k_id, pant_from_num_id,
                          /*iif(PANT_OPER_CALLED = 'IGUSEV', 'zzzIGUSEV', PANT_OPER_CALLED)*/ pant_oper_called,
                          pant_create_date_time, pant_last_call_date, pant_cl_call_qty, pant_cont_call_qty,
                          pant_date_begin, pant_date_end, pant_project_type, pant_rd_project_type, pant_from_un_id,
                          pant_is_dismiss
          from pantera_kontracts pk, refdata rd, oksummary oks
          where pk.pant_date_begin between :rep_date_begin and :rep_date_end and
                pant_date_begin <= pant_date_end and
--          PK.PANT_OPER_CALLED = :OPER_CALLED and
                pk.pant_rd_project_type = rd.rd_id and
                rd.rd_code containing :rep_project and
                pant_oper_called <> 'IGUSEV' and
                not exists(select *
                           from sb_akt sba
                           where sba.sba_from_k_id = pant_from_k_id and
                                 sba_date_begin between pant_date_begin and pant_date_end and
                                 sba.sba_is_delete = 0 and
                                 sba.sba_from_sbl_id <> 0 and -- важно!
                                 sba.sba_status = 2)
                and oks_close_state = 0

          union all
          select distinct rd.rd_name, pant_id, pant_from_k_id, pant_from_num_id,
                          /*iif(PANT_OPER_CALLED = 'IGUSEV', 'zzzIGUSEV', PANT_OPER_CALLED)*/ pant_oper_called,
                          pant_create_date_time, pant_last_call_date, pant_cl_call_qty, pant_cont_call_qty,
                          pant_date_begin, pant_date_end, pant_project_type, pant_rd_project_type, pant_from_un_id,
                          pant_is_dismiss
          from pantera_kontracts pk, refdata rd, oksummary oks
          where :rep_date_begin - 1 between pant_date_begin and pant_date_end and
                pant_date_begin <= pant_date_end and
                pk.pant_rd_project_type = rd.rd_id and
                rd.rd_code containing :rep_project and
                pant_oper_called <> 'IGUSEV' and
                oks_close_state = 0 and
                not exists(select *
                           from sb_akt sba
                           where sba.sba_from_k_id = pant_from_k_id and
                                 sba_date_begin between pant_date_begin and pant_date_end and
                                 sba.sba_is_delete = 0 and
                                 sba.sba_from_sbl_id <> 0 and -- важно!
                                 sba.sba_status = 2))

      select distinct rd_name, pant_id, pant_from_k_id, pant_from_num_id,
                      /*iif(PANT_OPER_CALLED = 'IGUSEV', 'zzzIGUSEV', PANT_OPER_CALLED)*/ pant_oper_called,
                      pant_create_date_time, pant_last_call_date, pant_cl_call_qty, pant_cont_call_qty, pant_date_begin,
                      pant_date_end, pant_project_type, pant_rd_project_type, pant_from_un_id, pant_is_dismiss
      from t1
      order by pant_oper_called, pant_from_k_id, pant_date_begin
      into cc_project, pant_id, pant_from_k_id, pant_from_num_id, oper_called, pant_create_date_time,
           pant_last_call_date, pant_cl_call_qty, pant_cont_call_qty, pant_date_begin, pant_date_end, pant_project_type,
           pant_rd_project_type, pant_from_un_id, pant_is_dismiss
  do
  begin
    if (pant_date_end > rep_date_end) then
      pant_date_end = rep_date_end; --чтобы поймать неуспешные блокировки

    k_calc = 'no';
    oper_called_tmp = coalesce(oper_called_tmp, oper_called);
    if (oper_called_tmp = oper_called) then
    begin
      if (pant_date_end - pant_date_begin < 8) then
      begin
        contracts_cur_unsuccessful = coalesce(contracts_cur_unsuccessful, 0) + 1;
      end
      else
      if (coalesce(k_id_tmp, 0) <> pant_from_k_id) then
      begin
        k_id_tmp = pant_from_k_id;
        if (:rep_date_begin - 1 between pant_date_begin and pant_date_end) then
        begin
          contracts_begin = coalesce(contracts_begin, 0) + 1;
          k_calc = 'ok';
        end

        if (k_calc = 'no' and
            pant_date_begin between :rep_date_begin and :rep_date_end) then
        begin
          contracts_receipts = coalesce(contracts_receipts, 0) + 1;
          k_calc = 'ok';
        end
      end

      if (k_calc = 'ok') then
      begin
        ol_date_begin = iif(pant_date_begin < rep_date_begin, rep_date_begin, pant_date_begin);
        ol_date_end = iif(pant_date_end > rep_date_end, rep_date_end, pant_date_end);

-- проверяю висела ли группа на ВС Сирена, на момент погашения или нет
        select iif(kal.kal_new_kf_officer = 1000000000004, 'Сирена', 'СВЗ')
        from kontract k, knaktlist kal
        where k_id = :pant_from_k_id and
              k_from_kf_id = kal.kal_from_kf_id and
              :rep_date_begin between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0
        into :project;

        payed_sum = 0;
        select -sum(ok_os + ok_perc + ok_fine)
        from operkred ok
        where ok.ok_from_k_id = :pant_from_k_id and
              ok_ol_date between :ol_date_begin and :ol_date_end and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              not exists(select *
                         from sb_akt sba
                         where sba.sba_is_delete = 0 and
                               sba.sba_from_k_id = ok_from_k_id and
                               ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                               sba.sba_from_sbl_id <> 0 and -- важно!
                               sba.sba_status = 2) -- если проект Сирена, группа не должна висеть на СБ в момент погашения
        into payed_sum;
        payed_sum = coalesce(payed_sum, 0);

        if (payed_sum > 0) then
        begin
          payed_qty = coalesce(payed_qty, 0) + 1;
          if (project = 'Сирена') then
            payed_sum_all_sirena = coalesce(payed_sum_all_sirena, 0) + payed_sum;
          else
            payed_sum_all_vips = coalesce(payed_sum_all_vips, 0) + payed_sum;

          payed_sum_all = coalesce(payed_sum_all, 0) + payed_sum;
        end
      end
    end
    else
    begin
      oper_called_tmp = oper_called;
      suspend;

      bonus_folder_svz = 0; -- Бонус за папки СВЗ
      bonus_folder_sirena = 0; --Бонус за папки Сирены
      bonus_for_performance = 0; --Бонус за эффективность
      bonus_to_accrual = 0; --Бонус к начислению

      contracts_begin = 0; --Кол-во договоров на начало
      contracts_receipts = 0; --Поступления договоров
      contracts_cur_unsuccessful = 0; --Текущие неуспешные
      contracts_recalled = 0; --Отозванные в СБ (без текущих)
      payed_sum_all = 0;
      payed_sum_all_sirena = 0;
      payed_sum_all_vips = 0;
    end
  end
  suspend;
end
