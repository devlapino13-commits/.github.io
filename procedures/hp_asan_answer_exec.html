CREATE PROCEDURE HP_ASAN_ANSWER_EXEC(
    EIL_ID D_ID)
AS
declare variable UN_VIRT D_ID;
declare variable SURNAME D_STR_255;
declare variable NAME D_STR_255;
declare variable PATRONYMIC D_STR_255;
declare variable MARITAL_STATUS D_STR_100;
declare variable PIN D_STR_20;
declare variable BIRTH_DATE D_DATE;
declare variable GENDER D_STR_10;
declare variable PASSPORT_SERIES D_STR_10;
declare variable PASSPORT_NUMBER D_STR_20;
declare variable ISSUED_DATE D_DATE;
declare variable PASSPORT_AUTHORITY D_STR_255;
declare variable EXPIRED_DATE D_DATE;
declare variable ERR_TEXT D_STR_255;
declare variable C_FAMILY D_INTEGER;
declare variable BIRTH_CITY D_STR_255;
declare variable BIRTH_CITY_TMP D_STR_255;
declare variable C_ID D_ID;
declare variable C_NAME_FIO D_STR_255;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable ADL_ID D_ID;
declare variable REGISTRATION_ADDRESS D_BLOB_TEXT;
declare variable PASSPORT_WHO D_STR_255;
declare variable FL_ID D_ID;
declare variable FL_ID_ENG D_ID;
begin
  un_virt = 1000000000000;

  for select cl.c_id, cl.c_name_fio, eil.eil_from_bls_id
      from extinfolist eil
      inner join clients cl on cl.c_id = eil.eil_from_c_id
      where eil.eil_id = :eil_id
      into :c_id, :c_name_fio, :bls_id
  do
  begin
    select tg.json_text
    from temp_grs tg
    where tg.bls_id = :bls_id
    into json_text;
    if (coalesce(json_text, '') = '') then
    begin
      select blx_blob
      from hp_get_blobex_blob(:bls_id)
      into json_text;
    end
    surname = f_json_str_get_value(:json_text, 'Surname', '');
    name = f_json_str_get_value(:json_text, 'Name', '');
    patronymic = f_json_str_get_value(:json_text, 'Patronymic', '');
    birth_city_tmp = f_json_str_get_value(:json_text, 'BirthAddress', '');
    marital_status = f_json_str_get_value(:json_text, 'MaritalStatus', '');
    pin = f_json_str_get_value(:json_text, 'PIN', ''); -- Персональный идентификационный номер для АЗ
    birth_date = f_json_str_get_value(:json_text, 'BirthDate', ''); -- Дата рождения
    gender = f_json_str_get_value(:json_text, 'Gender', ''); -- Пол. Возможные значения "M" - муж и "F" - жен
    passport_series = f_json_str_get_value(:json_text, 'DocumentSeria', ''); -- Серия паспорта, пример "ID"
    passport_number = f_json_str_get_value(:json_text, 'DocumentNumber', ''); --  Номер паспорта, пример "1501045"
    passport_who = f_json_str_get_value(:json_text, 'GivenOrganization', '');
    issued_date = cast(left(f_json_str_get_value(:json_text, 'GivenDate', ''), 10) as date); -- Дата выдачи паспорта
    expired_date = cast(left(f_json_str_get_value(:json_text, 'ExpireDate', '31.12.9999'), 10) as date); -- Дата выдачи паспорта
    passport_authority = cast(f_json_blob_get_value(:json_text, 'GivenOrganization', '') as d_blob_text); -- Орган выдавший паспорт
    registration_address = cast(f_json_blob_get_value(:json_text, 'RegistrationAddress', '') as d_blob_text);
    if (left(birth_city_tmp, 10) = 'AZƏRBAYCAN') then
      birth_city = right(:birth_city_tmp, char_length(:birth_city_tmp) - 11);
      else birth_city = birth_city_tmp;
    c_family = case :marital_status
                 when 'evli' then f_get_rd_id('family.married')
                 when 'subay' then f_get_rd_id('family.unmarried')
                 when 'boşanmış' then f_get_rd_id('family.divorced')
                 when 'dul' then f_get_rd_id('family.widower')
                 else 0
               end;
    if (registration_address = '') then
    begin
      update grs_clients gc
      set gc.gc_status = 3
      where gc.gc_from_c_id = :c_id and
            gc.gc_status = 2 and
            gc.gc_is_last = 1 and
            gc.gc_from_eil_id = :eil_id;

      update addrclient adc
      set adc.adc_from_adl_id = 41701000000000
      where adc.adc_address_type = 1 and
            adc.adc_from_c_id = :c_id;
    end
    else
    begin
      select adl_id
      from hp_asan_addr_save(:registration_address)
      into adl_id;

      update or insert into addrclient (adc_from_c_id, adc_address_type, adc_from_adl_id, adc_from_un_id)
      values (:c_id, 1, :adl_id, :un_virt)
      matching (adc_from_c_id, adc_address_type);

      update or insert into grs_addrlist (ga_from_eil_id, ga_from_adl_id) -- RFCAZ-528
      values (:eil_id, :adl_id)
      matching (ga_from_eil_id);
    end

    select fl_id, fl_id_eng
    from hp_prepare_fio(:surname, :name, :patronymic, null, null, null)
    into :fl_id, fl_id_eng;

    update clients c
    set c.c_name_fio = f_correct_fio(replace(coalesce(:surname, ''), 'I', 'ı')) || ' ' || f_correct_fio(replace(coalesce(:name, ''), 'I', 'ı')) || ' ' || f_correct_fio(replace(coalesce(:patronymic, ''), 'I', 'ı')),
        c.c_sex = iif(:gender = 'Kişi', 1, 0),
        c.c_birth_date = :birth_date,
        c.c_pass_no = :passport_series || :passport_number,
        c.c_pass_date = :issued_date,
        c.c_pass_date_expire = :expired_date,
        c.c_family = :c_family,
        c.c_index_c_pass_no = :passport_series || :passport_number,
        c.c_birth_city = :birth_city,
        c.c_birth_country = 'Азербайджан',
        c.c_pass_who = :passport_who,
        c.c_from_fl_id = :fl_id,
        c.c_name_eng_fl_id = :fl_id_eng
    where c.c_id = :c_id and
          c.c_pincode = :pin;

    update clientdocs cds
    set cds.cds_create_datetime = current_timestamp
    where cds.cds_from_c_id = :c_id and
          cds.cds_is_main_doc = 1 and
          cds.cds_is_delete = 0;

    execute procedure hp_objectprops_set_with_rd(f_objkind('clients', 'id'), :c_id, -1,
        f_get_rd_id('AnketaQuestions.Family'), current_date, '', 0, :c_family, :un_virt);
  end
end
