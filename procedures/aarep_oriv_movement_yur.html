CREATE PROCEDURE AAREP_ORIV_MOVEMENT_YUR(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SBL_ID INTEGER)
RETURNS (
    SUM_ORIV_DATE CURR_TYPE,
    BALANCE_BEGIN CURR_TYPE,
    BALANCE_END CURR_TYPE,
    ENROLLED_SUMM CURR_TYPE,
    REPAY_SUM_PERIOD CURR_TYPE,
    K_KONTRACT_NO D_STR_20,
    SBL_NAME D_STR_100,
    SBA_DATE_BEGIN D_DATE,
    C_NAME_FIO D_STR_100,
    REP_LASTPAY_DATE D_DATE,
    OKO_OS CURR_TYPE,
    SUMMA_OKO_OS CURR_TYPE,
    OKO_SH_OS CURR_TYPE,
    K_STATUS D_STR_20,
    DATE_TO_ORIV D_DATE,
    IS_COURT SMALLINT,
    SBL_ID D_ID,
    ORIV_MEMO D_STR_255,
    ORIV_COMMENT D_STR_255,
    ORIV_MEMO_DATE D_DATE,
    ORIV_COMMENT_DATE D_DATE,
    K_DISB_DATE D_DATE,
    REG_NAME D_STR_100,
    KF_OFFICE_NAME D_STR_100,
    IS_ORIV D_STR_100,
    DEPARTMENT_NAME D_STR_100,
    QTY_DELINQ_DAYS D_INTEGER,
    NOTIFICATION D_STR_10)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable c_dolya curr_type;
declare variable date_writeoff d_date;
declare variable reg_id d_id;
declare variable reg_id_old d_id;
declare variable kf_id d_id;
begin

  for select sum_oriv_date, balance_begin, balance_end, enrolled_summ, repay_sum_period, k_kontract_no, sbl_name,
             sba_date_begin, c_name_fio, rep_lastpay_date, oko_os, summa_oko_os, oko_sh_os, k_status, date_to_oriv,
             is_court, sbl_id, oriv_memo, oriv_comment, oriv_memo_date, oriv_comment_date, k_disb_date, reg_name,
             kf_office_name, is_oriv, department_name, qty_delinq_days, notification
      from aarep_oriv_movement(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_sbl_id)
      into sum_oriv_date, balance_begin, balance_end, enrolled_summ, repay_sum_period, k_kontract_no, sbl_name,
           sba_date_begin, c_name_fio, rep_lastpay_date, oko_os, summa_oko_os, oko_sh_os, k_status, date_to_oriv,
           is_court, sbl_id, oriv_memo, oriv_comment, oriv_memo_date, oriv_comment_date, k_disb_date, reg_name,
           kf_office_name, is_oriv, department_name, qty_delinq_days, notification
  do
  begin
    if (:k_status <> 'спец. учёт' and
        exists(select *
               from oksummary oks
               inner join kontract k on k.k_id = oks.oks_from_k_id and
                     k.k_kontract_no = :k_kontract_no
               where oks.oks_close_state = 1 and
                     oks.oks_date_begin <= :rep_date_end and
                     oks.oks_date_end >= :rep_date_begin)) then
      continue;
    suspend;
  end
end
