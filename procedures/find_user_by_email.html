CREATE PROCEDURE FIND_USER_BY_EMAIL(
    EMAIL D_STR_100)
RETURNS (
    UN_ID D_ID,
    UN_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    JL_KIND D_STR_100)
AS
declare variable c_id d_id;
begin
    for select un_id,
               un.un_name,
               c_id,
               c_name_fio
    from hp_find_client_by_continfo(null, :email) t
    left join username un on un.un_from_c_id = t.c_id
    into :un_id,
         :un_name,
         :c_id,
         :c_name_fio
    do
    begin
        select jl.jl_kind
        from hr_pstatus hps
        inner join joblist jl on jl.jl_id = hps.ps_from_jl_id
        where current_date between ps_date_begin and ps_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            hps.ps_from_c_id =:c_id
        into :jl_kind;

        suspend;
    end
end
