CREATE PROCEDURE GET_ONE_KONTRACT_DATA(
    K_ID DECIMAL(18,0),
    KONT_NO VARCHAR(20),
    DD DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    RES_K_ID DECIMAL(18,0),
    K_STOP_PERC DATE,
    K_STOP_FINE DATE,
    K_BALANCE DECIMAL(18,4),
    K_DELINQ_DATE DATE,
    K_DELINQ_DAYS INTEGER,
    KS_STATE INTEGER)
AS
begin
  if (k_id > 0) then
  begin
    res_k_id = :k_id;
    select k_kontract_no, k_stop_perc, k_stop_fine, ks_state
    from kontract, kontstate
    where k_id = ks_from_k_id and
          k_id = :k_id and
          ks_date_begin <= :dd and
          ks_date_end >= :dd and
          ks_ol_is_delete = 0
    into :k_kontract_no, :k_stop_perc, :k_stop_fine, :ks_state;
  end
  else
  begin
    if ((kont_no <> '') and
        (kont_no is not null)) then
    begin
      k_kontract_no = :kont_no;
      select k_id, k_stop_perc, k_stop_fine, ks_state
      from kontract, kontstate
      where k_id = ks_from_k_id and
            k_kontract_no = :kont_no and
            ks_date_begin <= :dd and
            ks_date_end >= :dd and
            ks_ol_is_delete = 0
      into :res_k_id, :k_stop_perc, :k_stop_fine, :ks_state;
      if (res_k_id is null) then
        res_k_id = 0;
    end
  end
  if (res_k_id > 0) then
  begin
    k_balance = 0;
    k_delinq_date = null;
    select oks_balance, oks_delinq_date
    from oksummary
    where oks_from_k_id = :res_k_id and
          oks_date_begin <= :dd and
          oks_date_end >= :dd and
          oks_close_state = 0
    into :k_balance, :k_delinq_date;
    if (k_balance is null) then
      k_balance = 0;
    if (k_delinq_date <= :dd) then
    begin
      k_delinq_days = :dd - :k_delinq_date;
    end
    else
    begin
      k_delinq_days = 0;
      k_delinq_date = null;
    end
    suspend;
  end
end
