CREATE PROCEDURE HP_CASHTERMINAL_LOAD(
    IN_CIL_ID D_INTEGER,
    IN_CASH_ID D_INTEGER,
    IN_DATE D_DATE,
    IN_TM_ID D_INTEGER,
    IN_MANAGER_ID D_ID,
    UN_ID D_ID)
RETURNS (
    CTM_FROM_TM_ID D_INTEGER,
    CTM_DOC_NO D_INTEGER,
    CTM_PERIOD_BEGIN D_DATETIME,
    CTM_PERIOD_END D_DATETIME,
    CIL_FROM_BLS_ID D_ID,
    CIL_ID D_INTEGER,
    CIL_FROM_CSP_ID D_INTEGER,
    CIL_DATE D_DATE,
    CIL_INSPECTOR_ID D_ID,
    CIL_INV_AMOUNT D_CURRENCY,
    CFL_FACT_AMOUNT D_CURRENCY,
    CIL_INV_AMOUNT$BOND D_STR_LONG,
    CFL_FACT_AMOUNT$BOND D_STR_LONG,
    PERSON_ID D_ID,
    DIFF_ACCOUNT D_STR_CODE,
    RES_OFFICE_ID D_INTEGER,
    RES_DIFF_KIND D_SMALLINT)
AS
declare variable oper_kind d_integer;
declare variable cfl_id d_integer;
begin
  if (in_cil_id is null) then
    in_cil_id = 0;

  oper_kind = f_enum_id('cashoperkind', 'plus.inkassa');
  diff_account = '';
  person_id = null;

  res_diff_kind = 0;
  if (in_cil_id <> 0) then
  begin
    select m.cim_from_c_id
    from cashinvmembers m
    where cim_from_cil_id = :in_cil_id and
          cim_role_kind = -1 -- Ответственное лицо
    order by cim_is_fact desc
    rows 1
    into person_id;

    cil_id = in_cil_id;
    select cil_from_csp_id, cil_date, cil_inspector_id, cil_inv_amount, cil_office_id, cil_from_bls_id
    from cashinvlist
    where cil_id = :in_cil_id and
          cil_oper_kind = :oper_kind
    into :cil_from_csp_id, :cil_date, :cil_inspector_id, :cil_inv_amount, res_office_id, cil_from_bls_id;

    if (row_count = 0) then
      exception e_message 'Акт инкассации не найден, Id: ' || in_cil_id;

    cil_inv_amount$bond = '';

    if (cil_inv_amount <> 0) then
    begin
      select res_bond_list
      from hp_cashbond_load(:in_cil_id)
      into :cil_inv_amount$bond;
    end

    select ctm_from_tm_id, ctm_doc_no, ctm_period_begin, ctm_period_end
    from cashterminal
    where ctm_from_cil_id = :in_cil_id
    into :ctm_from_tm_id, :ctm_doc_no, :ctm_period_begin, :ctm_period_end;

    if (row_count = 0) then
      exception e_message 'Акт инкассации не найден, Id: ' || in_cil_id;

    select cil_id, cil_inv_amount
    from cashinvlist
    where cil_parent_id = :in_cil_id and
          cil_is_delete = 0
    into :cfl_id, :cfl_fact_amount;

    if (row_count = 0) then
    begin
      cfl_fact_amount = cil_inv_amount;
      cfl_fact_amount$bond = cil_inv_amount$bond;
      res_diff_kind = 2;
    end else
    begin
      select res_bond_list
      from hp_cashbond_load(:cfl_id)
      into :cfl_fact_amount$bond;

      if (cfl_fact_amount <> cil_inv_amount) then
        res_diff_kind = 1;
    end
  end else
  begin
    cil_id = 0;

    if (in_cash_id is null) then
      exception e_message 'Касса не указана';

    if (in_manager_id is null) then
      exception e_message 'Контролер не указан';
  
    if (in_date is null) then
      in_date = current_date;
  
    cil_from_csp_id = in_cash_id;
    cil_date = in_date;
    cil_inspector_id = in_manager_id;
    cil_inv_amount = 0;

    ctm_from_tm_id = in_tm_id;
    ctm_doc_no = null;
    ctm_period_begin = null;
    ctm_period_end = null;
    cil_from_bls_id = null;

    select dateadd(second, 1, tml_last_datetime)
    from termilast
    where tml_from_tm_id = :in_tm_id
    into ctm_period_begin;

    cfl_fact_amount = 0;

    select csp_cashier_id, csp_office_id
    from cashplace
    where csp_id = :in_cash_id
    into person_id, res_office_id;
  end

  suspend;
end
