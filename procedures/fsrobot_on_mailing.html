CREATE PROCEDURE FSROBOT_ON_MAILING(
    KIND VARCHAR(10))
AS
declare variable msg_lifetime_days integer;
declare variable res_subject varchar(255);
declare variable res_body blob sub_type 1 segment size 80;
declare variable res_maillist varchar(255);
declare variable msg_type integer;
declare variable rep_command_kind varchar(255);
declare variable rep_un_id decimal(18,0);
declare variable res_c_id decimal(18,0);
declare variable msg_sender smallint;
begin
-- Процедура вызывается из через сервис по графику (ежеминутно)
-- Параметры берет из таблицы FsRobot_EventList
-- Процедура готовит список уведомлений для отправки роботом по почте или sms (Event 'DoMailing')
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь

  for select distinct msg_type, res_maillist, res_subject, res_body, msg_lifetime_days, rep_command_kind, res_c_id,
                      max(rep_un_id), msg_sender
      from fsrobot_on_mailing_tmp(:kind)
      group by msg_type, res_maillist, res_subject, res_body, msg_lifetime_days, rep_command_kind, res_c_id, msg_sender
      into :msg_type, :res_maillist, :res_subject, :res_body, :msg_lifetime_days, :rep_command_kind, res_c_id,
           :rep_un_id, :msg_sender
  do
  begin
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id, msg_sender)
    values (:msg_type, :res_maillist, :res_subject, :res_body, :msg_lifetime_days, :rep_un_id, :rep_command_kind, 0,
            :res_c_id, :msg_sender);
  end

end
