CREATE PROCEDURE HP_GET_CLIENT_DOC_IMAGES(
    C_ID D_ID,
    INCLUDE_MAIN_PASSPORT D_SMALLINT,
    INCLUDE_PHOTO D_SMALLINT,
    OTHER_CDS_LIST D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_IMAGE_LIST D_STR_8100)
AS
declare variable doc_image d_str_long;
declare variable cds_id d_id;
declare variable c_name_fio d_str_100;
declare variable c_doc_type d_integer;
declare variable c_blob_photo_id d_id;
declare variable passport_cds_id d_id;
begin
  res_image_list = '';

  if (other_cds_list is null) then
    other_cds_list = '';

  if (include_photo is null) then
    include_photo = 0;

  if (include_main_passport is null) then
    include_main_passport = 0;

  select c_name_fio, c_doc_type, c_blob_photo_id
  from clients
  where c_id = :c_id
  into c_name_fio, c_doc_type, c_blob_photo_id;

  passport_cds_id = null;
  select cds_id
  from clientdocs
  where cds_from_c_id = :c_id and
        cds_from_pk_id = :c_doc_type and
        cds_is_delete = 0
  order by cds_create_datetime desc, cds_id desc
  rows 1
  into passport_cds_id;

  if (include_main_passport = 1 and coalesce(passport_cds_id, 0) <> 0) then
  begin
    doc_image = f_docs_item(passport_cds_id, un_id);
    if (res_image_list <> '') then
      res_image_list = res_image_list || ',';
    res_image_list = res_image_list || doc_image;
  end

  if (include_photo = 1 and coalesce(c_blob_photo_id, 0) > 0) then
  begin
    doc_image = f_image_item('blobstore', c_blob_photo_id, 'Фото', un_id);
    doc_image = f_format('{"caption": "Фото", "blobs":[%1]}', doc_image);

    if (res_image_list <> '') then
      res_image_list = res_image_list || ',';
    res_image_list = res_image_list || doc_image;
  end

  if (other_cds_list <> '') then
  begin
    cds_id = null;
    for select res_str
    from hp_split_str(:other_cds_list, ',')
    where res_str <> coalesce(:passport_cds_id, 0)
    into :cds_id
    do
    begin
      doc_image = f_docs_item(cds_id, un_id);

      if (char_length(res_image_list || ',' || doc_image) <= 8000) then
      begin
        if (res_image_list <> '') then
          res_image_list = res_image_list || ',';
  
        res_image_list = res_image_list || doc_image;
      end
    end
  end

  doc_image = f_json_make_pair('caption', c_name_fio, 'str');
  res_image_list = f_json_str_set_value('{' || doc_image || '}', 'items', '['||res_image_list||']');

  suspend;
end
