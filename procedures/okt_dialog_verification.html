declare variable kf_id id_type;
begin
  select cont.cc_language, coalesce(cont.cc_name, 'Не указано'), cont.cc_from_c_id,

         coalesce(cont.cc_svz_comments, ''), cont.cc_cell_no,
         iif(cont.cc_call_back is null, '', 'Повторный звонок')
  from contacts cont
  where cont.cc_id = :cc_id
  into :cc_language, :cc_name, :c_id, :cc_status, :adl_id, :cc_type, :cc_svz_comments, :cc_cell_no, :call_back_status;

  cc_address = 'Не указан';

  select kont.k_last_reg_id
  from knclient kncl
  inner join kontstate ks on kncl.kc_from_k_id = ks.ks_from_k_id
  inner join kontract kont on kont.k_id = ks.ks_from_k_id
  where ks.ks_ol_is_delete = 0 and
        ks.ks_state = 0 and
        current_date between ks.ks_date_begin and ks.ks_date_end and
        kncl.kc_amount_approved > 0 and

        kont.k_status <> -1

    where macl.mac_from_c_id = :c_id and
          macl.mac_is_poruch = 0 and
          macl.mac_amount > 0 and
          maa.ma_date_end = '31.12.9999' and
          maa.ma_is_active = 1

    into :kf_id;

    from knaktlist kal

    timezone = 6;
  else
  if (cc_cell_no starting('998')) then
    timezone = 5;
  else
  if (cc_cell_no starting('992')) then
    timezone = 5;
  else
    timezone = timezone_ru;

  timezone = coalesce(timezone, 0);

  from places pl
  where pl.pl_id = :pl_id
  into :city;

  from clients cl
  where cl.c_id = :c_id

  begin

    language_name = 'Русский';
  end

    from refdata rd
    where rd.rd_id = :cc_language
    into :language_code, :language_name;
  end
