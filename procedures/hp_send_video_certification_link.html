CREATE PROCEDURE HP_SEND_VIDEO_CERTIFICATION_LINK(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    LINK D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
begin
  -- RFCRU-5604 Интеграция видео-сертификации клиентов через Яндекс.Телемост в МП
  res_text = '';

  if (coalesce(left(:link, 27), '') <> 'https://telemost.yandex.ru/') then
    res_text = 'Некорректная ссылка';

  if (res_text = '') then
  begin
    -- Отправка ссылки
    if (obj_kind = 'miniapp') then
    begin
      for select mac.mac_from_c_id
      from miniappclient mac
      where mac.mac_from_ma_id = :obj_id and
            mac.mac_amount > 0
      into c_id
      do
      begin
        execute procedure mc_send_push('clients', :c_id, 'Видео-сертификация', f_format('Для начала видео-конференции перейдите по ссылке: %1', :link));
      end
    end
    else if (obj_kind = 'kontract') then
    begin
      for select kc.kc_from_c_id
      from knclient kc
      where kc.kc_from_k_id = :obj_id and
            kc.kc_amount_approved > 0
      into c_id
      do
      begin
        execute procedure mc_send_push('clients', :c_id, 'Видео-сертификация', f_format('Для начала видео-конференции перейдите по ссылке: %1', :link));
      end
    end

    -- Логируем отправку
    insert into video_certification_link_log (vcll_link, vcll_object_kind, vcll_object_id, vcll_from_un_id)
    values (:link, cast(f_objkind(:obj_kind, 'id') as d_integer), :obj_id, :un_id);

    res_text = 'Ссылка на Видео-сертификацию отправлена';
  end

  suspend;
end
