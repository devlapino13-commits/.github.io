CREATE PROCEDURE AAREP_SIRENA_OPER_DELINQ_GR(
    REP_DATE DATE)
RETURNS (
    OPER_CALLED STR_100,
    K_KONTRACT_NO STR_100,
    OKS_DELINQ_DAY INTEGER,
    KF_NAME STR_100,
    OFFICER STR_100,
    CC_PROJECT STR_100,
    BALANCE_KONT CURR_TYPE)
AS
declare variable k_id id_type;
declare variable kf_id id_type;
declare variable sb_type str_100;
begin
  for select pantk.pant_from_k_id, kont.k_kontract_no, kont.k_from_kf_id, pantk.pant_oper_called, rd.rd_name
      from pantera_kontracts pantk
      inner join kontract kont on kont.k_id = pantk.pant_from_k_id
      inner join refdata rd on pantk.pant_rd_project_type = rd.rd_id
      where :rep_date between pantk.pant_date_begin and pantk.pant_date_end
--and pantk.pant_project_type = 4 -- Сирена/Просрочки
            and
            rd.rd_code in ('OKTELL.Projects.Sirena', 'OKTELL.Projects.Sirena1', 'OKTELL.Projects.Sirena2') and
            pantk.pant_date_end >= pantk.pant_date_begin
      order by pant_oper_called
      into :k_id, k_kontract_no, :kf_id, :oper_called, :cc_project
  do
  begin
    oks_delinq_day = null;

    select (cast(:rep_date as date) - oks_delinq_date) oks_delinq_day
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          oks_close_state = 0 and
          :rep_date between oks_date_begin and oks_date_end and
          cast(:rep_date as date) - oks_delinq_date >= 0
    into oks_delinq_day;

    oks_delinq_day = coalesce(oks_delinq_day, 0);

    if (oks_delinq_day > 0) then
    begin
      select kal.kal_new_kf_name, iif(kal.kal_new_kf_officer = 1000000000004, 'КЦ-Сирена', 'СВЗ')
      from knaktlist kal
      where kal.is_delete = 0 and
            kal.kal_from_kf_id = :kf_id and
            :rep_date between kal.kal_date and kal.kal_date_end
      into :kf_name, officer;

      sb_type = null;

      select iif(sbl.sbl_type = 0, 'СБ', 'Коллектор')
      from sb_akt sba
      inner join sb_list sbl on sbl.sbl_id = sba.sba_from_sbl_id
      where sba.sba_is_delete = 0 and
            sba.sba_from_k_id = :k_id and
            :rep_date between sba.sba_date_begin and sba.sba_date_end and
            sba.sba_from_sbl_id <> 0
      into :sb_type;

      if (sb_type is not null) then
        officer = sb_type;

----added on 23/01/2019 by Aleksei
      balance_kont = 0;

      select sum(sh_calc_perc + zad_os)
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      into :balance_kont;

      suspend;
    end
  end
end
