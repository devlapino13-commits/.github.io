CREATE PROCEDURE HP_SUD_ADD_MEMO(
    SMR_ID D_ID NOT NULL,
    MEMO D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_TXT D_STR_255,
    RES_OK D_BOOL)
AS
declare variable sul_id d_id;
begin
  sul_id = (select smr_last_sul_id from sud_mainreq where smr_id = :smr_id);

  if (trim(coalesce(memo, '')) = '') then
  begin
    res_ok = 0;
    res_txt = 'Введите комментарий!';
    suspend;
    exit;
  end 

  if (sul_id is null) then
  begin
    res_ok = 0;
    res_txt = 'Что-то пошло не так.. (Не найден лог для smr_id = ' || smr_id || ')';
    suspend;
    exit;
  end 

  insert into sud_mainmemo (smm_from_sul_id, smm_memo, smm_from_un_id)
  values (:sul_id, :memo, :un_id);

  res_ok = 1;
  res_txt = 'Комментарий добавлен';
  suspend;
end
