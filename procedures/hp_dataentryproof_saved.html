CREATE PROCEDURE HP_DATAENTRYPROOF_SAVED(
    KI_ID D_INTEGER,
    CODE_LST D_STR_8100)
AS
declare variable bct_id d_id;
declare variable kc_id d_id;
begin
  delete from kk_income_cib where kic_from_ki_id = :ki_id;
  for
    select
      bct_id
    from hp_split_str(:code_lst, ',')
      inner join bki_catalog
        on bct_kind = 'incomeInfoSource'
          and bct_code = res_str
  into
    :bct_id
  do
  begin
    update or insert into kk_income_cib(
      kic_from_ki_id,
      kic_from_bct_id)
    values(
      :ki_id,
      :bct_id)
    matching(
      kic_from_ki_id,
      kic_from_bct_id);
  end

  kc_id = (
    select
      kc_id
    from kk_income
      inner join knclient
        on kc_from_k_id = ki_from_k_id
          and kc_from_c_id = ki_from_c_id
    where ki_id = :ki_id);
  if (kc_id > 0) then
    execute procedure hp_knclient_pdn_update(:kc_id, 0, null, null, null, null, null, 'kk_doc_income');
end
