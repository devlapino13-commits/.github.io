CREATE PROCEDURE HP_SENDINFOKZ_EDOC_CODE(
    MA_ID D_INTEGER,
    K_ID D_ID,
    KONT_STATE D_INTEGER)
RETURNS (
    EDC_CODE D_STR_CODE,
    RES_TXT D_STR_100)
AS
declare variable dt_curr d_date;
begin
  dt_curr = current_date;
  edc_code = '';
  res_txt = '';

  if (kont_state <= 0
    and exists(
      select *
      from hp_edoc_list(11, :ma_id, 1000000000000)
      where edl_enum_edoc_kind = 2
      and eds_status = 1))
    then edc_code = 'information_request1';
  else
  if (kont_state > 0
    and f_edoc_status('loan.application', :ma_id, :dt_curr) = 'signed'
    and f_edoc_status('loan.agreement', :k_id, :dt_curr) = 'signed')
    then edc_code = 'information_request2';
  else
    res_txt = 'По данной заявке нельзя сформировать документ.';
  suspend;
end
