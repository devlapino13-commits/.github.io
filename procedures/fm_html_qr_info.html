CREATE PROCEDURE FM_HTML_QR_INFO(
    UUID D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable OBJ_CODE D_STR_CODE;
declare variable EDS_ID D_ID;
declare variable OBJ_KIND D_INTEGER;
declare variable DOC_NO D_STR_CODE;
declare variable DOC_DATE D_DATE;
declare variable UN_NAME D_STR_CODE;
begin
  -- Выдает информацию по эл.подписи через переданный UUID
  res_text = '';

  select un_name
  from username
  where un_id = :un_id
  into :un_name;

  if (:un_name = 'fmtest') then
  begin
    select atd.html_content
    from apple_test_data atd
    where atd.report_type = 'QR_INFO'
    into :res_text;
    suspend;
    exit;
  end

  begin
    select obj_kind, obj_id
    from hp_uuid(:uuid)
    into :obj_kind, :eds_id;
  when any do
    begin
      res_text = 'Не электронная подпись: ' || uuid;
      suspend;
      exit;
    end
  end

  if (eds_id is null)
  then res_text = 'Неверная ссылка';
  else obj_code = f_objkind(obj_kind, 'name');

  if (res_text = '' and obj_code <> 'edoc_sign') then
    res_text = 'QR-код не содержит электронной подписи';

  if (res_text = '') then
  begin
    obj_code = '';
    for select edl_id, eds_from_c_id, eds_status, eds_enum_edoc_sign_kind, eds_status_datetime,
           edl_object_kind, edl_object_id, edl_blank_date,
           edl_status, edl_create_un_id, edl_create_datetime, edc_name, c_name_fio, un_name
    from edoc_sign, edoc_list, clients, edoc_cfg, username
    where eds_id = :eds_id and
          edl_from_edc_id = edc_id and
          eds_from_edl_id = edl_id and
          eds_from_c_id = c_id and
          edl_create_un_id = un_id
    as cursor edoc
    do
    begin
      obj_code = f_objkind(edoc.edl_object_kind, 'name');
      if (obj_code = 'com_list') then
      begin
        select k_kontract_no, k_disb_date
        from com_list, kontract
        where cml_id = edoc.edl_object_id and
              cml_object_id = k_id
        into doc_no, doc_date;
      end else
      if (obj_code = 'kontract') then
      begin
        select k_kontract_no, k_disb_date
        from kontract
        where k_id = edoc.edl_object_id
        into doc_no, doc_date;
      end
      else
      if (obj_code = 'miniapp') then
      begin
        select ma_no, ma_date_app
        from miniapp
        where ma_id = edoc.edl_object_id
        order by ma_number desc
        rows 1
        into doc_no, doc_date;
      end
      else
      if (obj_code = 'agreements') then
      begin
        select ag_agreement_no, ag_agreement_date
        from agreements
        where ag_id = edoc.edl_object_id
        into doc_no, doc_date;
      end
      else
      if (obj_code = 'clients') then
      begin
        doc_no = '';
        doc_date = edoc.edl_blank_date;
      end
      else
      if (obj_code = 'kontmatag') then
      begin
        select ag_agreement_no, ag_agreement_date
        from agreements
        inner join kontmatag on ag_id = kma_from_ag_id
        where kma_id = edoc.edl_object_id
        into doc_no, doc_date;
      end
      else
      if (obj_code = 'ft_anketa_list') then
      begin
        doc_no = '';
        doc_date = edoc.edl_blank_date;
      end
      else res_text = 'Неизвестный тип: ' || obj_code;

      if (res_text = '') then
      begin
        res_text = '<body>' ||
        'Пакет документов:<br><b>' || edoc.edc_name || '</b><br><br>' ||
        iif(obj_code in ('kontract','miniapp','agreements'),
          'Номер документа: <b>' || doc_no || '</b><br> Дата:<b>' || f_datetostr(doc_date) || '</b><br><br>', '') ||
        'Время создания документа: <b>' || f_datetimetostr(edoc.edl_create_datetime) || '</b><br>' ||
        'Пользователь: <b>' || edoc.un_name || '</b><br><br>' ||
        'Клиент: <b>' || edoc.c_name_fio || '</b><br>' ||
        'Время подписания: <b>' || f_datetimetostr(edoc.eds_status_datetime) || '</b><br><br>' ||
        'Подписанные документы: <br><ul>';
        for select edf_print_name
        from edoc_blank, edoc_frx
        where edb_from_edl_id = edoc.edl_id and
          edb_from_edf_id = edf_id
        as cursor edb do
          res_text = BLOB_APPEND(res_text, '<li>' || edb.edf_print_name || '<br>');
        res_text = BLOB_APPEND(res_text, '</ul></body>');
      end
    end
  end

  suspend;
end
