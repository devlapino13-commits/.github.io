CREATE PROCEDURE HP_DO_EXPRESSAPP_CANCEL(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable EXA_FROM_EXC_ID D_INTEGER;
declare variable EDL_ID D_ID;
declare variable EDOC_KIND D_INTEGER;
declare variable OBJ_KIND2 D_INTEGER;
declare variable OBJ_ID2 D_ID;
begin
  res_text = '';
  if (f_objkind(obj_kind, 'name') <> 'expressapp') then
    res_text = 'Отказать можно только до отправки предложения.';

  if (res_text = '') then
  begin
    select exa_from_exc_id, exa_from_edl_id
    from expressapp
    where exa_id = :obj_id
    into exa_from_exc_id, edl_id;

    if (exa_from_exc_id <= 30) then
    begin
      update expressapp
      set exa_final_status = -1,
          exa_from_un_id = :un_id
      where exa_id = :obj_id;

      if (edl_id is not null) then
      begin
        select edl_from_edc_id,
               edl_object_kind,
               edl_object_id
        from edoc_list
        where edl_id = :edl_id
        into edoc_kind,
             obj_kind2,
             obj_id2;
        execute procedure hp_edoc_cancel(:edoc_kind, :obj_kind2, :obj_id2);
      end

      res_text = 'Предодобренная заявка успешно аннулирована.';
    end else
      res_text = 'Отказать можно только до отправки предложения.';
  end

  suspend;
end
