CREATE PROCEDURE HP_BIOGUARD_SAVE_RESPONSE(
    SESSION_ID D_STR_100 NOT NULL,
    JSON_INFO D_BLOB_TEXT NOT NULL,
    IMG_FILE D_BLOB_BINARY NOT NULL)
RETURNS (
    EIL_ID D_ID,
    BR_ID D_ID,
    CLIENT_PHOTO_BLS_ID D_ID)
AS
declare variable BIOGUARD_EIC_ID D_ID;
begin
    select eic.eic_id
    from extinfocfg eic
    where eic.eic_code = 'bioguard_info'
    into :bioguard_eic_id;

    if (bioguard_eic_id is null) then
        exception e_default 'Extinfocfg "bioguard_info" does not exists';

    -- Сохраняем результат полученный от сервиса Bio-Guard
    insert into bioguard_results (br_session_id, br_datetime, br_status)
    values (:session_id, current_timestamp, f_get_rd_id('BioguardStatus.IN_PROCESS'))
    returning br_id into :br_id;
    
    insert into extinfolist (eil_from_eic_id, eil_status, eil_is_json, eil_ext_id)
    values (:bioguard_eic_id,  0, 1, :session_id)
    returning eil_id into :eil_id;

    execute procedure hp_extinfo_set_result(:eil_id, :json_info);

    if (coalesce(img_file, '') <> '') then
    begin
        execute procedure hp_extinfo_set_blob(:eil_id, 'bioguard_client_photo', :img_file);

        select eib.eib_from_bls_id
        from extinfolistbls eib
        where eib.eib_from_eil_id = :eil_id
        into :client_photo_bls_id;
    end

    suspend;
end
