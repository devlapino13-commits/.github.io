CREATE PROCEDURE UXP$UXFRAME_INFO(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    UFR_UUID D_UUID,
    UX_CODE D_STR_100,
    UX_NAME D_STR_100,
    UFR_CODE D_STR_CODE,
    UFR_US_CODE D_STR_CODE,
    UFR_IS_GRID D_BOOL,
    UFR_TREE_PARENT_FIELD D_STR_CODE,
    UFR_TREE_IMAGE_FIELD D_STR_CODE,
    UFR_CAN_UPDATE D_BOOL,
    UFR_CAN_INSERT D_BOOL,
    UFR_CAN_DELETE D_BOOL,
    US_UPDATE D_STR_CODE,
    US_INSERT D_STR_CODE,
    US_DELETE D_STR_CODE,
    UFR_HIDE_ON_GROUP D_BOOL,
    UFR_GROUP_PANEL D_BOOL,
    UFR_FIND_PANEL D_SMALLINT,
    UFR_HIDE_SUMMARY D_BOOL,
    UFR_AUTOWIDTH D_SMALLINT,
    UFR_SQL_WHERE D_STR_LONG,
    UFR_SQL_ORDERBY D_STR_LONG,
    US_SQL D_STR_8100,
    US_IS_SQL D_BOOL,
    UFR_VISIBLE_CODE D_STR_CODE,
    UFR_INPLACE_UUID D_UUID,
    EXP_EXPRESSION D_STR_4096,
    RES_SOURCE_UUID D_UUID,
    UFR_REFRESH_PARAMS D_STR_LONG,
    UFR_TREE_POSITION_FIELD D_STR_CODE,
    UFR_TREE_CAN_DRAG_FIELD D_STR_CODE,
    UFR_TREE_CAN_DROP_FIELD D_STR_CODE,
    UFR_ADD_TABLE_FIELDS D_STR_CODE,
    RES_TREE_MOVE_SQL D_STR_8100,
    RES_SELECT_SQL D_STR_8100)
AS
begin
  ufr_uuid = ux_uuid;

  if (coalesce(ufr_uuid, '') = '') then
    ufr_uuid = uxf$uniq(ux_uuid_code);

  us_update = null;
  us_insert = null;
  us_delete = null;

  ufr_inplace_uuid = null;

  for select ufr_fcode,
             ufr_name,
             ufr_code,
             us_code,
             ufr_is_grid,
             ufr_can_update,
             ufr_can_insert,
             ufr_can_delete,
             ufr_tree_parent_field,
             ufr_tree_image_field,
             ufr_hide_on_group,
             ufr_group_panel,
             ufr_find_panel,
             ufr_hide_summary,
             ufr_autowidth,
             ufr_sql_where,
             ufr_sql_orderby,
             us_sql,
             us_uuid,
             ufr_refresh_params,
             ufr_tree_position_field,
             ufr_tree_can_drag_field,
             ufr_tree_can_drop_field,
             ufr_add_table_fields
      from ux$frame, ux$source
      where ufr_uuid = :ufr_uuid and
            ufr_us_uuid = us_uuid
      into :ux_code,
           :ux_name,
           :ufr_code,
           :ufr_us_code,
           :ufr_is_grid,
           :ufr_can_update,
           :ufr_can_insert,
           :ufr_can_delete,
           :ufr_tree_parent_field,
           :ufr_tree_image_field,
           :ufr_hide_on_group,
           :ufr_group_panel,
           :ufr_find_panel,
           :ufr_hide_summary,
           :ufr_autowidth,
           :ufr_sql_where,
           :ufr_sql_orderby,
           :us_sql,
           :res_source_uuid,
           :ufr_refresh_params,
           :ufr_tree_position_field,
           :ufr_tree_can_drag_field,
           :ufr_tree_can_drop_field,
           :ufr_add_table_fields
  do
  begin
    ufr_visible_code = ufr_us_code;

    if (ufr_add_table_fields is null) then
      ufr_add_table_fields = '';

    if (coalesce(us_sql, '') <> '') then
    begin
      us_is_sql = 1;
      ufr_visible_code = '<Select SQL>';
    end
    else
      us_is_sql = 0;

    res_select_sql = '';
    begin
      select sql_text
      from uxp$uxframe_sql_select(:ufr_uuid)
      into res_select_sql;

      when any do
        res_select_sql = '';
    end

    res_tree_move_sql = '';
    if (ufr_tree_position_field <> '') then
    begin
      select res_tree_move_sql
      from uxp$uxframe_sql_tree_move(:ufr_uuid, true)
      into :res_tree_move_sql;
    end

    select us_code
    from ux$frdata, ux$source
    where ufd_ufr_uuid = :ufr_uuid and
          ufd_kind = 1 and
          us_uuid = ufd_us_uuid
    into us_update;

    select us_code
    from ux$frdata, ux$source
    where ufd_ufr_uuid = :ufr_uuid and
          ufd_kind = 2 and
          us_uuid = ufd_us_uuid
    into us_insert;

    select us_code
    from ux$frdata, ux$source
    where ufd_ufr_uuid = :ufr_uuid and
          ufd_kind = 3 and
          us_uuid = ufd_us_uuid
    into us_delete;

    exp_expression = '';
    select exp_expression_all
    from uxp$uxexpression_text(:ufr_uuid)
    into :exp_expression;

    suspend;
  end
end
