CREATE PROCEDURE CM_RU_0008_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable ws integer;
declare variable wp integer;
declare variable wj integer;
declare variable wa integer;
declare variable cdc_code varchar(255);
declare variable cdv_value_string varchar(1024);
declare variable pr_code varchar(100);
begin
  for
  select kc_from_c_id,
         k_disb_date,
         c_name_fio,
         c_pass_no,
         kc_amount_approved,
         pr_code
  from knclient, kontract, clients, product
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  --and kc_kommis_lgota > 0
  and kc_from_k_id = k_id
  and kc_from_c_id = c_id
  and k_from_pr_id = pr_id
  into c_id,
       kf_disb_date,
       res_info_text,
       kf_client_pass_no,
       kc_amount,
       pr_code
  do
  begin
--        Добавлено по письму Бабура от 23.05.2015, если у клиента не указан место работы то заем не утверждается
      ws = 0; -- work_status
      wp = 0; -- work_place_name
      wj = 0; -- work_job_name
      wa = 0; -- work_address_name
      cdv_value_string = null;
      select OVS_VALUE from HP_OBJECTPROPS_GET(0, :c_id, -1, 'work_status', current_date)
      into :cdv_value_string;
      cdv_value_string = trim(coalesce(cdv_value_string, ''));
      if ((cdv_value_string <> '') and (cdv_value_string <> 'Значение')) then
      begin
        if (cdv_value_string in ('Безработный', 'не работает')) then ws = 2;
        else  ws = 1; -- работает
      end
      cdv_value_string = null;
      select OVS_VALUE from HP_OBJECTPROPS_GET(0, :c_id, -1, 'work_place_name', current_date)
      into :cdv_value_string;
      cdv_value_string = trim(coalesce(cdv_value_string, ''));
      if ((cdv_value_string <> '') and (cdv_value_string <> 'Значение')) then wp = 1;
      cdv_value_string = null;
      select OVS_VALUE from HP_OBJECTPROPS_GET(0, :c_id, -1, 'work_job_name', current_date)
      into :cdv_value_string;
      cdv_value_string = trim(coalesce(cdv_value_string, ''));
      if ((cdv_value_string <> '') and (cdv_value_string <> 'Значение'))  then wj = 1;
      cdv_value_string = null;
      select OVS_VALUE from HP_OBJECTPROPS_GET(0, :c_id, -1, 'work_address_name', current_date)
      into :cdv_value_string;
      cdv_value_string = trim(coalesce(cdv_value_string, ''));
      if ((cdv_value_string <> '') and (cdv_value_string <> 'Значение')) then wa = 1;
      if (exists (select * from addrclient ac where ac.adc_from_c_id = :c_id and ac.adc_address_type = 3 and ac.adc_from_adl_id <> 81001000000000)) then wa = 1;
      /*
      for
      select cdc_code, cdv_value_string
      from  CUSTOMDATACFG, customdatavalue
      where CDC_OBJECT_KIND = 0 -- client
      and current_date between cdv_date_begin and cdv_date_end
      and cdc_is_delete = 0
      and cdv_object_id = :c_id
      and CDV_FROM_CDC_ID = cdc_id
      and cdc_object_kind = cdv_object_kind
      into cdc_code, cdv_value_string
      do
      begin
        if (cdc_code = 'work_status') then
        begin
          if (coalesce(cdv_value_string,'') = 'Работает') then ws = 1;
          else if (coalesce(cdv_value_string,'') = 'Безработный') then ws = 2;
        end
        if (cdc_code = 'work_place_name' and coalesce(cdv_value_string,'') <> '') then wp = 1;
        if (cdc_code = 'work_job_name' and coalesce(cdv_value_string,'') <> '') then wj = 1;
        if (cdc_code = 'work_address_name' and coalesce(cdv_value_string,'') <> '') then wa = 1;
      end
      */
    if (pr_code not similar to '%.FAMILY%|%.PATENT%') then  --Добавил по письму Зары
    /*
    они должны выбрать, что он безработный, тогда база может пропустить его без
    указания работы. В других случаях - нет.
    Плюс, по продукту Микрозаем на патент и Групповой микрозаем на патент тоже
    может быть не указано место работы, так как клиент может ее не иметь, пока не получил патент
    */
    begin
      if (ws = 0) then
      begin
        res_error_is_critical=2;
        res_error_text='Поле "Работает/Безработный" не заполнено!';
        Suspend;
      end --else ws = 1;
      if (ws = 1) then
      begin
        if (wp = 0) then
        begin
          res_error_is_critical=2;
          res_error_text='Место работы не указано!';
          Suspend;
        end 
        if (wj = 0) then
        begin
          res_error_is_critical=2;
          res_error_text='Должность не указана!';
          Suspend;
        end
        if (wa = 0) then
        begin
          res_error_is_critical=2;
          res_error_text='Адрес работы не указано!';
          Suspend;
        end
      end
    end
  end
end
