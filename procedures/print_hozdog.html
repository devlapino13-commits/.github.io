CREATE PROCEDURE PRINT_HOZDOG(
    C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_FULL_ADDRESS VARCHAR(255),
    C_INN VARCHAR(20),
    REG_NAME VARCHAR(50),
    SD_DATE_BEGIN DATE,
    SD_DATE_END DATE,
    SD_CATEGORY VARCHAR(255),
    SD_MEMO VARCHAR(1024))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select c_name_fio, c_pass_no, c_pass_date, c_pass_who, adl_address_str, c_inn, reg_name, sd_date_begin,
             sd_date_end, sd_category, sd_memo

      from clients, servicedog, region, addrclient, addrlist
      where sd_from_c_id = :c_id and
            :rep_date between sd_date_begin and sd_date_end and
            sd_from_c_id = c_id and
            sd_from_reg_id = reg_unicode and
            adc_from_c_id = c_id and
            adc_from_adl_id = adl_id and
            adc_address_type = 1
      order by sd_date_begin desc
      rows 1
      into :c_name_fio, :c_pass_no, :c_pass_date, :c_pass_who, :c_full_address, :c_inn, :reg_name, :sd_date_begin,
           :sd_date_end, :sd_category, :sd_memo

  do
  begin
    suspend;
  end
end
