CREATE PROCEDURE HP_CONTDATA_DATE_UPDATE(
    EDS_ID D_ID NOT NULL,
    CDA_ID D_ID,
    CLT_ID D_INTEGER,
    NEW_DATE D_DATE = current_date)
AS
declare variable c_id d_id;
begin
  -- Процедура освежает дату подтвержения номера клиента после подписания электронного документа

  select eds_from_c_id
  from edoc_sign
  where eds_id = :eds_id
  into c_id;

  if (c_id is null or
      cda_id is null or
      clt_id is null) then
    exit;

  if (new_date is null) then
    new_date = current_date;

  update contdata
  set cda_date = :new_date
  where cda_id = :cda_id and
        cda_from_c_id = :c_id and
        cda_from_clt_id = :clt_id and
        cda_is_delete = 0 and
        cda_date < :new_date;
end
