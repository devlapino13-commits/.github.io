CREATE PROCEDURE HP_KONTSTATE_LOAN_STATUS(
    K_ID DECIMAL(18,0),
    MA_REG_ID INTEGER,
    MA_NO INTEGER,
    REP_DATE DATE,
    CLIENTS_ID_LIST VARCHAR(255))
RETURNS (
    C_LAST_BONUS_STATUS INTEGER,
    C_CURR_BONUS_STATUS INTEGER,
    C_LAST_BONUS_STATUS_STR VARCHAR(100),
    C_CURR_BONUS_STATUS_STR VARCHAR(100))
AS
declare variable kolich integer;
declare variable c_id decimal(18,0);
declare variable db_id integer;
declare variable client_list_tmp varchar(255);
declare variable c_id_str varchar(20);
declare variable n integer;
declare variable sql_text varchar(1024);
declare variable kf_officer id_type;
declare variable old_client smallint;
declare variable is_svz smallint;
declare variable first_id_officer id_type;
declare variable is_action smallint;
declare variable re_cli smallint;
declare variable is_erorr smallint;
declare variable kfolder_id id_type;
declare variable first_kfolder_id id_type;
begin

  db_id = f_get_db_id();
  if (clients_id_list is null) then
    clients_id_list = '';

  if ((db_id <> 81001) or (rep_date < cast('05.10.2015' as date))) then
  begin
    select c_last_bonus_status, count(c_id)
    from hp_kontstate_loan_status_tmp(:k_id, :ma_reg_id, :ma_no, :rep_date, :clients_id_list)
    group by c_last_bonus_status
    order by 2 desc, 1 desc
    rows 1
    into :c_last_bonus_status, :kolich;
    if (c_last_bonus_status is null) then
      c_last_bonus_status = 0;

    select c_curr_bonus_status, count(c_id)
    from hp_kontstate_loan_status_tmp(:k_id, :ma_reg_id, :ma_no, :rep_date, :clients_id_list)
    group by c_curr_bonus_status
    order by 2 desc, 1 desc
    rows 1
    into :c_curr_bonus_status, :kolich;
    if (c_curr_bonus_status is null) then
      c_curr_bonus_status = 0;
  end
  else
  begin
--    добавлено по приказу №23-09-15 от 30.09.2015г. пункт 5.6.2
--    Если все члены группы имеют разные статусы то статус такой группы определяется по клиенту с самым низким статусам
    select c_last_bonus_status, count(c_id)
    from hp_kontstate_loan_status_tmp(:k_id, :ma_reg_id, :ma_no, :rep_date, :clients_id_list)
    group by c_last_bonus_status
    order by 2 desc, 1
    rows 1
    into :c_last_bonus_status, :kolich;
    if (c_last_bonus_status is null) then
      c_last_bonus_status = 0;

    select c_curr_bonus_status, count(c_id)
    from hp_kontstate_loan_status_tmp(:k_id, :ma_reg_id, :ma_no, :rep_date, :clients_id_list)
    group by c_curr_bonus_status
    order by 2 desc, 1
    rows 1
    into :c_curr_bonus_status, :kolich;
    if (c_curr_bonus_status is null) then
      c_curr_bonus_status = 0;
  end

  c_last_bonus_status_str = '';
  c_curr_bonus_status_str = '';

  if (c_last_bonus_status = -1) then
    c_last_bonus_status_str = 'Не определен';
  if (c_last_bonus_status = 0) then
    c_last_bonus_status_str = 'Без статуса';
  if (c_last_bonus_status = 1) then
    c_last_bonus_status_str = 'Бронзовый';
  if (c_last_bonus_status = 2) then
    c_last_bonus_status_str = 'Серебрянный';
  if (c_last_bonus_status = 3) then
    c_last_bonus_status_str = 'Золотой';
  if (c_last_bonus_status = 4) then
    c_last_bonus_status_str = 'Бриллиантовый';
  if (c_last_bonus_status = 5) then
    c_last_bonus_status_str = 'Бриллиант-миллионер';

  if (c_curr_bonus_status = -1) then
    c_curr_bonus_status_str = 'Не определен';
  if (c_curr_bonus_status = 0) then
    c_curr_bonus_status_str = 'Без статуса';
  if (c_curr_bonus_status = 1) then
    c_curr_bonus_status_str = 'Бронзовый';
  if (c_curr_bonus_status = 2) then
    c_curr_bonus_status_str = 'Серебрянный';
  if (c_curr_bonus_status = 3) then
    c_curr_bonus_status_str = 'Золотой';
  if (c_curr_bonus_status = 4) then
    c_curr_bonus_status_str = 'Бриллиантовый';
  if (c_curr_bonus_status = 5) then
    c_curr_bonus_status_str = 'Бриллиант-миллионер';

  if ((db_id = 81001) and
      (rep_date >= cast('05.10.2015' as date))) then
  begin
    if (c_last_bonus_status = 0) then
      c_last_bonus_status_str = 'Без статуса';
    if (c_last_bonus_status = 1) then
      c_last_bonus_status_str = 'Бронза';
    if (c_last_bonus_status = 2) then
      c_last_bonus_status_str = 'Серебро';
    if (c_last_bonus_status = 3) then
      c_last_bonus_status_str = 'Золото';
    if (c_last_bonus_status = 4) then
      c_last_bonus_status_str = 'Бриллиант';
    if (c_last_bonus_status = 5) then
      c_last_bonus_status_str = 'Бриллиант*';
    if (c_last_bonus_status = 6) then
      c_last_bonus_status_str = 'Бриллиант**';
    if (c_last_bonus_status = 7) then
      c_last_bonus_status_str = 'Бриллиант***';
    if (c_last_bonus_status = 8) then
      c_last_bonus_status_str = 'Бриллиант****';
    if (c_last_bonus_status = 9) then
      c_last_bonus_status_str = 'Бриллиант*****';
    if (c_last_bonus_status = 10) then
      c_last_bonus_status_str = 'Платинум';

    if (c_curr_bonus_status = 0) then
      c_curr_bonus_status_str = 'Без статуса';
    if (c_curr_bonus_status = 1) then
      c_curr_bonus_status_str = 'Бронза';
    if (c_curr_bonus_status = 2) then
      c_curr_bonus_status_str = 'Серебро';
    if (c_curr_bonus_status = 3) then
      c_curr_bonus_status_str = 'Золото';
    if (c_curr_bonus_status = 4) then
      c_curr_bonus_status_str = 'Бриллиант';
    if (c_curr_bonus_status = 5) then
      c_curr_bonus_status_str = 'Бриллиант*';
    if (c_curr_bonus_status = 6) then
      c_curr_bonus_status_str = 'Бриллиант**';
    if (c_curr_bonus_status = 7) then
      c_curr_bonus_status_str = 'Бриллиант***';
    if (c_curr_bonus_status = 8) then
      c_curr_bonus_status_str = 'Бриллиант****';
    if (c_curr_bonus_status = 9) then
      c_curr_bonus_status_str = 'Бриллиант*****';
    if (c_curr_bonus_status = 10) then
      c_curr_bonus_status_str = 'Платинум';

-- если клиенты уволенного сотрудника, то C_CURR_BONUS_STATUS=C_CURR_BONUS_STATUS+5 делаем.
    sql_text = '';
    if (clients_id_list <> '') then
    begin
      /*
        while (client_list_tmp<>'') do
        begin
          c_id_str='';
          n=position(';',client_list_tmp,1);
          if (n>0) then
          begin
            c_id_str=left(client_list_tmp, n-1);
            client_list_tmp=substring(client_list_tmp from n+1);
          end else
          begin
            c_id_str=client_list_tmp;
            client_list_tmp='';
          end
          if (c_id_str<>'') then
          begin
            if (sql_text='')
            then sql_text='select c_id from clients where c_id='||:c_id_str;
            else sql_text=sql_text||' union all select c_id from clients where c_id='||:c_id_str;
          end
        end
        */
      client_list_tmp = replace(clients_id_list, ';', ',');
      sql_text = 'select c_id from clients
                    where c_id in (' || :client_list_tmp || ')';
    end

    if (sql_text <> '') then
    begin
      first_id_officer = 0;
      first_kfolder_id = 0;
      is_action = 1;
      re_cli = 0;
      for execute statement :sql_text
              into :c_id
      do
      begin
--    проверка каждого клиента
        old_client = 0;
        is_erorr = 0;
        is_svz = 0;

        for select kal_new_kf_officer, kal_from_kf_id
            from knclient, kontstate, kontract, knaktlist
            where kc_from_c_id = :c_id and
                  kc_amount_approved > 0 and
                  kc_from_k_id = ks_from_k_id and
                  ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  kc_from_k_id = k_id and
                  k_from_kf_id = kal_from_kf_id and
                  knaktlist.is_delete = 0 and
                  kal_date <= :rep_date
            order by kal_date desc
            rows 2 -- проверка 2 последних передач
            into kf_officer, kfolder_id
        do
        begin
          if (re_cli = 0) then
            re_cli = 1; -- для того чтобы если все клиенты новые чтобы не попали под акцию
          old_client = 1;
          if (exists(select *
                     from hp_get_departed_svz(:kf_officer)
                     where result = 1)) then
          begin
            is_svz = 1;
--        данная проверка для того чтобы предыдущий уволенный КС один и тот же человек должен быть
            if (first_id_officer = 0) then
              first_id_officer = kf_officer;
            else
            if (first_id_officer <> kf_officer) then
              is_erorr = 1;
--            если клиенты с разных групп собран из одного же СВЗ то тоже не подлежить к акцие
            if (first_kfolder_id = 0) then
              first_kfolder_id = kfolder_id;
            else
            if (first_kfolder_id <> kfolder_id) then
              is_erorr = 1;
          end

        end
        if (is_action = 1 and
            old_client = 1 and
            (is_svz = 0 or is_erorr = 1)) then
          is_action = 0;

      end
      if (re_cli = 0 and
          is_action = 1) then
        is_action = 0;

--        if (is_action = 1) then C_CURR_BONUS_STATUS = C_CURR_BONUS_STATUS + 5;
      if (c_curr_bonus_status > 10) then
        c_curr_bonus_status = 10;
    end
  end

  suspend;
end
