CREATE PROCEDURE HP_OPERDEL_OPERKIND(
    OL_ID D_ID,
    OPER_TYPE D_SMALLINT,
    OPER_SUBTYPE D_SMALLINT)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_NAME D_STR_100)
AS
begin
  if (coalesce(oper_type, -1) = -1) then
  begin
    select ol_oper_type, ol_oper_subtype
    from operlist
    where ol_id = :ol_id
    into oper_type, oper_subtype;
  end

  oper_type = coalesce(oper_type, 0);
  oper_subtype = coalesce(oper_subtype, 0);

  select opl_name
  from optlist o
  where o.opl_main = :oper_type and
        o.opl_detail = :oper_subtype
  into res_name;

  if (row_count = 0) then
    exception e_message 'Операция не определена, тип: ' || oper_type || ', субтип: ' || oper_subtype;

  res_code = '';
  if (oper_type = 1) then
  begin
    res_code = trim(case oper_subtype
      when 1 then 'disb'
      when 2 then 'repay'
      when 3 then 'writeoff'
      when 4 then 'reschedule'
      when 9 then 'transfer'
      when 10 then 'overpayback'
      when 11 then 'overpayoff'
      when 12 then 'floatperc'
      when 13 then 'reverse'
      else ''
    end);
  end

  if (res_code = '') then
    res_name = '?';

  suspend;
end
