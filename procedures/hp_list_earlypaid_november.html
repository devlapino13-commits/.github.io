CREATE PROCEDURE HP_LIST_EARLYPAID_NOVEMBER(
    REP_DATE D_DATE = current_date,
    IS_FILL D_STR_10 = 'no')
RETURNS (
    RES_K_ID D_ID,
    RES_KONTRACT_NO D_STR_20,
    RES_IS_IPF D_INTEGER,
    RES_C_ID D_ID,
    RES_C_NAME_FIO D_STR_100,
    RES_CURR_PERC D_CURRENCY,
    RES_OL_DATE D_DATE,
    RES_REG_ID D_INTEGER,
    RES_REG_NAME D_STR_CODE)
AS
declare variable ol_id d_id;
declare variable k_kolich d_integer;
declare variable sum_group d_currency;
declare variable count_group d_integer;
declare variable tmp_curr_perc d_currency;
declare variable n d_integer;
declare variable s d_currency;
begin
  if (is_fill = 'yes')
    then delete from earlypaid_november;

  for
    select
     ok_from_k_id,
     ok_from_ol_id,
     ok_ol_date,
      -sum(iif(ok_ol_date < ok_sh_date, ok_perc, 0))
    from operkred ok
    where ok_ol_date between '17.11.2020' and '30.11.2020'
      and ok_sh_date  between '01.12.2020' and '28.02.2021'
      and ok_ol_is_active = 1
      and ok_type = 4
    group by  ok_from_k_id, ok_from_ol_id, ok_ol_date
    having -sum(iif(ok_ol_date < ok_sh_date, ok_perc, 0)) > 0
  into
    :res_k_id,
    :ol_id,
    :res_ol_date,
    :res_curr_perc
  do
  begin
    -- участие в гос.дотации
    if (exists(
      select *
      from knclient_compens
      where kcc_from_k_id = :res_k_id
        and kcc_date_end > kcc_date_begin
      )) then
    begin
      -- если это погашение доли
      if (exists(
        select *
        from knclient
        where kc_from_k_id = :res_k_id
          and kc_closed_ol_id = :ol_id))
      then continue;

      -- активность на текущую дату
      if (exists(
        select *
        from kontstate
        where ks_from_k_id = :res_k_id
          and :rep_date between ks_date_begin and ks_date_end
          and ks_ol_is_delete = 0
          and ks_state = 1)) then
      begin
        select
          k_kontract_no,
          k_kolich_client_approved,
          reg_unicode,
          reg_name,
          iif(pr_code like '%.ipf.%', 1, 0)
        from kontract
          inner join region
            on reg_unicode = k_last_reg_id
          inner join product
            on pr_id = k_from_pr_id
        where k_id = :res_k_id
        into
          :res_kontract_no,
          :k_kolich,
          :res_reg_id,
          :res_reg_name,
          :res_is_ipf;

        if (k_kolich = 1) then
        begin
          select
            c_id,
            c_name_fio
          from knclient
            inner join clients
              on c_id = kc_from_c_id
          where kc_from_k_id = :res_k_id
          into
            :res_c_id,
            :res_c_name_fio;

          if (res_curr_perc >= 1) then begin
            if (is_fill = 'yes') then
              insert into earlypaid_november (
                en_k_id, en_kontract_no, en_is_ipf, en_c_id, en_c_name_fio, en_curr_perc,
                en_ol_date, en_reg_id, en_reg_name)
              values(:res_k_id, :res_kontract_no, :res_is_ipf, :res_c_id, :res_c_name_fio, :res_curr_perc,
                :res_ol_date, :res_reg_id, :res_reg_name);
            else suspend;
          end
        end else
        begin
          select
            sum(iif(ol_id is null, kc_amount_approved, 0))
          from knclient
            left join operlist
              on ol_id = kc_closed_ol_id
                and is_delete = 0
                and ol_date < :res_ol_date
          where kc_from_k_id = :res_k_id
            and kc_amount_approved > 0
          into
            :sum_group;
          tmp_curr_perc = res_curr_perc;

          n = 0;
          s = 0;
          for
            select
              c_id,
              c_name_fio,
              round(:res_curr_perc * kc_amount_approved / :sum_group, 0)
            from knclient_compens
              inner join knclient
                on kc_id = kcc_from_kc_id
              inner join clients
                on c_id = kc_from_c_id
            where kcc_from_k_id = :res_k_id
              and kcc_date_end > kcc_date_begin
          into
            :res_c_id,
            :res_c_name_fio,
            :res_curr_perc
          do
          begin
            n = n + 1;
            if (n = k_kolich)
              then res_curr_perc = tmp_curr_perc - s;
                else s = s + res_curr_perc;

            if (res_curr_perc >= 1) then begin
              if (is_fill = 'yes') then
                insert into earlypaid_november (
                  en_k_id, en_kontract_no, en_is_ipf, en_c_id, en_c_name_fio, en_curr_perc,
                  en_ol_date, en_reg_id, en_reg_name)
                values(:res_k_id, :res_kontract_no, :res_is_ipf, :res_c_id, :res_c_name_fio, :res_curr_perc,
                  :res_ol_date, :res_reg_id, :res_reg_name);
              else suspend;
            end

            res_curr_perc = tmp_curr_perc;
          end
        end
      end
    end
  end
end
