CREATE PROCEDURE UXP$COMPANY_PARAMS
RETURNS (
    REG_COMPANY_NAME D_STR_255,
    REG_COMPANY_BOSS_TITUL D_STR_255,
    REG_COMPANY_BOSS_NAME D_STR_255,
    REG_COMPANY_ACCOUNT_TITUL D_STR_255,
    REG_COMPANY_ACCOUNT_NAME D_STR_255,
    CM_NAME D_STR_255,
    CM_COMPANY_FULL_NAME D_STR_255)
AS
declare variable reg_id d_integer;
begin
  reg_id = uxf$hq_id();

  select reg_company_name, reg_company_boss_titul, reg_company_boss_name,
         reg_company_account_titul, reg_company_account_name
  from region
  where reg_unicode = :reg_id
  into reg_company_name, reg_company_boss_titul, reg_company_boss_name,
       reg_company_account_titul, reg_company_account_name;

  cm_name = uxf$const('company_name');
  cm_company_full_name = cm_name;

  suspend;
end
