CREATE PROCEDURE CM_RU_0092_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable k_amount_approved curr_type;
declare variable k_percent perc_type;
declare variable k_kommis_perc perc_type;
declare variable kc_id id_type;
declare variable k_kolich_repay integer;
begin
  for select k_disb_date, kc_from_c_id, k_amount_approved,
        k.k_kolich_repay, k.k_percent, k.k_kommis_perc, KC_AMOUNT_APPROVED
    from kontract k, product, knclient
  where k_id = :k_id
  and k_from_pr_id = pr_id       
--  and pr_code = '.IND.EMP.WINNER.'
  and pr_code containing '.IND.'
  and pr_code containing '.EMP.'
  and pr_code containing '.WINNER.'
  and k_id = kc_from_k_id
  and kc_amount_approved > 0
  into KF_DISB_DATE, kc_id, k_amount_approved, k_kolich_repay, k_percent, k_kommis_perc, KC_AMOUNT
  do
  begin
    if (kc_id not in (
                    1010100024447,
                    1010100054869,
                    1010100139819,
                    1010100154744,
                    1010100315506,
                    1010100295691,
                    1010100276471)
        ) then
    begin
      res_error_is_critical = 2; --Не менять! Если просят сделать исключение, то лучше добавить СВЗ и айди этого контракта
      res_error_text = 'Указан не правильный продукт или сотрудник, проверьте данные';
      suspend;
      exit;
    end
    if (k_amount_approved < 100000) then
    begin  
      res_error_is_critical = 2;
      res_error_text = 'Сумма кредита не может быть меньше 100000 р.';
      suspend;
    end 
    if (k_kolich_repay not between 12 and 18) then
    begin  
      res_error_is_critical = 2;
      res_error_text = 'Количество платежей должен быть от 12 до 18';
      suspend;
    end 
    if (k_percent <> 20) then
    begin  
      res_error_is_critical = 2;
      res_error_text = '%%ставка указана не верно, должно быть 20%';
      suspend;
    end 
    if (k_kommis_perc <> 0) then
    begin  
      res_error_is_critical = 2;
      res_error_text = 'Доп.процент указан не верно, должен быть 0';
      suspend;
    end 
    if (kc_id in (
                1010100024447,
                1010100054869,
                1010100139819,
                1010100154744,
                1010100315506)
        ) then
    begin
      if (k_amount_approved > 250000) then
      begin  
        res_error_is_critical = 1;
        res_error_text = 'Сумма кредита не может быть больше 250000 р.';
        suspend;
      end
    end
    if (kc_id = 1010100295691) then
    begin 
      if (k_amount_approved > 135000) then
      begin  
        res_error_is_critical = 1;
        res_error_text = 'Сумма кредита не может быть больше 135000 р.';
        suspend;
      end
    end
    if (kc_id = 1010100276471) then
    begin 
      if (k_amount_approved > 150000) then
      begin  
        res_error_is_critical = 1;
        res_error_text = 'Сумма кредита не может быть больше 150000 р.';
        suspend;
      end
    end

  end
end
