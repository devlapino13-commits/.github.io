CREATE PROCEDURE AAREP_CLI_INFO(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    KF_NAME VARCHAR(100),
    KNC_NAME_FIO VARCHAR(100),
    PR_LOAN_TYPE_STR VARCHAR(20),
    KNC_AMOUNT_APPROVED CURR_TYPE,
    KNC_REP_ACTIVE_BALANCE CURR_TYPE,
    K_PERCENT DECIMAL(18,7),
    K_PERCENT_EFFECT PERC_TYPE,
    RESULT_PERC_END_CLI CURR_TYPE,
    K_DISB_DATE DATE,
    K_PERIOD DECIMAL(15,2),
    K_KREDLINE_END DATE,
    REP_DELINQ_DAYS INTEGER,
    PROLONG_KOL INTEGER,
    PROLONG_SROK DATE,
    IS_ZALOG SMALLINT,
    LLC_CURRENT_STATUS_STR VARCHAR(50),
    KNC_REP_LLR_CL_AMOUNT CURR_TYPE,
    LLC_AMMOUNT_FROM_PERCENT_CLI CURR_TYPE,
    KNC_PLASTCARD_NO VARCHAR(20),
    PL_NAME VARCHAR(100),
    KF_OFFICER_NAME VARCHAR(100),
    REG_STAT_REP_REGION VARCHAR(255),
    KF_REGION_NAME VARCHAR(60),
    KF_BRANCH_NAME VARCHAR(60),
    PR_CODE VARCHAR(20),
    PR_NAME VARCHAR(50),
    PR_IS_EMPLOYER_STR VARCHAR(20),
    PR_IS_SECURED_STR VARCHAR(20),
    KNC_BUSINESS_TYPE VARCHAR(50),
    KNC_BUSINESS_SUBTYPE VARCHAR(50),
    KNC_CIRCLE INTEGER,
    LLC_CURRENT_RESERVE VARCHAR(50),
    LLC_CURRENT_PERCENT CURR_TYPE,
    KOL_ZALOG INTEGER,
    ZALOG_1 VARCHAR(100),
    ZALOG_1_SUBTYPE VARCHAR(50),
    ZALOG_2 VARCHAR(100),
    ZALOG_2_SUBTYPE VARCHAR(50),
    BALANCE_CLI_TEST CURR_TYPE,
    K_ID ID_TYPE,
    K_PERCENT360 PERC_TYPE,
    KNC_PASS_NO VARCHAR(30),
    KNC_PASS_WHO VARCHAR(255),
    KNC_PASS_DATE DATE)
AS
declare variable k_percent_per_day decimal(18,7);
declare variable pr_int_calc_type integer;
declare variable k_amount_approved curr_type;
declare variable k_id_prior id_type;
declare variable k_stop_perc date;
declare variable result_perc_end curr_type;
declare variable cr_rate curr_type;
declare variable rep_active_balance curr_type;
declare variable llc_ammount_from_percent curr_type;
declare variable kf_region_id integer;
declare variable kf_region_id_prior integer;
declare variable k_currency varchar(3);
declare variable k_kolich_repay integer;
declare variable db_id integer;
declare variable k_zero_balance_date date;
begin
  k_id_prior = 0;
  kf_region_id_prior = 0;
  cr_rate = (select cr_rate
             from opercurr
             where cr_from_cu_id = 840 and
                   :rep_date_end between cr_date and cr_date_end);

  :db_id = f_get_db_id();

  for select k_kontract_no, k_disb_date, k_period, k_currency, k_percent, k_kolich_repay, kf_name, kf_region_name,
             kf_branch_name, pr_code, pr_name, pr_loan_type_str, pr_is_employer_str, pr_is_secured_str, knc_name_fio,
             knc_business_type, knc_business_subtype, knc_amount_approved, knc_rep_active_balance, knc_circle,
             llc_current_status_str, llc_current_reserve, llc_current_percent, knc_rep_llr_cl_amount, k_kredline_end,
             k_id, k_amount_approved, rep_active_balance, kf_officer_name, kf_region_id, knc_plastcard_no, knc_pass_no,
             knc_pass_who, knc_pass_date

      from aarep_kont_clients_list_new(:rep_region_id, :rep_date_begin, :rep_date_end, 0, '000', 3)
      into k_kontract_no, k_disb_date, k_period, k_currency, k_percent, k_kolich_repay, kf_name, kf_region_name,
           kf_branch_name, pr_code, pr_name, pr_loan_type_str, pr_is_employer_str, pr_is_secured_str, knc_name_fio,
           knc_business_type, knc_business_subtype, knc_amount_approved, knc_rep_active_balance, knc_circle,
           llc_current_status_str, llc_current_reserve, llc_current_percent, knc_rep_llr_cl_amount, k_kredline_end,
           k_id, k_amount_approved, rep_active_balance, kf_officer_name, kf_region_id, knc_plastcard_no, knc_pass_no,
           knc_pass_who, knc_pass_date

  do
  begin
    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;
      prolong_kol = (select count(*)
                     from operrest, operlist
                     where operlist.is_delete = 0 and
                           ol_date <= :rep_date_end and
                           ol_id = or_from_ol_id and
                           or_from_k_id = :k_id);

      prolong_srok = iif(exists(select 1
                                from operrest, operlist
                                where operlist.is_delete = 0 and
                                      ol_date <= :rep_date_end and
                                      ol_id = or_from_ol_id and
                                      or_from_k_id = :k_id), (select max(ok_sh_date)
                                                              from operkred
                                                              where ok_ol_is_active = 1 and
                                                                    ok_ol_date <= :rep_date_end and
                                                                    ok_from_c_id = 0 and
                                                                    ok_from_k_id = :k_id), null);

      is_zalog = iif(exists(select 1
                            from zalog
                            where z_type = 1 and
                                  z_from_k_id = :k_id), 1, 0);
      kol_zalog = (select count(*)
                   from zalog
                   where z_type = 1 and
                         z_from_k_id = :k_id);

      zalog_1 = (select z_name
                 from zalog
                 where z_type = 1 and
                       z_from_k_id = :k_id rows 1);

      zalog_1_subtype = (select (select zs_name
                                 from zalogsubtype
                                 where zs_id = z_subtype)
                         from zalog
                         where z_type = 1 and
                               z_from_k_id = :k_id rows 1);

      zalog_2 = (select z_name
                 from zalog
                 where z_type = 1 and
                       z_from_k_id = :k_id rows 2 to 2);
      zalog_2_subtype = (select (select zs_name
                                 from zalogsubtype
                                 where zs_id = z_subtype)
                         from zalog
                         where z_type = 1 and
                               z_from_k_id = :k_id rows 2 to 2);

      k_stop_perc = null;
      pr_int_calc_type = null;
      k_percent_per_day = null;
      select k_stop_perc, pr_interest_method, kd_percent_per_day, k_percent_effect, k_kredline_end,
             k_zero_balance_date
      from kontract, product, kontdata
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            k_id = kd_from_k_id and
            kd_ol_date <= :rep_date_end and
            kd_date_end >= :rep_date_end and
            kd_ol_is_delete = 0
      into k_stop_perc, pr_int_calc_type, k_percent_per_day, k_percent_effect, k_kredline_end, k_zero_balance_date;
      k_percent360 = k_percent_per_day * 365;


      if (f_new_perc_active(rep_date_end) = 1)
      then result_perc_end = f_perc_saldo(k_id, rep_date_end);
      else
      begin
        result_perc_end = (select result_perc_end
                           from aarep_calc_perc_one(:k_id, :rep_date_begin, :rep_date_end, 1, :db_id, :k_kredline_end, :k_stop_perc, :k_zero_balance_date));
      end

      if (k_currency = '840') then
        result_perc_end = result_perc_end * cr_rate;

      rep_delinq_days = coalesce((select rep_delinq_days
                                  from aarep_kont_delinq_status(:k_id, :rep_date_end)), 0);
      llc_ammount_from_percent = 0;
      if (rep_delinq_days >= 90) then
        llc_ammount_from_percent = result_perc_end;
    end
    if (kf_region_id <> kf_region_id_prior) then
    begin
      kf_region_id_prior = kf_region_id;
      select reg_stat_rep_region, pl_name
      from region, addrlist, places
      where reg_unicode = :kf_region_id and
            reg_address_fact = adl_id and
            adl_from_pl_id = pl_id
      into reg_stat_rep_region, pl_name;
    end

    result_perc_end_cli = cast(knc_amount_approved as double precision) / k_amount_approved * result_perc_end;
    balance_cli_test = cast(knc_amount_approved as double precision) / k_amount_approved * rep_active_balance;
    llc_ammount_from_percent_cli = cast(knc_amount_approved as double precision) / k_amount_approved * llc_ammount_from_percent;
    suspend;
  end

end
