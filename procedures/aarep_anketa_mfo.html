CREATE PROCEDURE AAREP_ANKETA_MFO(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_45 INTEGER)
RETURNS (
    KOL_CLI_END INTEGER,
    KOL_CLI_PERIOD INTEGER,
    KOL_KRED_END INTEGER,
    KOL_KRED_PERIOD INTEGER,
    DISB_SUM CURR_TYPE,
    OKS_BALANCE CURR_TYPE,
    OKS_BALANCE_NOT_DELINQ CURR_TYPE,
    DELINQ_OS CURR_TYPE,
    DELINQ_OS_1_30 CURR_TYPE,
    DELINQ_OS_31_90 CURR_TYPE,
    DELINQ_OS_91_ CURR_TYPE,
    DELINQ_BALANCE_91_ CURR_TYPE,
    LLC_SUM_OS CURR_TYPE,
    LLC_SUM_PERC CURR_TYPE,
    LLC_SUM_RASHOD_OS CURR_TYPE,
    LLC_SUM_RASHOD_PERC CURR_TYPE,
    RES_BALANCE_IS_DELINQ CURR_TYPE,
    RES_BALANCE_NOT_DELINQ CURR_TYPE,
    RES_BALANCE_PERIOD_IS_DELINQ CURR_TYPE,
    RES_BALANCE_PERIOD_NOT_DELINQ CURR_TYPE,
    MAX_PERCENT CURR_TYPE,
    MIN_PERCENT CURR_TYPE,
    REP_MEMO VARCHAR(255),
    KOL_KRED_PERIOD_OLD INTEGER,
    DISB_SUM_PERIOD_POVTOR CURR_TYPE,
    KOL_CLI_REPEAT INTEGER,
    KOL_CLI_NEW INTEGER,
    DISB_SUM_PERIOD CURR_TYPE,
    SUM_CALC_PERC_DELINQ CURR_TYPE,
    KOL_CLI_DELINQ INTEGER,
    PSK CURR_TYPE,
    KOL_CLI_DELINQ_DISB6 INTEGER,
    PLAN_REPAY_OS CURR_TYPE,
    PLAN_REPAY_PERC CURR_TYPE,
    AAREP_PERC_CALCED CURR_TYPE,
    RES_AMOUNT_PERIOD CURR_TYPE,
    REPAY_FINE CURR_TYPE,
    REPAY_FINE_KOMMISS CURR_TYPE,
    REPAY_OS_GRAPH CURR_TYPE,
    REPAY_PERC_GRAPH CURR_TYPE)
AS
declare variable delinq_os_rest curr_type;
declare variable delinq_os_rest_period curr_type;
declare variable date_begin_month6 date;
begin
-- operkred_fix

  is_45 = coalesce(is_45, 0);
  if (is_45 > 1) then
  begin
    is_45 = 1;
  end
  if (is_45 < 0) then
  begin
    is_45 = 0;
  end

  if (is_45 = 1) then
    rep_memo = 'Указываются данные по займам с суммой выдачи менее 45 тыс. руб. и выданные на срок до 2 месяцев';
  else
    rep_memo = 'Указываются данные по всем займам кроме займов с суммой выдачи менее 45 тыс. руб. и выданные на срок до 2 месяцев';

-- Число заемщиков-физических лиц, ед. -- Число заемщиков за квартал, ед.
  select count(*), sum(iif(ac_disb_date between cast(:rep_date_begin as date) and cast(:rep_date_end as date), 1, 0)), sum(iif(exists(select *
                                                                                                                                      from oksummary, knclient -- 2,3
                                                                                                                                      where oks_balance > 0 and
                                                                                                                                            ac_disb_date between oks_date_begin and oks_date_end and
                                                                                                                                            oks_from_k_id = kc_from_k_id and
                                                                                                                                            kc_amount_approved > 0 and
                                                                                                                                            kc_from_c_id = ac_id and
                                                                                                                                            oks_close_state = 0 and
                                                                                                                                            oks_from_k_id <> ac_kontract_id), ac_is_client, 0))
  from aarep_cli_active_list(:rep_date_end)
  where ac_is_client = 1 and
        ((not(ac_amount_approved <= 45000 and
        ac_period < 2) and
        :is_45 = 0) -- для второй случай нужно наоборот сделать
        or
        ((ac_amount_approved <= 45000 and
        ac_period < 2) and
        :is_45 = 1))
  into kol_cli_end, kol_cli_period, kol_cli_repeat;

-- Количество микрозаймов, шт.
-- Кол-во активных микрозаймов выданный за заданный период т.е. квартал, добавлено от 18.09.2014
  select count(kc_from_c_id), sum(iif(k_disb_date between :rep_date_begin and :rep_date_end, 1, 0)),
         /*
         sum(iif(k_disb_date between :rep_date_begin and :rep_date_end -- пункт 2,1
                 and exists (select * from kontstate, knclient kc2
                             where ks_ol_is_delete = 0
                             and ks_state = 1
                             and k_disb_date between ks_date_begin and ks_date_end
                             and ks_from_k_id <> k_id
                             and ks_from_k_id = kc2.kc_from_k_id
                             and kc2.kc_amount_approved > 0
                             and kc2.kc_from_c_id = kc1.kc_from_c_id), 1, 0)),*/
         sum(iif(k_disb_date between :rep_date_begin and :rep_date_end -- пункт 2,1
         and exists(select *
                    from oksummary oks2, knclient kc3
                    where k_disb_date between oks2.oks_date_begin and oks2.oks_date_end and
                          oks2.oks_balance > 0 and
                          oks2.oks_from_k_id <> k_id and
                          oks2.oks_from_k_id = kc3.kc_from_k_id and
                          kc3.kc_amount_approved > 0 and
                          oks2.oks_close_state = 0 and
                          kc3.kc_from_c_id = kc1.kc_from_c_id), 1, 0)),

         sum(iif(k_disb_date between :rep_date_begin and :rep_date_end -- пункт 2,2
         and exists(select *
                    from oksummary oks2, knclient kc3
                    where k_disb_date between oks2.oks_date_begin and oks2.oks_date_end and
                          oks2.oks_balance > 0 and
                          oks2.oks_from_k_id <> k_id and
                          oks2.oks_from_k_id = kc3.kc_from_k_id and
                          kc3.kc_amount_approved > 0 and
                          oks2.oks_close_state = 0 and
                          kc3.kc_from_c_id = kc1.kc_from_c_id), kc1.kc_amount_approved, 0)), sum(iif(k_disb_date between :rep_date_begin and :rep_date_end -- 5,2,1
         and kc1.kc_circle <= 1, 1, 0)), sum(iif(k_disb_date between :rep_date_begin and :rep_date_end, kc1.kc_amount_approved, 0)) -- 5.6.1 и 5.6.2

  from oksummary oks1, knclient kc1, kontract
  where :rep_date_end between oks_date_begin and oks_date_end and
        oks_balance > 0 and
        oks_from_k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        oks_from_k_id = k_id and
        oks1.oks_close_state = 0 and
        ((not(kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 0) -- для второй случай нужно наоборот сделать
        or
        ((kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 1))
  into kol_kred_end, kol_kred_period, kol_kred_period_old, disb_sum_period_povtor, kol_cli_new, disb_sum_period;

-- Объем микрозаймов, предоставленных за минувший квартал, тыс. руб.

  select sum(kc_amount_approved),
-- 5,21    Полная стоимость микрозаймов, предоставленных за минувший квартал
         sum(cast(kc_amount_approved as double precision) * k_percent_effect_fifd) / sum(kc_amount_approved)
  from kontstate, kontract, knclient
  where ks_ol_is_delete = 0 and
        ks_state = 1 and
        cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
        ks_from_k_id = k_id and
        k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        ((not(kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 0) -- для второй случай нужно наоборот сделать
        or
        ((kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 1))
  into disb_sum, psk;

  date_begin_month6 = dateadd(month, -5, cast(:rep_date_end as date) - extract(day from cast(:rep_date_end as date)) + 1);
-- Объем портфеля микрозаймов (включая просроченную задолженность), всего, тыс. руб. (строка. 84)
  select sum(cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance), sum(iif(:rep_date_end >= oks_delinq_date, -- 5.9
         cast(kc_amount_approved as double precision) / k_amount_approved * (select sum(sh_calc_perc)
                                                                             from aarep_make_zad_list(k_id, :rep_date_end, 1)
                                                                             where sh_date is not null and
                                                                                   sh_repay_type < 2), 0)),
         sum(iif(:rep_date_end >= oks_delinq_date, 1, 0)), -- 5.10 и 5.10.1
         sum(iif(:rep_date_end >= oks_delinq_date and --  5.10.1  в т.ч. количество займов, выданных за последние полгода, по которым имеется просроченная задолженность, ед.
         k_disb_date between :date_begin_month6 and :rep_date_end, 1, 0))
  from oksummary, kontract, knclient
  where :rep_date_end between oks_date_begin and oks_date_end and
        oks_from_k_id = k_id and
        k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        oks_close_state = 0 and
        ((not(kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 0) -- для второй случай нужно наоборот сделать
        or
        ((kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 1))
  into oks_balance, sum_calc_perc_delinq, kol_cli_delinq, kol_cli_delinq_disb6;

-- Объем просроченной задолженности, всего, тыс. руб. (строка. 86)
--Объем просроченной задолженности, всего, тыс. руб.(просрочки с нулевыми днями не показывает) (строки. 87 - 89)
  with tabl1
  as (select oks_from_k_id, oks_balance, oks_delinq_date,
             (iif(exists(select *
                         from operrest, operlist
                         where operlist.is_delete = 0 and
                               ol_id = or_from_ol_id and
                               :rep_date_end between ol_date and or_date_end and
                               or_from_k_id = oks_from_k_id) and exists(select 1
                                                                        from rdb$database
                                                                        where (select iif(max(iif(ok_type = 1, ok_sh_date, cast('01.01.1900' as date))) < max(iif(ok_type = 2, ok_sh_date, cast('01.01.1900' as date))), 1, null)
                                                                               from operkred
                                                                               where ok_ol_is_active = 1 and
                                                                                     ok_ol_date <= :rep_date_end and
                                                                                     ok_from_k_id = oks_from_k_id and
                                                                                     ok_from_c_id = 0 and
                                                                                     ok_type in (1, 2)) = 1), 1, 0)) is_rest,
-- проверяет группу рестрктуризирован или нет, если да то проверяет кол-во погашение не менялся или нет
             (iif(exists(select *
                         from operrest, operlist
                         where operlist.is_delete = 0 and
                               ol_id = or_from_ol_id and
                               :rep_date_end between ol_date and or_date_end and
                               or_from_k_id = oks_from_k_id and
                               ol_date_oper_day between :rep_date_begin and :rep_date_end -- ограничение сделанные за текущий период
             ) and exists(select 1
                          from rdb$database
                          where (select iif(max(iif(ok_type = 1, ok_sh_date, cast('01.01.1900' as date))) < max(iif(ok_type = 2, ok_sh_date, cast('01.01.1900' as date))), 1, null)
                                 from operkred
                                 where ok_ol_is_active = 1 and
                                       ok_ol_date <= :rep_date_end and
                                       ok_from_k_id = oks_from_k_id and
                                       ok_from_c_id = 0 and
                                       ok_type in (1, 2)) = 1), 1, 0)) is_rest_period

      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            :rep_date_end >= oks_delinq_date and
            oks_close_state = 0 and
            oks_balance > 0),
  tabl2
  as (select oks_from_k_id, sum(ok_os + ok_sh_os) delinq_os,
             cast(:rep_date_end as date) - oks_delinq_date as delinq_day, oks_balance, is_rest, is_rest_period
      from tabl1, operkred
      where oks_from_k_id = ok_from_k_id and
            ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date_end and
            ok_type < 7 and
            ok_sh_date <= :rep_date_end and
            ok_from_c_id = 0
      group by oks_from_k_id, ok_sh_date, oks_delinq_date, oks_balance, is_rest, is_rest_period
      having sum(ok_os + ok_sh_os) > 0 or sum(ok_perc + ok_sh_perc) > 0),
  tabl3
  as (select cast(kc_amount_approved as double precision) / k_amount_approved * delinq_os as delinq_os, delinq_day,
             kc_from_c_id, k_id, k_kontract_no, k_kolich_client_approved,
             cast(kc_amount_approved as double precision) / k_amount_approved kc_doly,
             coalesce(k_stop_perc, '31.12.9999') k_stop_perc, oks_balance, is_rest, is_rest_period
      from tabl2, kontract, knclient
      where oks_from_k_id = k_id and
            k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            ((not(kc_amount_approved <= 45000 and
            k_period < 2) and
            :is_45 = 0) -- для второй случай нужно наоборот сделать
            or
            ((kc_amount_approved <= 45000 and
            k_period < 2) and
            :is_45 = 1))),
  tabl4
  as (select k_id, kc_from_c_id, delinq_day, kc_doly * oks_balance kc_balance, sum(delinq_os) delinq_os, k_stop_perc,
             is_rest, is_rest_period
      from tabl3
      group by k_id, kc_from_c_id, delinq_day, kc_doly, oks_balance, k_stop_perc, is_rest, is_rest_period),
  tabl5
  as (select delinq_day,
             iif(k_stop_perc <= cast(:rep_date_end as date) or delinq_day > 60, kc_balance, delinq_os) delinq_os,
             is_rest, is_rest_period
      from tabl4)
  select sum(delinq_os), sum(iif(delinq_day <= 30, delinq_os, 0)), sum(iif(delinq_day between 31 and 90, delinq_os, 0)),
         sum(iif(delinq_day > 90, delinq_os, 0)), sum(iif(is_rest = 1, delinq_os, 0)),
         sum(iif(is_rest_period = 1, delinq_os, 0))
  from tabl5

  into delinq_os, delinq_os_1_30, delinq_os_31_90, delinq_os_91_, delinq_os_rest, delinq_os_rest_period;
  if (delinq_os_rest is null) then
    delinq_os_rest = 0;
  if (delinq_os_rest_period is null) then
    delinq_os_rest_period = 0;

  res_balance_is_delinq = delinq_os_rest;
  res_balance_period_is_delinq = delinq_os_rest_period;

-- в т.ч объем портфеля микрозаймов без просроченной задолженности, тыс. руб.  (строка. 85)
  oks_balance_not_delinq = oks_balance - delinq_os;

--Объем необслуживаемых микрозаймов, тыс. руб. (строка 90)
  select sum(cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance)
  from oksummary, kontract, knclient
  where :rep_date_end between oks_date_begin and oks_date_end and
        cast(:rep_date_end as date) - oks_delinq_date > 90 and
        oks_from_k_id = k_id and
        k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        oks_close_state = 0 and
        ((not(kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 0) -- для второй случай нужно наоборот сделать
        or
        ((kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 1))
  into delinq_balance_91_;

--Объем сформированных резервов на возможные потери по микрозаймам, тыс. руб. (строка 91)
  select sum(cast(kc_amount_approved as double precision) / k_amount_approved * (lk_balance) / 100 * lk_percent),
         sum(iif(ks_date_begin >= '28.02.2012', cast(kc_amount_approved as double precision) / k_amount_approved * (coalesce(lk_calcperc, 0)) / 100 * lk_percent, 0))
  from llc_kont, kontract, knclient, kontstate, llc_main
  where :rep_date_end between lk_date and lk_date_end and
        lk_is_signed = 1
--  and lk_balance > 0
        and
        lk_from_k_id = k_id and
        k_id = ks_from_k_id and
        ks_ol_is_delete = 0 and
        ks_state = 1 and
        :rep_date_end between ks_date_begin and ks_date_end and
        k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        lm_id = lk_from_lm_id and
        lm_is_signed = 1 and
        ((not(kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 0) -- для второй случай нужно наоборот сделать
        or
        ((kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 1))
  into llc_sum_os, llc_sum_perc;

--Объем сформированных резервов на возможные потери по микрозаймам за минувший квартал, тыс. руб.(строка 92)
  select sum(cast(kc_amount_approved as double precision) / k_amount_approved * (lk_balance) / 100 * lk_percent),
         sum(iif(ks_date_begin >= '28.02.2012', cast(kc_amount_approved as double precision) / k_amount_approved * (coalesce(lk_calcperc, 0)) / 100 * lk_percent, 0))
  from llc_kont, kontract, knclient, kontstate, llc_main
  where :rep_date_begin - 1 between lk_date and lk_date_end and
        lk_is_signed = 1
--  and lk_balance > 0
        and
        k_id = ks_from_k_id and
        ks_ol_is_delete = 0 and
        ks_state = 1 and
        :rep_date_begin - 1 between ks_date_begin and ks_date_end and
        lk_from_k_id = k_id and
        k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        lm_id = lk_from_lm_id and
        lm_is_signed = 1 and
        ((not(kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 0) -- для второй случай нужно наоборот сделать
        or
        ((kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 1))
  into llc_sum_rashod_os, llc_sum_rashod_perc;
  llc_sum_rashod_os = llc_sum_os - llc_sum_rashod_os;
  llc_sum_rashod_perc = llc_sum_perc - llc_sum_rashod_perc;

--Объем реструктурированных микрозаймов, тыс. руб.
  with tabl1
  as (select distinct or_from_k_id
      from operrest, operlist, kontstate
      where or_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            :rep_date_end between ol_date and or_date_end and
            or_from_k_id = ks_from_k_id and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end

            and
            exists(select 1
                   from rdb$database

-- 5,22    Объём начисленных к погашению платежей (основной долг, проценты, штрафы и т.п.) от заёмщиков, за минувший квартал, тыс. руб.
  with tabl1
  as (select ks_from_k_id, ks_date_begin as k_disb_date
      from kontstate
      where ks_ol_is_delete = 0 and
            ((ks_state = 1 and
            ks_date_end >= :rep_date_begin and
            ks_date_begin <= :rep_date_end) or
            (ks_date_begin = :rep_date_begin and
            ks_state = 2))),
  tabl2
  as (select ok_from_k_id, sum(ok_os + ok_sh_os) ok_sh_os, sum(ok_perc + ok_sh_perc) ok_sh_perc
      from operkred, tabl1
      where ok_from_k_id = ks_from_k_id and
            ok_type < 8
--          and (ok_OL_Date < :rep_date_begin or ok_OL_Date <= k_disb_date)
            and
            ok_ol_date < maxvalue(:rep_date_begin, k_disb_date + 1) -- замена предыдущей строке
            and
            ok_sh_date between :rep_date_begin and :rep_date_end and
            ok_ol_is_active = 1 and
            ok_type <> 7 and
            ok_from_c_id = 0
      group by ok_from_k_id
      having(sum(ok_os + ok_sh_os) <> 0) or (sum(ok_perc + ok_sh_perc) <> 0))
  select sum(cast(kc_amount_approved as double precision) / k_amount_approved * ok_sh_os),

        or
        ((kc_amount_approved <= 45000 and
        k_period < 2) and
        :is_45 = 1))
  into plan_repay_os, plan_repay_perc;
