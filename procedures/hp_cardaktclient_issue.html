CREATE PROCEDURE HP_CARDAKTCLIENT_ISSUE(
    CAC_ID D_INTEGER NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable kal_new_kf_name d_str_100;
declare variable c_id d_id;
declare variable rd_id d_integer;
declare variable rd_name d_str_255;
declare variable is_prim_issue boolean;
declare variable reg_id d_integer;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable cac_status d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable dt_curr d_date;
declare variable curr_code d_str_lang;
declare variable ol_memo d_str_255;
declare variable is_pincover d_bool;
declare variable ol_id d_id;
declare variable ol_id1 d_id;
begin
  err_code = 0;
  err_text = '';

  dt_curr = current_date;
  curr_code = '417';
  select
    cac_from_k_id,
    cac_from_c_id,
    cac_from_rd_id,
    cac_from_reg_id,
    cac_card_id,
    cac_from_cca_id,
    cac_from_ol_id,
    cem_is_pincover,
    cac_status
  from cardaktclient
    left join refdata
      on rd_id = cac_from_rd_id
    inner join cardlist
      on cal_card_id = cac_card_id
    inner join cardembomain
      on cem_id = cal_from_cem_id
  where cac_id = :cac_id
  into
    :k_id,
    :c_id,
    :rd_id,
    :reg_id,
    :card_id,
    :cca_id,
    :ol_id,
    :is_pincover,
    :cac_status;

  if (row_count = 0) then
  begin
    err_code = 1;
    err_text = f_format('Акт выдачи карты №%1 не найден', cac_id);
  end

  if (err_text = '' and cac_status <> 3) then
  begin
    err_code = 2;
    err_text = f_format('Акт выдачи карты №%1 должен иметь статус "карта активирована"', cac_id);
  end

  if (err_text = '' and ol_id > 0
      and exists(select * from operlist where ol_id = :ol_id and is_delete = 0)) then
  begin
    err_code = 3;
    err_text = f_format('Для акта выдачи карты №%1 обнаружена бухгалтерская проводка', cac_id);
  end

  if (err_text = '') then
  begin
    is_prim_issue = rd_id = f_get_rd_id('ReceivCard_Primary');

    -- бух.проводка(карта)
    ol_id = null;
    acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/14');
    acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/14');

    select
      coalesce(rd_name, '?')
    from refdata
    where rd_id = :rd_id
    into
      :rd_name;
    select
      k_kontract_no,
      k_disb_date,
      kal_new_kf_name
    from kontract
      inner join knaktlist
        on kal_from_kf_id = k_from_kf_id
          and k_disb_date between kal_date and kal_date_end
          and is_delete = 0
    where k_id = :k_id
    into
      :k_kontract_no,
      :k_disb_date,
      :kal_new_kf_name;

    ol_memo =
      f_format(' ID %1 согл.акта №%2 %3', card_id, cac_id,
        coalesce(f_format('(кредит.договор №%1 от %2 Папка №%3)', k_kontract_no, f_datetostr(k_disb_date), kal_new_kf_name), ''));

    insert into operlist (
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id,
      ol_document_client
     )
    values (
      :reg_id,
      2,
      2,
      :dt_curr,
      :cac_id,
      0,
      f_format('Выдана карта (%1)', :rd_name) || :ol_memo,
      :un_id,
      :c_id
    )
    returning ol_id
    into :ol_id;

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_dt,
      :acc_kt,
      1,
      1
    );

    -- бух.проводка(ПИН - конверт)
    if (is_pincover = 1) then
    begin
      acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/15');
      acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/15');

      insert into operlist (
        ol_parent_id,
        ol_from_reg_id,
        ol_oper_type,
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_memo,
        ol_user_id,
        ol_document_client
       )
      values (
        :ol_id,
        :reg_id,
        2,
        2,
        :dt_curr,
        :cac_id,
        0,
        'Выдан пин-конверт от карты' || :ol_memo,
        :un_id,
        :c_id
      )
      returning ol_id
      into :ol_id1;

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id1,
        :curr_code,
        1,
        :acc_dt,
        :acc_kt,
        1,
        1
      );
    end

    update cardaktclient set
      cac_from_ol_id = :ol_id,
      cac_status = 4
    where cac_id = :cac_id;

    update cardlist set
      cal_status = f_enum('cardlist_status', 'transfer_to_client', 'id'),
      cal_from_cca_id = :cca_id,
      cal_date_issue = :dt_curr
    where cal_card_id = :card_id;

    if (not exists(
      select *
      from cardlistlnk
      where cll_card_id = :card_id))
    then
      insert into cardlistlnk (
        cll_card_id,
        cll_from_c_id,
        cll_from_k_id)
      values (
        :card_id,
        :c_id,
        iif(:is_prim_issue, null, :k_id));
  end

  suspend;
end
