CREATE PROCEDURE AAREP_CALC_PERC_OPERBOOK(
    REPORT_REGION_ID D_INTEGER,
    REPORT_DATE_BEGIN D_DATE,
    REPORT_DATE_END D_DATE,
    REPORT_NEED_PERC D_INTEGER)
RETURNS (
    AAREP_BALANCE_BEGIN D_CURRENCY,
    AAREP_PERC_BEGIN D_CURRENCY,
    AAREP_PERC_CALCED D_CURRENCY,
    AAREP_PERC_PAYED D_CURRENCY,
    AAREP_PERC_WRITEOFF D_CURRENCY,
    AAREP_PERC_END D_CURRENCY,
    AAREP_BALANCE_END D_CURRENCY,
    AAREP_DISB_AMOUNT D_CURRENCY,
    AAREP_CALCED_ALL D_CURRENCY,
    AAREP_PAYED_ALL D_CURRENCY,
    AAREP_WRITEOFF_ALL D_CURRENCY,
    AAREP_OFFICE D_STR_100,
    AAREP_PRODUCT_ID D_INTEGER,
    AAREP_FOND D_STR_20,
    AAREP_OFFICE_ID D_INTEGER,
    AAREP_ACCOUNT_PROFIT D_STR_20,
    AAREP_ACCOUNT_CALCED D_STR_20,
    AAREP_ACCOUNT_REPAYED D_STR_20,
    AAREP_ACCOUNT_PORTFOLIO D_STR_20,
    AAREP_OUT_PERC_BEGIN D_CURRENCY,
    AAREP_OUT_PERC_CALCED D_CURRENCY,
    AAREP_OUT_PERC_END D_CURRENCY,
    AAREP_IS_MBH D_BOOL)
AS
declare variable db_id d_integer;
declare variable is_new_variant d_integer;
begin
  db_id = f_get_db_id();

  -- Переключатель между старым и новым вариантом
  is_new_variant = 2;

  if (is_new_variant = 2) then
  begin
    for
      select
        aarep_office,
        aarep_product_id,
        aarep_fond,
        aarep_office_id,
        sum(aarep_balance_begin),
        sum(aarep_perc_begin),
        sum(aarep_perc_calced),                               
        sum(aarep_perc_payed),
        sum(aarep_perc_writeoff),
        sum(aarep_perc_end),
        sum(aarep_balance_end),
        sum(aarep_disb_amount),
        sum(aarep_calced_all),
        sum(aarep_payed_all),
        sum(aarep_writeoff_all),
        max(aarep_account_profit),
        max(aarep_account_calced),
        max(aarep_account_repayed),
        max(aarep_account_portfolio),
        sum(aarep_out_perc_begin),
        sum(aarep_out_perc_calced),
        sum(aarep_out_perc_end)
      from aarep_calc_perc_operbook_tmp(:report_region_id, :report_date_begin, :report_date_end, :report_need_perc)
      group by
        aarep_office,
        aarep_product_id,
        aarep_fond,
        aarep_office_id
    into
      :aarep_office,
      :aarep_product_id,
      :aarep_fond,
      :aarep_office_id,
      :aarep_balance_begin,
      :aarep_perc_begin,
      :aarep_perc_calced,
      :aarep_perc_payed,
      :aarep_perc_writeoff,
      :aarep_perc_end,
      :aarep_balance_end,
      :aarep_disb_amount,
      :aarep_calced_all,
      :aarep_payed_all,
      :aarep_writeoff_all,
      :aarep_account_profit,
      :aarep_account_calced,
      :aarep_account_repayed,
      :aarep_account_portfolio,
      :aarep_out_perc_begin,
      :aarep_out_perc_calced,
      :aarep_out_perc_end
    do
    begin
      aarep_is_mbh = iif(db_id in (39801, 41701) and exists(
        select *
        from product
        where pr_id = :aarep_product_id
          and pr_code containing '.murabaha.'), 1, 0);
      suspend;
    end
  end

  if (is_new_variant = 1) then
  begin
    for
      select
        aarep_office,
        aarep_product_id,
        aarep_fond,
        aarep_office_id,
        aarep_account_profit,
        aarep_account_calced,
        aarep_account_repayed,
        aarep_account_portfolio,
        sum(aarep_balance_begin),
        sum(aarep_perc_begin),
        sum(aarep_perc_calced),
        sum(aarep_perc_payed),
        sum(aarep_perc_writeoff),
        sum(aarep_perc_end),
        sum(aarep_balance_end),
        sum(aarep_disb_amount),
        sum(aarep_calced_all),
        sum(aarep_payed_all),
        sum(aarep_writeoff_all),
        sum(aarep_perc_begin),
        sum(aarep_perc_calced),
        sum(aarep_perc_end)
      from aarep_calc_perc_operbook_tmp2(:report_region_id, :report_date_begin, :report_date_end, :report_need_perc)
      group by
        aarep_office,
        aarep_product_id,
        aarep_fond,
        aarep_office_id,
        aarep_account_profit,
        aarep_account_calced,
        aarep_account_repayed,
        aarep_account_portfolio
    into
      :aarep_office,
      :aarep_product_id,
      :aarep_fond,
      :aarep_office_id,
      :aarep_account_profit,
      :aarep_account_calced,
      :aarep_account_repayed,
      :aarep_account_portfolio,
      :aarep_balance_begin,
      :aarep_perc_begin,
      :aarep_perc_calced,
      :aarep_perc_payed,
      :aarep_perc_writeoff,
      :aarep_perc_end,
      :aarep_balance_end,
      :aarep_disb_amount,
      :aarep_calced_all,
      :aarep_payed_all,
      :aarep_writeoff_all,
      :aarep_out_perc_begin,
      :aarep_out_perc_calced,
      :aarep_out_perc_end
    do
    begin
      aarep_is_mbh = iif(db_id in (39801, 41701) and exists(
        select *
        from product
        where pr_id = :aarep_product_id
          and pr_code containing '.murabaha.'), 1, 0);
      suspend;
    end
  end
end
