CREATE PROCEDURE OKT_INCOME_ABONENT_CHECK(
    PHONE D_STR_20)
RETURNS (
    OTL_ID D_INTEGER,
    UN_ID D_ID,
    UN_NAME D_STR_100,
    URL D_STR_2048,
    PCM_ID D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable PANT_ID D_INTEGER;
declare variable PROJECT D_STR_100;
declare variable K_ID D_ID;
declare variable PHONE_TYPE D_STR_100;
declare variable FORM_NAME D_STR_20;
declare variable CLIENT_PHONE D_STR_20;
declare variable CC_ID D_INTEGER;
declare variable OI_ID D_INTEGER;
begin
  insert into okt_income (oi_phone)
  values (:phone)
  returning oi_id
  into :oi_id;

  if (character_length(phone) = 10) then phone = '7'||phone;

  -- ищем физ лицо
  select res_object_id
  from hp_contdata_find_object(:phone, 'cell')
  where res_object_kind = 0
  rows 1
  into :c_id;

  if (c_id is not null) then  -- если клиент найден, проверяем его среди клиентов по проекту Сирена
  begin
        for select pank.pant_id,
                   un_id,
                   un.un_name,
                   rd.rd_code,
                   pank.pant_from_k_id
        from pantera_kontracts pank
        inner join knclient kncl on kncl.kc_from_k_id = pank.pant_from_k_id
        inner join refdata rd on rd.rd_id = pank.pant_rd_project_type
        inner join username un on un.un_id = pank.pant_from_un_id
        inner join oksummary oks on oks.oks_from_k_id = pank.pant_from_k_id
        where current_date between pank.pant_date_begin and pank.pant_date_end
        and kncl.kc_from_c_id = :c_id
        and current_date between oks.oks_date_begin and oks.oks_date_end
        rows 1
        into :pant_id, :un_id, :un_name, :project, :k_id
        do
        begin
            if (un_id = 1010100000278) then -- igusev - это виртуальный оператор, на которого перемещаются группы, которые вышли из просрочки
                exit;

            pcm_id = null;

            select pco.pco_from_pcm_id
            from pantera_commands_operators pco
            where pco.pco_from_un_id = :un_id
            and current_date between pco.pco_date_begin and pco.pco_date_end
            into :pcm_id;

            phone_type = 'ContactInfo.Cell_Main';
            if (phone <> f_cont_info('clients', :c_id, 'cell_main', 'last')) then
                phone_type = 'ContactInfo.Cell_Add';

            form_name = 'Sirena_d.aspx';

            if (exists (
              select res_un_id, pa_id, pa_name, pa_command_name
              from okt_permission_access(:un_name)
              where pa_command_name = 'oktell.projects.blind'
            )) then
                form_name = 'Sirena_d_lite.aspx'; -- < если оператор из незрячих, для него отрывается специфичная форма

            insert into okt_income_task_list (otl_from_pant_id, otl_object_kind, otl_object_id, otl_task_status, otl_pant_un_id, otl_phone, otl_is_main_phone_no)
            values (:pant_id, f_objkind('clients', 'id'), :c_id, 0, :un_id, :phone, iif(:phone_type = 'ContactInfo.Cell_Main', 1, 0))
            returning otl_id
            into :otl_id;

            url = 'http://oktell.molbulak.ru/'||:form_name||'?project='||:project||'&pant_id='||:pant_id||'&object_id='||:c_id||'&object_kind=Clients&client_cell_no='||:phone||'&K_ID='||:k_id||'&cont_call_active=0&phone_type='||:phone_type||'&call_system=OKTELL&un_id='||:un_id||'&otl_id='||:otl_id||'&un_name_new=';
            suspend;
        end
  end


  if ((c_id is null) or (pant_id is null)) then  -- ищу среди контантных лиц
  begin
    delete from okt_income_clients_tmp  where 1 = 1;

    insert into okt_income_clients_tmp (oic_from_c_id, oic_from_cc_id)
    select distinct cont.cc_from_c_id,
                    cont.cc_id
    from contacts cont
    where ((cont.cc_total_res = 1) or (cont.cc_verif_ignore = 1)) -- беру только верифицированные + которых не верифицировали из-за галочки "не верифицировать"
    and cont.cc_is_old_contact = 0 -- беру только последние контакты
    and cont.cc_cell_no = :phone; -- '996779967331'

    for select pank.pant_id,
               un_id,
               un.un_name,
               rd.rd_code,
               pank.pant_from_k_id,
               kncl.kc_from_c_id,
               t.oic_from_cc_id
    from pantera_kontracts pank
    inner join knclient kncl on kncl.kc_from_k_id = pank.pant_from_k_id
    inner join refdata rd on rd.rd_id = pank.pant_rd_project_type
    inner join username un on un.un_id = pank.pant_from_un_id
    inner join oksummary oks on oks.oks_from_k_id = pank.pant_from_k_id
    inner join okt_income_clients_tmp t on t.oic_from_c_id = kncl.kc_from_c_id
    where current_date between pank.pant_date_begin and pank.pant_date_end
    and current_date between oks.oks_date_begin and oks.oks_date_end
    and un_id <> 1010100000278
    rows 1
    into :pant_id, :un_id, :un_name, :project, :k_id, :c_id, :cc_id
    do
    begin
        pcm_id = null;

        select pco.pco_from_pcm_id
        from pantera_commands_operators pco
        where pco.pco_from_un_id = :un_id
        and current_date between pco.pco_date_begin and pco.pco_date_end
        into :pcm_id;

        form_name = 'Sirena_d.aspx';

        if (exists (
          select res_un_id, pa_id, pa_name, pa_command_name
          from okt_permission_access(:un_name)
          where pa_command_name = 'oktell.projects.blind'
        )) then
            form_name = 'sirena_d_lite.aspx'; -- < если оператор из незрячих, для него отрывается специфичная форма

        client_phone = coalesce(f_cont_info('clients', :c_id, 'cell_main', 'last'), '');

        insert into okt_income_task_list (otl_from_pant_id, otl_object_kind, otl_object_id, otl_task_status, otl_pant_un_id, otl_phone, otl_is_main_phone_no)
        values (:pant_id, f_objkind('contacts', 'id'), :cc_id, 0, :un_id, :phone, 0)
        returning otl_id
        into :otl_id;

        url ='http://oktell.molbulak.ru/'||form_name||'?project='||:project||'&pant_id='||:pant_id||'&object_id='||:cc_id||'&object_kind=Contacts&contact_cell_no='||:phone||'&client_cell_no='||:client_phone||'&c_id='||:c_id||'&K_ID='||:k_id||'&phone_type=ContactInfo.Cell_Main&call_system=OKTELL&un_id='||:un_id||'&otl_id='||:otl_id||'&un_name_new=';
        suspend;
    end
  end

  if (otl_id is not null) then
  begin
     update okt_income
     set oi_from_otl_id = :otl_id
     where (oi_id = :oi_id);
  end
  else
  begin
    pcm_id = -1;
    url ='http://oktell.molbulak.ru/SirenaPhoneFrm.aspx?';
    suspend;    
  end
end
