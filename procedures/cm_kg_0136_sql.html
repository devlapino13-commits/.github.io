CREATE PROCEDURE CM_KG_0136_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable C_ID ID_TYPE;
declare variable PR_CODE D_STR_100;
begin
  if (K_ID = 109010239773700101) then
    exit; ---RFCKG-613
  if (:k_id = 109010190645900101) then
    exit;--по письму от Чолпон 12,07,2022
  if (k_id = 109010242287300101) then
    exit; -- RFCKG-689
  if (k_id = 109010248810200101) then
    exit; --RFCKG-847
  if (k_id = 109010294303800101) then
    exit; --RFCKG-2905

  for select kc.kc_from_c_id, pr.pr_code
      from knclient kc
      inner join kontract k on kc.kc_from_k_id = k.k_id
      inner join product pr on k.k_from_pr_id = pr.pr_id
      where kc_from_k_id = :k_id and
            kc_amount_approved > 100000
      into :c_id, :pr_code
  do
  begin
    if (not exists(select *
                     from extinfolist eil
                     where eil.eil_from_c_id = :c_id and
                           eil.eil_from_eic_id = 10 and
                           eil.eil_status = 2 and
                           coalesce(eil.eil_kk_un_id, 0) = 0)) then
    begin
      if (not exists(select *
                     from clientdocs cdc
                     inner join passkind pk on pk.pk_id = cds_from_pk_id
                     inner join passdoctype pdt on pk.pk_from_pdt_id = pdt.pdt_id and
                     pdt.pdt_code in ('income_statement', 'place_of_business',  'certificate_of_employment') -- Справка о доходах, фото бизнеса, cправка с места работы (RFCKG-3576)
                     where cds_from_c_id = :c_id and
                           cds_is_delete = 0 and
                           cast(cds_create_datetime as date) between current_date - 5 and current_date -- Дата загр документа (cds_create_datetime)
      )) then
      begin
        res_error_text = 'Отсутствует справка о доходах/фото бизнеса/cправка с места работы.';
        res_error_is_critical = 0; -- Временно отлючила по просьбе Азамата Махмутова (12.04.2021 вр. 14:28)
        suspend;
      end
    end
  end
end
