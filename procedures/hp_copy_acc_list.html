CREATE PROCEDURE HP_COPY_ACC_LIST(
    ORIG_OFFICE_ID INTEGER,
    NEW_OFFICE_ID INTEGER,
    ORIG_FIND_STR VARCHAR(100),
    NEW_STR VARCHAR(100))
AS
declare variable al_from_reg_id integer;
declare variable al_date date;
declare variable al_from_bg_id varchar(20);
declare variable al_currency varchar(3);
declare variable al_number integer;
declare variable al_type integer;
declare variable al_memo varchar(255);
declare variable al_date_close date;
declare variable al_is_active smallint;
declare variable is_delete smallint;
declare variable al_from_bl_id varchar(10);
declare variable al_account varchar(32);
declare variable new_office_code varchar(20);
declare variable al_id varchar(20);
declare variable al_id_new varchar(20);
declare variable al_name varchar(200);
begin
  select reg_code
  from region
  where reg_unicode = :new_office_id
  into :new_office_code;

  for select al_id, al_from_reg_id, al_date, al_from_bg_id, al_currency, al_number, al_type, al_name, al_memo,
             al_date_close, al_is_active, is_delete, al_from_bl_id, al_account
      from acc_list
      where al_from_reg_id = :orig_office_id
      order by al_id
      into :al_id, :al_from_reg_id, :al_date, :al_from_bg_id, :al_currency, :al_number, :al_type, :al_name, :al_memo,
           :al_date_close, :al_is_active, :is_delete, :al_from_bl_id, :al_account
  do
  begin
    al_id_new = null;
    select al_id
    from acc_list
    where al_id = :new_office_code || right(:al_id, 14)
    into :al_id_new;
    if (al_id_new is null) then
    begin
      al_id_new = :new_office_code || right(:al_id, 14);
      insert into acc_list (al_id, al_from_reg_id, al_date, al_from_bg_id, al_currency, al_number, al_type, al_name,
                            al_memo, al_date_close, al_is_active, is_delete, al_from_bl_id)
      values (:new_office_code || right(:al_id, 14), :new_office_id, :al_date, :al_from_bg_id, :al_currency, :al_number,
              :al_type, replace(:al_name, :orig_find_str, :new_str), :al_memo, :al_date_close, :al_is_active,
              :is_delete, :al_from_bl_id);
    end
  end
end
