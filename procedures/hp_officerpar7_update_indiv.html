CREATE PROCEDURE HP_OFFICERPAR7_UPDATE_INDIV(
    REP_DATE_END DATE)
AS
declare variable oks_id decimal(18,0);
declare variable oks_balance decimal(18,4);
declare variable oks_delinq_balance7 decimal(18,4);
declare variable oks_balance_c decimal(18,4);
declare variable par7 decimal(18,4);
declare variable par7_table decimal(18,4);
declare variable categ_temp smallint;
begin
  for select oks_kc_id * 100 + 1, -- если в ID КСника в конце "01" то это по продуктам 1+1
             sum(oks_balance) as oks_balance,
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0)) as delinq_balance7,
             sum(iif(cast(:rep_date_end as date) - k_disb_date < 30, oks_balance, 0)), 0 as categ_temp
      from oksummary, kontract, product
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id
--    and pr_common_name like 'пп индив%'
            and
            pr_code containing '.one.' and
            oks_close_state = 0
      group by oks_kc_id
      union
      select oks_kc_id, -- в ID КСника если индив ПП
             sum(oks_balance) as oks_balance,
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0)) as delinq_balance7,
             sum(iif(cast(:rep_date_end as date) - k_disb_date < 30, oks_balance, 0)), 1 as categ_temp
      from oksummary, kontract --, product
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_from_k_id = k_id
--and k_from_pr_id = pr_id
-- and pr_common_name like 'пп индив%'  <<-- закоментировал igusev 10.03.2015
            and
            oks_close_state = 0
      group by oks_kc_id
      into oks_id, oks_balance, oks_delinq_balance7, oks_balance_c, categ_temp
  do
  begin
--    Вычисляет PAR7
    if (oks_balance - oks_balance_c > 0) then
      par7 = cast(oks_delinq_balance7 as double precision) / (oks_balance - oks_balance_c) * 100;
    else
    if (oks_delinq_balance7 > 0) then
      par7 = 100;
    else
      par7 = 0;
--    из таблицы OFFICERPAR7 предыдущий PAR7 вытаскивает
    par7_table = null;
    select op_par8_180
    from officerpar7
    where op_kc_id = :oks_id and
          :rep_date_end between op_date_begin and op_date_end
    into par7_table;
    if (par7_table is null) then
      par7_table = 0;

    if (categ_temp = 0) then
    begin
      if (par7 > 5.0000 and
          par7_table < 5.0000) then
      begin
        insert into officerpar7 (op_kc_id, op_date_begin, op_par8_180)
        values (:oks_id, :rep_date_end + 1, :par7);
      end
      else
      if (par7 <= 3.0000 and
          par7_table > 3.0000) then
      begin
        insert into officerpar7 (op_kc_id, op_date_begin, op_par8_180)
        values (:oks_id, :rep_date_end + 1, :par7);
      end
    end
    else
    if (categ_temp = 1) then
    begin
      if (par7 > 7.0000 and
          par7_table < 7.0000) then
      begin
        insert into officerpar7 (op_kc_id, op_date_begin, op_par8_180)
        values (:oks_id, :rep_date_end + 1, :par7);
      end
      else
      if (par7 <= 7.0000 and
          par7_table > 7.0000) then
      begin
        insert into officerpar7 (op_kc_id, op_date_begin, op_par8_180)
        values (:oks_id, :rep_date_end + 1, :par7);
      end
    end

--      suspend;
  end
end
