CREATE PROCEDURE CM_AZ_0007_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kf_disb_date d_date;
declare variable pr_product_kind integer;
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_amount_approved curr_type;
declare variable kf_id id_type;
declare variable pr_loan_type integer;
declare variable max_delinq_day integer;
declare variable last_k_id id_type;
declare variable delinq_day integer;
begin
  for
  select pr_product_kind,
         ma_kolich_repay,
         ma_kolich_client,
         ma_summa_minus,
         ma_from_kf_id,
         ma_date_oper,
         pr_loan_type
  from miniapp, product
  where ma_id=:ma_id
  and ma_from_pr_id = pr_id
  and ma_date_end='31.12.9999'
  into pr_product_kind,
       k_kolich_repay,
       k_kolich_client_approved,
       k_amount_approved,
       kf_id,
       kf_disb_date,
       pr_loan_type
  do
  begin
    if (pr_loan_type <> 0) then
    begin
      exit;
    end 
        last_k_id = null;
        select k_id from kontstate, kontract
        where ks_ol_is_delete = 0
        and ks_state = 1
        and cast(ks_date_begin as date) < :kf_disb_date
        and ks_from_k_id = k_id
        and k_from_kf_id = :kf_id
        order by ks_date_begin desc
        rows 1
        into last_k_id;
        if (last_k_id is null) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Yeni qrup kreditlərinin buraxılışı 29.10.2014 tarixindən dayandırılıb/Выдачи новым группам запрещены от 29.10.2014';
          suspend;
        end else
        begin
--            проверка не имеется ли в предыдущем цикле просроченный платеж
          max_delinq_day = null;
          select max(ok_ol_date - ok_sh_date) from operkred
          where ok_ol_is_active = 1
          and ok_from_k_id = :last_k_id
          and ok_ol_date <= :kf_disb_date
          and ok_type in (4,5)
          and ok_os + ok_perc <> 0
          and ok_ol_date > ok_sh_date
          and ok_from_c_id=0
          into max_delinq_day;
          if (max_delinq_day is null) then max_delinq_day = 0;
          delinq_day = null;
          select (:kf_disb_date - oks_delinq_date) from oksummary
          where oks_from_k_id = :last_k_id
          and :kf_disb_date between oks_date_begin and oks_date_end
          and :kf_disb_date > oks_delinq_date
          and oks_close_state = 0
          into delinq_day;
          if (delinq_day is null) then delinq_day = 0;
          if (max_delinq_day + delinq_day > 0) then
          begin
            res_error_is_critical =1;
            res_error_text = 'Keçmiş dövrdə müqavilənin gecikmə günləri mövcüd olduğundan buraxılış mümkün deyil/В предыдущем цикле данного контракта имеются просроченные дни, выдача запрещена!';
            suspend;
          end 
        end
  end
end
