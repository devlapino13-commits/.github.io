    CZ_NON_DELINQ_CALC_PERC DECIMAL(18,4),
    CZ_DELINQ_ALL DECIMAL(18,4),
    CZ_NON_DELINQ_ALL DECIMAL(18,4),
    CZ_NON_DELINQ_OS DECIMAL(18,4),
    CZ_DELINQ_OS DECIMAL(18,4),
    CZ_DELINQ_PERC DECIMAL(18,4),
    CZ_FINE DECIMAL(18,4),
    CZ_OS_TOTAL DECIMAL(18,4),
    CZ_PERC_TOTAL DECIMAL(18,4),
    CZ_TOTAL_RES DECIMAL(18,4),
    CZ_CL_OS DECIMAL(18,4),
    CZ_CL_PERC DECIMAL(18,4),
    CZ_CL_FINE DECIMAL(18,4),
    CZ_CL_ALL DECIMAL(18,4),
    CZ_CL_OS_PERC DECIMAL(18,4),
    CZ_CL_DELINQ_OS DECIMAL(18,4),
    CZ_CL_NON_DELINQ_OS DECIMAL(18,4),
    CZ_CL_BALANCE DECIMAL(18,4),
    CZ_CL_NON_DELINQ_CALC_PERC DECIMAL(18,4),
    CZ_CL_NON_DELIQ_OS_CALC_PERC DECIMAL(18,4),
    CZ_CL_NON_DELINQ_TOTAL DECIMAL(18,4),
    CZ_CL_DELINQ_TOTAL DECIMAL(18,4),
    CZ_CL_OS_TOTAL DECIMAL(18,4),
    CZ_CL_PERC_TOTAL DECIMAL(18,4),
    CZ_CL_TOTAL_RES DECIMAL(18,4),
    KSR_KONTRACTNO_AND_DISB VARCHAR(1024),
    KSR_SUD_POSHLINA_INFO STR_255,
    BRT_IS_FINISHED D_BOOL,
    KSR_DISB_DATE_LIST D_STR_255,
    KSR_KONT_PHONE STR_100)
AS
declare variable reg_id integer;
declare variable sign_c_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable cz_os decimal(18,4);
declare variable cz_perc decimal(18,4);
declare variable sh_min_date date;
declare variable sh_date date;
declare variable sh_calc_fine decimal(18,4);
declare variable sh_calc_perc decimal(18,4);
declare variable zad_os decimal(18,4);
declare variable kont_amount decimal(18,4);
declare variable rep_date date;
declare variable tmp decimal(18,4);
declare variable del_days integer;
declare variable kc_closed_ol_id decimal(18,0);
declare variable closed_own_debt smallint;
declare variable oko_sh_os curr_type;
declare variable oko_sh_perc curr_type;
declare variable count_kont_cli integer;
declare variable k_id_cli id_type;
declare variable ksr_kontract_no_temp varchar(255);
declare variable ksr_kf_name_temp varchar(255);
declare variable ksr_deliq_days_temp varchar(255);
declare variable ksr_disb_date_str varchar(255);
declare variable has_active smallint;
declare variable kf_id decimal(18,0);
declare variable cur_delinq_days_temp integer;
declare variable bln_id integer;
declare variable db_id integer;
declare variable k_is_devided smallint;
declare variable k_base_k_id id_type;
declare variable ksr_sud_poshlina decimal(18,2);
declare variable oek_rd_id integer;
declare variable has_sum_debt decimal(18,2);
declare variable has_writeoff_debt integer;
declare variable ks_state integer;
declare variable group_spec integer;
declare variable cz_non_delinq_os_one curr_type;
declare variable cz_non_delinq_calc_perc_one curr_type;
declare variable cz_non_delinq_all_one curr_type;
declare variable cz_os_one curr_type;
declare variable cz_fine_one curr_type;
declare variable cz_perc_one curr_type;
declare variable cz_delinq_os_one curr_type;
declare variable cz_delinq_all_one curr_type;
declare variable cz_delinq_perc_one curr_type;
declare variable ksr_sud_poshlina_new decimal(18,2);
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  db_id = f_get_db_id();
  select prc_from_reg_id, prc_sign_by_c_id, prc_from_k_id, prc_from_c_id, prc_no, prc_date, prc_for_whom,
         prc_from_bln_id
  from printcert
  where prc_id = :prc_id
  into :reg_id, :sign_c_id, :k_id, :c_id, :sp_no, :sp_date, :sp_company, bln_id;

  -- Если по договору была допущена непрерывная просрочка от 120 дней.
  if (
      (
        (
          db_id <> 39801
          and rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN') not in (
              'SDANILOVA', 'EASYAMOLOVA', 'ZGITINOVA', 'LABDURAKHMANOVA', 'CHMYKTYBEKOVA', 'MNAZANOV',
              'RSHIGAPOV', 'TLUKINA', 'MIVANOV', 'AKUZMIN', 'IONISCHENKO', 'GERALIEVA'
          )
        )
        or
        (
          db_id = 39801 -- RFCKZ-627: для КЗ свои исключения
          and rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN') not in (
              'ZGITINOVA', 'ZHBATYROVA'
          )
        )
      )
      and exists(
          select *
          from operkred
          where ok_ol_is_active = 1
            and ok_from_k_id = :k_id
            and ok_type in (4, 5)
            and ok_ol_date - ok_sh_date >= 120
            and ok_os + ok_perc < 0
      )
  ) then
  begin
    exception e_message 'За справкой незамедлительно обратитесь в Операционный отдел';
  end


  if (sp_company is null) then
    sp_company = '';

  select distinct ksr_id, ksr_kont_titul, ksr_kont_postion, ksr_other_1, ksr_other_2, ksr_other_3, c_name_fio,
                  ksm_from_c_id, ksr_from_reg_id, reg_code || ' ' || reg_name, al.adl_address_str, reg_kont_company,
                  reg_regist_no, c_pass_no, c_pass_date, c_pass_who, c_inn, ac.adl_address_str, af.adl_address_str

  from kontsignreg, clients, kontsignmen, region_list(:reg_id, 1, -1, 0), region, addrlist al, addrlist af, addrclient, addrlist ac
  where (id_list = ksr_from_reg_id) and
        ksm_id = ksr_from_ksm_id and
        ksm_from_c_id = c_id and
        reg_unicode = :reg_id and
        reg_address_legal = al.adl_id and
        reg_address_fact = af.adl_id and
        ac.adl_id = adc_from_adl_id and
        adc_from_c_id = c_id and
        adc_address_type = 1 and
        c_id = :sign_c_id and
        coalesce(ksr_date_begin, '01.01.1990') <= current_date and
        coalesce(ksr_date_end, '31.12.9999') >= current_date rows 1
  into :ksr_id, :ksr_kont_titul, :ksr_kont_position, :ksr_other_1, :ksr_other_2, :ksr_other_3, :ksr_fio, :ksr_from_c_id,
       :ksr_from_reg_id, :ksr_region_name, :ksr_region_address, :ksr_region_company, :ksr_regist_no, :ksr_pass_no,
       :ksr_pass_date, :ksr_pass_who, :ksr_inn, :ksr_full_adress, :ksr_reg_address;

  sh_min_date = null;
  rep_date = sp_date;
  cz_cl_os = 0;
  cz_cl_perc = 0;
  cz_cl_fine = 0;
  cz_del_days = 0;
  cz_cl_delinq_os = 0;
  cz_cl_non_delinq_os = 0;
  cz_cl_all = 0;
  cz_cl_os_perc = 0;
  cz_cl_balance = 0;
  cz_cl_non_delinq_calc_perc = 0;

  cz_cl_non_deliq_os_calc_perc = 0;

  cz_os = 0;
  cz_fine = 0;
  cz_perc = 0;
  cz_delinq_os = 0;
  cz_delinq_all = 0;
  cz_delinq_perc = 0;
  cz_non_delinq_os = 0;
  cz_non_delinq_calc_perc = 0;
  cz_non_delinq_all = 0;

  cz_inn = '';
  select c_name_fio, cl.c_pass_no, cl.c_pass_who, cl.c_pass_date, c_inn
  from clients cl
  where c_id = :c_id
  into :cz_name_fio, :cz_pass_no, :cz_pass_who, :cz_pass_date, cz_inn;

  ksr_kf_name = '';
  ksr_deliq_days = '';
  ksr_kontract_no = '';

  has_active = 0;
  ksr_disb_date_list = '';

  for select distinct ks_from_k_id, kncl.kc_closed_ol_id, k_kontract_no, kal_new_kf_name, cast((
                      case
                        when datediff(day, oks_delinq_date, :rep_date) < 0 then 0
                        else datediff(day, oks_delinq_date, :rep_date)
                      end) as varchar(50)),
                      kont.k_disb_date, kont.k_is_devided, kont.k_base_k_id
      from kontstate, knclient kncl, kontract kont, knaktlist, oksummary oks, product
      where kc_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_date_begin <= :rep_date and
            ks_date_end >= :rep_date and
            ks_state = 1 and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id and
            k_from_kf_id = knaktlist.kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date between kal_date and kal_date_end and
            k_id = oks_from_k_id and
            oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end and
            k_from_pr_id = pr_id and
            (:db_id <> 41701 or
            (:bln_id = 1186 and
            pr_code containing '.gr.') or
            (:bln_id = 1353 and
            pr_code containing '.ind.' and
            pr_code not containing '.ipf.') or
            (:bln_id = 1403 and
            pr_code containing '.ipf.'))
      into :k_id, :kc_closed_ol_id, :ksr_kontract_no_temp, :ksr_kf_name_temp, :ksr_deliq_days_temp, :ksr_disb_date,
           k_is_devided, k_base_k_id
  do
  begin
    if (closed_own_debt = 2) then
    begin
      cz_cl_perc = 0;
      cz_delinq_perc = 0;
    end
    has_active = 1;
    if (ksr_kf_name <> '') then
      ksr_kf_name = ksr_kf_name || '; ' || ksr_kf_name_temp;
    else
      ksr_kf_name = ksr_kf_name_temp;

    if (ksr_deliq_days <> '') then
      ksr_deliq_days = ksr_deliq_days || '; ' || ksr_deliq_days_temp;
    else
      ksr_deliq_days = ksr_deliq_days_temp;

    ksr_disb_date_str = iif(extract(day from ksr_disb_date) < 10, '0' || extract(day from ksr_disb_date), extract(day from ksr_disb_date)) || '.' || iif(extract(month from ksr_disb_date) < 10, '0' || extract(month from ksr_disb_date), extract(month from ksr_disb_date)) || '.' || extract(year from ksr_disb_date);

    if (ksr_kontract_no <> '') then
    begin
      ksr_kontractno_and_disb = ksr_kontractno_and_disb || '; ' || ksr_kontract_no_temp || ' от ' || :ksr_disb_date_str;
      ksr_kontract_no = ksr_kontract_no || '; ' || ksr_kontract_no_temp;
    end
    else
    begin
      ksr_kontract_no = ksr_kontract_no_temp;
      ksr_kontractno_and_disb = ksr_kontract_no_temp || ' от ' || :ksr_disb_date_str;
    end

    ksr_disb_date_list = ksr_disb_date_list || f_datetostr(:ksr_disb_date) || ' г.;';

    closed_own_debt = 0;
    sh_min_date = null;

    cz_os_one = 0;
    cz_fine_one = 0;
    cz_perc_one = 0;
    cz_delinq_os_one = 0;
    cz_delinq_all_one = 0;
    cz_delinq_perc_one = 0;

    cz_non_delinq_os_one = 0;
    cz_non_delinq_calc_perc_one = 0;
    cz_non_delinq_all_one = 0;

    for select sh_date, sh_calc_fine, sh_calc_perc, zad_os
        from aarep_make_zad_list(:k_id, :rep_date, 1)
        where sh_date is not null
        into :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os
    do
    begin
      if ((sh_min_date is null) and
          (sh_date < :rep_date)) then
      begin
        if ((zad_os <> 0) or (sh_calc_perc <> 0) or (sh_calc_fine <> 0)) then
          sh_min_date = :sh_date;
      end
      if (sh_date <= :rep_date) then
      begin
        cz_os_one = cz_os_one + zad_os;
        cz_fine_one = cz_fine_one + sh_calc_fine;
        cz_perc_one = cz_perc_one + sh_calc_perc;
        cz_delinq_os_one = cz_delinq_os_one + zad_os;
        cz_delinq_all_one = cz_delinq_all_one + cz_delinq_all + zad_os + sh_calc_perc + sh_calc_fine;
        cz_delinq_perc_one = cz_perc_one; -------------
      end
      else
      begin
        cz_non_delinq_os_one = cz_non_delinq_os_one + zad_os;
        cz_non_delinq_calc_perc_one = cz_non_delinq_calc_perc_one + sh_calc_perc;
        cz_non_delinq_all_one = cz_non_delinq_all_one + zad_os + sh_calc_perc;
      end
    end

    cz_os = cz_os + cz_os_one;
    cz_fine = cz_fine + cz_fine_one;
    cz_perc = cz_perc + cz_perc_one;    ---------------------------------
    cz_delinq_os = cz_delinq_os + cz_delinq_os_one;
    cz_delinq_all = cz_delinq_all + cz_delinq_all_one;
    cz_delinq_perc = cz_delinq_perc + cz_delinq_perc_one;----------------
    cz_non_delinq_os = cz_non_delinq_os + cz_non_delinq_os_one;
    cz_non_delinq_calc_perc = cz_non_delinq_calc_perc + cz_non_delinq_calc_perc_one;
    cz_non_delinq_all = cz_non_delinq_all + cz_non_delinq_all_one;
    cz_os_total = cz_delinq_os + cz_non_delinq_os;
    cz_perc_total = cz_delinq_perc + cz_non_delinq_calc_perc;
    cz_total_res = cz_delinq_all + cz_non_delinq_all;

    if (sh_min_date is not null) then
      del_days = (rep_date - sh_min_date);
    else
      del_days = 0;
    if (del_days > cz_del_days) then
      cz_del_days = del_days;

    select k_amount_approved
    from kontract
    where k_id = :k_id
    into :kont_amount;

    if (closed_own_debt = 0) then
    begin
      for select kc_amount_approved
          from knclient
          where kc_from_k_id = :k_id and
                kc_from_c_id = :c_id and
                kc_amount_approved is not null and
                kc_amount_approved <> 0 rows 1
          into :cz_amount
      do
      begin
        tmp = cast(cz_os_one as decimal(15,2)) * cz_amount; -- Было cz_cl_os= cz_os as * CZ_AMOUNT;
        cz_cl_os = cz_cl_os + cast(tmp / kont_amount as decimal(15,2));

        tmp = cz_perc_one * cz_amount;
        cz_cl_perc = cz_cl_perc + cast(tmp / kont_amount as decimal(15,2));

        tmp = cz_fine_one * cz_amount;
        cz_cl_fine = cz_cl_fine + cast(tmp / kont_amount as decimal(15,2));

        tmp = cast(cz_delinq_os_one as decimal(15,2)) * cz_amount; -- Было cz_cl_os= cz_os as * CZ_AMOUNT;
        cz_cl_delinq_os = cz_cl_delinq_os + cast(tmp / kont_amount as decimal(15,2));

        tmp = cast(cz_non_delinq_os_one as decimal(15,2)) * cz_amount; -- Было cz_cl_os= cz_os as * CZ_AMOUNT;
        cz_cl_non_delinq_os = cz_cl_non_delinq_os + cast(tmp / kont_amount as decimal(15,2));

        tmp = cast(cz_non_delinq_calc_perc_one as decimal(15,2)) * cz_amount;
        cz_cl_non_delinq_calc_perc = cz_cl_non_delinq_calc_perc + cast(tmp / kont_amount as decimal(15,2));

        cz_cl_non_deliq_os_calc_perc = cz_cl_non_delinq_os + cz_cl_non_delinq_calc_perc;
      end
    end
  end

  if (has_active = 0) then -- если у клиента нет активного кредита, вытаскиваю ближайший закрытый кредит
  begin
    select k_kontract_no, k_disb_date, k_from_kf_id, k_id, k_is_devided, k_base_k_id
    from kontstate, kontract, product, knclient
    where ks_ol_is_delete = 0 and
          ks_state = 2 and
          :rep_date between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          (:db_id <> 41701 or
          (:bln_id = 1186 and
          pr_code containing '.gr.') or
          (:bln_id = 1353 and
          pr_code containing '.ind.' and
          pr_code not containing '.ipf.') or
          (:bln_id = 1403 and
          pr_code containing '.ipf.')) and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = :c_id
    order by ks_date_begin desc
    rows 1
    into ksr_kontract_no, ksr_disb_date, kf_id, k_id, k_is_devided, k_base_k_id;

    select kal.kal_new_kf_name
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          :rep_date between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :ksr_kf_name;

    ksr_disb_date_str = iif(extract(day from ksr_disb_date) < 10, '0' || extract(day from ksr_disb_date), extract(day from ksr_disb_date)) || '.' || iif(extract(month from ksr_disb_date) < 10, '0' || extract(month from ksr_disb_date), extract(month from ksr_disb_date)) || '.' || extract(year from ksr_disb_date);
    if (ksr_kontract_no <> '') then
    begin
      ksr_kontractno_and_disb = ksr_kontract_no || ' от ' || :ksr_disb_date_str;
    end

    ksr_disb_date_list = ksr_disb_date_list || f_datetostr(:ksr_disb_date) || ' г.;';
  end

  cz_cl_all = cz_cl_all + cz_cl_os + cz_cl_perc + cz_cl_fine;
  cz_cl_os_perc = cz_cl_os_perc + cz_cl_os + cz_cl_perc;
  cz_cl_balance = cz_cl_balance + cz_cl_delinq_os + cz_cl_non_delinq_os;

  oko_sh_os = null;
  oko_sh_perc = null;
  select sum(oko_os + oko_sh_os) oko_sh_os, sum(oko_perc + oko_sh_perc) oko_sh_perc, count(distinct oko_from_k_id),
         max(oko_from_k_id)
  from operkroff, operlist
  where oko_from_ol_id = ol_id and
        operlist.is_delete = 0 and
        oko_from_c_id = :c_id and
        ol_date <= :rep_date
  into oko_sh_os, oko_sh_perc, count_kont_cli, k_id_cli;
  if (oko_sh_os is null) then
    oko_sh_os = 0;
  if (oko_sh_perc is null) then
    oko_sh_perc = 0;
  if (count_kont_cli is null) then
    count_kont_cli = 0;
  if (k_id_cli is null) then
    k_id_cli = 0;

  cz_cl_delinq_os = cz_cl_delinq_os + oko_sh_os;
  cz_cl_perc = cz_cl_perc + oko_sh_perc;

  if (count_kont_cli = 1) then
  begin
    oko_sh_os = null;
    oko_sh_perc = null;
    select sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc)
    from operkroff, operlist
    where oko_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          oko_from_k_id = :k_id_cli and
          ol_date <= :rep_date
    into oko_sh_os, oko_sh_perc;
    if (oko_sh_os is null) then
      oko_sh_os = 0;
    if (oko_sh_perc is null) then
      oko_sh_perc = 0;

    cz_delinq_os = cz_delinq_os + oko_sh_os;
    cz_delinq_perc = cz_delinq_perc + oko_sh_perc;
  end
  else
  if (count_kont_cli > 1) then
  begin
    oko_sh_os = null;
    oko_sh_perc = null;
    with tabl1
    as (select distinct oko_from_k_id as k_id
        from operkroff, operlist
        where oko_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              oko_from_c_id = :c_id and
              ol_date <= :rep_date)
    select sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc)
    from tabl1, operkroff, operlist
    where k_id = oko_from_k_id and
          oko_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          ol_date <= :rep_date
    into oko_sh_os, oko_sh_perc;
    if (oko_sh_os is null) then
      oko_sh_os = 0;
    if (oko_sh_perc is null) then
      oko_sh_perc = 0;
    cz_delinq_os = cz_delinq_os + oko_sh_os;
    cz_delinq_perc = cz_delinq_perc + oko_sh_perc;
  end

  if (cz_cl_non_delinq_calc_perc is null) then
    cz_cl_non_delinq_calc_perc = 0;
  if (cz_cl_non_delinq_os is null) then
    cz_cl_non_delinq_os = 0;
  if (cz_cl_delinq_os is null) then
    cz_cl_delinq_os = 0;
  if (cz_cl_perc is null) then
    cz_cl_perc = 0;
  if (cz_cl_fine is null) then
    cz_cl_fine = 0;

  cz_cl_non_delinq_total = cz_cl_non_delinq_os + cz_cl_non_delinq_calc_perc;
  cz_cl_delinq_total = cz_cl_delinq_os + cz_cl_perc + cz_cl_fine;

  cz_cl_os_total = cz_cl_non_delinq_os + cz_cl_delinq_os;
  cz_cl_perc_total = cz_cl_non_delinq_calc_perc + cz_cl_perc;

  cz_cl_total_res = cz_cl_non_delinq_total + cz_cl_delinq_total;

  --Добавлено в 201115 Музаффар
  if (cz_amount is null) then
    cz_amount = 0;
  if (cz_os_total is null) then
    cz_os_total = 0;
  if (cz_perc_total is null) then
    cz_perc_total = 0;
  if (cz_total_res is null) then
    cz_total_res = 0;
  if (cz_fine is null) then
    cz_fine = 0; --конец 201115

  /*Поле «Максимальное число дней просрочки» – не зависит от статуса кредита
  (активный/закрытый/списанный), и отображает максимальное число просроченных дней по данному кредиту*/
  cur_delinq_days = 0;
  cur_delinq_days_temp = 0;

  select sum(iif(oks_date_end > :rep_date - 1, :rep_date - oks_delinq_date, 0)) as cur_delinq_days,
         max(iif(oks_date_end > :rep_date - 1, :rep_date, oks_date_end + 1) - oks_delinq_date) as max_delinq_days
  from oksummary oks
  where oks_from_k_id = iif(:k_id is null, :k_id_cli, :k_id) and
        oks_delinq_date < iif(oks_date_end > :rep_date, :rep_date, oks_date_end + 1) and
        oks_close_state = 0
  into cur_delinq_days, max_delinq_days;

  /*Поле «Текущая просрочка», в случае, если это активный заем будет
  отображаться число дней текущей просрочки, если это списанный заем, то число дней просрочки на день списания.*/
  select ok_ol_date - min(ok_sh_date)
  from operkred ok, kontstate ks
  where ok_ol_is_active = 1 and
        ok_from_c_id = 0 and
        ok_type = 5 and
        ok_from_k_id = ks_from_k_id and
        ok_ol_date = ks_date_begin and
        ks_ol_is_delete = 0 and
        ks_state = 2 and
        ks_from_k_id = iif(:k_id is null, :k_id_cli, :k_id)
  group by ok_ol_date
  into cur_delinq_days_temp;

  if (cur_delinq_days_temp < 0 or (cur_delinq_days_temp is null)) then
    cur_delinq_days_temp = 0;
  if (cur_delinq_days < 0 or (cur_delinq_days is null)) then
    cur_delinq_days = 0;
  if (max_delinq_days < 0 or (max_delinq_days is null)) then
    max_delinq_days = 0;

  if (cur_delinq_days < cur_delinq_days_temp) then
    cur_delinq_days = cur_delinq_days_temp;

  /* если клиент погасил свою долю, значения по доле клиента должны быть 0 */
  if (exists(select *
             from operlist ol
             where ol.is_delete = 0 and
                   ol_id = :kc_closed_ol_id and
                   ol.ol_date <= :rep_date and
                   ol.ol_document_client = :c_id)) then
  begin
    closed_own_debt = 1;
  end

  /*Если было списание доли клиента то значения по долям клиента должны быть 0*/
  if (exists(select *
             from operlist ol, refdata rd
             where ol.is_delete = 0 and
                   ol_id = :kc_closed_ol_id and
                   ol.ol_date <= :rep_date and
                   ol.ol_document_client = :c_id and
                   ol.ol_from_rd_id = rd.rd_id and
                   rd.rd_main_code = 'Reason_WriteOff' and
                   rd.rd_code in ('Reason_WriteOff.IsClientDead', 'Reason_WriteOff.935', 'Reason_WriteOff.936', 'Reason_WriteOff.937', 'Reason_WriteOff.938', 'Reason_WriteOff.939', 'Reason_WriteOff.940', 'Reason_WriteOff.941', 'Reason_WriteOff.942', 'Reason_WriteOff.943',
                                  'Reason_WriteOff.944', 'Reason_WriteOff.945'))) then
  begin
    closed_own_debt = 1;
  end

  if (closed_own_debt = 1) then
  begin
    cz_cl_os = 0;
    cz_cl_perc = 0;
    cz_cl_fine = 0;
    cz_cl_all = 0;
    cz_cl_os_perc = 0;
    cz_cl_delinq_os = 0;
    cz_cl_non_delinq_os = 0;
    cz_cl_balance = 0;
    cz_cl_non_delinq_calc_perc = 0;
    cz_cl_non_deliq_os_calc_perc = 0;
    cz_cl_non_delinq_total = 0;
    cz_cl_delinq_total = 0;
    cz_cl_os_total = 0;
    cz_cl_perc_total = 0;
    cz_cl_total_res = 0;
  end

  -- added on 28-01-2019 by Aleksei -- >>
  has_writeoff_debt = 0;

  if (db_id = 81001) then
  begin
    select ks_state
    from kontstate ks
    where ks.ks_from_k_id = :k_id and
          ks.ks_ol_is_delete = 0 and
          :rep_date between ks.ks_date_begin and ks.ks_date_end and
          ks.ks_state = 2 --определяем закрыт ли контракт на отчетную дату
    into :ks_state;

    if (:ks_state = 2) then
    begin
      has_sum_debt = 0;

      select sum(coalesce(oko_os, 0) + coalesce(oko_sh_os, 0) + coalesce(oko_perc, 0) + coalesce(oko_sh_perc, 0))
      from operkroff okf
      where oko_from_k_id = :k_id and
            okf.oko_ol_is_active = 1
      into :has_sum_debt;

      has_writeoff_debt = iif(has_sum_debt > 0, 1, 0); --если сумма = 1 - то контракт на спецучете, 0 - закрыт

      if (has_writeoff_debt = 0) then
      begin --определяем группа на спецучете или активна,added on 29-01-2019 by Aleksei
        group_spec = 0;
        select sum(coalesce(oko_os, 0) + coalesce(oko_sh_os, 0) + coalesce(oko_perc, 0) + coalesce(oko_sh_perc, 0))
        from operkroff okf
        where oko_from_k_id = :k_base_k_id and
              okf.oko_ol_is_active = 1
        into :has_sum_debt;

        group_spec = iif(coalesce(has_sum_debt, 0) > 0, 1, 0); --если сумма = 1 - то Группа на спецучете, 0 - активна

        has_writeoff_debt = has_writeoff_debt + group_spec; -- проверка контракт или группа на спецучете или активна
        has_writeoff_debt = iif(has_writeoff_debt >= 1, 1, 0); ---1 - спецучет, 0 - активна

      end

    end

    if ((k_is_devided = 1) and
        (has_writeoff_debt = 1)) then --амнистированный на спецучете или группа на спецучете
    begin
      cz_non_delinq_os = 0;
      cz_non_delinq_calc_perc = 0;
      cz_non_delinq_all = 0; --1,2,3
      cz_delinq_os = 0;
      cz_delinq_perc = 0;
      cz_fine = 0; --4,5,6
      cz_delinq_all = 0; --7
      cz_os_total = 0;
      cz_perc_total = 0;
      cz_total_res = 0; --8,9,10
      k_id = null;

      for select distinct ks_from_k_id
          from kontdevides kd, kontstate ks
          where kd_k_orig = :k_base_k_id and
                ks_from_k_id in (kd_k_orig, kd_k_new) and
                ks_state = 2 and
                ks_ol_is_delete = 0
          into k_id
      do
      begin
        sh_min_date = null;
        zad_os = 0;
        sh_calc_perc = 0;
        sh_calc_fine = 0;

        for select sh_date, sh_calc_fine, sh_calc_perc, zad_os
            from aarep_make_zad_list(:k_id, :rep_date, 1)
            into :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os
        do
        begin
          :sh_date = coalesce(:sh_date, '01.01.01');

          if ((sh_min_date is null) and
              (sh_date < :rep_date)) then
          begin
            if ((zad_os <> 0) or (sh_calc_perc <> 0) or (sh_calc_fine <> 0)) then
              sh_min_date = :sh_date;
          end
          if (sh_date <= :rep_date) then
          begin
            cz_delinq_os = cz_delinq_os + zad_os; --4
            cz_delinq_perc = cz_delinq_perc + sh_calc_perc; --5
            cz_fine = cz_fine + sh_calc_fine; --6
            cz_delinq_all = cz_delinq_all + zad_os + sh_calc_perc + sh_calc_fine; --7
          end
          else
          begin
            cz_non_delinq_os = cz_non_delinq_os + zad_os; --1
            cz_non_delinq_calc_perc = cz_non_delinq_calc_perc + sh_calc_perc; --2
            cz_non_delinq_all = cz_non_delinq_all + zad_os + sh_calc_perc; --3
          end
        end
        cz_os_total = cz_non_delinq_os + cz_delinq_os; --8
        cz_perc_total = cz_delinq_perc + cz_non_delinq_calc_perc; --9
        cz_total_res = cz_delinq_all + cz_non_delinq_all; --10
      end
    end
  end
  --added on 28-01-2019 by Aleksei -- <<

  if ((k_is_devided = 1) and
      (has_writeoff_debt = 0)) then -- --контракт амнистированный активный или группа активна, added on 28-01-2019 by Aleksei
  begin
    cz_non_delinq_os = 0;
    cz_non_delinq_calc_perc = 0;
    cz_non_delinq_all = 0; --1,2,3
    cz_delinq_os = 0;
    cz_delinq_perc = 0;
    cz_fine = 0; --4,5,6
    cz_delinq_all = 0; --7
    cz_os_total = 0;
    cz_perc_total = 0;
    cz_total_res = 0; --8,9,10
    k_id = null;

    for select distinct ks_from_k_id
        from kontdevides kd, kontstate ks
        where kd_k_orig = :k_base_k_id and
              ks_from_k_id in (kd_k_orig, kd_k_new) and
              ks_state = 1 and
              ks_ol_is_delete = 0 and
              :sp_date between ks_date_begin and ks_date_end
        into k_id
    do
    begin
      sh_min_date = null;
      zad_os = 0;
      sh_calc_perc = 0;
      sh_calc_fine = 0;

      for select sh_date, sh_calc_fine, sh_calc_perc, zad_os
          from aarep_make_zad_list(:k_id, :rep_date, 1)
          where sh_date is not null
          into :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os
      do
      begin
        if ((sh_min_date is null) and
            (sh_date < :rep_date)) then
        begin
          if ((zad_os <> 0) or (sh_calc_perc <> 0) or (sh_calc_fine <> 0)) then
            sh_min_date = :sh_date;
        end
        if (sh_date <= :rep_date) then
        begin
          cz_delinq_os = cz_delinq_os + zad_os; --4
          cz_delinq_perc = cz_delinq_perc + sh_calc_perc; --5
          cz_fine = cz_fine + sh_calc_fine; --6
          cz_delinq_all = cz_delinq_all + zad_os + sh_calc_perc + sh_calc_fine; --7
        end
        else
        begin
          cz_non_delinq_os = cz_non_delinq_os + zad_os; --1
          cz_non_delinq_calc_perc = cz_non_delinq_calc_perc + sh_calc_perc; --2
          cz_non_delinq_all = cz_non_delinq_all + zad_os + sh_calc_perc; --3
        end
      end
      cz_os_total = cz_non_delinq_os + cz_delinq_os; --8
      cz_perc_total = cz_delinq_perc + cz_non_delinq_calc_perc; --9
      cz_total_res = cz_delinq_all + cz_non_delinq_all; --10
    end
  end

  if (db_id = 81001) then
  begin
    cz_cl_os_total = coalesce(cz_cl_non_delinq_os, 0) + coalesce(cz_cl_delinq_os, 0);
    cz_cl_perc_total = coalesce(cz_cl_non_delinq_calc_perc, 0) + coalesce(cz_cl_perc, 0);
    cz_cl_total_res = coalesce(cz_cl_non_delinq_total, 0) + coalesce(cz_cl_delinq_total, 0);

    cz_delinq_all = coalesce(cz_delinq_os, 0) + coalesce(cz_delinq_perc, 0) + coalesce(cz_fine, 0);
    cz_os_total = coalesce(cz_non_delinq_os, 0) + coalesce(cz_delinq_os, 0);
    cz_perc_total = coalesce(cz_delinq_perc, 0) + coalesce(cz_non_delinq_calc_perc, 0);
    cz_total_res = coalesce(cz_os_total, 0) + coalesce(cz_perc_total, 0) + coalesce(cz_fine, 0);
  end

  ksr_sud_poshlina = 0;

  select sum(coalesce(ox.ox_amount_plus, 0) - coalesce(ox.ox_amount_minus, 0))
  from operextra ox, operlist ol, knclient
  where ox_from_rd_id = :oek_rd_id and
        kc_from_c_id = :c_id and
        ox.ox_object_kind = 9 and
        ox.ox_object_id = kc_from_k_id and
        ox.ox_from_ol_id = ol_id and
        ol_date <= :rep_date and
        ol.is_delete = 0
  into :ksr_sud_poshlina_new;

  ksr_sud_poshlina = ksr_sud_poshlina + coalesce(ksr_sud_poshlina_new, 0);

  if (coalesce(ksr_sud_poshlina, 0) > 0) then
    ksr_sud_poshlina_info = 'Сумма задолженности по возмещению государственной пошлины на основании судебного акта '
                            || cast(ksr_sud_poshlina as varchar(10))
                            || iif(db_id = 39801, ' тенге. ', ' рублей. ');

  brt_is_finished = 0;
  if (db_id = 81001 and
      exists(select *
             from bankrupt
             where brt_from_c_id = :c_id and
                   brt_is_finished = 1 and
                   :rep_date between brt_date_begin and brt_date_end)) then
    brt_is_finished = 1;

  if (ksr_disb_date_list <> '') then
    ksr_disb_date_list = left(ksr_disb_date_list, character_length(ksr_disb_date_list) - 1);

  ksr_kont_phone = f_cont_info('clients', :sign_c_id, 'cell_main', 'ok');

  suspend;
end
