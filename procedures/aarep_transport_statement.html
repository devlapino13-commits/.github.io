CREATE PROCEDURE AAREP_TRANSPORT_STATEMENT(
    REP_TRANS_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PEO_DATE DATE,
    PEO_BILL_ID DECIMAL(18,0),
    PEO_BILL_LINE_NO INTEGER,
    PEO_FROM_C_ID DECIMAL(18,0),
    PEO_COST DECIMAL(18,4),
    PEO_FROM_PEN_ID INTEGER,
    PEO_OPER_DATE DATE,
    PEO_MATOTVET_FIO VARCHAR(100),
    PEO_PETROL_NAME VARCHAR(100),
    PEO_ACC_BALANCE VARCHAR(20),
    PEO_ACC_NUMBER INTEGER,
    PEO_LITER_IN_ONE DECIMAL(18,4),
    PEO_TRANS_ID INTEGER,
    PEO_BRANCH_NAME VARCHAR(100),
    PEO_LINE INTEGER,
    PEO_LINE_NAME VARCHAR(50),
    PEO_LITER_PLUS DECIMAL(18,4),
    PEO_SUMMA_PLUS DECIMAL(18,4),
    PEO_LITER_MINUS DECIMAL(18,4),
    PEO_SUMMA_MINUS DECIMAL(18,4),
    PEO_LITER_SALDO DECIMAL(18,4),
    PEO_SUMMA_SALDO DECIMAL(18,4),
    PEO_OFFICE_NAME VARCHAR(100),
    PEO_TRANS_NAME VARCHAR(255),
    PEO_TRANS_LABEL VARCHAR(50),
    PEO_OPER_DETAIL VARCHAR(1024),
    PEO_SALDO_LITER_TOTAL DECIMAL(18,4),
    PEO_SALDO_SUMMA_TOTAL DECIMAL(18,4),
    PEO_PROBEN_KM DECIMAL(18,4),
    PEO_PROBEG_SUMMA DECIMAL(18,4),
    PEO_PROBEG_LITER DECIMAL(18,4),
    PEO_TRANS_YEDIZ VARCHAR(50),
    PEO_EXPENSE_SUMMER DECIMAL(18,4),
    PEO_EXPENSE_WINTER DECIMAL(18,4),
    PEO_EXPENSE_SUMMER_MOUNT DECIMAL(18,4),
    PEO_EXPENSE_WINTER_MOUNT DECIMAL(18,4),
    PEO_YEDIZ VARCHAR(20))
AS
declare variable peo_oper_type integer;
declare variable peo_liter decimal(18,4);
declare variable peo_summa decimal(18,4);
declare variable peo_fact_km decimal(18,4);
declare variable peo_fact_liter decimal(18,4);
declare variable peo_fact_summa decimal(18,4);
declare variable norma_yediz integer;
declare variable norma_km integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  peo_saldo_liter_total = 0;
  peo_saldo_summa_total = 0;
  peo_proben_km = 0;
  peo_probeg_summa = 0;
  peo_probeg_liter = 0;

  peo_line = 1;/* Остаток на начало */
  peo_line_name = 'Сальдо на начало';
  peo_date = :rep_date_begin;
  peo_liter_plus = 0;
  peo_liter_minus = 0;
  peo_summa_plus = 0;
  peo_summa_minus = 0;
  select pet_name, pet_label, pet_norma_yediz, pet_norma_km, ped_expense_summer, ped_expense_winter,
         ped_expense_summer_mount, ped_expense_winter_mount
  from pettrans, petdata
  where pet_id = :rep_trans_id and
        ped_from_pet_id = pet_id and
        ped_date <= :rep_date_end
  order by ped_date desc
  rows 1
  into :peo_trans_name, :peo_trans_label, :norma_yediz, :norma_km, peo_expense_summer, peo_expense_winter,
       peo_expense_summer_mount, peo_expense_winter_mount;
  if (norma_yediz = 0) then
    peo_yediz = 'км';
  if (norma_yediz = 1) then
    peo_yediz = 'час';
  if (norma_yediz = 2) then
    peo_yediz = 'миль';
  peo_trans_yediz = :norma_km || ' ' || :peo_yediz;

  peo_oper_detail = 'Остаток на начало';

  for select peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, sum(peo_trans_liter),
             sum(peo_trans_summa), peo_oper_date, c_name_fio, pen_name, pen_acc_balance, pen_acc_number,
             pen_liter_in_one, r2.reg_code || ' ' || r2.reg_name, r1.reg_code || ' ' || r1.reg_name
      from petopers, operlist, clients, petname, region r1, region r2
      where peo_from_pet_id = :rep_trans_id and
            peo_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ol_date < :rep_date_begin and
            peo_from_c_id = c_id and
            peo_from_pen_id = pen_id and
            ol_from_reg_id = r2.reg_unicode and
            peo_from_reg_id = r1.reg_unicode
      group by peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, peo_oper_date, c_name_fio, pen_name, pen_acc_balance, pen_acc_number, pen_liter_in_one, r1.reg_code, r1.reg_name, r2.reg_code, r2.reg_name
      having sum(peo_trans_liter) <> 0 or sum(peo_trans_summa) <> 0
      order by peo_oper_date, peo_bill_id, peo_bill_line_no
      into :peo_bill_id, :peo_bill_line_no, :peo_from_c_id, :peo_cost, :peo_from_pen_id, :peo_liter, :peo_summa,
           :peo_oper_date, :peo_matotvet_fio, :peo_petrol_name, :peo_acc_balance, :peo_acc_number, :peo_liter_in_one,
           :peo_branch_name, :peo_office_name
  do
  begin
    peo_trans_id = :rep_trans_id;
    peo_liter_saldo = :peo_liter;
    peo_summa_saldo = :peo_summa;
    suspend;
  end

  peo_liter_saldo = 0;
  peo_summa_saldo = 0;
  peo_line = 2;/* Обороты */
  for select ol_date, peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, sum(peo_trans_liter),
             sum(peo_trans_summa), peo_oper_date, c_name_fio, pen_name, pen_acc_balance, pen_acc_number,
             pen_liter_in_one, r2.reg_code || ' ' || r2.reg_name, r1.reg_code || ' ' || r1.reg_name, peo_oper_type,
             ol_memo, sum(peo_probeg_fact_km), sum(peo_probeg_fact_liter), sum(peo_trans_summa)
      from petopers, operlist, clients, petname, region r1, region r2
      where peo_from_pet_id = :rep_trans_id and
            peo_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ol_date_oper_day between :rep_date_begin and :rep_date_end and
            peo_from_c_id = c_id and
            peo_from_pen_id = pen_id and
            ol_from_reg_id = r2.reg_unicode and
            peo_from_reg_id = r1.reg_unicode
      group by ol_date, peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, peo_oper_date, c_name_fio, pen_name, pen_acc_balance, pen_acc_number, pen_liter_in_one, r1.reg_code, r1.reg_name, r2.reg_code, r2.reg_name, peo_oper_type, ol_memo
      having sum(peo_trans_liter) <> 0 or sum(peo_trans_summa) <> 0
      order by ol_date, peo_oper_date, peo_bill_id, peo_bill_line_no
      into :peo_date, :peo_bill_id, :peo_bill_line_no, :peo_from_c_id, :peo_cost, :peo_from_pen_id, :peo_liter,
           :peo_summa, :peo_oper_date, :peo_matotvet_fio, :peo_petrol_name, :peo_acc_balance, :peo_acc_number,
           :peo_liter_in_one, :peo_branch_name, :peo_office_name, :peo_oper_type, :peo_oper_detail, :peo_fact_km,
           :peo_fact_liter, :peo_fact_summa
  do
  begin
    peo_trans_id = :rep_trans_id;
    peo_liter_plus = 0;
    peo_liter_minus = 0;
    peo_summa_plus = 0;
    peo_summa_minus = 0;
    if (peo_liter >= 0) then
      peo_liter_plus = :peo_liter;
    else
      peo_liter_minus = -1 * :peo_liter;
    if (peo_summa >= 0) then
      peo_summa_plus = :peo_summa;
    else
      peo_summa_minus = -1 * :peo_summa;

    if (peo_oper_type = 0) then
      peo_line_name = 'Первоначальные остатки';
    if (peo_oper_type = 1) then
      peo_line_name = 'Приход на склад';
    if (peo_oper_type = 2) then
      peo_line_name = 'Залито через Чек';
    if (peo_oper_type = 3) then
      peo_line_name = 'Передан в другой офис';
    if (peo_oper_type = 4) then
      peo_line_name = 'Принят из другого офиса';
    if (peo_oper_type = 5) then
      peo_line_name = 'Залито в бак со склада';
    if (peo_oper_type = 6) then
      peo_line_name = 'Возврашен из бака';
    if (peo_oper_type = 7) then
      peo_line_name = 'Списан из бака';
    if (peo_oper_type = 8) then
      peo_line_name = 'Заменен мат.ответ.';
    if (peo_oper_type = 9) then
      peo_line_name = 'Передача между объектами';

    if ((peo_fact_km <> 0) or (peo_fact_liter <> 0)) then
    begin
      peo_proben_km = peo_proben_km + peo_fact_km;
      peo_probeg_summa = peo_probeg_summa - peo_fact_summa;
      peo_probeg_liter = peo_probeg_liter + cast(peo_fact_liter as decimal(15,2));
    end
    suspend;
  end

  peo_oper_detail = 'Остаток на конец';
  peo_date = :rep_date_end;
  peo_line = 3;/* Остаток на конец */
  peo_line_name = 'Сальдо на конец';
  peo_liter_plus = 0;
  peo_liter_minus = 0;
  peo_summa_plus = 0;
  peo_summa_minus = 0;

  for select peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, sum(peo_trans_liter),
             sum(peo_trans_summa), peo_oper_date, c_name_fio, pen_name, pen_acc_balance, pen_acc_number,
             pen_liter_in_one, r2.reg_code || ' ' || r2.reg_name, r1.reg_code || ' ' || r1.reg_name
      from petopers, operlist, clients, petname, region r1, region r2
      where peo_from_pet_id = :rep_trans_id and
            peo_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ol_date <= :rep_date_end and
            peo_from_c_id = c_id and
            peo_from_pen_id = pen_id and
            ol_from_reg_id = r2.reg_unicode and
            peo_from_reg_id = r1.reg_unicode
      group by peo_bill_id, peo_bill_line_no, peo_from_c_id, peo_cost, peo_from_pen_id, peo_oper_date, c_name_fio, pen_name, pen_acc_balance, pen_acc_number, pen_liter_in_one, r1.reg_code, r1.reg_name, r2.reg_code, r2.reg_name
      having sum(peo_trans_liter) <> 0 or sum(peo_trans_summa) <> 0
      order by peo_oper_date, peo_bill_id, peo_bill_line_no
      into :peo_bill_id, :peo_bill_line_no, :peo_from_c_id, :peo_cost, :peo_from_pen_id, :peo_liter, :peo_summa,
           :peo_oper_date, :peo_matotvet_fio, :peo_petrol_name, :peo_acc_balance, :peo_acc_number, :peo_liter_in_one,
           :peo_branch_name, :peo_office_name

  do
  begin
    peo_trans_id = :rep_trans_id;
    peo_liter_saldo = :peo_liter;
    peo_summa_saldo = :peo_summa;
    peo_saldo_liter_total = peo_saldo_liter_total + peo_liter_saldo;
    peo_saldo_summa_total = peo_saldo_summa_total + peo_summa_saldo;

    suspend;
  end
end
