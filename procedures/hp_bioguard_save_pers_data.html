CREATE PROCEDURE HP_BIOGUARD_SAVE_PERS_DATA(
    C_ID D_ID NOT NULL,
    IS_NEW_CLIENT D_BOOL NOT NULL,
    C_NAME_FIO D_STR_100 NOT NULL,
    C_GENDER D_STR_10,
    C_BIRTH_DATE D_DATE,
    C_BIRTH_COUNTRY D_STR_100,
    C_BIRTH_CITY D_STR_100,
    NATIONALITY D_STR_255,
    NATIONALITY_KZ D_STR_255,
    CITIZENSHIP D_STR_255,
    CITIZENSHIP_KZ D_STR_255)
AS
declare variable nation_id d_id;
declare variable citizen_id d_id;
declare variable lang d_str_lang;
declare variable c_alternat_fio_new d_str_100; /* Новое значение поля c_alternat_fio (вычисленное из ФИО, полученного от Биогарда) */
declare variable c_name_eng_new d_str_100; /* Новое значение поля c_name_eng (вычисленное из ФИО, полученного от Биогарда) */
begin
    -- Не обновляем перс. данные для новых клиентов, т.к. при создании они и так устанавливаются
    if (is_new_client = 1) then
        exit;

    lang = 'ru';
    nation_id = (select rd_id from hp_get_rd_id_by_name('Nation', :nationality, :nationality_kz));
    citizen_id = (select rd_id from hp_get_rd_id_by_name('Citizen', :citizenship, :citizenship_kz));
    c_alternat_fio_new = f_translit_fio('cyr', 'lat', :c_name_fio, :lang);
    c_name_eng_new = f_translit_fio('cyr', 'eng', :c_name_fio, :lang);

    update clients c
    set c.c_name_fio = :c_name_fio,
        c.c_alternat_fio  = :c_alternat_fio_new,
        c.c_name_eng = :c_name_eng_new,
        c.c_sex = iif(:c_gender = '1', 1, 0),
        c.c_birth_date = :c_birth_date,
        c.c_birth_country = :c_birth_country,
        c.c_birth_city = :c_birth_city,
        c.c_nation = :nation_id,
        c.c_citizen = :citizen_id
    where c.c_id = :c_id;

end
