CREATE PROCEDURE CONTROLLER_FOR_BLOCK_LIST(
    OFFICER_ID ID_TYPE,
    LIST_TYPE SMALLINT,
    REP_REGION_ID ID_TYPE,
    KF_NAME_SHOW SMALLINT,
    METRO_LINE STR_255,
    METRO_STATION STR_255)
RETURNS (
    COUNTER INTEGER,
    KF_COUNTER INTEGER,
    K_ID ID_TYPE,
    K_DISB_DATE DATE,
    IS_CONTROL_BLOCK SMALLINT,
    CONTROL_FIO STR_100,
    KF_NAME STR_100,
    CL_CITIZEN_STR STR_100,
    REP_C_ADDRESS_JOB STR_255,
    C_ID ID_TYPE,
    METRO STR_255,
    MA_PLAN_DISB_TIME TIMESTAMP)
AS
declare variable rep_date date;
declare variable un_id id_type;
declare variable check_result str_255;
declare variable kf_id id_type;
declare variable kk integer;
declare variable fname str_100;
declare variable lname str_100;
declare variable patronymic str_100;
declare variable cl_citizen integer;
declare variable kf_id_tmp id_type;
declare variable ma_app_id id_type;
declare variable is_ok smallint;
begin
  counter = 0;
  kf_counter = 0;
  kf_id_tmp = 0;
  kf_name = '';
  rep_date = current_date;

  metro_line = coalesce(metro_line, '');
  metro_station = coalesce(metro_station, '');

  if (metro_line = '') then
    metro_line = 'Все';
  if (metro_station = '') then
    metro_station = 'Все';

  for select k_id, k_disb_date, un_id, check_result, kf_id, kf_counter, c_id, ma_app_id
      from controller_for_block_list_tmp(:officer_id, :list_type, :rep_region_id)
      into :k_id, :k_disb_date, :un_id, :check_result, :kf_id, :kf_counter, :c_id, :ma_app_id
  do
  begin
    metro = null;

    if (metro_line <> 'Все') then
    begin
      is_ok = 0;
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'work_address_metro', :rep_date)
      where ovs_value in (select rd.rd_name
                          from refdata rd
                          where rd.rd_memo = :metro_line and
                                rd.rd_main_code = 'Metropoliten')
      into :metro;

--      Старый вариант, отключено 29.03.2016, Sanjari
--        select CDV_VALUE
--        from HP_GET_CUSTOMDATAVALUE(0, :c_id, 'work_address_metro', :REP_DATE)
--        where CDV_VALUE in (select rd.rd_name
--                            from refdata rd
--                            where rd.rd_memo = :METRO_LINE and rd.rd_main_code = 'Metropoliten')
--        into :METRO;

      metro = coalesce(metro, '');

      if (metro <> '') then
        is_ok = 1;

      if (is_ok = 1) then
      begin
        if (metro_station <> 'Все') then
        begin
          if (metro = metro_station) then
            is_ok = 1;
          else
            is_ok = 0;
        end
      end
    end
    else
    begin
      is_ok = 1;
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'work_address_metro', :rep_date)
      into metro;

--        select CDV_VALUE
--        from HP_GET_CUSTOMDATAVALUE(0, :c_id, 'work_address_metro', :REP_DATE)
--        into :METRO;
    end

    if (is_ok = 1) then
    begin

      is_control_block = 0; -- не заблокирован
      if ((un_id is not null) and
          (check_result = 'ControllerCheckStatus.NotChecked')) then
        is_control_block = 9; -- заблокир
      if (check_result = 'ControllerCheckStatus.Confirmed') then
        is_control_block = 1; -- подтвержд.
      if (check_result = 'ControllerCheckStatus.NotConfirmed') then
        is_control_block = -1; -- не подтвержд.
      if (check_result = 'ControllerCheckStatus.Refused') then
        is_control_block = -2; -- Отказ от ВО.

      control_fio = '';

      rep_c_address_job = null;
      /*
          select adl_address_str
          from  addrlist, addrclient
          where adc_from_c_id = :c_id
          and adc_from_adl_id = adl_id
          and adc_Address_type = 3
          into REP_C_ADDRESS_JOB;
    
          if (REP_C_ADDRESS_JOB is null) then
          begin    */

--              select CDV_VALUE
--              from HP_GET_CUSTOMDATAVALUE(0, :c_id, 'work_address_name', :REP_DATE)
--              into :REP_C_ADDRESS_JOB;
--end

      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :c_id and
            adc_address_type = 3 and
            adc_from_adl_id = adl_id
      into :rep_c_address_job;

      if (coalesce(rep_c_address_job, 'Не указан') = 'Не указан') then
        rep_c_address_job = '';
      if (rep_c_address_job = '') then
      begin
        select ovs_value
        from hp_objectprops_get(0, :c_id, -1, 'work_address_name', :rep_date)
        into rep_c_address_job;
      end

      if (coalesce(rep_c_address_job, '') <> '') then
      begin
        ma_plan_disb_time = null;

        select ma_plan_disb_time
        from miniapp maa
        where maa.ma_id = :ma_app_id and
              maa.ma_date_end = '31.12.9999'
        into ma_plan_disb_time;

        if (un_id is not null) then
        begin
          select cl.c_name_fio
          from username un
          inner join clients cl on cl.c_id = un.un_from_c_id
          where un.un_id = :un_id
          into :control_fio;
        end

        kk = 0;
        fname = '';
        lname = '';
        patronymic = '';

        if (control_fio <> '') then
        begin
          kk = position(' ', :control_fio) - 1;
          if (kk > 0) then
          begin
            lname = substring(:control_fio from 1 for kk);
            control_fio = substring(:control_fio from kk + 2 for 255);
            kk = position(' ', control_fio) - 1;
            if (kk > 0) then
            begin
              fname = substring(:control_fio from 1 for kk);
              control_fio = substring(:control_fio from kk + 2 for 255);
              patronymic = :control_fio;
            end
            else
              fname = :control_fio;
            control_fio = lname || ' ' || left(fname, 1) || '. ' || iif(patronymic <> '', left(patronymic, 1) || '.', '');
          end
        end

        if (kf_name_show = 1) then
        begin
          if (kf_id <> kf_id_tmp) then
          begin
            kf_id_tmp = kf_id;

            select kal.kal_new_kf_name
            from knaktlist kal
            where kal_from_kf_id = :kf_id and
                  :rep_date between kal_date and kal_date_end and
                  kal.is_delete = 0
            into :kf_name;
          end
        end

        cl_citizen_str = '';
        cl_citizen = 0;

        select (case c_citizen
                  when 8 then 'КГ'
                  when 9 then 'УЗ'
                  when 10 then 'КЗ'
                  when 12 then 'ТЖ'
                  when 13 then 'РФ'
                  else '--'
                end) as c_citizen_str, c_citizen
        from clients cl
        where c_id = :c_id
        into cl_citizen_str, cl_citizen;

        rep_c_address_job = iif(rep_c_address_job is not null, rep_c_address_job, 'Адрес не указан');

        if ((cl_citizen_str = '--') and
            (cl_citizen <> 0)) then
        begin
          select ddr.rd_name
          from refdata ddr
          where ddr.rd_id = :cl_citizen
          into :cl_citizen_str;
        end

        counter = counter + 1;

        suspend;
      end
    end
  end
end
