CREATE PROCEDURE HP_DEACTIVE_RKL
AS
declare variable cds_id d_id;
declare variable rd_id d_integer;
begin
  rd_id = f_get_rd_id('citizen.ru');

  for select distinct ctd.cds_id
      from knclient kc
      inner join clientdocs ctd on ctd.cds_from_c_id = kc.kc_from_c_id
      inner join passkind pak on pak.pk_id = ctd.cds_from_pk_id
      inner join passdoctype pdt on pdt.pdt_id = pak.pk_from_pdt_id
      inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
      inner join clients c on c.c_id = kc.kc_from_c_id
      where kc.kc_amount_approved > 0 and
            c.c_citizen <> :rd_id and
            ctd.cds_is_delete = 0 and
            pdt.pdt_code = 'results_RCL_check' and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            ks.ks_date_begin = current_date - 1 and
            ctd.cds_doc_date_begin <= ks_date_begin
      into :cds_id
  do
  begin
    -- Делаем в автономной транзакции, чтобы не накапливать большой объем изменений
    in autonomous transaction do
    begin
      update clientdocs cds
      set cds.cds_is_delete = 1
      where cds.cds_id = :cds_id and
            cds.cds_is_delete = 0;
      when gdscode lock_conflict, gdscode update_conflict, gdscode deadlock do
      begin
        --Игнорируем ошибки блокировки
      end
    end
  end
end
