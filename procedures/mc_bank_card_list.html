CREATE PROCEDURE MC_BANK_CARD_LIST(
    MCL_ID D_ID)
RETURNS (
    TOKEN_ID D_STR_CODE,
    IS_VERIFIED D_INTEGER,
    DATE_EXP D_STR_100,
    CARD_TYPE D_STR_10,
    CARD_NUM_LAST4 D_STR_10,
    DESIGN_ID D_INTEGER,
    NAME_ON_CARD D_STR_255)
AS
declare variable C_ID D_ID;
declare variable DB_ID D_INTEGER;
begin
  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id;

  if (coalesce(:c_id, -999) = -999) then
    exit;

  db_id = f_get_db_id();

  if(db_id = 41701) then
  begin
      card_type = 'elcart';
    
      for select iif(eta.eta_state = 0, 0, 1),
                 right(est.et_cardnumber, 6),
                 cast(right(eta.eta_bank_id, 1) as integer),
                 eta.eta_expiry,
                 eta.eta_token_id,
                 eta.eta_name_on_card
          from extsys_token est
          inner join extsys_token_add eta on eta.eta_from_et_id = est.et_id
          where est.et_from_c_id = :c_id
          and eta.eta_state >= 0
          into :is_verified,
               :card_num_last4,
               :design_id,
               :date_exp,
               :token_id,
               :name_on_card
      do
        suspend;
  end
  else if (db_id = 39801) then
  begin
     for select est.et_token,
                right(est.et_cardnumber, 6)
          from extsys_token est
          where est.et_from_c_id = :c_id
          and est.et_is_actual = 1
          into  :token_id,
                :card_num_last4
      do
        suspend;
  end
end
