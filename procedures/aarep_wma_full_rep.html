CREATE PROCEDURE AAREP_WMA_FULL_REP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE = null)
RETURNS (
    WMA_ID INTEGER,
    WMA_DATE DATE,
    FIO STR_100,
    CITIZENSHIP STR_100,
    LANGUAGE STR_100,
    WMA_CANT_CALL_QTY INTEGER,
    LAST_CANT_CALL_DATETIME TIMESTAMP,
    BLOCKED_USER STR_100,
    PHONE_NO STR_100,
    CALLBACK_DATETIME TIMESTAMP,
    LAST_USER STR_100,
    MEMO D_STR_8100,
    CREATE_DATETIME TIMESTAMP,
    CITY STR_100,
    STATUS STR_255,
    OFFICE_NAME STR_255,
    SEND_TO_OFFICE_DATETIME TIMESTAMP,
    CLOSE_REASON STR_100,
    REFUSE_REASON STR_255,
    EMA_LO_FIO STR_255,
    KFOLDER_SVZ STR_100,
    KFOLDER_NAME STR_100,
    K_KONTRACT_NO STR_100,
    PLAN_DISB_DATE DATE,
    FACT_DISB_DATE DATE,
    EMA_CURRENT_STATUS STR_100,
    TKT_ID ID_TYPE,
    WMA_SOURCE STR_255,
    CITY_FOR_OFFICE STR_100,
    INFO_SOURCE STR_255,
    INFO_SOURCE_TEXT STR_255,
    LOAN_PURPOSE STR_255)
AS
declare variable EMA_STATUS smallint;
declare variable SVZ_REFUSE STR_255;
declare variable EMA_ID ID_TYPE;
declare variable MA_ID integer;
declare variable MA_REFUSE_REASON STR_255;
declare variable MA_DATE_OPER date;
declare variable MA_KRL_ID integer;
declare variable K_ID ID_TYPE;
declare variable K_REFUSE_REASON STR_255;
declare variable K_KRL_ID integer;
declare variable KS_STATE smallint;
declare variable K_DISB_DATE date;
declare variable WMA_REFUSE_REASON STR_255;
declare variable WMA_STATUS integer;
begin
    for select wma_id,
               wma_date,
               iif(wmat_name = '', '[Анкета в процессе заполнения]', wmat_name),
               rd_c.rd_name as citizenship,
               rd_l.rd_name as language,
               wma_cant_call_qty,
               wma_phone_no,
               wma_last_cant_call_datetime,
               coalesce(un.un_name, ''),
               wma_create_datetime,
               wmat_city,
               trim((case wma_status
                     when -1 then iif(rd_close.rd_tag =1, 'КЦ: Закрыта. Позитив. статус',  'КЦ: Закрыта. Негатив. статус')
                     when 0  then 'Новая'
                     when 1  then 'Перезвонить'
                     when 2  then 'Передана в офис'
               end)),
               reg.reg_name,
               wma_send_to_office_datetime,
               wmt.wmat_call_back_date,
               coalesce(un_last.un_name, ''),
               iif(wma_status >= 0, wmt.wmat_anketa_memo, wmc.wmc_memo),
               rd_close.rd_name,
               rd_refuse.rd_name,
               wma.wma_from_tkt_id,
               wma_status,
               rd_source.rd_name,
               pl.pl_name,
               rd_is.rd_name as info_source,
               wmt.wmat_info_source_text,
               rd_lp.rd_name as loan_purpose
    from cc_website_miniapp wma
    inner join cc_website_miniapp_tmp  wmt on wmt.wmat_id = wma.wma_last_wmat_id
    left join cc_website_miniapp_close wmc on wmc.wmc_from_wma_id = wma.wma_id
    left join refdata  rd_c      on rd_c.rd_id = wmt.wmat_citizenship
    left join refdata  rd_l      on rd_l.rd_id = wmat_language
    left join username un        on wma_blocked_un_id = un.un_id
    left join username un_block  on un_block.un_id = wma_blocked_un_id
    left join username un_last   on un_last.un_id = wma_last_un_id
    left join region   reg       on reg.reg_unicode = wmt.wmat_office
    left join refdata  rd_close  on rd_close.rd_id = wmc.wmc_closed_rd_id
    left join refdata  rd_refuse on rd_refuse.rd_id = wmc.wmc_refuse_id
    Left join refdata  rd_source on rd_source.rd_id = wma.wma_row_source
    left join places pl on pl.pl_id = wmt.wmat_city_id
    left join refdata rd_is on rd_is.rd_id = wmt.wmat_info_source
    left join refdata rd_lp on rd_lp.rd_id = wmt.wmat_loan_purpose
    where wma_date between :rep_date_begin and :rep_date_end
    and wma.wma_status between -1 and 2
    order by wma_id
    into :wma_id,
         :wma_date,
         :fio,
         :citizenship,
         :language,
         :wma_cant_call_qty,
         :phone_no,
         :last_cant_call_datetime,
         :blocked_user,
         :create_datetime,
         :city,
         :status,
         :office_name,
         :send_to_office_datetime,
         :callback_datetime,
         :last_user,
         :memo,
         :close_reason,
         :wma_refuse_reason,
         :tkt_id,
         :wma_status,
         :wma_source,
         :city_for_office,
         :info_source,
         :info_source_text,
         :loan_purpose
    do
    begin
        ema_lo_fio = null;
        ema_status = null;
        svz_refuse = null;
        ema_id = null;

        select cl.c_name_fio,
               ema.ema_status,
               rd_refuse.rd_name,
               ema.ema_id
        from extminiapp ema
        left join refdata rd_refuse on rd_refuse.rd_id = ema.ema_result_rd_id
        left join clients cl on cl.c_id = ema.ema_lo_id
        where ema.ema_from_tkt_id = :tkt_id
        into :ema_lo_fio,
             :ema_status,
             :svz_refuse,
             :ema_id;

        ma_id = null;
        kfolder_name= null;
        kfolder_svz = null;
        ma_refuse_reason = null;
        ma_date_oper= null;
        ma_krl_id = null;

        for select maa.ma_id
        from miniapp maa
        where maa.ma_from_tkt_id = :tkt_id
        rows 1
        into :ma_id
        do
        begin
            select map.ma_id,
                   kal.kal_new_kf_name,
                   cl.c_name_fio,
                   krs.kr_text,
                   map.ma_date_oper,
                   krl.krl_id
            from miniapp map
            inner join knaktlist  kal on kal.kal_from_kf_id = map.ma_from_kf_id
            inner join clients    cl  on cl.c_id = kal.kal_new_kf_officer
            left join kontreslist krl on krl.krl_ma_id = map.ma_id
            left join kontres     krs on krs.kr_id = krl.krl_from_kr_id
            where map.ma_id = :ma_id
            and map.ma_date_end = '31.12.9999'
            and current_date between kal.kal_date and kal.kal_date_end
            and kal.is_delete = 0
            rows 1
            into
                :ma_id,
                :kfolder_name,
                :kfolder_svz,
                :ma_refuse_reason,
                :ma_date_oper,
                :ma_krl_id;
        end

        k_id = null;
        k_kontract_no = null;
        k_refuse_reason = null;
        k_krl_id = null;
        ks_state = null;
        k_disb_date = null;
        fact_disb_date = null;

        select kont.k_id,
               kont.k_kontract_no,
               krs.kr_text,
               krl.krl_id,
               ks.ks_state,
               kont.k_disb_date,
               ol.ol_date
        from kontract kont
        left join kontreslist krl on krl.krl_from_k_id = kont.k_id
        left join kontres     krs on krs.kr_id = krl.krl_from_kr_id
        left join kontstate   ks  on ks.ks_from_k_id = kont.k_id
        left join operlist    ol  on ol.ol_id = kont.k_fact_disb_ol_id
            and ol.is_delete = 0
        where kont.k_from_ma_app_id = :ma_id
        and current_date between ks.ks_date_begin and ks.ks_date_end
        and ks.ks_ol_is_delete = 0
        rows 1
        into :k_id,
             :k_kontract_no,
             :k_refuse_reason,
             :k_krl_id,
             :ks_state,
             :k_disb_date,
             :fact_disb_date;

        ema_current_status = '';
        refuse_reason = '';
        plan_disb_date = null;

        if (k_id is not null) then
        begin
            plan_disb_date = :k_disb_date;
            if (ks_state in (1,2)) then
                ema_current_status = 'Заем выдан';
            else if (k_krl_id is not null) then
            begin
                ema_current_status = 'Отказ на уровне кредитного договора';
                refuse_reason = k_refuse_reason;
            end
            else
                ema_current_status = 'Кредитный договор в процессе';
        end
        else if (ma_id is not null) then
        begin
            plan_disb_date = :ma_date_oper;
            if (ma_krl_id is not null) then
            begin
                ema_current_status = 'Отказ на уровне кредитной заявки';
                refuse_reason = ma_refuse_reason;
            end
            else
                ema_current_status = 'Кредитная заявка';
        end
        else if (ema_id is not null) then
        begin
           if (ema_status = 0) then
           begin
                if (ema_lo_fio is null) then
                    ema_current_status = 'Заявка на уровне FsMobile. Не заблокирована ВИПС';
                else
                    ema_current_status = 'Заявка на уровне FsMobile. Заблокирвоана ВИПС';
           end
           else if (ema_status = 1) then
           begin
                ema_current_status = 'Заявка на уровне FsMobile. Обработана ВИПС';
           end
            else if (ema_status = -1) then
            begin
                refuse_reason = svz_refuse;
                ema_current_status = 'Отказ на уровне FsMobile со стороны ВИПС';
            end
        end
        else
        begin
            if (wma_status = 0) then
                ema_current_status = 'Заявка на уровне КЦ. Новая';
            else if (wma_status = 1) then
                ema_current_status = 'Заявка на уровне КЦ. Перезвон';
            else if (wma_status = 2) then
                ema_current_status = 'Заявка с КЦ передана в офис';
            else if (wma_status = -1) then
            begin
                if (refuse_reason is null) then
                    ema_current_status = 'Закрыта, на уровне КЦ';
                else
                begin
                    ema_current_status = 'Отказано, на уровне КЦ';
                    refuse_reason = wma_refuse_reason;
                end
            end
        end
        suspend;
    end
end
