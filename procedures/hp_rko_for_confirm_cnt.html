CREATE PROCEDURE HP_RKO_FOR_CONFIRM_CNT(
    UN_ID D_ID)
RETURNS (
    RKO_OWN_CNT D_INTEGER,
    RKO_OTHER_CNT D_INTEGER)
AS
declare variable c_id d_id;
declare variable mo_reg_id d_integer;
begin
  -- возвращает кол-во Расходно-кассовых ордеров для подтверждения контролером (RFCKG-4031)
  -- un_id - контролер, менеджер офиса (МО)
  -- rko_own_cnt - кол-во расходно-кассовых ордеров (РКО), в которых контролер заданный un_id
  -- rko_other_cnt - кол-во РКО, в которых контролер другой, но созданы в офисе МО

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  select ps_from_reg_id
  from hr_pstatus
  inner join joblist on jl_id = ps_from_jl_id
  inner join jobkind on jk_id = jl_kind
  where ps_from_c_id = :c_id and
        ps_status = 1 and
        ps_is_working = 1 and
        current_date between ps_date_begin and ps_date_end and
        jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') -- менеджер офиса (МО)
  into mo_reg_id;
  mo_reg_id = coalesce(mo_reg_id, 0); -- офис МО

  if (exists(select *
             from cashoffice
             where cso_is_online = 1 and
                   cso_is_active = 1 and
                   cso_activate_date < current_date and
                   cso_office_id = :mo_reg_id)) then -- кассы офиса работают в электронном режиме
  begin
    select count(*)
    from opercash
    where opc_is_delete = 0 and
          opc_is_signed = 0 and
          opc_sign = -1 and
          opc_date = current_date and
          opc_office_id = :mo_reg_id and
          opc_manager_id = :c_id
    into rko_own_cnt;

    select count(*)
    from opercash
    where opc_is_delete = 0 and
          opc_is_signed = 0 and
          opc_sign = -1 and
          opc_date = current_date and
          opc_office_id = :mo_reg_id and
          opc_manager_id <> :c_id
    into rko_other_cnt;
  end

  rko_own_cnt = coalesce(rko_own_cnt, 0);
  rko_other_cnt = coalesce(rko_other_cnt, 0);

  suspend;
end
