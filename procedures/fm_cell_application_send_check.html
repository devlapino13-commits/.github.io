CREATE PROCEDURE FM_CELL_APPLICATION_SEND_CHECK(
    EDB_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_LONG,
    RES_NEED_ROLLBACK D_SMALLINT)
AS
declare variable c_id d_id;
declare variable eds_status d_smallint;
begin
  res_text = '';
  res_need_rollback = 1;

  eds_status = 0;
  c_id = 0;

  select eds.eds_from_c_id,
         eds.eds_status
  from edoc_blank edb
  join edoc_list edl on edl.edl_id = edb.edb_from_edl_id
  join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
  where eds.eds_from_c_id > 0
    and edb.edb_id = :edb_id
  into :c_id,
       :eds_status;

  if (eds_status = 0) then
    res_text = 'Документ на изменение номера не подписан клиентом';

  if (res_text = '' and eds_status = 1) then
  begin
    if (not exists(select *
                   from clientdocs cd
                   where cd.cds_from_c_id = :c_id and
                         cd.cds_from_pk_id = 355 and
                         current_date between cd.cds_doc_date_begin and cd.cds_doc_date_end and
                         cd.cds_is_delete = 0 and
                         cd.cds_verify_status in (-3, 1))) then
      res_text = 'Закрепите фото клиента для смены номера';
  end

  if (res_text = '') then
  begin
    update edoc_blank_scan
    set edbs_status = 1,
        edbs_status_un_id = :un_id
    where edbs_from_edb_id = :edb_id and
          edbs_from_bls_id is not null and
          edbs_status in (-1, 0);

    res_need_rollback = 0;
  end

  suspend;
end
