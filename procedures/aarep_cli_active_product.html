CREATE PROCEDURE AAREP_CLI_ACTIVE_PRODUCT(
    REP_OFFICER_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    C_ID D_ID,
    CLI_FIO D_STR_100,
    PR_NAME D_STR_100,
    CLI_BALANCE D_CURRENCY,
    PLAN_CLOSE_DATE D_DATE,
    PR_BALANCE D_CURRENCY)
AS
declare variable OLD_C_ID D_CURRENCY;
begin
  old_c_id = 0;
  for
  select kc_from_c_id, pr_name, sum(oks_balance), max(k_kredline_end)
  from oksummary, knclient, kontract, product
  where oks_kc_id = :rep_officer_id
  and :rep_date between oks_date_begin and oks_date_end
  and oks_balance > 0
  and oks_from_k_id = k_id
  and k_from_pr_id = pr_id
  and oks_from_k_id = kc_from_k_id
  and kc_amount_approved > 0
  group by 1,2
  order by kc_from_c_id, pr_name desc
  into c_id, pr_name, pr_balance, plan_close_date
  do
  begin
    select c_name_fio from clients
    where c_id = :c_id
    into cli_fio;
    if (old_c_id <> c_id) then
    begin
      old_c_id = c_id;
      cli_balance = 0;
    end 
    cli_balance = cli_balance + pr_balance;
    suspend;
  end 
end
