CREATE PROCEDURE HP_EXTINFO_BANKRUPTCY_INSERT(
    EIL_ID D_ID)
AS
declare variable K_ID D_ID;
declare variable C_ID D_ID;
declare variable BLS_ID D_ID;
declare variable BANK_STATE D_STR_255;
declare variable BANK_DATE D_DATE;
declare variable JSON_BLOB D_BLOB_TEXT;
declare variable IDX D_INTEGER;
declare variable PERSON_COUNT D_INTEGER;
begin
    select eil.eil_from_c_id,
           eil.eil_object_id,
           eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eil.eil_id = :eil_id
      and eil.eil_status = 2
      and eic.eic_code = 'bankruptcy'
    rows 1
    into c_id, k_id, bls_id;

    select blx_blob
      from hp_get_blobex_blob(:bls_id)
      into json_blob;

    if (row_count is null or coalesce(json_blob, '') = '') then
        exit;

    person_count = f_json_blob_get_count(:json_blob,'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo');

    idx = 0;
    while (idx < person_count) do
    begin
        bank_state = f_json_blob_get_value(:json_blob,
            'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo[' || idx || '].StatementStatus', null);


        if (bank_state = 'Nachata_protsedura_bankrotstva') then
        begin
            bank_date = cast(substring(f_json_blob_get_value(:json_blob,                      -- дата не известна
                'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo[' || idx || '].LastModificationDate', null) from 1 for 10) as d_date);

            insert into bankruptcy_for_exec (bfe_from_c_id,bfe_from_k_id,bfe_from_eil_id,bfe_bankruptcy_date,bfe_status)
            values (:c_id, :k_id, :eil_id, :bank_date, 0);

            exit;
        end

        idx = idx + 1;
    end
end
