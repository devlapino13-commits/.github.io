CREATE PROCEDURE AAREP_STAFF_RELATION_REPORT_DET(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    R_C_NAME_FIO VARCHAR(100),
    R_REL_NAME VARCHAR(100),
    R_C_BIRTH_DATE DATE,
    R_C_PLASTCARD_NO VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    REG_NAME VARCHAR(50),
    C_BIRTH_DATE DATE)
AS
declare variable C_ID ID_TYPE;
begin
  for select CL.C_NAME_FIO, JL.JL_NAME, REG.REG_NAME, CL.C_BIRTH_DATE, CL.C_ID
      from HR_PSTATUS HR
      inner join region_list(:rep_region_id, 1, 3, 1) on id_list = hr.ps_from_reg_id
      inner join CLIENTS CL on CL.C_ID = HR.PS_FROM_C_ID
      inner join JOBLIST JL on JL.JL_ID = HR.PS_FROM_JL_ID
      inner join REGION REG on REG.REG_UNICODE = HR.PS_FROM_REG_ID
      where HR.PS_IS_WORKING = 1 and
            :REP_DATE_END between HR.PS_DATE_BEGIN and HR.PS_DATE_END
      into :C_NAME_FIO, :JL_NAME, :REG_NAME, :C_BIRTH_DATE, :C_ID
  do
  begin
    for select C_NAME_FIO, REL_NAME, C_BIRTH_DATE, C_PLASTCARD_NO
        from HP_CLIENT_RELATION_LIST(:C_ID, 0)
        into :R_C_NAME_FIO, :R_REL_NAME, :R_C_BIRTH_DATE, :R_C_PLASTCARD_NO
    do
    begin
      suspend;
    end
  end
end
