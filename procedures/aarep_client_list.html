CREATE PROCEDURE AAREP_CLIENT_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    CLIENT_FIO VARCHAR(100),
    KC_BALANCE CURR_TYPE,
    KC_CALC_PERC CURR_TYPE,
    K_DISB_DATE DATE,
    IS_CLIENT INTEGER,
    REG_NAME VARCHAR(100))
AS
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable DD_CLOSE_TMP date;
declare variable KC_AMOUNT_APPROVED CURR_TYPE;
declare variable K_KOLICH_CLIENT_APPROVED integer;
declare variable C_ID_TMP ID_TYPE;
declare variable REG_ID integer;
declare variable K_ID ID_TYPE;
declare variable OKS_BALANCE CURR_TYPE;
declare variable TMP PERC_TYPE;
declare variable REG_ID_PRIOR integer;
declare variable K_CALC_PERC CURR_TYPE;
begin
  reg_id_prior = 0;
  for select oks_from_k_id, oks_balance, oks_region_id
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end
      and oks_close_state = 0 
--  and oks_balance = 0
            and
            oks_region_id = id_list
      into k_id, oks_balance, reg_id
  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end

    select k_disb_date, k_amount_approved, k_kolich_client_approved, k_kontract_no
    from kontract
    where k_id = :k_id
    into k_disb_date, k_amount_approved, k_kolich_client_approved, k_kontract_no;

    select sum(sh_calc_perc)
    from aarep_make_zad_list(:k_id, :rep_date_end, 1)
    where sh_date is not null
    into k_calc_perc;

    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :c_id_tmp, :kc_amount_approved
    do
    begin
      select c_name_fio
      from clients
      where c_id = :c_id_tmp
      into client_fio;

--      kc_balance = cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance;
      tmp = oks_balance / k_amount_approved;
      kc_balance = cast(tmp * kc_amount_approved as decimal(15,2));

      tmp = k_calc_perc / k_amount_approved;
      kc_calc_perc = cast(tmp * kc_amount_approved as decimal(15,2));

      is_client = 1;
      for select kc_date_close_os
          from knclient
          where ((kc_date_disb between '01.01.1900' and (:k_disb_date - 1)) or
                (kc_date_disb = :k_disb_date and
                kc_from_k_id < :k_id)) and
                kc_date_close_os between :k_disb_date and '31.12.9999' and
                kc_from_c_id = :c_id_tmp
          into :dd_close_tmp
      do
      begin
        if (dd_close_tmp >= :rep_date_end) then
        begin
          is_client = 0;
        end
      end

      suspend;
    end
  end
end
