CREATE PROCEDURE HP_CHECK_CLI_OWN_LO(
    MAC_FROM_C_ID ID_TYPE,
    LO_REGION_ID INTEGER,
    LO_ID ID_TYPE)
RETURNS (
    CL_CHECK INTEGER,
    MA_RESULT_ID INTEGER,
    MA_RESULT_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable KS_DATE_END date;
declare variable KS_FROM_K_ID ID_TYPE;
declare variable KAL_NEW_KF_OFFICER ID_TYPE;
declare variable KAL_NEW_REG_ID ID_TYPE;
declare variable KF_OFFICER varchar(100);
declare variable REG_NAME varchar(100);
begin
  if (lo_region_id <> 1061800) then -- ТЦ Исламик
  begin
--    проверяем клиент не закрыт ли 30 дней назад
    ma_result_txt = '';
    select ks_date_end, ks_from_k_id
    from knclient, kontstate
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = :mac_from_c_id and
          ks_region_id <> 1061800 -- ТЦ Исламик
    order by ks_date_end desc
    rows 1
    into ks_date_end, ks_from_k_id;
--    если не закрыт то проверяем следующее
    if (ks_date_end > current_date - 31) then
    begin
--      у клиента вытаскивает текущий КС
      select kal_new_kf_officer, kal_new_reg_id
      from kontract, knaktlist kal, product pr
      where k_id = :ks_from_k_id and
            k_from_kf_id = kal_from_kf_id and
            current_date between kal_date and kal_date_end and
            kal.is_delete = 0 and
            kal.kal_new_reg_id <> 1061800 and
            pr.pr_id = k_from_pr_id and
            pr.pr_code not containing '.installment_goods.' and
            pr.pr_code not containing '.sugarmill.'
      into kal_new_kf_officer, kal_new_reg_id;

      if (lo_id <> coalesce(kal_new_kf_officer, lo_id)) then
      begin
        select c_name_fio
        from clients
        where c_id = :kal_new_kf_officer
        into kf_officer;

        select reg_name
        from region r
        where r.reg_unicode = :kal_new_reg_id
        into reg_name;
        
        if (exists (
            select 1
            from hr_voctravel, hr_orders
            where hrv_from_ho_id = ho_id and
            hrv_from_c_id = :kal_new_kf_officer and
            hrv_changer_id = :lo_id and
            ho_order_type in (7, 8, 9, 10, 13) and
            current_date between hrv_date_begin and hrv_date_end_fact)) then
        begin
          cl_check = 1;
          ma_result_id = 1;
          ma_result_txt = BLOB_APPEND(ma_result_txt, '<br><b><font color="blue">' || 'Клиент является клиентом другого СПФ:' || kf_officer || ' и заявка будет оформлена на: ' || kf_officer || '</font></b><br>');
          suspend;
        end
        else
        begin
          cl_check = 1;
          ma_result_id = -1;
          ma_result_txt = BLOB_APPEND(ma_result_txt, '<br><b><font color="red">' || 'Клиент числится за другим СПФ:' || kf_officer || '. Офис: ' || reg_name || '</font></b><br>');
          suspend;
        end
      end
    end
  end
end
