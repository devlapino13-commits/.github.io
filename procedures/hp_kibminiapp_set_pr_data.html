CREATE PROCEDURE HP_KIBMINIAPP_SET_PR_DATA(
    KBM_ID ID_TYPE,
    KBM_PR_ID STR_255,
    KBM_VER_INFO SMALLINT,
    KBM_ERR_TEXT STR_255)
AS
declare variable kbm_is_checked integer;
begin
  kbm_is_checked = 0;

  if (kbm_err_text in ('Ошибка при поиске клиента, status=SUBJECT_NOT_FOUND', 'ClientNotFound', 'SubjectNotFound')) then
  begin
    kbm_err_text = 'Проверка осуществлена. Клиент в КИБ не найден';
    kbm_is_checked = 1;
  end
  if (kbm_err_text = 'SocCode should be 14 characters') then
    kbm_err_text = 'ИНН должен быть 14 символов';
  if (kbm_err_text = 'Failed cross check validation between SocCode and Date of birth') then
    kbm_err_text = 'Дата рождения не соответствует ИНН';
  if (kbm_err_text = 'Subject identification missing') then
    kbm_err_text = 'ИНН не указан';

  update kibminiapp kbm
  set kbm.kbm_err_text = :kbm_err_text,
      kbm.kbm_id = :kbm_id,
      kbm.kbm_pr_id = :kbm_pr_id,
      kbm.kbm_ver_info = :kbm_ver_info,
      kbm.kbm_is_checked = :kbm_is_checked
  where kbm.kbm_id = :kbm_id;
end
