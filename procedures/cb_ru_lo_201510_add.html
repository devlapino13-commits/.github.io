CREATE PROCEDURE CB_RU_LO_201510_ADD(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    LO_FIO VARCHAR(100),
    REG_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    RES_STAGE_BEGIN DATE,
    RES_DISB_COUNT INTEGER,
    RES_BONUS_AMOUNT DECIMAL(18,4),
    RES_BONUS_CALC DECIMAL(18,4),
    RES_MEMO VARCHAR(255))
AS
declare variable lo_id decimal(18,0);
declare variable needcalc integer;
begin
-- расчет доп.бонуса за выдачу в РФ
  for select ks_kc_id, count(*)
      from kontstate, knclient, region_list(:rep_region_id, 1, 3, 1)
      where ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            ks_kc_id > 1000000000004 and
            ks_region_id = id_list
      group by ks_kc_id
      into :lo_id, :res_disb_count
  do
  begin
    res_memo = '';
    needcalc = 1;
    res_stage_begin = null;
    select min(ps_date_begin)
    from hr_pstatus
    where ps_from_c_id = :lo_id and
          ps_status = 1
    into :res_stage_begin;
    if (coalesce(res_stage_begin, :rep_date_begin) > (dateadd(month, -2, :rep_date_begin))) then
    begin
      needcalc = 0;
      res_memo = 'Не начислять: Стаж менее 2 мес.';
    end

    select c_name_fio
    from clients
    where c_id = :lo_id
    into :lo_fio;

    reg_name = '';
    jl_name = '';
    select reg_code || ' ' || reg_name, jl_name
    from hr_pstatus, joblist, region
    where ps_from_c_id = :lo_id and
          ps_from_reg_id = reg_unicode and
          ps_from_jl_id = jl_id and
          ps_status = 1 and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_is_working = 1
    into :reg_name, :jl_name;
    if (reg_name = '') then
    begin
      needcalc = 0;
      if (res_memo = '') then
        res_memo = 'Не начислять: Нет действующего приказа';
      else
        res_memo = res_memo || ', нет действующего приказа';
    end
    res_bonus_amount = 150.0 * res_disb_count;
    res_bonus_calc = 150.0 * res_disb_count * needcalc;
    suspend;
  end
end
