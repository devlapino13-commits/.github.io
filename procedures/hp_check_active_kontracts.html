CREATE PROCEDURE HP_CHECK_ACTIVE_KONTRACTS(
    C_ID D_ID,
    AMOUNT_APROVED D_CURRENCY)
RETURNS (
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable K_ID D_ID;
declare variable K_PERIOD D_INTEGER;
declare variable K_KOLICH_REPAY D_INTEGER;
declare variable PAID_REPAY D_INTEGER;
declare variable PAYMENTS_LEFT D_INTEGER;
declare variable IS_DELINQ D_SMALLINT;
begin

  res_error_text = null;
  res_error_is_critical = 0;

  /* активные займы клиента */
  for
      select ks.ks_from_k_id,
             k.k_period,
             k.k_kolich_repay
      from kontstate ks
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
      inner join kontract k on k.k_id = ks.ks_from_k_id
      where ks.ks_state = 1
        and ks.ks_ol_is_delete = 0
        and current_date between ks.ks_date_begin and ks.ks_date_end
        and kc.kc_from_c_id = :c_id
        and kc.kc_amount_approved > 0
      order by k.k_period
      into :k_id,
           :k_period,
           :k_kolich_repay
  do
  begin

  if (amount_aproved > 150000) then
  begin
     -- === 1. Срок займа = 3 месяца ===
    if (k_period = 3) then
    begin
      res_error_text =
        'Создание заявки запрещено до полного закрытия клиентом текущего активного займа';
      res_error_is_critical = 2;
      suspend;
      exit;
    end

--     === 2. Остаток обязательных платежей ===
    paid_repay = 0;

    select max(ok_sh_id)
    from operkred
    where ok_from_k_id = :k_id
      and ok_ol_is_active = 1
      and ok_type = 4
      and ok_os + ok_perc < 0
      and ok_ol_date <= current_date
    into :paid_repay;

    if (paid_repay is null) then
      paid_repay = 0;

    payments_left = k_kolich_repay - paid_repay;

    if (payments_left < 0) then
      payments_left = 0;

    if (payments_left > 1) then
    begin
      res_error_text =
        'Создание заявки запрещено до полного закрытия клиентом текущего активного займа';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
 end

    /* === 3. Активная просрочка ≥ 2 дней === */
    is_delinq = null;

    select 1
    from oksummary
    where oks_from_k_id = :k_id
      and current_date between oks_date_begin and oks_date_end
      and current_date - oks_delinq_date >= 2
      and oks_close_state = 0
    into :is_delinq;

    if (is_delinq = 1) then
    begin
      res_error_text =
        'Создание заявки запрещено до полного закрытия клиентом текущего активного займа';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
  end

  suspend;
end
