CREATE PROCEDURE HP_SEND_QUERY_IE_KONTUR(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable c_citizen d_integer;
declare variable c_inn d_str_20;
declare variable ma_date_oper d_date;
begin
    -- RFCRU-5314
    -- Отправка запроса в контур, для того чтобы проверить статус ИП

    if (f_objkind(:obj_kind, 'code') not in ('miniapp')) then
        exception e_default 'object kind must be "miniapp"';

    select ma.ma_date_oper, mac.mac_from_c_id, c.c_citizen, c.c_inn
    from miniapp ma
    inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
        and mac.mac_amount > 0
    inner join clients c on mac.mac_from_c_id = c.c_id
    where ma.ma_id = :obj_id
        and ma.ma_is_active = 1
        and ma.ma_date_end = '31.12.9999'
    into :ma_date_oper, :c_id, :c_citizen, :c_inn;

    if (coalesce(:c_id, 0) = 0) then
    begin
        -- Если не было найдено ни одной записи
        res_text = 'Вы пытаетесь отправить запрос по неактивной заявке';
        suspend;
        exit;

    end

    if (coalesce(c_inn, '') = '') then
    begin
        -- Если есть какие-то проблемы с ИНН
        res_text = 'Не добавлен ИНН';
        suspend;
        exit;

    end

    -- Если гражданин не РФ и нет документа ВНЖ или РВП тогда выводим ошибку
    -- Если cds_verify_status = -3, тогда документ был добавлен через МП, иначе через КМП и документ прошёл через оцифровку
    if (coalesce(f_get_rd_code(:c_citizen), '') not in ('Citizen.RU')
        and not exists
        (
            select cds.cds_id
            from clientdocs cds
            where cds.cds_from_c_id = :c_id
                and cds.cds_from_pk_id in (39, 46)
                and dateadd(day, 30, :ma_date_oper) between cds.cds_doc_date_begin and cds.cds_doc_date_end
                and cds.cds_is_delete = 0
                and cds.cds_verify_status in (-3, 2)
        )
    ) then
    begin
        res_text = 'Не добавлен РВП либо Вид на жительство';
        suspend;
        exit;

    end

    -- Отправляем запрос на проверку ИП в контук
    execute procedure hp_extinfo_insert_bki_cfg('kontur_ip_base', :c_id, :obj_kind, :obj_id, 0, :un_id);

    res_text = 'Запрос был отправлен';
    suspend;
end
