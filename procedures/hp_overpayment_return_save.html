CREATE PROCEDURE HP_OVERPAYMENT_RETURN_SAVE(
    K_ID ID_TYPE NOT NULL,
    C_ID ID_TYPE NOT NULL,
    SUM_OVP CURR_TYPE NOT NULL,
    UN_ID ID_TYPE,
    OPER_DATE DATE NOT NULL = current_date,
    LNK_K_ID ID_TYPE = null,
    DEFAULT_OFFICE INTEGER = null,
    AL_ID D_STR_20 = null)
RETURNS (
    ERR_MSG STR_255,
    OL_ID ID_TYPE)
AS
declare variable kontract_no varchar(20);
declare variable reg_id integer;
declare variable new_ol_id id_type;
declare variable ol_memo varchar(255);
declare variable account_overpayments varchar(20);
declare variable k_disb_date date;
declare variable error_text str_255;
declare variable kc_from_c_id id_type;
declare variable sum_zadol curr_type;
declare variable pereplata_fact curr_type;
declare variable lnk_kontract_no str_20;
declare variable last_ovp_date d_date;
declare variable reg_id_tmp integer;
declare variable c_comment d_str_100;
declare variable c_name_fio d_str_255;
begin
  err_msg = '';
  ol_id = -1;
  if (not exists(select *
                 from clients
                 where c_id = :c_id)) then
  begin
    err_msg = 'Клиент не указан!';
    suspend;
    exit;
  end

  -- Проверка по ПОД/ФТ
  for select c_name_fio, c_comment
      from hp_check_client_fio_ft(:c_id, 0)
      into c_name_fio, c_comment
  do
  begin
    err_msg = 'Совпадение с лицом по ' || c_comment || ' ' || c_name_fio || 'Продолжение невозможно.';
    suspend;
    exit;
  end

  kontract_no = null;
  select k_kontract_no, k_last_reg_id, k_disb_date
  from kontract k
  where k_id = :k_id
  into kontract_no, reg_id, k_disb_date;

  reg_id_tmp = reg_id;

  if (default_office is not null) then
    reg_id = default_office;

  if (kontract_no is null) then
  begin
    err_msg = 'Неверный номер контракта!';
    suspend;
    exit;
  end

  -- находим счет переплаты, из послденей операции переплаты
  select max(ok_from_ol_id), -sum(ok_os), max(iif(ok_os > 0, ok_ol_date, cast('01.01.91' as date)))
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type = 9
  into ol_id, pereplata_fact, last_ovp_date;
  if (ol_id = -1) then
  begin
    err_msg = 'В текущем контракте не было переплаты!';
    suspend;
    exit;
  end

  if (pereplata_fact <> sum_ovp) then
  begin
    err_msg = 'Есть расхождение в сумме переплаты!';
    suspend;
    exit;
  end

  if (lnk_k_id is null) then
  begin
    select account_overpayments, err_msg, new_ol_id
    from hp_save_return_money_oper('overpayment', :un_id, :reg_id, :reg_id_tmp, :k_disb_date, :kontract_no, :oper_date, :c_id, :sum_ovp, :k_id, :last_ovp_date, :al_id)
    into :account_overpayments, err_msg, ol_id;

    if (err_msg <> '') then
    begin
      suspend;
      exit;
    end
  end
  else
  begin
    sum_zadol = null;
    select sum(zad_os + sh_calc_perc + sh_calc_fine)
    from aarep_make_zad_list(:lnk_k_id, :oper_date, 1)
    into sum_zadol;
    if (sum_zadol is null) then
      sum_zadol = 0;

    if (sum_zadol <= 0) then
    begin
      select k_kontract_no
      from kontract
      where k_id = :lnk_k_id
      into lnk_kontract_no;

      err_msg = 'В контракте №:' || lnk_kontract_no || ' не имеется задолженность';
      suspend;
      exit;
    end
    else
    if (sum_zadol < sum_ovp) then
    begin
      err_msg = 'Сумма возврата превышает сумму задолженности, задолженность=' || sum_zadol;
      suspend;
      exit;
    end

    kc_from_c_id = null;
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :lnk_k_id and
          kc_amount_approved > 0
    into kc_from_c_id;

    ol_memo = 'Возврат переплаты по кредитному договору ' || kontract_no || ' от ' || f_datetostr(k_disb_date);

    error_text = null;
    select error_text, ol_id
    from hp_do_repayment(:lnk_k_id, 0, :oper_date, :sum_ovp, 1, -- :repay_to_os,
    1, -- :repay_to_perc,
    1, -- :repay_to_fine,
    1, -- :repay_to_kommis,
    1, -- :need_save,
    0, -- :is_full_repay,
    0, -- :is_writeoff,
    :reg_id, -- :repay_fact_office_id, -- головной офис в рф
    3, -- :repay_dt_account_kind, -- внутренний бух.счет европлата
    :account_overpayments, -- :repay_dt_account_no,
    1, -- :repay_curr_rate,
    :un_id, -- :repay_user_id,
    :kc_from_c_id, -- :repay_document_client,
    :ol_memo, -- :repay_doc_client_name_reason,
    null, -- :repay_manager_c_id,
    0, -- :repay_from_rd_id,
    0)
    into error_text, new_ol_id;
    error_text = coalesce(error_text, '');
    err_msg = '';
    if (error_text <> '') then
    begin
      err_msg = error_text;
      suspend;
      exit;
    end
    ol_id = new_ol_id;

  end

  execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, null);

  if (lnk_k_id is not null) then
    execute procedure hp_do_kn_operbook_one_online(:lnk_k_id, current_date, null, null);

  suspend;
end
