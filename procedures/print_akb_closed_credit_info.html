CREATE PROCEDURE PRINT_AKB_CLOSED_CREDIT_INFO(
    KBM_ID D_ID)
RETURNS (
    CREDITSTATUSDESCRIPTION D_STR_255,
    ACCOUNTNO D_STR_20,
    INITIALAMOUNT D_CURRENCY,
    LINEAMOUNT D_CURRENCY,
    MONTHLYPAYMENTAMOUNT D_CURRENCY,
    OUTSTANDINGDEBTMAIN D_CURRENCY,
    OUTSTANDINGDEBTINTEREST D_CURRENCY,
    DELINQDAYSMAIN D_INTEGER,
    DELINQDAYSINTEREST D_INTEGER,
    CREDITPURPOSE_DESCRIPTION D_STR_255,
    COLLATERALCODE D_STR_10,
    CREDITTYPEDESCRIPTION D_STR_255,
    ORGIDTYPEDESCRIPTION D_STR_255,
    BANKNAME D_STR_255,
    GRANTEDON D_DATE,
    CONTRACTDUEON D_DATE,
    LASTUPDATEDDATE D_DATE)
AS
declare variable CHECKED_DATE D_DATETIME;
declare variable JSON_TEXT D_BLOB_BINARY;
declare variable LIABILITIESCOUNT D_INTEGER;
declare variable I D_INTEGER;
declare variable LIABILITY D_BLOB_TEXT;
declare variable CREDITSTATUS varchar(10);
declare variable CREDITPURPOSE varchar(10);
declare variable CREDITTYPE varchar(10);
declare variable ORGIDTYPE varchar(10);
declare variable GRANTEDON_STR varchar(30);
declare variable CONTRACTDUEON_STR varchar(30);
declare variable LASTUPDATEDDATE_STR varchar(30);
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();
  json_text = '';
  select bls.bls_blob
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
  inner join kibminiapp kbm on eil.eil_object_id = kbm.kbm_id
  inner join blobstore bls on eil.eil_from_bls_id = bls.bls_id
  where eil.eil_status = 2 and
        kbm.kbm_ver_info = 3 and
        kbm.kbm_id = :kbm_id and
        eic.eic_code = 'akb'
  order by kbm.kbm_checked_datetime desc
  rows 1
  into :json_text;

  if (json_text <> '' or json_text <> '{}' or json_text is not null) then
  begin
  liabilitiescount = f_json_blob_get_count(:json_text, 'liabilities.liability') - 1;

      if (f_json_blob_get_value(:json_text, 'liabilities.liability', '') <> '') then
      begin
        i = 0;
        while (i <= liabilitiescount) do
        begin
          liability = f_json_blob_get_value(:json_text, 'liabilities.liability[' || :i || ']', '');
          creditstatus = f_json_blob_get_value(:liability, 'creditStatus', '');
          creditstatusdescription = 'Нерабочие';
          i = i + 1;
          if (creditstatus = '001') then
          begin
          accountno = f_json_blob_get_value(:liability, 'accountNo', '');
          initialamount = f_json_blob_get_value(:liability, 'initialAmount', '');
          initialamount = cast(initialamount as d_currency);
          lineamount = f_json_blob_get_value(:liability, 'lineAmount', '');
          lineamount = cast(lineamount as d_currency);
          monthlypaymentamount = f_json_blob_get_value(:liability, 'monthlyPaymentAmount', '');
          monthlypaymentamount = cast(monthlypaymentamount as d_currency);
          outstandingdebtmain = f_json_blob_get_value(:liability, 'outstandingDebtMain', '');
          outstandingdebtmain = cast(outstandingdebtmain as d_currency);
          outstandingdebtinterest = f_json_blob_get_value(:liability, 'outstandingDebtInterest', '');
          outstandingdebtinterest = cast(outstandingdebtinterest as d_currency);
          delinqdaysmain = f_json_blob_get_value(:liability, 'daysMainSumOverdue', '');
          delinqdaysinterest = f_json_blob_get_value(:liability, 'daysInterestOverdue', '');
          creditpurpose = f_json_blob_get_value(:liability, 'creditPurpose', '');
          creditpurpose_description = case
                                          when creditpurpose = '001' then iif(db_id <> 94401, 'Потребительские кредиты физическим лицам', 'Fiziki şəxslərə istehlak kreditləri')
                                          when creditpurpose = '002' then iif(db_id <> 94401, 'Нефтяной сектор', 'Neft sektoru')
                                          when creditpurpose = '003' then iif(db_id <> 94401, 'Ипотечные кредиты физическим лицам', 'Fiziki şəxslərə ipoteka kreditləri')
                                          when creditpurpose = '004' then iif(db_id <> 94401, 'Сектор торговли и обслуживания', 'Ticarət və xidmət sektoru')
                                          when creditpurpose = '005' then iif(db_id <> 94401, 'Сектор энергетики, химии и природных ресурсов', 'Energetika, kimya və təbii ehtiyatlar sektoru')
                                          when creditpurpose = '006' then iif(db_id <> 94401, 'Сектор сельского хозяйства и переработки', 'Kənd təsərrüfatı və emal sektoru')
                                          when creditpurpose = '007' then iif(db_id <> 94401, 'Сектор строительства и недвижимости', 'İnşaat və əmlak sektoru')
                                          when creditpurpose = '008' then iif(db_id <> 94401, 'Промышленность и производственный сектор', 'Sənaye və istehsal sektoru')
                                          when creditpurpose = '009' then iif(db_id <> 94401, 'Транспортно-коммуникационный сектор', 'Nəqliyyat və rabitə sektoru')
                                          when creditpurpose = '010' then iif(db_id <> 94401, 'Другие секторы', 'Digər sektorlar')
                                          when creditpurpose = '011' then iif(db_id <> 94401, 'Аренда жилого помещения с правом выкупа', 'Yaşayış sahəsinin satmaq öhdəliyi ilə kirayəyə alınması')
                                        end;

          collateralcode = f_json_blob_get_value(:liability, 'collateralCode', '');
          credittype = f_json_blob_get_value(:liability, 'creditType', '');
          credittypedescription = case
                                    when credittype = '001' then iif(db_id <> 94401, 'Кредит', 'Kredit')
                                    when credittype = '002' then iif(db_id <> 94401, 'Кредитная линия', 'Kredit xətti')
                                    when credittype = '003' then iif(db_id <> 94401, 'Кредитная карта', 'Kredit kartı')
                                    when credittype = '004' then iif(db_id <> 94401, 'Гарантия и поручительство', 'Qarantiya və zaminlik')
                                    when credittype = '005' then iif(db_id <> 94401, 'Овердрафт', 'Overdraft xətləri')
                                    when credittype = '007' then iif(db_id <> 94401, 'Лизинг', 'Lizinq')
                                    when credittype = '008' then iif(db_id <> 94401, 'Факторинг', 'Faktorinq')
                                    when credittype = '009' then iif(db_id <> 94401, 'Квартира в аренду', 'Kirayə mənzil')
                                    when credittype = '010' then iif(db_id <> 94401, 'Транш кредита', 'Tranş kredit')
                                  end;
          orgidtype = f_json_blob_get_value(:liability, 'creditType', '');
          orgidtypedescription = case
                                    when orgidtype = '001' then 'Банк'
                                    when orgidtype = '002' then 'НБКО'
                                    when orgidtype = '003' then 'Кредитное Бюро'
                                    when orgidtype = '004' then 'Страхование'
                                    when orgidtype = '005' then 'Лизинг'
                                    when orgidtype = '007' then 'Кредитный союз'
                                    when orgidtype = '008' then 'Другие юридические лица'
                                    when orgidtype = '009' then 'Частичная продажа'
                                    when orgidtype = '010' then 'Факторинг'
                                  end;
          bankname = f_json_blob_get_value(:liability, 'bankName', '');
          grantedon_str = f_json_blob_get_value(:liability, 'grantedOn', '');
          grantedon_str = substring(grantedon_str from 1 for 10);
          grantedon = cast(grantedon as d_date);
          contractdueon_str = f_json_blob_get_value(:liability, 'contractDueOn', '');
          contractdueon_str = substring(contractdueon_str from 1 for 10);
          contractdueon = cast(contractdueon_str as d_date);
          lastupdateddate_str = f_json_blob_get_value(:liability, 'lastUpdatedDate', '');
          lastupdateddate_str = substring(lastupdateddate_str from 1 for 10);
          lastupdateddate = cast(lastupdateddate_str as d_date);
          suspend;
          end
          end
        end
      end
end
