CREATE PROCEDURE AAREP_MC_AGREEMENT(
    PAR D_STR_255)
RETURNS (
    C_FIO D_STR_100,
    C_INN D_STR_100,
    C_PASS_NO D_STR_100,
    REGISTRATION_PHONE D_STR_100,
    REGISTRATION_DATETIME D_DATETIME,
    AGREEMENT_DATETIME D_DATETIME,
    PERSONAL_AGREEMENT_DATETIME D_DATETIME,
    LICENSE_URL D_STR_2048,
    AGREEMENT_URL D_STR_2048)
AS
declare variable C_ID D_ID;
declare variable MCL_ID D_ID;
declare variable PHONE D_STR_20;
declare variable PASS_NO D_STR_100;
declare variable FIO D_STR_100;
declare variable INN D_STR_20;
declare variable SQL_TEXT D_STR_2048;
declare variable LANG_CODE D_STR_LANG;
begin
  if (par <> '') then
  begin
    if (:par similar to '[0-9]+' and
        char_length(:par) = 13) then
    begin
      c_id = par;
      sql_text = 'select mcl.mcl_id, mcl.mcl_login, mcl.mcl_datetime, cl.c_name_fio, cl.c_pass_no, cl.c_inn, cl.c_id
                  from mc_login mcl
                  inner join clients cl on mcl.mcl_from_c_id = cl.c_id
                  where mcl.mcl_from_c_id = ' || :c_id;
    end
    if (:par similar to '[0-9]+' and
        c_id is null) then
    begin
      inn = par;
      sql_text = 'select mcl.mcl_id, mcl.mcl_login, mcl.mcl_datetime, cl.c_name_fio, cl.c_pass_no, cl.c_inn, cl.c_id
                  from mc_login mcl
                  inner join clients cl on mcl.mcl_from_c_id = cl.c_id
                  where cl.c_inn = ''' || :inn || '''
                  order by mcl.mcl_datetime asc';
    end
    if (par similar to '%[а-яА-ЯA-Za-z]%' and
        not par similar to '%[0-9]%') then
    begin
      fio = par;
      sql_text = 'select mcl.mcl_id, mcl.mcl_login, mcl.mcl_datetime, cl.c_name_fio, cl.c_pass_no, cl.c_inn, cl.c_id
                  from mc_login mcl
                  inner join clients cl on mcl.mcl_from_c_id = cl.c_id
                  where cl.c_name_fio = ''' || :fio || '''
                  order by mcl.mcl_datetime asc';
    end
    if (:par similar to '(79|992|996|998|994)[[:DIGIT:]]{9}' or (:par similar to '(9960|9961|9963|9964)[[:DIGIT:]]{8}')) then
    begin
      phone = par;
      sql_text = 'select mcl.mcl_id, mcl.mcl_login, mcl.mcl_datetime, cl.c_name_fio, cl.c_pass_no, cl.c_inn, cl.c_id
                  from mc_login mcl
                  left join clients cl on mcl.mcl_from_c_id = cl.c_id
                  where mcl.mcl_login starting ''' || :phone || '''
                  order by mcl.mcl_datetime asc';
    end
    if (par similar to '%[а-яА-ЯA-Za-z0-9]%' and
        fio is null and
        inn is null and
        c_id is null) then
    begin
      pass_no = par;
      sql_text = 'select mcl.mcl_id, mcl.mcl_login, mcl.mcl_datetime, cl.c_name_fio, cl.c_pass_no, cl.c_inn, cl.c_id
                  from mc_login mcl
                  inner join clients cl on mcl.mcl_from_c_id = cl.c_id
                  where cl.c_index_c_pass_no = ''' || :pass_no || '''
                  order by mcl.mcl_datetime asc';
    end

    for execute statement :sql_text
            into :mcl_id, :registration_phone, :registration_datetime, :c_fio, :c_pass_no, :c_inn, :c_id
    do
    begin
      if (registration_phone containing '_') then
        registration_phone = left(registration_phone, position('_' in registration_phone) - 1);

      agreement_datetime = null;
      lang_code = null;
      license_url = '';

      select mca.mca_datetime,
             mca.mca_lang_code
      from mc_agreement mca
      where mca.mca_from_mcl_id = :mcl_id and
            mca.mca_document = f_enum_id('mc_document', 'mobile_app_use')
      into :agreement_datetime,
           :lang_code;

      if (agreement_datetime is null) then
        continue;

      license_url = (select url from get_mc_agreement_url('mobile_app_use', cast(:agreement_datetime as date), :lang_code));

      ------------------------
      personal_agreement_datetime = null;
      lang_code = null;
      agreement_url = '';

      select mca.mca_datetime,
             mca.mca_lang_code
      from mc_agreement mca
      where mca.mca_from_mcl_id = :mcl_id and
            mca.mca_document = f_enum_id('mc_document', 'personal_data_processing')
      into :personal_agreement_datetime,
           :lang_code;

      if (personal_agreement_datetime is not null) then
        agreement_url = (select url from get_mc_agreement_url('personal_data_processing', cast(:personal_agreement_datetime as date), :lang_code));

      suspend;
    end
  end
end
