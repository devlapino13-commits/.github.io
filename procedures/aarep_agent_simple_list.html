CREATE PROCEDURE AAREP_AGENT_SIMPLE_LIST(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    KE_DATE DATE,
    KE_ORDER_NO VARCHAR(20),
    KE_IS_KRS SMALLINT,
    KE_STATUS SMALLINT,
    KE_FROM_REG_ID INTEGER,
    KE_CODE VARCHAR(20),
    KE_OPER_DAYS SMALLINT,
    KE_KOLICH_BEGIN INTEGER,
    KE_KOLICH_NEW INTEGER,
    KE_KOLICH_END INTEGER,
    KE_KOLICH_WORK INTEGER,
    KE_AGENT_FIO VARCHAR(100),
    KE_OFFICER_NAME VARCHAR(100),
    KE_OFFICER_CODE VARCHAR(20),
    KE_CLIENT_KOLICH INTEGER,
    KE_REGION_NAME VARCHAR(100),
    KE_CLIENT_SUMMA DECIMAL(18,4))
AS
declare variable total_kol integer;
declare variable ke_from_p_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ke_date, ke_order_no, ke_is_krs, ke_status, ke_from_reg_id, ke_code, ke_oper_days, c_name_fio,
             ke_from_p_id, reg_code || ' ' || reg_name
      from krexpert, clients, region_list(:report_region_id, 1, -1, 1), region
      where ke_is_krs = 3 and
            ke_from_p_id = c_id and
            (ke_from_reg_id = id_list) and
            ke_from_reg_id = reg_unicode
      into :ke_date, :ke_order_no, :ke_is_krs, :ke_status, :ke_from_reg_id, :ke_code, :ke_oper_days, :ke_agent_fio,
           :ke_from_p_id, :ke_region_name
  do
  begin
    ke_kolich_work = 0;
    if (ke_date < :report_date_end) then
    begin
      ke_kolich_end = 1;
      if (ke_date < :report_date_begin) then
      begin
        ke_kolich_begin = 1;
        ke_kolich_new = 0;
      end
      else
      begin
        ke_kolich_begin = 0;
        ke_kolich_new = 1;
      end
      total_kol = 0;
      ke_client_kolich = 0;
      ke_client_summa = 0;
      ke_kolich_work = 0;
      for select kal_new_officer_code, c_name_fio, count(ks_from_k_id), sum(kc_amount_approved)
          from kontstate, kontract, knclient, knaktlist, clients, region_list(:report_region_id, 1, -1, 1)
          where ks_from_k_id = k_id and
                kc_from_k_id = k_id and
                kal_new_kf_officer = c_id and
                kal_from_kf_id = k_from_kf_id and
                ks_date_begin between :report_date_begin and :report_date_end and
                ks_state = 1 and
                kc_agent_id = :ke_from_p_id and
                kal_new_reg_id = id_list and
                kal_date_end >= :report_date_end and
                kal_date <= :report_date_end and
                ks_ol_is_delete = 0 and
                knaktlist.is_delete = 0
          group by kal_new_officer_code, c_name_fio
          into :ke_officer_code, :ke_officer_name, :ke_client_kolich, :ke_client_summa
      do
      begin
        if (ke_client_kolich is null) then
          ke_client_kolich = 0;
        if (ke_client_summa is null) then
          ke_client_summa = 0;
        total_kol = 1;
        ke_kolich_work = 1;
        suspend;
        ke_kolich_work = 0;
        ke_kolich_begin = 0;
        ke_kolich_new = 0;
        ke_kolich_end = 0;
      end
      if (total_kol = 0) then
      begin
        ke_officer_code = '---';
        ke_officer_name = 'Нет выдач';
        ke_client_kolich = 0;
        suspend;
      end
    end
  end
end
