CREATE PROCEDURE HP_EXPRESSAPP_CHECK(
    REP_EXA_ID D_INTEGER,
    NEED_CHECK D_BOOL = 0,
    DEBUG_MODE D_BOOL = 0)
RETURNS (
    EXA_ID D_INTEGER,
    C_ID D_ID,
    MA_ID D_ID,
    K_ID D_ID,
    EXC_ID D_INTEGER,
    EXC_NAME D_STR_255,
    ACTION_KIND D_STR_CODE,
    ACTION_CODE D_STR_CODE,
    CHECK_ID D_INTEGER,
    CDM_ID D_INTEGER,
    CHECK_MESSAGE D_STR_255)
AS
declare variable obj_id d_id;
declare variable obj_kind d_integer;
declare variable check_kind d_integer;
begin
  if (rep_exa_id is null) then
    rep_exa_id = 0;

  for select exa_id, exa_from_c_id
  from expressapp
  where (exa_id = :rep_exa_id or :rep_exa_id = 0)
  into exa_id, c_id
  do
  begin
    if (need_check = 1) then
    begin
      update expressapp
      set exa_from_exc_id = 30
      where exa_id = :exa_id and
            exa_from_exc_id <= 30;

      execute procedure hp_do_expressapp_process(:exa_id, :debug_mode);
    end

    select exa_from_ma_id, exa_from_k_id, exa_from_exc_id, exc_name,
           exc_action_kind, exc_action_code, exc_obj_kind
    from expressapp, expresscfg
    where exa_from_exc_id = exc_id and
          exa_id = :exa_id
    into ma_id, k_id, exc_id, exc_name, action_kind, action_code, obj_kind;

    cdm_id = null;
    check_message = '';
    check_kind = null;
    check_id = null;

    if (action_kind = 'check') then
    begin
      obj_id = case obj_kind
        when 9 then k_id
        when 11 then ma_id
        when 60 then exa_id
      end;

      if (obj_id is null) then
        exception e_default 'Unknown obj_id';

      check_kind = f_enum('condition_check_kind', action_code, 'id');

      select ccs_id
      from condcheckstate
      where ccs_object_kind = :obj_kind and
            ccs_object_id = :obj_id and
            ccs_check_kind = :check_kind and
            ccs_errors > 0
      into check_id;

      if (check_id is not null) then
      begin
        for select ccr_from_cdm_id, ccr_message
        from condcheckresult
        where ccr_from_ccs_id = :check_id
        into cdm_id, check_message
        do
          suspend;
      end
    end

    if (cdm_id is null) then
      suspend;
  end
end
