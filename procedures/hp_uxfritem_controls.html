CREATE PROCEDURE HP_UXFRITEM_CONTROLS(
    LIST_KIND D_STR_CODE)
RETURNS (
    EDITOR_ID D_STR_CODE,
    EDITOR_NAME D_STR_CODE,
    EDITOR_MEMO D_STR_255)
AS
begin
  for select editor_id, editor_name, editor_memo
  from hp_uxfritem_control_kind
  where editor_kind = :list_kind or
        :list_kind = 'full'
  into :editor_id, :editor_name, :editor_memo
  do
    suspend;
end
