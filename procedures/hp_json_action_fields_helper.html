CREATE PROCEDURE HP_JSON_ACTION_FIELDS_HELPER(
    TABLE_NAME D_STR_CODE,
    OBJECT_NAME D_STR_CODE)
RETURNS (
    READY_CODE D_STR_8100)
AS
declare variable def_is_null d_smallint;
declare variable def_str d_str_code;
declare variable res_field_name d_str_code;
declare variable res_field_type d_str_code;
declare variable res_field_size d_integer;
declare variable res_txt d_str_long;
begin
  -- Если Code_kind = sql, то код будет формироваться для IBExpert:
  -- Процедура формирует из списка полей готовый код для процедуры сохранения в JSON_Action...
  -- Пример кода ниже?

  /*
      for select f_json_str_get(:ja_json, 'ob_number', 'value', null, 1) as ob_number,
                 f_json_str_get(:ja_json, 'ob_currency', 'value', null, 1) as ob_currency,
...
                 f_json_str_get(:ja_json, 'ob_from_sac_id', 'value', null, 1) as ob_from_sac_id
          from rdb$database as cursor ob
      do
      begin
      end
  */

  -- Если Code_kind <> sql, то код будет формироваться для Delphi:
  -- Будет формирован код для создания FDMemTable

  if (coalesce(object_name, '') = '') then
    object_name = 'sql';

  if (object_name = 'sql') then
  begin
    ready_code = '';
    for select res_field_name, res_field_type, iif(res_is_key_field = 1 or res_fk_field <> '', 1, 0)
        from get_table_system_info(:table_name)
        into :res_field_name, :res_field_type, :def_is_null
    do
    begin
      def_str = 'null';
      if (def_is_null = 0) then
      begin
        if (res_field_type in ('smallint', 'integer', 'float', 'bigint', 'percent', 'currency')) then
          def_str = '0';
        else
        if (res_field_type in ('date', 'datetime', 'float', 'bigint', 'percent', 'currency')) then
          def_str = 'null';
        else
          def_str = '''''';
      end
      if (ready_code <> '') then
        ready_code = ready_code || ',
';
      ready_code = ready_code || '    f_json_str_get(:ja_json, ''' || res_field_name || ''', ''value'', ' || def_str || ', 1) as ' || res_field_name;
    end
    ready_code = '
/*
for select
' || ready_code || '
from rdb$database as cursor cur
do
begin
end
*/';
  end
  else
  begin
    ready_code = '
object '||object_name||': TFDMemTable
  FetchOptions.AssignedValues = [evMode]
  FetchOptions.Mode = fmAll
  ResourceOptions.AssignedValues = [rvSilentMode]
  ResourceOptions.SilentMode = True
  UpdateOptions.AssignedValues = [uvCheckRequired, uvAutoCommitUpdates]
  UpdateOptions.CheckRequired = False
  UpdateOptions.AutoCommitUpdates = True
  Left = 10
  Top = 10
';

    -- object vKnOperBookkob_id: TLargeintField FieldName = 'kob_id' end
    for select res_field_name, res_field_type, res_field_size
        from get_table_system_info(:table_name)
        into :res_field_name, :res_field_type, res_field_size
    do
    begin
      if (res_field_type = 'bigint') then
        res_field_type = 'LargeInt';
      else
      if (res_field_type in ('percent', 'percent', 'double')) then
        res_field_type = 'Float';
      res_field_type = upper(left(res_field_type, 1)) || substring(res_field_type from 2);

      res_field_name = upper(left(res_field_name, 1)) || substring(res_field_name from 2);
      res_txt = '  object ' || object_name || res_field_name || ': T' || res_field_type || 'Field FieldName = ''' || res_field_name || '''';

      if (res_field_type = 'String') then
        res_txt = res_txt || ' size = ' || res_field_size;
      res_txt = res_txt || ' end
';
      ready_code = ready_code || res_txt;
    end

    for select res_field_name, res_field_type, res_field_size
        from get_table_system_info(:table_name)
        where res_fk_table <> ''
        into :res_field_name, :res_field_type, res_field_size
    do
    begin
      res_field_size = 255;
      res_field_type = 'String';

      res_field_name = upper(left(res_field_name, 1)) || substring(res_field_name from 2);
      res_field_name = res_field_name || '_name';

      res_txt = '  object ' || object_name || res_field_name || ': T' || res_field_type || 'Field FieldName = ''' || res_field_name || '''';

      if (res_field_type = 'String') then
        res_txt = res_txt || ' size = ' || res_field_size;
      res_txt = res_txt || ' end
';
      ready_code = ready_code || res_txt;
    end

    ready_code = ready_code || '
end';
  end
  suspend;
end
