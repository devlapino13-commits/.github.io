Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_UXCODE_RENAME(
    UX_CODE D_STR_CODE,
    NEW_UX_CODE D_STR_CODE)
AS
begin
  ux_code = trim(coalesce(ux_code, ''));
  new_ux_code = trim(coalesce(new_ux_code, ''));

  if (ux_code = '') then
      exception e_default f_format('Empty old ux_code');

  if (new_ux_code = '') then
      exception e_default f_format('Empty new ux_code');

  if (ux_code = new_ux_code) then
    Exit;

  if (exists (select * from uxcode where ux_code = :new_ux_code)) then
    exception e_message f_format('В базе существует объект с таким именем: %1', new_ux_code);

  update uxcode
  set ux_code = :new_ux_code
  where ux_code = :ux_code;

  if (row_count = 0) then
    exception e_message f_format('Не найден объект с названием: %1', ux_code);
end
