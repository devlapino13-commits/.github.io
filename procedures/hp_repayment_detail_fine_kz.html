CREATE PROCEDURE HP_REPAYMENT_DETAIL_FINE_KZ(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    VID_SUM VARCHAR(50),
    S_ALL DECIMAL(18,2),
    KOL_DAY INTEGER,
    RATE_PER_DAY D_CURRENCY,
    CALC_FINE DECIMAL(18,0))
AS
declare variable k_disb_date date;
declare variable sh_date date;
declare variable sh_os_accum decimal(18,2);
declare variable prev_date date;
begin
  select k_disb_date
  from kontract
  where k_id = :k_id
  into :k_disb_date;

  for select sh_date, s_all, sh_calc_fine
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      where sh_date is not null and
            sh_calc_fine <> 0
      order by sh_date
      into :sh_date, :s_all, :calc_fine
  do
  begin
    select max(sh.sh_date)
    from schedule sh
    where sh.sh_from_k_id = :k_id and
          sh.sh_id > 0 and
          sh.sh_date < :sh_date
    into :prev_date;

    if (prev_date is null) then
      prev_date = k_disb_date;

    kol_day = minvalue(:rep_date, :sh_date) - :prev_date;

    vid_sum = 'Ежемесячный взнос (ОД + %)';

    if (s_all <> 0 and kol_day > 0) then
      rate_per_day = round(cast(calc_fine as decimal(18,7)) / (s_all * kol_day) * 100, 4);
    else
      rate_per_day = 0;

    suspend;
  end
end
