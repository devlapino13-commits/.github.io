CREATE PROCEDURE HP_EDOC_WEB_SIGN(
    UUID D_STR_CODE,
    CODE D_STR_CODE)
RETURNS (
    RES_TEXT D_STR_255,
    RES_TEXT_STYLE D_STR_CODE,
    RES_BUTTON_STATE D_STR_CODE)
AS
declare variable obj_id d_id;
declare variable eds_status d_integer;
declare variable edm_status d_smallint;
declare variable edl_status d_integer;
declare variable hash_check d_smallint;
declare variable edl_id d_id;
declare variable sign_kind d_str_code;
declare variable obj_code d_str_code;
declare variable cv_status d_integer;
declare variable msg_id d_integer;
declare variable eds_from_c_id d_id;
declare variable tmp_text d_str_255;
declare variable res_is_error d_integer;
declare variable edl_object_kind d_integer;
declare variable edl_object_id d_id;
declare variable edc_code d_str_code;
declare variable disb_kind d_smallint;
begin
  res_text = '';
  res_text_style = 'error';
  res_button_state = 'disabled';

  if (code is null) then code = '';

  obj_code = '';
  select obj_kind, obj_id
  from hp_uuid(:uuid)
  into obj_code, :obj_id;

  if (obj_id is null)
  then res_text = 'Неверная ссылка';
  else obj_code = f_objkind(obj_code, 'name');

  if (obj_code = 'edoc_sign') then
  begin
    if (res_text = '') then
    begin
      select edl_id, eds_status, iif(coalesce(edl_date_end, current_date) < current_date, -1, edl_status),
             edl_object_kind, edl_object_id, edc_code
      from edoc_sign, edoc_list, edoc_cfg
      where eds_id = :obj_id and
            eds_from_edl_id = edl_id and
            edl_from_edc_id = edc_id
      into edl_id, eds_status, edl_status, edl_object_kind, edl_object_id, edc_code;
    
      if (res_text = '' and edl_status = -1) then res_text = 'Данные устарели';
      if (res_text = '' and eds_status = 1) then
      begin
        res_text = 'Документы уже подписаны';
        res_text_style = 'normal';
      end
    end
  
    if (res_text = '') then
    begin
      select edm_from_msg_id, edm_status, iif(:code = '', 0, iif(edm_hash_pin in (:code, f_md5(:code)), 1, 0)),
             eds_from_c_id
      from edoc_msg, edoc_sign
      where eds_id = :obj_id and
            edm_from_eds_id = eds_id
      order by iif(:code = '', 0, iif(edm_hash_pin in (:code, f_md5(:code)), 1, 0)) desc
      rows 1 -- берем наиболее успешный статус
      into msg_id, edm_status, hash_check,
           eds_from_c_id;
  
      if (res_text = '' and edm_status is null) then res_text = 'Неверная ссылка';
      if (res_text = '' and edm_status = 1) then
      begin
        res_text = 'Документы уже подписаны';
        res_text_style = 'normal';
      end
      if (res_text = '' and hash_check = 0) then
      begin
        res_text = 'Неверный код';
        res_button_state = 'enabled';
      end
    end

    if (edc_code = 'loan.application' and f_can_disb_on_card(edl_object_kind, edl_object_id) = 0) then
    begin
      -- Если выдача по СБП, можно подписывать только через КМП.
      select ma_disb_kind
      from miniapp
        inner join product on ma_from_pr_id = pr_id
      where ma_id = :edl_object_id
        and ma_is_active = 1
        and ma_date_end >= '31.12.2999'
      into disb_kind;
      if (disb_kind = 7) then
        res_text = 'Вы получаете онлайн-заем по СБП, поэтому Заявление на заем должно быть подписано через Клиентское мобильное приложение "M Bulak"';
      if (disb_kind = 8) then
        res_text = 'Вы получаете онлайн-заем на расчетный счет, поэтому Заявление на заем должно быть подписано через Клиентское мобильное приложение "M Bulak"';
    end

    if (res_text = '') then
    begin
      sign_kind = 'online';
  
      if (eds_from_c_id is null) then
      begin
        sign_kind = 'paper';

        -- если ручное подписание и сейчас подписывает менеджер (андерайтер), подписываем у всех
        update edoc_sign
        set eds_status = 1,
            eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', :sign_kind, 'id')
        where eds_from_edl_id = :edl_id and
              eds_status = 0
        order by eds_from_c_id nulls first;
      end

      update edoc_msg
      set edm_status = 1
      where edm_from_eds_id = :obj_id and
            edm_from_msg_id = :msg_id and
            edm_status = 0;

      if (eds_from_c_id is not null) then
      begin
        update edoc_sign
        set eds_enum_edoc_sign_kind =  f_enum('edoc_sign_kind', :sign_kind, 'id'),
            eds_pin = :code
        where eds_id = :obj_id and
              (
                eds_enum_edoc_sign_kind is distinct from  f_enum('edoc_sign_kind', :sign_kind, 'id') or
                eds_pin is distinct from :code
              );
      end

      select res_is_error, res_text
      from hp_edoc_com_action(:edl_id, 1000000000000) -- пользователь fsrobot
      into :res_is_error, :tmp_text;

      if (res_is_error = 0) then
      begin
        res_text = 'Документы успешно подписаны';
        res_text_style = 'normal';
      end else
      begin
        res_text_style = 'error';
        res_text = tmp_text;
      end
    end
  end else
  if (obj_code = 'contveri') then
  begin
    -- тут проверка правиильности подписи из ContVeri
    select cvr_status, iif(cvr_hash_pin in (f_coder_pass(:code), f_md5(:code)), 1, 0)
    from contveri
    where cvr_id = :obj_id
    into cv_status, hash_check;

    if (res_text = '' and cv_status = 1) then
    begin
      res_text = 'Данные уже подтверждены';
      res_text_style = 'normal';
    end
    if (res_text = '' and hash_check = 0) then
    begin
      res_text = 'Неверный код';
      res_button_state = 'enabled';
    end

    if (res_text = '' and hash_check = 1) then
    begin
      update contveri
      set cvr_status = 1,
          cvr_sign_kind = f_enum('edoc_sign_kind', 'online', 'id')
      where cvr_id = :obj_id;

      res_text = 'Успешно подтвержден';
      res_text_style = 'normal';

      execute procedure hp_contveri_sign_sync(:obj_code, :obj_id);
    end
  end else
  if (obj_code = 'contverify') then
  begin
    -- тут проверка правиильности подписи из ContVerify
    select cv_status, iif(cv_password in (f_coder_pass(:code), f_md5(:code)), 1, 0)
    from contverify
    where cv_id = :obj_id
    into cv_status, hash_check;

    if (res_text = '' and cv_status = 1) then
    begin
      res_text = 'Данные уже подтверждены';
      res_text_style = 'normal';
    end
    if (res_text = '' and hash_check = 0) then
    begin
      res_text = 'Неверный код';
      res_button_state = 'enabled';
    end
    if (res_text = '' and hash_check = 1) then
    begin
      update contverify
      set cv_status = 3
      where cv_id = :obj_id;

      res_text = 'Успешно подтвержден';
      res_text_style = 'normal';

      execute procedure hp_contveri_sign_sync(:obj_code, :obj_id);
    end
  end else
  begin
    res_text = 'Неверная ссылка';
  end

  Suspend;
end
