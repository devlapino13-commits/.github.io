CREATE PROCEDURE FT_FM_MINIAPP_ACTION(
    MAC_FROM_C_ID ID_TYPE,
    LO_REGION_ID INTEGER,
    LO_ID ID_TYPE,
    UN_ID ID_TYPE)
RETURNS (
    CL_CHECK INTEGER,
    MA_RESULT_ID INTEGER,
    MA_RESULT_TXT D_STR_255)
AS
declare variable ft_name_fio str_255;
declare variable c_comment str_255;
declare variable db_id integer;
declare variable temp_str d_str_255;
begin

  :db_id = f_get_db_id();
  ma_result_txt = '';

-- Поиск по перечню экстимистов ------------------------------------------->>
  for select coalesce(c_name_fio, ''), coalesce(c_comment, '')
      from hp_check_client_fio_ft(:mac_from_c_id, 1)
      into :ft_name_fio, :c_comment
  do
  begin
    if (db_id <> 81001) then
    begin
      cl_check = 1;
      ma_result_id = -1;

      ma_result_txt = BLOB_APPEND(ma_result_txt, '<br><b><font color="red">' || 'При регистрации заявки выявлено cовпадение с лицом по ' || c_comment || ' ' || ft_name_fio || '</font></b><br>');
    end

    if (db_id = 81001) then
    begin
      cl_check = 1;
      ma_result_id = -1;

      ma_result_txt = BLOB_APPEND(ma_result_txt, '<br><b><font color="red">' || 'НЕОБХОДИМО срочно связаться с Кассиром-оператором' || '</font></b><br>');

      temp_str = 'c_comment=' || coalesce(c_comment, '');
      temp_str = temp_str || '&ft_name_fio=' || coalesce(ft_name_fio, '');
      temp_str = temp_str || '&c_id=' || coalesce(cast(mac_from_c_id as str_100), '');
      temp_str = temp_str || '&reg_id=' || coalesce(cast(lo_region_id as str_100), '');
      temp_str = temp_str || '&lo_id=' || coalesce(cast(lo_id as str_100), '');
      temp_str = temp_str || '&db_id=' || coalesce(cast(db_id as str_100), '');
      temp_str = temp_str || '&finpolis_type=all_critical';

      execute procedure fsrobot_insert_event('DoMailing', 'FT_CLIENT_FOUND', temp_str, un_id);
    end
-- << ------------------------------------------------------------------
-- Поиск по Перечню 639 ---------------------------------------------->>
    if (db_id = 81001) then
    begin
      for select coalesce(c_name_fio, ''), coalesce(c_comment, '')
          from hp_check_client_fio_ft639(:mac_from_c_id, '', '', '', null, 1)
          where c_comment_id = 1
          into :ft_name_fio, :c_comment
      do
      begin
        temp_str = 'c_comment=' || coalesce(c_comment, '');
        temp_str = temp_str || '&ft_name_fio=' || coalesce(ft_name_fio, '');
        temp_str = temp_str || '&c_id=' || coalesce(cast(mac_from_c_id as str_100), '');
        temp_str = temp_str || '&reg_id=' || coalesce(cast(lo_region_id as str_100), '');
        temp_str = temp_str || '&lo_id=' || coalesce(cast(lo_id as str_100), '');
        temp_str = temp_str || '&db_id=' || coalesce(cast(db_id as str_100), '');
        temp_str = temp_str || '&finpolis_type=639';
        execute procedure fsrobot_insert_event('DoMailing', 'FT_CLIENT_FOUND', temp_str, un_id);
      end

    end
-- << ------------------------------------------------------------------

    suspend;
  end
end
