CREATE PROCEDURE UXP$UXFRAME_ADD(
    UX_OWNER D_UUID,
    FR_CODE D_STR_CODE,
    FR_NAME D_STR_100,
    FR_IS_GRID D_BOOL,
    FR_SOURCE_OR_SQL D_STR_8100,
    UX_OWNER_CODE D_STR_CODE = null)
RETURNS (
    RES_FR_UUID D_UUID,
    RES_FR_UUID_STR D_STR_CODE,
    US_UUID D_UUID)
AS
declare variable ux_owner_kind d_str_code;
declare variable tmp_uuid_str d_str_code;
declare variable grid_ufi_uuid d_uuid;
declare variable res_kind d_str_code;
declare variable res_source_code d_str_code;
declare variable res_sql_select d_str_8100;
declare variable res_sql_where d_str_long;
declare variable res_sql_orderby d_str_long;
declare variable next_num d_integer;
declare variable fr_source d_str_code;
declare variable grid_parent_uuid d_uuid;
declare variable item_count d_integer;
declare variable usr_edi_uuid d_uuid;
declare variable tmp_parent_uuid d_uuid;
declare variable grour_uuid_str d_str_code;
declare variable param_count d_integer;
begin
  if (coalesce(ux_owner, '') = '') then
    ux_owner = uxf$uniq(ux_owner_code);

  ux_owner_kind = uxf$code(ux_owner, 'kind');

  if (fr_code is null) then
    fr_code = '';

  fr_code = trim(fr_code);

  if (exists(select * from ux$frame
  where ufr_code = :fr_code and
        ufr_owner = :ux_owner)) then
  begin
    if (fr_code <> '') then
      exception e_message 'Кодовое название фрейма занято, придумайте новый код или оставьте пустым.';
    begin
      -- Находим первый свободный номер
      next_num = 0;
      while (fr_code = '') do
      begin
        next_num = next_num + 1;
        if (not exists(select * from ux$frame
        where ufr_code = :next_num and
              ufr_owner = :ux_owner)) then
          fr_code = next_num;
      end
    end
  end

  fr_source = '';
  grour_uuid_str = '';

  if (fr_source_or_sql is null) then
    fr_source_or_sql = '';

  fr_source_or_sql = trim(fr_source_or_sql);

  -- Если в тексте нет ниодного пробела, значит передан источник
  if (position(' ', fr_source_or_sql) = 0 and
      position('(', fr_source_or_sql) = 0) then
  begin
    fr_source = fr_source_or_sql;
    fr_source_or_sql = '';
  end

  if (fr_source <> '' and fr_source_or_sql <> '') then
    exception e_message 'Укажите только одну из них: источник или SQL-текст';

  res_kind = '';
  res_source_code = '';
  res_sql_select = '';
  res_sql_where = '';
  res_sql_orderby = '';

  if (fr_source_or_sql <> '') then
  begin
    select res_kind, res_source_code, res_sql_select, res_sql_where, res_sql_orderby
    from uxp$sql_parcer(:fr_source_or_sql)
    into :res_kind, :res_source_code, :res_sql_select, :res_sql_where, :res_sql_orderby;
  
    if (res_kind = 'table') then
    begin
      -- Если источник одна и это таблица, при наличии параметром пропускаем как SQL текст
      if (exists(select * from ux$sqlfields
      where usf_is_field = 0 and             
            usf_source = :res_source_code)) then
        res_kind = 'sql';
    end
  
    if (res_kind = 'sql') then
    begin
      fr_source = '';
      res_sql_where = '';
      res_sql_orderby = '';
    end
    else
    begin
      fr_source = res_source_code;
      res_kind = '';
    end
  end

  if (fr_source <> '') then
  begin
    us_uuid = uxf$uniq(fr_source, 0);
    if (fr_source <> '' and us_uuid is null) then
    begin
      execute procedure uxp$uxsource_create(:fr_source, '', null, 0, '');
      us_uuid = uxf$uniq(fr_source);
    end
  end else
  if (res_kind = 'sql') then
  begin
    select res_uuid
    from uxp$uxsource_sql_to_uuid(:res_sql_select)
    into :us_uuid;
  
    if (not exists(select * from ux$source
    where us_uuid = :us_uuid)) then
    begin
      -- парсинг SQL и создание источника
      if (not exists(select * from ux$sqlfields)) then
        exception e_default 'Sql fields are empty';

      insert into ux$source (us_uuid, us_code, us_is_table, us_kind, us_sql)
      values (:us_uuid, uuid_to_char(:us_uuid), 0, -1, :res_sql_select);
    end
  end else
  begin
    select ux_us_uuid
    from ux$form, ux$code
    where ufm_uuid = :ux_owner and
          ufm_ux_uuid = ux_uuid
    into us_uuid;
  end

  insert into ux$frame (ufr_owner, ufr_code, ufr_name, ufr_us_uuid, ufr_is_grid, ufr_sql_where, ufr_sql_orderby)
  values (:ux_owner, :fr_code, :fr_name, :us_uuid, :fr_is_grid, :res_sql_where, :res_sql_orderby)
  returning (ufr_uuid)
  into res_fr_uuid;

  res_fr_uuid_str = uuid_to_char(res_fr_uuid);

  if (fr_source_or_sql <> '') then
    execute procedure uxp$uxframe_update_sql(:res_fr_uuid, :res_sql_select);

  if (ux_owner_kind <> 'uxreport') then
  begin
    update or insert into ux$fritem (ufi_owner, ufi_ufr_uuid, ufi_kind)
    values (:ux_owner, :res_fr_uuid, 1)
    matching (ufi_owner, ufi_ufr_uuid);
  end

  -- Синхронизируем поля фрейма
  execute procedure uxp$uxframe_sync_params(:res_fr_uuid, -1);

  if (ux_owner_kind <> 'uxreport') then
  begin
    if (fr_is_grid = 1) then
    begin
      -- Если набор данных, создаем грид
  
      select ufi_uuid_str
      from uxp$uxfritem_add_aux(:res_fr_uuid, 'group', 'grid', null)
      into tmp_uuid_str;
  
      if (coalesce(tmp_uuid_str, '') = '') then
        exception e_message 'Не удалось создать набор данных';
  
      select ufi_uuid, ufi_parent_uuid
      from ux$fritem
      where ufi_owner = :res_fr_uuid and
            ufi_control in ('grid', 'tree')
      into grid_ufi_uuid, grid_parent_uuid;
    end
  
    usr_edi_uuid = uxf$uuid(uxf$uniq('sys$params'), 'uxfield', 'current$user');

    param_count = 0;
    select count(*)
    from ux$field
    where uf_owner = :us_uuid and
          uf_is_param = 1 and
          (uf_edi_uuid is null or uf_edi_uuid is distinct from :usr_edi_uuid)
    into param_count;

    if (param_count > 1) then
    begin
      -- Создаем группу для параметров
      select ufi_uuid_str
      from uxp$uxfritem_add_aux(:res_fr_uuid, 'group', '', :res_fr_uuid)
      into grour_uuid_str;

      if (grour_uuid_str <> '') then
      begin
        update ux$fritem
        set ufi_group_direction = iif(:param_count >=3 ,1, ufi_group_direction),
            ufi_caption_align = -1
        where ufi_uuid = char_to_uuid(:grour_uuid_str);
      end
    end

    tmp_parent_uuid = res_fr_uuid;
    if (grour_uuid_str <> '') then
      tmp_parent_uuid = char_to_uuid(grour_uuid_str);

    -- Добавляем параметры в интерфейс
    insert into ux$fritem (ufi_owner, ufi_parent_uuid, ufi_uf_uuid, ufi_kind)
    select :res_fr_uuid, :tmp_parent_uuid, uf_uuid, 0
    from ux$field
    where uf_owner = :us_uuid and
          uf_is_param = 1 and
          (uf_edi_uuid is null or uf_edi_uuid is distinct from :usr_edi_uuid)
    order by uf_position;
  
    tmp_parent_uuid = coalesce(:grid_ufi_uuid, :res_fr_uuid);
    -- Добавляем все поля в интерфейс
    if (ux_owner_kind <> 'uxcommand') then
    begin
      if (fr_is_grid = 1) then
      begin
        insert into ux$gritem (ugi_ugr_uuid, ugi_uf_uuid, ugi_position)
        select :grid_ufi_uuid, uf_uuid, null
        from ux$field
        where uf_owner = :us_uuid and
              uf_is_param = 0
        order by uf_position;
      end else
      begin
        insert into ux$fritem (ufi_owner, ufi_parent_uuid, ufi_uf_uuid, ufi_kind)
        select :res_fr_uuid, :tmp_parent_uuid, uf_uuid, 0
        from ux$field
        where uf_owner = :us_uuid and
              uf_is_param = 0 and
              (uf_edi_uuid is null or uf_edi_uuid is distinct from :usr_edi_uuid)
        order by uf_position;
      end
    end

    -- Если грид создан, переносим его на конец
    if (grid_ufi_uuid is not null) then
    begin
      item_count = 0;
      select count(*)
      from ux$fritem
      where ufi_parent_uuid is not distinct from :grid_parent_uuid and
            ufi_uuid <> :grid_ufi_uuid
      into item_count;

      if (item_count > 0) then
      begin
        update ux$fritem
        set ufi_position = :item_count
        where ufi_uuid = :grid_ufi_uuid;
      end
    end
  end

  suspend;
end
