CREATE PROCEDURE HP_KNAKT_AUTO_RES_LO
RETURNS (
    TKF_LO_ID DECIMAL(18,0),
    TKF_LO_NAME VARCHAR(100),
    TKF_BALANCE DECIMAL(18,4),
    TKF_DELINQ_BALANCE DECIMAL(18,4),
    TKF_LOAN_COUNT INTEGER,
    TKF_DELINQ_LOAN INTEGER,
    TKF_FOLDER_COUNT INTEGER,
    TKF_DELINQ_FOLDER INTEGER)
AS
begin
  for select tkf_lo_id, c_name_fio, count(*), sum(iif(tkf_delinq_days > 0, 1, 0)), sum(tkf_balance),
             sum(iif(tkf_delinq_days > 0, tkf_balance, 0.0)), sum(tkf_loan_count),
             sum(iif(tkf_delinq_days > 0, tkf_loan_count, 0.0))
      from temp_knakt_folder, clients
      where tkf_lo_id = c_id
      group by 1, 2
      order by tkf_lo_id
      into :tkf_lo_id, :tkf_lo_name, :tkf_folder_count, :tkf_delinq_folder, :tkf_balance, :tkf_delinq_balance,
           :tkf_loan_count, :tkf_delinq_loan
  do
  begin
    suspend;
  end
end
