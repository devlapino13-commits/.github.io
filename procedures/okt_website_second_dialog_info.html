CREATE PROCEDURE OKT_WEBSITE_SECOND_DIALOG_INFO(
    CHK_ID INTEGER)
RETURNS (
    HTML_RES BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable ck_parent_id integer;
declare variable ck_birth_date date;
declare variable ck_city id_type;
declare variable ck_citizenship integer;
declare variable ck_cred_amount curr_type;
declare variable ck_legal_docs str_100;
declare variable ck_job_status str_100;
declare variable ck_memo varchar(1000);
declare variable ck_secondary_data str_255;
declare variable city_str str_100;
declare variable citizenship_str str_100;
declare variable pass_no_oper str_100;
declare variable cell_no_oper str_100;
declare variable ck_birth_date_str str_100;
begin
  select chk.ck_parent_id
  from cc_check_knclient chk
  where chk.ck_id = :chk_id
  into :ck_parent_id;

  if (ck_parent_id is not null) then
  begin
    select chk.ck_birth_date, chk.ck_city, chk.ck_citizenship, coalesce(chk.ck_cred_amount, 0),
           coalesce(chk.ck_legal_docs, ''), coalesce(iif(chk.ck_job_status = 1, 'да', 'нет'), ''),
           coalesce(chk.ck_memo, ''), coalesce(chk.ck_secondary_data, '')
    from cc_check_knclient chk
    where chk.ck_type = 6 and
          chk.ck_id = :ck_parent_id
    into :ck_birth_date, :ck_city, :ck_citizenship, :ck_cred_amount, :ck_legal_docs, :ck_job_status, :ck_memo,
         :ck_secondary_data;

    city_str = '';

    if (ck_city is not null) then
    begin
      select pl.pl_name
      from places pl
      where pl.pl_id = :ck_city
      into city_str;
    end

    citizenship_str = '';

    if (ck_citizenship is not null) then
    begin
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :ck_citizenship
      into :citizenship_str;

    end

    pass_no_oper = '';
    cell_no_oper = '';

    select pass_no, cell_no
    from cc_secondary_data_parse(:ck_secondary_data)
    into pass_no_oper, cell_no_oper;

    ck_birth_date_str = '';

    if (ck_birth_date is not null) then
    begin
      if (ck_birth_date is not null) then
      begin
        ck_birth_date_str = replace(iif(extract(day from ck_birth_date) < 10, '0', '') || extract(day from ck_birth_date) || '.' || iif(extract(month from ck_birth_date) < 10, '0', '') || extract(month from ck_birth_date) || '.' || extract(year from ck_birth_date), ' ', '');
      end
    end

    html_res = '';

    html_res = BLOB_APPEND(html_res, 'Дата рождения: ' || ck_birth_date_str || ';</br>');
    html_res = BLOB_APPEND(html_res, 'Город........: ' || city_str || ';</br>');
    html_res = BLOB_APPEND(html_res, 'Гражданство..: ' || citizenship_str || ';</br>');
    html_res = BLOB_APPEND(html_res, 'Сумма займа..: ' || ck_cred_amount || ';</br>');
    html_res = BLOB_APPEND(html_res, 'Документы....: ' || ck_legal_docs || ';</br>');
    html_res = BLOB_APPEND(html_res, 'Трудоустроен.: ' || ck_job_status || ';</br>');
    html_res = BLOB_APPEND(html_res, 'Паспорт......: ' || pass_no_oper || ';</br>');
    html_res = BLOB_APPEND(html_res, 'Доп телефон..: ' || cell_no_oper || ';</br>');
    html_res = BLOB_APPEND(html_res, 'Комментарий..: ' || ck_memo || ';');

    suspend;
  end
end
