CREATE PROCEDURE CALLS_DELIQ_KONT_LIST(
    REP_REGION_ID DECIMAL(18,0),
    REP_DATE_END DATE,
    KONT_NUMBER VARCHAR(255),
    KS_STATUS SMALLINT,
    CALL_TYPE SMALLINT,
    DELIQ_TYPE SMALLINT)
RETURNS (
    REG_NAME VARCHAR(255),
    KONT_NO VARCHAR(255),
    K_ID VARCHAR(255),
    KS_STATE SMALLINT)
AS
declare variable deliq_closed_date date;
begin
  /*
        CREATED BY IGUSEV 17.07.2012
        PURPOSE: Для "аналитического отчета по звонкам должникам".     
         Предоставляет список кредитных договоров в соответствии с выставленными фильтрами
    */

  if (kont_number = '') then
    kont_number = null;
  if (ks_status is null) then
    ks_status = 3;
  if (call_type is null) then
    call_type = 0;
  if (:deliq_type is null) then
    deliq_type = 2;

  for select distinct regl.parent_name as reg_name, kont.k_kontract_no as kont_no,
                      cast(kont.k_id as varchar(255)) as k_id, ks_state
      from callcenter_debtors_list cdl
      left join callcenter_debtors_result cdr on cdr.ccdr_from_ccdl_id = cdl.ccdl_id
      inner join region_list(:rep_region_id, 1, -1, 1) regl on regl.id_list = ccdl_kf_branch_id
      inner join kontract kont on k_id = cdl.ccdl_from_k_id
      inner join kontstate ks on k_id = ks_from_k_id
      where ks_ol_is_delete = 0 and
            ((k_kontract_no = :kont_number) or
            (:kont_number is null)) and
            cdl.ccdl_create_date <= :rep_date_end + 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            (((ks.ks_state = :ks_status) and
            (:ks_status <> 3)) or
            (ks.ks_state in (1, 2) and
            (:ks_status = 3))) and
            ((cdr.ccdr_call_datetime < :rep_date_end + 1) or
            (:call_type = 0))
      order by reg_name, k_kontract_no
      into :reg_name, :kont_no, :k_id, :ks_state
  do
  begin
    if (:deliq_type = 2) then
    begin
      suspend;
    end
    else
    begin
      deliq_closed_date = null;

      select cdl.ccdl_deliq_closed_date
      from callcenter_debtors_list cdl
      where cdl.ccdl_from_k_id = :k_id and
            cdl.ccdl_create_date <= :rep_date_end + 1
      order by cdl.ccdl_create_date desc
      rows 1
      into :deliq_closed_date;

      if (:deliq_type = 1) then
      /* погашенный кредиты, на данный момент не имеют просрочки*/
      begin
        if ((deliq_closed_date is not null) and
            (deliq_closed_date <= :rep_date_end)) then
          suspend;
      end

      if (:deliq_type = 0) then
      /* активный кредит*/
      begin
        if ((deliq_closed_date > :rep_date_end) or (deliq_closed_date is null)) then
          suspend;
      end
    end
  end
end
