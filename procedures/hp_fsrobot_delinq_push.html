CREATE PROCEDURE HP_FSROBOT_DELINQ_PUSH
AS
declare variable K_ID D_ID;
declare variable MCL_ID D_ID;
declare variable PUSH_MESSAGE D_STR_LONG;
declare variable KONTRACT_NO STR_100;
begin
    -- RFCKG-4248
    -- Процедура вызывается в FsRobot и отправляет push-уведомление в КМП тем клиентам, у которых сегодня наступает 30-й день просрочки платежа по кредиту

    if (f_get_db_id() <> 41701) then
        exit;

    for
        select
            k.k_id,
            k.k_kontract_no
        from oksummary oks
        inner join kontract k on oks.oks_from_k_id = k.k_id
        where
            current_date between oks.oks_date_begin and oks.oks_date_end
            and oks.oks_close_state = 0
            and current_date - oks.oks_delinq_date = 30
        into
            :k_id,
            :kontract_no

    do
    begin
        for
            select
                mcl.mcl_id
            from knclient kc
            inner join clients c on kc.kc_from_c_id = c.c_id
            inner join mc_login mcl on mcl.mcl_from_c_id = kc.kc_from_c_id
            where
                kc.kc_from_k_id = :k_id
                and kc.kc_amount_approved > 0
                and mcl.mcl_is_delete = 0
            into
                :mcl_id
        do
        begin
            push_message = f_format(
                    'Уважаемый клиент, у вас имеется просрочка в количестве 30 дней по контракту №%1. Просьба срочно оплатить просроченный платеж через доступые платежные системы',
                    kontract_no
            );

            execute procedure mc_send_push('mc_login', :mcl_id, '30-й день просрочки по контракту', :push_message, 30);

        end
    end
end
