CREATE PROCEDURE HP_ADD_WEEKEND_PREVMONTH(
    MONTH_INP D_INTEGER,
    YEAR_INP D_INTEGER,
    REG_ID D_INTEGER,
    C_ID D_ID = null)
RETURNS (
    RES_TEXT D_STR_2048)
AS
declare variable c1 d_str_10;
declare variable c2 d_str_10;
declare variable c3 d_str_10;
declare variable dt_beg d_date;
declare variable dt_beg_prev d_date;
declare variable month_prev d_integer;
declare variable year_prev d_integer;
declare variable weekend_prev d_str_100;
declare variable year_month d_integer;
declare variable sql d_str_8100;
begin
  c1 = 'Р';
  c2 = 'Вых';
  c3 = ascii_char(13)||ascii_char(10);
  res_text = '';

  dt_beg = f_format('01.%1.%2', lpad(month_inp, 2, '0'), year_inp);
  dt_beg_prev = dateadd(month, -1, dt_beg);
  month_prev = extract(month from dt_beg_prev);
  year_prev = extract(year from dt_beg_prev);
  weekend_prev = (select iif(weekend_days='', '', '.') || weekend_days
    from hp_timesheet_pre_set(:reg_id, :year_prev, :month_prev));

  year_month = 100 * year_inp + month_inp;

  for
    select
      a.c_id,
      a.fio,

      a.sat_day_1 a1, a.sat_day_2 a2, a.sat_day_3 a3, a.sat_day_4 a4, a.sat_day_5 a5, a.sat_day_6 a6, a.sat_day_7 a7,
      a.sat_day_8 a8, a.sat_day_9 a9, a.sat_day_10 a10, a.sat_day_11 a11, a.sat_day_12 a12, a.sat_day_13 a13, a.sat_day_14 a14,
      a.sat_day_15 a15, a.sat_day_16 a16, a.sat_day_17 a17, a.sat_day_18 a18, a.sat_day_19 a19, a.sat_day_20 a20, a.sat_day_21 a21,
      a.sat_day_22 a22, a.sat_day_23 a23, a.sat_day_24 a24, a.sat_day_25 a25, a.sat_day_26 a26, a.sat_day_27 a27, a.sat_day_28 a28,
      a.sat_day_29 a29, a.sat_day_30 a30, a.sat_day_31 a31,

      b.sat_day_1 b1, b.sat_day_2 b2, b.sat_day_3 b3, b.sat_day_4 b4, b.sat_day_5 b5, b.sat_day_6 b6, b.sat_day_7 b7,
      b.sat_day_8 b8, b.sat_day_9 b9, b.sat_day_10 b10, b.sat_day_11 b11, b.sat_day_12 b12, b.sat_day_13 b13, b.sat_day_14 b14,
      b.sat_day_15 b15, b.sat_day_16 b16, b.sat_day_17 b17, b.sat_day_18 b18, b.sat_day_19 b19, b.sat_day_20 b20, b.sat_day_21 b21,
      b.sat_day_22 b22, b.sat_day_23 b23, b.sat_day_24 b24, b.sat_day_25 b25, b.sat_day_26 b26, b.sat_day_27 b27, b.sat_day_28 b28,
      b.sat_day_29 b29, b.sat_day_30 b30, b.sat_day_31 b31
    from hp_timesheet_list (:reg_id, :month_inp, :year_inp) as a
      inner join hp_timesheet_list (:reg_id, :month_prev, :year_prev) as b
        on b.c_id = a.c_id
    where a.c_id = coalesce(:c_id, a.c_id)
    as cursor c
  do
  begin
    sql = '';
    if (c.a1 = c1 and c.b1 = c2 and '.1.' not containing weekend_prev)
      then sql = sql || 'sat_day_1=0,';
    if (c.a2 = c1 and c.b2 = c2 and '.2.' not containing weekend_prev)
      then sql = sql || 'sat_day_2=0,';
    if (c.a3 = c1 and c.b3 = c2 and '.3.' not containing weekend_prev)
      then sql = sql || 'sat_day_3=0,';
    if (c.a4 = c1 and c.b4 = c2 and '.4.' not containing weekend_prev)
      then sql = sql || 'sat_day_4=0,';
    if (c.a5 = c1 and c.b5 = c2 and '.5.' not containing weekend_prev)
      then sql = sql || 'sat_day_5=0,';
    if (c.a6 = c1 and c.b6 = c2 and '.6.' not containing weekend_prev)
      then sql = sql || 'sat_day_6=0,';
    if (c.a7 = c1 and c.b7 = c2 and '.7.' not containing weekend_prev)
      then sql = sql || 'sat_day_7=0,';
    if (c.a8 = c1 and c.b8 = c2 and '.8.' not containing weekend_prev)
      then sql = sql || 'sat_day_8=0,';
    if (c.a9 = c1 and c.b9 = c2 and '.9.' not containing weekend_prev)
      then sql = sql || 'sat_day_9=0,';
    if (c.a10 = c1 and c.b10 = c2 and '.10.' not containing weekend_prev)
      then sql = sql || 'sat_day_10=0,';
    if (c.a11 = c1 and c.b11 = c2 and '.11.' not containing weekend_prev)
      then sql = sql || 'sat_day_11=0,';
    if (c.a12 = c1 and c.b12 = c2 and '.12.' not containing weekend_prev)
      then sql = sql || 'sat_day_12=0,';
    if (c.a13 = c1 and c.b13 = c2 and '.13.' not containing weekend_prev)
      then sql = sql || 'sat_day_13=0,';
    if (c.a14 = c1 and c.b14 = c2 and '.14.' not containing weekend_prev)
      then sql = sql || 'sat_day_14=0,';
    if (c.a15 = c1 and c.b15 = c2 and '.15.' not containing weekend_prev)
      then sql = sql || 'sat_day_15=0,';
    if (c.a16 = c1 and c.b16 = c2 and '.16.' not containing weekend_prev)
      then sql = sql || 'sat_day_16=0,';
    if (c.a17 = c1 and c.b17 = c2 and '.17.' not containing weekend_prev)
      then sql = sql || 'sat_day_17=0,';
    if (c.a18 = c1 and c.b18 = c2 and '.18.' not containing weekend_prev)
      then sql = sql || 'sat_day_18=0,';
    if (c.a19 = c1 and c.b19 = c2 and '.19.' not containing weekend_prev)
      then sql = sql || 'sat_day_19=0,';
    if (c.a20 = c1 and c.b20 = c2 and '.20.' not containing weekend_prev)
      then sql = sql || 'sat_day_20=0,';
    if (c.a21 = c1 and c.b21 = c2 and '.21.' not containing weekend_prev)
      then sql = sql || 'sat_day_21=0,';
    if (c.a22 = c1 and c.b22 = c2 and '.22.' not containing weekend_prev)
      then sql = sql || 'sat_day_22=0,';
    if (c.a23 = c1 and c.b23 = c2 and '.23.' not containing weekend_prev)
      then sql = sql || 'sat_day_23=0,';
    if (c.a24 = c1 and c.b24 = c2 and '.24.' not containing weekend_prev)
      then sql = sql || 'sat_day_24=0,';
    if (c.a25 = c1 and c.b25 = c2 and '.25.' not containing weekend_prev)
      then sql = sql || 'sat_day_25=0,';
    if (c.a26 = c1 and c.b26 = c2 and '.26.' not containing weekend_prev)
      then sql = sql || 'sat_day_26=0,';
    if (c.a27 = c1 and c.b27 = c2 and '.27.' not containing weekend_prev)
      then sql = sql || 'sat_day_27=0,';
    if (c.a28 = c1 and c.b28 = c2 and '.28.' not containing weekend_prev)
      then sql = sql || 'sat_day_28=0,';
    if (c.a29 = c1 and c.b29 = c2 and '.29.' not containing weekend_prev)
      then sql = sql || 'sat_day_29=0,';
    if (c.a30 = c1 and c.b30 = c2 and '.30.' not containing weekend_prev)
      then sql = sql || 'sat_day_30=0,';
    if (c.a31 = c1 and c.b31 = c2 and '.31.' not containing weekend_prev)
      then sql = sql || 'sat_day_31=0,';
    if (sql <> '') then
    begin
      sql = f_format(
        'update sa_tabelo set
          %1
        where sat_year_month = %2
          and sat_from_c_id = %3
          and sat_from_sawt_id = %4
          and sat_office_id = %5',  left(sql, char_length(sql)-1), year_month, c.c_id, 0, reg_id);
      execute statement sql;
      execute procedure hp_tabel_calc_days_hours(:year_month, c.c_id, :reg_id);
    end else
      res_text = res_text || '  ' || c.fio || c3;
  end
  if (res_text <> '') then
    res_text = 'Не удалось проставить выходные дни: ' || c3 || res_text;
  suspend;
end
