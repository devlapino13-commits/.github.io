CREATE PROCEDURE ACCBOOK_LIST(
    IS_DT SMALLINT)
RETURNS (
    NUM INTEGER,
    DT VARCHAR(2048),
    KT VARCHAR(2048))
AS
declare variable d_b varchar(20);
declare variable d_e varchar(20);
declare variable k_b varchar(20);
declare variable k_e varchar(20);
declare variable b_prior varchar(20);
declare variable e_prior varchar(20);
begin
  /* Procedure Text */
  b_prior = null;
  e_prior = null;
  num = 0;
  dt = '';
  kt = '';
  if (is_dt = 1) then
  begin
    for select ab_dt_begin, ab_dt_end, ab_kt_begin, ab_kt_end
        from accbook
        where ab_is_delete = 0
        order by 1, 2, 3, 4
        into :d_b, :d_e, :k_b, :k_e
    do
    begin
      if (((b_prior <> d_b) or (e_prior <> d_e)) and
          (b_prior is not null)) then
      begin
        if ((dt <> '') or (kt <> '')) then
        begin
          num = num + 1;
          suspend;
        end

        kt = '';
      end

      if (d_b = d_e) then
      begin
        dt = d_b;
      end
      else
      begin
        dt = d_b || '-' || d_e;
      end

      if (k_b = k_e) then
      begin
        if (kt <> '') then
          kt = kt || ', ' || k_b;
        else
          kt = k_b;
      end
      else
      begin
        if (kt <> '') then
          kt = kt || ', ' || k_b || '-' || k_e;
        else
          kt = k_b || '-' || k_e;
      end
      b_prior = d_b;
      e_prior = d_e;
    end
  end
  else
  begin

    for select ab_dt_begin, ab_dt_end, ab_kt_begin, ab_kt_end
        from accbook
        where ab_is_delete = 0
        order by 3, 4, 1, 2
        into :d_b, :d_e, :k_b, :k_e
    do
    begin
      if (((b_prior <> k_b) or (e_prior <> k_e)) and
          (b_prior is not null)) then
      begin
        if ((dt <> '') or (kt <> '')) then
        begin
          num = num + 1;
          suspend;
        end

        dt = '';
      end

      if (k_b = k_e) then
      begin
        kt = k_b;
      end
      else
      begin
        kt = k_b || '-' || k_e;
      end

      if (d_b = d_e) then
      begin
        if (dt <> '') then
          dt = dt || ', ' || d_b;
        else
          dt = d_b;
      end
      else
      begin
        if (dt <> '') then
          dt = dt || ', ' || d_b || '-' || d_e;
        else
          dt = d_b || '-' || d_e;
      end
      b_prior = k_b;
      e_prior = k_e;
    end

  end

  if ((dt <> '') or (kt <> '')) then
  begin
    num = num + 1;
    suspend;
  end

end
