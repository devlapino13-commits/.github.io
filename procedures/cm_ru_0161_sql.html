CREATE PROCEDURE CM_RU_0161_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id id_type;
declare variable pr_code varchar(100);
declare variable c_curr_status integer;
declare variable res_amount d_currency;
declare variable k_refund integer;
declare variable k_add_month d_smallint;
begin
  for select k_disb_date, pr_code, k_amount_approved, k_refund, k_add_month
      from kontract kont, product pr
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            pr.pr_code not containing '.food.' and
            pr.pr_code not containing '.summer_plus.'and
            pr.pr_code not containing '.comfortable.' and
            pr.pr_code not containing '.refinance.' and
            pr.pr_code not containing '.refinance_oivo.' and
            pr.pr_code not containing '.refinance_plus.' and
            pr.pr_code not containing '.b_refinance.'
      into kf_disb_date, pr_code, :kc_amount, :k_refund, :k_add_month
  do
  begin
    -- < добавлено блок по рефинансированию, задача на jiro 1944 14.05.20
    if (k_refund = 1) then
    begin
      res_amount = 0;
      select sum(cast(kk.res_amount as d_currency))
      from hp_kontrefund_k_list(:k_id) kk, kontract k, product pr
      where k.k_id = kk.res_k_id and
            k.k_from_pr_id = pr.pr_id and
            pr.pr_code not like '%.food.%' and
            pr.pr_code not like '%.par.%'
      into :res_amount;
      res_amount = coalesce(res_amount, 0);
      if (kc_amount < res_amount and
          not(k_refund = 1 and kf_disb_date >= '01.03.2022' and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%')) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Указанная сумма для рефинансирования не может быть меньше размера остатка суммы активного кредита';
        suspend;
      end -->
    end
    if (pr_code containing '.status.') then
    begin
      select sum(cast(res_amount as d_currency))
      from hp_kontrefund_k_list(:k_id)
      into :res_amount;
      res_amount = coalesce(res_amount, 0);
      if (res_amount > 0 and
          kc_amount <> res_amount) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Сумма кредита, не должна быть больше или меньше суммы остатка активного кредита';
        suspend;
      end
    end
    if (k_add_month > 0) then
    begin
      if (pr_code not containing '.blago.' and
          pr_code not containing '.uspeh.' and
          pr_code not containing '.shans.' and
          pr_code not containing '.status.' and
          pr_code not containing '.support.' and
          not(k_refund = 1 and kf_disb_date >= '01.03.2022' and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%')
          ) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Если к первому погашению добавлены месяцы, продукт должен быть «Благо»,«Успех», «Шанс», «Статус».';
        suspend;
      end
      if (pr_code containing '.status.') then
      begin
        for select kc_from_c_id, kc_amount_approved
            from knclient
            where kc_from_k_id = :k_id and
                  kc_amount_approved > 0
            into c_id, kc_amount
        do
        begin
          -- проверяю статус клиента
          c_curr_status = 0;
          select c_curr_status
          from hp_get_client_status_str(:c_id, :kf_disb_date)
          into c_curr_status;
          c_curr_status = coalesce(c_curr_status, 0);
          -- если у клиента нет активного займа
          if (c_curr_status < 1 or (c_curr_status > 0 and
              exists(select *
                     from kontstate ks
                     inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
                     where ks.ks_state = 1 and
                           ks.ks_ol_is_delete = 0 and
                           :kf_disb_date between ks.ks_date_begin and ks.ks_date_end and
                           kncl.kc_amount_approved > 0 and
                           kncl.kc_from_c_id = :c_id))) then
          begin
            res_error_is_critical = 1;
            res_error_text = 'Только статусные клиенты могут получить займ если к первому погашению добавлены месяцы и нет активного займа';
            suspend;
          end
        end
      end
    end
  end
end
