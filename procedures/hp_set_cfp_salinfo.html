CREATE PROCEDURE HP_SET_CFP_SALINFO
AS
declare variable bls_id d_id;
declare variable c_id d_id;
declare variable arr_length d_integer;
declare variable json_blob d_blob_text;
declare variable counter d_integer;
declare variable current_index d_integer;
declare variable result d_currency;
declare variable current_value d_currency;
declare variable ca_id d_integer;
declare variable cq_id d_integer;
declare variable mac_id d_id;
declare variable res_text d_str_255;
declare variable eil_id d_id;
declare variable kpd_err d_str_255;
declare variable perc d_currency;
declare variable cp_full_name_ru d_str_255;
declare variable cp_id d_integer;
declare variable prev_value d_currency;
begin
  -- Процедура должна сохранить среднюю значению заработной платы за последний 6 месяцев в анкету анализ доходов / расходов. (RFCKG-3240)
  prev_value = 0;
  res_text = '';
  counter = 6;
  result = 0;
  select cq.cq_id
  from cfp_questions cq
  where cq.cq_code_name = 'salary_certificate' and
        cq.cq_is_delete = 0
  into cq_id;

  for select eil.eil_id, eil.eil_from_bls_id, eil.eil_from_c_id, ca.ca_id, ca.ca_object_id, ca.ca_from_cp_id
      from extinfolist eil
      inner join cfp_anketa ca on ca.ca_from_c_id = eil.eil_from_c_id and
            ca.ca_date = current_date
      inner join cfp_anketa_kpd cak on cak.cak_from_ca_id = ca.ca_id and
            coalesce(cak.cak_from_eil_id, 0) = 0
      where eil.eil_from_eic_id = 10 and
            eil.eil_status = 2 and
            eil.eil_date = current_date
      into eil_id, bls_id, c_id, ca_id, mac_id, cp_id
  do
  begin

    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into :json_blob;

    -- Получаем длину массива
    arr_length = f_json_blob_get_count(json_blob, 'WorkPeriods.WorkPeriodWithSumDto');

    if (arr_length <= 6) then
      counter = arr_length;

    --Считаем  Salary
    while (counter > 0) do
    begin
      current_index = arr_length - counter;
      current_value = f_json_blob_get_value(json_blob, 'WorkPeriods.WorkPeriodWithSumDto[' || current_index || '].Salary', 0);
      result = result + current_value;
      counter = counter - 1;
    end
    select cp.cp_full_name_ru
    from cfp_positions cp
    where cp.cp_id = :cp_id
    into cp_full_name_ru;

    if (cp_full_name_ru containing 'Правоохранительные органы') then
      perc = 0;
    else
      perc = 0.2;

    result = result - (result * :perc);
    if (arr_length <= 6) then
      result = result / arr_length;
    else
      result = result / 6;

    if (coalesce(result, 0) = 0) then
    begin
      update cfp_anketa_kpd cak
      set cak.cak_from_eil_id = :eil_id
      where cak.cak_from_ca_id = :ca_id;

      update extinfolist eil
      set eil.eil_from_un_id = 1000000000000
      where eil.eil_id = :eil_id;
    end
    else
    begin
      select caa.caa_value
      from cfp_anketa_answers caa
      where caa.caa_from_ca_id = :ca_id and
            caa.caa_from_cq_id = :cq_id
      into prev_value;
      prev_value = coalesce(prev_value, 0);
      -- Обновляем данные
      update cfp_anketa_answers caa
      set caa.caa_value = :result
      where caa.caa_from_ca_id = :ca_id and
            caa.caa_from_cq_id = :cq_id;

      update cfp_anketa ca
      set ca.ca_avr_income = :result
      where ca.ca_id = :ca_id;

      select kpd_err
      from save_cfp_kpd(:mac_id)
      into :kpd_err;

      if (coalesce(:kpd_err, '') <> '') then
      begin
        update cfp_anketa_answers caa
        set caa.caa_value = :prev_value
        where caa.caa_from_ca_id = :ca_id and
              caa.caa_from_cq_id = :cq_id;

        update cfp_anketa ca
        set ca.ca_avr_income = :prev_value
        where ca.ca_id = :ca_id;

        update extinfolist eil
        set eil.eil_from_un_id = 1000000000000
        where eil.eil_id = :eil_id;
      end

      update cfp_anketa_kpd cak
      set cak.cak_from_eil_id = :eil_id
      where cak.cak_from_ca_id = :ca_id;
    end
  end
end
