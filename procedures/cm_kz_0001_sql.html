CREATE PROCEDURE CM_KZ_0001_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable MILITARY_STATUS D_SMALLINT;
declare variable QUERY_DATE D_DATE;
declare variable MCQ_RECEIVED D_SMALLINT;
begin
  -- RFCKG-4079 Проверка статуса воинской службы
  for select mac.mac_from_c_id
      from miniapp ma
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
      where ma.ma_id = :ma_id and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999' and
            mac.mac_amount > 0
      into c_id
  do
  begin
    select status,
           query_date,
           mcq_received
    from hp_military_query_status(:c_id)
    into military_status,
         query_date,
         mcq_received;

    -- Если успешного запроса нет (запрос не выполнялся или завершился ошибкой)
    if (mcq_received = 0 or
        (military_status in (-2, -3))) then
      res_error_text = 'Ответ от сервиса Призывники еще не получен.'; -- RFCKZ-28

    if (military_status = 1) then
      res_error_text = 'Клиент призван на срочную воинскую службу';

    if (coalesce(res_error_text, '') <> '') then
    begin
      -- RFCKZ-877
      res_error_is_critical = 2;
      suspend;
    end
  end
end
