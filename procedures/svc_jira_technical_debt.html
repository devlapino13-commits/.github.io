CREATE PROCEDURE SVC_JIRA_TECHNICAL_DEBT(
    IGNORE_CLARIFICATION D_BOOL,
    ADD_PROJ_WITH_TO D_BOOL = 0,
    ADD_PROJ_WITHOUT_TO D_BOOL = 0)
RETURNS (
    JRI_ISSUE_ID D_INTEGER,
    JRI_PARENT_ID D_INTEGER,
    JRI_DEV_ID D_ID,
    JRI_TYPE_ID D_INTEGER,
    JRI_JIRA_CODE D_STR_CODE,
    JRI_TITLE D_STR_LONG,
    JRI_PROJECT D_STR_CODE,
    JRI_LEVEL D_SMALLINT,
    JRI_PRIORITY D_SMALLINT,
    JRI_KPI_TIMEZONE D_SMALLINT,
    JRI_ESTIMATE_SECS D_ID,
    JRI_ESTIMATE_PERIOD D_STR_CODE,
    JRI_WORKED_SECS D_ID,
    JRI_CREATED D_DATETIME,
    JRI_FINISHED D_DATETIME,
    JRI_CLOSED D_DATETIME,
    JRI_STATE_ID D_SMALLINT,
    JRI_STATE_DATETIME D_DATETIME,
    JRI_RECALC D_SMALLINT,
    JRI_KPI_PERIOD D_INTEGER,
    JRI_CHECK_STATUS D_SMALLINT,
    JRI_CHECK_MEMO D_STR_255,
    JRI_CATEGORY_ID D_INTEGER,
    JRI_SOFTWARE_ID D_INTEGER,
    JRI_JOB_KIND D_SMALLINT,
    JRD_LOGIN D_STR_CODE,
    JRD_FIO D_STR_100,
    JRT_TEXT D_STR_CODE,
    TASK_LAST_STATE D_STR_CODE,
    TASK_STACK D_STR_CODE,
    TASK_VIEW D_STR_CODE)
AS
declare variable jpp_has_to d_integer;
begin
  for select jri_issue_id, jri_parent_id, jri_dev_id, jri_type_id, jri_jira_code, jri_title, jri_project, jri_level,
             jri_priority, jri_kpi_timezone, jri_estimate_secs, jri_estimate_period, jri_worked_secs, jri_created,
             jri_finished, jri_closed, jri_state_id, jri_state_datetime, jri_recalc, jri_kpi_period, jri_check_status,
             jri_check_memo, jri_category_id, jri_software_id, jri_job_kind, jd.jrd_login, jd.jrd_fio, jt.jrt_text,
             js.jrs_text
      from jira_issues ji
      left join jira_dev jd on jd.jrd_id = ji.jri_dev_id
      left join jira_types jt on jt.jrt_id = ji.jri_type_id
      left join jira_states js on js.jrs_id = ji.jri_state_id
      where jt.jrt_id in (1, 2, 5, 7, 8, 9)
         and js.jrs_result >= 0
      into :jri_issue_id, :jri_parent_id, :jri_dev_id, :jri_type_id, :jri_jira_code, :jri_title, :jri_project, :jri_level,
           :jri_priority, :jri_kpi_timezone, :jri_estimate_secs, :jri_estimate_period, :jri_worked_secs, :jri_created,
           :jri_finished, :jri_closed, :jri_state_id, :jri_state_datetime, :jri_recalc, :jri_kpi_period,
           :jri_check_status, :jri_check_memo, :jri_category_id, :jri_software_id, :jri_job_kind, :jrd_login, :jrd_fio, :jrt_text,
           :task_last_state
  do
  begin
    if ((select is_igrore from jira_ignore_params(1, :ignore_clarification, :jri_jira_code, :task_last_state, :jrd_fio)) = 1) then
        continue;

    select task_stack, task_level
    from jira_update_params(:jrd_fio, :jrt_text, :jri_level, :jri_issue_id)
    into :task_stack, :jri_level;

    task_view = 'created';

    suspend;
  end

  for select jpp_has_to, jri_issue_id, jri_parent_id, jri_dev_id, jri_type_id, jri_jira_code, jri_title, jri_project,
             jri_level, jri_priority, jri_kpi_timezone, jri_estimate_secs, jri_estimate_period, jri_worked_secs,
             jri_created, jri_finished, jri_closed, jri_state_id, jri_state_datetime, jri_recalc, jri_kpi_period,
             jri_check_status, jri_check_memo, jri_category_id, jri_software_id, jri_job_kind, jrd_login, jrd_fio,
             jrt_text, task_last_state, task_stack
        from svc_jira_pre_estimate_debt(:add_proj_with_to, :add_proj_without_to)
        into :jpp_has_to, :jri_issue_id, :jri_parent_id, :jri_dev_id, :jri_type_id, :jri_jira_code, :jri_title,
             :jri_project, :jri_level, :jri_priority, :jri_kpi_timezone, :jri_estimate_secs, :jri_estimate_period,
             :jri_worked_secs, :jri_created, :jri_finished, :jri_closed, :jri_state_id, :jri_state_datetime, :jri_recalc,
             :jri_kpi_period, :jri_check_status, :jri_check_memo, :jri_category_id, :jri_software_id, :jri_job_kind,
             :jrd_login, :jrd_fio, :jrt_text, :task_last_state, :task_stack
   do
   begin
        task_view = 'proj_without_to';
        if (jpp_has_to = 1) then
            task_view = 'proj_with_to';

        suspend;
   end
end
