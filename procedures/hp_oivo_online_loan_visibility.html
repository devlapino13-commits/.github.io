CREATE PROCEDURE HP_OIVO_ONLINE_LOAN_VISIBILITY(
    MCL_ID D_ID)
RETURNS (
    IS_VISIBLE D_BOOL)
AS
declare variable c_id d_id;
declare variable c_birth_date d_date;
declare variable c_name_fio d_str_100;
declare variable c_pass_no d_str_code;
declare variable c_inn d_str_20;
declare variable rep_date d_date;
begin
    rep_date = current_date;
    is_visible = 1;

    select mcl.mcl_from_c_id, c.c_name_fio, c.c_birth_date, c.c_pass_no, c.c_inn
    from mc_login mcl
    inner join clients c on mcl.mcl_from_c_id = c.c_id
    where mcl.mcl_id = :mcl_id
    into :c_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn;

    if (coalesce(c_id, 0) = 0) then
    begin
        is_visible = 0;
        suspend;
        exit;
    end

    -- Отсутствие просрочек по активным займам
    if (is_visible = 1 and
        exists(select *
               from knclient kc
               inner join oksummary oks on kc.kc_from_k_id = oks.oks_from_k_id
               where kc.kc_from_c_id = :c_id and
                     kc.kc_amount_approved > 0 and
                     :rep_date between oks.oks_date_begin and oks.oks_date_end and
                     oks.oks_close_state = 0 and
                     oks.oks_delinq_date <= :rep_date
    
        )) then
        is_visible = 0; 

    -- Актуальный верифицированный паспорт
    if (is_visible = 1 and
        not exists(select *
                   from clientdocs cds
                   where cds.cds_from_c_id = :c_id and
                         :rep_date between cds.cds_doc_date_begin and cds.cds_doc_date_end and
                         cds.cds_verify_status = 2 and
                         cds.cds_is_delete = 0 and
                         cds.cds_is_main_doc = 1)
    ) then
        is_visible = 0; 

    -- Возраст заемщика от 18 до 65 лет
    if (is_visible = 1 and
        f_age(c_birth_date, current_date) not between 18 and 65
    ) then
        is_visible = 0;

    -- Отсутствие активной заявки
    if (is_visible = 1 and
        exists(select *
               from miniapp
               inner join miniappclient on mac_from_ma_id = ma_id
               left join kontract on k_from_ma_app_id = ma_id
               where ma_date_end = '31.12.9999' and
                     iif(k_id is not null, ma_is_active = 1 and k_status = 0, ma_is_active = 1) and
                     ma_date_app between :rep_date - 30 and :rep_date and
                     mac_from_c_id = :c_id)
    ) then
        is_visible = 0; 

    -- Отсутствие клиента в отказниках или стоп-листе
    if (is_visible = 1 and
        exists(select 1
               from hp_blacklist_search(:c_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn, f_cont(:c_id, 'cell', 'ok', 'main'), 1, 1, 0)
               where c_comment_id = 1)
    ) then
        is_visible = 0; 

    -- Отсутствие заявки с отказом со стороны клиента менее месяца назад
    if (is_visible = 1 and
        exists(select *
               from miniappclient mac
               inner join miniapp ma on mac.mac_from_ma_id = ma.ma_id
               inner join kontreslist krl on krl.krl_ma_id = mac.mac_from_ma_id
               where mac.mac_from_c_id = :c_id and
                     mac.mac_amount > 0 and
                     -- Берем только заявки клиента за последние 40 дней. По условию сказано, что нужно проверять на наличие
                     -- отказа по иниц. клиента за последний месяц, но заявки выбираем с запасом по дням для случая, если отказ
                     -- был не в тот же день, что была создана заявка
                     ma.ma_date_app >= :rep_date - 40 and
                     krl.krl_from_kr_id = 10 and -- Отказ по инициативе клиента
                     krl.krl_date_old >= dateadd(month, -1, :rep_date)
    
        )
    ) then
        is_visible = 0; 

    -- Отсутствие активного займа, не предполагающего наличие параллельного займа
    -- ???????

    suspend;
end
