CREATE PROCEDURE HR_PERSONAL_TABEL_STAT_TMP(
    REP_REGION_ID INTEGER,
    REP_YEAR_MONTH INTEGER)
RETURNS (
    PS_FROM_REG_ID INTEGER,
    PS_REG_NAME VARCHAR(100),
    PS_TABEL_STATUS VARCHAR(100),
    C_STAFF_COUNT INTEGER,
    C_TABEL_NEED INTEGER,
    C_TABEL_TOTAL INTEGER,
    C_TABEL_APPROVED INTEGER,
    C_TABEL_NAME VARCHAR(100),
    SAWT_ID INTEGER)
AS
declare variable dd1 date;
declare variable dd2 date;
declare variable ps_from_c_id decimal(18,0);
declare variable suspended integer;
begin
-- Список сотрудников для отображения в табели
  dd1 = '01.' || right(:rep_year_month, 2) || '.' || left(:rep_year_month, 4);
  dd2 = dateadd(month, 1, dd1);
  for select ps_from_reg_id, ps_from_c_id, ps_reg_name, ps_tabel_status, c_need_tabel
      from hr_all_personal_new_pstatus(:rep_region_id, :dd1, :dd2)
      into :ps_from_reg_id, :ps_from_c_id, :ps_reg_name, :ps_tabel_status, :c_tabel_need
  do
  begin
    c_staff_count = 1;
    suspended = 0;
    c_tabel_total = 1;
    for select sawt_id, sawt_name, sat_is_ready
        from sa_tabel, sa_worktype
        where sat_year_month = :rep_year_month and
              sat_from_c_id = :ps_from_c_id and
              sat_from_sawt_id = sawt_id
        order by sawt_id
        into :sawt_id, :c_tabel_name, :c_tabel_approved
    do
    begin
      suspend;
      c_staff_count = 0;
      suspended = 1;
    end
    if (suspended = 0) then
    begin
      sawt_id = 999;
      c_tabel_name = 'Нет табели';
      c_tabel_total = 0;
      c_tabel_approved = 0;
      suspend;
    end
  end
end
