CREATE PROCEDURE CRM_MINIAPP_HISTORY(
    PHONE_NO STR_255,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ID_STATUS STR_100,
    ID_SVZ ID_TYPE,
    OPER_DATE_TIME D_DATETIME,
    ID_REFUSAL STR_255)
AS
declare variable ma_id id_type;
declare variable ma_number d_smallint;
declare variable k_id id_type;
declare variable krl_id id_type;
begin

    for select mac_from_ma_id, ma_number, max(ma_date_save)
    from miniapp
    inner join miniappclient on mac_from_ma_id = ma_id
                             and ma_date_end = '31.12.9999'
    inner join continfo on cci_object_id = mac_from_c_id
                        and (ok_cell_main = :phone_no
                        or last_cell_main = :phone_no)
    where mac_amount > 0
    and ma_date_app between :rep_date_begin and :rep_date_end
    group by 1,2
    into ma_id, ma_number, oper_date_time
    do
    begin
        for select (case
                    when (ma.ma_is_active <= 0 or k.k_status <= -1) then 'Отказан'
                    when ma.ma_is_active = 1 and k.k_status = 0 then 'В процессе'
                    when ma.ma_is_active = 1 and k.k_status = 1 then 'Выдан'
                    end) as status,
                    iif(un.un_from_c_id <> k.k_fact_disb_lo_id, coalesce(k.k_fact_disb_lo_id, un.un_from_c_id), coalesce(k.k_fact_disb_lo_id, un.un_from_c_id)),
                    k.k_id
        from miniapp ma
        left join kontract k on k.k_from_ma_app_id = ma.ma_id
        inner join username un on un.un_id = ma.ma_from_un_id
        where ma.ma_id = :ma_id
        and ma.ma_number = :ma_number
        and ma.ma_date_app between :rep_date_begin and :rep_date_end
        group by 1,2,3
        into id_status, id_svz, k_id
        do
        begin
            id_status = trim(id_status);
            if(id_status = 'Отказан') then
            begin
    
                for select max(krl_id)
                from kontreslist
                where krl_from_k_id = :k_id
                or krl_ma_id = :ma_id
                into krl_id
                do
                begin
                    select kr_text
                    from kontres
                    inner join kontreslist krl on krl_from_kr_id = kr_id
                    where krl.krl_id = :krl_id
                    into id_refusal;
                end
            end
            if (id_status = 'Выдан') then
            begin
              select ol_datetime
              from operlist
              inner join operkred on ok_from_ol_id = ol_id
              where ok_from_k_id = :k_id 
              and is_delete = 0
              and ok_type = 8
              into oper_date_time;
            end
           suspend;
           id_refusal = null;
        end
    end

    for select (case ema_status
                when -1 then 'Отказ'
                when 0 then 'В ожидании'
                when 1 then 'Обработан'
               end),
               ema_lo_id,
               coalesce(ema_status_datetime, ema_datetime),
               rd_name
    from extminiapp
    inner join refdata on rd_id = ema_result_rd_id
    inner join continfo on cci_object_id = ema_from_c_id
                        and (ok_cell_main = :phone_no
                        or last_cell_main = :phone_no)
    where ema_date between :rep_date_begin and :rep_date_end
    into id_status ,id_svz, oper_date_time, id_refusal
    do
    begin
        id_status = trim(id_status);
        suspend;
    end
end
