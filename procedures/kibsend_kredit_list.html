CREATE PROCEDURE KIBSEND_KREDIT_LIST
RETURNS (
    POCKET_ID INTEGER,
    UNIQ_CRED_NUM VARCHAR(20),
    BANK_CODE INTEGER,
    FIL_CODE INTEGER,
    NUMBER VARCHAR(20),
    DATEAWD DATE,
    DATESTOP DATE,
    CURR_CODE VARCHAR(3),
    CREDTYPE_CODE VARCHAR(20),
    CREDSTATEX_CODE VARCHAR(20),
    CRED_AMOUNT DECIMAL(18,4),
    K_ID DECIMAL(18,0),
    REPAYSTATUS INTEGER,
    GUARANTEE VARCHAR(100),
    COUNTOFDELINQUENCY INTEGER,
    UNPAIDAMOUNT DECIMAL(18,4),
    UNPAIDPERCENTAMOUNT DECIMAL(18,4),
    UNPAIDFINE DECIMAL(18,4),
    PAYMETHOD SMALLINT,
    FACT_AMOUNT DECIMAL(18,4),
    PLAN_AMOUNT DECIMAL(18,4),
    DELIQ_DAYS INTEGER,
    PAY_FREQUENCY SMALLINT,
    OVERDUEAMOUNT DECIMAL(18,4),
    PERCENTOVERDUEAMOUNT DECIMAL(18,4),
    PAYOUTCOUNT INTEGER,
    PAYOUTDAY SMALLINT,
    FACTREPAYMENTDATE DATE,
    OVERDUEPAYOUTNUMBER INTEGER,
    WRITEOFFOS DECIMAL(18,4),
    WRITEOFFPERC DECIMAL(18,4))
AS
declare variable dd date;
declare variable is_urgent integer;
declare variable payoutcount_tmp integer;
begin
  /*
    CHANGED BY IGUSEV 14.06.2012
    changes:   Была добавлена выборка дынных по полю kkr_paymethod
    (1 аннуитет
    2 дифференцированный/стандарт
    3 индивидуальный/гибкий  график 
    4 другое)

    CHANGED BY IGUSEV 27.07.2012
    changes: Была добавлена выборка дынных по следющим полям
             coalesce(KKR_FACT_AMOUNT,0),  - фактическая выплаченная группой сумма по данному погашению
             coalesce(KKR_PLAN_AMOUNT,0),  паланируема сумма - сумма, которую клиент должен платить каждый месяц
             coalesce(KKR_DELIQ_DAYS,0),   число дней просрочки
             coalesce(KKR_PAY_FREQUENCY,0), частота платежа. На данный момент для всех "Ежемесячно"
             kibkr.kkr_overdueamount,
             kibkr.kkr_percentoverdueamount,
             k_kolich_repay,     общее кол-во выплат. сколько раз должна группа заплатить*
             kibkr.kkr_payoutday
  */

  if (is_urgent is null) then
    is_urgent = 0;
  for select kkr_from_kp_id, kkr_uniq_cred_num, kkr_bank_code, kkr_fil_code, k_kontract_no, k_applicate_date,
             k_disb_date, k_kredline_end, cu_symbol, kkr_credtype_code, kkr_credstatex_code, k_amount_approved,
             kkr_from_k_id, kkr_repaystatus, kkr_guarantee, coalesce(kkr_countofdelinquency, 0), /* число просрочек  */
             kibkr.kkr_overduepayoutnumber, /*порядковый номер выплаты*/
             coalesce(kkr_unpaidamount, 0), coalesce(kkr_unpaidpercentamount, 0), coalesce(kkr_unpaidfine, 0), /*остаточная сумма по Пени*/
             kibkr.kkr_paymethod, coalesce(kkr_fact_amount, 0), /*фактическая выплаченная группой сумма по данному погашению*/
             coalesce(kkr_plan_amount, 0), /*паланируема сумма - сумма, которую клиент должен платить каждый месяц*/
             coalesce(kkr_deliq_days, 0), /*число дней просрочки*/
             coalesce(kkr_pay_frequency, 0), /*частота платежа. На данный момент для всех "Ежемесячно"*/
             kibkr.kkr_overdueamount, kibkr.kkr_percentoverdueamount, k_kolich_repay, /* общее кол-во выплат. сколько раз должна группа заплатить*/
             kibkr.kkr_payoutday, kibkr.kkr_factrepaymentdate, kibkr.kkr_write_off_os, kibkr.kkr_write_off_perc
      from kibkredit kibkr, kontract, currlist, kibpocket
      where kkr_transid = 0 and
            k_currency = cu_id and
            kkr_from_k_id = k_id and
            kkr_from_kp_id = kp_id and
            ((kp_check_status = 1 and
            :is_urgent = 1) or
            (:is_urgent = 0))
      order by kkr_from_kp_id, kkr_from_k_id
      into :pocket_id, :uniq_cred_num, :bank_code, :fil_code, :number, :dd, :dateawd, :datestop, :curr_code,
           :credtype_code, :credstatex_code, :cred_amount, :k_id, :repaystatus, :guarantee, :countofdelinquency,
           :overduepayoutnumber, :unpaidamount, :unpaidpercentamount, :unpaidfine, /*остаточная сумма по Пени*/
           :paymethod, :fact_amount, /*фактическая выплаченная группой сумма по данному погашению*/
           :plan_amount, /*паланируема сумма - сумма, которую клиент должен платить каждый месяц*/
           :deliq_days, /*число дней просрочки*/
           :pay_frequency, /*частота платежа. На данный момент для всех "Ежемесячно"*/
           :overdueamount, :percentoverdueamount, :payoutcount, --
           :payoutday, :factrepaymentdate, /*фактическая дата закрытия кредита*/
           :writeoffos, :writeoffperc
  do
  begin
    if (credstatex_code = '1-0') then
    begin
      dateawd = :dd;
    end
    if (paymethod is null) then
      paymethod = 4;

    if (countofdelinquency > payoutcount) then
    begin
      payoutcount_tmp = 0;
      select count(*)
      from aarep_make_zad_list(:k_id, current_date, 1)
      where sh_date is not null
      into :payoutcount_tmp;

      payoutcount_tmp = coalesce(payoutcount_tmp, 0);
      if (payoutcount_tmp <> 0) then
        payoutcount = payoutcount_tmp;
    end

    if (countofdelinquency > payoutcount) then
      countofdelinquency = payoutcount;
    suspend;
  end

end
