CREATE PROCEDURE OKT_CC_PERSONS_INS_CALL(
    CP_ID INTEGER,
    CPC_NO INTEGER,
    CELL_NO VARCHAR(20),
    FIO STR_100,
    CITIZEN INTEGER,
    CITY STR_100,
    ADDIT_CELL_NO VARCHAR(20),
    INFO_SOURCE_ID INTEGER,
    INFO_SOURCE_MEMO STR_255,
    OPERATOR_ID ID_TYPE,
    QESTION_TEXT VARCHAR(1000),
    ANSWER_TEXT VARCHAR(1000),
    COMMENT_TEXT STR_255,
    CALL_TYPE SMALLINT,
    ANSWER_TYPE SMALLINT,
    C_ID ID_TYPE,
    EXT_INFO STR_255,
    CATEG_TEXT STR_100)
RETURNS (
    CP_ID_NEW INTEGER,
    CPC_NO_NEW INTEGER)
AS
declare variable cpc_qestion_oper_id id_type;
declare variable cpc_answer_oper_id id_type;
declare variable qestion_text_old varchar(1000);
declare variable cpc_qestion_oper_id_old id_type;
declare variable cp_fio str_100;
declare variable cp_citizen integer;
declare variable cp_city str_100;
declare variable cp_addit_cell_no str_100;
declare variable cpc_called_oper_id id_type;
declare variable cp_from_c_id id_type;
declare variable cp_from_is_id integer;
declare variable cp_memo str_255;
begin
  fio = trim(coalesce(fio, ''));
  city = trim(coalesce(city, ''));
  addit_cell_no = trim(coalesce(addit_cell_no, ''));

  qestion_text = trim(coalesce(qestion_text, ''));
  answer_text = trim(coalesce(answer_text, ''));

  if (c_id = 0) then
    c_id = null;

  if (answer_type = 1) then
    cpc_called_oper_id = operator_id;

  if (cp_id = 0) then
  begin
    insert into cc_persons (cp_cell_no, cp_fio, cp_citizen, cp_city, cp_addit_cell_no, cp_creator_id, cp_from_c_id,
                            cp_from_is_id, cp_memo, cp_ext_info)
    values (:cell_no, :fio, :citizen, :city, :addit_cell_no, :operator_id, :c_id, :info_source_id, :info_source_memo,
            :ext_info)
    returning cp_id
    into :cp_id;
  end
  else
  begin
    select cp.cp_fio, cp.cp_citizen, cp.cp_city, cp.cp_addit_cell_no, cp.cp_from_c_id, coalesce(cp.cp_from_is_id, -777),
           cp.cp_memo
    from cc_persons cp
    where cp.cp_id = :cp_id
    into :cp_fio, :cp_citizen, :cp_city, :cp_addit_cell_no, :cp_from_c_id, :cp_from_is_id, :cp_memo;

    if ((cp_fio <> fio) or (cp_citizen <> citizen) or (cp_city <> city) or (cp_addit_cell_no <> addit_cell_no) or (cp_from_is_id <> info_source_id) or (cp_memo <> info_source_memo)) then
    begin
      update cc_persons cp
      set cp.cp_fio = :cp_fio,
          cp.cp_citizen = :cp_citizen,
          cp.cp_city = :cp_city,
          cp.cp_addit_cell_no = :cp_addit_cell_no,
          cp.cp_edit_oper_id = :operator_id,
          cp.cp_from_is_id = :info_source_id,
          cp.cp_memo = :info_source_memo,
          cp.cp_ext_info = :ext_info
      where cp.cp_id = :cp_id;
    end

    if (coalesce(cp_from_c_id, 0) <> coalesce(:c_id, 0)) then
    begin
      update cc_persons cp
      set cp.cp_from_c_id = :c_id
      where cp.cp_id = :cp_id;
    end
  end

  if (cpc_no = 0) then
  begin
    cpc_qestion_oper_id = null;
    cpc_answer_oper_id = null;

    if (qestion_text <> '') then
      cpc_qestion_oper_id = operator_id;
    if (answer_text <> '') then
      cpc_answer_oper_id = operator_id;

    insert into cc_person_calls (cpc_from_cp_id, cpc_qestion_text, cpc_answer_text, cpc_qestion_oper_id,
                                 cpc_answer_oper_id, cpc_call_type, cpc_called_oper_id, cpc_memo, cpc_categ_str)
    values (:cp_id, :qestion_text, :answer_text, :cpc_qestion_oper_id, :cpc_answer_oper_id, :call_type,
            :cpc_called_oper_id, :comment_text, :categ_text)
    returning cpc_from_cp_id, cpc_no
    into cp_id, :cpc_no;
  end
  else
  begin
    qestion_text_old = '';

    select cpc.cpc_qestion_text, cpc.cpc_qestion_oper_id
    from cc_person_calls cpc
    where cpc.cpc_from_cp_id = :cp_id and
          cpc.cpc_no = :cpc_no
    into :qestion_text_old, cpc_qestion_oper_id_old;

    if (qestion_text_old <> qestion_text) then
      cpc_qestion_oper_id = operator_id;
    else
      cpc_qestion_oper_id = cpc_qestion_oper_id_old;

    update cc_person_calls cpc
    set cpc_qestion_text = :qestion_text,
        cpc_qestion_oper_id = :cpc_qestion_oper_id,
        cpc_answer_text = :answer_text,
        cpc_answer_oper_id = :operator_id,
        cpc_call_type = :call_type,
        cpc_called_oper_id = :cpc_called_oper_id,
        cpc_memo = :comment_text,
        cpc.cpc_categ_str = :categ_text
    where cpc.cpc_from_cp_id = :cp_id and
          cpc_no = :cpc_no;
  end

  cp_id_new = cp_id;
  cpc_no_new = cpc_no;
  suspend;
end
