CREATE PROCEDURE AAREP_ACTIVE_LOANS_ON_VU(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    K_NO D_STR_20,
    OFFICE D_STR_100,
    K_DISB_DATE D_DATE,
    OKO_DEBT DECIMAL(18,4))
AS
declare variable reg_id d_id;
begin
  for select distinct k.k_kontract_no,
             k.k_disb_date,
             sum(oko.oko_os + oko.oko_perc),
             ks.ks_region_id
      from operkroff oko
      inner join kontract k on k.k_id = oko.oko_from_k_id
      inner join kontstate ks on ks.ks_from_k_id = k.k_id and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0
      where oko.oko_ol_is_active = 1 and
            oko.oko_ol_date <= :rep_date_end and
            oko.oko_type = 4
      group by k.k_kontract_no,
               k.k_disb_date,
               oko.oko_id,
               ks.ks_region_id
      having sum(oko.oko_os + oko.oko_sh_os) + sum(oko.oko_perc + oko.oko_sh_perc) <> 0
      into :k_no,
           :k_disb_date,
           :oko_debt,
           :reg_id
  do
  begin
    if (oko_debt <= 0) then continue;
    office = '';
    select reg.reg_name
    from region reg
    inner join region_list(:reg_id, 1, 3, 1) on reg.reg_unicode = id_list
    into :office;
    suspend;
  end
end
