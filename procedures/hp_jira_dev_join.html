CREATE PROCEDURE HP_JIRA_DEV_JOIN(
    CORRECT_LOGIN D_STR_CODE,
    REMOVE_FIO D_STR_100)
AS
declare variable dev_correct_id d_id;
declare variable dev_remove_id d_id;
begin
  select jrd_id
  from jira_dev
  where jrd_login = :correct_login
  into dev_correct_id;

  if (dev_correct_id is null) then
    exception e_message 'Login not found: ' || coalesce(correct_login, '<Null>');

  select jrd_id
  from jira_dev
  where jrd_fio = :remove_fio and
        jrd_id <> :dev_correct_id
  into dev_remove_id;

  if (dev_remove_id is null) then
    exception e_message 'New FIO not found: ' || coalesce(remove_fio, '<Null>');

  update jira_issues
  set jri_dev_id = :dev_correct_id
  where jri_dev_id = :dev_remove_id;

  update jira_log
  set jrl_dev_id = :dev_correct_id
  where jrl_dev_id = :dev_remove_id;

  if (not exists(select * from jira_calc
                 where jrc_dev_id = :dev_correct_id)) then
  begin
    update jira_calc
    set jrc_dev_id = :dev_correct_id
    where jrc_dev_id = :dev_remove_id;
  end else
  begin
    if (exists(select * from jira_calc
               where jrc_dev_id = :dev_remove_id)) then
      exception e_message 'Both Dev_Id exists in Jira_calc';
  end

  if (not exists(select * from jira_kpi
                 where jrk_dev_id = :dev_correct_id)) then
  begin
    update jira_kpi
    set jrk_dev_id = :dev_correct_id
    where jrk_dev_id = :dev_remove_id;
  end else
  begin
    if (exists(select * from jira_kpi
               where jrk_dev_id = :dev_remove_id)) then
      exception e_message 'Both Dev_Id exists in Jira_calc';
  end

  delete from jira_dev
  where jrd_id = :dev_remove_id;
end
