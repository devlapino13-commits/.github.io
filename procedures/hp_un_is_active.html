CREATE PROCEDURE HP_UN_IS_ACTIVE(
    UN_ID DECIMAL(18,0))
RETURNS (
    UN_IS_ACTIVE INTEGER,
    UN_NAME VARCHAR(30))
AS
declare variable pa_id integer;
begin
  un_is_active = 0;
  un_name = '';
  if (un_id is null) then
  begin
    suspend;
    exit;
  end

  select un_name
  from username
  where un_id = :un_id
  into un_name;

  for select pa_id
      from permactions
      where pa_parent_id is null
      into :pa_id
  do
  begin
    if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', :pa_id, '')
               where has_access = 1)) then
    begin
      un_is_active = 1;
      suspend;
      exit;
    end
  end
  suspend;
end
