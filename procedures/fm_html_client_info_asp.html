CREATE PROCEDURE FM_HTML_CLIENT_INFO_ASP(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable ASP_NAME D_STR_CODE;
declare variable EDL_STATUS_STR D_STR_CODE;
declare variable DATE_TMP date;
declare variable LANG D_STR_LANG;
declare variable DB_ID D_INTEGER;
declare variable URI_TEXT D_STR_CODE;
declare variable URI_DESCRIPTION D_STR_100;
declare variable URI_EDITOR D_STR_CODE;
declare variable URI_COMMENT D_STR_100;
declare variable EDOC_STATE D_STR_CODE;
declare variable MSG_TEXT D_STR_8100;
declare variable MEET_VALID D_SMALLINT;
declare variable MSG_KIND D_SMALLINT;
declare variable EDS_ID D_ID;
declare variable ADD_TIME D_INTEGER;
declare variable ERROR_TEXT D_STR_255;
declare variable PERSTEMP_EXISTS D_BOOL;
declare variable IS_OIVO D_BOOL;
begin
  -- Формирует ссылку на подписание Договора АСП/ЭЦП для fm_html_client_info
  res_text = '';
  edoc_state = f_edoc_state2('asp.agreement', current_date, un_id);
  is_oivo = (select is_oivo from hp_check_oivo_assignment(:c_id, :un_id));

  if (edoc_state <> 'inactive') then
  begin
    lang = f_get_lang_code(un_id, null);
    db_id = f_get_db_id();
  
    asp_name = 'АСП';
    if (db_id in (39801, 41701)) then asp_name = 'ЭЦП';

    res_text = '<h2>' || f_format(f_lang(lang, 'Договор %1'), asp_name) || '</h2>';

    edl_status_str = f_edoc_status('asp.agreement', c_id, current_date);
    if (edl_status_str = 'signed') then
    begin
      select cast(eds_status_datetime as date)
      from edoc_list
      inner join edoc_sign on eds_from_edl_id = edl_id
      where edl_object_kind = 0 and
            edl_object_id = :c_id and
            edl_from_edc_id = f_edoc_kind('asp.agreement','id') and
            current_date between edl_blank_date and coalesce(edl_date_end, current_date)
      order by edl_status desc, edl_id desc
      rows 1
      into date_tmp;
      res_text = res_text || f_format(f_lang(lang, 'Договор %1 подписан: %2<br>'), asp_name,  f_datetostr(date_tmp));
    end
    else
    begin
      select msg_kind
      from hp_edoc_get_msg_kind('clients', :c_id, f_edoc_kind('asp.agreement', 'id'))
      into msg_kind;

      if (db_id in (39801, 41701)) then
      begin
        uri_description =  f_lang(:lang, 'Укажите подписанта для отправки договора клиенту');
        uri_editor = 'combo';
        uri_comment = f_lang(:lang, 'Клиенту будет отправлен договор через СМС.');

        if (msg_kind = 4) then
        begin
          uri_description =  f_lang(:lang, 'Укажите подписанта для отправки договора клиенту');
          uri_editor = 'combo';
          uri_comment = f_lang(:lang, 'Документ будет скоро сформирован для подписания через биометрию');
        end 
      end
      else
      begin
        uri_description =  f_lang(lang, 'Подтверждаете отправку договора через СМС клиенту?');
        uri_editor = 'none';
        uri_comment = '';
      end

      uri_text = '';
      if (edl_status_str = 'unsigned') then
      begin
        if (msg_kind <> 4) then
        begin
          res_text = res_text ||
            f_format(f_lang(lang, 'Договор %1 отправлен, но не подписан.'), asp_name) || '<br>';

          uri_text = f_lang(lang, 'Отправить повторно');
        end
        else
        begin
          res_text = res_text ||
            f_format(f_lang(lang, 'Договор %1 сформирован, но не подписан.'), asp_name) || '<br>';

          select eds.eds_id
          from edoc_list edl
          inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
          where edl.edl_object_kind = 0 and
                edl.edl_object_id = :c_id and
                edl.edl_from_edc_id = f_edoc_kind('asp.agreement', 'id') and
                edl.edl_status = 0 and
                current_date between edl.edl_blank_date and coalesce(edl.edl_date_end, current_date)
          order by edl.edl_id desc
          rows 1
          into eds_id;

          res_text = res_text || f_uri_open('Подписать ЭЦП через биометрию', 'edoc', 'client', c_id, 'Договор ЭЦП',  'eds_id=' || eds_id, un_id) || '<br>';
        end 

        select msg_text
        from hp_edoc_draft_message('asp.agreement', :c_id, :c_id)
        into :msg_text;

        if (coalesce(msg_text, '') <> '') then
          res_text = res_text || '<br>' || msg_text;
      end
      else -- нет договора АСП
      begin
        add_time = null;
        select add_time
        from hp_qr_additional_time(:c_id)
        into add_time;
        add_time = coalesce(add_time, 0);

        meet_valid = f_meet_valid(c_id, 120 + add_time);
        if (meet_valid in (0, 2) and is_oivo <> 1) then
        begin
          res_text =  res_text || f_lang(lang, 'Требуется верификация клиента по QR-коду.');

          if (db_id = 81001 and f_tmp_const('use_perstemp') = '1') then
          begin
            error_text = '';
            perstemp_exists = 0;
            select res_error_text, res_is_exists
            from hp_check_exists_perstemp(:c_id, 'name_date_place_birth')
            into error_text, perstemp_exists;

            if (error_text = '' and perstemp_exists = 1) then
              res_text = res_text || ' ' || f_lang(lang, 'Были изменены ФИО или дата/место рождения, нельзя отправить договор АСП.'); -- RFCRU-2205
          end
        end

        if (meet_valid in (1, 2) or is_oivo = 1) then
        begin
          if (msg_kind <> 4) then
            uri_text = f_format(f_lang(lang, 'Отправить договор %1 на подпись клиенту'), :asp_name);
          else
            uri_text = f_format(f_lang(lang, 'Сформировать договор %1 на подпись клиенту'), :asp_name);
        end
      end

      if (uri_text <> '') then
        res_text = res_text || f_uri_sql(:uri_text, 'asp_sign_sms', 'client',
          c_id, uri_editor, uri_description, uri_comment, 0, 0, '', '',
          'dialog', '') || '<br>';
    end
    if (is_oivo = 1 and f_meet_passed(:c_id) <> 1) then
      res_text =  res_text || f_lang(lang, '</br>Требуется верификация клиента по QR-коду.');
  end

  res_text = res_text || '<br>';
  suspend;
end
