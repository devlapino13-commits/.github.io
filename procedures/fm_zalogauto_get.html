CREATE PROCEDURE FM_ZALOGAUTO_GET(
    ZA_ID D_INTEGER,
    ZA_OBJECT_KIND D_STR_CODE NOT NULL,
    ZA_OBJECT_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_JSON VARCHAR(8100))
AS
declare variable tmp_json d_str_long;
declare variable cds_id d_id;
declare variable cds_name d_str_255;
declare variable c_id d_id;
declare variable pk_id d_integer;
declare variable pk_name d_str_code;
declare variable arr_name d_str_code;
declare variable field_name d_str_code;
declare variable sts_id d_id;
declare variable pts_id d_id;
begin
  res_json = '';
  za_object_kind = f_objkind(za_object_kind, 'name');

  if (za_object_kind not in('kontract', 'miniapp')) then
    exception e_default f_format('Incorrect object kind %1', za_object_kind);

  if (za_object_kind = 'miniapp') then
  begin
    select mac_from_c_id
    from miniappclient
    where mac_from_ma_id = :za_object_id and
          mac_amount > 0
    rows 1
    into c_id;
  end
  if (za_object_kind = 'kontract') then
  begin
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :za_object_id and
          kc_amount_approved > 0
    rows 1
    into c_id;
  end

  if (za_id is not null) then
  begin
    -- Для существующих подтягиваем данные для редактирования
    select
      f_json_make_pair('za_from_c_id', za_from_c_id, 'str') || ',' ||
      f_json_make_pair('za_from_c_name', c_name_fio, 'str') || ',' ||
      f_json_make_pair('za_orig_sts_id', za_orig_sts_id, 'str') || ',' ||
      f_json_make_pair('za_orig_pts_id', za_orig_pts_id, 'str') || ',' ||
      f_json_make_pair('zal_webpage', zal_webpage, 'str') || ',' ||
      f_json_make_pair('zal_cost', round(zal_cost), 'str'),
      za_orig_sts_id, za_orig_pts_id
    from zalogauto
      left join zalogautolist on za_from_zal_id = zal_id
      left join clients on za_from_c_id = c_id
    where
      za_id = :za_id
    into res_json, sts_id, pts_id;
  end

  -- Готовим списки для выбора документов СТС, ПТС
  for select 310, 'СТС', 'sts_list', 'za_orig_sts_id' from rdb$database
      union
      select 315, 'ПТС', 'pts_list', 'za_orig_pts_id' from rdb$database
  into pk_id, pk_name, arr_name, field_name do
  begin
    tmp_json = '';

    for select
      cds_id,
      f_format('%1 №%2 от %3', :pk_name, cds_doc_no, f_datetostr(cds_doc_date_begin))
    from clientdocs
    where
      cds_from_c_id = :c_id and
      cds_is_delete = 0 and
      cds_from_pk_id = :pk_id
    union
    select
      cds_id,
      f_format('%1 №%2 от %3', :pk_name, cds_doc_no, f_datetostr(cds_doc_date_begin))
    from clientdocs
    where cds_id in (:STS_ID, :PTS_ID) and
          cds_from_pk_id = :pk_id
    order by 1 desc
    rows 1
    into cds_id, cds_name do
    begin
      tmp_json = tmp_json || iif(tmp_json = '', '', ',') || '{' ||
        f_json_make_pair('id', cds_id, 'str') || ',' ||
        f_json_make_pair('name', cds_name, 'str') || '}';
    end

    res_json = res_json || iif(res_json <> '', ',', '') ||
      f_json_make_pair(:arr_name, '[' || tmp_json || ']', 'array');

    if (za_id is null) then -- Значения по-умолчанию для новых
      res_json = res_json || ',' || f_json_make_pair(:field_name, cds_id, 'str');
  end
  res_json = '{' || res_json || '}';

  suspend;
end
