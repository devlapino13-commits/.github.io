Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UI_UXACTIONEXEC(
    EXEC_ALIAS D_STR_CODE)
RETURNS (
    EXEC_CODE D_STR_CODE,
    EXEC_UA_UUID D_UUID,
    EXEC_PARAMS D_STR_LONG)
AS
declare variable uxa_uuid d_uuid;
declare variable mm_uuid d_uuid;
declare variable max_param_id d_smallint;
declare variable param_id d_smallint;
begin
  exec_params = '';

  select uxa_uuid
  from ux$alias
  where uxa_fcode = :exec_alias
  into uxa_uuid;

  if (row_count = 0) then
    Exit;

  mm_uuid = uxf$mainmenu();

  select ua_uuid
  from ux$action
  where ua_owner = :mm_uuid and
        ua_uxa_uuid = :uxa_uuid and
        ua_usr_id_field = ''
  into exec_ua_uuid;

  exec_code = exec_alias;

  select max(uap_exec_param_id)
  from ux$actionparam
  where uap_ua_uuid = :exec_ua_uuid and
        uap_is_output = 0
  into max_param_id;

  if (max_param_id is null) then
    max_param_id = 0;

  if (max_param_id > 0) then
  begin
    exec_params = '';
    param_id = 0;
    while (param_id < max_param_id) do
    begin
      if (exec_params <> '') then
        exec_params = exec_params || ';';

      exec_params = exec_params || param_id;
      param_id = param_id + 1;
    end
  end

  suspend;
end
