CREATE PROCEDURE HP_CARDEMBOMAIN_SETFNAME(
    CEM_ID D_INTEGER,
    FNAME D_STR_100,
    UN_ID D_ID)
RETURNS (
    OL_ID D_ID,
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable curr_code d_str_lang;
declare variable ol_id1 d_id;
declare variable is_own d_bool;
declare variable is_pincover d_bool;
declare variable reg_id d_integer;
declare variable reg_name d_str_code;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable card_count d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable ol_memo d_str_255;
begin
  curr_code = '417';

  select
    cem_app_count,
    cem_from_ol_id,
    cem_is_pincover,
    ckd_is_own
  from cardembomain
    inner join cardkind
      on ckd_id = cem_from_ckd_id
  where cem_id = :cem_id
    and cem_from_ces_id = 2
    and cem_filename is null
  into
    card_count,
    :ol_id,
    :is_pincover,
    :is_own;

  if (row_count = 0) then
  begin
    err_code = 1;
    err_text = 'Ошибка присвоения имени файла в заявке №' || cem_id;
    suspend;
    exit;
  end

  if (ol_id is null or exists(
      select *
      from operlist
      where ol_id = :ol_id
        and is_delete = 1)) then
  begin
    -- Бух.проводка
    select
      reg_unicode,
      reg_name,
      c_id,
      c_name_fio
    from cardplace
      inner join clients
        on c_id = cpl_from_c_id
      inner join region
        on reg_unicode = cpl_from_reg_id
    where cpl_id = 0
    into
      :reg_id,
      :reg_name,
      :c_id,
      :c_name_fio;

    if (is_own = 0) then
    begin
      acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/12');
      acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/12');
      if (acc_dt is null or acc_kt is null) then
      begin
        err_code = 2;
        err_text = 'В ' || reg_name || ' не открыты счета:' ||
          iif(acc_dt is null, ' 96101/12', '') || iif(acc_kt is null, ' 96102/12', '');
        suspend;
        exception e_message err_text;
      end
      ol_memo = 'Подача заявки на выпуск карт в "' || c_name_fio || '"; заявка № ' || cem_id || ', количествo ' || card_count || 'шт.';

      insert into operlist (
        ol_from_reg_id,
        ol_oper_type,
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_memo,
        ol_user_id,
        ol_document_client
       )
      values (
        :reg_id,
        2,
        2,
        current_date,
        :cem_id,
        0,
        :ol_memo,
        :un_id,
        :c_id
      )
      returning ol_id
      into :ol_id;

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        :curr_code,
        1,
        :acc_dt,
        :acc_kt,
        :card_count,
        :card_count
      );
    end

    if (is_pincover = 1) then
    begin
      acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/13');
      acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/13');
      if (acc_dt is null or acc_kt is null) then
      begin
        err_code = 2;
        err_text = 'В ' || reg_name || ' не открыты счета:' ||
          iif(acc_dt is null, ' 96101/13', '') || iif(acc_kt is null, ' 96102/13', '');
        suspend;
        exception e_message err_text;
      end
      ol_memo = 'Пин-конверты карт в "' || c_name_fio || '"; заявка № ' || cem_id || ', количествo ' || card_count || 'шт.';

      insert into operlist (
        ol_parent_id,
        ol_from_reg_id,
        ol_oper_type,
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_memo,
        ol_user_id,
        ol_document_client
       )
      values (
        :ol_id,
        :reg_id,
        2,
        2,
        current_date,
        :cem_id,
        0,
        :ol_memo,
        :un_id,
        :c_id
      )
      returning ol_id
      into :ol_id1;

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id1,
        :curr_code,
        1,
        :acc_dt,
        :acc_kt,
        :card_count,
        :card_count
      );
      ol_id = coalesce(ol_id, ol_id1);
    end

    update cardembomain set
      cem_filename = :fname,
      cem_from_un_id = :un_id
    where cem_id = :cem_id;

    err_code = 0;
    err_text = 'Имя файла успешно присвоено в заявке №' || cem_id;
  end else
  begin
    err_code = 1;
    err_text = 'Ошибка создания операции по подаче заявки на выпуск карт в МПЦ; заявка № ' || cem_id;
  end

  suspend;
end
