CREATE PROCEDURE AAREP_MAT_REPORT_LIST(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE INTEGER,
    REPORT_ID INTEGER)
RETURNS (
    MREP_REGION_ID INTEGER,
    MREP_ACCOUNT VARCHAR(20),
    MREP_BEGIN_KOLICH DECIMAL(18,4),
    MREP_BEGIN_AMOUNT DECIMAL(18,4),
    MREP_IN_KOLICH DECIMAL(18,4),
    MREP_IN_AMOUNT DECIMAL(18,4),
    MREP_OUT_KOLICH DECIMAL(18,4),
    MREP_OUT_AMOUNT DECIMAL(18,4),
    MREP_END_KOLICH DECIMAL(18,4),
    MREP_END_AMOUNT DECIMAL(18,4),
    MREP_WP_NAME VARCHAR(100),
    MREP_WP_STATUS_STR VARCHAR(500),
    MREP_MO_NAME VARCHAR(100),
    MREP_ML_NAME VARCHAR(255),
    MREP_YEDIZ VARCHAR(20),
    MREP_MC_NAME VARCHAR(100),
    MREP_MC_STATUS_STR VARCHAR(20),
    MREP_REG_NAME VARCHAR(100),
    MREP_ACC_NAME VARCHAR(255),
    MREP_BG_CODE VARCHAR(20),
    MREP_BG_NAME VARCHAR(255))
AS
declare variable mrep_mc_status integer;
declare variable mrep_wp_status integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /*
1 - Остаток ТМЦ,
2 - Полный оборот,
3 - Приход, расход,
4 - Перемещения,
5 - Списания, расход (откого),
6 - Списания, расход (кому)
*/
  for select mrep_region_id, mrep_account, mrep_begin_kolich, mrep_begin_amount, mrep_in_kolich, mrep_in_amount,
             mrep_out_kolich, mrep_out_amount, mrep_end_kolich, mrep_end_amount, wp_name, wp_status, c_name_fio,
             ml_name, yl_code, mc_name, mc_status, reg_name, al_name, al_from_bg_id, bg_name
      from aarep_mat_report_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_id)
      left join workplace on (mrep_wp_id = wp_id)
      left join clients on (mrep_mo_id = c_id)
      left join(matlist left join yedizlist on (ml_from_yl_id = yl_id)) on (mrep_ml_id = ml_id)
      left join matcateg on (mrep_mc_id = mc_id)
      left join region on (mrep_region_id = reg_unicode)
      left join(acc_list left join balgroup on (al_from_bg_id = bg_id)) on (mrep_account = al_id)

      into :mrep_region_id, :mrep_account, :mrep_begin_kolich, :mrep_begin_amount, :mrep_in_kolich, :mrep_in_amount,
           :mrep_out_kolich, :mrep_out_amount, :mrep_end_kolich, :mrep_end_amount, :mrep_wp_name, :mrep_wp_status,
           :mrep_mo_name, :mrep_ml_name, :mrep_yediz, :mrep_mc_name, :mrep_mc_status, :mrep_reg_name, :mrep_acc_name,
           :mrep_bg_code, :mrep_bg_name
  do
  begin

    if (mrep_mc_status = 0) then
      mrep_mc_status_str = '1 ОС';
    if (mrep_mc_status = 1) then
      mrep_mc_status_str = '2 МБП';
    if (mrep_mc_status = 2) then
      mrep_mc_status_str = '3 Немат.акт.';
    if (mrep_mc_status = 3) then
      mrep_mc_status_str = '4 ТМЗ, Расх.';

    if (mrep_wp_status = 5) then
      mrep_wp_status_str = 'Рабоч.место';
    if (mrep_wp_status = 6) then
      mrep_wp_status_str = 'Склад';
    if (mrep_wp_status = 7) then
      mrep_wp_status_str = 'Отдел';
    if (mrep_wp_status = 8) then
      mrep_wp_status_str = 'Прочее место';
    suspend;
  end
end
