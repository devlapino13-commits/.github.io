CREATE PROCEDURE HP_KONTSTATE_KONTRACT_CHECK
RETURNS (
    KS_FROM_OL_ID TYPE OF COLUMN KONTSTATE.KS_FROM_OL_ID,
    KS_FROM_K_ID TYPE OF COLUMN KONTSTATE.KS_FROM_K_ID,
    KS_DATE_BEGIN TYPE OF COLUMN KONTSTATE.KS_DATE_BEGIN,
    KS_DATE_END TYPE OF COLUMN KONTSTATE.KS_DATE_END,
    KS_OL_IS_DELETE TYPE OF COLUMN KONTSTATE.KS_OL_IS_DELETE,
    KS_STATE TYPE OF COLUMN KONTSTATE.KS_STATE,
    KS_KC_ID TYPE OF COLUMN KONTSTATE.KS_KC_ID,
    KS_REGION_ID TYPE OF COLUMN KONTSTATE.KS_REGION_ID,
    KS_KFOLDER_NO TYPE OF COLUMN KONTSTATE.KS_KFOLDER_NO,
    KS_BONUS_CHANGE TYPE OF COLUMN KONTSTATE.KS_BONUS_CHANGE)
AS
begin
  for select ks_from_ol_id, ks_from_k_id, ks_date_begin, ks_date_end, ks_ol_is_delete, ks_state, ks_kc_id, ks_region_id,
             ks_kfolder_no, ks_bonus_change
      from kontstate, kontract
      where ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ((ks_state = 0 and
            (k_applicate_date <> ks_date_begin)) or
            ((ks_state = 1) and
            ((k_fact_disb_date <> ks_date_begin) or
            (k_fact_disb_ol_id <> ks_from_ol_id) or
            (k_fact_disb_lo_id <> ks_kc_id) or
            (k_fact_disb_reg_id <> ks_region_id) or
            (k_fact_disb_kf_no <> ks_kfolder_no) or
            (k_fact_disb_bonus_state <> ks_bonus_change))) or
            ((ks_state = 2) and
            ((k_fact_closed_date <> ks_date_begin) or
            (k_fact_closed_ol_id <> ks_from_ol_id) or
            (k_fact_closed_lo_id <> ks_kc_id) or
            (k_fact_closed_reg_id <> ks_region_id) or
            (k_fact_closed_kf_no <> ks_kfolder_no) or
            (k_fact_closed_bonus_change <> ks_bonus_change))))
      into :ks_from_ol_id, :ks_from_k_id, :ks_date_begin, :ks_date_end, :ks_ol_is_delete, :ks_state, :ks_kc_id,
           :ks_region_id, :ks_kfolder_no, :ks_bonus_change
  do
  begin
    suspend;
  end
end
