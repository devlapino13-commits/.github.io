CREATE PROCEDURE STR_WITH_SEPARATOR_PARSER(
    INP_STR D_BLOB_TEXT,
    SEPARATOR VARCHAR(1) = ';')
RETURNS (
    VALUE_OUT D_STR_255)
AS
declare variable emerg_exit_counter d_integer;
declare variable max_iter d_integer;
begin
  /*
    Принимает на вход строку и разделитель.
    Разбивает исходную строку по разделителю.
  */
  value_out = '';
  if (inp_str = '') then
  begin
    suspend;
    exit;
  end
  emerg_exit_counter = 0;
  max_iter = char_length(:inp_str);
  if (:separator = '') then
    exception e_message 'Необходимо указать разделитель!';
  if (:inp_str not containing :separator) then
    exception e_message 'Строка не содержит указанный разделитель!';
  if (right(:inp_str, 1) <> :separator) then
    :inp_str = :inp_str || :separator;
  while (inp_str <> '') do
  begin
    emerg_exit_counter = emerg_exit_counter + 1;
    if (emerg_exit_counter > :max_iter) then
      exception e_message 'Произошла ошибка!';
    value_out = left(:inp_str, position(:separator in :inp_str) - 1);
    inp_str = overlay(:inp_str placing '' from 1 for position(:separator in :inp_str));
    if (value_out = '') then
      continue;
    suspend;
  end
end
