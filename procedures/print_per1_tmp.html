CREATE PROCEDURE PRINT_PER1_TMP(
    C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    HO_DATE DATE,
    HO_TEXT VARCHAR(1024),
    HO_REASON VARCHAR(255),
    HO_ORDER VARCHAR(100),
    HO_ORDER_NO VARCHAR(20),
    HO_REGION VARCHAR(100),
    HO_DATE_BEGIN DATE,
    HO_DATE_END DATE,
    HO_JOBNAME VARCHAR(255),
    HO_TABEL_NO VARCHAR(10),
    HO_IS_ORDER INTEGER)
AS
declare variable dd date;
begin
  ho_is_order = 1;
  for select ho_date, r1.reg_code || ' ' || r1.reg_name, ho_text, ho_reason, sawt_name, ho_order_no, ps_date_begin,
             ps_date_end, r2.reg_code || ' ' || r2.reg_name || ', ' || jl_name, ps_tabel_no, ps_date_end_in_order

      from hr_orders, region r1, sa_worktype, hr_pstatus, region r2, depart, joblist
      where ho_from_reg_id = r1.reg_unicode and
            ho_order_type = sawt_id and
            ps_from_ho_id = ho_id and
            ps_from_c_id = :c_id and
            ps_from_reg_id = r2.reg_unicode and
            ps_from_dep_id = dep_id and
            ps_from_jl_id = jl_id and
            (ho_date <= :rep_date or
            ps_date_begin <= :rep_date) and
            ps_status in (0, 1) -- включает только аннулированные и утвержденные
      into :ho_date, :ho_region, :ho_text, :ho_reason, :ho_order, :ho_order_no, :ho_date_begin, :ho_date_end,
           :ho_jobname, :ho_tabel_no, :dd
  do
  begin
    if (ho_date_end >= '31.12.2099') then
      ho_date_end = :dd;
    suspend;
  end
  ho_tabel_no = null;
  ho_is_order = 0;
  for select ho_date, r1.reg_code || ' ' || r1.reg_name, ho_text, ho_reason, sawt_name, ho_order_no, hrv_date_begin,
             hrv_date_end, hrv_direction

      from hr_orders, region r1, sa_worktype, hr_voctravel
      where ho_from_reg_id = r1.reg_unicode and
            ho_order_type = sawt_id and
            hrv_from_ho_id = ho_id and
            hrv_from_c_id = :c_id and
            (ho_date <= :rep_date or
            hrv_date_begin <= :rep_date) and
            ho_status in (0, 1) -- включает только неаннулированные
      into :ho_date, :ho_region, :ho_text, :ho_reason, :ho_order, :ho_order_no, :ho_date_begin, :ho_date_end,
           :ho_jobname
  do
  begin
    suspend;
  end

end
