CREATE PROCEDURE HP_REACCEPT_REPAYMENT(
    K_ID D_ID,
    TO_DATE D_DATE = '01.12.2020',
    UN_ID D_ID = 1000000000000,
    MEMO D_STR_100 = '')
RETURNS (
    RES_TXT D_STR_255,
    LST_OL_ID D_STR_8100,
    LST_NEW_OL_ID D_STR_8100)
AS
declare variable delim_sym d_str_sym;
declare variable new_ol_id d_id;
declare variable new_ol_parent_id d_id;
declare variable err_txt d_str_255;
declare variable acc_bal varchar(5);
declare variable c_id d_id;
declare variable ol_id d_id;
declare variable ol_tmp_id d_id;
declare variable ol_date d_date;
declare variable ol_datetime d_datetime;
declare variable ol_parent_id d_id;
declare variable ol_doc_no d_integer;
declare variable ol_doc_no_str d_str_20;
declare variable repay_fact_office_id d_id;
declare variable repay_user_id d_id;
declare variable repay_document_client d_id;
declare variable repay_doc_client_name_reason d_str_255;
declare variable repay_manager_c_id d_id;
declare variable repay_from_rd_id d_integer;
declare variable repay_amount d_currency;
declare variable repay_to_os d_integer;
declare variable repay_to_perc d_integer;
declare variable repay_to_fine d_integer;
declare variable repay_to_kommis d_integer;
declare variable repay_dt_account_kind d_integer;
declare variable repay_dt_account_no d_str_20;
declare variable is_full_repay d_integer;
declare variable is_ipf d_integer;
declare variable k_fact_closed_ol_id d_id;
declare variable min_date d_date;
declare variable max_date d_date;
declare variable tmp_date d_date;
declare variable period_beg d_integer;
declare variable period_end d_integer;
begin
  res_txt = '';
  delim_sym = '|';
  lst_ol_id = '';
  lst_new_ol_id = '';

  if (to_date < '01.12.2020')
    then to_date = '01.12.2020';
  min_date = to_date;
  max_date = current_date;--'01.01.1900';

  if (not exists(
    select *
    from knclient_compens
    where kcc_from_k_id = :k_id))
  then res_txt = 'Данный контракт не участвует в акции.';

    if (exists(
        select *
        from earlypaid_november epn
        inner join operlist ol on ol.ol_id = epn.en_ol_id
        where ol.is_delete = 0
        and epn.en_k_id =:k_id
        and ol.ol_date >= :to_date))
  then res_txt = 'Была перекидка ранее выплаченных % на декабрь-февраль';

  /*
  if (res_txt = '' and exists(
    select *
    from kontstate
    where ks_from_k_id = :k_id
    and  ks_date_begin > :to_date
    and ks_state = 1
    and ks_ol_is_delete = 0))
  then res_txt = 'На '  || f_datetostr(:to_date) || ' контракт еще не был выдан.';
  */

  if (res_txt = '' and exists(
    select *
    from operkred
      where ok_from_k_id = :k_id
        and ok_ol_date >= :to_date
        and ok_type = 2
        and ok_ol_is_active = 1))
  then res_txt = 'Перепроводка погашений отменена(была реструктуризация).';

  if (res_txt = '' and exists(
    select *
    from operkred
      where ok_from_k_id = :k_id
        and ok_ol_date >= :to_date
        and ok_type = 5
        and ok_ol_is_active = 1))
  then res_txt = 'Перепроводка погашений отменена(было списание).';

  if (res_txt = '' and exists(
    select *
    from operkred ok
      where ok_from_k_id = :k_id
        and ok_ol_date >= :to_date
        and ok_type = 9
        and ok.ok_os > 0
        and ok_ol_is_active = 1))
  then res_txt = 'Перепроводка погашений отменена(был возврат переплаты).';

  
  if (res_txt = '' and exists(
    select *
    from knclient kncl
    inner join operlist ol on ol.ol_id = kncl.kc_closed_ol_id
        and ol.is_delete = 0
    where kncl.kc_from_k_id = :k_id
        and kncl.kc_amount_approved > 0
        and ol.ol_date >= :to_date))
  then res_txt = 'Перепроводка погашений отменена(было закрытие доли участника).';


  if (res_txt = '' and exists(
    select *
    from operkroff okf
      where okf.oko_from_k_id = :k_id
        and okf.oko_ol_date >= :to_date
        and okf.oko_ol_is_active = 1))
  then res_txt = 'Было движение по внесистемному учету';

  if (res_txt = '' and exists(
    select *
    from operextra oke
      where oke.ox_object_kind = 9
        and oke.ox_object_id = :k_id
        and oke.ox_ol_date >= :to_date
        and oke.ox_ol_is_active = 1))
  then res_txt = 'Было движение по operextra';

  /* igusev отключил по запросы Кыял А. 25.01.2021
  if (res_txt = '' and exists (
    select * from earlypaid_november
    inner join operlist ol on ol.ol_id = en_ol_id
    where en_k_id = :k_id
    and ol.is_delete = 0))
  then res_txt = 'Было погашение по ОС за счет выплаченных % в ноябре';
  */

  if (res_txt = '') then
  begin
    period_beg = 100 * extract(year from to_date) + extract(month from to_date);
    period_end = 100 * extract(year from current_date) + extract(month from current_date);
    if (exists(
      select *
      from kontcalcperc
      where kcp_period between :period_beg and :period_end))
    then res_txt = 'Необхобимо удалить начисленные проценты';
  end

  if (res_txt = '') then
  begin
    select
      iif(pr_code containing '.ipf.', 1 ,0),
      ol.ol_id
    from kontract
      left join operlist ol
        on ol.ol_id = k_fact_closed_ol_id
          and ol.is_delete = 0
      inner join product
        on pr_id = k_from_pr_id
    where k_id = :k_id
    into
      :is_ipf,
      :k_fact_closed_ol_id;

    -- операции для аннулирования
    for
      select
        ok_from_ol_id,
        ok_ol_date
      from operkred
      where ok_from_k_id = :k_id
        and ok_ol_date >= :to_date
        and ok_type in (4, 9)
        and ok_ol_is_active = 1
      group by ok_ol_date, ok_from_ol_id
      order by ok_ol_date desc, ok_from_ol_id desc
    into
        :ol_id,
        :ol_date
    do
    begin
      -- выполняется аннулирование
      update operlist set
        is_delete = 1
      where ol_id = (select iif(ol_parent_id > 0, ol_parent_id, ol_id)
        from operlist where ol_id = :ol_id);

      -- в список на перепроводку попадают только операции не от робота
      if (not exists(
        select *
        from compens_robot_log
        where crl_from_ol_id = :ol_id)) then
      begin
        --if (ol_date > max_date) then max_date = ol_date;
        lst_ol_id = ol_id || delim_sym || lst_ol_id;
      end
    end

    if (is_ipf = 1 and to_date = '01.12.2020')
    then execute procedure fsrobot_opercompens(:to_date, :k_id);

    tmp_date = min_date;
    --if (lst_ol_id <> '') then
    while (tmp_date between min_date and max_date) do
    begin
      -- если платеж по графику
      if (is_ipf = 0 and exists(
        select *
        from (
          select
            ok_sh_date,
            sum(ok_os + ok_sh_os) sh_os,
            sum(ok_perc + ok_sh_perc) sh_perc
          from operkred
          where ok_from_k_id = :k_id
            and ok_type < 7
            and ok_ol_is_active = 1
            and ok_from_c_id = 0
            and ok_ol_date <= :tmp_date
          group by ok_sh_id, ok_sh_date)
        where ok_sh_date = :tmp_date
          and (sh_os <> 0 or sh_perc <> 0)))
      then execute procedure fsrobot_opercompens(:tmp_date, :k_id);

      for
        select
          res_str
        from hp_split_str(:lst_ol_id, :delim_sym)
      into
        :ol_id
      do
      begin
        select
          ol_date,
          ol_parent_id
        from operlist
        where ol_id = :ol_id
          and ol_date = :tmp_date
        into
          :ol_date,
          :ol_parent_id;
        if (row_count = 0) then continue;
        ol_tmp_id = coalesce(ol_parent_id, ol_id);

        -- сбор данных для вх.параметров hp_do_repayment ... etc.
        repay_to_os = 1;
        repay_to_perc = 1;
        repay_to_fine = 1;
        repay_to_kommis = 1;
        select
          ol_datetime,
          ol_doc_no,
          ol_doc_no_str,
          ol_from_reg_id,
          ol_user_id,
          ol_document_client,
          ol_doc_client_name,
          ol_manager_c_id,
          ol_from_rd_id
        from operlist
        where ol_id = :ol_tmp_id
        into
          :ol_datetime,
          :ol_doc_no,
          :ol_doc_no_str,
          :repay_fact_office_id,
          :repay_user_id,
          :repay_document_client,
          :repay_doc_client_name_reason,
          :repay_manager_c_id,
          :repay_from_rd_id;

        -- проверка на "полное закрытие долга"
        is_full_repay = 0;
        /*
        if (k_fact_closed_ol_id = ol_id)
        then is_full_repay = 1; else is_full_repay = 0;
        */

        -- проверка на "закрытие долга одним участником"
        if (exists(
          select *
          from knclient
          where kc_from_k_id = :k_id
            and kc_from_c_id = :repay_document_client
            and kc_closed_ol_id = :ol_id))
        then c_id = repay_document_client; else c_id = 0;

        -- сумма погашения & счёт
        select
          sum(ob_summa),
          max(ob_account_dt)
        from operbook
        where ob_from_ol_id = :ol_tmp_id
          and substring(ob_account_dt from 8 for 5) not in ('11531', '80611')
        into
          :repay_amount,
          :repay_dt_account_no;

        -- вид оплаты
        acc_bal = substring(repay_dt_account_no from 8 for 5);
        repay_dt_account_kind =
        case
          when acc_bal in ('10001', '10023') then 1
          when acc_bal in ('10101', '10150') then 2
        else 3 end;

        -- погашение
        select
          loan_ol_id,
          ol_id,
          error_text
        from hp_do_repayment (
          :k_id,
          :c_id,
          :ol_date,
          :repay_amount,
          :repay_to_os,
          :repay_to_perc,
          :repay_to_fine,
          :repay_to_kommis,
          1,
          :is_full_repay,
          0,
          :repay_fact_office_id,
          :repay_dt_account_kind,
          :repay_dt_account_no,
          0,
          :repay_user_id,
          :repay_document_client,
          :repay_doc_client_name_reason,
          :repay_manager_c_id,
          :repay_from_rd_id)
        into
          :new_ol_id,
          :new_ol_parent_id,
          :err_txt;
        lst_new_ol_id = lst_new_ol_id || new_ol_id || delim_sym;

        for
          select ol_id from operlist
          where ol_id in (:new_ol_id, :new_ol_parent_id)
          union
          select ol_id from operlist
          where ol_parent_id in (:new_ol_id, :new_ol_parent_id)
        into
          :ol_tmp_id
        do
        begin
          update operlist set
             ol_datetime = :ol_datetime,
             ol_doc_no = :ol_doc_no,
             ol_doc_no_str = :ol_doc_no_str
          where ol_id = :ol_tmp_id;
        end
      end

      tmp_date = tmp_date + 1;
    end
  end

  insert into reaccept_repayment_log (rrl_from_k_id, rrl_res_txt, rrl_lst_ol_id, rrl_lst_new_ol_id, rrl_un_id, rrl_memo)
  values (:k_id, :res_txt, :lst_ol_id, :lst_new_ol_id, :un_id, :memo);

  execute procedure hp_extsysoper_fix('kontract', :k_id, :to_date);

  -- должно быть в самом низу. Удаляю сформировавшиеся смс
  delete from fsrobot_eventlist fse
  where fel_command_kind in ('Loan_LO_Repay_OK', 'Loan_Repay_OK', 'Loan_Repay_X', 'Push_Loan_LO_Repay');

  suspend;
end
