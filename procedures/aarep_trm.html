CREATE PROCEDURE AAREP_TRM(
    REG_ID D_INTEGER,
    TERM_ID D_STR_100,
    DTBEG D_DATE,
    DTEND D_DATE,
    TMBEG VARCHAR(8),
    TMEND VARCHAR(8))
RETURNS (
    SUM_CLASSIC CURR_TYPE,
    SUM_IPF CURR_TYPE)
AS
declare reg_lst d_str_2048;
declare dateb1 d_datetime;                      
declare datee1 d_datetime;
begin
  reg_lst = (select list(id_list, ',') from region_list(:reg_id, 1, -1, 1));
  reg_lst = reg_lst || ',';

  dateb1 = dtbeg + cast(tmbeg as time);
  datee1 = dtend + cast(tmend as time);
      
  select
    sum(iif(pr_code containing '.ipf.', 0, esm_amount)),  
    sum(iif(pr_code containing '.ipf.', esm_amount, 0))
  from extsysmain
    inner join kontract
      on k_id = esm_from_k_id
    inner join product
      on pr_id = k_from_pr_id
    inner join hp_extsys_parcer(esm_full_source, 'terminal_number') as nm
      on nm.field_value = :term_id
  where esm_date between :dtbeg and :dtend
    and esm_project in ('doscredo', 'umai')                     
    and esm_action = 'pay'
    and esm_status = 1
    and esm_from_k_id > 0
    and esm_datetime between :dateb1 and :datee1                               
    and exists(
      select *
      from terminals
      where tm_terminal_id = nm.field_value
        and tm_project = 2
        and esm_date between coalesce(tm_date_begin, '01.01.1900')
        and coalesce(tm_date_end, '31.12.9999')
        and :reg_lst containing tm_office || ',')
  into                  
    :sum_classic,
    :sum_ipf;
  sum_classic = coalesce(sum_classic, 0);                                         
  sum_ipf = coalesce(sum_ipf, 0);                                         
  suspend;                             
end
