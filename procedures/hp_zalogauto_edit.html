CREATE PROCEDURE HP_ZALOGAUTO_EDIT(
    ZA_ID D_INTEGER,
    ZA_OBJECT_KIND D_STR_CODE,
    ZA_OBJECT_ID D_ID,
    ZA_FROM_C_ID D_ID,
    ZAL_COST D_CURRENCY,
    ZA_ORIG_STS_ID D_INTEGER,
    ZA_ORIG_PTS_ID D_INTEGER,
    ZA_NEW_STS_ID D_INTEGER,
    ZA_NEW_PTS_ID D_INTEGER,
    ZAL_WEBPAGE D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable zap_id d_integer;
declare variable zal_id d_integer;
declare variable lang d_str_lang;
declare variable pr_code str_100;
begin
  -- Создание / Редактирование автозалога перед отправкой на Профсуждение
  select pr_code
  from kontract
  inner join product on pr_id = k_from_pr_id
  where k_id = :za_object_id
  into pr_code;

  res_is_error = 0;
  res_text = '';
  lang = f_get_lang_code(un_id, null);
  za_object_kind = f_objkind(za_object_kind, 'name');

  if (za_object_kind not in('kontract', 'miniapp')) then
    exception e_default f_format('Invalid object kind: ', za_object_kind);

  if (za_object_id is null) then
    exception e_default 'Field obj_id is null';

  if (za_from_c_id is null) then
    res_text = f_lang(lang, 'Не указан поручитель');

  if (za_orig_sts_id is null) then
    res_text = f_lang(lang, 'Не указан документ СТС (до оформления)');

  if (za_orig_pts_id is null) then
    res_text = f_lang(lang, 'Не указан документ ПТС (до оформления)');

  if (za_orig_sts_id = 0) then
    za_orig_sts_id = null;

  if (za_orig_pts_id = 0) then
    za_orig_pts_id = null;

  if ((coalesce(zal_cost, 0) = 0) and pr_code not containing '.zalog.') then
    res_text = f_lang(lang, 'Не указанa стоимость автозалога');

  if (res_text <> '') then
  begin
    res_is_error = 1;
    suspend;
    exit;
  end

  if (za_id is null) then
  begin
    if (za_object_kind = 'miniapp') then
    begin
      insert into zalogauto (za_from_ma_id, za_from_c_id, za_orig_sts_id, za_orig_pts_id, za_new_sts_id, za_new_pts_id)
      values (:za_object_id, :za_from_c_id, :za_orig_sts_id, :za_orig_pts_id, :za_new_sts_id, :za_new_pts_id)
      returning za_id
      into :za_id;
    end
    else if (za_object_kind = 'kontract') then
    begin
      insert into zalogauto (za_from_k_id, za_from_c_id, za_orig_sts_id, za_orig_pts_id, za_new_sts_id, za_new_pts_id)
      values (:za_object_id, :za_from_c_id, :za_orig_sts_id, :za_orig_pts_id, :za_new_sts_id, :za_new_pts_id)
      returning za_id
      into :za_id;
    end

    insert into zalogautoprof (zap_from_za_id, zad_from_un_id)
    values (:za_id, :un_id)
    returning zap_id
    into :zap_id;

    insert into zalogautolist (zal_from_zap_id, zal_cost, zal_webpage, zal_from_un_id)
    values (:zap_id, :zal_cost, :zal_webpage, :un_id)
    returning zal_id
    into :zal_id;

    update zalogauto
    set za_from_zal_id = :zal_id
    where za_id = :za_id;
  end else
  begin
    update zalogauto
    set za_from_c_id = :za_from_c_id,
        za_orig_sts_id = :za_orig_sts_id,
        za_orig_pts_id = :za_orig_pts_id,
        za_new_sts_id = :za_new_sts_id,
        za_new_pts_id = :za_new_pts_id
    where (za_id = :za_id)
    returning za_from_zal_id
    into zal_id;

    update zalogautolist
    set zal_cost = :zal_cost,
        zal_webpage = :zal_webpage,
        zal_from_un_id = :un_id,
        zal_datetime = current_timestamp
    where (zal_id = :zal_id);
  end

  suspend;
end
