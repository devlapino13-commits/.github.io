CREATE PROCEDURE HP_STATVALUE_LIST(
    STI_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    STR_ID D_INTEGER,
    STR_NAME D_STR_255,
    STR_FROM_PL_ID D_ID,
    STR_FROM_PL_NAME D_STR_255,
    STV_ID D_ID,
    STV_PERIOD_BEGIN D_INTEGER,
    STV_VALUE D_CURRENCY)
AS
declare variable period_begin d_integer;
begin
  period_begin = extract(year from rep_date) * 100 + extract(month from rep_date);

  for select str_id, str_name, str_from_pl_id
  from statregion
  order by str_name
  into :str_id, :str_name, :str_from_pl_id
  do
  begin
    stv_id = null;
    select stv_id, stp_period_begin, stv_value
    from statvalue, statpocket
    where stp_id = stv_from_stp_id and
          stp_from_sti_id = :sti_id and
          stp_period_begin <= :period_begin and
          stp_date_begin <= :rep_date and
          stp_date_end >= :rep_date and
          stv_from_str_id = :str_id
    order by stp_period_begin desc, stv_id desc
    rows 1
    into stv_id, stv_period_begin, stv_value;

    if (stv_id is not null) then
    begin
      str_from_pl_name = '';

      select iif(plt_short_name_side = -1, plt_name || ' ', '') ||
             pl_name ||
             lower(iif(plt_short_name_side = 1, ' '|| plt_name, ''))
      from places, placetype
      where pl_id = :str_from_pl_id and
            pl_status = plt_id
      into :str_from_pl_name;

      suspend;
    end
  end
end
