CREATE PROCEDURE UXP$UXDEPEND_FILL(
    UX_UUID D_UUID,
    UX_NEED_CLEAR D_BOOL = 1,
    UX_UUID_CODE D_STR_CODE = null)
AS
declare variable max_level d_integer;
declare variable ufm_count d_integer;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code, 0);

  if (ux_need_clear is null) then
    ux_need_clear = 1;

  if (ux_need_clear = 0) then
  begin
    delete from ux$depend
    where udp_owner_uuid = :ux_uuid;
  end else
  begin
    delete from ux$depend
    where 1 = 1;
  end

  if (coalesce(ux_uuid, '') = '') then
    exit;

  if (not exists(select * from ux$uniq
  where uu_id = :ux_uuid)) then
    exit;

  max_level = 10;

  execute procedure uxp$uxdepend_create_tree(:ux_uuid, :ux_uuid, null, 0, :max_level);

  for select ufm_ux_uuid, udp_lock_delete, udp_child_locked
  from ux$depend, ux$form, ux$code
  where ufm_uuid = udp_uuid and
        ufm_ux_uuid = ux_uuid and
        (udp_lock_delete = 1 or udp_child_locked = 1) and
        ux_release_uuid is distinct from ufm_uuid
  as cursor cur
  do
  begin
    select count(*)
    from ux$form
    where ufm_ux_uuid = cur.ufm_ux_uuid
    into ufm_count;

    if (ufm_count > 1) then
    begin
      update ux$depend
      set udp_lock_delete = 0,
          udp_child_locked = 0
      where current of cur;
    end
  end
end
