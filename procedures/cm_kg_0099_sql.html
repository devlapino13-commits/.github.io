        if (exists (

            and pr.pr_code containing '.business.'
        )) then

            begin

          (living_expenses is not null) and

          (legal_docs_expenses is not null) and

          (repayment_obligations is not null) and

        if (net_profit <> monthly_income + extra_amount - communal_expenses - living_expenses - /*UNFORESEEN_EXPENSES - */ legal_docs_expenses - other_business_expenses - repayment_obligations) then
