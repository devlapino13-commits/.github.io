CREATE PROCEDURE MAT_AKT_MAIN(
    AKT_OL_ID DECIMAL(18,0))
RETURNS (
    MAM_PERSON_IN DECIMAL(18,0),
    MAM_PERSON_OUT DECIMAL(18,0),
    MAM_PERSON_NAME_IN VARCHAR(100),
    MAM_PERSON_NAME_OUT VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  for select distinct om_matotvet_id, om_matotvet_info, c1.c_name_fio, c2.c_name_fio
      from opermat
      left join clients c1 on (c1.c_id = om_matotvet_id)
      left join clients c2 on (c2.c_id = om_matotvet_info)
      where om_from_ol_id = :akt_ol_id and
            om_main_line = 1 and
            om_oper_type <= 5 and
            (om_kolich_plus > 0 or
            om_summa_plus > 0)
      union
      select distinct om_matotvet_info, om_matotvet_id, c2.c_name_fio, c1.c_name_fio
      from opermat
      left join clients c1 on (c1.c_id = om_matotvet_id)
      left join clients c2 on (c2.c_id = om_matotvet_info)
      where om_from_ol_id = :akt_ol_id and
            om_main_line = 1 and
            om_oper_type <= 5 and
            (om_kolich_minus > 0 or
            om_summa_minus > 0)
      into :mam_person_in, :mam_person_out, :mam_person_name_in, :mam_person_name_out
  do
  begin
    if (mam_person_name_in is null) then
      mam_person_name_in = '';
    if (mam_person_name_out is null) then
      mam_person_name_out = '';
    suspend;
  end
end
