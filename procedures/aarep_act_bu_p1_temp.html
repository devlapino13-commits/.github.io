CREATE PROCEDURE AAREP_ACT_BU_P1_TEMP(
    KD_K_ORIG ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    K_ID ID_TYPE,
    SH_DATE DATE,
    ZAD_OS CURR_TYPE,
    SH_CALC_PERC CURR_TYPE)
AS
begin
  for
--  select :kd_k_orig as k_id from rdb$database
--  union
--  select kd_k_new from kontdevides
--  where kd_k_orig = :kd_k_orig
      select k_id
      from kontract
      where k_id = :kd_k_orig or
            k_base_k_id = :kd_k_orig
      into k_id
  do
  begin
    for select ok_sh_date, -ok_os ok_os, -ok_perc ok_perc
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_ol_date = :rep_date_end and
              ok_type = 5
        union all
        select sh_date, zad_os, sh_calc_perc
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date is not null and
              (zad_os > 0 or
              sh_calc_perc > 0)
        into sh_date, zad_os, sh_calc_perc

    end
  end
end
