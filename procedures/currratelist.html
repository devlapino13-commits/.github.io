Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE CURRRATELIST(
    DD D_DATE,
    DEF_CURR D_STR_LANG)
RETURNS (
    CRL_ID D_STR_LANG,
    CRL_SYMBOL D_STR_10,
    CRL_NAME D_STR_CODE,
    CRL_RATE D_CURRENCY,
    CRL_UNIT D_INTEGER,
    CRL_DATE D_DATE,
    CRL_DATE_LAST D_DATE,
    CRL_RATE_PRIOR D_CURRENCY,
    CRL_UNIT_PRIOR D_INTEGER,
    CRL_DATE_PRIOR D_DATE,
    CRL_RATIO D_STR_10)
AS
begin
  -- возвращает курс валют заданной валюты (def_curr)
  for
    select distinct
      cu_id,
      cu_symbol,
      cu_name
    from opercurr
      inner join currlist
        on cr_from_cu_id = cu_id
    where cr_from_cu_id <> :def_curr
    order by cr_from_cu_id
  into
    :crl_id,
    :crl_symbol,
    :crl_name
  do
  begin
    select
      cr_rate,
      cr_date,
      cr_curr_unit
    from opercurr
    where cr_from_cu_id = :crl_id
      and :dd between cr_date and cr_date_end
    rows 1
    into
      :crl_rate,
      :crl_date,
      :crl_unit;

    crl_rate_prior = crl_rate;
    crl_date_prior = crl_date;
    crl_ratio = '1:' || crl_unit;

    select
      cr_rate,
      cr_date,
      cr_curr_unit
    from opercurr
    where cr_from_cu_id = :crl_id
      and cr_date_end = :crl_date - 1
    rows 1
    into
      :crl_rate_prior,
      :crl_date_prior,
      :crl_unit_prior;

    select
      max(cr_date)
    from opercurr
    where cr_from_cu_id = :crl_id
    into
      :crl_date_last;

    if (crl_date_last is null) then
      crl_date_last = crl_date;

    suspend;
  end
end
