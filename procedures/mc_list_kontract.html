CREATE PROCEDURE MC_LIST_KONTRACT(
    MCL_ID D_ID)
RETURNS (
    K_ID D_ID,
    K_KONTRACT_NO STR_20,
    K_STATUS D_STR_100,
    K_FACT_DISB_DATE D_DATE)
AS
declare variable date_end d_date;
declare variable c_id d_id;
declare variable sum_zadol d_currency;
declare variable ks_state d_smallint;
begin
  date_end = current_date;
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  for select k_id, k_kontract_no, ks_state, k_fact_disb_date
      from knclient, kontract , kontstate
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = k_id and
            k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state in (1, 2) and
            :date_end between ks_date_begin and ks_date_end
      into k_id, k_kontract_no, ks_state, k_fact_disb_date
  do
  begin
    if (ks_state = 1) then
    begin
      k_status = 'Активный';
    end
    else
    begin
      sum_zadol = 0;
      select sum(oko_os + oko_perc + oko_other + oko_sh_os + oko_sh_perc + oko_sh_other)
      from operkroff
      where oko_from_k_id = :k_id and
            oko_ol_is_active = 1
      into sum_zadol;
      sum_zadol = coalesce(sum_zadol, 0);

      if (sum_zadol > 0) then
        continue;

      k_status = 'Закрытый';
    end
    suspend;
  end
end
