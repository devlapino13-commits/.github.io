CREATE PROCEDURE AAREP_INVOICELIST(
    REP_YEAR INTEGER,
    REP_MONTH INTEGER)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REG_DATE_BEGIN DATE,
    REG_DATE_END DATE,
    REG_KPP_NO VARCHAR(20),
    REG_NALOG_ADDRESS VARCHAR(255),
    REG_INVOICE_MIN INTEGER,
    REG_INVOICE_MAX INTEGER,
    ADL_ID DECIMAL(18,0),
    REG_INVOICE_COUNT INTEGER)
AS
declare variable rep_dd date;
begin
  if (rep_month < 12) then
    rep_dd = cast('01.' ||(:rep_month + 1) || '.' || :rep_year as date) - 1;
  else
    rep_dd = cast('31.' || :rep_month || '.' || :rep_year as date);
  for select reg_unicode, reg_code, reg_name
      from region
      where reg_status = 3
      order by reg_unicode
      into :reg_id, :reg_code, :reg_name
  do
  begin
    reg_nalog_address = '';
    select rp_date_begin, rp_date_end, rp_kpp_no, rp_nalog_adl_id
    from regprops
    where rp_from_reg_id = :reg_id and
          rp_date_begin <= :rep_dd and
          rp_date_end >= :rep_dd
    into :reg_date_begin, reg_date_end, reg_kpp_no, adl_id;
    if (adl_id is not null) then
    begin
      select adl_address_str
      from addrlist
      where adl_id = :adl_id
      into :reg_nalog_address;
    end
    reg_invoice_min = 0;
    reg_invoice_max = 0;
    reg_invoice_count = 0;

    select distinct count(im_invoice_no), min(im_invoice_no), max(im_invoice_no)
    from invoicemain, invoicekont, kontcalcperc
    where im_period = (:rep_year * 100 + :rep_month) and
          im_id = ik_from_im_id and
          ik_from_k_id = kcp_from_k_id and
          kcp_period = (:rep_year * 100 + :rep_month) and
          kcp_from_reg_id = :reg_id and
          im_from_reg_id = :reg_id
    into :reg_invoice_count, :reg_invoice_min, :reg_invoice_max;

    if (reg_invoice_min is null) then
      reg_invoice_min = 0;
    if (reg_invoice_max is null) then
      reg_invoice_max = 0;
    if (reg_invoice_count is null) then
      reg_invoice_count = 0;

    if (reg_date_end >= '31.12.2999') then
      reg_date_end = null;
    suspend;
  end
end
