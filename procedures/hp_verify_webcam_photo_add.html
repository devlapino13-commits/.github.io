CREATE PROCEDURE HP_VERIFY_WEBCAM_PHOTO_ADD(
    K_ID D_ID,
    C_ID D_ID,
    JSON_ANSWER D_STR_8100,
    STATUS D_INTEGER,
    UN_ID D_ID,
    WCAM_PHOTO D_BLOB_BINARY NOT NULL,
    VID_TYPE D_BOOL = 0)
AS
declare variable grs_bls_id d_id;
declare variable wcam_bls_id d_id;
declare variable acl_id d_id;
declare variable reg_id d_integer;
declare variable rd_id d_integer;
begin
  reg_id = (select k_last_reg_id from kontract where k_id = :k_id);
  rd_id = f_get_rd_id('ActionGroup.Photo_Verify_Issue');
  grs_bls_id = (select bls_id from get_grs_photo(:c_id, 1));

  execute procedure hp_insert_into_blobstore (
    'wcam_' || :c_id || '_' || :k_id || '.jpg',
    'vci',
    :wcam_photo,
    :un_id,
    0,
  0)
  returning_values :wcam_bls_id;


  if (status = 0) then
    insert into actionlist(
      acl_from_reg_id,
      acl_object_kind,
      acl_object_id,
      acl_from_rd_id,
      acl_from_un_id)
    values (
      :reg_id,
      9,
      :k_id,
      :rd_id,
      :un_id)
    returning acl_id
    into
      :acl_id;

  insert into verify_webcam_photo (
    vwp_k_id,
    vwp_c_id,
    vwp_type,
    vwp_json_answer,
    vwp_wcam_bls_id,
    vwp_grs_bls_id,
    vwp_acl_id,
    vwp_status,
    vwp_un_id)
  values (
    :k_id,
    :c_id,
    :vid_type,
    :json_answer,
    :wcam_bls_id,
    :grs_bls_id,
    :acl_id,
    :status,
    :un_id);
end
