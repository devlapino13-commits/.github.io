CREATE PROCEDURE HP_MAKE_SCHEDULE_PERC(
    LOAN_AMOUNT DECIMAL(18,4),
    LOAN_REPAY_KOLICH INTEGER,
    LOAN_PRODUCT_ID INTEGER,
    LOAN_PERCENT DECIMAL(18,7),
    LOAN_KOMMIS_PERC DECIMAL(18,7),
    LOAN_KOMMIS_LGOTA DECIMAL(18,4),
    LOAN_ADD_MONTH D_INTEGER = 0,
    LOAN_CLI_ID D_ID = -1,
    NEED_BASE D_BOOL = 0,
    MA_INSUR_KIND D_SMALLINT = 0)
AS
declare variable pr_percent_for_days integer;
declare variable pr_is_365 integer;
declare variable calc_kommis_amount decimal(18,4) = 0;
declare variable calc_perc_for_day decimal(18,9);
declare variable calc_perc_for_month decimal(18,9);
declare variable calc_annu_for_perc decimal(18,4) = 0;
declare variable calc_period_real integer;
declare variable annu_perc double precision;
declare variable pr_week_or_month integer;
declare variable pr_wm_kolich integer;
declare variable pr_percent_method integer;
declare variable tmp_amount decimal(18,4) = 0;
declare variable sh_os_total decimal(18,4) = 0;
declare variable pr_round integer;
declare variable pr_round_perc integer;
declare variable pr_round_kommis integer;
declare variable total_grace integer;
declare variable sh_id integer;
declare variable sh_days integer;
declare variable sh_os decimal(18,4);
declare variable sh_perc decimal(18,4);
declare variable sh_grace integer;
declare variable new_balance decimal(18,4) = 0;
declare variable orig_balance decimal(18,4) = 0;
declare variable new_perc decimal(18,7);
declare variable annu decimal(18,4) = 0;
declare variable sss decimal(18,4) = 0;
declare variable p1 decimal(18,4) = 0;
declare variable p2 decimal(18,4) = 0;
declare variable kol integer;
declare variable total_days integer;
declare variable delta_days double precision;
declare variable days_list d_str_long;
declare variable base_days_list d_str_long;
declare variable pr_percent_kind d_integer;
declare variable apply_lgota_schedule d_bool;
declare variable db_id d_integer;
declare variable first_day_perc d_currency;
declare variable k_disb_date d_date;
declare variable pr_code d_str_100;
declare variable c_stage_month d_currency;
declare variable start_date d_date;
declare variable base_perc_for_day d_percent;
declare variable add_month_lgota d_smallint;
declare variable point d_integer;
declare variable perc_minus d_percent;
declare variable date_20250101 d_date;
declare variable date_20250710 d_date;
declare variable date_20260210 d_date;
declare variable insurance_min_amount d_integer;
declare variable ipf_insurance_lgota d_smallint;
declare variable ipf_first_perc_amount decimal(18,4);
declare variable ipf_transfer_per_payment decimal(18,4);
declare variable ipf_transfer_period_count d_integer;
declare variable ipf_first_sh_days integer;
declare variable ipf_last_payment_total decimal(18,4);
declare variable ipf_payment_delta decimal(18,4);
declare variable ipf_payment_correction decimal(18,4);
declare variable ipf_calc_step d_integer;
declare variable ipf_calc_max_step d_integer;
declare variable ma_insur_payment_kind_later d_smallint;
begin
  total_grace = 0;
  ipf_insurance_lgota = 0;
  ipf_first_perc_amount = 0;
  ipf_transfer_per_payment = 0;
  ipf_transfer_period_count = 0;
  ipf_last_payment_total = 0;
  ipf_payment_delta = 0;
  ipf_payment_correction = 0;
  ipf_calc_step = 1;
  ipf_calc_max_step = 1;
  date_20250101 = cast('01.01.2025' as date); -- дата для обновления
  date_20250710 = cast('10.07.2025' as date); 
  date_20260210 = cast('10.02.2026' as date);
  ma_insur_payment_kind_later =  f_enum_id('ma_insur_payment_kind', 'later');
  if (need_base is null) then
    need_base = 0;

  if (loan_product_id is null) then
    exit;

-- Тестовые данные вручную
--  loan_disb_date='02.04.2016';     -- Дата выдачи
--  loan_repay_day=0;                -- Предпочитаемая дата погашения
--  loan_repay_kolich=12;            -- Количество погашений
--  loan_product_id=173;             -- ID продукта
--  loan_amount=40000;               -- Сумма кредита
--  loan_percent=34.9;               -- Процентная ставка
--  loan_kommis_perc=6.7;            -- Ставка доп.процента

-- Либо можно заполнить через ID объекта
--  obj_kind=9;
--  obj_id=109010017709800201;

-- Получаем необходимые настройки кредитного продукта
  select pr_percent_for_days, pr_is_360, pr_week_or_month, pr_wm_kolich,
         pr_interest_method, pr_round, pr_round_perc, pr_round_admin,
         pr_percent_kind, pr_code, pr_add_month_lgota
  from product
  where pr_id = :loan_product_id
  into pr_percent_for_days, pr_is_365, pr_week_or_month, pr_wm_kolich,
       pr_percent_method, pr_round, pr_round_perc, pr_round_kommis,
       pr_percent_kind, pr_code, add_month_lgota;

  db_id = f_get_db_id();

  -- RFCKG-5388
  if (db_id = 41701 and
      f_get_const('kont_insurance_active') = 1 and
      pr_code containing '.ipf.' and
      pr_code not containing '.ignore_insurance.' and
      loan_repay_kolich > 1 and
      ma_insur_kind in (0, ma_insur_payment_kind_later)) then
  begin
    insurance_min_amount = cast(f_get_const('kont_insurance_ipf_min_amount') as d_integer);

    if (loan_amount >= :insurance_min_amount) then
    begin
      ipf_insurance_lgota = 1;

      update temp_schedule
      set sh_grace = 1
      where sh_id = 1;
    end
  end

  select min(sh_date), sum(sh_grace)
  from temp_schedule
  into k_disb_date,  :total_grace;
  if (total_grace is null) then
    total_grace = 0;

  if (k_disb_date is null) then
    exit;

  pr_round = -pr_round;
  pr_round_perc = -pr_round_perc;
  pr_round_kommis = -pr_round_kommis;

  if (coalesce(pr_percent_for_days, 0) = 0) then
    exit;

  if (loan_add_month is null) then
    loan_add_month = 0;

  if (loan_add_month = 0) then
    add_month_lgota = 0;
  if (db_id in (39801, 41701) and pr_code containing '.sugarmill.') then
  begin
    update temp_schedule
    set sh_os = 0,
        sh_perc = 0,
        sh_kommis = 0,
        sh_grace = 0,
        sh_kommis_monthly = 0
    where 1 = 1;

    sh_os_total = 0;
    for select sh_id, sh_date, sh_days, sh_os, sh_perc, sh_kommis, sh_grace, sh_kommis_monthly
    from temp_schedule
    where sh_id > 0
    order by sh_date
    as cursor cur
    do
    begin
      if (cur.sh_id < loan_repay_kolich) then
      begin
        if (cur.sh_id <= 6) then
          sh_os = round(loan_amount * 0.03,0); -- 3% от всей суммы
        else
          sh_os = round((loan_amount - sh_os_total) / (loan_repay_kolich - cur.sh_id + 1), 0); -- остальное равномерно
      end else
        sh_os = loan_amount - sh_os_total;

      sh_os_total = sh_os_total + sh_os;

      update temp_schedule
      set sh_os = :sh_os
      where current of cur;
    end

    exit;
  end

-- Вычисляем сумму доп.процента с учетом предоставленных скидок
  if (db_id not in (39801, 41701) or loan_amount <= 11000000)
  then
    calc_kommis_amount = round(loan_amount * loan_kommis_perc / 100.0, :pr_round_kommis);
  else
    calc_kommis_amount = round(cast(loan_amount as d_integer) * loan_kommis_perc / 100.0, :pr_round_kommis);
  calc_kommis_amount = calc_kommis_amount - loan_kommis_lgota;
  if (calc_kommis_amount < 0) then
    calc_kommis_amount = 0.0;

  update temp_schedule
  set sh_kommis = :calc_kommis_amount
  where sh_id = 0;

  calc_perc_for_day = round(cast(loan_percent as double precision) / cast(pr_percent_for_days as double precision), 9);
  base_perc_for_day = calc_perc_for_day;

  start_date = cast('08.09.2021' as date);
  apply_lgota_schedule = 0;
  if (pr_percent_kind = 1 and db_id = 81001 and k_disb_date >= start_date) then
    apply_lgota_schedule = 1;

  first_day_perc = 0;
  if (apply_lgota_schedule = 1) then
  begin
    -- расчет для надежного сотрудника
    if (pr_code containing '.reliable_emp.') then
    begin
      c_stage_month = 0;
      select emp_stage_month
      from hp_get_employer_stage(:loan_cli_id, :k_disb_date)
      into c_stage_month;
      c_stage_month = coalesce(c_stage_month, 0);
      calc_perc_for_day = case
                             when c_stage_month >= 36.0 then 2.5
                             when c_stage_month >= 24.0 then 3.0
                             else 3.5
                           end;
      first_day_perc = round(loan_amount * round(cast(7.5 - calc_perc_for_day  as double precision) / 30.0, 9) / 100.0, :pr_round);

      calc_perc_for_day = round(calc_perc_for_day / 30, 9);
    end else
    if (pr_code containing '.summer_plus.') then
    begin
      first_day_perc = round(loan_amount * round(cast(2.5 as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
      calc_perc_for_day = round(((calc_perc_for_day * 30.0) - 2.5) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 2.5% в месяц
    end else
    if (pr_code containing '.comfortable.' and 1<>1) then
    begin
      first_day_perc = round(loan_amount * round(cast(2.2 as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
      calc_perc_for_day = round(((calc_perc_for_day * 30.0) - 2.2) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 2.2% в месяц
    end else
    if (pr_code containing '.zalog.') then
    begin
      point = f_point(loan_cli_id, :k_disb_date);

      if (k_disb_date < :date_20250101) then
      begin
        perc_minus = case
                       when point <= 6 then 0.4
                       when point <= 9 then 0.9
                       when point <= 12 then 1.4
                       else 1.7
                     end;
      end else
      if (k_disb_date < :date_20250710) then
      begin
        perc_minus = case
                       when point <= 6 then 0.4
                       else 0.9
                     end;
      end else
      if (k_disb_date < :date_20260210) then
      begin
        perc_minus = case
                       when point <= 70 then 1.0
                       else 0.9
                     end;
      end else
      begin
        perc_minus = case
                       when point <= 5 then 0.6
                       when point <= 70 then 1.0
                       else 0.9
                     end;
      end

      first_day_perc = round(loan_amount * round(cast(perc_minus as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
      calc_perc_for_day = round(((calc_perc_for_day * 30.0) - perc_minus) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 1.7% в месяц
    end else
    begin
      if (k_disb_date < :date_20250101) then
      begin
        first_day_perc = round(loan_amount * round(cast(1.7 as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
        calc_perc_for_day = round(((calc_perc_for_day * 30.0) - 1.7) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 1.7% в месяц
      end else
      if (k_disb_date < :date_20250710) then
      begin
        if (pr_code containing '.selfemployed.') then
        begin
          first_day_perc = round(loan_amount * round(cast(1.7 as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
          calc_perc_for_day = round(((calc_perc_for_day * 30.0) - 1.7) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 1.7% в месяц
        end else
        begin
          point = f_point(loan_cli_id, :k_disb_date);
          perc_minus = case
                         when point between 4 and 6 then 1.4
                         else 0.9
                       end;
          first_day_perc = round(loan_amount * round(cast(perc_minus as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
          calc_perc_for_day = round(((calc_perc_for_day * 30.0) - perc_minus) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 1.7% в месяц
        end 
      end else
      if (k_disb_date < :date_20260210) then
      begin
        point = f_point(loan_cli_id, :k_disb_date);
        perc_minus = case
                       when point <= 70 then 1.0
                       else 0.9
                     end;
        first_day_perc = round(loan_amount * round(cast(perc_minus as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
        calc_perc_for_day = round(((calc_perc_for_day * 30.0) - perc_minus) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 1.7% в месяц
      end else
      if (pr_code containing '.easy_start') then
      begin
        first_day_perc = round(loan_amount * round(cast(0.6 as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
        calc_perc_for_day = round(((calc_perc_for_day * 30.0) - 0.6) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 1.7% в месяц
      end else
      begin
        point = f_point(loan_cli_id, :k_disb_date);
        perc_minus = case
                       when point <= 5 then 0.6
                       when point <= 70 then 1.0
                       else 0.9
                     end;
        first_day_perc = round(loan_amount * round(cast(perc_minus as double precision) / 30.0, 9) / 100.0, :pr_round); -- Разница процента за один день менжду базовой и льготной ставкой
        calc_perc_for_day = round(((calc_perc_for_day * 30.0) - perc_minus) / 30, 9); -- Превращаем базовую ставку на льготную уменьшив на 1.7% в месяц
      end 
    end
  end

-- строим график погашения
-- PR_WEEK_OR_MONTH      1-Неделя, 2-Месяц
-- PR_WM_KOLICH          Количество

  calc_period_real = loan_repay_kolich - total_grace;

  if (ipf_insurance_lgota = 1 and
      calc_period_real > 0 and
      ma_insur_kind in (0, ma_insur_payment_kind_later)) then
  begin
    select first 1 sh_days
    from temp_schedule
    where sh_id = 1
    into :ipf_first_sh_days;

    ipf_first_sh_days = coalesce(ipf_first_sh_days, 0);

    ipf_first_perc_amount = round((cast(loan_amount as double precision) * calc_perc_for_day / 100.0) * cast(ipf_first_sh_days as double precision), :pr_round_perc);

    ipf_first_perc_amount = round(ipf_first_perc_amount * ((100.0000 - add_month_lgota) / 100.0000), pr_round_perc);

    if (apply_lgota_schedule = 1) then
      ipf_first_perc_amount = ipf_first_perc_amount + first_day_perc;

    ipf_transfer_period_count = loan_repay_kolich - 2;

    if (ipf_transfer_period_count < 1) then
      ipf_transfer_period_count = 1;

    ipf_transfer_per_payment = round(:ipf_first_perc_amount / :ipf_transfer_period_count, :pr_round_perc);
  end

  if (pr_percent_method = 2) then -- Если метод аннуитета, вычисляем сумму аннуитета
  begin
    if (pr_week_or_month = 2 and
        pr_wm_kolich = 1) then
    begin
      if (pr_is_365 = 0) -- Если ставка указывается за 360 дней, но график строим по календарным дням
      then
        calc_perc_for_month = round(calc_perc_for_day * 365.0 / 12.0 / 100.0, 9);
      else
        calc_perc_for_month = round(calc_perc_for_day * 30.0 / 100.0, 9);
    end
    else
    begin
      if (pr_week_or_month = 0) then
        delta_days = 1.0;
      if (pr_week_or_month = 1) then
        delta_days = 7.0;
      if (pr_week_or_month = 2) then
        delta_days = 30.0;
      delta_days = delta_days * pr_wm_kolich;
      calc_perc_for_month = round(calc_perc_for_day * delta_days / 100.0, 9);
    end
    annu_perc = (calc_perc_for_month * power((1 + calc_perc_for_month), calc_period_real)) / (power((1 + calc_perc_for_month), calc_period_real) - 1);
    calc_annu_for_perc = loan_amount * annu_perc;
    calc_annu_for_perc = round(calc_annu_for_perc, :pr_round);

    days_list = '';

    select list(iif(sh_grace = 0, sh_days, -sh_days))
    from temp_schedule
    where sh_id > 0
    into days_list;

    calc_annu_for_perc = f_annu(loan_amount, 0, 1, calc_perc_for_day, days_list, pr_round, pr_round_perc, loan_add_month, add_month_lgota, 0);
  end

  if (pr_percent_method = -1) then -- Если на первоначальную сумму и равные платежи
  begin
    total_days = 0;
    select sum(sh_days)
    from temp_schedule
    into :total_days;
    if (total_days is null) then
      total_days = 0;

    if (db_id not in (39801, 41701) or loan_amount <= 11000000)
    then
      annu_perc = loan_amount * calc_perc_for_day * cast(total_days as double precision) / 100.0;
    else
      annu_perc = cast(loan_amount as d_integer) * calc_perc_for_day * cast(total_days as double precision) / 100.0;
    calc_annu_for_perc = (loan_amount / calc_period_real) + (annu_perc / loan_repay_kolich);
    calc_annu_for_perc = round(calc_annu_for_perc, :pr_round);
  end

  if (ipf_insurance_lgota = 1 and
      pr_percent_method in (2, -1) and
      ma_insur_kind in (0, ma_insur_payment_kind_later)) then
  begin
    calc_annu_for_perc = calc_annu_for_perc + ipf_transfer_per_payment;
    ipf_calc_max_step = 3;
  end

  while (ipf_calc_step <= ipf_calc_max_step) do
  begin
    tmp_amount = loan_amount; -- здесь будет храниться база для начисления процентов
    sh_os_total = 0;

    for select sh_id, sh_grace, sh_days
        from temp_schedule
        where sh_id > 0
        order by sh_id
        into :sh_id, :sh_grace, sh_days
    do
    begin
      if (apply_lgota_schedule = 1 and sh_id = 2) then
      begin
        base_days_list = '';
        select list(iif(sh_grace = 0, sh_days, -sh_days))
        from temp_schedule
        where sh_id > 1
        into base_days_list;

        if (need_base = 1) then
        begin
          calc_annu_for_perc = f_annu(loan_amount - sh_os, 0, 1, base_perc_for_day, base_days_list, pr_round, pr_round_perc, loan_add_month, add_month_lgota, 0);
          calc_perc_for_day = base_perc_for_day;
        end else
        if (sh_os <= first_day_perc) then
          calc_annu_for_perc = f_annu(loan_amount - sh_os, 0, 1, calc_perc_for_day, base_days_list, pr_round, pr_round_perc, loan_add_month, add_month_lgota, 0);
      end

      sh_perc = round((cast(tmp_amount as double precision) * calc_perc_for_day / 100.0) * cast(sh_days as double precision), :pr_round_perc);

      -- если первое погашение через 2-3 месяца, то скидка по процентам на 50% на первом платеже
      if (sh_id = 1) then
      begin
        sh_perc = round(sh_perc * ((100.0000 - add_month_lgota) / 100.0000), pr_round_perc);

        if (apply_lgota_schedule = 1) then
          sh_perc = sh_perc + first_day_perc;
      end

      -- RFCKG-5388
      -- Для ИПФ со страхованием первый платеж льготный: ОС и наценка в первом месяце не отображаются
      if (ipf_insurance_lgota = 1 and
          sh_id = 1 and
          ma_insur_kind in (0, ma_insur_payment_kind_later)) then
        sh_perc = 0;

      -- RFCKG-5388
      -- Наценка первого льготного месяца полностью переносится во второй платеж
      -- до расчета ОС, чтобы итоговая сумма второго платежа не была завышена
      if (ipf_insurance_lgota = 1 and
          sh_id = 2 and
          ma_insur_kind in (0, ma_insur_payment_kind_later)) then
        sh_perc = round(:sh_perc + :ipf_first_perc_amount, :pr_round_perc);

      sh_os = 0;
      if (sh_grace = 0 or sh_id = loan_repay_kolich) then
      begin
        if (pr_percent_method in (0, 1)) then
          sh_os = round(loan_amount / calc_period_real, :pr_round); -- ОС равными долями
        else
        begin
          sh_os = calc_annu_for_perc - sh_perc; -- ОС исходим из суммы аннуитета (общего платежа)

          if (sh_id = 1 and apply_lgota_schedule = 1) then
            sh_os = sh_os + first_day_perc;
        end
      end

      if (sh_os < 0) then
        sh_os = 0;

      if (sh_id < loan_repay_kolich) then
        sh_os_total = sh_os_total + sh_os;
      else
        sh_os = loan_amount - sh_os_total;

      update temp_schedule
      set sh_os = :sh_os,
          sh_perc = :sh_perc
      where sh_id = :sh_id;

      if (pr_percent_method in (1, 2)) -- если на остаток или аннуитет
      then
        tmp_amount = tmp_amount - sh_os;
    end

    if (ipf_insurance_lgota = 1 and
        pr_percent_method in (2, -1) and
        ipf_calc_step < ipf_calc_max_step and
        ma_insur_kind in (0, ma_insur_payment_kind_later)) then
    begin
      select sh_os + sh_perc + coalesce(sh_kommis_monthly, 0)
      from temp_schedule
      where sh_id > 0 and
            sh_grace = 0
      order by sh_id desc
      rows 1
      into :ipf_last_payment_total;

      ipf_last_payment_total = coalesce(ipf_last_payment_total, 0);
      ipf_payment_delta = :ipf_last_payment_total - :calc_annu_for_perc;

      if (abs(:ipf_payment_delta) >= 1) then
      begin
        ipf_payment_correction = round(:ipf_payment_delta / :calc_period_real, :pr_round);

        if (ipf_payment_correction <> 0) then
          calc_annu_for_perc = calc_annu_for_perc + ipf_payment_correction;
        else
          ipf_calc_step = ipf_calc_max_step;
      end
      else
        ipf_calc_step = ipf_calc_max_step;
    end

    ipf_calc_step = ipf_calc_step + 1;
  end

  if (calc_kommis_amount > 0) then
  begin
    new_balance = loan_amount - calc_kommis_amount;
    orig_balance = loan_amount;

    select res_percent
    from hp_calc_real_percent(:orig_balance, :calc_kommis_amount, :calc_period_real, :calc_perc_for_day)
    into :new_perc;
    new_perc = round(new_perc / 30.0, 9);

    for select sh_id, sh_os, sh_os + sh_perc
        from temp_schedule
        where sh_id > 0
        order by sh_id
        into :sh_id, sh_os, annu
    do
    begin
      p1 = 0;
      p2 = 0;
      if (sh_os <> 0) then
      begin
        kol = 30;
        p1 = round(cast(orig_balance as double precision) / 100 * calc_perc_for_day * kol, pr_round_perc);
        p2 = round(cast(new_balance as double precision) / 100 * new_perc * kol, pr_round_perc);
        orig_balance = orig_balance - (annu - p1);
        new_balance = new_balance - (annu - p2);
        if (p2 < p1) then
          p2 = p1;
        sss = sss + (p2 - p1);
        if (sss > calc_kommis_amount) then
        begin
          p2 = p2 - (sss - calc_kommis_amount);
          sss = calc_kommis_amount;
        end
      end

      update temp_schedule
      set sh_kommis_monthly = :p2 - :p1
      where sh_id = :sh_id;
    end
    if (sss <> 0) then
    begin
      update temp_schedule
      set sh_kommis_monthly = sh_kommis_monthly + (:calc_kommis_amount - :sss)
      where sh_id = :sh_id;
    end
  end
  else
  begin
    update temp_schedule
    set sh_kommis_monthly = 0
    where 1 = 1;
  end

end
