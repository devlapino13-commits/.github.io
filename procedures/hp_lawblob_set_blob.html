CREATE PROCEDURE HP_LAWBLOB_SET_BLOB(
    LAM_ID D_ID,
    BLS_BLOB D_BLOB_BINARY,
    BLS_FILENAME D_STR_100,
    BLS_SRC_SYSTEM D_SMALLINT,
    BLS_KIND D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    BLS_ID BIGINT)
AS
declare variable db_id d_id;
begin
  db_id = f_get_db_id();
  bls_id = 0;
  if (lam_id is not null) then
  begin
    if (exists(select *
               from lawblob lab
               where lab.lab_from_lam_id = :lam_id and
                     (lab.lab_from_bls_id is not null) and
                     lab.lab_is_delete = 0)) then
      exception e_message 'Для данного судебного дела уже закреплен документ.';

    execute procedure hp_insert_into_blobstore (:bls_filename, 'lawblob_' || :lam_id, :bls_blob, :un_id, :bls_kind, :bls_src_system)
    returning_values :bls_id;

    begin
      insert into lawblob (lab_from_lam_id, lab_from_bls_id, lab_from_un_id)
      values (:lam_id, :bls_id, :un_id);
    end
    when any do
      exception e_message 'Добавление документа не удалось.';
  end
  else
  begin
    exception e_message 'Не выбрано судебное дело для прикрепления файла.';
  end
  suspend;
end
