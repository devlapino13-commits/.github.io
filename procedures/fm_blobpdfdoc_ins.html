CREATE PROCEDURE FM_BLOBPDFDOC_INS(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    SAVE_KIND D_STR_CODE,
    PAGE_COUNT D_SMALLINT,
    ADD_PARAMS D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    BPD_ID D_INTEGER)
AS
declare variable src_kind d_smallint; /* Вид источника: 0 - Фото с камеры, 2 - Галерея */
begin
  src_kind = f_json_str_get_value(add_params, 'src_kind', 2);

  insert into blobpdfdoc (bpd_object_kind, bpd_object_id, bpd_save_kind,
    bpd_page_count, bpd_src_kind, bpd_from_un_id)
  values (:obj_kind, :obj_id, :save_kind, :page_count, :src_kind, :un_id)
  returning bpd_id
  into bpd_id;

  suspend;
end
