CREATE PROCEDURE HP_UXFRAME_SQL_LOOKUP(
    UFP_CODE D_STR_100)
RETURNS (
    SQL_TEXT D_STR_8100)
AS
declare variable ulp_id_field d_str_code;
declare variable ulp_text_field d_str_255;
declare variable ulp_source d_str_code;
declare variable ulp_is_table d_bool;
declare variable ulp_param_name d_str_code;
declare variable ulp_param_list d_str_long;
declare variable ufp_origin_name d_str_code;
begin
  sql_text = '';

  select ufp_origin_name, ulp_id_field, ulp_text_field, ulp_source, iif(ulp_source = ulp_table_name, 1, 0)
  from uxfrparam, uxfrlookup, uxlookup
  where ufp_code = :ufp_code and
        ufl_ufp_code = ufp_code and
        ufl_ulp_code = ulp_code
  into ufp_origin_name, ulp_id_field, ulp_text_field, ulp_source, ulp_is_table;

  if (ulp_id_field <> '' and ulp_text_field <> '' and ulp_source <> '') then
  begin
    sql_text = f_format('select %1 as id, %2 as name from %3', ulp_id_field, ulp_text_field, ulp_source);

    if (ulp_is_table = 0) then
    begin
      ulp_param_list = '';
      for select ufp_origin_name
      from uxfrparam
      where ufp_parent_code = :ufp_code
      order by ufp_index
      into ulp_param_name
      do
      begin
        if (ulp_param_list <> '') then
          ulp_param_list = ulp_param_list || ',';

        ulp_param_list = ulp_param_list || ':' || ufp_origin_name || '$' || ulp_param_name;
      end
      if (ulp_param_list <> '') then
        sql_text = sql_text || '(' || ulp_param_list || ')';
    end
  end
  suspend;
end
