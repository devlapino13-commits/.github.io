CREATE PROCEDURE AAREP_TAXP_NSP_OPERBOOK(
    REG_ID D_INTEGER,
    DATE_END D_DATE)
RETURNS (
    TAX_AMOUNT D_CURRENCY,
    TAX_REGION D_STR_100,
    TAX_REGION_ID D_INTEGER,
    TAX_BRANCH_ID D_INTEGER,
    TAX_AMOUNT_BALANCE D_CURRENCY,
    TAX_AMOUNT_DIFF D_CURRENCY,
    ACC_EXP D_STR_20,
    ACC_NSP D_STR_20,
    LLC_MEMO D_STR_CODE,
    IS_IPF D_INTEGER)
AS
declare variable db_id d_integer;
declare variable date_beg d_date;
declare variable exp_balance d_str_20;
declare variable exp_acc_no d_integer;
declare variable nsp_balance d_str_20;
declare variable nsp_acc_no d_integer;
declare variable reg_code d_str_10;
declare variable product_kind d_str_20;
declare variable acc_tmp d_str_20;
declare variable is_special d_bool;
begin
  db_id = f_get_db_id();
  if (db_id in (39801, 41701)) then
  begin
    date_beg = dateadd(-extract(day from :date_end) + 1 day to :date_end);
    for
      select
        office_id,
        office_name,
        aarep_product_kind,
        iif(:date_end >= '30.06.2023' and lkc_status <= 4, 0, 1),
        sum(llc_sum_perc)
      from aarep_llc_calced_perc_cli(:reg_id, :date_beg, :date_end)
        inner join region_list(:reg_id, 1, 3, 1)
          on id_list = office_id
      group by 1, 2, 3, 4
    into
      :tax_region_id,
      :tax_region,
      :product_kind,
      :is_special,
      :tax_amount
    do
    begin
      is_ipf = iif(:product_kind = 'ипф', 1, 0);
      reg_code = left(:tax_region, 4);
      tax_branch_id = tax_region_id;
      llc_memo = 'Формирование РППУ ' || iif(is_ipf = 1, 'по наценке', 'на проценты') || ' за ' || f_monthname(:date_end);

      exp_balance = iif(:is_ipf = 1, '80433', '80431');
      nsp_balance = iif(:is_ipf = 1, iif(:is_special = 0, '11032', '11031'), iif(:is_special = 0, '11002', '11001'));

      exp_acc_no = 2;
      nsp_acc_no = 2;

      acc_exp = :reg_code || '417' || :exp_balance || '00000' || :exp_acc_no;
      acc_nsp = :reg_code || '417' || :nsp_balance || '00000' || :nsp_acc_no;

      tax_amount_balance = -f_al_bal(acc_nsp, date_end);
      tax_amount_diff = tax_amount - tax_amount_balance;
      if (tax_amount_diff < 0) then
      begin
        tax_amount_diff = -tax_amount_diff;
        acc_tmp = acc_exp;
        acc_exp = acc_nsp;
        acc_nsp = acc_tmp;
      end
      suspend;
    end
  end

  if (db_id = 94401) then
  begin

    date_beg = dateadd(-extract(day from :date_end) + 1 day to :date_end);

    for select id_list, r1.reg_code || ' ' || r1.reg_name
        from region_list(:reg_id, 1, 3, 1)
        inner join region r1 on r1.reg_unicode = id_list

        into :tax_region_id, :tax_region
    do
    begin
      tax_amount = 0;


      select sum(cast(lk_calcperc as double precision) * lk_percent / 100)
      from llc_main, llc_kont, kontstate, region_list(:tax_region_id, 1, 3, 1)
      where lm_is_signed = 1 and
            lm_id = lk_from_lm_id and
            :date_end between lk_date and lk_date_end and
            lk_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            :date_end between ks_date_begin and ks_date_end and
            lk_from_reg_id = id_list
      into :tax_amount;

      is_ipf = iif(:product_kind = 'ипф', 1, 0);
      reg_code = left(:tax_region, 4);
      tax_branch_id = tax_region_id;
      llc_memo = 'Формирование РППУ ' || iif(is_ipf = 1, 'по наценке', 'на проценты') || ' за ' || f_monthname(:date_end);

      exp_balance = '89150';
      nsp_balance = '21910';

      exp_acc_no = 1;
      nsp_acc_no = 2;

      acc_exp = :reg_code || '944' || :exp_balance || '00000' || :exp_acc_no;
      acc_nsp = :reg_code || '944' || :nsp_balance || '00000' || :nsp_acc_no;

      tax_amount_balance = -f_al_bal(acc_nsp, date_end);
      tax_amount_diff = tax_amount - tax_amount_balance;
      if (tax_amount_diff < 0) then
      begin
        tax_amount_diff = -tax_amount_diff;
        acc_tmp = acc_exp;
        acc_exp = acc_nsp;
        acc_nsp = acc_tmp;
      end

      if (tax_amount > 0 or  tax_amount_balance > 0) then
        suspend;
    end

  end
end
