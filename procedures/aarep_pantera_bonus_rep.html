CREATE PROCEDURE AAREP_PANTERA_BONUS_REP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    PROJECT_TYPE SMALLINT)
RETURNS (
    K_ID DECIMAL(18,0),
    OPER_CALLED VARCHAR(100),
    OK_SUM DECIMAL(18,4),
    KF_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(50))
AS
declare variable k_from_kf_id decimal(18,0);
declare variable pant_date_begin date;
declare variable pant_date_end date;
declare variable pant_id decimal(15,2);
declare variable pant_last_call_date date;
declare variable is_ok smallint;
declare variable sbl_id integer;
declare variable ok_sum_temp decimal(18,4);
declare variable ok_ol_date date;
begin
  /*
    Логика отчета:

    Для пантеры считаю выплаченные суммы, только если оператор звонил (и дозвнился) клиенту или его родственникам до даты выплаты
    Для СБ считаю выплаченные суммы вне зависимости от наличия звонков клиенту
    */

  for select pantk.pant_id, pantk.pant_from_k_id, pantk.pant_oper_called, kont.k_kontract_no, k_from_kf_id,
             pantk.pant_date_begin, pantk.pant_date_end, pantk.pant_last_call_date
      from pantera_kontracts pantk
      inner join kontract kont on kont.k_id = pantk.pant_from_k_id
      where pantk.pant_date_end >= :rep_date_begin -- чтобы вычислить у кого группа была на руках
            and
            ((pantk.pant_cl_call_qty > 0 or
            pantk.pant_cont_call_qty > 0) /*  or (:PROJECT_TYPE = 2)*/ ) -- если клиенту звонил оператор ранее даты начала отчета, все равно он подлежит дальнейшей проверке
            and
            pantk.pant_date_end >= pant_date_begin -- исключаю передачи в один и тот же день
            and
            pantk.pant_project_type = :project_type /* 1 - КЦ, 2 - СБ Айбек, 3 - СБ Эльдар*/
      order by pant_oper_called
      into :pant_id, :k_id, :oper_called, :k_kontract_no, :k_from_kf_id, :pant_date_begin, :pant_date_end,
           :pant_last_call_date
  do
  begin
    is_ok = 0;
--if (:PROJECT_TYPE = 1) then
--begin
    if (pant_last_call_date <= :rep_date_end) then
      is_ok = 1;
--end
--else  is_ok = 1;

    if (is_ok = 0) then
    begin
      if (exists(select *
                 from pantera_clients pc
                 where pc.pcl_from_pan_id = :pant_id and
                       pc.pcl_res > 0 and
                       pc.pcl_called_date <= :rep_date_end rows 1)) then
        is_ok = 1;
    end
    if (is_ok = 0) then
    begin
      if (exists(select *
                 from pantera_contacts pcon
                 where pcon.pcon_from_pan_id = :pant_id and
                       pcon.pcon_res > 0 and
                       pcon.pcon_called_date <= :rep_date_end rows 1)) then
        is_ok = 1;
    end

-- если выявлено, что действительно, был соврершен успешный звонок клиенту или конт. лицу (.._res > 0), тогда  ....
    if (is_ok = 1) then
    begin
      if (pant_date_begin < rep_date_begin) then
        pant_date_begin = rep_date_begin;
      if (pant_date_end > rep_date_end) then
        pant_date_end = rep_date_end;

      select kal_new_kf_name
      from knaktlist
      where knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal_from_kf_id = :k_from_kf_id
      into kf_name;

      if (project_type = 1) then
        sbl_id = 82;
      if (project_type = 2) then
        sbl_id = 90;
      if (project_type = 3) then
        sbl_id = 87;

      ok_sum = 0;
      ok_sum_temp = 0;

      for select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine), ok.ok_ol_date
          from operkred ok
          where ok.ok_type = 4 and
                ok.ok_ol_is_active = 1 and
                ok.ok_ol_date between :pant_date_begin and :pant_date_end and
                ok.ok_from_k_id = :k_id and
                ok_from_c_id = 0
          group by ok.ok_ol_date
          order by ok.ok_ol_date
          into :ok_sum_temp, :ok_ol_date
      do
      begin
        if (ok_sum_temp is null) then
          ok_sum_temp = 0;
        if (exists(select *
                   from sb_akt sba
                   where sba.sba_from_k_id = :k_id and
                         :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                         sba.sba_from_sbl_id = :sbl_id)) then
          ok_sum = ok_sum + ok_sum_temp;
      end

      if (ok_sum is null) then
        ok_sum = 0;

      if (ok_sum <> 0) then
      begin
        suspend;
      end
    end
  end
end
