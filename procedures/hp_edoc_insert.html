CREATE PROCEDURE HP_EDOC_INSERT(
    EDOC_KIND D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID,
    LANG D_STR_10 = null)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable edl_object_kind d_integer; /* un_id */
declare variable edoc_id d_integer;
declare variable edc_state_str d_str_code;
declare variable c_id d_id;
declare variable edl_blank_date d_date;
declare variable blank_count d_integer;
declare variable edc_name d_str_255;
declare variable edc_date_begin d_date;
declare variable edc_date_end d_date;
declare variable db_id d_integer;
declare variable asp_status d_str_code;
declare variable c_name_fio d_str_100;
declare variable edc_state d_smallint;
declare variable reg_id d_integer;
declare variable tmp_text d_str_255;
declare variable full_text d_str_8100;
declare variable asp_fio_list d_str_8100;
declare variable cont_fio_list d_str_8100;
declare variable edc_use_last_contdata d_smallint;
declare variable res_is_error d_smallint;
declare variable edc_id d_integer;
declare variable delay_min d_integer;
declare variable edc_object_uniq d_smallint;
declare variable max_edl_status d_smallint;
declare variable edn_edl_id d_id;
declare variable edn_status d_smallint;
declare variable edc_perstemp_kind d_smallint;
declare variable v_context d_str_255;
begin
  -- Входящий параметр edoc_kind может быть передан как кодовое имя конфигурации (eDoc_cfg.edc_code),
  -- так и просто ID конфигурации (eDoc_cfg.edc_id).
  -- Следующие две строки заполняют переменные ID и код независимо от того, как передан параметр

  res_text = '';
  delay_min = 5;

  edoc_id = f_edoc_kind(edoc_kind, 'id');
  if (edoc_id is null) then Exit;

  if (obj_kind = 'edoc_conf') then
  begin
    -- Электронный документ создается после одобрения предварительного запроса на создание
    -- В эту категорию входят эл.документы у которых есть бланки edoc_frx.edf_prepare_kind = 6
    select edn_edl_id, edn_object_kind, edn_object_id, edn_create_un_id, edn_status
    from edoc_conf
    where edn_edl_id = :obj_id
    into edn_edl_id, obj_kind, obj_id, un_id, edn_status;

    if (row_count = 0) then
      res_text = 'Запрос на формирование эл.документа не найден';

    if (edn_status <> 1) then
      res_text = 'Запрос на формирование документа не подтвержден';

    if (exists(select * from edoc_list where edl_id = :edn_edl_id)) then
      res_text = 'По данному запросу эл.документ уже сформирован';
  end

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  -- Тип объекта в конфигурации и переданный obj_kind должны совпадать.
  select edc_object_kind, edc_date_begin, edc_date_end, edc_name, edc_state,
    edc_use_last_contdata, edc_object_uniq, edc_code, edc_perstemp_kind
  from edoc_cfg
  where edc_id = :edoc_id
  into edl_object_kind, edc_date_begin, edc_date_end, edc_name, edc_state,
    edc_use_last_contdata, edc_object_uniq, edoc_kind, edc_perstemp_kind;

  obj_kind = f_objkind(obj_kind, 'id');

  -- Если тип переданного объекта и тип из конфигурации не совпадает, выдаем ошибку
  if (obj_kind <> f_objkind(edl_object_kind, 'id')) then
    exception e_default 'Incorrect obj_kind. Expected: ' ||
              f_objkind(edl_object_kind, 'name') || ', actual: ' || f_objkind(obj_kind, 'name');

  -- Если проект отключен, завершаем работу
  edc_state_str = f_edoc_state(edoc_id, current_date, null, obj_kind, obj_id);

  -- Если проект отключен, завершаем работу
  if (edc_state_str = 'inactive') then exit;

  edl_blank_date = null;
  select blank_date, reg_id
  from hp_edoc_object_info(:obj_kind, :obj_id, :un_id, null)
  into edl_blank_date, reg_id;

  -- по-идее edoc_kind нужно передавать в hp_edoc_object_info
  if (edoc_kind in ('loan.pdn.recalc','loan.debt', 'loan.cancel', 'loan.refinance', 'ipf.chdp.agreement', 'loan.delinq.notif', 'restruct_application', 'add_loan_consent', 'restruct_approve', 'loan.creditor_rep', 'autorestruct_notify', 'risk.agreement', 'restruct.loan.delinq.notif')) then
    edl_blank_date = current_date;

  -- если проект действующий, проверяем дату документа на соответствие периоду действия проекта
  -- если проект режиме тестирования, даты не проверяем.
  if (edc_state_str = 'active' and
      edl_blank_date not between edc_date_begin and coalesce(edc_date_end, edl_blank_date)) then Exit;

  -- Договор АСП подписывается только один раз
  if (edoc_kind in ('asp.agreement', 'asp.agreement_employee', 'asp.agreement_oep') and
      f_edoc_status(edoc_kind, obj_id, current_date) = 'signed') then
  begin
    res_text = edc_name || ' ' || 'уже подписан';
    suspend;
    exit;
  end
  if (edoc_kind = 'asp.agreement' and f_grs_active(:un_id) = 1) then
  begin
  if (not exists(select * from clients c
                 where c.c_id = :obj_id and
                       c.c_nation <> 0)) then
                        exception e_message 'Национальность не указана, прошу указать национальность и переотправить ЭЦП';
  end
  if (exists(select *
  from edoc_list
  where edl_from_edc_id = :edoc_id and
        edl_object_kind = :obj_kind and
        edl_object_id = :obj_id and
        edl_status <> -1 and
        datediff(minute, edl_create_datetime, current_timestamp) <= :delay_min) and un_id <> 1000000000000) then
  begin
    res_text = f_format('После последнего создания документа %1 прошло менее %2 минут, пожалуйста, подождите.', edc_name, delay_min);
    suspend;
    exit;
  end

  -- Получаем список подписывающих лиц, для данного объекта
  -- если объект Kontract, возвращает список из KnClient,
  -- если объект Miniapp, возвращает список из MiniappClient и т.д.
  -- если нужно подтягивать залогодателей, будет доработана процедура hp_edoc_sign_clients

  db_id = f_get_db_id();

  --RFCKG-2742
  if (db_id in (39801, 41701)) then
  begin
    v_context  = rdb$get_context('SYSTEM','CLIENT_PROCESS');
    --Если происходит создание нового эл. док-та эцп через ОЭП не через КМП тогда отрабатывает блок
    if (edoc_kind = 'asp.agreement_oep' and
        exists(select *
               from edoc_list edl
               inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id and
                     edc.edc_code = 'asp.agreement_oep'
               where edl.edl_object_kind = :obj_kind and
                     edl.edl_object_id = :obj_id) and
        (v_context containing 'Finstart' or v_context containing 'FsMobileSvc'))
    then
    begin
      res_text = 'Данный электронный документ создается только в клиентском мобильном приложении MBulak';
      suspend;
      exit;
    end
  end 

  asp_fio_list = '';
  cont_fio_list = '';
  full_text = '';
  for select c_id
  from hp_edoc_sign_clients(:edoc_kind, :obj_kind, :obj_id, '')
  into :c_id
  do
  begin
    tmp_text = '';
    c_name_fio = '';
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    -- Проверка верификации и данных клиента
    if (db_id = 81001 and edoc_kind <> 'loan.creditor_rep') then --RFCRU-6059
    begin
      for select c_name_fio, c_birth_date, c_pass_no, c_pass_who, c_pass_date, c_verify_status
      from clients
      where c_id = :c_id
      as cursor cur_c
      do
      begin
        res_text = '';
        if (coalesce(cur_c.c_name_fio, '') = '') then
          res_text = left(res_text || 'Не указано ФИО клиента. ', 255);
        if (cur_c.c_birth_date is null) then
          res_text = left(res_text || 'Не указана дата рождения. ', 255);
        if (coalesce(cur_c.c_pass_no, '') = '') then
          res_text = left(res_text || 'Не указан номер паспорта. ', 255);
        if (coalesce(cur_c.c_pass_who, '') = '') then
          res_text = left(res_text || 'Не указан орган выдавший паспорт. ', 255);
        if (cur_c.c_pass_date is null) then
          res_text = left(res_text || 'Не указана дата выдачи паспорта клиента. ', 255);
  
        if (cur_c.c_verify_status = -2) then
          res_text = left(res_text || 'Верификация:данные не верны. ', 255);
        else if (cur_c.c_verify_status = 0) then
          res_text = left(res_text || 'Верификация:еще не отправлен. ', 255);
        else if (cur_c.c_verify_status = 1 and edoc_kind not in ('cell.application', 'cell.application.no_access')) then -- RFCRU-3592 -- RFCRU-4632
          res_text = left(res_text || 'Верификация:в процессе. ', 255);

        if (res_text = '' and edc_perstemp_kind = -1 and edoc_kind not in ('cell.application', 'cell.application.no_access')) then -- RFCRU-3592 -- RFCRU-4632
        begin
          if (f_tmp_const('use_perstemp') = '1') then
          begin
            if (exists(select 1
                       from hp_perstemp_diff(:c_id))) then
              res_text = left(res_text || 'ПерсДанные не подтвеждены. ', 255);
          end
        end
      end
      if (res_text <> '') then
      begin
        res_text = left(f_format('%1: %2', c_name_fio,  res_text), 255);
        suspend;
        exit;
      end
    end

    -- RFCRU-4948
    if (edoc_kind not in ('asp.agreement', 'bankcard.moneta', 'bankcard.elcart', 'storage_agreement', 'asp.agreement_employee', 'asp.agreement_oep', 'pretrial_claim') and
       (edc_state = 3 or (edc_state = 2 and f_tmp_const('edoc_allow_in_draft_without_asp') <> 1)) and
       (db_id = 41701 or edoc_kind  not in ('cell.application', 'cell.application.no_access')) and
       db_id <> 94401 and
       db_id <> 39801) then
    begin
      -- в боевом режиме или в режиме тестирования на уровне регионов,
      -- без наличия подписанного договора АСП (ЭЦП) не даем создать другой электронный документ
      -- В АЗ АСП/ЭЦП не подписывается

      asp_status = f_edoc_status('asp.agreement', c_id, current_date);

      if(asp_status <> 'signed') then
        asp_status = f_edoc_status('asp.agreement_oep', c_id, current_date);

      if (db_id in (39801, 41701)) then
      begin
        if (f_asp_signed(:c_id, current_date) <> 1) then
        begin
          tmp_text = c_name_fio;
          if (asp_fio_list <> '') then
            asp_fio_list = asp_fio_list || ', ';

          asp_fio_list = asp_fio_list || c_name_fio;
        end
      end
      else
      if (asp_status <> 'signed' and edoc_kind <> 'loan.creditor_rep') then  --RFCRU-5419
      begin
        tmp_text = c_name_fio;
        if (asp_fio_list <> '') then
          asp_fio_list = asp_fio_list || ', ';

        asp_fio_list = asp_fio_list || c_name_fio;
      end
    end else
    if (edoc_kind in ('asp.agreement', 'asp.agreement_employee', 'asp.agreement_oep')) then
    begin
      select edc_id
      from edoc_cfg
      where edc_code = 'cell.application' and
            edc_state = 3
      into edc_id;

      if (edc_id is not null) then
      begin
        if (exists(select * from edoc_list
        where edl_object_kind = f_objkind('clients', 'id') and
              edl_object_id = :c_id and
              edl_from_edc_id = :edc_id and
              edl_status = 0)) then
        begin
          tmp_text = c_name_fio;
          if (asp_fio_list <> '') then
            asp_fio_list = asp_fio_list || ', ';
  
          asp_fio_list = asp_fio_list || c_name_fio;
        end
      end
    end

    if (tmp_text = '') then
    begin
      select res_text
      from hp_edoc_client_continfo(:edoc_kind, :c_id, :un_id)
      into :tmp_text;

      if (tmp_text <> '') then
      begin
        if (cont_fio_list <> '') then
          cont_fio_list = cont_fio_list || ', ';

        cont_fio_list = cont_fio_list || c_name_fio;
      end
    end
  end

  if (asp_fio_list <> '') then
  begin
    if (edoc_kind not in ('asp.agreement', 'asp.agreement_employee', 'asp.agreement_oep'))
    then full_text = f_format('Нет договора %1. ФИО: %2', iif(db_id = 81001, 'АСП', 'ЭЦП'), asp_fio_list);
    else full_text = f_format('Нет подписанного заявления на изменение номера. ФИО: %2', asp_fio_list);
  end

  if (cont_fio_list <> '') then
  begin
    if (full_text <> '') then
      full_text = full_text || '.  ';

    if (edc_use_last_contdata = 1)
    then full_text = full_text || 'Нет основного номера: ' || cont_fio_list;
    else full_text = full_text || 'Нет подтвержденного номера: ' || cont_fio_list;
  end

  if (full_text <> '') then
  begin
    if (char_length(full_text) <= 255)
    then res_text = full_text;
    else res_text = left(full_text, 252) || '...';

    suspend;
    exit;
  end

  if (edc_object_uniq in (1, 2)) then
  begin
    -- Если на объект создается и подписывается только один эл. документ,
    -- и есть подписанный эл. документ, запрещаем создание нового,
    -- т.к. он аннулирует подписанный. Вместо этого требуется явное аннулирование

    select max(edl_status)
    from edoc_list
    where edl_object_kind = :obj_kind and
          edl_object_id = :obj_id and
          edl_from_edc_id = :edoc_id and
          edl_status = 1
    into max_edl_status;
    max_edl_status = coalesce(max_edl_status, -1);

    if (max_edl_status = 1) then
    begin
      res_text = 'Документ уже подписан, создание нового запрещено.';
      suspend;
      exit;
    end

    if (edc_object_uniq = 2 and max_edl_status = 0) then
    begin
      res_text = 'Существует неподписаный документ, создание нового запрещено.';
      suspend;
      exit;
    end
  end


  -- аннулируем другие документы для данного объекта
  execute procedure hp_edoc_cancel(:edoc_kind, :obj_kind, :obj_id);

  if (edn_edl_id is null) then
  begin
    -- создаем новый
    insert into edoc_list (edl_object_kind, edl_object_id, edl_from_edc_id, edl_blank_date, edl_from_reg_id, edl_create_un_id)
    values (:obj_kind, :obj_id, :edoc_id, :edl_blank_date, :reg_id, :un_id)
    returning edl_id
    into :edl_id;
  end else
  begin
    -- создаем с существующим Id согласно одобренного предварительного запроса (edoc_conf)
    insert into edoc_list (edl_id, edl_object_kind, edl_object_id, edl_from_edc_id, edl_blank_date, edl_from_reg_id, edl_create_un_id)
    values (:edn_edl_id, :obj_kind, :obj_id, :edoc_id, :edl_blank_date, :reg_id, :un_id);

    edl_id = edn_edl_id;
  end

  -- для всех подписывающих лиц создаем запись для отслеживания подписания данного документа (edl_id)
  for select c_id
  from hp_edoc_sign_clients(:edoc_kind, :obj_kind, :obj_id, '')
  into :c_id
  do
  begin
    insert into edoc_sign (eds_from_edl_id, eds_from_c_id)
    values (:edl_id, :c_id);

    -- если документ является основанием для подписание перс данных
    if (edc_perstemp_kind = 1 and f_tmp_const('use_perstemp') = '1') then
    begin
      update perstemp
      set pte_edoc_for_approve = :edl_id
      where pte_from_c_id = :c_id and
            pte_edoc_for_approve is distinct from :edl_id;
    end
  end

  -- Закидываем бланки в очередь для формирования pdf сервисом (FsService)
  -- После приготовления всех бланков, в триггере edoc_list_aiu_sign будут оправлены сообщения подписывающим
  select blank_count
  from hp_edoc_blank_ins(:edl_id, :lang)
  into :blank_count;
  if (blank_count is null) then blank_count = 0;

  if (res_text = '' and edoc_kind = 'cell.application' and f_get_const('asp_const') = 0) then
  begin
    -- если это заявка на изменение номера, требуется ручное подписание
    select res_is_error, res_text
    from hp_edoc_manual_sign_request(:edl_id, 1, :un_id)
    into :res_is_error, :res_text;

    if (res_is_error = 0) then
      res_text = '';
  end

  -- если в данном проекте нет действующих бланков,
  -- или они были ранее приготовлены и ничего не изменилось
  if (blank_count = 0) then
  begin
    -- В триггере вызовется hp_edoc_msg_insert, и сообщения будут отправлены
    update edoc_list
    set edl_blank_status = 1
    where edl_id = :edl_id;
  end

  suspend;
end
