CREATE PROCEDURE FS_REPORT_HTML_MAIN(
    UN_ID DECIMAL(18,0),
    REP_ID VARCHAR(255),
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE,
    UN_IP VARCHAR(20) = 0)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
begin
  -- Процедура возвращает информацию в формате HTML для отображения отчета
  -- в Finstart. Является аналогом процедуры fm_report_html_main
  -- Выделена отдельно, т.к. была проблема с раздачей кучи прав для Read Only

  -- Логирование запроса
  execute procedure hp_fsdatalog_update(:un_id, :rep_id, :obj_kind, :obj_id, :date_begin, :date_end, :un_ip, 0, 0);

  if (rep_id = 'loan_in_progress_status') then
  begin
    select res_html
    from hp_html_get_loan_status(9, :obj_id, 'full', :un_id)
    into res_text;
    suspend;
    exit;
  end

  if (rep_id = 'extinfo_equifax_fs') then
  begin
    select res_html
    from fm_html_extinfo_equifax_fs(:obj_id, :un_id)
    into res_text;
    suspend;
    exit;
  end

end
