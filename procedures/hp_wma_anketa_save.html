CREATE PROCEDURE HP_WMA_ANKETA_SAVE(
    WMA_ID D_INTEGER,
    CLIENT_IN_COMPANY_STATUS STR_100,
    CLIENT_LANGUAGE STR_100,
    GENDER D_SMALLINT,
    FIO STR_100,
    PHONE STR_100,
    PHONE_ADD STR_100,
    CITIZENSHIP STR_100,
    CITY_ID ID_TYPE,
    CITY STR_100,
    PASS_NO STR_100,
    BIRTH_DATE D_DATE,
    DOCLIST STR_255,
    WORK_STATUS STR_100,
    CURRENT_LOAN_MBF D_SMALLINT,
    ACTIVE_LOAN_OTHER_COMPANY D_SMALLINT,
    LOAN_PRODUCT STR_100,
    LOAN_PERIOD D_INTEGER,
    LOAN_AMOUNT D_INTEGER,
    MEETING_POINT STR_100,
    MEETING_ADDRESS STR_255,
    MEETING_DATE D_DATE,
    OFFICE_ID D_INTEGER,
    ANKETA_MEMO D_STR_8100,
    CALL_BACK_DATE D_DATETIME,
    CLOSE_REASON STR_100,
    REFUSAL_REASON STR_100,
    UN_ID ID_TYPE,
    C_ID ID_TYPE,
    INFO_SOURCE STR_100,
    INFO_SOURCE_TEXT STR_255,
    LOAN_PURPOSE STR_100)
AS
declare variable STATUS smallint;
declare variable WMAT_ID integer;
declare variable ERROR_TEXT d_str_255;
begin
  if (city_id = 0) then
    city_id = null;
  if (office_id = 0) then
    office_id = null;
  if (un_id = 0) then
    un_id = null;
  close_reason = coalesce(close_reason, '');
  status = 0;

  if (close_reason <> '') then -- используется только для негативных статусов
  begin
    status = -1;
    update or insert into cc_website_miniapp_close (wmc_from_wma_id, wmc_un_id, wmc_memo, wmc_closed_rd_id, wmc_refuse_id)
    values (:wma_id, :un_id, :anketa_memo, f_get_rd_id(:close_reason), f_get_rd_id(:refusal_reason))
    matching (wmc_from_wma_id);
  end

  insert into cc_website_miniapp_tmp (wmat_from_wma_id, wmat_client_in_company_status, wmat_language, wmat_gender,
                                      wmat_name, wmat_phone_add, wmat_citizenship, wmat_city_id, wmat_city,
                                      wmat_pass_no, wmat_birthday, wmat_doclist, wmat_work_status,
                                      wmat_current_loan_mbf, wmat_active_loan_othe_company, wmat_loan_product,
                                      wmat_loan_period, wmat_loan_amount, wmat_meeting_point, wmat_meeting_address,
                                      wmat_meeting_date, wmat_office, wmat_anketa_memo, wmat_call_back_date,
                                      wmat_from_un_id, wmat_info_source, wmat_info_source_text, wmat_loan_purpose)
  values (:wma_id, f_get_rd_id(:client_in_company_status), f_get_rd_id(:client_language), :gender, :fio, :phone_add,
          f_get_rd_id(:citizenship), :city_id, :city, :pass_no, :birth_date, :doclist, f_get_rd_id(:work_status),
          :current_loan_mbf, :active_loan_other_company, f_get_rd_id(:loan_product), :loan_period, :loan_amount,
          f_get_rd_id(:meeting_point), :meeting_address, :meeting_date, :office_id, :anketa_memo, :call_back_date,
          :un_id, f_get_rd_id(:info_source), :info_source_text, f_get_rd_id(:loan_purpose))
  returning wmat_id
  into :wmat_id;

  if ((office_id is not null) and
      (status = 0)) then
    status = 2;
  if ((call_back_date is not null) and
      (status = 0)) then
    status = 1;

  update cc_website_miniapp t
  set wma_last_un_id = :un_id,
      wma_blocked_un_id = null,
      wma_cant_call_qty = 0,
      wma_last_wmat_id = :wmat_id,
      wma_status = :status,
      wma_send_to_office_datetime = iif(:office_id is not null, current_timestamp, null),
      wma_phone_no = :phone
  where (wma_id = :wma_id);

  if (status = 2) then
  begin
    if (:c_id = -1) then -- если клиент не найден и требуется создать нового
    begin
      select c_id
      from cc_do_add_client(0, :un_id, :fio, :gender, :birth_date, f_get_rd_id(:client_language), f_get_rd_id(:citizenship), :phone, :phone_add)
      into :c_id;
    end
    else
    begin
      -- Сохранение номера телефона
      if (coalesce(phone, '') <> '') then
      begin
        begin
          if (f_cont_info('clients', :c_id, 'cell_main', 'ok') <> :phone) then
            execute procedure hp_continfo_update(:c_id, 'contactinfo.cell_main', :phone, 0, :un_id);
          when any do
            error_text = 'не удалется сохранить основной номер телефона';
        end
      end

      if (coalesce(phone_add, '') <> '') then
      begin
        begin
            if (f_cont_info('clients', :c_id, 'cell_add', 'last') <> :phone_add) then
                execute procedure hp_continfo_update(:c_id, 'contactinfo.cell_add', :phone_add, 0, :un_id);
            when any do
                error_text = 'не удалется сохранить дополнительный номер телефона';
        end
      end
    end

    update cc_website_miniapp t
    set wma_from_c_id = :c_id
    where (wma_id = :wma_id);
  end
end
