CREATE PROCEDURE AAREP_CHECK_FINPOLIS_TMP(
    REP_C_ID D_ID,
    IS_ALL_CHECK D_BOOL = 0)
RETURNS (
    FP_NUMBER D_STR_20,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_CODE,
    C_INN D_STR_20,
    C_ADDRESS D_STR_255,
    C_FINPOLIS_SEND_DATE D_DATE,
    FOUND_PASS_NO D_SMALLINT,
    FOUND_INN_NO D_SMALLINT,
    FOUND_CLIENT_FIO D_SMALLINT,
    FOUND_BIRTH_DATE D_SMALLINT,
    FOUND_ADDRESS_LEGAL D_SMALLINT,
    FOUND_ADDRESS_FACT D_SMALLINT,
    FOUND_ADDRESS_BIRTH D_SMALLINT,
    C_FINPOLIS_PROCEED D_SMALLINT,
    TERR_CATEG D_STR_20,
    FP_NAME D_STR_255,
    FP_GR D_DATE)
AS
declare variable ADDRESS_LEGAL D_STR_255;
declare variable ADDRESS_FACT D_STR_255;
declare variable C_INDEX_C_PASS_NO D_STR_CODE;
declare variable INDEXED_ADDRESS D_STR_255;
declare variable ADC_ADDRESS_TYPE D_INTEGER;
declare variable ADL_ADDRESS_STR D_STR_255;
declare variable DB_ID D_INTEGER;
declare variable FP_ID D_ID;
declare variable FP_FOUND D_BOOL;
begin
  db_id = f_get_db_id();

  if (coalesce(:rep_c_id, 0) = 0) then
  begin
    for with cte_finpolis
        as (select fp_id, number, gr, amr as fp_reg_address, adress as fp_state_address, mr as fp_birth_address,
                   sd_rg_concat as fp_pass_no, nd as fp_inn
            from finpolis
            where not exists(
                               select 1
                               from refdata rd
                               where rd.rd_id = finpolis.terr_categ_id
                                 and rd.rd_main_code = 'FionpolisDirectory'
                                 and coalesce(rd.rd_memo,'') = iif(:is_all_check = 1, '#all#', 'soft_check')
                            )
            ),
        cte_no_address
        as (select fp.fp_id, fp.number as fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_finpolis_send_date,
                   c_finpolis_proceed, iif(fp.gr = cl.c_birth_date, 1, 0) as found_birth_date, 1 as kind
            from clients cl
            inner join cte_finpolis fp on cl.c_index_c_pass_no = fp.fp_pass_no and
                  fp.fp_pass_no > ''

            union all

            select fp.fp_id, fp.number as fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_finpolis_send_date,
                   c_finpolis_proceed, iif(fp.gr = cl.c_birth_date, 1, 0) as found_birth_date, 2 as kind
            from clients cl
            inner join cte_finpolis fp on cl.c_inn = fp.fp_inn and
                  fp.fp_inn > ''

            union all

            select fp.fp_id, fp.number as fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_finpolis_send_date,
                   c_finpolis_proceed, iif(fp.gr = cl.c_birth_date, 1, 0) as found_birth_date, 3 as kind
            from clients cl, finpolisparced fpp, cte_finpolis fp, fiolist fl1, fiolist fl2
            where fpp.fpp_from_fp_id = fp.fp_id
              and fl1.fl_id = fpp.fpp_from_fl_id
              and fl2.fl_id in (cl.c_from_fl_id, cl.c_name_eng_fl_id, cl.c_alternat_fl_id)
              and fl1.fl_name1 = fl2.fl_name1
              and fl1.fl_name2 = fl2.fl_name2
              and (fl1.fl_name3 = fl2.fl_name3 or (:db_id = 81001 and fp.gr = cl.c_birth_date)) -- В РФ кроме ФИ должно совпасть отчество или ДР
              and fpp.fpp_fio > ''

            union all

            select fp.fp_id, fp.number as fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_finpolis_send_date,
                   c_finpolis_proceed, iif(fp.gr = cll.cll_birth_date, 1, 0) as found_birth_date, 3 as kind
            from clients cl, clientlog cll, finpolisparced fpp, cte_finpolis fp, fiolist fl1, fiolist fl2
            where fpp.fpp_from_fp_id = fp.fp_id
              and cll.cll_from_c_id = c_id
              and fl1.fl_id = fpp.fpp_from_fl_id
              and fl2.fl_id = cll.cll_from_fl_id
              and fl1.fl_name1 = fl2.fl_name1
              and fl1.fl_name2 = fl2.fl_name2
              and (fl1.fl_name3 = fl2.fl_name3 or (:db_id = 81001 and fp.gr = cl.c_birth_date)) -- В РФ кроме ФИ должно совпасть отчество или ДР
              and fpp.fpp_fio > ''

            union all

            -- Проверка clients через f_levenstain2, дата рождения должна совпадать
            select fp.fp_id, fp.number as fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_finpolis_send_date,
                   c_finpolis_proceed, iif(fp.gr = cl.c_birth_date, 1, 0) as found_birth_date, 3 as kind
            from clients cl, finpolisparced fpp, cte_finpolis fp, fiolist fl1, fiolist fl2
            where fpp.fpp_from_fp_id = fp.fp_id
              and fl1.fl_id = fpp.fpp_from_fl_id
              and fp.gr = cl.c_birth_date
              and fl2.fl_id in (cl.c_from_fl_id, cl.c_name_eng_fl_id, cl.c_alternat_fl_id)
              and f_levenstain2(fl1.fl_name1, fl2.fl_name1, 2) < 3
              and f_levenstain2(fl1.fl_name2, fl2.fl_name2, 2) < 3
              and (:db_id = 81001 or f_levenstain2(fl1.fl_name3, fl2.fl_name3, 2) < 3)
              and fpp.fpp_fio > ''
            ),
        cte_address
        as (select fp.fp_id, adc.adc_from_c_id, adl.adl_address_str, adl.adl_indexed_address, 4 as kind, fp.gr,
                   fp.number as fp_number
            from addrlist adl
            inner join cte_finpolis fp on (hash(adl.adl_indexed_address) = hash(fp.fp_reg_address) and
                  adl.adl_indexed_address = fp.fp_reg_address)
            inner join addrclient adc on adc.adc_from_adl_id = adl.adl_id
            where adl.adl_is_filled = 1 and
                  adc.adc_address_type = 1 and
                  fp.fp_reg_address > '' and
                  fp.fp_reg_address similar to '%[[:DIGIT:]]%'

            union all

            select fp.fp_id, adc.adc_from_c_id, adl.adl_address_str, adl.adl_indexed_address, 5 as kind, fp.gr,
                   fp.number as fp_number
            from addrlist adl
            inner join cte_finpolis fp on (hash(adl.adl_indexed_address) = hash(fp.fp_state_address) and
                  adl.adl_indexed_address = fp.fp_state_address)
            inner join addrclient adc on adc.adc_from_adl_id = adl.adl_id
            where adl.adl_is_filled = 1 and
                  adc.adc_address_type = 2 and
                  fp.fp_state_address > '' and
                  fp.fp_state_address similar to '%[[:DIGIT:]]%'
        )

        select fp_id, fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, adl_address_str, c_finpolis_send_date,
               c_finpolis_proceed, max(iif(kind = 1, 1, 0)) as found_pass_no, max(iif(kind = 2, 1, 0)) as found_inn_no,
               max(iif(kind = 3, 1, 0)) as found_client_fio, max(iif(kind = 4, 1, 0)) as found_address_legal,
               max(iif(kind = 5, 1, 0)) as found_address_fact, max(iif(kind = 6, 1, 0)) as found_address_birth,
               found_birth_date
        from (select fp_id, fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, adl.adl_address_str,
                     c_finpolis_send_date, c_finpolis_proceed, found_birth_date, kind
              from cte_no_address cl
              inner join addrclient adc on cl.c_id = adc.adc_from_c_id and
                    adc_address_type = 1
              inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id

              union all

              select fp_id, fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, adl.adl_address_str,
                     c_finpolis_send_date, c_finpolis_proceed, iif(adr.gr = cl.c_birth_date, 1, 0) as found_birth_date,
                     kind
              from cte_address adr
              inner join clients cl on adr.adc_from_c_id = cl.c_id
              inner join addrclient adc on cl.c_id = adc.adc_from_c_id and
                    adc_address_type = 1
              inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id

        )
        group by fp_id, fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, adl_address_str, c_finpolis_send_date, c_finpolis_proceed, found_birth_date
        into fp_id, fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_address, c_finpolis_send_date,
             c_finpolis_proceed, found_pass_no, found_inn_no, found_client_fio, found_address_legal,
             found_address_fact, found_address_birth, found_birth_date
    do
    begin
      select nameu, terr_categ, gr
      from finpolis
      where fp_id = :fp_id
      into fp_name, terr_categ, fp_gr;

      select left(rd.rd_name, 20)
      from refdata rd
      where rd.rd_main_code = 'FionpolisDirectory' and
            rd.rd_str_id = :terr_categ
      rows 1
      into terr_categ;

      if (not exists (
        select *
        from finpolis_excludes
        where
          fpe_from_c_id = :c_id and
          fpe_nameu = :fp_name and
          fpe_gr = :fp_gr and
          fpe_is_delete = 0))
      then
        suspend;
    end
  end
  else
  begin
    -- Если клиент ранее проверен, не проверяем дальше
    if (exists(select *
               from finpolis_checked
               where fpc_from_c_id = :rep_c_id)) then
      exit;

    address_legal = '';
    address_fact = '';
    fp_found = 0;
    select c_id, coalesce(c_name_fio, '') as c_name_fio, c_birth_date, coalesce(c_pass_no, '') as c_pass_no,
           coalesce(c_index_c_pass_no, '') as c_index_c_pass_no, coalesce(c_inn, '') as c_inn, c_finpolis_send_date,
           coalesce(c_finpolis_proceed, 0) as c_finpolis_proceed
    from clients cl
    where c_id = :rep_c_id
    into c_id, c_name_fio, c_birth_date, c_pass_no, c_index_c_pass_no, c_inn, c_finpolis_send_date,
         c_finpolis_proceed;

    for select adl.adl_indexed_address, adc.adc_address_type, adl_address_str
        from addrclient adc
        inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id
        where adc.adc_from_c_id = :rep_c_id and
              adc.adc_address_type in (1, 2)
        into indexed_address, adc_address_type, adl_address_str
    do
    begin
      if (:adc_address_type = 1) then
      begin
        c_address = :adl_address_str;
        address_legal = :indexed_address;
      end
      else
        address_fact = :indexed_address;
    end

    for with cte_finpolis
        as (select fp_id, number, gr, amr as fp_reg_address, adress as fp_state_address, mr as fp_birth_address,
                   sd_rg_concat as fp_pass_no, nd as fp_inn
            from finpolis
            where not exists(
                               select 1
                               from refdata rd
                               where rd.rd_id = finpolis.terr_categ_id
                                 and rd.rd_main_code = 'FionpolisDirectory'
                                 and coalesce(rd.rd_memo,'') = iif(:is_all_check = 1, '#all#', 'soft_check')
                            )
            )

        select fp_id, fp_number, max(iif(kind = 1, 1, 0)) as found_pass_no, max(iif(kind = 2, 1, 0)) as found_inn_no,
               max(iif(kind = 3, 1, 0)) as found_client_fio, max(iif(kind = 4, 1, 0)) as found_address_legal,
               max(iif(kind = 5, 1, 0)) as found_address_fact, max(iif(kind = 6, 1, 0)) as found_address_birth,
               max(found_birth_date)
        from (select fp_id, fp.number as fp_number, iif(fp.gr = :c_birth_date, 1, 0) as found_birth_date, 1 as kind
              from cte_finpolis fp
              where fp.fp_pass_no = :c_index_c_pass_no and
                    fp.fp_pass_no > ''

              union all

              select fp.fp_id, fp.number as fp_number, iif(fp.gr = :c_birth_date, 1, 0) as found_birth_date, 2 as kind
              from cte_finpolis fp
              where fp.fp_inn = :c_inn and
                    fp.fp_inn > ''

              union all

              select fp_id, fp.number as fp_number, iif(fp.gr = :c_birth_date, 1, 0) as found_birth_date, 3 as kind
              from clients cl, finpolisparced fpp, cte_finpolis fp, fiolist fl1, fiolist fl2
              where fpp.fpp_from_fp_id = fp.fp_id
                and cl.c_id = :rep_c_id
                and fl1.fl_id = fpp.fpp_from_fl_id
                and fl2.fl_id in (cl.c_from_fl_id, cl.c_name_eng_fl_id, cl.c_alternat_fl_id)
                and fl1.fl_name1 = fl2.fl_name1
                and fl1.fl_name2 = fl2.fl_name2
                and (fl1.fl_name3 = fl2.fl_name3 or (:db_id = 81001 and fp.gr = cl.c_birth_date)) -- В РФ кроме ФИ должно совпасть отчество или ДР
                and fpp.fpp_fio > ''

              union all
  
              select fp_id, fp.number as fp_number, iif(fp.gr = cll.cll_birth_date, 1, 0) as found_birth_date, 3 as kind
              from clientlog cll, finpolisparced fpp, cte_finpolis fp, fiolist fl1, fiolist fl2
              where fpp.fpp_from_fp_id = fp.fp_id
                and cll.cll_from_c_id = :c_id
                and fl1.fl_id = fpp.fpp_from_fl_id
                and fl2.fl_id = cll.cll_from_fl_id
                and fl1.fl_name1 = fl2.fl_name1
                and fl1.fl_name2 = fl2.fl_name2
                and (fl1.fl_name3 = fl2.fl_name3 or (:db_id = 81001 and fp.gr = cll.cll_birth_date)) -- В РФ кроме ФИ должно совпасть отчество или ДР
                and fpp.fpp_fio > ''


              union all
  
              -- Проверка clients через f_levenstain2, дата рождения должна совпадать
              select fp_id, fp.number as fp_number, 1 as found_birth_date, 3 as kind
              from clients cl, finpolisparced fpp, cte_finpolis fp, fiolist fl1, fiolist fl2
              where fpp.fpp_from_fp_id = fp.fp_id
                and fl1.fl_id = fpp.fpp_from_fl_id
                and cl.c_id = :rep_c_id
                and cl.c_birth_date = fp.gr
                and fl2.fl_id in (cl.c_from_fl_id, cl.c_name_eng_fl_id, cl.c_alternat_fl_id)
                and f_levenstain2(fl1.fl_name1, fl2.fl_name1, 2) < 3
                and f_levenstain2(fl1.fl_name2, fl2.fl_name2, 2) < 3
                and (:db_id = 81001 or f_levenstain2(fl1.fl_name3, fl2.fl_name3, 2) < 3)
                and fpp.fpp_fio > ''

              union all

              select fp_id, fp.number as fp_number, iif(fp.gr = :c_birth_date, 1, 0) as found_birth_date, 4 as kind
              from cte_finpolis fp
              where fp.fp_reg_address = :address_legal and
                    fp.fp_reg_address > '' and
                    fp.fp_reg_address similar to '%[[:DIGIT:]]%'

              union all

              select fp_id, fp.number as fp_number, iif(fp.gr = :c_birth_date, 1, 0) as found_birth_date, 5 as kind
              from cte_finpolis fp
              where fp.fp_state_address = :address_fact and
                    fp.fp_state_address > '' and
                    fp.fp_state_address similar to '%[[:DIGIT:]]%')
        group by fp_id, fp_number
        into fp_id, fp_number, found_pass_no, found_inn_no, found_client_fio, found_address_legal, found_address_fact,
             found_address_birth, found_birth_date
    do
    begin
      fp_found = 1;
      select nameu, terr_categ, gr
      from finpolis
      where fp_id = :fp_id
      into fp_name, terr_categ, fp_gr;

      select left(rd.rd_name, 20)
      from refdata rd
      where rd.rd_main_code = 'FionpolisDirectory' and
            rd.rd_str_id = :terr_categ
      rows 1
      into terr_categ;

      if (not exists (
        select *
        from finpolis_excludes
        where
          fpe_from_c_id = :c_id and
          fpe_nameu = :fp_name and
          fpe_gr = :fp_gr and
          fpe_is_delete = 0))
      then
        suspend;
    end
    if (fp_found = 0 and rdb$get_context('SYSTEM', 'READ_ONLY') = 'FALSE') then
    begin
      update or insert into finpolis_checked (fpc_from_c_id)
      values (:rep_c_id);
    when any do
      begin
        -- Ошибки из-за параллельной вставки игнорируем
      end
    end
  end
end
