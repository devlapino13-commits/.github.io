CREATE PROCEDURE HP_EXTMINIAPP_TIMEOUT_REJECT
AS
declare variable DB_ID D_INTEGER;
declare variable EMA_ID D_ID;
declare variable LO_ID ID_TYPE;
declare variable EMA_BLOCK_DATETIME D_DATETIME;
declare variable EMA_DATE D_DATE;
declare variable PS_IS_WORKING D_SMALLINT;
declare variable HRV_DATE_BEGIN D_DATE;
declare variable SENT_TEXT D_STR_LONG;
declare variable RD_ID D_INTEGER;
declare variable CLIENT_TEXT D_STR_255;
declare variable CHECK_DATE D_DATE;
declare variable COUNTER D_INTEGER;
begin
  db_id = f_get_db_id();

  update extminiapp ema
  set ema.ema_status = -1,
      ema.ema_result_rd_id = f_get_rd_id('website_miniapp.Refuse.DaysDelayCancel'),
      ema.ema_lo_id = 1000000000000
  where ema.ema_from_c_id is null and
        ema.ema_lo_id is null and
        ema.ema_from_mcl_id is not null and
        ema.ema_status = 0 and
        datediff(hour, ema.ema_date, current_timestamp) > 24;

  if (db_id = 81001) then
  begin
    rd_id = f_get_rd_id('LOPenaltyReason.Extminiapp_72_Hours');

    -- update const_var
    -- set cv_value = 1
    -- where (cv_code = 'extminiapp_no_push');

    rdb$set_context('USER_TRANSACTION', 'extminiapp_no_push', 1);

    for select ema_id, ema_lo_id, ema_block_datetime, ema_date, coalesce(c_name_fio, '')
        from extminiapp
        left join clients on ema_from_c_id = c_id
        where ema_block_datetime >= '01.06.2022 00:00:01' -- дата запуска проекта
              and
              ema_status = 0 and
              ema_lo_id is not null and
              ema_block_datetime is not null
        into ema_id, lo_id, ema_block_datetime, ema_date, client_text
    do
    begin
      select ps_is_working
      from hr_pstatus
      where ps_from_c_id = :lo_id and
            current_date between ps_date_begin and ps_date_end
      into ps_is_working;

      hrv_date_begin = null;

      if (ps_is_working = 0) then
      begin
        -- только разблокировка
        execute procedure fm_extminiapp_unblock(:ema_id);
      end
      if (ps_is_working = 1) then
      begin
        if (client_text <> '') then
          client_text = 'Клиент: ' || client_text;

        -- проверяем на нахождение в отпуске
        select hrv_date_begin
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_c_id = :lo_id and
              hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              hrv_status = 1 and
              current_date between hrv_date_begin and iif(hrv_date_end <> hrv_date_end_fact, hrv_date_end_fact, hrv_date_end) and
              sawt_base_type in (4, 5) and
              sawt_hours_real >= 0 and
              sawt_id <> 20 rows 1
        into hrv_date_begin;

        check_date = current_date;
        counter = 0; -- счетчик праздничных и выходных дней, в общей формуле будет домножаться на 24 часа

        while (check_date >= cast(ema_block_datetime as date)) do
        begin
          counter = counter + f_date_is_weekend(check_date, 0);
          check_date = check_date - 1;
        end

        if (hrv_date_begin is not null) then
        begin
          if (datediff(minute, ema_block_datetime, cast(hrv_date_begin as timestamp)) - 1440 * counter >= 4320) then
          begin
            -- штраф и разблокировка + пуши по штрафам и разблокировке
            execute procedure fm_extminiapp_unblock(:ema_id);

            sent_text = 'Вы пропустили сроки работы с заявкой, поэтому данная заявка разблокирована для других ВИПС.' || client_text;
            execute procedure hp_send_push_to_staff(:lo_id, 'Онлайн заявка разблокирована',
                :sent_text, 1000000000000, 'msg_to_staff', 7);

            if (ema_block_datetime >= '01.07.2022 00:00:00') then
            begin
              sent_text = 'Вы нарушили сроки обработки онлайн-заявки. Вам будет начислен штраф 500 рублей.' || client_text;
              insert into lo_penalty (lp_from_c_id, lp_days, lp_ma_date, lp_date, lp_datetime, lp_amount, lp_is_active,
                                      lp_reason, lp_from_ema_id)
              values (:lo_id, datediff(hour, :ema_block_datetime, cast(:hrv_date_begin as timestamp)) - 24 * :counter, :ema_date,
                      current_date, current_timestamp, 500, 1, :rd_id, :ema_id);

              execute procedure hp_send_push_to_staff(:lo_id, 'Нарушение сроков', :sent_text,
                  1000000000000, 'msg_to_staff', 7);
            end
          end
          if (datediff(minute, ema_block_datetime, cast(hrv_date_begin as timestamp)) - 1440 * counter < 4320) then
          begin
            -- разблокировка + пуш по разблокировке
            execute procedure fm_extminiapp_unblock(:ema_id);

            sent_text = 'Вы пропустили сроки работы с заявкой, поэтому данная заявка разблокирована для других ВИПС.' || client_text;
            execute procedure hp_send_push_to_staff(:lo_id, 'Онлайн заявка разблокирована',
                :sent_text, 1000000000000, 'msg_to_staff', 7);

          end
        end
        if (hrv_date_begin is null and
            datediff(minute, ema_block_datetime, current_timestamp) - 1440 * counter between 2880 and 2939) then
        begin
          -- пуш о приближающемся штрафе
          sent_text = 'Срок обработки онлайн-заявки, ' || client_text || ', истекает через 24 часа. За нарушение срока будет начислен штраф 500 руб.';
          execute procedure hp_send_push_to_staff(:lo_id,
              'Срок обработки онлайн-заявки истекает', :sent_text, 1000000000000,
              'msg_to_staff', 7);
        end
        if (hrv_date_begin is null and
            datediff(minute, ema_block_datetime, current_timestamp) - 1440 * counter >= 4320) then
        begin
          -- штраф и разблокировка  + пуши по штрафам и разблокировке
          execute procedure fm_extminiapp_unblock(:ema_id);

          sent_text = 'Вы пропустили сроки работы с заявкой, поэтому данная заявка разблокирована для других ВИПС.' || client_text;
          execute procedure hp_send_push_to_staff(:lo_id, 'Онлайн заявка разблокирована',
              :sent_text, 1000000000000, 'msg_to_staff', 7);

          if (ema_block_datetime >= '01.07.2022 00:00:00') then
          begin
            sent_text = 'Вы нарушили сроки обработки онлайн-заявки. Вам будет начислен штраф 500 рублей.' || client_text;
            insert into lo_penalty (lp_from_c_id, lp_days, lp_ma_date, lp_date, lp_datetime, lp_amount, lp_is_active,
                                    lp_reason, lp_from_ema_id)
            values (:lo_id, datediff(hour, :ema_block_datetime, current_timestamp) - 24 * :counter, :ema_date, current_date,
                    current_timestamp, 500, 1, :rd_id, :ema_id);

            execute procedure hp_send_push_to_staff(:lo_id, 'Нарушение сроков', :sent_text,
                1000000000000, 'msg_to_staff', 7);
          end
        end
      end
    end
    -- update const_var
    -- set cv_value = 0
    -- where (cv_code = 'extminiapp_no_push');

    rdb$set_context('USER_TRANSACTION', 'extminiapp_no_push', 0);
  end
end
