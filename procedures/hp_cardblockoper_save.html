CREATE PROCEDURE HP_CARDBLOCKOPER_SAVE(
    OFFICE_ID D_INTEGER,
    CAL_ID D_INTEGER,
    OPER_TYPE D_INTEGER,
    OPER_REASON D_INTEGER,
    UN_ID D_ID,
    C_ID D_ID,
    CBO_IS_HARD_STOPLIST D_BOOL)
AS
declare variable STATUS D_INTEGER;
declare variable HARD_STOPLIST_IN D_INTEGER;
declare variable STOPLIST_IN D_INTEGER;
begin
  hard_stoplist_in = f_enum_id('cardlist_status','hard_stoplist_in');
  stoplist_in = f_enum_id('cardlist_status','stoplist_in');

  select cal.cal_status
  from cardlist cal
  where cal.cal_card_id = :cal_id
  into :status;

  if (status = hard_stoplist_in) then
    exception e_default 'Карта находится в жестком стоп-листе';

  if (oper_type = 2 and status <> stoplist_in) then
    exception e_default 'Разблокировка запрещена. Карта не находится в мягком стоп-листе';

  if (oper_type = 1 and coalesce(cbo_is_hard_stoplist, 0) = 0 and status = stoplist_in) then
    exception e_default 'Карта уже заблокирована';

--Если есть активные операции
  if (exists(select 1
             from cardblockopers cbo
             where cbo.cbo_from_cal_id = :cal_id
               and cbo.cbo_status not in (-1, 3))) then
    exception e_default 'Существуют не завершенные операции';

  insert into cardblockopers (cbo_from_cal_id, cbo_from_un_id, cbo_type, cbo_reason, cbo_region_id, cbo_from_c_id, cbo_is_hard_stoplist)
  values (:cal_id, :un_id, :oper_type, :oper_reason, :office_id, :c_id, :cbo_is_hard_stoplist);
end
