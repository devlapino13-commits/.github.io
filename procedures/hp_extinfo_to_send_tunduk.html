CREATE PROCEDURE HP_EXTINFO_TO_SEND_TUNDUK(
    C_ID ID_TYPE,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_VALUE D_BLOB_TEXT,
    DRL_ID D_ID)
AS
declare variable pass_no str_100;
declare variable c_pass_no varchar(30);
declare variable c_inn varchar(20);
declare variable counter integer;
declare variable str_leng integer;
declare variable ch str_100;
declare variable is_finish smallint;
declare variable seria str_100;
declare variable number str_100;
declare variable inn str_100;
begin
  if (:object_kind <> f_objkind('mc_login', 'id')) then
  begin
    for select cl.c_pass_no, cl.c_inn
        from clients cl
        where cl.c_id = :c_id
        into :pass_no, :c_inn
    do
    begin
      ------------- разделяю серию и номер паспорта ---------------------------- >
      seria = '';
      pass_no = replace(pass_no, ' ', '');
      if (pass_no similar to '[-]?[[:DIGIT:]]+') then
      begin
        seria = substring(pass_no from 1 for 4);
        number = substring(pass_no from 5);
      end
      else
      begin
        counter = 1;
        is_finish = 0;
        str_leng = character_length(pass_no);
        while ((counter < str_leng - 1) and (is_finish = 0)) do
        begin
          ch = substring(pass_no from counter for 1);
          if (ch not similar to '[-]?[[:DIGIT:]]+') then
            seria = seria || ch;
          else
          begin
            number = substring(pass_no from counter);
            is_finish = 1;
          end
          counter = counter + 1;
        end
      end
      -------------------------------------------------------------------------- <
      json_value = '{' || f_json_make_pair('seriesNumber', :seria, 'str') || ', ' || --
      f_json_make_pair('idNum', :number, 'str') || ', ' || --
      f_json_make_pair('inn', :c_inn, 'str') || '}';
      suspend;
    end
  end
  else
  begin
    select drl.drl_id
    from docrecoglist drl
    where drl.drl_from_mcl_id = :object_id
    order by drl.drl_id desc
    rows 1
    into drl_id;

    pass_no = f_docrecog_value(:drl_id, 'doc_no', '');
    c_inn = f_docrecog_value(:drl_id, 'inn', '');

    ------------- разделяю серию и номер паспорта ---------------------------- >
    seria = '';
    pass_no = replace(pass_no, ' ', '');
    if (pass_no similar to '[-]?[[:DIGIT:]]+') then
    begin
      seria = substring(pass_no from 1 for 4);
      number = substring(pass_no from 5);
    end
    else
    begin
      counter = 1;
      is_finish = 0;
      str_leng = character_length(pass_no);
      while ((counter < str_leng - 1) and (is_finish = 0)) do
      begin
        ch = substring(pass_no from counter for 1);
        if (ch not similar to '[-]?[[:DIGIT:]]+') then
          seria = seria || ch;
        else
        begin
          number = substring(pass_no from counter);
          is_finish = 1;
        end
        counter = counter + 1;
      end
    end
    -------------------------------------------------------------------------- <
    json_value = '{' || f_json_make_pair('seriesNumber', :seria, 'str') || ', ' || --
    f_json_make_pair('idNum', :number, 'str') || ', ' || --
    f_json_make_pair('inn', :c_inn, 'str') || '}';
    suspend;
  end
end
