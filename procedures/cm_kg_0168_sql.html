CREATE PROCEDURE CM_KG_0168_SQL(
    MA_ID D_ID,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable PR_CODE D_STR_100;
declare variable POINT_CLS D_INTEGER;
declare variable POINT_IPF D_INTEGER;
declare variable OKS_BALANCE D_CURRENCY;
declare variable OKS_BALANCE_CLS D_CURRENCY;
declare variable OKS_BALANCE_IPF D_CURRENCY;
declare variable TOTAL_DEBT D_CURRENCY;
declare variable TOTAL_DEBT_CLS D_CURRENCY;
declare variable TOTAL_DEBT_IPF D_CURRENCY;
declare variable MAX_DEBT D_CURRENCY;
declare variable MAX_DEBT_CLS D_CURRENCY;
declare variable MAX_DEBT_IPF D_CURRENCY;
declare variable MAC_AMOUNT D_CURRENCY;
declare variable MA_KOLICH_REPAY D_INTEGER;
declare variable MA_DATE_OPER D_DATE;
declare variable MAX_OLD_AMOUNT_IPF D_CURRENCY;
declare variable AVAILABLE_AMOUNT D_CURRENCY;
declare variable AVAILABLE_AMOUNT_CLS D_CURRENCY;
declare variable AVAILABLE_AMOUNT_IPF D_CURRENCY;
declare variable TMP_PR_CODE D_STR_100;
declare variable TMP_SUM D_CURRENCY;
declare variable TMP_STR D_STR_100;
declare variable IS_IPF D_BOOL;
begin
  -- проверки блока 0168 (RFCKG: 3578, 3748):
  -- 1) общий совокупный долг не должен превышать общий максимально допустимый долг
  -- 2) типовой совокупный долг не должен превышать типовой максимально допустимый долг

  -- определения:
  -- а) типовой совокупный долг = баланс ОС активных кредитов типа продукта из новой заявки + сумма новой заявки
  -- б) общий совокупный долг = баланс ОС всех активных кредитов (независимо от типа продукта) + сумма новой заявки
  -- в) типовой максимально допустимый долг = максимально возможная сумма долга по одному типу продукта (классика или ИПФ)
  -- г) общий максимально допустимый долг = больший из двух типовых максимально допустимых долгов

  if (ma_id = 4285565) then -- RFCKG-2905
    exit;

  res_error_is_critical = 1;
  res_error_text = '';

  for select ma_kolich_repay,
             ma_date_oper,
             pr_code,
             mac_from_c_id,
             mac_amount
      from miniapp
      inner join miniappclient on mac_from_ma_id = ma_id
      inner join product on pr_id = ma_from_pr_id
      where ma_id = :ma_id and
            ma_is_active = 1 and
            ma_date_end = '31.12.9999' and
            mac_amount > 0
      into ma_kolich_repay,
           ma_date_oper,
           pr_code,
           c_id,
           mac_amount
  do
  begin
    if (pr_code containing '.kardhasan.' or
        pr_code containing '.bittamlik.' or
        pr_code containing '.mbulak_cards.' or
        pr_code containing '.preferential.' or
        pr_code containing '.solemar.' or
        pr_code containing '.july.' or
        pr_code containing '.umra.' or
        pr_code containing '.fertilizer.' or
        pr_code containing '.sugarmill.' or
        pr_code containing '.installment_goods.' or
        pr_code containing '.business.' or
        pr_code containing '.express_online.' or
        (pr_code containing '.point_ignore.' and
        pr_code not containing '.july.')) then
      exit;

    if (pr_code containing '.ipf.') then
      is_ipf = 1;
    else
      is_ipf = 0;

    select c_name_fio
    from clients
    where c_id = :c_id
    into res_info_text;

    point_cls = f_point(c_id, ma_date_oper, 0); -- баллы клиента по классике, для расчета сумм и сроков классических кредитов, берутся общие баллы (RFCKG-3251)
    point_ipf = f_point(c_id, ma_date_oper, 2); -- баллы клиента по ИПФ

    if (pr_code containing '.july.' and
        pr_code containing '.standart.') then
    begin
      if (mac_amount <> 99999 or
          point_cls < 1) then
      begin
        res_error_text = 'Сумма выдачи и балл клиента не соответствует по данному продукту.';
        suspend;
      end
      exit;
    end

    oks_balance = 0; -- общий активный долг
    oks_balance_cls = 0; -- активный долг по классике
    oks_balance_ipf = 0; -- активный долг по ИПФ

    total_debt = 0; -- общий совокупный долг
    total_debt_cls = 0; -- совокупный долг по классике
    total_debt_ipf = 0; -- совокупный долг по ИПФ

    max_debt = 0; -- общий максимально допустимый долг
    max_debt_cls = 0; -- максимально допустимый долг по классике
    max_debt_ipf = 0; -- максимально допустимый долг по ИПФ

    available_amount = 0; -- доступная сумма всего
    available_amount_cls = 0; -- доступная сумма по классике
    available_amount_ipf = 0; -- доступная сумма по ИПФ

    max_old_amount_ipf = 0; -- максимальная сумма в предыдущих кредитах по ИПФ

    if (is_ipf = 0 and pr_code not containing '.sputnik.') then  --RFCKG-4838
    begin
      -- заявка по классике, проверка сроков кредита
      if (point_cls < 20 and
          ma_kolich_repay not between 3 and 18) then
      begin
        res_error_text = 'Балл клиента по классике - ' || point_cls || '. Cрок кредита должен быть в диапазоне 3-18 мес.';
        suspend;
      end
      --RFCKG-3745
      /*else if (point_cls >= 20 and
          mac_amount < 250001 and
          ma_kolich_repay not between 3 and 24) then
      begin
        res_error_text = 'Балл клиента по классике - ' || point_cls || '. Сумма кредита - ' || mac_amount || '. Cрок кредита должен быть в диапазоне 3-24 мес.';
        suspend;
      end */
      else if (point_cls >= 20 and
          ma_kolich_repay not between 3 and 36) then
      begin
        res_error_text = 'Балл клиента по классике - ' || point_cls || '. Сумма кредита - ' || mac_amount || '. Cрок кредита должен быть в диапазоне 3-36 мес.';
        suspend;
      end
    end

    if (is_ipf = 1) then
    begin
      -- заявка по ИПФ, проверка сроков кредита
      if (point_ipf <= 19 and ma_kolich_repay not between 5 and 24) then
      begin
        res_error_text = 'Балл клиента по ИПФ - ' || point_ipf || '. Cрок кредита должен быть в диапазоне 5-24 мес.';
        suspend;
      end
      --RFCKG-3745
      /*else if (point_ipf >= 20 and
          mac_amount < 250001 and
          ma_kolich_repay not between 3 and 24) then
      begin
        res_error_text = 'Балл клиента по ИПФ - ' || point_ipf || '. Сумма кредита - ' || mac_amount || '. Cрок кредита должен быть в диапазоне 3-24 мес.';
        suspend;
      end*/
      else if (point_ipf >= 20 and
          ma_kolich_repay not between 5 and 28) then
      begin
        res_error_text = 'Балл клиента по ИПФ - ' || point_ipf || '. Сумма кредита - ' || mac_amount || '. Cрок кредита должен быть в диапазоне 5-28 мес.';
        suspend;
      end
      else if (ma_kolich_repay > 28) then
      begin
        res_error_text = 'Максимальный срок кредита по ИПФ 28 мес.';
        suspend;
      end

      select max(kc_amount_approved)
      from kontract
      inner join product on pr_id = k_from_pr_id
      inner join kontstate on ks_from_k_id = k_id
      inner join knclient on kc_from_k_id = k_id
      where ks_state = 1 and
            ks_date_begin < '01.09.2021' and
            ks_date_end <= :ma_date_oper and
            kc_amount_approved > 0 and
            kc_from_c_id = :c_id and
            pr_code containing '.ipf.'
      into max_old_amount_ipf; -- максимальная сумма предыдущего кредита, выданного до 01.09.2021 года и закрытого на момент подачи текущей заявки
    end

    tmp_sum = 0;
    tmp_pr_code = '';

    for select pr_code,
               oks_balance * (kc_amount_approved / k_amount_approved)
        from oksummary
        inner join knclient on kc_from_k_id = oks_from_k_id
        inner join kontract on k_id = oks_from_k_id
        inner join product on pr_id = k_from_pr_id
        where k_disb_date <= :ma_date_oper and
              kc_amount_approved > 0 and
              kc_from_c_id = :c_id and
              oks_close_state = 0 and
              :ma_date_oper between oks_date_begin and oks_date_end
        into tmp_pr_code,
             tmp_sum
    do
    begin
      oks_balance = oks_balance + tmp_sum;
      if (tmp_pr_code containing '.ipf.') then
        oks_balance_ipf = oks_balance_ipf + tmp_sum;
      else
        oks_balance_cls = oks_balance_cls + tmp_sum;
    end

    total_debt = oks_balance + mac_amount;
    if (is_ipf = 1) then
      total_debt_ipf = oks_balance_ipf + mac_amount;
    else
      total_debt_cls = oks_balance_cls + mac_amount;

    max_debt_cls = case
                     when point_cls <= 6 then 150000
                     when point_cls between 7 and 12 then 200000
                     when point_cls between 13 and 19 then 250000
                     when point_cls >= 20 then 500000
                   end;

    max_debt_ipf = case
                     when point_ipf <= 19 then 300000
                     when point_ipf between 20 and 30 then 350000
                     when point_ipf >= 31 then 500000
                   end;

    -- Если максимальная сумма предыдущего кредита (выданный до 01.09.2021 года и закрытого на момент подачи текущей заявки) выше,
    -- чем максимальная сумма по накопленным баллам, то клиент может получить максимальную сумму до суммы предыдущего кредита.
    -- Данное правило касается только ИПФ-продуктов (RFCKG-3251).
    if (max_debt_ipf < max_old_amount_ipf) then
      max_debt_ipf = max_old_amount_ipf;

    max_debt = max_debt_cls;
    if (max_debt < max_debt_ipf) then
      max_debt = max_debt_ipf; -- беру больший максимально допустимый долг по классике или ИПФ

    available_amount = max_debt - oks_balance;
    available_amount_cls = max_debt_cls - oks_balance_cls;
    available_amount_ipf = max_debt_ipf - oks_balance_ipf;

    if (available_amount < 0) then
      available_amount = 0; -- доступная сумма для займа не может быть отрицательной

    if (available_amount_cls < 0) then
      available_amount_cls = 0;

    if (available_amount_ipf < 0) then
      available_amount_ipf = 0;

    if (available_amount_cls > available_amount) then
      available_amount_cls = available_amount; -- доступная сумма по классике не может быть больше общей доступной суммы

    if (available_amount_ipf > available_amount) then
      available_amount_ipf = available_amount; -- доступная сумма по ИПФ не может быть больше общей доступной суммы

    tmp_str = '). Баллы по классике/ИПФ - ' || point_cls || '/' || point_ipf || '. Доступная сумма по классике/ИПФ - ' || round(available_amount_cls) || '/' || round(available_amount_ipf) || ' сом.';

    if (total_debt > max_debt) then
    begin
      res_error_text = 'Совокупный долг (' || round(total_debt) || ') превышает максимально допустимое значение (' || round(max_debt) || tmp_str;
      suspend;
    end
    else if (is_ipf = 1 and
        total_debt_ipf > max_debt_ipf) then
    begin
      res_error_text = 'Совокупный долг по ИПФ (' || round(total_debt_ipf) || ') превышает максимально допустимое значение (' || round(max_debt_ipf) || tmp_str;
      suspend;
    end
    else if (is_ipf = 0 and
        total_debt_cls > max_debt_cls) then
    begin
      res_error_text = 'Совокупный долг по классике (' || round(total_debt_cls) || ') превышает максимально допустимое значение (' || round(max_debt_cls) || tmp_str;
      suspend;
    end
  end
end
