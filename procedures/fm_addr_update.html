CREATE PROCEDURE FM_ADDR_UPDATE(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    REP_ADL_ID D_ID,
    ADDRESS_TYPE D_INTEGER,
    PL_ID D_ID,
    STREET_ID D_ID,
    DEPART_NO D_STR_20,
    VLADEN_NO D_STR_20,
    FLAT_NO D_STR_20,
    BUILDING_NO D_STR_20,
    SECTION_NO D_STR_20,
    COMMENT_STR D_STR_255,
    POST_INDEX D_STR_20,
    IS_WITHOUT_STREET D_INTEGER,
    NEW_STREET_NAME D_STR_255,
    NEW_STREET_TYPE D_INTEGER,
    LANG_KIND D_SMALLINT,
    UN_ID D_ID,
    MAP_LINK D_STR_LONG = null)
RETURNS (
    RES_TEXT D_STR_255,
    ADL_ADDRESS_STR D_STR_255,
    ADL_ID D_ID,
    ADL_STREET_ID D_ID,
    NEED_GEO D_BOOL)
AS
declare variable db_id d_integer;
declare variable has_any_no d_integer;
declare variable geo_count d_integer;
declare variable quota_exceeded d_smallint;
begin
  res_text = '';
  lang_kind = coalesce(lang_kind, 0);
  db_id = f_get_db_id();
  uxf$set_user(un_id);
  need_geo = 0;
  geo_count = -1;

  if (coalesce(pl_id, 0) = 0) then
  begin
    res_text = iif(db_id <> 94401, 'Населенный пункт не указан', 'Yaşayış məntəqəsi qeyd olunmayıb');
    suspend;
    exit;
  end

  if (coalesce(object_id, 0) = 0) then
  begin
    res_text = iif(db_id <> 94401, 'Клиент не указан', 'Müştəri qeyd olunmayıb');
    suspend;
    exit;
  end

  new_street_name = trim(coalesce(new_street_name, ''));
  if (coalesce(street_id, 0) <= 0 and
      new_street_name = '' and
      is_without_street = 0) then
  begin
    res_text = iif(db_id <> 94401, 'Улица не указана', 'Küçə qeyd olunmayıb');
    suspend;
    exit;
  end

  if (char_length(coalesce(trim(post_index), '')) not in (0, 6)) then
  begin
    res_text = iif(db_id <> 94401, 'Почтовый индекс указан неверно', 'Poçt kodu səhvdir');
    suspend;
    exit;
  end

  depart_no = coalesce(depart_no, '0');
  flat_no = coalesce(flat_no, '0');
  building_no = coalesce(building_no, '0');
  section_no = coalesce(section_no, '0');
  post_index = coalesce(post_index, '');
  vladen_no = coalesce(vladen_no, '');
  if (char_length(post_index) <= 3) then
    post_index = '';
  comment_str = coalesce(comment_str, '');

  select adl_address_str, adl_street_id, has_any_no
  from hp_make_address_str(:pl_id, :street_id, :depart_no, '', :flat_no, :building_no, :section_no,
                           :post_index, :is_without_street,
                           :new_street_name, :new_street_type, :lang_kind, :un_id, :vladen_no)
  into :adl_address_str, :street_id, has_any_no;

  -- Конец формирования
  if (post_index = '') then
    post_index = '0';
  if (is_without_street = 1) then
    street_id = pl_id; -- Т.к. улицу не добавляли
  if (is_without_street = 0 and
      has_any_no = 0) then
    is_without_street = 1;

  adl_street_id = street_id;

  if (rep_adl_id <= 0) then
    rep_adl_id = -1;
  else
  begin
    if (object_kind = 0) then
      rep_adl_id = -1;
  end

  update or insert into addrlist (adl_id, adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no,
                                  adl_section_no, adl_address_str, adl_comment, adl_post_index, adl_from_un_id,
                                  adl_street_id, adl_is_without_street, adl_geo_count, adl_lang_kind, adl_vladen_no)
  values (:rep_adl_id, :pl_id, :depart_no, '', :flat_no, :building_no, :section_no, :adl_address_str,
          :comment_str, :post_index, :un_id, :street_id, :is_without_street, :geo_count, :lang_kind, :vladen_no)
  returning adl_id
  into adl_id;

  if (post_index <> '' and
      street_id > 0 and
      is_without_street = 0) then
  begin
    update places
    set pl_post_index = :post_index
    where pl_id = :street_id and
          coalesce(pl_post_index, '') = '';
  end

  if (object_kind = 0) then
  begin
    update or insert into addrclient (adc_from_c_id, adc_address_type, adc_from_adl_id, adc_from_un_id)
    values (:object_id, :address_type, :adl_id, :un_id)
    matching (adc_from_c_id, adc_address_type);
  end

  if (object_kind = 13) then
  begin
    update contacts
    set cc_from_adl_id = :adl_id
    where (cc_id = :object_id);
  end

  if (coalesce(map_link, '') <> '') then
  begin
    if (:map_link not containing '2gis' and
        :map_link not containing 'yandex.ru/maps') then
    begin
      res_text = 'Ссылку на указанный сайт сохранять запрещено.';
      suspend;
      exit;
    end 

    update or insert into addrlist_map_links(aml_from_adl_id, aml_link, aml_from_un_id, aml_datetime)
    values (:adl_id, :map_link, :un_id, current_timestamp)
    matching (aml_from_adl_id);
  end

  -- Геопозиционирование, пока обрабатывается только РФ
  -- Остальные сохраняются со статусом -1 - не обработан
  if (db_id = 81001) then
  begin
    -- Проверка на превышение количества запросов в Yandex Geocoder
    -- Передаем adl_id, но это не принципиально, т.к. лимит на уровне всех запросов
    select quota_exceeded
    from hp_msgquota_process('yandex.geocoder', :adl_id)
    into quota_exceeded;
    if (quota_exceeded = 0) then
    begin
      need_geo = 1;
      geo_count = 0;
    end
  end

  suspend;
end
