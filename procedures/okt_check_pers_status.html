CREATE PROCEDURE OKT_CHECK_PERS_STATUS(
    CELL_NO VARCHAR(20),
    IS_FULL SMALLINT)
RETURNS (
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE,
    STATUS_TXT STR_100,
    FIO STR_100,
    HTML_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_id id_type;
declare variable cc_id integer;
declare variable jl_name str_100;
declare variable html_txt_kont blob sub_type 1 segment size 80;
declare variable i integer;
declare variable k_kontract_no str_100;
declare variable kf_name str_100;
declare variable k_fact_disb_date date;
declare variable k_id id_type;
declare variable ks_state integer;
declare variable k_status integer;
declare variable k_disb_date date;
declare variable k_kredline_end date;
declare variable date_str str_100;
declare variable k_fact_closed_date date;
declare variable html_txt_poruch blob sub_type 1 segment size 80;
declare variable emp_reg_name str_100;
declare variable html_txt_website blob sub_type 1 segment size 80;
declare variable esm_id id_type;
declare variable esm_date date;
declare variable ef_name_id integer;
declare variable migr_name str_100;
declare variable esm_time_str str_100;
declare variable html_txt_cont blob sub_type 1 segment size 80;
declare variable cc_from_c_id id_type;
declare variable cc_name_fio str_100;
declare variable delinq_days integer;
declare variable delinq_total integer;
declare variable ukt_id integer;
declare variable acl_result_rd_id integer;
declare variable acl_status_str str_255;
declare variable has_active smallint;
declare variable result_str str_255;
declare variable acc_no str_100;
declare variable fio_tmp str_100;
declare variable res_str_values str_255;
declare variable esm_id_base id_type;
begin
  c_id = null;

  fio = '';
  is_full = coalesce(is_full, 1);
  has_active = 0;

-- ищем физ лицо
  select res_object_id
  from hp_contdata_find_object(:cell_no, 'cell')
  where res_object_kind = 0 rows 1
  into :c_id;

  if (c_id is not null) then
  begin
    status_txt = 'Найден среди физ. лиц. Клиентом/Сотрудником не является';

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into :fio;

    jl_name = null;

    select jl.jl_name, reg.reg_name
    from hr_pstatus hps
    inner join joblist jl on jl.jl_id = ps_from_jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind
    inner join region reg on reg.reg_unicode = hps.ps_from_reg_id
    where current_date between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          hps.ps_status = 1 and
          hps.ps_from_c_id = :c_id
    into :jl_name, :emp_reg_name;

    if (is_full = 1) then
    begin
      if (jl_name is not null) then
      begin
        status_txt = 'Сотрудник компании';
      end
      else
      begin
-->> КРЕДИТНАЯ ИСТОРИЯ --------------
        html_txt_kont = '';

        i = 0;

        for select kont.k_kontract_no, kal.kal_new_kf_name, kont.k_fact_disb_date, kont.k_fact_closed_date, kont.k_id,
                   ks.ks_state, kont.k_status, kont.k_disb_date, kont.k_kredline_end
            from knclient kncl
            inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
            inner join kontract kont on kont.k_id = kncl.kc_from_k_id
            inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
            where kncl.kc_from_c_id = :c_id and
                  kncl.kc_amount_approved > 0 and
                  current_date between ks.ks_date_begin and ks.ks_date_end and
                  ks.ks_ol_is_delete = 0 and
                  current_date between kal.kal_date and kal.kal_date_end and
                  kal.is_delete = 0
            order by k_fact_disb_date desc
            into :k_kontract_no, :kf_name, :k_fact_disb_date, :k_fact_closed_date, :k_id, :ks_state, :k_status,
                 :k_disb_date, :k_kredline_end
        do
        begin
          if (i = 0) then
          begin
            html_txt_kont = '<b><font color="#02B316">КРЕДИТНАЯ ИСТОРИЯ КЛИЕНТА</font></b></br>';
            i = 1;
          end

          if (ks_state = 1) then
            has_active = 1;

          html_txt_kont = BLOB_APPEND(html_txt_kont, '<b>Контракт №:' || k_kontract_no || ' (группа: ' || kf_name || ')</b></br>');

          html_txt_kont = BLOB_APPEND(html_txt_kont, 'Статус: <b>');

          if (k_status = -1) then
            html_txt_kont = BLOB_APPEND(html_txt_kont, 'аннулирован' || '</br>');
          else
          begin
            if (ks_state = 0) then
              html_txt_kont = BLOB_APPEND(html_txt_kont, 'в процессе' || '</br>');
            else
            if (ks_state = 1) then
              html_txt_kont = BLOB_APPEND(html_txt_kont, 'выдан' || '</br>');
            else
            if (ks_state = 2) then
              html_txt_kont = BLOB_APPEND(html_txt_kont, 'закрыт' || '</br>');
          end
          html_txt_kont = BLOB_APPEND(html_txt_kont, '</b>');

          if (ks_state = 1) then -- если займ активный
          begin
            delinq_days = null;

            select current_date - oks.oks_delinq_date
            from oksummary oks
            where current_date between oks.oks_date_begin and oks.oks_date_end and
                  oks.oks_close_state = 0 and
                  oks.oks_from_k_id = :k_id
            into :delinq_days;

            delinq_days = coalesce(delinq_days, 0);

            if (delinq_days > 0) then
            begin
              html_txt_kont = BLOB_APPEND(html_txt_kont, '<font color="#E15112"><b>ЗАЕМ В ПРОСРОЧКЕ!</b></font>' || '</br>');
              html_txt_kont = BLOB_APPEND(html_txt_kont, '<font color="#E15112">Дней просрочки: <b>' || delinq_days || '</b></font></br>');

              delinq_total = null;

              select sum(zad_os + sh_calc_perc + sh_calc_fine)
              from aarep_make_zad_list(:k_id, current_date, 1)
              where sh_date is not null
              into :delinq_total;

              delinq_total = coalesce(delinq_total, 0);
              html_txt_kont = BLOB_APPEND(html_txt_kont, '<font color="#E15112">Сумма просрочки: <b>' || delinq_total || '</b></font></br>');
            end
          end

          if (k_fact_disb_date is not null) then
          begin
            html_txt_kont = BLOB_APPEND(html_txt_kont, 'Дата выдачи: <b>');

            date_str = replace(iif(extract(day from k_fact_disb_date) < 10, '0', '') || extract(day from k_fact_disb_date) || '.' || iif(extract(month from k_fact_disb_date) < 10, '0', '') || extract(month from k_fact_disb_date) || '.' || extract(year from k_fact_disb_date), ' ', '');

            html_txt_kont = BLOB_APPEND(html_txt_kont, date_str || '</b></br>');
          end

          if ((k_disb_date is not null) and
              (ks_state = 0)) then
          begin
            html_txt_kont = BLOB_APPEND(html_txt_kont, 'Дата выдачи: <b>');

            date_str = replace(iif(extract(day from k_disb_date) < 10, '0', '') || extract(day from k_disb_date) || '.' || iif(extract(month from k_disb_date) < 10, '0', '') || extract(month from k_disb_date) || '.' || extract(year from k_disb_date), ' ', '');

            html_txt_kont = BLOB_APPEND(html_txt_kont, date_str || '</b></br>');
          end

          if ((k_fact_closed_date is not null) and
              (ks_state = 2)) then
          begin
            html_txt_kont = BLOB_APPEND(html_txt_kont, 'Дата закрытия займа: <b>');

            date_str = replace(iif(extract(day from k_fact_closed_date) < 10, '0', '') || extract(day from k_fact_closed_date) || '.' || iif(extract(month from k_fact_closed_date) < 10, '0', '') || extract(month from k_fact_closed_date) || '.' || extract(year from k_fact_closed_date), ' ', '');

            html_txt_kont = BLOB_APPEND(html_txt_kont, date_str || '</b></br>');
          end

          if (ks_state = 1) then
          begin
            html_txt_kont = BLOB_APPEND(html_txt_kont, 'Дата закрытия займа: <b>');

            date_str = replace(iif(extract(day from k_kredline_end) < 10, '0', '') || extract(day from k_kredline_end) || '.' || iif(extract(month from k_kredline_end) < 10, '0', '') || extract(month from k_kredline_end) || '.' || extract(year from k_kredline_end), ' ', '');

            html_txt_kont = BLOB_APPEND(html_txt_kont, date_str || '</b></br>');

            html_txt_kont = BLOB_APPEND(html_txt_kont, '</br>');
            html_txt_kont = BLOB_APPEND(html_txt_kont, '------------</br>');
          end
        end

        if (html_txt_kont <> '') then
          status_txt = 'ЕСТЬ КРЕДИТНАЯ ИСТОРИЯ';
--<< КРЕДИТНАЯ ИСТОРИЯ --------------

-->> ПОРУЧИТЕЛЬ АКТИВНОГО ЗАЙМА --------------
        html_txt_poruch = '';

        i = 0;

        for select zal.z_from_k_id, kont.k_kontract_no, kal.kal_new_kf_name
            from zalog zal
            inner join kontract kont on zal.z_from_k_id = kont.k_id
            inner join kontstate ks on ks.ks_from_k_id = kont.k_id
            inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
            where zal.z_from_c_id = :c_id and
                  ks.ks_state = 1 and
                  current_date between ks.ks_date_begin and ks.ks_date_end and
                  ks.ks_ol_is_delete = 0 and
                  current_date between kal.kal_date and kal.kal_date_end
            into :k_id, :k_kontract_no, :kf_name
        do
        begin
          if (i = 0) then
          begin
            html_txt_poruch = '<font color="#02B316"><b>ПОРУЧИТЕЛЬ В АКТИВНОМ ЗАЙМЕ</b></font></br>';
            i = 1;
          end

          html_txt_poruch = BLOB_APPEND(html_txt_poruch, 'Контракт №: <b>' || k_kontract_no || '</b> (группа: <b>' || kf_name || ')</b></br></br>');

          delinq_days = null;

          select current_date - oks.oks_delinq_date
          from oksummary oks
          where current_date between oks.oks_date_begin and oks.oks_date_end and
                oks.oks_close_state = 0 and
                oks.oks_from_k_id = :k_id
          into :delinq_days;

          delinq_days = coalesce(delinq_days, 0);

          if (delinq_days > 0) then
          begin
            html_txt_kont = BLOB_APPEND(html_txt_kont, '<font color="#E15112"><b>ЗАЕМ В ПРОСРОЧКЕ!</b></font>' || '</br>');
            html_txt_kont = BLOB_APPEND(html_txt_kont, '<font color="#E15112">Дней просрочки: <b>' || delinq_days || '</b></font></br>');

            delinq_total = null;

            select sum(zad_os + sh_calc_perc + sh_calc_fine)
            from aarep_make_zad_list(:k_id, current_date, 1)
            where sh_date is not null
            into :delinq_total;

            delinq_total = coalesce(delinq_total, 0);
            html_txt_kont = BLOB_APPEND(html_txt_kont, '<font color="#E15112">Сумма просрочки: <b>' || delinq_total || '</b></font></br>');
          end
          html_txt_kont = BLOB_APPEND(html_txt_kont, '------------</br>');
        end
--<< ПОРУЧИТЕЛЬ АКТИВНОГО ЗАЙМА --------------
      end
    end
  end
  else
  begin
    if (c_id is null) then
    begin
      if (is_full = 1) then
      begin
-->> КОНТАКТНОЕ ЛИЦО КЛИЕНТА АКТИВНОГО ЗАЙМА --------------

        html_txt_cont = '';

        i = 0;

        for select cont.cc_name, cont.cc_id, cont.cc_from_c_id
            from contacts cont
            where cont.cc_cell_no = :cell_no
--and cont.cc_from_c_id <> 0
            into :fio_tmp, :cc_id, :cc_from_c_id
        do
        begin
          if (fio = '') then
            fio = fio_tmp;

          if (i = 0) then
          begin
            html_txt_cont = '<font color="#02B316"><b>КОНТАКТНОЕ ЛИЦО</b></font></br>';
            i = 1;
          end

          if (cc_from_c_id <> 0) then
          begin
            select cl.c_name_fio
            from clients cl
            where cl.c_id = :cc_from_c_id
            into cc_name_fio;

            status_txt = '<font color="#02B316"><b>КОНТАКТНОЕ ЛИЦО КЛИЕНТА: ' || cc_name_fio || '</b></font>';
          end
          else
          begin
            status_txt = '<font color="#02B316"><b>КОНТАКТ БЕЗ ПРИВЯЗКИ К КЛИЕНТУ</b></font>';
          end

          k_kontract_no = null;

          for select kont.k_kontract_no, kal.kal_new_kf_name
              from knclient kncl
              inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
              inner join kontract kont on kont.k_id = kncl.kc_from_k_id
              inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
              where kncl.kc_from_c_id = :cc_from_c_id and
                    kncl.kc_amount_approved > 0 and
                    current_date between ks.ks_date_begin and ks.ks_date_end and
                    ks.ks_state = 1 and
                    current_date between kal.kal_date and kal.kal_date_end and
                    kal.is_delete = 0
              order by k_fact_disb_date desc
              into :k_kontract_no, :kf_name
          do
          begin
            html_txt_cont = BLOB_APPEND(html_txt_cont, 'Контактное лицо клиента <b>' || cc_name_fio || '</b> </br>');
            html_txt_cont = BLOB_APPEND(html_txt_cont, 'Активный заем №: <b>' || k_kontract_no || '</b> (группа: <b>' || kf_name || '</b>)</br>');
            html_txt_cont = BLOB_APPEND(html_txt_cont, '------------</br>');
          end
--<< КОНТАКТНОЕ ЛИЦО КЛИЕНТА АКТИВНОГО ЗАЙМА --------------
        end
      end
      else
      begin
        select cont.cc_name, cont.cc_id
        from contacts cont
        where cont.cc_cell_no = :cell_no
--and cont.cc_from_c_id <> 0
        order by cc_date descending
        rows 1
        into :fio_tmp, :cc_id;

        if (fio = '') then
          fio = fio_tmp;
      end
    end
  end

  if (is_full = 1) then
  begin
-- << ПОИСК ПО ЗАЯВКАМ С САЙТА ДЛЯ МИГРАНТОВ
    html_txt_website = '';
    i = 0;

    select ef.esf_id
    from extsysfields ef
    where ef.esf_project = 'website' and
          ef.esf_field_code = 'cl_name'
    into ef_name_id;

    ukt_id = f_objkind('extsysmain', 'id');

    esm_id_base = null;

    for select em.esm_id, em.esm_date,
               extract(hour from em.esm_datetime) || ':' || extract(minute from em.esm_datetime)
        from extsysmain em
        inner join extsysvalues ev on ev.esv_from_esm_id = em.esm_id
        inner join extsysfields ef on ef.esf_id = ev.esv_from_esf_id and
              ef.esf_project = em.esm_project
        where em.esm_project = 'website' and
              ef.esf_field_code = 'phone_no' and
              em.esm_action = 'save_cl_info' and
              em.esm_result_id = 0 and
              em.esm_status = 0 and
              em.esm_result_id = 0 and
              ev.esv_value_txt = :cell_no
        order by em.esm_datetime desc
        into :esm_id, :esm_date, :esm_time_str
    do
    begin
      if (i = 0) then
      begin
        html_txt_website = '<font color="#02B316"><b>ЗАЯВКИ С САЙТА ДЛЯ МИГРАНТОВ</b></font></br>';
        i = i + 1;

        esm_id_base = esm_id;

        if (coalesce(status_txt, '') = '') then
          status_txt = 'Заявка с сайта для мигрантов';
      end

      html_txt_website = BLOB_APPEND(html_txt_website, 'Дата заявки : <b>');

      date_str = replace(iif(extract(day from esm_date) < 10, '0', '') || extract(day from esm_date) || '.' || iif(extract(month from esm_date) < 10, '0', '') || extract(month from esm_date) || '.' || extract(year from esm_date), ' ', '');

      html_txt_website = BLOB_APPEND(html_txt_website, date_str || ' ' || esm_time_str || '</b></br>');

      migr_name = null;

      select ev.esv_value_txt
      from extsysvalues ev
      where ev.esv_from_esm_id = :esm_id and
            ev.esv_from_esf_id = :ef_name_id
      into :migr_name;

      if (migr_name is not null) then
      begin
        html_txt_website = BLOB_APPEND(html_txt_website, 'Имя абонента <b>:' || migr_name || '</b></br>');

        if ((i <= 1) and
            (fio = '')) then
          fio = migr_name;
      end

      acl_result_rd_id = null;
      acl_status_str = null;

      select (case al.acl_status
                when 0 then 'Заявка в процессе'
                when 1 then 'Заявка в процессе'
                when 2 then 'Заявка обработана'
                when -2 then 'Заявка отменена'
              end) as status_str,

                   al.acl_result_rd_id
      from actionlist al
      where al.acl_object_id = :esm_id and
            al.acl_object_kind = :ukt_id and
            al.acl_parent_id is null
      into :acl_status_str, :acl_result_rd_id;

      acl_status_str = trim(coalesce(acl_status_str, ''));

      if (acl_status_str <> '') then
      begin
        html_txt_website = BLOB_APPEND(html_txt_website, 'Статус обработки: <b>' || :acl_status_str || '</b></br>');
      end

      result_str = null;

      select rd.rd_name
      from refdata rd
      where rd.rd_id = :acl_result_rd_id
      into :result_str; -- определяю результат звонка

      result_str = coalesce(result_str, '');

--<font color="#02B316"
      if (result_str <> '') then
      begin
        html_txt_website = BLOB_APPEND(html_txt_website, 'Результат звонка:<b>' || :result_str || '</b></br>');
      end
      res_str_values = null;

      select res_str_values
      from hp_objectprops_anketa_get(:ukt_id, :esm_id, -1, 'AnketaQuestions.OKTELL.WebSite.Memo', current_date, 'Results.StrValue')
      into res_str_values;

      res_str_values = coalesce(res_str_values, '');
      if (res_str_values <> '') then
      begin
        html_txt_website = BLOB_APPEND(html_txt_website, 'Комментарий оператора:<b>' || :res_str_values || '</b></br>');
      end

      html_txt_website = BLOB_APPEND(html_txt_website, '------------</br>');
    end
  end
-- >> -----------------------------
  status_txt = coalesce(status_txt, 'Не известен');

  html_txt = '';
  html_txt = '<font color="#02B316"><b>Статус: ' || status_txt || '</b></font></br>';

  if (fio <> '') then
    html_txt = BLOB_APPEND(html_txt, '<b>ФИО: ' || fio || '</b></br></br>');

  if (jl_name is not null) then
  begin
    html_txt = BLOB_APPEND(html_txt, 'Должность: <b>' || jl_name || '</b></br>');
    html_txt = BLOB_APPEND(html_txt, 'Офис: <b>' || emp_reg_name || '</b></br>');
  end

  if (has_active = 1) then
  begin
    acc_no = '';

    select uniq_acc_str
    from hp_show_clientuniq_acc_no(:c_id)
    into :acc_no;

    acc_no = coalesce(acc_no, 'Не указан');

    html_txt = BLOB_APPEND(html_txt, 'Лицевой счет: <font color="#02B316"><b>' || acc_no || '</b></font></br>');
  end
  html_txt = BLOB_APPEND(html_txt, '------------</br>');

  html_txt = BLOB_APPEND(html_txt, coalesce(html_txt_kont, ''));
  html_txt = BLOB_APPEND(html_txt, coalesce(html_txt_poruch, ''));
  html_txt = BLOB_APPEND(html_txt, coalesce(html_txt_cont, ''));
  html_txt = BLOB_APPEND(html_txt, coalesce(html_txt_website, ''));

  c_id = coalesce(c_id, 0);
  cc_id = coalesce(cc_id, 0);
  esm_id_base = coalesce(esm_id_base, 0);

  if (c_id <> 0) then
  begin
    object_id = c_id;
    object_kind = 'Clients';
  end
  else
  if (esm_id_base <> 0) then
  begin
    object_id = esm_id_base;
    object_kind = 'extsysmain';
  end
  else
  begin
    object_id = cc_id;
    object_kind = 'Contacts';
  end

  object_id = coalesce(object_id, 0);
  object_kind = coalesce(object_kind, '');

  suspend;
end
