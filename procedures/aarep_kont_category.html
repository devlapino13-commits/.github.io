CREATE PROCEDURE AAREP_KONT_CATEGORY(
    REP_K_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    KONT_CATEGORY VARCHAR(20))
AS
declare variable aa_curr_kc_id id_type;
declare variable aa_disb_kc_id id_type;
declare variable dd_disb_date date;
declare variable aa_loan_kind_new varchar(20);
declare variable aa_category_id smallint;
declare variable rep_delinq_days integer;
declare variable aa_k_id id_type;
declare variable kal_date date;
declare variable categ_id smallint;
declare variable def_curr varchar(3);
declare variable ol_date date;
declare variable ok_type smallint;
declare variable ok_sh_date date;
declare variable ok_os curr_type;
declare variable ok_perc curr_type;
begin
  kont_category = null;
  aa_k_id = :rep_k_id;
  categ_id = 0;
  rep_delinq_days = null;
  select cm_default_currency
  from company
  into def_curr;
  for select ks_kc_id, kal_new_kf_officer, k_disb_date, kal_date
      from kontstate, kontract, knaktlist
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = k_id and
            ks_from_k_id = :rep_k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end
      into aa_curr_kc_id, aa_disb_kc_id, dd_disb_date, kal_date
  do
  begin
    if (aa_curr_kc_id = aa_disb_kc_id) then
    begin
      if (dd_disb_date >= '01.06.2012') then
      begin
        aa_loan_kind_new = 'Категория 1';
        aa_category_id = 1;
      end
      else
      begin
        rep_delinq_days = 0;
        select rep_delinq_days
        from aarep_kont_delinq_status(:aa_k_id, '01.12.2012')
        into :rep_delinq_days;
        if (rep_delinq_days is null) then
          rep_delinq_days = 0;
        if (rep_delinq_days <= 0) then
        begin
          aa_loan_kind_new = 'Категория 1';
          aa_category_id = 1;
        end
        else
        if (rep_delinq_days <= 180) then
        begin
          aa_loan_kind_new = 'Категория 2';
          aa_category_id = 2;
        end
        else
        begin
          aa_loan_kind_new = 'Категория 3';
          aa_category_id = 3;
        end
      end
    end
    else
    begin
      if (kal_date < '01.12.2012') then
        kal_date = '01.12.2012';
      rep_delinq_days = 0;
      select rep_delinq_days
      from aarep_kont_delinq_status(:aa_k_id, :kal_date)
      into :rep_delinq_days;
      if (rep_delinq_days is null) then
        rep_delinq_days = 0;
      if (rep_delinq_days <= 0) then
      begin
        aa_loan_kind_new = 'Категория 1';
        aa_category_id = 1;
      end
      else
      if (rep_delinq_days <= 180) then
      begin
        aa_loan_kind_new = 'Категория 2';
        aa_category_id = 2;
      end
      else
      begin
        aa_loan_kind_new = 'Категория 3';
        aa_category_id = 3;
      end
    end
    kont_category = aa_loan_kind_new;
    if (kont_category is null) then
      kont_category = 'Не определен';

    for select ok_ol_date, ok_type, ok_sh_date, ok_os, ok_perc
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :aa_k_id and
              ok_type <> 7 and
              ok_from_c_id = 0
        order by ok_from_k_id, ok_sh_date, ok_ol_date
        into :ol_date, :ok_type, :ok_sh_date, :ok_os, :ok_perc
    do
    begin
      if ((rep_date_end >= '12.04.2013') and
          (ol_date - ok_sh_date > 180) and
          (ol_date >= '01.12.2012') and
          (def_curr = '810') and
          (ok_type = 4) and
          ((ok_os + ok_perc) < 0)) then
        categ_id = 3;
    end

    if (rep_delinq_days is null and
        categ_id <> 3) then
    begin
      rep_delinq_days = 0;
      select rep_delinq_days
      from aarep_kont_delinq_status(:aa_k_id, :rep_date_end)
      into :rep_delinq_days;
    end

    if (((rep_delinq_days > 180) and
        (rep_date_end > '01.12.2012') and
        (def_curr = '810')) or (categ_id = 3)) then
    begin
      aa_loan_kind_new = 'Категория 3';
      aa_category_id = 3;
    end

    suspend;
  end
end
