CREATE PROCEDURE CM_KG_0098_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable dtoc d_date;
declare variable photo_id id_type;
declare variable days_qty integer;
declare variable c_id d_id;
declare variable pr_code d_str_100;
begin
  for select cl.c_name_fio, cl.c_pass_no, cl.c_blob_photo_id, kont.k_disb_date, cl.c_id, pr.pr_code
      from knclient kc, clients cl, kontract kont, product pr
      where kc.kc_from_k_id = :k_id and
            kc.kc_from_c_id = cl.c_id and
            kc.kc_amount_approved > 0 and
            kont.k_id = kc.kc_from_k_id and
            pr.pr_id = kont.k_from_pr_id
      into :res_info_text, :kf_client_pass_no, :photo_id, :kf_disb_date, :c_id, :pr_code
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;
    if (photo_id is null) then
    begin
      res_error_text = 'Фотография клиента отсутсвует. Утверждение займа невозможно';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
    if (not exists (select *
                from knclient kncl
                inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
                where kncl.kc_from_c_id = :c_id
                and kncl.kc_amount_approved > 0
                and ks.ks_ol_is_delete = 0
                and ks.ks_state = 1
                and ks.ks_date_begin <= :kf_disb_date
                and ks.ks_from_k_id <> :k_id)) then
    begin
        select cast(bs.bls_datetime as date)
        from blobstore bs
        where bs.bls_id = :photo_id and
              bs.bls_kind = 0 --фото сделано с камеры мобильного (0-камера, 1-сканирование, 2-галерея)
        into :dtoc;
    
        days_qty = 0;
    
        select count(sc.scd_work_hours)
        from sa_calendays sc
        where sc.scd_date between :dtoc and :kf_disb_date and
              sc.scd_work_hours > 0
        into :days_qty;
    
        if (days_qty > 2) then
        begin
          res_error_text = 'Фотография клиента не актуальна - сделана более 2 рабочих дней назад назад (' || f_datetostr(:dtoc) || '). Утверждение займа невозможно';
          res_error_is_critical = 1;
          suspend;
          exit;
        end
     end
  end
end
