CREATE PROCEDURE CARD_COMMAND_STOP_LIST_IN(
    OBJECT_KIND D_STR_100,
    OBJECT_ID D_ID,
    PARENT_CS_ID D_INTEGER = null)
AS
declare variable OBJECT_KIND_STR D_STR_100;
declare variable CARD_ID D_INTEGER;
declare variable CCA_ID D_INTEGER;
declare variable CS_REQUEST_JSON D_STR_8100;
declare variable CS_ID D_INTEGER;
declare variable CBR_REASON D_STR_255;
declare variable IPC_CODE D_STR_100;
declare variable CBO_REGION_ID D_INTEGER;
declare variable UN_ID D_ID;
declare variable IS_HARD_STOPLIST D_BOOL;
declare variable CS_FROM_CSC_ID D_INTEGER;
begin
    -- Процедура отвечает за формирование операции для МПЦ "Поместить в стоп-лист"

    object_kind_str = f_objkind(object_kind, 'name');
    
    if (object_kind_str = 'cardblockopers') then
    begin
        select cbo.cbo_from_cal_id,
               cbr.cbr_reason,
               cbr.cbr_ipc_code,
               cbo.cbo_region_id,
               cbo.cbo_from_un_id,
               coalesce(cbo.cbo_is_hard_stoplist, 0)
        from cardblockopers cbo
        inner join cardblockreasons cbr on cbr.cbr_id = cbo.cbo_reason
        where cbo.cbo_id = :object_id
        into :card_id,
             :cbr_reason,
             :ipc_code,
             :cbo_region_id,
             :un_id,
             :is_hard_stoplist;
    end
    else
        exception e_message 'object_kind = '||object_kind_str||' не реализован';

    select cal_from_cca_id
    from cardlist cal
    where cal.cal_card_id = :card_id
    and cal_status = f_enum('cardlist_status', 'transfer_to_client', 'id')
    into :cca_id;

    if (is_hard_stoplist = 1) then
    begin
      cs_from_csc_id = f_get_cardsynccfg('card_hard_stoplist_in', 'id');
    end
    else
    begin
      ipc_code = '5'; -- МПЦ код = 5 мягкий стоп лист
      cs_from_csc_id = f_get_cardsynccfg('card_stoplist_in', 'id');
    end

    cs_request_json = f_format('{"card_id":%1, "reason_comment": "%2", "reason_code": "%3", "bank_c": 21, "groupc": "01"}', :card_id, :cbr_reason, :ipc_code);

    insert into cardsync(
      cs_parent_id,
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_un_id
      )
    values(
      :parent_cs_id,
      :cs_from_csc_id,
      :card_id,
      :cca_id,
      f_objkind(:object_kind, 'id'),
      :object_id,
      :cs_request_json,
      :cbo_region_id,
      :un_id
      )
     returning cs_id
     into :cs_id;

     if (object_kind_str = 'cardblockopers') then
     begin
        update cardblockopers cbo
        set cbo.cbo_from_cs_id = :cs_id
        where cbo.cbo_id = :object_id;
     end

    if (is_hard_stoplist = 1) then
    begin
      insert into cardhardstoplist(chl_card_id, chl_from_un_id, chl_from_cbo_id, chl_from_cs_id)
      values (:card_id, :un_id, :object_id, :cs_id);
    end

end
