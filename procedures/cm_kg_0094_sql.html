CREATE PROCEDURE CM_KG_0094_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable lo_id id_type;
begin
  for
  select kal.kal_new_kf_officer
  from miniapp maa, product, knaktlist kal
  where ma_id = :ma_id 
  and ma_is_active = 1
  and ma_date_end = '31.12.9999'
  and ma_from_pr_id = pr_id
  and pr_code containing '.education.'
  and kal.kal_from_kf_id = maa.ma_from_kf_id
  and maa.ma_date_oper between kal.kal_date and kal.kal_date_end
  and kal.is_delete = 0
  into lo_id
  do
  begin
    if (lo_id <> 1090100871894) then -- если это не Бекназар
    begin
      res_error_text = 'Выбранный продукт может быть выдан только Бекназаром';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
