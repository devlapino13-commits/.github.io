CREATE PROCEDURE HP_LOGIN_ACTION_CHECKS(
    ACTION_CODE D_STR_CODE,
    ACTION_DATA D_STR_LONG,
    CHECK_EXTRA_DATA D_STR_LONG = null)
RETURNS (
    RES_CODE D_STR_CODE,
    ERR_TEXT D_STR_255)
AS
declare variable mcl_id d_id;
declare variable mcl_login d_str_code;
declare variable mcl_project d_integer;
declare variable mcl_birth_date d_date;
declare variable login_try_cnt d_integer;
declare variable send_code_cnt d_integer;
declare variable check_code_cnt d_integer;
declare variable mcc_pin_code d_integer;
declare variable max_check_cnt d_smallint;
declare variable pin_code d_integer;
declare variable db_is_actual d_smallint;
declare variable db_id d_integer;
declare variable is_biom d_smallint;
declare variable c_id_by_login d_id;
declare variable c_birth_date d_date;
declare variable tcv_create_datetime d_datetime;
declare variable tcv_count d_integer;
declare variable tcv_max_count d_integer;
declare variable wait_time_minutes_left d_integer;
declare variable wait_time_minutes_max d_integer;
declare variable quota_exceeded d_bool;
declare variable quota_cfg_code d_str_code;
declare variable quota_cont_code d_str_code ;
declare variable ip_address d_str_code;
declare variable mcl_pass_hash d_str_long;
declare variable mcl_dev_id_str d_str_code;
declare variable mcl_dev_id d_uuid;
declare variable mcl_dev_id_actual d_uuid;
declare variable checks_limit_exceeded d_smallint;
declare variable quota_code d_str_code;
declare variable quota_interval_parsed d_integer;
declare variable time_left_str d_str_100;
begin
    /*
        RFCKG-4671
        Процедура отвечает за различные проверки, выполняемые в процедуре mc_login_action
        За валидацию входных параметров отвечает другая процедура - hp_login_action_validation
    */
    db_id = f_get_db_id();
    db_is_actual = 0;

    is_biom = iif(db_id in (39801, 41701), 1, 0);

    if (exists(select * from hp_db_is_actual)) then
        db_is_actual = 1;


    if (action_code in ('send_code', 'check_code', 'update_pass', 'login')) then
    begin
        mcl_login = f_json_str_get_value(:action_data, 'login', '');
        mcl_project = f_enum('mcl_project', f_json_str_get_value(:action_data, 'project', 'kmp'), 'id', 0);

        checks_limit_exceeded = 0;
        login_try_cnt = 0;
        send_code_cnt = 0;
        check_code_cnt = 0;
        max_check_cnt = 3; -- !!HARDCODE!!

        select mcc.mcc_login_try_cnt, mcc.mcc_send_code_cnt, mcc.mcc_check_code_cnt, mcc.mcc_pin_code
        from mc_check mcc
        where mcc.mcc_login = :mcl_login and
              mcc.mcc_check_kind = 0 and
              mcc.mcc_project = :mcl_project
        into :login_try_cnt, :send_code_cnt, :check_code_cnt, :mcc_pin_code;

        if (action_code = 'send_code') then
        begin
            if (db_is_actual = 1 and send_code_cnt >= max_check_cnt) then
                checks_limit_exceeded = 1; -- Чтобы не дублировать сообщения об ошибках, выносим их в конец и используем флаг ошибки
        end

        if (action_code in ('check_code', 'update_pass')) then
        begin
            pin_code = f_json_str_get_value(action_data, 'pin_code', -1);

            if (db_is_actual = 1 and check_code_cnt >= max_check_cnt or mcc_pin_code = 0) then
                checks_limit_exceeded = 1;

            if (checks_limit_exceeded = 0 and mcc_pin_code <> pin_code) then
            begin
                update or insert into mc_check
                (
                    mcc_login,
                    mcc_check_code_cnt,
                    mcc_check_kind,
                    mcc_project
                )
                values
                (
                    :mcl_login,
                    :check_code_cnt + 1,
                    0,
                    :mcl_project
                )
                matching (mcc_login, mcc_check_kind, mcc_project);

                res_code = case action_code
                                when 'check_code' then 'check_error'
                                when 'update_pass' then 'pass_error'
                           end;

                err_text = 'Коды не совпадают.';
                
                suspend;
                exit;
            end
        end

        if (action_code = 'login') then
        begin
            mcl_pass_hash = f_json_str_get_value(action_data, 'password', '');
            mcl_id = 0;

            select
              mcl.mcl_id
            from mc_login mcl
            where mcl.mcl_login = :mcl_login
              and mcl.mcl_is_delete = 0
              and mcl.mcl_project = :mcl_project
              and mcl.mcl_pass_hash = :mcl_pass_hash
            into
              :mcl_id;

            if (coalesce(mcl_id, 0) = 0) then
            begin
                update or insert into mc_check
                (
                    mcc_login,
                    mcc_login_try_cnt,
                    mcc_check_kind,
                    mcc_project
                )
                values
                (
                    :mcl_login,
                    :login_try_cnt + 1,
                    0,
                    :mcl_project
                )
                matching (mcc_login, mcc_check_kind, mcc_project);
                
                res_code = 'login_error';
                err_text = 'Неправильный номер телефона или пароль';
    
                suspend;
                exit;
            end

            if (login_try_cnt >= max_check_cnt) then
                checks_limit_exceeded = 1;
        end

        if (checks_limit_exceeded = 1) then
        begin
            res_code = case action_code
                            when 'send_code' then 'send_block'
                            when 'check_code' then 'check_block'
                            when 'update_pass' then 'pass_block'
                            when 'login' then 'login_block'
                       end;
    
            err_text = case action_code
                            when 'send_code' then 'Превышен лимит отправок кода. Повторите попытку через 15 минут'
                            when 'check_code' then 'Превышен лимит проверок кода. Повторите попытку через 15 минут'
                            when 'update_pass' then 'Превышен лимит попыток смены пароля. Повторите попытку через 15 минут'
                            when 'login' then 'Превышен лимит ошибочных попыток входа. Повторите попытку через 15 минут'
                       end;
            suspend;
            exit;            
        end
    end

    if (action_code in ('send_code', 'update_pass')) then
    begin
        if (:is_biom = 0) then
        begin
            mcl_login = f_json_str_get_value(:action_data, 'login', '');
            mcl_birth_date = cast(f_json_str_get_value(:action_data, 'birth_date', '') as date);
            res_code = case action_code
                            when 'send_code' then 'send_error'
                            when 'update_pass' then 'pass_error'
                       end;
    
            select first 1 cd.cda_from_c_id
            from contdata cd
            where cd.cda_from_clt_id = f_cont_id(:mcl_login, 'null') and
                  cd.cda_cont_kind = 0 and
                  cd.cda_is_main = 1 and
                  cd.cda_status = 1 and
                  cd.cda_is_delete = 0
            into :c_id_by_login;
    
            if (c_id_by_login is not null) then
            begin
                select cl.c_birth_date
                from clients cl
                where cl.c_id = :c_id_by_login
                into :c_birth_date;
                
                if (c_birth_date is null) then
                begin
                    err_text = 'У найденного клиента не указана дата рождения';
                    suspend;
                    exit;
                end
                
                if (mcl_birth_date <> c_birth_date) then
                begin
                    err_text = 'Данный номер закреплен за другим клиентом. Обратитесь в офис компании для подтверждения';
                    suspend;
                    exit;
                end
            end
    
            if (coalesce(err_text, '') = '') then
                res_code = null;
        end
    end

    -- ***************************************************************************

    if (action_code = 'send_code') then
    begin
        if (db_id = 39801 and db_is_actual = 1) then
        begin
            tcv_create_datetime = null;
            tcv_count = null;
            tcv_max_count = 3; -- !!HARDCODE!!
            wait_time_minutes_max = 15; -- !!HARDCODE!!
            wait_time_minutes_left = 0;

            select tcv.tcv_create_datetime, tcv.tcv_count
            from try_counter_verify tcv
            where tcv.tcv_phone = :mcl_login
            into :tcv_create_datetime, :tcv_count;

            wait_time_minutes_left = wait_time_minutes_max - datediff(minute, :tcv_create_datetime, current_timestamp);

            if (tcv_count >= tcv_max_count and wait_time_minutes_left > 0) then
            begin
                res_code = 'send_block';
                err_text = 'Вы не прошли биометрию, повторите попытку через ' || :wait_time_minutes_left|| ' минут';
                suspend;
                exit;
            end
        end

        if (db_is_actual = 1) then
        begin
            ip_address = f_json_str_get_value(:action_data, 'ip', '');
            res_code = 'send_error';

            quota_cfg_code = 'mobile.clients.send_code_login';
            quota_cont_code = mcl_login;

            select quota_exceeded, quota_code
            from hp_msgquota_process(:quota_cfg_code, :quota_cont_code)
            into :quota_exceeded, :quota_code;

            -- Если квота по номеру не превышена, проверяем квоту по IP
            if (quota_exceeded = 0) then
            begin
                quota_cfg_code = 'mobile.clients.send_code_ip';
                quota_cont_code = ip_address;

                select quota_exceeded, quota_code
                from hp_msgquota_process(:quota_cfg_code, :quota_cont_code)
                into :quota_exceeded, :quota_code;
            end

            if (quota_exceeded = 1) then
            begin
                quota_interval_parsed = (select quota_interval from hp_parse_quota_code(:quota_code));

                if (coalesce(quota_interval_parsed, -1) <> -1) then
                begin
                    time_left_str =  (select time_left from hp_get_time_left_formatted(:quota_interval_parsed, :quota_cont_code, :quota_cfg_code));
                    err_text = f_format('Отправка кода временно ограничена. Пожалуйста, повторите попытку через %1 [#%2]', :time_left_str, :quota_code);
                end
                else
                    err_text = 'Отправка кода временно ограничена. Пожалуйста, повторите попытку позже';

                suspend;
                exit;
            end
        end

        if (coalesce(err_text, '') = '') then
            res_code = null;
    end

    -- ***************************************************************************

    if (action_code = 'login') then
    begin
        mcl_dev_id_str = f_json_str_get_value(action_data, 'uuid_id', '');
        mcl_dev_id = null;

        if (char_length(mcl_dev_id_str) > 0) then
          mcl_dev_id = char_to_uuid(mcl_dev_id_str);

        select
          mcl.mcl_dev_id
        from mc_login mcl
        where mcl.mcl_login = :mcl_login
          and mcl.mcl_is_delete = 0
          and mcl.mcl_project = :mcl_project
          and mcl.mcl_pass_hash = :mcl_pass_hash
        into
          :mcl_dev_id_actual;
    
        if (mcl_dev_id <> mcl_dev_id_actual) then
        begin
            res_code = 'login_error';
            err_text = 'Скорее всего Вы пытаетесь зайти с другого устройства. Пройдите процесс регистрации после нажатия "Забыл Пин код"';
            suspend;
            exit;
        end
    end

    -- ***************************************************************************

end
