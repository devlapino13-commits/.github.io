CREATE PROCEDURE UXP$UXEVENT_FRAME_SQL(
    UX_UUID D_UUID)
RETURNS (
    UFR_UUID D_UUID,
    UFR_NAME D_STR_255,
    UFR_UUID_CODE D_STR_CODE,
    UFR_SELECT_SQL D_STR_4096,
    UFR_PARAM_SQL D_STR_4096,
    UFR_FIELD_LIST D_STR_8100)
AS
declare variable ufr_select_list d_str_4096;
declare variable ufr_param_list d_str_4096;
begin
  for select ufr_uuid,
             iif(ufr_code not containing ufr_name and coalesce(ufr_code, '') <> '', ufr_name || '  (' || ufr_code || ')', ufr_name)
  from ux$frame
  where (ufr_uuid = :ux_uuid or ufr_owner = :ux_uuid)
  into ufr_uuid, ufr_name
  do
  begin
    ufr_uuid_code = uuid_to_char(ufr_uuid);

    ufr_select_sql = '';
    ufr_select_list = '';
    ufr_param_sql = '';
    ufr_param_list = '';
    ufr_field_list = '';

    select sql_text, res_field_list
    from uxp$uxframe_sql_expr(:ux_uuid, null, 1, 0)
    into :ufr_param_sql, :ufr_param_list;

    ufr_field_list = ufr_param_list;

    select sql_text, res_field_list
    from uxp$uxframe_sql_expr(:ux_uuid, null, 0, 0)
    into :ufr_select_sql, :ufr_select_list;

    if (ufr_select_list <> '') then
    begin
      if (ufr_field_list <> '') then
        ufr_field_list = ufr_field_list || ascii_char(13);

      ufr_field_list = ufr_field_list || ufr_select_list;
    end

    suspend;
  end
end
