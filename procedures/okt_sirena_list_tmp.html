CREATE PROCEDURE OKT_SIRENA_LIST_TMP(
    REP_DATE DATE,
    PROJECT_CODE STR_100,
    SUBPROJECT_CODE STR_100,
    K_FOLDER_NAME STR_100,
    KONTRACT_NO STR_100,
    DELINQ_DAYS_MIN INTEGER,
    DELINQ_DAYS_MAX INTEGER,
    CALLED_OPER STR_100 = '')
RETURNS (
    K_ID ID_TYPE,
    KF_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(50),
    DELINQ_DAYS INTEGER,
    OKS_KC_ID ID_TYPE,
    REG_ID D_INTEGER,
    CLI_QTY_COUNTER D_INTEGER)
AS
declare variable KF_ID_TMP ID_TYPE;
declare variable K_ID_TMP ID_TYPE;
declare variable DELINQ_DAYS_MIN_TMP integer;
declare variable DB_ID integer;
declare variable BALANCE_OS CURR_TYPE;
declare variable DELINQ_OS CURR_TYPE;
declare variable MIN_BALANCE integer;
declare variable SQL_TEXT D_STR_8100;
declare variable RD_ID D_INTEGER;
begin
  if (CALLED_OPER = 'OKTELL_LIST') then
  begin
    rd_id = f_get_rd_id(:project_code);

    for select pee.ppe_from_k_id
    from pantera_preexec pee
    where pee.ppe_project_rd_id = :rd_id
    into :k_id
    do
    begin
       select kont.k_last_reg_id,
              kont.k_kolich_client_approved,
              kont.k_kontract_no
       from kontract kont
       where kont.k_id = :k_id
       into :reg_id,
            :cli_qty_counter,
            :k_kontract_no;

      suspend;
    end
    exit;
  end

  k_folder_name = trim(coalesce(k_folder_name, ''));
  kontract_no = trim(coalesce(kontract_no, ''));
  delinq_days_min = coalesce(delinq_days_min, 0);
  delinq_days_max = coalesce(delinq_days_max, 0);

  db_id = f_get_db_id();
  min_balance = 500;

  if (project_code = 'OKTELL.Projects.Sirena') then
  begin
    if (db_id = 94401) then -- АЗ
    begin
      if (delinq_days_min <= 0) then
        delinq_days_min = 61;
      if (delinq_days_max <= 0) then
        delinq_days_max = 9999;
      min_balance = 0;
    end
    else if (db_id = 81001) then
    begin
      if (delinq_days_min <= 0) then
        delinq_days_min = 21;
      if (delinq_days_max <= 0) then
        delinq_days_max = 60;
    end
    else
    begin
      if (delinq_days_min <= 0) then
        delinq_days_min = 1;
      if (delinq_days_max <= 0) then
        delinq_days_max = 30;
    end
  end
  if (project_code = 'OKTELL.Projects.Sirena1') then
  begin
    if (delinq_days_min <= 0) then
      delinq_days_min = 21;
    if (delinq_days_max <= 0) then
      delinq_days_max = 60;
    if (delinq_days_max > 60) then
      delinq_days_max = 60;
  end

  if (kontract_no <> '') then
  begin
    select kont.k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id_tmp;
  end

  if (k_folder_name <> '') then
  begin
    select kal.kal_from_kf_id
    from knaktlist kal
    where kal.kal_new_kf_name = :k_folder_name
    and :rep_date between kal.kal_date and kal.kal_date_end
    and kal.is_delete = 0
    into :kf_id_tmp;
  end

  if (subproject_code = '') then
  begin
    balance_os = 0;

    sql_text = 'select oks.oks_from_k_id,
                       k_kontract_no,
                       kont.k_from_kf_id,
                       cast(:rep_date as date) - oks_delinq_date as delinq_days,
                       oks.oks_kc_id,
                       oks.oks_balance,
                       kont.k_last_reg_id,
                       kont.k_kolich_client_approved
                from oksummary oks
                inner join kontract kont on kont.k_id = oks.oks_from_k_id
                where :rep_date between oks_date_begin and oks_date_end and
                      oks_delinq_date < :rep_date and
                      cast(:rep_date as date) - oks_delinq_date between iif(oks_kc_id <> 1000000000004, iif(:delinq_days_min < 7, 7, :delinq_days_min), :delinq_days_min) and :delinq_days_max and
                      oks.oks_balance > :min_balance';
    if (k_id_tmp is not null) then
        sql_text = sql_text||' and kont.k_id = :k_id_tmp';
    else
        sql_text = sql_text||' and :k_id_tmp is null';

    if (kf_id_tmp is not null) then
        sql_text = sql_text||' and kont.k_from_kf_id = :kf_id_tmp';
    else
        sql_text = sql_text||' and :kf_id_tmp is null';

    for execute statement(sql_text)
    (rep_date := rep_date,
     delinq_days_min := delinq_days_min,
     delinq_days_max := delinq_days_max,
     min_balance := min_balance,
     k_id_tmp := k_id_tmp,
     kf_id_tmp := kf_id_tmp)
    into :k_id,:k_kontract_no, :kf_id, :delinq_days, :oks_kc_id, :balance_os,
         :reg_id, :cli_qty_counter
    do
    begin
      if (exists(select *
               from sb_akt sba
               where sba.sba_from_k_id = :k_id and
                     sba.sba_from_sbl_id <> 0 and
                     sba.sba_is_delete = 0 and
                     sba.sba_status > 0 and
                     :rep_date between sba.sba_date_begin and sba.sba_date_end)) then
        continue;

      delinq_os = 0;
      delinq_os = 0;

      select sum(ok_os + ok_sh_os)
      from operkred ok
      where ok_from_k_id = :k_id and
            ok_from_c_id = 0 and
            ok_ol_date <= :rep_date and
            ok_sh_date < :rep_date and
            ok_ol_is_active = 1 and
            ok_type <= 7
      into :delinq_os;

      balance_os = coalesce(balance_os, 0);
      delinq_os = coalesce(delinq_os, 0);

      if (min_balance = 0) then
        suspend;
      else
      if (not((balance_os < min_balance) or (delinq_os < min_balance))) then
        suspend;
    end
  end
  else
  if (subproject_code = 'OKTELL.SubProjects.Sirena.VE') then -- звонки по группам на ВС Сирена
  begin
    balance_os = 0;

    sql_text = 'select oks.oks_from_k_id,
                k_kontract_no,
                kont.k_from_kf_id,
                cast(:rep_date as date) - oks_delinq_date,
                oks.oks_kc_id,
                oks_balance,
                kont.k_last_reg_id,
                kont.k_kolich_client_approved
            from oksummary oks
            inner join kontract kont on kont.k_id = oks.oks_from_k_id
            where :rep_date between oks_date_begin and oks_date_end and
                  oks_delinq_date < :rep_date and
                  cast(:rep_date as date) - oks_delinq_date between :delinq_days_min and :delinq_days_max and
                  oks.oks_kc_id = 1000000000004 and
                  oks.oks_balance > :min_balance';

    if (k_id_tmp is not null) then
        sql_text = sql_text||' and kont.k_id = :k_id_tmp';
    else
        sql_text = sql_text||' and :k_id_tmp is null';

    if (kf_id_tmp is not null) then
        sql_text = sql_text||' and kont.k_from_kf_id = :kf_id_tmp';
    else
        sql_text = sql_text||' and :kf_id_tmp is null';

    for execute statement(sql_text)
    (rep_date := rep_date,
     delinq_days_min := delinq_days_min,
     delinq_days_max := delinq_days_max,
     min_balance := min_balance,
     k_id_tmp := k_id_tmp,
     kf_id_tmp := kf_id_tmp)
    into :k_id, :k_kontract_no, :kf_id, :delinq_days, :oks_kc_id, :balance_os,
         :reg_id, :cli_qty_counter
    do
    begin
      if (exists(select *
               from sb_akt sba
               where sba.sba_from_k_id = :k_id and
                     sba.sba_from_sbl_id <> 0 and
                     sba.sba_is_delete = 0 and
                     sba.sba_status > 0 and
                     :rep_date between sba.sba_date_begin and sba.sba_date_end)) then
        continue;


      delinq_os = 0;
      select sum(ok_os + ok_sh_os)
      from operkred
      where ok_from_k_id = :k_id and
            ok_from_c_id = 0 and
            ok_ol_date <= :rep_date and
            ok_sh_date < :rep_date and
            ok_ol_is_active = 1 and
            ok_type <= 7
      into :delinq_os;

      balance_os = coalesce(balance_os, 0);
      delinq_os = coalesce(delinq_os, 0);

      if (min_balance = 0) then
        suspend;
      else
      if (not((balance_os < min_balance) or (delinq_os < min_balance))) then
        suspend;
    end
  end
  else
  if (subproject_code = 'OKTELL.SubProjects.Sirena.Delinq') then --Звонки просрочникам
  begin
    delinq_days_min_tmp = 7;

    if (delinq_days_min < delinq_days_min_tmp) then
      delinq_days_min = delinq_days_min_tmp;

    balance_os = 0;

    sql_text = 'select oks.oks_from_k_id,
                       k_kontract_no,
                       kont.k_from_kf_id,
                       cast(:rep_date as date) - oks_delinq_date as delinq_days,
                       oks.oks_kc_id,
                       oks.oks_balance,
                       kont.k_last_reg_id,
                       kont.k_kolich_client_approved
                from oksummary oks
                inner join kontract kont on kont.k_id = oks.oks_from_k_id
                where :rep_date between oks_date_begin and oks_date_end and
                      oks_delinq_date < :rep_date and
                      cast(:rep_date as date) - oks_delinq_date between :delinq_days_min and :delinq_days_max and
                      oks.oks_kc_id <> 1000000000004 and
                      ((oks.oks_balance > :min_balance) or
                      (:db_id = 94401))';
     if (k_id_tmp is not null) then
        sql_text = sql_text||' and kont.k_id = :k_id_tmp';
    else
        sql_text = sql_text||' and :k_id_tmp is null';

    if (kf_id_tmp is not null) then
        sql_text = sql_text||' and kont.k_from_kf_id = :kf_id_tmp';
    else
        sql_text = sql_text||' and :kf_id_tmp is null';

     for execute statement(sql_text)
        (rep_date := rep_date,
         delinq_days_min := delinq_days_min,
         delinq_days_max := delinq_days_max,
         min_balance := min_balance,
         k_id_tmp := k_id_tmp,
         kf_id_tmp := kf_id_tmp,
         db_id := db_id)
    into :k_id, :k_kontract_no, :kf_id, :delinq_days, :oks_kc_id, :balance_os,
         :reg_id, :cli_qty_counter
    do
    begin
      if (exists(select *
               from sb_akt sba
               where sba.sba_from_k_id = :k_id and
                     sba.sba_from_sbl_id <> 0 and
                     sba.sba_is_delete = 0 and
                     sba.sba_status > 0 and
                     :rep_date between sba.sba_date_begin and sba.sba_date_end)) then
        continue;

      delinq_os = 0;
      select sum(ok_os + ok_sh_os)
      from operkred
      where ok_from_k_id = :k_id and
            ok_from_c_id = 0 and
            ok_ol_date <= :rep_date and
            ok_sh_date < :rep_date and
            ok_ol_is_active = 1 and
            ok_type <= 7
      into :delinq_os;

      balance_os = coalesce(balance_os, 0);
      delinq_os = coalesce(delinq_os, 0);

      if (min_balance = 0) then
        suspend;
      else
      if (not((balance_os < min_balance) or (delinq_os < min_balance))) then
        suspend;
    end
  end
end
