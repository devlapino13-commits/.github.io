CREATE PROCEDURE AAREP_QR_PAYMENTS(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    IQP_DATE D_DATE,
    IQP_IPC_DATETIME D_DATETIME,
    IQP_DATETIME D_DATETIME,
    C_NAME_FIO D_STR_100,
    CS_NAME D_STR_100,
    IQP_TRANSACTION D_STR_100,
    IQP_CHECK_NUMBER D_STR_100,
    IQP_AMOUNT D_CURRENCY,
    IQP_FEE D_CURRENCY,
    COMMIS_PERC D_PERCENT)
AS
declare variable rd_id_tsp d_integer;
declare variable rd_id_tsp_ipf d_integer;
declare variable rd_id_ipf_sellers d_integer;
declare variable cl_obj_kind d_integer;
declare variable c_id d_id;
declare variable cs_id d_id;
begin
  rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');
  rd_id_tsp_ipf = f_get_rd_id('AgreementKinds.tsp_and_ipf');
  rd_id_ipf_sellers = f_get_rd_id('AgreementKinds.IPF_Sellers');
  cl_obj_kind = f_objkind('clients', 'id');

  for select cl.c_name_fio,
             cs.cs_name,
             cl.c_id,
             cs.cs_id
  from clientshops cs
  inner join clients cl on cl.c_id = cs.cs_from_c_id
  where cs.cs_from_rd_id = :rd_id_tsp
  into :c_name_fio,
       :cs_name,
       :c_id,
       :cs_id
  do
  begin
    commis_perc = 0;
    select coalesce(ag.ag_commis_cards_other, 0)
    from agreements ag
    where ag.ag_object_kind = :cl_obj_kind and
          ag.ag_object_id = :c_id and
          ag.ag_status = 1 and
          :rep_date_end between ag.ag_date_begin and coalesce(ag.ag_date_end, '31.12.9999') and
          ag.ag_from_rd_id in (:rd_id_tsp, :rd_id_tsp_ipf)
    into commis_perc;

    for select qp.iqp_date,
           qp.iqp_ipc_datetime,
           qp.iqp_datetime,
           qp.iqp_transaction,
           qp.iqp_check_number,
           qp.iqp_amount,
           qp.iqp_fee
    from ipc_qr_payments qp
    where qp.iqp_from_cs_id = :cs_id and
          qp.iqp_date between :rep_date_begin and :rep_date_end
    order by qp.iqp_ipc_datetime
    into :iqp_date,
         :iqp_ipc_datetime,
         :iqp_datetime,
         :iqp_transaction,
         :iqp_check_number,
         :iqp_amount,
         :iqp_fee
    do
    begin
      suspend;
    end
  end
end
