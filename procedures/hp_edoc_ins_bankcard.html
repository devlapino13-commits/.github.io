CREATE PROCEDURE HP_EDOC_INS_BANKCARD(
    EDOC_KIND D_STR_CODE,
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
begin
  res_text = '';

  if (f_clientcom_allowed(c_id, 1, 1, 1) = 0) then
    res_text = 'Отправка сообщений данному клиенту запрещена по причине отсутствия согласий';

  if (res_text = '') then
  begin
    select edl_id, res_text
    from hp_edoc_insert(:edoc_kind, f_objkind('clients', 'id'), :c_id, :un_id)
    into edl_id, res_text;
  end

  suspend;
end
