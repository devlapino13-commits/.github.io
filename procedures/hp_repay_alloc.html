CREATE PROCEDURE HP_REPAY_ALLOC(
    REP_DATE DATE,
    REP_K_ID D_ID,
    REPAY_AMOUNT DECIMAL(18,4),
    REPAY_TO_OS INTEGER,
    REPAY_TO_PERC INTEGER,
    REPAY_TO_FINE INTEGER,
    REPAY_TO_KOMMIS INTEGER,
    REPAY_IS_WRTITEOFF D_SMALLINT = null)
AS
declare variable cs_pros integer;
declare variable cs_pred integer;
declare variable k_disb_date date;
declare variable cm_database_id integer;
declare variable s_rec_no integer;
declare variable rep_saldo decimal(18,4);
declare variable client_koef double precision;
declare variable pr_percent_method integer;
declare variable is_extra_at_end d_smallint;
declare variable vac_date_begin d_date;
begin
  cm_database_id = f_get_db_id();

  if (repay_is_wrtiteoff is null) then
    repay_is_wrtiteoff = 0;

  update temp_vrepay
  set pay_os = 0,
      pay_perc = 0,
      pay_fine = 0,
      pay_tax = 0,
      pay_kommis = 0,
      pay_extra = 0,
      resh_perc = 0,
      sh_new_os = 0,
      sh_new_perc = 0,
      depo_amount = 0;

-- Чтение необходимых параметров кредита
  select cs_pros, cs_pred, k_disb_date, client_koef, pr_percent_method, vac_date_begin
  from temp_params
  into :cs_pros, :cs_pred, :k_disb_date, :client_koef, pr_percent_method, vac_date_begin;

  -- RFCRU-1800 Изменение очередности распределения погашений по займам с 01.07.24: пени после срочных %% и ОД
  if (cm_database_id = 81001 and rep_date >= cast('01.07.2024' as date)) then
    cs_pros = 7;

  if (client_koef <> 1) then
  begin
-- Вычисление доли клиента из распределенной части
    execute procedure hp_repay_client_part;
    exit;
  end
  if (repay_amount <= 0) then
    exit;
  rep_saldo = repay_amount;

  if ((k_disb_date between '01.10.2013' and '30.06.2014') and
      (cm_database_id = 81001) and
      (cs_pros <> 5)) then
  begin
-- если выдачи после 01.10.2013 и РФ, то распределение: Проц, ОС, Штрафы
    cs_pros = 5;
  end

  is_extra_at_end = 0;
  if (cm_database_id = 81001 and rep_date >= '01.01.2020') then
    is_extra_at_end = 1;

  if (cm_database_id = 39801) then
    is_extra_at_end = 1;
-- НАЧАЛО: Распределение доп.обязательств
  if (is_extra_at_end = 0) then
  begin
    select rep_saldo
    from hp_repay_alloc_tmp('Extra', :rep_date, 0, 0, 0, 0, :rep_saldo, 1, 0)
    into :rep_saldo;
  end
-- КОНЕЦ: Распределение доп.обязательств

  if (pr_percent_method = -1) then -- Финансирование по исламским принципам
  begin
    if (f_ipf_newrepmode(rep_date, rep_k_id, repay_is_wrtiteoff) = 1) then
    begin
      -- Все платежи распределяются пропорционально на ОС, переплаты в конце
      select rep_saldo
      from hp_repay_alloc_tmp('Proportional_ipf', :rep_date, 1 * :repay_to_os, 2 * :repay_to_perc, 3 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 1)
      into :rep_saldo;
    end else
    begin
      -- Все платежи распределяются пропорционально на ОС, наценку и НсП
      select rep_saldo
      from hp_repay_alloc_tmp('Proportional', :rep_date, 1 * :repay_to_os, 2 * :repay_to_perc, 3 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 1)
      into :rep_saldo;
    end
  end else
  begin
-- НАЧАЛО: Просроченные платежи, специальное распределение
    select rep_saldo
    from hp_repay_alloc_tmp('Vertical', :rep_date, 1 * :repay_to_os, 2 * :repay_to_perc, 3 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 1)
    into :rep_saldo;
-- КОНЕЦ: Просроченные платежи, специальное распределение

-- НАЧАЛО: Просроченные платежи, стандартное распределение по настройкам продукта
    if (cm_database_id = 39801) then -- RFCKZ-1031 Весь ОС, затем Проценты, после штрафы (все вертикально)
    begin
      select rep_saldo
      from hp_repay_alloc_tmp('Vertical', :rep_date, 1 * :repay_to_os, 2 * :repay_to_perc, 3 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
      into :rep_saldo;
    end else
    begin
      if (cs_pros = 1) then -- ВСЕ штрафы (вертикально), и после горизонтально Проценты, ОС
      begin
        select rep_saldo
        from hp_repay_alloc_tmp('Vertical', :rep_date, 0, 0, 1 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
  
        select rep_saldo
        from hp_repay_alloc_tmp('Horizontal', :rep_date, 2 * :repay_to_os, 1 * :repay_to_perc, 0, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
      end
  
      if (cs_pros = 2) then -- Горизонтально Штрафы, Проц, ОС
      begin
        select rep_saldo
        from hp_repay_alloc_tmp('Horizontal', :rep_date, 3 * :repay_to_os, 2 * :repay_to_perc, 1 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
      end
  
      if (cs_pros = 3) then -- ВСЕ штрафы (вертикально), и после горизонтально ОС, Проценты
      begin
        select rep_saldo
        from hp_repay_alloc_tmp('Vertical', :rep_date, 0, 0, 1 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
  
        select rep_saldo
        from hp_repay_alloc_tmp('Horizontal', :rep_date, 1 * :repay_to_os, 2 * :repay_to_perc, 0, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
      end
  
      if (cs_pros = 4) then -- Горизонтально Штрафы, ОС, Проценты
      begin
        select rep_saldo
        from hp_repay_alloc_tmp('Horizontal', :rep_date, 2 * :repay_to_os, 3 * :repay_to_perc, 1 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
      end
  
      if (cs_pros = 5) then -- Проц, ОС, Штрафы
      begin
        select rep_saldo
        from hp_repay_alloc_tmp('Horizontal', :rep_date, 2 * :repay_to_os, 1 * :repay_to_perc, 3 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
      end
  
      if (cs_pros = 6) then -- Все вертикально: все проценты, все ОС и потом все пени
      begin
        select rep_saldo
        from hp_repay_alloc_tmp('Vertical', :rep_date, 2 * :repay_to_os, 1 * :repay_to_perc, 3 * :repay_to_fine, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
      end
  
      if (cs_pros = 7) then -- Все вертикально: все проценты, все ОС и потом все пени
      begin
        select rep_saldo
        from hp_repay_alloc_tmp('Vertical', :rep_date, 2 * :repay_to_os, 1 * :repay_to_perc, 0, 4 * :repay_to_kommis, :rep_saldo, 1, 0)
        into :rep_saldo;
  
        select rep_saldo
        from hp_repay_alloc_tmp('Horizontal', :rep_date, 2 * :repay_to_os, 1 * :repay_to_perc, 0, 4 * :repay_to_kommis, :rep_saldo, 2, 0)
        into :rep_saldo;
  
        select rep_saldo
        from hp_repay_alloc_tmp('Vertical', :rep_date, 0, 0, 1, 0, :rep_saldo, 1, 0)
        into :rep_saldo;
      end
   end

-- КОНЕЦ: Просроченные платежи, стандартное распределение по настройкам продукта

-- НАЧАЛО: Текущий платеж
    if (cm_database_id = 39801) then -- RFCKZ-1031 ОС, затем процент (все горизонтально)
    begin
      select rep_saldo
        from hp_repay_alloc_tmp('Horizontal', :rep_date, 1 * :repay_to_os, 2 * :repay_to_perc, 0, 4 * :repay_to_kommis, :rep_saldo, 2, 0)
        into :rep_saldo;
    end else
    begin
      if (cs_pros <> 7) then
      begin
        select rep_saldo
        from hp_repay_alloc_tmp('Horizontal', :rep_date, 2 * :repay_to_os, 1 * :repay_to_perc, 0, 4 * :repay_to_kommis, :rep_saldo, 2, 0)
        into :rep_saldo;
      end
    end
-- КОНЕЦ: Текущий платеж

-- НАЧАЛО: Будущие платежи
    if (cs_pred = 2) then -- ОС с очередного погашения
    begin
      select rep_saldo
      from hp_repay_alloc_tmp('Horizontal', :rep_date, 2 * :repay_to_os, 1 * :repay_to_perc, 0, 4 * :repay_to_kommis, :rep_saldo, 3, 0)
      into :rep_saldo;
    end

    if (cs_pred = 3) then -- ОС с последнего погашения (Special_Mode=-1)
    begin
      select rep_saldo
      from hp_repay_alloc_tmp('Horizontal', :rep_date, 2 * :repay_to_os, 1 * :repay_to_perc, 0, 4 * :repay_to_kommis, :rep_saldo, 3, -1)
      into :rep_saldo;
    end
-- КОНЕЦ: Будущие платежи
  end

-- НАЧАЛО: Распределение доп.обязательств
  if (is_extra_at_end = 1) then
  begin
    select rep_saldo
    from hp_repay_alloc_tmp('Extra', :rep_date, 0, 0, 0, 0, :rep_saldo, 1, 0)
    into :rep_saldo;
  end
-- КОНЕЦ: Распределение доп.обязательств

-- Если Rep_Saldo>0, значит это переплата
  if (rep_saldo > 0) then
  begin
    select max(s_rec_no)
    from temp_vrepay
    into :s_rec_no;

    update temp_vrepay
    set pay_pereplat = pay_pereplat + :rep_saldo
    where s_rec_no = :s_rec_no;
  end

-- Если после распределения баланс нулевой, скриптом приравниваем Pay_OS на Zad_OS;
  rep_saldo = 0;
  select sum(zad_os - pay_os)
  from temp_vrepay
  into :rep_saldo;
  if (rep_saldo is null) then
    rep_saldo = 0;

  if (rep_saldo <= 0 and vac_date_begin is null) then
  begin
    update temp_vrepay
    set pay_os = zad_os;
  end
end
