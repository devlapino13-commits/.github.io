CREATE PROCEDURE HP_TRANSFER_PERECOST_TO_OB
AS
declare variable cpc_from_ol_id decimal(18,0);
declare variable cpc_number integer;
declare variable cpc_from_al_id varchar(20);
declare variable cpc_summa_plus decimal(18,4);
declare variable cpc_summa_minus decimal(18,4);
declare variable cpc_from_cu_id varchar(3);
declare variable acc varchar(20);
declare variable acc_format varchar(20);
declare variable reg_code varchar(10);
begin
  select cm_account_format
  from company
  into :acc_format;

  for select cpc_from_ol_id, cpc_number, cpc_from_al_id, cpc_summa_plus, cpc_summa_minus, cpc_from_cu_id, reg_code
      from currperecost, operlist, region
      where cpc_from_ol_id = ol_id and
            ol_from_reg_id = reg_unicode
      into :cpc_from_ol_id, :cpc_number, :cpc_from_al_id, :cpc_summa_plus, :cpc_summa_minus, :cpc_from_cu_id,
           :reg_code
  do
  begin
    if (acc_format = 'BBBBBCCCOOOONNNNNN') then
      acc = '11520' || :cpc_from_cu_id || :reg_code || '000001';
    if (acc_format = 'OOOOCCCBBBBBNNNNNN') then
      acc = :reg_code || :cpc_from_cu_id || '11520' || '000001';

    if (cpc_summa_plus > 0) then
    begin
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                            ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                            ob_client_kt, ob_detail)
      values (:cpc_from_ol_id, -1, :cpc_from_cu_id, 1, null, :cpc_from_al_id, 0, :cpc_summa_plus, null, null, null,
              :acc, null, 'IMP: Перекурсовка счета');
    end
    if (cpc_summa_minus > 0) then
    begin
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                            ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                            ob_client_kt, ob_detail)
      values (:cpc_from_ol_id, -1, :cpc_from_cu_id, 1, null, :acc, 0, :cpc_summa_minus, null, null, null,
              :cpc_from_al_id, null, 'IMP: Перекурсовка счета');
    end
  end
end
