CREATE PROCEDURE AAREP_CLIENT_YUR_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_NUMBER INTEGER,
    RES_REG_NAME VARCHAR(50),
    RES_C_ID ID_TYPE,
    RES_C_NAME STR_100,
    RES_C_INN STR_20,
    RES_C_OKPO STR_20,
    RES_LEGALLY_ADDR STR_255,
    RES_ACTUAL_ADDR STR_255,
    RES_COUNTRY STR_255,
    RES_RESIDENT STR_20,
    RES_TYPE_ACTIVITY STR_255,
    RES_RISK_LEVEL STR_20,
    RES_NUMBER_EMPL INTEGER,
    RES_DATE_OPEN_ACC DATE,
    RES_UN_NAME STR_100,
    RES_C_DATETIME TIMESTAMP,
    RES_BEN_C_ID ID_TYPE,
    RES_C_NAME_OWNER STR_100,
    RES_C_NAME_POSITION STR_100,
    RES_BEN_C_INN STR_20,
    RES_BEN_C_BIRTH_DATE DATE,
    RES_BEN_LEGALLY_ADDR STR_255,
    RES_BEN_ACTUAL_ADDR STR_255,
    RES_BEN_DOC_NAME STR_100,
    RES_BEN_NATIONALITY STR_255,
    RES_BEN_PUBLIC STR_20,
    RES_BEN_PERCENT DECIMAL(5,2),
    RES_TYPE_CONTR STR_20,
    RES_IS_FOUNDER STR_20)
AS
declare variable number_benefic integer;
begin
  rep_date_end = rep_date_end + 1;
  res_number = 0;
  for select reg_name, c_id, c_name_fio, c_inn, c_okpo, rd_name, iif(c_is_resident = 1, 'да', 'нет'),
             c_org_type_long,
             case c_pft_risk_koef
               when 0 then 'низкий'
               when 1 then 'средний'
               when 2 then 'высокий'
             end,
             c_kol_family, un_fio, c_datetime,
             case c_type_contr
               when 0 then 'Хоз.деятельность'
               when 1 then 'ИПФ'
               else ''
             end,
             iif(c_is_founder = 1, 'да', 'нет')
      from clients
      inner join region_list(:rep_region_id, 1, 3, 1) on id_list = c_from_reg_id
      inner join region on reg_unicode = c_from_reg_id
      left join refdata on rd_id = c_citizen
      left join username on un_id = c_from_un_id
      where c_datetime >= :rep_date_begin and
            c_datetime < :rep_date_end and
            c_type = 1
      order by c_name_fio
      into :res_reg_name, :res_c_id, :res_c_name, :res_c_inn, :res_c_okpo, :res_country, :res_resident,
           :res_type_activity, :res_risk_level, :res_number_empl, :res_un_name, :res_c_datetime, :res_type_contr,
           :res_is_founder
  do
  begin
    res_number = res_number + 1;
    res_date_open_acc = (select min(ol_date)
                         from operlist
                         where ol_document_client = :res_c_id);
    res_legally_addr = null;
    res_legally_addr = (select adl_address_str
                        from addrclient
                        inner join addrlist on adl_id = adc_from_adl_id
                        where adc_from_c_id = :res_c_id and
                              adc_address_type = 1);
    res_actual_addr = null;
    res_actual_addr = (select adl_address_str
                       from addrclient
                       inner join addrlist on adl_id = adc_from_adl_id
                       where adc_from_c_id = :res_c_id and
                             adc_address_type = 2);
    number_benefic = 0;
    res_ben_c_id = null;
    res_c_name_owner = null;
    res_c_name_position = null;
    res_ben_c_inn = null;
    res_ben_c_birth_date = null;
    res_ben_doc_name = null;
    res_ben_nationality = null;
    res_ben_public = null;
    res_ben_percent = null;
    res_ben_legally_addr = null;
    res_ben_actual_addr = null;

    for select c_id, iif(coalesce(cb_position, 0) > 0, cb_ben_name, ''), iif(cb_is_owner = 1, cb_ben_name, ''), c_inn,
               c_birth_date, pk_name, rd.rd_name, iif(c_is_public = 1, 'да', 'нет'), cb_percent
        from clientbenefic
        left join clients on c_id = cb_ben_from_c_id
        left join passkind on pk_id = c_doc_type
        left join refdata rd on rd.rd_id = c_citizen
        where cb_from_c_id = :res_c_id
        into :res_ben_c_id, :res_c_name_owner, :res_c_name_position, :res_ben_c_inn, :res_ben_c_birth_date,
             :res_ben_doc_name, :res_ben_nationality, :res_ben_public, :res_ben_percent
    do
    begin
      number_benefic = number_benefic + 1;
      if (:res_ben_c_id > 0) then begin
        res_ben_legally_addr = (select adl_address_str
                              from addrclient
                              inner join addrlist on adl_id = adc_from_adl_id
                              where adc_from_c_id = :res_ben_c_id and
                                    adc_address_type = 1);
        res_ben_actual_addr = (select adl_address_str
                             from addrclient
                             inner join addrlist on adl_id = adc_from_adl_id
                             where adc_from_c_id = :res_ben_c_id and
                                   adc_address_type = 2);
      end
      suspend;
    end
    if (number_benefic = 0) then
      suspend;
  end
end
