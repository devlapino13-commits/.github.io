CREATE PROCEDURE HP_FIND_REPANALITICATEMPLATE(
    UN_ID DECIMAL(18,0))
RETURNS (
    RAT_ID INTEGER,
    RAT_NAME VARCHAR(255),
    RAT_MEMO VARCHAR(255),
    RAT_ORDERNO INTEGER)
AS
begin
  for select rat_id, rat_name, rat_memo, rat_orderno
      from repanaliticatemplate
      where rat_from_un_id = :un_id or
            rat_id = 0
      order by rat_used_count desc, rat_orderno
      into :rat_id, :rat_name, :rat_memo, :rat_orderno
  do
  begin
    suspend;
  end
end
