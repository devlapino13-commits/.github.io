CREATE PROCEDURE AAREP_ACT_KONTRACTS_DETAIL(
    C_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    NUMBER INTEGER,
    KFOLDER_NAME VARCHAR(255))
AS
declare variable kf_name varchar(50);
declare variable number1 integer;
declare variable kf_name_real_old varchar(50);
declare variable kf_name_real varchar(50);
begin
  kf_name_real_old = '';
  for select kal_new_kf_name,
             kal.kal_new_kf_name || ' от ' || iif(extract(day from k_fact_disb_date) < 10, '0' || extract(day from k_fact_disb_date), extract(day from k_fact_disb_date)) || '.' || iif(extract(month from k_fact_disb_date) < 10, '0' || extract(month from k_fact_disb_date), extract(month from k_fact_disb_date)) || '.' || extract(year from k_fact_disb_date)
      from kontract k, kontstate ks, knaktlist kal
      where k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_state in (1, 2) and
            kal.is_delete = 0 and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.kal_from_kf_id = k.k_from_kf_id and
            kal.kal_new_kf_officer = :c_id and
            k.k_fact_disb_date <= '31.07.2016' --ks.ks_date_begin <= '31.07.2016'
      order by kal.kal_new_officer_code, kal.kal_new_no, ks.ks_date_begin desc
      into kf_name_real, kf_name
  do
  begin
    if (kf_name_real <> kf_name_real_old) then
    begin
      kf_name_real_old = kf_name_real;

      if (number1 = 4 or number1 is null) then
        number1 = 1;
      else
        number1 = number1 + 1;

      if (number1 < 4) then
      begin
        kfolder_name = coalesce(kfolder_name || ', ', '') || :kf_name;
      end

      if (number1 = 4) then
      begin
        number = coalesce(number + 1, 1);

        kfolder_name = kfolder_name || ', ' || :kf_name;

        suspend;
        kfolder_name = null;
      end
    end
  end
  suspend;
end
