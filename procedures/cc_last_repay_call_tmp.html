CREATE PROCEDURE CC_LAST_REPAY_CALL_TMP(
    KFOLDER_NAME STR_100,
    CITIZEN INTEGER)
RETURNS (
    K_ID ID_TYPE,
    KF_NAME STR_100,
    MAX_DELINQ_DAY INTEGER,
    REG_NAME STR_100,
    C_ID ID_TYPE,
    C_PLASTCARD_NO STR_100,
    CITIZEN_STR STR_100,
    CK_CALL_QTY SMALLINT,
    CK_IS_BLOCKED SMALLINT,
    CK_BLOCKED_OPER STR_100,
    CK_CHECKED_OPER STR_100,
    CK_CHECKED_DATE DATE,
    SH_DATE DATE,
    C_NAME_FIO STR_100)
AS
declare variable rep_date date;
declare variable sh_qty smallint;
declare variable delinq_day_repayment integer;
declare variable oks_delinq_day integer;
declare variable reg_id integer;
declare variable reg_id_tmp integer;
declare variable ck_status smallint;
declare variable is_ok smallint;
declare variable rd_id integer;
declare variable ok_sh_date_tmp date;
declare variable max_delinq_days smallint;
begin
  citizen = coalesce(citizen, 0);
  kfolder_name = coalesce(kfolder_name, '');

  max_delinq_days = 30;

  rep_date = current_date;
  reg_id_tmp = 0;

  for select oks.oks_from_k_id, kal.kal_new_kf_name, kal.kal_new_reg_id,
             (cast(:rep_date as date) - oks_delinq_date) oks_delinq_day
      from oksummary oks
      inner join kontract kont on kont.k_id = oks.oks_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      where oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end and
            cast(:rep_date as date) - oks_delinq_date < :max_delinq_days and
            oks.oks_kc_id = 1000000000004 -- КЦ Сирена
            and
            kal.is_delete = 0 and
            ((kal.kal_new_kf_name = :kfolder_name) or
            (:kfolder_name = '')) and
            kal.kal_new_kf_officer = 1000000000004 -- КЦ Сирена
      order by oks.oks_region_id
      into :k_id, :kf_name, :reg_id, :oks_delinq_day
  do
  begin
    sh_qty = 0;
    sh_date = null;

    for select ok_sh_date
        from operkred ok
        where ok_type < 7 and
              ok_ol_is_active = 1 and
              ok_from_k_id = :k_id and
              ok_ol_date <= :rep_date and
              ok.ok_from_c_id = 0
        group by ok_sh_date
        having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
        order by ok_sh_date
        into :ok_sh_date_tmp
    do
    begin
      sh_date = ok_sh_date_tmp;
      sh_qty = sh_qty + 1;
    end

    if (sh_qty = 1) then -- если осталось последнее погашение, тогда
    begin
-- вычисляю максимальную просрочку

      max_delinq_day = 0; -- Максимальный срок задержки кредита
      delinq_day_repayment = 0;

      select max(iif(oks_date_end = '30.12.9999', cast(:rep_date as date) - oks_delinq_date, oks_date_end + 1 - oks_delinq_date))
      from oksummary oks
      where oks_close_state = 0 and
            oks_from_k_id = :k_id
      into :delinq_day_repayment;

      if (delinq_day_repayment is null) then
        delinq_day_repayment = 0;
      if (oks_delinq_day is null) then
        oks_delinq_day = 0;

      if (oks_delinq_day > delinq_day_repayment) then
        max_delinq_day = oks_delinq_day;
      else
        max_delinq_day = delinq_day_repayment;
      if (max_delinq_day < 0) then
        max_delinq_day = 0;

      if (max_delinq_day < :max_delinq_days) then -- если max число дней просрочки меньше 30, тогда пропускаю
      begin
--is_ok = 0;
--if (is_ok = 1) then
--begin
        if (reg_id <> reg_id_tmp) then
        begin
          select reg.reg_name
          from region reg
          where reg.reg_unicode = :reg_id
          into :reg_name;

          reg_id_tmp = reg_id;
        end

        for select kncl.kc_from_c_id,
                   (select cont_value
                    from hp_contdata_get_cell_valid(0, kncl.kc_from_c_id)), rd.rd_name, rd.rd_id, cl.c_name_fio
            from knclient kncl
            inner join clients cl on cl.c_id = kncl.kc_from_c_id
            inner join refdata rd on rd.rd_id = cl.c_citizen
            where kncl.kc_from_k_id = :k_id and
                  kncl.kc_amount_approved > 0
            into :c_id, :c_plastcard_no, :citizen_str, rd_id, :c_name_fio
        do
        begin
          is_ok = 0;

          ck_status = null;
          ck_call_qty = null;

          ck_is_blocked = null;
          ck_blocked_oper = null;
          ck_checked_oper = null;
          ck_checked_date = null;

          select ck.ck_call_qty, ck.ck_status, ck.ck_is_blocked, ck.ck_blocked_oper, ck.ck_checked_oper,
                 ck.ck_checked_date
          from cc_check_knclient ck
          where ck.ck_type = 3 and
                ck.ck_from_k_id = :k_id and
                ck.ck_from_c_id = :c_id
          into :ck_call_qty, :ck_status, :ck_is_blocked, :ck_blocked_oper, :ck_checked_oper, :ck_checked_date;

          ck_call_qty = coalesce(ck_call_qty, 0);
          ck_status = coalesce(ck_status, 0);

-- ck_status = 0  -- не звонили еще
-- ck_status = -1 -- не дозвонились

-- дается только 3 попытки дозвона
          if ((ck_status in (-1, 0)) and
              (ck_call_qty < 3)) then
            is_ok = 1;

          if (is_ok = 1) then
          begin
            if (citizen = 0) then
              suspend;
            else
            if (citizen = rd_id) then
              suspend;
          end
        end
--end
      end
    end
  end
end
