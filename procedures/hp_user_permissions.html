CREATE PROCEDURE HP_USER_PERMISSIONS(
    UN_ID DECIMAL(18,0))
RETURNS (
    ML_ID INTEGER,
    ML_PARENT_ID INTEGER,
    ML_CODE VARCHAR(100),
    ML_PARENT_CODE VARCHAR(100),
    ML_CAPTION VARCHAR(255),
    ML_DESCRIPTION VARCHAR(1024),
    ML_IS_REPORT INTEGER,
    ML_SECURE_KIND INTEGER,
    ML_LEVEL VARCHAR(10),
    ML_ORDER_NO INTEGER,
    ML_HAS_CHILD INTEGER)
AS
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
declare variable uniqtab_menulist integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;

  for with tab
      as (select distinct me.pm_subject_id as ml_id, me.pm_access_kind as ml_access_kind
          from permissions ro, permissions me
          where ro.pm_object_kind = :uniqtab_username and
                ro.pm_object_id = :un_id and
                ro.pm_subject_kind = :uniqtab_rolename and
                ro.pm_is_delete = 0 and
                ro.pm_subject_id = me.pm_object_id and
                ro.pm_access_kind <> 0 and
                (coalesce(ro.pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(ro.pm_date_end, '31.12.2999') >= current_date) and
                me.pm_object_kind = :uniqtab_rolename and
                me.pm_subject_kind = :uniqtab_menulist and
                me.pm_is_delete = 0 and
                me.pm_access_kind <> 0 and
                (coalesce(me.pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(me.pm_date_end, '31.12.2999') >= current_date)
          union
          select distinct me.pm_subject_id as ml_id, me.pm_access_kind as ml_access_kind
          from permissions me
          where me.pm_object_kind = :uniqtab_username and
                me.pm_object_id = :un_id and
                me.pm_subject_kind = :uniqtab_menulist and
                me.pm_is_delete = 0 and
                me.pm_access_kind <> 0 and
                (coalesce(me.pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(me.pm_date_end, '31.12.2999') >= current_date))
      select ml_id
      from tab
      group by ml_id
      having min(ml_access_kind) = 1
      into :ml_id
  do
  begin
    select ml_parent_id, ml_code, ml_parent_code, ml_caption, ml_description, ml_is_report, ml_secure_kind, ml_level,
           ml_has_child, ml_position
    from menulist
    where ml_id = :ml_id
    into :ml_parent_id, :ml_code, :ml_parent_code, :ml_caption, :ml_description, :ml_is_report, :ml_secure_kind,
         :ml_level, :ml_has_child, ml_order_no;

    if (ml_is_report is null) then
      ml_is_report = 0;

    if (ml_id > 0) then
      ml_is_report = 1;

    suspend;
  end
end
