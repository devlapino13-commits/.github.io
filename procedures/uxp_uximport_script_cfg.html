CREATE PROCEDURE UXP$UXIMPORT_SCRIPT_CFG(
    TABLE_NAME D_STR_CODE)
RETURNS (
    SQL_BLOCK_EXTERNAL VARCHAR(8000) CHARACTER SET WIN1251,
    SQL_BLOCK_LOCAL VARCHAR(8000) CHARACTER SET WIN1251,
    SQL_SELECT VARCHAR(8000) CHARACTER SET WIN1251,
    SQL_INSERT VARCHAR(8000) CHARACTER SET WIN1251,
    SQL_EMPTY_SELECT VARCHAR(8000) CHARACTER SET WIN1251)
AS
declare variable uuid_field d_str_code;
declare variable matching_fields d_str_255;
declare variable replace_uuid_fields d_str_long;
declare variable declare_vars varchar(32000) character set win1251;
declare variable field_name d_str_code;
declare variable field_domain d_str_code;
declare variable select_fields varchar(4096) character set win1251;
begin
  if (trim(coalesce(table_name, '')) = '') then
    exception e_default 'Empty table name';

  select uuid_field, matching_fields, replace_uuid_fields
  from uxp$uximport_cfg
  where table_name = :table_name
  into :uuid_field, :matching_fields, :replace_uuid_fields;

  if (row_count = 0) then
    exception e_default 'Sync config not found, table: ' || table_name;


  sql_empty_select = 'select * from [$$$table_name$$$] where [$$$uuid_field$$$] is null rows 0';

  if (table_name = 'ux$pairuuid') then
  begin
    sql_block_external = '-- Start import: [$$$table_name$$$]
-- SET TERM ^ ;
execute block (exp_id d_integer = :exp_id)
as
[$$$declare_variables$$$]
begin
  uxf$syncmode(''on'');

  for execute statement (''
  select exi_ux_uuid as up_old_uuid,
         iif(exi_use_own_uuid = 1, exi_ux_uuid, null) as up_new_uuid
  from ux$expoitem, ux$export
  where exp_id = ? and
        exp_id = exi_exp_id and
        exp_index = exi_exp_index
  order by exi_exp_id desc, exi_exp_index desc, exi_index'')
  (:exp_id)
  on external uxf$syncpath('''')
  as user current_user password uxf$pswd() role ''RDB$ADMIN''
  into [$$$select_fields$$$]
  do
  begin
    update or insert into [$$$table_name$$$] ([$$$select_fields$$$])
    values ([$$$field_values$$$])
    matching ([$$$matching_fields$$$]);
  end

  uxf$syncmode(''off'');
end
--^
--SET TERM ; ^


';


    sql_block_local = '-- Start import: [$$$table_name$$$]
-- SET TERM ^ ;
execute block (exp_id d_integer = :exp_id)
as
begin
  uxf$syncmode(''on'');

  for select exi_ux_uuid as up_old_uuid,
             iif(exi_use_own_uuid = 1, exi_ux_uuid, null) as up_new_uuid
  from ux$expoitem, ux$export
  where exp_id = :exp_id and
        exp_id = exi_exp_id and
        exp_index = exi_exp_index
  order by exi_exp_id desc, exi_exp_index desc, exi_index
  as cursor cur
  do
  begin
    update or insert into [$$$table_name$$$] ([$$$select_fields$$$])
    values ([$$$field_values$$$])
    matching ([$$$matching_fields$$$]);
  end

  uxf$syncmode(''off'');
end
-- ^
-- SET TERM ; ^


';


    sql_select = 'select null as exp_ufm_uuid, exi_ux_uuid as up_old_uuid,
           iif(exi_use_own_uuid = 1, exi_ux_uuid, null) as up_new_uuid
  from ux$expoitem, ux$export
  where exp_id = :exp_id and
        exp_id = exi_exp_id and
        exp_index = exi_exp_index
  order by exi_exp_id desc, exi_exp_index desc, exi_index';

    sql_insert = 'update or insert into [$$$table_name$$$] ([$$$select_fields$$$])
    values ([$$$field_values$$$])
    matching ([$$$matching_fields$$$])';

  end else
  begin
    sql_block_external = '-- Start import: [$$$table_name$$$]
-- SET TERM ^ ;
execute block (exp_id d_integer = :exp_id)
as
[$$$declare_variables$$$]
begin
  uxf$syncmode(''on'');

  for execute statement (''
  select [$$$select_fields$$$]
  from [$$$table_name$$$], ux$expoitem, ux$export
  where exp_id = ? and
        exi_ux_uuid = [$$$uuid_field$$$] and
        exp_id = exi_exp_id and
        exp_index = exi_exp_index
  order by exi_exp_id desc, exi_exp_index desc, exi_index'')
  (:exp_id)
  on external uxf$syncpath()
  as user current_user password uxf$pswd() role ''RDB$ADMIN''
  into [$$$select_fields$$$]
  do
  begin
    update or insert into [$$$table_name$$$] ([$$$select_fields$$$])
    values ([$$$field_values$$$])
    matching ([$$$matching_fields$$$]);
  end

  uxf$syncmode(''off'');
end
-- ^
-- SET TERM ; ^


';

    sql_block_local = '-- Start import: [$$$table_name$$$]
-- SET TERM ^ ;
execute block (exp_id d_integer = :exp_id)
as
begin
  uxf$syncmode(''on'');

  for select [$$$select_fields$$$]
  from [$$$table_name$$$], ux$expoitem, ux$export
  where exp_id = :exp_id and
        exi_ux_uuid = [$$$uuid_field$$$] and
        exp_id = exi_exp_id and
        exp_index = exi_exp_index
  order by exi_exp_id desc, exi_exp_index desc, exi_index
  as cursor cur
  do
  begin
    update or insert into [$$$table_name$$$] ([$$$select_fields$$$])
    values ([$$$field_values$$$])
    matching ([$$$matching_fields$$$]);
  end

  uxf$syncmode(''off'');
end
-- ^
-- SET TERM ; ^


';

    sql_select = 'select exp_ufm_uuid, [$$$select_fields$$$]
  from [$$$table_name$$$], ux$expoitem, ux$export
  where exp_id = :exp_id and
        exi_ux_uuid = [$$$uuid_field$$$] and
        exp_id = exi_exp_id and
        exp_index = exi_exp_index
  order by exi_exp_id desc, exi_exp_index desc, exi_index';

    sql_insert = 'update or insert into [$$$table_name$$$] ([$$$select_fields$$$])
    values ([$$$field_values$$$])
    matching ([$$$matching_fields$$$])';
  end

  select_fields = '';
  declare_vars = '';
  for select lower(trim(r.rdb$field_name)), lower(trim(r.rdb$field_source))
  from rdb$fields f, rdb$relation_fields r
  where f.rdb$field_name = r.rdb$field_source and
        r.rdb$system_flag = 0 and
        r.rdb$relation_name = upper(:table_name)
  order by r.rdb$field_position
  into field_name, field_domain
  do
  begin
    if (declare_vars <> '') then
      declare_vars = declare_vars || ascii_char(13);

    if (field_domain not starting 'd_') then
      exception e_default 'Non standart domain name: "' || field_domain || '" on ' || table_name || '.' || field_name;

    declare_vars = declare_vars || 'declare variable ' || field_name || ' ' || field_domain || ';';

    if (select_fields <> '') then
      select_fields = select_fields || ', ';

    select_fields = select_fields || field_name;
  end

  sql_empty_select = replace(sql_empty_select, '[$$$table_name$$$]', table_name);
  sql_empty_select = replace(sql_empty_select, '[$$$uuid_field$$$]', uuid_field);

  sql_block_external = replace(sql_block_external, '[$$$table_name$$$]', table_name);
  sql_block_external = replace(sql_block_external, '[$$$uuid_field$$$]', uuid_field);
  sql_block_external = replace(sql_block_external, '[$$$matching_fields$$$]', matching_fields);
  sql_block_external = replace(sql_block_external, '[$$$declare_variables$$$]', declare_vars);
  sql_block_external = replace(sql_block_external, '[$$$select_fields$$$]', select_fields);

  sql_block_local = replace(sql_block_local, '[$$$table_name$$$]', table_name);
  sql_block_local = replace(sql_block_local, '[$$$uuid_field$$$]', uuid_field);
  sql_block_local = replace(sql_block_local, '[$$$matching_fields$$$]', matching_fields);
  sql_block_local = replace(sql_block_local, '[$$$declare_variables$$$]', declare_vars);
  sql_block_local = replace(sql_block_local, '[$$$select_fields$$$]', select_fields);

  sql_select = replace(sql_select, '[$$$table_name$$$]', table_name);
  sql_select = replace(sql_select, '[$$$uuid_field$$$]', uuid_field);
  sql_select = replace(sql_select, '[$$$select_fields$$$]', select_fields);

  sql_insert = replace(sql_insert, '[$$$table_name$$$]', table_name);
  sql_insert = replace(sql_insert, '[$$$uuid_field$$$]', uuid_field);
  sql_insert = replace(sql_insert, '[$$$matching_fields$$$]', matching_fields);
  sql_insert = replace(sql_insert, '[$$$select_fields$$$]', select_fields);

  select_fields = replace(select_fields, ' ', ' :');
  select_fields = ':' || select_fields;

  if (replace_uuid_fields <> '') then
  begin
    for select trim(res_str)
    from uxp$split_str(:replace_uuid_fields, ',')
    into :field_name
    do
      select_fields = replace(select_fields, ':' || field_name, 'uxf$pair(:' || field_name || ')');
  end

  sql_block_external = replace(sql_block_external, '[$$$field_values$$$]', select_fields);

  sql_insert = replace(sql_insert, '[$$$field_values$$$]', select_fields);

  select_fields = replace(select_fields, ':', 'cur.');
  sql_block_local = replace(sql_block_local, '[$$$field_values$$$]', select_fields);

  suspend;
end
