CREATE PROCEDURE HP_EDOC_CANCEL(
    EDOC_KIND D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID)
AS
declare variable edc_id d_integer;
declare variable edc_code d_str_code;
declare variable edoc_state d_str_code;
declare variable edl_id d_id;
declare variable edl_status d_smallint;
declare variable obj_kind_str d_str_code;
declare variable edc_object_uniq d_smallint;
declare variable db_id d_id;
declare variable k_id d_id;
declare variable c_id d_id;
begin
  -- Процедура аннулирует электронный документ
  db_id = f_get_db_id();

  -- В РФ при переносе даты выдачи по тех. причине не аннулируем документы (RFCRU-5881)
  if (db_id = 81001
      and obj_kind = f_objkind('kontract', 'id')
      and obj_id = rdb$get_context('USER_TRANSACTION', 'MOVE_DISB_K_ID'))
  then
    exit;

  edc_id = f_edoc_kind(edoc_kind, 'id');

  select edc_code, edc_object_uniq
  from edoc_cfg
  where edc_id = :edc_id
  into edc_code, edc_object_uniq;

  if (edc_id is null) then
    exit;

  --Убрал возможность аннулировать заявку, после подписания эл. документа "договор займа"------------------
  if (obj_kind = f_objkind('miniapp', 'name') and :db_id in (39801, 41701)) then
  begin
  k_id = null;
  select kt.k_id
  from kontract kt
  where kt.k_from_ma_app_id = :obj_id and
        kt.k_id > 0
  into k_id;

  -- RFCKG-4858
  if (exists(select *
             from edoc_list edl
             inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
                and edc.edc_code = 'loan.agreement'
             where edl.edl_object_kind = f_objkind('kontract', 'id') and
                   edl.edl_object_id = :k_id and
                   edl.edl_status = 1)) then
    exception e_message('Нельзя аннулировать заявку после подписания кредитного договора');
  end
  ---------------------------------------------------------------------------------------------------------

  obj_kind = f_objkind(obj_kind, 'id');
  obj_kind_str = f_objkind(obj_kind, 'name');

  -- Возвращает статус электронного документа:
  edoc_state = f_edoc_state(edoc_kind, current_date, null, obj_kind, obj_id);
  if (edoc_state = 'inactive') then
    exit;

  -- если нет электронного документа, завершаем работу
  if (not exists(select *
                 from edoc_list
                 where edl_object_kind = :obj_kind and
                       edl_object_id = :obj_id and
                       edl_from_edc_id = :edc_id and
                       edl_status <> -1)) then
    exit;

  if (obj_kind_str = 'miniapp' and edc_code = 'loan.application') then
  begin
    if (exists(select *
               from kontract, kontstate
               where k_from_ma_app_id = :obj_id and
                     ks_from_k_id = k_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0)) then
      exception e_message f_format('Аннулирование документа запрещено. Кредит уже выдан (ma_id: %1)', obj_id);
  end

  if (obj_kind_str = 'kontract' and edc_code = 'loan.agreement') then
  begin
    if (exists(select *
               from kontstate
               where ks_from_k_id = :obj_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0)) then
      exception e_default f_format('Аннулирование документа запрещено. Кредит уже выдан (k_id: %1)', obj_id);
  end

  for select edl_id, edl_status
      from edoc_list
      where edl_object_kind = :obj_kind and
            edl_object_id = :obj_id and
            edl_from_edc_id = :edc_id and
            coalesce(edl_date_end, current_date) >= current_date and
            edl_status >= 0
      into edl_id, edl_status
  do
  begin
    if (edc_object_uniq = 0 and edl_status = 1) then
    begin
      update edoc_list
      set edl_date_end = current_date - 1
      where edl_id = :edl_id;
    end
    else
    begin
      update edoc_list
      set edl_status = -1
      where edl_id = :edl_id;
    end

    if (db_id = 81001 and edc_code in ('loan.agreement', 'ad.agreement', 'pdata.agreement')) then
    begin
      --При аннулировании вышеуказанных документов необходимо скорректировать реестр
      for select eds_from_c_id
      from edoc_sign
      where eds_from_edl_id = :edl_id
        and eds_from_c_id is not null
      into c_id
      do
      begin
        if (f_edoc_blank_signed(c_id, 'ad.agreement', current_date) = 0) then
          execute procedure hp_clientcomreg_set(:c_id, null, 'add', 1, 0, 0, 'edoc_canceled');
        if (f_edoc_blank_signed(c_id, 'pdata.agreement', current_date) = 0) then
          execute procedure hp_clientcomreg_set(:c_id, null, 'add', 0, 1, 0, 'edoc_canceled');
      end
    end

     -- RFCKG-4990
    if (db_id = 41701
        and obj_kind_str = 'kontract' 
        and edc_code in ('insurance.application', 'insurance.kyrgyz', 'insurance.kyrgyz.ipf')) then
    begin
      update insurance_kontract
      set ik_enum_status = f_enum_id('insurance_kontract', 'edoc_was_cancelled')
      where ik_from_k_id = :obj_id 
        and ik_from_ag_id in (7130, 7414, 7570);
    end
  end

  -- Если аннулируется Договор АСП, аннулируем все заявления на изменение основного номера
  if (obj_kind_str = 'Clients' and edc_code = 'asp.agreement') then
    execute procedure hp_edoc_cancel('cell.application', obj_kind, obj_id);

  -- Если аннулируется договор займа, аннулируем Расчет ПДН
  if (edc_code = 'loan.agreement') then
  begin
    execute procedure hp_edoc_cancel('loan.pdn', obj_kind, obj_id);

    -- RFCKG-4990
    if (db_id = 41701) then
    begin
      execute procedure hp_edoc_cancel('insurance.application', obj_kind, obj_id);
      execute procedure hp_edoc_cancel('insurance.kyrgyz', obj_kind, obj_id);
      execute procedure hp_edoc_cancel('insurance.kyrgyz.ipf', obj_kind, obj_id);
    end
  end

  -- RFCKG-3925
  -- Если аннулируется заявка на получение кредита, аннулируем зявление на страхование
  if (db_id in (39801, 41701)
    and edc_code = 'loan.application'
    and obj_kind_str = 'miniapp'
  ) then
  begin
    select coalesce(k.k_id, 0)
    from kontract k
    where k.k_from_ma_app_id = :obj_id
        and k.k_id > 0
    into k_id;

    if (k_id > 0) then
    begin
      execute procedure hp_edoc_cancel('insurance.application', f_objkind('kontract', 'id'), :k_id);
      execute procedure hp_edoc_cancel('insurance.kyrgyz', f_objkind('kontract', 'id'), :k_id);
      execute procedure hp_edoc_cancel('insurance.kyrgyz.ipf', f_objkind('kontract', 'id'), :k_id);

      -- RFCKF-5235: Аннулирование инфо-листа при аннулировании заявки
      if (db_id = 41701) then
        execute procedure hp_edoc_cancel('loan.application_new_conditions', f_objkind('kontract', 'id'), :k_id);
    end
  end

  -- RFCRU-4632
  if (db_id = 81001 and edoc_kind = 'cell.application.no_access') then
  begin
    select first 1 edl_id
    from edoc_list
    where edl_from_edc_id = :edc_id
      and edl_object_kind = :obj_kind
      and edl_object_id = :obj_id
    order by edl_id desc
    into :edl_id;

    -- документ может находится в двух состояния: был создан и уже подписан (f_enum_id('mc_phone_change', 'created'), f_enum_id('mc_phone_change', 'signed'))
    update mc_phone_change
    set mpc_status = f_enum_id('mc_phone_change', 'cancelled')
    where mpc_from_edl_id = :edl_id
      and mpc_status in (f_enum_id('mc_phone_change', 'created'), f_enum_id('mc_phone_change', 'signed'));
  end
end
