CREATE PROCEDURE AAREP_CLIENTS_PLACES(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_AMOUNT_ALL_DISB DECIMAL(18,4),
    AAREP_AMOUNT_ALL_BALANCE DECIMAL(18,4),
    AAREP_KOLICH_ALL_DISB INTEGER,
    AAREP_KOLICH_ALL_BALANCE INTEGER,
    AAREP_AMOUNT_FEMAIL_DISB DECIMAL(18,4),
    AAREP_AMOUNT_FEMAIL_BALANCE DECIMAL(18,4),
    AAREP_KOLICH_FEMAIL_DISB INTEGER,
    AAREP_KOLICH_FEMAIL_BALANCE INTEGER,
    AAREP_STAT_REP_REGION VARCHAR(255))
AS
declare variable kont_amount decimal(18,4);
declare variable aarep_k_id decimal(18,0);
declare variable femail_amount decimal(18,4);
declare variable all_kol integer;
begin
  /* Procedure Text */
  aarep_amount_all_disb = 0;
  aarep_amount_all_balance = 0;
  aarep_kolich_all_disb = 0;
  aarep_kolich_all_balance = 0;
  aarep_amount_femail_disb = 0;
  aarep_amount_femail_balance = 0;
  aarep_kolich_femail_disb = 0;
  aarep_kolich_femail_balance = 0;

  for select k_kolich_client_approved, aarep_plan_os, aarep_k_id, reg_stat_rep_region
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 2), kontract, region
      where aarep_k_id = k_id and
            reg_unicode = aarep_region_id
      into :aarep_kolich_all_disb, :aarep_amount_all_disb, :aarep_k_id, :aarep_stat_rep_region
  do
  begin
    aarep_amount_femail_disb = 0;
    aarep_kolich_femail_disb = 0;
    select count(kc_from_c_id), sum(kc_amount_approved)
    from knclient, clients
    where kc_from_k_id = :aarep_k_id and
          kc_from_c_id = c_id and
          c_sex = 0 and
          kc_amount_approved <> 0 and
          kc_amount_approved is not null
    into :aarep_kolich_femail_disb, :aarep_amount_femail_disb;
    if (aarep_kolich_femail_disb is null) then
      aarep_kolich_femail_disb = 0;
    if (aarep_amount_femail_disb is null) then
      aarep_amount_femail_disb = 0;
    suspend;
  end

  aarep_amount_all_disb = 0;
  aarep_amount_all_balance = 0;
  aarep_kolich_all_disb = 0;
  aarep_kolich_all_balance = 0;
  aarep_amount_femail_disb = 0;
  aarep_amount_femail_balance = 0;
  aarep_kolich_femail_disb = 0;
  aarep_kolich_femail_balance = 0;
  for select k_kolich_client_approved, k_amount_approved, aarep_k_balance, aarep_k_id, reg_stat_rep_region
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 3), kontract, region
      where aarep_k_id = k_id and
            reg_unicode = aarep_region_id
      into :aarep_kolich_all_balance, :kont_amount, :aarep_amount_all_balance, :aarep_k_id, :aarep_stat_rep_region
  do
  begin
    aarep_kolich_femail_balance = 0;
    aarep_amount_femail_balance = 0;
    select count(kc_from_c_id), sum(kc_amount_approved)
    from knclient, clients
    where kc_from_k_id = :aarep_k_id and
          kc_from_c_id = c_id and
          c_sex = 0 and
          kc_amount_approved <> 0 and
          kc_amount_approved is not null
    into :all_kol, :femail_amount;
    if (all_kol is null) then
      all_kol = 0;
    if (femail_amount is null) then
      femail_amount = 0;
    if (femail_amount <> 0) then
    begin
      aarep_kolich_femail_balance = all_kol;
      aarep_amount_femail_balance = cast(femail_amount / kont_amount as decimal(15,7));
      aarep_amount_femail_balance = cast(aarep_amount_femail_balance * aarep_amount_all_balance as decimal(18,2));
    end
    suspend;
  end

end
