CREATE PROCEDURE AAREP_CLIENTDOCSCHECK_TABLE(
    REPORT_DATE_BEGIN D_DATETIME,
    REPORT_DATE_END D_DATETIME)
RETURNS (
    NUMBER_OF_ORDER D_INTEGER NOT NULL,
    CHECK_ID TYPE OF COLUMN CLIENTQUEUECHECK.CQ_ID,
    DATE_EDIT D_DATE NOT NULL,
    EMPLOYEE_DIGIT_ID TYPE OF COLUMN USERNAME.UN_ID NOT NULL,
    EMPLOYEE_DIGIT_NAME TYPE OF COLUMN USERNAME.UN_FIO NOT NULL,
    DATE_START D_DATETIME NOT NULL,
    DATE_END D_DATETIME,
    TIME_PROCESSING_SEC D_INTEGER NOT NULL,
    TIME_PROCESSING D_DOUBLE NOT NULL,
    CLIENT_ID TYPE OF COLUMN CLIENTS.C_ID NOT NULL,
    CLIENT_NAME TYPE OF COLUMN CLIENTS.C_NAME_FIO NOT NULL,
    DOCUMENT_ID TYPE OF COLUMN CLIENTDOCSCHECK_LOG.DCL_CDS_ID,
    DOCUMENT_NAME D_STR_8100,
    DOCUMENT_NUMBER D_STR_8100,
    DOCUMENT_DATE_OF_ISSUE D_DATE,
    DOCUMENT_DATE_VALIDITY D_DATE,
    EMPLOYEE_CREATE_NAME TYPE OF COLUMN USERNAME.UN_FIO NOT NULL,
    EMPLOYEE_CREATE_OFFICE TYPE OF COLUMN REGION.REG_NAME)
AS
begin
  for
    select
        row_number() over (/*order by cqc.cq_un_id*/)                                                  as number_of_order,
        q.check_id,
        coalesce(cdc.dcl_date_check, cq_date_start_check)                                              as date_edit,
        q.employee_digit_id,
        q.employee_digit_name,
        q.cq_date_start_check                                                                          as date_start,
        q.cq_date_end_check                                                                            as date_end,
        q.time_processing_client_sec/coalesce(check_doc_count, 1)                                      as time_processing_sec,
        (q.time_processing_client_sec/coalesce(check_doc_count, 1)/86400.000000000)                    as time_processing,
        q.client_id,
        q.client_name,
        q.documnet_id,
        (select pk_name from passkind as pk where pk_id = coalesce(cdc.dcl_from_pk_id, q.passkind_id)) as document_name,
        cdc.dcl_doc_no                                                                                 as document_number,
        cdc.dcl_doc_date_begin                                                                         as document_date_of_issue,
        cdc.dcl_doc_date_end                                                                           as document_date_validity,
        q.employee_create_name,
        q.employee_create_office
      from
        (
          select
              cqc.cq_id                                                                as check_id,
              count(cqc.cq_id) over (partition by cqc.cq_id)                           as check_doc_count,
              cqc.cq_c_id                                                              as client_id,
              (select first 1 c.c_name_fio from clients c where c.c_id = cqc.cq_c_id)  as client_name,
              cqc.cq_un_id                                                             as employee_digit_id,
              (select first 1 u.un_fio from username u where (u.un_id = cqc.cq_un_id)) as employee_digit_name,
              cqc.cq_date_start_check,
              cqc.cq_date_end_check,
              u.un_fio                                                                 as employee_create_name,
              r.reg_name                                                               as employee_create_office,
              datediff(
                second,
                cqc.cq_date_start_check,
                coalesce(cqc.cq_date_end_check, current_timestamp))                    as time_processing_client_sec,
              cd.cds_id                                                                as documnet_id,
              cd.cds_from_pk_id                                                        as passkind_id
            from
              clientqueuecheck                         as cqc
              inner join clientdocs                    as cd
                on (cqc.cq_c_id = cd.cds_from_c_id)
              inner join username                      as u
                on (u.un_id = cd.cds_from_un_id)
              inner join region                        as r
                on (r.reg_unicode = u.un_curr_reg_id)
            where
                 (cqc.cq_date_start_check between :report_date_begin and :report_date_end)
              or (coalesce(cqc.cq_date_end_check, current_timestamp) between :report_date_begin and :report_date_end)
        )                             as q
        left join clientdocscheck_log as cdc
          on (     (cdc.dcl_c_id = q.client_id)
               and (cdc.dcl_cds_id = q.documnet_id))
      where
        (coalesce(cdc.dcl_date_check, :report_date_begin) between :report_date_begin and :report_date_end)
      into
        :number_of_order,
        :check_id,
        :date_edit,
        :employee_digit_id,
        :employee_digit_name,
        :date_start,
        :date_end,
        :time_processing_sec,
        :time_processing,
        :client_id,
        :client_name,
        :document_id,
        :document_name,
        :document_number,
        :document_date_of_issue,
        :document_date_validity,
        :employee_create_name,
        :employee_create_office
    do
      suspend;
end
