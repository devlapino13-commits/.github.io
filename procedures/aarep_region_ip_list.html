CREATE PROCEDURE AAREP_REGION_IP_LIST
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(100),
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    IP_ADDRESSES VARCHAR(1024),
    REG_MEMO BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    REG_IS_CLOSED INTEGER,
    REG_CODE_NAME VARCHAR(100))
AS
begin
  reg_memo = '';
  for select r.reg_unicode, r.reg_code, r.reg_name, b.reg_name, iif(r.reg_date_close is null, 0, 1)
      from region b, region r
      where r.parent = b.reg_unicode and
            r.reg_status = 3
      into :reg_id, reg_code, reg_name, branch_name, reg_is_closed
  do
  begin
    reg_code_name = reg_code || ' ' || reg_name;
    ip_addresses = '';
    select list(rad_ip_address, ',')
    from regipaddr
    where rad_from_reg_id = :reg_id
    into ip_addresses;
    if (ip_addresses is null) then
      ip_addresses = '';
    suspend;
  end
end
