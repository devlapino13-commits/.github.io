CREATE PROCEDURE AAREP_EARLY_CLOSED_KONTRACTS(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_PARENT D_STR_100,
    OFFICE D_STR_100,
    C_FIO D_STR_100,
    APPROVE_EMP_FIO D_STR_100,
    CLOSED_EMP_FIO D_STR_100,
    KONTRACT_NO D_STR_20,
    CLOSE_DATE D_DATE,
    AMOUNT_APPROVED D_INTEGER,
    APP_EMP_FIO D_STR_100,
    DATE_APP D_DATE)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable ma_id d_id;
declare variable kommis_amount d_currency;
declare variable kredline_end d_date;
begin
  for select reg.reg_stat_rep_region, -- регион
             reg.reg_name, -- офис
             cl.c_name_fio, -- ФИО клиента
             cl3.c_name_fio, -- ФИО закрывшего сотрудника
             k.k_kontract_no, -- контракт
             ks.ks_date_begin, -- дата закрытия
             kc.kc_amount_approved, -- сумма кредита
             k.k_id, --
             cl.c_id, --
             k.k_from_ma_app_id --
      from kontract k
      inner join kontstate ks on k.k_id = ks.ks_from_k_id and
            ks.ks_state = 2 and
            ks.ks_ol_is_delete = 0
      inner join clients cl3 on ks.ks_kc_id = cl3.c_id
      inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id and
            kc.kc_amount_approved > 0
      inner join clients cl on kc.kc_from_c_id = cl.c_id
      inner join region_list(:rep_region_id, 1, -1, 1) on ks.ks_region_id = id_list
      inner join region reg on ks.ks_region_id = reg.reg_unicode
      where k.k_kredline_end > ks.ks_date_begin and -- условие для досрочно закрытых кредитов
            ks.ks_date_begin between :rep_date_begin and :rep_date_end
      into :reg_parent, --
           :office, --
           :c_fio, --
           :closed_emp_fio, --
           :kontract_no, --
           :close_date, --
           :amount_approved, --
           :k_id, --
           :c_id, --
           :ma_id --
  do
  begin
    -- проверяем, есть ли активный займ на дату отчета
    if (exists(select *
               from kontstate ks
               inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id and
                     kc.kc_amount_approved > 0
               where kc.kc_from_c_id = :c_id and
                     ks.ks_state = 1 and
                     :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
                     ks.ks_ol_is_delete = 0)) then
      continue;

    -- получаем штраф
    kommis_amount = null;
    select ok.ok_perc
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_type = 7 and
          ok.ok_ol_date <= :rep_date_end
    order by ok.ok_ol_date desc rows 1
    into :kommis_amount;

    if ((kommis_amount <> 0) or (kommis_amount is not null)) then
      continue;

    -- КС, утвердивший займ
    approve_emp_fio = '';
    select cl.c_name_fio
    from username un
    inner join kontlog kl on un.un_id = kl.kl_user_id
    inner join clients cl on un.un_from_c_id = cl.c_id
    where kl.kl_state = 1 and
          kl.kl_from_k_id = :k_id
    order by kl.kl_datetime desc rows 1
    into :approve_emp_fio;

    -- КС, принявший заявку
    app_emp_fio = '';
    select cl4.c_name_fio
    from miniapp ma
    inner join username un on un.un_id = ma.ma_from_un_id
    inner join clients cl4 on un.un_from_c_id = cl4.c_id
    where ma.ma_is_active = 1 and
          ma.ma_id = :ma_id and
          ma.ma_date_end = '31.12.9999'
    into :app_emp_fio;

    -- Дата принятия новой заявки (до полного закрытия договора)
    date_app = null;
    select ma.ma_date_app
    from miniapp ma
    inner join miniappclient mac on ma.ma_id = mac.mac_from_ma_id
    where mac.mac_from_c_id = :c_id and
          ma.ma_id <> :ma_id and
          ma.ma_is_active = 0 and
          ma.ma_date_end = '31.12.9999' and
          ma.ma_date_app <= :close_date and
          mac.mac_is_poruch = 0
    order by ma.ma_date_oper desc rows 1
    into :date_app;

    if ((date_app < close_date) or (date_app is null)) then
      continue;
    suspend;
  end
end
