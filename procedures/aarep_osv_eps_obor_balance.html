CREATE PROCEDURE AAREP_OSV_EPS_OBOR_BALANCE(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SHOW_ADD_CODE INTEGER = 0)
RETURNS (
    DT_AB_ID ID_TYPE,
    KT_AB_ID ID_TYPE,
    OB_SUMMA CURR_TYPE,
    OB_SUMMA_VAL CURR_TYPE,
    OB_SUMMA_TAX CURR_TYPE,
    OB_SUMMA_TAX_DT CURR_TYPE,
    OB_SUMMA_TAX_KT CURR_TYPE,
    DT_AB_CODE D_STR_CODE,
    KT_AB_CODE D_STR_CODE,
    DT_AB_NAME D_STR_255,
    KT_AB_NAME D_STR_255)
AS
declare variable has_filter d_smallint;
begin
  if (rep_show_add_code is null) then
    rep_show_add_code = 0;

  has_filter = 0;
  if (exists(select *
             from temp_ref_account_rep
             rows 1)) then
    has_filter = 1;

  if (has_filter = 0) then
  begin
    for select obb_ab_dt, obb_ab_kt, sum(obb_summa), sum(obb_summa_val), sum(obb_summa_tax_dt), sum(obb_summa_tax_kt)
        from kn_operbalbook
        where obb_ol_date between :rep_date_begin and :rep_date_end and
              obb_is_extended = :rep_show_add_code
        group by 1, 2
        into :dt_ab_id, :kt_ab_id, :ob_summa, :ob_summa_val, :ob_summa_tax_dt, ob_summa_tax_kt
    do
    begin
      dt_ab_code = '';
      dt_ab_name = '';
      kt_ab_code = '';
      kt_ab_name = '';

      select iif(:rep_show_add_code = 0, ab_id, ab_parent_id), ab_code, ab_name
      from ref_accbalance
      where ab_id = :dt_ab_id
      into :dt_ab_id, dt_ab_code, dt_ab_name;

      select iif(:rep_show_add_code = 0, ab_id, ab_parent_id), ab_code, ab_name
      from ref_accbalance
      where ab_id = :kt_ab_id
      into :kt_ab_id, kt_ab_code, kt_ab_name;

      ob_summa_tax = ob_summa_tax_dt;
      suspend;
    end
  end
  else
  begin
    for select a_dt.acc_from_ab_id, a_kt.acc_from_ab_id,
               trim(iif(:rep_show_add_code = 0, a_dt.acc_ab_code, a_dt.acc_add_code)),
               trim(iif(:rep_show_add_code = 0, a_kt.acc_ab_code, a_kt.acc_add_code)), sum(ob_summa), sum(ob_summa_val),
               sum(ob_summa_tax_dt), sum(ob_summa_tax_kt)
        from aarep_osv_eps_obor_tmp(:rep_date_begin, :rep_date_end, null, null, 0), ref_account a_dt, ref_account a_kt
        where dt_acc_id = a_dt.acc_id and
              kt_acc_id = a_kt.acc_id
        group by 1, 2, 3, 4
        into :dt_ab_id, :kt_ab_id, :dt_ab_code, :kt_ab_code, :ob_summa, :ob_summa_val, :ob_summa_tax_dt,
             ob_summa_tax_kt
    do
    begin
      dt_ab_name = '';
      kt_ab_name = '';

      select ab_name
      from ref_accbalance
      where ab_id = :dt_ab_id
      into dt_ab_name;

      select ab_name
      from ref_accbalance
      where ab_id = :kt_ab_id
      into kt_ab_name;

      ob_summa_tax = ob_summa_tax_dt;
      suspend;
    end
  end
end
