CREATE PROCEDURE HP_PDN_ZEROPERC_ACTION(
    ACTION_KIND D_STR_CODE,
    DISB_PERIOD D_INTEGER,
    GROUP_ID D_INTEGER,
    ID_LIST D_STR_LONG)
RETURNS (
    PZP_ID D_ID)
AS
declare variable res_id d_integer;
declare variable tmp_id d_integer;
declare variable curr_period d_integer;
declare variable new_group_id d_integer;
declare variable pcg_name d_str_100;
begin
  -- Действия над таблицей PDN_ZEROPERC
  pzp_id = 0;

  if (coalesce(disb_period, 0) = 0) then
    exception e_message 'Период не указан';

  if (coalesce(group_id, 0) = 0) then
    exception e_message 'Группа не указана';

  if (coalesce(action_kind, '') = '') then
    exception e_message 'Действие не указано';

  if (action_kind not in ('add', 'delete', 'check', 'clone')) then
    exception e_message 'Неопределенное действие: ' || action_kind;

  id_list = trim(coalesce(id_list,''));

  if (action_kind = 'add') then
  begin
    if (disb_period <= 0 or group_id <= 0) then
      exception e_message f_format('Неверные команды, группа %1, период %2', group_id, disb_period);

    insert into pdn_zeroperc (pzp_disb_period, pzp_from_pcg_id)
    values (:disb_period, :group_id)
    returning pzp_id
    into :pzp_id;
  end

  if (action_kind = 'delete') then
  begin
    if (id_list = '') then
      exception e_message 'Записи для удаления не указаны';

    for select res_str
    from hp_split_str(:id_list, ',')
    into :res_id
    do
    begin
      delete from pdn_zeroperc
      where pzp_id = :res_id;
    end
  end

  if (action_kind = 'clone') then
  begin
    if (disb_period = -1 and group_id = -1 and id_list <> '') then
    begin
      -- Клонирование выделенных строк
      pzp_id = null;
      for select res_str
      from hp_split_str(:id_list, ',')
      into :res_id
      do
      begin
        insert into pdn_zeroperc (pzp_disb_period, pzp_from_pcg_id, pzp_day_min, pzp_day_max, pzp_amount_min,
                                  pzp_amount_max, pzp_zalog_kind, pzp_poruch_kind, pzp_line_id, pzp_percent)
        select pzp_disb_period, pzp_from_pcg_id, pzp_day_min, pzp_day_max, pzp_amount_min, pzp_amount_max,
               pzp_zalog_kind, pzp_poruch_kind, pzp_line_id, pzp_percent
        from pdn_zeroperc
        where pzp_id = :res_id
        returning pzp_id
        into tmp_id;

        if (pzp_id is null) then
          pzp_id = tmp_id;
      end

    end else
    if (id_list <> '') then
    begin
      pzp_id = null;
      select pzp_disb_period
      from pdn_zeroperc
      where pzp_id = :id_list
      into curr_period;

      if (curr_period is null) then
        curr_period = -1;

      new_group_id = group_id;
      if (group_id > 0) then
      begin
        select pcg_name
        from pdn_credgroup
        where pcg_id = :group_id
        into pcg_name;

        if (coalesce(pcg_name, '') = '') then
          exit;

        pcg_name = pcg_name || ' копия';
        insert into pdn_credgroup (pcg_name)
        values (:pcg_name)
        returning pcg_id
        into :new_group_id;
      end

      insert into pdn_zeroperc (pzp_disb_period, pzp_from_pcg_id, pzp_day_min, pzp_day_max, pzp_amount_min,
                                pzp_amount_max, pzp_zalog_kind, pzp_poruch_kind, pzp_line_id, pzp_percent)
      select iif(:disb_period > 0, :disb_period, pzp_disb_period),
             iif(:group_id > 0, :new_group_id, pzp_from_pcg_id),
             pzp_day_min, pzp_day_max, pzp_amount_min, pzp_amount_max,
             pzp_zalog_kind, pzp_poruch_kind, iif(:group_id = -1, pzp_line_id, 0), pzp_percent
      from pdn_zeroperc
      where (:disb_period = -1 or pzp_disb_period = :curr_period) and
            (:group_id = -1 or pzp_from_pcg_id = :group_id);
    end
  end

  select count(*)
  from hp_pdn_zeroperc_check(null)
  into res_id;

  suspend;
end
