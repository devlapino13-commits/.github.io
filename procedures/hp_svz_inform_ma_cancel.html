CREATE PROCEDURE HP_SVZ_INFORM_MA_CANCEL(
    REP_DATE DATE)
AS
declare variable c_id id_type;
declare variable holidays_qty integer;
declare variable difdays integer;
declare variable ma_date_app date;
declare variable ma_id integer;
declare variable kf_name str_100;
declare variable un_id id_type;
declare variable rd_id integer;
begin
  un_id = 1000000000000;

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'LOPenaltyReason.Miniapp_6_Days'
  into :rd_id;

-- если это выходной день, не отправляю уведомление
  if (not exists(select sc.scd_work_hours
                 from sa_calendays sc
                 where sc.scd_date = :rep_date and
                       sc.scd_work_hours > 0)) then
    exit;

-- получаю список работающих свз на дату отчета
  for select hps.ps_from_c_id
      from hr_pstatus hps
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where :rep_date between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            jk.jk_system_code = 'JobSystem.LO_Standart'
      into :c_id
  do
  begin
    for select maa.ma_id, datediff(day, maa.ma_date_oper, :rep_date), kal.kal_new_kf_name, maa.ma_date_oper
        from miniapp maa
        inner join knaktlist kal on kal.kal_from_kf_id = maa.ma_from_kf_id
        where maa.ma_date_oper < :rep_date and
              maa.ma_is_active = 1 and
              maa.ma_date_end = '31.12.9999' and
              :rep_date between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0 and
              kal.kal_new_kf_officer = :c_id and
              maa.ma_date_oper >= '01.06.2017'
        order by 2
        into :ma_id, :difdays, :kf_name, :ma_date_app
    do
    begin
      holidays_qty = 0;

      select count(*)
      from sa_calendays sc
      where sc.scd_date between :ma_date_app and :rep_date and
            sc.scd_work_hours = 0
      into :holidays_qty;

      holidays_qty = coalesce(holidays_qty, 0);

      difdays = difdays - holidays_qty + 1;

      if (not(exists(select *
                     from kontract kont
                     where kont.k_from_ma_app_id = :ma_id and
                           kont.k_status = 1))) then
      begin
        if ((difdays = 3) or (difdays >= 5)) then -- уведомляю СВЗ
        begin
          execute procedure fsrobot_insert_event('DoMailing', 'Push_Svz_Ma_Notification',
              :c_id || ';' || :difdays || ';' || :kf_name || ';6_days;0', :un_id);
        end

        if (difdays >= 6) then
        begin
          insert into lo_penalty (lp_from_c_id, lp_from_ma_id, lp_days, lp_ma_date, lp_amount, lp_reason)
          values (:c_id, :ma_id, :difdays, :ma_date_app, 1000, :rd_id);

          execute procedure hp_miniapp_status_change(:ma_id, :un_id, 0);

          insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id)
          values (10, :un_id, null, 0, :ma_id);

          update kontract kont
          set kont.k_status = -1
          where kont.k_status = 0 and
                kont.k_from_ma_app_id = :ma_id;
        end
      end
    end
  end
end
