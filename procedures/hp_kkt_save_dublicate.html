CREATE PROCEDURE HP_KKT_SAVE_DUBLICATE(
    KKM_ID D_ID NOT NULL,
    KKM_FD D_INTEGER NOT NULL,
    KKM_FPD D_ID NOT NULL,
    KKM_FN D_STR_20 NOT NULL,
    KKM_FISC_DATETIME D_DATETIME NOT NULL,
    DO_ANNUL D_BOOL = 1)
AS
declare variable kkf_id d_id;
  declare variable dubl_kkm_id d_id;
  declare variable corr_dubl_kkm_id d_id;
  declare variable corr_reg_id d_integer;
begin
  if (do_annul is null) then
    do_annul = 1;  --  создать чек аннулирования для дубля. по-умолчанию - создать

  -- вставим дубль чека
  select kkf_id, kkd_from_reg_id
  from kkt_fn, kkt_devices
  where kkf_code = :kkm_fn
    and kkd_id = kkf_from_kkd_id
  into kkf_id, corr_reg_id;

  if (kkf_id is null) then
    exception e_message 'Не найден ФН: ' || coalesce(kkm_fn, '<NULL>');

  insert into kkt_main (kkm_from_ol_id, kkm_from_k_id, kkm_kkt_kind, kkm_is_return, kkm_status, kkm_fd,
                        kkm_fpd, kkm_from_kkf_id, kkm_create_datetime, kkm_send_datetime, kkm_fisc_datetime,
                        kkm_ol_is_delete, kkm_corr_ol_id, kkm_corr_reg_id, kkm_add_number, kkm_duplicated_kkm_id)
  select kkm_from_ol_id, kkm_from_k_id, kkm_kkt_kind, kkm_is_return, 2, :kkm_fd, :kkm_fpd,
         :kkf_id, kkm_create_datetime, kkm_send_datetime, :kkm_fisc_datetime, kkm_ol_is_delete,
         kkm_corr_ol_id, kkm_corr_reg_id, kkm_add_number, kkm_id
  from kkt_main
  where kkm_id = :kkm_id
  returning kkm_id
  into dubl_kkm_id;

  insert into kkt_items (kki_from_kkm_id, kki_position, kki_type, kki_amount)
  select :dubl_kkm_id, kki_position, kki_type, kki_amount
  from kkt_items
  where kki_from_kkm_id = :kkm_id;

  if (do_annul = 0) then
    exit;

  -- создадим аннулирование созданного дубля
  insert into kkt_main (kkm_from_ol_id, kkm_from_k_id, kkm_kkt_kind, kkm_ol_is_delete,
                        kkm_corr_ol_id,
                        kkm_is_return,
                        kkm_corr_reg_id,
                        kkm_duplicated_kkm_id)
  select kkm_from_ol_id, kkm_from_k_id, kkm_kkt_kind, kkm_ol_is_delete,
        kkm_from_ol_id,    -- корректируем оригинальный ol_id
        1 - kkm_is_return, -- чек обратный целевому
        :corr_reg_id,      -- для kkm_corr_reg_id ставим оригинальный регион
        :dubl_kkm_id       -- обозначим что это относится к дубликату
  from kkt_main
  where kkm_id = :dubl_kkm_id
  returning kkm_id
  into corr_dubl_kkm_id;

  insert into kkt_items (kki_from_kkm_id, kki_position, kki_type, kki_amount)
  select :corr_dubl_kkm_id, kki_position, kki_type, kki_amount
  from kkt_items
  where kki_from_kkm_id = :dubl_kkm_id;
end
