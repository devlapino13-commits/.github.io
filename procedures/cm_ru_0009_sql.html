CREATE PROCEDURE CM_RU_0009_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable c_verify_status integer;
declare variable c_cell_no_date date;
declare variable c_birth_date date;
declare variable c_doc_type integer;
declare variable c_from_is_id integer;
declare variable kol_cli_knclient integer;
declare variable kol_cli_kontract integer;
declare variable adl_id id_type;
begin
  kol_cli_knclient = 0;
  if (k_id = 101010090675400201) then
  exit;
--        Проверка верификации для РФ базы
  for
  select kc_from_c_id,
         kc_amount_approved,
         k_disb_date,
         c_verify_status,
         c_name_fio,
         c_pass_no,
--         c_cell_no_date,
         c_birth_date,
         c_doc_type,
         c_from_is_id,
         k_kolich_client_approved
  from knclient, kontract, clients
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_from_k_id = k_id
  and kc_from_c_id = c_id
  into c_id,
       kc_amount,
       kf_disb_date,
       c_verify_status,
       res_info_text,
       kf_client_pass_no,
--       c_cell_no_date,
       c_birth_date,
       c_doc_type,
       c_from_is_id,
       kol_cli_kontract
  do
  begin
    c_cell_no_date = null;
    select c_cell_no_date
    from AAREP_GET_CONTINFO('clients', :c_id)
    into c_cell_no_date;
    kol_cli_knclient = kol_cli_knclient + 1;
        if ((c_verify_status<>2) and (c_verify_status<>-1)) then
        begin 
          if (exists(select *
              from region, kontract
              where k_last_reg_id = reg_unicode
              and k_id = :k_id
              and REG_NEED_CLIENT_VERIFY = 1)) then
          begin
            res_error_is_critical=2;
            res_error_text='Верификация не проведена';
            Suspend;
          end
        end
--        отключен в связи с запуском электронных документов
--        if (coalesce(:c_cell_no_date,'01.01.1990') < :kf_disb_date - 30) then
--        begin 
--          res_error_is_critical=2;
--          res_error_text='Сот. номер не обновлен более месяца.';
--          Suspend;
--        end
        if (:c_birth_date is null) then
        begin 
          res_error_is_critical=2;
          res_error_text='Дата рождения не заполнена';
          Suspend;
        end
        if (coalesce(:kf_client_pass_no,'') = '') then
        begin 
          res_error_is_critical=2;
          res_error_text='Номер паспорта не заполнен';
          Suspend;
        end else
        begin
          if (exists(select * from HP_CHECK_FMS_PASSPORT(:kf_client_pass_no) where CHECK_RESULT=1)) then
          begin
            res_error_is_critical=2;
            res_error_text='По базе УФМС паспорт недействителен!';
            Suspend;
          end
        end
        if ((coalesce(:c_doc_type,0) < 1) and res_error_text='') then
        begin
          res_error_is_critical=2;
          res_error_text='Тип документа не указан';
          Suspend;
        end 
        if (exists(select 1 from addrclient
            where adc_from_c_id = :c_id
            and adc_address_type = 1)) then
        begin
          if (exists(select 1 from addrclient
              where adc_from_c_id = :c_id
              and adc_address_type = 1
              and mod(adc_from_adl_id,1000000000) = 0)) then
              begin
                res_error_is_critical=2;
                res_error_text='Юридический адрес не заполнен';
                Suspend;
              end
        end else
        begin
          res_error_is_critical=2;
          res_error_text='Юридический адрес не существует';
          Suspend;
        end
        if (exists(select 1 from addrclient
            where adc_from_c_id = :c_id
            and adc_address_type = 2)) then
        begin
          if (exists(select 1 from addrclient
              where adc_from_c_id = :c_id
              and adc_address_type = 2
              and mod(adc_from_adl_id,1000000000) = 0)) then
              begin
                res_error_is_critical=2;
                res_error_text='Фактический адрес не заполнен';
                Suspend;
              end
        end else
        begin
          res_error_is_critical=2;
          res_error_text='Фактический адрес не существует';
          Suspend;
        end
      /*  if (c_from_is_id = -1 or c_from_is_id is null) then
        begin
          res_error_is_critical = 2;
          res_error_text = 'В информации клиента поле "Узнали про нас" не выбрано';
          Suspend;
        end */
    for
    select adc_from_adl_id from addrclient
    where adc_from_c_id = :c_id
    and adc_address_type in (1, 2)
    into adl_id
    do
    begin
      if (not exists(select * from addrlist
      where adl_id = :adl_id
      and left(adl_from_pl_id, 3) = left(f_get_db_id(), 3))) then
      begin
        res_error_is_critical=2;
        res_error_text='Неверно указана страна в адресе';
        suspend;
      end
    end
  end
  if (kol_cli_knclient <> kol_cli_kontract) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Прошу вас переоформить контракт, общее кол-во клиентов не совпададает по кол-ву договору!';
    Suspend;
  end 
end
