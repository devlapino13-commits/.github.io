CREATE PROCEDURE HP_UXFRAME_SQL_MODIFY(
    FR_CODE D_STR_CODE,
    DATA_KIND D_INTEGER)
RETURNS (
    SQL_TEXT D_STR_8100,
    EXEC_PROC D_BOOL)
AS
declare variable ufs_field_name d_str_code;
declare variable ufp_name d_str_code;
declare variable field_list d_str_8100;
declare variable enter_sym d_str_10;
declare variable sql_where d_str_long;
declare variable ufd_source d_str_code;
declare variable ufd_is_table d_bool;
declare variable ufp_pk_field d_smallint;
declare variable value_list d_str_8100;
declare variable ufp_code d_str_100;
declare variable ufp_def_value d_str_100;
declare variable f_temp d_str_255;
declare variable ufp_editor d_str_code;
begin
  -- Приготовление Update sql для фрейма

  exec_proc = 0;
  sql_text = '';
  field_list = '';
  value_list = '';
  sql_where = '';
  enter_sym = ascii_char(13) || ascii_char(10);

  f_temp = 'coalesce(f_ux_set_blob(:new_%1$blob, :new_%1$file), :new_%1)';

  select ufd_source, ufd_is_table
  from uxfrdata
  where ufd_ufr_code = :fr_code and
        ufd_kind = :data_kind
  into ufd_source, ufd_is_table;

  if (row_count = 0) then
    exception e_default f_format('Data source not found, fr_code: %1', fr_code);

  if (ufd_is_table = 1) then
  begin
    for select ufs_field_name, ufp_origin_name, coalesce(ufp_pk_field, 0), ufp_editor
    from uxframesql, uxfrparam
    where ufs_ufr_code = :fr_code and
          ufs_sql_kind = :data_kind and
          ufs_ufp_code = ufp_code and
          ufp_in_use = 1
    order by ufs_index
    into ufs_field_name, ufp_name, ufp_pk_field, ufp_editor
    do
    begin
      if (field_list <> '') then
        field_list = field_list || ', ' || enter_sym;

      if (data_kind = 2 and value_list <> '') then
        value_list = value_list || ', ' || enter_sym;
  
      if (data_kind = 1) then
      begin
        if (ufp_editor <> 'blob') then
          field_list = field_list || ufs_field_name || ' = :new_' || ufp_name;
        else
          field_list = field_list || ufs_field_name || ' = ' || f_format(f_temp, ufp_name);
      end

      if (data_kind = 2) then
      begin
        field_list = field_list || ufs_field_name;

        if (ufp_editor <> 'blob') then
          value_list = value_list || ':new_' || ufp_name;
        else
          value_list = value_list || f_format(f_temp, ufp_name);
      end
  
      if (ufp_pk_field <> 0 and data_kind in (1, 3)) then
      begin
        if (sql_where <> '') then
          sql_where = sql_where || ' and ' || enter_sym;
  
        sql_where = sql_where || ufs_field_name || ' = :old_' || ufp_name;
      end
    end

    if (data_kind = 1) then
    begin
      if (field_list <> '') then
      begin
        sql_text = 'update ' || ufd_source || ' set ';
        sql_text = sql_text || enter_sym || field_list;
  
        if (sql_where <> '') then
          sql_text = sql_text || enter_sym || 'where (' || sql_where || ')';
        else
          exception e_message ('Update SQL: Нет условия по ключевым полям (WHERE)');
      end
    end else
    if (data_kind = 2) then
    begin
      if (field_list <> '') then
      begin
        sql_text = 'insert into ' || ufd_source;
        sql_text = sql_text || enter_sym || ' (' || field_list || ')';
        sql_text = sql_text || enter_sym || 'values (' || value_list || ')';
      end
    end else
    if (data_kind = 3) then
    begin
      if (sql_where <> '') then
      begin
        sql_text = 'delete from ' || ufd_source;
        sql_text = sql_text || enter_sym || 'where (' || sql_where || ')';
      end else
        exception e_message ('Delete SQL: Нет условия по ключевым полям (WHERE)');
    end
  end else
  begin
    if (exists(select * from rdb$procedure_parameters
    where rdb$procedure_name = upper(:ufd_source) and
          rdb$parameter_type = 1)) then
      exec_proc = 0;
    else
      exec_proc = 1;

    for select lower(trim(rdb$parameter_name))
    from rdb$procedure_parameters
    where rdb$procedure_name = upper(:ufd_source) and
          rdb$parameter_type = 0
    order by rdb$parameter_number
    into ufs_field_name
    do
    begin
      ufp_name = '';
      ufp_code = '';
      ufp_def_value = '';
      ufp_editor = '';

      select ufs_ufp_code
      from uxframesql
      where ufs_ufr_code = :fr_code and
            ufs_sql_kind = :data_kind and
            ufs_field_name = :ufs_field_name
      into ufp_code;

      if (row_count > 0) then
      begin
        if (field_list <> '') then
          field_list = field_list || ', ' || enter_sym;

        ufp_def_value = 'null';
        if (ufp_code <> '') then
        begin
          select ufp_origin_name, ufp_editor
          from uxfrparam
          where ufp_code = :ufp_code
          into ufp_name, ufp_editor;

          if (ufp_editor <> 'blob') then
            field_list = field_list || ':' || ufp_name;
          else
            field_list = field_list || f_format(f_temp, ufp_name);
        end
        else
          field_list = field_list || ufp_def_value;
      end
      else
        exception e_default f_format('В настройках отсутствует параметр %1 процедуры %2', upper(ufs_field_name), upper(ufd_source));
    end

    if (field_list <> '') then
    begin
      if (exec_proc = 0) then
        sql_text = 'select * from ' || ufd_source || '(' || field_list || ')';
      else
        sql_text = 'execute procedure ' || ufd_source || '(' || field_list || ')';
    end
    else
      exception e_default f_format('В настройках процедуры %1 отсутствуют параметры', upper(ufd_source));
  end
  suspend;
end
