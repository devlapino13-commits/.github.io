CREATE PROCEDURE HP_COMPET_AZ_OLIMPIC_2020_SPF(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer;
declare variable rd_id integer;
declare variable res_period_id integer;
declare variable res_team_id integer;
declare variable res_lo_id decimal(18,0);
declare variable ad_active_balance decimal(18,4) = 0;
declare variable points_1 decimal(18,4) = 0;
declare variable points_2 decimal(18,4) = 0;
declare variable points_all decimal(18,4) = 0;
declare variable res_team_id_prior integer;
declare variable cpr_id integer;
declare variable cpr_group_name varchar(255);
declare variable cpd_comment d_str_4096;
declare variable cpp_orderno integer;
declare variable kind integer;
declare variable cpr_final_points decimal(18,4) = 0;
declare variable ad_disb_loan decimal(18,4) = 0;
declare variable place_no integer;
declare variable points_3 decimal(18,4) = 0;
declare variable points_4 decimal(18,4) = 0;
declare variable is_otpusk varchar(100);
declare variable is_otpusk_month varchar(100);
declare variable kol integer;
declare variable ad_disb_amount decimal(18,4) = 0;
declare variable aarep_fact_planned decimal(18,4) = 0;
declare variable aarep_plan_all decimal(18,4) = 0;
declare variable points_5 decimal(18,4) = 0;
declare variable client_diff decimal(18,4) = 0;
declare variable client_diff_prior decimal(18,4) = 0;
declare variable stage_cpp_id integer;
declare variable ad_disb_new_client integer;
declare variable dd_tmp date;
declare variable otpusk_days smallint;
declare variable n integer;
declare variable ad_delinq_balance_7 curr_type;
declare variable ad_diff_balance curr_type;
declare variable svz_portfel_begin curr_type;
declare variable svz_portfel_end curr_type;
declare variable prinyat_portfel curr_type;
declare variable date_begin date;
declare variable date_end_prv date;
declare variable date_begin_prv date;
declare variable zarabotok_end curr_type;
declare variable zarabotok_begin curr_type;
declare variable zarabotok_diff curr_type;
declare variable is_month_end smallint;
declare variable ad_delinq_balance_diff_7 curr_type;
declare variable avg_delinq_balance_7 curr_type;
declare variable avg_delinq_balance_7_diff curr_type;
declare variable avg_disb_amount curr_type;
declare variable avg_zarabotok_diff curr_type;
declare variable avg_balance_diff curr_type;
declare variable avg_plan_disb curr_type;
declare variable rep_period integer;
declare variable rep_indicator integer;
declare variable plan_sum curr_type;
declare variable disb_sum curr_type;
declare variable dosroch_sum curr_type;
declare variable plan_perc curr_type;
declare variable lo_stage_date date;
declare variable lo_stage integer;
declare variable avg_points curr_type;
declare variable lo_panelty_count curr_type;
declare variable total_points curr_type;
declare variable add_points curr_type;
declare variable kol_day_decade smallint;
declare variable kol_day_worked smallint;
declare variable kol_day_month integer;
declare variable otpusk_kind smallint;
declare variable cpr_comment str_255;
declare variable par7 perc_type;
declare variable group_name str_255;
declare variable max_points curr_type;
declare variable avg_disb_loan curr_type;
begin
  -- Расчет итогов олимпиады РФ 2020
  -- Stage:
  -- 0 - период (неделя)
  -- 1 - Промежуточные итоги
  -- 2 - Итоги
  -- 3 - Финальные результаты

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  kol_day_decade = res_date_end - res_date_begin + 1;

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  if (extract(day from res_date_end) not in (10, 20)) then
    is_month_end = 1;
  else
    is_month_end = 0;

  if (cpp_orderno = 0 and
      res_date_begin = cast('31.01.2020' as date)) then
  begin
    res_date_begin = '01.02.2020';
    res_date_end = '01.02.2020';

    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    -- Формируем в памяти данные по РФ
    --    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0, 'standart')
    --    returning_values :res_period_id;

    -- Формируем в памяти данные по РФ
    -- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
    -- Sanjari, 04.06.2017
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1, sum(ad_active_balance)
        from hp_analitica_report('kont_lo_teamname_id', 'tree_group', 8, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 360, 0, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, :ad_active_balance
    do
    begin
      if (exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1)) then
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, '', -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end
    end
    suspend;
    exit;
  end

  -- Ищем итоги тура
  stage_cpp_id = null;
  select cpp_id
  from competperiod
  where cpp_from_rd_id = :rd_id and
        cpp_stage = 2 and
        cpp_status = 1 and
        cpp_date_end <= :res_date_end
  order by cpp_date_begin desc, cpp_id desc
  rows 1
  into :stage_cpp_id;

  if (stage = 0) then -- Если уровень Декады
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    -- Формируем в памяти данные по РФ
    --    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0, 'standart')
    --    returning_values :res_period_id;

    -- Формируем в памяти данные по РФ
    -- Тип отчета передаем "olimpic", чтобы система учитывала особенности расчета для олимпиады
    -- Sanjari, 04.06.2017
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    date_begin = res_date_end - extract(day from res_date_end) + 1;
    date_end_prv = date_begin - 1;
    date_begin_prv = date_end_prv - extract(day from date_end_prv) + 1;
    /*
    --  формируем данные для план выдачи
    if (is_month_end = 1) then
    begin
      rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);
      select rd_id
      from refdata
      where rd_code in ('planindicators.disb_amount')
      into rep_indicator;

      execute procedure hp_plandata_fill(1000000, :res_date_end, :rep_indicator, :rep_period, 1, null);
    end      */

    -- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1, res_field_2,
               --           sum(ad_active_client),
               sum(ad_active_balance), sum(ad_delinq_balance_end2), sum(ad_delinq_balance_diff2), sum(ad_disb_loan), sum(ad_disb_amount)
               --           sum(ad_return_fact_os),
               --           sum(ad_return_plan_os),
               --           sum(ad_disb_new_client),
               --           sum(ad_diff_client)
        from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 8, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 360, 0, 'standart')
        where exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1, 2
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, :res_lo_id,
             --         :ad_active_client,          -- Клиенты на конец периода
             :ad_active_balance, -- Баланс на конец периода
             :ad_delinq_balance_7, -- Баланс просрочек 7 дней
             :ad_delinq_balance_diff_7, -- Изменение баланс 7 дней
             :ad_disb_loan, -- Выданные займы,
             :ad_disb_amount -- Сумма выдачи СВЗ
             --         :aarep_fact_planned,
             --         :aarep_plan_all,
             --         :ad_disb_new_client,
             --         :ad_diff_client
    do
    begin
      cpd_comment = '';
      if (ad_disb_loan is null) then
        ad_disb_loan = 0;
      if (ad_disb_amount is null) then
        ad_disb_amount = 0;
      if (aarep_fact_planned is null) then
        aarep_fact_planned = 0;
      if (aarep_plan_all is null) then
        aarep_plan_all = 0;
      if (ad_disb_new_client is null) then
        ad_disb_new_client = 0;

      -- Исключаем если нет действующего
      if (not(exists(select *
                     from hr_pstatus
                     where ps_from_c_id = :res_lo_id and
                           ps_date_end >= :res_date_end and
                           ps_date_begin <= :res_date_end and
                           ps_status = 1 and
                           ps_is_working = 1))) then
        res_team_id = 0;

      if (res_team_id <> 0) then
      begin
        -- Исключаем если не СВЗ/КС
        if (exists(select *
                   from hr_pstatus, joblist, jobkind
                   where ps_from_c_id = :res_lo_id and
                         ps_date_end >= :res_date_end and
                         ps_date_begin <= :res_date_end and
                         ps_status = 1 and
                         ps_from_jl_id = jl_id and
                         jl_kind = jk_id and
                         not(jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner')) and
                         ps_is_working = 1)) then
          res_team_id = 0;
      end

      is_otpusk = '';
      is_otpusk_month = '';
      if (res_team_id <> 0) then
      begin
        -- Исключаем если в отпуске.
        -- Проверяем статус отпуска на каждый день декады:
        -- если из декады проработал менее 5 дней то будем считать что в отпуске
        otpusk_days = 0;
        n = 0;
        while (n <= 9 and dateadd(day, n, res_date_begin) < res_date_end) do
        begin
          dd_tmp = dateadd(day, n, res_date_begin);
          if (exists(select *
                     from hr_voctravel, hr_orders, sa_worktype
                     where hrv_from_ho_id = ho_id and
                           ho_order_type = sawt_id and
                           hrv_from_c_id = :res_lo_id and
                           sawt_base_type in (4, 5) -- отпуск, декретный отпуск
                           and
                           sawt_id <> 20 and
                           sawt_hours_real >= 0 and
                           hrv_date_begin <= :dd_tmp and
                           hrv_date_end_fact >= :dd_tmp and
                           hrv_status = 1)) then
            otpusk_days = otpusk_days + 1;
          n = n + 1;
        end
        kol_day_worked = kol_day_decade - otpusk_days;

        if (kol_day_worked < 5) then
        begin
          if (not(exists(select *
                         from hr_voctravel, hr_orders
                         where hrv_from_ho_id = ho_id and
                               ho_order_type = 20 and
                               hrv_from_c_id = :res_lo_id and
                               hrv_date_begin <= :res_date_end and
                               hrv_date_end_fact >= :res_date_begin and
                               hrv_status = 1))) then
          begin
            --            res_team_id=0;
            is_otpusk = 'Отпуск декады.';
          end
        end
      end

      points_1 = 0.0;
      points_2 = 0.0;
      points_3 = 0.0;
      points_4 = 0.0;
      points_5 = 0.0;
      --      points_5=0.0; -- Среднее кол-во выдач

      --      par7=0.0;
      --      if (ad_active_balance<>0)
      --      then par7=round(ad_delinq_balance_7*100.00/ad_active_balance,2);

      --    проверяем, если конец месяца т.е. не 10е и не 20е то включаем дополнительные расчеты
      ad_diff_balance = null;

      if (is_month_end = 1) then
      begin
        svz_portfel_begin = null;
        select sum(oks_balance)
        from oksummary
        where :date_begin - 1 between oks_date_begin and oks_date_end
              --        and :date_begin - 1 - oks_delinq_date <= 360
              and
              oks_kc_id = :res_lo_id and
              oks_close_state = 0
        into svz_portfel_begin;
        if (svz_portfel_begin is null) then
          svz_portfel_begin = 0;

        svz_portfel_end = null;
        select sum(oks_balance)
        from oksummary
        where :res_date_end between oks_date_begin and oks_date_end
              --        and :res_date_end - oks_delinq_date <= 360
              and
              oks_kc_id = :res_lo_id and
              oks_close_state = 0
        into svz_portfel_end;
        if (svz_portfel_end is null) then
          svz_portfel_end = 0;

        prinyat_portfel = null;
        select sum(oks_balance)
        from knaktlist, kontract, oksummary
        where knaktlist.is_delete = 0 and
              kal_date between :date_begin and :res_date_end and
              kal_from_ka_id > 0 and
              kal_new_kf_officer <> kal_prior_lo_id and
              kal_new_kf_officer = :res_lo_id and
              kal_from_kf_id = k_from_kf_id and
              k_id = oks_from_k_id and
              :res_date_end between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_kc_id = :res_lo_id and
              oks_close_state = 0 and
              k_fact_disb_lo_id <> oks_kc_id -- если принемает то что сам выдавал то их из портфеля не минусуют
              -- если передача была внутри одной команды то не минусуют, Приказ № 235.2-09-18
              and
              (coalesce((select res_rd_id
                         from hp_objectprops_get(0, kal_new_kf_officer, -1, 'objectprops.staffteam', :res_date_end)), -999) -- вместо :res_date_end было kal_date т.е. нужно проверять на дату конец
              <> coalesce((select res_rd_id
                           from hp_objectprops_get(0, kal_prior_lo_id, -1, 'objectprops.staffteam', :res_date_end)), -777) or
              --   если сотрудник был уволен и была передача то такой портфел будет минусоваться
              not exists(select *
                         from hr_pstatus
                         where ps_from_c_id = kal_prior_lo_id and
                               :res_date_end between ps_date_begin and ps_date_end and
                               ps_status = 1 and
                               ps_is_working = 1))
        into prinyat_portfel;
        if (prinyat_portfel is null) then
          prinyat_portfel = 0;

        ad_diff_balance = svz_portfel_end - svz_portfel_begin - prinyat_portfel;

        select aarep_bonus_total
        from cb_az_lo_201602_one_lo(null, :res_lo_id, /*:date_begin_prv,*/ :date_end_prv, 0)
        into zarabotok_begin;

        select aarep_bonus_total
        from cb_az_lo_201602_one_lo(null, :res_lo_id, /*:date_begin,*/ :res_date_end, 0)
        into zarabotok_end;

        zarabotok_diff = zarabotok_end - zarabotok_begin;

        --      Расчет стажа нужно сделать через эту процедуру HP_GET_LO_STAGE!!!

        select lo_stage_date, lo_stage_month
        from hp_get_lo_stage(:res_lo_id, :res_date_end)
        into lo_stage_date, lo_stage;
        if (lo_stage_date is not null) then
        begin
          cpd_comment = cpd_comment || 'Стаж КС: <b>' || lo_stage || '</b><br>';
        end

        if (lo_stage < 4 and
            zarabotok_diff > 0) then
          zarabotok_diff = zarabotok_diff / 2; --Для КССпФ со стажем менее 4 месяцев, их заработок делится на 2, пункт 3
        /*
        --      реализация план выдач
        select sum(pd_data_1)
        from temp_plandata
        where pd_from_c_id = :res_lo_id and
              pd_is_office = 2 and
              pd_data_1 > 0
        into plan_sum;
        if (plan_sum is null) then
          plan_sum = 0;   */

        plan_sum = 0;
        select po_amount
        from hp_get_planoper_info(1000000, :res_lo_id, 'planindicators.disb_amount', :res_date_end, :rep_period)
        into plan_sum;
        if (plan_sum is null) then
          plan_sum = 0;

        --      обьем выдач
        select sum(k_amount_approved)
        from kontstate, kontract
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :date_begin and :res_date_end and
              ks_kc_id = :res_lo_id and
              ks_from_k_id = k_id
        into disb_sum;
        if (disb_sum is null) then
          disb_sum = 0;

        --      Обьем досроных погашений
        select -sum(ok_os)
        from operkred, kontract, knaktlist
        where ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_os < 0 and
              ok_ol_date between :date_begin and :res_date_end and
              ok_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              ok_ol_date between kal_date and kal_date_end and
              kal_new_kf_officer = :res_lo_id and
              ok_ol_date < ok_sh_date and
              extract(month from ok_ol_date) <> extract(month from ok_sh_date)
        into dosroch_sum;
        if (dosroch_sum is null) then
          dosroch_sum = 0;

        if (plan_sum > 0) then
          plan_perc = (disb_sum - dosroch_sum) / plan_sum * 100;
        else
          plan_perc = 0;

        is_otpusk_month = '';
        if (res_team_id <> 0) then
        begin
          -- Исключаем если в отпуске.
          -- Проверяем статус отпуска на каждый день декады:
          -- если из декады проработал менее 5 дней то будем считать что в отпуске
          otpusk_days = 0;
          n = 0;
          while (dateadd(day, n, date_begin) < res_date_end) do
          begin
            dd_tmp = dateadd(day, n, date_begin);
            if (exists(select *
                       from hr_voctravel, hr_orders, sa_worktype
                       where hrv_from_ho_id = ho_id and
                             ho_order_type = sawt_id and
                             hrv_from_c_id = :res_lo_id and
                             sawt_base_type in (4, 5) -- отпуск, декретный отпуск
                             and
                             sawt_id <> 20 and
                             sawt_hours_real >= 0 and
                             hrv_date_begin <= :dd_tmp and
                             hrv_date_end_fact >= :dd_tmp and
                             hrv_status = 1)) then
              otpusk_days = otpusk_days + 1;
            n = n + 1;
          end
          kol_day_month = extract(day from res_date_end);
          kol_day_worked = kol_day_month - otpusk_days;

          if (kol_day_worked < 15) then
          begin
            if (not(exists(select *
                           from hr_voctravel, hr_orders
                           where hrv_from_ho_id = ho_id and
                                 ho_order_type = 20 and
                                 hrv_from_c_id = :res_lo_id and
                                 hrv_date_begin <= :res_date_end and
                                 hrv_date_end_fact >= :res_date_begin and
                                 hrv_status = 1))) then
            begin
              --            res_team_id=0;
              is_otpusk_month = 'Отпуск месяца.';
            end
          end
        end

      end

      cpd_comment = is_otpusk || '<br>';
      if (is_otpusk_month <> '') then
        cpd_comment = cpd_comment || is_otpusk_month || '<br>';
      --      cpd_comment=cpd_comment||'Актив.кл: <b>'||round(ad_active_client)||'</b><br>';
      --      cpd_comment=cpd_comment||'PAR7: <b>'||par7||'%</b><br>';
      cpd_comment = cpd_comment || 'Активный баланс: <b>' || round(ad_active_balance, 2) || ' р.</b><br>';

        from competperiod, competresult, competdetails
        where cpp_from_rd_id = :rd_id and
              cpp_stage = 0 and
              cpp_date_begin >= :res_date_begin and
              cpp_date_end <= :res_date_end and
              cpp_id = cpr_from_cpp_id and
              cpr_object_id <> 0 and -- Исключаем всех кто вне команды
              cpr_group_name <> '' and
              cpr_id = cpd_from_cpr_id and
              cpd_add_data_9 = 1 and -- выборка только по последним днем месяца
              cpd_add_data_7 > 0 -- у кого имеется активный баланс
        group by cpr_object_id
        having(cpr_object_id) > 1
        order by 2
        rows 3
        into res_team_id, par7
    do
    begin
      update competresult
      set cpr_additional_points = cpr_additional_points + 3,
          cpr_comment = cpr_comment || iif(cpr_comment = '', '', '; ') || 'Доп.бал за наименьший ПАР7:' || round(:par7, 2)
      where cpr_object_id = :res_team_id and
            cpr_from_cpp_id = :cpp_id;
    end

    --  Команде, победившей по итогам отчетного месяца
    add_points = 1;
    select cpr_group_name, max(cpr_final_points)
    from competresult
    where cpr_from_cpp_id = :cpp_id
    group by cpr_group_name
    order by 2
    rows 1
    into :group_name, max_points;

    update competresult
    set cpr_additional_points = cpr_additional_points + :add_points,
        cpr_comment = cpr_comment || iif(cpr_comment = '', '', '; ') || 'Балл за 1 место: ' || round(:add_points, 0)
    where cpr_from_cpp_id = :cpp_id and
          cpr_group_name = :group_name and
          cpr_final_points = :max_points;

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  if (stage = 2) then -- если уровень этапа
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, sum(cpr_final_points) -- было avg(cpr_final_points) изменен от 05.06.2018 Улан
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      --cpr_final_points = round(cast(cpr_final_points as double precision) / 5, 2);
      cpr_group_name = '';
      -- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);

    update competresult
    set cpr_additional_points = 10,
        cpr_comment = 'Доп. бал за 1 место: 10'
    where cpr_from_cpp_id = :cpp_id and
          cpr_order_in_group = 1 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');

    update competresult
    set cpr_additional_points = 6,
        cpr_comment = 'Доп. бал за 2 место: 6'
    where cpr_from_cpp_id = :cpp_id and
          cpr_order_in_group = 2 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');

    update competresult
    set cpr_additional_points = 3,
        cpr_comment = 'Доп. бал за 3 место: 3'
    where cpr_from_cpp_id = :cpp_id and
          cpr_order_in_group = 3 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');

  end

  if (stage = 3) then -- если финальные итоги
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, avg(cpr_final_points)
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      --cpr_final_points = round(cast(cpr_final_points as double precision) / 5, 2);
      cpr_group_name = '';
      -- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end
    execute procedure hp_competition_recalc_order(:cpp_id);
    --    res_text = 'Автоматизация итогов еще не готов';
  end
  suspend;
end
