CREATE PROCEDURE HP_PERSTEMP_VERIFY_INFO(
    PERS_STATUS D_INTEGER)
RETURNS (
    RES_CAPTION D_STR_100,
    RES_MEMO D_STR_255,
    RES_COLOR_STATE D_SMALLINT)
AS
begin
  res_color_state = -1; -- желтый цвет
  res_caption = 'X';
  res_memo = 'Статус не определен';

  /*
   -- Статус вычисляется в функции: F_PERS_STATUS

   -8  Долго на подписании - когда клиенту ушли документы для подписи более 3 дней назад;
   -7  На подписании - когда клиенту ушли документы для подписи;
   -6  Требуется отправка элдок;
   -5  В очереди - когда находится в очереди на верификацию;
   -4  В обработке КК - когда КК уже рассматривает;
   -3  Данные изменены - когда офис что то поменял, а текущий статус верифицирован и новые данные ещё не отправлены на КК;
   -2  На доработке - когда КК отправил на доработку;
   -1  Не подлежит - это те по кому верификация не требуется (юр. лица);
    0  Неполные данные - когда нет скана и/или фото, когда текущий статус НЕ верифицирован (чаще всего для первичных клиентов или отправленных на доработку);
    1  Готов к отправке на КК – когда все заполнено, когда текущий статус НЕ верифицирован (чаще всего для первичных клиентов или отправленных на доработку);
    2  Подтверждён - когда КК подтвердил перс. данные и клиент подписал (при необходимости);
  */


  if (pers_status = -8) then
  begin
    res_color_state = -1; -- желтый цвет
    res_caption = 'Долго на подписании';
    res_memo = 'Эл.документ в ожидании подписания уже более 3 дней';
  end else
  if (pers_status = -7) then
  begin
    res_color_state = -1; -- желтый цвет
    res_caption = 'На подписании';
    res_memo = 'Необходимо подписать отправленный эл.документ';
  end else
  if (pers_status = -6) then
  begin
    res_color_state = -1; -- желтый цвет
    res_caption = 'Требуется эл.документ';
    res_memo = 'Данные подтверждены, требуется формирование эл.документа на подписание';
  end else
  if (pers_status = -5) then
  begin
    res_color_state = 0; -- обычный цвет
    res_caption = 'В очереди КК';
    res_memo = 'Данные отправлены на верификацию, в очереди КК';
  end else
  if (pers_status = -4) then
  begin
    res_color_state = 0; -- обычный цвет
    res_caption = 'В обработке КК';
    res_memo = 'Данные проверяются сотрудниками КК';
  end else
  if (pers_status = -3) then
  begin
    res_color_state = 0; -- обычный цвет
    res_caption = 'Данные изменены';
    res_memo = 'Персональные данные изменены: паспортные, ИНН или адреса';
  end else
  if (pers_status = -2) then
  begin
    res_color_state = -1; -- желтый цвет
    res_caption = 'На доработке';
    res_memo = 'Верификатор отправил на доработку';
  end else
  if (pers_status = -1) then
  begin
    res_color_state = 0; -- обычный цвет
    res_caption = 'Не подлежит';
    res_memo = 'Верификация не требуется (юр.лица)';
  end else
  if (pers_status = 0) then
  begin
    res_color_state = -1; -- желтый цвет
    res_caption = 'Неполные данные';
    res_memo = 'Необходимо перепроверить данные и отправить на КК';
  end else
  if (pers_status = 1) then
  begin
    res_color_state = 0; -- обычный цвет
    res_caption = 'В ожидании';
    res_memo = 'В ожидании проверки и решения КК';
  end else
  if (pers_status = 2) then
  begin
    res_color_state = 1; -- Зеленый цвет
    res_caption = 'Подтверждён';
    res_memo = 'Персональные данные успешно подтверждены';
  end

  suspend;
end
