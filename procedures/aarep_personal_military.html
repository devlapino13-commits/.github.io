CREATE PROCEDURE AAREP_PERSONAL_MILITARY(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    C_NAME_FIO STR_100,
    C_BIRTH_DATE DATE,
    JOB_NAME STR_100,
    PS_CURR_STATUS VARCHAR(100),
    CDS_DOC_NO VARCHAR(30),
    CDS_DOC_ISSUED_BY VARCHAR(255),
    CDS_DOC_DATE_BEGIN DATE,
    CDS_RANK VARCHAR(255),
    CDS_FIT VARCHAR(255),
    CDS_REGIST VARCHAR(255),
    MEMO_STATUS VARCHAR(255),
    CDS_DOC_TYPE STR_255)
AS
declare variable c_id id_type;
begin
--  берутся все активные сотрудники
  for select c_id, cl.c_name_fio, jl.jl_name as job_name, r.reg_name, c_birth_date
      from region_list(:rep_region_id, 1, -1, 1)

      inner join region r on id_list = reg_unicode

      inner join hr_pstatus on (id_list = ps_from_reg_id and
            ps_status = 1 and
            ps_date_begin < ps_date_end and
            :rep_date between ps_date_begin and ps_date_end)

      inner join clients cl on (ps_from_c_id = c_id and
            c_sex = 1)

      inner join hr_orders ho on (ps_from_ho_id = ho_id and
            ho.ho_status = 1)

      inner join sa_worktype swt on (ho_order_type = sawt_id and
            swt.sawt_id <> 6)

      inner join joblist jl on ps_from_jl_id = jl_id

      order by reg_name, job_name, c_name_fio
      into c_id, c_name_fio, job_name, reg_name, c_birth_date
  do
  begin
    ps_curr_status = '';

    select sawt_name
    from hr_voctravel hrv
    inner join hr_orders ho on hrv_from_ho_id = ho_id
    inner join sa_worktype sw on ho_order_type = sawt_id
    where hrv_from_c_id = :c_id and
          hrv_status = 1 and
          :rep_date between hrv_date_begin and hrv_date_end_fact
    order by hrv_date_begin desc
    rows 1
    into :ps_curr_status;

    cds_doc_no = null;
    cds_doc_issued_by = null;
    cds_doc_date_begin = null;
    cds_rank = null;
    cds_fit = null;
    cds_regist = null;
    memo_status = 'Данные не заполнены';

    for select cd.cds_doc_no, cd.cds_doc_issued_by, cd.cds_doc_date_begin, cd.cds_memo_1, cd.cds_memo_2, cd.cds_memo_3,
               pask.pk_name
        from clientdocs cd
        inner join passkind pask on cd.cds_from_pk_id = pask.pk_id
        inner join passdoctype pdt on pask.pk_from_pdt_id = pdt.pdt_id and
              pdt.pdt_code in ('military_id', 'ref_alternative_duty')
        where cd.cds_from_c_id = :c_id and
              cd.cds_is_delete = 0 and
              cast(cd.cds_datetime as date) <= :rep_date

        into cds_doc_no, cds_doc_issued_by, cds_doc_date_begin, cds_rank, cds_fit, cds_regist, cds_doc_type
    do
    begin
      if (memo_status = 'Данные не заполнены') then
        memo_status = iif((iif(cds_doc_no is null, 0, 1) + iif(cds_doc_issued_by is null, 0, 1) + iif(cds_doc_date_begin is null, 0, 1) + iif(cds_rank is null, 0, 1) + iif(cds_fit is null, 0, 1) + iif(cds_regist is null, 0, 1)) = 6, 'Заполнен', 'Не все данные заполнены');

      else
        memo_status = 'Введены несколько документов.';
    end

    suspend;
  end
end
