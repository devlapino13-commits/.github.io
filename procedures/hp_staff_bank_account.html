CREATE PROCEDURE HP_STAFF_BANK_ACCOUNT(
    C_ID DECIMAL(18,0),
    ACC_OLD VARCHAR(50),
    ACC_NEW VARCHAR(50),
    SACK_ID INTEGER,
    ACC_IS_MAIN INTEGER)
AS
declare variable bank_code varchar(10);
declare variable bc_number integer;
begin
  acc_old = coalesce(acc_old, '');
  acc_new = coalesce(acc_new, '');
  if (sack_id is not null) then
  begin
    select sack_bank_code
    from sa_cardkind
    where sack_id = :sack_id
    into :bank_code;
  end
  if (acc_old <> '') then
  begin
    if (exists(select *
               from bankclient
               where bc_from_c_id = :c_id and
                     bc_account_no = :acc_old)) then
    begin
      update bankclient
      set bc_account_no = :acc_new,
          bc_is_main = :acc_is_main,
          bc_from_b_id = :bank_code
      where (bc_from_c_id = :c_id) and
            (bc_account_no = :acc_old);
    end
    else
    begin
      select max(bc_number) + 1
      from bankclient
      where bc_from_c_id = :c_id
      into :bc_number;
      if ((bc_number is null) or (bc_number <= 0)) then
        bc_number = 1;

      insert into bankclient (bc_from_c_id, bc_number, bc_account_no, bc_name, bc_is_main, bc_is_active, bc_from_b_id)
      values (:c_id, :bc_number, :acc_new, 'current account', :acc_is_main, 1, :bank_code);
    end
  end
  else
  begin
    if (acc_new <> '') then
    begin
      select max(bc_number) + 1
      from bankclient
      where bc_from_c_id = :c_id
      into :bc_number;
      if ((bc_number is null) or (bc_number <= 0)) then
        bc_number = 1;

      insert into bankclient (bc_from_c_id, bc_number, bc_account_no, bc_name, bc_is_main, bc_is_active, bc_from_b_id)
      values (:c_id, :bc_number, :acc_new, 'currenct account', :acc_is_main, 1, :bank_code);
    end
  end
end
