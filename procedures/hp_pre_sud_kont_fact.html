CREATE PROCEDURE HP_PRE_SUD_KONT_FACT(
    REP_DATE D_DATE,
    REP_DELINQ_DAYS D_INTEGER)
RETURNS (
    SBL_NAME STR_255,
    K_ID ID_TYPE,
    K_KONTRACT_NO STR_20,
    DISB_DATE D_DATE,
    DELINQ_DAYS D_INTEGER,
    ZAD_ALL CURR_TYPE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_FINE CURR_TYPE,
    ZAD_EXTRA CURR_TYPE,
    CL_FIO_ALL D_STR_LONG,
    HAS_SUD_REQ D_BOOL,
    ZAD_BALANCE D_CURRENCY)
AS
declare variable delinq_date d_date;
declare variable oks_date_begin d_date;
declare variable oks_delinq_date d_date;
begin
  zad_os = 0;
  zad_perc = 0;
  zad_fine = 0;
  zad_extra = 0;
  delinq_date = dateadd(day, -rep_delinq_days, rep_date);

  for select k_id, k_kontract_no, k_disb_date, :rep_date - oks_delinq_date,
             oks_balance, oks_delinq_os, oks_delinq_perc
      from oksummary
      inner join kontract on k_id = oks_from_k_id
      where oks_date_begin <= :rep_date and
            oks_date_end >= :rep_date and
            oks_delinq_date <= :delinq_date and
            oks_close_state = 0
      into k_id, k_kontract_no, disb_date, delinq_days,
           zad_balance, zad_os, zad_perc
  do
  begin
    sbl_name = '';

    select sbl_name
    from sb_akt, sb_list
    where sba_from_k_id = :k_id and
          :rep_date between sba_date_begin and sba_date_end and
          sba_is_delete = 0 and
          sba_status = 2 and
          sba_from_sbl_id = sbl_id
    into sbl_name;

    zad_all = zad_os + zad_perc + zad_fine + zad_extra;

    select list(c_name_fio, ', ')
    from knclient
    inner join clients on c_id = kc_from_c_id
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into cl_fio_all;

    has_sud_req = 0;

    if (exists(select *
               from sud_mainreq
               where smr_from_k_id = :k_id)) then
      has_sud_req = 1;
    suspend;
  end

  for select oks_date_begin, k_id, k_kontract_no, k_disb_date,
             oks_balance,
             oks_out_os,
             oks_out_perc,
             oks_extra
      from oksummary
      inner join kontract on k_id = oks_from_k_id
      where oks_date_begin <= :rep_date and
            oks_date_end >= :rep_date and
            oks_close_state = 1 and
            (oks_out_os <> 0 or oks_out_perc <> 0 or oks_extra <> 0)
      into oks_date_begin, k_id, k_kontract_no, disb_date,
           zad_balance, zad_os, zad_perc, zad_extra
  do
  begin
    sbl_name = '';

    oks_delinq_date = null;
    delinq_days = 0;

    select oks_delinq_date
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin < :oks_date_begin and
          oks_close_state = 0 and
          oks_delinq_date < '31.12.9999'
    order by oks_from_k_id desc, oks_date_begin desc
    rows 1
    into oks_delinq_date;

    if (oks_delinq_date is null) then
      oks_delinq_date = oks_date_begin;

    delinq_days = datediff(day, oks_delinq_date, rep_date);

    select sbl_name
    from sb_akt, sb_list
    where sba_from_k_id = :k_id and
          :rep_date between sba_date_begin and sba_date_end and
          sba_is_delete = 0 and
          sba_status = 2 and
          sba_from_sbl_id = sbl_id
    into sbl_name;

    zad_all = zad_os + zad_perc + zad_fine + zad_extra;

    select list(c_name_fio, ', ')
    from knclient
    inner join clients on c_id = kc_from_c_id
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into cl_fio_all;

    has_sud_req = 0;

    if (exists(select *
               from sud_mainreq
               where smr_from_k_id = :k_id)) then
      has_sud_req = 1;
    suspend;
  end
end
