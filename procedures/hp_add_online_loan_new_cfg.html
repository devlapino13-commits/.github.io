CREATE PROCEDURE HP_ADD_ONLINE_LOAN_NEW_CFG(
    MIN_POINT D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    MEMO D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    RES_STATUS D_SMALLINT)
AS
begin
  res_text = '';
  res_status = 0;

  if (min_point < 0) then
    res_text = 'Балл не может быть меньше 0';

  if (date_begin < current_date) then
    res_text = 'Дата начала действия не может быть меньше текущей даты (' || f_datetostr(current_date) || ')';

  if (date_end < current_date - 1) then
    res_text = 'Дата окончания действия не может быть указана меньше ' || f_datetostr(current_date - 1);

  if (date_end is null) then
    date_end = '31.12.9999';

  if (trim(memo) = '') then
    res_text = 'Заполните поле "Комментарий"';

  if (res_text = '') then
  begin
    res_text = 'Данные успешно сохранены';
    res_status = 1;
    insert into online_loan_cfg (olc_date_begin, olc_date_end, olc_min_point, olc_from_un_id, olc_datetime,
                                 olc_user_date_end, olc_description)
    values (:date_begin, :date_end, :min_point, :un_id, current_timestamp, :date_end, :memo);
  end

  suspend;
end
