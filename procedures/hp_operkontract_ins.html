CREATE PROCEDURE HP_OPERKONTRACT_INS(
    REP_OK_ID INTEGER,
    OL_ID ID_TYPE,
    KONTRACT_NO STR_100,
    OPERATION_TYPE INTEGER,
    AMOUNT CURR_TYPE)
RETURNS (
    ERROR_TEXT STR_255,
    OK_ID INTEGER)
AS
declare variable k_id id_type;
declare variable k_status integer;
declare variable disb_ol_id id_type;
declare variable k_prepayment_summa decimal(15,2);
declare variable operation_type_code str_100;
declare variable ipf_commission decimal(15,2);
declare variable payed_val decimal(15,2);
declare variable calc_amount decimal(15,2);
begin
  if (ol_id = -1) then
    ol_id = null;
  error_text = '';
  operation_type = coalesce(operation_type, 0);

  if (operation_type = 0) then
  begin
    error_text = 'Ошибка! Тип операции не указан';
    suspend;
    exit;
  end

  operation_type_code = (select rd.rd_code
                         from refdata rd
                         where rd.rd_id = :operation_type);

  if (ol_id is null) then
  begin
    kontract_no = trim(coalesce(kontract_no, ''));
    if (kontract_no = '') then
    begin
      error_text = 'Ошибка! Номер договора не может быть пустым';
      suspend;
      exit;
    end

    if (amount <= 0) then
    begin
      error_text = 'Ошибка! Не корректная сумма платежа';
      suspend;
      exit;
    end

    select kont.k_id, kont.k_status, kont.k_fact_disb_ol_id, coalesce(kont.k_prepayment_summa, 0),
           coalesce(pr.pr_ipf_commission, 0)
    from kontract kont
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kont.k_kontract_no = :kontract_no
    order by kont.k_disb_date desc
    rows 1
    into :k_id, :k_status, :disb_ol_id, :k_prepayment_summa, :ipf_commission;

    calc_amount = (case operation_type_code
                     when 'CashTransactionIncom.IPFCommission' then ipf_commission
                     when 'CashTransactionIncom.PartialPrice' then k_prepayment_summa
                   end);
    if (k_id is null) then
    begin
      error_text = 'Ошибка! Договор не найден';
      suspend;
      exit;
    end

    if (k_status <> 1) then
    begin
      error_text = 'Ошибка! Договор должен быть утвержден';
      suspend;
      exit;
    end

    if (disb_ol_id is not null) then
    begin
      if (exists(select *
                 from operlist ol
                 where ol.ol_id = :disb_ol_id and
                       ol.is_delete = 0)) then
      begin
        error_text = 'Договор уже выдан. Операция не может быть осуществлена';
        suspend;
        exit;
      end
    end

    if (operation_type_code = 'CashTransactionIncom.IPFCommission') then
    begin
      if ((ipf_commission = 0) and
          (amount <> 0)) then
      begin
        error_text = 'Ошибка! Для данного продукта не указан комиссия за выдачу ИПФ';
        suspend;
        exit;
      end

      if ((amount > ipf_commission) and
          (ipf_commission <> 0)) then
      begin
        error_text = 'Ошибка! Для данного продукта комиссия за выдачу ИПФ составляет: ' || ipf_commission || ' сом, а в кассу вносится: ' || cast(amount as decimal(15,2)) || ' сом';
        suspend;
        exit;
      end
    end

    if (operation_type_code = 'CashTransactionIncom.PartialPrice') then
    begin
      if ((k_prepayment_summa = 0) and
          (amount <> 0)) then
      begin
        error_text = 'Ошибка! Для данного кредита не указана сумма частичной оплаты стоймости товара';
        suspend;
        exit;
      end
      if ((amount > k_prepayment_summa) and
          (k_prepayment_summa <> 0)) then
      begin
        error_text = 'Ошибка! Для данного кредита cумма частичной оплаты стоймости товара составляет: ' || ipf_commission || ' сом, а в кассу вносится: ' || cast(amount as decimal(15,2)) || ' сом';
        suspend;
        exit;
      end
    end

    payed_val = 0;
    select sum(opk.ok_amount)
    from operkontract opk
    inner join operlist ol on ol.ol_id = opk.ok_from_ol_id and
          ol.is_delete = 0
    where opk.ok_from_k_id = :k_id and
          opk.ok_operation_type = :operation_type
    into :payed_val;
    payed_val = coalesce(payed_val, 0);

    if ((payed_val = calc_amount) and
        (calc_amount > 0)) then
    begin
      error_text = 'Ошибка! Сумма оплаты, ранее уже была внесена в кассу в полном объеме';
      suspend;
      exit;
    end
    else
    if (amount > (calc_amount - payed_val)) then
    begin
      error_text = 'Ошибка! Сумма оплаты, частично уже была внесена, оставшаяся часть: ' ||(calc_amount - payed_val) || '. Вы пытаетесь внести: ' || cast(amount as decimal(15,2));
      suspend;
      exit;
    end

    insert into operkontract (ok_from_k_id, ok_amount, ok_operation_type)
    values (:k_id, :amount, :operation_type)
    returning ok_id
    into :ok_id;
  end
  else
  begin
    ok_id = rep_ok_id;

    update operkontract
    set ok_from_ol_id = :ol_id
    where (ok_id = :rep_ok_id);
  end
  suspend;
end
