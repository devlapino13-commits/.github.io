CREATE PROCEDURE FM_CHECK_SHOW_REP(
    C_ID ID_TYPE,
    UN_ID ID_TYPE)
RETURNS (
    SHOW_REPORT D_INTEGER)
AS
declare variable kc_id d_integer;
declare variable reg_id d_integer;
declare variable is_active d_integer;
declare variable svz_reg_id d_integer;
declare variable svz_id id_type;
declare variable k_id id_type;
declare variable kf_id id_type;
begin
  show_report = 0;
  select ps_from_reg_id
  from username
  inner join hr_pstatus on ps_from_c_id = un_from_c_id
  where un_id = :un_id and
        current_date between ps_date_begin and ps_date_end and
        ps_is_working = 1 and
        ps_status = 1
  into svz_reg_id;

  is_active = 0;
  for select kc_from_k_id
      from knclient
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0
      order by kc_id
      into k_id
  do
  begin
    if (exists(select *
               from kontstate
               where ks_from_k_id = :k_id and
                     ks_ol_is_delete = 0 and
                     ks_state = 1 and
                     ks_date_end > current_date)) then
      is_active = 1;

    if (coalesce((select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
                  from operkroff
                  where oko_from_k_id = :k_id and
                        oko_ol_is_active = 1), 0) > 0) then
      is_active = 1;

    if (coalesce((select coalesce(sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff), 0)
                  from operextra
                  inner join kontstate on ks_from_k_id = ox_object_id
                  where ox_ol_is_active = 1 and
                        ox_object_kind = 9 and
                        ox_object_id = :k_id and
                        ox_from_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense') and
                        ox_object_kind = f_objkind('kontract', 'id') and
                        ox_ol_date <= current_date and
                        current_date between ks_date_begin and ks_date_end and
                        ks_ol_is_delete = 0 and
                        ks_state = 2), 0) > 0) then
      is_active = 1;

  end
  select k_from_kf_id
  from kontstate, knclient, kontract
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = ks_from_k_id and
        ks_state = 2 and
        ks_ol_is_delete = 0 and
        ks_date_end >= '31.12.2999' and
        k_id = ks_from_k_id
  order by ks_date_begin desc rows 1
  into kf_id;

  select kal_new_kf_officer, kal_new_reg_id
  from knaktlist
  where kal_from_kf_id = :kf_id and
        is_delete = 0 and
        kal_date_end > current_date
  into svz_id, reg_id;

  if (is_active = 0 and
      reg_id = svz_reg_id and
      svz_id = 1000000000004) then
    show_report = 1;
  suspend;
end
