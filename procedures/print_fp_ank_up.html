CREATE PROCEDURE PRINT_FP_ANK_UP(
    C_ID DECIMAL(18,0))
RETURNS (
    FPA_REF_RESULT VARCHAR(1000),
    FPA_REFRESH_D DATE,
    FPA_REF_OFFICER STR_255)
AS
declare variable kal_new_reg_id id_type;
declare variable ca_id id_type;
declare variable check_description varchar(2000);
declare variable fpa_id integer;
declare variable ks_from_k_id id_type;
declare variable ks_region_id id_type;
declare variable ks_lo_id id_type;
declare variable ks_from_k_id_1 id_type;
declare variable ks_region_id_1 id_type;
declare variable ks_lo_id_1 id_type;
declare variable ks_from_k_id_2 id_type;
declare variable ks_region_id_2 id_type;
declare variable ks_lo_id_2 id_type;
declare variable ks_date_begin_1 d_date;
declare variable ks_date_begin_2 d_date;
begin
  fpa_refresh_d = null;

  for select fpa.fpa_date, fpa.fpa_changes, fpa.fpa_id
      from fp_anketa_cl fpa
      where fpa.fpa_c_id = :c_id and
            fpa.fpa_id > 1
      order by fpa.fpa_date
      into :fpa_refresh_d, :fpa_ref_result, :fpa_id
  do
  begin
    ca_id = null;

    ---------- >
    --Беру последний кредит на указанную дату

    ks_from_k_id_1 = null;
    ks_region_id_1 = null;
    ks_lo_id_1 = null;
    ks_date_begin_1 = null;

    ks_from_k_id_2 = null;
    ks_region_id_2 = null;
    ks_lo_id_2 = null;
    ks_date_begin_2 = null;

    select ks_from_k_id, ks_region_id, un.un_from_c_id, ks_date_begin
    from kontstate ks, operlist ol, knclient kc, username un
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin <= :fpa_refresh_d and
          ks_from_ol_id = ol_id and
          ol.is_delete = 0 and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = :c_id and
          ol.ol_user_id = un.un_id
    order by ks_date_begin desc
    rows 1
    into ks_from_k_id_1, ks_region_id_1, ks_lo_id_1, ks_date_begin_1;

    select ks_from_k_id, ks_region_id, un.un_from_c_id, ks_date_begin
    from kontstate ks, operlist ol, username un, zalog zal
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin <= :fpa_refresh_d and
          ks_from_ol_id = ol_id and
          ol.is_delete = 0 and
          ol.ol_user_id = un.un_id and
          zal.z_from_k_id = ks.ks_from_k_id and
          zal.z_from_c_id = :c_id
    order by ks_date_begin desc
    rows 1
    into ks_from_k_id_2, ks_region_id_2, ks_lo_id_2, ks_date_begin_2;

    if (coalesce(ks_date_begin_1, '01.01.1900') > coalesce(ks_date_begin_2, '01.01.1900')) then
    begin
        ks_from_k_id = ks_from_k_id_1;
        ks_region_id = ks_region_id_1;
        ks_lo_id = ks_lo_id_1;
    end
    else
    begin
        ks_from_k_id = ks_from_k_id_2;
        ks_region_id = ks_region_id_2;
        ks_lo_id = ks_lo_id_2;
    end
    ------- <

    --Проверяю последений офис клиента
    select kal.kal_new_reg_id
    from kontract, knaktlist kal
    where k_id = :ks_from_k_id and
          k_from_kf_id = kal_from_kf_id and
          kal.is_delete = 0 and
          :fpa_refresh_d between kal_date and kal_date_end
    into kal_new_reg_id;

    --Если папка не передана в другой офис, проверяю должность сотрудника сохранившего операцию, если кассир беру ИД
    ca_id = null;

    select ps_from_c_id
    from username un, hr_pstatus, joblist jl, jobkind jk
    where un.un_from_c_id = :ks_lo_id and
          un_from_c_id = ps_from_c_id and
          ps_from_reg_id = :ks_region_id and
          ps_is_working = 1 and
          ps_status = 1 and
          :fpa_refresh_d between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id and
          jk.jk_id = jl.jl_kind and
          jk.jk_system_code = 'JobSystem.Specialist_KOBD'
          rows 1
    into :ca_id;

    if (ca_id is null) then
    begin
      --Если не сработало условие беру кассира на дату с этого офиса
      select ps_from_c_id
      from hr_pstatus, joblist, jobkind
      where ps_is_working = 1 and
            ps_status = 1 and
            :fpa_refresh_d between ps_date_begin and ps_date_end and
            ps_from_reg_id = :ks_region_id and
            ps_from_jl_id = jl_id and
            jk_id = jl_kind and
            jk_system_code = 'JobSystem.Specialist_KOBD'
            rows 1
      into :ca_id;

      --Если не сработало условие то офис без кассира либо офис закрыт, тогда беру кассира офиса где числится папка на дату запроса
      if (ca_id is null) then
      begin
        select ps_from_c_id
        from hr_pstatus ps, joblist jl, refdata rd
        where ps_is_working = 1 and
              ps_status = 1 and
              current_date between ps_date_begin and ps_date_end and
              ps.ps_from_reg_id = :kal_new_reg_id and
              ps_from_jl_id = jl_id and
              jl_kind = rd.rd_orderno and --Кассиры
              rd_code containing 'Specialist_KOBD'
              rows 1
        into :ca_id;
      end
    end

    if (ca_id is null) then
      ca_id = ks_lo_id;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :ca_id
    into :fpa_ref_officer;

    check_description = null;

    select fpd.fpd_description
    from fp_anketa_descript fpd
    where fpd.fpd_anketa_no = :fpa_id and
          fpd.fpd_from_c_id = :c_id
    into :check_description;

    check_description = coalesce(check_description, '');

    if (check_description <> '') then
      fpa_ref_result = check_description;

    suspend;
  end
end
