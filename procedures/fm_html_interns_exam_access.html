CREATE PROCEDURE FM_HTML_INTERNS_EXAM_ACCESS(
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable c_name_fio d_str_100;
declare variable reg_name d_str_100;
declare variable i_id d_id;
declare variable link d_str_long;
declare variable obj_kind d_integer;
declare variable has_access d_smallint;
begin
  if (un_id = 1090100004277) then
    un_id = 1090100000162;
  res_text = '';
  date_begin = '01.' || extract(month from current_date) || '.' || extract(year from current_date);
  date_end = dateadd(month, 1, date_begin) - 1;
  obj_kind = f_objkind('interns', 'id');
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  res_text = BLOB_APPEND(res_text, '<h2>Список стажёров:</h2>');

  res_text = BLOB_APPEND(res_text, '<table width="100%"><col width="25%"><col width="50%"><col width="25%">');
  res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Офис</b>', '<b>Стажёр</b>', '<b>Доступ</b>', '', -1, -1, -1, -1, 3, ''));
  for select t.name_fio, t.reg_name, t.i_id
      from hp_interns_view(:un_id, 'reg_manager', 1000000, :date_begin, :date_end, 1, 0, 1) t
      order by t.name_fio, t.reg_name
      into c_name_fio, reg_name, i_id
  do
  begin
    link = f_uri_sql('Разрешить аттестацию', 'give_exam_access', 'interns', :i_id, 'none', 'Подтвердите выдачу доступа к аттестации', '', null, null, null, null, 'dialog', null);
    has_access = 0;
    if (exists(select *
               from exam_access ea
               where ea.ea_object_kind = :obj_kind and
                     ea.ea_object_id = :i_id and
                     current_date between ea.ea_date_begin and ea.ea_date_end)) then
      has_access = 1;

    res_text = BLOB_APPEND(res_text, f_html_table_row(:reg_name, :c_name_fio, iif(has_access = 1, trim('Доступ есть'), trim(:link)), '', -1, -1, -1, -1, 3, ''));
  end
  res_text = BLOB_APPEND(res_text, '</table></body>');

  suspend;
end
