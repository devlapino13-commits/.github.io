CREATE PROCEDURE HP_CORRECT_OPERSALMAIN_OKLAD
RETURNS (
    C_ID DECIMAL(18,0),
    OL_YEAR INTEGER,
    OL_MONTH INTEGER,
    OL_ID DECIMAL(18,0),
    REG_ID INTEGER,
    OSM_ID INTEGER)
AS
declare variable kol integer;
declare variable os_value decimal(18,4) = 0;
declare variable osm_id_first integer;
begin
  for select os_from_c_id, ol_year, ol_month, os_value
      from opersalary, operlist
      where os_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            os_from_sac_id = 2 -- для КР
            and
            os_value <> 0
      group by os_from_c_id, ol_year, ol_month, os_value
      having count(*) > 1
      into :c_id, :ol_year, :ol_month, :os_value
  do
  begin
    kol = 0;
    osm_id_first = null;
    for select os_from_ol_id, os_from_reg_id
        from opersalary, operlist
        where os_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              os_from_sac_id = 2 -- для КР
              and
              os_value = :os_value and
              ol_year = :ol_year and
              ol_month = :ol_month and
              os_from_c_id = :c_id
        into :ol_id, :reg_id
    do
    begin
      osm_id = null;
      select osm_id
      from opersalmain
      where osm_from_c_id = :c_id and
            osm_from_ol_id = :ol_id
      into :osm_id;

      if (osm_id is not null) then
      begin
        update opersaldetail
        set osd_value = 0
        where osd_from_osm_id = :osm_id and
              osd_from_sac_id = 2 -- для КР
              and
              osd_value <> 0;
        kol = kol + 1;
        if (osm_id_first is null) then
          osm_id_first = :osm_id;
      end
    end
    update opersaldetail
    set osd_value = :os_value
    where osd_from_osm_id = :osm_id_first and
          osd_from_sac_id = 2; -- для КР

    suspend;
  end
end
