CREATE PROCEDURE HP_EXTINFO_POL_CANCEL_PROCESS(
    EIL_ID D_ID)
AS
declare variable EIL_OBJECT_KIND D_INTEGER;
declare variable EIL_OBJECT_ID D_ID;
begin
    select eil.eil_object_kind, eil.eil_object_id
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        and eic.eic_code = 'policy_cancel'
    where eil.eil_id = :eil_id
    into :eil_object_kind, :eil_object_id;

    if (:eil_object_kind = f_objkind('kontract', 'id')
        and f_get_const('kont_insurance_active') = 1
        and exists
        (
            select 1
            from extinfolist eil
            inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
                and eic.eic_code = 'policy_cancel'
            where eil.eil_id = :eil_id
                and eil.eil_status = 2
        )
    ) then
    begin
        update insurance_kontract ik
        set ik.ik_enum_status = f_enum_id('insurance_kontract', 'insurance_policy_spoiled')
        where ik.ik_from_k_id = :eil_object_id;
    end
end
