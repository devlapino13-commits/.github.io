Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_PETDATA_VIEW(
    PET_ID D_INTEGER)
RETURNS (
    PED_FROM_PET_ID D_INTEGER,
    PED_ID D_INTEGER,
    PED_DATE D_DATE,
    PED_REGION_PLACE_ID D_INTEGER,
    PED_FROM_C_ID D_ID,
    PED_EXPENSE_SUMMER D_CURRENCY,
    PED_EXPENSE_WINTER D_CURRENCY,
    PED_EXPENSE_SUMMER_MOUNT D_CURRENCY,
    PED_EXPENSE_WINTER_MOUNT D_CURRENCY,
    PED_ACC_BALANCE D_STR_20,
    PED_ACC_NUMBER D_INTEGER,
    PED_IS_ACTIVE D_INTEGER,
    C_NAME_FIO D_STR_100,
    REG_NAME D_STR_CODE)
AS
begin
  for
    select
      ped_from_pet_id,
      ped_id,
      ped_date,
      ped_region_place_id,
      ped_from_c_id,
      ped_expense_summer,
      ped_expense_winter,
      ped_expense_summer_mount,
      ped_expense_winter_mount,
      ped_acc_balance,
      ped_acc_number,
      ped_is_active,
      c_name_fio,
      reg_name
    from petdata
      inner join clients
        on c_id = ped_from_c_id
      inner join region
        on reg_unicode = ped_region_place_id
    where ped_from_pet_id = :pet_id
    order by ped_date desc, ped_is_active desc
  into
    :ped_from_pet_id,
    :ped_id,
    :ped_date,
    :ped_region_place_id,
    :ped_from_c_id,
    :ped_expense_summer,
    :ped_expense_winter,
    :ped_expense_summer_mount,
    :ped_expense_winter_mount,
    :ped_acc_balance,
    :ped_acc_number,
    :ped_is_active,
    :c_name_fio,
    :reg_name
  do
  begin
    suspend;
  end
end
