CREATE PROCEDURE AAREP_OBORSAL_CALC_PERC(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    OB_END_DT DECIMAL(18,4),
    OB_END_KT DECIMAL(18,4),
    OB_OBOROT_DT DECIMAL(18,4),
    OB_OBOROT_KT DECIMAL(18,4),
    AAREP_OFFICE_ID INTEGER,
    AAREP_ACCOUNT VARCHAR(20),
    AAREP_ACCOUNT_TYPE INTEGER)
AS
declare variable ob_saldo_end decimal(18,4);
begin
  for select distinct aarep_office_id, aarep_account, aarep_account_type
      from aarep_calc_perc_acclist(:report_region_id, :report_date_begin, :report_date_end)
      into :aarep_office_id, :aarep_account, :aarep_account_type
  do
  begin
    ob_end_dt = 0;
    ob_end_kt = 0;
    ob_oborot_dt = 0;
    ob_oborot_kt = 0;

    select sum(obs_dt_val), sum(obs_kt_val)
    from obsummary
    where obs_from_al_id = :aarep_account and
          obs_date between :report_date_begin and :report_date_end
    into :ob_oborot_dt, :ob_oborot_kt;
    if (ob_oborot_dt is null) then
      ob_oborot_dt = 0;
    if (ob_oborot_kt is null) then
      ob_oborot_kt = 0;

    ob_saldo_end = f_al_bal(aarep_account, report_date_end, 'val');

    if (ob_saldo_end >= 0) then
      ob_end_dt = ob_saldo_end;
    else
      ob_end_kt = -ob_saldo_end;

    suspend;
  end
end
