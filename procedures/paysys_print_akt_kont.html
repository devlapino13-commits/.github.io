CREATE PROCEDURE PAYSYS_PRINT_AKT_KONT(
    PSM_ID ID_TYPE)
RETURNS (
    REG_NAME VARCHAR(50),
    C_NAME_FIO_SRC VARCHAR(100),
    C_NAME_FIO_DST VARCHAR(100),
    JL_NAME_SRC VARCHAR(100),
    JL_NAME_DST VARCHAR(100),
    PSM_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    DST_IS_KONT SMALLINT,
    AKT_CAPTION VARCHAR(50),
    PSM_AKT_ID DECIMAL(18,0),
    PSI_LIST VARCHAR(255))
AS
declare variable c_id_src decimal(18,0);
declare variable c_id_dst decimal(18,0);
declare variable psm_oper_kind smallint;
declare variable kc_status smallint;
declare variable k_id decimal(18,0);
declare variable psi_code varchar(30);
begin
  select reg_name, psm_date, psm_from_c_source, psm_from_c_destin, psm_oper_kind
  from paysys_main, region
  where psm_id = :psm_id and
        psm_from_reg_id = reg_unicode
  into reg_name, psm_date, c_id_src, c_id_dst, psm_oper_kind;

-- Заголовок шапки акта
  if (psm_oper_kind = 0) then
    akt_caption = 'Приема-передачи карт';
  else
  if (psm_oper_kind = 1) then
    akt_caption = 'Аннулирования карты';

-- ФИО выдающего и получающего
  c_name_fio_src = '';
  if (c_id_src is not null) then
    select c_name_fio
    from clients
    where c_id = :c_id_src
    into c_name_fio_src;

  c_name_fio_dst = '';
  if (c_id_dst is not null) then
    select c_name_fio
    from clients
    where c_id = :c_id_dst
    into c_name_fio_dst;

-- Должность выдающего/получающего сотрудника
  jl_name_src = '';
  select jl_name
  from hr_pstatus, joblist
  where ps_from_c_id = :c_id_src and
        ps_status = 1 and
        :psm_date between ps_date_begin and ps_date_end and
        ps_from_jl_id = jl_id
  into jl_name_src;

  select jl_name
  from hr_pstatus, joblist
  where ps_from_c_id = :c_id_dst and
        ps_status = 1 and
        :psm_date between ps_date_begin and ps_date_end and
        ps_from_jl_id = jl_id
  into jl_name_dst;

-- Определение получателя
  k_id = null;
  k_kontract_no = '';
  select k_id, k_kontract_no
  from paysys_list, kontract
  where psl_from_psm_id = :psm_id and
        psl_object_id = k_id and
        psl_object_kind = 3 and
        psl_sign = 1 rows 1
  into k_id, k_kontract_no;

  if (k_id is not null) then -- Получатель это контракт
  begin
    if (psm_oper_kind = 0) then
      akt_caption = 'Выдачи карты';
-- Получение статуса участника
    select kc_status
    from knclient
    where kc_from_k_id = :k_id and
          kc_from_c_id = :c_id_dst and
          kc_amount_approved > 0 rows 1
    into kc_status;

    jl_name_dst = 'Участник группы';
    if (kc_status = 1) then
      jl_name_dst = 'Лидер группы';
    else
    if (kc_status = 2) then
      jl_name_dst = 'Казначей группы';
    dst_is_kont = 1;
  end
  else
  begin
    select k_id, k_kontract_no
    from paysys_list, kontract
    where psl_from_psm_id = :psm_id and
          psl_object_id = k_id and
          psl_object_kind = 3 and
          psl_sign = -1 rows 1
    into k_id, k_kontract_no;
    if (k_id is not null) then -- Отправитель это контракт
    begin
      if (psm_oper_kind = 0) then
        akt_caption = 'Возврата карты';
-- Получение статуса участника
      select kc_status
      from knclient
      where kc_from_k_id = :k_id and
            kc_from_c_id = :c_id_src and
            kc_amount_approved > 0 rows 1
      into kc_status;

      jl_name_src = 'Участник группы';
      if (kc_status = 1) then
        jl_name_src = 'Лидер группы';
      else
      if (kc_status = 2) then
        jl_name_src = 'Казначей группы';
      dst_is_kont = 0;
    end
  end
  psm_akt_id = psm_id;
  psi_list = '';

  for select distinct psi_code
      from paysys_list, paysys_item
      where psl_from_psm_id = :psm_id and
            psl_from_psi_id = psi_id and
            psl_sign = -1
      into :psi_code
  do
  begin
    if (char_length(psi_code) = 16) then
      psi_code = left(psi_code, 4) || '-' || substring(psi_code from 5 for 4) || '-' || substring(psi_code from 10 for 4) || '-' || right(psi_code, 4);

    if (psi_list = '') then
      psi_list = psi_code;
    else
      psi_list = psi_list || ', ' || psi_code;
  end
  suspend;
end
