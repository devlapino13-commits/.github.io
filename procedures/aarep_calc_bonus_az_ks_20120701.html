CREATE PROCEDURE AAREP_CALC_BONUS_AZ_KS_20120701(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_KC_ID DECIMAL(18,0),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_CURR_DELIQ_BALANCE DECIMAL(18,4),
    AA_LO_JOBNAME VARCHAR(100),
    PS_IS_WORKING_STR VARCHAR(100),
    PAR DECIMAL(18,4),
    AA_KOLICH_CL_DISB INTEGER,
    BONUS_FOR_DISB DECIMAL(18,4),
    BONUS_PORT DECIMAL(18,4),
    BONUS_DRAFT DECIMAL(18,4),
    BONUS_KOEF DECIMAL(18,4),
    BONUS_TOTAL DECIMAL(18,4))
AS
declare variable ps_is_working integer;
begin
  for select aa_curr_kc_id, aa_curr_kc_fio, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name,
             sum(aa_curr_amount_active_end), sum(aa_curr_deliq_balance)

      from aarep_cb_az_ks_20120701_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, null, null)
      group by aa_curr_kc_id, aa_curr_kc_fio, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name
      into :aa_curr_kc_id, :aa_curr_kc_fio, :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name,

           :aa_curr_amount_active_end, :aa_curr_deliq_balance
  do
  begin
    par = 0;
    if (:aa_curr_amount_active_end <> 0) then
      par = cast(:aa_curr_deliq_balance / :aa_curr_amount_active_end * 100 as decimal(15,1));
    /*fact2=0;*/
    if (par <= 1) then
      bonus_koef = 100;
    else
    if (par <= 2) then
      bonus_koef = 75;
    else
    if (par <= 3) then
      bonus_koef = 50;
    else
      bonus_koef = 0;

    /*Вычисляет кол-во выдач для повторных клиентов*/
    select sum(k_kolich_client_approved)
    from knaktlist, kontract, kontstate
    where :rep_date_end between kal_date and kal_date_end and
          knaktlist.is_delete = 0 and
          kal_from_kf_id = k_from_kf_id and
          ks_from_k_id = k_id and
          ks_ol_is_delete = 0 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_state = 1 and
          kal_new_kf_officer = :aa_curr_kc_id
    into :aa_kolich_cl_disb;
    if (:aa_kolich_cl_disb is null) then
      aa_kolich_cl_disb = 0;

    bonus_for_disb = :aa_kolich_cl_disb * 9;
    if (aa_curr_amount_active_end >= 50000) then
      bonus_port = aa_curr_amount_active_end * 0.001;/* 0.1% от портфеля */
    else
      bonus_port = 0;
    bonus_port = round(bonus_port, 0);
    bonus_draft = bonus_for_disb + bonus_port;
    bonus_total = bonus_draft * bonus_koef / 100;

    select jl_name, ps_is_working
    from hr_pstatus, joblist
    where ps_from_c_id = :aa_curr_kc_id and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_from_jl_id = jl_id rows 1
    into :aa_lo_jobname, ps_is_working;
    if (ps_is_working = 0) then
      ps_is_working_str = 'Уволен';
    else
      ps_is_working_str = 'Работает';

    suspend;
  end

end
