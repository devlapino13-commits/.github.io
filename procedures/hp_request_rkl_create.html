CREATE PROCEDURE HP_REQUEST_RKL_CREATE(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RKL_ID D_ID)
AS
declare variable c_pass_no d_str_code;
declare variable c_pass_date d_date;
declare variable rkl_request d_str_8100;
declare variable wallet_id d_integer;
begin
  wallet_id = 100500;
  for
    select
      trim(c_pass_no),
      c_pass_date
    from clients
    where c_id = :c_id
  into
    :c_pass_no,
    :c_pass_date
  do
  begin
    rkl_id = gen_id(gen_request_rkl_id, 1);
    rkl_request = f_format('{"partner_request_id": "%1","wallet_id": %2,"doc_number": "%3","doc_issue_date": "%4"}',
      :rkl_id, :wallet_id, :c_pass_no, cast(:c_pass_date as varchar(10)));

    insert into request_rkl(
      rkl_id,
      rkl_un_id,
      rkl_c_id,
      rkl_numdoc,
      rkl_dtdoc,
      rkl_request)
    values(
      :rkl_id,
      :un_id,
      :c_id,
      :c_pass_no,
      :c_pass_date,
      :rkl_request
    );
    suspend;
  end
end
