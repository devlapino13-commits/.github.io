CREATE PROCEDURE HP_TRANSFERTMC_NUMBDOC(
    OFFICE_ID D_INTEGER,
    DT_OPER D_DATE)
RETURNS (
    NUMB_DOC D_INTEGER)
AS
declare variable nyear d_integer;
begin
  nyear = extract(year from :dt_oper);
  select
    coalesce(max(trt_number), 0) + 1
  from transfertmc
  where trt_offices_id = :office_id
    and extract(year from trt_date) = :nyear
  into
    :numb_doc;
  suspend;
end
