CREATE PROCEDURE AAREP_EDOC_BLANK_VERIFY(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    EDF_PRINT_NAME D_STR_100,
    EDL_OBJECT_NAME D_STR_255,
    EDBV_BEGIN_DATE D_DATE,
    EDBV_BEGIN_TIME D_TIME,
    EDBV_END_DATE D_DATE,
    EDBV_END_TIME D_TIME,
    EDC_NAME D_STR_100,
    MINUTE_VERIFY D_INTEGER,
    STATUS_STR D_STR_CODE,
    RD_NAME D_STR_255,
    UN_FIO_CHECK D_STR_100,
    EDL_CREATE_DATETIME D_DATETIME,
    BLS_DATETIME D_DATETIME,
    EDL_STATUS_STR D_STR_20,
    EDBV_END_DATETIME D_DATETIME,
    UN_FIO_CREATE D_STR_100,
    REG_NAME D_STR_100,
    EDS_STATUS_DATETIME D_DATETIME,
    EDS_STATUS_DATETIME_VERIFY D_DATETIME,
    REG_ID D_INTEGER)
AS
declare variable edb_id d_id;
declare variable un_id d_id;
declare variable rd_id d_integer;
declare variable edbv_edbs_status d_smallint;
declare variable edl_blank_date d_date;
declare variable edb_from_bls_id d_id;
declare variable edbv_begin_datetime d_datetime;
declare variable edbv_create_un_id d_id;
declare variable edl_id d_id;
declare variable tabl_name d_str_code;
declare variable id_name d_str_code;
declare variable edl_object_kind d_integer;
declare variable edl_object_id d_id;
begin

  for
  select edbv_from_edb_id, edbv_begin_datetime, edbv_end_datetime, edbv_from_un_id,
         edbv_from_rd_id, edbv_edbs_status, edbv_create_un_id
  from edoc_blank_verify
  where edbv_create_date between :rep_date_begin and :rep_date_end
--  and edbv_from_edb_id = 4804178
  into edb_id, edbv_begin_datetime, edbv_end_datetime, un_id,
       rd_id, edbv_edbs_status, edbv_create_un_id
  do
  begin

    edl_blank_date = null;
    edf_print_name = null;
    edc_name = null;
    edl_object_kind = null;
    edl_object_id = null;
    edb_from_bls_id = null;
    edl_create_datetime = null;
    edl_status_str = null;
    reg_id = null;

    select edl_blank_date, edf_print_name, edc_name, edl_object_kind, edl_object_id, edb_from_bls_id, edl_create_datetime,
           case edl_status
             when 0 then 'не подписан'
             when 1 then 'подписан'
             when -1 then 'аннулирован'
            end,
            edl_from_reg_id, edl_id
    from edoc_blank, edoc_list, edoc_cfg, edoc_frx
    where edb_id = :edb_id and
          edb_from_edl_id = edl_id and
          edl_from_edc_id = edc_id and
          edb_from_edf_id = edf_id and
          (edb_from_bls_id is not null or edb_blob is not null)
    into edl_blank_date, edf_print_name, edc_name, edl_object_kind, edl_object_id, edb_from_bls_id, edl_create_datetime, edl_status_str, reg_id, edl_id;

    if (edb_from_bls_id is null) then continue;

    edb_from_bls_id = null;
    select edbs_from_bls_id from edoc_blank_scan
    where  edbs_from_edb_id = :edb_id
    into edb_from_bls_id;

    select doc_name
    from hp_edoc_object_info(:edl_object_kind, :edl_object_id, :un_id, :edl_blank_date)
    into edl_object_name;

    select obk_name, obk_pk_field from objkind
    where obk_id = :edl_object_kind
    into tabl_name, id_name;

    edl_object_name = null;
    if (tabl_name = 'kontract') then
    begin
      select c_name_fio from knclient, clients
      where kc_from_k_id = :edl_object_id
      and kc_amount_approved > 0
      and kc_from_c_id = c_id
      into edl_object_name;
    end else
    if (tabl_name = 'miniapp') then
    begin
      select c_name_fio from miniappclient, clients
      where mac_from_ma_id = :edl_object_id
      and mac_amount > 0
      and mac_from_c_id = c_id
      into edl_object_name;
    end else
    if (tabl_name = 'com_list') then
    begin
      select c_name_fio from com_list, knclient, clients
      where cml_id = :edl_object_id
      and cml_object_kind = 9
      and cml_object_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = c_id
      into edl_object_name;
    end else
    if (tabl_name = 'ft_anketa_list') then
    begin
      select c_name_fio from ft_anketa_list, clients
      where fal_id = :edl_object_id
      and fal_from_c_id = c_id
      into edl_object_name;
    end

    if (edl_object_name is null) then
    begin
      select doc_name
      from hp_edoc_object_info(:edl_object_kind, :edl_object_id, :un_id, :edl_blank_date)
      into edl_object_name;
    end 

    edbv_begin_date = edbv_begin_datetime;
    edbv_begin_time = edbv_begin_datetime;
    edbv_end_date = edbv_end_datetime;
    edbv_end_time = edbv_end_datetime;

    minute_verify = datediff(minute from edbv_begin_datetime to edbv_end_datetime);

    status_str = case edbv_edbs_status
                   when 0 then 'Ожидает закрепление скана (в очереди)'
                   when 1 then 'Отправлен на проверку'
                   when 2 then 'Одобрен'
                   when -1 then 'Доработка'
                   else ''
                 end;

    rd_name = null;
    select rd_name from refdata
    where rd_id = :rd_id
    into rd_name;

    un_fio_check = null;
    if (edbv_edbs_status <> 1) then
    begin
      select un_fio from username
      where un_id = :un_id
      into un_fio_check;
    end

    bls_datetime = null;
    select bls_datetime from blobstore
    where bls_id = :edb_from_bls_id
    into bls_datetime;

    select un_fio from username
    where un_id = :edbv_create_un_id
    into un_fio_create;

    select reg_code||' '||reg_name from region
    where reg_unicode = :reg_id
    into reg_name;


    eds_status_datetime = null;
    select min(eds_status_datetime)
    from edoc_sign
    where eds_from_edl_id = :edl_id
    and eds_from_c_id is not null -- клиент подписал
    into eds_status_datetime;

    eds_status_datetime_verify = null;
    select max(eds_status_datetime)
    from edoc_sign
    where eds_from_edl_id = :edl_id
    and eds_from_c_id is null -- верификатор подписал
    into eds_status_datetime_verify;


    suspend;
  end 
end
