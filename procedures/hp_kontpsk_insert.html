CREATE PROCEDURE HP_KONTPSK_INSERT(
    K_ID D_ID,
    OL_ID D_ID)
RETURNS (
    KPS_ID D_ID,
    KPS_PSK D_PERCENT)
AS
declare variable OL_DATE D_DATE;
declare variable LAST_PSK D_PERCENT;
begin
  select effect_rate
  from hp_effect_rate_common(:k_id, :ol_id, 1)
  into :kps_psk;

  select ol_date
  from operlist
  where ol_id = :ol_id
  into ol_date;

  select kps_psk
  from kontpsk, operlist ol
  where kps_from_k_id = :k_id and
        kps_from_ol_id = ol.ol_id and
        ol.is_delete = 0 and
        ol.ol_date <= :ol_date
  order by ol.ol_date desc, ol.ol_id desc
  rows 1
  into last_psk;

  if (row_count = 0) then
  begin
    select k_percent_effect_fifd
    from kontract
    where k_id = :k_id
    into last_psk;
  end

  -- RFCRU-4448, проблема у контракта с пск
  if ((k_id = 101010118183000201) and (f_get_db_id() = 81001)) then
    last_psk = kps_psk;

  if (last_psk is null or last_psk <> kps_psk) then
  begin
    insert into kontpsk (kps_from_k_id, kps_from_ol_id, kps_psk)
    values (:k_id, :ol_id, :kps_psk)
    returning kps_id
    into :kps_id;
  end

  suspend;
end
