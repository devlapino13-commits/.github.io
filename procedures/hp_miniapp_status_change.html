CREATE PROCEDURE HP_MINIAPP_STATUS_CHANGE(
    MA_ID INTEGER,
    MA_FROM_UN_ID DECIMAL(18,0),
    MA_IS_ACTIVE INTEGER)
AS
declare variable ma_number integer;
declare variable last_date_app date;
declare variable last_un_id decimal(18,0);
declare variable db_id integer;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable c_pass_no d_str_20;
declare variable c_full_address d_str_255;
declare variable c_birth_date d_date;
declare variable c_from_reg_id d_id;
declare variable kr_text d_str_255;
declare variable kr_id d_id;
declare variable ma_date_oper d_date;
declare variable res_text d_str_100;
begin
  db_id = f_get_db_id();

  select ma_number, ma_date_app, ma_from_un_id, ma_date_oper
  from miniapp
  where ma_id = :ma_id and
        ma_date_end >= '31.12.2999'
  order by ma_id desc, ma_number desc
  into :ma_number, last_date_app, last_un_id, ma_date_oper;

  if (:ma_is_active <= 0) then
  begin
    select res_text
    from hp_miniapp_annul_access(:ma_id, :ma_date_oper, :ma_from_un_id)
    into res_text;

    if (coalesce(res_text, '') <> '') then
      exception e_message(:res_text);
  end

  if (last_date_app <> current_date or last_un_id <> ma_from_un_id) then
  begin
    insert into miniapp (ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay,
                         ma_percent, ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save,
                         ma_from_un_id, ma_office_id, ma_currency_id, ma_no, ma_percent_category, ma_date_end,
                         ma_manager_id, ma_plan_disb_time, ma_expert_id, ma_repay_day, ma_source_system, ma_kommis_perc,
                         ma_refund, ma_add_info)
    select ma_id, -1, current_date, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay, ma_percent,
           ma_kolich_client, ma_memo, :ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save, :ma_from_un_id,
           ma_office_id, ma_currency_id, ma_no, ma_percent_category, '31.12.9999', ma_manager_id, ma_plan_disb_time,
           ma_expert_id, ma_repay_day, ma_source_system, ma_kommis_perc, ma_refund, ma_add_info
    from miniapp
    where ma_id = :ma_id and
          ma_number = :ma_number;
  end
  else
  begin
    update miniapp
    set ma_is_active = :ma_is_active,
        ma_date_app = current_date
    where ma_id = :ma_id and
          ma_number = :ma_number;
  end
  if (db_id in (39801, 41701)) then
  begin
    for select mac.mac_from_c_id
        from miniappclient mac
        where mac.mac_from_ma_id = :ma_id and
              mac.mac_amount > 0
        into :c_id
    do
    begin
      c_name_fio = '';
      c_pass_no = 0;
      c_full_address = '';
      c_from_reg_id = 0;
      select c_name_fio,
             c_pass_no,
             adl_address_str,
             c_birth_date,
             c_from_reg_id
      from clients, addrclient, addrlist
      where c_id = :c_id and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id rows 1
      into c_name_fio,
           c_pass_no,
           c_full_address,
           c_birth_date,
           c_from_reg_id;
      kr_text = '';
      select kr_text, kr_id
      from kontreslist, kontres k
      where krl_ma_id = :ma_id and
            krl_from_kr_id = kr_id and
            kr_is_active = 1 rows 1
      into :kr_text,
           :kr_id;
      if (:kr_id not in (89, 90, 78, 85)) then
      begin
        if (exists(select *
                   from moratory m
                   where m.m_from_c_id = :c_id and
                         m.m_is_active = 1)) then
        exit;
        insert into moratory (m_id, m_number, m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end, m_month,
                              m_is_active, m_comment_group, m_comment_officer, m_comment_manager, m_client_fio,
                              m_client_pass_no, m_client_address, m_client_birth_date, m_our_clients, m_from_reg_id,
                              m_no, m_datetime, m_datetime_edit, m_user_create, m_user_edit, m_db_code, m_original_id)
        values (-255, 1, :c_id, null, 1, current_date, '31.12.9999', 0, 1, null, :kr_text, null, :c_name_fio,
                :c_pass_no, :c_full_address, :c_birth_date, 1, :c_from_reg_id, null, current_timestamp,
                current_timestamp, null, null, null, null);
      end
    end
  end
end
