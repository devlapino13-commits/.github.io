CREATE PROCEDURE AAREP_CALC_PERC_BUSINESS(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    AAREP_K_ID DECIMAL(18,0),
    AAREP_BALANCE_BEGIN DECIMAL(18,7),
    AAREP_PERC_BEGIN DECIMAL(18,7),
    AAREP_PERC_CALCED DECIMAL(18,7),
    AAREP_PERC_PAYED DECIMAL(18,7),
    AAREP_PERC_WRITEOFF DECIMAL(18,7),
    AAREP_PERC_END DECIMAL(18,7),
    AAREP_BALANCE_END DECIMAL(18,7),
    AAREP_KONTRACT_NO VARCHAR(20),
    AAREP_CLIENT_NAME VARCHAR(100),
    AAREP_IS_INDIV VARCHAR(10),
    AAREP_PRODUCT_NAME VARCHAR(50),
    AAREP_PRODUCT_KIND STR_20,
    AAREP_DISB_DATE DATE,
    AAREP_DISB_AMOUNT DECIMAL(18,4),
    AAREP_BUSINESS VARCHAR(100),
    AAREP_BUS_TYPE VARCHAR(100),
    KNC_WORK_KOLICH INTEGER,
    AAREP_PERC_TRANSFER DECIMAL(18,4))
AS
declare variable aarep_loan_t integer;
declare variable kc_amount decimal(18,4);
declare variable aarep_balance_begin_t decimal(18,4);
declare variable aarep_perc_begin_t decimal(18,4);
declare variable aarep_perc_calced_t decimal(18,4);
declare variable aarep_perc_payed_t decimal(18,4);
declare variable aarep_perc_writeoff_t decimal(18,4);
declare variable aarep_perc_end_t decimal(18,4);
declare variable aarep_balance_end_t decimal(18,4);
declare variable aarep_disb_amount_t decimal(18,4);
declare variable aarep_balance_begin_0 decimal(18,4);
declare variable aarep_perc_begin_0 decimal(18,4);
declare variable aarep_perc_calced_0 decimal(18,4);
declare variable aarep_perc_payed_0 decimal(18,4);
declare variable aarep_perc_writeoff_0 decimal(18,4);
declare variable aarep_perc_end_0 decimal(18,4);
declare variable aarep_balance_end_0 decimal(18,4);
declare variable kol_chet integer;
declare variable kol_cl integer;
declare variable curr_code varchar(3);
declare variable curr_prior varchar(3);
declare variable curr_rate_def decimal(18,4);
declare variable k_curr_rate decimal(18,4);
declare variable curr_rate decimal(18,4);
declare variable aarep_perc_transfer_t decimal(18,4) = 0;
declare variable aarep_perc_transfer_0 decimal(18,4) = 0;
declare variable aarep_from_reg_id integer;
begin
  -- Начисленные проценты в разрезе типа бизнеса
  curr_prior = '000';
  for select k_id, k_kontract_no, pr_loan_type, pr_common_name,
             iif(pr_code containing '.IPF.', 'ИПФ', 'Классика'), k_kolich_client_approved,
             k_index_curr_code, k_index_curr_rate, k_amount_approved, k_disb_date, aarep_perc_begin,
             aarep_perc_calced, aarep_perc_payed, aarep_perc_writeoff, aarep_perc_end, aarep_perc_transfer,
             aarep_from_reg_id
      from kontract, product, aarep_calc_perc_list_tmp(:report_region_id, :report_date_begin, :report_date_end)
      where k_id = aarep_k_id and
            k_from_pr_id = pr_id
      order by k_index_curr_code, pr_loan_type, pr_id
      into :aarep_k_id, :aarep_kontract_no, :aarep_loan_t, :aarep_product_name, :aarep_product_kind, :kol_cl,
           :curr_code, :k_curr_rate, :aarep_disb_amount_t, :aarep_disb_date, :aarep_perc_begin_t,
           :aarep_perc_calced_t, :aarep_perc_payed_t, :aarep_perc_writeoff_t, :aarep_perc_end_t,
           :aarep_perc_transfer_t, :aarep_from_reg_id
  do
  begin
    aarep_balance_begin_t = 0;
    select oks_balance
    from oksummary
    where oks_date_begin <= (:report_date_begin - 1) and
          oks_date_end >= (:report_date_begin - 1) and
          oks_from_k_id = :aarep_k_id and
          oks_region_id = :aarep_from_reg_id and
          oks_close_state = 0
    into :aarep_balance_begin_t;

    aarep_balance_end_t = 0;
    select oks_balance
    from oksummary
    where oks_date_begin <= :report_date_end and
          oks_date_end >= :report_date_end and
          oks_from_k_id = :aarep_k_id and
          oks_region_id = :aarep_from_reg_id and
          oks_close_state = 0
    into :aarep_balance_end_t;

    if (curr_prior <> :curr_code) then
    begin
      curr_rate = 1;
      select cr_rate
      from opercurr
      where cr_from_cu_id = :curr_code and
            cr_date <= :report_date_end and
            cr_date_end >= :report_date_end
      into :curr_rate;
      if (curr_rate is null) then
        curr_rate = 1;
      curr_prior = :curr_code;
    end
    curr_rate_def = curr_rate;
    if (curr_rate < k_curr_rate) then
      curr_rate_def = k_curr_rate;

    if (aarep_loan_t = 1) then
      aarep_is_indiv = 'Индив.';
    else
      aarep_is_indiv = 'Группа';

    aarep_balance_begin_0 = :aarep_balance_begin_t;
    aarep_perc_begin_0 = :aarep_perc_begin_t;
    aarep_perc_calced_0 = :aarep_perc_calced_t;
    aarep_perc_payed_0 = :aarep_perc_payed_t;
    aarep_perc_writeoff_0 = :aarep_perc_writeoff_t;
    aarep_perc_end_0 = :aarep_perc_end_t;
    aarep_balance_end_0 = :aarep_balance_end_t;
    aarep_perc_transfer_0 = :aarep_perc_transfer_t;
    kol_chet = 0;

    for select c_name_fio, bs_name, bsl_name, kc_work_kolich, sum(kc_amount_approved)
        from knclient, bussubtype, clients, busstandartlist, busstandartlink
        where kc_amount_approved is not null and
              kc_amount_approved <> 0 and
              kc_from_k_id = :aarep_k_id and
              kc_from_c_id = c_id and
              bsl_from_bsm_id = 1 and
              bslk_from_bsm_id = 1 and
              bslk_from_bsl_id = bsl_id and
              bslk_from_bs_id = bs_id and
              kc_busuness_subtype = bs_id
        group by c_name_fio, bs_name, bsl_name, kc_work_kolich
        into :aarep_client_name, :aarep_business, :aarep_bus_type, :knc_work_kolich, :kc_amount
    do
    begin
      aarep_disb_amount = :kc_amount;
      if (kc_amount <> aarep_disb_amount_t) then
      begin
        kol_chet = kol_chet + 1;
        aarep_balance_begin = cast(:aarep_balance_begin_t / :aarep_disb_amount_t as decimal(18,7));
        aarep_balance_begin = cast(:aarep_balance_begin * :kc_amount as decimal(15,2));

        aarep_perc_begin = cast(:aarep_perc_begin_t / :aarep_disb_amount_t as decimal(18,7));
        aarep_perc_begin = cast(:aarep_perc_begin * :kc_amount as decimal(15,2));

        aarep_perc_calced = cast(:aarep_perc_calced_t / :aarep_disb_amount_t as decimal(18,7));
        aarep_perc_calced = cast(:aarep_perc_calced * :kc_amount as decimal(15,2));

        aarep_perc_payed = cast(:aarep_perc_payed_t / :aarep_disb_amount_t as decimal(18,7));
        aarep_perc_payed = cast(:aarep_perc_payed * :kc_amount as decimal(15,2));

        aarep_perc_writeoff = cast(:aarep_perc_writeoff_t / :aarep_disb_amount_t as decimal(18,7));
        aarep_perc_writeoff = cast(:aarep_perc_writeoff * :kc_amount as decimal(15,2));

        aarep_perc_end = cast(:aarep_perc_end_t / :aarep_disb_amount_t as decimal(18,7));
        aarep_perc_end = cast(:aarep_perc_end * :kc_amount as decimal(15,2));

        aarep_balance_end = cast(:aarep_balance_end_t / :aarep_disb_amount_t as decimal(18,7));
        aarep_balance_end = cast(:aarep_balance_end * :kc_amount as decimal(15,2));

        aarep_perc_transfer = cast(:aarep_perc_transfer_t / :aarep_disb_amount_t as decimal(18,7));
        aarep_perc_transfer = cast(:aarep_perc_transfer * :kc_amount as decimal(15,2));

        if (kol_chet < kol_cl) then
        begin
          aarep_balance_begin_0 = aarep_balance_begin_0 - aarep_balance_begin;
          aarep_perc_begin_0 = aarep_perc_begin_0 - aarep_perc_begin;
          aarep_perc_calced_0 = aarep_perc_calced_0 - aarep_perc_calced;
          aarep_perc_payed_0 = aarep_perc_payed_0 - aarep_perc_payed;
          aarep_perc_writeoff_0 = aarep_perc_writeoff_0 - aarep_perc_writeoff;
          aarep_perc_end_0 = aarep_perc_end_0 - aarep_perc_end;
          aarep_balance_end_0 = aarep_balance_end_0 - aarep_balance_end;
          aarep_perc_transfer_0 = aarep_perc_transfer_0 - aarep_perc_transfer;
        end
        else
        begin
          aarep_balance_begin = aarep_balance_begin_0;
          aarep_perc_begin = aarep_perc_begin_0;
          aarep_perc_calced = aarep_perc_calced_0;
          aarep_perc_payed = aarep_perc_payed_0;
          aarep_perc_writeoff = aarep_perc_writeoff_0;
          aarep_perc_end = aarep_perc_end_0;
          aarep_balance_end = aarep_balance_end_0;
          aarep_perc_transfer = aarep_perc_transfer_0;
        end
      end
      else
      begin
        aarep_balance_begin = :aarep_balance_begin_t;
        aarep_perc_begin = :aarep_perc_begin_t;
        aarep_perc_calced = :aarep_perc_calced_t;
        aarep_perc_payed = :aarep_perc_payed_t;
        aarep_perc_writeoff = :aarep_perc_writeoff_t;
        aarep_perc_end = :aarep_perc_end_t;
        aarep_balance_end = :aarep_balance_end_t;
        aarep_perc_transfer = aarep_perc_transfer_t;
      end
      if (curr_rate_def <> 1) then
      begin
        aarep_balance_begin = cast(:aarep_balance_begin * :curr_rate_def as decimal(15,2));
        aarep_perc_begin = cast(:aarep_perc_begin * :curr_rate_def as decimal(15,2));
        aarep_perc_calced = cast(:aarep_perc_calced * :curr_rate_def as decimal(15,2));
        aarep_perc_payed = cast(:aarep_perc_payed * :curr_rate_def as decimal(15,2));
        aarep_perc_writeoff = cast(:aarep_perc_writeoff * :curr_rate_def as decimal(15,2));
        aarep_perc_end = cast(:aarep_perc_end * :curr_rate_def as decimal(15,2));
        aarep_balance_end = cast(:aarep_balance_end * :curr_rate_def as decimal(15,2));
        aarep_perc_transfer = cast(:aarep_perc_transfer * :curr_rate_def as decimal(15,2));
      end
      suspend;
    end
  end
end
