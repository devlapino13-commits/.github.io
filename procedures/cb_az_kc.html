CREATE PROCEDURE CB_AZ_KC(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PARENT_NAME D_STR_100,
    REG_NAME D_STR_100,
    C_NAME_FIO D_STR_255,
    JK_NAME STR_100,
    C_WORKED_MONTHS D_INTEGER,
    KOL_DISB D_INTEGER,
    BONUS_KOEF D_INTEGER,
    KOLVO_PROSS_KREDIT D_INTEGER,
    BONUS_KOEF_PROSS D_CURRENCY,
    BONUS_TO_PAY D_CURRENCY,
    FINE_PROSS D_CURRENCY,
    STANDART_BONUS D_CURRENCY,
    DISB_BONUS D_CURRENCY,
    TOTAL_TO_PAY D_CURRENCY)
AS
declare variable c_id d_id;
declare variable reg_id d_integer;
declare variable parent d_integer;
declare variable ps_date_begin d_date;
begin
  bonus_koef = 5;
  bonus_koef_pross = 10;

  for select ps_from_c_id, ps_from_reg_id, jk_name
      from hr_pstatus
      inner join joblist jl on ps_from_jl_id = jl_id
      inner join jobkind jk on jl_kind = jk_id
      inner join region_list(:rep_region_id, 1, 3, 1) on ps_from_reg_id = id_list
      where ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            jk_system_code = 'JobSystem.LO_Standart'
      into :c_id, :reg_id, jk_name
  do
  begin
    c_worked_months = 0;
    select ps.ps_date_begin
    from hr_pstatus ps
    inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id
    where ps.ps_status = 1 and
          ho.ho_status = 1 and
          ho_order_type = 1 and
          ps.ps_from_c_id = :c_id
          order by ps.ps_date_begin desc
          rows 1
    into ps_date_begin;

    c_worked_months = datediff(month, :ps_date_begin, :rep_date_end);
    c_worked_months = coalesce(c_worked_months, 0);
    if ((c_worked_months < 2) or (c_worked_months > 9)) then
      continue;

    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    select reg_code || ' ' || reg_name, parent
    from region
    where reg_unicode = :reg_id
    into :reg_name, :parent;

    select reg.reg_name
    from region reg
    where reg.reg_unicode = :parent
    into parent_name;

    kol_disb = 0;
    select count(k_id)
    from kontstate, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_kc_id = :c_id and
          k_fact_disb_lo_id = :c_id and
          ks_from_k_id = k_id
    into kol_disb;

    kolvo_pross_kredit = 0;
    select count(*) -- кол. просрочен. кредит на конец отчет. мес 0+
    from oksummary oks
    inner join kontract k on k.k_id = oks.oks_from_k_id and
          k.k_fact_disb_lo_id = :c_id
    where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_kc_id = :c_id and
          oks_close_state = 0 and
          oks.oks_delinq_os > 0 and
          cast(:rep_date_end as date) - oks.oks_delinq_date > 0
    into kolvo_pross_kredit;

    bonus_to_pay = kol_disb * bonus_koef;
    fine_pross = kolvo_pross_kredit * bonus_koef_pross;
    total_to_pay = bonus_to_pay - fine_pross;
    disb_bonus = coalesce(total_to_pay, 0);

    select aarep_bonus_total_to_pay
    from cb_az_lo_201602_one_lo(:reg_id, :c_id, :rep_date_end, 0)
    into standart_bonus;

    if (standart_bonus > total_to_pay) then
      total_to_pay = standart_bonus;

    if (total_to_pay < 0) then
      total_to_pay = 0;
    suspend;
  end
end
