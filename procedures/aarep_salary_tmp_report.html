CREATE PROCEDURE AAREP_SALARY_TMP_REPORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_C_ID DECIMAL(18,0))
RETURNS (
    ASR_REG_ID INTEGER,
    ASR_C_ID DECIMAL(18,0),
    ASR_SAC_ID INTEGER,
    ASR_VALUE NUMERIC(18,4),
    ASR_C_NAME_FIO VARCHAR(100),
    ASR_REG_NAME VARCHAR(100),
    ASR_SALDO_BEGIN DECIMAL(18,4),
    ASR_SALDO_END DECIMAL(18,4))
AS
declare variable c_prior decimal(18,0);
declare variable reg_prior integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  c_prior = 0;
  reg_prior = 0;
  asr_saldo_begin = 0;
  asr_saldo_end = 0;

  if (report_c_id = 0) then
  begin
    for select osm_from_c_id, osd_from_sac_id, sum(osd_value)
        from opersalmain, opersaldetail
        where osm_id = osd_from_osm_id and
              osm_ol_is_active = 1 and
              osm_ol_date between :report_date_begin and :report_date_end
        group by osm_from_c_id, osd_from_sac_id
        order by osm_from_c_id
        into asr_c_id, asr_sac_id, asr_value
    do
    begin
      if (asr_c_id <> c_prior) then
      begin
        asr_reg_id = null;
        select rep_region_id
        from hp_get_employer_region(:asr_c_id, :report_date_end, :report_region_id)
        into :asr_reg_id;
        if (asr_reg_id is not null) then
        begin
          select c_name_fio
          from clients
          where c_id = :asr_c_id
          into :asr_c_name_fio;
          select reg_name
          from region
          where reg_unicode = :asr_reg_id
          into :asr_reg_name;
        end
        select rep_salary_saldo
        from hp_get_employer_salary_saldo(:asr_c_id, (:report_date_begin - 1))
        into :asr_saldo_begin;
        select rep_salary_saldo
        from hp_get_employer_salary_saldo(:asr_c_id, :report_date_end)
        into :asr_saldo_end;
        c_prior = asr_c_id;
      end

      if (asr_reg_id is not null) then
        suspend;
    end
  end
  else
  begin
    for select osm_from_c_id, osd_from_sac_id, sum(osd_value)
        from opersalmain, opersaldetail
        where osm_id = osd_from_osm_id and
              osm_ol_is_active = 1 and
              osm_ol_date between :report_date_begin and :report_date_end and
              osm_from_c_id = :report_c_id
        group by osm_from_c_id, osd_from_sac_id
        order by osm_from_c_id
        into asr_c_id, asr_sac_id, asr_value
    do
    begin
      if (asr_c_id <> c_prior) then
      begin
        asr_reg_id = null;
        select rep_region_id
        from hp_get_employer_region(:asr_c_id, :report_date_end, :report_region_id)
        into :asr_reg_id;
        if (asr_reg_id is not null) then
        begin
          select c_name_fio
          from clients
          where c_id = :asr_c_id
          into :asr_c_name_fio;
          select reg_name
          from region
          where reg_unicode = :asr_reg_id
          into :asr_reg_name;
        end
        select rep_salary_saldo
        from hp_get_employer_salary_saldo(:asr_c_id, (:report_date_begin - 1))
        into :asr_saldo_begin;
        select rep_salary_saldo
        from hp_get_employer_salary_saldo(:asr_c_id, :report_date_end)
        into :asr_saldo_end;
        c_prior = asr_c_id;
      end

      if (asr_reg_id is not null) then
        suspend;
    end
  end

end
