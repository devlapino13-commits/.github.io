CREATE PROCEDURE AAREP_TRANSKRED_BUSINESS_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    TRANS_OS DECIMAL(18,4),
    TRANS_REGION_NAME VARCHAR(100),
    TRANS_BT_NAME VARCHAR(50),
    TRANS_BS_NAME VARCHAR(50))
AS
declare variable sum_os decimal(18,4);
declare variable k_id decimal(18,0);
declare variable k_all decimal(18,4);
declare variable c_amount decimal(18,4);
declare variable cm_is_client_mode integer;
begin
-- operkred_fix
  select cm_is_client_mode
  from company
  into cm_is_client_mode;

  if (cm_is_client_mode = 0) then
  begin
    for select reg_code || ' ' || reg_name, k_id, k_amount_approved, sum(ok_os * cr_rate)
        from operkred, kontract, opercurr, region, region_list(:report_region_id, 1, -1, 1)
        where ok_from_k_id = k_id and
              k_index_curr_code = cr_from_cu_id and
              ok_ol_date between cr_date and cr_date_end and
              ok_ol_is_active = 1 and
              ok_ol_date between :report_date_begin and :report_date_end and
              ok_type = 4 and
              ok_ol_reg_id = reg_unicode and
              ok_ol_reg_id = id_list and
              k_amount_approved <> 0 and
              ok_from_c_id = 0
        group by reg_code, reg_name, k_id, k_amount_approved
        into :trans_region_name, :k_id, :k_all, :sum_os
    do
    begin
      for select bt_name, bs_name, kc_amount_approved
          from knclient, bussubtype, bustype
          where kc_from_k_id = :k_id and
                kc_busuness_subtype = bs_id and
                bs_from_bt_id = bt_id and
                kc_amount_approved <> 0 and
                kc_amount_approved is not null
          into :trans_bt_name, trans_bs_name, c_amount
      do
      begin
        trans_os = cast(:sum_os / :k_all * :c_amount as decimal(15,2));
        suspend;
      end
    end
    exit;
  end

  for select sum(ok_os * cr_rate), reg_code || ' ' || reg_name, bt_name, bs_name
      from operkred, opercurr, kontract, region, region_list(:report_region_id, 1, -1, 1), knclient, bussubtype, bustype
      where ok_from_k_id = k_id and
            k_index_curr_code = cr_from_cu_id and
            ok_ol_date between cr_date and cr_date_end and
            ok_ol_is_active = 1 and
            ok_ol_date between :report_date_begin and :report_date_end and
            ok_type = 4 and
            ok_ol_reg_id = reg_unicode and
            ok_ol_reg_id = id_list and
            k_amount_approved <> 0 and
            ok_from_c_id = kc_from_c_id and
            ok_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_busuness_subtype = bs_id and
            bs_from_bt_id = bt_id
      group by reg_code, reg_name, bt_name, bs_name
      into trans_os, trans_region_name, trans_bt_name, trans_bs_name
  do
  begin

    suspend;
  end
end
