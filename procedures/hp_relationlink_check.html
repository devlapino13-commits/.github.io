CREATE PROCEDURE HP_RELATIONLINK_CHECK(
    C_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0),
    IS_CANDIDATE INTEGER)
RETURNS (
    RESULT_ERROR INTEGER)
AS
declare variable rll_client_2 d_id;
begin
  -- Если параметр Is_Cadidate=1, то система сначала создаст новые связи
  if (is_candidate = 1) then
    execute procedure hp_relationlink_update(:c_id, :un_id, :is_candidate);

  result_error = 0;
/*  if (exists(select *
             from relationlink
             where rll_client_1 = :c_id and
                   rll_is_active = 1 and
                   rll_link_status <> 2)) then
    result_error = 1; -- Есть НЕ утвержденные */

  for select rk.rll_client_2
      from relationlink rk
      where rll_client_1 = :c_id and
            rll_is_active = 1 and
            rll_link_status <> 2
      into :rll_client_2
  do
  begin
    if (exists(select hps.ps_from_c_id
               from hr_pstatus hps
               where current_date between ps_date_begin and ps_date_end and
                     ps_is_working = 1 and
                     hps.ps_status = 1 and
                     hps.ps_from_c_id = :rll_client_2)) then
      result_error = 1;-- Есть НЕ утвержденные активныые коллеги BOARD-1950  Не можем оформить найм (из-за родственных связей)
  end
  suspend;
end
