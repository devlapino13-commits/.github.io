CREATE PROCEDURE HR_APPLICAT_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_KIND INTEGER)
RETURNS (
    HRA_ID INTEGER,
    HRA_FROM_REG_ID INTEGER,
    HRA_FROM_C_ID DECIMAL(18,0),
    HRA_DATE_BEGIN DATE,
    HRA_DATE_END DATE,
    HRA_FROM_HO_ID DECIMAL(18,0),
    HRA_USER_ID DECIMAL(18,0),
    HRA_DATETIME TIMESTAMP,
    HRA_PERSONAL_FIO VARCHAR(100),
    HRA_USER_NAME VARCHAR(100),
    HRA_ORDER_USER_FIO VARCHAR(100),
    HRA_ORDER_SAVE_TIME TIMESTAMP,
    HRA_APPROVED INTEGER,
    HRA_ORDER_TYPE INTEGER,
    HRA_ORDER_NAME VARCHAR(100),
    HRA_POSITION VARCHAR(512),
    HRA_IS_APPROVED INTEGER,
    HRA_IS_CHECKED INTEGER,
    HRA_MEMO VARCHAR(255),
    HRA_JL_NAME VARCHAR(100),
    HRA_REGION_NAME VARCHAR(100),
    HRA_CHANGER_ID DECIMAL(18,0),
    HRA_CHANGER_FIO VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable ignore smallint;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select hra_id, hra_from_reg_id, hra_from_c_id, hra_date_begin, hra_date_end, hra_from_ho_id, hra_user_id,
             hra_datetime, hra_is_approved, hra_is_checked, hra_memo, un_name, c_name_fio, hra_order_type, sawt_name,
             hra_changer_id
      from hr_applicat, username, clients, region_list(:rep_region_id, 1, -1, 1), sa_worktype
      where (hra_from_reg_id = id_list) and
            hra_from_c_id = c_id and
            hra_user_id = un_id and
            hra_order_type = sawt_id and
            ((hra_date_begin >= :rep_date_begin and
            hra_date_begin <= :rep_date_end) or
            (hra_date_end >= :rep_date_begin and
            hra_date_end <= :rep_date_end)) and
            (:rep_kind = 0 or
            (:rep_kind = 1 and
            hra_from_ho_id is null) or
            (:rep_kind = 2 and
            hra_from_ho_id is not null))
      into :hra_id, :hra_from_reg_id, :hra_from_c_id, :hra_date_begin, :hra_date_end, :hra_from_ho_id, :hra_user_id,
           :hra_datetime, :hra_is_approved, :hra_is_checked, :hra_memo, :hra_user_name, :hra_personal_fio,
           :hra_order_type, :hra_order_name, :hra_changer_id
  do
  begin
    hra_approved = 0;
    hra_order_save_time = null;
    hra_order_user_fio = null;

    hra_changer_fio = 'Не указан';
    if (hra_changer_id is not null) then
    begin
      if (hra_changer_id = 0) then
        hra_changer_fio = 'Заменяющий не требуется';
      else
      begin
        select c_name_fio
        from clients
        where c_id = :hra_changer_id
        into :hra_changer_fio;
      end
    end

    ignore = 0;

    if (:hra_from_ho_id is not null) then
    begin
      if (exists(select *
                 from hr_orders ho
                 where ho.ho_id = :hra_from_ho_id and
                       ho.ho_status = -1)) then
        ignore = 1;
      else
      begin
        hra_approved = 1;
        select c_name_fio, ho_save_time
        from hr_orders, username, clients
        where ho_from_un_id = un_id and
              un_from_c_id = c_id and
              ho_id = :hra_from_ho_id
        into :hra_order_user_fio, :hra_order_save_time;
      end
    end

    if (ignore = 0) then
    begin
      hra_position = '';
      select ps_text, ps_jl_name --Должность
      from hr_personal_last_position(:hra_from_c_id, :rep_date_end)
      rows 1
      into :hra_position, hra_jl_name;

      hra_region_name = '';

      select reg.reg_name
      from region reg
      where reg.reg_unicode = :hra_from_reg_id
      into :hra_region_name;

      suspend;
    end
  end
end
