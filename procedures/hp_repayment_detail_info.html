CREATE PROCEDURE HP_REPAYMENT_DETAIL_INFO(
    K_ID D_ID,
    REP_DATE D_DATE,
    KONT_NO D_STR_CODE = null)
RETURNS (
    RES_HEADER D_STR_255,
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_KOLICH_REPAY D_INTEGER,
    K_PERIOD D_FLOAT,
    K_STOP_PERC D_DATE,
    K_DELINQ_DATE D_DATE,
    K_PERCENT_YEAR D_PERCENT,
    K_USED_DAYS D_INTEGER,
    K_PERCENT_METHOD D_STR_255,
    K_ANNU_FORMULA D_STR_LONG,
    K_ANNU_CALC_STR D_STR_LONG,
    DATE_STOP_KONTRACT D_STR_20)
AS
declare variable pr_interest_method d_smallint;
declare variable k_percent_per_day d_percent;
declare variable k_percent_month d_percent;
declare variable k_real_kolich_repay d_integer;
declare variable annu_amount d_currency;
begin
  if (k_id is null and kont_no <> '') then
  begin
    select k_id
    from kontract
    where k_kontract_no = :kont_no
    into k_id;
  end

  select k_kontract_no, k_disb_date, k_amount_approved, k_kolich_repay, k_period, k_stop_perc,
         k_percent_per_day, pr_interest_method
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into k_kontract_no, k_disb_date, k_amount_approved, k_kolich_repay, k_period, k_stop_perc,
       k_percent_per_day, pr_interest_method;

  k_percent_month = k_percent_per_day * 30.0000000;
  k_percent_year = k_percent_per_day * 365.0000000;

  res_header = 'РАСЧЕТ ЗАДОЛЖЕННОСТИ ПО ДОГОВОРУ ЗАЙМА № ' || k_kontract_no;

  k_used_days = datediff(day, k_disb_date, coalesce(k_stop_perc, rep_date));

  select f_datetostr(date_stop_kontract)
  from hp_find_stop_kontract_date(:k_id)
  into:date_stop_kontract;

  select oks_delinq_date
  from oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin <= :rep_date and
        oks_date_end >= :rep_date and
        oks_delinq_date <= :rep_date
  into k_delinq_date;

  select count(*)
  from schedule
  where sh_from_k_id = :k_id and
        sh_os > 0
  into k_real_kolich_repay;

  k_annu_calc_str = '';
  if (pr_interest_method = 0) then
  begin
    k_percent_method = 'На первоначальную сумму займа.';
    k_annu_formula = 'П = ОС / Nпер + ОС * Ставка';
    k_annu_calc_str = 'П = %1 / %3 + %1 * %2';
    k_annu_calc_str = f_format(k_annu_calc_str, f_numtostr(k_amount_approved, 2), f_numtostr(k_percent_month, 3) || '%', k_real_kolich_repay);

    annu_amount = k_amount_approved / k_real_kolich_repay + k_amount_approved * k_percent_month / 100.00 ;
    k_annu_calc_str = k_annu_calc_str || ' = ' || f_numtostr(annu_amount, 2);
  end else
  begin
    k_percent_method = 'На остаток задолженности.';
    k_annu_formula = 'П = ОС * (Ставка + (Ставка/((1+ Ставка)^Nпер-1))';
    k_annu_calc_str = 'П = %1 * (%2 + (%2 / ((1+ %2)^%3-1))';
    k_annu_calc_str = f_format(k_annu_calc_str, f_numtostr(k_amount_approved, 2), f_numtostr(k_percent_month, 3) || '%', k_real_kolich_repay);
    annu_amount = k_amount_approved * ((k_percent_month / 100.0000000) + ((k_percent_month / 100.0000000) / (power(1 + (k_percent_month / 100.0000000), k_real_kolich_repay) - 1)));
    k_annu_calc_str = k_annu_calc_str || ' = ' || f_numtostr(annu_amount, 2);
  end

  if (k_delinq_date is null) then
  begin
    select ok_sh_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date and
          ok_type <= 4
    group by ok_sh_date
    having  sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
    order by ok_sh_date
    rows 1
    into k_delinq_date;
  end

  suspend;
end
