CREATE PROCEDURE UXP$UXFRLOOKUP_UPDATE(
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID,
    ULP_SOURCE D_STR_CODE,
    ULP_LINK_KIND D_SMALLINT = 0)
AS
declare variable ulp_uf_uuid d_uuid;
declare variable us_uuid d_uuid;
declare variable ufp_uuid d_uuid;
begin
  us_uuid = uxf$uniq(:ulp_source);

  if (ulp_link_kind = 0) then
  begin
    if (exists(select * from ux$source
    where us_code = :ulp_source and
          us_keyfields = '')) then
      exception e_message 'У источника не указано ключевое поле';
  end

  select ufp_uuid
  from ux$frparam
  where ufp_ufr_uuid = :ufp_ufr_uuid and
        ufp_uf_uuid = :ufp_uf_uuid
  into ufp_uuid;

  if (ufp_uuid is null) then
  begin
    insert into ux$frparam (ufp_ufr_uuid, ufp_uf_uuid, ufp_editor)
    values (:ufp_ufr_uuid, :ufp_uf_uuid, 'lookup')
    returning (ufp_uuid)
    into ufp_uuid;
  end

  update or insert into ux$frlookup (ufl_ufp_uuid, ufl_us_uuid, ufl_list_kind)
  values (:ufp_uuid, :us_uuid, :ulp_link_kind)
  matching (ufl_ufp_uuid);

  delete from ux$frparam
  where ufp_parent = :ufp_uuid;

  update ux$frparam
  set ufp_editor = 'lookup'
  where ufp_uuid = :ufp_uuid and
        ufp_editor <> 'lookup';

  for select uf_uuid
  from ux$field
  where uf_owner = :us_uuid and
        uf_is_param = 1
  into ulp_uf_uuid
  do
  begin
    insert into ux$frparam (ufp_ufr_uuid, ufp_uf_uuid, ufp_parent)
    values (:ufp_ufr_uuid, :ulp_uf_uuid, :ufp_uuid);
  end
end
