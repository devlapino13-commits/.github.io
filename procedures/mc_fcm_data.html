CREATE PROCEDURE MC_FCM_DATA(
    BODY D_STR_LONG,
    TITLE D_STR_255,
    REP_ID D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_STR_CODE,
    DATE_KIND D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    MCL_ID D_ID)
RETURNS (
    RES_TEXT D_STR_2048)
AS
declare variable msg_qty integer;
begin
  -- Процедура формирует тело Firebase-уведомления для МПКлиентов

  -- Получаем количество непрочитанных сообщений
  if (mcl_id is null) then  -- для массовых уведомлений "для всех клиентов"
    msg_qty = 1;
  else
    select message_qty
    from mc_unread_message_qty(:mcl_id)
    into msg_qty;

  res_text = '';
  if (coalesce(body, '') = '' or coalesce(title, '') = '') then
  begin
    suspend;
    exit;
  end

  res_text =
    '"priority":"high",' ||
    '"notification":{' ||
    f_json_make_pair('title', title, 'str') || ',' ||
    f_json_make_pair('body', body, 'str') || ',' ||
    f_json_make_pair('badge', msg_qty, 'str') || ',' ||
    f_json_make_pair('notification_count', msg_qty, 'num') || '},' ||
    '"data":{"type":"notification",' ||
    f_json_make_pair('title', title, 'str') || ',' ||           --, родительская процедура mc_send_push отправляет этот параметр равным 0
    f_json_make_pair('body', body, 'str') || ',' ||
    f_json_make_pair('rep_id', rep_id, 'str') || ',' ||
    f_json_make_pair('obj_kind', obj_kind, 'str') || ',' ||
    f_json_make_pair('obj_id', coalesce(obj_id, ''), 'str');

  -- Даты передавать не обязательно
  if (date_kind >= 1 and date_begin is not null) then
  begin
    res_text = res_text || ',' ||
      f_json_make_pair('date_kind', date_kind, 'str') || ',' ||
      f_json_make_pair('date_begin',  f_datetostr(date_begin), 'str');
    if (date_kind = 2 and date_begin is not null) then
      res_text = res_text || ',' ||
       f_json_make_pair('date_end',  f_datetostr(date_end), 'str');
  end

  res_text = res_text || '}';
  suspend;
end
