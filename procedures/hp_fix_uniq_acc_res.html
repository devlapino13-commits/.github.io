CREATE PROCEDURE HP_FIX_UNIQ_ACC_RES(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable C_BIRTH_DATE D_DATE;
declare variable UN_ID D_ID;
begin
  -- RFCKG-4715
  -- Команда "Прикрепить лицевой счет"

  res_text = '';

  -- Проверка входного параметра
  if (coalesce(:c_id, 0) = 0) then
    exception e_message('Укажите клиента');

  -- Получаем данные клиента
  select cl.c_birth_date,
         cl.c_from_un_id
  from clients cl
  where cl.c_id = :c_id
    and cl.c_birthday_commited = 1
  into :c_birth_date, :un_id;

  if (row_count = 0) then
    exception e_message('Клиент не найден или дата рождения не подтверждена');

  -- Проверяем лицевой счет
  select uniq_acc_str
  from hp_show_clientuniq_acc_no(:c_id)
  into :res_text;

  if (res_text = 'xxx') then
  begin
    execute procedure hp_generate_c_uniq_id(
        :c_id,
        :c_birth_date,
        1,
        :un_id
    );

    select uniq_acc_str
    from hp_show_clientuniq_acc_no(:c_id)
    into :res_text;
  end

  suspend;
end
