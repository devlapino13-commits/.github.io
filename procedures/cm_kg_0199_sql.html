CREATE PROCEDURE CM_KG_0199_SQL(
    K_ID ID_TYPE,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id id_type;
declare variable un_id_2 d_id;
declare variable count_kontract d_smallint;
declare variable k_id_2 d_id;
declare variable emp_c_id id_type;
begin
  exit;

  for select kc.kc_from_c_id
      from kontract k
      inner join knclient kc on k.k_id = kc.kc_from_k_id and
            kc.kc_amount_approved > 0
      where k.k_id = :k_id
      into :c_id
  do
  begin
      if (k_id in(109010211235100101, 109010186467000201, 109010213017400101, 109010214885700101,109010215910200101)) then exit;
      count_kontract = 0;

      --получаем контракты клиента, которые были выданы за текущий день
      for select k.k_id
          from kontract k
          inner join kontstate ks on k.k_id = ks.ks_from_k_id and
                ks.ks_state = 1 and
                ks.ks_ol_is_delete = 0
          inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id and
                kc.kc_amount_approved > 0 and
                kc.kc_from_c_id = :c_id
          inner join knaktlist kal on k.k_from_kf_id = kal.kal_from_kf_id and
                current_date between kal.kal_date and kal.kal_date_end and
                kal.is_delete = 0
          where k.k_disb_date = current_date and
                k.k_id <> :k_id
          into :k_id_2
      do
      begin
        count_kontract = count_kontract + 1;

        -- получаем un_id уполномоченного лица, который должен утверждать данный контракт
        un_id_2 = null;
        select kl.kl_user_id
        from kontlog kl
        where kl.kl_from_k_id = :k_id_2
        order by kl.kl_id desc rows 1
        into :un_id_2;

        -- получаем c_id уполномоченного лица
        emp_c_id = null;
        select un.un_from_c_id
        from username un
        where un.un_id = :un_id_2
        into :emp_c_id;

        -- проверяем, находится ли уполномоченное лицо в данный момент в отпуске
        if (exists(select *
                   from hr_voctravel hrv
                   inner join hr_orders ho on hrv.hrv_from_ho_id = ho.ho_id
                   inner join sa_worktype saw on ho.ho_order_type = saw.sawt_id
                   where hrv.hrv_from_c_id = :emp_c_id and
                         current_date between hrv.hrv_date_begin and hrv.hrv_date_end and
                         hrv.hrv_status = 1 and
                         ho.ho_status = 1 and
                         saw.sawt_id in (7, 8, 9, 10, 13, 16, 17, 18, 24, 25))) then
        begin
          -- получаем заменяющее лицо, если сотрудник в отпуске
          un_id_2 = null;
          select un.un_id
          from hr_orders ho
          inner join hr_voctravel hrv on ho.ho_id = hrv.hrv_from_ho_id
          inner join username un on hrv.hrv_changer_id = un.un_from_c_id
          where hrv.hrv_from_c_id = :emp_c_id and
                current_date between hrv.hrv_date_begin and hrv.hrv_date_end and
                hrv.hrv_status = 1 and
                ho.ho_status = 1
          order by ho.ho_date desc
          rows 1
          into :un_id_2;
        end

        un_id_2 = coalesce(un_id_2, 0);

        if (un_id <> un_id_2 and
            count_kontract >= 1) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Клиент, получающий в 1 день несколько кредитов, должен быть утверждён только одним и тем же уполномоченным лицом';
          suspend;
        end
      end
  end
end
