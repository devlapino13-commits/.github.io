CREATE PROCEDURE CM_KG_0139_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable kc_circle integer;
declare variable categ d_integer;
declare variable ma_app_id d_integer;
declare variable pr_code d_str_100;
declare variable kbm_id d_id;
declare variable max_amount d_integer;
begin
  if (k_id = 109010128151000201) then exit;
  res_error_is_critical = 1;
  for select k_disb_date, k_amount_approved, kc_circle, kc_from_c_id, kont.k_from_ma_app_id, pr_code
      from kontract kont
      inner join product pr on k_from_pr_id = pr_id and
            pr_code containing '.IND.' and
            pr_code containing '.BIRGE.' and
            pr_code not containing '.EC.' and
            pr_code not containing '.PAR.' and
            pr_code not containing '.EXPRESS.' and
            pr_code not containing '.SOCIAL.' and
            kont.k_refund = 0
      inner join knclient kc on k_id = kc_from_k_id and
            kc_amount_approved > 0
      where k_id = :k_id
      into :kf_disb_date, :kc_amount, :kc_circle, c_id, :ma_app_id, :pr_code
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
        exit;
    -- под выборку подпадают "Бирге стандарт" и "Бирге Мугалим"
    categ = 0;
    max_amount = 80000;
    if (kc_amount < 80000) then
        continue;
    -- 1.Если у клиента 6 цикл (в Компании) и более, тогда ...
    if ((kc_circle >= 6) and (pr_code not containing '.MUGALIM.')) then
    begin
        categ = 3;
        max_amount = 200000;
    end
    -- 2. Если у клиента 4 -5 цикл в компании и клиент ранее хоть раз брал заем, где его собственная сумма займа 90 000 и выше, тогда ...
    if ((categ = 0) and (kc_circle >=4) and (exists(select *
                                               from knclient kc
                                               inner join kontstate ks on kc_from_k_id = ks_from_k_id and
                                                     ks_ol_is_delete = 0 and
                                                     ks_state = 1 and
                                                     ks_date_begin <= :kf_disb_date
                                               where kc_from_c_id = :c_id and
                                                     kc_amount_approved >= 90000.0))) then
    begin
        categ = 2;
        max_amount = 150000;
    end
    -- 3. Если у есть кредитная история (не важно в Компании или по КИБ), и он не соответствует условиям пунктов (1), (2), тогда ...
    if ((categ = 0) and (kc_circle > 2)) then
    begin
        categ = 1;
        max_amount = 100000;
    end
    if (categ = 0) then
    begin
        kbm_id = null;
        select kbm_id
        from hp_kibminiapp_last(:c_id, :ma_app_id)
        into kbm_id;

        if (kbm_id is not null) then
        begin
            if (exists (select *
                        from kibhistory kh
                        where kh_from_kbm_id = :kbm_id
                        and kh_creditstatus not containing('Отказано')
                        and kh_creditstatus not containing('Рассмотрение заявки')
                        and kh_is_our = 0
                        and kh_is_debtor = 1)) then
             begin
                 categ = 1;
                 max_amount = 100000;
             end
        end
    end
    if (kc_amount > max_amount) then
    begin
        res_error_text = 'Для продукта '||iif(pr_code containing '.MUGALIM.', 'Бирге Мугалим', 'Бирге стандарт')||', для данного клиента, максимально допустимая сумма '||max_amount;
        suspend;
    end
  end
end
