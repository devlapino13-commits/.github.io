CREATE PROCEDURE HP_SEND_REQMAIN_NOTIFY
AS
declare variable OFFICE_MANAGER_ID D_ID;
declare variable CASHIER_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable RM_ID D_ID;
declare variable RM_DATE D_DATE;
declare variable RM_STATUS D_STR_CODE;
declare variable RM_NO D_INTEGER;
declare variable MESSAGE_BODY D_BLOB_TEXT;
declare variable EMAIL_ADDRESS D_STR_255;
begin
    message_body = '';

    for select
        ps.ps_from_c_id,
        ps.ps_from_reg_id
    from hr_pstatus ps
    where ps.ps_from_jl_id = 225 -- Менеджер офиса
        and ps.ps_is_working = 1
        and current_date between ps.ps_date_begin and ps.ps_date_end
    order by ps.ps_from_reg_id
    into
        :office_manager_id,
        :reg_id
    do
    begin
        cashier_id = 0;
        message_body = '';

        -- в теле сообщения будут находится только последниии 50 запросов, так как их может быть большое множество
        -- если сотрудник с ними разберётся, то ему прилетят следующие 50 и так далее пока они не кончатся полностью
        for select
            rm.rm_id,
            rm.rm_date,
            iif(rm.rm_status = 0,'в процессе', 'согласованно с менеджером'),
            rm.rm_no
        from reqmain rm
        where rm.rm_from_reg_id = :reg_id
            and rm.rm_status in (0,5)
            and extract(month from rm.rm_date) = extract(month from current_date)
            and extract(year from rm.rm_date) = extract(year from current_date)
        order by rm.rm_date desc
        into
            :rm_id,
            :rm_date,
            :rm_status,
            :rm_no
        do
            message_body = blob_append
            (
                message_body,
                'Запрос:' || rm_id || ';     Дата:' || rm_date || ';     статус: ' || rm_status || ';     № заявки: ' || rm_no || '<br>'
            );

        if(message_body = '') then
            continue;

        -- если МО в отпуске или на больничном, то уведомление получат все кассиры
        if (exists
            (
                select 1
                from hr_voctravel hrv
                inner join hr_orders hro on hrv.hrv_from_ho_id = hro.ho_id
                inner join sa_worktype sawt on hro.ho_order_type = sawt.sawt_id
                    and sawt.sawt_id <> 20
                    and sawt.sawt_hours_real >= 0
                    and (sawt.sawt_base_type in (4, 5) or sawt.sawt_id = 17) -- отпуск, больничный
                where hrv_from_c_id = :office_manager_id
                    and hrv_status = 1
                    and current_date between hrv.hrv_date_begin and hrv.hrv_date_end
            )) then
        begin
            for select ps.ps_from_c_id
            from hr_pstatus ps
            where ps.ps_from_jl_id in (7, 36, 195) -- кассиры
                and ps.ps_from_reg_id = :reg_id
                and ps.ps_is_working = 1
                and current_date between ps.ps_date_begin and ps.ps_date_end
            into :cashier_id
            do
            begin
                email_address = f_cont_info('Clients', :cashier_id, 'email', 'ok');
                execute procedure hp_send_message('email', :cashier_id, :email_address, 'УВЕДОМЛЕНИЕ О НЕ ИСПОЛНЕННЫХ ЗАПРОСАХ', :message_body , '', 1, 1000000000000);

            end
        end
        else
        begin
            email_address = f_cont_info('Clients', :office_manager_id, 'email', 'ok');
            execute procedure hp_send_message('email', :office_manager_id, :email_address, 'УВЕДОМЛЕНИЕ О НЕ ИСПОЛНЕННЫХ ЗАПРОСАХ', :message_body , '', 1, 1000000000000);

        end
    end
end
