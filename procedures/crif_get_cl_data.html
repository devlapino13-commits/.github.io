CREATE PROCEDURE CRIF_GET_CL_DATA(
    POCKET_ID D_INTEGER)
RETURNS (
    RECORD_TYPE D_STR_10,
    REGION_ID D_STR_100,
    C_ID_STR D_STR_100,
    NAME D_STR_100,
    SURNAME D_STR_100,
    PATRONYMIC D_STR_100,
    C_SEX D_STR_10,
    C_BIRTH_DATE D_DATE,
    C_FAMILY D_STR_10,
    TYPE_ADDRESS D_STR_10,
    ADDRESS D_STR_255,
    IDENTIFICATION_TYPE D_STR_10,
    IDENTIFICATION_NUMBER D_STR_20,
    TYPE_DOCUMENT D_STR_10,
    DOCUMENT_NO D_STR_100,
    DOC_DATE_BEGIN D_DATE,
    DOC_DATE_END D_DATE,
    CONTACT_TYPE D_STR_10,
    CONTACT_NO D_STR_100,
    YUR_FIO D_STR_255,
    LEGAL_FORM D_STR_255,
    C_DATETIME D_DATE,
    REGISTRATION_OFFICE D_STR_255,
    ACTIVITY_KIND D_STR_255,
    CITIZENSHIP D_STR_255)
AS
declare variable rep_date_begin d_date;
declare variable rep_date_end d_date;
declare variable c_id d_id;
declare variable reg_id d_id;
declare variable doc_type d_integer;
declare variable fl_id d_id;
declare variable rd_family d_str_100;
declare variable c_name_fio d_str_255;
declare variable reg_ishenim_region d_integer;
declare variable form_status d_str_255;
declare variable org_type d_str_255;
declare variable c_citizen d_integer;
begin
  select cp.cp_date_begin, cp.cp_date_end
  from cib_pocket cp
  where cp.cp_id = :pocket_id
  into :rep_date_begin, :rep_date_end;

  if (rep_date_end is null) then
    exit;

  for select distinct kc.kc_from_c_id
      from crif_get_kontract_data_tmp(:pocket_id) cg
      inner join knclient kc on kc.kc_from_k_id = cg.k_id and
            kc.kc_amount_approved > 0
      into c_id
  do
  begin
    c_id_str = :c_id;
    identification_type = 3; -- ИНН

    select iif(c.c_type = 0, 'ID', 'BD'), --
           iif(c.c_sex = 0, 'F', 'M'), --
           c.c_birth_date, --
           c.c_pass_no, --
           c.c_pass_date, --
           c.c_pass_date_expire, --
           c.c_from_reg_id, --
           c.c_inn, --
           c.c_doc_type, --
           c.c_from_fl_id, --
           f_get_rd_code(c.c_family), --
           cast(c.c_datetime as date), --
           c.c_name_fio, --
           coalesce(c.c_form_status_long, ''), --
           coalesce(c.c_org_type_long, ''), c.c_citizen
    from clients c
    where c.c_id = :c_id
    into record_type, --
         c_sex, --
         c_birth_date, --
         document_no, --
         doc_date_begin, --
         doc_date_end, --
         reg_id, --
         identification_number, --
         doc_type, --
         fl_id, --
         rd_family, --
         c_datetime, --
         c_name_fio, --
         form_status, --
         org_type, c_citizen;

    if (coalesce(identification_number, '') = '') then
      identification_type = '';
      -- RFCKG-2741
    if (coalesce(doc_date_end, '31.12.9999') >= '31.12.9999') then
      doc_date_end = dateadd(year, 10, :doc_date_begin);
    citizenship = case c_citizen
                    when 8 then 'KG'
                    when 9 then 'UZ'
                    when 10 then 'KZ'
                    when 11 then 'TM'
                    when 12 then 'TJ'
                    when 13 then 'RO'
                    when 52 then 'AM'
                    when 4855 then 'KG'
                    else ''
                  end;

    type_document = case :doc_type
                      when 8 then '6' -- ID-карта
                      when 296 then '6' --  ID-карта
                      when 1 then '6'
                      when 6 then '1' -- Заграничный паспорт
                      when 7 then '3' -- водительское удостоверение
                      when 41 then '4' -- ВНЖ
                      when 2 then '5' -- Воинское удостоверение
                      when 9 then '7' -- Свидетельство о рождении
                      else '6'
                    end;

    if (:record_type = 'BD') then
      yur_fio = :c_name_fio;
    c_family = case rd_family
                 when 'Family.widower' then '4'
                 when 'Family.divorced' then '3'
                 when 'Family.married' then '2'
                 else '1'
               end;

    select fl.fl_name1, --
           fl.fl_name2, --
           fl.fl_name3
    from fiolist fl
    where fl.fl_id = :fl_id
    into surname, --
         name, --
         patronymic;

    select adl.adl_address_str
    from addrlist adl
    inner join addrclient adc on adc.adc_from_adl_id = adl.adl_id and
          adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1
    into address;

    type_address = 'MI';
    if (:record_type = 'BD') then
      type_address = 'MT';
    contact_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');
    contact_type = '1';

    if (contact_no = '') then
      contact_type = '';

    if (coalesce(:reg_id, 0) <> 0) then
    begin
      select reg_ishenim_region
      from region
      where reg_unicode = :reg_id
      into reg_ishenim_region;

      region_id = null;
      select rd.rd_tag -- rd_tag хранится айди регионов из справочника CRIF
      from refdata rd
      where rd.rd_id = :reg_ishenim_region
      into region_id;
      region_id = coalesce(region_id, '999');
      if (:record_type = 'BD') then
        registration_office = region_id;
    end
    legal_form = '';
    activity_kind = '';
    if (:record_type = 'BD') then
    begin
      if (:form_status <> '') then
        legal_form = case :form_status
                       when 'Коммерческие организации' then '10'
                       when 'Частные предприятия' then '11'
                       when 'Предприятия' then '12'
                       when 'Государственные предприятия' then '13'
                       when 'Предприятия общественных объединений' then '14'
                       when 'Предприятия потребительской кооперации' then '15'
                       when 'Махаллинские предприятия' then '16'
                       when 'Акционерные общесва' then '17'
                       when 'Государственно-акционерные общества' then '18'
                       when 'Смешанные товарищества' then '19'
                       when 'Общества с ограниченной ответственностью' then '19'
                       when 'Полные товарищества' then '20'
                       when 'Производственные кооперативы' then '21'
                       when 'Дехканские и фермерские хозяйства' then '22'
                       when 'Унитарное предприятие' then '23'
                       when 'Ассоциации (союзы)' then '24'
                       when 'Концерны' then '25'
                       when 'Объединения предприятий акционерного общества' then '26'
                       when 'Объединения предприятий смешанного товарищества' then '27'
                       when 'Объединения предприятий полного товарищества' then '28'
                       when 'Лизинговые компании' then '29'
                       when 'Корпорации' then '30'
                       when 'Холдинговые компании' then '31'
                       when 'Приватизированные инвестиционные фонды' then '32'
                       when 'Другие коммерческие организации' then '33'
                       when 'Некоммерческие организации' then '34'
                       when 'Государственные учреждения' then '35'
                       when 'Учреждения самоуправлений граждан' then '36'
                       when 'Общественные объединения (организации)' then '37'
                       when 'Учреждения общественных объединений (организаций)' then '38'
                       when 'Объединения обществ, союзов, ассоциаций и фондов' then '39'
                       when 'Союзы потребительских обществ' then '40'
                       when 'Учреждения потребительской кооперации' then '41'
                       when 'Финансово-промышленные группы' then '42'
                       when 'Семейное предприятие' then '44'
                       else '43' -- Другие некоммерческие организации
                     end;
      if (:org_type <> '') then
        activity_kind = case :org_type
                          when 'Сельское хозяйство, охота и лесоводство' then '253'
                          when 'Рыболовство, рыбоводство' then '254'
                          when 'Горнодобывающая промышленность' then '255'
                          when 'Производство топлива и минеральных энергетических ресурсов' then '256'
                          when 'Добыча полезных ископаемых, кроме топлива и энергетических ресурсов' then '257'
                          when 'Промышленность' then '258'
                          when 'Производство продуктов питания' then '259'
                          when 'Текстильная и швейная промышленность' then '260'
                          when 'Производство кожи, изделий из кожи и обуви' then '261'
                          when 'Обработка древесины и производство изделий из дерева' then '262'
                          when 'Целлюлозно-бумажное производство; издательская деятельность' then '263'
                          when 'Производство коксовой продукции, нефтепродуктов и ядерных материалов' then '264'
                          when 'Химическая промышленность' then '265'
                          when 'Производство резиновых и пластмассовых изделий' then '266'
                          when 'Производство прочих неметаллических минеральных продуктов' then '267'
                          when 'Металлургия и производство готовых металлических изделий' then '268'
                          when 'Производство машин и оборудования' then '269'
                          when 'Производство электрического, электронного и оптического оборудования' then '270'
                          when 'Производство транспортных средств и оборудования' then '271'
                          when 'Прочее производство' then '272'
                          when 'Производство и распределение электроэнергии, газа и воды' then '273'
                          when 'Строительство' then '274'
                          when 'Торговля' then '275'
                          when 'Гостиницы и рестораны' then '276'
                          when 'Транспорт и связь' then '277'
                          when 'Финансовая деятельность' then '278'
                          when 'Торговля недвижимостью, лизинг и предоставление услуг потребителям' then '279'
                          when 'Государственное управление' then '280'
                          when 'Образование' then '281'
                          when 'Здравоохранение и социальные услуги' then '282'
                          when 'Деятельность домашних хозяйств' then '284'
                          when 'Деятельность экстерриториальных организаций' then '285'
                          else '283' -- Предоставление коммунальных, социальных и персональных услуг
                        end;
    end
    suspend;
  end
end
