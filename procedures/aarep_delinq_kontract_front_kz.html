CREATE PROCEDURE AAREP_DELINQ_KONTRACT_FRONT_KZ(
    REP_REGION_ID ID_TYPE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_CODE,
    C_NAME_FIO D_STR_100,
    KAL_NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    OKS_BALANCE D_CURRENCY,
    K_PERIOD D_CURRENCY,
    K_DISB_DATE D_DATE,
    K_KONTRACT_DATE D_DATE,
    DELINQ_DAY D_SMALLINT,
    DELINQ_OS D_CURRENCY,
    DELINQ_SH_PERC D_CURRENCY,
    FINE_BALANCE D_CURRENCY,
    TOTAL_BALANCE D_CURRENCY,
    KC_NAME_FIO D_STR_100,
    KF_OFFICER_FIO D_STR_100,
    MAIN_NUM D_STR_100,
    ADDRESS_STR D_STR_255,
    FACT_ADDRESS_STR D_STR_255)
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable kc_id id_type;
declare variable kf_officer id_type;
declare variable zad_os d_currency;
declare variable sh_calc_perc d_currency;
declare variable sh_calc_fine d_currency;
begin
    for select r.reg_name,
               kal.kal_new_kf_name,
               oks.oks_balance,
               k.k_kontract_no,
               k.k_disb_date,
               k.k_kontract_date,
               k.k_period,
               maxvalue(cast(:rep_date_end as date) - oks.oks_delinq_date, 0),
               k.k_id,
               kc.kc_from_c_id,
               k_fact_disb_lo_id,
               kal.kal_new_kf_officer
        from oksummary oks
        inner join kontract k on k.k_id = oks.oks_from_k_id
        inner join knclient kc on kc.kc_from_k_id = k.k_id
        inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
        inner join region r on r.reg_unicode = oks.oks_region_id
        inner join region_list(:rep_region_id, 1, 3, 1) on r.reg_unicode = id_list
        where :rep_date_end between oks.oks_date_begin and oks.oks_date_end
          and :rep_date_end >= oks_delinq_date
          and oks_close_state = 0
          and kal.is_delete = 0
          and :rep_date_end between kal.kal_date and kal.kal_date_end
          and kc.kc_amount_approved > 0
          and r.reg_unicode <> 1010100
       into :reg_name,
            :kal_new_kf_name,
            :oks_balance,
            :k_kontract_no,
            :k_disb_date,
            :k_kontract_date,
            :k_period,
            :delinq_day,
            :k_id,
            :c_id,
            :kc_id,
            :kf_officer
    do
    begin
        select c.c_name_fio,
               f_cont_info('clients', c.c_id, 'cell_main', 'ok')
        from clients c
        where c.c_id = :c_id
        into :c_name_fio,
             :main_num;

        select adl.adl_address_str
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_from_c_id = :c_id
        and adc.adc_address_type = 1
        into :address_str;

        select adl.adl_address_str
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_from_c_id = :c_id
        and adc.adc_address_type = 2
        into :fact_address_str;

        select kc.c_name_fio
        from clients kc
        where kc.c_id = :kc_id
        into kc_name_fio;

        select kc.c_name_fio
        from clients kc
        where kc.c_id = :kf_officer
        into kf_officer_fio;

        delinq_os = 0;
        delinq_sh_perc = 0;
        fine_balance = 0;
        total_balance = 0;

        for select zad_os, sh_calc_perc, sh_calc_fine
            from aarep_make_zad_list_temp(:k_id, :rep_date_end, 1)
            where is_extra = 0 and sh_date <= :rep_date_end
            into zad_os, sh_calc_perc, sh_calc_fine
        do
        begin
            delinq_os = delinq_os + zad_os;
            delinq_sh_perc = delinq_sh_perc + sh_calc_perc;
            fine_balance = fine_balance + sh_calc_fine;
        end
        
        total_balance = delinq_os + delinq_sh_perc + fine_balance;

        suspend;
    end
end
