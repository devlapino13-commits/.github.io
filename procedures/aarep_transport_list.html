CREATE PROCEDURE AAREP_TRANSPORT_LIST(
    OFFICE_ID INTEGER,
    REP_DD DATE)
RETURNS (
    PED_ID INTEGER,
    PED_DATE DATE,
    PED_REGION_PLACE_ID INTEGER,
    PED_REGION_BALANCE_ID INTEGER,
    PED_FROM_C_ID DECIMAL(18,0),
    PED_EXPENSE_SUMMER DECIMAL(18,4),
    PED_EXPENSE_WINTER DECIMAL(18,4),
    PED_EXPENSE_SUMMER_MOUNT DECIMAL(18,4),
    PED_EXPENSE_WINTER_MOUNT DECIMAL(18,4),
    PED_ACC_BALANCE VARCHAR(20),
    PED_ACC_NUMBER INTEGER,
    PED_IS_ACTIVE INTEGER,
    PET_ID INTEGER,
    PET_NAME VARCHAR(255),
    PET_LABEL VARCHAR(50),
    PET_IS_RENT INTEGER,
    PET_NORMA_YEDIZ INTEGER,
    PET_NORMA_KM INTEGER,
    PED_CLIENT_FIO VARCHAR(100),
    PED_PLACE_NAME VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select pet_id, pet_name, pet_label, pet_is_rent, pet_norma_yediz, pet_norma_km
      from pettrans
      into :pet_id, :pet_name, :pet_label, :pet_is_rent, :pet_norma_yediz, :pet_norma_km
  do
  begin
    for select ped_id, ped_date, ped_region_place_id, reg_head_office, ped_from_c_id, ped_expense_summer,
               ped_expense_winter, ped_expense_summer_mount, ped_expense_winter_mount, ped_acc_balance, ped_acc_number,
               ped_is_active, c_name_fio, reg_code || ' ' || reg_name
        from petdata, clients, region
        where ped_from_pet_id = :pet_id and
              ped_date <= :rep_dd and
              c_id = ped_from_c_id and
              reg_unicode = ped_region_place_id
        order by ped_date desc
        rows 1
        into :ped_id, :ped_date, :ped_region_place_id, :ped_region_balance_id, :ped_from_c_id, :ped_expense_summer,
             :ped_expense_winter, :ped_expense_summer_mount, :ped_expense_winter_mount, :ped_acc_balance,
             :ped_acc_number, :ped_is_active, :ped_client_fio, :ped_place_name
    do
    begin
      if ((ped_region_place_id = :office_id) and
          (ped_is_active = 1)) then
        suspend;
    end
  end
end
