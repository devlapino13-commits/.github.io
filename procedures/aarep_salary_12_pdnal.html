CREATE PROCEDURE AAREP_SALARY_12_PDNAL(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_INN D_STR_CODE,
    RES_SALARY D_CURRENCY,
    RES_PDNAL D_CURRENCY,
    RES_SALARY_BRANCH D_CURRENCY,
    RES_PDNAL_BRANCH D_CURRENCY,
    DD_END D_DATE)
AS
declare variable c_id d_id;
declare variable tmp_amount d_currency;
declare variable dd_begin d_date;
declare variable sac_id d_integer;
declare variable is_hq d_integer;
begin
  rep_date_begin = rep_date_begin - extract(day from rep_date_begin) + 1;
  rep_date_end = dateadd(month, 1, rep_date_end - extract(day from rep_date_end) + 1) - 1;

  for select distinct osm_from_c_id, c_name_fio, c_inn
      from opersalmain, clients
      where osm_from_c_id = c_id and
            osm_ol_is_active = 1 and
            osm_ol_date between :rep_date_begin and :rep_date_end
      order by c_id
      into c_id, c_name_fio, c_inn
  do
  begin
    dd_begin = rep_date_begin - extract(day from rep_date_begin) + 1;
    while (dd_begin <= rep_date_end) do
    begin
      dd_end = dateadd(month, 1, dd_begin) - 1;

      res_salary = 0;
      res_pdnal = 0;
      res_salary_branch = 0;
      res_pdnal_branch = 0;
  
      is_hq = 0;
      if (exists(
      select *
      from opersalary, operlist
      where os_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ol_date between :dd_begin and :dd_end and
            os_from_c_id = :c_id and
            os_from_sac_id in (5, 9) and
            os_value <> 0 and
            os_from_reg_id = 1090100)) then is_hq = 1;

      for select osd_from_sac_id, sum(osd_value)
          from opersalmain, opersaldetail
          where osm_id = osd_from_osm_id and
                osm_ol_is_active = 1 and
                osm_ol_date between :dd_begin and :dd_end and
                osm_from_c_id = :c_id and
                osd_from_sac_id in (5, 9)
          group by osd_from_sac_id
          into sac_id, tmp_amount
      do
      begin
        if (tmp_amount is null) then
          tmp_amount = 0;
        if (is_hq = 1) then
        begin
          if (sac_id = 5) then
            res_pdnal = res_pdnal + tmp_amount;
          if (sac_id = 9) then
            res_salary = res_salary + tmp_amount;
        end else
        begin
          if (sac_id = 5) then
            res_pdnal_branch = res_pdnal_branch + tmp_amount;
          if (sac_id = 9) then
            res_salary_branch = res_salary_branch + tmp_amount;
        end
      end
      dd_begin = dateadd(month, 1, dd_begin);
      if (res_pdnal <> 0 or res_salary <> 0 or
          res_salary_branch <> 0 or res_pdnal_branch <> 0) then
        suspend;
    end
  end
end
