CREATE PROCEDURE MC_SEND_ONLINE_LOAN_PUSH
AS
declare variable C_ID D_ID;
declare variable MCL_ID D_INTEGER;
declare variable SUBJECT D_STR_255;
declare variable MSG_BODY D_STR_2048;
declare variable CLIENT_POINTS D_INTEGER;
declare variable CLIENT_LAST_KIB_DATE_CHECK D_DATETIME;
declare variable OFFICE_ID D_INTEGER;
declare variable EMPOLOYEE_ID D_ID;
begin   
    if(current_date not in ('25.07.2024', '28.07.2024')) then
        exit;

    subject =   'Акция, лучшим клиентам!!!';
    msg_body =  'Зайди в приложении в "Онлайн кредиты" и получи до 50.000сом прямо сейчас на карту МБулак Элкарт';

    for select
        c_id,
        mcl_id
    from mc_send_online_loan_push_tmp
    into
        c_id,
        mcl_id
    do
    begin
        execute procedure mc_send_push('mc_login', :mcl_id, :subject, :msg_body, 1);
        
        -- RFCKG-2721
        select
            k.k_fact_disb_lo_id,
            k.k_fact_disb_reg_id
        from knclient kc
        inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id
            and ks.ks_ol_is_delete = 0
            and ks.ks_state = 1
        inner join kontract k on kc.kc_from_k_id = k.k_id
        where kc.kc_from_c_id = :c_id
            and kc.kc_amount_approved > 0
        order by ks.ks_date_begin desc
        rows 1
        into
            empoloyee_id,
            office_id;

        client_points = f_point(:c_id, current_date, 0);
        client_last_kib_date_check = (select kbm_datetime from hp_kibminiapp_last(:c_id, null));

        insert into online_loan_push_log (olp_date, olp_datetime, olp_c_id, olp_point_qty, olp_cib_last_check, olp_cli_office_id, olp_from_lo_id)
        values (current_date,  current_timestamp, :c_id, :client_points, :client_last_kib_date_check, :office_id, :empoloyee_id);
    end

    -- проверка о том, что процедура отработала
    execute procedure hp_send_message('email', 1090101807439, 'd.mihailechko@mbulak.kg', 'КМП dmmhlchk', 'Пуш уведомления были отправлены', '', 1, 1000000000000);
end
