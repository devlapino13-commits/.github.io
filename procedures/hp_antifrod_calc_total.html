CREATE PROCEDURE HP_ANTIFROD_CALC_TOTAL(
    JSO D_INTEGER)
AS
declare variable risk_level d_smallint;
declare variable total_percent d_percent;
declare variable afc_id d_id;
declare variable percent d_percent;
declare variable max_percent d_percent;
declare variable min_percent d_percent;
declare variable jso_cat d_integer;
begin
  -- пересчет результата проверки по антифроду.
  total_percent = 0;

  for select afc_id, afc_max_percent, afc_min_percent
  from antifrod_categories
  into afc_id, max_percent, min_percent
  do
  begin
    jso_cat = json.obj_get_obj(jso, afc_id);

    select sum(cast(json.obj_get_val(:jso_cat, acr_id || '.' || 'criteria_res', acr_percent) as d_percent))
    from antifrod_criteria
    where acr_from_afc_id = :afc_id
    into percent;

    if (json.obj_get_val(jso_cat, 'is_critical', 0) = 1) then
      max_percent = 100;

    percent = maxvalue(percent, min_percent);
    percent = minvalue(percent, max_percent);
    total_percent = total_percent + percent;

    json.obj_set_s(jso_cat, 'category_res', percent);
  end

  risk_level = case
                 when total_percent < 50 then 0
                 when total_percent <= 80 then 1
                 else 2
               end;

  json.obj_set_s(jso, 'risk_level', risk_level);
end
