CREATE PROCEDURE HP_SEND_COURT_NOTIFICATION(
    K_ID D_ID,
    C_ID D_ID)
AS
declare variable phone_no d_str_code;
declare variable msg_kind d_str_code;
declare variable cm_name d_str_code;
declare variable push_title d_str_code;
declare variable cl_name d_str_100;
declare variable msg_body d_str_255;
declare variable lifetime_days d_smallint;
declare variable user_id d_id;
declare variable k_kontract_no d_str_code;
declare variable kc_id d_id;
declare variable mcl_project d_integer;
declare variable msg_sender_code d_str_code;
begin
  -- Отправка уведомлений о пердаче дела в суд

  user_id = 1000000000000; -- finstart_robot
  msg_kind = f_get_msg_kind(c_id, 'hp_send_court_notification', '', null, 365);
  lifetime_days = 1095;

  if (msg_kind in ('push', 'sms')) then
  begin
    select cm_name
    from company
    into cm_name;

    select k_kontract_no, kc_id
    from kontract, knclient
    where k_id = :k_id
      and kc_from_k_id = k_id
      and kc_from_c_id = :c_id
    into k_kontract_no, kc_id;

    select txt2
    from clients, hp_fio_clear(c_name_fio)
    where c_id = :c_id
    into cl_name;

    cl_name = coalesce(cl_name, '');

    if (cl_name = '') then
      cl_name = 'Здравствуйте';

    msg_body = cl_name ||
      ', у Вас имеется задолженность по договору ' || k_kontract_no ||
      '. В связи с неоплатой долга Ваше дело передано в Департамент розыска и принудительного взыскания T.+79067365443' || cm_name;

    if (msg_kind = 'sms' and f_clientcom_allowed(c_id, 0, 0, 1) = 1) then
    begin
      select cont_value
      from hp_contdata_get_cell_main(0, :c_id)
      into phone_no;

      if (phone_no starting '7') then
      begin
        mcl_project = f_get_mcl_project(f_objkind('kontract', 'id'), :k_id);
        msg_sender_code = f_enum_code('sms_sender', iif (:mcl_project = 2, 1, 0));

        f_send_sms(phone_no, msg_body, 'knclient', kc_id, 'hp_send_court_notification', lifetime_days, user_id, 0, :msg_sender_code);
      end

    end
    if (msg_kind = 'push' and f_clientcom_allowed(c_id, 0, 0, 1) = 1) then
    begin
      push_title = 'Департамент розыска и принудительного взыскания';
      f_send_push('knclient', kc_id, push_title, msg_body, 'hp_send_court_notification', lifetime_days, user_id);
    end
  end
end
