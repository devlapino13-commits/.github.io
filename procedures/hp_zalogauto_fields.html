CREATE PROCEDURE HP_ZALOGAUTO_FIELDS(
    ZAP_ID D_INTEGER)
RETURNS (
    Z_NAME D_STR_100,
    Z_DOCUMENT D_STR_255,
    Z_REGISTRY_NO D_STR_CODE,
    Z_REGISTRY_DATE D_DATE,
    Z_REGISTRY_AGENCY D_STR_100,
    Z_MEMO1 D_STR_LONG,
    Z_MEMO2 D_STR_4096)
AS
declare variable za_orig_sts_id d_id;
declare variable za_orig_pts_id d_id;
declare variable tmp_str d_str_long;
declare variable drl_sts_id d_id;
declare variable drl_pts_id d_id;
begin
  select za_orig_sts_id, za_orig_pts_id
  from zalogautoprof, zalogauto
  where zap_id = :zap_id and
        zap_from_za_id = za_id
  into za_orig_sts_id, za_orig_pts_id;

  select cds_from_drl_id
  from clientdocs
  where cds_id = :za_orig_sts_id
  into drl_sts_id;

  select cds_from_drl_id
  from clientdocs
  where cds_id = :za_orig_pts_id
  into drl_pts_id;

  -- zalog
  z_name = '';
  z_document = '';
  z_registry_no = '';
  z_registry_date = null;
  z_registry_agency = '';
  z_memo1 = '';
  z_memo2 = '';

  z_name = f_docrecog_value(drl_pts_id, 'model', '');
  tmp_str = f_docrecog_value(drl_sts_id, 'doc_no', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'СТС ' || tmp_str;
    z_document = tmp_str;
  end
  tmp_str = f_docrecog_value(drl_pts_id, 'doc_no', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'ПТС ' || tmp_str;
    if (z_document <> '') then
      z_document = z_document || ', ';
    z_document = z_document || tmp_str;
  end

  if (coalesce(drl_sts_id, 0) > 0) then
  begin
  z_registry_no = f_docrecog_value(drl_sts_id, 'plate', '');
  z_registry_date = f_docrecog_value(drl_sts_id, 'date_begin', '');
  z_registry_agency = f_docrecog_value(drl_sts_id, 'authority', '');
  end
  if (z_registry_agency <> '') then
  begin
    if (position('ГИБДД', z_registry_agency) = 0) then
      z_registry_agency = 'ГИБДД ' || z_registry_agency;
  end

  tmp_str = f_docrecog_value(drl_pts_id, 'type_name', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'тип ' || tmp_str;
    if (z_memo1 <> '') then
      z_memo1 = z_memo1 || ', ';

    z_memo1 = z_memo1 || tmp_str;
  end

  tmp_str = f_docrecog_value(drl_pts_id, 'category', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'кат ' || tmp_str;
    if (z_memo1 <> '') then
      z_memo1 = z_memo1 || ', ';

    z_memo1 = z_memo1 || tmp_str;
  end

  tmp_str = f_docrecog_value(drl_pts_id, 'issue_year', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'год ' || tmp_str;
    if (z_memo1 <> '') then
      z_memo1 = z_memo1 || ', ';

    z_memo1 = z_memo1 || tmp_str;
  end

  tmp_str = f_docrecog_value(drl_pts_id, 'vin', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'VIN ' || tmp_str;
    if (z_memo1 <> '') then
      z_memo1 = z_memo1 || ', ';

    z_memo1 = z_memo1 || tmp_str;
  end

  tmp_str = f_docrecog_value(drl_pts_id, 'body_color', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'цвет ' || tmp_str;
    if (z_memo1 <> '') then
      z_memo1 = z_memo1 || ', ';

    z_memo1 = z_memo1 || tmp_str;
  end

  tmp_str = z_registry_no;
  if (tmp_str <> '') then
  begin
    tmp_str = 'рег.знак ' || tmp_str;
    if (z_memo2 <> '') then
      z_memo2 = z_memo2 || ', ';

    z_memo2 = z_memo2 || tmp_str;
  end

  tmp_str = z_name;
  if (tmp_str <> '') then
  begin
    tmp_str = 'марка ' || tmp_str;
    if (z_memo2 <> '') then
      z_memo2 = z_memo2 || ', ';

    z_memo2 = z_memo2 || tmp_str;
  end

  tmp_str = z_memo1;
  if (tmp_str <> '') then
  begin
    if (z_memo2 <> '') then
      z_memo2 = z_memo2 || ', ';

    z_memo2 = z_memo2 || tmp_str;
  end

  tmp_str = f_docrecog_value(drl_pts_id, 'chassis_no', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'шасси ' || tmp_str;
    if (z_memo2 <> '') then
      z_memo2 = z_memo2 || ', ';
  
    z_memo2 = z_memo2 || tmp_str;
  end

  tmp_str = f_docrecog_value(drl_pts_id, 'body_no', '');
  if (tmp_str <> '') then
  begin
    tmp_str = 'кузов ' || tmp_str;
    if (z_memo2 <> '') then
      z_memo2 = z_memo2 || ', ';
  
    z_memo2 = z_memo2 || tmp_str;
  end

  suspend;
end
