CREATE PROCEDURE MC_ONLINE_LOAN_CHECK(
    MCL_ID D_ID)
RETURNS (
    DO_PASS_CHECK D_BOOL,
    DO_KIB_CHECK D_BOOL,
    ERROR_HEADER D_STR_255,
    ERROR_TEXT D_STR_255)
AS
declare variable C_ID D_ID;
declare variable BLS_ID D_INTEGER;
declare variable VOIDSTATUS D_SMALLINT;
declare variable PIN D_STR_100;
declare variable PASSPORT_SERIES D_STR_100;
declare variable PASSPORT_NUMBER D_STR_100;
declare variable PASS_NO D_STR_255;
declare variable JSON_TEXT D_STR_8100;
declare variable KBM_ID D_ID;
begin
  do_pass_check = 1;
  do_kib_check = 1;
  error_header = '';
  error_text = '';

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id;

  if (exists(select *
             from clients c
             where c.c_id = :c_id and
                   c.c_verify_status = 2 and
                   cast(c.c_verify_datetime as date) = current_date)) then
    do_pass_check = 0;

  select eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
        eic.eic_code = 'Tunduk'
  where eil.eil_from_c_id = :c_id and
        eil.eil_status = 2 and
        eil.eil_from_bls_id > 0
  order by eil.eil_id desc
  rows 1
  into bls_id;

  if (coalesce(bls_id, 0) > 0) then
  begin
    pass_no = '';
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into json_text;

    voidstatus = f_json_str_get_value(:json_text, 'voidStatus', 1); -- Актуальность паспорта (не просрочен ли)
    pin = f_json_str_get_value(:json_text, 'pin', ''); -- Персональный идентификационный номер
    passport_series = f_json_str_get_value(:json_text, 'passportSeries', ''); -- Серия паспорта, пример "ID"
    passport_number = f_json_str_get_value(:json_text, 'passportNumber', ''); --  Номер паспорта, пример "1501045"

    if (passport_series <> '' and
        passport_number <> '') then
      pass_no = passport_series || passport_number;
    if (voidstatus = 1) then
    begin
      error_header = 'Паспорт недействителен';
      error_text = 'Ваш паспорт недействителен, обновите паспортные данные.';
    end
    else
    if (voidstatus = 0 and
        :pin <> '' and
        pass_no <> '') then
    begin
      if (:c_id = (select c.c_id
                   from clients c
                   where c.c_inn = :pin and
                         c.c_pass_no = :pass_no)) then
        do_pass_check = 0;
    end
  end

  kbm_id = (select kbm_id from mc_online_loan_get_kib_id(:c_id));

  if (kbm_id is not null) then
  begin
    do_kib_check = 0;

    if (exists(select *
               from kibhistory kh
               where kh.kh_from_kbm_id = :kbm_id and
                     kh.kh_is_our = 0 and
                     kh.kh_is_debtor = 1 and
                     kh.kh_active_delinq_days_count > 0 and
                     kh.kh_creditstatus not starting 'Отказано')) then
    begin
      error_header = 'Имеются просрочки';
      error_text = 'В других ФКУ имеются просрочки. Выдача невозможна.';
    end
  end
  suspend;
end
