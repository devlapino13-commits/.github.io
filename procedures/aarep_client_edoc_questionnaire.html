CREATE PROCEDURE AAREP_CLIENT_EDOC_QUESTIONNAIRE(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME STR_100,
    C_DTBIRTH DATE,
    C_NAME_ENG STR_100,
    COMPANY_NAME STR_100,
    AN01_ID INTEGER,
    AN02_ID INTEGER,
    AN03_ID INTEGER,
    AN04_ID INTEGER,
    AN05_ID INTEGER,
    AN06_ID INTEGER,
    AN07_ID INTEGER,
    AN08_ID INTEGER,
    AN09_ID INTEGER,
    AN10_ID INTEGER,
    AN11_ID INTEGER,
    AN12_ID INTEGER,
    AN13_ID INTEGER,
    AN01_OTHER STR_255,
    AN02_OTHER STR_255,
    AN03_OTHER STR_255,
    AN04_OTHER STR_255,
    AN05_OTHER STR_255,
    AN06_OTHER STR_255,
    AN07_OTHER STR_255,
    AN08_OTHER STR_255,
    AN09_OTHER STR_255,
    AN10_OTHER STR_255,
    AN11_OTHER STR_255,
    AN12_OTHER STR_255,
    AN13_OTHER STR_255,
    AN_DATE DATE,
    QR_CODE D_STR_CODE)
AS
declare variable c_id d_id;
declare variable kc_id d_id;
begin
  c_id = f_json_str_get_value(add_params, 'c_id', null);
  if (c_id is null) then
    exit;

  select kc_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_from_c_id = :c_id and
        kc_amount_approved > 0
  into kc_id;

  if (kc_id is null) then
    exit;

  select c_name, c_dtbirth, c_name_eng, company_name, an01_id, an02_id, an03_id, an04_id, an05_id, an06_id, an07_id, an08_id, an09_id,
         an10_id, an11_id, an12_id, an13_id, an01_other, an02_other, an03_other, an04_other, an05_other, an06_other,
         an07_other, an08_other, an09_other, an10_other, an11_other, an12_other, an13_other, an_date, qr_code
  from aarep_client_questionnaire(:kc_id, :add_params)
  into c_name, c_dtbirth, c_name_eng, company_name, an01_id, an02_id, an03_id, an04_id, an05_id, an06_id, an07_id, an08_id, an09_id,
       an10_id, an11_id, an12_id, an13_id, an01_other, an02_other, an03_other, an04_other, an05_other, an06_other,
       an07_other, an08_other, an09_other, an10_other, an11_other, an12_other, an13_other, an_date, qr_code;

  suspend;
end
