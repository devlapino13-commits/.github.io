CREATE PROCEDURE AAREP_CARD_BALANCES_EMPLOYEE(
    C_ID D_ID NOT NULL,
    REG_ID D_INTEGER NOT NULL,
    ON_DATE D_DATE = null)
RETURNS (
    NUM_CARD D_INTEGER,
    NUM_PINC D_INTEGER)
AS
declare variable num d_integer;
declare variable is_card d_bool;
begin
  on_date = coalesce(on_date, current_date);
  if (on_date = current_date) then
  begin
    num_card = coalesce((
      select count(*)
      from cardlist
        inner join cardplace
          on cpl_id = cal_last_cpl_id
            and cpl_from_reg_id = :reg_id
            and cpl_from_c_id = :c_id
            and cpl_is_transfer = 0
      where cal_status = 1), 0);
    num_pinc = coalesce((
      select count(*)
      from cardlist
        inner join cardplace
          on cpl_id = cal_last_pin_cpl_id
            and cpl_from_reg_id = :reg_id
            and cpl_from_c_id = :c_id
            and cpl_is_transfer = 0
      where cal_status = 1), 0);
  end else
  begin
    for
      with Tmp as (
        select
          cem_is_pincover as is_pincover,
          max(ol_datetime) as ol_datetime
        from cardaktmain
          inner join cardplace
            on cpl_id = cam_dst_cpl_id
              and cpl_is_transfer = 0
          inner join operlist
            on ol_id = cam_from_ol_id
              and ol_date <= :on_date
              and is_delete = 0
          inner join cardaktrange
            on car_from_cam_id = cam_id
          inner join cardlist
            on cal_card_id between car_first_card_id and car_last_card_id
              and coalesce(cal_date_issue, :on_date) >= :on_date
          inner join cardembomain
            on cem_id = cal_from_cem_id
        group by cal_card_id, cem_is_pincover)

      select
        coalesce(count(*), 0),
        1
      from cardaktmain
        inner join cardplace
          on cpl_id = cam_dst_cpl_id
            and cpl_from_reg_id = :reg_id
            and cpl_from_c_id = :c_id
            and cpl_is_transfer = 0
        inner join tmp
          on ol_datetime = cam_ol_datetime
      union
      select
        coalesce(sum(is_pincover), 0),
        0
      from cardaktmain
        inner join cardplace
          on cpl_id = cam_dst_pin_cpl_id
            and cpl_from_reg_id = :reg_id
            and cpl_from_c_id = :c_id
            and cpl_is_transfer = 0
        inner join tmp
          on ol_datetime = cam_ol_datetime
    into
      :num,
      :is_card
    do
    begin
      if (is_card = 1) then num_card = num;
        else num_pinc = num;
    end
  end
  suspend;
end
