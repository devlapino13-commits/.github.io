CREATE PROCEDURE HP_CHECK_OIVO_ASSIGNMENT(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    IS_OIVO D_SMALLINT,
    RES_ID D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable CMA_USER_ID D_ID;
begin
    is_oivo = 0;
    res_id = 0;
    res_text = '';

    if (exists (
        select 1
        from mc_login mcl
        inner join docrecoglist drl on drl.drl_from_mcl_id = mcl.mcl_id and drl.drl_is_main_doc = 1
        inner join clientdocs cds on cds.cds_from_drl_id = drl.drl_id and
              cds.cds_is_delete = 0 and cds.cds_is_main_doc = 1
        where mcl.mcl_from_c_id = :c_id and
              mcl.mcl_is_delete = 0 and
              mcl.mcl_project = f_enum_id('mcl_project', 'oivo')
        )) then 
    begin
        is_oivo = 1;

        select coalesce(cma.cma_from_un_id, 0)
        from client_manager_assignment cma
        where cma.cma_from_c_id = :c_id and
              cma.cma_is_active = 1 and
              cast(cma.cma_unassigned_at as date) > current_date and
              cma.cma_project = 'oivo'
        into :cma_user_id;

        if (row_count > 0) then
        begin
            if (cma_user_id = :un_id) then
            begin
                res_id = 1;
                res_text = 'Клиент закреплен за вами';
            end
            else
            begin
                res_id = 2;
                res_text = 'Клиент закреплен за другим сотрудником';
            end
        end
        else
        begin
            res_id = 3;
            res_text = 'Клиент ни за кем не закреплен';
        end
    end

    suspend;
end
