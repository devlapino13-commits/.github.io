CREATE PROCEDURE AAREP_KIB_BASIC_DATA(
    OBJECT_ID ID_TYPE,
    OBJECT_TYPE VARCHAR(10))
RETURNS (
    C_ID ID_TYPE,
    CLIENT_DATA BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable unpaid_own_amount curr_type;
declare variable active_kont smallint;
declare variable negative_status smallint;
declare variable max_delinq_days integer;
declare variable poruch_kont_qty integer;
declare variable kbm_id id_type;
declare variable kh_count integer;
declare variable kh_debtor_count integer;
declare variable kh_repaystatus str_100;
declare variable kh_creditstatus str_100;
declare variable kh_deliqdayscount integer;
declare variable kh_dateaward date;
declare variable own_amount curr_type;
declare variable group_unpaid_amount curr_type;
declare variable group_amount curr_type;
declare variable cli_percent curr_type;
declare variable cu_symbol varchar(3);
declare variable cr_rate curr_type;
declare variable cu_id varchar(3);
declare variable kh_clients_kol integer;
declare variable rep_date date;
declare variable kh_is_debtor smallint;
declare variable max_delinq_day integer;
declare variable c_counter integer;
declare variable kbm_ver_info smallint;
declare variable kbm_checked_date date;
declare variable kbm_checked_date_str varchar(10);
declare variable c_name_fio varchar(100);
declare variable ma_id d_integer;
begin
  rep_date = current_date;
  object_type = trim(coalesce(object_type, ''));

  if ((object_type is null) or (object_id is null)) then
    exit;
  client_data = '';
  c_counter = 0;

  ma_id = 0;
  if (object_type = 'kontract') then
    select k_from_ma_app_id
    from kontract
    where k_id = :object_id
    into ma_id;
  else
  if (object_type = 'miniapp') then
    ma_id = object_id;

      for select cl.c_id, cl.c_name_fio
      from clients cl
      where cl.c_id = cast(:object_id as id_type) and
            cast(:object_type as varchar(10)) = 'client'
      union all
      select macl.mac_from_c_id, ''
      from miniappclient macl
      where macl.mac_from_ma_id = cast(:object_id as id_type) and
            --macl.mac_amount > 0 and
            macl.mac_from_reg_id = 0 and
            cast(:object_type as varchar(10)) = 'miniapp'
      union all
      select kncl.kc_from_c_id, ''
      from knclient kncl
      where kncl.kc_from_k_id = cast(:object_id as id_type) and
            --kncl.kc_amount_approved > 0 and
            cast(:object_type as varchar(10)) = 'kontract'
      union all
      select macl.mac_from_c_id, ''
      from miniappclient macl
      inner join kontract kont on kont.k_from_ma_app_id = macl.mac_from_ma_id
      where kont.k_id = cast(:object_id as id_type) and
            macl.mac_from_reg_id = 0 and
            cast(:object_type as varchar(10)) = 'kontract'
            and macl.mac_is_poruch = -1
      into :c_id, :c_name_fio
  do
  begin
    if (c_name_fio = '') then
    begin
      select c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :c_name_fio;
    end

    if (c_counter <> 0) then
    begin
      client_data = BLOB_APPEND(client_data, '<hr width="100%"  noshade color="#0b7fab">');
    end
    c_counter = 1;
    kbm_checked_date_str = '';
    kbm_checked_date = null;

    client_data = BLOB_APPEND(client_data, '<b>' || c_name_fio || '</b><br>');

    select kbm_id, kbm_ver_info, cast(kbm_checked_datetime as date)
    from hp_kibminiapp_last(:c_id, :ma_id)
    into kbm_id, kbm_ver_info, kbm_checked_date;

    kh_count = 0;
    unpaid_own_amount = 0;
    active_kont = 0;
    max_delinq_days = 0;
    poruch_kont_qty = 0;
    negative_status = 0;
    kh_debtor_count = 0;

    if (kbm_checked_date is not null) then
    begin
      kbm_checked_date_str = replace(iif(extract(day from kbm_checked_date) < 10, '0', '') || extract(day from kbm_checked_date) || '.' || iif(extract(month from kbm_checked_date) < 10, '0', '') || extract(month from kbm_checked_date) || '.' || extract(year from kbm_checked_date), ' ', '');
    end

    if (kbm_ver_info in (0, 2)) then
    begin
      client_data = BLOB_APPEND(client_data, '&nbsp&nbspКлиент еще не проверен<br>');
    end
    else
    if (kbm_ver_info = 7) then
    begin
      client_data = BLOB_APPEND(client_data, '&nbsp&nbspВо время проверки у КИБ возникла ошибка<br>');
    end
    else
    begin
      for select trim(coalesce(kh.kh_repaystatus, '-- не выбран --')), trim(coalesce(kh.kh_creditstatus, '')),
                 coalesce(kh.kh_ownamount, 0), coalesce(kh.kh_deliqdayscount, 0),
                 iif(coalesce(kh_is_debtor, 0) = 1, coalesce(kh.kh_unpaidamount, 0), 0), coalesce(kh.kh_amount, 0),
                 kh.kh_dateaward,
                 (
                 case kh_currency
                   when 'Сом (Кыргызский)' then 'KGS'
                   when 'Доллар США' then 'USD'
                   when 'Евро' then 'EUR'
                   when 'Российский рубл' then 'RUB'
                   else 'KGS'
                 end),
                 coalesce(kh_clients_kol, 0), coalesce(kh_is_debtor, 0)
          from kibhistory kh
          where kh.kh_from_kbm_id = :kbm_id and
                kh_creditstatus not containing ('Отказано') -- если отказано, это значит клиент кредит не получил
                and
                kh.kh_creditstatus not containing ('Рассмотрение заявки') -- кредитный договор еще не оформлен
                and
                coalesce(kh.kh_is_our, 0) = 0 -- только в других  ФКУ
          order by kh.kh_dateaward desc
          into :kh_repaystatus, :kh_creditstatus, :own_amount, :kh_deliqdayscount, :group_unpaid_amount, :group_amount,
               :kh_dateaward, :cu_symbol, :kh_clients_kol, :kh_is_debtor
      do
      begin
        cr_rate = null;

        if (kh_count = 0) then
        begin
          client_data = BLOB_APPEND(client_data, '&nbsp&nbspДата проверки: <b>' || kbm_checked_date_str || '</b><br>');
        end

        kh_count = kh_count + 1;
        if (kh_is_debtor = 1) then
        begin
          kh_debtor_count = kh_debtor_count + 1;
          kh_deliqdayscount = coalesce(kh_deliqdayscount, 0);

          if ((kh_clients_kol = 1) and
              (own_amount = 0)) then
            own_amount = group_amount;

          if (kh_debtor_count <= 3) then -- проверяю статус только для посследних займов
          begin
            if (kh_deliqdayscount > max_delinq_days) then
              max_delinq_days = kh_deliqdayscount;
            if (kh_repaystatus in ('Погашен гарантом', 'Погашен за счет обеспечения в судебном порядке', 'Погашен за счет обеспечения во внесудебном порядке', 'Погашен за счет обеспечения на добровольной основе', 'Погашен за счет обращ.взыск.на обесп.через суд.орг', 'Погашен за счет обращ.вызск.на обесп. во внесуд.по', 'Погашен за счет права уступки требования', 'Погашен через судебные органы (кредиты без залога', 'Списан через РППУ',

                               '120 дней просрочки и более',
                                   '60 дней до 89 дней просрочки', '60 дней просрочки и более', '90 дней  до 119 дней просрочки', '90 дней просрочки и более', '30 дней до 59 дней просрочки', '30 дней просрочки и более')) then
            begin
              if (negative_status = 0) then
              begin
                client_data = BLOB_APPEND(client_data, '&nbsp&nbspНарушения в трех последних займах в других ФКУ<br>');
                negative_status = 1;
              end
              client_data = BLOB_APPEND(client_data, '&nbsp&nbsp&nbsp&nbsp<b>' || kh_repaystatus || '; кредит от ' || cast(kh_dateaward as str_100) || '</b><br>');
            end
            else
            if (kh_creditstatus in ('Судебное разбирательство', 'Open_Under_Legal_Proceedings', 'Settled_Late', 'Sold_To_Third_Party', 'Written_Off', 'Fully_Paid_Collateral_Execution', 'Fully_Paid_Resolved_Through_Court', 'Fully_Paid_By_Guarantor', 'Fully_Paid_Collateral_Execution_Through_Court', 'Written_Off_Through_Loan_Loss_Provision')) then
            begin
              if (negative_status = 0) then
              begin
                client_data = BLOB_APPEND(client_data, '&nbsp&nbspНарушения в трех последних займах в других ФКУ<br>');
                negative_status = 1;
              end
              client_data = BLOB_APPEND(client_data, '&nbsp&nbsp&nbsp&nbsp<b>' || kh_creditstatus || '; кредит от ' || cast(kh_dateaward as str_100) || '</b><br>');
            end
          end
        end

        -- рассматриваю все активные кредиты по киб
        if (kh_creditstatus in ('Выдан', 'Просрочен', 'Просрочен  - <не выбрано>', 'Просрочен  -> Просрочки', 'Просрочен -> Просрочки', 'Реструктуризирован', 'Своевременный платеж', 'Начисление процентов',

                            'Granted_And_Activated', 'Restructured',
                                'Open_Under_Legal_Proceedings')) then
        begin
          if (kh_is_debtor = 1) then
          begin
            active_kont = active_kont + 1;
            if ((own_amount > 0) and
                (group_unpaid_amount > 0) and
                (group_amount > 0)) then
            begin
              select cl.cu_id
              from currlist cl
              where cl.cu_symbol = :cu_symbol
              into :cu_id;

              select opc.cr_rate
              from opercurr opc
              where opc.cr_from_cu_id = :cu_id and
                    :rep_date between opc.cr_date and opc.cr_date_end
              into :cr_rate;

              cr_rate = coalesce(cr_rate, 1);

              own_amount = own_amount * cr_rate;
              group_unpaid_amount = group_unpaid_amount * cr_rate;
              group_amount = group_amount * cr_rate;

              cli_percent = (cast(own_amount as double precision) * 100) / group_amount; -- доля клиента в группе
              unpaid_own_amount = unpaid_own_amount + (group_unpaid_amount * cli_percent / 100);
            end
          end
          else
          begin
            poruch_kont_qty = poruch_kont_qty + 1;
          end
        end
      end

      if (kh_count = 0) then
      begin
        if (not exists(select * from knclient, kontstate -- добавлено по письму Алмаза, если имеется кр. история в своем ФКУ то не выводит сообщение об отстутствии кр. истории
        where kc_from_c_id = :c_id
        and kc_amount_approved > 0
        and kc_from_k_id = ks_from_k_id
        and ks_ol_is_delete = 0
        and ks_state = 1)) then
        begin
          client_data = BLOB_APPEND(client_data, '&nbsp&nbspКредитная история отсутствует<br>');
        end

        client_data = BLOB_APPEND(client_data, '&nbsp&nbspДата проверки: <b>' || kbm_checked_date_str || '</b><br>');
      end
      else
      begin
        if (negative_status = 0) then
          client_data = BLOB_APPEND(client_data, '&nbsp&nbsp<b>В других ФКУ нарушений не обнаружено</b><br>');
        client_data = BLOB_APPEND(client_data, '&nbsp&nbspМакс. дни просрочки в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(max_delinq_days as varchar(10)) || ' дней</b><br>');

        client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивный баланс в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(round(unpaid_own_amount) as varchar(10)) || ' сом</b><br>');

        client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивные займы в других ФКУ<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(active_kont as varchar(10)) || ' кредит(ов)</b><br>');

        client_data = BLOB_APPEND(client_data, '&nbsp&nbspАктивные в других ФКУ, где он(а) поручитель<br>' || '&nbsp&nbsp&nbsp&nbsp<b>' || cast(poruch_kont_qty as varchar(10)) || ' кредит(ов)</b><br>');
      end
      if (ma_id <> 0) then
        client_data = BLOB_APPEND(client_data, f_uri_html('Просмотр отчета', 'kib_report',
          'Отчет КИБ', 'miniapp', :ma_id, current_date, current_date,
          '{"c_id":' || c_id || '}') || '<br>');
    end
  end
  suspend;
end
