CREATE PROCEDURE HP_CLIENT_RELATED_LOANS(
    C_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    UNIQ_ID D_STR_20,
    CLIENT_ID D_ID,
    CLIENT_FIO D_STR_100,
    LN_ID D_INTEGER,
    LN_DATE_BEGIN D_DATE,
    LN_DATE_END D_DATE,
    LN_COUNT D_INTEGER,
    CLIENT_POINT D_INTEGER,
    LN_ID_LIST D_STR_LONG,
    LN_AGREEMENT_LIST D_STR_LONG)
AS
declare variable ks_date_begin d_date;
declare variable ks_date_end d_date;
declare variable kc_id_list d_str_long;
declare variable last_date d_date;
declare variable kc_list d_str_long;
declare variable tmp_count d_integer;
declare variable kc_id d_id;
declare variable kont_list d_str_long;
declare variable k_kontract_no d_str_20;
begin
  ln_id_list = '';
  kc_id_list = '';
  ln_id = 0;
  ln_count = 0;
  client_fio = '';
  client_id = c_id;
  ln_agreement_list = '';

  select c_name_fio
  from clients
  where c_id = :c_id
  into client_fio;

  if (row_count = 0) then
    exit;

  for select ks_date_begin, iif(ks_date_end > '31.12.2999', ks_date_end, ks_date_end + 1), kc_id, k_kontract_no
  from knclient, kontstate, kontract
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = ks_from_k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_from_k_id = k_id and
        ks_date_begin between :rep_date_begin and :rep_date_end
  order by ks_date_begin
  into ks_date_begin, ks_date_end, kc_id, k_kontract_no
  do
  begin
    if (kc_id_list containing '[' || kc_id || ']') then
      Continue;

    if (ln_date_begin is null) then
      ln_date_begin = ks_date_begin;

    if (ln_date_end is null) then
      ln_date_end = ks_date_end;

    if (kc_id_list is null) then
      kc_id_list = '';

    kc_id_list = kc_id_list || '[' || kc_id || ']';

    if (ln_id_list <> '') then
      ln_id_list = ln_id_list || ', ';

    ln_id_list = ln_id_list || kc_id;

    if (ln_agreement_list <> '') then
      ln_agreement_list = ln_agreement_list || ', ';

    ln_agreement_list = ln_agreement_list || k_kontract_no;
    ln_count = ln_count + 1;

    last_date = null;
    tmp_count = 0;
    kc_list = '';
    kont_list = '';
    select max(ks_date_end), list(kc_id), count(*), list(k_kontract_no, ', ')
    from knclient, kontstate, kontract
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = ks_from_k_id and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          ks_date_begin between :ln_date_begin and :ln_date_end and
          ks_from_k_id = k_id and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          not(:kc_id_list containing '[' || kc_id || ']')
    into last_date, kc_list, tmp_count, kont_list;

    if (last_date is null) then
    begin
      ln_id = ln_id + 1;
      uniq_id = c_id || '_' || ln_id;
      client_point = f_point(:c_id, coalesce(ln_date_end, current_date), 0);

      suspend;
      ln_date_begin = null;
      ln_date_end = null;
      ln_id_list = '';
      ln_agreement_list = '';
      ln_count = 0;
    end else
    if (last_date >= :rep_date_end) then
    begin
      ln_date_end = null;

      if (coalesce(kc_list, '') <> '') then
      begin
        if (ln_id_list <> '') then
          ln_id_list = ln_id_list || ', ';
        ln_id_list = ln_id_list || kc_list;

        if (ln_agreement_list <> '') then
          ln_agreement_list = ln_agreement_list || ', ';
        ln_agreement_list = ln_agreement_list || kont_list;

        ln_count = ln_count + tmp_count;
      end

      ln_id = ln_id + 1;
      uniq_id = c_id || '_' || ln_id;
      client_point = f_point(:c_id, coalesce(ln_date_end, current_date), 0);
      suspend;
      exit;
    end else
      ln_date_end = last_date + 1;
  end
end
