CREATE PROCEDURE HP_MONETA_DIFF_KONT_PAY_SBP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    K_KONTRACT_NO STR_100,
    SBPPHONE STR_100,
    ESM_DATE DATE,
    AMOUNT CURR_TYPE,
    SVZ_FIO STR_100,
    KAL_NEW_KF_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED CURR_TYPE,
    CL_FIO VARCHAR(100))
AS
declare variable kf_id id_type;
declare variable k_id_old id_type;
declare variable reg_id_old integer;
declare variable kf_officer_old id_type;
declare variable k_id id_type;
declare variable reg_id integer;
declare variable kf_officer id_type;
declare variable kc_from_c_id id_type;
begin
  kf_officer_old = 0;
  reg_id_old = 0;
  k_id_old = 0;

  for select k_id, sbpphone, amount, esm_date, reg_id, kf_officer
      from hp_moneta_diff_kont_pay_sbp_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, 3)
      order by sbpphone, k_id
      into :k_id, :sbpphone, :amount, :esm_date, :reg_id, :kf_officer
  do
  begin
    if (reg_id <> reg_id_old) then
    begin
      select reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_name;

      reg_id_old = reg_id;
    end

    if (kf_officer <> kf_officer_old) then
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :kf_officer
      into :svz_fio;

      kf_officer_old = kf_officer;
    end

    if (k_id <> k_id_old) then
    begin
      select kont.k_kontract_no, k_amount_approved, k_disb_date, k_from_kf_id
      from kontract kont
      where kont.k_id = :k_id
      into :k_kontract_no, k_amount_approved, k_disb_date, kf_id;

      kal_new_kf_name = '';

      select kal.kal_new_kf_name
      from knaktlist kal
      where kal.kal_from_kf_id = :kf_id and
            :rep_date_end between kal_date and kal_date_end and
            kal.is_delete = 0
      into :kal_new_kf_name;

      for select kc_from_c_id

          from knclient kc
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0
          into kc_from_c_id
      do
      begin
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into cl_fio;
        suspend;
      end

      k_id_old = k_id;
    end
  end
end
