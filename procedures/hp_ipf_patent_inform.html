CREATE PROCEDURE HP_IPF_PATENT_INFORM
AS
declare variable msg_mail_to varchar(255);
declare variable msg_subject varchar(255);
declare variable msg_body blob sub_type 1 segment size 80;
declare variable head_body blob sub_type 1 segment size 80;
declare variable tmp_body blob sub_type 1 segment size 80;
declare variable reg_id integer;
declare variable reg_email varchar(255);
declare variable c_reg_id integer;
declare variable c_name varchar(100);
declare variable c_date_end date;
declare variable cs_date_end str_100;
declare variable rd_id id_type;
declare variable rd_id_1 id_type;
begin
  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.IPF_Sellers'
  into :rd_id;

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.tsp_and_ipf'
  into :rd_id_1;


  msg_subject = 'Уведомление об истечении срока патента контрагента ИПФ';
  head_body = '<b>Внимание !<br>Истекает срок патента у следующих поставщиков:</b><br><br>' || '<table border="1" cols="2"><tr><th>Ф.И.О контрагента</th><th>Дата окончания</th></tr>';
  tmp_body = '';
  reg_id = -1;
  for select coalesce(ag_from_init_reg_id, ag_from_reg_id) as reg_id, ag_caption, max(cds_doc_date_end)
      from agreements ag, clientdocs, passdoctype, passkind
      where ag_object_kind = 0 and
            ag_object_id = cds_from_c_id and
            pdt_id = pk_from_pdt_id and
            pdt_code = 'work_patent' and
            cds_from_pk_id = pk_id and
            cds_doc_date_end between current_date + 1 and current_date + 10 and
            ag_status = 1 and
            ag.ag_from_rd_id in (:rd_id, :rd_id_1) and
            current_date between ag_date_begin and ag_date_end
      group by reg_id, ag_object_id, ag_caption
      union
      select 999999999, null, null
      from rdb$database
      order by 1, 3, 2
      into c_reg_id, c_name, c_date_end
  do
  begin
    if (c_reg_id <> reg_id) then
    begin
      msg_mail_to = null;
      if (reg_id <> -1) then
      begin
        for select reg_email_manager
            from region
            where reg_unicode = :reg_id and
-- REG_STATUS = 3 and
                  reg_email_manager is not null and
                  is_delete = 0 and
                  reg_date_close is null
            into msg_mail_to
        do
        begin
          msg_body = head_body || tmp_body || '</table>';
          insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body)
          values (0, :msg_mail_to, :msg_subject, :msg_body);
        end
        tmp_body = '';
      end
      reg_id = c_reg_id;
    end
    select sval
    from datetostr(:c_date_end)
    into cs_date_end;
    tmp_body = blob_append(tmp_body, '<tr><td>' || c_name || '</td><td align="center">' || cs_date_end || '</td></tr>');
  end
end
