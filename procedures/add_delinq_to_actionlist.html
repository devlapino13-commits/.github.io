CREATE PROCEDURE ADD_DELINQ_TO_ACTIONLIST
AS
declare variable OBJ_ID D_ID;
declare variable RD_ID D_ID;
declare variable REG_ID D_ID;
declare variable DELINQ_DAYS D_INTEGER;
declare variable OBJ_KIND D_INTEGER;
declare variable db_id d_id;
begin
    db_id = f_get_db_id();
    if (db_id = 39801) then
    begin
        -- RFCKZ-201 --
        -- Создает запись, фиксирующую 5 день просрочки
        rd_id = f_get_rd_id('ActionList.Delinq_Notif');
        for select k.k_id,
                   k.k_last_reg_id,
                   current_date - oks.oks_delinq_date
        from oksummary oks
        inner join kontract k on oks.oks_from_k_id = k.k_id
        where oks.oks_close_state = 0
          and current_date between oks.oks_date_begin and oks.oks_date_end
          and current_date - oks.oks_delinq_date between 5 and 15 --Отбираем активные контракты от 5 до 15 дней просрочки
        into obj_id, reg_id, delinq_days

        do
        begin
            obj_kind = f_objkind('kontract', 'id');
            if (not exists(select * -- Проверка на наличие уже существующей записи
                       from actionlist acl
                       where acl.acl_from_rd_id = :rd_id
                         and acl.acl_date < current_date
                         and acl.acl_object_kind = :obj_kind
                         and acl.acl_object_id = :obj_id
                         and acl.acl_from_reg_id = :reg_id)
                         and (delinq_days = 5)) then   --Проверка дня просрочки
            begin  --Добавление записи в ActionList
                insert into actionlist (acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                                  acl_respons_c_id, acl_datetime, acl_from_un_id, acl_result_rd_id, acl_last_datetime)
                values (current_date, :reg_id, :obj_kind, :obj_id, :rd_id, 2, 1000000000000, current_timestamp, 1000000000000, 0, current_timestamp);
            end
        end
    end
end
