CREATE PROCEDURE HP_ANKETA_INCOME_OUTCOME(
    K_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_LONG,
    CA_DATE D_DATE,
    MA_ID D_INTEGER,
    CAF_KPD_ERR D_STR_255)
AS
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable mac_id d_integer;
declare variable caf_from_ca_id d_integer;
declare variable k_disb_date d_date;
begin
  -- Модуль анкеты анализ доходов/расходов. RFCKG-3378
  res_info_text = '<body><style>table {border-collapse: collapse} *{border: 1px solid} td {padding: 3px} .name {text-wrap: nowrap}</style><table><tr><th scope="col">Клиент</th> <th scope="col">Дата анкеты</th> <th scope="col">Комментарий</th></tr>';

  -- Получаем заявку контракта
  select ma.ma_id, k.k_disb_date
  from kontract k
  inner join miniapp ma on ma.ma_id = k.k_from_ma_app_id
  where k.k_id = :k_id
  order by ma.ma_number desc
  rows 1
  into ma_id, k_disb_date;

  --Получаем клиентов контракта
  for select kc.kc_from_c_id, c.c_name_fio
  from knclient kc
  inner join clients c on c.c_id = kc.kc_from_c_id
  where kc.kc_from_k_id = :k_id
  into c_id, c_name_fio
  do
  begin
    --Получаем клиента в заявке
    select mac.mac_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id
      and  mac.mac_from_c_id = :c_id
    into mac_id;

    select caf.caf_kpd_err, caf.caf_from_ca_id
    from cfp_access_form caf
    where caf.caf_from_c_id = :c_id
      and caf.caf_mac_id = :mac_id
    into  caf_kpd_err,
          caf_from_ca_id;

    caf_kpd_err = coalesce(caf_kpd_err, '');

    if (caf_from_ca_id is not null) then
    begin
      select ca.ca_date
      from cfp_anketa ca
      where ca.ca_id = :caf_from_ca_id
      into ca_date;

      res_info_text = res_info_text || 
                      '<tr><td class="name">' || c_name_fio || '</td>' ||
                      '<td>' || f_datetostr(coalesce(ca_date, k_disb_date)) || '</td>' ||
                      '<td>' || caf_kpd_err || '</td></tr>';
    end
  end

  res_info_text = res_info_text || '</table></body>';
  suspend;
end
