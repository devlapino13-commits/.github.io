CREATE PROCEDURE AAREP_OPEREXPREG_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_IS_ALL INTEGER)
RETURNS (
    OB_FROM_OL_ID DECIMAL(18,0),
    OB_NUMBER INTEGER,
    OL_DATE DATE,
    OL_MEMO VARCHAR(1024),
    OB_ACCOUNT_DT VARCHAR(20),
    OB_ACCOUNT_KT VARCHAR(20),
    OB_SUMMA DECIMAL(18,4),
    OB_DETAIL VARCHAR(1024),
    UN_NAME VARCHAR(30),
    OB_BALANCE_DT VARCHAR(20),
    OB_BALANCE_KT VARCHAR(20),
    OB_AL_NAME_DT VARCHAR(200),
    OB_AL_NAME_KT VARCHAR(200),
    IS_DONE INTEGER,
    EXR_DATE DATE,
    OL_REG_CODE VARCHAR(10),
    OL_REG_NAME VARCHAR(100),
    OL_DOC_NO INTEGER,
    OB_ACC_SHORT_DT VARCHAR(20),
    OB_ACC_SHORT_KT VARCHAR(20),
    OB_CURRENCY VARCHAR(3))
AS
declare variable exr_amount decimal(18,4);
begin
  for select ob_from_ol_id, ob_number, ol_doc_no, ol_date_oper_day, ol_memo, ob_account_dt, ob_account_kt, ob_summa,
             ob_detail, un_name, bg_dt.bg_id, bg_kt.bg_id, al_dt.al_name, al_kt.al_name, reg_code, reg_name,
             al_dt.al_account, al_kt.al_account, ob_currency
      from operbook, operlist, region, region_list(:rep_region_id, 1, -1, 1), acc_list al_dt, acc_list al_kt, balgroup bg_dt, balgroup bg_kt, username
      where ob_from_ol_id = ol_id and
            ob_ol_is_active = 1 and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            ob_ol_reg_id = reg_unicode and
            reg_unicode = id_list and
            ob_account_dt = al_dt.al_id and
            ob_account_kt = al_kt.al_id and
            al_dt.al_from_bg_id = bg_dt.bg_id and
            al_kt.al_from_bg_id = bg_kt.bg_id and
            ((bg_dt.bg_kind >= 4 and
            bg_kt.bg_kind <> 3) or
            (bg_kt.bg_kind >= 4 and
            bg_dt.bg_kind <> 3)) and
            ol_user_id = un_id and
            ob_summa is not null and
            ob_summa <> 0
      into ob_from_ol_id, ob_number, ol_doc_no, ol_date, ol_memo, ob_account_dt, ob_account_kt, ob_summa, ob_detail,
           :un_name, :ob_balance_dt, :ob_balance_kt, ob_al_name_dt, ob_al_name_kt, ol_reg_code, ol_reg_name,
           ob_acc_short_dt, ob_acc_short_kt, :ob_currency
  do
  begin
    exr_amount = 0;
    exr_date = null;
    select sum(occ_amount)
    from opercc
    where occ_from_ol_id = :ob_from_ol_id and
          occ_ob_number = :ob_number
    into :exr_amount;
    if (exr_amount is null) then
      exr_amount = 0;
    if (exr_amount = :ob_summa) then
      is_done = 1;
    else
      is_done = 0;
    if (rep_is_all = 0) then
    begin
      if (is_done = 0) then
        suspend;
    end
    else
      suspend;
  end
end
