CREATE PROCEDURE HP_AKB_SET_DATA(
    EIL_ID D_ID)
AS
declare variable OBJECT_ID D_ID;
declare variable JSON_TEXT D_BLOB_BINARY;
declare variable SCORE D_BLOB_TEXT;
declare variable POINT D_STR_20;
declare variable RESPONSE D_STR_20;
declare variable STATUS_DATETIME D_DATETIME;
begin
  object_id = null;
  status_datetime = null;
  json_text = '';
  select eil.eil_object_id, eil.eil_status_datetime, bls.bls_blob
  from blobstore bls
  inner join extinfolist eil on bls.bls_id = eil.eil_from_bls_id
  where eil.eil_id = :eil_id and
        eil.eil_status = 2
  into :object_id, :status_datetime, :json_text;

  if (json_text <> '') then
  begin
    if (object_id is null) then
      exit;

    score = coalesce(f_json_blob_get_value(:json_text, 'score', ''), '');
    point = coalesce(f_json_blob_get_value(:score, 'point', ''), 0);
    response = coalesce(f_json_blob_get_value(:score, 'response', ''), '');

    update kibminiapp kbm
    set kbm.kbm_score_rate = cast(:point as d_integer),
        kbm.kbm_response = :response
    where kbm.kbm_is_checked = 1 and
          kbm.kbm_id = :object_id and
          kbm.kbm_ver_info = 3;
  end
end
