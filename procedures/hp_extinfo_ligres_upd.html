CREATE PROCEDURE HP_EXTINFO_LIGRES_UPD(
    EIL_ID D_ID)
AS
declare variable bls_id d_id;
declare variable object_kind d_integer;
declare variable object_id d_id;
declare variable ip_num d_str_255;
declare variable ip_start d_date;
declare variable ip_end_date d_date;
declare variable end_section d_str_255;
declare variable fio_pristav d_str_255;
declare variable fssp_dep d_str_255;
declare variable fssp_adr d_str_long;
declare variable cnt d_integer;
declare variable smc_status d_integer;
declare variable smc_collect_status d_integer;
declare variable sag_id d_id;
declare variable res_ok d_bool;
declare variable smr_id d_id;
begin
  select eil_from_bls_id, eil_object_kind, eil_object_id
  from extinfolist
  where eil_id = :eil_id
  into bls_id, object_kind, object_id;

  if (object_kind <> f_objkind('sud_loanclient', 'id')) then
    exit;

  -- Ищем наши договоры (по ИНН компании)
  select max(ip_num), max(fssp_dep), max(fssp_adr), max(ip_start), max(ip_end_date), max(end_section), max(fio_pristav), count(1)
  from hp_get_ligres_content(:bls_id)
  where claimant_inn = '7728744752'
  into ip_num, fssp_dep, fssp_adr, ip_start, ip_end_date, end_section, fio_pristav, cnt;

  smr_id = (select slc_from_smr_id from sud_loanclient where slc_id = :object_id);

  -- обрабатываем только если инфо по нашим делам только один блок, если несколько, то сложно определить какой блок использовать, будут обрабатывать вручную.
  if (cnt <> 1) then
  begin
    select res_ok
    from hp_sud_add_memo(:smr_id, 'В ответе от Лигрес обнаружено несколько ИП. Требуется ручная обработка', 1000000000000)
    into res_ok;

    exit;
  end

  sag_id = (select sag_id from sud_agents where sag_kind = 4 and replace(replace(sag_name, ' ', ''), '.', '') = replace(replace(:fssp_dep, ' ', ''), '.', ''));

  if (sag_id is null) then
    insert into sud_agents (sag_kind, sag_name, sag_address)
    values (4, :fssp_dep, :fssp_adr)
    returning sag_id
    into sag_id;

  smc_collect_status = case replace(end_section, ' ' , '')
                         when 'ст.46,ч.1,п.3' then 2
                         when 'ст.46,ч.1,п.4' then 3
                         when 'ст.47,ч.1,п.7' then 4
                       end;

  smc_status = 0;

  if (sag_id is not null) then
    smc_status = 1;

  if (smc_status > 0
    and smc_collect_status is not null
    and ip_end_date is not null
    and fio_pristav is not null)
  then
    smc_status = 2;

  update sud_maincollect
  set smc_number = :ip_num,
      smc_initorder_date = :ip_start,
      smc_from_sag_id = :sag_id,
      smc_marshal = :fio_pristav,
      smc_doc_date = :ip_end_date,
      smc_collect_status = :smc_collect_status
  where smc_from_slc_id = :object_id
    and smc_status in (0,1);

  begin
    update sud_maincollect
    set smc_status = :smc_status
    where smc_from_slc_id = :object_id
      and smc_status in (0,1);

    when any do
      select res_ok
      from hp_sud_add_memo(:smr_id, 'При обработке ответа от Лигрес произошла ошибка перевода в стаус: ' || coalesce(:smc_status, '<NULL>'), 1000000000000)
      into res_ok;
  end

  when any do
    select res_ok
    from hp_sud_add_memo(:smr_id, 'При обработке ответа от Лигрес произошла ошибка!', 1000000000000)
    into res_ok;

end
