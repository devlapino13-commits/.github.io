CREATE PROCEDURE HP_KONTRACT_INFO_LIST(
    IN_K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    SH_CALC_FINE D_CURRENCY,
    SH_CALC_PERC D_CURRENCY,
    ZAD_OS D_CURRENCY,
    ZAD_PERC D_CURRENCY,
    SH_REPAY_TYPE D_INTEGER,
    S_OS D_CURRENCY,
    S_PERC D_CURRENCY,
    S_ALL D_CURRENCY,
    V_PAY_OS D_CURRENCY,
    V_PAY_PERC D_CURRENCY,
    IS_GRACE D_INTEGER,
    PAC_TYPE D_INTEGER,
    PAC_PERCENT D_PERCENT,
    PAC_INC_INT D_SMALLINT,
    K_STOP_FINE D_DATE,
    K_INDEX_CURR_RATE D_CURRENCY,
    K_CURRENCY_RATE D_CURRENCY,
    SH_FINE_BASE D_CURRENCY,
    PR_MIN_FINE_BASE D_CURRENCY,
    SH_DATE_SH_ORIG D_DATE,
    SUM_FINE_BASE_OS D_CURRENCY,
    SUM_FINE_BASE_PERC D_CURRENCY,
    SH_FINE_KOL_DAY D_INTEGER,
    K_STOP_PERC D_DATE,
    DD_CHANGE_BEGIN D_DATE,
    DD_CHANGE_END D_DATE,
    DD_CHANGE_DAYS D_INTEGER,
    ZAD_KOMMIS D_CURRENCY,
    SH_PRIOR_PERCENT_PER_DAY D_PERCENT,
    IS_EXTRA D_SMALLINT,
    IS_HIDDEN_LINE D_SMALLINT,
    SH_ADD_PERC D_CURRENCY,
    S_RUNNING_TOTAL D_CURRENCY)
AS
declare variable tax_perc d_percent;
declare variable tax_amount d_currency;
declare variable db_id d_integer;
declare variable k_disb_date d_date;
declare variable s_total_calcing d_currency;
declare variable s_total_zad d_currency;
begin
  select k_disb_date
  from kontract
  where k_id = :in_k_id
  into k_disb_date;

  db_id = f_get_db_id();

  tax_perc = 0;
  select nsp_percent_for_int
  from hp_get_kontract_nsp_acc(0, :k_disb_date)
  into :tax_perc;

  s_total_calcing = 0;
  for select sh_id, sh_date, sh_calc_fine, sh_calc_perc, zad_os, zad_perc, sh_repay_type, s_os, s_perc, s_all, v_pay_os,
             v_pay_perc, is_grace, pac_type, pac_percent, pac_inc_int, k_stop_fine, k_index_curr_rate, k_currency_rate,
             sh_fine_base, pr_min_fine_base, sh_date_sh_orig, sum_fine_base_os, sum_fine_base_perc, sh_fine_kol_day,
             k_stop_perc, dd_change_begin, dd_change_end, dd_change_days, zad_kommis, sh_prior_percent_per_day, is_extra,
             is_hidden_line, sh_add_perc
  from aarep_make_zad_list(:in_k_id, :rep_date, 1, 0)
  into :sh_id, :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os, :zad_perc, :sh_repay_type, :s_os, :s_perc, :s_all,
       :v_pay_os, :v_pay_perc, :is_grace, :pac_type, :pac_percent, :pac_inc_int, :k_stop_fine, :k_index_curr_rate,
       :k_currency_rate, :sh_fine_base, :pr_min_fine_base, :sh_date_sh_orig, :sum_fine_base_os, :sum_fine_base_perc,
       :sh_fine_kol_day, :k_stop_perc, :dd_change_begin, :dd_change_end, :dd_change_days, :zad_kommis,
       :sh_prior_percent_per_day, :is_extra, :is_hidden_line, :sh_add_perc
  do
  begin
    tax_amount = 0;

    if (tax_perc <> 0 and not(db_id in (39801, 41701) and sh_date is null)) then
      tax_amount = round(sh_calc_perc * tax_perc / 100.0000, 0) + round(zad_kommis * tax_perc / 100.0000, 0);

    s_total_calcing = s_total_calcing + zad_os + sh_calc_perc + zad_kommis + sh_add_perc + tax_amount;
  end

  s_running_total = 0;
  s_total_zad = 0;
  for select sh_id, sh_date, sh_calc_fine, sh_calc_perc, zad_os, zad_perc, sh_repay_type, s_os, s_perc, s_all, v_pay_os,
             v_pay_perc, is_grace, pac_type, pac_percent, pac_inc_int, k_stop_fine, k_index_curr_rate, k_currency_rate,
             sh_fine_base, pr_min_fine_base, sh_date_sh_orig, sum_fine_base_os, sum_fine_base_perc, sh_fine_kol_day,
             k_stop_perc, dd_change_begin, dd_change_end, dd_change_days, zad_kommis, sh_prior_percent_per_day, is_extra,
             is_hidden_line, sh_add_perc
  from aarep_make_zad_list(:in_k_id, :rep_date, 1, 0)
  into :sh_id, :sh_date, :sh_calc_fine, :sh_calc_perc, :zad_os, :zad_perc, :sh_repay_type, :s_os, :s_perc, :s_all,
       :v_pay_os, :v_pay_perc, :is_grace, :pac_type, :pac_percent, :pac_inc_int, :k_stop_fine, :k_index_curr_rate,
       :k_currency_rate, :sh_fine_base, :pr_min_fine_base, :sh_date_sh_orig, :sum_fine_base_os, :sum_fine_base_perc,
       :sh_fine_kol_day, :k_stop_perc, :dd_change_begin, :dd_change_end, :dd_change_days, :zad_kommis,
       :sh_prior_percent_per_day, :is_extra, :is_hidden_line, :sh_add_perc
  do
  begin
    tax_amount = 0;

    if (tax_perc <> 0 and not(db_id in (39801, 41701) and sh_date is null)) then
      tax_amount = round(sh_calc_perc * tax_perc / 100.0000, 0) + round(zad_kommis * tax_perc / 100.0000, 0);

    s_all = zad_os + maxvalue(sh_calc_perc, zad_perc) + zad_kommis + sh_add_perc + tax_amount;
    s_total_zad = s_total_zad + s_all;
    s_all = 0;

    s_running_total = s_total_zad;
    if (s_running_total > s_total_calcing) then
      s_running_total = s_total_calcing;

    if (tax_perc <> 0 and not(db_id in (39801, 41701) and sh_date is null)) then
      tax_amount = round(zad_perc * tax_perc / 100.0000, 0) + round(zad_kommis * tax_perc / 100.0000, 0);

    sh_add_perc = tax_amount;
    s_all = zad_os + zad_perc + zad_kommis + sh_add_perc;

    suspend;
  end
end
