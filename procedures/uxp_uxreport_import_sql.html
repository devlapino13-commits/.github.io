CREATE PROCEDURE UXP$UXREPORT_IMPORT_SQL(
    RL_ID D_INTEGER)
RETURNS (
    RL_SQL D_STR_LONG)
AS
begin
  select rl_sql
  from reportlist
  where rl_id = :rl_id
  into rl_sql;

  rl_sql = trim(coalesce(rl_sql, ''));

  if (rl_sql <> '' and rl_sql not starting 'select ') then
    rl_sql = 'select * from ' || rl_sql;

  suspend;
end
