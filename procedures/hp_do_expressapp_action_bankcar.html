CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_BANKCAR(
    EXA_ID D_INTEGER)
RETURNS (
    RES_STATUS D_SMALLINT)
AS
declare variable exa_from_c_id id_type;
declare variable exa_disb_date date;
declare variable card_date date;
declare variable exa_from_ma_id d_integer;
declare variable exa_from_un_id id_type;
declare variable edoc_kind d_str_code;
declare variable edl_id d_id;
declare variable exa_from_edl_id d_id;
begin
-- проверка имеется ли у клиента банковская карта, если да то проверяется когда была закреплена
-- если прошло более 7ми дней или не имеется карта то клиенту отправляется электронный документ чтобы заново закрепили
  res_status = 0;
  exa_from_c_id = null;
  exa_from_edl_id = null;
  exa_from_ma_id = null;
  select exa_from_c_id, exa_disb_date, exa_from_ma_id, exa_from_un_id, exa_from_edl_id
  from expressapp
  where exa_id = :exa_id
  into exa_from_c_id, exa_disb_date, exa_from_ma_id, exa_from_un_id, exa_from_edl_id;

  if (exa_from_ma_id is null) then
  begin
    res_status = -1;
    suspend;
    exit;
  end 

  -- дата закрепление карты
  card_date = null;
  select cast(token_datetime as date) from hp_client_bank_card(:exa_from_c_id)
  into card_date;

  -- если это не боевая база то дату закрепления делаем текущую дату т.е. для тестирование
  if (card_date is not null and
      card_date + 7 < exa_disb_date and
      not exists(select * from hp_db_is_actual)) then
  begin
    card_date = current_date;

    update extsys_token
    set et_token_datetime = current_timestamp
    where et_from_c_id = :exa_from_c_id and
          et_is_actual = 1
    order by et_id desc
    rows 1;
  end

  if (card_date is null or exa_disb_date > card_date + 7) then
  begin
    if (exa_from_edl_id is null) then -- если еще не отправлен эл. документ на закрепление банковской карты
    begin
      -- не имеется или не закреплена банковская карта
      -- нужно создать электронный документ, на закрпеление банковской карты
      edoc_kind = 'bankcard.moneta';

      select edl_id
      from hp_edoc_ins_bankcard(:edoc_kind,:exa_from_c_id, :exa_from_un_id)
      into edl_id;
    
      update expressapp
      set exa_from_edl_id = :edl_id
      where exa_id = :exa_id;
    end
    res_status = 2; -- был 1
  end else
  begin
    res_status = 0;
  end

  suspend;
end
