CREATE PROCEDURE HP_PERMISSIONS_CHECK_LIST(
    USER_ROLES VARCHAR(255))
RETURNS (
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_INTEGER,
    ML_ID D_INTEGER,
    PM_ID D_INTEGER,
    PM_DATE_BEGIN D_DATE,
    PM_DATE_END D_DATE)
AS
declare variable sql_text varchar(2048);
declare variable is_one_role integer;
declare variable rn_id decimal(18,0);
declare variable uniqtab_rolename integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;

  if (position(',', user_roles) <= 0) then
  begin
    is_one_role = 1;
    rn_id = cast(user_roles as id_type);
  end
  else
    is_one_role = 0;

  sql_text = 'select distinct pm_subject_kind, pm_subject_id from permissions
  where pm_object_kind=' || :uniqtab_rolename || ' and pm_object_id in (' || :user_roles || ') and pm_subject_kind in (2,3,5) and pm_is_delete=0
  and (coalesce(pm_date_begin, ''01.01.2000'')<=current_date)
  and (coalesce(pm_date_end, ''31.12.2999'')>=current_date)';
  for execute statement(:sql_text)
  into :object_kind, :object_id
  do
  begin
    pm_id = -1;
    pm_date_begin = null;
    pm_date_end = null;
    if (is_one_role = 1) then
    begin
      select pm_id, pm_date_begin, pm_date_end
      from permissions
      where pm_object_kind = :uniqtab_rolename and
            pm_object_id = :rn_id and
            pm_subject_kind = :object_kind and
            pm_subject_id = :object_id and
            pm_is_delete = 0 and
            (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
            (coalesce(pm_date_end, '31.12.2999') >= current_date)
      rows 1
      into :pm_id, pm_date_begin, pm_date_end;
    end

    if (object_kind = 3) then -- PermActiin
      ml_id = 2000000 + object_id;
    else
    if (object_kind = 5) then -- UxAlias
      ml_id = 5000000 + object_id;
    else
      ml_id = object_id;

    suspend;
  end
end
