CREATE PROCEDURE AAREP_PRO_RAZDEL3(
    REP_DATE_END DATE)
RETURNS (
    BSL_NAME STR_255,
    RESTRUCT_BALANCE CURR_TYPE,
    DELINQ_BALANCE CURR_TYPE,
    LLC_STANDART_BALANCE CURR_TYPE,
    LLC_POD_NABLUD_BALANCE CURR_TYPE,
    LLC_SUBSTANDART_BALANCE CURR_TYPE,
    LLC_SOMNITEL_BALANCE CURR_TYPE,
    LLC_POTERY_BALANCE CURR_TYPE,
    NO_ACCRUAL_STATUS_BALANCE CURR_TYPE,
    LLC_OBSHY_REZERV CURR_TYPE,
    LLC_SPEC_REZERV CURR_TYPE)
AS
declare variable ed_izmer curr_type;
begin
  ed_izmer = 1000;
  for select bsl_name, sum(iif(is_restruct = 1, balance_kc, 0)) / :ed_izmer, -- Реструктуризированные
             sum(iif(is_delinq = 1, balance_kc, 0)) / :ed_izmer, -- Просроченные
             sum(iif(lkc_status in (1, 3), balance_kc, 0)) / :ed_izmer, -- Стандартные
             sum(iif(lkc_status = 4, balance_kc, 0)) / :ed_izmer, -- Активы под наблюдением
             sum(iif(lkc_status = 5, balance_kc, 0)) / :ed_izmer, -- Субстандартные
             sum(iif(lkc_status = 6, balance_kc, 0)) / :ed_izmer, -- Сомнительные
             sum(iif(lkc_status = 7, balance_kc, 0)) / :ed_izmer, -- Потери
             sum(iif(no_accrual_status = 1, balance_kc, 0)) / :ed_izmer, -- Статус не начисления
             sum(iif(lkc_status <= 4, balance_kc, 0)) / :ed_izmer, -- Общий резерв
             sum(iif(lkc_status >= 5, balance_kc, 0)) / :ed_izmer -- Специальный резерв
      from aarep_pro_razdel3_temp(:rep_date_end)
      group by bsl_name
      into bsl_name, restruct_balance, delinq_balance, llc_standart_balance, llc_pod_nablud_balance,
           llc_substandart_balance, llc_somnitel_balance, llc_potery_balance, no_accrual_status_balance,
           llc_obshy_rezerv, llc_spec_rezerv
  do
  begin

    suspend;
  end
end
