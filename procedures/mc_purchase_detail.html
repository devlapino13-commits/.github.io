CREATE PROCEDURE MC_PURCHASE_DETAIL(
    MCL_ID D_ID,
    MP_ID D_ID)
RETURNS (
    MP_RECEIPT_NO D_STR_20,
    MP_CATEG D_STR_100,
    MP_AMOUNT D_CURRENCY,
    MP_DATETIME D_DATETIME,
    MP_NAME_MAGAZIN D_STR_100,
    MP_STATUS D_INTEGER)
AS
declare variable db_id d_id;
declare variable c_id d_id;
begin
  db_id = f_get_db_id();
  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  if (c_id is null or db_id not in (39801, 41701) or mp_id is null) then
  begin
    suspend;
    exit;
  end
  --Тянем данные об операции по mp_id
  select mp.mp_receipt_number, mp.mp_amount, mp.mp_datetime, feg.feg_name, cs.cs_name, mp.mp_status
  from mc_purchase mp
  inner join clientshops cs on cs.cs_id = mp.mp_from_cs_id
  inner join foreign_econom_goods_ref feg on feg.feg_id = mp.mp_from_feg_id
  where mp.mp_id = :mp_id
  into :mp_receipt_no, :mp_amount, :mp_datetime, :mp_categ, :mp_name_magazin, :mp_status;
  suspend;

end
