CREATE PROCEDURE HP_EPAY_LOYAL_DO_CHECK(
    AG_ID D_INTEGER,
    REP_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable epd_id d_integer;
declare variable epd_date_end d_date;
declare variable epd_status d_integer;
declare variable epd_from_epl_id d_id;
declare variable epr_id d_integer;
declare variable client_non_resident d_integer;
begin
  res_text = '';
  res_is_error = 0;

  select epr_id
  from epay_project
  where epr_code = 'epay_loyal'
  into :epr_id;

  select ag_object_id
  from agreements
  where ag_id = :ag_id and
        ag_status = 1 and
        :rep_date between ag_date_begin and coalesce(ag_date_end, :rep_date)
  into c_id;

  if (c_id is null) then
  begin
    res_is_error = 1;
    res_text = 'Не найден клиент с действующим соглашением';
  end

  if (res_text = '') then
  begin
    select epd_id, epd_date_end, epd_status, epd_from_epl_id
    from epay_doc
    where epd_from_c_id = :c_id and
          epd_object_kind = f_objkind('agreements', 'id') and
          epd_object_id = :ag_id and
          epd_date_end <= :rep_date
    order by epd_date_end desc, epd_id desc
    into epd_id, epd_date_end, epd_status, epd_from_epl_id;
    if (epd_status = 1) then
    begin
      if (epd_date_end >= rep_date) then
      begin
        res_is_error = 1;
        res_text = 'Документы уже подтверждены на дату: ' || f_datetostr(epd_date_end);
      end
    end

    if (epd_from_epl_id is not null) then
    begin
      res_is_error = 1;
      res_text = 'По данной строке выплата баллов уже произведена';
    end
  end

  if (res_text = '') then
  begin
    select 1 - resident
    from aarep_epay_client_info(:c_id, :rep_date)
    into client_non_resident;
    if (client_non_resident is null) then client_non_resident = 0;

    if (epd_id is not null) then
    begin
      update epay_doc
      set epd_date_end = :rep_date,
          epd_non_resident = :client_non_resident,
          epd_status = 1,
          epd_from_un_id = :un_id
      where epd_id = :epd_id;
    end else
    begin
      insert into epay_doc (epd_from_epr_id, epd_from_c_id, epd_object_kind, epd_object_id, epd_date_end, epd_amount,
                            epd_non_resident, epd_status, epd_from_epl_id, epd_memo_id, epd_from_un_id)
      values (:epr_id, :c_id, f_objkind('agreements', 'id'), :ag_id, :rep_date, 0, :client_non_resident, 1, null, 0, :un_id);
    end

    res_text = 'Данные по документам успешно подтверждены';
  end
  suspend;
end
