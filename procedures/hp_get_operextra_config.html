CREATE PROCEDURE HP_GET_OPEREXTRA_CONFIG(
    RD_CODE VARCHAR(255))
RETURNS (
    RES_RD_ID INTEGER,
    RES_RD_NAME VARCHAR(255),
    RES_OBJECT_KIND VARCHAR(100),
    RES_DEFAULT_MEMO VARCHAR(1024),
    XLS_COLUMN_KEYNO VARCHAR(10),
    XLS_COLUMN_AMOUNT VARCHAR(10),
    XLS_COLUMN_DETAIL VARCHAR(10),
    XLS_COLUMN_DATE VARCHAR(10))
AS
begin
  /* Procedure Text */
  xls_column_keyno = '';
  xls_column_date = '';
  xls_column_amount = '';
  xls_column_detail = '';
  res_default_memo = '';

  select rd_id, rd_name
  from refdata
  where rd_code = :rd_code
  into res_rd_id, res_rd_name;

  if (res_rd_id is not null) then
  begin
    if (rd_code = 'OperExtraKind.LawClientExpense') then
    begin
      res_object_kind = 'Kontract';
      xls_column_keyno = 'B';
      xls_column_date = '';
      xls_column_amount = 'C';
      xls_column_detail = 'D';
      res_default_memo = 'Начисленная гос.пошлина по судебным искам/решениям';
    end

    suspend;
  end
end
