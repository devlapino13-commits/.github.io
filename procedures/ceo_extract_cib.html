CREATE PROCEDURE CEO_EXTRACT_CIB(
    CONTRACT_NO D_STR_20)
RETURNS (
    K_ID D_ID,
    MA_ID D_ID,
    EIL_ID D_ID,
    EIL_FROM_C_ID D_ID,
    EIL_FROM_EIC_ID D_ID,
    EIL_DATE D_DATE,
    EIL_OBJECT_KIND D_INTEGER,
    EIL_OBJECT_ID D_ID,
    EIL_FROM_BLS_ID D_ID,
    JSON_TEXT D_BLOB_TEXT)
AS
declare variable db_id d_integer;
begin
  db_id = f_Get_dB_id();
  for select k.k_id,
             k.k_from_ma_app_id
      from kontract k
      where k.k_kontract_no = :contract_no
      into k_id,
           ma_id
  do
  begin
    for select eil.eil_id,
               eil.eil_from_c_id,
               eil.eil_from_eic_id,
               eil.eil_date,
               eil.eil_object_kind,
               eil.eil_object_id,
               eil.eil_from_bls_id
        from extinfolist eil
        inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
              ((:db_id = 81001 and eic.eic_id in (14, 16, 1, 2)) or
               (:db_id = 39801 and eic.eic_id = 1))
        where ((eil.eil_object_kind = 11 and
              eil.eil_object_id = :ma_id) or
              (eil.eil_object_kind = 9 and
              eil.eil_object_id = :k_id)) and
              eil.eil_status = 2 and
              eil.eil_is_json = 1
        into eil_id,
             eil_from_c_id,
             eil_from_eic_id,
             eil_date,
             eil_object_kind,
             eil_object_id,
             eil_from_bls_id
    do
    begin
      select finish_json_text
      from hp_mask_client_data(:eil_from_bls_id)
      into json_text;

      if (json_text is not null) then
        suspend;
    end
  end
end
