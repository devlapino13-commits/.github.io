CREATE PROCEDURE HP_SEND_CONFIRMATIONS(
    C_ID D_ID,
    UN_ID D_ID)
AS
declare variable REP_CODE D_STR_CODE;
declare variable CC_ID D_INTEGER;
declare variable MCL_ID D_ID;
declare variable GOOGLE_ID D_STR_255;
declare variable PUSH_BODY D_STR_255;
begin
  -- RFCKG-4180 Отправляем ознакомление с согласиями
  rep_code = 'consent_confirmations';

  push_body = 'Ознакомтесь с согласиями';

  insert into consent_confirmations (cc_from_c_id, cc_from_un_id)
  values (:c_id, :un_id)
  returning cc_id, cc_from_mcl_id
  into :cc_id, mcl_id;

  select mcl.mcl_google_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into google_id;

  execute procedure mc_send_push('mc_login', :mcl_id, 'Ознакомление с согласиями', :push_body, 1);
end
