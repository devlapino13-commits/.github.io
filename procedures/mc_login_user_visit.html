CREATE PROCEDURE MC_LOGIN_USER_VISIT(
    USER_LOGIN D_STR_CODE,
    USER_PASS D_STR_CODE,
    GOOGLE_ID D_STR_255)
RETURNS (
    CAN_LOGIN D_INTEGER,
    ERR_MESSAGE D_STR_255,
    RES_CODE D_STR_CODE,
    MCL_ID D_ID,
    C_NAME_FIO D_STR_255,
    C_ID D_ID,
    MCL_LANG_CODE D_STR_LANG)
AS
declare variable login_params d_str_long;
begin
  can_login = 0;
  mcl_id = 0;
  c_name_fio = '';
  mcl_lang_code = f_get_lang_code(null, null);

  login_params =
    '{' ||
    f_json_make_pair('login', user_login, 'str') ||
    f_json_make_pair('password', user_pass, 'str') ||
    '}';

  select res_code, err_text
  from mc_login_action_visit('login', :login_params)
  into :res_code, :err_message;

  if (res_code = 'login_ok') then
  begin
    can_login = 1;

    select mcl_id, mcl_from_c_id, mcl_lang_code
    from mc_login
    where mcl_login = :user_login and
          mcl_pass_hash = :user_pass and
          mcl_is_delete = 0
    into mcl_id, c_id, mcl_lang_code;

    if (coalesce(mcl_id, 0) <> 0 and
        coalesce(google_id, '') <> '') then
      update mc_login
      set mcl_google_id = :google_id
      where mcl_id = :mcl_id and
            mcl_is_delete = 0;

    select coalesce(c_name_fio, '')
    from mc_login left join clients on mcl_from_c_id = c_id
    where mcl_id = :mcl_id
    into c_name_fio;
  end
  suspend;
end
