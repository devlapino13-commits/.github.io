CREATE PROCEDURE AAREP_KANS_PRINT_AKT_TMP(
    REP_ID_LIST VARCHAR(1024))
RETURNS (
    RES_KOP_ID INTEGER)
AS
declare variable kk integer;
declare variable ss varchar(255);
declare variable k integer;
begin
  rep_id_list = rep_id_list || ' 0';
  while (rep_id_list <> '') do
  begin
    kk = position(' ', :rep_id_list);
    if (kk > 0) then
    begin
      ss = left(:rep_id_list, kk - 1);
      ss = trim(leading from ss);
      k = cast(:ss as integer);
      for select kop_id
          from kansoper
          where kop_id = :k
          into :res_kop_id
      do
      begin
        suspend;
      end
      rep_id_list = right(:rep_id_list, char_length(:rep_id_list) - kk);
      if (rep_id_list = '0') then
        rep_id_list = '';
      if (rep_id_list = ' 0') then
        rep_id_list = '';
      if (rep_id_list = '  0') then
        rep_id_list = '';
    end
    else
      rep_id_list = '';
  end
end
