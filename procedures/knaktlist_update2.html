CREATE PROCEDURE KNAKTLIST_UPDATE2(
    KF DECIMAL(18,0))
AS
declare variable k_no integer;
declare variable d date;
declare variable d_prior date;
declare variable ka_id decimal(18,0);
declare variable reg_id integer;
declare variable reg_first integer;
declare variable k_id decimal(18,0);
declare variable kal_date_end date;
declare variable k_last_reg_id integer;
declare variable lo_id_prior decimal(18,0);
declare variable reg_id_prior integer;
declare variable kf_prior_name varchar(100);
declare variable kf_name varchar(100);
declare variable lo_id decimal(18,0);
declare variable lo_id_last decimal(18,0);
declare variable kal_to_other_lo_last integer;
declare variable kal_to_other_lo integer;
begin
  /* Procedure Text */
  reg_first = 0;
  update knaktlist
  set kal_date_end = (kal_date - 1)
  where kal_from_kf_id = :kf and
        is_delete = 1 and
        kal_date_end <> (kal_date - 1);

  d_prior = cast('31.12.9999' as date);
  for select kal_number, kal_from_ka_id, kal_date, kal_new_reg_id, kal_date_end
      from knaktlist
      where kal_from_kf_id = :kf and
            is_delete = 0
      order by kal_date desc, kal_from_ka_id desc, kal_number desc
      into :k_no, :ka_id, :d, :reg_id, :kal_date_end
--  as cursor cur
  do
  begin
    if (reg_first = 0) then
      reg_first = :reg_id;

    if ((d_prior <> kal_date_end) or (kal_date_end is null)) then
    begin
      update knaktlist
      set kal_date_end = :d_prior
      where kal_from_ka_id = :ka_id and
            kal_number = :k_no and
            kal_from_kf_id = :kf;
--      where current of cur;
    end
    d_prior = d - 1;
  end

  lo_id_prior = null;
  reg_id_prior = null;
  kf_prior_name = null;
  for select kal_number, kal_from_ka_id, kal_date, kal_new_reg_id, kal_date_end, kal_new_kf_officer, kal_new_kf_name,
             kal_to_other_lo
      from knaktlist
      where kal_from_kf_id = :kf and
            is_delete = 0
      order by kal_date, kal_from_ka_id, kal_number
      into :k_no, :ka_id, :d, :reg_id, :kal_date_end, :lo_id, :kf_name, :kal_to_other_lo_last
--  as cursor cur
  do
  begin
    kal_to_other_lo = 0;
    if (kal_to_other_lo_last is null) then
      kal_to_other_lo_last = -2;
    if (lo_id_prior is null) then
    begin
      lo_id_prior = :lo_id;
      reg_id_prior = :reg_id;
      kf_prior_name = kf_name;
      lo_id_last = :lo_id;
      kal_to_other_lo = -1;
    end

    if ((lo_id > 1000000000004) and
        (kal_to_other_lo = 0)) then
    begin
      if (lo_id <> lo_id_last) then
        kal_to_other_lo = 1;
    end

    if ((lo_id_prior <> lo_id) or (reg_id_prior <> reg_id) or (kf_prior_name = kf_name) or (kal_to_other_lo <> kal_to_other_lo_last)) then
    begin
      update knaktlist
      set kal_prior_reg_id = :reg_id_prior,
          kal_prior_lo_id = :lo_id_prior,
          kal_prior_kf_name = :kf_prior_name,
          kal_to_other_lo = :kal_to_other_lo
      where kal_from_ka_id = :ka_id and
            kal_number = :k_no and
            kal_from_kf_id = :kf;
--      where current of cur;
    end
    lo_id_prior = :lo_id;
    reg_id_prior = :reg_id;
    kf_prior_name = kf_name;
  end
end
