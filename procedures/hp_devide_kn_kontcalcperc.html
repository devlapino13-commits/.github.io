CREATE PROCEDURE HP_DEVIDE_KN_KONTCALCPERC(
    K_ID ID_TYPE,
    KCP_PERIOD INTEGER,
    LOAN_KC_LIST D_STR_255,
    LOAN_AMOUNT_LIST D_STR_255,
    LOAN_KOMMIS_LIST D_STR_255,
    DIFF_KCP_PERC_BEGIN_SALDO CURR_TYPE = 0,
    DIFF_KCP_KOMMIS_BEGIN_SALDO CURR_TYPE = 0,
    DIFF_KCP_OUT_PERC_BEGIN_SALDO CURR_TYPE = 0,
    DIFF_KCP_PERC_CALCED CURR_TYPE = 0,
    DIFF_KCP_KOMMIS_CALCED CURR_TYPE = 0,
    DIFF_KCP_OUT_PERC_CALCED CURR_TYPE = 0)
RETURNS (
    KC_ID ID_TYPE,
    CL_KCP_FROM_REG_ID INTEGER,
    CL_KCP_PERC_SALDO CURR_TYPE,
    CL_KCP_TRANSFER_REG_ID INTEGER,
    CL_KCP_PERC_CALCED CURR_TYPE,
    CL_KCP_PERC_PAYED CURR_TYPE,
    CL_KCP_PERC_WRITEOFF CURR_TYPE,
    CL_KCP_PERC_BEGIN_SALDO CURR_TYPE,
    CL_KCP_PERC_TRANSFER_PAYED CURR_TYPE,
    CL_KCP_PERC_TRANSFER_WRITEOFF CURR_TYPE,
    CL_KCP_KOMMIS_CALCED CURR_TYPE,
    CL_KCP_KOMMIS_PAYED CURR_TYPE,
    CL_KCP_KOMMIS_WRITEOFF CURR_TYPE,
    CL_KCP_KOMMIS_SALDO CURR_TYPE,
    CL_KCP_KOMMIS_BEGIN_SALDO CURR_TYPE,
    CL_KCP_KOMMIS_TRANSFER_PAYED CURR_TYPE,
    CL_KCP_KOMMIS_TRANSFER_WRITEOFF CURR_TYPE,
    CL_KCP_PERC_ACCOUNT VARCHAR(20),
    CL_KCP_KOMMIS_ACCOUNT VARCHAR(20),
    CL_KCP_LOAN_IS_CLOSED INTEGER,
    CL_KCP_OUT_PERC_BEGIN_SALDO CURR_TYPE,
    CL_KCP_OUT_PERC_CALCED CURR_TYPE,
    CL_KCP_OUT_PERC_SALDO CURR_TYPE)
AS
declare variable tmp_kc_list d_str_255;
declare variable tmp_amount_list d_str_255;
declare variable kcp_from_reg_id integer;
declare variable kcp_perc_saldo curr_type;
declare variable kcp_transfer_reg_id integer;
declare variable kcp_perc_calced curr_type;
declare variable kcp_perc_payed curr_type;
declare variable kcp_perc_writeoff curr_type;
declare variable kcp_perc_begin_saldo curr_type;
declare variable kcp_perc_transfer_payed curr_type;
declare variable kcp_perc_transfer_writeoff curr_type;
declare variable kcp_kommis_calced curr_type;
declare variable kcp_kommis_payed curr_type;
declare variable kcp_kommis_writeoff curr_type;
declare variable kcp_kommis_saldo curr_type;
declare variable kcp_kommis_begin_saldo curr_type;
declare variable kcp_kommis_transfer_payed curr_type;
declare variable kcp_kommis_transfer_writeoff curr_type;
declare variable kcp_perc_account varchar(20);
declare variable kcp_kommis_account varchar(20);
declare variable kcp_loan_is_closed integer;
declare variable kcp_out_perc_begin_saldo curr_type;
declare variable kcp_out_perc_calced curr_type;
declare variable kcp_out_perc_saldo curr_type;
declare variable kcp_s_perc_saldo curr_type;
declare variable kcp_s_perc_calced curr_type;
declare variable kcp_s_perc_payed curr_type;
declare variable kcp_s_perc_writeoff curr_type;
declare variable kcp_s_perc_begin_saldo curr_type;
declare variable kcp_s_perc_transfer_payed curr_type;
declare variable kcp_s_perc_transfer_writeoff curr_type;
declare variable kcp_s_kommis_calced curr_type;
declare variable kcp_s_kommis_payed curr_type;
declare variable kcp_s_kommis_writeoff curr_type;
declare variable kcp_s_kommis_saldo curr_type;
declare variable kcp_s_kommis_begin_saldo curr_type;
declare variable kcp_s_kommis_transfer_payed curr_type;
declare variable kcp_s_kommis_transfer_writeoff curr_type;
declare variable kcp_s_out_perc_begin_saldo curr_type;
declare variable kcp_s_out_perc_calced curr_type;
declare variable kcp_s_out_perc_saldo curr_type;
declare variable kc_amount curr_type;
declare variable n_pos integer;
declare variable is_last integer;
declare variable koef_loan double precision;
declare variable koef_kommis double precision;
declare variable date_begin date;
declare variable date_end date;
declare variable kcp_period_prv integer;
declare variable tmp_list d_str_255;
declare variable loan_amount curr_type;
declare variable loan_kommis curr_type;
declare variable kc_kommis curr_type;
declare variable tmp_kommis_list d_str_255;
declare variable client_count integer;
declare variable oks_bal_zad_perc curr_type;
declare variable gr_bal_zad_perc curr_type;
begin
-- процедура делит проценты за указанный месяца на уровне участников
-- список id займов (kc_id) передается через переменную LOAN_KC_LIST
-- список сумм займов (kc_amount_approved) передается через переменную LOAN_AMOUNT_LIST

  date_begin = cast('01.' || right(kcp_period, 2) || '.' || left(kcp_period, 4) as date);
  date_end = dateadd(month, 1, date_begin) - 1;
  kcp_period_prv = extract(year from (date_begin - 1)) * 100 + extract(month from (date_begin - 1));

  if (coalesce(loan_kommis_list, '') = '') then
    loan_kommis_list = loan_amount_list;

  client_count = 0;
  loan_amount = 0;
  tmp_list = loan_amount_list;
  while (tmp_list <> '') do
  begin
    n_pos = position(',', tmp_list);
    kc_amount = 0;
    if (n_pos > 0) then
    begin
      kc_amount = left(tmp_list, n_pos - 1);
      tmp_list = substring(tmp_list from n_pos + 1);
    end
    else
    begin
      kc_amount = tmp_list;
      tmp_list = '';
    end
    loan_amount = loan_amount + kc_amount;
    client_count = client_count + 1;
  end

  loan_kommis = 0;
  tmp_list = loan_kommis_list;
  while (tmp_list <> '') do
  begin
    n_pos = position(',', tmp_list);
    kc_amount = 0;
    if (n_pos > 0) then
    begin
      kc_amount = left(tmp_list, n_pos - 1);
      tmp_list = substring(tmp_list from n_pos + 1);
    end
    else
    begin
      kc_amount = tmp_list;
      tmp_list = '';
    end
    loan_kommis = loan_kommis + kc_amount;
  end

  for select kcp_from_reg_id, kcp_perc_saldo, kcp_transfer_reg_id, kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff,
             kcp_perc_begin_saldo, kcp_perc_transfer_payed, kcp_perc_transfer_writeoff, kcp_kommis_calced,
             kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo, kcp_kommis_begin_saldo, kcp_kommis_transfer_payed,
             kcp_kommis_transfer_writeoff, kcp_perc_account, kcp_kommis_account, kcp_loan_is_closed,
             kcp_out_perc_begin_saldo, kcp_out_perc_calced, kcp_out_perc_saldo
      from kontcalcperc
      where kcp_from_k_id = :k_id and
            kcp_period = :kcp_period
      into :kcp_from_reg_id, :kcp_perc_saldo, :kcp_transfer_reg_id, :kcp_perc_calced, :kcp_perc_payed,
           :kcp_perc_writeoff, :kcp_perc_begin_saldo, :kcp_perc_transfer_payed, :kcp_perc_transfer_writeoff,
           :kcp_kommis_calced, :kcp_kommis_payed, :kcp_kommis_writeoff, :kcp_kommis_saldo, :kcp_kommis_begin_saldo,
           :kcp_kommis_transfer_payed, :kcp_kommis_transfer_writeoff, :kcp_perc_account, :kcp_kommis_account,
           :kcp_loan_is_closed, :kcp_out_perc_begin_saldo, :kcp_out_perc_calced, :kcp_out_perc_saldo
  do
  begin
    gr_bal_zad_perc = 0;
    if (kcp_perc_saldo <> 0) then
    begin
      for select kc_id
          from knclient
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0
          into :kc_id
      do
      begin
        oks_bal_zad_perc = 0;
        select oks_bal_zad_perc
        from kn_oksummary
        where oks_from_kc_id = :kc_id and
              oks_date_begin <= :date_end
        order by oks_from_kc_id desc, oks_date_begin desc
        rows 1
        into :oks_bal_zad_perc;
        if (oks_bal_zad_perc is null) then
          oks_bal_zad_perc = 0;
        gr_bal_zad_perc = gr_bal_zad_perc + oks_bal_zad_perc;
      end
    end

    tmp_kc_list = loan_kc_list;
    tmp_amount_list = loan_amount_list;
    tmp_kommis_list = loan_kommis_list;

    kcp_perc_begin_saldo = kcp_perc_begin_saldo - diff_kcp_perc_begin_saldo;
    kcp_kommis_begin_saldo = kcp_kommis_begin_saldo - diff_kcp_kommis_begin_saldo;
    kcp_out_perc_begin_saldo = kcp_out_perc_begin_saldo - diff_kcp_out_perc_begin_saldo;
    kcp_perc_calced = kcp_perc_calced - diff_kcp_perc_calced;
    kcp_kommis_calced = kcp_kommis_calced - diff_kcp_kommis_calced;
    kcp_out_perc_calced = kcp_out_perc_calced - diff_kcp_out_perc_calced;

    cl_kcp_from_reg_id = kcp_from_reg_id;
    cl_kcp_transfer_reg_id = kcp_transfer_reg_id;
    cl_kcp_perc_account = kcp_perc_account;
    cl_kcp_kommis_account = kcp_kommis_account;
    cl_kcp_loan_is_closed = kcp_loan_is_closed;

    kcp_s_perc_saldo = kcp_perc_saldo;
    kcp_s_perc_calced = kcp_perc_calced;
    kcp_s_perc_payed = kcp_perc_payed;
    kcp_s_perc_writeoff = kcp_perc_writeoff;
    kcp_s_perc_begin_saldo = kcp_perc_begin_saldo;
    kcp_s_perc_transfer_payed = kcp_perc_transfer_payed;
    kcp_s_perc_transfer_writeoff = kcp_perc_transfer_writeoff;
    kcp_s_kommis_calced = kcp_kommis_calced;
    kcp_s_kommis_payed = kcp_kommis_payed;
    kcp_s_kommis_writeoff = kcp_kommis_writeoff;
    kcp_s_kommis_saldo = kcp_kommis_saldo;
    kcp_s_kommis_begin_saldo = kcp_kommis_begin_saldo;
    kcp_s_kommis_transfer_payed = kcp_kommis_transfer_payed;
    kcp_s_kommis_transfer_writeoff = kcp_kommis_transfer_writeoff;
    kcp_s_out_perc_begin_saldo = kcp_out_perc_begin_saldo;
    kcp_s_out_perc_calced = kcp_out_perc_calced;
    kcp_s_out_perc_saldo = kcp_out_perc_saldo;

    while (tmp_kc_list <> '') do
    begin
      is_last = 0;
      n_pos = position(',', tmp_kc_list);
      if (n_pos > 0) then
      begin
        kc_id = left(tmp_kc_list, n_pos - 1);
        tmp_kc_list = substring(tmp_kc_list from n_pos + 1);
      end
      else
      begin
        kc_id = tmp_kc_list;
        tmp_kc_list = '';
        is_last = 1;
      end

      n_pos = position(',', tmp_amount_list);
      if (n_pos > 0) then
      begin
        kc_amount = left(tmp_amount_list, n_pos - 1);
        tmp_amount_list = substring(tmp_amount_list from n_pos + 1);
      end
      else
      begin
        kc_amount = tmp_amount_list;
        tmp_amount_list = '';
      end

      n_pos = position(',', tmp_kommis_list);
      if (n_pos > 0) then
      begin
        kc_kommis = left(tmp_kommis_list, n_pos - 1);
        tmp_kommis_list = substring(tmp_kommis_list from n_pos + 1);
      end
      else
      begin
        kc_kommis = tmp_kommis_list;
        tmp_kommis_list = '';
      end

-- движения берется из фактических данных
      cl_kcp_perc_payed = 0;
      cl_kcp_perc_writeoff = 0;
      cl_kcp_kommis_payed = 0;
      cl_kcp_perc_transfer_payed = 0;
      cl_kcp_perc_transfer_writeoff = 0;
      cl_kcp_kommis_transfer_payed = 0;

      select sum(iif(ok_type = 4, -ok_perc, 0)), sum(iif(ok_type in (5, 6), -ok_perc, 0)),
             sum(iif(ok_type = 4, -ok_kommis, 0)),
             sum(iif(ok_type = 4 and :kcp_perc_transfer_payed <> 0 and :kcp_from_reg_id <> :kcp_transfer_reg_id and ok_ol_reg_id = :cl_kcp_transfer_reg_id, -ok_perc, 0)),
             sum(iif(ok_type in (5, 6) and :kcp_perc_transfer_payed <> 0 and :kcp_from_reg_id <> :kcp_transfer_reg_id and ok_ol_reg_id = :cl_kcp_transfer_reg_id, -ok_perc, 0)),
             sum(iif(ok_type = 4 and :kcp_kommis_transfer_payed <> 0 and :kcp_from_reg_id <> :kcp_transfer_reg_id and ok_ol_reg_id = :cl_kcp_transfer_reg_id, -ok_kommis, 0))
      from kn_operkred
      where ok_from_kc_id = :kc_id and
            ok_ol_date between :date_begin and :date_end and
            ok_ol_is_active = 1 and
            ok_type in (4, 5, 6)
      into :cl_kcp_perc_payed, :cl_kcp_perc_writeoff, :cl_kcp_kommis_payed, :cl_kcp_perc_transfer_payed,
           :cl_kcp_perc_transfer_writeoff, :cl_kcp_kommis_transfer_payed;

      if (cl_kcp_perc_payed is null) then
        cl_kcp_perc_payed = 0;
      if (cl_kcp_perc_writeoff is null) then
        cl_kcp_perc_writeoff = 0;
      if (cl_kcp_kommis_payed is null) then
        cl_kcp_kommis_payed = 0;
      if (cl_kcp_perc_transfer_payed is null) then
        cl_kcp_perc_transfer_payed = 0;
      if (cl_kcp_perc_transfer_writeoff is null) then
        cl_kcp_perc_transfer_writeoff = 0;
      if (cl_kcp_kommis_transfer_payed is null) then
        cl_kcp_kommis_transfer_payed = 0;

-- получение начального сальдо из конечного сальдо предыдущего месяца
      cl_kcp_perc_begin_saldo = 0;
      cl_kcp_kommis_begin_saldo = 0;
      cl_kcp_out_perc_begin_saldo = 0;
      select kcp_perc_saldo, kcp_kommis_saldo, kcp_out_perc_saldo
      from kn_kontcalcperc
      where kcp_period = :kcp_period_prv and
            kcp_from_kc_id = :kc_id
      into :cl_kcp_perc_begin_saldo, cl_kcp_kommis_begin_saldo, cl_kcp_out_perc_begin_saldo;

      if (cl_kcp_perc_begin_saldo is null) then
        cl_kcp_perc_begin_saldo = 0;
      if (cl_kcp_kommis_begin_saldo is null) then
        cl_kcp_kommis_begin_saldo = 0;
      if (cl_kcp_out_perc_begin_saldo is null) then
        cl_kcp_out_perc_begin_saldo = 0;

      if (is_last = 0) then
      begin
        koef_loan = kc_amount / loan_amount;
        if (loan_kommis <> 0) then
          koef_kommis = kc_kommis / loan_kommis;
        else
          koef_kommis = 0;

        if (gr_bal_zad_perc = kcp_perc_saldo and
            kcp_perc_saldo <> 0) then
        begin
-- если общая сумма остатка накопленных %% равна задолженности по процентам по графику
-- то участникам передаем сумму по графику
          cl_kcp_perc_saldo = 0;
          select oks_bal_zad_perc
          from kn_oksummary
          where oks_from_kc_id = :kc_id and
                oks_date_begin <= :date_end
          order by oks_from_kc_id desc, oks_date_begin desc
          rows 1
          into :cl_kcp_perc_saldo;
          if (cl_kcp_perc_saldo is null) then
            cl_kcp_perc_saldo = 0;
        end
        else
        begin
-- деление пропорционально
          cl_kcp_perc_saldo = round(kcp_perc_saldo * koef_loan, 0);
        end

        if (kcp_perc_saldo <> 0 and
            cl_kcp_perc_saldo = 0) then
        begin
          if (kcp_perc_saldo >= client_count) then
            cl_kcp_perc_saldo = 1.00;
          else
            cl_kcp_perc_saldo = round(kcp_perc_saldo * koef_loan, 2);
        end
        if (cl_kcp_perc_saldo > kcp_s_perc_saldo and
            kcp_s_perc_saldo >= 0) then
          cl_kcp_perc_saldo = kcp_s_perc_saldo;

        cl_kcp_out_perc_saldo = round(kcp_out_perc_saldo * koef_loan, 0);

        cl_kcp_kommis_saldo = round(kcp_kommis_saldo * koef_kommis, 2);
        cl_kcp_kommis_writeoff = round(kcp_kommis_writeoff * koef_kommis, 2);
        cl_kcp_kommis_transfer_writeoff = round(kcp_kommis_transfer_writeoff * koef_kommis, 2);

        if (cl_kcp_kommis_saldo > kcp_s_kommis_saldo and
            kcp_s_kommis_saldo >= 0) then
          cl_kcp_kommis_saldo = kcp_s_kommis_saldo;

        if (cl_kcp_out_perc_saldo > kcp_s_out_perc_saldo and
            kcp_s_out_perc_saldo >= 0) then
          cl_kcp_out_perc_saldo = kcp_s_out_perc_saldo;

-- вычисление соответствующей суммы начисления
        cl_kcp_perc_calced = cl_kcp_perc_saldo + cl_kcp_perc_payed + cl_kcp_perc_writeoff - cl_kcp_perc_begin_saldo;

        cl_kcp_kommis_calced = cl_kcp_kommis_saldo - cl_kcp_kommis_payed - cl_kcp_kommis_writeoff - cl_kcp_kommis_begin_saldo;

        cl_kcp_out_perc_calced = cl_kcp_out_perc_saldo - cl_kcp_out_perc_begin_saldo;

        if (kcp_out_perc_saldo > 0) then
        begin
          if (abs(cl_kcp_out_perc_calced) <= 2 and
              cl_kcp_out_perc_begin_saldo >= 0) then
          begin
            cl_kcp_out_perc_saldo = cl_kcp_out_perc_begin_saldo;
            cl_kcp_out_perc_calced = 0;
          end
          else
          if (abs(cl_kcp_out_perc_saldo) <= 2) then
          begin
            cl_kcp_out_perc_calced = -cl_kcp_out_perc_begin_saldo;
            cl_kcp_out_perc_saldo = 0;
          end
        end

-- вычисление нераспределенную часть.
-- будет присвоен последнему займу
        kcp_s_perc_saldo = kcp_s_perc_saldo - cl_kcp_perc_saldo;
        kcp_s_perc_calced = kcp_s_perc_calced - cl_kcp_perc_calced;
        kcp_s_perc_payed = kcp_s_perc_payed - cl_kcp_perc_payed;
        kcp_s_perc_writeoff = kcp_s_perc_writeoff - cl_kcp_perc_writeoff;
        kcp_s_perc_begin_saldo = kcp_s_perc_begin_saldo - cl_kcp_perc_begin_saldo;
        kcp_s_perc_transfer_payed = kcp_s_perc_transfer_payed - cl_kcp_perc_transfer_payed;
        kcp_s_perc_transfer_writeoff = kcp_s_perc_transfer_writeoff - cl_kcp_perc_transfer_writeoff;
        kcp_s_kommis_calced = kcp_s_kommis_calced - cl_kcp_kommis_calced;
        kcp_s_kommis_payed = kcp_s_kommis_payed - cl_kcp_kommis_payed;
        kcp_s_kommis_writeoff = kcp_s_kommis_writeoff - cl_kcp_kommis_writeoff;
        kcp_s_kommis_saldo = kcp_s_kommis_saldo - cl_kcp_kommis_saldo;
        kcp_s_kommis_begin_saldo = kcp_s_kommis_begin_saldo - cl_kcp_kommis_begin_saldo;
        kcp_s_kommis_transfer_payed = kcp_s_kommis_transfer_payed - cl_kcp_kommis_transfer_payed;
        kcp_s_kommis_transfer_writeoff = kcp_s_kommis_transfer_writeoff - cl_kcp_kommis_transfer_writeoff;
        kcp_s_out_perc_begin_saldo = kcp_s_out_perc_begin_saldo - cl_kcp_out_perc_begin_saldo;
        kcp_s_out_perc_calced = kcp_s_out_perc_calced - cl_kcp_out_perc_calced;
        kcp_s_out_perc_saldo = kcp_s_out_perc_saldo - cl_kcp_out_perc_saldo;
      end
      else
      begin
-- присвоение нераспределенного остатка последнему участнику
        cl_kcp_perc_saldo = kcp_s_perc_saldo;
        cl_kcp_kommis_writeoff = kcp_s_kommis_writeoff;
        cl_kcp_kommis_saldo = kcp_s_kommis_saldo;
        cl_kcp_kommis_transfer_writeoff = kcp_s_kommis_transfer_writeoff;
        cl_kcp_out_perc_saldo = kcp_s_out_perc_saldo;

-- получение начального сальдо из конечного сальдо предыдущего месяца
-- вычисление соответствующей суммы начисления
        cl_kcp_perc_calced = cl_kcp_perc_saldo + cl_kcp_perc_payed + cl_kcp_perc_writeoff - cl_kcp_perc_begin_saldo;

        cl_kcp_kommis_calced = cl_kcp_kommis_saldo - cl_kcp_kommis_payed - cl_kcp_kommis_writeoff - cl_kcp_kommis_begin_saldo;

        cl_kcp_out_perc_calced = cl_kcp_out_perc_saldo - cl_kcp_out_perc_begin_saldo;
      end

      suspend;
    end
  end
