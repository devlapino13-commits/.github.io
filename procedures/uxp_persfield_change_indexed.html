CREATE PROCEDURE UXP$PERSFIELD_CHANGE_INDEXED(
    PIF_SOURCE D_STR_CODE NOT NULL,
    PIF_FIELD_NAME D_STR_CODE NOT NULL,
    IS_INDEXED D_BOOL NOT NULL)
RETURNS (
    PIF_UUID D_UUID)
AS
declare variable pif_indexed d_bool;
declare variable pva_id d_uuid;
begin
  select pif_uuid, pif_indexed
  from ux$persfield
  where pif_source = :pif_source and
        pif_field_name = :pif_field_name
  into pif_uuid, pif_indexed;

  if (row_count = 0) then
    exception e_default 'Field not found in ux$persfield';

  if (pif_indexed = is_indexed) then
    exception e_message 'PIF_INDEXED идентична с IS_INDEXED';

  if (is_indexed = 0) then
  begin
    insert into persvalidx (pvx_pva_id, pvx_value)
    select pvs_pva_id, pvs_value
    from persdata, persvalue, persvalstr
    where pda_from_pif_uuid = :pif_uuid and
          pva_from_pda_id = pda_id and
          pvs_pva_id = pva_id;

    if (row_count > 0) then
    begin
      for select pva_id
      from persdata, persvalue
      where pda_from_pif_uuid = :pif_uuid and
            pva_from_pda_id = pda_id
      into pva_id
      do
      begin
        delete from persvalstr
        where pvs_pva_id = :pva_id;
      end
    end
  end else
  begin
    insert into persvalstr (pvs_pva_id, pvs_value)
    select pvx_pva_id, pvx_value
    from persdata, persvalue, persvalidx
    where pda_from_pif_uuid = :pif_uuid and
          pva_from_pda_id = pda_id and
          pvx_pva_id = pva_id;

    if (row_count > 0) then
    begin
      for select pva_id
      from persdata, persvalue
      where pda_from_pif_uuid = :pif_uuid and
            pva_from_pda_id = pda_id
      into pva_id
      do
      begin
        delete from persvalidx
        where pvx_pva_id = :pva_id;
      end
    end
  end

  update ux$persfield
  set pif_indexed = :is_indexed
  where pif_uuid = :pif_uuid;
end
