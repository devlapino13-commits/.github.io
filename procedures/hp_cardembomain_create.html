CREATE PROCEDURE HP_CARDEMBOMAIN_CREATE(
    CARD_KIND D_INTEGER NOT NULL,
    CARD_COUNT D_INTEGER NOT NULL,
    IS_PINCOVER D_BOOL NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    CEM_ID D_INTEGER,
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable status_id d_integer;
declare variable card_no d_integer;
begin
  err_code = 0;

  select ces_id
  from cardembostatus
  where ces_code = 'new'
  into status_id;

  if (status_id is null) then
  begin
    err_code = 1;
    err_text = 'cardembostatus not found: ''new''';
    exit;
  end

  insert into cardembomain (cem_is_pincover, cem_app_count, cem_from_ckd_id, cem_from_ces_id, cem_from_un_id)
  values (:is_pincover, :card_count, :card_kind, :status_id, :un_id)
  returning cem_id
  into :cem_id;

  card_no = 0;
  while (card_no < card_count) do
  begin
    card_no = card_no + 1;
    insert into cardlist (cal_from_cem_id, cal_last_cpl_id)
    values (:cem_id, 0);
  end
  err_text = 'Заявка № ' || cem_id || ' успешно создана, количество : ' || card_no;

  suspend;
end
