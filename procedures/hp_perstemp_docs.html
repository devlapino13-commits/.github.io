CREATE PROCEDURE HP_PERSTEMP_DOCS(
    IN_C_ID D_ID,
    IN_SHOW_INACTIVE D_BOOL)
RETURNS (
    CDS_FROM_C_ID D_ID,
    CDS_ID D_INTEGER,
    CDS_FROM_PK_ID D_INTEGER,
    CDS_DOC_NO D_STR_CODE,
    CDS_DOC_DATE_BEGIN D_DATE,
    CDS_DOC_DATE_END D_DATE,
    CDS_DOC_ISSUED_BY D_STR_255,
    CDS_DOC_FOR_COMPANY D_STR_255,
    CDS_DOC_TERRITORY D_STR_255,
    CDS_DOC_BLOB_1 D_ID,
    CDS_DOC_BLOB_2 D_ID,
    CDS_FROM_UN_ID D_ID,
    CDS_DATETIME D_DATETIME,
    CDS_IS_DELETE D_SMALLINT,
    CDS_VERIFY_STATUS D_SMALLINT,
    CDS_VERIFY_DATETIME D_DATETIME,
    CDS_VERIFY_UN_ID D_ID,
    CDS_VERIFY_CC_DATETIME D_DATETIME,
    CDS_VERIFY_CC_UN_ID D_ID,
    VERIFY_CC_STATUS D_BOOL,
    CDS_MEMO_1 D_STR_255,
    CDS_MEMO_2 D_STR_255,
    CDS_MEMO_3 D_STR_255,
    CDS_MEMO_4 D_STR_255,
    CDS_MEMO_5 D_STR_255,
    PDT_IS_MAIN_DOC D_SMALLINT,
    CDS_IS_OLD_DOC D_SMALLINT,
    PK_NAME D_STR_255,
    CDS_IS_MAIN_DOC D_SMALLINT,
    CDS_FROM_DRL_ID D_ID,
    EDIT_KIND D_INTEGER,
    PDT_NEED_DATE_BEGIN D_SMALLINT,
    PDT_NEED_DATE_END D_SMALLINT,
    CDS_CREATE_DATETIME D_DATETIME,
    BLOB_ADD_KIND D_STR_100,
    SHOW_IN_KK D_SMALLINT,
    CDS_HAS_BLOB_1 D_BOOL,
    CDS_HAS_BLOB_2 D_BOOL)
AS
begin
  if (in_show_inactive is null) then
    in_show_inactive = 0;

  for select cds_from_c_id, cds_id, cds_from_pk_id, cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_doc_issued_by,
             cds_doc_for_company, cds_doc_territory, cds_doc_blob_1, cds_doc_blob_2, cds_from_un_id, cds_datetime,
             cds_is_delete, cds_verify_status, cds_verify_datetime, cds_verify_un_id, cds_verify_cc_datetime,
             cds_verify_cc_un_id, cds_memo_1, cds_memo_2, cds_memo_3, cds_memo_4, cds_memo_5, pdt_is_main_doc,
             cds_is_old_doc, pk_name, cds_is_main_doc, cds_from_drl_id, edit_kind, pdt_need_date_begin, pdt_need_date_end,
             cds_create_datetime, blob_add_kind, show_in_kk, verify_cc_status
  from aarep_clientdocs_list(:in_c_id)
  order by cds_is_old_doc, cds_id
  into :cds_from_c_id, :cds_id, :cds_from_pk_id, :cds_doc_no, :cds_doc_date_begin, :cds_doc_date_end,
       :cds_doc_issued_by, :cds_doc_for_company, :cds_doc_territory, :cds_doc_blob_1, :cds_doc_blob_2,
       :cds_from_un_id, :cds_datetime, :cds_is_delete, :cds_verify_status, :cds_verify_datetime, :cds_verify_un_id,
       :cds_verify_cc_datetime, :cds_verify_cc_un_id, :cds_memo_1, :cds_memo_2, :cds_memo_3, :cds_memo_4, :cds_memo_5,
       :pdt_is_main_doc, :cds_is_old_doc, :pk_name, :cds_is_main_doc, :cds_from_drl_id, :edit_kind,
       :pdt_need_date_begin, :pdt_need_date_end, :cds_create_datetime, :blob_add_kind, :show_in_kk, :verify_cc_status
  do
  begin
    cds_has_blob_1 = iif(coalesce(cds_doc_blob_1, 0) <> 0, 1, 0);
    cds_has_blob_2 = iif(coalesce(cds_doc_blob_2, 0) <> 0, 1, 0);

    if (in_show_inactive = 0 and cds_is_old_doc = 1) then
      Continue;

    suspend;
  end
end
