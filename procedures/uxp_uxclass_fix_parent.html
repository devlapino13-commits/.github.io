CREATE PROCEDURE UXP$UXCLASS_FIX_PARENT(
    CLASS_PATH D_STR_4096)
AS
declare variable cls_parent d_str_code;
declare variable cls_name d_str_code;
declare variable cls_visual d_smallint;
declare variable last_class d_str_code;
begin
  last_class = '';
  for select res_str
  from uxp$split_str(:class_path, ';')
  into :cls_name
  do
    last_class = cls_name;

  cls_parent = null;
  for select res_str
  from uxp$split_str(:class_path, ';')
  into :cls_name
  do
  begin
    cls_visual = -2;

    select cls_visual
    from ux$class
    where cls_name = :cls_name
    into cls_visual;

    if (row_count = 0) then
    begin
      if (cls_name = last_class) then
        cls_visual = -1;

      insert into ux$class (cls_name, cls_parent, cls_visual)
      values (:cls_name, :cls_parent, :cls_visual);
    end else
    begin
      if (cls_name = last_class and cls_visual = -2) then
      begin
        update ux$class
        set cls_visual = -1
        where cls_name = :cls_name;
      end

      update ux$class
      set cls_parent = :cls_parent
      where cls_name = :cls_name;
    end

    cls_parent = cls_name;
  end
end
