CREATE PROCEDURE JIRA_UPDATE_PARAMS(
    PROGRAMMER_FIO D_STR_100,
    TASK_TYPE D_STR_100,
    TASK_LEVEL_INP D_SMALLINT,
    TASK_ID D_INTEGER = 0)
RETURNS (
    TASK_STACK D_STR_CODE,
    TASK_LEVEL D_SMALLINT)
AS
begin
    task_level = task_level_inp;
    task_stack = 'База данных';

    task_stack =
    case
        when programmer_fio in ('Антон Ицкович', 'Владимир Цымбалюк') then 'Сервисы Python'
        when programmer_fio = 'Тилек Сулайманбеков' then 'КМП Swift/Kotlin'
        when programmer_fio = 'Анатолий �©егольский' then 'Сервисы МП / FsService'
        when programmer_fio = 'Анищук Вячеслав' then 'КМП Flutter'
        when programmer_fio = 'Оксана �©егольская' then 'МП Swift/Kotlin'
        else task_stack
    end;
    
    if (task_type = 'техническая оценка') then
    begin
        task_stack = 'Техническая оценка';
        task_level = 3;
    end

    if (task_type = 'Аутсорс') then
    begin
        task_stack = 'КМП Swift/Kotlin';
        task_level = 3;
    end

    if (exists (select * from jira_delphi jd where jd.jd_issue_id = :task_id)) then
        task_stack = 'Delphi';

    suspend;
end
