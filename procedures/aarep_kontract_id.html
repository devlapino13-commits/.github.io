AAREP_KONTRACT_ID

Возвращает список контрактов / Параметр Report_ID: / 1 - Список заявок/договоров в процессе (на дату) / 2 - Выданные кредиты (за период) / 3 - Список активных контрактов (на дату) / 4 - Список закрытых контрактов (за период) / 5 - Просроченные кредиты (на дату) / 6 - Планируемые погашения (за период) / 7 - Контракты с депозитами (на дату) / 8 - Планируемые погашения на дату с депозитами / 9 - Планируемые погашения и просроченные на дату с депозитами

CREATE PROCEDURE AAREP_KONTRACT_ID(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3),
    REPORT_ID INTEGER)
RETURNS (
    AAREP_K_ID DECIMAL(18,0),
    AAREP_K_BALANCE DECIMAL(18,4),
    AAREP_K_CLOSED_DATE DATE,
    AAREP_K_DELIQ_OS DECIMAL(18,4),
    AAREP_K_DELIQ_PERC DECIMAL(18,4),
    AAREP_K_DELIQ_DAYS INTEGER,
    AAREP_K_DELIQ_REPAYS VARCHAR(10),
    AAREP_K_DELIQ_BEGIN_DATE DATE,
    AAREP_DATE_ACTION DATE,
    AAREP_PLAN_OS DECIMAL(18,4),
    AAREP_PLAN_PERC DECIMAL(18,4),
    AAREP_PLAN_ALL DECIMAL(18,4),
    AAREP_PLAN_REPAY_NO INTEGER,
    AAREP_PLAN_REPAY_DATE DATE,
    AAREP_DEPOSIT DECIMAL(18,4),
    AAREP_ZALOG DECIMAL(18,4),
    AAREP_ZALOG_PLUS DECIMAL(18,4),
    AAREP_ZALOG_MINUS DECIMAL(18,4),
    AAREP_ZALOG_NAME VARCHAR(100),
    AAREP_COLL_PLAN_OS DECIMAL(18,4),
    AAREP_COLL_FACT_OS DECIMAL(18,4),
    AAREP_COLL_KOEF_PERC DECIMAL(18,7),
    AAREP_KOLICH_ACTIVE INTEGER,
    AAREP_REGION_ID INTEGER,
    AAREP_DISB_EKV DECIMAL(18,4),
    AAREP_DELIQ_ALL DECIMAL(18,4),
    AAREP_IS_RESTRUCTED INTEGER,
    AAREP_CLOSED_LO DECIMAL(18,0),
    AAREP_DISB_LO VARCHAR(100),
    AAREP_KF_ID DECIMAL(18,0),
    OL_ID D_ID,
    AAREP_DISB_LO_C_ID D_ID)
AS
declare variable dd date;
declare variable okz_is_money integer;
declare variable temp_sum decimal(18,4);
declare variable curr_code varchar(3);
declare variable curr_rate decimal(18,4);
declare variable plus1 decimal(18,4);
declare variable minus1 decimal(18,4);
declare variable cr_rate decimal(18,4);
declare variable can_suspend integer;
declare variable curr_prior varchar(3);
declare variable def_curr varchar(3);
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate decimal(18,4);
declare variable pr_set_rate_in_kont integer;
declare variable dk_disb_date date;
declare variable st_staff integer;
declare variable st_client integer;
declare variable dk_disb_kc_id decimal(18,0);
declare variable dk_disb_kc_id_prior decimal(18,0);
declare variable db_id integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  aarep_disb_lo = '';

  select cm_default_currency
  from company
  into def_curr;
  :db_id = f_get_db_id();

  aarep_is_restructed = 0;
  aarep_deliq_all = 0;
  aarep_disb_ekv = 0;
  curr_prior = '000';
  curr_rate = 1;
  aarep_kolich_active = 0;
  aarep_k_id = 0;
  aarep_k_balance = 0;
  aarep_k_closed_date = null;
  aarep_plan_repay_date = null;
  aarep_k_deliq_begin_date = null;
  aarep_k_deliq_os = 0;
  aarep_k_deliq_perc = 0;
  aarep_k_deliq_days = 0;
  aarep_k_deliq_repays = '';
  aarep_date_action = report_date_end;
  aarep_coll_plan_os = 0;
  aarep_coll_fact_os = 0;
  aarep_coll_koef_perc = 0;
  aarep_plan_perc = 0;
  aarep_plan_os = 0;
  aarep_plan_perc = 0;
  aarep_plan_repay_no = 0;
  aarep_plan_all = 0;
  aarep_deposit = 0;
  aarep_zalog = 0;
  aarep_zalog_plus = 0;
  aarep_zalog_minus = 0;
  aarep_zalog_name = '';

  if (report_id = 1) then
  /* Заявки, контракты в процессе*/
  begin
    for select distinct ks_from_k_id, kal_new_reg_id, k_from_kf_id
        from kontstate, kontract, knaktlist, region_list(:report_region_id, 1, -1, 1)
        where ks_ol_is_delete = 0 and
              cast(ks_date_begin as date) <= :report_date_end and
              ks_date_end >= :report_date_end and
              ks_state = 0 /* в процессе */
              and
              k_status >= 0 /* не аннулированные */
              and
              k_id = ks_from_k_id and
              ((:report_currency = '000') or
              (:report_currency = '---') or
              (k_currency = :report_currency)) and
              kal_from_kf_id = k_from_kf_id and
              kal_date <= :report_date_end and
              kal_date_end >= :report_date_end and
              (kal_new_reg_id = id_list) and
              knaktlist.is_delete = 0
        into :aarep_k_id, :aarep_region_id, aarep_kf_id
    do
    begin
      suspend;
    end
  end

  if (report_id = 2) then
  /* Выданные кредиты */
  begin
    for select ks_region_id, ks_from_k_id, ks_date_begin, k_amount_approved, k_amount_approved * k_index_curr_rate,
               k_from_kf_id, ks_from_ol_id
        from kontstate, kontract, region_list(:report_region_id, 1, 3, 1)
        where k_id = ks_from_k_id and
              ks_date_begin between :report_date_begin and :report_date_end and
              ks_state = 1 and
              ks_region_id = id_list and
              ks_ol_is_delete = 0
        into :aarep_region_id, :aarep_k_id, :aarep_date_action, :aarep_plan_os, :temp_sum, aarep_kf_id, ol_id
    do
    begin
      if (report_currency = '000') then
        aarep_plan_os = cast(:temp_sum as decimal(18,2));

      -- RFCRU-5470: Заполняем поле "Выдавший КС" в отчете "Клиенты выданных кредитов (для фин.отд)"
      if (db_id = 81001) then
      begin
        select c.c_name_fio, c.c_id
        from kontract k
        inner join clients c on k.k_fact_disb_lo_id = c.c_id
        where k.k_id = :aarep_k_id
        into :aarep_disb_lo, :aarep_disb_lo_c_id;
      end

      suspend;
    end
  end

  if (report_id = 3 or report_id = 20) then
  /* Активные контракты */
  begin
    for select ac_k_id, ac_disb, ac_k_balance, ac_client_kolich, ac_region_id, ac_kf_id
        from aarep_kontract_active_temp(:report_region_id, :report_date_end, :report_currency, :report_reverse)
        into :aarep_k_id, :aarep_disb_ekv, :aarep_k_balance, :aarep_kolich_active, :aarep_region_id, aarep_kf_id
    do
    begin
      aarep_date_action = :report_date_end;
      if (aarep_k_balance > 0) then
      begin
        aarep_k_deliq_days = 0;
        select :report_date_end - oks_delinq_date
        from oksummary oks
        where :report_date_end between oks_date_begin and oks_date_end and
              oks_from_k_id = :aarep_k_id and
              oks_close_state = 0 and
              oks_delinq_date < :report_date_end
        into aarep_k_deliq_days;
        aarep_k_deliq_days = coalesce(aarep_k_deliq_days, 0);

        -- RFCRU-5470: Заполняем поле "Выдавший КС" в отчете "Клиенты активных кредитов"
        if (report_id = 3 and db_id = 81001) then
        begin
          select c.c_name_fio, c.c_id
          from kontract k
          inner join clients c on k.k_fact_disb_lo_id = c.c_id
          where k.k_id = :aarep_k_id
          into :aarep_disb_lo, :aarep_disb_lo_c_id;
        end

        suspend;
      end

    end
  end

  aarep_disb_ekv = 0;
  if (report_id = 4) then
  /* Закрытые контракты за период*/
  begin
    for select distinct ks_from_k_id, ks_date_begin, ks_region_id, ks_kc_id, k_from_kf_id
        from kontstate, region_list(:report_region_id, 1, -1, 1), kontract
        where ks_ol_is_delete = 0 and
              ks_date_begin between :report_date_begin and :report_date_end and
              ks_state = 2 /* Закрытые */
              and
              ks_region_id = id_list and
              ks_from_k_id = k_id
        into :aarep_k_id, :aarep_k_closed_date, :aarep_region_id, :aarep_closed_lo, aarep_kf_id
    do
    begin
      aarep_date_action = :aarep_k_closed_date;
      suspend;
    end
  end

  if (report_id in (5, 18, 19)) then
  /* Просроченные кредиты на дату*/
  begin
    dk_disb_kc_id_prior = 0;
    for select distinct dk_id, dk_kf_id, dk_date, dk_os, dk_perc, dk_days, dk_repays_str, dk_region_id, curr_code,
                        dk_balance_end, dk_disb_date, dk_disb_kc_id
        from aarep_kontract_deliq_temp(:report_region_id, null, :report_date_end, :report_currency)
        where (dk_os + dk_perc) >= 0.01
        into :aarep_k_id, :aarep_kf_id, :aarep_k_deliq_begin_date, :aarep_k_deliq_os, :aarep_k_deliq_perc,
             :aarep_k_deliq_days, :aarep_k_deliq_repays, :aarep_region_id, :curr_code, :aarep_k_balance, :dk_disb_date,
             :dk_disb_kc_id
    do
    begin
      aarep_deliq_all = :aarep_k_deliq_os + :aarep_k_deliq_perc;
      aarep_is_restructed = 0;

      if (dk_disb_kc_id_prior <> dk_disb_kc_id) then
      begin
        aarep_disb_lo = null;
        select c_name_fio
        from clients
        where c_id = :dk_disb_kc_id
        into :aarep_disb_lo;
        if (aarep_disb_lo is null) then
          aarep_disb_lo = 'Не найден';
        dk_disb_kc_id_prior = dk_disb_kc_id;
      end

      if (exists(select *
                 from aarep_get_restructuring(:aarep_k_id, :report_date_end, :db_id)
                 where is_restruct = 1)) then
        aarep_is_restructed = 1;

      aarep_plan_repay_no = 0;
      select max(sh_id)
      from schedule
      where sh_from_k_id = :aarep_k_id and
            sh_date <= :aarep_k_deliq_begin_date
      into aarep_plan_repay_no;
      aarep_plan_repay_no = coalesce(aarep_plan_repay_no, 0);

      suspend;
    end
  end

  if (report_id = 6) then
  /* Погашения по графику */
  begin
    for select distinct k_currency, ks_from_k_id, kal_new_reg_id, k_from_kf_id
        from kontstate, kontract, knaktlist, region_list(:report_region_id, 1, -1, 1)
        where ks_ol_is_delete = 0 and
              ((ks_date_end >= :report_date_begin and
              cast(ks_date_begin as date) <= :report_date_end and
              ks_state = 1) or
              (ks_date_begin = :report_date_begin and
              ks_state = 2)) and
              kal_from_kf_id = k_from_kf_id and
              ks_from_k_id = k_id and
              kal_date <= :report_date_end and
              kal_date_end >= :report_date_end and
              (kal_new_reg_id = id_list) and
              knaktlist.is_delete = 0 and
              ((:report_currency = '000') or
              (:report_currency = '---') or
              (k_currency = :report_currency))
        into :curr_code, :aarep_k_id, :aarep_region_id, aarep_kf_id
    do
    begin
      if (report_currency = '000') then
      begin
        if (curr_prior <> :curr_code) then
        begin
          select cr_rate
          from opercurr
          where cr_from_cu_id = :curr_code and
                cr_date <= :report_date_begin and
                cr_date_end >= :report_date_begin
          into :curr_rate;
          if (curr_rate is null) then
            curr_rate = 1;
          curr_prior = :curr_code;
        end
      end
      for select ok_sh_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
          from operkred
          where ok_from_k_id = :aarep_k_id and
                ok_type < 8 /* Кроме выдач */
                and
                ok_ol_date <= :report_date_end and
                ok_sh_date >= :report_date_begin and
                ok_sh_date <= :report_date_end and
                ok_ol_is_active = 1 and
                ok_from_c_id = 0 and
                ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
          group by ok_sh_id, ok_sh_date
          having(sum(ok_os + ok_sh_os) <> 0) or (sum(ok_perc + ok_sh_perc) <> 0)
          order by ok_sh_date
          into :aarep_plan_repay_no, :aarep_plan_repay_date, :aarep_plan_os, :aarep_plan_perc
      do
      begin
        aarep_date_action = :aarep_plan_repay_date;
        if (aarep_plan_os is null) then
          aarep_plan_os = 0;
        if (aarep_plan_perc is null) then
          aarep_plan_perc = 0;
        aarep_plan_all = aarep_plan_os + aarep_plan_perc;

        if (report_currency = '000') then
        begin
          aarep_plan_os = aarep_plan_os * curr_rate;
          aarep_plan_perc = aarep_plan_perc * curr_rate;
          aarep_plan_all = aarep_plan_all * curr_rate;
        end
        suspend;
      end
    end
  end

  if (report_id = 7) then
  /* Депонированные */
  begin
    for select kal_new_reg_id, k_currency, ks_from_k_id, sum(okd_summa_plus - okd_summa_minus), k_from_kf_id
        from kontstate, kontract, knaktlist, operlist, operkreddepo, region_list(:report_region_id, 1, -1, 1)
        where ks_ol_is_delete = 0 and
              cast(ks_date_begin as date) <= :report_date_end and
              ks_date_end >= :report_date_end and
              k_id = ks_from_k_id and
              ((:report_currency = '000') or
              (:report_currency = '---') or
              (k_currency = :report_currency)) and
              okd_from_k_id = ks_from_k_id and
              ol_id = okd_from_ol_id and
              operlist.is_delete = 0 and
              ol_date <= :report_date_end and
              kal_from_kf_id = k_from_kf_id and
              kal_date <= :report_date_end and
              kal_date_end >= :report_date_end and
              (kal_new_reg_id = id_list) and
              knaktlist.is_delete = 0
        group by kal_new_reg_id, k_currency, ks_from_k_id, k_from_kf_id
        having sum(okd_summa_plus - okd_summa_minus) > 0
        order by k_currency
        into :aarep_region_id, :curr_code, :aarep_k_id, :aarep_deposit, aarep_kf_id
    do
    begin
      if (report_currency = '000') then
      begin
        if (curr_prior <> :curr_code) then
        begin
          select cr_rate
          from opercurr
          where cr_from_cu_id = :curr_code and
                cr_date <= :report_date_end and
                cr_date_end >= :report_date_end
          into :curr_rate;
          if (curr_rate is null) then
            curr_rate = 1;
          curr_prior = :curr_code;
        end
        aarep_deposit = aarep_deposit * curr_rate;
      end
      if (aarep_deposit > 0) then
        suspend;
    end
  end
  if (report_id = 8) then
  /* Депонированные и планируемые */
  begin
    for select aarep_k_id, k_from_kf_id, max(aarep_plan_repay_date), sum(aarep_plan_os), sum(aarep_plan_perc),
               sum(aarep_plan_all), sum(aarep_deposit), max(aarep_date_action)
        from aarep_kontract_id_temp(:report_region_id, :report_date_end, :report_reverse, :report_currency, 0), kontract
        where aarep_k_id = k_id
        group by aarep_k_id, k_from_kf_id
        having sum(aarep_plan_all) <> 0 and sum(aarep_deposit) <> 0
        into :aarep_k_id, :aarep_kf_id, :aarep_plan_repay_date, :aarep_plan_os, :aarep_plan_perc, :aarep_plan_all,
             :aarep_deposit, :aarep_date_action
    do
    begin
      aarep_region_id = 1000000;
      suspend;
    end
  end

  if (report_id = 9) then
  /* Депонированные, планируемые и просроченные */
  begin
    for select aarep_k_id, k_from_kf_id, max(aarep_plan_repay_date), sum(aarep_plan_os), sum(aarep_plan_perc),
               sum(aarep_plan_all), sum(aarep_deposit), max(aarep_date_action)
        from aarep_kontract_id_temp(:report_region_id, :report_date_end, :report_reverse, :report_currency, 1), kontract
        where aarep_k_id = k_id
        group by aarep_k_id, k_from_kf_id
        having sum(aarep_plan_all) <> 0 and sum(aarep_deposit) <> 0
        into :aarep_k_id, aarep_kf_id, :aarep_plan_repay_date, :aarep_plan_os, :aarep_plan_perc, :aarep_plan_all,
             :aarep_deposit, :aarep_date_action
    do
    begin
      aarep_region_id = 1000000;
      suspend;
    end
  end
  if (report_id = 10) then
  /* Погашения по графику (полный список без учета погашений */
  begin
