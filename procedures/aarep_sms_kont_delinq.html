CREATE PROCEDURE AAREP_SMS_KONT_DELINQ(
    REP_DATE DATE,
    REP_DAYS INTEGER)
RETURNS (
    PHONE VARCHAR(20),
    SMS_MESSAGE VARCHAR(255),
    CLIENT_ID DECIMAL(18,0),
    CL_NAME STR_100,
    PUSH_MESSAGE D_STR_LONG,
    KC_ID D_ID)
AS
declare variable DB_ID integer;
declare variable K_ID decimal(18,0);
declare variable REP_AMOUNT varchar(20);
declare variable CURR varchar(10);
declare variable NSD_PERCENT decimal(15,4);
declare variable K_KONTRACT_NO varchar(20);
declare variable DATE_FORMAT_STR varchar(20);
declare variable C_ALTERNAT_FIO STR_100;
declare variable OKS_KC_ID ID_TYPE;
declare variable SVZ_FIO_KIR STR_100;
declare variable SVZ_NAME_KIR STR_100;
declare variable SVZ_CELL_NO STR_100;
declare variable C_NAME_FIO_KIR STR_100;
declare variable CL_NAME_KIR STR_100;
declare variable OKS_REGION_ID integer;
declare variable OKS_KC_OK smallint;
declare variable OKS_KC_ID_TMP ID_TYPE;
declare variable CM_NAME STR_255;
declare variable PR_ID D_INTEGER;
declare variable K_DISB_DATE D_DATE;
declare variable MCL_PROJECT integer;
begin
  rep_days = coalesce(rep_days, 0);

  select cm.cm_name
  from company cm
  into :cm_name;
  :db_id = f_get_db_id();

  if (db_id = 81001) then
    curr = ' руб.';
  if (db_id in (39801, 41701)) then
    curr = ' сом';
  if (db_id = 94401) then
    curr = ' AZN';

  nsd_percent = 0;
  select coalesce(nsd_percent / 100, 0)
  from nspconfig_data
  where nsd_from_nsm_id = -1 and
        (:rep_date + :rep_days) between nsd_date_begin and nsd_date_end
  into nsd_percent;
  if (nsd_percent is null) then
    nsd_percent = 0;

  date_format_str = f_datetostr(rep_date - rep_days);

  for select oks_from_k_id, oks_kc_id, oks_region_id, k_kontract_no, k_disb_date, k_from_pr_id
      from oksummary, kontract
      where oks_close_state = 0 and -- важно!
            :rep_date - oks_delinq_date = cast(:rep_days as integer) and
            :rep_date between oks_date_begin and oks_date_end and
            oks_from_k_id = k_id
      into k_id, oks_kc_id, oks_region_id, k_kontract_no, k_disb_date, pr_id
  do
  begin
      -- Определяем проект: М Булак или OiVO RFCRU-4257
    mcl_project = f_get_mcl_project(f_objkind('kontract','id'), :k_id);

    select sum(round(zad_os + sh_calc_perc + sh_calc_perc * :nsd_percent + sh_calc_fine)) delinq_sum
    from aarep_make_zad_list_temp(:k_id, :rep_date, 1)
    where sh_date < :rep_date -- - :rep_days
    into rep_amount;

    if (oks_kc_id < 1000000000010) then -- если не виртуальный сотрудник
    begin
      --  если группа числится за виртуальным сотрудником, определяю менеджера, который утверждал эту группу
      select k_approved_manager
      from kontract
      where k_id = :k_id
      into oks_kc_id;

      oks_kc_id = coalesce(oks_kc_id, 0);

      -- список действующих менеджеров офиса

      oks_kc_ok = 0;

      for select ps_from_c_id
          from hr_pstatus, joblist, jobkind
          where ps_status = 1 and
                ps_is_working = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code = 'JobSystem.Region_KM' and
                :rep_date between ps_date_begin and ps_date_end and
                ps_from_reg_id = :oks_region_id
          into :oks_kc_id_tmp
      do
      begin
        if (oks_kc_id = oks_kc_id_tmp) then
        begin
          oks_kc_ok = 1;
        end
      end

      -- если менеджер утвердивший заем не совпал ни с одним из менеджеров офиса, указываю в СМС имя и телефон любого МО этого офиса.
      if (oks_kc_ok = 0) then
        oks_kc_id = oks_kc_id_tmp;
    end

    svz_fio_kir = null;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :oks_kc_id
    into :svz_fio_kir;

    svz_fio_kir = coalesce(svz_fio_kir, '');
    svz_name_kir = '';

    select txt2
    from hp_fio_clear(:svz_fio_kir)
    into :svz_name_kir;
    
    svz_name_kir = coalesce(svz_name_kir, '') || ' ';
    svz_cell_no = null;

    select cont_value
    from hp_contdata_get_cell_valid(0, :oks_kc_id)
    into :svz_cell_no;

    svz_cell_no = coalesce(svz_cell_no, '');

    rep_amount = cast(rep_amount as decimal(15,0));
    for select iif(((:rep_days = 1) and (:db_id = 81001)),
               (select cont_value
                from hp_contdata_get_cell_main(0, kc_from_c_id)),
               (select cont_value
                from hp_contdata_get_cell_valid(0, kc_from_c_id))),
                kc_from_c_id, kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into phone, client_id, kc_id
    do
    begin
      if (phone <> '') then
      begin
        if (db_id in (39801, 41701)) then -- KG
        begin
          sms_message = 'Ваш долг от ' || date_format_str || 'г.до сих пор не погашен. Сумма:' || rep_amount || 'сом';
          suspend;
        end
        else
        if (db_id = 81001) then -- RF
        begin
          cl_name = '';

          select cl.c_alternat_fio, cl.c_name_fio
          from clients cl
          where cl.c_id = :client_id
          into :c_alternat_fio, c_name_fio_kir;

          select txt2
          from hp_fio_clear(:c_alternat_fio)
          into :cl_name;

          select txt2
          from hp_fio_clear(:c_name_fio_kir)
          into :cl_name_kir;

          cl_name = coalesce(cl_name, '');
          if (cl_name = '') then
            cl_name = 'klient';

          cl_name_kir = coalesce(cl_name_kir, '');
          if (cl_name_kir = '') then
            cl_name_kir = 'клиент';

          sms_message = '';

          if ((rep_days = 1) and (mcl_project = 1))  then    -- М Булак RFCRU-4257
          begin
            sms_message =  'Уважаемый, ' || cl_name_kir || ', уведомляем о наличии просроченной задолженности по договору ' || k_kontract_no || '. По вопросам звоните 88005557171 ' || cm_name;
            push_message = 'Уважаемый, ' || cl_name_kir || ', уведомляем о наличии просроченной задолженности по договору ' || k_kontract_no || '. При наличии вопросов позвоните по телефону 88005557171 ' || cm_name;

            -- Разветвление после 08.09.2021, продукты оформляются со "скидкой", которая пропадает после просрочки
            if (k_disb_date >= cast('08.09.2021' as date)) then
              if (exists(select * from product
                         where pr_id = :pr_id
                           and pr_percent_kind = 1
                           and pr_date_begin >= '08.09.2021'
              )) then
              begin
                --  RFCRU-4097
                sms_message =  'Уважаемый, ' || cl_name_kir || ', уведомляем о наличии просроченной задолженности по договору ' || k_kontract_no || '. Низкая ставка может быть отменена, график измениться. По вопросам звоните  88005557171 ' || cm_name;
                push_message = 'Уважаемый, ' || cl_name_kir || ', уведомляем о наличии просроченной задолженности по договору ' || k_kontract_no || '. При наличии вопросов позвоните Вашему специалисту ' || svz_name_kir || ' Т.' || svz_cell_no || ' ' || cm_name;

              end
          end
          else
          if ((rep_days = 1) and (mcl_project = 2))  then      -- OiVO RFCRU-4257
          begin
            sms_message =  'Уважаемый, ' || cl_name_kir || ', уведомляем о наличии просроченной задолженности по договору ' || k_kontract_no || '. По вопросам звоните 88005559348 ' || cm_name;
            push_message = 'Уважаемый, ' || cl_name_kir || ', уведомляем о наличии просроченной задолженности по договору ' || k_kontract_no || '. При наличии вопросов позвоните по телефону 88005559348 ' || cm_name;

            -- Разветвление после 08.09.2021, продукты оформляются со "скидкой", которая пропадает после просрочки
            if (k_disb_date >= cast('08.09.2021' as date)) then
              if (exists(select * from product
                         where pr_id = :pr_id
                           and pr_percent_kind = 1
                           and pr_date_begin >= '08.09.2021'
              )) then
              begin
                --  RFCRU-4097
                sms_message =  'Уважаемый, ' || cl_name_kir || ', уведомляем о наличии просроченной задолженности по договору ' || k_kontract_no || '. Низкая ставка может быть отменена, график измениться. По вопросам звоните  88005559348 ' || cm_name;
                push_message = 'Уважаемый, ' || cl_name_kir || ', уведомляем о наличии просроченной задолженности по договору ' || k_kontract_no || '. При наличии вопросов позвоните Вашему специалисту ' || svz_name_kir || ' Т.' || svz_cell_no || ' ' || cm_name;

              end
          end
          else
          if ((rep_days = 3)  and (mcl_project = 1))  then    -- М Булак RFCRU-4257
          begin
            sms_message =  'Уважаемый, ' || cl_name_kir || ', У Вас имеется задолженность по договору ' || k_kontract_no || '. Срочно позвоните Вашему специалисту ' || svz_name_kir || ' ' || svz_cell_no || ' ' || cm_name;
            push_message = 'Уважаемый, ' || cl_name_kir || ', У Вас имеется задолженность. Срочно позвоните по телефону 88005557171 ' || cm_name;
          end
          else
          if ((rep_days = 3)  and (mcl_project = 2))  then    -- OiVO RFCRU-4257
          begin
            sms_message =  'Уважаемый, ' || cl_name_kir || ', У Вас имеется задолженность по договору ' || k_kontract_no || '. Срочно позвоните Вашему специалисту ' || svz_name_kir || ' ' || svz_cell_no || ' ' || cm_name;
            push_message = 'Уважаемый, ' || cl_name_kir || ', У Вас имеется задолженность. Срочно позвоните по телефону 88005559348 ' || cm_name;
          end
          else
          if ((rep_days = 5) and (mcl_project = 1))  then    -- М Булак RFCRU-4257
          begin
            sms_message =  'Уважаемый, ' || cl_name_kir || ', У Вас имеется задолженность по договору ' || k_kontract_no || '. Срочно позвоните Вашему специалисту ' || svz_name_kir || ' ' || svz_cell_no || ' ' || cm_name;
            push_message = 'Уважаемый, ' || cl_name_kir || ', У Вас имеется задолженность по договору ' || k_kontract_no || '. Срочно позвоните по телефону 88005557171 ' || cm_name;
          end
          else
          if ((rep_days = 5) and (mcl_project = 2))  then    -- OiVO RFCRU-4257
          begin
            sms_message =  'Уважаемый, ' || cl_name_kir || ', У Вас имеется задолженность по договору ' || k_kontract_no || '. Срочно позвоните Вашему специалисту ' || svz_name_kir || ' ' || svz_cell_no || ' ' || cm_name;
            push_message = 'Уважаемый, ' || cl_name_kir || ', У Вас имеется задолженность по договору ' || k_kontract_no || '. Срочно позвоните по телефону 88005559348 ' || cm_name;
          end
          else
          if ((rep_days = 10) and (mcl_project = 1))  then    -- М Булак RFCRU-4257
          begin
            sms_message =  'Уважаемый, ' || cl_name_kir || ', У Вас задолженность по договору ' || k_kontract_no || ', долг увеличивается! Срочно позвоните Вашему специалисту ' || svz_name_kir || ' ' || svz_cell_no || ' ' || cm_name;
            push_message = 'Уважаемый, ' || cl_name_kir || ', У Вас задолженность по договору ' || k_kontract_no || '. Долг увеличивается. Срочно позвоните по телефону 88005557171 ' || cm_name;
          end
          else
          if ((rep_days = 10) and (mcl_project = 2))  then    -- OiVO RFCRU-4257
          begin
            sms_message =  'Уважаемый, ' || cl_name_kir || ', У Вас задолженность по договору ' || k_kontract_no || ', долг увеличивается! Срочно позвоните Вашему специалисту ' || svz_name_kir || ' ' || svz_cell_no || ' ' || cm_name;
            push_message = 'Уважаемый, ' || cl_name_kir || ', У Вас задолженность по договору ' || k_kontract_no || '. Долг увеличивается. Срочно позвоните по телефону 88005559348 ' || cm_name;
          end
          if (sms_message <> '' and (rep_days = 1 or f_clientcom_allowed(client_id, 0, 0, 1) = 1)) then
            suspend;
        end
        else
        if (db_id = 94401) then -- AZ
        begin
          sms_message = 'MBulaka odenishiniz ' || :rep_days || ' gun gecikir. Bugune olan gecikme borcunuz ' || :rep_amount || :curr || '. Gecikmede olan her gune cerime hesablanir';
          suspend;
        end
      end
    end
  end
end
