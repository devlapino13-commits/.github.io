CREATE PROCEDURE HP_UX_SELECT_ITEMS(
    LIST_KIND D_STR_CODE,
    ITEM_CODE D_STR_CODE)
RETURNS (
    ID D_STR_255,
    ID_VISIBLE D_STR_255,
    TEXT D_STR_255)
AS
declare variable ux_kind d_integer;
declare variable source_name d_str_code;
declare variable fr_code d_str_code;
declare variable root_grid d_str_code;
declare variable ux_code d_str_code;
declare variable ux_kind_str d_str_code;
declare variable field_name d_str_code;
declare variable ufi_parent_code d_str_100;
declare variable table_name d_str_code;
declare variable uxcommand_kind d_smallint;
declare variable dont_show_fields d_bool;
begin
  if (trim(coalesce(list_kind, '')) = '') then
    exit;

  if (list_kind is null) then
    list_kind = '';

  item_code = trim(coalesce(item_code, ''));

  if (list_kind = 'ufp_buttonedit_uxa') then
  begin
    select ufd_source, ufp_origin_name
    from uxfrparam, uxfrdata
    where ufp_code = :item_code and
          ufp_ufr_code = ufd_ufr_code and
          ufd_kind = 0
    into source_name, field_name;

    for select fk_table
    from hp_uxfrparam_fk_list(:source_name, :field_name)
    into :table_name
    do
    begin
      for select uxa_code, uxa_code, uxa_name || '  - ' || Upper(ux_table_name)
      from uxalias, uxcode, uxcodeframe, uxframe
      where ux_table_name = :table_name and
            ucf_ux_code = ux_code and
            ucf_ufr_code = ufr_code and
            ucf_is_mainframe = 1 and
            ufr_show_kind = 2 and -- Набор данных
            uxa_ux_code = ux_code
      into id, id_visible, text
      do
        suspend;
    end

    exit;
  end

  if (list_kind = 'lookup_source') then
  begin
    if (not exists(select * from uxlookup
    where ulp_table_name = :item_code and
          ulp_source = :id)) then
    begin
      id = item_code;
      id_visible = item_code;
      text = 'Текущая таблица';
      suspend;
    end

    text = '';

    for select lower(trim(p.rdb$procedure_name))
    from rdb$procedures p
    into id
    do
    begin
      id_visible = id;

      if (exists(select * from uxlookup
      where ulp_table_name = :item_code and
            ulp_source = :id)) then
        Continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'ufp_lookup_code') then
  begin
    select ufd_source, ufp_origin_name
    from uxfrparam, uxfrdata
    where ufp_code = :item_code and
          ufp_ufr_code = ufd_ufr_code and
          ufd_kind = 0
    into source_name, field_name;

    for select fk_table
    from hp_uxfrparam_fk_list(:source_name, :field_name)
    into :table_name
    do
    begin
      for select ulp_code, ulp_code, ulp_name
      from uxlookup
      where ulp_table_name = :table_name
      into id, id_visible, text
      do
        suspend;
    end

    exit;
  end

  if (list_kind = 'ufp_combobox_code') then
  begin
    for select ucm_code, ucm_code, ucm_name
    from uxcombomain
    into id, id_visible, text
    do
      suspend;

    exit;
  end

  if (list_kind = 'uxcodeframe_new_frames') then
  begin
    if (item_code = '') then
      exit;

    ux_kind = uxf$kind('uxframe', 'id');
    select ux_table_name
    from uxcode
    where ux_code = :item_code
    into table_name;

    for select ux_code, ux_code, ux_name
    from uxcode
    where ux_kind = :ux_kind and
          ux_table_name = :table_name
    into id, id_visible, text
    do
    begin
      if (exists(select * from uxcodeframe
      where ucf_ux_code = :item_code and
            ucf_ufr_code = :id)) then
        continue;

      suspend;
    end

    for select ux_code, ux_code, ux_name || '  - ' || Upper(ux_table_name)
    from uxcode
    where ux_kind = :ux_kind and
          ux_table_name <> :table_name
    into id, id_visible, text
    do
    begin
      if (exists(select * from uxcodeframe
      where ucf_ux_code = :item_code and
            ucf_ufr_code = :id)) then
        continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'uxcodeframe_existing_frames') then
  begin
    if (item_code = '') then
      exit;

    for select ucf_ufr_code, ucf_ufr_code, ux_name
    from uxcodeframe, uxcode
    where ucf_ux_code = :item_code and
          ucf_ufr_code = ux_code
    into id, id_visible, text
    do
    begin
      if (exists(select * from uxfritem
      where ufi_ux_code = :item_code and
            ufi_ufr_code = :id)) then
        continue;

      suspend;
    end

    exit;
  end


  if (list_kind = 'frame_master_fields') then
  begin
    if (item_code = '') then
      exit;

    for select ufp_origin_name,
               ufp_origin_name,
               iif(ufp_pk_field = 0, '', 'pk' || ufp_pk_field ||'  ') ||
               iif(ufp_name <> ufp_origin_name,
                   ufp_origin_name || '  - ' || ufp_name, ufp_origin_name)
    from uxfrparam
    where ufp_ufr_code = :item_code and
          ufp_is_delete= 0 and
          ufp_in_use = 1
    into id, id_visible, text
    do
      suspend;

    exit;
  end

  if (list_kind = 'frame_detail_fields') then
  begin
    if (item_code = '') then
      exit;

    select ufd_source
    from uxfrdata
    where ufd_ufr_code = :item_code and
          ufd_kind = 0
    into source_name;

    -- Дочерное поле для связки мастер-деталь
    -- Отображаем только тех у кого есть ссылка на другие таблицы
    for select ufp_origin_name,
               ufp_origin_name,
               iif(ufp_pk_field = 0, '', 'pk' || ufp_pk_field ||'  ') ||
               iif(ufp_name <> ufp_origin_name,
                   ufp_origin_name || '  - ' || ufp_name, ufp_origin_name)
    from uxfrparam
    where ufp_ufr_code = :item_code and
          ufp_is_delete = 0 and
          ufp_in_use = 1
    into id, id_visible, text
    do
    begin
      if (exists(select * from hp_uxfrparam_fk_list(:source_name, :id))) then
        suspend;
    end

    exit;
  end

  if (list_kind = 'action_form_open') then
  begin
    for select uxa_code, uxa_code, uxa_name || '  - ' || Upper(ux_table_name)
    from uxcode, uxform, uxalias
    where ux_code = ufm_code and
          ux_code = uxa_ux_code and
          uxa_ux_code <> :item_code
    into id, id_visible, text
    do
    begin
      -- Если ранее добавлен, не показываем в списке чтобы повторно не добавить
      if (exists(select * from uxaction
      where ua_ux_code = :item_code and
            ua_uxa_code = :id)) then
        Continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'action_form_insert') then
  begin
    for select uxa_code, uxa_code, ux_code, uxa_name || '  - ' || Upper(ux_table_name)
    from uxcode, uxform, uxalias
    where ux_code = ufm_code and
          ux_code = uxa_ux_code and
          uxa_ux_code <> :item_code
    into id, id_visible, ux_code, text
    do
    begin
      -- Если ранее добавлен, не показываем в списке чтобы повторно не добавить
      if (exists(select * from uxaction
      where ua_ux_code = :item_code and
            ua_uxa_code = :id)) then
        Continue;

      if (not exists(select *
      from uxcodeframe, uxframe
      where ucf_ux_code = :ux_code and
            ucf_ufr_code = ufr_code and
            ucf_is_mainframe = 1 and
            ufr_show_kind = 1)) then
        continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'action_command_open') then
  begin
    for select uxa_code, uxa_code, uxa_name || '  - ' || Upper(ux_table_name)
    from uxcode, uxcommand, uxalias
    where ux_code = cmd_code and
          uxa_ux_code = ux_code
    into id, id_visible, text
    do
    begin
      suspend;
    end

    exit;
  end


  if (list_kind = 'action_blank_open') then
  begin
    for select uxa_code, uxa_code, uxa_name || '  - ' || Upper(ux_table_name)
    from uxcode, uxblank, uxalias
    where ux_code = ub_code and
          ux_code = uxa_ux_code
    into id, id_visible, text
    do
    begin
      -- Если ранее добавлен, не показываем в списке чтобы повторно не добавить
      if (exists(select * from uxaction
      where ua_ux_code = :item_code and
            ua_uxa_code = :id)) then
        Continue;

      suspend;
    end

    exit;
  end

  if (list_kind = 'uxframe_create_kind') then
  begin
    id = 'uxframe';
    id_visible = 'Только фрейм';
    text = 'Создать только фрейм';
    suspend;

    id = 'uxform';
    id_visible = 'Фрейм и табличная форма';
    text = 'Создать фрейм и форму с набором данных';
    suspend;

    id = 'uxform_one';
    id_visible = 'Фрейм с одной записю';
    text = 'Создать фрейм и форму с одной записю';
    suspend;

    exit;
  end

  if (list_kind = 'uxcode_actions') then
  begin
    select uxf$kind(ux_kind, 'code')
    from uxcode
    where ux_code = :item_code
    into ux_kind_str;

    if (ux_kind_str <> 'uxFrame') then
    begin
      for select ua_code, uxa_code,
                 trim(case uxa_kind
                    when 0 then 'Открыть:'
                    when 2 then 'Создать:'
                 end) || ' ' || uxa_name
      from uxalias, uxaction
      where uxa_ux_code = :item_code and
            uxa_code = ua_uxa_code
      into id, id_visible, text
      do
      begin
        if (id starting 'fmMainMenu#') then
          text = replace(text, 'Открыть:', 'Главное меню:');

        suspend;
      end

      if (row_count = 0) then
      begin
        for select 'fmMainMenu#' || uxa_code, uxa_code, 'Тест:  ' || uxa_name
        from uxalias
        where uxa_ux_code = :item_code
        into id, id_visible, text
        do
          suspend;
      end
    end

    exit;
  end

  if (list_kind = 'uxaction_add_def_param') then
  begin
    for select id, id_visible, text
    from hp_uxactionparam_for_ins(:item_code)
    into :id, :id_visible, :text
    do
      suspend;

    exit;
  end

  if (list_kind = 'uxfrparam_add_source') then
  begin
    for select lower(ut_code), coalesce(ut_name, ''), left(ut_memo, 255)
    from hp_uxtable_list
    into :id, :id_visible, :text
    do
    begin
      if (id_visible <> '') then
        text = id_visible;

      id_visible = id;
      suspend;
    end

    exit;
  end

  if (list_kind = 'uxfrparam_del_source') then
  begin
    select ufd_source, ufp_origin_name
    from uxfrparam, uxfrdata
    where ufp_code = :item_code and
          ufp_ufr_code = ufd_ufr_code and
          ufd_kind = 0
    into ux_code, field_name;

    for select url_fk_table
    from uxrelation
    where url_source = :ux_code and
          url_field_name = :field_name
    into :id
    do
    begin
      id_visible = id;
      text = '';

      select ut_caption
      from uxtable
      where ut_table_name = :id
      into text;

      if (text = '') then
      begin
        select left(trim(rdb$description), 255)
        from rdb$relations r
        where r.rdb$system_flag = 0 and
              r.rdb$relation_type in (0, 1, 4) and
              r.rdb$relation_name = upper(:id)
        into text;
      end

      suspend;
    end

    exit;
  end

  if (list_kind = 'uxfrparam_add_params') then
  begin
    select ufi_ux_code, ufi_parent_code
    from uxfritem
    where ufi_code = :item_code
    into fr_code, ufi_parent_code;

    uxcommand_kind = uxf$kind('uxcommand', 'id');

    -- если фрейм участвует в командах, не показываем исходящие поля
    dont_show_fields = 0;
    if (exists(select * from uxcodeframe, uxcode
    where ucf_ufr_code = :fr_code and
          ucf_ux_code = ux_code and
          ux_kind = :uxcommand_kind)) then
      dont_show_fields = 1;

    if (ufi_parent_code is null) then
      ufi_parent_code = item_code;

    root_grid = fr_code || '#g';


    for select ufp_code,
               ufp_origin_name,
               iif(ufp_origin_name = ufp_name,
                   ufd_source || '.' || ufp_origin_name,
                   ufd_source || '.' || ufp_origin_name || '   - ' || ufp_name)
    from uxfrparam, uxfrdata
    where ufp_ufr_code = :fr_code and
          ufp_is_delete = 0 and
          ufp_in_use = 1 and
          ufd_ufr_code = ufp_ufr_code and
          ufd_kind = 0 and
          ufp_editor <> 'current_user' and
          not(:dont_show_fields =1 and ufp_is_param = 0)
    into id, id_visible, text
    do
    begin
      if (ufi_parent_code = root_grid or item_code = root_grid) then
      begin
        if (exists(select * from uxfritem
        where ufi_ux_code = :fr_code and
              ufi_parent_code = :root_grid and
              ufi_ufp_code = :id)) then
         Continue;
      end else
      begin
        if (exists(select * from uxfritem
        where ufi_ux_code = :fr_code and
              ufi_parent_code <> :root_grid and
              ufi_ufp_code = :id)) then
         Continue;
      end

      suspend;
    end

    exit;
  end

  if (list_kind = 'source_fields') then
  begin
    text = '';
    for select field_name
    from hp_ux_source_fields(:item_code)
    where param_kind = 1
    into :id
    do
    begin
      id_visible = id;
      suspend;
    end

    exit;
  end

  if (list_kind = 'FsConfig_TestUsers') then
  begin
    for select un_name, un_name, un_fio
    from username
    into id, id_visible, text
    do
      suspend;

    Exit;
  end

  exception e_message f_format('Undefined params: list_kind: %1, item_code: %2', list_kind,  item_code);
end
