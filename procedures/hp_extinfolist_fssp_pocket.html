CREATE PROCEDURE HP_EXTINFOLIST_FSSP_POCKET(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REC_NO D_INTEGER,
    EIL_ID D_ID,
    LINE_NO D_INTEGER,
    EIL_FROM_C_ID D_ID,
    EIL_FROM_K_ID D_ID,
    EIL_DATE D_DATE,
    EIL_FROM_BLS_ID D_ID,
    EIL_CREATE_DATETIME D_DATETIME,
    EIL_STATUS_DATETIME D_DATETIME,
    EIL_STATUS_STR D_STR_20,
    JS_NAME D_STR_4096,
    JS_EXE_PRODUCTION D_STR_4096,
    JS_DETAILS D_STR_4096,
    JS_IP_END D_STR_4096,
    JS_SUBJECT D_STR_4096,
    JS_DEPARTMENT D_STR_4096,
    JS_BAILIFF D_STR_4096,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE D_DATE,
    REG_NAME D_STR_100,
    KF_NAME D_STR_100)
AS
declare variable eil_status d_smallint;
declare variable eic_id d_integer;
declare variable item_count d_integer;
declare variable json_item d_str_8100;
declare variable json_text d_blob_text;
begin
  select eic_id
  from extinfocfg
  where eic_code = 'fssp'
  into eic_id;

  if (eic_id is null) then
    exit;

  rec_no = 0;

  for select eil_id, eil_from_c_id, eil_date, eil_object_id, eil_from_bls_id,
             eil_status, eil_create_datetime, eil_status_datetime
  from extinfolist, extinfopocket
  where eil_from_eic_id = :eic_id and
        eil_date between :rep_date_begin and :rep_date_end and
        eil_from_eip_id = eip_id
  into eil_id, eil_from_c_id, eil_date, eil_from_k_id, eil_from_bls_id,
       eil_status, eil_create_datetime, eil_status_datetime
  do
  begin
    json_text = '';
    js_name = '';
    js_exe_production = '';
    js_details = '';
    js_ip_end = '';
    js_subject = '';
    js_department = '';
    js_bailiff = '';
    line_no = 0;

    eil_status_str = 'Нет инфо';
    eil_status_str = trim(case eil_status
                          when -3 then 'Отменен'
                          when -2 then 'Ошибка'
                          when -1 then 'Недоступен'
                          when 0 then 'В очереди'
                          when 1 then 'В обработке'
                          when 2 then 'Готов'
                        end);

    select c_name_fio, c_birth_date
    from clients
    where c_id = :eil_from_c_id
    into c_name_fio, c_birth_date;

    select k_kontract_no, k_disb_date, reg_code || ' ' || reg_name, kn.kal_new_kf_name
    from kontract, knaktlist kn, region
    where k_id = :eil_from_k_id and
          k_from_kf_id = kn.kal_from_kf_id and
          kn.kal_date_end >= '31.12.2999' and
          kn.is_delete = 0 and
          kn.kal_new_reg_id = reg_unicode
    into k_kontract_no, k_disb_date, reg_name, kf_name;

    if (eil_from_bls_id is not null) then
    begin
      select cast(blx_blob as d_blob_text)
      from hp_get_blobex_blob(:eil_from_bls_id)
      into json_text;
    end
    if (json_text <> '') then
    begin
      item_count = f_json_blob_get_count(json_text, 'Result');
      if (item_count > 0) then
      begin
        while (line_no < item_count) do
        begin
          json_item = f_json_blob_get_value(json_text, 'Result[' || line_no || ']', '');
          js_name = f_json_str_get_value(json_item, 'name', '');
          js_exe_production = f_json_str_get_value(json_item, 'exe_production', '');
          js_details = f_json_str_get_value(json_item, 'details', '');
          js_ip_end = f_json_str_get_value(json_item, 'ip_end', '');
          js_subject = f_json_str_get_value(json_item, 'subject', '');
          js_department = f_json_str_get_value(json_item, 'department', '');
          js_bailiff = f_json_str_get_value(json_item, 'bailiff', '');

          line_no = line_no + 1;
          rec_no = rec_no + 1;
          suspend;
        end
      end
    end
    if (line_no = 0) then
    begin
      rec_no = rec_no + 1;
      suspend;
    end
  end
end
