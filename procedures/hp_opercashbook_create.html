CREATE PROCEDURE HP_OPERCASHBOOK_CREATE(
    OFFICE_ID D_INTEGER,
    OPER_DATE D_DATE,
    CASH_ID D_INTEGER,
    IN_MANAGER_ID D_ID,
    UN_ID D_ID)
RETURNS (
    CBM_ID D_INTEGER,
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable opc_from_csp_id d_integer;
declare variable cash_no d_integer;
declare variable saldo_count d_integer;
declare variable opc_sign d_smallint;
declare variable cls_al_id d_str_20;
declare variable ipf_al_id d_str_20;
declare variable cls_saldo d_currency;
declare variable ipf_saldo d_currency;
declare variable in_cashier_id d_id;
begin
  res_is_error = 0;
  res_text = 'Кассовый журнал успешно сформирован';

  if (in_manager_id is null) then
  begin
    res_is_error = 1;
    res_text = 'Контролер не указан';
    suspend;
    exit;
  end

  select csp_cashier_id
  from cashplace
  where csp_id = :cash_id
  into in_cashier_id;

  if (in_cashier_id is null) then
  begin
    res_is_error = 1;
    res_text = 'Кассир не определен';
    suspend;
    exit;
  end

  if (exists(select * from opercash
             where opc_office_id = :office_id and
                   opc_date = :oper_date and
                   opc_is_delete = 0 and
                   (opc_from_ol_id is null or opc_status <> 1))) then
  begin
    res_is_error = 1;
    res_text = 'Имеются не проведенные операции';
    suspend;
    exit;
  end

  for select distinct opc_from_csp_id, csp_cash_no
  from opercash, cashplace
  where opc_office_id = :office_id and
        opc_date = :oper_date and
        opc_from_csp_id = csp_id and
        opc_is_delete = 0
  into opc_from_csp_id, cash_no
  do
  begin
    opc_sign = null;
    select opc_sign
    from opercash
    where opc_from_csp_id = :opc_from_csp_id and
          opc_date = :oper_date and
          opc_is_delete = 0
    order by opc_id desc
    rows 1
    into opc_sign;

    if (opc_sign <> 0) then
    begin
      res_text = 'Последняя запись в журнале должна быть "Закрытие кассы", касса: ' || cash_no;
      suspend;
      exit;
    end

    insert into cashtotal (ctt_from_csp_id, ctt_date, ctt_need_recalc)
    values (:opc_from_csp_id, :oper_date, 1);
  end

  saldo_count = 0;
  for select csp_id, csp_cash_no, csp_cls_al_id, csp_ipf_al_id
  from cashplace
  where csp_office_id = :office_id
  into opc_from_csp_id, cash_no, cls_al_id, ipf_al_id
  do
  begin
    cls_saldo = 0;
    ipf_saldo = 0;

    if (cls_al_id is not null) then
      cls_saldo = f_al_bal(cls_al_id, oper_date);

    if (ipf_al_id is not null) then
      ipf_saldo = f_al_bal(ipf_al_id, oper_date);

    if (f_cash_saldo(opc_from_csp_id, oper_date, 0) <> (cls_saldo + ipf_saldo)) then
    begin
      res_text = 'Кассовый журнал сформирован. Остаток кассы не совпадает с балансом, касса: ' || cash_no;
      -- Временно продолжаем работу, тестирование
--      suspend;
--      exit;
    end

    if (cls_saldo <> 0 or ipf_saldo <> 0) then
      saldo_count = saldo_count + 1;
  end

  if (saldo_count > 1) then
  begin
    res_text = 'Кассовый журнал сформирован. Остатки не переведены в главную кассу';
      -- Временно продолжаем работу, тестирование
--    suspend;
--    exit;
  end

  insert into cashbookmain (cbm_office_id, cbm_cashier_id, cbm_manager_id, cbm_date, cbm_from_un_id, cbm_need_recalc)
  values (:office_id, :in_cashier_id, :in_manager_id, :oper_date, :un_id, 1)
  returning cbm_id
  into :cbm_id;

  insert into cashbooklist (cbl_from_cbm_id, cbl_doc_no, cbl_is_ipf, cbl_from_ol_id, cbl_client_fio,
                            cbl_amount_plus, cbl_amount_minus, cbl_cash_acc_no, cbl_cash_account, cbl_corr_account)
  select :cbm_id, opc_doc_no, opc_is_ipf, opc_from_ol_id, res_client_fio,
         res_amount_plus, res_amount_minus, res_cash_acc_no, res_cash_account, res_account
  from hp_opercash_journal_list(:office_id, :oper_date, 0, :un_id);

  insert into cashbookacc (cba_from_cbm_id, cba_account, cba_amount_plus, cba_amount_minus)
  select :cbm_id, res_account, res_amount_plus, res_amount_minus
  from hp_opercash_journal_acc_total(:office_id, :oper_date, :un_id);

  suspend;
end
