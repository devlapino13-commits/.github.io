CREATE PROCEDURE HP_KONTREFUND_K_LIST(
    K_ID D_ID)
RETURNS (
    RES_K_ID D_ID,
    RES_AMOUNT D_STR_20,
    RES_TEXT D_STR_LONG)
AS
declare variable pr_name d_str_code;
declare variable k_disb_date d_date;
declare variable disb_date d_date;
declare variable pr_round_perc d_integer;
declare variable tax_perc d_percent;
declare variable db_id d_integer;
declare variable pr_code varchar(100);
declare variable is_ipf smallint;
declare variable amount d_currency;
begin
  -- процедура возвращает список действующий займов участников и сумму для закрытия

  -- если переданный займ не находится в процессе
  if (not exists(select *
                 from kontstate
                 where ks_from_k_id = :k_id and
                       ks_state = 0 and
                       ks_date_end >= '31.12.2999')) then
    exit;

  db_id = f_get_db_id();

  select k_disb_date from kontract
  where k_id = :k_id
  into disb_date;

  -- получаем все активныые займы клиентов, которые участвует параданном займе
  for
    select distinct
      ks_from_k_id
    from knclient as kc1
      inner join knclient as kc2
         on kc2.kc_from_c_id = kc1.kc_from_c_id
           and kc2.kc_from_k_id <> :k_id
           and kc2.kc_amount_approved > 0
      inner join kontstate
        on ks_from_k_id = kc2.kc_from_k_id + 0
          and ks_date_end = '31.12.9999'
          and ks_state = 1
          and ks_ol_is_delete = 0
    where kc1.kc_from_k_id = :k_id
      and kc1.kc_amount_approved > 0
    into
      :res_k_id
  do
  begin
    select k_disb_date, pr_name, pr_round_perc, pr_code
    from product, kontract
    where k_id = :res_k_id and
          k_from_pr_id = pr_id
    into k_disb_date, pr_name, pr_round_perc, pr_code;
    if (pr_code containing '.IPF.') then
      is_ipf = 1;
    else
      is_ipf = 0;

    res_text = f_objname('kontract', res_k_id);
    res_text = res_text || '<br>' || pr_name;

    tax_perc = 0;
    select nsp_percent_for_int
    from hp_get_kontract_nsp_acc(0, :k_disb_date)
    into :tax_perc;

    -- вычисляем сумму для полного погашения
    res_amount = 0;
    select sum(zad_os + iif(:is_ipf = 1, zad_perc, sh_calc_perc) + sh_calc_fine + zad_kommis + -->>
           round((iif(:is_ipf = 1, zad_perc, sh_calc_perc) * iif(:db_id in (39801, 41701) and sh_date is null, 0, :tax_perc)) / 100.00, -->>
           :pr_round_perc) + round((zad_kommis * :tax_perc) / 100.00, :pr_round_perc))
    from aarep_make_zad_list(:res_k_id, :disb_date, 1, 1)
    into amount;

    if (amount is null) then
      amount = 0;

    res_amount = round(amount, 2);
    suspend;
  end
end
