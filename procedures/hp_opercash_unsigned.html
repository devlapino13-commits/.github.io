CREATE PROCEDURE HP_OPERCASH_UNSIGNED(
    OFFICE_ID D_INTEGER,
    REP_DATE_END D_DATE,
    IN_OWN_DOCS D_BOOL,
    UN_ID D_ID)
RETURNS (
    OPC_ID D_INTEGER,
    OPC_PARENT_ID D_INTEGER,
    OPC_DATE D_DATE,
    OPC_DOC_NO D_INTEGER,
    OPC_OFFICE_ID D_INTEGER,
    OPC_FROM_CSP_ID D_INTEGER,
    OPC_OPER_KIND D_SMALLINT,
    OPC_SIGN D_SMALLINT,
    OPC_AMOUNT D_CURRENCY,
    OPC_FROM_C_ID D_ID,
    OPC_IS_IPF D_BOOL,
    OPC_CORR_AL_ID D_STR_20,
    OPC_FROM_K_ID D_ID,
    OPC_CORR_REG_ID D_INTEGER,
    OPC_MEMO D_STR_LONG,
    OPC_IS_ONLINE D_BOOL,
    OPC_FROM_EDL_ID D_ID,
    OPC_STATUS D_SMALLINT,
    OPC_MANAGER_ID D_ID,
    OPC_IS_SIGNED D_BOOL,
    OPC_FROM_OL_ID D_ID,
    OPC_IS_DELETE D_BOOL,
    OPC_FROM_UN_ID D_ID,
    OPC_DATETIME D_DATETIME,
    RES_OFFICE_NAME D_STR_100,
    RES_CORR_OFFICE D_STR_100,
    RES_CLIENT_FIO D_STR_100,
    RES_AMOUNT_PLUS D_CURRENCY,
    RES_AMOUNT_MINUS D_CURRENCY,
    RES_CASH_ACCOUNT D_STR_20,
    RES_CORR_ACCOUNT D_STR_20,
    RES_STATUS_STR D_STR_20,
    RES_USERNAME D_STR_CODE,
    RES_OPER_KIND D_STR_100,
    RES_IS_CHILD D_BOOL,
    RES_CAN_EXECUTE D_BOOL,
    RES_OPER_CODE D_STR_CODE,
    RES_IS_CLOSEDAY D_BOOL,
    RES_BLANK_ID D_INTEGER,
    RES_MANAGER_FIO D_STR_100,
    RES_OWN_DOC D_BOOL)
AS
declare variable last_oper_kind d_integer;
declare variable user_c_id d_id;
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into user_c_id;

  for select opc_id, opc_parent_id, opc_date, opc_doc_no, opc_office_id, opc_from_csp_id, opc_oper_kind, opc_sign,
             opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id, opc_from_k_id, opc_corr_reg_id, opc_memo,
             opc_is_online, opc_from_edl_id, opc_status, opc_manager_id, opc_is_signed, opc_from_ol_id, opc_is_delete,
             opc_from_un_id, opc_datetime, res_office_name, res_corr_office, res_client_fio, res_amount_plus,
             res_amount_minus, res_cash_account, res_corr_account, res_status_str, res_username, res_oper_kind,
             res_is_child, res_can_execute, res_oper_code, res_is_closeday, res_blank_id, res_manager_fio
  from hp_opercash_office(:office_id, 0, :rep_date_end, 0, :un_id)
  where opc_is_signed = 0
  into :opc_id, :opc_parent_id, :opc_date, :opc_doc_no, :opc_office_id, :opc_from_csp_id, :opc_oper_kind, :opc_sign,
       :opc_amount, :opc_from_c_id, :opc_is_ipf, :opc_corr_al_id, :opc_from_k_id, :opc_corr_reg_id, :opc_memo,
       :opc_is_online, :opc_from_edl_id, :opc_status, :opc_manager_id, :opc_is_signed, :opc_from_ol_id,
       :opc_is_delete, :opc_from_un_id, :opc_datetime, :res_office_name, :res_corr_office, :res_client_fio,
       :res_amount_plus, :res_amount_minus, :res_cash_account, :res_corr_account, :res_status_str, :res_username,
       :res_oper_kind, :res_is_child, :res_can_execute, :res_oper_code, :res_is_closeday, :res_blank_id, :res_manager_fio
  do
  begin
    if (opc_manager_id = user_c_id) then
      res_own_doc = 1;
    else
      res_own_doc = 0;

    if (in_own_docs <> res_own_doc) then
      Continue;

    suspend;
  end
end
