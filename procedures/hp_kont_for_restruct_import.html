CREATE PROCEDURE HP_KONT_FOR_RESTRUCT_IMPORT(
    UN_ID D_ID)
RETURNS (
    RES_ERR_TEXT D_STR_255,
    RES_INFO_TEXT D_STR_255)
AS
declare variable tid_id d_id;
declare variable k_id d_id;
declare variable k_kontract_no d_str_255;
declare variable import_count d_integer;
declare variable error_count d_integer;
begin
  import_count = 0;
  error_count = 0;
  res_err_text = '';
  res_info_text = '';


  for select tid.tid_id,
             tid.tid_text
  from temp_import_data tid
  where coalesce(tid.tid_is_imported, 0) = 0
    and coalesce(tid.tid_text, '') <> ''
  into :tid_id,
       :k_kontract_no
  do
  begin
    k_id = 0;

    select k.k_id
    from kontract k
    where k.k_kontract_no = :k_kontract_no
    into :k_id;

    if (coalesce(:k_id, 0) = 0) then
    begin
      res_err_text = res_err_text || 'Контракт: ' || :k_kontract_no || ' не найден в базе! '|| ascii_char(10);
      continue;
    end

    if (exists(select 1
              from kont_for_restruct_kz kfr
              where kfr.kfr_from_k_id = :k_id
                and kfr.kfr_is_delete = 0)) then
    begin
      res_err_text = res_err_text || 'Контракт: ' || :k_kontract_no || ' уже импортирован! '|| ascii_char(10);
      continue;
    end

    insert into kont_for_restruct_kz (kfr_from_k_id, kfr_from_un_id)
    values (:k_id, :un_id);

    update temp_import_data
    set tid_is_imported = 1
    where tid_id = :tid_id;

    import_count = import_count + 1;
  end

  if (import_count = 0) then
  begin
    res_err_text = res_err_text || 'Импортировано записей: '||import_count||'.' || ascii_char(10) ||
                   'Данные во временной таблице не сохранены. ';
    res_info_text = null;
  end
  else
  begin
    res_info_text = 'Импортировано записей: ' || import_count||'.';
    res_err_text = null;
  end
  suspend;
end
