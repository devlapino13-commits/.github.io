CREATE PROCEDURE HP_CARDAKTMAIN_TRANSACTION(
    CAM_ID D_INTEGER)
RETURNS (
    OL_ID D_ID,
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable cam_status d_smallint;
declare variable curr_code d_str_lang;
declare variable card_list d_str_8100;
declare variable card_count d_integer;
declare variable card_count1 d_integer;
declare variable pin_count d_integer;
declare variable pin_count1 d_integer;
declare variable cpls_id d_integer;
declare variable cpld_id d_integer;
declare variable cpld_pin_id d_integer;
declare variable reg_id d_integer;
declare variable regs_id d_integer;
declare variable regd_id d_integer;
declare variable regs_name d_str_code;
declare variable regd_name d_str_code;
declare variable cs_id d_id;
declare variable cd_id d_id;
declare variable is_transfer_s d_smallint;
declare variable is_transfer_d d_smallint;
declare variable type_akt d_smallint;
declare variable un_id d_id;
declare variable ckd_name d_str_code;
declare variable ckd_is_own d_bool;
declare variable first_card d_integer;
declare variable doc_kind d_integer;
declare variable ol_id1 d_id;
declare variable ol_id2 d_id;
declare variable ol_datetime d_datetime;
declare variable ol_memo d_str_long;
declare variable ol_memo_pin d_str_long;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable acc_dt1 d_str_20;
declare variable acc_kt1 d_str_20;
declare variable acc_dt2 d_str_20;
declare variable acc_kt2 d_str_20;
declare variable acc_pin_dt1 d_str_20;
declare variable acc_pin_kt1 d_str_20;
declare variable acc_pin_dt2 d_str_20;
declare variable acc_pin_kt2 d_str_20;
declare variable acc_96101_12 d_str_20;
declare variable acc_96101_13 d_str_20;
declare variable acc_96101_14 d_str_20;
declare variable acc_96101_15 d_str_20;
declare variable acc_96101_16 d_str_20;
declare variable acc_96101_17 d_str_20;
declare variable acc_96102_12 d_str_20;
declare variable acc_96102_13 d_str_20;
declare variable acc_96102_14 d_str_20;
declare variable acc_96102_15 d_str_20;
declare variable acc_96102_16 d_str_20;
declare variable acc_96102_17 d_str_20;
begin
  err_code = 0;
  err_text = '';
  curr_code = '417';

  select
    cam_status,
    cam_card_list,
    cam_card_count,
    cam_src_cpl_id,
    cam_dst_cpl_id,
    cam_dst_pin_cpl_id,
    regs_id,
    regs_name,
    regd_id,
    regd_name,
    cs_id,
    cd_id,
    cpls_is_transfer,
    cpld_is_transfer,
    un_id,
    cam_from_ol_id
  from hp_cardaktmain_view(:cam_id)
  into
    :cam_status,
    :card_list,
    :card_count,
    :cpls_id,
    :cpld_id,
    :cpld_pin_id,
    :regs_id,
    :regs_name,
    :regd_id,
    :regd_name,
    :cs_id,
    :cd_id,
    :is_transfer_s,
    :is_transfer_d,
    :un_id,
    :ol_id;

  if ((cam_status = 0 and ol_id is null) or (cam_status = 1 and
    exists(
      select *
      from operlist
      where ol_id = :ol_id
        and is_delete = 1))) then
  begin
    -- кол-во карт с пин-конвертами
    pin_count = coalesce((
      select
        sum(card_count)
      from hp_cardrange_jsontolist(:card_list)
      where is_pincover = 1), 0);
    if (pin_count > 0 and cpld_pin_id is null) then
    begin
      err_code = 2;
      err_text = 'Мат.ответственное лицо для ПИН-конвертов не определено.';
      suspend;
      exception e_message err_text;
    end

    -- бух.счета
    reg_id = iif(is_transfer_s = 0 and is_transfer_d = 1, regs_id, regd_id);
    acc_96101_12 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/12');
    acc_96101_13 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/13');
    acc_96101_14 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/14');
    acc_96101_15 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/15');
    acc_96101_16 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/16');
    acc_96101_17 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/17');
    acc_96102_12 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/12');
    acc_96102_13 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/13');
    acc_96102_14 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/14');
    acc_96102_15 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/15');
    acc_96102_16 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/16');
    acc_96102_17 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/17');

    if (cpls_id = 0) then begin
      -- cpls_id = 0(оприходование из МПЦ)
      acc_dt1 = acc_96101_14; acc_kt1 = acc_96101_12;
      acc_dt2 = acc_96102_12; acc_kt2 = acc_96102_14;
      acc_pin_dt1 = acc_96101_15; acc_pin_kt1 = acc_96101_13;
      acc_pin_dt2 = acc_96102_13; acc_pin_kt2 = acc_96102_15;

      first_card = f_json_str_get_value(:card_list, 'cards[0].b', null);
      select
        ckd_name,
        ckd_is_own
      from cardlist
        inner join cardembomain
          on cem_id = cal_from_cem_id
        inner join cardkind
          on ckd_id = cem_from_ckd_id
      where cal_card_id = :first_card
      into
        :ckd_name,
        :ckd_is_own;
      ol_memo = f_format('Приход платежных карт "%1" из МПЦ по АПП № %2, в количестве %3шт.', :ckd_name, :cam_id, :card_count);
      ol_memo_pin = f_format('Приход ПИН-конвертов "%1" из МПЦ по АПП № %2, в количестве %3шт.', :ckd_name, :cam_id, :pin_count);
    end else begin
      -- type_akt = 1(передача из офиса), 2(приход в офис), 3(смена МОЛ)
      if (is_transfer_s = 0 and is_transfer_d = 1) then begin
        acc_dt1 = acc_96101_16; acc_kt1 = acc_96101_14;
        acc_dt2 = acc_96102_14; acc_kt2 = acc_96102_16;
        acc_pin_dt1 = acc_96101_17; acc_pin_kt1 = acc_96101_15;
        acc_pin_dt2 = acc_96102_15; acc_pin_kt2 = acc_96102_17;
        type_akt = 1;
        doc_kind = 0;
      end else if (is_transfer_s = 1 and is_transfer_d = 0) then begin
        acc_dt1 = acc_96101_14; acc_kt1 = acc_96102_14;
        acc_dt2 = acc_96102_16; acc_kt2 = acc_96101_16;
        acc_pin_dt1 = acc_96101_15; acc_pin_kt1 = acc_96102_15;
        acc_pin_dt2 = acc_96102_17; acc_pin_kt2 = acc_96101_17;
        type_akt = 2;
        doc_kind = 1;
      end else if (is_transfer_s = 0 and is_transfer_d = 0) then begin
        acc_dt1 = acc_96102_14; acc_kt1 = acc_96101_14;
        acc_dt2 = acc_96101_14; acc_kt2 = acc_96102_14;
        acc_pin_dt1 = acc_96102_15; acc_pin_kt1 = acc_96101_15;
        acc_pin_dt2 = acc_96101_15; acc_pin_kt2 = acc_96102_15;
        type_akt = 3;
        doc_kind = 1;
      end
      ol_memo = f_format('Перемещение платежных карт по АПП № %1, в количестве %2шт.', :cam_id, :card_count);
      if (pin_count > 0) then
        ol_memo_pin = f_format('Перемещение ПИН-конвертов по АПП № %1, в количестве %2шт.', :cam_id, :pin_count);
    end

    if (acc_dt1 is null or acc_kt1 is null or acc_dt2 is null or acc_kt2 is null
      or (pin_count > 0 and (acc_pin_dt1 is null or acc_pin_kt1 is null or acc_pin_dt2 is null or acc_pin_kt2 is null)) ) then
    begin
      err_code = 2;
      err_text = 'Некоторые счета, участвующие в бух.проводках не открыты ';
      suspend;
      exception e_message err_text;
    end

    -- операция (карты)
    insert into operlist (
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id,
      ol_document_client
     )
    values (
      iif(:type_akt = 1, :regs_id, :regd_id),
      2,
      2,
      current_date,
      :cam_id,
      :doc_kind,
      :ol_memo,
      :un_id,
      iif(:type_akt = 1, :cs_id, :cd_id)
    )
    returning ol_id, ol_datetime
    into :ol_id, :ol_datetime;

    -- проводки по картам
    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_dt1,
      :acc_kt1,
      :card_count,
      :card_count
    );

    if (type_akt = 2) then
    begin
      -- "приход в офис" карт, проводки в
      -- офисе-отправителе(их м.б. несколько)
      for
        select
          reg_id,
          count(*)
        from hp_get_cardlist_transoffices (:card_list)
        group by reg_id
      into
        :reg_id,
        :card_count1
      do
      begin
        acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/16');
        acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/16');
        insert into operlist (
          ol_parent_id,
          ol_from_reg_id,
          ol_oper_type,
          ol_oper_subtype,
          ol_date,
          ol_doc_no,
          ol_doc_kind,
          ol_memo,
          ol_user_id,
          ol_document_client
         )
        values (
          :ol_id,
          :reg_id,
          2,
          2,
          current_date,
          :cam_id,
          :doc_kind,
          f_format('Перемещение платежных карт по АПП № %1, в количестве %2шт.', :cam_id, :card_count1),
          :un_id,
          :cs_id
        )
        returning ol_id
        into :ol_id2;

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_summa,
          ob_summa_val
        )
        values (
          :ol_id2,
          :curr_code,
          1,
          :acc_dt,
          :acc_kt,
          :card_count1,
          :card_count1
        );
      end
    end else begin
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id,
        :curr_code,
        1,
        :acc_dt2,
        :acc_kt2,
        :card_count,
        :card_count
      );
    end

    if (pin_count > 0) then
    begin
      -- операция (пин-конверты)
      insert into operlist (
        ol_parent_id,
        ol_from_reg_id,
        ol_oper_type,
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_memo,
        ol_user_id,
        ol_document_client
       )
      values (
        :ol_id,
        iif(:type_akt = 1, :regs_id, :regd_id),
        2,
        2,
        current_date,
        :cam_id,
        :doc_kind,
        :ol_memo_pin,
        :un_id,
        iif(:type_akt = 1, :cs_id, :cd_id)
      )
      returning ol_id
      into :ol_id1;

      -- проводки по пин-конвертам
      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_summa,
        ob_summa_val
      )
      values (
        :ol_id1,
        :curr_code,
        1,
        :acc_pin_dt1,
        :acc_pin_kt1,
        :pin_count,
        :pin_count
      );

      if (type_akt = 2) then
      begin
        -- "приход в офис" пин-конвертов, проводки в
        -- офисе-отправителе(их м.б. несколько)
        for
          select
            reg_id,
            sum(is_pincover)
          from hp_get_cardlist_transoffices (:card_list)
          group by reg_id
          having sum(is_pincover) > 0
        into
          :reg_id,
          :pin_count1
        do
        begin
          acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96102/17');
          acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = :curr_code and al_account = '96101/17');
          insert into operlist (
            ol_parent_id,
            ol_from_reg_id,
            ol_oper_type,
            ol_oper_subtype,
            ol_date,
            ol_doc_no,
            ol_doc_kind,
            ol_memo,
            ol_user_id,
            ol_document_client
           )
          values (
            :ol_id1,
            :reg_id,
            2,
            2,
            current_date,
            :cam_id,
            :doc_kind,
            f_format('Перемещение ПИН-конвертов по АПП № %1, в количестве %2шт.', :cam_id, :pin_count1),
            :un_id,
            :cs_id
          )
          returning ol_id
          into :ol_id2;

          insert into operbook (
            ob_from_ol_id,
            ob_currency,
            ob_curr_rate,
            ob_account_dt,
            ob_account_kt,
            ob_summa,
            ob_summa_val
          )
          values (
            :ol_id2,
            :curr_code,
            1,
            :acc_dt,
            :acc_kt,
            :pin_count1,
            :pin_count1
          );
        end
      end else begin
        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_summa,
          ob_summa_val
        )
        values (
          :ol_id1,
          :curr_code,
          1,
          :acc_pin_dt2,
          :acc_pin_kt2,
          :pin_count,
          :pin_count
        );
      end
    end

    update cardaktmain set
      cam_from_ol_id = :ol_id,
      cam_ol_datetime = :ol_datetime
    where cam_id = :cam_id;
    suspend;
  end else
  begin
    err_code = 1;
    err_text = 'AПП платежных карт № ' || cam_id || ' не может быть проведен.';
    suspend;
    exception e_message err_text;
  end
end
