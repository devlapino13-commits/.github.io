CREATE PROCEDURE HP_BKI_CORR_GET_LIST(
    REP_DATE_BEGIN D_DATE)
RETURNS (
    BKP_STATUS D_INTEGER,
    PACKET_NAME D_STR_CODE,
    BKI_NAME D_STR_100,
    PACKET_TYPE_STR D_STR_255,
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    EVENT_STR D_STR_CODE,
    EVENT_DATE D_DATE,
    EVENT_ID D_INTEGER,
    BKK_ID D_ID,
    BEV_ID D_ID,
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    REC_NO D_INTEGER)
AS
declare variable object_kind d_integer;
  declare variable object_id d_id;
  declare variable action_corr_id d_integer;
begin
  action_corr_id = f_enum('bki_packet_action', 'correction', 'id');
  if (rep_date_begin is null) then
    rep_date_begin = '01.01.1900';

  for select bkp_code, bev_event_date, bkk_id, bkk_name, bkt_memo, bkl_object_kind, bkl_object_id, bkc_from_c_id,
      bev_event_id, bev_rec_no, bev_id, bkp_status
  from bki_packet, bki_packet_type, bki_kind, bki_loans, bki_clients, bki_events
  where bkp_status = -5
    and bkt_id = bkp_from_bkt_id
    and bkt_action = :action_corr_id
    and bkt_format = 0
    and bkk_id = bkt_from_bkk_id
    and bkl_from_bkp_id = bkp_id
    and bkc_from_bkl_id = bkl_id
    and bev_from_bkc_id = bkc_id
  union all
  select distinct bkp_code, bev_event_date, bkk_id, bkk_name, bkt_memo, bkl_object_kind, bkl_object_id, bkc_from_c_id,
      bev_event_id, bev_rec_no, bev_id, bkp_status
  from bki_send_data_corr, bki_events, bki_clients, bki_loans, bki_packet, bki_packet_type, bki_kind
  where bev_id = bsd_from_bev_id
    and bkc_id = bev_from_bkc_id
    and bkl_id = bkc_from_bkl_id
    and bkp_id = bkl_from_bkp_id
    and bkt_id = bkp_from_bkt_id
    and bkt_action = :action_corr_id
    and bkt_format = 0
    and bkp_status <> -5
    and bkk_id = bkt_from_bkk_id
    and bkp_date >= :rep_date_begin
  order by 1 desc, 2
  into packet_name, event_date, bkk_id, bki_name, packet_type_str, object_kind, object_id, c_id,
      event_id, rec_no, bev_id, bkp_status
  do
  begin
    k_kontract_no = null;
    k_disb_date = null;
  
    if (object_kind = 11) then
      k_kontract_no = 'Заявка';
    else
      select k_kontract_no, k_disb_date
      from kontract
      where k_id = :object_id
      into k_kontract_no, k_disb_date;
    
    event_str = f_enum('bki_event', event_id, 'name');
    c_name_fio = null;
    
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    suspend;
  end
end
