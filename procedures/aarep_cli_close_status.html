CREATE PROCEDURE AAREP_CLI_CLOSE_STATUS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_NO INTEGER)
RETURNS (
    C_LAST_BONUS_STATUS INTEGER,
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_PLASTCARD_NO VARCHAR(20),
    C_FULL_ADDRESS VARCHAR(255),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_CIRCLE INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    KS_DATE_END DATE,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    KF_OFFICER VARCHAR(100),
    KC_BONUS_TEXT VARCHAR(20),
    MONTH_CLOSE INTEGER,
    KAL_NEW_KF_NAME VARCHAR(20),
    CL_BALL D_INTEGER)
AS
declare variable cli_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  rep_no = coalesce(rep_no, 1);

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
-- Вытаскивает всех клиентов которые когда либо брали кредит но на текущую дату не активный
  for select distinct kc_from_c_id
      from kontstate, knclient, region_list(:rep_region_id, 1, 3, 1)
      where ks_from_k_id = kc_from_k_id and
            ks_state = 2 and
            kc_amount_approved > 0 and
            ks_ol_is_delete = 0 and
            ks_region_id = id_list and
            ((:rep_no = 1 and
            ks_date_begin <= :rep_date_end) or
            (:rep_no = 2 and
            ks_date_begin between :rep_date_begin and :rep_date_end))
      into :cli_id
  do
  begin
    if (not exists(select *
                   from kontstate, knclient
                   where ks_ol_is_delete = 0 and
                         ks_state = 1 and
                         :rep_date_end between ks_date_begin and ks_date_end and
                         ks_from_k_id = kc_from_k_id and
                         kc_from_c_id = :cli_id and
                         kc_amount_approved > 0)) then
    begin
      kc_bonus_text = '';
      select c_last_bonus_status, c_last_bonus_status_str
      from hp_kontstate_client_status(:cli_id, :rep_date_end)
      into :c_last_bonus_status, kc_bonus_text;

      if (:c_last_bonus_status > 0) then
      begin
--   Данные контракта
        select kc_amount_approved, kc_circle, k_kontract_no, k_disb_date, ks_date_end,
               r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, c1.c_name_fio, kal_new_kf_name,
               c2.c_name_fio, c2.c_birth_date, c2.c_pass_no,
               (select cont_value
                from hp_contdata_get_cell_main(0, :cli_id)), adl_address_str
        from knclient, kontract, kontstate, knaktlist, clients c1, region r1, clients c2, region r2, addrlist, addrclient
        where kc_from_c_id = :cli_id and
              kc_from_k_id = k_id and
              kc_amount_approved > 0 and
              ks_from_k_id = k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              :rep_date_end between kal_date and kal_date_end and
              kal_new_kf_officer = c1.c_id and
              r1.reg_unicode =:rep_region_id and
              kal_new_reg_id = r1.reg_unicode and
              r1.parent = r2.reg_unicode and
              c2.c_id = :cli_id and
              adc_from_c_id = :cli_id and
              adc_address_type = 1 and
              adc_from_adl_id = adl_id
        order by k_disb_date desc
        rows 1
        into :kc_amount_approved, :kc_circle, :k_kontract_no, :k_disb_date, :ks_date_end, :reg_name, :parent_name,
             :kf_officer, :kal_new_kf_name, :c_name_fio, :c_birth_date, :c_pass_no, :c_plastcard_no, :c_full_address;

        if (:ks_date_end is not null and
            :ks_date_end <> '31.12.9999') then
          ks_date_end = :ks_date_end + 1;
        month_close = datediff(month, :ks_date_end, :rep_date_end);

        cl_ball = f_point(:cli_id, :rep_date_end);

        suspend;
      end
    end
  end
end
