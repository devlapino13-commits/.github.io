CREATE PROCEDURE PRINT_RESH_SCHEDULE(
    CML_ID D_INTEGER)
RETURNS (
    SH_FROM_K_ID DECIMAL(18,0),
    SH_NUMBER INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_PERC_TAX DECIMAL(18,4),
    SH_TOTAL DECIMAL(18,4),
    SH_TOTAL_OS_PERC DECIMAL(18,4),
    SH_BALANCE_BEFORE DECIMAL(18,4),
    SH_BALANCE_AFTER DECIMAL(18,4),
    SH_DAYS INTEGER)
AS
declare variable sh_dd date;
declare variable cm_database_id integer;
begin
  cm_database_id = f_get_db_id();

  select cml_object_id
  from com_list
  where cml_id = :cml_id
  into sh_from_k_id;

  sh_kommis = 0;
  sh_perc_tax = 0;
  sh_days = null;
  select sum(cmr_sh_os), max(cmr_sh_date)
  from com_list_resh
  where cmr_from_cml_id = :cml_id
  into :sh_balance_before, :sh_dd;
  sh_balance_after = :sh_balance_before;

  for select cmr_sh_id, cmr_sh_date, cmr_sh_os, cmr_sh_perc
  from com_list_resh
  where cmr_from_cml_id = :cml_id and
        (cmr_sh_os <> 0 or cmr_sh_perc <> 0)
  order by cmr_sh_date, cmr_sh_id
  into :sh_number, :sh_date, :sh_os, :sh_perc
  do
  begin
    sh_balance_after = :sh_balance_after - :sh_os;
    if (sh_days is not null) then
      sh_days = cast((:sh_date - :sh_dd) as integer);
    else
      sh_days = null;

    -- НСП
    sh_perc_tax = iif(:cm_database_id in (39801, 41701), round(sh_perc / 100 * 2), 0);

    -- Общая без НСП
    sh_total_os_perc = sh_os + sh_perc;
    sh_total = sh_os + sh_perc + sh_perc_tax;

    if (sh_total <> 0) then
      suspend;

    sh_days = 0;
    sh_dd = sh_date;
    sh_balance_before = :sh_balance_before - :sh_os;
  end
end
