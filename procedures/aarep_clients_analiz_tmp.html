CREATE PROCEDURE AAREP_CLIENTS_ANALIZ_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(100),
    BARANCH_NAME VARCHAR(100),
    K_ID DECIMAL(18,0),
    LOAN_TYPE_TXT VARCHAR(20),
    IS_WOFF SMALLINT,
    IS_OS SMALLINT,
    IS_PERC SMALLINT,
    IS_FINE SMALLINT,
    DELINQ_DAY INTEGER,
    C_ID ID_TYPE,
    IN_PERIOD_DISB SMALLINT,
    BEFORE_PERIOD_DISB SMALLINT,
    K_KONTRACT_NO VARCHAR(20),
    SVZ_FIO VARCHAR(100),
    PR_CODE VARCHAR(100))
AS
declare variable reg_id_prior integer;
declare variable reg_id integer;
declare variable k_close_date date;
declare variable ol_id decimal(18,0);
declare variable loan_type smallint;
declare variable ok_type smallint;
declare variable ok_os curr_type;
declare variable ok_perc curr_type;
declare variable ok_fine curr_type;
declare variable sd_date_min date;
declare variable ok_sh_date date;
declare variable k_id_prior id_type;
declare variable is_os_woff smallint;
declare variable is_perc_woff smallint;
declare variable is_fine_woff smallint;
declare variable svz_id_prior id_type;
declare variable svz_id id_type;
begin
  reg_id_prior = 0;
  k_id_prior = 0;
  before_period_disb = 0;
  svz_id_prior = 0;
  for select id_list, ks_from_k_id, ks_date_begin, ks_from_ol_id,
--         k_kolich_client_approved,
             pr_loan_type, kc_from_c_id, k_kontract_no, ks.ks_kc_id, pr_code
      from kontstate ks, kontract, product, region_list(:rep_region_id, 1, 3, 1), knclient
      where ks_ol_is_delete = 0 and
            ks_state = 2 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_region_id = id_list and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            k_id = kc_from_k_id and
            kc_amount_approved > 0
--  and k_id = 109010015930500201
      order by id_list, ks_kc_id, ks_from_k_id
      into reg_id, k_id, k_close_date, ol_id,
--       kol_cli,
           loan_type, c_id, k_kontract_no, svz_id, pr_code
  do
  begin
--  вычислЯем регион и офис
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode
      into office_name, baranch_name;
    end

    if (svz_id_prior <> svz_id) then
    begin
      svz_id_prior = svz_id;

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :svz_id
      into :svz_fio;
    end

--  ’ип кредита
    if (loan_type = 1) then
      loan_type_txt = '1. Индив.';
    else
      loan_type_txt = '2. Группа';

--  проверка закрытую операцию
    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;

      ok_type = null;
      ok_os = null;
      ok_perc = null;
      ok_fine = null;
      sd_date_min = :k_close_date;
      is_woff = 0;
      is_os = 0;
      is_perc = 0;
      is_fine = 0;

      is_os_woff = 0;
      is_perc_woff = 0;
      is_fine_woff = 0;
      for select ok_sh_date, ok_type, ok_os, ok_perc, ok_fine
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :k_id and
                ok_from_ol_id = :ol_id and
                ok_type in (4, 5, 9) and
                ok_os + ok_perc + ok_fine < 0
          into ok_sh_date, ok_type, ok_os, ok_perc, ok_fine
      do
      begin
        if (sd_date_min > ok_sh_date) then
          sd_date_min = ok_sh_date;
        if (ok_type = 5) then
        begin
          is_woff = 1;
          if (ok_os < 0) then
            is_os_woff = 1;
          if (ok_perc < 0) then
            is_perc_woff = 1;
          if (ok_fine < 0) then
            is_fine_woff = 1;
        end

        if (ok_os < 0) then
          is_os = 1;
        if (ok_perc < 0) then
          is_perc = 1;
        if (ok_fine < 0) then
          is_fine = 1;
      end

      if (is_woff = 1) then
      begin
        is_os = is_os_woff;
        is_perc = is_perc_woff;
        is_fine = is_fine_woff;
      end

      if (is_os = 1) then
      begin
        is_perc = 0;
        is_fine = 0;
      end
      else
      if (is_perc = 1) then
      begin
        is_fine = 0;
      end

      ok_type = coalesce(ok_type, 0);
      ok_os = coalesce(ok_os, 0);
      ok_perc = coalesce(ok_perc, 0);
      ok_fine = coalesce(ok_fine, 0);

      delinq_day = :k_close_date - :sd_date_min;
    end

--  вычислЯем не выдавалсЯ ли повторно за текущий период
    in_period_disb = 0;
    select 1
    from kontstate, knclient
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = :c_id and
          kc_when_prior_loan_closed = :k_close_date -- -1 нужно убрать, после того как на сервер зальем данную процедуру
    into in_period_disb;

    suspend;
  end

  reg_id_prior = 0;
  k_id_prior = 0;
  loan_type_txt = '';
  svz_id_prior = 0;

  is_woff = 0;
  is_os = 0;
  is_perc = 0;
  is_fine = 0;
  delinq_day = 0;
  in_period_disb = 0;
  before_period_disb = 0;

  for select ks_from_k_id, ks_region_id, kc_from_c_id, ks.ks_kc_id
      from kontstate ks, knclient, region_list(:rep_region_id, 1, 3, 1)
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_region_id = id_list and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_when_prior_loan_closed is not null and
            kc_when_prior_loan_closed < :rep_date_begin -- -1 нужно убрать, после того как на сервер зальем данную процедуру
      into k_id, reg_id, c_id, svz_id
  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode
      into office_name, baranch_name;
    end

    if (svz_id_prior <> svz_id) then
    begin
      svz_id_prior = svz_id;

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :svz_id
      into :svz_fio;
    end

    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;
      select pr_loan_type, pr_code
      from kontract, product
      where k_id = :k_id and
            k_from_pr_id = pr_id
      into loan_type, pr_code;
      if (loan_type = 1) then
        loan_type_txt = '1. Индив.';
      else
        loan_type_txt = '2. Группа';
    end

    before_period_disb = 1;

    suspend;
  end
end
