CREATE PROCEDURE FIND_CLIENT_BY_MB_CARD(
    CARD_ID D_INTEGER,
    FILTER_STATUS D_STR_100 = 'transfer_to_client')
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    IS_CASHING D_BOOL,
    C_PASS_NO D_STR_CODE,
    C_INN D_STR_20,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE D_STR_10)
AS
declare variable FILTER_STATUS_ID D_INTEGER;
declare variable CAL_STATUS D_INTEGER;
declare variable FILTER_STATUS_ID2 D_INTEGER;
begin
  if (filter_status <> 'all') then
  begin
    if (filter_status = 'for_block') then
    begin
      filter_status_id = f_enum('cardlist_status', 'transfer_to_client', 'id');
      filter_status_id2 = f_enum('cardlist_status', 'stoplist_in', 'id');
    end
    else
    begin
      filter_status_id = f_enum('cardlist_status', :filter_status, 'id');
      filter_status_id2 = filter_status_id;
    end
  end

  select
    cci_from_c_id,
    cl.c_name_fio,
    coalesce(crg.crg_is_cashing, 0),
    cal_status,
    cl.c_pass_no,
    cl.c_inn,
    cl.c_pass_who,
    f_datetostr(cl.c_pass_date) as c_pass_date
  from cardclientid
    inner join cardclientacc cca
      on cca_from_cci_id = cci_id
        and cca_status = 0
    inner join cardlist
      on cal_from_cca_id = cca_id
      --  and cal_status = :cal_status_2
     inner join clients cl on cl.c_id = cci_from_c_id
     left join cardriskgroups crg on crg.crg_id = cca.cca_risk_group
  where  cal_card_id = :card_id 
    and cci_status = 1
  into
    :c_id,
    :c_name_fio,
    :is_cashing,
    :cal_status,
    :c_pass_no,
    :c_inn,
    :c_pass_who,
    :c_pass_date;

  if (filter_status <> 'all' and cal_status not in(:filter_status_id, :filter_status_id2)) then
    exit;

  suspend;
end
