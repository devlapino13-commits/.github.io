CREATE PROCEDURE HP_PHONECALLS_FORM(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    FIND_TEXT D_STR_100,
    FILE_STATE D_SMALLINT)
RETURNS (
    PNC_ID D_ID,
    PNC_START D_DATETIME,
    PNC_FINISH D_DATETIME,
    PNC_DURATION D_INTEGER,
    PNC_FILENAME D_STR_255,
    PNC_IS_DELINQ D_BOOL,
    RES_SRC_PHONE_NO D_STR_20,
    RES_DST_PHONE_NO D_STR_20,
    RES_SRC_C_FIO D_STR_100,
    RES_DST_C_FIO D_STR_100,
    RES_ADD_INFO D_STR_255,
    RES_FILE_EXITS D_BOOL,
    RES_CONT_KIND D_STR_CODE,
    RES_FILENAME D_STR_LONG,
    RES_OFFICE D_STR_255)
AS
declare variable need_suspend d_bool;
declare variable pnc_src_phone_id d_integer;
declare variable pnc_dst_phone_id d_integer;
declare variable pnc_src_c_id d_id;
declare variable pnc_dst_c_id d_id;
declare variable pnc_object_kind d_integer;
declare variable pnc_object_id d_id;
declare variable pnc_is_contperson d_bool;
declare variable res_info d_str_255;
begin
  if (trim(coalesce(find_text, '')) = '') then
    if (datediff(day, rep_date_begin, rep_date_end) >=7) then
      exception e_message 'укажите доп.фильтр или период меньше недели';

  if (file_state is null) then
    file_state = 0;
  
  -- file_state:
  --  0 - все
  --  1 - только с файлами
  -- -1 - без файлов

  for select pnc_id,
             pnc_src_phone_id,
             pnc_dst_phone_id,
             pnc_src_c_id,
             pnc_dst_c_id,
             pnc_object_kind,
             pnc_object_id,
             pnc_start,
             pnc_finish,
             pnc_duration, 
             pnc_filename,
             pnc_is_delinq,
             pnc_is_contperson
      from phonecalls
      where pnc_start between :rep_date_begin and :rep_date_end + 1 and
            (:file_state = 0 or
            (:file_state = 1 and coalesce(pnc_filename, '') <> '') or
            (:file_state = -1 and coalesce(pnc_filename, '') = '')) and
            pnc_source = 0
      into :pnc_id,
           :pnc_src_phone_id,
           :pnc_dst_phone_id,
           :pnc_src_c_id,
           :pnc_dst_c_id,
           :pnc_object_kind,
           :pnc_object_id,
           :pnc_start,
           :pnc_finish,
           :pnc_duration, 
           :pnc_filename,
           :pnc_is_delinq,
           :pnc_is_contperson
  do
  begin
    res_src_phone_no = f_cont_code(pnc_src_phone_id);
    res_dst_phone_no = f_cont_code(pnc_dst_phone_id);

    res_src_c_fio = '';
    select c_name_fio
    from clients
    where c_id = :pnc_src_c_id
    into res_src_c_fio;

    res_dst_c_fio = '';
    if (:pnc_dst_c_id > 0) then
    begin
      select c_name_fio
      from clients
      where c_id = :pnc_dst_c_id
      into res_dst_c_fio;
    end

    res_add_info = '';

    if (pnc_object_kind = 0 and pnc_object_id is not null) then
    begin
      for select k_kontract_no || ' от ' || f_datetostr(k_kontract_date) || ' г.'
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
      where kc.kc_amount_approved > 0
        and kc.kc_from_c_id = :pnc_object_id
        and ks.ks_ol_is_delete = 0
        and ks.ks_state = 1
        and ks.ks_date_begin >= :rep_date_begin
        and ks.ks_date_end <= :rep_date_end
      into :res_info
      do
      begin
        res_add_info = res_add_info || res_info || '; ';
      end
    end

    res_cont_kind = 'основной';
    if (pnc_is_contperson = 1) then
      res_cont_kind = 'допол-ый';

    res_file_exits = 0; 
    res_filename = '';

    if (coalesce(pnc_filename, '') <> '') then
    begin
      res_file_exits = 1;
      res_filename = 'https://[basic_auth_ru_voice]record.molbulak.ru/' || pnc_filename;
    end

    need_suspend = 1;
    if (coalesce(find_text, '') <> '') then
    begin
      need_suspend = 0;

      if (find_text = cast(pnc_id as d_str_20) or
          res_src_phone_no containing find_text or
          res_dst_phone_no containing find_text or
          res_src_c_fio containing find_text or
          res_dst_c_fio containing find_text or
          res_add_info containing find_text) then
      need_suspend = 1;
    end

    res_office = '';
    select reg_name
    from knclient, oksummary, region
    where oks_from_k_id = kc_from_k_id
      and reg_unicode = oks_region_id
      and kc_from_c_id = :pnc_dst_c_id
    order by oks_date_begin desc
    rows 1
    into res_office;

    if (need_suspend = 1) then
      suspend;
  end
end
