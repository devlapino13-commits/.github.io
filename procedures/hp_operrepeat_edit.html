CREATE PROCEDURE HP_OPERREPEAT_EDIT(
    ID D_INTEGER,
    K_ID D_ID,
    BEGIN_DATE D_DATE,
    OL_DATE D_DATE,
    BOOK_REVERSE D_SMALLINT,
    LOAN_REVERSE D_SMALLINT,
    MEMO D_STR_LONG,
    SQL D_STR_8100,
    UN_ID D_ID)
RETURNS (
    ORP_ID D_INTEGER,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable orp_state d_smallint;
begin
  res_is_error = 1;
  res_text = '';
  orp_id = coalesce(id, 0);

  if (orp_id = 0) then
  begin
    /*
     0 - Новый запрос
     1 - В обработке
     2 - Скрипт установлен
     3 - Проверен
     4 - Обработан
     -1  Запрос отмене
    */

    if (exists (select *
    from operrepeat
    where orp_from_k_id = :k_id and
          orp_state not in (-1, 4))) then
    begin
      res_text = 'По данному договору существует не закрытый запрос';
    end else
    begin
      insert into operrepeat (orp_from_k_id, orp_memo, orp_ol_date, orp_begin_date,
                              orp_book_reverse, orp_loan_reverse, orp_sql)
      values (:k_id, :memo, :ol_date, :begin_date, :book_reverse, :loan_reverse, :sql)
      returning orp_id, orp_state
      into :orp_id, orp_state;

      execute procedure hp_operrepeat_set_status(:orp_id, :orp_state, :un_id);

      res_is_error = 0;
      res_text = 'Запрос успешно создан';
    end
  end

  if (res_text = '') then
  begin
    if (exists (select *
    from operrepeat
    where orp_id = :orp_id and
          orp_state in (-1, 4))) then
    res_text = 'Данный запрос закрыт, редактирование запрешается';
  end

  if (res_text = '') then
  begin
    update operrepeat
    set orp_memo = :memo,
        orp_ol_date = :ol_date,
        orp_begin_date = :begin_date,
        orp_book_reverse = :book_reverse,
        orp_loan_reverse = :loan_reverse,
        orp_sql = :sql
    where orp_id = :orp_id
    returning orp_state
    into orp_state;

    execute procedure hp_operrepeat_set_status(:orp_id, :orp_state, :un_id);
    res_is_error = 0;
    res_text = 'Параметры запроса успешно изменены';
  end

  suspend;
end
