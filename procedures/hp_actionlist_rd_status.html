CREATE PROCEDURE HP_ACTIONLIST_RD_STATUS(
    REP_ACL_ID DECIMAL(18,0),
    REP_OBJECT_KIND INTEGER,
    REP_OBJECT_ID DECIMAL(18,0),
    REP_FROM_RD_ID INTEGER)
RETURNS (
    RES_RD_NAME VARCHAR(255),
    RES_RD_MEMO VARCHAR(255),
    RES_RD_ORDERNO INTEGER,
    RES_OBJECT_NAME VARCHAR(255),
    RES_OBJECT_STATUS DECIMAL(18,4),
    RES_OPR_ID DECIMAL(18,0),
    RES_OBJECT_DETAIL BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_OBJECT_MEMO BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_OBJECT_BLOB_IDS D_STR_2048,
    RES_OBJECT_BLOB_NAMES BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_HAS_BLOCK D_BOOL)
AS
declare variable acl_respons_c_id decimal(18,0);
declare variable rd_str_id varchar(255);
declare variable rd_main_code varchar(255);
declare variable storedproc_param varchar(255);
begin
  -- Процедура возвращает статус объекта по свойству RD_ID
  -- REP_OBJECT_MEMO - возвращается необходимые данные в виде мемо поля.
  --                   формула для получения лежат в RD_Str_ID
  -- RD_STR_ID - ожидается в формате PROCEDURE_NAME:PARAM_NAME
  --             вернется REP_OBJECT_MEMO и REP_BLOB_IDS
  -- REP_BLOB_IDS - тут будут переданы список блобов (через точку запятую если их несколько)

  select acl_respons_c_id
  from actionlist
  where acl_id = :rep_acl_id
  into :acl_respons_c_id;
  if (acl_respons_c_id is null) then
    acl_respons_c_id = 0;

  -- Название объекта

  res_object_name = '';
  select res_object_name
  from hp_actionlist_object_name(:rep_object_kind, :rep_object_id)
  into :res_object_name;

  -- Статус объекта
  res_object_status = 0;
  -- RefData это -1
  res_opr_id = null;
  select ovs_from_opr_id, ovs_add_value
  from hp_objectprops_get(:rep_object_kind, :rep_object_id, -1, :rep_from_rd_id, null)
  into :res_opr_id, :res_object_status;
  if (res_object_status is null) then
    res_object_status = 0;

  -- переписки по объекту
  res_object_memo = '';
  select res_memo
  from hp_actionlist_get_memo(:rep_acl_id, :res_opr_id)
  into :res_object_memo;

  -- детали объекта, блобы
  res_object_blob_ids = '';
  res_object_detail = '';
  res_rd_name = '';
  res_rd_memo = '';
  res_rd_orderno = 0;
  res_object_blob_names = '';
  res_has_block = 0;

  select rd_name, rd_memo, rd_main_code, rd_orderno, rd_str_id
  from refdata
  where rd_id = :rep_from_rd_id
  into :res_rd_name, :res_rd_memo, rd_main_code, :res_rd_orderno, rd_str_id;

  if (not(rd_main_code starting 'ActionList')) then
    rd_str_id = '';

  if (rd_str_id is null) then
    rd_str_id = '';
  if (rd_str_id = '') then
  begin
    suspend;
    exit;
  end

  -- поле RD_STR_ID содержит значение в формате [процедура:параметр]
  storedproc_param = replace(rd_str_id, 'hp_actionlist_get_1:', '');

  -- Вызов процедуры HP_ACTIONLIST_GET_1
  select res_object_detail, res_blob_ids, res_object_blob_names, res_has_block
  from hp_actionlist_get_1(:rep_object_kind, :rep_object_id, :storedproc_param)
  into :res_object_detail, res_object_blob_ids, res_object_blob_names, res_has_block;
  if (res_object_blob_names is null) then
    res_object_blob_names = '';
  if (res_object_detail = '') then
    res_object_detail = res_rd_name;

  suspend;
end
