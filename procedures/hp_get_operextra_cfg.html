CREATE PROCEDURE HP_GET_OPEREXTRA_CFG(
    RES_FORM_NAME D_STR_100 = 'OperExtraCalcForm')
RETURNS (
    RES_RD_ID D_INTEGER,
    RES_RD_CODE D_STR_255,
    RES_RD_NAME D_STR_255,
    RES_OBJECT_KIND D_STR_100,
    RES_DEFAULT_MEMO D_STR_LONG,
    XLS_COLUMN_KEYNO D_STR_10,
    XLS_COLUMN_AMOUNT D_STR_10,
    XLS_COLUMN_DETAIL D_STR_10,
    XLS_COLUMN_DATE D_STR_10,
    XLS_COLUMN_RD_NAME D_STR_10)
AS
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();

  res_object_kind = 'Kontract';
  xls_column_keyno = 'B';
  xls_column_date = 'A';
  xls_column_amount = 'C';
  xls_column_detail = 'D';
  xls_column_rd_name = 'E';

  if (res_form_name = 'OperExtraCalcForm') then
  begin
    if (db_id = 39801) then
    begin
      res_rd_code = 'OperExtraKind.ExecutiveInscription';
      for
        select
        rd_id,
        rd_name
      from refdata
      where rd_code = :res_rd_code
      into
        :res_rd_id,
        :res_rd_name
      do
      begin
        res_default_memo = 'Начисленная по выписке исполнительной надписи';
        suspend;
      end
    end

    res_rd_code = 'OperExtraKind.LawClientExpense';
    for
      select
        rd_id,
        rd_name
      from refdata
      where rd_code = :res_rd_code
    into
      :res_rd_id,
      :res_rd_name
    do
    begin
      res_default_memo = 'Начисленная гос.пошлина по судебным искам/решениям';
      suspend;
    end
  end

  if (res_form_name = 'OperExtraCancelAndRestorationForm') then  -- RFCKZ-639
  begin
    if (db_id = 39801) then
    begin
      res_rd_code = 'OperExtraKind.LawClientExpenseCancel';
      for
        select
          rd_id,
          rd_name
        from refdata
        where rd_code = :res_rd_code
        into
          :res_rd_id,
          :res_rd_name
        do
        begin
          res_default_memo = 'Отмена исполнительной надписи';
          suspend;
        end
    
      res_rd_code = 'OperExtraKind.LawClientExpenseRestoration';
      for
        select
          rd_id,
          rd_name
        from refdata
        where rd_code = :res_rd_code
      into
        :res_rd_id,
        :res_rd_name
      do
      begin
        res_default_memo = 'Восстановление по решению суда';
        suspend;
      end
    end
  end
end
