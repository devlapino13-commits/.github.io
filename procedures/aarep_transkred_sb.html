CREATE PROCEDURE AAREP_TRANSKRED_SB(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PERSONAL_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    WORKING_STATUS VARCHAR(20),
    TRANS_DATE DATE,
    TRANS_OS CURR_TYPE,
    TRANS_PERC CURR_TYPE,
    TRANS_FINE CURR_TYPE,
    TRANS_ALL CURR_TYPE,
    REG_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    OKRUG_NAME VARCHAR(60),
    KF_NAME VARCHAR(100))
AS
declare variable ps_from_c_id id_type;
declare variable id_list integer;
declare variable id_list_prior integer;
declare variable ps_from_c_id_prior id_type;
begin
  id_list_prior = 0;
  ps_from_c_id_prior = 0;
  for select jl_name, ps_from_c_id, iif(ps_is_working = 1, 'Работает', 'Уволен'), id_list
      from hr_pstatus, joblist, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = id_list and
            ps_status > -1 -- не аннулированные приказы
--    and ps_is_working = 1 -- все активные сотрдуники
            and
            ps_from_jl_id = jl_id and
            jl_kind = 8 -- Выборка только по СБ
      into jl_name, ps_from_c_id, working_status, id_list
  do
  begin
    for select ok_ol_date, -sum(ok_os), -sum(ok_perc), -sum(ok_fine), kal_new_kf_name
        from operkred, kontract, knaktlist
        where ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_os + ok_perc + ok_fine < 0 and
              ok_ol_date between :rep_date_begin and :rep_date_end and
              ok_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              ok_ol_date between kal_date and kal_date_end and
              kal_new_kf_officer = :ps_from_c_id and
              ok_from_c_id = 0 and
              (select aa_category_id
               from aarep_get_category_id(k_id, ok_ol_date - 1, null, null, null, null, null, null)) in (2, 3) and
              not exists(select *
                         from sb_akt, sb_list -- добавлено по приказу №302-06-13, пункт 2, не включать закрытые по СБ
                         where sba_is_delete = 0 and
                               sba_from_sbl_id <> 0 and
                               sba_status = 2 and
                               ok_ol_date between sba_date_begin and sba_date_end and
                               sba_from_k_id = k_id and
                               sba_from_sbl_id = sbl_id and
                               sbl_type = 1)

        group by ok_ol_date, kal_new_kf_name
        into trans_date, trans_os, trans_perc, trans_fine, kf_name
    do
    begin
      if (ps_from_c_id_prior <> ps_from_c_id) then
      begin
        ps_from_c_id_prior = ps_from_c_id;
        personal_name = (select c_name_fio
                         from clients
                         where c_id = :ps_from_c_id);

        if (id_list_prior <> id_list) then
        begin
          id_list_prior = id_list;
          select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name,
                 r3.reg_code || ' ' || r3.reg_name
          from region r1, region r2, region r3
          where r1.reg_unicode = :id_list and
                r1.parent = r2.reg_unicode and
                r2.parent = r3.reg_unicode
          into reg_name, branch_name, okrug_name;
        end
      end
      trans_all = trans_os + trans_perc + trans_fine;
      suspend;
    end
  end
end
