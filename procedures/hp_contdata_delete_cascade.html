CREATE PROCEDURE HP_CONTDATA_DELETE_CASCADE(
    CDA_ID D_ID,
    UN_ID D_ID)
AS
declare variable cda_from_c_id d_id;
declare variable cda_cont_kind d_integer;
declare variable cda_from_clt_id d_integer;
declare variable cda_status d_smallint;
declare variable cont_kind d_str_code;
declare variable rd_id d_integer;
declare variable cda_is_main d_smallint;
declare variable cda_is_delete d_smallint;
declare variable moved_enum_id d_integer;
declare variable other_cda_id d_id;
declare variable other_c_id d_id;
declare variable opr_id d_id;
declare variable other_status d_smallint;
declare variable ovs_status d_smallint;
declare variable ovs_id d_integer;
declare variable other_cell_no d_str_100;
begin
  -- Процедура для каскадного удаления совпадающих номеров у других клиентов
  -- Каскадное удаление должно вызываться только когда номер подтвержден и не удален

  select cda_from_c_id, cda_cont_kind, cda_from_clt_id, cda_is_main, cda_status, cda_is_delete
  from contdata
  where cda_id = :cda_id
  into cda_from_c_id, cda_cont_kind, cda_from_clt_id, cda_is_main, cda_status, cda_is_delete;

  if (cda_from_clt_id = 0) then
    exit;

  if (cda_status is null) then
    exception e_default f_format('cda_id not found: %1', cda_id);

  cont_kind = f_enum('cont_kind', cda_cont_kind, 'code');

  if (not(cont_kind = 'cell' and cda_is_main = 1)) then
    exit;

  if (cda_status <> 1) then
    exception e_default f_format('Incorrect cda_status: %1', cda_status);

  if (cda_is_delete = 1) then
    exception e_default f_format('Incorrect cda_is_delete: %1', cda_is_delete);

  moved_enum_id = f_enum('cont_delete_kind', 'moved', 'id');

  rd_id = f_get_rd_id('ContactInfo.cell_main');

  for select cda_id, cda_from_c_id, cda_status
  from contdata
  where cda_from_c_id <> :cda_from_c_id and
        cda_from_clt_id = :cda_from_clt_id and
        cda_cont_kind = :cda_cont_kind and
        cda_is_delete = 0 and
        cda_is_main = 1
  into other_cda_id, other_c_id, other_status
  do
  begin
    -- Удаляем контактные данные из новой структуры (ContData)
    execute procedure hp_contdata_delete(:other_cda_id, :moved_enum_id, :cda_id, :un_id);

    -- Удаляем контактные данные из старой структуры (ObjectProps)
    select opr_id
    from objectprops
    where opr_object_kind = f_objkind('clients', 'id') and
          opr_object_id = :other_c_id and
          opr_subject_kind = f_objkind('refdata', 'id') and
          opr_subject_id = :rd_id
    into :opr_id;
  
    if (opr_id is not null) then
    begin
      if (other_status = 0)
      then ovs_status = 0;
      else ovs_status = 2;

      insert into objectvalues (ovs_from_opr_id, ovs_date_begin, ovs_value, ovs_add_value, ovs_from_un_id)
      values (:opr_id, current_date, '', :ovs_status, :un_id)
      returning (ovs_id)
      into ovs_id;

      if (ovs_status = 0) then
      begin
        update objectvalues
        set ovs_add_value = 0
        where ovs_from_opr_id = :opr_id and
              ovs_id = :ovs_id and
              ovs_add_value <> 0;
      end

      if (ovs_status = 2) then
      begin
        other_cell_no = '';
        select f_cont_code(cda_from_clt_id)
        from contdata
        where cda_from_c_id = :other_c_id and
              cda_cont_kind = :cda_cont_kind and
              cda_is_delete = 0 and
              cda_is_main = 1
        into other_cell_no;
  
        if (other_cell_no <> '') then
        begin
          insert into objectvalues (ovs_from_opr_id, ovs_date_begin, ovs_date_end, ovs_value, ovs_from_un_id,
                                    ovs_datetime, ovs_is_active, ovs_add_value)
          select ovs_from_opr_id, current_date, ovs_date_end, ovs_value, ovs_from_un_id, ovs_datetime, ovs_is_active,
                 ovs_add_value
          from objectvalues
          where ovs_from_opr_id = :opr_id and
                ovs_value = :other_cell_no
          order by ovs_from_opr_id desc, ovs_id desc
          rows 1;
        end
      end
    end

    -- Аннуируем договор АСП
    if (other_status = 1) then
      execute procedure hp_edoc_cancel('asp.agreement', 'clients', :other_c_id);
  end
end
