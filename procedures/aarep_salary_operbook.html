CREATE PROCEDURE AAREP_SALARY_OPERBOOK(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    ASR_REG_ID INTEGER,
    ASR_SAC_ID INTEGER,
    ASR_ORDER INTEGER,
    ASR_VALUE NUMERIC(18,4),
    ASR_SAC_NAME VARCHAR(100),
    SAC_ACCOUNT_DT VARCHAR(20),
    SAC_ACCOUNT_KT VARCHAR(20),
    SAC_OPERBOOK_SUMMA DECIMAL(18,4),
    SAC_OPERBOOK_STORNO DECIMAL(18,4),
    SAC_SUMMA_DIFF DECIMAL(18,4),
    SAC_SALARY_SUMMA DECIMAL(18,4),
    SAC_SALARY_STORNO DECIMAL(18,4),
    SAC_HAS_ACCOUNT VARCHAR(50),
    SAC_DETAIL VARCHAR(255),
    SAC_SUMMA_OB_ALL DECIMAL(18,4),
    SAC_IS_MINUS_STORNO INTEGER)
AS
declare variable sac_dt_balance varchar(20);
declare variable sac_kt_balance varchar(20);
declare variable sac_dt_number integer;
declare variable sac_kt_number integer;
declare variable head_off integer;
declare variable parent integer;
declare variable sac_show_in_operbook integer;
declare variable curr_currency varchar(3);
begin
  select reg_head_office, parent
  from region
  where reg_unicode = :report_region_id
  into :head_off, :parent;
  if (head_off is not null) then
    report_region_id = :parent;

  select cm_default_currency
  from company
  into :curr_currency;

  for select reg_head_office, os_from_sac_id, sac_priority_order, sac_name, sac_dt_balance, sac_kt_balance,
             sac_dt_number, sac_kt_number, sac_detail, sac_show_in_operbook, sac_is_minus_storno, sum(os_value)
      from opersalary, operlist, region_list(:report_region_id, 1, -1, 1), sa_config, region
      where operlist.is_delete = 0 and
            ol_id = os_from_ol_id and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            (id_list = os_from_reg_id) and
            sac_id = os_from_sac_id and
            reg_unicode = os_from_reg_id and
            sac_show_in_operbook > 0
      group by reg_head_office, os_from_sac_id, sac_name, sac_priority_order, sac_dt_balance, sac_kt_balance, sac_dt_number, sac_kt_number, sac_detail, sac_show_in_operbook, sac_is_minus_storno
      order by reg_head_office, sac_priority_order
      into :asr_reg_id, :asr_sac_id, :asr_order, :asr_sac_name, :sac_dt_balance, :sac_kt_balance, :sac_dt_number,
           :sac_kt_number, :sac_detail, :sac_show_in_operbook, :sac_is_minus_storno, :asr_value
  do
  begin
    sac_salary_summa = 0;
    sac_operbook_summa = 0;
    sac_operbook_storno = 0;
    sac_salary_storno = 0;
    sac_summa_diff = 0;
    sac_summa_ob_all = 0;
    sac_account_dt = '';
    sac_account_kt = '';

    sac_has_account = '';
    if (sac_show_in_operbook = 1) then
    begin
      sac_has_account = ' Привязан к счетам';
      select al_id
      from acc_list
      where al_from_reg_id = :asr_reg_id and
            al_from_bg_id = :sac_dt_balance and
            al_number = :sac_dt_number and
            al_currency = :curr_currency
      into sac_account_dt;
      if (sac_account_dt is null) then
        sac_account_dt = '';

      sac_has_account = ' Привязан к счетам';
      select al_id
      from acc_list
      where al_from_reg_id = :asr_reg_id and
            al_from_bg_id = :sac_kt_balance and
            al_number = :sac_kt_number and
            al_currency = :curr_currency
      into sac_account_kt;
      if (sac_account_kt is null) then
        sac_account_kt = '';
    end
    if (sac_show_in_operbook = 2) then
      sac_has_account = 'Не привязан к счетам';
    if ((sac_account_kt <> '') and
        (sac_account_kt <> '') and
        (sac_show_in_operbook = 1)) then
    begin
      select sum(ob_summa_val)
      from operbook
      where ob_account_dt = :sac_account_dt and
            ob_account_kt = :sac_account_kt and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = :asr_reg_id and
            ob_ol_date between :report_date_begin and :report_date_end and
            ob_from_sac_id is null
      into :sac_operbook_summa;
      if (sac_operbook_summa is null) then
        sac_operbook_summa = 0;

      select sum(ob_summa_val)
      from operbook
      where ob_account_dt = :sac_account_dt and
            ob_account_kt = :sac_account_kt and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = :asr_reg_id and
            ob_ol_date between :report_date_begin and :report_date_end and
            ob_from_sac_id = :asr_sac_id
      into :sac_salary_summa;
      if (sac_salary_summa is null) then
        sac_salary_summa = 0;

      select sum(ob_summa_val)
      from operbook
      where ob_account_dt = :sac_account_kt and
            ob_account_kt = :sac_account_dt and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = :asr_reg_id and
            ob_ol_date between :report_date_begin and :report_date_end and
            ob_from_sac_id = :asr_sac_id
      into :sac_salary_storno;
      if (sac_salary_storno is null) then
        sac_salary_storno = 0;

      select sum(ob_summa_val)
      from operbook
      where ob_account_dt = :sac_account_kt and
            ob_account_kt = :sac_account_dt and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = :asr_reg_id and
            ob_ol_date between :report_date_begin and :report_date_end and
            ob_from_sac_id is null
      into :sac_operbook_storno;
      if (sac_operbook_storno is null) then
        sac_operbook_storno = 0;
    end
    if (sac_show_in_operbook = 1) then
      sac_summa_diff = :asr_value - (:sac_operbook_summa - :sac_operbook_storno + :sac_salary_summa - :sac_salary_storno);
    sac_summa_ob_all = :sac_operbook_summa - :sac_operbook_storno + :sac_salary_summa - :sac_salary_storno;
    suspend;
  end
end
