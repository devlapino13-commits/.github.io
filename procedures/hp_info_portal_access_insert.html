CREATE PROCEDURE HP_INFO_PORTAL_ACCESS_INSERT(
    OFFICE_ID D_INTEGER,
    C_ID D_ID,
    DATE_END D_DATE,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable ipa_from_un_id d_id;
declare variable ipa_from_jl_id d_integer;
begin
  res_is_error = 1;

  select un.un_id
  from username un
  where un.un_from_c_id = :c_id
  into ipa_from_un_id;

  if (coalesce(ipa_from_un_id, 0) = 0) then
  begin
    res_text = 'Пользователь указан неверно';
    suspend;
    exit;
  end

  select ps_from_jl_id
  from hr_pstatus
  where ps_from_reg_id = :office_id and
        ps_from_c_id = :c_id and
        ps_is_working = 1 and
        current_date between ps_date_begin and ps_date_end
  into ipa_from_jl_id;

  if (coalesce(ipa_from_jl_id, 0) = 0) then
  begin
    res_text = 'Должность указана неверно';
    suspend;
    exit;
  end

  if (coalesce(date_end, '01.01.1900') < current_date) then
  begin
    res_text = 'Дата закрытия доступа не может быть прошедшей датой';
    suspend;
    exit;
  end

  if (exists(select *
             from info_portal_access ipa
             where ipa.ipa_from_un_id = :ipa_from_un_id and
                   ipa.ipa_access = 1 and
                   ipa.ipa_access_date_end >= current_date)) then
  begin
    res_text = 'У пользователя уже есть доступ, сначала аннулируйте его';
    suspend;
    exit;
  end

  insert into info_portal_access (ipa_from_un_id, ipa_from_reg_id, ipa_access, ipa_access_date_end, ipa_from_jl_id, ipa_changer_un_id)
  values (:ipa_from_un_id, :office_id, 1, :date_end, :ipa_from_jl_id, :un_id);

  res_text = 'Доступ успешно добавлен';
  res_is_error = 0;
  suspend;
end
