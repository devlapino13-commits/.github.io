CREATE PROCEDURE KONTRACT_FULL_OPER_LIST_DETAIL(
    KONT_ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    OL_DATE DATE,
    OL_DOC_NO VARCHAR(10),
    OL_MEMO VARCHAR(1024),
    OL_DATETIME TIMESTAMP,
    UN_NAME VARCHAR(30),
    SUM_OF_OK_OS DECIMAL(18,4),
    SUM_OF_OK_PERC DECIMAL(18,4),
    SUM_OF_OK_FINE DECIMAL(18,4),
    SUM_OF_OK_TAX DECIMAL(18,4),
    OK_PAY_TYPE INTEGER,
    SUM_OF_OK_KOMMIS DECIMAL(18,4),
    OK_TYPE INTEGER,
    OK_PAY_TYPE_TEXT VARCHAR(20),
    OK_TYPE_TEXT VARCHAR(100),
    SUM_OF_ALL DECIMAL(18,4),
    OL_ID VARCHAR(20),
    OL_DATE_OPER_DAY DATE,
    OK_SH_DATE DATE,
    OK_SH_NUMBER INTEGER,
    SUM_OF_DEPO DECIMAL(18,4),
    KOL_DELIQ INTEGER,
    OL_CLIENT_NAME D_STR_LONG,
    KOL_DELINQ_STR INTEGER)
AS
declare variable FACT_FIO D_STR_255;
declare variable LAWCLIENTEXPENSE_ID integer;
declare variable DB_ID D_ID;
declare variable CMP_STR_NO D_STR_CODE;
declare variable CMP_DATE date;
declare variable OLD_PERCENT D_CURRENCY;
begin
  db_id = f_get_db_id();
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'OperExtraKind.LawClientExpense'
  into :lawclientexpense_id;

  for select ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, sum(ok_sh_os + ok_os), sum(ok_sh_perc + ok_perc),
             sum(ok_fine), sum(ok_tax), ok_pay_type, sum(ok_kommis + ok_kommis_adv), ok_type,
             sum(ok_sh_os + ok_os + ok_sh_perc + ok_perc + ok_fine + ok_tax + ok_kommis + ok_kommis_adv), ol_id,
             ol_date_oper_day, ot_name, ok_sh_date, ok_sh_id, cast(0 as decimal(18,4))
      from operkred ok, kroper, operlist, username
      where (ok_from_ol_id = operlist.ol_id) and
            (ol_user_id = un_id) and
            (is_delete = 0) and
            ok_from_k_id = :kont_id and
            (ok_type = ot_id) and
            ok_type in (2, 4, 5, 6, 8, 9) and
            ol_date <= :dd and
            ok_from_c_id = 0
      group by ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, ok_pay_type, ok_type, ol_id, ol_date_oper_day, ot_name, ok_sh_date, ok_sh_id
      union
      select ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, cast(0 as decimal(18,4)), cast(0 as decimal(18,4)),
             cast(0 as decimal(18,4)), cast(0 as decimal(18,4)), cast(3 as integer), cast(0 as decimal(18,4)),
             cast(4 as integer), sum(okd_summa_minus - okd_summa_plus), ol_id, ol_date_oper_day,
             cast('' as varchar(50)), ol_date as ok_sh_date, cast(0 as integer), sum(okd_summa_minus - okd_summa_plus)
      from operkreddepo, operlist, username
      where (okd_from_ol_id = operlist.ol_id) and
            (ol_user_id = un_id) and
            (is_delete = 0) and
            okd_from_k_id = :kont_id and
            ol_date <= :dd
      group by ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, ol_id, ol_date_oper_day
      union
      select ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, 0, 0, 0, 0, 0,
             sum(oe.ox_amount_plus - oe.ox_amount_minus), 4, sum(oe.ox_amount_plus - oe.ox_amount_minus), ol_id,
             ol_date_oper_day,
             iif(sum(oe.ox_amount_plus - oe.ox_amount_minus) >= 0, 'Судеб. взыскание', 'Судеб. издержки') as ok_type_text,
             ol_date as ok_sh_date, cast(0 as integer), cast(0 as decimal(18,4))
      from operextra oe, operlist ol, username un
      where oe.ox_object_kind = 9 and
            oe.ox_object_id = :kont_id and
            oe.ox_from_rd_id = :lawclientexpense_id and
            ol.ol_id = oe.ox_from_ol_id and
            ol.is_delete = 0 and
            ol_user_id = un_id and
            ol.ol_date <= :dd
      group by ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, ol_id, ol_date_oper_day
      union
      select ol_date, ol_doc_no, kd_memo as ol_memo, ol_datetime, un_name, 0, 0, 0, 0, 0,
             null, 0, null, ol_id, ol_date_oper_day, 'Снижение ставки', null, null, kd_percent
      from kontdata, operlist, username
      where kd_from_k_id = :kont_id
      and kd_ol_is_delete = 0
      and kd_memo is not null
      and kd_from_ol_id = ol_id
      and ol_user_id = un_id
      and kd_ol_date < kd_date_end
      and :db_id = 81001
      order by 1, 14, 2, 16, 18
      into :ol_date, :ol_doc_no, :ol_memo, :ol_datetime, :un_name,
           :sum_of_ok_os, :sum_of_ok_perc, :sum_of_ok_fine, :sum_of_ok_tax,
           :ok_pay_type, :sum_of_ok_kommis,
           :ok_type, :sum_of_all,
           :ol_id, :ol_date_oper_day, :ok_type_text, :ok_sh_date, :ok_sh_number, :sum_of_depo
  do
  begin
    ol_client_name = '';
    select c_name_fio, ol_doc_client_name
    from operlist, clients
    where c_id = ol_document_client and
          ol_id = :ol_id
    into :ol_client_name, :fact_fio;
    if (fact_fio is null) then
      fact_fio = '';
    if (ol_client_name is null) then
      ol_client_name = '';
    if ((fact_fio <> '') and
        (ol_client_name <> '')) then
      ol_client_name = ol_client_name || ' через ' || fact_fio;

    if (ol_client_name = '') then
    begin
      select ol_doc_client_name
      from operlist
      where ol_id = :ol_id
      into :ol_client_name;
    end

    if (ok_type_text = 'Снижение ставки') then
    begin
      old_percent = 0;
      select kd_percent from kontdata
      where kd_ol_is_delete = 0
      and kd_from_k_id = :kont_id
      and kd_ol_date < :ol_date
      order by kd_ol_date desc
      rows 1
      into old_percent;
      if (sum_of_depo = old_percent) then continue;
      if (sum_of_depo > old_percent) then ok_type_text = 'Повышение ставки';
      ol_client_name = ok_type_text||' c '||round(old_percent,2)||' до '||round(sum_of_depo,2)||' в мес. '||ol_memo;
      sum_of_depo = 0;
    end

    if (sum_of_depo is null) then
      sum_of_depo = 0;
    if (ok_pay_type = 1) then
      ok_pay_type_text = 'Налич.';
    if (ok_pay_type = 2) then
      ok_pay_type_text = 'Безнал';
    if (ok_pay_type = 3) then
      ok_pay_type_text = 'Внутр.';
    if (sum_of_ok_os is null) then
      sum_of_ok_os = 0;
    if (sum_of_ok_perc is null) then
      sum_of_ok_perc = 0;
    if (sum_of_ok_fine is null) then
      sum_of_ok_fine = 0;
    if (sum_of_ok_tax is null) then
      sum_of_ok_tax = 0;
    if (sum_of_ok_kommis is null) then
      sum_of_ok_kommis = 0;
    if (sum_of_all is null) then
      sum_of_all = 0;
    if (ol_doc_no is null) then
      ol_doc_no = '';
    if ((ok_type_text is null) or (ok_type_text = '') or (ok_type_text = cast('' as varchar(50)))) then
    begin
      if (sum_of_depo < 0) then
        ok_type_text = 'Депонирование';
      if (sum_of_depo > 0) then
        ok_type_text = 'Расход депон.';
      ok_pay_type_text = ' ';
    end
    if ((sum_of_ok_os <> 0) or (sum_of_ok_perc <> 0) or (sum_of_ok_kommis <> 0)) then
    begin
      kol_deliq = cast((ol_date - ok_sh_date) as integer);
      if (ol_date > ok_sh_date and
          not(ok_type = 9 and
          sum_of_ok_os > 0)) then
        kol_delinq_str = ol_date - ok_sh_date;
      else
        kol_delinq_str = null;
    end
    else
    begin
      kol_deliq = 0;
      kol_delinq_str = null;
    end

    -- оплата доп.процента
    if (ok_sh_number = 0) then
    begin
      kol_deliq = 0;
      kol_delinq_str = null;
    end

    if (f_get_db_id() = 81001 and
        ok_type_text = 'Списание') then
      ok_type_text = 'Перевод на спец.учет';

    if (db_id in (39801, 41701) and
        ok_type = 2) then
    begin
      if (un_name = 'FINSTART_ROBOT') then
      begin
        select cp.cmp_str_no, cp.cmp_date
        from com_list cl
        inner join com_packet cp on cp.cmp_id = cl.cml_last_cmp_id
        where cl.cml_object_kind = 9 and
              cml_last_status = 5 and
              cl.cml_object_id = :kont_id and
              cl.cml_date <= :ol_date
        order by cml_date desc
        rows 1
        into :cmp_str_no, :cmp_date;

        ol_memo = ol_memo || coalesce('Протокол ГКК №' || cmp_str_no || ' от ' || f_datetostr(:cmp_date) || '(онлайн КК).', '');
      end
    end
    suspend;
  end
end
