CREATE PROCEDURE AAREP_MINIAPP_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DATE_KIND INTEGER,
    FIND_KF_NAME VARCHAR(100),
    FIND_MA_NO INTEGER,
    FIND_C_FIO VARCHAR(100),
    APP_KIND INTEGER)
RETURNS (
    MA_FROM_REG_ID INTEGER,
    MA_ID INTEGER,
    MA_NUMBER INTEGER,
    MA_DATE_APP DATE,
    MA_DATE_OPER DATE,
    MA_FROM_KF_ID DECIMAL(18,0),
    MA_FROM_PR_ID INTEGER,
    MA_KOLICH_REPAY INTEGER,
    MA_PERCENT DECIMAL(18,7),
    MA_KOLICH_CLIENT INTEGER,
    MA_MEMO VARCHAR(255),
    MA_IS_ACTIVE INTEGER,
    MA_IS_AUTO INTEGER,
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_DATE_SAVE TIMESTAMP,
    MA_FROM_UN_ID DECIMAL(18,0),
    MA_OFFICE_ID INTEGER,
    MA_CURRENCY_ID VARCHAR(3),
    MA_NO INTEGER,
    MA_PERCENT_CATEGORY DECIMAL(18,7),
    MA_REG_NAME VARCHAR(100),
    MA_PR_NAME VARCHAR(50),
    MA_USER_NAME VARCHAR(30),
    MA_LOAN_OFFICER VARCHAR(100),
    MA_KF_NAME VARCHAR(100),
    MA_KONT_ID DECIMAL(18,0),
    MA_KONT_DATE DATE,
    MA_KONT_AMOUNT DECIMAL(18,4),
    MA_KONT_STATE INTEGER,
    K_STATUS INTEGER,
    MA_ID_STR VARCHAR(255),
    MA_KIB_SEND INTEGER,
    MA_KIB_RECIEVE INTEGER,
    MA_MANAGER_ID DECIMAL(18,0),
    MA_PLAN_DISB_TIME TIMESTAMP,
    MA_DISB_TIME_STR VARCHAR(20),
    MA_EXPERT_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    MA_KOMMIS_PERC DECIMAL(15,2),
    DISB_KIND D_STR_20,
    BLANK_LANG D_STR_20)
AS
declare variable disb_kind_id d_integer;
declare variable kol integer;
declare variable tmp_c_id decimal(18,0);
declare variable kbm_ver_info integer;
declare variable kbm_is_checked integer;
declare variable db_id d_integer;
declare variable pkb_eil_status d_integer;
begin
  db_id = f_get_db_id();
  for select ma_from_reg_id, ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay,
             ma_percent, ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save,
             ma_from_un_id, ma_office_id, ma_currency_id, ma_no, ma_percent_category, ma_manager_id, ma_plan_disb_time,
             ma_expert_id, ma_kommis_perc
      from aarep_miniapp_list_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_date_kind, :find_kf_name, :find_ma_no, :find_c_fio), region_list(:rep_region_id, 1, -1, 1)
      where ma_office_id = id_list
      into :ma_from_reg_id, :ma_id, :ma_number, :ma_date_app, :ma_date_oper, :ma_from_kf_id, :ma_from_pr_id,
           :ma_kolich_repay, :ma_percent, :ma_kolich_client, :ma_memo, :ma_is_active, :ma_is_auto, :ma_summa_minus,
           :ma_date_save, :ma_from_un_id, :ma_office_id, :ma_currency_id, :ma_no, :ma_percent_category, :ma_manager_id,
           :ma_plan_disb_time, :ma_expert_id, ma_kommis_perc
  do
  begin
    k_kontract_no = '';
    select pr_name, reg_name, kal_new_kf_name, c_name_fio, un_name
    from product, region, knaktlist, clients, username
    where pr_id = :ma_from_pr_id and
          reg_unicode = :ma_office_id and
          knaktlist.is_delete = 0 and
          kal_from_kf_id = :ma_from_kf_id and
          :rep_date_end between kal_date and kal_date_end and
          kal_new_kf_officer = c_id and
          un_id = :ma_from_un_id
    into :ma_pr_name, :ma_reg_name, :ma_kf_name, :ma_loan_officer, :ma_user_name;

    ma_kont_id = null;
    ma_kont_amount = 0;
    ma_kont_date = null;
    ma_kont_state = null;
    k_status = null;
    select k_id, k_amount_approved, k_disb_date, k_status, k_kontract_no
    from kontract
    where k_from_ma_app_id = :ma_id and
          k_from_kf_id = :ma_from_kf_id
    order by k_disb_date desc
    rows 1
    into :ma_kont_id, :ma_kont_amount, :ma_kont_date, :k_status, :k_kontract_no;


    disb_kind = '';
    if (db_id in (39801, 41701)) then
    begin
      if (:ma_id > 0) then
      begin
        select m.ma_disb_kind
        from miniapp m
        where ma_id = :ma_id and
              m.ma_date_end = '31.12.9999' rows 1
        into :disb_kind_id;
      end
      else
      begin
        select k.k_disb_kind
        from kontract k
        where k.k_id = :ma_kont_id
        into :disb_kind_id;
      end
      disb_kind = (case disb_kind_id
                     when 1 then 'Наличными'
                     when 6 then 'Карта Мбулак'
                     else 'Неизвестно'
                   end);
    end

    if (ma_kont_id is not null) then
    begin
      ma_is_auto = 1;

      select distinct ks_state
      from kontstate
      where ks_from_k_id = :ma_kont_id and
            ks_ol_is_delete = 0 and
            ks_date_end >= '31.12.2999'
      into :ma_kont_state;

      if (ma_kont_state = 0) then -- если в договор в процессе
      begin
        if (k_status = -2) then
          ma_kont_state = -1; -- Заявка
        if (k_status = -1) then
          ma_kont_state = -2; -- Аннулирован
        if (k_status = 0) then
          ma_kont_state = 0; -- В процессе
      end
      if (ma_kont_state = 1) then
        ma_kont_state = 1; -- Активный
      if (ma_kont_state = 2) then
        ma_kont_state = 2; -- Закрытый
    end
    else
    begin
      if (ma_is_active = 1) then
        ma_kont_state = -1; -- Заявка
      else
        ma_kont_state = -2; -- Дог.анналуирован
      k_status = -2; -- Заявка
    end

    ma_id_str = :ma_id || '; ' || :ma_number;
    if ((app_kind = 0) or ((app_kind = 1) and
        (ma_is_active = 1)) or ((app_kind = 2) and
        (ma_is_active <= 0))) then
    begin
      kol = kol + 1;
      if (exists(select *
                 from kibminiapp
                 inner join kibminiapp_link on kbl_from_kbm_id = kbm_id
                 where kbl_object_kind = f_objkind('miniapp', 'id') and
                 kbl_object_id = :ma_id)) then
        ma_kib_send = 1;
      else
        ma_kib_send = 0;

      ma_kib_recieve = 0;
      for select distinct kbm_id, kbm_ver_info, kbm_is_checked
          from kibminiapp
          inner join kibminiapp_link on kbl_from_kbm_id = kbm_id
          where kbl_object_kind = f_objkind('miniapp', 'id') and
                kbl_object_id = :ma_id
          into :tmp_c_id, :kbm_ver_info, :kbm_is_checked
      do
      begin
        if (kbm_ver_info in (3, 4, 5)) then
          ma_kib_recieve = ma_kib_recieve + 1;
        else
        if (kbm_is_checked = 1) then
          ma_kib_recieve = ma_kib_recieve + 1;
        else
        begin
          if (exists(select *
                     from kibhistory
                     where kh_from_kbm_id = :tmp_c_id)) then
            ma_kib_recieve = ma_kib_recieve + 1;
        end
      end
      if (ma_kib_recieve > 0) then
      begin
        if (ma_kib_recieve >= ma_kolich_client) then
          ma_kib_recieve = 1;/* ответ полный */
        else
          ma_kib_recieve = 2;/* ответ частиынй */
      end
      ma_disb_time_str = '';
      if (ma_plan_disb_time is not null) then
      begin
        ma_disb_time_str = right('0' || extract(hour from ma_plan_disb_time), 2) || ':' || right('0' || extract(minute from ma_plan_disb_time), 2);
      end

      blank_lang = null;
      if (db_id = 41701) then
        blank_lang = (
          select first 1
            iif(c_blank_lang = 'kg', 'кыргызский', 'русский')
          from miniappclient
            inner join clients
              on c_id = mac_from_c_id
          where mac_from_ma_id = :ma_id
            and mac_is_cashier = 1);

      if (db_id = 39801) then
      begin
        pkb_eil_status = null;

        select first 1 eil.eil_status
        from extinfolist eil
        inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        where eic.eic_code = 'pkb_cib_report' and
              eil.eil_object_kind = f_objkind('miniapp', 'id') and
              eil.eil_object_id = :ma_id
        order by eil.eil_status_datetime desc
        into :pkb_eil_status;

        if (pkb_eil_status is not null) then
            ma_kib_send = 1;

        if (pkb_eil_status = 2) then
            ma_kib_recieve = 1;
      end

      suspend;
    end
  end
end
