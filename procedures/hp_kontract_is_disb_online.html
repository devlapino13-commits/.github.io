CREATE PROCEDURE HP_KONTRACT_IS_DISB_ONLINE(
    K_ID D_ID)
RETURNS (
    IS_DISB_ONLINE D_BOOL)
AS
declare variable edoc_kind d_integer;
declare variable edoc_date d_date;
declare variable disb_date d_date;
declare variable c_id d_id;
declare variable base_k_id d_id;
declare variable ok_pay_type d_integer;
declare variable db_id d_integer;
begin
  -- Процедура возвращает вариант выдачи займа: онлайн или офлайн.

  -- Выданные займы должны классифицироваться в категорию «онлайн займов»,
  -- если они ОДНОВЕРЕННО отвечают ВСЕМ следующим критериям:
  -- 1)  Займ выдан в период действующего договора АСП;
  -- 2)  Займ является вторым, третьим, четвертым, пятым и т.д. после
  --      заключения договора АСП (Первый займ, в рамках выдачи которого был заключен договор АСП, не считается).
  -- 3)  Договор займа подписан электронно;
  -- 4)  Выдача была произведена безналичным способом (любым не через кассу).

  db_id = f_get_db_id();
  is_disb_online = 1;

  edoc_kind = f_edoc_kind('loan.agreement', 'id');

  if (edoc_kind is null) then -- если электронные документы не запущены, то не онлайн
    is_disb_online = 0;

  if (is_disb_online = 1) then
  begin
    select k_base_k_id
    from kontract
    where k_id = :k_id
    into base_k_id; -- Если амнистированный займ, будем смотреть на основную

    -- Проверка подписания через онлайн документ, пункт 3
    if (not exists(select * from edoc_list
    where edl_object_kind = 9 and -- Kontract
          edl_object_id = coalesce(:base_k_id, :k_id) and
          edl_from_edc_id = :edoc_kind and
          edl_status = 1)) then
      is_disb_online = 0;
  end

  -- Проверка выдачи через кассу офиса, пункт 4.
  if (is_disb_online = 1) then
  begin
    select ok_ol_date, ok_pay_type
    from operkred
    where ok_from_k_id = coalesce(:base_k_id, :k_id) and
          ok_ol_is_active = 1 and
          ok_type = 8 and -- Выдача
          ok_os > 0
    order by ok_ol_date
    rows 1
    into disb_date, ok_pay_type;

    if (disb_date is null) then
      is_disb_online = 0;

    if (ok_pay_type = 1) then
      is_disb_online = 0;
  end

  if (is_disb_online = 1 and disb_date is not null) then
  begin
    edoc_kind = f_edoc_kind('asp.agreement', 'id');

    -- Проверяем каждого клиента на наличие подписанного договора АСП на момент выдачи
    for select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into c_id
    do
    begin
      if (is_disb_online = 1) then
      begin
        edoc_date = null;
        -- Проверка наличие подписанного договора АСП на дату выдачи, пункт 1
        select edl_blank_date
        from edoc_list
        where edl_object_kind = 0 and -- Clients
              edl_object_id = :c_id and
              edl_from_edc_id = :edoc_kind and
              edl_status = 1 and
              edl_blank_date <= :disb_date and
              coalesce(edl_date_end, :disb_date) >= :disb_date
        order by edl_blank_date desc
        rows 1
        into edoc_date;
  
        if (edoc_date is null)
        then is_disb_online = 0;
        else
        begin
          -- Если есть подписанный договор АСП, проверяем имеются ли другие выдачи, пункт 2
          if (not exists(select *
                         from knclient
                         where kc_from_c_id = :c_id and
                               kc_date_disb between :edoc_date and :disb_date and
                               kc_from_k_id <> :k_id and
                               kc_amount_approved > 0)) then
          is_disb_online = 0;
        end
      end
    end
  end

  -- Проверка по продукту и по дате .
  if (is_disb_online = 1) then
  begin
    if (db_id = 81001) then
    begin
      if (exists(select *
                 from kontract
                 inner join product  on pr_id = k_from_pr_id
                 where k_id = :k_id and
                       pr_code containing '.reliable_emp.' and
                       k_fact_disb_date >= '24.04.2023')) then
        is_disb_online = 0;
    end
  end

  suspend;
end
