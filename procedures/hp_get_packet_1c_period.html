CREATE PROCEDURE HP_GET_PACKET_1C_PERIOD(
    PCK_KIND INTEGER)
RETURNS (
    PCK_ID D_INTEGER,
    PCK_MONTH D_STR_CODE,
    PCK_DATE DATE,
    PCK_INVALID D_INTEGER,
    PCK_EXCEL_DONE D_INTEGER,
    UN_NAME D_STR_CODE)
AS
begin
  for select pck_id, extract(year from pck_date) || '-' || right('0' || extract(month from pck_date), 2) || ' ' || trim(
             case extract(month from pck_date)
               when 1 then 'Январь'
               when 2 then 'Февраль'
               when 3 then 'Март'
               when 4 then 'Апрель'
               when 5 then 'Май'
               when 6 then 'Июнь'
               when 7 then 'Июль'
               when 8 then 'Август'
               when 9 then 'Сентябрь'
               when 10 then 'Октябрь'
               when 11 then 'Ноябрь'
               when 12 then 'Декабрь'
             end),
             pck_date, pck_invalid, pck_excel_done, coalesce(un_name, '')
      from packet_1c
      left join username on pck_un_id = un_id
      where pck_kind = :pck_kind
      order by pck_date desc
      into pck_id, pck_month, pck_date, pck_invalid, pck_excel_done, un_name
  do
  begin
    pck_invalid = coalesce(pck_invalid, 0);
    pck_excel_done = coalesce(pck_excel_done, 0);
    suspend;
  end
end
