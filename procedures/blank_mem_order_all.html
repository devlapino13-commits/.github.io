CREATE PROCEDURE BLANK_MEM_ORDER_ALL(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    OL DECIMAL(18,0),
    POCKET_ID INTEGER,
    REG_ID INTEGER,
    MEM_DATE DATE,
    MEM_ID DECIMAL(18,0),
    MEM_NUMBER INTEGER,
    MEM_CURRENCY VARCHAR(3),
    MEM_DT VARCHAR(20),
    MEM_KT VARCHAR(20),
    MEM_SUMMA DECIMAL(18,4),
    MEM_CLIENT_DT VARCHAR(255),
    MEM_CLIENT_KT VARCHAR(255),
    MEM_DETAIL VARCHAR(1024),
    BL_ID DECIMAL(18,0),
    BL_DATE DATE,
    BL_DOC_NO INTEGER,
    BL_MEMO VARCHAR(1024),
    BL_OFFICE_FULL_NAME VARCHAR(255),
    BL_POCKET_NAME VARCHAR(100))
AS
declare variable db_id integer;
begin

  :db_id = f_get_db_id();
  for select distinct ol_id, ol.ol_pocket, ol.ol_from_reg_id, ol_date_oper_day
      from operlist ol, operbook ob, region_list(:rep_region_id, 1, -1, 1)
      where ol.ol_date = :rep_date and
            ol.is_delete = 0 and
            ol_id = ob_from_ol_id and
            ol_from_reg_id = id_list and
            ((('11209' in (left(ob_short_acc_dt, 5), left(ob_short_acc_kt, 5)) or
            '21199/99' in (ob_short_acc_dt, ob_short_acc_kt)) and
            :db_id in (39801, 41701)) or
            (:db_id = 94401 and
            ob_short_acc_dt in ('21110/1', '21140/1', '21210/1', '21240/1')))
      into ol, pocket_id, reg_id, mem_date
  do
  begin
    bl_id = null;
    bl_date = null;
    bl_doc_no = null;
    bl_memo = null;
    bl_office_full_name = null;
    bl_pocket_name = null;
    select bl_id, bl_date, bl_doc_no, bl_memo, bl_office_full_name, bl_pocket_name
    from blank_mem_order_main(:ol, :pocket_id, :reg_id, :mem_date)
    into :bl_id, :bl_date, :bl_doc_no, :bl_memo, :bl_office_full_name, :bl_pocket_name;

    mem_id = null;
    mem_number = null;
    mem_currency = null;
    mem_dt = null;
    mem_kt = null;
    mem_summa = null;
    mem_client_dt = null;
    mem_client_kt = null;
    mem_detail = null;
    for select mem_id, mem_number, mem_currency, mem_dt, mem_kt, mem_summa, mem_client_dt, mem_client_kt, mem_detail
        from blank_mem_order_detail(:ol, :pocket_id, :reg_id, :mem_date)
        into :mem_id, :mem_number, :mem_currency, :mem_dt, :mem_kt, :mem_summa, :mem_client_dt, :mem_client_kt,
             :mem_detail
    do
    begin
      suspend;
    end
  end
end
