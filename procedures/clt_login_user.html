CREATE PROCEDURE CLT_LOGIN_USER(
    UID D_STR_CODE)
RETURNS (
    ERROR_TEXT D_STR_255,
    KC_ID D_ID,
    C_ID D_ID,
    CONS_LEVEL D_INTEGER,
    LOAN_TARGET_CODE D_STR_255,
    CLIENT_NAME D_STR_100,
    LEVEL_NAME D_STR_64,
    LEVEL_NAME_KG D_STR_64)
AS
declare variable const_err d_str_255;
declare variable object_kind d_integer;
declare variable object_id d_id;
declare variable k_id d_id;
declare variable rd_id d_integer;
declare variable payed_summa d_currency;
declare variable price d_currency;
declare variable k_disb_date date;
declare variable access_time d_integer;
declare variable is_test_uid smallint;
declare variable reg_id d_integer;
begin
  const_err = 'У пользователя нет доступа на данный ресурс';
  error_text = '';

  kc_id = 0;
  cons_level = 0;
  client_name = '';
  loan_target_code = 0;
  c_id = 0;

  begin
      select udl_object_kind, udl_object_id
      from uuidlist
      where udl_uuid = char_to_uuid(:uid)
      into :object_kind, :object_id;
      when any do
      begin
        error_text = 'Идентификатор пользователя передан не корректно';
        suspend;
        exit;
      end
  end

  if ((object_id is null) or
     (object_kind <> f_objkind('knclient', 'id'))) then
  begin
    error_text = const_err;
    suspend;
    exit;
  end

  is_test_uid = 0;

  select kncl.kc_id, kncl.kc_from_c_id, rd.rd_code, kncl.kc_from_k_id
  from knclient kncl
  left join refdata rd on kncl.kc_loan_target = rd.rd_id
  where kncl.kc_id = :object_id
  and kncl.kc_amount_approved > 0
  into :kc_id, :c_id, :loan_target_code, :k_id;

  /******Для тестов Марата*****/
  if (uid = '51C5E9D0-961C-4731-B6DE-416D22D78A1A') then
  begin
    error_text = '';
    cons_level = 0;
    client_name = 'Иван';
    level_name = '«Начинающий хозяин»';
    level_name_kg = '«Баштоочу ишкер»';
    suspend;
    exit;
  end

  if (uid = '93FFBDF1-8713-46F2-9891-1789E8114831') then
  begin
    error_text = '';
    cons_level = 1;
    client_name = 'Пётр';
    level_name =  '«Умный хозяин»';
    level_name_kg = '«Акылдуу кожоюн»';
    suspend;
    exit;
  end

  if (uid = '60C861F7-DA03-40C4-94CC-67522AFD82ED') then
  begin
    error_text = '';
    cons_level = 2;
    client_name = 'Мария';
    level_name = '«Зажиточная хозяйка»';
    level_name_kg = '«Бакубат кожойке»';
    suspend;
    exit;
  end
  
  if (uid = '2BB4443E-8B01-4CE2-8AA8-1ADD2D9E7838') then
  begin
    error_text = '';
    cons_level = 3;
    client_name = 'Асель';
    level_name = '«Начинающий предприниматель»';
    level_name_kg = '«Жаңы баштаган ишкер»';
    suspend;
    exit;
  end

  if (uid = 'ED4EB7F3-E185-4DB0-AD64-E49DF10C781B') then
  begin
    error_text = '';
    cons_level = 4;
    client_name = 'Аман';
    level_name = '«Успешный предприниматель»';
    level_name_kg = '«Ийгиликтүү ишкер»';
    suspend;
    exit;
  end

   --UID старой консультации у которой больше нет доступа на платформу
  if (uid = 'F9F73188-EDF9-4EA7-9A49-93CA1CC8AB9F') then
  begin
    error_text = '';
    cons_level = 5;
    client_name = 'Марат';
    level_name = '«Продвинутый предприниматель»';
    level_name_kg = '«Өркүнд�©г�©н ишкер»';
    suspend;
    exit;
  end
  /************Для тестов Марата********/

  if (kc_id = 0) then
  begin
    error_text = const_err;
    suspend;
    exit;
  end

  k_disb_date = null;

  select kont.k_disb_date, kont.k_last_reg_id
  from kontract kont
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where kont.k_id = :k_id
  and pr.pr_code containing '.CONSULTATION.'
  into :k_disb_date, :reg_id;

  if ((k_disb_date is null) and (is_test_uid = 0)) then
  begin
    error_text = const_err;
    suspend;
    exit;    
  end

  rd_id = f_get_rd_id('CashTransactionIncom.Consultation');

  payed_summa = 0;

  select
      sum(ok.ok_amount)
    from operkontract ok
      inner join operlist ol
        on ol.ol_id = ok.ok_from_ol_id
          and ol.is_delete = 0
          and ok.ok_from_k_id = :k_id
          and ok.ok_operation_type = :rd_id
          and ol_document_client = :c_id
  into
  :payed_summa;

  payed_summa = coalesce(payed_summa, 0);

  select cons_level, price, coalesce(access_time,0), T.level_name, T.level_name_kg
  from clt_get_price_for_consultation(:kc_id) T
  into cons_level, price, access_time, level_name, level_name_kg;

  if (uid = 'ED4EB7F3-E185-4DB0-AD64-E49DF10C781B') then  -- для теста Марата
    cons_level = 0;

  if ((payed_summa = 0) and (is_test_uid = 0)) then
  begin
    error_text = 'Доступ закрыт. Оплата за консультацию не произведена';
    suspend;
    exit; 
  end
  else if ((payed_summa < price) and (is_test_uid = 0)) then
  begin
      error_text = 'Доступ закрыт. Оплата за консультацию произведена не в полном объеме.';
      suspend;
      exit; 
  end

  if ((is_test_uid = 0) and (current_date > (dateadd(month, coalesce(:access_time, 0), coalesce(:k_disb_date, cast('01.01.1900' as date)))))) then
  begin
      error_text = 'Закончился срок доступа к порталу';
      suspend;
      exit; 
  end

  select fl.fl_name2
  from clients cl
  inner join fiolist fl on fl.fl_id = cl.c_from_fl_id
  where cl.c_id = :c_id
  into :client_name;

  if (is_test_uid = 0) then
  begin
      if (not exists (
          select *
          from clt_kc_visit_svod cks
          where cks.cks_from_kc_id = :object_id)) then
      begin
         insert into clt_kc_visit_svod (cks_from_kc_id, cks_office_id)
         values (:object_id, :reg_id);
      end
      else
      begin
         update clt_kc_visit_svod cks
         set cks_visit_qty = cks_visit_qty + 1
         where cks.cks_from_kc_id = :object_id;
      end
    
      insert into clt_kc_visit (ckv_from_kc_id, ckv_office_id)
      values (:object_id, :reg_id);
  end

  suspend;
end
