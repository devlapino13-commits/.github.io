CREATE PROCEDURE HP_GET_PRODUCT_LIST
RETURNS (
    MCP_ID D_ID,
    MCP_FROM_PR_ID D_INTEGER,
    MCP_TITLE D_STR_255,
    MCP_TEXT_1 D_STR_255,
    MCP_TEXT_2 D_STR_255,
    MCP_TEXT_3 D_STR_255,
    MCP_TEXT_4 D_STR_255,
    MCP_ADDITIONAL_INFO D_STR_255,
    MCP_LOAN_PURPOSE D_STR_255,
    MCP_AMOUNT_STEP D_INTEGER,
    MCP_PICTURE_URL D_STR_LONG,
    MCP_SMALL_PICTURE_URL D_STR_LONG,
    MCP_IS_ACTIVE D_SMALLINT,
    MCP_SERIAL_NUMBER D_INTEGER,
    MCP_LANG D_STR_20,
    MCP_PROJECT D_STR_100)
AS
declare variable mcp_project_int d_smallint;
begin
    for select mcp_id, mcp_from_pr_id, mcp_title, mcp_text_1, mcp_text_2, mcp_text_3, mcp_text_4,
               mcp_additional_info, mcp_loan_purpose, mcp_amount_step, mcp_picture_url, mcp_small_picture_url,
               mcp_is_active, mcp_serial_number, mcp_lang, mcp_project
        from mc_products
        where mcp_is_active = 1
        into :mcp_id, :mcp_from_pr_id, :mcp_title, :mcp_text_1, :mcp_text_2, :mcp_text_3, :mcp_text_4,
             :mcp_additional_info, :mcp_loan_purpose, :mcp_amount_step, :mcp_picture_url, :mcp_small_picture_url,
             :mcp_is_active, :mcp_serial_number, :mcp_lang, :mcp_project_int
    do
    begin
        mcp_project = case f_enum_code('mcl_project', mcp_project_int, 0)
                        when 'kmp' then 'МБулак'
                        when 'oivo' then 'ОЙВО'
                        else 'Ошибка, не определен'
                      end;

        suspend;
    end
end
