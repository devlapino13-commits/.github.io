CREATE PROCEDURE AAREP_CLIENTDOCS_VNJ(
    REP_REGION_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_100,
    KS_DATE_BEGIN D_DATE,
    CDS_CREATE_DATE D_DATE,
    K_KONTRACT_NO_LIST D_STR_LONG,
    RES_CLIENT_IS_RESIDENT D_SMALLINT,
    ACC_NO D_STR_20)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable k_kontract_no_last d_str_20;
declare variable n d_smallint;
declare variable k_fact_disb_date d_date;
declare variable office_id d_integer;
declare variable k_kontract_no_1 d_str_20;
declare variable is_suspend d_smallint;
declare variable citizen_ru_id d_integer;
declare variable doc_vnj_id d_integer;
declare variable kc_id d_id;
declare variable knr_non_resident d_smallint;
begin

  --Беру все активные офисы
  for select reg_unicode
      from region reg, region_list(:rep_region_id, 1, -1, 1)
      where reg.is_delete = 0 and
            reg_status = 3 and
            reg.reg_date_close is null and
            reg.reg_date_open <= :rep_date_end and
            (reg_unicode = id_list)
      order by reg_name
      into office_id
  do
  begin

    /*
Берем все активные контракты на текущую дату и проверяем было ли добавлено ВНЖ после выдачи контракта, если есть кандидаты то показываем.
*/
    for select ks.ks_from_k_id, knc.kc_from_c_id, ks.ks_date_begin, kc_id
        from kontstate ks
        inner join knclient knc on knc.kc_from_k_id = ks.ks_from_k_id and
              knc.kc_amount_approved > 0
        where ks.ks_state = 1 and
              ks.ks_region_id = :office_id and
              :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 /*and ks_from_k_id = 101010060080800701  */
        into :k_id, c_id, ks_date_begin, kc_id
    do
    begin
      cds_create_date = null;

      select cast(cds_create_datetime as d_date)
      from clientdocs cds
      inner join passkind pk on cds.cds_from_pk_id = pk.pk_id
      inner join passdoctype pdt on pk.pk_from_pdt_id = pdt.pdt_id
      where cds.cds_from_c_id = :c_id and
            cds.cds_is_delete = 0 and
            cast(cds_create_datetime as d_date) > :ks_date_begin and
            :ks_date_begin between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.2999') and
            pdt.pdt_code in ('registration_vnj')
      order by cds_create_datetime desc
      rows 1
      into :cds_create_date;

      if (:cds_create_date > :ks_date_begin) then
      begin
        k_kontract_no = '';
        c_name_fio = '';

        select k.k_kontract_no
        from kontract k
        where k.k_id = :k_id
        into :k_kontract_no;

        k_kontract_no_list = '';
        n = 0;
        is_suspend = 1;
        for select k.k_kontract_no, k_fact_disb_date
            from kontract k
            inner join knclient knc on knc.kc_from_k_id = k.k_id and
                  knc.kc_amount_approved > 0
            where k.k_fact_disb_date > :ks_date_begin and
                  knc.kc_from_c_id = :c_id
            order by k_fact_disb_date
            into :k_kontract_no_1, k_fact_disb_date
        do
        begin
          n = n + 1;
          if (n = 1) then
            k_kontract_no_list = k_kontract_no_1 || ' от ' || f_datetostr(k_fact_disb_date);
          else
            k_kontract_no_list = k_kontract_no_list || ', ' || k_kontract_no_1 || ' от ' || f_datetostr(k_fact_disb_date);
          if (k_fact_disb_date = cds_create_date) then
            is_suspend = 0;
        end

        select cl.c_name_fio
        from clients cl
        where cl.c_id = :c_id
        into :c_name_fio;

        if (is_suspend = 1) then
        begin
          citizen_ru_id = 0;
          select rd_id
          from refdata
          where rd_code = 'Citizen.RU'
          into :citizen_ru_id;

          doc_vnj_id = 0;
          select pk_id
          from passkind, passdoctype
          where pdt_id = pk_from_pdt_id and
                pk_is_delete = 0 and
                pdt_code = 'registration_vnj' rows 1
          into :doc_vnj_id;
          -- Резидент
          select 1 - res_non_resident
          from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :rep_date_end)
          into :res_client_is_resident;

          knr_non_resident = null;

          select knr_non_resident
          from kn_resident
          where knr_from_kc_id = :kc_id and
                :rep_date_end between knr_date_begin and knr_date_end
          into knr_non_resident;

          acc_no = null;
          if (knr_non_resident = 1) then
            acc_no = '492';
          else
          if (knr_non_resident = 0) then
            acc_no = '488';

          suspend;
        end

      end
    end
  end
end
