CREATE PROCEDURE AAREP_LOAN_OFFICERS_FIRED(
    REG_ID INTEGER,
    REP_DATE DATE,
    REP_C_ID DECIMAL(18,0))
RETURNS (
    CAN_SUSPEND INTEGER)
AS
declare variable user_name varchar(30);
declare variable person_id decimal(18,0);
begin
-- процедура выдает список кредитных специалистов от которых нужно передать папки другим,
-- при этом сотрудник не должен быть действующим КС,
-- если он действующий, значит после даты передачи должен быть приказ на увольнение или изменение должности
-- Концепт: без основания не передавать папки КС другим сотрудникам

  can_suspend = 1; -- можно возвращать

  person_id = (select un_from_c_id
               from username
               where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN'));

  if (person_id is not null) then
  begin
    if (exists(select *
               from hr_pstatus, joblist, jobkind
               where ps_from_c_id = :person_id and
                     ps_date_begin <= current_date and
                     ps_date_end >= current_date and
                     ps_status = 1 and
                     ps_from_jl_id = jl_id and
                     jl_kind = jk_id and
                     ps_is_working = 1 and
                     jk_system_code in ('JobSystem.FinStart_Support', 'JobSystem.FinStart_Programmer'))) then
    begin
      suspend;
      exit;
    end
  end

  if (exists(select *
             from hr_pstatus, joblist, jobkind -- если действующий КС
             where ps_from_c_id = :rep_c_id and
                   ps_status = 1 and
                   ps_from_reg_id = :reg_id and
                   ps_from_jl_id = jl_id and
                   jl_kind = jk_id and
                   jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
                   ps_date_begin <= :rep_date and
                   ps_date_end >= :rep_date and
                   ps_is_working = 1)) then
  begin
    can_suspend = 0; -- если действующий КС, то не будем выводить его в списке КС передающих папки

    if (exists(select *
               from hr_pstatus, joblist, jobkind
               where ps_from_c_id = :rep_c_id and
                     ps_status = 1 and
                     ps_from_jl_id = jl_id and
                     jl_kind = jk_id and
                     ps_date_begin >= :rep_date and
                     (not((jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) and
                     ps_is_working = 1) or
                     (ps_is_working = 0)))) then
    begin
-- Будущим числом есть приказ на изменение должности или увольнение,
-- значит его можно отображать в списке передающих КС
      can_suspend = 1;
    end
    if (can_suspend = 0) then
    begin
      if (exists(select *
                 from hr_pstatus, joblist, jobkind -- если действующий КС
                 where ps_from_c_id = :rep_c_id and
                       ps_status = 1 and
                       ps_from_reg_id <> :reg_id and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       ps_date_begin > :rep_date and
                       ps_is_working = 1)) then
      begin
        can_suspend = 1;
      end
      if (can_suspend = 0) then
      begin
        if (exists(select *
                   from hr_voctravel, hr_orders
                   where hrv_from_ho_id = ho_id and
                         hrv_from_c_id = :rep_c_id and
                         hrv_date_begin <= :rep_date and
                         hrv_date_end_fact >= :rep_date and
                         ho_order_type in (16, 17, 18))) then
          can_suspend = 1;
      end
    end
  end -- иначе значит он не действующий КС и можно Suspend'ить
  suspend;
end
