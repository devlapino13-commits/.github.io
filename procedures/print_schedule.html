CREATE PROCEDURE PRINT_SCHEDULE(
    PRINT_K_ID DECIMAL(18,0))
RETURNS (
    SH_FROM_K_ID DECIMAL(18,0),
    SH_NUMBER INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_PERC_TAX DECIMAL(18,4),
    SH_TOTAL DECIMAL(18,4),
    SH_BALANCE_BEFORE DECIMAL(18,4),
    SH_BALANCE_AFTER DECIMAL(18,4),
    SH_DAYS INTEGER,
    SH_KONT_KOMMIS CURR_TYPE,
    K_NSP_PERCENT DECIMAL(18,7),
    SH_TOTAL_WITH_TAX DECIMAL(18,4),
    SH_KOMMIS_TOTAL DECIMAL(18,4))
AS
declare variable sh_dd d_date;
declare variable k_disb_date d_date;
begin
  select k_amount_approved, k_disb_date,
         k_amount_approved * k_kommis_perc / 100 / k_kolich_repay
  from kontract
  where k_id = :print_k_id
  into :sh_balance_before, :sh_dd, sh_kont_kommis;
  sh_balance_after = sh_balance_before;
  k_disb_date = sh_dd;

  k_nsp_percent = 0;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  sh_from_k_id = print_k_id;

  for select sh_id, sh_date, sh_os, sh_perc, sh_kommis
      from hp_get_base_schedule(:print_k_id)
      into :sh_number, :sh_date, :sh_os, :sh_perc, :sh_kommis
  do
  begin
    sh_balance_after = :sh_balance_after - :sh_os;
    sh_days = cast((:sh_date - :sh_dd) as integer);
    select res
    from fhp_round_bank(((:sh_perc + :sh_kommis) * :k_nsp_percent / 100.0), 0)
    into :sh_perc_tax;
    sh_total = sh_os + sh_perc;
    sh_total_with_tax = sh_os + sh_perc + sh_perc_tax;

    if (sh_total <> 0) then
    begin
      suspend;
      if (sh_kommis_total <> 0) then
        sh_kommis_total = 0;
    end
    else
      sh_kommis_total = sh_kommis;

    sh_dd = sh_date;
    sh_balance_before = :sh_balance_before - :sh_os;
  end
end
