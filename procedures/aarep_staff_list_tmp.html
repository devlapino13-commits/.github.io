CREATE PROCEDURE AAREP_STAFF_LIST_TMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    PS_DATE_BEGIN DATE,
    JL_NAME VARCHAR(255),
    DEP_NAME VARCHAR(100),
    REG_CODE_NAME VARCHAR(50),
    PS_IS_ON_STAFF_TEXT VARCHAR(50),
    PS_IS_WORKING_TEXT VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    PS_FROM_C_ID DECIMAL(18,0),
    PS_FROM_REG_ID INTEGER)
AS
declare variable ps_is_on_staff smallint;
declare variable ps_is_working smallint;
declare variable dd date;
declare variable cc_id_prior decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ps_from_c_id, ps_date_begin, ps_is_on_staff, ps_is_working, jl_name, dep_name,
             r1.reg_code || ' ' || r1.reg_name, c_name_fio, ps_from_reg_id
      from hr_pstatus, region_list(:report_region_id, 1, -1, 1), joblist, depart, region r1, clients
      where (ps_from_reg_id = id_list) and
            ps_date_begin <= :report_date_end and
            ps_date_end >= :report_date_end and
            ps_from_reg_id = r1.reg_unicode and
            ps_from_dep_id = dep_id and
            ps_from_jl_id = jl_id and
            ps_from_c_id = c_id
      order by jl_order_no, c_name_fio

      into :ps_from_c_id, :ps_date_begin, :ps_is_on_staff, :ps_is_working, :jl_name, :dep_name, :reg_code_name,
           :c_name_fio, :ps_from_reg_id
  do
  begin
    if (ps_is_on_staff = 0) then
      ps_is_on_staff_text = 'Вне штатный';
    else
      ps_is_on_staff_text = 'Штатный';

    if (ps_is_working = 1) then
      ps_is_working_text = 'Работает';
    else
      ps_is_working_text = 'Уволен';

    suspend;
  end

  cc_id_prior = 0;
  for select sd_from_c_id, sd_date_begin, sd_category, sk_name, r1.reg_code || ' ' || r1.reg_name, c_name_fio,
             sd_date_closed, sd_from_reg_id
      from servicedog, region_list(:report_region_id, 1, -1, 1), region r1, clients, servicekind
      where (sd_from_reg_id = id_list) and
            sd_date_begin <= :report_date_end and
            (sd_date_end >= :report_date_end or
            sd_date_closed is null) and
            sd_from_reg_id = r1.reg_unicode and
            sd_kind = sk_id and
            sd_from_c_id = c_id
      order by sd_from_c_id, sd_date_begin desc
      into :ps_from_c_id, :ps_date_begin, :jl_name, :dep_name, :reg_code_name, :c_name_fio, :dd, :ps_from_reg_id
  do
  begin
    if (ps_from_c_id <> cc_id_prior) then
    begin
      ps_is_on_staff_text = 'Хоз.договор';

      if (dd is not null) then
      begin
        if (dd <= report_date_end) then
          ps_is_working_text = 'Закрыт';
        else
          ps_is_working_text = 'Действующий';
      end
    end
    suspend;
    cc_id_prior = :ps_from_c_id;
  end
end
