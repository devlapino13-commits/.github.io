CREATE PROCEDURE HP_CLIENTSTATE_CREATE
RETURNS (
    RES_C_ID DECIMAL(18,0))
AS
declare variable c_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable c_id_prior decimal(18,0);
declare variable ks_date_end date;
declare variable res_date date;
declare variable res_region_id integer;
declare variable res_lo_id decimal(18,0);
declare variable res_kf_name varchar(100);
declare variable res_staff_state integer;
declare variable res_client_state integer;
declare variable dd date;
declare variable kf_id_prior decimal(18,0);
declare variable cl_circle integer;
begin
-- Процедура возвращает актуальный статус клиента.

-- Нижеуказанный код должен быть выполнен отдельным скриптом, далее пересчет индексов

--  delete from clientstate;
--
--  insert into clientstate (cs_from_c_id, cs_client_state, cs_staff_state)
--  select c_id, -1, -1
--  from clients;

--  update clientstate
--  set cs_client_state=-1,
--      cs_staff_state=-1;

-- Статус сотрудника
  res_staff_state = null;
  for select ps_from_c_id, ps_is_working
      from hr_pstatus
      where ps_date_end = '31.12.9999' and
            ps_status = 1
      into res_c_id, :res_staff_state
  do
  begin
    update clientstate
    set cs_staff_state = :res_staff_state
    where (cs_from_c_id = :res_c_id);
  end
  suspend;

-- Статус клиента
  c_id_prior = 0;
  kf_id_prior = 0;
  cl_circle = 0;
  for select kc_from_c_id, ks_date_end, ks_date_begin, k_from_kf_id
      from kontstate, knclient, kontract
      where ks_from_k_id = kc_from_k_id and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id
      order by kc_from_c_id, ks_date_end desc, ks_date_begin
      into c_id, ks_date_end, res_date, kf_id
  do
  begin
    if (c_id <> c_id_prior) then
    begin
      res_c_id = c_id;
      if (kf_id <> kf_id_prior) then
      begin
        select kal_new_reg_id, kal_new_kf_officer, kal_new_kf_name
        from knaktlist
        where knaktlist.is_delete = 0 and
              kal_date_end = '31.12.9999' and
              kal_from_kf_id = :kf_id
        into :res_region_id, :res_lo_id, res_kf_name;

        kf_id_prior = kf_id;
      end
      if (ks_date_end = '31.12.9999') then
        res_client_state = 1;
      else
      begin
        res_client_state = 2;
        res_date = ks_date_end + 1;
      end

      update clientstate
      set cs_client_state = :res_client_state,
          cs_date = :res_date,
          cs_region_id = :res_region_id,
          cs_lo_id = :res_lo_id,
          cs_kf_name = :res_kf_name
      where (cs_from_c_id = :res_c_id);

      if (c_id_prior <> 0) then
      begin
        update clientstate
        set cs_client_circle = :cl_circle
        where cs_from_c_id = :c_id_prior;

        cl_circle = 0;
      end
    end
    c_id_prior = c_id;
    cl_circle = cl_circle + 1;
  end

  update clientstate
  set cs_client_circle = :cl_circle
  where cs_from_c_id = :c_id;

  suspend;

-- Заявки
  dd = current_date - 45;
  c_id_prior = 0;
  res_client_state = 0;
  for select mac_from_c_id, ma_date_app, ma_from_kf_id
      from miniapp, miniappclient, clientstate
      where ma_id = mac_from_ma_id and
            ma_is_active = 1 and
            mac_amount > 0 and
            ma_date_app >= :dd and
            ma_date_end = '31.12.9999' and
            cs_from_c_id = mac_from_c_id and
            cs_client_state <> 1
      order by 1, 2 desc
      into c_id, res_date, kf_id
  do
  begin
    if (c_id <> c_id_prior) then
    begin
      res_c_id = c_id;

      select kal_new_reg_id, kal_new_kf_officer, kal_new_kf_name
      from knaktlist
      where knaktlist.is_delete = 0 and
            kal_date_end = '31.12.9999' and
            kal_from_kf_id = :kf_id
      into :res_region_id, :res_lo_id, res_kf_name;

      update clientstate
      set cs_client_state = :res_client_state,
          cs_date = :res_date,
          cs_region_id = :res_region_id,
          cs_lo_id = :res_lo_id,
          cs_kf_name = :res_kf_name
      where (cs_from_c_id = :res_c_id);

    end
    c_id_prior = c_id;
  end
  suspend;

end
