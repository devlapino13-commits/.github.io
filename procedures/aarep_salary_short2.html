CREATE PROCEDURE AAREP_SALARY_SHORT2(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    ZP_PERSONAL_ID DECIMAL(18,0),
    ZP_PERSONAL_FIO VARCHAR(100),
    ZP_FROM_SAC_ID INTEGER,
    ZP_SC_NAME VARCHAR(100),
    ZP_SUMMA DECIMAL(18,4),
    ZP_SUMMA_1 DECIMAL(18,4),
    ZP_SUMMA_2 DECIMAL(18,4),
    ZP_SUMMA_3 DECIMAL(18,4),
    ZP_SUMMA_4 DECIMAL(18,4),
    ZP_SUMMA_5 DECIMAL(18,4),
    ZP_SUMMA_6 DECIMAL(18,4),
    ZP_SUMMA_7 DECIMAL(18,4),
    ZP_SUMMA_8 DECIMAL(18,4),
    ZP_SUMMA_9 DECIMAL(18,4),
    ZP_SUMMA_10 DECIMAL(18,4),
    ZP_SUMMA_11 DECIMAL(18,4),
    ZP_SUMMA_12 DECIMAL(18,4),
    ZP_REGION INTEGER)
AS
declare variable zp_kol integer;
declare variable zp_year integer;
declare variable zp_month integer;
declare variable report_year integer;
declare variable report_month integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable zp_year_month integer;
declare variable sa_dd date;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  report_year = extract(year from :report_date_end);
  report_month = extract(month from :report_date_end);

  zp_from_sac_id = 9;
  select sac_name
  from sa_config
  where sac_id = :zp_from_sac_id
  into :zp_sc_name;

  for select osm_from_c_id, osm_year_month, sum(osd_value)
      from opersalmain, opersaldetail
      where osd_from_osm_id = osm_id and
            osm_ol_is_active = 1 and
            osm_ol_date between :report_date_begin and :report_date_end and
            osd_from_sac_id = 9
      group by osm_from_c_id, osm_year_month
      into :zp_personal_id, :zp_year_month, :zp_summa
  do
  begin
    zp_month = mod(zp_year_month, 100);
    zp_year = (zp_year_month - zp_month) / 100;

    if (zp_month < 12) then
      sa_dd = cast(('01.' || right('0' ||(:zp_month + 1), 2) || '.' || :zp_year) as date) - 1;
    else
      sa_dd = '31.12.' || zp_year;

    select rep_region_id
    from hp_get_employer_region(:zp_personal_id, :sa_dd, :report_region_id)
    into :zp_region;

    if (zp_region is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :zp_personal_id
      into :zp_personal_fio;

      zp_kol = (:report_year * 12 + :report_month) - (:zp_year * 12 + :zp_month);
      zp_kol = 12 - zp_kol;
      zp_summa_1 = 0;
      zp_summa_2 = 0;
      zp_summa_3 = 0;
      zp_summa_4 = 0;
      zp_summa_5 = 0;
      zp_summa_6 = 0;
      zp_summa_7 = 0;
      zp_summa_8 = 0;
      zp_summa_9 = 0;
      zp_summa_10 = 0;
      zp_summa_11 = 0;
      zp_summa_12 = 0;

      if (zp_kol = 1) then
        zp_summa_1 = zp_summa;
      if (zp_kol = 2) then
        zp_summa_2 = zp_summa;
      if (zp_kol = 3) then
        zp_summa_3 = zp_summa;
      if (zp_kol = 4) then
        zp_summa_4 = zp_summa;
      if (zp_kol = 5) then
        zp_summa_5 = zp_summa;
      if (zp_kol = 6) then
        zp_summa_6 = zp_summa;
      if (zp_kol = 7) then
        zp_summa_7 = zp_summa;
      if (zp_kol = 8) then
        zp_summa_8 = zp_summa;
      if (zp_kol = 9) then
        zp_summa_9 = zp_summa;
      if (zp_kol = 10) then
        zp_summa_10 = zp_summa;
      if (zp_kol = 11) then
        zp_summa_11 = zp_summa;
      if (zp_kol = 12) then
        zp_summa_12 = zp_summa;
      suspend;
    end
  end
