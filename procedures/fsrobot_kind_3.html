CREATE PROCEDURE FSROBOT_KIND_3(
    REP_EMAIL_FROM VARCHAR(255),
    REP_EMAIL_TO VARCHAR(1024),
    REP_SUBJECT VARCHAR(1024),
    REP_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    RESULT_IS_OK INTEGER,
    RESULT_UN_FIO VARCHAR(100),
    RESULT_STR VARCHAR(255))
AS
declare variable c_email_verified integer;
declare variable result_c_id decimal(18,0);
declare variable un_id decimal(18,0);
declare variable tmp_str varchar(1024);
declare variable n integer;
declare variable mg_code varchar(255);
declare variable rd_id integer;
declare variable topic_ids varchar(255);
declare variable need_send integer;
declare variable mg_topic varchar(255);
declare variable fsm_category varchar(255);
declare variable fsm_id decimal(18,0);
declare variable name_surname varchar(100);
declare variable c_email varchar(255);
declare variable msg_body blob sub_type 1 segment size 80;
begin
  need_send = 0;
  topic_ids = '';
  result_is_ok = 1;
  result_str = '';

  rep_email_to = replace(rep_email_to, ' ', '');
-- получаем C_ID и ФИО через email
  select res_object_id, c_name_fio, res_is_verified
  from hp_contdata_find_object(:rep_email_from, 'email'), clients
  where res_object_kind = 0 and
        res_object_id = c_id rows 1
  into :result_c_id, result_un_fio, c_email_verified;

  if (result_c_id is null and
      rep_email_from in ('btolbaev@mbfplanet.com', 'ceo@molbulak.com')) then
  begin
    select c_id, c_name_fio
    from clients
    where c_name_fio starting 'Тольбаев Бабыржан' rows 1
    into :result_c_id, result_un_fio;
    c_email_verified = 1;
  end

  select name_surname
  from hp_fio_clear(:result_un_fio)
  into :name_surname;

  if (result_c_id is null) then
  begin
    result_str = 'Сотрудник с таким email не найден';
    result_is_ok = 0;
    suspend;
    exit;
  end

  if (c_email_verified <> 1) then
  begin
    result_str = 'Ваш email не подтвержден в системе.';
    result_is_ok = 0;
    suspend;
    exit;
  end

-- Ищем пользователя соответствующему User_C_ID
  select un_id
  from hp_client_un_is_active(:result_c_id)
  where un_is_active = 1
  into un_id;

  if (un_id is null) then
  begin
    result_str = 'У вас нет действующего доступа в системе.';
    result_is_ok = 0;
    suspend;
    exit;
  end

  tmp_str = rep_email_to;
  result_str = '';
  while (tmp_str <> '') do
  begin
    c_email = '';
    n = position(';', tmp_str);
    if (n > 0) then
    begin
      c_email = substring(tmp_str from 1 for n - 1);
      tmp_str = substring(tmp_str from n + 1);
    end
    else
    begin
      c_email = tmp_str;
      tmp_str = '';
    end

    if (c_email <> '') then
    begin
      mg_code = '';
      select rd_id, rd_code, rd_memo
      from refdata
      where rd_main_code = 'mailgroup' and
            rd_str_id = :c_email
      into rd_id, mg_code, mg_topic;
      if (not(mg_topic starting '/topics/')) then
        mg_topic = '/topics/' || mg_topic;

      if (coalesce(mg_code, '') <> '') then
      begin
        if (exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, :mg_code)
                   where has_access = 0)) then
        begin
          result_str = result_str || 'У вас нет доступа на <b>' || c_email || '</b><br>';
          result_is_ok = 0;
        end
        else
        begin
          if (need_send = 0) then -- Выполняем только раз
          begin
            fsm_category = '';
            insert into fsrobot_mailmain (fsm_mail_from, fsm_mail_to, fsm_subject, fsm_body, fsm_category,
                                          fsm_from_un_id)
            values (:rep_email_from, :rep_email_to, :rep_subject, :rep_body, :fsm_category, :un_id)
            returning fsm_id
            into :fsm_id;
          end
          need_send = 1;

          topic_ids = topic_ids || rd_id || ';';
          result_str = result_str || 'Успешная обработка: <b>' || c_email || '</b><br>';

-- Отправляем push (fsroboto_msgbox), но делаем ссылку на e-mail (fsrobot_msglist)
          select res_text
          from fm_json_data(:rep_subject, 'Письмо от ' || :name_surname, 'mailmessages_all', 'fsrobot_mailmain', :fsm_id, null, null, null)
          into msg_body;

          insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_status,
                                      msg_from_un_id, msg_command_kind, msg_object_kind, msg_object_id)
          values (2, :mg_topic, 'Письмо от ' || :name_surname, :msg_body, 7, 0, :un_id, 'new_mail', -1, :rd_id);
        end
      end
    end
  end
  if (need_send = 1) then
  begin

    tmp_str = topic_ids;
    while (tmp_str <> '') do
    begin
      rd_id = 0;
      n = position(';', tmp_str);
      if (n > 0) then
      begin
        rd_id = substring(tmp_str from 1 for n - 1);
        tmp_str = substring(tmp_str from n + 1);
      end
      else
      begin
        rd_id = tmp_str;
        tmp_str = '';
      end

      if (rd_id <> 0) then
      begin
-- Информация о получателях
        insert into fsrobot_mailusers (fsu_from_fsm_id, fsu_id, fsu_object_kind, fsu_object_id)
        values (:fsm_id, -1, -1, :rd_id);

-- Здесь надо еще вставку на fsrobot_masgbox отправку уведомления на Topic
      end
    end

  end
  suspend;
end
