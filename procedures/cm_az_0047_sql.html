CREATE PROCEDURE CM_AZ_0047_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id d_id;
declare variable k_id_last d_id;
declare variable max_continuing_deling d_integer;
declare variable kc_amount_approved_last d_currency;
declare variable kc_amount_approved_max d_currency;
begin

  if (exists(select *
             from kontract k
             inner join product pr on k.k_from_pr_id = pr.pr_id and
                   pr_code containing ('score_rate')
             where k.k_id = :k_id)) then
    exit;
  for select knc.kc_from_c_id, knc.kc_amount_approved, k.k_disb_date, cl.c_name_fio
      from knclient knc
      inner join clients cl on cl.c_id = knc.kc_from_c_id
      inner join kontract k on k.k_id = knc.kc_from_k_id
      inner join product pr on k.k_from_pr_id = pr.pr_id and
            pr_code not containing ('akcia')
      where knc.kc_from_k_id = :k_id
      into :c_id, :kc_amount, :kf_disb_date, :res_info_text
  do
  begin
    if (:kc_amount > 1500) then
    begin
      max_continuing_deling = 0; -- ???????????? ?????????? ???????????? ????
      kc_amount_approved_max = 0; -- c???? ?????????? ?????, ???? ?? 1500
      -- ?????? ???????? ???? ???????
      select max(ok.ok_ol_date - ok.ok_sh_date)
      from operkred ok
      where ok.ok_ol_is_active = 1 and
            ok.ok_ol_date <= :kf_disb_date and
            ok.ok_ol_date > ok_sh_date and
            ok_type < 7 and
            ok_from_k_id = :k_id_last and
            ok_os + ok_perc < 0 and
            ok_from_c_id = 0
      into :max_continuing_deling;
      max_continuing_deling = coalesce(max_continuing_deling, 0);

      if (max_continuing_deling > 30) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'клиент у которого больше 30 дней просрочки в предыдущих циклах, не может получить кредит';
        suspend;
      end

      if (exists(select *
                 from kontstate ks
                 inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
                 where kncl.kc_from_c_id = :c_id and
                       kncl.kc_amount_approved > 1500 and
                       ks.ks_state = 2 and
                       ks.ks_ol_is_delete = 0 and
                       ks.ks_from_k_id <> :k_id)) then
        exit;
      else
      begin
        res_error_is_critical = 1;
        res_error_text = 'клиент у которого сумма меньше 1500 azn предыдущих циклах не может получить кредит';
        suspend;
      end
    end
  end
end
