CREATE PROCEDURE HP_KONTUR_B2_CHECK
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable kont_err_list d_str_long;
begin

  kont_err_list = '';
  -- Группировку по контракту не делаем, т.к. на уровне клиента лишний раз не вставится
  -- Но есть плюс, добавятся залогодатели по всем автозаймам
  for select kc.kc_from_c_id, kc.kc_from_k_id
  from oksummary oks
  inner join knclient kc on oks.oks_from_k_id = kc.kc_from_k_id
  where
    current_date between oks_date_begin and oks_date_end and
    (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc + oks_out_other + oks_extra) > 0
  into c_id, k_id do
  begin

    if (f_bankrupt(c_id, current_date) = 1) then
    begin
      -- Если банкротство завершено, не отправляем
      if (exists (
          select *
          from bankrupt
          where brt_from_c_id = :c_id and
            brt_is_finished = 1 and
            current_date between brt_date_begin and brt_date_end)) then
        continue;
    end
    in autonomous transaction do
      execute procedure hp_extinfo_insert_kontur('kontract', k_id, 0, null, 1);
    when any do
      kont_err_list = left(kont_err_list || iif(kont_err_list <> '', ',', '') || k_id, 8100);
  end

  if (kont_err_list > '') then
    exception e_default 'Контракты с ошибками отправки в Контур:' || kont_err_list;
end
