CREATE PROCEDURE PRINT_INFO_NBKR_CLOSE_WE_OWE(
    KC_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    K_KONTRACT_NO D_STR_100,
    CLIENT_FIO D_STR_100,
    EDOC_DATE D_DATE,
    QR_CODE D_STR_LONG,
    K_DISB_DATE D_DATE,
    K_NEW_PERC DECIMAL(15,2))
AS
declare variable EDL_ID D_ID;
declare variable C_ID D_ID;
begin
  select kont.k_kontract_no,
         cl.c_name_fio,
         kncl.kc_from_c_id,
         kont.k_disb_date,
         round(nnl.nnl_new_perc,2)
  from knclient kncl
  inner join kontract kont on kont.k_id = kncl.kc_from_k_id
  inner join nbkr_notif_list nnl on nnl.nnl_from_kc_id = kncl.kc_id
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  where kncl.kc_id = :kc_id
  into :k_kontract_no, :client_fio, :c_id, :k_disb_date, :k_new_perc;

  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);

    select cast(edl.edl_create_datetime as date)
    from edoc_list edl where edl.edl_id = :edl_id
    into :edoc_date;

    qr_code = f_edoc_qr_by_edl(edl_id, c_id);
  end
  suspend;
end
