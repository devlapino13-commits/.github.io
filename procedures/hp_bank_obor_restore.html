CREATE PROCEDURE HP_BANK_OBOR_RESTORE(
    FIX_DATA D_BOOL,
    ONE_OL_ID D_ID)
RETURNS (
    OL_ID D_ID,
    OL_DATE D_DATE,
    OB_SUMMA D_CURRENCY,
    OL_MEMO D_STR_LONG,
    K_KONTRACT_NO D_STR_CODE,
    K_ID D_ID,
    NEW_OL_ID D_ID,
    OB_NUMBER D_INTEGER,
    MAIN_OL_ID D_ID,
    RES D_STR_255)
AS
begin
  if (fix_data is null) then
    fix_data = 0;

  for select tol_object_id, tol_add_info_1, ol_date, ol_memo
  from temp_objlist, operlist
  where tol_kind = 'fix_51_01_ol_id' and
        tol_object_kind = 9 and
        coalesce(tol_add_info_1, '') <> '' and
        tol_status = 0 and
        tol_object_id = ol_id and
        (:one_ol_id is null or ol_id = :one_ol_id) and
        ol_memo not starting 'Корректировка накопленных процентов по договору'
  into ol_id, k_kontract_no, ol_date, ol_memo
  do
  begin
    res = '';
    ob_summa = 0;
    k_id = null;
    begin
      select k_id
      from kontract
      where k_kontract_no = :k_kontract_no
      into k_id;
  
      if (k_id is null) then
        exception e_default f_format('Kontract_no not found: %1', k_kontract_no);

      when any do
      begin
        res = 'Error: Multiple k_kontract_no ' || k_kontract_no;
      end
    end

    ob_number = null;
    new_ol_id = null;
    main_ol_id = null;

    if (res = '') then
    begin
      select sum(ob_summa)
      from operbook
      where ob_from_ol_id = :ol_id and
            (cast(ob_account_dt as d_str_code) containing '51-01' or
            cast(ob_account_kt as d_str_code) containing '51-01')
      into ob_summa;
  
      if (ob_summa is null) then
        exception e_default f_format('Bank operbook not found, ol_id: %1', ol_id);

      when any do
      begin
        res = 'Error: Multiple operbook,  ol_id ' || ol_id;
      end
    end

    if (res = '') then
    begin
      if (exists(select * from operbook
          where ob_from_ol_id = :ol_id and
                ob_detail starting 'ком. за кред. займ ')) then
      begin
        select ok_from_ol_id
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date = :ol_date and
              ok_ol_is_active = 1 and
              ok_type in (4, 9)
        rows 1
        into new_ol_id;
      end else
      begin
        select ok_from_ol_id
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date = :ol_date and
              ok_ol_is_active = 1 and
              ok_type in (4, 9)
        group by 1
        having -sum(ok_os + ok_perc + ok_fine) = :ob_summa
        order by 1
        rows 1
        into new_ol_id;
      end
  
      if (new_ol_id is null) then
        exception e_default f_format('New_ol_id not found, k_id: %1', k_id);

      when any do
      begin
        res = 'Error, new_ol_id not found';
      end
    end

    if (res = '') then
    begin
      select ol_parent_id
      from operlist
      where ol_id = :new_ol_id
      into main_ol_id;
  
      when any do
      begin
        res = f_format('Main_ol_id not found, new_id: %1', new_ol_id);
      end
    end

    if (res = '' and fix_data = 1) then
    begin
      update operbook
      set ob_from_ol_id = :main_ol_id,
          ob_ol_is_active = 1
      where ob_from_ol_id = :ol_id and
            (cast(ob_account_dt as d_str_code) containing '51-01' or
            cast(ob_account_kt as d_str_code) containing '51-01');

      when any do
      begin
        res = f_format('Doublicate operbook, main_ol_id: %1, ol_id: %2', main_ol_id, ol_id);
      end
    end

    if (fix_data = 1) then
    begin
      update temp_objlist
      set tol_status = 1
      where tol_kind = 'fix_51_01_ol_id' and
            tol_object_kind = 9 and
            tol_object_id = :ol_id;
    end

    suspend;
  end
end
