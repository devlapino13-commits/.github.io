CREATE PROCEDURE HP_HAS_ACTUAL_MINIAPP(
    UN_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0))
RETURNS (
    IS_OWN_APPLICATION INTEGER)
AS
declare variable ma_id integer;
declare variable ma_is_active integer;
declare variable ma_disb_date date;
declare variable ma_office_id integer;
declare variable k_status integer;
declare variable reg_all integer;
declare variable k_state integer;
declare variable uniqtab_region integer;
declare variable uniqtab_username integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Region')
  into :uniqtab_region;
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;

  is_own_application = 0;
  for select distinct mac_from_ma_id, ma_office_id
      from miniappclient, miniapp
      where mac_from_c_id = :c_id and
            mac_amount > 0 and
            mac_from_ma_id = ma_id and
            ma_date_end = '31.12.9999' and
            ma_date_oper >= (current_date - 10) and
            ma_is_active = 1
      into :ma_id, :ma_office_id
  do
  begin
-- проверяем, есть ли активные заявки клиента за последний месяц
    k_status = 0;
    k_state = 0;
    select k_status, ks_state
    from kontract, kontstate
    where k_from_ma_app_id = :ma_id and
          ks_from_k_id = k_id and
          ks_date_end >= '31.12.2999' and
          ks_ol_is_delete = 0 rows 1
    into :k_status, k_state;
    if (k_status is null) then
      k_status = 0;

    if ((k_status <> -1) and
        (k_state = 0)) then
    begin
      for select distinct pm_subject_id
          from permissions
          where pm_subject_kind = :uniqtab_region and
                pm_object_kind = :uniqtab_username and
                pm_object_id = :un_id and
                pm_is_delete = 0 and
                (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(pm_date_end, '31.12.2999') >= current_date)
          into :reg_all
      do
      begin
        if (exists(select id_list
                   from region_orig_list(:reg_all, 1, -1, 1)
                   where id_list = :ma_office_id)) then
        begin
          is_own_application = 1;
          suspend;
          exit;
        end
      end
    end
  end
  suspend;
end
