CREATE PROCEDURE AAREP_CL_VALIDITY_DOC(
    REP_REGION_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    OFFICE_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    DOC_DATE_BEGIN DATE,
    DOC_DATE_END DATE,
    MESSAGE_TEXT VARCHAR(100),
    K_KREDLINE_END DATE,
    PK_NAME VARCHAR(100))
AS
declare variable pk_id integer;
begin
  for select distinct kc_from_c_id
      from oksummary oks, knclient, region_list(:rep_region_id, 1, -1, 1)
      where oks_close_state = 0 and
            :rep_date_begin between oks_date_begin and oks_date_end and
            oks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            (oks_region_id = id_list)
      into c_id
  do
  begin
    message_text = '';
    select k_kontract_no, k_kredline_end, reg_name
    from kontract, knclient, kontstate, region
    where k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = :c_id and
          kc_from_k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_begin between ks_date_begin and ks_date_end and
          k_last_reg_id = reg_unicode
    order by ks_date_begin
    rows 1
    into k_kontract_no, k_kredline_end, office_name;

    pk_name = null;
    for select pk.pk_id, pk_name
        from passdoctype pdt, passkind pk
        where pk_from_pdt_id = pdt_id and
              pdt_code in ('passport_local', 'passport_card', 'passport_trans', 'registration_rvp', 'registration_vnj', 'magration_card')
        into pk_id, pk_name
    do
    begin
      doc_date_begin = null;
      doc_date_end = null;

      for select distinct c_name_fio, cds_doc_date_begin, cd.cds_doc_date_end
          from clientdocs cd, clients
          where cds_from_c_id = :c_id and
                cd.cds_is_delete = 0 and
                cds_from_pk_id = :pk_id and
                cds_from_c_id = c_id
          order by cd.cds_id desc
          rows 1
          into c_name_fio, doc_date_begin, doc_date_end
      do
      begin
        if (doc_date_end is null) then
        begin
          message_text = 'Дата окончания срока действия не указан';
          suspend;
        end

        if (doc_date_end <= :rep_date_begin) then
        begin
          message_text = 'Срок действия истек';
          suspend;
        end

        if (k_kredline_end > doc_date_end and
            doc_date_end between :rep_date_begin + 1 and :rep_date_end) then
        begin
          message_text = 'Срок действия истекает через ' || datediff(day, :rep_date_begin, :doc_date_end) || '-дней';
          suspend;
        end
      end
    end
  end
end
