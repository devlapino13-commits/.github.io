CREATE PROCEDURE HP_GET_LOAN_DESCRIPTION(
    C_ID D_ID,
    MA_ID D_INTEGER)
RETURNS (
    LOAN_DESCRIPTION D_STR_LONG)
AS
declare variable k_id d_id;
declare variable temp_descr_of_loan d_str_long;
declare variable is_ipf d_bool;
declare variable k_disb_date d_date;
declare variable imei d_id;
declare variable kml_id d_id;
begin
    -- rfckg-2563
    :loan_description = '';

    select
        k.k_id,
        k.k_disb_date,
        iif(pr_code containing '.ipf.', 1, 0)
    from kontract k
    inner join product pr on k.k_from_pr_id = pr.pr_id
    where k.k_from_ma_app_id = :ma_id
    into
        :k_id,
        :k_disb_date,
        :is_ipf;

    if(f_get_const('clp_active') = 1) then
    begin
        -- если есть запись в clp_miniappclient то действуем по новому алгоритму
        if(exists
            (
                select *
                from clp_miniappclient clpm
                where clpm.cmc_from_c_id = :c_id
                    and clpm.cmc_from_ma_id = :ma_id
            )) then
        begin
            temp_descr_of_loan = '';

            -- генерирую новое значение для descr_of_loan
            select clpq.cq_name || '; ' || clpc.cc_name
            from clp_miniappclient clpm
            inner join clp_questions clpq on clpm.cmc_questions_id = clpq.cq_id
            inner join clp_clarification clpc on clpm.cmc_clarification_id = clpc.cc_id
            where clpm.cmc_from_c_id = :c_id
                and clpm.cmc_from_ma_id = :ma_id
            into :loan_description;

            if(:is_ipf = 1) then
            begin
                for select km.kml_id, ml.ml_name || ' ' || ml.ml_model
                from kontmat km
                inner join matlist ml on ml.ml_id = km.kml_from_ml_id
                where km.kml_object_kind = 9
                    and km.kml_object_id = :k_id
                into :kml_id, temp_descr_of_loan
                do
                    begin
                      imei = null;
                      select kmi.kmi_imei
                      from kontmat_imei kmi
                      where kmi.kmi_kml_id = :kml_id
                      into :imei;
                      if (coalesce(:imei, 0) > 0) then
                        temp_descr_of_loan = :temp_descr_of_loan || ' ' || :imei;

                      :loan_description = left(:loan_description || '; ' || :temp_descr_of_loan, 1024);
                    end
                :loan_description = left(:loan_description, 1024);
            end
        end
    end

    if(:loan_description = '') then
        select coalesce(res_str_values, '')
        from hp_objectprops_anketa_get('Clients', :c_id, -1, 'AnketaQuestions.descr_of_loan', :k_disb_date, null)
        into :loan_description;

    suspend;
end
