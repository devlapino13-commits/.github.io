CREATE PROCEDURE HP_OPERCASH_TERMINALS_INFO(
    OFFICE_ID D_INTEGER)
RETURNS (
    RES_TERMINAL_ID D_INTEGER,
    RES_TERMINAL_NO D_STR_100,
    RES_TERMINAL_BEGIN D_DATE,
    RES_TERMINAL_END D_DATE,
    RES_TERMINAL_MEMO D_STR_255,
    TML_LAST_DATETIME D_DATETIME,
    TML_UPDATED_AMOUNT_CLS D_CURRENCY,
    TML_UPDATED_AMOUNT_IPF D_CURRENCY,
    TML_UPDATED_AMOUNT_ALL D_CURRENCY,
    TML_UPDATED_DATETIME D_DATETIME,
    TML_MEMO D_STR_255)
AS
begin
  for select tm_id, tm_terminal_id, tm_date_begin,  tm_date_end, tm_memo
  from terminals
  where tm_office = :office_id and
        coalesce(tm_date_end, current_date) >= current_date
  into res_terminal_id, res_terminal_no, res_terminal_begin, res_terminal_end, res_terminal_memo
  do
  begin
    tml_last_datetime = null;
    tml_updated_amount_cls = 0;
    tml_updated_amount_ipf = 0;
    tml_updated_datetime = null;
    tml_memo = '-';

    select tml_last_datetime, tml_updated_amount_cls, tml_updated_amount_ipf,
           tml_updated_datetime, tml_memo
    from termilast
    where tml_from_tm_id = :res_terminal_id
    into tml_last_datetime, tml_updated_amount_cls, tml_updated_amount_ipf,
         tml_updated_datetime, tml_memo;

    tml_updated_amount_all = tml_updated_amount_cls + tml_updated_amount_ipf;

    suspend;
  end
end
