CREATE PROCEDURE HP_EXTINFO_TO_SEND_CHEK_INN(
    C_ID ID_TYPE)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable inn d_str_100;
begin
  for select cl.c_inn
      from clients cl
      where cl.c_id = :c_id
      into :inn
  do
  begin
    if (f_get_db_id() in (39801, 41701)) then
      json_value = '{' || f_json_make_pair('pin', :inn, 'str') || '}';  -- для получения фактического адреса и кол-во членов семьи по ИНН
    else
      json_value = '{' || f_json_make_pair('INN', :inn, 'str') || '}';
    suspend;
  end
end
