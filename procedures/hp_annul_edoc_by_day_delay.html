CREATE PROCEDURE HP_ANNUL_EDOC_BY_DAY_DELAY(
    UPDATE_DAYS D_INTEGER)
AS
declare variable edoc_code d_str_100;
declare variable obj_kind d_id;
declare variable obj_id d_id;
declare variable edl_id d_id;
begin
  for select edl.edl_id, edc.edc_code, edl.edl_object_kind, edl.edl_object_id
      from edoc_list edl
      inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
      where edl.edl_status = 0 and
            datediff(day, edl.edl_blank_date, current_date) > :update_days and
            edc.edc_process_kind <> 0
      into edl_id, edoc_code, obj_kind, obj_id
  do
  begin
    begin
      -- Попытка выполнить аннулирование
      execute procedure hp_edoc_cancel(:edoc_code, :obj_kind, :obj_id);
      when any do
      begin
        -- Если произошла ошибка, просто пропускаем текущий документ
        -- (пустой блок для игнорирования ошибки)
      end
    end
  end
end
