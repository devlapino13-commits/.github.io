CREATE PROCEDURE UXP$PREBLANKS_CODE_LOOKUP(
    PREBLANK_KIND D_INTEGER)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255,
    RES_INACTIVE D_STR_CODE)
AS
declare variable kind_str d_str_code;
declare variable space d_str_255;
begin
  if (coalesce(preblank_kind, 0) = 0) then
    exit;

  kind_str = f_enum('uxaction_preblank_kind', preblank_kind, 'code');
  space = '   - ';

  -- code,blank,loan,hr,edoc

  res_inactive = '';

  if (kind_str = 'code') then
  begin
    for select ubc_code, ubc_memo, ubc_inactive
    from ux$blankcode
    into :res_code, :res_text, :res_inactive
    do
    begin
      if (res_text = '') then
        res_text = res_code;
      else
        res_text = res_code || space || res_text;

      suspend;
    end
  end else
  if (kind_str = 'blank') then
  begin
    for select bln_id, bln_id || :space || bln_name
    from blankname
    order by bln_id
    into res_code, res_text
    do
    begin
      res_inactive = '';
      suspend;
    end
  end else
  if (kind_str = 'loan') then
  begin
    for select pr_id, pr_id || :space || left(iif(pr_name = pr_memo, pr_name, pr_name || '   / ' || pr_memo), 255)
    from product
    order by pr_id desc
    into res_code, res_text
    do
    begin
      res_inactive = '';
      suspend;
    end
  end else
  if (kind_str = 'hr') then
  begin
    for select sawt_id, sawt_id || :space || sawt_name
    from sa_worktype
    order by sawt_id
    into res_code, res_text
    do
    begin
      res_inactive = '';
      suspend;
    end
  end else
  if (kind_str = 'edoc') then
  begin
    for select edc_code, edc_code || :space || edc_name
    from edoc_cfg
    order by edc_id
    into res_code, res_text
    do
    begin
      res_inactive = '';
      suspend;
    end
  end else
  if (kind_str = 'eblank') then
  begin
    res_code = 'id';
    res_text = 'Эл.документ';
    res_inactive = '';
    suspend;
  end else
    exception e_default 'Undefined preblank_kind: ' || kind_str;
end
