CREATE PROCEDURE AAREP_KONTRACT_ID_COLLECT_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_K_ID DECIMAL(18,0),
    AAREP_DATE_ACTION DATE,
    AAREP_COLL_PLAN_OS DECIMAL(18,4),
    AAREP_COLL_FACT_OS DECIMAL(18,4))
AS
begin
  report_reverse = 0;
  aarep_coll_plan_os = 0;
  aarep_coll_fact_os = 0;
  aarep_date_action = :report_date_end;

  for select ok_from_k_id, sum(ok_os + ok_sh_os), cast(0 as decimal(18,4))
      from operkred, kontract, knaktlist, region_list(:report_region_id, 1, -1, 1)
      where ok_ol_is_active = 1 and
            ((ok_ol_date < :report_date_begin and
            ok_type <> 8) or
            ((ok_ol_date between :report_date_begin and :report_date_end) and
            ok_type = 1)) and
            ok_sh_date >= :report_date_begin and
            ok_sh_date <= :report_date_end and
            ((:report_currency = '000') or
            (:report_currency = '---') or
            (k_currency = :report_currency)) and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :report_date_end and
            kal_date_end >= :report_date_end and
            (kal_new_reg_id = id_list) and
            ok_from_k_id = k_id and
            knaktlist.is_delete = 0 and
            ok_from_c_id = 0
      group by ok_from_k_id
      having sum(ok_os + ok_sh_os) <> 0
      union
      select ok_from_k_id, cast(0 as decimal(18,4)), sum(-1 * ok_os)
      from operkred, kontract, knaktlist, region_list(:report_region_id, 1, -1, 1)
      where ok_ol_is_active = 1 and
            ok_ol_date between :report_date_begin and :report_date_end and
            ok_sh_date >= :report_date_begin and
            ok_sh_date <= :report_date_end and
            ok_type >= 4 and
            ok_type <> 8 and
            ok_os <> 0 and
            knaktlist.is_delete = 0 and
            ((:report_currency = '000') or
            (:report_currency = '---') or
            (k_currency = :report_currency)) and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :report_date_end and
            kal_date_end >= :report_date_end and
            (kal_new_reg_id = id_list) and
            ok_from_k_id = k_id and
            ok_from_c_id = 0
      group by ok_from_k_id
      order by 1
      into :aarep_k_id, :aarep_coll_plan_os, :aarep_coll_fact_os
  do
  begin
    suspend;
  end
end
