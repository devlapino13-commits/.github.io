CREATE PROCEDURE HP_SYS_TRANSLATE_SYNC
AS
declare variable db_user d_str_code;
declare variable db_pass d_str_code;
declare variable orig_text d_str_255;
declare variable translate_text d_str_255;
declare variable trm_id d_id;
declare variable current_db_id d_integer;
declare variable source_db_id d_integer;
declare variable db_connection d_str_255;
declare variable lg_lang_code d_str_10;
begin
  -- процедура синхронизирует оригинал и переводы текстов между базами

  current_db_id = f_get_db_id();

  select mss_user_connect, mss_pass
  from morsynsettings
  rows 1
  into db_user, db_pass;

  -- сначала проводим полный цикл импорта: забираем все что нужно из других БД
  for select 81001, '10.1.2.11/3050:E:\IB_Base\DB_MBFRU.IB'
  from rdb$database
  where :current_db_id <> 81001
  union
  select 41701, '192.168.200.250/3050:D:\Finance\IB_Base\DB_1.IB'
  from rdb$database
  where :current_db_id <> 41701
  union
  select 94401, '10.60.0.50/3050:E:\DB\DB_AZ.IB'
  from rdb$database
  where :current_db_id <> 94401
  into source_db_id, db_connection
  do
  begin
    lg_lang_code = '';
    translate_text = '';

    -- импортируем оригиналы текста для перевода из других БД
    for execute statement 'select trm_orig_text from sys_transmain'
    on external :db_connection
    as user :db_user password :db_pass
    into :orig_text
    do
    begin
      select trm_id from hp_sys_translate_insert('', :orig_text, '')
      into trm_id;
    end

    for select lg_lang_code
    from hp_translate_language_list
    into :lg_lang_code
    do
    begin
      if (current_db_id = 81001) then
      begin
        if (source_db_id = 41701 and lg_lang_code <> 'kg') then Continue;
        if (source_db_id = 94401 and lg_lang_code <> 'az') then Continue;
      end
      if (current_db_id = 41701) then
      begin
        if (source_db_id = 81001 and lg_lang_code in ('kg', 'az')) then Continue;
        if (source_db_id = 94401 and lg_lang_code <> 'az') then Continue;
      end
      if (current_db_id = 94401) then
      begin
        if (source_db_id = 81001 and lg_lang_code in ('kg', 'az')) then Continue;
        if (source_db_id = 41701 and lg_lang_code <> 'kg') then Continue;
      end

      for execute statement
      ('select trm_orig_text, trl_translate_text
      from sys_transmain, sys_translate
      where trm_id = trl_from_trm_id and
            coalesce(trl_translate_text, '''') <> '''' and
            trl_lang_code = :lg_lang_code')
      (lg_lang_code := lg_lang_code)
      on external :db_connection
      as user :db_user password :db_pass
      into :orig_text, translate_text
      do
      begin
        select trm_id from hp_sys_translate_insert(:lg_lang_code, :orig_text, :translate_text)
        into trm_id;
      end
    end
  end

  -- после этого проводим полный цикл экспорта: заливаем все что нужно в другие БД
  for select 81001, '10.1.2.11/3050:E:\IB_Base\DB_MBFRU.IB'
  from rdb$database
  where :current_db_id <> 81001
  union
  select 41701, '192.168.200.250/3050:D:\Finance\IB_Base\DB_1.IB'
  from rdb$database
  where :current_db_id <> 41701
  union
  select 94401, '10.60.0.50/3050:E:\DB\DB_AZ.IB'
  from rdb$database
  where :current_db_id <> 94401
  into source_db_id, db_connection
  do
  begin
    lg_lang_code = '';
    translate_text = '';

    -- экспортируем оригиналы текста для перевода в другие БД
    for select trm_orig_text from sys_transmain
    into orig_text
    do
    begin
      execute statement ('select trm_id from hp_sys_translate_insert('''', :orig_text, '''')')
      (orig_text := orig_text)
      on external :db_connection
      as user :db_user password :db_pass
      into :trm_id;
    end

    for select lg_lang_code
    from hp_translate_language_list
    into :lg_lang_code
    do
    begin
      if (current_db_id = 81001) then
      begin
        if (source_db_id = 41701 and lg_lang_code = 'kg') then Continue;
        if (source_db_id = 94401 and lg_lang_code = 'az') then Continue;
      end
      if (current_db_id = 41701) then
      begin
        if (source_db_id = 81001 and lg_lang_code not in ('kg', 'az')) then Continue;
        if (source_db_id = 94401 and lg_lang_code = 'az') then Continue;
      end
      if (current_db_id = 94401) then
      begin
        if (source_db_id = 81001 and lg_lang_code not in ('kg', 'az')) then Continue;
        if (source_db_id = 41701 and lg_lang_code = 'kg') then Continue;
      end

      for select trm_orig_text, trl_translate_text
      from sys_transmain, sys_translate
      where trm_id = trl_from_trm_id and
            trl_lang_code = :lg_lang_code and
            coalesce(trl_translate_text, '') <> ''
      into :orig_text, translate_text
      do
      begin
        execute statement ('select trm_id from hp_sys_translate_insert(:lg_lang_code, :orig_text, :translate_text)')
        (lg_lang_code := lg_lang_code, orig_text := orig_text, translate_text := translate_text)
        on external :db_connection
        as user :db_user password :db_pass
        into :trm_id;
      end
    end
  end
end
