CREATE PROCEDURE KONTRACT_FULL_OPER_PERC(
    KONT_ID D_ID,
    DD D_DATE)
RETURNS (
    OL_DATE D_DATE,
    SUM_OF_OK_OS D_CURRENCY,
    SUM_OF_OK_PERC D_CURRENCY,
    SUM_OF_OK_FINE D_CURRENCY,
    SUM_OF_OK_KOMMIS D_CURRENCY,
    SUM_OF_OK_KOMMIS_ADV D_CURRENCY,
    SUM_BALANCE D_CURRENCY,
    CALC_PERC_END D_CURRENCY,
    CALCED_PERC D_CURRENCY,
    OPER_KIND D_STR_CODE,
    OL_IS_END_MONTH D_INTEGER,
    OUT_CALCED_PERC D_CURRENCY,
    CALC_OUT_PERC_END D_CURRENCY)
AS
declare variable oper_type d_integer;
declare variable perc_per_day d_percent;
declare variable k_stop_perc d_date;
declare variable calc_method d_integer;
declare variable min_dd d_date;
declare variable first_time d_integer;
declare variable saldo_prior d_currency;
declare variable kcp_period d_integer;
declare variable calced_perc_mbh d_currency;
declare variable write_off_mbh boolean;
declare variable is_mbh boolean;
begin
  calc_perc_end = 0;
  calced_perc = 0;
  write_off_mbh = false;
  sum_balance = 0;
  min_dd = null;
  saldo_prior = 0;
  is_mbh = f_new_markup_mbh(:kont_id);

  for
    select
      ol_date,
      k_percent_per_day,
      k_stop_perc,
      oper_type,
      oper_kind,
      sum(sum_of_ok_os),
      sum(sum_of_ok_perc),
      sum(sum_of_ok_fine),
      sum(sum_of_ok_kommis),
      sum(sum_of_ok_kommis_adv),
      max(ol_is_end_month)
    from kontract
      inner join kontract_full_oper_perc_temp(:kont_id, :dd)
        on 1 = 1
    where k_id = :kont_id
    group by ol_date, k_percent_per_day, k_stop_perc, oper_type, oper_kind
    order by ol_date, oper_kind
  into
    :ol_date,
    :perc_per_day,
    :k_stop_perc,
    :oper_type,
    :oper_kind,
    :sum_of_ok_os,
    :sum_of_ok_perc,
    :sum_of_ok_fine,
    :sum_of_ok_kommis,
    :sum_of_ok_kommis_adv,
    :ol_is_end_month
  do
  begin
    calced_perc = 0;
    kcp_period = 100 * extract(year from ol_date) + extract(month from ol_date);
    if (min_dd is not null) then
    begin
      if ((ol_date >= :dd) or (ol_date - min_dd >= 0)) then
      begin
        if (f_new_perc_active(ol_date) = 1) then
          select result_calced, result_perc_end, result_out_calced, result_out_perc_end
          from aarep_calc_perc_one_period(:kont_id, :min_dd, :ol_date)
          into :calced_perc, :calc_perc_end, out_calced_perc, :calc_out_perc_end;
        else
          select result_calced, result_perc_end, result_out_calced, result_out_perc_end
          from aarep_calc_perc_one(:kont_id, :min_dd, :ol_date, 1, null, null, null, null)
          into :calced_perc, :calc_perc_end, out_calced_perc, :calc_out_perc_end;
      end
    end

    calced_perc = calc_perc_end - saldo_prior - sum_of_ok_perc;
    sum_balance = sum_balance + sum_of_ok_os;
    if (min_dd is null) then
      min_dd = ol_date;
    else
      min_dd = ol_date + 1;
    if (sum_of_ok_os + sum_of_ok_perc + sum_of_ok_fine + sum_of_ok_kommis + sum_of_ok_kommis_adv = 0) then
    begin
      oper_type = 77;
      oper_kind = 'Начисл.%%';
    end

    if (is_mbh) then
    begin
      if (oper_type = 8) then
      begin
        -- выдача
        sum_of_ok_perc = (
          select
            sum(sh_perc)
          from schedule
          where sh_from_k_id = :kont_id);
        calced_perc = sum_of_ok_perc;
        calced_perc_mbh = sum_of_ok_perc;
        calc_perc_end = sum_of_ok_perc;
      end else
      begin
        calced_perc_mbh = calced_perc_mbh + sum_of_ok_perc;
        calced_perc = calced_perc_mbh;
        calc_perc_end = null;

        if (oper_type = 5) then
        begin
          -- списание
          write_off_mbh = true;
          calc_perc_end = (
            select
              iif(kcp_perc_begin_saldo > kcp_delayed_perc_begin, kcp_perc_calced - kcp_delayed_perc_begin, -kcp_delayed_perc_begin)
            from kontcalcperc
            where kcp_period = :kcp_period
              and kcp_from_k_id = :kont_id
              and kcp_from_c_id = 0);
        end else
        if (oper_type = 77) then
        begin
          -- начисление %%
          calc_perc_end = (
            select
              -kcp_perc_calced
              --iif(kcp_perc_begin_saldo > kcp_delayed_perc_begin, -kcp_perc_calced, 0)
            from kontcalcperc
            where kcp_period = :kcp_period
              and kcp_from_k_id = :kont_id
              and kcp_from_c_id = 0);
        end
      end
    end

    suspend;
    saldo_prior = calc_perc_end;
  end
end
