CREATE PROCEDURE PRINT_TARGET_MONITORING(
    KC_ID D_ID)
RETURNS (
    CLIENT_FIO D_STR_100,
    ANALIZ_DATE D_DATE,
    ANALIZ_LO_FIO D_STR_100,
    FACT_LOAN_TARGET D_STR_255,
    MONITORING_MEMO D_STR_2048,
    KONTRACT_NO D_STR_100,
    FOLDER_NAME D_STR_100,
    PR_NAME D_STR_100,
    LOAN_AMOUNT D_CURRENCY,
    LOAN_TARGET D_STR_255,
    DESCR_OF_LOAN D_STR_2048,
    HAS_ANKETA D_SMALLINT,
    SVZ_MEMO D_STR_2048,
    ACTION_DATE D_DATE,
    K_DISB_DATE D_DATE)
AS
declare variable c_id d_id;
declare variable rd_id d_integer;
declare variable kf_id d_id;
declare variable obj_kind d_str_100;
declare variable obj_id d_id;
declare variable is_ipf d_bool;
declare variable acl_id d_id;
declare variable dt_find d_date;
begin
  rd_id = f_get_rd_id('Anketa.TargetUseLoan');

  select
    k_disb_date,
    k_kontract_no,
    k_from_kf_id,
    kc_amount_approved,
    c_id,
    c_name_fio,
    pr_name,
    iif(pr_code containing '.ipf.', 1 ,0),
    rd_name
  from knclient
    inner join kontract
      on k_id = kc_from_k_id
    inner join clients
      on c_id = kc_from_c_id
    inner join product
      on pr_id = k_from_pr_id
    left join refdata
      on rd_id = kc_loan_target
  where kc_id = :kc_id
    and kc_amount_approved > 0
  into
    :k_disb_date,
    :kontract_no,
    :kf_id,
    :loan_amount,
    :c_id,
    :client_fio,
    :pr_name,
    :is_ipf,
    :loan_target;

  obj_kind = 'KnClient';
  obj_id = kc_id;
  select
    acl_id,
    acl_date,
    current_date, --acl_last_datetime,
    c_name_fio
  from actionlist
    inner join clients
      on c_id = acl_respons_c_id
  where acl_object_kind = f_objkind(:obj_kind, 'id')
    and acl_object_id = :obj_id
    and acl_from_rd_id = :rd_id
  order by acl_id desc
  rows 1
  into
    :acl_id,
    :action_date,
    :analiz_date,
    :analiz_lo_fio;

  if (acl_id is null) then
  begin
    obj_kind = 'Clients';
    obj_id = c_id;

    if (k_disb_date < '14.02.2022')
    then
      dt_find = k_disb_date +
        iif(k_disb_date >= '22.12.2021' and is_ipf = 1, 1, 8);
    else
      dt_find = k_disb_date + 1;

    select
      acl_id,
      acl_date,
      acl_last_datetime,
      c_name_fio
    from actionlist
      inner join clients
        on c_id = acl_respons_c_id
    where acl_object_kind = f_objkind(:obj_kind, 'id')
      and acl_object_id = :obj_id
      and acl_from_rd_id = :rd_id
      and acl_date = :dt_find
      and acl_status = 2
    order by acl_id desc
    rows 1
    into
      :acl_id,
      :action_date,
      :analiz_date,
      :analiz_lo_fio;
  end else analiz_date = current_date;

  select
    coalesce(res_str_values, '')
  from hp_objectprops_anketa_get('Clients', :c_id, -1, 'AnketaQuestions.descr_of_loan', :k_disb_date, null)
  into
    :descr_of_loan;

  has_anketa = iif(acl_id > 0, 1, 0);
  if (has_anketa = 1) then
  begin
    select
      kal_new_kf_name
    from knaktlist
    where kal_from_kf_id = :kf_id
      and :analiz_date between kal_date and kal_date_end
      and is_delete = 0
    into
      :folder_name;

    select
      coalesce(res_str_values, '')
    from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'AnketaQuestions.LoanUseFact', :analiz_date, null)
    into
      :fact_loan_target;

    select
      coalesce(res_str_values, '')
    from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'Monitoring.Memo', :analiz_date, null)
    into
      :monitoring_memo;

    select
      coalesce(left(obm_memo, 2048), '')
    from hp_objectprops_memo_get('Clients', :c_id, 'Refdata', 'ObjectProps.Client_memo', 1)
    rows 1
    into
      :svz_memo;

    suspend;
  end
end
