CREATE PROCEDURE AAREP_KANS_PRINT_AKT_MAIN(
    REP_ID_LIST VARCHAR(255))
RETURNS (
    AKT_MEMO VARCHAR(255),
    AKT_REG_FROM VARCHAR(100),
    AKT_REG_TO VARCHAR(100),
    AKT_TOTAL DECIMAL(18,4),
    AKT_DATE DATE)
AS
declare variable kop_is_delete integer;
begin
  /* заголовок акта приема-передач канц/хозтоваров между офисами */

  for select min(kop_date), kop_memo, r_from.reg_code || ' ' || r_from.reg_name, r_to.reg_code || ' ' || r_to.reg_name,
             kop_is_delete, sum(kopl_minus)
      from kansoper, kansoperlist, aarep_kans_print_akt_tmp(:rep_id_list), region r_from, region r_to
      where kopl_from_kop_id = kop_id and
            kop_id = res_kop_id and
            kopl_minus <> 0 and
            kop_from_reg_id = r_from.reg_unicode and
            kop_to_office = r_to.reg_unicode
      group by r_from.reg_code, r_from.reg_name, r_to.reg_code, r_to.reg_name, kop_memo, kop_is_delete
      into :akt_date, :akt_memo, :akt_reg_from, :akt_reg_to, :kop_is_delete, :akt_total
  do
  begin
    if (kop_is_delete = 1) then
      akt_memo = 'АННУЛИРОВАН. ' || akt_memo;
    suspend;
  end
end
