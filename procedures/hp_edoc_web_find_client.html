CREATE PROCEDURE HP_EDOC_WEB_FIND_CLIENT(
    INN D_STR_20,
    ACC D_INTEGER,
    BIRTH_DATE D_DATE)
RETURNS (
    C_ID D_ID)
AS
begin
  c_id = null;
  inn = coalesce(inn, '');
  birth_date = coalesce(birth_date, '31.12.9999');
  acc = coalesce(acc, 0);

  if (inn = '' and (birth_date = '31.12.9999' or acc = 0)) then
    exception e_message 'not all parameters were filled';

  if (inn <> '' and (birth_date <> '31.12.9999' or acc <> 0)) then
    exception e_message 'too many parameters filled';

  if (inn <> '') then
    c_id = (select c.c_id from clients c where c.c_inn = :inn);
  else
    c_id = (select c_id from hp_check_clientuniq_acc_no(:birth_date, :acc));

  if (coalesce(c_id, -1) <> -1) then
    suspend;
end
