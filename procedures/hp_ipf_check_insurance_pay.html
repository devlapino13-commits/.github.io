CREATE PROCEDURE HP_IPF_CHECK_INSURANCE_PAY(
    K_ID D_ID,
    REP_DATE D_DATE = null)
RETURNS (
    RES_TEXT D_STR_255,
    ZAD_IK_COMMIS D_CURRENCY)
AS
declare variable pr_code d_str_100;
declare variable ik_commission d_currency;
declare variable payed_amount d_currency;
begin
  -- проверка наличия обязательств по страхованию ИПФ займов (RFCKG-5390)
  rep_date = coalesce(rep_date, current_date);

  select pr_code
  from kontract
  inner join product on pr_id = k_from_pr_id
  where k_id = :k_id
  into pr_code;

  if (pr_code not containing '.ipf.' or
      pr_code containing '.ignore_insurance.') then
  begin
    res_text = '';
    suspend;
    exit;
  end

  payed_amount = 0;
  ik_commission = 0;
  zad_ik_commis = 0;

  select ik_commission
  from insurance_kontract
  where ik_from_k_id = :k_id
  into ik_commission; -- сумма страховой премии

  if (coalesce(ik_commission, 0) = 0) then
  begin
    res_text = '';
    suspend;
    exit;
  end

  select sum(ok_amount)
  from operkontract
  inner join operlist on ol_id = ok_from_ol_id
  where ok_from_k_id = :k_id and
        ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance') and
        is_delete = 0 and
        ol_date <= :rep_date
  into :payed_amount; -- оплаченная сумма страховой премии

  if (payed_amount is null) then
    payed_amount = 0;

  if (ik_commission > payed_amount) then
  begin
    res_text = 'Имеются обязательства по страхованию';
    zad_ik_commis = ik_commission - payed_amount;
  end

  if (res_text is null) then
    res_text = '';

  suspend;
end
