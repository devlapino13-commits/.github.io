CREATE PROCEDURE HP_UXFRAME_SQL(
    UX_CODE D_STR_CODE,
    FR_CODE D_STR_CODE,
    NEED_SECURE_MASK D_BOOL = 0)
RETURNS (
    UFR_NAME D_STR_100,
    UFR_MEMO D_STR_LONG,
    UFR_SHOW_KIND D_SMALLINT,
    UFR_CAN_SELECT D_BOOL,
    UFR_CAN_UPDATE D_BOOL,
    UFR_CAN_INSERT D_BOOL,
    UFR_CAN_DELETE D_BOOL,
    UFR_WIDTH D_INTEGER,
    UFR_HEIGHT D_INTEGER,
    UFR_TABLE_NAME D_STR_CODE,
    UFR_KEY_FIELDS D_STR_100,
    UFR_PARAM_SQL D_STR_8100,
    UFR_SELECT_SQL D_STR_8100,
    UFR_UPDATE_SQL D_STR_8100,
    UFR_INSERT_SQL D_STR_8100,
    UFR_DELETE_SQL D_STR_8100,
    UFR_HAS_EXEC_PROC D_BOOL,
    UFR_SELECT_IS_EXEC D_BOOL)
AS
declare variable ufp_name d_str_code;
declare variable exec_proc d_bool;
begin
  ufr_select_sql = '';
  ufr_update_sql = '';
  ufr_insert_sql = '';
  ufr_delete_sql = '';
  ufr_has_exec_proc = 0;
  ufr_select_is_exec = 0;

  if (need_secure_mask is null) then
    need_secure_mask = 0;

  for select ux_name, ux_table_name, ufr_memo, ufr_show_kind,
             ufr_can_select, ufr_can_update, ufr_can_insert, ufr_can_delete
  from uxcode, uxframe
  where ufr_code = :fr_code and
        ux_code = ufr_code
  into :ufr_name, :ufr_table_name, :ufr_memo, :ufr_show_kind,
       :ufr_can_select, :ufr_can_update, :ufr_can_insert, :ufr_can_delete
  do
  begin
    ufr_width = 0;
    ufr_height = 0;

    select ufi_width, ufi_height
    from uxfritem
    where ufi_ux_code = :fr_code and
          ufi_parent_code is null
    into :ufr_width, :ufr_height;

    ufr_key_fields = '';
    ufr_param_sql = '';

    for select ufp_origin_name
    from uxfrparam
    where ufp_ufr_code = :fr_code and
          ufp_pk_field <> 0 and
          ufp_is_delete = 0 and
          ufp_in_use = 1
    order by ufp_pk_field
    into ufp_name
    do
    begin
      if (ufr_key_fields <> '') then
        ufr_key_fields = ufr_key_fields || ',';

      ufr_key_fields = ufr_key_fields || ufp_name;
    end

    select sql_text
    from hp_uxframe_sql_param(:ux_code, :fr_code)
    into ufr_param_sql;

    if (ufr_can_select = 1) then
    begin
      select sql_text, sql_is_exec
      from hp_uxframe_sql_sel(:ux_code, :fr_code, :need_secure_mask)
      into ufr_select_sql, ufr_select_is_exec;
    end

    if (ufr_can_update = 1) then
    begin
      select sql_text, exec_proc
      from hp_uxframe_sql_modify(:fr_code, 1)
      into :ufr_update_sql, exec_proc;

      if (ufr_update_sql <> '' and exec_proc = 1) then
        ufr_has_exec_proc = 1;
    end

    if (ufr_can_insert = 1) then
    begin
      select sql_text, exec_proc
      from hp_uxframe_sql_modify(:fr_code, 2)
      into :ufr_insert_sql, exec_proc;

      if (ufr_insert_sql <> '' and exec_proc = 1) then
        ufr_has_exec_proc = 1;
    end

    if (ufr_can_delete = 1) then
    begin
      select sql_text, exec_proc
      from hp_uxframe_sql_modify(:fr_code, 3)
      into :ufr_delete_sql, exec_proc;

      if (ufr_delete_sql <> '' and exec_proc = 1) then
        ufr_has_exec_proc = 1;
    end

    suspend;
  end
end
