CREATE PROCEDURE FM_COM_LIST_CREATE(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    CMC_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    CML_ID D_ID,
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_INTEGER)
AS
declare variable cml_last_status_code d_str_code;
declare variable cml_last_status d_smallint;
declare variable cmc_state d_smallint;
declare variable cmc_object_kind d_integer;
declare variable reg_id d_integer;
declare variable init_c_id d_id;
declare variable com_kind d_str_code;
declare variable cmc_id_lgota d_integer;
declare variable lang d_str_lang;
declare variable db_id d_integer;
declare variable other_cmc_name d_str_255;
declare variable k_stop_perc d_date;
declare variable k_kredline_end d_date;
begin
  obj_kind = f_objkind(obj_kind, 'id');
  lang = f_get_lang_code(un_id, null);
  db_id = f_get_db_id();

  res_text = '';
  res_is_error = 1;

  if (coalesce(cmc_id, -1) = -1) then
  begin
    res_text = f_lang(lang, 'Вид кредитного комитета не указан');
    suspend;
    exit;
  end

  select cmc_id
  from com_cfg
  where cmc_code = 'rest_lgota'
  into cmc_id_lgota;

  if (res_text = '') then
  begin
    other_cmc_name = '';
    select cmc_name
    from com_list, com_cfg
    where cml_object_kind = :obj_kind and
          cml_object_id = :obj_id and
          cml_from_cmc_id <> :cmc_id and
          cml_is_closed = 0 and
          cml_from_cmc_id = cmc_id
    order by cml_id desc
    rows 1
    into other_cmc_name;
  
    if (other_cmc_name <> '') then
      res_text = f_format(f_lang(lang, 'Имеется запрос на тему: %1. Дождитесь закрытия запроса.'), other_cmc_name);
  end


  if (res_text = '') then
  begin
    select cml_id, cml_last_status
    from com_list
    where cml_object_kind = :obj_kind and
          cml_object_id = :obj_id and
          cml_from_cmc_id = :cmc_id and
          cml_is_closed = 0
    order by cml_id desc
    rows 1
    into cml_id, cml_last_status;
  
    if (cml_id is not null) then
    begin
      cml_last_status_code = f_enum('com_action_status', cml_last_status, 'code');
  
      if (cml_last_status_code in ('sent', 'process', 'approved'))
      then res_text = f_lang(lang, 'Договор уже находится на рассмотрении');
      else
        if (cml_last_status_code in ('new', 'rework'))
        then res_text = f_lang(lang, 'По договору уже создан запрос. Необходимо отправить на рассмотрение');
        else res_text = f_lang(lang, 'По договору уже создан запрос. Статус не определен');
    end
  end

  if (res_text = '') then
  begin
    select cmc_state, cmc_object_kind, f_enum('com_kind', cmc_com_kind, 'code')
    from com_cfg
    where cmc_id = :cmc_id
    into cmc_state, cmc_object_kind, com_kind;
  
    if (cmc_state = 0) then
      res_text = f_lang(lang, 'Данный вид комитета не доступен');
  
    if (res_text = '' and cmc_object_kind <> obj_kind) then
      res_text = f_format(f_lang(lang, 'Тип объекта не соответствует. Ожидается %1, передан %2'),
                          f_objkind(cmc_object_kind, 'name'),
                          f_objkind(obj_kind, 'name'));

  end

  if (res_text = '') then
  begin
    if (com_kind = 'reschedule') then
    begin
      if (db_id = 81001 and current_date >= '01.06.2020') then
      begin
        if (cmc_id_lgota = cmc_id) then
        begin
          if (exists(
          select * from oksummary
          where oks_from_k_id = :obj_id and
                current_date between oks_date_begin and oks_date_end and
                oks_delinq_date < current_date and
                oks_close_state = 0 )) then
          res_text = f_lang(lang, 'Просроченные займы не подлежат реструктуризации');
        end
      end

      if (res_text = '' and db_id in (39801, 41701)) then
      begin
        if (f_tmp_const('ipf_percent_mode_active') = 0) then
        begin
          if (exists(select * from kontract, product
          where k_id = :obj_id and
                k_from_pr_id = pr_id and
                pr_code containing '.ipf.')) then
          res_text = f_lang(lang, 'Рефинансирование по ИПФ временно не доступны для реструктуризации');
        end
      end

      if (res_text = '') then
      begin
        if (exists(select * from kontdata
        where kd_ol_date > current_date and
              kd_ol_is_delete = 0 and
              kd_from_k_id = :obj_id)) then
          res_text = f_lang(lang, 'Срок льготной ставки не истек, запрет на реструктуризацию');
      end
    end
  end

  if (res_text = '') then
  begin
    select k_stop_perc, k_kredline_end
    from kontract
    where k_id = :obj_id
    into k_stop_perc, k_kredline_end;

    if (k_stop_perc is not null) then
      res_text = f_lang(lang, 'Договор расторгнут, реструктуризация запрещается');
    else
      if (k_kredline_end < current_date) then
        res_text = f_lang(lang, 'Срок договора истек, реструктуризация запрещается');
  end

  if (res_text = '') then
  begin
    if (db_id <> 94401 and cmc_id_lgota is not null and cmc_id = cmc_id_lgota) then
    begin
      if (exists(select *
      from com_list
      where cml_object_kind = :obj_kind and
            cml_object_id = :obj_id and
            cml_from_cmc_id = :cmc_id_lgota and
            f_enum('com_action_status', cml_last_status, 'code') = 'done')) then
        res_text = f_lang(lang, 'Данный займ ранее уже был реструктурирован по льготному тарифу. Повторная реструктуризация запрещена.');
    end
  end

  if (res_text = '') then
  begin
    select reg_id
    from hp_edoc_object_info(:obj_kind, :obj_id, :un_id, current_date)
    into :reg_id;

    select un_from_c_id
    from username
    where un_id = :un_id
    into init_c_id;

    insert into com_list (cml_from_cmc_id, cml_date, cml_from_reg_id, cml_init_c_id,
                          cml_object_kind, cml_object_id, cml_from_un_id)
    values (:cmc_id, current_date, :reg_id, :init_c_id, :obj_kind, :obj_id, :un_id)
    returning cml_id
    into :cml_id;

    res_text = 'Звпрос на КК создан. По необходимости добавьте комментарий, файл(ы). Далее отправьте на рассмотрение КК.';
    res_is_error = 0;
  end

  suspend;
end
