CREATE PROCEDURE HP_GET_PASSPORT_VARIANTS(
    PASS_NO D_STR_20)
RETURNS (
    RES_DOC_NO_WITHOUT_SERIES D_STR_20,
    RES_DOC_NO_WITHOUT_SPACES D_STR_20,
    RES_DOC_NO D_STR_20,
    RES_WITHOUT_SERIES D_SMALLINT)
AS
declare variable INDX D_INTEGER;
begin
-- RFCRU-3312 Процедура обрабатывает номер паспорта и возвращает
-- RES_DOC_NO_WITHOUT_SERIES паспорт без серии    AN12312312 = 12312312
-- RES_DOC_NO_WITHOUT_SPACES паспорт без пробелов AN 12312312 = AN 12312312
-- RES_DOC_NO просто Серия и номер паспорта       AN 12312312 = AN 12312312
-- RES_WITHOUT_SERIES Переменная отображающ Была ли серия или нет
  indx = 1;
  res_without_series = 0;

  res_doc_no_without_spaces = replace(pass_no, ' ', '');
  res_doc_no_without_series = res_doc_no_without_spaces;
  
  while (substring(res_doc_no_without_series from 1 for 1) not similar to '[[:DIGIT:]]+') do --пока первый символ не [0-9]
  begin
      res_doc_no_without_series = substring(res_doc_no_without_series from 2);

      indx = indx + 1;
  end

  if (res_doc_no_without_spaces similar to '[[:DIGIT:]]+') then --Если документ состоит из цифр (Загран паспорт таджик.) клиент 1010100887822
  begin
    res_without_series = 1;
    indx = 0;
    res_doc_no = res_doc_no_without_spaces;
  end

  if (indx <> 0 and res_without_series <> 1) then
    res_doc_no = substring(res_doc_no_without_spaces from 1 for indx - 1) || ' ' || substring(res_doc_no_without_spaces from indx);       --добавляем пробел между мерией и номером паспорта --> FA 1214289

  suspend;
end
