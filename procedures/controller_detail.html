CREATE PROCEDURE CONTROLLER_DETAIL(
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    SHOW_PHOTO SMALLINT)
RETURNS (
    C_NAME_FIO STR_100,
    C_PLASTCARD_NO STR_255,
    C_CITIZEN_STR VARCHAR(50),
    REP_C_ADDRESS_JOB STR_255,
    CB_FROM_BLS_ID TYPE OF COLUMN CLIENTBLOB.CB_FROM_BLS_ID,
    C_PHOTO BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    C_MEMO VARCHAR(1024),
    WORK_PLACE_NAME STR_255,
    WORK_JOB_NAME STR_255,
    WORK_ADDRESS_NAME STR_255,
    WORK_ADDRESS_ORIENTIR STR_255,
    WORK_ADDRESS_METRO STR_255,
    WORK_STATUS STR_255)
AS
declare variable rep_date date;
begin
  rep_date = current_date;
  show_photo = coalesce(show_photo, 0);

  for select cl.c_id, cl.c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_valid(0, kncl.kc_from_c_id)), ddr.rd_name as c_citizen_str, cl.c_memo
      from knclient kncl
      inner join clients cl on kncl.kc_from_c_id = cl.c_id
      inner join refdata ddr on ddr.rd_id = cl.c_citizen
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_from_c_id = :c_id and
            kncl.kc_amount_approved > 0
      into :c_id, :c_name_fio, c_plastcard_no, c_citizen_str, :c_memo
  do
  begin
    work_place_name = '';
    work_job_name = '';
    work_address_name = '';
    work_address_orientir = '';
    work_address_metro = '';
    work_status = '';

    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_place_name', :rep_date)
    into work_place_name;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_job_name', :rep_date)
    into work_job_name;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_address_orientir', :rep_date)
    into work_address_orientir;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_address_metro', :rep_date)
    into work_address_metro;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_status', :rep_date)
    into work_status;

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :c_id and
          adc_address_type = 3 and
          adc_from_adl_id = adl_id
    into :work_address_name;

    if (coalesce(work_address_name, 'Не указан') = 'Не указан') then
      work_address_name = '';
    if (work_address_name = '') then
    begin
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'work_address_name', :rep_date)
      into work_address_name;
    end

--      Старый вариант, отключено 29.03.2016, Sanjari
--      for select CDC_CODE, CDV_VALUE_STRING
--      from HP_CUSTOMDATACFG_ONE(0, 0, :C_ID, :REP_DATE)
--      into :cdc_code, :cdv_value_string
--      do
--      begin
--        if (cdc_code = 'work_place_name')       then work_place_name       = cdv_value_string;
--        if (cdc_code = 'work_job_name')         then work_job_name         = cdv_value_string;
--        if (cdc_code = 'work_address_name')     then work_address_name     = cdv_value_string;
--        if (cdc_code = 'work_address_orientir') then work_address_orientir = cdv_value_string;
--        if (cdc_code = 'work_address_metro')    then work_address_metro    = cdv_value_string;
--        if (cdc_code = 'work_status')           then work_status           = cdv_value_string;
--      end

    c_memo = coalesce(c_memo, '');
    c_memo = replace(c_memo, ascii_char(13), ' ');
    c_memo = replace(c_memo, ascii_char(10), ' ');

    /*
      REP_C_ADDRESS_JOB = null;
      select adl_address_str
      from  addrlist, addrclient
      where adc_from_c_id = :c_id
      and adc_from_adl_id = adl_id
      and adc_Address_type = 3
      into REP_C_ADDRESS_JOB;

      if ((REP_C_ADDRESS_JOB is null) and (work_address_name <> ''))then
      begin */
    rep_c_address_job = work_address_name;
--end
    rep_c_address_job = coalesce(rep_c_address_job, '');

    rep_c_address_job = iif(rep_c_address_job <> '', rep_c_address_job, 'Адрес не указан');

    if (show_photo = 1) then
    begin

      select cb_from_bls_id
      from aarep_clientblob_list(:c_id)
      where cb_photo_kind = 'Фото контрагента'
      order by cb_datetime desc
      into :cb_from_bls_id;

      c_photo = null;

      if (cb_from_bls_id > -1) then
      begin
        select bls_blob
        from aarep_blobstore_show(:cb_from_bls_id)
        into :c_photo;
      end
    end
    suspend;
  end
end
