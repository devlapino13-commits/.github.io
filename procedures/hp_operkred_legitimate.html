CREATE PROCEDURE HP_OPERKRED_LEGITIMATE(
    K_ID ID_TYPE,
    SH_DATE DATE)
AS
declare variable k_amount_approved curr_type;
declare variable pr_product_kind smallint;
declare variable delinq_day integer;
declare variable ol_date date;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable c_id id_type;
declare variable dolya double precision;
declare variable is_legitimate smallint;
declare variable repay_sum curr_type;
declare variable is_legitimate_cli smallint;
declare variable woff_sum curr_type;
declare variable pr_code varchar(100);
begin
  select k_amount_approved, pr_product_kind, pr_code
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into k_amount_approved, pr_product_kind, pr_code;

  ol_date = null;
  for select max(ok_ol_date) as ol_date, sum(ok_os + ok_sh_os) as os, sum(ok_perc + ok_sh_perc) as perc
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :k_id and
            ok_sh_date = :sh_date and
            ok_type < 7 and
            (ok_os <> 0 or
            ok_perc <> 0 or
            ok_sh_os <> 0 or
            ok_sh_perc <> 0 or
            ok_sh_id = 0) -- погашении только по пеням не включать
      into ol_date, zad_os, zad_perc
  do
  begin
    delinq_day = iif(ol_date > sh_date, ol_date - sh_date, 0);
--  проверка закрыт ли платеж, не досрочно ли закрыт за 10 дней и не параллельный кредит ли
    if (zad_os <= 0 and
        zad_perc <= 0
--        and sh_date - ol_date <= 10
--        and pr_product_kind <> 2
        ) then
      is_legitimate = 1;
    else
      is_legitimate = 0;

    repay_sum = 0;
    delinq_day = 0;
    woff_sum = 0;
    if (is_legitimate = 1) then
    begin
      /*
      repay_sum = null;
      select -sum(ok_os + ok_perc) from operkred
      where ok_ol_is_active = 1
      and ok_from_k_id = :k_id
      and ok_sh_date = :sh_date
      and ok_ol_date = :ol_date
      and ok_type = 4
      into repay_sum;
      repay_sum = coalesce(repay_sum,0);
      */
      repay_sum = null;
      delinq_day = null;
      woff_sum = null;
      select -sum(iif(ok_type = 4, ok_os + ok_perc, 0)), max(ok_ol_date - ok_sh_date),
             -sum(iif(ok_type = 5, ok_os + ok_perc, 0))
      from operkred
      where ok_ol_is_active = 1 and
            ok_sh_date = :sh_date and
            ok_from_k_id = :k_id and
            ok_type in (4, 5) and
            ok_os + ok_perc < 0 and
            ok_sh_date - ok_ol_date <= 10
      into repay_sum, delinq_day, woff_sum;
      repay_sum = coalesce(repay_sum, 0);
      delinq_day = coalesce(delinq_day, 0);
      woff_sum = coalesce(woff_sum, 0);

      if (delinq_day < 0) then
        delinq_day = 0;
    end

    for select kc_from_c_id, cast(kc_amount_approved as double precision) / :k_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, dolya
    do
    begin
      is_legitimate_cli = is_legitimate;
      if (is_legitimate = 1 and
          (dolya * repay_sum < 2500.00 or woff_sum > 0 or pr_product_kind = 2 or pr_code containing '.express.')) then
        is_legitimate_cli = 0;

      update or insert into clilegpay (clp_from_c_id, clp_from_k_id, clp_sh_date, clp_ol_date, clp_is_legitimate,
                                       clp_delinq_day)
      values (:c_id, :k_id, :sh_date, :ol_date, :is_legitimate_cli, :delinq_day)
      matching (clp_from_c_id, clp_from_k_id, clp_sh_date);

    end
--    suspend;
  end
-- если запись удален из таблицы operkred то и в таблице CliLegPay тоже должно удалиться
-- если запись удален то не будет возвращать результат верхнего скрипта и ol_date будет null

  if (ol_date is null) then
  begin
    delete from clilegpay
    where clp_from_k_id = :k_id and
          clp_sh_date = :sh_date;
  end
end
