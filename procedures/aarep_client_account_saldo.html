CREATE PROCEDURE AAREP_CLIENT_ACCOUNT_SALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_BALANCE_ID VARCHAR(20))
RETURNS (
    OB_ACCOUNT VARCHAR(20),
    OB_CLIENT DECIMAL(18,0),
    OB_REGION_ID INTEGER,
    OB_ACCOUNT_NAME VARCHAR(255),
    OB_REGION_NAME VARCHAR(100),
    OB_CLIENT_NAME VARCHAR(100),
    OB_SUMMA_DT DECIMAL(18,4),
    OB_SUMMA_KT DECIMAL(18,4),
    OB_SUMMA DECIMAL(18,4),
    OB_BALANCE_ID VARCHAR(20))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ob_account, ob_client, sum(ob_summa), ob_region_id, al_name, reg_name, c_name_fio, al_from_bg_id
      from acc_list, region, aarep_client_account_temp(:report_region_id, :report_date_end, :report_balance_id)
      left join clients on (ob_client = c_id)
      where al_id = ob_account and
            ob_region_id = reg_unicode
      group by ob_account, ob_client, ob_region_id, al_name, reg_name, c_name_fio, al_from_bg_id
      having sum(ob_summa) <> 0
      into :ob_account, :ob_client, :ob_summa, :ob_region_id, :ob_account_name, :ob_region_name, :ob_client_name,
           :ob_balance_id
  do
  begin
    if (ob_client_name is null) then
      ob_client_name = '';
    if (ob_summa >= 0) then
    begin
      ob_summa_dt = :ob_summa;
      ob_summa_kt = 0;
    end
    else
    begin
      ob_summa_dt = 0;
      ob_summa_kt = -1 * :ob_summa;
    end
    suspend;
  end
end
