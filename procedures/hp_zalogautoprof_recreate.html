CREATE PROCEDURE HP_ZALOGAUTOPROF_RECREATE(
    K_KONTRACT_NO D_STR_CODE,
    REP_YEAR D_INTEGER,
    UN_ID D_ID,
    COST_1 D_CURRENCY = 0,
    COST_2 D_CURRENCY = 0,
    COST_3 D_CURRENCY = 0,
    COST_4 D_CURRENCY = 0,
    COST_5 D_CURRENCY = 0,
    LINK_1 D_STR_LONG = null,
    LINK_2 D_STR_LONG = null,
    LINK_3 D_STR_LONG = null,
    LINK_4 D_STR_LONG = null,
    LINK_5 D_STR_LONG = null)
AS
declare variable k_id d_id;
declare variable zap_id d_integer;
declare variable rep_date d_date;
declare variable za_id d_integer;
declare variable res_is_error d_smallint;
declare variable res_text d_str_255;
begin
  -- Процедура для создания профсуждения на конец года по существующему ZalogAuto или
  -- для импорта из Excel. В этом случае передаются параметры COST_ и LINK_

  rep_date = cast('31.12.' || rep_year as date);

  select k_id
  from kontract
  where k_kontract_no = :k_kontract_no
  into k_id;

  select za_id
  from zalogauto, zalog
  where za_from_k_id = :k_id and
        za_from_z_id = z_id and
        coalesce(z_date_begin, :rep_date) <= :rep_date and
        coalesce(z_date_end, :rep_date + 1) > :rep_date
  into za_id;

  if (za_id is null) then
    exception e_default f_format('ZalogList not found, k_id: %1', k_id);

  select zap_id
  from zalogautoprof
  where zap_from_za_id = :za_id and
        zap_date = :rep_date
  into zap_id;

  if (zap_id is null) then
  begin
    insert into zalogautoprof (zap_from_za_id, zap_date, zad_from_un_id)
    values (:za_id, :rep_date, :un_id)
    returning zap_id
    into zap_id;
  end

  if (cost_1 <> 0) then
  begin
    select res_is_error, res_text
    from hp_zalogautolist_edit(0, :zap_id, :cost_1, :link_1, :un_id)
    into res_is_error, res_text;
  
    if (res_is_error = 1) then
      exception e_default res_text;
  end

  if (cost_2 <> 0) then
  begin
    select res_is_error, res_text
    from hp_zalogautolist_edit(0, :zap_id, :cost_2, :link_2, :un_id)
    into res_is_error, res_text;
  
    if (res_is_error = 1) then
      exception e_default res_text;
  end

  if (cost_3 <> 0) then
  begin
    select res_is_error, res_text
    from hp_zalogautolist_edit(0, :zap_id, :cost_3, :link_3, :un_id)
    into res_is_error, res_text;
  
    if (res_is_error = 1) then
      exception e_default res_text;
  end

  if (cost_4 <> 0) then
  begin
    select res_is_error, res_text
    from hp_zalogautolist_edit(0, :zap_id, :cost_4, :link_4, :un_id)
    into res_is_error, res_text;
  
    if (res_is_error = 1) then
      exception e_default res_text;
  end

  if (cost_5 <> 0) then
  begin
    select res_is_error, res_text
    from hp_zalogautolist_edit(0, :zap_id, :cost_5, :link_5, :un_id)
    into res_is_error, res_text;
  
    if (res_is_error = 1) then
      exception e_default res_text;
  end

  select res_is_error, res_text
  from hp_actionlist_zalogauto_add(:zap_id, :un_id)
  into res_is_error, res_text;

  if (res_is_error = 1) then
    exception e_default res_text;
end
