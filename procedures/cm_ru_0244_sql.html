CREATE PROCEDURE CM_RU_0244_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_name_fio d_str_100;
declare variable final_pdn d_currency;
declare variable k_amount_approved d_currency;
declare variable final_limit d_currency;
declare variable error_status d_smallint;
declare variable error_text d_str_255;
declare variable c_id d_id;
declare variable ks_date_begin_last d_date;
declare variable kol_day d_integer;
declare variable k_refund d_smallint;
declare variable k_id_last d_id;
declare variable pr_code_last d_str_100;
declare variable res_is_error d_smallint;
declare variable k_disb_date d_date;
begin

  res_is_error = 0;
  res_error_text = '';

  if (check_kind containing '.loan_disb.' or check_kind containing '.loan_approve.' or check_kind containing '.send_kk.') then
  begin
    error_status = 0;
    error_text = '';
    select error_status, error_text
    from hp_check_limit_kontract(:k_id)
    into :error_status, error_text;

    if (error_status = 1) then
    begin
      res_error_is_critical = 2;
      res_error_text = :error_text;
      suspend;
    end
  end
  -- Заем может быть выдан  чем через два дня после выдачи предыдущего займа
  -- отправка КК

  for select knc.kc_from_c_id, kn.k_disb_date, k_refund
      from kontract kn
      inner join knclient knc on knc.kc_from_k_id = kn.k_id
      inner join product pr on pr.pr_id = kn.k_from_pr_id
      where kn.k_id = :k_id and
            knc.kc_amount_approved > 0 and
            pr.pr_code containing '.comfortable.'
      into :c_id, k_disb_date, k_refund
  do
  begin
    ks_date_begin_last = null;

    -- все выданные контракты продукту удобный
    for select k_id, ks.ks_date_begin
        from kontstate ks
        inner join knclient knc on knc.kc_from_k_id = ks.ks_from_k_id
        inner join kontract k on k.k_id = knc.kc_from_k_id
        inner join product p on p.pr_id = k.k_from_pr_id
        where knc.kc_from_c_id = :c_id and
              knc.kc_amount_approved > 0 and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0 and
              ks.ks_from_k_id <> :k_id and
              p.pr_code containing '.comfortable.'
        into :k_id_last, ks_date_begin_last
    do
    begin
      if (ks_date_begin_last is not null) then
      begin
        kol_day = k_disb_date - ks_date_begin_last;
        if (kol_day <= 2 and
            check_kind containing '.send_kk.') then
        begin
          res_error_is_critical = 2;
          res_error_text = 'От даты выдачи последнего контракта не должно быть меньше 2 дней включительно';
          suspend;
        end
      end
      --не допускается выдача займа «Удобный 3.0» при наличии другого активного займа по продукту «Удобный 3.0»
      /*if (k_refund = 0 and
          check_kind containing '.send_kk.') then
      begin
        if (exists(select *
                   from kontstate ks
                   where ks.ks_from_k_id = :k_id_last and
                         current_date between ks.ks_date_begin and ks.ks_date_end and
                         ks.ks_ol_is_delete = 0 and
                         ks.ks_state = 1)) then
        begin
          res_error_is_critical = 2;
          res_error_text = 'не допускается выдача займа «Удобный 3.0» при наличии другого активного займа по продукту «Удобный 3.0»';
          suspend;
        end
      end */

      if (check_kind containing '.loan_approve.') then
      begin
        if (cast(ks_date_begin_last as d_date) + 2 >= current_date) then
        begin
          res_error_is_critical = 2;
          res_error_text = 'Дата выдачи возможна не раньше :' || f_datetostr(ks_date_begin_last + 3);
          suspend;
        end
      end

    end

  end
end
