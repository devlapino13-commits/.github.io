CREATE PROCEDURE OKT_DO_ACTIONSAVE(
    ACL_ID DECIMAL(18,0),
    QUESTION_CODE VARCHAR(255),
    RD_CODE VARCHAR(255),
    STATE INTEGER,
    STR_VALUE VARCHAR(255),
    UN_ID DECIMAL(18,0))
AS
declare variable rd_id integer;
declare variable question_id integer;
declare variable err_text d_str_long;
begin
  /*    OKTELL
        процедура сохранения анкеты через OKTELL
  */
  if (question_code similar to '[-]?[[:DIGIT:]]+') then
    question_id = question_code;
  else
    question_id = (select rd_id
                   from refdata
                   where rd_code = :question_code);

  if (rd_code similar to '[-]?[[:DIGIT:]]+') then
    rd_id = rd_code;
  else
    rd_id = (select rd_id
             from refdata
             where rd_code = :rd_code);

  select err_text
  from hp_do_actionsave(:acl_id, :question_id, :rd_id, :state, :str_value, :un_id)
  into :err_text;
end
