CREATE PROCEDURE OKTELL_USER_LIST_WITH_ACCESS
RETURNS (
    UN_ID ID_TYPE,
    UN_NAME STR_100,
    UN_FIO STR_100,
    VERIFICATION SMALLINT,
    SIRENA_1 SMALLINT,
    SIRENA_2 SMALLINT,
    PANTERA_1 SMALLINT,
    PANTERA SMALLINT,
    PANTERA_2 SMALLINT,
    LAN_RU SMALLINT,
    LAN_KG SMALLINT,
    LAN_UZ SMALLINT,
    LAN_TJ SMALLINT,
    KS_FINANCE_1 SMALLINT,
    KS_FINANCE_2 SMALLINT,
    INVESTIGATION_AND_RECOVERY SMALLINT)
AS
declare variable DB_ID D_ID;
begin
  db_id = f_get_db_id();
-- Процедура возвращает список активных пользователей колл-центра с доступами к проектам и языкам

  for select un_id,
             un_name,
             a.un_fio,
             iif(sum(a.r115) = 0, 0, iif(sum(a.r115) = 1, 1, 0)) as verification, -- верификация
             iif(sum(a.r116) = 0, 0, iif(sum(a.r116) = 1, 1, 0)) as sirena_1, -- сирена 1
             iif(sum(a.r117) = 0, 0, iif(sum(a.r117) = 1, 1, 0)) as sirena_2, -- сирена 2
             iif(sum(a.r118) = 0, 0, iif(sum(a.r118) = 1, 1, 0)) as pantera_1, -- пантера 1
             iif(sum(a.r124) = 0, 0, iif(sum(a.r124) = 1, 1, 0)) as pantera_2, -- пантера 2  
             iif(sum(a.r197) = 0, 0, iif(sum(a.r197) = 1, 1, 0)) as ks_finance_1, -- КС Финанс 1
             iif(sum(a.r198) = 0, 0, iif(sum(a.r198) = 1, 1, 0)) as ks_finance_2, -- КС Финанс 2
             iif(sum(a.r210) = 0, 0, iif(sum(a.r210) = 1, 1, 0)) as investigation_and_recovery, --- ДРПВ

             iif(sum(a.r143) = 0, 0, iif(sum(a.r143) = 1, 1, 0)) as lan_ru, -- русский язык
             iif(sum(a.r144) = 0, 0, iif(sum(a.r144) = 1, 1, 0)) as lan_kg, -- кыргызский язык
             iif(sum(a.r145) = 0, 0, iif(sum(a.r145) = 1, 1, 0)) as lan_uz, -- узбекский язык
             iif(sum(a.r146) = 0, 0, iif(sum(a.r146) = 1, 1, 0)) as lan_tj -- таджикский язык
      from (select un_id,
                   u.un_name,
                   un_fio,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 115, pm_access_kind, 0), iif(pm_subject_id = 146, pm_access_kind, 0)) as r115,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 116, pm_access_kind, 0), 0) as r116,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 117, pm_access_kind, 0), 0) as r117,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 118, pm_access_kind, 0), iif(pm_subject_id = 287, pm_access_kind, 0)) as r118,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 124, pm_access_kind, 0), 0) as r124,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 143, pm_access_kind, 0), 0) as r143,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 144, pm_access_kind, 0), 0) as r144,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 145, pm_access_kind, 0), 0) as r145,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 146, pm_access_kind, 0), 0) as r146,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 197, pm_access_kind, 0), 0) as r197,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 198, pm_access_kind, 0), 0) as r198,
                   iif(:db_id not in (39801, 41701), iif(pm_subject_id = 210, pm_access_kind, 0), 0) as r210
            from permissions
            inner join username u on un_id = pm_object_id
            where pm_object_kind = 0 and
                  pm_subject_kind = 3 and
                  pm_is_delete = 0 and
                  pm_subject_id in (115, 116, 117, 118, 124, 143, 144, 145, 146, 197, 198, 210, 287)) a
      group by a.un_id, a.un_fio, un_name
      order by a.un_fio
      into un_id, un_name, un_fio, verification, sirena_1, sirena_2, pantera_1, pantera_2, ks_finance_1, ks_finance_2, investigation_and_recovery,
           lan_ru, lan_kg, lan_uz, lan_tj
  do
  begin
    if ((verification + sirena_1 + sirena_2 + pantera_1 + pantera_2 + ks_finance_1 + ks_finance_2 + investigation_and_recovery) = 0) then
        continue;

    if ((select un_is_active
         from hp_un_is_active(:un_id)) = 1) then
    begin
      suspend;
    end
  end
end
