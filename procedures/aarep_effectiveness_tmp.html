CREATE PROCEDURE AAREP_EFFECTIVENESS_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_K_ID DECIMAL(18,0),
    AA_CURR_REGION_ID INTEGER,
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_AMOUNT_DISB DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_BEGIN DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_KOLICH_CL_ACTIVE_BEGIN INTEGER,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_KOLICH_CL_DISB INTEGER,
    AA_DELIQ_CL_KOLICH INTEGER,
    AA_DELIQ_CL_KOLICH7 INTEGER,
    AA_DELIQ_BALANCE7 DECIMAL(18,4),
    AA_REPAY_NORMONTH_OS DECIMAL(18,4),
    AA_REPAY_FACT_PERIOD_OS DECIMAL(18,4),
    PR_INTEREST_METHOD SMALLINT,
    K_PERCENT_PER_DAY DECIMAL(18,7),
    K_STOP_PERC DATE,
    RESULT_CALCED DECIMAL(18,4),
    RESULT_PAYED DECIMAL(18,4),
    RES_AMOUNT_PERIOD DECIMAL(18,4),
    AA_REPAY_ALL_FINE DECIMAL(18,4),
    AA_CURR_LEVEL2_NAME VARCHAR(100),
    AA_CURR_LEVEL3_NAME VARCHAR(100),
    AA_DELIQ_DAYS INTEGER,
    AA_REPAY_ALL_OS CURR_TYPE,
    AA_DELIQ_BALANCE1 CURR_TYPE,
    AA_DELIQ_BALANCE30 CURR_TYPE,
    AA_REPAY_ALL_KOMMIS CURR_TYPE,
    AA_KOL_KC_IN_REGION INTEGER,
    AA_KOL_SKC_IN_REGION INTEGER,
    AA_LOAN_KIND VARCHAR(100))
AS
declare variable last_curr_rate curr_type;
begin
  for select aa_k_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_level2_name,
             aa_curr_level3_name, aa_curr_region_id, aa_amount_active_begin, -- КП на начало
             aa_kolich_cl_active_begin, aa_amount_active_end, -- КП на конец
             aa_kolich_cl_active_end, aa_kolich_cl_disb, aa_amount_disb, -- Сумма выдачи за период
             aa_repay_normonth_os, aa_repay_fact_period_os, aa_deliq_cl_kolich, aa_deliq_balance, aa_deliq_cl_kolich7,
             aa_deliq_balance7, -- Просроченный балан > 7
             aa_repay_all_fine, pr_interest_method, k_percent_per_day, k_stop_perc, aa_deliq_days, aa_repay_all_os, --  Погашения ОС за период
             aa_deliq_balance1, -- Просроченный баланс >= 1 дня
             aa_deliq_balance30, -- Просроченный баланс > 30
             aa_repay_all_kommis, -- «Комиссии полученные
             iif(js_code <> 'JobSystem.Specialist_SB', aa_kol_kc_in_region, 0), -- не включает в кол-во СВЗ СБ, js_id = 14 - id СБ
             aa_repay_all_perc, -- погашенные проценты
             last_curr_rate, -- курс валюты на конец даты
             aa_calc_perc_calced, aa_amort_kommis_period, aa_kol_skc_in_region, aa_loan_kind

      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 1, 1, 1, 1, 0)
      into :aa_k_id, :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name, :aa_curr_level2_name,
           :aa_curr_level3_name, :aa_curr_region_id, :aa_amount_active_begin, :aa_kolich_cl_active_begin,
           :aa_amount_active_end, :aa_kolich_cl_active_end, :aa_kolich_cl_disb, :aa_amount_disb, :aa_repay_normonth_os,
           :aa_repay_fact_period_os, :aa_deliq_cl_kolich, :aa_deliq_balance, :aa_deliq_cl_kolich7, :aa_deliq_balance7,
           :aa_repay_all_fine, :pr_interest_method, :k_percent_per_day, :k_stop_perc, :aa_deliq_days, aa_repay_all_os,
           aa_deliq_balance1, aa_deliq_balance30, aa_repay_all_kommis, aa_kol_kc_in_region, result_payed,
           last_curr_rate, result_calced, res_amount_period, aa_kol_skc_in_region, aa_loan_kind
  do
  begin
    /*
--  Вычисляет начисленные проценты и выплаченные проценты за период
      RESULT_CALCED = 0;
--      RESULT_PAYED = 0;
      select coalesce(RESULT_CALCED,0)
      from aarep_calc_perc_one(:AA_K_ID, :rep_date_begin, :rep_date_end, 1, null, null, null, null)
      into :RESULT_CALCED;
      RESULT_CALCED = RESULT_CALCED * LAST_CURR_RATE;
--  Вычисляет аммортизацию доп процентов, к признанию
      res_amount_period = 0;
      select -res_amount_period
      from aarep_calc_amort_kommis_temp(:AA_K_ID, :rep_date_begin, :rep_date_end)
      into :res_amount_period;
    */

--      if ((AA_LOAN_KIND='A Групповой') or (AA_LOAN_KIND='B Параллельный') or (AA_LOAN_KIND='E Парный'))  --Парные кредиты считаем групповым, по письму Эльнуры. mrahmanov
    if ((aa_loan_kind = 'A Групповой') /*or (AA_LOAN_KIND='B Параллельный')*/ or (aa_loan_kind = 'E Парный')) --Парные кредиты считаем групповым, по письму Эльнуры. mrahmanov
    then
      aa_loan_kind = 'A Групповой';
    else
      aa_loan_kind = 'B Индивидуальный';
    suspend;
  end

end
