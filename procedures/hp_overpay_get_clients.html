CREATE PROCEDURE HP_OVERPAY_GET_CLIENTS(
    REP_DATE_BEGIN D_DATE)
RETURNS (
    OPM_ID D_ID,
    CREATE_DATETIME D_DATETIME,
    STATUS D_SMALLINT,
    STATUS_DATETIME D_DATETIME,
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    REG_NAME D_STR_LONG,
    LOANS_ACTIVE_CNT D_INTEGER,
    LOANS_OVERDUE_CNT D_INTEGER,
    DEBT_SUM D_CURRENCY,
    OVERPAY_SUM D_CURRENCY,
    DEBT_FOR_PAYMENT D_CURRENCY,
    REMAIN_OVERPAY D_CURRENCY,
    LOANS_REPAY_CNT D_INTEGER,
    EDOC_SENT_CNT D_INTEGER,
    LO_NAME D_STR_255,
    KF_NAME D_STR_100,
    KONTRACT_NO D_STR_20,
    LAST_OVERPAY_DAYS D_INTEGER)
AS
begin
  for select opm_id, opm_create_datetime, opm_status, opm_status_datetime, opm_from_c_id
  from overpay_main
  where opm_status = 0 or opm_create_datetime > :rep_date_begin
  into opm_id, create_datetime, status, status_datetime, c_id
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    select count(opl_from_k_id), sum(iif(opl_overdue_days > 0, 1, 0)), sum(opl_sum), sum(opl_res_sum)
    from overpay_loans
    where opl_from_opm_id = :opm_id
      and opl_type = 2
    into loans_active_cnt, loans_overdue_cnt, debt_sum, debt_for_payment;

    select list(distinct reg_name)
    from overpay_loans, oksummary, region r
    where oks_from_k_id = opl_from_k_id
      and reg_unicode = oks_region_id
      and opl_from_opm_id = :opm_id
      and opl_type = 2
    into reg_name;

    select count(opl_from_k_id), sum(opl_sum), current_date - max(opl_date)
    from overpay_loans
    where opl_from_opm_id = :opm_id
      and opl_type = 1
    into loans_repay_cnt, overpay_sum, last_overpay_days;

    remain_overpay = iif(overpay_sum > debt_sum , overpay_sum - debt_sum, 0);

    select count(edl_id)
    from edoc_list
    where edl_object_kind = 117
      and edl_object_id = :opm_id
      and edl_from_edc_id in (32, 6) --RFCKZ-494
    into edoc_sent_cnt;

    select c.c_name_fio, kal.kal_new_kf_name, k.k_kontract_no
    from overpay_loans opl
    inner join kontract k on k.k_id = opl.opl_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
    inner join clients c on c.c_id = kal.kal_new_kf_officer
    where opl_from_opm_id = :opm_id
      and opl_type = 2
      and current_date between kal.kal_date and kal.kal_date_end
      and kal.is_delete = 0
    order by k.k_disb_date desc
    rows 1
    into lo_name, kf_name, kontract_no;
    suspend;
  end 
end
