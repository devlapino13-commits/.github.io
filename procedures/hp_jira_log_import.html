CREATE PROCEDURE HP_JIRA_LOG_IMPORT(
    LOG_ISSUE_ID D_INTEGER,
    LOG_FIELD D_STR_CODE,
    LOG_OLD_VALUE D_STR_100,
    LOG_NEW_VALUE D_STR_100,
    LOG_DATETIME D_STR_CODE)
AS
declare variable state_old d_smallint;
declare variable state_new d_smallint;
declare variable dev_id d_id;
declare variable orig_datetime d_str_code;
declare variable db_id d_integer;
declare variable real_datetime d_datetime;
declare variable new_login d_str_code;
begin
  orig_datetime = log_datetime;
  log_datetime = left(log_datetime, 23);
  real_datetime = replace(log_datetime, 't', ' ');

  state_old = 0;
  state_new = 0;

  db_id = f_get_db_id();
  if (db_id = 81001) then
    real_datetime = dateadd(hour, -3, real_datetime);

  if (log_field = 'assignee') then
  begin
    dev_id = null;

    if (coalesce(log_new_value, '') <> '') then
    begin
      new_login = '';

      select jrd_id
      from jira_dev
      where jrd_fio = :log_new_value
      into dev_id;

      if (dev_id is null) then
      begin
        select jru_login
        from jira_user
        where jru_fio = :log_new_value
        into new_login;
  
        if (new_login <> '') then
        begin
          select jrd_id
          from jira_dev
          where jrd_login = :new_login
          into :dev_id;
        end

        if (dev_id is null) then
        begin
          new_login = log_new_value;
  
          insert into jira_dev (jrd_login, jrd_fio)
          values (:log_new_value, :log_new_value)
          returning jrd_id
          into :dev_id;
        end
      end
    end

    select jrl_new_state
    from jira_log
    where jrl_issue_id = :log_issue_id and
          jrl_orig_datetime < :orig_datetime
    order by jrl_new_datetime desc, jrl_id desc
    rows 1
    into state_new;

    if (row_count = 1) then
      state_old = state_new;
    else
    begin
      log_old_value = '';
      log_new_value = 'Создана';
      log_field = 'status';
    end
  end

  if (log_field = 'status' and state_new = 0) then
  begin
    if (state_old = 0 and coalesce(log_old_value, '') <> '') then
    begin
      select jrs_id
      from jira_states
      where jrs_text = :log_old_value
      into :state_old;
    
      if (row_count = 0) then
      begin
        insert into jira_states (jrs_text)
        values (:log_old_value)
        returning jrs_id
        into :state_old;
      end
    end

    if (state_new = 0 and coalesce(log_new_value, '') <> '') then
    begin
      select jrs_id
      from jira_states
      where jrs_text = :log_new_value
      into :state_new;
    
      if (row_count = 0) then
      begin
        insert into jira_states (jrs_text)
        values (:log_new_value)
        returning jrs_id
        into :state_new;
      end
    end
  end

  if (not exists(select * from jira_log
                 where jrl_issue_id = :log_issue_id and
                       jrl_old_state = :state_old and
                       jrl_new_state = :state_new and
                       jrl_orig_datetime = :orig_datetime)) then
  begin
    insert into jira_log (jrl_issue_id, jrl_old_state, jrl_new_state, jrl_dev_id, jrl_new_datetime, jrl_orig_datetime)
    values (:log_issue_id, :state_old, :state_new, :dev_id, :real_datetime, :orig_datetime);
  end
end
