CREATE PROCEDURE HP_OFFICERPAR7_UPDATE(
    REP_DATE_END DATE)
AS
declare variable oks_kc_id decimal(18,0);
declare variable oks_balance decimal(18,4);
declare variable delinq_balance8_180 decimal(18,4);
declare variable par8_180 decimal(5,5);
begin
  for select oks_kc_id, sum(oks_balance), sum(delinq_balance8_180)
      from aarep_list_group_par8_180_tmp(1000000, :rep_date_end)
      group by oks_region_id, oks_kc_id
      into oks_kc_id, oks_balance, delinq_balance8_180
  do
  begin
    par8_180 = cast(delinq_balance8_180 as double precision) / oks_balance * 100;

    insert into officerpar7 (op_kc_id, op_date_begin, op_balance, op_delinq_bal8_180, op_par8_180)
    values (:oks_kc_id, :rep_date_end + 1, :oks_balance, :delinq_balance8_180, :par8_180);

--      suspend;
  end
end
