CREATE PROCEDURE AAREP_LORONOSTRO_LINK(
    REG_ID INTEGER,
    ACC_DT VARCHAR(20),
    ACC_KT VARCHAR(20))
RETURNS (
    RES_REG_ID INTEGER,
    RES_ACC_DT VARCHAR(20),
    RES_ACC_KT VARCHAR(20),
    REG_CURRENT INTEGER,
    TOTAL_LINE INTEGER,
    RES_LINE_NO INTEGER)
AS
declare variable reg_loronostro_head integer;
declare variable reg_dt integer;
declare variable reg_kt integer;
declare variable acc varchar(20);
begin
  total_line = 0;
  res_line_no = 0;
  select reg_loronostro_head
  from region
  where reg_unicode = :reg_id
  into :reg_loronostro_head;

  select al_from_reg_id
  from acc_list
  where al_id = :acc_dt
  into :reg_dt;
  if (reg_dt is null) then
  begin
    select reg_unicode
    from region
    where reg_code = left(:acc_dt, 4)
    into :reg_dt;
  end

  select al_from_reg_id
  from acc_list
  where al_id = :acc_kt
  into :reg_kt;
  if (reg_kt is null) then
  begin
    select reg_unicode
    from region
    where reg_code = left(:acc_kt, 4)
    into :reg_kt;
  end

  if (reg_dt <> reg_kt) then
  begin
    if ((reg_dt = reg_loronostro_head) or (reg_kt = reg_loronostro_head)) then
      total_line = 2;
    else
      total_line = 3;
  end
  else
    total_line = 1;

  if (reg_dt <> reg_kt) then
  begin
    if ((reg_dt = reg_loronostro_head) or (reg_kt = reg_loronostro_head)) then
    /* Один из участников это центр (БРУ и ГО в других странах) */
    begin -- Всего проводок 2
      if (reg_dt = reg_id) then -- Дт остается, Кт в другом офисе
      begin
-- нужно вычислить Лоро-ностро по Кт
        res_acc_dt = acc_dt;
        res_reg_id = :reg_dt;
        res_acc_kt = null;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_dt and
              ln_region_nostro = :reg_kt
        into :res_acc_kt;
        reg_current = 1;
        res_line_no = res_line_no + 1;
        suspend;

-- нужно вычислить Лоро-ностро по Дт в другом офисе
        res_acc_kt = acc_kt;
        res_reg_id = :reg_kt;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_kt and
              ln_region_nostro = :reg_dt
        into :res_acc_dt;
        reg_current = 0;
        res_line_no = res_line_no + 1;
        suspend;
      end
      if (reg_kt = reg_id) then -- Кт остается, Дт в другом офисе
      begin
-- нужно вычислить Лоро-ностро по Dт
        res_acc_kt = acc_kt;
        res_reg_id = :reg_kt;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_kt and
              ln_region_nostro = :reg_dt
        into :res_acc_dt;
        reg_current = 1;
        res_line_no = res_line_no + 1;
        suspend;

-- нужно вычислить Лоро-ностро по Кт в другом офисе
        res_acc_dt = acc_dt;
        res_reg_id = :reg_dt;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_dt and
              ln_region_nostro = :reg_kt
        into :res_acc_kt;
        reg_current = 0;
        res_line_no = res_line_no + 1;
        suspend;
      end
    end
    else
    begin -- Центр не участвует в проводках, итоговых проводок будет 3
      if (reg_dt = reg_id) then -- Дт остается, Кт в другом офисе
      begin
-- нужно вычислить Лоро-ностро по Кт
        res_acc_dt = acc_dt;
        res_reg_id = :reg_dt;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_dt and
              ln_region_nostro = :reg_loronostro_head
        into :res_acc_kt;
        reg_current = 1;
        res_line_no = res_line_no + 1;
        suspend;

        res_reg_id = :reg_loronostro_head;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_loronostro_head and
              ln_region_nostro = :reg_dt
        into :res_acc_dt;
        reg_current = 0;

        select ln_account
        from loronostro
        where ln_region_loro = :reg_loronostro_head and
              ln_region_nostro = :reg_kt
        into :res_acc_kt;
        reg_current = 0;
        res_line_no = res_line_no + 1;
        suspend;

-- нужно вычислить Лоро-ностро по Дт в другом офисе
        res_acc_kt = acc_kt;
        res_reg_id = :reg_kt;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_kt and
              ln_region_nostro = :reg_loronostro_head
        into :res_acc_dt;
        reg_current = 0;
        res_line_no = res_line_no + 1;
        suspend;
      end
      if (reg_kt = reg_id) then -- Кт остается, Дт в другом офисе
      begin
-- нужно вычислить Лоро-ностро по Dт
        res_acc_kt = acc_kt;
        res_reg_id = :reg_kt;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_kt and
              ln_region_nostro = :reg_loronostro_head
        into :res_acc_dt;
        reg_current = 1;
        res_line_no = res_line_no + 1;
        suspend;

        res_reg_id = :reg_loronostro_head;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_loronostro_head and
              ln_region_nostro = :reg_kt
        into :res_acc_kt;
        reg_current = 0;

        select ln_account
        from loronostro
        where ln_region_loro = :reg_loronostro_head and
              ln_region_nostro = :reg_dt
        into :res_acc_dt;
        reg_current = 0;
        res_line_no = res_line_no + 1;
        suspend;

-- нужно вычислить Лоро-ностро по Кт в другом офисе
        res_acc_dt = acc_dt;
        res_reg_id = :reg_dt;
        select ln_account
        from loronostro
        where ln_region_loro = :reg_dt and
              ln_region_nostro = :reg_loronostro_head
        into :res_acc_kt;
        reg_current = 0;
        res_line_no = res_line_no + 1;
        suspend;
      end
    end
  end
  else
  begin
    res_reg_id = :reg_id;
    res_acc_dt = :acc_dt;
    res_acc_kt = :acc_kt;
    reg_current = 1;
    res_line_no = res_line_no + 1;
    suspend;
  end
end
