begin
  res_html = '';
  color_green = '#02b316';
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);
  is_disb = 0;
  k_id = null;
  if (:obj_kind = 9) then
  begin
    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :obj_id
    into ma_id;
    k_id = :obj_id;
  end
  else
    ma_id = :obj_id;

  if (coalesce(:k_id, 0) > 0 and
      exists(select *
             from kontstate
             where ks_from_k_id = :k_id and
                   ks_state = 1 and
                   ks_ol_is_delete = 0)) then
    is_disb = 1;
  for select c_id, c_name_fio, c_verify_status, knc.kc_loan_target, bst.bs_name, coalesce(c_is_public, 0)
      from knclient knc, clients, bussubtype bst
      where bst.bs_id = knc.kc_busuness_subtype and
            kc_from_k_id = :k_id and
            kc_from_c_id = c_id and
            kc_amount_approved > 0
      union
      select c_id, c_name_fio, c_verify_status, mcl.mac_loan_target, bst.bs_name, coalesce(c_is_public, 0)
      from miniappclient mcl, bussubtype bst, clients
      where mcl.mac_business_subtype = bst.bs_id and
            mac_from_ma_id = :ma_id and
            mac_from_c_id = c_id and
            mac_amount > 0
      order by 2
      into :c_id, :c_name_fio, :c_verify_status, :loan_target, :business_subtype, :npdl_status
  do
  begin
    loan_target_str = '';
    select rd_name
    from refdata
    where rd_id = :loan_target
    into :loan_target_str;

    select count(distinct ks_from_k_id)
    from knclient, kontstate
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1
    into :c_circle;
    if (c_circle is null) then
      c_circle = 0;
    c_circle = c_circle + 1 - is_disb;
    if (db_id in (39801, 41701)) then
    begin
      poruch_c_id = null;
      poruch_fio = '';
      select mac.mac_from_c_id, cl.c_name_fio
      from miniappclient mac
      inner join miniapp ma on mac.mac_from_ma_id = ma.ma_id
      inner join clients cl on mac.mac_from_c_id = cl.c_id
      where mac.mac_from_ma_id = :ma_id and
            mac.mac_is_poruch = -1 and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999'
      union all
      select mac.mac_from_c_id, cl.c_name_fio
      from miniappclient mac
      inner join miniapp ma on mac.mac_from_ma_id = ma.ma_id
      inner join clients cl on mac.mac_from_c_id = cl.c_id
      inner join kontract k on ma.ma_id = k.k_from_ma_app_id
      where k.k_id = :k_id and
            mac.mac_is_poruch = -1 and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999' rows 1
      into :poruch_c_id, :poruch_fio;
    end
    -------------------------------------------
    res_field_name = f_uri_open(left(c_name_fio, 50), 'client', 'Clients', c_id, left(c_name_fio, 50), '', un_id);
    if (c_circle > 1) then
      res_field_values = iif(db_id <> 94401, 'цикл - ', 'dövr - ') || c_circle;
    else
      res_field_values = f_format('<font color="%1">%2</font>', :color_green, iif(db_id <> 94401, 'новый клиент', 'yeni müştəri'));
    res_html = blob_append(res_html, f_html_table_row(res_field_name, res_field_values, '', '', -1, -1, -1, -1, 2, ''));

    select pr_code
    from miniapp
    inner join product on pr_id = ma_from_pr_id
    where ma_id = :ma_id and
          ma_is_active = 1 and
          ma_date_end = '31.12.9999'
    into pr_code;

    if (db_id in (39801, 41701)) then
    begin
      select loan_description
      from hp_get_loan_description(:c_id, :ma_id)
      into :loan_description;
      res_html = blob_append(res_html, f_html_table_row('Цель', coalesce(loan_target_str, ''), '', '', -1, -1, -1, -1, 2, ''));
      ------------------------------------------- BOARD 4914
      if (:db_id <> 39801) then -- RFCKG-4210
        res_html = blob_append(res_html, f_html_table_row('Отрасль кредитования', coalesce(business_subtype, ''), '', '', -1, -1, -1, -1, 2, ''));
      res_html = blob_append(res_html, f_html_table_row('Подробное описание по цели', coalesce(loan_description, ''), '', '', -1, -1, -1, -1, 2, ''));
      if (poruch_c_id is not null) then
      begin
        res_html = blob_append(res_html, f_html_table_row(poruch_fio, f_uri_html(f_lang(lang, 'Лицо, дающее согласие (супруг(а))'), 'client_info', f_lang(lang, 'Информация по супругу(e)'), 'clients', :poruch_c_id, current_date, current_date, ''), '', '', -1, -1, -1, -1, 2, ''));
      end
      if (:ma_id > 0) then
      begin
        select m.ma_disb_kind, m.ma_date_app, m.ma_date_oper, mac.mac_amount, m.ma_insur_kind
        from miniapp m
        inner join miniappclient mac on mac.mac_from_ma_id = m.ma_id
        where ma_id = :ma_id and
              m.ma_date_end = '31.12.9999' and
              mac.mac_amount > 0 rows 1
        into :disb_kind_id, :ma_date_app, disb_date, loan_amount, :ma_insur_kind;
      end
      else
      begin
        select k.k_disb_kind
        from kontract k
        where k.k_id = :k_id
        into :disb_kind_id;
      end
      disb_kind = (case disb_kind_id
                     when 1 then 'Наличными'
                     when 6 then 'Карта Мбулак'
                     else 'Неизвестно'
                   end);
      if (:db_id <> 39801) then -- RFCKG-4210
      begin

        res_html = blob_append(res_html, f_html_table_row('Способ выдачи', disb_kind, '', '', -1, -1, -1, -1, 2, ''));
        -------------------------------------------
        if (:db_id = 41701 and
            pr_code not containing '.ignore_insurance.') then  ----RFCKG-4797
        begin
          insurance_required = 0;
          insur_amount = null;
          period_in_days = 0;

          if (f_get_const('new_loan_conditions') = 1) then
            insurance_min_amount = cast(f_get_const('kont_new_insurance_min_amount') as d_integer);
          else
            insurance_min_amount = cast(f_get_const('kont_insurance_min_amount') as d_integer);

          if (pr_code containing '.ipf.') then
            insurance_min_amount = cast(f_get_const('kont_insurance_ipf_min_amount') as d_integer);

          if (:k_id is not null) then
          begin
            select k.k_disb_date, k.k_amount_approved
            from kontract k
            where k.k_id = :k_id
            into :disb_date, loan_amount;

            select commission_amount
            from get_policy_commission_amount(:k_id)
            into :insur_amount;
          end
          else
          begin
            select sh.sh_date
            from schedule sh
            where sh.sh_from_k_id = -:ma_id
            order by sh.sh_id desc
            rows 1
            into :ma_repay_date;
            if (:ma_repay_date is not null) then
            begin
              period_in_days = ma_repay_date - disb_date;
              select commission_amount
              from calc_insurance_amount(:loan_amount, :period_in_days, :pr_code, :disb_date, -:ma_id) -- RFCKG-4803
              into :insur_amount;
            end
          end

          if (f_get_const('kont_insurance_active') = 1
            and pr_code containing '.IND.'
            and (pr_code containing '.STANDART.' or
                 pr_code containing '.IPF.')
            and pr_code not containing '.consultation.'
            and :loan_amount >= :insurance_min_amount) then
            insurance_required = 1;
            
          if (f_get_const('new_loan_conditions') = 1
            and f_get_const('kont_insurance_active') = 1
            and pr_code containing '.IND.'
            and pr_code containing '.STANDART.'
            and pr_code containing '.consultation.'
            and pr_code containing '.INSUR.') then
          begin
            point = coalesce(f_point(:c_id, :disb_date), 0);
            if (point < 20) then
              -- Баллы 0–19 страховка ОБЯЗАТЕЛЬНО
              insurance_required = 1;
            else
              -- Баллы 20 и выше → смотрим сумму кредита
              insurance_required = iif(:loan_amount >= 250000, 1, 0);
          end

          if (pr_code containing '.insur_kontrakt.') then -- RFCKG-4831
            insurance_required = 0;

          if (insurance_required = 1) then
            res_html = blob_append(res_html, f_html_table_row('Сумма страхования', iif(insur_amount is null, 'График погашений еще не построен', round(insur_amount, 2)), '', '', -1, -1, -1, -1, 2, ''));
        end

        insur_kind_text = iif(ma_insur_kind is not null, f_enum_name('ma_insur_payment_kind', ma_insur_kind), f_lang(lang, 'не указан'));
        res_html = blob_append(res_html, f_html_table_row(f_lang(lang, 'Способ оплаты страховой премии'), :insur_kind_text, '', '', -1, -1, -1, -1, 2, ''));

        select consent
        from hp_html_consent_spouse(:c_id, :ma_id, :ma_date_app)
        into consent;
        res_html = blob_append(res_html, f_html_table_row('Наличие согласия', :consent, '', '', -1, -1, -1, -1, 2, ''));
  
        ------- RFCKG-3107 ----------
        if (exists(select *
                   from mc_login ml
                   where ml.mcl_from_c_id = :c_id and
                         ml.mcl_is_delete = 0)) then
          presence_of_kmp = 'Да';
        else
          presence_of_kmp = 'Нет';
  
        res_html = blob_append(res_html, f_html_table_row('Наличие КМП', :presence_of_kmp, '', '', -1, -1, -1, -1, 2, ''));
        -----------------------------
      end
    end


    -- RFCRU-4167
    if (f_get_const('documented_income_active') = 1 and
       (pr_code containing '.reliable_client.' or pr_code containing '.loyal.' or pr_code containing '.comfortable.' or
        pr_code containing '.zalog.' or pr_code containing '.zalog_car.' or pr_code containing '.oivo.')) then -- RFCRU-5260
    begin
      for select mcl.mac_id
          from miniappclient mcl
          where mcl.mac_from_ma_id = :ma_id and
                mcl.mac_amount > 0
          into :mac_id
      do
      begin
        maci_is_documented = 0;
        maci_is_blocked = 0;

        select maci.maci_from_mac_id, maci.maci_is_documented, maci.maci_is_blocked, iif(maci_survey is null, 0, 1)
        from miniappclient_income maci
        where maci.maci_from_mac_id = :mac_id
        into :maci_from_mac_id, maci_is_documented, maci_is_blocked, is_maci_survey;

        if (k_id is not null) then
        begin
          read_only = iif(exists(select 1
                                 from kontstate ks
                                 where ks_from_k_id = :k_id and
                                       ks_ol_is_delete = 0 and
                                       ks_state = 1 and
                                       current_date between ks.ks_date_begin and ks.ks_date_end) or maci_is_blocked = 1, 1, 0);
        end
        else
          read_only = 1;
        res_field_name = 'Доход подтвержден документально';
        res_field_values = iif(maci_is_documented = 0, 'нет', 'да');

        if (read_only = 0) then
          res_link = f_uri_sql(f_lang(lang, 'Изменить'), 'set_maci_is_documented', 'MINIAPPCLIENT_INCOME', :mac_id, 'combo', 'Доход подтвержден документально?', '', 0, 0, :maci_is_documented, '', 'dialog', 'null');
        else
          res_link = '';

        res_html = blob_append(res_html, f_html_table_row(:res_field_name, '<b>' || res_field_values || '</b> ' || res_link || '<br>', '', '', -1, -1, -1, -1, 2, ''));

        if (maci_is_documented = 1) then
        begin
          anketa_read_only = 1;
          rd_id = 0;
          select rd_id
          from hp_get_check_loan_data(:k_id)
          into rd_id;
          acl_status = null;

          select first 1 acl.acl_status
          from actionlist acl
          where acl.acl_object_id = :k_id and
                acl.acl_object_kind = 9 and
                acl.acl_from_rd_id = :rd_id
          order by acl.acl_datetime desc
          into :acl_status;

          if (acl_status is null) then
            anketa_read_only = 0;

          if (acl_status = -1) then

            select obm_memo

            into :obm_memo;

                       from verifycomment
                       where vcm_code = '25' and
                             :obm_memo containing vcm_caption)) then
              anketa_read_only = 0;

          if (anketa_read_only = 1) then
            uri_text = 'Просмотр';

            uri_text = 'Заполнить';

            uri_text = 'Изменить';

            res_link = f_uri_open(uri_text, 'dpd_anketa', 'MINIAPPCLIENT_INCOME', :mac_id, uri_text, 'read_only=' || anketa_read_only, :un_id);
          end

            res_link = '';

          res_html = blob_append(res_html, f_html_table_row('Анкета', '<span style="color:blue">' || res_link || '</span>', '', '', -1, -1, -1, -1, 2, ''));

            from knclient knc, kontract k
            where knc.kc_from_k_id = :k_id and
                  knc.kc_from_k_id = k.k_id and

          res_html = blob_append(res_html, f_html_table_row(:res_field_name, '<b>' || res_field_values || '</b> ' || '' || '<br>', '', '', -1, -1, -1, -1, 2, ''));
        end

      begin
        select kcp.kc_pdn_koef, kcp.kc_loan_expense
        from knclient_pdn kcp
        where kcp.kc_id =  :kc_id
            and kcp.kcp_kind = 0
        into :final_pdn,
             :calc_month_repay_final;

        res_field_name = 'ПДН';
        res_field_values = iif(:final_pdn is not null, cast(:final_pdn as d_str_20), 'Нет данных');
        res_html = blob_append(res_html, f_html_table_row(res_field_name, '<b>' || res_field_values || '</b> ', '', '', -1, -1, -1, -1, 2, ''));
        
        res_field_name = 'ССП';
        res_field_values = iif(:calc_month_repay_final is not null, cast(:calc_month_repay_final as d_str_20), 'Нет данных');
        res_html = blob_append(res_html, f_html_table_row(res_field_name, '<b>' || res_field_values || '</b> ', '', '', -1, -1, -1, -1, 2, ''));
      end
    end

    if (db_id = 81001 and obj_kind = 9) then
    begin
      res_field_name = f_lang(lang, 'Рефинансированный заем (займы)');
      select coalesce(list(k_kontract_no, '; '), '')
      from kontrefinance
      left join kontract on k_id = kr_from_k_id and
            kr_from_k_id is not null
      where kr_from_ma_id = :ma_id
      into :res_field_values;
      res_link = '';
      if (res_field_values <> '') then
        res_html = blob_append(res_html, f_html_table_row(res_field_name, '<b>' || res_field_values || '</b> ' || res_link || '<br>', '', '', -1, -1, -1, -1, 2, ''));
    end

    if (db_id = 41701 and f_get_const('self_prohibition') = 1) then      -- RFCKG-4711
    begin

      select
            case
              when res = 0 then 'Запрос отправлен'
              when res = 1 then '<span style="color: red;">У клиента действует самозапрет. Выдача невозможна</span>'
              when res = 2 then 'У клиента не установлен самозапрет'
              when res = -1 and spm_query_qty < 2 then '<span style="color: red;">Внутренняя ошибка сервиса</span>'
              when res = -1 and spm_query_qty = 2  then '<span style="color: red;">Ошибка при получении статуса клиента. Прикрепите документ</span>'
              when res = -2 then '<span style="color: red;">Ошибка при получении статуса клиента. Прикрепите документ</span>'
              when coalesce(res, -777) = -777 then 'Запрос не осуществлялся'
              when res = -3 then 'Запрос не осуществлялся'
              else ''
            end,
            coalesce(res, -777)
      from hp_self_prohibition_get_status(:c_id, f_objkind(:obj_kind, 'name'), :obj_id)
      into self_prohibition_status_str,
           self_prohibition_status;

      if (exists (select *
                  from clientdocs cds
                  inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                  inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
                  where pdt.pdt_code = 'self_prohibition' and
                        cds.cds_from_c_id = :c_id and
                        cds.cds_doc_date_begin = current_date) and
          self_prohibition_status <> 1) then
        self_prohibition_status_str = 'Документ прикреплен';

      res_field_name = f_lang(lang, 'Самозапрет');
      res_link = '';
      res_html = blob_append(res_html, f_html_table_row(res_field_name, self_prohibition_status_str, '', '', -1, -1, -1, -1, 2, ''));
    end

    if (db_id = 41701) then      -- RFCKG-4801
    begin
       select eil.eil_status
       from extinfolist eil
       inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
       where eic.eic_code = 'check_npdl' and
             eil.eil_object_kind = f_objkind('miniapp', 'id') and
             eil.eil_object_id = :ma_id and
             eil.eil_from_c_id = :c_id
       order by eil.eil_status_datetime
       rows 1
       into :npdl_eil_status;

       if (npdl_eil_status = 0) then
