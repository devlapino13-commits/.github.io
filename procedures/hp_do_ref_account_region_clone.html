CREATE PROCEDURE HP_DO_REF_ACCOUNT_REGION_CLONE(
    ACC_ID ID_TYPE)
AS
declare variable reg_id d_integer;
declare variable al_id d_str_code;
declare variable acc_from_reg_id d_integer;
declare variable reg_code d_str_code;
declare variable new_acc_id d_id;
declare variable new_acc_name d_str_255;
declare variable acc_date_open d_date;
declare variable acc_code d_str_code;
declare variable new_al_id d_str_code;
declare variable acc_name d_str_255;
declare variable acc_from_ab_id d_id;
declare variable acc_from_sab_id d_id;
declare variable acc_from_cu_id d_str_long;
declare variable acc_from_inx_id d_id;
declare variable acc_number d_id;
declare variable reg_name d_str_255;
declare variable new_reg_code d_str_code;
declare variable new_reg_name d_str_255;
begin
  select acc_code, acc_from_reg_id, acc_date_open, acc_name, reg_code, reg_name, acc_from_ab_id, acc_from_sab_id,
         acc_from_cu_id, acc_from_inx_id, acc_number
  from ref_account, region
  where acc_id = :acc_id and
        acc_from_reg_id = reg_unicode and
        acc_object_id is null
  into :acc_code, :reg_id, acc_date_open, acc_name, reg_code, reg_name, :acc_from_ab_id, :acc_from_sab_id,
       :acc_from_cu_id, :acc_from_inx_id, :acc_number;

  if (reg_id is null) then
    exception e_default 'Can not clone account with object_id or empty region';

  if (acc_code not starting '7') then
    exception e_default 'Can not clone non income/expense account';

  select al_id
  from acc_list
  where al_from_acc_id = :acc_id and
        al_from_reg_id = :reg_id
  into :al_id;

  for select reg_unicode, reg_code, reg_name
      from region
      where reg_unicode <> :reg_id and
            reg_status = 3 and
            coalesce(reg_date_close, '31.12.2999') >= :acc_date_open
      into acc_from_reg_id, new_reg_code, new_reg_name
  do
  begin
    new_acc_name = replace(acc_name, reg_code, new_reg_code);
    new_acc_name = replace(new_acc_name, reg_name, new_reg_name);

    acc_code = null;
    select acc_code
    from get_new_ref_account_code(:acc_from_ab_id, :acc_from_sab_id, :acc_from_cu_id, :acc_from_reg_id, :acc_from_inx_id, :acc_number, null, null, 0)
    into :acc_code;

    new_acc_id = null;
    select acc_id
    from ref_account
    where acc_code = :acc_code
    into new_acc_id;

    if (new_acc_id is null) then
    begin
      insert into ref_account (acc_from_ab_id, acc_from_sab_id, acc_from_reg_id, acc_from_cu_id, acc_from_inx_id,
                               acc_object_kind, acc_object_id, acc_number, acc_is_currency, acc_name, acc_memo,
                               acc_type, acc_out_balance, acc_date_open, acc_non_resident)
      select acc_from_ab_id, acc_from_sab_id, :acc_from_reg_id, acc_from_cu_id, acc_from_inx_id, acc_object_kind,
             acc_object_id, acc_number, acc_is_currency, :new_acc_name, acc_memo, acc_type, acc_out_balance,
             acc_date_open, acc_non_resident
      from ref_account
      where acc_id = :acc_id
      returning acc_id
      into new_acc_id;
    end

    if (al_id is not null) then
    begin
      new_al_id = new_reg_code || substring(al_id from 5);

      update acc_list
      set al_from_acc_id = :new_acc_id
      where al_id = :new_al_id and
            al_from_reg_id = :acc_from_reg_id;
    end
  end
end
