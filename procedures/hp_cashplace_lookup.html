CREATE PROCEDURE HP_CASHPLACE_LOOKUP(
    OFFICE_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_ID D_INTEGER,
    RES_TEXT D_STR_CODE)
AS
declare variable c_id d_id;
declare variable uxa_uuid d_uuid;
declare variable full_access d_bool;
declare variable cls_acc d_str_20;
declare variable ipf_acc d_str_20;
declare variable un_name d_str_code;
begin
  if (un_id is null) then
    un_id = uxf$user();

  select un_name, un_from_c_id
  from username
  where un_id = :un_id
  into un_name, c_id;

  full_access = 0;
  if (exists(select * from cashplace
  where csp_office_id = :office_id and
        csp_cashier_id = :c_id and
        csp_is_main = 1)) then
    full_access = 1;

  if (full_access = 0 and uxf$user_is('superuser', un_id) = 1) then
    full_access = 1;

  if (full_access = 0 and un_name = 'kabdymalikova') then
    full_access = 1;

  if (full_access = 0) then
  begin
    for select csp_id, cls.al_account, ipf.al_account,
               trim(iif(csp_kind = 0, 'Касса', 'Расчет')) || ' ' || csp_cash_no || '  (' || reg_code || ')'
    from cashplace
         left join acc_list cls on (csp_cls_al_id = cls.al_id)
         left join acc_list ipf on (csp_ipf_al_id = ipf.al_id)
         inner join region on (csp_office_id = reg_unicode)
    where csp_office_id = :office_id and
          csp_cashier_id = :c_id
    into res_id, cls_acc, ipf_acc, res_text
    do
    begin
      if (coalesce(cls_acc, '') <> '' and coalesce(ipf_acc, '') <> '') then
        res_text = res_text || '   (' || cls_acc || ', ' || ipf_acc || ')';
      else
      if (coalesce(cls_acc, '') <> '') then
        res_text = res_text || '   (' || cls_acc || ')';
      else
      if (coalesce(ipf_acc, '') <> '') then
        res_text = res_text || '   (' || ipf_acc || ')';

      suspend;
    end
  end

  if (res_id is null or full_access = 1) then
  begin
    if (full_access = 0) then
    begin
      uxa_uuid = uxf$uniq('afm_cashoffice.list', 0);
      if (uxa_uuid is null) then
        exit;
  
      if (uxf$access(un_id, uxa_uuid) <> 1) then
        exit;
    end

    for select csp_id, cls.al_account, ipf.al_account,
               trim(iif(csp_kind = 0, 'Касса', 'Расчет')) || ' ' || csp_cash_no || ',  ' || reg_code
    from cashplace
         left join acc_list cls on (csp_cls_al_id = cls.al_id)
         left join acc_list ipf on (csp_ipf_al_id = ipf.al_id)
         inner join region on (csp_office_id = reg_unicode)
    where csp_office_id = :office_id
    into res_id, cls_acc, ipf_acc, res_text
    do
    begin
      if (coalesce(cls_acc, '') <> '' and coalesce(ipf_acc, '') <> '') then
        res_text = res_text || '   (' || cls_acc || ', ' || ipf_acc || ')';
      else
      if (coalesce(cls_acc, '') <> '') then
        res_text = res_text || '   (' || cls_acc || ')';
      else
      if (coalesce(ipf_acc, '') <> '') then
        res_text = res_text || '   (' || ipf_acc || ')';

      suspend;
    end
  end
end
