CREATE PROCEDURE HP_EFFECT_RATE_AZ(
    K_ID D_ID)
RETURNS (
    CALC_EFF DECIMAL(18,3))
AS
declare variable k_disb_date date;
begin
  select k_disb_date
  from kontract
  where k_id = :k_id
  into :k_disb_date;

  if (k_disb_date <= '12.03.2017') then
  begin
    select calc_eff
    from hp_effect_rate_az_old(:k_id)
    into :calc_eff;
  end
  else if (k_disb_date <= '07.08.2019') then
  begin
    select calc_eff
    from hp_effect_rate_kg(:k_id)
    into calc_eff;
    calc_eff = (power(1 + calc_eff / 100.000, 1.000 / 12.000) - 1) * 12 * 100;
  end
  else
  begin
    select calc_eff
    from hp_effect_rate_az_old(:k_id)
    into :calc_eff;
    calc_eff = round(calc_eff, 2);
  end
  suspend;
end
