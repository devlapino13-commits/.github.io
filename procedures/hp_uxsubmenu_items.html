CREATE PROCEDURE HP_UXSUBMENU_ITEMS(
    UX_CODE D_STR_CODE,
    UXA_CODE D_STR_CODE,
    HIDE_EMPTY D_BOOL = 0,
    SHOW_ROOT D_BOOL = 0)
RETURNS (
    RES_CODE D_STR_20,
    RES_PARENT_CODE D_STR_20,
    RES_CAPTION D_STR_100,
    RES_HINT D_STR_255,
    RES_UA_CODE D_STR_100,
    RES_INACTIVE D_BOOL,
    RES_CHECKED D_BOOL,
    RES_IMAGE D_INTEGER,
    RES_DISABLE_ON_EDIT D_BOOL,
    RES_IS_DEFAULT D_BOOL)
AS
declare variable res_order_code d_str_20;
declare variable ux_kind d_str_code;
declare variable uxa_kind d_integer;
declare variable ua_usr_id_field d_str_code;
begin
  if (uxa_code is null) then
    uxa_code = '';

  for select 'a_00', 'a_', '', 'Действия', 0, 0, '', 0, '', ''
  from rdb$database
  union select 'b_00', 'b_', '', 'Бланки', 0, 0, '', 0, '', ''
  from rdb$database
  union
  select usm_menu_index || '00', usm_menu_index, '', usm_caption, 0, 0, '', 0, '', ''
  from uxsubmenu
  where usm_ux_code = :ux_code
  union
  select ua_menu_index, ua_menu_index, ua_code, ua_name, ua_inactive, uxa_kind,
         ux_kind, ua_disable_on_edit, ua_hint, ua_usr_id_field
  from uxaction, uxalias, uxcode
  where ua_ux_code = :ux_code and
        ua_uxa_code = uxa_code and
        uxa_ux_code = ux_code
  order by 1
  into :res_order_code, :res_code, res_ua_code, res_caption, :res_inactive, uxa_kind,
       ux_kind, res_disable_on_edit, res_hint, ua_usr_id_field
  do
  begin
    res_checked = 1;
    res_parent_code = null;
    res_is_default = 0;

    if (res_ua_code = '') then
    begin
      if (show_root = 0 and res_code in ('a_', 'b_')) then
        continue;

      if (not(show_root = 1 and res_code in ('a_', 'b_'))) then
      begin
        if (not exists(select * from uxaction
        where ua_ux_code = :ux_code and
              ua_menu_index starting :res_code and
              ua_menu_index <> :res_code)) then
          Continue;
      end

      if (uxa_code <> '') then
      begin
        if (not exists(select * from uxaction, uxaliasaction
        where ua_ux_code = :ux_code and
              ua_code = uaa_ua_code and
              uaa_uxa_code = :uxa_code and
              ua_menu_index starting :res_code and
              ua_menu_index <> :res_code)) then
        begin
          if (hide_empty = 1) then
            continue;
          else
            res_checked = 0;
        end
      end

      if (char_length(res_code) > 4) then
        res_parent_code = left(res_code, char_length(res_code) - 2) || '00';

      res_code = res_order_code;

      res_image = 36; -- папка
    end else
    begin
      if (char_length(res_code) > 4) then
        res_parent_code = left(res_code, char_length(res_code) - 2) || '00';

      if (uxa_code <> '') then
      begin
        select uaa_is_default
        from uxaliasaction
        where uaa_uxa_code = :uxa_code and
              uaa_ua_code = :res_ua_code
        into res_is_default;

        if (row_count = 0) then
        begin
          if (hide_empty = 1) then
            continue;
          else
            res_checked = 0;
        end
      end

      ux_kind = uxf$kind(ux_kind, 'code');

      res_image = 19; -- Форма

      res_image = case
        when (ux_kind = 'uxframe' and ua_usr_id_field <> '') then 4 -- поисковик
        when (ux_kind = 'uxframe' and ua_usr_id_field = '') then 17 -- Фрейм
        when (ux_kind = 'uxform' and uxa_kind = 0) then 19 -- Форма
        when (ux_kind = 'uxform' and uxa_kind = 2) then 55 -- Создание записи
        when ux_kind = 'uxcommand' then 25 -- Команда
        when ux_kind = 'uxreport' then 26 -- Отчет
      end;
    end

    if (show_root = 1 and res_parent_code is null) then
      res_parent_code = left(res_code, 2) || '00';

    if (res_code in ('a_00', 'b_00')) then
      res_parent_code = null;

    suspend;
  end
end
