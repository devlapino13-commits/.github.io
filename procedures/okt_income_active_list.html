CREATE PROCEDURE OKT_INCOME_ACTIVE_LIST(
    USER_NAME D_STR_100)
RETURNS (
    K_KONTRACT_NO D_STR_100,
    UN_NAME D_STR_100,
    ABONENT_FIO D_STR_100,
    ABONENT_TYPE D_STR_100,
    CALL_DATETIME D_DATETIME,
    URL D_STR_4096,
    COUNTER D_INTEGER,
    PROJECT_NAME D_STR_100)
AS
declare variable CLIENT_KIND_ID D_INTEGER;
declare variable OTL_OBJECT_KIND D_INTEGER;
declare variable OTL_OBJECT_ID D_ID;
declare variable OTL_PANT_UN_ID D_ID;
declare variable OTL_ID integer;
declare variable PHONE_TYPE D_STR_100;
declare variable PROJECT D_STR_100;
declare variable PANT_ID D_ID;
declare variable FORM_NAME D_STR_20;
declare variable C_ID D_ID;
declare variable CLIENT_PHONE D_STR_20;
declare variable PHONE D_STR_20;
declare variable K_ID D_ID;
begin
    client_kind_id = f_objkind('clients', 'id');
    counter = 0;

    for select kont.k_kontract_no,
               kont.k_id,
               t.otl_id,
               t.otl_object_kind,
               t.otl_object_id,
               t.otl_pant_un_id,
               t.otl_datetime,
               t.otl_phone,
               iif(t.otl_is_main_phone_no = 1, 'ContactInfo.Cell_Main', 'ContactInfo.Cell_Add'),
               rd.rd_code,
               rd.rd_name,
               pant_id,
               un.un_name
    from okt_income_task_list t
    inner join pantera_kontracts pank on t.otl_from_pant_id = pank.pant_id
    inner join kontract kont on kont.k_id = pank.pant_from_k_id
    inner join refdata rd on rd.rd_id = pank.pant_rd_project_type
    inner join username un on un.un_id = otl_pant_un_id
    where t.otl_is_active = 1
    and t.otl_date >= current_date - 3  -- < добавил igusev. Mantis task № 0000137: Пропущенные звонки в Октелл
    order by t.otl_id
    into :k_kontract_no,
         :k_id,
         :otl_id,
         :otl_object_kind,
         :otl_object_id,
         :otl_pant_un_id,
         :call_datetime,
         :phone,
         :phone_type,
         :project,
         :project_name,
         :pant_id,
         :un_name
    do
    begin
        if ((un_name = user_name) or (user_name = '---')) then
        begin
            abonent_fio = '';
            abonent_type = '';

            form_name = 'Sirena_d.aspx';
    
            if (exists (
              select res_un_id, pa_id, pa_name, pa_command_name
              from okt_permission_access(:un_name)
              where pa_command_name = 'oktell.projects.blind'
            )) then
                form_name = 'Sirena_d_lite.aspx'; -- < если оператор из незрячих, для него отрывается специфичная форма


            if (otl_object_kind = client_kind_id) then
            begin
                select cl.c_name_fio
                from clients cl
                where cl.c_id = :otl_object_id
                into :abonent_fio;

                abonent_type = 'Клиент';

                url = 'http://oktell.molbulak.ru/'||:form_name||'?project='||:project||'&pant_id='||:pant_id||'&object_id='||:otl_object_id||'&object_kind=Clients&client_cell_no='||:phone||'&K_ID='||:k_id||'&cont_call_active=0&phone_type='||:phone_type||'&call_system=OKTELL&un_id='||:otl_pant_un_id||'&otl_id='||:otl_id;
             end
            else
            begin
                select cont.cc_name,
                      coalesce(f_cont_info('clients', cont.cc_from_c_id, 'cell_main', 'ok'), ''),
                      cont.cc_from_c_id
                from contacts cont
                where cont.cc_id = :otl_object_id
                into :abonent_fio, client_phone, c_id;

                abonent_type = 'Контактное лицо';

                url ='http://oktell.molbulak.ru/'||form_name||'?project='||:project||'&pant_id='||:pant_id||'&object_id='||:otl_object_id||'&object_kind=Contacts&contact_cell_no='||:phone||'&client_cell_no='||:client_phone||'&c_id='||:c_id||'&K_ID='||:k_id||'&phone_type=ContactInfo.Cell_Main&call_system=OKTELL&un_id='||:otl_pant_un_id||'&otl_id='||:otl_id;
            end

            counter = counter + 1;
            suspend;
        end
    end
end
