CREATE PROCEDURE AAREP_CALC_AMORT_KOMMIS2(
    REPORT_REGION_ID DECIMAL(18,0),
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_NEED_PERC INTEGER)
RETURNS (
    RES_AMOUNT_BEGIN DECIMAL(18,4),
    RES_AMOUNT_END DECIMAL(18,4),
    RES_AMOUNT_PERIOD DECIMAL(18,4),
    RES_DISB_DATE DATE,
    RES_LAST_DATE DATE,
    RES_KRED_AMOUNT DECIMAL(18,4),
    RES_KOMMIS_AMOUNT DECIMAL(18,4),
    RES_PERC_DAILY DECIMAL(18,9),
    AAREP_K_ID DECIMAL(18,0),
    AAREP_KONTRACT_NO VARCHAR(20),
    AAREP_CLIENT_NAME VARCHAR(100),
    AAREP_PRODUCT_NAME VARCHAR(50),
    AAREP_LOAN_TYPE VARCHAR(50),
    AAREP_CURRENCY VARCHAR(3),
    AAREP_REGION_NAME VARCHAR(100),
    AAREP_PR_ID INTEGER,
    AAREP_REGION_ID INTEGER,
    AAREP_K_FOND VARCHAR(20),
    RES_KOMMIS_PAYED DECIMAL(18,4),
    K_KREDLINE_END DATE)
AS
declare variable curr_days1 decimal(18,4);
declare variable curr_days2 decimal(18,4);
declare variable max_days decimal(18,4);
declare variable kont_state integer;
declare variable aarep_loan_t integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable cash_id decimal(18,0);
begin
-- Старый вариант от 05.11.2013 г.

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select distinct k_currency, ks_from_k_id, k_kontract_no, k_clients_cashier_id, pr_loan_type, pr_name,
                      reg_code || ' ' || reg_name, pr_id, reg_unicode, k_from_f_id, k_kredline_end
      from kontstate, product, knaktlist, region, region_list(:report_region_id, 1, -1, 1), kontract
      where k_id = ks_from_k_id and
            pr_id = k_from_pr_id and
            ks_ol_is_delete = 0 and
            ((ks_date_end >= :report_date_begin and
            cast(ks_date_begin as date) <= :report_date_end and
            ks_state = 1) or
            (ks_date_begin = :report_date_begin and
            ks_state = 2)) and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :report_date_end and
            kal_date_end >= :report_date_end and
            (kal_new_reg_id = id_list) and
            kal_new_reg_id = reg_unicode and
            knaktlist.is_delete = 0 and
            k_disb_kind <> 6
      order by pr_loan_type, ks_from_k_id
      into :aarep_currency, :aarep_k_id, :aarep_kontract_no, :cash_id, :aarep_loan_t, :aarep_product_name,
           :aarep_region_name, :aarep_pr_id, :aarep_region_id, :aarep_k_fond, :k_kredline_end
  do
  begin
    if (cash_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :cash_id
      into :aarep_client_name;
    end
    else
      aarep_client_name = '';
    res_amount_begin = 0;
    res_amount_end = 0;
    res_amount_period = 0;
    res_kred_amount = 0;
    res_kommis_amount = 0;
    res_kommis_payed = 0;
    if (report_need_perc <> 0) then
    begin
      select res_amount_begin, res_amount_end, res_amount_period, res_disb_date, res_last_date, res_kred_amount,
             res_kommis_amount, res_perc_daily, res_kommis_payed
      from aarep_calc_amort_kommis_temp(:aarep_k_id, :report_date_begin, :report_date_end)
      into :res_amount_begin, :res_amount_end, :res_amount_period, :res_disb_date, :res_last_date, :res_kred_amount,
           :res_kommis_amount, :res_perc_daily, :res_kommis_payed;
      if (aarep_loan_t = 1) then
        aarep_loan_type = 'Индив.';
      else
        aarep_loan_type = 'Группа';
    end
    suspend;
  end
end
