CREATE PROCEDURE HP_KONTRACT_SCHEDULE_TO_DATE(
    KONT DECIMAL(18,0),
    DD DATE)
RETURNS (
    S_DATE DATE,
    S_NUMBER INTEGER,
    S_OS DECIMAL(18,4),
    S_PERC DECIMAL(18,4),
    P_OS DECIMAL(18,4),
    P_PERC DECIMAL(18,4),
    P_FINE DECIMAL(18,4),
    ZAD_OS DECIMAL(18,4),
    ZAD_PERC DECIMAL(15,2))
AS
begin
-- knclient_fix
-- возможно новый входящий параметр

  for select ok_sh_date, ok_sh_id, sum(ok_sh_os), sum(ok_sh_perc), sum(ok_os), sum(ok_perc), sum(ok_fine),
             sum(ok_sh_os + ok_os), sum(ok_sh_perc + ok_perc)
      from operkred
      where ok_from_k_id = :kont and
            ok_ol_is_active = 1 and
            ok_type < 8 and
            ok_ol_date <= :dd and
            ok_from_c_id = 0 and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
      group by ok_sh_date, ok_sh_id
      order by ok_sh_date, ok_sh_id
      into :s_date, :s_number, :s_os, :s_perc, :p_os, :p_perc, :p_fine, :zad_os, :zad_perc
  do
  begin
    if (s_os is null) then
      s_os = 0;
    if (s_perc is null) then
      s_perc = 0;
    if (p_os is null) then
      p_os = 0;
    if (p_perc is null) then
      p_perc = 0;
    if (p_fine is null) then
      p_fine = 0;
    if (zad_os is null) then
      zad_os = 0;
    if (zad_perc is null) then
      zad_perc = 0;
    suspend;
  end
end
