CREATE PROCEDURE KONTRACT_OPER_LIST(
    KONT DECIMAL(18,0))
RETURNS (
    V_DATE DATE,
    SH_ID INTEGER,
    SH_DATE DATE,
    V_OS DECIMAL(18,4),
    V_PERC DECIMAL(18,4),
    V_FINE DECIMAL(18,4),
    V_OPER_TYPE INTEGER)
AS
begin
  /* Procedure Text */
  v_os = 0;
  v_perc = 0;
  v_fine = 0;
  for select ok_type, ok_ol_date, ok_sh_id, ok_sh_date, -1 * ok_os, -1 * ok_perc, -1 * ok_fine
      from operkred
      where ok_from_k_id = :kont and
            ok_type <= 7 and
            ok_type > 1 and
            ok_ol_is_active = 1 and
            ok_from_c_id = 0 and
            ((ok_os <> 0) or
            (ok_perc <> 0) or
            (ok_fine <> 0)) and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
      order by ok_ol_date, ok_sh_id
      into :v_oper_type, :v_date, :sh_id, :sh_date, :v_os, :v_perc, :v_fine
  do
    suspend;
end
