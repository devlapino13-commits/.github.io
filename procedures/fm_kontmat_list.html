CREATE PROCEDURE FM_KONTMAT_LIST(
    K_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    KML_ID D_INTEGER,
    AG_ID D_INTEGER,
    AG_CAPTION D_STR_255,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    KML_GUARANTEE_MONTH D_INTEGER,
    KML_COUNT D_CURRENCY,
    KML_COST D_CURRENCY,
    KML_TOTAL D_CURRENCY,
    ML_FROM_YL_ID D_INTEGER,
    FEG_ID D_INTEGER,
    KML_IS_RELATIVE D_SMALLINT,
    KML_IS_USED D_SMALLINT,
    KMI_IMEI D_ID,
    SHOW_IMEI D_SMALLINT,
    QUANTITY D_INTEGER,
    WP_NAME D_STR_255,
    MOTV_NAME D_STR_255,
    YL_CODE D_STR_CODE,
    IS_OWN_GOOD D_SMALLINT,
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    REG_ID D_INTEGER,
    REG_NAME D_STR_100,
    WP_ID D_ID,
    MOTV_ID D_ID,
    BILL_ID ID_TYPE,
    BILL_LINE_NO D_ID,
    ML_ID D_ID)
AS
begin
  for select kml_id, ml_name, ml_model, ag_id, ag_caption, kml_guarantee_month, kml_count, kml_cost, kml_total,
             ml_from_yl_id, kml_from_feg_id, kml_is_relative, kml_is_used, ag_object_id, ml_id 
      from kontmat, matlist, agreements, refdata
      where kml_from_ml_id = ml_id and
            kml_status = 1 and
            kml_object_kind = 9 and
            kml_object_id = :k_id and
            kml_from_ag_id = ag_id and
            ag_from_rd_id = rd_id and
            rd_code in ('AgreementKinds.IPF_Sellers', 'AgreementKinds.tsp_and_ipf', 'AgreementKinds.IPF_Idjara')
      into kml_id, ml_name, ml_model, ag_id, ag_caption, kml_guarantee_month, kml_count, kml_cost, kml_total,
           ml_from_yl_id, feg_id, kml_is_relative, kml_is_used, c_id, ml_id 
  do
  begin
    is_own_good = 0;

    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into c_name_fio;

    if (exists(select *
               from kontmat_own kmo
               where kmo.kmo_kml_id = :kml_id and
                     kmo.kmo_k_id = :k_id)) then
    begin
      is_own_good = 1;
      select kmo.kmo_bill_id, kmo.kmo_bill_line_no, kmo.kmo_wp_id, kmo.kmo_motv_id
      from kontmat_own kmo
      where kmo.kmo_kml_id = :kml_id and
            kmo.kmo_k_id = :k_id
      into bill_id, bill_line_no, wp_id, motv_id;

      select sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus)
      from opermat om
      inner join operlist ol on ol.ol_id = om.om_from_ol_id and
            ol.is_delete = 0
      where om.om_bill_id = :bill_id and
            om.om_bill_line_no = :bill_line_no and
            om.om_workplace_id = :wp_id having sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0 or
            sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0
      into quantity;

      wp_name = null;
      reg_id = null;
      reg_name = null;

      select wp.wp_name,
             reg_unicode,
             reg_name
      from workplace wp
      left join region
            on reg_unicode = wp_from_reg_id
      where wp.wp_id = :wp_id
      into wp_name,
           reg_id,
           reg_name;

      motv_name = (select c_name_fio
                   from clients
                   where c_id = :motv_id);

      select yl.yl_code
      from yedizlist yl
      where yl.yl_id = :ml_from_yl_id
      into yl_code;
    end
    kmi_imei = null;
    show_imei = 0;
    select kmi.kmi_imei
    from kontmat_imei kmi
    where kmi.kmi_kml_id = :kml_id
    into kmi_imei;
    if (coalesce(kmi_imei, 0) > 0) then
      show_imei = 1;
    if (exists(select *
               from foreign_econom_goods_ref feg
               where feg.feg_id = :feg_id and
                     feg.feg_code = '8517120000')) then
      show_imei = 1;
    suspend;
  end
end
