CREATE PROCEDURE AAREP_KM_KOBD_EMAIL_ABSENT(
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    RESULT_EMAIL STR_100,
    FIO STR_100,
    JL_NAME STR_100)
AS
declare variable reg_email_manager_tmp varchar(100);
declare variable reg_email_operator_tmp varchar(100);
declare variable jk_system_code_tmp varchar(100);
begin
  for select reg_name,
             (select cont_value
              from hp_contdata_get_email_valid(0, c_id)), c_name_fio, reg.reg_email_manager, reg.reg_email_operator,
             jk_system_code, jl_name
      from hr_pstatus, clients, joblist, jobkind, region reg
      where ps_from_c_id = c_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id > 1000000000004 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Specialist_KOBD') and
            ps_from_reg_id = reg_unicode
      into reg_name, result_email, fio, reg_email_manager_tmp, reg_email_operator_tmp, jk_system_code_tmp, jl_name
  do
  begin
    result_email = coalesce(result_email, '');

    if (jk_system_code_tmp = 'JobSystem.Region_KM') then
    begin
      if (result_email = '') then
        result_email = reg_email_manager_tmp;
    end
    else
    begin
      if (result_email = '') then
        result_email = reg_email_operator_tmp;
    end

    if (result_email <> '') then
      suspend;
  end
end
