CREATE PROCEDURE HP_CESSION_WRITEOFF(
    K_ID ID_TYPE,
    REP_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    NEW_OL_ID D_ID,
    KC_ID D_ID,
    KC_STATUS D_SMALLINT)
AS
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable sum_os d_currency;
declare variable sum_perc d_currency;
declare variable ol_memo d_str_255;
begin
    new_ol_id = null;
    kc_id = null;
    kc_status = null;
    sum_os = 0;
    sum_perc = 0;
    ol_memo = 'Списание задолженности с внебалансового учета по договору Цессии';

    select k_last_reg_id
    from kontract
    where k_id = :k_id
    into :reg_id;

    -- Получаем остатки
    for select oko_from_c_id,
               sum(oko_os + oko_sh_os),
               sum(oko_perc + oko_sh_perc)
    from operkroff
    where oko_from_k_id = :k_id
      and oko_ol_is_active = 1
    group by oko_from_c_id
    having sum(oko_os + oko_sh_os) > 0
        or sum(oko_perc + oko_sh_perc) > 0
    into :c_id, :sum_os, :sum_perc
    do 
    begin
        select kv
        from get_kv(:reg_id, 5)
        into :new_ol_id;

        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete, ol_user_id, ol_document_client, ol_date_oper_day)
        values (:new_ol_id, :reg_id, 1, 8, :rep_date, 9, :ol_memo, 0, 0, :un_id, :c_id, :rep_date);
    
        insert into operkroff (oko_from_ol_id, oko_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc, oko_other, oko_ol_is_active, oko_ol_date)
        values (:new_ol_id, 1, :k_id, :c_id, 4, -:sum_os, -:sum_perc, 0, 1, :rep_date);

        select distinct kc_id,
                        kc_status
        from knclient
        where kc_from_k_id = :k_id
          and kc_from_c_id = :c_id
        into :kc_id,
             :kc_status;

        execute procedure hp_do_kn_operbook_save(:new_ol_id, :new_ol_id, :kc_id, 'operkroffoff', :reg_id, :rep_date, 1);
        execute procedure hp_kn_oksummary_recalc(:kc_id, :rep_date);

        suspend;
    end
end
