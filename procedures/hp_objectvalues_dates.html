CREATE PROCEDURE HP_OBJECTVALUES_DATES(
    OPR_ID DECIMAL(18,0))
AS
declare variable d_prior date;
declare variable ovs_from_opr_id decimal(18,0);
declare variable ovs_id integer;
declare variable ovs_date_begin date;
declare variable ovs_date_end date;
declare variable ovs_is_active integer;
declare variable actual_ovs_id integer;
declare variable ovs_add_value decimal(18,4) = 0;
declare variable fact_ovs_id integer;
declare variable ovs_value varchar(255);
begin
  d_prior = '31.12.9999';

  actual_ovs_id = null; -- ID последней актуальной записи
  fact_ovs_id = null; -- ID последней подтвержденной даты

  for select ovs_from_opr_id, ovs_id, ovs_date_begin, ovs_date_end, ovs_is_active, ovs_add_value, ovs_value
      from objectvalues
      where ovs_from_opr_id = :opr_id
      order by ovs_from_opr_id desc, ovs_date_begin desc, ovs_datetime desc, ovs_id desc
      into ovs_from_opr_id, ovs_id, ovs_date_begin, ovs_date_end, ovs_is_active, ovs_add_value, ovs_value
  do
  begin
    if (ovs_is_active = 1) then
    begin
      if (actual_ovs_id is null) then
        actual_ovs_id = ovs_id;
      if (fact_ovs_id is null and
          ovs_add_value = 2) then
        fact_ovs_id = ovs_id;

      if (ovs_date_end <> d_prior) then
      begin
        update objectvalues
        set ovs_date_end = :d_prior
        where ovs_from_opr_id = :ovs_from_opr_id and
              ovs_id = :ovs_id;
      end
      d_prior = ovs_date_begin - 1;
    end
    else
    begin
      if (ovs_date_end <> (ovs_date_begin - 1)) then
      begin
        update objectvalues
        set ovs_date_end = (:ovs_date_begin - 1)
        where ovs_from_opr_id = :ovs_from_opr_id and
              ovs_id = :ovs_id;
      end
    end
  end
  update objectprops
  set opr_actual_ovs_id = :actual_ovs_id,
      opr_fact_ovs_id = :fact_ovs_id
  where opr_id = :opr_id and
        (coalesce(opr_actual_ovs_id, -777) <> coalesce(:actual_ovs_id, -777) or
        coalesce(opr_fact_ovs_id, -777) <> coalesce(:fact_ovs_id, -777));
end
