Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXCLASS_CONTROLS
RETURNS (
    CONTROL_ID D_STR_CODE,
    CONTROL_NAME D_STR_CODE,
    CONTROL_MEMO D_STR_255,
    CONTROL_IMAGE D_INTEGER)
AS
begin
  control_id = '';
  control_name = '-';
  control_memo = 'Редактор не установлен';
  control_image = -1;
  suspend;

  for select editor_id, editor_name, editor_memo, image_index
  from uxp$control_kind
  into :control_id, :control_name, :control_memo, control_image
  do
  begin
    if (control_id in ('cell', 'email', 'dialogedit')) then
      Continue;

    if (control_id starting 'current$') then
      Continue;

    if (control_id starting 'blob') then
      Continue;

    suspend;
  end
end
