CREATE PROCEDURE HP_BIOGUARD_SAVE_TEMP_DATA(
    SESSION_ID D_STR_100 NOT NULL,
    PASSPORT_PHOTO1 D_BLOB_BINARY NOT NULL,
    PASSPORT_PHOTO2 D_BLOB_BINARY NOT NULL,
    DOCRECOG_JSON D_BLOB_TEXT NOT NULL)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255)
AS
declare variable DOCUMENT_TYPE D_STR_CODE; /* Тип распознанного документа */
declare variable DOC_NUMBER_VALUE D_STR_CODE; /* Распознанное значения для поля "Номер документа" */
declare variable TIN_VALUE D_STR_20; /* Распознанное значения для поля "ИИН" */
begin
    /*
        RFCKZ-267
        Была добавлена проверка, что клиент при регистрации загрузил фото лицевой и обратной стороны паспорта. Если не делать эту проверку,
        клиент может прикрепить две лицевые или две обратные стороны паспорта и успешно зарегистрироваться, что некорректно.

        Проверка осуществляется следующим образом:
        С КМП в метод передается JSON с распозанными значенями полей, в данном методе проверяется, что в переданном JSON есть поля: номер документа и ИИН
        Поле ИИН находится только на лицевой строне удостоверения личности, а номер документа только на обратной. Таким образом мы можем проверить, были ли
        прикреплены обе стороны документа, либо только одна
    */

    res_code = 'success';
    res_text = '';

    document_type = f_json_blob_get_value(docrecog_json, 'result.document_type', '');
    if (coalesce(document_type, '') not in ('kaz.id.type1', 'kaz.id.type2')) then
    begin
        res_code = 'error';
        res_text = 'Прикреплен неизвестный документ. Пожалуйста, прикпрепите удостоверение личности';
        suspend;
        exit;
    end

    select
        max(iif(field_code = 'tin', field_value, null)),
        max(iif(field_code = 'number', field_value, null))
    from hp_parse_docrecog_json(:docrecog_json)
    where field_code in ('tin', 'number')
    into
        :tin_value,
        :doc_number_value;

    if (res_code = 'success' and (coalesce(:tin_value, '') = '' or char_length(:tin_value) <> 12)) then
    begin
        res_code = 'error';
        res_text = 'Не удалось обнаружить ИИН. Пожалуйста проверьте, что вы прикрепляли лицевую сторону документа';
    end

    if (res_code = 'success' and (coalesce(:doc_number_value, '') = '' or char_length(:doc_number_value) <> 9)) then
    begin
        res_code = 'error';
        res_text = 'Не удалось обнаружить номер документа. Пожалуйста проверьте, что вы прикрепляли обратную сторону документа';
    end

    if (:res_code = 'success') then
        insert into bioguard_temp_data (btd_session_id, btd_create_datetime, btd_passport_photo1, btd_passport_photo2, btd_docrecog_json)
        values (:session_id, current_timestamp, :passport_photo1, :passport_photo2, :docrecog_json);

    suspend;
end
