CREATE PROCEDURE HP_E_HR_ORDERS_ONE_CANCEL(
    IN_HO_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable ho_date d_date;
declare variable ho_status d_integer;
declare variable is_closed d_bool;
begin
  res_is_error = 1;
  res_text = '';

  select ho_date, ho_status
  from hr_orders
  where ho_id = :in_ho_id
  into ho_date, ho_status;

  is_closed = 0;
  select is_closed
  from hp_sa_tabel_month_status(:ho_date)
  into is_closed;

  if (is_closed = 1) then
    res_text = 'Месяц закрыт для кадровых приказов.';
  else
  if (ho_status = -1) then
    res_text = 'Приказ уже аннулирован';
  else
  begin
    update hr_orders
    set ho_status = -1,
        ho_status_user_id = :un_id
    where ho_id = :in_ho_id;

    res_is_error = 0;
  end

  suspend;
end
