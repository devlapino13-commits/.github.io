CREATE PROCEDURE AAREP_DISB_LOANS_WITHOUT_ANK_TM(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    KONTRACT_NO VARCHAR(20),
    KF_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    CL_FIO VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    MAC_LOAN_TARGET VARCHAR(255),
    KC_LOAN_TARGET VARCHAR(255),
    LOAN_USE_FACT D_STR_255,
    MAC_BUSINESS_SUBTYPE VARCHAR(50),
    KC_BUSUNESS_SUBTYPE VARCHAR(50),
    DESCRIPTION_OF_LOAN STR_255,
    MONITORING_MEMO STR_255,
    K_AMOUNT_APPROVED D_CURRENCY)
AS
declare variable reg_id d_integer;
declare variable kf_id d_id;
declare variable k_id d_id;
declare variable ma_id d_integer;
declare variable c_id d_id;
declare variable kc_id d_id;
begin
  for select id_list,
             reg.reg_name
      from region_list(:rep_region_id, 1, -1, 1)
      inner join region reg on reg.reg_unicode = id_list
      into :reg_id,
           :reg_name
  do
  begin
    for select kn.k_kontract_no,
               kn.k_disb_date,
               kn.k_from_kf_id,
               kn.k_id,
               kn.k_from_ma_app_id,
               kn.k_amount_approved
        from kontstate ks
        inner join kontract kn on kn.k_id = ks.ks_from_k_id
        inner join product pr on kn.k_from_pr_id = pr.pr_id
        where ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and
              ks.ks_date_begin = :rep_date_end - 30 and
              ks.ks_region_id = :reg_id
        into :kontract_no,
             :k_disb_date,
             :kf_id,
             :k_id,
             :ma_id,
             :k_amount_approved
    do
    begin
      kf_name = null;
      c_name_fio = null;

      select kal.kal_new_kf_name,
             cl.c_name_fio
      from knaktlist kal
      inner join clients cl on kal.kal_new_kf_officer = cl.c_id
      where kal.kal_from_kf_id = :kf_id and
            kal.is_delete = 0 and
            :rep_date_end between kal.kal_date and kal.kal_date_end
      into :kf_name,
           :c_name_fio;

      for select rd.rd_name,
                 bst.bs_name,
                 cl.c_id,
                 kc.kc_id
          from knclient kc
          inner join clients cl on kc.kc_from_c_id = cl.c_id
          inner join bussubtype bst on kc.kc_busuness_subtype = bst.bs_id
          inner join refdata rd on rd.rd_id = kc.kc_loan_target
          where kc.kc_from_k_id = :k_id and
                kc.kc_amount_approved > 0
          into :kc_loan_target,
               :kc_busuness_subtype,
               :c_id,
               :kc_id
      do
      begin
        mac_loan_target = '';
        mac_business_subtype = '';
        cl_fio = '';

        select rd.rd_name,
               bst.bs_name,
               cl.c_name_fio
        from miniappclient mac
        inner join miniapp ma on mac.mac_from_ma_id = ma.ma_id
        inner join clients cl on cl.c_id = mac.mac_from_c_id
        inner join bussubtype bst on mac.mac_business_subtype = bst.bs_id
        inner join refdata rd on rd.rd_id = mac.mac_loan_target
        where ma.ma_is_active = 1 and
              ma.ma_id = :ma_id and
              ma.ma_date_end = '31.12.9999' and
              mac.mac_from_c_id = :c_id
        into :mac_loan_target,
             :mac_business_subtype,
             :cl_fio;

        loan_use_fact = '';
        monitoring_memo = '';
        description_of_loan = '';

        select ptm.fact_loan_target, ptm.monitoring_memo, ptm.descr_of_loan
        from print_target_monitoring(:kc_id) ptm
        into :loan_use_fact, :monitoring_memo, :description_of_loan;

        begin
        if (loan_use_fact is null or loan_use_fact = '') then
          loan_use_fact = 'Не заполнено';

        if (description_of_loan is null or description_of_loan = '') then
          description_of_loan = 'Не заполнено';

        if (monitoring_memo is null or monitoring_memo = '') then
          monitoring_memo = 'Не заполнен';
        end
        suspend;
      end
    end
  end
end
