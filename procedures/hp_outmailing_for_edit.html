CREATE PROCEDURE HP_OUTMAILING_FOR_EDIT(
    IN_OML_ID D_INTEGER,
    IN_OML_PARENT_ID D_INTEGER,
    IN_IS_OUT D_BOOL,
    IN_GROUP_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    OML_ID D_INTEGER,
    OML_PARENT_ID D_INTEGER,
    OML_IS_OUT D_BOOL,
    OML_DATE D_DATE,
    OML_GROUP_KIND D_INTEGER,
    OML_NUMBER D_INTEGER,
    OML_REGIST_NO D_STR_CODE,
    OML_FROM_REG_ID D_INTEGER,
    OML_APPLIANT_KIND D_INTEGER,
    OML_APPLICANT_NAME D_STR_100,
    OML_FROM_C_ID D_ID,
    OML_FROM_OCT_ID D_INTEGER,
    OML_FROM_OSK_ID D_INTEGER,
    OML_KONTRACTS D_STR_255,
    OML_SUBJECT D_STR_LONG,
    OML_MEMO D_STR_8100,
    OML_IS_REGULATORY D_BOOL,
    OML_IS_DELINQ D_BOOL,
    OML_ORIGINAL_STATE D_INTEGER,
    OML_RESULT_STATE D_INTEGER,
    OML_DEADLINE D_DATE,
    OML_FROM_UN_ID D_ID,
    OML_DATETIME D_DATETIME,
    RES_CAN_EDIT_LOAN D_BOOL)
AS
declare variable loan_count d_integer;
begin
  res_can_edit_loan = 1;

  if (coalesce(in_group_kind, '') = '') then
    in_group_kind = 'office';

  select oml_id,
         oml_parent_id,
         oml_is_out,
         oml_date,
         oml_group_kind,
         oml_number,
         oml_regist_no,
         oml_from_reg_id,
         oml_appliant_kind,
         oml_applicant_name,
         oml_from_c_id,
         oml_from_oct_id,
         oml_from_osk_id,
         oml_kontracts,
         oml_subject,
         oml_memo,
         oml_is_regulatory,
         oml_is_delinq,
         oml_original_state,
         oml_result_state,
         oml_deadline,
         oml_from_un_id,
         oml_datetime
  from outmailing
  where oml_id = :in_oml_id
  into :oml_id,
       :oml_parent_id,
       :oml_is_out,
       :oml_date,
       :oml_group_kind,
       :oml_number,
       :oml_regist_no,
       :oml_from_reg_id,
       :oml_appliant_kind,
       :oml_applicant_name,
       :oml_from_c_id,
       :oml_from_oct_id,
       :oml_from_osk_id,
       :oml_kontracts,
       :oml_subject,
       :oml_memo,
       :oml_is_regulatory,
       :oml_is_delinq,
       :oml_original_state,
       :oml_result_state,
       :oml_deadline,
       :oml_from_un_id,
       :oml_datetime;

  if (row_count = 0) then
  begin
     oml_parent_id = in_oml_parent_id;
     oml_is_out = in_is_out;
     oml_date = current_date;
     oml_appliant_kind = 0;
     oml_group_kind = f_enum('outmailing_group_kind', in_group_kind, 'id');
     oml_from_oct_id = 0;
     oml_from_osk_id = 0;
     oml_kontracts = '';
     oml_subject = '';
     oml_memo = '';
     oml_is_regulatory = 0;
     oml_is_delinq = 0;
     oml_original_state = 0;
     oml_result_state = 0;

     select un_curr_reg_id
     from username
     where un_id = :un_id
     into :oml_from_reg_id;

     if (in_oml_parent_id is not null) then
     begin
       select oml_from_reg_id, oml_appliant_kind, oml_group_kind,
              oml_applicant_name, oml_from_c_id, oml_from_oct_id, oml_kontracts, oml_subject,
              oml_is_regulatory, oml_is_delinq
       from outmailing
       where oml_id = :in_oml_parent_id
       into oml_from_reg_id, :oml_appliant_kind, :oml_group_kind,
            :oml_applicant_name, :oml_from_c_id, :oml_from_oct_id, :oml_kontracts, :oml_subject,
            :oml_is_regulatory, :oml_is_delinq;

        select count(*)
        from outmailkont
        where omk_from_oml_id = :in_oml_parent_id
        into loan_count;

        if (loan_count > 1) then
          res_can_edit_loan = 0;
     end
  end else
  begin
    select count(*)
    from outmailkont
    where omk_from_oml_id = :oml_id
    into loan_count;

    if (loan_count > 1) then
      res_can_edit_loan = 0;
  end

  suspend;
end
