CREATE PROCEDURE HP_DOCPRLOAN_DPL_LIST(
    DP_ID D_INTEGER,
    CLIENT_KIND D_SMALLINT,
    ONLY_LINKED D_BOOL,
    ONLY_ACTIVE D_BOOL = null,
    UN_ID D_ID = null)
RETURNS (
    PR_REC_ID D_INTEGER,
    PR_ID D_INTEGER,
    PR_NAME D_STR_100,
    PR_MEMO D_STR_255,
    PR_IS_DELETE D_BOOL,
    PR_DATE_BEGIN D_DATE,
    PR_DATE_END D_DATE,
    DPL_SET_ID D_INTEGER,
    DPL_GROUP_TEXT D_STR_100,
    DPL_CHECKED D_INTEGER,
    DPL_DAYS D_INTEGER)
AS
begin
  if (dp_id is null) then
    dp_id = 0;

  if (only_active is null) then
    only_active = 0;

  for select pr_id, pr_name,
             pr_common_name ||
             iif(coalesce(pr_name, '') <> coalesce(pr_memo, '') and coalesce(pr_memo, '') <> '', ', ' || pr_memo, ''),
             iif(pr_date_end >= current_date, 0, 1),
             pr_date_begin, pr_date_end
  from product
  order by 4, pr_date_begin desc, pr_name
  into pr_id, pr_name,
       pr_memo, pr_is_delete,
       pr_date_begin, pr_date_end
  do
  begin
    if (only_active = 1 and pr_is_delete = 1) then
      Continue;

    if (pr_date_end >= '31.12.2999') then
      pr_date_end = null;

    for select 0
    from rdb$database
    union
    select distinct dpl_set_id
    from docprloan
    where dpl_from_dp_id = :dp_id and
          dpl_from_pr_id = :pr_id and
          dpl_is_delete = 0 and
          dpl_set_id = 0
    into dpl_set_id
    do
    begin
      if (dpl_set_id = 0)
      then dpl_group_text = 'Документы';
      else dpl_group_text = 'Документы ' || dpl_set_id;

      pr_rec_id = dp_id * 100 + dpl_set_id;

      dpl_checked = -1;

      select dpl_date_kind, dpl_date_days
      from docprloan
      where dpl_from_dp_id = :dp_id and
            dpl_from_pr_id = :pr_id and
            dpl_set_id = :dpl_set_id and
            dpl_client_kind = :client_kind and
            dpl_is_delete = 0
      into dpl_checked, dpl_days;

      if (dpl_days = 0) then
        dpl_days = null;

      if (only_linked = 0 or
          dpl_checked <> -1) then
        suspend;
    end
  end
end
