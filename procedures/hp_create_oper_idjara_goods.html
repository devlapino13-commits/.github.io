CREATE PROCEDURE HP_CREATE_OPER_IDJARA_GOODS(
    OBJ_ID D_ID)
AS
declare variable un_id d_id;
declare variable ob_currency varchar(3);
declare variable k_id d_id;
declare variable ag_id d_id;
declare variable region_id d_integer;
declare variable matotvet_id d_id;
declare variable workplace_id d_id;
declare variable ag_agreement_no d_str_100;
declare variable ag_agreement_date d_date;
declare variable client_id d_id;
declare variable curr_region_id d_integer;
declare variable ol_id d_id;
declare variable account_kt d_str_20;
declare variable account_dt d_str_20;
declare variable ol_doc_kind d_integer;
declare variable ol_memo d_str_long;
declare variable kml_id d_id;
declare variable ml_id d_id;
declare variable kml_cost decimal(18,2);
declare variable kml_count d_integer;
declare variable kml_total decimal(18,2);
declare variable ml_name d_str_255;
declare variable kml_bill_id d_id;
declare variable kml_bill_line_no d_integer;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable line_no d_integer;
declare variable matcateg_id d_id;
declare variable memo_detail d_str_long;
begin
  un_id = 1000000000000;
  matcateg_id = 133;
  select cm_default_currency
  from company
  into :ob_currency;

  select kma.kma_from_k_id, kma.kma_from_ag_id
  from kontmatag kma
  where kma.kma_id = :obj_id
  into k_id, ag_id;

  select kal.kal_new_reg_id, k.k_approved_manager
  from kontract k
  inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id and
        kal.is_delete = 0 and
        kal.kal_date_end >= current_date
  where k.k_id = :k_id
  into :region_id, :matotvet_id;

  select wp_id
  from workplace
  where wp_from_reg_id = :region_id and
        wp_status = 6 and
        wp_is_active = 1 and
        (wp_name like '%Собственность%' or
        wp_name containing 'Истиснаа')
  rows 1
  into :workplace_id;

  select ag.ag_agreement_no, ag.ag_agreement_date, ag.ag_object_id
  from agreements ag
  where ag.ag_id = :ag_id
  into ag_agreement_no, ag_agreement_date, client_id;

  curr_region_id = f_hq_id();

  select kv
  from get_kv(:curr_region_id, 5)
  into :ol_id;

  account_kt = (select al_id
                from acc_list
                where al_from_reg_id = :region_id and
                      al_currency = '417' and
                      al_account = '21199/99');

  account_dt = (select al_id
              from acc_list
              where al_from_reg_id = :region_id and
                    al_currency = '417' and
                    al_account = '11207/3');

  ol_doc_kind = 9; -- Мем.ордер

  if (ag_agreement_no <> '' and
      ag_agreement_date is not null) then
    ol_memo = 'Операции по Иджара согласно договора № ' || ag_agreement_no || ' от ' || coalesce(ag_agreement_date, current_date);
  else
    ol_memo = 'Покупка товаров по ИПФ';

  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_user_id,
                        ol_document_client, ol_oper_subtype, ol_is_client)
  values (:ol_id, :region_id, 3, current_date, :ol_doc_kind, :ol_memo, :un_id, :client_id, 9, 1)
  returning ol_id
  into :ol_id;

  for select kml_id, kml_from_ml_id, kml_cost, kml_count, kml_total,
             ml_name || iif(coalesce(ml_model, '') <> '', ' (' || ml_model || ')', ''), kml_bill_id, kml_bill_line_no,
             k_kontract_no, k_disb_date
      from kontmat, matlist, kontract
      where kml_object_kind = 9 and
            kml_object_id = :k_id and
            kml_from_ag_id = :ag_id and
            kml_from_ml_id = ml_id and
            kml_object_id = k_id
      into :kml_id, :ml_id, :kml_cost, :kml_count, :kml_total, ml_name, :kml_bill_id, :kml_bill_line_no, :k_kontract_no,
           :k_disb_date
  do
  begin
    insert into opermat (om_from_ol_id, om_oper_type, om_workplace_id, om_matotvet_id, om_summa_plus, om_summa_minus,
                         om_kolich_plus, om_kolich_minus, om_nds_percent, om_account, om_is_written_off,
                         om_account_info)
    values (:ol_id, 1, :workplace_id, :matotvet_id, :kml_total, 0, :kml_count, 0, 0, :account_dt, 0, :account_kt)
    returning (om_number)
    into :line_no;

    update kontmat
    set kml_bill_id = :ol_id,
        kml_bill_line_no = :line_no
    where kml_id = :kml_id;

    memo_detail = 'Операции по Иджара: ' || :ml_name || ', дог. фин-ия ' || :k_kontract_no || ' от ' || :k_disb_date || ' в рамках дог. №' || :ag_agreement_no || ' ' || :ag_agreement_date;

    insert into operbill (ob_bill_id, ob_bill_line_no, ob_from_ml_id, ob_from_reg_id, ob_from_mc_id, ob_date_begin,
                          ob_from_ol_id, ob_invent_no, ob_invent_str, ob_full_detail)
    values (:ol_id, :line_no, :ml_id, :region_id, :matcateg_id, current_date, :ol_id, 0, '', '');

    insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa,
                          ob_account_kt, ob_client_kt, ob_detail)
    values (:ol_id, :ob_currency, 1, null, :account_dt, :kml_total, :kml_total, :account_kt, :client_id,
            :memo_detail);

  end
end
