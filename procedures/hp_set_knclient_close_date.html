CREATE PROCEDURE HP_SET_KNCLIENT_CLOSE_DATE(
    K_ID DECIMAL(18,0),
    DISB_DATE DATE)
AS
declare variable closed_date date;
declare variable c_id decimal(18,0);
declare variable kc_id decimal(18,0);
declare variable kc_amount_approved decimal(18,4) = 0;
begin
-- Процедура устанавливает дату закрытия предыдущего кредита в новом кредите
  for select kc_id, kc_from_c_id, kc_amount_approved
      from knclient
      where kc_from_k_id = :k_id
      into :kc_id, :c_id, kc_amount_approved
  do
  begin
    closed_date = null;
    if (kc_amount_approved > 0) then
    begin
      select ks_date_end
      from kontstate, knclient
      where ks_from_k_id = kc_from_k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_date_begin <= :disb_date and
            kc_id <> :kc_id
      order by ks_date_end desc
      rows 1
      into :closed_date;
      if (closed_date < '31.12.2999') then
        closed_date = closed_date + 1;
      if (coalesce(closed_date, '31.12.2999') > disb_date) then
        closed_date = null;
    end
    update knclient
    set kc_when_prior_loan_closed = :closed_date
    where kc_id = :kc_id;
  end
end
