CREATE PROCEDURE AAREP_EPAY_BANKCARD_DISB_ZUP(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    NUM D_INTEGER,
    NAME STR_255,
    SURNAME STR_255,
    MIDLENAME STR_255,
    BIRTHDAY D_DATE,
    IIN STR_100,
    CITIZEN STR_100,
    RESIDENT D_SMALLINT,
    POST_INDEX STR_20,
    REGION_ STR_255,
    COUNTRY STR_255,
    OBL STR_255,
    DISTRICT STR_255,
    CITY STR_255,
    STREET STR_255,
    DEPART_NO STR_255,
    BUILDING_NO STR_255,
    FLAT_NO STR_255,
    SECTION_NO STR_255,
    PASS_TYPE STR_100,
    PASS_SER STR_10,
    PASS_COUNTRY STR_100,
    PASS_NO D_STR_100,
    PASS_WHO STR_255,
    PASS_DATE D_DATE,
    PROFIT_DATE D_DATE,
    PROFIT_SUM D_CURRENCY,
    SEX D_STR_CODE)
AS
declare variable c_id id_type;
begin
  -- для ЗУП
  num = 0;
  profit_date = rep_date_end;

  for select ag_object_id, sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, :rep_date_end))
  from operagr, agreements
  where ogr_ol_date between :rep_date_begin and :rep_date_end and
        ogr_ol_is_active = 1 and
        ogr_from_ag_id = ag_id and
        ag_object_kind = f_objkind('clients', 'id')
  group by ag_object_id
  into :c_id, profit_sum
  do
  begin
    for select name, surname, midlename, birthday, iin, citizen, resident, post_index, region_, country, obl, district,
               city, street, depart_no, building_no, flat_no, section_no, pass_type, pass_ser, pass_country, pass_no,
               pass_who, pass_date, sex
        from aarep_epay_client_info(:c_id, :rep_date_end)
        into :name, :surname, :midlename, :birthday, :iin, :citizen, :resident, :post_index, :region_, :country, :obl,
             :district, :city, :street, :depart_no, :building_no, :flat_no, :section_no, :pass_type, :pass_ser,
             :pass_country, :pass_no, :pass_who, :pass_date, :sex
    do
    begin
      num = num + 1;
      suspend;
    end
  end
end
