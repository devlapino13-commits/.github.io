CREATE PROCEDURE OKT_PHONECALL_SAVE(
    PNC_ID D_ID NOT NULL,
    FILE_PATH D_BLOB_TEXT NOT NULL,
    DURATION D_INTEGER NOT NULL,
    CALL_START D_DATETIME NOT NULL,
    CALL_FINISH D_DATETIME NOT NULL)
RETURNS (
    RES D_BOOL)
AS
declare variable VALUE_ D_STR_2048;
declare variable COMMA_INDEX D_INTEGER;
begin
    res = 0;

    insert into okt_phonecall_save_log (pnc_id, file_path, duration, call_start, call_finish)
    values (:pnc_id, :file_path, :duration, :call_start, :call_finish);

    while (char_length(trim(file_path)) > 0) do
    begin
        comma_index = 0;
        comma_index = position(',', file_path);
        if (comma_index = 0) then
            comma_index = char_length(file_path) + 1;

        value_ = trim(substring(file_path from 1 for comma_index - 1));
        if (value_ is null or char_length(value_) = 0) then
            exception e_default 'Путь к файлу не может быть пустым';

        -- обрезаю уже использованные файлы и оставляю только с которыми ещё предстоит работа
        file_path = trim(substring(file_path from comma_index + 1));

        insert into okt_dialog_path (odp_dialog_path, odp_from_pnc_id)
        values (:value_, :pnc_id);
        res = 1;
    end

    if (res = 1) then -- если от ОКТЕЛЛ получен хоть один аудио-файл, формирую задачу на ее обработку для сервиса Комиля
    begin
       insert into okt_unupdated_file (pnc_id)
       values (:pnc_id);
    end

    update PHONECALLS pc
    set pc.pnc_call_status_id = f_enum_id('phone_call_result', 'answered'),
     pc.pnc_start = :call_start,
     pc.pnc_finish = :call_finish,
     pc.pnc_duration = :duration
    where
    pc.pnc_id = :pnc_id;
    
    suspend;
end
