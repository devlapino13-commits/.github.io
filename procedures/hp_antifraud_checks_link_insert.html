CREATE PROCEDURE HP_ANTIFRAUD_CHECKS_LINK_INSERT(
    C_ID D_ID NOT NULL,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    CHECK_TYPE D_SMALLINT NOT NULL,
    CHECK_STAGE D_SMALLINT NOT NULL,
    CHECK_RESULT D_INTEGER NOT NULL,
    AC_ID D_ID = null)
RETURNS (
    AFCL_ID D_ID)
AS
begin
    -- Чтобы не засорять таблицу при проверках на некоторых этапах проверяем, сущесвтует ли уже такая запись
    if (f_enum_code('antifraud_check_stage', :check_stage) in ('loan_check_preview', 'freeze_send_doc') and
        exists(select *
               from antifraud_checks_link afcl
               where afcl.afcl_from_c_id = :c_id and
                     afcl.afcl_object_kind = :object_kind and
                     afcl.afcl_object_id = :object_id and
                     afcl.afcl_check_stage = :check_stage and
                     afcl.afcl_check_type = :check_type and
                     afcl.afcl_check_result = :check_result)
    ) then
        exit;

    insert into antifraud_checks_link (afcl_from_c_id, afcl_object_kind, afcl_object_id, afcl_check_type,
                                       afcl_check_stage, afcl_from_ac_id, afcl_check_result, afcl_datetime)
    values (:c_id, :object_kind, :object_id, :check_type, :check_stage, :ac_id,
            :check_result, current_timestamp)
    returning afcl_id
    into :afcl_id;

    suspend;
end
