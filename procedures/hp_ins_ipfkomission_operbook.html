CREATE PROCEDURE HP_INS_IPFKOMISSION_OPERBOOK(
    K_ID D_ID)
AS
declare variable ol_doc_kind d_integer;
declare variable k_status d_integer;
declare variable k_kontract_no d_str_100;
declare variable pr_id d_id;
declare variable pr_ipf_commission d_currency;
declare variable ol_doc_no d_integer;
declare variable ol_doc_no_str d_str_100;
declare variable ol_memo d_str_255;
declare variable res_acc_kt d_str_100;
declare variable res_acc_dt d_str_100;
declare variable un_id d_id;
declare variable c_id d_id;
declare variable res_currency d_str_100;
declare variable c_name_fio d_str_255;
declare variable io_from_ol_id d_id;
declare variable ol_id d_id;
declare variable reg_id d_id;
declare variable k_disb_date d_date;
declare variable reg_code d_str_10;
declare variable k_approved_manager d_id;
begin
  ol_doc_kind = 9; -- мемориальный ордер
  for select k.k_status, k.k_last_reg_id, k.k_kontract_no, k.k_disb_date, pr.pr_ipf_commission, pr.pr_id, k.k_approved_manager
      from kontract k
      inner join product pr on pr.pr_id = k.k_from_pr_id and
            pr.pr_code containing '.ipf.' and
            pr.pr_ipf_commission > 0
      where k.k_id = :k_id
      into k_status, reg_id, k_kontract_no, k_disb_date, pr_ipf_commission, pr_id, k_approved_manager
  do
  begin
    select un.un_id from username un
    where un.un_from_c_id = :k_approved_manager
    into un_id;

    select kc.kc_from_c_id
    from knclient kc
    where kc.kc_from_k_id = :k_id and
          kc.kc_amount_approved > 0
    into c_id;
    if (exists(select *
               from ipf_operkontract io
               where io.io_from_k_id = :k_id and
                     io.io_is_active = 1)) then
    begin
      if (not exists(select *
                     from operkontract ok
                     inner join operlist ol on ol.ol_id = ok.ok_from_ol_id and
                           ol.is_delete = 0
                     where ok.ok_from_k_id = :k_id) and
          k_status < 1) then
      begin
        select io.io_from_ol_id
        from ipf_operkontract io
        where io.io_from_k_id = :k_id and
              io.io_is_active = 1
        into io_from_ol_id;

        update operlist ol
        set ol.is_delete = 1
        where ol.ol_id = :io_from_ol_id;
      end
    end
    else
    begin
      if (k_status = 1) then
      begin
        res_currency = f_get_const('cm_default_currency');

        select c.c_name_fio
        from clients c
        where c.c_id = :c_id
        into c_name_fio;

        select kv
        from get_kv(:reg_id, 5)
        into :ol_id;

        select num
        from last_doc_number(:k_disb_date, :ol_doc_kind, :reg_id)
        into :ol_doc_no;

        ol_doc_no_str = :ol_doc_no;
        ol_memo = 'Единовременная денежная сумма по ИПФ к получению ' || k_kontract_no;

        insert into operlist (ol_id, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str, ol_oper_type, ol_oper_subtype,
                              ol_doc_kind, ol_memo, ol_user_id, ol_document_client, ol_doc_client_name, ol_is_client,
                              ol_parent_id, ol_manager_c_id, ol_from_rd_id)
        values (:ol_id, :reg_id, :k_disb_date, :ol_doc_no, :ol_doc_no_str, 2, 1, :ol_doc_kind, :ol_memo, :un_id, :c_id,
                null, iif(:c_id is not null, 1, 0), null, null, 0);

        insert into ipf_operkontract (io_id, io_from_ol_id, io_from_k_id, io_is_active)
        values (null, :ol_id, :k_id, 1);

        reg_code = (select reg_code
                    from region
                    where reg_unicode = :reg_id);
        res_acc_kt = (select al_id
                      from acc_list a
                      where a.al_from_reg_id = :reg_id and
                            a.al_account = '21141/1000' and
                            a.al_currency = :res_currency);

        select res_account
        from hp_make_kredit_acclist(:pr_id, :reg_id, 43)
        into :res_acc_dt;

        insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                              ob_summa, ob_detail, ob_client_kt, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
        values (:ol_id, :res_currency, 1, :res_acc_dt, :res_acc_kt, :pr_ipf_commission, :pr_ipf_commission, '', :c_id,
                :k_disb_date, 1, :reg_id);
      end
    end
  end
end
