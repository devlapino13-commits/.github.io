CREATE PROCEDURE HP_CARDSYNC_ACTION_EXEC_ONE(
    CSA_ID D_INTEGER NOT NULL)
AS
declare variable cs_id d_integer;
declare variable new_cs_id d_integer;
declare variable csc_code d_str_100;
declare variable cs_object_id d_id;
declare variable cs_object_kind d_integer;
declare variable cs_status d_integer;
declare variable cs_card_id d_integer;
declare variable csa_status d_smallint;
declare variable cs_answer_json d_str_8100;
declare variable card_acc_balance decimal(15,2);
declare variable csc_id d_integer;
declare variable amount d_currency;
declare variable cca_id d_integer;
declare variable un_id d_id;
declare variable ol_id d_id;
declare variable err_text d_str_255;
begin
  delete from cardsync_action_error cse
  where cse.cse_from_csa_id = :csa_id;

  select csa.csa_from_cs_id,
         csa.csa_status,
         f_get_cardsynccfg(cs_from_csc_id, 'code'),
         cs_object_id,
         cs_object_kind,
         cs_card_id,
         cs_status,
         cs.cs_answer_json,
         cs.cs_from_cca_id,
         cs.cs_from_un_id
  from cardsync_action csa
  inner join cardsync cs on cs.cs_id = csa.csa_from_cs_id
  where csa.csa_id = :csa_id
  into :cs_id,
       :csa_status,
       :csc_code,
       :cs_object_id,
       :cs_object_kind,
       :cs_card_id,
       :cs_status,
       :cs_answer_json,
       :cca_id,
       :un_id;

  if (csa_status = 1) then -- если задача ранее уже была обработана
    exit;

  if (cs_status = 2) then -- удачная синхронизация
  begin
      if (csc_code = 'registr_client') then
      begin
        execute procedure card_command_set_new_card(:cs_id);
      end else
      if (csc_code = 'set_new_card') then
      begin
        -- если картсчет подвязан
        update cardaktclient
        set cac_status = 2
        where cac_id = :cs_object_id
        and cac_status = 1;

        -- новая задача по активации карты
        execute procedure card_command_card_activation('card_activation_on_disb', :cs_object_kind, :cs_object_id, :cs_id);
      end else
      if (csc_code = 'card_activation_on_disb') then
      begin
        execute procedure card_command_set_card_name(:cs_id);
      end else
      if (csc_code = 'set_card_name') then
      begin
        -- если имя на карте отредактировано
        update cardaktclient
        set cac_status = 3
        where cac_id = :cs_object_id
        and cac_status = 2;
      end else
      if (csc_code in ('send_pin', 'resend_pin') and cs_object_kind = f_objkind('cardaktclient', 'id')) then
      begin
        -- если ПИН-код отправлен/переотправлен
        update cardaktclient set
          cac_status = 5
        where cac_id = :cs_object_id
          and cac_status = 4;
      end else
      if (csc_code = 'set_risk_group') then
      begin
        -- если успешно установлена риск-группа по факту выдачи займа,
        -- тогда должна быть создана операция по установке лимита по карте
        --if (cca_id <> 424372) then
          execute procedure card_command_card_disb(:cs_object_kind, :cs_object_id, :cs_id);
      end else
      if (csc_code = 'card_disb') then -- Если установка кредитного лимита по факту выдачи карты прошла успешно
      begin
        --execute procedure card_command_adv_perc_writeoff(:cs_object_kind, :cs_object_id, :cs_id);
        execute procedure card_adv_perc_repayment(:cs_object_kind, :cs_object_id, :cs_id);
      end else
      if (csc_code = 'adv_kommis_writeoff') then -- Если спсиание доп. % прошло успешно
      begin
        -- ...
      end
      else if (csc_code = 'payment_credit_limit_change') then
      begin
        update cardclientacc_limit cll
        set cll.ccl_ipc_is_ok = 1
        where cll.ccl_from_cs_id = :cs_id;

        -- если успешно установлен лимит по карте, по факту платежа по займу, тогда необходимо происзвести списание
        -- списание с карты суммы = сумма платежа минус уплаченный ОС
        execute procedure card_payment_writeoff(:cs_object_kind, :cs_object_id, :cs_id);
      end else
      if (csc_code = 'payment_write_off') then
      begin
        -- если спсиание с карты по факту погашения, прошло успешно,
        -- и если это было вызвано не автопогашением
        -- произвожу пополнение на сумму платежа
        select first 1
          cs_from_csc_id,
          cs_amount
        from cardsync
        where cs_card_id = :cs_card_id
          and cs_id < (
            select
              coalesce(cs_parent_id ,:cs_id)
            from cardsync
            where cs_id = :cs_id)
        order by cs_id desc
        into
          :csc_id,
          :amount;
        csc_id = coalesce(csc_id, -1);
        amount = coalesce(amount, 0);

        if (not (csc_id = f_get_cardsynccfg_id('check_balance_on_autorepayment') and amount > 0)) then
          execute procedure card_payment_adjunction(:cs_object_kind, :cs_object_id, :cs_id);
      end else
      if (csc_code = 'card_deactivate') then
      begin
        update cardlist
        set cal_status = f_enum('cardlist_status', 'deactivation', 'id')
        where cal_card_id = :cs_card_id;
      end else
      if (csc_code = 'card_activation') then
      begin
        update cardlist
        set cal_status = f_enum('cardlist_status', 'transfer_to_client', 'id')
        where cal_card_id = :cs_card_id;
      end else
      if (csc_code = 'card_deactivate') then
      begin
        update cardlist
        set cal_status = f_enum('cardlist_status', 'deactivation', 'id')
        where cal_card_id = :cs_card_id;
      end else
      if (csc_code = 'card_stoplist_in') then
      begin
        update cardlist
        set cal_status = f_enum('cardlist_status', 'stoplist_in', 'id')
        where cal_card_id = :cs_card_id;

        update cardblockopers cbo
        set cbo.cbo_status = 3
        where cbo.cbo_id = :cs_object_id;
      end else
      if (csc_code = 'card_hard_stoplist_in') then
      begin
        update cardlist
        set cal_status = f_enum('cardlist_status', 'hard_stoplist_in', 'id')
        where cal_card_id = :cs_card_id;

        update cardblockopers cbo
        set cbo.cbo_status = 3,
            cbo_active_for_robot = 0
        where cbo.cbo_id = :cs_object_id;
      end else
      if (csc_code = 'card_stoplist_out') then
      begin
        update cardlist
        set cal_status = f_enum('cardlist_status', 'transfer_to_client', 'id')
        where cal_card_id = :cs_card_id;

        update cardblockopers cbo
        set cbo.cbo_status = 3
        where cbo.cbo_id = :cs_object_id;
      end else
      if (csc_code in ('check_balance', 'check_balance_on_closing')) then
      begin
        card_acc_balance = cast(f_json_str_get_value(:cs_answer_json, 'data.availableAmount', -999.00) as decimal(15,2));

        update cardsync cs
        set cs.cs_amount = :card_acc_balance
        where cs.cs_id = :cs_id;

        if (f_objkind(:cs_object_kind, 'name') = 'pvnopers') then
        begin
            update pvnopers po
            set po.po_card_balance_amount = :card_acc_balance,
                po.po_status = f_enum('pvnopers_status', 'balance_check_ok', 'id')
            where po.po_id = :cs_object_id;
        end

        if (f_objkind(:cs_object_kind, 'name') = 'cardaktclient') then
        begin
          update cardaktclient set
            cac_amount = :card_acc_balance
          where cac_id = :cs_object_id;
          ol_id = (select ol_id from card_reissue_payment(:cs_object_id)); -- cписание комиссии за перевыпуск
        end
      end else
      if (csc_code = 'pvn_write_off') then
      begin
         update pvnopers po
         set po.po_status = f_enum('pvnopers_status', 'writeoff_ok', 'id')
         where po.po_id = :cs_object_id;

         if ((select po.po_oper_type
             from pvnopers po
             where po.po_id = :cs_object_id) = 0) then -- если это операция обналичивания
         begin
            execute procedure hp_pvnoper_cashing(:cs_object_id, :cs_id); -- формируем бух. проводки
         end
         else
         begin
            execute procedure hp_pvnopers_repayment(:cs_object_id, :cs_id); -- формируется операция погашения по займу
         end
      end else
      if (csc_code = 'write-off_on_closing') then
      begin
        execute procedure card_overpayment_transfer(:cs_id);
      end else
      if (csc_code = 'reset_card_pin_counter') then
        execute procedure hp_cardpinchange_resend_pin(-:cs_id, 'resend_pin', null)
          returning_values :cs_id, :err_text;
      if (csc_code = 'resend_pin' and cs_object_kind = f_objkind('cardpinchange', 'id')) then
      begin
        update cardpinchange set
          cph_status = f_enum_id('cardpinchange_status', 'pin_changed')
        where cph_id = :cs_object_id;

        update edoc_sign set
          eds_status = 1,
          eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'auto', 'id')
        where eds_from_edl_id = (select cph_res_edl_id from cardpinchange where cph_id = :cs_object_id);
      end else
      if (csc_code = 'check_balance_on_autorepayment') then
      begin
        execute procedure hp_autorepayment_cardloan(:cs_id);
      end else
      if (csc_code = 'payment_for_card_reissue') then
      begin
        ol_id = (select ol_id from card_reissue_payment(:cs_object_id, true)); -- бух.проводка после cписания комиссии за перевыпуск
      end
      if (csc_code in (
        'annul_repayment',
        'annul_repayment_perc',
        'annul_repayment_credit_limit_change')) then
      begin
        new_cs_id = (select cs_id from hp_cardoper_annul_run(:cs_object_id));
      end
  end
  ------------- в самом конце -----------
  update cardsync_action ca
  set ca.csa_status = 1,
      ca.csa_exec_datetime = current_timestamp,
      ca.csa_try_qty = ca.csa_try_qty + 1
  where ca.csa_id = :csa_id;
end
