CREATE PROCEDURE HP_GET_BLANK_FOR_PRINT(
    BLANK_ID INTEGER,
    BLANK_NUMBER INTEGER,
    BLANK_VERSION INTEGER,
    BLANK_HAS_BLOB INTEGER,
    BLANK_HAS_FRX_BLOB INTEGER)
RETURNS (
    BLB_VERSION INTEGER,
    BLB_FRX_STATUS INTEGER,
    BLB_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    BLB_FRX_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    BLB_FONT_STYLE VARCHAR(100),
    BLB_FONT_SIZE INTEGER)
AS
declare variable blb_has_blob integer;
declare variable blb_has_frx_blob integer;
begin
  blb_blob = null;
  blb_frx_blob = null;
  blb_font_style = '';
  blb_font_size = null;

  select iif(blb_blob is null, 0, 1), iif(blb_frx_blob is null, 0, 1), blb_frx_status, blb_version, blb_font_style,
         blb_font_size
  from blankblob
  where blb_from_bln_id = :blank_id and
        blb_number = :blank_number
  into blb_has_blob, blb_has_frx_blob, blb_frx_status, blb_version, blb_font_style, blb_font_size;
  if (blb_version is null) then
    exit;

  if (blank_has_blob = -1) then -- проверка только версии
  begin
    suspend;
    exit;
  end

  if (blank_version = blb_version and
      blank_has_blob = blb_has_blob and
      blank_has_frx_blob = blb_has_frx_blob) then
    exit;

  select blb_blob, blb_frx_blob
  from blankblob
  where blb_from_bln_id = :blank_id and
        blb_number = :blank_number
  into :blb_blob, :blb_frx_blob;

  suspend;
end
