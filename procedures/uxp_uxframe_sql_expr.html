CREATE PROCEDURE UXP$UXFRAME_SQL_EXPR(
    UX_UUID D_UUID,
    UF_UUID D_UUID,
    IS_PARAM_SQL D_BOOL,
    EXPR_CHECKING D_BOOL = 0)
RETURNS (
    SQL_TEXT D_STR_8100,
    RES_FIELD_LIST D_STR_8100)
AS
declare variable field_list d_str_8100;
declare variable ufd_source d_str_code;
declare variable ufd_is_table d_bool;
declare variable us_uuid d_uuid;
declare variable ufr_uuid d_uuid;
declare variable ux_kind d_str_code;
begin
  -- Приготовление Select sql для фрейма
  if (is_param_sql is null) then
    is_param_sql = 0;

  if (expr_checking is null) then
    expr_checking = 0;

  sql_text = '';
  field_list = '';
  res_field_list = '';

  if (uf_uuid = '') then
    uf_uuid = null;

  ux_kind = uxf$code(ux_uuid, 'kind');

  if (ux_kind = 'uxsource') then
  begin
    select us_uuid, us_code, us_is_table
    from ux$source
    where us_uuid = :ux_uuid
    into us_uuid, ufd_source, ufd_is_table;
  
    select list(uf_select_name, ', ')
    from ux$field
    where uf_owner = :us_uuid and
          uf_is_param = :is_param_sql and
          (:expr_checking = 0 or uf_type = 0)
    into res_field_list;

    if (res_field_list is null) then
      res_field_list = '';
  
    if (res_field_list <> '') then
    begin
      sql_text = 'select ' || res_field_list || ' from ';

      if (is_param_sql = 1) then
        sql_text = sql_text || 'rdb$database';
      else
      begin
        sql_text = sql_text || ufd_source;

        if (ufd_is_table = 0) then
        begin
          select list('null')
          from ux$field
          where uf_owner = :us_uuid and
                uf_is_param = 1
          into field_list;
          if (field_list <> '') then
            sql_text = sql_text || '(' || field_list || ')';
        end
      end
    end
  end else
  begin
    if (ux_kind <> 'uxframe') then
    begin
      select ufr_uuid
      from ux$frame
      where ufr_owner = :ux_uuid and
            ufr_is_mainframe = 1
      into ufr_uuid;
    end else
      ufr_uuid = ux_uuid;
  
    if (is_param_sql = 0) then
    begin
      select us_uuid, us_code, us_is_table
      from ux$frame, ux$source
      where ufr_uuid = :ufr_uuid and
            ufr_us_uuid = us_uuid
      into us_uuid, ufd_source, ufd_is_table;
    
      select list(uf_select_name, ', ')
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :ufr_uuid and
            ufp_uf_uuid = uf_uuid and
            uf_is_param = 0 and
            (:expr_checking = 0 or uf_type = 0) and
            (:uf_uuid is null or ufp_uf_uuid = :uf_uuid)
      into res_field_list;
    
      if (res_field_list <> '') then
      begin
        sql_text = 'select ' || res_field_list || ' from ' || ufd_source;
    
        if (ufd_is_table = 0) then
        begin
          select list('null')
          from ux$field
          where uf_owner = :us_uuid and
                uf_is_param = 1
          into field_list;
          if (field_list <> '') then
            sql_text = sql_text || '(' || field_list || ')';
        end
      end
    end else
    begin
      select sql_text, sql_params
      from uxp$uxframe_sql_param(:ufr_uuid, 0)
      into :sql_text, :res_field_list;
    end
  end

  suspend;
end
