CREATE PROCEDURE AAREP_CHECK_FINPOLIS_ONE(
    REP_C_ID DECIMAL(18,0))
RETURNS (
    FP_NUMBER VARCHAR(20),
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(20),
    C_INN VARCHAR(20),
    C_ADDRESS VARCHAR(255),
    C_FINPOLIS_SEND_DATE DATE,
    FOUND_PASS_NO INTEGER,
    FOUND_INN_NO INTEGER,
    FOUND_CLIENT_FIO INTEGER,
    FOUND_BIRTH_DATE INTEGER,
    FOUND_ADDRESS_LEGAL INTEGER,
    FOUND_ADDRESS_FACT INTEGER,
    FOUND_ADDRESS_BIRTH INTEGER,
    C_FINPOLIS_PROCEED INTEGER,
    IS_CRITICAL INTEGER)
AS
begin
  for select fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_address, c_finpolis_send_date,
             max(found_pass_no), max(found_inn_no), max(found_client_fio), max(found_birth_date),
             max(found_address_legal), max(found_address_fact), max(found_address_birth), max(c_finpolis_proceed)
      from aarep_check_finpolis_tmp(:rep_c_id)
      group by fp_number, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_address, c_finpolis_send_date
      into :fp_number, :c_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn, :c_address, :c_finpolis_send_date,
           :found_pass_no, :found_inn_no, :found_client_fio, :found_birth_date, :found_address_legal,
           :found_address_fact, :found_address_birth, :c_finpolis_proceed
  do
  begin
    is_critical = 0;
    if (c_finpolis_proceed is null) then
      c_finpolis_proceed = 0;
    is_critical = found_pass_no + found_inn_no + found_client_fio + found_birth_date + found_address_legal + found_address_fact;
    if (is_critical > 1) then
      is_critical = 1;
    else
      is_critical = 0;
    suspend;
  end
end
