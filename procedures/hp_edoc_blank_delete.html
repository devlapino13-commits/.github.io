CREATE PROCEDURE HP_EDOC_BLANK_DELETE(
    EDB_ID D_ID,
    UN_ID D_ID)
AS
declare variable REG_ID D_INTEGER;
declare variable EDL_ID D_ID;
declare variable EDC_CODE D_STR_CODE;
declare variable EDL_OBJECT_KIND D_INTEGER;
declare variable EDL_OBJECT_ID D_ID;
begin
    --RFCRU-2439 Процедура для удаления бланка из очереди
    select edl.edl_id,
           edbp.edbp_from_reg_id,
           edc.edc_code,
           edl.edl_object_kind,
           edl.edl_object_id
    from edoc_blank edb
    inner join edoc_blank_pending edbp on edbp.edbp_from_edb_id = edb.edb_id
    inner join edoc_list edl on edl.edl_id = edb_from_edl_id
    inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
    where edb.edb_id = :edb_id
    and edl.edl_status in (0, -1)
    into edl_id,
         reg_id,
         edc_code,
         edl_object_kind,
         edl_object_id;

    if (edl_id is null) then
        exception e_message 'Документ, котрый вы пытаетесь аннулировать, отсутсвует в очереди';

    if (edc_code <> 'cell.application') then
        exception e_message 'Документ, котрый вы пытаетесь аннулировать, не является "Заявлением на изменение номера"';

    insert into edoc_blank_pending_del (edbp_from_edb_id, edbp_from_un_id, edbp_from_reg_id, edbp_from_edl_id) values (:edb_id, :un_id, :reg_id, :edl_id);
    execute procedure hp_edoc_cancel(edc_code, edl_object_kind, edl_object_id);
end
