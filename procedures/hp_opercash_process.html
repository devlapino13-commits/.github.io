CREATE PROCEDURE HP_OPERCASH_PROCESS
AS
declare variable OPC_ID D_INTEGER;
declare variable ERR_TEXT D_BLOB_TEXT;
declare variable ERR_MAILTO D_STR_255;
declare variable DB_ID D_INTEGER;
declare variable ONE_ERR_TEXT D_STR_8100;
declare variable UN_ID D_ID;
declare variable RES_IS_ERROR D_SMALLINT;
declare variable ERR_MSG D_STR_255;
declare variable TMP_DATETIME D_DATETIME;
begin
  -- Обработка кассовых операций поступающие через новый модуль РМК Кассира

  db_id = f_get_db_id();

  if (db_id <> 41701) then
    exit;

  -- Общие настройки, зависящие от Страны
  err_mailto = 'finstart@mbfplanet.com';

  err_msg = 'Ошибка проведения, OPC_ID:';
  err_text = ''; -- Список доступных проектов

  for select cpr_from_opc_id, cpr_from_un_id
  from cashprocess
  where coalesce(cpr_error_text, '') = ''
  order by cpr_from_opc_id
  into :opc_id, un_id
  do
  begin
    one_err_text = '';
    res_is_error = 0;

    in autonomous transaction do
    begin
      tmp_datetime = cast('NOW' as timestamp);

      select res_text
      from hp_opercash_book_process(:opc_id, :un_id)
      into :one_err_text;

      one_err_text = '';

      insert into cashproclog (cpl_from_opc_id, cpl_started, cpl_finished, cpl_duration_ms, cpl_error)
      values (:opc_id, :tmp_datetime, cast('NOW' as timestamp), datediff(millisecond, :tmp_datetime, cast('NOW' as timestamp)), '');

      -- Ошибки блокировки пропускаем молча
      when gdscode lock_conflict, gdscode update_conflict, gdscode deadlock do
      begin
      end

      -- В остальных случаях фиксируем
      when any do
      begin
        one_err_text = rdb$get_context('USER_TRANSACTION', 'Opercash_error_text');
        if (coalesce(one_err_text, '') <> '') then
          res_is_error = 1; -- пользовательская ошибка
        else
          one_err_text = err_msg || ' ' || opc_id || ', gdscode: ' || gdscode;

        update cashprocess
        set cpr_error_text = :one_err_text
        where cpr_from_opc_id = :opc_id;

        insert into cashproclog (cpl_from_opc_id, cpl_started, cpl_finished, cpl_duration_ms, cpl_error)
        values (:opc_id, :tmp_datetime, cast('NOW' as timestamp), datediff(millisecond, :tmp_datetime, cast('NOW' as timestamp)), :one_err_text);
      end
    end

    if (one_err_text <> '' and res_is_error = 0) then
      err_text = blob_append(err_text, one_err_text); -- другие отправляем по почте
  end

  -- Если в ходе обработки появились ошибки, отправляем разработчикам и ответственным лицам
  if (err_text <> '') then
  begin
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body,
                                msg_lifetime_days, msg_from_un_id,
                                msg_command_kind)
    values (0, :err_mailto, 'Opercash online process', :err_text, 30, 1000000000000,
            'opercash'); -- RefData.RD_Code=:MSG_Command_Kind
  end
end
