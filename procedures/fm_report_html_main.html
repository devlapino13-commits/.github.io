CREATE PROCEDURE FM_REPORT_HTML_MAIN(
    UN_ID D_ID,
    REP_ID D_STR_255,
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    ADD_PARAMS D_STR_2048 = '',
    SOURCE_SYSTEM D_SMALLINT = 1,
    UN_IP D_STR_20 = 0,
    REG_ID D_INTEGER = null)

    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)

declare variable db_id d_integer;

  begin
      select ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = (select un_from_c_id from username where un_id = :un_id) and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_status = 1 and
            ps_is_working = 1

  from username

            ks_date_end >= '31.12.2999' and

        from miniappclient, miniapp

              mac_from_c_id = :obj_id and

              ma_date_end >= '31.12.2999' and
              ma_date_oper >= (current_date - 30)

        into :ma_id_tmp;

        begin
          obj_kind = 'Miniapp';
          obj_id = ma_id_tmp;
        end

      suspend;

        from hp_html_get_loan_status(9, :obj_id, 'full', :un_id)

        suspend;

        begin
          if (exists(select *
                     from expressapp
                     where exa_id = :obj_id and

                           exa_final_status = 0)) then
          begin
