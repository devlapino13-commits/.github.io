CREATE PROCEDURE AAREP_PURCHASE_DETAIL(
    DATEB D_DATE,
    DATEE D_DATE,
    TAX D_STR_255)
RETURNS (
    ID ID_TYPE,
    DT D_DATE,
    DOCNO D_STR_20,
    SELLER D_STR_100,
    INN D_STR_20,
    CODE D_STR_10,
    NAME D_STR_255,
    MODEL D_STR_255,
    QUANTITY CURR_TYPE,
    SUMMA CURR_TYPE)
AS
begin
  for               
    select
      ol_id,                                 
      ol_date,
      ol_doc_no,          
      c_name_fio,
      c_inn,
      feg_code,  
      ml_name,
      ml_model,
      kml_count,
      kml_total
    from operlist
      inner join operbook
        on ob_from_ol_id = ol_id
      inner join clients
        on c_id = ol_document_client
          and c_type = 0                                                                
      inner join kontmat
        on kml_bill_id = ol_id 
          and kml_bill_line_no = ob_number
      inner join matlist
        on ml_id = kml_from_ml_id
      left join foreign_econom_goods_ref
        on feg_id = kml_from_feg_id
    where ol_date between :dateb and :datee         
      and ol_oper_type = 3 
      and ol_oper_subtype = 9 
      and is_delete = 0
      and ((:tax = 'Все') or exists(select * from clientdocs where cds_from_c_id = c_id and cds_from_pk_id in (44,68,71) and cds_memo_3 = :tax))
    order by ol_id                              
  into
    :id,                       
    :dt,
    :docno,                    
    :seller,
    :inn,                           
    :code,                  
    :name,                    
    :model,                    
    :quantity,                       
    :summa
  do suspend;                             
end
