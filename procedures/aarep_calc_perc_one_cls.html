CREATE PROCEDURE AAREP_CALC_PERC_ONE_CLS(
    K_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    DEBUG_MODE D_BOOL = 0,
    KEEP_MINUS_SALDO D_BOOL = 0,
    KEEP_SALDO_ON_CLOSE D_BOOL = 0)
RETURNS (
    RESULT_DATE D_DATE,
    RESULT_CALCED D_CURRENCY,
    RESULT_PAYED D_CURRENCY,
    RESULT_WRITEOFF D_CURRENCY,
    RESULT_PERC_BEGIN D_CURRENCY,
    RESULT_PERC_END D_CURRENCY,
    RESULT_BALANCE_END D_CURRENCY,
    RESULT_CALCED_ALL D_CURRENCY,
    RESULT_PAYED_ALL D_CURRENCY,
    RESULT_WRITEOFF_ALL D_CURRENCY,
    RESULT_DISB_AMOUNT D_CURRENCY,
    RESULT_DISB_DATE D_DATE,
    RESULT_OUT_PERC_BEGIN D_CURRENCY,
    RESULT_OUT_PERC_END D_CURRENCY,
    RESULT_OUT_CALCED D_CURRENCY,
    RESULT_PERC_BASE D_CURRENCY,
    RESULT_CALC_DAYS D_INTEGER,
    RESULT_CALC_FOR_DAYS D_CURRENCY,
    RESULT_MINUS_SALDO D_BOOL,
    RESULT_VAC_PERC_SALDO D_CURRENCY)
AS
declare variable ol_date_prv d_date;
declare variable tmp_perc_base d_currency;
declare variable ol_date d_date;
declare variable percent_per_day d_percent;
declare variable oks_perc_base d_currency;
declare variable perc_pmt d_currency;
declare variable perc_off d_currency;
declare variable perc_acr7 d_currency;
declare variable pr_is_360 d_integer;
declare variable pr_round_perc d_integer;
declare variable pr_percent_kind d_integer;
declare variable percent_prv_day d_percent;
declare variable k_stop_perc d_date;
declare variable calced_out_all d_currency;
declare variable pr_interest_method d_integer;
declare variable end_zero d_bool;
declare variable sh_perc_all d_currency;
declare variable max_diff d_currency;
declare variable ks_date_end d_date;
declare variable db_id d_integer;
declare variable begin_calced d_bool;
declare variable ol_date_orig d_date;
declare variable pay_os d_currency;
declare variable kd_date_end d_date;
declare variable first_line d_bool;
declare variable keep_saldo_begin_on_close d_bool;
declare variable keep_saldo_end_on_close d_bool;
declare variable res_psk_limit d_currency;
declare variable payed_fine d_currency;
declare variable max_perc_amount d_currency;
declare variable vac_status d_integer;
declare variable in_vacation d_bool;
declare variable repay_vac_perc d_currency;
begin
  if (debug_mode is null) then
    debug_mode = 0;

  if (keep_minus_saldo is null) then
    keep_minus_saldo = 0;

  if (keep_saldo_on_close is null) then
    keep_saldo_on_close = 0;

  keep_saldo_begin_on_close = keep_saldo_on_close;
  keep_saldo_end_on_close = keep_saldo_on_close;

  if (keep_saldo_end_on_close = 1 and rep_date_end >= '01.05.2022') then
    keep_saldo_end_on_close = 0;

  if (keep_saldo_begin_on_close = 1 and rep_date_begin >= '01.05.2022') then
    keep_saldo_begin_on_close = 0;

  result_date = rep_date_end;
  result_perc_begin = 0;
  result_out_perc_begin = 0;
  begin_calced = 0;

  result_calced = 0;
  result_payed = 0;
  result_writeoff = 0;
  result_perc_end = 0;
  result_balance_end = 0;
  result_calced_all = 0;
  result_payed_all = 0;
  result_writeoff_all = 0;
  result_disb_amount = 0;
  result_out_perc_end = 0;
  result_out_calced = 0;
  calced_out_all = 0;
  result_minus_saldo = 0;
  result_perc_base = 0;
  result_calc_for_days = 0;
  result_calc_days = 0;
  in_vacation = 0;
  result_vac_perc_saldo = 0;
  repay_vac_perc = 0;
  first_line = 1;

  ol_date_prv = null;

  if (rep_date_begin > rep_date_end) then
  begin
    suspend;
    exit;
  end

  select ks_date_begin, ks_date_end
  from kontstate
  where ks_from_k_id = :k_id and 
        ks_ol_is_delete = 0 and
        ks_state = 1
  into result_disb_date, ks_date_end;

  if (ks_date_end < '31.12.2999') then
    ks_date_end = ks_date_end + 1;

  if (ks_date_end is null or
      rep_date_end < result_disb_date) then
  begin
    suspend;
    exit;
  end

  select k_amount_approved, pr_percent_kind, pr_interest_method, pr_is_360,
         iif(pr_round_perc = 0, 0, 2), k_stop_perc
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into result_disb_amount, pr_percent_kind, pr_interest_method, pr_is_360,
       pr_round_perc, k_stop_perc;

  result_balance_end = result_disb_amount;
  result_perc_base = result_balance_end;
  oks_perc_base = 0;

  if (pr_percent_kind = 2) then
    exception e_default 'Accrual method does not match: pr_percent_kind = 2';

  if (k_stop_perc is null) then
    k_stop_perc = rep_date_end + 1;

  db_id = f_get_db_id();

  max_diff = 5;
  if (rep_date_end >= '01.05.2022') then
    max_diff = 1;

  if (db_id = 94401) then
    max_diff = 0.10;

  select res_psk_limit
  from hp_calcperc_psk_limit(:rep_date_end)
  into :res_psk_limit;

  for select fact_date, sum(delta_base), sum(pay_os), sum(pay_perc), sum(off_perc), sum(acr7_perc),
             max(vac_status), sum(vac_perc)
  from aarep_calc_perc_one_cls_tmp(:k_id, :rep_date_end)
  where fact_date <= :rep_date_end
  group by fact_date
  into ol_date, tmp_perc_base, pay_os, perc_pmt, perc_off, perc_acr7,
       vac_status, repay_vac_perc
  do
  begin
    result_balance_end = result_balance_end + pay_os;

    if (kd_date_end is null or kd_date_end < ol_date) then
    begin
      select kd_percent_per_day, kd_date_end
      from kontdata
      where kd_from_k_id = :k_id and
            kd_ol_is_delete = 0 and
            kd_ol_date <= :ol_date and
            kd_ol_date <= kd_date_end
      order by kd_from_k_id desc, kd_ol_date desc
      rows 1
      into percent_per_day, kd_date_end;
  
      if (row_count = 0) then
        exception e_default f_format('Record Kontdata not found, k_id %1, date: %2', k_id, f_datetostr(ol_date));
    end

    if (pr_interest_method <= 0) then
    begin
      if (oks_perc_base > 0)
      then oks_perc_base = result_disb_amount;
      else oks_perc_base = 0;
    end

    result_date = ol_date;

    if (ol_date_prv is null) then
      ol_date_prv = ol_date;

    if (percent_prv_day is null) then
      percent_prv_day = percent_per_day;

    if (debug_mode = 0 and begin_calced = 0 and ol_date >= rep_date_begin) then
    begin
      begin_calced = 1;

      ol_date_orig = ol_date;
      ol_date = rep_date_begin - 1;

      result_calc_days = f_agein360(ol_date_prv, ol_date, pr_is_360);
      result_calc_for_days = cast(oks_perc_base * percent_prv_day as double precision) / 100.0000 * result_calc_days;
      result_out_perc_begin = calced_out_all - result_payed_all - result_writeoff_all + result_calc_for_days;
      result_out_perc_begin = round(result_out_perc_begin, pr_round_perc);

      result_calc_days = f_agein360(ol_date_prv, minvalue(ol_date, k_stop_perc), pr_is_360);
      result_calc_for_days = cast(oks_perc_base * percent_prv_day as double precision) / 100.0000 * result_calc_days;
      result_perc_begin = result_calced_all - result_payed_all - result_writeoff_all + result_calc_for_days;
      result_perc_begin = round(result_perc_begin, pr_round_perc);

      -- если на дату отчета имеется платеж и разница меньше допускаемого, то накопленные подгоняем до суммы в графике
      sh_perc_all = 0;
    
      select oks_delinq_perc
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :ol_date and
            oks_date_end >= :ol_date
      into sh_perc_all;
      if (sh_perc_all is null) then
        sh_perc_all = 0;
    
      if (abs(sh_perc_all - result_perc_begin) <= max_diff) then
      begin
        result_calced_all = result_calced_all + (sh_perc_all - result_perc_begin);
        result_perc_begin = sh_perc_all;
      end

      if (abs(sh_perc_all - result_out_perc_begin) <= max_diff) then
      begin
        calced_out_all = calced_out_all + (sh_perc_all - result_out_perc_begin);
        result_out_perc_begin = sh_perc_all;
      end

      if (result_perc_begin < 0) then
        result_minus_saldo = 1;

      end_zero = 0;
      if (keep_minus_saldo = 0 and result_perc_begin < 0) then
        end_zero = 1;

      if (keep_saldo_begin_on_close = 0 and result_perc_begin > 0 and ol_date >= ks_date_end) then
        end_zero = 1;

      if (abs(result_perc_begin) <= max_diff) then
        end_zero = 1;

      if (end_zero = 1) then
      begin
        result_calced_all = result_calced_all - result_perc_begin;
        result_perc_begin = 0;
      end
    
      end_zero = 0;
      if (keep_minus_saldo = 0 and result_out_perc_begin < 0) then
        end_zero = 1;

      if (keep_saldo_begin_on_close = 0 and result_out_perc_begin > 0 and ol_date >= ks_date_end) then
        end_zero = 1;

      if (abs(result_out_perc_begin) <= max_diff) then
        end_zero = 1;

      if (end_zero = 1) then
      begin
        calced_out_all = calced_out_all - result_out_perc_begin;
        result_out_perc_begin = 0;
      end

      ol_date = ol_date_orig;
    end

    result_calc_days = f_agein360(ol_date_prv, ol_date, pr_is_360);
    result_calc_for_days = cast(oks_perc_base * percent_prv_day as double precision) / 100.0000 * result_calc_days;
    result_calc_for_days = round(result_calc_for_days, pr_round_perc);
    calced_out_all = calced_out_all + result_calc_for_days;

    result_calc_days = f_agein360(ol_date_prv, minvalue(ol_date, k_stop_perc), pr_is_360);
    result_calc_for_days = cast(oks_perc_base * percent_prv_day as double precision) / 100.0000 * result_calc_days;
    result_calc_for_days = round(result_calc_for_days, pr_round_perc);
    result_calced_all = result_calced_all + result_calc_for_days;

    calced_out_all = calced_out_all + perc_acr7;
    result_calced_all = result_calced_all + perc_acr7;
    result_payed_all = result_payed_all + perc_pmt;
    result_writeoff_all = result_writeoff_all + perc_off;

    result_vac_perc_saldo = result_vac_perc_saldo + repay_vac_perc;
    if (in_vacation = 1) then
      result_vac_perc_saldo = result_vac_perc_saldo + result_calc_for_days;

    if (vac_status = 1) then
      in_vacation = 1;

    if (vac_status = 2) then
      in_vacation = 0;

    if (debug_mode = 1) then
    begin
      result_payed = perc_pmt;
      result_writeoff = perc_off;
    end else
    if (ol_date between rep_date_begin and rep_date_end) then
    begin
      result_payed = result_payed + perc_pmt;
      result_writeoff = result_writeoff + perc_off;
    end

    result_perc_end = result_calced_all - result_payed_all - result_writeoff_all;
    result_out_perc_end = calced_out_all - result_payed_all - result_writeoff_all;

    if (first_line = 0) then
      result_perc_base = oks_perc_base;

    if (ol_date = rep_date_end) then
    begin
      -- если на дату отчета имеется платеж и разница меньше допускаемого, то накопленные подгоняем до суммы в графике
      sh_perc_all = 0;
    
      select oks_delinq_perc
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :ol_date and
            oks_date_end >= :ol_date
      into sh_perc_all;
      if (sh_perc_all is null) then
        sh_perc_all = 0;
    
      if (abs(sh_perc_all - result_perc_end) <= max_diff) then
      begin
        result_calced_all = result_calced_all + (sh_perc_all - result_perc_end);
        result_perc_end = sh_perc_all;
      end

      if (abs(sh_perc_all - result_out_perc_end) <= max_diff) then
      begin
        calced_out_all = calced_out_all + (sh_perc_all - result_out_perc_end);
        result_out_perc_end = sh_perc_all;
      end

      if (result_perc_end < 0) then
        result_minus_saldo = 1;

      end_zero = 0;
      if (keep_minus_saldo = 0 and result_perc_end < 0) then
        end_zero = 1;

      if (keep_saldo_end_on_close = 0 and result_perc_end > 0 and ol_date >= ks_date_end) then
        end_zero = 1;

      if (abs(result_perc_end) <= max_diff) then
        end_zero = 1;

      if (end_zero = 1) then
      begin
        result_calced_all = result_calced_all - result_perc_end;
        result_perc_end = 0;
      end
    
      end_zero = 0;
      if (keep_minus_saldo = 0 and result_out_perc_end < 0) then
        end_zero = 1;

      if (keep_saldo_end_on_close = 0 and result_out_perc_end > 0 and ol_date >= ks_date_end) then
        end_zero = 1;

      if (abs(result_out_perc_end) <= max_diff) then
        end_zero = 1;

      if (end_zero = 1) then
      begin
        calced_out_all = calced_out_all - result_out_perc_end;
        result_out_perc_end = 0;
      end
    end

    result_calced = result_perc_end - (result_perc_begin - result_payed - result_writeoff);
    result_out_calced = result_out_perc_end - (result_out_perc_begin - result_payed - result_writeoff);

    if (debug_mode = 1) then
    begin
      suspend;
      result_perc_begin = result_perc_end;
      result_out_perc_begin = result_out_perc_end;
    end

    ol_date_prv = ol_date;
    percent_prv_day =  percent_per_day;
    oks_perc_base = oks_perc_base + tmp_perc_base;
    first_line = 0;
  end

  if (debug_mode = 0) then
  begin
    if (res_psk_limit > 0) then
    begin
      payed_fine = 0;
  
      select sum(ok_fine)
      from operkred
      where ok_from_k_id = :k_id and
            ok_type = 4 and
            ok_fine <> 0 and
            ok_ol_date <= :rep_date_end and
            ok_ol_is_active = 1
      into payed_fine;
  
      if (payed_fine is null) then
        payed_fine = 0;
  
      max_perc_amount = res_psk_limit * result_disb_amount;
      if (result_perc_end + result_payed_all + result_writeoff_all - payed_fine > max_perc_amount) then
      begin
        result_perc_end = max_perc_amount - (result_payed_all + result_writeoff_all - payed_fine);
        if (result_perc_end < 0) then
          result_perc_end = 0;
  
        result_out_perc_end = result_perc_end;
        result_calced = result_perc_end - (result_perc_begin - result_payed - result_writeoff);
        result_out_calced = result_out_perc_end - (result_out_perc_begin - result_payed - result_writeoff);
      end
    end

    result_out_perc_end = result_out_perc_end - result_perc_end;
    suspend;
  end
end
