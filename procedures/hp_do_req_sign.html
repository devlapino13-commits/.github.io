CREATE PROCEDURE HP_DO_REQ_SIGN(
    RM_ID D_ID)
AS
declare variable NEW_STATUS D_INTEGER;
declare variable OLD_STATUS D_INTEGER;
declare variable UN_ID D_ID;
declare variable REG1_ID D_INTEGER;
declare variable REG2_ID D_INTEGER;
declare variable LOG_ID D_ID;
begin
    -- RFCKG-2925
    if(f_get_db_id() in (39801, 41701)) then
    begin
        -- вытаскиваю пользователя, который проводит данную операцию
        select
            un_id,
            iif(un_curr_reg_id in (1090100, 1010100), un_curr_reg_id, 0) -- информация про ГО Бишкек
        from username
        where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN')
        into
            :un_id,
            :reg1_id;


        select
            rm_from_reg_id,
            rm_status
        from reqmain
        where rm_id = :rm_id
        into
            reg2_id,
            old_status;
    

        -- если запрос в статусе "исполнен", то дальше никаких операций не проводить
        if(old_status = 3) then
            exit;
    
        -- если reg2_id = ГО Бишкек и фигурант не является сотрудником ГО Бишкек
        if(reg2_id in (1090100, 1010100) and reg1_id = 0) then
            exit;
    end

    select rs_status
    from reqsign
    where rs_from_rm_id = :rm_id
    order by rs_number desc
    rows 1
    into :new_status;


    update reqmain
    set rm_status = :new_status
    where rm_id = :rm_id and
        rm_status <> :new_status;

end
