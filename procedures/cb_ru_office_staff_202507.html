CREATE PROCEDURE CB_RU_OFFICE_STAFF_202507(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    JL_NAME VARCHAR(255),
    OFFICER_NAME VARCHAR(100),
    AAREP_BONUS_PLUS CURR_TYPE,
    AAREP_BONUS_MINUS CURR_TYPE,
    AAREP_BONUS_TOTAL CURR_TYPE,
    AAREP_BONUS_TOTAL_FACT CURR_TYPE,
    AAREP_BONUS_OFFICER CURR_TYPE,
    SUM_MINUS CURR_TYPE,
    AAREP_DELINQ_OS CURR_TYPE,
    AAREP_DELINQ_PERC CURR_TYPE,
    AAREP_FACT_OS CURR_TYPE,
    AAREP_FACT_PERC CURR_TYPE,
    AAREP_FACT_FINE CURR_TYPE,
    AAREP_FACT_DELINQ_OS CURR_TYPE,
    AAREP_FACT_DELINQ_PERC CURR_TYPE,
    AAREP_FACT_ADV_OS CURR_TYPE,
    AAREP_FACT_ADV_PERC CURR_TYPE,
    AAREP_BONUS_OS CURR_TYPE,
    AAREP_BONUS_OS_ADV CURR_TYPE,
    REG_LOGIC_LABEL VARCHAR(255),
    STAGE_MONTH DECIMAL(4,2),
    KPI1 CURR_TYPE,
    KPI2 CURR_TYPE,
    PAR_31_360 CURR_TYPE,
    FACT_DEL_BAL_31_360 CURR_TYPE,
    PLAN_DEL_BAL_31_360 CURR_TYPE,
    BALANCE_OFFICE CURR_TYPE,
    PLAN_KP_31_360 CURR_TYPE,
    FACT_KP_31_360 CURR_TYPE,
    TATL_KPI CURR_TYPE,
    LML_ID INTEGER,
    MANAGER_ID DECIMAL(18,0),
    C_ID ID_TYPE,
    KPI3 D_CURRENCY,
    FACT_KOL_SVZ DECIMAL(18,4),
    PLAN_KOL_SVZ DECIMAL(18,4))
AS
declare variable reg_id integer;
declare variable rep_date_begin date;
declare variable aarep_plan_all decimal(18,4);
declare variable aarep_fact_all decimal(18,4);
declare variable aarep_fact_planned decimal(18,4);
declare variable aarep_fact_other decimal(18,4);
declare variable aarep_month_plan_os decimal(18,4);
declare variable aarep_month_plan_perc decimal(18,4);
declare variable jk_system_code varchar(100);
declare variable kol integer;
declare variable kol_mo integer;
declare variable kol_zam_mo integer;
declare variable kol_kobd integer;
declare variable kol_admin integer;
declare variable perc_bonus_mo curr_type;
declare variable perc_bonus_zam_mo curr_type;
declare variable perc_bonus_kobd curr_type;
declare variable perc_bonus_admin curr_type;
declare variable aarep_bonus_total_fact_tmp curr_type;
declare variable rd_id integer;
declare variable rep_period integer;
declare variable mo_id id_type;
declare variable manager_name d_str_100;
declare variable lo_id id_type;
declare variable mgr_name d_str_100;
declare variable officer_id decimal(18,0);
declare variable plan_execute d_currency;
declare variable is_is_ok d_smallint;
declare variable is_admin d_smallint;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;

  mo_id = -1;
  --  if (rep_date_end >= '01.06.2021') then mo_id = 1010100066847; --текущего менеджера нужно добавить в текущу ОИ -- убран по письму Ироды от 24.11.2022

  insert into temp_bonus_office (bo_region_id, bo_reg_logic_label, bo_lml_id, bo_manager_id, bo_manager_name,
                                 /*BO_LO_ID,*/ bo_plan_all, bo_fact_all, bo_fact_planned, bo_fact_other,
                                 bo_month_plan_os, bo_month_plan_perc, bo_delinq_os, bo_delinq_perc, bo_fact_os,
                                 bo_fact_perc, bo_fact_fine, bo_fact_delinq_os, bo_fact_delinq_perc, bo_fact_adv_os,
                                 bo_fact_adv_perc, bo_bonus_os, bo_bonus_os_adv)
  select reg_id, coalesce(reg_logic_label, ''), lml_id, manager_id, manager_name, --lo_id,
         sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other),
         sum(aarep_month_plan_os), sum(aarep_month_plan_perc), sum(aarep_delinq_os), sum(aarep_delinq_perc),
         sum(aarep_fact_os), sum(aarep_fact_perc), sum(aarep_fact_fine), sum(aarep_fact_delinq_os),
         sum(aarep_fact_delinq_perc), sum(aarep_fact_adv_os), sum(aarep_fact_adv_perc), sum(aarep_bonus_os),
         sum(aarep_bonus_os_adv)
  from cb_ru_ssvz_202110_one_tmp(:rep_region_id, :rep_date_end), region
  where reg_id = reg_unicode and
        coalesce(reg_logic_label, '') not containing '.mini_office.'
  group by 1, 2, 3, 4, 5;

  perc_bonus_mo = 85;
  for select bo_region_id, bo_reg_logic_label
      from temp_bonus_office
      where bo_reg_logic_label containing '.mega_office.'
      group by 1, 2
      into reg_id, reg_logic_label
  do
  begin
    aarep_bonus_total_fact_tmp = 0;
    for select bo_lml_id, bo_manager_id, bo_manager_name, sum(bo_plan_all), sum(bo_fact_all), sum(bo_fact_planned),
               sum(bo_fact_other), sum(bo_month_plan_os), sum(bo_month_plan_perc), sum(bo_delinq_os),
               sum(bo_delinq_perc), sum(bo_fact_os), sum(bo_fact_perc), sum(bo_fact_fine), sum(bo_fact_delinq_os),
               sum(bo_fact_delinq_perc), sum(bo_fact_adv_os), sum(bo_fact_adv_perc), sum(bo_bonus_os),
               sum(bo_bonus_os_adv)
        from temp_bonus_office
        where bo_region_id = :reg_id
        group by 1, 2, 3
        into lml_id, manager_id, manager_name, aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other,
             aarep_month_plan_os, aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os,
             aarep_fact_perc, aarep_fact_fine, aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os,
             aarep_fact_adv_perc, aarep_bonus_os, aarep_bonus_os_adv
    do
    begin
      -- для проверки по офисам
      is_is_ok = 0;
      if (reg_id = 1041400) then
      begin
        is_is_ok = 1;
      end

      -->>01.11.20
      aarep_bonus_plus = round((aarep_fact_perc + aarep_fact_fine + aarep_fact_delinq_perc + aarep_fact_adv_perc) * 0.015, 0);
      aarep_bonus_plus = round(aarep_bonus_plus + aarep_fact_delinq_os * 0.01, 0);

      aarep_bonus_minus = round((aarep_delinq_os + aarep_delinq_perc) * 0.03, 0);

      aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus; -- Базовый заработок

      --  Фактический баланс просрочек 31-360 дней
      fact_del_bal_31_360 = null;
      select sum(oks_balance)
      from oksummary
      where oks_region_id = :reg_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            :rep_date_end - oks_delinq_date between 31 and 360 and
            oks_close_state = 0 and
            :lml_id = f_lo_mgr_id(oks_from_k_id, :rep_date_end, oks_kc_id, oks_region_id)
      into fact_del_bal_31_360;
      fact_del_bal_31_360 = coalesce(fact_del_bal_31_360, 0);
      fact_del_bal_31_360 = fact_del_bal_31_360 / 1000000;

      --  Плановый баланс просрочек
      rep_period = extract(year from :rep_date_end) || right((extract(month from :rep_date_end) + 100), 2);
      plan_del_bal_31_360 = null;

      --  Общий кп
      balance_office = null;
      select sum(oks_balance)
      from oksummary
      where oks_region_id = :reg_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0 and
            :lml_id = f_lo_mgr_id(oks_from_k_id, :rep_date_end, oks_kc_id, oks_region_id)
      into balance_office;
      balance_office = coalesce(balance_office, 0);
      balance_office = balance_office / 1000000;

      -- ПАР 31-360
      par_31_360 = null;
      if (fact_del_bal_31_360 > 0 and
          balance_office > 0) then
        par_31_360 = (fact_del_bal_31_360 / balance_office) * 100;

      if (manager_id is null) then
        manager_id = -1;

      plan_del_bal_31_360 = null;
      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :manager_id, 'PlanIndicators.BalancePar31_360', :rep_date_end, :rep_period)
      into plan_del_bal_31_360;

      plan_del_bal_31_360 = coalesce(plan_del_bal_31_360, 0);
      plan_del_bal_31_360 = plan_del_bal_31_360 / 1000000;

      --  Расчет KPI1
      if (par_31_360 < 2) then
        kpi1 = 110;
      else
      if (par_31_360 < 3) then
        kpi1 = 100;
      else
      begin
        kpi1 = iif((1 - (fact_del_bal_31_360 - plan_del_bal_31_360) * 0.2) * 100 < 0, 0, (1 - (fact_del_bal_31_360 - plan_del_bal_31_360) * 0.2) * 100);
      end

      --  фактические данные по КП 0-360 на отчетную дату в миллионах руб.
      fact_kp_31_360 = null;
      select sum(oks_balance)
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0 and
            cast(:rep_date_end as date) - oks_delinq_date < 361 and
            oks_region_id = :reg_id and
            :lml_id = f_lo_mgr_id(oks_from_k_id, :rep_date_end, oks_kc_id, oks_region_id)
      into :fact_kp_31_360;
      fact_kp_31_360 = coalesce(fact_kp_31_360, 0);
      fact_kp_31_360 = fact_kp_31_360 / 1000000;

      --  планируемые данные по КП 0-360 на отчетную дату в миллионах руб.
      plan_kp_31_360 = null;
      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :manager_id, 'PlanIndicators.Portfolio_amount', :rep_date_end, :rep_period)
      into :plan_kp_31_360;

      plan_kp_31_360 = coalesce(plan_kp_31_360, 0);
      plan_kp_31_360 = plan_kp_31_360 / 1000000;

      --  Расчет KPI2
      kpi2 = null;
      plan_execute = 0;

      if (plan_kp_31_360 > 0) then
      begin
        plan_execute = ((fact_kp_31_360 / plan_kp_31_360) - 1) * 100;

        kpi2 = case
                 when plan_execute >= 4 then 150
                 when plan_execute >= 2.5 then 130
                 when plan_execute >= 1 then 110
                 when plan_execute >= 0 then 100
                 when plan_execute >= -1 then 50
                 else 0
               end;
      end

      --  фактическое кол-во Випс на дату
      fact_kol_svz = null;
      select --count(*)
      sum(iif(jk_name = 'СВЗ 0,5', 0.5, 1))
      from hr_pstatus, joblist, jobkind
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_id and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.LO_Standart' and
            (exists(select *
                    from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                    where is_decret = 0) or
            ps_from_c_id = :mo_id) and
            :lml_id = f_lo_mgr_id(null, :rep_date_end, ps_from_c_id, ps_from_reg_id)
      into fact_kol_svz;
      fact_kol_svz = coalesce(fact_kol_svz, 0);

      is_admin = 0;

      if (exists(select *
                 from hr_pstatus, joblist, jobkind
                 where :rep_date_end between ps_date_begin and ps_date_end and
                       ps_from_reg_id = :reg_id and
                       ps_is_working = 1 and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code = 'JobSystem.Administrator')) then
      begin
        is_admin = 1;
      end

      --  План кол-во СВЗ за указанный период
      plan_kol_svz = null;
      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :manager_id, 'PlanIndicators.LO_Count', :rep_date_end, :rep_period)
      into plan_kol_svz;

      plan_kol_svz = coalesce(plan_kol_svz, 0);

      --  Расчет KPI3
      if (fact_kol_svz >= plan_kol_svz) then
        kpi3 = 100;
      else
      if (fact_kol_svz + 0.5 >= plan_kol_svz) then
        kpi3 = 50;
      else
        kpi3 = 0;

      tatl_kpi = (30 + kpi1 * 25 / 100 + kpi2 * 35 / 100 + kpi3 * 10 / 100);

      aarep_bonus_total_fact = aarep_bonus_total * tatl_kpi / 100;

      if ((aarep_plan_all + aarep_fact_all + aarep_fact_planned -->>
          + aarep_delinq_os + aarep_delinq_perc + aarep_fact_os + aarep_fact_perc -->
          + aarep_fact_delinq_perc + aarep_fact_adv_os + aarep_fact_adv_perc) <> 0) then
      begin
        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name --, r1.reg_logic_label
        from region r1, region r2
        where r1.reg_unicode = :reg_id and
              r1.parent = r2.reg_unicode
        into reg_name, branch_name;

        if (manager_id <> -1) then
        begin
          select jl_name, c_name_fio, jk_system_code, c_id
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                ps_from_c_id = :manager_id and
                ps_from_c_id = c_id
          into jl_name, officer_name, jk_system_code, c_id;
        end
        else
        begin
          select jl_name, c_name_fio, c_id, jk_system_code, c_id
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                        from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                        where is_decret = 0) or
                ps_from_c_id = :mo_id) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                ps_from_c_id = c_id rows 1
          into jl_name, officer_name, manager_id, jk_system_code, c_id;
        end

        -- Расчет совокупного стажа сотрудинка, если менее 2х месяцев то бонус не получает
        select lo_stage_month_full
        from hp_get_lo_stage_full(:manager_id, :rep_date_end, :jk_system_code)
        into stage_month;

        aarep_bonus_officer = 0;
        if (aarep_bonus_total_fact > 0) then
        begin
          perc_bonus_mo = 100;
          if (is_admin = 1) then
            perc_bonus_mo = 93;
          if (stage_month >= 2) then
            aarep_bonus_officer = cast(aarep_bonus_total_fact as double precision) / 100 * perc_bonus_mo;

          aarep_bonus_total_fact_tmp = aarep_bonus_total_fact_tmp + aarep_bonus_total_fact;
        end
        if (officer_name is null) then
          c_id = null;
        aarep_bonus_officer = round(aarep_bonus_officer, 0);
        suspend;

        jl_name = null;
        officer_name = null;
        jk_system_code = null;
        manager_id = null;
        c_id = null;

      end

    end

    if ((aarep_plan_all + aarep_fact_all + aarep_fact_planned -->>
        + aarep_delinq_os + aarep_delinq_perc + aarep_fact_os + aarep_fact_perc -->
        + aarep_fact_delinq_perc + aarep_fact_adv_os + aarep_fact_adv_perc) <> 0) then
    begin
      aarep_delinq_os = 0;
      aarep_delinq_perc = 0;
      aarep_fact_os = 0;
      aarep_fact_perc = 0;
      aarep_fact_fine = 0;
      aarep_fact_delinq_os = 0;
      aarep_fact_delinq_perc = 0;
      aarep_fact_adv_os = 0;
      aarep_fact_adv_perc = 0;

      aarep_bonus_os = 0;
      aarep_bonus_os_adv = 0;
      sum_minus = 0;
      aarep_bonus_plus = 0;
      aarep_bonus_minus = 0;
      aarep_bonus_total = 0;
      aarep_bonus_total_fact = 0;

      fact_del_bal_31_360 = null;
      plan_del_bal_31_360 = null;

      par_31_360 = null;
      tatl_kpi = null;

      kpi1 = null;
      kpi2 = null;
      kpi3 = null;

      balance_office = null;
      plan_kp_31_360 = null;
      fact_kp_31_360 = null;

      fact_kol_svz = null;
      plan_kol_svz = null;

      --    вычисляем кол-во сотрудников по должностям
      kol_mo = 0;
      kol_zam_mo = 0;
      kol_kobd = 0;
      kol_admin = 0;
      for select jk_system_code, count(*), max(iif(jl_name <> 'Заместитель менеджера офиса' and jk_system_code = 'JobSystem.Region_KM', 1, 0)) -- если результат 1 то имеется менеджер офиса, если 0 то нету
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Administrator') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                        from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                        where is_decret = 0) or
                ps_from_c_id = :mo_id) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                jl_name <> 'Выездной кассир-оператор' and
                ps_from_c_id = c_id -- and
                --                :lml_id = f_lo_mgr_id(null, :rep_date_end, ps_from_c_id, ps_from_reg_id)
          group by jk_system_code
          into jk_system_code, kol, kol_mo
      do
      begin
        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
          kol_zam_mo = kol - kol_mo;
        end
        if (jk_system_code = 'JobSystem.Administrator') then
        begin
          kol_admin = kol;
        end
      end

      perc_bonus_mo = 0;
      perc_bonus_zam_mo = 0;
      perc_bonus_kobd = 0;
      perc_bonus_admin = 0;
      --    нужно раскидать


      perc_bonus_mo = 100;

      if (is_admin = 1) then
      begin
        perc_bonus_mo = 93;
        if (kol_admin > 0) then
          perc_bonus_admin = cast(7.0 as double precision) / kol_admin;
      end 



      for select jl_name, c_name_fio, c_id, jk_system_code, c_id
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Administrator') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                        from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                        where is_decret = 0) or
                ps_from_c_id = :mo_id) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                jl_name <> 'Выездной кассир-оператор' and
                ps_from_c_id = c_id and
                --                :lml_id = f_lo_mgr_id(null, :rep_date_end, ps_from_c_id, ps_from_reg_id) and
                jk_system_code <> 'JobSystem.Region_KM' -- из этого списка убираем менеджера и зам МО
          order by jk_system_id, jl_name desc
          into jl_name, officer_name, officer_id, jk_system_code, c_id
      do
      begin

        aarep_bonus_officer = 0;
        stage_month = 0;
        --      Расчет совокупного стажа сотрудинка, если менее 2х месяцев то бонус не получает
        select lo_stage_month_full
        from hp_get_lo_stage_full(:officer_id, :rep_date_end, :jk_system_code)
        into stage_month;

        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
          if (stage_month >= 2) then
          begin
            if (jl_name <> 'Заместитель менеджера офиса') then
              aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_mo;
            else
              aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_zam_mo;
          end
        end
        else
        if (stage_month >= 2) then
        begin
          if (jk_system_code = 'JobSystem.Administrator') then
          begin
            aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_admin;
          end
        end

        aarep_bonus_officer = (select res
                               from fhp_round_bank(:aarep_bonus_officer, 0));

        if (officer_name is null) then
          c_id = null;

        aarep_bonus_officer = round(aarep_bonus_officer, 0);
        suspend;
      end

      aarep_bonus_officer = null;
      jl_name = null;
      officer_name = null;
      officer_id = null;
      manager_id = null;
      c_id = null;

    end
  end

  for select bo_region_id, bo_reg_logic_label, --bo_lml_id, bo_manager_id, bo_manager_name, --bo_lo_id,
             sum(bo_plan_all), sum(bo_fact_all), sum(bo_fact_planned), sum(bo_fact_other), sum(bo_month_plan_os),
             sum(bo_month_plan_perc), sum(bo_delinq_os), sum(bo_delinq_perc), sum(bo_fact_os), sum(bo_fact_perc),
             sum(bo_fact_fine), sum(bo_fact_delinq_os), sum(bo_fact_delinq_perc), sum(bo_fact_adv_os),
             sum(bo_fact_adv_perc), sum(bo_bonus_os), sum(bo_bonus_os_adv)
      from temp_bonus_office
      where bo_reg_logic_label not containing '.mega_office.'
      group by 1, 2
      into reg_id, reg_logic_label, --lml_id, manager_id, manager_name, --lo_id,
           :aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os,
           aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine,
           aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc, aarep_bonus_os,
           aarep_bonus_os_adv
  do
  begin
    -- для проверки по офисам
    is_is_ok = 0;
    if (reg_id = 1041400) then
    begin
      is_is_ok = 1;
    end
    -->>01.11.20
    aarep_bonus_plus = round((aarep_fact_perc + aarep_fact_fine + aarep_fact_delinq_perc + aarep_fact_adv_perc) * 0.015, 0);
    aarep_bonus_plus = round(aarep_bonus_plus + aarep_fact_delinq_os * 0.01, 0);

    aarep_bonus_minus = round((aarep_delinq_os + aarep_delinq_perc) * 0.03, 0);

    aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus; -- Базовый заработок

    --  Фактический баланс просрочек 31-360 дней
    fact_del_bal_31_360 = null;
    select sum(oks_balance)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          :rep_date_end - oks_delinq_date between 31 and 360 and
          oks_close_state = 0
    into fact_del_bal_31_360;
    fact_del_bal_31_360 = coalesce(fact_del_bal_31_360, 0);
    fact_del_bal_31_360 = fact_del_bal_31_360 / 1000000;

    --  Плановый баланс просрочек
    select rd_id
    from refdata
    where rd_code = 'PlanIndicators.BalancePar31_360'
    into rd_id;
    rep_period = extract(year from :rep_date_end) || right((extract(month from :rep_date_end) + 100), 2);
    plan_del_bal_31_360 = null;
    select sum(pd_data_1)
    from hp_tmp_plandata_show(:reg_id, null, :rd_id, :rep_period, 0, 0)
    where pd_is_office = 1
    into plan_del_bal_31_360;
    plan_del_bal_31_360 = coalesce(plan_del_bal_31_360, 0);
    plan_del_bal_31_360 = plan_del_bal_31_360 / 1000000;

    --  Общий кп
    balance_office = null;
    select sum(oks_balance)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0
    into balance_office;
    balance_office = coalesce(balance_office, 0);
    balance_office = balance_office / 1000000;

    -- ПАР 31-360
    par_31_360 = null;
    if (fact_del_bal_31_360 > 0 and
        balance_office > 0) then
      par_31_360 = (fact_del_bal_31_360 / balance_office) * 100;

    if (manager_id is null) then
      manager_id = -1;

    select sum(po_amount)
    from hp_get_planoper_mgr_info(:reg_id, :manager_id, 'PlanIndicators.BalancePar31_360', :rep_date_end, :rep_period)
    into plan_del_bal_31_360;

    plan_del_bal_31_360 = coalesce(plan_del_bal_31_360, 0);
    plan_del_bal_31_360 = plan_del_bal_31_360 / 1000000;

    --  Расчет KPI1
    if (par_31_360 < 2) then
      kpi1 = 110;
    else
    if (par_31_360 < 3) then
      kpi1 = 100;
    else
    begin
      kpi1 = iif((1 - (fact_del_bal_31_360 - plan_del_bal_31_360) * 0.2) * 100 < 0, 0, (1 - (fact_del_bal_31_360 - plan_del_bal_31_360) * 0.2) * 100);
    end

    --  фактические данные по КП 0-360 на отчетную дату в миллионах руб.
    fact_kp_31_360 = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_close_state = 0 and
          cast(:rep_date_end as date) - oks_delinq_date < 361 and
          oks_region_id = :reg_id
    into :fact_kp_31_360;
    fact_kp_31_360 = coalesce(fact_kp_31_360, 0);
    fact_kp_31_360 = fact_kp_31_360 / 1000000;

    --  планируемые данные по КП 0-360 на отчетную дату в миллионах руб.
    plan_kp_31_360 = null;
    select sum(po_amount)
    from hp_get_planoper_mgr_info(:reg_id, :manager_id, 'PlanIndicators.Portfolio_amount', :rep_date_end, :rep_period)

    into :plan_kp_31_360;
    plan_kp_31_360 = coalesce(plan_kp_31_360, 0);
    plan_kp_31_360 = plan_kp_31_360 / 1000000;

    --  Расчет KPI2
    kpi2 = null;
    plan_execute = 0;

    if (plan_kp_31_360 > 0) then
    begin
      plan_execute = ((fact_kp_31_360 / plan_kp_31_360) - 1) * 100;

      kpi2 = case
               when plan_execute >= 4 then 150
               when plan_execute >= 2.5 then 130
               when plan_execute >= 1 then 110
               when plan_execute >= 0 then 100
               when plan_execute >= -1 then 50
               else 0
             end;
    end

    --  фактическое кол-во Випс на дату
    fact_kol_svz = null;
    select --count(*)
sum(iif(jk_name = 'СВЗ 0,5', 0.5, 1))
    from hr_pstatus, joblist, jobkind
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :reg_id and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code = 'JobSystem.LO_Standart' and
          (exists(select *
                  from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                  where is_decret = 0) or
          ps_from_c_id = :mo_id)
    into fact_kol_svz;
    fact_kol_svz = coalesce(fact_kol_svz, 0);


      is_admin = 0;

      if (exists(select *
                 from hr_pstatus, joblist, jobkind
                 where :rep_date_end between ps_date_begin and ps_date_end and
                       ps_from_reg_id = :reg_id and
                       ps_is_working = 1 and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code = 'JobSystem.Administrator')) then
      begin
        is_admin = 1;
      end


    select rd_id
    from refdata
    where rd_code = 'PlanIndicators.LO_Count'
    into rd_id;
    --  План кол-во СВЗ за указанный период
    plan_kol_svz = null;
    select sum(pd_data_1)
    from hp_tmp_plandata_show(:reg_id, null, :rd_id, :rep_period, 0, 0)
    where pd_is_office = 1
    into plan_kol_svz;
    plan_kol_svz = coalesce(plan_kol_svz, 0);

    --  Расчет KPI3
    if (fact_kol_svz >= plan_kol_svz) then
      kpi3 = 100;
    else
    if (fact_kol_svz + 0.5 >= plan_kol_svz) then
      kpi3 = 50;
    else
      kpi3 = 0;

    tatl_kpi = (30 + kpi1 * 25 / 100 + kpi2 * 35 / 100 + kpi3 * 10 / 100);

    aarep_bonus_total_fact = aarep_bonus_total * tatl_kpi / 100;

    if ((aarep_plan_all + aarep_fact_all + aarep_fact_planned -->>
        + aarep_delinq_os + aarep_delinq_perc + aarep_fact_os + aarep_fact_perc -->
        + aarep_fact_delinq_perc + aarep_fact_adv_os + aarep_fact_adv_perc) <> 0) then
    begin
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name --, r1.reg_logic_label
      from region r1, region r2
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode
      into reg_name, branch_name; --, reg_logic_label;

      suspend;
      aarep_bonus_total_fact_tmp = aarep_bonus_total_fact;
      if (aarep_bonus_total_fact_tmp < 0) then
        aarep_bonus_total_fact_tmp = 0;

      aarep_delinq_os = 0;
      aarep_delinq_perc = 0;
      aarep_fact_os = 0;
      aarep_fact_perc = 0;
      aarep_fact_fine = 0;
      aarep_fact_delinq_os = 0;
      aarep_fact_delinq_perc = 0;
      aarep_fact_adv_os = 0;
      aarep_fact_adv_perc = 0;

      aarep_bonus_os = 0;
      aarep_bonus_os_adv = 0;
      sum_minus = 0;
      aarep_bonus_plus = 0;
      aarep_bonus_minus = 0;
      aarep_bonus_total = 0;
      aarep_bonus_total_fact = 0;

      fact_del_bal_31_360 = null;
      plan_del_bal_31_360 = null;
      tatl_kpi = null;
      kpi1 = null;
      kpi2 = null;
      kpi3 = null;
      plan_kp_31_360 = null;
      fact_kp_31_360 = null;
      par_31_360 = null;

      fact_kol_svz = null;
      plan_kol_svz = null;

      --    вычисляем кол-во сотрудников по должностям
      kol_mo = 0;
      kol_zam_mo = 0;
      kol_kobd = 0;
      kol_admin = 0;
      for select jk_system_code, count(*), max(iif(jl_name <> 'Заместитель менеджера офиса' and jk_system_code = 'JobSystem.Region_KM', 1, 0)) -- если результат 1 то имеется менеджер офиса, если 0 то нету
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Administrator') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                        from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                        where is_decret = 0) or
                ps_from_c_id = :mo_id) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                jl_name <> 'Выездной кассир-оператор' and
                ps_from_c_id = c_id
          group by jk_system_code
          into jk_system_code, kol, kol_mo
      do
      begin
        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
          kol_zam_mo = kol - kol_mo;
        end
        if (jk_system_code = 'JobSystem.Administrator') then
        begin
          kol_admin = kol;
        end
      end

      perc_bonus_mo = 0;
      perc_bonus_zam_mo = 0;
      perc_bonus_kobd = 0;
      perc_bonus_admin = 0;

      perc_bonus_mo = 100;

      if (is_admin = 1) then
      begin
        perc_bonus_mo = 93;
        if (kol_admin > 0) then
          perc_bonus_admin = cast(7.0 as double precision) / kol_admin;
      end

      for select jl_name, c_name_fio, c_id, jk_system_code, c_id
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Administrator') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                        from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                        where is_decret = 0) or
                ps_from_c_id = :mo_id) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                jl_name <> 'Выездной кассир-оператор' and
                jl_name <> 'Директор регионального развития/Антикризисный менеджер' and
                ps_from_c_id = c_id
          order by jk_system_id, jl_name desc
          into jl_name, officer_name, officer_id, jk_system_code, c_id
      do
      begin

        aarep_bonus_officer = 0;
        stage_month = 0;
        --      Расчет совокупного стажа сотрудинка, если менее 2х месяцев то бонус не получает
        select lo_stage_month_full
        from hp_get_lo_stage_full(:officer_id, :rep_date_end, :jk_system_code)
        into stage_month;

        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
          if (stage_month >= 2) then
          begin
            if (jl_name <> 'Заместитель менеджера офиса') then
              aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_mo;
            else
              aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_zam_mo;
          end
        end
        else
        if (stage_month >= 2) then
        begin
          if (jk_system_code = 'JobSystem.Administrator') then
          begin
            aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_admin;
          end
        end

        aarep_bonus_officer = (select res
                               from fhp_round_bank(:aarep_bonus_officer, 0));

        if (officer_name is null) then
          c_id = null;
        aarep_bonus_officer = round(aarep_bonus_officer, 0);
        suspend;
      end

      aarep_bonus_officer = null;
      jl_name = null;
      officer_name = null;
      officer_id = null;
      c_id = null;
    end
  end
