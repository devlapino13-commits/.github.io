CREATE PROCEDURE AAREP_KONTRACT_LOG(
    REP_REGION INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_PERCENT_CATEGORY DECIMAL(18,7),
    K_PERCENT DECIMAL(18,7),
    K_KOLICH_REPAY INTEGER,
    K_STATUS INTEGER,
    KAL_NEW_KF_NAME VARCHAR(20),
    KAL_NEW_REG_ID INTEGER,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    KF_OFFICER_NAME VARCHAR(100),
    KL_USER_ID DECIMAL(18,0),
    KL_DATETIME TIMESTAMP,
    KL_STATE INTEGER,
    UN_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    KL_STATE_STR VARCHAR(20),
    KS_STATE_STR VARCHAR(20),
    APPROVED_MANAGER_FIO STR_100)
AS
declare variable ks_state integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable k_approved_manager id_type;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select k_id, k_kontract_no, k_disb_date, k_amount_approved, k_percent_category, k_percent, k_kolich_repay,
             k_status, kal_new_kf_name, kal_new_reg_id, r1.reg_code || ' ' || r1.reg_name, parent_name, c1.c_name_fio,
             kl_user_id, kl_datetime, kl_state, un_fio, jl_name || ' офиса ' || r2.reg_name, k_approved_manager
      from kontract, knaktlist, region r1, region_list(:rep_region, 1, 3, 1), clients c1, kontlog, username, hr_pstatus, joblist, region r2
      where k_disb_date between :rep_date_begin and :rep_date_end and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            (id_list = r1.reg_unicode) and
            kal_new_reg_id = r1.reg_unicode and
            r1.reg_status = 3
            /*and k_status=1*/
            and
            c1.c_id = kal_new_kf_officer and
            kl_from_k_id = k_id and
            kl_state in (-1, 1) /*Включают аннулированные и утвержденные контракты*/
            and
            kl_user_id = un_id and
            un_from_c_id = ps_from_c_id and
            k_disb_date between ps_date_begin and ps_date_end and
            ps_from_jl_id = jl_id and
            r2.reg_unicode = ps_from_reg_id
      into :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :k_percent_category, :k_percent, :k_kolich_repay,
           :k_status, :kal_new_kf_name, :kal_new_reg_id, :reg_name, :parent_name, :kf_officer_name, :kl_user_id,
           :kl_datetime, :kl_state, :un_fio, :jl_name, :k_approved_manager
  do
  begin
    approved_manager_fio = null;
    select c_name_fio
    from clients
    where c_id = :k_approved_manager
    into approved_manager_fio;

    kl_state_str = '';
    kl_state_str = case :kl_state
                     when -1 then 'Аннулирован'
                     when 0 then 'В процессе'
                     when 1 then 'Утвержден'
                   end;

    suspend;
  end
end
