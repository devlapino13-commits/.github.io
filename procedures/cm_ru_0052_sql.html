CREATE PROCEDURE CM_RU_0052_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable kf_officer id_type;
declare variable officer_name varchar(100);
declare variable k_circle integer;
declare variable cli_k_id_previous_circle integer;
declare variable c_id id_type;
declare variable k_id_prior id_type;
declare variable cli_k_id_prior_circle id_type;
declare variable max_delinq_day integer;
declare variable kont_no_last varchar(20);
begin
  k_id_prior = 0;
  for
  select kal_new_kf_officer, k_circle, k_disb_date
  from kontract, knaktlist
  where k_id = :k_id
--  and k_circle < 2
  and k_from_kf_id = kal_from_kf_id
  and knaktlist.is_delete = 0
  and current_date between kal_date and kal_date_end
  and kal_new_kf_officer in (
        1010100065165,
        1170100000166,
        1010100141899,
        1010800000351,
        1010100190437,
        1010100206695,
        1010100222748,
        1010300007049,
        1010100237366,
        1010100253585,
        1010100253352,
        1010100259622,
        1010100269753,
        1010100275448,
        1010100277262,
        1010100278773,
        1010100279886,
        1010100281156,
        1010100284546,
        1010100289413,
        1010100294749,
        1010100295109
        )
    into kf_officer, k_circle, kf_disb_date
    do
    begin
      if (k_circle < 2) then
      begin
--      если новый контракт то запрещается
        officer_name = '';
        select c_name_fio from clients
        where c_id = :kf_officer
        into officer_name;
        res_error_text = 'СВЗ: '||officer_name||' не имеет права выдавать новым группам из-за увеличение PAR';
        res_error_is_critical = 2;
        suspend;
      end else
      begin
--      а если повторный контракт, то проверяется предыдущий цикл каждого клиента
--      и в предыдущем цикле у клиента была просрочка более 30 дней то не должно выдаваться
--      изменено по письму Санжара от 21.04.2016г в 23:00, тема письма: Блокировка СВЗ 
        for
        select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
        and kc_circle >= 2
        into c_id, kc_amount
        do
        begin
--        вычисляется предыдущий цикл клиента
          cli_k_id_previous_circle = null;
          select ks_from_k_id
          from kontstate, knclient
          where ks_ol_is_delete = 0
          and ks_state = 1
          and ks_from_k_id <> :k_id
          and ks_date_begin < :kf_disb_date
          and ks_from_k_id = kc_from_k_id
          and kc_amount_approved > 0
          and kc_from_c_id = :c_id
          order by ks_date_begin desc
          rows 1
          into cli_k_id_prior_circle;
--        проверяется максимальное кол-во просроченных дней предыдущего цикла
          if (k_id_prior <> cli_k_id_prior_circle) then
          begin
            k_id_prior = cli_k_id_prior_circle;
            max_delinq_day = null;
            kont_no_last = '';
            select max(ok_ol_date - ok_sh_date)
            from operkred
            where ok_from_k_id = :cli_k_id_prior_circle
            and ok_ol_is_active = 1
            and ok_ol_date <= :kf_disb_date
            and ok_ol_date > ok_sh_date
            and ok_type in (4,5,6)
            and ok_os + ok_perc < -100 -- просроченные суммы менее 100 рублей не считать
            into max_delinq_day;
            if (max_delinq_day is null) then max_delinq_day =0;
            if (max_delinq_day <= 30) then
            begin
--            если в предыдущем цикле просчрока не более 30 дней то проверяется то то что кредит активный или нет
--            если активный то проверяется просрочка предыдущего цикла клиента
              max_delinq_day = null;
              select :kf_disb_date - oks_delinq_date
              from oksummary
              where oks_from_k_id = :cli_k_id_prior_circle
              and :kf_disb_date between oks_date_begin and oks_date_end
              and :kf_disb_date - oks_delinq_date > 30
              and oks_close_state = 0
              into max_delinq_day;
              if (max_delinq_day is null) then max_delinq_day = 0;
              if (max_delinq_day > 30) then
              begin
                select k_kontract_no from kontract
                where k_id = :cli_k_id_prior_circle
                into kont_no_last;
              end
            end
          end
          if (max_delinq_day > 30) then
          begin
            select c_name_fio, c_pass_no from clients
            where c_id = :c_id
            into res_info_text, kf_client_pass_no;
            res_error_text = 'в предыдущем цикле просрочка>30 дней, № контракта предыдущего цикла: '||kont_no_last||'; кол-во просроч дней: '||max_delinq_day;
            res_error_is_critical = 2;
          end
        end
      end
    end
end
