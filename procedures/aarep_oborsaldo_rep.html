CREATE PROCEDURE AAREP_OBORSALDO_REP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    BG_ID VARCHAR(20),
    BG_NAME VARCHAR(255),
    BG_SALDO_BEGIN DECIMAL(18,4),
    BG_OBOROT_DT DECIMAL(18,4),
    BG_OBOROT_KT DECIMAL(18,4),
    BG_SALDO_END DECIMAL(18,4))
AS
begin
  for with tab
      as (select bg_id, bg_type, bg_saldo_begin, bg_oborot_dt, bg_oborot_kt, bg_saldo_end
          from aarep_oborsaldo_tmp(:rep_region_id, :rep_date_begin, :rep_date_end))
      select tab.bg_id, bg_name, bg_saldo_begin, bg_oborot_dt, bg_oborot_kt, bg_saldo_end
      from tab, balgroup
      where tab.bg_id = balgroup.bg_id
      union
      select left(tab.bg_id, 4), bg_name, sum(bg_saldo_begin), sum(bg_oborot_dt), sum(bg_oborot_kt), sum(bg_saldo_end)
      from tab, balgroup
      where left(tab.bg_id, 4) = balgroup.bg_id
      group by 1, 2
      union
      select left(tab.bg_id, 2), bg_name, sum(bg_saldo_begin), sum(bg_oborot_dt), sum(bg_oborot_kt), sum(bg_saldo_end)
      from tab, balgroup
      where left(tab.bg_id, 2) = balgroup.bg_id
      group by 1, 2
      union
      select tab.bg_type,
             case tab.bg_type
               when 1 then 'Активы'
               when 2 then 'Пассивы'
               when 4 then 'Доходы'
               when 5 then 'Расходы'
               when 6 then 'Внебаланс'
             end,
             sum(bg_saldo_begin), sum(bg_oborot_dt), sum(bg_oborot_kt), sum(bg_saldo_end)
      from tab, balgroup
      where left(tab.bg_id, 2) = balgroup.bg_id
      group by 1, 2
      into :bg_id, :bg_name, :bg_saldo_begin, :bg_oborot_dt, :bg_oborot_kt, :bg_saldo_end
  do
  begin
    suspend;
  end
end
