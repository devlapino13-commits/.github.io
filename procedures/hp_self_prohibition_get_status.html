CREATE PROCEDURE HP_SELF_PROHIBITION_GET_STATUS(
    C_ID D_ID NOT NULL,
    OBJECT_KIND D_STR_CODE NOT NULL,
    OBJECT_ID D_ID NOT NULL)
RETURNS (
    RES D_INTEGER,
    SPM_QUERY_QTY D_INTEGER)
AS
declare variable SPM_TOTAL_RESULT D_INTEGER;
begin
  -- процедура возвращает итоговый статус по "Самозапрету" (исходя из проверок в КИБ и доп факторов)
  select spm_total_result,
         spm_query_qty
  from self_prohibition_main spm
  where spm.spm_object_kind = f_objkind(:object_kind, 'id')
  and spm.spm_object_id = :object_id
  and spm.spm_from_c_id = :c_id
  into :spm_total_result,
        :spm_query_qty;

  res = :spm_total_result;
  if (spm_total_result = -3) then
    res = 0;
  if ((spm_total_result = -1 and spm_query_qty > 1) or
       (spm_total_result = -2)) then
  begin
    -- если текущей датой закреплен документ (ищем в clientdocs) (точноее названеи не знаю), но полагаю что назваться должен "от утсвие Самозапрета" тогда
    if (exists (select *
                from clientdocs cds
                inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
                where pdt.pdt_code = 'self_prohibition' and
                      cds.cds_from_c_id = :c_id and
                      cds.cds_doc_date_begin = current_date)) then
      res = 2;
  end

  suspend;
end
