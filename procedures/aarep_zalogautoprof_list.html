CREATE PROCEDURE AAREP_ZALOGAUTOPROF_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_LIST_KIND D_STR_CODE,
    REP_BOOK_KIND D_SMALLINT,
    REP_K_ID D_ID)
RETURNS (
    ZAP_ID D_INTEGER,
    ZAP_FROM_ZA_ID D_INTEGER,
    ZAP_DATE D_DATE,
    ZAP_NO D_STR_CODE,
    ZAP_AVG_COST D_CURRENCY,
    ZAP_ZALOG_COST D_CURRENCY,
    ZAP_STATUS D_SMALLINT,
    ZAP_FROM_OL_ID D_ID,
    ZAD_FROM_UN_ID D_ID,
    ZAD_DATETIME D_DATETIME,
    ZA_ID D_INTEGER,
    ZA_FROM_K_ID D_ID,
    ZA_FROM_MA_ID D_INTEGER,
    ZA_FROM_C_ID D_ID,
    ZA_FROM_Z_ID D_INTEGER,
    ZA_ORIG_STS_ID D_INTEGER,
    ZA_ORIG_PTS_ID D_INTEGER,
    ZA_NEW_STS_ID D_INTEGER,
    ZA_NEW_PTS_ID D_INTEGER,
    ZA_FROM_ZAL_ID D_INTEGER,
    ZA_FROM_ZR_ID D_INTEGER,
    Z_ID D_INTEGER,
    Z_FROM_K_ID ID_TYPE,
    Z_FROM_C_ID ID_TYPE,
    Z_TYPE D_INTEGER,
    Z_SUBTYPE D_INTEGER,
    Z_NAME D_STR_100,
    Z_ADDRESS D_STR_255,
    Z_DOCUMENT D_STR_255,
    Z_COST_MARKET CURR_TYPE,
    Z_COST_LIKVID CURR_TYPE,
    Z_COST_KONTRACT CURR_TYPE,
    Z_MEMO1 D_STR_LONG,
    Z_MEMO2 D_STR_2048,
    Z_NAME_FIO D_STR_255,
    Z_PHONE D_STR_100,
    Z_FROM_FL_ID ID_TYPE,
    Z_FROM_ADL_ID ID_TYPE,
    Z_REGISTRY_NO D_STR_CODE,
    Z_REGISTRY_DATE D_DATE,
    Z_REGISTRY_AGENCY D_STR_100,
    Z_DATE_BEGIN D_DATE,
    Z_DATE_END D_DATE,
    Z_IS_READY D_SMALLINT,
    Z_SOURCE_Z_ID D_INTEGER,
    ZAP_KIND_STR D_STR_CODE,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    PR_NAME D_STR_100,
    K_FROM_KF_ID D_ID,
    K_REGION_ID D_INTEGER,
    K_REGION_NAME D_STR_100,
    K_LO_FIO D_STR_100,
    RES_ZAP_STATUS_STR D_STR_CODE,
    ZAL_COST D_CURRENCY,
    Z_SUBTYPE_STR D_STR_CODE,
    RES_KC_ID D_ID,
    RES_BOOK_SALDO D_CURRENCY,
    RES_BOOK_DIFF D_CURRENCY,
    RES_LOAN_STATUS D_STR_CODE,
    RES_LOAN_CLOSED D_DATE,
    RES_XLS_BLS_ID D_ID,
    RES_IS_XLS D_BOOL)
AS
declare variable z_subtype_prv d_integer;
declare variable sab_id d_integer;
declare variable acc_id d_id;
declare variable ks_state d_smallint;
begin
  z_subtype_str = '';

  if (coalesce(rep_list_kind, '') = '') then
    rep_list_kind = '0';

  if (rep_list_kind = '0') then
    rep_list_kind = 'prof';
  else
  if (rep_list_kind = '1') then
    rep_list_kind = 'active';
  else
  if (rep_list_kind = '2') then
    rep_list_kind = 'disb';

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id;

  for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
             zad_from_un_id, zad_datetime,
             za_id, za_from_k_id, za_from_ma_id, za_from_c_id, za_from_z_id, za_orig_sts_id, za_orig_pts_id,
             za_new_sts_id, za_new_pts_id, za_from_zal_id, za_from_zr_id,
             z_from_k_id, z_from_c_id, z_type, z_subtype, z_name, z_address, z_document, z_cost_market, z_cost_likvid,
             z_cost_kontract, z_memo1, z_memo2, z_name_fio, z_phone, z_from_fl_id, z_from_adl_id, z_id, z_registry_no,
             z_registry_date, z_registry_agency, z_date_begin, z_date_end, z_is_ready, z_source_z_id,
             k_kontract_no, k_disb_date, k_amount_approved, k_from_kf_id,
             pr_name
  from aarep_zalogautoprof_tmp(:rep_date_begin, :rep_date_end, :rep_list_kind, :rep_book_kind, :rep_k_id),
       zalogauto, zalog, kontract, product
  where zap_from_za_id = za_id and
        za_from_z_id = z_id and
        za_from_k_id = k_id and
        k_from_pr_id = pr_id
  into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
       :zad_from_un_id, :zad_datetime,
       :za_id, :za_from_k_id, :za_from_ma_id, :za_from_c_id, :za_from_z_id, :za_orig_sts_id, :za_orig_pts_id,
       :za_new_sts_id, :za_new_pts_id, :za_from_zal_id, :za_from_zr_id,
       :z_from_k_id, :z_from_c_id, :z_type, :z_subtype, :z_name, :z_address, :z_document, :z_cost_market,
       :z_cost_likvid, :z_cost_kontract, :z_memo1, :z_memo2, :z_name_fio, :z_phone, :z_from_fl_id, :z_from_adl_id,
       :z_id, :z_registry_no, :z_registry_date, :z_registry_agency, :z_date_begin, :z_date_end, :z_is_ready, :z_source_z_id,
       :k_kontract_no, :k_disb_date, :k_amount_approved, :k_from_kf_id,
       :pr_name
  do
  begin
    zap_kind_str = 'Переоценка';
    if (k_disb_date = zap_date) then
      zap_kind_str = 'Выдача';

    res_zap_status_str = trim(case zap_status
      when -3 then 'Создано'
      when -2 then 'Отказо'
      when -1 then 'Доработка'
      when 0 then 'Очередь КК'
      when 1 then 'Процесс КК'
      when 2 then 'Одобрено'
      else ''
    end);

    if (zap_from_ol_id is not null) then
    begin
      if (exists(select * from operlist ol
                 where ol.ol_id = :zap_from_ol_id and
                       ol.is_delete = 0)) then
      begin
        zap_status = 3;
        res_zap_status_str = 'Проведено';
      end
    end

    k_region_id = null;
    k_region_name = '';
    k_lo_fio = '';

    select f.kal_new_reg_id, reg_code || ' ' || reg_name, c_name_fio
    from knaktlist f, region, clients
    where f.kal_from_kf_id = :k_from_kf_id and
          f.is_delete = 0 and
          f.kal_date <= :zap_date and
          f.kal_date_end >= :zap_date and
          f.kal_new_reg_id = reg_unicode and
          f.kal_new_kf_officer = c_id
    into k_region_id, k_region_name, k_lo_fio;

    zal_cost = 0.0;
    select zal_cost
    from zalogautolist
    where zal_id = :za_from_zal_id
    into zal_cost;

    if (z_subtype_prv is distinct from z_subtype) then
    begin
      if (z_subtype is not null) then
      begin
        select zs_name
        from zalogsubtype
        where zs_id = :z_subtype
        into :z_subtype_str;
      end
    end

    res_loan_status = '';
    res_loan_closed = null;

    select ks_state
    from kontstate
    where ks_from_k_id = :za_from_k_id and
          ks_date_begin <= :rep_date_end and
          ks_date_end >= :rep_date_end and
          ks_ol_is_delete = 0
    into ks_state;

    if (ks_state = 0) then
      res_loan_status = 'Заявка';
    else
    if (ks_state = 1) then
      res_loan_status = 'Активный';
    if (ks_state = 2) then
      res_loan_status = 'Закрыт';

    select ks_date_begin
    from kontstate
    where ks_from_k_id = :za_from_k_id and
          ks_state = 2 and
          ks_ol_is_delete = 0
    into res_loan_closed;

    res_kc_id = null;
    res_book_saldo = 0;
    res_book_diff = 0;

    select kc_id
    from knclient
    where kc_from_k_id = :za_from_k_id and
          kc_amount_approved > 0
    into res_kc_id;

    for select acc_id
    from ref_account
    where acc_from_sab_id = :sab_id and
          acc_object_kind = 8 and
          acc_object_id = :res_kc_id
    into acc_id
    do
      res_book_saldo = res_book_saldo - f_acc_bal(acc_id, zap_date);

    res_book_diff = zap_zalog_cost - res_book_saldo;

    res_xls_bls_id = null;
    res_is_xls = 0;

    select iif(b1.bls_filename like '%.xls%', cd.cds_doc_blob_1, cd.cds_doc_blob_2)
    from clientdocs cd
         inner join passkind on (cds_from_pk_id = pk_id)
         inner join passdoctype on (pdt_id = pk_from_pdt_id and pdt_code = 'reval_of_fair_value_cars')
         left join blobstore b1 on (b1.bls_id = cd.cds_doc_blob_1 and b1.bls_filename like '%.xls%')
         left join blobstore b2 on (b2.bls_id = cd.cds_doc_blob_2 and b2.bls_filename like '%.xls%')
    where cds_from_c_id = :za_from_c_id and
          cds_doc_date_begin <= :rep_date_end
    order by cds_is_delete, cds_verify_status desc, cds_datetime desc
    rows 1
    into res_xls_bls_id;

    if (res_xls_bls_id is not null) then
      res_is_xls = 1;

    suspend;

    z_subtype_prv = z_subtype;
  end
end
