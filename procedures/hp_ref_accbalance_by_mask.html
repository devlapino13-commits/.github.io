CREATE PROCEDURE HP_REF_ACCBALANCE_BY_MASK(
    AB_CODE_MASK D_STR_CODE)
RETURNS (
    AB_ID D_ID,
    AB_CODE D_STR_CODE,
    AB_IS_EXTENDED D_SMALLINT,
    AB_PARENT_ID D_ID)
AS
declare variable pos d_integer;
declare variable sym d_str_sym;
declare variable acc_start d_str_code;
declare variable orig_mask d_str_code;
begin
  if (ab_code_mask is null) then
    exception e_default 'Балансовый счет не указан';

  ab_code_mask = replace(ab_code_mask, ' ', '');

  if (char_length(ab_code_mask) < 5) then
    exception e_default 'Длина балансового счета меньше 5 символов'||ab_code_mask;

  ab_code_mask = replace(ab_code_mask, '(', ' (');
  orig_mask = ab_code_mask;

  ab_code_mask = replace(ab_code_mask, '*', '%');
  ab_code_mask = replace(ab_code_mask, 'x', '_');
  ab_code_mask = replace(ab_code_mask, 'х', '_');

  ab_is_extended = 0;
  if (char_length(ab_code_mask) > 5) then
  begin
    ab_is_extended = 1;

    if (ab_code_mask not like '_____ (%)') then
      exception e_default 'Неверный формат счета: '||orig_mask;
  end

  acc_start = '';
  pos = 0;
  while (pos < 5) do
  begin
    pos = pos + 1;
    sym = substring(ab_code_mask from pos for 1);
    if (sym between '0' and '9')
    then acc_start = acc_start || sym;
    else break;
  end

  for select ab_id, ab_code, ab_parent_id
  from ref_accbalance
  where ab_level = (3 + :ab_is_extended) and
        ab_code starting :acc_start and
        ab_code like :ab_code_mask
  into ab_id, ab_code, ab_parent_id
  do
    suspend;
end
