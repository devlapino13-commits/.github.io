CREATE PROCEDURE AAREP_CALC_AMORT_TEMP(
    REG_ID INTEGER,
    YEAR_NO INTEGER,
    MONTH_NO INTEGER)
RETURNS (
    PCA_BILL_ID DECIMAL(18,0),
    PCA_BILL_LINE_NO INTEGER,
    PCA_START_YEAR INTEGER,
    PCA_START_MONTH INTEGER,
    PCA_YEAR_NO INTEGER,
    PCA_MONTH_NO INTEGER,
    PCA_AGE INTEGER,
    PCA_TOTAL_OS DECIMAL(18,4),
    PCA_TOTAL_AMORT DECIMAL(18,4),
    PCA_LIFETIME INTEGER,
    PCA_LIFETIME_SALDO INTEGER,
    PCA_CURR_OS DECIMAL(18,4),
    PCA_CURR_AMORT DECIMAL(18,4),
    PCA_CURR_AMORT_OTHER DECIMAL(18,4),
    PCA_REGION_ID INTEGER)
AS
declare variable dont_calc smallint;
declare variable pca_koef integer;
declare variable pca_calc_type smallint;
begin
  /* Procedure Text */
  for select pca_bill_id, pca_bill_line_no, pca_start_year, pca_start_month, pca_year_no, pca_month_no, pca_age,
             sum(pca_total_os), sum(pca_total_amort), avg(pca_lifetime), avg(pca_lifetime_saldo), sum(pca_curr_os),
             sum(pca_curr_amort), sum(pca_curr_amort_other), pca_region_id
      from aarep_calc_amort_temp_list(:reg_id, :year_no, :month_no)
      group by pca_bill_id, pca_bill_line_no, pca_start_year, pca_start_month, pca_year_no, pca_month_no, pca_age, pca_region_id
      into :pca_bill_id, :pca_bill_line_no, :pca_start_year, :pca_start_month, :pca_year_no, :pca_month_no, :pca_age,
           :pca_total_os, :pca_total_amort, :pca_lifetime, :pca_lifetime_saldo, :pca_curr_os, :pca_curr_amort,
           :pca_curr_amort_other, :pca_region_id
  do
  begin
    if (pca_total_os >= 0) then
      suspend;
  end
end
