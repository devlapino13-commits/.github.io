CREATE PROCEDURE FM_HTML_KIB_REPORT(
    MA_ID D_ID,
    C_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable KH_CREATE_DATE D_DATE;
declare variable UN_FIO D_STR_100;
declare variable KF_CLIENT_NAME D_STR_100;
declare variable KF_DATE_BIRTH D_DATE;
declare variable KF_CLIENT_PASS_NO D_STR_20;
declare variable KF_INN D_STR_20;
declare variable KF_AMOUNT decimal(18,2);
declare variable MA_KOLICH_REPAY D_INTEGER;
declare variable BASIC_AMOUNT decimal(18,2);
declare variable ACTIVE_CNT D_INTEGER;
declare variable MONTHLY_REPAY decimal(18,2);
declare variable ALL_CNT D_INTEGER;
declare variable SHOWN_CNT D_INTEGER;
declare variable SCORE_RATE D_STR_100;
declare variable RISK_RATE D_STR_255;
declare variable RISK_RATE_DESCRIPTION D_STR_CODE;
declare variable COMMON_POINTS D_INTEGER;
declare variable IPF_POINTS D_INTEGER;
declare variable KH_NO D_INTEGER;
declare variable KH_COMPANY D_STR_CODE;
declare variable KH_CURRENCY D_STR_10;
declare variable KF_IS_PORUCH_STR D_STR_20;
declare variable KH_IS_ACTIV_TXT D_STR_CODE;
declare variable KH_CREDITTYPE D_STR_CODE;
declare variable KH_OWNAMOUNT decimal(18,2);
declare variable KH_UNPAIDAMOUNT decimal(18,2);
declare variable KH_CREDITSTATUS D_STR_CODE;
declare variable KH_DATEAWARD D_DATE;
declare variable KH_AMOUNT decimal(18,2);
declare variable KH_UNPAIDPERCENAMOUNT decimal(18,2);
declare variable KH_REPAYSTATUS D_STR_255;
declare variable KH_DATESTOP D_DATE;
declare variable KH_PERIOD_IN_MONTH decimal(18,2);
declare variable KH_DELIQDAYSCOUNT D_INTEGER;
declare variable KH_CLIENTS_FIO D_STR_LONG;
declare variable KH_PORUCH_FIO D_STR_LONG;
declare variable REGION_ID D_INTEGER;
declare variable COLOR_BLACK varchar(100);
declare variable COLOR_YELLOW varchar(100);
declare variable COLOR_RED varchar(100);
declare variable COLOR_LIGHT_RED varchar(100);
declare variable COLOR_GREEN varchar(100);
declare variable COLOR_LIGHT_GREEN varchar(100);
begin
  color_black = 'black';
  color_yellow = '#f08205';
  color_red = '#ff0000';
  color_light_red = '#ff9999';
  color_green = '#008000';
  color_light_green = '#00ff00';
  select ma.ma_office_id
  from miniapp ma
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  into region_id;

  res_html = (select res_style
              from fm_html_styles) || '<body>';

  res_html = blob_append(res_html, '<b>Отчет по проверке в ЗАО КБ "Ишеним"</b><br>');

  /*
  risk_descr_250 = 'E3';
  risk_descr_529 = 'E2';
  risk_descr_573 = 'E1';
  risk_descr_603 = 'D3';
  risk_descr_619 = 'D2';
  risk_descr_640 = 'D1';
  risk_descr_652 = 'C3';
  risk_descr_664 = 'C2';
  risk_descr_676 = 'C1';
  risk_descr_688 = 'B3';
  risk_descr_700 = 'B2';
  risk_descr_712 = 'B1';
  risk_descr_730 = 'A3';
  risk_descr_742 = 'A2';
  risk_descr_900 = 'A1';
  */

  select t.kh_create_date, t.un_fio, t.kf_client_name, t.kf_date_birth, t.kf_client_pass_no,
         t.kf_inn, t.kf_amount, t.ma_kolich_repay, t.basic_amount, t.active_cnt, t.monthly_repay,
         t.all_cnt, t.shown_cnt, t.score_rate, t.risk_rate, t.risk_rate_description,
         t.common_points, t.ipf_points
  from aarep_kib_miniapp_cli_report(:region_id, :ma_id, :c_id) t
  rows 1 -- Для статичной информации
  into kh_create_date, un_fio, kf_client_name, kf_date_birth, kf_client_pass_no,
       kf_inn, kf_amount, ma_kolich_repay, basic_amount, active_cnt, monthly_repay,
       all_cnt, shown_cnt, score_rate, risk_rate, risk_rate_description,
       common_points, ipf_points;

  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Дата проверки', :kh_create_date, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Пользователь', :un_fio, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Клиент ' || :kf_client_name || '<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Дата рождения', f_datetimetostr(:kf_date_birth), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Документ', :kf_client_pass_no, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Код соц. страхов.', :kf_inn, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Предполагаемые суммы по всем активным кредитам<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Сумма в заявке', :kf_amount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Срок кредита в заявке', :ma_kolich_repay, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Основная сумма', :basic_amount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Кол-во', :active_cnt, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Ежемесяч. платеж', :monthly_repay, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Всего кредитных историй', :all_cnt, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Отображено', :shown_cnt, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

--    res_html = res_html = blob_append(res_html, '<b>Расшифровка балла<b><br>');
--    место для кода по расшифровке баллов                                                                                                                                                                                               </div>

    -- RFCKG-3069
    score_rate =case
                    when risk_rate containing 'E' then '<font color="' || color_red || '"><b>' || score_rate || '</b></font>'
                    when risk_rate containing 'D' then '<font color="' || color_light_red || '"><b>' || score_rate || '</b></font>'
                    when risk_rate containing 'C' then '<font color="' || color_yellow || '"><b>' || score_rate || '</b></font>'
                    when risk_rate containing 'B' then '<font color="' || color_light_green || '"><b>' || score_rate || '</b></font>'
                    when risk_rate containing 'A' then '<font color="' || color_green || '"><b>' || score_rate || '</b></font>'
                    else score_rate
                end;

    risk_rate = case
                    when risk_rate containing 'E' then '<div style="background-color: ' || color_red || '; color: black; display: inline-block; padding: 10px; font-size: 14px; font-weight: bold; text-align: center; border: 1px solid black;">' || risk_rate || '</div>'
                    when risk_rate containing 'D' then '<div style="background-color: ' || color_light_red || '; color: black; display: inline-block; padding: 10px; font-size: 14px; font-weight: bold; text-align: center; border: 1px solid black;">' || risk_rate || '</div>'
                    when risk_rate containing 'C' then '<div style="background-color: ' || color_yellow || '; color: black; display: inline-block; padding: 10px; font-size: 14px; font-weight: bold; text-align: center; border: 1px solid black;">' || risk_rate || '</div>'
                    when risk_rate containing 'B' then '<div style="background-color: ' || color_light_green || '; color: black; display: inline-block; padding: 10px; font-size: 14px; font-weight: bold; text-align: center; border: 1px solid black;">' || risk_rate || '</div>'
                    when risk_rate containing 'A' then '<div style="background-color: ' || color_green || '; color: black; display: inline-block; padding: 10px; font-size: 14px; font-weight: bold; text-align: center; border: 1px solid black;">' || risk_rate || '</div>'
                    else '<div style="background-color: ' || color_black || '; color: white; display: inline-block; padding: 10px; font-size: 14px; font-weight: bold; text-align: center; border: 1px solid black;">' || risk_rate || '</div>'
                end;


  res_html = blob_append(res_html, '<b>Скоринг<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Скоринг балл', :score_rate, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Степень риска', :risk_rate, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Пояснение', :risk_rate_description, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  res_html = blob_append(res_html, '<b>Внутренний скоринг Компании<b><br>');
  res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Баллы по классике', :common_points, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Баллы по ИПФ', :ipf_points, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

  for select t.kh_no, iif(t.kh_is_our = 1, 'В М Булак', 'В другом ФКУ') as kh_company,
             t.kh_currency, t.kf_is_poruch_str,
             iif(t.kh_is_active = 1, 'Активный', 'Не активный') as kh_is_activ_txt,
             t.kh_credittype, t.kh_ownamount, t.kh_unpaidamount, t.kh_creditstatus,
             t.kh_dateaward, t.kh_amount, t.kh_unpaidpercenamount, t.kh_repaystatus,
             t.kh_datestop, t.kh_period_in_month, t.kh_deliqdayscount,
             t.kh_clients_fio, t.kh_poruch_fio
      from aarep_kib_miniapp_cli_report(:region_id, :ma_id, :c_id) t
      into kh_no, kh_company, kh_currency, kf_is_poruch_str, kh_is_activ_txt,
           kh_credittype, kh_ownamount, kh_unpaidamount, kh_creditstatus,
           kh_dateaward, kh_amount, kh_unpaidpercenamount, kh_repaystatus,
           kh_datestop, kh_period_in_month, kh_deliqdayscount,
           kh_clients_fio, kh_poruch_fio
  do
  begin
    res_html = blob_append(res_html, '<b>Кредит № '|| :kh_no || '<b><br>');

    res_html = blob_append(res_html, '<table width="100%"><col width="25%"><col width="25%"><col width="25%"><col width="25%">' ||
               '<tr><th>ФКУ</th><th>Валюта</th><th>Участник</th><th>Статус</th></tr>' ||
               f_html_table_row(:kh_company, :kh_currency, :kf_is_poruch_str, :kh_is_activ_txt, -1, -1, -1, -1, 4, '') ||
               '<br></table>');

    res_html = blob_append(res_html, '<table width="100%"><col width="40%"><col width="60%">' ||
               f_html_table_row('Тип', :kh_credittype, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Дата выдачи', f_datetostr(:kh_dateaward), '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Дата окончания', f_datetostr(:kh_datestop), '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Собств. сумма', :kh_ownamount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Общая сумма', :kh_amount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Срок кредита', :kh_period_in_month, '', '', -1, -1, -1, -1, 2, '') ||
            
               f_html_table_row('Остаток ОС', :kh_unpaidamount, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Остаток %', :kh_unpaidpercenamount, '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Статус кредита', :kh_creditstatus, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Статус погашения', :kh_repaystatus, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Макс. дн. просроч.', :kh_deliqdayscount, '', '', -1, -1, -1, -1, 2, '') ||

               f_html_table_row('Заемщики', :kh_clients_fio, '', '', -1, -1, -1, -1, 2, '') ||
               f_html_table_row('Поручители', :kh_poruch_fio, '', '', -1, -1, -1, -1, 2, '')|| '</table><br>');
  end 
  res_html = blob_append(res_html, '</body>');
  suspend;
end
