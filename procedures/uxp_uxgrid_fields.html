CREATE PROCEDURE UXP$UXGRID_FIELDS(
    GRID_UUID D_UUID,
    GRID_UUID_CODE D_STR_CODE = null)
RETURNS (
    RES_CHECKED D_BOOL,
    UI_UUID D_UUID,
    UI_GRID_UUID D_UUID,
    UI_PARENT_BAND_ID D_STR_CODE,
    UF_UUID D_UUID,
    UF_FIELD_NAME D_STR_CODE,
    UF_POSITION D_INTEGER,
    UI_CAPTION D_STR_100,
    UI_UGR_UUID D_UUID,
    UI_POSITION D_INTEGER,
    UI_WIDTH D_INTEGER,
    UI_SIZABLE D_BOOL,
    UI_ORDER_INDEX D_INTEGER,
    UI_GROUP_INDEX D_INTEGER,
    UI_FIXED_KIND D_INTEGER,
    UI_SUMMARY_KIND D_INTEGER,
    UI_BAND_ID D_INTEGER,
    UI_HINT D_STR_LONG,
    UI_HIDDEN D_SMALLINT,
    UI_READONLY D_BOOL,
    UI_IS_BAND D_BOOL,
    UI_HAS_CHILD D_BOOL,
    UI_HEADER_ALIGN D_SMALLINT,
    UI_VALUE_ALIGN D_SMALLINT,
    RES_IMAGE_INDEX D_INTEGER,
    RES_HAS_EXPRESSION D_BOOL,
    RES_EXPRESSION D_STR_LONG)
AS
declare variable us_uuid d_uuid;
declare variable is_report d_bool;
declare variable ui_ufr_uuid d_uuid;
begin
  if (coalesce(grid_uuid, '') = '') then
    grid_uuid = uxf$uniq(grid_uuid_code);

  ui_grid_uuid = grid_uuid;

  select ufr_uuid, ufr_us_uuid, iif(uxf$code(ufr_owner, 'kind') = 'uxreport', 1, 0)
  from ux$grid, ux$frame
  where ugr_uuid = :grid_uuid and
        ugr_ufr_uuid = ufr_uuid
  into ui_ufr_uuid, us_uuid, is_report;

  if (us_uuid is null) then
    exception e_default 'Grid_uuid is not Grid items';

  ui_ugr_uuid = grid_uuid;

  res_checked = 1;
  ui_parent_band_id = '-1';
  uf_uuid = null;
  uf_field_name = '';
  ui_caption = '';
  uf_position = 0;
  ui_ugr_uuid = grid_uuid;
  ui_width = 0;
  ui_sizable = 1;
  ui_order_index = null;
  ui_group_index = null;
  ui_fixed_kind = 0;
  ui_summary_kind = 0;
  ui_band_id = -1;
  ui_hint = '';
  ui_hidden = 0;
  ui_readonly = 0;
  ui_is_band = 1;
  ui_uuid = null;

  for select bnd_id, coalesce(bnd_parent_id, -1), iif(bnd_childs = 0, 0, 1), bnd_position,
             bnd_caption, bnd_hidden, bnd_width, bnd_fixed_kind
  from ux$band
  where bnd_grid_uuid = :grid_uuid
  order by bnd_index
  into :uf_field_name, :ui_parent_band_id, :ui_has_child,
       :ui_position, :ui_caption, :ui_hidden, :ui_width, :ui_fixed_kind
  do
  begin
    res_image_index = 43;
    if (ui_has_child = 1) then
      res_image_index = 40;
    ui_band_id = ui_parent_band_id;
    suspend;
  end

  if (row_count > 0) then
  begin
    res_image_index = 42;
    ui_band_id = -1;
    uf_field_name = '-1';
    ui_parent_band_id = '';
    ui_has_child = 1;
    ui_position = -1;
    ui_caption = 'Поля и категории';
    ui_hidden = 0;
    ui_width = 0;
    ui_fixed_kind = 0;
    suspend;
  end

  res_image_index = 41;
  ui_has_child = 0;
  ui_is_band = 0;
  for select uf_uuid, uf_field_name, uf_name, uf_position
  from ux$field, ux$frame, ux$frparam
  where ufr_uuid = :ui_ufr_uuid and
        ufp_ufr_uuid = ufr_uuid and
        uf_uuid = ufp_uf_uuid and
        uf_state = 0 and
        uf_is_param = 0 and
        ufp_parent is null
  order by iif(ufr_us_uuid = uf_owner, 0, 1), uf_position
  into uf_uuid, uf_field_name, ui_caption, uf_position
  do
  begin
    res_checked = 0;
    ui_band_id = -1;
    ui_parent_band_id = '-1';

    ui_position = uf_position;
    ui_width = 0;
    ui_sizable = 1;
    ui_order_index = null;
    ui_group_index = null;
    ui_fixed_kind = 0;
    ui_summary_kind = 0;
    ui_hint = '';
    ui_hidden = 0;
    ui_readonly = 0;
    ui_uuid = null;

    if (is_report = 0) then
    begin
      if (not exists(select * from ux$frparam
      where ufp_uf_uuid = :uf_uuid and
            ufp_ufr_uuid = :ui_ufr_uuid)) then
        Continue;
    end

    select ugi_uuid, ugi_position, ugi_width, ugi_sizable,
           ugi_order_index, ugi_group_index, ugi_fixed_kind, ugi_summary_kind, ugi_bnd_id,
           ugi_caption, ugi_hint, ugi_hidden, ugi_readonly,
           ugi_header_align, ugi_value_align
    from ux$gritem
    where ugi_ugr_uuid = :grid_uuid and
          ugi_uf_uuid = :uf_uuid
    into :ui_uuid, :ui_position, :ui_width, :ui_sizable,
         :ui_order_index, :ui_group_index, :ui_fixed_kind, :ui_summary_kind, ui_band_id,
         :ui_caption, :ui_hint, :ui_hidden, :ui_readonly,
         :ui_header_align, :ui_value_align;

    if (row_count = 1) then
      res_checked = 1;

    res_has_expression = 0;
    res_expression = '';

    select exp_expression
    from ux$expression
    where exp_ux_uuid = :ui_uuid and
          exp_kind in (2, 3)
    into res_expression;

    if (coalesce(res_expression, '') <> '') then
      res_has_expression = 1;

    if (ui_band_id is null) then
      ui_band_id = -1;

    ui_parent_band_id = ui_band_id;

    if (ui_order_index = 0) then
      ui_order_index = null;

    if (ui_group_index = 0) then
      ui_group_index = null;

    suspend;
  end
end
