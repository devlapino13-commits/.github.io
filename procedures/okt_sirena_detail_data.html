        from hp_objectprops_memo_get('Kontract', :k_id, 'Refdata', 'ObjectProps.Kontract_memo', 1)
        into :res_text
    do
    begin
      res_text = coalesce(res_text, '');
      if (res_text <> '') then
        blob_txt = BLOB_APPEND(blob_txt, '</br><b>КОММЕНТРИЙ ВИПС ПО КРЕДИТУ: </b>' || coalesce(res_text, '') || '</br>');
    end

-- история звонков
    blob_txt = BLOB_APPEND(blob_txt, '</br><b>ИСТОРИЯ ЗВОНКОВ УЧАСТНИКУ ГРУППЫ (3 последних звонка)</b></br>');

    i = 0;

    for select project,
               result,
               memo,
               called_datetime,
               un_name
    from
    (
        select rd.rd_name as project,
               rd_res.rd_name as result,
               pantc.pcl_memo as memo,
               pantc.pcl_called_datetime as called_datetime,
               iif(un.un_name is null, pantk.pant_oper_called, un.un_name) as un_name
        from pantera_kontracts pantk
        inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
        inner join pantera_clients pantc on pantc.pcl_from_pan_id = pantk.pant_id
        inner join refdata rd_res on rd_res.rd_id = pantc.pcl_rd_res
        left join pantera_clients_opers pco on pco.pco_from_pcl_id = pantc.pcl_id
        left join username un on un.un_id = pco.pco_from_un_id
        where pantk.pant_from_k_id = :k_id and
              pantc.pcl_from_c_id = :c_id_gr
        union all
        select  rd.rd_name as project,
                    'Поисковик менеджера',
                    omu.omu_memo,
                    omu.omu_datetime,
                    un.un_name
            from okt_manager_update omu
            inner join pantera_kontracts pantk on pantk.pant_id = omu.omu_from_pant_id
            inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
            inner join username un on un.un_id = omu.omu_from_un_id
            where pantk.pant_from_k_id = :k_id and
                  omu.omu_object_id = :c_id_gr and
                  omu.omu_object_kind = 0 and
                  omu.omu_memo <> ''
    ) A
    order by called_datetime desc

        into :project, :result, memo, called_datetime, pant_oper_called
    do
    begin
      if ((pant_oper_called = 'FINSTART_ROBOT') or (i = 3)) then
        continue;

      i = i + 1;

      blob_txt = BLOB_APPEND(blob_txt, '<b>№.. ' || cast(i as str_10) || '</b><br>');
      blob_txt = BLOB_APPEND(blob_txt, 'Проект..............<b>' || coalesce(project, '') || '</b><br>');
      blob_txt = BLOB_APPEND(blob_txt, 'Оператор..........' || coalesce(pant_oper_called, '') || '<br>');
      blob_txt = BLOB_APPEND(blob_txt, 'Результат..........<b>' || coalesce(result, '') || '</b><br>');
      blob_txt = BLOB_APPEND(blob_txt, 'Примечание.....' || coalesce(memo, '') || '<br>');

      called_datetime_str = replace(iif(extract(day from called_datetime) < 10, '0', '') || extract(day from called_datetime) || '.' || iif(extract(month from called_datetime) < 10, '0', '') || extract(month from called_datetime) || '.' || extract(year from called_datetime) || ':  ' || extract(hour from called_datetime) || '. ' || extract(minute from called_datetime), ' ', '');

      blob_txt = BLOB_APPEND(blob_txt, 'Время звонка: ' || coalesce(called_datetime_str, '') || '<br></br>');
    end
    if (i = 0) then
      blob_txt = BLOB_APPEND(blob_txt, 'Звонков не было');
  end

  suspend;
end
