CREATE PROCEDURE FIND_PLACE_STREETS(
    PARENT_ID DECIMAL(18,0))
RETURNS (
    PL_ID DECIMAL(18,0),
    PARENT DECIMAL(18,0),
    PL_NAME VARCHAR(255),
    PLT_NAME VARCHAR(255),
    PL_STREET VARCHAR(255),
    PLT_SHORT_NAME VARCHAR(100),
    PL_NAME_ORIG VARCHAR(100))
AS
declare variable p_id decimal(18,0);
declare variable p_name varchar(100);
declare variable kol integer;
declare variable plt_short_name_side integer;
begin
  /* Procedure Text */
  kol = 0;
  if ((parent_id is null) or (parent_id <= 0)) then
    parent_id = 0;

  if (parent_id <> 0) then
  begin
-- Ищем улиц под выбранный населенный пункт
    for select pl_id, parent, pl_name, plt_name, plt_short_name, plt_short_name_side
        from places, placetype
        where pl_status = plt_id and
              plt_can_select = 2 and
              parent = :parent_id
        into :pl_id, :parent, :pl_name, :plt_name, :plt_short_name, :plt_short_name_side
    do
    begin
      pl_name_orig = pl_name;
      kol = 1;
      if (plt_short_name is null) then
        plt_short_name = '';
      if (plt_short_name <> '') then
      begin
        if (plt_short_name_side = -1) then
        begin
          if (right(plt_short_name, 1) = '.') then
            pl_street = :plt_short_name || :pl_name;
          else
            pl_street = :plt_short_name || ' ' || :pl_name;
        end
        else
        if (plt_short_name_side = 1) then
          pl_street = :pl_name || ' ' || :plt_short_name;
        else
          pl_street = :pl_name;
      end
      else
        pl_street = :pl_name;
      suspend;
    end
-- Ищем улиц под все дочерние ветки
    for with recursive pl_list
        as (select pl_id, parent, pl_name, pl_status
            from places
            where pl_id = :parent_id
            union all
            select p.pl_id, p.parent, p.pl_name, p.pl_status
            from places p
            join pl_list on (pl_list.pl_id = p.parent))
        select pl_id, pl_name
        from pl_list, placetype
        where pl_status = plt_id and
              pl_id <> :parent_id and
              plt_can_select = 1
        into :p_id, :p_name
    do
    begin
      parent = :p_id;
      for select pl_id, pl_name, plt_name, plt_short_name, plt_short_name_side
          from places, placetype
          where pl_status = plt_id and
                plt_can_select = 2 and
                parent = :p_id
          into :pl_id, :pl_name, :plt_name, :plt_short_name, :plt_short_name_side
      do
      begin
        pl_name_orig = pl_name;
        if (plt_short_name is null) then
          plt_short_name = '';
        if (plt_short_name <> '') then
        begin
          if (plt_short_name_side = -1) then
          begin
            if (right(plt_short_name, 1) = '.') then
              pl_street = :plt_short_name || :pl_name;
            else
              pl_street = :plt_short_name || ' ' || :pl_name;
          end
          else
          if (plt_short_name_side = 1) then
            pl_street = :pl_name || ' ' || :plt_short_name;
          else
            pl_street = :pl_name;
        end
        else
          pl_street = :pl_name;
        pl_name = pl_name || ' (' || :p_name || ')';
        kol = 1;
        suspend;
      end
    end

    if (kol = 0) then
    begin
      pl_id = 0;
      parent = :parent_id;
      pl_name = 'Нет улиц';
      pl_name_orig = '';
      plt_name = '';
      pl_street = '';
      suspend;
    end
  end
end
