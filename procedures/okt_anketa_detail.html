CREATE PROCEDURE OKT_ANKETA_DETAIL(
    UN_ID DECIMAL(18,0),
    ACL_ID DECIMAL(18,0))
RETURNS (
    RES_QUESTION_CODE STR_100,
    RES_ANSWER_CODE STR_100,
    STATE INTEGER,
    ED_TEXT VARCHAR(255),
    ED_KIND INTEGER,
    TITLE VARCHAR(255),
    DETAIL VARCHAR(255),
    MRU VARCHAR(5000),
    IS_NA INTEGER)
AS
declare variable que_id integer;
declare variable ans_id integer;
declare variable anketa_group str_255;
declare variable rd_code str_255;
declare variable question str_255;
declare variable acl_object_id id_type;
declare variable cc_birth_date date;
declare variable cc_cell_no_add str_100;
begin
  /*
  OKTELL

  Отображает набор вопросов и варианты ответов
  */

  select rd.rd_code, al.acl_object_id
  from actionlist al
  inner join refdata rd on al.acl_from_rd_id = rd.rd_id
  where al.acl_id = :acl_id
  into :anketa_group, :acl_object_id;

  for select res_question_id, -- id вопроса
             res_rd_id, -- id ответа
             res_state, -- Статус (-1, 0, 1)
             res_str_value, -- Текст редактора
             rd_editor_kind, -- Тип редактора
--   Вопросы: 2-флажки, 3-радиокнопки
--   Ответы:  0-обычный, 1-с редактором
             rd_name, -- Текст вопроса/ответа
             rd_description, -- Описание вопроса
             rd_mru_list, -- Список подсказок
             iif(rd_code = 'Results.NoAnswer', 1, 0), -- Системная строка "Нет ответа"
             rd_code
      from hp_actions_short_detail(:un_id, :acl_id)
      into que_id, ans_id, state, ed_text, ed_kind, title, detail, mru, is_na, rd_code
  do
  begin
-- Тестовое заполнение MRU-списка
--    if (title = 'Таджикистан') then
--    begin
--      mru = 'Москва; Баку; Каракол; Наманган; Бишкек; Санкт Петербург; Ош; Нарын';
--    end

    if (que_id = ans_id) then
    begin
      res_question_code = rd_code;
      res_answer_code = rd_code;
    end
    else
    if (que_id <> ans_id) then
    begin
      res_answer_code = rd_code;
    end

    if (anketa_group containing ('OKTELL.')) then
    begin
      if (que_id <> ans_id) then
      begin
        res_answer_code = rd_code;
      end
    end

-- верификация
    if (anketa_group = 'OKTELL.Projects.Verification') then
    begin
      select cl.cc_cell_no_add, cl.cc_birth_date
      from contacts cl
      where cl.cc_id = :acl_object_id
      into :cc_cell_no_add, cc_birth_date;

      if ((res_question_code = 'AnketaQuestions.OKTELL.AddCellNo') and
          (res_answer_code = 'Results.StrValue')) then
      begin
        ed_text = coalesce(cc_cell_no_add, '');
      end

      if ((res_question_code = 'AnketaQuestions.OKTELL.BirthDate') and
          (res_answer_code = 'Results.StrValue')) then
      begin
        if (cc_birth_date is not null) then
        begin
          ed_text = replace(iif(extract(day from cc_birth_date) < 10, '0', '') || extract(day from cc_birth_date) || '.' || iif(extract(month from cc_birth_date) < 10, '0', '') || extract(month from cc_birth_date) || '.' || extract(year from cc_birth_date), ' ', '');
        end
      end
    end

-- звонки по заявкам с сайта
    if (anketa_group = 'OKTELL.Projects.WebSite') then
    begin
      if ((res_question_code = 'AnketaQuestions.OKTELL.WebSite.OfficePlan') and
          (res_answer_code = 'Results.StrValue')) then
      begin
        select left(list(reg_name, ';'), 5000)
        from (select reg_name
              from region reg
              where reg.reg_status = 3 and
                    reg.reg_date_close is null and
                    reg.reg_unicode not in (1012100, 1012200)
              order by reg_name) a
        into :mru;
      end

      if ((res_question_code = 'AnketaQuestions.OKTELL.WebSite.Product') and
          (res_answer_code = 'Results.StrValue')) then
      begin
        mru = 'Баракат;Баракат+;Семья;Семья+';
      end
    end

-- анкета п жалобам
    if (anketa_group = 'OKTELL.Projects.InCall.Complaint') then
    begin
      if ((res_question_code = 'AnketaQuestions.OKTELL.Complain.Office') and
          (res_answer_code = 'Results.StrValue')) then
      begin
        select left(list(reg_name, ';'), 5000)
        from (select reg_name
              from region reg
              where reg.reg_status = 3 and
                    reg.reg_date_close is null and
                    reg.reg_unicode not in (1012100, 1012200)
              order by reg_name) a
        into :mru;
      end
    end

-- анкета по вакансиям
    if (anketa_group = 'OKTELL.Projects.InCall.Vacancy') then
    begin
      if ((res_question_code = 'AnketaQuestions.OKTELL.Vacancy.Office') and
          (res_answer_code = 'Results.StrValue')) then
      begin
        select left(list(reg_name, ';'), 5000)
        from (select reg_name
              from region reg
              where reg.reg_status = 3 and
                    reg.reg_date_close is null and
                    reg.reg_unicode not in (1012100, 1012200)
              order by reg_name) a
        into :mru;
      end
    end

-- информация по офисам
    if (anketa_group = 'OKTELL.Projects.InCall.OfficeData') then
    begin
      if ((res_question_code = 'AnketaQuestions.OKTELL.OfficeData.Office') and
          (res_answer_code = 'Results.StrValue')) then
      begin
        select left(list(reg_name, ';'), 5000)
        from (select reg_name
              from region reg
              where reg.reg_status = 3 and
                    reg.reg_date_close is null and
                    reg.reg_unicode not in (1012100, 1012200)
              order by reg_name) a
        into :mru;
      end
    end

    suspend;
  end
end
