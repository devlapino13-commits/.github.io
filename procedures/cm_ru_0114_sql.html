CREATE PROCEDURE CM_RU_0114_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable kf_officer id_type;
declare variable officer_name varchar(100);
declare variable k_circle integer;
declare variable cli_k_id_previous_circle integer;
declare variable c_id id_type;
declare variable k_id_prior id_type;
declare variable cli_k_id_prior_circle id_type;
declare variable max_delinq_day integer;
declare variable kont_no_last varchar(20);
declare variable kc_circle integer;
declare variable ks_date_end date;
declare variable kf_officer_old id_type;
declare variable kal_date_old date;
begin
    k_id_prior = 0;
    for select kal_new_kf_officer, k_circle, k_disb_date
    from kontract, knaktlist kal, svz_par_problem spp
    where k_id = :k_id
        and k_from_kf_id = kal_from_kf_id
        and kal.is_delete = 0
        and current_date between kal_date and kal_date_end
        and spp.svz_id = kal_new_kf_officer
        and spp.spp_is_del = 0
    into kf_officer, k_circle, kf_disb_date
    do
    begin
        select c_name_fio
        from clients
        where c_id = :kf_officer
        into officer_name;
        for select kc_from_c_id, kc_amount_approved, kc_circle
        from knclient
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
--        and kc_circle >= 2
        into c_id, kc_amount, kc_circle
        do
        begin
          if (kc_circle < 2) then
          begin
            select c_name_fio, c_pass_no from clients
            where c_id = :c_id
            into res_info_text, kf_client_pass_no;
            res_error_text = 'СВЗ: '||officer_name||' не имеет права выдавать новым клиентам из-за увеличение PAR30 более 9%';
            res_error_is_critical = 2;
            suspend;
          end else
          begin
--        вычисляется предыдущий цикл клиента
              cli_k_id_prior_circle = null;
              select ks_from_k_id, ks_date_end
              from kontstate, knclient
              where ks_ol_is_delete = 0
              and ks_state = 1
              and ks_from_k_id <> :k_id
              and ks_date_begin < :kf_disb_date
              and ks_from_k_id = kc_from_k_id
              and kc_amount_approved > 0
              and kc_from_c_id = :c_id
              order by ks_date_begin desc
              rows 1
              into cli_k_id_prior_circle, ks_date_end;
              if (cli_k_id_prior_circle is not null) then
              begin
                  /*
                  if (ks_date_end < cast('31.12.9999' as date)) then ks_date_end = ks_date_end + 1;
                  else ks_date_end = current_date;
    
                  select kal_new_kf_officer, kal_date from knaktlist, kontract
                  where knaktlist.is_delete = 0
                  and :ks_date_end between kal_date and kal_date_end
                  and kal_from_kf_id = k_from_kf_id
                  and k_id = :cli_k_id_prior_circle
                  into kf_officer_old, kal_date_old;
--                если был передан виртуальному сотруднику то вытаскивает СВЗ до передачи виртуальному сотруднику
                  if (kf_officer_old <= 1000000000004) then
                  begin
                    select kal_new_kf_officer from knaktlist, kontract
                    where knaktlist.is_delete = 0
                    and :kal_date_old - 1 between kal_date and kal_date_end
                    and kal_from_kf_id = k_from_kf_id
                    and k_id = :cli_k_id_prior_circle
                    into kf_officer_old;
                  end 
                  if (kf_officer_old <> kf_officer) then
                  begin
                    select c_name_fio, c_pass_no from clients
                    where c_id = :c_id
                    into res_info_text, kf_client_pass_no;
    
                    res_error_text = 'СВЗ: '||officer_name||' не имеет права выдавать чужим клиентам из-за увеличение PAR30 более 9%';
                    res_error_is_critical = 2;
                    suspend;
                  end
                  */
              end else
              begin
                select c_name_fio, c_pass_no from clients
                where c_id = :c_id
                into res_info_text, kf_client_pass_no;
    
                res_error_text = 'СВЗ: '||officer_name||' не имеет права выдавать новым клиентам из-за увеличение PAR30 более 9%';
                res_error_is_critical = 2;
                suspend;
              end 
          end
        end
    end
end
