CREATE PROCEDURE FM_CONTINFO_LIST(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    CONT_TYPE VARCHAR(255),
    CONT_VALUE VARCHAR(100),
    CONT_STATUS_ID INTEGER,
    CONT_STATUS VARCHAR(255),
    EDIT_TYPE VARCHAR(255),
    EDIT_MASK VARCHAR(255),
    EDIT_LIST VARCHAR(255),
    NEED_VERIFY INTEGER,
    ERR_TEXT VARCHAR(255))
AS
declare variable is_active_client smallint;
declare variable has_miniapp smallint;
begin
  err_text = '';
  select is_active_client
  from fm_client_status(:un_id, :object_id)
  into is_active_client;

  select has_miniapp
  from fm_client_has_miniapp(:un_id, :object_id)
  into :has_miniapp;

  for select rd_code, -- Тип контакта (н-р: ContactInfo.Cell_Main)
             res_value, -- Значение контакта
             res_verify_status, -- Статус (числовой)
             res_value_status, -- Статус (текстовый)
             rd_str_id, -- Тип реадктора (cell, email, phone, other)
             res_editmask, -- Маска ввода (RegExp)
             res_combo_mask, -- Коды стран для номеров телефонов
             res_need_verify -- Требуется ли верификация контакта
      from aarep_object_continfo(:object_kind, :object_id, 0, 'full', :un_id, 1)
      into cont_type, cont_value, cont_status_id, cont_status, edit_type, edit_mask, edit_list, need_verify
  do
  begin
    suspend;
  end
end
