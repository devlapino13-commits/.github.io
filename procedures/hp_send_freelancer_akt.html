CREATE PROCEDURE HP_SEND_FREELANCER_AKT(
    C_ID D_ID,
    UN_ID D_ID,
    SD_ID D_INTEGER,
    SSH_ID D_INTEGER,
    SIGNER_C_ID D_ID,
    SSHE_ID D_INTEGER)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable edc_id d_integer;
declare variable object_kind d_integer;
declare variable signed_date d_date;
declare variable category d_str_255;
declare variable edoc_kind d_str_100;
begin
  res_text = '';
  if (signer_c_id = 0 or signer_c_id is null) then
  begin
    res_text = 'Необходимо выбрать подписанта.';
    suspend;
    exit;
  end

  if (sshe_id <> -1 and f_edoc_status('freelancer_akt', :sshe_id, current_date) = 'signed') then
  begin
    res_text = 'Акт уже подписан.';
    suspend;
    exit;
  end

  edc_id = f_edoc_kind('asp.agreement', 'id');

  select edc_object_kind
  from edoc_cfg
  where edc_id = :edc_id
  into :object_kind;

  select cast(eds.eds_status_datetime as date)
  from edoc_list edl
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id and
        eds.eds_status = 1
  where edl.edl_object_kind = :object_kind and
        edl.edl_object_id = :c_id and
        edl.edl_from_edc_id = :edc_id and
        current_date between edl.edl_blank_date and coalesce(edl.edl_date_end, current_date) and
        edl.edl_status = 1
  order by edl.edl_status desc, edl.edl_id desc
  rows 1
  into :signed_date;

  select sd.sd_category
  from servicedog sd
  where sd.sd_id = :sd_id
  into :category;

  if (:category containing 'стажер') then
    edoc_kind = 'probationer_kontract';
  else
    edoc_kind = 'freelancer_kontract';

  if (edoc_kind = 'freelancer_kontract' and (:signed_date < '11.01.2022' or :signed_date is null)) then
  begin
  /*
    11.01.2022 был создан новый договор ЭЦП, единый для клиентов, внештатников и продавцов.
    Если договор был подписан до этой даты, то необходимо переподписать.
    */
    res_text = 'Необходимо подписать ЭЦП последней редакции.';
    suspend;
    exit;
  end
  if (edoc_kind = 'probationer_kontract' and (:signed_date < '20.05.2022' or :signed_date is null)) then
  begin
  /*
    20.05.2022 был создан новый договор ЭЦП, единый для клиентов, внештатников и продавцов.
    Если договор был подписан до этой даты, то необходимо переподписать.
    */
    res_text = 'Необходимо подписать ЭЦП последней редакции.';
    suspend;
    exit;
  end

  if (f_edoc_status(:edoc_kind, :sd_id, current_date) <> 'signed') then
  begin
    res_text = 'Необходимо подписать договор на оказание услуг.';
    suspend;
    exit;
  end
  if (:sshe_id = -1) then
    insert into servicesh_edoc (sshe_from_sd_id, sshe_from_ssh_id)
    values (:sd_id, :ssh_id)
    returning sshe_id
    into sshe_id;

  if (sshe_id is null or sshe_id < 0) then
  begin
    res_text = 'Произошла непредвиденная ошибка. Попробуйте позже.';
    suspend;
    exit;
  end

  if (res_text = '') then
    select edl_id, res_text
    from hp_edoc_insert('freelancer_akt', f_objkind('freelancer_akt', 'id'), :sshe_id, :un_id)
    into :edl_id, :res_text;

  if (:edl_id is null) then
    edl_id = 0;

  if (res_text = '' and
      :edl_id > 0) then
    update edoc_list
    set edl_manager_c_id = :signer_c_id
    where edl_id = :edl_id;

end
