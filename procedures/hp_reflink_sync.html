CREATE PROCEDURE HP_REFLINK_SYNC(
    REF_CATEGORY D_STR_255,
    REF_OBJECT_ID ID_TYPE,
    REF_VALUES D_STR_255,
    REF_ACTION_KIND D_STR_100,
    MASTER_CATEGORY D_STR_100 = '')
AS
declare variable ref_table_name d_str_100;
declare variable ref_key_field d_str_100;
declare variable ref_label_field d_str_100;
declare variable obj_kind d_integer;
declare variable rd_id d_integer;
declare variable rd_str_id d_str_255;
declare variable n_finish d_integer;
declare variable sql_text d_str_2048;
declare variable orig_value d_str_255;
begin
  -- Add - Добавляем одно значение
  -- Delete - Исключаем одно значение
  -- Full - Приводим в полное соответствие
  -- Import - Full+Создает записи в справочнике если нет

  if (ref_values is null) then
    ref_values = '';

  if (coalesce(ref_category, '') = '') then
    exit;

  if (coalesce(ref_values, '') = '' and
      not(ref_action_kind in ('full', 'import'))) then
    exit;

  if (coalesce(master_category, '') = '') then
    master_category = ref_category;

  select ref_label_field,
         ref_table_name,
         f_objkind(ref_table_name, 'field'),
         f_objkind(ref_table_name, 'id')
  from hp_reflink_category(0)
  where ref_name = :master_category and
        f_objkind(ref_table_name, 'field') <> ''
  into :ref_label_field,
       ref_table_name,
       ref_key_field,
       obj_kind;

  -- Если ref_label_field пустой, значит нет поле для хранения текстовой версии

  if (obj_kind is null) then
    exit;

  if (ref_values = '' and
      ref_action_kind in ('full', 'import')) then
  begin
    execute procedure hp_reflinklog('delete', :obj_kind, :ref_object_id);
    delete from reflink
    where rlk_object_kind = :obj_kind and
          rlk_object_id = :ref_object_id;
    exit;
  end

  if (ref_action_kind in ('full', 'import')) then
  begin
    if (left(ref_values, 1) <> '.') then
      ref_values = '.' || ref_values;
    if (right(ref_values, 1) <> '.') then
      ref_values = ref_values || '.';

    for select rd_id,
               rd_str_id
        from reflink, refdata
        where rlk_object_kind = :obj_kind and
              rlk_object_id = :ref_object_id and
              rlk_from_rd_id = rd_id and
              rd_main_code = 'reflink'
        into :rd_id,
             :rd_str_id
    do
    begin
      if (position('.' || rd_str_id || '.', ref_values) = 0) then
      begin
        execute procedure hp_reflinklog('delete', :obj_kind, :ref_object_id, :rd_id);
        delete from reflink
        where rlk_object_kind = :obj_kind and
              rlk_object_id = :ref_object_id and
              rlk_from_rd_id = :rd_id;
      end
    end

    while (ref_values <> '') do
    begin
      rd_str_id = '';
      n_finish = position('.', ref_values, 2);
      if (n_finish > 0) then
      begin
        rd_str_id = substring(ref_values from 2 for (n_finish - 2));
        ref_values = replace(ref_values, '.' || rd_str_id || '.', '.');
        if (ref_values = '.') then
          ref_values = '';
        if (ref_values <> '' and
            left(ref_values, 1) <> '.') then
          ref_values = '.' || ref_values;
      end
      else
      begin
        rd_str_id = replace(ref_values, '.', '');
        ref_values = '';
      end
      if (rd_str_id <> '') then
      begin
        rd_id = null;
        select rd_id
        from refdata
        where rd_main_code = 'reflink' and
              rd_str_id = :rd_str_id and
              rd_is_active = 1 rows 1
        into rd_id;

        if (rd_id is null and
            ref_action_kind = 'import') then
        begin
          insert into refdata (rd_main_code, rd_name, rd_memo, rd_str_id)
          values ('reflink', :rd_str_id, :ref_category, :rd_str_id)
          returning (rd_id)
          into :rd_id;
        end

        if (rd_id is null) then
          exception e_default 'Unknown reflink value in refdata: ' || ref_category || ' - ' || :rd_str_id;

        if (not(exists(select *
                       from reflink
                       where rlk_object_kind = :obj_kind and
                             rlk_object_id = :ref_object_id and
                             rlk_from_rd_id = :rd_id))) then
        begin
          insert into reflink (rlk_object_kind, rlk_object_id, rlk_from_rd_id)
          values (:obj_kind, :ref_object_id, :rd_id);
          execute procedure hp_reflinklog('insert', :obj_kind, :ref_object_id, :rd_id);
        end
      end
    end
  end

  if (ref_action_kind in ('add', 'delete')) then
  begin
    if (ref_label_field = '') then
    begin
      select rd_id
      from refdata
      where rd_main_code = 'reflink' and
            rd_str_id = :ref_values and
            rd_is_active = 1 rows 1
      into rd_id;

      if (rd_id is null) then
        exception e_default 'Unknown reflink value: ' || ref_category || ' - ' || :ref_values;

      if (ref_action_kind = 'add') then
      begin
        if (not(exists(select *
                       from reflink
                       where rlk_object_kind = :obj_kind and
                             rlk_object_id = :ref_object_id and
                             rlk_from_rd_id = :rd_id))) then
        begin
          insert into reflink (rlk_object_kind, rlk_object_id, rlk_from_rd_id)
          values (:obj_kind, :ref_object_id, :rd_id);
          execute procedure hp_reflinklog('insert', :obj_kind, :ref_object_id, :rd_id);
        end
      end
      else
      begin
        execute procedure hp_reflinklog('delete', :obj_kind, :ref_object_id, :rd_id);
        delete from reflink
        where rlk_object_kind = :obj_kind and
              rlk_object_id = :ref_object_id and
              rlk_from_rd_id = :rd_id;
      end
    end
    else
    begin
      orig_value = '';
      sql_text = 'select ' || :ref_label_field || ' from ' || :ref_table_name || ' where ' || ref_key_field || '=' || :ref_object_id;
      execute statement :sql_text into orig_value;

      if (orig_value is null) then
        orig_value = '';

      if (ref_action_kind = 'add') then
      begin
        if (position('.' || ref_values || '.', orig_value) = 0) then
        begin
          if (right(orig_value, 1) <> '.') then
            orig_value = orig_value || '.';
          orig_value = orig_value || ref_values || '.';
        end
      end

      if (ref_action_kind = 'delete') then
      begin
        if (position('.' || ref_values || '.', orig_value) > 0) then
        begin
          orig_value = replace(orig_value, '.' || ref_values || '.', '.');
          if (orig_value = '.') then
            orig_value = '';
          if (orig_value <> '' and
              right(orig_value, 1) <> '.') then
            orig_value = orig_value || '.';
        end
      end

      sql_text = 'update ' || ref_table_name || ' set ' || ref_label_field || '=''' || orig_value || ''' where ' || ref_key_field || '=' || :ref_object_id;
      execute statement :sql_text;
    end
  end
end
