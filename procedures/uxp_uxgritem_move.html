CREATE PROCEDURE UXP$UXGRITEM_MOVE(
    GRID_UUID D_UUID,
    ITEM_CODE D_STR_CODE,
    NEW_PARENT_CODE D_STR_CODE,
    NEW_POSITION D_INTEGER)
AS
declare variable parent_is_band d_bool;
declare variable us_uuid d_uuid;
declare variable uf_uuid d_uuid;
begin
  parent_is_band = 0;

  if (coalesce(item_code, '') = '') then
    exit;

  if (new_parent_code similar to '[-]?[[:DIGIT:]]+') then
    parent_is_band = 1;

  if (item_code similar to '[-]?[[:DIGIT:]]+') then
  begin
    if (parent_is_band = 0) then
      exit;

    update ux$band
    set bnd_parent_id = :new_parent_code,
        bnd_position = iif(:new_position = -1, bnd_position, :new_position)
    where bnd_grid_uuid = :grid_uuid and
          bnd_id = :item_code;
  end
  else
  begin
    select ufr_us_uuid
    from ux$grid, ux$frame
    where ugr_uuid = :grid_uuid and
          ugr_ufr_uuid = ufr_uuid
    into us_uuid;

    if (parent_is_band = 0) then
    begin
      select ugi_bnd_id
      from ux$field, ux$gritem
      where uf_owner = :us_uuid and
            uf_field_name = :new_parent_code and
            ugi_ugr_uuid = :grid_uuid and
            ugi_uf_uuid = uf_uuid
      into new_parent_code;

      if (new_parent_code is null) then
        new_parent_code = -1;
    end

    select uf_uuid
    from ux$field
    where uf_owner = :us_uuid and
          uf_field_name = :item_code
    into uf_uuid;

    update ux$gritem
    set ugi_position = iif(:parent_is_band = 1, ugi_position, :new_position),
        ugi_bnd_id = :new_parent_code
    where ugi_ugr_uuid = :grid_uuid and
          ugi_uf_uuid = :uf_uuid;
  end
end
