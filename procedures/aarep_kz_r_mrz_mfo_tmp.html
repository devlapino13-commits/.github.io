CREATE PROCEDURE AAREP_KZ_R_MRZ_MFO_TMP(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    CLI_BALANCE D_CURRENCY,
    CLI_PERC_SALDO D_CURRENCY,
    CLI_CALC_FINE D_CURRENCY,
    BAL_STOEMOST D_CURRENCY,
    C_ID D_ID)
AS
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable OKS_BALANCE D_CURRENCY;
declare variable K_PERC_SALDO D_CURRENCY;
declare variable CLI_DOLY D_DOUBLE;
declare variable SH_CALC_FINE D_CURRENCY;
declare variable K_ID bigint;
begin

  for
  select k_id,
         k_amount_approved,
         oks_balance,
         f_perc_saldo(k_id, :rep_date_end)
  from oksummary, kontract, knclient, region_list(:rep_region_id,1,3,1)
  where :rep_date_end between oks_date_begin and oks_date_end
  and oks_close_state = 0
  and oks_from_k_id = k_id
  and k_id = kc_from_k_id
  and kc_amount_approved > 0
  and oks_region_id = id_list
  into k_id,
       k_amount_approved,
       oks_balance,
       k_perc_saldo
  do
  begin
--    select sum(sh_calc_fine) from aarep_make_zad_list(:k_id,:rep_date_end + 30,1)
--    into sh_calc_fine;
    sh_calc_fine = 0;
    cli_doly = null;
    c_name_fio = null;
    c_inn = null;
    cli_calc_fine = null;
    for
    select cast(kc_amount_approved as double precision) / :k_amount_approved,
           c_name_fio,
           c_inn,
           c_id
    from knclient, clients
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    and kc_from_c_id = c_id
    into  cli_doly, c_name_fio, c_inn, c_id
    do
    begin
      cli_balance = cli_doly * oks_balance;
      cli_perc_saldo = cli_doly * k_perc_saldo;
      cli_calc_fine = cli_doly * sh_calc_fine;

      bal_stoemost = cli_balance + cli_perc_saldo + cli_calc_fine;

      suspend;
    end 


  end
end
