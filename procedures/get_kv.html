CREATE PROCEDURE GET_KV(
    REG INTEGER,
    TAB INTEGER)
RETURNS (
    KV DECIMAL(18,0))
AS
begin
  /* Procedure Text */
  if (tab = 5) then
  begin
    kv = next value for gen_operlist_id;
    suspend;
    exit;
  end
  if (reg = 0) then
  begin
    reg = f_hq_id();
  end

  select kv_value
  from keyvalue
  where kv_from_reg_id = :reg and
        kv_table_no = :tab
  into :kv;
  if (kv is null) then
  begin
    kv = 1;
    insert into keyvalue
    values (:reg, :tab, :kv);
  end
  else
  begin
    in autonomous transaction do
    update keyvalue
    set kv_value = kv_value + 1
    where kv_from_reg_id = :reg and
          kv_table_no = :tab
    returning kv_value into kv;
  end
  if (tab in (8, 19)) -- Moratory и CellAktMain
  then
    kv = reg * 1000000000 + kv;
  else
    kv = reg * 1000000 + kv;
  suspend;
end
