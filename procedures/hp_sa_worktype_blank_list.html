CREATE PROCEDURE HP_SA_WORKTYPE_BLANK_LIST
RETURNS (
    SAWT_ID D_INTEGER,
    SAWT_SYMBOL D_STR_10,
    SAWT_HOURS_REAL D_CURRENCY,
    SAWT_NAME D_STR_100,
    SAWT_BASE_TYPE D_INTEGER,
    SAWT_INCLUDE_HOLIDAY D_SMALLINT,
    SAWT_IS_ORDER D_SMALLINT,
    SAWT_IS_PAID D_BOOL,
    SAWT_AUTO_APPROVE D_BOOL,
    SAWT_HIDE_IN_TABEL D_BOOL)
AS
begin
  for select sawt_id,
             sawt_symbol,
             sawt_hours_real,
             sawt_name,
             sawt_base_type,
             sawt_include_holiday,
             sawt_is_order,
             sawt_is_paid,
             sawt_auto_approve,
             sawt_hide_in_tabel
      from sa_worktype
      where sawt_id <> 0
      into :sawt_id,
           :sawt_symbol,
           :sawt_hours_real,
           :sawt_name,
           :sawt_base_type,
           :sawt_include_holiday,
           :sawt_is_order,
           :sawt_is_paid,
           :sawt_auto_approve,
           :sawt_hide_in_tabel
  do
  begin
    suspend;
  end
end
