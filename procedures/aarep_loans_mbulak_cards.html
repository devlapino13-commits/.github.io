CREATE PROCEDURE AAREP_LOANS_MBULAK_CARDS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    K_CIRCLE INTEGER,
    KF_NAME VARCHAR(100),
    KF_BRANCH_NAME VARCHAR(50),
    REG_NAME VARCHAR(50),
    PR_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(20),
    CONNECTION_TYPE VARCHAR(10),
    PARALLELISM_SIGN VARCHAR(20),
    PR_IS_SECURED_STR VARCHAR(20),
    PR_LOAN_TYPE_STR VARCHAR(20),
    C_TYPE_STR VARCHAR(20),
    ACCRUAL_STATUS VARCHAR(100),
    RESTRUCT_DATE_LAST D_DATE,
    KOL_RESTR INTEGER,
    PERCENT_CLASSIF VARCHAR(20),
    LLC_SUM_PERC CURR_TYPE,
    LLC_CURRENT_AMOUNT DECIMAL(18,4),
    CS_NAME VARCHAR(50),
    AAREP_K_DELIQ_DAYS_PERC INTEGER,
    AAREP_K_DELIQ_DAYS INTEGER,
    REMAINING_SURPLUS_CONTRIBUTED VARCHAR(10),
    SAGGING INTEGER,
    ZAD_PERC CURR_TYPE,
    KK_BALANCE DECIMAL(18,4),
    REG_KOD INTEGER,
    EXTRA_CHARGE_PERCENT DECIMAL(18,4),
    K_PERCENT_CT DECIMAL(18,4),
    CODE_BT_NAME INTEGER,
    K_PERCENT DECIMAL(18,4),
    KD_PERCENT DECIMAL(18,4),
    KC_AMOUNT_APPLICATE DECIMAL(18,4),
    K_KREDLINE_END D_DATE,
    CREDIT_CARD_ID D_INTEGER,
    DATE_CREDIT_CARD D_DATE,
    KS_DATE_BEGIN D_DATE,
    CREDIT_CARD_HOLDER D_STR_100,
    C_NAME_FIO D_STR_100,
    KC_LOAN_TARGET_STR VARCHAR(100),
    REG_STAT_REP_REGION VARCHAR(100),
    BT_NAME VARCHAR(100),
    C_ID D_STR_20,
    K_DISB_KIND D_STR_20)
AS
declare variable K_ID decimal(18,0);
declare variable LK_PERCENT decimal(18,4);
declare variable K_INDEX_CURR_RATE CURR_TYPE;
declare variable KC_LOAN_TARGET integer;
declare variable K_STOP_PERC D_DATE;
declare variable C_TYPE integer;
declare variable PR_LOAN_TYPE integer;
declare variable PR_IS_SECURED integer;
declare variable K_KONTRACT_NO_2 varchar(20);
declare variable K_FROM_KF_ID decimal(18,0);
declare variable PR_CODE varchar(100);
declare variable REP_PERIOD integer;
declare variable KC_BUSINESS_TYPE integer;
declare variable CR_RATE CURR_TYPE;
declare variable SCORE_RATE integer;
declare variable REP_DELIQ_DAYS_KOLICH integer;
declare variable AAREP_PERC_END decimal(18,4);
declare variable LK_BALANCE CURR_TYPE;
declare variable LK_CALCPERC CURR_TYPE;
declare variable REP_PERIOD_DATE_BEGIN integer;
declare variable KS_REGION_ID integer;
declare variable K_DISB_DATE D_DATE;
declare variable STATUS_ISSUED D_INTEGER;
declare variable STATUS_PIN D_INTEGER;
declare variable DB_ID D_INTEGER;
begin
  -- RFCKG-3540
  db_id = f_get_db_id();

  restruct_date_last = null;
  kol_restr = 0;
  status_issued = f_enum('cardaktclient', 'issued', 'id');
  status_pin = f_enum('cardaktclient', 'pin_assigned', 'id');

  for select kon.k_id, cl.c_id, cl.c_name_fio, ks.ks_date_begin, kon.k_percent, reg.reg_stat_rep_region, cl.c_type,
             pr.pr_loan_type, pr.pr_is_secured, kon.k_kontract_no, pr.pr_name, reg.reg_name,
             r2.reg_code || ' ' || r2.reg_name, kon.k_from_kf_id, kl.kc_circle, kc_loan_target, pr_code,
             kl.kc_business_type, kon.k_amount_applicate, kon.k_kredline_end, kon.k_index_curr_rate, kon.k_stop_perc,
             kon.k_disb_date,
             iif (kon.k_disb_kind = 6, 'Банк.карта МБулак', '')
      from kontstate ks
      inner join kontract kon on kon.k_id = ks.ks_from_k_id
      inner join product pr on pr.pr_id = kon.k_from_pr_id
      inner join knclient kl on kl.kc_from_k_id = kon.k_id
      inner join clients cl on cl.c_id = kl.kc_from_c_id
      inner join region reg on reg.reg_unicode = ks.ks_region_id
      left join region r2 on reg.parent = r2.reg_unicode
      inner join region_list(:rep_region_id, 1, 3, 1) on kon.k_last_reg_id = id_list
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
            kon.k_disb_kind = 6

           into :k_id, :c_id, :c_name_fio, :ks_date_begin, :k_percent, :reg_stat_rep_region, :c_type, :pr_loan_type,
           pr_is_secured, k_kontract_no, pr_name, reg_name, kf_branch_name, k_from_kf_id, k_circle, kc_loan_target,
           pr_code, kc_business_type, :kc_amount_applicate, :k_kredline_end, :k_index_curr_rate, :k_stop_perc, k_disb_date,
           k_disb_kind

  do
  begin
    lk_balance = 0;
    lk_calcperc = 0;
    lk_percent = 0;
    credit_card_id = 0;
    date_credit_card = null;
    credit_card_id = (case
                        when k_id = 109010139192800101 then 2015
                        when k_id = 109010138773200101 then 2028
                        when k_id = 109010138786400101 then 2030
                        when k_id = 109010139194700101 then 2043
                      end);

    date_credit_card = (case
                          when k_id = 109010139192800101 then '28.04.2021'
                          when k_id = 109010138773200101 then '23.04.2021'
                          when k_id = 109010138786400101 then '23.04.2021'
                          when k_id = 109010139194700101 then '28.04.2021'
                        end);

    select cal.cal_card_id, cac.cac_date
    from cardaktclient cac
    inner join cardclientacc cca on cca.cca_id = cac.cac_from_cca_id
    inner join cardlist cal on cca.cca_id = cal.cal_from_cca_id
    where cac.cac_status in (:status_issued, :status_pin) and
          cac.cac_from_c_id = :c_id and
          cac.cac_date <= :ks_date_begin
    order by cac.cac_date desc
    rows 1
    into :credit_card_id, :date_credit_card;

    select rd_name
    from refdata
    where rd_id = :kc_loan_target
    into :kc_loan_target_str;

    k_kontract_no_2 = null; --получаем паралелиный активный кредит
    select kon.k_kontract_no
    from kontstate k
    inner join knclient kc on kc.kc_from_k_id = k.ks_from_k_id
    inner join kontract kon on kon.k_id = k.ks_from_k_id
    where kc.kc_from_c_id = :c_id and
          k.ks_ol_is_delete = 0 and
          k.ks_state = 1 and
          k.ks_from_k_id <> :k_id and
          :rep_date_end between k.ks_date_begin and k.ks_date_end rows 1
    into :k_kontract_no_2;

    if (k_kontract_no_2 is not null) then
      parallelism_sign = 'параллельный';
    else
      parallelism_sign = 'непараллельный';

    if (llc_sum_perc > 0) then
      percent_classif = 'стандартные';
    if (llc_sum_perc = 0) then
      percent_classif = '';

    if (k_disb_date >= '01.09.2021' and
        k_disb_date <= '28.11.2021') then --сопоставляем проценты по ишеним
    begin
      if (pr_code containing '.consultation.') then
      begin
        k_percent_ct = (case
                          when k_percent = 34 then 36
                          when k_percent = 32 then 34
                          when k_percent = 30 then 32
                          when k_percent = 28 then 30
                          when k_percent = 26 then 28
                          when k_percent = 24 then 26
                          when k_percent = 22 then 24
                          when k_percent = 21 then 23
                          when k_percent = 20 then 22
                          when k_percent = 19 then 21
                          when k_percent = 18 then 20
                          when k_percent = 17 then 19
                        end);
      end
      else
        k_percent_ct = (case
                          when k_percent = 46 then 48
                          when k_percent = 44 then 46
                          when k_percent = 42 then 44
                          when k_percent = 40 then 42
                          when k_percent = 38 then 40
                          when k_percent = 36 then 38
                          when k_percent = 34 then 36
                          when k_percent = 32 then 34
                          when k_percent = 30 then 32
                          when k_percent = 28 then 30
                          when k_percent = 27 then 29
                          when k_percent = 26 then 28
                        end);
    end
    if (k_disb_date >= '29.11.2021') then --сопоставляем проценты по ишеним
    begin
      if (pr_code containing '.consultation.') then
      begin
        k_percent_ct = (case
                          when k_percent = 34 then 36
                          when k_percent = 33 then 35
                          when k_percent = 31 then 33
                          when k_percent = 29 then 31
                          when k_percent = 28 then 30
                          when k_percent = 26 then 28
                          when k_percent = 24 then 26
                          when k_percent = 23 then 25
                          when k_percent = 22 then 24
                          when k_percent = 21 then 23
                          when k_percent = 20 then 22
                          when k_percent = 19 then 21
                        end);
      end
      else
        k_percent_ct = (case
                          when k_percent = 46 then 48
                          when k_percent = 44 then 46
                          when k_percent = 43 then 45
                          when k_percent = 42 then 44
                          when k_percent = 40 then 42
                          when k_percent = 38 then 40
                          when k_percent = 36 then 38
                          when k_percent = 34 then 36
                          when k_percent = 32 then 34
                          when k_percent = 30 then 32
                          when k_percent = 29 then 31
                          when k_percent = 28 then 30
                        end);

      if (pr_code containing '.score7.') then
      begin
        k_percent_ct = (case
                          when k_percent = 34 then 36
                          when k_percent = 33 then 35
                          when k_percent = 31 then 33
                          when k_percent = 29 then 31
                          when k_percent = 28 then 30
                          when k_percent = 26 then 28
                          when k_percent = 24 then 26
                          when k_percent = 23 then 25
                          when k_percent = 22 then 24
                          when k_percent = 21 then 23
                          when k_percent = 20 then 22
                          when k_percent = 19 then 21
                        end);
      end
    end

    if (k_disb_date <= '31.08.2021') then
    begin
      if (pr_code containing '.consultation.') then
      begin
        k_percent_ct = case
                         when :k_circle >= 6 then 26.0
                         when :k_circle between 4 and 5 then 29.0
                         when :k_circle between 2 and 3 then 32.0
                         else 35
                       end;
      end
      else
        k_percent_ct = case
                         when :k_circle >= 6 then 48.0
                         when :k_circle between 4 and 5 then 48.0
                         when :k_circle between 2 and 3 then 48.0
                         else 48
                       end;

    end
    extra_charge_percent = 0;
    extra_charge_percent = k_percent - k_percent_ct; --справочник нбкр

    if (reg_stat_rep_region = 'ысыкульская область') then
      reg_kod = 02;

    if (reg_stat_rep_region = 'жалалабадская область') then
      reg_kod = 03;

    if (reg_stat_rep_region = 'нарынская область') then
      reg_kod = 04;

    if (reg_stat_rep_region = 'баткенская область') then
      reg_kod = 05;

    if (reg_stat_rep_region = 'ошская область' or reg_stat_rep_region = 'г.ош') then
      reg_kod = 06;

    if (reg_stat_rep_region = 'таласская область') then
      reg_kod = 07;

    if (reg_stat_rep_region = 'чуйская область' or reg_stat_rep_region = 'г.бишкек') then
      reg_kod = 08;

    pr_loan_type_str = '';
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'группа';
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'индив.';

    pr_is_secured_str = '';
    if (pr_is_secured = 0) then
      pr_is_secured_str = 'без залога';
    if (pr_is_secured = 1) then
      pr_is_secured_str = 'залог';
    if (pr_is_secured = 2) then
      pr_is_secured_str = 'поручители';
    if (pr_is_secured = 3) then
      pr_is_secured_str = 'залог,поручит.';

    kk_balance = 0;
    select oks_balance -- получаем баланс кредита
    from oksummary
    where oks_date_begin between '01.01.1900' and :rep_date_end and
          oks_date_end between(:rep_date_end) and '31.12.9999' and
          oks_from_k_id = :k_id and
          oks_close_state = 0
    into :kk_balance;

    zad_perc = 0; -- накопленные  %
    select sum(sh_calc_perc)
    from aarep_make_zad_list(:k_id, :rep_date_end, 1)
    where sh_date is not null
    into :zad_perc;

    sagging = 0;
    aarep_k_deliq_days = 0;

    select :rep_date_end - oks_delinq_date --количество просроченных дней
    from oksummary oks
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_from_k_id = :k_id and
          oks_close_state = 0 and
          oks_delinq_date < :rep_date_end
    into aarep_k_deliq_days;
    aarep_k_deliq_days = coalesce(aarep_k_deliq_days, 0);

    aarep_k_deliq_days_perc = :aarep_k_deliq_days;

    select restruct_kol, restruct_date_last -- < получаем количество реструктуризаций
    from aarep_get_restructuring(:k_id, :rep_date_end, 41701)
    into :kol_restr, :restruct_date_last;

    accrual_status = 'статус начисления';
    if (k_stop_perc is not null) then
      accrual_status = 'статус неначисления';
    if (k_stop_perc > rep_date_end) then
      accrual_status = 'статус начисления';

    c_type_str = 'физ. лицо';
    if (c_type = 1) then
      c_type_str = 'юр. лицо';

    select kal.kal_new_kf_name
    from knaktlist kal
    where :rep_date_end between kal.kal_date and kal.kal_date_end and
          kal.kal_from_kf_id = :k_from_kf_id and
          kal.is_delete = 0
    into :kf_name;

    bt_name = null;
    select bt_name
    from knclient kc, bussubtype, bustype
    where kc_from_c_id = :c_id and
          kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          kc_busuness_subtype = bs_id and
          bs_from_bt_id = bt_id
    into bt_name;

    if (bt_name like 'торговля') then --справочник нбкр
      code_bt_name = 05;

    if (bt_name like 'услуги') then
      code_bt_name = 11;

    if (bt_name like 'производство') then
      code_bt_name = 01;

    if (bt_name like 'сельское хозяйство' or bt_name like 'животноводство') then
      code_bt_name = 02;

    if (bt_name like 'другое' or bt_name like 'физические лица зарплата') then
      code_bt_name = 08;

    credit_card_holder = :c_name_fio;
    remaining_surplus_contributed = ''; --сумма/остаток излише внесенных на кредитную карту денежных средств в рамках срр (при погашении на отчетную дату)
    connection_type = ''; --тип связанности/аффилированности/инсайдер
    -- ниже указанные поля по прозьбе назгуль оставлены пустыми, данные по 4 полям будет брать с отчета рппу
    llc_sum_perc = 0;
    llc_current_amount = 0;
    lk_calcperc = 0;
    cs_name = '';

    kd_percent = 0;

    select kd.kd_percent
    from kontdata kd
    where kd.kd_from_k_id = :k_id and
          :rep_date_end between kd.kd_ol_date and kd.kd_date_end and
          kd.kd_ol_is_delete = 0
    order by kd_ol_date desc
    rows 1
    into :kd_percent;

    if (ks_date_begin between '15.12.2022' and '19.05.2024') then
    begin
      k_percent_ct = k_percent;
      extra_charge_percent = k_percent - k_percent_ct;
    end

    -- RFCKG-2426
    if (:ks_date_begin >= '20.05.2024') then
    begin
      k_percent_ct = k_percent + 1;
      extra_charge_percent = k_percent - k_percent_ct;
    end

    suspend;
  end
end
