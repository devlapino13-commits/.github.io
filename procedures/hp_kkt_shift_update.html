CREATE PROCEDURE HP_KKT_SHIFT_UPDATE(
    ACTION_CODE D_STR_CODE NOT NULL,
    FN_NUMBER D_STR_20 NOT NULL,
    FISC_DATETIME D_DATETIME NOT NULL,
    KKD_CODE D_STR_CODE NOT NULL,
    SHIFT_NUMBER D_INTEGER NOT NULL)
AS
declare variable kkf_id d_integer;
  declare variable kkd_id d_integer;
  declare variable kks_id d_id;
  declare variable error_text d_str_255;
  declare variable msg_subject d_str_255;
  declare variable msg_text d_str_long;
begin
  select kkf_id
  from kkt_fn
  where kkf_code = :fn_number
  into kkf_id;

  if (kkf_id is null) then
  begin
    select kkd_id
    from kkt_devices
    where kkd_code = :kkd_code
    into kkd_id;

    if (kkd_id is null) then
    begin
      error_text = 'Не найдена ККТ';
      exception e_default 'not found KKT. kkd_code = ' || coalesce(kkd_code, '<NULL>');
    end

    insert into kkt_fn (kkf_from_kkd_id, kkf_code)
    values (:kkd_id, :fn_number)
    returning kkf_id
    into :kkf_id;
  end

  select kks_id
  from kkt_shift
  where kks_from_kkf_id = :kkf_id
    and kks_number = :shift_number
  into kks_id;

  if (kks_id is null) then
    insert into kkt_shift (kks_from_kkf_id, kks_number)
    values (:kkf_id, :shift_number)
    returning kks_id
    into kks_id;

  if (action_code = 'open') then
    update kkt_shift
    set kks_open_datetime = current_timestamp,
        kks_open_datetime_fisc = :fisc_datetime,
        kks_close_datetime = null,
        kks_close_datetime_fisc = null
    where (kks_id = :kks_id);

  else if (action_code = 'close') then
    update kkt_shift
    set kks_close_datetime = current_timestamp,
        kks_close_datetime_fisc = :fisc_datetime
    where (kks_id = :kks_id);

  else
  begin
    error_text = 'Неизвестное действие:' || action_code;
    exception e_default 'Unknown action action_code = ' || action_code;
  end 

  when any do
  begin
    msg_subject = 'ККТ: Ошибка в hp_kkt_shift_update.';
    msg_text  = 'Ошибка в hp_kkt_shift_update: ' || coalesce(error_text, 'gdscode: ' || gdscode) || ascii_char(10) || ascii_char(13) ||
                'Параметры:' || ascii_char(10) || ascii_char(13) ||
                ' action_code = ' || coalesce(action_code, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' fn_number = ' || coalesce(fn_number, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' fisc_datetime = ' || coalesce(fisc_datetime, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' kkd_code = ' || coalesce(kkd_code, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' shift_number = ' || coalesce(shift_number, '<NULL>');
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind)
    values (0, 'ru.finstart@molbulak.com', :msg_subject, :msg_text, 7, 1000000000001, '');
  end 

end
