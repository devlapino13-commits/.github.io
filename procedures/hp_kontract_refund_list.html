CREATE PROCEDURE HP_KONTRACT_REFUND_LIST(
    DISB_K_ID D_ID,
    DISB_KC_ID D_ID = 0)
RETURNS (
    RES_REFUNDED_K_ID D_ID,
    RES_REFUNDED_DELINQ_DAY D_INTEGER,
    RES_REFUNDED_ADV_DAY D_INTEGER)
AS
declare variable ks_date_begin d_date;
declare variable ol_id d_id;
declare variable max_sh_date d_date;
declare variable min_sh_date d_date;
declare variable disb_ol_datetime d_datetime;
begin
  if (disb_kc_id is null) then
    disb_kc_id = 0;

  select ks_date_begin, ol_datetime
  from kontstate, operlist
  where ks_from_k_id = :disb_k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_from_ol_id = ol_id
  into ks_date_begin, disb_ol_datetime;

  if (ks_date_begin is null) then
    exit;

  for select distinct ks_from_k_id, ks_from_ol_id
  from knclient kn, kontstate, knclient r, operlist
  where kn.kc_from_k_id = :disb_k_id and
        kn.kc_amount_approved > 0 and
        (:disb_kc_id = 0 or kn.kc_id = :disb_kc_id) and
        r.kc_from_c_id = kn.kc_from_c_id and
        r.kc_amount_approved > 0 and
        r.kc_date_close = (:ks_date_begin - 1) and
        r.kc_from_k_id = ks_from_k_id and
        ks_state = 2 and
        ks_ol_is_delete = 0 and
        ks_date_begin = :ks_date_begin and
        ks_from_ol_id = ol_id and
        ol_datetime >= :disb_ol_datetime
  into res_refunded_k_id, ol_id
  do
  begin
    res_refunded_delinq_day = 0;
    min_sh_date = null;
    max_sh_date = null;

    select min(ok_sh_date), max(ok_sh_date)
    from operkred
    where ok_from_ol_id = :ol_id and
          ok_from_k_id = :res_refunded_k_id and
          ok_type = 4 and
          ok_sh_date <> ok_ol_date and
          (ok_os <> 0 or ok_perc <> 0)
    into min_sh_date, max_sh_date;

    if (min_sh_date < ks_date_begin or
        max_sh_date > ks_date_begin) then
    begin
      if (min_sh_date > ks_date_begin) then
        min_sh_date = ks_date_begin;

      res_refunded_delinq_day = datediff(day, min_sh_date, ks_date_begin);

      if (max_sh_date < ks_date_begin) then
        max_sh_date = ks_date_begin;

      res_refunded_adv_day = datediff(day, ks_date_begin, max_sh_date);

      suspend;
    end
  end
end
