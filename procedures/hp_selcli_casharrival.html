CREATE PROCEDURE HP_SELCLI_CASHARRIVAL(
    K_ID D_ID,
    DT D_DATE = current_date)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    PREPAYMENT_SUMMA D_CURRENCY,
    IPF_COMMISSION D_CURRENCY,
    AMOUNT_CONSULTING D_CURRENCY,
    SIGN_CONSULTING D_BOOL)
AS
declare variable is_ipf d_bool;
declare variable kc_id d_id;
declare variable kc_status d_integer;
begin
  for
    select
      iif(pr_code containing '.ipf.', 1, 0),
      coalesce(k_prepayment_summa,0),
      coalesce(pr_ipf_commission,0)
    from kontract
      inner join product
        on pr_id = k_from_pr_id
    where k_id = :k_id
  into
    :is_ipf,
    :prepayment_summa,
    :ipf_commission
  do
  begin
    for
      select
        kc_id,
        kc_status,
        c_id,
        c_name_fio
      from knclient
        inner join clients
          on c_id = kc_from_c_id
      where kc_from_k_id = :k_id
        and kc_amount_approved > 0
    into
      :kc_id,
      :kc_status,
      :c_id,
      :c_name_fio
    do
    begin
      if (is_ipf = 1) then
      begin
        amount_consulting = 0;
        sign_consulting = 0;
        if (kc_status <> 2) then continue;
      end else
      begin
        amount_consulting =
          (select price from clt_get_price_for_consultation(:kc_id));
        sign_consulting = iif(f_edoc_status('consulting.agreement', :kc_id, :dt) = 'signed', 1, 0);  -- RFCKG-4363

        if (sign_consulting = 0) then
        begin
            if (not exists(select * from hp_db_is_actual)) then
                sign_consulting = 1; -- если БД тестовая, тогда не проверяю подписан ли договор на оказание консультационных услуг или нет.
        end
      end
      suspend;
    end
  end
end
