Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXALIASACTION_MARK_ALL(
    UA_UUID D_UUID,
    UXA_UUID_LIST D_STR_4096,
    ACTION_KIND D_SMALLINT)
AS
declare variable tmp_uuid d_uuid;
declare variable res_str d_str_code;
begin
  if (action_kind not in (-1, 1)) then
    exit;

  if (ua_uuid is null) then
    exit;

  for select res_str
  from uxp$split_str(:uxa_uuid_list, ',')
  into :res_str
  do
  begin
    tmp_uuid = char_to_uuid(res_str);

    if (action_kind = -1) then
    begin
      delete from ux$aliasaction
      where uaa_uxa_uuid = :tmp_uuid and
            uaa_ua_uuid = :ua_uuid;
    end else
    begin
      update or insert into ux$aliasaction (uaa_uxa_uuid, uaa_ua_uuid)
      values (:tmp_uuid, :ua_uuid)
      matching (uaa_uxa_uuid, uaa_ua_uuid);
    end
  end
end
