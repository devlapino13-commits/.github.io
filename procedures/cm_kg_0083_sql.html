CREATE PROCEDURE CM_KG_0083_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable KF_CLIENT_PASS_NO varchar(30);
declare variable C_ID ID_TYPE;
declare variable KC_BALANCE CURR_TYPE;
declare variable PR_CODE D_STR_100;
begin
 if (K_ID = 109010264931500101) then exit;
 if (K_ID = 109010258413400101) then exit;
 if (k_id = 109010223769500101) then exit;
 if (K_ID = 109010259756400101) then exit; ---RFCKG-1240
 if (k_id = 109010227421600101) then exit; --jira 283
 if (k_id = 109010233171200101) then exit; --RFCKG-490
 if (K_ID = 109010239773700101) then exit; ---RFCKG-613
 if (k_id = 109010242287300101) then exit; -- RFCKG-689
 if (k_id = 109010244875700101) then exit; --RFCKG-749
 if (k_id = 109010245540400101) then exit; --RFCKG-768
 if (k_id = 109010247683300101) then exit; --RFCKG-819
 if (k_id = 109010248810200101) then exit; --RFCKG-847
 if (k_id = 109010249331800101) then exit; --RFCKG-878
 if (K_ID = 109010251969000101) then exit; -- RFCKG-934
 if (k_id = 109010259541700101) then exit; -- RFCKG-1223
 if (k_id = 109010294303800101) then exit; --RFCKG-2905

  for
  select kc_from_c_id, kc_amount_approved, k_disb_date, pr_code
  from knclient, kontract, product
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_from_k_id = k_id
  and k_from_pr_id = pr_id
  into c_id, kc_amount, kf_disb_date, pr_code
  do
  begin
    if (pr_code containing '.point_ignore.' and
        pr_code containing '.insurance.') then
      exit;
    kc_balance = null;
    select sum(cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance)
    from knclient, kontract, oksummary
    where kc_from_c_id = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id <> :k_id
    and kc_from_k_id = k_id
    and k_id = oks_from_k_id
    and :kf_disb_date between oks_date_begin and oks_date_end
    and oks_close_state = 0
    and oks_balance > 0
    into kc_balance;
    kc_balance = coalesce(kc_balance,0);
    kc_balance = round(kc_balance,0);
    -- При проверке RFCKG-3745 выявили, что данная сумма, должна быть БОЛЬШЕ 500000
    if (kc_amount + kc_balance > 500000.0) then
    begin
      select c_name_fio, c_pass_no from clients
      where c_id = :c_id
      into res_info_text, kf_client_pass_no;
      res_error_text = 'У клиента совокупный баланс задолженности превышает 500 тыс. сомов.';
      res_error_is_critical = 1;

      if(k_id in(109010061906800101, 109010055961600101,109010074802000101, 109010055961600201)) then 
        res_error_is_critical = 0;
      suspend;
    end
  end
end
