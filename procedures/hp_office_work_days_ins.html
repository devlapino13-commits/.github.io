CREATE PROCEDURE HP_OFFICE_WORK_DAYS_INS(
    OFFICE_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    SOWD_DAYS D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_255,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable current_days d_integer;
declare variable str d_str_100;
begin
  -- создает новый режим работы офиса - кол-во рабочих дней в неделю (RFCKG-2502)

  res_info_text = '';
  res_error_text = '';

  if (not exists(select *
                 from region
                 where reg_unicode = :office_id)) then
    res_error_text = 'Ошибка: офис не найден';

  if (not exists(select *
                 from username
                 where un_id = :un_id)) then
    res_error_text = 'Ошибка: пользователь не найден';

  select sowd_days
  from sa_office_working_days
  where sowd_from_reg_id = :office_id and
        current_date between sowd_date_begin and sowd_date_end
  into current_days;

  if (sowd_days = current_days) then
    res_error_text = 'Ошибка: новый режим не отличается от текущего';

  if (coalesce(sowd_days, 0) not between 5 and 7) then
    res_error_text = 'Ошибка: кол-во рабочих дней должно быть в диапазоне 5-7';

  if (date_begin is null) then
    res_error_text = 'Ошибка: не указана дата начала действия';

  if (date_begin < current_date) then
    res_error_text = 'Ошибка: дата начала действия не может быть прошедшей датой';

  if (res_error_text = '') then
  begin
    -- удаляем режим на будущую дату (если есть)
    select 'Удален режим: дней ' || sowd_days || ', действует с ' || f_datetostr(sowd_date_begin)
    from sa_office_working_days
    where sowd_from_reg_id = :office_id and
          sowd_date_begin > current_date
    into str;

    delete from sa_office_working_days
    where sowd_from_reg_id = :office_id and
          sowd_date_begin > current_date;

    if (row_count = 0) then
      str = '';

    -- добавляем новый режим
    insert into sa_office_working_days (sowd_from_reg_id, sowd_days, sowd_date_begin, sowd_editor_un_id)
    values (:office_id, :sowd_days, :date_begin, :un_id);

    if (row_count = 1) then
    begin
      if (str <> '') then
        res_info_text = str || ascii_char(13) || ascii_char(10) || ascii_char(13) || ascii_char(10);
      res_info_text = res_info_text || 'Добавлен режим: дней ' || sowd_days || ', действует с ' || f_datetostr(date_begin);
    end
    else
      res_error_text = 'Ошибка: добавить новый режим не удалось';
  end

  suspend;
end
