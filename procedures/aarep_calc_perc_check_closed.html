CREATE PROCEDURE AAREP_CALC_PERC_CHECK_CLOSED(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_MAX_FETCH D_INTEGER = 0)
RETURNS (
    REC_NO D_INTEGER,
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    K_ID D_ID,
    DIFF_KIND D_STR_CODE,
    DIFF_PERC_SALDO D_CURRENCY,
    KCP_PERC_SALDO D_CURRENCY,
    KCP_PERC_CALCED D_CURRENCY,
    NEW_PERC_SALDO D_CURRENCY,
    NEW_PERC_CALCED D_CURRENCY,
    K_STOP_PERC D_DATE,
    K_KREDLINE_END D_DATE,
    DATE_CLOSE D_DATE)
AS
declare variable dd_begin d_date;
begin
  if (rep_max_fetch is null) then
    rep_max_fetch = 0;

  rec_no = 0;
  for select ks_from_k_id, ks_date_begin
  from kontstate, kontract, product
  where ks_state = 2 and
        ks_ol_is_delete = 0 and
        ks_date_begin between :rep_date_begin and :rep_date_end and
        ks_from_k_id = k_id and
        k_from_pr_id = pr_id and
        pr_percent_kind <> 2
  into k_id, date_close
  do
  begin
    new_perc_calced = 0;
    new_perc_saldo = 0;
    kcp_perc_saldo = 0;
    kcp_perc_calced = 0;

    dd_begin = dateadd(day, -(extract(day from date_close) - 1), date_close);

    select result_calced, result_perc_end
    from aarep_calc_perc_one(:k_id, :dd_begin, :date_close, 1, null, null, null, null)
    into kcp_perc_calced, kcp_perc_saldo;

    select result_calced, result_perc_end
    from aarep_calc_perc_one_cls(:k_id, :dd_begin, :date_close, 0, 1, 0)
    into new_perc_calced, new_perc_saldo;

    diff_perc_saldo = new_perc_saldo - kcp_perc_saldo;
    if (abs(diff_perc_saldo) > 1) then
    begin
      select k_kontract_no, k_disb_date, k_stop_perc, k_kredline_end
      from kontract
      where k_id = :k_id
      into k_kontract_no, k_disb_date, k_stop_perc, k_kredline_end;


      if (abs(diff_perc_saldo) <= 1) then
        diff_kind = 'A. <= 1';
      else
      if (abs(diff_perc_saldo) <= 5) then
        diff_kind = 'B. <= 5';
      else
      if (abs(diff_perc_saldo) <= 10) then
        diff_kind = 'C. <= 10';
      else
      if (abs(diff_perc_saldo) <= 50) then
        diff_kind = 'D. <= 50';
      else
      if (abs(diff_perc_saldo) <= 500) then
        diff_kind = 'E. <= 500';
      else
      if (abs(diff_perc_saldo) <= 1000) then
        diff_kind = 'F. <= 1000';
      else
        diff_kind = 'G. > 1000';

      rec_no = rec_no + 1;
      suspend;

      if (rep_max_fetch > 0 and rec_no = rep_max_fetch) then
        exit;
    end
  end
end
