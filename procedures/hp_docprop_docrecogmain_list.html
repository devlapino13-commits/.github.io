CREATE PROCEDURE HP_DOCPROP_DOCRECOGMAIN_LIST(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    DP_ID D_INTEGER)
RETURNS (
    DRM_ID D_INTEGER,
    DRM_NAME D_STR_255,
    C_ID D_ID)
AS
declare variable db_id d_integer;
declare variable citizen_code d_str_code;
declare variable citizen_kind d_smallint;
declare variable passport_dp_id d_integer;
declare variable doc_count d_integer;
declare variable db_code d_str_code;
begin
  obj_kind = f_objkind(obj_kind, 'code');
  db_id = f_get_db_id();
    
  if (obj_kind = 'clients') then
    c_id = obj_id;
  else
  if (obj_kind = 'kontract') then
  begin
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :obj_id and
          kc_amount_approved > 0
    rows 1
    into c_id;
  end else
  if (obj_kind = 'miniapp') then
  begin
    select mac_from_c_id
    from miniappclient
    where mac_from_ma_id = :obj_id and
          mac_amount > 0
    rows 1
    into c_id;
  end

  if (c_id is not null) then
  begin
    -- Свойство документа по которому документ считается паспортом.
    select dp_id
    from docprop
    where dp_code = 'person.passport'
    into passport_dp_id;
  
    if (passport_dp_id is not null) then
    begin
      -- Среди действующих на дату документов выбираем по пориоритету:
      -- Сначала паспорт текущей страны
      -- если такого нет, выбираем последнюю по дате прикрепления
      select dpk_citizen_kind, pk_country
      from clientdocs, docproppk, passkind
      where dpk_from_dp_id = :passport_dp_id and
            dpk_is_delete = 0 and
            cds_from_pk_id = dpk_from_pk_id and
            dpk_from_pk_id = pk_id and
            cds_from_c_id = :c_id and
            cds_verify_status = 2 and
            coalesce(cds_create_datetime, cds_datetime) < (current_date + 1) and
            (cds_delete_datetime is null or cds_delete_datetime > (current_date + 1)) and
            cds_doc_date_begin <= current_date and
            coalesce(cds_doc_date_end, current_date) >= current_date
      order by dpk_citizen_kind, coalesce(cds_create_datetime, cds_datetime) desc, cds_doc_date_end desc
      rows 1
      into citizen_kind, citizen_code;
    
      -- Если нет прикрепленного действующего паспорта,
      -- смотрим поле "Гражданство" и по нему вычисляем тип гражданства
      if (citizen_kind is null) then
      begin
        select replace(rd_code, 'citizen.', '')
        from clients, refdata
        where c_id = :c_id and
              c_citizen = rd_id
        into citizen_code;
    
        if ((db_id = 81001 and citizen_code = 'ru') or
            (db_id in (39801, 41701) and citizen_code = 'kg') or
            (db_id = 94401 and citizen_code = 'az'))
        then citizen_kind = 0;
        else
          if ((db_id = 81001 and citizen_code = 'kg') or
              (db_id in (39801, 41701) and citizen_code = 'ru'))
          then citizen_kind = 1;
          else citizen_kind = 2;
      end
    end
  end else
    exit;

  db_code = trim(case db_id
    when 41701 then 'kg'
    when 39801 then 'kz'
    when 81001 then 'ru'
    when 94401 then 'az'
  end);

  if (citizen_code is null) then
    citizen_code = db_code;

  doc_count = 0;
  select distinct count(*)
  from docprop, docproppk, passkind, docrecogmain
  where dp_id = :dp_id and
        dpk_from_dp_id = dp_id and
        dpk_is_delete = 0 and
        dpk_is_delete = 0 and
        pk_id = dpk_from_pk_id and
        pk_is_delete = 0 and
        pk_from_drm_id = drm_id and
        drm_inactive = 0 and
        (pk_country starting :citizen_code or pk_country starting :db_code) and
        (:citizen_kind is null or dpk_citizen_kind = :citizen_kind)
  into doc_count;

  if (doc_count > 1) then
  begin
    drm_id = -1;
    drm_name = 'Любой документ из списка';
    suspend;
  end

  for select distinct drm_id, drm_name
  from docprop, docproppk, passkind, docrecogmain
  where dp_id = :dp_id and
        dpk_from_dp_id = dp_id and
        dpk_is_delete = 0 and
        dpk_is_delete = 0 and
        pk_id = dpk_from_pk_id and
        pk_is_delete = 0 and
        pk_from_drm_id = drm_id and
        drm_inactive = 0 and
        (pk_country starting :citizen_code or pk_country starting :db_code) and
        (:citizen_kind is null or dpk_citizen_kind = :citizen_kind)
  order by dp_position, pk_id
  into drm_id, drm_name
  do
    suspend;
end
