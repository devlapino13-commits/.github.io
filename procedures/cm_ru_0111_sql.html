CREATE PROCEDURE CM_RU_0111_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)

declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable K_KOLICH_REPAY integer;
declare variable PR_CODE varchar(20);
declare variable KC_FROM_C_ID ID_TYPE;
declare variable K_KOMMIS_PERC PERC_TYPE;
declare variable KOL_CLI_GROUP integer;
declare variable PR_ID integer;
declare variable K_PERCENT PERC_TYPE;
declare variable C_BIRTH_DATE date;
declare variable C_CITIZEN integer;
declare variable MIGR_DATE date;
declare variable CON_RF integer;
declare variable CON_ROD integer;
declare variable PATENT_DATE date;
declare variable K_KREDLINE_END date;
declare variable C_LAST_STATUS smallint;
declare variable K_LAST_REG_ID ID_TYPE;
begin
  --ПРОДУКТ «МИКРОЗАЕМ НА ПАТЕНТ»
  for select K_AMOUNT_APPROVED, K_KOLICH_REPAY, PR_CODE, K_DISB_DATE, K_KOMMIS_PERC, K_KOLICH_CLIENT_APPROVED, PR_ID,
             K_PERCENT, K_LAST_REG_ID, K.K_KREDLINE_END
      from KONTRACT K, PRODUCT
      where K_ID = :K_ID and
            K_FROM_PR_ID = PR_ID and
            PR_CODE containing '.IND.' and
            PR_CODE containing '.PATENT.' --Микрозаем на патент
      into K_AMOUNT_APPROVED, K_KOLICH_REPAY, PR_CODE, KF_DISB_DATE, K_KOMMIS_PERC, KOL_CLI_GROUP, PR_ID, K_PERCENT,
           K_LAST_REG_ID, K_KREDLINE_END

    begin

      res_error_is_critical = 2;
      suspend;
      exit;
    end    */
    --Проверка по паспорту продукта, пункт 1.1.
    if (KOL_CLI_GROUP <> 1) then
    begin
      --'Количество клиентов не соответствует условию паспорта продукта:

      res_error_text = 'Количество клиентов не соответствует условию паспорта продукта!';
      res_error_is_critical = 1;

      res_error_text = 'Сумма выдачи должен быть от 9 000 до 25 000 руб.!';
      res_error_is_critical = 1;
      suspend;

    begin

      res_error_text = 'Срок микрозайма должен быть от 3 до 9 месяцев!';

      res_error_text = 'Дополнительный процент не соответствует условию продукта!';
      res_error_is_critical = 1;
      suspend;

               from SCHEDULE SCH

      res_error_text = 'Клиент вправе воспользоваться льготным периодом в течение первого месяца займа!';
      res_error_is_critical = 1;

    --Проверка по паспорту продукта, пункт 1.9.
    if (exists(select *
               from SCHEDULE SCH
               where SCH.SH_FROM_K_ID = :K_ID and
                     SCH.SH_DATE <> :K_KREDLINE_END and
                     (SCH.SH_ID > 0 and
                     SCH.SH_OS + SH_PERC not between 3000 and 6000 and
                     SH_OS <> 0 or (SH_OS = 0 and
                     SH_PERC > 6000)))) then
    begin
      --'Сумма план.погашения не соответствует условию паспорта продукта, смотрите пункт: 1.9.!';
      res_error_text = 'Сумма план.погашения должна быть от 3000 р. до 6000 р.';
      res_error_is_critical = 1;
      suspend;

          KC_FROM_K_ID = :K_ID and
          KC_FROM_C_ID = C_ID

    if (C_BIRTH_DATE > dateadd(year, -21, :KF_DISB_DATE)) then
    begin
      --'Возраст клиента не позволяет получить кредит, смотрите пункт: 2.1. в паспорте продукта!';
      res_error_text = 'Возраст клиента не позволяет получить кредит!';
      res_error_is_critical = 1;
      suspend;
    end
    --Проверка по паспорту продукта, пункт 2.1.
    if (C_CITIZEN not in (9, 12)) then
    begin

      res_error_text = 'Несоответствие гражданства клиента с условием продукта';
      res_error_is_critical = 1;
      suspend;
    end
    select distinct CDS_DOC_DATE_BEGIN
    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
    where CDS_FROM_PK_ID = PK_ID and
          PK_FROM_PDT_ID = PDT_ID and
          CDS_IS_DELETE = 0 and
          PDT_CODE = 'magration_card' and
          :KF_DISB_DATE between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END and
          CDS_FROM_C_ID = :KC_FROM_C_ID rows 1
    into MIGR_DATE;
    if (MIGR_DATE is null) then
    begin
      --'Условия по проверке миграционной карты не позволяет получить кредит, смотрите пункт:
      --2.1. в паспорте продукта!';
      res_error_text = 'Нет миграционной карта либо не указан срок!';
      res_error_is_critical = 1;
      suspend;
    end
    select sum(iif(CC_TYPE = 0, 1, 0)) CON_RF, sum(iif(CC_TYPE = 1, 1, 0)) CON_ROD
    from CONTACTS
    where CC_FROM_C_ID = :KC_FROM_C_ID and
          --cc_check_result = 1 and
          CC_IS_OLD_CONTACT = 0 and
          CC_CELL_NO is not null
    into CON_RF, CON_ROD;
    CON_RF = coalesce(CON_RF, 0);
    CON_ROD = coalesce(CON_ROD, 0);
    if (CON_RF < 1 or CON_ROD < 1) then
    begin
      --'Условия по проверке контактных данных не позволяет получить кредит, смотрите пункт:
      --2.1. в паспорте продукта!';           ---
      res_error_text = 'Необходимо указать 1 контакта на родине и 1 контакта в РФ'; ---
      res_error_is_critical = 1;
      suspend;
    end
    select distinct CDS_DOC_DATE_BEGIN
    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
    where CDS_FROM_PK_ID = PK_ID and
          PK_FROM_PDT_ID = PDT_ID and
          CDS_IS_DELETE = 0 and
          PDT_CODE = 'work_patent' and
          CDS_FROM_C_ID = :KC_FROM_C_ID
    order by CDS_DOC_DATE_BEGIN desc rows 1
    into PATENT_DATE;
    if (PATENT_DATE > KF_DISB_DATE) then
    begin
      res_error_text = 'Дата выдачи патента больше даты выдачи кредита!';
      res_error_is_critical = 1;
      suspend;
    end
    --пункт: 2.2.1
    if (KF_DISB_DATE - MIGR_DATE > 20 and
        PATENT_DATE is null) then
    begin
      --'Условия по проверке миграционной карты не позволяет получить кредит, смотрите пункт:
      --2.2.1. в паспорте продукта!';
      res_error_text = 'Срок миграционной карты менее 20-дней';
      res_error_is_critical = 1;
      suspend;
    end
    select C_LAST_STATUS
    from HP_GET_CLIENT_STATUS(:KC_FROM_C_ID, :KF_DISB_DATE)
    into C_LAST_STATUS;
    --пункт: 2.3.1
    if (datediff(month, PATENT_DATE, KF_DISB_DATE) < 1 and
        C_LAST_STATUS = 0) then
    begin
      --'Срок окончание текущего патента менее одного месяца, смотрите пункт: 2.3.1. в паспорте продукта!!';
      res_error_text = 'Срок окончание текущего патента менее одного месяца';
      res_error_is_critical = 1;
      suspend;
    end
    --пункт: 2.3.2 пока невозможно сделать
    --пункт: 2.3.3
    if (not exists(select *
                   from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                   where CDS_FROM_C_ID = :KC_FROM_C_ID and
                         CDS_FROM_PK_ID = PK_ID and
                         PK_FROM_PDT_ID = PDT_ID and
                         CDS_IS_DELETE = 0 and
                         PDT_CODE in ('certificate_rus_lang', -- Сертификат о знании русского языка
                                  'diplom_vuz', -- Диплом ВУЗ
                                  'diplom_suz', -- Диплом СУЗ
                                  'diplom_school' -- Аттестат
                                  ))) then
    begin
      --'Нет необходимых документов, смотрите пункт: 2.3.3. в паспорте продукта!!';
      res_error_text = 'Необходимо указать один из документов: - Сертификат о знании русс.яз., Диплом или Аттестат!';
      res_error_is_critical = 0;
      suspend;
    end
    /*
      if (not exists(select *
                     from CLIENTDOCS CD
                     where CDS_FROM_C_ID = :KC_FROM_C_ID and
                           CDS_FROM_PK_ID in (87) and
                           CD.CDS_IS_DELETE = 0)) then

      begin
        if (not exists(select *
                       from CLIENTDOCS CD
                       where CDS_FROM_C_ID = :KC_FROM_C_ID and
                             CDS_FROM_PK_ID in (74) and
                             CD.CDS_IS_DELETE = 0)) then

        begin
          res_error_text = 'Не прикреплен справка о подаче на разрешительные документы(РВП,ВНЖ)!';
          res_error_is_critical = 1;
          suspend;
        end
        else
        begin
          res_error_text = 'Не прикреплен договор об оказании консультационных услуг!';

        if (K_LAST_REG_ID in (1020200, 1080100, 1150100, 1280100, 1380100, 1120100, 1180100, 1450100, 1190100, 1230100,
                              1470100) and
            K_AMOUNT_APPROVED > 20000) then
        begin
          res_error_text = 'Максимальная сумма микрозайма по продукту 20000 рублей';
          res_error_is_critical = 1;
          suspend;
        end
        if (K_LAST_REG_ID in (1160100) and
            K_AMOUNT_APPROVED > 18500) then
        begin
          res_error_text = 'Максимальная сумма микрозайма по продукту 20000 рублей';
          res_error_is_critical = 1;
          suspend;
        end
        if (K_LAST_REG_ID in (1360100) and
            K_AMOUNT_APPROVED > 18000) then
        begin
          res_error_text = 'Максимальная сумма микрозайма по продукту 18000 рублей';
          res_error_is_critical = 1;
          suspend;
        end
      end
      else
      begin
        if (K_LAST_REG_ID in (1020200, 1080100, 1150100, 1280100, 1380100) and
            K_AMOUNT_APPROVED > 20000) then
        begin
          res_error_text = 'Максимальная сумма микрозайма по продукту 20000 рублей';
          res_error_is_critical = 1;
          suspend;

          res_error_text = 'Максимальная сумма микрозайма по продукту 22000 рублей';
          res_error_is_critical = 1;
          suspend;

        if (K_LAST_REG_ID in (1160100) and
            K_AMOUNT_APPROVED > 18500) then
