CREATE PROCEDURE FM_CFP_POSITION_LIST(
    MAC_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ID D_INTEGER,
    NAME D_STR_255,
    IS_CHECKED D_BOOL)
AS
declare variable c_id d_id;
declare variable cp_checked_id d_id;
declare variable lang d_str_code;
declare variable acl_id d_id;
begin
  if(coalesce(mac_id, 0) < 0) then    -- RFCKG-5565
  begin
    acl_id = mac_id * -1;

    select acl.acl_object_id
    from actionlist acl
    where acl.acl_id = :acl_id
    into mac_id;
  end

  for select mac.mac_from_c_id
      from miniappclient mac
      where mac.mac_id = :mac_id and
            mac.mac_amount > 0
      into c_id
  do
  begin
    select c.c_blank_lang
    from clients c
    where c.c_id = :c_id
    into lang;
    select ca.ca_from_cp_id
    from cfp_anketa ca
    where ca.ca_from_c_id = :c_id and
          ca.ca_object_kind = f_objkind('miniappclient', 'id') and
          ca.ca_object_id = :mac_id and
          (
            coalesce(:acl_id, 0) = 0
            or ca.ca_from_acl_id = :acl_id
          ) and
          (
            coalesce(:acl_id, 0) <> 0
            or ca.ca_from_acl_id is null
          )
    into cp_checked_id;

    if (coalesce(cp_checked_id, 0) > 0) then
    begin
      select cp.cp_id, iif(:lang = 'kg', cp.cp_full_name_kg, cp.cp_full_name_ru)
      from cfp_positions cp
      where cp.cp_id = :cp_checked_id and
            cp.cp_type = 4
      into id, name;
      is_checked = 1;
      suspend;
    end
    is_checked = 0;
    for select cp.cp_id, iif(:lang = 'kg', cp.cp_full_name_kg, cp.cp_full_name_ru)
        from cfp_positions cp
        where cp.cp_id <> coalesce(:cp_checked_id, 0) and
              cp.cp_type = 4
        into id, name
    do
    begin
      suspend;
    end
  end
end
