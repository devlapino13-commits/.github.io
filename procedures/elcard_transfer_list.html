CREATE PROCEDURE ELCARD_TRANSFER_LIST
RETURNS (
    ESM_ID D_ID,
    ESM_AMOUNT D_CURRENCY,
    PAN_ENCRYPTED D_STR_LONG)
AS
declare variable K_ID D_ID;
begin
    -- процедура возвращает список операций для перевода денег по факту выдачи займа на стороннюю Элкарт
    for select em.esm_id,
               em.esm_amount,
               em.esm_from_k_id
    from extsysmain em
    where em.esm_project = 'elcard_disb' and
    em.esm_action = 'do_transfer' and
    em.esm_status = 0
    into :esm_id,
         :esm_amount,
         :k_id
    do
    begin
        pan_encrypted = '';

        select oa.oa_pan_encrypted
        from online_app oa
        inner join clients cl on cl.c_id = oa.oa_from_c_id
        where oa.oa_from_k_id = :k_id
        into :pan_encrypted;

        --pan_encrypted = 'hUseMfHycS-byLS9IRWXvmempFq_yWzJMr48L2MXvBNO7S6HRXlVghAG2VY='; -- для теста

        suspend;
    end
end
