CREATE PROCEDURE HP_PHONE_CHANGE_REQUESTS(
    REP_REGION_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    EDB_ID D_ID,
    MPC_ID D_ID,
    EDL_BLANK_DATE D_DATE,
    EDF_PRINT_NAME D_STR_100,
    EDL_OBJECT_NAME D_STR_255,
    EDC_NAME D_STR_255,
    STATUS_STR D_STR_CODE,
    REG_NAME D_STR_100,
    EDS_CLIENT_LIST D_STR_4096)
AS
declare variable REG_ID D_INTEGER;
declare variable KOBD_OFFICE D_INTEGER;
declare variable edl_object_id d_id;
declare variable edl_object_kind d_integer;
begin
    kobd_office = null;

    select ps_from_reg_id
    from hr_pstatus
    inner join username on ps_from_c_id = un_from_c_id
    where current_date between ps_date_begin and ps_date_end
        and ps_is_working = 1
        and un_id = :un_id
    into kobd_office;

    if (kobd_office is not null and :rep_region_id = 0) then
        rep_region_id = kobd_office;

    for select id_list, reg_code || ' ' || reg_name
    from region, region_list(:rep_region_id, 1, 3, 1)
    where id_list = reg_unicode
    into reg_id, reg_name
    do
    begin
        for select
            edl.edl_id,
            edb.edb_id,
            mpc.mpc_id,
            edl.edl_blank_date,
            edf.edf_print_name,
            edc.edc_name,
            edl.edl_object_kind,
            edl.edl_object_id,
            f_enum_name('mc_phone_change', mpc.mpc_status)
        from mc_phone_change mpc
        inner join mc_login mcl on mpc.mpc_from_mcl_id = mcl.mcl_id
        inner join edoc_list edl on edl.edl_id = mpc.mpc_from_edl_id
        inner join edoc_blank edb on edl.edl_id = edb.edb_from_edl_id
        inner join edoc_frx edf on edb.edb_from_edf_id = edf.edf_id
        inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
        where mpc.mpc_status = f_enum_id('mc_phone_change', 'signed')
            and coalesce(mcl.mcl_from_reg_id, 1010100) = :reg_id
        into
            edl_id,
            edb_id,
            mpc_id,
            edl_blank_date,
            edf_print_name,
            edc_name,
            edl_object_kind,
            edl_object_id,
            status_str
        do
        begin
            edl_object_name = '';
            eds_client_list = '';
            
            select doc_name
            from hp_edoc_object_info(:edl_object_kind, :edl_object_id, :un_id, :edl_blank_date)
            into edl_object_name;
            
            select list(c_name_fio)
            from edoc_sign, clients
            where eds_from_edl_id = :edl_id and
            eds_from_c_id = c_id
            into eds_client_list;
            
            suspend;
        end
    end
end
