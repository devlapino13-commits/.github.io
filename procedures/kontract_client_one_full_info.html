    CLIENT_ID DECIMAL(18,0),
    REPORT_DATE_END DATE)
RETURNS (
    K_ID D_STR_20,
    K_FROM_KF_ID D_STR_20,
    K_CURRENCY VARCHAR(3),
    K_APPLICATE_DATE DATE,
    K_APPLICATE_NO INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    K_KONTRACT_DATE DATE,
    K_FROM_PR_ID INTEGER,
    K_DISB_DATE DATE,
    K_AMOUNT_APPLICATE DECIMAL(18,4),
    K_STAGE INTEGER,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_LGOTA INTEGER,
    K_PERCENT DECIMAL(18,7),
    K_PERCENT_LGOTA DECIMAL(18,7),
    K_KOLICH_REPAY INTEGER,
    K_FROM_F_ID VARCHAR(20),
    K_FROM_PT_ID VARCHAR(20),
    K_STATUS INTEGER,
    K_STATUS_NAME VARCHAR(30),
    K_CLOSE_DATE DATE,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_PERCENT_PER_DAY DECIMAL(18,7),
    K_MEMO VARCHAR(1024),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_PERCENT_FOR_DAYS INTEGER,
    K_KOLICH_CLIENT_APPLICATE INTEGER,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_REPAY_KOLICH_CLOSED INTEGER,
    K_CLASS VARCHAR(20),
    K_KOMMIS_PERC DECIMAL(18,7),
    K_CREDIT_TARGET VARCHAR(255),
    K_IS_KREDLINE_STR VARCHAR(10),
    K_KREDLINE_END DATE,
    KFI_NAME VARCHAR(100),
    KFI_OFFICER_CODE VARCHAR(20),
    KFI_OFFICER_NAME VARCHAR(100),
    KFI_REGION_NAME VARCHAR(50),
    KFI_REGION_ID INTEGER,
    KFI_DATE DATE,
    KFI_DATE_END DATE,
    PR_CODE VARCHAR(100),
    PR_NAME VARCHAR(50),
    PR_IS_SECURED_TEXT VARCHAR(50),
    PR_LOAN_TYPE_TEXT VARCHAR(50),
    K_CIRCLE_TEXT VARCHAR(10),
    K_BALANCE DECIMAL(18,4),
    K_PAYED_FINE DECIMAL(18,4),
    KC_FROM_K_ID DECIMAL(18,0),
    KC_NUMBER INTEGER,
    KC_FROM_C_ID DECIMAL(18,0),
    KC_AMOUNT_APPLICATE DECIMAL(18,4),
    KC_FROM_PT_ID VARCHAR(20),
    KC_BUSINESS_TYPE INTEGER,
    KC_IS_NEW_BUSINESS SMALLINT,
    KC_IS_REGISTERED SMALLINT,
    KC_BUSINESS_NAME VARCHAR(255),
    KC_MONTHLY_OBOROT DECIMAL(18,4),
    KC_KAPITAL DECIMAL(18,4),
    KC_MONTH_PROFIT DECIMAL(18,4),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_WORK_KOLICH INTEGER,
    KC_WORK_KOLICH_FAMILY INTEGER,
    KC_IS_BUSINESS_FULL_TIME SMALLINT,
    KC_BUSUNESS_SUBTYPE INTEGER,
    KC_IS_NEW_ADDED SMALLINT,
    KC_STATUS SMALLINT,
    CLIENT_NAME_FIO VARCHAR(100),
    KC_IS_NEW_BUSINESS_TEXT VARCHAR(5),
    KC_IS_REGISTERED_TEXT VARCHAR(5),
    KC_IS_BUSINESS_FULL_TIME_TEXT VARCHAR(5),
    KC_IS_NEW_ADDED_TEXT VARCHAR(5),
    KC_STATUS_TEXT VARCHAR(20),
    KC_BUS_NAME VARCHAR(50),
    KC_SUBBUS_NAME VARCHAR(50),
    K_FINE_MAX_DAYS INTEGER,
    K_FINE_MAX_DAYS_STR VARCHAR(50),
    C_LAST_STATUS INTEGER,
    C_CIRCLE INTEGER)
AS
declare variable pr_is_secured integer;
declare variable pr_loan_type integer;
declare variable k_is_kredline integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable wnesys_os curr_type;
declare variable wnesys_perc curr_type;
declare variable wnesys_summ curr_type;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select k_id, k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no, k_kontract_date,
             k_from_pr_id, k_disb_date, k_amount_applicate, k_stage, k_period, k_kolich_lgota, k_percent,
             k_percent_lgota, k_kolich_repay, k_from_f_id, k_from_pt_id, kontract.k_status, k_close_date, k_circle,
             k_circle_internal, k_percent_per_day, k_memo, k_amount_approved, k_percent_for_days,
             k_kolich_client_applicate, k_kolich_client_approved, k_repay_kolich_closed, k_class, k_kommis_perc,
             k_credit_target, k_is_kredline, k_kredline_end, kal_new_kf_name, kal_new_officer_code, c1.c_name_fio,
             reg_name, reg_unicode, kal_date, kal_date_end, pr_code, pr_name, pr_is_secured, pr_loan_type, kc_from_k_id,
             kc_number, kc_from_c_id, kc_amount_applicate, kc_from_pt_id, kc_business_type, kc_is_new_business,
             kc_is_registered, kc_business_name, kc_monthly_oborot, kc_kapital, kc_month_profit, kc_amount_approved,
             kc_work_kolich, kc_work_kolich_family, kc_is_business_full_time, kc_busuness_subtype, kc_is_new_added,
             kc_status, c2.c_name_fio, bt_name, bs_name, kc_circle
      from kontract, knaktlist, clients c1, region, product, knclient, clients c2, bustype, bussubtype
      where kc_from_c_id = :client_id and
            kc_from_k_id = k_id and
            kc_amount_approved <> 0 and
            kal_from_kf_id = k_from_kf_id and
            kal_new_kf_officer = c1.c_id and
            c2.c_id = kc_from_c_id and
            k_from_pr_id = pr_id and
            reg_unicode = kal_new_reg_id and
            kal_date <= k_disb_date and
            kal_date_end >= k_disb_date and
            bt_id = kc_business_type and
            bs_id = kc_busuness_subtype and
            knaktlist.is_delete = 0
      into :k_id, :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no, :k_kontract_date,
           :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period, :k_kolich_lgota, :k_percent,
           :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id, :k_status, :k_close_date, :k_circle,
           :k_circle_internal, :k_percent_per_day, :k_memo, :k_amount_approved, :k_percent_for_days,
           :k_kolich_client_applicate, :k_kolich_client_approved, :k_repay_kolich_closed, :k_class, :k_kommis_perc,
           :k_credit_target, :k_is_kredline, :k_kredline_end, :kfi_name, :kfi_officer_code, :kfi_officer_name,
           :kfi_region_name, :kfi_region_id, :kfi_date, :kfi_date_end, :pr_code, :pr_name, :pr_is_secured,
           :pr_loan_type, :kc_from_k_id, :kc_number, :kc_from_c_id, :kc_amount_applicate, :kc_from_pt_id,
           :kc_business_type, :kc_is_new_business, :kc_is_registered, :kc_business_name, :kc_monthly_oborot,
           :kc_kapital, :kc_month_profit, :kc_amount_approved, :kc_work_kolich, :kc_work_kolich_family,
           :kc_is_business_full_time, :kc_busuness_subtype, :kc_is_new_added, :kc_status, :client_name_fio,
           :kc_bus_name, :kc_subbus_name, :c_circle
  do
  begin
    c_last_status = null;
    select c_last_status
    from hp_get_client_status(:client_id, :k_disb_date)
    into c_last_status;

    ---
    select sum(coalesce(oko.oko_os, 0) + coalesce(oko.oko_sh_os, 0)),
           sum(coalesce(oko.oko_perc, 0) + coalesce(oko.oko_sh_perc, 0))
    from operkroff oko
    where oko.oko_from_k_id = :k_id and
          oko.oko_ol_is_active = 1
    into :wnesys_os, :wnesys_perc;

    wnesys_summ = wnesys_os + wnesys_perc;
    ---

    select sum(ok_os), -1 * sum(ok_fine), max(ok_ol_date),
           max(iif((ok_type in (4, 5) and (ok_os + ok_perc <> 0)), ok_ol_date - ok_sh_date, 0))
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type >= 4 and
          ok_type <= 8 and
          ok_ol_date <= :report_date_end and
          ok_from_c_id = 0
    into :k_balance, :k_payed_fine, :k_close_date, k_fine_max_days;

    if ((k_fine_max_days is null) or
       (k_fine_max_days < 0)) then
      k_fine_max_days = 0;

    if (k_balance is null) then
      k_balance = 0;
    if (k_payed_fine is null) then
      k_payed_fine = 0;
    if (k_balance > 0) then
      k_close_date = null;

    k_fine_max_days_str = cast(k_fine_max_days as str_100);

    -- наличие внесистемного долга
    if (wnesys_summ > 0) then
    begin
      k_fine_max_days_str = 'Спецучет';
      k_balance = k_balance + wnesys_summ;
      k_close_date = null;
    end

    k_status_name = '';
    if (k_status = -1) then
      k_status_name = 'Аннулирован';
    if (k_status = -2) then
      k_status_name = 'Заявка';
    if (k_status = 0) then
      k_status_name = 'В процессе';
    if (k_status = 1) then
      k_status_name = 'Контракт';

    if (k_is_kredline = 0) then
      k_is_kredline_str = 'Нет';
    if (k_is_kredline = 1) then
      k_is_kredline_str = 'Да';

    if (pr_is_secured = 1) then
      pr_is_secured_text = 'С залогом';
    else
      pr_is_secured_text = 'Без залога';

    if (kc_is_registered = 1) then
      kc_is_registered_text = 'Да';
    else
      kc_is_registered_text = 'Нет';

    if (kc_is_new_business = 1) then
      kc_is_new_business_text = 'Да';
    else
      kc_is_new_business_text = 'Нет';

    if (kc_is_business_full_time = 1) then
      kc_is_business_full_time_text = 'Да';
    else
      kc_is_business_full_time_text = 'Нет';

    if (kc_is_new_added = 1) then
      kc_is_new_added_text = 'Да';
    else
      kc_is_new_added_text = 'Нет';

    if (kc_status = 0) then
      kc_status_text = 'Клиент';
    if (kc_status = 1) then
      kc_status_text = 'Лидер';
    if (kc_status = 2) then
      kc_status_text = 'Казначей';

    if (pr_loan_type = 1) then
      pr_loan_type_text = 'Индив.кредит';
    else
      pr_loan_type_text = 'Групп.кредит';

    k_percent = k_percent - k_percent_lgota;

    if (k_circle_internal > 1) then
      k_circle_text = (cast(k_circle as varchar(3)) || ' / ' || cast(k_circle_internal as varchar(3)));
    else
      k_circle_text = cast(k_circle as varchar(5));
    suspend;
  end
end
