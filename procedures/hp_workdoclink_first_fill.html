CREATE PROCEDURE HP_WORKDOCLINK_FIRST_FILL
RETURNS (
    OBJECT_KIND INTEGER)
AS
declare variable object_id varchar(20);
declare variable object_str varchar(1024);
declare variable str2 varchar(255);
declare variable str3 varchar(255);
begin
  delete from workdoclink;

  object_kind = 1; -- Коды платежей
  for select pc_id, pc_name
      from paycode
      into :object_id, :object_str
  do
  begin
    if (object_str is null) then
      object_str = '';
    execute procedure hp_insert_workdoclink(:object_kind, :object_id, :object_id || ' ' || :object_str);
  end
  suspend;

  object_kind = 2; -- Кассовый символ
  for select kp_id, kp_name
      from kassplan
      into :object_id, :object_str
  do
  begin
    if (object_str is null) then
      object_str = '';
    execute procedure hp_insert_workdoclink(:object_kind, :object_id, :object_id || ' ' || :object_str);
  end
  suspend;

  object_kind = 3; -- Лицевые счета
  for select al_id, al_name, al_account, al_memo
      from acc_list
      into :object_id, :object_str, str2, str3
  do
  begin
    if (object_str is null) then
      object_str = '';
    if (str2 is null) then
      str2 = '';
    if (str3 is null) then
      str3 = '';
    execute procedure hp_insert_workdoclink(:object_kind, :object_id,
        :object_id || ' ' || :object_str || ' ' || :str2 || ' ' || :str3);
  end
  suspend;

  object_kind = 4; -- Балансовые счета
  for select bg_id, bg_name, bg_memo
      from balgroup
      into :object_id, :object_str, str2
  do
  begin
    if (object_str is null) then
      object_str = '';
    if (str2 is null) then
      str2 = '';
    execute procedure hp_insert_workdoclink(:object_kind, :object_id, :object_id || ' ' || :object_str || ' ' || :str2);
  end
  suspend;

  object_kind = 5; -- Коды валют
  for select cu_id, cu_symbol, cu_name
      from currlist
      into :object_id, :object_str, str2
  do
  begin
    if (object_str is null) then
      object_str = '';
    if (str2 is null) then
      str2 = '';
    execute procedure hp_insert_workdoclink(:object_kind, :object_id, :object_id || ' ' || :object_str || ' ' || :str2);
  end
  suspend;

  object_kind = 6; -- Банки
  for select b_id, b_name, b_swift, b_mfo_old
      from banks
      into :object_id, :object_str, str2, str3
  do
  begin
    if (object_str is null) then
      object_str = '';
    if (str2 is null) then
      str2 = '';
    if (str3 is null) then
      str3 = '';
    execute procedure hp_insert_workdoclink(:object_kind, :object_id,
        :object_id || ' ' || :object_str || ' ' || :str2 || ' ' || :str3);
  end
  suspend;
