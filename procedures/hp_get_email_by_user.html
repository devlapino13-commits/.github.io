CREATE PROCEDURE HP_GET_EMAIL_BY_USER(
    UN_ID DECIMAL(18,0))
RETURNS (
    C_EMAIL VARCHAR(100),
    C_EMAIL_VERIFIED INTEGER,
    USER_FIO VARCHAR(100))
AS
declare variable c_id decimal(18,0);
begin
  c_email_verified = -2; -- Если нет пользователя или не привязан клиент
  c_email = 'Пользователь не найден';

  select un_from_c_id, un_fio
  from username
  where un_id = :un_id rows 1
  into :c_id, user_fio;
  if (user_fio is null) then
    user_fio = '';

  if (c_id is null) then
  begin
    c_email = 'Клиент не найден';
    c_email_verified = -2;
    suspend;
    exit;
  end

  select cont_value
  from hp_contdata_get_email_main(0, :c_id)
  into :c_email;
  c_email_verified = 1;

  if (c_email = '') then
  begin
    c_email = 'E-mail не указан';
    c_email_verified = 0;
  end
  suspend;
end
