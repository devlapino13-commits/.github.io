CREATE PROCEDURE HP_APPROVE_EXPRESSAPP(
    EXA_ID D_INTEGER,
    UN_ID D_ID)
AS
declare variable group_id d_integer;
declare variable count_group_id d_integer;
declare variable next_group_id d_integer;
declare variable date_begin d_date;
declare variable max_date_begin d_date;
declare variable cl_count_in_one_day d_integer;
declare variable reg_id d_integer;
begin
  -- Данная процедура начинает процесс обработки по кандидату
  group_id = 0;
  next_group_id = 0;
  max_date_begin = null;

  select un.un_curr_reg_id
  from username un
  where un.un_id = :un_id
  into reg_id;

  select max(exa.exa_date)
  from expressapp exa
  where exa.exa_from_exp_id > 1 and
        exa.exa_from_reg_id = :reg_id
  into max_date_begin;

  if (max_date_begin < current_date or (max_date_begin is null)) then
    date_begin = current_date;       --условие добавлено, так как запуск отложился на несколько дней, а самая последняя выдача была сегодня.
  else
    date_begin = max_date_begin;

  if (extract(day from :date_begin) <= 3) then
  begin
    date_begin = '04.' || extract(month from current_date) || '.' || extract(year from current_date);
  end

  select count(*)
  from expressapp exa
  where exa.exa_from_exp_id > 1 and
        exa.exa_date = :date_begin and
        exa.exa_from_reg_id = :reg_id
  into cl_count_in_one_day;

  if (cl_count_in_one_day > 20) then
    date_begin = date_begin + 1;

  if (extract(day from :date_begin) > 27) then
  begin
    exception e_message('Предложение отправляется до 27 числа включительно, утверждение отменяется');
    exit;
  end 

  next_group_id = 0;

  with table1
  as (select count(*) as exa_count, exa.exa_group_id as exa_group_id
      from expressapp exa
      where exa.exa_from_exp_id > 1 and
            exa.exa_final_status >= 0 and
            exa.exa_from_exc_id > 0 and
            exa.exa_from_exc_id <> 200
      group by exa.exa_group_id)
  select exa_group_id
  from table1
  order by exa_count
  rows 1
  into :next_group_id;  -- Беру группу с минимальным количеством кандидатов для того чтобы выставить группу для текущего кандидата

  update expressapp exa set
    exa.exa_from_exc_id = 10,
    exa.exa_wait_status = 0,
    exa.exa_from_un_id = :un_id,
    exa.exa_group_id = :next_group_id,
--    exa.exa_disb_date = :date_begin + 2,  Дата выдачи будет присваиваться после принятия предложения процедура hp_do_expressapp_process
    exa.exa_date = :date_begin
  where exa.exa_id = :exa_id;
end
