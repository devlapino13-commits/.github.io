CREATE PROCEDURE HP_MY_EDOC_MSG(
    IN_EDL_ID D_ID)
RETURNS (
    EDS_ID D_ID,
    EDS_STATUS D_INTEGER,
    EDS_SIGN_KIND D_STR_CODE,
    EDS_STATUS_DATETIME D_DATETIME,
    MSG_TYPE D_INTEGER,
    MSG_DATETIME D_DATETIME,
    MSG_SUBJECT D_STR_255,
    MSG_BODY D_STR_8100)
AS
declare variable c_id d_id;
declare variable eds_enum_edoc_sign_kind d_integer;
begin
  c_id = uxf$usr_cid(0);

  if (c_id is null) then
    exit;

  if (not exists(select * from edoc_sign
                 where eds_from_edl_id = :in_edl_id and
                       eds_from_c_id = :c_id)) then
    exit;


  for select s.eds_id, s.eds_status, s.eds_enum_edoc_sign_kind, s.eds_status_datetime,
             r.msg_type, r.msg_datetime, r.msg_subject, left(r.msg_body, 8100)
  from edoc_sign s
       left join edoc_msg m on m.edm_from_eds_id = s.eds_id
       left join fsrobot_msgbox r on m.edm_from_msg_id = r.msg_id
  where s.eds_from_edl_id = :in_edl_id and
        s.eds_from_c_id = :c_id
  into eds_id, eds_status, eds_enum_edoc_sign_kind, eds_status_datetime,
       msg_type, msg_datetime, msg_subject, msg_body
  do
  begin
    eds_sign_kind = f_enum_name('edoc_sign_kind', eds_enum_edoc_sign_kind);

    if (eds_status <> 1) then
      eds_status_datetime = null;

    if (eds_status = 0) then
      eds_status = null;

    if (coalesce(msg_subject, '') = '' and coalesce(msg_body, '') <> '') then
      msg_subject = left(msg_body, 255);

    suspend;
  end
end
