CREATE PROCEDURE AAREP_GET_OFFICE_CATEG(
    REG_ID INTEGER,
    REG_DATE DATE)
RETURNS (
    RC_REG_ID INTEGER,
    RC_VALUE_01 VARCHAR(100),
    RC_VALUE_02 VARCHAR(100),
    RC_VALUE_03 VARCHAR(100),
    RC_VALUE_04 VARCHAR(100),
    RC_VALUE_05 VARCHAR(100),
    RC_VALUE_06 VARCHAR(100),
    RC_VALUE_07 VARCHAR(100),
    RC_VALUE_08 VARCHAR(100),
    RC_VALUE_09 VARCHAR(100),
    RC_VALUE_10 VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_OFFICE_POPULATE_NAME VARCHAR(100),
    AA_OFFICE_STAT_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_LEVEL1_NAME VARCHAR(100),
    AA_CURR_LEVEL2_NAME VARCHAR(100),
    AA_CURR_LEVEL3_NAME VARCHAR(100),
    AA_CURR_LEVEL4_NAME VARCHAR(100))
AS
declare variable rc_id integer;
declare variable rc_null_value varchar(100);
declare variable rc_value varchar(100);
declare variable lev_id integer;
declare variable reg_name_tmp varchar(100);
declare variable reg_parent integer;
begin
  rc_reg_id = reg_id;
  rc_value_01 = 'Не определен';
  rc_value_02 = 'Не определен';
  rc_value_03 = 'Не определен';
  rc_value_04 = 'Не определен';
  rc_value_05 = 'Не определен';
  rc_value_06 = 'Не определен';
  rc_value_07 = 'Не определен';
  rc_value_08 = 'Не определен';
  rc_value_09 = 'Не определен';
  rc_value_10 = 'Не определен';

  for select rc_id, rc_null_value
      from regcateg
      into :rc_id, :rc_null_value
  do
  begin
    if (rc_id = 1) then
      rc_value_01 = :rc_null_value;
    if (rc_id = 2) then
      rc_value_02 = :rc_null_value;
    if (rc_id = 3) then
      rc_value_03 = :rc_null_value;
    if (rc_id = 4) then
      rc_value_04 = :rc_null_value;
    if (rc_id = 5) then
      rc_value_05 = :rc_null_value;
    if (rc_id = 6) then
      rc_value_06 = :rc_null_value;
    if (rc_id = 7) then
      rc_value_07 = :rc_null_value;
    if (rc_id = 8) then
      rc_value_08 = :rc_null_value;
    if (rc_id = 9) then
      rc_value_09 = :rc_null_value;
    if (rc_id = 10) then
      rc_value_10 = :rc_null_value;

    select distinct rcv_text
    from regcdate, regcvalue
    where rcv_from_rc_id = :rc_id and
          rcd_from_rcv_id = rcv_id and
          rcd_from_reg_id = :reg_id and
          rcd_date_begin <= :reg_date and
          rcd_date_end >= :reg_date rows 1
    into :rc_value;

    if (rc_id = 1) then
      rc_value_01 = :rc_value;
    if (rc_id = 2) then
      rc_value_02 = :rc_value;
    if (rc_id = 3) then
      rc_value_03 = :rc_value;
    if (rc_id = 4) then
      rc_value_04 = :rc_value;
    if (rc_id = 5) then
      rc_value_05 = :rc_value;
    if (rc_id = 6) then
      rc_value_06 = :rc_value;
    if (rc_id = 7) then
      rc_value_07 = :rc_value;
    if (rc_id = 8) then
      rc_value_08 = :rc_value;
    if (rc_id = 9) then
      rc_value_09 = :rc_value;
    if (rc_id = 10) then
      rc_value_10 = :rc_value;
  end

  select office.reg_code || ' ' || office.reg_name, branch.reg_code || ' ' || branch.reg_name, office.reg_populate_kind,
         office.reg_stat_rep_region, office.parent
  from region office, region branch
  where office.reg_unicode = :reg_id and
        branch.reg_unicode = office.parent
  into :aa_curr_office_name, :aa_curr_branch_name, :aa_office_populate_name, :aa_office_stat_name, :reg_parent;

  lev_id = 0;
  aa_curr_level_name = null;
  aa_curr_level2_name = null;
  aa_curr_level3_name = null;
  aa_curr_level4_name = null;
  while (reg_parent is not null) do
  begin
    select reg_code || ' ' || reg_name, parent, reg_report_level
    from region
    where reg_unicode = :reg_parent
    into :reg_name_tmp, :reg_parent, :lev_id;
    if ((lev_id = 1) and
        (aa_curr_level_name is null)) then
      aa_curr_level_name = :reg_name_tmp;
    if ((lev_id = 2) and
        (aa_curr_level2_name is null)) then
      aa_curr_level2_name = :reg_name_tmp;
    if ((lev_id = 3) and
        (aa_curr_level3_name is null)) then
      aa_curr_level3_name = :reg_name_tmp;
    if ((lev_id = 4) and
        (aa_curr_level4_name is null)) then
      aa_curr_level4_name = :reg_name_tmp;
  end
  if (aa_curr_level4_name is null) then
    aa_curr_level4_name = reg_name_tmp;
  if (aa_curr_level3_name is null) then
    aa_curr_level3_name = aa_curr_level4_name;
  if (aa_curr_level2_name is null) then
    aa_curr_level2_name = aa_curr_level3_name;
  if (aa_curr_level_name is null) then
    aa_curr_level_name = aa_curr_level2_name;
  aa_curr_level1_name = aa_curr_level_name;
  suspend;
end
