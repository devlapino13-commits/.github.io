CREATE PROCEDURE HP_REVERSAL_OPERLIST(
    REP_OL_ID ID_TYPE,
    REP_DATE DATE,
    REP_UN_ID ID_TYPE)
RETURNS (
    RES_TEXT D_STR_255,
    NEW_MAIN_OL_ID D_ID)
AS
declare variable kol_kont_id d_id;
declare variable ok_from_k_id id_type;
declare variable ox_object_id id_type;
declare variable oko_from_k_id id_type;
declare variable ol_date date;
declare variable ol_from_reg_id integer;
declare variable oek_rd_id integer;
declare variable k_id id_type;
declare variable ol_id d_id;
declare variable is_child d_smallint;
declare variable ol_parent_id d_id;
declare variable new_ol_id bigint;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  if (rep_un_id is null) then
  begin
    rep_un_id = (select un_id
                 from username
                 where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN'));
  end

  if (rep_un_id is null) then
    exception e_default 'Unknown user id';

  ol_date = null;
  select ol_date, ol_from_reg_id, ol_parent_id
  from operlist
  where ol_id = :rep_ol_id and
        operlist.is_delete = 0
  into ol_date, ol_from_reg_id, ol_parent_id;

  if (ol_date is null) then
  begin
    res_text = 'Активная операция с ID:' || rep_ol_id || ' не обнаружена.';
    suspend;
    exit;
  end

  if (ol_parent_id is not null) then
    rep_ol_id = ol_parent_id;

  -- Берем главную операцию и все его дочерние
  -- Сначала проверяем на возможность сторнирования

  res_text = '';
  for select ol_id, 0
      from operlist
      where ol_id = :rep_ol_id
      union
      select ol_id, 1
      from operlist
      where ol_parent_id = :rep_ol_id and
            not(ol_oper_type = 2 and ol_oper_subtype = 36) -- комиссию за погашение не сторнируем
      order by 2, 1
      into ol_id, is_child
  do
  begin
    kol_kont_id = null;
    ok_from_k_id = null;
    select count(distinct ok_from_k_id), max(ok_from_k_id)
    from operkred
    where ok_ol_is_active = 1 and
          ok_from_ol_id = :ol_id
    into kol_kont_id, ok_from_k_id;
    if (kol_kont_id is null) then
      kol_kont_id = 0;
    if (kol_kont_id >= 2) then
      res_text = 'По одной операции(Аналитика стандартного учета) проходит несколько контрактов, сторнирование отменяется';

    if (res_text = '') then
    begin
      kol_kont_id = null;
      ox_object_id = null;
      select count(distinct ox_object_id), max(ox_object_id)
      from operextra
      where ox_from_rd_id = :oek_rd_id and
            ox_ol_is_active = 1 and
            ox_from_ol_id = :ol_id
      into kol_kont_id, ox_object_id;
      if (kol_kont_id is null) then
        kol_kont_id = 0;
      if (kol_kont_id >= 2) then
        res_text = 'По одной операции(Судебное взыскание) проходит несколько контрактов, сторнирование отменяется';
    end

    if (res_text = '') then
    begin
      kol_kont_id = null;
      oko_from_k_id = null;
      select count(distinct oko_from_k_id), max(oko_from_k_id)
      from operkroff
      where oko_ol_is_active = 1 and
            oko_from_ol_id = :ol_id
      into kol_kont_id, oko_from_k_id;
      if (kol_kont_id is null) then
        kol_kont_id = 0;
      if (kol_kont_id >= 2) then
        res_text = 'По одной операции(Внесистемный учет) проходит несколько контрактов, сторнирование отменяется';
    end

    -- проверка, не были ли после проводки другие операции
--    if (res_text = '') then
--    begin
--      if (ok_from_k_id is not null) then
--      begin
--        if (exists(select *
--                   from operkred
--                   where ok_from_k_id = :ok_from_k_id and
--                         ok_ol_date > :ol_date and
--                         ok_from_ol_id > :ol_id and
--                         ok_ol_is_active = 1)) then
--          res_text = 'Имеются более поздние операции(Аналитика стандартного учета), сторнирование отменяется';
--      end
--    end
--
--    if (res_text = '') then
--    begin
--      if (ox_object_id is not null) then
--      begin
--        if (exists(select *
--                   from operextra
--                   where ox_from_rd_id = :oek_rd_id and
--                         ox_object_kind = 9 and
--                         ox_object_id = :ox_object_id and
--                         ox_ol_date >= :ol_date and
--                         ox_from_ol_id > :ol_id and
--                         ox_ol_is_active = 1)) then
--          res_text = 'Имеются более поздние операции(Судебное взыскание), сторнирование отменяется';
--      end
--    end
--
--    if (res_text = '') then
--    begin
--      if (oko_from_k_id is not null) then
--      begin
--        if (exists(select *
--                   from operkroff
--                   where oko_from_k_id = :oko_from_k_id and
--                         oko_ol_date >= :ol_date and
--                         oko_from_ol_id > :ol_id and
--                         oko_ol_is_active = 1)) then
--          res_text = 'Имеются более поздние операции(Внесистемный учет), сторнирование отменяется';
--      end
--    end

    if (res_text = '') then
    begin
      if (not exists(select *
                     from operbook
                     where ob_from_ol_id = :ol_id) and
          not exists(select *
                     from operkred
                     where ok_from_ol_id = :ol_id) and
          not exists(select *
                     from operkroff
                     where oko_from_ol_id = :ol_id) and
          not exists(select *
                     from operextra
                     where ox_from_rd_id = :oek_rd_id and
                           ox_from_ol_id = :ol_id) and
          not exists(select *
                     from petopers
                     where peo_from_ol_id = :ol_id) and
          not exists(select *
                     from paysys_repays
                     where psr_from_ol_id = :ol_id) and
          not exists(select *
                     from operkontract
                     where ok_from_ol_id = :ol_id) and
          not exists(select *
                     from kn_operkred
                     where ok_from_ol_id = :ol_id) and
          not exists(select *
                     from kn_operbook
                     where kob_from_ol_id = :ol_id)) then
        res_text = 'Записи для сторнирования отсутствуют!';
    end

    if (res_text <> '') then
    begin
      suspend;
      exit;
    end
  end -- конец проверок

  -- Если до этого места дошли без ошибок, делаем сторнирование каждой операции

  new_main_ol_id = null;
  for select ol_id, 0
      from operlist
      where ol_id = :rep_ol_id
      union
      select ol_id, 1
      from operlist
      where ol_parent_id = :rep_ol_id and
            not(ol_oper_type = 2 and ol_oper_subtype = 36) -- комиссию за погашение не сторнируем
      order by 2, 1
      into ol_id, is_child
  do
  begin
    -- Если операцию закрытия сторнируем то запись из KONTSTATE должны удалить.
    -- Письмо от Елены 31.05.2019г. тема: RE: сторно проводки ЕПС формируются с ошибками
    -- Думаю, лучше оставить активным, чтобы договор попадал в отчеты с нулевым остатком. Чтобы его можно было найти, при необходимости.
    delete from kontstate
    where ks_from_ol_id = :ol_id;

    --создаем новый OL_ID
    select kv
    from get_kv(:ol_from_reg_id, 5)
    into :new_ol_id;

    insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                          ol_status, is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                          ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                          ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month,
                          ol_parent_id, ol_manager_c_id, ol_from_rd_id)
    select :new_ol_id, ol_from_reg_id, ol_oper_type, :rep_date, ol_doc_no, ol_doc_no_str, ol_doc_kind,
           'Сторнирование операции:' || coalesce(ol_memo, ''), ol_status, is_delete,
           current_timestamp, :rep_un_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified, ol_document_office,
           ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, :rep_date, ol_pocket, ol_doc_client_name, ol_is_client,
           ol_year, ol_month, :new_main_ol_id, ol_manager_c_id, ol_from_rd_id
    from operlist
    where ol_id = :ol_id;

    insert into reversallist (rl_new_ol_id, rl_old_ol_id, rl_user_id, rl_date, rl_ol_is_delete)
    values (:new_ol_id, :ol_id, :rep_un_id, :rep_date, 0);

    if (is_child = 0) then
      new_main_ol_id = new_ol_id;

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                          ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                          ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id,
                          ob_short_acc_dt, ob_short_acc_kt)
    select :new_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, -ob_summa_val, -ob_summa,
           ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt,
           'Сторнирование операции:' || coalesce(ob_detail, ''), ob_from_sac_id, :rep_date,
           ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt
    from operbook
    where ob_from_ol_id = :ol_id;

    insert into operkroff (oko_from_ol_id, oko_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc, oko_other,
                           oko_sh_os, oko_sh_perc, oko_sh_other, oko_ol_is_active, oko_ol_date)
    select :new_ol_id, oko_id, oko_from_k_id, oko_from_c_id, oko_type, -oko_os, -oko_perc, -oko_other, -oko_sh_os,
           -oko_sh_perc, -oko_sh_other, oko_ol_is_active, :rep_date
    from operkroff
    where oko_from_ol_id = :ol_id;

    insert into operextra (ox_from_ol_id, ox_id, ox_ol_date, ox_ol_is_active, ox_object_kind, ox_object_id,
                           ox_from_rd_id, ox_amount_plus, ox_amount_minus, ox_detail, ox_from_reg_id,
                           ox_amount_writeoff)
    select :new_ol_id, ox_id, :rep_date, ox_ol_is_active, ox_object_kind, ox_object_id, ox_from_rd_id, -ox_amount_plus,
           -ox_amount_minus, 'Сторнирование операции:' || coalesce(ox_detail, ''), ox_from_reg_id,
           -ox_amount_writeoff
    from operextra
    where ox_from_ol_id = :ol_id;

    insert into petopers (peo_from_ol_id, peo_id, peo_bill_id, peo_bill_line_no, peo_oper_type, peo_from_c_id, peo_cost,
                          peo_from_pen_id, peo_kolich, peo_liter, peo_summa, peo_from_pet_id, peo_trans_liter,
                          peo_trans_summa, peo_probeg_fact_liter, peo_probeg_fact_km, peo_speedometr, peo_oper_date,
                          peo_from_reg_id, peo_talon_liter)
    select :new_ol_id, peo_id, peo_bill_id, peo_bill_line_no, peo_oper_type, peo_from_c_id, peo_cost, peo_from_pen_id,
           -peo_kolich, -peo_liter, -peo_summa, peo_from_pet_id, -peo_trans_liter, -peo_trans_summa,
           -peo_probeg_fact_liter, -peo_probeg_fact_km, peo_speedometr, :rep_date, peo_from_reg_id, -peo_talon_liter
    from petopers
    where peo_from_ol_id = :ol_id;

    insert into paysys_repays (psr_id, psr_from_psi_id, psr_date, psr_doc_no, psr_amount, psr_from_ol_id,
                               psr_from_un_id, psr_datetime)
    select psr_id, psr_from_psi_id, psr_date, psr_doc_no, -psr_amount, :new_ol_id, :rep_un_id, current_timestamp
    from paysys_repays
    where psr_from_ol_id = :ol_id;

    insert into operkontract (ok_id, ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
    select null, ok_from_k_id, :new_ol_id, -ok_amount, ok_operation_type
    from operkontract
    where ok_from_ol_id = :ol_id;

    insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa,
                             kob_summa_val, kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id,
                             kob_ol_is_active, kob_detail_id, kob_client_dt, kob_client_kt, kob_from_kc_id)
    select :new_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, -kob_summa, -kob_summa_val,
           -kob_summa_tax_dt, -kob_summa_tax_kt, :rep_date, kob_ol_reg_id, kob_ol_is_active, kob_detail_id,
           kob_client_dt, kob_client_kt, kob_from_kc_id
    from kn_operbook
    where kob_from_ol_id = :ol_id;

    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                          ok_ol_reg_id, ok_ol_is_active)
    select :new_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, -ok_os, -ok_perc, -ok_fine, ok_pay_type,
           -ok_kommis, -ok_sh_os, -ok_sh_perc, -ok_kommis_adv, -ok_tax, :rep_date, ok_ol_reg_id, ok_ol_is_active
    from operkred
    where ok_from_ol_id = :ol_id;

    -- удаляем то что создалось сейчас через триггер Operkred
    delete from kn_operkred
    where ok_from_ol_id = :new_ol_id;

    -- и заливаем данные из сторнируемой операции
    insert into kn_operkred (ok_from_ol_id, ok_number, ok_from_kc_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                             ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                             ok_ol_reg_id, ok_ol_is_active, ok_is_actual)
    select :new_ol_id, ok_number, ok_from_kc_id, ok_sh_id, ok_sh_date, ok_type, -ok_os, -ok_perc, -ok_fine, ok_pay_type,
           -ok_kommis, -ok_sh_os, -ok_sh_perc, -ok_kommis_adv, -ok_tax, :rep_date, :ol_from_reg_id, ok_ol_is_active,
           ok_is_actual
    from kn_operkred
    where ok_from_ol_id = :ol_id;

    -- перестраиваем график погашений
    for select distinct ok_from_k_id
        from operkred
        where ok_from_ol_id = :new_ol_id
        into k_id
    do
      execute procedure hp_do_repayment_recalc_sh(:new_ol_id, :k_id);

    if (ol_id <> new_ol_id) then
    begin
      for select opl_acc_kind, opl_from_al_id, opl_from_acc_id
      from operloan
      where opl_from_ol_id = :ol_id
      as cursor cur
      do
      begin
        update or insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id, opl_from_acc_id)
        values (:new_ol_id, cur.opl_acc_kind, cur.opl_from_al_id, cur.opl_from_acc_id)
        matching (opl_from_ol_id);
      end
    end
  end

  if (f_get_db_id() = 81001 and
      rep_date >= '01.01.2018') then
  begin
    for select distinct ok_from_k_id
        from operlist, operkred
        where ol_id = :new_main_ol_id and
              ok_from_ol_id = ol_id
        union
        select distinct ok_from_k_id
        from operlist, operkred
        where ol_parent_id = :new_main_ol_id and
              ok_from_ol_id = ol_id
        into k_id
    do
      execute procedure hp_do_kn_operbook_one_online(:k_id, :rep_date, null, null);
  end

  suspend;
end
