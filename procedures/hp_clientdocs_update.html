CREATE PROCEDURE HP_CLIENTDOCS_UPDATE(
    K_ID D_ID NOT NULL,
    DATE_BEGIN D_DATE NOT NULL)
AS
declare variable C_ID D_ID;
declare variable CDS_ID D_ID;
begin
  -- Установка даты действия документа
    for select knc.kc_from_c_id
        from knclient knc
        where knc.kc_from_k_id = :k_id and
              knc.kc_amount_approved > 0
        into :c_id
    do
    begin
      for select cds_id
          from clientdocs
          inner join passkind pk on pk_id = cds_from_pk_id
          inner join passdoctype on pdt_id = pk_from_pdt_id
          where cds_from_c_id = :c_id and
                cds_is_delete = 0 and
                ((pdt_code = 'video_sertif') or
                (pk.pk_id = 364))
          into :cds_id
      do
      begin
        update clientdocs
        set cds_is_delete = 1,
            cds_doc_date_end = :date_begin
        where cds_id = :cds_id;
      end
    end
end
