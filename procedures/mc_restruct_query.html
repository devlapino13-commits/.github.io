CREATE PROCEDURE MC_RESTRUCT_QUERY(
    MCL_ID D_ID NOT NULL,
    K_ID D_ID NOT NULL)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255)
AS
declare variable KFR_ID D_INTEGER;
declare variable KRD_ID D_INTEGER;
declare variable KRD_IS_DONE D_SMALLINT;
declare variable EDL_ID D_ID;
declare variable EDL_STATUS D_STR_CODE;
begin
    -- RFCKZ-587

    select first 1 kfr.kfr_id, krd.krd_is_done
    from kont_for_restruct_kz kfr
    left join kont_for_restruct_kz_dtl krd on krd.krd_from_kfr_id = kfr.kfr_id
    where kfr.kfr_from_k_id = :k_id
    order by kfr.kfr_id desc, krd.krd_is_done desc
    into :kfr_id, :krd_is_done;

    if (coalesce(:kfr_id, 0) = 0 or coalesce(:krd_is_done, 0) = 1) then
    begin
        res_code = 'error';
        res_text = 'Возникли непредвиденные трудности, просим обратиться к Тех. Поддержке';
        suspend;
        exit;
    end


    edl_status = f_edoc_status('restruct_application', :k_id, current_date);
    if (edl_status = 'signed') then
    begin
        -- Если "Заявление на реструктуризацию" было подписано, тогда создаём график
        execute procedure hp_extinfo_insert('restruct_schedule', 'kontract', :k_id, 1, 1000000000000);

        res_code = 'success';
        res_text = 'Запрос принят в обработку. Вам будет отправлен электронный документ';
        suspend;
    end
    else if (edl_status = 'unsigned') then
    begin
        -- Данное условие вообще не является действительным, но сделано для предосторожности
        -- Если "Заявление на реструктуризацию" находится в состоянии "создан, но ещё не подписан"
        -- Тогда просто показываем клиенту текстовку
        res_code = 'success';
        res_text = 'Эл. Документ "Заявление на реструктуризацию" ждет подписания';
        suspend;
    end
    else
    begin
        -- Иначе создаём эл. докумен "Заявление на реструктуризацию"
        select edl_id
        from hp_edoc_insert('restruct_application', 'kontract', :k_id, 1000000000000)
        into :edl_id;

        if (coalesce(:edl_id, 0) = 0) then
        begin
            res_code = 'error';
            res_text = 'Возникли непредвиденные трудности, просим обратиться к Тех. Поддержке';
            suspend;
            exit;
        end
    
        insert into kont_for_restruct_kz_dtl (krd_from_kfr_id, krd_from_edl_id, krd_from_mcl_id)
        values (:kfr_id, :edl_id, :mcl_id)
        returning krd_id into :krd_id;
    
        update kont_for_restruct_kz_dtl krd
        set krd.krd_is_done = -1
        where krd.krd_from_kfr_id = :kfr_id
            and krd.krd_id <> :krd_id
            and krd.krd_is_done <> -1;
    
        res_code = 'success';
        res_text = 'Запрос принят в обработку. Вам будет отправлен электронный документ';
        suspend;
    end
end
