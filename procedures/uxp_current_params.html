CREATE PROCEDURE UXP$CURRENT_PARAMS
RETURNS (
    PARAM_NAME D_STR_CODE,
    PARAM_VALUE D_STR_255,
    PARAM_IS_CONSTANT D_BOOL)
AS
declare variable uf_domain d_str_code;
declare variable uf_default d_str_100;
declare variable sql_text d_str_long;
declare variable edi_uuid d_uuid;
declare variable is_error d_bool;
declare variable edi_use_last d_bool;
declare variable exp_calc_expr d_str_long;
declare variable un_id d_id;
begin
  sql_text = 'select cast((%1) as %2) from rdb$database';

  for select edi_uuid, edi_use_last
  from ux$editor
  into edi_uuid, edi_use_last
  do
  begin
    for select uf_field_name, uf_domain, uf_default, exp_calc_expr
    from ux$frparam
         inner join ux$field on (ufp_uf_uuid = uf_uuid)
         left join ux$expression on (uf_uuid = exp_ux_uuid and exp_kind = 1 and exp_calc_kind >= 0)
    where ufp_uuid = :edi_uuid
    into param_name, uf_domain, uf_default, exp_calc_expr
    do
    begin
      param_value = '';
      param_is_constant = 0;

      if (param_name in ('current$region_id', 'current$office_id')) then
      begin
        un_id = uxf$user();
        param_value = uxf$user_reg(un_id);

        suspend;
        Continue;

        when any do
          param_value = '';
      end

      if (exp_calc_expr is null) then
        exp_calc_expr = '';

      if (exp_calc_expr <> '') then
      begin
        param_is_constant = 1;
        is_error = 0;
  
        begin
          execute statement exp_calc_expr
          into param_value;
    
          when any do
            is_error = 1;
        end
  
        if (is_error = 1) then
        begin
          is_error = 0;
  
          execute statement f_format(sql_text, exp_calc_expr, uf_domain)
          into param_value;
    
          when any do
            is_error = 1;
        end
  
        if (is_error = 0 and param_value <> '') then
          suspend;
      end else
      if (edi_use_last = 1) then
      begin
        if (coalesce(uf_default, '') <> '') then
        begin
          execute statement f_format(sql_text, uf_default, uf_domain)
          into param_value;
    
          when any do
            param_value = uf_default;
        end

        suspend;
      end
    end
  end
end
