CREATE PROCEDURE CM_RU_0247_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable disb_date d_date;
declare variable ma_refund d_integer;
declare variable pr_code d_str_100;
declare variable pr_code_one d_str_100;
declare variable disb_date_last d_date;
declare variable pr_name d_str_code;
declare variable c_id d_id;
begin

  for
  select k_disb_date, k_refund, pr_code, pr_name, kc_from_c_id
  from kontract, product, knclient
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and k_id = kc_from_k_id
  and kc_amount_approved > 0
  into disb_date, ma_refund, pr_code, pr_name, c_id
  do
  begin
    pr_code_one = null;
    pr_code_one = case
                     when pr_code containing '.shans.' then '.shans.'
                     when pr_code containing '.express.' then '.express.'
                     when pr_code containing '.ZALOG_CAR.' then '.ZALOG_CAR.'
                     when pr_code containing '.education.' then '.education.'
                     when pr_code containing '.reliable_client.' then '.reliable_client.'
                     when pr_code containing '.LOYAL.' then '.LOYAL.'
                     when pr_code containing '.reliable_emp.' then '.reliable_emp.'
                     when pr_code containing '.exclusive.' then '.exclusive.'
                     when pr_code containing '.zalog.' then '.zalog.'
                     when pr_code containing '.selfemployed.' then '.selfemployed.'
                     when pr_code containing '.business.' then '.business.'
                     when pr_code containing '.privilege.' then '.privilege.'
                     when pr_code containing '.privilege_plus.' then '.privilege_plus.'
                     when pr_code containing '.refinance.' then '.refinance.'
                     when pr_code containing '.refinance_oivo.' then '.refinance_oivo.'
                    end;
    pr_code_one = trim(pr_code_one);

    if (ma_refund = 0 and pr_code_one is not null) then
    begin
      disb_date_last = null;
      select k_disb_date from kontstate, kontract, product, knclient
      where ks_ol_is_delete = 0
      and ks_state = 1
      and :disb_date between ks_date_begin and ks_date_end
      and ks_from_k_id = k_id
      and k_from_pr_id = pr_id
      and pr_code containing :pr_code_one
      and k_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = :c_id
      order by k_disb_date desc
      rows 1
      into disb_date_last;

      if (disb_date_last is not null) then
      begin
        if (pr_code_one = '.shans.') then
        begin
          res_error_text = 'Имеется Шанс активный займ, выдача запрещена!';
          res_error_is_critical = 2;
          suspend;
        end else
        if (pr_code_one = '.zalog.') then
        begin
          res_error_text = 'Имеется активный Залоговый займ, выдача запрещена!';
          res_error_is_critical = 2;
          suspend;
        end else
        if (pr_code_one = '.ZALOG_CAR.') then
        begin
          res_error_text = 'Имеется активный Микрозайм под залог, выдача запрещена!';
          res_error_is_critical = 2;
          suspend;
        end else
        if (disb_date_last + 3  > disb_date) then
        begin
          res_error_text = 'Новый заем '||:pr_name||' может быть выдан не ранее '||f_datetostr((disb_date_last + 3));
          res_error_is_critical = 2;
          suspend;
        end
      end
    end
  end
end
