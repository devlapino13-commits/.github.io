CREATE PROCEDURE HP_SAVE_APP_RETURN_CONSULT(
    ARC_FROM_C_ID D_ID,
    ARC_FROM_K_ID D_ID,
    ARC_DATE D_DATE,
    ARC_STATUS D_SMALLINT,
    ARCI_APP_OFFICE_ID INTEGER,
    ARCI_CONSULT_OFFICE_ID INTEGER,
    ARCI_AMOUNT D_CURRENCY,
    ARC_CREATOR_UN_ID D_ID,
    ARC_SOLUTION_UN_ID D_ID,
    ARC_CREATE_DATETIME TIMESTAMP,
    ARC_STATUS_DATETIME TIMESTAMP,
    ARCI_RETURN_REASON D_SMALLINT,
    ARCI_RETURN_TYPE D_SMALLINT,
    ARCI_SIGN_TYPE D_SMALLINT,
    OL_DATE DATE)
RETURNS (
    ARC_ID D_INTEGER,
    RES_TXT D_STR_255)
AS
begin
  res_txt = '';

  if (:arci_sign_type = 0) then
    :arc_status = f_enum('app_return_consult', 'process', 'id');
  if (:arci_sign_type = 1) then
    :arc_status = f_enum('app_return_consult', 'new_app', 'id');

  insert into app_return_consult (arc_id, arc_from_c_id, arc_from_k_id, arc_date, arc_status, arc_creator_un_id,
                                  arc_create_datetime, arc_status_un_id, arc_status_datetime)
  values (:arc_id, :arc_from_c_id, :arc_from_k_id, :arc_date, :arc_status, :arc_creator_un_id, :arc_create_datetime,
          :arc_creator_un_id, :arc_create_datetime)
  returning arc_id
  into :arc_id;

  if (:arc_id is null) then
  begin
    res_txt = 'Произошла ошибка при регистрации заявления.';

    suspend;
    exit;
  end

  insert into app_return_consult_info (arci_id, arci_app_office_id, arci_consult_office_id, arci_amount, arci_return_reason,
                                       arci_return_type, arci_sign_type)
  values (:arc_id, :arci_app_office_id, :arci_consult_office_id, :arci_amount, :arci_return_reason, :arci_return_type,
          :arci_sign_type);

  suspend;
end
