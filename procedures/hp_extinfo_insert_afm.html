CREATE PROCEDURE HP_EXTINFO_INSERT_AFM(
    C_ID D_ID,
    OBJ_ID D_ID)
AS
begin
    if (exists
        (
            select 1
            from extinfolist eil
            inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
                and eic.eic_code in ('afm_iin_check', 'afm_name_check')
            where eil.eil_from_c_id = :c_id
                and eil.eil_object_id = :obj_id
                and eil.eil_status in (0, 1, 2)
        )
    ) then
        exit;

    if (f_get_const('afm_is_active') = '1') then
    begin
        execute procedure hp_extinfo_insert('afm_iin_check', 'miniapp', :obj_id, 1, 1000000000000);
        post_event 'afm_iin_check';

        execute procedure hp_extinfo_insert('afm_name_check', 'miniapp', :obj_id, 1, 1000000000000);
        post_event 'afm_name_check';

        execute procedure hp_afm_send_compliance_mail(f_objkind('clients', 'id'), :c_id, 'afm_limit');
    end
    else
        execute procedure hp_extinfo_insert('checkme_info', 'clients', :c_id, 1);
end
