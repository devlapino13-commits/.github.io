CREATE PROCEDURE HP_UXDEPEND_LIST
RETURNS (
    UDP_CODE D_STR_255,
    UDP_PARENT_CODE D_STR_255,
    UDP_OWNER_CODE D_STR_100,
    UDP_UX_CODE D_STR_100,
    UDP_UX_KIND D_STR_CODE,
    UDP_POSITION D_INTEGER,
    UDP_IMAGE_ID D_INTEGER,
    UDP_LOCK_DELETE D_BOOL,
    UDP_CHILD_LOCKED D_BOOL,
    RES_IS_OWN D_BOOL,
    RES_NAME D_STR_255,
    RES_MEMO D_STR_255,
    RES_KIND_CAPTION D_STR_100,
    RES_TABLE_NAME D_STR_CODE,
    RES_DATETIME D_DATETIME)
AS
declare variable udp_caption d_str_255;
begin
  for select udp_code,
             udp_parent_code,
             udp_owner_code,
             udp_ux_code,
             udp_ux_kind,
             udp_position,
             udp_caption,
             udp_image_id,
             udp_lock_delete,
             udp_child_locked
      from uxdepend
      order by udp_parent_code, udp_position
      into :udp_code,
           :udp_parent_code,
           :udp_owner_code,
           :udp_ux_code,
           :udp_ux_kind,
           :udp_position,
           :udp_caption,
           :udp_image_id,
           :udp_lock_delete,
           :udp_child_locked
  do
  begin
    res_is_own = 1 - udp_lock_delete;

    res_name = udp_caption;
    res_memo = '';
    res_kind_caption = '';
    res_table_name = '';
    res_datetime = null;

    if (udp_ux_kind = 'uxaction') then
    begin
      res_kind_caption = 'Действие';

      select ua_name
      from uxaction
      where ua_code = :udp_ux_code
      into res_name;
    end else
    if (udp_ux_kind = 'table') then
    begin
      res_table_name = udp_ux_code;
      res_kind_caption = 'Таблица';
      res_name = udp_ux_code;
    end else
    begin
      res_kind_caption = uxf$kind(udp_ux_kind, 'caption');

      select ux_name, ux_memo, ux_table_name, ux_create_datetime
      from uxcode
      where ux_code = :udp_ux_code
      into res_name, res_memo, res_table_name, res_datetime;
    end

    suspend;
  end
end
