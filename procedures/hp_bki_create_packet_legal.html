CREATE PROCEDURE HP_BKI_CREATE_PACKET_LEGAL(
    UID_STR D_STR_CODE NOT NULL,
    EVENT_CODE D_STR_10 NOT NULL,
    EVENT_DATE D_DATE NOT NULL)
RETURNS (
    PACKET_ID D_ID,
    PACKET_NAME D_STR_CODE,
    PACKET_DATE D_DATE)
AS
declare variable counter d_integer;
  declare variable obj_kind d_integer;
  declare variable obj_id d_id;
  declare variable object_kind d_integer;
  declare variable object_id d_id;
  declare variable c_id d_id;
  declare variable bkl_id d_id;
  declare variable bkc_id d_id;
  declare variable event_id d_integer;
begin
  counter = 0;
  packet_date = current_date;

  select obj_kind, obj_id
  from hp_find_cbr_uuid(:uid_str)
  into obj_kind, obj_id;

  if (f_objkind(obj_kind, 'code') = 'miniappclient') then
    select f_objkind('miniapp', 'id'), mac_from_ma_id, mac_from_c_id
    from miniappclient
    where mac_id = :obj_id
    into object_kind, object_id, c_id;
  else if (f_objkind(obj_kind, 'code') = 'knclient') then
    select f_objkind('kontract', 'id'), kc_from_k_id, kc_from_c_id
    from knclient
    where kc_id = :obj_id
    into object_kind, object_id, c_id;

  if (object_id is null) then
    exception e_message 'Object not found. uid = ' || coalesce(uid_str, '<NULL>');

  select el_id
  from enum_list
  where el_kind = 'bki_event'
    and el_code starting with :event_code || '_'
  into event_id;

  if (event_id is null) then
    exception e_message 'Event not found. event_code = ' || coalesce(event_code, '<NULL>');

  select packet_name, new_counter
  from hp_bki_get_packet_name_equifax(:packet_date, :counter, 0)
  into packet_name, counter;

  insert into bki_packet (bkp_code, bkp_from_bkk_id, bkp_date, bkp_number, bkp_from_bkt_id, bkp_status)
  values (:packet_name, 1, :packet_date, :counter, 1, 1)
  returning bkp_id
  into :packet_id;

  insert into bki_loans (bkl_from_bkp_id, bkl_object_kind, bkl_object_id)
  values (:packet_id, :object_kind, :object_id)
  returning bkl_id
  into :bkl_id;

  insert into bki_clients (bkc_from_bkl_id, bkc_from_c_id, bkc_uid_bkd_id)
  values (:bkl_id, :c_id, f_bki_data_id(:uid_str))
  returning bkc_id
  into bkc_id;

  insert into bki_events (bev_from_bkc_id, bev_event_date, bev_tag, bev_event_id, bev_status, bev_rec_no)
  values (:bkc_id, :event_date, 5, :event_id, 1, 1);

  suspend;
end
