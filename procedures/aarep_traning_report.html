CREATE PROCEDURE AAREP_TRANING_REPORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE D_DATE,
    C_AGE D_INTEGER,
    C_POSITION VARCHAR(300),
    C_CATEGORY VARCHAR(100),
    C_OFFICE VARCHAR(100),
    C_DEPART VARCHAR(100),
    C_REGION VARCHAR(100),
    C_OKRUG VARCHAR(100),
    TM_DATE_BEGIN DATE,
    TM_DATE_END DATE,
    TM_SUBJECT VARCHAR(255),
    TM_PLACE VARCHAR(255),
    DAYS_QTY INTEGER,
    TRAINERS VARCHAR(1000),
    TOTAL_HOURS INTEGER,
    TM_TYPE_STR VARCHAR(255),
    TM_MEMO D_STR_2048)
AS
declare variable tb_from_c_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable tm_id integer;
declare variable trainer varchar(255);
declare variable c_id integer;
begin
  /*  changed by IGUSEV  27.08.2012
      changes:  были убраны следующие исходящие параметры:
                TR_2_KOLICH integer  -  Кол-во тренинг. более 1 дня
                TR_2_TOTAL_DAYS integer -  Всего дни тренинг.более 1 дня
                TR_2_TOTAL_HOURS integer - Всего часы тренинг.более 1 дня
                TR_1_TOTAL_HOURS integer - Всего часы одноднев.тренингов,
                TR_1_KOLICH integer - Кол-во одноднев,

                Были добавлены следующие исходящие параметры:
                C_REGION varchar(100) - ,
                C_OKRUG varchar(100),
                TM_DATE_BEGIN date,
                TM_DATE_END date,
                TM_SUBJECT varchar(255),
                TM_PLACE varchar(255),
                DAYS_QTY integer,
                TRAINERS varchar(255),
                TM_TYPE_STR varchar(255)
  */

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  for select distinct tm.tm_id, tb_from_c_id, c_name_fio, c_birth_date, tm.tm_date_begin, tm.tm_date_end, tm.tm_subject, tm.tm_place,
                      (
                      case tm.tm_type
                        when 0 then 'Корпоративный'
                        when 1 then 'Открытый'
                      end), tm.tm_memo
      from tranmembers tb, tranmain tm, region_list(:report_region_id, 1, -1, 1), clients
      where tb_is_trainer = 0 and
            tm_id = tb_from_tm_id and
            tm_date_end >= :report_date_begin and
            tm_date_begin <= :report_date_end and
            (tm_from_reg_id = id_list) and
            tb_from_c_id = c_id
      into :tm_id, :tb_from_c_id, :c_name_fio, :c_birth_date, :tm_date_begin, :tm_date_end, :tm_subject, :tm_place, :tm_type_str, :tm_memo
  do
  begin

    c_position = '';
    c_category = '';
    c_office = '';
    c_depart = '';

    days_qty = cast((tm_date_end - tm_date_begin + 1) as integer);

    c_age = cast(datediff(month, :c_birth_date, :report_date_end) / 12 as integer);

    select jl_name, reg.reg_name, dep_name, jl_category, reg2.reg_name, /*РЕГИОН*/
           reg3.reg_name /*ОКРУГ*/
    from hr_pstatus hps
    inner join depart on ps_from_dep_id = dep_id
    inner join joblist on ps_from_jl_id = jl_id
    inner join region reg on ps_from_reg_id = reg.reg_unicode
    left join region reg2 on reg2.reg_unicode = reg.parent
    left join region reg3 on reg3.reg_unicode = reg2.parent
    where ps_from_c_id = :tb_from_c_id and
          ps_date_begin <= :report_date_end and
          ps_date_end >= :report_date_end
    order by ps_is_working desc, ps_date_end desc
    rows 1
    into :c_position, :c_office, :c_depart, :c_category, :c_region, :c_okrug;

    trainers = '';
    trainer = '';
    for select distinct ', ' || coalesce(cl.c_name_fio, '')
        from tranmembers tb
        left join clients cl on tb_from_c_id = c_id
        where tb_from_tm_id = :tm_id and
              tb_is_trainer = 1
        into :trainer
    do
    begin
      trainers = left(trainers || trainer, 1000);
    end
    if (trainers <> '') then
      trainers = substring(trainers from 3);

    /*
        TR_1_KOLICH=0;
        TR_1_TOTAL_HOURS=0;

        select Count(*), sum(tm_hours) from tranmain, tranmembers
        where tb_from_tm_id=tm_id
        and tb_is_trainer=0
        and tb_from_c_id=:tb_from_c_id
        and (tm_date_end-tm_date_begin)<=1
        and tm_date_end>=:report_date_begin
        and tm_date_begin<=:report_date_end
        into :TR_1_KOLICH, :TR_1_TOTAL_HOURS;
        if (TR_1_KOLICH is null) then TR_1_KOLICH=0;
        if (TR_1_TOTAL_HOURS is null) then TR_1_TOTAL_HOURS=0;
        */

    select sum(tm_hours)
    from tranmain, tranmembers
    where tb_from_tm_id = tm_id and
          tb_is_trainer = 0 and
          tb_from_c_id = :tb_from_c_id and
          tm_date_end >= :report_date_begin and
          tm_date_begin <= :report_date_end and
          tm_id = :tm_id
    into :total_hours;



    /*
        TR_2_KOLICH=0;
        TR_2_TOTAL_HOURS=0;
        TR_2_TOTAL_DAYS=0;
        select Count(*), sum(cast(tm_date_end as date)-cast(tm_date_begin as date)), sum(tm_hours) from tranmain, tranmembers
        where tb_from_tm_id=tm_id
        and tb_is_trainer=0
        and tb_from_c_id=:tb_from_c_id
        and (tm_date_end-tm_date_begin)>1
        and tm_date_end>=:report_date_begin
        and tm_date_begin<=:report_date_end
        into :TR_2_KOLICH, :TR_2_TOTAL_DAYS, :TR_2_TOTAL_HOURS;
        if (TR_2_KOLICH is null) then TR_2_KOLICH=0;
        if (TR_2_TOTAL_DAYS is null) then TR_2_TOTAL_DAYS=0;
        if (TR_2_TOTAL_HOURS is null) then TR_2_TOTAL_HOURS=0;    */

    suspend;
  end
end
