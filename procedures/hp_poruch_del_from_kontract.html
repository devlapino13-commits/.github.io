CREATE PROCEDURE HP_PORUCH_DEL_FROM_KONTRACT(
    K_ID ID_TYPE,
    C_ID ID_TYPE)
RETURNS (
    ERROR_MESSAGE STR_100)
AS
declare variable counter integer;
begin
  error_message = '';
  counter = 0;

  for select kont.k_id, zal.z_from_c_id
      from kontract kont
      inner join zalog zal on kont.k_id = zal.z_from_k_id
      where kont.k_id = :k_id and
            zal.z_from_c_id = :c_id rows 1
      into :k_id, :c_id
  do
  begin
    counter = + 1;

    delete from zalog zal
    where zal.z_from_k_id = :k_id and
          zal.z_from_c_id = :c_id;
  end

  if (counter = 0) then
  begin
    error_message = 'Операция не удалась. Либо ошиблись с ID контаркта либо с ID поручителя';
  end

  suspend;
end
