CREATE PROCEDURE CB_KG_KOBD_2021_09(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    REPAY_SUM_NOT_OS CURR_TYPE,
    REPAY_KOMMIS_IPF CURR_TYPE,
    DELINQ_OS_PERIOD CURR_TYPE,
    AAREP_CONSULT_SUMMA CURR_TYPE,
    SUM_VOZNOG_OFFICE CURR_TYPE,
    KOL_KOBD SMALLINT,
    FIO_KOBD VARCHAR(1024),
    BONUS_KOBD CURR_TYPE,
    MEMO VARCHAR(2048))
AS
declare variable rep_date_begin date;
declare variable reg_id integer;
declare variable c_name_fio varchar(100);
declare variable c_id id_type;
declare variable bonus_memo varchar(255);
declare variable ps_date_begin date;
declare variable is_kobd smallint;
declare variable date_dismiss date;
declare variable officer_list d_str_long;
declare variable rd_id_ipf d_integer;
declare variable rd_id_cons d_integer;
declare variable tmp_sum d_currency;
declare variable ok_operation_type d_integer;
begin
  :rd_id_ipf = f_get_rd_id('CashTransactionIncom.IPFCommission');
  :rd_id_cons = f_get_rd_id('CashTransactionIncom.Consultation');
  
  select list(ps.ps_from_c_id)
  from hr_pstatus ps
  inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
  inner join jobkind jk on jk.jk_id = jl.jl_kind and
                           jk.jk_system_code = 'Jobsystem.Operational_manager'
  where ps.ps_status = 1 and
        ps.ps_is_on_staff = 1 and
        :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_is_working = 1
  into officer_list;

  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin
    repay_sum_not_os = null;
    repay_kommis_ipf = 0;
    aarep_consult_summa = 0;

    --  Суммы погашении процентов пеня и админ сбор
    select -sum(ok.ok_perc + ok.ok_fine + ok.ok_kommis)
    from operkred ok
    inner join kontract k on k.k_id = ok_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = k_from_kf_id
    where ok.ok_ol_is_active = 1 and
          ok.ok_type = 4 and
          ok.ok_ol_reg_id = :reg_id and
          ok.ok_ol_date between :rep_date_begin and :rep_date_end and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0 and
          kal.kal_new_kf_officer not in (select value_out
                                          from str_with_separator_parser(:officer_list, ','))
    into repay_sum_not_os;
    repay_sum_not_os = coalesce(repay_sum_not_os, 0);

    --  Комиссия за выдачу ИПФ
    -- оплата за консультационные услуги
    for select sum(ok.ok_amount), ok_operation_type
        from operkontract ok
        inner join kontract k on k.k_id = ok.ok_from_k_id
        inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
        inner join operlist ol on ol.ol_id = ok.ok_from_ol_id and
              ol.is_delete = 0 and
              ol.ol_from_reg_id = :reg_id and
              ol.ol_date between :rep_date_begin and :rep_date_end and
              :rep_date_end between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0 and
              kal.kal_new_kf_officer not in (select value_out
                                             from str_with_separator_parser(:officer_list, ','))
        where ok.ok_operation_type in (:rd_id_ipf, :rd_id_cons)
        group by 2
        into :tmp_sum, :ok_operation_type

    do
    begin
      if (ok_operation_type = :rd_id_ipf) then
        repay_kommis_ipf = :tmp_sum;
      if (ok_operation_type = :rd_id_cons) then
        aarep_consult_summa = :tmp_sum;
    end
    repay_kommis_ipf = coalesce(repay_kommis_ipf, 0);
    aarep_consult_summa = coalesce(aarep_consult_summa, 0);

    --  Сумма просроченной задолженности по ОС за отчетный период
    delinq_os_period = null;
    select sum(ok_os + ok_sh_os)
    from oksummary oks, operkred ok
    inner join kontract k on k.k_id = ok.ok_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks.oks_region_id = :reg_id and
          :rep_date_end >= oks.oks_delinq_date and
          oks.oks_from_k_id = ok.ok_from_k_id and
          oks.oks_close_state = 0 and
          ok.ok_ol_is_active = 1 and
          ok.ok_type < 7 and
          ok.ok_sh_date between :rep_date_begin and :rep_date_end and
          ok.ok_ol_date <= :rep_date_end and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0 and
          kal.kal_new_kf_officer not in (select value_out
                                         from str_with_separator_parser(:officer_list, ','))
    into delinq_os_period;
    delinq_os_period = coalesce(delinq_os_period, 0);
    --  Сумма вознаграждения подотчетного офиса
    sum_voznog_office = (repay_sum_not_os + repay_kommis_ipf) * 9 / 100 + (aarep_consult_summa) * 9 / 100 - delinq_os_period * 40 / 100;

    if (repay_sum_not_os <> 0 or repay_kommis_ipf <> 0 or delinq_os_period <> 0) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;

      kol_kobd = 0;
      fio_kobd = '';
      memo = '';
      for select c_name_fio, c_id, ps_date_begin
          from hr_pstatus, joblist, jobkind, clients
          where ps_status = 1 and
                ps_is_working = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code = 'JobSystem.Specialist_KOBD' and
                jl_name <> 'Выездной кассир' and
                :rep_date_end between ps_date_begin and ps_date_end and
                ps_from_reg_id = :reg_id and
                ps_from_c_id = c_id
                /*  and (ps_date_begin <= dateadd(month, - 3, :rep_date_end)
           or
           (select min(hr1.ps_date_begin) from hr_pstatus hr1
            where hr1.ps_is_working = 1
            and hr1.ps_status = 1
            and hr1.ps_from_c_id = c_id) <= dateadd(month, - 3, :rep_date_end)
           )  */
                and
                coalesce((select sawt_id
                          from hr_voctravel, hr_orders, sa_worktype
                          where hrv_from_ho_id = ho_id and
                                ho_order_type = sawt_id and
                                hrv_from_c_id = ps_from_c_id and
                                :rep_date_end between hrv_date_begin and hrv_date_end_fact and
                                hrv_status = 1
                          order by hrv_date_begin desc
                          rows 1), 0) not in (16, 17, 18, 8, 24) -- Декретников не включает
          into c_name_fio, c_id, ps_date_begin
      do
      begin
        if (ps_date_begin <= dateadd(month, -3, :rep_date_end)) then
          is_kobd = 1;
        else
        begin
          --        вычисляем дату последнего увольнения
          date_dismiss = null;
          select max(ps_date_begin)
          from hr_pstatus
          where ps_from_c_id = :c_id and
                ps_status = 1 and
                ps_is_working = 0 and
                ps_date_begin <= :rep_date_end
          into date_dismiss;
          if (date_dismiss is null) then
            date_dismiss = '01.01.1990';

          select min(ps_date_begin)
          from hr_pstatus, joblist, jobkind
          where ps_status = 1 and
                ps_is_working = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code = 'JobSystem.Specialist_KOBD' and
                jl_name <> 'Выездной кассир' and
                ps_from_c_id = :c_id and
                ps_date_begin > :date_dismiss
          into ps_date_begin;
          if (ps_date_begin <= dateadd(month, -3, :rep_date_end)) then
            is_kobd = 1;
          else
            is_kobd = 0;
        end
        if (is_kobd = 1) then
        begin
          kol_kobd = kol_kobd + 1;
          if (fio_kobd = '') then
            fio_kobd = c_name_fio;
          else
            fio_kobd = fio_kobd || '; ' || c_name_fio;

          --  вытаскивает данные по выговорам и бонус коэффициент
          bonus_memo = null;
          select bonus_memo
          from hp_get_vigovor_koef(:c_id, :rep_date_end)
          where bonus_koef < 100 or
                bonus_sum > 0
          into bonus_memo;
          bonus_memo = coalesce(bonus_memo, '');
          if (bonus_memo <> '') then
          begin
            if (memo = '') then
              memo = c_name_fio || '-' || bonus_memo;
            else
              memo = memo || '; ' || c_name_fio || '-' || bonus_memo;
          end
        end
      end
      if (kol_kobd is null) then
        kol_kobd = 0;
      if (kol_kobd > 0) then
        bonus_kobd = sum_voznog_office / (50 * kol_kobd);
      else
        bonus_kobd = 0;
      if (sum_voznog_office < 0) then
        bonus_kobd = 0;
      suspend;
    end
  end
end
