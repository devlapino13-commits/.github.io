CREATE PROCEDURE AAREP_SCORE_RATE_REPORT(
    REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    DISB_DATE D_DATE,
    KONTRACT_NO D_STR_20,
    GROUP_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    PRODUCT_NAME D_STR_100,
    CUSTOMER_FIO D_STR_100,
    CLIENT_FIO D_STR_100,
    CLIENT_AMOUNT D_CURRENCY,
    DELINQ D_INTEGER,
    CLIENT_SCORE_RATE D_INTEGER,
    PROBLEM D_STR_255)
AS
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable max_delinq d_integer;
declare variable active_delinq d_integer;
declare variable kbm_id d_id;
begin
    for select distinct client.c_id, kn.k_id, kn.k_from_ma_app_id, kn.k_disb_date, kn.k_kontract_no, kal.kal_new_kf_name,
                        reg.reg_name,pr.pr_name , client.c_name_fio, kc.kc_amount_approved, customer.c_name_fio
    from kontract kn
    inner join kontstate ks on ks.ks_from_k_id = kn.k_id
    inner join knaktlist kal on kal.kal_from_kf_id = kn.k_from_kf_id
    inner join knclient kc on kc.kc_from_k_id = kn.k_id
    inner join clients client on client.c_id = kc.kc_from_c_id
    inner join clients customer on customer.c_id = ks.ks_kc_id
    inner join region_list(:region_id,1,3,1) t on t.id_list = cast(ks.ks_region_id as d_integer)
    inner join region reg on t.id_list = reg.reg_unicode
    inner join product pr on pr.pr_id = kn.k_from_pr_id
    where ks.ks_state = 1
          and ks.ks_ol_is_delete = 0
          and ks.ks_date_begin between :rep_date_begin and :rep_date_end
          and :rep_date_end between kal.kal_date and kal.kal_date_end
          and kal.is_delete = 0
          and kc.kc_amount_approved > 0
    into :c_id,
         :k_id,
         :ma_id,
         :disb_date,
         :kontract_no,
         :group_name,
         :office_name,
         :product_name,
         :client_fio,
         :client_amount,
         :customer_fio
    do
    begin
        active_delinq = 0;

        select iif(((current_date - oks.oks_delinq_date) < 0), 0, current_date - oks.oks_delinq_date)
        from oksummary oks
        where current_date between oks.oks_date_begin and oks.oks_date_end
        and oks.oks_from_k_id = :k_id
        and oks_close_state = 0
        into :active_delinq;

        max_delinq = 0;

        select coalesce(max(ok.ok_ol_date - ok_sh_date), 0) 
        from operkred ok
        where ok.ok_ol_is_active = 0
        and ok.ok_ol_date <= current_date
        and ok.ok_ol_date > ok_sh_date
        and ok_type < 7
        and ok_from_k_id = :k_id
        and ok_os + ok_perc < 0
        and ok_from_c_id=0
        into :max_delinq;

        delinq = iif(:max_delinq>:active_delinq, :max_delinq, :active_delinq);

        kbm_id = null;
        client_score_rate = 0;

        select kbm.kbm_score_rate,
               kbm.kbm_id
        from kibminiapp kbm
        where kbm.kbm_ma_id = :ma_id  and
              kbm.kbm_from_c_id = :c_id and
              kbm.kbm_is_checked = 1
        order by kbm.kbm_id desc
        rows 1
        into :client_score_rate,
             :kbm_id;

        problem = '';
        if(kbm_id is null) then
            problem = 'Нет запросов в КИБ';

        suspend;
    end
end
