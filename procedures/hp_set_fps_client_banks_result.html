CREATE PROCEDURE HP_SET_FPS_CLIENT_BANKS_RESULT(
    ID D_INTEGER,
    CLIENT_NAME D_STR_100)
AS
declare variable status d_smallint;
declare variable c_id d_id;
declare variable un_id d_id;
declare variable k_id d_id;
begin
  -- Устанавливает статус проверки банка клиента по СБП.
  -- Вызывается сервисом проверки через СБП
  -- Если сервис вернул что у клиент нет счета в банке, то Client_Name = 'x'
  -- Иначе в Client_name нужно передать ФИО из ответа банка

  if (id is null) then
    exception e_default 'fcb_id is null';

  if (coalesce(client_name, '') = '') then
    exception e_default 'Client_name is emptry';

  status = null;

  if (client_name = 'x') then
    status = -1;

  if (client_name <> 'x') then
  begin
    select fcb_from_c_id, fcb_from_un_id
    from fps_client_banks
    where fcb_id = :id
    into c_id, un_id;

    status = 1;
    if (exists(select * from moneta_client_fps_names(:c_id)
               where client_names containing '"' || :client_name || '"'))
    then
      status = 2;

    if (not exists(
    select * from fps_client_names
    where fcn_from_fcb_id = :id and
          fcn_name = :client_name)) then
    begin
      -- Имя добавляем даже если совпадает, на случай, если потом андеррайтеры изменят
      insert into fps_client_names (fcn_from_fcb_id, fcn_name, fcn_status)
      values (:id, :client_name, iif(:status = 2, 2, 0));
    end
    else
    begin
      -- Если имя есть, контрольно обновляем статус
      update fps_client_names
      set fcn_status = iif(:status = 2, 2, 0)
      where fcn_from_fcb_id = :id and
          fcn_name = :client_name and
          fcn_status <> iif(:status = 2, 2, 0);
    end
  end

  if (status is not null) then
  begin
    update fps_client_banks
    set fcb_status = :status
    where fcb_id = :id and
          fcb_status <> :status;
  end

  if (status = 1) then
  begin
    for select kc_from_k_id
    from fps_loan_banks, knclient, kontract
    where fbl_from_fcb_id = :id
      and flb_from_kc_id = kc_id
      and kc_from_k_id = k_id
      and k_disb_kind = 7 -- СБП
      and k_status = 0
    into k_id do
      execute procedure hp_fps_resend_kk(k_id, un_id);
  end
end
