CREATE PROCEDURE HP_CLIENTSTATE_ORDER_CREATE(
    C_ID D_ID NOT NULL,
    FIELD D_STR_100 NOT NULL,
    STATUS D_SMALLINT NOT NULL)
AS
declare variable sql_text d_str_long;
declare variable flag d_smallint;
begin
    -- RFCRU-5337
    -- Поля из таблицы clientstate_ru
    if (field not in
        (
            'csr_has_miniapp',
            'csr_edoc_asp_signed',
            'csr_extinfo_checked',
            'csr_edoc_agreement',
            'csr_clientdoc_agreement',
            'csr_kmp_agreement',
            'csr_has_proxy',
            'csr_is_poruch',
            'csr_is_zalog',
            'csr_is_clientrelation',
            'csr_has_permission',
            'csr_is_employee',
            'csr_is_yur',
            'csr_is_operbook',
            'csr_is_kmp_user',
            'csr_has_kontract',
            'csr_has_oksummary',
            'csr_has_kfolder',
            'csr_has_miniappclient',
            'csr_is_in_sb_list',
            'csr_has_operation',
            'csr_operfraud',
            'csr_kontsignmen',
            'csr_miniapp_verify',
            'csr_has_fp_anketa_cl',
            'csr_is_knclient'
        )
    ) then
        exception e_default('the wrong field was passed');

    -- Отсеиваем дубликаты
    if (exists
        (
            select 1
            from clientstate_order cso
            where cso.cso_c_id = :c_id
                and cso.cso_upd_field = :field
                and cso.cso_upd_state = :status
        )
    ) then
        exit;

    -- Отсеиваем записи, которые уже имеют передаваемый статус
    flag = 0;
    sql_text = f_format(
    '
        select 1
        from clientstate_ru
        where csr_from_c_id = %1
            and %2 = %3;
    ',
    :c_id, :field, :status);

    execute statement (:sql_text)
    into flag;

    if (flag = 1) then
        exit;


    -- Обработка происходит в процедуре hp_clientstate_order_handle, которая вызывается роботом (fsm_id = 210)
    insert into clientstate_order (cso_c_id, cso_upd_field, cso_upd_state)
    values (:c_id, :field, :status);
end
