CREATE PROCEDURE PRINT_MC_KONTRACT_NO_DEBT(
    K_ID ID_TYPE,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO STR_100,
    C_PASS_NO VARCHAR(30),
    C_PASS_WHO STR_255,
    C_PASS_DATE DATE,
    C_PASS_WHO_CODE STR_20,
    K_KONTRACT_NO STR_20,
    K_KONTRACT_DATE DATE,
    EDOC_DATE D_DATE,
    QR_CODE D_STR_LONG,
    KSR_KONT_TITUL VARCHAR(1024),
    KSR_KONT_POSITION VARCHAR(255),
    KSR_OTHER_1 VARCHAR(255),
    KSR_OTHER_2 VARCHAR(255),
    KSR_OTHER_3 VARCHAR(255),
    KSR_IS_ACTIVE INTEGER,
    KSR_FIO VARCHAR(100),
    HRS_JOBMEMO VARCHAR(255),
    KSR_FROM_C_ID DECIMAL(18,0),
    KSR_FROM_REG_ID INTEGER,
    KSR_ID INTEGER,
    KSR_REGION_NAME VARCHAR(100),
    KSR_REGION_ADDRESS VARCHAR(255),
    KSR_REGION_COMPANY VARCHAR(255),
    KSR_REGIST_NO VARCHAR(255),
    KSR_PASS_NO VARCHAR(30),
    KSR_PASS_DATE DATE,
    KSR_PASS_WHO VARCHAR(255),
    KSR_FULL_ADRESS VARCHAR(255),
    KSR_REG_ADDRESS VARCHAR(255),
    KSR_REG_CITY VARCHAR(255),
    KSR_CAN_SIGN_LOANS INTEGER,
    KSR_REG_COMPANY_NAME STR_255,
    KSR_REG_COMPANY_FULL_NAME STR_255,
    KSR_INN VARCHAR(20),
    C_ALTERNAT_FIO STR_100,
    KSR_QR_CODE D_STR_8100,
    KSR_REG_COMPANY_NAME_SHORT D_STR_255,
    REG_NAME_ALTERNAT D_STR_100,
    KSR_REGION_NAME_SHORT D_STR_100)
AS
declare variable COUNTER smallint;
declare variable OKS_FROM_K_ID ID_TYPE;
declare variable OKS_DATE_END date;
declare variable OKS_CLOSE_STATE smallint;
declare variable EDL_ID D_ID;
declare variable C_ID D_ID;
begin
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);

    select cast(edl.edl_create_datetime as date)
    from edoc_list edl where edl.edl_id = :edl_id
    into :edoc_date;
  end

  select ksr_kont_titul, ksr_kont_position, ksr_other_1, ksr_other_2, ksr_other_3, ksr_is_active, ksr_fio, hrs_jobmemo,
         ksr_from_c_id, ksr_from_reg_id, ksr_id, ksr_region_name, ksr_region_address, ksr_region_company, ksr_regist_no,
         ksr_pass_no, ksr_pass_date, ksr_pass_who, ksr_full_adress, ksr_reg_address, ksr_reg_city, ksr_can_sign_loans,
         ksr_reg_company_name, ksr_reg_company_full_name, ksr_inn, c_alternat_fio, ksr_qr_code,
         ksr_reg_company_name_short, reg_name_alternat, ksr_region_name_short
  from print_company(1000000, 1, current_date, 998)
  into :ksr_kont_titul, :ksr_kont_position, :ksr_other_1, :ksr_other_2, :ksr_other_3, :ksr_is_active, :ksr_fio,
       :hrs_jobmemo, :ksr_from_c_id, :ksr_from_reg_id, :ksr_id, :ksr_region_name, :ksr_region_address,
       :ksr_region_company, :ksr_regist_no, :ksr_pass_no, :ksr_pass_date, :ksr_pass_who, :ksr_full_adress,
       :ksr_reg_address, :ksr_reg_city, :ksr_can_sign_loans, :ksr_reg_company_name, :ksr_reg_company_full_name, :ksr_inn,
       :c_alternat_fio, :ksr_qr_code, :ksr_reg_company_name_short, :reg_name_alternat, :ksr_region_name_short;

  counter = 0;

  select oks_from_k_id, max(oks_date_end), oks_close_state
  from oksummary
  inner join kontstate on ks_from_k_id = oks_from_k_id
  where oks_from_k_id = :k_id
  and ks_ol_is_delete = 0
  and ks_state = 2
  and current_date between ks_date_begin and ks_date_end
  group by 1,3
  order by 2 desc
  rows 1
  into :oks_from_k_id, oks_date_end, oks_close_state;
  counter = row_count;

  if (counter = 0) then
  begin
    select ks_state, ks_date_begin from kontstate
    where ks_from_k_id = :k_id
    and ks_ol_is_delete = 0
    and current_date between ks_date_begin and ks_date_end
    into oks_close_state, oks_date_end;
    counter = row_count;
  end 

  if (counter = 1 and (oks_close_state <> 1 or oks_date_end < current_date)) then
  begin
    for select c_name_fio,
           c_pass_no,
           c_pass_who,
           c_pass_date,
           c_pass_who_code,
           left(k_kontract_no, 14),
           k_kontract_date,
           kc_from_c_id
    from knclient
    inner join kontract k on kc_from_k_id = k.k_id
    inner join clients on c_id = kc_from_c_id
    where k.k_id = :k_id
    and kc_amount_approved > 0
    into c_name_fio,
         c_pass_no,
         c_pass_who,
         c_pass_date,
         c_pass_who_code,
         k_kontract_no,
         k_kontract_date,
         c_id
    do
    begin

      qr_code = f_edoc_qr_by_edl(edl_id, c_id);
      suspend;
    end
  end
end
