CREATE PROCEDURE INSERT_DOVLIST_CLONE(
    DL_ID INTEGER,
    DATE_BEGIN DATE,
    DATE_END DATE,
    UN_ID DECIMAL(18,0))
AS
declare variable new_dl_id integer;
begin
  new_dl_id = gen_id(gen_dovlist_id, 1);
  insert into dovlist (dl_id, dl_date_begin, dl_date_end, dl_date_end_fact, dl_from_dc_id, dl_from_bln_id, dl_from_c_id,
                       dl_manager_id, dl_from_reg_id, dl_main_text, dl_city, dl_memo_1, dl_memo_2, dl_memo_3, dl_memo_4,
                       dl_memo_5, dl_memo_6, dl_memo_7, dl_memo_8, dl_memo_9, dl_memo_10, dl_status, dl_from_un_id)
  select :new_dl_id, :date_begin, :date_end, null, dl_from_dc_id, dl_from_bln_id, dl_from_c_id, dl_manager_id,
         dl_from_reg_id, dl_main_text, dl_city, dl_memo_1, dl_memo_2, dl_memo_3, dl_memo_4, dl_memo_5, dl_memo_6,
         dl_memo_7, dl_memo_8, dl_memo_9, dl_memo_10, 0, :un_id
  from dovlist
  where dl_id = :dl_id;

  insert into dovclients (dvc_from_dl_id, dvc_from_c_id, dvc_from_reg_id)
  select :new_dl_id, dvc_from_c_id, dvc_from_reg_id
  from dovclients
  where dvc_from_dl_id = :dl_id;

end
