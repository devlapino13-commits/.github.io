CREATE PROCEDURE HP_BKI_CORR_FILL_VALUES(
    BEV_ID D_ID,
    FILL_KIND D_SMALLINT,
    SOURCE_BEV_ID D_ID)
AS
declare variable bsd_id d_id;
  declare variable bkd_id d_id;
  declare variable c_id d_id;
  declare variable bkk_id d_id;
  declare variable jso_client d_integer;
  declare variable xml_version_id d_integer;
begin
  -- fill_kind
  -- 1 - заполнить СТАРЫЕ значения данными из события
  -- 2 - заполнить НОВЫЕ значения данными из события
  -- 3 - заполнить СТАРЫЕ значения данными из БД
  -- 4 - заполнить НОВЫЕ значения данными из БД

  if (fill_kind in (1,2)) then
  begin
    for with old_data (old_from_bxf_id, old_from_bkd_id) as (
      select bfv_from_bxf_id, bsd_from_bkd_id
      from bki_send_data_corr, bki_xml_fields_ver
      where bfv_id = bsd_from_bfv_id
        and bsd_from_bev_id = :source_bev_id
    union
      select bfv_from_bxf_id, bsd_from_bkd_id
      from bki_send_data, bki_xml_fields_ver
      where bfv_id = bsd_from_bfv_id
        and bsd_obj_kind = 2
        and bsd_obj_id = :source_bev_id
    )
    select bsd_id, old_from_bkd_id
    from old_data, bki_xml_fields_ver, bki_send_data_corr
    where bfv_from_bxf_id = old_from_bxf_id
      and bsd_from_bfv_id = bfv_id
      and bsd_from_bev_id = :bev_id
    into bsd_id, bkd_id
    do
      if (fill_kind = 1) then
        update bki_send_data_corr
        set bsd_from_bkd_id = :bkd_id
        where bsd_id = :bsd_id;
      else
        update bki_send_data_corr
        set bsd_new_value_id = :bkd_id
        where bsd_id = :bsd_id;
  end
  else if (fill_kind in (3,4)) then
  begin
    select xml_version_id
    from hp_bki_get_current_xml_ver
    into xml_version_id;

    select bkc_from_c_id, bkp_from_bkk_id
    from bki_events, bki_clients, bki_loans, bki_packet
    where bev_id = :bev_id
      and bkc_id = bev_from_bkc_id
      and bkl_id = bkc_from_bkl_id
      and bkp_id = bkl_from_bkp_id
    into c_id, bkk_id;

    jso_client = json.obj_create();
    json.obj_set_s(jso_client, 'bkk_id', :bkk_id);
    execute procedure hp_bki_get_client_info5(:c_id, current_date, :jso_client);

    for select bsd_id, f_bki_data_id(iif(bki_db_field is null, bxf_default_value, json.obj_get_val(:jso_client, bki_db_field)))
    from bki_send_data_corr, bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bsd_from_bev_id = :bev_id
      and bfv_id = bsd_from_bfv_id
      and bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
      and (bki_db_field is null or bki_db_field not starting with '<')
    into bsd_id, bkd_id
    do
      if (fill_kind = 3) then
        update bki_send_data_corr
        set bsd_from_bkd_id = :bkd_id
        where bsd_id = :bsd_id;
      else
        update bki_send_data_corr
        set bsd_new_value_id = :bkd_id
        where bsd_id = :bsd_id;

  end
end
