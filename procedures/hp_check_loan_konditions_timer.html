CREATE PROCEDURE HP_CHECK_LOAN_KONDITIONS_TIMER(
    K_ID DECIMAL(18,0),
    MA_REG_ID INTEGER,
    MA_ID INTEGER,
    CHECK_KIND VARCHAR(100))
RETURNS (
    KF_CLIENT_FIO VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    KF_ERROR_TEXT VARCHAR(255),
    KF_ERROR_IS_CRITICAL INTEGER,
    SQL_ID INTEGER,
    RES_NO INTEGER,
    LAST_K_ID ID_TYPE,
    KC_MAX_AMOUNT CURR_TYPE,
    CURR_K_ID ID_TYPE,
    TIME_DELTA INTEGER)
AS
declare variable sql_text blob sub_type 1 segment size 80;
declare variable t_before integer;
declare variable t_after integer;
begin
  if (check_kind is null) then
    check_kind = '';
  if (check_kind = '.disb.' and
      k_id <> 0) then
  begin
    for select cdm_sql, cdm_id
        from conditionmain
        where cdm_check_kind containing '.disb.' and
              current_date between cdm_date_begin and cdm_date_end and
              cdm_is_active = 1
        into sql_text, sql_id
    do
    begin
      t_before = current_timestamp;
      t_after = 0;
      for execute statement(sql_text)(k_id := k_id)
              into kf_client_fio, kf_client_pass_no, kf_disb_date, kc_amount, kf_error_text, kf_error_is_critical
      do
      begin
        t_after = current_timestamp;
        time_delta = datediff(millisecond, t_before, t_after);
        suspend;
      end
      if (t_after = 0) then
      begin
        t_after = current_timestamp;
        time_delta = datediff(millisecond, t_before, t_after);
        suspend;
      end
    end
    exit;
  end

  if (k_id <> 0) then -- проверка контракта
  begin
    for select cdm_sql, cdm_id
        from conditionmain
        where cdm_check_kind containing '.loan.' and
              current_date between cdm_date_begin and cdm_date_end and
              cdm_is_active = 1
        into sql_text, sql_id
    do
    begin
      t_before = current_timestamp;
      t_after = 0;
      for execute statement(sql_text)(k_id := k_id)
              into kf_client_fio, kf_client_pass_no, kf_disb_date, kc_amount, kf_error_text, kf_error_is_critical
      do
      begin
        t_after = current_timestamp;
        time_delta = datediff(millisecond, t_before, t_after);
        suspend;
      end
      if (t_after = 0) then
      begin
        t_after = current_timestamp;
        time_delta = datediff(millisecond, t_before, t_after);
        suspend;
      end
    end
  end
  else
  begin -- проверка заявки
    for select cdm_sql, cdm_id
        from conditionmain
        where cdm_check_kind containing '.miniapp.' and
              current_date between cdm_date_begin and cdm_date_end and
              cdm_is_active = 1
        into sql_text, sql_id
    do
    begin
      t_before = current_timestamp;
      t_after = 0;
      for execute statement(sql_text)(ma_id := ma_id)
              into kf_client_fio, kf_client_pass_no, kf_disb_date, kc_amount, kf_error_text, kf_error_is_critical
      do
      begin
        t_after = current_timestamp;
        time_delta = datediff(millisecond, t_before, t_after);
        suspend;
      end
      if (t_after = 0) then
      begin
        t_after = current_timestamp;
        time_delta = datediff(millisecond, t_before, t_after);
        suspend;
      end
    end
  end

end
