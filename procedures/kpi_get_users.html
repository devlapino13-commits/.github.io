CREATE PROCEDURE KPI_GET_USERS
RETURNS (
    RES_JSON D_BLOB_TEXT)
AS
declare variable user_list d_blob_text;
declare variable fio d_str_100;
declare variable country_code d_str_10;
declare variable level_  d_smallint;
declare variable is_director  d_bool;
begin
    :res_json = '';
    :user_list = '';

    for select
        ku.ku_fio,
        ku.ku_country_code,
        ku.ku_level,
        ku.ku_is_director
    from kpi_users ku
    into
        :fio,
        :country_code,
        :level_,
        :is_director
    do
    begin
        -- генерирую json объект
        :user_list =
            blob_append(user_list, ascii_char(10) ||
            ascii_char(9) || ascii_char(9) || '{' || ascii_char(10) ||
            ascii_char(9) || ascii_char(9) || ascii_char(9) || '"fio": "' || coalesce(fio, '') || '",' || ascii_char(10) ||
            ascii_char(9) || ascii_char(9) || ascii_char(9) || '"country_code": "' || coalesce(:country_code, '') || '",' || ascii_char(10) ||
            ascii_char(9) || ascii_char(9) || ascii_char(9) || '"level":' || coalesce(level_, '') || ',' || ascii_char(10) ||
            ascii_char(9) || ascii_char(9) || ascii_char(9) || '"is_director":' || iif(is_director = 1, 'true', 'false') || ascii_char(10) ||
            ascii_char(9) || ascii_char(9) || '},');
    end
    :user_list = left(user_list, char_length(user_list) - 1); -- убираю последнюю запятую
    :user_list = user_list || ascii_char(10);

     res_json = blob_append(
                '{', ascii_char(10)
                || ascii_char(9) || '"user_list": ['
                || ascii_char(9) || :user_list
                || ascii_char(9) || ']' || ascii_char(10)
                || '}');

    suspend;
end
