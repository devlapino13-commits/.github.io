CREATE PROCEDURE HP_FOREIGN_ECONOM_GOODS_REF(
    AG_ID D_INTEGER = null,
    K_ID D_ID = null)
RETURNS (
    FEG_ID D_INTEGER,
    FEG_PARENT_ID D_INTEGER,
    FEG_LEVEL D_INTEGER,
    FEG_CODE D_STR_10,
    FEG_NAME D_STR_255,
    FEG_FROM_RD_ID D_INTEGER,
    RD_CODE D_STR_255,
    RD_NAME D_STR_255,
    FEG_MAX_NUMBER CURR_TYPE,
    FEG_AG_ID D_INTEGER,
    FIX_PRICE D_CURRENCY,
    FEG_TAX_ID D_STR_CODE,
    FEG_UNIT_CODE D_STR_10,
    FEG_NEED_IMEI D_BOOL)
AS
declare variable is_north boolean;
declare variable disb_date d_date;
begin
  ag_id = coalesce(ag_id, 0);
  k_id = coalesce(k_id, 0);

  feg_ag_id = ag_id;
  if (ag_id > 0 and not exists(select * from agreements where ag_id = :ag_id))
    then exit;

  if (k_id > 0) then
    select
      k_disb_date,
      reg_part_world_id = f_get_rd_id('part_world_north')
    from kontract
      inner join region
        on reg_unicode = k_last_reg_id
    where k_id = :k_id
    into
      :disb_date,
      :is_north;

  if (ag_id = 0 or not exists(select * from feacagreement_get (:ag_id))) then begin
    for
      select
        feg_id,
        feg_parent_id,
        feg_level,
        feg_code,
        feg_name,
        feg_from_rd_id,
        feg_tax_id,
        feg_unit_code,
        rd_code,
        rd_name,
        feg_max_number,
        0,
        feg_need_imei
      from foreign_econom_goods_ref
        left join refdata
          on rd_id = feg_from_rd_id
    into
        :feg_id,
        :feg_parent_id,
        :feg_level,
        :feg_code,
        :feg_name,
        :feg_from_rd_id,
        :feg_tax_id,
        :feg_unit_code,
        :rd_code,
        :rd_name,
        :feg_max_number,
        :fix_price,
        :feg_need_imei
    do suspend;
  end else begin
    for
      with nomen1 as (
      select feg_id, feg_parent_id
      from foreign_econom_goods_ref
        inner join feacagreement_get (:ag_id)
          on rd_id = feg_from_rd_id
            and feg_code is not null
      ), nomen2 as (
      select a.feg_id, a.feg_parent_id
      from foreign_econom_goods_ref as a
        inner join nomen1 as b
          on b.feg_parent_id = a.feg_id
      ), nomen3 as (
      select a.feg_id, a.feg_parent_id
      from foreign_econom_goods_ref as a
        inner join nomen2 as b
          on b.feg_parent_id = a.feg_id
      )
    select
      n.feg_id,
      n.feg_parent_id,
      n.feg_level,
      n.feg_code,
      n.feg_name,
      n.feg_from_rd_id,
      n.feg_tax_id,
      n.feg_unit_code,
      rd_code,
      rd_name,
      n.feg_max_number,
      n.feg_need_imei
    from foreign_econom_goods_ref as n
      left join refdata
        on rd_id = n.feg_from_rd_id
      inner join (
          select feg_id from nomen1
          union
          select feg_id from nomen2
          union
          select feg_id from nomen3) as c
        on c.feg_id = n.feg_id
    into
      :feg_id,
      :feg_parent_id,
      :feg_level,
      :feg_code,
      :feg_name,
      :feg_from_rd_id,
      :feg_tax_id,
      :feg_unit_code,
      :rd_code,
      :rd_name,
      :feg_max_number,
      :feg_need_imei
    do
    begin
      if (k_id > 0) then
        fix_price = coalesce((
          select
            iif(:is_north, fpr_price_north, fpr_price_south)
          from fertilizerprice
          where fpr_from_ag_id = :ag_id
            and fpr_from_feg = :feg_id
            and :disb_date between fpr_dt_begin and fpr_dt_end ), 0);
      else fix_price = 0;
      suspend;
    end
  end
end
