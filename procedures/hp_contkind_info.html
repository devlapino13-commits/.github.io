CREATE PROCEDURE HP_CONTKIND_INFO(
    CONTACT_KIND D_STR_CODE)
RETURNS (
    RD_ID D_INTEGER,
    RD_CODE D_STR_100,
    CONT_KIND_CODE D_STR_CODE,
    CONT_KIND_ID D_INTEGER,
    CONT_IS_MAIN D_SMALLINT)
AS
begin
  -- cont_kind могут передаваться как старый, так и новый вариант кодовых названий
  -- поэтому конверетируем их между собой

  if (contact_kind is null) then
    exception e_default 'contact_kind is null';

  rd_code = f_enum('cont_kind', contact_kind, 'code', 0);
  if (rd_code is null) then rd_code = '';

  cont_is_main = 0;
  if (rd_code <> '') then -- значит передан enum
  begin
    cont_kind_code = rd_code;

    if (rd_code = 'cell') then
    begin
      cont_is_main = 1;
      rd_code = 'cell_main';
    end
    rd_code = 'contactinfo.' || rd_code;

    select rd_id
    from refdata
    where rd_code = :rd_code
    into rd_id;
  end else
  begin
    -- передан rd_id или rd_code
    if (contact_kind similar to '[-]?[[:DIGIT:]]+') then
    begin
      select rd_id, rd_code
      from refdata
      where rd_id = :contact_kind and
            rd_main_code = 'contactinfo'
      into :rd_id, :rd_code;
    end
    else
    begin
      select rd_id, rd_code
      from refdata
      where rd_code = :contact_kind and
            rd_main_code = 'contactinfo'
      into :rd_id, :rd_code;
    end

    -- кодовое имя конвертируем в enum вариант
    if (rd_id is not null) then
    begin
      cont_kind_code = replace(rd_code, 'contactinfo.', '');
      if (cont_kind_code = 'cell_main') then cont_is_main = 1;
      if (cont_kind_code in ('cell_main', 'cell_add')) then cont_kind_code = 'cell';
    end
  end
  if (rd_id is null) then
    exception e_default 'Incorrect contact_kind: ' || contact_kind;

  cont_kind_id = f_enum('cont_kind', cont_kind_code, 'id');

  suspend;
end
