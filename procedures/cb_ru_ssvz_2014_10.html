CREATE PROCEDURE CB_RU_SSVZ_2014_10(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OKRUG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    OKS_BALANCE CURR_TYPE,
    OKS_DELINQ_BALANE7 CURR_TYPE,
    OKS_DELINQ_BALANE30 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    KOL_CLI_BEGIN INTEGER,
    KOL_CLI_END INTEGER,
    PRIROST_CLI INTEGER,
    BONUS_PRIROST CURR_TYPE,
    SHTRAF_PRIROST CURR_TYPE,
    OKS_BALANCE_LAST CURR_TYPE,
    OKS_DELINQ_BALANE7_LAST DECIMAL(18,4),
    PAR7_LAST CURR_TYPE,
    BONUS_SNIJEN_PAR7 CURR_TYPE,
    SHTRAF_UVELICH_PAR7 CURR_TYPE,
    KOL_SVZ INTEGER,
    AVG_CLI_SVZ CURR_TYPE,
    BONUS_EFFECT CURR_TYPE,
    KOL_SSVZ INTEGER,
    BONUS_END CURR_TYPE,
    BONUS_MEMO VARCHAR(1024),
    SHTRAF_TOTAL DECIMAL(18,4),
    SHTRAF_MEMO VARCHAR(1024))
AS
declare variable rep_date_begin date;
declare variable oks_region_id integer;
declare variable oks_region_id_prior integer;
declare variable prirost_factor integer;
declare variable snijen_par7 curr_type;
declare variable oks_delinq_balane30_last decimal(18,4);
declare variable par30_last decimal(18,4);
begin
  if ((rep_region_id is null) or (rep_date_end is null)) then
    exit;
  rep_date_begin = rep_date_end - extract(day from :rep_date_end);

--  delete from TEMP_KC_REG_INFO;
-- на начало
  insert into temp_kc_reg_info (kc_region_id, kc_kol_cli, kc_kol_cli_sks)
  select ac_region_id, sum(ac_is_client), 0 -- KC_KOL_CLI_SKS - 0 - на дату начало; 1 - на дату конец
  from aarep_cli_active_list(:rep_date_begin)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_begin, null, null, null, null, null, null)) in (1, 2) and
        not exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_begin between sba_date_begin and sba_date_end and
                         sba_from_k_id = ac_kontract_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1 -- находящийся по внешним коллектором не будет считатся
        )
  group by ac_region_id;

-- на конец
  insert into temp_kc_reg_info (kc_region_id, kc_kol_cli, kc_kol_cli_sks)
  select ac_region_id, sum(ac_is_client), 1 -- KC_KOL_CLI_SKS - 0 - на дату начало; 1 - на дату конец
  from aarep_cli_active_list(:rep_date_end)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
        not exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_end between sba_date_begin and sba_date_end and
                         sba_from_k_id = ac_kontract_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1 -- находящийся по внешним коллектором не будет считатся
        )
--  and (exists(select * from knaktlist   -- если не было передано из друго офиса
--              where knaktlist.is_delete = 0 -- т.е. офис на конец предыдущего месяца не поменялся
--              and :rep_date_begin between kal_date and kal_date_end
--              and kal_from_kf_id = ac_kf_id
--              and kal_new_reg_id = ac_region_id)
--       or AC_DISB_DATE between :rep_date_begin + 1 and :rep_date_end) -- или в текущем периоде было выдача
  group by ac_region_id;

  oks_region_id_prior = 0;
  for select oks_region_id, sum(oks_balance),
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0)),
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 30, oks_balance, 0))
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
            oks_region_id = id_list
      group by oks_region_id
      into oks_region_id, oks_balance, oks_delinq_balane7, oks_delinq_balane30
  do
  begin
    if (oks_region_id_prior <> oks_region_id) then
    begin
      oks_region_id_prior = oks_region_id;
      select r3.reg_code || ' ' || r3.reg_name, r2.reg_code || ' ' || r2.reg_name, r1.reg_code || ' ' || r1.reg_name
      from region r1, region r2, region r3
      where r1.reg_unicode = :oks_region_id and
            r2.reg_unicode = r1.parent and
            r3.reg_unicode = r2.parent
      order by r3.reg_code, r2.reg_code, r1.reg_code
      into okrug_name, branch_name, reg_name;
    end

    if (oks_delinq_balane7 = 0) then
    begin
      par7 = 0;
    end
    else
    begin
      par7 = oks_delinq_balane7 / oks_balance * 100;
    end
    if (oks_delinq_balane30 = 0) then
    begin
      par30 = 0;
    end
    else
    begin
      par30 = oks_delinq_balane30 / oks_balance * 100;
    end

-- клиенты на начало
    kol_cli_begin = null;
    select kc_kol_cli
    from temp_kc_reg_info
    where kc_region_id = :oks_region_id and
          kc_kol_cli_sks = 0 -- выборка начало
    into kol_cli_begin;
    if (kol_cli_begin is null) then
      kol_cli_begin = 0;

-- клиенты на конец
    kol_cli_end = null;
    select kc_kol_cli
    from temp_kc_reg_info
    where kc_region_id = :oks_region_id and
          kc_kol_cli_sks = 1 -- выборка конец
    into kol_cli_end;
    if (kol_cli_end is null) then
      kol_cli_end = 0;
-- Расчет прироста клиентов по категории 1
    prirost_cli = kol_cli_end - kol_cli_begin;
    bonus_memo = '';
    shtraf_memo = '';

-- a) Бонус за рост бизнеса с учетом качества по 1 и 2 категориям(офисы, где клиентов более 500)
    prirost_factor = 0;
    shtraf_prirost = 0;
    if (kol_cli_end > 500) then
    begin
      if (prirost_cli <= 10) then
      begin
        prirost_factor = 0;
        shtraf_prirost = 5000;

        bonus_memo = 'Прирост <= 10 клиентов, бонус за прирост не положено; ';
        shtraf_memo = 'Штраф за прирост <= 10 клиентов ';

      end
      else
      begin
        prirost_factor = case
                           when par7 <= 7 then case
                                                 when prirost_cli <= 10 then 0
                                                 when prirost_cli <= 100 then 120
                                                 when prirost_cli <= 200 then 150
                                                 when prirost_cli <= 300 then 160
                                                 when prirost_cli <= 500 then 170
                                                 else 180
                                               end
                           when par30 <= 7 then case
                                                  when prirost_cli <= 10 then 0
                                                  when prirost_cli <= 100 then 100
                                                  when prirost_cli <= 200 then 130
                                                  when prirost_cli <= 300 then 135
                                                  when prirost_cli <= 500 then 140
                                                  else 145
                                                end
                           when par30 <= 15 then case
                                                   when prirost_cli <= 10 then 0
                                                   when prirost_cli <= 100 then 35
                                                   when prirost_cli <= 200 then 40
                                                   when prirost_cli <= 300 then 45
                                                   when prirost_cli <= 500 then 50
                                                   else 55
                                                 end
                           else 0
                         end;
      end
    end

--  Расчет бонуса за рост клиентов
    bonus_prirost = prirost_factor * prirost_cli;
    if (bonus_prirost > 126000) then
      bonus_prirost = 126000;

--  Показатели на конец предыдущего месяца
    oks_balance_last = null;
    oks_delinq_balane7_last = null;
    oks_delinq_balane30_last = null;
    select sum(oks_balance), sum(iif(cast(:rep_date_begin as date) - oks_delinq_date > 7, oks_balance, 0)),
           sum(iif(cast(:rep_date_begin as date) - oks_delinq_date > 30, oks_balance, 0))
    from oksummary
    where :rep_date_begin between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_begin, null, null, null, null, null, null)) in (1, 2) and
          oks_region_id = :oks_region_id
    into oks_balance_last, oks_delinq_balane7_last, oks_delinq_balane30_last;

    if (oks_balance_last is null) then
      oks_balance_last = 0;
    if (oks_delinq_balane7_last is null) then
      oks_delinq_balane7_last = 0;
    if (oks_delinq_balane30_last is null) then
      oks_delinq_balane30_last = 0;

    if (oks_delinq_balane7_last = 0) then
    begin
      par7_last = 0;
    end
    else
    begin
      par7_last = oks_delinq_balane7_last / oks_balance_last * 100;
    end
    if (oks_delinq_balane30_last = 0) then
    begin
      par30_last = 0;
    end
    else
    begin
      par30_last = oks_delinq_balane30_last / oks_balance_last * 100;
    end

--  пукнт 4,1
--  b) Бонус за снижение PAR7 по 1 и 2 категориямм, если PAR7 на конец > 7% (все офисы)
    snijen_par7 = par7_last - par7;
--    snijen_par30 = par30_last - par30;
    bonus_snijen_par7 = 0;
    shtraf_uvelich_par7 = 0;
    if (par7 > 7) then
    begin
      if (snijen_par7 > 0) then
      begin
        bonus_snijen_par7 = trunc(snijen_par7, 0) * 2000;
      end
      else
      if (snijen_par7 < 0) then
      begin
        shtraf_uvelich_par7 = 6000;

        shtraf_memo = shtraf_memo || 'Штраф за ухудшение PAR7; ';
      end
    end

--  c) Бонус за эффективность СВЗ ** по 1 и 2 категориям (Офис, где СВЗ>=4)
--  Вычисляет кол-во СВЗ у которых стаж не менее одного месяца
    kol_svz = null;
    with tabl1
    as (select ps1.ps_from_c_id,
               (select ps2.ps_date_begin
                from hr_pstatus ps2, joblist jl2
                where ps2.ps_from_c_id = ps1.ps_from_c_id and
                      ps2.ps_date_begin is not null and
                      ps2.ps_status = 1 and
                      ps2.ps_from_jl_id = jl2.jl_id and
                      jl2.jl_kind = 5
                order by ps2.ps_date_begin
                rows 1) as ps_stage
        from hr_pstatus ps1, joblist jl1
        where :rep_date_end between ps1.ps_date_begin and ps1.ps_date_end and
              ps1.ps_status = 1 and
              ps1.ps_is_working = 1 and
              ps1.ps_from_reg_id = :oks_region_id and
              ps1.ps_from_jl_id = jl1.jl_id and
              jl1.jl_kind = 5 and
              coalesce((select sawt_id
                        from hr_voctravel, hr_orders, sa_worktype
                        where hrv_from_ho_id = ho_id and
                              ho_order_type = sawt_id and
                              hrv_from_c_id = ps1.ps_from_c_id and
                              :rep_date_end between hrv_date_begin and hrv_date_end_fact
                        order by hrv_date_begin desc
                        rows 1), 0) not in (16, 17, 18) -- Декретников не включает
    )
    select count(*)
    from tabl1
--    where dateadd(month, - 6, :rep_date_end) >= ps_stage   -- стаж ССВЗ не менее 6 месяца пункт 1,4
    where :rep_date_end - ps_stage >= 170 --(170 и более дней стажа на последний день месяца)
    into kol_svz;
    if (kol_svz is null) then
      kol_svz = 0;

    avg_cli_svz = 0;
    bonus_effect = 0;
    if (kol_svz >= 4) then
    begin
      avg_cli_svz = cast(kol_cli_end as double precision) / kol_svz;
      if (par30 <= 7.000) then
      begin
        if (rep_date_end <= '30.11.2014') then
        begin
          if (avg_cli_svz >= 150) then
            bonus_effect = 7500;
        end
        else
        begin
          if (avg_cli_svz >= 170) then
            bonus_effect = 7500;
        end
      end
      else
      begin
        bonus_memo = bonus_memo || 'PAR30>7%, бонус за эффект. СВЗ не положено; ';
      end
    end
    else
    begin
      bonus_memo = bonus_memo || 'Кол-во СВЗ<4, бонус за эффект. СВЗ не положено; ';
    end

    bonus_end = bonus_prirost + bonus_snijen_par7 + bonus_effect;
    shtraf_total = shtraf_prirost + shtraf_uvelich_par7;

--  Вычисляет кол-во СВЗ у которых стаж не менее одного месяца
    kol_ssvz = null;
    with tabl1
    as (select ps1.ps_from_c_id,
               (select ps2.ps_date_begin
                from hr_pstatus ps2, joblist jl2
                where ps2.ps_from_c_id = ps1.ps_from_c_id and
                      ps2.ps_date_begin is not null and
                      ps2.ps_status = 1 and
                      ps2.ps_from_jl_id = jl2.jl_id and
                      jl2.jl_kind = 6
                order by ps2.ps_date_begin
                rows 1) as ps_stage
        from hr_pstatus ps1, joblist jl1
        where :rep_date_end between ps1.ps_date_begin and ps1.ps_date_end and
              ps1.ps_status = 1 and
              ps1.ps_is_working = 1 and
              ps1.ps_from_reg_id = :oks_region_id and
              ps1.ps_from_jl_id = jl1.jl_id and
              jl1.jl_kind = 6)
    select count(*)
    from tabl1
    where dateadd(month, -1, :rep_date_end) >= ps_stage -- стаж ССВЗ не менее 1 месяца пункт 1,4
    into kol_ssvz;
    if (kol_ssvz is null) then
      kol_ssvz = 0;

    suspend;
  end
end
