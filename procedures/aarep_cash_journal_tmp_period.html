CREATE PROCEDURE AAREP_CASH_JOURNAL_TMP_PERIOD(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OL_ID DECIMAL(18,0),
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_CLIENT_NAME VARCHAR(255),
    OL_SUMMA_PLUS DECIMAL(18,4),
    OL_SUMMA_MINUS DECIMAL(18,4),
    OL_ACCOUNT VARCHAR(255),
    OL_KOL_PLUS INTEGER,
    OL_KOL_MINUS INTEGER,
    OL_SALDO_BEGIN_DT DECIMAL(18,4),
    OL_SALDO_BEGIN_KT DECIMAL(18,4),
    OL_SALDO_END_DT DECIMAL(18,4),
    OL_SALDO_END_KT DECIMAL(18,4),
    OL_KIND INTEGER,
    CORR_ACC VARCHAR(20),
    OL_BRANCH_NAME VARCHAR(255),
    OL_DOC_NO_STR VARCHAR(50),
    OL_ACCOUNT_2 VARCHAR(255),
    OL_PARENT_ID ID_TYPE,
    C_IS_RESIDENT INTEGER,
    KC_FIO STR_100)
AS
declare variable ol_document_client decimal(18,0);
declare variable ol_doc_client_name varchar(255);
declare variable total_plus decimal(18,4);
declare variable total_minus decimal(18,4);
declare variable al_id varchar(20);
declare variable ol_document_client2 decimal(18,0);
declare variable total_plus_dd decimal(18,4);
declare variable total_minus_dd decimal(18,4);
declare variable has_line integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable atl_acc_mask varchar(20);
declare variable t_plus decimal(18,4);
declare variable t_minus decimal(18,4);
declare variable reg_id integer;
declare variable ol_oper_type integer;
declare variable ol_oper_subtype integer;
declare variable cm_database_id integer;
declare variable reg_code varchar(20);
begin
  -- knclient_fix

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  :cm_database_id = f_get_db_id();

  has_line = 0;
  ol_kol_plus = 0;
  ol_kol_minus = 0;
  ol_kind = 0;
  ol_summa_minus = 0;
  ol_summa_plus = 0;
  ol_saldo_begin_dt = 0;
  ol_saldo_begin_kt = 0;
  ol_saldo_end_dt = 0;
  ol_saldo_end_kt = 0;
  total_plus = 0;
  total_minus = 0;
  total_minus_dd = 0;
  total_plus_dd = 0;
  ol_doc_no_str = '';

  select reg_kont_company
  from region
  where reg_unicode = :rep_region_id
  into :ol_branch_name;

  /* Начало нового расчета, Sanjari, 19.07.2012 */
  for select atl_acc_mask
      from acctypelist
      where atl_from_at_id = 11
      into :atl_acc_mask
  do
  begin
    for select al_id
        from acc_list, region_list(:rep_region_id, 1, 3, 1)
        where al_from_reg_id = id_list and
              al_from_bg_id = :atl_acc_mask
        into :al_id
    do
    begin
      t_plus = 0;
      t_minus = 0;
      t_plus = f_al_bal(al_id, rep_date_begin - 1);
      total_plus = total_plus + t_plus;
    end

    for select distinct obs_from_al_id
    from obsummary, acc_list, region_list(:rep_region_id, 1, -1, 1)
    where obs_from_al_id = al_id and
          al_from_reg_id = id_list and
          obs_date between :rep_date_begin and :rep_date_end and
          al_from_bg_id = :atl_acc_mask
    into al_id
    do
    begin
      select dt_ekv, kt_ekv
      from hp_al_obor(:al_id, :rep_date_begin, :rep_date_end)
      into :t_plus, :t_minus;

      total_plus_dd = total_plus_dd + t_plus;
      total_minus_dd = total_minus_dd + t_minus;
    end
  end

  if (total_plus - total_minus >= 0) then
  begin
    ol_saldo_begin_dt = total_plus - total_minus;
    ol_saldo_begin_kt = 0;
  end
  else
  begin
    ol_saldo_begin_dt = 0;
    ol_saldo_begin_kt = total_minus - total_plus;
  end

  if (total_plus + total_plus_dd - total_minus - total_minus_dd >= 0) then
  begin
    ol_saldo_end_dt = total_plus + total_plus_dd - total_minus - total_minus_dd;
    ol_saldo_end_kt = 0;
  end
  else
  begin
    ol_saldo_end_dt = 0;
    ol_saldo_end_kt = 0 - (total_plus + total_plus_dd - total_minus - total_minus_dd);
  end

  for select atl_acc_mask
      from acctypelist
      where atl_from_at_id = 11
      into :atl_acc_mask
  do
  begin
    for select al_id, al_from_reg_id
        from acc_list, region_list(:rep_region_id, 1, 3, 1)
        where al_from_reg_id = id_list and
              al_from_bg_id = :atl_acc_mask
        into :al_id, :reg_id
    do
    begin
      ol_kind = 1;
      ol_summa_minus = 0;
      for select ol_id, ol_date, ol_doc_no, ol_document_client, ol_doc_client_name, al_account, ob_summa_val,
                 ob_client_kt, ob_account_kt, ol_oper_type, ol_oper_subtype, reg_code
          from operlist, operbook, acc_list, region
          where ob_ol_date between :rep_date_begin and :rep_date_end and
                ob_ol_is_active = 1 and
                ol_id = ob_from_ol_id and
                ob_ol_reg_id = :reg_id and
                ob_account_dt = :al_id and
                al_id = ob_account_kt and
                ol_from_reg_id = reg_unicode and
                ob_summa <> 0
          into :ol_id, :ol_date, :ol_doc_no, :ol_document_client, :ol_doc_client_name, :ol_account, :ol_summa_plus,
               :ol_document_client2, :corr_acc, :ol_oper_type, :ol_oper_subtype, :reg_code
      do
      begin
        ol_parent_id = null;
        if (cm_database_id = 81001) then -- igusev 20.01.2015 добавил условие, согласно письма от Ларисы от 20.01.15 12:34
        begin
          if (ol_account = '76-05/4') then
            ol_account = '60323';
          ol_account = left(ol_account, 5);
        end

        if ((ol_account = '57-04') and
            (cm_database_id = 81001)) then
        begin
          ol_parent_id = :ol_id;

          for select ob.ob_from_ol_id, left(al_account, 5), ob_summa_val, ob_client_kt, ob_account_kt, ol_oper_type,
                     ol_oper_subtype
              from operbook ob
              inner join operlist ol on ol.ol_id = ob.ob_from_ol_id
              inner join acc_list acl on acl.al_id = ob_account_kt
              where ol.ol_parent_id = :ol_parent_id and
                    ol.is_delete = 0 and
                    ((ol.ol_oper_subtype = 2) or
                    (ol.ol_oper_subtype = 8 and
                    ob_short_acc_kt not containing '57-04'))
              into :ol_id, :ol_account, :ol_summa_plus, :ol_document_client2, :corr_acc, :ol_oper_type,
                   :ol_oper_subtype
          do
          begin
            for select ol_client_name, ol_kol_plus, has_line, ol_doc_no_str, ol_account_2, ol_summa_plus, c_is_resident,
                       kc_fio
                from aarep_cash_journal_subtmp_1(:ol_doc_client_name, :ol_document_client, :ol_document_client2, :cm_database_id, :ol_date, :ol_oper_type, :ol_oper_subtype, :ol_id, :ol_kol_plus, :has_line, :ol_doc_no_str, :reg_code, :ol_doc_no, :ol_account, :ol_summa_plus)
                into :ol_client_name, :ol_kol_plus, :has_line, :ol_doc_no_str, :ol_account_2, :ol_summa_plus,
                     :c_is_resident, :kc_fio
            do
            begin
              suspend;
            end
          end
        end
        else
        begin
          for select ol_client_name, ol_kol_plus, has_line, ol_doc_no_str, ol_account_2, ol_summa_plus, c_is_resident,
                     kc_fio
              from aarep_cash_journal_subtmp_1(:ol_doc_client_name, :ol_document_client, :ol_document_client2, :cm_database_id, :ol_date, :ol_oper_type, :ol_oper_subtype, :ol_id, :ol_kol_plus, :has_line, :ol_doc_no_str, :reg_code, :ol_doc_no, :ol_account, :ol_summa_plus)
              into :ol_client_name, :ol_kol_plus, :has_line, :ol_doc_no_str, :ol_account_2, :ol_summa_plus,
                   :c_is_resident, :kc_fio
          do
          begin
            suspend;
          end
        end
      end

      ol_kind = 2;
      ol_summa_plus = 0;

      for select ol_id, ol_date, ol_doc_no, ol_document_client, ol_doc_client_name, al_account, ob_summa_val,
                 ob_client_dt, ob_account_dt, ol_oper_type, ol_oper_subtype, reg_code
          from operlist, operbook, acc_list, region
          where ob_ol_date between :rep_date_begin and :rep_date_end and
                ob_ol_is_active = 1 and
                ol_id = ob_from_ol_id and
                ob_ol_reg_id = :reg_id and
                ob_account_kt = :al_id and
                al_id = ob_account_dt and
                ol_from_reg_id = reg_unicode and
                ob_summa <> 0
          into :ol_id, :ol_date, :ol_doc_no, :ol_document_client, :ol_doc_client_name, :ol_account, :ol_summa_minus,
               :ol_document_client2, :corr_acc, :ol_oper_type, :ol_oper_subtype, :reg_code
      do
      begin
        if (cm_database_id = 81001) then -- igusev 20.01.2015 добавил условие, согласно письма от Ларисы от 20.01.15 12:34
        begin
          if (ol_account = '76-05/4') then
            ol_account = '60323';
          ol_account = left(ol_account, 5);
        end

        if ((ol_account = '57-04') and
            (cm_database_id = 81001)) then
        begin
          ol_parent_id = :ol_id;

          for select ob.ob_from_ol_id, al_account, iif(ol_oper_subtype = 20, 0, ob_summa_val), ob_client_kt,
                     ob_account_kt, ol_oper_type, ol_oper_subtype
              from operbook ob
              inner join operlist ol on ol.ol_id = ob.ob_from_ol_id
              inner join acc_list acl on acl.al_id = ob_account_dt
              where ol.ol_parent_id = :ol_parent_id and
                    ol.is_delete = 0 and
                    (ol.ol_oper_subtype = 2 or
                    (ol.ol_oper_subtype = 20 and
                    ob_short_acc_kt containing '57-04' and
                    ob_short_acc_dt containing '51-01'))
              into :ol_id, :ol_account, :ol_summa_plus, :ol_document_client2, :corr_acc, :ol_oper_type,
                   :ol_oper_subtype
          do
          begin
            for select ol_client_name, ol_kol_minus, has_line, ol_doc_no_str, ol_account_2, ol_summa_minus,
                       c_is_resident, kc_fio
                from aarep_cash_journal_subtmp_2(:ol_doc_client_name, :ol_document_client, :ol_document_client2, :cm_database_id, :ol_date, :ol_oper_type, :ol_oper_subtype, :ol_id, :ol_kol_minus, :has_line, :ol_doc_no_str, :reg_code, :ol_doc_no, :ol_account, :ol_summa_minus)
                into :ol_client_name, :ol_kol_minus, :has_line, :ol_doc_no_str, :ol_account_2, :ol_summa_minus,
                     :c_is_resident, :kc_fio
            do
            begin
              suspend;
            end
          end
        end
        else
        begin
          for select ol_client_name, ol_kol_minus, has_line, ol_doc_no_str, ol_account_2, ol_summa_minus, c_is_resident,
                     kc_fio
              from aarep_cash_journal_subtmp_2(:ol_doc_client_name, :ol_document_client, :ol_document_client2, :cm_database_id, :ol_date, :ol_oper_type, :ol_oper_subtype, :ol_id, :ol_kol_minus, :has_line, :ol_doc_no_str, :reg_code, :ol_doc_no, :ol_account, :ol_summa_minus)
              into :ol_client_name, :ol_kol_minus, :has_line, :ol_doc_no_str, :ol_account_2, :ol_summa_minus,
                   c_is_resident, :kc_fio
          do
          begin
            suspend;
          end
        end
      end
    end
  end
  if (has_line = 0) then
  begin
    suspend;
  end
end
