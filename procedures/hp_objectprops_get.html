CREATE PROCEDURE HP_OBJECTPROPS_GET(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID VARCHAR(255),
    SUBJECT_KIND VARCHAR(100),
    SUBJECT_ID VARCHAR(255),
    REP_DATE DATE)
RETURNS (
    OVS_FROM_OPR_ID DECIMAL(18,0),
    OVS_ID INTEGER,
    OVS_VALUE VARCHAR(255),
    OVS_ADD_VALUE DECIMAL(18,4),
    OVS_DATE DATE,
    OVS_DATETIME TIMESTAMP,
    OPR_ACTUAL_OVS_ID INTEGER,
    OPR_FACT_OVS_ID INTEGER,
    OVS_FROM_UN_ID DECIMAL(18,0),
    OVS_ORIG_VALUE_STR VARCHAR(1024),
    RES_RD_CODE VARCHAR(255),
    RES_RD_ID INTEGER,
    OVS_VALUE_FOR_REP VARCHAR(255),
    IS_HISTORY INTEGER)
AS
declare variable RD_CODE_MIN varchar(255);
declare variable RD_CODE_MAX varchar(255);
declare variable RD_ID_MIN integer;
declare variable RD_ID_MAX integer;
declare variable ANSWER_ACTUALITY_DAYS integer;
declare variable DD_TMP date;
begin
-- процедура устанавливает получает значение для пары object/subject
  is_history = 0;

  if (rep_date >= '31.12.2999') then
    rep_date = null;

  answer_actuality_days = 0;
  if (subject_kind = '-1' or subject_kind = 'Refdata') then
  begin
    if (subject_id similar to '[-]?[[:DIGIT:]]+') then
    begin
      select rd_tag
      from refdata
      where rd_id = :subject_id
      into :answer_actuality_days;
    end
    else
    begin
      select rd_tag
      from refdata
      where rd_code = :subject_id
      into :answer_actuality_days;
    end
  end
  if (answer_actuality_days is null) then
    answer_actuality_days = 0;

  if (rep_date is null and
      object_kind similar to '[-]?[[:DIGIT:]]+' and
      object_id similar to '[-]?[[:DIGIT:]]+' and
      subject_kind similar to '[-]?[[:DIGIT:]]+' and
      subject_id similar to '[-]?[[:DIGIT:]]+') then
  begin
    select ovs_from_opr_id, opr_actual_ovs_id, opr_fact_ovs_id, ovs_id, ovs_value, ovs_add_value, ovs_date_begin,
           ovs_datetime, ovs_from_un_id
    from objectvalues, objectprops
    where opr_object_kind = :object_kind and
          opr_object_id = :object_id and
          opr_subject_kind = :subject_kind and
          opr_subject_id = :subject_id and
          ovs_from_opr_id = opr_id and
          ovs_id = opr_actual_ovs_id
    into :ovs_from_opr_id, opr_actual_ovs_id, opr_fact_ovs_id, :ovs_id, :ovs_value, :ovs_add_value, ovs_date,
         ovs_datetime, ovs_from_un_id;

    if (answer_actuality_days > 0) then
    begin
      dd_tmp = current_date;
      if (datediff(day, ovs_date, dd_tmp) > answer_actuality_days) then
        exit;
    end
    ovs_orig_value_str = '';
    ovs_value_for_rep = '';
    rd_code_min = '';
    rd_code_max = '';
    res_rd_code = '';
    res_rd_id = 0;
    rd_id_min = 0;
    rd_id_max = 0;
    select list(iif(ovr_value <> '', rd_name || ': ' || ovr_value, rd_name), '; '), list(rd_name, '; '), min(rd_code),
           max(rd_code), min(ovr_result_rd_id), max(ovr_result_rd_id)
    from objectvalres, refdata
    where ovr_from_opr_id = :ovs_from_opr_id and
          ovr_from_ovs_id = :ovs_id and
          ovr_result_rd_id = rd_id
    into ovs_orig_value_str, :ovs_value_for_rep, :rd_code_min, rd_code_max, rd_id_min, rd_id_max;
    if (rd_code_min is null) then
      rd_code_min = '';
    if (rd_code_max is null) then
      rd_code_max = '';
    if (rd_id_min is null) then
      rd_id_min = 0;
    if (rd_id_max is null) then
      rd_id_max = 0;
    if (rd_code_min <> '' and
        rd_code_min = rd_code_max) then
      res_rd_code = :rd_code_min;
    if (rd_id_min <> 0 and
        rd_id_min = rd_id_max) then
      res_rd_id = :rd_id_min;
    if (ovs_orig_value_str is null) then
      ovs_orig_value_str = '';
    ovs_orig_value_str = left(ovs_orig_value_str, 255);
    if (ovs_orig_value_str <> '') then
      ovs_value = left(ovs_orig_value_str, 255);
    else
      ovs_orig_value_str = ovs_value;

    if (ovs_from_opr_id is not null) then
    begin
      if (exists(select *
                 from objectvalues, objectprops
                 where opr_object_kind = :object_kind and
                       opr_object_id = :object_id and
                       opr_subject_kind = :subject_kind and
                       opr_subject_id = :subject_id and
                       ovs_from_opr_id = opr_id and
                       ovs_id > 1)) then
        is_history = 1;
      suspend;
    end
    exit;
  end

  select ovs_from_opr_id, opr_actual_ovs_id, opr_fact_ovs_id
  from hp_objectprops_get_opr_id(:object_kind, :object_id, :subject_kind, :subject_id)
  into :ovs_from_opr_id, opr_actual_ovs_id, opr_fact_ovs_id;
  if (ovs_from_opr_id is null) then
    exit;

  if (rep_date is not null) then
  begin
    select ovs_id, ovs_value, ovs_add_value, ovs_date_begin, ovs_datetime, ovs_from_un_id
    from objectvalues
    where ovs_from_opr_id = :ovs_from_opr_id and
          ovs_date_begin <= :rep_date and
          ovs_date_end >= :rep_date and
          ovs_is_active = 1
    into :ovs_id, :ovs_value, :ovs_add_value, ovs_date, ovs_datetime, ovs_from_un_id;

    if (answer_actuality_days > 0) then
    begin
      dd_tmp = rep_date;
      if (datediff(day, ovs_date, dd_tmp) > answer_actuality_days) then
        exit;
    end

    ovs_orig_value_str = '';
    rd_code_min = '';
    rd_code_max = '';
    res_rd_code = '';
    res_rd_id = 0;
    rd_id_min = 0;
    rd_id_max = 0;
    select list(iif(ovr_value <> '', rd_name || ': ' || ovr_value, rd_name), '; '), list(rd_name, '; '), min(rd_code),
           max(rd_code), min(ovr_result_rd_id), max(ovr_result_rd_id)
    from objectvalres, refdata
    where ovr_from_opr_id = :ovs_from_opr_id and
          ovr_from_ovs_id = :ovs_id and
          ovr_result_rd_id = rd_id
    into ovs_orig_value_str, :ovs_value_for_rep, :rd_code_min, rd_code_max, rd_id_min, rd_id_max;
    if (rd_code_min is null) then
      rd_code_min = '';
    if (rd_code_max is null) then
      rd_code_max = '';
    if (rd_id_min is null) then
      rd_id_min = 0;
    if (rd_id_max is null) then
      rd_id_max = 0;
    if (rd_code_min <> '' and
        rd_code_min = rd_code_max) then
      res_rd_code = :rd_code_min;
    if (rd_id_min <> 0 and
        rd_id_min = rd_id_max) then
      res_rd_id = :rd_id_min;

    if (ovs_orig_value_str is null) then
      ovs_orig_value_str = '';
    ovs_orig_value_str = left(ovs_orig_value_str, 255);
    if (ovs_orig_value_str <> '') then
      ovs_value = left(ovs_orig_value_str, 255);
    else
      ovs_orig_value_str = ovs_value;
  end
  else
  begin
    select ovs_id, ovs_value, ovs_add_value, ovs_date_begin, ovs_datetime, ovs_from_un_id
    from objectvalues
    where ovs_from_opr_id = :ovs_from_opr_id and
          ovs_id = :opr_actual_ovs_id
    into :ovs_id, :ovs_value, :ovs_add_value, ovs_date, ovs_datetime, ovs_from_un_id;

    if (answer_actuality_days > 0) then
    begin
      dd_tmp = current_date;
      if (datediff(day, ovs_date, dd_tmp) > answer_actuality_days) then
        exit;
    end

    ovs_orig_value_str = '';
    rd_code_min = '';
    rd_code_max = '';
    res_rd_code = '';
    res_rd_id = 0;
    rd_id_min = 0;
    rd_id_max = 0;
    select list(iif(ovr_value <> '', rd_name || ': ' || ovr_value, rd_name), '; '), list(rd_name, '; '), min(rd_code),
           max(rd_code), min(ovr_result_rd_id), max(ovr_result_rd_id)
    from objectvalres, refdata
    where ovr_from_opr_id = :ovs_from_opr_id and
          ovr_from_ovs_id = :ovs_id and
          ovr_result_rd_id = rd_id
    into ovs_orig_value_str, :ovs_value_for_rep, :rd_code_min, rd_code_max, rd_id_min, rd_id_max;
    if (rd_code_min is null) then
      rd_code_min = '';
    if (rd_code_max is null) then
      rd_code_max = '';
    if (rd_id_min is null) then
      rd_id_min = 0;
    if (rd_id_max is null) then
      rd_id_max = 0;
    if (rd_code_min <> '' and
        rd_code_min = rd_code_max) then
      res_rd_code = :rd_code_min;
    if (rd_id_min <> 0 and
        rd_id_min = rd_id_max) then
      res_rd_id = :rd_id_min;

    if (ovs_orig_value_str is null) then
      ovs_orig_value_str = '';
    if (ovs_orig_value_str <> '') then
      ovs_value = left(ovs_orig_value_str, 255);
    else
      ovs_orig_value_str = ovs_value;
  end
  suspend;
end
