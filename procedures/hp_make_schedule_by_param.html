Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_MAKE_SCHEDULE_BY_PARAM(
    LOAN_DISB_DATE D_DATE,
    LOAN_PRODUCT_ID D_INTEGER,
    LOAN_AMOUNT D_CURRENCY,
    LOAN_PERCENT D_PERCENT,
    LOAN_KOMMIS_PERC D_PERCENT,
    LOAN_REPAY_KOLICH D_INTEGER,
    LOAN_REPAY_DAY D_INTEGER)
RETURNS (
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    SH_OS D_CURRENCY,
    SH_PERC D_CURRENCY,
    SH_TOTAL D_CURRENCY,
    SH_KOMMIS D_CURRENCY,
    SH_PERC_TAX D_CURRENCY,
    SH_DISB_AMOUNT D_CURRENCY,
    SH_BALANCE D_CURRENCY)
AS
declare variable pr_percent_for_days integer;
declare variable calc_kommis_lgota decimal(18,4) = 0;
declare variable pr_round integer;
declare variable pr_round_perc integer;
declare variable pr_round_kommis integer;
begin
  -- Построение графика погашений по переданным параметрам

  -- Получаем необходимые настройки кредитного продукта
  select pr_percent_for_days, pr_round, pr_round_perc, pr_round_admin
  from product
  where pr_id = :loan_product_id
  into :pr_percent_for_days, :pr_round, :pr_round_perc, :pr_round_kommis;

  pr_round = -pr_round;
  pr_round_perc = -pr_round_perc;
  pr_round_kommis = -pr_round_kommis;

  if (coalesce(pr_percent_for_days, 0) = 0) then
    exit;

  delete from temp_schedule
  where 1 = 1;

  -- Вычисляем сумму доп.процента с учетом предоставленных скидок
  calc_kommis_lgota = 0;

  execute procedure hp_make_schedule_date(:loan_amount, :loan_repay_kolich, :loan_repay_day, :loan_disb_date,
      :loan_product_id, :loan_kommis_perc, :calc_kommis_lgota, 'new', 0, 0);

  execute procedure hp_make_schedule_perc(:loan_amount, :loan_repay_kolich, :loan_product_id, :loan_percent,
      :loan_kommis_perc, :calc_kommis_lgota, 0);

  sh_perc_tax = 0;
  sh_balance = 0;
  for select sh_id, sh_date, sh_os, sh_perc, sh_kommis,
            (sh_os + sh_perc + sh_kommis),
            iif(sh_id = 0, :loan_amount, 0)
  from temp_schedule
  into :sh_id, :sh_date, :sh_os, :sh_perc, :sh_kommis, sh_total, sh_disb_amount
  do
  begin
    sh_balance = sh_balance + sh_disb_amount - sh_os;
    suspend;
  end

  delete from temp_schedule
  where 1 = 1;
end
