CREATE PROCEDURE HP_CHECK_LOAN_KONDITIONS(
    K_ID D_ID,
    MA_REG_ID D_INTEGER,
    MA_ID D_INTEGER,
    CHECK_KIND D_STR_100,
    UN_ID D_ID = null)
RETURNS (
    KF_CLIENT_FIO D_STR_100,
    KF_ERROR_TEXT D_STR_255,
    KF_ERROR_IS_CRITICAL D_INTEGER,
    SQL_ID D_INTEGER,
    RES_NO D_INTEGER)
AS
declare variable fincert_res d_smallint;
declare variable fincert_text d_str_255;
declare variable c_id d_id;
begin
  if (check_kind is null) then
    check_kind = '';
  if (check_kind = '' and
      k_id <> 0) then
    check_kind = '.loan.';
  if (check_kind = '' and
      ma_id <> 0) then
    check_kind = '.miniapp.';

  if (left(check_kind, 1) <> '.') then
    check_kind = '.' || check_kind;
  if (right(check_kind, 1) <> '.') then
    check_kind = check_kind || '.';

  -- disb - проверка возможности выдачи
  -- mdisb - проверка выдачи без проверки кассовых остатков
  if (un_id is null) then
  begin
    un_id = (select un_id
             from username
             where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN'));
    if (un_id is null) then
      un_id = 1000000000000;
  end
  res_no = 0;

  if (check_kind in ('.check_preview.', '.loan_approve.', '.send_kk.', '.loan_disb.')) then
  begin
    -- выполнить расчет ПДН при отправке на КК до проверки блоков
    if (f_get_db_id() = 81001 and check_kind in ('.check_preview.', '.send_kk.', '.loan_approve.')) then
      begin
        select err_text
        from hp_knclient_pdn_validate(:k_id, replace(:check_kind, '.', ''))
        into kf_error_text;
    
        if (kf_error_text <> '') then
        begin
          kf_error_is_critical = 2;
          kf_client_fio = 'Ошибка расчета ПДН';
  
          res_no = res_no + 1;
          suspend;
          exit;
        end
    end

    if (f_get_db_id() = 81001 and check_kind <> '.loan_disb.') then
    begin
      --RFCRU-3955 автоматический отказ при совпадении со списками финЦЕРТ - номер паспорта, ИНН
      select res, res_text
      from check_fincert(null, :k_id, replace(:check_kind, '.', ''))
      into :fincert_res, fincert_text;

      if (fincert_res = 1) then
      begin
        kf_error_is_critical = 2;
        kf_client_fio = fincert_text;

        res_no = res_no + 1;
        suspend;
        exit;
      end

      execute procedure hp_antifrod_check(k_id);
    end

    -- Проверка документов по новой схеме производится независимо отключена проверка или нет
    for select kf_client_fio, kf_error_text, kf_error_is_critical, sql_id
        from hp_check_loan_kondition_one(:k_id, :ma_id, 0, :check_kind, :un_id)
        into :kf_client_fio, :kf_error_text, :kf_error_is_critical, :sql_id
    do
    begin
      res_no = res_no + 1;
      suspend;
    end
  end

  if (f_get_db_id() = 81001 and check_kind in ('.create_loan.')) then
  begin
    -- RFCRU-3894
    for select mac_from_c_id
    from miniappclient
    where mac_from_ma_id = :ma_id
    into c_id
    do
      execute procedure hp_extinfo_insert_smevpass(:c_id, 0, :un_id);
  end
   
  for select cdm_id
      from conditionmain
      where cdm_check_kind containing :check_kind and
            current_date between cdm_date_begin and cdm_date_end and
            cdm_is_active = 1 and
            cdm_id > 0 -- Кроме проверки документов по новой схеме
      into sql_id
  do
  begin
    for select kf_client_fio, kf_error_text, kf_error_is_critical, sql_id
        from hp_check_loan_kondition_one(:k_id, :ma_id, :sql_id, :check_kind, :un_id)
        into :kf_client_fio, :kf_error_text, :kf_error_is_critical, :sql_id
    do
    begin
      res_no = res_no + 1;
      suspend;
    end
  end
end
