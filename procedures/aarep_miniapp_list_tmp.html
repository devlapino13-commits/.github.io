CREATE PROCEDURE AAREP_MINIAPP_LIST_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DATE_KIND INTEGER,
    FIND_KF_NAME VARCHAR(100),
    FIND_MA_NO INTEGER,
    FIND_C_FIO VARCHAR(100))
RETURNS (
    MA_FROM_REG_ID INTEGER,
    MA_ID INTEGER,
    MA_NUMBER INTEGER,
    MA_DATE_APP DATE,
    MA_DATE_OPER DATE,
    MA_FROM_KF_ID DECIMAL(18,0),
    MA_FROM_PR_ID INTEGER,
    MA_KOLICH_REPAY INTEGER,
    MA_PERCENT DECIMAL(18,7),
    MA_KOLICH_CLIENT INTEGER,
    MA_MEMO VARCHAR(255),
    MA_IS_ACTIVE INTEGER,
    MA_IS_AUTO INTEGER,
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_DATE_SAVE TIMESTAMP,
    MA_FROM_UN_ID DECIMAL(18,0),
    MA_OFFICE_ID INTEGER,
    MA_CURRENCY_ID VARCHAR(3),
    MA_NO INTEGER,
    MA_PERCENT_CATEGORY DECIMAL(18,7),
    MA_MANAGER_ID DECIMAL(18,0),
    MA_PLAN_DISB_TIME TIMESTAMP,
    MA_EXPERT_ID DECIMAL(18,0),
    MA_KOMMIS_PERC DECIMAL(18,7))
AS
declare variable ma_id_prior integer;
declare variable can_suspend integer;
declare variable kf_id decimal(18,0);
declare variable ap_ma_id integer;
declare variable c_id decimal(18,0);
declare variable kol integer;
declare variable max_kol integer;
declare variable id_list integer;
begin
  ma_id_prior = 0;
  kol = 0;
  max_kol = 500;
  ma_from_reg_id = 0;

  if (find_kf_name <> '') then
    find_kf_name = upper(:find_kf_name);
  if (find_c_fio <> '') then
    find_c_fio = upper(:find_c_fio);
  find_c_fio = replace(find_c_fio, '%', '');

  if ((find_kf_name <> '') or (find_c_fio <> '') or (find_ma_no > 0)) then
  begin
    /* поиск по критериям */
    if (find_ma_no > 0) then
    /* поиск по номеру заявки */
    begin
      for select ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay, ma_percent,
                 ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save, ma_from_un_id,
                 ma_office_id, ma_currency_id, ma_no, ma_percent_category, ma_manager_id, ma_plan_disb_time,
                 ma_expert_id, ma_kommis_perc
          from miniapp
          where ma_no = :find_ma_no and
                ma_date_end = '31.12.9999'
          order by ma_id desc, ma_number desc
          into :ma_id, :ma_number, :ma_date_app, :ma_date_oper, :ma_from_kf_id, :ma_from_pr_id, :ma_kolich_repay,
               :ma_percent, :ma_kolich_client, :ma_memo, :ma_is_active, :ma_is_auto, :ma_summa_minus, :ma_date_save,
               :ma_from_un_id, :ma_office_id, :ma_currency_id, :ma_no, :ma_percent_category, :ma_manager_id,
               :ma_plan_disb_time, :ma_expert_id, :ma_kommis_perc
      do
      begin
        ma_is_auto = 0;
        can_suspend = 0;
        if (ma_id <> ma_id_prior) then
        begin
          if (rep_date_kind = 0) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 1) and
              (ma_date_app between :rep_date_begin and :rep_date_end)) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 2) and
              (ma_date_oper between :rep_date_begin and :rep_date_end)) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 3) and
              (ma_date_save between :rep_date_begin and (:rep_date_end + 1))) then
            can_suspend = 1;
        end
        if (can_suspend = 1) then
        begin
          suspend;
          if (kol >= max_kol) then
            exit;
        end
        ma_id_prior = :ma_id;
      end
    end /* конец поиска по номеру заявки */

    if (find_kf_name <> '') then
    /* поиск по названию папки */
    begin
      for select distinct kal_from_kf_id, id_list
          from knaktlist, region_list(:rep_region_id, 1, -1, 1)
          where knaktlist.is_delete = 0 and
                kal_date <= :rep_date_end and
                kal_date_end >= :rep_date_end and
                kal_new_kf_name starting :find_kf_name and
                kal_new_reg_id = id_list
          into :kf_id, :id_list
      do
      begin
        for select ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay,
                   ma_percent, ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save,
                   ma_from_un_id, ma_office_id, ma_currency_id, ma_no, ma_percent_category, ma_manager_id,
                   ma_plan_disb_time, ma_expert_id, ma_kommis_perc
            from miniapp
            where ma_from_kf_id = :kf_id and
                  ma_office_id = :id_list and
                  ma_date_end = '31.12.9999'
            order by ma_id desc, ma_number desc
            into :ma_id, :ma_number, :ma_date_app, :ma_date_oper, :ma_from_kf_id, :ma_from_pr_id, :ma_kolich_repay,
                 :ma_percent, :ma_kolich_client, :ma_memo, :ma_is_active, :ma_is_auto, :ma_summa_minus, :ma_date_save,
                 :ma_from_un_id, :ma_office_id, :ma_currency_id, :ma_no, :ma_percent_category, :ma_manager_id,
                 :ma_plan_disb_time, :ma_expert_id, :ma_kommis_perc
        do
        begin
          ma_is_auto = 0;
          can_suspend = 0;
          if (ma_id <> ma_id_prior) then
          begin
            if (rep_date_kind = 0) then
              can_suspend = 1;
            else
            if ((rep_date_kind = 1) and
                (ma_date_app between :rep_date_begin and :rep_date_end)) then
              can_suspend = 1;
            else
            if ((rep_date_kind = 2) and
                (ma_date_oper between :rep_date_begin and :rep_date_end)) then
              can_suspend = 1;
            else
            if ((rep_date_kind = 3) and
                (ma_date_save between :rep_date_begin and (:rep_date_end + 1))) then
              can_suspend = 1;
          end
          if (can_suspend = 1) then
          begin
            suspend;
            if (kol >= max_kol) then
              exit;
          end
        end
      end
    end /* конец поиска по названию папки */

    if (find_c_fio <> '') then
    /* поиск по фио клиента */
    begin
      for select distinct object_id
          from hp_find_from_namelist(0, 0, :find_c_fio)
          into :c_id
      do
      begin
        for select distinct mac_from_ma_id
            from miniappclient
            where mac_from_c_id = :c_id and
                  mac_is_poruch = 0 /* не поручитель*/
            into :ap_ma_id
        do
        begin
          for select distinct ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id,
                              ma_kolich_repay, ma_percent, ma_kolich_client, ma_memo, ma_is_active, ma_is_auto,
                              ma_summa_minus, ma_date_save, ma_from_un_id, ma_office_id, ma_currency_id, ma_no,
                              ma_percent_category, ma_manager_id, ma_plan_disb_time, ma_expert_id, ma_kommis_perc
              from miniapp
              where ma_id = :ap_ma_id and
                    ma_date_end = '31.12.9999'
              order by ma_id desc, ma_number desc
              into :ma_id, :ma_number, :ma_date_app, :ma_date_oper, :ma_from_kf_id, :ma_from_pr_id, :ma_kolich_repay,
                   :ma_percent, :ma_kolich_client, :ma_memo, :ma_is_active, :ma_is_auto, :ma_summa_minus, :ma_date_save,
                   :ma_from_un_id, :ma_office_id, :ma_currency_id, :ma_no, :ma_percent_category, :ma_manager_id,
                   :ma_plan_disb_time, :ma_expert_id, :ma_kommis_perc
          do
          begin
            ma_is_auto = 0;
            can_suspend = 0;
            if (ma_id <> ma_id_prior) then
            begin
              if (rep_date_kind = 0) then
                can_suspend = 1;
              else
              if ((rep_date_kind = 1) and
                  (ma_date_app between :rep_date_begin and :rep_date_end)) then
                can_suspend = 1;
              else
              if ((rep_date_kind = 2) and
                  (ma_date_oper between :rep_date_begin and :rep_date_end)) then
                can_suspend = 1;
              else
              if ((rep_date_kind = 3) and
                  (ma_date_save between :rep_date_begin and (:rep_date_end + 1))) then
                can_suspend = 1;
            end
            if (can_suspend = 1) then
            begin
              suspend;
              if (kol >= max_kol) then
                exit;
            end
            ma_id_prior = :ma_id;
          end
        end
      end
    end /* конец поиска по названию папки */

  end
  else
  begin /* условий поиска нет, ищем только по дате */
    if (rep_date_kind = 1) then
    begin
      for select ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay, ma_percent,
                 ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save, ma_from_un_id,
                 ma_office_id, ma_currency_id, ma_no, ma_percent_category, ma_manager_id, ma_plan_disb_time,
                 ma_expert_id, ma_kommis_perc
          from miniapp, region_list(:rep_region_id, 1, -1, 1)
          where ma_office_id = id_list and
                ma_date_app between :rep_date_begin and :rep_date_end and
                ma_date_end = '31.12.9999'
          into :ma_id, :ma_number, :ma_date_app, :ma_date_oper, :ma_from_kf_id, :ma_from_pr_id, :ma_kolich_repay,
               :ma_percent, :ma_kolich_client, :ma_memo, :ma_is_active, :ma_is_auto, :ma_summa_minus, :ma_date_save,
               :ma_from_un_id, :ma_office_id, :ma_currency_id, :ma_no, :ma_percent_category, :ma_manager_id,
               :ma_plan_disb_time, :ma_expert_id, :ma_kommis_perc
      do
      begin
        ma_is_auto = 0;
        can_suspend = 0;
        if (ma_id <> ma_id_prior) then
        begin
          if (rep_date_kind = 0) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 1) and
              (ma_date_app between :rep_date_begin and :rep_date_end)) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 2) and
              (ma_date_oper between :rep_date_begin and :rep_date_end)) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 3) and
              (ma_date_save between :rep_date_begin and (:rep_date_end + 1))) then
            can_suspend = 1;

          if (can_suspend = 1) then
          begin
            suspend;
            if (kol >= max_kol) then
              exit;
          end
          ma_id_prior = :ma_id;
        end

      end
    end /* конец поиска по номеру заявки */
    if (rep_date_kind = 2) then
    begin
      for select ma_id, ma_number, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id, ma_kolich_repay, ma_percent,
                 ma_kolich_client, ma_memo, ma_is_active, ma_is_auto, ma_summa_minus, ma_date_save, ma_from_un_id,
                 ma_office_id, ma_currency_id, ma_no, ma_percent_category, ma_manager_id, ma_plan_disb_time,
                 ma_expert_id, ma_kommis_perc
          from miniapp, region_list(:rep_region_id, 1, -1, 1)
          where ma_office_id = id_list and
                ma_date_oper between :rep_date_begin and :rep_date_end and
                ma_date_end = '31.12.9999'
          into :ma_id, :ma_number, :ma_date_app, :ma_date_oper, :ma_from_kf_id, :ma_from_pr_id, :ma_kolich_repay,
               :ma_percent, :ma_kolich_client, :ma_memo, :ma_is_active, :ma_is_auto, :ma_summa_minus, :ma_date_save,
               :ma_from_un_id, :ma_office_id, :ma_currency_id, :ma_no, :ma_percent_category, :ma_manager_id,
               :ma_plan_disb_time, :ma_expert_id, :ma_kommis_perc
      do
      begin
        ma_is_auto = 0;
        can_suspend = 0;
        if (ma_id <> ma_id_prior) then
        begin
          if (rep_date_kind = 0) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 1) and
              (ma_date_app between :rep_date_begin and :rep_date_end)) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 2) and
              (ma_date_oper between :rep_date_begin and :rep_date_end)) then
            can_suspend = 1;
          else
          if ((rep_date_kind = 3) and
              (ma_date_save between :rep_date_begin and (:rep_date_end + 1))) then
            can_suspend = 1;

          if (can_suspend = 1) then
          begin
            suspend;
            if (kol >= max_kol) then
              exit;
          end
          ma_id_prior = :ma_id;
        end
      end
    end /* конец поиска по номеру заявки */
  end
end
