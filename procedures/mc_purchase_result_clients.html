CREATE PROCEDURE MC_PURCHASE_RESULT_CLIENTS(
    MCL_ID ID_TYPE,
    GB_ID ID_TYPE)
RETURNS (
    TYPE_TRANSACTION VARCHAR(255),
    OPERATION_NUMBER INTEGER,
    CLIENT_NAME VARCHAR(255),
    MAGAZINE_NAME VARCHAR(255),
    AMOUNT DECIMAL(18,4),
    AMOUNT_TOTAL DECIMAL(18,4),
    PREPAYMENT DECIMAL(18,4),
    STATUS_OPERATION VARCHAR(255),
    DATETIME_OPERATION D_DATETIME,
    LIMIT_BALANCE INTEGER,
    ERROR_TXT VARCHAR(255))
AS
declare variable DB_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable GB_STATUS D_INTEGER;
begin
  db_id = f_get_db_id();
  type_transaction = 'Покупка в рассрочку';

  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  if (c_id is null or db_id not in (39801, 41701)) then
  begin
    suspend;
    exit;
  end

  select cl.c_name_fio
  from clients cl
  where cl.c_id = :c_id
  into :client_name;

  select cs.cs_name, glb.gb_id, glb.gb_amount, glb.gb_amount_total, glb.gb_prepayment, glb.gb_status, glb.gb_datetime
  from gl_basket glb
  inner join clientshops cs on cs.cs_id = glb.gb_from_cs_id
  where glb.gb_from_mcl_id = :mcl_id and
        glb.gb_id = :gb_id
  into :magazine_name, :operation_number, :amount, :amount_total, :prepayment, :gb_status, :datetime_operation;

  status_operation = (case
                        when gb_status = 0 then 'Отменен'
                        when gb_status = 1 then 'В ожидании'
                        when gb_status = 2 then 'Успешно'
                        when gb_status = -1 then 'Ошибка'
                      end);

  execute procedure hp_calc_credit_limit(:c_id, 'installment_goods')
      returning_values :error_txt, :limit_balance;

  suspend;
end
