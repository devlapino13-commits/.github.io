CREATE PROCEDURE HP_UXFRAME_UPDATE(
    OLD_UX_CODE D_STR_CODE,
    UX_CODE D_STR_CODE,
    UX_TABLE_NAME D_STR_CODE,
    UX_NAME D_STR_100,
    UFR_SHOW_KIND D_SMALLINT,
    UFR_CAN_SELECT D_BOOL,
    UFR_CAN_UPDATE D_BOOL,
    UFR_CAN_INSERT D_BOOL,
    UFR_CAN_DELETE D_BOOL,
    UFR_NEW_SELECT D_STR_CODE,
    UFR_NEW_UPDATE D_STR_CODE,
    UFR_NEW_INSERT D_STR_CODE,
    UFR_NEW_DELETE D_STR_CODE,
    UFR_TREE_PARENT_FIELD D_STR_CODE)
RETURNS (
    RES_UX_CODE D_STR_CODE)
AS
begin
  select res_ux_code
  from hp_uxcode_update(:old_ux_code, :ux_code, :ux_table_name, :ux_name, 'uxframe')
  into :ux_code;

  update or insert into uxframe (ufr_code, ufr_show_kind, ufr_tree_parent_field,
                                 ufr_can_select, ufr_can_update, ufr_can_insert, ufr_can_delete)
  values (:ux_code, :ufr_show_kind, :ufr_tree_parent_field,
          :ufr_can_select, :ufr_can_update, :ufr_can_insert, :ufr_can_delete)
  matching (ufr_code);

  ufr_new_select = lower(trim(coalesce(ufr_new_select, '')));
  ufr_new_update = lower(trim(coalesce(ufr_new_update, '')));
  ufr_new_insert = lower(trim(coalesce(ufr_new_insert, '')));
  ufr_new_delete = lower(trim(coalesce(ufr_new_delete, '')));

  if (ufr_can_select = 1) then
  begin
    update uxfrdata
    set ufd_new_source = :ufr_new_select
    where ufd_ufr_code = :ux_code and
          ufd_kind = 0 and
          ufd_new_source <> :ufr_new_select;
  end

  if (ufr_can_update = 1) then
  begin
    update uxfrdata
    set ufd_new_source = :ufr_new_update
    where ufd_ufr_code = :ux_code and
          ufd_kind = 1 and
          ufd_new_source <> :ufr_new_update;
  end

  if (ufr_can_insert = 1) then
  begin
    update uxfrdata
    set ufd_new_source = :ufr_new_insert
    where ufd_ufr_code = :ux_code and
          ufd_kind = 2 and
          ufd_new_source <> :ufr_new_insert;
  end

  if (ufr_can_delete = 1) then
  begin
    update uxfrdata
    set ufd_new_source = :ufr_new_delete
    where ufd_ufr_code = :ux_code and
          ufd_kind = 3 and
          ufd_new_source <> :ufr_new_delete;
  end

  res_ux_code = ux_code;
  suspend;

end
