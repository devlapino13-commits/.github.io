CREATE PROCEDURE PRINT_AMNISTIA_CL_2(
    AAREP_K_ID ID_TYPE)
RETURNS (
    K_KONTRACT_NO VARCHAR(50),
    C_AMNISTIA_FIO VARCHAR(100),
    KC_AMOUNT_APPROVED CURR_TYPE,
    STATUS_CL VARCHAR(255),
    ADDRESS_YUR VARCHAR(255),
    ZAD_OS CURR_TYPE,
    ZAD_CALC_PERC CURR_TYPE,
    ZAD_OS_PERC CURR_TYPE,
    C_CITIZEN INTEGER)
AS
declare variable kd_k_new id_type;
declare variable kd_c_id id_type;
declare variable ks_state integer;
declare variable k_id_min d_id;
declare variable k_id_max d_id;
begin
  k_id_min = aarep_k_id / 10 * 10; -- 101010017976100301 -> 101010017976100300
  k_id_max = aarep_k_id / 10 * 10 + 9; -- 101010017976100301 -> 101010017976100309
  for select kc_from_k_id, ks_state, kc_from_c_id, kc_amount_approved
      from kontstate ks, knclient kc
      where ks_from_k_id between :k_id_min and :k_id_max and
            ks_ol_is_delete = 0 and
            ks_state = 2 and
            current_date between ks_date_begin and ks_date_end and
            kc_from_k_id = ks_from_k_id and
            kc_amount_approved > 0
      into kd_k_new, ks_state, kd_c_id, kc_amount_approved
  do
  begin

    select c_name_fio, c_citizen
    from clients
    where c_id = :kd_c_id
    into c_amnistia_fio, c_citizen;

    status_cl = 'Заемщик ' || c_amnistia_fio || ' произвел оплату своей части долга в полном объеме в предусмотренные сроки. Истец претензий к нему не имеет.';

    select k_kontract_no
    from kontract
    where k_id = :kd_k_new
    into k_kontract_no;

    address_yur = null;
    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :kd_c_id and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into :address_yur;
    if (address_yur is null or address_yur = 'Не указан') then
      address_yur = 'Адрес не указан';

--    select sum(CUR_CL_OS), sum(CUR_CL_PERC), sum(CUR_CL_OS + CUR_CL_PERC)
--    from print_client_zadol(:kd_k_new)
--    where CZ_NAME_FIO = :c_amnistia_fio
--    into zad_os, zad_calc_perc, zad_os_perc;

    suspend;
  end
end
