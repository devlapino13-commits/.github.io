CREATE PROCEDURE HP_KONTREFUND_BLOCK_DEL(
    KRB_ID D_INTEGER)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
begin
  -- Аннулирование запрета выдачи займа по рефинансированию (RFCRU-4446)

  if (not exists(select *
                 from kontrefund_block
                 where krb_id = :krb_id and
                       krb_is_block = 1 and
                       krb_is_delete = 0)) then
  begin
    res_text = 'У пользователя нету запрета чтобы аннулировать его';
    suspend;
    exit;
  end

  update kontrefund_block
  set krb_is_delete = 1
  where krb_id = :krb_id;

  res_is_error = 0;
  res_text = 'Запрет успешно аннулирован';
  suspend;
end
