CREATE PROCEDURE FM_JSON_CLEAR_PARSE(
    JSON_OBJECT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    NEW_JSON BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable quote_symbol varchar(1);
declare variable kol integer;
declare variable start_sym integer;
declare variable n integer;
declare variable i integer;
declare variable len integer;
declare variable last_sym integer;
declare variable sym varchar(1);
declare variable isopen integer;
begin
-- Процедура возвращает значение параметра json-объекта
-- по переданному json-объекту и имени параметра

  quote_symbol = '•';

  json_object = replace(json_object, '\"', :quote_symbol);

  len = char_length(json_object);
  i = 0;
  new_json = '';
  kol = 0;
  start_sym = 0;
  n = 0;
  last_sym = 0;
  while (i < len) do
  begin
    i = i + 1;
    sym = substring(json_object from i for 1);

    if ((kol <> 2) and
        (kol = 1 or sym <> ' ')) then
    begin
      n = n + 1;
      new_json = BLOB_APPEND(new_json, sym);
    end

    if (kol = 2) then
    begin
      if (sym in (':', ',', '}', ']')) then
      begin
        kol = 0;
        n = n + 1;
        new_json = BLOB_APPEND(new_json, sym);
      end
      if (sym = ':') then
        last_sym = n;
    end
    if (sym = '"') then
      kol = kol + 1;
    if (sym = ':' and
        kol <> 1) then
      start_sym = i;
    if (start_sym > 0 and
        sym = '"') then
      start_sym = 0;
    if (start_sym > 0 and
        (sym = ',' or sym = '}' or sym = ']' or i = len)) then
    begin
      if (i < len or sym = ',' or sym = '}' or sym = ']') then
      begin
        new_json = left(new_json, n - 1) || '"' || substring(json_object from i for 1);
        n = n + 1;
        new_json = left(new_json, last_sym) || '"' || substring(new_json from last_sym + 1);
        n = n + 1;
      end
      else
      begin
        new_json = left(new_json, start_sym - 1) || '"' || substring(json_object from start_sym + 1) || '"';
      end
      start_sym = 0;
      last_sym = 0;
    end
  end

  new_json = trim(new_json);
  new_json = replace(new_json, ascii_char(10), '');
  new_json = replace(new_json, ascii_char(13), '');
  new_json = trim(new_json);

  json_object = new_json;
  len = char_length(json_object);
  i = 0;
  new_json = '';
  kol = 0;
  isopen = 0;
  while (i < len) do
  begin
    i = i + 1;
    sym = substring(json_object from i for 1);
    if (sym = '"') then
      isopen = 1 - isopen;
    if (isopen <> 1) then
    begin
      if (sym = '{') then
        kol = kol + 1;
      if (sym = '}') then
      begin
        kol = kol - 1;
        if (kol = 0) then
        begin
          new_json = BLOB_APPEND(new_json, ',"EndOfObject":""');
        end
      end
    end
    new_json = BLOB_APPEND(new_json, sym);
  end

  suspend;
end
