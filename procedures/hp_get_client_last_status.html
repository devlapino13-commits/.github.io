CREATE PROCEDURE HP_GET_CLIENT_LAST_STATUS(
    REP_C_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    C_LAST_STATUS INTEGER,
    C_LAST_STATUS_STR STR_100)
AS
begin
  c_last_status = 0;
  select cs_star_total
  from clientstatus
  where cs_from_c_id = :rep_c_id and
        cs_ol_is_delete = 0 and
        :rep_date_end between cs_date_begin and cs_date_end
  into c_last_status;

  c_last_status_str = case c_last_status
                        when 5 then 'Статус: *****'
                        when 4 then 'Статус: ****'
                        when 3 then 'Статус: ***'
                        when 2 then 'Статус: **'
                        when 1 then 'Статус: *'
                        else 'Без статуса'
                      end;
  suspend;
end
