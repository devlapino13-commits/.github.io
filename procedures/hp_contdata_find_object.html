CREATE PROCEDURE HP_CONTDATA_FIND_OBJECT(
    CONT_VALUE VARCHAR(100),
    CONT_KIND VARCHAR(100))
RETURNS (
    RES_OBJECT_KIND INTEGER,
    RES_OBJECT_ID DECIMAL(18,0),
    RES_IS_VERIFIED INTEGER)
AS
declare variable CLT_ID D_INTEGER;
declare variable db_id d_id;
begin
  db_id = f_get_db_id();

  if (f_get_const('new_contdata') = 0) then -- если работаем по старой структуре. После переключения удалить код ниже.
  begin
    if (cont_kind = 'cell') then
    begin
      for select cci_object_kind, cci_object_id, iif(ok_cell_main = last_cell_main, 1, 0)
          from continfo
          where ok_cell_main = :cont_value or
                last_cell_main = :cont_value
          union
          select cci_object_kind, cci_object_id, iif(ok_cell_add = last_cell_add, 1, 0)
          from continfo
          where ok_cell_add = :cont_value or
                last_cell_add = :cont_value
          into :res_object_kind, :res_object_id, res_is_verified
      do
      begin
        suspend;
      end
    end
    if (cont_kind = 'email') then
    begin
      for select cci_object_kind, cci_object_id, iif(ok_email = last_email, 1, 0)
          from continfo
          where ok_email = :cont_value or
                last_email = :cont_value
          into :res_object_kind, :res_object_id, res_is_verified
      do
      begin
        suspend;
      end
    end
  end
  if (:res_object_id is null and db_id = 81001) then
  begin
    -- получение данных по новой структуре
    res_object_kind = 0;
    clt_id = f_cont_id(cont_value, 'null');
    if (clt_id is null) then exit;

    if (cont_kind not in ('cell', 'email')) then exit;

    for select cda_from_c_id, cda_status
        from contdata
        where cda_from_clt_id = :clt_id and
              cda_is_delete = 0
        into res_object_id, res_is_verified
    do
    begin
      suspend;
    end
  end
end
