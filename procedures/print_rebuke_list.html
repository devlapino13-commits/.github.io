CREATE PROCEDURE PRINT_REBUKE_LIST(
    RB_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    RB_NUMBER D_STR_20,
    RS_PROTOCOL_NO D_INTEGER,
    RS_PROTOCOL_DATE D_DATE,
    RS_EXPLANATORY_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    RB_TYPE_STR D_STR_20,
    RB_TYPE_INT D_SMALLINT,
    REG_NAME VARCHAR(50),
    REG_PARENT_NAME VARCHAR(50),
    SIGNER_STATUS D_STR_100,
    SIGNER_STATUS_KG D_STR_100,
    SIGNER_FIO D_STR_100,
    SIGNER_QR D_STR_LONG,
    HR_SIGNER D_STR_LONG,
    WHO_ADD_FIO D_STR_100,
    JL_NAME D_STR_100,
    JL_NAME_KG D_STR_100,
    PLACE_STR D_STR_100,
    WHO_ADD_JL_NAME D_STR_100,
    WHO_ADD_JL_NAME_KG D_STR_100,
    RB_DATE D_DATE,
    RB_TYPE_NAME D_STR_100,
    RB_NOTE_DATE D_DATE,
    RB_OFFICE_NAME D_STR_255,
    RB_OFFICE_NAME_KG D_STR_255,
    RB_SIGNER_NAME D_STR_100,
    RB_SIGNER_POSITION D_STR_100,
    RB_SIGNER_POSITION_KG D_STR_255,
    EDOC_SIGN_DATETIME D_DATETIME)
AS
declare variable c_id d_id;
declare variable rb_type d_smallint;
declare variable reg_id d_integer;
declare variable un_id d_id;
declare variable edl_id d_id;
declare variable signer_id d_id;
declare variable rb_signer_id d_id;
declare variable rb_from_reg_id d_integer;
declare variable rb_explanatory_date d_date;
declare variable rb_from_un_id d_id;
declare variable db_id d_integer;
begin 
  db_id = f_get_db_id();
  hr_signer = '';

  select rb.rb_number, rs.rs_protocol_no, rs.rs_protocol_date, rs.rs_explanatory_date, rb_explanatory_date, rb.rb_from_c_id, rb.rb_type,
         rs.rs_from_un_id, rb.rb_from_un_id, rb.rb_date, rb.rb_note_date, rb.rb_signer_id, rb_from_reg_id
  from rebuke rb
  left join rebuke_scores rs on rb.rb_id = rs.rs_from_rb_id
  where rb.rb_id = :rb_id
  into :rb_number, :rs_protocol_no, :rs_protocol_date, :rs_explanatory_date, :rb_explanatory_date, :c_id, :rb_type,
    :un_id, :rb_from_un_id, :rb_date, :rb_note_date, :rb_signer_id, :rb_from_reg_id;

  if (db_id = 41701 and rb_from_reg_id in(1090100, 1060100)) then
  begin
    -- в офисах 0901 и 0601 поля ниже пишутся в rebuke, а не в rebuke_scores, т.к.
    -- в указанных офисах при дисциплинарном взыскании запись в rebuke_scores не создается (RFCKG-4362)
    rs_explanatory_date = rb_explanatory_date;
    un_id = rb_from_un_id;
  end

  if (f_enum_code('rebuke_type', rb_type) starting 'remark_') then
  begin
    rb_type_str = 'замечание';
    rb_type_int = 1;
  end else
  begin
    rb_type_str = 'выговор';
    rb_type_int = 2;
  end

  rb_type_name = f_enum_name('rebuke_type', rb_type);
  rb_type_name = lower(rb_type_name);

  select c.c_name_fio
  from clients c
  where c.c_id = :c_id
  into :c_name_fio;

  -- RFCKG-4335
  select c.c_name_fio, jl.jl_name, coalesce(jl.jl_name_origin, '')
  from username un
  inner join clients c on un.un_from_c_id = c.c_id
  inner join hr_pstatus ps on ps.ps_from_c_id = c.c_id and
        current_date between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_status = 1
  inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
  where un.un_id = :un_id
  into :who_add_fio, :who_add_jl_name, :who_add_jl_name_kg;

  -- RFCKG-4335
  select reg.reg_name, reg.reg_unicode, jl.jl_name, coalesce(jl.jl_name_origin, '')
  from hr_pstatus ps
  inner join region reg on ps.ps_from_reg_id = reg.reg_unicode
  inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
  where ps.ps_from_c_id = :c_id and
        current_date between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_status = 1
  into :reg_name, :reg_id, :jl_name, :jl_name_kg;

  select reg_name
  from region
  where reg_unicode = (select reg.parent from region reg where reg.reg_unicode = :reg_id)
  into :reg_parent_name;

  signer_status = iif(:reg_id <> 1090100, 'Директор филиала «Центр» ОсОО «МКК «М Булак»', 'Исполнительный директор ОсОО «МКК «М Булак»');
  signer_status_kg = iif(:reg_id <> 1090100, '«М Булак» МКК» ЖЧКнун «Центр» Филиалынын директору', '«М Булак» МКК» ЖЧКнун «Центр»» аткаруучу директору');

  place_str = iif(:reg_id <> 1090100, 'Филиала «Центр» ОсОО МКК «М Булак»', 'Головной офис ОсОО МКК «М Булак»');

  select c.c_name_fio
  from clients c
  where c.c_id = iif(:reg_id <> 1090100, 1090100181509, 1010100001753)
  into :signer_fio;

  signer_qr = '';
  edoc_sign_datetime = null;

  -- RFCKG-4335
  if (coalesce(add_params, '') <> '') then
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);

  if (edl_id is not null) then
  begin
    select eds_from_c_id
    from edoc_sign
    where eds_from_edl_id = :edl_id
    rows 1
    into :signer_id;

    signer_qr = f_edoc_qr_by_edl(edl_id, signer_id);
    hr_signer = f_edoc_hr_signer(edl_id);

    select max(eds_status_datetime)
    from edoc_sign
    where
        eds_from_edl_id = :edl_id and
        eds_status = 1
    into :edoc_sign_datetime;
  end

  -- Подписант: ФИО, должность на русском и кыргызском (RFCKG-4362)
  select c_name_fio, jl_name, jl_name_origin
  from kontsignreg
  inner join kontsignmen on ksm_id = ksr_from_ksm_id
  inner join clients on c_id = ksm_from_c_id
  inner join hr_pstatus on ps_from_c_id = c_id
  inner join joblist on jl_id = ps_from_jl_id
  where current_date between ps_date_begin and ps_date_end and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_is_on_staff = 1 and
        ksr_id = :rb_signer_id
  into rb_signer_name, rb_signer_position, rb_signer_position_kg;

  -- Офис подписанта: названия для договоров на русском и кыргызском (RFCKG-4362)
  select reg_kont_company, reg_name_alternat
  from region
  where reg_unicode = iif(:rb_from_reg_id = 1090100, 1090100, 1090006)
  into rb_office_name, rb_office_name_kg;

  suspend;
end
