CREATE PROCEDURE HP_GET_USER_EMAIL(
    USER_NAME D_STR_100)
RETURNS (
    C_EMAIL VARCHAR(40),
    C_EMAIL_VERIFIED D_INTEGER,
    C_ID D_ID,
    UN_ID D_ID,
    CM_DATABASE_ID D_INTEGER,
    RECOVERY_BY_EMAIL D_INTEGER)
AS
begin
  cm_database_id = f_get_db_id();

  select c_id, un_id
  from clients, username
  where un_from_c_id = c_id and
        un_name = :user_name
  into :c_id, :un_id;

  select cont_value
  from hp_contdata_get_email_valid(0, :c_id)
  into :c_email;

  c_email_verified = 1;
  if (coalesce(c_email, '') = '' or
     (select cont_value from hp_contdata_get_email_main(0, :c_id)) <> :c_email)
  then c_email_verified = 0;

  recovery_by_email = 1;
  if (cm_database_id in (39801, 41701) and
    not exists(
      select *
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.PassRecoveryEMail')
      where has_access = 1))
  then recovery_by_email = 0;

  suspend;
end
