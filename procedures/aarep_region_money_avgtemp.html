CREATE PROCEDURE AAREP_REGION_MONEY_AVGTEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    RES_SALDO DECIMAL(18,4),
    RES_REGION_ID INTEGER,
    RES_DATE DATE)
AS
declare variable atl_acc_mask varchar(20);
declare variable al_id varchar(20);
begin
  res_saldo = 0;
  res_date = :report_date_begin;
  for select atl_acc_mask
      from acctypelist
      where atl_from_at_id in (11, 12)
      into :atl_acc_mask
  do
  begin
    for select al_id, al_from_reg_id
        from acc_list, region_list(:report_region_id, 1, -1, 1)
        where al_from_bg_id = :atl_acc_mask and
              al_from_reg_id = id_list
        into :al_id, :res_region_id
    do
    begin
      res_saldo = f_al_bal(al_id, report_date_begin);
      suspend;
    end
  end

  res_saldo = 0;
  for select id_list
      from region_list(:report_region_id, 1, 3, 1)
      into :res_region_id
  do
  begin
    res_date = :report_date_begin;
    while (res_date <= :report_date_end) do
    begin
      suspend;
      res_date = :res_date + 1;
    end
  end
  for select atl_acc_mask
      from acctypelist
      where atl_from_at_id in (11, 12)
      into :atl_acc_mask
  do
  begin
    for select al_from_reg_id, obs_date, sum(obs_dt_ekv - obs_kt_ekv)
        from obsummary, acc_list, region_list(:report_region_id, 1, -1, 1)
        where al_from_bg_id = :atl_acc_mask and
              al_from_reg_id = id_list and
              obs_from_al_id = al_id and
              obs_date between(:report_date_begin + 1) and :report_date_end
        group by al_from_reg_id, obs_date
        having sum(obs_dt_ekv - obs_kt_ekv) <> 0
        into :res_region_id, :res_date, :res_saldo
    do
    begin
      if (res_saldo is null) then
        res_saldo = 0;
      suspend;
    end
  end
end
