CREATE PROCEDURE HP_EXTINFO_CHECK_FOR_COMPLETE(
    C_ID D_ID,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    RESULT D_SMALLINT)
AS
declare variable MA_ID D_INTEGER;
declare variable MAA_VERIF_IS_OK D_SMALLINT;
declare variable PASSVERIF_IS_OK D_SMALLINT;
declare variable ADDRREG_IS_OK D_SMALLINT;
declare variable ADDRFACT_IS_OK D_SMALLINT;
begin
    if (object_kind = 9) then -- если кредит
    begin
        select kont.k_from_ma_app_id
        from kontract kont
        where kont.k_id = :object_id
        into :ma_id;
    end
    else
        ma_id = :object_id;

    addrreg_is_ok  = 0;
    addrfact_is_ok = 0;
    passverif_is_ok = 0;
    maa_verif_is_ok = 0;
    result = 0;

    -- Проверка: Адрес регистрации заполнен
    if (exists (
        select * from addrclient adcl
        where adcl.adc_from_c_id = :c_id
        and adcl.adc_address_type = 1
        and adcl.adc_from_adl_id <> 81001000000000 -- не указан
    )) then
        addrreg_is_ok = 1;

    -- Проверка: фактический адрес проживания заполнен
    if (exists (
        select *
        from addrclient adcl
        where adcl.adc_from_c_id = :c_id
        and adcl.adc_address_type = 2
        and adcl.adc_from_adl_id <> 81001000000000 -- не указан
    )) then
        addrfact_is_ok = 1;

    -- Проверка: паспорт верифицирован
    if (exists (
        select *
        from clients cl
        where cl.c_verify_status = 2
        and cl.c_id = :c_id
    )) then
        passverif_is_ok = 1;

    -- Проверка: заявка подписана
     if ((select client_sign_status
          from hp_edoc_get_status('loan.application', 'miniapp', :ma_id, null)
          where c_id = :c_id) = 1) then
        maa_verif_is_ok = 1;

     -- если все ОК, тогда формирую задачу в extinfo_list
     if ((addrreg_is_ok = 1) and
         (addrfact_is_ok = 1) and
         (passverif_is_ok = 1) and
         (maa_verif_is_ok = 1)) then
        result = 1;

    suspend;
end
