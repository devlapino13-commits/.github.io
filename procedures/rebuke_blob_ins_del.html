CREATE PROCEDURE REBUKE_BLOB_INS_DEL(
    BLOB_FILE BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    F_NAME VARCHAR(255),
    RB_ID INTEGER,
    RL_TYPE SMALLINT,
    UN_ID DECIMAL(18,0))
RETURNS (
    ERR_TEXT VARCHAR(255))
AS
declare variable rbl_num integer;
begin
  /*
    CREATED BY IGUSEV 02.08.2012
    PURPOSE : Модуль - "Регистрация выговоров и замечаний"
              Запись и удаление (is_del = 1) прикрепленных файлов в таблицу (из таблицы) REBUKE_BLOB

              RL_TYPE = 4  - добавление нового файла
                        5  - удаление
  */
  /* Начальная проверка  ----------------------------------------------------- >> */
  if (rb_id is null) then
    rb_id = 0;
  if (f_name is null) then
    f_name = '';
  if (rl_type is null) then
    rl_type = 0;
  if (un_id is null) then
    un_id = 0;

  err_text = '';

  if (rb_id = 0) then
    err_text = 'ID документа не определен';

  if (rl_type not in (4, 5)) then
    err_text = 'Тип операции не определен';
  if (un_id = 0) then
    err_text = 'Пользователь формирующий запись не определен';

  if (not exists(select un.un_id
                 from username un
                 where un.un_id = :un_id)) then
    err_text = 'Сотрудник формирующий запись не определен';

  if (rl_type = 4) then
  begin
    if (blob_file is null) then
      err_text = 'Файл отсутствует';
    if (f_name = '') then
      err_text = 'Имя Файла отсутствует. Файл закрепить не удалось';
  end
  /* << ----------------------------------------------------------------------------*/

  if (err_text = '') then
  begin
    if (rl_type = 4) then
    /* INSERT */
    begin
      if ((rb_id <> 0) and
          (rb_id is not null)) then
      begin
        if (exists(select *
                   from rebuke rb
                   where rb.rb_id = :rb_id rows 1)) then
        begin
          if (not exists(select *
                         from rebuke_blob rbl
                         where rbl.rbl_name = :f_name and
                               rbl.rbl_from_rb_id = :rb_id and
                               rbl.rbl_is_delete = 0 rows 1)) then
          begin
            insert into rebuke_blob (rbl_from_rb_id, rbl_blob, rbl_name)
            values (:rb_id, :blob_file, :f_name);

            insert into rebuke_log (rl_from_rb_id, rl_from_un_id, rl_type, rl_memo)
            values (:rb_id, :un_id, :rl_type, 'Был добавлен файл :' || :f_name);
          end
          else
            err_text = 'Файл с названием :' || f_name || ' к этому документу уже закреплан.';
        end
        else
          err_text = 'Документ найти не удалось. Прикрепление файла отменено';
      end
      else
        err_text = 'Документ найти не удалось. Прикрепление файла отменено';
    end
    if (rl_type = 5) then
    /* DELETE */
    begin
      if (rb_id = 0) then
        err_text = 'Документ не определн';
      else
      begin
        update rebuke_blob rbl
        set rbl_is_delete = 1
        where rbl.rbl_is_delete = 0 and
              (rbl_from_rb_id = :rb_id) and
              (rbl.rbl_name = :f_name);

        insert into rebuke_log (rl_from_rb_id, rl_from_un_id, rl_type, rl_memo)
        values (:rb_id, :un_id, :rl_type, 'Был удален файл :' || :f_name);
      end
    end
  end
  suspend;
end
