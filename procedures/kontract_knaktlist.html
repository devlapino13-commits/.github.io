CREATE PROCEDURE KONTRACT_KNAKTLIST(
    KONT_ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    KAL_DATE TYPE OF COLUMN KNAKTLIST.KAL_DATE,
    KAL_KF_NAME TYPE OF COLUMN KNAKTLIST.KAL_NEW_KF_NAME,
    KAL_LO_FIO VARCHAR(100),
    KAL_REG_NAME VARCHAR(100),
    KAL_IS_ACTIVE VARCHAR(100),
    KAL_AKT_NO VARCHAR(20),
    KAL_MEMO VARCHAR(2048))
AS
declare variable DB_ID D_INTEGER;
declare variable KF_ID decimal(18,0);
begin
    db_id = f_get_db_id();

    select k_from_kf_id
    from kontract
    where k_id = :kont_id
    into :kf_id;


    for select
        kal_date,
        kal_new_kf_name,
        c_name_fio,
        reg_code || ' ' || reg_name,
        iif(knaktlist.is_delete = 1, 'Аннулирована', ''), -- Михайлечко Дмитрий 27.06.2024
        iif(kal_from_ka_id <> 0 and :db_id in (39801, 41701), ka_akt_no, 0), -- RFCKG-2455
        iif(kal_from_ka_id <> 0 and :db_id in (39801, 41701), ka_memo, 'Создание кредитной папки') -- RFCKG-2455
    from knaktlist, knakt, clients, region
    where kal_new_kf_officer = c_id
        and kal_new_reg_id = reg_unicode
        and kal_from_kf_id = :kf_id
        and kal_from_ka_id = ka_id
    order by kal_date, knaktlist.is_delete
    into
        :kal_date,
        :kal_kf_name,
        :kal_lo_fio,
        :kal_reg_name,
        :kal_is_active,
        :kal_akt_no,
        :kal_memo
    do
        suspend;

end
