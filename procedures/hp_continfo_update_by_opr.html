CREATE PROCEDURE HP_CONTINFO_UPDATE_BY_OPR(
    OPR_ID DECIMAL(18,0))
AS
declare variable opr_object_kind integer;
declare variable opr_object_id decimal(18,0);
declare variable opr_subject_kind integer;
declare variable opr_subject_id decimal(18,0);
declare variable opr_actual_ovs_id integer;
declare variable opr_fact_ovs_id integer;
declare variable rd_code varchar(255);
declare variable rd_main_code varchar(255);
declare variable last_val varchar(100);
declare variable ok_value varchar(100);
declare variable db_id integer;
begin
  -- Данная процедура вызывается при изменении контактных данных через ObjectProps
  db_id = f_get_db_id();

  if (f_get_const('new_contdata') = 1) then
    exit; -- после перехода на новую схему, удаляем эту процедуру

  select opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id, opr_actual_ovs_id, opr_fact_ovs_id
  from objectprops
  where opr_id = :opr_id
  into :opr_object_kind, :opr_object_id, :opr_subject_kind, :opr_subject_id, :opr_actual_ovs_id, :opr_fact_ovs_id;

  -- Если субъектом не является RefData, то выходим из процедуры
  if (opr_subject_kind <> -1) then
    exit;

  select rd_code, rd_main_code
  from refdata
  where rd_id = :opr_subject_id
  into :rd_code, :rd_main_code;

  -- если свойство не является Контактной информацией, выходим из процедуры
  if (rd_main_code <> 'ContactInfo') then
    exit;

  -- Запрет редактирования ключевых ПД после утверждения договора
  -- RFCRU-5170
  if (db_id = 81001
      and exists(select 1
                from knclient kc
                join kontract k on k.k_id = kc.kc_from_k_id
                join kontstate ks on ks.ks_from_k_id = k.k_id
                where kc.kc_from_c_id = :opr_object_id
                  and kc.kc_amount_approved > 0
                  and ks.ks_state = 0
                  and ks.ks_ol_is_delete = 0
                  and k.k_status = 1
                  and current_date between ks.ks_date_begin and ks.ks_date_end)
      and rd_code in ('ContactInfo.Cell_Main', 'ContactInfo.Email'))
  then
    exception e_message 'Договор утвержден, корректировка персональных данных запрещена.';

  last_val = '';
  ok_value = '';

  if (opr_actual_ovs_id is not null) then
  begin
    select ovs_value
    from objectvalues
    where ovs_from_opr_id = :opr_id and
          ovs_id = :opr_actual_ovs_id
    into :last_val;
    if (last_val is null) then
      last_val = '';
  end

  if (opr_fact_ovs_id is not null) then
  begin
    select ovs_value
    from objectvalues
    where ovs_from_opr_id = :opr_id and
          ovs_id = :opr_fact_ovs_id
    into :ok_value;
    if (ok_value is null) then
      ok_value = '';
  end

  if (last_val = '' and ok_value <> '') then
    last_val = ok_value;

  if (rd_code = 'ContactInfo.Phone_Home' or
      rd_code = 'ContactInfo.Phone_Work' or
      rd_code = 'ContactInfo.Skype' or
      rd_code = 'ContactInfo.Whatsapp' or
      rd_code = 'ContactInfo.Viber' or
      rd_code = 'ContactInfo.Telegram' or
      rd_code = 'ContactInfo.Imo' or
      rd_code = 'ContactInfo.OK' or
      rd_code = 'ContactInfo.Facebook' or
      rd_code = 'ContactInfo.VK') then
    ok_value = last_val;

  if (rd_code = 'ContactInfo.Cell_Main') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_cell_main, ok_cell_main)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Cell_Add') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_cell_add, ok_cell_add)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Email') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_email, ok_email)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Phone_Home') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_phone_home, ok_phone_home)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Phone_Work') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_phone_work, ok_phone_work)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Skype') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_skype, ok_skype)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Whatsapp') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_whatsapp, ok_whatsapp)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Viber') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_viber, ok_viber)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Telegram') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_telegram, ok_telegram)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Imo') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_imo, ok_imo)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.OK') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_ok, ok_ok)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.Facebook') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_facebook, ok_facebook)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

  if (rd_code = 'ContactInfo.VK') then
  begin
    update or insert into continfo (cci_object_kind, cci_object_id, last_vk, ok_vk)
    values (:opr_object_kind, :opr_object_id, :last_val, :ok_value)
    matching (cci_object_kind, cci_object_id);
    exit;
  end

end
