CREATE PROCEDURE HP_DO_KN_OPERBOOK_ALL_LOAN
RETURNS (
    ERR_TEXT D_STR_LONG)
AS
declare variable k_id d_id;
declare variable rows_cnt d_integer;
declare variable t_time timestamp;
begin
  -- процедура вызывается роботом ежедневно
  -- по всем активным займам строит проводки ЕПС

  k_id = 0;
  while (k_id is not null) do
  begin
    err_text = '';
    execute statement('select res_k_id from hp_do_kn_operbook_part_online(:k_id)')(k_id := k_id)
        with autonomous transaction
        into k_id;

    -- Задержка до 10 секунд, после 100 обработанных кредитов
    -- Включается после 3:00, т.к. до этого нагрузка низкая
    t_time = cast('now' as timestamp);
    if (extract(hour from t_time) > 2) then
      while (mod(extract(second from t_time), 10) <> 0) do
        t_time = cast('now' as timestamp);

    when any do
    begin
      err_text = 'select res_k_id from hp_do_kn_operbook_part_online(' || k_id || '); commit; --' || case gdscode
                                                                                                       when 335544345 then 'Lock conflict on no wait transaction'
                                                                                                       when 335544451 then 'Update conflicts with concurrent update'
                                                                                                       when 335544451 then 'Update conflicts with concurrent update'
                                                                                                       when 335544336 then 'Deadlock'
                                                                                                       else 'gdscode=' || gdscode
                                                                                                     end;

      suspend;
      -- Продвигаем вперед k_id, т.к. из-за исключения он остался старым
      rows_cnt = 0;
      for select ks_from_k_id
          from kontstate
          where ks_state = 1 and
                ks_date_begin <= current_date and
                ks_date_end >= current_date and
                ks_ol_is_delete = 0 and
                ks_from_k_id > :k_id
          order by ks_from_k_id
          rows 100
          into k_id
      do
        rows_cnt = rows_cnt + 1;

      if (rows_cnt < 100) then
        k_id = null;
    end
  end
end
