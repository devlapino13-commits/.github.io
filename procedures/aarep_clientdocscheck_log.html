CREATE PROCEDURE AAREP_CLIENTDOCSCHECK_LOG(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    NAME_OFFICE TYPE OF COLUMN REGION.REG_NAME,
    NAME_CLIENT TYPE OF COLUMN CLIENTS.C_NAME_FIO,
    NAME_EMPLOYEE_DIGIT TYPE OF COLUMN USERNAME.UN_FIO,
    NAME_EMPLOYEE_CREATE TYPE OF COLUMN USERNAME.UN_FIO,
    DATE_PROCESSING D_DATE,
    DATE_START D_DATETIME NOT NULL,
    DATE_END D_DATETIME,
    TIME_IN_SECONDS D_INTEGER NOT NULL,
    TIME_IN_MINUTES D_INTEGER NOT NULL,
    EDIT_IS_PASSPORT D_BOOL NOT NULL,
    EDIT_IS_MIGRATION_CARD D_BOOL NOT NULL,
    EDIT_IS_CHECK_IN D_BOOL NOT NULL,
    EDIT_IS_PATENT D_BOOL NOT NULL,
    EDIT_IS_PATENT_CHECKS D_BOOL NOT NULL,
    EDIT_IS_RESIDENCE_PERMIT D_BOOL NOT NULL,
    EDIT_IS_RVP D_BOOL NOT NULL,
    EDIT_IS_OTHER_DOC D_BOOL NOT NULL)
AS
declare variable client_id type of column clientqueuecheck.cq_c_id;
declare variable pdt_code type of column passdoctype.pdt_code;
begin
  for
    select
        (select first 1 c.c_name_fio from clients c where c.c_id = cqc.cq_c_id),
        (select first 1 u.un_fio from username u where (cqc.cq_un_id = u.un_id)),
        cqc.cq_c_id,
        cast(cqc.cq_date_end_check as d_date),
        cqc.cq_date_start_check,
        cqc.cq_date_end_check,
        round((coalesce(cqc.cq_date_end_check, current_timestamp) - cqc.cq_date_start_check) * 24 * 60 * 60),
        round((coalesce(cqc.cq_date_end_check, current_timestamp) - cqc.cq_date_start_check) * 24 * 60)
      from
        clientqueuecheck as cqc
      where
        (   (cqc.cq_date_start_check between :report_date_begin and :report_date_end)
         or (cqc.cq_date_end_check   between :report_date_begin and :report_date_end))
    into
      :name_client,
      :name_employee_digit,
      :client_id,
      :date_processing,
      :date_start,
      :date_end,
      :time_in_seconds,
      :time_in_minutes
  do
  begin
    name_office              = null;
    name_employee_create     = null;

    edit_is_passport         = 0;
    edit_is_migration_card   = 0;
    edit_is_check_in         = 0;
    edit_is_patent           = 0;
    edit_is_patent_checks    = 0;
    edit_is_residence_permit = 0;
    edit_is_rvp              = 0;
    edit_is_other_doc        = 0;

    for
      select
          (select r.reg_name from region as r where(u.un_curr_reg_id = r.reg_unicode)),
          --coalesce(:name_office, (select r.reg_name from region as r where(u.un_curr_reg_id = r.reg_unicode))),
          u.un_fio,
          --coalesce(:name_employee_create, u.un_fio),
          (select pdt.pdt_code from passdoctype as pdt inner join passkind as pk on (cdc.dcl_from_pk_id = pk.pk_id) and (pk.pk_from_pdt_id = pdt.pdt_id))
        from
          clientdocscheck_log                                as cdc
          inner join clientdocs                              as cd
            on     (cdc.dcl_cds_id = cd.cds_id)
               and (cdc.dcl_c_id = :client_id)
          inner join username                                as u
            on     (cd.cds_from_un_id = u.un_id)
--          inner join region_list(:report_region_id, 1, 3, 1) as rl
--            on     (u.un_curr_reg_id = rl.id_list)
--          inner join passkind                                as pk
--            on     (cdc.dcl_from_pk_id = pk.pk_id)
--          inner join passdoctype                             as pdt
--            on     (pk.pk_from_pdt_id = pdt.pdt_id)
        where
          (cdc.dcl_date_check between :report_date_begin and :report_date_end)
        into
          :name_office,
          :name_employee_create,
          :pdt_code
      do
      begin
        if (:pdt_code starting with 'passport_') then
          edit_is_passport = 1;
        else if (:pdt_code starting with 'magration_card') then
          edit_is_migration_card = 1;
        else if (:pdt_code starting with 'registration_fiz') then
          edit_is_check_in = 1;
        else if (:pdt_code starting with 'work_patent') then
          edit_is_patent = 1;
        else if (:pdt_code starting with 'kvit_patent') then
          edit_is_patent_checks = 1;
        else if (:pdt_code starting with 'registration_vnj') then
          edit_is_residence_permit = 1;
        else if (:pdt_code starting with 'registration_rvp') then
          edit_is_rvp = 1;
        else
          edit_is_other_doc = 1;
      end

    if (:name_office is not null) then
      suspend;
  end
end
