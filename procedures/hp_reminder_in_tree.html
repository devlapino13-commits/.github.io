CREATE PROCEDURE HP_REMINDER_IN_TREE(
    UN_ID D_ID)
RETURNS (
    REC_ID INTEGER,
    REC_PARENT_ID INTEGER,
    RD_ID INTEGER,
    RD_CODE VARCHAR(255),
    ACL_OBJECT_KIND INTEGER,
    RES_BOX_NAME VARCHAR(100),
    RES_COUNT INTEGER,
    RES_STATUS INTEGER,
    RES_IS_OWN INTEGER,
    ACL_ID DECIMAL(18,0),
    RES_PICTURE_ID INTEGER,
    RES_NODE_MEMO VARCHAR(255),
    RES_IS_BOLD INTEGER,
    RES_CAN_SELECT INTEGER,
    RD_STR_ID VARCHAR(100))
AS
declare variable IS_SUPERUSER D_SMALLINT;
declare variable EDOC_SCAN_COUNT D_INTEGER;
declare variable EDOC_CHECK_COUNT D_INTEGER;
declare variable CARD_DISB_COUNT D_INTEGER;
declare variable REG_ID D_INTEGER;
declare variable UNIQTAB_USERNAME D_INTEGER;
declare variable UNIQTAB_REGION D_INTEGER;
declare variable TMP_COUNT D_INTEGER;
declare variable ACCESS_EDOC_SCAN D_SMALLINT;
declare variable ACCESS_EDOC_CHECK D_SMALLINT;
declare variable ACCESS_CARD_DISB D_SMALLINT;
declare variable ACCESS_DOUBLE_CLIENTS D_SMALLINT;
declare variable access_mc_change_phone D_SMALLINT;
declare variable DOUBLE_CLIENTS_COUNT D_INTEGER;
declare variable change_phone_count d_integer;
begin
  -- mEDocScan      - Сканирование электронных документов
  -- Вызывается окно  TeDocScanForm.OpenToScan(Self);
  -- За список отвечает процедура hp_edoc_manual_sign_list(:rep_region_id, 'scan', null, :un_id)

  -- mEDocScanCheck - Проверка скана электронных документов   TeDocScanForm.OpenToCheck(Self);
  -- Вызывается окно  TeDocScanForm.OpenToCheck(Self);
  -- За список отвечает процедура hp_edoc_manual_sign_list(:rep_region_id, 'check', null, :un_id)

  -- mKontCardDisb  - Выдача кредитов на банковские карты     TKontCardDisbForm.Create(Self, 'full');
  -- Вызывается окно  TKontCardDisbForm.Create(Self, 'full');
  -- За список отвечает процедура hp_kontracts_for_card_disb(:rep_region_id, :rep_date, 'new', :un_id)

  -- mChangePhoneVerify  - Подтверждение смены номера телефона в КМП     TPhoneChangeVerifyForm.Execute(Self);
  -- Вызывается окно  TPhoneChangeVerifyForm.Execute(Self);
  -- За список отвечает процедура hp_phone_change_requests(:rep_region_id, :un_id)

  res_node_memo = '';
  res_is_bold = 0;
  res_picture_id = 6;
  res_can_select = 0;
  rec_id = 0;
  rec_parent_id = null;
  rd_id = null;
  rd_code = ''; -- будем передавать кодовое название окна
  res_box_name = ''; -- видимое название задачи
  res_count = 0;
  res_status = 0; -- Новый
  res_is_own = 0;
  rd_str_id = 'custom_form';

  acl_id = null;
  acl_object_kind = null;

  is_superuser = 0;
  if (exists(select has_access
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Sertif_All_Tasks')
             where has_access = 1)) then
    is_superuser = 1;

  access_edoc_scan = 0;
  access_edoc_check = 0;
  access_card_disb = 0;
  access_double_clients = 0;
  access_mc_change_phone = 0;

  if (is_superuser = 1 or exists(select has_access
             from hp_permission_access_check(:un_id, 'MenuList', 0, 'mEDocScan')
             where has_access = 1)) then
    access_edoc_scan = 1;

  if (is_superuser = 1 or exists(select has_access
             from hp_permission_access_check(:un_id, 'MenuList', 0, 'mEDocScanCheck')
             where has_access = 1)) then
    access_edoc_check = 1;

  if (is_superuser = 1 or exists(select has_access
             from hp_permission_access_check(:un_id, 'MenuList', 0, 'mKontCardDisb')
             where has_access = 1)) then
    access_card_disb = 1;

  if (is_superuser = 1 or exists(select has_access
             from hp_permission_access_check(:un_id, 'MenuList', 0, 'N167') -- Исключение дубликата
             where has_access = 1)) then
    access_double_clients = 1;

  if (is_superuser = 1 or exists(select has_access
             from hp_permission_access_check(:un_id, 'MenuList', 0, 'mChangePhoneVerify')
             where has_access = 1)) then
    access_mc_change_phone = 1;

  edoc_scan_count = 0;
  edoc_check_count = 0;
  card_disb_count = 0;
  double_clients_count = 0;
  change_phone_count = 0;

  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Region')
  into :uniqtab_region;

  for select distinct pm_subject_id
  from permissions
  where pm_subject_kind = :uniqtab_region and
        pm_object_kind = :uniqtab_username and
        pm_object_id = :un_id and
        pm_is_delete = 0 and
        (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
        (coalesce(pm_date_end, '31.12.2999') >= current_date)
  into :reg_id
  do
  begin
    if (access_edoc_scan = 1) then
    begin
      tmp_count = 0;
      select count(*)
      from edoc_blank_pending, edoc_blank_scan, edoc_blank, edoc_cfg,
           edoc_list, region_list(:reg_id, 1, 3, 1)
      where edbp_from_edb_id = edbs_from_edb_id and
            edbs_from_edb_id = edb_id and
            edb_from_edl_id = edl_id and
            edl_from_edc_id = edc_id and
            edc_state <> 0 and
            (edb_from_bls_id is not null or edb_blob is not null) and
            edbp_from_reg_id = id_list and
            edbs_status in (-1, 0) and
            edbs_kind in (0, 1)
      into tmp_count;
      edoc_scan_count = edoc_scan_count + tmp_count;
    end

    if (access_edoc_check = 1) then
    begin
      tmp_count = 0;
      select count(*)
      from edoc_blank_pending, edoc_blank_scan, edoc_blank, edoc_cfg,
           edoc_list, region_list(:reg_id, 1, 3, 1)
      where edbp_from_edb_id = edbs_from_edb_id and
            edbs_from_edb_id = edb_id and
            edb_from_edl_id = edl_id and
            edl_from_edc_id = edc_id and
            edc_state <> 0 and
            (edb_from_bls_id is not null or (edb_blob is not null)) and
            edbp_from_reg_id = id_list and
            ((edbs_status = 1 and edbs_from_bls_id is not null) or
            (edbs_status = 0 and edbs_kind = 2))
      into tmp_count;

      edoc_check_count = edoc_check_count + tmp_count;
    end

    if (access_mc_change_phone = 1) then
    begin
      tmp_count = 0;
      select count(*)
      from hp_phone_change_requests(1000000, :un_id)
      into tmp_count;

      change_phone_count = change_phone_count + tmp_count;
    end

    if (access_card_disb = 1) then
    begin
      tmp_count = 0;
      select count(*)
      from hp_kontracts_for_card_disb(:reg_id, current_date, 'new', :un_id)
      into tmp_count;

      card_disb_count = card_disb_count + tmp_count;
    end
  end

--  if (edoc_scan_count = 0 and edoc_check_count = 0 and card_disb_count = 0) then
--    exit;

  if (access_double_clients = 1) then
  begin
    select count(*)
    from doubleclients
    where (dc_state > 0 and
           (dc_state_un_id = :un_id or dc_state_un_id is null))
    into double_clients_count;
  end


  rec_id = rec_id + 1;
  res_picture_id = 6;
  res_status = 0; -- Новый
  rd_code = ''; -- будем передавать кодовое название окна
  res_box_name = f_format('Появились новые данные (%1)', edoc_scan_count + edoc_check_count + card_disb_count); -- видимое название задачи
  res_node_memo = 'Здесь отображаются новые задачи для вашего внимания';
  rec_parent_id = null;
  res_count = edoc_scan_count + edoc_check_count + card_disb_count;
  res_is_bold = iif(res_count > 0, 1, 0);
  rd_str_id = '';
  res_can_select = 0;
  suspend;

  rec_parent_id = rec_id;
  res_can_select = 1;
  res_picture_id = 6;
  res_status = 0; -- Новый
  rd_str_id = 'menu_item';

  rec_id = rec_id + 1;
  res_count = edoc_scan_count;
  res_is_bold = iif(res_count > 0, 1, 0);
  rd_code = 'mEDocScan'; -- название меню для открытия
  res_box_name = f_format('Сканирование электронных документов (%1)', res_count);
  res_node_memo = 'Необходимо отсканировать электронные документы для ручного подписания';
  suspend;

  rec_id = rec_id + 1;
  res_count = edoc_check_count;
  res_is_bold = iif(res_count > 0, 1, 0);
  rd_code = 'mEDocScanCheck'; -- название меню для открытия
  res_box_name = f_format('Проверка скана электронных документов (%1)', res_count);
  res_node_memo = 'Необходимо проверить ручные подписи в электронных документах';
  suspend;

  rec_id = rec_id + 1;
  res_count = card_disb_count;
  res_is_bold = iif(res_count > 0, 1, 0);
  rd_code = 'mKontCardDisb'; -- название меню для открытия
  res_box_name = f_format('Выдача займов на банковские карты (%1)', res_count);
  res_node_memo = 'Необходимо подтвердить выдачу кредитов на банковские карты';
  suspend;

  if (access_double_clients = 1) then
  begin
    rec_id = rec_id + 1;
    res_count = double_clients_count;
    res_is_bold = iif(res_count > 0, 1, 0);
    rd_code = 'double_clients'; -- название меню для открытия
    res_box_name = f_format('Исключение дубликата (%1)', res_count);
    res_node_memo = 'Необходимо проверить на предмет дублирования';
    suspend;
  end

  if (access_mc_change_phone = 1) then    -- RFCRU-4763
  begin
    rec_id = rec_id + 1;
    res_count = change_phone_count;
    res_is_bold = iif(res_count > 0, 1, 0);
    rd_code = 'mChangePhoneVerify'; -- название меню для открытия
    res_box_name = f_format('Подтверждение смены номера телефона в КМП (%1)', res_count);
    res_node_memo = 'Необходимо подтвердить смену номера телефона в КМП';
    suspend;
  end
end
