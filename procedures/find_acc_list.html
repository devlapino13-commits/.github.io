CREATE PROCEDURE FIND_ACC_LIST(
    FIND_STR VARCHAR(100))
RETURNS (
    AL_ID VARCHAR(20),
    AL_FROM_REG_ID INTEGER,
    AL_DATE DATE,
    AL_FROM_BG_ID VARCHAR(20),
    AL_CURRENCY VARCHAR(3),
    AL_NUMBER INTEGER,
    AL_TYPE INTEGER,
    AL_NAME VARCHAR(200),
    AL_MEMO VARCHAR(255),
    AL_DATE_CLOSE DATE,
    AL_IS_ACTIVE SMALLINT,
    IS_DELETE SMALLINT,
    AL_FROM_BL_ID VARCHAR(10),
    AL_ACCOUNT VARCHAR(128),
    AL_IS_IPF INTEGER,
    AL_DESCRIPTION D_STR_255)
AS
begin
  for select al_id, al_from_reg_id, al_date, al_from_bg_id, al_currency, al_number, al_type, al_name, al_memo,
             al_date_close, al_is_active, is_delete, al_from_bl_id, al_account, al_is_ipf, al_description
      from acc_list, find_in_workdoc_words('acc_list', :find_str, 1)
      where al_id = wdlink_object_id
      into :al_id, :al_from_reg_id, :al_date, :al_from_bg_id, :al_currency, :al_number, :al_type, :al_name, :al_memo,
           :al_date_close, :al_is_active, :is_delete, :al_from_bl_id, :al_account, :al_is_ipf, :al_description
  do
  begin
    suspend;
  end
end
