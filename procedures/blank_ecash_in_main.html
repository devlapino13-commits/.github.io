CREATE PROCEDURE BLANK_ECASH_IN_MAIN(
    OL DECIMAL(18,0))
RETURNS (
    BL_DATE D_DATE,
    BL_DOC_NO D_INTEGER,
    BL_MEMO D_STR_LONG,
    BL_CLIENT_NAME D_STR_255,
    BL_PASS_NO D_STR_CODE,
    BL_PASS_DATE D_DATE,
    BL_PASS_WHO D_STR_255,
    BL_CLIENT_REGIST_NO D_STR_20,
    BL_CLIENT_OKPO D_STR_20,
    BL_CLIENT_INN D_STR_20,
    BL_OFFICE_FULL_NAME D_STR_255,
    BL_OFFICE_FULL_NAME_KG1 D_STR_255,
    BL_OFFICE_FULL_NAME_KG2 D_STR_255,
    BL_OFFICE_BANK_NAME D_STR_255,
    BL_SUMMA D_CURRENCY,
    BL_ACCOUNT D_STR_20,
    BL_IS_DELETE D_SMALLINT,
    BL_KASS_PLAN D_STR_10,
    BL_CORR_ACC D_STR_255,
    BL_COMPANY D_STR_255,
    BL_OFFICE_NAME D_STR_100,
    BL_BRANCH_FULL_NAME D_STR_255,
    BL_DOC_NO_STR D_STR_20,
    REG_CODE D_STR_20,
    BL_DOC_CLIENT_NAME D_STR_255,
    BL_UN_FIO D_STR_100,
    BL_KOD D_STR_20,
    BL_MANAGER_FIO D_STR_100,
    BL_TICKET_NO D_STR_CODE,
    BL_IPF_CLS D_STR_20,
    QR_CODE D_STR_LONG,
    EDOC_SIGN_DATETIME D_DATETIME,
    DATE_LIC_CHET D_STR_CODE,
    BL_CASHIER_FIO D_STR_100,
    BL_SAVE_DATETIME D_DATETIME,
    RES_FULL_SHOW D_BOOL)
AS
declare variable OFFICE varchar(255);
declare variable COMPANY_NAME varchar(255);
declare variable BL_BANK_NAME varchar(255);
declare variable BL_BANK_ID varchar(10);
declare variable BL_CORR_ACCOUNT varchar(20);
declare variable OL_MANAGER_C_ID decimal(18,0);
declare variable OPC_FROM_C_ID D_ID;
declare variable CASH_ID D_INTEGER;
declare variable OPC_IS_IPF D_BOOL;
declare variable EDL_ID D_ID;
declare variable CLIENT_ACC_NO D_STR_CODE;
declare variable C_BIRTH_DATE D_DATE;
declare variable OPC_STATUS D_SMALLINT;
declare variable OPC_OTHER_CLIENT D_STR_255;
declare variable EDS_FROM_C_ID D_ID;
declare variable OPC_SIGN D_SMALLINT;
declare variable IN_OPC_ID D_INTEGER;
begin
  -- Временно отключен

  /*
  -- № чека ПВН операции 05.08.2021 ЕАБ
  select pvn.po_ticket_no
  from pvnopers pvn
  where pvn.po_from_ol_id = :ol -- Доработать
  rows 1
  into :bl_ticket_no;
  */

  bl_ticket_no = '';
  company_name = '';
  bl_bank_name = '';
  bl_bank_id = '';
  bl_corr_acc = '';
  bl_kass_plan = '';

  select opc_sign
  from opercash
  where opc_id = :ol
  into opc_sign;

  if (opc_sign = -1) then
  begin
    -- готовим данные приходного ордера, но входящий ol (opc_id) расходника,
    -- ищем opc_id приходника, операция может быть дочерней расходной операции
    -- например: расходник - выдача займа, приходник - прием комиссии за кассовое обслуживание (RFCKG-4162)

    select opc_id
    from opercash
    where opc_sign = 1 and -- приходный ордер
          opc_parent_id = :ol
    order by opc_is_delete, opc_status desc
    rows 1
    into in_opc_id;

    if (row_count = 0) then
      exception e_default 'in_opc_id not found';

    ol = in_opc_id; -- меняем opc_id расходника на opc_id приходника
  end

  select opc_date, opc_doc_no, opc_memo, opc_is_delete,
         reg_name, reg_kont_company, reg_code, opc_status,
         un_fio, opc_from_c_id, opc_manager_id,
         opc_from_csp_id, opc_is_ipf, opc_amount, opc_corr_al_id,
         opc_from_edl_id, c_name_fio, opc_other_client, opc_datetime
  from opercash
       inner join region on opc_office_id = reg_unicode
       inner join username on opc_from_un_id = un_id
       inner join clients on opc_cashier_id = c_id
  where opc_id = :ol
  into bl_date, bl_doc_no, bl_memo, bl_is_delete,
       office, bl_branch_full_name, reg_code, opc_status,
       bl_un_fio, opc_from_c_id, ol_manager_c_id,
       cash_id, opc_is_ipf, bl_summa, bl_corr_account,
       edl_id, bl_cashier_fio, opc_other_client, bl_save_datetime;

  if (row_count = 0) then
    exception e_default 'eBlank is empty';

  edoc_sign_datetime = null;
  qr_code = '';

  opc_other_client = trim(coalesce(opc_other_client, ''));

  if (edl_id is not null) then
  begin

    select max(eds.eds_status_datetime), eds.eds_from_c_id
    from edoc_sign eds
    where eds.eds_from_edl_id = :edl_id and
          eds.eds_status = 1
    group by eds_from_c_id
    into edoc_sign_datetime, eds_from_c_id;

    qr_code = f_edoc_qr_by_edl(edl_id, eds_from_c_id);
  end

  res_full_show = 0;
  if (opc_status = 1 or qr_code <> '') then
    res_full_show = 1;

  select iif(:opc_is_ipf = 0, p.csp_cls_al_id, p.csp_ipf_al_id)
  from cashplace p
  where p.csp_id = :cash_id
  into bl_account;

  if (opc_is_ipf = 0) then
    bl_ipf_cls = 'Классика';
  else
    bl_ipf_cls = 'ИПФ';

  select al_number
  from acc_list
  where al_id = :bl_account
  into bl_kod;

  bl_manager_fio = '';
  if (ol_manager_c_id is not null) then
  begin
    select c_name_fio
    from clients
    where c_id = :ol_manager_c_id
    into :bl_manager_fio;
  end

  bl_doc_no_str = bl_doc_no;
  bl_corr_account = replace(substring(bl_corr_account from 8 for 5), '-', '.');

  bl_client_name = '';
  select c_name_fio, c_pass_no, c_pass_date, c_pass_who,
         c_reg_no, c_okpo, c_inn, c_birth_date
  from clients
  where c_id = :opc_from_c_id
  into bl_client_name, bl_pass_no, bl_pass_date, bl_pass_who,
       bl_client_regist_no, bl_client_okpo, bl_client_inn, c_birth_date;

  if (opc_other_client <> '') then
    bl_client_name = trim(bl_client_name || ' через ' || opc_other_client);
  client_acc_no = '';
  
  if (opc_from_c_id is not null and c_birth_date is not null) then
  begin
    select uniq_acc_str
    from hp_show_clientuniq_acc_no(:opc_from_c_id)
    into :client_acc_no; -- № Лицевого счета
    
    date_lic_chet = extract(year from c_birth_date)
                    || lpad(extract(month from c_birth_date), 2, '0')
                    || lpad(extract(day from c_birth_date), 2, '0')
                    || '-' || client_acc_no;
  end

  select cm_name
  from company
  into company_name;

  bl_office_bank_name = bl_bank_id || ' ' || bl_bank_name;
  bl_office_full_name = company_name || ' ' || office;
  bl_company = company_name;
  bl_office_name = :office;
  bl_doc_client_name = ''; -- Приход из третьих запрещен

  if (bl_doc_client_name <> '') then
    bl_client_name = bl_client_name || ' через ' || bl_doc_client_name;

  bl_office_full_name_kg1 = reg_code|| ' ' || office;
  bl_office_full_name_kg2 = company_name;

  suspend;
end
