CREATE PROCEDURE AAREP_CHECK_FINPOLIS
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    FP_NAME D_STR_255,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_CODE,
    C_INN D_STR_20,
    C_ADDRESS D_STR_255,
    C_FINPOLIS_SEND_DATE D_DATE,
    FOUND_PASS_NO D_SMALLINT,
    FOUND_INN_NO D_SMALLINT,
    FOUND_CLIENT_FIO D_SMALLINT,
    FOUND_BIRTH_DATE D_SMALLINT,
    KONTRACT_NO D_STR_20,
    KONTRACT_DATE_BEGIN D_DATE,
    KONTRACT_DATE_END D_DATE,
    KONTRACT_STATUS D_STR_100,
    FP_NUMBER D_STR_20,
    C_FINPOLIS_PROCEED D_SMALLINT,
    TERR_CATEG D_STR_20,
    MA_ID D_INTEGER,
    CONFIRM D_STR_20,
    UN_FIO D_STR_100)
AS
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  for select c_id, terr_categ, fp_number, c_name_fio, fp_name, c_birth_date, c_pass_no, c_inn, c_address, c_finpolis_send_date,
             max(c_finpolis_proceed), max(found_pass_no), max(found_inn_no), max(found_client_fio),
             max(found_birth_date)
      from aarep_check_finpolis_tmp(0)
      group by c_id, terr_categ, fp_number, c_name_fio, fp_name, c_birth_date, c_pass_no, c_inn, c_address, c_finpolis_send_date
      into c_id, terr_categ, fp_number, c_name_fio, fp_name, c_birth_date, c_pass_no, c_inn, c_address,
           c_finpolis_send_date, c_finpolis_proceed, found_pass_no, found_inn_no, found_client_fio,
           found_birth_date
  do
  begin
    if (db_id = 81001 and c_id in
    (
      -- RFCRU-162 Найдены в результате доработки алгоритма, скрыты по указанию Татьяны
      1010100344494,
      1010100438651,
      1010100137392,
      1010100880110,
      1010100294292,
      -- 1010100645106, Татьяна сказала вывести, в связи с погашением, 05.12.2023
      1010100270458,
      1010100466018,
      -- RFCRU-792 Найдены в результате доработки алгоритма, скрыты по письму Татьяны от 29.01.2024
      1010100153663,
      1010100209440,
      1010100238814,
      1010100580236,
      1010100269820,
      1010100414318,
      1010100758011)) then
      continue;
    un_fio = (
      select un_fio
      from clients
        inner join username
          on un_id = c_from_un_id
      where c_id = :c_id);
    if (c_finpolis_proceed is null) then
      c_finpolis_proceed = 0;
    kontract_date_end = null;
    kontract_status = 'Активный';
    kontract_date_begin = null;
    kontract_no = '';
    select
      k_kontract_no,
      ks_date_begin,
      k_from_ma_app_id
    from kontract, kontstate, knclient
    where k_id = ks_from_k_id and
          k_id = kc_from_k_id and
          kc_from_c_id = :c_id and
          kc_amount_approved is not null and
          kc_amount_approved <> 0 and
          ks_date_end >= '31.12.2999' and
          ks_state = 1 rows 1
    into
      :kontract_no,
      :kontract_date_begin,
      :ma_id;
    if (kontract_date_begin is null) then
    begin
      kontract_status = 'В процессе';
      kontract_date_begin = null;
      select k_kontract_no, ks_date_begin
      from kontract, kontstate, knclient
      where k_id = ks_from_k_id and
            k_id = kc_from_k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved is not null and
            kc_amount_approved <> 0 and
            ks_date_end >= '31.12.2999' and
            ks_state = 0 rows 1
      into :kontract_no, :kontract_date_begin;
      if (kontract_date_begin is null) then
      begin
        kontract_status = 'Внесистемный учет';
        kontract_date_begin = null;
        select kont.k_kontract_no, kont.k_disb_date, kont.k_fact_closed_date
        from operkroff okf
        inner join operlist ol on ol.ol_id = okf.oko_from_ol_id
        inner join kontract kont on kont.k_id = okf.oko_from_k_id
        where okf.oko_from_c_id = :c_id and
              ol.is_delete = 0
        group by kont.k_kontract_no, kont.k_disb_date, kont.k_fact_closed_date
        having sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc) > 0
        order by kont.k_kontract_no, kont.k_disb_date, kont.k_fact_closed_date
        rows 1
        into :kontract_no, :kontract_date_begin, :kontract_date_end;

        if (kontract_date_begin is null) then
        begin
          kontract_status = 'Закрыт';
          kontract_date_begin = null;
          select k_kontract_no, k_disb_date, ks_date_begin
          from kontract, kontstate, knclient
          where k_id = ks_from_k_id and
                k_id = kc_from_k_id and
                kc_from_c_id = :c_id and
                kc_amount_approved is not null and
                kc_amount_approved <> 0 and
                ks_date_end >= '31.12.2999' and
                ks_state = 2
          order by ks_date_begin desc
          rows 1
          into :kontract_no, :kontract_date_begin, :kontract_date_end;

          if (kontract_date_begin is null) then
            kontract_status = 'Нет истории';
        end
      end
    end
    confirm = (
      select
        case fpv_is_confirm
          when -1 then 'отказ'
          when  0 then 'не принято'
          when  1 then 'разрешение'
        else '' end
      from fp_verify
      where fpv_ma_id = :ma_id and fpv_c_id = :c_id);


    if (kontract_status <> 'Активный') then begin
      if (f_asp_valid(:c_id, current_date) > 0)
      then kontract_status = kontract_status || '; Активный АСП';
    end

    suspend;
  end
end
