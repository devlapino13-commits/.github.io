CREATE PROCEDURE HP_SUD_GET_LIST_UX(
    PARAM D_STR_10,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_K_ID D_ID,
    UN_ID D_ID,
    CNT_ON_PAGE D_INTEGER = null,
    SHOW_ALL D_BOOL = null,
    RESPONSIBLE_ID_IN D_ID = null)
RETURNS (
    REQUEST_ID D_ID,
    REP_REGION D_STR_CODE,
    REQUEST_DATE D_DATE,
    KONTRACT_NO D_STR_20,
    FOLDER_NO D_STR_100,
    STAGE_STR D_STR_255,
    STATUS_DATE_STR D_STR_CODE,
    STATUS_USER D_STR_100,
    LAST_MEMO D_STR_2048,
    COURT_NAME D_STR_255,
    SUD_SAG_ID D_ID,
    RESPONSIBLE_ID D_ID,
    RESPONSIBLE_NAME D_STR_255,
    PROC_TYPE D_INTEGER,
    K_ID D_ID,
    PARENT_SMR D_ID,
    INICIATOR D_STR_100,
    ALL_MEMO D_BLOB_TEXT,
    LAWSUIT_NO D_STR_100,
    LAWSUIT_BEGIN_DATE D_DATE,
    LAWSUIT_DECISION_DATE D_DATE,
    LAWSUIT_EFFECTIVE_DATE D_DATE,
    RESOLUTION D_STR_8100,
    REG_ID D_INTEGER,
    STATE_DUTY_SUM D_CURRENCY,
    DECISION_BLS_ID D_ID,
    EXDOC_BLS_ID D_ID,
    EXDOC_SUD_SUM D_CURRENCY,
    EXDOC_SUM D_CURRENCY,
    REVERSAL_DECISION_BLS_ID D_ID,
    SMR_LAST_SUL_ID D_ID,
    STAGE_CODE D_STR_CODE,
    PARAM_JSON D_STR_255,
    IS_READY_REPEAT_FSSP D_BOOL,
    CLAIM_DATE D_DATE,
    IS_ZALOG D_BOOL,
    ADD_INFO D_STR_LONG)
AS
declare variable param1 d_str_10;
declare variable reg_list d_str_8100;
declare variable stage_id d_integer;
declare variable status d_integer;
declare variable sul_from_un_id d_id;
declare variable min_sld_status d_integer;
declare variable present_doc_office d_bool;
declare variable present_doc_archive d_bool;
declare variable slc_id d_id;
declare variable slc_id_tmp d_id;
declare variable stage_code_tmp d_str_code;
declare variable is_ready_repeat d_bool;
declare variable i d_integer;
declare variable pretrial_claim_status d_str_255;
declare variable tol_status d_integer;
declare variable tol_add_info_1 d_str_255;
declare variable sud_mainreq_obj_kind d_integer;
begin
  if (param is null) then
    exit;

  if (un_id is null) then
    exception e_message 'Пользователь не определен!';

  select list('[' || reg_list || ']')
  from aarep_region_user_tmp(:un_id)
  into reg_list;

  if (rep_k_id is not null) then
  begin
    select min(smr_date), max(smr_date)
    from sud_mainreq
    where smr_from_k_id = :rep_k_id
    into rep_date_begin, rep_date_end;

    if (row_count = 0) then
      exit;
  end

  if (rep_date_begin is null) then rep_date_begin = '01.01.1900';
  if (rep_date_end is null) then rep_date_end = '31.12.9999';
  if (show_all is null) then show_all = 1;
  param_json = f_json_str_set_value('{}', 'param', param);
  param1 = left(param, 1);
  param = '|' || param || '|';
  i = 0;
  sud_mainreq_obj_kind = f_objkind('sud_mainreq', 'id');

  for select smr_id, smr_date, smr_from_k_id, sul_object_kind, sul_status, smr_from_smr_id,
      smr_from_reg_id, smr_last_sul_id, c_name_fio, f_datetimetostr(sul_datetime), sul_from_un_id
  from sud_mainreq, sud_mainlog, clients c
  where sul_id = smr_last_sul_id
    and c_id = smr_from_c_id
    and smr_date between :rep_date_begin and :rep_date_end
    and (:rep_k_id is null or smr_from_k_id = :rep_k_id)
  order by smr_date desc, smr_id desc
  into request_id, request_date, k_id, stage_id, status, parent_smr,
      reg_id, smr_last_sul_id, iniciator, status_date_str, sul_from_un_id
  do
  begin
    if (show_all = 0 and i > cnt_on_page) then
      exit;

    if (reg_list not containing '[' || reg_id || ']') then
      continue;

    last_memo = '';
    folder_no = '';
    all_memo = '';
    iniciator = '';
    kontract_no = '';
    court_name = '';
    responsible_name = '';
    sud_sag_id = null;
    responsible_id = null;
    proc_type = null;
    is_ready_repeat_fssp = 0;
    resolution = '';
    add_info = '';

    stage_code =  case stage_id
                    when f_objkind('sud_mainreq',     'id') then 'R'
                    when f_objkind('sud_maindocs',    'id') then 'D'
                    when f_objkind('sud_mainpay',     'id') then 'P'
                    when f_objkind('sud_mainlist',    'id') then 'J'
                    when f_objkind('sud_maincollect', 'id') then 'JC'
                    else 'error'
                  end;

    if (not exists(select 1 from oksummary oks
    where oks_from_k_id = :k_id
      and current_date between oks.oks_date_begin and oks.oks_date_end
      and (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc + oks_out_other + oks_extra) > 0))
    then
    begin
      stage_code = 'Z';

      if (param1 <> 'Z') then
        continue;
    end

    if (stage_code = 'error') then
      exception e_default 'Unknown stage_id = ' || coalesce(stage_id, '<NULL>');

    if (param1 <> 'A' and stage_code not containing param1) then
      continue;

    param_json = f_json_str_set_value(param_json, 'request_id', request_id);
    status_user = (select un_fio from username where un_id = :sul_from_un_id);
    rep_region = (select reg_code || ' ' || reg_name from region where reg_unicode = :reg_id);

    select f_format('%1 %2: %3', f_datetimetostr(smm_datetime), un_fio, smm_memo) as log_txt
    from sud_mainmemo, username
    where smm_from_sul_id = :smr_last_sul_id
      and smm_from_un_id = un_id
    order by smm_datetime desc, smm_id desc
    rows 1
    into last_memo;
  
    with tab as (
      select f_format('%1 %2: %3', f_datetimetostr(smm_datetime), un_fio, smm_memo) as log_txt
      from sud_mainlog, sud_mainmemo, username
      where sul_from_smr_id = :request_id and
            sul_id = smm_from_sul_id and
            smm_from_un_id = un_id
      order by sul_datetime desc, sul_id desc
      )
    select list(log_txt, ascii_char(10)||ascii_char(13))
    from tab
    into all_memo;

    select kal_new_kf_name, k_kontract_no, iif(pr_code containing '.zalog', 1, 0)
    from kontract, knaktlist, product
    where kal_from_kf_id = k_from_kf_id
      and pr_id = k_from_pr_id
      and k_id = :k_id
      and :request_date between kal_date and kal_date_end
    into folder_no, kontract_no, is_zalog;

    --**************************************
    if (stage_code = 'R' and responsible_id_in is null) then
    begin
      stage_code = case status
                     when 0 then '|R1|'
                     when 1 then '|R2|'
                     when -1 then '|R1|'
                     when -2 then '|R3|'
                   end;


      if (stage_code = '') then
        stage_code = '|R0|';

      if (('|A0||R0|' || stage_code) containing param) then
      begin
        stage_str = (select list(name || trim(iif(:status = -1, '_(Доработка)', ''))) from hp_sud_get_stages where :stage_code containing ('|' || code || '|'));
        param_json = f_json_str_set_value(param_json, 'stage_code', stage_code);
        i = i + 1;
        suspend;
      end

      continue;
    end

    --**************************************
    select
      min(iif(scd_on_request > 1 or scd_on_client > 1, sld_status, 2)),
      max(iif(sld_kind = 1 and sld_status < 1, 1, 0)),
      max(iif(sld_kind = 2 and sld_status < 1, 1, 0)),
      max(smd_proc_type),
      max(smd_sud_sag_id),
      max(smd_respons_sag_id),
      max(smd_claim_date)
    from sud_maindocs, sud_loandocs, sud_cfgdockind
    where smd_id = :request_id
      and sld_from_smd_id = smd_id
      and sld_inactive = 0
      and scd_id = sld_doc_kind
    into min_sld_status, present_doc_office, present_doc_archive, proc_type, sud_sag_id, responsible_id, claim_date;

    if (coalesce(responsible_id_in, responsible_id) <> responsible_id) then
      continue;

    court_name = (select sag_name from sud_agents where sag_id = :sud_sag_id);
    responsible_name = (select sag_name from sud_agents where sag_id = :responsible_id);

    if (stage_code = 'D') then
    begin
      stage_code = '';
      pretrial_claim_status = '';

      if (status = 0) then
        stage_code = stage_code || '|D1|';
      else if (status < 0) then
        stage_code = stage_code || '|D6|';
      else if (status = 1) then
      begin
        if (min_sld_status < 2) then
          stage_code = stage_code || '|D2|';
        if (present_doc_office = 1) then
          stage_code = stage_code || '|D3|';
        if (present_doc_archive = 1) then
          stage_code = stage_code || '|D4|';
        if (min_sld_status = 2) then
          stage_code = stage_code || '|D5|';
      end

      if (stage_code = '') then
        stage_code = '|D0|';

      if (exists(select 1
                from sud_loanbls, sud_loandocs, sud_cfgdockind
                where sld_id = slb_from_sld_id
                  and scd_id = sld_doc_kind
                  and sld_from_smd_id = :request_id
                  and scd_code = 'pretrial_claim'))
      then
        pretrial_claim_status = 'Досудебная претензия отправлена. ';

      if (pretrial_claim_status = '') then
      begin
        tol_status = null;
        tol_add_info_1 = '';

        select tol_status, tol_add_info_1
        from temp_objlist
        where tol_kind = 'hp_sud_load_loans'
          and tol_object_kind = :sud_mainreq_obj_kind
          and tol_object_id = :request_id
        into tol_status, tol_add_info_1;

        if (tol_status = 0) then
          pretrial_claim_status = 'Досудебная претензия в очереди на отправку. ';
        else if (tol_status < 0) then
          pretrial_claim_status = f_format('Ошибка: %1 ', tol_add_info_1);
      end

      add_info = add_info || pretrial_claim_status;

      if (('|A0||D0|' || stage_code) containing param) then
      begin
        stage_str = (select list(name || trim(iif(:status = -1, '_(Доработка)', ''))) from hp_sud_get_stages where :stage_code containing ('|' || code || '|'));
        param_json = f_json_str_set_value(param_json, 'stage_code', stage_code);
        i = i + 1;
        suspend;
      end

      continue;
    end

    select smp_sum_sud
    from sud_mainpay
    where smp_id = :request_id
    into state_duty_sum;

    --**************************************
    if (stage_code = 'P') then
    begin
      stage_code = case status
                     when 0 then '|P1|'
                     when 1 then '|P1|'
                     when -1 then '|P2|'
                     when -2 then '|P2|'
                   end;

      if (stage_code = '') then
        stage_code = '|P0|';

      if (('|A0||P0|' || stage_code) containing param) then
      begin
        stage_str = (select list(name || trim(iif(:status = -1, '_(Доработка)', ''))) from hp_sud_get_stages where :stage_code containing ('|' || code || '|'));
        param_json = f_json_str_set_value(param_json, 'stage_code', stage_code);
        i = i + 1;
        suspend;
      end

      continue;
    end

    --**************************************
    select sml_lawsuit_no, sml_begin_date, sml_decision_date,
           sml_dec_effective_date, sml_decision_bls_id,
           sml_exdoc_bls_id, sml_exdoc_sud_sum, sml_exdoc_sum, sml_reversal_decision_bls_id
    from sud_mainlist
    where sml_id = :request_id
    into lawsuit_no, lawsuit_begin_date, lawsuit_decision_date,
         lawsuit_effective_date, decision_bls_id,
         exdoc_bls_id, exdoc_sud_sum, exdoc_sum, reversal_decision_bls_id;

   select list(coalesce(slc_resolution, ''))
   from sud_loanclient
   where slc_from_smr_id = :request_id
   into resolution;

   if (stage_code = 'J') then
    begin
      stage_code = case
                     when status = 0 then '|J1|'
                     when status = 2 then '|J5|'
                     when status = -2 then '|J6|'
                     when decision_bls_id is null then '|J2|'
                     when decision_bls_id is not null and lawsuit_no is null then '|J3|'
                     when decision_bls_id is not null and lawsuit_no is not null and exdoc_bls_id is null then '|J4|'
                   end;

      if (stage_code = '') then
        stage_code = '|J0|';

      if (('|A0||J0|' || stage_code) containing param) then
      begin
        stage_str = (select list(name || trim(iif(:status = -1, '_(Доработка)', ''))) from hp_sud_get_stages where :stage_code containing ('|' || code || '|'));
        param_json = f_json_str_set_value(param_json, 'stage_code', stage_code);
        i = i + 1;
        suspend;
      end

      continue;
    end

    --**************************************
    if (stage_code = 'JC') then
    begin
      stage_code = '';
      slc_id_tmp = 0;
      is_ready_repeat_fssp = 0;

      for select slc_id, stage_code, is_ready_repeat
      from hp_sud_get_clientslist_ux(f_json_str_set_value(f_json_str_set_value(:param_json, 'param', 'C0'), 'stage_code', '|C0|'))
      order by slc_id, smc_no desc -- для каждого клиента берем только последнюю запись
      into slc_id, stage_code_tmp, is_ready_repeat
      do
      begin
        if (slc_id_tmp = slc_id) then
          continue;

        slc_id_tmp = slc_id;
        is_ready_repeat_fssp = maxvalue(is_ready_repeat_fssp, is_ready_repeat);

        if (stage_code not containing stage_code_tmp) then
          stage_code = stage_code || stage_code_tmp;
      end 

      if (('|A0||J0||J5||C0|' || stage_code) containing param) then
      begin
        stage_str = (select list(name || trim(iif(:status = -1, '_(Доработка)', ''))) from hp_sud_get_stages where :stage_code containing ('|' || code || '|'));
        param_json = f_json_str_set_value(param_json, 'stage_code', stage_code);
        i = i + 1;
        suspend;
      end

      continue;
    end

    if (stage_code = 'Z') then
    begin
      stage_code = '|Z1|';
      stage_str = (select list(name || trim(iif(:status = -1, '_(Доработка)', ''))) from hp_sud_get_stages where :stage_code containing ('|' || code || '|'));
      i = i + 1;
      suspend;
    end
  end
end
