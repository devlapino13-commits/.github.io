CREATE PROCEDURE CB_RU_LO_ALTERNATIVE(
    REP_LO_ID ID_TYPE,
    REP_DATE_END D_DATE,
    REP_FIX_BONUS D_CURRENCY = -1,
    REP_LP_AMOUNT D_CURRENCY = -1,
    REP_BONUS_MINUS_DISB D_INTEGER = -1)
RETURNS (
    LO_STAGE_DATE D_DATE,
    DATE_BEGIN_STAGE DATE,
    DATE_END_STAGE DATE,
    KOL_VYDACH D_INTEGER,
    KOL_DEL_PLAN D_INTEGER,
    AAREP_BONUS_ALTERNATIVE D_CURRENCY,
    LO_STAGE_MONTH D_INTEGER)
AS
declare variable REP_DATE_BEGIN D_DATE;
declare variable REP_PERIOD_END D_INTEGER;
declare variable SFB_FIX_BONUS D_CURRENCY;
declare variable SFB_TOTAL_WORK D_INTEGER;
declare variable SFB_FACT_WORK D_INTEGER;
declare variable LP_AMOUNT D_CURRENCY;
declare variable JK_SYSTEM_CODE D_STR_100;
declare variable LO_STAGE_MONTH_FULL decimal(4,2);
declare variable KOL_DISB_KONT CURR_TYPE;
declare variable SFB_FACT_WORK_EFF D_CURRENCY;
declare variable TOTAL_DAYS D_SMALLINT;
declare variable DD D_DATE;
declare variable DISB_NORMA D_CURRENCY;
declare variable AAREP_BONUS_MINUS_DISB D_CURRENCY;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  rep_period_end = extract(year from :rep_date_end) * 100 + extract(month from :rep_date_end);

  date_begin_stage = dateadd(month, -7, :rep_date_begin);
  date_end_stage = dateadd(month, -1, :rep_date_begin - 1);
  date_end_stage = date_end_stage - extract(day from date_end_stage) + 15;

  lo_stage_date = null;
  select lo_stage_date, lo_stage_month from hp_get_lo_stage_alter(:rep_lo_id, :rep_date_end)
  into lo_stage_date, lo_stage_month;

  if (coalesce(lo_stage_date, current_date) between date_begin_stage and date_end_stage) then
  begin
    /* nachis -  это начисления в размере 250 рублей за каждый заем, выданный за отчетный
    месяц. Если одному и тому же клиенту за один отчетный месяц было выдано два 
    и более займа по одному и тому же продукту, то доход начисляется только за 
    один из таких займов */
    kol_vydach = 0;
    with tabl1 as (
      select distinct k_from_pr_id, kc_from_c_id from kontstate, kontract, knclient
      where ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin between :rep_date_begin and :rep_date_end
      and ks_kc_id = :rep_lo_id
      and ks_from_k_id = k_id
      and ks_from_k_id = kc_from_k_id
      and kc_amount_approved > 0
      and not exists(select * from operagr  -- привлеченные через агентов не включается
          where ks_from_ol_id = ogr_from_ol_id
          and ogr_ol_is_active = 1
          and ogr_sign = 1
          and ogr_amount > 0))
    select count(*) from tabl1
    into kol_vydach;

    /* vychet - это вычеты в размере 500 рублей за каждое плановое погашение, по которому
     допущена просрочка в отчетном месяце, если просроченная задолженность не
     погашена на конец данного отчетного месяца. */

    ------------------------------------------------------ дополнительно по вычету нужно учесть выдачи когда был учеником
    kol_del_plan = 0;
    select count(*) from oksummary, kontract
    where :rep_date_end between oks_date_begin and oks_date_end
    and :rep_date_end >= oks_delinq_date
    and oks_close_state = 0
--    and oks_delinq_date between :rep_date_begin and :rep_date_end
    and oks_kc_id = :rep_lo_id
    and oks_from_k_id = k_id
    and (k_fact_disb_lo_id = :rep_lo_id or -- по собственным выдачам
      exists(select * from krexpert        -- по выдачам наставника
      where ke_from_p_id = :rep_lo_id
      and :rep_date_end between ke_date and ke_date_end
      and ke_status = 1
      and ke_teacher_id = k_fact_disb_lo_id))
    and oks_last_sh_date >= :rep_date_begin
    into kol_del_plan;

    if (coalesce(rep_fix_bonus,-1) = -1 or coalesce(rep_bonus_minus_disb,-1) = -1) then
    begin
      -- Результат: количество рабочих дней в месяце
      total_days = 0;
      -- Пока не дошли до конца месяца повторяем цикл
      dd = rep_date_begin;
      while (dd <= rep_date_end) do
      begin
        total_days = total_days + (1 - f_date_is_weekend(dd, 1)); -- функция возвращает 0 если рабочий день, 1 если это выходной
        dd = dateadd(day, 1, dd); -- Переходим на следующий день
      end


      sfb_fix_bonus = 0;
      sfb_total_work = 0;
      sfb_fact_work = 0;

      -- фиксированный бонус будет брасться из таблицы SA_FIXBONUS
      select coalesce(sfb_fix_bonus, 0),
             coalesce(sfb_total_work, 0),
             coalesce(sfb_fact_work, 0)
      from sa_fixbonus s
      where sfb_period = :rep_period_end and
            sfb_from_c_id = :rep_lo_id
      into :sfb_fix_bonus, sfb_total_work, sfb_fact_work;

      if (--sfb_total_work > sfb_fact_work and
        sfb_fact_work > 0) then
      sfb_fact_work_eff = round(sfb_fact_work / cast(sfb_total_work as d_currency) * cast(total_days as d_currency), 0);
                  sfb_fact_work_eff = coalesce(sfb_fact_work_eff,0);

      if (sfb_total_work > 0) then
        sfb_fix_bonus = round((sfb_fix_bonus / sfb_total_work) * sfb_fact_work, 2); --Фикс.бонус к начислению
      else
      if (sfb_total_work = 0) then
        sfb_fix_bonus = 0.00;
      rep_fix_bonus = sfb_fix_bonus;
    end 

    if (coalesce(rep_lp_amount, -1) = -1) then
    begin
      --    штраф по заявкам
      lp_amount = null;
      select coalesce(sum(lp_amount), 0)
      from lo_penalty
      where lp_is_active = 1 and
            lp_date between :rep_date_begin and :rep_date_end and
            lp_from_c_id = :rep_lo_id and
            not exists(select * from miniapp, product
            where ma_id = lp_from_ma_id
            and ma_is_active = 1
            and ma_date_end = '31.12.9999'
            and ma_from_pr_id = pr_id
            and pr_code containing '.express_online.')
      into lp_amount;
      lp_amount = coalesce(lp_amount, 0);
      rep_lp_amount = lp_amount;
    end 

    if (coalesce(rep_bonus_minus_disb,-1) = -1) then
    begin
      aarep_bonus_minus_disb = 0;
      select jk_system_code
      from hr_pstatus h, joblist, jobkind
      where :rep_date_end between ps_date_begin and ps_date_end and
            h.ps_from_c_id = :rep_lo_id and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id
      into :jk_system_code;
  
      select lo_stage_month_full
      from hp_get_lo_stage_full(:rep_lo_id, :rep_date_end, :jk_system_code)
      into :lo_stage_month_full;

      if (coalesce(lo_stage_month_full, -1) < 0) then
      lo_stage_month_full = 0;

      select count(ks.ks_from_k_id)
      from kontstate ks
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and -- говорит о том что кредит активный
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_kc_id = :rep_lo_id and
            not exists(select * from kontract, product
            where k_id = ks.ks_from_k_id
            and k_from_pr_id = pr_id
            and pr_code containing '.express_online.')
      into :kol_disb_kont;
  
      if (lo_stage_month_full > 4) then
      begin
        if (sfb_fact_work_eff > 0) then
        begin
          disb_norma = round(kol_disb_kont / sfb_fact_work_eff, 2);
  
          aarep_bonus_minus_disb = case
                                     when disb_norma <= 0.5 then 3000
                                     when disb_norma between 0.49 and 0.75 then 2000
                                     when disb_norma between 0.74 and 0.99 then 1000
                                     else 0
                                   end;

        end
      end

      if (sfb_fact_work = 0) then aarep_bonus_minus_disb = 0;
      rep_bonus_minus_disb = aarep_bonus_minus_disb;
    end 
    -- БОНУС = (НАЧИСЛЕНИЯ – ВЫЧЕТЫ) + ФИКСИРОВАННЫЙ БОНУС - ШТРАФЫ
    aarep_bonus_alternative = (kol_vydach * 250 - kol_del_plan * 500) + rep_fix_bonus - (rep_lp_amount + rep_bonus_minus_disb);
    if (aarep_bonus_alternative < 0) then aarep_bonus_alternative = 0;
  end

  suspend;
end
