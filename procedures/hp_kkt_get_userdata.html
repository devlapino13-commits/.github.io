CREATE PROCEDURE HP_KKT_GET_USERDATA(
    UN_ID D_ID NOT NULL)
RETURNS (
    FIO D_STR_255,
    INN D_STR_CODE)
AS
begin
    select c_name_fio, c_inn
    from username, clients c
    where un_id = :un_id
      and c_id = un_from_c_id
    into fio, inn;

    if (trim(coalesce(fio, '')) = '') then
      exception e_default 'fio is empty for un_id = ' || coalesce(un_id, '<NULL>');

    select short_fio
    from hp_fio_clear(:fio)
    into fio;

    if (inn is null) then
      inn = '';
  suspend;
end
