CREATE PROCEDURE HP_UXFRAME_INFO(
    UFR_CODE D_STR_CODE)
RETURNS (
    UX_CODE D_STR_CODE,
    UX_TABLE_NAME D_STR_CODE,
    UX_KIND D_INTEGER,
    UX_NAME D_STR_100,
    UX_CREATE_DATETIME D_DATETIME,
    UFR_SHOW_KIND D_SMALLINT,
    UFR_CAN_SELECT D_BOOL,
    UFR_CAN_UPDATE D_BOOL,
    UFR_CAN_INSERT D_BOOL,
    UFR_CAN_DELETE D_BOOL,
    UFR_SELECT D_STR_CODE,
    UFR_UPDATE D_STR_CODE,
    UFR_INSERT D_STR_CODE,
    UFR_DELETE D_STR_CODE,
    UFR_NEW_SELECT D_STR_CODE,
    UFR_NEW_UPDATE D_STR_CODE,
    UFR_NEW_INSERT D_STR_CODE,
    UFR_NEW_DELETE D_STR_CODE,
    UFR_SELECT_STATE D_SMALLINT,
    UFR_UPDATE_STATE D_SMALLINT,
    UFR_INSERT_STATE D_SMALLINT,
    UFR_DELETE_STATE D_SMALLINT,
    UFR_TREE_PARENT_FIELD D_STR_CODE,
    UFR_HAS_SYNC_DIFF D_BOOL)
AS
begin
  for select ux_code, 
             ux_table_name, 
             ux_kind, 
             ux_name, 
             ux_create_datetime,
             ufr_show_kind,
             ufr_can_select,
             ufr_can_update,
             ufr_can_insert,
             ufr_can_delete,
             ufr_tree_parent_field
      from uxcode, uxframe
      where ux_code = :ufr_code and
            ux_code = ufr_code
      into :ux_code,
           :ux_table_name,
           :ux_kind,
           :ux_name,
           :ux_create_datetime,
           :ufr_show_kind,
           :ufr_can_select,
           :ufr_can_update,
           :ufr_can_insert,
           :ufr_can_delete,
           :ufr_tree_parent_field
  do
  begin
    ufr_has_sync_diff = 0;

    ufr_select = '';
    ufr_update = '';
    ufr_insert = '';
    ufr_delete = '';
    ufr_new_select = '';
    ufr_new_update = '';
    ufr_new_insert = '';
    ufr_new_delete = '';

    select ufd_source, ufd_new_source
    from uxfrdata
    where ufd_ufr_code = :ux_code and
          ufd_kind = 0
    into ufr_select, ufr_new_select;

    select ufd_source, ufd_new_source
    from uxfrdata
    where ufd_ufr_code = :ux_code and
          ufd_kind = 1
    into ufr_update, ufr_new_update;

    select ufd_source, ufd_new_source
    from uxfrdata
    where ufd_ufr_code = :ux_code and
          ufd_kind = 2
    into ufr_insert, ufr_new_insert;

    select ufd_source, ufd_new_source
    from uxfrdata
    where ufd_ufr_code = :ux_code and
          ufd_kind = 3
    into ufr_delete, ufr_new_delete;

    ufr_select_state = 0;
    ufr_update_state = 0;
    ufr_insert_state = 0;
    ufr_delete_state = 0;

    -- State:
    -- 0 - все ок
    -- 1 - Отсутстсует в БД
    -- 2 - Источник изменен
    -- 3 - Требуется синхронизация

    if (ufr_new_select <> '') then
    begin
      if (ufr_new_select <> ux_table_name) then
      begin
        if (not exists(select * from rdb$procedures p
        where p.rdb$package_name is null and p.rdb$procedure_name = upper(:ufr_new_select))) then
          ufr_select_state = 1;
      end
      if (ufr_select_state = 0 and ufr_new_select <> ufr_select)
      then
        if (ufr_select <> '')
        then ufr_select_state = 2;
        else ufr_select_state = 3;
    end

    if (ufr_new_update <> '') then
    begin
      if (ufr_new_update <> ux_table_name) then
      begin
        if (not exists(select * from rdb$procedures p
        where p.rdb$package_name is null and p.rdb$procedure_name = upper(:ufr_new_update))) then
          ufr_update_state = 1;
      end
      if (ufr_update_state = 0 and ufr_new_update <> ufr_update)
      then
        if (ufr_update <> '')
        then ufr_update_state = 2;
        else ufr_update_state = 3;
    end

    if (ufr_new_insert <> '') then
    begin
      if (ufr_new_insert <> ux_table_name) then
      begin
        if (not exists(select * from rdb$procedures p
        where p.rdb$procedure_name = upper(:ufr_new_insert))) then
          ufr_insert_state = 1;
      end
      if (ufr_insert_state = 0 and ufr_new_insert <> ufr_insert)
      then
        if (ufr_insert <> '')
        then ufr_insert_state = 2;
        else ufr_insert_state = 3;
    end

    if (ufr_new_delete <> '') then
    begin
      if (ufr_new_delete <> ux_table_name) then
      begin
        if (not exists(select * from rdb$procedures p
        where p.rdb$package_name is null and p.rdb$procedure_name = upper(:ufr_new_delete))) then
          ufr_delete_state = 1;
      end
      if (ufr_delete_state = 0 and ufr_new_delete <> ufr_delete)
      then
        if (ufr_delete <> '')
        then ufr_delete_state = 2;
        else ufr_delete_state = 3;
    end

    if (exists(select * from uxfrparam
    where ufp_ufr_code = :ufr_code and
          ufp_state <> 0)) then
      ufr_has_sync_diff = 1;
    else
      if (exists(select * from uxframesql
      where ufs_ufr_code = :ufr_code and
            ufs_state <> 0)) then
        ufr_has_sync_diff = 1;

    suspend;
  end
end
