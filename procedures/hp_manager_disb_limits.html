CREATE PROCEDURE HP_MANAGER_DISB_LIMITS(
    UN_ID D_ID,
    REG_ID D_ID)
RETURNS (
    KDP_DATE D_DATE,
    KDP_SUMMA_SALDO D_CURRENCY,
    KDP_SUMMA_APPROVED D_CURRENCY,
    KDP_SUMMA_PROCESS D_CURRENCY,
    KDP_COUNT_APPROVED D_INTEGER,
    KDP_COUNT_PROCESS D_INTEGER)
AS
declare variable MGR_ID D_INTEGER;
declare variable REP_DATE_END D_DATE;
begin
  if (un_id is null) then
  begin
    un_id = uxf$user();
    when any do
      exit;
  end

  mgr_id = f_un_mgr_id(un_id);
  if (mgr_id = -1) then
    exit;

  kdp_date = current_date;
  rep_date_end = kdp_date + 14;

  while (kdp_date <= rep_date_end) do
  begin
    kdp_summa_saldo = 0;
    kdp_summa_approved = 0;
    kdp_summa_process = 0;
    kdp_count_approved = 0;
    kdp_count_process = 0;

    select kdp_summa_saldo, kdp_summa_approved, kdp_summa_process, kdp_count_approved, kdp_count_process
    from kontdisbplan
    inner join region_list(:reg_id, 1, 3, 1) reg on kdp_office_id = reg.id_list
    where kdp_mgr_id = :mgr_id and
          kdp_date_begin = :kdp_date
    into kdp_summa_saldo, kdp_summa_approved, kdp_summa_process, kdp_count_approved, kdp_count_process;

    suspend;
    kdp_date = kdp_date + 1;
  end
end
