CREATE PROCEDURE HP_EXTSYS_CHECK_UNEARNED_PAY(
    PROJECT D_STR_CODE,
    ESM_ID D_ID,
    ACCOUNT D_STR_CODE,
    C_ID D_ID,
    K_ID D_ID)
RETURNS (
    ERR_NO D_INTEGER,
    ERR_TEXT STR_100)
AS
declare variable ESM_DATE D_DATE;
declare variable ERR_NO_DEF D_INTEGER;
declare variable TMP_K_ID D_ID;
begin
  -- Процедура проверяет наличие необработанных платежей, и платежей будущей датой

  if (project containing ('test')) then
    exit;

  err_no_def = case
                 when project in ('million', 'emanat', 'mobilnik') then 1
                 when project = 'europlat' then 111
                 when project = 'korona' then 1
                 when project = 'eleksnet' then 60
                 when project in ('moneta', 'tinkoff', 'wirebank') then 302
                 when project = 'kicb' then 100
               end;

  -- Проверяем, имеются у клиента необработанные pay-записи или нет
  for select esm_from_k_id
      from extsysmain
      where esm_account = :account and
            esm_object_kind = 0 and
            esm_object_id = :c_id and
            esm_action = 'pay' and
            esm_status = 0 and
            esm_id <> :esm_id
      into tmp_k_id
  do
  begin
    if (k_id is null or tmp_k_id is null or tmp_k_id = k_id) then
    begin
      err_text = 'Временная ошибка. Повторите запрос позже';
      err_no = err_no_def;
      suspend;
      exit;
    end
  end

  if (k_id is not null) then
  begin
    -- Если есть оплаты поздней датой, тоже запрещаем.
    -- Например вечером через кассу платят завтрашним днем. Если после этого
    -- принять через платежную систему, сядет текущей датой и все сломается

    select esm_date
    from extsysmain
    where esm_id = :esm_id
    into esm_date;

    if (k_id is not null and
        esm_date is not null and
        exists(select *
               from operkred
               where ok_ol_date > :esm_date and
                     ok_from_k_id = :k_id and
                     ok_ol_is_active = 1)) then
    begin
      err_text = 'Невозможно оплатить сегодня. Обратитесь в офис компании.';
      err_no = err_no_def;
      suspend;
      exit;
    end
  end
end
