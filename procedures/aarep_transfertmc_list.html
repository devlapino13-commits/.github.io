CREATE PROCEDURE AAREP_TRANSFERTMC_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    ID D_INTEGER,
    NM D_INTEGER,
    DT D_DATE,
    STATUS D_STR_CODE,
    OFFICES_NAME D_STR_CODE,
    OFFICER_NAME D_STR_CODE,
    MOLS_NAME D_STR_100,
    MOLR_NAME D_STR_100,
    DT_SIGN1 D_DATE,
    DT_SIGN2 D_DATE,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    INVENT_NO D_INTEGER,
    INVENT_STR D_STR_CODE,
    YL_CODE D_STR_20,
    QTY D_CURRENCY,
    SUMMA D_CURRENCY,
    STATE_NAME D_STR_255,
    STATE_NOTE D_STR_100)
AS
declare variable ol_id d_id;
declare variable reg_lst d_str_8100;
begin
  select
    list(id_list, ';')
  from region_list(:rep_region_id, 1, 3, 1)
  into
    :reg_lst;

  for
    select
      trt_id,
      trt_save_ol_id,
      trt_number,
      trt_date,
      case
        when operlist.is_delete = 1  then 'Аннулирован'
        when trt_status = 0 then 'Создан'
        when trt_status = 1 then 'Передан'
        when trt_status = 2 then 'Принят'
      end,
      rs.reg_name,
      rr.reg_name,
      cs.c_name_fio,
      cr.c_name_fio,
      trt_mols_sign_date,
      trt_molr_sign_date
    from transfertmc
      inner join region as rs
        on rs.reg_unicode = trt_offices_id
      inner join region as rr
        on rr.reg_unicode = trt_officer_id
      inner join clients as cs
        on cs.c_id = trt_mols_id
      inner join clients as cr
        on cr.c_id = trt_molr_id
      inner join operlist
        on ol_id = trt_save_ol_id
    where trt_date between :rep_date_begin and :rep_date_end
      and (:reg_lst containing trt_offices_id or :reg_lst containing trt_officer_id)
  into
    :id,
    :ol_id, 
    :nm,
    :dt,
    :status,
    :offices_name,
    :officer_name,
    :mols_name,
    :molr_name,
    :dt_sign1,
    :dt_sign2
  do
  begin
    for
      select
        ml_name,
        iif(ml_model = ml_name, ' -//-', ml_model),
        ob_invent_no,
        ob_invent_str,
        yl_code,
        om_kolich_minus,
        om_summa_minus,
        rd_name,
        om_state_note
      from opermat
        inner join operbill
          on ob_bill_id = om_bill_id
            and ob_bill_line_no = om_bill_line_no
            and :dt between ob_date_begin and ob_date_end
            and ob_is_delete = 0
        inner join matcateg
          on mc_id = ob_from_mc_id
        inner join matlist
          on ml_id = ob_from_ml_id
        inner join yedizlist
          on yl_id = ml_from_yl_id
        left join refdata
          on rd_id = om_state_rd_id
      where om_from_ol_id = :ol_id
        and om_main_line = 1
    into
      :ml_name,
      :ml_model,
      :invent_no,
      :invent_str,
      :yl_code,
      :qty,
      :summa,
      :state_name,
      :state_note
    do
    begin
      suspend;
    end
  end
end
