CREATE PROCEDURE HP_BKI_GET_PACKET_NAME_NBKI(
    PACKET_DATE D_DATE,
    COUNTER D_INTEGER,
    PACKET_FORMAT D_SMALLINT)
RETURNS (
    PACKET_NAME D_STR_CODE,
    NEW_COUNTER D_INTEGER)
AS
declare variable ts d_datetime;
declare variable need_exit boolean;
begin
  counter = coalesce(counter, 0);
  need_exit = false;
  while (not need_exit)
  do
  begin
    counter = counter + 1;
    -- если счетчик вышел за пределы
    if (counter > 9999) then
      exception e_default 'counter overflow';

  ts = localtimestamp + cast(counter as float)/86400;

  -- firebird может подставить значение 60 секунд!!!
  if (extract(second from ts) = 60) then
    ts = ts + 1e0/86400;

  packet_name = f_format('P701NN000001_%1_%2',
                               extract(year    from packet_date)||
                          lpad(extract(month   from packet_date),2,'0')||
                          lpad(extract(day     from packet_date),2,'0'),

                          lpad(extract(hour    from ts),2,'0')||
                          lpad(extract(minute  from ts),2,'0')||
                          lpad(cast(extract(second  from ts) as integer),2,'0'));

    if (packet_format = 2) then
      packet_name = packet_name || '_AF';

    -- проверка имени на уникальность.
    if (not exists (select * from bki_packet bkp where bkp.bkp_code = :packet_name)) then
      need_exit = true;
  end
  new_counter = counter;
  suspend;
end
