CREATE PROCEDURE AAREP_GET_CLIENT_REGION(
    REP_C_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_INN VARCHAR(30),
    C_CITIZEN INTEGER,
    C_DOC_TYPE INTEGER,
    C_SEX INTEGER,
    KF_REG_ID INTEGER)
AS
begin
  select kal_new_reg_id
  from knaktlist, knclient, kontstate, kontract
  where ks_from_k_id = kc_from_k_id and
        ks_ol_is_delete = 0 and
        ks_state = 1 and
        ks_date_end >= '31.12.2999' and
        kc_amount_approved > 0 and
        kc_from_c_id = :rep_c_id and
        k_id = ks_from_k_id and
        kal_from_kf_id = k_from_kf_id and
        knaktlist.is_delete = 0 and
        kal_date_end >= '31.12.2999' and
        kal_date <= current_date
  order by ks_date_begin desc
  rows 1
  into :kf_reg_id;

  if (kf_reg_id is null) then
  begin
    select un_curr_reg_id
    from username
    where un_id = :un_id
    into :kf_reg_id;
  end

  for select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_citizen, c_doc_type, c_sex
      from clients
      where c_id = :rep_c_id
      into c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_citizen, c_doc_type, c_sex
  do
  begin
    suspend;
  end
end
