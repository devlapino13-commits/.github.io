CREATE PROCEDURE CM_RU_0184_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable C_ID ID_TYPE;
declare variable KC_BALANCE CURR_TYPE;
declare variable PR_CODE D_STR_100;
declare variable LOC_PR_CODE D_STR_100;
declare variable MAIN_K_AMOUNT D_CURRENCY;
declare variable MAIN_K_ID D_ID;
declare variable DAY_LETS D_INTEGER;
declare variable MAIN_DISB_DATE date;
begin
  loc_pr_code = '';
  select pr_code, k_disb_date
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id and
        (pr_code containing '.shans plus.' or
        pr_code containing '.status_plus.')
  into pr_code, kf_disb_date;
  if (pr_code is null) then
    exit;
  if (pr_code containing '.shans plus.') then
    loc_pr_code = '.shans.';
  if (pr_code containing '.status_plus.') then
    loc_pr_code = '.status.';
  if (loc_pr_code <> '') then
  begin
    for select knc.kc_from_c_id
        from knclient knc
        where knc.kc_from_k_id = :k_id and
              knc.kc_amount_approved > 0
        into :c_id
    do
    begin
      main_k_id = 0;
      main_k_amount = 0;
      main_disb_date = '01.01.1099';
      select coalesce(kc_amount_approved, 0), coalesce(k_id, 0), k.k_fact_disb_date
      from knclient knc
      inner join kontstate ks on ks.ks_from_k_id = knc.kc_from_k_id
      inner join kontract k on k.k_id = knc.kc_from_k_id
      inner join product p on p.pr_id = k.k_from_pr_id
      where knc.kc_from_c_id = :c_id and
            knc.kc_amount_approved > 0 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :kf_disb_date between ks.ks_date_begin and ks.ks_date_end and
            k.k_disb_date <= :kf_disb_date and
            p.pr_code containing :loc_pr_code rows 1
      into :main_k_amount, main_k_id, :main_disb_date;
      if (main_k_id = 0) then
      begin
        res_error_text = 'Нет основного займа';
        res_error_is_critical = 1;
        suspend;
      end
      else
      begin
        if (loc_pr_code = '.shans.') then
        begin
          if (main_k_amount > 0 and
              main_k_amount > 27000) then
          begin
            res_error_text = 'Сумма основного займа превышает 27000';
            res_error_is_critical = 1;
            suspend;
          end
        end
        if (loc_pr_code = '.status.') then
        begin
          day_lets = 0;
          day_lets = datediff(day, :main_disb_date, :kf_disb_date);
          if (day_lets not between 5 and 15) then
          begin
            res_error_text = 'Со дня выдачи основного займа должно пройти от 5 до 15 календарных дней';
            res_error_is_critical = 1;
            suspend;
          end
        end
      end
    end
  end
end
