CREATE PROCEDURE FP_ANKETA_POCKET_FILL(
    REP_DATE DATE)
AS
declare variable C_ID ID_TYPE;
declare variable RIST smallint;
declare variable KS_DATE_BEGIN date;
declare variable FPA_DATE date;
begin

  for select kncl.kc_from_c_id, iif(cl.c_citizen = 12, 1, 0)
        from kontract kont
        inner join kontstate ks on ks.ks_from_k_id = kont.k_id
        inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            kncl.kc_amount_approved > 0
        union
        select t.fcl_from_c_id, 1
        from finpolis_639_cli_link t
        inner join knclient kncl on kncl.kc_from_c_id = t.fcl_from_c_id
        inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
        where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            kncl.kc_amount_approved > 0
        union
        select kncl.kc_from_c_id, iif(cl.c_citizen = 12, 1, 0)
        from
        (
                select okf.oko_from_k_id
                from operkroff okf
                where okf.oko_ol_is_active = 1
                and okf.oko_ol_date <= :REP_DATE
                group by okf.oko_from_k_id
                having sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc) > 0
        ) b
        inner join kontstate ks_close on ks_close.ks_from_k_id = oko_from_k_id
        inner join knclient kncl on kncl.kc_from_k_id = ks_close.ks_from_k_id
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where ks_close.ks_ol_is_delete = 0
        and ks_date_begin <= :REP_DATE
        and ks_close.ks_state = 2
        and kncl.kc_amount_approved > 0
    into c_id, :rist
  do
  begin
    fpa_date = null;

    select fpa_date
    from fp_anketa_cl fpcl
    where fpcl.fpa_c_id = :c_id
    order by fpcl.fpa_date desc
    rows 1
    into :fpa_date;

    fpa_date = coalesce(fpa_date, cast('01.01.1900' as date));

    if (((rist = 1) and
        (datediff(day, :fpa_date, :rep_date) >= 179)) or ((rist = 0) and
        (datediff(day, :fpa_date, :rep_date) >= 359))) then
    begin
      execute procedure fp_fill_anketa(:c_id, :rep_date, 1010100000278); -- подставляю свой ID
    end
  end
end
