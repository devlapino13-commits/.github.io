CREATE PROCEDURE AAREP_KROFF_SB(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SBA_FROM_K_ID DECIMAL(18,0),
    SBA_KF_NAME VARCHAR(100),
    SBA_KONTRACT_NO VARCHAR(20),
    SBA_DISB_DATE DATE,
    SBA_DATE_BEGIN DATE,
    SBA_REGION_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    SBA_LO_FIO VARCHAR(100),
    SBL_TYPE_STR VARCHAR(20),
    SBL_NAME VARCHAR(255),
    SBL_TEAM_NAME VARCHAR(255),
    AAREP_CLIENTS_KOL INTEGER,
    AAREP_CASHIER_FIO VARCHAR(255),
    AAREP_LEADER_FIO VARCHAR(255),
    OKO_PAYED_PERC DECIMAL(18,4),
    OKO_PAYED_OS DECIMAL(18,4),
    OKO_BALANCE_TOTAL DECIMAL(18,4),
    OKO_BALANCE_PERC DECIMAL(18,4),
    OKO_BALANCE_OTHER DECIMAL(18,4),
    OKO_BALANCE_OS DECIMAL(18,4),
    OKO_PAYED_OTHER DECIMAL(15,2),
    OKO_PAYED_OS_PER DECIMAL(15,2),
    OKO_PAYED_PERC_PER DECIMAL(15,2),
    OKO_PAYED_OTHER_PER DECIMAL(18,4),
    OKO_PAYED_TOTAL_PER DECIMAL(18,4),
    OKO_PAYED_TOTAL DECIMAL(18,4),
    SBA_AMOUNT_APPROVED CURR_TYPE,
    OKO_WRITEOFF_OS CURR_TYPE,
    OKO_WRITEOFF_PERC CURR_TYPE,
    OKO_WRITEOFF_OTHER CURR_TYPE,
    OKO_WRITEOFF_OS_PER CURR_TYPE,
    OKO_WRITEOFF_PERC_PER CURR_TYPE,
    OKO_WRITEOFF_OTHER_PER CURR_TYPE,
    AA_CURR_KC_SEX VARCHAR(10))
AS
declare variable reg_id_prior integer;
declare variable lo_id_prior decimal(18,0);
declare variable sbl_id_prior integer;
declare variable sbl_type integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable sba_lo_id decimal(18,0);
declare variable sba_region_id integer;
declare variable aarep_cashier_id decimal(18,0);
declare variable aarep_leader_id decimal(18,0);
declare variable sba_from_sbl_id integer;
declare variable oko_type integer;
declare variable oko_os decimal(18,4);
declare variable oko_perc decimal(18,4);
declare variable oko_other decimal(18,4);
declare variable oko_sh_perc decimal(18,4);
declare variable oko_sh_other decimal(18,4);
declare variable oko_sh_os decimal(18,4);
declare variable oko_sh_perc_sum decimal(18,4);
declare variable oko_sh_os_sum decimal(18,4);
declare variable oko_sh_other_sum decimal(18,4);
declare variable oko_perc_sum decimal(18,4);
declare variable oko_other_sum decimal(18,4);
declare variable oko_os_sum decimal(18,4);
declare variable ol_date date;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  reg_id_prior = 0;
  lo_id_prior = 0;
  sbl_id_prior = 0;

  for select sba_from_k_id, sba_date_begin, sba_from_sbl_id,

             kal_new_kf_officer, kal_new_reg_id, kal_new_kf_name,

             k_kontract_no, k_kolich_client_approved, k_clients_cashier_id, k_clients_leader_id, k_disb_date,
             k_amount_approved
      from sb_akt sba, kontract, knaktlist, kontstate, region_list(:rep_region_id, 1, -1, 1)
      where sba_from_k_id = k_id and
            sba_is_delete = 0 and
            sba_from_sbl_id <> 0 and
            ks_from_k_id = k_id and
            ks_date_begin <= :rep_date_end and
            ks_date_end >= :rep_date_end and
            ks_ol_is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            sba_from_sbl_id <> 0 and
            kal_new_reg_id = id_list and
            :rep_date_end between sba.sba_date_begin and sba.sba_date_end and
            k_id in (-- вытаскиваю все кредиты, по которым на внесистемном учете висит задолженнсть на начало периода
      select oko_from_k_id
      from operkroff okf
      inner join operlist ol on ol.ol_id = okf.oko_from_ol_id
      where ol.is_delete = 0 and
            ol.ol_date < :rep_date_begin
      group by okf.oko_from_k_id
      having sum(oko_sh_os) + sum(oko_sh_perc) + sum(oko_sh_other) + sum(oko_os) + sum(oko_perc) + sum(oko_other) > 0

      union all
-- вытаскиваю кредиты которые были переведны на внесистемный учет в рамках периода отчета
      select distinct okf.oko_from_k_id
      from operkroff okf
      inner join operlist ol on ol.ol_id = okf.oko_from_ol_id
      where okf.oko_type = 1 and
            ol.ol_date between :rep_date_begin and :rep_date_end and
            ol.is_delete = 0) and
            sba_status = 2 -- беру только утвержденные (-1 - аннулирован, 0 - в процессе, 1 - согласован, 2 - утвержден)
      into :sba_from_k_id, :sba_date_begin, :sba_from_sbl_id, :sba_lo_id, :sba_region_id, :sba_kf_name,
           :sba_kontract_no, :aarep_clients_kol, :aarep_cashier_id, :aarep_leader_id, :sba_disb_date,
           :sba_amount_approved
  do
  begin
    if (sba_region_id <> reg_id_prior) then
    begin
      select reg1.reg_code || ' ' || reg1.reg_name, reg2.reg_code || ' ' || reg2.reg_name
      from region reg1, region reg2
      where reg1.reg_unicode = :sba_region_id and
            reg2.reg_unicode = reg1.parent
      into :sba_region_name, :aa_curr_branch_name;

      reg_id_prior = sba_region_id;
    end
    if (sba_lo_id <> lo_id_prior) then
    begin
      select c_name_fio, iif(c_sex = 0, 'КС Женщина', 'КС Мужчина')
      from clients
      where c_id = :sba_lo_id
      into :sba_lo_fio, aa_curr_kc_sex;
      lo_id_prior = sba_lo_id;
    end

    aarep_cashier_fio = null;
    select c_name_fio
    from clients
    where c_id = :aarep_cashier_id
    into :aarep_cashier_fio;

    aarep_leader_fio = null;
    select c_name_fio
    from clients
    where c_id = :aarep_leader_id
    into :aarep_leader_fio;

    if (sbl_id_prior <> sba_from_sbl_id) then
    begin
      select sbl_name, sbl_team_name, sbl_type
      from sb_list
      where sbl_id = :sba_from_sbl_id
      into :sbl_name, :sbl_team_name, :sbl_type;
      sbl_type_str = 'Не определен';
      if (sbl_type = 0) then
        sbl_type_str = 'СБ';
      else
      if (sbl_type = 1) then
        sbl_type_str = 'Коллекторы';
      sbl_id_prior = sba_from_sbl_id;
    end

    oko_os_sum = 0;
    oko_perc_sum = 0;
    oko_other_sum = 0;
    oko_sh_os_sum = 0;
    oko_sh_perc_sum = 0;
    oko_sh_other_sum = 0;

    oko_payed_os = 0;
    oko_payed_perc = 0;
    oko_balance_total = 0;
    oko_balance_os = 0;
    oko_balance_perc = 0;
    oko_balance_other = 0;
    oko_payed_other = 0;

    oko_payed_os_per = 0;
    oko_payed_perc_per = 0;
    oko_payed_other_per = 0;

    for select ol.ol_date, okf.oko_type, -- 1 построение задолженоси после списания, 4 - погашение, 5  - смерть участника (списывается), 6- начисление
               okf.oko_os, okf.oko_perc, okf.oko_other, okf.oko_sh_os, okf.oko_sh_perc, okf.oko_sh_other
        from operkroff okf
        inner join operlist ol on ol.ol_id = okf.oko_from_ol_id
        where okf.oko_from_k_id = :sba_from_k_id and
              ol.is_delete = 0
        into :ol_date, :oko_type, :oko_os, :oko_perc, :oko_other, :oko_sh_os, :oko_sh_perc, :oko_sh_other
    do
    begin
      if (oko_type = 4) then
      begin
        oko_payed_os = oko_payed_os + oko_os;
        oko_payed_perc = oko_payed_perc + oko_perc;
        oko_payed_other = oko_payed_other + oko_other;

        if (ol_date between :rep_date_begin and :rep_date_end) then
        begin
          oko_payed_os_per = oko_payed_os_per + oko_os;
          oko_payed_perc_per = oko_payed_perc_per + oko_perc;
          oko_payed_other_per = oko_payed_other_per + oko_other;
        end
      end

      if (oko_type = 5) then
      begin
        oko_writeoff_os = oko_writeoff_os + oko_os;
        oko_writeoff_perc = oko_writeoff_perc + oko_perc;
        oko_writeoff_other = oko_writeoff_other + oko_other;

        if (ol_date between :rep_date_begin and :rep_date_end) then
        begin
          oko_writeoff_os_per = oko_writeoff_os_per + oko_os;
          oko_writeoff_perc_per = oko_writeoff_perc_per + oko_perc;
          oko_writeoff_other_per = oko_writeoff_other_per + oko_other;
        end
      end
      oko_os_sum = oko_os_sum + oko_os;
      oko_perc_sum = oko_perc_sum + oko_perc;
      oko_other_sum = oko_other_sum + oko_other;
      oko_sh_os_sum = oko_sh_os_sum + oko_sh_os;
      oko_sh_perc_sum = oko_sh_perc_sum + oko_sh_perc;
      oko_sh_other_sum = oko_sh_other_sum + oko_sh_other;
    end
    oko_payed_os = -oko_payed_os;
    oko_payed_perc = -oko_payed_perc;
    oko_payed_other = -oko_payed_other;
    oko_payed_os_per = -oko_payed_os_per;
    oko_payed_perc_per = -oko_payed_perc_per;
    oko_payed_other_per = -oko_payed_other_per;

    oko_balance_total = oko_os_sum + oko_perc_sum + oko_other_sum + oko_sh_os_sum + oko_sh_perc_sum + oko_sh_other_sum;
    oko_balance_os = oko_os_sum + oko_sh_os_sum;
    oko_balance_perc = oko_perc_sum + oko_sh_perc_sum;
    oko_balance_other = oko_other_sum + oko_sh_other_sum;

    oko_payed_total_per = oko_payed_os_per + oko_payed_perc_per + oko_payed_other;
    oko_payed_total = oko_payed_os + oko_payed_perc + oko_payed_other;

    suspend;
  end
end
