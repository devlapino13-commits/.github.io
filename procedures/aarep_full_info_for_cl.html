CREATE PROCEDURE AAREP_FULL_INFO_FOR_CL(
    REP_REGION_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KC_C_ID ID_TYPE,
    K_ID ID_TYPE,
    REG_NAME VARCHAR(50),
    OF_NAME VARCHAR(50),
    CL_FIO VARCHAR(100),
    CL_PLAN_REP_OS CURR_TYPE,
    K_AMOUNT_APPROVED CURR_TYPE,
    KNC_TYPE_STR VARCHAR(10),
    K_FROM_CS_ID INTEGER,
    PR_IS_SECURED_STR VARCHAR(20),
    PR_LOAN_TYPE_STR VARCHAR(10),
    PR_NAME VARCHAR(50),
    CKS_FIO VARCHAR(100),
    K_KF_NO INTEGER,
    K_KF_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    PLAN_REP_OS CURR_TYPE,
    PLAN_PERC CURR_TYPE,
    CL_PLAN_ALL CURR_TYPE,
    PLAN_ALL_REP CURR_TYPE,
    PLAN_PERC_ALL CURR_TYPE,
    PLAN_ALL_SUM CURR_TYPE,
    CL_PLAN_PERC_ALL CURR_TYPE,
    KC_FROM_PT_ID VARCHAR(20),
    KC_CIRCLE INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    CL_AMOUNT_APPROVED CURR_TYPE,
    KNC_FULL_ADDRESS VARCHAR(255),
    PK_NAME VARCHAR(100),
    AAREP_IS_RESTRUCTED INTEGER,
    CAGENT_FIO VARCHAR(100),
    C_PLASTCARD_NO VARCHAR(20),
    BS_NAME VARCHAR(50),
    LLC_CURRENT_PERCENT PERC_TYPE,
    PLAN_OS_ALL CURR_TYPE,
    CL_PLAN_OS_ALL CURR_TYPE)
AS
declare variable c_type integer;
declare variable pr_is_secured smallint;
declare variable pr_loan_type smallint;
declare variable kf_id id_type;
declare variable reg_id id_type;
declare variable kc_agent_id id_type;
begin
  for with kontracts
      as (select ks_from_k_id
          from kontstate
          where ks_state = 1 and
                ks_ol_is_delete = 0 and
                :rep_date_end between ks_date_begin and ks_date_end
          union
          select ks_from_k_id
          from kontstate
          where ks_state = 2 and
                ks_ol_is_delete = 0 and
                ks_date_begin between :rep_date_begin and :rep_date_end)
      select distinct ks_from_k_id, kal.kal_new_reg_id, parent_name
      from kontracts, kontract k, knaktlist kal, region_list(:rep_region_id, 1, -1, 1)
      where ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            kal.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            (kal.kal_new_reg_id = id_list) rows 10
      into k_id, reg_id, reg_name
  do
  begin
    plan_rep_os = 0;
    plan_perc = 0;
    select sum(ok_os + ok_sh_os) sh_os, sum(ok_perc + ok_sh_perc) sh_perc
    from operkred ok
    where ok_from_c_id = 0 and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end and
          ok_sh_date between :rep_date_begin and :rep_date_end and
          ok_type < 7 and
          ok_from_k_id = :k_id
    into plan_rep_os, plan_perc;

    plan_all_rep = plan_rep_os + plan_perc;

    select sum(ok_os + ok_sh_os) sh_os, sum(ok_perc + ok_sh_perc) sh_perc
    from operkred ok
    where ok_from_c_id = 0 and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end and
          ok_type < 7 and
          ok_from_k_id = :k_id
    into plan_os_all, plan_perc_all;

    plan_all_sum = plan_os_all + plan_perc_all;
    bs_name = '';
    for select kc_from_c_id, cl.c_name_fio, kc_amount_approved, cl.c_type, kc.kc_from_pt_id, kc.kc_circle, pk_name,
               kc_agent_id,
               (select cont_value
                from hp_contdata_get_cell_main(0, cl.c_id)), bs.bs_name
        from knclient kc, clients cl, passkind ps, bussubtype bs
        where kc_amount_approved > 0 and
              kc_from_k_id = :k_id and
              kc_from_c_id = cl.c_id and
              cl.c_doc_type = pk_id and
              bs_id = kc_busuness_subtype
        into kc_c_id, cl_fio, cl_amount_approved, c_type, kc_from_pt_id, kc_circle, pk_name, kc_agent_id,
             c_plastcard_no, bs_name
    do
    begin
      plan_os_all = 0;
      plan_all_rep = 0;

      select reg_name
      from region
      where reg_unicode = :reg_id
      into of_name;

      if (c_type = 0) then
        knc_type_str = 'Физ.лицо';
      else
        knc_type_str = 'Юр.лицо';

      select kont.k_kontract_no, k_from_kf_id, k_amount_approved, kont.k_from_cs_id, k_disb_date, pr.pr_is_secured,
             pr.pr_loan_type, pr.pr_name
      from kontract kont, product pr
      where k_id = :k_id and
            kont.k_from_pr_id = pr.pr_id
      into k_kontract_no, kf_id, k_amount_approved, k_from_cs_id, k_disb_date, pr_is_secured, pr_loan_type, pr_name;

      pr_is_secured_str = '';
      if (pr_is_secured = 0) then
        pr_is_secured_str = 'Без залога';
      if (pr_is_secured = 1) then
        pr_is_secured_str = 'Залог';
      if (pr_is_secured = 2) then
        pr_is_secured_str = 'Поручители';
      if (pr_is_secured = 3) then
        pr_is_secured_str = 'Залог,поруч.';

      pr_loan_type_str = '';
      if (pr_loan_type = 0) then
        pr_loan_type_str = 'Группа';
      if (pr_loan_type = 1) then
        pr_loan_type_str = 'Индив.';

      select kal.kal_new_no, kal.kal_new_kf_name, cks.c_name_fio
      from knaktlist kal, clients cks
      where kal.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal.kal_from_kf_id = :kf_id and
            kal.kal_new_kf_officer = cks.c_id
      into k_kf_no, k_kf_name, cks_fio;

      cl_plan_rep_os = cl_amount_approved / k_amount_approved * plan_rep_os;

      cl_plan_os_all = cl_amount_approved / k_amount_approved * plan_os_all;

      cl_plan_perc_all = cl_amount_approved / k_amount_approved * plan_perc_all;

      cl_plan_all = cl_plan_os_all + cl_plan_perc_all;

      knc_full_address = '';
      select adl_address_str
      from addrclient, addrlist
      where adc_address_type = 1 and
            adc_from_c_id = :kc_c_id and
            adc_from_adl_id = adl_id
      into :knc_full_address;

      aarep_is_restructed = 0;
      if (exists(select *
                 from operrest, operlist
                 where or_from_k_id = :k_id and
                       or_from_ol_id = ol_id and
                       operlist.is_delete = 0 and
                       ol_date <= :rep_date_end and
                       or_date_end >= :rep_date_end)) then
        aarep_is_restructed = 1;

      cagent_fio = '';
      select c_name_fio
      from clients cagent
      where cagent.c_id = :kc_agent_id
      into cagent_fio;

      select lk_percent
      from llc_kont llck
      where llck.lk_from_k_id = :k_id and
            :rep_date_end between lk_date and lk_date_end
      into llc_current_percent;

      suspend;
    end
  end
end
