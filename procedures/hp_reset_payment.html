CREATE PROCEDURE HP_RESET_PAYMENT(
    K_ID D_ID,
    OL_ID D_ID,
    OL_DATE D_DATE)
RETURNS (
    RES_ERR_TXT D_STR_255,
    RES_INFO_TXT D_STR_255)
AS
declare variable os_sum d_currency;
begin
  -- RFCKG-4660 команда выполнить для модуля "Отмена списания по ОС"
  res_info_txt = '';
  res_err_txt = '';

  if (k_id is null) then
    res_err_txt = 'Контракт не найден';

  select -sum(ok_os)
  from operkred ok
  where ok_from_k_id = :k_id and
        ok_from_ol_id = :ol_id and
        ok_ol_date = :ol_date
  into os_sum;

  if (os_sum is null) then
    res_err_txt = 'Операция не найдена';

  if (res_err_txt = '') then
  begin

    update operkred
    set ok_os = 0
    where (ok_from_ol_id = :ol_id);

    update operbook
    set ob_summa_val = 0,
        ob_summa = 0
    where (ob_from_ol_id = :ol_id) and
          (ob_summa = :os_sum);

    update operkroff
    set oko_os = 0,
        oko_sh_os = 0
    where (oko_from_ol_id = :ol_id);

    delete from kontstate ks
    where (ks_from_k_id = :k_id) and
          (ks_state = 2) and
          (ks.ks_from_ol_id = :ol_id);

    update kontract
    set k_fact_closed_date = null,
        k_fact_closed_ol_id = null,
        k_fact_closed_reg_id = null,
        k_fact_closed_lo_id = null,
        k_fact_closed_kf_no = null,
        k_fact_closed_bonus_change = null
    where (k_id = :k_id) and
          (k_fact_closed_ol_id = :ol_id);

    res_info_txt = 'Успешно!';
  end

  suspend;
end
