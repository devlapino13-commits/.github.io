CREATE PROCEDURE AAREP_CLIENT_STATEMENT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_BALANCE_ID VARCHAR(20),
    REPORT_ACCOUNT_NO INTEGER,
    REPORT_CLIENT_ID DECIMAL(18,0))
RETURNS (
    STA_LINE_NO INTEGER,
    STA_ACCOUNT_DT VARCHAR(20),
    STA_ACCOUNT_KT VARCHAR(20),
    STA_DATE DATE,
    STA_DOC_NO INTEGER,
    STA_MEMO VARCHAR(1024),
    STA_OB_DT DECIMAL(18,4),
    STA_OB_KT DECIMAL(18,4),
    STA_CURRENCY VARCHAR(3),
    STA_ACC_NAME VARCHAR(255),
    STA_SALDO_DT DECIMAL(18,4),
    STA_SALDO_KT DECIMAL(18,4),
    STA_BEGIN_DT DECIMAL(18,4),
    STA_BEGIN_KT DECIMAL(18,4))
AS
declare variable sta_ob_saldo decimal(18,4);
declare variable beg_saldo_kt decimal(18,4);
declare variable ob_detail varchar(1024);
declare variable doc_kind varchar(255);
declare variable ob_from_ol_id decimal(18,0);
declare variable ob_number integer;
begin
  sta_ob_dt = 0;
  sta_ob_kt = 0;
  select sum(ob_summa_val)
  from operbook, acc_list, region_list(:report_region_id, 1, -1, 1)
  where ob_ol_reg_id = id_list and
        ob_ol_is_active = 1 and
        ob_ol_date < :report_date_begin and
        ob_client_dt = :report_client_id and
        ob_account_dt = al_id and
        (:report_balance_id = '' or al_from_bg_id starting with :report_balance_id) and
        (:report_account_no = -1 or al_number = :report_account_no)
  into :sta_ob_saldo;
  if (sta_ob_saldo is null) then
    sta_ob_saldo = 0;

  select sum(ob_summa_val)
  from operbook, acc_list, region_list(:report_region_id, 1, -1, 1)
  where ob_ol_reg_id = id_list and
        ob_ol_is_active = 1 and
        ob_ol_date < :report_date_begin and
        ob_client_kt = :report_client_id and
        ob_account_kt = al_id and
        (:report_balance_id = '' or al_from_bg_id starting with :report_balance_id) and
        (:report_account_no = -1 or al_number = :report_account_no)
  into :beg_saldo_kt;
  if (beg_saldo_kt is null) then
    beg_saldo_kt = 0;

  sta_ob_saldo = :sta_ob_saldo - :beg_saldo_kt;
  sta_saldo_dt = 0;
  sta_saldo_kt = 0;

  if (sta_ob_saldo >= 0) then
    sta_saldo_dt = :sta_ob_saldo;
  else
    sta_saldo_kt = -1 * :sta_ob_saldo;

  sta_begin_dt = :sta_saldo_dt;
  sta_begin_kt = :sta_saldo_kt;
  sta_line_no = 0;
  sta_memo = 'Сальдо на начало периода';
  sta_date = :report_date_begin;
  for select ob_from_ol_id, ob_number, ob_ol_date, ob_account_dt, ob_account_kt, ob_currency, al_name,
                      ob_summa, 0.0000, ob_detail
      from operbook, acc_list, region_list(:report_region_id, 1, -1, 1)
      where ob_ol_reg_id = id_list and
            ob_ol_is_active = 1 and
            ob_ol_date between :report_date_begin and :report_date_end and
            ob_client_dt = :report_client_id and
            ob_account_dt = al_id and
            (:report_balance_id = '' or al_from_bg_id starting with :report_balance_id) and
            (:report_account_no = -1 or al_number = :report_account_no)
      union
      select ob_from_ol_id, ob_number, ob_ol_date, ob_account_dt, ob_account_kt, ob_currency, al_name,
             0.0000, ob_summa, ob_detail
      from operbook, acc_list, region_list(:report_region_id, 1, -1, 1)
      where ob_ol_reg_id = id_list and
            ob_ol_is_active = 1 and
            ob_ol_date between :report_date_begin and :report_date_end and
            ob_client_kt = :report_client_id and
            ob_account_kt = al_id and
            (:report_balance_id = '' or al_from_bg_id starting with :report_balance_id) and
            (:report_account_no = -1 or al_number = :report_account_no)
      order by 3, 1, 2
      into :ob_from_ol_id, :ob_number, :sta_date, :sta_account_dt, :sta_account_kt, :sta_currency,
           :sta_acc_name, :sta_ob_dt, :sta_ob_kt, :ob_detail
  do
  begin
    sta_doc_no = null;
    sta_memo = '';
    doc_kind = '';
    select ol_doc_no, ol_memo, dn_name_short
    from operlist left join docname on (ol_doc_kind = dn_id)
    where ol_id = :ob_from_ol_id
    into sta_doc_no, :sta_memo, :doc_kind;

    sta_line_no = :sta_line_no + 1;
    sta_ob_saldo = :sta_ob_saldo + :sta_ob_dt - :sta_ob_kt;
    sta_saldo_dt = 0;
    sta_saldo_kt = 0;

    if (sta_ob_saldo >= 0) then
      sta_saldo_dt = :sta_ob_saldo;
    else
      sta_saldo_kt = -1 * :sta_ob_saldo;

    if ((ob_detail is not null) and
        (ob_detail <> '')) then
    begin
      if (sta_memo is null) then
        sta_memo = ob_detail;
      else
      begin
        if (sta_memo <> ob_detail) then
          sta_memo = cast((sta_memo || ', ' || ob_detail) as varchar(1024));
      end
    end
    if (sta_memo is null) then
      sta_memo = '';
    if (doc_kind is not null) then
      sta_memo = :doc_kind || ': ' || :sta_memo;
    suspend;
  end
  if (sta_line_no = 0) then
    suspend;
end
