CREATE PROCEDURE HP_PERFORM_FIXBEM(
    OBJ_MOD D_INTEGER,
    OBJ_ID D_ID,
    OBJ_VAL D_STR_255)
RETURNS (
    ERR_TXT D_STR_255)
AS
begin
  if (obj_mod = 1) then
    update miniappclient set
      mac_loan_target = :obj_val
    where mac_id = :obj_id;
  else
  if (obj_mod = 2) then
    update miniappclient set
      mac_business_subtype = :obj_val
    where mac_id = :obj_id;
  else
  if (obj_mod = 3) then
    update knclient set
      kc_loan_target = :obj_val
    where kc_id = :obj_id;
  else
  if (obj_mod = 4) then
    update knclient set
      kc_busuness_subtype = :obj_val
    where kc_id = :obj_id;
  err_txt = iif(row_count = 1, '', 'Runtime error!');
  suspend;
end
