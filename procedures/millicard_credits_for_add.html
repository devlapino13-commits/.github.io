CREATE PROCEDURE MILLICARD_CREDITS_FOR_ADD(
    PROJECT STR_100 = 'millicard_disb')
RETURNS (
    ESM_ID ID_TYPE,
    FRONTID ID_TYPE,
    PAN STR_100,
    CARD_DATE_END DATE,
    EXPIRY STR_100,
    AMOUNT INTEGER,
    SECRETWORD STR_100,
    CURRENCY VARCHAR(3),
    BRANCHFRONTID VARCHAR(10))
AS
declare variable psi_id id_type;
declare variable mm smallint;
declare variable yy varchar(10);
declare variable k_id id_type;
declare variable esm_pan str_100;
declare variable project_id smallint;
begin
  project = coalesce(project, '');
  if (project = '') then
    project = 'millicard_disb';

  branchfrontid = 'MB_001';

  project_id = 1;
  if (project = 'millicard_biz_disb') then
    project_id = 2;

  for select em.esm_id, esm_object_id, em.esm_from_k_id, em.esm_amount * 100, em.esm_account
      from extsysmain em
      where em.esm_project = :project and
            em.esm_action = 'addCredit' and
            em.esm_status = 0
      into :esm_id, :frontid, :k_id, :amount, :esm_pan
  do
  begin
    for select psi_id
        from hp_paysys_get_cl_info(:frontid, :project_id) -- 1 - millikart, 2 - millikart Business
        order by psm_date desc
        rows 1
        into :psi_id
    do
    begin
      select pys.psi_code, psi_date_end, pys.psi_item_status
      from paysys_item pys
      where pys.psi_id = :psi_id
      into :pan, :card_date_end, :secretword;

      mm = extract(month from card_date_end);
      yy = right(cast(extract(year from card_date_end) as varchar(4)), 2);

      expiry = iif(mm < 10, '0' || mm, mm) || yy;
    end

    -- Помни!!!! Только на время теста (Тестовая карта)-->
    --secretword = 'Hello';
    --pan = '9394970000528793';
    --expiry = '0521';
    -------------------------------------<

    select kont.k_currency
    from kontract kont
    where kont.k_id = :k_id
    into :currency;

    if (pan = esm_pan) then
      suspend;
  end
end
