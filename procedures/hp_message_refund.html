CREATE PROCEDURE HP_MESSAGE_REFUND
AS
declare variable NUM_PROCESSING D_INTEGER;
declare variable NUM_ADDED D_INTEGER;
declare variable RES_TEXT D_STR_255;
declare variable FR_ID D_ID;
declare variable EDL_ID D_ID;
begin
  if (f_get_db_id() not in (39801, 41701)) then
    exit;

  num_processing = (select count(*)
                    from edoc_blank, edoc_list
                    where edb_from_edl_id = edl_id and
                          edb_status = 0 and
                          edl_status <> -1);
  num_added = 0;
  if (num_processing < 50) then
  begin
    for select first 250 fr.fr_id
        from finerefund fr
        where fr.fr_is_send = 0 and
              fr.fr_status = 1
        into fr_id
    do
    begin
      execute procedure hp_edoc_ins('info_about_finerefund', f_objkind('finerefund', 'id'), :fr_id, 1000000000000)
          returning_values :edl_id, :res_text;

      update finerefund fr
      set fr.fr_is_send = 1
      where fr.fr_id = :fr_id;

      num_added = num_added + 1;
    end
    if (num_added = 0) then
      update fsrobot_sh_main
      set fsm_is_active = 0
      where fsm_id = 71;
  end
end
