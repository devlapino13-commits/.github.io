CREATE PROCEDURE HP_PRE_SUD_KONTRACT_IN_OUT
AS
declare variable k_id d_id;
begin
  -- добавляет/исключает контракт в реестре досудебных взысканий (RFCRU-4792)
  -- выполняется роботом по расписанию

  -- добавление в реестр
  for select oks_from_k_id
      from oksummary
      left join pre_sud_kontract_list on pskl_k_id = oks_from_k_id
      where pskl_k_id is null and -- отсустствующие в реестре
            oks_close_state in (0, 1) and -- контракты с задолженностью
            current_date between oks_date_begin and oks_date_end and
            current_date - oks_delinq_date > 0 -- есть просрочка
      into k_id
  do
    execute procedure hp_pre_sud_action_log_ins(f_enum_id('pre_sud_actions', 'register_in', 1), 0, :k_id, current_timestamp, 'Робот FinStart');

  -- исключение из реестра
  for select pskl_k_id
      from pre_sud_kontract_list
      into k_id
  do
  begin
    if (not exists(select *
                   from oksummary
                   where oks_from_k_id = :k_id and
                         oks_close_state in (0, 1) and
                         current_date between oks_date_begin and oks_date_end and
                         current_date - oks_delinq_date > 0)) then
      execute procedure hp_pre_sud_action_log_ins(f_enum_id('pre_sud_actions', 'register_out', 1), 0, :k_id, current_timestamp, 'Робот FinStart');
  end
end
