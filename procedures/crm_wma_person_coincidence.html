CREATE PROCEDURE CRM_WMA_PERSON_COINCIDENCE(
    FIO STR_100,
    BIRTH_DATE DATE,
    PHONE_NO STR_100)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable c_id id_type;
declare variable res_comment str_255;
declare variable c_birth_date str_10;
declare variable is_first_row d_smallint;
declare variable c_name_fio d_str_100;
declare variable c_pass_no str_100;
declare variable cell_main str_100;
begin

json_value = '{ "coincidence_with_persons": [ ';

if (:phone_no is null) then
    begin
        is_first_row = 1;
        for select c_id, res_comment, c_name_fio, f_datetostr(c_birth_date) c_birth_date, c_pass_no, last_cell_main
        from hp_find_clients(null, :fio, :birth_date, null, null, null, null, null, null)
        inner join continfo on cci_object_id = c_id
        where is_critical = 1
        into :c_id, :res_comment, :c_name_fio, :c_birth_date, :c_pass_no, :cell_main
        do
        begin
            if (is_first_row = 0) then
            json_value = BLOB_APPEND(json_value, ', ');

            json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('c_id', :c_id, 'str') || ', '
                                || f_json_make_pair('comment',:res_comment, 'str') || ', '
                                || f_json_make_pair('fio', c_name_fio, 'str') || ', '
                                || f_json_make_pair('birth_date', c_birth_date, 'date') || ', '
                                || f_json_make_pair('pass_no', c_pass_no, 'str') || ', '
                                || f_json_make_pair('cell_main', cell_main, 'str') || ', '
                                || (select txt from cc_wma_info_part_tmp(:c_id, 0))
                                || '} ');
            if (is_first_row = 1) then
            is_first_row = 0; 
        end
    end

if (:fio is null and :birth_date is null) then
    begin
        is_first_row = 1;
        for select c_id, res_comment, c_name_fio, f_datetostr(c_birth_date) c_birth_date, c_pass_no, last_cell_main
        from hp_find_clients(null, null, null, null, null, null, :phone_no, null, null)
        inner join continfo on cci_object_id = c_id
        into :c_id, :res_comment, :c_name_fio, :c_birth_date, :c_pass_no, :cell_main
        do
        begin
            if (is_first_row = 0) then
            json_value = BLOB_APPEND(json_value, ', ');

            json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('c_id', :c_id, 'str') || ', '
                                || f_json_make_pair('comment',:res_comment, 'str') || ', '
                                || f_json_make_pair('fio', c_name_fio, 'str') || ', '
                                || f_json_make_pair('birth_date', c_birth_date, 'date') || ', '
                                || f_json_make_pair('pass_no', c_pass_no, 'str') || ', '
                                || f_json_make_pair('cell_main', cell_main, 'str') || ', '
                                || (select txt from cc_wma_info_part_tmp(:c_id, 0))
                                || '} ');
            if (is_first_row = 1) then
            is_first_row = 0; 
        end
    end
if (:fio is not null and :birth_date is not null and :phone_no is not null) then
    begin
      is_first_row = 1;
      for select c_id, res_comment, c_name_fio, f_datetostr(c_birth_date) c_birth_date, c_pass_no, last_cell_main
      from hp_find_clients(null, :fio, :birth_date, null, null, null, :phone_no, null, null)
      inner join continfo on cci_object_id = c_id
      where is_critical = 1
      into :c_id, :res_comment, :c_name_fio, :c_birth_date, :c_pass_no, :cell_main
      do
        begin
            if (is_first_row = 0) then
            json_value = BLOB_APPEND(json_value, ', ');

            json_value = BLOB_APPEND(json_value, '{ ' || f_json_make_pair('c_id', :c_id, 'str') || ', '
                                || f_json_make_pair('comment',:res_comment, 'str') || ', '
                                || f_json_make_pair('fio', c_name_fio, 'str') || ', '
                                || f_json_make_pair('birth_date', c_birth_date, 'date') || ', '
                                || f_json_make_pair('pass_no', c_pass_no, 'str') || ', '
                                || f_json_make_pair('cell_main', cell_main, 'str') || ', '
                                || (select txt from cc_wma_info_part_tmp(:c_id, 0))
                                || '} ');
            if (is_first_row = 1) then
            is_first_row = 0; 
        end
    end

 json_value = BLOB_APPEND(json_value, ' ]');
 json_value = BLOB_APPEND(json_value, ' ');
 json_value = BLOB_APPEND(json_value, ' }');
 suspend;
end
