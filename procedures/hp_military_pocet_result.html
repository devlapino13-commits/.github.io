CREATE PROCEDURE HP_MILITARY_POCET_RESULT(
    MCP_ID D_INTEGER = null,
    UN_ID D_ID = null)
AS
declare variable C_ID D_ID;
declare variable C_INN D_STR_20;
declare variable TMP_KONTRACT_LIST D_STR_LONG;
declare variable K_KONTRACT_NO D_STR_20;
declare variable C_NAME_FIO D_STR_100;
declare variable MAIL_BODY D_BLOB_TEXT;
declare variable TABLE_ROW D_STR_LONG;
declare variable MAIL_TO D_STR_255;
declare variable I D_INTEGER;
declare variable MCQ_ANSWER_DATE D_DATE;
declare variable MAIL_SUBJECT D_STR_255;
declare variable MCP_STATUS D_INTEGER;
declare variable K_DATE_BEGIN D_DATE;
begin
  -- RFCRU-3817 оповещение для Тех. поддежки и для клиентов
  mail_body = '';
  i = 0;

  if (un_id is null) then
    un_id = 1000000000000;

  if (mcp_id is null) then
  begin
    select first 1 mcp.mcp_id, mcp.mcp_status
    from military_check_pocket mcp
    order by mcp.mcp_create_date desc
    into mcp_id, mcp_status;

    if (mcp_status in (0, -2)) then
    begin
      mail_subject = 'Ошибка пакетной обработки сервис Призывник КЗ (пакетная обработка)';
      mail_body = '<strong>Ответ от сервиса Призывник КЗ (пакетная обработка) не был получен!</strong>';

      mail_to = 'kg.finstart@mbulak.kg, p.juravlev@mbulak.kg';
      execute procedure hp_send_quick_message('email', :mail_to, '', :mail_body, '', 1, :un_id);
      exit;
    end
  end

  mail_body = blob_append(mail_body, '<p>В результате пакетной обработки клиентов статус воинской службы у следующих клиентов был изменён на «Призван»</p>');

  mail_body = blob_append(mail_body, f_format('<table style="border-collapse: collapse; border: 2px solid #8c8c8c;" cellpadding="0" cellspacing="0">
  <tr>
    <th style="border: 1px solid #8c8c8c; padding: 8px; background-color: #f2f2f2;">
      %1
    </th>
    <th style="border: 1px solid #8c8c8c; padding: 8px; background-color: #f2f2f2;">
      %2
    </th>
    <th style="border: 1px solid #8c8c8c; padding: 8px; background-color: #f2f2f2;">
      %3
    </th>
    <th style="border: 1px solid #8c8c8c; padding: 8px; background-color: #f2f2f2;">
      %4
    </th>
    <th style="border: 1px solid #8c8c8c; padding: 8px; background-color: #f2f2f2;">
      %5
    </th>
  </tr><tbody>', 'Имя клиента', 'Контракты клиента', 'ИИН', 'Дата начала срока прохождения воиснкой службы', 'Дата окончания срока прохождения воиснкой службы'));

  for select mcq.mcq_from_c_id, mcq.mcq_answer_date
  from military_check_query mcq
  where mcq.mcq_from_mcp_id = :mcp_id
    and mcq.mcq_status = 1 -- берем только призывников
  into c_id, mcq_answer_date
  do
  begin
    table_row = '';
    tmp_kontract_list = '';
    c_name_fio = '';

    select c.c_name_fio, c.c_inn
    from clients c
    where c.c_id = :c_id
    into c_name_fio, c_inn;

    -- RFCKZ-755
    -- отправляем отдельный пуш на каждый контракт и собираем список для письма
    for select k.k_kontract_no, ks.ks_date_begin
        from knclient kc
        inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id
        inner join kontract k on k.k_id = ks.ks_from_k_id
        where ks.ks_ol_is_delete = 0
          and ks.ks_state = 1
          and current_date - 1 between ks.ks_date_begin and ks.ks_date_end
          and kc.kc_amount_approved > 0
          and kc.kc_from_c_id = :c_id
        order by k.k_id
        into :k_kontract_no, :k_date_begin
    do
    begin
      if (tmp_kontract_list = '') then
        tmp_kontract_list = k_kontract_no;
      else
        tmp_kontract_list = tmp_kontract_list || ', ' || k_kontract_no;

      execute procedure mc_send_push('clients', :c_id, 'Внимание!',
                        f_format('МФО уведомляет о предоставлении отсрочки платежей по основному долгу и вознаграждению по Договору о предоставлении микрокредита №%1 от %2 на период прохождения воинской службы, до %3 г., без начисления вознаграждения.', :k_kontract_no,
                        f_datetostr(:k_date_begin),
                        f_datetostr(dateadd(month, 14, :mcq_answer_date))), cast(dateadd(month, 14, :mcq_answer_date) as d_date) - current_date);
    end

    if (tmp_kontract_list <> '') then
    begin
      i = i + 1;

      table_row = f_format('<tr>
        <td style="border: 1px solid #8c8c8c; padding: 8px;">%1</td>
        <td style="border: 1px solid #8c8c8c; padding: 8px;">%2</td>
        <td style="border: 1px solid #8c8c8c; padding: 8px;">%3</td>
        <td style="border: 1px solid #8c8c8c; padding: 8px;">%4</td>
        <td style="border: 1px solid #8c8c8c; padding: 8px;">%5</td>
      </tr>',
     c_name_fio,
     tmp_kontract_list,
     c_inn,
     f_datetostr(:mcq_answer_date),
     f_datetostr(dateadd(month, 14, :mcq_answer_date)));

      mail_body = blob_append(mail_body, table_row);
    end
  end

  mail_body = blob_append(mail_body, '</tbody></table>');

  -- Отправляем письмо Тех. поддержке

  if (i > 0) then
  begin
    mail_subject = 'Изменение статуса воинской службы на «Призван» у активных клиентов';
    mail_to = 'p.juravlev@mbulak.kg, fs@mbulak.kz, a.bimendina@mbulak.kz, oper@mbulak.kz, buh@mbulak.kz';
    execute procedure hp_send_quick_message('email', :mail_to, :mail_subject, :mail_body, '', 1, :un_id);

    execute procedure fsrobot_insert_event('DoMailing', 'push_military_pocket', :mcp_id, :un_id);
  end
end
