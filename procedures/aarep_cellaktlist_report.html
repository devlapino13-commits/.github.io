CREATE PROCEDURE AAREP_CELLAKTLIST_REPORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    CAL_FROM_CAM_ID DECIMAL(18,0),
    CAL_ID INTEGER,
    CAL_DATE_BEGIN DATE,
    CAL_DATE_END DATE,
    CAL_FROM_C_ID DECIMAL(18,0),
    CAL_CELL_NO VARCHAR(20),
    CAL_FROM_COP_ID INTEGER,
    CAL_PS_TEXT VARCHAR(255),
    CAL_FIO VARCHAR(100),
    CAL_REGION_ID INTEGER,
    CAL_BRANCH_ID INTEGER,
    CAL_FROM_JL_ID INTEGER,
    CAL_FROM_JL_NAME VARCHAR(100),
    CAL_REGION_NAME VARCHAR(100),
    CAL_STATE_STR VARCHAR(20),
    CAL_IS_CONTACT_STR VARCHAR(20),
    CAL_IS_CORPORATE_STR VARCHAR(20),
    CAL_HAS_LIMIT_STR VARCHAR(100),
    CAL_IS_WORKING_STR VARCHAR(20))
AS
declare variable cal_is_contact integer;
declare variable cal_is_corporate integer;
declare variable cal_has_limit integer;
declare variable cal_is_working integer;
declare variable clk_limit_kind varchar(100);
begin
  for select cal_from_cam_id, cal_id, cal_date_begin, cal_date_end, cal_from_c_id, cal_cell_no, cal_is_contact,
             cal_is_corporate, cal_has_limit, cal_from_cop_id, cal_ps_text, cal_fio, cal_region_id, cal_branch_id,
             cal_from_jl_id, cal_from_jl_name, cal_region_name, cal_is_working, cal_state_str, clk_limit_kind
      from aarep_cellaktlist_temp(:report_region_id, :report_date_end), region_list(:report_region_id, 1, -1, 1)
      where (cal_region_id = id_list)
      into :cal_from_cam_id, :cal_id, :cal_date_begin, :cal_date_end, :cal_from_c_id, :cal_cell_no, :cal_is_contact,
           :cal_is_corporate, :cal_has_limit, :cal_from_cop_id, :cal_ps_text, :cal_fio, :cal_region_id, :cal_branch_id,
           :cal_from_jl_id, :cal_from_jl_name, :cal_region_name, :cal_is_working, :cal_state_str, :clk_limit_kind
  do
  begin
    if (cal_is_contact = 1) then
      cal_is_contact_str = 'Контактный';
    else
      cal_is_contact_str = 'Не контактный';
    if (cal_is_corporate = 1) then
      cal_is_corporate_str = 'Корпоративный';
    else
      cal_is_corporate_str = 'Личный';
--    if (CAL_HAS_LIMIT=1) then CAL_HAS_LIMIT_STR='Есть лимит'; else CAL_HAS_LIMIT_STR='Нет лимита';
    cal_has_limit_str = clk_limit_kind;
    if (cal_is_working = 1) then
      cal_is_working_str = 'Действующий';
    else
      cal_is_working_str = 'Уволенный';
    suspend;
  end
end
