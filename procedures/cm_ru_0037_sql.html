CREATE PROCEDURE CM_RU_0037_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable k_amount_approved curr_type;
declare variable k_kolich_repay integer;
declare variable pr_code varchar(20);
declare variable in_rows1 integer;
declare variable rows1_cli_id id_type;
declare variable kc_from_c_id id_type;
declare variable k_kommis_perc perc_type;
declare variable k curr_type;
declare variable kol_cli_group integer;
declare variable C_CURR_BONUS_STATUS integer;
declare variable PRL_PERCENT_MAX perc_type;
declare variable pr_id integer;
declare variable k_percent perc_type;
declare variable c_citizen integer;
declare variable rd_code varchar(255);
declare variable crel_other_c_id id_type;
declare variable rel_name varchar(100);
declare variable k_from_kf_id id_type;
declare variable kal_new_reg_id integer;
declare variable rd_name varchar(255);
declare variable is_punkt_221 smallint;
declare variable kc_circle integer;
declare variable re_k_id id_type;
declare variable sum_delinq_day integer;
declare variable sum_delinq_day_temp integer;
declare variable disb_sum_new_cli curr_type;
declare variable re_close_date date;
declare variable k_kredline_end date;
declare variable k_kolich_lgota integer;
declare variable re_kredline_end date;
declare variable re_pr_code varchar(20);
declare variable re_pr_name varchar(50);
declare variable pr_name varchar(50);
declare variable pr_product_kind integer;
declare variable pdt_code varchar(100);
declare variable CDS_DOC_DATE_BEGIN date;
declare variable CDS_DOC_DATE_END date;
declare variable pdt_name varchar(100);
declare variable is_active smallint;
begin
  for
  select k_amount_approved,
         k_kolich_repay,
         pr_code,
         k_disb_date,
         k_kommis_perc,
         k_kolich_client_approved,
         pr_id,
         k_percent,
         k_from_kf_id,
         k_kredline_end,
         k_kolich_lgota,
         pr_name
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.)%'
  into k_amount_approved,
       k_kolich_repay,
       pr_code,
       kf_disb_date,
       k_kommis_perc,
       kol_cli_group,
       pr_id,
       k_percent,
       k_from_kf_id,
       k_kredline_end,
       k_kolich_lgota,
       pr_name
  do
  begin
    select kal_new_reg_id
    from knaktlist
    where knaktlist.is_delete = 0
    and current_date between kal_date and kal_date_end
    and kal_from_kf_id = :k_from_kf_id
    into kal_new_reg_id;
    if (exists(select * from region
               where (parent = 1010000
                   or reg_unicode in (1020200, 1030300, 1031300, 1031400))
               and reg_unicode not in (1420100, 1440100)
               and reg_unicode = :kal_new_reg_id)) then
    begin
      is_punkt_221 = 1;
    end else
    begin
      is_punkt_221 = 0;
    end 
    if (is_punkt_221 = 1) then
    begin
      disb_sum_new_cli = 20000;
    end else
    begin
      disb_sum_new_cli = 15000;
    end

    C_CURR_BONUS_STATUS = null;
    select C_CURR_BONUS_STATUS
    from HP_KONTSTATE_LOAN_STATUS(:K_ID, 0, 0, :kf_disb_date, null)
    into C_CURR_BONUS_STATUS;
    if (C_CURR_BONUS_STATUS is null) then C_CURR_BONUS_STATUS = 0;
    PRL_PERCENT_MAX = null;
    select max(PRL_PERCENT_MAX)
    from prodlogic
    where prl_from_pr_id = :pr_id
    and :kf_disb_date between prl_date_begin and prl_date_end
    and :kol_cli_group between prl_kolich_min and prl_kolich_max
    and :k_kolich_repay between prl_repay_min and prl_repay_max
    into PRL_PERCENT_MAX;
    if (PRL_PERCENT_MAX is null) then PRL_PERCENT_MAX = 100;
    k_kolich_repay = k_kolich_repay - coalesce(k_kolich_lgota,0);
    if (K_Percent <> (PRL_PERCENT_MAX - C_CURR_BONUS_STATUS * 0.2)) then
    begin
       res_error_text = 'Процентная ставка не правильно указано!';
       res_error_is_critical = 2;
       suspend;
    end 
--  проверка по продукту Семья Плюс
    if (pr_code containing '.FAMILY') then
    begin
      if (pr_code containing '.FAMILY.') then
      begin
        if (kol_cli_group not in (2,3)) then
        begin
          res_error_text = 'Кол-во клиентов данного продукта не правильно указано!';
          res_error_is_critical = 2;
          suspend;
        end
        if (k_kommis_perc <> 0) then
        begin
          res_error_text = 'Для данного продукта Доп. процент не правильно указано!';
          res_error_is_critical = 2;
          suspend;
        end
      end else
      if (pr_code containing '.FAMILY PLUS.') then
      begin
          if (kol_cli_group not in (2,3,4)) then
          begin
            res_error_text = 'Кол-во клиентов данного продукта не правильно указан!';
            res_error_is_critical = 2;
            suspend;
          end 
          if (k_kommis_perc <> 1.00) then
          begin
            if (current_date <= cast('31.12.2015' as date) and k_kommis_perc = 0) then
            begin
              if (is_punkt_221 = 0) then
              begin
                res_error_text = 'Для данного продукта Доп. процент не правильно указан!';
                res_error_is_critical = 2;
                suspend;
              end
            end else
            begin
              res_error_text = 'Для данного продукта Доп. процент не правильно указан!';
              res_error_is_critical = 2;
              suspend;
            end
          end
      end 
      in_rows1 = 1;
      rows1_cli_id = 0;
      for
      select kc_from_c_id, kc_amount_approved, kc_circle
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into kc_from_c_id, kc_amount, kc_circle
      do
      begin
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into res_info_text;
        if (in_rows1 = 1) then
        begin
          in_rows1 = 0;
          rows1_cli_id = kc_from_c_id;
        end else
        begin
          if (pr_code containing '.FAMILY.') then
          begin
--            в продекте семья все члены группы должны быть находится в близком родстве
              if ((:kc_from_c_id not in (select CREL_OTHER_C_ID
                                         from HP_CLIENT_RELATION_LIST(:rows1_cli_id,0)
                                         where crel_from_rel_id in ('1','2','3','4','5','6','7','8','37','38','47','48','75','76','85','76',
                                                 '375','376','385','386','475','476','485','486')
                                          )
                  )) then
              begin
                res_error_text = 'Клиент '||res_info_text||' не является родственником!';
                res_error_is_critical = 0;
                suspend;
              end
          end else
          if (pr_code containing '.FAMILY PLUS.') then
          begin
--            В продукте семья плюсь наоборот не должны быть близкие родственники
              /*  -- убран по письму Зары, в данном продукте все могут получать заем
              for
              select crel_other_c_id, rel_name
              from HP_CLIENT_RELATION_LIST(:rows1_cli_id,0)
              where cast(crel_from_rel_id as integer) in (1,2,3,4,5,6,7,8,37,38,47,48,75,76,85,76,
                                                          375,376,385,386,475,476,485,486)
              into crel_other_c_id, rel_name
              do
              begin
                if (kc_from_c_id = crel_other_c_id) then
                begin
                  res_error_text = 'Клиент '||res_info_text||' является - '||rel_name;
                  res_error_is_critical = 1;
                  suspend;
                end
              end
              */
          end
        end
--      проверка на соответсвии суммы на срок кредита
        if (kc_amount = 15000) then
        begin
          if (k_kolich_repay not in (3,6)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end 
        end else
        if (kc_amount = 25000) then
        begin
          if (k_kolich_repay not in (3,6,9)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end 
        end else
        if (kc_amount = 30000) then
        begin
          if (k_kolich_repay not in (6,9)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end 
        end else
        if (kc_amount = 40000) then
        begin
          if (k_kolich_repay not in (6,9,12)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end 
        end else
        if (kc_amount in (50000,60000)) then
        begin
          if (k_kolich_repay not in (9,12)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end
        end else
        begin
          res_error_text = 'У клиента указана не правильная сумма!';
          res_error_is_critical = 2;
          suspend;
        end
        if (kc_circle <= 1) then
        begin
          if (kc_amount > 30000) then
          begin
            res_error_text = 'Клиент новый и не может брать более 30тыс. руб.';
            res_error_is_critical = 1;
            suspend;
          end 
        end else
        if (kc_amount > 30000) then
        begin
--        проверяется предыдущий цикл клиента на просрочку
          re_k_id = null;
          select kc_from_k_id,
                 ks_date_end,
                 k_kredline_end,
                 pr_code,
                 pr_name,
                 pr_product_kind
          from knclient, kontstate, kontract, product
          where kc_from_c_id = :kc_from_c_id
          and kc_amount_approved > 0
          and kc_from_k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and ks_from_k_id <> :k_id
          and ks_date_begin < :kf_disb_date
          and ks_from_k_id = k_id
          and k_from_pr_id = pr_id
          order by ks_date_begin desc
          rows 1
          into re_k_id,
               re_close_date,
               re_kredline_end,
               re_pr_code,
               re_pr_name,
               pr_product_kind;
          if (re_k_id is null) then re_k_id = 0;
          if (re_k_id <> 0) then
          begin
--          если в предыдущем цикле брал не индив кредит или 1+1 то проверяется следующее
            if (re_pr_code not containing '.ind.' or re_pr_code containing '.one.') then
            begin
                --вычисляется макс кол-во просроченных дней
                sum_delinq_day = 0;
                sum_delinq_day_temp = 0;
                for select max(ok_ol_date - ok_sh_date) from operkred
                where ok_ol_is_active = 1
                and ok_from_k_id = :re_k_id
                and ok_type in (4,5,9)
                and ok_os + ok_perc < -100
                and ok_ol_date > ok_sh_date
                group by ok_sh_date
                into sum_delinq_day_temp
                do
                begin
                   sum_delinq_day = sum_delinq_day + sum_delinq_day_temp;
                end 
    
                if (sum_delinq_day is null) then sum_delinq_day = 0;
    
                if (sum_delinq_day > 7) then
                begin
                  res_error_text = 'Клиент допустил просрочку более 7 дней суммарно и не может получить более 30тыс.руб.';
                  res_error_is_critical = 1;
                  suspend;
                end else
                begin
    --            если суммарных просрочек более 7 дней не имеется то проверяет на досрочное погашение
                  if (re_close_date < current_date) then
                  begin
                    re_close_date = re_close_date + 1;
                    if (re_close_date + 10 < re_kredline_end
                        and re_pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.|.ASALAM.|.ZNAKOM.|.UVAJENIE.)%' -- igusev 20.11.2015 добавил .ASALAM.|.ZNAKOM.|.UVAJENIE.
                        ) then
                    begin
                      if (not(re_pr_code similar to '%(.KEL.|.PARTNER.|.DOVERIE.)%')) then
                      begin
                        res_error_text = 'Клиент предыдущий цикл закрыл досрочно, запрещается брать более 30 тыс. руб';
                        res_error_is_critical = 1;
                        suspend;
                      end
                    end
                  end else
                  begin
                    if (pr_product_kind <> 2) then
                    begin
                        res_error_text = 'У клиента имеется активный кредит!';
                        res_error_is_critical = 1;
                        suspend;
                    end
                  end
                end
            end else
            begin
              res_error_text = 'Передыдущий продукт клиента -'||re_pr_name||' и не может получить более 30 тыс. по продукту '||pr_name;
              res_error_is_critical = 1;
              suspend;
            end 
          end else
          begin
            res_error_text = 'Клиент новый и не может брать более 30тыс. руб.';
            res_error_is_critical = 1;
            suspend;
          end 

        end
      end
    end 
--  проверка по продукту Баракат
    if (pr_code containing '.BARAKAT') then
    begin
      if (kol_cli_group <> 1) then
      begin
        res_error_text = 'Кол-во клиентов данного продукта не правильно указан!';
        res_error_is_critical = 2;
        suspend;
      end
      if (pr_code containing '.BARAKAT.') then
      begin
          k = 50000;
          if (k_kommis_perc <> 1.50) then
          begin
            res_error_text = 'Для данного продукта Доп. процент не правильно указан!';
            res_error_is_critical = 2;
            suspend;
          end
      end else
      if (pr_code containing '.BARAKAT PLUS.') then
      begin
          k = 60000;
          if (k_kommis_perc <> 1.0) then
          begin
            res_error_text = 'Для данного продукта Доп. процент не правильно указан!';
            res_error_is_critical = 2;
            suspend;
          end
      end 
      for
      select kc_from_c_id, kc_amount_approved, kc_circle
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into kc_from_c_id, kc_amount, kc_circle
      do
      begin
        select c_name_fio, c_citizen
        from clients
        where c_id = :kc_from_c_id
        into res_info_text, c_citizen;
        select rd_code, rd_name from refdata
        where rd_id = :c_citizen
        into rd_code, rd_name;
        if (rd_code not in ('Citizen.KG', 'Citizen.UZ', 'Citizen.TJ')) then
        begin
          res_error_text = 'По выбранному продукту гражданины '||rd_name||' не могут брать.';
          res_error_is_critical = 1;
          suspend;
        end 
--    Обязательно наличие у клиента Патента или Разрешения на работу
--    Разработан по пункту 5
        is_active = 0;
        for
        /*
        select pdt_code,
               CDS_DOC_DATE_BEGIN,
               CDS_DOC_DATE_END,
               pdt_name
        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
        where CDS_FROM_C_ID = :kc_from_c_id
        and cds_from_pk_id = pk_id
        and pk_from_pdt_id = pdt_id
        and cds_is_delete = 0
        and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
        /*/
        select pdt_code,
               min(CDS_DOC_DATE_BEGIN),
               max(CDS_DOC_DATE_END),
               pdt_name
        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
        where CDS_FROM_C_ID = :kc_from_c_id
        and cds_from_pk_id = pk_id
        and pk_from_pdt_id = pdt_id
        and cds_is_delete = 0
        and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
        group by pdt_code, pdt_name  -- */
        into pdt_code,
             CDS_DOC_DATE_BEGIN,
             CDS_DOC_DATE_END,
             pdt_name
        do
        begin
          if (:rd_code in ('Citizen.TJ', 'Citizen.UZ')) then
          begin
            if (pr_code containing '.BARAKAT.') then
            begin
              if (pdt_code in ('work_permit', -- Разрешение на работу
                               'work_patent')  -- Патент) then
                  ) then
              begin
                if (dateadd(month, 3, :kf_disb_date) > CDS_DOC_DATE_END) then
                begin
                  res_error_text = 'Для гражд. ТЖ и УЗ'||pdt_name||' должны быть действительны не менее 3 месяцев после получения микрозайма';
                  res_error_is_critical = 1;
                  suspend;
                end else
                begin
                  is_active = 1;
                end 
              end 
            end else
            if (pr_code containing '.BARAKAT PLUS.') then
            begin
              if (pdt_code in ('registration_rvp', -- Разрешение на временное проживание (РВП)
                               'registration_vnj') -- Вид на жительство (ВНЖ)
                  ) then
              begin
                if (dateadd(month, 3, :kf_disb_date) > CDS_DOC_DATE_END) then
                begin
                  res_error_text = 'Для гражд. ТЖ и УЗ'||pdt_name||' должны быть действительны не менее 3 месяцев после получения микрозайма';
                  res_error_is_critical = 1;
                  suspend;
                end else
                begin
                  is_active = 1;
                end 
              end 
            end 
          end else
          if (:rd_code = 'Citizen.KG') then
          begin
            if (CDS_DOC_DATE_END < :k_kredline_end) then
            begin
              res_error_text = 'Для гражд. КГ '||pdt_name||' должны быть действительны на весь срок микрозайма';
              res_error_is_critical = 1;
              suspend;
            end else
            begin
              is_active = 1;
            end 
          end
        end 
        if (is_active = 0) then
        begin
          if (:rd_code = 'Citizen.KG') then
          begin
            res_error_text = 'Для гражд. КГ не имеется документ миграционного учета.';
            res_error_is_critical = 1;
            suspend;
          end else
          begin
            if (pr_code containing '.BARAKAT.') then
            begin
              res_error_text = 'У клиента должно быть патент или разрешения на работу!';
              res_error_is_critical = 1;
              suspend;
            end else
            if (pr_code containing '.BARAKAT PLUS.') then
            begin
              res_error_text = 'У клиента должно быть Разрешения на временное проживание или Вид на жительство';
              res_error_is_critical = 1;
              suspend;
            end 
          end
        end
      /*
        if (pr_code containing '.BARAKAT.') then
        begin
          if (not exists(
                select *
                from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                where CDS_FROM_C_ID = :kc_from_c_id
                and cds_from_pk_id = pk_id
                and pk_from_pdt_id = pdt_id
                and cds_is_delete = 0
                and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                and (pdt_code in ('work_permit', -- Разрешение на работу
                                  'work_patent')  -- Патент
                     or (:rd_code in ('Citizen.KG')
                         and pdt_code = 'magration_card')
                    ))) then
          begin
            res_error_text = 'У клиента должно быть патент или разрешения на работу!';
            res_error_is_critical = 1;
            suspend;
          end
        end else
        if (pr_code containing '.BARAKAT PLUS.') then
        begin
          if (not exists(
                select *
                from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                where CDS_FROM_C_ID = :kc_from_c_id
                and cds_from_pk_id = pk_id
                and pk_from_pdt_id = pdt_id
                and cds_is_delete = 0
                and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                and (pdt_code in ('registration_rvp', -- Разрешение на временное проживание (РВП)
                                 'registration_vnj') -- Вид на жительство (ВНЖ)
                    or (:rd_code in ('Citizen.KG')
                         and pdt_code = 'magration_card'))
                    )) then
          begin
              res_error_text = 'У клиента должно быть Разрешения на временное проживание или Вид на жительство';
              res_error_is_critical = 1;
              suspend;
          end
        end
       */
--      проверка на соответсвии суммы на срок кредита
        if (kc_amount = 10000) then
        begin
          if (k_kolich_repay not in (3)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end 
        end else
        if (kc_amount = 15000) then
        begin
          if (k_kolich_repay not in (3,6)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end 
        end else
        if (kc_amount = 20000) then
        begin
          if (k_kolich_repay not in (3,6,9)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end 
        end else
        if (kc_amount in (30000,40000)) then
        begin
          if (k_kolich_repay not in (6,9,12)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end 
        end else
        if (kc_amount in (50000,:k)) then
        begin
          if (k_kolich_repay not in (9,12)) then
          begin
            res_error_text = 'Срок не соответствует к сумме кредита!';
            res_error_is_critical = 2;
            suspend;
          end
        end else
        begin
          res_error_text = 'У клиента указана не правильная сумма!';
          res_error_is_critical = 2;
          suspend;
        end
        if (kc_circle <= 1) then
        begin
          if (kc_amount > :disb_sum_new_cli) then
          begin
            res_error_text = '1. Клиент новый и не может брать более '||:disb_sum_new_cli||' руб.';
            res_error_is_critical = 1;
            suspend;
          end 
        end else
        if (kc_amount > :disb_sum_new_cli) then
        begin
--        проверяется предыдущий цикл клиента на просрочку
          re_k_id = null;
          select kc_from_k_id, ks_date_end, k_kredline_end, pr_code, pr_name
          from knclient, kontstate, kontract, product
          where kc_from_c_id = :kc_from_c_id
          and kc_amount_approved > 0
          and kc_from_k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and ks_from_k_id <> :k_id
          and ks_date_begin < :kf_disb_date
          and ks_from_k_id = k_id
          and k_from_pr_id = pr_id
          order by ks_date_begin desc
          rows 1
          into re_k_id, re_close_date, re_kredline_end, re_pr_code, re_pr_name;
          if (re_k_id is null) then re_k_id = 0;
          sum_delinq_day = null;
          if (re_k_id <> 0) then
          begin
--          если в предыдущем цикле брал не индив кредит или 1+1 то проверяется следующее
            if (re_pr_code containing '.ind.' and re_pr_code not containing '.one.') then
            begin
                --вычисляется макс кол-во просроченных дней
                sum_delinq_day = 0;
                sum_delinq_day_temp = 0;
                for select max(ok_ol_date - ok_sh_date) from operkred
                where ok_ol_is_active = 1
                and ok_from_k_id = :re_k_id
                and ok_type in (4,5,9)
                and ok_os + ok_perc < -100
                and ok_ol_date > ok_sh_date
                group by ok_sh_date
                into sum_delinq_day_temp
                do
                begin
                   sum_delinq_day = sum_delinq_day + sum_delinq_day_temp;
                end 
    
                if (sum_delinq_day is null) then sum_delinq_day = 0;
    
                if (sum_delinq_day > 7) then
                begin
                  res_error_text = 'Клиент допустил просрочку более 7 дней суммарно и не может получить более '||:disb_sum_new_cli;
                  res_error_is_critical = 1;
                  suspend;
                end else
                begin
    --            если суммарных просрочек более 7 дней не имеется то проверяет на досрочное погашение
                  if (re_close_date < current_date) then
                  begin
                    re_close_date = re_close_date + 1;
                    if (re_close_date + 10 < re_kredline_end
                        and re_pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.|.ASALAM.|.ZNAKOM.|.UVAJENIE.)%' -- igusev 20.11.2015 добавил .ASALAM.|.ZNAKOM.|.UVAJENIE.
                        ) then
                    begin
                      if (not(re_pr_code similar to '%(.KEL.|.PARTNER.|.DOVERIE.|.PP.)%')) then
                      begin
                        res_error_text = 'Клиент предыдущий цикл закрыл досрочно, запрещается брать более2 '||:disb_sum_new_cli;
                        res_error_is_critical = 1;
                        suspend;
                      end
                    end
                  end else
                  begin
                    res_error_text = 'У клиента имеется активный кредит!';
                    res_error_is_critical = 1;
                    suspend;
                  end
                end
              end else
              begin
                res_error_text = 'Передыдущий продукт клиента -'||re_pr_name||' и не может получить более '||:disb_sum_new_cli||' по продукту '||pr_name;
                res_error_is_critical = 1;
                suspend;
              end
          end else
          begin
            res_error_text = '2. Клиент новый и не может брать более '||:disb_sum_new_cli;
            res_error_is_critical = 1;
            suspend;
          end
        end
      end
    end 
  end
