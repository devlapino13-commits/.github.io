CREATE PROCEDURE HP_INCOME_INFO(
    C_ID D_ID,
    MAC_ID D_ID)
RETURNS (
    RES_HTML D_STR_8100)
AS
declare variable SUBJECT_ID D_ID;
declare variable RD_CODE D_STR_255;

declare function get_html_result(c_id d_id, mac_id d_id, config d_str_code, title d_str_255)
  returns d_str_4096
  as
  declare variable status d_smallint;
  declare variable status_str d_str_255;
  declare variable eil_id d_id;
  declare variable query_link d_str_2048;
  declare variable res_html d_str_4096;
  declare variable tmp_error_count d_integer;
  declare variable c_name_fio d_str_100;
  begin
      status = null;
      status_str = '';
      eil_id = 0;
      query_link = '';

      select c.c_name_fio
      from clients c
      where c.c_id = :c_id
      into :c_name_fio;

      select count(*)
      from income_check_querys icq
      where icq.icq_from_mac_id = :mac_id and
            icq.icq_from_c_id = :c_id and
            icq.icq_config = :config and
            icq.icq_status = -1
      into tmp_error_count;

      tmp_error_count = coalesce(tmp_error_count, 0);

      select icq.icq_status,
             case
               when icq.icq_status = -2 and :config = 'asp_check'
                 then '<span style="color: red;">Данные не получены</span>'
               when icq.icq_status = -2
                 then '<span style="color: red;">Ошибка. Прикрепите выписку по карте </span>'
               when icq.icq_status = -1 and :config = 'asp_check'
                 then 'Сервер ЦРТР недоступен'
               when icq.icq_status = -1
                 then '<span style="color: red;">Ошибка при получении справки о доходах клиента</span>'
               when icq.icq_status = 0 then 'Запрос отправлен'
               else ''
             end,
             coalesce(icq.icq_from_eil_id, 0)
      from income_check_querys icq
      where icq.icq_from_mac_id = :mac_id and
            icq.icq_from_c_id = :c_id and
            icq.icq_config = :config
      order by icq.icq_status_datetime desc
      rows 1
      into status,
           status_str,
           eil_id;

      if (status = -2 and exists(select *
                                 from clientdocs cds
                                 where cds.cds_from_c_id = :c_id and
                                       cds.cds_is_delete = 0 and
                                       cds.cds_from_pk_id in (370))) then  -- (PASSKIND 370 - Выписка по карте)
        status_str = 'Выписка по карте успешно прикреплена';

      eil_id = coalesce(eil_id, 0);

      if (eil_id = 0 or (status = -3)) then
      begin
        query_link = f_uri_sql('Запросить', :config, 'miniappclient', :mac_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', '');
        status_str = 'Запрос не отправлен';
      end
      if (status = -1 and eil_id <> 0) then
      begin
        query_link = f_uri_sql('Запросить повторно', :config, 'miniappclient', :mac_id, 'none', 'Подтверждаете?', '', 0, 0, '', '', 'dialog', '');
      end
      if (status = 0 and eil_id <> 0) then
        query_link = 'Запрос обрабатывается';
      if (status = 2) then
      begin
        status_str = 'Ответ получен';
        query_link = '';
      end

      if (((status <> 2 or eil_id = 0) and tmp_error_count < 3) or status = -3) then
      begin
        res_html = blob_append(res_html, f_format('<h2>%1</h2>', :title));
        res_html = blob_append(res_html, '<table width="100%">
           <col width="40%">
           <col width="30%">
           <col width="30%">
           <tr>');
    
        res_html = blob_append(res_html, f_html_table_row(:c_name_fio,
                                                            status_str,
                                                            query_link,
                                                            '', -1, -1, -1, -1, 3, ''));
        res_html = blob_append(res_html, '</table>');
      end
      else if (status = 2) then
      begin
        query_link = f_uri_html(:c_name_fio, f_format('html_%1', :config), :title, f_objkind('extinfolist', 'id'), :eil_id, current_date, current_date);

        res_html = blob_append(res_html, f_format('<h2>%1</h2>', :title));
        res_html = blob_append(res_html, '<table width="100%">
           <col width="100%">
           <tr>');
    
        res_html = blob_append(res_html, f_html_table_row(query_link,
                                                            '',
                                                            '',
                                                            '', -1, -1, -1, -1, 1, ''));
        res_html = blob_append(res_html, '</table>');
      end
      else if (tmp_error_count >= 3) then
      begin
        if (:config = 'asp_check') then
          status_str = '<span style="color: red;">Данные не получены</span>';
        else
        begin
          status_str = '<span style="color: red;">Ошибка. Прикрепите выписку по карте</span>';
        end
        res_html = blob_append(res_html, f_format('<h2>%1</h2>', :title));
        res_html = blob_append(res_html, '<table width="100%">
           <col width="40%">
           <col width="60%">
           <tr>');
    
        res_html = blob_append(res_html, f_html_table_row(:c_name_fio,
                                                          :status_str,
                                                          '',
                                                          '', -1, -1, -1, -1, 2, ''));
        res_html = blob_append(res_html, '</table>');
      end

      return res_html;
  end

begin
-- RFCKG-4255 Справки о доходах (ЗП, пенсия).
-- RFCRU-3940 Справка АСП
  res_html = '';

  for select rd_code,
             opr_subject_id
      from objectvalues, objectprops, objectvalres, refdata
      where opr_object_kind = 0 and
            opr_object_id = :c_id and
            opr_subject_kind = '-1' and
            opr_subject_id in (9264, 9262, 9265) and -- 9264 - Является ли клиент пенсионером (Да/Нет) см. refdata
            ovs_from_opr_id = opr_id and       -- 9262 - Является ли клиент трудоустроенным в настоящее время (Да/Нет) см. refdata
            ovr_result_rd_id = rd_id and      --9265 - Является получателем АСП(Да/Нет)
            ovs_id = opr_actual_ovs_id and
            ovr_from_opr_id = ovs_from_opr_id and
            ovr_from_ovs_id = ovs_id
      into rd_code,
           subject_id
  do
  begin

    if (subject_id = 9262 and rd_code containing 'Yes') then
      res_html = blob_append(res_html, get_html_result(:c_id, :mac_id, 'salary_check', 'Справка о заработной плате'));

    if (subject_id = 9264 and rd_code containing 'Yes') then
      res_html = blob_append(res_html, get_html_result(:c_id, :mac_id, 'pension_check', 'Справка о пенсии'));
        
    if (subject_id = 9265 and rd_code containing 'Yes') then
      res_html = blob_append(res_html, get_html_result(:c_id, :mac_id, 'asp_check', 'Справка АСП'));

  end
  suspend;
end
