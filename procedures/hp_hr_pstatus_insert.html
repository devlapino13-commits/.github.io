CREATE PROCEDURE HP_HR_PSTATUS_INSERT(
    PS_ID INTEGER,
    PS_FROM_HO_ID ID_TYPE,
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_FROM_REG_ID INTEGER,
    PS_FROM_DEP_ID INTEGER,
    PS_FROM_JL_ID INTEGER,
    PS_IS_ON_STAFF SMALLINT,
    PS_IS_WORKING SMALLINT,
    PS_FROM_C_ID ID_TYPE,
    PS_SALARY CURR_TYPE,
    PS_TRANSFER_OFFICE INTEGER,
    PS_TABEL_NO VARCHAR(10),
    PS_CLASS INTEGER,
    PS_STATUS INTEGER,
    PSA_AGREEMENT_NO VARCHAR(10))
AS
begin
  update or insert into hr_pstatus (ps_id, ps_from_ho_id, ps_date_begin, ps_period_month, ps_date_end_in_order,
                                    ps_from_reg_id, ps_from_dep_id, ps_from_jl_id, ps_is_on_staff, ps_is_working,
                                    ps_from_c_id, ps_salary, ps_transfer_office, ps_tabel_no, ps_class)
  values (:ps_id, :ps_from_ho_id, :ps_date_begin, :ps_period_month, :ps_date_end_in_order, :ps_from_reg_id,
          :ps_from_dep_id, :ps_from_jl_id, :ps_is_on_staff, :ps_is_working, :ps_from_c_id, :ps_salary,
          :ps_transfer_office, :ps_tabel_no, :ps_class)
  matching (ps_from_ho_id, ps_id)
  returning ps_from_ho_id, ps_id
  into :ps_from_ho_id, :ps_id;

  if (f_get_db_id() in (39801, 41701) and
      exists(select ho.ho_order_type, ho.ho_status
             from hr_orders ho
             where ho.ho_id = :ps_from_ho_id and
                   ho.ho_status = 1 and
                   ho.ho_order_type = 1)) then
    execute procedure hp_zeroing_rebuke_scores(:ps_from_c_id);

  update or insert into hr_pstatus_agreement (psa_from_ho_id, psa_from_ps_id, psa_agreement_no)
  values (:ps_from_ho_id, :ps_id, :psa_agreement_no)
  matching (psa_from_ho_id, psa_from_ps_id);
end
