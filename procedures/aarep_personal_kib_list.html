CREATE PROCEDURE AAREP_PERSONAL_KIB_LIST(
    REP_DATE_END DATE,
    REP_C_ID ID_TYPE = 0)
RETURNS (
    REG_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    KBM_PR_ID_DATETIME TIMESTAMP,
    KBM_CHECKED_DATETIME TIMESTAMP,
    KBM_ID ID_TYPE,
    C_ID ID_TYPE,
    REG_ID INTEGER,
    IS_RESULT INTEGER,
    IS_SENT INTEGER,
    KBM_FROM_UN_ID ID_TYPE,
    IS_PERSONAL INTEGER,
    SAWT_NAME VARCHAR(100))
AS
declare variable reg_id_prior integer;
declare variable jl_id integer;
declare variable jl_id_prior integer;
declare variable kbm_is_checked smallint;
declare variable c_id_prior id_type;
declare variable rep_region_id integer;
declare variable is_sort integer;
begin
  reg_id_prior = 0;
  jl_id_prior = 0;
  rep_c_id = coalesce(rep_c_id, 0);
  c_id_prior = 0;
  if (rep_c_id = 0) then
  begin
    for with tabl1
        as (select ps_from_c_id, ps_from_reg_id, ps_from_jl_id, 1 as is_sort
            from hr_pstatus
            where ps_is_working = 1 and
                  ps_status = 1 and
                  :rep_date_end between ps_date_begin and ps_date_end
            union
            select kbm_from_c_id, 0, 0, 2
            from kibminiapp
            where kbm_is_candidate = 1)
        select ps_from_c_id, max(ps_from_reg_id), max(ps_from_jl_id), min(is_sort)
        from tabl1
        group by 1
        order by 4, 2
        into c_id, reg_id, jl_id, is_sort
    do
    begin
      if (is_sort = 1) then
        is_personal = 1;
      else
        is_personal = 0;
      sawt_name = '';
      if (is_personal = 1) then
      begin
        select sawt_name
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              hrv_from_c_id = :c_id and
              :rep_date_end between hrv_date_begin and hrv_date_end_fact
        order by hrv_date_begin desc
        rows 1
        into sawt_name;
      end

      if (reg_id_prior <> reg_id) then
      begin
        reg_name = '';
        reg_id_prior = reg_id;
        select reg_code || ' ' || reg_name
        from region
        where reg_unicode = :reg_id
        into reg_name;
      end

      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name_fio;

      if (jl_id_prior <> jl_id) then
      begin
        jl_name = '';
        jl_id_prior = jl_id;
        select jl_name
        from joblist
        where jl_id = :jl_id
        into jl_name;
      end

      kbm_id = null;
      kbm_pr_id_datetime = null;
      kbm_checked_datetime = null;
      kbm_from_un_id = null;
      kbm_is_checked = null;
      select kbm_id, kbm_pr_id_datetime, kbm_checked_datetime, kbm_is_checked, kbm_from_un_id
      from kibminiapp
      where kbm_from_c_id = :c_id
      order by kbm_id desc
      rows 1
      into kbm_id, kbm_pr_id_datetime, kbm_checked_datetime, kbm_is_checked, kbm_from_un_id;

      kbm_id = coalesce(kbm_id, 0);
      kbm_is_checked = coalesce(kbm_is_checked, 0);

      is_sent = 0;
      is_result = 0;
      if (kbm_id <> 0) then
      begin
        /* is_sent
                0 - не проверен
                1 - отправлен на проверку
                2 - имеется результат
                3 - устарелая информация  -- если более полгода не проверялась то устраелая информация
                4 - После проверки не имеется результат т.е. без кредитной истории
              */
        if (kbm_is_checked = 0) then
          is_sent = 1;
        if (exists(select *
                   from kibhistory
                   where kh_from_kbm_id = :kbm_id)) then
          is_result = 1;

        if (is_sent = 0) then
        begin
          if (is_result = 1) then
          begin
            if (dateadd(month, -6, current_date) > cast(kbm_checked_datetime as date)) then
            begin
              is_sent = 3;
            end
            else
            begin
              is_sent = 2;
            end
          end
          else
          begin
            if (dateadd(month, -6, current_date) > cast(kbm_checked_datetime as date)) then
            begin
              is_sent = 3;
            end
            else
            begin
              is_sent = 4;
            end
          end
        end
      end
      suspend;
    end
  end
  else
  if (rep_c_id <> 0) then
  begin
    reg_id = null;
    jl_name = null;
    select jl_name, ps_from_reg_id
    from joblist, hr_pstatus
    where jl_id = ps_from_jl_id and
          ps_is_working = 1 and
          ps_status = 1 and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_c_id = :rep_c_id
    into jl_name, reg_id;
    if (jl_name is null) then
      is_personal = 0;
    else
      is_personal = 1;
    reg_name = '';
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    kbm_id = null;
    kbm_pr_id_datetime = null;
    kbm_checked_datetime = null;
    kbm_from_un_id = null;
    kbm_is_checked = null;
    select kbm_id, kbm_pr_id_datetime, kbm_checked_datetime, kbm_is_checked, kbm_from_un_id
    from kibminiapp
    where kbm_from_c_id = :rep_c_id
    order by kbm_id desc
    rows 1
    into kbm_id, kbm_pr_id_datetime, kbm_checked_datetime, kbm_is_checked, kbm_from_un_id;

    kbm_id = coalesce(kbm_id, 0);
    kbm_is_checked = coalesce(kbm_is_checked, 0);

    is_sent = 0;
    is_result = 0;
    if (kbm_id <> 0) then
    begin
      /* is_sent
            0 - не проверен
            1 - отправлен на проверку
            2 - имеется результат
            3 - устарелая информация  -- если более полгода не проверялась то устраелая информация
            4 - После проверки не имеется результат т.е. без кредитной истории
          */
      if (kbm_is_checked = 0) then
        is_sent = 1;
      if (exists(select *
                 from kibhistory
                 where kh_from_kbm_id = :kbm_id)) then
        is_result = 1;

      if (is_sent = 0) then
      begin
        if (is_result = 1) then
        begin
          if (dateadd(month, -6, current_date) > cast(kbm_checked_datetime as date)) then
          begin
            is_sent = 3;
          end
          else
          begin
            is_sent = 2;
          end
        end
        else
        begin
          if (dateadd(month, -6, current_date) > cast(kbm_checked_datetime as date)) then
          begin
            is_sent = 3;
          end
          else
          begin
            is_sent = 4;
          end
        end
      end
    end
    suspend;
  end
end
