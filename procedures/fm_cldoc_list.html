CREATE PROCEDURE FM_CLDOC_LIST(
    UN_ID D_ID,
    C_ID D_ID,
    SHOW_FULL_LIST D_BOOL = 1)
RETURNS (
    ID D_INTEGER,
    TITLE D_STR_100,
    MEMO D_STR_LONG,
    ERR_TEXT D_STR_255,
    NEED_DOC_NO D_BOOL,
    NEED_DATE_BEGIN D_BOOL,
    NEED_DATE_END D_BOOL,
    NEED_DISB_WHO D_BOOL,
    IS_PASS D_BOOL,
    NEED_PHOTO1 D_BOOL,
    NEED_PHOTO2 D_BOOL,
    IS_LATIN D_SMALLINT,
    DOC_NO D_STR_CODE,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    IS_MAIN D_BOOL,
    CAN_LOAD D_BOOL,
    DOC_ORG D_STR_255,
    DOC_ORG_CODE D_STR_20,
    DOC_ORG_REF D_STR_255,
    DRM_IS_MANUAL D_BOOL,
    DRM_CODE D_STR_255,
    DRM_API_CODE D_STR_255,
    DRM_RECOG_2 D_BOOL,
    DRM_SHOW_FIELDS_STAFF D_BOOL,
    DRM_PHOTO_SIZE D_SMALLINT,
    FILE_EXT D_STR_10)
AS
declare variable ATTACH_GALLERY_ACCESS D_BOOL; /* SHOW_FULL_LIST */
declare variable C_NAME_FIO D_STR_100;
declare variable C_CITIZEN D_INTEGER;
declare variable C_VERIFY_STATUS D_SMALLINT;
declare variable C_PASS_WHO_CODE D_STR_20;
declare variable C_DOC_TYPE D_INTEGER;
declare variable IS_ACTIVE_CLIENT D_BOOL;
declare variable IS_EMPLOYEE D_BOOL;
declare variable HAS_MINIAPP D_BOOL;
declare variable HAS_ERR_OIVO D_BOOL;
declare variable DB_ID D_INTEGER;
declare variable PDT_CODE D_STR_100;
declare variable IS_NEW_CLIENT D_BOOL;
declare variable IS_NEW_CLIENTDOCS D_BOOL;
declare variable C_CITIZEN_2 D_INTEGER;
declare variable DRM_ID D_INTEGER;
declare variable C_AGE D_INTEGER;
begin
  db_id = f_get_db_id();
  has_err_oivo = 0;

  -- Доступ на прикрепление из галереи телефона (обычно у менеджера)
  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.AttachGalleryAccess')
  into attach_gallery_access;

  if (exists(select *
          from kontstate ks
          inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
          where ks.ks_state = 1
          and ks.ks_ol_is_delete = 0
          and kncl.kc_from_c_id = :c_id
          and kncl.kc_amount_approved  > 0 )) then
    is_new_client = 0;
  else
    is_new_client = 1;

  is_new_clientdocs = f_get_const('new_clientdocs');

  if (coalesce(c_id, 0) <> 0) then
  begin
    select c_name_fio, c_citizen, c_verify_status, c_doc_type, c_pass_who_code, f_age(c_birth_date, current_date)
    from clients
    where c_id = :c_id
    into c_name_fio, c_citizen, c_verify_status, c_doc_type, c_pass_who_code, :c_age;

    select pte_citizen
    from perstemp
    where pte_from_c_id = :c_id
    into c_citizen_2;

    if (c_citizen_2 is null) then
      c_citizen_2 = c_citizen;

    select is_active_client, is_employee
    from fm_client_status(:un_id, :c_id)
    into is_active_client, is_employee;

    select has_miniapp
    from fm_client_has_miniapp(:un_id, :c_id)
    into :has_miniapp;

    if (db_id = 81001 and
        exists(select 1
               from hp_check_oivo_assignment(:c_id, :un_id)
               where is_oivo = 1 and
                     res_id <> 1)) then
      has_err_oivo = 1;

    for select pk_id, pk_name, pdt_need_doc_no, pdt_need_date_begin, pdt_need_date_end, pdt_need_disb_who,
               pdt_is_main_doc, pdt_code, pk_is_latin, coalesce(pk_can_load, 0), pk_passport_issuers,
               drm_is_manual, drm_code, drm_api_code, drm_recog_2, drm_show_fields_staff, drm_photo_size,
               pk_file_ext, drm_id
        from hp_citizen_passkind(:c_citizen, :c_citizen_2)
        inner join passkind on res_pk_id = pk_id and pk_id > 0
        inner join passdoctype on pk_from_pdt_id = pdt_id
        left join docrecogmain on pk_from_drm_id = drm_id and drm_inactive = 0 and (:is_new_clientdocs = 1 or drm_is_manual = 0)
        where
          :show_full_list = 1 or
          pk_is_other_category = 0
        order by pdt_is_main_doc desc, pk_name, pk_order, pk_id
        into id, title, need_doc_no, need_date_begin, need_date_end,
          need_disb_who, is_pass, pdt_code, is_latin, can_load, doc_org_ref,
          drm_is_manual, drm_code, drm_api_code, drm_recog_2,
          drm_show_fields_staff, drm_photo_size, file_ext, drm_id
    do
    begin
      if (pdt_code containing '_phone_change') then
        continue;

      -- RFCKZ-505
      if (pdt_code = 'scholarship_statement' and
         (db_id <> 39801 or
          :c_age is null or
          :c_age >= 22
      )) then
        continue;

      -- Запрещенные к показу в МП РФ
      if (pdt_code in ('credit_history', 'fssp', 'certificate_name_change', 'selfie_photo') and
          db_id = 81001) then
        continue;
      doc_org = null;
      doc_org_code = null;

      if (db_id in (81001, 41701)) then
      begin
        -- Для РФ, КГ запрещаем закрепление паспорта, если клиент новый
        if (coalesce(is_pass, 0) = 1 and is_new_client = 1) then
          can_load = 0;
      end
      else
      begin
        -- Запрещаем возможность выбора из галлереи если это паспорт в константе не разрешено
        if (coalesce(is_pass, 0) = 1) then
          can_load = 0;
      end

      if (can_load = 0 and attach_gallery_access = 1) then -- Если есть доступ, разрешаем прикрепление из галереи телефона
        can_load = 1;

      if (db_id in (39801, 41701) and pdt_code in ('foto_on_date_disb','passport_card')) then
        can_load = 0;

      err_text = '';
      if (is_employee = 1) then
        err_text = 'Редактирование документа сотрудника запрещено.';
      if (has_miniapp = 0 and
          is_active_client = 1 and
          is_pass = 1) then
        err_text = 'Редактирование паспорта активного клиента запрещено.';
      if (has_err_oivo = 1 and err_text = '') then
        err_text = 'Добавление документа запрещено, клиент не закреплен за вами.';

      select drm_has_page1, drm_has_page2
      from hp_get_drm_has_pages(:id)
      into :need_photo1, :need_photo2;

      memo = '';
      select pk_name || ': <b>сохранено ранее</b><br>' ||
        'Номер: <b>' || coalesce(cds_doc_no, '-') || '</b><br>' ||
        'Дата выдачи: <b>' || f_datetostr(cds_doc_date_begin) || '</b><br>' ||
        'Действует до: <b>' || f_datetostr(cds_doc_date_end) || '</b><br>' ||
        'Стр.1: <b>' || iif(coalesce(cds_doc_blob_1, 0) <> 0, 'Есть скан', 'Нет скана') || '</b>, ' ||
        'Стр.2: <b>' || iif(coalesce(cds_doc_blob_2, 0) <> 0, 'Есть скан', 'Нет скана') || '</b><br>' ||
        'Сохранен: <b>' || f_datetimetostr(cds_datetime) || '</b><br>',
        cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_is_main_doc,
        cds_doc_issued_by
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = :id and
            cds_is_delete = 0 and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id
      order by cds_create_datetime desc
      rows 1
      into memo, doc_no, date_begin, date_end, is_main, doc_org;

      if (id = c_doc_type) then
        doc_org_code = c_pass_who_code;

      if (memo = '') then
      begin
        select pk_name || ': <b>ранее не сохранено</b>'
        from passkind
        where pk_id = :id
        into memo;
        doc_no = null;
        date_begin = null;
        date_end = null;
        is_main = null;
      end
      if (need_photo1 = 0 and
          need_photo2 = 0) then
        memo = memo || '<br>Фото документа не требуется';
      memo = '<b>' || c_name_fio || '</b><br>' || memo;

      if (coalesce(is_main, 0) = 1 and
          coalesce(c_verify_status, 0) = 2) then
      begin
        -- Документ верифицирован
        memo = memo || '<font color="red"><b>Редактирование запрещено,</b> так как паспорт успешно верифицирован.</font><br>';
        memo = memo || 'Можно редактировать только если клиент предоставил новый документ.';
      end

      -- Распознаваться могут только JPG-файлы с камеры
      if (file_ext is distinct from 'jpg') then
        drm_is_manual = 1;

      -- RFCKZ-217
      -- Если в документе нет полей, подменяем drm_show_fields_staff = 0 для МП
      -- Сделано для того, чтобы не показывать пустой экран редактирования полей в МП
      -- Связанные изменения есть FM_DOCRECOG_ACTION
      if (drm_show_fields_staff = 1 and (select count(*) from docrecogfield where drf_from_drm_id = :drm_id and drf_inactive = 0) = 0) then
          drm_show_fields_staff = 0;

      if (db_id = 39801 and id = 300) then
        can_load = 0;

      if (drm_code is not null) then
      begin
        suspend;
        drm_code = null;
        drm_recog_2 = null;
      end

      if (pdt_code <> 'transport' or db_id <> 81001) then
        if (is_new_clientdocs = 0) then
        suspend;
    end
  end
  else
  begin
    is_employee = 0;
    is_active_client = 0;
    err_text = '';
    doc_org = null;
    doc_org_code = null;
    memo = '';
    doc_no = '';
    date_begin = null;
    date_end = null;
    is_main = 1;

    for select
      pk_id, pk_name, pdt_need_doc_no, pdt_need_date_begin, pdt_need_date_end,
      pdt_need_disb_who, pdt_is_main_doc, pk_is_latin, coalesce(pk_can_load, 0), '',
      drm_is_manual, drm_code, drm_api_code, drm_recog_2, drm_show_fields_staff,
      drm_photo_size, pk_file_ext
    from passkind
    inner join passdoctype on (pk_from_pdt_id = pdt_id)
    inner join docrecogmain on (pk_from_drm_id = drm_id and (drm_is_manual = 0 or (:db_id = 81001 and drm_id = 196)))
    where pdt_is_main_doc = 1
      and pk_is_delete = 0
      and pk_id > 0
    order by pk_country, pk_name, pk_order, pk_id
    into :id, :title, need_doc_no, need_date_begin, need_date_end, need_disb_who,
      is_pass, is_latin, can_load, doc_org_ref,
      drm_is_manual, drm_code, drm_api_code, drm_recog_2, drm_show_fields_staff,
      drm_photo_size, file_ext
    do
    begin
      can_load = 0;
      err_text = '';

      select drm_has_page1, drm_has_page2
      from hp_get_drm_has_pages(:id)
      into :need_photo1, :need_photo2;

        -- Распознаваться могут только JPG-файлы с камеры
        if (file_ext is distinct from 'jpg') then
          drm_is_manual = 1;

      suspend;
    end
  end
end
