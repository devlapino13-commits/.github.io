CREATE PROCEDURE FM_ANKETA_ADD(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    RD_ID VARCHAR(100),
    UN_ID DECIMAL(18,0))
RETURNS (
    ACL_ID DECIMAL(18,0),
    RES_TEXT D_STR_2048)
AS
declare variable REG_ID integer;
declare variable CM_DATABASE_ID integer;
declare variable OBJECT_KIND integer;
declare variable ACL_DATETIME timestamp;
declare variable ACL_STATUS integer;
declare variable C_ID decimal(18,0);
declare variable ACL_RESPONS_C_ID decimal(18,0);
begin
-- Добавляет объект в список действий (ActionList)
-- Здесь можно произвезти проверку готовности, например утвержден ли кредит

  res_text = '';
-- если ошибка (или условия не позволяют создать объект, res_text будет возвращать текст

  cm_database_id = f_get_db_id();

  object_kind = f_objkind(obj_kind, 'id');

  if (rd_id not similar to '[-]?[[:DIGIT:]]+') then
  begin
    rd_id = (select rd_id
             from refdata
             where rd_code = :rd_id);
  end
  reg_id = null;

  select acl_id, acl_datetime, acl_status, acl_respons_c_id
  from actionlist
  where acl_object_kind = :object_kind and
        acl_object_id = :obj_id and
        acl_from_rd_id = :rd_id
  order by acl_id desc
  rows 1
  into :acl_id, :acl_datetime, acl_status, acl_respons_c_id;

  select un_from_c_id, un_curr_reg_id
  from username
  where un_id = :un_id rows 1
  into :c_id, reg_id;

  select ps_from_reg_id
  from hr_pstatus
  where ps_from_c_id = :c_id and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_status = 1 and
        ps_is_working = 1
  into :reg_id;

  if (reg_id is null) then
  begin
    acl_id = 0;
    res_text = 'Офис пользователя не заполнен';
    suspend;
    exit;
  end

-- Если существует еще не завершенная анкета
-- или завершенная в течение трех последних дней, то возвращаем ей
-- статус в ожидании
  if (acl_id is not null and
      (acl_status in (0, 1) or (coalesce(acl_respons_c_id, 0) = c_id and
      datediff(day, acl_datetime, current_date) <= 3))) then
  begin
    update actionlist
    set acl_status = 0,
        acl_from_un_id = :un_id,
        acl_respons_c_id = :c_id
    where acl_id = :acl_id;
    suspend;
    exit;
  end

  insert into actionlist (acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status, acl_respons_c_id,
                          acl_from_un_id)
  values (:reg_id, :object_kind, :obj_id, :rd_id, 0, :c_id, :un_id)
  returning acl_id
  into :acl_id;

  suspend;
end
