CREATE PROCEDURE HP_DOCRECOGREQUEST_SET_STATUS(
    DRR_ID D_INTEGER,
    NEW_STATUS D_SMALLINT,
    REASON_RD_ID D_INTEGER = null)
AS
begin
  -- NEW_STATUS:
  --   0 - Новый запрос
  --   1 - Прикреплен
  --   2 - Сохранен (обработан сотрудником)
  --  -1 - Переснять
  --  -2 - Аннулирован

  update docrecogrequest
  set drr_status = :new_status,
      drr_reason_rd_id = :reason_rd_id
  where drr_id = :drr_id;

  if (new_status = -1) then
   execute procedure hp_docrecogrequest_notify(drr_id, 'on_rework');

  if (new_status = 1) then
   execute procedure hp_docrecogrequest_notify(drr_id, 'on_attach');
end
