CREATE PROCEDURE HP_LO_IS_PARTNER(
    REP_DATE DATE,
    IS_CALC SMALLINT,
    IS_INSERT SMALLINT)
RETURNS (
    SVZ_ID ID_TYPE,
    SVZ_BALANCE_PAR_7 CURR_TYPE,
    SVZ_BALANCE CURR_TYPE,
    PAR7 PERC_TYPE,
    SVZ_CL_QTY INTEGER,
    C_NAME_FIO STR_100,
    LO_TYPE SMALLINT,
    LO_TYPE_STR STR_100)
AS
declare variable k_id id_type;
declare variable oks_balance curr_type;
declare variable k_disb_date date;
declare variable oks_delinq_date date;
declare variable cl_qty integer;
declare variable rd_parent_id integer;
declare variable rd_id integer;
declare variable un_id id_type;
declare variable svz_major_prior integer;
declare variable svz_partner_prior integer;
declare variable cur_svz_major integer;
declare variable cur_svz_partner integer;
begin
  /*

  returns LO_TYPE smallint (1 - Ведущий, 2 - Партнер)

  Процедура работает в 3-ех режимах в зависимости от входящих параметров

  1) (IS_CALC = 1) and (is_insert = 0)
     Вычисляет по алгоритму, СВЗ-Партнеров на дату.
     Процедура работает только выборку данных, и может быть использована для отчета

  2)  (IS_CALC = 1) and (is_insert = 1)
      a) Вычисляет по алгоритму, СВЗ-Партнеров на дату. Посредством рекурсии процедура вызывает себя с пераметрами из пункта (1)
      б) Извлекет список действующих СВЗ-партнеров на дату.
      в) производит соотношение:
            если ранее действующий СВЗ-Партнер  не сохранил свой статус, аннулируем его статус
            если ранее СВЗ не имел статуса СВЗ-партнер, статус СВЗ-партнер присваивается ему

      В данном режиме процедура работает на изменение данных.
      В данном режиме процедура запускается раз в месяц, чтобы обновить список действующих СВЗ-Партнеров.

  3)  (IS_CALC = 0) and (is_insert = 0)
      Извлекает список всех актуальных на дату СВЗ-Партнеров, по уже готовым значениям в objectProps
      + к списку добавляет виртуального сотрудника СВЗ-Сирена.

      В данном режиме процедура работает для finlite.

  */

  is_calc = coalesce(is_calc, 0);
  is_insert = coalesce(is_insert, 0);

  rd_id = null;

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ObjectProps.LO_Partner'
  into :rd_id;

  if ((is_calc = 1) and
      (is_insert = 0)) then
  begin
-- Опеределяю на данный момент всех действующих СВЗ мега-офисов
    for select distinct hps.ps_from_c_id
        from hr_pstatus hps
        inner join joblist jl on jl.jl_id = ps_from_jl_id
        inner join jobkind jk on jk.jk_id = jl.jl_kind
        inner join region reg on reg.reg_unicode = hps.ps_from_reg_id
        where :rep_date between ps_date_begin and ps_date_end and
              ps_is_working = 1 and
              jk.jk_system_code = 'JobSystem.LO_Standart'
--and reg_logic_label containing ('.mega_office.')  --  igusev закомментировал 02.09.2015 т.к. теперь анализу подлежат СВЗ всех офисов, не только Мега
              and
              reg_status = 3 and
              reg_date_close is null
        into :svz_id
    do
    begin
      svz_cl_qty = 0;
      svz_balance = 0;
      svz_balance_par_7 = 0;
      par7 = 0;

-- Определяю баланс СВЗ, баланс ПАР7, число клиентов

      for select oks_from_k_id, oks_balance, k_disb_date, oks_delinq_date
          from oksummary oks
          inner join kontract kont on kont.k_id = oks.oks_from_k_id
          where :rep_date between oks_date_begin and oks_date_end and
                oks_balance > 0 and
                oks_close_state = 0 and
                oks_from_k_id = k_id and
                oks.oks_kc_id = :svz_id -- <<
          into :k_id, :oks_balance, :k_disb_date, :oks_delinq_date
      do
      begin
        cl_qty = null;

        select res_just_client_count
        from hp_get_just_client_in_loan(:rep_date, :k_id, :k_disb_date)
        into cl_qty;

        cl_qty = coalesce(cl_qty, 0);
        svz_cl_qty = svz_cl_qty + cl_qty;
        svz_balance = svz_balance + oks_balance;

        if (:rep_date - oks_delinq_date > 7) then
        begin
          svz_balance_par_7 = svz_balance_par_7 + oks_balance;
        end
      end

      svz_balance_par_7 = coalesce(svz_balance_par_7, 0);

      if (svz_balance > 0) then
      begin
        par7 = svz_balance_par_7 / svz_balance * 100;

-- если ПАР7 < 7% и число клиентов более или равно 230, то это СВЗ-Партнер

        lo_type = 0;
        if ((par7 < 7) and
            (svz_cl_qty >= 230)) then
        begin
          lo_type = 2;
          lo_type_str = 'СВЗ-партнер';
        end
        if ((par7 < 7) and
            (svz_cl_qty between 150 and 229)) then
        begin
          lo_type = 1;
          lo_type_str = 'Ведущий СВЗ';
        end
        if (lo_type <> 0) then
        begin
          select cl.c_name_fio
          from clients cl
          where cl.c_id = :svz_id
          into :c_name_fio;

          suspend;
        end
      end
    end
  end

  if ((is_calc = 1) and
      (is_insert = 1)) then
  begin
    un_id = (select un_id
             from username un
             where un.un_name = 'IGUSEV');

    if (rd_id is not null) then
    begin
      for select svz_id, sum(svz_major_prior), sum(svz_partner_prior), sum(cur_svz_major), sum(cur_svz_partner)
          from (
-- извлекаю список СВЗ-партнеров из ObjectProps, передыдущего месяца
          select result_local_id as svz_id, iif(result_value = 1, 1, 0) as svz_major_prior,
                 iif(result_value = 2, 1, 0) as svz_partner_prior, 0 as cur_svz_major, 0 as cur_svz_partner
          from hp_objectprops_get_list('Subject', 'Refdata', :rd_id, 'Clients', :rep_date)
          where result_value in (1, 2) -- << Важно
          union all
-- вычисляю актульаных СВЗ - партнеров на текущий месяц
          select svz_id, 0, 0, iif(lo_type = 1, 1, 0), iif(lo_type = 2, 1, 0)
          from hp_lo_is_partner(:rep_date, 1, 0)) a
          group by svz_id
          into :svz_id, :svz_major_prior, :svz_partner_prior, :cur_svz_major, :cur_svz_partner
      do
      begin
        svz_major_prior = coalesce(svz_major_prior, 0);
        svz_partner_prior = coalesce(svz_partner_prior, 0);
        cur_svz_major = coalesce(cur_svz_major, 0);
        cur_svz_partner = coalesce(cur_svz_partner, 0);

        if ((svz_major_prior = 0) and
            (cur_svz_major = 1)) then -- если в предыдущем месяце СВЗ НЕ был присвоен статус "Ведущий СВЗ", а в текущем месяце СВЗ получает статус "Ведущий СВЗ", тогда присваиваю статус
          execute procedure hp_objectprops_set('Clients', :svz_id, 'RefData', 'ObjectProps.LO_Partner', :rep_date, 1, 1,
              :un_id);
        if ((svz_partner_prior = 0) and
            (cur_svz_partner = 1)) then -- если в предыдущем месяце СВЗ НЕ был присвоен статус "СВЗ - партнер", а в текущем месяце СВЗ получает статус "СВЗ - партнер", тогда присваиваю статус
          execute procedure hp_objectprops_set('Clients', :svz_id, 'RefData', 'ObjectProps.LO_Partner', :rep_date, 2, 2,
              :un_id);
        if (((svz_major_prior = 1) or (svz_partner_prior = 1)) and
            (cur_svz_major = 0) and
            (cur_svz_partner = 0)) then -- если в предыдущем месяце СВЗ БЫЛ присвоен статус "СВЗ-Партнер или Ведущий СВЗ", а в текущем месяце СВЗ НЕ получает статус, тогда аннулирую статус для СВЗ
          execute procedure hp_objectprops_set('Clients', :svz_id, 'RefData', 'ObjectProps.LO_Partner', :rep_date, 0, 0,
              :un_id);
        suspend;
      end
    end
  end

  if ((is_calc = 0) and
      (is_insert = 0)) then
  begin
    if (rd_id is not null) then
    begin
-- извлекаю список СВЗ-Партнеров для finlite

-- Сирена
      for select 1000000000004
          from rdb$database
          union all
-- СВЗ - партнеры
          select result_local_id as svz_id
          from hp_objectprops_get_list('Subject', 'Refdata', :rd_id, 'Clients', :rep_date)
          inner join hr_pstatus hps on hps.ps_from_c_id = result_local_id
          inner join region reg on reg.reg_unicode = hps.ps_from_reg_id
          where result_value = 2 -- << Важно
                and
                :rep_date between ps_date_begin and ps_date_end and
                ps_is_working = 1 and
                reg_logic_label containing ('.mega_office.')
          into :svz_id
      do
      begin
        suspend;
      end
    end
  end
end
