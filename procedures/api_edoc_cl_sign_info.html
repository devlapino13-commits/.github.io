CREATE PROCEDURE API_EDOC_CL_SIGN_INFO(
    EDOC_ID D_ID,
    EDOC_SIGNER D_STR_CODE)
RETURNS (
    RES_COMPANY_NAME D_STR_255,
    RES_TITLE D_STR_255,
    RES_NAME D_STR_255,
    RES_INN D_STR_255,
    RES_SIGN_DATETIME D_STR_CODE,
    RES_SIGN_GUID D_STR_CODE)
AS
declare variable edoc_signer_guid d_str_code;
declare variable need_exit d_bool;
declare variable c_id d_id;
begin
  -- edoc_id - это EDS_ID

  res_company_name = '';
  res_title = 'Неверная ссылка, документ не найден';
  res_name = '';
  res_inn = '';
  res_sign_datetime = null;
  res_sign_guid = '';

  need_exit = 0;
  if (coalesce(edoc_id, 0) = 0 or coalesce(edoc_signer, '') = '') then
    need_exit = 1;
  else
  begin
    edoc_signer_guid = f_edoc_cl_signer(edoc_id);
    if (edoc_signer <> edoc_signer_guid) then
      need_exit = 1;
  end

  if (need_exit = 0) then
  begin
    select eds_from_c_id, f_datetimetostr(eds_status_datetime)
    from edoc_sign
    where eds_id = :edoc_id and
          eds_status = 1
    into c_id, res_sign_datetime;

    if (row_count = 0) then
      need_exit = 1;
  end

  if (need_exit = 0) then
  begin
    select cm_name
    from company
    into res_company_name;

    res_title = 'Расшифровка электронной подписи';
    res_sign_guid = lower(edoc_signer);

    select esi_name, esi_inn
    from edoc_sinfo
    where esi_from_eds_id = :edoc_id
    into res_name, res_inn;

    if (row_count = 0) then
    begin
      select c_name_fio, c_inn
      from clients
      where c_id = :c_id
      into res_name, res_inn;
    end
  end

  suspend;
end
