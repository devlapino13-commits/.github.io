CREATE PROCEDURE HP_EXPRESSAPP_DO_DISBURSEMENT(
    EXA_ID D_ID)
RETURNS (
    WAIT_STATUS D_SMALLINT)
AS
declare variable MA_ID D_ID;
declare variable K_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable OL_ID_2 D_ID;
declare variable AMOUNT D_INTEGER;
declare variable KOMMIS_AMOUNT D_INTEGER;
declare variable RES_TEXT D_BLOB_TEXT;
declare variable ACC_DT_1 D_STR_255;
declare variable ACC_KT_1 D_STR_255;
declare variable ACC_DT_2 D_STR_255;
declare variable ACC_KT_2 D_STR_255;
declare variable DISCOUNT D_INTEGER;
declare variable DISB_OL_ID D_ID;
declare variable KONTRACT_NO D_STR_20;
declare variable KAGENT_C_ID D_ID;
declare variable KAGENT_FIO D_STR_255;
declare variable PR_CODE D_STR_100;
declare variable TKT_ID D_ID;
declare variable K_DISB_DATE D_DATE;
declare variable AG_AGREEMENT_NO varchar(30);
declare variable DOC_NO D_INTEGER;
declare variable MP_ID D_ID;
declare variable UN_ID D_ID;
declare variable K_DISB_KIND D_INTEGER;
declare variable LO_ID D_ID;
declare variable PUSH_SUBJECT D_STR_255;
declare variable PUSH_BODY D_BLOB_TEXT;
declare variable C_NAME_FIO D_STR_100;
declare variable C_INN D_STR_20;
declare variable REP_DISB_KIND D_INTEGER;
declare variable REP_REPAY_KIND D_INTEGER;
declare variable K_FACT_DISB_DATE D_DATE;
declare variable K_AMMOUNT_APPROVED D_CURRENCY;
declare variable COMPANY_NAME D_STR_255;
declare variable EMAIL D_STR_100;
declare variable PHONE_NUMBER D_STR_100;
declare variable OA_IS_FIRED_LO D_SMALLINT;
declare variable MANAGER_ID D_ID;
declare variable EXP D_INTEGER;
declare variable AG_ID D_INTEGER;
begin
  wait_status = 0;
  res_text = '';

  select exa.exa_from_k_id, exa.exa_from_reg_id, exa.exa_from_c_id, exa.exa_from_ma_id, pr.pr_code, exa.exa_from_un_id
  from expressapp exa
  inner join product pr on pr.pr_id = exa.exa_from_pr_id
  where exa.exa_id = :exa_id
  into :k_id, :reg_id, :c_id, :ma_id, :pr_code, :un_id;

  --Если это не товары в рассрочку, завершаем работу.
  --Условие добавлено чтобы не ломать логику для обычного экспресс займа
  if (pr_code not containing '.installment_goods.' and
      pr_code not containing '.sugarmill.' and
      pr_code not containing '.online_loan.') then
  begin
    suspend;
    exit;
  end

  select
    kt.k_disb_kind,
    kt.k_kontract_no,
    kt.k_fact_disb_date,
    kt.k_amount_approved
  from kontract kt
  where kt.k_id = :k_id
  into
    k_disb_kind,
    kontract_no,
    k_fact_disb_date,
    k_ammount_approved;

  select acc_kt, acc_dt, res_text
  from hp_get_acc_online_disb(:k_disb_kind, :pr_code, :reg_id)
  into acc_kt_1, acc_dt_1, res_text;

  if (res_text <> '') then
    exception e_message res_text;


  if (pr_code containing '.sugarmill.') then
  begin
    select cs_discount, cs_from_c_id, c_name_fio
    from kontract
    inner join extminiapp on ema_from_tkt_id = k_from_tkt_id
    inner join extsysblob on esb_from_esw_id = ema_from_esw_id
    inner join clientshops on cs_id = f_json_str_get_value(esb_blob_txt, 'seller.shop_id', -1)
    inner join clients on c_id = cs_from_c_id
    where k_id = :k_id and
          cs_is_delete = 0
    into :discount, :kagent_c_id, :kagent_fio;
  end

  if (pr_code containing '.installment_goods.') then
  begin
    kagent_c_id = null;

    select
        cs.cs_discount,
        cs.cs_from_c_id,
        coalesce(cs.cs_name, ''),
        mp.mp_id
    from mc_purchase mp
    inner join clientshops cs on cs.cs_id = mp.mp_from_cs_id
    inner join clients cl on cs.cs_from_c_id = cl.c_id
    where mp.mp_from_ma_id = :ma_id
    into
        discount,
        kagent_c_id,
        company_name,
        mp_id;

    -- RFCKG-2581
    if(:kagent_c_id is not null and :kagent_c_id <> 1090101386758) then
    begin
        select
            c.c_id, 
            c.c_name_fio,
            c.c_inn
        from knclient kc
        inner join clients c on kc.kc_from_c_id = c.c_id
        where kc.kc_from_k_id = :k_id
            and kc.kc_amount_approved > 0
        into
            c_id,
            c_name_fio,
            c_inn;

        email = f_cont_info('clients', :c_id, 'email', 'last');
        phone_number = coalesce(f_cont_info('clients', :c_id, 'cell_main', 'last'), '');

        push_subject = 'Оформлена рассрочка ' || c_name_fio;
        push_body = blob_append(
            'По договору ОсОО/ИП ' || company_name || ' оформлена рассрочка:' || ascii_char(10),
            'Заемщик: ' || c_name_fio || ascii_char(10) ||
            'ИНН заемщика: ' || c_inn || ascii_char(10) ||
            '№ контракта: ' || kontract_no || ascii_char(10) ||
            'Дата контракта: ' || k_fact_disb_date || ascii_char(10) ||
            'Сумма кредита: ' || k_ammount_approved || ascii_char(10) ||
            'Номер телефона заемщика: ' || phone_number);

        execute procedure hp_send_quick_message('email', :email, :push_subject, :push_body, '', 1, 1000000000000);
    end
  end

  if (pr_code containing '.TMS.') then
  begin
    ag_id = null;
    select kml_from_ag_id from kontmat
    where kml_object_kind = f_objkind('kontract', 'id')
    and kml_object_id = :k_id
    order by kml_id desc
    rows 1
    into ag_id;
    if (ag_id is not null) then
    begin
      select ag_object_id from agreements
      where ag_id = :ag_id
      into kagent_c_id;
    end
  end

  discount = coalesce(discount, 0);

  if (discount <> 0) then
  begin
    --Если выдача на карту МБулак
    if (k_disb_kind = 6) then
      res_text = (select err_text
                  from hp_do_disbursement_card_mb(:k_id, :un_id));
    else
--      res_text = (select res_text
--                  from hp_do_disbursement(:un_id, :k_id, :reg_id, 3, :acc_kt_1, :reg_id, 1, :acc_dt_1, :c_id, 1000000000100));
      select res_text
      from hp_do_disbursement(:un_id, :k_id, :reg_id, :rep_disb_kind, :acc_kt_1, :reg_id, :rep_repay_kind, :acc_dt_1, :c_id, 1000000000100)
      into res_text;


    if (res_text <> '') then
    begin
      wait_status = -1;
      suspend;
      exit;
    end

    select ks.ks_from_ol_id, k.k_amount_approved, k.k_kontract_no, ks.ks_date_begin
    from kontstate ks
    inner join kontract k on k.k_id = ks.ks_from_k_id
    where ks.ks_state = 1 and
          ks.ks_ol_is_delete = 0 and
          ks.ks_from_k_id = :k_id
    into :disb_ol_id, :amount, :kontract_no, :k_disb_date;

    ag_agreement_no = '';
    select ag.ag_agreement_no
    from agreements ag
    where ag.ag_object_kind = f_objkind('clients', 'id') and
          ag.ag_object_id = :kagent_c_id and
          :k_disb_date between ag.ag_date_begin and coalesce(ag.ag_date_end, '31.12.9999') and
          ag.ag_from_rd_id = f_get_rd_id('AgreementKinds.installment_goods') and
          ag.ag_status = 1
    into ag_agreement_no;

    acc_dt_2 = '';

    acc_dt_2 = (select al_id
                from acc_list
                where al_from_reg_id = :reg_id and
                      al_currency = 417 and
                      al_account = '11531/99');

    if (coalesce(acc_dt_2, '') = '') then
      exception e_message 'счет дт не определен';

    acc_kt_2 = '';

    acc_kt_2 = (select al_id
                from acc_list
                where al_from_reg_id = :reg_id and
                      al_currency = 417 and
                      al_account = '21114/19'); -- поменять на 21114/19

    if (coalesce(acc_kt_2, '') = '') then
      exception e_message 'счет кт не определен';

    kommis_amount = amount * discount / 100;

    update operbook ob
    set ob.ob_client_dt = :c_id,
        ob.ob_client_kt = :kagent_c_id
    where ob.ob_from_ol_id = :disb_ol_id;

    select ol.ol_doc_no
    from operlist ol
    where ol.ol_id = :disb_ol_id
    into doc_no;

    insert into operlist (ol_id, ol_parent_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status,
                          is_delete, ol_user_id, ol_document_client, ol_doc_client_name, ol_doc_no)
    values (null, :disb_ol_id, :reg_id, 2, current_date, 9,
            'Комиссия к получению согл.договора №' || :ag_agreement_no || ' ' || :kagent_fio || ' кр.дог. №' || :kontract_no,
            0, 0, 1000000000000, :kagent_c_id, null, :doc_no)
    returning ol_id
    into :ol_id_2;

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                          ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                          ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
                          ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt)
    values (:ol_id_2, -1, '417', 1, :kagent_c_id, :acc_dt_2, :kommis_amount, :kommis_amount, null, null, null,
            :acc_kt_2, null,
            'Комиссия к получению согл.договора №' || :ag_agreement_no || ' ' || :kagent_fio || ' кр.дог. №' || :kontract_no,
            0, 0, null, current_date, 1, :reg_id, :acc_dt_2, :acc_kt_2);
  end
  else
  begin
    --Если выдача на карту МБулак
    if (k_disb_kind = 6) then
      res_text = (select err_text
                  from hp_do_disbursement_card_mb(:k_id, :un_id));
    else
    begin
      rep_disb_kind = iif(k_disb_kind = 5, 5, 3); --5 выдача на сторонние карты элькарт, 3 мемориальный ордер
      rep_repay_kind = iif(k_disb_kind = 5, 5, 1);

      res_text = (select res_text
                  from hp_do_disbursement(:un_id, :k_id, :reg_id, :rep_disb_kind, :acc_kt_1, :reg_id, :rep_repay_kind, :acc_dt_1, :c_id, 1000000000100));
    end

    if (res_text <> '') then
    begin
      wait_status = -1;
      suspend;
      exit;
    end

    if (pr_code containing '.installment_goods.' or
        pr_code containing '.sugarmill.') then
    begin
      select ks.ks_from_ol_id
      from kontstate ks
      inner join kontract kt on kt.k_id = ks.ks_from_k_id
      where ks.ks_from_k_id = :k_id and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0
      into :disb_ol_id;

      update operbook ob
      set ob.ob_client_dt = :c_id,
          ob.ob_client_kt = :kagent_c_id
      where ob.ob_from_ol_id = :disb_ol_id;
    end
  end

  if (pr_code containing '.sugarmill.') then
  begin
    select kt.k_from_tkt_id
    from kontract kt
    where kt.k_id = :k_id
    into tkt_id;

    update ticket tkt
    set tkt.tkt_status = 2
    where tkt.tkt_id = :tkt_id;
  end

  -- создаем задачу на оплату контрагенту по "товарам в рассрочку"
  if (pr_code containing '.installment_goods.') then
  begin
    update mc_purchase mp
    set mp.mp_status = f_enum_id('mc_purchase_status', 'success', null)
    where mp.mp_id = :mp_id;

    execute procedure hp_epaytask_create(:k_id);
  end

  if (pr_code containing '.online_loan.') then
  begin
    select oa.oa_last_lo_id, oa.oa_is_fired_lo
    from online_app oa
    where oa.oa_from_exa_id = :exa_id
    into lo_id, oa_is_fired_lo;

    oa_is_fired_lo = coalesce(oa_is_fired_lo, 0);

    if (coalesce(lo_id, 0) <> 0 and oa_is_fired_lo = 0) then
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into c_name_fio;

      push_subject = 'Выдача по онлайн кредиту';
      push_body = blob_append(c_name_fio, ', получил онлайн займ. Возьмите его на свой баланс');
      execute procedure hp_send_push_to_staff(:lo_id, :push_subject, :push_body, :un_id, null, null);
    end

    execute procedure hp_objectprops_set_with_rd('clients', :c_id, 'refdata', 'AnketaQuestions.descr_of_loan', current_date,
    'Удовлетворение потребительских нужд', 0, 0, :un_id, 'Удовлетворение потребительских нужд');

    if (oa_is_fired_lo = 1) then
    begin
      select ps.ps_from_c_id
      from hr_pstatus ps
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where ps.ps_from_reg_id = :reg_id and
            current_date between ps.ps_date_begin and ps.ps_date_end and
            ps.ps_status = 1 and
            ps.ps_is_working = 1 and
            jk.jk_system_code = 'jobsystem.Region_KM'
      into manager_id;

      --RFCKG-2560
      push_subject = 'Онлайн кредит';
      push_body = blob_append(c_name_fio, ', получил онлайн займ. Один из ваших ВИПСов может его забрать в свой портфель');
      execute procedure hp_send_push_to_staff(:manager_id, :push_subject, :push_body, :un_id, null, null);

      --push_body для випсов, push_subject остается тем же
      push_body = blob_append(c_name_fio, ', получил онлайн займ. Вы можете забрать в свой портфель');
      for select ps.ps_from_c_id
          from hr_pstatus ps
          inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
          inner join jobkind jk on jk.jk_id = jl.jl_kind
          where ps.ps_from_reg_id = :reg_id and
                current_date between ps.ps_date_begin and ps.ps_date_end and
                ps.ps_status = 1 and
                ps.ps_is_working = 1 and
                jk.jk_system_code in ('jobsystem.lo_standart', 'jobsystem.lo_senior')
          into lo_id
       do
       begin
         exp = 0;
         select cast(ovs_value as integer)
         from hp_objectprops_get('clients', :c_id, 'refdata', 'lo_pure_exp', current_date)
         into exp;

         if (exp not between 4 and 12) then
           continue;

         execute procedure hp_send_push_to_staff(:manager_id, :push_subject, :push_body, :un_id, null, null);
       end 
    end
  end 

  suspend;
end
