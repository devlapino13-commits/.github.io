CREATE PROCEDURE FM_GET_PERCENT(
    CURR_PR_ID INTEGER,
    CURR_STAUS INTEGER,
    DISB_DATE DATE,
    LOAN_AMOUNT CURR_TYPE,
    LOAN_REPAY INTEGER,
    ADD_PR_ID INTEGER = null)
RETURNS (
    PERC_DEFAULT PERC_TYPE)
AS
declare variable prl_percent_min perc_type;
declare variable prl_percent_max perc_type;
declare variable perc_min perc_type;
declare variable perc_max perc_type;
declare variable percent_lgota perc_type;
declare variable pr_code varchar(100);
declare variable pr_product_kind integer;
declare variable percent d_percent;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  perc_min = null;
  perc_max = null;

  select pr_code, pr_product_kind
  from product
  where pr_id = :curr_pr_id
  into pr_code, pr_product_kind;

  if (disb_date < '01.11.2020') then
  begin
    percent_lgota = 0;
    if (pr_code containing 'barakat' or pr_code containing 'family' or pr_product_kind = 2 or pr_code containing 'blago' or pr_code containing 'uspeh' or
      pr_code containing '.zalog_car.' or pr_code containing '.education.' or pr_code containing '.food.') then
    begin
      percent_lgota = case curr_staus
                        when 5 then 1.00
                        when 4 then 0.75
                        when 3 then 0.75
                        when 2 then 0.50
                        when 1 then 0.25
                        else 0
                      end;
      if (pr_code containing '.food.') then
      begin
        if (loan_repay < 10) then
          perc_max = 9.4;
        else
          perc_max = 9.9;
      end
      else
      if (pr_code containing '.barakat.') then
      begin
        if (loan_repay <= 6) then
          perc_max = 8.9;
        else
          perc_max = 9.4;
      end
      else
      if (pr_code containing '.blago.') then
      begin
        if (loan_repay < 10) then
          perc_max = 9.4;
        else
          perc_max = 9.9;
      end
      else
      if (pr_code containing '.barakat plus.') then
      begin
        if (loan_repay <= 6) then
          perc_max = 8.2;
        else
          perc_max = 8.7;
      end
      else
      if (pr_code containing '.uspeh.') then
      begin
        if (loan_repay < 10) then
          perc_max = 8.7;
        else
          perc_max = 9.2;
      end
      else
      if (pr_code containing '.family.') then
      begin
        if (loan_repay <= 6) then
        begin
          perc_max = 7.5;
          --        kommis_perc_max = 0.0;
        end
        else
        begin
          perc_max = 8.0;
          --        kommis_perc_max = 1.0;
        end
      end
      else
      if (pr_code containing '.family plus.') then
      begin
        if (loan_repay <= 6) then
        begin
          perc_max = 8.5;
          --        kommis_perc_max = 1.0;
        end
        else
        begin
          perc_max = 9.0;
          --        kommis_perc_max = 1.5;
        end
      end
      else
      if (pr_code containing '.zalog_car.') then
      begin
        perc_max = 4.0;
        --        kommis_perc_max = 0;

        if (curr_staus >= 2) then
          percent_lgota = 0.4;
        else
        if (curr_staus = 1) then
          percent_lgota = 0.25;
        else
          percent_lgota = 0;
      end
      else
      if (pr_code containing '.education.') then
      begin
        perc_max = 3.8;

        if (curr_staus = 1) then
          percent_lgota = 0.05;
        else
        if (curr_staus >= 2) then
          percent_lgota = 0.2;
        else
          percent_lgota = 0;
      end
      perc_default = perc_max - percent_lgota;
      --    if (perc_min<perc_default)
      --    then perc_min=perc_default;
      --    else perc_default=perc_min;
      --    if (perc_max>perc_default) then perc_max=perc_default;
    end
  end
  else
  if (disb_date >= '01.11.2020') then
  begin
    if ((pr_product_kind = 2 and pr_code containing  '.LOYAL.') or pr_code containing '.zalog_car.'  or pr_code containing '.reliable_client.' or
         pr_code containing '.reliable_client_plus.' or pr_code containing '.zalog.' or pr_code containing '.selfemployed.' or
         pr_code containing '.business.' or pr_code containing '.refinance_plus.' or pr_code containing '.selfemployed_plus.') then
    begin
      if (exists(select prl_percent_min, prl_percent_max
                 from aarep_product_logic_list(:curr_pr_id, :disb_date, -1, -1, :loan_amount, -1, :loan_repay, -1, -1, -1, -1, -1, -1))) then
      begin

        percent = null;
        select percent
        from perc_amount_interval_by_point(:curr_pr_id, :curr_staus)
        into :percent;

        perc_max = percent;
        perc_default = perc_max;
      end
    end
  end

  if (db_id in (39801, 41701)) then --KG
  begin

        percent = null;
        select percent
        from perc_amount_interval_by_point(:curr_pr_id, :curr_staus)
        into :percent;

        perc_max = percent;
        perc_default = perc_max;
  end


  if (perc_max is null) then
  begin
    for select prl_percent_min, prl_percent_max
        from aarep_product_logic_list(:curr_pr_id, :disb_date, -1, -1, :loan_amount, -1, :loan_repay, -1, -1, -1, -1, -1, -1)
        into :prl_percent_min, prl_percent_max
    do
    begin
      if (perc_min is null) then
        perc_min = prl_percent_min;
      if (perc_max is null) then
        perc_max = prl_percent_max;
      if (perc_min > prl_percent_min) then
        perc_min = prl_percent_min;
      if (perc_max < prl_percent_max) then
        perc_max = prl_percent_max;
    end

    if (perc_max is null) then
    begin
      suspend;
      exit;
    end

    perc_default = perc_max;
  end

  suspend;
end
