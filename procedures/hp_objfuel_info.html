CREATE PROCEDURE HP_OBJFUEL_INFO(
    ID_STR D_STR_255)
RETURNS (
    PET_BILL_ID D_ID,
    PET_BILL_LINE_NO D_INTEGER,
    PET_NAME D_STR_255,
    PET_LABEL D_STR_CODE,
    PED_REGION_PLACE_ID D_INTEGER,
    PED_FROM_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    PED_DATE D_DATE)
AS
begin
  if (id_str is not null) then
  for
    select
      f_json_str_get_value(:id_str, 'bill_id', 0) as pet_bill_id,
      f_json_str_get_value(:id_str, 'bill_no', 0) as pet_bill_line_no,
      f_json_str_get_value(:id_str, 'invent_str', '') as pet_label,
      f_json_str_get_value(:id_str, 'reg_id', 0) as ped_region_place_id,
      f_json_str_get_value(:id_str, 'c_id', 0) as ped_from_c_id,
      f_json_str_get_value(:id_str, 'dt_oper', 0) as ped_date
    from rdb$database
  into
    :pet_bill_id,
    :pet_bill_line_no,
    :pet_label,
    :ped_region_place_id,
    :ped_from_c_id,
    :ped_date
  do
  begin
    pet_name = (
      select
        ml_name ||
          iif(ml_model in (ml_name, ''), '', ' ' || ml_model) as name
      from operbill
        inner join matlist
          on ml_id = ob_from_ml_id
      where ob_bill_id = :pet_bill_id
        and ob_bill_line_no = :pet_bill_line_no);
    c_name_fio = (select c_name_fio from clients where c_id = :ped_from_c_id);
    suspend;
  end
end
