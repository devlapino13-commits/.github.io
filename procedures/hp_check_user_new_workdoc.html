CREATE PROCEDURE HP_CHECK_USER_NEW_WORKDOC(
    UN_ID DECIMAL(18,0))
RETURNS (
    NEW_DOC_KOLICH INTEGER)
AS
declare variable c_id decimal(18,0);
declare variable jl_id decimal(18,0);
declare variable wdl_id integer;
declare variable un_access integer;
declare variable is_vip integer;
declare variable has_access integer;
begin
  new_doc_kolich = 0;
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;
  if (c_id is null) then
    c_id = 0;

  is_vip = 0;
  if (exists(select *
             from workdocvipusers
             where wdvip_from_c_id = :c_id)) then
    is_vip = 1;
  has_access = 0;
  select ps_from_jl_id
  from hr_pstatus
  where ps_from_c_id = :c_id and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_is_working = 1 rows 1
  into :jl_id;
  if (jl_id is null) then
    jl_id = 0;
  if (jl_id <> 0) then
    has_access = 1;
  if (has_access = 0) then
  begin
    if (exists(select *
               from servicedog
               where sd_from_c_id = :c_id and
                     sd_date_begin <= current_date and
                     sd_date_end >= current_date)) then
      has_access = 1;
    if ((has_access = 0) and
        (c_id in (1090100224546, 1010100105262, 1090100532672))) then
      has_access = 1;
  end
  if (has_access = 1) then
  begin
    for with workdoc_user
        as (select distinct wdc_from_wdl_id as wdl_id, 0 as un_access
            from workdoccl
            where wdc_from_c_id = :c_id
            union
            select distinct wdu_from_wdl_id as wdl_id, 0 as un_access
            from workdocjl
            where wdu_from_jl_id = :jl_id
            union
            select distinct wdl_id as wdl_id, 0 as un_access
            from workdoclist
            where (wdl_is_open_access = 1 or
                  :is_vip = 1)
            union
            select distinct wds_from_wdl_id as wdl_id, 1 as un_access
            from workdocstat
            where wds_from_un_id = :un_id)
        select wdl_id, max(un_access)
        from workdoc_user
        group by wdl_id
        having max(un_access) = 0
        into :wdl_id, :un_access
    do
    begin
      new_doc_kolich = new_doc_kolich + 1;
    end
  end
  suspend;
end
