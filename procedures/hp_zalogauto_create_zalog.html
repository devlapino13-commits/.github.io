CREATE PROCEDURE HP_ZALOGAUTO_CREATE_ZALOG(
    ZA_ID D_INTEGER)
AS
declare variable k_id d_id;
declare variable za_from_c_id d_id;
declare variable k_disb_date d_date;
declare variable zap_id d_integer;
declare variable zap_status d_smallint;
declare variable z_id d_integer;
begin
    select first 1
    za_from_k_id,
    za_from_c_id,
    zal_from_zap_id,
    zap_status,
    za_from_z_id
  from zalogauto
    inner join zalogautoprof
      on zap_from_za_id = za_id
    inner join zalogautolist
      on zal_from_zap_id = zap_id
  where za_id = :za_id
  order by zad_datetime
  into
    :k_id,
    :za_from_c_id,
    :zap_id,
    :zap_status,
    :z_id;

  if (z_id is not null) then
    exit;

  if (zap_status <> 2) then
    exit;

  if (exists(select * from zalog
  where z_from_k_id = :k_id)) then
    Exit;

  select k_disb_date
  from kontract
  where k_id = :k_id
  into k_disb_date;

  insert into zalog (z_from_k_id, z_from_c_id, z_type, z_subtype, z_name, z_document, z_cost_market,
                     z_cost_likvid, z_cost_kontract, z_memo1, z_memo2, z_registry_no, z_registry_date, z_registry_agency, z_date_begin)
  select :k_id, :za_from_c_id, 1, 4, z_name, z_document, 0, 0, 0,
         z_memo1, z_memo2, z_registry_no, z_registry_date, z_registry_agency,
         :k_disb_date
  from hp_zalogauto_fields(:zap_id)
  returning z_id
  into z_id;

  update zalogauto
  set za_from_z_id = :z_id
  where za_id = :za_id;
end
