CREATE PROCEDURE HP_OPERVAC_OPERS(
    REP_K_ID D_ID)
RETURNS (
    RES_K_ID D_ID,
    RES_OL_ID D_ID,
    RES_OL_DATE D_DATE,
    RES_DOC_NO D_STR_20,
    RES_AMOUNT D_CURRENCY,
    RES_MEMO D_STR_LONG,
    RES_DATE_BEGIN D_DATE,
    RES_DATE_END D_DATE,
    RES_MAX_PART_LIMIT D_CURRENCY)
AS
declare variable ov_vac_days d_integer;
declare variable ov_vac_begin_days d_integer;
declare variable ov_max_part_limit d_currency;
declare variable is_vac d_bool;
begin
  res_k_id = rep_k_id;
  res_max_part_limit = 0;
  is_vac = 0;

  for select distinct ok_ol_date, ok_from_ol_id, ol_doc_no_str, ol_memo
  from operkred, operlist
  where ok_from_k_id = :rep_k_id and
        ok_ol_is_active = 1 and
        ok_from_ol_id = ol_id
  order by 1, 2
  into res_ol_date, res_ol_id, res_doc_no, res_memo
  do
  begin
    res_amount = 0;
    select sum(ok_os + ok_perc + ok_fine)
    from operkred
    where ok_from_k_id = :res_k_id and
          ok_from_ol_id = :res_ol_id
    into res_amount;

    if (res_amount is null) then
      res_amount = 0;

    ov_vac_days = null;
    ov_vac_begin_days = 0;
    ov_max_part_limit = 0;

    select sum(ov_vac_days), sum(ov_vac_begin_days), sum(ov_max_part_limit)
    from opervac
    where ov_from_k_id = :rep_k_id and
          ov_from_ol_id = :res_ol_id
    into ov_vac_days, ov_vac_begin_days, ov_max_part_limit;

    if (ov_vac_days is not null) then
      is_vac = 1;

    if (ov_vac_days is null) then
      ov_vac_days = 0;

    if (ov_vac_begin_days is null) then
      ov_vac_begin_days = 0;

    if (ov_max_part_limit is null) then
      ov_max_part_limit = 0;

    res_max_part_limit = res_max_part_limit + ov_max_part_limit;

    if (is_vac = 0) then
    begin
      res_date_begin = null;
      res_date_end = null;
    end else
    begin
      if (res_date_begin is null) then
      begin
        res_date_begin = res_ol_date;
        res_date_begin = res_date_begin + ov_vac_begin_days;
      end
  
      if (res_date_end is null) then
        res_date_end = res_date_begin;
  
      res_date_end = res_date_end + ov_vac_days;
      end

    suspend;
  end
end
