CREATE PROCEDURE HP_CLIENTSHOPS_TERMVIEW(
    ID D_ID NOT NULL,
    TTYPE D_SMALLINT NOT NULL)
RETURNS (
    ITO_ID D_INTEGER,
    TRM_ID D_INTEGER,
    TRM_TYPE D_SMALLINT,
    TRM_IPC_ID D_STR_20,
    TRM_FROM_ML_ID D_ID,
    TRM_INVENT_STR D_STR_CODE,
    TRM_SERIAL_NO D_STR_255,
    TRM_SIMCARD_NO D_STR_20,
    ML_NAME D_STR_255,
    TRM_FROM_OBJECT_KIND D_INTEGER,
    TRM_FROM_OBJECT_ID D_ID,
    TRM_DATETIME D_DATETIME,
    TRM_FROM_UN_ID D_ID,
    UN_FIO D_STR_100,
    TRM_DATE_BEGIN D_DATE,
    TRM_DATE_END D_DATE,
    ITO_STATUS D_SMALLINT)
AS
declare variable obj_kind d_integer;
begin
  obj_kind = f_objkind(iif(ttype = 1, 'clientshops', 'region'), 'id');
  for
    select
      ito_id,
      trm_id,
      trm_type,
      trm_ipc_id,
      trm_from_ml_id,
      trm_invent_str,
      trm_serial_no,
      trm_simcard_no,
      ml_name,
      ito_object_kind,
      ito_object_id,
      ito_datetime,
      ito_from_un_id,
      un_fio,
      ito_date_begin,
      iif(ito_date_end = '31.12.9999', null, ito_date_end),
      ito_status
    from ipc_terminal_ownership
      inner join ipc_terminals
        on trm_id = ito_trm_id
          and trm_actual = 1
      inner join matlist
        on ml_id = trm_from_ml_id
      inner join username
        on un_id = ito_from_un_id
    where ito_object_kind = :obj_kind
      and ito_object_id = :id
  into
    :ito_id,
    :trm_id,
    :trm_type,
    :trm_ipc_id,
    :trm_from_ml_id,
    :trm_invent_str,
    :trm_serial_no,
    :trm_simcard_no,
    :ml_name,
    :trm_from_object_kind,
    :trm_from_object_id,
    :trm_datetime,
    :trm_from_un_id,
    :un_fio,
    :trm_date_begin,
    :trm_date_end,
    :ito_status
  do
  begin
    suspend;
  end
end
