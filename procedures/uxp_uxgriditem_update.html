CREATE PROCEDURE UXP$UXGRIDITEM_UPDATE(
    GRID_UUID_CODE D_STR_CODE,
    ITEM_POSITION D_INTEGER,
    ITEM_FIELDNAME D_STR_CODE,
    ITEM_CAPTION D_STR_100,
    ITEM_HINT D_STR_LONG,
    ITEM_SUMMARY D_SMALLINT)
AS
declare variable grid_uuid d_uuid;
declare variable uf_uuid d_uuid;
begin
  if (coalesce(item_fieldname, '') = '') then
    exit;

  grid_uuid = char_to_uuid(grid_uuid_code);

  if (not exists(select * from ux$grid
                 where ugr_uuid = :grid_uuid)) then
  begin
    select ugr_uuid
    from ux$grid
    where ugr_ufr_uuid = :grid_uuid
    order by ugr_datetime desc
    rows 1
    into grid_uuid;

    if (row_count = 0) then
    begin
      select ugr_uuid
      from ux$frame, ux$grid
      where ufr_owner = :grid_uuid and
            ugr_ufr_uuid = ufr_uuid
      order by ugr_datetime desc
      rows 1
      into grid_uuid;

      if (row_count = 0) then
        exception e_message 'Grid not found: ' || grid_uuid_code;
    end
  end

  if (coalesce(item_summary, 0) = 0) then
    item_summary = null;

  if (item_fieldname = 'delete others') then
  begin
    delete from ux$gritem
    where ugi_ugr_uuid = :grid_uuid and
          ugi_position >= :item_position
    order by ugi_position desc;

    exit;
  end

  select uf_uuid
  from ux$grid, ux$frame, ux$field
  where ugr_uuid = :grid_uuid and
        ufr_uuid = ugr_ufr_uuid and
        uf_owner = ufr_us_uuid and
        uf_field_name = trim(:item_fieldname)
  into uf_uuid;

  if (uf_uuid is null) then
    exception e_message 'Field not found: ' || item_fieldname;

  update ux$gritem
  set ugi_position = :item_position,
      ugi_caption = :item_caption,
      ugi_hint = :item_hint,
      ugi_summary_kind = coalesce(:item_summary, ugi_summary_kind)
  where ugi_ugr_uuid = :grid_uuid and
        ugi_uf_uuid = :uf_uuid;

  if (row_count = 0) then
  begin
    insert into ux$gritem (ugi_ugr_uuid, ugi_uf_uuid, ugi_position, ugi_caption, ugi_hint, ugi_summary_kind)
    values (:grid_uuid, :uf_uuid, :item_position, :item_caption, :item_hint, :item_summary);
  end
end
