CREATE PROCEDURE FM_HTML_PRINT_LIST(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    ADD_PARAMS VARCHAR(3000),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable res_obj_id integer;
declare variable json_key varchar(255);
declare variable json_value varchar(255);
declare variable show_manager integer;
declare variable blank_name varchar(255);
declare variable add_params_2 varchar(3000);
declare variable date_kind integer;
declare variable rd_name varchar(255);
declare variable uri varchar(2000);
declare variable db_id integer;
declare variable no_blank integer;
declare variable tmp_text blob sub_type 1 segment size 80;
declare variable tmp_is_ok integer;
declare variable lang d_str_lang;
begin
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  res_text = (select res_style from fm_html_styles) || '<body>';

  no_blank = 0;
  show_manager = 0;
  blank_name = '';

  for select json_key, json_value
  from fm_json_parse(:add_params, 0, '')
  into :json_key, :json_value
  do
  begin
    if (json_key = 'show_manager') then
      show_manager = json_value;
    if (json_key = 'blank_name') then
      blank_name = json_value;
    if (json_key = 'date_kind') then
      date_kind = json_value;
    if (json_key = 'add_params') then
      add_params_2 = json_value;
  end

  if (blank_name = 'LoanBlank') then
  begin
    if (not exists(select * from print_blank_list(:obj_id, 0, current_date))) then
      no_blank = 1;
    rd_name = f_lang(lang, 'Печать договоров');
  end
  else
  begin
    select rd_name
    from refdata
    where rd_main_code = 'reflink' and
          rd_memo = 'BlankName' and
          rd_str_id = :blank_name
    into :rd_name;
    if (coalesce(rd_name, '') = '') then
      no_blank = 1;
  end

  if (no_blank = 1) then
  begin
    res_text = BLOB_APPEND(res_text, f_format('<h2 class="green_text">%1</h2>',
      f_lang(lang, 'Формы для печати не найдены')));
    res_text = BLOB_APPEND(res_text, '</body>');
    suspend;
    exit;
  end

  -- Проверка условий печати
  if (blank_name = 'LoanBlank') then
  begin
    select result_is_ok, result_txt
    from fm_check_loan_konditions('.print_docs.', :obj_id, 0, :un_id)
    into tmp_is_ok, tmp_text;
    if (tmp_is_ok = 0) then
    begin
      res_text = BLOB_APPEND(res_text, f_format('<h2 class="green_text">%1</h2>',
        f_lang(lang, 'Проверка условий')) || tmp_text || '</body>');
      suspend;
      exit;
    end
    else
    if (tmp_text <> '') then
      res_text = BLOB_APPEND(res_text, f_format('<h2 class="green_text">%1</h2>',
        f_lang(lang, 'Проверка условий')) || tmp_text);
  end

  res_text = BLOB_APPEND(res_text,
    f_format('<h2 class="green_text">%1</h2><table width="100%"><tr><th>%2</th></tr>',
    rd_name, f_lang(lang,'Названия форм')));

  if (blank_name = 'LoanBlank') then
  begin
    for select res_blank_id, res_blank_name
    from print_blank_list(:obj_id, 0, current_date) -- Дату уточнить
    into res_obj_id, rd_name
    do
    begin
      if (rd_name <> '') then
      begin
        uri = f_uri_blank(rd_name, 'print', obj_kind, obj_id, res_obj_id,
          f_lang(lang, 'Печать'), show_manager, date_kind, add_params_2);

        if (uri <> '') then
          res_text = BLOB_APPEND(res_text, '<tr><td>' || uri || '</td>');
      end
    end
  end
  else
  begin
    for select res_obj_id
        from hp_reflink_get_obj(:blank_name, 'BlankName')
        into :res_obj_id
    do
    begin
      rd_name = '';
      uri = '';

      select bln_name
      from blankname
      where bln_id = :res_obj_id
      into :rd_name;
      if (rd_name <> '') then
      begin
        uri = f_uri_blank(rd_name, 'print', obj_kind, obj_id, res_obj_id,
          f_lang(lang, 'Печать'), show_manager, date_kind, add_params_2);
        if (uri <> '') then
          res_text = BLOB_APPEND(res_text, '<tr><td align="center">' || uri || '</td>');
      end
    end
  end

  res_text = BLOB_APPEND(res_text, '</table></body>');
  suspend;
end
