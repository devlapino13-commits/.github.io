CREATE PROCEDURE HP_CLIENT_LEGITIMCALC_LIST(
    C_ID D_ID)
RETURNS (
    LGC_DATE D_DATE,
    LGC_SH_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    LGC_CALC D_INTEGER,
    LGC_SALDO D_INTEGER)
AS
begin
  lgc_saldo = 0;
  for select lgc_date, lgc_sh_date, k_kontract_no, lgc_calc
  from legitimcalc, kontract
  where
    lgc_from_c_id = :c_id and
    lgc_from_k_id = k_id and
    lgc_date <= current_date
  order by lgc_date, lgc_sh_date
  into lgc_date, lgc_sh_date, k_kontract_no, lgc_calc
  do
  begin
    lgc_saldo = lgc_saldo + lgc_calc;
    suspend;
  end
end
