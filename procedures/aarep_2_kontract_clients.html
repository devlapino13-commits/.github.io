CREATE PROCEDURE AAREP_2_KONTRACT_CLIENTS(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE D_STR_100,
    FIO D_STR_100,
    KONTRACT_NO D_STR_20,
    DISB_DATE D_DATE,
    BALANCE_ON_DATE CURR_TYPE,
    SVZ_FIO D_STR_100,
    POINT D_INTEGER,
    K_AMOUNT_APPROVED CURR_TYPE,
    KONTREF_LIST D_STR_2048,
    ACTIVE_REF_LIST D_STR_2048)
AS
declare variable C_ID D_ID;
declare variable PR_CODE STR_100;
declare variable CLI_BALANCE CURR_TYPE;
declare variable AMOUNT_MAX CURR_TYPE;
declare variable MA_ID D_INTEGER;
declare variable K_KONTRACT_NO_REF D_STR_20;
declare variable KR_FROM_K_ID D_ID;
declare variable CL_PDN CURR_TYPE;
begin
  for select kc.kc_from_c_id, pr_code, k_kontract_no, reg_name, k_fact_disb_date, lo.c_name_fio, cl.c_name_fio, oks_balance,
             f_point(kc.kc_from_c_id, k_fact_disb_date), kc_pdn_koef, k_amount_approved, k_from_ma_app_id
      from kontstate
      inner join kontract on k_id = ks_from_k_id
      inner join knclient kc on k_id = kc_from_k_id
      inner join product on pr_id = k_from_pr_id
      inner join knaktlist kal on kal_from_kf_id = k_from_kf_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = kal_new_reg_id
      inner join region on reg_unicode = kal_new_reg_id
      inner join clients lo on lo.c_id = kal_new_kf_officer
      inner join clients cl on cl.c_id = kc.kc_from_c_id
      inner join oksummary on oks_from_k_id = k_id
      inner join knclient_pdn kcp on kcp.kc_id = kc.kc_id
      where :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin >= '01.08.2025' and        -- RFCRU-4262
            kc_amount_approved > 0 and
            kal.is_delete = 0 and
            kal_date_end = '31.12.9999' and
            k_refund = 1 and
            oks_balance > 0 and
            oks_close_state = 0 and
            :rep_date_end between oks_date_begin and oks_date_end and
            kcp_kind = 0 and
            kal_new_kf_officer > 1000000000004
      into c_id, pr_code, kontract_no, office, disb_date, svz_fio, fio, balance_on_date, point,
           cl_pdn, k_amount_approved, ma_id
  do
  begin
    --RFCRU-4262
    kontref_list = ''; --список реф-х займов
    active_ref_list = ''; --список нереф-х займов
    for select kr.kr_from_k_id, k.k_kontract_no
    from kontrefinance kr
    left join kontract k on k.k_id = kr.kr_from_k_id
    where kr_from_ma_id = :ma_id
    and kr_from_k_id is not null
    into :kr_from_k_id, k_kontract_no_ref
    do
    begin
      kontref_list = kontref_list || k_kontract_no_ref || '; ';

      if (exists(
        select 1
          from kontstate ks
          inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
          where ks.ks_ol_is_delete = 0
            and ks.ks_state = 1
            and :rep_date_end between ks.ks_date_begin and ks.ks_date_end
            and ks.ks_from_k_id = :kr_from_k_id
            and kc.kc_from_c_id = :c_id
            and kc.kc_amount_applicate <> 0
        )) then
        begin
          active_ref_list = active_ref_list || coalesce(:k_kontract_no_ref, '') || '; ';
        end
    end

    cli_balance = 0;
    select sum(oks_balance)
    from knclient, oksummary, kontract
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = oks_from_k_id and
          kc_from_k_id = k_id and
          oks_balance > 0 and
          oks_close_state = 0 and
          k_disb_date <= :disb_date and
          :rep_date_end between oks_date_begin and oks_date_end
    into cli_balance;
    cli_balance = coalesce(cli_balance, 0);

    amount_max = 0;

    if (exists (
          select 1
          from kontstate ks
          inner join kontract k on k.k_id = ks.ks_from_k_id
          inner join knclient kc on k_id = kc_from_k_id
          inner join product pr on pr.pr_id = k.k_from_pr_id
          where :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            kc_amount_approved > 0 and
            kc_from_c_id = :c_id and
            pr.pr_code containing '.zalog.')) then
        pr_code = '.zalog.';

    if (pr_code containing '.zalog.') then
    begin
      amount_max = case
                     when point between 0 and 9 then 200000
                     when point between 10 and 15 then 250000
                     when point between 16 and 35 then 300000
                     when point between 36 and 45 then 400000
                     when point >= 46 then 500000
                     else 200000
                   end;
    end
    -- RFCRU-4061
    if (pr_code containing '.oivo.' or -- Ойво
        pr_code containing '.express.' or -- Экспресс
        pr_code containing '.shans.' or -- Шанс
        pr_code containing '.universal.' or -- Универсальный
        pr_code containing '.selfemployed.' or  -- Самозанятый
        pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%') then -- Надежный клиент + Параллельный
    begin
      amount_max = case
                       when point between 0 and 9 then 100000
                       when point between 10 and 15 then 150000
                       when point between 16 and 35 then 250000
                       when point between 36 and 45 then 300000
                       when point >= 46 then 400000
                       else 100000
                     end;
    end
    if (pr_code containing '.comfortable.') then
    begin
      amount_max = case
                       when point <= 6 then 100000
                       when point <= 11 then 160000
                       else 200000
                     end;
    end 

    if (active_ref_list <> '') then
    begin
      suspend;
    end
  end
end
