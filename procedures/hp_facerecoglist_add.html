CREATE PROCEDURE HP_FACERECOGLIST_ADD(
    CDS_ID D_ID)
AS
declare variable doc_blob_1 d_id;
declare variable c_id d_id;
declare variable photo_id d_id;
begin
  select cds_doc_blob_1, c_id, c_blob_photo_id
  from clientdocs, clients
  where cds_id = :cds_id and
        cds_from_c_id = c_id and
        cds_doc_blob_1 is not null and
        c_blob_photo_id is not null
  into doc_blob_1, c_id, photo_id;

  if (c_id is null) then
    exit;

  if (not exists(
  select * from facerecoglist
  where  frl_from_c_id = :c_id and
         frl_photo_bls_id = :photo_id and
         frl_doc_bls_id = :doc_blob_1)) then
  begin
    insert into facerecoglist (frl_from_c_id, frl_photo_bls_id, frl_doc_bls_id, frl_from_cds_id, frl_state)
    values (:c_id, :photo_id, :doc_blob_1, :cds_id, 0);
  end
end
