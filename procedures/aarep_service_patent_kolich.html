CREATE PROCEDURE AAREP_SERVICE_PATENT_KOLICH(
    SD_ID INTEGER,
    SD_DATE_BEGIN DATE,
    SD_DATE_END DATE)
RETURNS (
    RES_DAYS_ALL INTEGER,
    RES_DAYS_PATENT INTEGER,
    RES_DAYS_SPOLIS INTEGER)
AS
declare variable res_dd date;
begin
  res_days_patent = 0;
  res_days_spolis = 0;
  res_days_all = 0;
  res_days_all = sd_date_end - sd_date_begin + 1;
  if (res_days_all <= 0) then
    res_days_all = 1;
  for select distinct res_dd
      from aarep_service_patent_tmp(:sd_id, :sd_date_begin, :sd_date_end, 1)
      into :res_dd
  do
  begin
    res_days_patent = res_days_patent + 1;
  end
  for select distinct res_dd
      from aarep_service_patent_tmp(:sd_id, :sd_date_begin, :sd_date_end, 0)
      into :res_dd
  do
  begin
    res_days_spolis = res_days_spolis + 1;
  end
  suspend;
end
