CREATE PROCEDURE HP_ADDPAYMENTQR_VIEW(
    REG_ID D_INTEGER,
    DTBEG D_DATE,
    DTEND D_DATE)
RETURNS (
    KQA_ID D_INTEGER,
    KQA_DTCREATE D_DATE,
    KQA_DTPAY D_DATETIME,
    KQA_REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    AG_ID D_INTEGER,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    KPT_ID D_INTEGER,
    KPT_ERROR_TEXT D_STR_255,
    KPT_STATUS D_SMALLINT,
    KQA_AMOUNT D_CURRENCY,
    KQA_QR_JSON D_STR_8100,
    KQA_DETAIL D_STR_LONG,
    STATUS D_STR_20)
AS
begin
  for
    select
      kqa_id,
      kqa_dtcreate,
      kqa_dtpay,
      kqa_reg_id,
      reg_name,
      kqa_ag_id,
      c_id,
      c_name_fio,
      kqa_kpt_id,
      kpt_error_text,
      kpt_status,
      kqa_amount,
      kqa_qr_json,
      kqa_detail
    from kontpaytasks_qr_add
      inner join region_list(:reg_id, 1, 3, 1)
        on id_list = kqa_reg_id
      inner join region
        on reg_unicode = id_list
      inner join agreements
        on ag_id = kqa_ag_id
      inner join clients
        on c_id = ag_object_id
      left join kontpaytasks
        on kpt_id = kqa_kpt_id
    where kqa_dtcreate between :dtbeg and :dtend
  into
    :kqa_id,
    :kqa_dtcreate,
    :kqa_dtpay,
    :kqa_reg_id,
    :reg_name,
    :ag_id,
    :c_id,
    :c_name_fio,
    :kpt_id,
    :kpt_error_text,
    :kpt_status,
    :kqa_amount,
    :kqa_qr_json,
    :kqa_detail
  do
  begin
    status =
      case kpt_status
        when -1 then 'ошибка'
        when  0 then 'в очереди'
        when  1 then 'в обработке'
        when  2 then 'обработан'
      else 'создан' end;
    suspend;
  end
end
