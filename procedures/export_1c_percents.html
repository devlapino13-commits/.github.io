CREATE PROCEDURE EXPORT_1C_PERCENTS(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    OL_DATE DATE,
    C_ID VARCHAR(20),
    OB_SUMMA DECIMAL(18,4),
    OL_MEMO VARCHAR(255),
    OB_CURRENCY VARCHAR(3))
AS
declare variable k_stop_perc date;
declare variable aarep_calc_method integer;
declare variable dd1 date;
declare variable dd2 date;
declare variable aarep_k_id decimal(18,0);
declare variable aarep_percent_per_day decimal(18,7);
declare variable c_id_tmp decimal(18,0);
declare variable db_id integer;
declare variable k_kredline_end date;
declare variable k_zero_balance_date date;
begin
  dd1 = :report_date_begin;
  ol_memo = 'Проценты по кредиту';

  :db_id = f_get_db_id();

  while ((dd1 <= :report_date_end) or (dd2 <= :report_date_end)) do
  begin
    if (extract(weekday from dd1) = 6) then
      dd2 = dd1 + 2;
    else
      dd2 = dd1;

    if (dd2 > :report_date_end) then
      dd2 = :report_date_end;
    /* Procedure Text */
    for select distinct k_clients_cashier_id, ks_from_k_id, k_percent_per_day, k_currency, k_stop_perc, k_kredline_end,
                        k_zero_balance_date
        from kontstate, knaktlist, region_list(:report_region_id, 1, -1, 1), kontract
        where k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ((ks_date_end >= :dd1 and
              cast(ks_date_begin as date) <= :dd2 and
              ks_state = 1) or
              (ks_date_begin = :dd1 and
              ks_state = 2)) and
              kal_from_kf_id = k_from_kf_id and
              kal_date <= :dd2 and
              kal_date_end >= :dd2 and
              (kal_new_reg_id = id_list) and
              knaktlist.is_delete = 0
        into :c_id_tmp, :aarep_k_id, :aarep_percent_per_day, :ob_currency, :k_stop_perc, :k_kredline_end,
             :k_zero_balance_date
    do
    begin
      c_id = 'N' || cast(c_id_tmp as varchar(15));
      if (f_new_perc_active(dd2) = 1) then
      begin
        for select result_calced
            from aarep_calc_perc_one_period(:aarep_k_id, :dd1, :dd2)
            into ob_summa
        do
        begin
          ol_date = :dd2;
          suspend;
        end
      end else
      begin
        for select result_calced
            from aarep_calc_perc_one(:aarep_k_id, :dd1, :dd2, 1, :db_id, :k_kredline_end, :k_stop_perc, :k_zero_balance_date)
            into ob_summa
        do
        begin
          ol_date = :dd2;
          suspend;
        end
      end
    end
    dd1 = dd2 + 1;
    dd2 = dd1;
  end
end
