CREATE PROCEDURE HP_SUD_PAY_DEFINE_COURT(
    SLC_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable eil_id d_id;
declare variable slc_address d_str_long;
declare variable smd_proc_type d_integer;
declare variable eic_info_actual_days d_integer;
declare variable kind d_integer;
begin
  res_text = 'Запросы подсудности отключены';
  suspend;
  exit; -- пока отключаем запросы подсудности

  res_text = '';

  select kc_from_c_id, slc_address, smd_proc_type
  from sud_loanclient, knclient, sud_maindocs
  where kc_id = slc_from_kc_id
    and smd_id = slc_from_smr_id
    and slc_id = :slc_id
  into c_id, slc_address, smd_proc_type;

  if (c_id is null) then
    res_text = 'Не найден клиент для запроса. slc_id = ' || coalesce(slc_id,  '<NULL>');

  else if (coalesce(smd_proc_type, '') = '') then
    res_text = 'Выберите тип делопроизводства!';

  else if (coalesce(slc_address, '') = '') then
    res_text = 'У клиента не заполнен адрес подсудности';

  if (res_text = '') then
  begin
    kind = iif(smd_proc_type = 0, 1, 2);

    if (exists(select eil_status
              from hp_sud_get_request_status(:slc_id, :kind)
              where eil_status >= 0))
    then
    begin
      res_text = 'Запрос уже был отправлен!';
      suspend;
      exit;
    end 
      
    select eil_id
    from hp_sud_extinfo_insert('DigiDebt_court', 'sud_loanclient', :slc_id, :c_id, 1)
    into eil_id;

    insert into sud_extinfo (sue_slc_id, sue_kind, sue_date, sue_from_eil_id)
    values (:slc_id, :kind, current_date, :eil_id);

    update sud_loanclient
    set slc_auto_worldcourt = iif(:smd_proc_type = 0, null, slc_auto_worldcourt),
        slc_auto_federalcourt = iif(:smd_proc_type = 1, null,  slc_auto_federalcourt)
    where slc_id = :slc_id;

    res_text = 'Запрос отправлен';
  end

  suspend;
end
