CREATE PROCEDURE HP_OPEN_DAY_FROM(
    REG_ID INTEGER,
    REG_DATE_BEGIN DATE,
    REG_DATE_END DATE,
    REG_USER_ID DECIMAL(18,0),
    REG_MEMO VARCHAR(255),
    REG_IS_CASH INTEGER)
AS
declare variable min_dd date;
declare variable last_is_closed integer;
declare variable last_dd date;
declare variable i integer;
begin
  /* Procedure Text */
  if (reg_is_cash is null) then
    reg_is_cash = 1;
  min_dd = :reg_date_begin;
  i = -1;
  last_dd = :min_dd - 1;
  while (last_dd < :reg_date_end) do
  begin
    i = i + 1;
    last_dd = cast((:min_dd + i) as date);
    select day_closed
    from hp_is_day_closed(:reg_id, :last_dd, :reg_is_cash)
    into :last_is_closed;
    if (last_is_closed = 1) then
    begin
      if (reg_is_cash = 1) then
      begin
        insert into closeday (cd_from_reg_id, cd_date, cd_number, cd_is_closed, cd_open_memo, cd_datetime, cd_user_id,
                              cd_is_cash_closed)
        values (:reg_id, :last_dd, -1, -1, :reg_memo, null, :reg_user_id, 0);
      end
      else
      begin
        insert into closeday (cd_from_reg_id, cd_date, cd_number, cd_is_closed, cd_open_memo, cd_datetime, cd_user_id,
                              cd_is_cash_closed)
        values (:reg_id, :last_dd, -1, 0, :reg_memo, null, :reg_user_id, -1);
      end
    end
  end
end
