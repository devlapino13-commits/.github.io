CREATE PROCEDURE AAREP_CASHCHECK_PART_FORM(
    REP_ID D_INTEGER)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_JOBNAME D_STR_100,
    CCP_DATESIGN D_DATETIME,
    C_SIGN D_INTEGER,
    C_QR D_STR_LONG,
    C_LOGIN D_STR_CODE)
AS
declare variable C_ID D_ID;
declare variable REP_DATE D_DATE;
declare variable DTSIGN D_DATETIME;
declare variable EDL_ID D_ID;
begin
  for
    select
      c_id,
      c_name_fio,
      ccm_date,
      ccm_edl_id,
      ccp_datesign,
      iif(ccp_datesign is null, 0, 1)
    from cashcheckpart, clients, cashcheckmain
    where ccp_from_ccm_id = :rep_id
      and ccp_from_c_id = c_id
      and ccm_id = :rep_id
  into
    :c_id,
    :c_name_fio,
    :rep_date,
    :edl_id,
    :ccp_datesign,
    :c_sign
  do
  begin
    c_jobname = '';
    select jl_name
    from hr_pstatus, joblist
    where ps_from_c_id = :c_id and
          ps_date_begin <= :rep_date and
          ps_date_end >= :rep_date and
          ps_status = 1 and
          ps_from_jl_id = jl_id
    order by ps_is_working desc
    rows 1
    into :c_jobname;

    c_login = (
      select first 1 un_name
      from username
      where un_from_c_id = :c_id
      order by un_id desc);

    if (edl_id > 0 and ccp_datesign is null) then
    begin
      dtsign = (
        select eds_status_datetime
        from edoc_sign
        where eds_from_edl_id = :edl_id
          and eds_from_c_id = :c_id
          and eds_status = 1
          and eds_enum_edoc_sign_kind <= 1);
      if (dtsign is not null) then
      begin
        ccp_datesign = dtsign;
        c_sign = 2;
        c_qr = f_edoc_qr_by_edl(:edl_id, :c_id);
      end
    end

    suspend;
  end
end
