CREATE PROCEDURE AAREP_CLASSIC_ISSUE_REPORT(
    REP_REGION_ID ID_TYPE,
    REP_DATE_END D_DATE)
RETURNS (
    PARENT_NAME D_STR_100,
    REG_NAME D_STR_100,
    KOL_KC D_INTEGER,
    KOL_ISSUANCE DECIMAL(18,4),
    AVERAGE INTEGER,
    SUMMA CURR_TYPE)
AS
declare variable reg_id d_id;
declare variable first_date d_date;
declare variable sawt_id d_id;
declare variable aa_kol_staff d_integer;
declare variable c_id d_id;
begin
  kol_issuance = 0;
  first_date = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date);
  for select parent_name,
             reg.reg_name,
             reg.reg_unicode
      from region_list(:rep_region_id, 1, 3, 1)
      inner join region reg on id_list = reg_unicode
      where :rep_date_end between reg.reg_date_open and coalesce(reg.reg_date_close, '31.12.9999') and
            reg.reg_unicode not in (1060100, 1090100)
      order by parent_name
      into :parent_name,
           :reg_name,
           :reg_id
  do
  begin
    select count(*),
           sum(kc.kc_amount_approved)
    from kontract k
    inner join kontstate ks on ks.ks_from_k_id = k.k_id
    inner join knclient kc on kc.kc_from_k_id = k.k_id
    inner join product pr on k.k_from_pr_id = pr.pr_id
    where ks.ks_region_id = :reg_id and
          ks.ks_ol_is_delete = 0 and
          ks.ks_state = 1 and
          ks.ks_date_begin between :first_date and :rep_date_end and
          kc.kc_amount_approved > 0 and
          pr.pr_code not containing '.ipf.'
    into :kol_issuance,
         :summa;

    average = 0;
    kol_kc = 0;
    for select ps_from_c_id
        from hr_pstatus, joblist, jobkind jk
        where ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              :rep_date_end between ps_date_begin and ps_date_end and
              ps_from_reg_id = :reg_id and
              jl_kind = jk.jk_id and
              jl_id in (227, 226)and -- КС и СКС
              ps_is_on_staff = 1
        into :c_id
    do
    begin
      sawt_id = null;
      select sawt_id
      from hr_voctravel, hr_orders, sa_worktype
      where hrv_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            hrv_from_c_id = :c_id and
            hrv_status = 1 and
            hrv_date_begin <= :rep_date_end and
            hrv_date_end_fact >= :rep_date_end
      order by hrv_date_begin desc
      rows 1
      into :sawt_id;
      sawt_id = coalesce(sawt_id, 0);
      if (sawt_id not in (24, 17, 18)) then
      begin
        kol_kc = kol_kc + 1;
      end
    end
    average = :kol_issuance / :kol_kc;
    suspend;
  end
end
