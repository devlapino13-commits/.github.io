CREATE PROCEDURE HP_FPS_GET_LOAN_BANK(
    K_ID D_ID,
    C_ID D_ID)
RETURNS (
    FPB_ID D_INTEGER,
    FPB_CODE D_STR_CODE,
    FPB_NAME D_STR_255,
    FPB_NAME_FULL D_STR_255,
    CELL_NO D_STR_100)
AS
begin
  -- Возвращает актуальный банк для указанного контракта и клиента
  select fpb_id, fpb_code, f_cont_code(fcb_from_clt_id), fpb_name, fpb_name_full
  from knclient, fps_loan_banks, fps_client_banks, fps_banks
  where
    kc_from_k_id = :k_id and
    kc_from_c_id = :c_id and
    kc_amount_approved > 0 and
    flb_from_kc_id = kc_id and
    fbl_from_fcb_id = fcb_id and
    fcb_from_fpb_id = fpb_id and
    fpb_status = 1 -- Активен
  into fpb_id, fpb_code, cell_no, fpb_name, fpb_name_full;

  suspend;
end
