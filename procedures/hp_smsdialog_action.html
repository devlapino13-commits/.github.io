CREATE PROCEDURE HP_SMSDIALOG_ACTION(
    SDR_ID INTEGER,
    SDR_CELL_NO VARCHAR(20),
    SDR_SYSTEM_ID INTEGER,
    SDR_FROM_C_ID DECIMAL(18,0),
    SDR_TEXT VARCHAR(255))
AS
declare variable action_name varchar(100);
declare variable sms_text varchar(255);
declare variable client_fio varchar(100);
declare variable rand_n integer;
declare variable next_n decimal(18,0);
declare variable curr_n decimal(18,0);
declare variable new_code varchar(10);
declare variable need_fio integer;
declare variable tmp_text varchar(255);
declare variable command_len integer;
declare variable tmp_cell_no varchar(255);
declare variable i integer;
declare variable cha varchar(255);
declare variable reg_address varchar(255);
declare variable reg_phone varchar(100);
declare variable reg_id integer;
declare variable sms_text_add varchar(1024);
declare variable c_id decimal(18,0);
declare variable sdr_cell_no2 varchar(20);
declare variable cm_name d_str_100;
begin
-- Выполняет действия в соответствии с системной командой
  sms_text_add = '';
  sdr_cell_no2 = '';
  need_fio = 0;
  sms_text = '';
  client_fio = '';
  cm_name = iif(f_get_db_id() = 81001, 'М Булак', 'М Булак');

  for select sdsw_action, char_length(sdsw_command_names)
      from smsdialogsyswords
      where sdsw_id = :sdr_system_id and
            sdsw_is_active = 1 and
            sdsw_action <> 'sms_pin'
      into :action_name, :command_len
  do
  begin
    client_fio = '';
    if (sdr_from_c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :sdr_from_c_id
      into :client_fio;
    end
    if (action_name = 'Loan_balance') then -- отправка баланса клиента
    begin
      need_fio = 0;
      select sms_text
      from hp_smsdialog_action_balance(:sdr_from_c_id)
      into :sms_text;
    end
    if (action_name = 'Client_Account') then -- лицевой счет КИВИ
    begin
      sms_text = 'Лицевой счет еще не создан.';
      need_fio = 1;
    end
    if (action_name = 'Marketing_register') then -- отправка баланса клиента
    begin
      sms_text = 'Salam! ';
      rand_n = round((rand() * 9), 0);
      sms_text = sms_text || rand_n;
      rand_n = round((rand() * 9), 0);
      sms_text = sms_text || rand_n;
      rand_n = round((rand() * 9), 0);
      sms_text = sms_text || rand_n;
      rand_n = round((rand() * 9), 0);
      sms_text = sms_text || rand_n;
      need_fio = 1;
    end
    if (action_name = 'Agent_register') then -- регистрация агента
    begin
      if (not(exists(select *
                     from hr_pstatus
                     where ps_from_c_id = :sdr_from_c_id and
                           ps_date_begin <= current_date and
                           ps_date_end >= '31.12.2999' and
                           ps_status = 1 and
                           ps_is_working = 1 and
                           ps_from_reg_id <> 1010100))) then
      begin
        need_fio = 0;
        if (client_fio <> '') then
        begin
          sms_text = 'Поздравляем! Вы стали агентом '||cm_name||'. Ваш код: ';
          curr_n = 0;
          select count(*)
          from smsdialogsend
          where sds_action = 'Agent_register'
          into :curr_n;
          if (curr_n is null) then
            curr_n = 0;
          next_n = curr_n + 1;
          next_n = next_n * 1046527;
          new_code = right(next_n, 6);
          new_code = left(new_code, 3) || '-' || right(new_code, 3);
          sms_text = sms_text || new_code;

          sms_text_add = cm_name||': Для активации скидки отправьте СМС на 8(903)76-76-000 со словом «Скидка»' || ' и тел. номер знакомого. Пример: Скидка 79007654321. СМС нужно отправить с номера, ' || 'который зарегистрирован в '||cm_name;
          sdr_cell_no2 = sdr_cell_no;
        end
        else
          sms_text = 'Ошибка! В базе нет клиента с таким номером.';
      end
      else
      begin
        sms_text = 'Сотрудники офисов не могут работать как агент и активировать скидки';
      end
    end
    if (action_name = 'Kupon_register') then -- регистрация агента
    begin
      sms_text = '';
      need_fio = 0;

      if (exists(select *
                 from smsdialogreciever
                 where sdr_from_c_id = :sdr_from_c_id and
                       sdr_action = 'Agent_register')) then
      begin
        if (exists(select *
                   from hr_pstatus
                   where ps_from_c_id = :sdr_from_c_id and
                         ps_date_begin <= current_date and
                         ps_date_end >= '31.12.2999' and
                         ps_status = 1 and
                         ps_is_working = 1 and
                         ps_from_reg_id <> 1010100)) then
        begin
          sms_text = 'Сотрудники офисов не могут работать как агент и активировать скидки';
        end
        else
        begin
          tmp_text = substring(sdr_text from (:command_len + 1) for 255);
          tmp_cell_no = '';
          i = 1;
          while (i <= char_length(tmp_text)) do
          begin
            cha = substring(tmp_text from i for 1);
            if ((cha = '0') or (cha = '1') or (cha = '2') or (cha = '3') or (cha = '4') or (cha = '5') or (cha = '6') or (cha = '7') or (cha = '8') or (cha = '9')) then
            begin
              tmp_cell_no = tmp_cell_no || cha;
            end
            i = i + 1;
          end
          if ((char_length(tmp_cell_no) = 11) and
              (left(tmp_cell_no, 1) = '8')) then
          begin
            tmp_cell_no = '7' || substring(tmp_cell_no from 2 for 10);
          end
          if (((char_length(tmp_cell_no) = 11) and
              (left(tmp_cell_no, 1) = '7')) or ((char_length(tmp_cell_no) = 12) and
              (left(tmp_cell_no, 1) = '9'))) then
          begin
            for select distinct res_object_id
                from hp_contdata_find_object(:tmp_cell_no, 'cell')
                where res_object_kind = 0
                into :c_id
            do
            begin
              if (exists(select *
                         from knclient, kontstate
                         where kc_from_c_id = :c_id and
                               kc_from_k_id = ks_from_k_id and
                               kc_amount_approved > 0 and
                               ks_state = 1 and
                               ks_date_end >= (current_date - 180) and
                               ks_ol_is_delete = 0 and
                               ks_date_begin <= ks_date_end)) then
              begin
                sms_text = 'Клиент ' || tmp_cell_no || ' уже является клиентом компании';
              end

              if (c_id = sdr_from_c_id) then
              begin
                sms_text = 'Агент не может активировать скидку для себя';
              end
            end
            if (sms_text = '') then
            begin
              if (exists(select *
                         from smsdialogsend
                         where sds_action = 'Kupon_register' and
                               sds_cell_no = :tmp_cell_no and
                               (sds_text like 'Ваш купон активирован%' or
                               sds_text like 'Поздравляем! Вы%'))) then
              begin
                sms_text = 'Клиенту ' || tmp_cell_no || ' ранее уже предоставлена скидка другим агентом';
              end
              else
              begin
                sms_text_add = 'Скидка для ' || :tmp_cell_no || ' успешно активирована';
                sdr_cell_no2 = sdr_cell_no;

                sms_text = 'Поздравляем! Вы получили скидку в 200 руб. в '||cm_name||'! 88005557171';
                sdr_cell_no = tmp_cell_no;
              end
            end
          end
          if (sms_text = '') then
          begin
            sms_text = 'Номер должен быть в международном формате: ' || tmp_text;
          end
        end
      end
      else
      begin
        sms_text = 'Вы не являетесь зарегистрированным агентом.';
      end
    end

    if (action_name = 'Client_from_marketing') then -- регистрация агента
    begin
      sms_text = '';
      need_fio = 0;
      reg_phone = '';
      reg_address = '';
      reg_id = null;
      select reg_unicode
      from hr_pstatus, region, joblist, jobkind
      where ps_from_c_id = :sdr_from_c_id and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id
--      and ps_from_c_id in (1000000000001, 1010100000004, 1010100189147, 1010100002561)
            and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.Specialist_Promoter' -- Промоутер
            rows 1
      into :reg_id;
      if (reg_id is not null) then
      begin
        select reg_phone, adl_address_str
        from region, addrlist
        where reg_unicode = :reg_id and
              reg_address_fact = adl_id
        into :reg_phone, :reg_address;

        if (reg_phone is null) then
          reg_phone = '';
        if (reg_address is null) then
          reg_address = '';

        if (reg_address <> '') then
        begin
          cha = substring(reg_address from 1 for 1);
          if ((cha = '0') or (cha = '1') or (cha = '2') or (cha = '3') or (cha = '4') or (cha = '5') or (cha = '6') or (cha = '7') or (cha = '8') or (cha = '9')) then
          begin
            reg_address = substring(reg_address from 9 for 255);
          end
          tmp_text = substring(sdr_text from (:command_len + 1) for 255);
          tmp_cell_no = '';
          i = 1;
          while (i <= char_length(tmp_text)) do
          begin
            cha = substring(tmp_text from i for 1);
            if ((cha = '0') or (cha = '1') or (cha = '2') or (cha = '3') or (cha = '4') or (cha = '5') or (cha = '6') or (cha = '7') or (cha = '8') or (cha = '9')) then
            begin
              tmp_cell_no = tmp_cell_no || cha;
            end
            i = i + 1;
          end
          if ((char_length(tmp_cell_no) = 11) and
              (left(tmp_cell_no, 1) = '8')) then
          begin
            tmp_cell_no = '7' || substring(tmp_cell_no from 2 for 10);
          end
          if ((char_length(tmp_cell_no) = 11) and
              (left(tmp_cell_no, 1) = '7')) then
          begin
            if (reg_phone <> '') then
              sms_text = 'Займы от '||cm_name||', адрес: ' || :reg_address || ' Тел. ' || :reg_phone;
            else
              sms_text = 'Займы от '||cm_name||', адрес: ' || :reg_address;
            sdr_cell_no = tmp_cell_no;
          end
          else
          begin
            if ((char_length(tmp_cell_no) = 12) and
                (left(tmp_cell_no, 1) = '9')) then
            begin
              if (reg_phone <> '') then
                sms_text = 'Займы от '||cm_name||', адрес: ' || :reg_address || ' Тел. ' || :reg_phone;
              else
                sms_text = 'Займы от '||cm_name||', адрес: ' || :reg_address;
              sdr_cell_no = tmp_cell_no;
            end
          end
          if (sms_text = '') then
          begin
            sms_text = 'Номер должен быть в международном формате: ' || tmp_text;
          end
        end
      end
      else
      begin
        sms_text = 'Вы не являетесь действующим промоутером.';
      end
    end
  end

  if (sms_text <> '') then
  begin
    if ((client_fio <> '') and
        (need_fio = 1)) then
      sms_text = client_fio || ': ' || sms_text;

    insert into smsdialogsend (sds_id, sds_cell_no, sds_text, sds_system_id, sds_from_c_id, sds_status, sds_service_id,
                               sds_sender_sdr_id, sds_reciever_sdr_id)
    values (-1, :sdr_cell_no, :sms_text, :sdr_system_id, :sdr_from_c_id, 0, null, :sdr_id, null);

    if ((sms_text_add <> '') and
        (sdr_cell_no2 <> '')) then
    begin
      insert into smsdialogsend (sds_id, sds_cell_no, sds_text, sds_system_id, sds_from_c_id, sds_status,
                                 sds_service_id, sds_sender_sdr_id, sds_reciever_sdr_id)
      values (-1, :sdr_cell_no2, :sms_text_add, :sdr_system_id, :sdr_from_c_id, 0, null, :sdr_id, null);
    end
  end
end
