CREATE PROCEDURE AAREP_SUMMARY_TAX_NSP_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    TAX_KIND VARCHAR(50),
    TAX_BASE DECIMAL(18,4),
    TAX_KOEF DECIMAL(15,7),
    TAX_REGION_ID INTEGER,
    AL_FROM_BG_ID STR_20)
AS
declare variable al_id str_20;
begin
  for select distinct al_from_reg_id, al.al_id, 2, al.al_from_bg_id
      from region_list(:report_region_id, 1, -1, 1), acc_list al
      where id_list = al.al_from_reg_id and
            al.al_from_bg_id in (--
                             '60121', --1.    60121 (все субсчета). Сальдо по оборотам, т.е. Кт. – Дт.
                             '60521', --2.    60521 (все субсчета). Сальдо по оборотам, т.е. Кт. – Дт.
                             '60591', --3.    60591 (все субсчета). Сальдо по оборотам, т.е. Кт. – Дт.
                             '60901', --4.    60901/1 Сальдо по оборотам, т.е. Кт. – Дт.
                             '61011', --5.    61011 (все субсчетам). Только Оборот по Кт.
                             '61099', --6.    61099 (все субсчета). Сальдо по оборотам, т.е. Кт. – Дт.
                             '61102', --7.    61102 (все субсчета). Сальдо по оборотам, т.е. Кт. – Дт.
                             '61106', --8.    61106 (все субсчета). Сальдо по оборотам, т.е. Кт. – Дт.
                             '61107' --9.    61107 (все субсчета). Сальдо по оборотам, т.е. Кт. – Дт.
                             ) and
            iif(al.al_from_bg_id = '60901', '60901/1', al.al_account) = al_account

      into :tax_region_id, al_id, :tax_koef, al_from_bg_id
  do
  begin
    tax_base = 0;
    tax_kind = 'Ставка ' || cast(:tax_koef as decimal(15,1)) || ' %:';
    suspend;

    for with tax_table
        as (select ol_from_reg_id, 2 txn_percent, sum(ob_summa) ob_summa
            from operlist ol, operbook ob
            where cast(ol_date as date) between :report_date_begin and :report_date_end and
                  ol_id = ob_from_ol_id and
                  :al_id = ob_account_kt and
                  ob.ob_short_acc_dt <> '60901/1' and
                  ol.is_delete = 0
            group by ol_from_reg_id, txn_percent
            union all
            select ol_from_reg_id, 2 txn_percent, -sum(ob_summa)
            from operlist ol, operbook ob
            where cast(ol_date as date) between :report_date_begin and :report_date_end and
                  ol_id = ob_from_ol_id and
                  :al_id = ob_account_dt and
                  ob.ob_short_acc_dt <> '60901/1' and
                  ol.is_delete = 0
            group by ol_from_reg_id, txn_percent)
        select ol_from_reg_id, 2 txn_percent, sum(ob_summa) ob_summa
        from tax_table
        group by ol_from_reg_id, txn_percent
        order by txn_percent, ol_from_reg_id
        into :tax_region_id, :tax_koef, :tax_base
    do
    begin
      tax_kind = 'Ставка ' || cast(:tax_koef as decimal(15,1)) || ' %:';
      suspend;
    end
  end
end
