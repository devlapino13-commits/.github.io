CREATE PROCEDURE AAREP_CARD_BALANCES_MOL(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    CARD_ID D_INTEGER,
    PIN_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    C_NAME_FIO D_STR_100,
    JL_NAME D_STR_100,
    C_NAME_FIO_PIN D_STR_100,
    JL_NAME_PIN D_STR_100,
    DT_RECV D_DATE,
    DT_RECV_PIN D_DATE,
    PARENT_NAME D_STR_100)
AS
declare variable cam_id d_integer;
declare variable c_id d_id;
declare variable c_id_pin d_id;
declare variable reg_id d_integer;
declare variable ol_id d_id;
declare variable ol_datetime d_datetime;
declare variable dt_curr d_date;
declare variable reg_lst d_str_8100;
declare variable reg_parent d_integer;
begin
  dt_curr = current_date;
  select
    coalesce(list(id_list, ';'), '')
  from region_list(:rep_region_id,1,3,1)
  into
    :reg_lst;

  for
    select
      cal_card_id,
      max(cam_ol_datetime)
    from cardaktmain
      inner join cardplace
        on cpl_id = cam_dst_cpl_id
          and cpl_is_transfer = 0
      inner join operlist
        on ol_id = cam_from_ol_id
          and ol_date <= :rep_date_end
      inner join cardaktrange
        on car_from_cam_id = cam_id
      inner join cardlist
        on cal_card_id between car_first_card_id and car_last_card_id
          and cal_status <= f_enum_id('cardlist_status', 'transfer_to_client')
          and coalesce(cal_date_issue, :rep_date_end) >= :rep_date_end
    where cam_from_ol_id > 0
    group by cal_card_id
  into
    :card_id,
    :ol_datetime
  do
  begin
    dt_recv = ol_datetime;

    select
      cam_id,
      reg_unicode,
      reg_name,
      reg.parent,
      c.c_id,
      c.c_name_fio,
      c1.c_id,
      c1.c_name_fio
    from cardaktmain
      inner join cardplace as p
        on p.cpl_id = cam_dst_cpl_id
      inner join region reg
        on reg_unicode = p.cpl_from_reg_id
      inner join clients as c
        on c.c_id = p.cpl_from_c_id
      left join cardplace as p1
        on p1.cpl_id = cam_dst_pin_cpl_id
      left join clients as c1
        on c1.c_id = p1.cpl_from_c_id
    where cam_ol_datetime = :ol_datetime
    into
      :cam_id,
      :reg_id,
      :reg_name,
      :reg_parent,
      :c_id,
      :c_name_fio,
      :c_id_pin,
      :c_name_fio_pin;

    if ((reg_lst not containing reg_id))
      then continue;

    select r.reg_name
    from region r
    where r.reg_unicode = :reg_parent
    into parent_name;

    jl_name = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :c_id
        and :dt_curr between ps_date_begin and coalesce(ps_date_end, '31.12.9999')
        and ps_is_working = 1);

    if (exists(
      select * from cardlist
      where cal_card_id = :card_id
        and cal_last_pin_cpl_id > 0)) then
    begin
      pin_id = card_id;
      dt_recv_pin = dt_recv;
      jl_name_pin = (
        select jl_name
        from hr_pstatus
          inner join joblist
            on jl_id = ps_from_jl_id
        where ps_from_c_id = :c_id_pin
          and :dt_curr between ps_date_begin and coalesce(ps_date_end, '31.12.9999')
          and ps_is_working = 1);
    end else begin
      pin_id = null;
      dt_recv_pin = null;
      jl_name_pin = null;
    end
    suspend;
  end
end
