CREATE PROCEDURE AAREP_OSV_EPS_OBOR_STATEMENT(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DT_AB_ID ID_TYPE,
    REP_KT_AB_ID ID_TYPE,
    REP_DT_AB_CODE D_STR_CODE = null,
    REP_KT_AB_CODE D_STR_CODE = null)
RETURNS (
    ST_LINE_NO INTEGER,
    ST_KOB_ID ID_TYPE,
    ST_DATE DATE,
    ST_DT_ACC_ID ID_TYPE,
    ST_DT_ACC_CODE D_STR_CODE,
    ST_DT_ACC_NAME D_STR_255,
    ST_KT_ACC_ID ID_TYPE,
    ST_KT_ACC_CODE D_STR_CODE,
    ST_KT_ACC_NAME D_STR_255,
    ST_AMOUNT_SUM CURR_TYPE,
    ST_AMOUNT_VAL CURR_TYPE,
    ST_AMOUNT_TAX CURR_TYPE,
    ST_AMOUNT_TAX_DT CURR_TYPE,
    ST_AMOUNT_TAX_KT CURR_TYPE,
    ST_OB_DETAIL D_STR_LONG,
    ST_OPER_MEMO D_STR_LONG)
AS
declare variable has_filter integer;
declare variable acc_mask_dt d_str_code;
declare variable acc_mask_kt d_str_code;
declare variable n_start integer;
begin
  if (rep_dt_ab_code is null) then
    rep_dt_ab_code = '';
  if (rep_kt_ab_code is null) then
    rep_kt_ab_code = '';
  acc_mask_dt = '';
  acc_mask_kt = '';
  n_start = position('(', rep_dt_ab_code);
  if (n_start > 0) then
  begin
    acc_mask_dt = left(rep_dt_ab_code, 5);
    rep_dt_ab_code = substring(rep_dt_ab_code from n_start + 1);
    rep_dt_ab_code = left(rep_dt_ab_code, char_length(rep_dt_ab_code) - 1);
    n_start = char_length(rep_dt_ab_code);
    acc_mask_dt = acc_mask_dt || '_____' || rep_dt_ab_code || left('__________', 10 - n_start);
  end

  n_start = position('(', rep_kt_ab_code);
  if (n_start > 0) then
  begin
    acc_mask_kt = left(rep_kt_ab_code, 5);
    rep_kt_ab_code = substring(rep_kt_ab_code from n_start + 1);
    rep_kt_ab_code = left(rep_kt_ab_code, char_length(rep_kt_ab_code) - 1);
    n_start = char_length(rep_kt_ab_code);
    acc_mask_kt = acc_mask_kt || '_____' || rep_kt_ab_code || left('__________', 10 - n_start);
  end

  has_filter = 0;
  if (exists(select *
             from temp_ref_account_rep
             rows 1)) then
    has_filter = 1;

  if (has_filter = 0) then
  begin
    for select kob_id, kob_ol_date, kob_acc_dt, kob_acc_kt, obd_detail, ol_memo, kob_summa, kob_summa_val,
               kob_summa_tax_dt, kob_summa_tax_kt
        from kn_operbook, operlist, kn_obdetail, ref_account a_dt, ref_account a_kt
        where kob_ol_is_active = 1 and
              kob_ol_date between :rep_date_begin and :rep_date_end and
              kob_acc_dt = a_dt.acc_id and
              kob_acc_kt = a_kt.acc_id and
              a_dt.acc_from_ab_id = :rep_dt_ab_id and
              a_kt.acc_from_ab_id = :rep_kt_ab_id and
              kob_from_ol_id = ol_id and
              kob_detail_id = obd_id and
              (:acc_mask_dt = '' or
              a_dt.acc_code like :acc_mask_dt) and
              (:acc_mask_kt = '' or
              a_kt.acc_code like :acc_mask_kt) and
              (kob_summa <> 0 or
              kob_summa_val <> 0 or
              kob_summa_tax_dt <> 0)
        order by kob_ol_date, kob_from_ol_id, kob_id
        into :st_kob_id, :st_date, :st_dt_acc_id, :st_kt_acc_id, :st_ob_detail, :st_oper_memo, :st_amount_sum,
             :st_amount_val, :st_amount_tax_dt, :st_amount_tax_kt
    do
    begin
      st_line_no = st_line_no + 1;
      st_dt_acc_code = f_fmt_acc(st_dt_acc_id);
      st_kt_acc_code = f_fmt_acc(st_kt_acc_id);

      st_dt_acc_name = '';
      select acc_name
      from ref_account
      where acc_id = :st_dt_acc_id
      into st_dt_acc_name;

      st_kt_acc_name = '';
      select acc_name
      from ref_account
      where acc_id = :st_kt_acc_id
      into st_kt_acc_name;

      st_amount_tax = st_amount_tax_dt;
      suspend;
    end
  end
  else
  begin
    for select distinct kob_id, kob_ol_date, kob_acc_dt, kob_acc_kt, obd_detail, ol_memo, kob_summa, kob_summa_val,
               kob_summa_tax_dt, kob_summa_tax_kt
        from kn_operbook, operlist, kn_obdetail, ref_account a_dt, ref_account a_kt, temp_ref_account_rep
        where kob_ol_is_active = 1 and
              kob_ol_date between :rep_date_begin and :rep_date_end and
              kob_acc_dt = a_dt.acc_id and
              kob_acc_kt = a_kt.acc_id and
              a_dt.acc_from_ab_id = :rep_dt_ab_id and
              a_kt.acc_from_ab_id = :rep_kt_ab_id and
              kob_from_ol_id = ol_id and
              kob_detail_id = obd_id and
              (:acc_mask_dt = '' or
              a_dt.acc_code like :acc_mask_dt) and
              (:acc_mask_kt = '' or
              a_kt.acc_code like :acc_mask_kt) and
              (kob_summa <> 0 or
              kob_summa_val <> 0 or
              kob_summa_tax_dt <> 0) and
              (kob_acc_dt = rep_acc_id or
              kob_acc_kt = rep_acc_id)
        order by kob_ol_date, kob_from_ol_id, kob_id
        into :st_kob_id, :st_date, :st_dt_acc_id, :st_kt_acc_id, :st_ob_detail, :st_oper_memo, :st_amount_sum,
             :st_amount_val, :st_amount_tax_dt, :st_amount_tax_kt
    do
    begin
      st_line_no = st_line_no + 1;
      st_dt_acc_code = f_fmt_acc(st_dt_acc_id);
      st_kt_acc_code = f_fmt_acc(st_kt_acc_id);

      st_dt_acc_name = '';
      select acc_name
      from ref_account
      where acc_id = :st_dt_acc_id
      into st_dt_acc_name;

      st_kt_acc_name = '';
      select acc_name
      from ref_account
      where acc_id = :st_kt_acc_id
      into st_kt_acc_name;

      st_amount_tax = st_amount_tax_dt;
      suspend;
    end
  end
end
