declare variable vrem double precision;
declare variable circle integer;
declare variable sx integer;
declare variable period_month decimal(18,4);
declare variable curr_rate decimal(18,4);
begin
-- knclient_fix

  select cr_rate
  from opercurr
  where cr_from_cu_id = '840' and
        cr_date <= :rep_date_end and
        cr_date_end >= :rep_date_end rows 1
  into :curr_rate;
  if (curr_rate is null) then
    curr_rate = 1;
  for select aa_office_populate_name, aa_office_stat_name, aa_by_deliq_days, aa_curr_branch_name, aa_curr_office_name,
             aa_curr_kc_fio, aa_loan_kind, aa_loan_common_name, aa_kc_stage, aa_loan_currency, kc_amount_approved,
             aa_kont_amount_val, aa_amount_active_begin, aa_kolich_active_begin, aa_kolich_cl_active_begin,
             aa_amount_active_end, aa_kolich_active_end, aa_kolich_cl_active_end, aa_amount_disb, aa_kolich_disb,
             aa_kolich_cl_disb, aa_repay_all_os, aa_repay_all_perc, aa_repay_all_fine, aa_repay_all_kommis,
             aa_repay_writeoff_os, aa_repay_writeoff_perc, aa_repay_writeoff_fine,

             aa_deliq_os_all, aa_deliq_perc_all,

             aa_deliq_balance, aa_deliq_kolich, aa_deliq_cl_kolich,

             aa_deliq_balance7, aa_deliq_kolich7, aa_deliq_cl_kolich7, aa_deliq_os7, aa_deliq_perc7,

             aa_deliq_balance30, aa_deliq_kolich30, aa_deliq_cl_kolich30, aa_deliq_os30, aa_deliq_perc30,

             aa_deliq_balance60, aa_deliq_kolich60, aa_deliq_cl_kolich60, aa_deliq_os60, aa_deliq_perc60,

             aa_deliq_balance90, aa_deliq_kolich90, aa_deliq_cl_kolich90, aa_deliq_os90, aa_deliq_perc90,

             aa_deliq_balance180, aa_deliq_kolich180, aa_deliq_cl_kolich180, aa_deliq_os180, aa_deliq_perc180,

             aa_deliq_balance360, aa_deliq_kolich360, aa_deliq_cl_kolich360, aa_deliq_os360, aa_deliq_perc360,

             aa_amount_sh_os, /* coll_plan_os */
             aa_repay_normonth_os, /* coll_fact_os */
             kc_circle, c_sex, aa_period,

             aa_rest_norm_bal, aa_rest_norm_kol, aa_rest_deliq_bal, aa_rest_deliq_kol, aa_rest_period_bal,
             aa_rest_period_kol, aa_perc_annual_amount, aa_no_rest_deliq_30, aa_perc_annual_effect, aa_loan_kind_new

      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', :rep_period_kind, 0, 1, 1, 0, 0, 0), knclient, clients
      where aa_k_id = kc_from_k_id and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null and
            kc_from_c_id = c_id
      into :aarep_office_populate_name, :aarep_office_stat_name, :aarep_by_deliq_days, :aarep_name_1, :aarep_name_2,
           :aarep_name_3, :aarep_name_4, :aarep_name_5, :aarep_name_7, :aarep_loan_currency, :aarep_client_amount,
           :aa_loan_amount, :aarep_active_balance_b, :aarep_active_kolich_kontract_b, :aarep_active_kolich_client_b,
           :aarep_active_balance_e, :aarep_active_kolich_kontract_e, :aarep_active_kolich_client_e, :aarep_disb_amount,
           :aarep_disb_kolich_kontract, :aarep_disb_kolich_client, :aarep_trans_os, :aarep_trans_perc,
           :aarep_trans_fine, :aarep_trans_kommis, :aarep_writeoff_os, :aarep_writeoff_perc, :aarep_writeoff_fine,

           :aarep_deliq_os, :aarep_deliq_perc,

           :aarep_deliq_balance, :aarep_deliq_kolich_kontract, :aarep_deliq_kolich_client,

           :aarep_deliq_balance7, :aarep_deliq_kolich_kontract7, :aarep_deliq_kolich_client7, :aarep_deliq_os7,
           :aarep_deliq_perc7,

           :aarep_deliq_balance30, :aarep_deliq_kolich_kontract30, :aarep_deliq_kolich_client30, :aarep_deliq_os30,
           :aarep_deliq_perc30,

           :aarep_deliq_balance60, :aarep_deliq_kolich_kontract60, :aarep_deliq_kolich_client60, :aarep_deliq_os60,
           :aarep_deliq_perc60,

           :aarep_deliq_balance90, :aarep_deliq_kolich_kontract90, :aarep_deliq_kolich_client90, :aarep_deliq_os90,
           :aarep_deliq_perc90,

           :aarep_deliq_balance180, :aarep_deliq_kolich_kontract180, :aarep_deliq_kolich_client180, :aarep_deliq_os180,
           :aarep_deliq_perc180,

           :aarep_deliq_balance360, :aarep_deliq_kolich_kontract360, :aarep_deliq_kolich_client360, :aarep_deliq_os360,
           :aarep_deliq_perc360,

           :aarep_coll_plan_os, :aarep_coll_fact_os, :circle, :sx, :period_month, :aarep_rest_norm_bal,
           :aarep_rest_norm_kol, :aarep_rest_deliq_bal, :aarep_rest_deliq_kol, :aarep_rest_period_bal,
           :aarep_rest_period_kol, :aarep_perc_annual_amount, :aarep_no_rest_deliq_30, :aarep_perc_annual_effect, :aa_loan_kind_new

  do
  begin
    if (aa_loan_amount <> 0) then
    begin
      vrem = aarep_client_amount / aa_loan_amount;
    end
    else
      vrem = 0;

    if ((aarep_client_amount / curr_rate) <= 500) then
      aarep_by_loan_amount = 'A до 500$';
    else
    if ((aarep_client_amount / curr_rate) <= 1000) then
      aarep_by_loan_amount = 'B 500 до 1000$';
    else
    if ((aarep_client_amount / curr_rate) <= 2500) then
      aarep_by_loan_amount = 'C 1000 до 2500$';
    else
    if ((aarep_client_amount / curr_rate) <= 5000) then
      aarep_by_loan_amount = 'D 2500 до 5000$';
    else
    if ((aarep_client_amount / curr_rate) <= 10000) then
      aarep_by_loan_amount = 'E 5000 до 10000$';
    else
      aarep_by_loan_amount = 'F больше 10000$';

    if (circle is null) then
      circle = 0;
    if (circle <= 1) then
      aarep_circle = 'Новый кредит';
    else
      aarep_circle = 'Повторный кредит';

    if (sx = 0) then
      aarep_sex = 'Женщина';
    if (sx = 1) then
      aarep_sex = 'Мужчина';
    if (sx = 2) then
      aarep_sex = 'Юр.лицо';

    if (period_month <= 3) then
      aarep_period_str = '(00-03] мес';
    else
    if (period_month <= 6) then
      aarep_period_str = '(03-06] мес';
    else
    if (period_month <= 12) then
      aarep_period_str = '(06-12] мес';
    else
    if (period_month <= 24) then
      aarep_period_str = '(12-24] мес';
    else
    if (period_month <= 36) then
      aarep_period_str = '(24-36] мес';
    else
      aarep_period_str = 'более 36 мес.';

    aarep_active_balance_b = cast(aarep_active_balance_b * vrem as decimal(18,2));
    aarep_active_balance_e = cast(aarep_active_balance_e * vrem as decimal(18,2));
    aarep_disb_amount = cast(aarep_disb_amount * vrem as decimal(18,2));

    if (aarep_active_kolich_kontract_b <> 0) then
      aarep_active_kolich_kontract_b = 1;
    if (aarep_active_kolich_kontract_b <> 0) then
      aarep_active_kolich_kontract_b = 1;
    if (aarep_active_kolich_client_b <> 0) then
      aarep_active_kolich_client_b = 1;
    if (aarep_active_kolich_kontract_e <> 0) then
      aarep_active_kolich_kontract_e = 1;
    if (aarep_active_kolich_client_e <> 0) then
      aarep_active_kolich_client_e = 1;
    if (aarep_disb_kolich_kontract <> 0) then
      aarep_disb_kolich_kontract = 1;
    if (aarep_disb_kolich_client <> 0) then
      aarep_disb_kolich_client = 1;
    aarep_trans_os = cast(aarep_trans_os * vrem as decimal(18,2));
    aarep_trans_perc = cast(aarep_trans_perc * vrem as decimal(18,2));
    aarep_trans_fine = cast(aarep_trans_fine * vrem as decimal(18,2));
    aarep_trans_kommis = cast(aarep_trans_kommis * vrem as decimal(18,2));
    aarep_writeoff_os = cast(aarep_writeoff_os * vrem as decimal(18,2));
    aarep_writeoff_perc = cast(aarep_writeoff_perc * vrem as decimal(18,2));
    aarep_writeoff_fine = cast(aarep_writeoff_fine * vrem as decimal(18,2));
    aarep_deliq_os = cast(aarep_deliq_os * vrem as decimal(18,2));
    aarep_deliq_perc = cast(aarep_deliq_perc * vrem as decimal(18,2));
    aarep_deliq_balance = cast(aarep_deliq_balance * vrem as decimal(18,2));
    if (aarep_deliq_kolich_kontract <> 0) then
      aarep_deliq_kolich_kontract = 1;
    if (aarep_deliq_kolich_client <> 0) then
      aarep_deliq_kolich_client = 1;

    aarep_deliq_balance7 = cast(aarep_deliq_balance7 * vrem as decimal(18,2));
    if (aarep_deliq_kolich_kontract7 <> 0) then
      aarep_deliq_kolich_kontract7 = 1;
    if (aarep_deliq_kolich_client7 <> 0) then
      aarep_deliq_kolich_client7 = 1;
    aarep_deliq_os7 = cast(aarep_deliq_os7 * vrem as decimal(18,2));
    aarep_deliq_perc7 = cast(aarep_deliq_perc7 * vrem as decimal(18,2));

    aarep_deliq_balance30 = cast(aarep_deliq_balance30 * vrem as decimal(18,2));
    if (aarep_deliq_kolich_kontract30 <> 0) then
      aarep_deliq_kolich_kontract30 = 1;
    if (aarep_deliq_kolich_client30 <> 0) then
      aarep_deliq_kolich_client30 = 1;
    aarep_deliq_os30 = cast(aarep_deliq_os30 * vrem as decimal(18,2));
    aarep_deliq_perc30 = cast(aarep_deliq_perc30 * vrem as decimal(18,2));

    aarep_deliq_balance60 = cast(aarep_deliq_balance60 * vrem as decimal(18,2));
    if (aarep_deliq_kolich_kontract60 <> 0) then
      aarep_deliq_kolich_kontract60 = 1;
    if (aarep_deliq_kolich_client60 <> 0) then
      aarep_deliq_kolich_client60 = 1;
    aarep_deliq_os60 = cast(aarep_deliq_os60 * vrem as decimal(18,2));
    aarep_deliq_perc60 = cast(aarep_deliq_perc60 * vrem as decimal(18,2));

    aarep_deliq_balance90 = cast(aarep_deliq_balance90 * vrem as decimal(18,2));
    if (aarep_deliq_kolich_kontract90 <> 0) then
      aarep_deliq_kolich_kontract90 = 1;
    if (aarep_deliq_kolich_client90 <> 0) then
      aarep_deliq_kolich_client90 = 1;
