CREATE PROCEDURE AAREP_KIB_MISTAKE(
    REP_REGION_ID INTEGER)
RETURNS (
    REG_NAME VARCHAR(50),
    KAL_NEW_KF_NAME VARCHAR(100),
    F_1 DATE,
    KKR_SEND_RESULT VARCHAR(255))
AS
begin
  for select reg_name, kal_new_kf_name, cast(kkr_send_time as date), kkr_send_result
      from kibkredit, kontract, knaktlist, region, region_list(:rep_region_id, 1, 3, 1)
      where kkr_send_result <> '' and
            kkr_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            kal_new_reg_id = reg_unicode and
            (id_list = kal_new_reg_id)
      order by kal_new_kf_name
      into :reg_name, :kal_new_kf_name, :f_1, :kkr_send_result
  do
  begin
    suspend;
  end
end
