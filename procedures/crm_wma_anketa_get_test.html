CREATE PROCEDURE CRM_WMA_ANKETA_GET_TEST(
    WMA_ID INTEGER)
RETURNS (
    STATUS STR_100,
    BLOCKED_DATETIME D_DATETIME,
    CREATE_DATETIME D_DATETIME,
    STATUS_DATETIME D_DATETIME,
    LANGUAGE STR_255,
    FIO STR_100,
    BIRTH_DATE D_DATE,
    PHONE_NO STR_100,
    CITIZENSHIP STR_255,
    CITY ID_TYPE,
    OFFICE D_INTEGER,
    SVZ ID_TYPE,
    REFUSE STR_255,
    LAST_CHANGES_DATETIME D_DATETIME,
    DOCLIST STR_255,
    WORKSTATUS STR_255,
    LOAN_PURPOSE STR_255,
    LOAN_PRODUCT STR_255,
    LOAN_PERIOD D_SMALLINT,
    LOAN_AMOUNT D_INTEGER,
    ANKETA_MEMO D_STR_8100,
    NAME1 STR_100,
    NAME2 STR_100,
    NAME3 STR_100)
AS
declare variable mp_ema_status str_100;
declare variable loan_status str_100;
declare variable kf_name str_100;
declare variable search_method str_100;
declare variable svz_fio str_100;
declare variable svz_cell_no str_100;
declare variable disb_date date;
declare variable kf_reg_name str_100;
declare variable tkt_id id_type;
declare variable wma_date date;
declare variable ma_id integer;
declare variable k_id id_type;
declare variable k_disb_date date;
declare variable c_id id_type;
declare variable ma_is_active integer;
declare variable kf_id id_type;
declare variable ol_id id_type;
declare variable k_status integer;
declare variable ma_refuse_reason str_255;
declare variable k_refuse_reason str_255;
declare variable status_priority integer;
declare variable refuse_reason str_255;
declare variable ema_id id_type;
declare variable clt_id d_integer;
declare variable is_new_contdata d_smallint;
declare variable kontract_date date;
begin
  delete from website_miniapp_cli_check_tmp
  where 1 = 1;

  is_new_contdata = f_get_const('new_contdata');
  is_new_contdata = 1; -- процедуру переводим на новый механизм, Sanjari, 21.12.2019

  for select wma_id, wma_date,
             iif(wmat_name = '', '[Анкета в процессе заполнения]', wmat_name),
             rd_c.rd_code as citizenship, rd_l.rd_code as language, wma_phone_no, wma_create_datetime, wmat_city_id, trim((
             case wma_status
               when -1 then 'Закрытa'
               when 0 then 'Новая'
               when 1 then 'Перезвонить'
               when 2 then 'Заявка Передана в офис'
             end)),
             reg.reg_unicode, iif(wma_status >= 0, wmt.wmat_anketa_memo, wmc.wmc_memo), wma.wma_from_tkt_id,
             rd_lp.rd_name as loan_purpose, wma.wma_from_c_id, wmt.wmat_birthday, wmat_doclist,
             rd_w.rd_code as work_status, rd_prod.rd_code as wmat_loan_product, wmat_loan_period, wmat_loan_amount
      from cc_website_miniapp wma
      inner join cc_website_miniapp_tmp wmt on wmt.wmat_id = wma.wma_last_wmat_id
      left join cc_website_miniapp_close wmc on wmc.wmc_from_wma_id = wma.wma_id
      left join refdata rd_c on rd_c.rd_id = wmt.wmat_citizenship
      left join refdata rd_l on rd_l.rd_id = wmat_language
      left join region reg on reg.reg_unicode = wmt.wmat_office
      left join refdata rd_close on rd_close.rd_id = wmc.wmc_closed_rd_id
      left join refdata rd_lp on rd_lp.rd_id = wmt.wmat_loan_purpose
      left join refdata rd_w on wmat_work_status = rd_w.rd_id
      left join refdata rd_prod on wmat_loan_product = rd_prod.rd_id
      where --wma_date between :rep_date_begin and :rep_date_end and
            wma.wma_status between -1 and 2 and
            wma_id = :wma_id
      order by wma_id
      into :wma_id, :wma_date, :fio, :citizenship, :language, :phone_no, :create_datetime, :city, :refuse, :office,
           :anketa_memo, :tkt_id, :loan_purpose, :c_id, :birth_date, doclist, workstatus, loan_product, loan_period,
           loan_amount
  do
  begin
    mp_ema_status = null;

    ema_id = null;
    blocked_datetime = null;

    select ema.ema_id,
           (
           case ema.ema_status
             when -1 then 'Отказ'
             when 0 then  iif(ema_lo_id is null,'Не заблокирована ВИПС', 'Заблокирована ВИПС')
             when 1 then 'Обработан'
           end),
           ema_block_datetime
    from extminiapp ema
    left join clients cl on cl.c_id = ema.ema_lo_id
    where ema.ema_from_tkt_id = :tkt_id
    into :ema_id, :mp_ema_status, blocked_datetime;

    if (c_id is not null) then
    begin
      update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method, wmc_status_priority)
      values (:wma_id, :c_id, 'Определен на уровне КЦ', 999)
      matching (wmc_esm_id, wmc_c_id);
    end
    else
    begin
      -- осуществляем поиск контрагента по ФИО+дата рождения, номеру телефона -->>

      -- поиск по фио и дате рождения
      for select cl.c_id
          from clients cl
          where cl.c_birth_date = :birth_date and
                cl.c_name_fio = :fio
          into :c_id
      do
      begin
        update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method, wmc_status_priority)
        values (:wma_id, :c_id, 'Найден по фио и дате рождения', 999)
        matching (wmc_esm_id, wmc_c_id);
      end

      -- поиск по телефону из  заявки
      if (phone_no <> '') then
      begin
        if (is_new_contdata = 0) then
        begin
          for select distinct opr.opr_object_id
              from objectprops opr, objectvalues ov, refdata
              where opr_subject_kind = -1 and
                    opr_subject_id = rd_id and
                    rd_code in ('ContactInfo.Cell_Main', 'ContactInfo.Cell_Add') and
                    opr_id = ovs_from_opr_id and
                    hash(lower(ovs_value)) = hash(lower(:phone_no)) and
                    ovs_value = :phone_no and
                    ov.ovs_date_begin >= '01.01.1900'
              into :c_id
          do
          begin
            update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method, wmc_status_priority)
            values (:wma_id, :c_id, 'Найден по телефону, из заявки с сайта', 999)
            matching (wmc_esm_id, wmc_c_id);
          end
        end
        else
        begin
          clt_id = f_cont_id(phone_no, 'null');
          if (clt_id is not null) then
          begin
            for select distinct cda_from_c_id
                from contdata
                where cda_is_delete = 0 and
                      cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
                      cda_from_clt_id = :clt_id
                into c_id
            do
            begin
              update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method,
                                                                   wmc_status_priority)
              values (:wma_id, :c_id, 'Найден по телефону, из заявки с сайта', 999)
              matching (wmc_esm_id, wmc_c_id);
            end
          end
        end
      end
    end

    loan_status = '';
    ma_id = null;
    ma_is_active = null;
    kf_id = null;
    kf_name = null;
    k_id = null;
    k_status = null;
    ol_id = null;
    search_method = '';
    svz_fio = null;
    svz_cell_no = null;
    disb_date = null;
    kf_reg_name = null;

    -- по найденным контрагентам осуществляем поиск на предмет получения займа после оформления заявки с сайта
    for select mcc.wmc_c_id, mcc.wmc_method
        from website_miniapp_cli_check_tmp mcc
        where mcc.wmc_esm_id = :wma_id
        into :c_id, :search_method
    do
    begin
      loan_status = '';
      ma_id = null;
      ma_is_active = null;
      kf_id = null;
      kf_name = null;
      k_id = null;
      k_status = null;
      ol_id = null;
      svz_fio = null;
      svz_cell_no = null;
      disb_date = null;
      kf_reg_name = null;
      ma_refuse_reason = null;
      k_refuse_reason = null;

      name1 = '';
      name2 = '';
      name3 = '';
      select fl_name1, fl_name2, fl_name3
      from clients
      left join fiolist on fl_id = c_from_fl_id
      where c_id = :c_id rows 1
      into :name1, name2, name3;

      for select maa.ma_id, maa.ma_is_active, maa.ma_from_kf_id, maa.ma_date_oper, maa.ma_date_app, krs.kr_text,
                 ma_date_save
          from miniapp maa
          inner join miniappclient macl on maa.ma_id = macl.mac_from_ma_id
          left join kontreslist krl on krl.krl_ma_id = maa.ma_id
          left join kontres krs on krs.kr_id = krl.krl_from_kr_id
          where maa.ma_date_app >= :wma_date and
                maa.ma_date_end = '31.12.9999' and
                macl.mac_amount > 0 and
                macl.mac_from_c_id = :c_id
          into :ma_id, :ma_is_active, :kf_id, :disb_date, :kontract_date, :ma_refuse_reason, status_datetime
      do
      begin
        loan_status = 'В процессе ';  -- Кредитная заявка
        status_priority = 3;
        if (ma_refuse_reason is null and
            ma_is_active < 1) then
        begin
          select k_id, kr_text
          from kontract
          left join kontreslist on krl_from_k_id = k_id
          left join kontres on krl_from_kr_id = kr_id
          where k_from_ma_app_id = :ma_id rows 1
          into :k_id, :ma_refuse_reason;

          if (k_id is null) then
          begin
            select k_id, kr_text
            from knclient
            inner join kontract on k_id = kc_from_k_id
            left join kontreslist on krl_from_k_id = k_id
            left join kontres on krl_from_kr_id = kr_id
            where kc_from_c_id = :c_id and
                  k_kontract_date >= :kontract_date and
                  krl_date_old >= :kontract_date rows 1
            into :k_id, :ma_refuse_reason;
          end
        end

        select kal.kal_new_kf_name, cl.c_name_fio, reg_name,
               f_cont_info('clients', kal.kal_new_kf_officer, 'cell_main', 'ok'), kal_new_kf_officer
        from knaktlist kal
        inner join clients cl on kal.kal_new_kf_officer = cl.c_id
        inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
        where kal.kal_from_kf_id = :kf_id and
              kal.is_delete = 0 and
              current_date between kal.kal_date and kal.kal_date_end -- беру именно на текущую дату
        into :kf_name, :svz_fio, :kf_reg_name, :svz_cell_no, svz;

        if (ma_is_active < 1) then
        begin
          loan_status = 'Отказ';  --Отказ на уровне кредитной заявки
          status_priority = 4;
        end
        else
        begin
          k_id = null;
          ol_id = null;
          k_disb_date = null;

          select kont.k_id, kont.k_status, kont.k_disb_date, ol.ol_id, krs.kr_text
          from kontract kont
          left join operlist ol on ol.ol_id = kont.k_fact_disb_ol_id and
                ol.is_delete = 0
          left join kontreslist krl on krl.krl_from_k_id = kont.k_id
          left join kontres krs on krs.kr_id = krl.krl_from_kr_id
          where kont.k_from_ma_app_id = :ma_id rows 1
          into :k_id, :k_status, :k_disb_date, :ol_id, :k_refuse_reason;

          if (k_id is not null) then
          begin
            loan_status = 'В процессе';
            status_priority = 2;

            if (k_status = -1) then
            begin
              loan_status = 'Отказ'; --Отказ на уровне кредитной заявки
              status_priority = 4;
            end
            else
            begin
              if (ol_id is not null) then
              begin
                loan_status = 'Выдан';
                status_priority = 1;
              end
            end
            disb_date = k_disb_date;
          end
        end

        refuse_reason = coalesce(ma_refuse_reason, '');
        if (refuse_reason = '') then
          refuse_reason = k_refuse_reason;

        insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method, wmc_status, wmc_status_priority,
                                                   wmc_refuse_reason, wmc_kf_name, wmc_svz_fio, wmc_svz_cell_no,
                                                   wmc_disb_date, wmc_reg_name)
        values (:wma_id, :c_id, :search_method, :loan_status, :status_priority, :refuse_reason, :kf_name, :svz_fio,
                :svz_cell_no, :disb_date, :kf_reg_name);
      end
    end

    if (ma_id is null) then
    begin
      c_id = null;
      search_method = '';

      if (ema_id is null) then
      begin
        status = :refuse;

      end
      else
      begin
        status = mp_ema_status;
      end
      suspend;
    end
    else
    begin
      for select wmc_method, wmc_status, wmc_kf_name, wmc_svz_fio, wmc_svz_cell_no, wmc_disb_date, wmc_reg_name
          from website_miniapp_cli_check_tmp
          where wmc_esm_id = :wma_id
          order by wmc_status_priority, wmc_disb_date
          rows 1
          into :search_method, :loan_status, :kf_name, :svz_fio, :svz_cell_no, :disb_date, :kf_reg_name

      do
      begin
        status = :loan_status;

        suspend;
      end
    end
  end
end
