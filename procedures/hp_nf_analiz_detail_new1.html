CREATE PROCEDURE HP_NF_ANALIZ_DETAIL_NEW1(
    FILE_NAME D_STR_100,
    FILE_TEXT D_BLOB_TEXT,
    UN_ID D_ID)
RETURNS (
    IFA_ID D_INTEGER,
    RES_TEXT D_STR_255)
AS
declare variable main_bnk d_str_20;
declare variable sub_bnk d_str_20;
declare variable main_bnk_id d_integer;
declare variable sub_bnk_id d_integer;
declare variable val1 d_currency;
declare variable val2 d_currency;
declare variable val3 d_currency;
declare variable val4 d_currency;
declare variable val5 d_currency;

declare variable num_pos d_integer;
declare variable num_str d_integer;
declare variable res_str d_str_8100;
declare variable row_str d_str_8100;
declare variable val_str d_str_8100;
declare variable dt_str d_str_10;
declare variable dt_file d_date;
declare variable run_susp boolean;
declare variable do_exit boolean;
declare variable num_step d_integer;
declare variable rec_no d_integer;
declare variable delim d_str_lang;
begin
  num_str = 0;
  rec_no = 0;
  run_susp = false;
  delim = ascii_char(13) || ascii_char(10);
  for
    select
      res_str
    from hp_split_str(:file_text, :delim)
  into
    :res_str
  do
  begin
    num_str = num_str + 1;
    row_str = trim(res_str);

    if (num_str = 3) then
    begin
      main_bnk = row_str;
      main_bnk_id = (select ib_id from ipc_banks where ib_bank_code = :main_bnk);
    end

    if (num_str = 4) then
    begin
      dt_str = left(row_str,  10);
      dt_file = right(dt_str, 2) || '.' || substring(dt_str from 6 for 2) || '.' || left(dt_str, 4);
      execute procedure hp_nf_analiz_new (:file_name, :dt_file, :file_text, :un_id)
        returning_values (:ifa_id, :res_text);
    end

    if (num_str >= 9) then
    begin
      run_susp = not (left(row_str, 1) in ('-', '+', '!') or (left(row_str, 5) = 'total'));
      if (run_susp) then
      begin
        rec_no = rec_no + 1;
        val_str = row_str;
        do_exit = false;
        num_step = 0;
        while (not do_exit) do
        begin
          num_step = num_step + 1;
          num_pos = position(' ', row_str);
          if (num_pos > 1) then
          begin
            val_str = left(row_str, num_pos - 1);
            row_str = trim(right(row_str, char_length(row_str) - num_pos));
            if (num_step = 1) then
            begin
              sub_bnk = val_str;
              sub_bnk_id = (select ib_id from ipc_banks where ib_bank_code = :sub_bnk);
            end else
            if (num_step = 2) then
              val1 = val_str;
            else
            if (num_step = 3) then
              val2 = val_str;
            else
            if (num_step = 4) then
              val3 = val_str;
            else
            if (num_step = 5) then
              val4 = val_str;
            else
            if (num_step = 6) then
              val5 = val_str;
          end else
            do_exit = true;
        end
      end
    end

    if (run_susp) then
      insert into ipc_nonfin_detail (
        ind_from_ifa_id,
        ind_rec_no,
        ind_main_bank_id,
        ind_sub_bank_id,
        ind_recieved_tran,
        ind_must_pay_tran,
        ind_netto_count_trans,
        ind_netto_amount
      )
      values (
        :ifa_id,
        :rec_no,
        :main_bnk_id,
        :sub_bnk_id,
        abs(:val2),
        abs(:val4),
        0,
        0
      );
  end
  suspend;
end
