CREATE PROCEDURE PRINT_SUD_LAWSUIT_CL(
    OBJ_ID D_ID)
RETURNS (
    C_FIO D_STR_100,
    ADDRESS_YUR D_STR_255,
    C_BIRTH_DATE D_DATE,
    CITIZEN D_STR_255,
    C_PASS_NO D_STR_CODE,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE D_DATE,
    C_INN D_STR_20,
    C_SNILS D_STR_100)
AS
declare variable c_id d_id;
begin
  for select c_id, c_name_fio, c_birth_date, c_pass_no, c_pass_who, c_pass_date, rd_name, c_inn, slc_address
  from sud_loanclient, knclient, clients, refdata
  where slc_from_smr_id = :obj_id
    and kc_id = slc_from_kc_id
    and c_id = kc_from_c_id
    and rd_id = c_citizen
  into c_id, c_fio, c_birth_date, c_pass_no, c_pass_who, c_pass_date, citizen, c_inn, address_yur
  do
  begin
    c_snils = null;

    select cds_doc_no
    from clientdocs, passkind, passdoctype
    where cds_from_c_id = :c_id
      and pk_id = cds_from_pk_id
      and pdt_id = pk_from_pdt_id
      and pdt_code = 'snils'
      and cds_is_delete = 0
    order by cds_datetime desc
    rows 1
    into c_snils;

    suspend;
  end
end
