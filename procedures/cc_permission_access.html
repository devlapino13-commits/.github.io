CREATE PROCEDURE CC_PERMISSION_ACCESS(
    UN_NAME STR_100,
    UN_ID ID_TYPE)
RETURNS (
    RES_UN_ID ID_TYPE,
    PA_ID INTEGER,
    PA_NAME STR_100,
    PA_COMMAND_NAME STR_100)
AS
declare variable pa_id_parent integer;
begin
  un_id = coalesce(un_id, 0);
  un_name = coalesce(un_name, '');

  if (un_id <> 0) then
    res_un_id = un_id;
  else
  if (un_name <> '') then
  begin
    select un.un_id
    from username un
    where un.un_name = :un_name rows 1
    into res_un_id;
  end
  else
    exit;

  for select pa_id
      from hp_user_system_access(:res_un_id)
      where pa_command_name = 'CallCentre.Login' and
            pa_access_kind = 1
      into :pa_id_parent
  do
  begin
    for select pa.pa_id, pa.pa_name, pa.pa_command_name
        from permactions pa
        where pa.pa_parent_id = :pa_id_parent and
              pa.pa_is_delete = 0
        into :pa_id, :pa_name, :pa_command_name
    do
    begin
      if (exists(select has_access
                 from hp_permission_access_check(:res_un_id, 'PermActions', 0, :pa_command_name)
                 where has_access = 1)) then
      begin
        begin
          suspend;
        end
      end
    end
  end
end
