         and K_ID = KC_FROM_K_ID
         and KC_AMOUNT_APPROVED > 0
         and KC_LOAN_TARGET=RD_ID

    sql_text =
      'With FT_LIST as
      (select distinct C_TYPE, C_NAME_FIO, C_BIRTH_DATE, C_PASS_NO,
         C_PASS_DATE, C_PASS_WHO, C_PASS_DATE_EXPIRE,
         cl.C_PASS_WHO_CODE, C_CITIZEN, C_ID,

    into
      :c_family_str;

    where rd_id = :c_citizen

    if (c_pft_risk_koef = 0) then

    begin

    if (c_pft_risk_koef = 2) then

    if (qty = 0) then
    begin
      qty = qty + 1;

      inner join hr_pstatus hps on hps.ps_from_c_id = cl.c_id
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      where kal.kal_from_kf_id = :kf_id and
            kal.is_delete = 0 and
            :print_date between kal.kal_date and kal.kal_date_end and
            :print_date between ps_date_begin and ps_date_end and
            ps_is_working = 1
      into :ank_filled_fio, :ank_filled_pos, :ank_filled_inn;

      ank_data_in_db_fio = null;

      if (is_update = 0) then
      begin
        select cl.c_name_fio, jl.jl_name, maa.ma_manager_id
        from miniapp maa
        inner join username un on un.un_id = ma_from_un_id
        inner join clients cl on cl.c_id = un.un_from_c_id
        inner join hr_pstatus hps on hps.ps_from_c_id = cl.c_id
        inner join joblist jl on jl.jl_id = ps_from_jl_id and
              maa.ma_date_end = '31.12.9999'
        where maa.ma_id = :ma_app_id and
              :k_disb_date between ps_date_begin and ps_date_end -- <<-- k_disb_date
              and
              ps_is_working = 1
        into :ank_data_in_db_fio, :ank_data_in_db_pos, :ma_manager_id;
      end
      else
      begin
        select cl.c_name_fio, jl.jl_name, maa.ma_manager_id
        from miniapp maa
        inner join username un on un.un_id = ma_from_un_id
        inner join clients cl on cl.c_id = un.un_from_c_id
        inner join hr_pstatus hps on hps.ps_from_c_id = cl.c_id
        inner join joblist jl on jl.jl_id = ps_from_jl_id and
              maa.ma_date_end = '31.12.9999'
        where maa.ma_id = :ma_app_id and
              :print_date between ps_date_begin and ps_date_end -- <<-- printe_date
              and
              ps_is_working = 1
        into :ank_data_in_db_fio, :ank_data_in_db_pos, :ma_manager_id;

        ank_data_in_db_fio = coalesce(ank_data_in_db_fio, '');

        if (ank_data_in_db_fio = '') then
        begin
          select
            cl.c_name_fio,
            jl.jl_name
          from hr_pstatus hps
            inner join joblist jl
              on jl.jl_id = ps_from_jl_id
            inner join jobkind jk
              on jk.jk_id = jl.jl_kind
            inner join clients cl
              on hps.ps_from_c_id = cl.c_id
          where :print_date between ps_date_begin and ps_date_end
            and hps.ps_from_reg_id = :k_last_reg_id
            and jk.jk_system_code = 'JobSystem.Specialist_KOBD'
          rows 1
          into
            :ank_data_in_db_fio,
            :ank_data_in_db_pos;
        end
      end

      if (ma_manager_id is not null) then begin
        select
          cl.c_name_fio,
          jl.jl_name
        from clients cl
          inner join hr_pstatus hps
            on hps.ps_from_c_id = cl.c_id
          inner join joblist jl
            on jl.jl_id = ps_from_jl_id
        where :print_date between ps_date_begin and ps_date_end
          and ps_is_working = 1
          and cl.c_id = :ma_manager_id
        into
          :ank_info_checked_fio,
          :ank_info_checked_pos;
      end else begin
        select
          cl.c_name_fio,
          jl.jl_name
        from hr_pstatus hps
          inner join joblist jl
            on jl.jl_id = ps_from_jl_id
          inner join jobkind jk
            on jk.jk_id = jl.jl_kind
          inner join clients cl
            on hps.ps_from_c_id = cl.c_id
        where :print_date between ps_date_begin and ps_date_end
          and hps.ps_from_reg_id = :k_last_reg_id
          and jk.jk_system_code = 'JobSystem.Region_KM'
        rows 1
        into
          :ank_info_checked_fio,
          :ank_info_checked_pos;
      end
    end

    pant_type = null;
    pant_doc_no = null;
    pant_date_begin = null;
    pant_date_end = null;
    pant_issued_by = null;
    pant_memo_1 = null;
    pant_memo_2 = null;
    select
      pk.pk_name,
      cd.cds_doc_no,
      cd.cds_doc_date_begin,
      cd.cds_doc_date_end,
      cd.cds_doc_issued_by,
      cd.cds_memo_1,
      cd.cds_memo_2
    from clientdocs cd
      inner join passkind pk
        on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt
        on pdt.pdt_id = pk.pk_from_pdt_id
    where cds_from_c_id = :kc_from_c_id
      and :k_disb_date between cds_doc_date_begin and cds_doc_date_end
      and cds_is_delete = 0
      and pdt.pdt_code in ('work_patent', 'License')
    order by cds_doc_date_end desc rows 1
    into
      :pant_type,
      :pant_doc_no,
      :pant_date_begin,
      :pant_date_end,
      :pant_issued_by,
      :pant_memo_1,
      :pant_memo_2;

    -- Анкетные данные получаем поле Инд.Предприниматель(Да/Нет)
    ind_entrepreneur_status = null;
    if (db_id in (39801, 41701)) then
    begin
      select
        iif(res_str_values = 'Да', 1, 0)
      from hp_objectprops_anketa_get('Clients', :kc_from_c_id, -1, 'Ind_Entrepreneur', current_date, null)
      into :ind_entrepreneur_status;
      ind_entrepreneur_status = coalesce(ind_entrepreneur_status, 0);
    end

    -- Публичное должностное лицо
    if (exists(
      select *
      from clients c
        inner join clientbenefic cb
          on c.c_id = cb.cb_ben_from_c_id
      where cb.Cb_from_c_id = :kc_from_c_id
        and coalesce(c.c_is_public, 0) = 1))
    then check_public_face = 1;
      else check_public_face = 0;

    -- Бенефициарный владелец
    if (exists(
      select *
      from clientbenefic
      where cb_from_c_id = :kc_from_c_id
        and cb_is_owner = 1))
    then beneficial_owner = 1;
      else beneficial_owner = 0;
