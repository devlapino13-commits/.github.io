CREATE PROCEDURE UXP$UXBUGTEST_INFO(
    BTS_UUID D_UUID)
RETURNS (
    BTS_UA_UUID D_UUID,
    BTS_UA_PARAMS D_STR_8100,
    BTS_UA_VER_KIND D_STR_CODE,
    BTS_OPEN_UX_UUID D_UUID)
AS
declare variable bts_open_kind d_smallint;
declare variable bts_open_ua_uuid d_uuid;
begin
  for select bts_ua_uuid,
             bts_ua_params,
             bts_ua_ver_kind,
             bts_open_ua_uuid,
             bts_open_kind,
             uxa_owner
      from ux$bugtest, ux$action, ux$alias
      where bts_uuid = :bts_uuid and
            bts_open_ua_uuid = ua_uuid and
            ua_uxa_uuid = uxa_uuid
      into :bts_ua_uuid,
           :bts_ua_params,
           :bts_ua_ver_kind,
           :bts_open_ua_uuid,
           :bts_open_kind,
           :bts_open_ux_uuid
  do
  begin
    if (bts_open_kind = 0) then
      bts_open_ux_uuid = null;
    else
    if (bts_open_kind = 1) then
      bts_open_ux_uuid = bts_open_ua_uuid;

    suspend;
  end
end
