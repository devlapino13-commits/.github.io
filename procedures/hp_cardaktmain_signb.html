CREATE PROCEDURE HP_CARDAKTMAIN_SIGNB(
    CAM_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable cem_id d_integer;
declare variable err_head d_str_100;
declare variable cam_status d_smallint;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable ol_id d_id;
declare variable ol_datetime d_datetime;
declare variable cpl_id d_integer;
begin
  err_code = 0;
  err_text = '';
  err_head = 'Акт приема-передачи № ' || cam_id;

  select
    cam_status,
    c_id,
    c_name_fio
  from cardaktmain
    inner join cardplace
      on cpl_id = cam_src_cpl_id
    inner join clients
      on c_id = cpl_from_c_id
  where cam_id = :cam_id
    and cam_dst_cpl_id = 0
  into
    :cam_status,
    :c_id,
    :c_name_fio;

  if (err_code = 0 and cam_status <> 0) then
  begin
    err_code = 1;
    err_text = err_head || ' не может быть подписан.';
  end

  if (err_code = 0 and not exists(
    select * from username where un_id = :un_id and un_from_c_id = :c_id)) then
  begin
    err_code = 2;
    err_text = err_head || ', может подписать только ' || c_name_fio || '.';
  end

  -- бух.проводка
  if (err_code = 0) then
  begin
    select
      ol_id,
      err_code,
      err_text
    from hp_cardaktmain_transactionb (:cam_id)
    into
      :ol_id,
      :err_code,
      :err_text;
    ol_datetime = (select ol_datetime from operlist where ol_id = :ol_id);
  end

  -- меняем статус заявки на "Заготовки переданы"
  if (err_code = 0) then
  begin
    select
      cem_id
    from cardembomain
      inner join cardkind
        on ckd_id = cem_from_ckd_id
          and cem_from_ces_id = 3
    where cem_from_camb_id = :cam_id
    into
      :cem_id;

    if (cem_id is null) then
    begin
      err_code = 1;
      err_text = err_head || ' не может быть подписан(несоответствие статуса заявки).';
    end else
    begin
      update cardembomain set
        cem_from_ces_id = 4,
        cem_from_un_id = :un_id
      where cem_id = :cem_id;
    end
  end

  if (err_code = 0) then
  begin
    update cardaktmain set
      cam_status = 1,
      cam_from_ol_id = :ol_id,
      cam_ol_datetime = :ol_datetime
    where cam_id = :cam_id;
    err_text = err_head || ' успешно подписан.';
  end

  suspend;
  if (err_code = 1) then exception e_message err_text;
end
