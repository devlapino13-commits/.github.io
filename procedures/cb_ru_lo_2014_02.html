CREATE PROCEDURE CB_RU_LO_2014_02(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    CB_PERSONAL_NAME VARCHAR(100),
    JS_NAME VARCHAR(100),
    CB_IS_WORKING SMALLINT,
    KOL_CLI_CATEG_1_2 INTEGER,
    KOL_CLI_DISB INTEGER,
    BONUS_FACTOR INTEGER,
    CB_DELINQ_BALANCE CURR_TYPE,
    CB_ACTIV_BALANCE CURR_TYPE,
    CB_PORTFEL_C CURR_TYPE,
    CB_PORTFEL_D CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR_FACTOR CURR_TYPE,
    CB_SUM_VZYSKAN CURR_TYPE,
    BONUS_TOTAL CURR_TYPE,
    BONUS_RE CURR_TYPE,
    REG_PAR8_180 DECIMAL(15,5),
    KOL_CLI_KATEG_1_LAST INTEGER,
    KOL_CLI_KATEG_1 INTEGER,
    PRIROST_KATEG_1 INTEGER)
AS
declare variable rep_date_begin date;
declare variable cb_region_id integer;
declare variable cb_personal_id id_type;
declare variable js_code varchar(100);
declare variable rep_date_end_2 date;
declare variable cb_region_id_prior integer;
declare variable kol_cli_categ_1 integer;
declare variable is_bonus smallint;
declare variable reg_unicode integer;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  rep_date_end_2 = rep_date_begin - 1;

-- Расчет Пар на уровне регион
  delete from temp_sks_reg_info;

  for select reg_unicode
      from region
      where reg_status = 3
      into reg_unicode
  do
  begin
--  Просроченный баланс категории 1 от 8 до 180 дней
    cb_delinq_balance = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :reg_unicode and
          oks_delinq_date < :rep_date_end and
          :rep_date_end - oks_delinq_date between 8 and 180 and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1
    into cb_delinq_balance;
    if (cb_delinq_balance is null) then
      cb_delinq_balance = 0;

--  Расчет баланса КС на конец предыдущего месяца по Категории 1; B - общий портфель катег.1 на предыдущую отчетную дату
    cb_activ_balance = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end_2 between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_region_id = :reg_unicode and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end_2, null, null, null, null, null, null)) = 1
    into :cb_activ_balance;
    if (:cb_activ_balance is null) then
      cb_activ_balance = 0;

--     Расчет Пар
    if (cb_delinq_balance = 0) then
    begin
      par7 = 0;
    end
    else
    begin
      if (cb_activ_balance > 0) then
      begin
        par7 = cb_delinq_balance / cb_activ_balance * 100;
      end
      else
      begin
        par7 = 100;
      end
    end

    insert into temp_sks_reg_info (reg_id, reg_par8_180)
    values (:reg_unicode, :par7);
  end

  reg_par8_180 = 0;
  cb_region_id_prior = 0;
  for select aa_curr_kc_id, sum(aa_just_cl_end)
--,sum(iif(aa_category_id = 1,aa_just_cl_end,0)), sum(AA_JUST_CL_DISB)
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 1, 1, 1, 0, 0, 0)
      where aa_category_id in (1, 2)
      group by aa_curr_kc_id
      into cb_personal_id, kol_cli_categ_1_2 --, kol_cli_categ_1, kol_cli_disb
  do
  begin
--    добавлно за бонусы март 2014, кол-во выданных клиентов не будем брать из отчета о портфеле, отдельно будем считать
--  будем брать все выданные без параллельных и без кредитов для сотрдуников
    kol_cli_disb = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_kc_id = :cb_personal_id and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind in (1, 3) -- включаем только групповых и индивидуальных
    into kol_cli_disb;
    if (kol_cli_disb is null) then
      kol_cli_disb = 0;

-- Системная должность
    js_name = null;
    js_code = null;
    select rd_name, jk_system_code
    from hr_pstatus, joblist, jobkind, refdata
    where ps_from_c_id = :cb_personal_id and
          ps_is_working = 1 and
          ps_status = 1 and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_id = rd_id
    into js_name, js_code;
    if (js_name is null) then
      js_name = 'Не определен';
    if (js_code is null) then
      js_code = '';
    if ((js_code <> '') and
        (js_code not in ('JobSystem.Region_KM', 'JobSystem.Specialist_SB'))) then
    begin
      cb_is_working = null;
      cb_region_id = null;
--    Cтатус на конец даты,  0 - не работает; 1 - работает;
      select ps_is_working, ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = :cb_personal_id and
            ps_date_begin is not null and
            ps_status = 1 and
            :rep_date_end between ps_date_begin and ps_date_end
      into cb_is_working, cb_region_id;
      if (cb_is_working is null) then
        cb_is_working = 0;

      if (cb_region_id_prior <> cb_region_id) then
      begin
        cb_region_id_prior = cb_region_id;
        select r1.reg_code || ' ' || r1.reg_name office_name, r2.reg_code || ' ' || r2.reg_name branch_name
        from region r1, region r2
        where r1.reg_unicode = :cb_region_id and
              r1.parent = r2.reg_unicode
        into office_name, branch_name;

        reg_par8_180 = null;
        select reg_par8_180
        from temp_sks_reg_info
        where reg_id = :cb_region_id
        into reg_par8_180;
        if (reg_par8_180 is null) then
          reg_par8_180 = 0;
      end

      cb_personal_name = (select c_name_fio
                          from clients
                          where c_id = :cb_personal_id);

--  Просроченный баланс категории 1 от 8 до 180 дней
      cb_delinq_balance = null;
      select sum(oks_balance)
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_kc_id = :cb_personal_id and
            oks_delinq_date < :rep_date_end and
            :rep_date_end - oks_delinq_date between 8 and 180 and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1
      into cb_delinq_balance;
      if (cb_delinq_balance is null) then
        cb_delinq_balance = 0;
--  Расчет баланса КС на конец предыдущего месяца по Категории 1; B - общий портфель катег.1 на предыдущую отчетную дату
      cb_activ_balance = null;
      select sum(oks_balance)
      from oksummary
      where :rep_date_end_2 between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_kc_id = :cb_personal_id and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end_2, null, null, null, null, null, null)) = 1
      into :cb_activ_balance;
      if (:cb_activ_balance is null) then
        cb_activ_balance = 0;
-- С - Портфель займов, переданный в категорию 1 от другого СВЗ в течение отчетного месяца
      cb_portfel_c = null;
      select sum(iif((select aa_category_id
                      from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1,
             (select oks_balance
              from oksummary
              where oks_from_k_id = k_id and
                    oks_close_state = 0 and
                    :rep_date_end between oks_date_begin and oks_date_end), 0))
      from kontstate, kontract, knaktlist kal1, knaktlist kal2
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal1.kal_from_kf_id and
            kal1.is_delete = 0 and
            :rep_date_end between kal1.kal_date and kal1.kal_date_end and
            kal1.kal_new_kf_officer = :cb_personal_id and
            k_from_kf_id = kal2.kal_from_kf_id and
            kal2.is_delete = 0 and
            :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
            kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
            ks_date_begin < kal1.kal_date -- после передачи выданные группы не включает
      into cb_portfel_c;
      if (cb_portfel_c is null) then
        cb_portfel_c = 0;
-- D - Портфель займов, переданный в категорию 1 от данного СВЗ другому СВЗ в течение отчетного месяца
      cb_portfel_d = null;
      select sum(iif((select aa_category_id
                      from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1,
             (select oks_balance
              from oksummary
              where oks_from_k_id = k_id and
                    oks_close_state = 0 and
                    :rep_date_end between oks_date_begin and oks_date_end), 0))
      from kontstate, kontract, knaktlist kal1, knaktlist kal2
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal1.kal_from_kf_id and
            kal1.is_delete = 0 and
            :rep_date_end between kal1.kal_date and kal1.kal_date_end and
            k_from_kf_id = kal2.kal_from_kf_id and
            kal2.is_delete = 0 and
            :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
            kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
            kal2.kal_new_kf_officer = :cb_personal_id and
            ks_date_begin < kal1.kal_date -- после передачи выданные группы не включает
      into cb_portfel_d;
      if (cb_portfel_d is null) then
        cb_portfel_d = 0;

--     Расчет Пар
      if (cb_delinq_balance = 0) then
      begin
        par7 = 0;
      end
      else
      begin
        if (cb_activ_balance + cb_portfel_c - cb_portfel_d > 0) then
        begin
          par7 = cb_delinq_balance / (cb_activ_balance + cb_portfel_c - cb_portfel_d) * 100;
        end
        else
        begin
          par7 = 100;
        end
      end

      bonus_factor = case
                       when kol_cli_categ_1_2 <= 19 then 0
                       when kol_cli_categ_1_2 <= 49 then 80
                       when kol_cli_categ_1_2 <= 99 then 90
                       when kol_cli_categ_1_2 <= 149 then 110
                       when kol_cli_categ_1_2 <= 199 then 115
                       when kol_cli_categ_1_2 <= 249 then 130
                       when kol_cli_categ_1_2 <= 299 then 140
                       when kol_cli_categ_1_2 <= 349 then 145
                       when kol_cli_categ_1_2 <= 399 then 147
                       else 150
                     end;
      par_factor = case
                     when par7 <= 3 then 1.1
                     when par7 <= 7 then 1
                     else 0
                   end;
--  Сумма взыскании по Категории 2
      cb_sum_vzyskan = null;
      select -sum(iif((select aa_category_id
                       from aarep_get_category_id(k_id, ok_ol_date, null, null, null, null, null, null)) = 2, ok_perc + ok_fine, 0))
      from operkred, kontract, knaktlist
      where ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_from_c_id = 0 and
            ok_type = 4 and
            ok_perc + ok_fine <> 0 and
            ok_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            ok_ol_date between kal_date and kal_date_end and
            kal_new_kf_officer = :cb_personal_id and
            not exists(select *
                       from sb_akt, sb_list -- добавлено по приказу №302-06-13, пункт 2, не включать закрытые по СБ
                       where sba_is_delete = 0 and
                             sba_from_sbl_id <> 0 and
                             sba_status = 2 and
                             ok_ol_date between sba_date_begin and sba_date_end and
                             sba_from_k_id = k_id and
                             sba_from_sbl_id = sbl_id
--                       and sbl_type = 1
            )
      into cb_sum_vzyskan;
      if (cb_sum_vzyskan is null) then
        cb_sum_vzyskan = 0;

--          if (kol_cli_categ_1_2 < 200 and kol_cli_disb < 18) then is_bonus = 0; -- убрана с приказом от 11 июня 2014 года
--          else is_bonus = 1;
--        Расчет клиентов категории 1 на дату отчета
      kol_cli_kateg_1 = null;
      select count(ac_is_client)
      from aarep_cli_active_list(:rep_date_end)
      where ac_is_client = 1 and
            ac_kc_id = :cb_personal_id and
            (select aa_category_id
             from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) = 1
            /*and exists(select * from kontstate  -- выборка только по своим выданным кредитам
                     where ks_from_k_id = ac_kontract_id
                     and ks_ol_is_delete = 0
                     and ks_state = 1
                     and :rep_date_end between ks_date_begin and ks_date_end
                     and ks_kc_id = :cb_personal_id) */
            and
            exists(select *
                   from kontract, knaktlist -- добавлено по приказу 163-08-14 о/д от 19.08.2014
                   where ac_kontract_id = k_id and
                         k_from_kf_id = kal_from_kf_id and
                         knaktlist.is_delete = 0 and
                         :rep_date_end between kal_date and kal_date_end
--            принятые за отчетынй период не будет смотреть или будет брать только собстенные кредиты который сам СВЗ выдавал
                         and
                         (kal_date < :rep_date_begin or
                         (exists(select *
                                 from kontstate -- выборка только по своим выданным кредитам
                                 where ks_from_k_id = ac_kontract_id and
                                       ks_ol_is_delete = 0 and
                                       ks_state = 1 and
                                       :rep_date_end between ks_date_begin and ks_date_end and
                                       ks_kc_id = :cb_personal_id))) and
                         kal_new_kf_officer = :cb_personal_id)

      into kol_cli_kateg_1;
      if (kol_cli_kateg_1 is null) then
        kol_cli_kateg_1 = 0;

--        Расчет клиентов категории 1 на ПРЕДЫДУ�©УЮ дату отчета
      kol_cli_kateg_1_last = null;
      select count(ac_is_client)
      from aarep_cli_active_list(:rep_date_end_2)
      where ac_is_client = 1 and
            ac_kc_id = :cb_personal_id and
            (select aa_category_id
             from aarep_get_category_id(ac_kontract_id, :rep_date_end_2, null, null, null, null, null, null)) = 1
            /*  and exists(select * from kontstate  -- выборка только по своим выданным кредитам -- убран по приказу №163-08-14 от 19.08.2014
                     where ks_from_k_id = ac_kontract_id
                     and ks_ol_is_delete = 0
                     and ks_state = 1
                     and :rep_date_end_2 between ks_date_begin and ks_date_end
                     and ks_kc_id = :cb_personal_id) */
      into kol_cli_kateg_1_last;
      if (kol_cli_kateg_1_last is null) then
        kol_cli_kateg_1_last = 0;

      prirost_kateg_1 = kol_cli_kateg_1 - kol_cli_kateg_1_last;

      if (kol_cli_categ_1_2 < 250 and
          prirost_kateg_1 < 15) then
        is_bonus = 0;
      else
        is_bonus = 1;

      if (par7 <= 7) then
        bonus_total = kol_cli_categ_1_2 * bonus_factor * par_factor * is_bonus + cb_sum_vzyskan * 30 / 100;
      else
        bonus_total = 0;

      if (cb_personal_id = 1000000000000 or cb_is_working = 0 or cb_personal_id = 1000000000003) then
        bonus_total = 0;
      bonus_re = bonus_total;
      if (:rep_date_end between '01.05.2014' and '31.05.2014') then
      begin
        if (reg_par8_180 > 7.00000 and
            bonus_total > 20000.0000) then
          bonus_total = 20000.0000;
      end
      else
      if (:rep_date_end >= '01.06.2014') then
      begin
        if (reg_par8_180 > 7.00000 and
            bonus_total > 15000.0000) then
          bonus_total = 15000.0000;
      end
      suspend;
    end
  end

end
