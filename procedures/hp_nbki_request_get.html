CREATE PROCEDURE HP_NBKI_REQUEST_GET
RETURNS (
    NR_ID INTEGER,
    LASTNAME STR_100,
    FIRSTNAME STR_100,
    PATERNALNAME STR_100,
    GENDER INTEGER,
    PLACEOFBIRTH STR_100,
    REG_POSTAL STR_100,
    REG_CITY STR_255,
    REG_STREET STR_255,
    REG_PROV INTEGER,
    RES_CITY STR_255,
    RES_STREET STR_255,
    RES_PROV INTEGER,
    AMOUNT CURR_TYPE,
    PRIMARYID_NUMBER STR_100,
    PRIMARYID_ISSUINGDATE DATE,
    PRIMARYID_ISSUINGBY STR_100,
    REQUESTOR STR_10,
    C_BIRTH_DATE DATE,
    PRIMARYID_TYPE INTEGER,
    PRIMARYID_ISSUINGLOCATION VARCHAR(10),
    CONSENT_DATE_BEGIN DATE,
    CONSENT_DATE_END DATE)
AS
declare variable c_id id_type;
declare variable c_name_fio str_100;
declare variable c_birth_city str_100;
declare variable c_birth_country str_100;
declare variable adc_type integer;
declare variable adl_street_id id_type;
declare variable adl_from_pl_id id_type;
declare variable pl_post_index str_100;
declare variable street_post_index str_100;
declare variable plt_name str_100;
declare variable adl_post_index integer;
begin
  requestor = 'FinStart';

  for select nr_id, nr_from_c_id
      from nbki_request
      where nr_status = 0
      into :nr_id, :c_id
  do
  begin
    select cl.c_name_fio, iif(cl.c_sex = 0, 2, 1), cl.c_birth_city, c_birth_country
    from clients cl
    where cl.c_id = :c_id
    into c_name_fio, gender, c_birth_city, c_birth_country;

    select txt1, txt2, txt3
    from hp_fio_clear(:c_name_fio)
    into :lastname, :firstname, :paternalname;

    if (c_name_fio like '%уулу%') then
      lastname = lastname || ' уулу';
    if (c_name_fio like '%кызы%') then
      lastname = lastname || ' кызы';

    placeofbirth = c_birth_city;
    if (placeofbirth = '') then
    begin
      if (c_birth_country <> '') then
        placeofbirth = c_birth_country;
    end

    placeofbirth = trim(coalesce(placeofbirth, ''));

    reg_city = '';
    reg_street = '';
    reg_prov = null;
    res_city = '';
    res_street = '';
    res_prov = null;

    for select adcl.adc_address_type, adl.adl_street_id, adl.adl_from_pl_id, adl.adl_post_index
        from addrclient adcl
        inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
        where adcl.adc_from_c_id = :c_id and
              adcl.adc_address_type in (1, 2)
        into :adc_type, :adl_street_id, :adl_from_pl_id, :adl_post_index
    do
    begin
      if (adc_type = 1) then -- место регистрации
      begin
        reg_postal = :adl_post_index;

        pl_post_index = '';

        select pl.pl_name, pl.pl_post_index
        from places pl
        where pl.pl_id = :adl_from_pl_id
        into :reg_city, :pl_post_index;

        street_post_index = '';
        plt_name = null;

        select pl.pl_name, pt.plt_name, pl.pl_post_index
        from places pl
        inner join placetype pt on pl.pl_status = pt.plt_id
        where pl.pl_id = :adl_street_id
        into :reg_street, :plt_name, :street_post_index;

        street_post_index = coalesce(street_post_index, '');

        /* по умолчанию наибольший приоритет отдаю почтовому индексу, указанному на уровне улицы*/
        reg_postal = coalesce(reg_postal, ''); -- Почтовый индекс места жительства/нахождения
        if (reg_postal = '') then
        begin
          if (street_post_index <> '') then
            reg_postal = street_post_index;
          else
          if (pl_post_index <> '') then
            reg_postal = pl_post_index;
        end

        reg_prov = (case plt_name
                      when 'Аллея' then 1
                      when 'Бульвар' then 2
                      when 'Въезд' then 3
                      when 'Дорога' then 4
                      when 'Заезд' then 5

                      when 'Казарма' then 6
                      when 'Квартал' then 7
                      when 'Километр' then 8
                      when 'Кольцо' then 9
                      when 'Линия' then 10

                      when 'Местечко' then 11
                      when 'Микрорайон' then 12
                      when 'Набережная' then 13
                      when 'Парк' then 14
                      when 'Переулок' then 15

                      when 'Переезд' then 16
                      when 'Площадь' then 17
                      when 'Площадка' then 18
                      when 'Проспект' then 19
                      when 'Проезд' then 20

                      when 'Просек' then 21
                      when 'Проселок' then 22
                      when 'Проулок' then 23
                      when 'Строение' then 24
                      when 'Территория' then 25

                      when 'Тракт' then 26
                      when 'Тупик' then 27
                      when 'Улица' then 28
                      when 'Участок' then 29
                      when 'Шоссе' then 30

                    end);

        if (reg_prov is null) then
          reg_prov = 28;
      end

      if (adc_type = 2) then -- фактич проживание
      begin
        select pl.pl_name
        from places pl
        where pl.pl_id = :adl_from_pl_id
        into :res_city;

        plt_name = null;

        select pl.pl_name, pt.plt_name
        from places pl
        inner join placetype pt on pl.pl_status = pt.plt_id
        where pl.pl_id = :adl_street_id
        into :res_street, :plt_name;

        res_prov = (case plt_name
                      when 'Аллея' then 1
                      when 'Бульвар' then 2
                      when 'Въезд' then 3
                      when 'Дорога' then 4
                      when 'Заезд' then 5

                      when 'Казарма' then 6
                      when 'Квартал' then 7
                      when 'Километр' then 8
                      when 'Кольцо' then 9
                      when 'Линия' then 10

                      when 'Местечко' then 11
                      when 'Микрорайон' then 12
                      when 'Набережная' then 13
                      when 'Парк' then 14
                      when 'Переулок' then 15

                      when 'Переезд' then 16
                      when 'Площадь' then 17
                      when 'Площадка' then 18
                      when 'Проспект' then 19
                      when 'Проезд' then 20

                      when 'Просек' then 21
                      when 'Проселок' then 22
                      when 'Проулок' then 23
                      when 'Строение' then 24
                      when 'Территория' then 25

                      when 'Тракт' then 26
                      when 'Тупик' then 27
                      when 'Улица' then 28
                      when 'Участок' then 29
                      when 'Шоссе' then 30

                    end);
        if (res_prov is null) then
          res_prov = 28;
      end
    end

    consent_date_begin = null;
    consent_date_end = null;
    amount = 0;

    select macl.mac_amount, maa.ma_date_app, dateadd(month, 2, maa.ma_date_app)
    from miniappclient macl
    inner join miniapp maa on macl.mac_from_ma_id = maa.ma_id
    where macl.mac_from_c_id = :c_id and
          macl.mac_amount > 0 and
          maa.ma_date_end = '31.12.9999' and
          maa.ma_is_active = 1
    order by maa.ma_date_app desc
    rows 1
    into :amount, :consent_date_begin, :consent_date_end;

    primaryid_number = '';
    primaryid_issuingdate = null;
    primaryid_issuingby = '';
    c_birth_date = null;

    select cl.c_pass_no, cl.c_pass_date, cl.c_pass_who, cl.c_birth_date, iif(rd.rd_code = 'Citizen.RU', 21, 10),
           (
           case rd.rd_code
             when 'Citizen.RU' then 'Россия'
             else substring(rd.rd_code from 9)
           end)
    from clients cl
    inner join refdata rd on rd.rd_id = cl.c_citizen
    where cl.c_id = :c_id
    into :primaryid_number, :primaryid_issuingdate, :primaryid_issuingby, :c_birth_date, :primaryid_type,
         :primaryid_issuinglocation;

    if (current_date between :consent_date_begin and :consent_date_end) then
    begin
      suspend;
    end
  end
end
