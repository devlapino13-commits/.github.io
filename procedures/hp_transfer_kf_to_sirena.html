CREATE PROCEDURE HP_TRANSFER_KF_TO_SIRENA(
    K_ID D_ID,
    REP_DATE D_DATE = null,
    KA_MEMO D_STR_255 = null)
RETURNS (
    KA_ID D_ID)
AS
declare variable kf_id d_id;
declare variable reg_id d_integer;
declare variable ka_akt_no d_integer;
declare variable current_lo_id d_id;
declare variable kal_new_kf_name d_str_100;
declare variable kal_new_no d_integer;
declare variable kal_new_officer_code d_str_20;
begin
    rep_date = coalesce(:rep_date, current_date);
    ka_memo = coalesce(:ka_memo, '');

    -- Получаем кредитную папку договора
    select k_from_kf_id
    from kontract
    where k_id = :k_id
    into :kf_id;

    if (kf_id is null) then
        exit;

    -- Получаем текущее закрепление кредитной папки
    select first 1 kal_new_kf_officer, kal_new_kf_name, kal_new_no, kal_new_officer_code, kal_new_reg_id
    from knaktlist
    where kal_from_kf_id = :kf_id and
          is_delete = 0 and
          :rep_date between kal_date and kal_date_end
    order by kal_date desc
    into :current_lo_id, :kal_new_kf_name, :kal_new_no, :kal_new_officer_code, :reg_id;

    if (current_lo_id is null) then
        exit;

    -- Папка уже закреплена за КЦ Сирена
    if (current_lo_id = 1000000000004) then
        exit;

    -- Повторяем ограничение из hp_transfer_lp_to_other:
    -- перенос выполняем только с обычного сотрудника на виртуального
    if (current_lo_id <= 1000000000004) then
        exit;

    -- Создаем акт передачи
    select kv
    from get_kv(:reg_id, 7)
    into :ka_id;

    select coalesce(max(cast(ka_akt_no as integer)), 0) + 1
    from knakt
    where ka_from_reg_id = :reg_id
    into :ka_akt_no;

    insert into knakt (ka_id, ka_date, ka_from_reg_id, ka_akt_no, ka_memo, is_delete, ka_from_ol_id)
    values (:ka_id, :rep_date, :reg_id, :ka_akt_no, :ka_memo, 0, null);

    -- Для КЦ в существующей процедуре к названию папки добавляется ".КЦ"
    kal_new_kf_name = kal_new_kf_name || '.КЦ';

    insert into knaktlist (kal_from_ka_id, kal_number, kal_from_kf_id, kal_from_k_id, kal_date, kal_date_end,
                           kal_new_kf_officer, kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no,
                           kal_new_officer_code)
    values (:ka_id, -1, :kf_id, :kf_id, :rep_date, date '9999-12-31', 1000000000004, :kal_new_kf_name, 0, :reg_id,
            :kal_new_no, :kal_new_officer_code);

    suspend;
end
