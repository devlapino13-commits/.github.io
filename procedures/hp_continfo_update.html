CREATE PROCEDURE HP_CONTINFO_UPDATE(
    C_ID D_ID,
    CONT_KIND D_STR_CODE,
    CONT_VALUE D_STR_100,
    CONT_STATUS D_SMALLINT,
    UN_ID D_ID,
    NEW_DATETIME D_DATETIME = current_timestamp,
    IS_KMP D_SMALLINT = 0)
AS
declare variable found_cound d_integer;
declare variable client_is_active d_smallint;
declare variable err_text d_str_255;
declare variable other_value d_str_100;
declare variable cont_is_main d_smallint;
declare variable cont_kind_code d_str_code;
declare variable rd_id d_integer;
declare variable db_id d_integer;
declare variable phone_number_c_fio str_255; /* ФИО клиента, за которым верифицирован переданный телефонный номер */
declare variable phone_number_c_id d_id; /* ID клиента, за которым верифицирован переданный телефонный номер */
declare variable is_active_client d_boolean; /* Является ли клиент активным (есть ли у него хотя бы 1 активный займ) */
begin
  if (new_datetime is null) then new_datetime = current_timestamp;

  if (cont_kind is null) then
    exception e_default 'Cont_kind is null';

  if (coalesce(c_id, 0) = 0) then
    exception e_default 'C_ID is null';

  db_id = f_get_db_id();

  if (db_id = 41701 and coalesce(cont_value, '') = '' and cont_kind  = 'ContactInfo.Cell_Add') then -- RFCKG-3917
    exception e_message 'Номер указан не верно. Пожалуйста перепроверите';

  if (cont_status is null) then
    cont_status = 0;

  select rd_id, cont_kind_code, cont_is_main
  from hp_contkind_info(:cont_kind)
  into rd_id, :cont_kind_code, cont_is_main;

  cont_value = f_cont_trim(cont_value, cont_kind_code);

  -- если указывают тот же номер, ничего не меняем и завершаем работу
  if (cont_status = 0) then
  begin
    if (exists(select *
    from objectvalues, objectprops
    where opr_object_kind = f_objkind('clients', 'id') and
          opr_object_id = :c_id and
          opr_subject_kind = f_objkind('refdata', 'id') and
          opr_subject_id = :rd_id and
          ovs_from_opr_id = opr_id and
          ovs_id = opr_actual_ovs_id and
          coalesce(ovs_value, '') = :cont_value)) then
      exit;
  end

  if (cont_kind_code = 'cell' and cont_value <> '' and cont_is_main = 0) then
  begin
    other_value = f_cont(c_id, cont_kind_code, 'last', 'main');

    if (cont_value = other_value) then
      exception e_message f_lang(f_get_lang_code(un_id, null), 'Дополнительный номер не должен совпадать с основным номером');
  end

  select count(*), max(client_is_active)
  from hp_continfo_can_change(:c_id, :cont_kind, :cont_value, :un_id)
  where res_is_error = 1
  into found_cound, client_is_active;
  if (client_is_active is null) then
    client_is_active = 0;

  err_text = '';

  if (is_kmp <> 1) then
  begin
      if (client_is_active = 1) then
        err_text = 'У клиента имеется активный займ. ';
    
      if (found_cound = 1) then
        err_text = err_text || 'Данный номер прикреплен другому действующему клиенту, обратитесь к КОБД.';
    
      if (found_cound > 1) then
        err_text = err_text || 'Данный номер прикреплен другим действующим клиентам, обратитесь к КОБД.';
    
      if (err_text <> '') then
        exception e_message trim(err_text);
  end


  -- RFCKG-4086: Запрет на смену основного номера телефона, если этот номер уже верифицирован за активным клиентом
  if (db_id = 41701 and cont_kind  = 'ContactInfo.Cell_Main' and coalesce(cont_value, '') <> '') then
  begin
    for
        select
            c.c_name_fio,
            cci.cci_object_id
        from continfo cci
        inner join clients c on cci.cci_object_id = c.c_id
        where
            cci.cci_object_kind = f_objkind('clients', 'id') and
            cci.ok_cell_main = :cont_value
        into
            :phone_number_c_fio,
            :phone_number_c_id
    do
    begin
        -- Определяем, есть ли у клиента хоть 1 активный займ
        is_active_client = exists(
            select 1
            from kontstate ks
            inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id
            where
                ks.ks_state = 1
                and ks.ks_ol_is_delete = 0
                and current_date between ks.ks_date_begin and ks.ks_date_end
                and kc.kc_amount_approved > 0
                and kc.kc_from_c_id = :phone_number_c_id
        );


        -- Активный клиент, для которого указан и верифицирован этот номер существует
        if (is_active_client) then
        begin
            exception e_message f_format('Невозможно указать этот телефонный номер, т.к. он верифицирован за активным клиентом %1', phone_number_c_fio);
        end
         
    end
  end


  -- Если нет ошибок, вызываем процедуру которая меняет номер
  execute procedure hp_continfo_force_update(:c_id, :cont_kind, :cont_value, :cont_status, :un_id, :new_datetime);

  -- убрано по задаче в Mantis #243
--   if (f_get_db_id() = 41701) then
--    execute procedure fsrobot_insert_event('domailing', 'pushing_phone_changed', :c_id, :un_id);
end
