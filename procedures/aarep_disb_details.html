CREATE PROCEDURE AAREP_DISB_DETAILS(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(30),
    K_DISB_DATE DATE,
    K_CLOSE_DATE DATE,
    K_AMOUNT DECIMAL(18,4),
    K_KOLICH_REPAY INTEGER,
    K_PERIOD INTEGER,
    K_PERCENT DECIMAL(18,7),
    PR_COMMON_NAME VARCHAR(100),
    PR_NAME VARCHAR(100),
    OFFICE_NAME VARCHAR(100),
    REGION_NAME VARCHAR(100),
    LO_FIO VARCHAR(100),
    SCHEDULED_1_DATE DATE,
    SCHEDULED_3_DATE DATE,
    SCHEDULED_6_DATE DATE,
    PAR7_1_BALANCE DECIMAL(18,4),
    PAR7_3_BALANCE DECIMAL(18,4),
    PAR7_6_BALANCE DECIMAL(18,4),
    PAR7_LAST_BALANCE DECIMAL(18,4),
    BALANCE_AT_END DECIMAL(18,4),
    CLIENT_NEW INTEGER,
    CLIENT_COUNT INTEGER)
AS
declare variable K_ID decimal(18,0);
declare variable REG_ID integer;
declare variable LO_ID decimal(18,0);
declare variable REG_ID_PRIOR integer;
declare variable LO_ID_PRIOR decimal(18,0);
begin
  reg_id_prior = 0;
  lo_id_prior = 0;

  for select k_kontract_no, ks_from_k_id, ks_region_id, ks_kc_id, ks_date_begin, ks_date_end, k_amount_approved,
             k_kolich_repay, round(k_period), round(k_percent_per_day * 30.00, 2), pr_common_name, pr_name,
             k_kolich_client_approved
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id
      into :k_kontract_no, :k_id, :reg_id, :lo_id, k_disb_date, k_close_date, :k_amount, :k_kolich_repay, :k_period,
           k_percent, :pr_common_name, pr_name, client_count
  do
  begin
    if (k_close_date >= '31.12.2999') then
      k_close_date = null;

    if (reg_id <> reg_id_prior) then
    begin
      select aa_curr_office_name, aa_curr_level3_name
      from aarep_get_office_categ(:reg_id, :rep_date_end)
      into :office_name, region_name;
      reg_id_prior = reg_id;
    end

    if (lo_id <> lo_id_prior) then
    begin
      select c_name_fio
      from clients
      where c_id = :lo_id
      into :lo_fio;
      lo_id_prior = lo_id;
    end

    client_new = 0;
    select count(*)
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          coalesce(kc_circle, 1) <= 1
    into :client_new;
    if (client_new is null) then
      client_new = 0;

    scheduled_1_date = null;
    scheduled_3_date = null;
    scheduled_6_date = null;
    par7_1_balance = 0.0;
    par7_3_balance = 0.0;
    par7_6_balance = 0.0;
    par7_last_balance = 0.0;
    balance_at_end = 0.0;

    scheduled_1_date = (select sh_date
                        from schedule
                        where sh_from_k_id = :k_id and
                              sh_id = 1);
    scheduled_3_date = (select sh_date
                        from schedule
                        where sh_from_k_id = :k_id and
                              sh_id = 3);
    scheduled_6_date = (select sh_date
                        from schedule
                        where sh_from_k_id = :k_id and
                              sh_id = 6);

    if (scheduled_1_date is not null) then
    begin
      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= (:scheduled_1_date + 8) and
            oks_date_end >= (:scheduled_1_date + 8) and
            oks_delinq_date <= :scheduled_1_date
            and oks_close_state = 0 
      into par7_1_balance;
      if (par7_1_balance is null) then
        par7_1_balance = 0.0;
    end

    if (scheduled_3_date is not null) then
    begin
      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= (:scheduled_3_date + 8) and
            oks_date_end >= (:scheduled_3_date + 8) and
            oks_delinq_date <= :scheduled_3_date
            and oks_close_state = 0 
      into par7_3_balance;
      if (par7_3_balance is null) then
        par7_3_balance = 0.0;
    end

    if (scheduled_6_date is not null) then
    begin
      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= (:scheduled_6_date + 8) and
            oks_date_end >= (:scheduled_6_date + 8) and
            oks_delinq_date <= :scheduled_6_date
            and oks_close_state = 0 
      into par7_6_balance;
      if (par7_6_balance is null) then
        par7_6_balance = 0.0;
    end

    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= (:rep_date_end + 8) and
          oks_date_end >= (:rep_date_end + 8) and
          oks_delinq_date <= :rep_date_end
          and oks_close_state = 0 
    into par7_last_balance;
    if (par7_last_balance is null) then
      par7_last_balance = 0.0;

    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end
          and oks_close_state = 0 
    into balance_at_end;
    if (balance_at_end is null) then
      balance_at_end = 0.0;

    suspend;
  end
end
