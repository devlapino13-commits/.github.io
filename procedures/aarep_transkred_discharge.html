CREATE PROCEDURE AAREP_TRANSKRED_DISCHARGE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(61),
    PARENT_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(20),
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OK_OS DECIMAL(18,4),
    OK_PERC DECIMAL(18,4),
    OK_FINE DECIMAL(18,4),
    OL_MEMO VARCHAR(1024),
    CL_CASHIER_NAME VARCHAR(100),
    DOC_CLI VARCHAR(100),
    OL_ID DECIMAL(18,0),
    OL_DATETIME TIMESTAMP,
    TOTAL_SUM DECIMAL(18,4))
AS
declare variable k_id decimal(18,0);
begin /*Отчет список погашении за указанный период погашал не казначей*/
  for select reg_code || ' ' || reg_name, parent_name, k_kontract_no, ol_date, ol_doc_no, sum(ok_os), sum(ok_perc),
             sum(ok_fine), ol_memo, cl1.c_name_fio cashier, cl2.c_name_fio doc_cli, ol_id, ol_datetime, k_id
      from operlist, region_list(:rep_region_id, 1, 3, 1), region, operkred, kontract, clients cl1, clients cl2
      where id_list = ol_from_reg_id and
            operlist.is_delete = 0 and
            ol_date between :rep_date_begin and :rep_date_end and
            ok_from_ol_id = ol_id and
            ok_type = 4 and
            reg_unicode = id_list and
            k_id = ok_from_k_id and
            k_clients_cashier_id <> ol_document_client and
            cl1.c_id = k_clients_cashier_id and
            cl2.c_id = ol_document_client and
            ok_from_c_id = 0
      group by reg_code, reg_name, parent_name, k_kontract_no, ol_date, ol_doc_no, ol_doc_no_str, ol_memo, cl1.c_name_fio, cl2.c_name_fio, ol_id, ol_datetime, k_id
      into :reg_name, :parent_name, :k_kontract_no, :ol_date, :ol_doc_no, :ok_os, :ok_perc, :ok_fine, :ol_memo,
           :cl_cashier_name, :doc_cli, :ol_id, :ol_datetime, :k_id
  do
  begin
    total_sum = coalesce(:ok_os, 0) + coalesce(:ok_perc, 0) + coalesce(:ok_fine, 0);

    if (not exists(select 1
                   from knclient
                   where kc_amount_approved <> 0 and
                         kc_from_k_id = :k_id and
                         kc_status = 2)) then
      cl_cashier_name = 'Казначей отсутствует.';

    suspend;
  end
end
