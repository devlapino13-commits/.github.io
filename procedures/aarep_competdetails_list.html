CREATE PROCEDURE AAREP_COMPETDETAILS_LIST(
    CPR_ID INTEGER)
RETURNS (
    CPD_FROM_CPR_ID INTEGER,
    CPD_ID INTEGER,
    CPD_OBJECT_KIND INTEGER,
    CPD_OBJECT_ID DECIMAL(18,0),
    CPD_OBJECT_NAME VARCHAR(255),
    CPD_POINTS_ALL DECIMAL(18,4),
    CPD_POINTS_1 DECIMAL(18,4),
    CPD_POINTS_2 DECIMAL(18,4),
    CPD_POINTS_3 DECIMAL(18,4),
    CPD_POINTS_4 DECIMAL(18,4),
    CPD_POINTS_5 DECIMAL(18,4),
    CPD_POINTS_6 DECIMAL(18,4),
    CPD_POINTS_7 DECIMAL(18,4),
    CPD_POINTS_8 DECIMAL(18,4),
    CPD_POINTS_9 DECIMAL(18,4),
    CPD_COMMENT VARCHAR(3072))
AS
begin
  for select cpd_from_cpr_id, cpd_id, cpd_object_kind, cpd_object_id, cpd_points_all, cpd_points_1, cpd_points_2,
             cpd_points_3, cpd_points_4, cpd_points_5, cpd_points_6, cpd_points_7, cpd_points_8, cpd_points_9,
             cpd_comment
      from competdetails
      where cpd_from_cpr_id = :cpr_id
      into :cpd_from_cpr_id, :cpd_id, :cpd_object_kind, :cpd_object_id, :cpd_points_all, :cpd_points_1, :cpd_points_2,
           :cpd_points_3, :cpd_points_4, :cpd_points_5, :cpd_points_6, :cpd_points_7, :cpd_points_8, :cpd_points_9,
           :cpd_comment
  do
  begin
    cpd_object_name = '';
    if (cpd_object_kind = -1) then
    begin
      select rd_name
      from refdata
      where rd_id = :cpd_object_id
      into :cpd_object_name;
    end
    if (cpd_object_kind = 0) then
    begin
      select c_name_fio
      from clients
      where c_id = :cpd_object_id
      into :cpd_object_name;
    end
    if (cpd_object_kind = 7) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :cpd_object_id
      into :cpd_object_name;
    end
    suspend;
  end
end
