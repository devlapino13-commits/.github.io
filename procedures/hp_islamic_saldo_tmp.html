CREATE PROCEDURE HP_ISLAMIC_SALDO_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    RES_REGION_ID INTEGER,
    RES_REGION_NAME VARCHAR(100),
    AMOUNT_PLUS DECIMAL(18,4),
    AMOUNT_MINUS DECIMAL(18,4),
    AMOUNT_DISB DECIMAL(18,4),
    AMOUNT_SALDO DECIMAL(18,4))
AS
declare variable al_id varchar(20);
declare variable tmp_amount decimal(18,4) = 0;
declare variable islamic_account varchar(20);
declare variable ln_account varchar(20);
declare variable start_date date;
declare variable pr_id integer;
begin
  islamic_account = '10101';
  start_date = '01.03.2017';

  amount_plus = 0;
  amount_minus = 0;
  amount_disb = 0;
  if (rep_region_id = 1040200) then
    amount_plus = 655417;
  if (rep_region_id = 1090100) then
    amount_plus = 25012565.19;

  res_region_id = rep_region_id;
  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :rep_region_id
  into :res_region_name;

  for select al_id
      from acc_list
      where al_from_reg_id = :rep_region_id and
            al_from_bg_id = :islamic_account
      into al_id
  do
  begin
    for select ln_account
        from loronostro
        where ln_region_loro = :rep_region_id
        into :ln_account
    do
    begin
      tmp_amount = 0;
      select sum(ob_summa)
      from operbook
      where ob_ol_date <= :rep_date and
            ob_account_dt = :al_id and
            ob_account_kt = :ln_account and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = :rep_region_id
      into :tmp_amount;
      if (tmp_amount is null) then
        tmp_amount = 0;
      amount_plus = amount_plus + tmp_amount;

      tmp_amount = 0;
      select sum(ob_summa)
      from operbook
      where ob_ol_date <= :rep_date and
            ob_account_dt = :ln_account and
            ob_account_kt = :al_id and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = :rep_region_id
      into :tmp_amount;
      if (tmp_amount is null) then
        tmp_amount = 0;
      amount_minus = amount_minus + tmp_amount;
    end
  end

  tmp_amount = 0;
  select sum(k_amount_approved)
  from kontract, kontstate, product
  where ks_from_k_id = k_id and
        ks_state = 1 and
        ks_date_begin between :start_date and :rep_date and
        ks_ol_is_delete = 0 and
        ks_region_id = :rep_region_id and
        k_from_pr_id = pr_id and
        pr_code containing '.ipf.' -- было pr_code containing '.murabaha.' изменен от 02.04.2018 Улан
  into tmp_amount;
  if (tmp_amount is null) then
    tmp_amount = 0;
  amount_disb = amount_disb + tmp_amount;

  amount_saldo = amount_plus - amount_minus - amount_disb;
  suspend;
end
