CREATE PROCEDURE HP_EDOC_CONF_STATUS_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_C_ID D_ID,
    REP_STATUS D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    EDN_EDL_ID D_ID,
    EDN_OBJECT_KIND D_INTEGER,
    EDN_OBJECT_ID D_ID,
    EDN_DATE D_DATE,
    EDN_FROM_EDC_ID D_INTEGER,
    EDN_FROM_C_ID D_ID,
    EDN_STATUS D_SMALLINT,
    EDN_CREATE_DATETIME D_DATETIME,
    EDN_MANAGER_C_ID D_ID,
    EDN_FROM_REG_ID D_INTEGER,
    EDN_NEED_CONFIRM D_BOOL,
    RES_CONF_AMOUNT D_STR_CODE,
    EDN_BLANK_LIST D_STR_LONG,
    EDN_STATUS_DATETIME D_DATETIME,
    RES_HO_ORDER_TYPE D_STR_100,
    RES_HO_ORDER_NO D_STR_20,
    RES_OFFICE D_STR_100,
    RES_STATUS_CODE D_STR_CODE,
    RES_EDOC_STATUS D_STR_CODE,
    RES_BLANK_STATUS D_STR_CODE,
    RES_CREATE_USER_NAME D_STR_CODE,
    RES_STATUS_USER_NAME D_STR_CODE,
    RES_NAME_FIO D_STR_100)
AS
declare variable edl_status d_smallint;
declare variable edl_blank_status d_smallint;
declare variable edn_conf_amount d_currency;
declare variable has_access d_bool;
declare variable ml_id d_integer;
declare variable edn_status_un_id d_id;
begin
  if (rep_status = -2) then -- все статусы
    rep_status = null;

  has_access = 0;

  select ml_id
  from menulist
  where ml_code = 'n240' and
        ml_secure_kind = -1
  into ml_id;

  if (ml_id is not null) then
  begin
    if (exists(select *
               from hp_permission_access_check(:un_id, 'menulist', :ml_id, '')
               where has_access = 1)) then
      has_access = 1;
  end

  for select edn_edl_id,
             edn_object_kind,
             edn_object_id,
             edn_date,
             edn_from_edc_id,
             edn_from_c_id,
             edn_status,
             edn_create_datetime,
             edn_manager_c_id,
             edn_from_reg_id,
             edn_need_confirm,
             edn_conf_amount,
             edn_blank_list,
             edn_status_un_id,
             edn_status_datetime,
             reg_code || ' ' || reg_name,
             c_name_fio,
             un_name
      from edoc_conf, region, clients, username, region_list(:rep_region_id, 1, 3, 1)
      where edn_date between :rep_date_begin and :rep_date_end and
            edn_need_confirm = 1 and
            edn_status = coalesce(:rep_status, edn_status) and
            edn_from_reg_id = id_list and
            reg_unicode = id_list and
            edn_from_c_id = coalesce(:rep_c_id, edn_from_c_id) and
            edn_from_c_id = c_id and
            edn_create_un_id = un_id
      order by edn_date desc, edn_edl_id desc
      into :edn_edl_id,
           :edn_object_kind,
           :edn_object_id,
           :edn_date,
           :edn_from_edc_id,
           :edn_from_c_id,
           :edn_status,
           :edn_create_datetime,
           :edn_manager_c_id,
           :edn_from_reg_id,
           :edn_need_confirm,
           :edn_conf_amount,
           :edn_blank_list,
           :edn_status_un_id,
           :edn_status_datetime,
           :res_office,
           :res_name_fio,
           :res_create_user_name
  do
  begin
    res_ho_order_type = '';
    res_ho_order_no = '';
    res_status_user_name = '';

    if (edn_status_un_id is not null) then
    begin
      select un_name
      from username
      where un_id = :edn_status_un_id
      into res_status_user_name;
    end

    res_conf_amount = '0.00';
    if (edn_conf_amount > 0) then
    begin
      if (has_access = 1) then
        res_conf_amount = f_numtostr(edn_conf_amount, 2);
      else
        res_conf_amount = '***';
    end

    select ho_order_no, sawt_name
    from hr_orders, sa_worktype
    where ho_id = :edn_object_id and            
          ho_order_type = sawt_id
    into res_ho_order_no, res_ho_order_type;

    if (res_ho_order_no is null) then
      res_ho_order_no = '';

    res_status_code = trim(case edn_status
      when -1 then 'Аннулирован'
      when 0 then 'Новый'
      when 1 then 'Подтвержден'
      else ''
    end);

    edl_status = null;
    edl_blank_status = null;
    res_edoc_status = 'Не создан';
    res_blank_status = '';

    select edl_status, edl_blank_status
    from edoc_list
    where edl_id = :edn_edl_id
    into edl_status, edl_blank_status;

    if (edl_status is not null) then
    begin
      res_edoc_status = trim(case edl_status
        when 1 then 'Подписан'
        when 0 then 'Не подписан'
        when -1 then 'Аннулирован'
      end);

      res_blank_status = trim(case edl_blank_status
        when 1 then 'Приготовлены'
        when 0 then 'В ожидании'
        when -1 then 'Ошибка'
      end);
    end

    suspend;
  end
end
