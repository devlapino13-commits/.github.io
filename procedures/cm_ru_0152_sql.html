CREATE PROCEDURE CM_RU_0152_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable REG_ID integer;
declare variable PR_ID integer;
declare variable PR_CODE varchar(100);
declare variable REG_LOGIC_LABEL varchar(100);
declare variable K_DISB_DATE date;
declare variable KF_ID ID_TYPE;
declare variable KF_OFFICER ID_TYPE;
begin
  reg_id = null;
  select k_last_reg_id, k_from_pr_id, k_disb_date, k_from_kf_id
  from kontract
  where k_id = :k_id
  into reg_id, pr_id, k_disb_date, kf_id;
  select kal_new_kf_officer
  from knaktlist
  where kal_from_kf_id = :kf_id and
        current_date between kal_date and kal_date_end and
        knaktlist.is_delete = 0
  into kf_officer;
  if (reg_id is null) then
  begin
    res_error_text = 'офис не определен, выберите офис';
    res_error_is_critical = 2;
    suspend;
    exit;
  end
  select pr_code
  from product
  where pr_id = :pr_id
  into pr_code;
  reg_logic_label = null;
  select reg_logic_label
  from region
  where reg_unicode = :reg_id
  into reg_logic_label;
  reg_logic_label = coalesce(reg_logic_label, '');
  if (reg_logic_label containing '.mini_office.') then
  begin
    if (pr_code not similar to '%(.status.|.status_plus.|.shans.|.express.|.shans plus.|.food.|.blago.|.reliable_client.|.express_online.|.universal.|.oivo.)%') then -- добавлен  blago
    begin
      res_error_text = 'по выбранному продукту в мини офисах выдачи запрещены';
      res_error_is_critical = 2;
      suspend;
    end
    else
    begin
      if (pr_code containing '.blago.') then --по задаче на jira BOARD-2629 по указанным мини офисам Благо запрешено
      begin
        res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
        res_error_is_critical = 2;
        suspend;
      end
      if (pr_code containing '.reliable_client.') then -- BOARD-2409
      begin
        if (k_disb_date < '12.11.2020') then
        begin
          if (reg_id not in (1460100, 1490500, 1490400, 1500100, 1510100, 1520100, 1280300)) then -- по задаче на jira BOARD-2629 по новому продукту выдача разрешено
          begin
            res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
            res_error_is_critical = 2;
            suspend;
          end
        end
        else
        begin
          if (reg_id not in (1460100, 1490600, 1500100, 1510100, 1520100, 1280300, 1400200, 1490500)) then -- по задаче на jira BOARD-2629 по новому продукту выдача разрешено
          begin
            res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
            res_error_is_critical = 2;
            suspend;
          end
        end
      end
      if (pr_code containing '.shans.') then
      begin
        if (reg_id not in (1460100, 1400200, 1510100, 1500100, 1490500, 1490600, 1120400, 1520100, 1280300, 1380200,
                           1170200)) then
        begin
          res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
          res_error_is_critical = 2;
          suspend;
        end
      end
      if (pr_code containing '.express.') then
      begin
        if (reg_id not in (1510100, 1500100, 1490600, 1120400, 1520100, 1280300, 1400200, 1380200, 1170200)) then
        begin
          res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
          res_error_is_critical = 2;
          suspend;
        end
      end
      if (pr_code containing '.status.') then
      begin
        if (reg_id not in (1510100, 1500100, 1170200, 1380200 /*, 1490500*/ )) then --добавил советский
        begin
          res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
          res_error_is_critical = 2;
          suspend;
        end
      end
      if (pr_code containing '.status_plus.') then
      begin
        res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
        res_error_is_critical = 2;
        suspend;
      end
      if (pr_code containing '.shans plus.') then
      begin
        if (reg_id not in (1460100, 1400200, 1510100, 1500100, 1490500, 1490600, 1120400, 1520100, 1280300, 1380200,
                           1170200)) then
        begin
          res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
          res_error_is_critical = 2;
          suspend;
        end
      end
      if (pr_code containing '.universal.') then
      begin
        if (reg_id in (1460100, 1490600)) then
        begin
          res_error_text = 'по выбранному продукту в текущем офисе выдачи запрещены';
          res_error_is_critical = 2;
          suspend;
        end
      end

    end
  end
  else
  begin
    if (pr_code not similar to '%(.express.|.express_online.|.shans.|.blago.|.reliable_client.|.uspeh.|.reliable_client_plus.|.ZALOG_CAR.|.reliable_emp.|.education.|.LOYAL.|.food.|.float_percent.|.support.|.summer_plus.|.comfortable.|.zalog.|.selfemployed.|.universal.|.privilege.|.privilege_plus.|.oivo.|.refinance.|.refinance_oivo.|.easy_start.|.business.|.refinance_plus.|.easy_start_oivo.|.switch_plus.|.b_bonus.|.plus_60.|.plus_100.|.plus_100_b.|.b_refinance.)%') then
    begin
      if (not(reg_id = 1490100 and
          kf_officer = 1010100569523 and
          (pr_code containing '.status.' or pr_code containing '.status_plus.'))) then
      begin
        res_error_text = 'по выбранному продукту в обычных офисах выдачи запрещены';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end

  if (reg_id not in (1012800, 1050600)) then --Только в OiVO доступен данный продукт
  begin
    if (pr_code containing '.easy_start_oivo.') then
    begin
      res_error_text = 'Выбранный продукт доступен только в офисах OiVO';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
