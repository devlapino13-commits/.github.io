CREATE PROCEDURE HP_ACTIONLIST_KK_OTHER_DB_INFO(
    IN_C_ID D_ID)
RETURNS (
    RES_INFO D_STR_8100,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_INN D_STR_CODE,
    C_PASS_NO D_STR_CODE)
AS
declare variable db_id d_integer;
begin
  res_info = 'Нет инфо';
  db_id = f_get_db_id();

  for select c_name_fio, c_birth_date, iif(:db_id = 81001, iif(coalesce(c_ext_inn, '') <> '', c_ext_inn, c_inn), c_inn), c_pass_no
  from clients
  where c_id = :in_c_id
  into c_name_fio, c_birth_date, c_inn, c_pass_no
  do
  begin
    select res_text
    from get_kk_other_db_check(:c_name_fio, :c_birth_date, :c_inn, :c_pass_no, 0)
    into :res_info;
  end

  if (res_info = '') then
    res_info = 'Нет инфо';

  suspend;
end
