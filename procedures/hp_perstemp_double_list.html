CREATE PROCEDURE HP_PERSTEMP_DOUBLE_LIST(
    FIND_C_ID DECIMAL(18,0),
    FIND_FIO VARCHAR(255),
    FIND_PASS_NO VARCHAR(30),
    FIND_INN VARCHAR(20),
    FIND_BIRTH_DATE DATE,
    FIND_CELL_NO VARCHAR(20))
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(255),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_INN VARCHAR(20),
    C_COMMENT VARCHAR(100),
    C_COMMENT_ID INTEGER,
    C_ADDRESS VARCHAR(255),
    C_PLASTCARD_NO VARCHAR(20),
    C_SEX INTEGER)
AS
begin
  for select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_comment, c_comment_id, c_address, c_plastcard_no, c_sex
  from aarep_check_client_fio(:find_c_id, :find_fio, :find_pass_no, :find_inn, 0, :find_birth_date, :find_cell_no, 0, null, 0, 0)
  order by c_comment_id desc, c_id
  into :c_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn, :c_comment, :c_comment_id, :c_address, :c_plastcard_no, :c_sex
  do
    suspend;
end
