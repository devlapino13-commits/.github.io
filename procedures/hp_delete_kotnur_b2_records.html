CREATE PROCEDURE HP_DELETE_KOTNUR_B2_RECORDS
AS
declare variable eil_id d_id;
declare variable eic_id d_id;
declare variable mail_subject d_str_255;
declare variable mail_body d_str_long;
declare variable mail_to d_str_255;
declare variable config d_str_code;
begin
-- RFCRU-5873
-- Написать робота, который удаляет записи из extinfolist,
-- для конфигурации kontur_b2, если прошло более 14 дней от eil_date, eil_status=2
-- и eil_from_bls_is is null (т.е. клиент НЕ банкрот).
  config = 'kontur_b2';

  select eic.eic_id
  from extinfocfg eic
  where eic.eic_code = :config
  into eic_id;

  if (coalesce(eic_id, 0) = 0) then
  begin
    mail_subject = f_format('Ошибка удаления записей для конфигурации %', config);
    mail_body = f_format('<strong>При удалении записей возникла ошибка: конфигурация %1 не найдена в процедуре hp_delete_kotnur_b2_records</strong>', config);
    mail_to = 'ru.finstart@molbulak.com, p.juravlev@mbulak.kg';
    execute procedure hp_send_quick_message('email', :mail_to, :mail_subject, :mail_body, '', 1, 1000000000000);
    exit;
  end

  delete from extinfolist eil
  where eil.eil_from_eic_id = :eic_id
  and eil.eil_from_bls_id is null
  and eil.eil_status = 2
  and eil.eil_date < dateadd(-14 day to current_date);
end
