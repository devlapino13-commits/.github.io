CREATE PROCEDURE MC_ONLINE_LOAN_VISIBILITY(
    MCL_ID D_ID,
    LOAN_LIMIT D_CURRENCY,
    HAVE_MBULAK_CARD D_INTEGER)
RETURNS (
    ONLINE_APP_ID D_INTEGER,
    IS_VISIBLE D_INTEGER,
    ONLINE_MBULAK_CARDS D_BOOL,
    ONLINE_OTHER_CARDS D_BOOL,
    MAX_LOAN_AMOUNT D_CURRENCY)
AS
declare variable C_ID D_ID;
declare variable CAL_CARD_ID D_INTEGER;
declare variable K_DISB_KIND D_INTEGER;
declare variable KS_DATE_BEGIN D_DATE;
declare variable KBM_ID D_ID;
declare variable KIB_SCORE D_INTEGER;
declare variable ACTIVE_ONLINE_BALANCE D_CURRENCY;
declare variable DAY_OF_WEEK  D_SMALLINT;
declare variable CURRENT_HOUR  D_SMALLINT;
begin
  is_visible = 0;
  online_mbulak_cards = 1;

    -- RFCKG-2521
    -- каждую субботу с 22.00 до 08.00 следующего дня отключать баннер онлайн займа
    day_of_week = extract(weekday from current_date);
    current_hour =  extract(hour from current_timestamp);
    
    if( (day_of_week = 6 and current_hour >= 22) or
        (day_of_week = 0 and current_hour < 8 )) then
    begin
        is_visible = 0;

        suspend;
        exit;
    end

  if ((select cv.cv_value
       from const_var cv
       where cv.cv_code = 'online_loan_is_active') = 0) then
  begin
    suspend;
    exit;
  end 

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id;

  if (coalesce(:c_id, -777) = -777) then
  begin
    suspend;
    exit;
  end

  if (f_get_const('online_loan_limit_cli_list') = 1) then
  begin
    if (not exists (
        select *
        from temp_objlist tol
        where tol_kind = 'online_loan'
        and tol.tol_object_id = :c_id
    )) then
    begin
      is_visible = 0;
      suspend;
      exit;
    end
  end

  online_other_cards = f_get_const('online_other_cards');

  -- RFCKG-2684
  if (c_id = 1090101308471) then -- для тестовой выдачи
    online_other_cards = 1;
  --

  if (online_other_cards = 0) then -- если активрована выдача на сторонние карты, нет смысла проверять наличие карты МБулак т.к. все-равно виджет будет виден
  begin
      -- проверяю наличие активной карты МБулак у клиента -->
      -- логика передачи входящего параметра HAVE_MBULAK_CARD оказалась некорректной ввиду ошибки в ТЗ, поэтому решил получать значенеи чз вызов процедуры
      have_mbulak_card = 0;
    
      select cal_card_id
      from get_active_mbulak_card(:c_id)
      into :cal_card_id;
    
    
      if (cal_card_id is not null) then
        have_mbulak_card = 1;
      -- < -----------------------------------------
  end

  if (((have_mbulak_card = 1) or (online_other_cards = 1))  and
      loan_limit >= 15000 and
      (select sum(coalesce((select (current_date - oks.oks_delinq_date)
                            from oksummary oks
                            where current_date between oks.oks_date_begin and oks.oks_date_end and
                                  current_date - oks.oks_delinq_date > 0 and
                                  oks_close_state = 0 and
                                  oks_from_k_id = kc.kc_from_k_id), 0))
       from knclient kc
       where kc.kc_from_c_id = :c_id
       and kc.kc_amount_approved > 0) = 0) then
    is_visible = 1;

  select oa.oa_id
  from online_app oa
  where oa.oa_from_mcl_id = :mcl_id
  and oa.oa_status between 0 and 2
  rows 1
  into :online_app_id;

  if (online_app_id is not null) then
    is_visible = 1;

  --JIRA 2168
  for select kt.k_disb_kind, ks.ks_date_begin
      from knclient kc
      inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
      inner join kontract kt on kt.k_id = kc.kc_from_k_id
      where kc.kc_from_c_id = :c_id and
            kc.kc_amount_approved > 0 and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end
      order by ks.ks_date_begin desc
      into k_disb_kind, ks_date_begin
  do
  begin
    if (online_app_id is null and ks_date_begin = current_date) then
    begin
      is_visible = 0;
      break;
    end 

    /* Данное условие было убрано на основании задачи RFCKG-2599
    if (online_app_id is null and k_disb_kind = 6) then
    begin
      is_visible = 0;
      break;
    end 
    */
  end

  if (is_visible = 1 and f_point(c_id, current_date, 0) < 6) then
    is_visible = 0;

  if (is_visible = 1) then
  begin
    select kbm_id,
           kbm_score_rate
    from mc_online_loan_get_kib_id(:c_id)
    into :kbm_id,
         :kib_score;

    if (kbm_id is not null) then
    begin
      if (kib_score < 641 or kib_score = 999) then
        is_visible = 0;
    end
  end

  -- реализация оп задаче RFCKG-2599. Связанная задача RFCKG-2252 -->
  if (is_visible = 1) then
  begin
      active_online_balance = 0;

      select sum(oks.oks_balance)
      from knclient kc
      inner join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id and
            current_date between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_close_state = 0
      inner join kontract kt on kt.k_id = oks.oks_from_k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id and
            pr.pr_code containing '.online_loan.'
      where kc.kc_from_c_id = :c_id and
            kc.kc_amount_approved > 0
      into active_online_balance;

      max_loan_amount = 49999 - active_online_balance;
      max_loan_amount = iif(max_loan_amount > loan_limit, loan_limit, max_loan_amount);
      if (max_loan_amount < 15000) then
        is_visible = 0;
  end
  -- < --

  suspend;
end
