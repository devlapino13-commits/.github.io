CREATE PROCEDURE AAREP_MINIAPP_DISB(
    REP_DATE_BEGIN1 DATE,
    REP_DATE_END1 DATE)
RETURNS (
    DATE_PERIOD VARCHAR(20),
    KOL_GROUP_APP INTEGER,
    KOL_CLI_APP INTEGER,
    KAL_GROUP_VADAN INTEGER,
    KOL_CLI_VADAN INTEGER)
AS
declare variable date_begin date;
declare variable date_end date;
declare variable rep_date_begin date;
declare variable rep_date_end date;
declare variable ma_id_prior integer;
declare variable ma_id integer;
declare variable ma_from_kf_id id_type;
declare variable ma_date_app date;
declare variable ma_number integer;
declare variable ma_kolich_client integer;
declare variable kol_cli_disb integer;
begin
  date_begin = rep_date_begin1;
  date_end = rep_date_end1;

  while (date_begin < date_end) do
  begin
    rep_date_begin = cast('01.' || extract(month from date_begin) || '.' || extract(year from date_begin) as date);
    rep_date_end = cast('01.' || extract(month from rep_date_begin + 35) || '.' || extract(year from rep_date_begin + 35) as date) - 1;
    date_begin = rep_date_end + 1;
    date_period = extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end);
    ma_id_prior = 0;
    kol_group_app = 0;
    kol_cli_app = 0;
    kal_group_vadan = 0;
    kol_cli_vadan = 0;
    for select ma_id, ma_from_kf_id, ma_date_app, ma_number, ma_kolich_client
        from miniapp
        where ma_date_app between :rep_date_begin and :rep_date_end
        order by ma_from_kf_id, ma_id, ma_date_app, ma_number
        into ma_id, ma_from_kf_id, ma_date_app, ma_number, ma_kolich_client
    do
    begin
      if (ma_id <> ma_id_prior) then
      begin
        kol_cli_disb = null;
        select k_kolich_client_approved
        from kontract, kontstate
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_from_k_id = k_id and
              ks_date_begin between :ma_date_app and :ma_date_app + 21 and
              k_from_kf_id = :ma_from_kf_id and
              k_from_ma_app_id = :ma_id
        into kol_cli_disb;
        if (kol_cli_disb is null) then
          kol_cli_disb = 0;
        kol_group_app = kol_group_app + 1;
        kol_cli_app = kol_cli_app + ma_kolich_client;
        if (kol_cli_disb <> 0) then
        begin
          kal_group_vadan = kal_group_vadan + 1;
          kol_cli_vadan = kol_cli_vadan + kol_cli_disb;
        end
        ma_id_prior = ma_id;
      end
    end
    suspend;
  end
end
