CREATE PROCEDURE CM_KG_0053_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable PR_CODE varchar(100);
declare variable C_ID ID_TYPE;
declare variable KOL_CLI integer;
declare variable KOL_UCHITEL integer;
begin
  select pr_code, ma_kolich_client
  from miniapp, product
  where ma_id = :ma_id
  and ma_is_active = 1
  and ma_date_end = '31.12.9999'
  and ma_from_pr_id = pr_id
  into pr_code, kol_cli;
  if (pr_code containing '.MUGALIM.') then
  begin
    kol_uchitel = 0;
    for
    select mac_from_c_id
    from miniappclient
    where mac_from_ma_id = :ma_id
    and mac_amount > 0
    into c_id
    do
    begin
      if (exists(select *
          from HP_OBJECTPROPS_ANKETA_GET (0, :c_id, -1,  'work_job_name', null, null)
          where RES_STR_VALUES = 'учитель')) then
      begin
        kol_uchitel = kol_uchitel + 1;
      end
    end
    if (kol_uchitel < kol_cli - kol_uchitel) then
    begin
    if (pr_code containing '.BIRGE.MUGALIM.') then
      res_error_text = 'По продукту Мугалим клиент не указан как учитель в анкете.'; else
      res_error_text = 'Кол-во учителей меньше 45% от состава группы, продукт Мугалим не правильно выбран.';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
