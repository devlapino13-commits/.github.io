CREATE PROCEDURE PRINT_ZALOGAUTO(
    ZAP_ID D_INTEGER)
RETURNS (
    ZAP_DATE D_DATE,
    ZAP_NO D_STR_CODE,
    ZAP_AVG_COST D_CURRENCY,
    ZAP_ZALOG_COST D_CURRENCY,
    ZAP_STATUS D_SMALLINT,
    ZAD_DATETIME D_DATETIME,
    K_ID D_ID,
    K_DISB_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    PTS_VIN D_STR_CODE,
    PTS_ENGINE_NO D_STR_CODE,
    PTS_CHASSIS_NO D_STR_CODE,
    PTS_BODY_NO D_STR_CODE,
    PTS_MODEL D_STR_CODE,
    PTS_ISSUE_YEAR D_STR_CODE,
    STS_PLATE D_STR_CODE)
AS
declare variable sts_drl_id d_integer;
declare variable pts_drl_id d_integer;
begin
  for select zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zad_datetime, k_id, k_kontract_no, k_disb_date,
             cds_sts.cds_from_drl_id, cds_pts.cds_from_drl_id
      from zalogautoprof
      left join zalogauto on zap_from_za_id = za_id
      left join kontract on za_from_k_id = k_id
      left join clientdocs cds_sts on za_orig_sts_id = cds_sts.cds_id
      left join clientdocs cds_pts on za_orig_pts_id = cds_pts.cds_id
      where zap_id = :zap_id
      into zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zad_datetime, k_id, k_kontract_no, k_disb_date,
           sts_drl_id, pts_drl_id
  do
  begin
    pts_vin = '';
    select z_document
    from zalog z
    where z_from_k_id = :k_id and
          current_date between coalesce(z_date_begin, '01.01.1900') and coalesce(z_date_end, '31.12.9999')
    into :pts_vin; --Идентификационный номер (VIN)

    --pts_vin = f_docrecog_value(pts_drl_id, 'vin'); -- Идентификационный номер (VIN)
    pts_engine_no = f_docrecog_value(pts_drl_id, 'engine_no'); -- Модель, № двигателя
    pts_chassis_no = f_docrecog_value(pts_drl_id, 'chassis_no'); -- Шасси (рама) №
    pts_body_no = f_docrecog_value(pts_drl_id, 'body_no'); -- Кузов (кабина, прицеп) №
    pts_model = f_docrecog_value(pts_drl_id, 'model'); -- Марка, модель ТС
    pts_issue_year = f_docrecog_value(pts_drl_id, 'issue_year'); -- Год изготовления ТС
    sts_plate = f_docrecog_value(sts_drl_id, 'plate'); -- Регистрационный знак
    suspend;
  end
end
