CREATE PROCEDURE HP_DO_EXPRESSAPP_START(
    APP_COUNT_ALL D_INTEGER,
    AVG_COUNT_PER_LO D_INTEGER,
    UNCLUDE_LO_IN_HOLIDAY D_BOOL)
RETURNS (
    TOTAL_STARTED D_INTEGER)
AS
declare variable group_id d_integer;
declare variable lo_count d_integer;
declare variable lo_id d_id;
declare variable fact_started d_integer;
declare variable exa_id d_integer;
declare variable k_id d_id;
declare variable fill_count d_integer;
begin
  group_id = 0;
  total_started = 0;

  select count(distinct exa_lo_id)
  from expressapp
  where exa_from_exc_id = 0 and
        exa_final_status = 0
  into lo_count;

  if (app_count_all is null) then
    app_count_all = 0;

  if (avg_count_per_lo is null) then
    avg_count_per_lo = 0;

  if (app_count_all <> 0 or avg_count_per_lo <> 0) then
  begin
    if (app_count_all > 0) then
      avg_count_per_lo = round(app_count_all / lo_count);
    if (avg_count_per_lo = 0) then
      avg_count_per_lo = 1;
  end else
    avg_count_per_lo = 100000;

  for select distinct exa_lo_id
  from expressapp
  where exa_from_exc_id = 0 and
        exa_final_status = 0
  into lo_id
  do
  begin
    if (unclude_lo_in_holiday = 0) then
    begin
      -- Если параметром не задано, исключаем сотрудников в отпуске и больничном
      if (exists(select * from hr_voctravel, hr_orders, sa_worktype
      where hrv_from_c_id = :lo_id and
            hrv_date_begin <= current_date and
            coalesce(hrv_date_end_fact, hrv_date_end) >= current_date and
            hrv_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            sawt_base_type in (4, 5) and
            sawt_id not in (15, 20))) then
        Continue;
    end

    group_id = group_id + 1;
    if (group_id > 10) then
      group_id = 1;

    fill_count = 0;
    for select exa_id, exa_base_k_id
    from expressapp
    where exa_lo_id = :lo_id and
          exa_from_exc_id = 0 and
          exa_final_status = 0
    into exa_id, k_id
    do
    begin
--      if (exists(select * from operkred
--      where ok_from_k_id = :k_id and
--            ok_ol_is_active = 1 and
--            ok_type = 4 and
--            (ok_os <> 0 or ok_perc <> 0) and
--            datediff(day, ok_sh_date, ok_ol_date) > 7)) then
--        Continue;

      fill_count = fill_count + 1;

      update expressapp
      set exa_from_exc_id = 10,
          exa_date = current_date,
          exa_disb_date = current_date,
          exa_group_id = :group_id
      where exa_id = :exa_id;

      total_started = total_started + 1;

      if (fill_count = avg_count_per_lo) then
        Break;
    end
  end

  suspend;
end
