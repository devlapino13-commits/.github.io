CREATE PROCEDURE SA_OKLAD_LIST(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SAO_NAME_FIO VARCHAR(100),
    SAO_JOBLIST VARCHAR(100),
    SAO_OFFICE VARCHAR(100),
    SAO_DATE TYPE OF COLUMN SA_OKLAD.SAO_DATE,
    SAO_SUMMA TYPE OF COLUMN SA_OKLAD.SAO_SUMMA,
    SAO_MEMO TYPE OF COLUMN SA_OKLAD.SAO_MEMO,
    PS_IS_WORKING_STR VARCHAR(20),
    SAO_FROM_C_ID DECIMAL(18,0),
    SAO_SUMMA_OLD DECIMAL(18,4))
AS
declare variable sao_summa_tmp decimal(18,4);
declare variable ps_is_working integer;
begin
  for select distinct sao_from_c_id
      from sa_oklad
      into sao_from_c_id
  do
  begin
    sao_date = null;
    sao_summa = 0;
    select sum(sao_summa), max(sao_date)
    from sa_oklad
    where sao_from_c_id = :sao_from_c_id and
          sao_date < :rep_date_begin
    into :sao_summa, :sao_date;
    if (sao_summa is null) then
      sao_summa = 0;

    select c_name_fio
    from clients
    where c_id = :sao_from_c_id
    into :sao_name_fio;
    sao_joblist = '';
    sao_office = '';
    ps_is_working = 0;
    select jl_name, reg_code || ' ' || reg_name, ps_is_working
    from hr_pstatus, joblist, region
    where ps_from_c_id = :sao_from_c_id and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          jl_id = ps_from_jl_id and
          reg_unicode = ps_from_reg_id
    order by ps_is_working desc
    rows 1
    into :sao_joblist, :sao_office, :ps_is_working;
    if (ps_is_working = 1) then
      ps_is_working_str = 'Работает';
    else
    begin
      if (sao_joblist <> '') then
        ps_is_working_str = 'Уволен';
      else
        ps_is_working_str = 'Нет дейс.приказа';
    end
    sao_summa_old = :sao_summa;

    select sum(sao_summa), max(sao_date)
    from sa_oklad
    where sao_from_c_id = :sao_from_c_id and
          sao_date between :rep_date_begin and :rep_date_end and
          sao_summa <> 0
    into :sao_summa_tmp, :sao_date;
    if ((sao_summa_tmp is not null) and
        (sao_summa_tmp <> 0)) then
    begin
      sao_summa = sao_summa + :sao_summa_tmp;
      suspend;
    end
  end
end
