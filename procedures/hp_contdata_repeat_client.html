CREATE PROCEDURE HP_CONTDATA_REPEAT_CLIENT(
    CHECK_CELL_NO D_STR_255)
RETURNS (
    REC_NO D_INTEGER,
    CDA_ID D_ID,
    CELL_NO D_STR_255,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    KS_DATE_BEGIN D_DATE,
    FOLDER_NO D_STR_CODE,
    LO_NAME D_STR_100,
    REG_NAME D_STR_100)
AS
declare variable clt_id d_integer;
declare variable c_id d_id;
declare variable db_id d_id;
begin
  clt_id = f_cont_id(check_cell_no, null);
  if (clt_id is null) then
    exit;

  db_id = f_get_db_id();

  cell_no = check_cell_no;
  rec_no = 0;

  for select cda_id, cda_from_c_id, c_name_fio, c_birth_date
  from contdata, clients
  where cda_is_main = 1 and
        cda_is_delete = 0 and
        cda_status = 1 and
        cda_cont_kind = 0 and
        cda_from_clt_id = :clt_id and
        cda_from_c_id = c_id
  into cda_id, c_id, c_name_fio, c_birth_date
  do
  begin
    select ks_date_begin, ks_kfolder_no, c_name_fio, reg_code || ' ' || reg_name
    from knclient, kontstate, kontract, clients, region
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = ks_from_k_id and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999' and
          ks_from_k_id = k_id and
          ks_kc_id = c_id and
          ks_region_id = reg_unicode and
          (:db_id <> 81001 or k_stop_perc is null)
    order by ks_date_begin desc
    rows 1
    into ks_date_begin, folder_no, lo_name, reg_name;

    if (ks_date_begin is not null) then
    begin
      rec_no = rec_no + 1;
      suspend;
    end
  end
end
