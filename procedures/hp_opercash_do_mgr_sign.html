CREATE PROCEDURE HP_OPERCASH_DO_MGR_SIGN(
    IN_OPC_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable user_c_id d_id;
declare variable opc_is_signed d_bool;
declare variable opc_date d_date;
declare variable opc_office_id d_integer;
begin
  res_is_error = 0;
  res_text = 'Успешно подтвержден';

  select opc_is_signed, opc_office_id, opc_date
  from opercash
  where opc_id = :in_opc_id
  into opc_is_signed, opc_office_id, opc_date;

  if (row_count = 0) then
  begin
    res_is_error = 1;
    res_text = 'Документ не найден';
  end else
  if (opc_is_signed = 1) then
    res_text = 'Ранее подтвержден';
  else
  begin
    select un_from_c_id
    from username
    where un_id = :un_id
    into user_c_id;

    if (not exists(select * from hp_opercash_managers(:opc_office_id, :opc_date)
                   where id = :user_c_id)) then
    begin
      res_is_error = 1;
      res_text = 'Нет права подписи';
    end else
    begin
      update opercash
      set opc_is_signed = 1,
          opc_manager_id = :user_c_id,
          opc_signed_un_id = :un_id
      where opc_id = :in_opc_id;
    end
  end

  suspend;
end
