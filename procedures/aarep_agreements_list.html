CREATE PROCEDURE AAREP_AGREEMENTS_LIST(
    REG_ID D_INTEGER,
    REP_DATE D_DATE,
    IS_ACTUAL D_BOOL)
RETURNS (
    AG_ID D_INTEGER,
    AG_PARENT_ID D_INTEGER,
    AG_FROM_RD_ID D_INTEGER,
    AG_OBJECT_KIND D_ID,
    AG_OBJECT_ID D_ID,
    AG_FROM_INIT_REG_ID D_INTEGER,
    AG_FROM_REG_ID D_INTEGER,
    AG_CAPTION D_STR_255,
    AG_AGREEMENT_NO D_STR_CODE,
    AG_AGREEMENT_DATE D_DATE,
    AG_DATE_BEGIN D_DATE,
    AG_DATE_END D_DATE,
    AG_MEMO D_STR_255,
    AG_FROM_UN_ID D_ID,
    AG_DATETIME D_DATETIME,
    AG_STATUS D_INTEGER,
    AG_ELSOM_ACCOUNT D_STR_20,
    AG_ELSOM_ACCOUNT_CLEAR D_STR_20,
    AG_ELCARD_ACCOUNT D_STR_20,
    AL_ACCOUNT D_STR_20,
    AG_COMMIS_CARDS_MB D_PERCENT,
    AG_COMMIS_CARDS_OTHER D_PERCENT,
    AG_ELSOM_BANK D_STR_255,
    RD_CODE D_STR_255,
    RD_NAME D_STR_255,
    OBJECT_NAME D_STR_255,
    REG_NAME_INIT D_STR_CODE,
    REG_NAME D_STR_CODE,
    KIND_ACTIVITY D_STR_255,
    TAX_SERVICE_CODE D_STR_255,
    TSP_IS_SIGNED D_STR_CODE,
    PROF_IS_SIGNED D_STR_CODE,
    GPSA_IS_SIGNED D_STR_CODE,
    AEO_STATUS_STR D_STR_CODE,
    AG_MBANK_ACCOUNT D_STR_20,
    PAY_RD_NAME D_STR_100,
    AG_TAX_PERCENT D_PERCENT)
AS
declare variable aeo_status d_integer;
declare variable pay_rd_id d_integer;
begin
  for
    select
      ag_id,
      ag_parent_id,
      ag_from_rd_id,
      rd1.rd_code,
      rd1.rd_name,
      rd2.rd_memo,
      ag_object_kind,
      ag_object_id,
      c_name_fio,
      ag_from_init_reg_id,
      r2.reg_name,
      ag_from_reg_id,
      r1.reg_name,
      ag_caption,
      ag_agreement_no,
      ag_agreement_date,
      ag_date_begin,
      ag_date_end,
      ag_memo,
      ag_from_un_id,
      ag_datetime,
      ag_status,
      coalesce(ag_elsom_account, ''),
      ag_elcard_account,
      ag_from_al_id,
      ag_commis_cards_mb,
      ag_commis_cards_other,
      coalesce(ag_mbank_account, ''),
      ag_from_pay_rd_id,
      ag_tax_percent
    from agreements
      inner join region_list(:reg_id,1,-1,1)
        on id_list = ag_from_reg_id
      left join region as r1
        on r1.reg_unicode = ag_from_reg_id
      left join region as r2
        on r2.reg_unicode = ag_from_init_reg_id
      left join clients
        on c_id = ag_object_id
      left join refdata as rd2
        on rd2.rd_id = ag_elsom_bank_id
      inner join refdata as rd1
        on rd1.rd_id = ag_from_rd_id
    where iif(:is_actual = 0, true, :rep_date between coalesce(ag_date_begin, '01.01.1900')
      and coalesce(ag_date_end, '31.12.9999'))
  into
    :ag_id,
    :ag_parent_id,
    :ag_from_rd_id,
    :rd_code,
    :rd_name,
    :ag_elsom_bank,
    :ag_object_kind,
    :ag_object_id,
    :object_name,
    :ag_from_init_reg_id,
    :reg_name_init,
    :ag_from_reg_id,
    :reg_name,
    :ag_caption,
    :ag_agreement_no,
    :ag_agreement_date,
    :ag_date_begin,
    :ag_date_end,
    :ag_memo,
    :ag_from_un_id,
    :ag_datetime,
    :ag_status,
    :ag_elsom_account,
    :ag_elcard_account,
    :al_account,
    :ag_commis_cards_mb,
    :ag_commis_cards_other,
    :ag_mbank_account,
    :pay_rd_id,
    :ag_tax_percent
  do
  begin
    ag_elsom_account_clear = ag_elsom_account;
    if (ag_elsom_account <> '') then
      ag_elsom_account = left(ag_elsom_account, 4) || '****' || right(ag_elsom_account, 2);
    if (ag_mbank_account <> '' and char_length(ag_mbank_account) >= 12 ) then
      ag_mbank_account = left(ag_mbank_account, 6) || '****' || right(ag_mbank_account, 2);

    kind_activity = coalesce((
      select first 1 rd_name
      from agreement_feac
        inner join refdata
          on rd_id = afe_from_rd_id
      where afe_from_ag_id = :ag_id), 'все');

    tax_service_code = (
      select first 1
        left(cds_memo_3, 3)
      from clientdocs
      where cds_from_c_id = :ag_object_id
        and cds_from_pk_id in (44, 68, 71)
        and cds_is_delete = 0);

    tax_service_code = (
      select first 1
        left(cds_memo_3, 3)
      from clientdocs
      where cds_from_c_id = :ag_object_id
        and cds_from_pk_id in (44, 68, 71)
        and cds_is_delete = 0
        and coalesce(cds_memo_3,'') <> '');

    tsp_is_signed =
      case f_edoc_status('tsp.agreement', :ag_id, current_date)
        when 'empty' then 'Не заключался'
        when 'signed' then 'Подписан'
        when 'unsigned' then 'Не подписан'
        when 'cancelled' then 'Завершен'
      end;
    prof_is_signed =
      case f_edoc_status('prof_cust.agreement', :ag_id, current_date)
        when 'empty' then 'Не заключался'
        when 'signed' then 'Подписан'
        when 'unsigned' then 'Не подписан'
        when 'cancelled' then 'Завершен'
      end;
    gpsa_is_signed =
      case f_edoc_status('gpsa.agreement', :ag_id, current_date)
        when 'empty' then 'Не заключался'
        when 'signed' then 'Подписан'
        when 'unsigned' then 'Не подписан'
        when 'cancelled' then 'Завершен'
      end;

    aeo_status = -1;
    select aeo.aeo_status
    from agreement_eds_original aeo
    where aeo.aeo_from_ag_id = :ag_id
    order by aeo.aeo_datetime desc
    rows 1
    into :aeo_status;

    if (:aeo_status = -1) then
      aeo_status_str = '';
    else
      aeo_status_str = f_enum('agreement_eds_orig_status', :aeo_status, 'name');

    pay_rd_name = (select rd_str_id from refdata where rd_id = :pay_rd_id);
    suspend;
  end
end
