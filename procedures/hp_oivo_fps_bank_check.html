CREATE PROCEDURE HP_OIVO_FPS_BANK_CHECK(
    C_ID D_ID,
    PHONE D_STR_20,
    FPB_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255,
    FCB_ID D_INTEGER,
    FCB_STATUS D_SMALLINT)
AS
declare variable clt_id d_integer;
declare variable check_count d_integer;
declare variable last_check_dt d_datetime;
declare variable result_stale_minutes d_integer;
begin
    -- RFCRU-5828
    /* Проверка принадлежности номера phone клиенту c_id в банке fpb_id.
       res_code: 'success' - номер подтвержден банком,
                 'fail'    - номер не принадлежит клиенту,
                 'limit'   - превышен лимит проверок за сутки,
                 'error'   - результат из банка еще не получен либо
                             некорректные входные данные */
    res_code = 'error';
    res_text = '';
    fcb_id = null;
    fcb_status = null;

    -- результат старше этого интервала считаем устаревшим и
    -- отправляем номер на повторную проверку в банк
    result_stale_minutes = 30;

    if (coalesce(phone, '') = '') then
    begin
        res_text = 'Номер телефона не определен';
        suspend;
        exit;
    end

    if (fpb_id is null) then
    begin
        res_text = 'Банк СБП не выбран';
        suspend;
        exit;
    end

    if (not exists(select * from fps_banks
                   where fpb_id = :fpb_id and fpb_status = 1 and
                                  fpb_name_full is not null)) then
    begin
        res_text = 'Выбранный банк СБП не активен';
        suspend;
        exit;
    end

    clt_id = f_cont_id(phone, 'add');

    select fcb_id, fcb_status
    from fps_client_banks
    where fcb_from_c_id = :c_id and
          fcb_from_clt_id = :clt_id and
          fcb_from_fpb_id = :fpb_id
    into fcb_id, fcb_status;

    -- связки клиент + номер + банк еще нет (номер введен на этапе 17):
    -- создаем запись, триггеры отправят ее на проверку в банк
    if (fcb_id is null) then
    begin
        insert into fps_client_banks (fcb_from_c_id, fcb_from_clt_id, fcb_from_fpb_id, fcb_from_un_id)
        values (:c_id, :clt_id, :fpb_id, :un_id)
        returning fcb_id, fcb_status
        into fcb_id, fcb_status;

        res_text = 'Проверка номера в банке инициирована, результат еще не получен';
        suspend;
        exit;
    end

    -- номер подтвержден банком
    if (fcb_status = 2) then
    begin
        res_code = 'success';
        suspend;
        exit;
    end

    -- проверка в банке еще выполняется
    if (fcb_status = 0) then
    begin
        res_text = 'Проверка номера в банке выполняется, результат еще не получен';
        suspend;
        exit;
    end

    select max(fcc_datetime)
    from fps_client_banks_checks
    where fcc_from_fcb_id = :fcb_id
    into last_check_dt;

    -- свежий результат - это ответ банка на текущую проверку
    if (last_check_dt is not null and
        datediff(minute, last_check_dt, current_timestamp) <= result_stale_minutes) then
    begin
        res_code = 'fail';
        res_text = case fcb_status
                     when  1 then 'ФИО владельца номера в банке не совпадает с данными клиента'
                     when -1 then 'Номер не найден в выбранном банке'
                     else 'Принадлежность номера не подтверждена'
                   end;
        suspend;
        exit;
    end

    -- результат устарел: отправляем на повторную проверку
    -- с контролем лимита проверок за сутки
    check_count = 0;
    select count(*)
    from fps_client_banks_checks
    where fcc_from_fcb_id = :fcb_id
      and datediff(hour, fcc_datetime, current_timestamp) <= 24
    into check_count;

    if (check_count >= 5) then
    begin
        res_code = 'limit';
        res_text = 'Превышен лимит проверок номера в СБП. Попробуйте позже или выберите другой банк';
        suspend;
        exit;
    end

    -- более двух суток данные о проверках не храним
    delete from fps_client_banks_checks
    where fcc_from_fcb_id = :fcb_id
      and fcc_datetime < current_timestamp - 2;

    update fps_client_banks
    set fcb_status = 0,
        fcb_from_un_id = :un_id
    where fcb_id = :fcb_id;

    fcb_status = 0;
    res_text = 'Проверка номера в банке инициирована повторно, результат еще не получен';
    suspend;
end
