CREATE PROCEDURE MC_GET_CLIENT_INFO(
    MCL_ID D_ID)
RETURNS (
    MCL_LOGIN D_STR_CODE,
    REG_NAME D_STR_CODE,
    C_NAME_FIO_OFFICER D_STR_100,
    CELL_MAIN_OFFICER D_STR_20,
    MCL_BIRTH_DATE D_DATE,
    MCL_CLIENT_NAME D_STR_255)
AS
declare variable temp_cl_name d_str_100;
declare variable c_id d_id;
declare variable kf_officer d_id;
declare variable db_id d_integer;
declare variable mcl_user_code d_integer;
declare variable ma_id d_integer;
declare variable reg_id d_integer;
begin
  db_id = f_get_db_id();
  mcl_client_name = 'не задан'; -- поле добавлено по части кмп кг

  select mcl.mcl_from_c_id,
         mcl.mcl_login,
         mcl.mcl_birth_date,
         mcl.mcl_user_code
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id,
       :mcl_login,
       :mcl_birth_date,
       :mcl_user_code;

  if (c_id is not null) then
  begin
    -- получаем последний контракт клиента, офис и випс-а с которым работал клиент.
    -- не важно, получил клиент заем или нет, активный или закрытый
    select first 1
        maa.ma_id,
        reg.reg_name,
        kal.kal_new_kf_officer,
        kal.kal_new_reg_id
    from miniapp  maa
    inner join miniappclient macl on macl.mac_from_ma_id = maa.ma_id
        and macl.mac_from_c_id = :c_id
        and macl.mac_amount > 0
    inner join knaktlist kal on kal.kal_from_kf_id = maa.ma_from_kf_id
        and kal.is_delete = 0
        and current_date between kal.kal_date and kal.kal_date_end
    inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
    where maa.ma_date_end = '31.12.9999'
    order by ma_id desc
    into
        :ma_id,
        :reg_name,
        :kf_officer,
        :reg_id;

    if (ma_id is not null) then
    begin
        -- RFCKG-4609

        if (not exists(select 1
                   from hr_pstatus hps
                   where current_date between hps.ps_date_begin and hps.ps_date_end and
                         hps.ps_is_working = 1 and
                         hps.ps_from_c_id = :kf_officer)) then
        begin
            -- Если ВИПС, за которым закреплен клиент, уже не работает, выводим ФИО менеджера офиса
            select first 1 hps.ps_from_c_id
            from hr_pstatus hps
            inner join joblist jl on jl.jl_id = ps_from_jl_id
            inner join jobkind jk on jk.jk_id = jl.jl_kind
            where current_date between ps_date_begin and ps_date_end and
                  hps.ps_is_working = 1 and
                  hps.ps_from_reg_id = :reg_id and
                  jk.jk_system_code = 'JobSystem.Region_KM' -- Менеджер офиса
            into :kf_officer;
            
        end

        select
            cl.c_name_fio,
            f_cont_info('clients', cl.c_id, 'cell_main', 'ok')
        from clients cl
        where cl.c_id = :kf_officer
        into
            :c_name_fio_officer,
            :cell_main_officer;
    end

    if (db_id in (39801, 41701)) then
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :temp_cl_name;

      select iif(fname = 'уулу' or fname = 'кызы',patronymic,fname)
      from fio_separation(:temp_cl_name)
      into :mcl_client_name;

      if (mcl_user_code = 1) then mcl_client_name = 'Агент';
    end
  end

  suspend;
end
