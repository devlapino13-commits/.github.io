CREATE PROCEDURE HP_CALC_AMORT_CHANGE(
    REP_DATE DATE)
RETURNS (
    PCA_BILL_ID DECIMAL(18,0),
    PCA_BILL_LINE_NO INTEGER,
    PCA_AGE INTEGER,
    PCA_TOTAL_OS DECIMAL(18,4),
    PCA_TOTAL_AMORT DECIMAL(18,4),
    PCA_LIFETIME INTEGER,
    PCA_FROM_MC_ID DECIMAL(18,0),
    MC_LIFETIME INTEGER,
    OB_INVENT_STR VARCHAR(50),
    ML_NAME VARCHAR(255),
    PCA_LIFETIME_SALDO INTEGER,
    MC_LOOK_NAME VARCHAR(255))
AS
declare variable dont_calc smallint;
declare variable month_no integer;
declare variable year_no integer;
declare variable kolich decimal(18,4);
declare variable balance_cost decimal(18,4);
declare variable pca_start_year integer;
declare variable pca_start_month integer;
begin
  /* Procedure Text */
  year_no = extract(year from :rep_date);
  month_no = extract(month from :rep_date);
  for select oam_bill_id, oam_bill_line_no, oam_start_year, oam_start_month, sum(oal_lifetime_month), sum(oal_summa_os),
             sum(oal_summa_amort), sum(oal_dont_calc)
      from operamormain, operamorlist
      where oam_is_delete = 0 and
            oal_is_delete = 0 and
            oal_bill_id = oam_bill_id and
            oal_bill_line_no = oam_bill_line_no and
            (oal_year * 12 + oal_month) <= (:year_no * 12 + :month_no) and
            (oam_start_year * 12 + oam_start_month) <= (:year_no * 12 + :month_no)
      group by oam_bill_id, oam_bill_line_no, oam_calc_type, oam_koef, oam_start_year, oam_start_month
      having sum(oal_summa_os) > sum(oal_summa_amort)
      into :pca_bill_id, :pca_bill_line_no, :pca_start_year, :pca_start_month, :pca_lifetime, :pca_total_os,
           :pca_total_amort, :dont_calc
  do
  begin
    pca_from_mc_id = null;
    mc_lifetime = null;
    ob_invent_str = '';
    ml_name = '';
    select ob_from_mc_id, mc_lifetime, ml_name, ob_invent_str, mc_name
    from operbill, matcateg, matlist
    where ob_date_end >= '31.12.2999' and
          ob_bill_id = :pca_bill_id and
          ob_bill_line_no = :pca_bill_line_no and
          ob_is_delete = 0 and
          ob_date_begin <= ob_date_end and
          ob_from_mc_id = mc_id and
          ob_from_ml_id = ml_id rows 1
    into :pca_from_mc_id, mc_lifetime, ml_name, ob_invent_str, mc_look_name;

    balance_cost = 0;
    kolich = 0;
    select sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus),
           sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus)
    from opermat, operlist
    where om_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          om_bill_id = :pca_bill_id and
          om_bill_line_no = :pca_bill_line_no
    into :kolich, :balance_cost;
    if (kolich is null) then
      kolich = 0;
    if (balance_cost is null) then
      balance_cost = 0;

    if ((mc_lifetime is not null) and
        ((balance_cost <> 0) or (kolich <> 0))) then
    begin
      pca_age = (year_no * 12 + month_no) - (pca_start_year * 12 + pca_start_month) + 1;
      pca_age = pca_age - dont_calc;
      if (pca_age < 0) then
        pca_age = 0;
      pca_lifetime_saldo = pca_lifetime - pca_age;
      if ((pca_lifetime - pca_age) > 0) then
        suspend;
    end
  end
end
