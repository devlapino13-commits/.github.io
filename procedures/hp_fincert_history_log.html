CREATE PROCEDURE HP_FINCERT_HISTORY_LOG
RETURNS (
    FHL_ID D_ID,
    FHL_FROM_C_ID D_ID,
    FHL_OBJ_KIND D_ID,
    FHL_OBJ_ID D_ID,
    FHL_CREATE_DATE D_DATETIME,
    FHL_LAST_UPADTE D_DATETIME,
    FHL_STATUS D_SMALLINT,
    FHL_STATUS_STR D_STR_CODE,
    FHL_ERR_TEXT D_STR_LONG,
    FHL_RES_INCIDENT D_STR_255,
    FHL_INCIDENT_DATE_TIME D_DATETIME,
    FHL_LAW_REQ_REQUEST D_STR_CODE,
    FHL_LAW_REQ_NUMBER D_STR_64,
    FHL_LAW_REQ_TICKET D_STR_64,
    FHL_LAW_REQ_DATETIME_AT D_DATETIME,
    C_NAME_FIO D_STR_255)
AS
begin
  for select fhl_id,
             fhl_from_c_id,
             fhl_obj_kind,
             fhl_obj_id,
             fhl_create_date,
             fhl_last_upadte,
             fhl_status,
             f_enum('fincert_status', fhl_status, 'name'),
             fhl_err_text,
             fhl_res_incident,
             fhl_incident_date_time,
             f_enum('aim_status', fhl_law_req_request, 'name'),
             fhl_law_req_number,
             fhl_law_req_ticket,
             fhl_law_req_datetime_at,
             c_name_fio
  from fincert_history_log
  join clients  on c_id = fhl_from_c_id
  into :fhl_id,
       :fhl_from_c_id,
       :fhl_obj_kind,
       :fhl_obj_id,
       :fhl_create_date,
       :fhl_last_upadte,
       :fhl_status,
       :fhl_status_str,
       :fhl_err_text,
       :fhl_res_incident,
       :fhl_incident_date_time,
       :fhl_law_req_request,
       :fhl_law_req_number,
       :fhl_law_req_ticket,
       :fhl_law_req_datetime_at,
       :c_name_fio
  do
  suspend;
end
