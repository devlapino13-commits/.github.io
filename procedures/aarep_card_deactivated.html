CREATE PROCEDURE AAREP_CARD_DEACTIVATED(
    REGION_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    NAME_FIO VARCHAR(100),
    CARD_ID D_INTEGER,
    CARD_ACC_NO D_STR_CODE,
    DATE_ISSUE D_DATE,
    DATE_CLOSE D_DATE,
    KONTRACT_NO VARCHAR(20),
    DATE_LIMIT_OPEN D_DATE,
    LIMIT_AMOUNT D_CURRENCY)
AS
begin

  for select r.reg_name, cl.c_name_fio, cal.cal_card_id, cca.cca_code, cal.cal_date_issue, cal.cal_date_end,
    k.k_kontract_no, k.k_disb_date as date_limit_open,
    (
      select sum(ccl_new_limit)
      from
      (
        select l.ccl_new_limit,
          row_number() over(partition by l.ccl_card_id, l.ccl_from_k_id order by l.ccl_ol_date desc) as num
        from cardclientacc_limit l
        where l.ccl_card_id = cda.cda_card_id
      )q
      where q.num = 1
    )
  from carddeactivation cda
    inner join cardlist cal on cda.cda_card_id = cal.cal_card_id
    inner join cardplace cpl on cal.cal_last_cpl_id = cpl.cpl_id
    inner join cardclientacc cca on cal.cal_from_cca_id = cca.cca_id
    inner join region r on cpl.cpl_from_reg_id = r.reg_unicode
    inner join cardlistlnk lnk on cal.cal_card_id = lnk.cll_card_id
    inner join kontract k on lnk.cll_from_k_id = k.k_id
    inner join clients cl on lnk.cll_from_c_id = cl.c_id
  where
    cda.cda_date between :date_begin and :date_end
    and :date_begin between lnk.cll_date_begin and lnk.cll_date_end
    and exists(
                 select 1
                 from region_list(:region_id, 1, 3, 1) rl
                 where r.reg_unicode = rl.id_list
               )

  into :reg_name, :name_fio, :card_id, :card_acc_no, :date_issue, :date_close, :kontract_no,
       :date_limit_open, :limit_amount
  do
    suspend;
end
