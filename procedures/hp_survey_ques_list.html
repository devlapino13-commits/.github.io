CREATE PROCEDURE HP_SURVEY_QUES_LIST(
    ID INTEGER,
    FIND_TXT VARCHAR(100),
    STATE VARCHAR(100))
RETURNS (
    SVQ_ID INTEGER,
    SVQ_CAPTION VARCHAR(255),
    SVQ_MEMO VARCHAR(255),
    SVQ_TIME_START TIMESTAMP,
    SVQ_TIME_FINISH TIMESTAMP,
    STATUS_ID INTEGER,
    STATUS_TXT VARCHAR(100),
    RESULT_COUNT INTEGER,
    SVQ_TIME_START_STR VARCHAR(20),
    SVQ_TIME_FINISH_STR VARCHAR(20),
    SVQ_CORRECT_SVA_ID INTEGER)
AS
begin
  if (coalesce(state, '') = '') then
    state = 'all';
  if (id <> 0 or find_txt = '') then
  begin
    status_id = 0;
    status_txt = '';
    result_count = 0;
    for select svq_id, svq_caption, svq_memo, svq_time_start, svq_time_finish, svq_correct_sva_id
        from survey_ques
        where (:id <> 0 and
              svq_id = :id) or
              (:id = 0 and
              :find_txt = '')
        into :svq_id, :svq_caption, :svq_memo, :svq_time_start, :svq_time_finish, :svq_correct_sva_id
    do
    begin
      result_count = 0;
      if (svq_time_finish < current_timestamp) then
      begin
        status_id = -1;
        status_txt = 'Завершено';
      end
      else
      if (svq_time_start is null) then
      begin
        status_id = 0;
        status_txt = 'Время не назначено';
      end
      else
      if (svq_time_start > current_timestamp) then
      begin
        status_id = 1;
        status_txt = 'Запланировано';
      end
      else
      if (svq_time_start <= current_timestamp and
          (svq_time_finish is null or svq_time_finish >= current_timestamp)) then
      begin
        status_id = 2;
        status_txt = 'Действующее';
      end
      if ((id <> 0) or (state = 'all') or (state = 'active' and
          status_id = 2) or (state = 'actual' and
          status_id >= 0)) then
      begin
        select count(*)
        from survey_result
        where svr_from_svq_id = :svq_id
        into :result_count;

        svq_time_start_str = '';
        svq_time_finish_str = '';
        if (svq_time_start is not null) then
        begin
          svq_time_start_str = right('0' || extract(day from svq_time_start), 2) || '.' || right('0' || extract(month from svq_time_start), 2) || '.' || extract(year from svq_time_start) || ' ' || right('0' || extract(hour from svq_time_start), 2) || ':' || right('0' || extract(minute from svq_time_start), 2);
        end
        if (svq_time_finish is not null) then
        begin
          svq_time_finish_str = right('0' || extract(day from svq_time_finish), 2) || '.' || right('0' || extract(month from svq_time_finish), 2) || '.' || extract(year from svq_time_finish) || ' ' || right('0' || extract(hour from svq_time_finish), 2) || ':' || right('0' || extract(minute from svq_time_finish), 2);
        end
        suspend;
      end
    end
  end
  else
  begin
    for select distinct svq_id, svq_caption, svq_memo, svq_time_start, svq_time_finish
        from survey_ques, survey_answer
        where svq_id = sva_from_svq_id and
              (svq_caption like '%' || :find_txt || '%' or
              svq_memo like '%' || :find_txt || '%' or
              sva_caption like '%' || :find_txt || '%' or
              sva_memo like '%' || :find_txt || '%')
        into :svq_id, :svq_caption, :svq_memo, :svq_time_start, :svq_time_finish
    do
    begin
      status_id = 0;
      status_txt = '';
      result_count = 0;
      if (svq_time_finish < current_timestamp) then
      begin
        status_id = -1;
        status_txt = 'Завершено';
      end
      else
      if (svq_time_start is null) then
      begin
        status_id = 0;
        status_txt = 'Время не назначено';
      end
      else
      if (svq_time_start > current_timestamp) then
      begin
        status_id = 1;
        status_txt = 'Запланировано';
      end
      else
      if (svq_time_start <= current_timestamp and
          (svq_time_finish is null or svq_time_finish >= current_timestamp)) then
      begin
        status_id = 2;
        status_txt = 'Действующее';
      end
      if ((id <> 0) or (state = 'all') or (state = 'active' and
          status_id = 2) or (state = 'actual' and
          status_id >= 0)) then
      begin
        select count(*)
        from survey_result
        where svr_from_svq_id = :svq_id
        into :result_count;

        svq_time_start_str = '';
        svq_time_finish_str = '';
        if (svq_time_start is not null) then
        begin
          svq_time_start_str = right('0' || extract(day from svq_time_start), 2) || '.' || right('0' || extract(month from svq_time_start), 2) || '.' || extract(year from svq_time_start) || ' ' || right('0' || extract(hour from svq_time_start), 2) || ':' || right('0' || extract(minute from svq_time_start), 2);
        end
        if (svq_time_finish is not null) then
        begin
          svq_time_finish_str = right('0' || extract(day from svq_time_finish), 2) || '.' || right('0' || extract(month from svq_time_finish), 2) || '.' || extract(year from svq_time_finish) || ' ' || right('0' || extract(hour from svq_time_finish), 2) || ':' || right('0' || extract(minute from svq_time_finish), 2);
        end
        suspend;
      end
    end
  end
end
