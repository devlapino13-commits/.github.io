Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE API_GET_SQL(
    SYS_NAME VARCHAR(32),
    SYS_METHOD VARCHAR(255),
    SYS_VERSION INTEGER)
RETURNS (
    RES_SQL VARCHAR(4000),
    SR_META BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
begin
  if (sys_version is null) then
    sys_version = 0;

  select sr_sql, sr_meta
  from sqlrequest
  where sr_is_active = 1 and
        sr_system_name = :sys_name and
        sr_method = :sys_method and
        (:sys_version = 0 or
        sr_min_version <= :sys_version)
  order by sr_min_version desc
  rows 1
  into :res_sql, sr_meta;

  if (coalesce(res_sql, '') = '') then
    exception e_api 'Method not found';

  suspend;
-- При необходимости, можно логировать sys_params вместе с un_id в связке sr_id
end
