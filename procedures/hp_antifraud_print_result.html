CREATE PROCEDURE HP_ANTIFRAUD_PRINT_RESULT(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    ANTIFRAUD_CHECK_DATE D_DATE,
    ANTIFRAUD_CHECK_RESULT D_STR_100)
AS
declare variable client_found_antifraud d_bool;
declare variable c_id d_id;
declare variable c_inn d_str_20;
declare variable ma_id d_id;
begin
    /*
        Процедура выводит результат проверки по антифроду по конкретной заявке или конкретному контакту
        Если нет привязки результата проверки к заявке или контракту, то берется последняя проверка по клиенту (через сервис АФЦ)
        Если привязка есть, приоритет все равно отдается проверке через АФЦ
    */

    if (f_objkind(:object_kind, 'name') not in ('miniapp', 'kontract')) then
        exit;

    if (object_kind = f_objkind('miniapp', 'id')) then
    begin
        select mac.mac_from_c_id
        from miniappclient mac
        where mac.mac_from_ma_id = :object_id and
              mac.mac_amount > 0
        into :c_id;

        ma_id = :object_id;
    end
    else
    begin
        select kc.kc_from_c_id, k.k_from_ma_app_id
        from knclient kc
        inner join kontract k on kc.kc_from_k_id = k.k_id
        where kc.kc_from_k_id = :object_id and
              kc.kc_amount_approved > 0
        into :c_id, :ma_id;
    end

    c_inn = (select c_inn from clients where c_id = :c_id);

    -- Т.к. приоритетно мы должны брать проверку через сервис АФЦ, пытаемся получить именно ее
    select first 1
        afcl.afcl_check_result,
        cast(afcl.afcl_datetime as date)
    from antifraud_checks_link afcl
    where afcl.afcl_from_c_id = :c_id and
          afcl.afcl_object_kind = f_objkind('miniapp', 'id') and
          afcl.afcl_object_id = :ma_id and
          afcl.afcl_check_type = f_enum_id('antifraud_check_type', 'afc_service')
    order by afcl.afcl_id desc -- Берем самую последнюю проверку
    into
        :client_found_antifraud,
        :antifraud_check_date;

    -- Если не удалось получить проверку через АФЦ, пытаемся получить любую проверку, привязанную к заявке или контракту
    if (client_found_antifraud is null or antifraud_check_date is null) then
    begin
        select first 1
            afcl.afcl_check_result,
            cast(afcl.afcl_datetime as date)
        from antifraud_checks_link afcl
        where afcl.afcl_from_c_id = :c_id and
              afcl.afcl_object_kind = :object_kind and
              afcl.afcl_object_id = :object_id
        order by afcl.afcl_id desc
        into
            :client_found_antifraud,
            :antifraud_check_date;
    end

    -- Если не удалось получить результат по конкретной заявке или контракту, то берем последний результат по клиенту
    if (client_found_antifraud is null or antifraud_check_date is null) then
    begin
        -- Получаем информацию о последней проверке антифрод по клиенту
        select first 1
            case f_enum('antifraud_status', ac.ac_status, 'code')
                when 'was_found' then 1
                else 0
            end,
            f_datetostr(ac.ac_date)
        from antifraud_check ac
        where ac.ac_inn = :c_inn and
              -- Интересуют только статусы "Не найден" или "Найден", чтобы не вывести в документ некорретный результат
              ac.ac_status in (f_enum_id('antifraud_status', 'not_found'), f_enum_id('antifraud_status', 'was_found'))
        order by ac.ac_id desc
        into
            :client_found_antifraud,
            :antifraud_check_date;
    end

    antifraud_check_result =
        case :client_found_antifraud
            when 0 then 'информация отсутствует'
            when 1 then 'информация обнаружена'
            else null
        end;

    suspend;
end
