CREATE PROCEDURE HP_UXFRAME_SQL_EXPR(
    UX_CODE D_STR_CODE,
    FIELD_NAME D_STR_CODE)
RETURNS (
    SQL_TEXT D_STR_8100,
    RES_FIELD_LIST D_STR_8100)
AS
declare variable field_list d_str_8100;
declare variable ufd_source d_str_code;
declare variable ufd_is_table d_bool;
declare variable ufr_show_kind d_smallint;
declare variable ux_kind d_str_code;
declare variable fr_code d_str_code;
begin
  -- Приготовление Select sql для фрейма
  sql_text = '';
  field_list = '';
  res_field_list = '';

  if (field_name is null) then
    field_name = '';

  if (ufr_show_kind = 0) then
  begin
    suspend;
    exit;
  end

  select uxf$kind(ux_kind, 'code')
  from uxcode
  where ux_code = :ux_code
  into ux_kind;

  if (ux_kind <> 'uxframe') then
  begin
    select ucf_ufr_code
    from uxcodeframe
    where ucf_ux_code = :ux_code and
          ucf_is_mainframe = 1
    into fr_code;
  end else
    fr_code = :ux_code;

  select ufd_source, ufd_is_table, ufr_show_kind
  from uxfrdata, uxframe
  where ufr_code = :fr_code and
        ufd_ufr_code = ufr_code and
        ufd_kind = 0
  into ufd_source, ufd_is_table, ufr_show_kind;

  if (ufr_show_kind <> 0) then
  begin
    select list(ufp_origin_name, ', ')
    from uxfrparam
    where ufp_ufr_code = :fr_code and
          ufp_is_param = 0 and
          ufp_is_delete = 0 and
          ufp_in_use = 1 and
          (:field_name = '' or ufp_origin_name = :field_name)
    into res_field_list;
  end

  if (res_field_list <> '') then
  begin
    sql_text = 'select ' || res_field_list || ' from ' || ufd_source;

    if (ufd_is_table = 0) then
    begin
      select list('null')
      from uxfrparam
      where ufp_ufr_code = :fr_code and
            ufp_is_param = 1
      into field_list;
      if (field_list <> '') then
        sql_text = sql_text || '(' || field_list || ')';
    end
  end

  suspend;
end
