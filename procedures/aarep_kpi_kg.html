CREATE PROCEDURE AAREP_KPI_KG(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_LO_ID ID_TYPE)
RETURNS (
    REG_NAME VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    KOL_CLI_BEGIN INTEGER,
    KOL_CLI_END INTEGER,
    BALANCE_BEGIN CURR_TYPE,
    BALANCE_END CURR_TYPE,
    KOL_DISB_RE INTEGER,
    KOL_CLOSED INTEGER,
    KOL_CLOSED_WOFF INTEGER,
    KOL_DISB_ALL INTEGER,
    PLAN_OS CURR_TYPE,
    FACT_OS CURR_TYPE,
    DELINQ_BAL_1 CURR_TYPE,
    PAR1 CURR_TYPE,
    PLAN_CLI INTEGER,
    FACT_CLI INTEGER,
    MARK_CLI_PRIROST DECIMAL(15,2),
    PLAN_KP_PERC_MIN CURR_TYPE,
    PLAN_KP_PERC_MAX CURR_TYPE,
    PRIROST_KP CURR_TYPE,
    PRIROST_KP_PERC CURR_TYPE,
    MARK_KP DECIMAL(15,2),
    PERC_UDERJANIE_CLI DECIMAL(15,2),
    MARK_CLI_UDERJ DECIMAL(15,2),
    PLAN_DISB_CLI INTEGER,
    MARK_CLI_DISB DECIMAL(15,2),
    VOZVRAT_PERC DECIMAL(15,2),
    MARK_VOZVRAT DECIMAL(15,2),
    MARK_PAR DECIMAL(15,2),
    MARK_AVG DECIMAL(15,2),
    BRANCH_NAME VARCHAR(60))
AS
declare variable c_id id_type;
declare variable c_email varchar(100);
declare variable reg_id integer;
declare variable reg_id_prior integer;
begin
  reg_id_prior = 0;
  for select ps_from_c_id, c_email, ps_from_reg_id
      from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1)
      where :rep_lo_id = 0 and
            ps_from_c_id = c_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id > 1000000000003 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            ps_from_reg_id = id_list and
            coalesce((select sawt_id
                      from hr_voctravel, hr_orders, sa_worktype
                      where hrv_from_ho_id = ho_id and
                            ho_order_type = sawt_id and
                            hrv_from_c_id = ps_from_c_id and
                            :rep_date_end between hrv_date_begin and hrv_date_end_fact
                      order by hrv_date_begin desc
                      rows 1), 0) not in (16, 17, 18) -- Декретников не включает
      union
      select c_id, c_email, :rep_region_id
      from clients
      where :rep_lo_id <> 0 and
            c_id = :rep_lo_id
      order by 3
      into :c_id, :c_email, :reg_id
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into :officer_name;

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode
      into :reg_name, :branch_name;
    end

    /*
      AAREP_MEMO='';
      if (c_email is null) then c_email='';
      if (c_email like '%@molbulak.com')
      then AAREP_MEMO='Ok';
      else AAREP_MEMO='Не будет отправлено, нет EMAIL';
      */
    kol_cli_begin = null;
    kol_cli_end = null;
    balance_begin = null;
    balance_end = null;
    kol_disb_re = null;
    kol_closed = null;
    kol_closed_woff = null;
    kol_disb_all = null;
    plan_os = null;
    fact_os = null;
    delinq_bal_1 = null;
    par1 = null;
    plan_cli = null;
    fact_cli = null;
    mark_cli_prirost = null;
    plan_kp_perc_min = null;
    plan_kp_perc_max = null;
    prirost_kp = null;
    prirost_kp_perc = null;
    mark_kp = null;
    perc_uderjanie_cli = null;
    mark_cli_uderj = null;
    plan_disb_cli = null;
    mark_cli_disb = null;
    vozvrat_perc = null;
    mark_vozvrat = null;
    mark_par = null;
    mark_avg = null;
    select kol_cli_begin, kol_cli_end, balance_begin, balance_end, kol_disb_re, kol_closed, kol_closed_woff,
           kol_disb_all, plan_os, fact_os, delinq_bal_1, par1, plan_cli, fact_cli, mark_cli_prirost, plan_kp_perc_min,
           plan_kp_perc_max, prirost_kp, prirost_kp_perc, mark_kp, perc_uderjanie_cli, mark_cli_uderj, plan_disb_cli,
           mark_cli_disb, vozvrat_perc, mark_vozvrat, mark_par, mark_avg
    from aarep_kpi_kg_one(:rep_date_begin, :rep_date_end, :c_id)
    into :kol_cli_begin, :kol_cli_end, :balance_begin, :balance_end, :kol_disb_re, :kol_closed, :kol_closed_woff,
         :kol_disb_all, :plan_os, :fact_os, :delinq_bal_1, :par1, :plan_cli, :fact_cli, :mark_cli_prirost,
         :plan_kp_perc_min, :plan_kp_perc_max, :prirost_kp, :prirost_kp_perc, :mark_kp, :perc_uderjanie_cli,
         :mark_cli_uderj, :plan_disb_cli, :mark_cli_disb, :vozvrat_perc, :mark_vozvrat, :mark_par, :mark_avg;

    suspend;
  end
end
