CREATE PROCEDURE HP_FINCERT_GET_FILE(
    KIND D_STR_255,
    CHECK_DATE_TIME D_DATETIME = null,
    SHIFT_DATETIME D_BOOL = null)
RETURNS (
    FF_FILE_BODY D_BLOB_TEXT,
    FF_CREATE_DATETIME D_DATETIME,
    FF_FILE_DATETIME D_DATETIME)
AS
declare variable ff_file_kind d_integer;
begin
  ff_file_kind = f_enum_id('fincert_file', kind);

  if (check_date_time is null) then  --RFCRU-5763
    check_date_time = localtimestamp;

  if (shift_datetime is null) then
    shift_datetime = 0; -- делать сдвиг времени

  if (shift_datetime = 1) then
    check_date_time = dateadd(hour, -1, check_date_time);

  -- берем текущий загруженный файл, если время его действия наступило
  select ff_file_body, ff_create_datetime, ff_file_datetime
  from fincert_files
  where ff_file_kind = :ff_file_kind
    and ff_create_datetime <= :check_date_time
  into ff_file_body, ff_create_datetime, ff_file_datetime;

  -- иначе ищем в архиве
  if (ff_file_body is null) then
    select ff_file_body, ff_create_datetime, ff_file_datetime
    from fincert_files_arch
    where ff_file_kind = :ff_file_kind
      and ff_create_datetime <= :check_date_time
    order by ff_create_datetime desc
    rows 1
    into ff_file_body, ff_create_datetime, ff_file_datetime;

  suspend;
end
