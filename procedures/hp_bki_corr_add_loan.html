CREATE PROCEDURE HP_BKI_CORR_ADD_LOAN(
    K_ID D_ID,
    C_ID D_ID,
    EVENT_ID D_INTEGER,
    EVENT_DATE D_DATE,
    BKK_ID D_ID,
    NEW_PACKET_FORCED D_BOOL = 0,
    MA_ID D_INTEGER = null)
RETURNS (
    NEW_BEV_ID D_ID)
AS
declare variable counter d_integer;
declare variable packet_name d_str_code;
declare variable packet_id d_id;
declare variable bkt_id d_id;
declare variable packet_date d_date;
declare variable bkl_id d_id;
declare variable bkc_id d_id;
declare variable uid_str d_str_code;
declare variable rec_no d_integer;
declare variable object_kind d_integer;
declare variable object_id d_id;
declare variable packet_format d_smallint;
begin
  if (ma_id is not null) then
  begin
    object_kind = 11;
    object_id = ma_id;
  end else
  begin
    object_kind = 9;
    object_id = k_id;
  end

  select bkt_id, bkt_format
  from bki_packet_type
  where bkt_action = f_enum('bki_packet_action', 'correction', 'id')
    and bkt_from_bkk_id = :bkk_id
    and bkt_format = 0
  into bkt_id, packet_format;

  if (coalesce(new_packet_forced, 0) = 0) then
    select bkp_id
    from bki_packet
    where bkp_from_bkt_id = :bkt_id
      and bkp_status = -5
    into packet_id;

  if (packet_id is null) then
  begin
    packet_date = current_date;

    select coalesce(max(bkp_number), 0)
    from bki_packet, bki_packet_type
    where bkt_id = bkp_from_bkt_id
      and bkp_date = :packet_date
      and bkt_from_bkk_id = :bkk_id
      and bkt_format = 0
    into counter;

    select packet_name, new_counter
    from hp_bki_get_packet_name(:packet_date, (select bkk_code from bki_kind where bkk_id = :bkk_id), :counter, :packet_format)
    into packet_name, counter;
  
    insert into bki_packet (bkp_code, bkp_from_bkk_id, bkp_date, bkp_number, bkp_from_bkt_id, bkp_status)
    values (:packet_name, :bkk_id, :packet_date, :counter, :bkt_id, -5)
    returning bkp_id
    into :packet_id;
  end

  update or insert into bki_loans (bkl_from_bkp_id, bkl_object_kind, bkl_object_id)
  values (:packet_id, :object_kind, :object_id)
  matching (bkl_from_bkp_id, bkl_object_kind, bkl_object_id)
  returning bkl_id
  into :bkl_id;

  if (object_kind = 9) then
  begin
    select f_uuid_cbr(f_uuid('knclient', kc_id))
    from knclient
      where kc_from_k_id = :object_id
        and kc_from_c_id = :c_id
    into uid_str;
  end
  else
  begin
    if (exists(
        select 1
        from uuidlist udl
          where udl_object_kind = :object_kind
            and udl_object_id = :object_id))
    then
      begin
        select f_uuid_cbr(f_uuid('miniappclient', mac_id))
        from miniappclient mac
        where mac.mac_from_ma_id = :object_id
          and mac.mac_amount > 0
          and mac.mac_from_c_id = :c_id
        into uid_str;
      end
      else  -- значит уже uid перенесен на knclient, соответственно k_id должен быть найден
        select f_uuid_cbr(f_uuid('knclient', kc_id))
        from knclient kc, kontract k
          where k_id = kc_from_k_id
            and k_from_ma_app_id = :object_id
            and kc_from_c_id = :c_id
        into uid_str;
  end

  select coalesce(max(bev_rec_no), 0) + 1
  from bki_packet, bki_loans, bki_clients, bki_events
  where bkp_id = :packet_id
    and bkl_from_bkp_id = bkp_id
    and bkc_from_bkl_id = bkl_id
    and bev_from_bkc_id = bkc_id
  into rec_no;

  insert into bki_clients (bkc_from_bkl_id, bkc_from_c_id, bkc_uid_bkd_id)
  values (:bkl_id, :c_id, f_bki_data_id(:uid_str))
  returning bkc_id
  into bkc_id;

  insert into bki_events (bev_from_bkc_id, bev_rec_no, bev_event_id, bev_event_date, bev_tag)
  values (:bkc_id, :rec_no, :event_id, :event_date, 5)
  returning bev_id
  into :new_bev_id;

  suspend;
end
