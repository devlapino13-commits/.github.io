CREATE PROCEDURE AAREP_SALARY_SHORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER)
RETURNS (
    ZP_PERSONAL_ID DECIMAL(18,0),
    ZP_PERSONAL_FIO VARCHAR(100),
    ZP_FROM_SAC_ID INTEGER,
    ZP_SC_NAME VARCHAR(100),
    ZP_SUMMA DECIMAL(18,4),
    ZP_KOL_1_OTPUSK DECIMAL(18,4),
    ZP_SUMMA_1_OTPUSK DECIMAL(18,4),
    ZP_SUMMA_1_TOTAL DECIMAL(18,4),
    ZP_KOL_2_OTPUSK DECIMAL(18,4),
    ZP_SUMMA_2_OTPUSK DECIMAL(18,4),
    ZP_SUMMA_2_TOTAL DECIMAL(18,4),
    ZP_KOL_3_OTPUSK DECIMAL(18,4),
    ZP_SUMMA_3_OTPUSK DECIMAL(18,4),
    ZP_SUMMA_3_TOTAL DECIMAL(18,4))
AS
declare variable zp_kol integer;
declare variable zp_year integer;
declare variable zp_month integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable osm_year_month integer;
declare variable zp_summa_tmp decimal(18,4) = 0;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  for select ps_from_c_id, ps_c_name_fio
      from hr_all_personal_new_pstatus(:report_region_id, :report_date_begin, :report_date_end)
      into :zp_personal_id, :zp_personal_fio
  do
  begin
    for select sac_id, sac_name
        from sa_config
        where sac_id in (13, 18, 21, 8, 9)
        into :zp_from_sac_id, :zp_sc_name
    do
    begin
      zp_summa = 0;
      zp_kol_1_otpusk = 0;
      zp_summa_1_otpusk = 0;
      zp_summa_1_total = 0;
      zp_kol_2_otpusk = 0;
      zp_summa_2_otpusk = 0;
      zp_summa_2_total = 0;
      zp_kol_3_otpusk = 0;
      zp_summa_3_otpusk = 0;
      zp_summa_3_total = 0;

      for select osm_year_month, sum(osd_value)
          from opersalmain, opersaldetail
          where osm_id = osd_from_osm_id and
                osm_ol_is_active = 1 and
                osm_ol_date between :report_date_begin and :report_date_end and
                osm_from_c_id = :zp_personal_id and
                osd_from_sac_id = :zp_from_sac_id
          group by osm_year_month
          into :osm_year_month, :zp_summa_tmp
      do
      begin
        zp_month = mod(osm_year_month, 100);
        zp_year = (osm_year_month - zp_month) / 100;
        zp_kol = (:report_year * 12 + :report_month) - (:zp_year * 12 + :zp_month);

        zp_summa = zp_summa + zp_summa_tmp;
        if (zp_kol = 3) then
        begin
          if (zp_from_sac_id = 13) then
            zp_kol_1_otpusk = zp_kol_1_otpusk + :zp_summa / 8;
          if (zp_from_sac_id = 21) then
            zp_summa_1_otpusk = zp_summa_1_otpusk + :zp_summa;
          if (zp_from_sac_id = 9) then
            zp_summa_1_total = zp_summa_1_total + :zp_summa;
        end
        if (zp_kol = 2) then
        begin
          if (zp_from_sac_id = 13) then
            zp_kol_2_otpusk = zp_kol_2_otpusk + :zp_summa / 8;
          if (zp_from_sac_id = 21) then
            zp_summa_2_otpusk = zp_summa_2_otpusk + :zp_summa;
          if (zp_from_sac_id = 9) then
            zp_summa_2_total = zp_summa_2_total + :zp_summa;
        end
        if (zp_kol = 1) then
        begin
          if (zp_from_sac_id = 13) then
            zp_kol_3_otpusk = zp_kol_3_otpusk + :zp_summa / 8;
          if (zp_from_sac_id = 21) then
            zp_summa_3_otpusk = zp_summa_3_otpusk + :zp_summa;
          if (zp_from_sac_id = 9) then
            zp_summa_3_total = zp_summa_3_total + :zp_summa;
        end
      end
      suspend;
    end
  end

end
