CREATE PROCEDURE HP_EDOC_BLANK_INS(
    EDL_ID D_ID,
    LANG D_STR_10 = null)
RETURNS (
    BLANK_COUNT D_INTEGER)
AS
declare variable edc_id d_integer;
declare variable blank_date d_date;
declare variable edf_id d_integer;
declare variable bln_id d_integer;
declare variable bls_id d_id;
declare variable edf_add_code d_str_code;
declare variable edb_id d_id;
declare variable edb_status d_integer;
declare variable edl_object_id d_id;
declare variable err_text d_str_255;
declare variable edl_object_kind d_integer;
declare variable edf_for_each_client d_smallint;
declare variable c_id d_id;
declare variable pr_id d_integer;
declare variable un_id d_id;
declare variable tmp_obj_kind d_integer;
declare variable tmp_obj_id d_id;
declare variable blob_date d_date;
declare variable edf_date_kind d_smallint;
declare variable edf_need_approve d_smallint;
declare variable blank_lang d_str_lang;
declare variable db_id d_integer;
declare variable edf_prepare_kind d_integer;
declare variable edn_blank_list d_str_long;
declare variable edc_code d_str_code;
declare variable need_to_continue d_smallint;
begin
  -- Процедура заносит бланки в очередь для приготовления pdf сервисом
  db_id = f_get_db_id();
  blank_count = 0;

  -- Получаем проект. Он должен быть действующим на дату или в режиме тестирования
  select edl_from_edc_id, edl_blank_date, edl_object_kind, edl_object_id, edl_create_un_id, edc_code
  from edoc_list, edoc_cfg
  where edl_id = :edl_id and
        edl_from_edc_id = edc_id
  into edc_id, blank_date, edl_object_kind, edl_object_id, un_id, edc_code;

  -- По недействующим проектам ничего не выполняем
  if (edc_id is null) then
  begin
    suspend;
    exit;
  end

  if (edc_code = 'staff.orders') then
  begin
    for select prb_blank_bln_id, prb_blank_blob
    from ux$preblanks
    where prb_prepared = 1 and
          prb_selected = 1 and
          prb_blank_bls_id is not null
    as cursor cur
    do
    begin
      edf_id = null;

      select edf_id
      from edoc_frx
      where edf_from_edc_id = :edc_id and
            edf_from_bln_id = cur.prb_blank_bln_id
      into edf_id;

      if (edf_id is not null) then
      begin
        blank_count = blank_count + 1;

        insert into edoc_blank (edb_from_edf_id, edb_from_edl_id, edb_from_bln_id, edb_blob, edb_status)
        values (:edf_id, :edl_id, cur.prb_blank_bln_id, cur.prb_blank_blob, 1);
      end else
        exception e_message 'Бланк не настроен в электронном документе';
    end

    if (blank_count = 0) then
      exception e_message 'Бланки не приготовлены';
    else
      blank_count = 0;

    suspend;
    exit;
  end

  for select edf_id, edf_add_code, edf_for_each_client, edf_date_kind, edf_need_approve, edf_prepare_kind, edc_code
      from edoc_frx, edoc_cfg
      where edc_id = :edc_id and
            edf_from_edc_id = edc_id and
            :blank_date between edf_date_begin and coalesce(edf_date_end, :blank_date) and
            edf_prepare_kind <> 0
      into edf_id, edf_add_code, edf_for_each_client, edf_date_kind, edf_need_approve, edf_prepare_kind, edc_code
  do
  begin
    select need_to_continue
    from hp_check_edf(:edl_object_kind, :edl_object_id, :edf_add_code, :edc_code)
    into need_to_continue;

    if (need_to_continue = 1) then
      continue;

    if (edf_prepare_kind = 6) then
    begin
      edn_blank_list = '';

      select edn_blank_list
      from edoc_conf
      where edn_edl_id = :edl_id
      into edn_blank_list;

      if (edn_blank_list <> '') then
      begin
        for select res_str
        from hp_split_str(:edn_blank_list, ',')
        into :bln_id
        do
        begin
          blank_count = blank_count + 1;

          insert into edoc_blank (edb_from_edf_id, edb_from_edl_id, edb_from_c_id, edb_from_bln_id, edb_from_bls_id,
                                  edb_status, edb_need_approve)
          values (:edf_id, :edl_id, null, :bln_id, null, 0, :edf_need_approve);
        end
      end

      Continue;
    end

    pr_id = null;
    select pr_id
    from hp_edoc_object_info(:edl_object_kind, :edl_object_id, :un_id, :blank_date)
    into pr_id;

    blob_date = blank_date;
    if (edf_date_kind = 1) then -- версию бланка нужно брать на дату выдачи займа
    begin
      tmp_obj_kind = edl_object_kind;
      tmp_obj_id = edl_object_id;
      if (f_objkind(edl_object_kind, 'name') = 'com_list') then
      begin
        select cml_object_kind, cml_object_id
        from com_list
        where cml_id = :edl_object_id
        into tmp_obj_kind, tmp_obj_id;
      end

      select blank_date
      from hp_edoc_object_info(:tmp_obj_kind, :tmp_obj_id, :un_id, null)
      into blob_date;
    end

    -- Получаем список необходых языков бланка для формирования бланка
    for select blank_lang
        from hp_edoc_blank_lang(:edc_id, :edl_object_kind, :edl_object_id, :edf_add_code)
        into :blank_lang
    do
    begin
      bln_id = null;
      bls_id = null;
      err_text = '';

      if (lang is not null and
          (select edc.edc_code
           from edoc_list edl
           inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
           where edl.edl_id = :edl_id) = 'asp.agreement') then
        blank_lang = :lang;

      select bln_id, bls_id, err_text
      from hp_edoc_get_blank_info(:edf_id, :pr_id, :blank_date, :blank_lang, :blob_date)
      into bln_id, bls_id, err_text;

      if (err_text <> '') then
        exception e_default err_text;

      if (bln_id is null and
          bls_id is null) then
        continue;

      tmp_obj_id = edl_object_id;
      if (edf_for_each_client = 0) then
        tmp_obj_id = null;

      for select c_id
          from hp_edoc_sign_clients(:edc_id, :edl_object_kind, :tmp_obj_id, :edf_add_code)
          into c_id
      do
      begin
        edb_id = null;

        --Исключаю АНКЕТУ ПОФТЭ для лиц дающих согласие----------
        if (:db_id in (39801, 41701) and
            :edl_object_kind = f_objkind('miniapp', 'id') and
            :edf_id = 2 and --Анкета ПОФТЭ
            exists(select *
                   from miniappclient mac
                   where mac.mac_from_ma_id = :edl_object_id and
                         mac.mac_from_c_id = :c_id and
                         mac.mac_is_poruch = -1)) then
          continue;
        ----------------------------------------------------------
        if (bln_id is not null) then
        begin
          select edb_id, edb_status
          from edoc_blank
          where edb_from_edl_id = :edl_id and
                edb_from_edf_id = :edf_id and
                edb_from_bln_id = :bln_id and
                coalesce(edb_from_c_id, 0) = coalesce(:c_id, 0)
          into edb_id, edb_status;
        end
        else
        begin
          select edb_id, edb_status
          from edoc_blank
          where edb_from_edl_id = :edl_id and
                edb_from_edf_id = :edf_id and
                edb_from_bls_id = :bls_id and
                coalesce(edb_from_c_id, 0) = coalesce(:c_id, 0)
          into edb_id, edb_status;
        end

        if (edb_id is not null) then
        begin
          if (edb_status not in (1, 2)) then
          begin
            blank_count = blank_count + 1;

            update edoc_blank
            set edb_from_bln_id = :bln_id,
                edb_from_bls_id = :bls_id,
                edb_blob = null,
                edb_status = 0,
                edb_need_approve = :edf_need_approve
            where edb_id = :edb_id;
          end
        end
        else
        begin
          blank_count = blank_count + 1;

          insert into edoc_blank (edb_from_edf_id, edb_from_edl_id, edb_from_c_id, edb_from_bln_id, edb_from_bls_id,
                                  edb_status, edb_need_approve)
          values (:edf_id, :edl_id, :c_id, :bln_id, :bls_id, 0, :edf_need_approve);
        end
      end
    end
  end

  -- Если бланк заранее готов (имеется BLS_ID), то отмечаем его приготовленным
  update edoc_blank
  set edb_status = 1
  where edb_from_edl_id = :edl_id and
        edb_from_bls_id is not null and
        edb_status <> 1;

  -- Если бланк готовится online, также отмечаем его приготовленным
  for select edf_id
      from edoc_frx
      where edf_from_edc_id = :edc_id and
            edf_is_online = 1
      into edf_id
  do
    update edoc_blank
    set edb_status = 1
    where edb_from_edl_id = :edl_id and
          edb_from_edf_id = :edf_id and
          edb_status <> 1;

  suspend;
end
