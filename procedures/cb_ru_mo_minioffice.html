CREATE PROCEDURE CB_RU_MO_MINIOFFICE(
    REP_DATE_END DATE)
RETURNS (
    MANAGER_NAME STR_100,
    REG_NAME STR_100,
    LO_STAGE_MONTH_FULL DECIMAL(15,1),
    KOL_DISB_CL INTEGER)
AS
declare variable reg_id integer;
declare variable manager_id id_type;
declare variable rep_date_begin date;
declare variable jk_system_code str_100;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  for select reg_unicode
      from region
      where reg_logic_label = '.mini_office.'
      into reg_id
  do
  begin
    manager_id = null;
    select ps_from_c_id, jk_system_code
    from hr_pstatus, joblist, jobkind
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_status = 1 and
          ps_is_working = 1 and
          ps_from_reg_id = :reg_id and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id
--    and jk_system_code = 'JobSystem.Region_KM'
    rows 1
    into manager_id, jk_system_code;

    manager_name = null;
    select c_name_fio
    from clients
    where c_id = :manager_id
    into manager_name;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    lo_stage_month_full = null;
    select lo_stage_month_full
    from hp_get_lo_stage_full(:manager_id, :rep_date_end, :jk_system_code, 0)
    into lo_stage_month_full;

--  Расчет кол-во выдач
    kol_disb_cl = 0;
    select count(distinct kc_from_c_id)
    from kontstate, knclient
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_region_id = :reg_id and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0
    into kol_disb_cl;

    if (kol_disb_cl > 0 or exists(select *
                                  from oksummary
                                  where oks_region_id = :reg_id and
                                        :rep_date_end between oks_date_begin and oks_date_end and
                                        oks_balance > 0 and
                                        oks_close_state = 0)) then
      suspend;

  end
end
