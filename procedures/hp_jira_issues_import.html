CREATE PROCEDURE HP_JIRA_ISSUES_IMPORT(
    ISSUE_ID D_INTEGER,
    ISSUE_CODE D_STR_CODE,
    ISSUE_DEV_LOGIN D_STR_CODE,
    ISSUE_DEV_NAME D_STR_100,
    ISSUE_CREATED D_STR_CODE,
    ISSUE_TYPE D_STR_CODE,
    ISSUE_STATE D_STR_CODE,
    ISSUE_ESTIMATE_SECS D_STR_CODE,
    ISSUE_ESTIMATE_PERIOD D_STR_CODE,
    ISSUE_LEVEL D_STR_CODE,
    ISSUE_PROJECT D_STR_CODE,
    ISSUE_PRIORITY D_STR_CODE,
    ISSUE_TITLE D_STR_4096,
    ISSUE_MAIN_ID D_INTEGER = null)
AS
declare variable dev_id d_id;
declare variable state_id d_smallint;
declare variable type_id d_integer;
declare variable priority_id d_smallint;
declare variable created_datetime d_datetime;
begin
  select jrd_id
  from jira_dev
  where jrd_login = :issue_dev_login
  into :dev_id;

  if (dev_id is null) then
  begin
    insert into jira_dev (jrd_login, jrd_fio)
    values (:issue_dev_login, :issue_dev_name)
    returning jrd_id
    into :dev_id;
  end

  issue_title = left(coalesce(issue_title, ''), 1024);

  issue_created = left(issue_created, 19);
  created_datetime = replace(issue_created, 't', ' ');
  if (f_get_db_id() = 81001) then
    created_datetime = dateadd(hour, -3, created_datetime);

  if (issue_estimate_secs = 'none') then
    issue_estimate_secs = '0';

  issue_estimate_secs = replace(issue_estimate_secs, ' ', '');

  if (issue_estimate_secs = 0) then
    issue_estimate_period = '-';

  select jrs_id
  from jira_states
  where jrs_text = :issue_state
  into :state_id;

  if (state_id is null) then
  begin
    insert into jira_states (jrs_text)
    values (:issue_state)
    returning jrs_id
    into :state_id;
  end

  select jrt_id
  from jira_types
  where jrt_text = :issue_type
  into :type_id;

  if (type_id is null) then
  begin
    insert into jira_types (jrt_text)
    values (:issue_type)
    returning jrt_id
    into :type_id;
  end

  if (coalesce(issue_priority, '') = '') then
    issue_priority = 'Средний';

  priority_id = null;
  priority_id = case issue_priority
    when 'Средний' then 0
    when 'Низкий' then -1
    when 'Критичный' then 1
  end;

  if (priority_id is null) then
    exception e_message 'Неопределенный приоритет: ' || issue_priority;

  if (exists(select * from jira_issues where jri_issue_id = :issue_id)) then
  begin
    update jira_issues
    set jri_parent_id = :issue_main_id,
        jri_dev_id = :dev_id,
        jri_type_id = :type_id,
        jri_jira_code = :issue_code,
        jri_title = :issue_title,
        jri_project = :issue_project,
        jri_level = :issue_level,
        jri_state_id = :state_id,
        jri_estimate_secs = :issue_estimate_secs,
        jri_estimate_period = :issue_estimate_period,
        jri_created = :created_datetime
    where jri_issue_id = :issue_id and
          (jri_parent_id is distinct from :issue_main_id or
           jri_dev_id is distinct from :dev_id or
           jri_type_id is distinct from :type_id or
           jri_jira_code is distinct from :issue_code or
           jri_title is distinct from :issue_title or
           jri_project is distinct from :issue_project or
           jri_level is distinct from :issue_level or
           jri_state_id is distinct from :state_id or
           jri_estimate_secs is distinct from :issue_estimate_secs or
           jri_estimate_period is distinct from :issue_estimate_period or
           jri_created is distinct from :created_datetime);
  end
  else
  begin
    insert into jira_issues (jri_issue_id, jri_parent_id, jri_dev_id, jri_type_id, jri_jira_code, jri_title, jri_project, jri_level,
                             jri_state_id, jri_estimate_secs, jri_estimate_period, jri_created)
    values (:issue_id, :issue_main_id, :dev_id, :type_id, :issue_code, :issue_title, :issue_project, :issue_level,
            :state_id, :issue_estimate_secs, :issue_estimate_period, :created_datetime);
  end
end
