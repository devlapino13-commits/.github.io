CREATE PROCEDURE HP_DO_KN_OPERBOOK_PERC_ONLINE(
    REP_DATE D_DATE,
    REP_K_ID D_STR_CODE,
    GET_COUNT D_SMALLINT,
    ROWS_COUNT D_INTEGER)
RETURNS (
    RES_K_ID D_ID,
    RES_KONTRACT_NO D_STR_255)
AS
begin
  -- получаем список займов, по которым нужно пересроить проводки на конец месяца
  -- если for select ничего не дал, ничего не возвращаем. На стороне клиента это будет означать завершение работы
  -- если хоть одна запись вернулась, из клиента заново запускаем данную процедуру
  -- если возвращаемый res_k_id <> 0, кладем его на виртуальную таблицу, в конце обработаем их отдельно

  -- если дата не передана, значит это запрос на получение количества кредитов для пересчета,
  -- чтобы установить максимальное значение ProgressBar

  if (rep_date is null) then
    exception e_message 'Дата не указана';

  res_kontract_no = '';
  if (coalesce(get_count, 0) = 1) then
  begin
    select count(distinct oks_from_k_id)
    from kn_oksummary
    where oks_month_state = -1 and
          oks_date_begin = :rep_date
    into :res_k_id;
    if (res_k_id is null) then
      res_k_id = 0;

    suspend;
    exit;
  end

  if (rep_date <> f_date_month_end(rep_date)) then
    exception e_message 'Дата не является последним днем месяца';

  if (coalesce(rep_k_id, '') = '') then rep_k_id = '0';

  for select distinct oks_from_k_id
      from kn_oksummary
      where oks_month_state = -1 and
            oks_date_begin = :rep_date and
            oks_from_k_id > :rep_k_id
      rows :rows_count
      into :res_k_id
  do
  begin
    execute procedure hp_do_kn_operbook_one_online(:res_k_id, current_date, null, :rep_date - 1);
    when any do
    begin
      res_kontract_no = '';

      select k_kontract_no
      from kontract
      where k_id = :res_k_id
      into :res_kontract_no;

      if (gdscode in (335544345, 335544451, 335544336)) then
        res_kontract_no = res_kontract_no || ' -- не удалось из-за блокировки';
      else
        res_kontract_no = res_kontract_no || ' -- код ошибки: ' || gdscode;

      suspend;
    end
  end

  res_kontract_no = '';
  if (res_k_id is null) then res_kontract_no = 'finish';
  suspend;
end
