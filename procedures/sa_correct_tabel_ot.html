        s.sat_day_16 = iif(s.sat_day_16 = 0 and :sat_day_16 > 0, :sat_day_16, s.sat_day_16),
        s.sat_day_17 = iif(s.sat_day_17 = 0 and :sat_day_17 > 0, :sat_day_17, s.sat_day_17),
