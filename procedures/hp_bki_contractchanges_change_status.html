CREATE PROCEDURE HP_BKI_CONTRACTCHANGES_CHANGE_STATUS(
    BOD_ID D_ID NOT NULL,
    NEW_STATUS D_INTEGER NOT NULL)
AS
declare variable bod_add d_str_long;
declare variable jso d_integer;
declare variable change_date d_date;
declare variable k_id d_id;
begin
  select bod_add, bod_object_id
  from bki_once_data
  where bod_id = :bod_id
  into bod_add, k_id;

  jso = json.parse(bod_add);
  change_date = json.obj_get_val(jso, 'applyDate');

  if (new_status = 0 or new_status = 1) then --откат в статус «Новый»/перевод в статус «Готов»
  begin
    json.obj_set_s(jso, 'status', new_status);
    json.obj_set_s(jso, 'dontSend', 1);
  end
  else
  if (new_status = 2) then  --на отправку в БКИ
  begin
    update or insert into bki_buffer (bkb_from_bkk_id, bkb_object_kind, bkb_object_id, bkb_date, bkb_tag, bkb_type)
    values (1, 9, :k_id, :change_date, 55, 0)
    matching (bkb_from_bkk_id, bkb_object_kind, bkb_object_id, bkb_date, bkb_type);

    json.obj_set_s(jso, 'dontSend', 0);
    json.obj_set_s(jso, 'status', new_status);
  end
  else
  if (new_status = -1) then                    --отмена отправки
  begin
    json.obj_set_s(jso, 'dontSend', 1);
    json.obj_set_s(jso, 'status', 1);
  end
  else
    exception e_message 'Статус не может быть изменен';

  update bki_once_data
    set bod_add = json.obj_to_json(:jso)
  where bod_id = :bod_id;
end
