CREATE PROCEDURE HP_GET_FIO_COMBO(
    TXT1 VARCHAR(100),
    TXT2 VARCHAR(200))
RETURNS (
    RES_TEXT VARCHAR(100))
AS
begin
  if ((txt1 <> '') and
      (txt2 = '')) then
  begin
    for select distinct fl_name2
        from fiolist
        where fl_name1 = :txt1 and
              fl_name2 <> '' and
              fl_name2 is not null
        into :res_text
    do
    begin
      res_text = left(res_text, 1) || lower(right(res_text, char_length(:res_text) - 1));
      suspend;
    end
  end
  if ((txt1 <> '') and
      (txt2 <> '')) then
  begin
    for select distinct fl_name3
        from fiolist
        where fl_name1 = :txt1 and
              fl_name2 = :txt2 and
              fl_name3 <> '' and
              fl_name3 is not null
        into :res_text
    do
    begin
      res_text = left(res_text, 1) || lower(right(res_text, char_length(:res_text) - 1));
      suspend;
    end
  end
end
