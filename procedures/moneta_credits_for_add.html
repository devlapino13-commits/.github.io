CREATE PROCEDURE MONETA_CREDITS_FOR_ADD
RETURNS (
    ESM_ID ID_TYPE,
    TOKEN STR_100,
    DISB_AMOUNT CURR_TYPE,
    MNT_DESCRIPTION STR_255,
    MNT_ID D_STR_10)
AS
declare variable c_id id_type;
declare variable k_id id_type;
declare variable k_kontract_no str_100;
declare variable fio str_100;
declare variable ag_id d_integer;
declare variable k_disb_date d_date;
begin
  -- Процедура предоставляет список платежей, по которым должны быть перечисления (выдачи) через Монету на карты клиентам
  for select em.esm_id, em.esm_account, em.esm_amount, em.esm_object_id, em.esm_from_k_id
      from extsysmain em
      where em.esm_project = 'moneta_disb' and
            em.esm_action = 'CardDisb' and
            em.esm_status = 0
      into esm_id, token, disb_amount, c_id, k_id
  do
  begin
    if (k_id is not null) then
    begin
      select k_kontract_no,
             k_disb_date,
             iif(pr.pr_code containing '.oivo_pack.', f_get_const('moneta_oivo_acc'), f_get_const('moneta_acc'))
      from kontract kont
      inner join product pr on pr.pr_id = kont.k_from_pr_id
      where kont.k_id = :k_id
      into k_kontract_no,
           k_disb_date,
           mnt_id;

      if (k_disb_date is distinct from current_date) then  -- Контроль: если дата отличается не выдаем
        continue;

      select cl.c_name_fio
      from knclient kncl
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      where kncl.kc_amount_approved > 0
      and kncl.kc_from_k_id = :k_id
      and kncl.kc_from_c_id = :c_id
      into :fio;

      mnt_description = 'Выдача займа по договору №'||:k_kontract_no||', '||fio||', без НДС';

      suspend;
    end else
    begin
      -- Выплаты по партнерской программе делаем через основной договор
      mnt_id = f_get_const('moneta_acc');

      select epl_object_id, c_name_fio
      from epay_trans, epay_list, clients
      where ept_from_esm_id = :esm_id and
            ept_from_epl_id = epl_id and
            ept_from_c_id = c_id
      into ag_id, fio;

      if (ag_id is not null) then
        mnt_description = 'Выплата по парнерской программе, договор №'||:ag_id||', '||fio||', без НДС';

      suspend;
    end
  end
end
