CREATE PROCEDURE HP_REFRESH_EXTINFO_DATA
AS
declare variable EIL_ID D_ID;
declare variable CONFIG D_STR_CODE;
declare variable BLS_ID D_ID;
declare variable EIL_STATUS D_SMALLINT;
begin
  for select eil.eil_id,
             eic.eic_code,
             eil.eil_from_bls_id,
             eil.eil_status
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eic.eic_code in ('salary_check', 'pension_check', 'pkb_cib_report', 'gbdfl_get_cli_data', 'checkme_info') and
        eil.eil_status not in (0, -1) and                                   -- Берем обработанные запросы
        datediff(minute, eil.eil_status_datetime, current_timestamp) <= 5 and  -- за последние 5 минут
        coalesce(eil.eil_kk_result, 0) = 0
  into eil_id,
       config,
       bls_id,
       eil_status
   do
   begin
     if (:config in ('salary_check', 'pension_check')) then
     begin
       if (exists(select *
                  from income_check_querys icq
                  where icq.icq_status = 0
                  and icq.icq_from_eil_id = :eil_id)) then
         execute procedure hp_extinfolist_income_update(:eil_id); -- Обновление состояния справки о ЗП, пенсия
     end

     if (:config = 'pkb_cib_report' and
         eil_status = 2) then
     begin
       if (not exists(select *
                      from kib_job_info kji
                      where kji.kji_from_eil_id = :eil_id)) then
         execute procedure hp_refresh_pkb_data(:eil_id);
     end

     update extinfolist eil
     set eil.eil_kk_result = 1
     where eil.eil_id = :eil_id;
   end
end
