CREATE PROCEDURE CHECK_MATCH_CARD_NAME(
    K_ID ID_TYPE)
RETURNS (
    KF_CLIENT_FIO VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    KF_ERROR_TEXT VARCHAR(255),
    KF_ERROR_IS_CRITICAL INTEGER)
AS
declare variable esa_message blob sub_type 1 segment size 80;
declare variable c_cardholder d_str_100;
declare variable mbulak_txt1 d_str_100;
declare variable mbulak_txt2 d_str_100;
declare variable mbulak_txt3 d_str_100;
declare variable other_bank_txt1 d_str_100;
declare variable other_bank_txt2 d_str_100;
declare variable other_bank_txt3 d_str_100;
declare variable c_id d_id;
declare variable txt1_match d_smallint;
declare variable txt2_match d_smallint;
declare variable other_bank_txt1_num d_smallint;
declare variable other_bank_txt2_num d_smallint;
declare variable other_bank_txt3_num d_smallint;
declare variable k_is_online d_smallint;
declare variable kf_client_fio_ciril varchar(100);
declare variable card_no_cl varchar(50);
begin
  select k.k_disb_date, k_amount_approved, k_is_online
  from kontract k
  where k_id = :k_id
  into :kf_disb_date, :kc_amount, :k_is_online;

  if (coalesce(k_is_online, 0) = 0) then
    exit;

  -- по транзакции определяю пользователя
  if (exists(select *
             from username un
             inner join hr_pstatus hps on hps.ps_from_c_id = un.un_from_c_id and
                   :kf_disb_date between ps_date_begin and ps_date_end and
                   ps_is_working = 1
             inner join joblist jl on jl.jl_id = ps_from_jl_id
             inner join jobkind jk on jk.jk_id = jl.jl_kind and
                   jk.jk_system_code = 'jobsystem.region_director'
             where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN'))) then
    kf_error_is_critical = 0;
  else
    kf_error_is_critical = 1;

  for select ea.esa_message, esm_from_k_id, em.esm_object_id
      from extsysmain em
      inner join extsysanswer ea on em.esm_id = ea.esa_from_esm_id
      where em.esm_action = 'cardcheck' and
            em.esm_project = 'elcard_disb' and
            em.esm_status = 1 and
            em.esm_from_k_id = :k_id
      into esa_message, k_id, c_id
  do
  begin
    txt1_match = 0;
    txt2_match = 0;

    c_cardholder = '';
    select json_value
    from fm_json_parse(:esa_message)
    where json_key_full = 'message'
    into :c_cardholder;
    c_cardholder = coalesce(c_cardholder, '');
    kf_client_pass_no = c_cardholder;

    other_bank_txt1 = '';
    other_bank_txt2 = '';
    other_bank_txt3 = '';
    select txt1, txt2, txt3
    from hp_fio_clear(:c_cardholder)
    into other_bank_txt1, other_bank_txt2, other_bank_txt3;
    other_bank_txt1_num = char_length(other_bank_txt1);
    other_bank_txt2_num = char_length(other_bank_txt2);
    other_bank_txt3_num = char_length(other_bank_txt3);

    kf_client_fio = '';
    kf_client_fio_ciril = '';
    mbulak_txt1 = '';
    mbulak_txt2 = '';
    mbulak_txt3 = '';
    select fl.fl_name1, fl.fl_name2, fl.fl_name3, c.c_name_eng, c_name_fio
    from knclient kc
    inner join clients c on kc.kc_from_c_id = c_id
    inner join fiolist fl on c.c_name_eng_fl_id = fl.fl_id
    where kc_from_k_id = :k_id and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0
    into mbulak_txt1, mbulak_txt2, mbulak_txt3, kf_client_fio, kf_client_fio_ciril;

    if (coalesce(kf_client_fio, '') = '') then
    begin
      kf_error_text = 'Для ' || kf_client_fio_ciril || ' ФИО на английском в базе данных не определен';
      suspend;
    end

    if (c_cardholder = '') then
    begin
      kf_error_text = 'Для ' || kf_client_fio_ciril || ' отсутствует ФИО на карте. Запросите договор клиента с банком';
      suspend;
    end

    if (exists(select *
               from string_char_to_decimal(:c_cardholder)
               where string_txt_out > 0)) then
    begin
      kf_error_text = 'У ' || kf_client_fio_ciril || ' имя на карте содержит цифры. Запросите договор клиента с банком';
      suspend;
    end

    if (coalesce(kf_error_text, '') <> '') then
      exit;

    kf_error_text = 'ФИО не совпадает с данными на карте Элкарт. Перевод возможен на карту заемщика ' || kf_client_fio_ciril;

    if (other_bank_txt1_num > 0) then
    begin
      if (left(mbulak_txt1, char_length(other_bank_txt1)) = other_bank_txt1) then
        txt1_match = 1;
      else
        txt1_match = (select *
                      from check_match_card_name_tmp(:mbulak_txt1, :other_bank_txt1));

      if (left(mbulak_txt2, char_length(other_bank_txt1)) = other_bank_txt1) then
        txt2_match = 1;
      else
        txt2_match = (select *
                      from check_match_card_name_tmp(:mbulak_txt2, :other_bank_txt1));
    end

    if (other_bank_txt2_num > 0) then
    begin
      if (left(mbulak_txt1, char_length(other_bank_txt2)) = other_bank_txt2) then
        txt1_match = 1;
      else
      if (txt1_match <> 1) then
        txt1_match = (select *
                      from check_match_card_name_tmp(:mbulak_txt1, :other_bank_txt2));

      if (left(mbulak_txt2, char_length(other_bank_txt2)) = other_bank_txt2) then
        txt2_match = 1;
      else
      if (txt2_match <> 1) then
        txt2_match = (select *
                      from check_match_card_name_tmp(:mbulak_txt2, :other_bank_txt2));
    end

    if (other_bank_txt3_num > 0) then
    begin
      if (left(mbulak_txt1, char_length(other_bank_txt3)) = other_bank_txt3) then
        txt1_match = 1;
      else
      if (txt1_match <> 1) then
        txt1_match = (select *
                      from check_match_card_name_tmp(:mbulak_txt1, :other_bank_txt3));

      if (left(mbulak_txt2, char_length(other_bank_txt3)) = other_bank_txt3) then
        txt2_match = 1;
      else
      if (txt2_match <> 1) then
        txt2_match = (select *
                      from check_match_card_name_tmp(:mbulak_txt2, :other_bank_txt3));
    end

    if (other_bank_txt3_num > 0) then
    begin
      if (left(mbulak_txt1, char_length(other_bank_txt3)) = other_bank_txt3) then
        txt1_match = 1;
      else
      if (txt1_match <> 1) then
        txt1_match = (select *
                      from check_match_card_name_tmp(:mbulak_txt1, :other_bank_txt3));

      if (left(mbulak_txt2, char_length(other_bank_txt3)) = other_bank_txt3) then
        txt2_match = 1;
      else
      if (txt2_match <> 1) then
        txt2_match = (select *
                      from check_match_card_name_tmp(:mbulak_txt2, :other_bank_txt3));
    end

    if (0 in (txt1_match, txt2_match)) then
    begin
      suspend;
    end

    select token
    from hp_client_bank_card(:c_id)
    into :card_no_cl;

    if (exists(select *
               from card_check(:kf_disb_date, :card_no_cl))) then
    begin
      kf_error_text = 'Номер карты клиента совпадает с номером карты сотрудника';
      suspend;
    end
  end
end
