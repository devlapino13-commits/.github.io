CREATE PROCEDURE HP_BKI_FIX_ERROR(
    BKE_ID D_ID NOT NULL,
    UN_ID D_ID,
    DONT_SEND D_BOOL = 0)
AS
declare variable source_alias d_str_100;
declare variable bkl_object_id d_id;
declare variable bkl_object_kind d_integer;
declare variable event_date d_date;
declare variable bkt_format d_integer;
begin
  if (exists (select * from bki_errors bke
              where bke.bke_id = :bke_id
                and bke.bke_fixed = 1))
  then
    exit;

  if (not exists(select * from username where un_id = :un_id)) then
    exception e_default 'not found user for un_id = ' || coalesce(un_id, '<null>');

  update bki_errors bke
  set bke.bke_fixed = 1,
      bke.bke_fixed_un_id = :un_id
  where bke.bke_id = :bke_id;

  if (coalesce(dont_send, 0) = 1) then
    exit;

  bkt_format = (select bkt_format
                from bki_errors, bki_events, bki_clients, bki_loans, bki_packet, bki_packet_type
                where bke_id = :bke_id
                  and bev_id = bke_from_bev_id
                  and bkc_id = bev_from_bkc_id
                  and bkl_id = bkc_from_bkl_id
                  and bkp_id = bkl_from_bkp_id
                  and bkt_id = bkp_from_bkt_id);

  source_alias = '.bki_errors.format' || bkt_format || '.';
  select bkl_object_id, bkl_object_kind, bev_event_date
  from bki_errors, bki_events, bki_clients, bki_loans, bki_packet
  where bke_id = :bke_id
    and bkc_id = bev_from_bkc_id
    and bev_id = bke_from_bev_id
    and bkl_id = bkc_from_bkl_id
    and bkp_id = bkl_from_bkp_id
  into bkl_object_id, bkl_object_kind, event_date;

  execute procedure hp_bki_add_to_buffer(source_alias, f_objkind(bkl_object_kind, 'code'), bkl_object_id, event_date);
end
