CREATE PROCEDURE PRINT_PURCHASE_ACT(
    KMA_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    KML_SELLER_FIO D_STR_100,
    KML_SELLER_INN D_STR_20,
    KML_SELLER_PASS_NO VARCHAR(50),
    KML_SELLER_PASS_DATE DATE,
    KML_SELLER_PASS_DATE_EXPIRE DATE,
    KML_SELLER_PASS_WHO VARCHAR(255),
    KML_SELLER_ADDRESS VARCHAR(255),
    KML_BANK_NAME VARCHAR(255),
    KML_COUNT DECIMAL(18,4),
    KML_COST DECIMAL(18,4),
    KML_TOTAL DECIMAL(18,4),
    KML_STATUS INTEGER,
    KML_NAME VARCHAR(255),
    KML_MODEL VARCHAR(255),
    KML_GUARANTEE_MONTH DECIMAL(18,4),
    KML_PURCHASE_DATETIME TIMESTAMP,
    QR_CODE D_STR_LONG,
    KFI_REG_NAME D_STR_255,
    K_KONTRACT_NO D_STR_255,
    K_KONTRACT_DATE DATE,
    PROJECT_DESCRIPTION D_STR_2048,
    K_SUM_DISB_PREPAY CURR_TYPE,
    OBJECT_QTY INTEGER,
    TOTAL_SUMM CURR_TYPE,
    C_NAME_FIO D_STR_100,
    CZ_DATE DATE,
    KC_INN VARCHAR(20),
    CA_ADDRESS D_STR_255,
    KML_SIGN_FIO D_STR_255,
    C_TYPE SMALLINT,
    SIGNER_ID D_ID,
    KML_IS_USED D_SMALLINT,
    KMI_IMEI D_STR_20,
    KML_BANK_NO VARCHAR(50),
    KML_BANK_BIK ID_TYPE,
    CL_QR_CODE D_STR_LONG,
    KML_SELL_TOTAL CURR_TYPE,
    TOTAL_PROFIT CURR_TYPE,
    CONTRACT_SIGN_DATE D_DATETIME)
AS
declare variable EDL_ID D_ID;
declare variable C_ID D_ID;
declare variable CLI_ID D_ID;
declare variable K_ID D_ID;
declare variable AG_ID D_ID;
begin


  qr_code = '';
  for select distinct ag.ag_object_id, kma.kma_from_k_id, c.c_type, ag.ag_id
      from kontmatag kma
      inner join agreements ag on kma.kma_from_ag_id = ag.ag_id and
            ag.ag_object_kind = f_objkind('clients', 'id')
      inner join clients c on c.c_id = ag.ag_object_id
      where kma.kma_id = :kma_id
      into c_id, k_id, c_type, ag_id
  do
  begin
    if (c_type = 1) then
    begin
      select cb.cb_ben_name, cb.cb_ben_from_c_id
      from clientbenefic cb
      where cb.cb_from_c_id = :c_id and
            cb.cb_is_signer = 1 rows 1
      into kml_sign_fio, signer_id;
    end
    for select prs.kml_seller_c_id, prs.kml_seller_fio, prs.kml_seller_inn, prs.kml_seller_pass_no,
               prs.kml_seller_pass_date, prs.kml_seller_pass_date_expire, prs.kml_seller_pass_who,
               prs.kml_seller_address, prs.kml_bank_name, pk.kml_count, pk.kml_cost, pk.kml_total,
               kml_sell_total, pk.kml_status,
               pk.kml_name, pk.kml_model, pk.kml_guarantee_month, pk.kml_purchase_datetime, kml_is_used,
               cast(pk.kmi_imei as d_str_20)
        from print_kml_seller(:k_id) prs, print_kontmat(:k_id, :c_id) pk
        where prs.kml_seller_c_id = :c_id
        into c_id, kml_seller_fio, kml_seller_inn, kml_seller_pass_no, kml_seller_pass_date,
             kml_seller_pass_date_expire, kml_seller_pass_who, kml_seller_address, kml_bank_name, kml_count, kml_cost,
             kml_total, kml_sell_total, kml_status, kml_name, kml_model, kml_guarantee_month, kml_purchase_datetime,
             kml_is_used, kmi_imei
    do
    begin
      select kk.kfi_reg_name, kk.k_kontract_no, kk.k_kontract_date
      from print_kontract(:k_id) kk
      into kfi_reg_name, k_kontract_no, k_kontract_date;

      select pki.project_description, pki.k_sum_disb_prepay, pki.object_qty, pki.total_summ, pki.total_profit
      from print_kontract_ipf(:k_id) pki
      into project_description, k_sum_disb_prepay, object_qty, total_summ, total_profit;

      select pc.c_name_fio, pc.kc_inn, pc.ca_address
      from print_client(:k_id) pc
      into c_name_fio, kc_inn, ca_address;

      select pcz.cz_date
      from print_client_zadol(:k_id) pcz
      into cz_date;

      select bc.bc_account_no, bc.bc_from_b_id
      from bankclient bc
      where bc.bc_from_c_id = :c_id and
            bc.bc_is_active = 1 rows 1
      into kml_bank_no, kml_bank_bik;

      select kc.kc_from_c_id
      from knclient kc
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      into cli_id;

      if (coalesce(add_params, '') <> '') then
      begin
        edl_id = f_json_str_get_value(add_params, 'edl_id', null);
        qr_code = f_edoc_qr_by_edl(edl_id, c_id);
        cl_qr_code = f_edoc_qr_by_edl(edl_id, cli_id);
      end

      select max(eds.eds_status_datetime)
      from edoc_sign eds
      inner join edoc_list edl on edl.edl_id = eds.eds_from_edl_id
            and eds.eds_status = 1
            and edl.edl_status = 1
            and edl.edl_blank_status = 1
      inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
      where edl.edl_object_kind = f_objkind('agreements', 'id')
          and edl.edl_object_id = :ag_id
          and edc.edc_code = 'auto.agreement'
          and eds.eds_from_c_id = :c_id
      into :contract_sign_date;

      suspend;
    end
  end
end
