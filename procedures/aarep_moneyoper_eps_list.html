CREATE PROCEDURE AAREP_MONEYOPER_EPS_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_ONLY_EMPTY D_BOOL = null,
    REP_OL_ID D_ID = null)
RETURNS (
    LINE_ID D_INTEGER,
    KOB_OL_DATE D_DATE,
    RES_DOC_NAME D_STR_CODE,
    RES_BANK_ID D_STR_CODE,
    RES_IS_BANK D_BOOL,
    RES_OPER_SIGN D_SMALLINT,
    RES_OPER_SIGN_STR D_STR_CODE,
    RES_OPER_SIGN_SYM D_STR_20,
    RES_DDS_CODE D_STR_20,
    KOB_BAL_CODE_DT D_STR_CODE,
    KOB_BAL_CODE_KT D_STR_CODE,
    KOB_ADD_CODE_DT D_STR_CODE,
    KOB_ADD_CODE_KT D_STR_CODE,
    KOB_ACC_CODE_DT D_STR_CODE,
    KOB_ACC_CODE_KT D_STR_CODE,
    KOB_SUMMA D_CURRENCY,
    KOB_CURRENCY D_STR_10,
    RES_DDS_IS_RESIDENT D_BOOL,
    RES_CLIENT_RESIDENT D_STR_CODE,
    RES_CLIENT_COUNTRY D_STR_CODE,
    RES_CLIENT_TYPE D_STR_CODE,
    RES_CLIENT_NAME D_STR_255,
    RES_CLIENT_ID D_STR_20,
    RES_CLIENT_DATE D_DATE,
    RES_DDS_IS_LAW D_BOOL,
    KOB_FROM_KC_ID D_ID,
    RES_KONTRACT_NO D_STR_CODE,
    RES_ACC_CODE D_STR_20,
    OL_DOC_NO_STR D_STR_20,
    OL_DOC_NO D_INTEGER,
    OL_MEMO D_STR_LONG,
    REG_CODE D_STR_20,
    KOB_ID D_ID,
    KOB_FROM_OL_ID D_STR_20,
    RES_OL_DATE D_DATE,
    RES_CLIENT_TYPE_NEW D_STR_CODE)
AS
declare variable kob_from_rml_id d_integer;
declare variable sab_full_code_dt d_str_100;
declare variable sab_full_code_kt d_str_100;
declare variable kob_client_dt d_id;
declare variable kob_client_kt d_id;
declare variable citizen_ru_id d_integer;
declare variable refault_code d_str_code;
declare variable cm_name d_str_255;
declare variable citizen_xx d_str_code;
declare variable bank_acc_id d_id;
declare variable kob_acc_dt d_id;
declare variable kob_acc_kt d_id;
declare variable country_code_ru d_str_code;
declare variable country_code_kg d_str_code;
declare variable country_code_uz d_str_code;
declare variable country_code_tj d_str_code;
declare variable kob_acc_memo_dt d_str_255;
declare variable kob_acc_memo_kt d_str_255;
declare variable obd_detail d_str_long;
declare variable acc_client_dt d_id;
declare variable acc_client_kt d_id;
declare variable last_bank_acc_id d_id;
declare variable last_bank_id d_str_20;
declare variable doc_vnj_id d_integer;
declare variable zesta_transfer_ol d_id;
declare variable zesta_com_kc_id d_id;
begin
  line_id = 0;
  citizen_ru_id = null;

  if (rep_only_empty is null) then
    rep_only_empty = 0;

  select rd_memo
  from refdata
  where rd_main_code = 'CountryOKSM' and
        rd_str_id = 'ru' and
        rd_parent_id is not null rows 1
  into :country_code_ru;

  select rd_memo
  from refdata
  where rd_main_code = 'CountryOKSM' and
        rd_str_id = 'kg' and
        rd_parent_id is not null rows 1
  into :country_code_kg;

  select rd_memo
  from refdata
  where rd_main_code = 'CountryOKSM' and
        rd_str_id = 'uz' and
        rd_parent_id is not null rows 1
  into :country_code_uz;

  select rd_memo
  from refdata
  where rd_main_code = 'CountryOKSM' and
        rd_str_id = 'tj' and
        rd_parent_id is not null rows 1
  into :country_code_tj;

  select c.cm_name
  from company c
  into cm_name;

  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into citizen_ru_id;

  select rd_memo
  from refdata
  where rd_main_code = 'CountryOKSM' and
        rd_str_id = 'RU' rows 1
  into :refault_code;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  if (coalesce(rep_ol_id, 0) = 0) then
    rep_ol_id = null;
  else
  begin
    select kob_cashflow_date, coalesce(ol_parent_id, ol_id)
    from kn_operbook, operlist
    where kob_from_ol_id = :rep_ol_id and
          kob_from_ol_id = ol_id rows 1
    into rep_date_begin, rep_ol_id;

    rep_date_end = rep_date_begin;
  end

  for select kob_id, kob_from_ol_id, kob_ol_date, kob_cashflow_date, kob_summa, kob_currency, kob_from_rml_id,
             kob_from_kc_id, kob_client_dt, kob_client_kt, kob_acc_dt, kob_acc_kt, adt.acc_code, akt.acc_code,
             adt.acc_ab_code, akt.acc_ab_code, adt.acc_add_code, akt.acc_add_code, adt.acc_memo, akt.acc_memo,
             adt.acc_object_id, akt.acc_object_id, sdt.sab_full_code, skt.sab_full_code, ol_doc_no_str, ol_doc_no,
             reg_code, ol_memo, obd_detail
      from kn_operbook, operlist, region, ref_account adt, ref_account akt, sys_accbalance sdt, sys_accbalance skt, kn_obdetail
      where kob_ol_is_active = 1 and
             kob_cashflow_date between :rep_date_begin and :rep_date_end and
            --kob_ol_date between :rep_date_begin and :rep_date_end and
            kob_from_ol_id = ol_id and
            ol_from_reg_id = reg_unicode and
            kob_acc_dt = adt.acc_id and
            kob_acc_kt = akt.acc_id and
            adt.acc_from_sab_id = sdt.sab_id and
            akt.acc_from_sab_id = skt.sab_id and
            kob_detail_id = obd_id and
            kob_from_rml_id is not null and
            coalesce(kob_from_rml_id, 0) <> -1 and
            (:rep_only_empty = 0 or
            coalesce(kob_from_rml_id, 0) = 0) and
            coalesce(:rep_ol_id, kob_from_ol_id) = kob_from_ol_id --and
            -- adt.acc_add_code = '60311 (11)'
            --  and kob_from_ol_id=1010100037081980
            -- and adt.acc_code='60323810009110000008  '
            -- and kob_from_kc_id is null
            -- and adt.acc_ab_code like '603%'
            --and
            --kob_id = 175897476

      into kob_id, kob_from_ol_id, res_ol_date, kob_ol_date, kob_summa, kob_currency, kob_from_rml_id, kob_from_kc_id,
           kob_client_dt, kob_client_kt, kob_acc_dt, kob_acc_kt, kob_acc_code_dt, kob_acc_code_kt, kob_bal_code_dt,
           kob_bal_code_kt, kob_add_code_dt, kob_add_code_kt, kob_acc_memo_dt, kob_acc_memo_kt, acc_client_dt,
           acc_client_kt, sab_full_code_dt, sab_full_code_kt, ol_doc_no_str, ol_doc_no, reg_code, ol_memo, obd_detail
  do
  begin
    res_dds_code = 'Пусто';
    res_dds_is_resident = 0;
    res_dds_is_law = 0;
    res_kontract_no = '';
    zesta_transfer_ol = null;
    zesta_com_kc_id = null;

    if (kob_acc_memo_dt is null) then
      kob_acc_memo_dt = '';

    if (kob_acc_memo_kt is null) then
      kob_acc_memo_kt = ''; 

    if (kob_from_rml_id = 0 and kob_acc_code_kt = '47423810009900000010') then
    begin                       ----RFCRU-4939 расчет кода ддс по комиссиям зесты
      select kob.kob_from_kc_id
      from operlist ol_parent
      inner join operlist ol on ol.ol_parent_id = ol_parent.ol_id
      inner join kn_operbook kob on kob.kob_from_ol_id = ol_parent.ol_id
      where ol.ol_id = :kob_from_ol_id
      and coalesce(kob_from_rml_id, 0) <> -1
      and kob_ol_is_active = 1
      rows 1
      into zesta_com_kc_id;

      select res_rml_id
      from hp_get_ref_moneylist_id(:kob_from_ol_id, :res_ol_date, :kob_acc_dt, :kob_acc_kt, :zesta_com_kc_id, coalesce(:kob_client_dt, :kob_client_kt))
      into :kob_from_rml_id;
    end

    if (kob_from_rml_id is not null) then
    begin
      select rmo_code, rml_is_resident, rml_is_law
      from ref_moneylist, ref_moneyoper
      where rml_id = :kob_from_rml_id and
            rml_from_rmo_id = rmo_id
      into res_dds_code, res_dds_is_resident, res_dds_is_law;
    end
    if (kob_acc_code_dt = '60323810009110000008') then
      res_dds_code = '04910';

    if (kob_acc_code_dt = '47423810009900000010') then  --RFCRU-4939   проверка даты перечисления по Зеста, должна быть в запрашиваемом периоде
    begin
      select olp.olp_from_ol_id
      from operperiod olp
      inner join operbook ob on ob.ob_from_ol_id = olp.olp_from_ol_id
      where :res_ol_date >= olp.olp_date_begin
            and :res_ol_date <= olp.olp_date_end
            and ob.ob_ol_date between :rep_date_begin and :rep_date_end
            and ob.ob_account_kt = '010181076-10000012'
            and ob.ob_account_dt = '010181051-01000007'
      into :zesta_transfer_ol;

      if (zesta_transfer_ol is null) then
        continue;
    end

    res_client_id = null;
    if (kob_from_kc_id is not null) then
    begin
      select kc_from_c_id, k_kontract_no
      from knclient, kontract
      where kc_id = :kob_from_kc_id and
            kc_from_k_id = k_id
      into res_client_id, res_kontract_no;
    end

    /*  select
           case
             when cds_id is not null then 1
             else 0
           end
    from aarep_clientdocs_list(:res_client_id)
    where pdt_is_main_doc = 1 and
          -- cds_is_delete = 0 and
          cds_verify_status = 2 and
          cds_doc_date_begin <= :rep_date_end and
          cds_create_datetime <= :rep_date_end and
          pk_name in ('Внутрен. паспорт РФ', 'Вид на жительство')
    order by cds_doc_date_begin desc
    rows 1
    into res_dds_is_resident;  */

    if ((res_client_id is not null) and
        (kob_bal_code_dt not like '603%')) then
      select 1 - res_non_resident
      from hp_get_client_type(:res_client_id, :citizen_ru_id, :doc_vnj_id, :kob_ol_date)
      into :res_dds_is_resident;

    if (kob_bal_code_dt in ('48901', '48909') and
        (kob_from_kc_id is null)) then
      res_dds_is_resident = 0;

    if ((res_client_id is null) and
        (kob_bal_code_dt like '603%')) then
    begin
      if (substring(kob_acc_code_dt from 13 for 1) = 1) then
        res_dds_is_resident = 1;
      if (substring(kob_acc_code_dt from 13 for 1) = 2) then
        res_dds_is_resident = 0;
    end

    -- Определения типа документа и операции
    res_is_bank = 0;
    res_oper_sign = 0;
    res_bank_id = '';
    res_doc_name = 'МемОрд';
    bank_acc_id = null;

    if (sab_full_code_dt in ('Деньги.Безнал.Банк.РасчСчет', 'Деньги.Безнал.Банк.СпецСчет', 'Деньги.Нал.Касса')) then
    begin
      res_oper_sign = 1;
      if (sab_full_code_dt starting 'Деньги.Безнал.Банк.') then
      begin
        res_is_bank = 1;
        res_doc_name = 'п/п'; -- Платежное поручение

        bank_acc_id = kob_acc_dt;

        if (sab_full_code_kt starting 'ПлатСистемы.') then
        begin
          if (exists(select *
                     from operperiod
                     where olp_from_ol_id = :kob_from_ol_id)) then
            bank_acc_id = null;
        end

        if (obd_detail starting 'ком. за кред. займ ') then
        begin
          res_kontract_no = left(replace(obd_detail, 'ком. за кред. займ ', ''), 50);
          if (position(' ', res_kontract_no) > 0) then
            res_kontract_no = left(res_kontract_no, position(' ', res_kontract_no) - 1);
          if (acc_client_dt is not null) then
            res_client_id = acc_client_dt;
          else
            res_client_id = -1;
        end
      end
      else
      if (sab_full_code_dt starting 'Деньги.Нал.Касса') then
      begin
        res_is_bank = 0;
        res_doc_name = 'ПКО'; -- Расходные кассовый ордер
      end
    end
    else
    if (sab_full_code_kt in ('Деньги.Безнал.Банк.РасчСчет', 'Деньги.Безнал.Банк.СпецСчет', 'Деньги.Нал.Касса')) then
    begin
      res_oper_sign = -1;
      if (sab_full_code_kt starting 'Деньги.Безнал.Банк.') then
      begin
        res_is_bank = 1;
        res_doc_name = 'п/п'; -- Платежное поручение

        bank_acc_id = kob_acc_kt;

        if (obd_detail starting 'ком. за кред. займ ') then
        begin
          res_kontract_no = left(replace(obd_detail, 'ком. за кред. займ ', ''), 50);
          if (position(' ', res_kontract_no) > 0) then
            res_kontract_no = left(res_kontract_no, position(' ', res_kontract_no) - 1);
          if (acc_client_kt is not null) then
            res_client_id = acc_client_kt;
          else
            res_client_id = -1;
        end
      end
      else
      if (sab_full_code_kt starting 'Деньги.Нал.Касса') then
      begin
        res_is_bank = 0;
        res_doc_name = 'РКО'; -- Расходные кассовый ордер
      end
    end
    else
    if (sab_full_code_kt starting 'ПлатСистемы.') then
    begin
      res_oper_sign = -1;
      res_is_bank = 0;
      res_doc_name = 'ПС'; -- Платежная система
      bank_acc_id = kob_acc_kt;

    end
    else
    if (sab_full_code_dt starting 'ПлатСистемы.') then
    begin
      res_oper_sign = 1;
      res_is_bank = 0;
      res_doc_name = 'ПС'; -- Платежная система
      bank_acc_id = kob_acc_dt;
    end
    if (res_doc_name = 'ПС') then
    begin
      if (kob_acc_code_dt in ('47423810009900000002', '47423810009900000005', '47423810009900000006', '47423810009900000008', '47423810009900000010')
      or kob_acc_code_kt = '47423810009900000010' ) then
        res_bank_id = '044525593'; -- Альфа-банк
      if (kob_acc_code_dt = '47423810009900000007') then
        res_bank_id = '044525550'; --ЮНИСТРИМ
    end
    res_oper_sign_str = case res_oper_sign
                          when -1 then 'Списание'
                          when 1 then 'Зачисление'
                          when 0 then 'МемОрдер'
                        end;

    res_oper_sign_sym = case res_oper_sign
                          when -1 then 'Р'
                          when 1 then 'П'
                          when 0 then '-'
                        end;

    if (bank_acc_id is not null and
        res_doc_name not in ('ПКО', 'РКО')) then
    begin
      if (last_bank_acc_id = bank_acc_id) then
        res_bank_id = last_bank_id;
      else
      begin
        select acb_bik
        from ref_accbank
        where acb_id = :bank_acc_id
        into res_bank_id;

        last_bank_acc_id = bank_acc_id;
        last_bank_id = res_bank_id;
      end
    end

    res_oper_sign_str = trim(res_oper_sign_str);

    /* res_acc_code = '';
    if (kob_from_kc_id is not null) then
    begin
      if (res_dds_is_resident = 1) then
        res_acc_code = '488';
      else
        res_acc_code = '492';
    end  */

    -- Информация о клиенте
    res_client_date = null;
    res_client_name = '';
    res_client_type = '';
    res_client_resident = 'Резидент';
    res_client_country = refault_code;

    if (res_client_id is null) then
    begin
      if (kob_client_dt is not null) then
        res_client_id = kob_client_dt;
      else
        res_client_id = kob_client_kt;
    end
    else
    if (res_client_id = -1) then
      res_client_id = null;

    if (res_client_id is not null) then
    begin
      select c_name_fio, c_birth_date,
             iif(c_type = 1, iif(c_name_fio like 'ИП %' or c_name_fio like '% ИП', 'ИП', 'ЮЛ'), 'ФЗ')
      from clients
      where c_id = :res_client_id
      into res_client_name, res_client_date, res_client_type;
    end
    else
    begin
      res_client_name = cm_name;
      res_client_country = refault_code;
      res_client_type = 'ЮЛ';
    end

    if ((res_client_id is not null) and
        (res_client_id <> kob_client_dt) and
        (res_client_id <> kob_client_kt) and
        (kob_bal_code_dt not like '603%')) then
      select 1 - res_non_resident
      from hp_get_client_type(:res_client_id, :citizen_ru_id, :doc_vnj_id, :kob_ol_date)
      into :res_dds_is_resident;
if((res_client_name = 'ПАО СБЕРБАНК')or(res_client_name = 'АО "Альфа-Банк"')) then
      res_dds_is_resident = 1;

    res_client_type_new = null;
    if ((kob_add_code_dt like '%(11)') or (kob_add_code_dt like '%(21)%') or (kob_add_code_dt like '%(31)%')) then
      res_client_type_new = 'ЮЛ';
    if ((kob_add_code_dt like '%(12)%') or (kob_add_code_dt like '%(22)%') or (kob_add_code_dt like '%(32)%')) then
      res_client_type_new = 'ИП';
    if ((kob_add_code_dt like '%(13)%') or (kob_add_code_dt like '%(23)%') or (kob_add_code_dt like '%(33)%') or (kob_add_code_dt like '%(03)%')) then
      res_client_type_new = 'ФЗ';

    if (res_client_type_new is not null) then
      res_client_type = res_client_type_new;
    if (kob_bal_code_dt in ('60307', '60308')) then
      res_client_type = 'ФЗ';

    if (res_dds_is_resident = 0) then
    begin
      res_client_resident = 'Нерезидент';
      citizen_xx = '';

      select res_country_code
      from hp_get_client_type(:res_client_id, :citizen_ru_id, null, :res_ol_date)
      into citizen_xx;

      if (citizen_xx <> '') then
      begin
        if (lower(citizen_xx) = 'ru') then
          res_client_country = country_code_ru;
        else
        if (lower(citizen_xx) = 'kg') then
          res_client_country = country_code_kg;
        else
        if (lower(citizen_xx) = 'uz') then
          res_client_country = country_code_uz;
        else
        if (lower(citizen_xx) = 'tj') then
          res_client_country = country_code_tj;
        else
        begin
          select rd_memo
          from refdata
          where rd_main_code = 'CountryOKSM' and
                rd_str_id = :citizen_xx and
                rd_parent_id is not null rows 1
          into :res_client_country;
        end
      end
    end

    if ((res_client_name = 'АО "Альфа-Банк"') and
        (res_client_country = 643) and
        (res_dds_code = '70060')) then
    begin
      res_dds_code = '04580';
      res_client_resident = 'Резидент';
      res_dds_is_resident = 1;
    end
    --Контрагенты по счетам 48901 и 48909 должны быть закодированы как нерезиденты (займы ЮЛ)
    if ((kob_from_kc_id is null) and
        (res_client_type = 'ЮЛ') and
        (res_dds_code in ('48901', '48909'))) then
      res_dds_is_resident = 0;

    res_acc_code = '';
    if (kob_from_kc_id is not null) then
    begin
      if (res_dds_is_resident = 1) then
        res_acc_code = '488';
      else
        res_acc_code = '492';
    end

    line_id = line_id + 1;
    suspend;

    if ((sab_full_code_dt in ('Деньги.Безнал.Банк.РасчСчет', 'Деньги.Безнал.Банк.СпецСчет', 'Деньги.Нал.Касса')) and
        (sab_full_code_kt in ('Деньги.Безнал.Банк.РасчСчет', 'Деньги.Безнал.Банк.СпецСчет', 'Деньги.Нал.Касса'))) then
    begin
      res_is_bank = 0;
      res_oper_sign = -1;
      res_bank_id = '';
      bank_acc_id = null;

      if (sab_full_code_kt starting 'Деньги.Безнал.Банк.') then
      begin
        res_is_bank = 1;
        res_doc_name = 'п/п'; -- Платежное поручение

        bank_acc_id = kob_acc_kt;

        if (obd_detail starting 'ком. за кред. займ ') then
        begin
          res_kontract_no = left(replace(obd_detail, 'ком. за кред. займ ', ''), 50);
          if (position(' ', res_kontract_no) > 0) then
            res_kontract_no = left(res_kontract_no, position(' ', res_kontract_no) - 1);
          if (acc_client_kt is not null) then
            res_client_id = acc_client_kt;
          else
            res_client_id = -1;
        end
      end
      else
      if (sab_full_code_kt starting 'Деньги.Нал.Касса') then
      begin
        res_is_bank = 0;
        res_doc_name = 'РКО'; -- Расходные кассовый ордер
      end

      res_oper_sign_str = case res_oper_sign
                            when -1 then 'Списание'
                            when 1 then 'Зачисление'
                            when 0 then 'МемОрдер'
                          end;

      res_oper_sign_sym = case res_oper_sign
                            when -1 then 'Р'
                            when 1 then 'П'
                            when 0 then '-'
                          end;

      if (bank_acc_id is not null and
          res_doc_name not in ('ПКО', 'РКО')) then
      begin
        if (last_bank_acc_id = bank_acc_id) then
          res_bank_id = last_bank_id;
        else
        begin
          select acb_bik
          from ref_accbank
          where acb_id = :bank_acc_id
          into res_bank_id;

          last_bank_acc_id = bank_acc_id;
          last_bank_id = res_bank_id;
        end
      end

      res_oper_sign_str = trim(res_oper_sign_str);

      /*  res_acc_code = '';
      if (kob_from_kc_id is not null) then
      begin
        if (res_dds_is_resident = 1) then
          res_acc_code = '488';
        else
          res_acc_code = '492';
      end */

      -- Информация о клиенте
      res_client_date = null;
      res_client_name = '';
      res_client_type = '';
      res_client_resident = 'Резидент';
      res_client_country = refault_code;

      if (res_client_id is null) then
      begin
        if (kob_client_dt is not null) then
          res_client_id = kob_client_dt;
        else
          res_client_id = kob_client_kt;
      end
      else
      if (res_client_id = -1) then
        res_client_id = null;

      if (res_client_id is not null) then
      begin
        select c_name_fio, c_birth_date,
               iif(c_type = 1, iif(c_name_fio like 'ИП %' or c_name_fio like '% ИП', 'ИП', 'ЮЛ'), 'ФЗ')
        from clients
        where c_id = :res_client_id
        into res_client_name, res_client_date, res_client_type;
      end
      else
      begin
        res_client_name = cm_name;
        res_client_country = refault_code;
        res_client_type = 'ЮЛ';
      end
      res_client_type_new = null;
      if ((kob_add_code_dt like '%(11)') or (kob_add_code_dt like '%(21)%') or (kob_add_code_dt like '%(31)%')) then
        res_client_type_new = 'ЮЛ';
      if ((kob_add_code_dt like '%(12)%') or (kob_add_code_dt like '%(22)%') or (kob_add_code_dt like '%(32)%')) then
        res_client_type_new = 'ИП';
      if ((kob_add_code_dt like '%(13)%') or (kob_add_code_dt like '%(23)%') or (kob_add_code_dt like '%(33)%') or (kob_add_code_dt like '%(03)%')) then
        res_client_type_new = 'ФЗ';

      if (res_client_type_new is not null) then
        res_client_type = res_client_type_new;

      if (kob_bal_code_dt in ('60307', '60308')) then
        res_client_type = 'ФЗ';

      if (res_dds_is_resident = 0) then
      begin
        res_client_resident = 'Нерезидент';
        citizen_xx = '';

        select res_country_code
        from hp_get_client_type(:res_client_id, :citizen_ru_id, null, :res_ol_date)
        into citizen_xx;

        if (citizen_xx <> '') then
        begin
          if (lower(citizen_xx) = 'ru') then
            res_client_country = country_code_ru;
          else
          if (lower(citizen_xx) = 'kg') then
            res_client_country = country_code_kg;
          else
          if (lower(citizen_xx) = 'uz') then
            res_client_country = country_code_uz;
          else
          if (lower(citizen_xx) = 'tj') then
            res_client_country = country_code_tj;
          else
          begin
            select rd_memo
            from refdata
            where rd_main_code = 'CountryOKSM' and
                  rd_str_id = :citizen_xx and
                  rd_parent_id is not null rows 1
            into :res_client_country;
          end
        end
      end

      if ((kob_add_code_dt like '%(11)%') or (kob_add_code_dt like '%(21)%') or (kob_add_code_dt like '%(31)%')) then
        res_client_type_new = 'ЮЛ';
      if ((kob_add_code_dt like '%(12)%') or (kob_add_code_dt like '%(22)%') or (kob_add_code_dt like '%(32)%')) then
        res_client_type_new = 'ИП';
      if ((kob_add_code_dt like '%(13)%') or (kob_add_code_dt like '%(23)%') or (kob_add_code_dt like '%(33)%')) then
        res_client_type_new = 'ФЗ';

      if ((res_client_resident = 'Нерезидент') and
          (res_client_name = 'АО "Альфа-Банк"') and
          (res_client_country = 643) and
          (res_dds_code = '70060')) then
      begin
        res_dds_code = '04580';
        res_client_resident = 'Резидент';
        res_dds_is_resident = 1;
      end

      --Контрагенты по счетам 48901 и 48909 должны быть закодированы как нерезиденты (займы ЮЛ)
      if ((kob_from_kc_id is null) and
          (res_client_type = 'ЮЛ') and
          (res_dds_code in ('48901', '48909'))) then
        res_dds_is_resident = 0;

      res_acc_code = '';
      if (kob_from_kc_id is not null) then
      begin
        if (res_dds_is_resident = 1) then
          res_acc_code = '488';
        else
          res_acc_code = '492';
      end

      line_id = line_id + 1;
      suspend;
    end
  end
