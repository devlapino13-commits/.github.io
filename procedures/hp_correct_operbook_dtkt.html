CREATE PROCEDURE HP_CORRECT_OPERBOOK_DTKT
RETURNS (
    OL_REG_CODE VARCHAR(10),
    OL_REGION_ID INTEGER,
    OL_ID DECIMAL(18,0),
    OL_DATE DATE,
    OB_CURRENCY VARCHAR(3),
    OB_ACCOUNT_DT VARCHAR(20),
    OB_ACCOUNT_KT VARCHAR(20),
    OB_REGION_DT INTEGER,
    OB_REGION_KT INTEGER,
    OB_SUMMA_VAL DECIMAL(18,4),
    OB_SUMMA DECIMAL(18,4))
AS
begin
  for select op.reg_code, ol_from_reg_id, ol_id, ol_date, ob_currency, ob_account_dt, ob_account_kt, dt.reg_unicode,
             kt.reg_unicode, ob_summa_val, ob_summa
      from operbook, operlist, region op, region dt, region kt, acc_list ac_dt, acc_list ac_kt
      where ob_from_ol_id = ol_id and
            ol_from_reg_id = op.reg_unicode and
            (left(ob_account_dt, 4) <> op.reg_code or
            left(ob_account_kt, 4) <> op.reg_code) and
            ob_account_dt = ac_dt.al_id and
            ac_dt.al_from_reg_id = dt.reg_unicode and
            ob_account_kt = ac_kt.al_id and
            ac_kt.al_from_reg_id = kt.reg_unicode and
            operlist.is_delete = 0 and
            (ob_summa <> 0 or
            ob_summa_val <> 0)
      into :ol_reg_code, :ol_region_id, :ol_id, :ol_date, :ob_currency, :ob_account_dt, :ob_account_kt, :ob_region_dt,
           :ob_region_kt, :ob_summa_val, :ob_summa
  do
  begin
    suspend;
  end
end
