CREATE PROCEDURE HP_POINTCALC_LIST(
    C_ID D_ID,
    REP_DATE D_DATE,
    CALC_TYPE D_SMALLINT = 0)
RETURNS (
    CALC_DATE D_DATE,
    SH_DATE D_DATE,
    K_ID D_ID,
    SH_KIND D_SMALLINT)
AS
declare variable tmp_is_legit d_smallint;
declare variable tmp_is_point d_smallint;
declare variable sh_period d_integer;
declare variable prv_period d_integer;
declare variable legit_sh_date d_date;
declare variable legit_k_id d_id;
declare variable point_sh_date d_date;
declare variable point_k_id d_id;
declare variable is_legit d_smallint;
declare variable is_point d_smallint;
declare variable last_100 d_date;
declare variable tmp_is_delinq d_smallint;
declare variable is_delinq d_smallint;
declare variable tmp_sh_date d_date;
declare variable tmp_k_id d_id;
declare variable days_1 d_integer;
declare variable days_2 d_integer;
declare variable days_3 d_integer;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  if (db_id = 81001) then
  begin
    days_1 = 8;
    days_2 = 31;
    days_3 = 61;
  end else
  begin
    days_1 = 2;
    days_2 = 8;
    days_3 = 61;
  end

  prv_period = null;
  is_legit = 0;
  is_point = 0;
  is_delinq = 0;
  sh_date = null;
  k_id = null;

  for select sh_date, k_id, is_legit, is_point, is_delinq
  from hp_pointcalc_tmp(:c_id, :rep_date, :calc_type)
  union
  select cast('31.12.9999' as date), 0, 0, 0, 0
  from rdb$database
  order by 1
  into :tmp_sh_date, :tmp_k_id, :tmp_is_legit, :tmp_is_point, tmp_is_delinq
  do
  begin
    sh_period = extract(year from tmp_sh_date) * 100 + extract(month from tmp_sh_date);

    if (tmp_is_delinq = 1) then
      is_delinq = 1;

    if (prv_period <> sh_period) then
    begin
      if (is_legit = 1) then
      begin
        calc_date = rep_date;
        sh_kind = 0;
        sh_date = legit_sh_date;
        k_id = legit_k_id;
        suspend;
      end
      if (is_point = 1) then
      begin
        calc_date = rep_date;
        sh_kind = 10;
        sh_date = point_sh_date;
        k_id = point_k_id;
        suspend;
      end

      if (is_point = -1) then
      begin
        sh_date = point_sh_date;
        k_id = point_k_id;

        sh_kind = 20;
        calc_date = sh_date + days_1;
        suspend;
  
        sh_kind = 30;
        calc_date = sh_date + days_2;
        suspend;
  
        sh_kind = 100;
        calc_date = sh_date + days_3;
  
        if (calc_date < coalesce(last_100, calc_date + 1)) then
        begin
          if (last_100 is null) then
            last_100 = calc_date;
          suspend;
        end
      end

      is_legit = 0;
      is_point = 0;
      sh_date = null;
      k_id = null;
    end

    if (tmp_is_legit = 1) then
    begin
      is_legit = 1;
      legit_k_id = tmp_k_id;
      legit_sh_date = tmp_sh_date;
    end

    if (tmp_is_point = -1 or (tmp_is_point = 1 and is_delinq = 0)) then
    begin
      if (is_point = 0 or tmp_is_point < is_point) then
      begin
        is_point = tmp_is_point;
        point_k_id = tmp_k_id;
        point_sh_date = tmp_sh_date;
      end
    end

    prv_period = sh_period;
  end
end
