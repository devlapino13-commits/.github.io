CREATE PROCEDURE HP_CORRECT_JOBTREEDATE
RETURNS (
    REG_ID INTEGER,
    DEP_ID INTEGER,
    JL_ID INTEGER,
    DD_MIN DATE,
    DD_MAX DATE)
AS
declare variable kol_diff integer;
declare variable kol_begin integer;
declare variable kol_end integer;
begin
  for select jt_from_reg_id, jt_from_dep_id, jt_from_jl_id, min(jt_date_begin), max(jt_date_begin)
      from jobtree
      group by jt_from_reg_id, jt_from_dep_id, jt_from_jl_id
      having min(jt_date_begin) <> max(jt_date_begin)
      into :reg_id, dep_id, jl_id, dd_min, dd_max
  do
  begin
    kol_begin = 0;
    select jt_kolich
    from jobtree
    where jt_from_reg_id = :reg_id and
          jt_from_dep_id = :dep_id and
          jt_from_jl_id = :jl_id and
          jt_date_begin = :dd_min
    into :kol_begin;
    if (kol_begin is null) then
      kol_begin = 0;

    kol_end = 0;
    select jt_kolich
    from jobtree
    where jt_from_reg_id = :reg_id and
          jt_from_dep_id = :dep_id and
          jt_from_jl_id = :jl_id and
          jt_date_begin = :dd_max
    into :kol_end;
    if (kol_end is null) then
      kol_end = 0;
    kol_diff = kol_end - kol_begin;

    if (kol_diff < 0) then
      kol_diff = 0;

    delete from jobtree
    where jt_from_reg_id = :reg_id and
          jt_from_dep_id = :dep_id and
          jt_from_jl_id = :jl_id and
          jt_date_begin = :dd_max;

    if (kol_diff <> 0) then
    begin
      update jobtree
      set jt_kolich = jt_kolich + :kol_diff
      where jt_from_reg_id = :reg_id and
            jt_from_dep_id = :dep_id and
            jt_from_jl_id = :jl_id and
            jt_date_begin = :dd_min;
    end
  end
  suspend;
end
