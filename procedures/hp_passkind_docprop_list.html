CREATE PROCEDURE HP_PASSKIND_DOCPROP_LIST(
    ONLY_ACTIVE D_BOOL = null)
RETURNS (
    PK_ID D_INTEGER,
    PK_NAME D_STR_100,
    PK_COUNTRY D_STR_LANG,
    PK_MEMO D_STR_255,
    PK_IS_DELETE D_BOOL,
    PK_IS_DELETE_STR D_STR_CODE,
    LGOTA_COUNT D_INTEGER)
AS
begin
  if (only_active is null) then
    only_active = 0;

  for select distinct pk_id, pk_name, pk_country, pk_is_delete, pdt_code || ', ' || pdt_name
  from passkind, passdoctype, docproppk
  where pk_from_pdt_id = pdt_id and
        pk_id = dpk_from_pk_id and
        dpk_is_delete = 0 and
        pk_id > 0
  order by pk_is_delete, pk_is_other_category, pk_id
  into pk_id, pk_name, pk_country, pk_is_delete, pk_memo
  do
  begin
    if (only_active = 1 and pk_is_delete = 1) then
      Continue;

    lgota_count = 0;
    select count(*)
    from docpklgota
    where dpg_from_pk_id = :pk_id and    
          dpg_is_delete = 0
    into lgota_count;

    if (lgota_count = 0) then
      lgota_count = null;

    pk_is_delete_str = '';
    if (pk_is_delete = 1) then
      pk_is_delete_str = 'Неактивный';
    suspend;
  end
end
