CREATE PROCEDURE HP_CLOSEMONTH_KN_OBSUMMARY(
    KONT_NO D_STR_CODE)
RETURNS (
    KC_ID D_ID,
    ACC_ID D_ID,
    ACC_CODE D_STR_CODE,
    ACC_NAME D_STR_255,
    SAB_FULL_CODE D_STR_255,
    SAB_NAME D_STR_255,
    KOBS_ID D_ID,
    KOBS_DATE D_DATE,
    KOBS_DATE_END D_DATE,
    KOBS_DT_SUM D_CURRENCY,
    KOBS_DT_VAL D_CURRENCY,
    KOBS_DT_TAX D_CURRENCY,
    KOBS_KT_SUM D_CURRENCY,
    KOBS_KT_VAL D_CURRENCY,
    KOBS_KT_TAX D_CURRENCY,
    KOBS_BAL_SUM D_CURRENCY,
    KOBS_BAL_VAL D_CURRENCY,
    KOBS_BAL_TAX D_CURRENCY)
AS
declare variable k_id d_id;
declare variable obj_kind d_integer;
begin
  if (kont_no containing '-') then
  begin
    select k_id
    from kontract
    where k_kontract_no = :kont_no
    into k_id;
  end else k_id = kont_no;

  obj_kind = f_objkind('knclient', 'id');

  for select kc_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into kc_id
  do
  begin
    for select acc_id, f_fmt_acc(acc_id), acc_name, sab_full_code, sab_name
    from ref_account, sys_accbalance
    where acc_object_kind = :obj_kind and
          acc_object_id = :kc_id and
          acc_from_sab_id = sab_id
    into acc_id, acc_code, acc_name, sab_full_code, sab_name
    do
    begin
      for select kobs_id, kobs_date, kobs_date_end, kobs_dt_sum, kobs_dt_val, kobs_dt_tax,
                 kobs_kt_sum, kobs_kt_val, kobs_kt_tax, kobs_bal_sum, kobs_bal_val, kobs_bal_tax
      from kn_obsummary
      where kobs_acc_id = :acc_id
      order by kobs_date
      into :kobs_id, :kobs_date, :kobs_date_end, :kobs_dt_sum, :kobs_dt_val, :kobs_dt_tax,
           :kobs_kt_sum, :kobs_kt_val, :kobs_kt_tax, :kobs_bal_sum, :kobs_bal_val, :kobs_bal_tax
      do
        suspend;
    end
  end

  suspend;
end
