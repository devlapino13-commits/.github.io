CREATE PROCEDURE MC_BMG_SAVE_REQUEST(
    MCL_ID D_ID,
    INN D_STR_20 NOT NULL,
    PHONE_NUMBER D_STR_20 NOT NULL,
    IS_FOUND D_BOOL NOT NULL,
    IS_ERROR D_BOOL NOT NULL)
RETURNS (
    BC_ID D_ID)
AS
declare variable REQUEST_JSON D_STR_100;
declare variable TOTAL_STATUS D_INTEGER;
begin
    request_json = f_format('{"inn":"%1", "phone_number":"%2"}', inn, phone_number);
    
    if (is_error = 1) then
        total_status = -2;
    else
        total_status = iif(is_found = 1, 1, -1);
    
    insert into bmg_clients (bc_from_mcl_id, bc_request, bc_status)
    values (:mcl_id, :request_json, :total_status)
    returning bc_id
    into :bc_id;

    suspend;
end
