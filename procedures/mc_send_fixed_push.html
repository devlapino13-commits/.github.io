CREATE PROCEDURE MC_SEND_FIXED_PUSH(
    MC_DATE_BEGIN D_DATE,
    MC_DATE_END D_DATE)
AS
declare variable subject d_str_255;
declare variable msg_body d_str_long;
declare variable mcl_id d_id;
declare variable days_count d_integer;
begin
  days_count = datediff(day, :mc_date_begin, :mc_date_end);

  subject = 'М БУЛАК НИКОГДА НЕ ПРИНИМАЕТ ОПЛАТУ ЗАЙМА ЧЕРЕЗ СОТРУДНИКОВ';
  msg_body = 'Не передавайте сотрудникам или третьим лицам деньги по займу.' || ascii_char(10) || 'Это могут быть мошенники!' || ascii_char(10) || 'Платите онлайн на сайте www.molbulak.ru или в приложении Компании.';

  -- закрепляю пуш
  insert into fixed_msg (fm_id, fm_subject, fm_body, fm_bg_colore, fm_text_colore, fm_date_begin, fm_date_end,
                         fm_days_active, fm_create_datetime, fm_from_un_id)
  values (0, :subject, :msg_body, 1, 0, :mc_date_begin, :mc_date_end, :days_count, current_timestamp, 1010100004025);

  -- получаю активных пользователей КМП
  for select ml.mcl_id
      from mc_login ml
      where ml.mcl_is_delete = 0
      into :mcl_id
  do
  begin
    -- отправляю пуш
    execute procedure mc_send_push('mc_login', :mcl_id, :subject, :msg_body, 1);
  end
end
