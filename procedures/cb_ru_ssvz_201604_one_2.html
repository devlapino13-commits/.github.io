AS
declare variable aarep_month_plan_os decimal(18,4) = 0;
declare variable aarep_month_plan_perc decimal(18,4) = 0;
declare variable aarep_plan_all decimal(18,4);
declare variable aarep_fact_all decimal(18,4);
declare variable aarep_fact_planned decimal(18,4);
declare variable aarep_fact_other decimal(18,4);
declare variable koef_bonus decimal(18,4) = 0;
declare variable koef_penalty decimal(18,4) = 0;
declare variable aarep1_plan_all decimal(18,4) = 0;
declare variable aarep1_fact_all decimal(18,4) = 0;
declare variable aarep1_fact_planned decimal(18,4) = 0;
declare variable aarep1_fact_other decimal(18,4) = 0;
declare variable c_id decimal(18,0);
declare variable c_email varchar(100);
declare variable reg_id integer;
declare variable reg_id_prior integer;
declare variable lo_id id_type;
declare variable ps_date_begin date;
declare variable sawt_id integer;
declare variable aarep_client_count_tmp integer;
declare variable average_client_count integer;
declare variable rep_date_begin date;
declare variable hrv_date_end_fact date;
begin
  if (rep_date_end >= '01.04.2018') then -- Бонусы кординально поменялись с 01.04.2018
  begin
    exit;
  end

  koef_bonus = 5.0;
  koef_penalty = 25.0;
  reg_id_prior = 0;

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      where cast(:rep_c_id as id_type) = 0
      union
      select ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = cast(:rep_c_id as id_type) and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            ps_status = 1
      into :reg_id
--  into :c_id, :c_email, :reg_id
  do
  begin
    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
    from region r1, region r2
    where r1.reg_unicode = :reg_id and
          r1.parent = r2.reg_unicode
    into reg_name, branch_name;

    aarep_port_balance_all = 0;
    aarep_port_kolich_all = 0;
    aarep_par_balance1 = 0;
    aarep_par_balance7 = 0;

    aarep_date = :rep_date_end;
    aarep_plan_all = 0;
    aarep_fact_all = 0;
    aarep_fact_planned = 0;
    aarep_fact_other = 0;
    aarep_delinq_os = 0;
    aarep_delinq_perc = 0;
    aarep_fact_os = 0;
    aarep_fact_perc = 0;
    aarep_fact_fine = 0;
    aarep_fact_delinq_os = 0;
    aarep_fact_delinq_perc = 0;
    aarep_fact_adv_os = 0;
    aarep_fact_adv_perc = 0;
    aarep_bonus_os = 0;
    aarep_bonus_os_adv = 0;

    select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other),
           sum(aarep_month_plan_os), sum(aarep_month_plan_perc), sum(aarep_delinq_os), sum(aarep_delinq_perc),
           sum(aarep_fact_os), sum(aarep_fact_perc), sum(aarep_fact_fine), sum(aarep_fact_delinq_os),
           sum(aarep_fact_delinq_perc), sum(aarep_fact_adv_os), sum(aarep_fact_adv_perc), sum(aarep_bonus_os),
           sum(aarep_bonus_os_adv)
    from cb_ru_ssvz_201604_one_tmp(:reg_id, :rep_date_end)
    into :aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os,
         aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine,
         aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc, aarep_bonus_os,
         aarep_bonus_os_adv;

    if (aarep_plan_all is null) then
      aarep_plan_all = 0; -- Сумма новых просрочек + выплаченные по графику за период
    if (aarep_fact_all is null) then
      aarep_fact_all = 0; -- Сумма новых просрочек + выплаченные по графику за период
    if (aarep_fact_planned is null) then
      aarep_fact_planned = 0; -- выплаченные по графику за период
    if (aarep_fact_other is null) then
      aarep_fact_other = 0; -- выплаченные по графику за период
    if (aarep_month_plan_os is null) then
      aarep_month_plan_os = 0;
    if (aarep_month_plan_perc is null) then
      aarep_month_plan_perc = 0;
    if (aarep_delinq_os is null) then
      aarep_delinq_os = 0;
    if (aarep_delinq_perc is null) then
      aarep_delinq_perc = 0;
    if (aarep_fact_os is null) then
      aarep_fact_os = 0;
    if (aarep_fact_perc is null) then
      aarep_fact_perc = 0;
    if (aarep_fact_fine is null) then
      aarep_fact_fine = 0;
    if (aarep_fact_delinq_os is null) then
      aarep_fact_delinq_os = 0;
    if (aarep_fact_delinq_perc is null) then
      aarep_fact_delinq_perc = 0;
    if (aarep_fact_adv_os is null) then
      aarep_fact_adv_os = 0;
    if (aarep_fact_adv_perc is null) then
      aarep_fact_adv_perc = 0;
    if (aarep_oklad is null) then
      aarep_oklad = 0;
    if (aarep_bonus_os is null) then
      aarep_bonus_os = 0;
    if (aarep_bonus_os_adv is null) then
      aarep_bonus_os = 0;

--      aarep_lo_koef=100.0; -- Вот эту нужно вычислить
--      aarep_lo_count=0;    -- количество СВЗ с нужным стажем
--      aarep_client_count=0;  -- количество клиентов у этих СВЗ

--    вычисление кол-во СВЗ и их клиентов
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;

      aarep_lo_count = 0;
      aarep_client_count = 0;
      rep_date_begin = rep_date_end - extract(day from rep_date_end);
      for select ps_from_c_id, ps_date_begin
          from hr_pstatus, joblist, jobkind
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_is_working = 1 and
                ps_status = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code = 'JobSystem.LO_Standart' and
                ps_from_reg_id = :reg_id
          into lo_id, ps_date_begin
      do
      begin
--        проверка стажа не менее 6 месяцев
--        проверка не было ли техническое увольнение
        /*
          if (ps_date_begin > dateadd(month, -6, rep_date_begin)
              and exists(select * from oksummary
              where oks_kc_id = :lo_id
              and :ps_date_begin - 1 between oks_date_begin and oks_date_end
              and oks_balance > 0)
              ) then
          begin
--          проверка с первой даты работы СВЗ
            select min(ps_date_begin)
            from hr_pstatus, joblist, jobkind
            where ps_status = 1
            and ps_from_jl_id = jl_id
            and jl_kind = jk_id
            and jk_system_code = 'JobSystem.LO_Standart'
            and ps_from_c_id = :lo_id
            into ps_date_begin;
          end
          */

        select ps_date_begin
        from hr_pstatus, joblist, jobkind
        where ps_status = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.LO_Standart' and
              ps_from_c_id = :lo_id and
              ps_is_working = 1 and
              ps_date_begin <= ps_date_end and
              ps_date_begin <= :rep_date_end and
              not exists(select *
                         from oksummary
                         where oks_kc_id = ps_from_c_id and
                               ps_date_begin between oks_date_begin and oks_date_end and
                               oks_region_id = ps_from_reg_id and
                               oks_balance > 0 and
                               oks_close_state = 0)
        order by ps_date_begin desc
        rows 1
        into ps_date_begin;

--        проверка из декретного отпуска когда выходила
        hrv_date_end_fact = null;
        select hrv_date_end_fact
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              hrv_from_c_id = :lo_id and
              hrv_status = 1 and
              sawt_id in (16, 17, 18, 21) and
              hrv_date_begin <= :rep_date_end
        order by hrv_date_begin desc
        rows 1
        into hrv_date_end_fact;
        if (hrv_date_end_fact > :rep_date_end) then
          hrv_date_end_fact = null;
        if (hrv_date_end_fact is not null) then
        begin
          if (not exists(select *
                         from oksummary
                         where oks_kc_id = :lo_id and
                               :hrv_date_end_fact - 1 between oks_date_begin and oks_date_end -- портфель проверять на предыдущий день
                               and
                               oks_region_id = :reg_id and
                               oks_balance > 0 and
                               oks_close_state = 0) and
              hrv_date_end_fact > ps_date_begin) then
            ps_date_begin = hrv_date_end_fact;
        end

--        если стаж более 6 месяцев то по нему считаеме кол-во клиентов
        if (ps_date_begin <= dateadd(month, -6, rep_date_begin)) then
        begin
--            расчет кол-во клиентов
          aarep_client_count_tmp = null;
          select sum((select res_just_client_count
                      from hp_get_just_client_in_loan(:rep_date_end, k_id, k_disb_date)))
          from oksummary, kontract
          where oks_kc_id = :lo_id and
                :rep_date_end between oks_date_begin and oks_date_end and
                oks_balance > 0 and
                oks_from_k_id = k_id and
                :rep_date_end - oks_delinq_date <= 360 and
                oks_close_state = 0
          into aarep_client_count_tmp;
          aarep_client_count_tmp = coalesce(aarep_client_count_tmp, 0);

          aarep_client_count = aarep_client_count + aarep_client_count_tmp;

          /*
            sawt_id = null;
            if (aarep_client_count_tmp = 0) then
            begin
--            проверка не находится ли в декретном отпуске
              select sawt_id from HR_VOCTRAVEL, hr_orders, sa_worktype
              where hrv_from_ho_id = ho_id
              and ho_order_type = sawt_id
              and hrv_from_c_id = :lo_id
              and :rep_date_end between hrv_date_begin and hrv_date_end_fact
              order by hrv_date_begin desc
              rows 1
              into sawt_id;
            end
            sawt_id = coalesce(sawt_id,0);

            if (sawt_id not in (16,17,18,8,21)) then
            begin
              aarep_lo_count = aarep_lo_count + 1;
            end
            */
          if (aarep_client_count_tmp > 0 or exists(select *
                                                   from hp_employer_in_dekret(:lo_id, :rep_date_end)
                                                   where is_decret = 0)) then
          begin
            aarep_lo_count = aarep_lo_count + 1;
          end

        end
      end

      if (aarep_lo_count > 0) then
        average_client_count = round(cast(aarep_client_count as double precision) / aarep_lo_count, 0);
      else
        average_client_count = 0;
      if (:rep_date_end < cast('01.08.2016' as date)) then
      begin
        aarep_lo_koef = case
                          when average_client_count < 150 then 30
                          when average_client_count <= 174 then 50
                          when average_client_count <= 199 then 70
                          when average_client_count <= 224 then 85
                          when average_client_count <= 249 then 100
                          else 105
                        end;
      end
      else
      begin
        aarep_lo_koef = case
                          when average_client_count < 175 then 10
                          when average_client_count <= 199 then 30
                          when average_client_count <= 249 then 50
                          when average_client_count <= 299 then 75
                          when average_client_count <= 349 then 100
                          else 110
                        end;
      end
    end

    aarep_returnment_koef = 1.0;
    if (aarep_plan_all <> 0) then
      aarep_returnment_koef = round(aarep_fact_planned / aarep_plan_all, 4);
    else
      aarep_returnment_koef = 0.0;
    if (aarep_returnment_koef > 1.0) then
      aarep_returnment_koef = 1.0;

    aarep_bonus_plus = round((aarep_fact_all / 1000.0) * koef_bonus, 0);
    if (aarep_plan_all > aarep_fact_planned) then
      aarep_bonus_minus = round(((aarep_plan_all - aarep_fact_planned) / 1000.0) * koef_penalty, 0);
    else
      aarep_bonus_minus = 0;

    aarep_bonus_total_fact = 0;
    aarep_bonus_total_to_pay = 0;
    aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus; -- Базовый заработок
    aarep_bonus_total_fact = aarep_bonus_total / 100.0 * aarep_lo_koef; -- Бонус к начислению (ЗДЕСЬ НУЖНО УМНОЖИТЬ НА КОЭФФИЦИЕНТ)
    aarep_bonus_total_to_pay = aarep_bonus_total_fact; -- Бонус к оплате, возможно здесь будет обнулен если уволен, выговор и т.д.
    if (aarep_bonus_total_to_pay < 0) then
      aarep_bonus_total_to_pay = 0;
    aarep_bonus_ideal = round(((aarep_month_plan_os + aarep_month_plan_perc + aarep_fact_other) / 1000.0) * koef_bonus, 0);

    aarep_day_bonus_plus = aarep_bonus_plus;
    aarep_day_bonus_minus = aarep_bonus_minus;
    aarep_day_bonus_total = aarep_bonus_total;
    aarep_day_returnment_koef = aarep_returnment_koef;

    aarep1_plan_all = 0;
    aarep1_fact_all = 0;
    aarep1_fact_planned = 0;
    aarep1_fact_other = 0;

    if ((extract(day from :rep_date_end) > 1) and
        (need_one_day = 1)) then
    begin
      select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other)
      from cb_ru_ssvz_201604_one_tmp(:reg_id, :rep_date_end - 1)
      into :aarep1_plan_all, :aarep1_fact_all, :aarep1_fact_planned, :aarep1_fact_other;

      if (aarep1_plan_all is null) then
        aarep1_plan_all = 0;
      if (aarep1_fact_all is null) then
        aarep1_fact_all = 0;
      if (aarep1_fact_planned is null) then
        aarep1_fact_planned = 0;
      if (aarep1_fact_other is null) then
        aarep1_fact_other = 0;

      aarep_day_returnment_koef = 1.0;
      if (aarep_plan_all - aarep1_plan_all <> 0) then
        aarep_day_returnment_koef = round((aarep_fact_planned - aarep1_fact_planned) / (aarep_plan_all - aarep1_plan_all), 4);
      else
        aarep_day_returnment_koef = 0.0;
      if (aarep_day_returnment_koef > 1.0) then
        aarep_day_returnment_koef = 1.0;
      if (aarep_day_returnment_koef < 0) then
        aarep_day_returnment_koef = 1.0;

      aarep_day_bonus_plus = round((aarep1_fact_all / 1000.0) * koef_bonus, 0);
      if (aarep1_plan_all > aarep1_fact_planned) then
        aarep_day_bonus_minus = round(((aarep1_plan_all - aarep1_fact_planned) / 1000.0) * koef_penalty, 0);
      else
        aarep_day_bonus_minus = 0;
      aarep_day_bonus_total = aarep_day_bonus_plus - aarep_day_bonus_minus;

      aarep_day_bonus_plus = aarep_bonus_plus - aarep_day_bonus_plus;
      aarep_day_bonus_minus = aarep_bonus_minus - aarep_day_bonus_minus;
      aarep_day_bonus_total = aarep_bonus_total - aarep_day_bonus_total;
    end

--    вычисляется кол-во МО в офисе
    kol_ssvz = null;
    select count(*)
    from hr_pstatus, joblist, jobkind
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_status = 1 and
          ps_is_working = 1 and
          ps_from_c_id > 1000000000003 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
          ps_from_reg_id = :reg_id and
          exists(select *
                 from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                 where is_decret = 0) and
          jl_name <> 'И.о. заместителя менеджера офиса'
    into kol_ssvz;
    kol_ssvz = coalesce(kol_ssvz, 0);

    if (kol_ssvz >= 2) then
    begin
      aarep_bonus_plus = round(aarep_bonus_plus / kol_ssvz, 0);
      aarep_bonus_minus = round(aarep_bonus_minus / kol_ssvz, 0);
      aarep_bonus_total = round(aarep_bonus_total / kol_ssvz, 0);
      aarep_bonus_total_fact = round(aarep_bonus_total_fact / kol_ssvz, 0);
      aarep_bonus_total_to_pay = round(aarep_bonus_total_to_pay / kol_ssvz, 0);
    end

    if (kol_ssvz >= 1) then
    begin
      for select ps_from_c_id,
                 (select cont_value
                  from hp_contdata_get_email_valid(0, ps_from_c_id)), jl_name
          from hr_pstatus, joblist, jobkind
          where :rep_c_id = 0 and
                :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
                ps_from_reg_id = :reg_id and
                exists(select *
                       from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                       where is_decret = 0) and
                jl_name <> 'И.о. заместителя менеджера офиса'
          union
          select ps_from_c_id,
                 (select cont_value
                  from hp_contdata_get_email_valid(0, :rep_c_id)), jl_name
          from hr_pstatus, joblist
          where :rep_c_id <> 0 and
                :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id = :rep_c_id and
                ps_from_jl_id = jl_id
          into :c_id, :c_email, jl_name
      do
      begin
        select c_name_fio
        from clients
        where c_id = :c_id
        into :c_name_fio;

        aarep_memo = '';
        if (c_email is null) then
          c_email = '';
        if (c_email like '%@molbulak.com') then
          aarep_memo = 'Ok';
        else
          aarep_memo = 'Не будет отправлено, нет EMAIL';

        suspend;

        aarep_delinq_os = null;
        aarep_delinq_perc = null;
        aarep_fact_os = null;
        aarep_fact_perc = null;
        aarep_fact_fine = null;
        aarep_fact_delinq_os = null;
        aarep_fact_delinq_perc = null;
        aarep_fact_adv_os = null;
        aarep_fact_adv_perc = null;
        aarep_lo_count = null;
        aarep_client_count = null;
        aarep_bonus_os_adv = null;
        aarep_bonus_os = null;
        kol_ssvz = null;
      end
    end
    else
    begin
      c_name_fio = 'Офис без МО';
      c_email = '';
      aarep_memo = 'Не будет отправлено, нет EMAIL';

      if (aarep_plan_all + aarep_fact_all + aarep_fact_planned + aarep_fact_other + aarep_month_plan_os + aarep_month_plan_perc + aarep_delinq_os + aarep_delinq_perc + aarep_fact_os + aarep_fact_perc + aarep_fact_fine + aarep_fact_delinq_os + aarep_fact_delinq_perc + aarep_fact_adv_os + aarep_fact_adv_perc <> 0) then
      begin
        aarep_bonus_total_to_pay = 0;
        suspend;
      end
    end

  end
