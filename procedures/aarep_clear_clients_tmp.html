CREATE PROCEDURE AAREP_CLEAR_CLIENTS_TMP(
    REP_DATE DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    REG_ID INTEGER,
    C_ID DECIMAL(18,0),
    C_AMOUNT DECIMAL(18,4),
    K_AMOUNT DECIMAL(18,4),
    CL_BALANCE DECIMAL(18,4),
    CL_KOLICH INTEGER,
    CL_CIR1_BALANCE DECIMAL(18,4),
    CL_CIR1_KOLICH INTEGER,
    CL_CIR3_BALANCE DECIMAL(18,4),
    CL_CIR3_KOLICH INTEGER,
    CL_CIR4_BALANCE DECIMAL(18,4),
    CL_CIR4_KOLICH INTEGER,
    CL_CIR7_BALANCE DECIMAL(18,4),
    CL_CIR7_KOLICH INTEGER,
    REG_NAME VARCHAR(100),
    PERC DECIMAL(18,4),
    PERC1 DECIMAL(18,4),
    PERC3 DECIMAL(18,4),
    PERC4 DECIMAL(18,4),
    PERC7 DECIMAL(18,4),
    C_AMOUNT1 DECIMAL(18,4),
    C_AMOUNT3 DECIMAL(18,4),
    C_AMOUNT4 DECIMAL(18,4),
    C_AMOUNT7 DECIMAL(18,4))
AS
declare variable koef decimal(18,7);
declare variable kont_balance decimal(18,4);
declare variable c_circle integer;
declare variable k_period decimal(18,4);
declare variable perc_amount decimal(18,4);
begin
  for select kc_from_c_id, k_id, kc_circle, kc_amount_approved, k_amount_approved, kal_new_reg_id,
             reg_code || ' ' || reg_name, k_period
      from kontract, kontstate, knclient, knaktlist, region
      where k_id = ks_from_k_id and
            ks_state = 1 and
            cast(ks_date_begin as date) <= :rep_date and
            cast(ks_date_end as date) >= :rep_date and
            ks_ol_is_delete = 0 and
            kc_from_k_id = k_id and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            knaktlist.is_delete = 0 and
            reg_unicode = kal_new_reg_id
      into :c_id, :k_id, :c_circle, :c_amount, :k_amount, :reg_id, :reg_name, :k_period
  do
  begin
    koef = c_amount / k_amount;
    if (not(exists(select *
                   from knclient, operkred
                   where kc_from_c_id = :c_id and
                         kc_amount_approved is not null and
                         kc_amount_approved > 0 and
                         kc_from_k_id = ok_from_k_id and
                         ok_ol_is_active = 1 and
                         ok_type = 4 and
                         ok_from_c_id = 0 and
                         ok_sh_date < ok_ol_date))) then
    begin
      kont_balance = 0;
      cl_cir1_balance = 0;
      cl_cir1_kolich = 0;
      cl_cir3_balance = 0;
      cl_cir3_kolich = 0;
      cl_cir4_balance = 0;
      cl_cir4_kolich = 0;
      cl_cir7_balance = 0;
      cl_cir7_kolich = 0;
      perc = 0;
      perc1 = 0;
      perc3 = 0;
      perc4 = 0;
      perc7 = 0;
      c_amount1 = 0;
      c_amount3 = 0;
      c_amount4 = 0;
      c_amount7 = 0;

      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :rep_date and
            oks_date_end >= :rep_date and
            oks_close_state = 0
      into :kont_balance;
      if (kont_balance is null) then
        kont_balance = 0;
      cl_balance = kont_balance * koef;
      cl_balance = round(cl_balance, -2);
      cl_kolich = 1;
      select sum(sh_perc)
      from schedule
      where sh_from_k_id = :k_id
      into :perc_amount;
      if (k_period <> 0) then
        perc_amount = perc_amount / k_period;
      perc_amount = perc_amount * koef;
      perc = perc_amount;

      if (c_circle >= 7) then
      begin
        cl_cir7_kolich = 1;
        cl_cir7_balance = cl_balance;
        perc7 = perc_amount;
        c_amount7 = c_amount;
      end
      else
      begin
        if (c_circle >= 4) then
        begin
          cl_cir4_kolich = 1;
          cl_cir4_balance = cl_balance;
          perc4 = perc_amount;
          c_amount4 = c_amount;
        end
        else
        begin
          if (c_circle >= 2) then
          begin
            cl_cir3_kolich = 1;
            cl_cir3_balance = cl_balance;
            perc3 = perc_amount;
            c_amount3 = c_amount;
          end
          else
          begin
            cl_cir1_kolich = 1;
            cl_cir1_balance = cl_balance;
            perc1 = perc_amount;
            c_amount1 = c_amount;
          end
        end
      end
      suspend;
    end
  end
end
