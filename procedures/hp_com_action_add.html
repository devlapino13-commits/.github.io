CREATE PROCEDURE HP_COM_ACTION_ADD(
    CML_ID D_ID,
    CMP_ID D_INTEGER,
    NEW_STATUS D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_SMALLINT)
AS
declare variable new_status_code d_str_code;
declare variable cmc_from_edc_id d_integer;
declare variable cmc_com_kind d_str_code;
declare variable curr_status d_str_code;
declare variable obj_id d_id;
declare variable curr_balance d_currency;
declare variable new_balance d_currency;
declare variable cml_date d_date;
declare variable is_return d_smallint;
declare variable lang d_str_lang;
declare variable tmp_text d_str_255;
declare variable db_id d_integer;
declare variable cmp_is_ipf d_smallint;
declare variable cmc_code d_str_code;
begin
  -- устанавливает новый статус запросу КК
  res_text = '';
  res_is_error = 0;
  lang = f_get_lang_code(un_id, null);
  db_id = f_get_db_id();

  is_return = 0;
  if (new_status starting 'return_') then
  begin
    is_return = 1;
    new_status = substring(new_status from 8);
  end

  select f_enum('com_kind', cmc_com_kind, 'code'), cmc_from_edc_id, cml_object_id, cml_date,
         f_enum('com_action_status', cml_last_status, 'code'),
         cmc_code
  from com_list, com_cfg
  where cml_id = :cml_id and
        cml_from_cmc_id = cmc_id
  into cmc_com_kind, cmc_from_edc_id, obj_id, cml_date, curr_status, cmc_code;

  if (cmc_from_edc_id is null and new_status = 'signing') then
  begin
    res_is_error = 1;
    res_text = f_lang(lang, 'Опция не доступна, Электронный документ не настроен');
  end

  -- RFCRU-6132
  if (res_text = '' and new_status = 'process' and cmc_code = 'rest_standart') then
  begin
    res_text = (select res_text
      from hp_check_psk_limit('com_list', :cml_id, :cml_date));
    if (res_text <> '') then
    begin
      res_is_error = 1;
      suspend;
    end
  end

  if (res_text = '' and
       ((curr_status = new_status) or
        (new_status = 'process' and is_return = 0 and curr_status in ('approved', 'signing', 'done')) or
        (new_status = 'approved' and curr_status in ('signing', 'done')) or
        (new_status = 'signing' and curr_status = 'done'))) then
  begin
    suspend;
    exit;
  end

  if (new_status = 'sent' and is_return = 1 and curr_status in ('signing', 'done')) then
  begin
    res_is_error = 1;
    res_text = f_lang(lang, 'Нельзя исключать, документ уже утвержден');
  end

  -- из окна КК аннулируем электронный документ или оформляется отказ
  if (res_text = '' and cmc_from_edc_id is not null and new_status in ('cancel_edoc', 'rejected')) then
  begin
    if (curr_status = 'done') then
    begin
      res_is_error = 1;
      if (cmc_com_kind = 'reschedule')
      then res_text = f_lang(lang, 'Сначала удалите реструктуризацию');
      else res_text = f_lang(lang, 'Сначала отмените статус "Исполнен"');
    end

    if (res_text = '') then
    begin
      execute procedure hp_edoc_cancel(:cmc_from_edc_id, 'com_list', :cml_id);
      select res_text
      from hp_com_memo_add(:cml_id, 'Электронный документ аннулирован', 1, :un_id)
      into :tmp_text;

      if (db_id = 81001) then
      begin
        -- Также аннулируем расчет ПДН
        execute procedure hp_edoc_cancel('loan.reschedule.pdn', 'com_list', :cml_id);
        select res_text
        from hp_com_memo_add(:cml_id, 'Электронный документ ПДН аннулирован', 1, :un_id)
        into :tmp_text;
      end
    end

    if (new_status = 'cancel_edoc') then
      new_status = 'process';
  end

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  new_status = f_enum('com_action_status', new_status, 'id');
  new_status_code = f_enum('com_action_status', new_status, 'code');

  if (new_status_code in ('new', 'sent')) then
    cmp_id = null;

  if ((res_text = '' and
      cmc_from_edc_id is null and
      new_status_code = 'done')
      or
      (cmc_code in ('kred_kanikul_svo', 'rastor_dog_smert_cli', 'rastor_dog_zakon') and  -- Если кред. каникулы СВО то без подписи клиента выполняем после утверждение
      new_status_code = 'approved')
--      or
--      (res_text = '' and
--       new_status_code = 'signed_edoc_pdn' and
--       cmc_code in ('rastor_dog_sogl_storon'))
      ) then
  begin
    select res_is_error, res_text
    from hp_com_action_execute(:cml_id, :un_id)
    into res_is_error, res_text;

    if (res_is_error = 0) then
      res_text = '';
  end

  if (res_text = '' and
      db_id not in (39801, 41701) and
      new_status_code in ('signing', 'approved') and
      cmc_from_edc_id is not null) then
  begin
    if (cml_date <> current_date) then
    begin
      res_is_error = 1;
      res_text = f_lang(lang, 'Дата запроса устарела');
    end
  end

  if (res_text = '' and
      db_id in (39801, 41701) and
      new_status_code = 'process' and
      cmp_id is not null and
      cmc_from_edc_id is not null) then
  begin
    cmp_is_ipf = 0;
    select coalesce(cmp_is_ipf, 0)
    from com_packet
    where cmp_id = :cmp_id
    into cmp_is_ipf;

    if (exists(select * from kontract, product
    where k_id = :obj_id and
          k_from_pr_id = pr_id and
          pr_code containing '.ipf.')) then
    begin
      if (cmp_is_ipf = 0) then
      begin
        res_is_error = 1;
        res_text = f_lang(lang, 'ИФП финансирования должны рассматриваться в КК ИПФ');
      end
    end else
    begin
      if (cmp_is_ipf = 1) then
      begin
        res_is_error = 1;
        res_text = f_lang(lang, 'Классические займы должны рассматриваться в КК классик');
      end
    end

    if (res_text = '' and cml_date <> current_date and cml_date <> (current_date - 1)) then
    begin
      res_is_error = 1;
      res_text = f_lang(lang, 'Дата запроса устарела');
    end
  end

  if (res_text = '' and
      cmc_from_edc_id is not null and
      new_status_code in ('signing', 'send_edoc_pdn')) then
  begin
    -- Статус send_edoc_pdn для РФ. Отправка документа "Уведомление о рассчитанном ПДН (реструктуризация)"

    if (res_text = '' and cmc_com_kind = 'reschedule') then
    begin
      curr_balance = 0;
      new_balance = 0;
      select sum(ok_os + ok_sh_os)
      from operkred
      where ok_from_k_id = :obj_id and
            ok_ol_is_active = 1 and
            ok_type <= 6
      into curr_balance;
      if (curr_balance is null) then
        curr_balance = 0;

      select sum(cmr_sh_os)
      from com_list_resh
      where cmr_from_cml_id = :cml_id
      into new_balance;
      if (new_balance is null) then
        new_balance = 0;

      if (new_balance <> curr_balance) then
      begin
        res_is_error = 1;
        res_text = f_lang(lang, 'Имеются расхождения в сумме основного долга, отправьте на доработку.');
      end
    end

    if (res_text = '') then
    begin
      if (new_status_code = 'send_edoc_pdn') then
      begin
        if (curr_status <> 'approved') then
        begin
          res_is_error = 1;
          res_text = f_lang(lang, 'Отправка электронного документа возможна только по одобренным запросам КК');
        end
        else
          begin
          select res_text
          from hp_edoc_ins_loan_reschedule_pdn(:cml_id, :un_id)
          into :res_text;
          if (res_text <> '') then
            res_is_error = 1;
        end
      end
      else if (new_status_code = 'signing') then
      begin
        if (curr_status not in ('approved', 'signed_edoc_pdn')) then
        begin
          res_is_error = 1;
          res_text = f_lang(lang, 'Отправка электронного документа возможна только по одобренным запросам КК');
        end
        else
        begin
          if (db_id = 81001 and exists(
              select * from knclient
              where kc_from_c_id = :obj_id
              and kc_amount_approved > 0
              and f_clientcom_allowed(kc_from_c_id, 0, 1, 0) = 0))
          then
            res_text = 'Отправка сообщений данному клиенту запрещена по причине отсутствия согласий';
          else if (db_id = 81001 and f_edoc_state2('loan.reschedule.pdn', cml_date, un_id) = 'active' and
              f_edoc_status('loan.reschedule.pdn', cml_id, cml_date) <> 'signed' and not exists (select *
                                                                                                 from kontract
                                                                                                 inner join product on k_from_pr_id = pr_id
                                                                                                 where k_id = :obj_id
                                                                                                 and (pr_code containing '.selfemployed.'
                                                                                                   or pr_code containing '.business.'))) then
            res_text = 'Сначала нужно чтобы клиент подписал: Уведомление о рассчитанном ПДН (реструктуризация)';

          if (res_text = '') then
            select res_text
            from hp_edoc_insert(:cmc_from_edc_id, 'com_list', :cml_id, :un_id)
            into :res_text;

          if (res_text <> '') then
            res_is_error = 1;
        end
      end
    end
  end

  if (res_text = '') then
  begin
    if (new_status_code in ('rejected', 'rework')) then -- если отправляется на доработку или отказ, график удаляется
    begin
      delete from com_list_resh
      where cmr_from_cml_id = :cml_id;
    end

    insert into com_action (cma_from_cml_id, cma_status, cma_from_cmp_id, cma_from_un_id)
    values (:cml_id, :new_status, :cmp_id, :un_id);
  end

  suspend;
end
