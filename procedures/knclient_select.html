CREATE PROCEDURE KNCLIENT_SELECT(
    K_ID DECIMAL(18,0))
RETURNS (
    KC_FROM_K_ID DECIMAL(18,0),
    KC_NUMBER INTEGER,
    KC_FROM_C_ID DECIMAL(18,0),
    KC_AMOUNT_APPLICATE DECIMAL(18,4),
    KC_FROM_PT_ID VARCHAR(20),
    KC_BUSINESS_TYPE INTEGER,
    KC_IS_NEW_BUSINESS SMALLINT,
    KC_IS_REGISTERED SMALLINT,
    KC_BUSINESS_NAME VARCHAR(255),
    KC_MONTHLY_OBOROT DECIMAL(18,4),
    KC_KAPITAL DECIMAL(18,4),
    KC_MONTH_PROFIT DECIMAL(18,4),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_WORK_KOLICH INTEGER,
    KC_WORK_KOLICH_FAMILY INTEGER,
    KC_IS_BUSINESS_FULL_TIME SMALLINT,
    KC_BUSUNESS_SUBTYPE INTEGER,
    KC_IS_NEW_ADDED SMALLINT,
    KC_STATUS SMALLINT,
    KC_CIRCLE INTEGER)
AS
declare variable dd date;
declare variable kk_id decimal(18,0);
declare variable kf_id decimal(18,0);
begin
  select k_disb_date, k_from_kf_id
  from kontract
  where k_id = :k_id
  into :dd, :kf_id;

  select k_id
  from kontract, product
  where k_disb_date <= :dd and
        k_from_pr_id = pr_id and
        pr_category = 0 and
        k_from_kf_id = :kf_id
  order by k_disb_date desc
  rows 1
  into :kk_id;
  if (kk_id is null) then
    kk_id = :k_id;

  for select kc_from_k_id, kc_number, kc_from_c_id, kc_amount_applicate, kc_from_pt_id, kc_business_type,
             kc_is_new_business, kc_is_registered, kc_business_name, kc_monthly_oborot, kc_kapital, kc_month_profit,
             kc_amount_approved, kc_work_kolich, kc_work_kolich_family, kc_is_business_full_time, kc_busuness_subtype,
             kc_is_new_added, kc_status, kc_circle
      from knclient
      where kc_from_k_id = :kk_id and
            kc_amount_approved <> 0
      into :kc_from_k_id, :kc_number, :kc_from_c_id, :kc_amount_applicate, :kc_from_pt_id, :kc_business_type,
           :kc_is_new_business, :kc_is_registered, :kc_business_name, :kc_monthly_oborot, :kc_kapital, :kc_month_profit,
           :kc_amount_approved, :kc_work_kolich, :kc_work_kolich_family, :kc_is_business_full_time,
           :kc_busuness_subtype, :kc_is_new_added, :kc_status, :kc_circle
  do
  begin
    suspend;
  end
end
