CREATE PROCEDURE AAREP_REBUKES_REGION(
    REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REGION_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    REB_DATE D_DATE,
    SOTR_FIO D_STR_100,
    JOB_NAME D_STR_100,
    REB_TYPE D_STR_100,
    SOTR_SCORE D_INTEGER,
    SOTR_INIT D_STR_100,
    DOC_SODERJ D_STR_LONG,
    DOC_OSNOV D_STR_LONG)
AS
declare variable c_id d_id;
declare variable rb_type_id d_smallint;
begin
  for select rb.rb_type,
             cl.c_name_fio,
             cli.c_name_fio,
             rs.rs_scores,
             rb.rb_reason,
             rb.rb_body,
             cl.c_id,
             rb.rb_date,
             reg.reg_name,
             reg2.reg_name
      from rebuke rb
      inner join clients cl on cl.c_id = rb.rb_from_c_id
      left join clients cli on cli.c_id = rb.rb_initiator_id
      inner join hr_pstatus hrps on hrps.ps_from_c_id = cl.c_id -- для выявления конечного офиса работы сотрудника
      inner join region_list(:region_id, 1, -1, 1) rl on hrps.ps_from_reg_id = rl.id_list
      inner join region reg on reg.reg_unicode = hrps.ps_from_reg_id
      inner join region reg2 on reg2.reg_unicode = reg.parent
      left join rebuke_scores rs on rb.rb_id = rs.rs_from_rb_id and rs.rs_is_delete = 0
      where rb.rb_is_delete = 0 and
            cast('now' as date) between hrps.ps_date_begin and hrps.ps_date_end and -- всегда только последний офис работы сорудника на текущую дату
            rb.rb_date between :rep_date_begin and :rep_date_end
      into rb_type_id,
           sotr_fio,
           sotr_init,
           sotr_score,
           doc_osnov,
           doc_soderj,
           c_id,
           reb_date,
           office_name,
           region_name
  do
  begin
    reb_type = '';
    select el_name
    from enum_list
    where el_kind = 'rebuke_type' and
          el_id = :rb_type_id
    into reb_type; -- RFCKG-3685

    job_name = '';
    select jl.jl_name
    from hr_pstatus hrps
    inner join joblist jl on jl.jl_id = hrps.ps_from_jl_id
    where hrps.ps_from_c_id = :c_id and
          :reb_date between hrps.ps_date_begin and hrps.ps_date_end
    into :job_name;

    suspend;
  end
end
