CREATE PROCEDURE HP_GET_RELATION_SHORT_ID(
    REL_ID VARCHAR(50),
    REL_SEX INTEGER)
RETURNS (
    REL_CODE VARCHAR(50))
AS
declare variable need_exit integer;
declare variable rel_code1 varchar(50);
declare variable rel_id1 varchar(50);
declare variable start_sym integer;
declare variable n integer;
begin
-- Процедура сокращает кодовое обозначение по нижеуказанной логике
-- Суть: Сестру брата назвать сестрой.

-- Алгоритм работы: берется 2 символа из начала родственной связи,
-- и пытаемся сократить, если не сокращается, завершаем работу
-- если сократилось, заменяем первые 2 символа на сокращенные 1, и повторяем цикл

  need_exit = 0;
  rel_code = rel_id;
  start_sym = 1;

-- Кодовые обозначения:
-- 1 - Супруг
-- 2 - Супруга
-- 3 - Отец
-- 4 - Мать
-- 5 - Сын
-- 6 - Дочь
-- 7 - Брат
-- 8 - Сестра

  while (need_exit = 0) do
  begin
    n = 0;
    rel_code1 = rel_code;
    while (n < 4) do
    begin
      n = n + 1;
-- Если связь больше 2 символов, то сокращаем пары типа "родители детей с учетом пола"
      rel_code = replace(rel_code, (n * 2 - 1) || '53', (n * 2 - 1)); -- у мужчин сокращаем отца сына
      rel_code = replace(rel_code, (n * 2 - 1) || '63', (n * 2 - 1)); -- у мужчин сокращаем отца дочери
      rel_code = replace(rel_code, (n * 2) || '54', (n * 2)); -- у женщин сокращаем мать сына
      rel_code = replace(rel_code, (n * 2) || '64', (n * 2)); -- у женщин сокращаем мать дочери

      if (char_length(rel_code) >= 2) then
      begin
        rel_code = replace(rel_code, '335', '3'); -- сын дедушки - отец. (НО МОЖЕТ БЫТЬ ДЯДЯ)
        rel_code = replace(rel_code, '345', '3'); -- сын бабушки - отец. (НО МОЖЕТ БЫТЬ ДЯДЯ)

        rel_code = replace(rel_code, '436', '4'); -- дочь дедушки - мать. (НО МОЖЕТ БЫТЬ ТЁТЯ)
        rel_code = replace(rel_code, '446', '4'); -- дочь бабушки - мать. (НО МОЖЕТ БЫТЬ ТЁТЯ)

        rel_code = replace(rel_code, '12', ''); -- Сокращаем комбинации Муж жены
        rel_code = replace(rel_code, '21', ''); -- Сокращаем комбинации Жена мужа

-- Если связь больше 2 символов, то сокращаем пары типа "родители детей с учетом пола" если они в начале
        if (((rel_sex = 1) and
            (rel_code starting '53' or rel_code starting '63')) or ((rel_sex = 0) and
            (rel_code starting '54' or rel_code starting '64'))) then
        begin
          rel_code = substring(rel_code from 3 for 50);
        end
      end

      if ((rel_code1 <> rel_code) and
          (n = 4)) then -- Если были изменения, цикл начинаем заново
      begin
        n = 0;
        rel_code1 = rel_code;
        start_sym = 1;
      end
    end

-- Если после обработки остались нижеуказанные пары, завершаем работу
    if (((rel_sex = 1) and
        (rel_code = '53' or rel_code = '63')) or ((rel_sex = 0) and
        (rel_code = '54' or rel_code = '64')) or (rel_code = '12' or rel_code = '21')) then
    begin
      need_exit = 1;
      break;
    end

    rel_id1 = substring(rel_code from :start_sym for 2);

-- Правила сокращений:
    rel_code1 = (case rel_id1
                   when '15' then '5' -- Супруг  -> Сын     = Сын
                   when '16' then '6' -- Супруг  -> Дочь    = Дочь

                   when '25' then '5' -- Супруга -> Сын     = Сын
                   when '26' then '6' -- Супруга -> Дочь    = Дочь

                   when '32' then '4' -- Отец    -> Супруга = Мать
                   when '35' then '7' -- Отец    -> Сын     = Брат
                   when '36' then '8' -- Отец    -> Дочь    = Сестра

                   when '41' then '3' -- Мать    -> Супруг  = Отец
                   when '45' then '7' -- Мать    -> Сын     = Брат
                   when '46' then '8' -- Мать    -> Дочь    = Сестра

                   when '53' then '1' -- Сын     -> Отец    = Супруг
                   when '54' then '2' -- Сын     -> Мать    = Супруга
                   when '57' then '5' -- Сын     -> Брат    = Сын
                   when '58' then '6' -- Сын     -> Сестра  = Дочь

                   when '63' then '1' -- Дочь    -> Отец    = Супруг
                   when '64' then '2' -- Дочь    -> Мать    = Супруга
                   when '67' then '5' -- Дочь    -> Брат    = Сын
                   when '68' then '6' -- Дочь    -> Сестра  = Дочь

                   when '73' then '3' -- Брат    -> Отец    = Отец
                   when '74' then '4' -- Брат    -> Мать    = Мать
                   when '77' then '7' -- Брат    -> Брат    = Брат
                   when '78' then '8' -- Брат    -> Сестра  = Сестра

                   when '83' then '3' -- Сестра  -> Отец    = Отец
                   when '84' then '4' -- Сестра  -> Мать    = Мать
                   when '87' then '7' -- Сестра  -> Брат    = Брат
                   when '88' then '8' -- Сестра  -> Сестра  = Сестра

                   else :rel_id1
                 end);

    if (rel_code1 <> rel_id1) then
    begin
      rel_code = left(rel_code, (start_sym - 1)) || rel_code1 || substring(rel_code from (start_sym + 2) for 50);
      if (char_length(rel_code) = 1) then
        need_exit = 1;
      else
        start_sym = 1;
    end
    else
    begin
      if (start_sym >= (char_length(rel_code) - 1)) then
        need_exit = 1;
      else
        start_sym = start_sym + 1;
    end
  end
  suspend;
end
