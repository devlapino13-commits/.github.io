CREATE PROCEDURE FM_SEND_PASSPORT_GRS(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_2048,
    NEED_REFRESH D_SMALLINT)
AS
declare variable c_inn d_str_20;
declare variable pass_who d_str_255;
declare variable pass_who_code d_str_20;
declare variable lang d_str_lang;
declare variable db_id d_integer;
declare variable c_pincode d_str_20;
declare variable c_pass_no d_str_100;
begin
  db_id = f_get_db_id();
  need_refresh = 1;
  lang = f_get_lang_code(:un_id, null);
  select c.c_inn, c.c_pass_who, c.c_pass_who_code, c.c_pincode, c.c_pass_no
  from clients c
  where c.c_id = :c_id
  into c_inn, pass_who, pass_who_code, c_pincode, c_pass_no;

  if (db_id in (39801, 41701)) then
  begin
    if (coalesce(c_inn, '') <> '' and
        char_length(c_inn) = 14 and
        coalesce(pass_who, '') <> '' and
        coalesce(pass_who_code, '') <> '') then
    begin
      execute procedure hp_extinfo_insert('Tunduk', f_objkind('clients', 'id'), :c_id, 1);
      res_text = 'Данные успешно отправлены в ГРС';
    end
    else
    if (coalesce(pass_who, '') = '' or coalesce(pass_who_code, '') = '') then
      res_text = 'Перед отправкой в ГРС нужно заполнить поля: Код подразделения, Орган выдавший документ';
    else
      res_text = 'Для отправки запроса в ГРС надо заполнить ИНН клиента';
    suspend;
  end
  if (db_id = 94401) then
  begin
    if (char_length(c_pincode) = 7 and
        coalesce(c_pass_no, '') <> '') then
    begin
      execute procedure hp_extinfo_insert('FIN', f_objkind('clients', 'id'), :c_id, 1);
      res_text = f_lang(lang, 'Данные успешно отправлены в ГРС');
    end
    else
    if (coalesce(c_pass_no, '') = '') then
      res_text = f_lang(lang, 'Перед отправкой в ГРС нужно заполнить серию и номер паспорта');
    else
      res_text = f_lang(lang, 'Для отправки запроса в ГРС надо заполнить ПИН клиента');
    suspend;
  end
end
