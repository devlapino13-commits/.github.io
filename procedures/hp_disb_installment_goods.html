CREATE PROCEDURE HP_DISB_INSTALLMENT_GOODS(
    MP_ID D_ID,
    EXA_ID D_ID)
AS
declare variable un_id d_id;
declare variable curr_exc_id d_integer;
declare variable wait_status d_smallint;
declare variable exc_id d_integer;
declare variable exc_obj_kind d_integer;
declare variable exc_action_kind d_str_code;
declare variable exc_action_code d_str_code;
declare variable ma_id d_id;
declare variable k_id d_id;
declare variable obj_id d_id;
declare variable edl_id d_id;
declare variable obj_code d_str_code;
declare variable exa_lo_id d_id;
declare variable k_disb_date d_date;
declare variable k_status d_integer;
declare variable res_txt d_str_255;
declare variable amount_kib decimal(18,2);
declare variable c_id d_id;
declare variable agent_c_id d_id;
declare variable exc_id_begin d_integer;
declare variable c_name_fio d_str_100;
declare variable reg_name d_str_100;
declare variable adl_address_str d_str_255;
declare variable c_sex d_smallint;
declare variable list_exc_id d_str_100;
declare variable point d_integer;
declare variable percent perc_type;
declare variable kagent_fio d_str_100;
declare variable amount_total decimal(18,2);
declare variable reg_id d_integer;
declare variable min_amount decimal(18,2);
declare variable max_amount decimal(18,2);
declare variable period_min d_integer;
declare variable period_max d_integer;
declare variable repay_day d_integer;
declare variable subject d_str_255;
declare variable msg_body varchar(500);
declare variable error_txt d_str_255;
begin
  select '.' || list(exc.exc_id, '.') || '.'
  from expresscfg exc
  where exc.exc_action_code in ('data_check',
                                'application_send',
                                'send_kk',
                                'kk.send',
                                'kk.wait',
                                'consult_send',
                                'consulting.agreement',
                                'loan_send',
                                'loan.disb')
  into list_exc_id;

  wait_status = 0;

  while (wait_status = 0) do
  begin
    select exa_from_exc_id, exa_from_ma_id, exa_from_k_id, exa_from_edl_id, exc_obj_kind, exc_action_kind,
           exc_action_code, exa_from_un_id, exa_from_c_id
    from expressapp, expresscfg
    where exa_id = :exa_id and
          exa_from_exc_id = exc_id and
          exa_final_status in (0, 1)
    into curr_exc_id, ma_id, k_id, edl_id, exc_obj_kind, exc_action_kind, exc_action_code, un_id, c_id;

    if (row_count = 0 or -- Если exa_final_status not in (0, 1) завершаем работу
        curr_exc_id = 0) then -- Если экспресс-заявке не дан старт, завершаем работу
      exit;

    wait_status = 0; -- Предположим что все нрмально

    obj_code = f_objkind(exc_obj_kind, 'code');

    obj_id = null;

    if (obj_code = 'expressapp') then
      obj_id = exa_id;

    if (obj_code = 'miniapp') then
      obj_id = ma_id;

    if (obj_code = 'kontract') then
      obj_id = k_id;

    if (obj_code = 'clients') then
      obj_id = c_id;

    if (obj_code = 'mc_purchase') then
      obj_id = mp_id;

    if (obj_id is null and
        exc_action_kind in ('edoc', 'check')) then
      exception e_default 'Unknown obj_id';

    if (obj_code = 'kontract') then
    begin
      select k_disb_date, k_status
      from kontract
      where k_id = :obj_id
      into k_disb_date, k_status;

      if (k_disb_date < current_date) then
      begin
        if (k_status = 1) then
        begin
          update kontract
          set k_status = 0,
              k_user_id = :un_id
          where k_id = :k_id;
        end

        res_txt = '';

        select res_txt
        from fm_change_disb_date(:un_id, :obj_code, :obj_id, current_date)
        into res_txt;

        if (res_txt = '') then
        begin
          update expressapp
          set exa_from_exc_id = 100,
              exa_disb_date = current_date
          where exa_id = :exa_id;

          continue;
        end
      end
    end

    if (exc_action_kind = 'action') then
    begin
      -- Нужно выполнить некие действия, например создать Miniapp и т.д.
      -- Всевозможные действия должны быть автоматизированы внутри нижеуказанной процедуры
      select res_status
      from hp_do_expressapp_action(:exa_id, :exc_action_code)
      into :wait_status;

      -- Добавляем айди заявки в корзине
      if (exc_action_code = 'miniapp.create' and
          wait_status = 0) then
      begin
        select exa.exa_from_ma_id
        from expressapp exa
        where exa.exa_id = :exa_id
        into ma_id;

        update mc_purchase mp
        set mp.mp_from_ma_id = :ma_id
        where mp.mp_id = :mp_id;
      end
    end

    if (exc_action_kind = 'check') then
    begin
      -- Запуск CheckLoanCondition
      -- Если неходим ошибку, отправляем Push специалисту и выходим из цикла (wait_status <> 0)
      select res_is_error
      from hp_check_loan_konditions_update(:exc_obj_kind, :obj_id, :exc_action_code, 0, :un_id)
      into :wait_status;
    end

    if (exc_action_kind = 'edoc') then
    begin
      -- Создаем новый электронный документ
      select res_text, edl_id
      from hp_edoc_insert(:exc_action_code, :obj_code, :obj_id, :un_id)
      into error_txt, edl_id;

      if (error_txt <> '') then
        wait_status = -1;

      if (coalesce(error_txt, '') = '') then
      begin
        update edoc_sign eds
        set eds.eds_status = 1,
            eds.eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'mobile', 'id')
        where eds.eds_from_edl_id = :edl_id;
      end
    end

    if (wait_status = 0) then
    begin
      -- Если дошли до этого момента, значит этап завершился нормально
      -- Устанавливаем следующий этап и оставляем wait_status = 0 чтобы не выйти из цикла
      exc_id = null;
      select exc_id
      from expresscfg
      where exc_id > :curr_exc_id and
            :list_exc_id not containing '.' || exc_id || '.'
      order by exc_id
      rows 1
      into :exc_id;

      if (row_count = 0) then -- Если запрос выше ничего не вернул, значит дошли до конца
      begin
        -- Устанавливаем финальный статус как выданный займ
        update expressapp
        set exa_final_status = 2
        where exa_id = :exa_id and
              exa_final_status <> 2;

        if (exists(select *
                   from expressapp exa
                   where exa.exa_id = :exa_id and
                         exa.exa_from_exc_id = 200)) then
        begin
          subject = 'Ура!';
          msg_body = 'Вы успешно сделали покупку в рассрочку!' || ascii_char(10) || ascii_char(10) || ascii_char(10) || 'Совершайте еще больше покупок вместе с нами.';
          execute procedure mc_send_push('clients', :c_id, :subject, :msg_body);

          select cs.cs_from_c_id
          from mc_purchase mp
          inner join clientshops cs on cs.cs_id = mp.mp_from_cs_id
          where mp.mp_id = :mp_id
          into :agent_c_id;

          subject = 'Продажа прошла успешно';
          msg_body = 'Вы продали товар в рассрочку успешно.' || ascii_char(10) || ascii_char(13) || ascii_char(10) || 'Совершайте еще больше продаж вместе с нами';
          execute procedure mc_send_push('clients', :agent_c_id, :subject, :msg_body);
          exit;
        end
      end
      else
      begin
        -- устанавливаем следующий этап, цикл отработает повторно
        update expressapp
        set exa_from_exc_id = :exc_id
        where exa_id = :exa_id;
      end
    end
    else
    begin -- если wait_status <> 0, фиксируем результат, завершается работа процедуры

      -- Инсертим запись в kontreslist со значением "Перерегистрация клиента"
      insert into kontreslist (krl_id, krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                               krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
      values (-1, 89, :un_id, null, 0, :ma_id, current_timestamp, current_date, current_date, 0);
      execute procedure hp_miniapp_status_change(:ma_id, :un_id, 0);

      k_id = null;
      select kt.k_id
      from kontract kt
      where kt.k_from_ma_app_id = :ma_id
      into k_id;

      update edoc_list edl
      set edl.edl_status = -1
      where edl.edl_object_kind = f_objkind('miniapp', 'id') and
            edl.edl_object_id = :ma_id;

      update edoc_list edl
      set edl.edl_status = -1
      where edl.edl_object_kind = f_objkind('kontract', 'id') and
            edl.edl_object_id = :k_id;

      update edoc_list edl
      set edl.edl_status = -1
      where edl.edl_object_kind = f_objkind('mc_purchase', 'id') and
            edl.edl_object_id = :mp_id;

      if (coalesce(k_id, 0) > 0) then
      begin
        update kontract kt
        set kt.k_status = -1
        where kt.k_id = :k_id;

        --Таблица kontlog автоматически обновиться со статусом -1
      end

      update expressapp exa
      set exa.exa_final_status = -1,
          exa.exa_from_exc_id = :curr_exc_id,
          exa_try_count = exa_try_count + 1,
          exa_wait_status = :wait_status
      where exa.exa_id = :exa_id;

      update mc_purchase mp
      set mp.mp_status = f_enum_id('mc_purchase_status', 'error', null)
      where mp.mp_id = :mp_id;
    end
  end
end
