CREATE PROCEDURE AAREP_DAILY_REPORT_KRED(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_OFFICE VARCHAR(100),
    AAREP_ACTIVE_PORT_GR DECIMAL(18,4),
    AAREP_ACTIVE_PORT_PAR DECIMAL(18,4),
    AAREP_ACTIVE_PORT_IND DECIMAL(18,4),
    AAREP_ACTIVE_PORT_PER DECIMAL(18,4),
    AAREP_ACTIVE_PORT_IPF DECIMAL(18,4),
    AAREP_ACTIVE_PORT_ALL DECIMAL(18,4),
    AAREP_ACTIVE_CLIENT_END INTEGER,
    AAREP_RISK_ALL DECIMAL(18,7),
    AAREP_RISK_7 DECIMAL(18,7),
    AAREP_RISK_30 DECIMAL(18,7))
AS
declare variable aarep_deliq_balance_30 integer;
declare variable aarep_deliq_balance_7 integer;
declare variable aarep_deliq_balance_all decimal(18,4);
declare variable bal_all decimal(18,4);
declare variable bal_7 decimal(18,4);
declare variable bal_30 decimal(18,4);
declare variable ac_all decimal(18,4);
declare variable ac_gr decimal(18,4);
declare variable ac_ind decimal(18,4);
declare variable ac_par decimal(18,4);
declare variable ac_per decimal(18,4);
declare variable ac_ipf decimal(18,4);
declare variable k integer;
declare variable ac_cl integer;
begin
  bal_all = 0;
  bal_7 = 0;
  bal_30 = 0;
  ac_all = 0;
  ac_gr = 0;
  ac_ind = 0;
  ac_per = 0;
  ac_ipf = 0;
  ac_par = 0;
  ac_cl = 0;
  k = 0;
  for select aarep_office, sum(aarep_active_port_gr), sum(aarep_active_port_par), sum(aarep_active_port_ind),
             sum(aarep_active_port_per), sum(aarep_active_port_ipf), sum(aarep_active_port_all),
             sum(aarep_active_client_end), sum(aarep_deliq_balance_all), sum(aarep_deliq_balance_7),
             sum(aarep_deliq_balance_30)
      from aarep_daily_rep_temp(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency)
      group by aarep_office
      order by aarep_office
      into :aarep_office, :aarep_active_port_gr, :aarep_active_port_par, :aarep_active_port_ind, :aarep_active_port_per,
           :aarep_active_port_ipf, :aarep_active_port_all, :aarep_active_client_end, :aarep_deliq_balance_all,
           :aarep_deliq_balance_7, :aarep_deliq_balance_30
  do
  begin
    k = k + 1;
    aarep_risk_all = 0;
    aarep_risk_7 = 0;
    aarep_risk_30 = 0;
    bal_all = bal_all + aarep_deliq_balance_all;
    bal_7 = bal_7 + aarep_deliq_balance_7;
    bal_30 = bal_30 + aarep_deliq_balance_30;
    ac_all = ac_all + aarep_active_port_all;
    ac_gr = ac_gr + aarep_active_port_gr;
    ac_ind = ac_ind + aarep_active_port_ind;
    ac_par = ac_par + aarep_active_port_par;
    ac_per = ac_per + aarep_active_port_per;
    ac_ipf = ac_ipf + aarep_active_port_ipf;
    ac_cl = ac_cl + aarep_active_client_end;

    if (aarep_active_port_all <> 0) then
      aarep_risk_all = cast(aarep_deliq_balance_all / aarep_active_port_all * 100 as decimal(15,2));
    if (aarep_active_port_all <> 0) then
      aarep_risk_7 = cast(aarep_deliq_balance_7 / aarep_active_port_all * 100 as decimal(15,2));
    if (aarep_active_port_all <> 0) then
      aarep_risk_30 = cast(aarep_deliq_balance_30 / aarep_active_port_all * 100 as decimal(15,2));
    suspend;
  end
  if (k > 1) then
  begin
    aarep_office = 'Всего по офисам';
    aarep_active_port_all = ac_all;
    aarep_active_port_gr = ac_gr;
    aarep_active_port_ind = ac_ind;
    aarep_active_port_par = ac_par;
    aarep_active_port_per = ac_per;
    aarep_active_port_ipf = ac_ipf;
    aarep_active_client_end = ac_cl;
    if (aarep_active_port_all <> 0) then
      aarep_risk_all = cast(bal_all / aarep_active_port_all * 100 as decimal(15,2));
    if (aarep_active_port_all <> 0) then
      aarep_risk_7 = cast(bal_7 / aarep_active_port_all * 100 as decimal(15,2));
    if (aarep_active_port_all <> 0) then
      aarep_risk_30 = cast(bal_30 / aarep_active_port_all * 100 as decimal(15,2));
    suspend;
  end

end
