CREATE PROCEDURE HP_DO_KN_OBSUMMARY_FIX_DATE(
    REP_DATE D_DATE)
AS
declare variable kobs_id d_id;
declare variable acc_id d_id;
begin
  for select kob_acc_dt as acc_id,
             sum(kob_summa) as kob_summa,
             sum(kob_summa_val) as kob_summa_val,
             sum(kob_summa_tax_dt) as kob_summa_tax
  from kn_Operbook
  where kob_ol_is_active = 1 and
        kob_ol_date = :rep_date
  group by acc_id
  as cursor book
  do
  begin
    update or insert into kn_obsummary (kobs_acc_id, kobs_date,
                                        kobs_dt_sum, kobs_dt_val, kobs_dt_tax)
    values (book.acc_id, :rep_date, book.kob_summa, book.kob_summa_val, book.kob_summa_tax)
    matching (kobs_acc_id, kobs_date);
  end

  for select kob_acc_kt as acc_id,
             sum(kob_summa) as kob_summa,
             sum(kob_summa_val) as kob_summa_val,
             sum(kob_summa_tax_kt) as kob_summa_tax
  from kn_Operbook
  where kob_ol_is_active = 1 and
        kob_ol_date = :rep_date
  group by acc_id
  as cursor book
  do
  begin
    update or insert into kn_obsummary (kobs_acc_id, kobs_date,
                                        kobs_kt_sum, kobs_kt_val, kobs_kt_tax)
    values (book.acc_id, :rep_date, book.kob_summa, book.kob_summa_val, book.kob_summa_tax)
    matching (kobs_acc_id, kobs_date);
  end

  for select kobs_id, kobs_acc_id
  from kn_obsummary
  where kobs_date = :rep_date and
        (kobs_dt_sum <> 0 or
         kobs_kt_sum <> 0 or
         kobs_dt_val <> 0 or
         kobs_kt_val <> 0 or
         kobs_dt_tax <> 0 or
         kobs_kt_tax <> 0)
  into kobs_id, acc_id
  do
  begin
    if (not exists(select * from kn_operbook
    where kob_acc_dt = :acc_id and
          kob_ol_is_active = 1 and
          kob_ol_date = :rep_date)) then
    begin
      if (not exists(select * from kn_operbook
      where kob_acc_kt = :acc_id and
            kob_ol_is_active = 1 and
            kob_ol_date = :rep_date)) then
      begin
        update kn_obsummary
        set kobs_dt_sum = 0,
            kobs_dt_val = 0,
            kobs_dt_tax = 0,
            kobs_kt_sum = 0,
            kobs_kt_val = 0,
            kobs_kt_tax = 0
        where kobs_id = :kobs_id;
      end
    end
  end
end
