CREATE PROCEDURE HP_KN_OKSUMMARY_RECALC(
    KC_ID D_ID,
    REP_DATE D_DATE NOT NULL)
AS
declare variable k_id d_id;
begin
  -- Обновляет данные в таблице kn_oksummary.
  if (f_get_db_id() <> 81001) then
    exit;

  select kc_from_k_id
  from knclient
  where kc_id = :kc_id
  into k_id;

  if ((select is_kn_mode
       from vbs_is_kn_mode(:k_id)) = 0) then
    Exit;

  update kn_oksummary
  set oks_need_recalc = 1
  where oks_from_kc_id = :kc_id and
        oks_date_begin = :rep_date;

  if (row_count = 0) then
    exception e_default f_format('kn_oksummary record not found, kc_id: %1, date: %2', kc_id, f_datetostr(rep_date));
end
