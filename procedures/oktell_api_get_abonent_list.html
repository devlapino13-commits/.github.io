CREATE PROCEDURE OKTELL_API_GET_ABONENT_LIST
RETURNS (
    PANT_ID D_ID,
    PERSON_ID D_ID,
    PERSON_TYPE D_STR_100,
    PHONE D_STR_20,
    PHONE_TYPE D_STR_100,
    COMMAND_ID D_INTEGER,
    MSK_CALL_TIME_START D_TIME,
    MSK_CALL_TIME_END D_TIME,
    URL D_STR_2048)
AS
declare variable k_id d_id;
begin
  delete from oktell_api_pant_list where 1 = 1;

  for select
         obj_id,
         person_id,
         person_type,
         cell_no,
         phone_type,
         command_id,
         msk_call_time_start,
         msk_call_time_end,
         url,
         k_id
  from okt_oktell_list_2(1)
  into :pant_id,
       :PERSON_ID,
       :PERSON_TYPE,
       :PHONE,
       :phone_type,
       :command_id,
       :msk_call_time_start,
       :msk_call_time_end,
       :url,
       :k_id
  do
  begin
    update or insert into oktell_api_pant_list (oal_from_pant_id, oal_from_k_id)
    values (:pant_id, :k_id)
    matching (oal_from_pant_id);

    suspend;
  end
end
