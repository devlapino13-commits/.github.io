CREATE PROCEDURE HP_PRINTABS_LIST
RETURNS (
    PTA_ID INTEGER,
    PTA_NAME VARCHAR(255),
    PTA_MEMO VARCHAR(255),
    PTA_STORED_SQL VARCHAR(255))
AS
declare variable n integer;
declare variable proc_name varchar(255);
begin
  pta_id = 0;
  for select distinct lower(pta_name), lower(pta_stored_sql)
      from printabs
      into pta_name, pta_stored_sql
  do
  begin
    proc_name = '';
    if (not(pta_stored_sql containing 'select *')) then
    begin
      n = position('(', pta_stored_sql, 1);
      if (n > 0) then
        proc_name = left(pta_stored_sql, n - 1);
      else
        proc_name = pta_stored_sql;

      pta_stored_sql = 'select * from ' || pta_stored_sql;
    end
    else
    begin
      n = position('from ', pta_stored_sql, 1);
      if (n > 0) then
        proc_name = trim(left(pta_stored_sql, n + 5));
    end
    pta_id = pta_id + 1;
    pta_memo = '';

    if (proc_name <> '') then
    begin
      proc_name = upper(proc_name);
      select trim(p.rdb$description)
      from rdb$procedures p
      where p.rdb$procedure_name = :proc_name and
            p.rdb$package_name is null
      into pta_memo;
      if (pta_memo is null) then
        pta_memo = '';
    end
    suspend;
  end
end
