CREATE PROCEDURE HP_HR_PSTATUS_PART_SET(
    PS_ID D_INTEGER,
    HO_ID ID_TYPE,
    OFFICE_ID D_INTEGER,
    EMP_ID D_ID,
    PART_HOURS D_INTEGER,
    PS_IS_NOT_MAIN D_BOOL)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_BOOL)
AS
declare variable ps_from_ho_id id_type;
declare variable ps_part_hours d_integer;
declare variable ps_is_part d_bool;
begin
  -- Обновляет hr_pstatus: ps_is_part = 1, ps_part_hours = заданное кол-во рабочих часов в день у совместителя (RFCAZ-372)

  res_text = '';
  res_is_error = 1;

  if (coalesce(office_id, 0) = 0 or coalesce(emp_id, 0) = 0) then
  begin
    res_text = 'Не указан сотрудник, которому требуется установить признак совместительсва';
    suspend;
    exit;
  end

  if (coalesce(ps_id, 0) = 0 or coalesce(ho_id, 0) = 0) then
  begin
    res_text = 'Не указан действующий приказ о найме (ротации, изменении условий)';
    suspend;
    exit;
  end

  if (coalesce(part_hours, 0) not in (1, 2, 3, 4)) then
  begin
    res_text = 'Некорректное кол-во рабочих часов в день (должно быть в диапазоне 1-4)';
    suspend;
    exit;
  end

  select ps_from_ho_id,
         ps_is_part,
         ps_part_hours
  from hr_pstatus
  inner join hr_orders on ho_id = ps_from_ho_id
  where ho_status = 1 and -- утвержденные
        ho_order_type between 1 and 3 and -- прием на работу, ротация, изменение условий договора
        ps_id = :ps_id and
        ps_from_ho_id = :ho_id and
        ps_from_reg_id = :office_id and
        ps_from_c_id = :emp_id and
        ps_status = 1 and
        ps_is_working = 1 and
        current_date between ps_date_begin and ps_date_end
  into ps_from_ho_id,
       ps_is_part,
       ps_part_hours;

  if (coalesce(ps_from_ho_id, 0) = 0) then
    res_text = 'Не найден действующий приказ о найме (ротации, изменении условий)';

  if (res_text = '' and ps_is_part = 1 or ps_part_hours is not null) then
    res_text = 'Указанный сотрудник уже является совместителем';

  if (res_text = '') then
  begin
    if (ps_is_not_main = 0) then
      ps_is_not_main = null;

    update hr_pstatus
    set ps_is_part = 1,
        ps_part_hours = :part_hours,
        ps_is_not_main = :ps_is_not_main
    where ps_id = :ps_id and
          ps_from_ho_id = :ho_id and
          ps_from_reg_id = :office_id and
          ps_from_c_id = :emp_id and
          ps_is_part = 0 and
          ps_status = 1 and
          ps_is_working = 1 and
          current_date between ps_date_begin and ps_date_end;

    res_text = 'Признак совместительства успешно установлен';
    res_is_error = 0;
  end

  suspend;
end
