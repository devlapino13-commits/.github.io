CREATE PROCEDURE PRINT_BLANK_LIST(
    KEY_ID DECIMAL(18,0),
    KEY_KIND INTEGER,
    REP_DATE DATE = null)
RETURNS (
    RES_BLANK_ID DECIMAL(18,0),
    RES_BLANK_NAME VARCHAR(255),
    RES_BLANK_MEMO VARCHAR(255),
    RES_BLANK_DATE DATE,
    RES_SHOW_MODE INTEGER,
    RES_IS_READ_ONLY INTEGER)
AS
declare variable is_active integer;
declare variable bln_from_blt_id d_integer;
declare variable db_id d_integer;
declare variable bln_lang d_str_lang;
declare variable cli_lang d_str_lang;
declare variable is_cession boolean;
begin
  db_id = f_get_db_id();

  if (key_kind = 0) then
  /* Кредитные договора */
  begin
    if (rep_date is null) then
    begin
      select k_disb_date
      from kontract
      where k_id = :key_id
      into :rep_date;
    end
    is_active = 0;
    if (exists(select *
               from kontstate
               where ks_from_k_id = :key_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0)) then
      is_active = 1;

    cli_lang = (
      select
        c_blank_lang
      from knclient
        inner join clients
          on c_id = kc_from_c_id
      where kc_from_k_id = :key_id
        and kc_amount_approved > 0
        and kc_status = 2);

    is_cession = (db_id = 81001 and
      exists(select * from sb_akt
        where sba_from_k_id = :key_id
          and current_date between sba_date_begin and sba_date_end
          and sba_is_delete = 0
          and sba_from_sbl_id in (224, 225)));

    for select distinct bln_id, bln_lang, f_str_join(bln_name, bln_memo, '/'), bln_memo, blb_date_begin,
                        iif(blb_frx_status = 1 and blb_frx_blob is not null, 0, bln_show_mode), bln_is_read_only,
                        bln_from_blt_id
        from blankname, productblank, kontract, blankblob
        where k_id = :key_id and
              k_from_pr_id = prb_from_pr_id and
              coalesce(prb_date_begin, :rep_date) <= :rep_date and
              coalesce(prb_date_end, :rep_date) >= :rep_date and
              prb_from_bln_id = bln_id and
              blb_from_bln_id = bln_id and
              blb_date_begin <= :rep_date and
              blb_date_end >= :rep_date and
              bln_is_active = 1 and -- Добавил скрытие отключенных бланков
              (blb_blob is not null or blb_frx_blob is not null) and
              (bln_print_status = 'all' or
              (:is_active <> 0 and bln_print_status = 'active') or
              (:is_active = 0 and bln_print_status = 'process'))
        order by prb_order_no, prb_from_bln_id
        into :res_blank_id, :bln_lang, :res_blank_name, :res_blank_memo, :res_blank_date, :res_show_mode, :res_is_read_only,
             bln_from_blt_id
    do
    begin
      -- RFCKG-5081
      if (db_id = 41701 and res_blank_id in (
        1381, 1382, 1554, 1555, 1700, 1699,
        1854, 1855, 1856, 1857, 1858, 1859,
        1860, 1861, 1862, 1863)) then
      begin
        if (bln_lang <> cli_lang) then
          continue;
      end

      -- RFCRU-5993
      if (is_cession and res_blank_id = 1588) then
        continue;

      -- исключаем бланки, которые используются в элеектронных документах и имеют встроенный штамп и подпись руководителя
      if (bln_from_blt_id = 0) then
        suspend;
      else
      if (db_id = 81001) then
      begin
        if (not(exists(select * from edoc_frx
        where edf_from_blt_id = :bln_from_blt_id and
              edf_need_qr = 1))) then
          suspend;
      end
    end

  end
  if (key_kind = 1) then
  /* Кадровые бланки */
  begin
    if (rep_date is null) then
    begin
      select ho_date
      from hr_orders
      where ho_id = :key_id
      into :rep_date;
    end

    for select bln_id, f_str_join(bln_name, bln_memo, '/'), bln_memo, blb_date_begin,
               iif(blb_frx_status = 1 and blb_frx_blob is not null, 0, bln_show_mode), bln_is_read_only,
               bln_from_blt_id
        from blankname, blankblob, hr_orders, sa_blank
        where ho_order_type = sab_from_sawt_id and
              sab_from_bln_id = bln_id and
              ho_id = :key_id and
              blb_from_bln_id = bln_id and
              blb_date_begin <= :rep_date and
              blb_date_end >= :rep_date and
              (blb_blob is not null or
              blb_frx_blob is not null)
        into :res_blank_id, :res_blank_name, :res_blank_memo, :res_blank_date, :res_show_mode, :res_is_read_only,
             bln_from_blt_id
    do
    begin
      -- исключаем бланки, которые используются в элеектронных документах и имеют встроенный штамп и подпись руководителя
      if (bln_from_blt_id = 0) then
        suspend;
      else
      if (db_id = 81001) then
      begin
        if (not(exists(select * from edoc_frx
        where edf_from_blt_id = :bln_from_blt_id and
              edf_need_qr = 1))) then
          suspend;
      end
    end
  end
  if (key_kind = 2) then
  /* Личный листок */
  begin
    if (rep_date is null) then
      rep_date = current_date;

    for select bln_id, f_str_join(bln_name, bln_memo, '/'), bln_memo, blb_date_begin,
               iif(blb_frx_status = 1 and blb_frx_blob is not null, 0, bln_show_mode), bln_is_read_only,
               bln_from_blt_id
        from blankname, blankblob
        where blb_from_bln_id = bln_id and
              (blb_blob is not null or
              blb_frx_blob is not null) and
              bln_kind = 5 and
              blb_date_begin <= :rep_date and
              blb_date_end >= :rep_date
        into :res_blank_id, :res_blank_name, :res_blank_memo, :res_blank_date, :res_show_mode, :res_is_read_only,
             bln_from_blt_id
    do
    begin
      -- исключаем бланки, которые используются в элеектронных документах и имеют встроенный штамп и подпись руководителя
      if (bln_from_blt_id = 0) then
        suspend;
      else
      if (db_id = 81001) then
      begin
        if (not(exists(select * from edoc_frx
        where edf_from_blt_id = :bln_from_blt_id and
              edf_need_qr = 1))) then
          suspend;
      end
    end
  end
  if (key_kind = 3) then
  /* Доверенности */
  begin
    if (rep_date is null) then
    begin
      select dl_date_begin
      from dovlist
      where dl_id = :key_id
      into :rep_date;
    end

    for select bln_id, f_str_join(bln_name, bln_memo, '/'), bln_memo, blb_date_begin,
               iif(blb_frx_status = 1 and blb_frx_blob is not null, 0, bln_show_mode), bln_is_read_only,
               bln_from_blt_id
        from blankname, blankblob, dovforms, dovlist, dovcateg
        where blb_from_bln_id = bln_id and
              blb_date_begin <= :rep_date and
              blb_date_end >= :rep_date and
              (blb_blob is not null or
              blb_frx_blob is not null) and
              df_from_bln_id = bln_id and
              df_from_dc_id = dc_id and
              dl_from_dc_id = dc_id and
              dl_id = :key_id
        into :res_blank_id, :res_blank_name, :res_blank_memo, :res_blank_date, :res_show_mode, :res_is_read_only,
             bln_from_blt_id
    do
    begin
      -- исключаем бланки, которые используются в элеектронных документах и имеют встроенный штамп и подпись руководителя
      if (bln_from_blt_id = 0) then
        suspend;
      else
      if (db_id = 81001) then
      begin
        if (not(exists(select * from edoc_frx
        where edf_from_blt_id = :bln_from_blt_id and
              edf_need_qr = 1))) then
          suspend;
      end
    end
  end
  if (key_kind in (6, 7, 8)) then
  /* Справки, уведомления */
  begin
    if (rep_date is null) then
      rep_date = current_date;

    for select bln_id, f_str_join(bln_name, bln_memo, '/'), bln_memo, blb_date_begin,
               iif(blb_frx_status = 1 and blb_frx_blob is not null, 0, bln_show_mode), bln_is_read_only,
               bln_from_blt_id
        from blankname, blankblob
        where bln_kind = :key_kind and
              blb_from_bln_id = bln_id and
              blb_date_begin <= :rep_date and
              blb_date_end >= :rep_date and
              (blb_blob is not null or
              blb_frx_blob is not null)
        into :res_blank_id, :res_blank_name, :res_blank_memo, :res_blank_date, :res_show_mode, :res_is_read_only,
             bln_from_blt_id
    do
    begin
      -- исключаем бланки, которые используются в элеектронных документах и имеют встроенный штамп и подпись руководителя
      if (bln_from_blt_id = 0) then
        suspend;
      else
      if (db_id = 81001) then
      begin
        if (not(exists(select * from edoc_frx
        where edf_from_blt_id = :bln_from_blt_id and
              edf_need_qr = 1))) then
          suspend;
      end
    end
  end
end
