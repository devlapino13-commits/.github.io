CREATE PROCEDURE HP_PETTRANS_EDIT(
    ID D_INTEGER,
    IS_RENT D_BOOL,
    IS_MOVE D_BOOL = 0,
    ID_OWN D_STR_255 = null)
RETURNS (
    PET_ID D_INTEGER,
    PET_NAME D_STR_255,
    PET_LABEL D_STR_CODE,
    PET_IS_RENT D_BOOL,
    PET_NORMA_KM D_INTEGER,
    PET_NORMA_YEDIZ D_INTEGER,
    PET_BILL_ID D_ID,
    PET_BILL_LINE_NO D_INTEGER,
    PED_ID D_INTEGER,
    PED_DATE D_DATE,
    PED_REGION_PLACE_ID D_INTEGER,
    PED_FROM_C_ID D_ID,
    PED_ACC_BALANCE D_STR_20,
    PED_ACC_NUMBER D_INTEGER,
    PED_EXPENSE_SUMMER D_CURRENCY,
    PED_EXPENSE_SUMMER_MOUNT D_CURRENCY,
    PED_EXPENSE_WINTER D_CURRENCY,
    PED_EXPENSE_WINTER_MOUNT D_CURRENCY,
    PED_IS_ACTIVE D_INTEGER)
AS
begin
  select
    pet_id,
    pet_name,
    pet_label,
    pet_is_rent,
    pet_norma_km,
    pet_norma_yediz,
    pet_bill_id,
    pet_bill_line_no
  from pettrans
  where pet_id = :id
  into
    :pet_id,
    :pet_name,
    :pet_label,
    :pet_is_rent,
    :pet_norma_km,
    :pet_norma_yediz,
    :pet_bill_id,
    :pet_bill_line_no;
  pet_is_rent = coalesce(pet_is_rent, :is_rent);
  pet_norma_km = coalesce(pet_norma_km, iif(pet_is_rent = 0, 10, 100));
  pet_norma_yediz = coalesce(pet_norma_yediz, iif(pet_is_rent = 0, 1, 0));

  if (is_move = 1 and pet_is_rent = 0) then
    exception e_message 'Перемещение собственного объекта выполняется из АПП ТМЦ.';

  if (pet_bill_id > 0) then
  begin
    select first 1
      ml_name || iif(ml_model in (ml_name, ''), '', ' ' || ml_model),
      ob_invent_str
    from operbill
      inner join matlist
        on ml_id = ob_from_ml_id
    where ob_bill_id = :pet_bill_id
      and ob_bill_line_no = :pet_bill_line_no
      and ob_is_delete = 0
    into
      :pet_name,
      :pet_label;
  end

  select first 1
    ped_id,
    ped_date,
    ped_region_place_id,
    ped_from_c_id,
    ped_acc_balance,
    ped_acc_number,
    ped_expense_summer,
    ped_expense_summer_mount,
    ped_expense_winter,
    ped_expense_winter_mount,
    ped_is_active
  from petdata
  where ped_from_pet_id = :id
  order by ped_id desc
  into
    :ped_id,
    :ped_date,
    :ped_region_place_id,
    :ped_from_c_id,
    :ped_acc_balance,
    :ped_acc_number,
    :ped_expense_summer,
    :ped_expense_summer_mount,
    :ped_expense_winter,
    :ped_expense_winter_mount,
    :ped_is_active;

  ped_id = coalesce(ped_id, 1);
  ped_date = coalesce(ped_date, current_date);
  ped_acc_balance = coalesce(ped_acc_balance, iif(pet_is_rent = 0, '80304', '80345'));
  ped_acc_number = coalesce(ped_acc_number, iif(pet_is_rent = 0, 7, 3));

  ped_expense_summer = coalesce(ped_expense_summer, iif(pet_is_rent = 0, 14, 10));
  ped_expense_summer_mount = coalesce(ped_expense_summer_mount, iif(pet_is_rent = 0, 14, 10));
  ped_expense_winter = coalesce(ped_expense_winter, iif(pet_is_rent = 0, 14, 10));
  ped_expense_winter_mount = coalesce(ped_expense_winter_mount, iif(pet_is_rent = 0, 14, 10));
  ped_is_active = coalesce(ped_is_active, 1);

  if (pet_id > 0 and pet_is_rent = 1 and is_move = 1) then
  begin
    -- перемещение арендованного объекта
    ped_id = ped_id + 1;
    ped_date = current_date;
    ped_region_place_id = null;
    ped_from_c_id = null;
    ped_is_active = 1;
  end

  suspend;
end
