CREATE PROCEDURE FM_LOANS_IN_PROCESS_TMP2(
    REG_ID D_INTEGER,
    C_ID D_ID)
RETURNS (
    MA_ID D_INTEGER,
    K_STATUS D_INTEGER,
    KF_ID D_ID,
    KF_NAME D_STR_100,
    DISB_DATE D_DATE,
    MA_PLAN_DISB_TIME D_DATE,
    PR_ID D_INTEGER)
AS
begin
    if (reg_id = -1) then
    begin
        for select ma_id, ma_is_active, ma_from_kf_id, kal_new_kf_name, ma_date_oper, ma_plan_disb_time, ma_from_pr_id
            from miniapp, knaktlist
            where ma_date_end >= '31.12.2999' and
                  ((ma_is_active = 1 and
                  ma_date_app between(current_date - 20) and current_date) -- Действующий заявки
                  or (ma_is_active <= 0 and
                  ma_date_app between(current_date - 1) and current_date) -- Аннулированные
                  ) and
                  kal_new_kf_officer = :c_id and
                  ma_from_kf_id = kal_from_kf_id and
                  knaktlist.is_delete = 0 and
                  kal_date <= current_date and
                  kal_date_end >= '31.12.2999'
            into :ma_id, k_status, kf_id, kf_name, :disb_date, ma_plan_disb_time, :pr_id
        do
            suspend;
    end
    else
    begin
        for select ma.ma_id, ma.ma_is_active, ma.ma_from_kf_id, kl.kal_new_kf_name, ma.ma_date_oper,
                   ma.ma_plan_disb_time, ma.ma_from_pr_id
            from (select ma_id, ma_is_active, ma_from_kf_id, ma_date_oper, ma_plan_disb_time, ma_from_pr_id
                  from miniapp
                  where ma_date_end >= '31.12.2999' and
                        ma_is_active = 1 and
                        ma_date_app between current_date - 20 and current_date -- Действующий заявки

                  union all

                  select ma_id, ma_is_active, ma_from_kf_id, ma_date_oper, ma_plan_disb_time, ma_from_pr_id
                  from miniapp
                  where ma_date_end >= '31.12.2999' and
                        ma_is_active <= 0 and
                        ma_date_app between current_date - 1 and current_date) ma  -- Аннулированные
            join knaktlist kl on kl.kal_from_kf_id = ma.ma_from_kf_id
            where kl.is_delete = 0 and
                  kl.kal_new_reg_id = :reg_id and
                  kl.kal_date <= current_date and
                  kl.kal_date_end >= '31.12.2999'

            into :ma_id, k_status, kf_id, kf_name, :disb_date, ma_plan_disb_time, :pr_id
        do
            suspend;
    end
end
