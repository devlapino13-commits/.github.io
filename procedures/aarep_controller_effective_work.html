CREATE PROCEDURE AAREP_CONTROLLER_EFFECTIVE_WORK(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_NAME_FIO STR_100,
    CHECK_QTY INTEGER,
    REFUSE_QTY INTEGER,
    REG_NAME_BASE STR_100,
    DISB_KONT_QTY_TOTAL INTEGER,
    REP_TYPE STR_100)
AS
declare variable rd_controller_id integer;
begin
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ControllerType.Controller'
  into :rd_controller_id;

  select reg.reg_name
  from region reg
  where reg.reg_unicode = :rep_region_id
  into :reg_name_base;

  rep_type = '(Ежемесячный)';
  if (datediff(day, rep_date_begin, rep_date_end) = 6) then
    rep_type = '(Еженедельный)';

  select count(*)
  from kontstate ks
  inner join region reg on reg.reg_unicode = ks.ks_region_id
  inner join region_list(:rep_region_id, 1, -1, 1) rl on rl.id_list = reg.reg_unicode
  where ks.ks_ol_is_delete = 0 and
        ks.ks_state = 1 and
        ks.ks_date_begin between :rep_date_begin and :rep_date_end
  into :disb_kont_qty_total;

  for select c_name_fio, sum(check_qty) as check_qty, sum(refuse_qty) as refuse_qty
      from (select cl.c_name_fio, 1 as check_qty,
                   iif(rd_res.rd_code = 'ControllerCheckStatus.NotConfirmed', 1, 0) as refuse_qty
            from controller cont
            inner join username un on un.un_id = cont.contr_from_un_id
            inner join clients cl on cl.c_id = un.un_from_c_id
            inner join kontract kont on kont.k_id = cont.contr_from_k_id
            inner join region reg on reg.reg_unicode = kont.k_last_reg_id
            inner join region_list(:rep_region_id, 1, -1, 1) rl on rl.id_list = reg.reg_unicode
            inner join refdata rd_res on rd_res.rd_id = cont.contr_check_result
            where cont.contr_checked_date between :rep_date_begin and :rep_date_end and
                  cont.contr_is_delete = 0 and
                  cont.contr_rd_type = :rd_controller_id) t
      group by c_name_fio
      into :c_name_fio, :check_qty, :refuse_qty
  do
  begin
    suspend;
  end
end
