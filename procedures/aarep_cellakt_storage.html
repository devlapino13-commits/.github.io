CREATE PROCEDURE AAREP_CELLAKT_STORAGE(
    REP_REGION_ID TYPE OF COLUMN REGION.REG_UNICODE,
    REP_DATE_BEGIN D_DATE)
RETURNS (
    CELL_NO TYPE OF COLUMN CELLAKTLIST.CAL_CELL_NO,
    OFFICE TYPE OF COLUMN REGION.REG_NAME,
    OWNER TYPE OF COLUMN CLIENTS.C_NAME_FIO,
    DATE_BEGIN TYPE OF COLUMN CELLAKTLIST.CAL_DATE_BEGIN,
    WHO_CREATE TYPE OF COLUMN USERNAME.UN_FIO,
    DATE_CREATE TYPE OF COLUMN CELLAKTMAIN.CAM_DATETIME)
AS
begin
    for select cal_cell_no,
               c_name_fio,
               reg_name,
               cal_date_begin,
               un_fio,
               cam_datetime
    from cellaktlist
    inner join cellaktmain on cam_id = cal_from_cam_id
    inner join clients on c_id = cal_from_c_id
    inner join region_list(:rep_region_id, 1, -1, 1) on (id_list = cam_from_reg_id)
    inner join region on reg_unicode = id_list
    inner join username on un_id = cam_user_id
    where :rep_date_begin between cal_date_begin and cal_date_end
    and cal_state = 0
    into cell_no, owner, office, date_begin, who_create, date_create
    do
    begin
        suspend;
    end
end
