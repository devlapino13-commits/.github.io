CREATE PROCEDURE AAREP_OPERCC_ONE_EDIT(
    OL_ID DECIMAL(18,0),
    OB_NUMBER INTEGER)
RETURNS (
    OCC_ID INTEGER,
    OCC_FROM_OL_ID DECIMAL(18,0),
    OCC_OB_NUMBER INTEGER,
    OCC_AMOUNT DECIMAL(18,4),
    OCC_MEMO VARCHAR(255),
    OCC_FROM_CC_ID INTEGER,
    OCC_PARENT_CC_ID INTEGER,
    OCC_CC_NAME VARCHAR(255),
    OCC_CC_MEMO VARCHAR(255),
    OB_SUMMA_VAL DECIMAL(18,4),
    OB_DETAIL VARCHAR(1024),
    OB_ACCOUNT_DT VARCHAR(20),
    OB_ACCOUNT_KT VARCHAR(20),
    REG_NAME VARCHAR(100),
    OL_DATE DATE)
AS
begin
  for select ob_from_ol_id, ob_number, ob_account_dt, ob_account_kt, ob_summa_val, ob_detail,
             reg_code || ' ' || reg_name, ob_ol_date
      from operbook, region
      where ob_from_ol_id = :ol_id and
            (ob_number = :ob_number or
            :ob_number = -1) and
            ob_ol_reg_id = reg_unicode
      into :occ_from_ol_id, :occ_ob_number, :ob_account_dt, :ob_account_kt, :ob_summa_val, :ob_detail, :reg_name,
           :ol_date
  do
  begin
    for with occ
        as (select occ_id, occ_amount, occ_memo, occ_from_cc_id
            from opercc
            where occ_from_ol_id = :occ_from_ol_id and
                  occ_ob_number = :occ_ob_number
            union
            select 0 as occ_id, 0 as occ_amount, '' as occ_memo, cc_id as occ_from_cc_id
            from costcenter)
        select occ_from_cc_id, sum(occ_id), sum(occ_amount), max(occ_memo)
        from occ
        group by occ_from_cc_id
        into :occ_from_cc_id, :occ_id, :occ_amount, :occ_memo
    do
    begin
      select cc_name, cc_memo, cc_parent
      from costcenter
      where cc_id = :occ_from_cc_id
      into :occ_cc_name, :occ_cc_memo, :occ_parent_cc_id;

      if (occ_id = 0) then
        occ_id = -1;
      suspend;
    end
  end
end
