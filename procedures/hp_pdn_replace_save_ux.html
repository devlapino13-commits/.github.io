CREATE PROCEDURE HP_PDN_REPLACE_SAVE_UX(
    PRP_ID D_ID,
    KC_ID D_ID,
    PL_ID D_ID,
    PRP_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    RES_TXT D_STR_255)
AS
begin
  if (not exists(select kc_id from knclient where kc_id = :kc_id)) then
  begin
    res_txt = 'Не найден займ ID: ' || kc_id;
    suspend;
    exit;
  end

  if (not exists(select pl_id from places where pl_id = :pl_id)) then
  begin
    res_txt = 'Не найден регион ID: ' || pl_id;
    suspend;
    exit;
  end

  if (prp_date is null) then
  begin
    res_txt = 'Не указана дата';
    suspend;
    exit;
  end

  if (prp_id is not null) then
    update pdn_replace_places
    set prp_kc_id = :kc_id,
        prp_date = :prp_date,
        prp_from_pl_id = :pl_id,
        prp_from_un_id = :un_id
    where prp_id = :prp_id;
  else
    insert into pdn_replace_places (prp_kc_id, prp_date, prp_from_pl_id, prp_create_un_id, prp_from_un_id)
    values (:kc_id, :prp_date, :pl_id, :un_id, :un_id);

  res_txt = 'OK';
  suspend;
end
