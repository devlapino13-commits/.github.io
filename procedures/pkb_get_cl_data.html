CREATE PROCEDURE PKB_GET_CL_DATA(
    POCKET_ID D_INTEGER,
    K_ID D_ID)
RETURNS (
    RECORD_TYPE D_STR_10,
    FIRSTNAME D_STR_100,
    SURNAME D_STR_100,
    FATHERSNAME D_STR_100,
    GENDER D_STR_10,
    CLASSIFICATION D_INTEGER,
    RESIDENCY D_INTEGER,
    DATEOFBIRTH D_DATE,
    PROFESSION D_STR_100,
    ECONOMYACTIVITYGROUP D_INTEGER,
    EDUCATION D_STR_10,
    MARITALSTATUS D_STR_10,
    COMMUNICATION D_STR_100,
    COMMUNICATION_TYPE D_STR_10,
    IDENTIFICATION_TYPE D_INTEGER,
    IDENTIFICATION_TYPE2 D_INTEGER,
    IDENTIFICATION_RANK D_INTEGER,
    IDENTIFICATION_NUMBER D_STR_20,
    IDENTIFICATION_NUMBER2 D_STR_20,
    CITIZENSHIP D_INTEGER,
    ADDRESS D_STR_255,
    ADDRESS_KATO_ID D_STR_20,
    C_ID D_ID,
    ADDRESS_LOCATION_ID D_STR_20,
    ADDRESS_TYPE_ID D_STR_20,
    FACT_ADDRESS D_STR_255,
    FACT_ADDRESS_KATO_ID D_STR_20,
    FACT_ADDRESS_TYPE_ID D_STR_20,
    REGISTRATION_DATE D_DATE,
    REGISTRATION_DATE2 D_DATE)
AS
declare variable REP_DATE_BEGIN D_DATE;
declare variable REP_DATE_END D_DATE;
declare variable REG_ID D_ID;
declare variable REGION_ID D_ID;
declare variable CP_FULL_NAME_RU D_STR_255;
declare variable DOC_TYPE D_INTEGER;
declare variable FL_ID D_ID;
declare variable RD_EDUCATION D_STR_100;
declare variable RD_FAMILY D_STR_100;
declare variable C_NAME_FIO D_STR_255;
declare variable DATEBIRTH D_DATE;
begin
  select cp.cp_date_begin, cp.cp_date_end
  from cib_pocket cp
  where cp.cp_id = :pocket_id
  into :rep_date_begin, :rep_date_end;

  if (rep_date_end is null) then
    exit;

  for select kc.kc_from_c_id
      from knclient kc
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      into c_id
  do
  begin
    select iif(c.c_type = 0, 'Individual', 'Company'), --
           iif(c.c_sex = 0, 'F', 'M'), --
           c.c_birth_date, --
           c.c_from_reg_id, --
           c.c_inn, --
           c.c_from_fl_id, --
           f_get_rd_code(c.c_education), --
           f_get_rd_code(c.c_family), --
           c.c_name_fio, --
           c.c_citizen,
           cast(c.c_datetime as date)
    from clients c
    where c.c_id = :c_id
    into record_type, --
         gender, --
         datebirth, --
         reg_id, --
         identification_number, --
         fl_id, --
         rd_education, --
         rd_family, --
         c_name_fio, --
         citizenship, --
         registration_date;

    select iif(pdt.pdt_code = 'passport_card', 7, 6), cds.cds_doc_no, cds.cds_doc_date_begin
    from clientdocs cds
    inner join passkind pk on pk.pk_id = cds.cds_from_pk_id and
                              pk.pk_is_delete = 0 and
                              pk.pk_country = 'kaz'
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where cds.cds_from_c_id = :c_id and
          cds.cds_is_delete = 0 and
          coalesce(cds.cds_doc_no, '') <> '' and
          cds.cds_is_main_doc = 1
          order by cds.cds_id desc
          rows 1
    into  identification_type2, identification_number2, registration_date2;

    communication = f_cont_info('clients', :c_id, 'cell_main', 'ok');
    communication_type = '1';
    identification_type = 14; --ИИН
    identification_rank = 1;
    if (communication_type = '') then
      communication = '';

    select adl.adl_address_str, right(pl.pl_code, 9)
    from addrlist adl
    inner join addrclient adc on adc.adc_from_adl_id = adl.adl_id and
          adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1
    inner join places pl on pl.pl_id = adl.adl_from_pl_id
    into address,  address_kato_id;
    if (coalesce(address_kato_id, '') = '') then
       address_kato_id = '750000000';
    address_location_id = '110';
    address_type_id = '1';
    select adl.adl_address_str, right(pl.pl_code, 9)
    from addrlist adl
    inner join addrclient adc on adc.adc_from_adl_id = adl.adl_id and
          adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 2
    inner join places pl on pl.pl_id = adl.adl_from_pl_id
    into fact_address,  fact_address_kato_id;
    if (coalesce(fact_address_kato_id, '') = '') then
       fact_address_kato_id = '750000000';
    fact_address_type_id = '1';
    dateofbirth = null;
    education = '';
    maritalstatus = '';
    classification = null;
    residency = null;
    profession = '';
    if (:record_type = 'Individual') then
    begin
      dateofbirth = :datebirth;
      classification = 1;
      residency = 1;

      education = case rd_education
                    when 'Education.22' then '3'
                    when 'Education.20' then '2'
                    when 'Education.19' then '1'
                    else '0'
                  end;

      maritalstatus = case rd_family
                        when 'Family.civilmarriage' then '5'
                        when 'Family.widower' then '4'
                        when 'Family.divorced' then '3'
                        when 'Family.married' then '2'
                        else '1'
                      end;

      select distinct cp.cp_full_name_ru
      from cfp_anketa ca
      inner join cfp_positions cp on cp.cp_id = ca.ca_from_cp_id and
            cp.cp_is_delete = 0
      where ca.ca_from_c_id = :c_id
      order by ca.ca_datetime desc
      rows 1
      into cp_full_name_ru;

      profession = case cp_full_name_ru
                     when 'Работник бюджетной сферы' then '0'
                     when 'Работник сферы услуг/продавец' then '1'
                     when 'Работник производства/рабочий' then '2'
                     when 'Владелец предприятия' then '3'
                     when 'Служащий' then '4'
                     when 'Высший руководитель компании' then '5'
                     when 'Госслужащий / чиновник' then '6'
                     else '7'
                   end;
    end

    select fl.fl_name1, --
           fl.fl_name2, --
           fl.fl_name3
    from fiolist fl
    where fl.fl_id = :fl_id
    into surname, --
         firstname, --
         fathersname;

    if (:fl_id is null) then
        select txt1,
               txt2,
               txt3
        from hp_insert_fiolist(:c_name_fio)
        into fathersname, --
             firstname, --
             surname;

    suspend;
  end
end
