CREATE PROCEDURE FM_ADDR_INFO(
    ADL_ID DECIMAL(18,0))
RETURNS (
    ADL_FROM_PL_ID DECIMAL(18,0),
    ADL_DEPART_NO D_STR_20,
    ADL_VLADEN_NO D_STR_20,
    ADL_FLAT_NO D_STR_20,
    ADL_BUILDING_NO D_STR_20,
    ADL_SECTION_NO D_STR_20,
    ADL_ADDRESS_STR VARCHAR(255),
    ADL_COMMENT VARCHAR(255),
    ADL_POST_INDEX INTEGER,
    ADL_IS_WITHOUT_STREET INTEGER,
    ADL_LANG_KIND INTEGER,
    STREET_ID DECIMAL(18,0),
    STREET_NAME VARCHAR(255),
    STREET_PLT_ID INTEGER,
    STREET_PLT_NAME VARCHAR(255),
    AML_MAP_LINK D_STR_LONG)
AS
begin
  select adl_from_pl_id, adl_depart_no, adl_vladen_no, adl_flat_no, adl_building_no, adl_section_no, adl_address_str,
         adl_comment, adl_post_index, adl_street_id, adl_is_without_street, adl_lang_kind
  from addrlist
  where adl_id = :adl_id
  into :adl_from_pl_id, :adl_depart_no, :adl_vladen_no, :adl_flat_no, :adl_building_no, :adl_section_no,
       adl_address_str, :adl_comment, :adl_post_index, :street_id, :adl_is_without_street, :adl_lang_kind;

  if (street_id <> adl_from_pl_id) then
    adl_is_without_street = 0;

  if (adl_lang_kind is null) then
    adl_lang_kind = 0;

  -- Адрес до улицы
  select place_full
  from aarep_places_lang_link(:adl_from_pl_id, :adl_lang_kind)
  into :adl_address_str;
  if (adl_address_str = 'Не указан') then
    adl_address_str = '';

  -- Название улицы
  if (adl_is_without_street <> 1) then
  begin
    select iif(:adl_lang_kind = 0 or coalesce(pl_name_komitet, '') = '', pl_name, pl_name_komitet),
           plt_id,
           iif(:adl_lang_kind = 0, plt_name, plt_alt_name)
    from places, placetype
    where pl_id = :street_id and
          pl_status = plt_id
    into street_name, street_plt_id, street_plt_name;
  end

  select coalesce(aml.aml_link, '')
  from addrlist_map_links aml
  where aml.aml_from_adl_id = :adl_id
  into aml_map_link;

  suspend;
end
