CREATE PROCEDURE AAREP_KROFF_ALL(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    SBA_FROM_K_ID D_ID,
    SBA_KF_NAME D_STR_100,
    SBA_KONTRACT_NO D_STR_20,
    SBA_DISB_DATE D_DATE,
    SBA_DATE_BEGIN D_DATE,
    SBA_DATE_END D_DATE,
    SBA_REGION_NAME D_STR_100,
    AA_CURR_BRANCH_NAME D_STR_100,
    SBA_LO_FIO D_STR_100,
    SBL_TYPE_STR D_STR_20,
    SBL_NAME D_STR_255,
    SBL_TEAM_NAME D_STR_255,
    AAREP_CLIENTS_KOL D_INTEGER,
    AAREP_CASHIER_FIO D_STR_100,
    AAREP_LEADER_FIO D_STR_100,
    OKO_PAYED_PERC D_CURRENCY,
    OKO_PAYED_OS D_CURRENCY,
    OKO_BALANCE_TOTAL D_CURRENCY,
    OKO_BALANCE_PERC D_CURRENCY,
    OKO_BALANCE_OTHER D_CURRENCY,
    OKO_BALANCE_OS D_CURRENCY,
    OKO_PAYED_OTHER D_CURRENCY,
    OKO_PAYED_OS_PER D_CURRENCY,
    OKO_PAYED_PERC_PER D_CURRENCY,
    OKO_PAYED_OTHER_PER D_CURRENCY,
    OKO_PAYED_TOTAL_PER D_CURRENCY,
    OKO_PAYED_TOTAL D_CURRENCY,
    SBA_AMOUNT_APPROVED CURR_TYPE,
    RES_BALANCE D_CURRENCY,
    RES_DELINQ_DAYS D_INTEGER,
    DELINQ_BALANCE D_CURRENCY,
    PR_NAME D_STR_100,
    AA_STATUS_TEXT D_STR_20,
    AA_DELIQ_REPAY_KOLICH D_INTEGER,
    SBA_FROM_SBL_ID D_STR_100)
AS
declare variable REG_ID_PRIOR D_INTEGER;
declare variable LO_ID_PRIOR D_CURRENCY;
declare variable SBL_ID_PRIOR D_STR_100;
declare variable SBL_TYPE D_INTEGER;
declare variable SBA_LO_ID D_CURRENCY;
declare variable SBA_REGION_ID D_INTEGER;
declare variable AAREP_CASHIER_ID D_CURRENCY;
declare variable AAREP_LEADER_ID D_ID;
declare variable OKO_OS D_CURRENCY;
declare variable OKO_PERC D_CURRENCY;
declare variable OKO_OTHER D_CURRENCY;
declare variable OL_DATE D_DATE;
declare variable TMP_SH_DATE D_DATE;
declare variable PR_ID D_INTEGER;
declare variable PR_COMMON_NAME D_STR_100;
declare variable PR_CODE D_STR_100;
declare variable SBA_FROM_K_ID_OLD D_ID;
begin
  reg_id_prior = 0;
  lo_id_prior = 0;
  sbl_id_prior = 0;
  sba_from_k_id_old = 0;
  for select distinct sba_from_k_id, sba_date_begin, kal_new_kf_officer, kal_new_reg_id, kal_new_kf_name, k_kontract_no,
                      k_kolich_client_approved, k_clients_cashier_id, k_clients_leader_id, k_disb_date, k_from_pr_id,
                      k_amount_approved, sba_from_sbl_id, coalesce(sba_date_end, '31.12.9999')
      from sb_akt sba
      inner join kontract k on sba_from_k_id = k_id
      inner join knaktlist kal on (k_from_kf_id = kal_from_kf_id and
            kal.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end)
      where sba_is_delete = 0 and
            sba_from_sbl_id <> 0 and
            sba_date_begin between :rep_date_begin and :rep_date_end and
            sba_status = 2 -- беру только утвержденные (-1 - аннулирован, 0 - в процессе, 1 - согласован, 2 - утвержден)
      into :sba_from_k_id, :sba_date_begin, :sba_lo_id, :sba_region_id, :sba_kf_name, :sba_kontract_no,
           :aarep_clients_kol, :aarep_cashier_id, :aarep_leader_id, :sba_disb_date, pr_id, :sba_amount_approved,
           :sba_from_sbl_id, sba_date_end
  do
  begin
    if (sba_date_end >= rep_date_end) then
      sba_date_begin = rep_date_end;

    if (sba_from_k_id <> sba_from_k_id_old) then
    begin
      sba_from_k_id_old = sba_from_k_id;

      aarep_cashier_fio = null;
      select c_name_fio
      from clients
      where c_id = :aarep_cashier_id
      into :aarep_cashier_fio;

      aarep_leader_fio = null;
      select c_name_fio
      from clients
      where c_id = :aarep_leader_id
      into :aarep_leader_fio;

      pr_common_name = null;
      pr_name = null;
      pr_code = null;
      select pr.pr_common_name, pr_name, pr.pr_code
      from product pr
      where pr_id = :pr_id
      into pr_common_name, pr_name, pr_code; --Название продукта
    end

    if (sba_region_id <> reg_id_prior) then
    begin
      select reg1.reg_code || ' ' || reg1.reg_name, reg2.reg_code || ' ' || reg2.reg_name
      from region reg1, region reg2
      where reg1.reg_unicode = :sba_region_id and
            reg2.reg_unicode = reg1.parent
      into :sba_region_name, :aa_curr_branch_name;

      reg_id_prior = sba_region_id;
    end

    if (sba_lo_id <> lo_id_prior) then
    begin
      select c_name_fio
      from clients
      where c_id = :sba_lo_id
      into :sba_lo_fio;
      lo_id_prior = sba_lo_id;
    end

    if (sbl_id_prior <> sba_from_sbl_id) then
    begin
      sbl_team_name = null;
      sbl_type = null;
      sbl_name = null;

      select sbl.sbl_name, sbl.sbl_team_name, sbl.sbl_type
      from sb_list sbl
      where sbl_id = :sba_from_sbl_id
      into :sbl_name, :sbl_team_name, :sbl_type;

      sbl_type_str = 'Не определен';
      if (sba_from_sbl_id = -1) then
        sbl_type_str = 'Модуль ОРиВ';
      else


      --if (sba_from_sbl_id = -2) then
      --  sbl_type_str = 'Внесистемный учет';
      --else
      if (sbl_type = 0) then
        sbl_type_str = 'СБ';
      else
      if (sbl_type = 1) then
        sbl_type_str = 'Коллекторы';

      sbl_id_prior = sba_from_sbl_id;

      --вылаченные
      oko_payed_os = 0;
      oko_payed_perc = 0;
      oko_payed_other = 0;

      --Выплаченные за период отчета
      oko_payed_os_per = 0;
      oko_payed_perc_per = 0;
      oko_payed_other_per = 0;

      aa_deliq_repay_kolich = 0;

      for select okf.oko_ol_date, okf.oko_os, okf.oko_perc, okf.oko_other, null
          from operkroff okf
          where okf.oko_from_k_id = :sba_from_k_id and
                okf.oko_ol_is_active = 1 and
                okf.oko_ol_date <= :sba_date_begin and
                okf.oko_type = 4
          union all
          select ok_ol_date, ok_os, ok_perc, ok_fine + ok_kommis + ok.ok_tax ok_other, ok_sh_date
          from operkred ok
          where ok_from_k_id = :sba_from_k_id and
                ok_ol_is_active = 1 and
                ok_ol_date <= :sba_date_begin and
                ok_type = 4
          into :ol_date, :oko_os, :oko_perc, :oko_other, tmp_sh_date
      do
      begin
        oko_payed_os = oko_payed_os + oko_os;
        oko_payed_perc = oko_payed_perc + oko_perc;
        oko_payed_other = oko_payed_other + oko_other;

        if (ol_date between :rep_date_begin and :rep_date_end) then
        begin
          oko_payed_os_per = oko_payed_os_per + oko_os;
          oko_payed_perc_per = oko_payed_perc_per + oko_perc;
          oko_payed_other_per = oko_payed_other_per + oko_other;
        end
        if (tmp_sh_date < ol_date) then
          aa_deliq_repay_kolich = aa_deliq_repay_kolich + 1;
      end
      oko_payed_os = -oko_payed_os;
      oko_payed_perc = -oko_payed_perc;
      oko_payed_other = -oko_payed_other;
      oko_payed_os_per = -oko_payed_os_per;
      oko_payed_perc_per = -oko_payed_perc_per;
      oko_payed_other_per = -oko_payed_other_per;

      oko_payed_total_per = oko_payed_os_per + oko_payed_perc_per + oko_payed_other_per;
      oko_payed_total = oko_payed_os + oko_payed_perc + oko_payed_other;

      oko_balance_os = 0;
      oko_balance_perc = 0;
      oko_balance_other = 0;
      oko_balance_total = 0;
      delinq_balance = 0; --Баланс просрочки
      tmp_sh_date = null;

      select sum(zad_os), sum(zad_perc), sum(sh_calc_fine + (zad_perc * 0.02)),
             sum(iif(sh_date < :rep_date_end, zad_os, 0))
      from aarep_make_zad_list(:sba_from_k_id, :sba_date_begin, 1)
      into oko_balance_os, oko_balance_perc, oko_balance_other, delinq_balance;

      oko_balance_total = oko_balance_os + oko_balance_perc + oko_balance_other;

      res_balance = 0;
      res_delinq_days = 0; --Кол-во просроч дней
      select oks_balance, :sba_date_begin - oks_delinq_date
      from oksummary oks
      where :sba_date_begin between oks_date_begin and oks_date_end and
            oks_from_k_id = :sba_from_k_id
            and oks_close_state = 0 
      into res_balance, res_delinq_days;

      if (coalesce(res_delinq_days, -1) < 0) then
        res_delinq_days = 0;

      select
             case ks_state
               when 0 then 'В процессе'
               when 1 then 'Активный'
               when 2 then 'Закрытый'
             end
      from kontstate ks
      where ks.ks_from_k_id = :sba_from_k_id and
            ks_ol_is_delete = 0 and
            :sba_date_begin between ks_date_begin and ks_date_end
      into aa_status_text;

      if (oko_balance_total > 0 and
          aa_status_text = 'Закрытый') then
        aa_status_text = 'Внесистемный учет'; --Статус кредита

      suspend;
    end
  end
end
