CREATE PROCEDURE HP_EXTINFO_GET_JSO(
    EIC_CODE D_STR_CODE,
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_ID)
RETURNS (
    JSO D_INTEGER,
    REQ_STATUS D_SMALLINT,
    REQ_DATE D_DATE,
    ACTUALITY_DATE D_DATE)
AS
declare variable bls_id d_id;
declare variable ma_id d_id;
declare variable c_id d_id;
begin
  if (object_kind = 'blobstore') then
  begin
    select eil.eil_status,
           eil.eil_date,
           eil.eil_date + eic.eic_info_actual_days,
           eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eic.eic_code = :eic_code and
          eil.eil_from_bls_id = :object_id
    order by eil.eil_id desc rows 1
    into req_status,
         req_date,
         actuality_date,
         bls_id;
  end
  else if (object_kind = 'extinfolist') then
  begin
    select eil.eil_status,
           eil.eil_date,
           eil.eil_date + eic.eic_info_actual_days,
           eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eic.eic_code = :eic_code and
          eil.eil_id = :object_id
    order by eil.eil_id desc rows 1
    into req_status,
         req_date,
         actuality_date,
         bls_id;
  end
  else if (object_kind in ('miniapp', 'kontract')) then
  begin

    if (object_kind = 'kontract') then
    begin
      select k.k_from_ma_app_id
      from kontract k
      where k.k_id = :object_id
      into :ma_id;
    end
    if (object_kind = 'miniapp') then
      ma_id = object_id;

    select eil.eil_status,
           eil.eil_date,
           eil.eil_date + eic.eic_info_actual_days,
           eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eic.eic_code = :eic_code and
          eil.eil_object_kind = f_objkind('miniapp', 'id') and
          eil.eil_object_id = :ma_id
    order by eil.eil_id desc rows 1
    into req_status,
         req_date,
         actuality_date,
         bls_id;
  end
  else if (object_kind = 'knclient') then
  begin
    select k.k_from_ma_app_id,
           kc.kc_from_c_id
    from knclient kc
    inner join kontract k on k.k_id = kc.kc_from_k_id
    where kc.kc_id = :object_id
    into ma_id, c_id;

    select eil.eil_status,
           eil.eil_date,
           eil.eil_date + eic.eic_info_actual_days,
           eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eic.eic_code = :eic_code and
          eil.eil_object_kind = f_objkind('miniapp', 'id') and
          eil.eil_object_id = :ma_id and
          eil.eil_from_c_id = :c_id
    order by eil.eil_id desc rows 1
    into req_status,
         req_date,
         actuality_date,
         bls_id;
  end
  else if (object_kind = 'clients') then
  begin
    select eil.eil_status,
           eil.eil_date,
           eil.eil_date + eic.eic_info_actual_days,
           eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eic.eic_code = :eic_code and
          eil.eil_from_c_id = :object_id
    order by eil.eil_id desc rows 1
    into req_status,
         req_date,
         actuality_date,
         bls_id;
  end
  else
    exception e_default 'Unknown object_kind = ' || coalesce(object_kind, '<NULL>');

  if (bls_id is not null) then
    select iif(blx_blob is null, null, json.parse(blx_blob))
    from hp_get_blobex_blob(:bls_id)
    into jso;

  suspend;
end
