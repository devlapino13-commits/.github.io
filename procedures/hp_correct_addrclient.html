CREATE PROCEDURE HP_CORRECT_ADDRCLIENT
RETURNS (
    C_ID DECIMAL(18,0),
    PL_ID DECIMAL(18,0),
    ADC_TYPE INTEGER,
    ADL_ID DECIMAL(18,0),
    ADL_FIRST DECIMAL(18,0))
AS
begin
  for select distinct ca_from_c_id
      from caddress
      into :c_id
  do
  begin
    pl_id = null;
    select ca_from_pl_id
    from caddress
    where ca_is_main = 1
    into :pl_id;
    if (pl_id is not null) then -- если есть основной адрес в старом формате
    begin
      if (not(exists(select *
                     from addrclient, addrlist
                     where adc_from_adl_id = adl_id and
                           adl_from_pl_id = :pl_id and
                           adc_address_type = 1 and
                           adc_from_c_id = :c_id))) then -- если основной адрес не является первой строкой в новом формате
      begin
        adc_type = null;
        select adc_address_type, adc_from_adl_id
        from addrclient, addrlist
        where adc_from_adl_id = adl_id and
              adl_from_pl_id = :pl_id and
              adc_from_c_id = :c_id
        into :adc_type, :adl_id;
        if (adc_type is not null) then -- находим в какой строке находится основной адрес
        begin
          adl_first = null;
          select adc_from_adl_id
          from addrclient
          where adc_from_c_id = :c_id and
                adc_address_type = 1
          into :adl_first;

-- меняем местами первую строку и найденный основной адрес
          update addrclient
          set adc_from_adl_id = :adl_id
          where adc_from_c_id = :c_id and
                adc_address_type = 1;
          if (adl_first is not null) then
            update addrclient
            set adc_from_adl_id = :adl_first
            where adc_from_c_id = :c_id and
                  adc_address_type = :adc_type;
        end
      end
    end
  end
end
