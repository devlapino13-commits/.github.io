CREATE PROCEDURE AAREP_CENTERBANK_REP_DATA_2022(
    REP_DATE D_DATE,
    REP_QUERY_DATE D_DATE,
    K_ID D_ID)
RETURNS (
    OKS_BALANCE D_CURRENCY,
    OKS_BAL_PERC D_CURRENCY,
    OKS_BAL_ZAD_OS D_CURRENCY,
    OKS_BAL_ZAD_PERC D_CURRENCY,
    OKS_DELINQ_DAYS D_INTEGER,
    OKS_FIRST_DELINQ_90 D_DATE,
    OKS_PMT_OS D_CURRENCY,
    OKS_PMT_PERC D_CURRENCY,
    SELL_DATE D_DATE,
    SELL_PMT_AMOUNT D_CURRENCY,
    WRITEOFF_DATE D_DATE,
    LAST_DATE D_DATE,
    OKS_KONT_LAST_DATE D_DATE,
    OKS_LLR_OS_NADZOR CURR_TYPE,
    OKS_LLR_PERC_NADZOR CURR_TYPE,
    OKS_ECL_VBS CURR_TYPE,
    OK_FINE CURR_TYPE)
AS
declare variable rep_date_begin d_date;
declare variable oks_close_state d_smallint;
declare variable oks_date_begin d_date;
declare variable oks_delinq_days_llr d_integer;
begin
  oks_balance = 0;
  oks_bal_perc = 0;
  oks_bal_zad_os = 0;
  oks_bal_zad_perc = 0;
  oks_delinq_days = 0;
  oks_first_delinq_90 = null;
  oks_pmt_os = 0;
  oks_pmt_perc = 0;
  sell_date = null; -- всегда пусто
  sell_pmt_amount = 0; -- всегда 0
  writeoff_date = null;
  last_date = null;
  oks_llr_os_nadzor = 0;
  oks_llr_perc_nadzor = 0;
  oks_ecl_vbs = 0;
  oks_delinq_days_llr = 0;
  ok_fine=0;

  rep_date_begin = rep_date - extract(day from rep_date) + 1;

  select max(oks_date_begin), sum(oks_balance), sum(oks_bal_perc), sum(oks_bal_zad_os), sum(oks_bal_zad_perc),
         max(oks_restruct_date), max(oks_close_state), max(oks_kont_last_date), sum(oks_llr_os_nadzor),
         sum(oks_llr_perc_nadzor), sum(oks_ecl_vbs), max(oks_delinq_days_llr)
  from kn_oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin <= :rep_date and
        oks_date_end >= :rep_date
  into oks_date_begin, oks_balance, oks_bal_perc, oks_bal_zad_os, oks_bal_zad_perc,
       last_date, oks_close_state, oks_kont_last_date, oks_llr_os_nadzor, oks_llr_perc_nadzor, oks_ecl_vbs,
       oks_delinq_days_llr;

  if (oks_date_begin is null) then
  begin
    oks_balance = 0;
    oks_bal_perc = 0;
    oks_bal_zad_os = 0;
    oks_bal_zad_perc = 0;
    oks_close_state = 2;
    oks_llr_os_nadzor = 0;
    oks_llr_perc_nadzor = 0;
    oks_ecl_vbs = 0;
    oks_delinq_days_llr = 0;
   end

  if (oks_date_begin is null or oks_close_state <> 0 or oks_date_begin < rep_date) then
  begin
    oks_balance = 0;
    oks_bal_zad_os = 0;
    oks_bal_zad_perc = 0;
    oks_delinq_days_llr = 0;
    oks_ecl_vbs = 0;
  end
  else
  begin
    if (oks_balance <> oks_bal_zad_os and
        abs(oks_balance - oks_bal_zad_os) < 10) then
    begin
      if (f_tmp_kc(:k_id) = 1) then
        oks_bal_zad_os = oks_balance;
    end
  end

  if (oks_bal_zad_os > oks_balance) then
    oks_bal_zad_os = oks_balance;

  if (oks_delinq_days_llr is not null) then
    oks_delinq_days = oks_delinq_days_llr;
  if (oks_delinq_days < 0) then
    oks_delinq_days = 0;

  if (last_date > rep_date) then
    last_date = null;

  select sum(-ok_os), sum(-ok_perc), sum(-ok_fine)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_ol_date between :rep_date_begin and :rep_date and
        ok_type = 4
  into oks_pmt_os, oks_pmt_perc, ok_fine;

  if (oks_pmt_os is null) then
    oks_pmt_os = 0;
  if (oks_pmt_perc is null) then
    oks_pmt_perc = 0;

  oks_first_delinq_90 = null;
  writeoff_date = null;

  select min(ok_ol_date)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_date <= :rep_query_date and
        ok_ol_is_active = 1 and
        ok_type = 5 and
        (ok_os < 0 or ok_perc < 0)
  into writeoff_date;

  select oks_delinq_date + 91
  from kn_oksummary
  where oks_from_k_id = :k_id and
        oks_date_begin <= :rep_query_date and
        oks_delinq_days >= 91 and
        oks_delinq_date is not null
  order by oks_date_begin
  rows 1
  into :oks_first_delinq_90;

  last_date = minvalue(coalesce(last_date, '31.12.9999'), coalesce(oks_first_delinq_90 + 91, '31.12.9999'));
  last_date = minvalue(coalesce(last_date, '31.12.9999'), coalesce(sell_date, '31.12.9999'));

  if (last_date = '31.12.9999') then
    last_date = null;

  suspend;
end
