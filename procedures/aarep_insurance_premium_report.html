CREATE PROCEDURE AAREP_INSURANCE_PREMIUM_REPORT(
    REP_REG_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_64,
    KAL_NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_255,
    LO_NAME_FIO D_STR_255,
    K_FACT_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    INSURANCE_AMOUNT D_CURRENCY,
    PAYED_AMOUNT D_CURRENCY,
    PAYED_DATE D_DATE,
    PR_NAME D_STR_CODE)
AS
declare variable c_id d_id;
declare variable lo_id d_id;
declare variable k_id d_id;
declare variable reg_unicode d_integer;
begin
  -- Отчет по оплате страховой премии клиентами.
  -- Построен по аналогии с AAREP_INSURANCE_PAYMENTS_REPORT,
  -- отбор идет по дате активации договора (KS_DATE_BEGIN), все продукты.

  for select distinct
      t.id_list,
      kal.kal_new_kf_name,
      k.k_kontract_no,
      k.k_fact_disb_lo_id,
      k.k_fact_disb_date,
      k.k_amount_approved,
      ik.ik_commission,
      pr.pr_name,
      kc.kc_from_c_id,
      k.k_id
  from kontstate ks
  inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
        kc.kc_amount_approved > 0
  inner join insurance_kontract ik on ik.ik_from_k_id = ks.ks_from_k_id and
        (ik.ik_enum_status in (f_enum_id('insurance_kontract', 'insurance_policy_obtained'),
                               f_enum_id('insurance_kontract', 'insurance_policy_app_was_sent')) or
         ik.ik_from_ag_id in (7414, 7570))
  inner join kontract k on k.k_id = ks.ks_from_k_id
  inner join product pr on pr.pr_id = k.k_from_pr_id and
        pr.pr_code containing '.ipf.' and
        pr.pr_code not containing '.ignore_insurance.'
  inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id and
        kal.is_delete = 0 and
        :rep_date_end between kal.kal_date and kal.kal_date_end
  inner join region_list(:rep_reg_id, 1, 3, 1) t on t.id_list = ks.ks_region_id
  where ks.ks_ol_is_delete = 0 and
        ks.ks_state = 1 and
        ks.ks_date_begin between :rep_date_begin and :rep_date_end
  into
      :reg_unicode,
      :kal_new_kf_name,
      :k_kontract_no,
      :lo_id,
      :k_fact_disb_date,
      :k_amount_approved,
      :insurance_amount,
      :pr_name,
      :c_id,
      :k_id
  do
  begin
    -- Офис
    select reg.reg_name
    from region reg
    where reg.reg_unicode = :reg_unicode
    into :reg_name;

    -- ФИО клиента
    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into :c_name_fio;

    -- ФИО специалиста (выдавший ВИПС)
    lo_name_fio = '';
    select c.c_name_fio
    from clients c
    where c.c_id = :lo_id
    into :lo_name_fio;

    -- Оплаченная сумма и дата оплаты страховой премии.
    -- Возможны два сценария проводок:
    --  1) сразу в кассу: Дт 10001/xxx (или 11510/x) -> Кт 21122/xxx - оплата;
    --  2) через требование: Дт 11531/xxx -> Кт 21122/xxx - это НАЧИСЛЕНИЕ
    --     в день выдачи, а оплата - закрытие требования: Кт 11531/xxx.
    -- Поэтому оплатой считаем проводки, где по кредиту 11531/xxx или
    -- 21122/xxx, а по дебету НЕ эти счета (т.е. деньги пришли извне).
    -- Договор определяем по клиенту (индексируемые OB_CLIENT_DT/KT)
    -- и уточняем номером договора в OL_MEMO.
    payed_amount = 0;
    payed_date = null;

    select coalesce(sum(ob.ob_summa), 0),
           max(ol.ol_date)
    from operbook ob
    inner join operlist ol on ol.ol_id = ob.ob_from_ol_id and
          ol.is_delete = 0
    where (ob.ob_client_dt = :c_id or
           ob.ob_client_kt = :c_id) and
          ob.ob_ol_is_active = 1 and
          ob.ob_short_acc_dt not starting '11531/' and
          ob.ob_short_acc_dt not starting '21122/' and
          (ob.ob_short_acc_kt starting '11531/' or
           ob.ob_short_acc_kt starting '21122/') and
          ol.ol_memo containing :k_kontract_no || ' от ' and
          ol.ol_date <= :rep_date_end
    into :payed_amount,
         :payed_date;

    payed_amount = coalesce(payed_amount, 0);

    suspend;
  end
end
