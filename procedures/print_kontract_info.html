Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE PRINT_KONTRACT_INFO(
    K_ID D_ID)
RETURNS (
    SH_ID D_INTEGER,
    S_DATE D_DATE,
    ZAD_OS D_CURRENCY,
    ZAD_PERC D_CURRENCY,
    CALCING_PERC_ORIG D_CURRENCY,
    PAY_TAX D_CURRENCY,
    ZAD_FINE D_CURRENCY,
    S_CALC_ALL D_CURRENCY)
AS
declare variable percent_for_int decimal(18,7);
declare variable round_perc d_smallint;
begin
  select nsp_percent_for_int, pr_round_perc
  from hp_get_kontract_params(:k_id, current_date)
  into :percent_for_int, :round_perc;

  for select sh_id, sh_date, zad_os, zad_perc, sh_calc_perc, round((zad_perc * :percent_for_int / 100), :round_perc),
             sh_calc_fine,
             (zad_os + zad_perc + sh_calc_fine + round((zad_perc * :percent_for_int / 100), :round_perc))
      from aarep_make_zad_list(:k_id, current_date, 1, 0)
      into :sh_id, :s_date, :zad_os, :zad_perc, :calcing_perc_orig, :pay_tax, :zad_fine, :s_calc_all
  do
  begin
    suspend;
  end
end
