CREATE PROCEDURE FM_ANKETA_LIST(
    UN_ID D_ID,
    RD_ID D_INTEGER)
RETURNS (
    ID D_ID,
    TITLE D_STR_255,
    DESCRIPTION D_STR_100,
    STATUS D_INTEGER,
    KIND D_INTEGER,
    CAN_CANCEL D_INTEGER)
AS
declare variable obj_kind d_integer;
declare variable obj_kind1 d_integer;
declare variable obj_kind2 d_integer;
declare variable obj_kind3 d_integer;
declare variable obj_kind4 d_integer;
declare variable obj_id d_id;
declare variable acl_datetime d_datetime;
declare variable c_id d_id;
declare variable is_manager d_integer;
declare variable reg_id d_id;
declare variable kc_un_id d_id;
declare variable db_id d_id;
begin
  db_id = f_get_db_id();
  -- kind = 0 (Анкета)
  -- kind = 1 (Мониторинг)
  -- can_cancel = 1 (Можно отменить в интерфейсе)
  -- can_cancel = 0 (Отменить нельзя)

  obj_kind1 = f_objkind('Clients', 'id');
  obj_kind2 = f_objkind('KnClient', 'id');
  obj_kind3 = f_objkind('MiniAppClient', 'id');
  obj_kind4 = f_objkind('Kontract', 'id');

  if(db_id = 41701) then
  begin
    select un_from_c_id
    from username
    where un_id = :un_id
    into :c_id;
    
    select ps.ps_from_reg_id
    from hr_pstatus ps
    inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
    inner join jobkind jk on jl.jl_kind = jk.jk_id
        and jk.jk_system_code = 'JobSystem.Region_KM'
    where current_date between ps.ps_date_begin and ps.ps_date_end
        and ps.ps_from_c_id = :c_id
        and ps.ps_is_on_staff = 1
        and ps.ps_is_working = 1
        and ps.ps_status = 1
    into :reg_id;
    if (reg_id is not null) then
      is_manager = 1;
  end

  for select aco_id, aco_status
  from actionorder
  inner join hp_staff_changer_list(:un_id, 1) on res_c_id = aco_respons_c_id
  where :rd_id in (0, aco_from_rd_id)
  order by aco_status, aco_id
  into :id,
       :status
  do
  begin
    for select acl_object_kind,
        acl_object_id,
        rd_name,
        iif(rd_str_id = 'Anketa', 0, 1),
        iif(rd_code in ('Anketa.Main', 'CheckAnketa.Home', 'CheckAnketa.Business'), 1, 0),
        acl_datetime
    from actionlist
    inner join refdata on rd_id = acl_from_rd_id
    where acl_id = :id
        and acl_object_kind in (:obj_kind1, :obj_kind2, :obj_kind3, :obj_kind4)
        and acl_from_rd_id = rd_id
    into :obj_kind,
         :obj_id,
         :description,
         :kind,
         :can_cancel,
         :acl_datetime
    do
    begin
      if (obj_kind = obj_kind1) then
        title = ( select c_name_fio
                       from clients where c_id = :obj_id);
      else
      if (obj_kind = obj_kind2) then
        title = ( select c_name_fio || ', договор ' || k_kontract_no
                       from knclient
                       inner join kontract on k_id = kc_from_k_id
                       inner join clients on c_id = kc_from_c_id
                       where kc_id = :obj_id);
      else
      if (obj_kind = obj_kind3) then
        title = ( select c_name_fio || ', заявка ' || ma_no
                       from miniappclient
                       inner join miniapp on ma_id = mac_from_ma_id
                       and ma_is_active = 1
                       and ma_date_end = '31.12.9999'
                       inner join clients on c_id = mac_from_c_id
                       where mac_id = :obj_id);
      else
      if (obj_kind = obj_kind4) then
        title = (select kt.k_kontract_no || ', ' || left(:acl_datetime, 19) || ', ' || cl.c_name_fio
                 from kontract kt
                 inner join clients cl on cl.c_id = kt.k_clients_cashier_id
                 where kt.k_id = :obj_id);
      suspend;
    end
  end
                -- RFCKG-3667 --
  if(db_id = 41701 and is_manager = 1 and rd_id in (0, 8941)) then  -- RFCKG-4460 Добавил and rd_id in (0, 8941))
  begin
    for select un.un_id       -- находим КС-ов по офису менеджера
    from username un
    join hr_pstatus ps on ps.ps_from_c_id = un.un_from_c_id
    join joblist jl on jl.jl_id = ps.ps_from_jl_id
    inner join jobkind jk on jl.jl_kind = jk.jk_id
        and jk.jk_system_code = 'JobSystem.LO_Standart'
    where ps.ps_from_reg_id = :reg_id
        and ps.ps_is_on_staff = 1
        and ps.ps_is_working = 1
        and ps.ps_status = 1
        and current_date between ps.ps_date_begin and ps.ps_date_end
    into :kc_un_id
    do
    begin
      for select id, title, description, status, kind, can_cancel
      from fm_anketa_list(:kc_un_id, f_get_rd_id('ActionGroup.Check_loan_fix'))
      into :id, :title, :description, :status, :kind, :can_cancel
      do
      begin
        suspend;
      end
    end
  end
                  -- RFCKG-4460 --
  if(db_id = 41701 and is_manager = 1 and rd_id in (0, 1622)) then
  begin
    for select un.un_id       -- находим КС-ов по офису менеджера
    from username un
    join hr_pstatus ps on ps.ps_from_c_id = un.un_from_c_id
    join joblist jl on jl.jl_id = ps.ps_from_jl_id
    inner join jobkind jk on jl.jl_kind = jk.jk_id
        and jk.jk_system_code = 'JobSystem.LO_Standart'
    where ps.ps_from_reg_id = :reg_id
        and ps.ps_is_on_staff = 1
        and ps.ps_is_working = 1
        and ps.ps_status = 1
        and current_date between ps.ps_date_begin and ps.ps_date_end
    into :kc_un_id
    do
    begin
      for select id, title, description, status, kind, can_cancel
      from fm_anketa_list(:kc_un_id, f_get_rd_id('Anketa.TargetUseLoan'))
      into :id, :title, :description, :status, :kind, :can_cancel
      do
      begin
        suspend;
      end
    end
  end
end
