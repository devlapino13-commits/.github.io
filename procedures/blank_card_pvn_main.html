CREATE PROCEDURE BLANK_CARD_PVN_MAIN(
    OL DECIMAL(18,0))
RETURNS (
    BL_TICKET_NO D_STR_CODE,
    BL_UN_FIO VARCHAR(100),
    BL_DATE DATE,
    BL_ACCOUNT VARCHAR(20),
    BL_SUMMA DECIMAL(18,4),
    BL_CLIENT_NAME VARCHAR(255),
    BL_CLIENT_INN VARCHAR(20),
    BL_OFFICE_FULL_NAME_KG1 VARCHAR(255),
    BL_OFFICE_FULL_NAME_KG2 VARCHAR(255),
    BL_MEMO VARCHAR(1024),
    BL_DOC_NO INTEGER)
AS
begin

  select pvn.po_ticket_no
  from pvnopers pvn
  where pvn.po_from_ol_id = :ol
  rows 1
  into :bl_ticket_no;

  select ol.ol_date, ol.ol_doc_no, un.un_fio, c.c_name_fio, c.c_inn, r.reg_code || ' ' || r.reg_name, ol.ol_memo
  from operlist ol
    inner join region r on ol.ol_from_reg_id = r.reg_unicode
    inner join username un on ol.ol_user_id = un.un_id
    inner join clients c on ol.ol_document_client = c.c_id
  where ol_id = :ol
  into :bl_date, :bl_doc_no, :bl_un_fio, :bl_client_name, :bl_client_inn, :bl_office_full_name_kg1, :bl_memo;

  select ob_account_dt, sum(ob_summa_val)
  from operbook ob
  where ob_from_ol_id = :ol
    and exists(
                 select 1
                 from acc_list a_dt
                   inner join acctypelist atl on a_dt.al_from_bg_id starting atl_acc_mask
                 where a_dt.al_id = ob.ob_account_dt
                   and atl.atl_from_at_id in (16, 45)
               )
  group by ob_account_dt
  into :bl_account, :bl_summa;

  select cm_name
  from company
  rows 1
  into :bl_office_full_name_kg2;

  suspend;
end
