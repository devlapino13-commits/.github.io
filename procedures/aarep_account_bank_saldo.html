CREATE PROCEDURE AAREP_ACCOUNT_BANK_SALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_PRODUCT_KIND STR_20 = null)
RETURNS (
    LINE_NAME VARCHAR(255),
    SALDO_HQ_417 DECIMAL(18,4),
    SALDO_HQ_840 DECIMAL(18,4),
    SALDO_HQ_978 DECIMAL(18,4),
    SALDO_HQ_OTHER DECIMAL(18,4),
    SALDO_BRU DECIMAL(18,4),
    SALDO_ISK DECIMAL(18,4),
    SALDO_OSH DECIMAL(18,4),
    SALDO_JAL DECIMAL(18,4),
    SALDO_NAR DECIMAL(18,4),
    SALDO_TAL DECIMAL(18,4),
    SALDO_KAD DECIMAL(18,4),
    SALDO_BAT DECIMAL(18,4),
    SALDO_UZG DECIMAL(18,4),
    SALDO_TASH DECIMAL(18,4),
    SALDO_CHUY DECIMAL(18,4),
    SALDO_BISH DECIMAL(18,4),
    SALDO_OTHER DECIMAL(18,4),
    SALDO_REGION DECIMAL(18,4),
    CR_RATE DECIMAL(18,4))
AS
declare variable bg_id varchar(20);
declare variable bg_id_old varchar(20);
declare variable num integer;
declare variable saldo_hq_417_tmp decimal(18,4);
declare variable saldo_hq_840_tmp decimal(18,4);
declare variable saldo_hq_978_tmp decimal(18,4);
declare variable saldo_hq_other_tmp decimal(18,4);
declare variable saldo_bru_tmp decimal(18,4);
declare variable saldo_isk_tmp decimal(18,4);
declare variable saldo_osh_tmp decimal(18,4);
declare variable saldo_jal_tmp decimal(18,4);
declare variable saldo_tash_tmp decimal(18,4);
declare variable saldo_nar_tmp decimal(18,4);
declare variable saldo_tal_tmp decimal(18,4);
declare variable saldo_kad_tmp decimal(18,4);
declare variable saldo_bat_tmp decimal(18,4);
declare variable saldo_uzg_tmp decimal(18,4);
declare variable saldo_chuy_tmp decimal(18,4);
declare variable saldo_bish_tmp decimal(18,4);
declare variable saldo_other_tmp decimal(18,4);
declare variable saldo_region_tmp decimal(18,4);
declare variable line_name_tmp varchar(255);
declare variable sum_saldo_hq_417 decimal(18,4);
declare variable sum_saldo_hq_840 decimal(18,4);
declare variable sum_saldo_hq_978 decimal(18,4);
declare variable sum_saldo_hq_other decimal(18,4);
declare variable sum_saldo_bru decimal(18,4);
declare variable sum_saldo_isk decimal(18,4);
declare variable sum_saldo_osh decimal(18,4);
declare variable sum_saldo_jal decimal(18,4);
declare variable sum_saldo_tash decimal(18,4);
declare variable sum_saldo_nar decimal(18,4);
declare variable sum_saldo_tal decimal(18,4);
declare variable sum_saldo_kad decimal(18,4);
declare variable sum_saldo_bat decimal(18,4);
declare variable sum_saldo_uzg decimal(18,4);
declare variable sum_saldo_chuy decimal(18,4);
declare variable sum_saldo_bish decimal(18,4);
declare variable sum_saldo_other decimal(18,4);
declare variable sum_saldo_region decimal(18,4);
begin
  -- вычисляет остаток по денежных средств в разбивке по округам.
  -- работает только для кр
  select op.cr_rate
  from opercurr op
  where op.cr_from_cu_id = '840' and
        :report_date_end between op.cr_date and op.cr_date_end
  into cr_rate;

  bg_id_old = '10001';
  for select
             case bg_id
               when '10001' then 1
               when '10150' then 2
               when '10101' then 3
             end num,
             bg_id, line_name, sum(saldo_hq_417), sum(saldo_hq_840), sum(saldo_hq_978), sum(saldo_hq_other),
             sum(saldo_bru), sum(saldo_isk), sum(saldo_osh), sum(saldo_jal), sum(saldo_tash), sum(saldo_nar),
             sum(saldo_tal), sum(saldo_kad), sum(saldo_bat), sum(saldo_uzg), sum(saldo_chuy), sum(saldo_bish),
             sum(saldo_other), sum(saldo_region)
      from aarep_account_bank_saldo_tmp(:report_region_id, :report_date_end)
      where bg_id = iif(:report_product_kind = 'ИПФ', '10101', bg_id)
      group by 1, bg_id, line_name
      order by 1
      into num, bg_id, line_name, sum_saldo_hq_417, sum_saldo_hq_840, sum_saldo_hq_978, sum_saldo_hq_other,
           sum_saldo_bru, sum_saldo_isk, sum_saldo_osh, sum_saldo_jal, sum_saldo_tash, sum_saldo_nar, sum_saldo_tal,
           sum_saldo_kad, sum_saldo_bat, sum_saldo_uzg, sum_saldo_chuy, sum_saldo_bish, sum_saldo_other,
           sum_saldo_region
  do
  begin
    if (bg_id <> bg_id_old) then
    begin
      line_name_tmp = line_name;
      line_name = 'Всего ' || case bg_id_old
                                     when '10001' then 'наличные в кассах'
                                     when '10150' then 'по традиционным'
                                     when '10101' then 'по ИПФ'
                                   end;

      saldo_hq_417 = saldo_hq_417_tmp;
      saldo_hq_840 = saldo_hq_840_tmp;
      saldo_hq_978 = saldo_hq_978_tmp;
      saldo_hq_other = saldo_hq_other_tmp;
      saldo_bru = saldo_bru_tmp;
      saldo_isk = saldo_isk_tmp;
      saldo_osh = saldo_osh_tmp;
      saldo_jal = saldo_jal_tmp;
      saldo_tash = saldo_tash_tmp;
      saldo_nar = saldo_nar_tmp;
      saldo_tal = saldo_tal_tmp;
      saldo_kad = saldo_kad_tmp;
      saldo_bat = saldo_bat_tmp;
      saldo_uzg = saldo_uzg_tmp;
      saldo_chuy = saldo_chuy_tmp;
      saldo_bish = saldo_bish_tmp;
      saldo_other = saldo_other_tmp;
      saldo_region = saldo_region_tmp;
      suspend;

      bg_id_old = bg_id;
      line_name = line_name_tmp;

      saldo_hq_417_tmp = 0;
      saldo_hq_840_tmp = 0;
      saldo_hq_978_tmp = 0;
      saldo_hq_other_tmp = 0;
      saldo_bru_tmp = 0;
      saldo_isk_tmp = 0;
      saldo_osh_tmp = 0;
      saldo_jal_tmp = 0;
      saldo_tash_tmp = 0;
      saldo_nar_tmp = 0;
      saldo_tal_tmp = 0;
      saldo_kad_tmp = 0;
      saldo_bat_tmp = 0;
      saldo_uzg_tmp = 0;
      saldo_chuy_tmp = 0;
      saldo_bish_tmp = 0;
      saldo_other_tmp = 0;
      saldo_region = 0;
    end

    saldo_hq_417 = sum_saldo_hq_417;
    saldo_hq_840 = sum_saldo_hq_840;
    saldo_hq_978 = sum_saldo_hq_978;
    saldo_hq_other = sum_saldo_hq_other;
    saldo_bru = sum_saldo_bru;
    saldo_isk = sum_saldo_isk;
    saldo_osh = sum_saldo_osh;
    saldo_jal = sum_saldo_jal;
    saldo_tash = sum_saldo_tash;
    saldo_nar = sum_saldo_nar;
    saldo_tal = sum_saldo_tal;
    saldo_kad = sum_saldo_kad;
    saldo_bat = sum_saldo_bat;
    saldo_uzg = sum_saldo_uzg;
    saldo_chuy = sum_saldo_chuy;
    saldo_bish = sum_saldo_bish;
    saldo_other = sum_saldo_other;
    saldo_region = sum_saldo_region;

    if (line_name = '') then
      line_name = 'Банк не указан';

    suspend;

    saldo_hq_417_tmp = coalesce(saldo_hq_417_tmp, 0) + sum_saldo_hq_417;
    saldo_hq_840_tmp = coalesce(saldo_hq_840_tmp, 0) + sum_saldo_hq_840;
    saldo_hq_978_tmp = coalesce(saldo_hq_978_tmp, 0) + sum_saldo_hq_978;
    saldo_hq_other_tmp = coalesce(saldo_hq_other_tmp, 0) + sum_saldo_hq_other;
    saldo_bru_tmp = coalesce(saldo_bru_tmp, 0) + sum_saldo_bru;
    saldo_isk_tmp = coalesce(saldo_isk_tmp, 0) + sum_saldo_isk;
    saldo_osh_tmp = coalesce(saldo_osh_tmp, 0) + sum_saldo_osh;
    saldo_jal_tmp = coalesce(saldo_jal_tmp, 0) + sum_saldo_jal;
    saldo_tash_tmp = coalesce(saldo_tash_tmp, 0) + sum_saldo_tash;
    saldo_nar_tmp = coalesce(saldo_nar_tmp, 0) + sum_saldo_nar;
    saldo_tal_tmp = coalesce(saldo_tal_tmp, 0) + sum_saldo_tal;
    saldo_kad_tmp = coalesce(saldo_kad_tmp, 0) + sum_saldo_kad;
    saldo_bat_tmp = coalesce(saldo_bat_tmp, 0) + sum_saldo_bat;
    saldo_uzg_tmp = coalesce(saldo_uzg_tmp, 0) + sum_saldo_uzg;
    saldo_chuy_tmp = coalesce(saldo_chuy_tmp, 0) + sum_saldo_chuy;
    saldo_bish_tmp = coalesce(saldo_bish_tmp, 0) + sum_saldo_bish;
    saldo_other_tmp = coalesce(saldo_other_tmp, 0) + sum_saldo_other;
    saldo_region_tmp = coalesce(saldo_region_tmp, 0) + sum_saldo_chuy;

  end
  saldo_hq_417 = saldo_hq_417_tmp;
  saldo_hq_840 = saldo_hq_840_tmp;
  saldo_hq_978 = saldo_hq_978_tmp;
  saldo_hq_other = saldo_hq_other_tmp;
  saldo_bru = saldo_bru_tmp;
  saldo_isk = saldo_isk_tmp;
  saldo_osh = saldo_osh_tmp;
  saldo_jal = saldo_jal_tmp;
  saldo_tash = saldo_tash_tmp;
  saldo_nar = saldo_nar_tmp;
  saldo_tal = saldo_tal_tmp;
  saldo_kad = saldo_kad_tmp;
  saldo_bat = saldo_bat_tmp;
  saldo_uzg = saldo_uzg_tmp;
  saldo_chuy = saldo_chuy_tmp;
  saldo_bish = saldo_bish_tmp;
  saldo_other = saldo_other_tmp;
  saldo_region = saldo_region_tmp;

  line_name = 'Всего ' || case bg_id_old
                                 when '10001' then 'наличные в кассах'
                                 when '10150' then 'по традиционным'
                                 when '10101' then 'по ИПФ'
                               end;
  suspend;
end
