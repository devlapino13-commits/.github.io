CREATE PROCEDURE FIND_ML_BY_BARCODE(
    BARCODE_STR VARCHAR(255),
    DD DATE)
RETURNS (
    ML_ID DECIMAL(18,0),
    ML_NAME VARCHAR(255),
    ML_YEDIZ VARCHAR(20),
    ML_MATCATEG_ID DECIMAL(18,0),
    ML_MATCATEG_NAME VARCHAR(255),
    ML_IS_ONE_KOLICH INTEGER,
    ML_MODEL VARCHAR(255))
AS
declare variable ddd date;
declare variable sym varchar(20);
begin
  /* Procedure Text */
  for select distinct mf_symbol
      from matfields
      where mf_id = 1
      into :sym
  do
  begin
    for select distinct ob_from_ml_id, ob_from_mc_id, ob_date_begin
        from operbill
        where ob_full_detail like '%' || '`' || :sym || '.:' || :barcode_str || '%' and
              ob_date_begin <= :dd
        order by 3 desc
        rows 1
        into :ml_id, :ml_matcateg_id, :ddd
    do
    begin
      for select yl_code, ml_name, mc_name, mc_is_one_kolich, ml_model
          from matlist, matcateg, yedizlist
          where ml_id = :ml_id and
                mc_id = :ml_matcateg_id and
                yl_id = ml_from_yl_id
          into :ml_yediz, :ml_name, :ml_matcateg_name, :ml_is_one_kolich, :ml_model
      do
      begin
        suspend;
      end
    end
  end
end
