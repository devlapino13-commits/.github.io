CREATE PROCEDURE HP_CASHFLOWDATA_PREPARE(
    REP_DATE D_DATE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable del_count d_integer;
declare variable ins_count d_integer;
declare variable zero_count d_integer;
begin
  res_is_error = 0;
  ins_count = 0;
  del_count = 0;

  select count(*)
  from kn_operbook
  where kob_cashflow_date = :rep_date and
        kob_ol_is_active = 1 and
        coalesce(kob_from_rml_id, 0) = 0 and
        kob_summa <> 0 and
        (kob_acc_dt + 0 <> 0 and kob_acc_kt + 0 <> 0)
  into zero_count;

  if (zero_count > 0) then
  begin
    res_is_error = 1;
    res_text = 'Имеются проводки без кода ДДС';
    suspend;
    exit;
  end

  delete from cashflowdata
  where cfd_date = :rep_date;

  del_count = row_count;

  insert into cashflowdata (cfd_date, cfd_doc_name, cfd_bank_id, cfd_is_bank, cfd_oper_sign, cfd_dds_code,
                            cfd_bal_code_dt, cfd_bal_code_kt, cfd_add_code_dt, cfd_add_code_kt, cfd_acc_code_dt, cfd_acc_code_kt, 
                            cfd_summa, cfd_currency, cfd_dds_is_resident, cfd_client_residency,
                            cfd_client_country, cfd_client_type, cfd_client_name, cfd_client_id, cfd_client_date, cfd_dds_is_law,
                            cfd_from_kc_id, cfd_kontract_no, cfd_acc_code, cfd_doc_no_str, cfd_ol_memo, cfd_kob_id,
                            cfd_ol_id, cfd_ol_date)
  select kob_ol_date, res_doc_name, res_bank_id, res_is_bank, res_oper_sign, res_dds_code, 
         kob_bal_code_dt, kob_bal_code_kt, kob_add_code_dt, kob_add_code_kt, kob_acc_code_dt, kob_acc_code_kt, 
         kob_summa, kob_currency, res_dds_is_resident, res_client_resident,
         res_client_country, res_client_type, res_client_name, res_client_id, res_client_date, res_dds_is_law,
         kob_from_kc_id, res_kontract_no, res_acc_code, ol_doc_no_str, ol_memo, kob_id,
         kob_from_ol_id, res_ol_date
  from aarep_moneyoper_eps_list(:rep_date, :rep_date, 0, null);

  ins_count = row_count;

  update or insert into cashflowlist (cfl_date, cfl_data_count)
  values (:rep_date, :ins_count)
  matching (cfl_date);

  res_text = 'Добавлены: ' || ins_count || ', удалены старые: ' || del_count;

  select count(*)
  from kn_operbook
  where kob_cashflow_date = :rep_date and
        kob_ol_is_active = 1 and
        kob_from_rml_id > 0
  into ins_count;

  update or insert into cashflowlist (cfl_date, cfl_book_count)
  values (:rep_date, :ins_count)
  matching (cfl_date);

  select count(*)
  from kn_operbook
  where kob_cashflow_date = :rep_date and
        kob_ol_is_active = 1 and
        kob_from_rml_id = 0
  into ins_count;

  update or insert into cashflowlist (cfl_date, cfl_book_zero)
  values (:rep_date, :ins_count)
  matching (cfl_date);

  suspend;
end
