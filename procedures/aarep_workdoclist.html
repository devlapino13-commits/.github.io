CREATE PROCEDURE AAREP_WORKDOCLIST(
    UN_ID DECIMAL(18,0),
    FIND_TXT VARCHAR(255),
    INCLUDE_ANNUL INTEGER)
RETURNS (
    WDL_ID INTEGER,
    WDL_FROM_WDT_ID INTEGER,
    WDL_NAME VARCHAR(255),
    WDL_MEMO VARCHAR(1024),
    WDL_DATE_APPROVE DATE,
    WDL_DATE_START DATE,
    WDL_FROM_UN_ID DECIMAL(18,0),
    WDL_DATETIME_SAVE TIMESTAMP,
    WDL_IS_OPEN_ACCESS INTEGER,
    WDL_READY INTEGER,
    WDL_READY_USER DECIMAL(18,0),
    WDL_READY_DATETIME TIMESTAMP,
    WDL_LOCK INTEGER,
    WDL_LOCK_USER DECIMAL(18,0),
    WDL_LOCK_DATETIME TIMESTAMP,
    WDL_STOP_DATE DATE,
    WDL_STOP_USER DECIMAL(18,0),
    WDL_STOP_DATETIME TIMESTAMP,
    WDL_STOP_WDL_ID INTEGER,
    WDL_HAS_NEW INTEGER,
    WDL_WORD_COUNT INTEGER,
    WDL_COUNT INTEGER)
AS
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();
  for select distinct w1.wdl_id, sum(w2.wdl_word_count), sum(w2.wdk_count), max(w2.wdl_has_new), w1.wdl_from_wdt_id,
                      w1.wdl_name, w1.wdl_memo, w1.wdl_date_approve, w1.wdl_date_start, w1.wdl_from_un_id,
                      w1.wdl_datetime_save, w1.wdl_is_open_access, w1.wdl_ready, w1.wdl_ready_user,
                      w1.wdl_ready_datetime, w1.wdl_lock, w1.wdl_lock_user, w1.wdl_lock_datetime, w1.wdl_stop_date,
                      w1.wdl_stop_user, w1.wdl_stop_datetime, w1.wdl_stop_wdl_id
      from workdoclist w1, aarep_workdoclist_tmp(:un_id, :find_txt, :include_annul) w2
      where w1.wdl_id = w2.wdl_id
      group by w1.wdl_id, w1.wdl_from_wdt_id, w1.wdl_name, w1.wdl_memo, w1.wdl_date_approve, w1.wdl_date_start, w1.wdl_from_un_id, w1.wdl_datetime_save, w1.wdl_is_open_access, w1.wdl_ready, w1.wdl_ready_user, w1.wdl_ready_datetime, w1.wdl_lock, w1.wdl_lock_user, w1.wdl_lock_datetime, w1.wdl_stop_date, w1.wdl_stop_user, w1.wdl_stop_datetime, w1.wdl_stop_wdl_id
      order by 2 desc, 3 desc, 4 desc, 8 desc
      into :wdl_id, :wdl_word_count, :wdl_count, :wdl_has_new, :wdl_from_wdt_id, :wdl_name, :wdl_memo,
           :wdl_date_approve, :wdl_date_start, :wdl_from_un_id, :wdl_datetime_save, :wdl_is_open_access, :wdl_ready,
           :wdl_ready_user, :wdl_ready_datetime, :wdl_lock, :wdl_lock_user, :wdl_lock_datetime, :wdl_stop_date,
           :wdl_stop_user, :wdl_stop_datetime, :wdl_stop_wdl_id
  do
