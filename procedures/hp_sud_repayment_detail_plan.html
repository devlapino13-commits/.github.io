CREATE PROCEDURE HP_SUD_REPAYMENT_DETAIL_PLAN(
    SMR_ID D_ID)
RETURNS (
    OKS_DATE D_DATE,
    OKS_ORIG_OS D_CURRENCY,
    OKS_ORIG_PERC D_CURRENCY,
    OKS_ORIG_ALL D_CURRENCY,
    OKS_MOVED_OS D_CURRENCY,
    OKS_SH_OS D_CURRENCY,
    OKS_SH_PERC D_CURRENCY,
    OKS_SH_ALL D_CURRENCY,
    OKS_ZAD_OS D_CURRENCY,
    OKS_ZAD_PERC D_CURRENCY,
    OKS_INFO D_STR_CODE)
AS
declare variable k_id d_id;
begin
  select smr_from_k_id
  from sud_mainreq
  where smr_id = :smr_id
  into k_id;

  for select oks_date, oks_orig_os, oks_orig_perc, oks_orig_all, oks_moved_os, oks_sh_os, oks_sh_perc, oks_sh_all, oks_zad_os,
         oks_zad_perc, oks_info
  from hp_repayment_detail_plan(:k_id, current_date)
  into :oks_date, :oks_orig_os, :oks_orig_perc, :oks_orig_all, :oks_moved_os, :oks_sh_os, :oks_sh_perc, :oks_sh_all,
       :oks_zad_os, :oks_zad_perc, :oks_info
  do
    suspend;
end
