CREATE PROCEDURE HP_USER_REGION(
    USER_ROLE D_STR_100,
    UN_ID D_ID,
    REG_ID_INP D_INTEGER)
RETURNS (
    REG_ID D_INTEGER,
    REG_NAME D_STR_100)
AS
declare variable reg_id_tmp d_integer;
declare variable c_id_list d_str_long;
declare variable sql d_str_2048;
begin
  if (:user_role in ('accountant', 'reg_manager') or un_id = 1090100002454) then
  begin
    select reg.reg_unicode, reg.reg_code || ' ' || reg.reg_name
    from region reg
    where reg.reg_unicode = 1000000
    into :reg_id, :reg_name;

    suspend;

    for select reg.reg_unicode, reg.reg_code || ' ' || reg.reg_name
        from region_list(1000000, 1, 3, 1) rl
        inner join region reg on rl.id_list = reg.reg_unicode
        into :reg_id, :reg_name
    do
      suspend;
    exit;
  end
  if (:user_role = 'kred_manager') then
  begin
    select list(c_id, ',')
    from hp_user_region_tmp(:un_id)
    into :c_id_list;
    sql = f_format('select pm.pm_subject_id as reg_id
    from permissions pm
    where pm.pm_object_kind = 0 and
    pm.pm_object_id = %1 and
    pm.pm_access_kind = 1 and
    pm.pm_is_delete = 0 and
    current_date between pm.pm_date_begin and coalesce(pm.pm_date_end, ''31.12.2999'') and
    pm.pm_subject_kind = 4
    union
    select ps.ps_from_reg_id as reg_id
    from hr_pstatus ps
    where current_date between ps.ps_date_begin and ps.ps_date_end and
    ps.ps_is_working = 1 and
    ps.ps_status = 1 and
    ps.ps_from_c_id in (%2)', :un_id, :c_id_list);
    for execute statement(:sql)
            into :reg_id
    do
    begin
      select reg.reg_code || ' ' || reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_name;
      suspend;
    end
    exit;
  end
  if (:user_role in ('admin_assistant', 'admin_assistant_chief')) then
  begin
    for select pm.pm_subject_id
        from permissions pm
        where pm.pm_object_kind = 0 and
              pm.pm_object_id = :un_id and
              pm.pm_subject_kind = 4 and
              pm.pm_is_delete = 0 and
              pm.pm_access_kind = 1 and
              current_date between coalesce(pm.pm_date_begin, current_date) and coalesce(pm.pm_date_end, '31.12.2999')
        into :reg_id_tmp
    do
    begin
      if (:reg_id_tmp = 1000000) then
      begin
        select reg.reg_unicode, reg.reg_code || ' ' || reg.reg_name
        from region reg
        where reg.reg_unicode = 1000000
        into :reg_id, :reg_name;
        suspend;
      end

      for select reg.reg_unicode, reg.reg_code || ' ' || reg.reg_name
          from region_list(:reg_id_tmp, 1, 3, 1) rl
          inner join region reg on rl.id_list = reg.reg_unicode
          into :reg_id, :reg_name
      do
        suspend;
    end
    exit;
  end
end
