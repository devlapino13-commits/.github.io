CREATE PROCEDURE AAREP_PSK_DISB(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    NUMBER_ITEM VARCHAR(20),
    CATEGORY_LOANS VARCHAR(255),
    KC_AMOUNT_APPROVED CURR_TYPE,
    PERCENT_EFFECT CURR_TYPE,
    KOL_KONTRACT INTEGER)
AS
declare variable item_1 integer;
declare variable item_2 integer;
declare variable item_3 integer;
declare variable sum_percent_effect decimal(18,4);
begin
  for select item_1, item_2, item_3, sum(kc_amount_approved),
             sum(cast(kc_amount_approved as double precision) * k_percent_effect), sum(is_kontract)
      from aarep_psk_disb_temp(:rep_date_begin, :rep_date_end)
      group by item_1, item_2, item_3
      into item_1, item_2, item_3, kc_amount_approved, sum_percent_effect, kol_kontract
  do
  begin
    percent_effect = cast(sum_percent_effect as double precision) / kc_amount_approved;

    number_item = item_1 || iif(item_2 = 0, '', '.' || item_2) || iif(item_3 = 0, '', '.' || item_3);
    category_loans = case number_item
                       when '1' then 'Потребительские микрозаймы с обеспечением в виде залога'
                       when '2' then 'Потребительские микрозаймы с иным обеспечением'
                       when '2.1' then 'до 365 дней включительно' --'до года'
                       when '2.2' then 'свыше 365 дней' --'cвыше года'
                       when '3' then 'Потребительские микрозаймы без обеспечения (кроме POS-микрозаймов)' --'Потребительские микрозаймы без обеспечения'
                       when '3.1' then 'до 30 дней включительно, в том числе:' --'до 1 месяца, в том числе:'
                       when '3.1.1' then 'до 30 тыс. руб. включительно' --'до 30 тыс. руб.'
                       when '3.1.2' then 'свыше 30 тыс. руб.'
                       when '3.2' then 'от 31 до 60 дней включительно, в том числе:' --'от 1 до 2 месяцев, в том числе:'
                       when '3.2.1' then 'до 30 тыс. руб. включительно' --'до 30 тыс. руб.'
                       when '3.2.2' then 'свыше 30 тыс. руб.'
                       when '3.3' then 'от 61 до 180 дней включительно, в том числе:' --'от 2 до 6 месяцев, в том числе:'
                       when '3.3.1' then 'до 30 тыс. руб. включительно' --'до 30 тыс. руб.'
                       when '3.3.2' then 'свыше 30 тыс. руб. до 100 тыс. руб. включительно' --'от 30 тыс.руб до 100 тыс. руб.'
                       when '3.3.3' then 'свыше 100 тыс. руб.'
                       when '3.4' then 'от 181 дня до 365 дней включительно, в том числе:' --'от 6 месяцев до 1 года, в том числе:'
                       when '3.4.1' then 'до 30 тыс. руб. включительно' --'до 30 тыс. руб.'
                       when '3.4.2' then 'свыше 30 тыс. руб. до 100 тыс. руб. включительно' --'от 30 тыс. руб. до 100 тыс. руб.'
--                            when '3.4.3' then 'от 60 тыс. руб. до 100 тыс. руб.'
                       when '3.4.3' then 'свыше 100 тыс. руб.'
                       when '3.5' then 'свыше 365 дней, в том числе:' --'свыше 1 года, в том числе:'
                       when '3.5.1' then 'до 30 тыс. руб. включительно' --'до 30 тыс. руб.'
                       when '3.5.2' then 'свыше 30 тыс. руб. до 60 тыс. руб. включительно' --'от 30 тыс. руб. до 60 тыс. руб.'
                       when '3.5.3' then 'свыше 60 тыс. руб. до 100 тыс. руб. включительно' --'от 60 тыс. руб. до 100 тыс. руб.'
                       when '3.5.4' then 'свыше 100 тыс. руб.'

                       when '4' then 'POS-микрозаймы'
                       when '4.1' then 'до 365 дней включительно, в том числе:'
                       when '4.1.1' then 'до 30 тыс. руб. включительно'
                       when '4.1.2' then 'свыше 30 тыс. руб. до 100 тыс. руб. включительно'
                       when '4.1.3' then 'свыше 100 тыс. руб.'
                       when '4.2' then 'свыше 365 дней'
                     end;
    number_item = '5.' || number_item;
    suspend;
  end
end
