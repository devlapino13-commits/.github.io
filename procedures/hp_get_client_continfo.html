CREATE PROCEDURE HP_GET_CLIENT_CONTINFO(
    C_ID D_ID,
    CONT_KIND D_STR_CODE)
RETURNS (
    RES_VALUE D_STR_255,
    RES_STATUS_ID D_INTEGER,
    RES_DATE D_DATE,
    RES_DATETIME D_DATETIME,
    RES_UN_ID D_ID)
AS
declare variable rd_id d_integer;
begin
  -- Альтернативный вариант вызова HP_OBJECTPROPS_GET

  if (cont_kind similar to '[[:DIGIT:]]+') then
  begin
    select rd_id, rd_code
    from refdata
    where rd_id = :cont_kind
    into rd_id, cont_kind;
  end else
  begin
    if (cont_kind not starting 'ContactInfo.') then
      cont_kind = 'ContactInfo.' || cont_kind;

    select rd_id
    from refdata
    where rd_code = :cont_kind
    into rd_id;
  end

  if (rd_id is not null) then
  begin
    select ovs_value, ovs_add_value, ovs_datetime, ovs_date_begin, ovs_from_un_id
    from objectvalues, objectprops
    where opr_object_kind = f_objkind('clients', 'id') and
          opr_object_id = :c_id and
          opr_subject_kind = f_objkind('refdata', 'id') and
          opr_subject_id = :rd_id and
          ovs_from_opr_id = opr_id and
          ovs_id = opr_actual_ovs_id
    into res_value, res_status_id, res_datetime, res_date, res_un_id;
  end

  suspend;
end
