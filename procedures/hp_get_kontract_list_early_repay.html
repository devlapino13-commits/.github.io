CREATE PROCEDURE HP_GET_KONTRACT_LIST_EARLY_REPAY(
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    PR_CODE_PART D_STR_CODE)
RETURNS (
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    KC_AMOUNT_APPROVED D_CURRENCY,
    LOAN_TARGET D_STR_255,
    PR_ID D_INTEGER,
    PR_CODE D_STR_100,
    PR_NAME D_STR_CODE,
    OLD_KS_STATE D_INTEGER,
    OLD_ZAD_OS_TOTAL D_CURRENCY,
    OLD_ZAD_PERC_TOTAL D_CURRENCY,
    EARLY_PERC D_CURRENCY,
    PERC_GET D_CURRENCY,
    PERC_FUTURE D_CURRENCY,
    PAY_OS_2 D_CURRENCY,
    PAY_OS_3 D_CURRENCY,
    OVER_OS D_CURRENCY,
    NSP_PAY_2 D_CURRENCY,
    NSP_PAY_3 D_CURRENCY,
    NSP_OVER D_CURRENCY,
    NSP_TOTAL D_CURRENCY,
    DELTA D_CURRENCY)
AS
declare variable kc_loan_target d_integer;
declare variable rd_id d_integer;
declare variable share d_currency;
begin
  -- скрипт возвращает список контрактов с частично-досрочными погашениями (ЧДП), RFCKG-5180

  -- условия выборки контрактов:
  -- 1) ИПФ-продукт
  -- 2) Цель финансирования - Потребительский
  -- 3) Контракты выданы в заданном периоде
  -- 4) У контрактов есть ЧДП в заданном периоде
  -- 5) ЧДП учитываются только по процентам

  select rd_id,
         rd_name
  from refdata
  where rd_main_code = 'Loan_Target' and -- цель финансирования
        rd_code = 'Loan_Target.Consumer' -- потребительский
  into rd_id,
       loan_target;

  for select k_id,
             k_kontract_no,
             k_fact_disb_date,
             kc_amount_approved,
             kc_loan_target,
             pr_name,
             pr_code,
             pr_id
      from kontract
      inner join product on pr_id = k_from_pr_id
      inner join knclient on kc_from_k_id = k_id
      where pr_code containing '.ipf.' and -- ИПФ-продукт
            pr_code containing :pr_code_part and -- конкретный ИПФ-продукт (murabaha, istisnaa и т.д.)
            kc_amount_approved > 0 and -- клиент
            kc_loan_target = :rd_id and -- потребительский
            k_fact_disb_date between :date_begin and :date_end -- выданные за период
      order by k_fact_disb_date
      into k_id,
           k_kontract_no,
           k_fact_disb_date,
           kc_amount_approved,
           kc_loan_target,
           pr_name,
           pr_code,
           pr_id
  do
  begin
    old_ks_state = 0;
    old_zad_os_total = 0;
    old_zad_perc_total = 0;
    perc_get = 0;
    perc_future = 0;
    pay_os_2 = 0;
    pay_os_3 = 0;
    over_os = 0;
    early_perc = 0;

    select early_perc
    from hp_get_kontract_early_repay(:k_id, :date_begin, :date_end)
    into early_perc;

    if (early_perc <= 0) then
      continue; -- ЧДП учитываются только по процентам

    share = early_perc; -- сумма распределения

    select ks_state
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          current_date between ks_date_begin and ks_date_end
    into old_ks_state;

    select sum(zad_os),
           sum(zad_perc)
    from aarep_make_zad_list(:k_id, current_date, 1)
    into old_zad_os_total,
         old_zad_perc_total;

    select sum(sum_of_ok_perc),
           sum(calc_perc_end)
    from kontract_full_oper_perc(:k_id, current_date)
    into perc_get, -- наценка к получению
         perc_future; -- отсроченные доходы по наценке

    if (perc_get is null) then
      perc_get = 0;

    if (perc_future is null) then
      perc_future = 0;

    if (perc_get > early_perc) then -- если наценка к получению больше суммы уступки, то
      perc_get = early_perc; -- наценка к получению будет уменьшена только на сумму уступки

    -- *** (2) -- расчет суммы погашения ОС со счета "Отсроченные доходы по наценке" за вычетом наценки к получению (pay_os_2)
    pay_os_2 = perc_future - perc_get;

    share = share - pay_os_2; -- остаток суммы уступки после pay_os_2

    -- *** (3) -- расчет суммы погашения ОС со счета "Доходы по операциям Мурабаха" (pay_os_3)
    pay_os_3 = old_zad_os_total - pay_os_2;

    if (pay_os_3 < 0) then
    begin
      over_os = -pay_os_3; -- отрицательную сумму погашения переводим в переплату со сменой знака
      pay_os_3 = 0;
    end

    if (pay_os_3 > share) then -- если погашение больше суммы уступки, то
      pay_os_3 = share; -- погашение будет только на сумму уступки

    share = share - pay_os_3; -- остаток суммы уступки после pay_os_3

    -- *** (4) -- расчет переплаты по ОС (over_os)
    over_os = over_os + early_perc; -- остаток суммы уступки после pay_os_2 и pay_os_3 идет на переплату

    if (over_os < 0) then
      over_os = 0; -- переплата не может быть отрицательным

    nsp_pay_2 = 0;
    nsp_pay_3 = 0;
    nsp_over = 0;

    if (pay_os_2 > 0) then
      nsp_pay_2 = round(pay_os_2 * 2 / 100, 0);

    if (pay_os_3 > 0) then
      nsp_pay_3 = round(pay_os_3 * 2 / 100, 0);

    if (over_os > 0) then
      nsp_over = round(over_os * 2 / 100, 0);

    nsp_total = nsp_pay_2 + nsp_pay_3 + nsp_over;

    delta = old_zad_os_total - early_perc; -- разница между задолженностью ОС и суммой уступки


    suspend;
  end
end
