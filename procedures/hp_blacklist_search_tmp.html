CREATE PROCEDURE HP_BLACKLIST_SEARCH_TMP(
    FIND_C_ID DECIMAL(18,0),
    FIND_FIO VARCHAR(100),
    FIND_BIRTH_DATE DATE,
    FIND_CODE VARCHAR(20),
    FIND_PASS_DATE DATE,
    FIND_PASS_NO VARCHAR(100))
RETURNS (
    M_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    IS_CRITICAL INTEGER,
    C_SEX INTEGER,
    RES_SAME_FIO INTEGER,
    RES_SAME_BIRTH_DATE INTEGER,
    RES_SAME_PASS_NO INTEGER,
    RES_SAME_PASS_DATE INTEGER,
    RES_SAME_CONTACT INTEGER,
    RES_SAME_CODE INTEGER)
AS
declare variable tmp_c_name_fio varchar(100);
declare variable tmp_birth_date date;
declare variable tmp_pass_no varchar(100);
declare variable tmp_pass_date date;
declare variable is_fio_ident integer;
declare variable txt1 varchar(100);
declare variable txt2 varchar(100);
declare variable txt3 varchar(100);
declare variable short_fio varchar(100);
declare variable tmp_short_fio varchar(100);
declare variable find_cell_no varchar(100);
declare variable find_orig_fio varchar(100);
declare variable critical_found integer;
declare variable contact_value varchar(100);
declare variable orig_txt1 varchar(100);
declare variable orig_txt2 varchar(100);
declare variable orig_txt3 varchar(100);
declare variable tmp_code varchar(20);
declare variable cda_cont_kind d_integer;
declare variable cda_from_clt_id d_integer;
declare variable is_new_contdata d_smallint;
begin
  find_cell_no = '';
  c_id = 0;
  is_critical = 0;
  if (find_c_id is null) then
    find_c_id = 0;
  if (find_pass_no is null) then
    find_pass_no = '';
  if (find_fio is null) then
    find_fio = '';
  if (find_code is null) then
    find_code = '';
  if (find_cell_no is null) then
    find_cell_no = '';

  find_pass_no = upper(f_replace_all(find_pass_no, ' ', ''));
  find_cell_no = upper(f_replace_all(find_cell_no, ' ', ''));
  find_fio = upper(f_replace_all(find_fio, '  ', ' '));
  find_fio = trim(find_fio);
  find_orig_fio = find_fio;
  short_fio = '';

  res_same_birth_date = 0;
  res_same_pass_no = 0;
  res_same_pass_date = 0;
  res_same_contact = 0;
  res_same_code = 0;
  res_same_fio = -1;

  is_new_contdata = f_get_const('new_contdata');
  is_new_contdata = 1; -- процедуру переводим на новый механизм, Sanjari, 21.12.2019

  -- если кроме ФИО ничего не указано, производим поиск только по ФИО
  orig_txt3 = '';
  if (find_fio <> '') then
  begin
    select full_text, txt1, txt2, txt3
    from hp_fio_clear(:find_fio)
    into :find_fio, :orig_txt1, :orig_txt2, :orig_txt3;

    if (orig_txt1 <> '' and
        orig_txt2 <> '') then
      short_fio = orig_txt1 || ' ' || orig_txt2;

    -- Список похожих по ФИО
    for select distinct object_id, m_from_c_id, m_client_fio, m_client_pass_no, m_client_birth_date
        from hp_find_from_namelist(0, 14, :find_fio), moratory
        where object_id = m_id and
              m_is_active = 1
              --    and m_id=1090100000000545
        into :m_id, :c_id, tmp_c_name_fio, tmp_pass_no, tmp_birth_date
    do
    begin
      tmp_code = '';
      tmp_pass_date = null;
      c_sex = 0;
      if (c_id is not null) then
      begin
        select c_name_fio, c_birth_date, c_index_c_pass_no, c_pass_date, c_inn, c_sex
        from clients
        where c_id = :c_id
        into tmp_c_name_fio, tmp_birth_date, tmp_pass_no, tmp_pass_date, tmp_code, c_sex;
      end

      critical_found = 0;
      is_critical = 0;
      is_fio_ident = 0;

      res_same_fio = 0;
      res_same_birth_date = 0;
      res_same_pass_no = 0;
      res_same_pass_date = 0;
      res_same_contact = 0;
      res_same_code = 0;

      if (find_orig_fio = tmp_c_name_fio) then
      begin
        is_fio_ident = 1;
        res_same_fio = 2;
      end
      else
      begin
        select full_text, txt1, txt2, txt3
        from hp_fio_clear(:tmp_c_name_fio)
        into :tmp_c_name_fio, txt1, txt2, txt3;
        tmp_short_fio = txt1 || ' ' || txt2;

        if (f_levenstain_is_same(:find_orig_fio, :tmp_c_name_fio) = 1) then
          res_same_fio = 1;

        if (find_fio <> short_fio) then -- Если текст поиска с отчеством
        begin
          -- Если результат поиска тоже имеет отчество
          if (tmp_c_name_fio <> tmp_short_fio) then
          begin
            -- поверяет два слова слова
            if (short_fio = tmp_short_fio) then
              res_same_fio = 1;
            if (short_fio = tmp_short_fio and
                txt3 starting orig_txt3) then
              res_same_fio = 1;
            -- поверяет все три слова
            if (find_fio = tmp_c_name_fio) then
              res_same_fio = 2;

            if (orig_txt3 <> '' and
                txt3 <> '') then
            begin
              if (f_levenstain_is_same(:orig_txt3, :txt3) = 0) then
                res_same_fio = -1;
            end
          end
          else
          begin
            -- проверяем только фамилию и имя
            if (short_fio = tmp_short_fio) then
              res_same_fio = 1;
          end
        end
        else -- если текст поиска без отчества
        begin
          if (tmp_c_name_fio <> tmp_short_fio) then
          begin
            -- поверяет все три слова
            if (find_fio = tmp_short_fio) then
              res_same_fio = 1;
          end
          else
          begin
            -- проверяем только фамилию и имя
            if (short_fio = tmp_short_fio) then
              res_same_fio = 2;
          end
        end
      end
      if (res_same_fio <> -1) then
        is_fio_ident = 1; --  TODO: Тут где-то ошибка, res_same_fio = 0 это не совпадение

      -- Если есть другие параметры, ищем по ним
      if (find_birth_date is not null or
          find_pass_date is not null or
          find_pass_no <> '' or
          find_code <> '') then
      begin
        if (tmp_code = find_code and
            find_code <> '') then
        begin
          is_critical = 1;
          critical_found = 1;
          res_same_pass_no = 1;
        end
        if (tmp_pass_no = find_pass_no and
            find_pass_no <> '') then
        begin
          is_critical = 1;
          critical_found = 1;
          res_same_pass_no = 1;
        end
        if (tmp_birth_date = find_birth_date and
            find_birth_date is not null) then
        begin
          if (is_fio_ident = 1 or tmp_pass_date = find_pass_date) then
          begin
            is_critical = 1;
            critical_found = 1;
            res_same_birth_date = 1;
          end
          if (res_same_fio = -1) then
            res_same_fio = 0;
        end
        if (tmp_pass_date = find_pass_date and
            find_pass_date is not null) then
        begin
          if (is_fio_ident = 1 or
              (tmp_birth_date = find_birth_date and
              tmp_pass_date = find_pass_date)) then
          begin
            is_critical = 1;
            critical_found = 1;
          end
          res_same_pass_date = 1;
        end

        -- Поиск по старым документам
        if (critical_found = 0 and
            (find_pass_no <> '' or
            find_pass_date is not null) and
            is_fio_ident = 1 and
            c_id > 0) then
        begin
          for select cds_doc_no, cds_doc_date_begin
              from clientdocs, passkind, passdoctype
              where cds_from_c_id = :c_id and
                    cds_from_pk_id = pk_id and
                    pk_from_pdt_id = pdt_id and
                    pdt_is_main_doc = 1 and
                    (coalesce(cds_doc_no, '') <> '' or
                    cds_doc_date_begin is not null)
              into :tmp_pass_no, :tmp_pass_date
          do
          begin
            tmp_pass_no = replace(tmp_pass_no, ' ', '');
            if (tmp_pass_no = find_pass_no) then
            begin
              is_critical = 1;
              critical_found = 1;
              res_same_pass_no = 1;
            end
            if (tmp_pass_date = find_pass_date) then
            begin
              is_critical = 1;
              critical_found = 1;
              res_same_pass_date = 1;
            end
          end
        end -- Конец поиска по старым документам
      end -- Конец поиска других параметров

      -- Если критических совпадений не найдено, ищем совпадения по контактным данным
      if (critical_found = 0 and
          c_id <> 0 and
          find_c_id <> 0 and
          res_same_fio <> -1 and
          c_id > 0) then
      begin
        if (is_new_contdata = 0) then
        begin
          for select ovs_value
              from objectprops, objectvalues, refdata
              where opr_object_kind = 0 and
                    opr_object_id = :find_c_id and
                    opr_subject_kind = -1 and
                    opr_subject_id = rd_id and
                    opr_id = ovs_from_opr_id and
                    ovs_value <> '' and
                    rd_code in ('ContactInfo.Cell_Main', 'ContactInfo.Cell_Add', 'ContactInfo.Email', 'ContactInfo.Skype', 'ContactInfo.Whatsapp', 'ContactInfo.Viber', 'ContactInfo.Telegram', 'ContactInfo.Imo', 'ContactInfo.OK', 'ContactInfo.Facebook',
                                'ContactInfo.VK')
              into :contact_value
          do
          begin
            if (exists(select *
                       from objectprops, objectvalues, refdata
                       where opr_object_kind = 0 and
                             opr_object_id = :c_id and
                             opr_subject_kind = -1 and
                             opr_subject_id = rd_id and
                             rd_code in ('ContactInfo.Cell_Main', 'ContactInfo.Cell_Add', 'ContactInfo.Email', 'ContactInfo.Skype', 'ContactInfo.Whatsapp', 'ContactInfo.Viber', 'ContactInfo.Telegram', 'ContactInfo.Imo', 'ContactInfo.OK', 'ContactInfo.Facebook',
                                         'ContactInfo.VK') and
                             opr_id = ovs_from_opr_id and
                             hash(lower(ovs_value)) = hash(lower(:contact_value)) and
                             ovs_value = :contact_value and
                             :contact_value <> '')) then
            begin
              if (is_fio_ident = 1) then
              begin
                is_critical = 1;
                critical_found = 1;
              end
              res_same_contact = 1;
            end
          end
        end
        else
        begin
          -- поиск совпадения по контактным данным
          for select cda_cont_kind, cda_from_clt_id
              from contdata
              where cda_from_c_id = :find_c_id and
                    cda_from_clt_id <> 0 and
                    cda_is_delete = 0
              into cda_cont_kind, cda_from_clt_id
          do
          begin
            if (exists(select *
                       from contdata
                       where cda_from_c_id = :c_id and
                             cda_is_delete = 0 and
                             cda_cont_kind = :cda_cont_kind and
                             cda_from_clt_id = :cda_from_clt_id)) then
            begin
              if (is_fio_ident = 1) then
              begin
                is_critical = 1;
                critical_found = 1;
              end
              res_same_contact = 1;
            end
          end
        end

      end -- Конец поиска по контактным данным
      if (not(res_same_fio = -1 and
          res_same_pass_no = 0 and
          res_same_birth_date = 0 and
          res_same_pass_date = 0)) then
      begin
        suspend;
      end
    end -- Конец поиска по ФИО

    -- Если было заполнено только ФИО, завершаем поиск
    if (find_birth_date is null and
        find_pass_date is null and
        find_pass_no = '') then
      exit;
  end

  if (critical_found = 1) then
    exit;

  res_same_fio = -1;
  res_same_birth_date = 0;
  res_same_pass_no = 0;
  res_same_pass_date = 0;
  res_same_contact = 0;
  res_same_code = 0;

  -- Поиск по дате рождения и другие параметры
  if (find_birth_date is not null and
      (find_pass_date is not null or
      find_pass_no <> '')) then
  begin
    tmp_pass_date = null;
    for select m_id, m_from_c_id, m_client_fio, m_client_pass_no, c_inn, m_client_birth_date
        from moratory left join clients on m_from_c_id = c_id
        where (m_client_birth_date = :find_birth_date) and
              m_is_active = 1 and
              coalesce(m_client_pass_no, '') <> ''
        into m_id, c_id, tmp_c_name_fio, tmp_pass_no, tmp_code, tmp_birth_date
    do
    begin
      res_same_fio = -1;
      if (f_levenstain_is_same(:find_orig_fio, :tmp_c_name_fio) = 1) then
        res_same_fio = 1;

      res_same_birth_date = 0;
      res_same_pass_no = 0;
      res_same_pass_date = 0;
      res_same_contact = 0;
      res_same_code = 0;

      tmp_pass_no = replace(tmp_pass_no, ' ', '');
      is_critical = 0;
      if (find_pass_no = tmp_pass_no) then
      begin
        is_critical = 1;
        critical_found = 1;
        res_same_pass_no = 1;
      end

      if (find_code = tmp_code and find_code <> '') then
      begin
        is_critical = 1;
        critical_found = 1;
        res_same_code = 1;
      end

      if (find_birth_date = tmp_birth_date) then
        res_same_birth_date = 1;

      if (is_critical = 1) then
        suspend;
    end
  end -- Конец поиска по дате рождения
end
