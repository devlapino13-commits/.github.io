CREATE PROCEDURE HP_TRANSFER_LP_TO_OTHER(
    REP_DATE DATE,
    REP_LO_FROM DECIMAL(18,0),
    REP_LO_TO DECIMAL(18,0),
    KA_MEMO VARCHAR(255))
RETURNS (
    KA_ID ID_TYPE,
    KOL INTEGER)
AS
declare variable reg_id_prior integer;
declare variable kf_id id_type;
declare variable kal_new_kf_name varchar(100);
declare variable kal_new_no integer;
declare variable kal_new_officer_code varchar(20);
declare variable reg_id integer;
declare variable ka_akt_no integer;
begin
  if (rep_lo_from <= 1000000000004) then
    exit;
  if (rep_lo_to > 1000000000004) then
    exit;

  kol = 0;
  if (ka_memo is null) then
    ka_memo = '';
  reg_id_prior = 0;
  for select distinct oks_region_id, k_from_kf_id
      from kontract, oksummary
      where k_id = oks_from_k_id and
            oks_close_state = 0 and
            oks_kc_id = :rep_lo_from and
            :rep_date between oks_date_begin and oks_date_end
      order by oks_region_id
      into reg_id, kf_id
  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      ka_id = (select kv
               from get_kv(:reg_id, 7));
      ka_akt_no = coalesce((select max(cast(ka_akt_no as integer))
                            from knakt
                            where ka_from_reg_id = :reg_id), 0) + 1;
      insert into knakt (ka_id, ka_date, ka_from_reg_id, ka_akt_no, ka_memo, is_delete, ka_from_ol_id)
      values (:ka_id, :rep_date, :reg_id, :ka_akt_no, :ka_memo, 0, null);

      suspend;
    end

    kal_new_kf_name = null;
    kal_new_no = null;
    kal_new_officer_code = null;
    select kal_new_kf_name, kal_new_no, kal_new_officer_code
    from knaktlist
    where knaktlist.is_delete = 0 and
          :rep_date between kal_date and kal_date_end and
          kal_from_kf_id = :kf_id
    into kal_new_kf_name, kal_new_no, kal_new_officer_code;

    if (rep_lo_to <= 1000000000003) then
      kal_new_kf_name = kal_new_kf_name || '.ВС';
    if (rep_lo_to = 1000000000004) then
      kal_new_kf_name = kal_new_kf_name || '.КЦ';

    insert into knaktlist (kal_from_ka_id, kal_number, kal_from_kf_id, kal_from_k_id, kal_date, kal_date_end,
                           kal_new_kf_officer, kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no,
                           kal_new_officer_code)
    values (:ka_id, -1, :kf_id, :kf_id, :rep_date, '31.12.9999', :rep_lo_to, :kal_new_kf_name, 0, :reg_id, :kal_new_no,
            :kal_new_officer_code);

    kol = kol + 1;
  end
  suspend;
end
