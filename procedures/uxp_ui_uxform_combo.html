CREATE PROCEDURE UXP$UI_UXFORM_COMBO(
    UFM_UUID D_UUID)
RETURNS (
    UFC_UF_UUID D_UUID,
    UFC_HIDE_TEXT D_BOOL,
    UFC_COMBO_VALUES D_STR_8100,
    UFC_KIND D_INTEGER,
    UFC_WIDTH D_INTEGER,
    UFC_HEIGHT D_INTEGER)
AS
declare variable us_code d_str_code;
declare variable uf_field_name d_str_code;
declare variable us_uuid d_uuid;
declare variable vertical_count d_integer;
declare variable ufc_item_count d_integer;
declare variable use_enum_code d_bool;
begin
  for select distinct ufc_combo_source, ufc_combo_field, ufc_hide_text, ufc_kind
  from ux$frame, ux$frparam, ux$frcombo
  where ufr_owner = :ufm_uuid and
        ufp_ufr_uuid = ufr_uuid and
        ufc_ufp_uuid = ufp_uuid
  into us_code, uf_field_name, ufc_hide_text, ufc_kind
  do
  begin
    for select us_uuid
    from ux$source
    where us_code = :us_code
    into us_uuid
    do
    begin
      for select uf_uuid
      from ux$field
      where uf_owner = :us_uuid and
            uf_field_name = :uf_field_name
      into ufc_uf_uuid
      do
      begin
        ufc_combo_values = '';
        ufc_item_count = 0;
        ufc_width = 0;
        ufc_height = 0;

        select combo_values, values_count
        from uxp$uxfrcombo_values(:ufc_uf_uuid)
        into :ufc_combo_values, ufc_item_count;

        if (ufc_kind <> 0) then
        begin
          vertical_count = 1;
          if (ufc_kind < 1000) then
          begin
            vertical_count = round(ufc_item_count / ufc_kind);
            if (vertical_count * ufc_kind < ufc_item_count) then
              vertical_count = vertical_count + 1;
          end

          ufc_height = 20 + 22 * vertical_count;

          ufc_width = ufc_kind * 100;
          if (ufc_width > 500) then
            ufc_width = 500;
        end

        suspend;
      end
    end
  end

  for select erl_code, ufp_uf_uuid, fer_use_enum_code, fer_hide_text, fer_kind
  from ux$frame, ux$frparam, ux$field, ux$frereflist, ux$ereflist
  where ufr_owner = :ufm_uuid and
        ufp_ufr_uuid = ufr_uuid and
        ufp_uf_uuid = uf_uuid and
        fer_ufp_uuid = ufp_uuid and
        fer_erl_uuid = erl_uuid
  into us_code,  ufc_uf_uuid, use_enum_code, ufc_hide_text, ufc_kind
  do
  begin
    if (ufc_kind = 0 and ufc_hide_text = 0 and use_enum_code = 0) then
    begin
      if (not exists(select * from enum_list
                     where el_kind = :us_code and
                           el_img_uuid is not null)) then
        continue;
    end

    ufc_combo_values = '';
    ufc_item_count = 0;
    ufc_width = 0;
    ufc_height = 0;

    select combo_values, values_count
    from uxp$uxfrcombo_enum_values(:us_code, :use_enum_code)
    into :ufc_combo_values, ufc_item_count;

    if (ufc_kind <> 0) then
    begin
      vertical_count = 1;
      if (ufc_kind < 1000) then
      begin
        vertical_count = round(ufc_item_count / ufc_kind);
        if (vertical_count * ufc_kind < ufc_item_count) then
          vertical_count = vertical_count + 1;
      end

      ufc_height = 20 + 22 * vertical_count;

      ufc_width = ufc_kind * 100;
      if (ufc_width > 500) then
        ufc_width = 500;
    end

    suspend;
  end
end
