CREATE PROCEDURE HP_CHECK_PARALLEL_LOAN(
    CURR_K_ID D_ID,
    CURR_MA_ID D_INTEGER = null)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable PR_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable CNT D_INTEGER;
declare variable SCORE D_INTEGER;
declare variable K_ID D_ID;
declare variable CURR_AMOUNT D_CURRENCY;
declare variable ACTIVE_BALANCE D_CURRENCY;
declare variable MAX_SUM D_CURRENCY;
declare variable SH_DATE D_DATE;
declare variable SH_SUM D_CURRENCY;
declare variable OK_DATE D_DATE;
declare variable OK_SUM D_CURRENCY;
declare variable OKS_IS_DELINQ D_BOOL;
declare variable OKS_DELINQ_DAY D_INTEGER;
declare variable LAST_YEAR_BEGIN D_DATE;
begin
  -- Входящие параметры: curr_k_id - получаемый параллельный кредит, curr_ma_id - заявка получаемого параллельного кредита, если curr_k_id передан, то curr_ma_id не используется.

  -- Процедура проверяет условия для получения параллельного кредита (RFCAZ-280)
  -- Если условия соблюдены, то res_text = null, иначе res_text содержит нарушенное условие.

  curr_k_id = coalesce(curr_k_id, 0);
  curr_ma_id = coalesce(curr_ma_id, 0);

  if (curr_k_id = 0 and curr_ma_id = 0) then
  begin
    res_text = 'Ошибка: не указан контракт или заявка';
    suspend;
    exit;
  end

  if (f_get_db_id() <> 94401) then
    exit;

  if (curr_k_id > 0) then
    select k_from_pr_id
    from kontract
    where k_id = :curr_k_id
    into pr_id;
  else
    select ma_from_pr_id
    from miniapp
    where ma_id = :curr_ma_id and
          ma_is_active = 1 and
          ma_date_end = '31.12.9999'
    into pr_id;

  if (not exists(select *
                 from product
                 where pr_id = :pr_id and
                       lower(pr_name) containing 'paralel')) then
    exit; -- проверки касаются только тех продуктов, в названии которых есть слово "paralel" (RFCAZ-454)

  -- В АЗ все кредиты индивидуальные, групповых кредитов нет
  -- Находим клиента (c_id) и сумму кредита в текущем контракте/заявке
  if (curr_k_id > 0) then
    select kc_from_c_id,
           kc_amount_approved
    from knclient
    where kc_from_k_id = :curr_k_id and
          kc_amount_approved > 0 rows 1
    into c_id,
         curr_amount;
  else
    select mac_from_c_id,
           mac_amount
    from miniappclient
    where mac_from_ma_id = :curr_ma_id and
          mac_amount > 0 rows 1
    into c_id,
         curr_amount;

  if (coalesce(c_id, 0) = 0) then
  begin
    res_text = 'Ошибка: клиент не найден';
    suspend;
    exit;
  end

  if (coalesce(curr_amount, 0) = 0) then
  begin
    res_text = 'Ошибка: не указана сумма параллельного кредита';
    suspend;
    exit;
  end

  -- Проверка условия: У клиента 8 и более баллов по программе лояльности
  score = f_point(:c_id, current_date);
  if (coalesce(score, 0) < 8) then
  begin
    res_text = 'У клиента недостаточно баллов для параллельного кредита (нужно минимум 8 баллов)';
    suspend;
    exit;
  end

  -- RFCAZ-611
  max_sum = case
              when score between 0 and 2 then 4500
              when score between 3 and 5 then 4500
              when score between 6 and 11 then 5000
              when score between 12 and 17 then 5500
              when score between 18 and 23 then 6000
              when score between 24 and 29 then 6000
              when score between 30 and 35 then 6500
              when score between 36 and 41 then 7500
              when score between 42 and 47 then 8000
              when score >= 48 then 8500
              else 0
            end;

  -- Проверка условия: Сумма получаемого параллельного кредита не должна превышать максимальную сумму согласно накопленным баллам
  if (curr_amount > max_sum) then
  begin
    res_text = 'Сумма получаемого кредита превышает максимальную сумму согласно накопленным баллам';
    suspend;
    exit;
  end

  -- Проверка условия: Клиент не допускал просрочек по кредитам в Компании за последний год,
  -- при этом допускается не более одной просрочки до 3-х дней включительно.
  last_year_begin = dateadd(year, -1, current_date); -- начало последнего года от текущей даты
  cnt = 0;
  k_id = null;
  for select ks_from_k_id
      from kontstate
      inner join knclient on kc_from_k_id = ks_from_k_id
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_end >= :last_year_begin and -- активные кредиты за последний год
            kc_amount_approved > 0 and
            kc_from_c_id = :c_id -- фильтр по клиенту
      into k_id
  do
  begin
    oks_delinq_day = 0;
    for select oks_delinq_day
        from oksummary
        where oks_delinq_day > 0 and
              oks_delinq_date between :last_year_begin and current_date and -- просроченные платежи по графику за последний год
              oks_from_k_id = :k_id -- фильтр по контракту клиента
        into oks_delinq_day -- кол-во просроченных дней
    do
    begin
      if (oks_delinq_day > 3) then
      begin
        res_text = 'У клиента за последний год есть просрочка более 3-х дней';
        suspend;
        exit;
      end
      cnt = cnt + 1;
    end
  end

  if (cnt > 1) then
  begin
    res_text = 'У клиента за последний год есть несколько просрочек до 3-х дней';
    suspend;
    exit;
  end

  -- Проверка условия: На одного клиента не может быть более двух кредитов в Компании (1 Основной и 1 Параллельный)
  cnt = 0;
  select count(*)
  from kontstate
  inner join knclient on kc_from_k_id = ks_from_k_id
  where ks_ol_is_delete = 0 and
        ks_state = 1 and
        current_date between ks_date_begin and ks_date_end and
        kc_amount_approved > 0 and
        kc_from_c_id = :c_id
  into cnt;

  if (coalesce(cnt, 0) >= 2) then
  begin
    res_text = 'У клиента не может быть более двух кредитов в Компании';
    suspend;
    exit;
  end

  if (cnt = 0) then
    exit; -- у клиента нет активных кредитов

  select ks_from_k_id
  from kontstate
  inner join knclient on kc_from_k_id = ks_from_k_id
  where ks_ol_is_delete = 0 and
        ks_state = 1 and
        current_date between ks_date_begin and ks_date_end and
        kc_amount_approved > 0 and
        kc_from_c_id = :c_id
  into k_id; -- единственный активный на текущий день кредит у клиента

  select oks_balance
  from oksummary
  where current_date between oks_date_begin and oks_date_end and
        oks_from_k_id = :k_id
  into active_balance; -- баланс активного на текущий день кредита

  if (active_balance is null) then
    active_balance = 0;

  -- Проверка условия: Общий баланс активного кредита и получаемого параллельного кредита не должен превышать максимальную сумму согласно накопленным баллам
  if ((active_balance + curr_amount) > max_sum) then
  begin
    res_text = 'Общий баланс активного кредита и получаемого параллельного кредита превышает максимальную сумму согласно накопленным баллам';
    suspend;
    exit;
  end

  -- Проверка условия: По текущему кредиту полностью оплачено не менее 4 платежей.
  -- Полностью оплаченными признаются платежи, внесенные не ранее чем за 10 дней до даты платежа по графику.
  cnt = 0;
  select count(*)
  from schedule -- график погашений из деталей контракта
  where sh_date < current_date and
        (sh_os + sh_perc) > 0 and
        sh_from_k_id = :k_id
  into cnt;

  if (coalesce(cnt, 0) < 4) then
  begin
    res_text = 'Клиент по текущему активному кредиту еще не произвел 4 погашения';
    suspend;
    exit;
  end

  cnt = 0;
  for select ok_sh_date,
             sum_of_ok_os + sum_of_ok_perc
      from kontract_full_oper_list_detail(:k_id, current_date) -- график погашений из отчета "Детальное распределение погашений"
      where ok_sh_date < current_date and
            ok_type = 4 and
            (sum_of_ok_os + sum_of_ok_perc) < 0
      into sh_date,
           sh_sum
  do
  begin
    ok_date = null;
    ok_sum = 0;
    select max(ok_ol_date),
           sum(ok_os + ok_perc)
    from operkred
    where ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_from_k_id = :k_id and
          ok_sh_date = :sh_date
    into ok_date, -- дата последнего фактического платежа по плановому погашению
         ok_sum; -- сумма платежа (ОС + %)

    if (sh_date - ok_date <= 10 and -- платеж не ранее чем за 10 дней до даты платежа по графику
        abs(sh_sum) + ok_sum <= 5) then -- платеж полностью закрыл сумму погашения по графику (ok_sum отр. число, а недоплата до 5 манат включительно легитимно, RFCAZ-286)
      cnt = cnt + 1; -- "правильные" платежи
  end

  if (cnt < 4) then
  begin
    res_text = 'Клиент по текущему активному кредиту еще не произвел 4 погашения';
    suspend;
    exit;
  end

  -- Проверка условия: У клиента нет активной просрочки по текущему кредиту в Компании.
  oks_is_delinq = 0;
  select oks_is_delinq
  from oksummary
  where current_date between oks_date_begin and oks_date_end and
        oks_from_k_id = :k_id
  into oks_is_delinq;

  if (oks_is_delinq = 1) then
  begin
    res_text = 'У клиента есть активная просрочка по текущему кредиту в Компании';
    suspend;
    exit;
  end

  suspend;
end
