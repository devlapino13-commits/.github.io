CREATE PROCEDURE AAREP_LOANS_CARD_DETAIL(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_REGION_ID ID_TYPE)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_100,
    KC_AMOUNT_APPROVED D_CURRENCY,
    K_DISB_DATE D_DATE,
    ESM_DATETIME D_DATETIME,
    ET_CARDNUMBER D_STR_100,
    KC_NAME_FIO D_STR_100,
    ET_TOKEN D_STR_100)
AS
begin
    for select k.k_kontract_no,
               c.c_name_fio,
               kc.kc_amount_approved,
               k.k_disb_date,
               esm.esm_datetime,
               et.et_cardnumber,
               kc_c.c_name_fio,
               et.et_token
    from kontstate ks
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join knclient kc on kc.kc_from_k_id = k.k_id
    inner join clients c on c.c_id = kc.kc_from_c_id
    inner join extsysmain esm on esm.esm_from_k_id = k.k_id
    inner join extsys_token et on et.et_token = esm.esm_account
    inner join clients kc_c on kc_c.c_id = k.k_fact_disb_lo_id
    inner join region r on r.reg_unicode = ks.ks_region_id
    inner join region_list(:rep_region_id, 1, 3, 1) on r.reg_unicode = id_list
    where ks.ks_ol_is_delete = 0
      and ks.ks_state = 1
      and ks.ks_date_begin between :rep_date_begin and :rep_date_end
      and kc.kc_amount_approved > 0
      and esm.esm_project = 'cnp'
      and esm.esm_object_kind = 0
      and esm.esm_result_id = 2
      and esm.esm_status = 2
    into :k_kontract_no,
         :c_name_fio,
         :kc_amount_approved,
         :k_disb_date,
         :esm_datetime,
         :et_cardnumber,
         :kc_name_fio,
         :et_token
    do
    begin
        suspend;
    end
end
