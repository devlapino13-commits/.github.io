CREATE PROCEDURE HP_CHECK_LOAN_KOND_RU_INDIV(
    K_ID ID_TYPE)
RETURNS (
    KF_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    KF_ERROR_TEXT VARCHAR(255),
    KF_ERROR_IS_CRITICAL INTEGER,
    LAST_K_ID DECIMAL(18,0),
    MAX_DELINQ_DAY INTEGER)
AS
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_from_pr_id integer;
declare variable kc_from_c_id id_type;
declare variable k_percent_pr perc_type;
declare variable k_percent perc_type;
declare variable k_kolich_lgota integer;
declare variable c_citizen integer;
declare variable pr_loan_type integer;
declare variable k_from_kf_id id_type;
declare variable kf_region_id integer;
declare variable k_circle integer;
declare variable kf_officer_id id_type;
declare variable kf_officer varchar(100);
declare variable par7_officer curr_type;
declare variable date_begin_par7 date;
declare variable k_kontract_no varchar(30);
declare variable z_from_c_id id_type;
declare variable name_poruch varchar(100);
declare variable dopusk smallint;
declare variable cds_doc_date_end date;
declare variable kol_cli integer;
declare variable magration_card_begin date;
declare variable kol_magration_card integer;
declare variable is_patent smallint;
declare variable pr_code varchar(100);
declare variable ks_state_prev_kont decimal(18,0);
declare variable pr_code_prev_kont varchar(100);
declare variable k_period_prev_kont integer;
declare variable check_prev_pr varchar(20);
declare variable repaid_kont smallint;
declare variable k_id_last_group id_type;
declare variable max_delinq_day_last_group integer;
declare variable re_pr_code varchar(20);
declare variable kf_officer_id_last id_type;
declare variable k_close_date_last date;
declare variable jk_system_code varchar(100);
declare variable ps_date_begin date;
declare variable is_poruch integer;
declare variable ps_from_reg_id integer;
begin
  kf_error_is_critical = 0;
  kf_error_text = '';
-- Проверка срока и кол-во клиентов
  select k_kolich_repay, k_kolich_client_approved, k_from_pr_id, k_disb_date, k_percent, k_kolich_lgota, pr_loan_type,
         k_from_kf_id, k_circle,
--         pr_common_name,
--         pr_is_employer,
--         pr_name,
--         pr_product_kind,
--         K_KREDLINE_END,
         pr_code
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into k_kolich_repay, k_kolich_client_approved, k_from_pr_id, kf_disb_date, k_percent, k_kolich_lgota, pr_loan_type,
       k_from_kf_id, k_circle,
--       pr_common_name,
--       pr_is_employer,
--         pr_name,
--         pr_product_kind,
--         K_KREDLINE_END,
       pr_code;

  if (pr_loan_type <> 1) then
  begin
    exit;
  end

  if (k_kolich_repay is null) then
    k_kolich_repay = 0;
  if (k_kolich_lgota is null) then
    k_kolich_lgota = 0;
  k_kolich_repay = k_kolich_repay - k_kolich_lgota;
  if (k_kolich_client_approved is null) then
    k_kolich_client_approved = 0;
  if (k_from_pr_id is null) then
    k_from_pr_id = 0;
  if (k_circle is null) then
    k_circle = 0;

  select kal_new_reg_id, kal_new_kf_officer
  from knaktlist
  where knaktlist.is_delete = 0 and
        :kf_disb_date between kal_date and kal_date_end and
        kal_from_kf_id = :k_from_kf_id
  into kf_region_id, kf_officer_id;

  dopusk = 0; -- убран при релазации по приказу от 06.03.2015
  /* Проверить наличие следующих документов каждого клиента данного контракта, и если все соответствует,
                   то пареметр нужно сделать dopusk = 1, тем самым в следующий раз не будет проверять.

                2.1.    Клиенты могут получить микрозаем при условии соответствия одному (любому) из следующих критериев: 
                a)    клиенты въехали в РФ после 1 января 2015г. и находятся в РФ не более 20 календарных дней на момент обращения за микрозаймом;
                b)    клиенты имеют один из следующих документов, необходимых для получения нового патента (на работу):
                      сертификат, подтверждающий знания клиентом русского языка, истории и основ законодательства России,
                      либо медицинскую справку (выданную не позднее 25 дней на момент обращения за займом) либо действующий
                      Полис (договор) добровольного медицинского страхования. */

  kol_magration_card = 0;
  kol_cli = 0;
  for select kc_from_c_id, c_citizen
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into kc_from_c_id, c_citizen
  do
  begin
    kol_cli = kol_cli + 1;
-- добавлено условие igusev 21.01.2015
-- Согласно письма от Зары от 20.01.2015
-- По нашим правилам поручитель может получить или иметь групповой микрозаем и заем «1+1». Но не может получать индивиудальный.
--  Проверка, клиент не находится ли у других в поручителях
--                  if (pr_name not containing '1+1') then
    if (pr_code not containing '.ONE.') then -- изменен от 17.03.2015 Улан
    begin
      k_kontract_no = null;
      select k_kontract_no
      from kontstate, kontract, product, zalog
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :kf_disb_date between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_loan_type = 1 and
            k_id = z_from_k_id and
            z_type = 0 and
            z_from_c_id = :kc_from_c_id rows 1
      into k_kontract_no;
      if (k_kontract_no is not null) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'Клиент поручитель контракта №' || k_kontract_no || ', выдача запрещена!';
        suspend;
      end
    end

    if (dopusk = 0) then -- добавлено от 06.02.2015 Улан
    begin
      if (c_citizen = 13) then
      begin
        dopusk = 1;
      end
      else
      begin
        cds_doc_date_end = null;
        magration_card_begin = null;
        is_patent = 0;
        select
--                               max(CDS_DOC_DATE_END)
               max(iif(pdt_code in ('work_permit', 'registration_rvp', 'registration_vnj'), cds_doc_date_end, null)), -- для проверки по пункту 2,2 №04-02-15
               max(iif(pdt_code = 'magration_card', cds_doc_date_begin, null)), -- дата миграционной карты
               max(iif(pdt_code = 'work_patent', 1, null))

        from clientdocs, passkind, passdoctype
        where cds_from_c_id = :kc_from_c_id and
              cds_from_pk_id = pk_id and
              pk_from_pdt_id = pdt_id and
              cds_is_delete = 0 and
              :kf_disb_date between cds_doc_date_begin and cds_doc_date_end and
              (pdt_code in ('work_permit', -- Разрешение на работу
                        'registration_rvp', -- Разрешение на временное проживание (РВП)
                        'registration_vnj', -- Вид на жительство (ВНЖ)
                        'magration_card' -- Миграционная карта
                        ) or
              (pdt_code = 'work_patent' -- Патент выданный после 1 января
              and
              cds_doc_date_begin >= '01.01.2015'))
        into cds_doc_date_end, magration_card_begin, is_patent;

        if (cds_doc_date_end is null) then
          cds_doc_date_end = '01.01.1900';
        if (magration_card_begin is null) then
          magration_card_begin = '01.01.1900';
        if (is_patent is null) then
          is_patent = 0;
        if (cds_doc_date_end >= dateadd(month, 2, :kf_disb_date) or is_patent = 1) then
        begin
          dopusk = 1;
        end
        if ((magration_card_begin > '01.01.2015') and
            datediff(day, magration_card_begin, kf_disb_date) <= 20) then
        begin
          kol_magration_card = kol_magration_card + 1;
        end

      end
    end
  end
--            end

  if (kol_magration_card = kol_cli) then -- если кол-во соответствующих миграционок совпадает с кол-вом клентов то допускается к выдаче
  begin
    dopusk = 1;
  end

--        if (pr_common_name like 'пп индив%') then
  if (pr_code containing '.PP.') then -- Добавлено от 17.03.2015 Улан замена к предыдущей строке
  begin
--        Добавлено по приказу от 25 апреля отправленной Зарой
--        Вычисляется предыдущий цикл группы, если индив ПП и в текущем КСнике был то проверка по на ПАР не будет
    re_pr_code = null;
    k_id_last_group = null;
    kf_officer_id_last = null;
    select pr_code, k_id, ks_kc_id
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_from_k_id = k_id and
          ks_from_k_id <> :k_id and
          ks_date_begin < :kf_disb_date and
          k_from_kf_id = :k_from_kf_id and
          k_from_pr_id = pr_id
    order by ks_date_begin desc
    rows 1
    into re_pr_code, k_id_last_group, kf_officer_id_last;
    if (re_pr_code is null) then
      re_pr_code = '';
    if (k_id_last_group is null) then
      k_id_last_group = 0;
    if (kf_officer_id_last is null) then
      kf_officer_id_last = 0;

--        вычисляется дата закрытия предыдущего цикла
    k_close_date_last = null;
    select ks_date_begin
    from kontstate
    where ks_from_k_id = :k_id_last_group and
          ks_ol_is_delete = 0 and
          ks_state = 2 and
          :kf_disb_date between ks_date_begin and ks_date_end
    into k_close_date_last;
    if (k_close_date_last is null) then
      k_close_date_last = cast('01.01.1990' as date);

    max_delinq_day_last_group = 1; -- специально сделано так, чтобы если больше нуля то срабатывала запрет по индив кредитам
--        если предыдущий цикл индив ПП и у данного КСника брал то проверяет на просрочку
    if (re_pr_code containing '.PP.' and
        kf_officer_id_last = kf_officer_id and
        k_close_date_last >= dateadd(month, -6, :kf_disb_date)) then
    begin
--          проверка на максимальный просроченный день
      max_delinq_day_last_group = null;
      select max(ok_ol_date - ok_sh_date)
      from operkred
      where ok_from_k_id = :k_id_last_group and
            ok_ol_is_active = 1 and
            ok_type in (4, 5) and
            ok_ol_date > ok_sh_date and
            ok_os + ok_perc < -100
      into max_delinq_day_last_group;
      if (max_delinq_day_last_group is null) then
        max_delinq_day_last_group = 0;
--          если проделанных просрочек не имеется то проверяет на текущий день предыдущую группу
      if (max_delinq_day_last_group = 0) then
      begin
        max_delinq_day_last_group = null;
        select current_date - oks_delinq_date
        from oksummary
        where oks_from_k_id = :k_id_last_group and
              current_date between oks_date_begin and oks_date_end and
              current_date > oks_delinq_date and
              oks_close_state = 0
        into max_delinq_day_last_group;
        if (max_delinq_day_last_group is null) then
          max_delinq_day_last_group = 0;
        if (max_delinq_day_last_group > 0) then
        begin
          kf_error_is_critical = 1;
          kf_error_text = 'Предыдущий цикл просрочен, выдача запрещается.';
        end
      end
    end

--        если условие по приказу от 25.04.2015 не срабатывает т.е. max_delinq_day_last_group > 0 то как старый запрет будет работать
    if (max_delinq_day_last_group > 0) then
    begin
--    из таблицы ищем СВЗ, имеется ли PAR7 > 5%
      par7_officer = null;
      select op_par8_180, op_date_begin - 1
      from officerpar7
      where op_kc_id = :kf_officer_id and
            current_date between op_date_begin and op_date_end
      into par7_officer, date_begin_par7;
      if (par7_officer is null) then
        par7_officer = 0;

      if (current_date <= cast('10.03.2015' as date)) then
      begin
        if (par7_officer > 3.0000) then
        begin
          kf_officer = (select c_name_fio
                        from clients
                        where c_id = :kf_officer_id);
          kf_error_is_critical = 1;
          kf_error_text = 'у СВЗ ' || kf_officer || ' по индив. займам PAR7 > 5%, от ' || date_begin_par7; -- << --  закоментировал igusev 10.03.2015
          suspend;
        end
      end
      else
-- добавил igusev 10.03.2015  согласно письма Бабура от 10.03.2015
      begin
        if (par7_officer > 7.0000 and
            kf_officer_id <> 1000000000004) then -- добавил Анатолий 06.06.2015 по обращению С. Даниловой, согласовал с Уланом, Игорем
        begin
          kf_officer = (select c_name_fio
                        from clients
                        where c_id = :kf_officer_id);
          kf_error_is_critical = 1;
          kf_error_text = 'у СВЗ ' || kf_officer || ' PAR7 > 7%, от ' || date_begin_par7 || ', выдача запрещена';
          suspend;
        end
      end

      if (kf_region_id in (1041400, /*1100100,*/ 1170100, 1230100, 1350100) and
          current_date > cast('03.12.2014' as date)) then -- убран офис Пенза из запрета, пиьсмо от Зары 23.01.2015 20:48
      begin
        kf_officer = (select reg_code || ' ' || reg_name
                      from region
                      where reg_unicode = :kf_region_id);
        kf_error_is_critical = 1;
        kf_error_text = 'По офису ' || kf_officer || ' запрет по индив. займам';
        suspend;
      end
    end

  end
--      Вытаскивает поручителя данного контракта
  for select z_from_c_id, c_citizen
      from zalog, clients
      where z_type = 0 and
            z_from_k_id = :k_id and
            z_from_c_id = c_id
      into z_from_c_id, c_citizen
  do
  begin
-- igusev 17.02.2015 согласно письма от Санжара от  17.02.2015 "Уберите ограничение если поручителем выступает ДО: Акинай, Темир или сам Бабур."
-- убрал ограничение на проверку данных сотрудников, в качестве поручителей в других кредитах.
    if (z_from_c_id not in (1010100000004, 1010100016760, 1010100189147)) then
    begin
-- каждого поручителя проверяет не имеется ли поручителям другово активного контракта
      for select k_kontract_no
          from kontstate, kontract, product, zalog
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :kf_disb_date between ks_date_begin and ks_date_end and
                ks_from_k_id = k_id and
                k_id <> :k_id -- данный контракт не в счет т.е. у данного контракта есть залог
                and
                k_from_pr_id = pr_id and
                pr_product_kind <> 1 -- смотрит кроме групповых кредитов
                and
                k_id = z_from_k_id and
                z_from_c_id = :z_from_c_id
          into k_kontract_no
      do
      begin
        name_poruch = (select c_name_fio
                       from clients
                       where c_id = :z_from_c_id);
        kf_error_is_critical = 1;
        kf_error_text = 'Поручитель ' || name_poruch || ' уже является поручителем по договору №' || k_kontract_no || ', выдача запрещена!';
        suspend;
      end
    end
    if (dopusk = 0) then -- добавлено от 06.02.2015 Улан
    begin
      if (c_citizen = 13) then
      begin
        dopusk = 1;
      end
      else
      begin
        cds_doc_date_end = null;
        is_patent = 0;
        select max(cds_doc_date_end), max(iif(pdt_code = 'work_patent', 1, null))
        from clientdocs, passkind, passdoctype
        where cds_from_c_id = :z_from_c_id and
              cds_from_pk_id = pk_id and
              pk_from_pdt_id = pdt_id and
              cds_is_delete = 0 and
              :kf_disb_date between cds_doc_date_begin and cds_doc_date_end and
              (pdt_code in ('work_permit', -- Разрешение на работу
                        'registration_rvp', -- Разрешение на временное проживание (РВП)
                        'registration_vnj' -- Вид на жительство (ВНЖ)
                        ) or
              (pdt_code = 'work_patent' -- Патент выданный после 1 января
              and
              cds_doc_date_begin >= '01.01.2015'))
        into cds_doc_date_end, is_patent;
        if (is_patent is null) then
          is_patent = 0;
        if (cds_doc_date_end is null) then
          cds_doc_date_end = '01.01.1900';
        if (cds_doc_date_end >= dateadd(month, 2, kf_disb_date) or is_patent = 1) then
        begin
          dopusk = 1;
        end
      end
    end

  end
  /* убрал по просьбе Лауры от 13.12.17 Приказ потерял силу
      if (dopusk = 0) then -- добавлено от 06.02.2015 Улан
      begin
        KF_ERROR_IS_CRITICAL = 0;
        KF_ERROR_TEXT = '1. Отсутствуют необходимые документы. Приказ от 07.02.2015 №04-02-15о/д, пункт 2.2';
        suspend;
      end     */

--------------------------------------------------------------------------------------------------------------------------------------------------------------
-- проверка кредитов для сотрудников
  if (pr_code containing '.EMP.' and
      pr_code containing '.IND.') then
  begin
--      вытаскивает поручителей кредита для сотруников
    is_poruch = 0;
    for select z_from_c_id
        from zalog, clients
        where z_type = 0 and
              z_from_k_id = :k_id and
              z_from_c_id = c_id
        into z_from_c_id
    do
    begin
      is_poruch = 1;
--        должность поручителя
      jk_system_code = null;
      ps_date_begin = null;
      ps_from_reg_id = null;
      select jk_system_code, ps_date_begin, ps_from_reg_id
      from hr_pstatus, joblist, jobkind
      where ps_from_c_id = :z_from_c_id and
            current_date between ps_date_begin and ps_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id
      into jk_system_code, ps_date_begin, ps_from_reg_id;
      if (jk_system_code is null) then
        jk_system_code = '';
      if (ps_date_begin is null) then
        ps_date_begin = '01.01.1900';
      if (ps_from_reg_id is null) then
        ps_from_reg_id = 0;

      if (not(jk_system_code similar to '%(JobSystem.LO_Standart|JobSystem.Region_KM|JobSystem.LO_Senior)%') and
          ps_from_reg_id <> 1010100) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'Поручитель не является СВЗ/ССВЗ/МО, более детально можете узнать из паспорта для сотрудников пункт 4.';
        suspend;
      end
      if (dateadd(month, 6, ps_date_begin) > current_date) then
      begin
        kf_error_is_critical = 0;
        kf_error_text = 'Стаж поручителя не более 6 месяцев, и не может быть поручителям';
        suspend;
      end
    end
    if (is_poruch = 0) then
    begin
      kf_error_is_critical = 1;
      kf_error_text = 'Поручитель не зарегистрирован';
      suspend;
    end
  end
--------------------------------------------------------------------------------------------------------------------------------------------------------------
  for select kc_from_c_id, kc_amount_approved
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into kc_from_c_id, kc_amount
  do
  begin
--  проверка последнего цикла клиента
    last_k_id = null;
    max_delinq_day = 0;

    select ks_from_k_id
    from kontstate, knclient
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_from_k_id = kc_from_k_id and
          kc_from_c_id = :kc_from_c_id and
          kc_amount_approved > 0 and
          ks_date_begin < :kf_disb_date
    order by ks_date_begin desc
    rows 1
    into last_k_id;

    if (last_k_id is null) then
      last_k_id = 0;

    max_delinq_day = 0;

    if (last_k_id <> 0) then -- значить повторный клиент
    begin
      select max(ol_date - ok_sh_date)
      from operkred, operlist
      where operlist.is_delete = 0 and
            ol_date <= :kf_disb_date and
            ol_id = ok_from_ol_id and
            ol_date > ok_sh_date and
            ok_type < 7 and
            ok_from_k_id = :last_k_id and
            ok_os + ok_perc < 0 and
            ok_from_c_id = 0
      into max_delinq_day;
      if (max_delinq_day is null) then
        max_delinq_day = 0;
    end

-- igusev 12.03.2015  Добавил согласно приказа  10-03-15 ----------------->>
    if (current_date >= '13.03.2015') then
    begin
      if (
--                (pr_code containing '.ONE.') and
          (coalesce(last_k_id, 0) > 0)) then
      begin
--                if ((pr_name containing 'Уважение') or (pr_name containing 'Доверие'))  then
        if (pr_code containing '.UVAJENIE.' or pr_code containing '.DOVERIE.') then
        begin
          select ks.ks_state,
--                           pr.pr_name,
                 pr.pr_code, round(kont.k_period)
          from kontstate ks
          inner join kontract kont on ks.ks_from_k_id = kont.k_id
          inner join product pr on pr.pr_id = kont.k_from_pr_id
          where ks.ks_ol_is_delete = 0 and
                ks.ks_from_k_id = :last_k_id and
                current_date between ks.ks_date_begin and ks.ks_date_end
          into :ks_state_prev_kont,
--                         :pr_name_prev_kont,
               :pr_code_prev_kont, :k_period_prev_kont;

--                    if ((pr_name containing 'Уважение')) then check_prev_pr = 'Знакомство';
--                    if ((pr_name containing 'Доверие'))  then check_prev_pr = 'Уважение';
          if ((pr_code containing '.UVAJENIE.')) then
            check_prev_pr = '.ZNAKOM.';
          if ((pr_code containing '.DOVERIE.')) then
            check_prev_pr = '.UVAJENIE.';

          if ((ks_state_prev_kont = 2)
--                        and (pr_code_prev_kont = '.ONE.')
--                        and (pr_name_prev_kont containing :check_prev_pr)
--                        and pr_code_prev_kont containing '.ONE.'
              and
              pr_code_prev_kont containing :check_prev_pr and
              k_period_prev_kont >= 6) then
          begin
            if (max_delinq_day > 0) then
            begin
              kf_error_is_critical = 1;
              kf_error_text = 'Клиент в предыдущем цикле допускал просрочки, выдача кредита по тарифу Уважение запрещена';
              suspend;
            end
            else
            begin
              select iif(max(ok_sh_date) > (ks_date_begin + 10), 1, 0)
              from operkred, kontstate
              where ok_ol_is_active = 1 and
                    ok_from_c_id = 0 and
                    ok_from_k_id = :last_k_id and
                    ok_from_k_id = ks_from_k_id and
                    ks_state = 2 and
                    ks_ol_is_delete = 0
              group by ks_date_begin
              into :repaid_kont;
              if (repaid_kont = 1) then
              begin
                kf_error_is_critical = 1;
                kf_error_text = 'Клиент в предыдущем цикле досрочно полностью закрыл микрозаем, выдача кредита по тарифу Уважение запрещена';
                suspend;
              end
            end
          end

          if (pr_code_prev_kont containing :check_prev_pr and
              k_period_prev_kont < 6) then
          begin
            kf_error_is_critical = 1;
            kf_error_text = 'Клиент не может перейти на следующий тариф, так как срок предыдущего цикла < 6 месяцев.';
            suspend;
          end

        end
      end
    end

-- Тариф "Знакомство"
    if (pr_code containing '.ZNAKOM.' and
        pr_code containing '.PP.') then -- изменен от 17.03.2015 замена к предыдущей строке
    begin
      if (last_k_id = 0 and
          kc_amount > 30000) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'Клиент новый и не имеет получать заем более 30000,00 руб';
        suspend;
      end
      k_percent_pr = case
                       when k_kolich_repay = 3 and (kc_amount = 10000 or kc_amount = 20000) then 10
                       when k_kolich_repay = 6 and (kc_amount = 20000 or kc_amount = 30000) then 7
                       when k_kolich_repay = 9 and (kc_amount = 30000) then 6
                       else 0
                     end;
      if (k_percent <> k_percent_pr or k_percent_pr = cast(0 as perc_type)) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'Ежемесячная выплата клиента не соответствует к тарифу "Знакомство"';
        suspend;
      end
    end
--  Тариф "Уважение"
    if (pr_code containing '.UVAJENIE.' and
        pr_code containing '.PP.') then
    begin
      if (last_k_id = 0) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'Клиент новый и не имеет право получить займ по тарифу "Уважение"';
        suspend;
      end
      if (max_delinq_day > 0) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'В предыдущем цикле клиента имеются просроченные дни, не может получить по тарифу "Уважение"';
        suspend;
      end
      k_percent_pr = case
                       when k_kolich_repay = 3 and (kc_amount = 10000 or kc_amount = 20000) then 9
                       when k_kolich_repay = 6 and (kc_amount = 20000 or kc_amount = 30000 or kc_amount = 40000) then 5.8
                       when k_kolich_repay = 9 and (kc_amount = 30000 or kc_amount = 40000) then 4.5
                       when k_kolich_repay = 12 and (kc_amount = 40000) then 6.0
                       else 0
                     end;
      if (k_percent <> k_percent_pr or k_percent_pr = cast(0 as perc_type)) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'Ежемесячная выплата клиента не соответствует к тарифу "Уважение"';
        suspend;
      end
    end
--  Тариф "Доверие"
    if (pr_code containing '.DOVERIE.' and
        pr_code containing '.PP.') then
    begin
      if (last_k_id = 0) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'Клиент новый и не имеет право получить займ по тарифу "Доверие"';
        suspend;
      end
      if (max_delinq_day > 0) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'В предыдущем цикле клиента имеются просроченные дни, не может получить по тарифу "Доверие"';
        suspend;
      end
      k_percent_pr = case
                       when k_kolich_repay = 3 and (kc_amount = 20000) then 6
                       when k_kolich_repay = 6 and (kc_amount = 20000 or kc_amount = 30000 or kc_amount = 40000) then 5
                       when k_kolich_repay = 9 and (kc_amount = 30000 or kc_amount = 40000 or kc_amount = 50000) then 3.5
                       when k_kolich_repay = 12 and (kc_amount = 40000 or kc_amount = 50000) then 5
                       else 0
                     end;
      if (k_percent <> k_percent_pr or k_percent_pr = cast(0 as perc_type)) then
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'Ежемесячная выплата клиента не соответствует к тарифу "Доверие"';
        suspend;
      end
    end

    if (pr_code containing '.ONE.' and
        pr_code containing '.UVAJENIE.') then -- изменен от 17.03.2015 Улан
    begin
      if (max_delinq_day > 0) then -- В предыдущем цикле клиента имеются просроченные дни
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'В предыдущем цикле у клиента были просроченные дни, не может получить по тарифу "«1+1» (Уважение)"';
        suspend;
      end
    end

    if (pr_code containing '.ONE.' and
        pr_code containing '.DOVERIE.') then -- изменен от 17.03.2015 Улан
    begin
      if (max_delinq_day > 0) then -- В предыдущем цикле клиента имеются просроченные дни
      begin
        kf_error_is_critical = 1;
        kf_error_text = 'В предыдущем цикле у клиента были просроченные дни, не может получить по тарифу "«1+1» (Доверие)"';
        suspend;
      end
    end

  end

  if (pr_code containing '.PP.') then -- Добавлено от 17.03.2015 Улан замена к предыдущей строке
  begin

    if (:kf_disb_date <= '31.08.2014') then
