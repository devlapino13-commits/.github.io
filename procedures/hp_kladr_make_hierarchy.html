CREATE PROCEDURE HP_KLADR_MAKE_HIERARCHY
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable pki_id d_id;
declare variable pki_code d_str_20;
declare variable parent_code d_str_20;
declare variable parent_id d_id;
declare variable ksm_status d_smallint;
begin
--Структура кодового обозначения в блоке "Код":
--СС РРР ГГГ ППП АА, где
--СС – код субъекта Российской Федерации (региона), коды регионов представлены в Приложении 2 к Описанию классификатора адресов Российской Федерации (КЛАДР);
--РРР – код района;
--ГГГ – код города;     
--ППП – код населенного пункта,
--АА – признак актуальности адресного объекта.
--
--Признак актуальности может принимать следующие значения:
--            "00” – актуальный объект (его наименование, подчиненность соответствуют состоянию на данный момент адресного пространства).
--            "01”-"50” – объект был переименован, в данной записи приведено одно из прежних его наименований (актуальный адресный объект присутствует в базе данных с тем же кодом, но с признаком актуальности "00”;
--            "51” –  объект был переподчинен или влился в состав другого объекта (актуальный адресный объект определяется по базе Altnames.dbf);
--            "52”-"98” – резервные значения признака актуальности;
--            ”99” – адресный объект не существует, т.е. нет соответствующего ему актуального адресного объекта.

  error_text = '';

  select ksm_status
  from kladr_sync_main
  where ksm_status < 4
  into ksm_status;

  if (ksm_status is null) then
  begin
    error_text = 'Справочник не загружен';
    suspend;
    exit;
  end 

  if (ksm_status <> 1) then
  begin
    error_text = 'Иерархия уже построена';
    suspend;
    exit;
  end 

  -- вставим верхний узел, т.к. индекс с pki_parent is null сразу после обновления не работает
  insert into places_kladr_import (pki_name, pki_type, pki_code, pki_okatd)
  values ('РФ', '', rpad('', 13, '0'), '');

  for select pki_id, left(pki_code, 11)
  from places_kladr_import
  into pki_id, pki_code
  do
  begin
    if (pki_code = rpad('', 11, '0')) then
      continue;

    parent_id = null;
    parent_code =  case
                     when right(pki_code, 9) = rpad('', 9, '0') then rpad('', 11, '0')  -- первый уровень
                     when right(pki_code, 6) = rpad('', 6, '0') then rpad(left(pki_code, 2), 11, '0')
                     when right(pki_code, 3) = rpad('', 3, '0') then rpad(left(pki_code, 5), 11, '0')
                     else rpad(left(pki_code, 8), 11, '0')
                   end;

    if (parent_code <> '') then
      select pki_id
      from places_kladr_import
      where pki_code starting with :parent_code
      order by pki_code
      rows 1
      into parent_id;

    update places_kladr_import
    set pki_parent = :parent_id
    where pki_id = :pki_id;
  end

  -- осталась запись, которую не приткнуть никуда
  delete from places_kladr_import
  where pki_parent is null
    and pki_code not like '__000000000__';

  --- заполним временную таблицу для облегчения поиска
  delete from kladr_sync_tmp;

  insert into kladr_sync_tmp (kst_from_pl_id, kst_pl_name, kst_pl_type, kst_pl_type_index, kst_pl_code, kst_pl_parent)
  with recursive t1 (id, pl_parent, name, pl_type, code) as (
    select pl_id, parent, pl_name, plt_short_name, pl_code
  from places, placetype
  where pl_status = plt_id
    and plt_can_select < 2
    and parent = 810000000001
  union all
  select pl_id, parent, pl_name, plt_short_name, pl_code
  from t1, places, placetype
  where parent = id
    and pl_status = plt_id
    and plt_can_select < 2
  )
  select id, name, pl_type, replace(pl_type, '.', ''), code, pl_parent
  from t1;

  update kladr_sync_main
  set ksm_status = 2
  where ksm_status = 1;

  suspend;
end
