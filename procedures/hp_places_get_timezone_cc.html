CREATE PROCEDURE HP_PLACES_GET_TIMEZONE_CC(
    CC_ID D_ID)
RETURNS (
    TIMEZONE D_SMALLINT,
    MSK_CALL_TIME_START D_TIME,
    MSK_CALL_TIME_END D_TIME)
AS
declare variable CC_MSK_TIME_SHIFT D_SMALLINT;
declare variable C_REG_TIMEZONE D_SMALLINT;
declare variable C_ID D_ID;
declare variable C_CELL_NO D_STR_100;
declare variable PHONE_NO D_STR_100;
declare variable CC_MSK_CALL_TIME_START D_TIME;
declare variable CC_MSK_CALL_TIME_END D_TIME;
declare variable C_MSK_CALL_TIME_START D_TIME;
declare variable C_MSK_CALL_TIME_END D_TIME;
begin
   select cc.cc_from_c_id,
          cc.cc_cell_no
   from contacts cc
   where cc.cc_id = :cc_id
   into :c_id,
        :phone_no;

   if (phone_no is null) then
    exit;

   timezone = case
                  when phone_no starting with '996' then 6
                  when phone_no starting with '998' then 5
                  when phone_no starting with '992' then 5
                  when phone_no starting with '994' then 4
                  when phone_no starting with '77' then 5
                end;
    
   c_cell_no = f_cont_info('clients', c_id, 'cell_main', 'last');
    
   select msk_call_time_start,
          msk_call_time_end,
          reg_timezone
   from hp_places_get_timezone_c(:c_id, :c_cell_no)
   into c_msk_call_time_start,
        c_msk_call_time_end,
        c_reg_timezone;

   if (c_reg_timezone is null) then
    exit;

   if (timezone is null) then
    timezone = c_reg_timezone;

   if (timezone is null) then
    exit;

   cc_msk_time_shift = 3 - timezone;
   cc_msk_call_time_start = dateadd(hour, iif(cc_msk_time_shift < -9, -9, cc_msk_time_shift), cast('09:00' as time));
   cc_msk_call_time_end = dateadd(hour, iif(cc_msk_time_shift > 3, 3, cc_msk_time_shift), cast('20:00' as time));

   msk_call_time_start = maxvalue(cc_msk_call_time_start, c_msk_call_time_start);
   msk_call_time_end = minvalue(cc_msk_call_time_end, c_msk_call_time_end);

   suspend;
end
