CREATE PROCEDURE OKT_ACTIONS_HTML(
    UN_ID DECIMAL(18,0),
    ACL_ID DECIMAL(18,0))
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_id decimal(18,0);
declare variable anketa_code varchar(255);
declare variable res_acl_id decimal(18,0);
declare variable res_object_kind integer;
declare variable res_object_id decimal(18,0);
declare variable anketa_id integer;
declare variable k_id decimal(18,0);
declare variable str varchar(255);
declare variable res_html_tmp blob sub_type 1 segment size 80;
declare variable res_date date;
declare variable res_date_tmp date;
declare variable repay_amount decimal(18,4) = 0;
declare variable oks_delinq_date date;
declare variable oks_delinq_date_tmp date;
declare variable res_html2 blob sub_type 1 segment size 80;
begin
  /* Procedure Text */
  res_html = (select res_style
              from fm_html_styles) || '<body>';
  res_html2 = '';
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (coalesce(acl_id, 0) = 0) then
  begin
    suspend;
    exit;
  end

-- Уточняем детали задачи
  res_acl_id = :acl_id;
  select acl_object_kind, acl_object_id, acl_from_rd_id, rd_code
  from actionlist, refdata
  where acl_id = :res_acl_id and
        acl_from_rd_id = rd_id
  into :res_object_kind, :res_object_id, :anketa_id, :anketa_code;

  if (anketa_code = 'OKTELL.Projects.Verification') then
  begin
    select res_html
    from okt_dialog_verification(:res_object_id)
    into :res_html2;
  end

  if (anketa_code = 'Anketa.Delinq.1') then
  begin
    if (res_object_kind in (0, 8)) then
    begin
      res_date = null;
      oks_delinq_date = null;
      repay_amount = 0;
      k_id = null;
      for select oks_from_k_id, oks_date_begin, oks_delinq_date
          from oksummary, knclient
          where oks_close_state = 0 and
                oks_date_begin <= current_date and
                oks_date_end >= current_date and
                oks_balance > 0 and
                oks_delinq_date < current_date and
                oks_from_k_id = kc_from_k_id and
                kc_amount_approved > 0 and
                kc_from_c_id = :res_object_id
          into :k_id, res_date_tmp, oks_delinq_date_tmp
      do
      begin
        if (res_date is null) then
          res_date = res_date_tmp;
        if (oks_delinq_date is null) then
          oks_delinq_date = oks_delinq_date_tmp;

        if (res_date <= res_date_tmp) then
        begin
          res_date = maxvalue(res_date, res_date_tmp);
          select sum(-ok_os - ok_perc - ok_fine - ok_tax)
          from operkred
          where ok_from_k_id = :k_id and
                ok_ol_is_active = 1 and
                ok_type in (4, 9) and
                ok_ol_date = :res_date
          into :repay_amount;
          if (repay_amount is null) then
            repay_amount = 0;

          oks_delinq_date = minvalue(oks_delinq_date, oks_delinq_date_tmp);
        end
      end

      if (k_id is not null) then
      begin
        res_html_tmp = '<font color="red">Просрочка от ' || coalesce(right('0' || extract(day from oks_delinq_date), 2) || '.' || right('0' || extract(month from oks_delinq_date), 2) || '.' || extract(year from oks_delinq_date), 'хх.хх.хххх') || ' г.</font><br>';

        res_html_tmp = BLOB_APPEND(res_html_tmp, '<font color="#02B316">Последний раз платили ' || coalesce(right('0' || extract(day from res_date), 2) || '.' || right('0' || extract(month from res_date), 2) || '.' || extract(year from res_date), 'хх.хх.хххх') || ' г., сумма: ' || round(repay_amount, 0) || '</font>');

        select res_text
        from hp_html_delinq_anketa(:un_id, current_date, :k_id, :res_object_id)
        into :res_html2;
        res_html2 = BLOB_APPEND(res_html_tmp, '<br><br>', res_html2);

        res_html_tmp = '';
        res_date = null;
        for select ques_text || ': <b>' || res_text || '</b>', res_date
            from hp_anketa_print(:res_object_kind, :res_object_id, :anketa_code, null)
            into :str, res_date_tmp
        do
        begin
          if (res_date is null) then
            res_date = res_date_tmp;
          if (res_date_tmp is not null) then
            res_date = minvalue(res_date, res_date_tmp);
          res_html_tmp = BLOB_APPEND(res_html_tmp, str || '<br>');
        end
        if (res_html_tmp <> '') then
          res_html2 = BLOB_APPEND(res_html2, '<br><font color="#02B316"><b>Инфо от ' || coalesce(right('0' || extract(day from res_date), 2) || '.' || right('0' || extract(month from res_date), 2) || '.' || extract(year from res_date), 'хх.хх.хххх') || '</b></font><br>', res_html_tmp);
      end
      else
        res_html2 = 'На данный момент у клиента нет просрочки.';
    end
  end
  str = '';
  if (anketa_code = 'Anketa.Main') then
  begin
    select c_name_fio
    from clients
    where c_id = :res_object_id
    into :str;
    res_html2 = BLOB_APPEND(res_html2, 'ФИО: ' || str);
  end

  if (anketa_code = 'CheckAnketa.Home') then
  begin
    str = '';
    select c_name_fio
    from clients
    where c_id = :res_object_id
    into :str;
    res_html2 = BLOB_APPEND(res_html2, 'ФИО: ' || str || '<br>');

    str = '';
    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :res_object_id and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into :str;
    res_html2 = BLOB_APPEND(res_html2, 'Прописан: ' || str || '<br>');

    str = '';
    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :res_object_id and
          adc_address_type = 2 and
          adc_from_adl_id = adl_id
    into :str;
    res_html2 = BLOB_APPEND(res_html2, 'Место проживания: ' || str || '<br>');
  end
  if (anketa_code = 'CheckAnketa.Business') then
  begin
    select c_name_fio
    from clients
    where c_id = :res_object_id
    into :str;
    res_html2 = BLOB_APPEND(res_html2, 'ФИО: ' || str || '<br>');

    str = '';
    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :res_object_id and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into :str;
    res_html2 = BLOB_APPEND(res_html2, 'Прописан: ' || str || '<br>');

    str = '';
    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :res_object_id and
          adc_address_type = 3 and
          adc_from_adl_id = adl_id
    into :str;
    res_html2 = BLOB_APPEND(res_html2, 'Место работы: ' || str || '<br>');
  end
-- Делает шрифт поменьше для всей страниы
  res_html = BLOB_APPEND(res_html, res_html2, '</body>');
  suspend;
end
