CREATE PROCEDURE HP_KLADR_MAIN_UX
RETURNS (
    KSM_STATUS D_SMALLINT,
    STATUS_STR D_STR_255,
    KSM_STATUS_DATETIME D_DATETIME,
    LAST_SYNC_DATETIME D_DATETIME,
    READY_TO_LOAD D_BOOL,
    READY_TO_LOAD_STR D_STR_CODE)
AS
declare variable ksm_kladr_load_datetime d_datetime;
begin
  ksm_status = 0;
  status_str = '';

  select ksm_status, ksm_kladr_load_datetime, ksm_status_datetime
  from kladr_sync_main ksm
  order by ksm_status_datetime desc
  rows 1
  into ksm_status, ksm_kladr_load_datetime, ksm_status_datetime;

  status_str = case ksm_status
                 when 1 then 'КЛАДР загружен.'
                 when 2 then 'Иерархия построена.'
                 when 3 then 'Сверка произведена.'
                 when 4 then 'Изменения сохранены.'
               end;

  if (ksm_status = 4) then
    last_sync_datetime = ksm_status_datetime;
  else
    select ksm_status_datetime
    from kladr_sync_main ksm
    where ksm_status = 4
    order by ksm_status_datetime desc
    rows 1
    into last_sync_datetime;


  ready_to_load = iif(exists(select 1 from places_kladr_import), 0, 1);
  ready_to_load_str = iif(ready_to_load = 1, 'Готово к загрузке', 'КЛАДР загружен: ' || f_datetimetostr(ksm_kladr_load_datetime));

  suspend;
end
