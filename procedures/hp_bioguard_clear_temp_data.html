CREATE PROCEDURE HP_BIOGUARD_CLEAR_TEMP_DATA
AS
declare variable BTD_ID D_ID;
begin
    for select btd.btd_id
    from bioguard_temp_data btd
    where datediff(hour, btd.btd_create_datetime, current_timestamp) >= 1
    into :btd_id
    do
        delete from bioguard_temp_data btd
        where btd.btd_id = :btd_id;

end
