CREATE PROCEDURE HP_EXTINFO_TO_SEND_LIGRES_FSSP_CLI(
    C_ID ID_TYPE)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable c_name_fio d_str_100;
declare variable lastname d_str_100;
declare variable firstname d_str_100;
declare variable birthdate d_str_100;
begin
  for select cl.c_name_fio, f_datetostr(cl.c_birth_date), fl.fl_name1, fl.fl_name2
      from clients cl
      inner join fiolist fl on fl.fl_id = cl.c_from_fl_id
      where cl.c_id = :c_id
      into :c_name_fio, :birthdate, :lastname, :firstname
  do
  begin
    if (c_name_fio like '%УУЛУ%') then
      lastname = lastname || ' УУЛУ';

    if (c_name_fio like '%КЫЗЫ%') then
      lastname = lastname || ' КЫЗЫ';

    if (c_name_fio like '%UULU%') then
      lastname = lastname || ' UULU';

    if (c_name_fio like '%KYZY%') then
      lastname = lastname || ' KYZY';

    json_value = '{' || f_json_make_pair('birthdate', :birthdate, 'str') || ', '
                     || f_json_make_pair('firstname', :firstname, 'str') || ', '
                     || f_json_make_pair('lastname', :lastname, 'str') || '}';

    suspend;
  end
end
