CREATE PROCEDURE CB_KG_ORIV_NEW(
    REP_DATE_END DATE,
    REP_DATE_BEGIN DATE = null)
RETURNS (
    SBL_NAME D_STR_255,
    JL_NAME D_STR_100,
    REPAY_ACTIVE_DOSUDEB CURR_TYPE,
    REPAY_CLOSE_DOSUDEB CURR_TYPE,
    REPAY_ACTIVE_SUD CURR_TYPE,
    REPAY_CLOSE_SUD CURR_TYPE,
    BONUS_OT_OTDELA CURR_TYPE,
    BONUS_REPAY CURR_TYPE,
    BONUS_TOTAL CURR_TYPE,
    SUMM CURR_TYPE)
AS
declare variable repay_active_dosudeb_total curr_type;
declare variable repay_close_dosudeb_total curr_type;
declare variable repay_active_sud_total curr_type;
declare variable repay_close_sud_total curr_type;
declare variable jl_id integer;
declare variable sbl_id integer;
declare variable kol_sb smallint;
declare variable bonus_otdela_ostatok decimal(18,4);
declare variable bonus_otdela decimal(18,4);
begin

  repay_active_dosudeb_total = 0;
  repay_close_dosudeb_total = 0;
  repay_active_sud_total = 0;
  repay_close_sud_total = 0;

  insert into temp_oriv (tsbl_id, trepay_active_dosudeb, trepay_close_dosudeb, trepay_active_sud, trepay_close_sud)
  select sbl_id,
         sum(iif(is_court = 0, repay_active, 0)),
         sum(iif(is_court = 0, repay_close, 0)),
         sum(iif(is_court = 1, repay_active, 0)),
         sum(iif(is_court = 1, repay_close, 0))
  from cb_kg_oriv_temp(:rep_date_end, :rep_date_begin)
  group by sbl_id;

-- Расчет бонуса отдела ОРИВ
  select sum(trepay_active_dosudeb),
         sum(trepay_close_dosudeb),
         sum(trepay_active_sud),
         sum(trepay_close_sud),

         count(*)
  from temp_oriv
  where tsbl_id <> 60
  into repay_active_dosudeb_total,
       repay_close_dosudeb_total,
       repay_active_sud_total,
       repay_close_sud_total,
       kol_sb;

  bonus_otdela = (repay_active_dosudeb_total + repay_close_dosudeb_total + repay_active_sud_total  + repay_close_sud_total) * 2.5 / 100;
  bonus_repay = 0 ;
  for
  select c_name_fio, jl_name, ps_from_jl_id
  from hr_pstatus, joblist, jobkind, clients
  where ps_status = 1
  and ps_is_working = 1
  and :rep_date_end between ps_date_begin and ps_date_end
  and ps_from_jl_id = jl_id
  and jl_kind = jk_id
  and jk_id = 32
  and jl_id in (210,250)
  and ps_from_c_id = c_id
  into sbl_name, jl_name, jl_id
  do
  begin
    bonus_total = bonus_otdela;
    bonus_ot_otdela = bonus_otdela;
    suspend;
  end 

  jl_name = '';
  sbl_name = '';

  for
  select tsbl_id, trepay_active_dosudeb, trepay_close_dosudeb, trepay_active_sud, trepay_close_sud,
    (trepay_active_dosudeb + trepay_close_dosudeb + trepay_active_sud + trepay_close_sud)
  from temp_oriv
  into sbl_id, repay_active_dosudeb, repay_close_dosudeb, repay_active_sud, repay_close_sud, summ
  do
  begin
    select sbl_name from sb_list
    where sbl_id = :sbl_id
    into sbl_name;

    bonus_repay = round((repay_close_dosudeb * 6 + repay_active_dosudeb * 3 + repay_close_sud * 3 + repay_active_sud * 1.5) / 100, 0);

    bonus_total = bonus_repay;
    bonus_ot_otdela = 0;
    suspend;
  end
end
