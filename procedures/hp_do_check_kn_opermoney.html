CREATE PROCEDURE HP_DO_CHECK_KN_OPERMONEY(
    REP_DATE_END D_DATE)
RETURNS (
    ACC_ID D_ID,
    OB_AMOUNT D_CURRENCY,
    KOB_AMOUNT D_CURRENCY)
AS
declare variable sab_id d_id;
declare variable al_id d_str_code;
declare variable rep_date_begin d_date;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  kob_amount = 0;
  for select al_id, al_from_acc_id
      from acc_list
      where al_from_bg_id in ('50-01', '51-01', '55-04')
      into al_id, acc_id
  do
  begin
    ob_amount = f_al_bal(al_id, rep_date_end);
    suspend;
  end

  ob_amount = 0;
  for select sab_id
      from sys_accbalance
      where sab_full_code in ('Деньги.Нал.Касса', 'Деньги.Безнал.Банк.РасчСчет', 'Деньги.Безнал.Банк.СпецСчет')
      into sab_id
  do
  begin
    for select acc_id
        from ref_account
        where acc_from_sab_id = :sab_id
        into acc_id
    do
    begin
      kob_amount = f_acc_bal(acc_id, rep_date_end);
      suspend;
    end
  end

  kob_amount = 0;
  for select al_id, al_from_acc_id
      from acc_list
      where al_from_bg_id = '57-01'
      into al_id, acc_id
  do
  begin
    select dt_ekv - kt_ekv
    from hp_al_obor(:al_id, :rep_date_begin, :rep_date_end)
    into :ob_amount;

    suspend;
  end

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Деньги.Нал.вПути'
  into sab_id;

  ob_amount = 0;
  for select acc_id
      from ref_account
      where acc_from_sab_id = :sab_id
      into acc_id
  do
  begin
    kob_amount = 0;
    select sum(kobs_dt_sum - kobs_kt_sum)
    from kn_obsummary
    where kobs_acc_id = :acc_id and
          kobs_date between :rep_date_begin and :rep_date_end
    into kob_amount;
    if (kob_amount is null) then
      kob_amount = 0;

    suspend;
  end

end
