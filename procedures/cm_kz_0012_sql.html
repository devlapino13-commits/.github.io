CREATE PROCEDURE CM_KZ_0012_SQL(
    MA_ID D_ID,
    CHECK_KIND D_STR_255,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable STATUS D_SMALLINT;
begin
    -- RFCKZ-91
    status = -2;
    res_info_text = '';
    res_error_text = 'Требуется переотправка запроса в ПОД/ФТ';
    res_error_is_critical = 2;

    -- Проверяю статус.
    select
        hp.status,
        hp.res_text
    from miniapp ma, miniappclient mac, hp_afm_get_status('afm', mac.mac_from_c_id, mac.mac_from_ma_id) hp
    where ma.ma_id = :ma_id
        and ma.ma_id = mac.mac_from_ma_id
        and ma.ma_is_active = 1
        and ma.ma_date_end >= '31.12.2999'
        and mac.mac_amount > 0
    into
        :status,
        :res_error_text;

    res_error_is_critical = iif(:status = 1, 0, 2);
    res_error_text = iif(:status = 1, '', :res_error_text);

    suspend;
end
