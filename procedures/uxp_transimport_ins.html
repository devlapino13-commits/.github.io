CREATE PROCEDURE UXP$TRANSIMPORT_INS(
    TRM_ORIG_TEXT D_STR_LONG,
    TRM_IS_DELPHI D_BOOL)
AS
declare variable trm_id d_id;
declare variable is_delphi d_bool;
begin
  if (char_length(trm_orig_text) > 255) then
    exit;

  if (trim(coalesce(trm_orig_text, '')) = '') then
    exit;

  select trm_id, trm_is_delphi
  from ux$transmain
  where trm_orig_text = :trm_orig_text
  into :trm_id, :is_delphi;

  if (row_count = 0) then
  begin
    insert into ux$transmain (trm_orig_text, trm_is_delphi)
    values (:trm_orig_text, :trm_is_delphi)
    returning trm_id
    into :trm_id;
  end else
  begin
    if (coalesce(is_delphi, 0) < trm_is_delphi) then
    begin
      update ux$transmain
      set trm_is_delphi = :trm_is_delphi
      where trm_id = :trm_id;
    end
  end
end
