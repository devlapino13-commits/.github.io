CREATE PROCEDURE AAREP_EXPORT_EPS_1C_OPERBOOK(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OPER_DATE DATE,
    ACC_DT D_STR_CODE,
    ACC_KT D_STR_CODE,
    SUMMA_EKV CURR_TYPE,
    SUMMA_VAL CURR_TYPE,
    SUMMA_TAX CURR_TYPE,
    SUMMA_TAX_DT CURR_TYPE,
    SUMMA_TAX_KT CURR_TYPE,
    OPER_MEMO D_STR_255,
    ACC_DT_NAME D_STR_255,
    ACC_KT_NAME D_STR_255,
    ACC_DT_REG INTEGER,
    ACC_KT_REG INTEGER)
AS
declare variable pck_id id_type;
begin
  if (exists(select *
             from packet_1c
             where pck_date between :rep_date_begin and :rep_date_end and
                   pck_kind = 1 and
                   pck_invalid = 1)) then
    exception e_default 'Found invalid packet(s)';

  for select pck_id, pck_date
      from packet_1c
      where pck_date between :rep_date_begin and :rep_date_end and
            pck_kind = 1
      order by pck_date
      into :pck_id, :oper_date
  do
  begin
    for select f_fmt_acc(pcb_acc_dt), f_fmt_acc(pcb_acc_kt), a_dt.acc_name, a_kt.acc_name, a_dt.acc_from_reg_id,
               a_kt.acc_from_reg_id, pcb_summa, pcb_summa_val, pcb_summa_tax_dt, pcb_summa_tax_kt, obd_detail
        from packet_1c_ob, ref_account a_dt, ref_account a_kt, kn_obdetail
        where pcb_pck_id = :pck_id and
              a_dt.acc_id = pcb_acc_dt and
              a_kt.acc_id = pcb_acc_kt and
              pcb_detail_id = obd_id
        into :acc_dt, :acc_kt, :acc_dt_name, :acc_kt_name, :acc_dt_reg, :acc_kt_reg, :summa_ekv, :summa_val,
             :summa_tax_dt, :summa_tax_kt, :oper_memo
    do
    begin
      summa_tax = summa_tax_dt;
      suspend;
    end
  end
end
