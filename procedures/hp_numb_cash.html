CREATE PROCEDURE HP_NUMB_CASH(
    REG_ID INTEGER,
    BG_ID STR_20,
    WITH_ZERO INTEGER = 0,
    UN_ID ID_TYPE = null)
RETURNS (
    NUMB_CASH INTEGER)
AS
begin
  if (un_id > 0 and
    exists(
      select * from hp_get_user_roles(:un_id)
      where lst_roles containing 'poweruser_go'))
  then with_zero = 1;

  if (with_zero = 1) then
  begin
    numb_cash = 0;
    suspend;
  end

  for
    select distinct
      al_number
    from acc_list
    where al_from_bg_id = :bg_id
      and al_from_reg_id = :reg_id
      and al_is_active = 1
      and current_date between al_date and coalesce(al_date_close,'31.12.9999')
      and al_number < 100
  into :numb_cash
  do suspend;
end
