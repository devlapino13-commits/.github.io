CREATE PROCEDURE PRINT_INVENTORY_MEMBERS(
    OBJ_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_100,
    UN_NAME D_STR_64,
    QR_CODE D_STR_LONG)
AS
declare variable C_ID D_ID;
declare variable SIGN_STATUS D_SMALLINT;
declare variable SIGN_TYPE D_SMALLINT;
declare variable EDL_ID D_ID;
begin
  select cl.c_name_fio, cl.c_id, imem.imem_sign_status, imem.imem_sign_type
  from invmember imem
  left join clients cl on cl.c_id = imem.imem_from_c_id
  where imem.imem_id = :obj_id
  into c_name_fio, c_id, sign_status, sign_type;

  un_name = null;
  edl_id = null;

  if (sign_type = 0 and sign_status = 1) then --подписание через FinStart
  begin
    select un.un_name
    from username un
    where un.un_from_c_id = :c_id
    into un_name;
  end

  if (sign_type = 1 and sign_status = 1) then  --подписание через Edoc
  begin
    un_name = null;
    select un_name
    from username un
    where un.un_from_c_id = :c_id
    into un_name;

    select edl.edl_id
    from edoc_list edl
    where edl.edl_object_kind = f_objkind('invmember', 'id') and
        edl.edl_object_id = :obj_id and
        edl.edl_status = 1
    into edl_id;

    qr_code = f_edoc_qr_by_edl(:edl_id, :c_id);
  end
  suspend;
end
