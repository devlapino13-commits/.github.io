CREATE PROCEDURE HP_FIND_SAVE_DOUBLECLIENTS(
    FIND_C_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
AS
declare variable alternat_fio varchar(100);
declare variable is_found integer;
declare variable cm_database_id integer;
declare variable c_min decimal(18,0);
declare variable c_max decimal(18,0);
declare variable find_fio varchar(100);
declare variable find_birth_date date;
declare variable find_code varchar(20);
declare variable find_pass_date date;
declare variable find_pass_no varchar(100);
declare variable found_c_id decimal(18,0);
declare variable res_identify_koef integer;
declare variable is_critical integer;
declare variable res_same_fio integer;
declare variable res_same_birth_date integer;
declare variable res_same_pass_no integer;
declare variable res_same_pass_date integer;
declare variable res_same_contact integer;
declare variable res_same_code integer;
begin
  if (coalesce(find_c_id, 0) = 0) then
    exit;

  cm_database_id = f_get_db_id();

  delete from doubleclients
  where dc_from_c_id_1 = :find_c_id or
        dc_from_c_id_2 = :find_c_id;

  select coalesce(c_name_fio, ''), coalesce(c_alternat_fio, ''), c_birth_date, c_pass_no, c_pass_date,
         iif(:cm_database_id <> 94401, c_inn, c_pincode)
  from clients
  where c_id = :find_c_id
  into :find_fio, :alternat_fio, :find_birth_date, :find_pass_no, :find_pass_date, :find_code;

  found_c_id = 0;
  is_critical = 0;
  res_same_fio = 0;
  res_same_birth_date = 0;
  res_same_pass_no = 0;
  res_same_pass_date = 0;
  res_same_contact = 0;
  res_same_code = 0;
  is_found = 0;

  for select c_id, max(is_critical), max(res_same_fio), max(res_same_birth_date), max(res_same_pass_no),
             max(res_same_pass_date), max(res_same_contact), max(res_same_code)
      from hp_find_clients_tmp(:find_c_id, :find_fio, :find_birth_date, :find_code, :find_pass_date, :find_pass_no, '', 1)
      where c_id <> 0
      group by c_id
      into :found_c_id, :is_critical, :res_same_fio, :res_same_birth_date, :res_same_pass_no, :res_same_pass_date,
           :res_same_contact, :res_same_code
  do
  begin
    res_identify_koef = res_same_fio + 1;
    res_identify_koef = res_identify_koef + res_same_code + res_same_birth_date + res_same_pass_no + res_same_pass_date + res_same_contact + is_critical * 2;
    is_found = 1;

    c_min = minvalue(find_c_id, found_c_id);
    c_max = maxvalue(find_c_id, found_c_id);
    update or insert into doubleclients (dc_from_c_id_1, dc_from_c_id_2, dc_identity_koef, dc_is_critical, dc_same_fio,
                                         dc_same_birth_date, dc_same_pass_no, dc_same_pass_date, dc_same_contact,
                                         dc_same_code, dc_from_un_id)
    values (:c_min, :c_max, :res_identify_koef, :is_critical, :res_same_fio, :res_same_birth_date, :res_same_pass_no,
            :res_same_pass_date, :res_same_contact, :res_same_code, :un_id)
    matching (dc_from_c_id_1, dc_from_c_id_2);
  end
  if (is_found = 0 and
      alternat_fio <> '') then
  begin
    for select c_id, max(is_critical), max(res_same_fio), max(res_same_birth_date), max(res_same_pass_no),
               max(res_same_pass_date), max(res_same_contact), max(res_same_code)
        from hp_find_clients_tmp(:find_c_id, :alternat_fio, :find_birth_date, :find_code, :find_pass_date, :find_pass_no, '', 1)
        where c_id <> 0
        group by c_id
        into :found_c_id, :is_critical, :res_same_fio, :res_same_birth_date, :res_same_pass_no, :res_same_pass_date,
             :res_same_contact, :res_same_code
    do
    begin
      res_identify_koef = res_same_fio + 1;
      res_identify_koef = res_identify_koef + res_same_code + res_same_birth_date + res_same_pass_no + res_same_pass_date + res_same_contact + is_critical * 2;
      is_found = 1;

      c_min = minvalue(find_c_id, found_c_id);
      c_max = maxvalue(find_c_id, found_c_id);
      update or insert into doubleclients (dc_from_c_id_1, dc_from_c_id_2, dc_identity_koef, dc_is_critical,
                                           dc_same_fio, dc_same_birth_date, dc_same_pass_no, dc_same_pass_date,
                                           dc_same_contact, dc_same_code, dc_from_un_id)
      values (:c_min, :c_max, :res_identify_koef, :is_critical, :res_same_fio, :res_same_birth_date, :res_same_pass_no,
              :res_same_pass_date, :res_same_contact, :res_same_code, :un_id)
      matching (dc_from_c_id_1, dc_from_c_id_2);
    end
  end
end
