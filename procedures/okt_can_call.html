CREATE PROCEDURE OKT_CAN_CALL(
    PROJECT STR_100,
    OBJECT_MASTER_KIND STR_255,
    OBJECT_MASTER_ID ID_TYPE,
    OBJECT_CHILD_KIND STR_255,
    OBJECT_CHILD_ID ID_TYPE,
    SUBJECT_KIND_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    CAN_CALL INTEGER,
    CAN_CALL_LIST BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    STOP_CALL_LIST BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    REASON STR_255)
AS
declare variable c_id id_type;
declare variable qty_ok integer;
declare variable cc_id id_type;
declare variable can_call_tmp integer;
declare variable project_id integer;
declare variable pi_id integer;
declare variable reason_id integer;
begin

  /*
    для понимания работы процедуры, пояснение:

    PROJECT - проект КЦ в рамках которого осуществляется проверка
    OBJECT_MASTER_KIND - Kontract
    OBJECT_MASTER_ID - K_ID
    OBJECT_CHILD_KIND - Clients
    OBJECT_CHILD_ID - C_ID
    SUBJECT_KIND - (тип данных: осносной сотовый номер, дополнительный)

    CAN_CALL - (0 - нельзя звонить, 1 - можно)

    если заполнен OBJECT_MASTER_KIND, OBJECT_MASTER_ID обязательно должен быть заполнен

    Если PROJECT if not null, проверка осущестляется в рамках проекта КЦ.
    Напримре по проекту Сирена1, менеджер КЦ установил, что по контракту №..., не нужно звонить,
    но, когда контракт уйдет на Сирена2, звонки должны возобноситься.

    STOP_CALL_LIST, CAN_CALL_LIST - только если заполнен OBJECT_MASTER_KIND, OBJECT_CHILD_KIND

    *************************
    Пример 1:

    если OBJECT_MASTER_KIND = Kontract, OBJECT_MASTER_ID = K_ID,
         OBJECT_CHILD_KIND  = Clients, OBJECT_CHILD_ID = null
    Это значит проверяется возможность звонить участникам группы.
    если не одному из участников звонить нельзя, тогда CAN_CALL = 0.
    **************************
    Пример 2:

    если OBJECT_MASTER_KIND = null, OBJECT_MASTER_ID = null
         OBJECT_CHILD_KIND = Clients, а OBJECT_CHILD_ID = :C_ID
    Это значит проверяется возможность звонить клиенту/поручителю...

    данный варинт равносилен

    OBJECT_MASTER_KIND = Clients, OBJECT_MASTER_ID = C_ID
    OBJECT_CHILD_KIND  = null,  OBJECT_CHILD_ID = null

    результат будет одинаковым
  */

  can_call = 1;
  can_call_list = '';
  stop_call_list = '';
  rep_date = coalesce(rep_date, current_date);

  if (project similar to '[-]?[[:DIGIT:]]+') then
    project_id = cast(project as integer);
  else
  begin
    select rd_id
    from refdata
    where rd_code = :project
    into :project_id;
  end

  if ((object_master_kind is null) and
      (object_child_kind is not null)) then
    object_master_kind = object_child_kind;
  if ((object_master_id is null) and
      (object_child_id is not null)) then
    object_master_id = object_child_id;

  if (object_master_kind = 'Kontract') then
  begin
    if (exists(select *
               from pantera_ignore pig
               where pig.pi_object_kind = 9 and
                     pig.pi_object_id = :object_master_id and
                     pig.pi_is_active = 1 and
                     :rep_date between pig.pi_date_begin and pig.pi_date_end and
                     ((pig.pi_rd_project_id is null) or
                     (pig.pi_rd_project_id = :project_id)))) then
      can_call = 0;

    if ((object_child_kind = 'Clients') and
        (can_call = 1)) then
    begin
      qty_ok = 0;

      for select kncl.kc_from_c_id
          from knclient kncl
          where kncl.kc_from_k_id = :object_master_id and
                ((kncl.kc_from_c_id = :object_child_id) or
                (:object_child_id is null)) and
                kncl.kc_amount_approved > 0
          into :c_id
      do
      begin
        select can_call
        from okt_can_call(:project, 'Clients', :c_id, null, null, :subject_kind_id, :rep_date)
        into :can_call_tmp;

        if (can_call_tmp = 0) then
          stop_call_list = BLOB_APPEND(stop_call_list, '.' || cast(c_id as varchar(18)));
        if (can_call_tmp = 1) then
          can_call_list = BLOB_APPEND(can_call_list, '.' || cast(c_id as varchar(18)));

        qty_ok = qty_ok + can_call_tmp;
      end

      if (qty_ok = 0) then
        can_call = 0;
    end
    if ((object_child_kind = 'Contacts') and
        (can_call = 1)) then
    begin
      qty_ok = 0;

      for select cont.cc_id
          from knclient kncl
          inner join clients cl on kncl.kc_from_c_id = cl.c_id
          inner join contacts cont on cont.cc_from_c_id = cl.c_id
          where kncl.kc_from_k_id = :object_master_id and
                ((cont.cc_id = :object_child_id) or
                (:object_child_id is null)) and
                kncl.kc_amount_approved > 0 and
                ((cont.cc_total_res > 0) -- положительно проверифицированнные или те, которые не проходили верификацию
                or
                (cont.cc_verif_ignore = 1)) and
                cont.cc_is_old_contact = 0 and
                cont.cc_is_delete = 0 -- RFCRU-2246
          into :cc_id
      do
      begin
        select can_call
        from okt_can_call(:project, 'Contacts', :cc_id, null, null, :subject_kind_id, :rep_date)
        into :can_call_tmp;

        if (can_call_tmp = 0) then
          stop_call_list = BLOB_APPEND(stop_call_list, '.' || cast(cc_id as varchar(18)));
        if (can_call_tmp = 1) then
          can_call_list = BLOB_APPEND(can_call_list, '.' || cast(cc_id as varchar(18)));

        qty_ok = qty_ok + can_call_tmp;
      end
      if (qty_ok = 0) then
        can_call = 0;
    end
  end

  if ((object_master_kind = 'Clients')) then
  begin
    pi_id = null;
    reason_id = null;

    select pig.pi_id, pig.pi_reason_rd_id
    from pantera_ignore pig
    where pig.pi_object_kind = 0 and
          pig.pi_object_id = :object_master_id and
          pig.pi_is_active = 1 and
          :rep_date between pig.pi_date_begin and pig.pi_date_end and
          ((pig.pi_rd_project_id is null) or
          (pig.pi_rd_project_id = :project_id)) and
          pig.pi_subject_kind = :subject_kind_id rows 1
    into :pi_id, :reason_id;

    if (pi_id is not null) then
    begin
      can_call = 0;

      select rd.rd_name
      from refdata rd
      where rd.rd_id = :reason_id
      into :reason;
    end
  end

  if ((object_master_kind = 'Contacts')) then
  begin
    pi_id = null;
    reason_id = null;

    select pig.pi_id, pig.pi_reason_rd_id
    from pantera_ignore pig
    where pig.pi_object_kind = 13 and
          pig.pi_object_id = :object_master_id and
          pig.pi_is_active = 1 and
          :rep_date between pig.pi_date_begin and pig.pi_date_end and
          ((pig.pi_rd_project_id is null) or
          (pig.pi_rd_project_id = :project_id)) and
          pig.pi_subject_kind = :subject_kind_id rows 1
    into :pi_id, :reason_id;

    if (pi_id is not null) then
    begin
      can_call = 0;

      select rd.rd_name
      from refdata rd
      where rd.rd_id = :reason_id
      into :reason;
    end
  end

  if (stop_call_list <> '') then
    stop_call_list = BLOB_APPEND(stop_call_list, '.');
  if (can_call_list <> '') then
    can_call_list = BLOB_APPEND(can_call_list, '.');

  suspend;
end
