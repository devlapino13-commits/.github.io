CREATE PROCEDURE HP_CHECK_ACTUAL_CLIENT_RF(
    C_NAME_FIO D_STR_255,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_100,
    C_INN D_STR_20)
RETURNS (
    IS_ACTUAL_CLIENT D_SMALLINT,
    CLIENT_INFO D_STR_LONG,
    SBL_NAME D_STR_2048)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable sbl_name_tmp d_str_255;
declare variable kontract_no d_str_100;
declare variable cc_name_fio d_str_255;
begin
  sbl_name = '';
  is_actual_client = 0;
  select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn
  from clients cl1
  where (cl1.c_name_fio = :c_name_fio and
        cl1.c_birth_date = :c_birth_date) or
        (cl1.c_inn = :c_inn) and
        cl1.c_id > 0
  union
  select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn
  from clients cl2
  inner join clientdocs cds on cds.cds_from_c_id = c_id
  where cds.cds_index_doc_no = :c_pass_no and
        cl2.c_id > 0 and
        cl2.c_birth_date = :c_birth_date rows 1
  into c_id, cc_name_fio, c_birth_date, c_pass_no, c_inn;

  client_info = :cc_name_fio || ', ' || f_datetostr(:c_birth_date);

  if (c_id is not null) then
  begin
    for select kt.k_id
        from knclient kc
        inner join kontract kt on kt.k_id = kc.kc_from_k_id
        where kc.kc_from_c_id = :c_id and
              kc.kc_amount_approved > 0
        into k_id
    do
    begin
      if (exists(select *
                 from sb_list sl
                 inner join sb_akt sa on sa.sba_from_sbl_id = sl.sbl_id and
                       sa.sba_from_k_id = :k_id and
                       current_date between sa.sba_date_begin and sa.sba_date_end) and
          exists(select *
                 from oksummary oks
                 where oks.oks_from_k_id = :k_id and
                       current_date between oks.oks_date_begin and oks.oks_date_end)) then
      begin
        select k.k_kontract_no
        from kontract k
        where k.k_id = :k_id
        into kontract_no;

        select sl.sbl_name
        from sb_list sl
        inner join sb_akt sa on sa.sba_from_sbl_id = sl.sbl_id and
              sa.sba_from_k_id = :k_id and
              current_date between sa.sba_date_begin and sa.sba_date_end and
              sa.sba_is_delete = 0
        into sbl_name_tmp;
        is_actual_client = 1;
        sbl_name = sbl_name || '<br>' || :kontract_no || ' ' || :sbl_name_tmp;
      end
    end
    suspend;
  end
end
