CREATE PROCEDURE CB_AZ_LO_2014_05(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(60),
    AA_CURR_BRANCH_NAME VARCHAR(60),
    AA_CURR_KC_FIO VARCHAR(100),
    OKS_BALANCE CURR_TYPE,
    OKS_DELINQ_BALANCE CURR_TYPE,
    PAR7 CURR_TYPE,
    CB_PERSONAL_STAGE DATE,
    OKS_KC_ID DECIMAL(18,0),
    KOL_CLI_DISB_GR INTEGER,
    KOL_CLI_DISB_IND INTEGER,
    IS_STAJER INTEGER,
    DISB_FROM_BONUS INTEGER,
    BONUS_DISB CURR_TYPE,
    BONUS_PORTFEL CURR_TYPE,
    PERC_UDERJ INTEGER,
    BONUS_TOTAL CURR_TYPE)
AS
declare variable oks_region_id integer;
declare variable rep_date_begin date;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  for select oks_region_id, oks_kc_id, sum(oks_balance), sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0))
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_region_id = id_list and
            oks_close_state = 0
      group by oks_region_id, oks_kc_id
      into oks_region_id, oks_kc_id, oks_balance, oks_delinq_balance
  do
  begin
    aa_curr_office_name = null;
    aa_curr_branch_name = null;
    select aa_curr_office_name, aa_curr_branch_name
    from aarep_get_office_categ(:oks_region_id, :rep_date_end)
    into aa_curr_office_name, aa_curr_branch_name;
    aa_curr_kc_fio = null;
    select c_name_fio
    from clients
    where c_id = :oks_kc_id
    into aa_curr_kc_fio;

    if (oks_balance > 0) then
      par7 = oks_delinq_balance / oks_balance * 100;
    else
      par7 = 0;
    par7 = round(par7, 4);

    perc_uderj = case
                   when par7 < 1.0 then 0
                   when par7 >= 1.0 and par7 < 2 then 25
                   when par7 >= 2.0 and par7 < 3 then 50
                   else 100
                 end;

--  Вычисление стажа сотрдуника
--    Дата поступления на работу
    cb_personal_stage = null;
    select ps_date_begin
    from hr_pstatus --, hr_orders , joblist
    where ps_from_c_id = :oks_kc_id and
          ps_date_begin is not null and
          ps_status = 1
    order by ps_date_begin
    rows 1
    into :cb_personal_stage;

    if (coalesce(:cb_personal_stage, current_date) > dateadd(month, -3, :rep_date_end)) then
    begin
      is_stajer = 1;
    end
    else
      is_stajer = 0;

    kol_cli_disb_gr = null;
    select sum(iif(pr_loan_type = 0, k_kolich_client_approved, 0)),
           sum(iif(pr_loan_type = 1, k_kolich_client_approved, 0))
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id
--    and pr_loan_type = 0 -- выборка по групповым кредитам
          and
          ks_region_id = :oks_region_id and
          ks_kc_id = :oks_kc_id
    into kol_cli_disb_gr, kol_cli_disb_ind;
    if (kol_cli_disb_gr is null) then
      kol_cli_disb_gr = 0;
    if (kol_cli_disb_ind is null) then
      kol_cli_disb_ind = 0;

    if (is_stajer = 1) then
    begin
      disb_from_bonus = kol_cli_disb_gr - 12; --изменена от 17.06.2014 было  disb_from_bonus = kol_cli_disb_gr - 21;
      if (disb_from_bonus < 0) then
        disb_from_bonus = 0;
    end
    else
      disb_from_bonus = kol_cli_disb_gr;

    bonus_disb = disb_from_bonus * 6 + kol_cli_disb_ind * 4;

    if (oks_balance >= 50000.0000) then
    begin
      bonus_portfel = oks_balance * 0.1 / 100;
    end
    else
      bonus_portfel = 0;

    bonus_total = bonus_disb + bonus_portfel;

    bonus_total = bonus_total * (100 - perc_uderj) / 100;

    suspend;
  end
end
