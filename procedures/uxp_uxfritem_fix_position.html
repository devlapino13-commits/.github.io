CREATE PROCEDURE UXP$UXFRITEM_FIX_POSITION(
    UFI_PARENT_UUID D_UUID)
AS
declare variable new_position d_integer;
begin
  new_position = -1;
  for select ufi_uuid, ufi_parent_uuid, ufi_position
  from ux$fritem a
  where ufi_parent_uuid is not distinct from :ufi_parent_uuid
  order by ufi_position, rdb$db_key
  as cursor ufi
  do
  begin
    if (ufi.ufi_parent_uuid is null) then
      new_position = -1;

    new_position = new_position + 1;

    if (ufi.ufi_position is distinct from new_position) then
    begin
      update ux$fritem
      set ufi_position = :new_position
      where current of ufi;
    end

    execute procedure uxp$uxfritem_fix_position(ufi.ufi_uuid);
  end
end
