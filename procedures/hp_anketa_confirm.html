CREATE PROCEDURE HP_ANKETA_CONFIRM(
    K_ID D_ID,
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    IS_ERROR D_INTEGER,
    RES_TEXT D_STR_255)
AS
declare variable mac_id d_integer;
begin
  is_error = 1;
  res_text = 'Подтверждения анкеты не произошло';

  select macl.mac_id
  from kontract kont
  inner join miniappclient macl on macl.mac_from_ma_id = kont.k_from_ma_app_id
  where kont.k_id = :k_id and
        macl.mac_from_c_id = :c_id
  into :mac_id;

  if (mac_id is not null) then
  begin
    update cfp_anketa ca
    set ca.ca_date = current_date,
        ca.ca_datetime = current_timestamp,
        ca.ca_from_un_id = :un_id
    where ca.ca_object_id = :mac_id;

    is_error = 0;
    res_text = 'Подтверждение анкеты прошло успешно';
  end

  suspend;
end
