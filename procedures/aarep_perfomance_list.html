CREATE PROCEDURE AAREP_PERFOMANCE_LIST(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_100,
    PR_NAME VARCHAR(70),
    IS_INDIV D_STR_20,
    OFFICER_NAME D_STR_100,
    K_STATUS D_STR_20,
    LAL_NAME D_STR_255,
    REPAY_DATE D_DATE,
    REPAY_SUM CURR_TYPE,
    DEBT_REMAIN CURR_TYPE)
AS
declare variable K_ID D_ID;
declare variable C_ID ID_TYPE;
declare variable C_DOLYA D_CURRENCY;
declare variable OL_ID D_ID;
declare variable IS_OPERKRED D_SMALLINT;
declare variable K_KOLICH_CLIENT_APPROVED D_INTEGER;
declare variable LAST_CLIENT D_SMALLINT;
declare variable C_DOLYA_TMP D_CURRENCY;
declare variable REPAY_SUM_TMP D_CURRENCY;
declare variable DEBT_REMAIN_TMP D_CURRENCY;
declare variable K_ID_OLD D_ID;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
begin
  k_id_old = 0;

  for select k_id,
             repay_sum,
             repay_date,
             ol_id,
             is_operkred
      from(
          select ok.ok_from_k_id as k_id,
                 -sum(ok.ok_os + ok.ok_perc + ok.ok_sh_os + ok.ok_sh_perc + ok.ok_fine + ok.ok_kommis + ok.ok_tax + ok.ok_kommis_adv) as repay_sum,
                 ok.ok_ol_date as repay_date,
                 ok.ok_from_ol_id as ol_id,
                 1 is_operkred
          from operkred ok
          inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(ok.ok_ol_reg_id as d_integer)
          where ok.ok_type in (4, 9) and
                ok.ok_ol_date between :rep_date_begin and :rep_date_end and
                ok.ok_ol_is_active = 1
          group by ok.ok_from_k_id,
                   ok_ol_date,
                   ok_from_ol_id
      ) a
      union all
      select distinct okf.oko_from_k_id,
                      0,
                      okf.oko_ol_date,
                      okf.oko_from_ol_id,
                      0
      from operkroff okf
      inner join operlist ol on ol.ol_id = okf.oko_from_ol_id
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(ol.ol_from_reg_id as d_integer)
      where okf.oko_type = 4 and
            okf.oko_ol_date between :rep_date_begin and :rep_date_end and
            okf.oko_ol_is_active = 1
      into k_id,
           repay_sum,
           repay_date,
           ol_id,
           is_operkred
  do
  begin
    repay_sum_tmp = repay_sum;
    if (k_id_old <> k_id) then
    begin
        last_client = 0;
        c_dolya_tmp = 0;
        c_id = null;
        k_kontract_no = null;
        pr_name = null;
        k_kolich_client_approved = null;
        c_dolya = null;
        k_amount_approved = 0;

        k_id_old = k_id;

        select kt.k_kontract_no,
                    pr.pr_name,
                    kt.k_kolich_client_approved,
                    k_amount_approved,
                    reg.reg_name,
                    cli.c_name_fio
        from kontract kt
        inner join product pr on pr.pr_id = kt.k_from_pr_id
        inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id
        inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
        inner join clients cli on cli.c_id = kal.kal_new_kf_officer
        where kt.k_id = :k_id
        and :rep_date_end between kal.kal_date and kal.kal_date_end
        and kal.is_delete = 0
        into k_kontract_no,
             pr_name,
             k_kolich_client_approved,
             k_amount_approved,
             reg_name,
             officer_name;

        k_status = null;

        select iif(oks.oks_close_state = 1, 'внебаланс', 'активный')
        from oksummary oks
        where oks.oks_from_k_id = :k_id
        and :rep_date_end between oks.oks_date_begin and oks.oks_date_end
        into k_status;
        
        k_status = coalesce(k_status, 'закрытый');

        if (:is_operkred = 1) then
        begin
        debt_remain = null;

        select sum(zad_os + sh_calc_perc + sh_calc_fine)
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        into debt_remain;

        debt_remain = coalesce(debt_remain,0);
        debt_remain_tmp = debt_remain;
        end
    end

    for select kc.kc_from_c_id,
               kc.kc_amount_approved / :k_amount_approved,
               cl.c_name_fio
        from knclient kc
        inner join clients cl on cl.c_id = kc.kc_from_c_id
        where kc.kc_from_k_id = :k_id and
              kc.kc_amount_approved > 0
        into c_id,
             c_dolya,
             c_name_fio
    do
    begin
      c_dolya_tmp = c_dolya_tmp + c_dolya;
      last_client = last_client + 1;
      is_indiv = iif(:k_kolich_client_approved > 1, 'групповой', 'индив.');
      lal_name = null;

      if (f_get_db_id() = 94401) then
        select lal.lal_name
        from lawmain lam
        inner join lawkredit lak on lak.lak_from_lam_id = lam.lam_id
        inner join lawcourts lac on lac.lac_from_lam_id = lam.lam_id
        inner join lawlist lal on lal.lal_id = lac.lac_from_lal_id
        where lak.lak_from_k_id = :k_id and
            lak.lak_from_c_id = :c_id and
            lam.lam_is_active = 1 and
            :rep_date_end between lac.lac_date_begin and lac.lac_date_end
        order by lac.lac_date_begin desc rows 1
        into lal_name;
      else
        select lal.lal_name
        from lawmain lam
        inner join lawkredit lak on lak.lak_from_lam_id = lam.lam_id
        inner join lawlist lal on lal.lal_id = lam.lam_from_lal_id
        where lak.lak_from_k_id = :k_id and
            lak.lak_from_c_id = :c_id and
            lam.lam_is_active = 1
        order by lam.lam_date_begin desc rows 1
        into lal_name;

      repay_sum = repay_sum_tmp * c_dolya;


      debt_remain = debt_remain_tmp * c_dolya;

      if (debt_remain <= 0) then
      begin
        debt_remain = 0;
        k_status = 'закрытый';
      end

      if (c_dolya_tmp <> 1 and last_client = :k_kolich_client_approved) then
      begin
        c_dolya_tmp = 1 - c_dolya_tmp;
        c_dolya = c_dolya + c_dolya_tmp;
        repay_sum = repay_sum_tmp * c_dolya;
        debt_remain = debt_remain_tmp * c_dolya;
      end

      if (is_operkred = 0) then
      begin
         repay_sum = null;
         select -(coalesce(okf.oko_os, 0) + coalesce(okf.oko_perc, 0) + coalesce(okf.oko_other, 0) + coalesce(okf.oko_sh_os, 0) + coalesce(okf.oko_sh_perc, 0) + coalesce(okf.oko_sh_other, 0))
         from operkroff okf
         where okf.oko_from_k_id = :k_id and
            okf.oko_from_c_id = :c_id and
            okf.oko_from_ol_id = :ol_id and
            okf.oko_ol_date between :rep_date_begin and :rep_date_end and
            okf.oko_type = 4 and
            okf.oko_ol_is_active = 1
         into repay_sum;
         repay_sum = coalesce(repay_sum, 0);

         debt_remain = null;

         select sum(oko_sh_os + oko_sh_perc + oko_sh_other + oko_other + oko_os + oko_perc)
         from operkroff opf, operlist ol
         where oko_from_k_id = :k_id and
               oko_from_c_id = :c_id and
               oko_from_ol_id = ol_id and
               oko_ol_date <= :rep_date_end and
               ol.is_delete = 0
         into debt_remain;

         debt_remain = coalesce(debt_remain,0);
      end
      suspend;
    end
  end
end
