HP_AFK_CHECK_RULE

Проверка наличия нужных триггеров при создании механизма, заменяющего FK ключи

Входные параметры: 0
Выходные параметры: 2
Вызывает процедур: 1
Вызывается из: 0
Использует другие процедуры1 шт.
HP_AFK_GET_TABLE
Вызывается из других процедур0 шт.

ни одна процедура не вызывает эту

Входные параметры0 шт.

параметров нет

Выходные параметры2 шт.

Тип

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_AFK_CHECK_RULE
RETURNS (
    O_FK_TABLE D_STR_64,
    RES D_STR_255)
AS
declare variable f_master d_str_64;
declare variable fk_master d_str_64;
declare variable fk_field d_str_64;
declare variable fk_master_list d_str_100;
declare variable fk_field_list d_str_100;
declare variable v_trigger d_str_64;
declare variable v_trigger_m d_str_64;
declare variable idx_sorting d_str_64;
declare variable v_idx_name d_str_64;
declare variable v_cnt d_integer;
declare variable v_delim d_str_10;
declare variable v_idx_exists d_str_100;
declare variable v_idx_sort_exists d_str_64;
declare variable v_bool boolean;
begin
  /******************************************************************************
*        Процедура делает проверку правильности создания AFK триггеров
*                      и индексов
*
*******************************************************************************/
  for select upper(f_master) f_master, upper(fk_table) fk_table,
             coalesce(idx_sorting, '') idx_sorting, idx_name,
             upper(fk_field) fk_field, upper(fk_master) fk_master, 1
      from hp_afk_get_table('ALL')
      where idx_name = ''

      union
      select upper(f_master) f_master, upper(fk_table) fk_table,
             coalesce(idx_sorting, '') idx_sorting, idx_name,
             max(upper(fk_field)) fk_field, max(upper(fk_master)) fk_master,
             count(1)
      from hp_afk_get_table('ALL')
      where idx_name <> ''
      group by 1, 2, 3, 4
      into :f_master, :o_fk_table, :idx_sorting, :v_idx_name, :fk_field,
           :fk_master, :v_cnt
  do
  begin

    if (v_cnt != 1) then --несколько полей в индексе
    begin
      v_delim = '';
      fk_master_list = '';
      fk_field_list = '';
      for select upper(fk_master), upper(fk_field)
          from hp_afk_get_table('ALL') ft
          where ft.idx_name = :v_idx_name
          order by field_position
          into :fk_master, :fk_field
          do
          begin
            fk_master_list= fk_master_list || v_delim || fk_master ;
            fk_field_list = fk_field_list || v_delim || fk_field ;
  
            v_delim = ',';
          end
    end

    res = '';
    v_trigger = :o_fk_table || '_AFK_BU';
    v_trigger_m = :f_master || '_AFK_AUD';
    if (not exists(select 1
                   from rdb$dependencies rd
                   where rd.rdb$dependent_name = :v_trigger)) then
      res = '1. Нет триггера на таблицу "' || o_fk_table || '"';
    else
    if (not exists(select 1
                   from rdb$dependencies rd
                   where rd.rdb$dependent_name = :v_trigger and
                         rdb$depended_on_name = :f_master and
                         rdb$field_name = :fk_master)) then
      res = '2. Нет в триггере ссылки на мастер таблицу "' || f_master || '" или поле "' || fk_master || '"';
    else
    if (not exists(select 1
                   from rdb$dependencies rd
                   where rd.rdb$dependent_name = :v_trigger and
                         rdb$depended_on_name = :o_fk_table and
                         rdb$field_name = :fk_field)) then
      res = '3. Нет в триггере ссылки на FK таблицу или поле "' || fk_field || '"';
    else
    if (v_cnt = 1 and  --одно поле индексе
        not exists(select 1
                   from rdb$indices ri
                   inner join rdb$index_segments ids on ids.rdb$index_name = ri.rdb$index_name
                   where ri.rdb$relation_name = :o_fk_table and
                         ids.rdb$field_name = :fk_field and
                         ids.rdb$field_position = 0)) then
      res = '4. Нет индекса на FK таблицу по полю "' || fk_field || '"';
    else
    if (v_cnt != 1) then --несколько полей в индексе
    begin
      v_bool = false;
      for select list(trim(ids.rdb$field_name)), max(
                 case ri.rdb$index_type
                   when 0 then 'Ascending'
                   when 1 then 'Descending'
                 end) rdb$index_type
          from rdb$indices ri
          inner join rdb$index_segments ids on ids.rdb$index_name = ri.rdb$index_name
          where ri.rdb$relation_name = :o_fk_table

          group by ri.rdb$index_name
          having count(1) > (:v_cnt - 1)
          into :v_idx_exists, :v_idx_sort_exists
      do
      begin
        if (v_idx_exists like fk_field_list || '%' and
            v_idx_sort_exists = idx_sorting) then
        begin
          v_bool = true;
          break;
        end
      end
      if (not v_bool) then
        res = '4. Нет индекса на FK таблицу по полям "' || fk_field_list || '"';
    end

    if (res = '') then
    begin
      if (v_cnt = 1  --проверка индексов только с одним полем(с более 1 проверяется в 4 пункте)
         and not exists(select 1
                     from rdb$indices ri
                     inner join rdb$index_segments ids on ids.rdb$index_name = ri.rdb$index_name
                     where ri.rdb$relation_name = :o_fk_table and
                           ids.rdb$field_name = :fk_field and
                           ids.rdb$field_position = 0 and
                            case coalesce(ri.rdb$index_type, 0)
                              when 0 then 'Ascending'
                              else 'Descending'
                            end = :idx_sorting)) then
        res = '5. Индекс по полю "' || fk_field || '" не в том направлении сортировки';
      else
      if (not exists(select 1
                     from rdb$dependencies rd
                     where rd.rdb$dependent_name = :v_trigger_m)) then
        res = '6. Нет триггера "' || v_trigger_m || '"';
      else
      if (not exists(select 1
                     from rdb$dependencies rd
                     where rd.rdb$dependent_name = :v_trigger_m and
                           rdb$depended_on_name = 'HP_AFK_CHECK')) then
        res = '7. Нет в триггере "' || v_trigger_m || '" ссылки на процедуру "HP_AFK_CHECK"';
      else
      if (not exists(select 1
                     from rdb$dependencies rd
                     where rd.rdb$dependent_name = :v_trigger_m and
                           rdb$depended_on_name = :f_master and
                           rd.rdb$field_name = :fk_master)) then
        res = '8. Нет в триггере "' || v_trigger_m || '" ссылки на таблицу "' || f_master || '" по полю "' || fk_master || '"';
      else
        res = 'Ok';
    end
    if (res != 'Ok') then --если не хорошо, то пробел впереди для сортировки в начало
      o_fk_table = ' ' || o_fk_table;

    suspend;
  end
end
