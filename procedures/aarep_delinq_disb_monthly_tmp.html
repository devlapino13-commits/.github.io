CREATE PROCEDURE AAREP_DELINQ_DISB_MONTHLY_TMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DELINQ_DAY INTEGER)
RETURNS (
    DISB_PERIOD INTEGER,
    PRODUCT_KIND VARCHAR(100),
    MONTH_ID INTEGER,
    DEL_BALANCE DECIMAL(18,4),
    DEL_COUNT INTEGER)
AS
declare variable dd_month_end date;
declare variable k_disb_date date;
declare variable k_balance decimal(18,4) = 0;
declare variable k_clients integer;
declare variable oks_delinq_date date;
declare variable oks_date_begin date;
declare variable oks_date_end_orig date;
declare variable oks_date_begin_orig date;
declare variable oks_date_end date;
declare variable k_id decimal(18,0);
begin
  product_kind = '';
  for select ks_from_k_id, ks_date_begin, k_amount_approved, k_kolich_client_approved, pr_common_name
      from kontstate, kontract, product
      where ks_from_k_id = k_id and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_ol_is_delete = 0 and
            k_from_pr_id = pr_id
      into :k_id, :k_disb_date, :k_balance, :k_clients, :product_kind
  do
  begin
    disb_period = extract(year from k_disb_date) * 100 + extract(month from k_disb_date);
    del_balance = k_balance;
    del_count = k_clients;
    month_id = -1; -- сумма выдачи
    suspend;

    for select oks_balance, oks_delinq_date, oks_date_begin, oks_date_end
        from oksummary
        where oks_from_k_id = :k_id and
              oks_balance > 0 and
              oks_date_begin <= :rep_date_end and
              oks_close_state = 0
        into :del_balance, oks_delinq_date, oks_date_begin, oks_date_end
    do
    begin
      oks_date_end_orig = oks_date_end;
      oks_date_begin_orig = oks_date_begin;
      if (oks_date_end > rep_date_end) then
        oks_date_end = dateadd(month, 1, rep_date_end);
      while ((extract(year from oks_date_end) * 100 + extract(month from oks_date_end)) > (extract(year from oks_date_begin) * 100 + extract(month from oks_date_begin))) do
      begin
        dd_month_end = dateadd(month, 1, oks_date_begin);
        dd_month_end = dd_month_end - extract(day from dd_month_end);
        if (dd_month_end > rep_date_end) then
          dd_month_end = rep_date_end;
        if (oks_delinq_date < (dd_month_end - rep_delinq_day)) then
        begin
          month_id = extract(year from dd_month_end) * 12 + extract(month from dd_month_end) - extract(year from k_disb_date) * 12 - extract(month from k_disb_date);
          if (month_id <= 0) then
            month_id = 1;
          if (month_id <= 18) then
            suspend;
        end
        oks_date_begin = dateadd(month, 1, oks_date_begin);
      end
    end
    if ((del_balance > 0) and
        (oks_date_begin_orig <= rep_date_end) and
        (rep_date_end <= oks_date_end_orig)) then
    begin
      month_id = 0; -- активный баланс
      suspend;

      if ((oks_delinq_date + rep_delinq_day) < rep_date_end) then
      begin
        month_id = 999; -- просроченный баланс на конец периода
        suspend;
      end
    end
  end
end
