CREATE PROCEDURE FSROBOT_MAILING_LO_REPAY_OK(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable ol_date date;
declare variable kol integer;
declare variable date_format_str varchar(20);
declare variable k_id decimal(18,0);
declare variable amount decimal(18,4) = 0;
declare variable doc_no varchar(20);
declare variable amount_str varchar(20);
declare variable reg_id integer;
declare variable cm_database_id integer;
declare variable amount_sym varchar(10);
declare variable ks_state integer;
declare variable oks_delinq_date_str varchar(20);
declare variable kont_last_date date;
declare variable oks_delinq_date date;
declare variable repay_date_str varchar(10);
declare variable kf_id decimal(18,0);
declare variable lo_id decimal(18,0);
declare variable kf_name varchar(100);
declare variable lo_fio varchar(255);
declare variable lo_sex integer;
declare variable repay_kind varchar(255);
declare variable ol_parent_id decimal(18,0);
declare variable ol_doc_kind integer;
declare variable ol_doc_client_name varchar(255);
declare variable repay_reg_name varchar(100);
declare variable ol_datetime timestamp;
declare variable repay_time varchar(20);
declare variable ol_document_client decimal(18,0);
declare variable repay_fio varchar(100);
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь

-- СМС после погашения кредита
  if (user_fio is null) then
    user_fio = '';
  res_maillist = :rep_email;

  :cm_database_id = f_get_db_id();
  amount_sym = (case
                  when (cm_database_id = 81001) then 'p.'
                  when (cm_database_id in (39801, 41701)) then 'c.'
                  when (cm_database_id = 94401) then 'AZN'
                end);

  res_body = '';
  kol = 0;

  for select distinct ok_from_k_id, k_from_kf_id, ok_ol_date, iif(coalesce(ol_doc_no, '0') = '0', 'Б/Н', ol_doc_no),
                      ol_from_reg_id, k_kredline_end, ol_datetime, ol_parent_id, ol_doc_kind, ol_doc_client_name,
                      ol_document_client
      from operkred, operlist, kontract
      where ok_from_ol_id = :rep_params and
            ol_id = :rep_params and
            ol_oper_type = 1 -- Погашение
            and
            ol_oper_subtype = 2 -- Погашение
            and
            ok_ol_is_active = 1 -- Действующая проводка
            and
            ok_type in (4, 9) -- Погашение
            and
            ok_from_k_id = k_id
      into :k_id, :kf_id, :ol_date, doc_no, reg_id, :kont_last_date, :ol_datetime, ol_parent_id, ol_doc_kind,
           ol_doc_client_name, ol_document_client
  do
  begin
    repay_time = right('0' || extract(day from ol_datetime), 2) || '.' || right('0' || extract(month from ol_datetime), 2) || '.' || extract(year from ol_datetime) || ' ' || extract(hour from ol_datetime) || ':' || extract(minute from ol_datetime);

    select kal_new_kf_name, kal_new_kf_officer
    from knaktlist
    where kal_date_end >= '31.12.2999' and
          knaktlist.is_delete = 0 and
          kal_from_kf_id = :kf_id and
          kal_date <= :ol_date
    into :kf_name, :lo_id;

    res_c_id = lo_id;

    if (lo_id <= 1000000000004) then
      exit; -- для виртуальных сотрудников не отправляем почту
    if (not exists(select 1
                   from hr_pstatus
                   where ps_from_c_id = :lo_id and
                         ps_date_begin <= :ol_date and
                         ps_date_end >= :ol_date and
                         ps_status = 1 and
                         ps_is_working = 1)) then
      exit; -- Также не отправляем уволенным

    select c_name_fio, c_sex,
           (select cont_value
            from hp_contdata_get_email_valid(0, :lo_id))
    from clients
    where c_id = :lo_id
    into :lo_fio, lo_sex, :res_maillist;

    if (coalesce(res_maillist, '') = '') then
      exit;
    if (res_maillist = 'mbf@molbulak.com') then
      exit;

    if (lo_sex = 1) then
      res_body = 'Уважаемый ';
    else
      res_body = 'Уважаемая ';
    res_body = BLOB_APPEND(res_body, '<b>' || lo_fio || '</b>,<br><br>');

    repay_kind = '';
    ol_parent_id = null;
    if (ol_parent_id is not null) then
    begin
      select ol_from_reg_id, ol_doc_kind, ol_doc_client_name, ol_document_client
      from operlist
      where ol_id = :ol_parent_id
      into :reg_id, :ol_doc_kind, ol_doc_client_name, ol_document_client;

      select esm_project
      from extsysmain, extsysoper
      where eso_from_esm_id = esm_id and
            eso_from_ol_id = :ol_parent_id
      into :repay_kind;

      if (repay_kind is null) then
        repay_kind = '';
    end

    repay_fio = '';
    if (ol_document_client is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ol_document_client
      into :repay_fio;
    end
    if (repay_kind = '') then
    begin
      if (ol_doc_kind = 1) then
        repay_kind = 'Касса';
      else
        repay_kind = 'Банк';
    end
    if (coalesce(ol_doc_client_name, '') <> '') then
      repay_kind = repay_kind || ', ' || ol_doc_client_name;
    else
      repay_kind = repay_kind || ', клиент клиент платил сам.';

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into :repay_reg_name;

    date_format_str = right('0' || extract(day from ol_date), 2) || '.' || right('0' || extract(month from ol_date), 2) || '.' || extract(year from ol_date);
    repay_date_str = date_format_str;

    select -sum(ok_os + ok_perc + ok_fine + ok_kommis_adv + ok_tax)
    from operkred
    where ok_from_ol_id = :rep_params and
          ok_type in (4, 9)
    into :amount;
    if (amount is null) then
      amount = 0;
    amount_str = round(amount, 0);
    if (char_length(amount_str) > 6) then
      amount_str = left(amount_str, char_length(amount_str) - 6) || ',' || left(right(amount_str, 6), 3) || ',' || right(amount_str, 3);
    else
    if (char_length(amount_str) > 3) then
      amount_str = left(amount_str, char_length(amount_str) - 3) || ',' || right(amount_str, 3);
    amount_str = amount_str || amount_sym;
    kol = kol + 1;

-- инфо о транзакции
    res_subject = 'Погашение кредита, ' || kf_name;
    res_body = BLOB_APPEND(res_body, 'Поступлило погашение от Вашей группы.<br>');
    res_body = BLOB_APPEND(res_body, 'Номер группы <b>' || :kf_name || '</b><br>');
    res_body = BLOB_APPEND(res_body, 'Сумма: <b>' || amount_str || '</b><br>');
    res_body = BLOB_APPEND(res_body, '№ квитанции: <b>' || doc_no || '</b> от <b>' || :date_format_str || '</b> г.<br>');
    res_body = BLOB_APPEND(res_body, 'Место погашения: <b>' || repay_reg_name || '</b><br>');
    res_body = BLOB_APPEND(res_body, 'Клиент: <b>' || repay_fio || '</b><br>');

    res_body = BLOB_APPEND(res_body, '<b>' || repay_kind || '</b><br>');
    res_body = BLOB_APPEND(res_body, 'Время: <b>' || repay_time || ' Мск.</b><br>');

-- инфо о состоянии
    ks_state = null;
    select ks_state
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999'
    into :ks_state;

    if (ks_state = 1) then
    begin
-- Кредит все еще активный, показываем состояние
      select oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :ol_date and
            oks_date_end >= :ol_date and
            oks_close_state = 0
      into :oks_delinq_date;

      oks_delinq_date_str = right('0' || extract(day from oks_delinq_date), 2) || '.' || right('0' || extract(month from oks_delinq_date), 2) || '.' || extract(year from oks_delinq_date);

      if (oks_delinq_date > ol_date) then
      begin
        res_body = BLOB_APPEND(res_body, '<br>Просроченной задолженности нет. Следующий платеж: <b>' || oks_delinq_date_str || '</b> г.<br>');
      end
      else
      begin
        amount = 0;
        if (kont_last_date > ol_date) then
        begin
          select sum(zad_os + zad_perc + sh_calc_fine)
          from aarep_make_zad_list_temp(:k_id, :ol_date, 0)
          where sh_date <= :ol_date
          into :amount;
        end
        else
        begin
          select sum(zad_os + sh_calc_perc + sh_calc_fine)
          from aarep_make_zad_list_temp(:k_id, :ol_date, 1)
          where sh_date <= :ol_date
          into :amount;
        end
        if (amount is null) then
          amount = 0;
        if (amount > 0) then
        begin
          amount_str = round(amount, 0);
          if (char_length(amount_str) > 6) then
            amount_str = left(amount_str, char_length(amount_str) - 6) || ',' || left(right(amount_str, 6), 3) || ',' || right(amount_str, 3);
          else
          if (char_length(amount_str) > 3) then
            amount_str = left(amount_str, char_length(amount_str) - 3) || ',' || right(amount_str, 3);
          amount_str = amount_str || amount_sym;

          res_body = BLOB_APPEND(res_body, '<br>Остается текущий долг. На <b>' || repay_date_str || '</b> г. сумма: <b>' || amount_str || ' р.</b><br>');
        end
        else
          res_body = BLOB_APPEND(res_body, '<br>Задолженности нет.<br>');
      end
    end

    if (ks_state = 2) then
    begin
-- Кредит закрыт
      res_body = BLOB_APPEND(res_body, '<br>Заем полностью погашен. <b>Спасибо!</b><br>');
    end
    suspend;

  end
end
