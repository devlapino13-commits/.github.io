      aarep_active_client_percent = cast((100 * cast(aarep_active_client_end as decimal(15,2)) / (cast(aarep_active_client_begin as decimal(15,2)))) - 100 as decimal(15,1));
    else
      aarep_active_client_percent = 0;

      aarep_risk_begin = cast((100 * aarep_deliq_balance_begin / aarep_active_port_begin) as decimal(15,2));
    else
      aarep_risk_begin = 0;

    if (aarep_deliq_balance_end <> 0) then
      aarep_risk_end = cast((100 * aarep_deliq_balance_end / aarep_active_port_end) as decimal(15,2));

  aarep_disb_kolich = aarep_disb_kolich1;
