CREATE PROCEDURE CC_DO_ADD_CLIENT(
    WMA_ID D_ID,
    UN_ID ID_TYPE,
    C_NAME_FIO D_STR_100,
    C_SEX D_SMALLINT,
    C_BIRTH_DATE D_DATE,
    C_NATION D_INTEGER,
    C_CITIZEN D_INTEGER,
    C_PHONE_NO D_STR_CODE,
    C_ADD_PHONE_NO D_STR_CODE,
    C_INN D_STR_20 = null)
RETURNS (
    C_ID D_ID)
AS
declare variable error_text d_str_255;
begin
  wma_id = coalesce(wma_id, 0);

  if (wma_id <> 0) then
  begin
      select wmat_name, wmat_gender, wmat_birthday, wmat_language, wmat_citizenship, wma_phone_no, wmat_phone_add,
             wma_last_un_id, wmat_inn
      from cc_website_miniapp, cc_website_miniapp_tmp
      where wma_id = :wma_id and
            wma_last_wmat_id = wmat_id
      into c_name_fio, c_sex, c_birth_date, c_nation, c_citizen, c_phone_no, c_add_phone_no, un_id, c_inn;
  end

  if (coalesce(c_name_fio, '') = '' or
      c_birth_date is null or
      coalesce(c_phone_no, '') = '' or
      un_id is null) then
    exception e_default 'Не хватает значений для создания физлица';

  select c.c_id
  from clients c
  where c.c_inn = :c_inn
  into :c_id;

  if (coalesce(c_id, 0) = 0) then
  begin
    select result
    from get_kv_id(2)
    into :c_id;

    insert into clients (c_id, c_type, c_name_fio, c_sex, c_birth_date, c_nation, c_citizen, c_from_reg_id, c_from_un_id, c_source_system, c_inn)
    values (:c_id, 0, :c_name_fio, :c_sex, :c_birth_date, :c_nation, :c_citizen, iif(f_get_db_id() = 41701, 1090100, 1010100), :un_id, 3, :c_inn);
  end
  -- Сохранение номера телефона
  begin
      if (coalesce(c_phone_no, '') <> '') then
        execute procedure hp_continfo_update(:c_id, 'contactinfo.cell_main', :c_phone_no, 0, :un_id);
      when any do
        error_text = 'не удалется сохранить основной номер телефона';
  end
  begin
      if (coalesce(c_add_phone_no, '') <> '') then
        execute procedure hp_continfo_update(:c_id, 'contactinfo.cell_add', :c_add_phone_no, 0, :un_id);
      when any do
        error_text = 'не удалется сохранить дополнительный номер телефона';
  end
  suspend;
end
