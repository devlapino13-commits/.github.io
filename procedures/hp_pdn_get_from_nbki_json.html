declare function my_strtonum (
    num_str d_str_code)
returns d_double
as
begin
  return replace(num_str, ',', '.');
  when any do
    return 0;
end

declare function my_get_last_el(jso_in d_integer)
returns d_integer
as
  declare variable cnt d_integer;
  declare variable i d_integer;
  declare variable jso d_integer;
  declare variable res d_integer;
  declare variable res_tmp d_integer;
  declare variable max_date d_date;
  declare variable max_date1 d_date;
  declare variable cur_date d_date;
  declare variable cur_date1 d_date;
begin
    res = null;
    if (jso_in is not null) then
    begin
    cnt = json.count_get(jso_in);

    i = 0;
     -- '{}';
    max_date = '01.01.1900';

    while (i < cnt) do
    begin
      jso = json.arr_get_obj(jso_in, i);
      res_tmp = jso;

      cur_date = json.obj_get_val(jso, 'calc_date', current_date);
      cur_date1 = json.obj_get_val(jso, 'last_updated_dt', current_date);

      if (max_date < cur_date) then
      begin
        max_date = cur_date;
        max_date1 = cur_date1;
        res = res_tmp;
      end
      else if (max_date = cur_date and max_date1 <= cur_date1) then
      begin
        max_date1 = cur_date1;
        res = res_tmp;
      end

      i = i + 1;
    end 
    end
    if (res is null) then res = json.obj_create();
    return res;
end

declare function f_my_obj_get_arr(handle d_integer, path d_str_long)
returns d_integer
as
  declare variable jsa d_integer;
  declare variable val_type d_str_code;
begin
  if (json.contains(handle, path) = 0) then
    return null;

  val_type = json.val_get_type_name(handle, path);

  if (val_type = 'Array') then
    return json.obj_get_arr(handle, path);
  else if (val_type = 'Object') then
  begin
    handle = json.obj_get_obj(handle, path);
    jsa = json.arr_create();
    json.arr_set_obj(jsa, -1, handle);

    return jsa;
  end
  else
    exception e_default path || ' is not Array/Object';
end

begin
---------------------Начало------------------
  jsa_accaunts = coalesce(json.obj_get_arr(jso_full, 'preply.report.account_reply_rutdf'), json.arr_create());

  loan_count = 0;
  fact_rec_no = 0;

  loan_count = json.count_get(jsa_accaunts);

  while (fact_rec_no < loan_count) do
  begin

    credpartnertype = null;
    business_category = null;

    jso_loan = coalesce(json.arr_get_obj(jsa_accaunts, fact_rec_no), json.arr_create());
    fact_rec_no = fact_rec_no + 1;
    credUid = json.obj_get_val(jso_loan, 'uuid', null);
    business_category =  json.obj_get_val(jso_loan, 'business_category', null);                                                  -- Тип партнера

    jso_deal = my_get_last_el(json.obj_get_arr(jso_loan, 'trade'));

    jso_guarantees = json.obj_get_arr(jso_loan, 'guarantor');

    jso_payment = my_get_last_el(json.obj_get_arr(jso_loan, 'payment'));
    credStartDebtDate = json.obj_get_val(jso_loan, 'fund_date', null);
    credupdate_tmp = json.obj_get_val(jso_loan, 'last_updated_dt', f_datetostr(rep_date));
    jso_credFullCost = my_get_last_el(json.obj_get_arr(jso_loan, 'overall_val'));

                          when 'KPK' then 6

                          when 'OIP' then 16
                          when 'RETAIL' then 10
                          when 'STRAKH' then 11
                          when 'FINANCE' then 13
                          else 99
                        end;


    if (coalesce(credUid, '') <> '') then
    begin
      if (exists(select 1 from temp_pdn_uids where uid_str = :credUid)) then
        continue;
      else
        insert into temp_pdn_uids (uid_str) values (:credUid);

    if (credenddatefact = '01.01.1900') then credenddatefact = null;

    credDate = replace(json.obj_get_val(jso_deal, 'opened_dt', '01.01.1900'), '-', '01.01.1900');

    credEnddate = replace(json.obj_get_val(jso_deal, 'close_dt', '31.12.2999'), '-', '31.12.2999'); -- Дата прекращения обязательства субъекта по условиям сделки

    i = 0;

      credGuarantee = 1;

      if (credGuaranteeFactEndDate is null or coalesce(credGuaranteeFactEndDate, '01.01.1900') < credGuaranteeFactEndDate_tmp) then

    first_sum = 0;

    debt_perc_sum = replace(replace(json.obj_get_val(jso_debtCurrent, 'int_outstanding', 0), ',', '.'), '-', 0);

    debt_sum = replace(replace(json.obj_get_val(jso_debtCurrent, 'amt_outstanding', 0), ',', '.'), '-', 0);

    if (debt_sum = 0) then

    credSumDebt = debt_op_sum;
    fullCurrentDebt = debt_sum;
    credUpdate = json.obj_get_val(jso_debtCurrent, 'calc_date', credupdate_tmp);

    if (credSum = 0) then
      credSum = maxvalue(debt_op_sum, first_sum);

    debt_op_sum = replace(replace(json.obj_get_val(jso_debtOverdue, 'principal_amt_past_due', 0), ',', '.'), '-', 0);
    debt_perc_sum = replace(replace(json.obj_get_val(jso_debtOverdue, 'int_amt_past_due', 0), ',', '.'), '-', 0);
    debt_other_sum = replace(replace(json.obj_get_val(jso_debtOverdue, 'other_amt_past_due', 0), ',', '.'), '-', 0);

    if (debt_sum = 0) then

    credSumOverdue = debt_sum;

    if (credDayOverdue < 0) then

    -- + если несколько действующих, показываем с большей суммой, далее с более поздней датой начала
    credCollateralItemType = null;

    credCollateralFactEndDate_tmp = null;
    credCollateralDate_tmp = null;
    credCollateralDate = null;

    i = 0;
    cnt = json.count_get(jso_collaterals);
    while (i < cnt) do
    begin
      jso_collateral = json.arr_get_obj(jso_collaterals, i);
      credCollateralSum_tmp = my_strtonum(json.obj_get_val(jso_collateral, 'collateral_value', 0));
      credCollateralDate_tmp = replace(json.obj_get_val(jso_collateral, 'collateral_date', '01.01.1900'), '-', '01.01.1900');

        credCollateralFactEndDate_tmp = replace(json.obj_get_val(jso_collateral, 'collateral_expiration_date', '01.01.1900'), '-', '01.01.1900');

      if (rep_date between credCollateralDate_tmp and credCollateralFactEndDate_tmp
        and (credCollateralDate is null
            or credCollateralSum_tmp > credCollateralSum
            or (credCollateralSum_tmp = credCollateralSum and credCollateralDate_tmp > credCollateralDate)
            )
          )
      then
      begin
        credCollateral = 1;
        credCollateralDate = credCollateralDate_tmp;
        credCollateralFactEndDate = credCollateralFactEndDate_tmp;
        credCollateralSum = credCollateralSum_tmp;
        credCollateralItemType = json.obj_get_val(jso_collateral, 'collateral_code', null);      --- Код предмета залога
      end

      i = i + 1;
    end 

    minSumPayCc = my_strtonum(json.obj_get_val(jso_paymentTerms, 'min_paymt', -777));   -- Сумма минимального платежа по кредитной карте
    percentEndDate = replace(json.obj_get_val(jso_paymentTerms, 'interest_payment_due_date', '01.01.1900'), '-', '01.01.1900'); -- Дата окончания срока уплаты процентов

    if (credCollateralFactEndDate = '01.01.1900') then credCollateralFactEndDate = null;
    if (percentEndDate = '01.01.1900') then percentEndDate = null;
    if (minSumPayCc = -777) then minSumPayCc = null;

    suspend;
    end
