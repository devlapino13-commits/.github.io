Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_ACCESSCASHROOM_PRINT(
    OBJ_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    ACR_ID D_INTEGER,
    ACR_DATE D_DATE,
    ACR_OFFICE_ID D_INTEGER,
    OFFICE_NAME D_STR_CODE,
    ACR_REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    ACR_START_DATE D_DATE,
    ACR_END_DATE D_DATE,
    ACR_BASIS_ID D_INTEGER,
    BASIS D_STR_255,
    ACR_STATUS D_SMALLINT,
    STATUS D_STR_CODE,
    ACR_EDL_ID D_ID,
    RECEIVER_LST D_STR_8100,
    ACR_ISSUER_ID D_ID,
    ISSUER_NAME D_STR_100,
    ACR_CASHIER_ID D_ID,
    CASHIER_NAME D_STR_100,
    ACR_PERSON_ID D_ID,
    PERSON_NAME D_STR_100,
    ACR_DIRECTOR_ID D_ID,
    DIRECTOR_NAME D_STR_100,
    ISSUER_LOGIN D_STR_CODE,
    CASHIER_LOGIN D_STR_CODE,
    PERSON_LOGIN D_STR_CODE,
    DIRECTOR_LOGIN D_STR_CODE,
    ISSUER_QR D_STR_CODE,
    CASHIER_QR D_STR_LONG,
    PERSON_QR D_STR_LONG,
    DIRECTOR_QR D_STR_LONG,
    ISSUER_SIGN D_INTEGER,
    CASHIER_SIGN D_INTEGER,
    PERSON_SIGN D_INTEGER,
    DIRECTOR_SIGN D_INTEGER,
    ACR_ISSUER_DATESIGN D_DATETIME,
    ACR_CASHIER_DATESIGN D_DATETIME,
    ACR_PERSON_DATESIGN D_DATETIME,
    ACR_DIRECTOR_DATESIGN D_DATETIME,
    ISSUER_JT D_STR_100,
    CASHIER_JT D_STR_100,
    CASHIER_REG D_STR_CODE,
    PERSON_JT D_STR_100,
    DIRECTOR_JT D_STR_100)
AS
declare variable SPLITTER D_STR_LANG;
declare variable DTSIGN D_DATETIME;
begin
  for
    select
      acr_id,
      acr_date,
      acr_office_id,
      rg1.reg_name,
      acr_reg_id,
      rg2.reg_name,
      acr_start_date,
      acr_end_date,
      acr_basis_id,
      rd_name,
      acr_status,
      case acr_status
        when -1 then 'аннулирован'
        when 0 then 'создан'
        when 1 then 'завизирован'
        when 2 then 'частично подписан'
        when 3 then 'подписан'
      end,
      acr_edl_id,
      acr_issuer_id,
      cl1.c_name_fio,
      acr_cashier_id,
      cl2.c_name_fio,
      acr_person_id,
      cl3.c_name_fio,
      acr_director_id,
      cl4.c_name_fio,
      iif(acr_issuer_datesign is null, 0, 1),
      iif(acr_cashier_datesign is null, 0, 1),
      iif(acr_person_datesign is null, 0, 1),
      iif(acr_director_datesign is null, 0, 1),
      acr_issuer_datesign,
      acr_cashier_datesign,
      acr_person_datesign,
      acr_director_datesign
    from accesscashroom
      inner join region as rg1
        on rg1.reg_unicode = acr_office_id
      inner join region as rg2
        on rg2.reg_unicode = acr_reg_id
      inner join refdata
        on rd_id = acr_basis_id
      inner join clients cl1
        on cl1.c_id = acr_issuer_id
      inner join clients cl2
        on cl2.c_id = acr_cashier_id
      inner join clients cl3
        on cl3.c_id = acr_person_id
      inner join clients cl4
        on cl4.c_id = acr_director_id
    where acr_id = :obj_id
  into
    :acr_id,
    :acr_date,
    :acr_office_id,
    :office_name,
    :acr_reg_id,
    :reg_name,
    :acr_start_date,
    :acr_end_date,
    :acr_basis_id,
    :basis,
    :acr_status,
    :status,
    :acr_edl_id,
    :acr_issuer_id,
    :issuer_name,
    :acr_cashier_id,
    :cashier_name,
    :acr_person_id,
    :person_name,
    :acr_director_id,
    :director_name,
    :issuer_sign,
    :cashier_sign,
    :person_sign,
    :director_sign,
    :acr_issuer_datesign,
    :acr_cashier_datesign,
    :acr_person_datesign,
    :acr_director_datesign
  do
  begin
    splitter = ';' || ascii_char(13) || ascii_char(13);

    if (acr_edl_id > 0) then
    begin
      dtsign = (
        select eds_status_datetime
        from edoc_sign
        where eds_from_edl_id = :acr_edl_id
          and eds_from_c_id = :acr_cashier_id
          and eds_status = 1
          and eds_enum_edoc_sign_kind <= 1);
      if (dtsign is not null) then
      begin
        acr_cashier_datesign = dtsign;
        cashier_sign = 2;
        cashier_qr = f_edoc_qr_by_edl(:acr_edl_id, :acr_cashier_id);
      end

      dtsign = (
        select eds_status_datetime
        from edoc_sign
        where eds_from_edl_id = :acr_edl_id
          and eds_from_c_id = :acr_person_id
          and eds_status = 1
          and eds_enum_edoc_sign_kind <= 1);
      if (dtsign is not null) then
      begin
        acr_person_datesign = dtsign;
        person_sign = 2;
        person_qr = f_edoc_qr_by_edl(:acr_edl_id, :acr_person_id);
      end

      dtsign = (
        select eds_status_datetime
        from edoc_sign
        where eds_from_edl_id = :acr_edl_id
          and eds_from_c_id = :acr_director_id
          and eds_status = 1
          and eds_enum_edoc_sign_kind <= 1);
      if (dtsign is not null) then
      begin
        acr_director_datesign = dtsign;
        director_sign = 2;
        director_qr = f_edoc_qr_by_edl(:acr_edl_id, :acr_director_id);
      end
    end

    issuer_login = (
      select first 1 un_name
      from username
      where un_from_c_id = :acr_issuer_id
      order by un_id desc);
    cashier_login = (
      select first 1 un_name
      from username
      where un_from_c_id = :acr_cashier_id
      order by un_id desc);
    person_login = (
      select first 1 un_name
      from username
      where un_from_c_id = :acr_person_id
      order by un_id desc);
    director_login = (
      select first 1 un_name
      from username
      where un_from_c_id = :acr_director_id
      order by un_id desc);

    issuer_jt = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :acr_issuer_id
        and :acr_date between ps_date_begin and ps_date_end
        and ps_is_working = 1);

    select
      jl_name,
      reg_name
    from hr_pstatus
      inner join joblist
        on jl_id = ps_from_jl_id
      inner join region
        on reg_unicode = ps_from_reg_id
    where ps_from_c_id = :acr_cashier_id
      and :acr_date between ps_date_begin and ps_date_end
      and ps_is_working = 1
    into
      :cashier_jt,
      :cashier_reg;

    person_jt = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :acr_person_id
        and :acr_date between ps_date_begin and ps_date_end
        and ps_is_working = 1);
    director_jt = (
      select jl_name
      from hr_pstatus
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :acr_director_id
        and :acr_date between ps_date_begin and ps_date_end
        and ps_is_working = 1);

    receiver_lst = (
      select list(
        f_format('%1, паспорт %2 выданный %3 от %4, сроком %5 включительно',
          c_name_fio,
          c_pass_no,
          c_pass_who,
          f_datetostr(c_pass_date),
          'с ' || f_datetostr(:acr_start_date) || ' по ' || f_datetostr(:acr_end_date)), :splitter) || ';'
      from accesscashroomd
        inner join clients
          on c_id = acd_receiver_id
      where acd_acr_id = :obj_id);
    suspend;
  end
end
