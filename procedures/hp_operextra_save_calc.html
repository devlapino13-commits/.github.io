CREATE PROCEDURE HP_OPEREXTRA_SAVE_CALC(
    REP_RD_ID INTEGER,
    OL_PARENT_ID DECIMAL(18,0),
    REP_OBJECT_KIND INTEGER,
    REP_OBJECT_ID DECIMAL(18,0),
    REP_DATE DATE,
    REP_REGION_ID INTEGER,
    REP_ACC_DT VARCHAR(20),
    REP_ACC_KT VARCHAR(20),
    REP_AMOUNT DECIMAL(18,4),
    REP_DETAIL VARCHAR(255),
    REP_OPER_MEMO VARCHAR(1024),
    UN_ID DECIMAL(18,0))
RETURNS (
    OL_ID DECIMAL(18,0))
AS
declare variable db_id integer;
declare variable cm_currency varchar(3);
declare variable last_kn_date d_date;
begin
  ol_id = 0;
  /*   select cm_database_id, cm_default_currency from company
   into :db_id, :cm_currency; */
  select cm_default_currency
  from company
  into :cm_currency;
  db_id = f_get_db_id();

  if (coalesce(ol_parent_id, 0) = 0) then
  begin
    rep_region_id = f_hq_id(db_id);

    rep_amount = 0;

    select res_account, res_acc_delinq
    from hp_make_kredit_acclist(0, :rep_region_id, -:rep_rd_id)
    into :rep_acc_dt, rep_acc_kt;
    ol_parent_id = null;
  end

  select kv
  from get_kv(:rep_region_id, 5)
  into :ol_id;

  insert into operlist (ol_id, ol_parent_id, ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                        ol_memo, ol_status, is_delete, ol_user_id)
  values (:ol_id, :ol_parent_id, :rep_region_id, 2, 29, :rep_date, 9, :rep_oper_memo, 0, 0, :un_id);

  insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa,
                        ob_account_kt, ob_client_kt, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
  values (:ol_id, :cm_currency, 1.0, null, :rep_acc_dt, :rep_amount, :rep_amount, :rep_acc_kt, null, :rep_detail,
          :rep_date, 1, :rep_region_id);

  if (ol_parent_id is not null) then
  begin
    if (db_id = 81001 and
        rep_date >= '01.01.2018') then
    begin
      select min(kc_last_ol_date)
      from knclient
      where kc_from_k_id = :rep_object_id and
            kc_amount_approved > 0
      into last_kn_date;
    end

    insert into operextra (ox_from_ol_id, ox_from_reg_id, ox_ol_date, ox_ol_is_active, ox_object_kind, ox_object_id,
                           ox_from_rd_id, ox_amount_plus, ox_amount_minus, ox_amount_writeoff, ox_detail)
    values (:ol_id, :rep_region_id, :rep_date, 1, :rep_object_kind, :rep_object_id, :rep_rd_id, :rep_amount, 0.0, 0.0,
            :rep_detail);

    if (db_id = 81001 and
        rep_date >= '01.01.2018') then
    begin
      if (last_kn_date is not null and
          last_kn_date = rep_date) then
      begin
        update knclient
        set kc_last_ol_date = :rep_date
        where kc_from_k_id = :rep_object_id and
              kc_amount_approved > 0;
      end
  
      execute procedure hp_do_kn_operbook_one_online(:rep_object_id, :rep_date, :ol_id, null);
    end
  end

  suspend;
end
