CREATE PROCEDURE HP_KONTMATAG_UPDATE(
    K_ID D_ID)
AS
declare variable obj_kind d_integer;
declare variable ag_id d_integer;
begin
  -- Процедура для заполнения (синхронизации) списка соглашений с поставщиками, которые продают товар через договор займа

  -- Отмечаем существующие записи как кандидат на удаление
  update kontmatag
  set kma_state = -1
  where kma_from_k_id = :k_id;

  obj_kind = f_objkind('kontract', 'id');

  for select distinct kml_from_ag_id
  from kontmat
  where kml_object_kind = :obj_kind and
        kml_object_id = :k_id and
        kml_from_ag_id is not null
  into :ag_id
  do
  begin
    -- исключаем из списка удаляемых строк
    update kontmatag
    set kma_state = 0
    where kma_from_k_id = :k_id and
          kma_from_ag_id = :ag_id;

    -- Если данный AG_ID еще не прописан для договора, то добавляем запись
    if (row_count = 0) then
    begin
      insert into kontmatag (kma_from_k_id, kma_from_ag_id)
      values (:k_id, :ag_id);
    end
  end

  -- Удаляем все что осталось как кандидат на удаление
  delete from kontmatag
  where kma_from_k_id = :k_id and
        kma_state = -1;
end
