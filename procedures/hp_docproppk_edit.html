CREATE PROCEDURE HP_DOCPROPPK_EDIT(
    PK_ID D_INTEGER,
    DP_ID D_INTEGER,
    CITIZEN_KIND D_INTEGER,
    VALID_KIND D_INTEGER,
    OLD_GROUP_NO D_INTEGER,
    NEW_GROUP_NO D_INTEGER,
    UN_ID D_ID)
AS
begin
  if (old_group_no is null) then
    old_group_no = 0;

  if (new_group_no is null) then
    new_group_no = 0;

  if (valid_kind = -1) then -- Аннулирование связи
  begin
    update docproppk
    set dpk_is_delete = 1,
        dpk_delete_un_id = :un_id
    where dpk_from_pk_id = :pk_id and
          dpk_from_dp_id = :dp_id and
          dpk_citizen_kind = :citizen_kind and
          dpk_group_no = :old_group_no and
          dpk_is_delete = 0;
  end else
  begin
    -- Добавление связи
    if (not exists(select * from docproppk
                   where dpk_from_pk_id = :pk_id and
                         dpk_from_dp_id = :dp_id and
                         dpk_citizen_kind = :citizen_kind and
                         dpk_valid_kind = :valid_kind and
                         dpk_group_no = :new_group_no and
                         dpk_is_delete = 0)) then
    begin
      update docproppk
      set dpk_valid_kind = :valid_kind,
          dpk_group_no = :new_group_no
      where dpk_from_pk_id = :pk_id and
            dpk_from_dp_id = :dp_id and
            dpk_citizen_kind = :citizen_kind and
            dpk_group_no = :old_group_no and
            dpk_is_delete = 0;
  
      if (row_count = 0) then
      begin
        insert into docproppk (dpk_from_pk_id, dpk_from_dp_id, dpk_citizen_kind, dpk_valid_kind, dpk_group_no, dpk_create_un_id)
        values (:pk_id, :dp_id, :citizen_kind, :valid_kind, :new_group_no, :un_id);
      end
    end
  end
end
