CREATE PROCEDURE PRINT_TARGET_MONITORING_ALL(
    KC_ID D_ID)
RETURNS (
    CLIENT_FIO D_STR_100,
    ANALIZ_DATE D_DATE,
    ANALIZ_LO_FIO D_STR_100,
    FACT_LOAN_TARGET D_STR_255,
    MONITORING_MEMO D_STR_2048,
    KONTRACT_NO D_STR_100,
    FOLDER_NAME D_STR_100,
    PR_NAME D_STR_100,
    LOAN_AMOUNT D_CURRENCY,
    LOAN_TARGET D_STR_255,
    DESCR_OF_LOAN D_STR_LONG,
    SVZ_MEMO D_STR_2048,
    ACTION_DATE D_DATE,
    K_DISB_DATE D_DATE)
AS
declare variable C_ID D_ID;
declare variable RD_ID D_INTEGER;
declare variable KF_ID D_ID;
declare variable OBJ_KIND D_STR_100;
declare variable OBJ_ID D_ID;
declare variable IS_IPF D_BOOL;
declare variable DT_FIND D_DATE;
declare variable DATEX D_DATE;
declare variable KNCLIENT_KIND D_SMALLINT;
declare variable CLIENTS_KIND D_SMALLINT;
declare variable ACTIONLIST_CNT D_INTEGER;
declare variable K_ID D_ID;
declare variable MA_ID D_INTEGER;
begin
  -- Процедура для шаблона отображающего все анкеты ЦМ по knclient
  rd_id = f_get_rd_id('Anketa.TargetUseLoan');
  datex = '03.03.2022';
  knclient_kind = f_objkind('KnClient', 'id');
  clients_kind = f_objkind('Clients', 'id');

  select iif(:datex > k.k_disb_date, kc.kc_from_c_id, kc.kc_id),
         iif(:datex > k.k_disb_date, :clients_kind, :knclient_kind)
  from knclient kc
  inner join kontract k on k.k_id = kc.kc_from_k_id
  where kc.kc_id = :kc_id
  into obj_id, obj_kind;

  select k_disb_date,
         k_kontract_no,
         k_from_kf_id,
         kc_amount_approved,
         c_id,
         c_name_fio,
         pr_name,
         iif(pr_code containing '.ipf.', 1, 0),
         rd_name,
         k_id,
         k_from_ma_app_id
  from knclient
  inner join kontract on k_id = kc_from_k_id
  inner join clients on c_id = kc_from_c_id
  inner join product on pr_id = k_from_pr_id
  left join refdata on rd_id = kc_loan_target
  where kc_id = :kc_id and
      kc_amount_approved > 0
  into :k_disb_date,
       :kontract_no,
       :kf_id,
       :loan_amount,
       :c_id,
       :client_fio,
       :pr_name,
       :is_ipf,
       :loan_target,
       :k_id,
       :ma_id;


    -- RFCKG 2562
    select loan_description
    from hp_get_loan_description(:c_id, :ma_id)
    into :descr_of_loan;


  if (:obj_kind = :knclient_kind) then
  begin
      for select acl.acl_date,
                 acl.acl_last_datetime,
                 c.c_name_fio
          from actionlist acl
          inner join clients c on c.c_id = acl.acl_respons_c_id
          where acl.acl_object_kind = :obj_kind and
                acl.acl_object_id = :obj_id and
                acl.acl_from_rd_id = :rd_id and
                acl.acl_status = 2
          into :action_date,
               :analiz_date,
               :analiz_lo_fio
      do
      begin
        select count(*)
        from actionlist acl
        where acl.acl_from_rd_id = :rd_id and
              acl.acl_object_kind = :obj_kind and
              acl.acl_object_id = :obj_id
        into actionlist_cnt;

        if (actionlist_cnt < 2) then
            analiz_date = current_date;

        if (obj_id = 3929614) then
            descr_of_loan = 'клиент оформляет Смарт телевизор для дома';

        select kal_new_kf_name
        from knaktlist
        where kal_from_kf_id = :kf_id and
              :analiz_date between kal_date and kal_date_end and
              is_delete = 0
        into :folder_name;
    
        select coalesce(res_str_values, '')
        from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'AnketaQuestions.LoanUseFact', :analiz_date, null)
        into :fact_loan_target;
    
        select coalesce(res_str_values, '')
        from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'Monitoring.Memo', :analiz_date, null)
        into :monitoring_memo;
    
        select coalesce(left(obm_memo, 2048), '')
        from hp_objectprops_memo_get('Clients', :c_id, 'Refdata', 'ObjectProps.Client_memo', 1)
        rows 1
        into :svz_memo;

        suspend;
      end
  end
  else if (:obj_kind = :clients_kind) then
  begin
    if (k_disb_date < '14.02.2022') then
      dt_find = k_disb_date + iif(k_disb_date >= '22.12.2021' and is_ipf = 1, 1, 8);
    else
      dt_find = k_disb_date + 1;

    for select acl.acl_date,
               acl.acl_last_datetime,
               c.c_name_fio
        from actionlist acl
        inner join clients c on c.c_id = acl.acl_respons_c_id
        where acl_object_kind = f_objkind(:obj_kind, 'id') and
              acl_object_id = :obj_id and
              acl_from_rd_id = :rd_id and
              acl_date = :dt_find and
              acl_status = 2
        order by acl_id desc
        into :action_date,
             :analiz_date,
             :analiz_lo_fio
    do
    begin
        select kal_new_kf_name
        from knaktlist
        where kal_from_kf_id = :kf_id and
              :analiz_date between kal_date and kal_date_end and
              is_delete = 0
        into :folder_name;
    
        select coalesce(res_str_values, '')
        from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'AnketaQuestions.LoanUseFact', :analiz_date, null)
        into :fact_loan_target;
    
        select coalesce(res_str_values, '')
        from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'Monitoring.Memo', :analiz_date, null)
        into :monitoring_memo;
    
        select coalesce(left(obm_memo, 2048), '')
        from hp_objectprops_memo_get('Clients', :c_id, 'Refdata', 'ObjectProps.Client_memo', 1)
        rows 1
        into :svz_memo;

        suspend;
    end
  end 
end
