CREATE PROCEDURE FM_HTML_PURCHASE_ACTS(
    K_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable obj_kind d_str_code;
declare variable seller_fio d_str_100;
declare variable ol_id d_id;
begin

  res_text = (select res_style
              from fm_html_styles) || '<body><h2>Печать закупочных актов</h2>' || '<table width="100%"><col width="60%"><col width="40%">' || f_format('<tr><th>%1</th>', 'ФИО продавца') || f_format('<th>%1</th></tr>', 'Ссылка для печати акта');

  for select distinct(c.c_name_fio), ol.ol_id
      from kontmat kml
      inner join operlist ol on ol.ol_id = kml.kml_bill_id
      inner join clients c on ol.ol_document_client = c.c_id
      where kml.kml_object_kind = 9 and
            kml.kml_object_id = :k_id
      into :seller_fio, :ol_id

  do
  begin
    res_text = BLOB_APPEND(res_text, f_html_table_row(:seller_fio, f_uri_blank('Распечатать закупочный акт', 'print', f_objkind('kontract', 'id'), :ol_id, 1579, 'Печать закупочного акта', 0, 0, ''), '', '', -1, -1, -1, -1, 2, ''));
  end

  suspend;
end
