CREATE PROCEDURE MC_DOC_NAME_CHANGE(
    PASSKIND_ID D_INTEGER,
    PASSKIND_NAME D_STR_100)
RETURNS (
    RES_VALUE D_STR_100)
AS
declare variable LEN D_INTEGER;
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();
  -- процедура отвечает за подмену наизменования документа в соответсиви с пиьмом от Зары
  /*
    (Sent: Tuesday, June 2, 2020 2:42 PM)
    Тема: КМП документы клиента (заявка, профиль)

    Документы предлагаю назвать так:

    1.    «Временную регистрацию, физ. лицо» и «Временная рег, юр. лицо» объединить в одно: «Временная регистрация в РФ»
    2.    Трудовой договор (односторонний) – Трудовой договор на 1 стр.
    3.    Трудовой договор (двусторонни) – Трудовой договор на 2 стр.
    4.    РВП – разрешение на временное проживание
    5.    Паспорт пластик КГ – ID-карта КР
    6.    КГ всюду на – КР.
    7.    Штам по пересечении границы – Штап о последнем пересечении границы
  */
  if (db_id = 81001) then
  begin
    if (passkind_id = 43) then
      res_value = 'временная регистрация в РФ';

    res_value = trim((case passkind_id
                        when 43 then 'Временная регистрация в РФ' -- Временная регистрация, физ.лицо
                        when 312 then 'Трудовой договор на 1 стр.' -- Трудовой договор (односторонний)
                        when 63 then 'Трудовой договор на 2 стр.' -- Трудовой договор (двусторонни)
                        when 46 then 'Разрешение на временное проживание' -- РВП
                        when 8 then 'ID-карта КР' -- Паспорт пластик КГ
                        when 67 then 'Штамп о последнем пересечении границы' -- Штамп о пересечении границы
                        else ''
                      end));

    if (res_value = '') then
    begin
      res_value = passkind_name;

      len = character_length(passkind_name);

      if ((len > 3) and
          (right(passkind_name, 3) = ' КГ')) then
      begin
        res_value = substring(passkind_name from 1 for len - 3) || ' КР';
      end
    end
    suspend;
  end
  else if (db_id in (39801, 41701)) then
  begin
     res_value = passkind_name;
     if (passkind_id = 366) then
      res_value = 'Договор ЭЦП';

     suspend;
  end
end
