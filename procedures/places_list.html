CREATE PROCEDURE PLACES_LIST(
    ID DECIMAL(18,0),
    SELF_INCLUDE INTEGER,
    KIND INTEGER,
    IS_DOWN INTEGER)
RETURNS (
    ID_LIST DECIMAL(18,0))
AS
declare variable id1 decimal(18,0);
declare variable need_suspend integer;
begin
  if (is_down = 1) then
  begin
    id1 = :id;
    for select pl_id
        from places
        where (pl_id = :id1) or
              ((:id1 = 0) and
              (parent is null))
        into :id
    do
    begin
      for with recursive pl_list
          as (select pl_id, parent, pl_status
              from places
              where pl_id = :id and
                    is_delete = 0
              union all
              select p.pl_id, p.parent, p.pl_status
              from places p
              join pl_list on (pl_list.pl_id = p.parent) -- все дочерние
              where p.is_delete = 0)
          select pl_id
          from pl_list
          where ((:self_include = 0 and
                pl_id <> :id) or
                (:self_include = 1)) and
                ((:kind = -1) or
                (:kind <> -1 and
                pl_status = :kind))
          into :id_list
      do
        suspend;
    end
  end
  else
  begin
    id_list = id;
    need_suspend = 1;
    if (self_include = 0) then
      need_suspend = 0;
    if (need_suspend = 1 and
        kind <> -1) then
    begin
      if (not(exists(select *
                     from places
                     where pl_id = :id_list and
                           pl_status = :kind))) then
        need_suspend = 0;
    end
    if (need_suspend = 1) then
      suspend;

    while (id_list is not null) do
    begin
      need_suspend = 1;
      select parent
      from places
      where pl_id = :id_list
      into id_list;

      if (id_list is null) then
        need_suspend = 0;

      if (need_suspend = 1 and
          kind <> -1) then
      begin
        if (not(exists(select *
                       from places
                       where pl_id = :id_list and
                             pl_status = :kind))) then
          need_suspend = 0;
      end
      if (need_suspend = 1) then
        suspend;
    end
  end
end
