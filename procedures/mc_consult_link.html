CREATE PROCEDURE MC_CONSULT_LINK(
    MCL_ID D_ID)
RETURNS (
    CONSULT_LINK D_STR_255)
AS
declare variable c_id d_id;
declare variable db_id d_integer;
declare variable current_level d_integer;
declare variable level_1 d_integer;
declare variable access_time d_integer;
declare variable kc_id d_id;
declare variable kc_id_1 d_id;
declare variable end_date d_date;
declare variable k_disb_date d_date;
begin
  db_id = f_get_db_id();
  consult_link = '';

  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  if (c_id is null or db_id not in (39801, 41701)) then
  begin
    suspend;
    exit;
  end

  current_level = null;
  kc_id = null;

  for select kc.kc_id, kon.k_disb_date
      from kontstate ks
      inner join kontract kon on kon.k_id = ks.ks_from_k_id
      inner join knclient kc on kc.kc_from_k_id = kon.k_id
      inner join product pr on pr.pr_id = kon.k_from_pr_id
      where kc.kc_from_c_id = :c_id and
            kc.kc_amount_approved > 0 and
            ks.ks_ol_is_delete = 0 and
            --ks.ks_state = 1 and
            ks.ks_date_begin >= '01.04.2021' and
            pr.pr_code containing '.consultation.'
      into :kc_id_1, k_disb_date
  do
  begin
    select cons_level, access_time
    from clt_get_price_for_consultation(:kc_id_1)
    into level_1, access_time;
    end_date = dateadd(month, access_time, k_disb_date);

    if (end_date > current_date) then
    begin
      if (current_level is null) then
      begin
        current_level = level_1;
        kc_id = kc_id_1;
      end
      else
      begin
        if (level_1 > current_level) then
        begin
          current_level = level_1;
          kc_id = kc_id_1;
        end
      end
    end
  end

  if (kc_id is not null) then
  begin
    select 'https://consult.mbulak.kg/?uid=' || f_uuid('knclient', kc_id)
    from knclient kc
    inner join kontract k on kc_from_k_id = k_id
    inner join region r on k_last_reg_id = reg_unicode
    where kc_id = :kc_id
    into :consult_link;
  end

  suspend;
end
