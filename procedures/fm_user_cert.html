CREATE PROCEDURE FM_USER_CERT(
    UN_ID DECIMAL(18,0))
RETURNS (
    LO_NAME_FIO VARCHAR(100),
    LO_POSITION VARCHAR(100),
    LO_STATUS VARCHAR(50),
    LO_PHOTO_ID DECIMAL(18,0),
    MANAGER_NAME VARCHAR(100),
    MANAGER_PHONE VARCHAR(20),
    OFFICE_ADDR VARCHAR(255),
    OFFICE_PHONE VARCHAR(255))
AS
declare variable C_ID decimal(18,0);
declare variable REG_ID integer;
declare variable DB_ID integer;
declare variable MANAGER_ID decimal(18,0);
declare variable SAWT_ID integer;
declare variable MANAGER_NAME_TMP varchar(100);
declare variable MANAGER_PHONE_TMP varchar(20);
declare variable LO_POSITION_FOR_CHECK varchar(100);
declare variable MANAGER_OF_OFFICE varchar(100);
declare variable C_FOR_PHONE D_ID;
declare variable UN_NAME D_STR_100;
begin

  :db_id = f_get_db_id();
  select c_id, c_name_fio, c_blob_photo_id
  from username, clients
  where un_id = :un_id and
        c_id = un_from_c_id
  into c_id, lo_name_fio, lo_photo_id;

  select un_name
  from username
  where un_id = :un_id
  into :un_name;

  if (:un_name = 'fmtest') then
  begin
    lo_photo_id = 1556681;
    lo_name_fio = 'Тестовый пользователь';
    lo_position = 'Специалист ГО (Баку)';
    lo_status = 'Активен/Действующий';
    manager_name = 'Тестовый Менеджер';
    manager_phone = '99412ХХХХХХХ';
    office_addr = 'Город Баку, Ясамальский район, пр. Гусейна Джавида, 520-й квартал, дом 10/2;3';
    office_phone = '99411ХХХХХХХ';
    suspend;
    exit;
  end

  lo_position = 'Не работает';
  lo_status = 'Аннулирован';
  manager_name = '';
  manager_phone = '';
  office_addr = '';
  office_phone = '';

-- Заполнение актуальных данных
  for select jl_name || '\nОфис ' || reg_name, ps_from_reg_id, jl_name
      from hr_pstatus, joblist, region
      where ps_from_jl_id = jl_id and
            ps_from_reg_id = reg_unicode and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_from_c_id = :c_id
      into lo_position, reg_id, lo_position_for_check
  do
  begin
    -- RFCKZ-159. Реализовано для КЗ --
    for select c_name_fio, c_id
          from clients cl
          inner join hr_pstatus ps on cl.c_id = ps.ps_from_c_id
          inner join joblist on ps.ps_from_jl_id = jl_id
          inner join region on ps.ps_from_reg_id = reg_unicode
          where ps_status = 1 and
                ps_date_begin <= current_date and
                ps_date_end >= current_date and
                jl_name containing 'Менеджер офиса'
                and ps.ps_from_reg_id = :reg_id
          into :manager_of_office, :c_for_phone
          do
          begin
            if (db_id = 39801) then
            begin
                if (lo_position_for_check containing 'Менеджер офиса') then
                begin
                manager_name = 'Джумабекова Бактыгуль Ниязахуновна';
                manager_phone = (select cont_value
                                 from hp_contdata_get_cell_main(0, 1010100001987));
                end
                if (lo_position_for_check not containing 'Менеджер офиса') then
                begin
                manager_name = :manager_of_office;
                manager_phone = (select cont_value
                                 from hp_contdata_get_cell_main(0, :c_for_phone));
                end
            end
          end
-- Адрес и телефон офиса
    lo_status = 'Активен/Действующий';
    select adl_address_str, reg_phone
    from region, addrlist
    where reg_unicode = :reg_id and
          reg_address_fact = adl_id
    into :office_addr, office_phone;
    if (office_addr is null) then
      office_addr = '';
    if (office_phone is null) then
      office_phone = '';
    if (db_id = 39801) then
    office_phone  = (select cont_value
                       from hp_contdata_get_cell_main(0, :c_for_phone));

-- Данные руководителя
    if (not((db_id in (39801, 41701) and
        reg_id = 1090100) or (db_id not in (39801, 41701) and
        reg_id = 1010100))) then
    begin
-- находим руководителя офиса     --RFCRU-3896
         for select c.c_id, c.c_name_fio,
                 (select cont_value
                  from hp_contdata_get_cell_main(0, c.c_id))
         from krexpert ke
         inner join loanmgrlist lml on lml.lml_id = ke.ke_from_lml_id
         inner join clients c on c.c_id = lml.lml_from_c_id
         inner join hr_pstatus ps_mgr on ps_mgr.ps_from_c_id = c.c_id
         where ke.ke_from_p_id = :c_id and
               lml.lml_from_reg_id = :reg_id and
               current_date between ke.ke_date and ke.ke_date_end and
               current_date between lml.lml_date_begin and lml.lml_date_end and
               ps_mgr.ps_is_working = 1 and
               ps_mgr.ps_status = 1 and
               current_date between ps_mgr.ps_date_begin and ps_mgr.ps_date_end
         into :manager_id, :manager_name_tmp, manager_phone_tmp
      do
      begin
-- Проверка на декретный отпуск
        if (manager_id <> c_id) then
        begin
          sawt_id = null;
          select sawt_id
          from hr_voctravel, hr_orders, sa_worktype
          where hrv_from_ho_id = ho_id and
                ho_order_type = sawt_id and
                hrv_from_c_id = :manager_id and
                current_date between hrv_date_begin and hrv_date_end_fact
          order by hrv_date_begin desc
          rows 1
          into sawt_id;
          sawt_id = coalesce(sawt_id, 0);

          if (sawt_id not in (16, 17, 18, 8, 21)) then
          begin
            manager_name = manager_name_tmp;
            manager_phone = manager_phone_tmp;
          end
        end
      end

-- если руководитель не найден, ищем заместителя (И.О.)
      if (manager_name = '') then
      begin
        for select c_id, c_name_fio,
                   (select cont_value
                    from hp_contdata_get_cell_main(0, c_id))
            from clients, hr_pstatus, joblist, jobkind
            where ps_from_c_id = c_id and
                  ps_from_reg_id = :reg_id and
                  ps_from_jl_id = jl_id and
                  jl_kind = jk_id and
                  ps_date_begin <= current_date and
                  ps_date_end >= current_date and
                  ps_status = 1 and
                  ps_is_working = 1 and
                  jk_system_code = 'JobSystem.Region_KM' and
                  not(jl_name in ('Менеджер офиса', 'Кредитный менеджер', 'Filial müdiri')) rows 1
            into :manager_id, :manager_name_tmp, manager_phone_tmp
        do
        begin
-- Проверка на декретный отпуск
          if (manager_id <> c_id) then
          begin
            sawt_id = null;
            select sawt_id
            from hr_voctravel, hr_orders, sa_worktype
            where hrv_from_ho_id = ho_id and
                  ho_order_type = sawt_id and
                  hrv_from_c_id = :manager_id and
                  current_date between hrv_date_begin and hrv_date_end_fact
            order by hrv_date_begin desc
            rows 1
            into sawt_id;
            sawt_id = coalesce(sawt_id, 0);

            if (sawt_id not in (16, 17, 18, 8, 21)) then
            begin
              manager_name = manager_name_tmp;
              manager_phone = manager_phone_tmp;
            end
          end
        end
      end

      if (db_id in (41701)) then -- если это КГ и не ГО-офис, в качестве руководителя указваю Бахтияра
      begin
        manager_name = 'Тольбаев Бахтиёр Латиханович';
        manager_phone = (select cont_value
                         from hp_contdata_get_cell_main(0, 1090100133063));
      end
    end
    if (manager_name = '') then
    begin
      select c_name_fio
      from clients, hr_pstatus, joblist, jobkind
      where ps_from_c_id = c_id and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_status = 1 and
            ps_is_working = 1 and
            jk_system_code = 'JobSystem.Director_General' rows 1
      into :manager_name;
      manager_phone = '';
    end
  end
  suspend;
end
