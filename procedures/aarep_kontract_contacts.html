CREATE PROCEDURE AAREP_KONTRACT_CONTACTS(
    C_ID DECIMAL(18,0))
RETURNS (
    CONTACT_DATA VARCHAR(700),
    CONT_TYPE SMALLINT,
    CC_TYPE INTEGER)
AS
declare variable adl_type smallint;
declare variable adl_address_str varchar(255);
declare variable crel_other_c_id decimal(18,0);
declare variable cont_plastcard_no varchar(20);
declare variable relation_reg_adr varchar(255);
declare variable relation_rod_reg_adr varchar(255);
declare variable relation_rod_fact_adr varchar(255);
declare variable relation_fact_adr varchar(255);
begin
  cont_type = 0;
  -- данные по контактным лицам
  for select (case
                when coalesce(cont.cc_status, '') = '' then 'Статус не указан'
                else cont.cc_status
              end) || ': ' ||(
                   case
                     when coalesce(cont.cc_name, '') = '' then 'ФИО не указано'
                     else cont.cc_name
                   end) || ', ' || ' тел: ' ||(
                   case
                     when coalesce(cont.cc_cell_no, '') = '' then 'Телефон не указан   '
                     else cont.cc_cell_no
                   end) || '  ' ||(
                   case
                     when cont.cc_type = 1 then ' Род.: '
                     when cont.cc_type = 0 then ' РФ:'
                     else ''
                   end) ||(
                   case
                     when coalesce(cont.cc_address, '') = '' then 'Адрес не указан'
                     else cont.cc_address
                   end),
                   cc_type
      from contacts cont
      where cont.cc_from_c_id = :c_id
      into :contact_data, :cc_type
  do
  begin
    cont_type = 1; -- контактные лица
    suspend;
  end

  -- данные по родственникам
  cc_type = 3;
  for select crel_other_c_id,
             (
             case
               when coalesce(rel_name, '') = '' then 'Статус не указан'
               else rel_name
             end) || ': ' ||(
             case
               when coalesce(c_name_fio, '') = '' then 'ФИО не указано'
               else c_name_fio
             end) || ',',
             left(coalesce(c_plastcard_no, ''), 20)
      from hp_client_relation_list(:c_id, 0)
      where crel_is_active = 1 and
            crel_level <= 1
      into :crel_other_c_id, :contact_data, :cont_plastcard_no
  do
  begin
    cont_type = 2; -- родственники

    relation_reg_adr = '';
    relation_fact_adr = '';
    relation_rod_reg_adr = '';
    relation_rod_fact_adr = '';

    for select adc_address_type, adl.adl_address_str
        from addrclient adcl
        inner join addrlist adl on adcl.adc_from_adl_id = adl.adl_id
        where adcl.adc_from_c_id = :crel_other_c_id and
              adcl.adc_address_type in (1, 2, 4, 5)
        into :adl_type, :adl_address_str
    do
    begin
      if (adl_type = 1) then
        relation_reg_adr = adl_address_str;
      if (adl_type = 2) then
        relation_fact_adr = adl_address_str;
      if (adl_type = 4) then
        relation_rod_reg_adr = adl_address_str;
      if (adl_type = 5) then
        relation_rod_fact_adr = adl_address_str;
    end

    if (relation_reg_adr <> '') then
      relation_reg_adr = 'Адр.рег.РФ: ' || relation_reg_adr || ', ';
    if (relation_fact_adr <> '') then
      relation_fact_adr = 'Факт.адр.РФ: ' || relation_fact_adr || ', ';
    if (relation_rod_reg_adr <> '') then
      relation_rod_reg_adr = 'Адр.рег.род.: ' || relation_rod_reg_adr || ', ';
    if (relation_rod_fact_adr <> '') then
      relation_rod_fact_adr = 'Адр.факт.род.: ' || relation_rod_fact_adr || ', ';

    contact_data = contact_data || ' тел: ' || cont_plastcard_no;
    contact_data = contact_data || ' ' || iif(relation_reg_adr <> '', ascii_char(13) || ascii_char(10) || '      ' || relation_reg_adr, '') || iif(relation_fact_adr <> '', ascii_char(13) || ascii_char(10) || '      ' || relation_fact_adr, '') || iif(relation_rod_reg_adr <> '', ascii_char(13) || ascii_char(10) || '     ' || relation_rod_reg_adr, '') || iif(relation_rod_fact_adr <> '', ascii_char(13) || ascii_char(10) || '     ' || relation_rod_fact_adr, '');

    if (cont_plastcard_no = '') then
      cont_plastcard_no = 'Телефон не указан';

    suspend;
  end

  if (cont_type = 0) then
  begin
    suspend;
  end
end
