CREATE PROCEDURE AAREP_PRO_FIZYUR_LOANS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_IPF SMALLINT = 0)
RETURNS (
    REP_OBLAST VARCHAR(255),
    REP_BUS_NAME VARCHAR(255),
    REP_BALANCE_BEGIN DECIMAL(18,4),
    REP_BALANCE DECIMAL(18,4),
    REP_BALANCE_BEGIN1000 DECIMAL(18,4),
    REP_BALANCE1000 DECIMAL(18,4),
    REP_DISB_CL_AMOUNT1000 DECIMAL(18,4),
    REP_DISB_CL_KOLICH INTEGER,
    REP_DISB_CL_AMOUNT DECIMAL(18,4),
    REP_PERC_DISB_KOEF DECIMAL(18,7))
AS
declare res_cl_type d_integer;
declare res_oblast d_str_255;
declare res_oblast_p d_str_255;
declare res_bus_name d_str_255;
declare res_bus_name_p d_str_255;

declare res_balance_begin curr_type;
declare res_balance curr_type;
declare res_perc_disb_annual curr_type;
declare res_disb_cl_amount curr_type;
declare res_disb_cl_kolich curr_type;

declare res0_balance_begin curr_type;
declare res0_balance curr_type;
declare res0_perc_disb_annual curr_type;
declare res0_disb_cl_amount curr_type;
declare res0_disb_cl_kolich curr_type;

declare res1_balance_begin curr_type;
declare res1_balance curr_type;
declare res1_perc_disb_annual curr_type;
declare res1_disb_cl_amount curr_type;
declare res1_disb_cl_kolich curr_type;

declare variable rep_perc_disb_annual decimal(18,4);
begin
  res_oblast_p = '';
  res_bus_name_p = '';

  for
    with rep_data
    as (
      select
        rep_oblast,
        rep_bus_name,
        rep_cl_type,
        coalesce(sum(rep_balance_begin),0) as rep_balance_begin,
        coalesce(sum(rep_balance),0) as rep_balance,
        coalesce(sum(rep_perc_disb_annual),0) as rep_perc_disb_annual,
        coalesce(sum(rep_disb_cl_amount),0) as rep_disb_cl_amount,
        coalesce(sum(rep_disb_cl_kolich),0) as rep_disb_cl_kolich
      from aarep_pro_active_loans_tmp(
        :rep_region_id,
        :rep_date_begin,
        :rep_date_end,
        8,
        :is_ipf)
      group by rep_oblast, rep_bus_name, rep_cl_type
    )
    select
      rep_oblast,
      rep_bus_name,
      rep_cl_type,
      rep_balance_begin,
      rep_balance,
      rep_perc_disb_annual,
      rep_disb_cl_amount,
      rep_disb_cl_kolich
    from rep_data
    union
    select
      rep_oblast,
      'х Всего кредитов',
      rep_cl_type,
      sum(rep_balance_begin),
      sum(rep_balance),
      sum(rep_perc_disb_annual),
      sum(rep_disb_cl_amount),
      sum(rep_disb_cl_kolich)
    from rep_data
    group by rep_oblast, rep_cl_type
    union
    select
      'я',
      '',
      null,
      0,
      0,
      0,
      0,
      0
    from company
  into
    :res_oblast,
    :res_bus_name,
    :res_cl_type,
    :res_balance_begin,
    :res_balance,
    :res_perc_disb_annual,
    :res_disb_cl_amount,
    :res_disb_cl_kolich
  do
  begin
    if (res_oblast <> res_oblast_p or
        res_bus_name <> res_bus_name_p) then begin
      if (res_oblast_p <> '') then begin
        rep_oblast = res_oblast_p;

        rep_bus_name = res_bus_name_p;
        rep_balance_begin = res0_balance_begin + res1_balance_begin;
        rep_balance = res0_balance + res1_balance;
        rep_perc_disb_annual = res0_perc_disb_annual + res1_perc_disb_annual;
        rep_disb_cl_amount = res0_disb_cl_amount + res1_disb_cl_amount;
        rep_disb_cl_kolich = res0_disb_cl_kolich + res1_disb_cl_kolich;
        rep_balance1000 = cast(rep_balance / 1000 as decimal(15,2));
        rep_balance_begin1000 = cast(rep_balance_begin / 1000 as decimal(15,2));
        rep_disb_cl_amount1000 = cast(rep_disb_cl_amount / 1000 as decimal(15,2));
        if (rep_disb_cl_amount <> 0) then begin
          rep_perc_disb_koef = cast(rep_perc_disb_annual / rep_disb_cl_amount as decimal(15,7));
          rep_perc_disb_koef = cast(rep_perc_disb_koef * 100 as decimal(15,1));
        end else rep_perc_disb_koef = 0;
        suspend;

        rep_bus_name = '       юридические лица';
        rep_balance_begin = res1_balance_begin;
        rep_balance = res1_balance;
        rep_perc_disb_annual = res1_perc_disb_annual;
        rep_disb_cl_amount = res1_disb_cl_amount;
        rep_disb_cl_kolich = res1_disb_cl_kolich;
        rep_balance1000 = cast(rep_balance / 1000 as decimal(15,2));
        rep_balance_begin1000 = cast(rep_balance_begin / 1000 as decimal(15,2));
        rep_disb_cl_amount1000 = cast(rep_disb_cl_amount / 1000 as decimal(15,2));
        if (rep_disb_cl_amount <> 0) then begin
          rep_perc_disb_koef = cast(rep_perc_disb_annual / rep_disb_cl_amount as decimal(15,7));
          rep_perc_disb_koef = cast(rep_perc_disb_koef * 100 as decimal(15,1));
        end else rep_perc_disb_koef = 0;
        suspend;

        rep_bus_name = '        физические лица';
        rep_balance_begin = res0_balance_begin;
        rep_balance = res0_balance;
        rep_perc_disb_annual = res0_perc_disb_annual;
        rep_disb_cl_amount = res0_disb_cl_amount;
        rep_disb_cl_kolich = res0_disb_cl_kolich;
        rep_balance1000 = cast(rep_balance / 1000 as decimal(15,2));
        rep_balance_begin1000 = cast(rep_balance_begin / 1000 as decimal(15,2));
        rep_disb_cl_amount1000 = cast(rep_disb_cl_amount / 1000 as decimal(15,2));
        if (rep_disb_cl_amount <> 0) then begin
          rep_perc_disb_koef = cast(rep_perc_disb_annual / rep_disb_cl_amount as decimal(15,7));
          rep_perc_disb_koef = cast(rep_perc_disb_koef * 100 as decimal(15,1));
        end else rep_perc_disb_koef = 0;
        suspend;
      end
      res_oblast_p = res_oblast;
      res_bus_name_p = res_bus_name;

      res0_balance_begin = 0;
      res0_balance = 0;
      res0_perc_disb_annual = 0;
      res0_disb_cl_amount = 0;
      res0_disb_cl_kolich = 0;

      res1_balance_begin = 0;
      res1_balance = 0;
      res1_perc_disb_annual = 0;
      res1_disb_cl_amount = 0;
      res1_disb_cl_kolich = 0;
    end

    if (res_cl_type = 0) then begin
      res0_balance_begin = res_balance_begin;
      res0_balance = res_balance;
      res0_perc_disb_annual = res_perc_disb_annual;
      res0_disb_cl_amount = res_disb_cl_amount;
      res0_disb_cl_kolich = res_disb_cl_kolich;
    end else begin
      res1_balance_begin = res_balance_begin;
      res1_balance = res_balance;
      res1_perc_disb_annual = res_perc_disb_annual;
      res1_disb_cl_amount = res_disb_cl_amount;
      res1_disb_cl_kolich = res_disb_cl_kolich;
    end
  end
end
