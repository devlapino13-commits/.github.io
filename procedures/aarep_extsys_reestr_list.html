CREATE PROCEDURE AAREP_EXTSYS_REESTR_LIST(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_REESTR_DATE DATE,
    RES_REESTR_AMOUNT CURR_TYPE,
    RES_EXTSYS_NAME D_STR_255,
    RES_KOMMIS CURR_TYPE,
    RES_PAY_DATE DATE,
    RES_PAY_SUMMA CURR_TYPE,
    RES_DOC_NO D_STR_CODE,
    RES_KONTRACT_NO D_STR_CODE,
    RES_CLIENT_NAME D_STR_255,
    RES_ALL CURR_TYPE,
    RES_OS CURR_TYPE,
    RES_PERC CURR_TYPE,
    RES_OTHER CURR_TYPE)
AS
declare variable min_amount curr_type = 0;
declare variable olp_date_begin date;
declare variable olp_date_end date;
declare variable al_id d_str_code;
declare variable al_id_prv d_str_code;
declare variable esp_project d_str_code;
declare variable total_amount curr_type;
declare variable kc_id id_type;
declare variable total_oper curr_type;
declare variable res_pay_amount curr_type;
declare variable res_reestr_payed curr_type;
declare variable ob_client_kt id_type;
declare variable pay_ol_id decimal(18,0);
declare variable loan_ol_id decimal(18,0);
declare variable ol_id decimal(18,0);
declare variable oek_rd_id integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  min_amount = 600000;

  al_id_prv = '';
  res_extsys_name = '';
  for select ol_date, ol_id, olp_date_begin, olp_date_end, ob_account_kt, ob_summa, ob_client_kt, ol_doc_no
      from operperiod, operbook, operlist ol
      where olp_from_ol_id = ob_from_ol_id and
            ob_ol_is_active = 1 and
            ob_summa >= :min_amount and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            olp_from_ol_id = ol_id and
            ol.is_delete = 0 and
            ol.ol_id <> 1010100004300604
      order by ol_date, ol.ol_id
      into res_reestr_date, ol_id, olp_date_begin, olp_date_end, al_id, res_reestr_payed, ob_client_kt, res_doc_no
  do
  begin
    res_kommis = 0;
    if (al_id <> al_id_prv) then
    begin
      res_extsys_name = '';
      esp_project = '';
      select esp_name, esp_project
      from extsyspro
      where esp_account_no = :al_id
      into res_extsys_name, esp_project;

      if (esp_project = '') then
        exception e_default 'Project not found: ' || coalesce(al_id, '<null>');

      al_id_prv = al_id;
    end

    res_reestr_amount = res_reestr_payed;
    res_pay_date = null;
    res_pay_summa = 0;
    res_kontract_no = '---';
    res_client_name = '';
    res_all = 0;
    res_os = 0;
    res_perc = 0;
    res_other = 0;
    if (ob_client_kt is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ob_client_kt
      into res_client_name;
    end
    suspend;

    if (ol_id = 1010100004295933) then
    begin
      select ol_date, ol_id, olp_date_begin, olp_date_end, ob_account_kt, ob_summa, ob_client_kt, ol_doc_no
      from operperiod, operbook, operlist ol
      where olp_from_ol_id = ob_from_ol_id and
            ob_ol_is_active = 1 and
            ob_summa >= :min_amount and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            olp_from_ol_id = ol_id and
            ol.is_delete = 0 and
            ol.ol_id = 1010100004300604
      order by ol_date, ol.ol_id
      into res_reestr_date, ol_id, olp_date_begin, olp_date_end, al_id, res_reestr_amount, ob_client_kt, res_doc_no;

      suspend;
      res_reestr_payed = res_reestr_payed + res_reestr_amount;
    end

    res_reestr_amount = 0;

    total_oper = 0;
    for select esm_date, eso_from_ol_id, esm_amount, esm_trans_id
        from extsysmain, extsysoper, operlist ol
        where esm_id = eso_from_esm_id and
              esm_project = :esp_project and
              esm_status = 1 and
              esm_action = 'pay' and
              eso_from_ol_id = ol.ol_id and
              ol.is_delete = 0 and
              esm_date between :olp_date_begin and :olp_date_end
        into res_pay_date, pay_ol_id, res_pay_amount, res_doc_no
    do
    begin
      total_amount = 0;
      res_pay_summa = res_pay_amount;

      loan_ol_id = null;
      select ol_id
      from operlist
      where ol_parent_id = :pay_ol_id
      into loan_ol_id;
      if (loan_ol_id is null) then
        loan_ol_id = :pay_ol_id;

      for with tab
          as (select ok_from_kc_id as kc_id, iif(ok_type = 4, ok_os, 0) as ok_os,
                     iif(ok_type = 4, ok_perc, 0) as ok_perc,
                     iif(ok_type = 9, ok_os, 0) + iif(ok_type = 9, ok_perc, 0) + ok_fine + ok_kommis_adv + ok_tax as ok_other
              from kn_operkred
              where ok_from_ol_id = :loan_ol_id and
                    ok_type in (4, 9)
              union all
              select kc_id, 0 as ok_os, 0 as ok_perc, oko_os + oko_perc + oko_other as ok_other
              from operkroff, knclient
              where oko_from_ol_id = :loan_ol_id and
                    oko_from_k_id = kc_from_k_id and
                    oko_from_c_id = kc_from_c_id and
                    oko_type = 4 and
                    kc_amount_approved > 0
              union all
              select kc_id, 0, 0,
                     -round((cast(ox_amount_minus as double precision) / k_amount_approved) * kc_amount_approved, 2)
              from operextra, knclient, kontract
              where ox_from_rd_id = :oek_rd_id and
                    ox_from_ol_id = :loan_ol_id and
                    kc_from_k_id = ox_object_id and
                    kc_amount_approved > 0 and
                    k_id = kc_from_k_id)
          select kc_id, -sum(ok_os), -sum(ok_perc), -sum(ok_other)
          from tab
          group by 1
          into :kc_id, res_os, res_perc, res_other
      do
      begin
        total_amount = total_amount + res_os + res_perc + res_other;
        if (abs(total_amount - res_pay_amount) <= 1) then
        begin
          res_other = res_other + res_pay_amount - total_amount;
          total_amount = res_pay_amount;
        end
        res_all = res_os + res_perc + res_other;

        res_kontract_no = '';
        res_client_name = '';
        select k_kontract_no, c_name_fio
        from knclient, kontract, clients
        where kc_id = :kc_id and
              kc_from_k_id = k_id and
              c_id = kc_from_c_id
        into :res_kontract_no, res_client_name;

        suspend;
        res_pay_summa = 0;
      end
      if (res_pay_amount <> total_amount) then
      begin
        res_kontract_no = 'X';
        res_client_name = 'X';
        res_all = res_pay_amount - total_amount;
        res_os = 0;
        res_perc = 0;
        res_other = 0;
        suspend;
        res_pay_summa = 0;
        res_reestr_amount = 0;
      end
      total_oper = total_oper + res_pay_amount;
    end

    if (res_reestr_payed <> total_oper) then
    begin
      res_kontract_no = '---';
      res_client_name = 'Разница';
      res_doc_no = '';
      res_kommis = res_reestr_payed - total_oper;
      res_all = 0;
      res_pay_summa = 0;
      res_pay_amount = 0;
      res_os = 0;
      res_perc = 0;
      res_other = 0;
      suspend;
      res_reestr_amount = 0;
    end
  end
end
