CREATE PROCEDURE AAREP_KZ_R_MRZ_MFO(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    CLI_BALANCE D_CURRENCY,
    CLI_PERC_SALDO D_CURRENCY,
    CLI_CALC_FINE D_CURRENCY,
    BAL_STOEMOST D_CURRENCY,
    WOFF_ALL_SUM D_CURRENCY,
    REP_DATE_END_STR D_STR_100)
AS
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable OKS_BALANCE D_CURRENCY;
declare variable K_PERC_SALDO D_CURRENCY;
declare variable CLI_DOLY D_DOUBLE;
declare variable SH_CALC_FINE D_CURRENCY;
declare variable K_ID bigint;
declare variable NEW_VAR integer;
declare variable C_ID D_ID;
begin
  rep_date_end_str = 'Отчетный период: по состоянию на '|| f_datetostr(:rep_date_end);
  for
  with tabl1 as (
    select c_name_fio,
           c_inn,
           c_id,
           sum(cli_balance) cli_balance,
           sum(cli_perc_saldo) cli_perc_saldo,
           sum(cli_calc_fine) cli_calc_fine,
           sum(bal_stoemost) bal_stoemost
    from aarep_kz_r_mrz_mfo_tmp(:rep_region_id, :rep_date_end)
    group by 1,2,3
    order by 7 desc
    rows 1
    )

  select * from tabl1
  into :c_name_fio, :c_inn, :c_id, :cli_balance, :cli_perc_saldo, :cli_calc_fine, :bal_stoemost
  do
  begin
    
    select sum(oko_sh_os + oko_sh_perc + oko_sh_other)
    from operkroff
    where oko_from_c_id = :c_id
    and oko_ol_is_active = 1
    and oko_type = 1
    into woff_all_sum;
    woff_all_sum = coalesce(woff_all_sum, 0);

    suspend;
  end 
end
