    REP_REGION_ID INTEGER,

    ERR_TEXT VARCHAR(2048))
AS
declare variable res_ol_memo varchar(1024);
declare variable al_id varchar(20);
declare variable al_kind integer;
declare variable refault_code varchar(10);
declare variable citizen_ru_id integer;
declare variable doc_vnj_id integer;
declare variable c_citizen integer;
declare variable ol_parent_id decimal(18,0);
declare variable citizen_xx varchar(10);
declare variable citizen_xx_prior varchar(10);
declare variable err_tmp integer;
declare variable res_ol_id d_str_20;
declare variable res_ob_number integer;
declare variable res_date date;
declare variable res_doc_no integer;
declare variable res_doc_name varchar(20);
declare variable res_account_dt varchar(20);
declare variable res_account_kt varchar(20);
declare variable res_currency varchar(3);
declare variable res_amount decimal(18,4);
declare variable res_is_bank integer; /* 0-Касса, 1-Банк */
declare variable res_oper_sign integer; /* 1-приход, -1-расход */
declare variable res_client_id decimal(18,0);
declare variable res_bank_id varchar(10);
declare variable res_error integer;
declare variable res_client_name varchar(100);
declare variable res_client_is_resident integer;
declare variable res_client_type varchar(10); /* 0-Физ.лицо, 1-юр.лицо */
declare variable res_client_resident_str varchar(20);
declare variable res_client_country varchar(10);
declare variable res_opercode_rd_id integer;
declare variable res_opercode_str varchar(20);
declare variable res_oper_sign_str varchar(20);
declare variable res_region_id integer;
declare variable res_acc_dt_tmp varchar(20);
declare variable res_acc_kt_tmp varchar(20);
declare variable res_amount_tmp decimal(18,4) = 0;
declare variable k_id decimal(18,0);
declare variable client_koef decimal(18,7);
declare variable res_bank_name varchar(255);
declare variable c_id_tmp decimal(18,0);
declare variable total_client integer;
declare variable client_no integer;
declare variable total_amount decimal(18,4) = 0;
declare variable res_amount_orig decimal(18,4) = 0;
declare variable olp_date_begin date;
declare variable olp_date_end date;
declare variable ol_id decimal(18,0);
declare variable full_acc_kt varchar(20);
declare variable ob_ol_date date;
declare variable ob_summa decimal(18,4) = 0;
declare variable ob_summa_all decimal(18,4) = 0;
declare variable ol_id_tmp decimal(18,0);
declare variable res_is_bank_tmp integer;
declare variable res_bank_id_tmp varchar(20);
declare variable ob_account_kt varchar(20);
declare variable res_client_id_tmp decimal(18,0);
declare variable ob_summa_all2 decimal(18,4) = 0;
declare variable is_sud integer;
declare variable law_rd_id integer;
declare variable num_dt integer;
declare variable num_kt integer;
declare variable tmp_num_dt integer;
declare variable tmp_num_kt integer;
declare variable cm_name str_255;
declare variable res_from_kc_id d_id;
declare variable res_kontract_no d_str_code;
declare variable res_client_date d_date;
declare variable res_acc_code d_str_code;
declare variable com_ol d_str_20;
declare variable com_ob_num d_integer;
declare variable com_doc_no d_integer;
declare variable com_ob_account_dt d_str_20;
declare variable com_ob_account_kt d_str_20;
declare variable com_ob_summa d_currency;
declare variable com_ol_memo d_str_255;
declare variable res_opercode_str_tmp d_str_20;
begin
  select c.cm_name
  from company c
  into cm_name;

  res_opercode_rd_id = 0;
  res_opercode_str = '';
  citizen_xx_prior = '';

  select rd_id
  from refdata
  where rd_code = 'OperExtraKind.LawClientExpense'
  into :law_rd_id;

  -- Проверка операций через третих лиц на предмет совпадания сумм.
  err_text = '';
  for select ob_from_ol_id, ob_account_kt, ob_ol_date, ob_summa
      from operbook
      where ob_ol_date between :rep_date_begin and :rep_date_end and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = 1010100 and
            (ob_account_dt like '%51-01%' or
            ob_account_dt like '%55-04%') and
            ob_from_ol_id <> 1010100002211214
            -- and ob_from_ol_id in (1010100001536449, 1010100001536450)
            and
            ob_account_kt in ('010181076-10000002', '010181076-10000003', '010181076-10000004', '010181076-10000005', '010181076-10000006', '010181076-10000007', '010181076-10000012'/*, '010181076-10000010'*/)  -- последний счет исключен из проверки, RFCRU-289
      into :ol_id, full_acc_kt, :ob_ol_date, :ob_summa
  do
  begin
    olp_date_begin = null;
    olp_date_end = null;
    select olp_date_begin, olp_date_end
    from operperiod
    where olp_from_ol_id = :ol_id
    into :olp_date_begin, olp_date_end;
    if (coalesce(olp_date_begin, '01.01.1990') <= '01.01.1990' or coalesce(olp_date_end, '01.01.1990') <= '01.01.1990') then
    begin
      err_text = err_text || 'Период не указан. Дата: ' || ob_ol_date || ', Кт: ' || :full_acc_kt || ', Сумма: ' || round(ob_summa, 2) || ascii_char(13) || ascii_char(10);
    end
  end
  if (err_text <> '') then
  begin
    suspend;
    exit;
  end

  for select ob_from_ol_id, ob_account_kt, ob_ol_date, ob_summa
      from operbook
      where ob_ol_date between :rep_date_begin and :rep_date_end and
            ob_ol_is_active = 1 and
            ob_ol_reg_id = 1010100 and
            ob_from_ol_id <> 1010100002211214 and
            (ob_account_dt like '%51-01%' or
            ob_account_dt like '%55-04%') and
            ob_account_kt in ('010181076-10000002', '010181076-10000003', '010181076-10000004', '010181076-10000005', '010181076-10000006', '010181076-10000007', '010181076-10000012'/*, '010181076-10000010'*/)  -- последний счет исключен из проверки, RFCRU-289
      into :ol_id, full_acc_kt, :ob_ol_date, :ob_summa
  do
  begin
    olp_date_begin = null;
    olp_date_end = null;
    select olp_date_begin, olp_date_end
    from operperiod
    where olp_from_ol_id = :ol_id
    into :olp_date_begin, olp_date_end;

    ob_summa_all = 0;
    select sum(ob_summa)
    from operbook
    where ob_ol_is_active = 1 and
          ob_account_dt = :full_acc_kt and
          ob_ol_date between :olp_date_begin and :olp_date_end and
          ob_from_ol_id not in (1010100001724609) --Возврат неправильной суммы. Просьба Снежанны от 25 мая 16 года --Музаффар
    into :ob_summa_all;
    if (ob_summa_all is null) then
      ob_summa_all = 0;

    if (full_acc_kt = '010181076-10000007') then
    begin
      ob_summa_all2 = 0;
      select sum(ob_summa)
      from operbook, operperiod
      where ob_ol_is_active = 1 and
            ob_account_dt = '010181091-02000001' and
            ob_account_kt = '010181076-10000007' and
            ob_ol_date = :ob_ol_date and
            olp_date_begin = :olp_date_begin and
            olp_date_end = :olp_date_end and
            ob_from_ol_id = olp_from_ol_id and
            ob_from_ol_id not in (1010100001724609) --Возврат неправильной суммы. Просьба Снежанны от 25 мая 16 года --Музаффар
      into :ob_summa_all2;
      if (ob_summa_all2 is null) then
        ob_summa_all2 = 0;
      ob_summa_all = ob_summa_all - ob_summa_all2;
    end

    if (full_acc_kt in ('010181076-10000010', '010181076-10000012')) then
    begin
      ob_summa_all2 = 0;
      select sum(ob_summa)
      from operbook, operperiod
      where ob_ol_is_active = 1 and
            ob_account_dt = '010181091-02000003' and
            ob_account_kt = :full_acc_kt and
            ob_ol_date = :ob_ol_date and
            olp_date_begin = :olp_date_begin and
            olp_date_end = :olp_date_end and
            ob_from_ol_id = olp_from_ol_id and
            ob_from_ol_id not in (1010100001724609) --Возврат неправильной суммы. Просьба Снежанны от 25 мая 16 года --Музаффар
      into :ob_summa_all2;
      if (ob_summa_all2 is null) then
        ob_summa_all2 = 0;
      ob_summa_all = ob_summa_all - ob_summa_all2;
    end

    if (char_length(err_text) <= 1024) then
    begin
      if (ob_summa_all <> ob_summa) then
      begin
        if (rep_date_end <> '30.06.2018') then
          err_text = err_text || 'Сумма детализации отличается. Дата: ' || ob_ol_date || ', Кт: ' || :full_acc_kt || ', Сумма в проводке: ' || round(:ob_summa, 2) || ', сумма детализации: ' || round(:ob_summa_all, 2) || ascii_char(13) || ascii_char(10);
      end
    end
  end
  if (err_text <> '') then
  begin
    suspend;
    exit;
  end

  select rd_memo
  from refdata
  where rd_main_code = 'CountryOKSM' and
        rd_str_id = 'RU' rows 1
  into :refault_code;
  err_tmp = 0;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  for select al_id, atl_from_at_id, al_from_reg_id
      from acc_list, acctypelist, region_list(:rep_region_id, 1, 3, 1)
      where al_from_bg_id = atl_acc_mask and
            atl_from_at_id in (11, 12)
            --  and atl_from_at_id=11
            and
            al_from_reg_id = id_list
      into :al_id, :al_kind, :res_region_id
  do
  begin
    -- Список приходных операций
    res_bank_id = '';
    res_bank_name = '';
    if (al_kind = 12) then
    begin
      select br_from_b_id
      from bankregion
      where br_from_reg_id = :res_region_id and
            br_name = :al_id
      order by br_is_active desc
      rows 1
      into :res_bank_id;
      if (res_bank_id is null) then
        res_bank_id = '';

      if (res_bank_id <> '') then
      begin
        select b_name
        from banks
        where b_id = :res_bank_id
        into :res_bank_name;
      end
    end

    res_is_bank_tmp = al_kind - 11;
    res_ol_memo='';

    for select cast(ob_from_ol_id as varchar(20)), ob_number, ob_ol_date, ol_doc_no, substring(ob_account_dt from 8 for 5),
               substring(ob_account_kt from 8 for 5), ob_currency, ob_summa, coalesce(ob_client_kt, ol_document_client),
               1, ol_parent_id, 'Зачисление',
               iif(ob_account_kt in ('010181076-10000002', '010181076-10000003', '010181076-10000004', '010181076-10000005', '010181076-10000006', '010181076-10000007', '010181076-10000010', '010181076-10000012'), -1, :res_is_bank_tmp), --'010181076-10000012' добавлен по RFCRU-4939
               ob_account_kt, cast(right(ob_account_dt, 6) as integer), cast(right(ob_account_kt, 6) as integer),ol_memo
        from operbook, operlist
        where ob_account_dt = :al_id and
              ob_ol_is_active = 1 and
              ob_ol_reg_id = :res_region_id and
              ob_ol_date between :rep_date_begin and :rep_date_end and
              ob_from_ol_id = ol_id
                --and ob_from_ol_id in (1010100044183375, 1010100001536449, 1010100001536450)
        union all
        select cast(ob_from_ol_id as varchar(20)), ob_number, ob_ol_date, ol_doc_no,
               substring(ob_account_dt from 8 for 5), substring(ob_account_kt from 8 for 5), ob_currency, ob_summa,
               coalesce(ob_client_dt, ol_document_client), -1, ol_parent_id, 'Списание', :res_is_bank_tmp,
               ob_account_kt, cast(right(ob_account_dt, 6) as integer), cast(right(ob_account_kt, 6) as integer),ol_memo
        from operbook, operlist
        where ob_account_kt = :al_id and
              ob_ol_is_active = 1 and
              ob_ol_reg_id = :res_region_id and
              ob_ol_date between :rep_date_begin and :rep_date_end and
              ob_from_ol_id = ol_id and
              :res_is_bank_tmp >= 0
                --and ob_from_ol_id in (1010100044183375, 1010100001536449, 1010100001536450)
        order by 3, 4, 1
        into :res_ol_id, res_ob_number, res_date, res_doc_no, res_account_dt, res_account_kt, res_currency, res_amount,
             res_client_id, res_oper_sign, ol_parent_id, res_oper_sign_str, :res_is_bank, :ob_account_kt, :tmp_num_dt,
             :tmp_num_kt, :res_ol_memo
    do
    begin
      res_client_name = '';
      res_client_is_resident = 1;
      res_client_resident_str = '';
      res_client_country = '';
      res_client_type = '';

      res_error = 0;
      if (res_is_bank <> 0) then
        res_doc_name = 'п/п';

      if (res_is_bank = 0) then
      begin
        if (res_oper_sign = 1) then
          res_doc_name = 'ПКО';
        else
          res_doc_name = 'РКО';
      end

      k_id = 0;
      is_sud = 0;
      select distinct ok_from_k_id
      from operkred, operlist
      where ok_from_ol_id = ol_id and
            ok_ol_is_active = 1 and
            ((:res_ol_id <> 1010100002211214 and
            (ol_id = :res_ol_id or
            ol_parent_id = :res_ol_id)) or
            (:res_ol_id = 1010100002211214 and
            ol_id = 1010100002210228)) rows 1
      into :k_id;
      if (k_id is null) then
        k_id = 0;
      total_client = 1;

      if (k_id = 0 and
          res_date >= '01.10.2016') then
      begin
        select ox_object_id
        from operextra
        where ox_from_ol_id = :res_ol_id and
              ox_object_kind = 9 and
              ox_ol_is_active = 1
        into :k_id;
        if (k_id is null) then
          k_id = 0;
      end

      if (k_id <> 0) then
      begin
        select count(*)
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :total_client;
        if (total_client is null) then
          total_client = 1;

        if (res_date >= '01.10.2016') then
        begin
          if (exists(select *
                     from operextra
                     where ox_object_kind = 9 and
                           ox_object_id = :k_id and
                           ox_ol_is_active = 1 and
                           ox_ol_date <= :res_date and
                           ox_from_rd_id = :law_rd_id)) then
            is_sud = 1;
        end
      end
      else
      begin
        k_id = 0;
        total_client = 1;
      end

      res_acc_dt_tmp = res_account_dt;
      res_acc_kt_tmp = res_account_kt;
      res_amount_tmp = :res_amount;

      -- Кредит погашен через банк, заменяем 57-04 на кредитные счета
      for select :res_acc_dt_tmp, substring(ob_account_kt from 8 for 5), ob_summa, :tmp_num_dt,
                 cast(right(ob_account_kt, 6) as integer)
          from operbook, operlist
          where ob_from_ol_id = ol_id and
                (ol_parent_id = :res_ol_id or
                ol_id = :ol_parent_id) and
                operlist.is_delete = 0
                --      and (:res_is_bank<>0 and :RES_ACC_KT_TMP='57-04' and not(ob_account_kt like '%57-04%'))
                and
                (:res_is_bank >= 0 and
                :res_acc_kt_tmp = '57-04' and
                not(ob_account_kt like '%57-04%'))
          union all
          select substring(ob_account_dt from 8 for 5), :res_acc_kt_tmp, ob_summa,
                 cast(right(ob_account_dt, 6) as integer), :tmp_num_kt
          from operbook, operlist
          where ob_from_ol_id = ol_id and
                (ol_parent_id = :res_ol_id or
                ol_id = :ol_parent_id) and
                operlist.is_delete = 0
                --      and (:res_is_bank<>0 and :RES_ACC_DT_TMP='57-04' and not(ob_account_dt like '%57-04%'))
                and
                (:res_is_bank >= 0 and
                :res_acc_dt_tmp = '57-04' and
                not(ob_account_dt like '%57-04%'))
          union all
          select :res_acc_dt_tmp, :res_acc_kt_tmp, :res_amount_tmp, :tmp_num_dt, :tmp_num_kt
          from rdb$database
          --      where (not(:res_is_bank<>0 and (:RES_ACC_DT_TMP='57-04' or :RES_ACC_KT_TMP='57-04')))
          where (not(:res_is_bank >= 0 and
                (:res_acc_dt_tmp = '57-04' or
                :res_acc_kt_tmp = '57-04')))
          into :res_account_dt, :res_account_kt, res_amount, :num_dt, :num_kt
      do
      begin
        err_tmp = 0;
        client_no = 0;
        res_amount_orig = res_amount;
        total_amount = 0;
        res_client_id_tmp = res_client_id;
        -- Если обычное физ.лицо в операции
        for select c_id,
                   iif(c_type = 1, iif(c_name_fio like 'ИП %' or c_name_fio like '% ИП', 'ИП', 'ЮЛ'), 'ФЗ'),
                   c_name_fio, c_is_resident, replace(rd_code, 'citizen.', ''), c_citizen, 1.0, c_birth_date, null, ''
            from clients, refdata
            where (:k_id = 0 and
                  :res_client_id_tmp is not null and
                  c_id = :res_client_id_tmp) and
                  c_citizen = rd_id
            union all
            -- если клиент не указан, идет банк
            select null, 'ЮЛ', :res_bank_name, 1, 'RU', :citizen_ru_id, 1.0, null, null, ''
            from rdb$database
            where (:k_id = 0 and
                  :res_client_id_tmp is null and
                  :res_bank_id <> '')
            union all
            -- если клиент не указан и банк тоже не указан
            select null, '', '', 1, 'RU', :citizen_ru_id, 1.0, null, null, ''
            from rdb$database
            where (:k_id = 0 and
                  :res_client_id_tmp is null and
                  :res_bank_id = '')
            union all
            -- если кредитная операция
            select c_id,
                   iif(c_type = 1, iif(c_name_fio like 'ИП %' or c_name_fio like '% ИП', 'ИП', 'ЮЛ'), 'ФЗ'),
                   c_name_fio, c_is_resident, replace(rd_code, 'citizen.', ''), c_citizen,
                   kc_amount_approved / k_amount_approved, c_birth_date, kc_id, k_kontract_no
            from clients, refdata, knclient, kontract
            where :k_id <> 0 and
                  c_id = kc_from_c_id and
                  c_citizen = rd_id and
                  kc_amount_approved > 0 and
                  kc_from_k_id = :k_id and
                  k_id = :k_id
            into :c_id_tmp, :res_client_type, res_client_name, res_client_is_resident, citizen_xx, c_citizen,
                 client_koef, res_client_date, res_from_kc_id, res_kontract_no
        do
        begin
          res_client_id = c_id_tmp;
          res_client_type = trim(res_client_type);
          client_no = client_no + 1;

          res_acc_code = '';
          if (res_from_kc_id is not null) then
          begin
            select iif(non_resident = 1, '492', '488')
            from hp_get_kn_client_resident(:res_from_kc_id, :rep_date_end)
            into res_acc_code;
          end

          if (client_koef = 1) then
            res_amount = res_amount_orig;
          else
          begin
            if (client_no < total_client) then
            begin
              res_amount = round(cast(res_amount_orig as double precision) * client_koef, 0);
              total_amount = total_amount + res_amount;
            end
            else
            begin
              res_amount = res_amount_orig - total_amount;
            end
          end
          if (res_client_id = 1010100658562) then
          begin
            res_client_id = 1010100658562;
          end

          if (res_client_type = 'ФЗ') then
          begin
            select 1 - res_non_resident, res_country_code, res_country_id
            from hp_get_client_type(:res_client_id, :citizen_ru_id, :doc_vnj_id, :res_date)
            into :res_client_is_resident, citizen_xx, c_citizen;
          end
          else
          begin
            if (citizen_ru_id = c_citizen) then
              res_client_is_resident = 1;
          end

          --          if (citizen_xx <> citizen_xx_prior) then
          --          begin
          res_client_country = '';
          if (citizen_xx <> '') then
          begin
            select rd_memo
            from refdata
            where rd_main_code = 'CountryOKSM' and
                  rd_str_id = :citizen_xx and
                  rd_parent_id is not null rows 1
            into :res_client_country;
          end
          citizen_xx_prior = citizen_xx;
          --          end

          if (res_client_is_resident = 0) then
            res_client_resident_str = 'Нерезидент';
          else
            res_client_resident_str = 'Резидент';

          --          if (res_client_country = '') then
          --          begin
          --            select rd_memo
          --            from refdata
          --            where rd_main_code = 'CountryOKSM' and
          --                  rd_str_id = :citizen_xx and
          --                  rd_parent_id is not null rows 1
          --            into :res_client_country;
          --
          --            citizen_xx_prior = citizen_xx;
          --          end

          if (res_client_country = '') then
            res_error = 1;

          if ((res_account_dt starting '51-' and
              res_account_kt starting '51-') or (res_account_dt starting '51-' and
              res_account_kt starting '50-') or (res_account_dt starting '50-' and
              res_account_kt starting '51-')

              or (res_account_dt starting '50-' and
              res_account_kt starting '57-01') or (res_account_dt starting '57-01' and
              res_account_kt starting '50-') or (res_account_dt starting '57-01' and
              res_account_kt starting '51-')) then
          begin
            res_client_name = cm_name;
            res_client_is_resident = 1;
            res_client_resident_str = 'Резидент';
            res_client_country = refault_code;
            res_client_type = 'ЮЛ';

            res_client_date = null;
            res_from_kc_id = null;
            res_kontract_no = '';
            res_acc_code = '';
          end
          if (res_account_dt starting '70-01' and
              res_account_kt starting '51-') then
            res_client_type = 'ПФЛ';
          if (res_client_country = '') then
            res_client_country = refault_code;

          select res_opercode_rd_id, res_opercode_str, res_error
          from aarep_get_moneyoper_code(:res_account_dt, :res_account_kt, :is_sud * 100 + :res_client_is_resident, :num_dt, :num_kt)
          into :res_opercode_rd_id, :res_opercode_str, :err_tmp;
          if (res_error = 0) then
            res_error = err_tmp;

          res_bank_id_tmp = res_bank_id;
          res_opercode_str_tmp = res_opercode_str;
          if (res_is_bank = -1) then -- если это платежные системы (76-10.2 и тд)
          begin
            res_bank_id = '';
            res_opercode_rd_id = 0;
            res_oper_sign = 0;
            res_opercode_str = '';
            res_client_type = 'ПФЛ';
          end

          insert into temp_moneyclassific (res_ol_id, res_ob_number, res_date, res_doc_no, res_doc_name, res_account_dt,
                                           res_account_kt, res_currency, res_amount, res_is_bank, res_oper_sign,
                                           res_client_id, res_bank_id, res_error, res_client_name,
                                           res_client_is_resident, res_client_type, res_client_resident_str,
                                           res_client_country, res_opercode_rd_id, res_opercode_str, res_oper_sign_str,
                                           res_region_id, res_from_kc_id, res_kontract_no, res_client_date,
                                           res_acc_code,res_ol_memo)
          values (:res_ol_id, :res_ob_number, :res_date, :res_doc_no, :res_doc_name, :res_account_dt, :res_account_kt,
                  :res_currency, :res_amount, :res_is_bank, :res_oper_sign, :res_client_id, :res_bank_id, :res_error,
                  :res_client_name, :res_client_is_resident, :res_client_type, :res_client_resident_str,
                  :res_client_country, :res_opercode_rd_id, :res_opercode_str, :res_oper_sign_str, :res_region_id,
                  :res_from_kc_id, :res_kontract_no, :res_client_date, :res_acc_code, left(:res_ol_memo, 250));

          if (:ob_account_kt = '010181076-10000012') then --RFCRU-4939 комиссии по зеста
          begin
            select olp_date_begin, olp_date_end
            from operperiod
            where olp_from_ol_id = :res_ol_id
            into :olp_date_begin, olp_date_end;

            for select cast(ob_from_ol_id as d_str_20), ob.ob_number, ol.ol_doc_no, substring(ob.ob_account_dt from 8 for 5), substring(ob.ob_account_kt from 8 for 5),
                  ob_summa, ol.ol_memo
            from operbook ob
            inner join operlist ol on ol.ol_id = ob.ob_from_ol_id
            inner join operperiod olp on olp.olp_from_ol_id = ol.ol_id
            where ob_ol_is_active = 1 and
                  ob_account_dt = '010181091-02000003' and
                  ob_account_kt = '010181076-10000012' and
                  ob_ol_date = :res_date and
                  olp_date_begin = :olp_date_begin and
                  olp_date_end = :olp_date_end
            into :com_ol, com_ob_num, com_doc_no, com_ob_account_dt, com_ob_account_kt,
                  com_ob_summa, com_ol_memo
            do
            begin
              insert into temp_moneyclassific (res_ol_id, res_ob_number, res_date, res_doc_no, res_doc_name, res_account_dt,
                                           res_account_kt, res_currency, res_amount, res_is_bank, res_oper_sign,
                                           res_client_id, res_bank_id, res_error, res_client_name,
                                           res_client_is_resident, res_client_type, res_client_resident_str,
                                           res_client_country, res_opercode_rd_id, res_opercode_str, res_oper_sign_str,
                                           res_region_id, res_from_kc_id, res_kontract_no, res_client_date,
                                           res_acc_code,res_ol_memo)
              values (:com_ol, :com_ob_num, :res_date, :com_doc_no, :res_doc_name, :com_ob_account_dt, :com_ob_account_kt,
                      :res_currency, :com_ob_summa, :res_is_bank, -1, :res_client_id, :res_bank_id_tmp, :res_error,
                      :res_client_name, :res_client_is_resident, :res_client_type, :res_client_resident_str,
                      :res_client_country, :res_opercode_rd_id, :res_opercode_str_tmp, 'Списание', :res_region_id,
                      :res_from_kc_id, :res_kontract_no, :res_client_date, :res_acc_code, left(:com_ol_memo, 250));
            end
          end

          res_bank_id = res_bank_id_tmp;
        end
      end
      if (res_is_bank = -1) then -- если это касса или банк (без платежных терминалов 76-10)
      begin
        -- Через платежные терминалы

        res_client_name = '';
        res_client_is_resident = 1;
        res_client_resident_str = '';
        res_client_country = '';
        res_client_type = '';

        res_error = 0;
        res_doc_name = 'МемОрдер';
        res_oper_sign = 1;
        res_oper_sign_str = 'МемОрдер';
        res_ol_memo='';

        olp_date_begin = null;
        olp_date_end = null;
        select olp_date_begin, olp_date_end
        from operperiod
        where olp_from_ol_id = :res_ol_id
        into :olp_date_begin, olp_date_end;

        for select distinct ob_from_ol_id
            from operbook
            where ob_ol_date between :olp_date_begin and :olp_date_end and
                  ob_ol_is_active = 1 and
                  ob_ol_reg_id = 1010100 and
                  ob_account_dt = :ob_account_kt
            into :ol_id_tmp
        do
        begin
          k_id = 0;
          select distinct ol_id, ok_from_k_id,ol_memo
          from operkred, operlist
          where ok_from_ol_id = ol_id and
                ok_ol_is_active = 1 and
                (ol_id = :ol_id_tmp or
                ol_parent_id = :ol_id_tmp) rows 1
          into :ol_id, :k_id,:res_ol_memo;
          if (k_id is null) then
            k_id = 0;
          total_client = 1;
          if (k_id = 0) then
          begin
            select oko_from_ol_id, oko_from_k_id
            from operkroff, operlist
            where oko_from_ol_id = ol_id and
                  (ol_id = :ol_id_tmp or
                  ol_parent_id = :ol_id_tmp) and
                  operlist.is_delete = 0 rows 1
            into :ol_id, :k_id;
          end
          if (k_id <> 0) then
          begin
            select count(*)
            from knclient
            where kc_from_k_id = :k_id and
                  kc_amount_approved > 0
            into :total_client;
            if (total_client is null) then
              total_client = 1;

            is_sud = 0;
            if (res_date >= '01.10.2016') then
            begin
              if (exists(select *
                         from operextra
                         where ox_object_kind = 9 and
                               ox_object_id = :k_id and
                               ox_ol_is_active = 1 and
                               ox_ol_date <= :res_date and
                               ox_from_rd_id = :law_rd_id)) then
                is_sud = 1;
            end

            res_amount_tmp = :res_amount;

            -- Кредит погашен через банк, заменяем 57-04 на кредитные счета
            for select :res_acc_kt_tmp, substring(ob_account_kt from 8 for 5), ob_summa,
                       cast(right(ob_account_dt, 6) as integer), cast(right(ob_account_kt, 6) as integer)
                from operbook
                where ob_from_ol_id = :ol_id and
                      ob_ol_is_active = 1
                into :res_account_dt, :res_account_kt, res_amount, :num_dt, :num_kt
            do
            begin
              err_tmp = 0;
              client_no = 0;
              res_amount_orig = res_amount;
              total_amount = 0;
              -- если кредитная операция
              for select c_id,
                         iif(c_type = 1, iif(c_name_fio like 'ИП %' or c_name_fio like '% ИП', 'ИП', 'ЮЛ'), 'ФЗ'),
                         c_name_fio, c_is_resident, replace(rd_code, 'citizen.', ''), c_citizen,
                         kc_amount_approved / k_amount_approved, c_birth_date, kc_id, k_kontract_no
                  from clients, refdata, knclient, kontract
                  where c_id = kc_from_c_id and
                        c_citizen = rd_id and
                        kc_amount_approved > 0 and
                        kc_from_k_id = :k_id and
                        k_id = :k_id
                  into :c_id_tmp, :res_client_type, res_client_name, res_client_is_resident, citizen_xx, c_citizen,
                       client_koef, res_client_date, res_from_kc_id, res_kontract_no
              do
              begin
                res_client_id = c_id_tmp;

                res_acc_code = '';
                if (res_from_kc_id is not null) then
                begin
                  select iif(non_resident = 1, '492', '488')
                  from hp_get_kn_client_resident(:res_from_kc_id, :rep_date_end)
                  into res_acc_code;
                end

                client_no = client_no + 1;
                if (client_koef = 1) then
                  res_amount = res_amount_orig;
                else
                begin
                  if (client_no < total_client) then
                  begin
                    res_amount = round(cast(res_amount_orig as double precision) * client_koef, 0);
                    total_amount = total_amount + res_amount;
                  end
                  else
                  begin
                    res_amount = res_amount_orig - total_amount;
                  end
                end

                if (res_client_id = 1010100658562) then
                begin
                  res_client_id = 1010100658562;
                end

                if (res_client_type = 'ФЗ') then
                begin
                  select 1 - res_non_resident, res_country_code, res_country_id
                  from hp_get_client_type(:res_client_id, :citizen_ru_id, :doc_vnj_id, :res_date)
                  into :res_client_is_resident, citizen_xx, c_citizen;
                end
                else
                begin
                  if (citizen_ru_id = c_citizen) then
                    res_client_is_resident = 1;
                end

                --                if (citizen_xx <> citizen_xx_prior) then
                --                begin
                --                  res_client_country = '';
                if (citizen_xx <> '') then
                begin
                  select rd_memo
                  from refdata
                  where rd_main_code = 'CountryOKSM' and
                        rd_str_id = :citizen_xx and
                        rd_parent_id is not null rows 1
                  into :res_client_country;
                end
                --                  citizen_xx_prior = citizen_xx;
                --                end

                if (res_client_is_resident = 0) then
                  res_client_resident_str = 'Нерезидент';
                else
                  res_client_resident_str = 'Резидент';

                --                if (res_client_country = '') then
                --                begin
                --                  select rd_memo
                --                  from refdata
                --                  where rd_main_code = 'CountryOKSM' and
                --                        rd_str_id = :citizen_xx and
                --                        rd_parent_id is not null rows 1
                --                  into :res_client_country;
                --
                --                  citizen_xx_prior = citizen_xx;
                --                end

                if (res_client_country = '') then
                  res_error = 1;

                if ((res_account_dt starting '51-' and
                    res_account_kt starting '51-') or (res_account_dt starting '51-' and
                    res_account_kt starting '50-') or (res_account_dt starting '50-' and
                    res_account_kt starting '51-') or (res_account_dt starting '50-' and
                    res_account_kt starting '57-01') or (res_account_dt starting '57-01' and
                    res_account_kt starting '50-') or (res_account_dt starting '57-01' and
                    res_account_kt starting '51-')) then
                begin
                  res_client_name = cm_name;
                  res_client_is_resident = 1;
                  res_client_resident_str = 'Резидент';
                  res_client_country = refault_code;
                  res_client_type = 'ЮЛ';
                end
                if (res_account_dt starting '70-01' and
                    res_account_kt starting '51-') then
                  res_client_type = 'ПФЛ';
                if (res_client_country = '') then
                  res_client_country = refault_code;

                select res_opercode_rd_id, res_opercode_str, res_error
                from aarep_get_moneyoper_code(:res_account_dt, :res_account_kt, :is_sud * 100 + :res_client_is_resident, :num_dt, :num_kt)
                into :res_opercode_rd_id, :res_opercode_str, :err_tmp;
                if (res_error = 0) then
                  res_error = err_tmp;

                insert into temp_moneyclassific (res_ol_id, res_ob_number, res_date, res_doc_no, res_doc_name,
                                                 res_account_dt, res_account_kt, res_currency, res_amount, res_is_bank,
                                                 res_oper_sign, res_client_id, res_bank_id, res_error, res_client_name,
                                                 res_client_is_resident, res_client_type, res_client_resident_str,
                                                 res_client_country, res_opercode_rd_id, res_opercode_str,
                                                 res_oper_sign_str, res_region_id, res_from_kc_id, res_kontract_no,
                                                 res_client_date, res_acc_code,res_ol_memo)
                values (:res_ol_id, :res_ob_number, :res_date, :res_doc_no, :res_doc_name, :res_account_dt,
                        :res_account_kt, :res_currency, :res_amount, :res_is_bank, :res_oper_sign, :res_client_id,
                        :res_bank_id, :res_error, :res_client_name, :res_client_is_resident, :res_client_type,
                        :res_client_resident_str, :res_client_country, :res_opercode_rd_id, :res_opercode_str,
                        :res_oper_sign_str, :res_region_id, :res_from_kc_id, :res_kontract_no, :res_client_date,
                        :res_acc_code, left(:res_ol_memo, 250));
              end
            end
          end -- конец K_ID<>0
        end -- конец for select

      end
