CREATE PROCEDURE HP_EDOC_SIGN_BLANK_SET(
    EDS_ID TYPE OF D_ID,
    EDB_ID TYPE OF D_ID,
    EDB_BLOB TYPE OF D_BLOB_BINARY)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable lang d_str_lang;
begin
  -- Процедура сохранения подписанного документа для сервиса интеграции с sima.az
  error_text = '';
  if (f_get_db_id() <> 94401) then
    exception e_default 'Not implemented';

  lang = f_get_lang_code(null, null); -- Это робот, берем настройку языка БД по-умолчанию

  -- Проверяем, что наша блокировка действительна
  if (not exists(
      select *
      from edoc_sign_lock
      where esl_from_eds_id = :eds_id)) then
  begin
    -- Если нет, выходим, нет смысла сохранять
    error_text = f_lang(lang, 'Время для подписания истекло');
    suspend;
    exit;
  end

  -- Сохраняем блоб в edoc_blank_taken
  update edoc_blank_taken
  set ebt_blob = :edb_blob
  where ebt_from_edb_id = :edb_id
    and ebt_from_eds_id = :eds_id;

  -- Если все подписанные файлы получены (блобы сохранены)
  if (not exists(
    select * from edoc_blank_taken
    where ebt_from_eds_id = :eds_id
      and ebt_blob is null)) then
  begin
    -- Подписанные бланки из edoc_blank_taken переносим в edoc_blank
    for select ebt_from_edb_id, ebt_blob
    from edoc_blank_taken
    where ebt_from_eds_id = :eds_id
    as cursor cur
    do
      update edoc_blank
      set edb_blob = cur.ebt_blob
      where edb_id = cur.ebt_from_edb_id;

    -- Отправлять сообещния в АЗ не планируется, поэтому edoc_msg не подписываем,
    -- но подписываем edoc_sign
    update edoc_sign
    set eds_status = 1,
        eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'sima', 'id')
    where eds_id = :eds_id and
          (eds_status <> 1 or
          eds_enum_edoc_sign_kind <> f_enum('edoc_sign_kind', 'sima', 'id'));

    execute procedure hp_edoc_sign_unlock(eds_id);
  end
  suspend;
end
