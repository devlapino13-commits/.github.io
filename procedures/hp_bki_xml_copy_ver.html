CREATE PROCEDURE HP_BKI_XML_COPY_VER(
    OLD_VER_ID D_ID,
    NEW_VER_ID D_ID,
    OLD_PARENT_ID D_ID = null,
    NEW_PARENT_ID D_ID = null)
AS
declare variable bfv_position d_integer;
  declare variable bfv_from_bxf_id d_id;
  declare variable bfv_forced_value d_str_255;
  declare variable new_bfv_id d_id;
  declare variable old_bfv_id d_id;
  declare variable shift_pos d_integer;
begin
  -- процедура делает копию конфигурации событий
  select coalesce(max(bfv_position), 0)
  from bki_xml_fields_ver
  where bfv_from_bxv_id = :new_ver_id
    and ((:new_parent_id is null and bfv_parent_id is null)
      or bfv_parent_id = :new_parent_id)
  into shift_pos;  -- приращение позиции

  for select bfv_id, bfv_position, bfv_from_bxf_id, bfv_forced_value, bfv_parent_id
  from bki_xml_fields_ver
  where bfv_from_bxv_id = :old_ver_id
    and ((:old_parent_id is null and bfv_parent_id is null)
      or bfv_parent_id = :old_parent_id)
  into old_bfv_id, bfv_position, bfv_from_bxf_id, bfv_forced_value, old_parent_id
  do
  begin
    insert into bki_xml_fields_ver (bfv_from_bxv_id, bfv_position, bfv_from_bxf_id, bfv_forced_value, bfv_parent_id)
    values (:new_ver_id, :bfv_position + :shift_pos, :bfv_from_bxf_id, :bfv_forced_value, :new_parent_id)
    returning bfv_id
    into :new_bfv_id;

    execute procedure hp_bki_xml_copy_ver(old_ver_id, new_ver_id, old_bfv_id, new_bfv_id);
  end 
end
