CREATE PROCEDURE UXP$PREBLANK_ACTIONS(
    PREBLANK$KIND D_STR_CODE,
    PREBLANK$CODE D_STR_CODE,
    IN_OBJ_KIND D_STR_CODE,
    IN_OBJ_ID D_STR_CODE)
RETURNS (
    RES_UA_UUID D_UUID,
    RES_BLN_ID D_INTEGER,
    RES_REQUIRED D_BOOL,
    RES_ID D_SMALLINT,
    RES_BLS_ID D_ID,
    RES_BLANK_NAME D_STR_255)
AS
declare variable ufm_uuid d_uuid;
declare variable ux_draft_uuid d_uuid;
declare variable preblank_kind d_str_code;
declare variable edc_id d_integer;
declare variable pr_id d_integer;
declare variable obj_kind d_str_code;
declare variable sawt_id d_integer;
declare variable res_disable_date d_date;
declare variable order_c_id d_id;
declare variable ps_is_on_staff d_bool;
declare variable ps_is_part d_bool;
declare variable ps_is_mount d_bool;
declare variable sab_with_mount d_bool;
declare variable sab_without_mount d_bool;
declare variable sab_for_staff d_bool;
declare variable sab_for_freelancer d_bool;
declare variable sab_for_parttime d_bool;
begin
  preblank_kind = f_enum('uxaction_preblank_kind', preblank$kind, 'code');
  preblank$code = trim(coalesce(preblank$code, ''));

  select ux_draft_uuid
  from ux$code
  where ux_fcode = 'fm_ux$preblanks'
  into ufm_uuid;

  if (preblank_kind = 'hr') then
  begin
    if (coalesce(in_obj_kind, '') = '') then
      in_obj_kind = 'sa_worktype';

    if (coalesce(in_obj_id, '') = '') then
      in_obj_id = preblank$code;

    obj_kind = f_objkind(in_obj_kind, 'name');

    if (obj_kind = 'sa_worktype') then
    begin
      sawt_id = :in_obj_id;
    end else
    if (obj_kind = 'hr_orders') then
    begin
      select ho_order_type
      from hr_orders
      where ho_id = :in_obj_id
      into sawt_id;

      select ps_from_c_id, ps_is_on_staff, ps_is_part, ps_is_mount
      from hr_pstatus
      where ps_from_ho_id = :in_obj_id
      order by ps_id
      rows 1
      into order_c_id, ps_is_on_staff, ps_is_part, ps_is_mount;

      if (order_c_id is null) then
      begin
        select hrv_from_c_id
        from hr_voctravel
        where hrv_from_ho_id = :in_obj_id
        order by hrv_number
        rows 1
        into order_c_id;

        if (order_c_id is not null) then
        begin
          select ps_is_on_staff, ps_is_part, ps_is_mount
          from hr_pstatus
          where ps_from_c_id = :order_c_id and
                ps_date_begin <= current_date and
                ps_date_begin <= ps_date_end and
                ps_status = 1
          order by ps_date_begin desc
          rows 1
          into ps_is_on_staff, ps_is_part, ps_is_mount;
        end
      end
    end

    if (sawt_id is null) then
      exit;
  end else
  if (preblank_kind = 'edoc') then -- бланки которых нужно сформировать
  begin
    if (edc_id is null) then
      edc_id = f_edoc_kind(preblank$code, 'id');

    if (edc_id is null) then
      Exit;
  end else
  if (preblank_kind = 'eblank') then -- бланки уже сформированного эл.документа
  begin
    if (obj_kind <> 'edoc_list') then
      exception e_message 'Требуется Id edoc_list';
  end else
  if (preblank_kind = 'loan') then
  begin
    if (coalesce(in_obj_kind, '') = '') then
      in_obj_kind = 'product';

    if (coalesce(in_obj_id, '') = '') then
      in_obj_id = preblank$code;

    obj_kind = f_objkind(in_obj_kind, 'name');

    if (obj_kind = 'product') then
      pr_id = in_obj_id;
    else
    if (obj_kind = 'kontract') then
    begin
      select k_from_pr_id
      from kontract
      where k_id = :in_obj_id
      into pr_id;
    end

    if (pr_id is null) then
      exit;
  end

  res_blank_name = '';
  if (preblank_kind = 'eblank') then -- бланки уже сформированного эл.документа
  begin
    res_required = 1;
    res_id = 1;

    for select edb_from_bln_id, edb_from_bls_id, edf_print_name
    from edoc_blank, edoc_frx
    where edb_from_edl_id = :in_obj_id and
          coalesce(edb_from_bls_id, 0) <> 0 and
          edb_from_edf_id = edf_id
    into res_bln_id, res_bls_id, res_blank_name
    do
      suspend;
  end

  res_blank_name = '';
  res_id = 0;
  for select ua_uuid, ux_draft_uuid, ua_name
  from ux$action, ux$alias, ux$code, ux$uniq
  where ua_owner = :ufm_uuid and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = ux_uuid and
        ux_uuid = uu_id and
        uu_kind = 4 -- только отчеты
  into res_ua_uuid, ux_draft_uuid, res_blank_name
  do
  begin
    res_bln_id = null;
    res_required = 0;
    res_bls_id = null;

    select urt_import_id
    from ux$report
    where urt_uuid = :ux_draft_uuid and
          (urt_kind = 3 or urt_import_kind = 2)
    into res_bln_id;

    if (res_bln_id is null) then
      Continue;

    select max(blb_date_end)
    from blankblob
    where blb_from_bln_id = :res_bln_id and
          (blb_blob is not null or blb_frx_blob is not null)
    into res_disable_date;

    if (res_disable_date < current_date) then
      Continue;

    if (preblank_kind = 'code') then
    begin
      res_id = uxf$preblanks_code(preblank$code, res_bln_id);
      if (res_id = 0) then
        Continue;

      if (res_id = -1) then
        res_required = 1;

      res_id = abs(res_id);
    end else
    if (preblank_kind = 'blank') then
    begin
      if (preblank$code <> res_bln_id) then
        Continue;
    end else
    if (preblank_kind = 'loan') then
    begin
      if (not exists(select * from productblank
                     where prb_from_pr_id = :pr_id and
                           prb_from_bln_id = :res_bln_id and
                           prb_date_begin >= current_date and
                           coalesce(prb_date_end, current_date) >= current_date)) then
        Continue;
    end else
    if (preblank_kind = 'hr') then
    begin
      select sab_with_mount, sab_without_mount, sab_for_staff, sab_for_freelancer,
             sab_for_parttime
      from sa_blank
      where sab_from_sawt_id = :sawt_id and
            sab_from_bln_id = :res_bln_id
      into :sab_with_mount, :sab_without_mount, :sab_for_staff, :sab_for_freelancer, :sab_for_parttime;

      if (row_count = 0) then
        Continue;

      if (not((ps_is_mount = 1 and sab_with_mount = 1) or
          (ps_is_mount = 0 and sab_without_mount = 1) or
          (ps_is_on_staff = 1 and sab_for_staff = 1) or
          (ps_is_on_staff = 0 and sab_for_freelancer = 1) or
          (ps_is_part = 1 and sab_for_parttime = 1))) then
        Continue;
    end else
    if (preblank_kind = 'edoc') then
    begin
      select 1 - edf_optional_sign
      from edoc_frx
      where edf_from_edc_id = :edc_id and
            edf_from_bln_id = :res_bln_id
      into res_required;

      if (row_count = 0) then
        Continue;
    end else
    if (preblank_kind = 'eblank') then
    begin
      select edb_from_bls_id, edf_print_name
      from edoc_blank, edoc_frx
      where edb_from_edl_id = :in_obj_id and
            edb_from_bln_id = :res_bln_id and
            coalesce(edb_from_bls_id, 0) = 0 and
            edb_from_edf_id = edf_id
      into res_bls_id, res_blank_name;

      if (row_count = 0) then
        Continue;
    end else
      exception e_default 'Undefined blank_kind: ' || preblank_kind;

    suspend;
  end
end
