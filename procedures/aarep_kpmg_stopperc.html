CREATE PROCEDURE AAREP_KPMG_STOPPERC(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(50),
    K_DISB_DATE DATE,
    OKS_BALANCE CURR_TYPE,
    PERC_BEGIN_SALDO CURR_TYPE,
    PERC_CALCED CURR_TYPE,
    WRITEOFF_PERC CURR_TYPE,
    REG_NAME STR_100,
    PERC_STOP_DATE DATE,
    CU_SYMBOL VARCHAR(3),
    K_PERCENT PERC_TYPE,
    LOAN_TYPE VARCHAR(20),
    PERC_SALDO CURR_TYPE,
    CASHIER_FIO STR_100)
AS
declare variable k_id id_type;
declare variable year_buf varchar(4);
declare variable month_buf varchar(2);
declare variable rep_date_begin_csp integer;
declare variable rep_date_end_csp integer;
declare variable i integer;
declare variable kcp_out_perc_calced curr_type;
declare variable kcp_out_perc_begin_saldo curr_type;
begin
  for select sum(oks_balance), ksp.ksp_from_k_id, reg.reg_name, ksp.ksp_date
      from oksummary oks
      inner join kontstopperc ksp on ksp.ksp_from_k_id = oks.oks_from_k_id
      inner join region reg on reg.reg_unicode = oks_region_id
      where oks_close_state = 0 and
            :rep_date_end between oks_date_begin and oks_date_end and
            ksp.ksp_date <= :rep_date_end
      group by ksp_from_k_id, reg_name, ksp.ksp_date
      into :oks_balance, :k_id, :reg_name, :perc_stop_date
  do
  begin
    select kont.k_kontract_no, kont.k_fact_disb_date, cu.cu_symbol, kont.k_percent,
           iif(pr.pr_loan_type = 0, 'Групповой', 'Индивидуальный'), cl.c_name_fio
    from kontract kont
    inner join currlist cu on cu.cu_id = kont.k_currency
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    inner join clients cl on cl.c_id = k_clients_cashier_id
    where kont.k_id = :k_id
    into :k_kontract_no, :k_disb_date, :cu_symbol, :k_percent, :loan_type, :cashier_fio;

    year_buf = extract(year from rep_date_begin);
    month_buf = extract(month from rep_date_begin);
    if (month_buf < 10) then
      month_buf = '0' || month_buf;

    rep_date_begin_csp = cast(year_buf || month_buf as integer);

    year_buf = extract(year from rep_date_end);
    month_buf = extract(month from rep_date_end);
    if (month_buf < 10) then
      month_buf = '0' || month_buf;

    rep_date_end_csp = cast(year_buf || month_buf as integer);

    i = 0;
    perc_begin_saldo = 0;
    perc_calced = 0;

    for select ksp.kcp_out_perc_begin_saldo, ksp.kcp_out_perc_calced
        from kontcalcperc ksp
        where ksp.kcp_period between :rep_date_begin_csp and :rep_date_end_csp and
              ksp.kcp_from_k_id = :k_id
        order by kcp_period
        into :kcp_out_perc_begin_saldo, kcp_out_perc_calced
    do
    begin
      if (i = 0) then
      begin
        perc_begin_saldo = kcp_out_perc_begin_saldo;
        i = 1;
      end
      perc_calced = perc_calced + kcp_out_perc_calced;
    end

    perc_saldo = perc_begin_saldo + perc_calced;

    writeoff_perc = 0;

    select sum(-ok.ok_perc)
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_type = 5 and
          ok.ok_ol_is_active = 1 and
          ok.ok_perc < 0 and
          ok.ok_ol_date between :rep_date_begin and :rep_date_end
    into :writeoff_perc;

    writeoff_perc = coalesce(writeoff_perc, 0);

    suspend;
  end
end
