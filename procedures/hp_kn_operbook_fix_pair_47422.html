CREATE PROCEDURE HP_KN_OPERBOOK_FIX_PAIR_47422(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    ORIG_ACC_ID D_ID,
    NEW_ACC_ID D_ID,
    ORIG_ACC_CODE D_STR_CODE,
    NEW_ACC_CODE D_STR_CODE,
    RES_ACTION D_STR_CODE,
    RES_COUNT D_INTEGER)
AS
declare variable kc_id d_id;
begin
  for select acc_id, acc_pair_acc_id, f_fmt_acc(acc_id), f_fmt_acc(acc_pair_acc_id)
  from ref_account
  where acc_code in ('47422810009900000005', '47422810009900000007', '47422810009900000008')
  into orig_acc_id, new_acc_id, orig_acc_code, new_acc_code
  do
  begin
    update kn_operbook
    set kob_acc_dt = :new_acc_id
    where kob_acc_dt = :orig_acc_id and
          kob_ol_date between :rep_date_begin and :rep_date_end and
          kob_ol_is_active = 1;

    res_action = 'Замена счетов по ДТ';
    res_count = row_count;
    suspend;

    update kn_operbook
    set kob_acc_kt = :new_acc_id
    where kob_acc_kt = :orig_acc_id and
          kob_ol_date between :rep_date_begin and :rep_date_end and
          kob_ol_is_active = 1;

    res_action = 'Замена счетов по КТ';
    res_count = row_count;
    suspend;

    delete from kn_operbook
    where kob_ol_date between :rep_date_begin and :rep_date_end and
          kob_ol_is_active = 1 and
          kob_acc_kt = kob_acc_dt and
          kob_acc_dt = :new_acc_id;

    res_action = 'Удалены проводки где ДТ = КТ';
    res_count = row_count;
    suspend;
  end

  for select acc_id, pair_acc_id, kc_id
  from aarep_kn_operbook_fix_pair_list(:rep_date_end)
  into :orig_acc_id, :new_acc_id, :kc_id
  do
  begin
    res_action = 'Парные счета займа ' || kc_id;
    res_count = 1;
    execute procedure aarep_kn_operbook_fix_zero_pair(:rep_date_end, :kc_id);
    suspend;
  end
end
