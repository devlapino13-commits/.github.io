CREATE PROCEDURE AAREP_CLIENTS_INFO_FOR_CIO(
    CLIENT_ID ID_TYPE,
    REP_DATE D_DATE = current_date)
RETURNS (
    C_ID ID_TYPE,
    C_TYPE D_INTEGER,
    C_NAME_FIO D_STR_100,
    C_SEX D_SMALLINT,
    C_BIRTH_DATE D_DATE,
    C_NATION D_INTEGER,
    C_CITIZEN D_INTEGER,
    C_PASS_NO D_STR_CODE,
    C_PASS_DATE D_DATE,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE_EXPIRE D_DATE,
    C_FROM_REG_ID D_INTEGER,
    C_EMAIL D_STR_100,
    C_FAMILY D_INTEGER,
    C_EDUCATION D_INTEGER,
    C_KOL_FAMILY D_INTEGER,
    C_KOL_CHILD D_INTEGER,
    C_MEMO D_STR_LONG,
    C_HIRED_DATE D_DATE,
    C_IS_RESIDENT D_SMALLINT,
    C_ARMY D_INTEGER,
    C_TABEL D_STR_10,
    C_PLASTCARD_NO D_STR_20,
    C_OTHERS D_STR_20,
    C_IS_PERSONAL D_SMALLINT,
    C_DOC_TYPE D_INTEGER,
    C_NATION_VIEW D_STR_100,
    C_CITIZEN_VIEW D_STR_100,
    C_FAMILY_VIEW D_STR_100,
    C_EDUCATION_VIEW D_STR_100,
    C_ARMY_VIEW D_STR_100,
    C_SEX_VIEW D_STR_10,
    C_BANK_NAME D_STR_255,
    C_IS_RESIDENT_TEXT D_STR_10,
    C_IS_PERSONAL_TEXT D_STR_10,
    C_PHOTO D_BLOB_BINARY,
    C_ADDR_REGIST_1_ADDRESS D_STR_255,
    C_ADDR_FACT_1_ADDRESS D_STR_255,
    C_ADDR_REGIST_2_ADDRESS D_STR_255,
    C_ADDR_FACT_2_ADDRESS D_STR_255,
    INFO_REGION D_STR_100,
    INFO_JOBNAME D_STR_100,
    INFO_REGION_PHONE D_STR_255,
    INFO_REGION_LEGAL_ADDRESS D_STR_255,
    INFO_REGION_FACT_ADDRESS D_STR_255,
    INFO_REGION_OBLAST D_STR_255,
    INFO_START_WORK D_DATE,
    C_CELL_NO_DATE D_DATE,
    C_SKYPE D_STR_255,
    INFO_REGION_IP_PHONE D_STR_100,
    C_INN D_STR_20,
    C_BIRTH_CITY D_STR_100,
    C_ADR_REG_PHONE D_STR_255,
    C_ADR_FACT_PHONE D_STR_255,
    C_WORKED_YEARS D_INTEGER,
    C_WORKED_MONTHS D_INTEGER,
    C_WORKED_DAYS D_INTEGER,
    C_WORKED_DAYS_TOTAL D_INTEGER)
AS
declare variable st_staff d_integer;
    declare variable st_client d_integer;
    declare variable c_blob_photo_id d_integer;
    declare variable adc_address_type d_integer;
    declare variable adc_phone d_str_255;
    declare variable adc_memo d_str_255;
    declare variable adl_address_str d_str_255;
    declare variable adl_comment d_str_255;
    declare variable adc_datetime d_datetime;
    declare variable ps_is_working d_integer;
    declare variable reg_address_fact id_type;
    declare variable reg_address_legal id_type;
    declare variable date_begin d_date;
    declare variable date_end d_date;
    declare variable prev_date_end d_date;
    declare variable y d_integer;
    declare variable m d_integer;
    declare variable d d_integer;
    declare variable d_total d_integer;
begin
  select st_user_client_access,
         st_user_staff_access
  from get_stored_proc_options
  into :st_client,
       :st_staff;

  select c_id,
         c_type,
         c_name_fio,
         c_sex,
         c_birth_date,
         c_nation,
         c_citizen,
         c_pass_no,
         c_pass_date,
         c_pass_who,
         c_pass_date_expire,
         c_from_reg_id,
         f_cont_info('clients', c_id, 'email', 'last'),
         c_family,
         c_education,
         c_kol_family,
         c_kol_child,
         c_memo,
         c_hired_date,
         c_is_resident,
         c_army,
         c_tabel,
         f_cont_info('clients', c_id, 'cell', 'last'),
         c_others,
         c_is_personal,
         c_doc_type,
         d1.rd_name,
         d2.rd_name,
         d3.rd_name,
         d4.rd_name,
         d5.rd_name,
         c_blob_photo_id,
         f_cont_info('clients', c_id, 'skype', 'last'),
         c_inn,
         c_birth_city
  from clients, refdata d1, refdata d2, refdata d3, refdata d4, refdata d5
  where (c_nation = d1.rd_id) and
        (c_citizen = d2.rd_id) and
        (c_family = d3.rd_id) and
        (c_education = d4.rd_id) and
        (c_army = d5.rd_id) and
        c_id = :client_id
  into :c_id,
       :c_type,
       :c_name_fio,
       :c_sex,
       :c_birth_date,
       :c_nation,
       :c_citizen,
       :c_pass_no,
       :c_pass_date,
       :c_pass_who,
       :c_pass_date_expire,
       :c_from_reg_id,
       :c_email,
       :c_family,
       :c_education,
       :c_kol_family,
       :c_kol_child,
       :c_memo,
       :c_hired_date,
       :c_is_resident,
       :c_army,
       :c_tabel,
       :c_plastcard_no,
       :c_others,
       :c_is_personal,
       :c_doc_type,
       :c_nation_view,
       :c_citizen_view,
       :c_family_view,
       :c_education_view,
       :c_army_view,
       :c_blob_photo_id,
       :c_skype,
       :c_inn,
       :c_birth_city;

  c_cell_no_date = null;
  if (f_get_const('new_contdata') = 0) then
  begin
    select res_date
    from hp_get_client_continfo(:c_id, 'ContactInfo.Cell_Main')
    into c_cell_no_date;
  end
  else
  begin
    select cda_date
    from contdata
    where cda_from_c_id = :c_id and
          cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
          cda_is_main = 1 and
          cda_is_delete = 0
    order by cda_status desc,
        cda_id desc rows 1
    into c_cell_no_date;
  end

  if (c_bank_name is null) then
    c_bank_name = '';
  if (c_blob_photo_id > 0) then
  begin
    select blx_blob
    from hp_get_blobex_blob(:c_blob_photo_id)
    into :c_photo;
  end
  else
  begin
    select c1.cb_blob
    from clientblob c1
    where c1.cb_from_c_id = :c_id and
          c1.cb_id = (select min(c2.cb_id)
                      from clientblob c2
                      where c2.cb_from_c_id = :c_id and
                            cb_type = 0)
    into :c_photo;
  end

  if (c_sex = 0) then
    c_sex_view = 'Женский';
  else
    c_sex_view = 'Мужской';

  if (c_is_resident = 1) then
    c_is_resident_text = 'Да';
  else
    c_is_resident_text = 'Нет';

  if (c_is_personal = 1) then
    c_is_personal_text = 'Сотрудник';
  else
    c_is_personal_text = 'Клиент';

  c_addr_regist_1_address = '';
  c_addr_regist_2_address = '';
  c_addr_fact_1_address = '';
  c_addr_fact_2_address = '';

  c_adr_reg_phone = '';
  c_adr_fact_phone = '';
  for select adc_address_type,
             adc_phone,
             adc_memo,
             adc_datetime,
             adl_address_str,
             adl_comment
      from addrclient, addrlist
      where adc_from_c_id = :client_id and
            adc_from_adl_id = adl_id and
            mod(adc_from_adl_id, 1000000) > 0
      into :adc_address_type,
           :adc_phone,
           :adc_memo,
           :adc_datetime,
           adl_address_str,
           adl_comment
  do
  begin
    if (adc_phone is null) then
      adc_phone = '';
    if (adc_memo is null) then
      adc_memo = '';
    if (adl_address_str is null) then
      adl_address_str = '';
    if (adl_comment is null) then
      adl_comment = '';

    if (adc_address_type = 1) then
    begin
      if ((adl_address_str is not null) and (adl_address_str <> '')) then
        c_addr_regist_1_address = c_addr_regist_1_address || ' ' || adl_address_str;

      if ((adc_phone is not null) and (adc_phone <> '')) then
        c_addr_regist_1_address = c_addr_regist_1_address || ' ' || 'тел.' || adc_phone;

      if ((adl_comment is not null) and (adl_comment <> '')) then
        c_addr_regist_1_address = c_addr_regist_1_address || ' Примеч.' || adl_comment;

      c_adr_reg_phone = adc_phone;
    end
    if (adc_address_type = 2) then
    begin
      if ((adl_address_str is not null) and (adl_address_str <> '')) then
        c_addr_fact_1_address = c_addr_fact_1_address || ' ' || adl_address_str;

      if ((adc_phone is not null) and (adc_phone <> '')) then
        c_addr_fact_1_address = c_addr_fact_1_address || ' ' || 'тел.' || adc_phone;

      if ((adl_comment is not null) and (adl_comment <> '')) then
        c_addr_fact_1_address = left(c_addr_fact_1_address || ' Примеч.' || adl_comment, 255);
      c_adr_fact_phone = adc_phone;
    end
    if (adc_address_type = 4) then
    begin
      if ((adl_address_str is not null) and (adl_address_str <> '')) then
        c_addr_regist_2_address = c_addr_regist_2_address || ' ' || adl_address_str;

      if ((adc_phone is not null) and (adc_phone <> '')) then
        c_addr_regist_2_address = c_addr_regist_2_address || ' ' || 'тел.' || adc_phone;

      if ((adl_comment is not null) and (adl_comment <> '')) then
        c_addr_regist_2_address = c_addr_regist_2_address || ' Примеч.' || adl_comment;
    end
    if (adc_address_type = 5) then
    begin
      if ((adl_address_str is not null) and (adl_address_str <> '')) then
        c_addr_fact_2_address = c_addr_fact_2_address || ' ' || adl_address_str;

      if ((adc_phone is not null) and (adc_phone <> '')) then
        c_addr_fact_2_address = c_addr_fact_2_address || ' ' || 'тел.' || adc_phone;

      if ((adl_comment is not null) and (adl_comment <> '')) then
        c_addr_fact_2_address = c_addr_fact_2_address || ' Примеч.' || adl_comment;
    end
  end

  info_region = '';
  info_jobname = '';
  ps_is_working = null;
  info_region_legal_address = '';
  info_region_fact_address = '';
  info_region_oblast = '';
  reg_address_fact = 0;
  reg_address_legal = 0;

  info_start_work = null;
  select min(ps_date_begin)
  from hr_pstatus
  where ps_status = 1 and
        ps_from_c_id = :client_id
  into :info_start_work;
  info_region_ip_phone = '';

  for select reg_name,
             jl_name,
             ps_is_working,
             reg_phone,
             reg_address_fact,
             reg_address_legal,
             reg_stat_rep_region,
             reg_phone_ip
      from hr_pstatus, region, joblist
      where ps_status = 1 and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            ps_from_c_id = :client_id and
            ps_date_begin <= :rep_date and
            ps_date_end >= :rep_date
      order by ps_date_begin desc,
          ps_is_working rows 1
      into :info_region,
           :info_jobname,
           :ps_is_working,
           :info_region_phone,
           :reg_address_fact,
           :reg_address_legal,
           :info_region_oblast,
           :info_region_ip_phone
  do
  begin
    if (ps_is_working = 0) then
      info_jobname = 'уволен. ' || :info_jobname;
    if (mod(reg_address_legal, 1000000) > 0) then
    begin
      for select adl_address_str,
                 adl_comment
          from addrlist
          where adl_id = :reg_address_legal
          into :adl_address_str,
               adl_comment
      do
      begin
        if (adl_address_str is null) then
          adl_address_str = '';
        if (adl_comment is null) then
          adl_comment = '';

        if ((adl_address_str is not null) and (adl_address_str <> '')) then
          info_region_legal_address = info_region_legal_address || ' ' || adl_address_str;

        if ((adl_comment is not null) and (adl_comment <> '')) then
          info_region_legal_address = info_region_legal_address || ' Примеч.' || adl_comment;
      end
    end
    if (mod(reg_address_legal, 1000000) > 0) then
    begin
      for select adl_address_str,
                 adl_comment
          from addrlist
          where adl_id = :reg_address_fact
          into :adl_address_str,
               adl_comment
      do
      begin
        if (adl_address_str is null) then
          adl_address_str = '';
        if (adl_comment is null) then
          adl_comment = '';

        if ((adl_address_str is not null) and (adl_address_str <> '')) then
          info_region_fact_address = info_region_fact_address || ' ' || adl_address_str;

        if ((adl_comment is not null) and (adl_comment <> '')) then
          info_region_fact_address = info_region_fact_address || ' Примеч.' || adl_comment;
      end
    end
  end

  -- общий стаж сотрудника в МБулак (полных лет, месяцев, дней)
  c_worked_years = 0;
  c_worked_months = 0;
  c_worked_days = 0;
  c_worked_days_total = 0;
  y = 0;
  m = 0;
  d = 0;
  d_total = 0;

  -- перебираем периоды работы сотрудника
  for select ps_date_begin,
             ps_date_end
      from hr_pstatus
      where ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id = :client_id and
            ps_date_begin <= :rep_date -- начало периода не позже отчетной даты
      into date_begin,
           date_end
  do
  begin
    if (date_end > rep_date) then
      date_end = rep_date; -- если конец периода больше отчетной даты, то считаем до отчетной даты

    -- получаем кол-во полных лет/мес/дней в текущем периоде
    select cnt_years,
           cnt_months,
           cnt_days,
           days_total
    from aarep_staj(:date_begin, :date_end)
    into y,
         m,
         d,
         d_total;

    -- добавляем текущий период к предыдущим
    c_worked_years = c_worked_years + y;
    c_worked_months = c_worked_months + m;
    c_worked_days = c_worked_days + d;
    c_worked_days_total = c_worked_days_total + d_total;

    if (date_begin = prev_date_end + 1) then
      c_worked_days = c_worked_days - 1; -- текущий период идет сразу за предыдущим

    prev_date_end = date_end;
  end

  c_worked_months = c_worked_months + c_worked_days / 30;
  c_worked_days = mod(c_worked_days, 30);
  c_worked_years = c_worked_years + c_worked_months / 12;
  c_worked_months = mod(c_worked_months, 12);

  suspend;
end
