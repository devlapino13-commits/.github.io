Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE LLC_LAST_LLC_LIST(
    REP_DATE DATE)
RETURNS (
    LLC_KONT_ID DECIMAL(18,0),
    DELIQ_DAY_KOL INTEGER,
    BALANCE DECIMAL(18,4),
    LOAN_TYPE SMALLINT,
    PERCENT_KOEF DECIMAL(18,7),
    STATUS_ID INTEGER,
    REG_ID INTEGER)
AS
declare variable lc_id decimal(18,0);
begin
  /* Procedure Text */
  for select lm_id
      from llc_main
      where lm_date < :rep_date
      order by lm_date
      rows 1
      into :lc_id
  do
  begin
    for select lk_from_k_id
        from llc_kont
        where lk_from_lm_id = :lc_id
        into :llc_kont_id
    do
    begin
      for select deliq_day_kol, balance, loan_type, percent_koef, status_id, reg_id
          from llc_one_info(:llc_kont_id, :rep_date)
          into :deliq_day_kol, :balance, :loan_type, :percent_koef, :status_id, :reg_id
      do
        suspend;
    end
  end
end
