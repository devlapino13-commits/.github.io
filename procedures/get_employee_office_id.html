CREATE PROCEDURE GET_EMPLOYEE_OFFICE_ID(
    C_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    LO_REGION_ID INTEGER)
AS
begin
  select ps_from_reg_id
  from hr_pstatus
  where ps_from_c_id = :c_id and
        ps_date_begin <= :rep_date and
        ps_date_end >= :rep_date and
        ps_is_working = 1 and
        ps_status = 1
  into :lo_region_id;

  suspend;
end
