CREATE PROCEDURE HP_UXFRLOOKUP_UPDATE(
    UFP_CODE D_STR_100,
    ULP_CODE D_STR_100)
AS
declare variable fr_code d_str_code;
declare variable ulp_source d_str_code;
declare variable field_name d_str_code;
declare variable ufp_index d_integer;
declare variable field_no d_integer;
begin
  select ufp_ufr_code, ufp_index
  from uxfrparam
  where ufp_code = :ufp_code
  into fr_code, ufp_index;

  update or insert into uxfrlookup (ufl_ufp_code, ufl_ulp_code)
  values (:ufp_code, :ulp_code)
  matching (ufl_ufp_code);

  update uxfrparam
  set ufp_editor = 'lookup'
  where ufp_code = :ufp_code and
        ufp_editor <> 'lookup';

  select ulp_source
  from uxlookup
  where ulp_code = :ulp_code
  into ulp_source;

  update uxfrparam
  set ufp_state = -2
  where ufp_ufr_code = :fr_code and
        ufp_parent_code = :ufp_code;

  field_no = ufp_index * 1000;
  for select field_name
  from hp_ux_source_fields(:ulp_source)
  where param_kind = 0
  into :field_name
  do
  begin
    field_no = field_no + 1;
    update or insert into uxfrparam (ufp_parent_code, ufp_ufr_code, ufp_index, ufp_origin_name, ufp_is_param, ufp_state)
    values (:ufp_code, :fr_code, :field_no, :field_name, 1, 0)
    matching (ufp_ufr_code, ufp_origin_name);
  end

  delete from uxfrparam
  where ufp_ufr_code = :fr_code and
        ufp_parent_code = :ufp_code and
        ufp_state = -2;
end
