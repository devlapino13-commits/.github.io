CREATE PROCEDURE BLANK_ECASH_OUT_MAIN(
    OL D_ID)
RETURNS (
    BL_DATE D_DATE,
    BL_DOC_NO D_STR_20,
    BL_MEMO D_STR_LONG,
    BL_CLIENT_NAME D_STR_255,
    BL_PASS_NO D_STR_CODE,
    BL_PASS_DATE D_DATE,
    BL_PASS_WHO D_STR_255,
    BL_CLIENT_REGIST_NO D_STR_20,
    BL_CLIENT_OKPO D_STR_20,
    BL_CLIENT_INN D_STR_20,
    BL_OFFICE_FULL_NAME D_STR_255,
    BL_SUMMA D_CURRENCY,
    BL_ACCOUNT_DT D_STR_20,
    BL_ACCOUNT_KT D_STR_20,
    BL_IS_DELETE D_SMALLINT,
    BL_KASS_PLAN D_STR_10,
    BL_CORR_ACC D_STR_255,
    BL_COMPANY D_STR_255,
    BL_OFFICE_NAME D_STR_255,
    BL_BALANCE_ID D_STR_20,
    BL_REGION_CODE D_STR_10,
    BL_ACCOUNT_NO D_INTEGER,
    BL_BRANCH_FULL_NAME D_STR_255,
    BL_DATE_DAY D_INTEGER,
    BL_DATE_MONTH D_INTEGER,
    BL_DATE_YEAR D_INTEGER,
    BL_UN_FIO D_STR_100,
    BL_MANAGER_FIO D_STR_100,
    BL_CREDIT_ACC D_STR_20,
    BL_IPF_CLS D_STR_20,
    QR_CODE D_STR_LONG,
    EDOC_SIGN_DATETIME D_DATETIME,
    DATE_LIC_CHET D_STR_CODE,
    BL_CASHIER_FIO D_STR_100,
    BL_SAVE_DATETIME D_DATETIME)
AS
declare variable OFFICE varchar(255);
declare variable COMPANY_NAME varchar(255);
declare variable OL_MANAGER_C_ID decimal(18,0);
declare variable OPC_FROM_C_ID D_ID;
declare variable CASH_ID D_INTEGER;
declare variable OPC_IS_IPF D_BOOL;
declare variable EDL_ID D_ID;
declare variable CLIENT_ACC_NO D_STR_CODE;
declare variable C_BIRTH_DATE D_DATE;
declare variable K_ID D_ID;
declare variable OPC_FROM_OL_ID D_ID;
declare variable OPC_OFFICE_ID D_INTEGER;
declare variable OPC_CORR_REG_ID D_INTEGER;
declare variable TMP_OL_ID D_ID;
begin
  -- если передан НЕ OL_ID, возможно это ID из рефинансирования. OL_ID берем оттуда
  if (not exists(select * from opercash
                 where opc_id = :ol)) then
    exception e_default 'eBlank OperId not found';

  select opc_date, opc_doc_no, opc_memo, opc_is_delete,
         reg_name, reg_kont_company, reg_code,
         un_fio, opc_from_c_id, opc_manager_id,
         opc_corr_al_id, opc_from_csp_id, opc_is_ipf, opc_amount,
         opc_from_edl_id, opc_from_k_id, opc_from_ol_id, c_name_fio,
         opc_datetime, opc_office_id, opc_corr_reg_id
  from opercash
       inner join region on opc_office_id = reg_unicode
       inner join username on opc_from_un_id = un_id
       inner join clients on opc_cashier_id = c_id
  where opc_id = :ol
  into bl_date, bl_doc_no, bl_memo, bl_is_delete,
       office, bl_branch_full_name, bl_region_code,
       bl_un_fio, opc_from_c_id, ol_manager_c_id,
       bl_account_dt, cash_id, opc_is_ipf, bl_summa,
       edl_id, k_id, opc_from_ol_id, bl_cashier_fio,
       bl_save_datetime, opc_office_id, opc_corr_reg_id;

  if (row_count = 0) then
    exception e_default 'eBlank is empty';

  edoc_sign_datetime = null;
  qr_code = '';

  if (edl_id is not null) then
  begin
    qr_code = f_edoc_qr_by_edl(edl_id, opc_from_c_id);

    select max(eds_status_datetime)
    from edoc_sign
    where eds_from_edl_id = :edl_id and
          eds_status = 1
    into edoc_sign_datetime;
  end

  if (opc_is_ipf = 0) then
    bl_ipf_cls = 'Классика';
  else
    bl_ipf_cls = 'ИПФ';

  company_name = '';
  bl_balance_id = '';
  bl_account_no = null;
  bl_corr_acc = '';
  bl_kass_plan = '';

  bl_date_day = extract(day from :bl_date);
  bl_date_month = extract(month from :bl_date);
  bl_date_year = extract(year from :bl_date);

  if (bl_summa is null) then
    bl_summa = 0;

  select iif(:opc_is_ipf = 0, p.csp_cls_al_id, p.csp_ipf_al_id)
  from cashplace p
  where p.csp_id = :cash_id
  into bl_account_kt;

  if (bl_account_dt is not null) then
  begin
    select al_from_bg_id, al_number
    from acc_list
    where al_id = :bl_account_dt
    into bl_balance_id, bl_account_no;
  end else
  if (opc_from_ol_id is not null) then
  begin
    if (opc_office_id = opc_corr_reg_id) then
    begin
      select ob_account_dt, al_from_bg_id, al_number
      from operbook, acc_list
      where ob_from_ol_id = :opc_from_ol_id and
            ob_account_dt = al_id and
            ob_account_kt = :bl_account_kt
      rows 1
      into bl_account_dt, bl_balance_id, bl_account_no;
    end else
    begin
      select ol_id
      from operlist
      where ol_parent_id = :opc_from_ol_id and
            ol_from_reg_id = :opc_corr_reg_id
      order by ol_id desc
      rows 1
      into tmp_ol_id;

      select ob_account_dt, al_from_bg_id, al_number
      from operbook, acc_list
      where ob_from_ol_id = :tmp_ol_id and
            ob_account_dt = al_id
      rows 1
      into bl_account_dt, bl_balance_id, bl_account_no;
    end
  end else
  if (k_id is not null) then
  begin
    select k_kontract_no
    from kontract
    where k_id = :k_id
    into bl_account_dt;
  end

  bl_credit_acc = bl_account_kt;

  bl_manager_fio = '';
  if (ol_manager_c_id is not null) then
  begin
    select c_name_fio
    from clients
    where c_id = :ol_manager_c_id
    into :bl_manager_fio;
  end

  select c_name_fio, c_pass_no, c_pass_date, c_pass_who,
         c_reg_no, c_okpo, c_inn, c_birth_date
  from clients
  where c_id = :opc_from_c_id
  into bl_client_name, bl_pass_no, bl_pass_date, bl_pass_who,
       bl_client_regist_no, bl_client_okpo, bl_client_inn, c_birth_date;

  client_acc_no = '';
  
  if (opc_from_c_id is not null and c_birth_date is not null) then
  begin
    select uniq_acc_str
    from hp_show_clientuniq_acc_no(:opc_from_c_id)
    into :client_acc_no; -- № Лицевого счета
    
    date_lic_chet = extract(year from c_birth_date)
                    || lpad(extract(month from c_birth_date), 2, '0')
                    || lpad(extract(day from c_birth_date), 2, '0')
                    || '-' || client_acc_no;
  end

  select cm_name
  from company
  into company_name;

  bl_office_full_name = company_name || ' ' || office;
  bl_company = company_name;
  bl_office_name = bl_region_code || ' '|| office;

  suspend;
end
