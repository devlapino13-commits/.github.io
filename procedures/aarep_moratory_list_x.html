CREATE PROCEDURE AAREP_MORATORY_LIST_X(
    OPER_ID DECIMAL(18,0),
    OPER_NO INTEGER)
RETURNS (
    M_ID DECIMAL(18,0),
    M_NUMBER INTEGER,
    M_FROM_C_ID DECIMAL(18,0),
    M_LAST_K_ID DECIMAL(18,0),
    M_DATE_BEGIN DATE,
    M_DATE_END DATE,
    M_MONTH INTEGER,
    M_COMMENT_GROUP VARCHAR(2048),
    M_COMMENT_OFFICER VARCHAR(2048),
    M_COMMENT_MANAGER VARCHAR(2048),
    M_CLIENT_FIO VARCHAR(100),
    M_CLIENT_PASS_NO VARCHAR(30),
    M_CLIENT_ADDRESS VARCHAR(255),
    M_CLIENT_BIRTH_DATE DATE,
    M_FROM_REG_ID INTEGER,
    M_NO INTEGER,
    M_DATETIME TIMESTAMP,
    M_DATETIME_EDIT TIMESTAMP,
    M_USER_CREATE DECIMAL(18,0),
    M_USER_EDIT DECIMAL(18,0),
    M_IS_ACTIVE_TEXT VARCHAR(20),
    M_IS_BLACK_TEXT VARCHAR(50),
    M_OUR_CLIENTS_TEXT VARCHAR(50),
    M_REGION_NAME VARCHAR(100),
    M_USER_CREATE_NAME VARCHAR(130),
    M_USER_EDIT_NAME VARCHAR(130),
    M_MOTIVE_NAME VARCHAR(255),
    M_IS_FORCE_TEXT VARCHAR(100))
AS
declare variable m_is_active smallint;
declare variable m_is_black integer;
declare variable m_our_clients integer;
declare variable m_is_force integer;
begin
  for select m_id, m_number, m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end, m_month, m_is_active,
             m_comment_group, m_comment_officer, m_comment_manager, m_client_fio, m_client_pass_no, m_client_address,
             m_client_birth_date, m_our_clients, m_from_reg_id, m_no, m_datetime, m_datetime_edit, m_user_create,
             m_user_edit, reg_code || ' ' || reg_name, u1.un_name || ' ' || u1.un_fio, u2.un_name || ' ' || u2.un_fio,
             ms_name, ms_is_force
      from moratory, region, username u1, username u2, mormotive, morstatus
      where m_id = :oper_id and
            m_number = :oper_no and
            m_from_reg_id = reg_unicode and
            u1.un_id = m_user_create and
            u2.un_id = m_user_edit and
            mm_from_m_id = :oper_id and
            mm_number = :oper_no and
            mm_from_ms_id = ms_id
      order by ms_is_force, ms_order_no, ms_name
      into :m_id, :m_number, :m_from_c_id, :m_last_k_id, :m_is_black, :m_date_begin, :m_date_end, :m_month,
           :m_is_active, :m_comment_group, :m_comment_officer, :m_comment_manager, :m_client_fio, :m_client_pass_no,
           :m_client_address, :m_client_birth_date, :m_our_clients, :m_from_reg_id, :m_no, :m_datetime,
           :m_datetime_edit, :m_user_create, :m_user_edit, :m_region_name, :m_user_create_name, :m_user_edit_name,
           :m_motive_name, :m_is_force
  do
  begin
    if (m_is_active = 1) then
      m_is_active_text = 'Действующий';
    else
      m_is_active_text = 'Аннулирован';

    if (m_is_black = 0) then
      m_is_black_text = 'Мораторий';
    if (m_is_black = 1) then
      m_is_black_text = 'Отказан на определенный срок';
    if (m_is_black = 2) then
      m_is_black_text = 'Стоп лист';

    if (m_our_clients = 1) then
      m_our_clients_text = 'Клиент нашей компании';
    if (m_our_clients = 0) then
      m_our_clients_text = 'Клиент другой компании';

    if (m_is_force = 1) then
      m_is_force_text = 'Вынужденный';
    if (m_is_force = 0) then
      m_is_force_text = 'По собственному желанию';
    suspend;
  end
end
