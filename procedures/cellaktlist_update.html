CREATE PROCEDURE CELLAKTLIST_UPDATE(
    CELL_NO VARCHAR(20))
AS
declare variable akt_id decimal(18,0);
declare variable d date;
declare variable chet integer;
declare variable d_prior date;
declare variable line_id integer;
begin
  /* Procedure Text */
  chet = 0;
  for select cal_from_cam_id, cal_id, cal_date_begin
      from cellaktlist
      where cal_cell_no = :cell_no
      order by cal_date_begin desc, cal_from_cam_id desc, cal_id desc
      into :akt_id, :line_id, :d
  do
  begin
    chet = :chet + 1;
    if (chet > 1) then
    begin
      update cellaktlist
      set cal_date_end = :d_prior - 1
      where cal_cell_no = :cell_no and
            cal_id = :line_id and
            cal_from_cam_id = :akt_id and
            cal_date_end <> :d_prior - 1;
    end
    else
    begin
      update cellaktlist
      set cal_date_end = '31.12.9999'
      where cal_cell_no = :cell_no and
            cal_id = :line_id and
            cal_from_cam_id = :akt_id and
            cal_date_end <> '31.12.9999';
    end
    d_prior = d;
  end
end
