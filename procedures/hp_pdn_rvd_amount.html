CREATE PROCEDURE HP_PDN_RVD_AMOUNT(
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_ID)
RETURNS (
    RVD_MIN D_CURRENCY,
    RVD_MAX D_CURRENCY)
AS
declare variable c_id d_id;
declare variable res_address d_str_255;
declare variable k d_currency;
declare variable k1 d_currency;
begin
  -- РВД: рекомендуемая величина дохода
  -- OBJECT_KIND может быть miniapp, kontract, clients

  rvd_min = 0;
  c_id = case object_kind
            when 'clients'   then object_id
            when 'miniapp'  then (select mac_from_c_id from miniappclient where mac_from_ma_id = :object_id)
            when 'kontract' then (select kc_from_c_id from knclient where kc_from_k_id = :object_id)
         end;

  if (c_id is null) then
    exception e_default f_format('Not found c_id for object_kind = %1, object_id = %2', object_kind, object_id);

  --* если клиент в течение 60 календарных дней, предшествующих дате текущей Заявки, получал в М Булак займы,   
  --  то значение РВД = максимальной величине анкетного дохода, которое использовалось для расчета ПДН по этим займам, в течение последних 60 дней.
  select max(coalesce(kc_anketa_income_calc, kc_anketa_income, 0))
  from knclient_pdn kcp, knclient kc, kontstate
  where kc.kc_id = kcp.kc_id
    and ks_from_k_id = kc_from_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    and kcp_kind = 0
    and kcp.kc_from_c_id = :c_id
    and kc_date > current_date - 61
  into rvd_min;

  if (rvd_min > 0) then
  begin
    -- RFCRU-3689 если в течение 60 дней, предшествующих дате ввода обновленных данных дохода, клиент получал в Компании займы,
    -- то ограничитель сверху рассчитывается от РВД по формуле «РВД * %%, но не более 350 000»
    rvd_max = case
                when rvd_min <= 25000 then rvd_min * 10.0000
                when rvd_min <= 120000 then 150000
                when rvd_min <= 150000 then rvd_min * 1.2000
                when rvd_min <= 190000 then rvd_min * 1.1500
                else rvd_min * 1.1000
              end;

    if (rvd_max > 350000) then
      rvd_max = 350000;

    if (rvd_min > rvd_max) then
      rvd_max = rvd_min;

    suspend;
    exit;
  end 

  select res_income_avg, res_address
  from hp_extinfolist_pdn_one_income(null, current_date, :c_id)
  into rvd_min, res_address;

  if (rvd_min <= 0) then
    exception e_message iif(trim(coalesce(res_address, '')) in ('', 'Не указан'), 'Не указан адрес регистрации клиента', 'Некорректно указан регион регистрации клиента');

  --множитель - 2 или 2,5 или 3 или 4, в зависимости от величины дохода клиента по Росстату:
  k = case
        when rvd_min > 100000 then 1.1
        when rvd_min > 70000 then 1.2
        when rvd_min > 50000 then 1.7
        else 2.3
      end;
      
  --значение РДВ = ОКРУГЛВВЕРХ(доход клиента по Росстату * множитель * случайный коэффициент; до целой тысячи)
  select coalesce(extract(year from c_birth_date), 0)
  from clients
  where c_id = :c_id
  into k1;

  k1 = k1 - trunc(k1, -2) + c_id - trunc(c_id, -2);
  k1 = (k1 - trunc(k1, -2))/100.00 + (c_id - trunc(c_id, -2))/10000.0000;

  rvd_min = rvd_min * k;
  rvd_min = rvd_min * (0.80 + k1 * 0.40);
  rvd_min = round(rvd_min, -3) + iif(round(rvd_min, -3) < rvd_min, 1000, 0);

  -- RFCRU-3689 если в течение последних 60 дней до даты ввода новых данных дохода в раздел "Ежемесячный доход" клиент НЕ получал займы в Компании, ограничитель сверху = 350 000 руб.
  rvd_max = 350000;

  if (rvd_min > rvd_max) then
    rvd_min = rvd_max;

  suspend;
end
