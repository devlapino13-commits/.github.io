CREATE PROCEDURE AAREP_INDICATORS_SVZ(
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    OKS_BALANCE CURR_TYPE,
    OKS_DELINQ_BALANCE_7 CURR_TYPE,
    OKS_DELINQ_BALANCE_30 CURR_TYPE,
    JUST_CLI_ACTIVE INTEGER,
    JUST_CLI_DISB INTEGER,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    DISB_AMOUNT CURR_TYPE,
    HIRE_DATE DATE,
    STAGE_MONTH INTEGER)
AS
declare variable REP_DATE_BEGIN date;
declare variable OKS_KC_ID ID_TYPE;
declare variable OKS_REGION_ID integer;
declare variable OKS_REGION_ID_PRIOR integer;
begin
  oks_region_id_prior = 0;
  rep_date_begin = rep_date_end + 1 - extract(day from :rep_date_end);
  for with tabl1
      as (select oks_region_id, oks_kc_id, oks_balance, oks_delinq_date,
                 (select res_just_client_count
                  from hp_get_just_client_in_loan(cast(:rep_date_end as date), oks_from_k_id, k_disb_date)) just_cli_end,
                 k_disb_date, k_amount_approved
          from oksummary, kontract
          where :rep_date_end between oks_date_begin and oks_date_end and
                oks_close_state = 0 and
                oks_balance > 0 and
                cast(:rep_date_end as date) - oks_delinq_date <= 180
--  and oks_region_id = 1010300
                and
                oks_kc_id > 1000000000003 and
                oks_from_k_id = k_id)
      select oks_region_id, oks_kc_id, sum(oks_balance),
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0)),
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 30, oks_balance, 0)), sum(just_cli_end),
             sum(iif(k_disb_date >= :rep_date_begin, just_cli_end, 0)),
             sum(iif(k_disb_date >= :rep_date_begin, k_amount_approved, 0))
      from tabl1
      group by oks_kc_id, oks_region_id
      into oks_region_id, oks_kc_id, oks_balance, oks_delinq_balance_7, oks_delinq_balance_30, just_cli_active,
           just_cli_disb, disb_amount
  do
  begin
    if (oks_region_id_prior <> oks_region_id) then
    begin
      oks_region_id_prior = oks_region_id_prior;
      reg_name = '';
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :oks_region_id
      into reg_name;
    end
    officer_name = '';
    select c_name_fio
    from clients
    where c_id = :oks_kc_id
    into officer_name;
    par7 = cast(oks_delinq_balance_7 as double precision) / oks_balance;
    par30 = cast(oks_delinq_balance_30 as double precision) / oks_balance;
--  Вычислить стаж
    hire_date = null;
    select min(ps_date_begin)
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :oks_kc_id and
          ps_date_begin <= :rep_date_end and
          ps_status = 1 and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior')
    into hire_date;
    if (hire_date is null) then
      hire_date = current_date;
    stage_month = round(cast((rep_date_end - hire_date) as double precision) / 30, 0);

    suspend;
  end
end
