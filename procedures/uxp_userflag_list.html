CREATE PROCEDURE UXP$USERFLAG_LIST(
    UN_ID D_ID,
    TIMEZONE D_DOUBLE = null)
RETURNS (
    UA_UUID D_UUID,
    UA_LAST_TIME D_DATETIME,
    UA_CAPTION D_STR_255,
    UA_MAINMENU D_STR_4096,
    UA_VISUAL_TIME D_STR_CODE)
AS
declare variable db_id d_id;
declare variable ml_id d_integer;
declare variable hide_date d_bool;
begin
  if (timezone is not null) then
  begin
    db_id = uxf$db_id();
    if (db_id = 41701) then
      timezone = timezone - 6;
    else
    if (db_id = 39801) then
      timezone = timezone - 5;
    else
    if (db_id = 81001) then
      timezone = timezone - 3;
    else
    if (db_id = 94401) then
      timezone = timezone - 4;
  end

  for select usf_ua_uuid, usf_datetime, ua_name
  from ux$userflag, ux$action
  where usf_un_id = :un_id and
        usf_ua_uuid = ua_uuid
  order by usf_datetime desc
  into :ua_uuid, :ua_last_time, :ua_caption
  do
  begin
    hide_date = 0;
    if (cast(ua_last_time as date) = current_date) then
      hide_date = 1;

    if (timezone is not null) then
      ua_last_time = dateadd(hour, timezone, ua_last_time);

    ua_visual_time = f_datetimetostr(ua_last_time);
    if (hide_date = 1) then
      ua_visual_time = right(ua_visual_time, 5);

    ua_mainmenu = '';
    ml_id = null;

    select item_id
    from uxp$actionmenu(:ua_uuid)
    into ml_id;

    ua_mainmenu = uxf$menupath(ml_id);

    suspend;
  end
end
