CREATE PROCEDURE HP_EXPORT_PLAT_PORUCH(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    PP_DATE DATE,
    PP_DOC_NO INTEGER,
    PP_MEMO VARCHAR(1024),
    PP_IS_DELETE SMALLINT,
    PP_CLIENT_REGIST_NO VARCHAR(20),
    PP_CLIENT_OKPO VARCHAR(20),
    PP_CLIENT_INN VARCHAR(20),
    PP_CLIENT_BANK_ID VARCHAR(10),
    PP_CLIENT_BANK_NAME VARCHAR(255),
    PP_CLIENT_BANK_ACCOUNT VARCHAR(50),
    PP_OFFICE_REGIST_NO VARCHAR(255),
    PP_OFFICE_OKPO VARCHAR(20),
    PP_OFFICE_INN VARCHAR(20),
    PP_OFFICE_BANK_ID VARCHAR(10),
    PP_OFFICE_BANK_NAME VARCHAR(255),
    PP_OFFICE_BANK_ACCOUNT VARCHAR(50),
    PP_SUMMA DECIMAL(18,4),
    PP_PAY_CODE VARCHAR(20),
    PP_CLIENT_BANK_CORR_ACC VARCHAR(255),
    PP_OFFICE_BANK_CORR_ACC VARCHAR(255),
    PP_CLIENT_BANK_SWIFT VARCHAR(20),
    PP_OFFICE_BANK_SWIFT VARCHAR(20),
    PP_CLIENT_NAME_FULL VARCHAR(255),
    PP_OFFICE_NAME_FULL VARCHAR(255),
    OL_POCKET INTEGER,
    OL DECIMAL(18,0),
    PP_REGION_NAME VARCHAR(100),
    PP_CURR_SYMBOL VARCHAR(3))
AS
declare variable reg_id integer;
declare variable al_id varchar(20);
declare variable curr_code varchar(3);
declare variable curr_code_old varchar(3);
begin
  for select al_id, al_from_reg_id
      from acc_list, acctypelist, region_list(:rep_region_id, 1, 3, 1)
      where al_from_bg_id = atl_acc_mask and
            atl_from_at_id = 12 and
            al_from_reg_id = id_list
      into :al_id, :reg_id
  do
  begin
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into :pp_region_name;

    curr_code_old = '';

    for select ob_from_ol_id, ob_currency
        from operbook, operlist
        where ob_account_kt = :al_id and
              ob_ol_reg_id = :reg_id and
              ob_ol_date = :rep_date and
              ob_ol_is_active = 1 and
              ol_id = ob_from_ol_id and
              ol_doc_kind = 6 and
              ol_oper_type = 2 and
              ol_oper_subtype in (7, 9, 20) and
              ob_from_pc_id is not null
        into :ol, :curr_code
    do
    begin
      if (curr_code_old <> curr_code) then
      begin
        curr_code_old = curr_code;

        select cu_symbol
        from currlist
        where cu_id = :curr_code
        into :pp_curr_symbol;
      end

      for select pp_date, pp_doc_no, pp_memo, pp_is_delete, pp_client_regist_no, pp_client_okpo, pp_client_inn,
                 pp_client_bank_id, pp_client_bank_name, pp_client_bank_account, pp_office_regist_no, pp_office_okpo,
                 pp_office_inn, pp_office_bank_id, pp_office_bank_name, pp_office_bank_account, pp_summa, pp_pay_code,
                 pp_client_name_full, pp_office_name_full, pp_client_bank_corr_acc, pp_client_bank_swift,
                 pp_office_bank_corr_acc, pp_office_bank_swift, ol_pocket
          from blank_plat_poruch(:ol)
          where pp_doc_no is not null
          into pp_date, pp_doc_no, pp_memo, pp_is_delete, pp_client_regist_no, pp_client_okpo, pp_client_inn,
               pp_client_bank_id, pp_client_bank_name, pp_client_bank_account, pp_office_regist_no, pp_office_okpo,
               pp_office_inn, pp_office_bank_id, pp_office_bank_name, pp_office_bank_account, pp_summa, pp_pay_code,
               pp_client_name_full, pp_office_name_full, pp_client_bank_corr_acc, pp_client_bank_swift,
               pp_office_bank_corr_acc, pp_office_bank_swift, ol_pocket
      do
      begin
        suspend;
      end
    end
  end
end
