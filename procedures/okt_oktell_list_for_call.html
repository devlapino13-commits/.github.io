CREATE PROCEDURE OKT_OKTELL_LIST_FOR_CALL
RETURNS (
    OBJ_ID ID_TYPE,
    CELL_NO D_STR_20,
    PROJECT STR_100,
    TIMEZONE D_INTEGER,
    PERSON_TYPE STR_100,
    PERSON_ID ID_TYPE,
    UN_ID ID_TYPE,
    LANGUAGE STR_100,
    URL D_STR_2048,
    ORDER_NO D_INTEGER,
    PHONE_TYPE STR_100,
    REAL_PROJECT D_STR_100,
    COMMAND_ID D_INTEGER,
    DB_ID D_INTEGER)
AS
declare variable UN_TMP ID_TYPE;
declare variable UN_NAME D_STR_100;
declare variable DO_REPLACE_URL D_SMALLINT;
declare variable COMMAND_ID_OLD d_integer;
begin
  -- OKTELL.   Список задач для обработки OKTELL НЕ связанных с actionlist
  select un.un_id
  from username un
  where un.un_name = 'IGUSEV'
  into un_tmp;

  order_no = 0;
  project = 'OKTELL.Projects.Sirena';

  command_id_old = -1;

  for select pant_id,
             project,
             timezone,
             person_type,
             person_id,
             un_id,
             cell_no,
             language,
             url,
             phone_type,
             un_name,
             command_id,
             db_id
      from okt_security_list
      order by delinq_days desc
      into :obj_id,
           :real_project,
           :timezone,
           :person_type,
           :person_id,
           :un_id,
           :cell_no,
           :language,
           :url,
           :phone_type,
           :un_name,
           :command_id,
           :db_id
  do
  begin
    if (un_id is null) then
      un_id = un_tmp;
    cell_no = trim(coalesce(cell_no, ''));

    if (cell_no <> '') then
    begin
      if (language = 'None') then
        language = 'Language.RU';

      order_no = order_no + 1;

      if (char_length(:cell_no) >= 11) then
      begin
        select un.un_name
        from username un
        where un.un_id =  :un_id
        into :un_name;

        if (command_id_old <> command_id) then
        begin
            command_id_old = command_id;

            if ((select pc.pcm_is_blind
                 from pantera_commands pc
                 where pc.pcm_id = :command_id) = 1) then
                do_replace_url = 1;
            else
                do_replace_url = 0;
        end

        if (do_replace_url = 1) then
            url = replace(url, 'Sirena_d.aspx', 'Sirena_d_lite.aspx');

        if (command_id is not null) then
            suspend;
       end
    end
  end
end
