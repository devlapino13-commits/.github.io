CREATE PROCEDURE AAREP_RELATION_LIST(
    C_ID DECIMAL(18,0))
RETURNS (
    REL_C_ID DECIMAL(18,0),
    REL_C_FIO VARCHAR(100),
    REL_NAME VARCHAR(100),
    REL_DISTANCE INTEGER,
    REL_CELL_NO VARCHAR(50),
    REL_ADDRESS VARCHAR(255),
    REL_LINK_WITH_COMPANY VARCHAR(255),
    REL_BLACK_LIST VARCHAR(100))
AS
declare variable m_is_black integer;
declare variable full_list d_blob_text;
begin
-- Ищет все связи с контрагентом если он(а) не действующий сотрудник
-- Находит если были:
--   общий адрес
--   общий старый адрес (места жительства)
--   общий телефон
--   родственные связи
--   родственники родственников (уданность 5 узлов)
--   контактные лица
--   в одной группе солидарности получали кредит
--   Если по адресам не найдены, ищет активных клиентов в одном доме
--   Если по адресам не найдены и нет актив.кл. в одном доме, то ищет активных клиентов с одной улицы
--   Если есть общие контакты
  full_list = '';
  for select distinct rel_c_id, rel_fio, rel_name, rel_distance, rel_cell_no,
                      iif(rel_address = 'Не указан', '', rel_address)
      from aarep_relation_list_tmp(:c_id)
      order by 4, 1
      into :rel_c_id, :rel_c_fio, :rel_name, :rel_distance, rel_cell_no, rel_address
  do
  begin
    if (position(rel_c_fio, full_list) <= 0) then
    begin
      full_list = blob_append(full_list, ';', rel_c_fio);
      rel_link_with_company = '';
      if (rel_c_id is not null) then
      begin
        if (exists(select *
                   from hr_pstatus
                   where ps_from_c_id = :rel_c_id and
                         ps_date_end >= '31.12.2999' and
                         ps_is_working = 1 and
                         ps_status = 1)) then
        begin
          rel_link_with_company = 'Действ.сотрудник';
          rel_cell_no = '***';
          rel_address = '***';
        end
        else
        if (exists(select *
                   from hr_pstatus
                   where ps_from_c_id = :rel_c_id and
                         ps_date_end >= '31.12.2999' and
                         ps_is_working = 0 and
                         ps_status = 1)) then
          rel_link_with_company = 'Бывший сотрудник';

        if (exists(select *
                   from knclient, kontstate
                   where kc_from_c_id = :rel_c_id and
                         kc_amount_approved > 0 and
                         kc_from_k_id = ks_from_k_id and
                         ks_ol_is_delete = 0 and
                         ks_state = 1 and
                         ks_date_end >= '31.12.2999')) then
        begin
          if (rel_link_with_company = '') then
            rel_link_with_company = 'Активный клиент';
          else
            rel_link_with_company = rel_link_with_company || ', активный клиент';
        end
        else
        begin
          if (exists(select *
                     from knclient, kontstate
                     where kc_from_c_id = :rel_c_id and
                           kc_amount_approved > 0 and
                           kc_from_k_id = ks_from_k_id and
                           ks_ol_is_delete = 0 and
                           ks_state = 1)) then
          begin
            if (rel_link_with_company = '') then
              rel_link_with_company = 'Бывший клиент';
            else
              rel_link_with_company = rel_link_with_company || ', бывший клиент';
          end
        end
        m_is_black = -1;
        rel_black_list = '';
        if (rel_c_id is not null) then
        begin
          select m_is_black
          from moratory
          where m_from_c_id = :rel_c_id and
                m_is_active = 1 and
                m_is_black > 0
          order by m_date_begin desc
          rows 1
          into :m_is_black;

          if (m_is_black = 1) then
            rel_black_list = 'Ранее отказано';
          if (m_is_black = 2) then
            rel_black_list = 'Стоп лист';
        end
      end
      suspend;
    end
  end
end
