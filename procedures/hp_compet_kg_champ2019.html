CREATE PROCEDURE HP_COMPET_KG_CHAMP2019(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer;
declare variable rd_id integer;
declare variable res_period_id integer;
declare variable res_team_id integer;
declare variable res_lo_id decimal(18,0);
declare variable ad_active_client integer;
declare variable ad_active_balance decimal(18,4) = 0;
declare variable ad_diff_client integer;
declare variable ad_delinq_balance_1 decimal(18,4) = 0;
declare variable ad_delinq_balance_diff_1 decimal(18,4) = 0;
declare variable points_1 decimal(18,4) = 0;
declare variable points_2 decimal(18,4) = 0;
declare variable points_3 decimal(18,4) = 0;
declare variable points_all decimal(18,4) = 0;
declare variable res_team_id_prior integer;
declare variable cpr_id integer;
declare variable cpr_group_name varchar(255);
declare variable cpd_comment d_str_4096;
declare variable lo_panelty_count integer;
declare variable avg_points decimal(18,4) = 0;
declare variable total_points decimal(18,4) = 0;
declare variable add_points decimal(18,4) = 0;
declare variable max_points decimal(18,4) = 0;
declare variable group_name varchar(255);
declare variable cpp_orderno integer;
declare variable kind integer;
declare variable cpr_final_points decimal(18,4) = 0;
declare variable stage_cpp_id integer;
declare variable ad_disb_amount decimal(18,4) = 0;
declare variable is_otpusk varchar(20);
declare variable par1 decimal(18,4) = 0;
declare variable avg_amount decimal(18,4) = 0;
declare variable sum_ad_diff_client integer;
declare variable kol_svz integer;
declare variable avg_ad_diff_client curr_type;
declare variable avg_rost_portfel curr_type;
declare variable points_4 curr_type;
declare variable kol_komand integer;
declare variable k integer;
declare variable ad_diff_balance curr_type;
declare variable otpusk_days integer;
declare variable n integer;
declare variable dd_tmp date;
declare variable ad_disb_amount_ipf curr_type;
declare variable avg_amount_ipf curr_type;
declare variable points_5 curr_type;
declare variable kol_kred_closed integer;
declare variable in_period_disb integer;
declare variable coeff_uderj curr_type;
declare variable points_6 curr_type;
declare variable group_name_prior varchar(255);
declare variable ad_disb_loan integer;
declare variable prinyat_portfel curr_type;
declare variable svz_portfel_begin curr_type;
declare variable svz_portfel_end curr_type;
declare variable par30 curr_type;
declare variable ad_disb_new_amount curr_type;
declare variable is_month_end smallint;
declare variable date_begin date;
declare variable date_begin_prv date;
declare variable date_end_prv date;
declare variable rep_period integer;
declare variable rep_indicator integer;
declare variable zarabotok_begin curr_type;
declare variable zarabotok_end curr_type;
declare variable zarabotok_diff curr_type;
declare variable plan_sum curr_type;
declare variable disb_sum curr_type;
declare variable dosroch_sum curr_type;
declare variable plan_perc curr_type;
begin
-- Расчет итогов олимпиады КР 2017
-- Stage:
-- 0 - период (неделя)
-- 1 - Промежуточные итоги
-- 2 - Итоги
-- 3 - Финальные результаты

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  if (extract(day from res_date_end) not in (10, 20)) then
    is_month_end = 1;
  else
    is_month_end = 0;

  if (cpp_orderno = 0 and
      res_date_begin = cast('31.01.2019' as date)) then
  begin
    res_date_begin = '01.02.2019';
    res_date_end = '01.02.2019';

    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

-- Формируем в памяти данные по КР
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

-- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1 --, ad_active_balance
        from hp_analitica_report('kont_lo_teamname_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
        where ad_active_balance > 0 and
              exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        into :res_team_id
    do
    begin
      if (exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1)) then
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, '', -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end
    end
    suspend;
    exit;
  end

-- Ищем итоги тура
  stage_cpp_id = null;
  select cpp_id
  from competperiod
  where cpp_from_rd_id = :rd_id and
        cpp_stage = 2 and
        cpp_status = 1 and
        cpp_date_end <= :res_date_end
  order by cpp_date_begin desc, cpp_id desc
  rows 1
  into :stage_cpp_id;

  if (stage = 0) then -- Если уровень Декады
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

-- Формируем в памяти данные по КР
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    date_begin = res_date_end - extract(day from res_date_end) + 1;
    date_end_prv = date_begin - 1;
    date_begin_prv = date_end_prv - extract(day from date_end_prv) + 1;

--  формируем данные для план выдачи
    if (is_month_end = 1) then
    begin
      rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);
      delete from temp_bonusru_lo; -------------------------------------------------------------нужно активировать ----------------------------------------
      for select rd_id
          from refdata
          where rd_code in ('planindicators.disb_amount', 'PlanIndicators.Disb_Amount_IPF')
          into rep_indicator
      do
      begin
        execute procedure hp_plandata_fill(1000000, :res_date_end, :rep_indicator, :rep_period, 1, null);

        insert into temp_bonusru_lo (bl_personal_id, bl_region_id, bl_bonus_total)
        select pd_from_c_id, pd_from_reg_id, pd_data_1
        from temp_plandata
        where pd_is_office = 2 and
              pd_data_1 > 0;
      end
    end

-- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    kol_svz = 0;
    cpr_id = null;
    for select res_field_1, res_field_2,
--           sum(ad_active_client),
--           sum(ad_active_balance),
--           sum(ad_diff_client),
               sum(ad_delinq_balance_end1), sum(ad_delinq_balance_diff1), sum(ad_disb_amount),
--           sum(ad_disb_loan)
--           sum(ad_disb_new_amount)
               sum(ad_diff_balance)
        from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
        where exists(select *
                     from hr_pstatus
                     where ps_from_c_id = res_field_2 and
                           ps_date_end >= :res_date_end and
                           ps_date_begin <= :res_date_end and
                           ps_status = 1 and
                           ps_is_working = 1) and
              exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1, 2
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, -- ID команды
             :res_lo_id, -- ID КСника
--         :ad_active_client,          -- Клиенты на конец периода
--         :ad_active_balance,         -- Баланс на конец периода
--         :ad_diff_client,            -- Прирост клиентов           -1-
             :ad_delinq_balance_1, -- Баланс просрочек
             :ad_delinq_balance_diff_1, -- 1.    Изменение баланса просрочек более 0 дней СВЗ (рублей) по итогам отчетной недели   v
             :ad_disb_amount, -- 2.    Сумма всех выдач за неделю (рублей)  по итогам отчетной недели
--         :ad_disb_loan,             -- Кол-во выдач
--         :ad_disb_new_amount          -- Сумма выдачи новым клиентам
             :ad_diff_balance -- Прирост портфля

    do
    begin
-- Исключаем если нет действующего
      is_otpusk = '';
      if (res_team_id <> 0) then
      begin
-- Исключаем если в отпуске.
-- Проверяем статус отпуска на каждый день недели:
-- если все 5 дней был в отпуске, исключаем из списка
        otpusk_days = 0;
        n = 0;
        while (n <= 4) do
        begin
          dd_tmp = dateadd(day, n, res_date_begin);
          if (exists(select *
                     from hr_voctravel, hr_orders, sa_worktype
                     where hrv_from_ho_id = ho_id and
                           ho_order_type = sawt_id and
                           hrv_from_c_id = :res_lo_id and
                           sawt_base_type in (4, 5) -- отпуск, декретный отпуск
                           and
                           sawt_id <> 20 and
                           sawt_hours_real >= 0 and
                           hrv_date_begin <= :dd_tmp and
                           hrv_date_end_fact >= :dd_tmp and
                           hrv_status = 1)) then
            otpusk_days = otpusk_days + 1;
          n = n + 1;
        end

        if (otpusk_days = 5) then
        begin
          if (not(exists(select *
                         from hr_voctravel, hr_orders
                         where hrv_from_ho_id = ho_id and
                               ho_order_type = 20 and
                               hrv_from_c_id = :res_lo_id and
                               hrv_date_begin <= :res_date_end and
                               hrv_date_end_fact >= :res_date_begin and
                               hrv_status = 1))) then
          begin
--            res_team_id=0;
            is_otpusk = 'Отпуск.';
          end
        end
      end

      points_1 = 0.0; --1.    Изменение баланса просрочек 1 и более дней СВЗ (рублей) по итогам отчетной недели
      points_2 = 0.0;
      points_3 = 0.0;
      points_4 = 0.0;
      points_5 = 0.0;
      points_6 = 0.0;

--    Сумма выдачи по ИПФ кредитам
      ad_disb_amount_ipf = null;
      select sum(k_amount_approved)
      from kontstate, kontract, product, knaktlist
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :res_date_begin and :res_date_end and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_code containing '.IPF.' and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :res_date_end between kal_date and kal_date_end and
            kal_new_kf_officer = :res_lo_id
      into ad_disb_amount_ipf;
      if (ad_disb_amount_ipf is null) then
        ad_disb_amount_ipf = 0;
      ad_disb_amount = ad_disb_amount - ad_disb_amount_ipf;

--      par1 = cast(ad_delinq_balance_1 as double precision) / ad_active_balance * 100;

      cpd_comment = '';
--      cpd_comment=cpd_comment||'Актив.кл: <b>'||round(ad_active_client)||'</b><br>';
--      cpd_comment=cpd_comment||'PAR1: <b>'||round(par1,4)||'%</b><br>';
      cpd_comment = cpd_comment || 'Баланс просрочки PAR1: <b>' || round(ad_delinq_balance_1, 2) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Изменение PAR0: <b>' || round(ad_delinq_balance_diff_1, 2) || ' р.</b><br>';
--      cpd_comment=cpd_comment||'Сумма выдач новым клиентам: <b>'||round(ad_disb_new_amount,2)||' р.</b><br>';
      cpd_comment = cpd_comment || 'Сумма выдач СпФ за период: <b>' || round(ad_disb_amount, 0) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Сумма выдач СпФ за период по ИПФ: <b>' || round(ad_disb_amount_ipf, 0) || '</b><br>';
      cpd_comment = cpd_comment || 'Прирост портфеля: <b>' || round(ad_diff_balance, 0) || '</b><br>';
--      cpd_comment=cpd_comment||'Баланс просрочки PAR>=1: <b>'||round(ad_delinq_balance_1,2)||'%</b><br>';
--      cpd_comment=cpd_comment||'Кол-во выдач: <b>'||ad_disb_loan||'</b><br>';
--      cpd_comment=cpd_comment||'Прирост кл: <b>'||round(ad_diff_client)||'</b><br>';

      /*
--    Вычисляем баллы по приросту
      points_1 = case
                   when ad_diff_client >= 1 then 2
                   when ad_diff_client >= 0 then 1
                   when ad_diff_client >= -5 then -2
                   else -4
                 end;
      */

--    1. Изменение баланса просрочек свз более 0 дней
      if (ad_delinq_balance_1 <= 0.0) then
        points_1 = 5;
      else
      if (ad_delinq_balance_diff_1 <= 0) then
        points_1 = 1;
      else
      if (ad_delinq_balance_diff_1 <= 10000) then
        points_1 = -1;
      else
        points_1 = -3;

--    2. Сумма выдач классика
      points_2 = case
                   when ad_disb_amount > 1200000.0 then 5
                   when ad_disb_amount > 850000.0 then 4
                   when ad_disb_amount > 550000.0 then 2
                   when ad_disb_amount > 400000.0 then 0
                   when ad_disb_amount > 250000.0 then -3
                   else -5
                 end;
--    3. Сумма выдач ИПФ
      points_3 = case
                   when ad_disb_amount_ipf > 550000.0 then 5
                   when ad_disb_amount_ipf > 400000.0 then 4
                   when ad_disb_amount_ipf > 250000.0 then 2
                   when ad_disb_amount_ipf > 150000.0 then 0
                   else -3
                 end;
--    4. Прирост портфеля
      points_4 = case
                   when ad_diff_balance > 200000.0 then 7
                   when ad_diff_balance > 125000.0 then 5
                   when ad_diff_balance > 100000.0 then 4
                   when ad_diff_balance > 75000.0 then 3
                   when ad_diff_balance > 25000.0 then 2
                   when ad_diff_balance >= 0.0 then 0
                   else -4
                 end;

      points_5 = 0;
      points_6 = 0;
      if (is_month_end = 1) then
      begin
--      реализация план выдач
        plan_sum = null;
        select sum(bl_bonus_total)
        from temp_bonusru_lo
        where bl_personal_id = :res_lo_id
        into plan_sum;
        if (plan_sum is null) then
          plan_sum = 0;

--      обьем выдач
        select sum(k_amount_approved)
        from kontstate, kontract
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :date_begin and :res_date_end and
              ks_kc_id = :res_lo_id and
              ks_from_k_id = k_id
        into disb_sum;
        if (disb_sum is null) then
          disb_sum = 0;

--      Обьем досроных погашений
        select -sum(ok_os)
        from operkred, kontract, knaktlist
        where ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_os < 0 and
              ok_ol_date between :date_begin and :res_date_end and
              ok_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              ok_ol_date between kal_date and kal_date_end and
              kal_new_kf_officer = :res_lo_id and
              ok_ol_date < ok_sh_date and
              extract(month from ok_ol_date) <> extract(month from ok_sh_date)
        into dosroch_sum;
        if (dosroch_sum is null) then
          dosroch_sum = 0;
        if (plan_sum = 0) then
          plan_sum = 700000.00;

        if (plan_sum > 0) then
          plan_perc = (disb_sum - dosroch_sum) / plan_sum * 100;
        else
          plan_perc = 0;

        cpd_comment = cpd_comment || 'Объем выдач: <b>' || round(disb_sum, 0) || '</b><br>';
        cpd_comment = cpd_comment || 'Объем досрочных погашений: <b>' || round(dosroch_sum, 0) || '</b><br>';
        cpd_comment = cpd_comment || 'План по выдачам: <b>' || round(plan_sum, 0) || '</b><br>';
        cpd_comment = cpd_comment || 'Выполнения плана в %%: <b>' || round(plan_perc, 0) || '</b><br>';

--      Выполнение плана по выдачам СВЗ за отчетный месяц в процентах
        points_5 = case
                     when plan_perc > 110 then 25
                     when plan_perc > 100 then 20
                     when plan_perc > 97 then 13
                     when plan_perc > 92 then 7
                     when plan_perc > 87 then 0
                     when plan_perc > 82 then -5
                     else -15
                   end;

        select aarep_bonus_total
        from cb_kg_lo_201602_one_lo(null, :res_lo_id, /*:date_begin_prv,*/ :date_end_prv, 0)
        into zarabotok_begin;
        select aarep_bonus_total
        from cb_kg_lo_201602_one_lo(null, :res_lo_id, /*:date_begin,*/ :res_date_end, 0)
        into zarabotok_end;
        zarabotok_diff = zarabotok_end - zarabotok_begin;

        cpd_comment = cpd_comment || 'Заработок на начало: <b>' || round(zarabotok_begin, 0) || '</b><br>';
        cpd_comment = cpd_comment || 'Заработок на конец: <b>' || round(zarabotok_end, 0) || '</b><br>';
        cpd_comment = cpd_comment || 'Прирост заработка: <b>' || round(zarabotok_diff, 0) || '</b><br>';
--      3. Прирост заработка СВЗ за отчетный месяц
        points_6 = case
                     when zarabotok_diff > 4000.0 then 25
                     when zarabotok_diff > 2000.0 then 18
                     when zarabotok_diff > 1500.0 then 12
                     when zarabotok_diff > 1000.0 then 8
                     when zarabotok_diff >= 0.0 then 0
                     else -8
                   end;

      end

      if (res_team_id <> res_team_id_prior) then -- Если команда отличается
      begin
-- Находим нужную строку команды за этот период
        cpr_id = null;
        select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id and
              cpr_object_kind = -1 and
              cpr_object_id = :res_team_id
        into :cpr_id;

        if (cpr_id is null) then -- если такой строки нет, создаем новую
        begin
          cpr_group_name = '';
          stage_cpp_id = null;
-- Получаем название Группы где последний раз была команда
          if (stage_cpp_id is not null) then
          begin
-- Если это первая неделя второго этапа, название лиги берем из поля комментария предыдущего расчета
            select coalesce(cpr_comment, '')
            from competresult, competperiod
            where cpr_from_cpp_id = :stage_cpp_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id rows 1
            into :cpr_group_name;
          end
          else
          begin -- Если это не первая неделя второго Этапа
            select cpr_group_name
            from competresult, competperiod
            where cpr_from_cpp_id = cpp_id and
                  cpp_from_rd_id = :rd_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id and
                  cpr_group_name <> '' and
                  cpp_orderno < :cpp_orderno
            order by cpp_orderno desc
            rows 1
            into :cpr_group_name;
          end
          if (cpr_group_name is null) then
            cpr_group_name = '';

          insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                    cpr_from_un_id, cpr_comment)
          values (:cpp_id, :cpr_group_name, -1, :res_team_id, 0, :un_id, '')
          returning cpr_id
          into :cpr_id;
        end
        res_team_id_prior = res_team_id;
      end

-- Вставляем данные по сотруднику
--      if (is_otpusk='') then -- отпускников убираю из вычисления письмо от 25.04.2017
--      begin
      insert into competdetails (cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, cpd_points_1,
