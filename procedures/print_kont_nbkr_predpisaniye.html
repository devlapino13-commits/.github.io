CREATE PROCEDURE PRINT_KONT_NBKR_PREDPISANIYE(
    K_ID D_ID)
RETURNS (
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE D_DATE,
    NP_AMOUNT D_CURRENCY,
    C_NAME_FIO D_STR_100)
AS
begin
  for select kont.k_kontract_no, kont.k_disb_date, np.np_amount, cl.c_name_fio
  from knclient kncl
  inner join kontract kont on kont.k_id = kncl.kc_from_k_id
  inner join nbkr_predpisaniye np on np.np_from_kc_id = kncl.kc_id
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  where kncl.kc_from_k_id = :k_id
  into :k_kontract_no, :k_disb_date, :np_amount, :c_name_fio
  do
  begin
    suspend;
  end
end
