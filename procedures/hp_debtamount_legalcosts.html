CREATE PROCEDURE HP_DEBTAMOUNT_LEGALCOSTS(
    K_ID D_ID,
    DT_OPER D_DATE = '31.12.9999',
    ONLY_LAST D_BOOL = 0)
RETURNS (
    DEBT_WRITEXEC D_CURRENCY,
    DEBT_STATEDUTY D_CURRENCY)
AS
declare variable rd_id1 d_integer;
declare variable rd_id2 d_integer;
begin
  rd_id1 = f_get_rd_id('OperExtraKind.ExecutiveInscription');
  rd_id2 = f_get_rd_id('OperExtraKind.LawClientExpense');

  if (only_last = 0) then
  begin
    select
      sum(iif(ox_from_rd_id = :rd_id1, ox_amount_plus - ox_amount_minus - ox_amount_writeoff, 0)),
      sum(iif(ox_from_rd_id = :rd_id2, ox_amount_plus - ox_amount_minus - ox_amount_writeoff, 0))
    from operextra
    where ox_object_id = :k_id
      and ox_from_rd_id in (:rd_id1, :rd_id2)
      and ox_ol_date <= :dt_oper
      and ox_ol_is_active = 1
    into
      :debt_writexec,
      :debt_stateduty;
  end else begin
    debt_writexec = (
      select first 1 ox_amount_plus
      from operextra
      where ox_object_id = :k_id
        and ox_from_rd_id = :rd_id1
        and ox_ol_date <= :dt_oper
        and ox_ol_is_active = 1
      order by ox_ol_date desc);
    debt_stateduty = (
      select first 1 ox_amount_plus
      from operextra
      where ox_object_id = :k_id
        and ox_from_rd_id = :rd_id2
        and ox_ol_date <= :dt_oper
        and ox_ol_is_active = 1
      order by ox_ol_date desc);
  end

  debt_writexec = coalesce(debt_writexec, 0);
  debt_stateduty = coalesce(debt_stateduty, 0);

  suspend;
end
