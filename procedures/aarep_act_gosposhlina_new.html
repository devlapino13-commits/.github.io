CREATE PROCEDURE AAREP_ACT_GOSPOSHLINA_NEW(
    REP_DATE_END DATE)
RETURNS (
    ACC_ID D_ID,
    ACC_CODE D_STR_CODE,
    ACC_OBJECT_ID D_ID,
    K_KONTRACT_NO STR_20,
    LOAN_STATUS STR_10,
    BAL_SUM_60323_20 CURR_TYPE,
    BAL_SUM_60324_21 CURR_TYPE,
    BAL_SUM_60324_22 CURR_TYPE,
    BAL_SUM_60324_23 CURR_TYPE,
    OX_OL_DATE_NACH DATE,
    OX_OL_DATE_OPLATA DATE,
    DAYS INTEGER,
    CLIENT_TYPE D_STR_20)
AS
declare variable k_id id_type;
declare variable oek_rd_id integer;
declare variable date_max date;
declare variable pr_code d_str_100;
begin

  for select acc_object_id, sum(iif(acc_from_sab_id = 27, kobs_bal_sum, 0)),
             sum(-iif(acc_from_sab_id = 86, kobs_bal_sum, 0)), sum(-iif(acc_from_sab_id = 87, kobs_bal_sum, 0)),
             sum(-iif(acc_from_sab_id = 88, kobs_bal_sum, 0))
      from ref_account, kn_obsummary
      where acc_from_sab_id in (27, 86, 87, 88) and
            kobs_acc_id = acc_id and
            :rep_date_end between kobs_date and kobs_date_end and
            kobs_bal_sum <> 0 and
            acc_object_kind + 0 = 8 and
            acc_object_id >= 9
      group by 1
     -- rows 10
      into acc_object_id, bal_sum_60323_20, bal_sum_60324_21, bal_sum_60324_22, bal_sum_60324_23

  do
  begin
    select k_kontract_no, k_id, pr_code
    from knclient, kontract, product
    where kc_id = :acc_object_id and
          kc_from_k_id = k_id and
          pr_id = k_from_pr_id
    into k_kontract_no, k_id, pr_code;


    client_type = 'ФЛ';
    if (pr_code containing '.selfemployed.') then
      client_type = 'СЗ';

    select iif(ks_state = 1, 'Активный', 'Закрытый')
    from kontstate
    where ks_from_k_id = :k_id and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_ol_is_delete = 0
    into loan_status;

    ox_ol_date_nach = null;
    ox_ol_date_oplata = null;
    select op.ox_ol_date -- Дата начисления госпошлины
    from operextra op
    where op.ox_object_id = :k_id and
          op.ox_amount_plus <> 0 and
          op.ox_ol_is_active = 1
    order by ox_ol_date asc
    rows 1
    into ox_ol_date_nach;

    select op.ox_ol_date -- Дата последней оплаты госпошлины
    from operextra op
    where op.ox_object_id = :k_id and
          op.ox_amount_minus <> 0 and
          op.ox_ol_is_active = 1
    order by ox_ol_date desc
    rows 1
    into ox_ol_date_oplata;

    date_max = ox_ol_date_nach; -- получаем более позднюю дату между даты начистения и даты оплаты госпошлин
    if (ox_ol_date_oplata > ox_ol_date_nach) then
      date_max = ox_ol_date_oplata;

    days = -datediff(day, rep_date_end, date_max); -- Кол-во дней с даты начисления или оплаты госпошлины до даты Акта
    suspend;
  end
end
