CREATE PROCEDURE HP_CHECK_CLIENT_MINIAPP(
    C_ID D_ID,
    MA_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT STR_255)
AS
declare variable FL_NAME1 STR_100;
declare variable FL_NAME2 STR_100;
declare variable C_BIRTH_DATE D_STR_100;
declare variable C_BIRTH_CITY STR_100;
declare variable C_BIRTH_COUNTRY STR_100;
declare variable C_CITIZEN STR_255;
declare variable C_PASS_NO STR_100;
declare variable C_PASS_DATE STR_100;
declare variable C_PASS_WHO STR_255;
declare variable C_PASS_WHO_CODE STR_20;
declare variable CDS_FROM_PK_ID D_INTEGER;
declare variable CDS_DOC_NO D_STR_100;
declare variable CDS_DOC_DATE_BEGIN D_STR_100;
declare variable CDS_DOC_DATE_END D_STR_100;
declare variable CDS_DOC_ISSUED_BY D_STR_255;
declare variable C_BLOB_PHOTO_ID STR_100;
declare variable DOC_LIST D_STR_255;
declare variable MA_DATE_APP D_DATE;
declare variable MA_DATE_OPER D_DATE;
declare variable CDS_MAIN_DOC_DATE_END D_DATE;
begin
  res_text = '';

  select coalesce(fl_name1, ''), coalesce(fl_name2, ''), f_datetostr(c_birth_date),
         coalesce(c_birth_country, ''), coalesce(c_birth_city, ''), c_citizen, coalesce(c_pass_no, ''),
         f_datetostr(c_pass_date), coalesce(c_pass_who, ''), coalesce(c_pass_who_code, ''),
         coalesce(c_blob_photo_id, '')
  from clients
  left join fiolist on c_from_fl_id = fl_id
  where c_id = :c_id
  into fl_name1, fl_name2, c_birth_date, c_birth_country, c_birth_city, c_citizen, c_pass_no, c_pass_date, c_pass_who,
       c_pass_who_code, c_blob_photo_id;

  select cds_from_pk_id, coalesce(cds_doc_no, ''), f_datetostr(cds_doc_date_begin),
         f_datetostr(cds_doc_date_end), coalesce(cds_doc_issued_by, ''), cds_doc_date_end
  from clientdocs
  where cds_from_c_id = :c_id and
        cds_is_main_doc = 1 and
        cds_is_delete = 0 and
        cds_verify_status = 2
  order by cds_create_datetime desc rows 1
  into cds_from_pk_id, cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_doc_issued_by, cds_main_doc_date_end;

  if (fl_name1 = '' or fl_name2 = '') then
    res_text = 'Не заполнено имя или фамилия';
  if (c_birth_date = '') then
    res_text = 'Не заполнена дата рождения';
  if (c_birth_country = '') then
    res_text = 'Не заполнена страна рождения';
  if (c_birth_city = '') then
    res_text = 'Не заполнено место рождения';
  if (c_citizen = 0) then
    res_text = 'Не заполнено гражданство';
  if (c_blob_photo_id = '') then
    res_text = 'Не закреплено фото клиента';
  if (c_pass_no = '' and
      cds_doc_no = '') then
    res_text = 'Не заполнен номер документа удостоверяющего личность';
  if (c_pass_date = '' and
      cds_doc_date_begin = '') then
    res_text = 'Не заполнена дата выдачи документа удостоверяющего личность';
  if ((c_pass_who = '' and
      cds_doc_issued_by = '') and
      c_pass_who_code = '') then
    res_text = 'Не заполнен орган, выдавший документ удостоверяющий личность';

  if (exists(select 1
             from clientdocs
             where cds_from_c_id = :c_id and
                   cds_is_main_doc = 1 and
                   cds_is_delete = 0 and
                   cds_verify_status <> 2 and
                   cds_doc_date_end is null)) then
    res_text = 'Не указана дата окончания срока действия паспорта';

  if (res_text = '' and
      cds_from_pk_id <> 12) then
  begin
    cds_doc_no = '';
    cds_doc_date_begin = '';
    cds_doc_date_end = '';

    select cds_doc_no, cds_doc_date_begin, cds_doc_date_end
    from clientdocs, passkind, passdoctype
    where cds_from_c_id = :c_id and
          cds_from_pk_id = pk_id and
          pk_from_pdt_id = pdt_id and
          cds_is_delete = 0 and
          current_date between cds_doc_date_begin and cds_doc_date_end and
          pdt_code in ('magration_card', 'registration_rvp', 'registration_vnj')
    order by cds_doc_date_end desc rows 1
    into cds_doc_no, cds_doc_date_begin, cds_doc_date_end;

    if (cds_doc_no = '') then
      res_text = 'Не заполнен номер документа подтверждающего право пребывания в РФ';
    if (cds_doc_date_begin = '' or cds_doc_date_end = '') then
      res_text = 'Не заполнена дата начала или дата окончания срока действия права пребывания в РФ';
  end

  if (res_text = '') then
  begin
    doc_list = '';
    for select ma_date_app, ma_date_oper
        from miniapp
        where ma_id = :ma_id and
              ma_date_end >= '31.12.2999'
        into ma_date_app, ma_date_oper
    do
    begin
      if (f_edoc_state2('ad.agreement', ma_date_app, un_id) = 'active' and
          f_edoc_status('ad.agreement', c_id, current_date) <> 'signed') then
        doc_list = doc_list || iif(doc_list = '', '', ', ') || 'Согласие на получение рекламных сообщений и информации';

      if (f_get_const('doc_optimize_pack2') <> 1 and
          f_edoc_state2('bki.agreement', ma_date_app, un_id) = 'active' and
          f_edoc_status('bki.agreement', ma_id, current_date) <> 'signed') then
        doc_list = doc_list || iif(doc_list = '', '', ', ') || 'Согласие на получение кредитного отчета';

      if (f_get_const('doc_optimize_pack2') <> 1 and
          f_edoc_state2('pdata.agreement', ma_date_app, un_id) = 'active' and
          f_edoc_status('pdata.agreement', ma_id, current_date) <> 'signed') then
        doc_list = doc_list || iif(doc_list = '', '', ', ') || 'Согласие на обработку персональных данных клиента';

      if (doc_list <> '') then
        res_text = 'Сначала нужно чтобы клиент подписал: ' || doc_list;

      if (c_citizen = 13 and
          cds_main_doc_date_end is not null and
          cds_main_doc_date_end < '31.12.9999' and
          iif(cds_from_pk_id = 12, dateadd(day, 90, cds_main_doc_date_end), cds_main_doc_date_end) < ma_date_oper) then
        res_text = 'Срок паспорта клиента истек либо не соответствует требованиям Компании';

      if (c_citizen <> 13 and
          cds_main_doc_date_end is not null and
          cds_main_doc_date_end < '31.12.9999' and
          cds_main_doc_date_end < dateadd(day, 30, ma_date_oper)) then
        res_text = 'До конца срока действия паспорта иностранного гражданина должно оставаться не менее 30 календарных дней на дату выдачи займа';
    end
  end
  suspend;
end
