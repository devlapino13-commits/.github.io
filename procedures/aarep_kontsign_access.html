CREATE PROCEDURE AAREP_KONTSIGN_ACCESS(
    K_ID DECIMAL(18,0),
    USER_ID DECIMAL(18,0),
    REG_ID INTEGER,
    USER_LOGIN VARCHAR(30))
RETURNS (
    KSR_ID INTEGER)
AS
begin
  ksr_id = 0;
  if (reg_id = 0) then
  begin
    select distinct kal_new_reg_id
    from knaktlist, kontract
    where k_id = :k_id and
          k_from_kf_id = kal_from_kf_id and
          kal_date_end >= cast('NOW' as date) and
          kal_date <= cast('NOW' as date) and
          knaktlist.is_delete = 0
    into :reg_id;
  end
  if (user_id = 0) then
  begin
    select un_id
    from username
    where un_name = :user_login
    into :user_id;
  end
  if (user_id <> 0) then
  begin
    if (reg_id is not null) then
    begin
      select distinct ksr_id
      from kontsignreg, kontsignmen, region_list(:reg_id, 1, -1, 0), username
      where (ksr_from_reg_id = id_list) and
            ksm_from_c_id = un_from_c_id and
            un_id = :user_id and
            ksm_can_approve = 1 and
            ksr_is_active = 1 and
            ksr_from_ksm_id = ksm_id and
            coalesce(ksr_date_begin, '01.01.1990') <= current_date and
            coalesce(ksr_date_end, '31.12.9999') >= current_date rows 1
      into :ksr_id;
      if (ksr_id is null) then
        ksr_id = 0;
    end
  end
  else
    ksr_id = 0;
  suspend;
end
