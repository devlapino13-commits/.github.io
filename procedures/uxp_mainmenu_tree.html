CREATE PROCEDURE UXP$MAINMENU_TREE(
    DEV_MODE D_BOOL = 1,
    WEB_MODE D_BOOL = 0)
RETURNS (
    MNU_ID D_ID,
    MNU_PARENT_ID D_ID,
    MNU_CODE D_STR_CODE,
    MNU_NAME D_STR_100,
    MNU_MEMO D_STR_255,
    MNU_POSITION D_INTEGER,
    MNU_UA_UUID D_UUID,
    MNU_IS_FOLDER D_BOOLEAN)
AS
begin
  if (dev_mode is null) then
    dev_mode = 1;

  if (web_mode is null) then
    web_mode = 0;

  for with recursive tree$list
  as (select cast(lpad(ml_position, 6, '_') as varchar(255)) as tree$pos,
             ml_id,
             ml_parent_id,
             ml_code,
             ml_caption,
             ml_description,
             ml_position,
             uml_ua_uuid,
             iif(ml_has_child = 1, true, false) as ml_has_child
      from menulist
           left join ux$menulist on uml_ml_id = ml_id
      where ml_parent_id is null and
            (ml_has_child > 0 or uml_ua_uuid is not null) and
            (coalesce(uml_web_disabled, 0) = 0 or :web_mode = 0 or ml_has_child > 0) and
            (coalesce(uml_fs_disabled, 0) = 0 or :dev_mode = 1 or ml_has_child > 0)
      union all
      select tree$pos || lpad(t.ml_position, 6, '_'),
             t.ml_id,
             t.ml_parent_id,
             t.ml_code,
             t.ml_caption,
             t.ml_description,
             t.ml_position,
             m.uml_ua_uuid,
             iif(t.ml_has_child = 1, true, false) as ml_has_child
      from menulist t
        left join ux$menulist m on m.uml_ml_id = t.ml_id
        join tree$list on (tree$list.ml_id = t.ml_parent_id and
                          (coalesce(m.uml_web_disabled, 0) = 0 or :web_mode = 0 or t.ml_has_child > 0) and
                          (coalesce(m.uml_fs_disabled, 0) = 0 or :dev_mode = 1 or t.ml_has_child > 0))
      where (t.ml_has_child > 0 or m.uml_ua_uuid is not null))
  select ml_id,
         ml_parent_id,
         ml_code,
         ml_caption,
         ml_description,
         ml_position,
         uml_ua_uuid,
         ml_has_child
  from tree$list
  order by tree$pos
  into mnu_id,
       mnu_parent_id,
       mnu_code,
       mnu_name,
       mnu_memo,
       mnu_position,
       mnu_ua_uuid,
       mnu_is_folder
  do
    suspend;
end
