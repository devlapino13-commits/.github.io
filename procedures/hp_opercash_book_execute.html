CREATE PROCEDURE HP_OPERCASH_BOOK_EXECUTE(
    OPC_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable OPC_FROM_OL_ID D_ID;
declare variable OPC_DATE D_DATE;
declare variable OPC_AMOUNT D_CURRENCY;
declare variable OPC_DOC_NO D_INTEGER;
declare variable OPC_IS_SIGNED D_BOOL;
declare variable OPC_STATUS D_SMALLINT;
declare variable OL_IS_DELETE D_BOOL;
declare variable OPC_OFFICE_ID D_INTEGER;
declare variable ASYNC_OPERCASH D_INTEGER;
declare variable OPC_OPER_KIND D_INTEGER;
begin
  res_is_error = 1;
  res_text = '';

  for select opc_date, opc_oper_kind, opc_office_id, opc_doc_no, opc_from_ol_id, opc_amount, opc_is_signed, opc_status, ol.is_delete
  from opercash
       inner join cashplace on opc_from_csp_id = csp_id
       left join operlist OL on opc_from_ol_id = ol_id
  where opc_id = :opc_id and            
        opc_parent_id is null
  union
  select opc_date, opc_oper_kind, opc_office_id, opc_doc_no, opc_from_ol_id, opc_amount, opc_is_signed, opc_status, OL.is_delete
  from opercash
       inner join cashplace on opc_from_csp_id = csp_id
       left join operlist OL on opc_from_ol_id = ol_id
  where opc_parent_id = :opc_id
  order by 1
  into opc_date, opc_oper_kind, opc_office_id, opc_doc_no, opc_from_ol_id, opc_amount, opc_is_signed, opc_status, ol_is_delete
  do
  begin
    if (ol_is_delete is null) then
      ol_is_delete = 0;

    if (ol_is_delete = 1) then
      opc_from_ol_id = null;

    if (opc_is_signed = 0 and res_text = '') then
      res_text = 'Документ не подтвержден менеджером, № ' || opc_doc_no || ', сумма: ' || f_numtostr(opc_amount, 0);

    if (opc_status <> 1 and res_text = '') then
      res_text = 'Документ еще не подписан, № ' || opc_doc_no || ', сумма: ' || f_numtostr(opc_amount, 0);

    if (opc_from_ol_id is not null and res_text = '') then
      res_text = 'Операция уже проведена, № ' || opc_doc_no || ', сумма: ' || f_numtostr(opc_amount, 0);

    if (res_text <> '') then
      break;
  end

  if (opc_date is null and res_text = '') then
    res_text = 'Операция не найдена либо она связана с другой операцией';

  if (res_text = '') then
  begin
    res_is_error = 0;

    async_opercash = f_async_opercash();
    -- async_opercash:
    -- 0 - асинхронная обработка операций отключена
    -- 1 - асинхронная обработка операций включена для конкретного офиса
    -- 2 - асинхронная обработка операций включена для всей компании

    if (async_opercash = 3 and not(opc_oper_kind in (1))) then -- не погашение, тут можно расширять IN
      async_opercash = 0;

    if (async_opercash = 1 and opc_office_id <> 1070100) then -- не 0701 Тогтогул
      async_opercash = 0;

    if (async_opercash <> 0) then
    begin
      -- Если ранее была ошибка (кроме блокировок), проводим вручную без очереди
      if (exists(select * from cashprocess
                 where cpr_from_opc_id = :opc_id and
                       coalesce(cpr_error_text, '') <> '')) then
      async_opercash = 0;
    end
  
    if (async_opercash <> 0) then
    begin
      -- Добавление в очередь для асинхронного выполнения
      delete from cashprocess
      where cpr_from_opc_id = :opc_id;
  
      insert into cashprocess (cpr_from_opc_id, cpr_date, cpr_from_un_id, cpr_error_text)
      values (:opc_id, :opc_date, :un_id, '');
  
      res_is_error = 0;
      res_text = 'Проведение операции добавлено в очередь';
    end else
    begin
      -- Проведение документа напрямую (синхронно)
  
      select res_is_error, res_text
      from hp_opercash_book_process(:opc_id, :un_id)
      into :res_is_error, :res_text;
  
      delete from cashprocess
      where cpr_from_opc_id = :opc_id;
  
      res_text = 'Операция успешно проведена';
    end
  end

  suspend;
end
