CREATE PROCEDURE AAREP_REPORT_ZB_PSK2(
    OBEG INTEGER,
    OEND INTEGER)
AS
declare variable k_id bigint;
declare variable k_disb_date d_date;
declare variable kd_percent_effect decimal(18,4);
declare variable rep_query_date date;
declare variable kc_id bigint;
declare variable line_no_upd integer;
begin
  for select k_id,  k_disb_date, line_no , rep_query_date
      from rep_ru_cbank
      where line_no >= :obeg and
            line_no <= :oend
      into :k_id,  :k_disb_date, :line_no_upd , :rep_query_date
  do
  begin
    kd_percent_effect = null;
    select effect_rate
    from hp_effect_rate_to_date(:k_id, :k_disb_date, 1)
    into :kd_percent_effect;

    update rep_ru_cbank
    set CRED_FULL_COST = :kd_percent_effect
    where line_no = :line_no_upd;
  end

end
