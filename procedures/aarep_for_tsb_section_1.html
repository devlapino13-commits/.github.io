CREATE PROCEDURE AAREP_FOR_TSB_SECTION_1
RETURNS (
    CM_COMPANY_FULL_NAME VARCHAR(100),
    REG_OKPO VARCHAR(20),
    REG_INN VARCHAR(20),
    OS_GOS_REG_NUMBER VARCHAR(100),
    REG_REGIST_NO VARCHAR(100),
    REG_ADDRESS_LEGAL_STR VARCHAR(255),
    REG_ADDRESS_FACT_STR VARCHAR(255),
    REG_PHONE VARCHAR(50),
    REG_EMAIL VARCHAR(50),
    REG_WEB_SITE VARCHAR(255),
    ADDRESS_DLYA_POLUCH_POCH VARCHAR(255))
AS
declare variable reg_address_legal id_type;
declare variable reg_address_fact id_type;
begin
--  Полное фирменное наименование
  select cm_company_full_name
  from company
  into cm_company_full_name;

  select reg_okpo, -- Код по Общероссийскому классификатору организационно-правовой формы (ОКОПФ)
         reg_inn, -- Идентификационный номер налогоплательщика (ИНН)
         reg_regist_no os_gos_reg_number, -- Основной государственный регистрационный номер (ОГРН)
         '2120177001168 от 28.02.2012 г.' reg_regist_no, -- Регистрационный номер в государственном реестре микрофинансовых организаций
         reg_address_legal, -- Место нахождения и адрес
         reg_address_fact, -- Адрес фактического нахождения
         reg_phone, -- Номер телефона
         'office@molbulak.ru', -- Адрес электронной почты
         'www.molbulak.ru' -- Адрес сайта в информационно-телекоммуникационной сети "Интернет"
  from region
  where reg_unicode = 1010100
  into reg_okpo, reg_inn, os_gos_reg_number, reg_regist_no, reg_address_legal, reg_address_fact, reg_phone, reg_email,
       reg_web_site;

  select adl_address_str
  from addrlist
  where adl_id = :reg_address_legal
  into reg_address_legal_str;

  select adl_address_str
  from addrlist
  where adl_id = :reg_address_fact
  into reg_address_fact_str;

  address_dlya_poluch_poch = 'г. Москва. Наставнический переулок, дом 8, стр. 2';

  suspend;
end
