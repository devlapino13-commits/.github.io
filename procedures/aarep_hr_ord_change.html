CREATE PROCEDURE AAREP_HR_ORD_CHANGE(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    GROUP_TYPE VARCHAR(255),
    GROUP_TYPE_NUM INTEGER,
    C_ID DECIMAL(18,0),
    REG_NAME VARCHAR(255),
    C_NAME_FIO VARCHAR(255),
    JL_NAME VARCHAR(255),
    SAWT_NAME VARCHAR(255),
    DOC_DATE_BEGIN DATE,
    DOC_DATE_END DATE)
AS
declare variable ho_cause_id int;
begin
  for with all_activ_empl
      as (select r1.reg_name, cl.c_name_fio, jl_name, cl.c_id
          from hr_pstatus, hr_orders, region_list(:report_region_id, 1, -1, 1), joblist, region r1, clients cl, region br, jobkind
          where ho_id = ps_from_ho_id and
                (ps_from_reg_id = id_list) and
                ps_date_begin <= :report_date_end and
                ps_date_end >= :report_date_end and
                ps_from_reg_id = r1.reg_unicode and
                br.reg_unicode = r1.parent and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                ps_is_working = 1 and
                ps_from_c_id = c_id)

      select (case group_type_num
                when 1 then 'Ушедшие в декретный отпуск'
                when 2 then 'Окончание декретного отпуска'
                when 3 then 'Отзыв из отпуска'
                when 4 then 'Ротация'
                when 5 then 'Изменение условий договора'
              end), group_type_num, c_id, reg_name, c_name_fio, jl_name, sawt_name, doc_date_begin, doc_date_end, ho_cause_id
      from (select (case
                      when sawt_id in (16, 17, 18) and hrv_date_begin between :report_date_begin and :report_date_end then 1
                      when sawt_id in (16, 17, 18) and hrv_date_end_fact between :report_date_begin and :report_date_end then 2
                      when sawt_id in (15, 19) and hrv_date_begin between :report_date_begin and :report_date_end then 3
                    end) as group_type_num, c_id, reg_name, c_name_fio, jl_name, swt.sawt_name,
                         hrv_date_begin as doc_date_begin, hrv_date_end_fact as doc_date_end, ho_cause_id
            from all_activ_empl, hr_voctravel, sa_worktype swt, hr_orders
            where all_activ_empl.c_id = hrv_from_c_id and
                  ((hrv_date_begin between :report_date_begin and :report_date_end and
                  sawt_id in (16, 17, 18, 15, 19)) or
                  (hrv_date_end_fact between :report_date_begin and :report_date_end and
                  sawt_id in (16, 17, 18))) and
                  hrv_date_begin <= hrv_date_end_fact and
                  hrv_status = 1 and
                  ho_order_type = sawt_id and
                  hrv_from_ho_id = ho_id and
                  ho_status = 1
            union all
            select (case
                      when sawt_id = 2 then 4
                      when sawt_id = 3 then 5
                    end) as group_type_num, c_id, reg_name, c_name_fio, jl_name, swt.sawt_name,
                         hps.ps_date_begin as doc_date_begin, null as doc_date_end, ho_cause_id
            from all_activ_empl, sa_worktype swt, hr_orders, hr_pstatus hps
            where all_activ_empl.c_id = hps.ps_from_c_id and
                  hps.ps_date_begin between :report_date_begin and :report_date_end and
                  sawt_id in (2, 3) and
                  hps.ps_date_begin <= hps.ps_date_end and
                  hps.ps_status = 1 and
                  ho_order_type = sawt_id and
                  hps.ps_from_ho_id = ho_id and
                  ho_status = 1) t
      order by group_type_num, reg_name, c_name_fio
      into :group_type, :group_type_num, :c_id, :reg_name, :c_name_fio, :jl_name, :sawt_name, :doc_date_begin,
           :doc_date_end, :ho_cause_id
  do
  begin
    if (ho_cause_id > 0) then
      sawt_name = left(sawt_name || ': ' || (select rd_name from refdata where rd_id = :ho_cause_id),255);
    suspend;
  end
end
