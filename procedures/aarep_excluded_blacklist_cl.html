CREATE PROCEDURE AAREP_EXCLUDED_BLACKLIST_CL(
    REP_REGION_ID ID_TYPE,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    M_CLIENT_FIO D_STR_100,
    M_CLIENT_BIRTH_DATE D_DATE,
    M_CLIENT_PASS_NO VARCHAR(30),
    MAL_REQ_DATETIME D_DATETIME,
    MAL_APPR_DATETIME D_DATETIME,
    MAL_APPR_COMMENT VARCHAR(512),
    UN_FIO D_STR_100)
AS
declare variable reg_id d_integer;
declare variable mal_appr_user d_id;
declare variable mal_appr_user_tmp d_id;
begin
  mal_appr_user_tmp = 0;

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into :reg_id
  do
  begin
    for select mal.mal_appr_datetime, mal.mal_appr_user, mal.mal_req_comment, m.m_client_fio, m.m_client_birth_date,
               m.m_client_pass_no, mal.mal_req_datetime
        from morapprlist mal
        inner join moratory m on mal.mal_from_m_id = m.m_id and
              mal.mal_from_m_number = m.m_number
        where mal.mal_req_state = 0 and
              mal.mal_appr_status = 1 and
              cast(mal.mal_appr_datetime as date) between :rep_date_begin and :rep_date_end and
              mal.mal_from_reg_id = :reg_id
        order by mal.mal_appr_user
        into :mal_appr_datetime, :mal_appr_user, :mal_appr_comment, :m_client_fio, :m_client_birth_date,
             :m_client_pass_no, :mal_req_datetime
   do
   begin
    if (mal_appr_user <> mal_appr_user_tmp) then
    begin
        mal_appr_user_tmp = mal_appr_user;

        select un.un_fio
        from username un
        where un.un_id = :mal_appr_user
        into :un_fio;
    end
    suspend;
   end
  end
end
