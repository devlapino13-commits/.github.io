CREATE PROCEDURE AAREP_SIRENA_PAYED_REP_NEW(
    REP_PROJECT STR_100,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DAYS_QTY INTEGER)
RETURNS (
    K_KONTRACT_NO STR_100,
    OPER_CALLED STR_100,
    PROJECT STR_100,
    COMMAND_NAME STR_100,
    OK_OL_DATE DATE,
    DELINQ_DAY INTEGER,
    PAYED_OS CURR_TYPE,
    PAYED_PERC CURR_TYPE,
    PAYED_FINE CURR_TYPE,
    PAYED_SUM CURR_TYPE,
    BLOCK_DATE DATE,
    BLOCK_DELINQ_OS CURR_TYPE,
    BLOCK_DELINQ_PERC CURR_TYPE,
    BLOCK_DELINQ_FINE CURR_TYPE,
    BLOCK_DELINQ_SUM CURR_TYPE,
    BLOCK_DELINQ_DAYS INTEGER,
    CC_PROJECT STR_100,
    TRANSFER_CATEGORY STR_100)
AS
declare variable k_id id_type;
declare variable pant_date_begin date;
declare variable pant_date_end date;
declare variable kf_id id_type;
declare variable pant_id id_type;
declare variable is_ok2 smallint;
declare variable kf_officer id_type;
declare variable sbl_id integer;
declare variable is_ok3 smallint;
declare variable db_id integer;
declare variable oks_delinq_day integer;
declare variable project_id integer;
declare variable cc_project_code str_100;
declare variable law_rd_id integer;
declare variable oks_close_state integer;
declare variable delinq_date date;
declare variable ok_ol_date_tmp date;
declare variable ok_delinq_day integer;
declare variable call_no d_smallint;
declare variable month_m d_smallint;
declare variable first_date d_date;
begin

  if ((rep_project not containing ('OKTELL.Projects.Sirena')) and
      (rep_project not containing ('OKTELL.PROJECTS.INVESTIGATION_AND_RECOVERY')) and
      (rep_project not containing ('OKTELL.PROJECTS.AUTO_INFORMER'))) then
  begin
    sbl_id = (select cast(rd.rd_str_id as integer)
              from refdata rd
              where rd.rd_code = :rep_project);
  end

  law_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  db_id = f_get_db_id();

  for select pantk.pant_oper_called, pantk.pant_date_begin, pantk.pant_date_end, pantk.pant_from_k_id,
             kont.k_from_kf_id, pant_id, kont.k_kontract_no, rd.rd_name, rd.rd_id, rd_code
      from pantera_kontracts pantk
      inner join kontract kont on pantk.pant_from_k_id = kont.k_id
      inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
      where rd.rd_code containing (:rep_project) and
            pantk.pant_date_end >= :rep_date_begin and
            pantk.pant_date_end >= pant_date_begin -- исключаю передачи в один и тот же день
      order by pant_oper_called
      into :oper_called, :pant_date_begin, :pant_date_end, :k_id, :kf_id, :pant_id,
           :k_kontract_no, :cc_project, :project_id, :cc_project_code
  do
  begin
    command_name = '';

    if (:rep_project in ('OKTELL.PROJECTS.PANTERA','OKTELL.PROJECTS.SIRENA','OKTELL.PROJECTS.KS_FINANCE') and :db_id = 81001) then
    begin
        select pcm.pcm_name
        from pantera_commands pcm
        inner join pantera_kontracts_com pkc on pkc.pkc_from_pcm_id = pcm.pcm_id
        where pkc.pkc_from_pant_id = :pant_id
        into command_name;
    end

    block_date = pant_date_begin;

    if (pant_date_begin < rep_date_begin) then
      pant_date_begin = rep_date_begin;
    if (pant_date_end > rep_date_end) then
      pant_date_end = rep_date_end;

    if ((db_id in (39801, 41701)) and (not exists (select *
                                         from pantera_clients pc
                                         where pc.pcl_from_pan_id = :pant_id
                                               --and pc.pcl_res > 0  по просьбе Нуржан
                                               and pc.pcl_called_date <= :rep_date_end
    ))) then continue;

    payed_sum = 0;
    payed_os = 0;
    payed_perc = 0;
    payed_fine = 0;
    ok_ol_date = null;

    /*
    for select -sum(ok.ok_os), -sum(ok.ok_perc), -sum(ok.ok_fine), -sum(ok.ok_os + ok.ok_perc + ok.ok_fine),
               ok.ok_ol_date
        from operkred ok
        where ok.ok_type = 4 and
              ok.ok_ol_is_active = 1 and
              ok.ok_from_c_id = 0 and
              ok.ok_from_k_id = :k_id and
              ok.ok_ol_date between :pant_date_begin and :pant_date_end
        group by ok_ol_date
        into :payed_os, :payed_perc, :payed_fine, :payed_sum, :ok_ol_date  */


    for select ol_date, sum(payed_os), sum(payed_perc), sum(payed_fine), sum(payed_os + payed_perc + payed_fine)
    from
    (
          select ok.ok_ol_date as ol_date,
                 -ok.ok_os as payed_os,
                 -ok.ok_perc as payed_perc,
                 -ok.ok_fine as payed_fine
          from operkred ok
          where ok.ok_from_k_id = :k_id and
                ok.ok_type = 4 and
                ok.ok_ol_is_active = 1 and
                ok.ok_from_c_id = 0 and
                ok.ok_ol_date between :pant_date_begin and :pant_date_end
    
          union all
    
          select oko_ol_date as ol_date,
                 -okf.oko_os,
                 -okf.oko_perc,
                 -okf.oko_other
          from operkroff okf
          where okf.oko_ol_is_active = 1
          and okf.oko_from_k_id = :k_id
          and okf.oko_type = 4
          and okf.oko_ol_date between :pant_date_begin and :pant_date_end
        
          union all
        
          select oe.ox_ol_date as ol_date,
                 0,
                 0,
                 oe.ox_amount_minus
          from operextra oe
          where oe.ox_object_kind = 9
          and oe.ox_object_id = :k_id
          and oe.ox_from_rd_id = :law_rd_id
          and oe.ox_ol_date between :pant_date_begin and :pant_date_end
          and oe.ox_ol_is_active = 1
    ) A
    group by ol_date
    having sum(payed_os + payed_perc + payed_fine) > 0
    order by ol_date desc
    into :ok_ol_date, :payed_os, :payed_perc, :payed_fine, :payed_sum
    do
    begin
      is_ok2 = 0;

      if (db_id in (81001, 41701)) then
        is_ok2 = 1; -- для РФ, КГ нет необходимости подсчитывать, были ли успешные звонки или нет

      -- проверяю висела ли группа на ВС Сирена, на момент погашения или нет
      select kal_new_kf_officer
      from knaktlist kal
      where kal.kal_from_kf_id = :kf_id and
            :ok_ol_date between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into :kf_officer;

      if (kf_officer = 1000000000004) then
        project = 'Сирена';
      else
        project = 'СВЗ';

      -- проверяю, были ли успешные звонки
      if (is_ok2 = 0) then
      begin
        if (exists(select *
                   from pantera_clients pc
                   where pc.pcl_from_pan_id = :pant_id and
                         pc.pcl_res > 0 and
                         pc.pcl_called_date between :ok_ol_date - :rep_days_qty and :ok_ol_date rows 1)) then
          is_ok2 = 1;
      end
      if (is_ok2 = 0) then
      begin
        if (exists(select *
                   from pantera_contacts pcon
                   where pcon.pcon_from_pan_id = :pant_id and
                         pcon.pcon_res > 0 and
                         pcon.pcon_called_date between :ok_ol_date - :rep_days_qty and :ok_ol_date rows 1)) then
          is_ok2 = 1;
      end

      if (is_ok2 = 1) then
      begin
        is_ok3 = 0;
        if (rep_project = 'OKTELL.PROJECTS.AUTO_INFORMER') then
            is_ok3 = 1;

        if (db_id in (39801, 41701)) then
        begin
            call_no = 0;
            month_m = extract (month from :rep_date_begin) -2;
            first_date = cast('01.' || :month_m || '.' || extract(year from :rep_date_begin) as date);
        if (call_no = 0) then
        begin
           if (exists(select *
                      from pantera_clients pc
                      where pc.pcl_from_pan_id = :pant_id and
                            pc.pcl_res > 0 and
                            pc.pcl_called_date between :first_date and :rep_date_end)) then
           call_no = 1;
        end
        if (call_no = 0) then
        begin
          if (exists(select *
                     from pantera_contacts pcon
                     where pcon.pcon_from_pan_id = :pant_id and
                           pcon.pcon_res > 0 and
                           pcon.pcon_called_date between :first_date and :rep_date_end)) then
          call_no = 1;
        end

      if (call_no = 0) then continue; 
        end
        if (rep_project containing ('OKTELL.Projects.Sirena')) then
        begin
          if (not exists( -- если проект Сирена, группа не должна висеть на СБ в момент погашения
          select *
          from sb_akt sba
          where sba.sba_is_delete = 0 and
                sba.sba_from_k_id = :k_id and
                :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                sba.sba_from_sbl_id <> 0 -- важно!
                and
                sba.sba_status = 2)) then
            is_ok3 = 1;
        end
        else
        if (rep_project containing ('OKTELL.PROJECTS.INVESTIGATION_AND_RECOVERY')) then
        begin
          if (exists( -- если группа числилась за СБ из списка ДРВП.
          select *
          from sb_akt sba
          where sba.sba_is_delete = 0 and
                sba.sba_from_k_id = :k_id and
                :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                sba.sba_from_sbl_id in (87, 91, 92, 93, 95, 96, 97, 98, 99, 100,
                                        102, 103, 105, 106, 107, 108, 109) and
                sba.sba_status = 2)) then
            is_ok3 = 1;
        end
        else
        if (rep_project containing ('OKTELL.PROJECTS.PANTERA')) then
        begin
          if (cc_project_code = 'OKTELL.PROJECTS.PANTERA') then
          begin
            if (exists( -- если группа числилась за СБ Пантера
            select *
            from sb_akt sba
            where sba.sba_is_delete = 0 and
                  sba.sba_from_k_id = :k_id and
                  :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                  sba.sba_from_sbl_id = 82 and
                  sba.sba_status = 2)) then
              is_ok3 = 1;
          end
          if (cc_project_code = 'OKTELL.PROJECTS.PANTERA2') then
          begin
            if (exists( -- если группа числилась за СБ Пантера
            select *
            from sb_akt sba
            where sba.sba_is_delete = 0 and
                  sba.sba_from_k_id = :k_id and
                  :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                  sba.sba_from_sbl_id = 112 and
                  sba.sba_status = 2)) then
              is_ok3 = 1;
          end
        end
        if (rep_project containing ('OKTELL.PROJECTS.KS_FINANCE')) then
        begin
          if (cc_project_code = 'OKTELL.PROJECTS.KS_FINANCE_1') then
          begin
            if (exists( -- если группа числилась за СБ Пантера
            select *
            from sb_akt sba
            where sba.sba_is_delete = 0 and
                  sba.sba_from_k_id = :k_id and
                  :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                  sba.sba_from_sbl_id = 181 and
                  sba.sba_status = 2)) then
              is_ok3 = 1;
          end
          if (cc_project_code = 'OKTELL.PROJECTS.KS_FINANCE_2') then
          begin
            if (exists( -- если группа числилась за СБ Пантера
            select *
            from sb_akt sba
            where sba.sba_is_delete = 0 and
                  sba.sba_from_k_id = :k_id and
                  :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                  sba.sba_from_sbl_id = 182 and
                  sba.sba_status = 2)) then
              is_ok3 = 1;
          end
        end
        else
        begin
          if (exists( -- проверяю, висела ли группа на СБ в момент принятия погашения или нет
          select *
          from sb_akt sba
          where sba.sba_is_delete = 0 and
                sba.sba_from_k_id = :k_id and
                :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                sba.sba_from_sbl_id = :sbl_id -- Важно!
                and
                sba.sba_status = 2)) then
            is_ok3 = 1;
        end

        if (is_ok3 = 1) then
        begin
          transfer_category = '';

          if (rep_project = 'OKTELL.Projects.Pantera') then
          begin
            transfer_category = 'Первичная';
            if (exists(select *
                       from pantera_kontracts pantk
                       where pantk.pant_id <> :pant_id and
                             pantk.pant_from_k_id = :k_id and
                             pantk.pant_date_begin <= :pant_date_begin and
                             coalesce(pantk.pant_is_dismiss, 0) = 0 -- важно, чтобы не поймать увольнения
                             and
                             pantk.pant_rd_project_type = :project_id and
                             pantk.pant_oper_called <> :oper_called)) then
            begin
              transfer_category = 'Вторичная';
            end
          end

          delinq_day = null;
          --ok_delinq_day = null;
          oks_delinq_day = 0;
          oks_close_state = null;

          -- проверяю, была ли группа в просрочке в день принятия погашения или нет
          if (rep_project containing ('OKTELL.Projects.Sirena')) then
          begin
            select iif(oks_date_end > cast(:ok_ol_date as date),
                       iif(cast(:ok_ol_date as date) - oks_delinq_date < 0, 0, cast(:ok_ol_date as date) - oks_delinq_date),
                       iif(oks_date_end - oks_delinq_date < 0, 0, oks_date_end - oks_delinq_date)) as delinq_days,
                       oks.oks_close_state
            from oksummary oks
            where :ok_ol_date between oks_date_begin and oks_date_end and
                  oks_from_k_id = :k_id
            into oks_delinq_day, oks_close_state;

            ok_delinq_day = null;

            select max(ok_ol_date - ok_sh_date)
            from operkred ok
            where ok_from_k_id = :k_id and
                  ok_ol_date = :ok_ol_date and
                  ok.ok_ol_is_active = 1 and
                  ok.ok_from_c_id = 0
            into ok_delinq_day;

            if (ok_delinq_day is null) then
              ok_delinq_day = 0;

            if (oks_close_state = 1) then
            begin
                delinq_date = null;
            
                select oks_delinq_date
                from oksummary
                where oks_from_k_id = :k_id and
                      oks_delinq_date < '31.12.9999'
                order by oks_date_begin desc
                rows 1
                into delinq_date;

                if (delinq_date is not null) then
                  oks_delinq_day = :ok_ol_date - delinq_date;
            end

            if (ok_delinq_day <= oks_delinq_day) then
              delinq_day = oks_delinq_day;
            if (ok_delinq_day > oks_delinq_day) then
              delinq_day = ok_delinq_day;
          end

          -- если на утро дня принятия погашения, группа была в просрочке (для проектов не Сирены проверять это не нужно)
          if ((delinq_day > 0) or (rep_project not containing ('OKTELL.Projects.Sirena'))) then
          begin
            oks_close_state = null;
            ok_ol_date_tmp = ok_ol_date - 1;

            select :ok_ol_date_tmp - oks_delinq_date,
                   oks_close_state 
            from oksummary oks
            where oks.oks_from_k_id = :k_id and
                  :ok_ol_date_tmp between oks.oks_date_begin and oks.oks_date_end
            into :delinq_day, :oks_close_state;

            if (oks_close_state = 1) then
            begin
                delinq_date = null;
                
                select oks_delinq_date
                from oksummary
                where oks_from_k_id = :k_id and
                      oks_delinq_date < '31.12.9999'
                order by oks_date_begin desc
                rows 1
                into delinq_date;

                if (delinq_date is not null) then
                  delinq_day = :ok_ol_date_tmp - delinq_date;
            end

            if ((delinq_day >= 0) or (rep_project not containing ('OKTELL.Projects.Sirena'))) then
            begin
              if (delinq_day < 0) then
                delinq_day = 0;

              --pt_id = null;
              block_delinq_os = null;
              block_delinq_perc = null;
              block_delinq_fine = null;
              block_delinq_sum = null;
              block_delinq_days = null;

              -- проверяю по таблице сводных данных, блокировки группы
              /*
              select pantt.pt_id, pantt.pt_delinq_os, pantt.pt_delinq_perc, pantt.pt_delinq_fine,
                     pantt.pt_delinq_os + pantt.pt_delinq_perc + pantt.pt_delinq_fine, pantt.pt_delinq_days
              from pantera_transfer pantt
              where pantt.pt_input_pant_id = :pant_id
              into :pt_id, :block_delinq_os, :block_delinq_perc, :block_delinq_fine, :block_delinq_sum,
                   :block_delinq_days;
                   */

              --if (pt_id is null) then -- если сводных данных нет, рассчитываю на день блокировки
              --begin
                select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine),
                       sum(zad_os) + sum(sh_calc_perc) + sum(sh_calc_fine)
                from aarep_make_zad_list(:k_id, :block_date, 1)
                where coalesce(sh_date, cast('01.01.1900' as date)) < :block_date
                into :block_delinq_os, :block_delinq_perc, :block_delinq_fine, :block_delinq_sum;

                block_delinq_days = null;
                oks_close_state = null;

                select :block_date - oks_delinq_date,
                       oks_close_state
                from oksummary oks
                where oks.oks_from_k_id = :k_id and
                      :block_date between oks.oks_date_begin and oks.oks_date_end
                into :block_delinq_days,
                     :oks_close_state;

                if (oks_close_state = 1) then
                begin
                    delinq_date = null;
                
                    select oks_delinq_date
                    from oksummary
                    where oks_from_k_id = :k_id and
                          oks_delinq_date < '31.12.9999'
                    order by oks_date_begin desc
                    rows 1
                    into delinq_date;
    
                    if (delinq_date is not null) then
                      block_delinq_days = :block_date - delinq_date;
                end
              --end

              block_delinq_os = coalesce(block_delinq_os, 0);
              block_delinq_perc = coalesce(block_delinq_perc, 0);
              block_delinq_fine = coalesce(block_delinq_fine, 0);
              block_delinq_sum = coalesce(block_delinq_sum, 0);
              block_delinq_days = coalesce(block_delinq_days, 0);

              suspend;
            end
          end
        end
      end
    end
  end
end
