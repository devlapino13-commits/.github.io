CREATE PROCEDURE AAREP_CALC_PERC_ONE_IPF_TMP(
    K_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    OK_OL_DATE D_DATE,
    LAST_SH_DATE D_DATE,
    SH_PROLONG D_SMALLINT,
    KS_DATE_BEGIN D_DATE,
    KS_DATE_END D_DATE)
AS
declare variable max_sh_date d_date;
declare variable k_kont_last_date d_date;
declare variable tmp_dd d_date;
declare variable begin_suspended d_smallint;
begin
  select ks_date_begin, ks_date_end, k_kont_last_date
  from kontstate, kontract
  where ks_from_k_id = :k_id and
        ks_from_k_id = k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_date_begin <= :rep_date_end
  into ks_date_begin, ks_date_end, k_kont_last_date;

  if (ks_date_begin is null) then
    exit;

  if (ks_date_end < '31.12.2999') then
    ks_date_end = ks_date_end + 1;

  sh_prolong = 0;
  ok_ol_date = ks_date_begin;
  last_sh_date = k_kont_last_date;
  suspend;

  if (rep_date_end > ks_date_end) then
    rep_date_end = ks_date_end;

  if (ks_date_begin > rep_date_begin) then
    begin_suspended = 1;

  ks_date_begin = ks_date_begin + 1;

  for select ok_ol_date, max(ok_sh_date)
  from operkred
  where ok_from_k_id = :k_id and
        ok_type <= 3 and
        ok_ol_is_active = 1 and
        ok_ol_date between :ks_date_begin and minvalue(:rep_date_end, :ks_date_end) and
        ok_sh_date > :k_kont_last_date
  group by ok_ol_date
  into tmp_dd, max_sh_date
  do
  begin
    sh_prolong = 0;
    if (rep_date_begin < tmp_dd and begin_suspended = 0) then
    begin
      ok_ol_date = rep_date_begin;
      begin_suspended = 1;
      suspend;
    end

    ok_ol_date = tmp_dd;

    if (last_sh_date < max_sh_date) then
    begin
      sh_prolong = 1;
      last_sh_date = max_sh_date;
      suspend;
    end
  end

  sh_prolong = 0;
  if (begin_suspended = 0) then
  begin
    ok_ol_date = rep_date_begin;
    suspend;
  end

  ok_ol_date = minvalue(rep_date_end, ks_date_end);
  suspend;
end
