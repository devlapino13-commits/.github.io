CREATE PROCEDURE HP_CLEAR_PATRONIMIC(
    NAME D_STR_CODE)
RETURNS (
    RESULT D_STR_CODE)
AS
begin
    result = name;
    result = replace(result, 'uulu', '');
    result = replace(result, 'kyzy', '');
    result = replace(result, 'kizi', '');
    result = replace(result, 'qizi', '');
    result = replace(result, 'ogli', '');
    result = replace(result, 'ugli', '');
    result = replace(result, 'уулу', '');
    result = replace(result, 'кызы', '');
    result = replace(result, 'угли', '');
    result = replace(result, ' ', '');
    suspend;
end
