CREATE PROCEDURE HP_EARLYPAID_NOVEMBER_SET(
    EN_ID D_INTEGER,
    REP_DATE D_DATE = current_date)
RETURNS (
    OL_ID D_ID)
AS
declare variable k_id d_id;
declare variable is_ipf d_integer;
declare variable c_id d_id;
declare variable curr_perc d_currency;
declare variable reg_id d_integer;
declare variable un_id d_id;
declare variable acc_dt d_str_20;
begin
  un_id = 1000000000000;
  for
    select
      en_k_id,
      en_is_ipf,
      en_c_id,
      en_curr_perc,
      en_reg_id
    from earlypaid_november
    where en_id = :en_id
  into
    :k_id,
    :is_ipf,
    :c_id,
    :curr_perc,
    :reg_id
  do
  begin
    acc_dt = (
      select al_id
      from acc_list
      where al_from_reg_id = :reg_id
        and al_account = iif(:is_ipf = 1, '11531/202', '11531/201'));
    select
      ol_id
    from  hp_do_repayment (
      :k_id,
      0,                    -- c_id
      :rep_date,            -- rep_date
      :curr_perc,           -- repay_amount
      1,                    -- repay_to_os
      0,                    -- repay_to_perc
      0,                    -- repay_to_fine
      0,                    -- repay_to_kommis
      1,                    -- need_save
      0,                    -- is_full_repay
      0,                    -- is_writeoff
      :reg_id,              -- repay_fact_office_id
      -3,                   -- repay_dt_account_kind
      :acc_dt,              -- repay_dt_account_no
      0,                    -- repay_curr_rate
      :un_id,               -- repay_user_id
      :c_id,                -- repay_document_client
      '',                   -- repay_doc_client_name_reason
      0,                    -- repay_manager_c_id
      0)                    -- repay_from_rd_id
    into
      :ol_id;

    if (is_ipf = 1) then
      delete from operkred
      where ok_from_k_id = :k_id
        and ok_from_ol_id = :ol_id
        and ok_type = 3;

    update earlypaid_november set
      en_ol_id = :ol_id
    where en_id = :en_id;
    suspend;
  end
end
