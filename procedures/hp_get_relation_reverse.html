CREATE PROCEDURE HP_GET_RELATION_REVERSE(
    REL_ID VARCHAR(50),
    REL_SEX INTEGER)
RETURNS (
    REL_CODE VARCHAR(50))
AS
declare variable n integer;
declare variable i integer;
declare variable tmp_code varchar(50);
declare variable rel_short_id varchar(50);
declare variable tmp_code_new varchar(50);
begin
-- Процедура отвечает на вопрос:
-- если я мужчина (Rel_Sex=1) или женщина (Rel_Sex), то кем является для меня Rel_ID ?

-- Входящий параметр Rel_ID - это длинный код, для которого необходимо найти обратного родственника
-- Входящий параметр Rel_Sex - это пол ЛИЧНОСТИ

-- Кодовые обозначения:
-- 1 - Супруг
-- 2 - Супруга
-- 3 - Отец
-- 4 - Мать
-- 5 - Сын
-- 6 - Дочь
-- 7 - Брат
-- 8 - Сестра
-- 9 - Гражданский супруг(а)  -> Гражданский супруг(а)

  rel_code = '';
  n = char_length(rel_id);
  i = 1;
  while (i <= n) do
  begin
-- длинный код делим по 1 символe, далее каждой связи находим обратную
    tmp_code = substring(:rel_id from :i for 1);

-- В зависимости от поля, вычисление обратной родственной связи
    if (rel_sex = 1) then
    begin
-- Если ЛИЧНОСТЬ мужской пол, то
      tmp_code_new = (case
                        when tmp_code = '1' then '2' -- супруг  -> супруга
                        when tmp_code = '2' then '1' -- супруга -> супруг
                        when tmp_code = '3' then '5' -- отец    -> сын
                        when tmp_code = '4' then '5' -- мать    -> сын
                        when tmp_code = '5' then '3' -- сын     -> отец
                        when tmp_code = '6' then '3' -- дочь    -> отец
                        when tmp_code = '7' then '7' -- брат    -> брат
                        when tmp_code = '8' then '7' -- сестра  -> брат
                        when tmp_code = '9' then '9' -- Гражданский супруг(а)  -> Гражданский супруг(а)
                        else 'x'
                      end);
    end
    else
    begin
-- Если ЛИЧНОСТЬ женский пол, то
      tmp_code_new = (case
                        when tmp_code = '1' then '2' -- супруг  -> супруга
                        when tmp_code = '2' then '1' -- супруга -> супруг
                        when tmp_code = '3' then '6' -- отец    -> дочь
                        when tmp_code = '4' then '6' -- мать    -> дочь
                        when tmp_code = '5' then '4' -- сын     -> мать
                        when tmp_code = '6' then '4' -- дочь    -> мать
                        when tmp_code = '7' then '8' -- брат    -> сестра
                        when tmp_code = '8' then '8' -- сестра  -> сестра
                        when tmp_code = '9' then '9' -- Гражданский супруг(а)  -> Гражданский супруг(а)
                        else 'x'
                      end);
    end
-- строим цепочку начиная с последнего
    rel_code = tmp_code_new || rel_code;

-- вычисляем пол промежуточного узла (четные - женщины, нечетные - мужчины)
    if (tmp_code <> 'x') then
      rel_sex = (mod(cast(tmp_code as integer), 2));
    else
      rel_sex = 0;

-- переходим на следующую пару кодов
    i = i + 1;
  end

-- ищем сокращенный вид родственной связи, например, сестра вашего брата, является вашей сестрой: 78 -> 8
  select rel_code
  from hp_get_relation_short_id(:rel_code, :rel_sex)
  into :rel_code;

  if (rel_code = '') then
    rel_code = 'x';
  suspend;
end
