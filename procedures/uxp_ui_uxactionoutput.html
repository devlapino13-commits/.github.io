Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UI_UXACTIONOUTPUT(
    UA_UUID D_UUID,
    VER_KIND D_STR_CODE = null)
RETURNS (
    UAP_FR_UUID D_UUID,
    UAP_FIELD D_STR_CODE,
    UAP_SOURCE_FR_UUID D_UUID,
    UAP_SOURCE_FIELD D_STR_CODE)
AS
declare variable uxa_owner d_uuid;
declare variable ufm_uuid d_uuid;
begin
  select ua_uxa_owner
  from uxp$uxalias_info(:ua_uuid)
  into :uxa_owner;

  ufm_uuid = uxf$ver(uxa_owner, ver_kind);

  select ufr_uuid
  from ux$frame
  where ufr_owner = :ufm_uuid and
        ufr_is_mainframe = 1
  into uap_fr_uuid;

  if (uap_fr_uuid is null) then
    exit;

  for select replace(uap_param_name, 'out$', ''), ufp_ufr_uuid, uf_field_name
  from ux$actionparam, ux$frparam, ux$field
  where uap_ua_uuid = :ua_uuid and
        uap_is_output = 1 and
        uap_owner_ufp_uuid = ufp_uuid and
        ufp_uf_uuid = uf_uuid
  into uap_field, uap_source_fr_uuid, uap_source_field
  do
    suspend;
end
