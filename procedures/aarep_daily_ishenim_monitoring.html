CREATE PROCEDURE AAREP_DAILY_ISHENIM_MONITORING(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REGION_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    COUNT_OF_EMPLOYEES D_INTEGER,
    COUNT_OF_STANDARD_ISSUES D_INTEGER,
    COUNT_OF_STANDARD_ISSUES_CD D_INTEGER,
    SUM_OF_STANDARD_ISSUES D_CURRENCY,
    SUM_OF_STANDARD_ISSUES_CD D_CURRENCY,
    COUNT_OF_CONSULT_ISSUES D_INTEGER,
    COUNT_OF_CONSULT_ISSUES_CD D_INTEGER,
    SUM_OF_CONSULT_ISSUES D_CURRENCY,
    SUM_OF_CONSULT_ISSUES_CD D_CURRENCY,
    SUM_OF_CONSULT_FEE D_CURRENCY,
    SUM_OF_CONSULT_FEE_CD D_CURRENCY,
    CNT_INS_ISSUES D_INTEGER,
    SUM_OF_INS_ISSUES D_CURRENCY,
    SUM_OF_INS_FEE D_CURRENCY,
    CNT_OF_INSUR_ISSUES_CD D_INTEGER,
    SUM_OF_INSUR_ISSUES_CD D_CURRENCY,
    SUM_OF_CONSULT_INS_ISSUES D_CURRENCY)
AS
declare variable k_id d_id;
declare variable k_fact_disb_date d_date;
declare variable office_id d_integer;
declare variable pr_code d_str_64;
declare variable k_amount_approved d_currency;
declare variable consult_fee d_currency;
declare variable ins_fee d_currency;
begin
    /* CD в названии переменной - текущая дата */
    for select
        rl.parent_name,
        reg.reg_unicode,
        reg.reg_name
    from region reg
    inner join region_list(1000000, 1, 3, 1) rl
        on reg.reg_unicode = rl.id_list
    order by rl.parent_name, reg.reg_name
    into region_name, office_id, office_name
    do
    begin
        count_of_employees = 0;

        count_of_standard_issues = 0;
        count_of_standard_issues_cd = 0;
        sum_of_standard_issues = 0;
        sum_of_standard_issues_cd = 0;

        count_of_consult_issues = 0;
        count_of_consult_issues_cd = 0;
        sum_of_consult_issues = 0;
        sum_of_consult_issues_cd = 0;
        sum_of_consult_fee = 0;
        sum_of_consult_fee_cd = 0;

        cnt_ins_issues = 0;
        sum_of_ins_issues = 0;
        sum_of_ins_fee = 0;
        cnt_of_insur_issues_cd = 0;
        sum_of_insur_issues_cd = 0;
        sum_of_consult_ins_issues = 0;

        select count(ps.ps_id)
        from hr_pstatus ps
        inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
        inner join jobkind jk on jl.jl_kind = jk.jk_id
            and jk.jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior')
        where ps.ps_from_reg_id = :office_id
            and ps.ps_is_working = 1
            and ps.ps_status = 1
            and :rep_date_end between ps.ps_date_begin and ps.ps_date_end
            and not exists (
                select *
                from hr_voctravel, hr_orders
                where hrv_from_ho_id = ho_id
                  and hrv_from_c_id = ps.ps_from_c_id
                  and ho_order_type in (16, 17, 18, 24)
                  and :rep_date_end between hrv_date_begin and hrv_date_end_fact
                  and hrv_status = 1)
        into count_of_employees;

        if (coalesce(count_of_employees, 0) = 0) then
            continue;

        for select
            k.k_id,
            k.k_amount_approved,
            k.k_fact_disb_date,
            pr.pr_code
        from kontstate ks
        inner join kontract k on ks.ks_from_k_id = k.k_id
        inner join product pr on k.k_from_pr_id = pr.pr_id
            and pr.pr_active = 1
            and pr.pr_code not containing ('.ipf.')
            and pr.pr_date_end >= :rep_date_begin
            and pr.pr_date_begin <= :rep_date_end
        where ks.ks_ol_is_delete = 0
            and ks.ks_state = 1
            and ks.ks_region_id = :office_id
            and ks.ks_date_begin between :rep_date_begin and :rep_date_end
        into
            k_id,
            k_amount_approved,
            k_fact_disb_date,
            pr_code
        do
        begin
            consult_fee = 0;
            ins_fee = 0;

            -- стандартные выдачи
            count_of_standard_issues = count_of_standard_issues + 1;
            sum_of_standard_issues = sum_of_standard_issues + k_amount_approved;

            if (k_fact_disb_date = rep_date_end) then
            begin
                count_of_standard_issues_cd = count_of_standard_issues_cd + 1;
                sum_of_standard_issues_cd = sum_of_standard_issues_cd + k_amount_approved;
            end

            -- получаем оплату страхования
            select coalesce(sum(oko.ok_amount), 0)
            from operkontract oko
            inner join operlist ol
                on oko.ok_from_ol_id = ol.ol_id
                and ol.is_delete = 0
            where oko.ok_from_k_id = :k_id
              and oko.ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance')
            into ins_fee;

            -- классика со страхованием
            if (ins_fee > 0) then
            begin
                cnt_ins_issues = cnt_ins_issues + 1;
                sum_of_ins_issues = sum_of_ins_issues + k_amount_approved;
                sum_of_ins_fee = sum_of_ins_fee + ins_fee;
    
                if (k_fact_disb_date = rep_date_end) then
                begin
                    cnt_of_insur_issues_cd  = cnt_of_insur_issues_cd + 1;
                    sum_of_insur_issues_cd = sum_of_insur_issues_cd + ins_fee;
                end
            end

            ins_fee = 0;

            -- консультация
            if (pr_code containing '.consultation.') then
            begin
                select coalesce(sum(ok.ok_amount), 0)
                from operkontract ok
                inner join operlist ol
                    on ok.ok_from_ol_id = ol.ol_id
                    and ol.is_delete = 0
                where ok.ok_operation_type = f_get_rd_id('CashTransactionIncom.Consultation')
                  and ok.ok_from_k_id = :k_id
                into consult_fee;

                count_of_consult_issues = count_of_consult_issues + 1;
                sum_of_consult_issues = sum_of_consult_issues + k_amount_approved;
                sum_of_consult_fee = sum_of_consult_fee + consult_fee;

                if (k_fact_disb_date = rep_date_end) then
                begin
                    count_of_consult_issues_cd = count_of_consult_issues_cd + 1;
                    sum_of_consult_issues_cd = sum_of_consult_issues_cd + k_amount_approved;
                    sum_of_consult_fee_cd = sum_of_consult_fee_cd + consult_fee;
                end

                select coalesce(sum(oko.ok_amount), 0)
                from operkontract oko
                inner join operlist ol
                    on oko.ok_from_ol_id = ol.ol_id
                    and ol.is_delete = 0
                where oko.ok_from_k_id = :k_id
                  and oko.ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance')
                into ins_fee;

                if (ins_fee > 0) then
                begin
                    sum_of_consult_ins_issues = sum_of_consult_ins_issues + k_amount_approved;
                end
            end
        end

        suspend;
    end
end
