CREATE PROCEDURE HP_PRODTYPEBLANK_LIST(
    BLN_ID INTEGER)
RETURNS (
    PRTB_FROM_PRT_ID INTEGER,
    PRTB_FROM_BLN_ID INTEGER,
    PRT_NAME VARCHAR(255),
    PRT_USED_PRODUCT INTEGER,
    PRT_USED_BLANK INTEGER,
    PRT_CHECKED INTEGER)
AS
begin
  for select prt_id, prt_name
      from prodtype
      where prt_is_delete = 0
      into prtb_from_prt_id, prt_name
  do
  begin
    prtb_from_bln_id = bln_id;
    prt_used_product = 0;
    prt_used_blank = 0;
    prt_checked = 0;

    select count(*)
    from prodtypelink, product
    where prtl_from_prt_id = :prtb_from_prt_id and
          prtl_from_pr_id = pr_id and
          coalesce(prtl_date_begin, '01.01.1990') <= current_date and
          coalesce(prtl_date_end, '31.12.2999') >= current_date and
          pr_date_begin <= current_date and
          pr_date_end >= current_date
    into prt_used_product;
    if (prt_used_product is null) then
      prt_used_product = 0;

    select count(*)
    from prodtypeblank, blankblob
    where prtb_from_prt_id = :prtb_from_prt_id and
          prtb_from_bln_id = blb_from_bln_id and
          blb_date_end >= '31.12.2999' and
          (blb_blob is not null or
          blb_frx_blob is not null)
    into prt_used_blank;
    if (prt_used_blank is null) then
      prt_used_blank = 0;

    if (exists(select *
               from prodtypeblank
               where prtb_from_prt_id = :prtb_from_prt_id and
                     prtb_from_bln_id = :bln_id)) then
      prt_checked = 1;

    suspend;
  end
end
