CREATE PROCEDURE HP_PRODTYPELINK_HISTORY(
    PR_ID INTEGER,
    PRT_ID INTEGER)
RETURNS (
    PRTL_ID INTEGER,
    PRTL_DATE_BEGIN DATE,
    PRTL_DATE_END DATE,
    PRTL_UN_NAME VARCHAR(30),
    PRTL_DATETIME TIMESTAMP)
AS
begin
  for select prtl_id, prtl_date_begin, prtl_date_end, prtl_datetime, un_name
      from prodtypelink, username
      where prtl_from_prt_id = :prt_id and
            prtl_from_pr_id = :pr_id and
            prtl_from_un_id = un_id
      order by prtl_date_begin desc
      into prtl_id, prtl_date_begin, prtl_date_end, prtl_datetime, prtl_un_name
  do
  begin
    suspend;
  end
end
