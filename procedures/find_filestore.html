CREATE PROCEDURE FIND_FILESTORE(
    FIND_REGION_ID INTEGER,
    IS_ALL INTEGER)
RETURNS (
    FS_ID INTEGER,
    FS_FILENAME VARCHAR(1020),
    FS_FILESIZE_KB CURR_TYPE,
    FS_DATETIME TIMESTAMP,
    K_KONTRACT_NO VARCHAR(20),
    KAL_NEW_KF_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    C_NAME_FIO VARCHAR(100),
    REG_NAME VARCHAR(50),
    UN_NAME VARCHAR(50),
    FS_IS_VIEWED SMALLINT,
    FS_VIEW_UN_ID DECIMAL(18,0),
    FS_VIEW_COMMENT VARCHAR(255))
AS
declare variable fs_k_id decimal(18,0);
begin
  for select fs_id, 'v:\' || fs_filename as fs_filename, fs_filesize_kb, fs_datetime, un_name, fs_is_viewed,
             fs_view_un_id, fs_object_id, fs_view_comment
      from filestore, username
      where fs_object_kind = 0 and
            fs_subject_kind = 0 and
            fs_status = 1 and
            ((:is_all = 1) or
            (:is_all = 0 and
            coalesce(fs_is_viewed, 0) = 0)) and
            fs_from_un_id = un_id
      order by fs_id asc
      into :fs_id, :fs_filename, :fs_filesize_kb, :fs_datetime, :un_name, :fs_is_viewed, :fs_view_un_id, :fs_k_id,
           :fs_view_comment
  do
  begin
    k_kontract_no = null;
    select k_kontract_no, kal_new_kf_name, k_disb_date, c_name_fio, reg_name
    from kontract, knaktlist, clients, region, region_list(:find_region_id, 1, -1, 1)
    where k_id = :fs_k_id and
          k_from_kf_id = kal_from_kf_id and
          kal_new_kf_officer = c_id and
          cast(kal_new_reg_id as integer) = reg_unicode and
          kal_date <= current_date and
          kal_date_end >= current_date and
          knaktlist.is_delete = 0 and
          cast(kal_new_reg_id as integer) = id_list
    into :k_kontract_no, :kal_new_kf_name, :k_disb_date, :c_name_fio, :reg_name;
    if (k_kontract_no is not null) then
      suspend;
  end
end
