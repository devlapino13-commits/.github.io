CREATE PROCEDURE HP_BKI_GET_PACKET_NAME(
    PACKET_DATE D_DATE,
    BKI_CODE D_STR_CODE NOT NULL,
    COUNTER D_INTEGER,
    PACKET_FORMAT D_SMALLINT)
RETURNS (
    PACKET_NAME D_STR_CODE,
    NEW_COUNTER D_INTEGER)
AS
begin
  if (bki_code = 'ru.equifax') then
    execute procedure hp_bki_get_packet_name_equifax(packet_date, counter, packet_format)
    returning_values packet_name, new_counter;
  else if (bki_code = 'ru.nbki') then
    execute procedure hp_bki_get_packet_name_nbki(packet_date, counter, packet_format)
    returning_values packet_name, new_counter;
  else
    exception e_default 'no algorithm generate packet_name for ' || bki_code;

  suspend;
end
