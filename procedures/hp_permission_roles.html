CREATE PROCEDURE HP_PERMISSION_ROLES(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID DECIMAL(18,0))
RETURNS (
    RN_ID DECIMAL(18,0),
    RN_NAME VARCHAR(30),
    RN_MEMO VARCHAR(2048),
    PM_ID INTEGER,
    PM_DATE_BEGIN DATE,
    PM_DATE_END DATE,
    PM_ACCESS_KIND INTEGER,
    PM_IS_DELETE INTEGER,
    PM_SELECTED INTEGER,
    USER_COUNT INTEGER,
    RN_ONLY_SUPERUSER INTEGER)
AS
declare variable un_id decimal(18,0);
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
declare variable uniqtab_menulist integer;
declare variable uniqtab_permactions integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;
  select ukt_id
  from hp_get_uniqtable_id('PermActions')
  into :uniqtab_permactions;

  if (object_kind = '0') then
    object_kind = 'Username';
  if (object_kind = '1') then
    object_kind = 'Rolename';

  user_count = 0;
  rn_only_superuser = 0;
  if (object_kind = 'username') then
  begin
    for select rn_id, rn_name, rn_memo, pm_id, pm_date_begin, pm_date_end, pm_access_kind, pm_is_delete
        from rolename
        left join permissions on (pm_object_kind = :uniqtab_username and
              pm_object_id = :object_id and
              pm_subject_kind = :uniqtab_rolename and
              pm_subject_id = rn_id and
              pm_is_delete = 0 and
              (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
              (coalesce(pm_date_end, '31.12.2999') >= current_date))
        into :rn_id, rn_name, rn_memo, pm_id, pm_date_begin, pm_date_end, pm_access_kind, pm_is_delete
    do
    begin
      if (pm_id is null) then
        pm_selected = 0;
      else
        pm_selected = 1;
      pm_access_kind = 1;

      if (pm_id is null) then
        pm_id = -1;
      if (pm_is_delete is null) then
        pm_is_delete = 0;

      user_count = 0;
      for select distinct pm_object_id
          from permissions
          where pm_subject_kind = :uniqtab_rolename and
                pm_subject_id = :rn_id and
                pm_is_delete = 0 and
                pm_object_kind = :uniqtab_username and
                (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(pm_date_end, '31.12.2999') >= current_date)
          into :un_id
      do
      begin
        if (exists(select *
                   from hp_un_is_active(:un_id)
                   where un_is_active = 1)) then
          user_count = user_count + 1;
      end

      if (exists(select *
                 from permissions, menulist
                 where pm_object_kind = :uniqtab_rolename and
                       pm_object_id = :rn_id and
                       pm_subject_kind = :uniqtab_menulist and
                       pm_subject_id = ml_id and
                       pm_is_delete = 0 and
                       ml_secure_kind = -1)) then
        rn_only_superuser = 1;
      else
        rn_only_superuser = 0;

      if (rn_only_superuser = 0) then
      begin
        if (exists(select *
                   from permissions, permactions
                   where pm_object_kind = :uniqtab_rolename and
                         pm_object_id = :rn_id and
                         pm_subject_kind = :uniqtab_permactions and
                         pm_subject_id = pa_id and
                         pm_is_delete = 0 and
                         pa_secure_kind = -1)) then
          rn_only_superuser = 1;
        else
          rn_only_superuser = 0;
      end

      suspend;
    end
  end
  if (object_kind = 'rolename') then
  begin
    for select rn_id, rn_name, rn_memo
        from rolename
        where rn_id = :object_id
        into :rn_id, rn_name, rn_memo
    do
    begin
      if (rn_id is null) then
        pm_selected = 0;
      else
        pm_selected = 1;
      pm_access_kind = 1;

      if (pm_is_delete is null) then
        pm_is_delete = 0;

      user_count = 0;
      for select distinct pm_object_id
          from permissions
          where pm_subject_kind = :uniqtab_rolename and
                pm_subject_id = :rn_id and
                pm_object_kind = :uniqtab_username and
                pm_is_delete = 0 and
                (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(pm_date_end, '31.12.2999') >= current_date)
          into :un_id
      do
      begin
        if (exists(select *
                   from hp_un_is_active(:un_id)
                   where un_is_active = 1)) then
          user_count = user_count + 1;
      end
      suspend;
    end
    pm_selected = 0;
    pm_access_kind = 1;
    for select pm_object_id, un_name, un_fio
        from permissions, username
        where pm_object_kind = :uniqtab_username and
              pm_subject_kind = :uniqtab_rolename and
              pm_subject_id = :object_id and
              pm_object_id = un_id and
              pm_is_delete = 0 and
              (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
              (coalesce(pm_date_end, '31.12.2999') >= current_date)
        into :rn_id, :rn_name, :rn_memo
    do
    begin
      if (exists(select *
                 from hp_un_is_active(:rn_id)
                 where un_is_active = 1)) then
      begin
        user_count = 1;
        select count(*)
        from permissions
        where pm_subject_kind = :uniqtab_rolename and
              pm_object_kind = :uniqtab_username and
              pm_object_id = :rn_id and
              pm_is_delete = 0 and
              (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
              (coalesce(pm_date_end, '31.12.2999') >= current_date)
        into :user_count;
        if (user_count is null) then
          user_count = 0;
        suspend;
      end
    end
  end
end
