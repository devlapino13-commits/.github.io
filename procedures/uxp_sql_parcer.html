CREATE PROCEDURE UXP$SQL_PARCER(
    SQL_TEXT D_STR_8100)
RETURNS (
    RES_KIND D_STR_CODE,
    RES_SOURCE_CODE D_STR_CODE,
    RES_SQL_SELECT D_STR_8100,
    RES_SQL_FROM D_STR_LONG,
    RES_SQL_WHERE D_STR_LONG,
    RES_SQL_ORDERBY D_STR_LONG)
AS
declare variable has_select d_bool;
declare variable one_source d_bool;
declare variable pos d_integer;
declare variable pos_start d_integer;
declare variable pos_end d_integer;
declare variable source_count d_integer;
declare variable orig_text d_str_8100;
declare variable spec_symbols d_str_code;
declare variable sym d_str_sym;
begin
  -- RES_KIND - table, procedure, sql

  res_kind = '';
  res_source_code = '';
  res_sql_select = '';
  res_sql_where = '';
  res_sql_orderby = '';
  orig_text = sql_text;

  while (position(ascii_char(13) || ascii_char(10), sql_text) > 0) do
    sql_text = replace(sql_text, ascii_char(13) || ascii_char(10), ' ');

  while (position(ascii_char(13), sql_text) > 0) do
    sql_text = replace(sql_text, ascii_char(13), ' ');

  while (position(ascii_char(10), sql_text) > 0) do
    sql_text = replace(sql_text, ascii_char(10), ' ');

  spec_symbols = ',:(';
  pos = 0;
  while (pos < char_length(spec_symbols)) do
  begin
    pos = pos + 1;
    sym = substring(spec_symbols from pos for 1);

    while (position(sym || ' ', sql_text) > 0) do
      sql_text = replace(sql_text, sym || ' ', sym);

    while (position(' ' || sym, sql_text) > 0) do
      sql_text = replace(sql_text, ' ' || sym, sym);
  end

  while (position('  ', sql_text) > 0) do
    sql_text = replace(sql_text, '  ', ' ');

  sql_text = trim(sql_text);

  one_source = 1;
  has_select = 0;
  pos_start = 0;
  pos_end = 0;

  if (left(sql_text, 7) <> 'select ') then
  begin
    if (position(' ', sql_text) > 0 or position('(', sql_text) > 0) then
      sql_text = 'select * from ' || sql_text;
  end

  if (left(sql_text, 7) = 'select ') then
  begin
    has_select = 1;

    if (position(' join ', sql_text) > 0) then
      one_source = 0;
    else
    begin
      pos = position(' from ', sql_text);
      if (pos > 0) then
        pos_start = pos + 6;
      else
        exception e_message 'Найден SELECT без FROM';
    end
  end else
  begin
    pos_start = 1;
    pos_end = 10000;
  end

  pos = position(' WHERE ', sql_text);
  if (pos = 0) then
    pos = position(' ORDER BY ', sql_text);

  if (pos > 0) then
    pos_end = pos - 1;
  else
    pos_end = 10000;

  if (has_select = 0 and one_source = 0) then
    exception e_message 'Найден JOIN без Select';

  if (has_select = 1 and one_source = 0) then
  begin
    res_kind = 'sql';
    res_source_code = '';
    res_sql_select = orig_text;
    res_sql_where = '';
    res_sql_orderby = '';
    suspend;

    exit;
  end

  res_sql_select = orig_text;
  if (has_select = 0) then
    res_sql_select = 'select * from ' || orig_text;

  res_sql_from = substring(sql_text from pos_start for pos_end - pos_start + 1);

  select max(source_code), count(*)
  from uxp$sql_source_parcer(:res_sql_from)
  into res_source_code, source_count;

  if (source_count > 1) then
  begin
    res_kind = 'sql';
    res_source_code = '';
    one_source = 0;
  end else
  begin
    res_kind = 'procedure';
    if (exists(select * from rdb$relations r
    where r.rdb$relation_name = upper(:res_source_code))) then
      res_kind = 'table';

    pos_start = position(' WHERE ', sql_text);
    if (pos_start > 0) then
    begin
      pos_end = position(' ORDER BY ', sql_text);
      if (pos_end = 0) then
        pos_end = position(' ROWS ', sql_text);

      if (pos_end = 0) then
        pos_end = char_length(sql_text);

      pos_start = pos_start + 7;
      res_sql_where = substring(sql_text from pos_start for (pos_end - pos_start + 1));
    end

    pos_start = position(' ORDER BY ', sql_text);
    if (pos_start > 0) then
    begin
      pos_end = position(' ROWS ', sql_text);

      if (pos_end = 0) then
        pos_end = char_length(sql_text);

      pos_start = pos_start + 10;
      res_sql_orderby = substring(sql_text from pos_start for (pos_end - pos_start + 1));
    end
  end

  suspend;
end
