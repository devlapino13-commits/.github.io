CREATE PROCEDURE HP_PHONECALLS_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    FIND_TEXT D_STR_100,
    FILE_STATE D_SMALLINT)
RETURNS (
    PNC_ID D_ID,
    PNC_SRC_PHONE_ID D_INTEGER,
    PNC_DST_PHONE_ID D_INTEGER,
    PNC_SRC_C_ID D_ID,
    PNC_DST_C_ID D_ID,
    PNC_DIRECTION D_SMALLINT,
    PNC_OBJECT_KIND D_INTEGER,
    PNC_OBJECT_ID D_ID,
    PNC_PRV_PHONE_ID D_INTEGER,
    PNC_CALL_STATUS_ID D_INTEGER,
    PNC_START D_DATETIME,
    PNC_FINISH D_DATETIME,
    PNC_DURATION D_INTEGER,
    PNC_FILENAME D_STR_255,
    PNC_CALL_RESULT_ID D_INTEGER,
    PNC_IS_DELINQ D_BOOL,
    PNC_IS_CONTPERSON D_BOOL,
    PNC_SOURCE D_INTEGER,
    RES_SRC_PHONE_NO D_STR_20,
    RES_DST_PHONE_NO D_STR_20,
    RES_SRC_C_FIO D_STR_100,
    RES_DST_C_FIO D_STR_100,
    RES_DIRECTION D_STR_CODE,
    RES_ADD_INFO D_STR_255,
    RES_PRV_PHONE_NO D_STR_20,
    RES_CALL_STATUS_STR D_STR_100,
    RES_CALL_RESULT_STR D_STR_100,
    RES_FILENAME D_STR_LONG,
    RES_FILE_EXITS D_BOOL,
    RES_CONT_KIND D_STR_CODE)
AS
declare variable need_suspend d_bool;
begin
  if (trim(coalesce(find_text, '')) = '') then
    if (datediff(day, rep_date_begin, rep_date_end) >=7) then
      exception e_message 'Укажите доп.фильтр или период меньше недели';

  if (file_state is null) then
    file_state = 0;

  -- file_state:
  --  0 - Все
  --  1 - Только с файлами
  -- -1 - Без файлов

  for select pnc_id,
             pnc_src_phone_id,
             pnc_dst_phone_id,
             pnc_src_c_id,
             pnc_dst_c_id,
             pnc_direction,
             pnc_object_kind,
             pnc_object_id,
             pnc_prv_phone_id,
             pnc_call_status_id,
             pnc_start,
             pnc_finish,
             pnc_duration,
             pnc_filename,
             pnc_call_result_id,
             pnc_is_delinq,
             pnc_is_contperson,
             pnc_source
      from phonecalls
      where pnc_start between :rep_date_begin and :rep_date_end + 1 and
            (:file_state = 0 or
            (:file_state = 1 and coalesce(pnc_filename, '') <> '') or
            (:file_state = -1 and coalesce(pnc_filename, '') = '')) and
            pnc_source = 1
      into :pnc_id,
           :pnc_src_phone_id,
           :pnc_dst_phone_id,
           :pnc_src_c_id,
           :pnc_dst_c_id,
           :pnc_direction,
           :pnc_object_kind,
           :pnc_object_id,
           :pnc_prv_phone_id,
           :pnc_call_status_id,
           :pnc_start,
           :pnc_finish,
           :pnc_duration,
           :pnc_filename,
           :pnc_call_result_id,
           :pnc_is_delinq,
           :pnc_is_contperson,
           :pnc_source
  do
  begin
    res_src_phone_no = f_cont_code(pnc_src_phone_id);
    res_dst_phone_no = f_cont_code(pnc_dst_phone_id);

    res_src_c_fio = '';
    select c_name_fio
    from clients
    where c_id = :pnc_src_c_id
    into res_src_c_fio;

    res_dst_c_fio = '';
    select c_name_fio
    from clients
    where c_id = :pnc_dst_c_id
    into res_dst_c_fio;

    res_direction = 'Исх';
    if (pnc_direction = -1) then
      res_direction = 'Вх';

    res_add_info = '';
    if (pnc_object_id is not null and pnc_object_kind is not null) then
      res_add_info = f_objname(pnc_object_kind, pnc_object_id);

    res_prv_phone_no = '';

    if (pnc_source = 0) then
    begin
        if (pnc_prv_phone_id is not null) then
          res_prv_phone_no = left(f_cont_code(pnc_prv_phone_id), 20);
    end

    res_call_status_str = f_enum_name('phone_call_status', pnc_call_status_id);
    res_call_result_str = f_enum_name('phone_call_result', pnc_call_result_id);

    if (pnc_object_kind = 9) then
    begin
      res_call_result_str = 'Разговор не состоялся';

      select iif(rd_tag > 0, 'Разговор состоялся', 'Разговор не состоялся')
      from pantera_clients
      inner join refdata on rd_id = pcl_rd_res
      where pcl_from_pnc_id = :pnc_id
      union all
      select iif(rd_tag > 0, 'Разговор состоялся', 'Разговор не состоялся')
      from pantera_contacts
      inner join refdata on rd_id = pcon_rd_res
      where pcon_from_pnc_id = :pnc_id rows 1
      into res_call_result_str;    
    end

    res_cont_kind = 'Основной';
    if (pnc_is_contperson = 1) then
      res_cont_kind = 'Допол-ый';

    res_file_exits = 0;
    res_filename = '';

    if (coalesce(pnc_filename, '') <> '') then
    begin
      res_file_exits = 1;
      res_filename = 'https://[basic_auth_ru_voice]record.molbulak.ru/' || pnc_filename;
    end

    need_suspend = 1;
    if (coalesce(find_text, '') <> '') then
    begin
      need_suspend = 0;

      if (find_text = cast(pnc_id as d_str_20) or
          res_src_phone_no containing find_text or
          res_dst_phone_no containing find_text or
          res_src_c_fio containing find_text or
          res_dst_c_fio containing find_text or
          res_add_info containing find_text or
          res_prv_phone_no containing find_text) then
      need_suspend = 1;
    end

    if (need_suspend = 1) then
      suspend;
  end
end
