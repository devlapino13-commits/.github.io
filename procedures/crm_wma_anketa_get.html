CREATE PROCEDURE CRM_WMA_ANKETA_GET(
    WMA_ID INTEGER)
RETURNS (
    STATUS STR_100,
    BLOCKED_DATETIME D_DATETIME,
    CREATE_DATETIME D_DATETIME,
    STATUS_DATETIME D_DATETIME,
    LANGUAGE STR_255,
    FIO STR_100,
    BIRTH_DATE D_DATE,
    PHONE_NO STR_100,
    CITIZENSHIP STR_255,
    CITY ID_TYPE,
    OFFICE D_INTEGER,
    SVZ ID_TYPE,
    REFUSE STR_255,
    LAST_CHANGES_DATETIME D_DATETIME,
    DOCLIST STR_255,
    WORKSTATUS STR_255,
    LOAN_PURPOSE STR_255,
    LOAN_PRODUCT STR_255,
    LOAN_PERIOD D_SMALLINT,
    LOAN_AMOUNT D_INTEGER,
    ANKETA_MEMO D_STR_8100,
    NAME1 STR_100,
    NAME2 STR_100,
    NAME3 STR_100)
AS
declare variable ema_status d_smallint;
declare variable svz_refuse str_255;
declare variable ema_id id_type;
declare variable ma_id d_integer;
declare variable ma_refuse_reason str_255;
declare variable ma_date_oper d_date;
declare variable ma_krl_id d_integer;
declare variable k_id id_type;
declare variable k_refuse_reason str_255;
declare variable k_krl_id d_integer;
declare variable ks_state d_smallint;
declare variable k_disb_date d_date;
declare variable wma_refuse_reason str_255;
declare variable wma_status d_integer;
declare variable tkt_id id_type;
declare variable ema_svz id_type;
declare variable ma_svz id_type;
declare variable k_svz id_type;
declare variable k_status d_smallint;
declare variable ma_is_active d_smallint;
declare variable c_id id_type;
declare variable wma_date d_date;
declare variable wma_status_datetime d_datetime;
declare variable ema_datetime d_datetime;
declare variable ema_status_datetime d_datetime;
declare variable ma_status_datetime d_datetime;
declare variable kontract_date d_date;
declare variable status_disb_datetime d_datetime;
declare variable ema_office d_integer;
declare variable ma_office d_integer;
declare variable k_office d_integer;
declare variable k_from_ma_app_id d_integer;
declare variable k_refuse_datetime d_datetime;
begin
  for select iif(wmat_name = '', '[Анкета в процессе заполнения]', wmat_name),
             rd_c.rd_code as citizenship, rd_lang.rd_code as language, wma_phone_no, wmat_city_id,
             iif(wma_status >= 0, wmt.wmat_anketa_memo, wmc.wmc_memo), rd_r.rd_code as refuse, wma_status,
             rd_purp.rd_code as loan_purpose, wma_from_tkt_id, wmat_birthday, wmat_doclist, rd_w.rd_code as workstatus,
             rd_prod.rd_code as loan_product, wmat_loan_period, wmat_loan_amount, wma.wma_date, wmat_datetime
      from cc_website_miniapp wma
      inner join cc_website_miniapp_tmp wmt on wmt.wmat_id = wma.wma_last_wmat_id
      left join cc_website_miniapp_close wmc on wmc.wmc_from_wma_id = wma.wma_id
      left join refdata rd_c on wmt.wmat_citizenship = rd_c.rd_id
      left join refdata rd_lang on wmat_language = rd_lang.rd_id
      left join refdata rd_r on wmc.wmc_refuse_id = rd_r.rd_id
      left join refdata rd_purp on wmt.wmat_loan_purpose = rd_purp.rd_id
      left join refdata rd_w on wmat_work_status = rd_w.rd_id
      left join refdata rd_prod on wmat_loan_product = rd_prod.rd_id
      where wma_id = :wma_id and
            wma.wma_status between -1 and 2
      into :fio, :citizenship, :language, :phone_no, :city, :anketa_memo, :wma_refuse_reason, :wma_status,
           :loan_purpose, :tkt_id, :birth_date, :doclist, :workstatus, :loan_product, :loan_period, :loan_amount,
           :wma_date, :wma_status_datetime
  do
  begin
    ema_status = null;
    svz_refuse = null;
    ema_id = null;
    status = '';
    refuse = '';
    svz = null;

    if (wma_status_datetime is not null) then
    begin
      status_datetime = wma_status_datetime;
      if (wma_id is not null) then
      begin
        if (wma_status = 0) then
          status = 'Новая';
        else
        if (wma_status = 1) then
          status = 'Перезвонить';
        else
        if (wma_status = 2) then
          status = 'Заявка Передана в офис';
        else
        if (wma_status = -1) then
        begin
          if (refuse is null) then
            status = 'Закрытa';
          else
          begin
            status = 'Отказ';
            refuse = wma_refuse_reason;
          end
        end
      end
    end

    select ema_lo_id, ema_status, rd_code, ema_id, ema_from_c_id, ema_block_datetime, ema_status_datetime,
           ema_office_id, ema_datetime, ema_amount, ema_period
    from extminiapp
    left join refdata on rd_id = ema_result_rd_id
    where ema_from_tkt_id = :tkt_id
    into :ema_svz, :ema_status, :svz_refuse, :ema_id, :c_id, :blocked_datetime, :ema_status_datetime, :ema_office,
         :ema_datetime, :loan_amount, :loan_period;

    if (ema_datetime is not null) then
    begin
      status_datetime = ema_datetime;

      if (ema_status_datetime is not null) then
      begin
        status_datetime = ema_status_datetime;
      end

      if (ema_status = 0) then
      begin
        if (ema_svz is null) then
          status = 'Не заблокирована ВИПС';
        else
          status = 'Заблокирована ВИПС';
      end
      else
      if (ema_status = 1) then
      begin
        status = 'Заблокирована ВИПС';
      end
      else
      if (ema_status = -1) then
      begin
        refuse = svz_refuse;
        status = 'Отказ';
      end
      if (ema_office is not null) then
        office = ema_office;
      if (ema_svz is not null) then
        svz = ema_svz;
    end

    ma_id = null;
    ma_refuse_reason = null;
    ma_date_oper = null;
    ma_krl_id = null;
    k_id = null;
    create_datetime = null;

    for select ma_id, kr_id, ma_date_oper, ma_date_app, un_from_c_id, ma_is_active, ma_date_save, ma_office_id
        from miniapp
        inner join miniappclient on mac_from_ma_id = ma_id
        inner join username on un_id = ma_from_un_id
        inner join knaktlist on kal_from_kf_id = ma_from_kf_id
        inner join clients on c_id = kal_new_kf_officer
        left join kontreslist on krl_ma_id = ma_id
        left join kontres on kr_id = krl_from_kr_id
        where mac_from_c_id = :c_id and
              ma_date_app >= :wma_date and
              ma_date_end = '31.12.9999'
        order by ma_is_active, ma_date_oper
        into ma_id, ma_refuse_reason, ma_date_oper, kontract_date, ma_svz, ma_is_active, ma_status_datetime, ma_office
    do
    begin
      if (create_datetime is null) then
      begin
        select ma_date_save
        from miniapp
        where ma_id = :ma_id and
              ma_number = 0
        into create_datetime;
      end

      if (ma_refuse_reason is null and
          ma_is_active < 1) then
      begin
        select k_id, kr_text, krl_datetime
        from kontract
        left join kontreslist on krl_from_k_id = k_id
        left join kontres on krl_from_kr_id = kr_id
        where k_from_ma_app_id = :ma_id rows 1
        into k_id, ma_refuse_reason, ma_status_datetime;

        if (k_id is null) then
        begin
          select k_id, kr_text, krl_datetime
          from knclient
          inner join kontract on k_id = kc_from_k_id
          left join kontreslist on krl_from_k_id = k_id
          left join kontres on krl_from_kr_id = kr_id
          where kc_from_c_id = :c_id and
                k_kontract_date >= :kontract_date and
                krl_date_old >= :kontract_date rows 1
          into k_id, ma_refuse_reason, ma_status_datetime;
        end
      end
      if (ma_refuse_reason is not null) then
      begin
        status = 'Отказ';
        refuse = ma_refuse_reason;
        status_datetime = ma_status_datetime;
      end
      if (ma_svz is not null) then
        svz = ma_svz;
      if (ma_office is not null) then
        office = ma_office;
    end

    if (ma_is_active = 1) then
    begin
      select k_id, k_status, ol_datetime, kr_id, krl_id, krl_datetime, ks_region_id, ks_kc_id
      from kontract
      inner join knclient on kc_from_k_id = k_id
      left join kontreslist on krl_from_k_id = k_id
      left join kontres on kr_id = krl_from_kr_id
      left join operlist on ol_id = k_fact_disb_ol_id and
            is_delete = 0
      left join kontstate on ks_from_k_id = k_id
      where k_from_ma_app_id = :ma_id and
            kc_amount_approved > 0 and
            ks_ol_is_delete = 0
      order by kc_id rows 1
      into k_id, k_status, status_disb_datetime, k_refuse_reason, k_krl_id, k_refuse_datetime, k_office, k_svz;

      if (k_status = 1 and
          status_disb_datetime is not null) then
      begin
        status = 'Выдан';
        status_datetime = status_disb_datetime;
      end
      else
      if (k_krl_id is not null) then
      begin
        status = 'Отказ';
        refuse = k_refuse_reason;
        status_datetime = k_refuse_datetime;
      end
      else
      begin
        status = 'В процессе';
        status_datetime = ma_status_datetime;
      end
      if (k_svz is not null) then
        svz = k_svz;
      if (k_office is not null) then
        office = k_office;
    end
  end

  if (c_id is not null) then
  begin
    select coalesce(lg.rd_code, :language), coalesce(c_name_fio, :fio), coalesce(c_birth_date, :birth_date),
           coalesce(last_cell_main, :phone_no), coalesce(cz.rd_code, :citizenship), fl_name1, fl_name2, fl_name3
    from clients
    left join refdata cz on cz.rd_id = c_citizen
    left join clanguage on cl_from_c_id = c_id
    left join refdata lg on lg.rd_id = cl_lang_no
    left join continfo on cci_object_id = c_id and
          cci_object_kind = 0
    left join fiolist on fl_id = c_from_fl_id
    where c_id = :c_id rows 1
    into :language, :fio, :birth_date, :phone_no, :citizenship, name1, name2, name3;
  end

  last_changes_datetime = status_datetime;

  select distinct pl.pl_id
  from places pl
  inner join addrlist adl on adl.adl_from_pl_id = pl.pl_id
  inner join region reg on reg.reg_address_fact = adl.adl_id
  where reg.reg_date_close is null and
        reg.is_delete = 0 and
        reg.reg_status = 3 and
        reg.reg_unicode = :office
  order by pl.pl_name rows 1
  into :city;

  suspend;
end
