CREATE PROCEDURE AAREP_ACCOUNT_BANK_SALDO_CL_IPF(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    LINE_NAME D_STR_255,
    BG_ID D_STR_20,
    ACCOUNT D_STR_20,
    CURRENCY VARCHAR(3),
    CR_RATE D_CURRENCY,
    CLASSIC_SALDO_HQ_417 D_CURRENCY,
    CLASSIC_SALDO_HQ_OTHER D_CURRENCY,
    CLASSIC_SALDO_HQ_840 D_CURRENCY,
    IPF_SALDO_HQ_417 D_CURRENCY,
    IPF_SALDO_HQ_OTHER D_CURRENCY,
    IPF_SALDO_HQ_840 D_CURRENCY,
    REG_CLASSIC_SALDO_HQ_417 D_CURRENCY,
    REG_IPF_SALDO_HQ_417 D_CURRENCY,
    CR_RATE1 D_CURRENCY)
AS
declare variable summa_ekv d_currency;
declare variable obs_saldo_val d_currency;
declare variable al_memo d_str_255;
declare variable al_memo_old d_str_255;
declare variable al_from_reg_id d_integer;
declare variable reg_id_old d_integer;
declare variable region_id id_type;
declare variable reg_id id_type;
declare variable classic_saldo_hq_417_tmp d_currency;
declare variable classic_saldo_hq_other_tmp d_currency;
declare variable classic_saldo_hq_840_tmp d_currency;
declare variable ipf_saldo_hq_417_tmp d_currency;
declare variable ipf_saldo_hq_other_tmp d_currency;
declare variable ipf_saldo_hq_840_tmp d_currency;
declare variable reg_classic_saldo_hq_417_tmp d_currency;
declare variable reg_ipf_saldo_hq_417_tmp d_currency;
begin
    select op.cr_rate
    from opercurr op
    where op.cr_from_cu_id = '840' and
        :rep_date_end between op.cr_date and op.cr_date_end
    into cr_rate;
  -- вычисляет остаток по денежных средств в разбивке по округам.
  -- работает только для кр
  al_memo_old = '';
  classic_saldo_hq_417_tmp = 0;
  classic_saldo_hq_other_tmp = 0;
  classic_saldo_hq_840_tmp = 0;
  ipf_saldo_hq_417_tmp = 0;
  ipf_saldo_hq_other_tmp = 0;
  ipf_saldo_hq_840_tmp = 0;
  reg_classic_saldo_hq_417_tmp = 0;
  reg_ipf_saldo_hq_417_tmp = 0;

  for select al_id, al_from_bg_id, al_currency, al_from_reg_id, coalesce(al_memo, ''), op.cr_rate
      from region_list(:rep_region_id, 1, 3, 1)
      inner join acc_list al on al_from_reg_id = id_list and
            al_from_bg_id in ('10150', '10101') and -- 10150 банк классика, 10101- банк ипф
            al.al_id <> 090141710150000101 --  BOARD-5892 задача от финансистов
      left join opercurr op on op.cr_from_cu_id = al_currency and
            :rep_date_end between op.cr_date and op.cr_date_end
      order by al_memo, al_from_bg_id
      into :account, :bg_id, :currency, :reg_id, :al_memo, :cr_rate1
  do
  begin

    if (al_memo_old <> al_memo) then
    begin
      classic_saldo_hq_417 = classic_saldo_hq_417_tmp;
      classic_saldo_hq_other = classic_saldo_hq_other_tmp;
      classic_saldo_hq_840 = classic_saldo_hq_840_tmp;
      ipf_saldo_hq_417 = ipf_saldo_hq_417_tmp;
      ipf_saldo_hq_other = ipf_saldo_hq_other_tmp;
      ipf_saldo_hq_840 = ipf_saldo_hq_840_tmp;
      reg_classic_saldo_hq_417 = reg_classic_saldo_hq_417_tmp;
      reg_ipf_saldo_hq_417 = reg_ipf_saldo_hq_417_tmp;


      classic_saldo_hq_417_tmp = 0;
      classic_saldo_hq_other_tmp = 0;
      classic_saldo_hq_840_tmp = 0;
      ipf_saldo_hq_417_tmp = 0;
      ipf_saldo_hq_other_tmp = 0;
      ipf_saldo_hq_840_tmp = 0;
      reg_classic_saldo_hq_417_tmp = 0;
      reg_ipf_saldo_hq_417_tmp = 0;


      al_memo_old = al_memo;
      if(classic_saldo_hq_417 <> 0 or
          classic_saldo_hq_other <> 0 or
          classic_saldo_hq_840 <> 0 or
          ipf_saldo_hq_417 <> 0 or
          ipf_saldo_hq_other <> 0 or
          ipf_saldo_hq_840 <> 0 or
          reg_classic_saldo_hq_417 <> 0 or
          reg_ipf_saldo_hq_417 <> 0) then
      suspend;
    end

    summa_ekv = 0;
    obs_saldo_val = 0;
    line_name = '';

    summa_ekv = f_al_bal(account, rep_date_end);
    obs_saldo_val = f_al_bal(account, rep_date_end, 'val');

    summa_ekv = coalesce(summa_ekv, 0);
    obs_saldo_val = coalesce(obs_saldo_val, 0);
    cr_rate1 = coalesce(cr_rate1, 0);


    select parent
    from region reg
    where reg.reg_unicode = :reg_id
    into al_from_reg_id;

    if (:bg_id = '10150') then
    begin
      if (currency = '417') then
      begin
        if (reg_id = 1090100) then
          classic_saldo_hq_417_tmp = classic_saldo_hq_417_tmp + summa_ekv;
        else
        if (reg_id = 1060100 or al_from_reg_id in (1010000, 1030000, 1040000, 1050000, 1060000, 1070000, 1090002, 1090004, 1091000, 1110000,
                               1270000)) then
          reg_classic_saldo_hq_417_tmp = reg_classic_saldo_hq_417_tmp + summa_ekv;
      end

      if (currency = '840') then
      begin
        if (reg_id = 1090100) then
          classic_saldo_hq_840_tmp = classic_saldo_hq_840_tmp + obs_saldo_val;
      end

      if (currency not in ('417', '840')) then
      begin
        if (reg_id = 1090100) then
          classic_saldo_hq_other_tmp = classic_saldo_hq_other_tmp + obs_saldo_val * cr_rate1;
      end
    end

    if (:bg_id = '10101') then
    begin
      if (currency = '417') then
      begin
        if (reg_id = 1090100) then
          ipf_saldo_hq_417_tmp = ipf_saldo_hq_417_tmp + summa_ekv;
        else
        if (reg_id = 1060100 or al_from_reg_id in (1010000, 1030000, 1040000, 1050000, 1060000, 1070000, 1090002, 1090004, 1091000, 1110000,
                               1270000)) then
          reg_ipf_saldo_hq_417_tmp = reg_ipf_saldo_hq_417_tmp + summa_ekv;
      end

      if (currency = '840') then
      begin
        if (reg_id = 1090100) then
          ipf_saldo_hq_840_tmp = ipf_saldo_hq_840_tmp + obs_saldo_val;
      end

      if (currency not in ('417', '840')) then
      begin
        if (reg_id = 1090100) then
          ipf_saldo_hq_other_tmp = ipf_saldo_hq_other_tmp + obs_saldo_val * cr_rate1;
      end
    end
    line_name = al_memo;

    if (line_name = '') then
      line_name = 'X Банк не указан';
  end

      classic_saldo_hq_417 = classic_saldo_hq_417_tmp;
      classic_saldo_hq_other = classic_saldo_hq_other_tmp;
      classic_saldo_hq_840 = classic_saldo_hq_840_tmp;
      ipf_saldo_hq_417 = ipf_saldo_hq_417_tmp;
      ipf_saldo_hq_other = ipf_saldo_hq_other_tmp;
      ipf_saldo_hq_840 = ipf_saldo_hq_840_tmp;
      reg_classic_saldo_hq_417 = reg_classic_saldo_hq_417_tmp;
      reg_ipf_saldo_hq_417 = reg_ipf_saldo_hq_417_tmp;

      if(classic_saldo_hq_417 <> 0 or
          classic_saldo_hq_other <> 0 or
          classic_saldo_hq_840 <> 0 or
          ipf_saldo_hq_417 <> 0 or
          ipf_saldo_hq_other <> 0 or
          ipf_saldo_hq_840 <> 0 or
          reg_classic_saldo_hq_417 <> 0 or
          reg_ipf_saldo_hq_417 <> 0) then
      suspend;
end
