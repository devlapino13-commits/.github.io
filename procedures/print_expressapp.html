CREATE PROCEDURE PRINT_EXPRESSAPP(
    EXA_ID D_INTEGER)
RETURNS (
    CLI_FIO STR_100,
    EXA_AMOUNT DECIMAL(18,4),
    EXA_PERIOD INTEGER,
    EXA_PERCENT DECIMAL(18,7),
    EXA_DATE_END DATE,
    K_PERCENT_EFFECT PERC_TYPE,
    K_KOMMIS_PERC PERC_TYPE,
    MONTHPAY_WITH_NSP CURR_TYPE,
    SUM_PERC CURR_TYPE,
    SUM_TOTAL CURR_TYPE,
    DATE_PREV D_DATE,
    OFFICE_ADDR D_STR_255)
AS
declare variable EXA_DATE date; /* EXA_DATE_END */
declare variable EXA_FROM_C_ID bigint;
declare variable EXA_FROM_MA_ID integer;
declare variable EXA_FROM_K_ID bigint;
declare variable EXA_DISB_DATE date;
declare variable EXA_FROM_PR_ID integer;
declare variable EXA_REPAY_DAY integer;
declare variable EXA_LO_ID bigint;
declare variable EXA_FROM_REG_ID integer;
declare variable SH_DATE_PREV D_DATE;
declare variable SH_DAYS D_SMALLINT;
declare variable SH_ALL CURR_TYPE;
declare variable SH_ID D_INTEGER;
declare variable SH_DATE D_DATE;
declare variable SH_PERC CURR_TYPE;
declare variable SH_TOTAL CURR_TYPE;
declare variable SH_DISB_AMOUNT CURR_TYPE;
begin
  for
  select exa_date, exa_from_c_id, exa_from_ma_id, exa_from_k_id, exa_disb_date, exa_from_pr_id, exa_amount, exa_period,
         exa_percent, exa_repay_day, exa_lo_id, exa_from_reg_id
  from expressapp
  where exa_id = :exa_id
  into exa_date, exa_from_c_id, exa_from_ma_id, exa_from_k_id, exa_disb_date, exa_from_pr_id, exa_amount, exa_period,
       exa_percent, exa_repay_day, exa_lo_id, exa_from_reg_id
   do
   begin

     select c_name_fio from clients
     where c_id = :exa_from_c_id
     into cli_fio;

     exa_date_end = exa_date + 3;

     sum_perc = 0;
     sum_total = 0;
     monthpay_with_nsp = 0;
     k_percent_effect = 0;
     if (f_get_db_id() in (39801, 41701)) then
     begin
       delete from temp_sh
       where 1=1;

--       for select sh_id, sh_date, sh_perc, sh_total, sh_disb_amount
--       from hp_make_schedule_by_param(:exa_disb_date, :exa_from_pr_id, :exa_amount, :exa_percent, 0, :exa_period, :exa_repay_day)
--       into :sh_id, :sh_date, :sh_perc, :sh_total, sh_disb_amount
       for select  sh_id, sh_date, sh_perc, sh_total, sh_disb_amount
       from print_exa_schedule(:exa_id)
       into :sh_id, :sh_date, :sh_perc, :sh_total, sh_disb_amount
       do
       begin
         sum_perc = sum_perc + sh_perc;
         sum_total = sum_total + sh_total;
         if (monthpay_with_nsp = 0) then monthpay_with_nsp = sh_total;
         if (sh_date_prev is null) then
            sh_days = 0;
          else
            sh_days = sh_date - sh_date_prev;
         sh_total = sh_total - sh_disb_amount;

--         insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
--         values (:sh_id, :sh_date, :sh_days, :sh_total);
--         sh_date_prev = sh_date;

         insert into temp_sh (sh_rec_id, sh_date, sh_repay_mode, sh_zad_os)
         values (:sh_id, :sh_date, :sh_days, :sh_total);
         sh_date_prev = sh_date;
       end

       delete from temp_schedule
       where 1=1;

       insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
       select sh_rec_id, sh_date, sh_repay_mode, sh_zad_os
       from temp_sh
       where 1=1;
       delete from temp_sh
       where 1=1;

       select calc_eff
       from hp_effect_rate_kg(:exa_from_k_id)
       into k_percent_effect ;

       delete from temp_schedule
       where 1=1;

       select max(prl_admin_perc_max) from prodlogic
       where prl_from_pr_id = :exa_from_pr_id
       and :exa_disb_date between prl_date_begin and prl_date_end
       into k_kommis_perc;
       k_kommis_perc = coalesce(k_kommis_perc,0);

       date_prev = current_date;

       -- адрес офиса
       select adl_address_str from region, addrlist
       where reg_unicode = :exa_from_reg_id
       and reg_address_fact = adl_id
       into office_addr;

     end 

     suspend;
   end
end
