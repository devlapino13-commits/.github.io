CREATE PROCEDURE CM_KG_0042_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable pr_code varchar(100);
declare variable k_percent perc_type;
declare variable k_kommis_perc perc_type;
declare variable k_circle integer;
declare variable kib integer;
begin
  select pr_code, k.k_percent, k.k_kommis_perc, K_DISB_DATE, k_circle
  from kontract k, product pr
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and K_DISB_DATE between '01.08.16' and '31.08.16'
  and pr_code not containing '.PAR.'
  into pr_code, k_percent, k_kommis_perc, KF_DISB_DATE, k_circle;

  if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
   exit;

  if (pr_code containing '.A.') then
  begin      
    if (pr_code containing '.GR.') then
    begin
      kib = 1; 
      k_circle=1;    
      select iif(sum(iif(kc.kc_circle <=1, 1, 0)) <= sum(iif(kc.kc_circle > 1, 1, 0)), 2, 1),
      iif(sum(iif(kc.kc_circle <=1, 1, 0)) <=
        sum(iif(kc.kc_circle <=1
        and coalesce((select 1 from kibminiapp kaa where kaa.kbm_from_c_id = kc_from_c_id rows 1),0)=1, 1, 0)), 2, 1)
      from knclient kc
      where kc_from_k_id = :k_id
      and kc.kc_amount_approved > 0
      into k_circle, kib;
      --k_circle = iif(kib>k_circle, kib, k_circle);
      --Исключения
      if (k_id in (109010023923000101, 109010023908200101, 109010023948500101, 109010024031000101, 109010023961200101, 109010024143700101)) then
       k_circle = 2;
      --Повторникам
      if (k_circle > 1) then
      begin
        if (k_percent <> 28.9) then
        begin
          res_error_text = 'Повторным группам процент по акции должен быть 28,9%';
          res_error_is_critical = 1;
          suspend;
        end
        
        if (k_kommis_perc <> 2.1) then
        begin
          res_error_text = 'Повторным группам доп.процент по акции должен быть 2,1%';
          res_error_is_critical = 1;
          suspend;
        end  
      end  
     
     --Новым
      if (k_circle <= 1) then
      begin
        if (k_percent <> 37.9) then
        begin
          res_error_text = 'Новой группе процент по акции должен быть 37,9%';
          res_error_is_critical = 1;
          suspend;
        end
        if (k_kommis_perc <> 2.8) then
        begin
          res_error_text = 'Новой группе доп.процент по акции должен быть 2,8%';
          res_error_is_critical = 1;
          suspend;
        end
      end
    end
     
    if (pr_code containing '.IND.') then
    begin
        select c_name_fio, kc_circle
        from knclient kc, clients c
        where kc_from_k_id = :k_id
        and kc.kc_amount_approved > 0
        and kc_from_c_id = c_id
        into res_info_text, k_circle;
        k_circle = coalesce(k_circle,1);
      if (k_circle > 1) then
      begin
        if (k_percent <> 31.8) then
        begin
          res_error_text = 'Повторный клиент: процент по акции должен быть 31.8%';
          res_error_is_critical = 1;
          suspend;
        end
        
        if (k_kommis_perc <> 3.2) then
        begin
          res_error_text = 'Повторный клиент: доп.процент по акции должен быть 3.2%';
          res_error_is_critical = 1;
          suspend;
        end  
      end  
     
      if (k_circle <= 1) then
      begin
        if (k_percent <> 39.9) then
        begin
          res_error_text = 'Новый клиент: процент по акции должен быть 39,9%';
          res_error_is_critical = 1;
          suspend;
        end
        if (k_kommis_perc <> 3.9) then
        begin
          res_error_text = 'Новый клиент: доп.процент по акции должен быть 3,9%';
          res_error_is_critical = 1;
          suspend;
        end
      end
    end
  end else
if (KF_DISB_DATE is not null and pr_code not containing '.PAR.') then
  begin 
    res_error_text = 'Действует АКЦИЯ  "Август-сезон низких процентов от МолБулак приказ №75-07-16 о/д", необходимо выбрать правильный продукт!';
    res_error_is_critical = iif(kf_disb_date > '08.08.16', 0, 1);
    suspend;
  end
end
