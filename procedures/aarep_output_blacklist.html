CREATE PROCEDURE AAREP_OUTPUT_BLACKLIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    MAL_REQ_COMMENT VARCHAR(512),
    MAL_APPR_COMMENT VARCHAR(256),
    MAL_APPR_DATE DATE,
    REQ_USER VARCHAR(200),
    APPR_USER VARCHAR(200),
    M_CLIENT_FIO VARCHAR(100),
    KONTNO_DISBDATE VARCHAR(50),
    KONT_STATUS VARCHAR(20),
    DELINQ_DAY INTEGER,
    OKS_BALANCE CURR_TYPE,
    MAX_DELINQ_DATE INTEGER,
    KC_BALANCE CURR_TYPE,
    KONT_STATUS2 VARCHAR(20))
AS
declare variable mal_from_m_id id_type;
declare variable mal_from_m_number integer;
declare variable mal_from_reg_id integer;
declare variable mal_req_user id_type;
declare variable mal_appr_user id_type;
declare variable reg_id_prior integer;
declare variable un_from_c_id id_type;
declare variable un_fio varchar(100);
declare variable jl_name varchar(100);
declare variable m_from_c_id id_type;
declare variable m_last_k_id id_type;
declare variable ks_from_k_id id_type;
declare variable ks_date_end date;
declare variable m_original_id id_type;
declare variable kc_amount decimal(18,4);
declare variable k_amount decimal(18,4);
declare variable dolya_cl double precision;
begin
  reg_id_prior = 0;
  for select mal_from_m_id, mal_from_m_number, mal_from_reg_id, mal_req_user, -- Автор запроса на изменение статуса
             mal_appr_user, -- Пользователь, принявший решение
             mal_req_comment, -- Комментарий автора
             mal_appr_comment, -- Комментарий пользователя принявшего решение
             mal_appr_datetime
      from morapprlist, region_list(:rep_region_id, 1, 3, 1)
      where cast(mal_appr_datetime as date) between :rep_date_begin and :rep_date_end and
            mal_req_state = 0 -- вывод из Стоп-листа
            and
            mal_appr_status = 1 -- утверждено
            and
            mal_from_reg_id = id_list
      into mal_from_m_id, mal_from_m_number, mal_from_reg_id, mal_req_user, mal_appr_user, mal_req_comment,
           mal_appr_comment, mal_appr_date
  do
  begin
    if (reg_id_prior <> mal_from_reg_id) then
    begin
      reg_id_prior = mal_from_reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :mal_from_reg_id
      into reg_name;
    end

    un_from_c_id = null;
    un_fio = null;
    select un_from_c_id, un_fio
    from username
    where un_id = :mal_req_user
    into un_from_c_id, un_fio;
    if (un_fio is null) then
      un_fio = '';

    jl_name = null;
    select jl_name
    from hr_pstatus, joblist
    where ps_from_c_id = :un_from_c_id and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_date_begin <= :rep_date_end and
          ps_from_jl_id = jl_id
    order by ps_date_begin desc
    rows 1
    into jl_name;
    if (jl_name is null) then
      jl_name = '';
    else
      jl_name = ', должность: ' || jl_name;
    req_user = un_fio || jl_name;

    un_from_c_id = null;
    un_fio = null;
    select un_from_c_id, un_fio
    from username
    where un_id = :mal_appr_user
    into un_from_c_id, un_fio;
    if (un_fio is null) then
      un_fio = '';

    jl_name = null;
    select jl_name
    from hr_pstatus, joblist
    where ps_from_c_id = :un_from_c_id and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_date_begin <= :rep_date_end and
          ps_from_jl_id = jl_id
    order by ps_date_begin desc
    rows 1
    into jl_name;
    if (jl_name is null) then
      jl_name = '';
    else
      jl_name = ', должность: ' || jl_name;
    appr_user = un_fio || jl_name;

    m_original_id = null;
    m_from_c_id = null;
    select m_from_c_id, m_last_k_id, m_client_fio, m_original_id
    from moratory
    where m_id = :mal_from_m_id and
          m_number = :mal_from_m_number
    into m_from_c_id, m_last_k_id, m_client_fio, m_original_id;

    if (m_last_k_id is null) then
    begin
      select ks_from_k_id
      from clients, knclient, kontstate
      where c_name_fio = :m_client_fio and
            c_id = kc_from_c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin <= :rep_date_end
      order by ks_date_begin desc
      rows 1
      into m_last_k_id;
      if (m_last_k_id is null) then
        m_last_k_id = 0;
    end
    if (m_from_c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :m_from_c_id
      into m_client_fio;
    end

    kontno_disbdate = null;
    select k_kontract_no || ' / ' ||(select sval
                                     from datetostr(k_disb_date)), k_amount_approved
    from kontract
    where k_id = :m_last_k_id
    into kontno_disbdate, k_amount;

    kc_amount = null;
    select kc_amount_approved
    from knclient
    where kc_from_k_id = :m_last_k_id and
          kc_from_c_id = :m_from_c_id and
          kc_amount_approved > 0
    into kc_amount;
    if (kc_amount is null) then
      kc_amount = 0;
    dolya_cl = cast(kc_amount as double precision) / k_amount;

    if (m_last_k_id <> 0) then
      kont_status = 'Закрытый';
    else
      kont_status = '';
    if (m_original_id is not null) then
      kont_status2 = 'Импортирован';
    else
      kont_status2 = '';

    oks_balance = null;
    delinq_day = null;
    select 'Активный', :rep_date_end - oks_delinq_date, oks_balance
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_from_k_id = :m_last_k_id
          and oks_close_state = 0
    into kont_status, delinq_day, oks_balance;

    kc_balance = dolya_cl * oks_balance;

    if (delinq_day < 0) then
      delinq_day = 0;

    select max(ok_ol_date - ok_sh_date)
    from operkred
    where ok_ol_is_active = 1 and
          ok_from_k_id = :m_last_k_id and
          ok_ol_date <= :rep_date_end and
          ok_ol_date > ok_sh_date and
          ok_type in (4, 5) and
          ok_os + ok_perc < 0
    into max_delinq_date;

    suspend;

    for select ks_from_k_id, ks_date_end, kc_amount_approved
        from kontstate, knclient
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_end >= :mal_appr_date and
              ks_from_k_id <> :m_last_k_id and
              ks_from_k_id = kc_from_k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = :m_from_c_id
        into ks_from_k_id, ks_date_end, kc_amount
    do
    begin
      kontno_disbdate = null;
      select k_kontract_no || ' / ' ||(select sval
                                       from datetostr(k_disb_date)), k_amount_approved
      from kontract
      where k_id = :ks_from_k_id
      into kontno_disbdate, k_amount;

      dolya_cl = cast(kc_amount as double precision) / k_amount;

      if (ks_date_end <= :rep_date_end) then
        kont_status = 'Закрытый';
      else
        kont_status = 'Активный';

      oks_balance = null;
      delinq_day = null;
      select :rep_date_end - oks_delinq_date, oks_balance
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_from_k_id = :ks_from_k_id
            and oks_close_state = 0
      into delinq_day, oks_balance;

      kc_balance = dolya_cl * oks_balance;

      if (delinq_day < 0) then
        delinq_day = 0;

      select max(ok_ol_date - ok_sh_date)
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :ks_from_k_id and
            ok_ol_date <= :rep_date_end and
            ok_ol_date > ok_sh_date and
            ok_type in (4, 5) and
            ok_os + ok_perc < 0
      into max_delinq_date;

      suspend;
    end
  end
end
