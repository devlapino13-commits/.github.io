CREATE PROCEDURE FM_AKB_DATA_HTML(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    PERSON_INFO D_BLOB_TEXT)
AS
declare variable PERSON_INFO_TEMP D_BLOB_TEXT;
declare variable UN_FIO D_STR_100;
declare variable LANG D_STR_LANG;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable BORROWER D_STR_4096;
declare variable NUMBER_OF_CREDITS D_INTEGER;
declare variable CREDIT_TEXT D_STR_8100;
declare variable NUMBER_OF_GUARANTIES D_INTEGER;
declare variable GUARANTEE_TEXT D_STR_8100;
declare variable REPORTING_DATE D_STR_20;
declare variable STEP D_INTEGER;
declare variable I D_INTEGER;
declare variable N D_INTEGER;
declare variable C_FIO D_STR_100;
declare variable C_BIRTHDAY D_STR_20;
declare variable C_DOC_NO D_STR_20;
declare variable FIN D_STR_20;
declare variable SCORE D_STR_100;
declare variable SCORE_RATE D_STR_LONG;
declare variable SCORE_RATE_DESCRIPTION D_STR_100;
declare variable RISK_RATE D_STR_LONG;
declare variable RISK_RATE_DESCRIPTION D_STR_255;
declare variable BANK_ID D_INTEGER; /* 2180 - id mbulak */
declare variable BANK_NAME D_STR_100;
declare variable NUMBER_ACTIVE_CREDITS D_INTEGER; /* активные займы в других ФКУ */
declare variable BALANCE_ACTIVE_CREDITS decimal(18,2); /* активный баланс в других ФКУ */
declare variable NUMBER_ACTIVE_CREDITS_P D_INTEGER; /* активные займы в других ФКУ, где он(а) поручитель: */
declare variable MAX_NAMBER_OF_DELIQ_DAYS D_INTEGER; /* макс. дни просрочки в других ФКУ: */
declare variable CREDIT_TYPE D_STR_100; /* тип кредита */
declare variable CREDIT_STATUS D_STR_100; /* статус кредита */
declare variable MAIN_AMOUNT decimal(18,2); /* основная сумма */
declare variable LINE_AMOUNT decimal(18,2); /* основная сумма */
declare variable REMAINING_AMOUNT decimal(18,2); /* оставшаяся сумма */
declare variable MONTHLY_PAYMENT decimal(18,2); /* ежемесячный платёж */
declare variable DATE_OF_RECEIPT D_STR_10; /* дата выдачи */
declare variable CLOSING_DATE D_STR_10; /* дата закрытия */
declare variable ACTUAL_CLOSING_DATE D_STR_10; /* фактическая дата закрытия */
declare variable PERIOD_IN_MONTH D_INTEGER; /* период на который брался кредит */
declare variable CURRENCY D_STR_10; /* валюта */
declare variable COMMON_POINTS D_INTEGER; /* баллы клиента по всем продуктам */
declare variable DAYS_OVERDUE D_INTEGER; /* количество дней активной просрочки */
declare variable CREDIT_PURPOSE D_STR_100; /* цель займа */
declare variable LAST_PAYMENT_DATE D_STR_10; /* дата последнего платежа */
declare variable DEBT_INTEREST decimal(18,2); /* задолжность по %% */
begin
    person_info = (select res_style from fm_html_styles) || '<body>';
    person_info_temp = '';

    lang = f_get_lang_code(un_id, null);
    bls_id = 0;

    select first 1 coalesce(eil.eil_from_bls_id, 0)
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
        and eic.eic_code = 'akb'
    where eil.eil_object_kind = f_objkind('clients', 'id')
        and eil.eil_object_id = :c_id
        and eil.eil_status = 2
        and eil.eil_date >= '20.11.2024'  -- дата изменения структуры json
    order by eil.eil_create_datetime desc
    into :bls_id;

    if(bls_id = 0) then
    begin
        person_info = blob_append(person_info, '<b>' || f_lang(:lang, 'Необходимо переотправить запрос в АКБ') || '<b><br>');
        person_info = blob_append(person_info, '</body>');
        suspend;
        exit;
    end
    
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into :json_text;

    select un.un_fio
    from username un
    where un.un_id = :un_id
    into un_fio;

    if(json_text is null or json_text containing 'Şəxs haqqında məlumat mövcud deyil.') then
    begin
        person_info = blob_append(person_info, '<b>' || f_lang(:lang, 'Информации по клиенту не найдена') || '<b><br>');
        person_info = blob_append(person_info, '</body>');
        suspend;
        exit;
    end

    reporting_date = left(f_json_blob_get_value(:json_text, 'reportingDate', '01-01-0001'), 10);
    person_info = blob_append(person_info, '<table width="100%"><col width="40%"><col width="60%">' ||
        f_html_table_row(f_lang(:lang, 'Дата проверки'), :reporting_date, '', '', -1, -1, -1, -1, 2, '') ||
        f_html_table_row(f_lang(:lang, 'Пользователь'), :un_fio, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');

    -- информация по клиенту
    borrower =  f_json_blob_get_value(:json_text, 'borrower', '');
    if(borrower is not null) then
    begin
        c_fio = f_json_str_get_value(:borrower, 'name', 'не найдена');
        person_info = blob_append(person_info, '<b>' || f_lang(:lang, 'Клиент') || ' ' || c_fio || '<b><br>');
    
        c_birthday = f_json_str_get_value(:borrower, 'dateOfBirth', '01-01-0001');
        c_doc_no = f_json_str_get_value(:borrower, 'documentNo', '');
        fin = f_json_str_get_value(:borrower, 'fin', '');
        person_info = blob_append(person_info, '<table width="100%"><col width="40%"><col width="60%">' ||
            f_html_table_row(f_lang(:lang, 'Дата рождения'), :c_birthday, '', '', -1, -1, -1, -1, 2, '') ||
            f_html_table_row(f_lang(:lang, 'Документ'), :c_doc_no, '', '', -1, -1, -1, -1, 2, '') ||
            f_html_table_row(f_lang(:lang, 'Пин код для АЗ'), :fin, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');
    
    end

    -- Скоринг
    score = f_json_blob_get_value(:json_text, 'score', '');
    if(score is not null) then
    begin
        score_rate = f_json_str_get_value(:score, 'point', '0');
        score_rate_description =
            case
                when score_rate between 1 and 199 then f_lang(:lang, 'Плохо')
                when score_rate between 199 and 599 then f_lang(:lang, 'Удовлетворительно')
                when score_rate between 599 and 749 then f_lang(:lang, 'Средне')
                when score_rate between 749 and 859 then f_lang(:lang, 'Хорошо')
                when score_rate between 859 and 1000 then f_lang(:lang, 'Отлично')
                else f_lang(:lang, 'Не определено')
            end;
    
        risk_rate = f_json_str_get_value(:score, 'response', '');
        risk_rate_description =
            case
                when risk_rate = 'AB' then f_lang(:lang, 'В активном обязательстве физического лица за текущий период есть высокое количество дней просрочки платежа.')
                when risk_rate = 'CC' then f_lang(:lang, 'За последние 12 месяцев у лица не было активных обязательств.')
                when risk_rate = 'TN' then f_lang(:lang, 'Очень рано для оценивания лица')
                when risk_rate = 'NC' then f_lang(:lang, 'У лица нет никаких кредитных обязательств.')
                when risk_rate = 'TY' then f_lang(:lang, 'Лицо младше 16 лет.')
                when risk_rate = 'NU' then f_lang(:lang, 'У лица есть необновленные активные кредиты.')
                when risk_rate = 'LP' then f_lang(:lang, 'Начисление баллов юридическим лицам не предусмотрено.')
                else f_lang(:lang, 'Не определено')
            end;
    end

    person_info = blob_append(person_info, '<b>' || f_lang(:lang, 'Скоринг') || '<b><br>');
    person_info = blob_append(person_info, '<table width="100%"><col width="40%"><col width="60%">' ||
        f_html_table_row(f_lang(:lang, 'Скоринг балл'), :score_rate || ' - ' || '<b>' || score_rate_description || '</b>', '', '', -1, -1, -1, -1, 2, '') ||
        f_html_table_row(f_lang(:lang, 'Степень риска'), :risk_rate || ' - ' || '<b>' || risk_rate_description || '</b>', '', '', -1, -1, -1, -1, 2, '') || '</table><br>');
        
    -- внутренний скоринг
    common_points = f_point(:c_id, current_date, 0);
    person_info = blob_append(person_info, '<b>' || f_lang(:lang, 'Внутренний скоринг Компании') || '<b><br>');
    person_info = blob_append(person_info, '<table width="100%"><col width="40%"><col width="60%">' ||
        f_html_table_row(f_lang(:lang, 'Баллы по всем продуктам'), :common_points, '', '', -1, -1, -1, -1, 2, '') || '</table><br><br>');

    number_active_credits = 0;
    balance_active_credits = 0;
    number_active_credits_p = 0;
    max_namber_of_deliq_days = 0;

    -- кредитная история
    number_of_credits = coalesce(f_json_blob_get_count(:json_text, 'liabilities.liability'), 0) - 1;
    if (number_of_credits >= 0) then
    begin
        person_info_temp = blob_append(person_info_temp, '<h2>' || f_lang(:lang, 'Кредитная история клиента как заёмщик') || '<h2>');
        step = 0;
        i = 0;
        n = 0;
        while (i <= number_of_credits) do
        begin
            credit_status = '000';
            credit_text = f_json_blob_get_value(:json_text, 'liabilities.liability[' || :i || ']', '');
            if (credit_text = '') then
            begin
                person_info_temp = blob_append(person_info_temp, f_lang(:lang, 'Ничего не найдено'));

                i = i + 1;
                leave;
            end

            bank_id = cast(f_json_str_get_value(:credit_text, 'bankId', '0') as d_integer);
            bank_name = f_json_str_get_value(:credit_text, 'bankName', '');
            credit_status = f_json_str_get_value(:credit_text, 'creditStatus', '');
            credit_type = f_json_str_get_value(:credit_text, 'creditType', '');
            main_amount = cast(f_json_str_get_value(:credit_text, 'initialAmount', '0') as decimal(18,2));
            line_amount = cast(f_json_str_get_value(:credit_text, 'lineAmount', '0') as decimal(18,2));
            remaining_amount = cast(f_json_str_get_value(:credit_text, 'outstandingDebtMain', '0') as decimal(18,2));
            monthly_payment = cast(f_json_str_get_value(:credit_text, 'monthlyPaymentAmount', '0') as decimal(18,2));
            date_of_receipt = left(f_json_str_get_value(:credit_text, 'grantedOn', '01-01-0001'), 10);
            closing_date = left(f_json_str_get_value(:credit_text, 'contractDueOn', '01-01-0001'), 10);
            actual_closing_date = left(f_json_str_get_value(:credit_text, 'creditStatusCloseDate', ''), 10);
            currency = f_json_str_get_value(:credit_text, 'currency', '');
            days_overdue = f_json_str_get_value(:credit_text, 'daysMainSumOverdue', '0');
            credit_purpose = f_json_str_get_value(:credit_text, 'creditPurpose', '');
            last_payment_date = left(f_json_str_get_value(:credit_text, 'lastPaymentDate', ''), 10);
            debt_interest = cast(f_json_str_get_value(:credit_text, 'outstandingDebtInterest', '') as decimal(18,2));
            period_in_month = datediff(month, cast(date_of_receipt as date), cast(closing_date as date));

            if (main_amount = 0) then
                main_amount = line_amount;

            if (step = 0) then
            begin
                if (bank_id <> 2180 and credit_status in ('003', '007', '008', '009')) then
                begin
                    number_active_credits = number_active_credits + 1;
                    balance_active_credits = balance_active_credits + remaining_amount;
                    if (max_namber_of_deliq_days < days_overdue) then
                    begin
                        max_namber_of_deliq_days = days_overdue;
                    end
                end

                if (bank_id <> 2180 and max_namber_of_deliq_days < days_overdue and credit_status in ('001', '010', '011', '012')) then
                    max_namber_of_deliq_days = days_overdue;

                if (i = number_of_credits) then
                begin
                    i = 0;
                    step = step + 1;
                    continue;
                end

                i = i + 1;
                continue;
            end

            if ((step = 1 and credit_status not in ('003', '007', '008', '009'))
                or (step = 2 and credit_status not in ('001', '010', '011', '012'))) then
            begin
                if (i = number_of_credits and step < 2) then
                begin
                    i = 0;
                    step = step + 1;
                    continue;
                end

                i = i + 1;
                continue;
            end

            i = i + 1;
            n = n + 1;
            if (n = 9) then
                leave;

            bank_name =
                case
                    when bank_id = 2180 then bank_name
                    else f_lang(:lang, 'Другой ФКУ')
                end;

            credit_type =
                case
                    when credit_type = '001' then f_lang(:lang, 'Кредит')
                    when credit_type = '002' then f_lang(:lang, 'Кредитная линия')
                    when credit_type = '003' then f_lang(:lang, 'Кредитная карта')
                    when credit_type = '004' then f_lang(:lang, 'Гарантия и поручительство')
                    else ''
                end;

            credit_status =
                case
                    when credit_status = '001' then f_lang(:lang, 'Полностью оплаченные')
                    when credit_status = '003' then f_lang(:lang, 'Списанные с баланса')
                    when credit_status = '007' then f_lang(:lang, 'Рабочие')
                    when credit_status = '008' then f_lang(:lang, 'Нерабочие')
                    when credit_status = '009' then f_lang(:lang, 'Обязательства, отложенные из-за пандемии')
                    when credit_status = '010' then f_lang(:lang, 'Отложенные, закрытые из-за пандемии')
                    when credit_status = '011' then f_lang(:lang, 'Право требования передано третьему лицу')
                    else ''
                end;

            credit_purpose =
                case
                    when credit_purpose = '001' then f_lang(:lang, 'Потребительские кредиты физическим лицам')
                    when credit_purpose = '002' then f_lang(:lang, 'Нефтяной сектор')
                    when credit_purpose = '003' then f_lang(:lang, 'Ипотечные кредиты физическим лицам')
                    when credit_purpose = '004' then f_lang(:lang, 'Сектор торговли и обслуживания')
                    when credit_purpose = '005' then f_lang(:lang, 'Сектор энергетики, химии и природных ресурсов')
                    when credit_purpose = '006' then f_lang(:lang, 'Сектор сельского хозяйства и переработки')
                    when credit_purpose = '007' then f_lang(:lang, 'Сектор строительства и недвижимости')
                    when credit_purpose = '008' then f_lang(:lang, 'Промышленность и производственный сектор')
                    when credit_purpose = '009' then f_lang(:lang, 'Транспортно-коммуникационный сектор')
                    when credit_purpose = '010' then f_lang(:lang, 'Другие секторы')
                    when credit_purpose = '011' then f_lang(:lang, 'Аренда жилого помещения с правом выкупа')
                    else ''
                end;

            person_info_temp = blob_append(person_info_temp, '<br><br><b>'||f_lang(:lang, 'Кредит № ')|| :n ||'<b><br>');
            person_info_temp = blob_append(person_info_temp, '<table width="100%"><col width="40%"><col width="60%">' ||
                f_html_table_row(f_lang(:lang, 'ФКУ'), :bank_name, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Валюта'), :currency, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Участник'), f_lang(:lang, 'Заёмщик'), '', '', -1, -1, -1, -1, 2, ''));
                
            person_info_temp = blob_append(person_info_temp,
                f_html_table_row(f_lang(:lang, 'Тип'), :credit_type, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Цель выдачи'), :credit_purpose, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Статус кредита'), :credit_status, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Дата выдачи'), :date_of_receipt, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Дата окончания'), :closing_date, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Фактическая дата закрытия'), :actual_closing_date, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Дата последнего платежа'), :last_payment_date, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Срок кредита (в месяцах)'), :period_in_month, '', '', -1, -1, -1, -1, 2, '') ||
                
                f_html_table_row(f_lang(:lang, 'Ежемесячный платёж'), :monthly_payment, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Общая сумма'), :main_amount, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Остаток ОС'), :remaining_amount, '', '', -1, -1, -1, -1, 2, '') ||
                f_html_table_row(f_lang(:lang, 'Задолженность по %%'), :debt_interest, '', '', -1, -1, -1, -1, 2, '') ||

                f_html_table_row(f_lang(:lang, iif(credit_status = 'Полностью оплаченные', 'Кол-во дней просрочки на момент закрытия займа','Кол-во дней активной просрочки')), :days_overdue, '', '', -1, -1, -1, -1, 2, '') || '</table><br>');
        end
    end
    else
        person_info_temp = blob_append(person_info_temp, f_lang(:lang, 'Ничего не найдено'));

    -- поручительство
    person_info_temp = blob_append(person_info_temp, '<h2>' || f_lang(:lang, 'Кредитная история клиента как поручитель') || '<h2>');
    number_of_guaranties = coalesce(f_json_blob_get_count(:json_text, 'guarantee'), 0) - 1;
    if (number_of_guaranties >= 0) then
    begin
        i = 0;
        while (i <= number_of_guaranties) do
        begin
            guarantee_text = f_json_blob_get_value(:json_text, 'guarantee['|| :i ||']', '');
            if (guarantee_text = '') then
            begin
                person_info_temp = blob_append(person_info_temp, f_lang(:lang, 'Ничего не найдено'));

                i = i + 1;
                leave;
            end

            bank_id = cast(f_json_str_get_value(:guarantee_text, 'guaBankId', '0') as d_integer);
            bank_name = f_json_str_get_value(:guarantee_text, 'guaBankName', '');
            credit_type = f_json_str_get_value(:guarantee_text, 'guaCreditType', '');
            credit_status = f_json_str_get_value(:guarantee_text, 'guaCreditStatus', '');
            main_amount = cast(f_json_str_get_value(:guarantee_text, 'guaInitialAmount', '0') as decimal(18,2));
            remaining_amount = cast(f_json_str_get_value(:guarantee_text, 'guaOutstandingDebt', '0') as decimal(18,2));
            monthly_payment = cast(f_json_str_get_value(:guarantee_text, 'guaMonthlyPaymentAmount', '0') as decimal(18,2));
            date_of_receipt = left(f_json_str_get_value(:guarantee_text, 'guaGrantedOn', '01-01-0001'), 10);
            closing_date = left(f_json_str_get_value(:guarantee_text, 'guaContractDueOn', '01-01-0001'), 10);
            actual_closing_date = left(f_json_str_get_value(:guarantee_text, 'guaCreditStatusCloseDate', ''), 10);
            currency = left(f_json_str_get_value(:guarantee_text, 'guaCurrency', ''), 10);
            credit_purpose = f_json_str_get_value(:guarantee_text, 'guaCreditPurpose', '');
            period_in_month = datediff(month, cast(date_of_receipt as date), cast(closing_date as date));

            i = i + 1;
            if (bank_id <> 2180) then
                number_active_credits_p = number_active_credits_p + 1;

            bank_name =
                case
                    when bank_id = 2180 then bank_name
                    else f_lang(:lang, 'Другой ФКУ')
                end;

            credit_type =
                case
                    when credit_type = '001' then f_lang(:lang, 'Кредит')
                    when credit_type = '002' then f_lang(:lang, 'Кредитная линия')
                    when credit_type = '003' then f_lang(:lang, 'Кредитная карта')
                    when credit_type = '004' then f_lang(:lang, 'Гарантия и поручительство')
                    else ''
                end;

            credit_status =
                case
                    when credit_status = '001' then f_lang(:lang, 'Полностью оплаченные')
                    when credit_status = '003' then f_lang(:lang, 'Списанные с баланса')
                    when credit_status = '007' then f_lang(:lang, 'Рабочие')
                    when credit_status = '008' then f_lang(:lang, 'Нерабочие')
                    when credit_status = '009' then f_lang(:lang, 'Обязательства, отложенные из-за пандемии')
                    when credit_status = '010' then f_lang(:lang, 'Отложенные, закрытые из-за пандемии')
                    when credit_status = '011' then f_lang(:lang, 'Право требования передано третьему лицу')
                    else ''
                end;

            credit_purpose =
                case
                    when credit_purpose = '001' then f_lang(:lang, 'Потребительские кредиты физическим лицам')
                    when credit_purpose = '002' then f_lang(:lang, 'Нефтяной сектор')
                    when credit_purpose = '003' then f_lang(:lang, 'Ипотечные кредиты физическим лицам')
                    when credit_purpose = '004' then f_lang(:lang, 'Сектор торговли и обслуживания')
                    when credit_purpose = '005' then f_lang(:lang, 'Сектор энергетики, химии и природных ресурсов')
