CREATE PROCEDURE UXP$UXEXPOFORM_PREPARE(
    UFM_UUID_CODE D_STR_CODE,
    CLONE_KIND D_SMALLINT,
    NEW_UX_CODE D_STR_CODE,
    NEW_UX_NAME D_STR_100,
    DB_CONNECTION D_STR_LONG,
    PASS D_STR_CODE)
AS
declare variable ufm_uuid d_uuid;
begin
  db_connection = trim(coalesce(db_connection, ''));

  delete from ux$expoform
  where 1 = 1;

  if (clone_kind = 2 and ufm_uuid_code = '') then
    uxf$flag('import_from_file', 'on');
  else
    uxf$flag('import_from_file', 'off');

  if (coalesce(ufm_uuid_code, '') <> '') then
  begin
    uxf$flag('import_from_file', 'off');

    if (db_connection <> '') then
    begin
      uxf$pswd(pass);
      uxf$syncpath(db_connection);
    end
  
    begin
      ufm_uuid = char_to_uuid(ufm_uuid_code);
      when any do
        exception e_message 'Неверный идентификатор формы';
    end
  
    insert into ux$expoform (exf_ufm_uuid, exf_clone_kind, exf_new_kind, exf_new_code, exf_new_name, exf_show_child_action)
    values (:ufm_uuid, :clone_kind, 1, :new_ux_code, :new_ux_name, iif(:clone_kind = 0, 1, 2));
  end
end
