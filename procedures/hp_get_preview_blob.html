CREATE PROCEDURE HP_GET_PREVIEW_BLOB(
    ID D_ID NOT NULL,
    ID_KIND D_STR_CODE NOT NULL)
RETURNS (
    RES_BLOB D_BLOB_BINARY)
AS
begin
  -- Процедура возвращает блоб, по переданному id и кодовому названию таблицы
  -- Используется в Finstart, в окне просмотра файлов TImagePreviewForm

  if (id_kind = 'blobstore') then
  begin
    select blx_blob from hp_get_blobex_blob(:id)
    into res_blob;
  end
  else if (id_kind = 'workdocblob') then
    select wdb_blob from workdocblob where wdb_id = :id
    into res_blob;
  else
  if (id_kind = 'edoc_blank') then
  begin
    select coalesce(edb_blob, blx_blob)
    from edoc_blank, hp_get_blobex_blob(edb_from_bls_id)
    where edb_id = :id
    into res_blob;
  end
  else
    exception e_default 'Incorrect id_kind: ' || id_kind;
  suspend;
end
