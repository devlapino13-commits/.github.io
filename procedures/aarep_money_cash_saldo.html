CREATE PROCEDURE AAREP_MONEY_CASH_SALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_C_ID ID_TYPE = null)
RETURNS (
    RES_SALDO_CASH CURR_TYPE,
    RES_SALDO_BANK_CHECK CURR_TYPE,
    RES_SALDO_CASH_ORDER CURR_TYPE,
    RES_SALDO_PLED_DOCUM CURR_TYPE,
    RES_NUM_CASH INTEGER)
AS
declare variable s1 decimal(18,4);
declare variable al_id varchar(20);
declare variable atl_acc_mask varchar(20);
declare variable reg_add str_20;
declare variable acc_norm varchar(32);
declare variable acc_ipf varchar(32);
begin
  res_saldo_cash = 0;
  res_saldo_bank_check = 0;
  res_saldo_cash_order = 0;
  res_saldo_pled_docum = 0;
  res_num_cash = coalesce((
    select uc_num_cash from user_cash
    where uc_from_c_id = :report_c_id and uc_from_reg_id = :report_region_id),0);

  for select atl_acc_mask
      from acctypelist
      where atl_from_at_id = 11
      into :atl_acc_mask
  do
  begin
    acc_norm = atl_acc_mask || '/' || res_num_cash;
    acc_ipf = atl_acc_mask || '/' || (100 + res_num_cash);

    for select al_id
        from acc_list
        where al_from_reg_id = :report_region_id
          and al_from_bg_id = :atl_acc_mask
          and iif(:report_c_id >= 0, al_account in (:acc_norm, :acc_ipf), 1 = 1)
        into :al_id
    do
    begin
      s1 = f_al_bal(al_id, report_date_end);
      res_saldo_cash = res_saldo_cash + s1;
    end
  end
  -- S.K.
  if (report_c_id >= 0 and res_num_cash = 1) then
  begin
    reg_add = (select reg_code || '41796001'
      from region where reg_unicode = :report_region_id);

    --96001/1 залоговые доки
    res_saldo_pled_docum = f_al_bal(:reg_add || '000001', report_date_end);

    --96001/2 бсо
    res_saldo_cash_order = f_al_bal(:reg_add || '000002', report_date_end);

    --96001/3 банковские чеки
    res_saldo_bank_check = f_al_bal(:reg_add || '000003', report_date_end);
  end
  suspend;
end
