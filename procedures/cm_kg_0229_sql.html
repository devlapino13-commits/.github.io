CREATE PROCEDURE CM_KG_0229_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable MA_ID D_ID;
declare variable K_DISB_DATE D_DATE;
declare variable KAGENT_C_ID D_ID;
begin
  select kt.k_from_ma_app_id, kt.k_disb_date
  from kontract kt
  inner join product pr on pr.pr_id = kt.k_from_pr_id
  where kt.k_id = :k_id and
        pr.pr_code containing '.installment_goods.' and 1<>1

  into ma_id, k_disb_date;

  if (coalesce(ma_id, 0) = 0) then
    exit;

  for select distinct cs.cs_from_c_id
      from mc_purchase mp
      inner join clientshops cs on cs.cs_id = mp.mp_from_cs_id
      where mp.mp_from_ma_id = :ma_id
      into :kagent_c_id
  do
  begin
    if (not exists(select *
                   from agreements ag
                   where ag.ag_object_kind = f_objkind('clients', 'id') and
                         ag.ag_object_id = :kagent_c_id and
                         :k_disb_date between ag.ag_date_begin and coalesce(ag.ag_date_end, '31.12.9999') and
                         ag.ag_from_rd_id = f_get_rd_id('AgreementKinds.installment_goods') and
                         ag.ag_status = 1)) then
    begin
      res_error_text = 'У контрагента нет активного договора по товарам в рассрочку';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
