CREATE PROCEDURE HP_GC_ANALIZ_POSTPROCESS(
    IGA_ID D_INTEGER)
AS
declare variable ifa_id d_integer;
declare variable c_id d_id;
declare variable ag_id d_integer;
declare variable commis_mb d_percent;
declare variable commis_other d_percent;
declare variable amount_mb d_currency;
declare variable amount_other d_currency;
declare variable amount d_currency;
begin
  ifa_id = (select iga_from_ifa_id from ipc_gc_analiz where iga_id = :iga_id);
  for
    with temp as (
      select
        igdp_from_c_id as c_id,
        max(igdp_from_ag_id) as ag_id ,
        sum(igdp_amount) as amount,
        sum(iif(igdp_is_our_card = 1, igdp_tsp_commis_amount, 0)) as amount_mb,
        sum(iif(igdp_is_our_card = 1, 0, igdp_tsp_commis_amount)) as amount_other
      from ipc_gc_analiz_detail_pos
      where igdp_from_iga_id = :iga_id
        and igdp_type = 1
        and igdp_is_error = 0
      group by igdp_from_c_id
      union all
      select
        igqp_from_c_id as c_id,
        max(igqp_from_ag_id) as ag_id,
        sum(igqp_amount) as amount,
        0 as amount_mb,
        sum(igqp_tsp_commis_amount) as amount_other
      from ipc_gc_analiz_qr_payments
      where igqp_from_iga_id = :iga_id
        and igqp_is_error = 0
      group by igqp_from_c_id
    )
    select
      c_id,
      max(ag_id),
      coalesce(sum(amount), 0),
      coalesce(sum(amount_mb), 0),
      coalesce(sum(amount_other), 0)
    from temp
    where ag_id > 0
    group by c_id
  into
    :c_id,
    :ag_id,
    :amount,
    :amount_mb,
    :amount_other
  do
  begin
    select
      coalesce(ag_commis_cards_mb, 0),
      coalesce(ag_commis_cards_other, 0)
    from agreements
    where ag_id = :ag_id
    into
      commis_mb,
      commis_other;

    insert into ipc_gc_commission (
      igc_from_iga_id,
      igc_from_c_id,
      igc_from_ag_id,
      igc_commis_cards_mb,
      igc_commis_cards_other,
      igc_amount_cards_mb,
      igc_amount_cards_other,
      igc_amount
    )
    values (
      :iga_id,
      :c_id,
      :ag_id,
      :commis_mb,
      :commis_other,
      round(:amount_mb, 2),
      round(:amount_other, 2),
      :amount
    );
  end

  if (exists(select * from ipc_gc_analiz_detail_pos where igdp_from_iga_id = :iga_id and igdp_error_id > 0) or
      exists(select * from ipc_gc_analiz_detail_qr where igdq_from_iga_id = :iga_id and igdq_error_id > 0) or
      exists(select * from ipc_gc_analiz_qr_payments where igqp_from_iga_id = :iga_id and igqp_error_id > 0))
  then update ipc_file_analiz set ifa_is_detail_problem = 1 where ifa_id = :ifa_id;
end
