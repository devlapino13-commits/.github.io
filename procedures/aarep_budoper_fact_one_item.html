CREATE PROCEDURE AAREP_BUDOPER_FACT_ONE_ITEM(
    REP_REGION_ID INTEGER,
    REP_BIT_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    BOI_FROM_BIT_ID INTEGER,
    BOI_FACT_01 DECIMAL(18,4),
    BOI_FACT_02 DECIMAL(18,4),
    BOI_FACT_03 DECIMAL(18,4),
    BOI_FACT_04 DECIMAL(18,4),
    BOI_FACT_05 DECIMAL(18,4),
    BOI_FACT_06 DECIMAL(18,4),
    BOI_FACT_07 DECIMAL(18,4),
    BOI_FACT_08 DECIMAL(18,4),
    BOI_FACT_09 DECIMAL(18,4),
    BOI_FACT_10 DECIMAL(18,4),
    BOI_FACT_11 DECIMAL(18,4),
    BOI_FACT_12 DECIMAL(18,4))
AS
declare variable tmp_amount decimal(18,4);
declare variable dd_begin date;
declare variable dd_end date;
declare variable dd_year integer;
declare variable i integer;
begin
  dd_year = extract(year from rep_date_end);
  boi_from_bit_id = :rep_bit_id;
  boi_fact_01 = 0;
  boi_fact_02 = 0;
  boi_fact_03 = 0;
  boi_fact_04 = 0;
  boi_fact_05 = 0;
  boi_fact_06 = 0;
  boi_fact_07 = 0;
  boi_fact_08 = 0;
  boi_fact_09 = 0;
  boi_fact_10 = 0;
  boi_fact_11 = 0;
  boi_fact_12 = 0;
  i = 0;
  while (i < 12) do
  begin
    i = i + 1;
    if (i < 10) then
      dd_begin = cast('01.0' || :i || '.' || :dd_year as date);
    else
      dd_begin = cast('01.' || :i || '.' || :dd_year as date);
    if (dd_begin >= :rep_date_begin) then
    begin
      if (i < 12) then
      begin
        if (i < 9) then
          dd_end = cast('01.0' ||(:i + 1) || '.' || :dd_year as date) - 1;
        else
          dd_end = cast('01.' ||(:i + 1) || '.' || :dd_year as date) - 1;
      end
      else
        dd_end = cast('31.12.' || :dd_year as date);
      if (dd_end > rep_date_end) then
        dd_end = :rep_date_end;
      if (dd_begin <= :dd_end) then
      begin
        select boi_fact
        from aarep_budoper_fact_period(:rep_region_id, :rep_bit_id, :dd_begin, :dd_end)
        into :tmp_amount;
        if (i = 1) then
          boi_fact_01 = :tmp_amount;
        if (i = 2) then
          boi_fact_02 = :tmp_amount;
        if (i = 3) then
          boi_fact_03 = :tmp_amount;
        if (i = 4) then
          boi_fact_04 = :tmp_amount;
        if (i = 5) then
          boi_fact_05 = :tmp_amount;
        if (i = 6) then
          boi_fact_06 = :tmp_amount;
        if (i = 7) then
          boi_fact_07 = :tmp_amount;
        if (i = 8) then
          boi_fact_08 = :tmp_amount;
        if (i = 9) then
          boi_fact_09 = :tmp_amount;
        if (i = 10) then
          boi_fact_10 = :tmp_amount;
        if (i = 11) then
          boi_fact_11 = :tmp_amount;
        if (i = 12) then
          boi_fact_12 = :tmp_amount;
      end
    end
  end
  suspend;
end
