CREATE PROCEDURE HP_KONTRACTS_EPS_FIX
AS
declare variable k_id d_id;
declare variable new_eps d_percent;
declare variable kd_id d_id;
begin
  for select cke.cke_from_k_id, cke.cke_new_eps
  from card_kontracts_eps cke
  where cke.cke_status = 1
  rows 10
  into :k_id, :new_eps
  do
  begin
    kd_id = null;

    select kd.kd_id
    from kontdata kd
    where kd.kd_from_k_id = :k_id
    and kd.kd_is_primary = 1
    and kd.kd_ol_is_delete = 0
    and kd.kd_date_end >= kd.kd_ol_date
    order by kd.kd_id
    into :kd_id;

    if (kd_id is null) then
        update card_kontracts_eps cke
        set cke.cke_status = -1
        where cke.cke_from_k_id = :k_id;
    else
    begin
        update kontract kont
        set kont.k_percent_effect = :new_eps,
            kont.k_percent_effect_fifd = :new_eps
        where kont.k_id = :k_id;
        
        update kontdata kd
        set kd.kd_percent_effect = :new_eps,
            kd.kd_percent_effect_fifd = :new_eps
        where kd.kd_id = :kd_id;
    
        update card_kontracts_eps cke
        set cke.cke_status = 2,
            cke.cke_new_eps = :new_eps
        where cke.cke_from_k_id = :k_id;
    end
  end
end
