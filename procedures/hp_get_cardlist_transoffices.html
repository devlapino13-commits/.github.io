CREATE PROCEDURE HP_GET_CARDLIST_TRANSOFFICES(
    CARD_LIST D_STR_8100)
RETURNS (
    CARD_ID D_INTEGER,
    IS_PINCOVER D_BOOL,
    REG_ID D_INTEGER)
AS
declare variable ol_datetime d_datetime;
begin
  for
    select
      cal_card_id,
      is_pincover,
      max(cam_ol_datetime)
    from hp_cardrange_jsontolist(:card_list)
      inner join cardlist
        on cal_card_id between first_card_id and last_card_id
          --and cal_status = 1
      inner join cardaktrange
        on cal_card_id between car_first_card_id and car_last_card_id
      inner join cardaktmain
        on cam_id = car_from_cam_id
      inner join cardplace
        on cpl_id = cam_dst_cpl_id
          and cpl_is_transfer = 0
      inner join operlist
        on ol_id = cam_from_ol_id
          and is_delete = 0
    group by cal_card_id, is_pincover
  into
    :card_id,
    :is_pincover,
    :ol_datetime
  do
  begin
    reg_id = (
      select
        cpl_from_reg_id
      from cardaktmain
        inner join cardplace
          on cpl_id = cam_src_cpl_id
      where cam_ol_datetime = :ol_datetime);
    suspend;
  end
end
