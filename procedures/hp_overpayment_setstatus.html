CREATE PROCEDURE HP_OVERPAYMENT_SETSTATUS(
    OVP_ID D_INTEGER NOT NULL,
    NEWSTATUS D_INTEGER NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable status d_integer;
declare variable phone d_str_20;
declare variable edl_id d_id;
declare variable esm_id d_id;
declare variable disb_date d_date;
declare variable date_need_sign d_date;
begin
  err_text = '';
  date_need_sign = '12.04.2021';

  select
    ovp_status,
    coalesce(ovp_phone, ''),
    ovp_edl_id,
    k_disb_date
  from overpayment
    inner join kontract
      on k_id = ovp_from_k_id
  where ovp_id = :ovp_id
    and ovp_edl_id > 0
    and coalesce(ovp_phone, '') <> ''
  into
    :status,
    :phone,
    :edl_id,
    :disb_date;
  if (row_count = 1) then
  begin
    if (status = 0 and newstatus = 2 and disb_date < date_need_sign) then
      newstatus = 1;
    if (newstatus <> status) then
    begin
      if (status = 0 and newstatus = 1) then
        select
          res_text
        from hp_edoc_ins('overpayment.phone', f_objkind('overpayment', 'id'), :ovp_id, :un_id)
        into
          :err_text;

      if (status in (0, 1) and newstatus = 2) then
        select
          esm_id,
          res_text
        from hp_overpayment_transfer(:ovp_id)
        into
          :esm_id,
          :err_text;

      if (err_text <> '') then
        exception e_message err_text;
      else
        update overpayment set
          ovp_status = :newstatus,
          ovp_check_esm_id = iif(:newstatus = 2, :esm_id, ovp_check_esm_id),
          ovp_datetime = current_timestamp,
          ovp_from_un_id = :un_id
        where ovp_id = :ovp_id;
    end
  end else
    err_text = f_format('Запись по возврату переплаты не опредена или условия не соответствуют (ovp_id = %1)', ovp_id);
  suspend;
end
