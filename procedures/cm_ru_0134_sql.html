CREATE PROCEDURE CM_RU_0134_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable PK_ID integer;
declare variable DOC_NAME STR_100;
declare variable C_ID ID_TYPE;
begin
  /*
    SYSDBA
    |По запросу Юлии, добавлено проверка на второй файл патента
*/
  select K_DISB_DATE
  from KONTRACT
  where K_ID = :K_ID
  into KF_DISB_DATE;
  if (KF_DISB_DATE is null) then
    exit;
  else
  begin
    for select KC_FROM_C_ID
        from KNCLIENT KC
        where KC_FROM_K_ID = :K_ID and
              KC_AMOUNT_APPROVED > 0
        into C_ID
    do
    begin
      select C_NAME_FIO
      from CLIENTS
      where C_ID = :C_ID
      into :res_info_text;
      for select distinct CD.CDS_FROM_PK_ID
          from CLIENTDOCS CD
          where CD.CDS_FROM_C_ID = :C_ID and
                CD.CDS_IS_DELETE = 0 and
                :KF_DISB_DATE between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END and
                CDS_FROM_PK_ID = 49 and --patent
                (coalesce(CDS_DOC_BLOB_1, 0) <= 0 or coalesce(CDS_DOC_BLOB_2, 0) <= 0)
          into PK_ID
      do
      begin
        select PK.PK_NAME
        from PASSKIND PK
        where PK.PK_ID = :PK_ID
        into DOC_NAME;
        res_error_text = 'К документу "' || coalesce(:DOC_NAME, :PK_ID, '') || '" не прикреплен файл!';
        res_error_is_critical = 1;
        suspend;
      end
    end
  end
end
