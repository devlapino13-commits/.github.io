        values (:ol_id, :reg_id, 2, '01.01.2013', 0, '', 9,
                'Перевод остатка между новым и старым планом счетов', 0, 0,
                current_timestamp, 1000000000001, null, null, 0, null, null, null, 11, '01.01.2013', 0, null, 0, 2013, 1,
                null);
      end
      if (sum_ekv > 0) then
      begin
        select al_new_al_id
        from acc_list_link
        where al_old_al_id = :al_old
        into ac_dt;
        ac_kt = al_old;
      end
      else
      begin
        select al_new_al_id
        from acc_list_link
        where al_old_al_id = :al_old
        into ac_kt;
        ac_dt = al_old;
        sum_ekv = 0 - sum_ekv;
        sum_val = 0 - sum_val;
      end
      curr_rate = 1;
      if (ob_currency <> '944') then
      begin
        select cr_rate
        from opercurr
        where cr_from_cu_id = :ob_currency and
              cr_date <= '01.01.2013' and
              cr_date_end >= '01.01.2013'
        into :curr_rate;
      end
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                            ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                            ob_client_kt, ob_detail, ob_from_sac_id)
      values (:ol_id, -1, :ob_currency, :curr_rate, null, :ac_dt, :sum_val, :sum_ekv, null, null, null, :ac_kt, null,
              'Перевод на новый план счетов', null);

      suspend;
    end
  end
