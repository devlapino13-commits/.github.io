CREATE PROCEDURE HP_FINCERT_PASSPORT_CHECK(
    PASS_NO D_STR_CODE,
    CHECK_DATE_TIME D_DATETIME)
RETURNS (
    RES_DATE D_DATE,
    RES_STATUS D_SMALLINT,
    FF_CREATE_DATETIME D_DATETIME)
AS
declare variable hash_pass d_str_100;
declare variable ff_file_body d_str_255;
declare variable res_datetime d_datetime;
begin
    -- RFCRU-3628
  res_status = 0;
  hash_pass = f_fincert_passport_hash(:pass_no);
  ff_file_body = '';

  if(coalesce(hash_pass, '') = '') then
    exit;

  select ff_create_datetime,
    case
        when position(:hash_pass in ff_file_body) > 0 then
          -- найдена позиция номера
          case
            when position(',' in substring(t.ff_file_body from position(:hash_pass in t.ff_file_body))) > 0 then
              -- есть запятая после номера — берем до неё
              substring(
                t.ff_file_body from position(:hash_pass in t.ff_file_body)
                for position(',' in substring(t.ff_file_body from position(:hash_pass in t.ff_file_body))) - 1
              )
            else
              -- запятая не найдена — берем до конца
              substring(t.ff_file_body from position(:hash_pass in t.ff_file_body))
          end
        else
          ''
    end
  from hp_fincert_get_file('passport_hash', :check_date_time, 1) t
  into ff_create_datetime, ff_file_body;

  if (ff_file_body <> '') then
  begin
    res_status = 1;

    res_datetime = cast(substring(ff_file_body from
                                  position('|', ff_file_body) + 1 for
                                  position('|', substring(ff_file_body from position('|', ff_file_body) + 1)) - 1) as d_datetime);

    res_date = cast(res_datetime as d_date);
  end

  suspend;
end
