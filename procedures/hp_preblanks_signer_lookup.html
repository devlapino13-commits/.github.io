CREATE PROCEDURE HP_PREBLANKS_SIGNER_LOOKUP(
    IN_OBJ_KIND D_STR_CODE,
    IN_OBJ_ID D_STR_CODE,
    REP_DATE D_DATE,
    COMMON_SIGNER D_BOOL)
RETURNS (
    ID D_INTEGER,
    TEXT D_STR_255)
AS
declare variable rep_office_id d_integer;
declare variable db_id d_integer;
declare variable is_hr d_bool;
begin
  if (in_obj_kind is null) then
    exit;

  if (rep_date is null) then
    rep_date = current_date;

  if (common_signer is null) then
    common_signer = 0;

  if (is_hr is null) then
    is_hr = 0;

  if (f_objkind(in_obj_kind, 'code') in ('hr_orders', 'sa_worktype')) then
    is_hr = 1;

  begin
    select reg_id
    from hp_edoc_object_info(:in_obj_kind, :in_obj_id, uxf$user(), :rep_date)
    into :rep_office_id;

    when any do
    begin
    end
  end

  db_id = f_get_db_id();

  if (coalesce(rep_office_id, 1000000) = 1000000) then
    rep_office_id = f_hq_id(db_id);

  if (is_hr = 0) then
  begin
    for select id, text
    from uxp$preblanks_signer_by_region(:rep_office_id, :rep_date, :common_signer)
    into :id, :text
    do
      suspend;
  end else
  begin
    for select id, text
    from hp_e_hr_orders_signer_lookup(:rep_office_id, :rep_date)
    into :id, :text
    do
      suspend;
  end
end
