CREATE PROCEDURE AAREP_KOMMIS_DIFF_SH2(
    K_ID D_ID,
    NEW_AMOUNT D_CURRENCY,
    NEW_PERCENT_PER_DAY D_PERCENT)
RETURNS (
    SH_DATE D_DATE,
    SH_OS D_CURRENCY,
    SH_PERC D_CURRENCY,
    SH_ALL D_CURRENCY,
    SH_IS_LGOTA D_BOOL)
AS
declare variable sh_date_prv d_date;
declare variable days_list d_str_255;
declare variable annu d_currency;
declare variable disb_date d_date;
declare variable days d_integer;
declare variable last_date d_date;
declare variable balance d_currency;
declare variable is_lgota d_bool;
begin
  days_list = '';

  sh_date_prv = null;
  for select sh_date, iif(sh_id > 0 and sh_os = 0, 1, 0)
  from schedule
  where sh_from_k_id = :k_id
  order by sh_from_k_id, sh_id
  into sh_date, is_lgota
  do
  begin
    if (sh_date_prv is null) then
    begin
      sh_date_prv = sh_date;
      disb_date = sh_date;
    end
    else
    begin
      if (days_list <> '') then
        days_list = days_list || ',';

      days = datediff(day, sh_date_prv, sh_date);
      if (is_lgota = 1) then
        days = -days;

      days_list = days_list || days;
    end

    sh_date_prv = sh_date;
  end
  last_date = sh_date;

  annu = f_annu(new_amount, 0, 1, new_percent_per_day, days_list, 0, 0, 0, 0, 0);

  sh_date_prv = disb_date;
  balance = new_amount;

  for select sh_date, iif(sh_os = 0, 1, 0)
  from schedule
  where sh_from_k_id = :k_id and
        sh_id > 0
  order by sh_from_k_id, sh_id
  into sh_date, sh_is_lgota
  do
  begin
    days = datediff(day, sh_date_prv, sh_date);
    sh_perc = cast(balance / 100.0000 as double precision);
    sh_perc = cast(sh_perc * new_percent_per_day as double precision);
    sh_perc = cast(sh_perc * days as double precision);
    sh_perc = round(sh_perc, 0);
    sh_os = 0;

    if (sh_is_lgota = 0) then
      sh_os = annu - sh_perc;

    if (sh_date = last_date) then
      sh_os = balance;

    sh_date_prv = sh_date;
    balance = balance - sh_os;

    sh_all = sh_os + sh_perc;

    suspend;
  end
end
