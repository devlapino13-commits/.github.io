declare variable c_name_fio varchar(100);
declare variable kc_amount_approved curr_type;
declare variable oek_rd_id integer;
declare variable k_kontract_no varchar(50);
begin
  select k_kontract_no,
         extract(day from k_disb_date) || '.' || extract(month from k_disb_date) || '.' || extract(year from k_disb_date),
         k_percent_per_day
  from kontract
  where k_id = left(:aarep_k_id, 17) || 1
  into k_orig_kontract_no, k_disb_date, k_percent_per_day;

  select sum(zad_os), sum(zad_calc_perc), sum(zad_os_perc), sum(cz_cl_fine)
  from print_amnistia_cl(:aarep_k_id)
  into k_zad_os, k_zad_calc_perc, k_zad_os_perc, cz_fine;

  sum_all = k_zad_os_perc + cz_fine;

--Если сумма задолженности до 20 000 рублей, госпошлина = 4 % от суммы, но не менее 400 руб.;
  if (k_zad_os_perc < 20000) then
  begin
    perc = k_zad_os_perc / 100;
    formula = k_zad_os_perc * 4;
    perc = 0;
    if (formula < 400) then
      formula = 400;
  end

--Если сумма от 20 001 до 100 000 руб., госпошлина = 800 руб. + 3 % суммы, превышающей 20 000 руб.;
  if (k_zad_os_perc between 20001 and 100000) then
  begin
    gos_posh = 800;
    perc = (k_zad_os_perc - 20000) / 100 * 3;
    formula = perc + gos_posh;
  end

--Если сумма от 100 001 до 200 000 руб., госпошлина = 3 200 руб. + 2 % суммы, превышающей 100 000 руб.;
  if (k_zad_os_perc between 100001 and 200000) then
  begin
    gos_posh = 3200;
    perc = (k_zad_os_perc - 100000) / 100 * 2;
    formula = perc + gos_posh;
  end

--Если сумма от 200 001 до 1 000 000 руб., госпошлина = 5 200 руб. + 1 % суммы, превышающей 200 000 руб.;
  if (k_zad_os_perc between 200001 and 1000000) then
  begin
    gos_posh = 5200;
    perc = (k_zad_os_perc - 200000) / 100 * 1;
    formula = perc + gos_posh;
  end

--Если сумма свыше 1 000 000 руб., госпошлина = 13 200 руб. плюс 0,5 % суммы, превышающей 1 000 000 руб., но не более 60 000 руб.;
  if (k_zad_os_perc > 1000000) then
  begin
    gos_posh = 13200;
    perc = (k_zad_os_perc - 1000000) / 100 * 0.5;
    formula = perc + gos_posh;
  end

  if (k_zad_os_perc < 100000) then
    formula_new = 4000;

  if (k_zad_os_perc between 100001 and 300000) then
    formula_new = 4000 + 3 * ((k_zad_os_perc - 100000) / 100);

  if (k_zad_os_perc between 300001 and 500000) then
    formula_new = 10000 + 2.5 * ((k_zad_os_perc - 300000) / 100);

  if (k_zad_os_perc between 500001 and 1000000) then
    formula_new = 15000 + 2 * ((k_zad_os_perc - 500000) / 100);

  if (k_zad_os_perc between 1000001 and 3000000) then
    formula_new = 25000 + 1.5 * ((k_zad_os_perc - 1000000) / 100);

--При формировании заявления о вынесении судебного приказа госпошлина = 50 % размера, указанного в пунктах 1-5.
  formula_sud = (formula / 2); --/100;

  formula_sud_new = (formula_new / 2);

  full_payment_debt = '';
  for select k_kontract_no, c_amnistia_fio
      from print_amnistia_cl(:aarep_k_id)
      into :k_kontract_no, :c_name_fio
  do
  begin
    if ((exists(select *
                from kontract k
                inner join knclient kc on kc.kc_from_k_id = k.k_id
                inner join kontstate ks on ks.ks_from_k_id = k.k_id
                where coalesce(k.k_base_k_id, 0) > 0 --клиент амнистирован
                      and
                      ks.ks_state = 2 and
                      ks.ks_ol_is_delete = 0 --контракт закрыт
                      and
                      current_date between ks.ks_date_begin and ks.ks_date_end and
                      kc.kc_amount_approved > 0 and
                      k.k_kontract_no = coalesce(:k_kontract_no, ''))) and
        (not exists(select *
                    from operkroff oko
                    where oko.oko_from_k_id = (select k.k_id
                                               from kontract k
                                               where k.k_kontract_no = coalesce(:k_kontract_no, '')) --спецучет не должен входить
                          and
                          oko.oko_ol_is_active = 1 and
                          (coalesce(oko.oko_sh_os, 0) + coalesce(oko.oko_sh_perc, 0) + coalesce(oko.oko_sh_other, 0) + coalesce(oko.oko_os, 0) + coalesce(oko.oko_perc, 0) + coalesce(oko.oko_other, 0)) > 0)))

    then
      full_payment_debt = full_payment_debt || 'Заемщик  ' || c_name_fio || '  произвел(-a) оплату своей части долга в полном объеме в предусмотренные сроки. ' || ascii_char(13) || 'Истец претензий к нему не имеет.' || ascii_char(13) || '' || ascii_char(13) || '';
