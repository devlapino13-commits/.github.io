CREATE PROCEDURE HP_CORRECT_KOMMIS_MONTHLY_AZ
AS
declare variable d decimal(18,4) = 0;
declare variable a decimal(18,4);
declare variable s decimal(18,4);
declare variable i decimal(18,4);
declare variable k_id decimal(18,0);
declare variable n decimal(18,4);
declare variable amount decimal(18,4);
declare variable month_amount decimal(18,4);
declare variable total_amount decimal(18,4);
begin
--  if ((select cm.cm_database_id from company cm) <> 94401) then exit;
  if (f_get_db_id() <> 94401) then
    exit;

  for select distinct sh_from_k_id
      from schedule sh, kontstate ks
      where sh_from_k_id = ks_from_k_id and
            ks_state = 1 and
            ks_date_begin >= '01.03.2018' and
            ks.ks_ol_is_delete = 0 and
            sh_kommis_monthly <= 0 and
            sh.sh_id > 0
      into :k_id
  do
  begin
    select sum(sh_kommis), max(sh_id)
    from schedule
    where sh_from_k_id = :k_id
    into :amount, :n;
    if (amount is null) then
      amount = 0;
    if (n is null) then
      n = 0;

    update schedule
    set sh_kommis_monthly = 0.0
    where sh_from_k_id = :k_id;

    if (amount > 0) then
    begin
      s = 0;
      total_amount = 0;
      d = round((100.0 / n) / (n - 1.0), 2);
      a = round(100.0 / n + (100.0 / n) / 2.0, 2);
      i = 1;
      a = a + d;
      while (i <= n) do
      begin
        a = a - d;
        a = round(a, 2);
        if (i = n) then
        begin
          a = 100 - s;
          month_amount = amount - total_amount;
        end
        else
          month_amount = round(amount * a / 100, 2);
        total_amount = total_amount + month_amount;

        update schedule
        set sh_kommis_monthly = :month_amount
        where sh_from_k_id = :k_id and
              sh_id = :i;
--        Suspend;
        i = i + 1;
      end
    end
  end
end
