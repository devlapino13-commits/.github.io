CREATE PROCEDURE AAREP_GSM_REPORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    PEO_FROM_C_ID DECIMAL(18,0),
    PEO_FROM_PEN_ID INTEGER,
    PEO_KOLICH DECIMAL(18,4),
    PEO_LITER DECIMAL(18,4),
    PEO_SUMMA DECIMAL(18,4),
    PEO_FROM_REG_ID INTEGER,
    PEO_MATOTVET_FIO VARCHAR(100),
    PEO_PETROL_NAME VARCHAR(100),
    PEO_ACC_BALANCE VARCHAR(20),
    PEO_ACC_NUMBER INTEGER,
    PEO_LITER_IN_ONE DECIMAL(18,4),
    PEO_IS_SKLAD VARCHAR(20),
    PEO_TRANS_NAME VARCHAR(255),
    PEO_TRANS_LABEL VARCHAR(50),
    PEO_REGION_NAME VARCHAR(100),
    PEO_BRANCH_NAME VARCHAR(100))
AS
declare variable peo_from_pet_id integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  peo_is_sklad = 'B (В баке)';
  peo_kolich = 1;
  for select peo_from_c_id, peo_from_pen_id, peo_from_pet_id, sum(peo_trans_liter), sum(peo_trans_summa),
             peo_from_reg_id, r1.reg_code || ' ' || r1.reg_name, r2.reg_name
      from petopers, operlist, region_list(:report_region_id, 1, -1, 1), region r1, region r2
      where peo_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ol_date <= :report_date_end and
            (ol_from_reg_id = id_list) and
            ol_from_reg_id = r2.reg_unicode and
            peo_from_reg_id = r1.reg_unicode
      group by peo_from_c_id, peo_from_pen_id, peo_from_pet_id, peo_from_reg_id, r1.reg_code, r1.reg_name, r2.reg_name
      having sum(peo_trans_liter) <> 0 or sum(peo_trans_summa) <> 0
      order by peo_from_c_id
      into :peo_from_c_id, :peo_from_pen_id, :peo_from_pet_id, :peo_liter, :peo_summa, :peo_from_reg_id,
           :peo_region_name, :peo_branch_name
  do
  begin
    select c_name_fio
    from clients
    where c_id = :peo_from_c_id
    into :peo_matotvet_fio;

    select pen_name, pen_acc_balance, pen_acc_number, pen_liter_in_one
    from petname
    where pen_id = :peo_from_pen_id
    into :peo_petrol_name, :peo_acc_balance, :peo_acc_number, :peo_liter_in_one;

    select pet_name, pet_label
    from pettrans
    where pet_id = :peo_from_pet_id
    into :peo_trans_name, :peo_trans_label;

    suspend;
  end

  peo_trans_name = 'Склад';
  peo_trans_label = '';
  peo_is_sklad = 'A (На складе)';
  for select peo_from_c_id, peo_from_pen_id, sum(peo_kolich), sum(peo_liter), sum(peo_summa), peo_from_reg_id,
             r1.reg_code || ' ' || r1.reg_name, r2.reg_name
      from petopers, operlist, region_list(:report_region_id, 1, -1, 1), region r1, region r2
      where peo_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            ol_date <= :report_date_end and
            (ol_from_reg_id = id_list) and
            ol_from_reg_id = r2.reg_unicode and
            peo_from_reg_id = r1.reg_unicode
      group by peo_from_c_id, peo_from_pen_id, peo_from_reg_id, r1.reg_code, r1.reg_name, r2.reg_name
      having sum(peo_kolich) <> 0 or sum(peo_liter) <> 0 or sum(peo_summa) <> 0
      order by peo_from_c_id
      into :peo_from_c_id, :peo_from_pen_id, :peo_kolich, :peo_liter, :peo_summa, :peo_from_reg_id, :peo_region_name,
           :peo_branch_name
  do
  begin
    select c_name_fio
    from clients
    where c_id = :peo_from_c_id
    into :peo_matotvet_fio;

    select pen_name, pen_acc_balance, pen_acc_number, pen_liter_in_one
    from petname
    where pen_id = :peo_from_pen_id
    into :peo_petrol_name, :peo_acc_balance, :peo_acc_number, :peo_liter_in_one;

    suspend;
  end
end
