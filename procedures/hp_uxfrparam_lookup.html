CREATE PROCEDURE HP_UXFRPARAM_LOOKUP(
    FR_CODE D_STR_CODE,
    NEED_FULL_CODE D_BOOL)
RETURNS (
    ID D_STR_100,
    TEXT D_STR_255)
AS
declare variable ufp_is_param d_bool;
declare variable ufp_code d_str_100;
begin
  id = null;
  text = '';
  suspend;

  for select ufp_code, ufp_origin_name, ufp_name, ufp_is_param
  from uxfrparam
  where ufp_ufr_code = :fr_code and
        ufp_in_use = 1 and
        ufp_is_delete = 0 and
        ufp_state = 0
  order by ufp_is_param desc, ufp_index
  into ufp_code, id, text, ufp_is_param
  do
  begin
    if (text not containing id) then
      text = id || '   - ' || text;

    if (ufp_is_param = 1) then
      text = text || '  (param)';

    if (need_full_code = 1) then
      id = ufp_code;

    suspend;
  end
end
