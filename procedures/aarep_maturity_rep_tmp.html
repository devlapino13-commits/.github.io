CREATE PROCEDURE AAREP_MATURITY_REP_TMP(
    REP_DATE_END DATE)
RETURNS (
    REP_YEAR INTEGER,
    REP_MONTH INTEGER,
    REP_LINE INTEGER,
    SH_OS DECIMAL(18,4),
    LLC_PERCENT DECIMAL(18,7),
    KONT_LLC_AMOUNT DECIMAL(18,4),
    KONT_BALANCE DECIMAL(18,4),
    KONT_LLC_MINUS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    PR_PRODUCT_KIND INTEGER,
    PR_CODE VARCHAR(100),
    K_CIRCLE INTEGER)
AS
declare variable pr_set_rate_in_kont integer;
declare variable kont_curr_rate decimal(18,4);
declare variable kont_currency varchar(3);
declare variable k_id decimal(18,0);
declare variable sh_date date;
declare variable def_curr varchar(3);
declare variable actual_rate decimal(18,4);
begin
  select cm_default_currency
  from company
  into :def_curr;
  actual_rate = 1;

  for select ks_from_k_id, k_index_curr_code, k_index_curr_rate, pr_set_rate_in_kont, pr_product_kind, pr_code,
             k_circle
      from kontstate, kontract, product
      where ks_from_k_id = k_id and
            cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end >= :rep_date_end and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            k_from_pr_id = pr_id
      into :k_id, :kont_currency, kont_curr_rate, pr_set_rate_in_kont, pr_product_kind, pr_code, k_circle
  do
  begin
    kont_llc_minus = 0;
    sh_os = 0;
    sh_perc = 0;
    llc_percent = 0;
    select lk_percent
    from llc_main, llc_kont
    where lk_date <= :rep_date_end and
          lk_date_end >= :rep_date_end and
          lk_from_k_id = :k_id and
          lm_is_signed = 1 and
          lm_id = lk_from_lm_id
    order by lm_date desc
    rows 1
    into :llc_percent;

    if (kont_currency <> def_curr) then
    begin
      select cr_rate
      from opercurr
      where cr_from_cu_id = :kont_currency and
            cr_date <= :rep_date_end and
            cr_date_end >= :rep_date_end
      into :actual_rate;
      if (actual_rate is null) then
        actual_rate = 1;

      if ((pr_set_rate_in_kont = 1) and
          (actual_rate < kont_curr_rate)) then
        actual_rate = kont_curr_rate;
    end
    else
      actual_rate = 1;

    rep_line = -1;
    rep_year = extract(year from :rep_date_end);
    rep_month = extract(month from :rep_date_end);

    kont_balance = 0;
    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :kont_balance;
    if (kont_balance < 0) then
      kont_balance = 0;

    kont_balance = kont_balance * actual_rate;
    kont_llc_amount = cast(kont_balance / 100 * llc_percent as decimal(15,2)); -- откорректировано, было kont_llc_amount=Cast(kont_balance * llc_percent / 100.000  as decimal(15,2))
    suspend;

    kont_llc_amount = 0;
    kont_balance = 0;

    for select ok_sh_date, sum(ok_sh_os + ok_os), sum(ok_sh_perc + ok_perc)
        from operkred
        where ok_from_k_id = :k_id and
              ok_type < 7 and
              ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date_end and
              ok_from_c_id = 0
        group by ok_sh_date
        having(sum(ok_sh_os + ok_os) <> 0 or sum(ok_sh_perc + ok_perc) <> 0)
        into :sh_date, :sh_os, :sh_perc
    do
    begin
      if (actual_rate <> 1) then
      begin
        sh_os = sh_os * actual_rate;
      end
      if (sh_date <= rep_date_end) then
      begin
        rep_year = extract(year from (:rep_date_end + 1));
        rep_month = extract(month from (:rep_date_end + 1));
        rep_line = 0;
      end
      else
      begin
        rep_year = extract(year from sh_date);
        rep_month = extract(month from sh_date);
        rep_line = (rep_year * 12 + rep_month) - (extract(year from :rep_date_end) * 12 + extract(month from :rep_date_end));
      end
--      KONT_LLC_MINUS=Cast(sh_os*llc_percent/100 as decimal(15,2));
      kont_llc_minus = cast(sh_os as double precision) * llc_percent / 100;
      suspend;
    end
  end
end
