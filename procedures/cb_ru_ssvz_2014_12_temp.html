CREATE PROCEDURE CB_RU_SSVZ_2014_12_TEMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KOL_SVZ INTEGER,
    OSTATOK INTEGER,
    KOL_LUCH INTEGER,
    KOL_SRED INTEGER,
    KOL_HUDSH INTEGER,
    BONUS_LUCH CURR_TYPE,
    BONUS_SRED CURR_TYPE,
    BONUS_HUDSH CURR_TYPE,
    SREDNEVZESH_ZP CURR_TYPE,
    KOL_SVZ_FROM_KOEF INTEGER)
AS
declare variable aarep_month_plan_os decimal(18,4) = 0;
declare variable aarep_month_plan_perc decimal(18,4) = 0;
declare variable aarep_plan_all decimal(18,4);
declare variable aarep_fact_all decimal(18,4);
declare variable aarep_fact_planned decimal(18,4);
declare variable koef_bonus decimal(18,4) = 0;
declare variable koef_penalty decimal(18,4) = 0;
declare variable aarep1_plan_all decimal(18,4) = 0;
declare variable aarep1_fact_all decimal(18,4) = 0;
declare variable aarep1_fact_planned decimal(18,4) = 0;
declare variable aarep1_fact_other decimal(18,4) = 0;
declare variable c_id decimal(18,0);
declare variable c_email varchar(100);
declare variable reg_id integer;
declare variable sawt_id integer;
declare variable is_dekred integer;
declare variable k integer;
declare variable srednevzesh_zp_temp decimal(18,4);
declare variable aarep_port_balance_all decimal(18,4);
declare variable reg_name varchar(100);
declare variable c_name_fio varchar(100);
declare variable aarep_bonus_total decimal(18,4);
declare variable aarep_bonus_minus decimal(18,4);
declare variable aarep_bonus_plus decimal(18,4);
declare variable aarep_returnment_koef decimal(18,7);
declare variable aarep_date date;
declare variable is_stage4 smallint;
begin
  delete from temp_bonusru_lo;
  koef_bonus = 50.0; -----------
  koef_penalty = 110.0; -----------

  if (rep_date_end >= '01.11.2015') then -- добавлено от 03.12.2015
  begin
    koef_bonus = 55.0;
    koef_penalty = 170.0;
  end

  kol_svz = 0;
  kol_svz_from_koef = 0;
  for select ps_from_c_id, ps_from_reg_id, iif(dateadd(month, 4, ps_date_begin) < :rep_date_end -- проверка стажа
             or (select min(ps_date_begin)
                 from hr_pstatus, joblist, jobkind -- если на текущем должности менее 4 мес¤цев, то дополнительно провер¤ет стаж
                 where ps_from_c_id = c_id and
                       ps_status = 1 and
                       ps_is_working = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) < dateadd(month, -4, :rep_date_end), 1, 0)

      from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1)
      where --  :rep_c_id=0 and
            ps_from_c_id = c_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id > 1000000000003 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            ps_from_reg_id = id_list
      order by ps_from_reg_id
      into :c_id, :reg_id, :is_stage4
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into :reg_name;

    aarep_port_balance_all = 0;
    select sum(oks_balance)
    from oksummary
    where oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_kc_id = :c_id and
          oks_balance <> 0 and
          oks_close_state = 0
    into :aarep_port_balance_all;
    if (aarep_port_balance_all is null) then
      aarep_port_balance_all = 0;

    aarep_date = :rep_date_end;
    aarep_plan_all = 0;
    aarep_fact_all = 0;
    aarep_fact_planned = 0;

    select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned)
    from cb_ru_lo_201410_one_tmp(:reg_id, :c_id, null, :rep_date_end)
    into :aarep_plan_all, aarep_fact_all, aarep_fact_planned;

    if (aarep_plan_all is null) then
      aarep_plan_all = 0; -- —умма новых просрочек + выплаченные по графику за период  ---------------------------------
    if (aarep_fact_all is null) then
      aarep_fact_all = 0; -- —умма новых просрочек + выплаченные по графику за период  ---------------------------------
    if (aarep_fact_planned is null) then
      aarep_fact_planned = 0; -- выплаченные по графику за период                  ---------------------------------

    aarep_returnment_koef = 1.0;
    if (aarep_plan_all <> 0) then
      aarep_returnment_koef = round(aarep_fact_planned / aarep_plan_all, 4);
    else
      aarep_returnment_koef = 0.0;
    if (aarep_returnment_koef > 1.0) then
      aarep_returnment_koef = 1.0;

    aarep_bonus_plus = round((aarep_fact_all / 1000.0) * koef_bonus, 0); ------------------------------

    if (aarep_plan_all > aarep_fact_planned) ---------------------
    then
      aarep_bonus_minus = round(((aarep_plan_all - aarep_fact_planned) / 1000.0) * koef_penalty, 0); ---------------------
    else
      aarep_bonus_minus = 0; ---------------------

    aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus; --------------------------------

--    Вычисляет текущий статус сотрудника т.е. не в декретном отпуске ли
    is_dekred = 0;
    if (aarep_port_balance_all > 0) then
    begin
      if (is_stage4 = 1) then
      begin
        kol_svz = kol_svz + 1;
        insert into temp_bonusru_lo (bl_personal_id, bl_bonus_total)
        values (:c_id, :aarep_bonus_total);
      end

      kol_svz_from_koef = kol_svz_from_koef + 1;
    end

  end

--  kol_svz = 18; использовал дл¤ проверки
  kol_luch = 0;
  kol_sred = 0;
  kol_hudsh = 0;
  if (kol_svz >= 3) then
  begin
    ostatok = mod(kol_svz, 3);

    if (ostatok = 0) then
    begin
      kol_luch = kol_svz / 3;
      kol_sred = kol_luch;
      kol_hudsh = kol_luch;
    end
    else
    if (ostatok = 1) then
    begin
      kol_luch = kol_svz / 3;
      kol_sred = kol_luch;
      kol_hudsh = kol_luch + 1;
    end
    else
    if (ostatok = 2) then
    begin
      kol_luch = kol_svz / 3;
      kol_sred = kol_luch + 1;
      kol_hudsh = kol_luch + 1;
    end
  end

  srednevzesh_zp = 0;
  k = 0;
  srednevzesh_zp_temp = 0;

  bonus_luch = 0;
  bonus_sred = 0;
  bonus_hudsh = 0;

  for select bl_personal_id, bl_bonus_total
      from temp_bonusru_lo
      order by bl_bonus_total desc
      into :c_id, :aarep_bonus_total
  do
  begin
    if (kol_svz = 1) then
    begin
      srednevzesh_zp = :aarep_bonus_total;
      kol_luch = 1;
      bonus_luch = bonus_luch + aarep_bonus_total;
    end
    else
    if (kol_svz = 2) then
    begin
      kol_luch = 1;
      kol_hudsh = 1;

      if (srednevzesh_zp = 0) then
      begin
        srednevzesh_zp = aarep_bonus_total * 30 / 100;
        bonus_luch = bonus_luch + aarep_bonus_total;
      end
      else
      begin
        srednevzesh_zp = (srednevzesh_zp + aarep_bonus_total * 50 / 100) / 0.8;
        bonus_hudsh = bonus_hudsh + aarep_bonus_total;
      end
    end
    else
    begin
      k = k + 1;
      srednevzesh_zp_temp = srednevzesh_zp_temp + aarep_bonus_total;

      if (k = kol_luch) then
      begin
        bonus_luch = srednevzesh_zp_temp;
        srednevzesh_zp = srednevzesh_zp + (srednevzesh_zp_temp / kol_luch) * 20 / 100;
        srednevzesh_zp_temp = 0;
      end
      else
      if (k = kol_luch + kol_sred) then
      begin
        bonus_sred = srednevzesh_zp_temp;
        srednevzesh_zp = srednevzesh_zp + (srednevzesh_zp_temp / kol_sred) * 30 / 100;
        srednevzesh_zp_temp = 0;
      end
      else
      if (k = kol_luch + kol_sred + kol_hudsh) then
      begin
        bonus_hudsh = srednevzesh_zp_temp;
        srednevzesh_zp = srednevzesh_zp + (srednevzesh_zp_temp / kol_hudsh) * 50 / 100;
        srednevzesh_zp_temp = 0;
      end
    end

  end
--      if (srednevzesh_zp < 0) then srednevzesh_zp = 0; убран от 07.04.2015

  suspend;
end
