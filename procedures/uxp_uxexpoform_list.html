CREATE PROCEDURE UXP$UXEXPOFORM_LIST
RETURNS (
    EXF_LINE_ID D_STR_CODE,
    EXF_PARENT_ID D_STR_CODE,
    EXF_UFM_UUID D_UUID,
    EXF_UA_UUID D_UUID,
    EXF_CAPTION D_STR_100,
    EXF_CLONE_KIND D_SMALLINT,
    EXF_UX_UUID_EXIST D_BOOL,
    EXF_UX_FCODE_EXIST D_BOOL,
    EXF_WARNING D_SMALLINT,
    EXF_SELECTED D_BOOL,
    EXF_NEW_KIND D_SMALLINT,
    EXF_NEW_CODE D_STR_CODE,
    EXF_NEW_NAME D_STR_100,
    EXF_UX_UUID D_UUID,
    EXF_UX_KIND D_SMALLINT,
    EXF_UX_SCODE D_STR_CODE,
    EXF_UX_FCODE D_STR_100,
    EXF_UX_NAME D_STR_100,
    EXF_UX_RELEASE_UUID D_UUID,
    EXF_UX_DRAFT_UUID D_UUID,
    EXF_US_CODE D_STR_CODE,
    EXF_US_NAME D_STR_100,
    EXF_US_IS_TABLE D_BOOL,
    EXF_US_KIND D_SMALLINT,
    EXF_US_MEMO D_STR_8100,
    EXF_US_KEYFIELDS D_STR_100,
    EXF_US_AUTOINCFIELD D_STR_CODE,
    EXF_US_LOOKUP_ID D_STR_CODE,
    EXF_US_LOOKUP_TEXT D_STR_255,
    EXF_US_MAIN_TABLE D_STR_CODE,
    EXF_US_SQL D_STR_4096,
    EXF_UA_UXA_UUID D_UUID,
    EXF_UA_UXA_CODE D_STR_CODE,
    EXF_UA_UXA_FCODE D_STR_100,
    EXF_UA_UXA_NAME D_STR_100,
    EXF_UA_UXA_KIND D_SMALLINT,
    EXF_UA_IN_UI D_BOOL,
    EXF_CHILD_ACTION_COUNT D_INTEGER,
    EXF_USE_RELEASE D_BOOL,
    EXF_SHOW_CHILD_ACTION D_SMALLINT,
    EXF_UA_INDEX D_INTEGER,
    EXF_POSITION_CODE D_STR_CODE,
    EXF_IS_LOCKED D_BOOL,
    EXF_NEED_DELETE D_BOOL,
    EXF_UFM_UUID_CODE D_STR_CODE,
    EXF_UA_UUID_CODE D_STR_CODE,
    EXF_UX_UUID_CODE D_STR_CODE,
    EXF_UX_RELEASE_UUID_CODE D_STR_CODE,
    EXF_UX_DRAFT_UUID_CODE D_STR_CODE,
    EXF_UA_UXA_UUID_CODE D_STR_CODE,
    EXF_IMAGE_INDEX D_SMALLINT,
    EXF_DB_ID D_INTEGER,
    EXF_DB_IS_ACTUAL D_BOOL,
    EXF_DB_PATH D_STR_LONG)
AS
begin
  for select exf_line_id, exf_parent_id, exf_ufm_uuid, exf_ua_uuid, exf_caption, exf_clone_kind, exf_ux_uuid_exist,
             exf_ux_fcode_exist, exf_warning, exf_selected, exf_new_kind, exf_new_code, exf_new_name, exf_ux_uuid,
             exf_ux_kind, exf_ux_scode, exf_ux_fcode, exf_ux_name, exf_ux_release_uuid, exf_ux_draft_uuid, exf_us_code,
             exf_us_name, exf_us_is_table, exf_us_kind, exf_us_memo, exf_us_keyfields, exf_us_autoincfield,
             exf_us_lookup_id, exf_us_lookup_text, exf_us_main_table, exf_us_sql, exf_ua_uxa_uuid, exf_ua_uxa_code,
             exf_ua_uxa_fcode, exf_ua_uxa_name, exf_ua_uxa_kind, exf_ua_in_ui, exf_child_action_count, exf_use_release,
             exf_show_child_action, exf_ua_index, exf_position_code, exf_is_locked, exf_need_delete,
             exf_db_id, exf_db_is_actual, exf_db_path
  from ux$expoform
  order by exf_position_code
  into :exf_line_id, :exf_parent_id, :exf_ufm_uuid, :exf_ua_uuid, :exf_caption, :exf_clone_kind, :exf_ux_uuid_exist,
       :exf_ux_fcode_exist, :exf_warning, :exf_selected, :exf_new_kind, :exf_new_code, :exf_new_name, :exf_ux_uuid,
       :exf_ux_kind, :exf_ux_scode, :exf_ux_fcode, :exf_ux_name, :exf_ux_release_uuid, :exf_ux_draft_uuid,
       :exf_us_code, :exf_us_name, :exf_us_is_table, :exf_us_kind, :exf_us_memo, :exf_us_keyfields,
       :exf_us_autoincfield, :exf_us_lookup_id, :exf_us_lookup_text, :exf_us_main_table, :exf_us_sql,
       :exf_ua_uxa_uuid, :exf_ua_uxa_code, :exf_ua_uxa_fcode, :exf_ua_uxa_name, :exf_ua_uxa_kind, :exf_ua_in_ui,
       :exf_child_action_count, :exf_use_release, :exf_show_child_action, :exf_ua_index, :exf_position_code,
       :exf_is_locked, :exf_need_delete,
       :exf_db_id, :exf_db_is_actual, :exf_db_path
  do
  begin
    exf_ufm_uuid_code = uuid_to_char(exf_ufm_uuid);
    exf_ua_uuid_code = uuid_to_char(exf_ua_uuid);
    exf_ux_uuid_code = uuid_to_char(exf_ux_uuid);
    exf_ux_release_uuid_code = uuid_to_char(exf_ux_release_uuid);
    exf_ux_draft_uuid_code = uuid_to_char(exf_ux_draft_uuid);
    exf_ua_uxa_uuid_code = uuid_to_char(exf_ua_uxa_uuid);

    exf_image_index = -1;
    exf_image_index = case exf_new_kind
      when 0 then 5
      when 1 then 62
      when 2 then 63
    end;

    suspend;
  end
end
