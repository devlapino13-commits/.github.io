CREATE PROCEDURE UXP$TRANSLATE_UPDATE(
    TRL_FROM_TRM_ID D_ID,
    TRL_LANG_CODE D_STR_10,
    TRL_TRANSLATE_TEXT D_STR_255,
    TRL_ACTUAL_TEXT D_STR_255,
    TRL_CHECKED D_BOOL = null)
AS
begin
  if (trim(coalesce(trl_translate_text, '')) = '' and
      trim(coalesce(trl_actual_text, '')) = '') then
  begin
    delete from ux$translate
    where trl_from_trm_id = :trl_from_trm_id and
          trl_lang_code = :trl_lang_code;
  end else
  begin
    if (not exists(select * from ux$transmain
                   where trm_id = :trl_from_trm_id)) then
      exit;

    if (trl_lang_code = 'uz' and not(uxf$is_rus(trl_translate_text))) then
      trl_lang_code = 'uzl';

    update or insert into ux$translate (trl_from_trm_id, trl_lang_code, trl_translate_text, trl_actual_text, trl_checked)
    values (:trl_from_trm_id, :trl_lang_code, :trl_translate_text, :trl_actual_text, :trl_checked)
    matching (trl_from_trm_id, trl_lang_code);
  end
end
