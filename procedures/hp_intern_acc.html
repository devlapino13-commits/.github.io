CREATE PROCEDURE HP_INTERN_ACC(
    I_ID D_INTEGER)
RETURNS (
    IS_ERR D_SMALLINT)
AS
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable jl_id d_integer;
declare variable jl_name d_str_100;
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable summ d_integer;
declare variable sd_memo d_str_long;
declare variable ssh_id d_integer;
declare variable sd_id d_integer;
declare variable sql_text d_str_long;
declare variable reg_name_fix d_str_code;
declare variable sd_acc_main d_str_20;
begin
  is_err = 1;

  select i.i_from_reg_id, i.i_from_c_id, jl.jl_id, jl.jl_name, i.i_date_begin, i.i_date_end
  from interns i
  inner join joblist jl on i.i_from_jl_id = jl.jl_id
  where i.i_id = :i_id
  into :reg_id, :c_id, :jl_id, :jl_name, :date_begin, :date_end;

  summ = 0;
  sd_memo = '0 (Ноль) сом 00 тыйын';

  if (:jl_id = 270) then
  begin
    summ = 1250;
    sd_memo = '1250 (Тысяча двести пятьдесят) сом 00 тыйын';
  end
  if (:jl_id = 271) then
  begin
    summ = 625;
    sd_memo = '625 (Шестьсот двадцать пять) сом 00 тыйын';
  end

  select replace(reg1.reg_name, ' р.', '')
  from region reg
  inner join region reg1 on reg.parent = reg1.reg_unicode
  where reg.reg_unicode = :reg_id
  into :reg_name_fix;

  if (:reg_name_fix <> 'Головной офис') then
  begin
    sql_text = 'select al_id
              from acc_list
              where al_from_reg_id = 1060100 and
                  al_from_bg_id = ''21199'' and
                  al_is_active = 1 and
                  al_name like ''%стажер%'' and
                  al_name like ''%' || :reg_name_fix || '%'' and
                  is_delete = 0';

    execute statement sql_text
        into :sd_acc_main;
  end
  else
    sd_acc_main = '060141721199000042';

  insert into servicedog (sd_from_reg_id, sd_from_c_id, sd_category, sd_memo, sd_is_indiv, sd_date_begin, sd_date_end,
                          sd_once_only_pay, sd_acc_main, sd_acc_advance, sd_acc_expense, sd_need_patent, sd_need_spolis,
                          sd_only_workdays, sd_user_id, sd_datetime, sd_status, sd_kind)
  values (:reg_id, :c_id, :jl_name, :sd_memo, 1, :date_begin, :date_end, 0, :sd_acc_main, '060141711501000112',
          '060141780316000003', 1, 1, 0, 1000000000000, current_timestamp, 0, 0)
  returning sd_id
  into :sd_id;

  update servicedog
  set sd_dogovor_no = cast(sd_id as d_str_20)
  where (sd_id = :sd_id);

  insert into servicesh (ssh_from_sd_id, ssh_date, ssh_amount_dogovor, ssh_date_regist)
  values (:sd_id, :date_end, :summ, :date_begin)
  returning ssh_id
  into :ssh_id;

  update intern_committee_result
  set icr_from_sd_id = :sd_id
  where (icr_from_i_id = :i_id);

  if (:sd_id is not null and
      :ssh_id is not null) then
    is_err = 0;

  suspend;
end
