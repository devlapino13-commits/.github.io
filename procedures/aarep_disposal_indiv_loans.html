CREATE PROCEDURE AAREP_DISPOSAL_INDIV_LOANS(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    COUNTER INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    CLIENT_FIO VARCHAR(1000),
    K_DISB_DATE DATE,
    KC_AMOUNT_APPROVED DECIMAL(18,4))
AS
begin
  counter = 0;
  for select kont.k_kontract_no, cl.c_name_fio as client_fio, k_disb_date, kncl.kc_amount_approved
      from kontract kont, knclient kncl, clients cl, kontstate ks, region_list(:rep_region_id, 1, -1, 1) rl
      where kncl.kc_from_k_id = kont.k_id and
            kncl.kc_amount_approved > 0 and
            cl.c_id = kncl.kc_from_c_id and
            ks.ks_from_k_id = kont.k_id and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 0 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            kont.k_disb_date = :rep_date and
            kont.k_last_reg_id = rl.id_list and
            kont.k_status = 1 and
            kont.k_from_pr_id = 45 /*только продукт Партнер Ок*/
      order by k_kontract_no, c_name_fio
      rows 10
      into :k_kontract_no, :client_fio, :k_disb_date, :kc_amount_approved
  do
  begin
    counter = counter + 1;
    suspend;
  end
end
