CREATE PROCEDURE HP_GET_MA_KC_AMOUNT(
    MA_ID D_ID NOT NULL,
    C_ID D_ID NOT NULL)
RETURNS (
    MA_KC_AMOUNT D_CURRENCY)
AS
declare variable ma_date_oper d_date;
declare variable ma_amount d_currency;
declare variable kc_amount_approved d_currency;
begin
  -- возвращает сумму заявки + уже выданные клиенту суммы по другим займам в тот же планиуремый день выдачи текущей заявки

  ma_kc_amount = 0;
  ma_amount = 0;
  kc_amount_approved = 0;

  select ma_summa_minus,
         ma_date_oper
  from miniapp
  where ma_id = :ma_id and
        ma_is_active = 1 and
        ma_date_end = '31.12.9999'
  into ma_amount,
       ma_date_oper;

  select sum(kc_amount_approved)
  from kontstate
  inner join knclient on kc_from_k_id = ks_from_k_id
  inner join kontract on k_id = ks_from_k_id
  where kc_from_c_id = :c_id and
        :ma_date_oper between ks_date_begin and ks_date_end and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        kc_amount_approved > 0 and
        k_disb_date = :ma_date_oper
  into kc_amount_approved;

  ma_kc_amount = ma_amount + coalesce(kc_amount_approved, 0);

  suspend;
end
