CREATE PROCEDURE CB_KG_KM_2021_04(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    DELINQ_BALANCE_7 CURR_TYPE,
    BALANCE_BEGIN CURR_TYPE,
    ZNACH_PAR7 CURR_TYPE,
    BONUS_FACTOR CURR_TYPE,
    KOL_SSVZ INTEGER,
    FIO_SSVZ VARCHAR(1024),
    REG_NAME VARCHAR(100),
    BONUS_KM INTEGER,
    BONUS_KM_COMMENT CURR_TYPE,
    SUM_VOZNOG_OFFICE CURR_TYPE,
    REPAY_SUM_NOT_OS CURR_TYPE,
    DELINQ_OS_PERIOD CURR_TYPE,
    REPAY_KOMMIS_IPF CURR_TYPE,
    CONSULT_SUMMA CURR_TYPE,
    PLAN_SPF INTEGER,
    FACT_SPF INTEGER,
    AAREP_BONUS_MEMO VARCHAR(50))
AS
declare variable rep_date_begin date;
declare variable reg_id integer;
declare variable rep_period integer;
declare variable diff_spf integer;
declare variable c_id id_type;
declare variable bonus_koef curr_type;
declare variable bonus_sum curr_type;
declare variable bonus_memo varchar(255);
begin

  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  rep_period = extract(year from rep_date_end) || right((extract(month from rep_date_end) + 100), 2);

  for select oks.oks_region_id, sum(iif(:rep_date_end - oks.oks_delinq_date > 7, oks.oks_balance, 0))
      from oksummary oks
      inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(oks.oks_region_id as d_integer)
      where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_balance > 0 and
            oks.oks_close_state = 0 and
            not exists(select *
                       from hr_pstatus hps
                       where hps.ps_is_working = 1 and
                             hps.ps_from_c_id = oks_kc_id and
                             hps.ps_from_jl_id = 140)
      group by oks.oks_region_id
      into reg_id, delinq_balance_7
  do
  begin
    --  балан кп и баланс par7 на начало
    balance_begin = null;
    select sum(oks_balance)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_begin - 1 between oks_date_begin and oks_date_end and
          oks_close_state = 0 and
          not exists(select *
                     from hr_pstatus hps
                     where hps.ps_is_working = 1 and
                           hps.ps_from_c_id = oks_kc_id and
                           hps.ps_from_jl_id = 140)
    into balance_begin;
    balance_begin = coalesce(balance_begin, 0);
    --  значение par7* в % от кп
    if (balance_begin = 0 and
        delinq_balance_7 > 0) then
      znach_par7 = 100;
    else
    if (delinq_balance_7 = 0) then
      znach_par7 = 0;
    else
      znach_par7 = cast(delinq_balance_7 as double precision) / balance_begin * 100;
    znach_par7 = round(znach_par7, 2);
    --  суммы погашении процентов пеня и админ сбор
    repay_sum_not_os = null;
    select -sum(ok_perc + ok_fine + ok_kommis)
    from operkred op
    inner join kontract kt on kt.k_id = op.ok_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id
    where ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_ol_reg_id = :reg_id and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          kal.is_delete = 0 and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          not exists(select *
                     from hr_pstatus hps
                     where hps.ps_is_working = 1 and
                           hps.ps_from_jl_id = 140 and
                           hps.ps_from_c_id = kal.kal_new_kf_officer)
    into repay_sum_not_os;
    repay_sum_not_os = coalesce(repay_sum_not_os, 0);

    --  комиссия за выдачу ипф
    repay_kommis_ipf = null;
    select sum(ok_amount)
    from refdata rd
    inner join operkontract okt on okt.ok_operation_type = rd.rd_id
    inner join operlist  ol on ol.ol_id = okt.ok_from_ol_id
    inner join kontract kon on kon.k_id = okt.ok_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = kon.k_from_kf_id
    where rd_code = 'cashtransactionincom.ipfcommission' and
          ol.is_delete = 0 and
          ol.ol_from_reg_id = :reg_id and
          ol.ol_date between :rep_date_begin and :rep_date_end and
          kal.is_delete = 0 and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          not exists(select *
                     from hr_pstatus hps
                     where hps.ps_is_working = 1 and
                           hps.ps_from_jl_id = 140 and
                           hps.ps_from_c_id = kal.kal_new_kf_officer)
    into repay_kommis_ipf;
    repay_kommis_ipf = coalesce(repay_kommis_ipf, 0);

    --оплата за консультационные услуги
    consult_summa = 0;
    select sum(ok.ok_amount)
    from operkontract ok
    inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
    inner join refdata rd on rd.rd_id = ok_operation_type
    inner join kontract kon on kon.k_id = ok.ok_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = kon.k_from_kf_id
    where rd_code = 'cashtransactionincom.consultation' and
          ol.is_delete = 0 and
          ol_from_reg_id = :reg_id and
          ol_date between :rep_date_begin and :rep_date_end and
          kal.is_delete = 0 and
          :rep_date_end between kal.kal_date and kal.kal_date_end and
          not exists(select *
                     from hr_pstatus hps
                     where hps.ps_is_working = 1 and
                           hps.ps_from_jl_id = 140 and
                           hps.ps_from_c_id = kal.kal_new_kf_officer)
    into :consult_summa;
    consult_summa = coalesce(consult_summa, 0);

    --  сумма просроченной задолженности по ос за отчетный период
    delinq_os_period = null;
    select sum(ok_os + ok_sh_os)
    from oksummary, operkred
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          :rep_date_end >= oks_delinq_date and
          oks_from_k_id = ok_from_k_id and
          oks_close_state = 0 and
          ok_ol_is_active = 1 and
          ok_type < 7 and
          ok_sh_date between :rep_date_begin and :rep_date_end and
          ok_ol_date <= :rep_date_end and
          not exists(select *
                     from hr_pstatus hps
                     where hps.ps_is_working = 1 and
                           hps.ps_from_c_id = oks_kc_id and
                           hps.ps_from_jl_id = 140)
    into delinq_os_period;
    delinq_os_period = coalesce(delinq_os_period, 0);
    --  сумма вознаграждения подотчетного офиса
    sum_voznog_office = (repay_sum_not_os + repay_kommis_ipf + consult_summa) * 9 / 100 - delinq_os_period * 40 / 100;

    --  бонус фактор
    if (rep_date_end <= '31.03.2019') then
    begin
      bonus_factor = case
                       when znach_par7 = 0.0 then 1.20
                       when znach_par7 <= 0.5 then 1.00
                       when znach_par7 <= 1.0 then 0.90
                       when znach_par7 <= 1.5 then 0.50
                       else 0
                     end;
    end
    else
    begin
      bonus_factor = case
                       when znach_par7 <= 0.2 then 1.20
                       when znach_par7 <= 0.6 then 1.00
                       when znach_par7 <= 0.9 then 0.90
                       when znach_par7 <= 1.3 then 0.70
                       when znach_par7 <= 1.7 then 0.50
                       when znach_par7 <= 2.1 then 0.35
                       when znach_par7 <= 2.5 then 0.20
                       else 0
                     end;
    end
    kol_ssvz = 0;
    fio_ssvz = '';

    for select c_id, count(*), list(c_name_fio, '; ')
        from hr_pstatus, joblist, jobkind, clients
        where ps_status = 1 and
              ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code = 'jobsystem.region_km' and
              :rep_date_end between ps_date_begin and ps_date_end and
              ps_from_reg_id = :reg_id and
              ps_from_c_id = c_id and
              (ps_date_begin <= dateadd(month, -3, :rep_date_end) or (select min(hr1.ps_date_begin)
                                                                      from hr_pstatus hr1
                                                                      where hr1.ps_is_working = 1 and
                                                                            hr1.ps_status = 1 and
                                                                            hr1.ps_from_c_id = c_id) <= dateadd(month, -3, :rep_date_end)) and
              coalesce((select sawt_id
                        from hr_voctravel, hr_orders, sa_worktype
                        where hrv_from_ho_id = ho_id and
                              ho_order_type = sawt_id and
                              hrv_from_c_id = ps_from_c_id and
                              :rep_date_end between hrv_date_begin and hrv_date_end_fact
                        order by hrv_date_begin desc rows 1), 0) not in (16, 17, 18, 8) -- декретников не включает
        group by c_id
        into c_id, kol_ssvz, fio_ssvz

    do
    begin
      if (kol_ssvz is null) then
        kol_ssvz = 0;

    end

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    --  план спф
    plan_spf = null;
    select prs_plan_spf
    from planregionspf
    where prs_period = :rep_period and
          prs_kind = 1 and
          prs_region = :reg_id
    into plan_spf;
    plan_spf = coalesce(plan_spf, 0);

    /*    --  факт спф
    fact_spf = null;
    select count(distinct oks_kc_id)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0
    into fact_spf;
    fact_spf = coalesce(fact_spf, 0);   */

    fact_spf = 0;
    select count(*)
    from aarep_personal_list(:reg_id, :rep_date_end, 1)
    where jl_name in ('специалист по финансированию', 'старший специалист по финансированию') and
          ps_curr_status not in ('отпуск без содерж.по уходу за ребенком', 'отпуск по беременности и родам', 'трудовой отпуск декретницы')
    into fact_spf;
    fact_spf = coalesce(fact_spf, 1);

    diff_spf = plan_spf - fact_spf;
    if (diff_spf < 0) then
      diff_spf = 0;

    bonus_km = sum_voznog_office * 0.15 * bonus_factor;

    if (bonus_km < 0) then
      bonus_km = 0;

    --  штраф 10% с  бонуса к начислению за каждого недостающего спф относительно планового показателя количества спф на конец месяца.
    if (bonus_km > 0 and
        diff_spf > 0) then
    begin
      if (diff_spf > 10) then
        diff_spf = 10;
      if (rep_period <> 202004) then
        bonus_km = bonus_km - bonus_km * diff_spf * 10 / 100;
    end

    bonus_km_comment = 0;
    bonus_memo = '';
    bonus_koef = 0;
    bonus_sum = 0;

    select bonus_koef, bonus_sum, bonus_memo
    from hp_get_vigovor_koef(:c_id, :rep_date_end)
    where bonus_memo <> 'нет замечаний и выговоров'
    into bonus_koef, bonus_sum, bonus_memo;

    if (bonus_km > 0 and
        bonus_memo <> '') then


    begin
      bonus_km_comment = bonus_km * bonus_koef / 100 - bonus_sum;
      if (bonus_km_comment < 0) then
        bonus_km_comment = 0;
    end
    aarep_bonus_memo = bonus_memo;
    if (bonus_memo = '') then
      bonus_km_comment = bonus_km;
    suspend;
  end
end
