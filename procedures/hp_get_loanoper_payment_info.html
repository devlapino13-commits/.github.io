CREATE PROCEDURE HP_GET_LOANOPER_PAYMENT_INFO(
    OL_ID D_ID,
    K_ID D_ID)
RETURNS (
    RES_PAYMENT_KIND D_SMALLINT,
    RES_LOAN_OL_ID D_ID,
    RES_MAIN_OL_ID D_ID,
    RES_LOAN_AMOUNT D_CURRENCY,
    RES_BOOK_AMOUNT D_CURRENCY,
    RES_DT_ACCOUNT D_STR_20,
    RES_LOAN_OFFICE D_INTEGER,
    RES_FACT_OFFICE D_INTEGER,
    RES_KKM_ID D_ID,
    RES_OPER_CODE D_STR_CODE)
AS
declare variable tmp_ol_id d_id;
declare variable ol_doc_kind d_smallint;
declare variable tmp_amount d_currency;
declare variable max_acc d_str_20;
declare variable ol_oper_type d_smallint;
declare variable ol_oper_subtype d_smallint;
declare variable ol_doc_kind_main d_smallint;
begin
  if (ol_id is null) then
    exception e_message 'Id операции не указан';

  if (k_id is null) then
    exception e_message 'Id договора не указана';

  res_payment_kind = 0;
  res_loan_amount = 0;
  res_book_amount = 0;
  res_dt_account = '';

  select coalesce(ol_parent_id, ol_id)
  from operlist
  where ol_id = :ol_id
  into ol_id;

  for select ol_id
  from operlist
  where ol_id = :ol_id
  union
  select ol_id
  from operlist
  where ol_parent_id = :ol_id
  into tmp_ol_id
  do
  begin
    if (exists(select * from operkred
    where ok_from_ol_id = :tmp_ol_id and
          ok_from_k_id = :k_id)) then
      res_loan_ol_id = tmp_ol_id;
    else
    if (exists(select * from operkroff
    where oko_from_ol_id = :tmp_ol_id and
          oko_from_k_id = :k_id)) then
      res_loan_ol_id = tmp_ol_id;
    else
    if (exists(select * from operextra
    where ox_from_ol_id = :tmp_ol_id and
          ox_object_kind = 9 and
          ox_object_id = :k_id)) then
      res_loan_ol_id = tmp_ol_id;
    else
    if (exists(select * from kontdata
    where kd_from_ol_id = :tmp_ol_id and
          kd_from_k_id = :k_id)) then
      res_loan_ol_id = tmp_ol_id;

    if (res_loan_ol_id is not null) then
      break;
  end

  if (res_loan_ol_id is null) then
    exit;

  select ol_parent_id, ol_from_reg_id, ol_doc_kind, ol_oper_type, ol_oper_subtype
  from operlist
  where ol_id = :res_loan_ol_id
  into res_main_ol_id, res_loan_office, ol_doc_kind, ol_oper_type, ol_oper_subtype;

  ol_doc_kind_main = ol_doc_kind;

  tmp_ol_id = coalesce(res_main_ol_id, res_loan_ol_id);

  if (exists(select * from extsysoper, extsysmain
      where eso_from_ol_id = :tmp_ol_id and
            eso_from_esm_id = esm_id and
            esm_from_k_id = :k_id)) then
  begin
    res_payment_kind = 3; -- через платежные системы
  end else
  begin
    if (res_main_ol_id is not null) then
    begin
      select ol_doc_kind
      from operlist
      where ol_id = :res_main_ol_id
      into ol_doc_kind_main;
    end
  
    if (ol_doc_kind = 1 or ol_doc_kind_main = 1) then -- Приходный ордер
      res_payment_kind = 1; -- через кассу
    else
    if (ol_doc_kind = 7 or ol_doc_kind_main = 7) then -- Банковские поступления
      res_payment_kind = 2; -- через банк
  end

  res_oper_code = '';
  select res_code
  from hp_operdel_operkind(:res_loan_ol_id, :ol_oper_type, :ol_oper_subtype)
  into res_oper_code;

  if (res_main_ol_id is null) then
  begin
    res_main_ol_id = res_loan_ol_id;
    res_fact_office = res_loan_office;
  end
  else
  begin
    select ol_from_reg_id
    from operlist
    where ol_id = :res_main_ol_id
    into res_fact_office;
  end

  -- здесь код для получения kkm_id

  select kkm_id
  from kkt_main
  where kkm_from_ol_id = :res_main_ol_id and
        kkm_from_k_id = :k_id and
        kkm_ol_is_delete = 0 and
        kkm_duplicated_kkm_id is null
  into res_kkm_id;

  for select -sum(ok_os + ok_perc + ok_fine + ok_tax + ok_kommis + ok_kommis_adv)
  from operkred
  where ok_from_ol_id = :res_loan_ol_id and
        ok_from_k_id = :k_id and
        ok_type in (4, 9)
  into tmp_amount
  do
  begin
    if (tmp_amount is null) then
      tmp_amount = 0;
    res_loan_amount = res_loan_amount + tmp_amount;
  end

  for select -sum(oko_os + oko_perc + oko_other)
  from operkroff
  where oko_from_ol_id = :res_loan_ol_id and
        oko_from_k_id = :k_id and
        oko_type in (4, 9)
  into tmp_amount
  do
  begin
    if (tmp_amount is null) then
      tmp_amount = 0;

    res_loan_amount = res_loan_amount + tmp_amount;
  end

  for select sum(ox_amount_minus)
  from operextra
  where ox_from_ol_id = :res_loan_ol_id and
        ox_object_kind = 9 and
        ox_object_id = :k_id
  into tmp_amount
  do
  begin
    if (tmp_amount is null) then
      tmp_amount = 0;

    res_loan_amount = res_loan_amount + tmp_amount;
  end

  if (res_loan_amount is null) then
    res_loan_amount = 0;

  select min(ob_account_dt), max(ob_account_dt), sum(ob_summa)
  from operbook
  where ob_from_ol_id = :res_main_ol_id
  into res_dt_account, max_acc, res_book_amount;

  if (res_book_amount is null) then
    res_book_amount = 0;

  if (res_dt_account is null) then
    res_dt_account = '';

  if (res_dt_account <> max_acc) then
    res_dt_account = '';

  suspend;
end
