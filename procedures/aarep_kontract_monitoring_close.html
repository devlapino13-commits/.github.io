CREATE PROCEDURE AAREP_KONTRACT_MONITORING_CLOSE(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    OFFICE_NAME D_STR_100,
    REG_NAME VARCHAR(50),
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    K_CLOSE_DATE D_DATE,
    DAYS_AFTER_DISB D_INTEGER,
    C_NAME D_STR_100,
    KS_NAME D_STR_100,
    MONITORING_STATUS D_STR_20,
    MONITORING_MEMO D_STR_255,
    DESCR_OF_LOAN D_STR_2048)
AS
declare variable ks_id d_id;
declare variable office_id d_integer;
declare variable c_id d_id;
declare variable analiz_date d_date;
declare variable rd_id d_id;
declare variable is_ipf d_bool;
declare variable dt_find d_date;
declare variable rd_memo_id d_integer;
declare variable reg_id d_integer;
declare variable pr_id d_id;
declare variable k_id d_id;
declare variable date_x d_date;
declare variable obj_kind d_smallint;
declare variable obj_id d_id;
declare variable is_express d_bool;
begin
  date_x = '03.03.2022'; --Дата перехода на knclient
  rd_id = f_get_rd_id('Anketa.TargetUseLoan');
  rd_memo_id = f_get_rd_id('Monitoring.Memo');

  for select datediff(day from ks.ks_date_begin to current_date), --Количество дней с дня выдачи
             ks.ks_kc_id, ks.ks_region_id,
             kc.kc_from_c_id,
             k.k_kontract_no,
             k.k_disb_date,
             k.k_from_pr_id,
             k.k_fact_closed_date,
             iif(k.k_fact_disb_date >= :date_x, kc.kc_id, kc_from_c_id),
             iif(k.k_fact_disb_date >= :date_x, f_objkind('KnClient', 'id'), f_objkind('Clients', 'id'))
      from kontstate ks
      inner join kontract k on k.k_id = ks.ks_from_k_id
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
            kc.kc_amount_approved > 0
      inner join region_list(:rep_region_id, 1, 3, 1) on id_list = ks.ks_region_id
      where ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 2 and
            ks.ks_kc_id <> 1000000000005 -- Пропускает КС Виртуальный сотрудник ОРИВ (по просьбе Чолпон)
      into :days_after_disb,
           :ks_id,
           :office_id,
           :c_id,
           :k_kontract_no,
           :k_disb_date,
           :pr_id,
           :k_close_date,
           :obj_id,
           :obj_kind
  do
  begin
    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into :c_name;

    select iif(pr.pr_code containing '.ipf.', 1, 0), --Проверка ИПФ продукт или стандарт
           iif(pr.pr_code containing '.express_online.', 1, 0) --Проверка на экспресс выдачу
    from product pr
    where pr.pr_id = :pr_id
    into :is_ipf, :is_express;

    select c.c_name_fio
    from clients c
    where c.c_id = :ks_id
    into ks_name;

    select reg.reg_name, reg.parent
    from region reg
    where reg.reg_unicode = :office_id
    into office_name, :reg_id;

    select reg.reg_name
    from region reg
    where reg.reg_unicode = :reg_id
    into :reg_name;

    analiz_date = null;

    if (is_express = 1) then
      monitoring_status = 'Экспресс выдача';

    else
    begin
      if (k_disb_date < '14.02.2022') then
        dt_find = k_disb_date + iif(k_disb_date >= '22.12.2021' and is_ipf = 1, 1, 8);
      else
        dt_find = k_disb_date + 1;

      select acl_last_datetime
      from actionlist
      where acl_object_kind = :obj_kind and
            acl_object_id = :obj_id and
            acl_from_rd_id = :rd_id and
            acl_date = :dt_find and
            acl_status = 2 rows 1
      into :analiz_date;

      monitoring_memo = null; --Комментарий СПФ

      select ovr_value
      from objectvalres, objectprops, refdata, objectvalues
      where opr_object_kind = :obj_kind and
            opr_object_id = :obj_id and
            opr_subject_kind = -1 and
            opr_subject_id = :rd_memo_id and
            ovr_from_opr_id = opr_id and
            ovr_from_ovs_id = ovs_id and
            ovs_from_opr_id = opr_id and
            ovs_date_begin <= :analiz_date and
            ovs_date_end >= :analiz_date and
            ovr_result_rd_id = rd_id
      into :monitoring_memo;

      monitoring_memo = coalesce(trim(:monitoring_memo), '');

      if (monitoring_memo is null or monitoring_memo = 'значение' or monitoring_memo = 'нет' or monitoring_memo = '') then
        monitoring_status = 'Не заполнен';
      else
        monitoring_status = 'заполнен';
    end

    select coalesce(res_str_values, '')
    from hp_objectprops_anketa_get('Clients', :c_id, -1, 'AnketaQuestions.descr_of_loan', :k_disb_date, null)
    into :descr_of_loan;

    suspend;
  end
end
