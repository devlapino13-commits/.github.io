CREATE PROCEDURE GET_REF_ACCOUNT_AGREEMENTS(
    SAB_FULL_NAME D_STR_255,
    SAB_INVERT_NAME D_STR_255,
    AG_ID D_INTEGER,
    KC_NON_RESIDENT D_SMALLINT,
    REP_DATE D_DATE)
RETURNS (
    ACC_ID D_ID)
AS
declare variable reg_id d_integer = null;
declare variable kc_client_type d_smallint = 0;
begin
  if (ag_id is null) then
    exception e_default 'Agreements ID is emptry';

  if (rep_date is null) then
    rep_date = current_date;

  kc_client_type = 3;

  select ag_from_reg_id
  from agreements
  where ag_id = :ag_id
  into reg_id;

  select acc_id
  from get_ref_account_any(:sab_full_name, :sab_invert_name, 'agreements', :ag_id, '', :reg_id, -1, :kc_non_resident, :kc_client_type, :rep_date)
  into :acc_id;
  suspend;
end
