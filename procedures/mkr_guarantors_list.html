CREATE PROCEDURE MKR_GUARANTORS_LIST(
    K_ID ID_TYPE)
RETURNS (
    ID STR_100,
    NAME STR_100,
    COUNTRYCODE VARCHAR(3),
    DATEOFBIRTH DATE,
    PLACEOFBIRTH STR_100,
    PINCODE VARCHAR(20))
AS
begin
  placeofbirth = 'Azerbaycan';
  for select coalesce(cl.c_pass_no, '') as id, coalesce(cl.c_name_fio, '') as name,
             coalesce(iif(pk_country = 'AZN', 'AZE', upper(pk_country)), '') as countrycode,
             cl.c_birth_date as dateofbirth, trim(coalesce(cl.c_pincode, '')) as pincode -- #pincode
--coalesce(iif (cl.c_pass_no starting 'AZE', cl.c_pincode, ''), '') as PinCode  #pincode
      from zalog zal
      inner join clients cl on cl.c_id = zal.z_from_c_id
      inner join passkind p on p.pk_id = cl.c_doc_type
      where zal.z_from_k_id = :k_id and
            zal.z_type = 0 and
            cl.c_type = 0
      into :id, :name, :countrycode, :dateofbirth, :pincode
  do
  begin
    suspend;
  end
end
