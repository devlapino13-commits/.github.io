CREATE PROCEDURE AAREP_ESP_OBORSALDO_KC(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_KC_ID ID_TYPE,
    REP_K_ID ID_TYPE = 0,
    REP_KIND D_STR_CODE = null,
    REP_IS_RESIDENT INTEGER = -1)
RETURNS (
    LINE_NO INTEGER,
    ACC_NO D_STR_CODE,
    ACC_BALANCE D_STR_CODE,
    KC_ID ID_TYPE,
    C_ID ID_TYPE,
    C_ID_STR D_STR_CODE,
    K_KONTRACT_NO D_STR_CODE,
    RES_BEGIN_DT CURR_TYPE,
    RES_BEGIN_KT CURR_TYPE,
    RES_OBOR_DT CURR_TYPE,
    RES_OBOR_KT CURR_TYPE,
    RES_END_DT CURR_TYPE,
    RES_END_KT CURR_TYPE,
    OKS_DISB_DATE DATE,
    OKS_LOAN_AMOUNT CURR_TYPE,
    OKS_LAST_SH_DATE DATE,
    OKS_IS_RESIDENT INTEGER,
    OKS_DELINQ_DAY_BEGIN INTEGER,
    OKS_DELINQ_DAY_END INTEGER,
    ACC_MEMO D_STR_255)
AS
declare variable kc_non_resident integer;
declare variable oks_os_bal_begin curr_type;
declare variable oks_os_dt curr_type;
declare variable oks_os_kt curr_type;
declare variable oks_os_bal_end curr_type;
declare variable oks_perc_bal_begin curr_type;
declare variable oks_perc_dt curr_type;
declare variable oks_perc_kt curr_type;
declare variable oks_perc_bal_end curr_type;
declare variable oks_pmt_perc_bal_begin curr_type;
declare variable oks_pmt_perc_dt curr_type;
declare variable oks_pmt_perc_kt curr_type;
declare variable oks_pmt_perc_bal_end curr_type;
declare variable oks_kommis_bal_begin curr_type;
declare variable oks_kommis_dt curr_type;
declare variable oks_kommis_kt curr_type;
declare variable oks_kommis_bal_end curr_type;
declare variable oks_pmt_kommis_bal_begin curr_type;
declare variable oks_pmt_kommis_dt curr_type;
declare variable oks_pmt_kommis_kt curr_type;
declare variable oks_pmt_kommis_bal_end curr_type;
declare variable kc_id_str d_str_code;
declare variable curr_section d_str_code;
declare variable oks_llr_os_bal_begin curr_type;
declare variable oks_llr_os_dt curr_type;
declare variable oks_llr_os_kt curr_type;
declare variable oks_llr_os_bal_end curr_type;
declare variable oks_llr_perc_bal_begin curr_type;
declare variable oks_llr_perc_dt curr_type;
declare variable oks_llr_perc_kt curr_type;
declare variable oks_llr_perc_bal_end curr_type;
declare variable oks_ecl_vbs_bal_begin curr_type;
declare variable oks_ecl_vbs_dt curr_type;
declare variable oks_ecl_vbs_kt curr_type;
declare variable oks_ecl_vbs_bal_end curr_type;
declare variable tmp_llr_os_prv curr_type;
declare variable tmp_llr_os curr_type;
declare variable tmp_llr_perc_prv curr_type;
declare variable tmp_llr_perc curr_type;
declare variable tmp_ecl_vbs_prv curr_type;
declare variable tmp_ecl_vbs curr_type;
declare variable tmp_amount curr_type;
declare variable ks_date_end date;
declare variable has_begin_saldo integer;
declare variable has_end_saldo integer;
declare variable has_saldo integer;
declare variable loan_amount curr_type;
declare variable koef double precision;
declare variable sh_perc curr_type;
begin
  if (extract(day from rep_date_begin) <> 1) then
    exception e_default 'Select first day of month for starting period';

  if (extract(day from rep_date_begin) <> 1) then
    exception e_default 'Select last day of month for ending period';

  if (rep_kc_id is null) then
    rep_kc_id = 0;

-- возможные значения rep_kind:
-- balance
-- calc_perc
-- pay_perc
-- calc_kommis
-- pay_kommis
-- llr_os
-- llr_perc
-- ecl_vbs

  if (rep_kind is null) then
    rep_kind = '';
  line_no = 0;
-- rep_resident_kind:
--  -1    все
--   0  - нерезиденты
--   1  - резиденты
  if (rep_is_resident is null) then
    rep_is_resident = -1;

  rep_date_begin = rep_date_begin - 1;
  for select
-- идентификаторы займа
             oks_from_kc_id, max(kc_non_resident), max(kc_from_c_id), max(k_kontract_no),

-- основной долг
             sum(iif(oks_date_begin = :rep_date_begin, oks_balance, 0)),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, -oks_disb_amount, 0)),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, -oks_pmt_os_norm - oks_pmt_os_delinq - oks_pmt_os_adv - oks_off_os, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_balance, 0)),

-- начисленные проценты
             sum(iif(oks_date_begin = :rep_date_begin, oks_bal_perc, 0)),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, oks_acr_perc, 0)),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, -oks_pmt_perc_norm - oks_pmt_perc_delinq - oks_pmt_perc_adv - oks_off_perc, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_bal_perc, 0)),

-- погашенные проценты
             sum(0),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, -oks_pmt_perc_norm - oks_pmt_perc_delinq - oks_pmt_perc_adv, 0)),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, -oks_pmt_perc_norm - oks_pmt_perc_delinq - oks_pmt_perc_adv, 0)),
             sum(0),

-- начисленные доп.проценты
             sum(0),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, oks_acr_kommis + oks_acr_kommis_adv, 0)),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, oks_acr_kommis + oks_acr_kommis_adv, 0)),
             sum(0),

-- погашение доп.проценты
             sum(iif(oks_date_begin = :rep_date_begin, oks_bal_kommis, 0)),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, oks_acr_kommis + oks_acr_kommis_adv, 0)),
             sum(iif(oks_date_begin > :rep_date_begin and oks_date_begin <= :rep_date_end, -oks_pmt_kommis, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_bal_kommis, 0)),

-- РВПЗ ОД
             sum(iif(oks_date_begin = :rep_date_begin, oks_llr_os, 0)), sum(0), sum(0),
             sum(iif(oks_date_begin = :rep_date_end, oks_llr_os, 0)),

-- РВПЗ %%
             sum(iif(oks_date_begin = :rep_date_begin, oks_llr_perc, 0)), sum(0), sum(0),
             sum(iif(oks_date_begin = :rep_date_end, oks_llr_perc, 0)),

-- Резерв РО
             sum(iif(oks_date_begin = :rep_date_begin and oks_close_state = 0, oks_diff_ecl_llr, 0)), sum(0), sum(0),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_diff_ecl_llr, 0)),

-- параметры займа
             max(oks_disb_date), max(kc_amount_approved),
             max(iif(oks_date_begin = :rep_date_end, oks_last_sh_date, null)), max(1 - kc_non_resident),
             max(iif(oks_date_begin = :rep_date_begin, iif(oks_delinq_date is not null, oks_delinq_days, null), null)),
             max(iif(oks_date_begin = :rep_date_end, iif(oks_delinq_date is not null, oks_delinq_days, null), null))
      from kn_oksummary, knclient, kontract
      where oks_date_begin between :rep_date_begin and :rep_date_end and
            oks_from_k_id = k_id and
            kc_id = oks_from_kc_id and
            kc_from_k_id = :rep_k_id and
            (:rep_kc_id = 0 or
            kc_id = :rep_kc_id) and
            kc_from_k_id = k_id
      group by oks_from_kc_id
      into kc_id, kc_non_resident, c_id, k_kontract_no,

           oks_os_bal_begin, oks_os_dt, oks_os_kt, oks_os_bal_end,

           oks_perc_bal_begin, oks_perc_dt, oks_perc_kt, oks_perc_bal_end,

           oks_pmt_perc_bal_begin, oks_pmt_perc_dt, oks_pmt_perc_kt, oks_pmt_perc_bal_end,

           oks_kommis_bal_begin, oks_kommis_dt, oks_kommis_kt, oks_kommis_bal_end,

           oks_pmt_kommis_bal_begin, oks_pmt_kommis_dt, oks_pmt_kommis_kt, oks_pmt_kommis_bal_end,

           oks_llr_os_bal_begin, oks_llr_os_dt, oks_llr_os_kt, oks_llr_os_bal_end,

           oks_llr_perc_bal_begin, oks_llr_perc_dt, oks_llr_perc_kt, oks_llr_perc_bal_end,

           oks_ecl_vbs_bal_begin, oks_ecl_vbs_dt, oks_ecl_vbs_kt, oks_ecl_vbs_bal_end,

           oks_disb_date, oks_loan_amount, oks_last_sh_date, oks_is_resident, oks_delinq_day_begin, oks_delinq_day_end
  do
  begin
    acc_memo = '';
    c_id_str = 'n' || c_id;
    kc_id_str = right('00000000' || kc_id, 8);
    if (kc_non_resident is null) then
      kc_non_resident = 0;
    if (oks_os_bal_begin is null) then
      oks_os_bal_begin = 0;
    if (oks_os_dt is null) then
      oks_os_dt = 0;
    if (oks_os_kt is null) then
      oks_os_kt = 0;
    if (oks_os_bal_end is null) then
      oks_os_bal_end = 0;
    if (oks_loan_amount is null) then
      oks_loan_amount = 0;

    curr_section = 'balance';
    if ((rep_kind = '' or rep_kind = curr_section) and
        (rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
    begin
      acc_memo = 'ОД';
      acc_balance = iif(oks_is_resident = 1, '48801', '49201');
      acc_no = acc_balance || '.810.00.00.' || kc_id_str;
      res_begin_dt = oks_os_bal_begin;
      res_begin_kt = 0;
      res_obor_dt = oks_os_dt;
      res_obor_kt = oks_os_kt;
      res_end_dt = oks_os_bal_end;
      res_end_kt = 0;

      if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
      begin
        line_no = line_no + 1;
        suspend;
      end
    end

    res_begin_dt = 0;
    res_begin_kt = 0;
    res_obor_dt = 0;
    res_obor_kt = 0;
    res_end_dt = 0;
    res_end_kt = 0;

    curr_section = 'calc_perc';
    if ((rep_kind = '' or rep_kind = curr_section) and
        (rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
    begin
      acc_memo = 'Начисленные %%';
      acc_balance = iif(oks_is_resident = 1, '48802', '49202');
      acc_no = acc_balance || '.810.00.10.' || kc_id_str;
      if (oks_perc_bal_begin > 0) then
        res_begin_dt = oks_perc_bal_begin;
      else
        res_begin_kt = -oks_perc_bal_begin;

      res_obor_dt = oks_perc_dt;
      res_obor_kt = oks_perc_kt;

      if (oks_perc_bal_end > 0) then
        res_end_dt = oks_perc_bal_end;
      else
        res_end_kt = -oks_perc_bal_end;

      if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
      begin
        line_no = line_no + 1;
        suspend;
      end
    end

    res_begin_dt = 0;
    res_begin_kt = 0;
    res_obor_dt = 0;
    res_obor_kt = 0;
    res_end_dt = 0;
    res_end_kt = 0;

    curr_section = 'pay_perc';
    if ((rep_kind = '' or rep_kind = curr_section) and
        (rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
    begin
      acc_memo = 'Выплаченные %%';
      acc_balance = iif(oks_is_resident = 1, '48809', '49209');
      acc_no = acc_balance || '.810.00.10.' || kc_id_str;
      if (oks_pmt_perc_bal_begin > 0) then
        res_begin_dt = oks_pmt_perc_bal_begin;
      else
        res_begin_kt = -oks_pmt_perc_bal_begin;

      res_obor_dt = oks_pmt_perc_dt;
      res_obor_kt = oks_pmt_perc_kt;

      if (oks_pmt_perc_bal_end > 0) then
        res_end_dt = oks_pmt_perc_bal_end;
      else
        res_end_kt = -oks_pmt_perc_bal_end;

      if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
      begin
        line_no = line_no + 1;
        suspend;
      end
    end

    res_begin_dt = 0;
    res_begin_kt = 0;
    res_obor_dt = 0;
    res_obor_kt = 0;
    res_end_dt = 0;
    res_end_kt = 0;

    curr_section = 'calc_kommis';
    if ((rep_kind = '' or rep_kind = curr_section) and
        (rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
    begin
      acc_memo = 'Начисленный доп.%%';
      acc_balance = iif(oks_is_resident = 1, '48802', '49202');
      acc_no = acc_balance || '.810.00.20.' || kc_id_str;
      if (oks_kommis_bal_begin > 0) then
        res_begin_kt = oks_kommis_bal_begin;
      else
        res_begin_dt = -oks_kommis_bal_begin;

      res_obor_dt = oks_kommis_dt;
      res_obor_kt = oks_kommis_kt;

      if (oks_kommis_bal_end > 0) then
        res_end_kt = oks_kommis_bal_end;
      else
        res_end_dt = -oks_kommis_bal_end;

      if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
      begin
        line_no = line_no + 1;
        suspend;
      end
    end

    res_begin_dt = 0;
    res_begin_kt = 0;
    res_obor_dt = 0;
    res_obor_kt = 0;
    res_end_dt = 0;
    res_end_kt = 0;

    curr_section = 'pay_kommis';
    if ((rep_kind = '' or rep_kind = curr_section) and
        (rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
    begin
      acc_memo = 'Выплаченный доп.%%';
      acc_balance = iif(oks_is_resident = 1, '48809', '49209');
      acc_no = acc_balance || '.810.00.20.' || kc_id_str;
      if (oks_pmt_kommis_bal_begin > 0) then
        res_begin_kt = oks_pmt_kommis_bal_begin;
      else
        res_begin_dt = -oks_pmt_kommis_bal_begin;

      res_obor_dt = oks_pmt_kommis_dt;
      res_obor_kt = oks_pmt_kommis_kt;

      if (oks_pmt_kommis_bal_end > 0) then
        res_end_kt = oks_pmt_kommis_bal_end;
      else
        res_end_dt = -oks_pmt_kommis_bal_end;

      if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
      begin
        line_no = line_no + 1;
        suspend;
      end
    end

    oks_llr_os_dt = 0;
    oks_llr_os_kt = 0;

    oks_llr_perc_dt = 0;
    oks_llr_perc_kt = 0;

    oks_ecl_vbs_dt = 0;
    oks_ecl_vbs_kt = 0;

    if (rep_kind = '' or rep_kind in ('llr_os', 'llr_perc', 'ecl_vbs')) then
    begin
      tmp_llr_os_prv = oks_llr_os_bal_begin;
      tmp_llr_perc_prv = oks_llr_perc_bal_begin;
      tmp_ecl_vbs_prv = oks_ecl_vbs_bal_begin;

      for select oks_llr_os, oks_llr_perc, iif(oks_close_state = 0, oks_diff_ecl_llr, 0)
          from kn_oksummary
          where oks_date_begin > :rep_date_begin and
                oks_date_begin <= :rep_date_end and
                oks_from_kc_id = :kc_id
          order by oks_from_kc_id, oks_date_begin
          into :tmp_llr_os, :tmp_llr_perc, tmp_ecl_vbs
      do
      begin
        if (tmp_llr_os >= tmp_llr_os_prv) then
          oks_llr_os_kt = oks_llr_os_kt + (tmp_llr_os - tmp_llr_os_prv);
        else
          oks_llr_os_dt = oks_llr_os_dt + (tmp_llr_os_prv - tmp_llr_os);

        if (tmp_llr_perc >= tmp_llr_perc_prv) then
          oks_llr_perc_kt = oks_llr_perc_kt + (tmp_llr_perc - tmp_llr_perc_prv);
        else
          oks_llr_perc_dt = oks_llr_perc_dt + (tmp_llr_perc_prv - tmp_llr_perc);

        if (tmp_ecl_vbs >= tmp_ecl_vbs_prv) then
          oks_ecl_vbs_kt = oks_ecl_vbs_kt + (tmp_ecl_vbs - tmp_ecl_vbs_prv);
        else
          oks_ecl_vbs_dt = oks_ecl_vbs_dt + (tmp_ecl_vbs_prv - tmp_ecl_vbs);

        tmp_llr_os_prv = tmp_llr_os;
        tmp_llr_perc_prv = tmp_llr_perc;
        tmp_ecl_vbs_prv = tmp_ecl_vbs;
      end
    end

    res_begin_dt = 0;
    res_begin_kt = 0;
    res_obor_dt = 0;
    res_obor_kt = 0;
    res_end_dt = 0;
    res_end_kt = 0;

    curr_section = 'llr_os';
    if ((rep_kind = '' or rep_kind = curr_section) and
        (rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
    begin
      acc_memo = 'РВПЗ ОД';
      acc_balance = iif(oks_is_resident = 1, '48810', '49210');
      acc_no = acc_balance || '.810.00.X1.' || kc_id_str;
      if (oks_llr_os_bal_begin > 0) then
        res_begin_kt = oks_llr_os_bal_begin;
      else
        res_begin_dt = -oks_llr_os_bal_begin;

      res_obor_dt = oks_llr_os_dt;
      res_obor_kt = oks_llr_os_kt;

      if (oks_llr_os_bal_end > 0) then
        res_end_kt = oks_llr_os_bal_end;
      else
        res_end_dt = -oks_llr_os_bal_end;

      if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
      begin
        line_no = line_no + 1;
        suspend;
      end
    end

    res_begin_dt = 0;
    res_begin_kt = 0;
    res_obor_dt = 0;
    res_obor_kt = 0;
    res_end_dt = 0;
    res_end_kt = 0;

    curr_section = 'llr_perc';
    if ((rep_kind = '' or rep_kind = curr_section) and
        (rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
    begin
      acc_memo = 'РВПЗ Проценты';
      acc_balance = iif(oks_is_resident = 1, '48810', '49210');
      acc_no = acc_balance || '.810.00.X2.' || kc_id_str;
      if (oks_llr_perc_bal_begin > 0) then
        res_begin_kt = oks_llr_perc_bal_begin;
      else
        res_begin_dt = -oks_llr_perc_bal_begin;

      res_obor_dt = oks_llr_perc_dt;
      res_obor_kt = oks_llr_perc_kt;

      if (oks_llr_perc_bal_end > 0) then
        res_end_kt = oks_llr_perc_bal_end;
      else
        res_end_dt = -oks_llr_perc_bal_end;

      if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
      begin
        line_no = line_no + 1;
        suspend;
      end
    end

    res_begin_dt = 0;
    res_begin_kt = 0;
    res_obor_dt = 0;
    res_obor_kt = 0;
    res_end_dt = 0;
    res_end_kt = 0;

    curr_section = 'ecl_vbs';
    if ((rep_kind = '' or rep_kind = curr_section) and
        (rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
    begin
      acc_memo = 'Корректировка РО';
      acc_balance = iif(oks_is_resident = 1, '48810', '49210');
      acc_no = acc_balance || '.810.00.X3.' || kc_id_str;
      if (oks_ecl_vbs_bal_begin > 0) then
        res_begin_kt = oks_ecl_vbs_bal_begin;
      else
        res_begin_dt = -oks_ecl_vbs_bal_begin;

      res_obor_dt = oks_ecl_vbs_dt;
      res_obor_kt = oks_ecl_vbs_kt;

      if (oks_ecl_vbs_bal_end > 0) then
        res_end_kt = oks_ecl_vbs_bal_end;
      else
        res_end_dt = -oks_ecl_vbs_bal_end;

      if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
      begin
        line_no = line_no + 1;
        suspend;
      end
    end

  end

  curr_section = 'poruch';
  if (rep_kind = '' or rep_kind = curr_section) then
  begin
    for select kc_id, kc_amount_approved, ks_date_begin, ks_date_end, 1 - coalesce(kc_non_resident, 0), kc_from_c_id,
               k_kontract_no, k_amount_approved
        from knclient, kontstate, kontract
        where ks_date_begin <= :rep_date_end and
              ks_date_end >= :rep_date_begin and
              ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_from_k_id = kc_from_k_id and
              kc_amount_approved > 0 and
              ks_from_k_id = k_id and
              k_reserv_kind = 2 and
              kc_from_k_id = :rep_k_id and
              (:rep_kc_id = 0 or
              kc_id = :rep_kc_id)
        into kc_id, oks_loan_amount, oks_disb_date, ks_date_end, oks_is_resident, c_id, k_kontract_no, loan_amount
    do
    begin
      if (not(rep_is_resident = -1 or rep_is_resident = oks_is_resident)) then
        continue;

      c_id_str = 'n' || c_id;
      kc_id_str = right('00000000' || kc_id, 8);

      acc_memo = 'Поручит. внебаланс';
      acc_balance = iif(oks_is_resident = 1, '91414', '91414');
      acc_no = acc_balance || '.810.00.00.' || kc_id_str;

      if (ks_date_end <= '31.12.2999') then
        ks_date_end = ks_date_end + 1;

      koef = oks_loan_amount / loan_amount;
      tmp_amount = 0;

      sh_perc = 0;
      select sum(sh_perc)
      from schedule
      where sh_from_k_id = :rep_k_id
      into sh_perc;

      sh_perc = round(sh_perc * koef, 2);
      tmp_amount = oks_loan_amount + sh_perc;

      if (tmp_amount <> 0) then
      begin
        res_begin_dt = 0;
        res_begin_kt = 0;
        res_obor_dt = 0;
        res_obor_kt = 0;
        res_end_dt = 0;
        res_end_kt = 0;

        has_begin_saldo = 0;
        has_end_saldo = 0;
        has_saldo = 0;

        if (oks_disb_date <= rep_date_begin) then
        begin
          if (exists(select *
                     from kn_oksummary
                     where oks_from_kc_id = :kc_id and
                           oks_date_begin = :rep_date_begin and
                           (oks_balance <> 0 or
                           oks_bal_perc <> 0) and
                           oks_close_state = 0)) then
            has_begin_saldo = 1;

          if (kc_id = 622529 and
              rep_date_begin <= '15.01.2018') then
            has_begin_saldo = 1;
        end

        if (ks_date_end > rep_date_end) then
        begin
          if (exists(select *
                     from kn_oksummary
                     where oks_from_kc_id = :kc_id and
                           oks_date_begin = :rep_date_end and
                           (oks_balance <> 0 or
                           oks_bal_perc <> 0))) then
            has_end_saldo = 1;

          if (kc_id = 622529 and
              rep_date_end <= '15.01.2018') then
            has_end_saldo = 1;
        end

        if (has_begin_saldo = 1) then
        begin
          res_begin_dt = tmp_amount;
          has_saldo = 1;
        end

        if (oks_disb_date > rep_date_begin and
            oks_disb_date <= rep_date_end) then
        begin
          res_obor_dt = tmp_amount;
          has_saldo = 1;
        end

        if ((ks_date_end > rep_date_begin and
            ks_date_end <= rep_date_end and
            has_saldo = 1) or (has_saldo = 1 and
            has_end_saldo = 0)) then
          res_obor_kt = tmp_amount;
        else
        if (has_end_saldo = 1) then
          res_end_dt = tmp_amount;

        if (res_begin_dt <> 0 or res_begin_kt <> 0 or res_obor_dt <> 0 or res_obor_kt <> 0 or res_end_dt <> 0 or res_end_kt <> 0) then
        begin
          line_no = line_no + 1;

          oks_disb_date = null;
          oks_loan_amount = 0;
          oks_last_sh_date = null;
          oks_delinq_day_begin = null;
          oks_delinq_day_end = null;

-- параметры займа
          select oks_last_sh_date, iif(oks_delinq_date is not null, oks_delinq_days, null)
          from kn_oksummary
          where oks_from_kc_id = :kc_id and
                oks_date_begin = :rep_date_end
          into oks_last_sh_date, oks_delinq_day_end;

          select iif(oks_delinq_date is not null, oks_delinq_days, null)
          from kn_oksummary
          where oks_from_kc_id = :kc_id and
                oks_date_begin = :rep_date_begin
          into oks_delinq_day_begin;

          suspend;
        end
      end
    end
  end
end
