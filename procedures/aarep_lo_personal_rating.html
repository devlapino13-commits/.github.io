CREATE PROCEDURE AAREP_LO_PERSONAL_RATING(
    REP_C_ID DECIMAL(18,0))
RETURNS (
    REG_NAME VARCHAR(100),
    JOB_NAME VARCHAR(100),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REG_ID INTEGER,
    JK_SYSTEM_CODE VARCHAR(255),
    REP_IS_VOCATION INTEGER)
AS
declare variable dd1 date;
declare variable dd2 date;
begin
  for select rep_date_begin, max(rep_date_end)
      from aarep_lo_personal_rating_tmp(:rep_c_id)
      group by rep_date_begin
      into :rep_date_begin, :rep_date_end
  do
  begin
    rep_is_vocation = 0;

--    select c_name_fio, reg_code||' '||reg_name, jl_name
--    тут определить должность и офис
    suspend;
  end

  for select rep_date_begin, max(rep_date_end)
      from aarep_lo_personal_rating_tmp(:rep_c_id)
      group by rep_date_begin
      into :dd1, :dd2
  do
  begin
    rep_is_vocation = 1;

    for select hrv_date_begin, hrv_date_end_fact
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_c_id = :rep_c_id and
              hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              hrv_date_begin between :dd1 and :dd2 and
              sawt_base_type = 4 and
              sawt_is_paid = 1
        into :rep_date_begin, :rep_date_end
    do
    begin
--    тут определить должность и офис
      suspend;
    end
  end

end
