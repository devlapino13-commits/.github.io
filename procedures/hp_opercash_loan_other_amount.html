CREATE PROCEDURE HP_OPERCASH_LOAN_OTHER_AMOUNT(
    IN_K_ID D_ID,
    OPER_CODE D_STR_CODE)
RETURNS (
    RES_AMOUNT D_CURRENCY)
AS
declare variable RD_ID D_INTEGER;
declare variable PAYED_AMOUNT D_CURRENCY;
declare variable KC_ID D_ID;
declare variable INSURANCE_MIN_AMOUNT D_INTEGER;
begin
  -- oper_code:
  --   CashTransactionIncom.PartialPrice  - Частиная оплата по ИПФ
  --   CashTransactionIncom.IPFCommission - Комиссия по ИПФ
  --   CashTransactionIncom.Consultation  - Оплата за консультации
  --   CashTransactionIncom.Insurance     - Оплата за страхование

  res_amount = 0;
  rd_id = f_get_rd_id(oper_code);

  if (oper_code = 'CashTransactionIncom.PartialPrice') then
  begin
    select coalesce(k_prepayment_summa, 0)
    from kontract
    where k_id = :in_k_id
    into res_amount;

    if (res_amount > 0) then
    begin

      select sum(opk.ok_amount)
      from operkontract opk
      inner join operlist ol on ol.ol_id = opk.ok_from_ol_id and
            ol.is_delete = 0
      where opk.ok_from_k_id = :in_k_id and
            opk.ok_operation_type = :rd_id
      into :payed_amount;

      if (payed_amount is null) then
        payed_amount = 0;

      res_amount = res_amount - payed_amount;
    end
  end else
  if (oper_code = 'CashTransactionIncom.Consultation') then
  begin
    select kc_id
    from knclient
    where kc_from_k_id = :in_k_id and
          kc_amount_approved > 0
    rows 1
    into kc_id;

    select price
    from clt_get_price_for_consultation(:kc_id)
    into :res_amount;

    if (res_amount > 0) then
    begin
      rd_id = f_get_rd_id(oper_code);

      select sum(opk.ok_amount)
      from operkontract opk
      inner join operlist ol on ol.ol_id = opk.ok_from_ol_id and
            ol.is_delete = 0
      where opk.ok_from_k_id = :in_k_id and
            opk.ok_operation_type = :rd_id
      into :payed_amount;

      if (payed_amount is null) then
        payed_amount = 0;

      res_amount = res_amount - payed_amount;
    end
  end else
  if (oper_code = 'CashTransactionIncom.Insurance') then
  begin
    if (f_get_const('new_loan_conditions') = 1) then
      insurance_min_amount = cast(f_get_const('kont_new_insurance_min_amount') as d_integer);
    else
      insurance_min_amount = cast(f_get_const('kont_insurance_min_amount') as d_integer);

    if (exists(select *
               from kontract
               inner join product on pr_id = k_from_pr_id
               where k_id = :in_k_id and
                     pr_code containing '.IPF.')) then
      insurance_min_amount = cast(f_get_const('kont_ipf_insurance_min_amount') as d_integer); -- RFCKG-5359

    if (f_get_const('kont_insurance_active') = 1) then
    begin
      if (exists(select *
                 from kontract
                 join product on pr_id = k_from_pr_id
                 where k_id = :in_k_id and
                       k_amount_approved >= :insurance_min_amount and
                       (pr_code containing '.IPF.' -- RFCKG-5359
                        or (
                              pr_code containing '.IND.'
                              and pr_code containing '.STANDART.'
                              and pr_code not containing '.insur_base.'
                              and (
                                  pr_code not containing '.consultation.'
                                  or (f_get_const('new_loan_conditions') = 1 and pr_code containing '.consultation.' and pr_code containing '.INSUR.')
                                  )
                            )
                       )
      )) then
      begin
        select ik.ik_commission
        from insurance_kontract ik
        where ik.ik_from_k_id = :in_k_id
        into :res_amount;
      end
    end

    if (res_amount is null) then
      res_amount = 0;

    if (res_amount > 0) then
    begin
      rd_id = f_get_rd_id(oper_code);

      select sum(opk.ok_amount)
      from operkontract opk
      inner join operlist ol on ol.ol_id = opk.ok_from_ol_id and ol.is_delete = 0
      where opk.ok_from_k_id = :in_k_id and
            opk.ok_operation_type = :rd_id
      into :payed_amount;

      if (payed_amount is null) then
        payed_amount = 0;

      res_amount = res_amount - payed_amount;
    end
  end

  suspend;
end
