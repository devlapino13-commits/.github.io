CREATE PROCEDURE HP_TERMINATION_OF_AGREEMENT(
    REP_DATE D_DATE)
AS
declare variable k_id d_id;
declare variable k_stop_fine d_date;
declare variable db_id d_integer;
declare variable delinq_d_min d_integer;
declare variable delinq_d_max d_integer;
declare variable k_fact_disb_date d_date;
declare variable k_kredline_end d_date;
declare variable oks_delinq_date d_date;
declare variable tmp_date d_date;
begin
  db_id = f_get_db_id();
  delinq_d_max = 9999;

  if (db_id in (81001, 94401)) then
    delinq_d_min = 91;
  else if (db_id in (39801, 41701)) then
    delinq_d_min = 90;
  else
    exit;

  -- Цель цикла ниже: Если сервис не отработал, то начиная за три дня до rep_date
  -- мы сделаем корректное расторжение нужной датой.
  -- То, что выполняем повторно, для даты, где было расторжение проблем не создает,
  -- т.к. учитывается значение k_stop_perc

  tmp_date = rep_date - 3;
  while (tmp_date <= rep_date) do
  begin
    for select oks_from_k_id, k_stop_fine, k_fact_disb_date, k_kredline_end, oks_delinq_date
        from oksummary, kontract, product
        where :tmp_date between oks_date_begin and oks_date_end and
              :tmp_date - oks_delinq_date between :delinq_d_min and :delinq_d_max and
              oks_from_k_id = k_id and
              k_stop_perc is null and
              oks_close_state = 0 and
              k_from_pr_id = pr_id and
              pr_code not containing '.IPF.'
        into k_id, k_stop_fine, k_fact_disb_date, k_kredline_end, oks_delinq_date
    do
    begin
      if ((db_id in (39801, 41701)) or
          (db_id = 94401 and k_fact_disb_date >= '01.07.2020') or
          (db_id = 81001 and k_kredline_end > oks_delinq_date + delinq_d_min)) then
        execute procedure hp_kontract_stop_perc(k_id, tmp_date, iif(k_stop_fine is null, tmp_date, null));
    end
    tmp_date = tmp_date + 1;
  end
end
