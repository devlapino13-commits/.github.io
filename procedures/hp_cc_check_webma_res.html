CREATE PROCEDURE HP_CC_CHECK_WEBMA_RES(
    CELL_NO STR_100,
    SITE_DATE DATE)
RETURNS (
    HAS_LOAN INTEGER,
    LOAN_DATE DATE,
    EMPLOYEE_DATE DATE,
    IS_EMPLOYEE INTEGER,
    K_KOLICH_CLIENT_APPROVED INTEGER)
AS
declare variable c_id id_type;
begin
  c_id = null;

  select res_object_id
  from hp_contdata_find_object(:cell_no, 'cell')
  where res_object_kind = 0 rows 1
  into :c_id;

  has_loan = 0;
  loan_date = null;
  employee_date = null;
  is_employee = 0;
  k_kolich_client_approved = 0;

  if (c_id is not null) then
  begin
    select ks.ks_date_begin, kont.k_kolich_client_approved
    from knclient kncl
    inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
    inner join kontract kont on kont.k_id = kncl.kc_from_k_id
    where ks.ks_ol_is_delete = 0 and
          kncl.kc_from_c_id = :c_id and
          ks.ks_state = 1 and
          ks.ks_date_begin >= :site_date
    order by ks.ks_date_begin
    rows 1
    into :loan_date, k_kolich_client_approved;

    if (loan_date is not null) then
      has_loan = 1;

    select hps.ps_date_begin
    from hr_pstatus hps
    inner join joblist jl on jl.jl_id = ps_from_jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind
    where :site_date between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          hps.ps_status = 1 and
          hps.ps_from_c_id = :c_id and
          ps_date_begin >= :site_date
    order by hps.ps_date_begin
    rows 1
    into :employee_date;

    if (employee_date is not null) then
      is_employee = 1;
    suspend;
  end
end
