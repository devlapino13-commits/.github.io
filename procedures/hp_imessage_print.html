CREATE PROCEDURE HP_IMESSAGE_PRINT(
    IM_ID D_ID,
    IM_TYPE D_INTEGER = 0)
RETURNS (
    CM_NAME D_STR_255,
    KONTRACT_NO D_STR_20,
    DISB_DATE D_DATE,
    REG_ID D_INTEGER,
    PERS_FIO D_STR_255,
    RES_SUMM ID_TYPE,
    PART_STR D_STR_2048,
    OL_DATE D_DATE,
    OL_DATETIME D_DATETIME,
    CASHIER_FIO D_STR_255,
    CASHIER_POS D_STR_255,
    CREATE_DATETIME D_DATETIME,
    CASHIER_SIGN_DATETIME D_DATETIME,
    CONTROLLER_1_FIO D_STR_100,
    CONTROLLER_1_POS D_STR_255,
    CONTROLLER_1_SIGN_DATETIME D_DATETIME,
    CONTROLLER_2_FIO D_STR_100,
    CONTROLLER_2_POS D_STR_255,
    CONTROLLER_2_SIGN_DATETIME D_DATETIME,
    STATUS D_INTEGER)
AS
declare variable k_id d_id;
declare variable cashier_id d_id;
declare variable controller_1_id d_id;
declare variable controller_2_id d_id;
declare variable num_day d_integer;
declare variable date_str d_str_code;
declare variable cl_name d_str_100;
declare variable im_from_c_id d_id;
begin
  if (im_type <> 0) then
    exit;
  for
    select
      k_id,
      k_kontract_no,
      k_disb_date,
      im_from_reg_id,
      im_through_whom,
      im_amount_payment,
      ol_date,
      ol_datetime,
      im_cashier_id,
      u1.un_fio,
      im_create_datetime,
      im_cashier_sign_datetime,
      coalesce(u2.un_fio, ''),
      im_controller_1_id,
      im_controller_1_sign_datetime,
      coalesce(u3.un_fio, ''),
      im_controller_2_id,
      im_controller_2_sign_datetime,
      im_status,
      im_type,
      im_from_c_id
    from internal_message
      left join operlist
        on ol_id = im_from_ol_id
          and is_delete = 0
      inner join kontract
        on k_id = im_from_k_id
      inner join username u1
        on u1.un_id = im_cashier_id
      left join username u2
        on u2.un_id = im_controller_1_id
      left join username u3
        on u3.un_id = im_controller_2_id
    where im_id = :im_id
  into
    :k_id, 
    :kontract_no,
    :disb_date,
    :reg_id,
    :pers_fio,
    :res_summ,
    :ol_date,
    :ol_datetime,
    :cashier_id,
    :cashier_fio,
    :create_datetime,
    :cashier_sign_datetime,
    :controller_1_fio,
    :controller_1_id,
    :controller_1_sign_datetime,
    :controller_2_fio,
    :controller_2_id,
    :controller_2_sign_datetime,
    :status,
    :im_type,
    :im_from_c_id
  do
  begin
    if (im_type = 1) then
    begin
      select c.c_name_fio
      from clients c
      where c.c_id = :im_from_c_id
      into pers_fio;
      if (ol_datetime is null and ol_date is null) then
      begin
        ol_datetime = cashier_sign_datetime;
        ol_date = cast(cashier_sign_datetime as d_date);
      end
    end

    if (controller_1_id = 1010100001839 and controller_1_sign_datetime < '01.03.2021')
      then controller_1_fio = 'Мухетдинова Гузель Имамовна';
    if (controller_2_id = 1010100001839 and controller_2_sign_datetime < '01.03.2021')
      then controller_2_fio = 'Мухетдинова Гузель Имамовна';

    cm_name = f_company_name('short', 'local', :ol_date);
    date_str = f_datetostr(:disb_date);
    cl_name = (
      select
        c_name_fio
      from knclient
        inner join clients
          on c_id = kc_from_c_id
      where kc_from_k_id = :k_id
        and kc_amount_approved > 0);

    cashier_pos =
      coalesce(f_get_position(:cashier_id, :ol_date), 'Кассир-оператор');
    if (controller_1_id > 0)
      then controller_1_pos = coalesce(f_get_position(:controller_1_id, :ol_date),'');
    if (controller_2_id > 0)
      then controller_2_pos = coalesce(f_get_position(:controller_2_id, :ol_date),'');

     num_day = extract(day from :ol_date);
     part_str =
       case
         when num_day between 1 and 10 then
           'Накануне платежа Клиент ' || cl_name || ' обратился в ' || cm_name || ' (далее - Компания) с устной просьбой (по телефону) принять досрочное погашение суммы займа от ' || pers_fio || ' по Договору группового микрозайма с поручительством № ' || kontract_no || ' от ' || date_str || ' года.' ||
           ' Клиент объяснил, что сам по уважительным причинам явиться в офис Компании для осуществления платежа в кассу не может, поэтому просил произвести погашение своего хорошего знакомого ' || pers_fio || '.' ||
           ' Состоялся телефонный разговор с клиентом, установлен факт знакомства третьего лица, осуществившего платеж, с заемщиком.' ||
           '  На основании вышеизложенного, в отношении описанной выше операции, отсутствуют подозрения в том, что она совершена в целях легализации (отмывания) доходов, полученных преступным путем, или финансирования терроризма.'
         when num_day between 11 and 20 then
           'В кассу офиса Компании обратился ' || pers_fio || ' для осуществления погашения по микрозайму за клиента ' || cl_name || ' по Договору микрозайма № ' || kontract_no || ' от ' || date_str || ' года.' ||
           ' ' || pers_fio || ' пояснил, что сам клиент по уважительной причине явиться в офис Компании не может, поэтому попросил его осуществить погашение займа за него.' ||
           '  На основании вышеизложенного, в отношении описанной выше операции, отсутствуют подозрения в том, что она совершена в целях легализации (отмывания) доходов, полученных преступным путем, или финансирования терроризма.'
       else
           pers_fio || ' обратился к кассиру офиса Компании для проведения погашения по микрозайму за клиента по Договору микрозайма № ' || kontract_no || ' от ' || date_str || ' года.' ||
           ' ' || pers_fio || ' пояснил, что ' || cl_name || ' явиться в офис Компании не может в связи с тем, что у него возникли срочные семейные обстоятельства,' ||
           ' и в этой связи попросил оказать ему услугу и произвести погашение за него по договору микрозайма для обеспечения клиенту Компании положительной кредитной истории.' ||
           '  На основании вышеизложенного, в отношении описанной выше операции, отсутствуют подозрения в том, что она совершена в целях легализации (отмывания) доходов, полученных преступным путем, или финансирования терроризма.'
       end;
    suspend;
  end
end
