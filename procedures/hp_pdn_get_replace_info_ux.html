CREATE PROCEDURE HP_PDN_GET_REPLACE_INFO_UX(
    PRP_ID D_ID,
    K_ID D_ID,
    PRP_DATE D_DATE)
RETURNS (
    REASON D_STR_CODE,
    PRP_CREATE_DATETIME D_DATETIME,
    PRP_DATETIME D_DATETIME,
    ADDR_REG D_STR_255,
    ADDR_FACT D_STR_255,
    ADDR_WORK D_STR_255,
    OFFICE D_STR_255,
    KC_ID D_ID,
    CAN_EDIT D_BOOL)
AS
declare variable kcp_kind d_integer;
  declare variable c_id d_id;
  declare variable disb_date d_date;
begin
  can_edit = 0;

  if (k_id is null or prp_date is null) then
    exit;

  select prp_kc_id, prp_date, prp_datetime, prp_create_datetime, kc_from_k_id, kc_from_c_id
  from pdn_replace_places, knclient
  where prp_id = :prp_id
    and kc_id = prp_kc_id
  into kc_id, prp_date, prp_datetime, prp_create_datetime, k_id, c_id;

  if (kc_id is null) then
    select kc_id, kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id
    into kc_id, c_id;

  select k_disb_date
  from kontract
  where k_id = :k_id
  into disb_date;

  if (prp_date = disb_date) then
    select kcp_kind
    from knclient_pdn kcp
    where kc_id = :kc_id
      and kcp_kind = 0
    into kcp_kind;
  else
    select kcp_kind
    from knclient_pdn
    inner join com_list on cml_from_kcp_id = kcp_id
    inner join com_cfg on cmc_id = cml_from_cmc_id
    inner join com_action on cma_from_cml_id = cml_id
    left join com_operlist on cmo_from_cma_id = cma_id
    where kc_id = :kc_id
      and cml_date = :prp_date
      and kcp_kind = 1
      and cmo_ol_is_delete = 0
    order by kc_date desc, kcp_id desc
    rows 1
    into kcp_kind;


  reason = 'Неизвестно';

  if (kcp_kind is null) then
  begin
    reason = 'Расчет еще не создан';
    can_edit = 1;
  end
  else if (kcp_kind = 0) then
    reason = 'Выдача';
  else if (kcp_kind = 1) then
    reason = 'КК';

  select adl_address_str
  from addrclient, hp_get_addrlist(adc_from_adl_id, :prp_date)
  where adc_from_c_id = :c_id
    and adc_address_type = 1
  into addr_reg;

  select adl_address_str
  from addrclient, hp_get_addrlist(adc_from_adl_id, :prp_date)
  where adc_from_c_id = :c_id
    and adc_address_type = 2
  into addr_fact;

  select adl_address_str
  from addrclient, hp_get_addrlist(adc_from_adl_id, :prp_date)
  where adc_from_c_id = :c_id
    and adc_address_type = 3
  into addr_work;

  select f_format('%1 (%2)', reg_name, reg_code)
  from knaktlist kal, kontract, region
  where :prp_date between kal_date and kal_date_end
    and k_id = :k_id
    and kal_from_kf_id = k_from_kf_id
    and kal.is_delete = 0
    and reg_unicode = kal_new_reg_id
  into office;

  suspend;
end
