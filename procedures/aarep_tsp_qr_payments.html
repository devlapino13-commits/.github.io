CREATE PROCEDURE AAREP_TSP_QR_PAYMENTS(
    REP_DATE_END D_DATE)
RETURNS (
    PARENT_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    TERM_NAME D_STR_255,
    SUM_AMOUNT CURR_TYPE,
    KOLVO_TRANS D_INTEGER,
    TERM_TYPE D_STR_10,
    RESULT_DATE_BEGIN D_DATE,
    RESULT_DATE_END D_DATE,
    COMMIS_PERC_MB D_PERCENT,
    COMMIS_PERC_OTHER D_PERCENT)
AS
declare variable cs_id d_id;
declare variable cs_ipc_id d_id;
declare variable ag_id d_id;
declare variable iqa_id d_id;
declare variable rd_id d_id;
declare variable obj_kind d_integer;
declare variable cs_from_c_id d_id;
declare variable reg_id d_id;
declare variable c_id d_id;
declare variable date_begin d_date;
declare variable ito_date_end d_date;
declare variable trm_status d_integer;
declare variable trm_id d_id;
declare variable ito_id d_id;
declare variable cs_obj_kind d_integer;
declare variable cl_obj_kind d_integer;
declare variable rd_id_tsp d_integer;
declare variable rd_id_tsp_ipf d_integer;
declare variable ito_object_id d_id;
declare variable ito_object_kind d_integer;
declare variable ag_from_init_reg_id d_id;
begin
   cs_obj_kind = f_objkind('ClientShops', 'id');
   cl_obj_kind = f_objkind('clients', 'id');
   rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');
   rd_id_tsp_ipf = f_get_rd_id('AgreementKinds.tsp_and_ipf');
  -- rd_id_tsp_ipff = f_get_rd_id ('AgreementKinds.IPF_Sellers');

   result_date_end = rep_date_end - 1;
   result_date_begin = cast('01.' || extract(month from result_date_end) || '.' || extract(year from result_date_end) as date);

   for select ita.ita_term_name, rg.reg_name, rg2.reg_name ,its.trm_id ,ito.ito_object_id ,ito.ito_object_kind
    from ipc_terminals its
    inner join ipc_terminal_app ita on ita.ita_from_trm_id = its.trm_id
    inner join ipc_terminal_ownership ito on ito.ito_trm_id =its.trm_id
    inner join region rg on rg.reg_unicode = ita.ita_from_reg_id
    inner join region rg2 on rg2.reg_unicode = rg.parent
    where ito.ito_status >-1  and
          rg.reg_unicode > 0 and
          rg2.reg_unicode > 0
    into term_name, office_name, parent_name ,:trm_id,:ito_object_id ,:ito_object_kind
    do
    begin

    kolvo_trans = null;
    sum_amount = null;

    select count(t.igdp_terminal_id), sum(t.igdp_amount)
    from ipc_gc_analiz_detail_pos t
    inner join ipc_gc_analiz iga on iga.iga_id = t.igdp_from_iga_id
    inner join ipc_file_analiz ifa on ifa.ifa_id = iga.iga_from_ifa_id
    where ifa.ifa_date between :result_date_begin and :result_date_end
    and t.igdp_terminal_id = :trm_id
    into kolvo_trans, sum_amount;

    commis_perc_other = 0;
    commis_perc_mb = 0;

    select coalesce (ag.ag_commis_cards_mb, 0), coalesce (ag.ag_commis_cards_other ,0)
    from clientshops cs
    inner join agreements ag on ag.ag_object_id = cs.cs_from_c_id
    where ag.ag_object_kind = :cl_obj_kind and
          ag.ag_status = 1 and
          cs.cs_id = :ito_object_id  and
          :ito_object_kind = :cs_obj_kind and
          :rep_date_end between ag.ag_date_begin and coalesce(ag.ag_date_end, '31.12.9999') and
          ag.ag_from_rd_id in (:rd_id_tsp, :rd_id_tsp_ipf)
    into commis_perc_mb,  commis_perc_other;


    sum_amount = coalesce(sum_amount,0);
    kolvo_trans = coalesce (kolvo_trans,0);
    term_type = 'POS';

    suspend;
  end

     commis_perc_other = 0;
     commis_perc_mb = 0;

  for select cs.cs_id, cs.cs_name,coalesce (ag.ag_commis_cards_mb,0),coalesce (ag.ag_commis_cards_other ,0) ,ag.ag_from_init_reg_id
      from clientshops cs
      inner join agreements ag on ag.ag_object_id = cs.cs_from_c_id
      where cs.cs_is_delete = 0
        and cs.cs_from_rd_id = :rd_id_tsp
        and ag.ag_object_kind = :cl_obj_kind
        and ag.ag_from_rd_id in (:rd_id_tsp,:rd_id_tsp_ipf)
        and :rep_date_end between ag.ag_date_begin  and coalesce (ag.ag_date_end , '31.12.9999')
        and ag_from_init_reg_id > 0
        and ag.ag_status = 1
      order by ag.ag_agreement_date desc
      into :cs_id, term_name , commis_perc_mb, commis_perc_other,:ag_from_init_reg_id
  do
  begin
     office_name = null;
     parent_name = null;


     select  r2.reg_name, r1.reg_name
     from region  r1
     inner join region as r2 on r2.reg_unicode = r1.parent
     where r1.reg_unicode = :ag_from_init_reg_id
     into parent_name, :office_name;


     kolvo_trans = null;
     sum_amount = null;

     select count(iqp.iqp_id), sum(iqp.iqp_amount)
     from ipc_qr_payments iqp
     where iqp.iqp_from_cs_id = :cs_id and
           iqp.iqp_date between :result_date_begin and :result_date_end and
           iqp.iqp_id > 0
     into kolvo_trans, sum_amount;

   sum_amount = coalesce(sum_amount,0);
   term_type ='QR';

  suspend;
  end
 end
