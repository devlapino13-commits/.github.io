CREATE PROCEDURE CHECK_EXTERNAL_REFS(
    MA_ID D_ID,
    IS_PORUCH D_SMALLINT)
RETURNS (
    HAVE_FAMILY_MEMBERS D_SMALLINT,
    HAVE_FACT_ADDRESS D_SMALLINT,
    HAVE_KIB D_SMALLINT,
    HAVE_ANIMAL D_SMALLINT)
AS
declare variable obj_kind d_integer;
declare variable c_id d_id;
begin
  obj_kind = f_objkind('miniapp', 'id');
  have_family_members = 0;
  have_fact_address = 0;
  have_kib = 0;
  have_animal = 0;
  for select mac.mac_from_c_id
      from miniappclient mac
      where mac.mac_from_ma_id = :ma_id and
            iif(:is_poruch = 1, mac.mac_is_poruch = 1, mac.mac_amount > 0)
      into c_id
  do
  begin
    if (exists(select *
               from kibminiapp kbm
               where kbm.kbm_ma_id = :ma_id and
                     kbm.kbm_from_c_id = :c_id and
                     kbm.kbm_is_checked = 1)) then
      have_kib = 1;
    if (exists(select *
               from extinfolist eil
               where eil.eil_from_eic_id = 3 and
                     eil.eil_object_kind = :obj_kind and
                     eil.eil_object_id = :ma_id and
                     eil.eil_from_c_id = :c_id and
                     eil.eil_status >= 0)) then
      have_fact_address = 1;
    if (exists(select *
               from extinfolist eil
               where eil.eil_from_eic_id = 4 and
                     eil.eil_object_kind = :obj_kind and
                     eil.eil_object_id = :ma_id and
                     eil.eil_from_c_id = :c_id and
                     eil.eil_status >= 0)) then
      have_family_members = 1;
    if (exists(select *
               from extinfolist eil
               where eil.eil_from_eic_id = 7 and
                     eil.eil_object_kind = :obj_kind and
                     eil.eil_object_id = :ma_id and
                     eil.eil_from_c_id = :c_id and
                     eil.eil_status >= 0)) then
      have_animal = 1;
    if (have_kib = 0 or   -- для групповых контрактов
        have_animal = 0 or
        have_family_members = 0 or
        have_fact_address = 0) then
    begin
      suspend;
      exit;
    end
  end
  suspend;
end
