CREATE PROCEDURE RCIB_FILL_POCKET(
    RCP_DATE_BEGIN DATE,
    POCKET_DATE DATE,
    IS_KREDIT SMALLINT,
    RCP_TYPE SMALLINT,
    BASE_RCP_ID ID_TYPE)
RETURNS (
    RCP_ID ID_TYPE,
    RCP_VERSION INTEGER,
    RCP_FILENAME STR_255)
AS
declare variable y varchar(4);
declare variable m varchar(2);
declare variable d varchar(2);
declare variable v varchar(1);
declare variable kredit_active_date date;
declare variable rcp_date date;
declare variable ma_active_date date;
begin
  if (base_rcp_id = 0) then
    base_rcp_id = null;

  kredit_active_date = cast('01.07.2014' as date); -- дата запуска проекта по выгрузке кредитных историй
  ma_active_date = cast('01.03.2015' as date); -- дата запуска проекта по выгрузке ФИЧ-файла

  if (base_rcp_id is null) then -- если это не пакет по ошибкам
  begin
    select rcp.rcp_date, rcp.rcp_version
    from rcib_pocket rcp
    where rcp.rcp_date <= :pocket_date and
          rcp.rcp_is_kredit = :is_kredit
    order by rcp.rcp_date desc
    rows 1
    into :rcp_date, rcp_version;

    if (rcp_date = pocket_date) then
      rcp_version = rcp_version + 1;
    else
    begin
      rcp_date_begin = rcp_date + 1;
      rcp_version = 1;
    end
  end
  else
  begin
    select rcp.rcp_date_begin, rcp.rcp_date
    from rcib_pocket rcp
    where rcp.rcp_id = :base_rcp_id
    into :rcp_date_begin, :pocket_date;
  end

  if (is_kredit = 1) then
  begin
    if (rcp_date_begin is null) then
    begin
      rcp_date_begin = kredit_active_date;
    end
  end
  else
  if (rcp_date_begin is null) then
    rcp_date_begin = ma_active_date;

  if (rcp_date_begin < ma_active_date) then
    rcp_date_begin = ma_active_date;

  if (rcp_type = 2) then
    rcp_date_begin = '01.01.1900';
------------------------------------------------------------------------
  select max(rcp.rcp_id)
  from rcib_pocket rcp
  into :rcp_id;
  rcp_id = coalesce(rcp_id, 0) + 1;

  if (is_kredit = 0) then
    y = cast(extract(year from pocket_date) as varchar(4));
  else
    y = substring(cast(extract(year from :pocket_date) as varchar(4)) from 4 for 1);
  m = cast(extract(month from :pocket_date) as varchar(2));
  d = cast(extract(day from :pocket_date) as varchar(2));

  if (is_kredit = 1) then -- кредитный пакет
  begin
    if (m = '10') then
      m = 'A';
    else
    if (m = '11') then
      m = 'B';
    else
    if (m = '12') then
      m = 'C';

    if (d = '10') then
      d = 'A';
    else
    if (d = '11') then
      d = 'B';
    else
    if (d = '12') then
      d = 'C';
    else
    if (d = '13') then
      d = 'D';
    else
    if (d = '14') then
      d = 'E';
    else
    if (d = '15') then
      d = 'F';
    else
    if (d = '16') then
      d = 'J';
    else
    if (d = '17') then
      d = 'H';
    else
    if (d = '18') then
      d = 'I';
    else
    if (d = '19') then
      d = 'J';
    else
    if (d = '20') then
      d = 'K';
    else
    if (d = '21') then
      d = 'L';
    else
    if (d = '22') then
      d = 'M';
    else
    if (d = '23') then
      d = 'N';
    else
    if (d = '24') then
      d = 'O';
    else
    if (d = '25') then
      d = 'P';
    else
    if (d = '26') then
      d = 'Q';
    else
    if (d = '27') then
      d = 'R';
    else
    if (d = '28') then
      d = 'S';
    else
    if (d = '29') then
      d = 'T';
    else
    if (d = '30') then
      d = 'U';
    else
    if (d = '31') then
      d = 'V';

    if (rcp_version < 10) then
      v = cast(rcp_version as varchar(1));

-- if (is_kredit = 1) then
--begin
    if (rcp_version = 10) then
      v = 'A';
    if (rcp_version = 11) then
      v = 'B';
    else
    if (rcp_version = 12) then
      v = 'C';
    else
    if (rcp_version = 13) then
      v = 'D';
    else
    if (rcp_version = 14) then
      v = 'E';
    else
    if (rcp_version = 15) then
      v = 'F';
    else
    if (rcp_version = 16) then
      v = 'J';
    else
    if (rcp_version = 17) then
      v = 'H';
    else
    if (rcp_version = 18) then
      v = 'I';
    else
    if (rcp_version = 19) then
      v = 'J';
    else
    if (rcp_version = 20) then
      v = 'K';
    else
    if (rcp_version = 21) then
      v = 'L';
    else
    if (rcp_version = 22) then
      v = 'M';
    else
    if (rcp_version = 23) then
      v = 'N';
    else
    if (rcp_version = 24) then
      v = 'O';
    else
    if (rcp_version = 25) then
      v = 'P';
--end
--else
--begin
    /*  if (RCP_VERSION = 1) then V = 'Q'; else
                if (RCP_VERSION = 2) then V = 'R'; else
                if (RCP_VERSION = 3) then V = 'S'; else
                if (RCP_VERSION = 4) then V = 'T'; else
                if (RCP_VERSION = 5) then V = 'U'; else
                if (RCP_VERSION = 6) then V = 'V'; else
                if (RCP_VERSION = 7) then V = 'X'; else
                if (RCP_VERSION = 8) then V = 'Y'; else
                if (RCP_VERSION = 9) then V = 'Z';    */
--end
    rcp_filename = '47L' || 'U' || y || m || d || v; -- формирую название файла
  end
  else -- пакет по заявкам
  begin
    if (char_length(m) < 2) then
      m = '0' || m;
    if (char_length(d) < 2) then
      d = '0' || d;

    rcp_filename = '47L' || '_' || y || m || d || '_' ||(case
                                                           when char_length(rcp_version) = 4 then rcp_version
                                                           when char_length(rcp_version) = 3 then '0' || rcp_version
                                                           when char_length(rcp_version) = 2 then '00' || rcp_version
                                                           when char_length(rcp_version) = 1 then '000' || rcp_version
                                                         end);
  end

  insert into rcib_pocket (rcp_id, rcp_filename, rcp_date_begin, rcp_date, rcp_version, rcp_is_kredit, rcp_base_rcp_id,
                           rcp_type)
  values (:rcp_id, :rcp_filename, :rcp_date_begin, :pocket_date, :rcp_version, :is_kredit, :base_rcp_id, :rcp_type);

  suspend;
end
