CREATE PROCEDURE PRINT_RESULT_SOFR(
    OBJECT_KIND INTEGER,
    OBJECT_ID ID_TYPE)
RETURNS (
    REP_FIO STR_100,
    REP_DTBIRTH DATE,
    REP_CITIZEN D_STR_20,
    REP_KONTRACT_NO D_STR_20,
    REP_DISB_DATE D_DATE,
    REP_KND_DATETIME D_DATETIME,
    REP_CHECK_BKI_DATE D_DATE,
    REP_CHECK_BKI_TXT D_STR_20,
    REP_FIO_RES_TXT D_STR_20,
    REP_DATEBIRTH_RES_TXT D_STR_20,
    REP_DUL_RES_TXT D_STR_20,
    REP_ADDRES_RES_TXT D_STR_20,
    REP_LOAN_RES_TXT D_STR_20,
    REP_FPS_NAME_FIO D_STR_100,
    REP_FPS_NAME_FIO_TXT D_STR_100,
    REP_CELL_MAIN_TXT D_STR_100,
    REP_CELL_MAIN_TXT_ADD D_STR_100,
    REP_VIDEO_SERTIF_DATE D_DATETIME,
    REP_VIDEO_SERTIF_TXT D_STR_20,
    REP_LAST_EMAIL D_STR_CODE,
    REP_LOCATION D_STR_255,
    REP_LOCATION_ADRESS D_STR_255,
    REP_EMULYATOR D_STR_LONG,
    REP_CLIENT_TXT D_STR_LONG,
    REP_SOCIAL_NETWORK D_STR_2048,
    REP_ANALIZ_PARAMS D_STR_255,
    REP_CHECK_DOC_TXT D_STR_100,
    REP_CHECK_DOC_COMMENT D_STR_100,
    KND_VALUE_STR D_STR_100)
AS
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable k_id id_type;
declare variable criteria d_str_255;
declare variable bki_text d_str_8100;
declare variable db_text d_str_8100;
declare variable res_text d_str_long;
declare variable critical_level d_smallint;
declare variable max_is_local d_smallint;
declare variable last_is_local d_bool;
declare variable ipr_is_local d_bool;
declare variable ok_cell_main d_str_20;
declare variable knd_value d_smallint;
declare variable country d_str_255;
declare variable k_fact_disb_date d_date;
declare variable last_datime d_datetime;
declare variable device_kind d_integer;
declare variable device_id d_str_code;
declare variable clients_cnt d_integer;
declare variable overdue_clients_cnt d_integer;
declare variable c_name_fio d_str_255;
declare variable list_fio d_str_long;
declare variable kontract_no_list d_str_long;
declare variable n d_smallint;
declare variable rd_id d_integer;
declare variable rd_name d_str_255;
declare variable res_value d_str_100;
declare variable res_value_status d_str_100;
declare variable rep_cell_main_datetime d_datetime;
declare variable res_critical_level d_smallint;
declare variable check_result d_integer;
declare variable c_pass_no varchar(30);
begin
  if (object_kind = 9) then
    k_id = object_id;
  else
    exit;

  if (k_id is null) then
    exit;

  select knc.kc_id, knc.kc_from_c_id
  from knclient knc
  where knc.kc_from_k_id = :k_id and
        knc.kc_amount_approved > 0
  into :kc_id, c_id;

  select c_name_fio, c_birth_date, k.k_kontract_no, r.rd_name, k.k_fact_disb_date, c_pass_no
  from knclient knc
  inner join kontract k on k.k_id = knc.kc_from_k_id
  inner join clients cl on c_id = kc_from_c_id
  inner join refdata r on r.rd_id = cl.c_citizen
  where kc_from_k_id = :k_id and
        knc.kc_amount_approved > 0
  into :rep_fio, :rep_dtbirth, :rep_kontract_no, rep_citizen, k_fact_disb_date, c_pass_no;

  select ol.ol_datetime
  from operkred ok
  inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
  where ok.ok_from_k_id = :k_id and
        ok.ok_type = 8 and
        ok.ok_ol_is_active = 1
  into :rep_disb_date;

  rep_knd_datetime = null;
  knd_value_str = null;
  select cast(knd.knd_value as d_smallint), knd.knd_datetime, trim((
             case cast(knd.knd_value as d_smallint)
               when -1 then 'Опрос не пройден'
               when 1 then 'Опрос успешно пройден'
             end))
  from knclient_data knd
  where knd_from_kc_id = :kc_id and
        knd_from_el_id = f_enum_id('knclient_data', 'bki_survey_res')
  order by knd_id desc
  rows 1
  into :knd_value, rep_knd_datetime, knd_value_str;

  --5.8.1
  if (rep_citizen = 'Россия') then
  begin
    rep_check_doc_txt = 'Проводилась';
    rep_check_doc_comment = 'Гражданин РФ';
  end
  else
  begin
    rep_check_doc_comment = 'Иностранный гражданин';
    select check_result
    from hp_check_fms_passport(:c_pass_no)
    into :check_result;
    if (check_result = 1) then
      rep_check_doc_comment = 'Проверка ФМС:' || ' НАЙДЕН';
    else


      rep_check_doc_comment = 'Проверка ФМС:' || ' не найден';
  end

  /*--5.8.2
  rep_check_bki_date = null;
  rep_check_bki_txt = 'Не осуществлен';
  select eil_status_datetime
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eic.eic_code = 'equifax_40' and
        eil_from_c_id = :c_id and
        cast(eil_status_datetime as d_date) <= :k_fact_disb_date
        order by eil_status_datetime desc
        rows 1
  into :rep_check_bki_date;

  if (rep_check_bki_date is not null) then
    rep_check_bki_txt = 'Осуществлен';  */

  --5.8.3
  rep_fps_name_fio = null;
  rep_fps_name_fio_txt = 'Проверка не проведена';

  select fcn_name
  from fps_loan_banks
  left join fps_client_banks on fbl_from_fcb_id = fcb_id
  left join fps_client_names on fcn_from_fcb_id = fcb_id
  where flb_from_kc_id = :kc_id and
        fcn_status = 2
  order by fcn_id desc
  rows 1
  into rep_fps_name_fio;
  if (rep_fps_name_fio is not null) then
    rep_fps_name_fio_txt = 'Проверка проведена – ФИО совпадает';

  --5.8.4
  rep_cell_main_datetime = null;
  rep_cell_main_txt = 'Проверка не проведена';
  rep_cell_main_txt_add = '';

  select res_datetime
  from aarep_object_one_continfo(0, :c_id, f_get_rd_id('ContactInfo.Cell_Main'), 0, 1000000000000)
  where res_value_status = 'Успешно подтвержден'
  into :rep_cell_main_datetime;

  if (rep_cell_main_datetime is not null) then
  begin
    rep_cell_main_txt = 'Проверка проведена – номер используется клиентом';
    rep_cell_main_txt_add = 'СМС получена, номер верифицирован ' || f_datetimetostr(rep_cell_main_datetime);
  end

  --5.8.10
  rep_social_network = '';
  for select rd_id
      from refdata
      where rd_is_active = 1 and
            rd_main_code = 'ContactInfo' and
            rd_code in ('ContactInfo.Skype', 'ContactInfo.Whatsapp', 'ContactInfo.Viber', 'ContactInfo.Telegram', 'ContactInfo.Imo', 'ContactInfo.OK', 'ContactInfo.Facebook', 'ContactInfo.VK', 'ContactInfo.Insta')
      order by rd_orderno, rd_id
      into :rd_id
  do
  begin
    for select rd_name, res_value, res_value_status
        from aarep_object_one_continfo(0, :c_id, :rd_id, 0, 1000000000000)
        into :rd_name, res_value, res_value_status
    do
    begin
      rep_social_network = rep_social_network || rd_name || ':' || res_value || ':' || res_value_status || '; ';
    end
  end

  for select criteria, bki_text, db_text, res_text, critical_level
      from hp_client_metrics_bki(:kc_id, current_date, '')
      into :criteria, bki_text, db_text, res_text, critical_level
  do
  begin
    if (criteria = 'БКИ. ФИО.') then
      rep_fio_res_txt = res_text;
    if (criteria = 'БКИ. Дата рождения.') then
      rep_datebirth_res_txt = res_text;
    if (criteria = 'БКИ. ДУЛ.') then
      rep_dul_res_txt = res_text;
    if (criteria = 'БКИ. Адреса.') then
      rep_addres_res_txt = res_text;
    if (criteria = 'БКИ. Количество активных займов не в Компании за 7 дней:') then
      rep_loan_res_txt = res_text;

    --5.8.2
    if (criteria = 'БКИ.') then
    begin
      if (res_text = 'КИ не получена') then
      begin
        rep_check_bki_date = null;
        rep_check_bki_txt = 'Не осуществлен';
      end
      else
      begin
        select eil_status_datetime
        from extinfolist eil
        inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
        where eic.eic_code = 'equifax_40' and
              eil_from_c_id = :c_id and
              cast(eil_status_datetime as d_date) <= :k_fact_disb_date
        order by eil_status_datetime desc
        rows 1
        into :rep_check_bki_date;

        if (rep_check_bki_date is not null) then
          rep_check_bki_txt = 'Осуществлен';
      end
    end

  end
  ok_cell_main = null;
  rep_last_email = null;
  rep_cell_main_txt = '';
  ok_cell_main = f_cont_info('clients', :c_id, 'cell_main', 'ok');
  rep_last_email = f_cont_info('clients', :c_id, 'email', 'last');

  if (ok_cell_main is not null) then
    rep_cell_main_txt = 'Проверка проведена – номер используется клиентом';

  rep_video_sertif_date = null;
  rep_video_sertif_txt = 'Не подтверждено';
  select cds.cds_create_datetime
  from clientdocs cds
  inner join passkind on pk_id = cds_from_pk_id
  inner join passdoctype on pdt_id = pk_from_pdt_id and
        pdt_code = 'video_sertif'
  where cds_from_c_id = :c_id and
        cast(cds.cds_create_datetime as d_date) <= :rep_disb_date
  order by cds.cds_id desc
  rows 1
  into :rep_video_sertif_date;

  if (rep_video_sertif_date is not null) then
    rep_video_sertif_txt = 'Подтверждено';

  -- Устройство. Местоположение. -------------------------
  res_critical_level = 0;
  db_text = '';
  res_text = '';
  critical_level = 0;
  max_is_local = 0;
  last_is_local = null;
  country = '';
  n = 1;
  rep_location_adress = '';

  for select coalesce(ipr_country, '<Страна неизвестна>'),
             coalesce(ipr_region || ', ', '') || coalesce(ipr_place, ''), coalesce(ipr_is_local, 0),
             max(cdv_update_datetime)
      from clientdevice, hp_get_ip_info(cdv_value)
      where cdv_from_el_id = f_enum('client_device', 'ip_address', 'id') and
            cdv_from_c_id = :c_id and
            cdv_create_datetime < :k_fact_disb_date + 1
      group by 1, 2, 3
      order by 4 desc
      rows 5
      into country, db_text, ipr_is_local, last_datime
  do
  begin
    if (n = 1) then
      rep_location_adress = country;
    last_is_local = coalesce(last_is_local, ipr_is_local);
    max_is_local = maxvalue(ipr_is_local, max_is_local);
    db_text = f_datetimetostr(last_datime) || ' ' || iif(ipr_is_local = 0, country, db_text);
    n = n + 1;
  end

  if (last_is_local is null) then
  begin
    critical_level = 0;
    res_text = 'нет данных';
  end
  else
  begin

    critical_level = case
                       when last_is_local = 1 then 0 -- последнее местоположение у нас
                       when max_is_local = 1 then 1 -- есть наша страна
                       else 2 -- все не наша страна
                     end;

    res_text = case critical_level
                 when 0 then 'Последнее местоположение в стране'
                 when 1 then 'Последнее местоположение не в стране'
                 else 'Местоположение в стране не зафиксировано'
               end;
  end
  rep_location = res_text;
  res_critical_level = maxvalue(critical_level, res_critical_level);
  ------------------  Устройство. Эмулятор.

  critical_level = 0;

  res_text = 'нет данных';

  select cdv_value * 2
  from clientdevice
  where cdv_from_el_id in (f_enum('client_device', 'andr_is_emul', 'id'), f_enum('client_device', 'ios_is_emul', 'id')) and
        cdv_from_c_id = :c_id and
        cdv_create_datetime < :k_fact_disb_date + 1
  order by cdv_update_datetime desc
  rows 1
  into critical_level;

  if (critical_level is null) then
    critical_level = 0;
  else
    res_text = iif(critical_level = 0, 'нет признака', 'есть признак');

  rep_emulyator = res_text;
  res_critical_level = maxvalue(critical_level, res_critical_level);

  ------- Устройство. Клиенты.

  res_text = 'нет данных';

  select cdv_from_el_id, cdv_value
  from clientdevice
  where cdv_from_el_id in (f_enum('client_device', 'andr_device_id', 'id'), f_enum('client_device', 'ios_id', 'id')) and
        cdv_from_c_id = :c_id and
        cdv_create_datetime < :k_fact_disb_date + 1
  order by cdv_update_datetime desc
  rows 1
  into device_kind, device_id;

  clients_cnt = 0;
  overdue_clients_cnt = 0;

  if (device_kind is not null) then
  begin
    for select c_name_fio, list(distinct iif(oks_delinq_date < :k_fact_disb_date, k_kontract_no, ''))
        from oksummary, knclient, clientdevice, kontract, clients
        where :k_fact_disb_date between oks_date_begin and oks_date_end and
              cdv_from_el_id = :device_kind and
              cdv_value = :device_id and
              cdv_from_c_id <> :c_id and
              cdv_from_c_id = kc_from_c_id and
              oks_from_k_id = kc_from_k_id and
              k_id = kc_from_k_id and
              c_id = kc_from_c_id
        group by 1
        into c_name_fio, kontract_no_list
    do
    begin
      clients_cnt = clients_cnt + 1;
      if (kontract_no_list <> '') then
      begin
        overdue_clients_cnt = overdue_clients_cnt + 1;
        list_fio = list_fio || c_name_fio || ' ' || kontract_no_list || ';';
      end
    end
    critical_level = iif(clients_cnt > 4, 2, 0);

    res_text = f_format('Количество активных клиентов с таким же ID устройства: %1, из них %2 в просрочке: %3', clients_cnt, overdue_clients_cnt, list_fio);
    res_critical_level = maxvalue(critical_level, res_critical_level);
  end
  rep_client_txt = res_text;

  --5.8.5
  rep_analiz_params = iif(res_critical_level = 0, 'Проверка проведена – полное совпадение', 'Проверка проведена – требует внимания');
  suspend;

end
