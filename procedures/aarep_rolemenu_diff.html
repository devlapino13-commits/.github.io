CREATE PROCEDURE AAREP_ROLEMENU_DIFF(
    CURR_ROLE_ID DECIMAL(18,0),
    NEW_ROLE_ID DECIMAL(18,0))
RETURNS (
    RN_MENU_NAME VARCHAR(50))
AS
declare variable uniqtab_rolename integer;
declare variable uniqtab_menulist integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;

  for select ml_code
      from menulist, permissions main
      left join permissions sec on (sec.pm_object_kind = :uniqtab_rolename and
            sec.pm_object_id = :curr_role_id and
            sec.pm_subject_kind = :uniqtab_menulist and
            sec.pm_subject_id = main.pm_subject_id and
            sec.pm_is_delete = 0)
      where sec.pm_object_id is null and
            main.pm_object_kind = :uniqtab_rolename and
            main.pm_object_id = :new_role_id and
            main.pm_is_delete = 0 and
            main.pm_subject_id = ml_id
      into :rn_menu_name
  do
  begin
    suspend;
  end
end
