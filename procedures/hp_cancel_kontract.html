CREATE PROCEDURE HP_CANCEL_KONTRACT(
    K_ID D_ID,
    UN_ID D_ID)
AS
declare variable M_PARAM D_INTEGER;
declare variable DB_ID D_INTEGER;
declare variable MSG_KIND D_STR_CODE;
declare variable C_ID D_ID;
declare variable K_KONTRACT_NO STR_20;
declare variable C_NAME_FIO STR_255;
declare variable MSG_MAIL_TO STR_255;
declare variable RES_TEXT STR_255;
declare variable mcl_project d_integer;
declare variable msg_sender_code d_str_code;
begin
   db_id = f_get_db_id();
   m_param = 10; -- Отказ по инициативе клиента

  if (exists(select *
             from kontract
             where k_id = :k_id and
                   k_status <> -1)) then
  begin
    insert into kontreslist (krl_id, krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                             krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
    values (-1, :m_param, :un_id, :k_id, 0, null, current_timestamp, current_date, current_date, 0);

    rdb$set_context('USER_TRANSACTION', 'FREEZE_APPROVE_CANCEL_K_ID', :k_id);

    update kontract
    set k_status = -1,
        k_user_id = :un_id
    where k_id = :k_id;

    for select kc_from_c_id, k_kontract_no , c_name_fio
        from knclient
        inner join kontract on k_id = kc_from_k_id
        inner join clients on  kc_from_c_id = c_id
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, k_kontract_no ,c_name_fio
    do
    begin
      msg_kind = f_get_msg_kind(c_id, 'hp_cancel_kontract', '', db_id);

      -- пуш для клиента
      if (msg_kind = 'push') then
      begin
        execute procedure mc_send_push('clients', c_id, 'Отказ от займа',
            'Вы успешно отказались от займа по договору №' || k_kontract_no, 365);
      end
      if (msg_kind = 'sms') then
      begin
        select cont_value
        from hp_contdata_get_cell_main(0, :c_id)
        into msg_mail_to;

        mcl_project = f_get_mcl_project(f_objkind('kontract', 'id'), :k_id);
        msg_sender_code = f_enum_code('sms_sender', iif (:mcl_project = 2, 1, 0));

        f_send_sms(msg_mail_to, 'Вы успешно отказались от займа по договору №' || k_kontract_no,
                   'clients', c_id, '', 365, 1000000000000, 0, msg_sender_code);

      end
      -- пуш для кредитного специалиста
      select un_id
      from kontstate
      inner join username on un_from_c_id = ks_kc_id
      where ks_from_k_id = :k_id
      order by ks_number desc
      rows 1
      into :un_id;

      if (un_id is not null) then
      begin
        res_text = 'Клиент ' || coalesce(c_name_fio,'') || ' отказался от займа по договору № ' || k_kontract_no;

        f_send_push('username', :un_id, 'Отказ клиента от займа', :res_text, 'HP_CANCEL_KONTRACT_NEW', 7, 1000000000000);
      end
    end
  end
end
