CREATE PROCEDURE AAREP_TRAINING_ON_PODFT(
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    INFO_JOBNAME STR_100,
    C_NAME_FIO STR_100,
    INFO_START_WORK DATE,
    TRAINING_DATE DATE)
AS
declare variable client_id d_id;
declare variable rep_year d_integer;
begin
  -- получаю список всех действующих сотрудников на дату, в разрезе офиса
  rep_year = extract(year from current_date);

  for select ps_from_c_id, reg_code || ' ' || reg_name, jl_name
      from hr_pstatus ps, region, joblist
      where ps_status = 1 and
            ps.ps_is_working = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id
      into :client_id, reg_name, info_jobname
  do
  begin
    -- получаю дату последнего приема на работу    board 1781
    select max(ps_date_begin)
    from hr_pstatus ps, hr_orders ho
    where ps_status = 1 and
          ps_is_working = 1 and
          ps_from_c_id = :client_id and
          ps.ps_from_ho_id = ho.ho_id and
          ho.ho_order_type = 1 --беру только приемы на работу
    into :info_start_work;

    -- Дата проведения тренинга, каждые два года с даты найма
    -- Догоняем год даты найма до года отчета
    training_date = info_start_work;
    while (extract(year from training_date) < rep_year) do
      training_date = dateadd(year, 2, training_date);
    -- +/- 10 дней от даты отчета, нанятых в этом году или позже исключаем
    if (training_date between :rep_date_end - 10 and :rep_date_end + 10 and extract(year from info_start_work) < rep_year) then
    begin
      select c_name_fio
      from clients
      where c_id = :client_id
      into c_name_fio;
      suspend;
    end
  end
end
