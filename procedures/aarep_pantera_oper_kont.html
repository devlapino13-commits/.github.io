CREATE PROCEDURE AAREP_PANTERA_OPER_KONT(
    REP_DATE DATE,
    OPER STR_100,
    PROJEC_TYPE INTEGER,
    SHOW_CLOSED SMALLINT)
RETURNS (
    OPER_CALLED STR_100,
    KF_NAME STR_100,
    KONTRACT_NO STR_100,
    K_ID ID_TYPE,
    CL_CALL_QTY INTEGER,
    CONT_CALL_QTY INTEGER,
    LAST_CALL_DATE DATE,
    KS_STATE INTEGER)
AS
declare variable is_ok smallint;
declare variable sbl_id integer;
begin
  show_closed = coalesce(show_closed, 1);
  oper = trim(coalesce(oper, ''));

  sbl_id = 0;

  if (projec_type = 1) then
    sbl_id = 82; -- Пантера
  if (projec_type = 2) then
    sbl_id = 90; -- КонстантинСудыКР
  if (projec_type = 3) then
    sbl_id = 87; -- Департамент Розыска и Взыскания

  if (sbl_id <> 0) then
  begin
    for select pantk.pant_oper_called, kal.kal_new_kf_name, kont.k_kontract_no, sba.sba_from_k_id,
               pantk.pant_cl_call_qty, pantk.pant_cont_call_qty, pantk.pant_last_call_date, ks.ks_state
        from pantera_kontracts pantk
        inner join kontract kont on kont.k_id = pantk.pant_from_k_id
        inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
        inner join kontstate ks on ks.ks_from_k_id = kont.k_id
        inner join sb_akt sba on sba.sba_from_k_id = kont.k_id
        where ((pantk.pant_oper_called = :oper) or
              (:oper = '')) and
              pantk.pant_date_end >= pantk.pant_date_begin and
              :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
              :rep_date between kal_date and kal_date_end and
              sba.sba_date_end >= :rep_date and
              sba.sba_date_begin <= sba.sba_date_end and
              sba_is_delete = 0 and
              sba.sba_from_sbl_id = :sbl_id and
              :rep_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              pantk.pant_date_begin >= sba.sba_date_begin and
              pantk.pant_project_type = :projec_type
        order by pant_oper_called, kal_new_kf_name
        into :oper_called, :kf_name, :kontract_no, :k_id, :cl_call_qty, :cont_call_qty, :last_call_date, :ks_state
    do
    begin
      is_ok = 1;
      if ((ks_state = 2) and
          (show_closed = 0)) then
        is_ok = 0;
      if (is_ok = 1) then
        suspend;
    end
  end
  else
  begin
    for select pantk.pant_oper_called, kal.kal_new_kf_name, kont.k_kontract_no, kont.k_id, pantk.pant_cl_call_qty,
               pantk.pant_cont_call_qty, pantk.pant_last_call_date, ks.ks_state
        from pantera_kontracts pantk
        inner join kontract kont on kont.k_id = pantk.pant_from_k_id
        inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
        inner join kontstate ks on ks.ks_from_k_id = kont.k_id
        where ((pantk.pant_oper_called = :oper) or
              (:oper = '')) and
              pantk.pant_date_end >= pantk.pant_date_begin and
              :rep_date between pantk.pant_date_begin and pantk.pant_date_end and
              :rep_date between kal_date and kal_date_end and
              :rep_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              pantk.pant_project_type = :projec_type
        order by pant_oper_called, kal_new_kf_name
        into :oper_called, :kf_name, :kontract_no, :k_id, :cl_call_qty, :cont_call_qty, :last_call_date, :ks_state
    do
    begin
      is_ok = 1;
      if ((ks_state = 2) and
          (show_closed = 0)) then
        is_ok = 0;
      if (is_ok = 1) then
        suspend;
    end
  end
end
