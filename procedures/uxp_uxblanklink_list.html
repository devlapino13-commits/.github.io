CREATE PROCEDURE UXP$UXBLANKLINK_LIST(
    IN_UBC_ID D_INTEGER,
    IN_ONLY_LINKED D_BOOL)
RETURNS (
    UBL_FROM_UBC_ID D_INTEGER,
    UBL_BLANK_ID D_INTEGER,
    UBL_REQUIRED D_BOOL,
    UBL_ADD_CODE D_STR_CODE,
    RES_SELECTED D_BOOL,
    RES_BLANK_NAME D_STR_255,
    RES_BLANK_MEMO D_STR_255,
    RES_BLANK_LANG D_STR_LANG,
    RES_BLANK_TYPE D_STR_255,
    RES_KIND D_STR_CODE,
    RES_CAN_SELECT D_BOOL,
    BLN_UX_IMPORTED D_BOOL)
AS
begin
  if (in_only_linked is null) then
    in_only_linked = 1;

  for select bln_id, bln_name, bln_memo, bln_lang, bln_ux_imported,
             blt_name
  from blankname, blanktype
  where bln_from_blt_id = blt_id
  order by bln_id
  into :ubl_blank_id, :res_blank_name, :res_blank_memo, res_blank_lang, bln_ux_imported,
       res_blank_type
  do
  begin
    if (ubl_blank_id < 1000) then
      res_kind = 'Бланки';
    else
      res_kind = 'Шаблоны';

    ubl_from_ubc_id = in_ubc_id;
    ubl_required = 0;
    ubl_add_code = '';
    res_selected = 0;
    res_can_select = 1;

    select ubl_required,
           ubl_add_code
    from ux$blanklink
    where ubl_from_ubc_id = :in_ubc_id and
          ubl_blank_id = :ubl_blank_id
    into :ubl_required,
         :ubl_add_code;

    if (row_count = 1) then
      res_selected = 1;
    else
    if (in_only_linked = 1) then
      Continue;

    suspend;
  end
end
