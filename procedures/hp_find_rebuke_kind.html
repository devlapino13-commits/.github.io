Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_FIND_REBUKE_KIND(
    FIND_TEXT D_STR_100)
RETURNS (
    RBK_ID D_INTEGER,
    RBK_CATEGORY_ID D_INTEGER,
    RBK_CODE D_STR_CODE,
    RBK_NAME D_STR_LONG,
    RBK_INDEX D_INTEGER,
    RBK_INACTIVE D_BOOL,
    RBC_NAME D_STR_100)
AS
begin
  for select rbk_id, rbk_category_id, rbk_code, rbk_code || '  - ' || rbk_name, rbk_index, rbk_inactive, rbc_name
  from rebuke_kind, rebuke_categ
  where rbk_category_id = rbc_id and
        current_date between rbc_date_begin and rbc_date_end and
        rbc_inactive = 0 and
        rbk_inactive = 0 and
        rbk_code = :find_text
  into :rbk_id, :rbk_category_id, :rbk_code, :rbk_name, :rbk_index, :rbk_inactive, :rbc_name
  do
    suspend;

  if (row_count > 0) then
    exit;

  for select rbk_id, rbk_category_id, rbk_code, rbk_code || '  - ' || rbk_name, rbk_index, rbk_inactive, rbc_name
  from rebuke_kind, rebuke_categ
  where rbk_category_id = rbc_id and
        current_date between rbc_date_begin and rbc_date_end and
        rbc_inactive = 0 and
        rbk_inactive = 0 and
        (rbk_code starting :find_text or
        rbk_name containing :find_text)
  order by rbk_category_id, rbk_index
  into :rbk_id, :rbk_category_id, :rbk_code, :rbk_name, :rbk_index, :rbk_inactive, :rbc_name
  do
    suspend;
end
