CREATE PROCEDURE CM_KG_0010_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable rd_code str_255;
declare variable business_type d_integer;
declare variable busuness_subtype d_integer;
declare variable kc_loan_target d_integer;
declare variable pr_code d_str_100;
begin
  res_error_is_critical = 2;

  for select rd_code,
             kc_business_type,
             kc_busuness_subtype,
             kc_loan_target,
             c_name_fio,
             pr_code
      from knclient
      inner join clients on c_id = kc_from_c_id
      inner join refdata on rd_id = kc_loan_target
      inner join kontract on k_id = kc_from_k_id
      inner join product on pr_id = k_from_pr_id
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into rd_code,
           business_type,
           busuness_subtype,
           kc_loan_target,
           res_info_text,
           pr_code
  do
  begin
    if (coalesce(business_type, 0) = 0) then
    begin
      res_error_text = 'Не указан тип бизнеса';
      suspend;
    end

    if (coalesce(busuness_subtype, 0) = 0) then
    begin
      res_error_text = 'Не указан субтип бизнеса';
      suspend;
    end

    if (business_type is not null and busuness_subtype is not null) then
    begin
      if (rd_code = 'Loan_Target.Consumer') then -- Потребительский
      begin
        if (business_type <> 6) then -- Другое
        begin
          res_error_text = 'Для цели займа "Потребительский", может быть выбран только тип бизнеса "Другое"';
          suspend;
        end
      end
      else
      begin -- цель займа "Создание бизнеса" или "Развитие бизнеса"
        if (business_type = 6) then -- Другое
        begin
          res_error_text = 'Тип бизнеса "Другое" может быть выбран только для цели "Потребительский"';
          suspend;
        end
      end
    end

    if (coalesce(kc_loan_target, 0) <= 1) then
    begin
      res_error_text = 'Не указана цель кредитования';
      suspend;
    end

  end
end
