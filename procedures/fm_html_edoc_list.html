CREATE PROCEDURE FM_HTML_EDOC_LIST(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable edl_status d_smallint;
declare variable edl_status_str d_str_code;
declare variable edl_id d_id;
declare variable eds_status_datetime d_datetime;
declare variable obj_info d_str_255;
declare variable edl_object_kind d_str_code;
declare variable edl_object_id d_id;
declare variable doc_count d_smallint;
declare variable lang d_str_lang;
declare variable edc_name d_str_100;
declare variable link_request d_str_255;
declare variable link_detail d_str_255;
declare variable has_manual_request d_smallint;
declare variable c_name_fio d_str_100;
declare variable sign_cnt d_smallint;
declare variable total_cnt d_smallint;
declare variable edbs_status d_smallint;
declare variable edbs_status_str d_str_code;
declare variable edl_date_end d_date;
declare variable db_id d_id;
begin
  obj_kind = f_objkind(obj_kind, 'name');
  lang = f_get_lang_code(un_id, null);
  db_id = f_get_db_id();

  doc_count = (select count(*) from fm_edoc_list_tmp(:obj_kind, :obj_id, :un_id));

  -- Показываем ФИО клиента в заголовке, если передан клиент
  c_name_fio = '';
  if (obj_kind = 'clients') then
    select c_name_fio from clients
    where c_id = :obj_id
    into c_name_fio;

  res_text = (select res_style from fm_html_styles) ||
    f_format('<body><h2>%1</h2>', c_name_fio) ||
    '<table width="100%"><col width="60%"><col width="40%">' ||
    f_format('<tr><th>%1</th>', f_lang(lang, 'Документ')) ||
    f_format('<th>%1</th></tr>', f_lang(lang, 'Действия'));

  -- Отображаем документы, связанные с переданным объектом
  for select edl_id, edl_object_kind, edl_object_id, edl_status, edl_date_end, edc_name
  from fm_edoc_list_tmp(:obj_kind, :obj_id, :un_id), edoc_list, edoc_cfg
  where edl_id = res_edl_id and
        edl_from_edc_id = edc_id
  into edl_id, edl_object_kind, edl_object_id, edl_status, edl_date_end, edc_name
  do
  begin
    select max(eds_status_datetime)
    from edoc_sign
    where eds_from_edl_id = :edl_id
    into eds_status_datetime;

    obj_info = f_objname(edl_object_kind, edl_object_id);
    obj_info = edc_name || iif(obj_info <> '', ', ', '') || obj_info;

    edl_status_str = trim(
      case
        when edl_date_end < current_date then f_lang(lang, 'Недействителен')
        when edl_status= 0 then f_lang(lang, 'Не подписан')
        when edl_status = 1 then f_lang(lang, 'Подписан')
        when edl_status = -1 then f_lang(lang, 'Аннулирован')
      end);

    if (edl_status = 0) then
    begin
      select count(*), sum(iif(eds_status = 1, 1, 0))
      from edoc_sign
      where eds_from_edl_id = :edl_id and
            eds_from_c_id is not null
      into total_cnt, sign_cnt;
      edl_status_str = f_format(f_lang(lang,  'Подписали %1 из %2'), sign_cnt, total_cnt);

      edbs_status = null;
      select min(edbs_status)
      from edoc_blank,  edoc_blank_scan
      where edb_from_edl_id = :edl_id and
            edbs_from_edb_id = edb_id
      into edbs_status;

      edbs_status_str = case edbs_status
        when 0 then f_lang(lang, 'Ожидает закрепления')
        when 1 then f_lang(lang, 'На проверке')
        when 2 then f_lang(lang, 'Документ одобрен')
        when -1 then f_lang(lang, 'Доработка')
        else ''
      end;
    end

    if (db_id = 94401) then
    begin
      link_request = '';
      if (edl_status = 0 and
          doc_count = 1) then
      begin
        has_manual_request = 0;
        if (exists(select *
                   from edoc_blank, edoc_blank_scan
                   where edb_from_edl_id = :edl_id and
                         edbs_from_edb_id = edb_id)) then
          has_manual_request = 1;
  
        if (has_manual_request = 0) then
          link_request = f_uri_sql(f_lang(lang, 'запрос на ручное подписание'), 'edoc_manual_sign', 'edoc_list', edl_id, 'none',
          f_lang(lang, 'вы действительно хотите отправить запрос на ручное подписание?'), '', null, null, null, null, 'dialog', '');
      end
    end

    link_detail = '';
    if (doc_count <> 1) then
      link_detail = f_uri_html(f_lang(lang, 'Детали'), 'edoc_detail',
      left(obj_info, 50), 'edoc_list', edl_id, null, null);

    res_text = blob_append(res_text,
      f_format('<tr><td colspan="2" align="left" class="filled"><b>%1</b></td></tr>', obj_info) ||
      f_format('<tr><td>%1: <b>%2</b><br>', f_lang(lang, 'Статус'), edl_status_str) ||
      iif(edbs_status_str<>'', f_format('%1: <b>%2</b><br>', f_lang(lang,  'Скан'), edbs_status_str), '') ||
      f_format('%1: %2</td>', f_lang(lang, 'Время'), f_datetimetostr(eds_status_datetime)) ||
      '<td align="center">' ||
      iif(link_detail <> '', link_detail, '') ||
      iif(link_request <> '', iif(link_detail<>'','<br><br>', '') || link_request, '') ||
      '</td></tr>');
  end
  res_text = blob_append(res_text, '</table>');

  if (doc_count = 1) then
  begin
    res_text = blob_append(res_text,
      (select res_text from fm_html_edoc_detail(:edl_id, :un_id, 1)));
  end
  suspend;
end
