CREATE PROCEDURE HP_PRODUCTBLANK_PRODUCT(
    REP_DATE D_DATE,
    SHOW_ONLY_ACTIVE D_SMALLINT,
    LINK_KIND D_SMALLINT,
    IN_BLN_ID D_INTEGER)
RETURNS (
    PR_ID D_INTEGER,
    PR_NAME D_STR_255,
    PR_COMMON_NAME D_STR_255,
    PR_CODE D_STR_100,
    PR_DATE_BEGIN D_DATE,
    PR_DATE_END D_DATE,
    RES_CHECKED D_SMALLINT,
    BLN_ID D_INTEGER)
AS
declare variable pr_memo d_str_255;
begin
  bln_id = in_bln_id;

  for select pr_id, pr_name, pr_memo, pr_common_name, lower(pr_code), pr_date_begin, pr_date_end
  from product
  where (:show_only_active = 0 or (:show_only_active = 1 and :rep_date between pr_date_begin and pr_date_end))
  into pr_id, pr_name, pr_memo, pr_common_name, pr_code, pr_date_begin, pr_date_end
  do
  begin
    if (pr_memo is null) then
      pr_memo = '';

    if (pr_date_end >= '31.12.2999') then
      pr_date_end = null;

    if (position(pr_name, pr_memo) = 0 and position(pr_memo, pr_name) = 0) then
      pr_name = pr_name || ' / ' || pr_memo;

    res_checked = 1;

    if (link_kind >= 0) then
    begin
      res_checked = 0;

      if (exists(select *
      from productblank
      where prb_from_pr_id = :pr_id and
            prb_from_bln_id = :bln_id and
            :rep_date between prb_date_begin and coalesce(prb_date_end, :rep_date)))
      then res_checked = 1;
    end

    if (link_kind = 0 or res_checked = 1) then
      suspend;
  end
end
