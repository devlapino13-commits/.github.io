CREATE PROCEDURE FIND_APPLICATE_ID(
    KF_ID DECIMAL(18,0),
    IS_PARALLEL SMALLINT,
    DD DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_CIRCLE INTEGER,
    K_CIRCLE_INNER INTEGER,
    K_REGION_ID INTEGER,
    K_IS_NEW INTEGER,
    K_ID_PRIOR DECIMAL(18,0))
AS
declare variable kt_id decimal(18,0);
declare variable kt_status integer;
declare variable kt_cir integer;
declare variable kt_cirin integer;
declare variable kt_region integer;
begin
  /* Procedure Text */
  k_id_prior = null;
  k_id = null;
  for select ks_state, k_id, k_circle, k_circle_internal, kal_new_reg_id
      from kontract, kontstate, knaktlist
      where k_from_kf_id = :kf_id and
            kal_from_kf_id = :kf_id and
            kal_date <= :dd and
            kal_date_end >= :dd and
            knaktlist.is_delete = 0 and
            ks_from_k_id = k_id and
            cast(ks_date_begin as date) <= :dd and
            ks_date_end >= :dd and
            ks_ol_is_delete = 0
      order by ks_state, k_id desc
      rows 1
      into :kt_status, :kt_id, :kt_cir, :kt_cirin, :kt_region
  do
  begin
    if (kt_id is not null) then
    begin
      if (is_parallel = 0) then
      begin
        if (kt_status <= 0) then
        begin
          k_id = kt_id;
          k_circle = kt_cir;
          k_circle_inner = kt_cirin;
          k_region_id = kt_region;
          k_is_new = 0;
          k_id_prior = 0;
        end
        else
        begin
          k_circle_inner = kt_cirin;
          k_circle = kt_cir + 1;
          k_region_id = kt_region;
          k_id_prior = kt_id;
          k_id = (:kf_id * 1000 + k_circle) * 100 + k_circle_inner;
          k_is_new = 1;
        end
      end
      else
      begin
        k_circle = kt_cir;
        k_circle_inner = kt_cirin + 1;
        k_region_id = kt_region;
        k_id_prior = kt_id;
        k_id = (:kf_id * 1000 + k_circle) * 100 + k_circle_inner;
        k_is_new = 1;
      end
      suspend;
    end
  end
  if (kt_id is null) then
  begin
    /*  Новая папка */
    k_circle = 1;
    k_circle_inner = 1;
    k_region_id = null;
    k_is_new = 1;
    select distinct kal_new_reg_id
    from knaktlist
    where kal_from_kf_id = :kf_id and
          kal_date <= :dd and
          kal_date_end >= :dd and
          is_delete = 0
    into :k_region_id;
    if (k_region_id is not null) then
    begin
      k_id_prior = 0;
      k_id = (:kf_id * 1000 + 1) * 100 + 1;
      suspend;
    end
  end
end
