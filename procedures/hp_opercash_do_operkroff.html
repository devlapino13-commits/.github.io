CREATE PROCEDURE HP_OPERCASH_DO_OPERKROFF(
    LOAN_OL_ID D_ID,
    REPAY_AMOUNT D_CURRENCY,
    ALLOW_OVER BOOLEAN = false)
AS
declare variable pay_os d_currency;
declare variable pay_perc d_currency;
declare variable k_id d_id;
declare variable pr_id d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable office_id d_integer;
declare variable ol_date d_date;
declare variable ol_is_active d_bool;
declare variable ob_detail d_str_255;
declare variable pay_other d_currency;
declare variable c_id d_id;
declare variable zad_os d_currency;
declare variable zad_perc d_currency;
declare variable zad_other d_currency;
declare variable zad_all d_currency;
declare variable pay_all d_currency;
declare variable client_count d_integer;
begin
  -- Суммы распределений по ранее списанному займу
  select max(oko_from_k_id), -sum(oko_os), -sum(oko_perc), -sum(oko_other)
  from operkroff
  where oko_from_ol_id = :loan_ol_id and
        oko_type = 4
  into k_id, pay_os, pay_perc, pay_other;

  if (k_id is null) then
    exit;

  if (pay_os is null) then
    pay_os = 0;

  if (pay_perc is null) then
    pay_perc = 0;

  if (pay_other is null) then
    pay_other = 0;

  if (pay_os = 0 and pay_perc = 0 and pay_other = 0) then
    exit;

  -- Id кредитного продукта договора займа
  select k_from_pr_id, k_kolich_client_approved
  from kontract
  where k_id = :k_id
  into pr_id, client_count;

  -- Офис где создана операция
  select ol_date, 1 - is_delete, ol_from_reg_id, ol_document_client
  from operlist
  where ol_id = :loan_ol_id
  into ol_date, ol_is_active, office_id, c_id;

  if (c_id is not null and client_count > 1) then
  begin
--    update operkroff
--    set oko_from_c_id = :c_id
--    where oko_from_ol_id = :loan_ol_id and
--          oko_from_c_id is distinct from :c_id;

    delete from operkroff
    where oko_from_ol_id = :loan_ol_id and
          oko_type = 4;

    select sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc), sum(oko_other + oko_sh_other)
    from operkroff
    where oko_from_k_id = :k_id and
          oko_from_c_id = :c_id and
          oko_ol_is_active = 1 and
          oko_ol_date <= :ol_date
    into zad_os, zad_perc, zad_other;

    if (zad_os is null) then
      zad_os = 0;

    if (zad_perc is null) then
      zad_perc = 0;

    if (zad_other is null) then
      zad_other = 0;

    zad_all = zad_os + zad_perc + zad_other;

    if ((zad_all < repay_amount) and not allow_over) then
      exception e_message 'Общая сумма долга клиента меньше чем принятая сумма';

    pay_all = repay_amount;

    pay_os = 0;
    pay_perc = 0;
    pay_other = 0;

    if (zad_os < pay_all) then
      pay_os = zad_os;
    else
      pay_os = pay_all;

    pay_all = pay_all - pay_os;

    if (zad_perc < pay_all) then
      pay_perc = zad_perc;
    else
      pay_perc = pay_all;

    pay_all = pay_all - pay_perc;
    pay_other = pay_all;
    pay_all = 0;

    insert into operkroff (oko_from_ol_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc, oko_other,
                           oko_sh_os, oko_sh_perc, oko_sh_other, oko_ol_is_active, oko_ol_date)
    values (:loan_ol_id, :k_id, :c_id, 4, -:pay_os, -:pay_perc, -:pay_other, 0,
            0, 0, 1, :ol_date);
  end

  if (pay_os <> 0) then
  begin
    -- Первая проводка на балансовых счетах
    ob_detail = 'Погашение ранее списанной суммы ОС';

    -- Счет учета ОС по списанным займа
    select res_acc_writeoff
    from hp_make_kredit_acclist(:pr_id, :office_id, 1)
    into acc_dt;

    if (acc_dt = '') then
      exception e_message 'Счет Дт для погашения ранее списанного займа не определен, pr_id: ' || pr_id;

    -- Счет расхода на РППУ по ОС
    select res_account
    from hp_make_kredit_acclist(:pr_id, :office_id, 48)
    into acc_kt;

    if (acc_kt = '') then
      exception e_message 'Счет Кт расхода на РППУ по ОС не определен, pr_id: ' || pr_id;

    insert into operbook (ob_from_ol_id, ob_number, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id, ob_detail)
    values (:loan_ol_id, -1, :acc_dt, :acc_kt, :pay_os, :pay_os,
            :ol_date, :ol_is_active, :office_id, :ob_detail);


    -- Вторая проводка на ВНЕбалансовых счетах
    ob_detail = 'Погашение сумм на внесистемном учете по ОС';

    -- Счет учета ОС по списанным займа, внебалансовые счета
    select res_acc_outbalance
    from hp_make_kredit_acclist(:pr_id, :office_id, 1)
    into acc_dt;

    if (acc_dt = '') then
      exception e_message 'Внебалансовый счет Дт для погашения ранее списанного займа не определен, pr_id: ' || pr_id;

    -- Счет расхода на РППУ по ОС, внебалансовые счета
    select res_acc_outbalance
    from hp_make_kredit_acclist(:pr_id, :office_id, 48)
    into acc_kt;

    if (acc_kt = '') then
      exception e_message 'Внебалансовый счет Кт расхода на РППУ по ОС не определен, pr_id: ' || pr_id;

    insert into operbook (ob_from_ol_id, ob_number, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id, ob_detail)
    values (:loan_ol_id, -1, :acc_dt, :acc_kt, :pay_os, :pay_os,
            :ol_date, :ol_is_active, :office_id, :ob_detail);
  end

  if (pay_perc <> 0) then
  begin
    -- Первая проводка на балансовых счетах
    ob_detail = 'Погашение ранее списанной суммы %%';

    -- Счет учета полученных %% по списанным займа
    select res_acc_writeoff
    from hp_make_kredit_acclist(:pr_id, :office_id, 3) -- Полученные проценты
    into acc_dt;

    if (acc_dt = '') then
      exception e_message 'Счет Дт для погашения %% ранее списанного займа не определен, pr_id: ' || pr_id;

    -- Счет расхода на РППУ по %%
    select res_account
    from hp_make_kredit_acclist(:pr_id, :office_id, 49)
    into acc_kt;

    if (acc_kt = '') then
      exception e_message 'Счет Кт расхода на РППУ по %% не определен, pr_id: ' || pr_id;

    insert into operbook (ob_from_ol_id, ob_number, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id, ob_detail)
    values (:loan_ol_id, -1, :acc_dt, :acc_kt, :pay_perc, :pay_perc,
            :ol_date, :ol_is_active, :office_id, :ob_detail);

    -- Вторая проводка на ВНЕбалансовых счетах
    ob_detail = 'Погашение сумм на внесистемном учете по %%';

    -- Счет учета ОС по списанным займа, внебалансовые счета
    select res_acc_outbalance
    from hp_make_kredit_acclist(:pr_id, :office_id, 3)
    into acc_dt;

    if (acc_dt = '') then
      exception e_message 'Внебалансовый счет Дт для погашения ранее списанного %% не определен, pr_id: ' || pr_id;

    -- Счет расхода на РППУ по ОС, внебалансовые счета
    select res_acc_outbalance
    from hp_make_kredit_acclist(:pr_id, :office_id, 49)
    into acc_kt;

    if (acc_kt = '') then
      exception e_message 'Внебалансовый счет Кт расхода на РППУ по %% не определен, pr_id: ' || pr_id;

    insert into operbook (ob_from_ol_id, ob_number, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id, ob_detail)
    values (:loan_ol_id, -1, :acc_dt, :acc_kt, :pay_perc, :pay_perc,
            :ol_date, :ol_is_active, :office_id, :ob_detail);
  end

  if (pay_other <> 0) then
  begin
    -- Первая проводка на балансовых счетах
    ob_detail = 'Погашение ранее списанной суммы %%';

    -- Счет учета полученных %% по списанным займа
    select res_acc_writeoff
    from hp_make_kredit_acclist(:pr_id, :office_id, 3) -- Полученные проценты
    into acc_dt;

    if (acc_dt = '') then
      exception e_message 'Счет Дт для погашения %% ранее списанного займа не определен, pr_id: ' || pr_id;

    -- Счет дохода по ранее списанным %%
    select res_acc_writeoff
    from hp_make_kredit_acclist(:pr_id, :office_id, 36)
    into acc_kt;

    if (acc_kt = '') then
      exception e_message 'Счет Кт Счет дохода по ранее списанным %% не определен, pr_id: ' || pr_id;

    insert into operbook (ob_from_ol_id, ob_number, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id, ob_detail)
    values (:loan_ol_id, -1, :acc_dt, :acc_kt, :pay_other, :pay_other,
            :ol_date, :ol_is_active, :office_id, :ob_detail);
  end
end
