CREATE PROCEDURE HP_CREDITHISTORY_SAVE(
    REP_C_ID ID_TYPE,
    REP_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    REP_UN_ID ID_TYPE)
AS
declare variable un blob sub_type 1 segment size 80 character set win1251;
begin
  un = rep_blob;
  insert into credithistory (ch_from_c_id, ch_blob, ch_from_un_id)
  values (:rep_c_id, :un, :rep_un_id);
end
