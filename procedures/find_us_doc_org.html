    RES_ID D_ID,

    RES_TEXT D_STR_LONG)

begin
  ref_name = f_json_str_get_value(add_params,'ref_name', 1);

  for select rd_id, rd_str_id, rd_name
      from find_refdata(:find_text, '', '', :ref_name, 1, 0)

      Suspend;
end
