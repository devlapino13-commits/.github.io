CREATE PROCEDURE AAREP_KNCLIENT_PDNLIST_SALDO(
    REP_DATE_END D_DATE)
RETURNS (
    LINE D_INTEGER,
    LOAN_REC_NO D_INTEGER,
    NEW_KONTRACT_NO D_STR_CODE,
    NEW_KC_ID D_ID,
    REC_NO D_INTEGER,
    CREDID D_STR_CODE,
    CREDCURRENCY D_STR_10,
    CREDDATE D_DATE,
    CREDENDDATE D_DATE,
    CREDENDDATEFACT D_DATE,
    CREDUPDATE D_DATE,
    CREDSUMDEBT D_CURRENCY,
    CREDSUMOVERDUE D_CURRENCY,
    CREDDAYOVERDUE D_CURRENCY,
    CREDFULLCOST D_CURRENCY,
    CREDSUM D_CURRENCY,
    CREDPARTNERTYPE_STR D_STR_255,
    CREDTYPE_STR D_STR_255,
    CREDACTIVE_STR D_STR_255,
    CALC_KOEF_PERIOD D_CURRENCY,
    CALC_PERIOD_MONTH D_CURRENCY,
    CALC_MONTH_REPAY D_CURRENCY,
    CALC_IS_CREDIT_CARD D_SMALLINT,
    CALC_EIL_DATE D_DATE,
    CALC_STATUS D_STR_255,
    CALC_IN_AVG D_SMALLINT,
    KC_ID D_ID,
    K_KONTRACT_NO D_STR_CODE,
    AVG_INCOME_DOC D_STR_10,
    AVG_INCOME_BKI D_STR_10,
    KC_STAT_AVG_INCOME D_CURRENCY,
    KC_ANKETA_INCOME D_CURRENCY,
    KC_FINAL_INCOME D_CURRENCY,
    C_NAME_FIO D_STR_100,
    ADL_ADDRESS_STR D_STR_255,
    FINAL_PDN D_CURRENCY,
    RES_LAST_INCOME_PERIOD D_INTEGER,
    RES_INCOME_01 D_CURRENCY,
    RES_INCOME_02 D_CURRENCY,
    RES_INCOME_03 D_CURRENCY,
    RES_INCOME_04 D_CURRENCY,
    RES_INCOME_05 D_CURRENCY,
    RES_INCOME_06 D_CURRENCY,
    RES_INCOME_07 D_CURRENCY,
    RES_INCOME_08 D_CURRENCY,
    RES_INCOME_09 D_CURRENCY,
    RES_INCOME_10 D_CURRENCY,
    RES_INCOME_11 D_CURRENCY,
    RES_INCOME_12 D_CURRENCY,
    EIL_ID D_ID,
    ADL_ID D_ID,
    STR_ID D_INTEGER,
    KC_CHECK_LAST_DATE D_DATE,
    KC_CHECK_COUNT D_INTEGER,
    ANKETA_WORK_INCOME D_CURRENCY,
    ANKETA_WORK_PERIOD D_STR_100,
    ANKETA_EXTRA_INCOME D_CURRENCY,
    PORUCH_STR D_STR_LANG,
    ZALOG_STR D_STR_LANG,
    KPL_IS_ZEROPERC D_SMALLINT,
    KPL_IS_ZEROPERC_STR D_STR_LANG,
    PR_NAME D_STR_255,
    K_FACT_DISB_ONLINE D_BOOL,
    CALC_DATE D_DATE,
    OKS_BALANCE D_CURRENCY,
    OKS_BAL_PERC D_CURRENCY,
    OKS_BAL_KOMMIS D_CURRENCY,
    OKS_DIFF_ESP_BAL D_CURRENCY,
    OKS_DELINQ_DAYS D_INTEGER,
    KOL_ACTIV_DOLG_KLIENT D_INTEGER,
    KOL_ACTIV_ZAIM_KLIENT D_INTEGER,
    KOL_CLOSE_DOLG_KLIENT D_INTEGER,
    KOL_CLOSE_DOLG_ALL D_INTEGER,
    REG_NAME D_STR_100)
AS
declare variable tmp_rec_no d_integer;
declare variable flag_state d_integer;
declare variable k_id id_type;
declare variable k_id_prior id_type;
begin
  tmp_rec_no = 0;
  line = 0;
  k_id_prior = 0;

  for
  select oks_from_kc_id,
         oks_from_k_id,
         oks_balance,
         oks_bal_perc,
         oks_bal_kommis,
         oks_diff_esp_bal,
         oks_delinq_days
  from kn_oksummary
  where :rep_date_end between oks_date_begin and oks_date_end
  and oks_close_state = 0
  into new_kc_id,
       k_id,
       oks_balance,
       oks_bal_perc,
       oks_bal_kommis,
       oks_diff_esp_bal,
       oks_delinq_days
  do
  begin

    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;
      select k_fact_disb_online from kontract
      where k_id = :k_id
      into k_fact_disb_online;
    end

    calc_date = null;
    select kc_date from knclient_pdn
    where kc_id = :kc_id
      and kcp_kind = 0
    into calc_date;


    flag_state = 0;
    pr_name = '';

    for select
      new_kontract_no, rec_no, credid, credcurrency, creddate, credenddate,
      credenddatefact, credupdate, credsumdebt, credsumoverdue, creddayoverdue,
      credfullcost, credsum, credpartnertype_str, credtype_str, credactive_str,
      calc_koef_period, calc_period_month, calc_month_repay, calc_is_credit_card,
      calc_eil_date, calc_status, calc_in_avg, kc_id, k_kontract_no,
      avg_income_doc, avg_income_bki, kc_stat_avg_income, kc_anketa_income,
      kc_final_income, c_name_fio, adl_address_str, final_pdn,
      res_last_income_period, res_income_01, res_income_02, res_income_03,
      res_income_04, res_income_05, res_income_06, res_income_07, res_income_08,
      res_income_09, res_income_10, res_income_11, res_income_12, eil_id, adl_id,
      str_id, kc_check_last_date, kc_check_count, anketa_work_income,
      anketa_work_period, anketa_extra_income, poruch_str, zalog_str, kpl_is_zeroperc, pr_name,
      kol_activ_dolg_klient , kol_activ_zaim_klient, kol_close_dolg_klient,kol_close_dolg_all ,reg_name
    from aarep_knclient_one_pdnlist(:new_kc_id)
    where rec_no = -1
    into
      new_kontract_no, rec_no, credid, credcurrency, creddate, credenddate,
      credenddatefact, credupdate, credsumdebt, credsumoverdue, creddayoverdue,
      credfullcost, credsum, credpartnertype_str, credtype_str, credactive_str,
      calc_koef_period, calc_period_month, calc_month_repay, calc_is_credit_card,
      calc_eil_date, calc_status, calc_in_avg, kc_id, k_kontract_no,
      avg_income_doc, avg_income_bki, kc_stat_avg_income, kc_anketa_income,
      kc_final_income, c_name_fio, adl_address_str, final_pdn,
      res_last_income_period, res_income_01, res_income_02, res_income_03,
      res_income_04, res_income_05, res_income_06, res_income_07, res_income_08,
      res_income_09, res_income_10, res_income_11, res_income_12, eil_id, adl_id,
      str_id, kc_check_last_date, kc_check_count, anketa_work_income,
      anketa_work_period, anketa_extra_income, poruch_str, zalog_str, kpl_is_zeroperc, pr_name,
      kol_activ_dolg_klient , kol_activ_zaim_klient, kol_close_dolg_klient,kol_close_dolg_all,reg_name
    do
    begin
      if (flag_state = 0) then
      begin
        tmp_rec_no = tmp_rec_no + 1;
        loan_rec_no = tmp_rec_no;
      end
      flag_state = 1;

      if (kpl_is_zeroperc = 1) then
        kpl_is_zeroperc_str = 'Да';
      else
        kpl_is_zeroperc_str = 'Нет';

      line = line + 1;
      Suspend;
      loan_rec_no = null;
    end
  end
end
