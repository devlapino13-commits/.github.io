CREATE PROCEDURE HP_FCM_USER_TOPICS(
    UN_ID D_ID,
    MCL_ID D_ID)
RETURNS (
    TOPIC_NAME D_STR_CODE)
AS
declare variable db_id d_integer; /* Код БД */
declare variable fcm_id d_str_255; /* ID Firebase */
declare variable c_id d_id; /* ID клиента */
declare variable db_prefix d_str_lang; /* Префикс БД */
declare variable mcl_project d_integer; /* Проект: МБулак, OIVO... */
declare variable mcl_project_kmp d_integer; /* ID проекта КМП MBulak */
declare variable mcl_project_oivo d_integer; /* ID проекта КМП OIVO */
begin
  -- Сервис: FsSvc
  -- Возвращает группы рассылок (топиков), на которые должен быть подписан пользователь
  -- При добавлении новой группы добавить описание ниже:
  -- Группы рассылок (топики)
  -- ru_kmp_all, kg_kmp_all, az_kmp_all, kz_kmp_all - Все пользователи КМП (по стране)
  -- ru_oivo_all - Все пользователи OIVO КМП (РФ)
  -- ru_kmp_ad  - Все пользователи КМП (РФ), подписавшие согласие на рекламу
  -- ru_kmp_test - Тестовая рассылка программистам РФ

  db_id = f_get_db_id();
  db_prefix = case db_id
      when 41701 then 'kg_'
      when 81001 then 'ru_'
      when 94401 then 'az_'
      when 39801 then 'kz_'
      else null
    end;
  if (db_prefix is null) then
    exit;

  mcl_project_kmp = f_enum_id('mcl_project', 'kmp');
  mcl_project_oivo = f_enum_id('mcl_project', 'oivo');

  if (mcl_id is not null) then
  begin
    select mcl_google_id, mcl_from_c_id, mcl_project
    from mc_login
    where mcl_id = :mcl_id
      and mcl_is_delete = 0
      and coalesce(mcl_google_id, '') <> ''
    into fcm_id, c_id, mcl_project;

    if (coalesce(fcm_id, '') = '') then
      exit;

    if (db_id <> 81001) then
    begin
      -- Везде, кроме РФ одно МП для клиентов
      topic_name = db_prefix || 'kmp_all';
      suspend;
    end

    if (db_id = 81001) then
    begin
      if (mcl_project = mcl_project_kmp) then
      begin
        topic_name = db_prefix || 'kmp_all';
        suspend;
        if (c_id is not null and f_clientcom_allowed(c_id, 1, 0, 0) = 1) then
        begin
          topic_name = db_prefix || 'kmp_ad';
          suspend;
        end
      end
      if (mcl_project = mcl_project_oivo) then
      begin
        topic_name = db_prefix || 'oivo_all';
        suspend;
      end

      -- Тестовая рассылка
      if (exists(select *
        from mc_login
        where mcl_id = :mcl_id
          and mcl_is_delete = 0
          and mcl_login in (
            '79772654344', -- Анатолий �©егольский
            '79778968471', -- Оксана �©егольская
            '79222260870', -- Сергей Кравцов
            '79684828697') -- Ксения Панкова
        )) then
      begin
        topic_name = db_prefix || 'kmp_test';
        suspend;
      end
    end
  end
  else if (un_id is not null) then
  begin
    -- TODO, избавиться от процедуры fm_user_topics, перенести сюда по части топиков
    for select replace(mg_topic, '/topics/', '')
    from fm_user_topics(:un_id)
    into topic_name
    do
      suspend;
  end
end
