CREATE PROCEDURE HP_KONTSTATE_LOAN_STATUS_TMP(
    K_ID DECIMAL(18,0),
    MA_REG_ID INTEGER,
    MA_NO INTEGER,
    REP_DATE DATE,
    CLIENT_ID_LIST VARCHAR(255))
RETURNS (
    C_ID DECIMAL(18,0),
    C_LAST_BONUS_STATUS INTEGER,
    C_CURR_BONUS_STATUS INTEGER)
AS
declare variable sql_text varchar(1024);
declare variable pr_product_kind integer;
declare variable k_from_kf_id id_type;
declare variable client_list_tmp varchar(255);
declare variable n integer;
declare variable c_id_str varchar(20);
begin
  if (client_id_list is null) then
    client_id_list = '';
  if (client_id_list = '') then
  begin
-- Проверка на параллельность
    select pr_product_kind, k_from_kf_id
    from product, kontract
    where k_id = :k_id and
          k_from_pr_id = pr_id
    into :pr_product_kind, :k_from_kf_id;
--Если контракт параллельный то статуc присваивает активного группового кредита(родителя)
    if (:pr_product_kind = 2) then
    begin
      select ks_from_k_id
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin <= :rep_date and
            k_id = ks_from_k_id and
            pr_id = k_from_pr_id and
            pr_product_kind = 1 -- Выборка только групповых кредитов
            and
            k_from_kf_id = :k_from_kf_id
      order by ks_date_begin desc
      rows 1
      into :k_id;
    end
  end
  sql_text = '';

  if (k_id > 0) then
    sql_text = 'select distinct kc_from_c_id from knclient where kc_from_k_id=' || :k_id || ' and kc_amount_approved>0';
  if (ma_no > 0) then
    sql_text = 'select distinct mac_from_c_id from miniappclient where mac_from_reg_id=' || :ma_reg_id || ' ' || ' and mac_from_ma_id=' || :ma_no || ' and mac_amount>0';
  if (client_id_list <> '') then
  begin
-- Если состав клиентов передано в виде списке через ;
-- то обрабатываем каждого и получаем общий статус группы (состава)
    client_list_tmp = client_id_list;
    while (client_list_tmp <> '') do
    begin
      c_id_str = '';
      n = position(';', client_list_tmp, 1);
      if (n = 0) then
        n = position(',', client_list_tmp, 1);

      if (n > 0) then
      begin
        c_id_str = left(client_list_tmp, n - 1);
        client_list_tmp = substring(client_list_tmp from n + 1);
      end
      else
      begin
        c_id_str = client_list_tmp;
        client_list_tmp = '';
      end
      if (c_id_str <> '') then
      begin
        if (sql_text = '') then
          sql_text = 'select c_id from clients where c_id=' || :c_id_str;
        else
          sql_text = sql_text || ' union all select c_id from clients where c_id=' || :c_id_str;
      end
    end
  end
  c_last_bonus_status = 0;
  c_curr_bonus_status = 0;
  if (sql_text <> '') then
  begin
    for execute statement :sql_text
            into :c_id
    do
    begin
      select c_last_bonus_status, c_curr_bonus_status
      from hp_kontstate_client_status(:c_id, :rep_date)
      into :c_last_bonus_status, :c_curr_bonus_status;
      suspend;
    end
  end
end
