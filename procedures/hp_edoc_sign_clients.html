CREATE PROCEDURE HP_EDOC_SIGN_CLIENTS(
    EDOC_KIND D_STR_CODE,
    OBJ_KIND VARCHAR(50),
    OBJ_ID D_ID,
    ADD_CODE D_STR_CODE = null)
RETURNS (
    C_ID D_ID)
AS
declare variable DB_ID D_INTEGER;
declare variable NEED_SUSPEND D_SMALLINT;
declare variable MCL_ID D_ID; /* Для товаров в рассрочку */
declare variable CS_ID D_ID; /* Для товаров в рассрочку */
declare variable TMP_C_ID D_ID;
declare variable OTHER_C_ID D_ID;
declare variable ACR_CASHIER_ID D_ID;
declare variable ACR_PERSON_ID D_ID;
declare variable ACR_DIRECTOR_ID D_ID;
declare variable ACR_CASHIER_DATESIGN D_DATETIME;
declare variable ACR_PERSON_DATESIGN D_DATETIME;
declare variable ACR_DIRECTOR_DATESIGN D_DATETIME;
begin
  --  если obj_id null, возвращаем одну строку с c_id = null, чтобы бланки готовились на уровне объекта
  if (obj_id is null) then
  begin
    c_id = null;
    suspend;
    exit;
  end

  db_id = f_get_db_id();

  add_code = trim(coalesce(add_code, ''));

  edoc_kind = f_edoc_kind(edoc_kind, 'code');
  obj_kind = f_objkind(obj_kind, 'name');


  if (edoc_kind in ('loan.agreement_autosign', 'loan.application_autosign')) then
    add_code = 'auto_sign';

  if (edoc_kind = 'loan.spouses_consent') then
    add_code = 'separate_consent';

  if (obj_kind = 'clients') then
  begin
    c_id = obj_id;
    suspend;
    exit;
  end

  if (obj_kind = 'agreements') then
  begin
    select ag_object_id
    from agreements
    where ag_id = :obj_id and
          ag_object_kind = 0
    into c_id;

    suspend;
    exit;
  end

  if (obj_kind = 'kontract_istinaa_prepay') then
  begin
    for select distinct kip.kip_from_c_id
        from kontract_istinaa_prepay kip
        inner join clients cl on cl.c_id = kip.kip_from_c_id
        where kip.kip_id = :obj_id and
              kip.kip_is_active = 1
        into c_id
    do
    begin
      suspend;
    end
    exit;
  end

  if (obj_kind = 'RECONCILIATION_INFO') then
  begin
    for select distinct ri.ri_from_c_id
        from reconciliation_info ri
        where ri.ri_id = :obj_id
        into c_id
    do
    begin
      suspend;
    end
    exit;
  end

  if (obj_kind = 'kontrefund_disb') then
  begin
    for select kc_from_c_id
        from kontrefund_disb, knclient
        where krd_from_k_id = kc_from_k_id and
              kc_amount_approved > 0 and
              krd_id = :obj_id
        into c_id
    do
      suspend;

    exit;
  end

  if (obj_kind = 'kontract') then
  begin
    if (edoc_kind in ('loan.delinq.notif','restruct.loan.delinq.notif', 'loan.agreement', 'loan.pdn', 'loan.pdn.recalc', 'loan.cancel', 'loan.refinance', 'insurance.application', 'insurance.kyrgyz', 'insurance.kyrgyz.ipf', 'loan_disb_consent', 'ipf.chdp.agreement', 'restruct_application', 'add_loan_consent', 'restruct_approve', 'loan.application_new_conditions', 'loan.creditor_rep', 'autorestruct_notify', 'risk.agreement', 'loan.kid')) then -- RFCKZ-805
    begin
      for select kc_from_c_id
          from knclient
          where kc_from_k_id = :obj_id and
                kc_amount_approved > 0
          into c_id
      do
      begin
        need_suspend = 1;
        if (add_code = 'anketa_ft') then
        begin
          if (f_ft_valid(c_id, current_date) = 1) then -- если анкета валидный, не печатаем для этого клиента
            need_suspend = 0;
        end
        if (need_suspend = 1) then
          suspend;
      end
    end

    if (edoc_kind = 'loan.poruch.agreement') then
    begin
      for select kc_from_c_id
          from knclient
          where kc_from_k_id = :obj_id and
                kc_amount_approved > 0
          into c_id
      do
      begin

        suspend;
      end
      for select z.z_from_c_id
          from zalog z
          where z.z_from_k_id = :obj_id
          into c_id
      do
      begin
        suspend;
      end
    end

    if (edoc_kind = 'loan.collateral') then
    begin
      for select z_from_c_id
          from zalog
          where z_from_k_id = :obj_id and
                z_from_c_id is not null and
                z_type = 1
          into c_id
      do
        suspend;
    end

    if (edoc_kind in ('pdata.agreement.zalog', 'bki.agreement.zalog')) then
    begin
      for select z.z_from_c_id
          from zalog z
          where z.z_from_k_id = :obj_id
          into c_id
      do
        suspend;
    end

    if (edoc_kind = 'loan.agreement_autosign') then
    begin
      --По товарам в рассрочку только индивидуальные кредиты
      --Поэтому запрос не в цикле
      select kc.kc_from_c_id
      from knclient kc
      where kc.kc_from_k_id = :obj_id and
            kc.kc_amount_approved > 0
      into c_id;

      suspend;
    end

    if (edoc_kind = 'information_request2') then
    for
      select
        kc_from_c_id
      from knclient
      where kc_from_k_id = :obj_id
        and kc_amount_approved > 0
    into :c_id
    do
      suspend;

    exit;
  end

  if (obj_kind = 'miniapp') then
  begin
    if (add_code = '') then
    begin
      for select mac_from_c_id
          from miniappclient
          where mac_from_ma_id = :obj_id and
                (mac_amount > 0 or
                (:db_id in (39801, 41701, 94401) and mac_is_poruch = 1)) -- для КЗ, КР и АЗ добавляем и поручителей и лиц дающих согласие
          into c_id
      do
        suspend;
    end
    else
    begin
      if (add_code = 'loan.application.soglasiye') then
      begin
        for select c_id
            from hp_miniapp_soglasiye_list(:obj_id)
            into :c_id
        do
          suspend;
      end

      if (add_code = 'auto_sign') then
      begin
        for select mac.mac_from_c_id
            from miniappclient mac
            where mac.mac_from_ma_id = :obj_id and
                  mac.mac_amount > 0
            into c_id
        do
          suspend;
      end

      --Для товаров в рассрочку/сахарного завода уйдёт отдельная смска по согласию
      if (add_code = 'separate_consent') then
      begin
        for select mac.mac_from_c_id
            from miniappclient mac
            where mac.mac_from_ma_id = :obj_id and
                  mac.mac_is_poruch = -1
            into c_id
        do
          suspend;
      end

      if (add_code = 'loan.application_npdl') then
      begin
        for select mac.mac_from_c_id
            from miniappclient mac
            where mac.mac_from_ma_id = :obj_id and
                  mac.mac_amount > 0
            into c_id
        do
          suspend;
      end

      if (add_code = 'necessary_conditions') then
      begin
        for select mac.mac_from_c_id
        from miniappclient mac
        where mac.mac_from_ma_id = :obj_id
          and mac.mac_amount > 0
        into c_id
        do
          suspend;
      end
    end

    exit;
  end

  if (obj_kind = 'com_list') then
  begin
    if (edoc_kind in ('loan.reschedule', 'loan.reschedule.pdn', 'loan.termination')) then
    begin
      for select kc_from_c_id
          from knclient, com_list
          where cml_id = :obj_id and
                cml_object_kind = f_objkind('kontract', 'id') and
                kc_from_k_id = cml_object_id and
                kc_amount_approved > 0
          into c_id
      do
        suspend;
    end
    exit;
  end

  if (obj_kind = 'kontpsk') then
  begin
    for select kc_from_c_id
        from knclient, kontpsk
        where kps_id = :obj_id and
              kc_from_k_id = kps_from_k_id and
              kc_amount_approved > 0
        into c_id
    do
      suspend;
    exit;
  end

  if (obj_kind = 'ft_anketa_list') then
  begin
    select fal_from_c_id
    from ft_anketa_list
    where fal_id = :obj_id
    into c_id;
    suspend;
    exit;
  end

  if (obj_kind = 'cardaktclient') then
  begin
    select cac_from_c_id
    from cardaktclient
    where cac_id = :obj_id
    into c_id;
    suspend;
    exit;
  end

  if (obj_kind = 'knclient') then
  begin
    select kc.kc_from_c_id
    from knclient kc
    where kc.kc_id = :obj_id
    into c_id;
    suspend;
    exit;
  end

  if (obj_kind = 'expressapp') then
  begin
    select exa_from_c_id
    from expressapp
    where exa_id = :obj_id
    into c_id;
    suspend;
    exit;
  end
  if (obj_kind = 'kontmatag') then
  begin
    --JIRA 1672
    /*
      Клиент получает уведомление с данными по каждому контрагенту.
      Поэтому object_kind = 'kontmatag'
    */
    if (db_id in (39801, 41701) and
        edoc_kind in ('idjara_goods.client_notif', 'act_from_company.agreement')) then
    begin
      for select distinct kc.kc_from_c_id
          from kontmatag kma
          inner join knclient kc on kc.kc_from_k_id = kma.kma_from_k_id
          where kma.kma_id = :obj_id
          into c_id
      do
        suspend;
      exit;
    end

    for select c.c_id
        from clients c
        inner join agreements ag on ag.ag_object_id = c.c_id
        inner join kontmatag kma on kma.kma_from_ag_id = ag.ag_id
        where ag.ag_object_kind = f_objkind('clients', 'id') and
              kma.kma_id = :obj_id
        into c_id
    do
    begin
      suspend;
      exit;
    end
  end
  if (obj_kind = 'finerefund') then
  begin
    select fr.fr_from_c_id
    from finerefund fr
    where fr.fr_id = :obj_id
    into c_id;
    suspend;
    exit;
  end
  if (obj_kind = 'app_return_consult') then
  begin
    select arc.arc_from_c_id
    from app_return_consult arc
    where arc.arc_id = :obj_id
    into c_id;
    suspend;
    exit;
  end
  if (obj_kind = 'cardblockopers') then
  begin
    select cbo.cbo_from_c_id
    from cardblockopers cbo
    where cbo.cbo_id = :obj_id
    into c_id;
    suspend;
    exit;
  end
  if (obj_kind = 'overpayment') then
  begin
    select ovp_from_c_id
    from overpayment
    where ovp_id = :obj_id
    into c_id;
    suspend;
    exit;
  end
  if (obj_kind = 'cardpinchange') then
  begin
    select cph.cph_from_c_id
    from cardpinchange cph
    where cph.cph_id = :obj_id
    into c_id;
    suspend;
    exit;
  end
  if (obj_kind = 'freelancer_akt') then
  begin
    select sd.sd_from_c_id
    from servicesh_edoc sshe
    inner join servicedog sd on sd.sd_id = sshe.sshe_from_sd_id
    where sshe.sshe_id = :obj_id
    into c_id;
    suspend;
    exit;
  end
  if (obj_kind in ('freelancer_kontract', 'probationer_kontract', 'kochegar_kontract', 'add_for_freelancers', 'termination_freelancer')) then
  begin
    select sd.sd_from_c_id
    from servicedog sd
    where sd.sd_id = :obj_id
    into c_id;
    suspend;
    exit;
  end

  if (obj_kind = 'mc_purchase') then
  begin
    select mp.mp_from_mcl_id, mp.mp_from_cs_id
    from mc_purchase mp
    where mp.mp_id = :obj_id
    into mcl_id, cs_id;

    select mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_id = :mcl_id and
          mcl.mcl_is_delete = 0
    order by mcl.mcl_datetime desc
    rows 1
    into c_id;
    suspend;

    select cs.cs_from_c_id
    from clientshops cs
    where cs.cs_id = :cs_id and
          cs.cs_is_delete = 0
    into c_id;
    suspend;
    exit;
  end

  if (obj_kind = 'hr_orders') then
  begin
    for select ps_from_c_id
        from hr_pstatus
        where ps_from_ho_id = :obj_id
        union
        select hrv_from_c_id
        from hr_voctravel
        where hrv_from_ho_id = :obj_id
        into c_id
    do
      suspend;

    exit;
  end

  if (obj_kind = 'servicedog') then
  begin
    for select sd_from_c_id
        from servicedog
        where sd_id = :obj_id
        into c_id
    do
      suspend;

    exit;
  end

  if (obj_kind = 'opercash') then
  begin
    for select iif(opc_sign = 0, csp_cashier_id, opc_from_c_id), iif(opc_sign = 0, opc_manager_id, null), opc_other_client_id
        from opercash, cashplace
        where opc_id = :obj_id and
              opc_from_csp_id = csp_id
        into :c_id, tmp_c_id, other_c_id
    do
    begin
      if (other_c_id is not null) then
        c_id = other_c_id; -- RFCKG-3927

      suspend;

      if (tmp_c_id is not null) then
      begin
        c_id = tmp_c_id;
        suspend;
      end
    end

    exit;
  end

  if (obj_kind = 'transfertmc') then
  begin
    for select iif(:obj_id < 0, trt_mols_id, trt_molr_id)
        from transfertmc
        where trt_id = abs(:obj_id) and
              trt_status < 2
        into :c_id
    do
      suspend;
    exit;
  end

  if (obj_kind = 'fuelgenreg') then
  begin
    for select iif(:obj_id < 0, fg_driver_id, fg_manager_id)
        from fuelgenreg
        where fg_id = abs(:obj_id) and
              fg_status = 0
        into :c_id
    do
      suspend;
    exit;
  end

  if (obj_kind = 'accesscashroom') then
  begin
    for
      select
        acr_cashier_id,
        acr_person_id,
        acr_director_id,
        acr_cashier_datesign,
        acr_person_datesign,
        acr_director_datesign
      from accesscashroom
      where acr_id = :obj_id
        and acr_status in (1, 2)
    into
      :acr_cashier_id,
      :acr_person_id,
      :acr_director_id,
      :acr_cashier_datesign,
      :acr_person_datesign,
      :acr_director_datesign
    do
    begin
      if (acr_cashier_datesign is null) then
      begin
        c_id = acr_cashier_id;
        suspend;
      end
      if (acr_person_datesign is null) then
      begin
        c_id = acr_person_id;
        suspend;
      end
      if (acr_director_datesign is null) then
      begin
        c_id = acr_director_id;
        suspend;
      end
    end
    exit;
  end

  if (obj_kind = 'cashcheckmain') then
  begin
    for
      select
        ccm_auditor as c_id
      from cashcheckmain
      where ccm_id = :obj_id
        and ccm_auditor > 0
          and ccm_auditor_datesign is null
      union
      select
        ccm_cashier_curr as c_id
      from cashcheckmain
      where ccm_id = :obj_id
        and ccm_cashier_curr > 0
          and ccm_cashier_curr_datesign is null
      union
      select
        ccm_cashier_new as c_id
      from cashcheckmain
      where ccm_id = :obj_id
        and ccm_cashier_new > 0
          and ccm_cashier_new_datesign is null
      union
      select
        ccp_from_c_id as c_id
      from cashcheckpart
      where ccp_from_ccm_id = :obj_id
        and ccp_datesign is null
    into
      :c_id
    do
    begin
      suspend;
    end
    exit;
  end

  if (obj_kind = 'signdoc') then
  begin
    for
      select
        c_id
      from hp_signmember_list(:obj_id)
    into
      :c_id
    do
    begin
      suspend;
    end
    exit;
  end

  if (obj_kind = 'info_loan_client') then
  begin
    select ilc_from_c_id
    from info_loan_client
    where ilc_id = :obj_id
    into c_id;

    suspend;
    exit;
  end

  if (obj_kind = 'invmember') then
  begin
    select imem.imem_from_c_id
    from invmember imem
    where imem.imem_id = :obj_id
    into :c_id;
    suspend;
    exit;
  end

  if (obj_kind = 'cashbookmain') then
  begin
    select cbm_manager_id
    from cashbookmain
    where cbm_id = :obj_id
    into :c_id;
    suspend;
    exit;
  end

  if (obj_kind = 'sud_mainreq') then
  begin
    for select kc_from_c_id
    from sud_loanclient, knclient
    where slc_from_smr_id = :obj_id
      and kc_id = slc_from_kc_id
    into :c_id
    do
      suspend;
    exit;
  end

  if (obj_kind = 'overpay_main') then
  begin
    select opm_from_c_id
    from overpay_main
    where opm_id = :obj_id
    into :c_id;
      suspend;
    exit;
  end

  if (obj_kind = 'smevpass_msg') then
  begin
    for select sbc_from_c_id
    from smevpass_msg, smevpass_blockclients
    where spm_id = :obj_id
      and sbc_id = spm_from_sbc_id
    into :c_id
    do
      suspend;
    exit;
  end

  exception e_default 'Client not found for object: ' || obj_kind;
end
