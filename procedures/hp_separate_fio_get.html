CREATE PROCEDURE HP_SEPARATE_FIO_GET(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    FNAME D_STR_100,
    MNAME D_STR_100,
    SNAME D_STR_100,
    FULL_FIO D_STR_100,
    FNAME_ENG D_STR_100,
    MNAME_ENG D_STR_100,
    SNAME_ENG D_STR_100,
    FULL_FIO_ENG D_STR_100,
    FNAME_LAT D_STR_100,
    MNAME_LAT D_STR_100,
    SNAME_LAT D_STR_100,
    FULL_FIO_LAT D_STR_100,
    REASON D_STR_100,
    R_ONLY D_BOOL)
AS
declare variable C_TYPE D_INTEGER;
begin
  if (f_tmp_const('use_perstemp') = '1' and
    exists(select * from perstemp where pte_from_c_id = :c_id)) then
  begin
    select
      iif(pte_sname is null, upper(left(fl1.fl_name1, 1)) || lower(substring(fl1.fl_name1 from 2)), pte_sname),
      iif(pte_sname is null, upper(left(fl1.fl_name2, 1)) || lower(substring(fl1.fl_name2 from 2)), pte_fname),
      iif(pte_sname is null, upper(left(fl1.fl_name3, 1)) || lower(substring(fl1.fl_name3 from 2)), pte_mname),
      iif(pte_sname is null, upper(left(fl2.fl_name1, 1)) || lower(substring(fl2.fl_name1 from 2)), pte_sname_eng),
      iif(pte_sname is null, upper(left(fl2.fl_name2, 1)) || lower(substring(fl2.fl_name2 from 2)), pte_fname_eng),
      iif(pte_sname is null, upper(left(fl2.fl_name3, 1)) || lower(substring(fl2.fl_name3 from 2)), pte_mname_eng),
      pte_alternat_fio,
      c_type
    from perstemp
      inner join clients
        on c_id = pte_from_c_id
      left join fiolist as fl1
        on fl1.fl_id = c_from_fl_id
      left join fiolist as fl2
        on fl2.fl_id = c_name_eng_fl_id
    where pte_from_c_id = :c_id
    into
      :sname,
      :fname,
      :mname,
      :sname_eng,
      :fname_eng,
      :mname_eng,
      :full_fio_lat,
      :c_type;
  end else
  begin
    select
      upper(left(fl1.fl_name3, 1)) || lower(substring(fl1.fl_name3 from 2)),
      upper(left(fl1.fl_name2, 1)) || lower(substring(fl1.fl_name2 from 2)),
      upper(left(fl1.fl_name1, 1)) || lower(substring(fl1.fl_name1 from 2)),
      upper(left(fl2.fl_name3, 1)) || lower(substring(fl2.fl_name3 from 2)),
      upper(left(fl2.fl_name2, 1)) || lower(substring(fl2.fl_name2 from 2)),
      upper(left(fl2.fl_name1, 1)) || lower(substring(fl2.fl_name1 from 2)),
      upper(left(fl3.fl_name3, 1)) || lower(substring(fl3.fl_name3 from 2)),
      upper(left(fl3.fl_name2, 1)) || lower(substring(fl3.fl_name2 from 2)),
      upper(left(fl3.fl_name1, 1)) || lower(substring(fl3.fl_name1 from 2)),
      c_alternat_fio,
      c_type
    from clients
      left join fiolist as fl1
        on fl1.fl_id = c_from_fl_id
      left join fiolist as fl2
        on fl2.fl_id = c_name_eng_fl_id
      left join fiolist as fl3
        on fl3.fl_id = c_alternat_fl_id
    where c_id = :c_id
    into
      :mname,
      :fname,
      :sname,
      :mname_eng,
      :fname_eng,
      :sname_eng,
      :fname_lat,
      :mname_lat,
      :sname_lat,
      :full_fio_lat,
      :c_type;
  end

  if (c_type = 0) then
  begin
    -- склейка по новому
    select
      c_name_fio,
      c_name_fio_eng
    from hp_prepare_fio (
      :sname,
      :fname,
      :mname,
      :sname_eng,
      :fname_eng,
      :mname_eng,
      1)
    into
      :full_fio,
      :full_fio_eng;

    reason = (
      select
       res_str
      from hp_can_fio_editing(:c_id));

    if (exists(
      select *
      from hp_permission_access_check(
        :un_id,
        'PermActions',
        0,
        'FinStart.Role_CAN_FIO_CHANGE')
      where has_access = 1) or (reason = ''))
    then
      r_only = 0;
    else
      r_only = 1;

    fname = coalesce(fname, '');
    mname = coalesce(mname, '');
    sname = coalesce(sname, '');
    full_fio = coalesce(full_fio, '');
    fname_eng = coalesce(fname_eng, '');
    mname_eng = coalesce(mname_eng, '');
    sname_eng = coalesce(sname_eng, '');
    full_fio_eng = coalesce(full_fio_eng, '');
    fname_lat = coalesce(fname_lat, '');
    mname_lat = coalesce(mname_lat, '');
    sname_lat = coalesce(sname_lat, '');
    full_fio_lat = coalesce(full_fio_lat, '');

    suspend;
  end
end
