CREATE PROCEDURE HP_OPERVAC_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_K_ID D_ID,
    RES_OL_ID D_ID,
    RES_OL_DATE D_DATE,
    RES_DATE_BEGIN D_DATE,
    RES_DATE_END D_DATE,
    RES_PERCENT D_PERCENT,
    RES_PERC_AMOUNT D_CURRENCY,
    RES_MAX_PART_LIMIT D_CURRENCY,
    RES_VAC_DAYS D_INTEGER,
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    K_PERIOD D_CURRENCY,
    K_KONT_LAST_DATE D_DATE)
AS
declare variable ov_vac_begin_days d_integer;
begin
  for select ov_from_k_id
  from opervac, operlist
  where ov_from_ol_id = ol_id and
        operlist.is_delete = 0
  group by ov_from_k_id
  having min(ol_date) between :rep_date_begin and :rep_date_end
  into res_k_id
  do
  begin
    for select ov_from_ol_id, ol_date, ov_perc, ov_vac_days,
               ov_vac_begin_days, ov_max_part_limit,
               k_kontract_no, k_period, k_disb_date, k_kont_last_date
    from kontract, opervac, operlist
    where k_id = :res_k_id and
          ov_from_k_id = k_id and
          ov_from_ol_id = ol_id and
          operlist.is_delete = 0
    order by ol_date, ol_id
    rows 1
    into :res_ol_id, res_ol_date, :res_perc_amount, :res_vac_days,
         :ov_vac_begin_days, :res_max_part_limit,
         :k_kontract_no, :k_period, :k_disb_date, :k_kont_last_date
    do
    begin
      res_percent = null;
      select round(kd_percent_per_day * 365.00000000, 3)
      from kontdata
      where kd_from_ol_id = :res_ol_id and
            kd_from_k_id = :res_k_id and
            kd_ol_date <= :res_ol_date
      order by kd_ol_date desc
      rows 1
      into res_percent;

      res_date_begin = res_ol_date;
      res_date_begin = dateadd(day, ov_vac_begin_days, res_date_begin);
      res_date_end = dateadd(day, res_vac_days, res_date_begin);

      suspend;
    end
  end
end
