CREATE PROCEDURE HP_GC_ANALIZ_DETAIL_QR_VIEW(
    IGA_ID D_INTEGER)
RETURNS (
    IGDQ_ID D_ID,
    IGDQ_REC_NO D_INTEGER,
    IGDQ_FROM_IB_ID D_INTEGER,
    BANK_NAME D_STR_255,
    IGDQ_IS_OUR_CARD D_BOOL,
    IGDQ_CURRENCY D_STR_LANG,
    IGDQ_AMOUNT D_CURRENCY,
    IGDQ_IPC_EMITENT_COMMIS_PERC D_PERCENT,
    IGDQ_IPC_EMITENT_COMMIS_AMOUNT D_CURRENCY,
    IGDQ_TRANS_NO D_STR_10,
    IGDQ_TRANS_DATE D_DATE,
    IGDQ_IS_ERROR D_BOOL,
    IGDQ_ERROR_ID D_INTEGER,
    ERR_NAME D_STR_255)
AS
declare variable object_id d_id;
begin
  for
    select
      igdq_id,
      igdq_rec_no,
      igdq_from_ib_id,
      igdq_is_our_card,
      igdq_currency,
      igdq_amount,
      igdq_ipc_emitent_commis_perc,
      igdq_ipc_emitent_commis_amount,
      igdq_trans_no,
      igdq_trans_date,
      igdq_is_error,
      igdq_error_id
    from ipc_gc_analiz_detail_qr
    where igdq_from_iga_id = :iga_id
  into
    :igdq_id,
    :igdq_rec_no,
    :igdq_from_ib_id,
    :igdq_is_our_card,
    :igdq_currency,
    :igdq_amount,
    :igdq_ipc_emitent_commis_perc,
    :igdq_ipc_emitent_commis_amount,
    :igdq_trans_no,
    :igdq_trans_date,
    :igdq_is_error,
    :igdq_error_id
  do
  begin
    bank_name = null;
    err_name = null;

    if (igdq_from_ib_id > 0) then
      bank_name = (select ib_ipc_bank_name from ipc_banks where ib_id = :igdq_from_ib_id);

    if (igdq_error_id > 0) then
      err_name = (select rd_name from refdata where rd_id = :igdq_error_id);

    suspend;
  end
end
