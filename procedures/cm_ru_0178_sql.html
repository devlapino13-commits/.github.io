CREATE PROCEDURE CM_RU_0178_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable PR_CODE varchar(100);
declare variable K_KOLICH_LGOTA integer;
declare variable K_KOLICH_REPAY integer;
declare variable KF_DISB_DATE D_DATE;
declare variable K_REFUND D_SMALLINT;
begin
  /*
разрешен льготный период в январе и феврале по продукту  «Параллельный»
*/
  select k_kolich_lgota, pr.pr_code, k_disb_date, k_refund
  from kontract kont
  inner join product pr on kont.k_from_pr_id = pr.pr_id
  where kont.k_id = :k_id
  into k_kolich_lgota, :pr_code, kf_disb_date, k_refund;

  if ((:pr_code containing '.par.loyal.' ) and
      exists(select *
             from schedule sh
             where sh.sh_from_k_id = :k_id and
                   sh.sh_id > 0 and
                   sh.sh_os = 0 and 
                   sh.sh_grace = 1 and
                   extract(month from sh_date) not in (1, 2)
      )) then                                                          
  begin
    if (not (k_refund = 1 and kf_disb_date >= '01.03.2022')) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'По продукту "Параллельный" льготный период можно получить только на январь и/или февраль';
      suspend;
    end
  end
end
