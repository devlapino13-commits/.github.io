CREATE PROCEDURE AAREP_DYNAMICS_PORTFOLIO_SSVZ(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KF_OFFICER_NAME VARCHAR(100),
    AA_KC_STAGE VARCHAR(100),
    AA_AMOUNT_ACTIVE_END CURR_TYPE,
    PRIROST_PORTFEL CURR_TYPE,
    AA_JUST_CL_END INTEGER,
    PRIROST_CLI INTEGER,
    PAR CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    AA_AMOUNT_ACTIVE_BEGIN CURR_TYPE,
    AA_JUST_CL_BEGIN INTEGER,
    AA_DELIQ_BALANCE CURR_TYPE,
    AA_DELIQ_BALANCE7 CURR_TYPE,
    AA_DELIQ_BALANCE30 CURR_TYPE,
    REG_NAME VARCHAR(60),
    KOLICH_CLI_DISB INTEGER,
    K_PLAN_CLOSE_CLI INTEGER,
    PAR_NA_1 CURR_TYPE,
    PAR_NA_2 CURR_TYPE,
    PAR_NA_3 CURR_TYPE,
    PAR_NA_4 CURR_TYPE,
    DELINQ_BALANCE_1 DECIMAL(18,4),
    DELINQ_BALANCE_2 DECIMAL(18,4),
    DELINQ_BALANCE_3 DECIMAL(18,4),
    DELINQ_BALANCE_4 DECIMAL(18,4),
    CB_PERSONAL_STAGE_MONTH INTEGER,
    PAR7_KAT1 CURR_TYPE,
    DELIQ_BALANCE7_KAT_1 DECIMAL(18,4),
    ACTIVE_BEGIN_KAT_1 DECIMAL(18,4),
    CB_PORTFEL_C DECIMAL(18,4),
    CB_PORTFEL_D DECIMAL(18,4))
AS
declare variable rep_date_begin date;
declare variable deliq_balance_kat_1 curr_type;
declare variable deliq_balance30_kat_1 curr_type;
declare variable aa_curr_kc_id decimal(18,0);
declare variable k_id id_type;
declare variable k_kredline_end date;
declare variable k_kolich_client_approved smallint;
declare variable rep_date date;
declare variable cb_personal_stage date;
declare variable cb_activ_balance_last curr_type;
begin
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end) + 1;
  reg_name = (select reg_name
              from region
              where reg_unicode = :rep_region_id);
  for select aa_curr_kc_id, aa_curr_kc_fio, aa_kc_stage, sum(aa.aa_amount_active_begin), sum(aa.aa_amount_active_end),
             sum(aa.aa_just_cl_begin), sum(aa.aa_just_cl_end), sum(aa.aa_deliq_balance), sum(aa.aa_deliq_balance7),
             sum(aa.aa_deliq_balance30), sum(iif(aa_category_id = 1, aa.aa_amount_active_begin, 0)) active_begin_kat_1,
             sum(iif(aa_category_id = 1, aa.aa_deliq_balance, 0)) deliq_balance_kat_1,
             sum(iif(aa_category_id = 1, aa.aa_deliq_balance7, 0)) deliq_balance7_kat_1,
             sum(iif(aa_category_id = 1, aa.aa_deliq_balance30, 0)) deliq_balance30_kat_1
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 1, 0, 1, 0, 0, 0) aa
      where aa_category_id in (1, 2) and
            aa.aa_loan_kind in ('A Групповой', 'C Индивидуальный')
      group by aa.aa_curr_kc_id, aa_curr_kc_fio, aa_kc_stage
      into aa_curr_kc_id, kf_officer_name, aa_kc_stage, aa_amount_active_begin, aa_amount_active_end, aa_just_cl_begin,
           aa_just_cl_end, aa_deliq_balance, aa_deliq_balance7, aa_deliq_balance30, active_begin_kat_1,
           deliq_balance_kat_1, deliq_balance7_kat_1, deliq_balance30_kat_1
  do
  begin

    if (aa_kc_stage = 'Не определен') then
      aa_kc_stage = 'Не опред.';

--   Просроченные балансы за предыдущие дни

    delinq_balance_1 = null;

    select sum(iif(:rep_date_end - 1 > oks_delinq_date and (select aa_category_id
                                                            from aarep_get_category_id(oks_from_k_id, :rep_date_end - 1, null, null, null, null, null, null)) in (1, 2) and :rep_date_end - 1 between oks_date_begin and oks_date_end, oks_balance, 0)), sum(iif(:rep_date_end - 2 > oks_delinq_date and (select aa_category_id
                                                                                                                                                                                                                                                                                                          from aarep_get_category_id(oks_from_k_id, :rep_date_end - 2, null, null, null, null, null, null)) in (1, 2) and :rep_date_end - 2 between oks_date_begin and oks_date_end, oks_balance, 0)), sum(iif(:rep_date_end - 3 > oks_delinq_date and (select aa_category_id
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                        from aarep_get_category_id(oks_from_k_id, :rep_date_end - 3, null, null, null, null, null, null)) in (1, 2) and :rep_date_end - 3 between oks_date_begin and oks_date_end, oks_balance, 0)), sum(iif(:rep_date_end - 4 > oks_delinq_date and (select aa_category_id
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                      from aarep_get_category_id(oks_from_k_id, :rep_date_end - 4, null, null, null, null, null, null)) in (1, 2) and :rep_date_end - 4 between oks_date_begin and oks_date_end, oks_balance, 0)), sum(iif(:rep_date_end - oks_delinq_date between 8 and 180 and (select aa_category_id
                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                  from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1 and :rep_date_end between oks_date_begin and oks_date_end, oks_balance, 0))

    from oksummary, kontract, product
    where oks_kc_id = :aa_curr_kc_id and
          oks_region_id = :rep_region_id and
          :rep_date_end >= oks_date_begin and
          :rep_date_end - 4 <= oks_date_end and
          oks_balance > 0 and
          oks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind = 1 and
          oks_close_state = 0
    into delinq_balance_1, delinq_balance_2, delinq_balance_3, delinq_balance_4, deliq_balance7_kat_1;
    if (delinq_balance_1 is null) then
      delinq_balance_1 = 0;
    if (delinq_balance_2 is null) then
      delinq_balance_2 = 0;
    if (delinq_balance_3 is null) then
      delinq_balance_3 = 0;
    if (delinq_balance_4 is null) then
      delinq_balance_4 = 0;

--  Расчет баланса КС на конец предыдущего месяца по Категории 1; B - общий портфель катег.1 на предыдущую отчетную дату
    cb_activ_balance_last = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_begin - 1 between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :aa_curr_kc_id and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_begin - 1, null, null, null, null, null, null)) = 1
    into :cb_activ_balance_last;
    if (:cb_activ_balance_last is null) then
      cb_activ_balance_last = 0;

-- С - Портфель займов, переданный в категорию 1 от другого СВЗ в течение отчетного месяца
    cb_portfel_c = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1,
           (select oks_balance
            from oksummary
            where oks_from_k_id = k_id and
                  oks_close_state = 0 and
                  :rep_date_end between oks_date_begin and oks_date_end), 0))
    from kontstate, kontract, knaktlist kal1, knaktlist kal2
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal1.kal_from_kf_id and
          kal1.is_delete = 0 and
          :rep_date_end between kal1.kal_date and kal1.kal_date_end and
          kal1.kal_new_kf_officer = :aa_curr_kc_id and
          k_from_kf_id = kal2.kal_from_kf_id and
          kal2.is_delete = 0 and
          :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
          kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
          ks_date_begin < kal1.kal_date -- после передачи выданные группы не включает
    into cb_portfel_c;
    if (cb_portfel_c is null) then
      cb_portfel_c = 0;
-- D - Портфель займов, переданный в категорию 1 от данного СВЗ другому СВЗ в течение отчетного месяца
    cb_portfel_d = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1,
           (select oks_balance
            from oksummary
            where oks_from_k_id = k_id and
                  oks_close_state = 0 and
                  :rep_date_end between oks_date_begin and oks_date_end), 0))
    from kontstate, kontract, knaktlist kal1, knaktlist kal2
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal1.kal_from_kf_id and
          kal1.is_delete = 0 and
          :rep_date_end between kal1.kal_date and kal1.kal_date_end and
          k_from_kf_id = kal2.kal_from_kf_id and
          kal2.is_delete = 0 and
          :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
          kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
          kal2.kal_new_kf_officer = :aa_curr_kc_id and
          ks_date_begin < kal1.kal_date -- после передачи выданные группы не включает
    into cb_portfel_d;
    if (cb_portfel_d is null) then
      cb_portfel_d = 0;

    prirost_portfel = aa_amount_active_end - aa_amount_active_begin;
    prirost_cli = aa_just_cl_end - aa_just_cl_begin;
    par = 0;
    par7 = 0;
    par30 = 0;
    par_na_1 = 0;
    par_na_2 = 0;
    par_na_3 = 0;
    par_na_4 = 0;
    par7_kat1 = 0;

--     if (active_begin_kat_1 > 0) then par7_kat1 = deliq_balance7_kat_1 / active_begin_kat_1 * 100;

    if (deliq_balance7_kat_1 = 0) then
    begin
      par7_kat1 = 0;
    end
    else
    begin
      if (cb_activ_balance_last + cb_portfel_c - cb_portfel_d > 0) then
      begin
        par7_kat1 = deliq_balance7_kat_1 / (cb_activ_balance_last + cb_portfel_c - cb_portfel_d) * 100;
      end
      else
      begin
        par7_kat1 = 100;
      end
    end

--      if (kf_officer_name = 'Чекирова Айнура Шеримбековна') then
--      begin
--        aa_amount_active_begin = aa_amount_active_begin;
--      end

    if (aa_amount_active_begin > 0) then
    begin
      par = aa_deliq_balance / aa_amount_active_begin * 100;
      par7 = aa_deliq_balance7 / aa_amount_active_begin * 100;
      par30 = aa_deliq_balance30 / aa_amount_active_begin * 100;

      par_na_1 = delinq_balance_1 / aa_amount_active_begin * 100;
      par_na_2 = delinq_balance_2 / aa_amount_active_begin * 100;
      par_na_3 = delinq_balance_3 / aa_amount_active_begin * 100;
      par_na_4 = delinq_balance_4 / aa_amount_active_begin * 100;

    end
    else
    begin
      if (delinq_balance_1 > 0) then
        par_na_1 = 100;
      if (delinq_balance_2 > 0) then
        par_na_2 = 100;
      if (delinq_balance_3 > 0) then
        par_na_3 = 100;
      if (delinq_balance_4 > 0) then
        par_na_4 = 100;

      if (aa_deliq_balance30 > 0) then
      begin
        par = 100;
        par7 = 100;
        par30 = 100;
      end
      else
      if (aa_deliq_balance7 > 0) then
      begin
        par = 100;
        par7 = 100;
      end
      else
      if (aa_deliq_balance > 0) then
        par = 100;
    end

--   кол-во выданных кредитов без учета льготных
    kolich_cli_disb = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_kc_id = :aa_curr_kc_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind = 1
    into kolich_cli_disb;
    if (kolich_cli_disb is null) then
      kolich_cli_disb = 0;

--   кол-во план закрыт кредитов без льготных на ближайщие 10 дней
    k_plan_close_cli = 0;
    for select k_id, k_kredline_end, k_kolich_client_approved
        from oksummary, kontract, product
        where oks_kc_id = :aa_curr_kc_id
--     and oks_balance > 0
              and
              :rep_date_end between oks_date_begin and oks_date_end and
              oks_from_k_id = k_id and
              k_from_pr_id = pr_id and
              pr_product_kind = 1 and
              oks_close_state = 0 and
              (select aa_category_id
               from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
        into k_id, k_kredline_end, k_kolich_client_approved
    do
    begin
      if (exists(select *
                 from operrest, operlist
                 where or_from_k_id = :k_id and
                       or_from_ol_id = ol_id and
                       operlist.is_delete = 0 and
                       ol_date <= :rep_date_end)) then
      begin
        select ok_sh_date
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date_end and
              ok_type not in (7, 8) and
              ok_from_c_id = 0
        group by ok_sh_date
        having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
        order by ok_sh_date desc
        rows 1
        into k_kredline_end;
      end
      if (:k_kredline_end between :rep_date_end and :rep_date_end + 10) then
        k_plan_close_cli = k_plan_close_cli + k_kolich_client_approved;
    end

    cb_personal_stage = null;
    select ps_date_begin
    from hr_pstatus --, hr_orders , joblist
    where ps_from_c_id = :aa_curr_kc_id and
          ps_date_begin is not null and
          ps_status = 1
    order by ps_date_begin
    rows 1
    into :cb_personal_stage;

    if (cb_personal_stage is null) then
      cb_personal_stage = :rep_date_end;
    cb_personal_stage_month = datediff(day, :cb_personal_stage, :rep_date_end) / 30;

--   kf_officer_name = (select c_name_fio from clients where c_id = :aa_curr_kc_id);
    suspend;
  end

end
