CREATE PROCEDURE OKT_REGION_LIST(
    FIND_TEXT STR_255)
RETURNS (
    REG_UNICODE INTEGER,
    REG_NAME VARCHAR(50),
    ADDRESS_STR STR_255,
    OR_CELLNO_INFO STR_255,
    OR_MEMO VARCHAR(2000),
    OR_SMS_TEXT STR_255,
    OR_PUSH_TEXT STR_255)
AS
declare variable do_suspend smallint;
begin
  find_text = trim(coalesce(find_text, ''));
  for select reg.reg_unicode, reg.reg_name, adl.adl_address_str, okr.or_cellno_info, okr.or_memo, okr.or_sms_text,
             okr.or_push_text
      from region reg
      left join addrlist adl on adl.adl_id = reg.reg_address_fact
      left join okt_region okr on okr.or_from_reg_id = reg.reg_unicode
      where ((reg.reg_date_close is null) or
            (reg.reg_date_close > current_date)) and
            reg.reg_status = 3
      order by reg_name
      into :reg_unicode, :reg_name, :address_str, :or_cellno_info, :or_memo, :or_sms_text, :or_push_text
  do
  begin
    do_suspend = 0;

    if (find_text <> '') then
    begin
      if (reg_name like '%' || :find_text || '%') then
        do_suspend = 1;
      else
      if (address_str like '%' || :find_text || '%') then
        do_suspend = 1;
      else
      if (or_cellno_info like '%' || :find_text || '%') then
        do_suspend = 1;
      else
      if (or_memo like '%' || :find_text || '%') then
        do_suspend = 1;
      else
      if (or_sms_text like '%' || :find_text || '%') then
        do_suspend = 1;
      else
      if (or_push_text like '%' || :find_text || '%') then
        do_suspend = 1;
    end
    else
    begin
      do_suspend = 1;
    end

    if (do_suspend = 1) then
      suspend;
  end
end
