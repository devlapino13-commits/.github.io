CREATE PROCEDURE HP_DEBT_CHECK(
    K_ID ID_TYPE,
    K_NO STR_20,
    ON_DATE DATE NOT NULL = current_date)
RETURNS (
    RESULT INTEGER,
    RESULT_K_ID ID_TYPE)
AS
declare variable oek_rd_id integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  if (k_no is not null) then
    k_id = (select k_id
            from kontract
            where k_kontract_no = :k_no);
  if (exists(select *
             from kontract
             where k_id = :k_id)) then
  begin
    if (exists(select *
               from oksummary
               where oks_from_k_id = :k_id and
                     :on_date between oks_date_begin and oks_date_end and
                     oks_close_state = 0 ) or exists(select oko_from_k_id
                                                                                 from operkroff
                                                                                 where oko_from_k_id = :k_id and
                                                                                       oko_ol_is_active = 1 and
                                                                                       oko_ol_date <= :on_date
                                                                                 group by oko_from_k_id
                                                                                 having sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc) > 0) or exists(select ox_object_id
                                                                                                                                                        from operextra
                                                                                                                                                        where ox_from_rd_id = :oek_rd_id and
                                                                                                                                                              ox_object_id = :k_id and
                                                                                                                                                              ox_object_kind = 9 and
                                                                                                                                                              ox_ol_is_active = 1 and
                                                                                                                                                              ox_ol_date <= :on_date
                                                                                                                                                        group by ox_object_id
                                                                                                                                                        having sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff) > 0)) then
      result = 1;
    else
      result = 0;
    result_k_id = k_id;
  end
  else
  begin
    result = -1;
    result_k_id = 0;
  end
  suspend;
end
