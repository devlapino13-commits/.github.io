CREATE PROCEDURE AAREP_EXTSYS_PAY(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ESM_PROJECT VARCHAR(20),
    ESM_DATE DATE,
    ESM_TRANS_ID VARCHAR(20),
    ESM_AMOUNT CURR_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO STR_100)
AS
begin
  for select em.esm_project, em.esm_date, em.esm_trans_id, em.esm_amount, kont.k_kontract_no, cl.c_name_fio
      from extsysmain em
      left join clients cl on cl.c_id = em.esm_object_id
      left join kontract kont on kont.k_id = em.esm_from_k_id
      where em.esm_status = 1 and
            em.esm_action = 'pay' and
            em.esm_date between :rep_date_begin and :rep_date_end
      into :esm_project, :esm_date, :esm_trans_id, :esm_amount, :k_kontract_no, :c_name_fio
  do
  begin
    suspend;
  end
end
