CREATE PROCEDURE AAREP_SIGNED_QUESTIONNAIRES_R(
    REP_REGION_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    RES_REP_DATE D_DATE,
    RES_REG_NAME D_STR_CODE,
    RES_LOAN_OFFICER D_STR_100,
    RES_APP_DATE D_DATE,
    RES_C_NAME_FIO D_STR_100)
AS
declare variable cds_create_date d_date;
declare variable c_id id_type;
declare variable pk_id d_integer;
declare variable reg_id_go d_integer;
declare variable reg_id d_integer;
declare variable ma_date d_date;
declare variable ma_id d_id;
declare variable ma_id_temp d_id;
declare variable k_id_temp d_id;
declare variable ma_date_temp d_date;
declare variable k_disb_date_temp d_date;
declare variable ks_state_temp d_integer;
declare variable dont_suspend boolean;
begin
  -- отчет создан в дополнение к aarep_signed_questionnaires
  -- отчет предназначен для сбора анкет-опросников кассирами в офисах
  -- отчет показывает заявки по которым были отказы на день отчета
  -- и были прикреплены анкеты-опросники, но не было в этот день выдачи по другой заявке
  -- и не осталось другой активной заявки
  res_rep_date = rep_date;
  select
    pk_id
  from passkind
    inner join passdoctype
      on pdt_id = pk_from_pdt_id
        and pdt_code = 'anketa_questionnaire'
  into
    :pk_id;

  reg_id_go = f_hq_id();

  res_reg_name = (select reg_name from region where reg_unicode = :rep_region_id);

  -- вытаскиваю все отказы за период
  for
    with t1 as (select krl.krl_ma_id as ma_id
                from kontreslist krl
                where krl_datetime >= :rep_date
                  and krl_datetime < :rep_date + 1
                  and krl_ma_id is not null
                union
                select k_from_ma_app_id
                from kontreslist krl, kontract k
                where krl_datetime >= :rep_date
                  and krl_datetime < :rep_date + 1
                  and krl_from_k_id is not null
                  and krl_ma_id is null
                  and k.k_id = krl_from_k_id
                group by 1
                )
    select t1.ma_id, mac.mac_from_c_id, lo.c_name_fio, kal_new_reg_id, cl.c_name_fio, min(ma.ma_date_app)
    from t1, miniapp ma, knaktlist kal, miniappclient mac, clients lo, clients cl
    where ma.ma_id = t1.ma_id
      and kal.kal_from_kf_id = ma.ma_from_kf_id
      and :rep_date between kal_date and kal_date_end
      and kal.is_delete = 0
      and kal_new_reg_id in (:rep_region_id, :reg_id_go)
      and mac.mac_from_ma_id = ma.ma_id
      and lo.c_id = kal_new_kf_officer
      and cl.c_id = mac_from_c_id
      and mac_is_poruch = 0
      and mac.mac_amount > 0
    group by 1,2,3,4,5
    order by 3
    into ma_id, c_id, res_loan_officer, reg_id, res_c_name_fio, ma_date
  do
  begin
    cds_create_date = (select cast(cds_create_datetime as date)
                      from clientdocs
                      where cds_from_c_id = :c_id
                        and cds_from_pk_id = :pk_id
                        and cds_create_datetime >= :ma_date
                        and cds_create_datetime < :rep_date + 1
                        and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
                        order by 1 desc
                        rows 1);

    -- Если нет прикрепленной анкеты опросника в период заявки, то не показываем
    if(cds_create_date is null) then
      continue;

    -- если кредит числится в ГО и клиент не является сотрудником текущего офиса, то не показываем
    if (reg_id = reg_id_go and not exists (select *
                                            from hr_pstatus, clients
                                            where
                                              ps_from_reg_id = :rep_region_id and
                                              ps_is_working = 1 and
                                              :res_rep_date between ps_date_begin and ps_date_end and
                                              ps_from_c_id = :c_id and
                                              ps_status = 1
                                           )
    )then
      continue;

    dont_suspend = false;
    -- Если на дату прикрепления анкеты-опросника есть заявка по которой еще нет решения или была выдача по ней после прикрепления анкеты, то не показываем
    for select ma_id
    from miniapp ma, miniappclient mac
    where :rep_date between ma.ma_date_app and ma.ma_date_end
      and ma_is_active = 1
      and ma_id <> :ma_id
      and mac.mac_from_ma_id = ma.ma_id
      and mac.mac_from_c_id = :c_id
    into ma_id_temp
    do
    begin
      ma_date_temp = (select min(ma_date_app) from miniapp where ma_id = :ma_id_temp);
  
      k_id_temp = null;
      k_disb_date_temp = null;
  
      select k_id, k_disb_date, ks_state
      from kontract, kontstate
      where k_from_ma_app_id = :ma_id_temp
        and ks_from_k_id = k_id
        and ks_ol_is_delete = 0
        and :rep_date between ks_date_begin and ks_date_end
      order by ks_date_begin desc
      rows 1
      into k_id_temp, k_disb_date_temp, ks_state_temp;
  
      --не отправляем если существует другая заявка, которая еще в процессе, т.е.:
      dont_suspend = (ma_date_temp > :rep_date - 30)                              -- заявка не старше 30 дней
                  and (k_id_temp is null or ks_state_temp = 0)                    -- по которой не было выдачи
                  and not exists(select krl.krl_id                                -- и не было отказов по заявке
                                  from kontreslist krl
                                  where krl_datetime >= :rep_date
                                    and krl_datetime < :rep_date + 1
                                    and krl_ma_id = :ma_id_temp)
                  and (k_id_temp is null                                          -- и не было отказов по контракту
                    or not exists(select krl.krl_id
                                  from kontreslist krl
                                  where krl_datetime >= :rep_date
                                    and krl_datetime < :rep_date + 1
                                    and krl_from_k_id = :k_id_temp));
       if (not dont_suspend) then
          dont_suspend = (k_id_temp is not null
                          and ks_state_temp in (1,2)
                          and k_disb_date_temp >= cds_create_date); -- либо после прикрепления анкеты была выдача.
  
      if (dont_suspend) then
        leave;               -- если нашлась активная заявка, то дальше искать не надо
    end

    if(not dont_suspend) then
    begin
      res_app_date = (select cast(max(eds_status_datetime) as date) as app_date
                      from hp_edoc_client_sign(:c_id)
                      where eds_status = 1
                        and edl_from_edc_id = f_edoc_kind('loan.application','id')   --заявка на займ
                        and edl_object_kind = 11
                        and edl_object_id = :ma_id
                      );

      suspend;
    end
  end
end
