CREATE PROCEDURE AAREP_INFO_LO_ORDERS(
    C_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO VARCHAR(100),
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    JOB_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    ORDER_KIND VARCHAR(100))
AS
begin
  for select c_name_fio, ps_date_begin, ps_date_end, jl_name, reg_code || ' ' || reg_name, sawt_name
      from clients, hr_pstatus, hr_orders, sa_worktype, joblist, region
      where c_id = ps_from_c_id and
            ps_status = 1 and
            ps_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            ps_from_jl_id = jl_id and
            ps_from_reg_id = reg_unicode and
            ps_from_c_id = :c_id
      order by ps_date_begin desc
      into :c_name_fio, ps_date_begin, ps_date_end, job_name, reg_name, order_kind
  do
  begin
    suspend;
  end
end
