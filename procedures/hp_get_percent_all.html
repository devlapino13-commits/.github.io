CREATE PROCEDURE HP_GET_PERCENT_ALL(
    K_ID D_ID)
RETURNS (
    PERC_STR D_STR_CODE)
AS
declare variable K_DISB_DATE D_DATE;
declare variable PERC_DAY PERC_TYPE;
declare variable YEAR_DAY D_SMALLINT;
declare variable K_PERCENT_PER_DAY D_PERCENT;
declare variable K_FROM_PR_ID D_INTEGER;
declare variable K_LEVEL D_SMALLINT;
begin
  perc_str = '';
  
  select k_disb_date,
         k_percent_per_day,
         k_from_pr_id
  from kontract
  where k_id = :k_id
  into k_disb_date,
       k_percent_per_day,
       k_from_pr_id;

  if (k_disb_date >= cast('22.01.2016' as date)) then year_day = 365;
  else year_day = 360;

  perc_day = null;
  if (k_disb_date >= '08.09.2021') then
  begin
    select kd_percent_per_day from kontdata
    where kd_from_k_id = :k_id
    and :k_disb_date + 1 between kd_ol_date and kd_date_end
    and kd_ol_is_delete = 0
    into perc_day;
  end
  if (perc_day is null) then perc_day = k_percent_per_day;

  perc_str = round((perc_day * year_day),3)||'%';

  k_level = 0;
  while (k_level <3) do
  begin
    k_level = k_level + 1;
    select :perc_str||coalesce(('; '||round((cast(new_kd_percent as double precision) / 30 * :year_day),3)||'%'),'')
    from hp_get_percent_one(:k_id, :k_level,:k_disb_date, :k_percent_per_day, :k_from_pr_id)
    into perc_str;
  end
  suspend;
end
