CREATE PROCEDURE HP_STATE_SA_TABELO_LOCK(
    YEAR_MONTH D_INTEGER,
    OFFICE_ID D_INTEGER,
    UN_ID D_ID,
    LOCK_STATE D_SMALLINT = null)
RETURNS (
    RES_OPTION D_SMALLINT,
    RES_LOCK D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable month_inp d_integer;
declare variable year_inp d_integer;
begin
  res_text = '';
  if (exists(
    select *
    from hr_pstatus ps
      inner join username
        on un_id = :un_id
      inner join joblist
        on jl_id = ps_from_jl_id
      inner join jobkind
        on jk_id = jl_kind
          and jk_system_code = 'JobSystem.HR_Department'
    where ps_from_c_id = un_from_c_id
      and current_date between ps_date_begin and ps_date_end
      and ps_is_working = 1))
    then res_option = 1;
      else res_option = 0;

  if (un_id = 1090100002454) then
    res_option = 1;

  if (lock_state is null) then
  begin
    res_lock = coalesce((
      select
        stl_lock
      from sa_tabelo_lock
      where stl_year_month = :year_month
        and stl_office_id = :office_id), 0);
  end else begin
    if (res_option = 1) then
    begin
      month_inp = mod(year_month, 100);
      year_inp = year_month / 100;
      if (lock_state = 1 and
        mod(month_inp, 2) = 0 and
        exists(
          select *
          from hp_timesheet_list(:office_id, :month_inp, :year_inp)
          where sat_days_total_overwork > 0)) then
      begin
        res_lock = 0;
        res_text = 'Блокировка отменена. По некоторым сотрудникам найдена переработка за 2-месяца.';
        suspend;
        exit;
      end

      update or insert into sa_tabelo_lock(
        stl_year_month,
        stl_office_id,
        stl_lock,
        stl_from_un_id,
        stl_datetime)
      values (
        :year_month,
        :office_id,
        :lock_state,
        :un_id,
        current_timestamp)
      matching (
        stl_year_month,
        stl_office_id)
      returning stl_lock
        into res_lock;
    end
  end

  suspend;
end
