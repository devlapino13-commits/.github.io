CREATE PROCEDURE CM_RU_0104_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable KC_AMOUNT_APPROVED_TEMP CURR_TYPE;
declare variable KC_AMOUNT_APPROVED_MAX CURR_TYPE;
declare variable K_KOLICH_REPAY integer;
declare variable PR_CODE varchar(100);
declare variable IN_ROWS1 integer;
declare variable ROWS1_CLI_ID ID_TYPE;
declare variable KC_FROM_C_ID ID_TYPE;
declare variable K_KOMMIS_PERC PERC_TYPE;
declare variable K CURR_TYPE;
declare variable KOL_CLI_GROUP integer;
declare variable C_CURR_BONUS_STATUS integer;
declare variable PRL_PERCENT_MAX PERC_TYPE;
declare variable PR_ID integer;
declare variable K_PERCENT PERC_TYPE;
declare variable C_CITIZEN integer;
declare variable RD_CODE varchar(255);
declare variable K_FROM_KF_ID ID_TYPE;
declare variable KAL_NEW_REG_ID integer;
declare variable RD_NAME varchar(255);
declare variable IS_PUNKT_221 smallint;
declare variable KC_CIRCLE integer;
declare variable RE_K_ID ID_TYPE;
declare variable SUM_DELINQ_DAY integer;
declare variable SUM_DELINQ_DAY_TEMP integer;
declare variable DISB_SUM_NEW_CLI CURR_TYPE;
declare variable RE_DISB_DATE date;
declare variable RE_CLOSE_DATE date;
declare variable PAR_RE_DISB_DATE date;
declare variable K_KREDLINE_END date;
declare variable K_KOLICH_LGOTA integer;
declare variable RE_KREDLINE_END date;
declare variable RE_PR_CODE varchar(100);
declare variable RE_PR_NAME varchar(50);
declare variable PR_NAME varchar(50);
declare variable PR_PRODUCT_KIND integer;
declare variable CL_DELINQ_DAY integer;
begin
  for
  select k_amount_approved,
         k_kolich_repay,
         pr_code,
         k_disb_date,
         k_kommis_perc,
         k_kolich_client_approved,
         pr_id,
         k_percent,
         k_from_kf_id,
         k_kredline_end,
         k_kolich_lgota,
         pr_name
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.|.blago.|.uspeh.)%'
  and pr_code not containing '.pilot.'
  into k_amount_approved,
       k_kolich_repay,
       pr_code,
       kf_disb_date,
       k_kommis_perc,
       kol_cli_group,
       pr_id,
       k_percent,
       k_from_kf_id,
       k_kredline_end,
       k_kolich_lgota,
       pr_name
  do
  begin
    k_kolich_repay = k_kolich_repay - k_kolich_lgota;
    select kal_new_reg_id
    from knaktlist
    where knaktlist.is_delete = 0
    and current_date between kal_date and kal_date_end
    and kal_from_kf_id = :k_from_kf_id
    into kal_new_reg_id;
    if (exists(select * from region
               where (parent = 1010000
                   or reg_unicode in (1020200, 1030300, 1031300, 1031400))
               and reg_unicode not in (1420100, 1440100)
               and reg_unicode = :kal_new_reg_id)) then
    begin
      is_punkt_221 = 1;
    end else
    begin
      is_punkt_221 = 0;
    end 
    if (is_punkt_221 = 1) then
    begin
      disb_sum_new_cli = 25000;
    end else
    begin
      disb_sum_new_cli = 25000;
    end
--  проверка по продукту Семья Плюс
    if (pr_code containing '.FAMILY') then
    begin
      if (pr_code containing '.FAMILY.') then
      begin
        if (kol_cli_group not in (2,3)) then
        begin
          res_error_text = 'Кол-во клиентов данного продукта не правильно указано!';
          res_error_is_critical = 2;
          suspend;
        end
      end else
      if (pr_code containing '.FAMILY PLUS.') then
      begin
          if (kol_cli_group not in (2,3,4)) then
          begin
            res_error_text = 'Кол-во клиентов данного продукта не правильно указан!';
            res_error_is_critical = 2;
            suspend;
          end 
      end
      in_rows1 = 1;
      rows1_cli_id = 0;
      for
      select kc_from_c_id, kc_amount_approved, kc_circle
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into kc_from_c_id, kc_amount, kc_circle
      do
      begin
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into res_info_text;
        --        проверяется предыдущий цикл клиента на просрочку
        re_k_id = null;
        select kc_from_k_id,
             ks_date_begin
        from knclient, kontstate, kontract, product pr
        where kc_from_c_id = :kc_from_c_id
        and kc_amount_approved > 0
        and kc_from_k_id = ks_from_k_id
        and ks_ol_is_delete = 0
        and ks_state = 1
        and ks_from_k_id <> :k_id
        and ks_date_begin < :kf_disb_date
        and ks_from_k_id = k_id
        and k_from_pr_id = pr_id
        and pr.pr_code containing '.PAR.'
        order by ks_date_begin desc
        rows 1
        into re_k_id,
             par_re_disb_date;
    
        --вычисляется макс кол-во просроченных дней
        cl_delinq_day = 0;    
        par_re_disb_date = null;
        select max(ok_ol_date - ok_sh_date) from operkred
        where ok_ol_is_active = 1
        and ok_from_k_id = :re_k_id
        and ok_type in (4,5,9)
        and ok_os + ok_perc < -100
        and ok_ol_date > ok_sh_date
        into cl_delinq_day;
        
        if (cl_delinq_day is null) then cl_delinq_day = 0;
        re_k_id = null;
        if (in_rows1 = 1) then
        begin
          in_rows1 = 0;
          rows1_cli_id = kc_from_c_id;
        end else
        begin
          if (pr_code containing '.FAMILY.') then
          begin
--            в продекте семья все члены группы должны быть находится в близком родстве
              if ((:kc_from_c_id not in (select CREL_OTHER_C_ID
                                         from HP_CLIENT_RELATION_LIST(:rows1_cli_id,0)
                                         where crel_from_rel_id in ('1','2','3','4','5','6','7','8'))
                  )) then
              begin
                res_error_text = 'Клиент '||res_info_text||' не является родственником!';
                res_error_is_critical = 0;
                suspend;
              end
          end
        end
        if (kc_circle <= 1 and KF_DISB_DATE < '01.07.16') then
        begin
          if (kc_amount > 25000) then
          begin
            res_error_text = 'Клиент новый и не может брать более 25тыс. руб.';
            res_error_is_critical = 1;
            suspend;
          end 
        end else
        if (kc_amount > 25000 and KF_DISB_DATE < '01.07.16') then
        begin
--        проверяется предыдущий цикл клиента на просрочку
          re_k_id = null;
          select kc_from_k_id,
                 ks_date_begin,
                 ks_date_end,
                 k_kredline_end,
                 pr_code,
                 pr_name,
                 pr_product_kind
          from knclient, kontstate, kontract, product
          where kc_from_c_id = :kc_from_c_id
          and kc_amount_approved > 0
          and kc_from_k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and ks_from_k_id <> :k_id
          and ks_date_begin < :kf_disb_date
          and ks_from_k_id = k_id
          and k_from_pr_id = pr_id
          and pr_code not containing '.PAR.'
          order by ks_date_begin desc
          rows 1
          into re_k_id,
               re_disb_date,
               re_close_date,
               re_kredline_end,
               re_pr_code,
               re_pr_name,
               pr_product_kind;
          if (re_k_id is null or re_kredline_end-re_disb_date < 87) then re_k_id = 0;
          if (re_k_id <> 0) then
          begin
--          если в предыдущем цикле брал не индив кредит или 1+1 то проверяется следующее
            if (re_pr_code not containing '.ind.' or re_pr_code containing '.one.') then
            begin
                --вычисляется макс кол-во просроченных дней
                sum_delinq_day = 0;
                sum_delinq_day_temp = 0;
                select max(ok_ol_date - ok_sh_date) from operkred
                where ok_ol_is_active = 1
                and ok_from_k_id = :re_k_id
                and ok_type in (4,5,9)
                and ok_os + ok_perc < -100
                and ok_ol_date > ok_sh_date
                into sum_delinq_day;
    
                if (sum_delinq_day is null) then sum_delinq_day = 0;
                if (sum_delinq_day < cl_delinq_day and re_disb_date < par_re_disb_date) then sum_delinq_day = cl_delinq_day;
    
                if (sum_delinq_day > 7) then
                begin
                  res_error_text = 'Клиент допустил просрочку более 7 дней и не может получить более 25тыс.руб.';
                  res_error_is_critical = 1;
                  suspend;
                end
            end
          end else
          begin
            res_error_text = 'Клиент новый и не может брать более 25тыс. руб.';
            res_error_is_critical = 1;
            suspend;
          end
        end else
        if (kc_circle > 3 and kc_amount between 40000 and 60000 and KF_DISB_DATE < '01.07.16') then
        begin
--        проверяется предыдущие 3цикла клиента на просрочку
          re_k_id = null;
          for select kc_from_k_id,
                 kc_amount_approved,
                 ks_date_begin,
                 ks_date_end,
                 k_kredline_end,
                 pr_code,
                 pr_name,
                 pr_product_kind
          from knclient, kontstate, kontract, product
          where kc_from_c_id = :kc_from_c_id
          and kc_amount_approved > 0
          and kc_from_k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 2
          and ks_from_k_id <> :k_id
          and ks_date_begin < :kf_disb_date
          and ks_from_k_id = k_id
          and k_from_pr_id = pr_id
          and pr_code not containing '.PAR.'
          order by ks_date_begin desc
          rows 3
          into re_k_id,
               kc_amount_approved_temp,
               re_disb_date,
               re_close_date,
               re_kredline_end,
               re_pr_code,
               re_pr_name,
               pr_product_kind
          do
          begin
          kc_amount_approved_max = iif(kc_amount_approved_max is not null,
                                     iif(kc_amount_approved_temp > kc_amount_approved_max,
                                       kc_amount_approved_temp, kc_amount_approved_max), kc_amount_approved_temp);
          if (kc_amount_approved_max < 60000) then
            begin
              re_k_id = 0;
              res_error_text = 'Клиент не может получить до 60000 руб., так как раньше не получал займ на сумму 60000';
              res_error_is_critical = 1;
              suspend;
            end
          if (re_k_id is null) then    
            begin
              re_k_id = 0;
              res_error_text = 'Клиент не может получить до 60000 руб.';
              res_error_is_critical = 1;
              suspend;
            end
          if (re_kredline_end - re_disb_date < 87) then
            begin
              re_k_id = 0;
              res_error_text = 'Клиент закрыл займ досрочно и не может получить до 60000 руб.';
              res_error_is_critical = 1;
              suspend;
            end
          if (re_k_id <> 0) then
          begin
            --вычисляется макс кол-во просроченных дней
            sum_delinq_day = 0;
            sum_delinq_day_temp = 0;
            select max(ok_ol_date - ok_sh_date) from operkred
            where ok_ol_is_active = 1
            and ok_from_k_id = :re_k_id
            and ok_type in (4,5,9)
            and ok_os + ok_perc < -100
            and ok_ol_date > ok_sh_date
            into sum_delinq_day_temp;
            if (sum_delinq_day is null) then sum_delinq_day = 0;     
            if (sum_delinq_day_temp > sum_delinq_day) then sum_delinq_day = sum_delinq_day_temp; 
            
            if (sum_delinq_day is null) then sum_delinq_day = 0;
            if (sum_delinq_day < cl_delinq_day and re_disb_date < par_re_disb_date) then sum_delinq_day = cl_delinq_day;
            if (sum_delinq_day > 7) then
            begin
              res_error_text = 'Клиент допустил просрочку более 7 дней и не может получить до 60000 руб.';
              res_error_is_critical = 1;
              suspend;
            end
          end
          end
        end
      end
    end 
--  проверка по продукту Баракат
    if (pr_code containing '.BARAKAT'
        or pr_code containing '.blago.'
        or pr_code containing '.uspeh.'
        ) then
    begin
      if (kol_cli_group <> 1) then
      begin
        res_error_text = 'Кол-во клиентов данного продукта не правильно указан!';
        res_error_is_critical = 2;
        suspend;
      end
      for
      select kc_from_c_id, kc_amount_approved, kc_circle
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into kc_from_c_id, kc_amount, kc_circle
      do
      begin
        select c_name_fio, c_citizen
        from clients
        where c_id = :kc_from_c_id
        into res_info_text, c_citizen;
        select rd_code, rd_name from refdata
        where rd_id = :c_citizen
        into rd_code, rd_name;
        if (kc_circle <= 1 and KF_DISB_DATE < '01.07.16') then
        begin
          if (kc_amount > :disb_sum_new_cli) then
          begin
            res_error_text = '1. Клиент новый и не может брать более '||:disb_sum_new_cli||' руб.';
            res_error_is_critical = 1;
            suspend;
          end 
        end else
        if (kc_amount > :disb_sum_new_cli and KF_DISB_DATE < '01.07.16') then
        begin
--        проверяется предыдущий цикл клиента на просрочку
          re_k_id = null;
          select kc_from_k_id, ks_date_end, k_kredline_end, pr_code, pr_name
          from knclient, kontstate, kontract, product
          where kc_from_c_id = :kc_from_c_id
          and kc_amount_approved > 0
          and kc_from_k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and ks_from_k_id <> :k_id
          and ks_date_begin < :kf_disb_date
          and ks_from_k_id = k_id
          and k_from_pr_id = pr_id
          and pr_code not containing '.PAR.'
          order by ks_date_begin desc
          rows 1
          into re_k_id, re_close_date, re_kredline_end, re_pr_code, re_pr_name;
          if (re_k_id is null) then re_k_id = 0;
          sum_delinq_day = null;
          if (re_k_id <> 0) then
          begin
--          если в предыдущем цикле брал не индив кредит или 1+1 то проверяется следующее
            if (re_pr_code containing '.ind.' and re_pr_code not containing '.one.') then
            begin
                --вычисляется макс кол-во просроченных дней
                sum_delinq_day = 0;
                sum_delinq_day_temp = 0;
                
                select max(ok.ok_ol_date - ok_sh_date) from operkred ok
                where ok.ok_ol_is_active = 1
                and ok.ok_ol_date  > ok_sh_date
                and ok_type < 7
                and ok_from_k_id = :re_k_id
                and ok_os + ok_perc < 0
                and ok_from_c_id=0
                into :sum_delinq_day;   
            
                if (sum_delinq_day is null) then sum_delinq_day = 0;
                if (sum_delinq_day < cl_delinq_day and re_disb_date < par_re_disb_date) then sum_delinq_day = cl_delinq_day;
            
                if (sum_delinq_day is null) then sum_delinq_day = 0;
    
                if (sum_delinq_day > 7) then
                begin
                  res_error_text = 'Клиент допустил просрочку более 7 дней суммарно и не может получить более '||:disb_sum_new_cli;
                  res_error_is_critical = 1;
                  suspend;
                end
              end
          end else
          begin
            res_error_text = '2. Клиент новый и не может брать более '||:disb_sum_new_cli;
            res_error_is_critical = 1;
            suspend;
          end
        end
      end
    end
  end
end
