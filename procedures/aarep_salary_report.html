CREATE PROCEDURE AAREP_SALARY_REPORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_C_ID DECIMAL(18,0))
RETURNS (
    ASR_REG_ID INTEGER,
    ASR_C_ID DECIMAL(18,0),
    ASR_SAC_ID INTEGER,
    ASR_ORDER INTEGER,
    ASR_VALUE NUMERIC(18,4),
    ASR_SAC_NAME VARCHAR(100),
    ASR_SAC_SHORT_NAME VARCHAR(100),
    ASR_C_NAME_FIO VARCHAR(100),
    ASR_REG_NAME VARCHAR(100),
    ASR_SALDO_BEGIN DECIMAL(18,4),
    ASR_SALDO_END DECIMAL(18,4))
AS
declare variable c_prior decimal(18,0);
declare variable reg_prior integer;
declare variable sac_sign integer;
declare variable ol_date date;
declare variable tmp_amount decimal(18,4);
begin
  if (report_c_id is null) then
    report_c_id = 0;
  for select asr_reg_id, asr_c_id, asr_sac_id, asr_value, asr_c_name_fio, asr_reg_name, asr_saldo_begin, asr_saldo_end,
             sac_name, sac_short_name, sac_priority_order
      from aarep_salary_tmp_report(:report_region_id, :report_date_begin, :report_date_end, :report_c_id), sa_config
      where asr_sac_id = sac_id and
            sac_is_visible = 1
      order by asr_c_id, asr_reg_id, sac_priority_order
      into :asr_reg_id, :asr_c_id, :asr_sac_id, :asr_value, :asr_c_name_fio, :asr_reg_name, :asr_saldo_begin,
           :asr_saldo_end, :asr_sac_name, :asr_sac_short_name, :asr_order
  do
  begin
    suspend;
  end
end
