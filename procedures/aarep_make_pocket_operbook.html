CREATE PROCEDURE AAREP_MAKE_POCKET_OPERBOOK(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    POC_ID INTEGER)
RETURNS (
    POB_AMOUNT DECIMAL(18,4),
    POB_REGION_NAME VARCHAR(100),
    POB_REGION_ID INTEGER,
    POB_BRANCH_ID INTEGER,
    POB_AMOUNT_BALANCE DECIMAL(18,4),
    POB_AMOUNT_DIFF DECIMAL(18,4),
    POB_ACC_DT VARCHAR(20),
    POB_ACC_KT VARCHAR(20),
    POB_BOOK_MEMO VARCHAR(1024))
AS
declare variable acc_tmp varchar(20);
declare variable dd_begin_str varchar(10);
declare variable dd_end_str varchar(10);
declare variable sql_text varchar(2048);
begin
  sql_text = '';
  select poc_id, coalesce(poc_stored_proc_name, '')
  from pocketoperconfig
  where poc_id = :poc_id
  into :poc_id, :sql_text;
  dd_begin_str = right('00' || extract(day from :report_date_begin), 2) || '.' || right('00' || extract(month from :report_date_begin), 2) || '.' || extract(year from :report_date_begin);
  dd_end_str = right('00' || extract(day from :report_date_end), 2) || '.' || right('00' || extract(month from :report_date_end), 2) || '.' || extract(year from :report_date_end);
-- SQL_TEXT должен выдавать поля в порядке:
-- 1. Сумма в аналитической части
-- 2. Название офиса
-- 3. ID офиса, за которую формируются данная проводка
-- 4. ID офиса где будет формироваться проводки (обычно это тот же самый офис)
-- 5. Сумму в балансе (бух.учете)
-- 6. Сумму разницы (аналитика - баланс). Это и есть сумма проводки, если сумма минусовая, система поменяет Дт и Кт счета местами
-- 7. Дт счет
-- 8. Кт счет
-- 9. Примечание в проводке, краткое описание

  sql_text = replace(sql_text, ':REPORT_REGION_ID', cast(:report_region_id as varchar(20)));
  sql_text = replace(sql_text, ':REPORT_DATE_BEGIN', '''' || :dd_begin_str || '''');
  sql_text = replace(sql_text, ':REPORT_DATE_END', '''' || :dd_end_str || '''');

  if (sql_text <> '') then
  begin
    for execute statement :sql_text
            into pob_amount, pob_region_name, pob_region_id, pob_branch_id, pob_amount_balance, pob_amount_diff,
            pob_acc_dt, pob_acc_kt, pob_book_memo
    do
    begin
      if (pob_amount_diff < 0) then
      begin
        acc_tmp = pob_acc_dt;
        pob_acc_dt = pob_acc_kt;
        pob_acc_kt = acc_tmp;
        pob_amount_diff = -pob_amount_diff;
      end
      suspend;
    end
  end
end
