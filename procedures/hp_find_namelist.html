CREATE PROCEDURE HP_FIND_NAMELIST(
    FIND_C_ID DECIMAL(18,0),
    FIND_OBJECT_KIND INTEGER,
    FIND_TXT VARCHAR(400))
RETURNS (
    OBJECT_ID DECIMAL(18,0),
    RES_PRIORITY INTEGER)
AS
declare variable tmps varchar(400);
declare variable find_code_4 varchar(100);
declare variable txt1 varchar(100);
declare variable txt2 varchar(100);
declare variable txt3 varchar(100);
declare variable txt4 varchar(100);
declare variable word_count integer;
declare variable find_orig_txt varchar(100);
declare variable c_name_1 d_str_255;
declare variable c_name_2 d_str_255;
declare variable c_name_3 d_str_255;
declare variable find_code_1 d_str_100;
declare variable find_code_2 d_str_100;
declare variable find_code_3 d_str_100;
declare variable tmp_name d_str_255;
declare variable sym d_str_sym;
declare variable name_id d_smallint;
declare variable p d_integer;
declare variable txt_id d_smallint;
declare variable txt d_str_255;
declare variable need_ignore d_bool;
begin
  if (find_txt is null) then
    find_txt = '';
  if (find_c_id is null) then
    find_c_id = 0;
  if (find_txt = '') then
    exit;

  word_count = 0;
  txt1 = '';
  txt2 = '';
  txt3 = '';
  txt4 = '';

  find_orig_txt = find_txt;
  find_txt = (select res_text
              from hp_txt_clear(:find_txt));
  if (find_txt = '') then
    exit;

  res_priority = 0;

  for select trim(res_str)
  from hp_split_str(:find_txt, ' ')
  order by char_length(trim(res_str)) desc
  rows 4
  into :tmps
  do
  begin
    word_count = word_count + 1;
    if (word_count = 1) then
      txt1 = tmps;
    if (word_count = 2) then
      txt2 = tmps;
    if (word_count = 3) then
      txt3 = tmps;
    if (word_count = 4) then
      txt4 = tmps;
  end

-- Ищем точное совпадение
  if (find_object_kind = 0) then
  begin
    if (find_c_id = 0) then
    begin
      for select c_id
          from clients
          where c_name_fio = :find_orig_txt or
                c_alternat_fio = :find_orig_txt or
                c_name_fio = :find_txt or
                c_alternat_fio = :find_txt
          into :object_id
      do
        suspend;
    end
    else
    begin
      for select c_id
          from clients
          where (c_name_fio = :find_orig_txt or
                c_alternat_fio = :find_orig_txt or
                c_name_fio = :find_txt or
                c_alternat_fio = :find_txt) and
                c_id = :object_id
          into :object_id
      do
        suspend;
    end
  end
  if (find_object_kind = 14) then
  begin
    for select m_id
        from moratory
        where m_client_fio = :find_orig_txt
        into :object_id
    do
    begin
      if (find_c_id = 0 or object_id = find_c_id) then
        suspend;
    end
  end
  if (find_object_kind = 15) then
  begin
    if (find_c_id = 0) then
    begin
      for select pl_id
          from places
          where (pl_name starting :find_orig_txt or
                pl_name_komitet starting :find_orig_txt)
          into :object_id
      do
        suspend;
    end
    else
    begin
      for select pl_id
          from places
          where (pl_name starting :find_orig_txt or
                pl_name_komitet starting :find_orig_txt) and
                pl_id = :object_id
          into :object_id
      do
        suspend;
    end
  end

  if (object_id is not null) then
    exit;

  if (word_count = 0) then
    exit;

  find_code_1 = f_metaphone(:txt1);
  find_code_2 = f_metaphone(:txt2);
  find_code_3 = f_metaphone(:txt3);
  find_code_4 = f_metaphone(:txt4);

  res_priority = 1;

  for select res_id
  from hp_find_namelist_tmp(:find_code_1, :find_object_kind, :find_c_id)
  into :object_id
  do
  begin
    if (word_count = 1) then
      suspend;
    else
    begin
      if (find_object_kind = 0) then
      begin
        c_name_1 = '';
        c_name_2 = '';
        c_name_3 = '';
  
        select c_name_fio, c_alternat_fio, c_name_eng
        from clients
        where c_id = :object_id
        into c_name_1, c_name_2, c_name_3;
  
        name_id = 0;
        while (name_id < 3) do
        begin
          need_ignore = 0;
          name_id = name_id + 1;
  
          if (name_id = 1) then
            tmp_name = c_name_1;
          else
          if (name_id = 2) then
            tmp_name = c_name_2;
          else
            tmp_name = c_name_3;
  
          tmp_name = trim(coalesce(tmp_name, ''));
          tmp_name = replace(tmp_name, ' ', '_');
  
          if (tmp_name <> '') then
          begin
            tmp_name = '_' || tmp_name;
  
            txt_id = 0;
            while (txt_id < 4) do
            begin
              txt_id = txt_id + 1;
  
              txt = trim(case txt_id
                when 1 then txt1
                when 2 then txt2
                when 3 then txt3
                when 4 then txt4
              end);
  
              if (txt <> '') then
              begin
                sym = left(txt, 1);
                p = position('_' || sym, tmp_name);
                if (p = 0) then
                begin
                  need_ignore = 1;
                  txt_id = 4;
                  name_id = 3;
                end
                else
                  tmp_name = left(tmp_name, p) || '$' || sym || substring(tmp_name from p + 1);
              end
            end
          end
  
          if (need_ignore = 0) then
            name_id = 3;
        end
  
        if (need_ignore = 1) then
          Continue;
      end

      for select res_id
      from hp_find_namelist_tmp(:find_code_2, :find_object_kind, :object_id)
      into :object_id
      do
      begin
        if (word_count = 2) then
          suspend;
        else
        begin
          for select res_id
          from hp_find_namelist_tmp(:find_code_3, :find_object_kind, :object_id)
          into :object_id
          do
          begin
            if (word_count = 3) then
              suspend;
            else
            begin
              for select res_id
              from hp_find_namelist_tmp(:find_code_4, :find_object_kind, :object_id)
              into :object_id
              do
                suspend;
            end
          end
        end
      end
    end
  end
end
