CREATE PROCEDURE MC_GL_BASKET_DETAIL(
    MCL_ID BIGINT,
    GB_ID BIGINT)
RETURNS (
    GBG_ID BIGINT,
    GBG_NAME D_STR_LONG,
    GBG_BARCODE INTEGER,
    GBG_QTY INTEGER,
    GBG_UNIT_PRICE D_CURRENCY,
    GBG_TOTAL_PRICE D_CURRENCY)
AS
declare variable db_id d_integer;
declare variable c_id d_id;
begin
  db_id = f_get_db_id();
  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  if ((select gl.gb_from_mcl_id
       from gl_basket gl
       where gl.gb_id = :gb_id) is null) then
  begin
    update gl_basket gb
    set gb.gb_from_mcl_id = :mcl_id
    where gb.gb_id = :gb_id;
  end

  if (c_id is null or db_id not in (39801, 41701) or gb_id is null) then
  begin
    suspend;
    exit;
  end
  for select glb.gb_id, gbg.gbg_name, gbg.gbg_barcode, gbg.gbg_qty, gbg.gbg_unit_price,
             (gbg.gbg_unit_price * gbg.gbg_qty)
      from gl_basket glb
      inner join clientshops cs on cs.cs_id = glb.gb_from_cs_id
      inner join gl_basket_goods gbg on glb.gb_id = gbg.gbg_from_gb_id
      where glb.gb_from_mcl_id = :mcl_id and
            glb.gb_id = :gb_id
      into :gbg_id, :gbg_name, :gbg_barcode, :gbg_qty, :gbg_unit_price, :gbg_total_price
  do
  begin
    suspend;
  end
end
