CREATE PROCEDURE CR_JSON_NEW_CUSTOMER(
    C_ID D_ID,
    SHOW_ADD_INFO SMALLINT = 0)
RETURNS (
    NEW_CUSTOMER D_STR_2048)
AS
declare variable status integer;
declare variable cln_cat d_str_lang;
declare variable rec_date date;
declare variable coment varchar(255);
declare variable search_name varchar(100);
declare variable cl_type integer;
declare variable surname varchar(100);
declare variable resident smallint;
declare variable r_mob_phone varchar(100);
declare variable f_names varchar(100);
declare variable midle_name varchar(100);
declare variable b_date date;
declare variable sex smallint;
declare variable nationality d_str_lang;
declare variable marital_status smallint;
declare variable person_code varchar(20);
declare variable client_b d_id;
declare variable c_region varchar(50);
declare variable c_since integer;
declare variable id_card varchar(30);
declare variable serial_no varchar(2);
declare variable doc_since date;
declare variable issued_by varchar(255);
declare variable doc_type varchar(3);
declare variable c_client d_id;
begin
  :status = 10;
  :cln_cat = null;
  :coment = null;
  :search_name = null;

  select iif(c.c_type = 0, 1, 2), fl.fl_name1, iif(c.c_citizen = 8, 1, 2),
         f_cont_info('clients', c.c_id, 'cell_main', 'ok'), fl.fl_name2, fl.fl_name3, c.c_birth_date,
         iif(c.c_sex = 0, 2, 1),
         case
           when c.c_nation in (0, 1) then 'KGZ'
           when c.c_nation = 2 then 'UZB'
           when c.c_nation = 3 then 'RUS'
           when c.c_nation = 4 then 'KAZ'
           when c.c_nation = 5 then 'TJK'
           when c.c_nation = 6 then 'UIG'
           when c.c_nation = 7 then 'DUN'
           when c.c_nation = 28 then 'RES'
           when c.c_nation = 55 then 'KOR'
         end,
         case
           when c.c_family = 14 then 1
           when c.c_family in (15, 50) then 2
           when c.c_family = 16 then 3
           when c.c_family = 17 then 4
         end,
         c.c_inn, c.c_id, reg.reg_name
  from clients c
  inner join fiolist fl on c.c_from_fl_id = fl.fl_id
  inner join region reg on reg.reg_unicode = c.c_from_reg_id
  where c.c_id = :c_id
  into :cl_type, :surname, :resident, :r_mob_phone, :f_names, :midle_name, :b_date, :sex, :nationality,
       :marital_status, :person_code, :client_b, :c_region;

  select coalesce(min(cll.cll_datetime), current_timestamp)
  from clientlog cll
  where cll.cll_from_c_id = :c_id
  into :rec_date;
  c_since = extract(year from rec_date);

  select overlay(cd.cds_doc_no placing '' from 1 for 2), left(replace(cd.cds_doc_no, ' ', ''), 2),
         cd.cds_doc_date_begin, cd.cds_doc_issued_by,
         case
           when cd.cds_from_pk_id in (6, 11, 13, 15, 17, 36, 303, 334, 335) then '001'
           when cd.cds_from_pk_id in (54, 61, 62) then '002'
           when cd.cds_from_pk_id = 56 then '003'
           when cd.cds_from_pk_id in (8, 12, 296, 300) then 'NIC'
         end,
         cci.cci_ext_id
  from clientdocs cd
  inner join clients c on cd.cds_from_c_id = c.c_id
  inner join cardclientid cci on cci.cci_from_c_id = cd.cds_from_c_id
  where cd.cds_from_c_id = :c_id and
        cd.cds_from_pk_id = c.c_doc_type and
        cd.cds_is_delete = 0
  order by cds_id desc
  rows 1
  into :id_card, :serial_no, :doc_since, :issued_by, :doc_type, :c_client;

  new_customer = '{' ||
    f_json_make_pair('cid', :c_id, 'num', 1) ||
    f_json_make_pair('cl_type', :cl_type, 'num', 1) ||
    f_json_make_pair('surname', :surname, 'str', 1) ||
    f_json_make_pair('c_since', :c_since, 'num', 1) ||
    f_json_make_pair('resident', :resident, 'num', 1) ||
    f_json_make_pair('r_mob_phone', :r_mob_phone, 'str', 1) ||
    f_json_make_pair('status', :status, 'num', 1) ||
    f_json_make_pair('f_names', :f_names, 'str', 1) ||
    f_json_make_pair('midle_name', :midle_name, 'str', 1) ||
    f_json_make_pair('b_date', :b_date, 'date', 1) ||
    f_json_make_pair('person_code', :person_code, 'str', iif(show_add_info = 1, 1, 0));

  if (:show_add_info = 1) then
    begin
    :new_customer = :new_customer ||
        f_json_make_pair('sex', :sex, 'num', 1) ||
        f_json_make_pair('nationality', :nationality, 'str', 1) ||
        f_json_make_pair('marital_status', :marital_status, 'num', 1) ||
        f_json_make_pair('doc_type', :doc_type, 'str', 1) ||
        f_json_make_pair('id_card', :id_card, 'str', 1) ||
        f_json_make_pair('serial_no', :serial_no, 'str', 1) ||
        f_json_make_pair('doc_since', :doc_since, 'date', 1) ||
        f_json_make_pair('issued_by', :issued_by, 'str', 1) ||
        f_json_make_pair('client', :c_client, 'num', 1) ||
        f_json_make_pair('client_b', :client_b, 'num, 1') ||
        f_json_make_pair('cln_cat', :cln_cat, 'str', 1) ||
        f_json_make_pair('rec_date', :rec_date, 'date', 1) ||
        f_json_make_pair('coment', :coment, 'str', 1) ||
        f_json_make_pair('region', :c_region, 'str', 1) ||
        f_json_make_pair('search_name', :search_name, 'str');
    end

    :new_customer = :new_customer || '}';

  suspend;
end
