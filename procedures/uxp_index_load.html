CREATE PROCEDURE UXP$INDEX_LOAD(
    INDEX_NAME D_STR_100)
RETURNS (
    SOURCE_NAME D_STR_CODE)
AS
declare variable fields d_str_100;
declare variable is_uniq d_bool;
declare variable fk_index d_str_code;
declare variable is_pk d_bool;
declare variable is_fk d_bool;
declare variable uix_pk d_bool;
declare variable uix_fields d_str_100;
declare variable us_uuid d_uuid;
declare variable pk_id d_integer;
declare variable field_name d_str_code;
begin
  if (trim(coalesce(index_name, '')) = '') then
    exit;

  index_name = upper(trim(:index_name));
  source_name = '';

  select uix_source, uix_pk, uix_fields
  from ux$index
  where uix_index = :index_name
  into :source_name, uix_pk, uix_fields;

  if (source_name = '') then
  begin
    for select trim(rdb$relation_name), rdb$unique_flag, lower(trim(coalesce(rdb$foreign_key, '')))
    from rdb$indices
    where rdb$index_name = :index_name
    into source_name, is_uniq, fk_index
    do
    begin
      select list(lower(trim(i.rdb$field_name)), ';')
      from rdb$index_segments i
      where i.rdb$index_name = :index_name
      into fields;
  
      if (coalesce(fields, '') = '') then
        exception e_default f_format('Index without fields: %1', index_name);
  
      is_pk = 0;
      is_fk = 0;
  
      if (exists(select *
      from rdb$relation_constraints r
      where r.rdb$index_name = :index_name and
            r.rdb$relation_name = :source_name and
            r.rdb$constraint_type = 'PRIMARY KEY')) then
        is_pk = 1;
  
      if (fk_index <> '') then
        is_fk = 1;
  
      insert into ux$index (uix_index, uix_source, uix_fields, uix_pk, uix_uniq, uix_fk, uix_fk_index)
      values (lower(:index_name), lower(:source_name), :fields, :is_pk, :is_uniq, :is_fk, :fk_index);
    end
  end else
  begin
    update ux$index
    set uix_state = 0
    where uix_index = lower(:index_name);

    if (uix_pk = 1) then
    begin
      select us_uuid
      from ux$source
      where us_code = :source_name and
            us_is_table = 1 and
            us_kind = 0
      into us_uuid;

      if (us_uuid is not null) then
      begin
        update ux$field
        set uf_pk_field = 0
        where uf_owner = :us_uuid and
              uf_pk_field <> 0;
  
        pk_id = 0;
        for select res_str
        from uxp$split_str(:uix_fields, ';')
        into :field_name
        do
        begin
          pk_id = pk_id + 1;
  
          update ux$field
          set uf_pk_field = :pk_id
          where uf_owner = :us_uuid and
                uf_field_name = :field_name;
        end
      end
    end
  end

  suspend;
end
