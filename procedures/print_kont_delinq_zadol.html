CREATE PROCEDURE PRINT_KONT_DELINQ_ZADOL(
    PAR1 ID_TYPE)
RETURNS (
    DELINQ_OS_ALL PERC_TYPE,
    DELINQ_PERC_ALL PERC_TYPE,
    DELINQ_SUM_ALL PERC_TYPE,
    DELINQ_OS PERC_TYPE,
    DELINQ_PERC PERC_TYPE,
    DELINQ_FINE PERC_TYPE,
    DELINQ_DATE DATE,
    DELINQ_DAY INTEGER)
AS
begin
  delinq_os_all = null;
  delinq_perc_all = null;
  delinq_os = null;
  delinq_perc = null;
  delinq_fine = null;
  delinq_sum_all = null;
  select round(sum(zad_os), 2), round(sum(sh_calc_perc), 2), round(sum(iif(sh_date < current_date, zad_os, 0)), 2),
         round(sum(iif(sh_date < current_date, sh_calc_perc, 0)), 2), round(sum(sh_calc_fine), 2), min(sh_date)
  from aarep_make_zad_list(:par1, current_date, 1)
  where sh_date is not null and
        zad_os + sh_calc_perc > 0
  into delinq_os_all, delinq_perc_all, delinq_os, delinq_perc, delinq_fine, delinq_date;

  delinq_os = coalesce(delinq_os, 0);
  delinq_perc = coalesce(delinq_perc, 0);
  delinq_fine = coalesce(delinq_fine, 0);
  delinq_sum_all = delinq_os_all + delinq_perc_all + delinq_fine;

  delinq_day = current_date - delinq_date;
  if (delinq_day < 0) then
    delinq_day = 0;

  suspend;
end
