CREATE PROCEDURE FM_HTML_INCOMPLETE_REPAYMENT(
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable LO_ID ID_TYPE;
declare variable KAL_NEW_KF_NAME D_STR_100;
declare variable KONTRACT_NO D_STR_20;
declare variable PR_NAME D_STR_64;
declare variable OK_SH_DATE D_STR_20;
declare variable INCOMPLETE_REPAYMENT decimal(18,2);
declare variable STR D_STR_LONG;
declare variable DB_ID D_INTEGER;
declare variable NSP_MULTIPLIER D_CURRENCY;
begin
  db_id = f_get_db_id();

  nsp_multiplier = 0;
  if (db_id in (39801, 41701)) then
    nsp_multiplier = 0.02;

  select un.un_from_c_id
  from username un
  where un.un_id = :un_id
  into :lo_id;

  res_text = '';
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  res_text = BLOB_APPEND(res_text, '<table width="100%">
         <col width="15%">
         <col width="15%">
         <col width="40%">
         <col width="30%">
         <col width="15%">
         <tr>' || f_format('<th>%1</th><th>%2</th><th>%3</th><th>%4</th><th>%5</th>',
                  '№ кред. папки',
                  '№ договора',
                  'Продукт',
                  'Дата погашения по графику',
                  'Недоплата') ||
         '</tr>');

  for select kal.kal_new_kf_name,
             kn.k_kontract_no,
             pr.pr_name,
             ok.ok_sh_date,
             round((sum(ok.ok_sh_os) + sum(ok.ok_sh_perc) + sum(ok.ok_sh_perc) * :nsp_multiplier) +
             (sum(ok.ok_os) + sum(ok.ok_perc) + sum(ok.ok_tax)), 0)
      from kontract kn
      inner join kontstate ks on ks.ks_from_k_id = kn.k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kn.k_from_kf_id
      inner join product pr on pr.pr_id = kn.k_from_pr_id
      inner join operkred ok on ok.ok_from_k_id = kn.k_id
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
           (current_date between ks.ks_date_begin and ks.ks_date_end) and
            kal.is_delete = 0 and
            kal.kal_new_kf_officer = :lo_id and
           (current_date between kal.kal_date and kal.kal_date_end) and
            ok.ok_ol_is_active = 1 and
            ok.ok_type <= 4 and
           (ok.ok_sh_date between current_date and current_date + 2)
      group by kal.kal_new_kf_name, kn.k_kontract_no, pr.pr_name, ok.ok_sh_date
      having (sum(ok.ok_sh_os) + sum(ok.ok_sh_perc) + sum(ok.ok_sh_perc) * :nsp_multiplier) +
             (sum(ok.ok_os) + sum(ok.ok_perc) + sum(ok.ok_tax)) >= 1 and
             (sum(ok.ok_sh_os) + sum(ok.ok_sh_perc) + sum(ok.ok_sh_perc) * :nsp_multiplier) +
             (sum(ok.ok_os) + sum(ok.ok_perc) + sum(ok.ok_tax)) <= 500
  into :kal_new_kf_name,
       :kontract_no,
       :pr_name,
       :ok_sh_date,
       :incomplete_repayment
  do
  begin
    str = '<tr>' ||
               '<td align="left">' || :kal_new_kf_name || '</td>' ||
               '<td align="left">' || :kontract_no || '</td>' ||
               '<td align="left">' || :pr_name || '</td>' ||
               '<td align="left">' || f_datetostr(:ok_sh_date) || '</td>' ||
               '<td align="left">' || :incomplete_repayment || '</td>' ||
          '</tr>';

    res_text = BLOB_APPEND(res_text, str);
  end

  res_text = BLOB_APPEND(res_text, '</table></body>');
  suspend;
end
