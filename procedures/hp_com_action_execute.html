CREATE PROCEDURE HP_COM_ACTION_EXECUTE(
    CML_ID D_ID NOT NULL,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255,
    NEED_EDOC_CANCEL D_SMALLINT)
AS
declare variable ADD_PARAMS D_STR_LONG;
declare variable CML_DATE D_DATE;
declare variable K_ID D_ID;
declare variable CMC_COM_KIND D_STR_CODE;
declare variable CURR_AMOUNT D_CURRENCY;
declare variable NEW_AMOUNT D_CURRENCY;
declare variable OL_ID D_ID;
declare variable TMP_TEXT D_STR_255;
declare variable LANG D_STR_LANG;
declare variable CMC_FROM_EDC_ID D_INTEGER;
declare variable DB_ID D_INTEGER;
declare variable CMC_CODE D_STR_CODE;
begin
  res_is_error = 0; 
  need_edoc_cancel = 0;
  res_text = '';
  lang = f_get_lang_code(un_id, null);
  db_id = f_get_db_id();

  select cml_date, cml_object_id, cml_add_params, f_enum('com_kind', cmc_com_kind, 'code'), cmc_from_edc_id, cmc_code
  from com_list, com_cfg
  where cml_id = :cml_id and
        cml_from_cmc_id = cmc_id
  into cml_date, k_id, :add_params, cmc_com_kind, cmc_from_edc_id, cmc_code;

  if (cmc_com_kind not in ('reschedule', 'termination')) then
  begin
    suspend;
    exit;
  end

  if (exists(select * from kontcalcperc
  where kcp_period >= (extract(year from :cml_date) * 100 + extract(month from :cml_date)) and
        kcp_from_k_id = :k_id)) then
  begin
    res_is_error = 1;
    need_edoc_cancel = 1;
    res_text = f_format(f_lang(lang, 'График не актуален, период закрыт: %1'), f_datetostr(cml_date));
    suspend;
    exit;
  end


  if (cmc_code = 'rest_standart') then -- Реструктуризация стандартное
  begin
    select res_is_error, res_text
    from hp_do_reschedule(:k_id, :cml_date, 'auto', :add_params, :un_id)
    into :res_is_error, :res_text;
  
    if (res_is_error = 1) then
    begin
      suspend;
      exit;
    end
  
    curr_amount = 0;
    new_amount = 0;
    select sum(sh_new_os + sh_new_perc)
    from temp_sh
    into curr_amount;
    if (curr_amount is null) then
      curr_amount = 0;
  
    select sum(cmr_sh_os + cmr_sh_perc)
    from com_list_resh
    where cmr_from_cml_id = :cml_id
    into new_amount;
    if (new_amount is null) then
      new_amount = 0;
  
    if ((new_amount <> curr_amount or
       (db_id not in (39801, 41701) and cmc_from_edc_id is not null and cml_date < current_date))
       and cmc_code not in ('rastor_dog_sogl_storon')) then
    begin
      res_is_error = 1;
      need_edoc_cancel = 1;
      if (new_amount <> curr_amount) then
        res_text = f_format(f_lang(lang, 'График не актуален, изменилась общая сумма (%1 -> %2)'), f_numtostr(new_amount, 0), f_numtostr(curr_amount, 0));
  
      if (db_id not in (39801, 41701) and cmc_from_edc_id is not null and cml_date < current_date) then
        res_text = f_format(f_lang(lang, 'График не актуален, дата устарела: %1'), f_datetostr(cml_date));
  
      select res_text
      from hp_com_action_add(:cml_id, null, 'cancelled', :un_id)
      into :tmp_text;
  
      select res_text
      from hp_com_memo_add(:cml_id, :res_text, 2, :un_id)
      into :tmp_text;
  
      suspend;
      exit;
    end
  
    add_params = f_json_str_set_value(add_params, 'memo', res_text);
    add_params = f_json_str_set_value(add_params, 'protocol_date', cml_date);
    add_params = f_json_str_set_value(add_params, 'memo', '');
    res_text = '';
  
    select ol_id, res_is_error, res_text
    from hp_do_reschedule(:k_id, :cml_date, 'save', :add_params, :un_id)
    into ol_id, :res_is_error, :res_text;
  
    if (res_is_error = 0 and ol_id is not null) then
    begin
      -- Фиксируем проведенную реструктуризацию
      insert into com_operlist (cmo_from_cma_id, cmo_from_ol_id, cmo_ol_is_delete)
      select cml_last_cma_id, :ol_id, 0
      from com_list
      where cml_id = :cml_id;
    end
  end

  if (cmc_code = 'kred_kanikul_svo') then -- Кредитные каникулы - СВО
  begin
    res_text = '';
    select ol_id, res_text
    from hp_kontvac(:k_id, :add_params, :un_id)
    into :ol_id, :res_text;
    if (res_text <> '') then res_is_error = 1;

    if (res_is_error = 0 and ol_id is not null) then
    begin
      -- Фиксируем проведенную реструктуризацию
      insert into com_operlist (cmo_from_cma_id, cmo_from_ol_id, cmo_ol_is_delete)
      select cml_last_cma_id, :ol_id, 0
      from com_list
      where cml_id = :cml_id;
    end
  end 

  if (cmc_code = 'kred_kanikul_standart') then -- Кредитные каникулы - Стандарт
  begin
    res_text = '';
    select ol_id, res_text
    from hp_kontvac(:k_id, :add_params, :un_id)
    into :ol_id, :res_text;
    if (res_text <> '') then res_is_error = 1;

    if (res_is_error = 0 and ol_id is not null) then
    begin
      -- Фиксируем проведенную реструктуризацию
      insert into com_operlist (cmo_from_cma_id, cmo_from_ol_id, cmo_ol_is_delete)
      select cml_last_cma_id, :ol_id, 0
      from com_list
      where cml_id = :cml_id;
    end
  end 

  if (cmc_code in ('rastor_dog_sogl_storon', 'rastor_dog_smert_cli','rastor_dog_zakon')) then
  begin
    select ol_id, res_is_error, res_text
    from hp_stop_perc(:k_id, :add_params)
    into :ol_id, :res_is_error, :res_text;

    if (res_is_error = 0 and ol_id is not null) then
    begin
      -- Фиксируем проведенную реструктуризацию
      insert into com_operlist (cmo_from_cma_id, cmo_from_ol_id, cmo_ol_is_delete)
      select cml_last_cma_id, :ol_id, 0
      from com_list
      where cml_id = :cml_id;
    end
  end 

  suspend;
end
