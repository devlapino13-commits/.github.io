CREATE PROCEDURE UXP$UXSOURCE_SQL_KEYNAME(
    US_UUID D_UUID,
    US_UUID_CODE D_STR_CODE = null)
RETURNS (
    KEYNAME_SQL D_STR_LONG,
    ERROR_TEXT D_STR_255)
AS
declare variable us_code d_str_code;
declare variable us_lookup_id d_str_code;
declare variable us_lookup_text d_str_255;
declare variable us_is_table d_bool;
declare variable main_prefix d_str_10;
declare variable pos d_integer;
declare variable sql_text d_str_long;
declare variable join_text d_str_long;
declare variable usl_fk_index d_str_code;
declare variable usl_prefix d_str_10;
declare variable fk_source d_str_code;
declare variable param_list d_str_long;
declare variable enter_sym d_str_lang;
declare variable param_name d_str_code;
declare variable tmp_main_prefix d_str_10;
declare variable usl_link_id d_integer;
begin
  if (coalesce(us_uuid, '') = '') then
    us_uuid = uxf$uniq(us_uuid_code, 0);

  enter_sym = ascii_char(13) || ascii_char(10);

  keyname_sql = '';
  usl_fk_index = '';
  error_text = '';

  if (us_uuid is null) then
  begin
    error_text = 'Источник не зарегистрирован.';
    suspend;
    exit;
  end

  select us_code, us_is_table, us_lookup_id, us_lookup_text
  from ux$source
  where us_uuid = :us_uuid
  into us_code, us_is_table, us_lookup_id, us_lookup_text;

  if (coalesce(us_lookup_text, '') = '') then
  begin
    error_text = 'Поле или выражение не указано.';
    suspend;
    exit;
  end

  if (coalesce(us_lookup_id, '') = '') then
  begin
    error_text = 'Id поле отсутствует.';
    suspend;
    exit;
  end

  sql_text = '';
  main_prefix = '';

  if (us_is_table = 0) then
  begin
    param_list = '';

    for select uf_field_name
    from ux$field
    where uf_owner = :us_uuid and
          uf_is_param = 1
    order by uf_position
    into param_name
    do
    begin
      if (param_list <> '') then
        param_list = param_list || ', ';

      param_list = param_list || ':' || param_name;
    end

  end else
  begin
    pos = position('_', us_lookup_id);
    if (pos > 0) then
      main_prefix = left(us_lookup_id, pos - 1);
    else
      main_prefix = left(us_code, 3);
  
    for select usl_fk_index, usl_prefix, uix_source, usl_link_id
    from ux$sourcelink, ux$index
    where usl_us_uuid = :us_uuid and
          usl_fk_index = uix_index
    order by usl_link_id
    into :usl_fk_index, :usl_prefix, fk_source, usl_link_id
    do
    begin
      tmp_main_prefix = main_prefix;
      if (us_code <> fk_source) then
      begin
        select usl_prefix
        from ux$sourcelink, ux$index
        where usl_us_uuid = :us_uuid and
              usl_fk_index = uix_index and
              uix_fk_source = :fk_source and
              usl_link_id <= :usl_link_id
        order by usl_link_id desc
        rows 1
        into tmp_main_prefix;
      end else
        tmp_main_prefix = main_prefix;

      join_text = '';
      select join_text, error_text
      from uxp$uxsource_sql_join(:usl_fk_index, :tmp_main_prefix, :usl_prefix)
      into join_text, error_text;
  
      if (join_text <> '') then
      begin
        if (sql_text <> '') then
          sql_text = sql_text || ' ' || ascii_char(13);
    
        sql_text = sql_text || join_text;
      end else
      begin
        suspend;
        exit;
      end
    end
  end

  if (usl_fk_index = '') then
    main_prefix = '';

  if (main_prefix <> '') then
    us_lookup_id = main_prefix || '.' || us_lookup_id;

  keyname_sql = 'select ' || us_lookup_text || enter_sym || ' from ' || us_code
                          || iif(param_list <> '', '(' || param_list || ')', '')
                          || ' ' || main_prefix
                          || iif(sql_text <> '', enter_sym || sql_text || ' ', '')
                          || enter_sym || 'where ' || us_lookup_id || ' = :id';


  keyname_sql = replace(keyname_sql, '  ', ' ');

  suspend;
end
