CREATE PROCEDURE HP_OUTMAILING_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_GROUP_KIND D_STR_CODE,
    SEARCH_KIND D_SMALLINT,
    IN_K_ID D_ID,
    IN_C_ID D_ID,
    IN_TEXT D_STR_100)
RETURNS (
    OML_ID D_INTEGER,
    OML_PARENT_ID D_INTEGER,
    OML_IS_OUT D_BOOL,
    OML_DATE D_DATE,
    OML_NUMBER D_INTEGER,
    OML_REGIST_NO D_STR_CODE,
    OML_FROM_REG_ID D_INTEGER,
    OML_APPLIANT_KIND D_INTEGER,
    OML_APPLIANT_KIND_STR D_STR_CODE,
    OML_APPLICANT_NAME D_STR_100,
    OML_FROM_C_ID D_ID,
    OML_FROM_OCT_ID D_INTEGER,
    OML_FROM_OSK_ID D_INTEGER,
    OML_SUBJECT D_STR_LONG,
    OML_MEMO D_STR_8100,
    OML_IS_REGULATORY D_BOOL,
    OML_IS_DELINQ D_BOOL,
    OML_ORIGINAL_STATE D_INTEGER,
    OML_ORIGINAL_STATE_STR D_STR_CODE,
    OML_RESULT_STATE D_INTEGER,
    OML_RESULT_STATE_STR D_STR_CODE,
    OML_DEADLINE D_DATE,
    OML_FROM_UN_ID D_ID,
    OML_DATETIME D_DATETIME,
    REG_NAME D_STR_100,
    RE_OML_ID D_INTEGER,
    RE_DATE D_DATE,
    RE_REGIST_NO D_STR_CODE,
    RE_ORIGINAL_STATE_STR D_STR_CODE,
    RE_REG_NAME D_STR_100,
    RE_REPLAYED D_BOOL,
    RES_CLIENT_FIO D_STR_100,
    OML_KONTRACTS D_STR_255,
    RES_ADDRESS D_STR_LONG,
    RES_IN_ATTACH_COUNT D_INTEGER,
    RES_IN_ATTACH_LIST D_STR_LONG,
    RE_ATTACH_COUNT D_INTEGER,
    RE_ATTACH_LIST D_STR_LONG,
    RE_COUNT D_INTEGER,
    RE_MEMO D_STR_8100,
    OCT_NAME D_STR_100,
    OSK_NAME D_STR_100,
    RE_OSK_NAME D_STR_100,
    RES_IMAGE_ID D_SMALLINT,
    RES_USER_FIO D_STR_100,
    RES_USER_JOBLIST D_STR_100,
    RES_SIGNER_FIO D_STR_100,
    RES_SIGNER_JOBLIST D_STR_100)
AS
declare variable re_original_state d_integer;
declare variable un_c_id d_id;
declare variable signer_c_id d_id;
begin
  if (search_kind is null) then
    search_kind = 0;

  if (coalesce(rep_group_kind, '') = '') then
    rep_group_kind = 'office';

  for select oml_id, oml_parent_id, oml_is_out, oml_date, oml_number, oml_regist_no, oml_from_reg_id, oml_appliant_kind,
         oml_applicant_name, oml_kontracts, oml_from_c_id, oml_from_oct_id, oml_from_osk_id, oml_subject, oml_memo, oml_is_regulatory,
         oml_is_delinq, oml_original_state, oml_result_state, oml_deadline, oml_from_un_id, oml_datetime,
         reg_code || ' ' || reg_name, c_name_fio, oct_name, osk_name
  from hp_outmailing_list_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_group_kind, :search_kind, :in_k_id, :in_c_id, :in_text)
       inner join outmailing on (oml_id = res_oml_id)
       inner join region on (oml_from_reg_id = reg_unicode)
       inner join outcateg on (oml_from_oct_id = oct_id)
       inner join outsendkind on (oml_from_osk_id = osk_id)
       left join clients on (oml_from_c_id = c_id)
  where oml_parent_id is null
  into oml_id, oml_parent_id, oml_is_out, oml_date, oml_number, oml_regist_no, oml_from_reg_id, oml_appliant_kind,
       oml_applicant_name, oml_kontracts, oml_from_c_id, oml_from_oct_id, oml_from_osk_id, oml_subject, oml_memo, oml_is_regulatory,
       oml_is_delinq, oml_original_state, oml_result_state, oml_deadline, oml_from_un_id, oml_datetime,
       reg_name, res_client_fio, oct_name, osk_name
  do
  begin
    oml_original_state_str = f_enum('outmailing_original_send', oml_original_state, 'name');
    oml_result_state_str = f_enum('outmailing_result_state', oml_result_state, 'name');
    oml_appliant_kind_str = f_enum('outmailing_applicant_kind', oml_appliant_kind, 'name');

    res_in_attach_count = 0;
    res_in_attach_list = '';

    select count(*), list(omf_name)
    from outmailfile
    where omf_from_oml_id = :oml_id
    into res_in_attach_count, res_in_attach_list;

    res_address = '';

    for select oma_id, oma_from_oml_id, oma_is_cc, oma_kind, oma_address, oma_email
    from outmailaddr
    where oma_from_oml_id = :oml_id
    as cursor cur
    do
    begin
      if (res_address <> '') then
        res_address = res_address || ',' || ascii_char(13);

      res_address = res_address || f_enum('outmailing_address_kind', cur.oma_kind, 'name') || ': ';
      if (cur.oma_address <> '' and cur.oma_email <> '') then
        res_address = res_address || cur.oma_address || ', имейл: ' || cur.oma_email;
      else
      if (cur.oma_address <> '') then
        res_address = res_address || cur.oma_address;
      else
      if (cur.oma_email <> '') then
        res_address = res_address || cur.oma_email;
    end

    re_oml_id = null;
    re_original_state_str = '';
    re_date = null;
    re_regist_no = '';
    re_memo = '';
    re_reg_name = '';
    re_attach_count = 0;
    re_attach_list = '';
    re_osk_name = '';
    re_count = 0;
    re_replayed = 0;
    res_image_id = oml_is_out;

    if (oml_is_out = 0 and oml_result_state = 0) then
      res_image_id = -1; -- Новый входящий (новый - это без изменения результата)

    select count(*)
    from outmailing
    where oml_parent_id = :oml_id
    into :re_count;

    if (re_count > 0) then
    begin
      res_image_id = 2; -- Входящий с ответом

      for select oml_id, oml_date, oml_regist_no, oml_original_state, oml_memo,
                 reg_code || ' ' || reg_name, osk_name
      from outmailing, region, outsendkind
      where oml_parent_id = :oml_id and
            oml_from_reg_id = reg_unicode and
            oml_from_osk_id = osk_id
      order by oml_id desc
      rows 1
      into :re_oml_id, :re_date, :re_regist_no, :re_original_state, :re_memo,
           :re_reg_name, :re_osk_name
      do
      begin
        re_original_state_str = f_enum('outmailing_original_send', re_original_state, 'name');
        re_replayed = 1; -- Ответ на письмо
  
        select count(*), list(omf_name)
        from outmailfile
        where omf_from_oml_id = :re_oml_id
        into re_attach_count, re_attach_list;
      end
    end else
    begin
      if (oml_is_out = 1) then
      begin
        re_original_state_str = oml_original_state_str;
        oml_original_state_str = '';
      end
    end

    if (re_count = 0) then
      re_count = null;

    res_user_fio = '';
    res_user_joblist = '';
    un_c_id = null;

    select un_fio, un_from_c_id
    from username
    where un_id = :oml_from_un_id
    into res_user_fio, un_c_id;

    select c_name_fio, jl_name
    from hr_pstatus, clients, joblist
    where ps_from_c_id = :un_c_id and
          ps_date_begin <= current_date and
          ps_date_end >= current_date and
          ps_status = 1 and
          ps_from_c_id = c_id and
          ps_from_jl_id = jl_id
    into res_user_fio, res_user_joblist;

    -- RFCRU-4114
    select first 1 distinct edc_default_manager_id
    from edoc_cfg
    where edc_default_manager_id is not null
    into :signer_c_id;

    select c.c_name_fio, jl.jl_name
    from clients c
    inner join hr_pstatus ps on ps.ps_from_c_id = c.c_id
    inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
    where c.c_id = :signer_c_id
        and ps.ps_date_begin <= current_date
        and ps.ps_date_end >= current_date
        and ps.ps_status = 1
    into :res_signer_fio, :res_signer_joblist;

    suspend;
  end
end
