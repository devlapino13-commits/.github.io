CREATE PROCEDURE HP_KONTRACT_FILL_PERCENT(
    IS_OLD_CALC INTEGER)
RETURNS (
    K_ID DECIMAL(18,0),
    DATE_DISB DATE,
    DATE_LAST DATE,
    KOL_DAYS INTEGER,
    AMOUNT_DISB DECIMAL(18,4),
    AMOUNT_PERC DECIMAL(18,4),
    PERC DECIMAL(18,7),
    PERC_WITH_LGOTA DECIMAL(18,4),
    K_PERCENT DECIMAL(18,7))
AS
begin
  if (coalesce(is_old_calc, 0) = 0) then
  begin
    for select k_id, k_disb_date, max(sh_date), k_amount_approved, sum(sh_perc)
        from kontract, schedule, kontstate
        where k_id = sh_from_k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_from_k_id = k_id
--      and ((k_percent_begin_annual is null) or (k_percent_begin_annual=0))
        group by k_id, k_disb_date, k_amount_approved
        into :k_id, :date_disb, :date_last, :amount_disb, :amount_perc
    do
    begin
      if (amount_disb is null) then
        amount_disb = 0;
      if (amount_perc is null) then
        amount_perc = 0;
      if ((date_last is not null) and
          (date_disb is not null)) then
        kol_days = cast(:date_last - :date_disb as integer);
      else
        kol_days = 0;
      if ((kol_days <> 0) and
          (amount_disb <> 0)) then
      begin
        perc = amount_perc / amount_disb * 100;
        perc = perc / kol_days * 365;
        perc = cast(perc as decimal(15,2));
      end
      else
      begin
        perc = 0;
      end
      update kontract
      set k_percent_begin_annual = :perc
      where k_id = :k_id;
    end
    update kontract
    set k_percent_begin_annual = 0
    where k_percent_begin_annual is null;
    suspend;
  end
  else
  begin
    for select k_id, k_disb_date, k_percent_per_day, max(sh_date), k_amount_approved, sum(sh_perc),
               sum(sh_perc) + sum(sh_perc_tax)
        from kontract, schedule, kontstate
        where k_id = sh_from_k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_from_k_id = k_id
        group by k_id, k_disb_date, k_amount_approved, k_percent_per_day
        into :k_id, :date_disb, :k_percent, :date_last, :amount_disb, :amount_perc, :perc_with_lgota
    do
    begin
      if (amount_disb is null) then
        amount_disb = 0;
      if (amount_perc is null) then
        amount_perc = 0;
      if (perc_with_lgota is null) then
        perc_with_lgota = 0;
      if (perc_with_lgota <> 0) then
        perc = k_percent * amount_perc / perc_with_lgota;
      else
        perc = 0;
      perc = perc * 365;
      perc = cast(perc as decimal(15,2));
      update kontract
      set k_percent_begin_annual = :perc
      where k_id = :k_id and
            k_percent_begin_annual <> :perc;
    end
    update kontract
    set k_percent_begin_annual = 0
    where k_percent_begin_annual is null;
    suspend;
  end
end
