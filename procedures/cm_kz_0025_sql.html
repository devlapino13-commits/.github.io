CREATE PROCEDURE CM_KZ_0025_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable K_ID D_ID;
declare variable DATE_COUNT D_INTEGER;
declare variable COUNT_K D_INTEGER;
declare variable HAS_SMALL_COUNT D_INTEGER;
begin
  for select mac.mac_from_c_id
      from miniapp ma
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
      where ma.ma_id = :ma_id and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999' and
            mac.mac_amount > 0
      into c_id
  do
  begin
      has_small_count = 0;

      for select ks.ks_from_k_id,
                 count(ks.ks_from_k_id) over()
          from kontstate ks
          inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
          where current_date between ks.ks_date_begin and ks.ks_date_end and
                ks.ks_state = 1 and
                ks.ks_ol_is_delete = 0 and
                kc.kc_from_c_id = :c_id and
                kc.kc_amount_approved > 0
          group by ks.ks_from_k_id
          into k_id,
               count_k
      do
      begin
          select count(distinct ok.ok_sh_id)
          from operkred ok
          where ok_ol_is_active = 1
            and ok_type < 7
            and ok_from_k_id = :k_id
            and ok_ol_date <= current_date
            and ((ok_os + ok_sh_os) <> 0 or (ok_perc + ok_sh_perc) <> 0)
          into date_count;

          if (date_count <= 2) then
              has_small_count = 1;
      end

      if (count_k >= 2 and has_small_count = 0) then
      begin
          res_error_is_critical = 2;
          res_error_text = 'У клиента уже есть 2 активных кредита';
          suspend;
      end
   end
end
