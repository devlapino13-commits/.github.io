CREATE PROCEDURE HP_SUD_SUD_SET_DATA(
    SMR_ID D_ID NOT NULL,
    LAWSUIT_NO D_STR_100 NOT NULL,
    DECISION_DATE D_DATE NOT NULL,
    DEC_EFFECTIVE_DATE D_DATE NOT NULL,
    RESOLUTION D_STR_8100,
    IS_REFUSAL D_BOOL = 0)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable slc_id d_id;
declare variable sub_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date_str d_str_100;
declare variable cl_count d_integer;
declare variable gp_summ d_currency;
declare variable debt_summ d_currency;
declare variable c_name_fio d_str_255;
declare variable c_birth_date_str d_str_100;
declare variable c_pass_no d_str_100;
declare variable address_legal d_str_long;
declare variable error_text d_str_255;
begin
  k_id = (select smr_from_k_id from sud_mainreq where smr_id = :smr_id);

  select k_kontract_no, f_datetostr(k_disb_date)
  from kontract
  where k_id = :k_id
  into :k_kontract_no, :k_disb_date_str;

  cl_count = (select count(slc_id) from sud_loanclient where slc_from_smr_id = :smr_id);
  gp_summ = (select smp_sum_sud from sud_mainpay where smp_id = :smr_id)/cl_count;

  for select slc_id, kc_from_c_id, slc_sum_os + slc_sum_perc
  from sud_loanclient, knclient
  where slc_from_smr_id = :smr_id
    and kc_id = slc_from_kc_id
  into slc_id, c_id, debt_summ
  do 
  begin
    select cl.c_name_fio, f_datetostr(cl.c_birth_date), cl.c_pass_no
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio, :c_birth_date_str, :c_pass_no;
  
    select adl_address_str
    from addrlist adl
    inner join addrclient adcl on adcl.adc_from_adl_id = adl.adl_id
    where adcl.adc_from_c_id = :c_id and
          adcl.adc_address_type = 1 -- Юридический (Адрес регистрации)
    into :address_legal;
  
    -- резолюцию генерим если не отказ
    if (is_refusal = 0) then
    begin
      if (coalesce(resolution, '') <> '') then
      begin
        resolution = replace(resolution, '¬', '-');
        resolution = replace(resolution, '­', '-');
        resolution = replace(resolution, '‘', '');
        resolution = replace(resolution, '“', '«');
        resolution = replace(resolution, '”', '»');
        resolution = replace(resolution, '„', ',');
        resolution = replace(resolution, '‚', ',');
        resolution = replace(resolution, '~', '-');
        resolution = replace(resolution, '—', '-');
        resolution = replace(resolution, 'À', 'A');
        resolution = replace(resolution, 'İ', 'I');
        resolution = replace(resolution, 'І', 'I');
        resolution = replace(resolution, ascii_char(10), ' ');
        resolution = replace(resolution, ascii_char(13), ' ');
        resolution = replace(resolution, '  ', ' ');
        resolution = replace(resolution, '  ', ' ');
      end
      else
        resolution = f_format('Взыскать с %1 %2 г.р. пасп. %3 адр. %4, задолж. по ОС и %% в сумме %5р. ',
                               c_name_fio, c_birth_date_str, c_pass_no, coalesce(:address_legal, 'XXX'), debt_summ) ||
                     f_format('по дог. №%1 от %2г. и расходы по оплате ГП - %3р.',
                               k_kontract_no, k_disb_date_str, gp_summ);
    
      update sud_loanclient
      set slc_resolution = :resolution
      where slc_id = :slc_id;
    end
  end

  update sud_mainlist
  set sml_lawsuit_no = :lawsuit_no,
      sml_decision_date = :decision_date,
      sml_dec_effective_date = :dec_effective_date,
      sml_status = iif(:is_refusal = 0, 1, -2)
  where sml_id = :smr_id;

  -- RFCRU-4892
  if (not exists(select 1 from lawkredit lk, lawmain lam     -- удалить условие после полного переезда на отправку из нового модуля (сверка скриптом)
            where lk.lak_from_lam_id = lam.lam_id
              and lk.lak_from_k_id = :k_id
              and lam.lam_is_active = 1
              and lam.lam_date_begin = :decision_date))
  then
  begin
    for select slc_id
    from sud_loanclient slc
    where slc_from_smr_id = :smr_id
      and slc_sum_all > 0
    into slc_id
    do
    begin
      insert into sud_bki (sub_from_slc_id, sub_kind)
      values (:slc_id, 0)
      returning sub_id
      into sub_id;

      error_text = '';

      select error_text from hp_sud_bki_send(:sub_id)
      into error_text;

      if (error_text <> '') then
        update sud_bki
        set sub_status = -1
        where sub_id = :sub_id;
    end
  end
end
