CREATE PROCEDURE PRINT_KML_SELLER(
    PRINT_K_ID ID_TYPE)
RETURNS (
    KML_SELLER_C_ID ID_TYPE,
    KML_SELLER_FIO VARCHAR(100),
    KML_SELLER_INN VARCHAR(20),
    KML_SELLER_PASS_NO VARCHAR(50),
    KML_SELLER_PASS_DATE DATE,
    KML_SELLER_PASS_DATE_EXPIRE DATE,
    KML_SELLER_PASS_WHO VARCHAR(255),
    KML_SELLER_ADDRESS VARCHAR(255),
    KML_BANK_NAME VARCHAR(255),
    KML_BANK_NO VARCHAR(50),
    KML_BANK_BIK ID_TYPE,
    KML_PANT_TYPE STR_100,
    KML_PANT_DOC_NO STR_100,
    KML_PANT_DATE_BEGIN DATE,
    KML_PANT_DATE_END DATE,
    KML_PANT_ISSUED_BY STR_255,
    KML_PANT_MEMO_1 STR_255,
    KML_REGISTR_TYPE STR_100,
    KML_REGISTR_DOC_NO STR_100,
    KML_REGISTR_DATE_BEGIN DATE,
    KML_REGISTR_DATE_END DATE,
    KML_REGISTR_ISSUED_BY STR_255,
    KML_POLIS_TYPE STR_100,
    KML_POLIS_DOC_NO STR_100,
    KML_POLIS_DATE_BEGIN DATE,
    KML_POLIS_DATE_END DATE,
    KML_POLIS_ISSUED_BY STR_255,
    KML_C_MEMO VARCHAR(1024),
    KML_TAX_SERVICE_CODE D_STR_255,
    KML_DOC_NO D_INTEGER)
AS
declare variable k_disb_date date;
declare variable rd_id id_type;
declare variable rd_id1 id_type;
declare variable rd_id2 id_type;
begin
  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.IPF_Sellers'
  into :rd_id;

  select rd_id
  from refdata
  where rd_code = 'TerritorialTaxAdmin'
  into :rd_id1;

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.tsp_and_ipf'
  into :rd_id2;

  select kont.k_disb_date
  from kontract kont
  where kont.k_id = :print_k_id
  into :k_disb_date;

  kml_doc_no = (
    select first 1
      ol_doc_no
    from kontmat
      inner join opermat
        on om_bill_id = kml_bill_id
          and om_bill_line_no = kml_bill_line_no
          and om_oper_type = 1
      inner join operlist
        on ol_id = om_from_ol_id
          and ol_doc_no > 0
          and is_delete = 0
    where kml_object_id = :print_k_id
      and kml_object_kind = 9);

  for select distinct ag_object_id
      from kontmat, agreements ag
      where kml_from_ag_id = ag_id and
            kml_object_kind = 9 and
            kml_object_id = :print_k_id and
            ag_object_kind = 0 and
            ag.ag_from_rd_id in (:rd_id, :rd_id2) and
            kml_from_ml_id is not null
      order by kml_id
      into :kml_seller_c_id
  do
  begin
    if (kml_seller_c_id is not null) then
    begin
      select c_name_fio, c.c_inn, c.c_pass_no, c.c_pass_date, c.c_pass_who, c.c_pass_date_expire, c.c_memo
      from clients c
      where c_id = :kml_seller_c_id
      into kml_seller_fio, kml_seller_inn, kml_seller_pass_no, kml_seller_pass_date, kml_seller_pass_who,
           kml_seller_pass_date_expire, kml_c_memo;

      select al.adl_address_str
      from addrclient ac, addrlist al
      where ac.adc_from_c_id = :kml_seller_c_id and
            ac.adc_address_type = 1 and
            ac.adc_from_adl_id = al.adl_id
      into kml_seller_address;

      select bc.bc_account_no, bc.bc_from_b_id, b_name
      from bankclient bc, banks
      where bc.bc_from_c_id = :kml_seller_c_id and
            bc.bc_is_active = 1 and
            bc.bc_from_b_id = b_id rows 1
      into kml_bank_no, kml_bank_bik, kml_bank_name;

      kml_pant_type = null;
      kml_pant_doc_no = null;
      kml_pant_date_begin = null;
      kml_pant_date_end = null;
      kml_pant_issued_by = null;

      select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by,
             cd.cds_memo_1
      from clientdocs cd
      inner join passkind pk on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cds_from_c_id = :kml_seller_c_id and
            current_date between cds_doc_date_begin and cds_doc_date_end and
            cds_is_delete = 0 and
            pdt.pdt_code in ('work_patent', 'License')
      order by cds_id desc rows 1
      into :kml_pant_type, :kml_pant_doc_no, :kml_pant_date_begin, :kml_pant_date_end, :kml_pant_issued_by,
           :kml_pant_memo_1;

      kml_registr_type = null;
      kml_registr_doc_no = null;
      kml_registr_date_begin = null;
      kml_registr_date_end = null;
      kml_registr_issued_by = null;

      select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by
      from clientdocs cd
      inner join passkind pk on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cds_from_c_id = :kml_seller_c_id and
            current_date between cds_doc_date_begin and cds_doc_date_end and
            cds_is_delete = 0 and
            (pdt.pdt_code = 'certificate_chp' or
            pdt.pdt_code = 'certificate_ip')
      order by cds_id desc rows 1
      into :kml_registr_type, :kml_registr_doc_no, :kml_registr_date_begin, :kml_registr_date_end,
           :kml_registr_issued_by;

      kml_polis_type = null;
      kml_polis_doc_no = null;
      kml_polis_date_begin = null;
      kml_polis_date_end = null;
      kml_polis_issued_by = null;

      select pk.pk_name, cd.cds_doc_no, cd.cds_doc_date_begin, cd.cds_doc_date_end, cd.cds_doc_issued_by
      from clientdocs cd
      inner join passkind pk on cd.cds_from_pk_id = pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cds_from_c_id = :kml_seller_c_id and
            current_date between cds_doc_date_begin and cds_doc_date_end and
            cds_is_delete = 0 and
            pdt.pdt_code = 'Soc_polis' rows 1
      into :kml_polis_type, :kml_polis_doc_no, :kml_polis_date_begin, :kml_polis_date_end, :kml_polis_issued_by;

      kml_tax_service_code = (
         select first 1
           left(rd_code || ' ' || rd_memo, 255)
         from clientdocs
           inner join refdata
             on rd_parent_id = :rd_id1
               and rd_code = cds_memo_1
         where cds_from_c_id = :kml_seller_c_id
           and cds_from_pk_id in (68,71)
           and cds_is_delete = 0
         order by cds_id desc);

      suspend;
    end
  end
end
