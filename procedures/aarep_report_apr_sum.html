CREATE PROCEDURE AAREP_REPORT_APR_SUM(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    PARENT_NAME D_STR_100,
    REG_NAME D_STR_100,
    BALANCE_BEGIN CURR_TYPE,
    BALANCE_END CURR_TYPE,
    CHANGES CURR_TYPE,
    DEL_BAL_BEGIN CURR_TYPE,
    DEL_BAL_END CURR_TYPE,
    CHANGES_BALANCE CURR_TYPE,
    DEL_BAL_BEGIN_7 CURR_TYPE,
    DEL_BAL_END_7 CURR_TYPE,
    CHANGES_BALANCE_7 CURR_TYPE,
    DEL_BAL_BEGIN_30 CURR_TYPE,
    DEL_BAL_END_30 CURR_TYPE,
    CHANGES_BALANCE_30 CURR_TYPE,
    DISB_SUM DECIMAL(18,4),
    KOL_DISB INTEGER,
    SVZ_KOL_ACT INTEGER,
    KOL_DISB_AVG DECIMAL(4,2),
    FIRST_DATE_MONTH D_DATE,
    WORK_DAYS_QTY INTEGER)
AS
begin
      first_date_month = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date);

for select    apr.parent_name,
              apr.reg_name,
             sum(apr.balance_begin),
             sum(apr.balance_end),
             sum(apr.changes),
             sum(apr.del_bal_begin),
             sum(apr.del_bal_end),
             sum(apr.changes_balance),
             sum(apr.del_bal_begin_7),
             sum(apr.del_bal_end_7),
             sum(apr.changes_balance_7),
             sum(apr.del_bal_begin_30),
             sum(apr.del_bal_end_30),
             sum(apr.changes_balance_30),
             sum(apr.disb_sum),
             sum(apr.kol_disb),
             sum(apr.svz_kol_act),
             sum(apr.kol_disb_avg),
             apr.work_days_qty
     from aarep_report_apr (:rep_region_id, :rep_date_end) apr
     group by parent_name ,reg_name ,work_days_qty
     into
            parent_name,
            reg_name,
            balance_begin ,
            balance_end,
            changes,

            del_bal_begin,
            del_bal_end,
            changes_balance,

            del_bal_begin_7,
            del_bal_end_7,
            changes_balance_7,

            del_bal_begin_30,
            del_bal_end_30,
            changes_balance_30,
            disb_sum,
            kol_disb,
            svz_kol_act,
            kol_disb_avg  ,
            work_days_qty

  end
end
