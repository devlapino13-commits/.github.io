CREATE PROCEDURE HP_IPC_BANK_CARDS_VIEW(
    IB_ID D_INTEGER)
RETURNS (
    IBC_ID D_INTEGER,
    IBC_FROM_IB_ID D_INTEGER,
    IBC_CARD_TYPE D_INTEGER,
    RD_NAME D_STR_255,
    IBC_CMI D_STR_10,
    IBC_BIN D_STR_10)
AS
begin
  for
    select
      ibc_id,
      ibc_from_ib_id,
      ibc_card_type,
      rd_name,
      ibc_cmi,
      ibc_bin
    from ipc_bank_cards
      inner join refdata
        on rd_id = ibc_card_type
    where ibc_from_ib_id = :ib_id
  into
    :ibc_id,
    :ibc_from_ib_id,
    :ibc_card_type,
    :rd_name,
    :ibc_cmi,
    :ibc_bin
  do
  begin
    suspend;
  end
end
