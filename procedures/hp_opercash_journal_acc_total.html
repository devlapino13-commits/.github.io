CREATE PROCEDURE HP_OPERCASH_JOURNAL_ACC_TOTAL(
    OFFICE_ID D_INTEGER,
    REP_DATE_END D_DATE,
    UN_ID D_ID)
RETURNS (
    RES_AMOUNT_PLUS D_CURRENCY,
    RES_AMOUNT_MINUS D_CURRENCY,
    RES_ACCOUNT D_STR_255)
AS
begin
  for select res_account, sum(res_amount_plus), sum(res_amount_minus)
  from hp_opercash_journal_list(:office_id, :rep_date_end, 1, :un_id)
  where opc_sign <> 0
  group by 1
  order by max(res_acc_balance), max(res_acc_no), 1
  into res_account, res_amount_plus, res_amount_minus
  do
    suspend;
end
