CREATE PROCEDURE FM_FIND_KONTRACT(
    USER_ID DECIMAL(18,0),
    SEARCH_TEXT VARCHAR(20))
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(100),
    KAL_NEW_KF_NAME VARCHAR(100),
    K_DISB_DATE DATE)
AS
begin
-- Получаем список кредитов
--  1. Активных или в процессе
--  2. С даты выдачи которых прошло не более недели
  for with kont
      as (select k_id, k_kontract_no, k_disb_date, kal_new_kf_name, kal_new_reg_id
          from findkontract('', :search_text, '', 0, 0, 0, 0) -- по № группы
          where ks_state <> 2 and
                k_disb_date > current_date - 7
          union
          select k_id, k_kontract_no, k_disb_date, kal_new_kf_name, kal_new_reg_id
          from findkontract(:search_text, '', '', 0, 0, 0, 0) -- по № контракта
          where ks_state <> 2 and
                k_disb_date > current_date - 7),
      reg
      as (select r_list
          from user_reg_list(:user_id))
      select k_id, k_kontract_no, k_disb_date, kal_new_kf_name
      from kont, reg
      where r_list = kal_new_reg_id rows 20
      into k_id, k_kontract_no, k_disb_date, kal_new_kf_name
  do
  begin
    suspend;
  end
end
