CREATE PROCEDURE FM_CREL_LIST(
    C_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_C_ID DECIMAL(18,0),
    RES_C_NAME_FIO VARCHAR(100),
    RES_C_BIRTH_DATE DATE,
    RES_REL_ID VARCHAR(50),
    RES_REL_NAME VARCHAR(100),
    RES_MEMO VARCHAR(255),
    RES_CAN_EDIT SMALLINT)
AS
declare variable c_plastcard_no varchar(30);
declare variable db_id integer;
begin
 db_id = f_get_db_id();
 
  for select crel_other_c_id, c_name_fio, c_birth_date, crel_from_rel_id, rel_name, crel_memo, crel_can_edit,
             c_plastcard_no
      from hp_client_relation_list(:c_id, 0)
      into res_c_id, res_c_name_fio, res_c_birth_date, res_rel_id, res_rel_name, res_memo, res_can_edit,
           c_plastcard_no
  do
  begin
  if (db_id in (39801, 41701)) then
    begin
    res_memo = c_plastcard_no;
    end
    suspend;
  end
end
