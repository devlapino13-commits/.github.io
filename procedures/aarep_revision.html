CREATE PROCEDURE AAREP_REVISION(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OL_DATE DATE,
    OK_FROM_OL_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(30),
    OK_TYPE INTEGER,
    OK_OS DECIMAL(18,4),
    OK_PERC DECIMAL(18,4),
    OK_FINE DECIMAL(18,4),
    OK_KOMMIS DECIMAL(18,4),
    OB_OS DECIMAL(18,4),
    OB_PERC DECIMAL(18,4),
    OB_FINE DECIMAL(18,4),
    OB_KOMMIS DECIMAL(18,4))
AS
begin
  for select ol_date, ok_from_ol_id, ok_type, ok_os, ok_perc, ok_fine, ok_kommis, ob_os, ob_perc, ob_fine, ob_kommis
      from aarep_revision_operkr_operbook(:rep_date_begin, :rep_date_end)
      where ok_os + ob_os <> 0 or
            (ok_perc + ob_perc <> 0) or
            (ok_fine + ob_fine <> 0) or
            (ok_kommis + ob_kommis <> 0)
      into :ol_date, :ok_from_ol_id, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_kommis, :ob_os, :ob_perc, :ob_fine, :ob_kommis

  do
  begin
    select distinct k_kontract_no
    from operkred, kontract
    where ok_from_k_id = k_id and
          ok_from_ol_id = :ok_from_ol_id and
          ok_from_c_id = 0
    into :k_kontract_no;

    suspend;
  end
end
