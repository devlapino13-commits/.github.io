declare function add_distinct(src_str d_str_long, sub_str d_str_100) returns d_str_long
as
begin
  src_str = coalesce(src_str, '');
  if (sub_str is null or sub_str = '' or position(sub_str, src_str) > 0) then
    return src_str;
  else
    return src_str || iif(src_str = '', '', ',') || sub_str;
end

declare procedure get_contract_data(
  kontract_no d_str_20
)
returns (
  k_id bigint,
  k_disb_date d_date,
  ma_id d_id,
  kf_id d_id,
  c_id d_id,
  k_disb_kind d_smallint
)
as
begin
  select k_id, k_disb_date, k_from_ma_app_id, k_from_kf_id, k_disb_kind
  from kontract
  where k_kontract_no = :kontract_no
  into k_id, k_disb_date, ma_id, kf_id, k_disb_kind;

  if (k_id is null) then
    exception e_default 'Can''t find k_id for: ' || coalesce(kontract_no, 'NULL');
  if (ma_id is null) then
    exception e_default 'Can''t find ma_id for: ' || coalesce(kontract_no, 'NULL');

  select kc_from_c_id
  from knclient
  where kc_from_k_id = :k_id
    and kc_amount_approved > 0
  into c_id;

  if (c_id is null) then
    exception e_default 'Can''t find client for contract: ' || coalesce(kontract_no, 'NULL');
  
  suspend;
end

declare function moneta_status(k_id d_id, c_id d_id) returns d_str_code
as
declare variable eil_status_str d_str_code;
begin
  for select
        case eil_status
          when -1 then 'Временная ошибка'
          when 0 then 'В очереди на отправку'
          when 1 then 'Ожидание ответа Монеты'
          when 2 then 'Документы получены'
        end
      from extinfolist, extinfocfg
      where eil_from_c_id = :c_id
        and eic_code = 'moneta_doc'
        and eil_from_eic_id = eic_id
        and eil_status in (-1, 0, 1, 2)
        and eil_object_kind = 9
        and eil_object_id = :k_id
        and datediff(day, eil_date, current_date) <= eic_info_actual_days
      order by eil_id desc
      rows 1
      into eil_status_str
  do
    return eil_status_str;
  return 'Запрос не найден';
end

declare procedure export_moneta_docs(
  kontract_id bigint,
  arr_json d_integer
)
as
declare variable el_id d_id;
declare variable el_code d_str_code;
declare variable bls_id d_id;
declare variable file_name d_str_100;
declare variable jso d_integer;
begin
  for select el_id, el_code
  from enum_list
  where el_kind = 'extinfo_bls_kind'
    and el_code in ('moneta_doc_payment_order', 'moneta_doc_receipt')
  into el_id, el_code
  do
  begin
    for select eib_from_bls_id
    from extinfolist, extinfolistbls
    where eil_from_eic_id = 11
      and eil_object_kind = 9
      and eil_status = 2
      and eil_object_id = :kontract_id
      and eib_from_eil_id = eil_id
      and eib_kind = :el_id
    order by eil_id desc
    into bls_id do
    begin
      if (el_code = 'moneta_doc_payment_order') then
        file_name = 'Платежное поручение.pdf';
      else if (el_code = 'moneta_doc_receipt') then
        file_name = 'Квитанция о переводе.pdf';
      else
        exception e_default 'Unexpected el_code: ' || coalesce(el_code, 'NULL');
      
      res_info = add_distinct(res_info, 'Монета');
      
      jso = json.obj_create();
      json.obj_set_s(jso, 'file_name', file_name);
      json.obj_set_l(jso, 'bls_id', bls_id);
      json.arr_set_obj(arr_json, -1, jso);
    end
  end
end

declare procedure export_asp_agreement(
  client_id d_id,
  arr_json d_integer
)
as
declare variable file_name d_str_100;
declare variable bls_id d_id;
declare variable jso d_integer;
begin
  for select edf_print_name, edb_from_bls_id
  from edoc_list, edoc_frx, edoc_blank
  where edf_code = 'asp.agreement'
    and edl_object_kind = 0
    and edl_object_id = :client_id
    and edf_from_edc_id = edl_from_edc_id
    and edb_from_edf_id = edf_id
    and edb_from_edl_id = edl_id
    and edl_status = 1
  order by edl_id desc
  rows 1
  into file_name, bls_id
  do
  begin
    file_name = file_name || '.pdf';
    res_info = add_distinct(res_info, 'АСП');
    
    jso = json.obj_create();
    json.obj_set_s(jso, 'file_name', file_name);
    json.obj_set_l(jso, 'bls_id', bls_id);
    json.arr_set_obj(arr_json, -1, jso);
  end
end

declare procedure export_edocs(
  kontract_id bigint,
  ma_id d_id,
  arr_json d_integer
)
as
declare variable edf_id d_id;
declare variable edc_id d_integer;
declare variable file_name d_str_100;
declare variable bls_id d_id;
declare variable jso d_integer;
begin
  for select edf_id, edf_from_edc_id, edf_print_name
  from edoc_frx
  where edf_id in (
    2,  -- Заявка на заём
    3,  -- Договор займа
    4,  -- Приложение к договору (график)
    5,  -- Личный лист (анкета)
    28  -- Согласие на обработку персональных данных клиента
  )
  union select edf_id, 16, edf_print_name
  from edoc_frx
  where edf_id = 28 -- Согласие на обработку ПД (старый вариант)
  into edf_id, edc_id, file_name
  do
  begin
    file_name = file_name || '.pdf';
    
    for with tab_edl as (
      select edl_id, edl_status
      from edoc_list
      where edl_from_edc_id = :edc_id
        and (edl_object_kind = 9 and edl_object_id = :kontract_id)
      union all select edl_id, edl_status
      from edoc_list
      where edl_from_edc_id = :edc_id
        and (edl_object_kind = 11 and edl_object_id = :ma_id)
    )
    select edb_from_bls_id
    from tab_edl, edoc_blank
    where edb_from_edf_id = :edf_id
      and edb_from_edl_id = edl_id
      and edl_status = 1
    order by edl_id desc
    rows 1
    into bls_id
    do
    begin
      if (edf_id = 2) then
        res_info = add_distinct(res_info, 'Заявка');
      if (edf_id = 3) then
        res_info = add_distinct(res_info, 'Договор');
      if (edf_id = 4) then
        res_info = add_distinct(res_info, 'График');
      if (edf_id = 5) then
        res_info = add_distinct(res_info, 'Анкета');
      if (edf_id = 28) then
        res_info = add_distinct(res_info, 'ПД');
      
      jso = json.obj_create();
      json.obj_set_s(jso, 'file_name', file_name);
      json.obj_set_l(jso, 'bls_id', bls_id);
      json.arr_set_obj(arr_json, -1, jso);
    end
  end
end

declare procedure export_passports(
  client_id d_id,
  disb_date d_date,
  arr_json d_integer
)
as
declare variable bls_id d_id;
declare variable bls_id_2 d_id;
declare variable cds_doc_no d_str_code;
declare variable cnt d_integer;
declare variable jso d_integer;
begin
  cnt = 1;
  
  for select cds_doc_blob_1, cds_doc_blob_2, cds_doc_no
  from clientdocs, passkind, passdoctype
  where pk_id = cds_from_pk_id
    and pdt_id = pk_from_pdt_id
    and cds_from_c_id = :client_id
    and coalesce(cds_doc_date_end, '2999-12-31') >= :disb_date
    and cds_verify_status = 2
    and pdt_is_main_doc = 1
    and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
  order by cds_create_datetime desc
  into bls_id, bls_id_2, cds_doc_no
  do
  begin
    cds_doc_no = replace(cds_doc_no, ' ', '');
    
    if (bls_id is not null and bls_id > 0) then
    begin
      jso = json.obj_create();
      json.obj_set_s(jso, 'file_name', f_format('Паспорт_%1_%2(лицевая).jpg', cnt, cds_doc_no));
      json.obj_set_l(jso, 'bls_id', bls_id);
      json.arr_set_obj(arr_json, -1, jso);
      res_info = add_distinct(res_info, 'Паспорт');
    end
    
    if (bls_id_2 is not null and bls_id_2 > 0) then
    begin
      jso = json.obj_create();
      json.obj_set_s(jso, 'file_name', f_format('Паспорт_%1_%2(обратная).jpg', cnt, cds_doc_no));
      json.obj_set_l(jso, 'bls_id', bls_id_2);
      json.arr_set_obj(arr_json, -1, jso);
      res_info = add_distinct(res_info, 'Паспорт');
    end
    
    cnt = cnt + 1;
  end
end

declare procedure export_photo(
  client_id d_id,
  arr_json d_integer
)
as
declare variable bls_id d_id;
declare variable jso d_integer;
begin
  bls_id = null;
  
  select c_blob_photo_id
  from clients
  where c_id = :client_id
  into bls_id;
  
  if (bls_id is not null) then
  begin
    jso = json.obj_create();
    json.obj_set_s(jso, 'file_name', 'Фото клиента.jpg');
    json.obj_set_l(jso, 'bls_id', bls_id);
    json.arr_set_obj(arr_json, -1, jso);
    res_info = add_distinct(res_info, 'Фото');
  end
end

begin
  -- RFCRU-5381 Выгрузка документов для передачи контрагенту по договору цессии
  -- RFCRU-5920 Выгрузка документов для передачи контрагенту для просуживания в РУ
  -- RFCRU-5996 Выгрузка квитанций и платежных поручений (Монета) для подачи в суд

  res_json = '{}';
  res_info = '';
  
  jsm = json.obj_create();
  jsa = json.arr_create();

  -- Инициализация (создание пунктов меню в Finstart)
  if (action_menu = 'miInit') then
  begin
    json.arr_set_obj(jsa, -1, jso_menu_item('miRequestMonetaDocs', 'Запросить квитанции и ПП (Монета)'));
    json.arr_set_obj(jsa, -1, jso_menu_item('miExportMonetaDocs', 'Экспорт квитанций и ПП(Монета)'));
    json.arr_set_obj(jsa, -1, jso_menu_item('miExportAssignmentRU', 'Экспорт документов для цессии (Россия)'));
    json.arr_set_obj(jsa, -1, jso_menu_item('miExportCourtUZ', 'Экспорт документов для суда (Узбекистан)'));
    
    json.obj_set_arr(jsm, 'menu_items', jsa);
    res_json = json.obj_to_json(jsm);
    suspend;
    exit;
  end

  -- Получаем данные договора
  select k_id, k_disb_date, ma_id, kf_id, c_id, k_disb_kind
  from get_contract_data(:k_kontract_no)
  into k_id, k_disb_date, ma_id, kf_id, c_id, k_disb_kind;

  -- Получаем регион договора
  select reg_unicode, reg_name
  from knaktlist, region
  where knaktlist.is_delete = 0
    and kal_from_kf_id = :kf_id
    and kal_date_end >= current_date
    and kal_new_reg_id = reg_unicode
  into reg_id, reg_name;

  -- Обработчики пунктов меню
  if (action_menu = 'miRequestMonetaDocs') then
  begin
    if (k_disb_kind in (5, 7)) then
    begin
      select eil_id
      from hp_sud_extinfo_insert('moneta_doc', 'kontract', :k_id, :c_id, 0)
      into eil_id;
      
      if (eil_id is not null) then
        res_info = add_distinct(res_info, 'Запрос создан');
      else
      begin
        res_info = add_distinct(res_info, 'Существует: ' || moneta_status(k_id, c_id));
      end
    end
    else
      res_info = 'Оффлайн';
    
    suspend;
    exit;
  end

  if (action_menu = 'miExportMonetaDocs') then
  begin
    -- Блобы
    execute procedure export_moneta_docs(:k_id, :jsa);
    json.obj_set_arr(jsm, 'blobs', jsa);

    res_json = json.obj_to_json(jsm);
    suspend;
    exit;
  end

  if (action_menu = 'miExportAssignmentRU') then
  begin
    -- Блобы
    execute procedure export_moneta_docs(:k_id, :jsa);
    execute procedure export_edocs(:k_id, :ma_id, :jsa);
    execute procedure export_passports(:c_id, :k_disb_date, :jsa);
    execute procedure export_photo(:c_id, :jsa);
    json.obj_set_arr(jsm, 'blobs', jsa);

    -- Бланки
    jso = jso_blank(71, 'Расчет задолженности.pdf', 9, k_id, reg_id, rep_date, rep_date);
    -- Инициализация переменных бланка
    json.obj_set_s(jso, 'add_params.rep_region_name', reg_name);
    json.obj_set_l(jso, 'add_params.rep_kontract_id', k_id);
    json.obj_set_obj(jsm, 'blanks[-1]', jso);
    
    res_json = json.obj_to_json(jsm);
    suspend;
    exit;
  end

  if (action_menu = 'miExportCourtUZ') then
  begin
    -- Блобы
    execute procedure export_moneta_docs(:k_id, :jsa);
    execute procedure export_edocs(:k_id, :ma_id, :jsa);
    execute procedure export_passports(:c_id, :k_disb_date, :jsa);
    execute procedure export_photo(:c_id, :jsa);
    execute procedure export_asp_agreement(:c_id, :jsa);
    json.obj_set_arr(jsm, 'blobs', jsa);

    -- Бланки
    jso = jso_blank(1800, 'Претензия.pdf', 9, k_id, reg_id, rep_date, rep_date);
    json.obj_set_obj(jsm, 'blanks[-1]', jso);
    
    jso = jso_blank(1801, 'Расчет задолженности.pdf', 9, k_id, reg_id, rep_date, rep_date);
    json.obj_set_obj(jsm, 'blanks[-1]', jso);
    
    res_json = json.obj_to_json(jsm);
    suspend;
    exit;
  end

  -- Если что-то пошло не так
  res_info = 'Unknown action: ' || coalesce(action_menu, 'NULL');
  res_json = '{"error": "' || res_info || '"}';
  suspend;
end
