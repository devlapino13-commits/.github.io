CREATE PROCEDURE UXP$UNIT_LIST
RETURNS (
    UNT_UUID TYPE OF D_UUID,
    UNT_CODE TYPE OF D_STR_CODE,
    UNT_NAME TYPE OF D_STR_100,
    UNT_POSITION TYPE OF D_INTEGER,
    UNT_KIND TYPE OF D_SMALLINT,
    UNT_IS_CMD D_BOOL,
    UNT_SOURCE D_STR_CODE,
    UNT_DFM_NAME TYPE OF D_STR_CODE,
    UNT_DFM_HASH TYPE OF D_STR_CODE,
    UNT_DFM_STATE D_SMALLINT,
    UNT_PAS_NAME TYPE OF D_STR_CODE,
    UNT_PAS_HASH TYPE OF D_STR_CODE,
    UNT_PAS_STATE D_SMALLINT,
    UNT_UX_UUID D_UUID,
    RES_UX_CREATED D_BOOL,
    RES_FRAME_COUNT D_INTEGER,
    RES_OTHER_DB_CREATED D_BOOL)
AS
declare variable db_id d_integer;
declare variable unt_ux_fcode d_str_100;
declare variable uim_db_id d_integer;
declare variable uim_ux_fcode d_str_100;
begin
  db_id = uxf$db_id();

  for select unt_uuid,
             unt_code,
             unt_name,
             unt_position,
             unt_kind,
             unt_is_cmd,
             unt_source,
             unt_dfm_name,
             unt_dfm_hash,
             unt_dfm_state,
             unt_pas_name,
             unt_pas_hash,
             unt_pas_state
      from ux$unit
      into :unt_uuid,
           :unt_code,
           :unt_name,
           :unt_position,
           :unt_kind,
           :unt_is_cmd,
           :unt_source,
           :unt_dfm_name,
           :unt_dfm_hash,
           :unt_dfm_state,
           :unt_pas_name,
           :unt_pas_hash,
           :unt_pas_state
  do
  begin
    res_ux_created = 0;
    res_other_db_created = 0;
    unt_ux_uuid = null;
    unt_ux_fcode = '';

    for select uim_db_id, uim_ux_fcode
    from ux$unitimport
    where uim_unt_code = :unt_code
    into :uim_db_id, :uim_ux_fcode
    do
    begin
      if (uim_db_id = db_id) then
        unt_ux_fcode = uim_ux_fcode;
      else
       res_other_db_created = 1;
    end

    if (coalesce(unt_ux_fcode, '') <> '') then
    begin
      select ux_uuid
      from ux$code
      where ux_fcode = :unt_ux_fcode
      into unt_ux_uuid;

      if (unt_ux_uuid is not null) then
        res_ux_created = 1;
    end

    select count(*)
    from ux$unitframe
    where unf_owner = :unt_uuid
    into res_frame_count;

    select left(uni_caption, 100)
    from ux$unititem
    where uni_uuid = :unt_uuid
    into unt_name;

    suspend;
  end
end
