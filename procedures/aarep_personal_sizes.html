CREATE PROCEDURE AAREP_PERSONAL_SIZES(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    JOB_NAME STR_100,
    C_NAME_FIO STR_100,
    PS_CURR_STATUS VARCHAR(100),
    HEAD_SIZE DECIMAL(15,2),
    SHIRT_SIZE DECIMAL(15,2),
    PANTS_SIZE DECIMAL(15,2),
    SIZE_OUTERWEAR DECIMAL(15,2),
    SHOE_SIZE DECIMAL(15,2),
    EMPTY VARCHAR(20))
AS
declare variable c_id id_type;
declare variable cdc_code varchar(255);
declare variable cdv_value_number decimal(15,2);
begin
--  берутся все активные сотрудники
  for select c_id, cl.c_name_fio, jl.jl_name as job_name, r.reg_name
      from clients cl, hr_pstatus, region_list(:rep_region_id, 1, -1, 1), hr_orders ho, sa_worktype swt, joblist jl, region r
      where c_id = ps_from_c_id and
            (ps_from_reg_id = id_list) and
            ps_status = 1 and
            ps_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            ps_from_jl_id = jl_id and
            ps_from_reg_id = reg_unicode and
            ho.ho_status = 1 and
            ps_date_begin < ps_date_end and
            ps_date_begin <= :rep_date and
            :rep_date < ps_date_end and
            swt.sawt_id <> 6
      order by reg_name, job_name, c_name_fio
      into c_id, c_name_fio, job_name, reg_name
  do
  begin
    ps_curr_status = '';
    select sawt_name
    from hr_voctravel, hr_orders, sa_worktype
    where hrv_from_ho_id = ho_id and
          ho_order_type = sawt_id and
          hrv_from_c_id = :c_id and
          hrv_status = 1 and
          hrv_date_begin <= :rep_date and
          hrv_date_end_fact >= :rep_date
    order by hrv_date_begin desc
    rows 1
    into :ps_curr_status;

    head_size = 0;
    shirt_size = 0;
    pants_size = 0;
    size_outerwear = 0;
    shoe_size = 0;

    for select cdc_code, cdv_value_number
        from hp_customdatacfg_one(0, 0, :c_id, :rep_date)
        into :cdc_code, :cdv_value_number
    do
    begin
      if (cdc_code = 'size_head') then
        head_size = cdv_value_number;
      if (cdc_code = 'size_shirt') then
        shirt_size = cdv_value_number;
      if (cdc_code = 'size_pants') then
        pants_size = cdv_value_number;
      if (cdc_code = 'size_jacket') then
        size_outerwear = cdv_value_number;
      if (cdc_code = 'size_shoes') then
        shoe_size = cdv_value_number;
    end
    empty = iif((head_size + shirt_size + pants_size + size_outerwear + shoe_size) > 0, 'Заполнен', 'Не заполнен');
    suspend;
  end
end
