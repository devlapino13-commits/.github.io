CREATE PROCEDURE HP_CHECK_MINIAPP_KONTRACT_DIFF(
    K_ID DECIMAL(18,0),
    MA_ID INTEGER,
    KIND VARCHAR(20))
AS
declare variable db_id integer;
declare variable need_cancel_kk integer;
declare variable rd_id integer;
declare variable cl_diff integer;
declare variable acl_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable new_memo varchar(255);
declare variable un_id decimal(18,0);
declare variable opr_id decimal(18,0);
begin
-- Процедуры выполняется в момент синхронизации данных между договором и заявки
-- Если имеются нижеуказанные изменения и существует положительное решение кредитного комитета,
-- то процедура отменяет решение

  :db_id = f_get_db_id();

  /*select rd_id
  from refdata
  where rd_code = 'ActionGroup.Check_Loan_Data' and
        rd_is_active = 1
  into :rd_id;*/

  select rd_id
  from hp_get_check_loan_data(:k_id)
  into:rd_id;

  if (rd_id is null) then
    exit; -- Если нет анкеты КК, отменяем действие

  select acl_id
  from actionlist
  where acl_object_kind = 9 and
        acl_object_id = :k_id and
        acl_status = 2 and
        acl_from_rd_id = :rd_id
  order by acl_id desc
  rows 1
  into acl_id;
  if (acl_id is null) then
    exit; -- если нет положительного решения, завершаем процесс

  need_cancel_kk = 0;
-- Проверяем измененные данные

-- Сменился участник
  for select kc_from_c_id
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into :c_id
  do
  begin
    if (not(exists(select *
                   from miniappclient
                   where mac_from_ma_id = :ma_id and
                         mac_amount > 0 and
                         mac_from_c_id = :c_id))) then
      need_cancel_kk = 1;
  end

  if (need_cancel_kk = 0) then
  begin
    for select mac_from_c_id
        from miniappclient
        where mac_from_ma_id = :ma_id and
              mac_amount > 0
        into :c_id
    do
    begin
      if (not(exists(select *
                     from knclient
                     where kc_from_k_id = :k_id and
                           kc_amount_approved > 0 and
                           kc_from_c_id = :c_id))) then
        need_cancel_kk = 1;
    end
  end
  if (need_cancel_kk = 0) then
  begin
-- Количество участников отличается
    cl_diff = 0;
    with tab
    as (select count(*) as cl_count
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        union
        select -count(*) as cl_count
        from miniappclient
        where mac_from_ma_id = :ma_id and
              mac_amount > 0)
    select sum(cl_count)
    from tab
    into :cl_diff;
    if (cl_diff is null) then
      cl_diff = 0;
    if (cl_diff <> 0) then
      need_cancel_kk = 1;
  end

  if (need_cancel_kk = 0) then
  begin
-- Проверяем срок кредита
    if (kind = 'k->m') then
    begin
      if (exists(select *
                 from kontract, miniapp
                 where k_id = :k_id and
                       ma_id = :ma_id and
                       ma_date_end >= '31.12.2999' and
                       k_kolich_repay < ma_kolich_repay)) then
        need_cancel_kk = 1;
    end
    else
    begin
      if (exists(select *
                 from kontract, miniapp
                 where k_id = :k_id and
                       ma_id = :ma_id and
                       ma_date_end >= '31.12.2999' and
                       ma_kolich_repay < k_kolich_repay)) then
        need_cancel_kk = 1;
    end
  end

  if (need_cancel_kk = 0) then
  begin
-- Проверка увеличения суммы
    if (kind = 'k->m') then
    begin
-- Меняется договора
      if (exists(select *
                 from knclient, miniappclient
                 where kc_from_k_id = :k_id and
                       kc_amount_approved > 0 and
                       mac_from_ma_id = :ma_id and
                       mac_amount > 0 and
                       kc_amount_approved > mac_amount and
                       mac_from_c_id = kc_from_c_id)) then
        need_cancel_kk = 1;
    end
    else
    begin
-- Меняется заявка
      if (exists(select *
                 from knclient, miniappclient
                 where kc_from_k_id = :k_id and
                       kc_amount_approved > 0 and
                       mac_from_ma_id = :ma_id and
                       mac_amount > 0 and
                       mac_amount > kc_amount_approved and
                       mac_from_c_id = kc_from_c_id)) then
        need_cancel_kk = 1;
    end
  end

  if (need_cancel_kk = 1) then
  begin
    select k_user_id
    from kontract
    where k_id = :k_id
    into :un_id;

    --S.K. 12.11.2019
    --delete from zalog where z_from_k_id = :k_id;

    update actionlist
    set acl_status = -1,
        acl_from_un_id = :un_id
    where acl_id = :acl_id;

    new_memo = 'Доработка: Отозван из КК в связи с изменениями условий заявки';
    execute procedure hp_objectprops_memo_set(9, :k_id, -1, :rd_id, :new_memo, :un_id);
  end
end
