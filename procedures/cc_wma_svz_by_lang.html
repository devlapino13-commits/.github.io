CREATE PROCEDURE CC_WMA_SVZ_BY_LANG(
    REGION_ID INTEGER,
    LANGUAGE_CODE STR_100)
RETURNS (
    C_ID ID_TYPE,
    C_NAME_FIO STR_100,
    C_PHONE STR_100)
AS
declare variable rep_tj smallint;
declare variable rep_uz smallint;
declare variable rep_kg smallint;
begin
  rep_tj = iif(language_code = 'Language.TJ', 1, 0);
  rep_uz = iif(language_code = 'Language.UZ', 1, 0);
  rep_kg = iif(language_code = 'Language.KG', 1, 0);

  for select t.cws_c_id
      from cc_wma_svz t
      where t.cws_region_id = :region_id and
            ((((t.cws_tj = 1) and
            (:rep_tj = 1)) or
            ((t.cws_uz = 1) and
            (:rep_uz = 1)) or
            ((t.cws_kg = 1) and
            (:rep_kg = 1))) or
            (:language_code = 'Language.RU'))
      into :c_id
  do
  begin
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio;

    c_phone = f_cont_info('clients', c_id, 'cell_main', 'ok');

    suspend;
  end
end
