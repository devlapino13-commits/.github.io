CREATE PROCEDURE CM_KG_0140_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kc_circle d_integer;
declare variable max_amount d_integer;
begin
  /* Блок по суммам для всех групповых
     проверка только для изменяющихся величин. Н-р сумма в зависимости от цикла.
     Неизменные величины (н-р % ставка) проверяются в общих блоках
     т.к. наличие или отсутвие консультации влияет только на % ставку => на блок консультация не влияет
  */
  res_error_is_critical = 1;
  for select kc.kc_amount_approved,
             kc_circle,
             cl.c_name_fio
  from kontract kont
  inner join product pr on k_from_pr_id = pr_id and
        pr_code containing '.GR.' and
        pr_code containing '.STANDART.' and
        pr_code not containing '.Actiya_Ishenim_Refin.' and
        kont.k_refund = 0 -- не рефинансирование
  inner join knclient kc on k_id = kc_from_k_id and
        kc_amount_approved > 0
  inner join clients cl on cl.c_id = kc.kc_from_c_id
  where k_id = :k_id
  into :kc_amount, :kc_circle, :res_info_text
  do
  begin
    max_amount = 0; -- 0 т.к. если цикл 4 и более тогда максимально допустимая сумма = максимально допустимой сумме в параметрах продукта (т.е. 200 000), а проверка на min и max показатели в соотвествии с параметрами продукта реализована в другом кондишене. Цель подобной реалзации: чтобы одни и те же проверки не дублровались в разныз блоках
    if (kc_circle <= 1) then
         max_amount = 80000;
    if (kc_circle between 2 and 3) then
         max_amount = 150000;
    if ((max_amount <> 0) and (kc_amount >  max_amount)) then
    begin
        res_error_text = 'У клиента цикл = '||kc_circle||'. Для клиента, максимальная допустимая сумма, по данному продукту, = '||max_amount;
        suspend;
    end
  end
end
