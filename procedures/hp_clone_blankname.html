CREATE PROCEDURE HP_CLONE_BLANKNAME(
    ORIG_BLN_ID D_INTEGER,
    NEW_NAME D_STR_255,
    UN_ID D_ID)
RETURNS (
    BLN_ID D_ID)
AS
declare variable new_blb_number d_integer;
declare variable orig_blb_number d_integer;
begin
  insert into blankname (bln_name, bln_memo, bln_is_active, bln_kind, bln_show_mode, bln_component,
                         bln_is_read_only, bln_can_export, bln_print_status)
  select :new_name, bln_memo, bln_is_active, bln_kind, bln_show_mode, bln_component, bln_is_read_only,
         bln_can_export, bln_print_status
  from blankname
  where bln_id = :orig_bln_id
  returning bln_id
  into bln_id;

  select blb_number
  from blankblob
  where blb_from_bln_id = :orig_bln_id
  order by blb_date_begin desc, blb_number desc
  rows 1
  into orig_blb_number;


  insert into blankblob (blb_from_bln_id, blb_date_begin, blb_blob, blb_params,
                         blb_frx_blob, blb_json_blob, blb_frx_status, blb_font_style, blb_font_size, blb_from_un_id)
  select :bln_id, current_date, blb_blob, blb_params, blb_frx_blob,
         blb_json_blob, blb_frx_status, blb_font_style, blb_font_size, :un_id
  from blankblob
  where blb_from_bln_id = :orig_bln_id and
        blb_number = :orig_blb_number
  returning blb_number
  into new_blb_number;

  insert into blankjsonlink (bjl_from_bln_id, bjl_from_blb_number, bjl_from_bjf_id)
  select :bln_id, :new_blb_number, bjl_from_bjf_id
  from blankjsonlink
  where bjl_from_bln_id = :orig_bln_id and
        bjl_from_blb_number = :orig_blb_number;

  insert into productblank (prb_from_pr_id, prb_from_bln_id, prb_date_begin, prb_date_end, prb_from_un_id)
  select prb_from_pr_id, :bln_id, current_date, null, :un_id
  from productblank
  where prb_from_bln_id = :orig_bln_id and
        prb_date_begin <= current_date and
        coalesce(prb_date_end, current_date) >= current_date;

  Suspend;
end
