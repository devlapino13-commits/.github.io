CREATE PROCEDURE AAREP_SHOWHIST_PNUMBER(
    C_ID D_ID)
RETURNS (
    RES_ID D_INTEGER,
    RES_CELL_NO D_STR_255,
    RES_DATETIME D_DATETIME,
    RES_DATETIMEV D_DATETIME,
    RES_UN_FIO D_STR_100,
    RES_UN_FIOV D_STR_100)
AS
declare variable rd_id d_integer;
declare variable ovs_datetime d_datetime;
declare variable un_fio d_str_100;
declare variable status d_integer;
begin
  rd_id = f_get_rd_id('ContactInfo.Cell_Main');
  for
    select
      ovs_id,
      ovs_value,
      ovs_datetime,
      un_fio,
      iif(ovs_add_value = 2, 1, 0)
    from objectprops
      inner join objectvalues
        on ovs_from_opr_id = opr_id
          and ovs_is_active = 1
          and ovs_value <> ''
      left join username
        on un_id = ovs_from_un_id
    where opr_object_kind = '0'
      and opr_object_id = :c_id
      and opr_subject_kind = '-1'
      and opr_subject_id = :rd_id
    order by ovs_datetime desc
  into
    :res_id,
    :res_cell_no,
    :ovs_datetime,
    :un_fio,
    :status
  do
  begin
    res_datetime = null;
    res_datetimev = null;
    res_un_fio = null;
    res_un_fiov = null;

    if (status = 1) then
    begin
      res_datetimev = ovs_datetime;
      res_un_fiov = un_fio;
    end else
    begin
      res_datetime = ovs_datetime;
      res_un_fio = un_fio;
    end
    suspend;
  end
end
