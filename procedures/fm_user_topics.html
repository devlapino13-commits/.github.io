CREATE PROCEDURE FM_USER_TOPICS(
    UN_ID DECIMAL(18,0))
RETURNS (
    MG_ID INTEGER,
    MG_CODE VARCHAR(255),
    MG_TOPIC VARCHAR(255),
    MG_MAIL VARCHAR(255))
AS
declare variable cm_database_id integer;
declare variable un_name varchar(100);
declare variable c_id decimal(18,0);
declare variable job_code varchar(255);
declare variable reg_id integer;
declare variable access_code varchar(100);
declare variable need_suspend integer;
declare variable branch_id integer;
begin
  -- Процедура возвращает список топиков для подписки на уведомления

  -- Возможные результаты:
  --  staff_all Все пользователи мобильного приложения
  --  developers Программисты, в целях тестирования
  --  finstart_support Тех.поддержка
  --  loan_officer СВЗ, КС (кто занимается выдачей)
  --  lo_manager Кредитные менеджера, Старшие СВЗ
  --  staff_region Все пользователи в регионах
  --  staff_hq Все пользователи в ГО

  -- если необходимо добавить новую, сделать это можно в справочнике maingroup.
  -- При этом:
  -- Поле RD_Memo должен содержать имя топика включая код страны
  -- Поле RD_Str_ID должен содержать прикрепленный email на эту рассылку.

  -- Если у пользователя не заполнен google_id, завершаем работу
  -- далее будет подписка в зависимости от должности и других прав
  select un_name, un_from_c_id
  from username
  where un_id = :un_id
  into :un_name, :c_id;

  if (un_name is null or c_id is null) then
    exit;

  -- получаем должность, офис
  job_code = '';
  reg_id = null;
  select ps_from_reg_id, jk_system_code, region.parent
  from hr_pstatus, joblist, jobkind, region
  where ps_from_c_id = :c_id and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id and
        reg_unicode = ps_from_reg_id
  into :reg_id, job_code, branch_id;

  :cm_database_id = f_get_db_id();
  for select rd_id, rd_code, coalesce(rd_memo, ''), coalesce(rd_str_id, '')
      from refdata
      where rd_main_code = 'mailgroup' and
            rd_parent_id is not null
      into :mg_id, mg_code, mg_topic, mg_mail
  do
  begin
    need_suspend = 0;

    -- По правам проверяем на Программиста
    if (mg_code = 'mailgroup.developers') then
    begin
      if (un_name in ('sislamov', 'kkarimov')) then
        need_suspend = 1;
      else
      begin
        access_code = 'FinStart.Programmer';
        if (exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, :access_code)
                   where has_access = 1)) then
          need_suspend = 1;
      end
    end

    -- По правам проверяем на Тех.поддержку FinStart
    if (mg_code = 'mailgroup.finstart_support') then
    begin
      access_code = 'FinStart.Permission_Administrator';
      if (exists(select *
                 from hp_permission_access_check(:un_id, 'PermActions', 0, :access_code)
                 where has_access = 1)) then
        need_suspend = 1;
      else
      begin
        access_code = 'FinStart.Permission_Operator';
        if (exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, :access_code)
                   where has_access = 1)) then
          need_suspend = 1;
      end
    end

    -- если есть действующий приказ
    if (reg_id is not null) then
    begin
      -- подписка на "Все"
      if (mg_code = 'mailgroup.staff_all') then
        need_suspend = 1;

      -- Проверка на кредитного спеца
      if (mg_code = 'mailgroup.loan_officer') then
      begin
        if (job_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner', 'JobSystem.LO_Senior', 'JobSystem.Region_KM', 'JobSystem.Region_Director')) then
          need_suspend = 1;

        -- в целях тестирования
        if (un_name in ('sislamov', 'aschegolskiy', 'vqaragozov')) then
          need_suspend = 1;
      end

      -- Проверка на кредитного спеца
      if (job_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner', 'JobSystem.LO_Senior', 'JobSystem.Region_KM', 'JobSystem.Region_Director') and
          cm_database_id = 81001) then
      begin
        --if (mg_code='mailgroup.loan_officer_moscow' and branch_id=1010000) then need_suspend=1;
        if (mg_code = 'mailgroup.loan_officer_moscow' and
            branch_id in (1010001, 1010002)) then
          need_suspend = 1;
        if (mg_code = 'mailgroup.loan_officer_vostok' and
            branch_id = 1010003) then
          need_suspend = 1;
        if (mg_code = 'mailgroup.loan_officer_ural' and
            branch_id = 1010004) then
          need_suspend = 1;
        if (mg_code = 'mailgroup.loan_officer_sibir' and
            branch_id = 1010005) then
          need_suspend = 1;
        if (mg_code = 'mailgroup.loan_officer_sever' and
            branch_id = 1420001) then
          need_suspend = 1;
        if (mg_code = 'mailgroup.loan_officer_central' and
            branch_id = 1590000) then
          need_suspend = 1;
      end

      -- Проверка на менеджера
      if (mg_code = 'mailgroup.lo_manager') then
      begin
        if (job_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.Region_Director')) then
          need_suspend = 1;

        -- в целях тестирования
        if (un_name in ('sislamov', 'aschegolskiy')) then
          need_suspend = 1;
      end

      -- Проверка на Регионы
      if (mg_code = 'mailgroup.staff_region') then
      begin
        if (not((cm_database_id in (39801, 41701) and
            reg_id = 1090100) or (cm_database_id not in (39801, 41701) and
            reg_id = 1010100))) then
          need_suspend = 1;
      end

      -- Проверка на Головной офис
      if (mg_code = 'mailgroup.staff_hq') then
      begin
        if ((cm_database_id in (39801, 41701) and
            reg_id = 1090100) or (cm_database_id not in (39801, 41701) and
            reg_id = 1010100)) then
          need_suspend = 1;
      end
    end

    if (need_suspend = 1) then
      suspend;

  end
end
