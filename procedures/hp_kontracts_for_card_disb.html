CREATE PROCEDURE HP_KONTRACTS_FOR_CARD_DISB(
    REP_REGION_ID D_INTEGER,
    REP_DATE D_DATE,
    REP_LIST_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    REC_ID D_INTEGER,
    K_ID D_ID,
    REG_ID D_INTEGER,
    REG_NAME D_STR_100,
    K_KONTRACT_NO D_STR_100,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_DISB_DATE D_DATE,
    K_DISB_KIND D_INTEGER,
    PR_NAME D_STR_255,
    PAY_PROJECT D_STR_CODE,
    PAY_STATUS D_SMALLINT,
    PAY_STATUS_TEXT D_STR_100,
    PAY_ERROR_TEXT D_STR_LONG,
    CLIENT_TOTAL D_INTEGER,
    CLIENT_DONE D_INTEGER,
    CLIENT_ERROR D_INTEGER,
    CLIENT_CARDS D_INTEGER,
    RES_CHECK_STATE D_SMALLINT,
    K_ID_STR D_STR_20,
    C_NAME_FIO D_STR_100,
    FPS_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    CITIZEN D_STR_255,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE D_DATE,
    KL_DATETIME D_DATETIME,
    DISB_ATTEMPTS_COUNT D_INTEGER)
AS
declare variable check_kind d_integer;
declare variable obj_kind d_integer;
declare variable c_id d_id;
declare variable db_id d_integer;
declare variable tmp_text d_str_long;
declare variable kc_id d_id;
declare variable fps_manual_disb d_integer;
declare variable c_citizen d_integer;
declare variable wait_freeze boolean;
declare variable fcb_id d_integer;
declare variable esm_project d_str_20;
begin
  if (rep_list_kind not in ('new', 'process', 'done', 'error', 'wait')) then
    exception e_default 'Unknown list_kind: ' || coalesce(rep_list_kind, '<NULL>');

  db_id = f_get_db_id();

  if (db_id in (39801, 41701)) then
    pay_project = 'elcard';
  if (db_id = 81001) then
  begin
    pay_project = 'Карта';
    fps_manual_disb = f_get_const('fps_manual_disb');
  end

  check_kind = f_enum('condition_check_kind', 'loan_disb', 'id');
  obj_kind = f_objkind('kontract', 'id');

  rec_id = 0;
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin
    reg_name = '';
    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    if (rep_list_kind in ('new', 'wait')) then -- готовые к выдаче
    begin
      if (rep_date <> current_date) then
        Exit;

      pay_status = 0;
      pay_status_text = '';

      for select kont.k_id, kont.k_kontract_no, kont.k_amount_approved, kont.k_disb_date, kont.k_disb_kind, p.pr_name
          from kontcarddisb kd, kontract kont, product p
          where kd.kkd_from_k_id = kont.k_id and
                kd.kkd_from_reg_id = :reg_id and
                kont.k_from_pr_id = p.pr_id
          into :k_id, :k_kontract_no, :k_amount_approved, :k_disb_date, :k_disb_kind, :pr_name
      do
      begin
        if (db_id = 81001) then
        begin
          wait_freeze = exists(
            select * from hp_get_freeze_data(:k_id)
            where freeze_datetime_end > current_timestamp);
          if ((rep_list_kind = 'new' and wait_freeze) or
              (rep_list_kind = 'wait' and not wait_freeze)) then
            continue;
        end

        -- RFCRU-4714
        select fbl_from_fcb_id
        from knclient
        left join fps_loan_banks on flb_from_kc_id = kc_id
        left join fps_client_names on fcn_from_fcb_id = fbl_from_fcb_id
        where kc_from_k_id = :k_id
          and fcn_status = 2
        order by fcn_id desc
        rows 1
        into fcb_id;

        disb_attempts_count = 0;
        select count(*)
        from fps_client_banks_checks fcc
        where fcc.fcc_from_fcb_id = :fcb_id and
              datediff(hour, fcc_datetime, current_timestamp) <= 24 -- Счетчик обновляется каждые 24 часа см. FM_SET_FPS_LOAN_BANKS
        into disb_attempts_count;

        disb_attempts_count = coalesce(disb_attempts_count, 0);

        pay_status = 2; -- Еще не выданные
        pay_status_text = '';
        pay_error_text = '';
        client_total = 0;
        client_cards = 0; -- проверить наличие банк карт
        client_done = 0;
        client_error = 0;
        kl_datetime = null;

        for select kc_id, kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_id, c_id
        do
        begin
            c_name_fio = '';
            fps_name_fio = '';

            select
              c_name_fio,
              '',
              c_birth_date,
              c_citizen,
              c_pass_no,
              c_pass_date
            from clients
            where c_id = :c_id
            into
              :c_name_fio,
              :fps_name_fio,
              :c_birth_date,
              :c_citizen,
              :c_pass_no,
              :c_pass_date;
            citizen = (select rd_name from refdata where rd_id = :c_citizen);
            client_total = client_total + 1;
            if (db_id = 81001 and k_disb_kind = 7) then
            begin
              if ((select coalesce(fpb_code, '') from hp_fps_get_loan_bank(:k_id, :c_id)) <> '') then
                client_cards = client_cards + 1;
              if (fps_manual_disb = 0) then
              begin
                select fcn_name
                from fps_loan_banks
                  left join fps_client_banks on fbl_from_fcb_id = fcb_id
                  left join fps_client_names on fcn_from_fcb_id = fcb_id
                where flb_from_kc_id = :kc_id
                  and fcn_status = 2
                order by fcn_id desc
                rows 1
                into fps_name_fio;
              end
            end
            else
            if (db_id in (39801, 41701) and k_disb_kind = 6) then
            begin
              if (exists(select * from hp_client_bank_card_mb(:c_id))) then
                client_cards = client_cards + 1;
            end
            else
            begin
              if ((select coalesce(token, '') from hp_client_bank_card(:c_id)) <> '') then
                client_cards = client_cards + 1;
            end
        end

        res_check_state = 0; -- не проверен на соответствие
        select iif(ccs_errors > 0, -2, 1)
        from condcheckstate
        where ccs_object_kind = :obj_kind and
              ccs_object_id = :k_id and
              ccs_check_kind = :check_kind
        into res_check_state;
        if (res_check_state is null) then
          res_check_state = 0;

        rec_id = rec_id + 1;

        if (db_id = 81001) then
        begin
          pay_project = 'Карта';

          if (k_disb_kind = 7) then
          begin
            tmp_text = '';
            pay_project = 'СБП';

            select list(c_name_fio || ': ' || error_text)
            from hp_fps_kontract_check(:k_id, '.loan_disb.')
            into :tmp_text;

            if (tmp_text is null) then
              tmp_text = '';

            if (tmp_text <> '') then
            begin
              res_check_state = -2;

              if (pay_error_text <> '') then
                pay_error_text = pay_error_text || ', ';

              pay_error_text = pay_error_text || tmp_text;
            end
          end
          else
          if (k_disb_kind = 8) then
            pay_project = 'Платёжное поручение';
          kl_datetime = (
            select first 1
              eds_status_datetime
            from edoc_list
              inner join edoc_sign
                on eds_from_edl_id = edl_id
                  and eds_status = 1
            where edl_object_kind = 9
              and edl_object_id = :k_id
              and edl_from_edc_id = f_edoc_kind('loan.agreement', 'id')
              and edl_status = 1);
        end

        k_id_str = k_id;
        suspend;
      end
    end
    else
    begin
      -- выданные по базе
      for select kont.k_id, kont.k_kontract_no, kont.k_amount_approved, kont.k_disb_date, kont.k_disb_kind, p.pr_name
          from kontstate ks, kontract kont, product p
          where ks.ks_state = 1 -- выданные по Kontstate
                and
                ks.ks_ol_is_delete = 0 and
                ks.ks_region_id = :reg_id and
                ks.ks_date_begin = :rep_date and
                ks.ks_from_k_id = kont.k_id and
                kont.k_is_online = 1 and -- только онлайн выдачи
                kont.k_disb_kind in (5, 6, 7, 8) and -- только перевод на карту + СБП
                kont.k_from_pr_id = pr_id
          into :k_id, :k_kontract_no, :k_amount_approved, :k_disb_date, :k_disb_kind, :pr_name
      do
      begin
        c_name_fio = '';
        fps_name_fio = '';
        select status, description, client_total, client_done, client_error, client_cards
        from hp_kont_card_disb_status(:k_id)
        into :pay_status, :pay_error_text, :client_total, :client_done, :client_error, client_cards;

        if (pay_status = 2) then -- если займ выдан, НО нет записи в ExtSysMain, считаем ошибкой
        begin
          if (pay_error_text = '') then
            pay_error_text = 'Займ выдан, но нет записи об оплате на карту';

          pay_status = -1;
        end

        pay_status_text = trim(case pay_status
                                 when -1 then 'Ошибка'
                                 when 0 then 'На обработке'
                                 when 1 then 'Выдан'
                               end);

        if ((rep_list_kind = 'process' and
            pay_status = 0) or
            (rep_list_kind = 'done' and
            pay_status = 1) or
            (rep_list_kind = 'error' and
            pay_status = -1)) then
        begin
          rec_id = rec_id + 1;

          res_check_state = 0; -- не проверен на соответствие
          select iif(ccs_errors > 0, -2, 1)
          from condcheckstate
          where ccs_object_kind = :obj_kind and
                ccs_object_id = :k_id and
                ccs_check_kind = :check_kind
          into res_check_state;
          if (res_check_state is null) then
            res_check_state = 0;

          select c_name_fio
          from knclient
            left join clients on kc_from_c_id = c_id
          where kc_from_k_id = :k_id
            and kc_amount_approved > 0
          rows 1
          into c_name_fio;

          if (db_id = 81001) then
          begin
            pay_project = 'Карта';
            if (k_disb_kind = 7) then
            begin
              select fcn_name
              from knclient
                left join fps_loan_banks on flb_from_kc_id = kc_id
                left join fps_client_names on fcn_from_fcb_id = fbl_from_fcb_id
              where kc_from_k_id = :k_id
                and fcn_status = 2
              order by fcn_id desc
              rows 1
              into fps_name_fio;

              tmp_text = '';
              pay_project = 'СБП';
  
              select list(c_name_fio || ': ' || error_text)
              from hp_fps_kontract_check(:k_id, '.loan_disb.')
              into :tmp_text;
  
              if (tmp_text is null) then
                tmp_text = '';

              if (tmp_text <> '') then
              begin
                res_check_state = -2;

                if (pay_error_text <> '') then
                  pay_error_text = pay_error_text || ', ';
  
                pay_error_text = pay_error_text || tmp_text;
              end
            end
            else
            if (k_disb_kind = 8) then
              pay_project = 'Платёжное поручение';

            esm_project = '';

            select esm_project
            from extsysmain
            where esm_from_k_id = :k_id
              and esm_status = 1
              and esm_object_kind = 0
              and esm_project in ('moneta_disb', 'wirebank_disb')
            into esm_project;

            esm_project = case esm_project
                            when 'moneta_disb' then 'Монета'
                            when 'wirebank_disb' then 'ТКБ'
                            else ''
                          end;

            pay_project = pay_project || '/' || esm_project;
          end

          k_id_str = k_id;
          suspend;
        end
      end
    end
  end
end
