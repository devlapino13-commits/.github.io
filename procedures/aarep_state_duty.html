CREATE PROCEDURE AAREP_STATE_DUTY(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME STR_255,
    KAL_NEW_KF_NAME STR_100,
    K_KONTRACT_NO STR_20,
    K_FACT_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED CURR_TYPE,
    K_KOLICH_CLIENT_APPROVED D_SMALLINT,
    BALANCE_BEGIN CURR_TYPE,
    ACCRUED_PERIOD CURR_TYPE,
    PAYD_PERIOD CURR_TYPE,
    WRITEOFF_PERIOD CURR_TYPE,
    BALANCE_END CURR_TYPE,
    KIND_NAME STR_100,
    CANCELED_PERIOD CURR_TYPE,
    RESTORED_PERIOD CURR_TYPE)
AS
declare variable K_ID ID_TYPE;
declare variable DB_ID D_INTEGER;
declare variable RD_ID_GOSPOSH ID_TYPE;
declare variable RD_ID_EXEC ID_TYPE;
declare variable RD_ID_CANCEL ID_TYPE;
declare variable RD_ID_RESTOR ID_TYPE;
declare variable RD_ID ID_TYPE;
begin

  db_id         = f_get_db_id();
  rd_id_gosposh = f_get_rd_id('OperExtraKind.LawClientExpense');             -- Госпошлина
  rd_id_exec    = f_get_rd_id('OperExtraKind.ExecutiveInscription');         -- Исполнительная надпись
  rd_id_cancel  = f_get_rd_id('OperExtraKind.LawClientExpenseCancel');       -- Отмена
  rd_id_restor  = f_get_rd_id('OperExtraKind.LawClientExpenseRestoration');  -- Восстановление

  for select distinct o.ox_object_id, o.ox_from_rd_id, rd.rd_name
      from operextra o
      join refdata rd on rd.rd_id = o.ox_from_rd_id
      join operlist ol on ol.ol_id = o.ox_from_ol_id
      where iif(:db_id = 39801, cast(ol.ol_datetime as date), o.ox_ol_date) <= :rep_date_end
        and o.ox_from_rd_id in (:rd_id_gosposh, case when :db_id = 39801 then :rd_id_exec else :rd_id_gosposh end)
        and o.ox_object_kind = 9
      into k_id, rd_id, kind_name
  do
  begin
    select reg.reg_name, kt.kal_new_kf_name, k.k_kontract_no, k.k_fact_disb_date, k.k_amount_approved, k.k_kolich_client_approved
    from kontract k
    inner join knaktlist kt on kt.kal_from_kf_id = k.k_from_kf_id
    inner join region reg on reg.reg_unicode = kt.kal_new_reg_id
    where kt.is_delete = 0
        and:rep_date_end between kt.kal_date and kt.kal_date_end
        and k.k_id = :k_id
    into reg_name, kal_new_kf_name, k_kontract_no, k_fact_disb_date, k_amount_approved,k_kolich_client_approved;

    balance_begin = 0;
    select sum(o.ox_amount_plus - o.ox_amount_minus - o.ox_amount_writeoff)
    from operextra o
    join operlist ol on ol.ol_id = o.ox_from_ol_id
    where o.ox_object_kind = 9
      and o.ox_object_id = :k_id
      and iif(:db_id = 39801, cast(ol.ol_datetime as date), o.ox_ol_date) <= :rep_date_begin
      and o.ox_ol_is_active = 1
      and o.ox_from_rd_id = :rd_id
    into :balance_begin;
    :balance_begin = coalesce(:balance_begin, 0); -- Остаток по гос. пошлине на начало

    accrued_period = 0;
    writeoff_period = 0;
    select sum(o.ox_amount_plus), sum(o.ox_amount_writeoff)
    from operextra o
    join operlist ol on ol.ol_id = o.ox_from_ol_id
    where o.ox_object_kind = 9
      and o.ox_object_id = :k_id
      and iif(:db_id = 39801, cast(ol.ol_datetime as date), o.ox_ol_date) between :rep_date_begin and :rep_date_end
      and o.ox_ol_is_active = 1
      and o.ox_from_rd_id = :rd_id
    into :accrued_period, :writeoff_period;
    :accrued_period  = coalesce(:accrued_period, 0);   -- Начислено за период
    :writeoff_period = coalesce(:writeoff_period, 0);  -- Списано за период

    payd_period = 0;
    if (:db_id = 39801) then  -- KZ исключаем сторно отмен
      select sum(o.ox_amount_minus)
      from operextra o
      join operlist ol on ol.ol_id = o.ox_from_ol_id
      where o.ox_object_kind = 9
        and o.ox_object_id = :k_id
        and cast(ol.ol_datetime as date) between :rep_date_begin and :rep_date_end
        and o.ox_ol_is_active = 1
        and o.ox_from_rd_id = :rd_id
        and not exists (select 1 from operextra_execution_note oen
        where oen.oen_object_kind = o.ox_object_kind
        and oen.oen_object_id = o.ox_object_id
        and oen.oen_from_ox_id_cancel = o.ox_id)
      into :payd_period;
    else  -- KG
      select sum(o.ox_amount_minus)
      from operextra o
      where o.ox_object_kind = 9
       and o.ox_object_id = :k_id
       and o.ox_ol_date between :rep_date_begin and :rep_date_end
       and o.ox_ol_is_active = 1
       and o.ox_from_rd_id = :rd_id
      into :payd_period;
    :payd_period = coalesce(:payd_period, 0); -- Выплачено за период

    canceled_period = 0;
    restored_period = 0;
    if (:db_id = 39801 and :rd_id = :rd_id_exec) then  -- KZ only
    begin
      select sum(o.ox_amount_minus)
      from operextra o
      join operlist ol on ol.ol_id = o.ox_from_ol_id
      inner join operextra_execution_note oen on o.ox_object_kind = oen.oen_object_kind
      and o.ox_object_id = oen.oen_object_id
      and o.ox_id = oen.oen_from_ox_id_cancel
      and oen.oen_from_rd_id = :rd_id_cancel
      where o.ox_object_kind = 9
        and o.ox_object_id = :k_id
        and cast(ol.ol_datetime as date) between :rep_date_begin and :rep_date_end
        and o.ox_ol_is_active = 1
      into :canceled_period;
      :canceled_period = coalesce(:canceled_period, 0); -- Отменена исполнительная надпись

      select sum(o.ox_amount_plus)
      from operextra o
      join operlist ol on ol.ol_id = o.ox_from_ol_id
      inner join operextra_execution_note oen on o.ox_object_kind = oen.oen_object_kind
      and o.ox_object_id = oen.oen_object_id
      and o.ox_id = oen.oen_from_ox_id_restoration
      and oen.oen_from_rd_id = :rd_id_restor
      where o.ox_object_kind = 9
        and o.ox_object_id = :k_id
        and cast(ol.ol_datetime as date) between :rep_date_begin and :rep_date_end
        and o.ox_ol_is_active = 1
      into :restored_period;
      :restored_period = coalesce(:restored_period, 0); -- Восстановлена по решению суда
    end

    :balance_end = 0;
    if (:db_id = 39801) then
      :balance_end = :balance_begin + :accrued_period - :canceled_period + :restored_period - :payd_period - :writeoff_period; -- Остаток на конец (KZ, с учетом отмен/восстановлений)
    else
      :balance_end = :balance_begin + :accrued_period - :payd_period - :writeoff_period; -- Остаток на конец

    if (:balance_begin + accrued_period + payd_period + writeoff_period <> 0) then
      suspend;
  end
end
