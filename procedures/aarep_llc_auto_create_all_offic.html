CREATE PROCEDURE AAREP_LLC_AUTO_CREATE_ALL_OFFIC(
    REP_DATE_BEGIN DATE,
    REP_DATE DATE)
RETURNS (
    REG_ID INTEGER,
    LM_ID DECIMAL(18,0))
AS
begin
  for select reg_unicode
      from region
      where reg_status = 3
      order by reg_unicode
      into :reg_id
  do
  begin
    delete from llc_main
    where lm_from_reg_id = :reg_id and
          lm_date_status = :rep_date;

    select result
    from get_kv_id(10)
    into :lm_id;

    insert into llc_main (lm_id, lm_date, lm_date_status, lm_from_reg_id, lm_no, lm_memo, lm_no_str, lm_is_signed)
    values (:lm_id, :rep_date, :rep_date, :reg_id, 0, '', '0', 1);

    insert into llc_sign (ls_from_lm_id, ls_number, ls_sign_type, ls_fio)
    values (:lm_id, 1, 1, 'Лариса Белоусова');

    insert into llc_kont (lk_from_lm_id, lk_from_k_id, lk_deliq_days, lk_from_reg_id, lk_balance, lk_status, lk_percent,
                          lk_date, lk_date_end, lk_memo, lk_is_indiv, lk_is_signed, lk_is_changed_mannual,
                          lk_percent_orig)
    select :lm_id, aarep_llc_k_id, aarep_llc_k_deliq_days, aarep_llc_region_id, aarep_llc_k_balance, aarep_llc_status,
           aarep_percent, :rep_date, '31.12.9999', '', aarep_llc_loan_type, 1, 0, aarep_percent
    from aarep_kontract_llc(:reg_id, :rep_date_begin, :rep_date, 0, '000');
    suspend;
  end
  suspend;
end
