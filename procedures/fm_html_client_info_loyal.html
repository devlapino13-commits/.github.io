CREATE PROCEDURE FM_HTML_CLIENT_INFO_LOYAL(
    UN_ID D_ID,
    C_ID D_ID,
    DATE_END D_DATE)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable rd_id d_integer;
declare variable ag_id d_integer;
declare variable link_count d_integer;
declare variable edl_status_str d_str_code;
declare variable lang d_str_lang;
declare variable close_date d_date;
declare variable last_k_id id_type;
declare variable show_client_link d_smallint;
begin
  -- Формирует ссылки для Партнерской программы (РФ) для fm_html_client_info

  lang = f_get_lang_code(un_id, null);
  res_text = f_format('<h2>%1</h2>', f_lang(lang, 'Партнерская программа'));

  link_count = 0;
  select count(*)
  from clientlink
  where clk_from_c_id = :c_id
  into link_count;
  if (link_count is null) then
    link_count = 0;

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.loyal_agreement'
  into rd_id;

  select ag_id
  from agreements
  where ag_object_kind = f_objkind('clients', 'id') and
        ag_object_id = :c_id and
        ag_from_rd_id = :rd_id
  order by ag_date_begin desc, ag_status desc
  rows 1
  into ag_id;


  edl_status_str = f_edoc_status('loyal.agreement', ag_id, date_end);

  if (edl_status_str = 'signed') then
  begin
    res_text = BLOB_APPEND(res_text,
      f_format('%1: %2 ', f_lang(lang, 'Друзья клиента'), link_count) ||
      f_uri_open(f_lang(lang, 'Добавить'), 'clientlink', 'client', c_id,
        f_lang(lang, 'Добавить друга'), '', un_id) || '<br>' ||
      f_uri_html(f_lang(lang, 'Детали участника программы'), 'loyal_client_info',
        f_lang(lang, 'Детали участника программы'), 'agreements', :ag_id,
        current_date, current_date) || '<br>');
  end else
  begin
    if (edl_status_str = 'unsigned') then
    begin
    res_text = BLOB_APPEND(res_text,
       f_lang(lang, 'Приглашение в Партнерскую программу клиентов еще не подписано')  || '<br>' ||
       f_uri_sql(f_lang(lang, 'Пригласить повторно'), 'lp_sign_sms', 'client',
        c_id, 'none', f_lang(lang, 'Подтверждаете отправку приглашения через СМС клиенту?'),
        '', 0, 0, '', '', 'dialog', '') || '<br>');
    end else
    begin
      if (edl_status_str <> 'inactive') then
      begin
        -- Условия участия в партнерской программе
        -- Клиент должен быть активным или когда либо бравший кредит
        show_client_link = 1;
        close_date = null;
        select ks_date_end, ks_from_k_id from kontstate, knclient
        where ks_ol_is_delete = 0
        and ks_state = 1
        and ks_from_k_id = kc_from_k_id
        and kc_amount_approved > 0
        and kc_from_c_id = :c_id
        order by ks_date_end desc
        rows 1
        into close_date, last_k_id;
        if (close_date is null) then show_client_link = 0; -- если не имеется кр. история то отменяется
        else
        if (dateadd(month, -6, :date_end) > :close_date) then show_client_link = 0; -- если после закрытия последнего контракта прошло более 6 мес.
        else
        if (exists(select * from moratory
          where m_is_black = 2
          and m_is_active = 1
          and m_from_c_id = :c_id)) then show_client_link = 0; -- если в Стоп-листе
        else
        if (exists(select * from oksummary
            where oks_from_k_id = :last_k_id
            and oks_close_state = 0
            and ((oks_delinq_day > 30 and oks_date_begin <= cast(:date_end as date))
                 or (:date_end between oks_date_begin and oks_date_end
                     and cast(:date_end as date) - oks_delinq_date > 30))
            )) then show_client_link = 0;

        if (show_client_link = 1) then
        begin
          res_text = f_uri_sql(f_lang(lang, 'Пригласить в Партнерскую программу клиентов'),
            'lp_sign_sms', 'client', c_id, 'none',
            f_lang(lang, 'Подтверждаете отправку приглашения через СМС клиенту?'),
            '', 0, 0, '', '', 'dialog', '') || '<br>';
        end else
        begin
          res_text = BLOB_APPEND(res_text,
          f_lang(lang, 'Клиент не соответствует к партнерской программе')|| '<br>');
        end 

      end
    end
  end

  suspend;
end
