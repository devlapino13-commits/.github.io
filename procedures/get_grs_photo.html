CREATE PROCEDURE GET_GRS_PHOTO(
    C_ID D_ID,
    CONTINUE_SEARCH D_BOOL = 0,
    MCL_ID D_ID = null,
    GET_BLOB D_BOOL = 1)
RETURNS (
    BLS_ID D_ID,
    BLOB_PHOTO D_BLOB_BINARY,
    MATCHING_PERC D_CURRENCY)
AS
declare variable pass_date_begin d_date;
declare variable db_id d_integer;
begin
    db_id = f_get_db_id();

    select
      eib_from_bls_id
    from extinfolist
      inner join extinfolistbls
        on eib_from_eil_id = eil_id
      inner join extinfocfg
        on eic_id = eil_from_eic_id
    where (eil_from_c_id = :c_id or eil_from_mcl_id = :mcl_id)
      and eil_status = 2
      and eic_code = 'Tunduk'
    order by eil_create_datetime desc
    rows 1
    into
      :bls_id;

    if (bls_id is null and continue_search = 1) then
      select cds_doc_blob_1
      from clientdocs
      inner join passkind on pk_id = cds_from_pk_id
      inner join passdoctype on pdt_id = pk_from_pdt_id
        and pdt_code = 'foto_on_date_disb'
      where :db_id <> 39801
        and cds_from_c_id = :c_id
        and cds_is_delete = 0
        and cds_doc_blob_1 > 0
        and cds_verify_status = 2
      order by cds_doc_date_begin desc
      rows 1
      into
        :bls_id;

    if (bls_id is null and continue_search = 1) then
      select
        c_blob_photo_id
      from clients
      where c_id = :c_id
      into
        :bls_id;

    select first 1 cds.cds_doc_date_begin
    from clientdocs cds
    where cds.cds_from_c_id = :c_id and
          cds.cds_is_delete = 0 and
          cds.cds_is_main_doc = 1 and
          current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
    order by cds.cds_create_datetime desc
    into pass_date_begin;

    if (db_id = 41701) then
        -- RFCKG-4339: Все проценты снижены на 2 пункта
        -- RFCKG-4661: Все проценты снижены на 3 пункта
        matching_perc = case
                          when datediff(year, pass_date_begin, current_date) < 5 then 93
                          when datediff(year, pass_date_begin, current_date) between 5 and 10 then 90
                          when datediff(year, pass_date_begin, current_date) > 10 then 85
                          else 98
                        end;
    else if (db_id = 39801) then  -- RFCKG-4276
        matching_perc = 0.85;
    else
        matching_perc = case
                          when datediff(year, pass_date_begin, current_date) < 5 then 98
                          when datediff(year, pass_date_begin, current_date) between 5 and 10 then 95
                          when datediff(year, pass_date_begin, current_date) > 10 then 90
                          else 98
                        end;

    ------  Для теста ---------------------- >
    if (f_get_const('grs_fake_photo') = 1) then
        bls_id = (select bls_id from get_fake_grs_photo_id);
    ---------------------------------------- <
    if ((coalesce(:bls_id, 0) > 0) and (get_blob = 1)) then
      blob_photo = (select blx_blob from hp_get_blobex_blob(:bls_id));

    suspend;
end
