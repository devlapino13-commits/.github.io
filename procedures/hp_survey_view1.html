CREATE PROCEDURE HP_SURVEY_VIEW1(
    C_ID D_ID)
RETURNS (
    RD_ID D_INTEGER,
    RD_CODE D_STR_255,
    RD_NAME D_STR_255,
    BLN_ID D_INTEGER,
    IS_MAIN D_BOOL,
    IS_TARG D_BOOL,
    IS_DELQ D_BOOL)
AS
declare variable obj_kind d_integer;
declare variable rd_code1 d_str_255;
declare variable rd_code2 d_str_255;
declare variable rd_code3 d_str_255;
begin
  rd_code1 = 'Anketa.Main';
  rd_code2 = 'Anketa.TargetUseLoan';
  rd_code3 = 'Anketa.Delinq.1';

  obj_kind = f_objkind('Clients', 'id');
  for
    select
      rd_id,
      rd_code,
      rd_name
    from refdata
    where rd_main_code = 'ActionGroup'
      and rd_str_id = 'Anketa'
      and rd_code in (:rd_code1, :rd_code2, :rd_code3)
      and rd_is_active = 1
  into
    :rd_id,
    :rd_code,
    :rd_name
  do
  begin
    if (
      exists(
        select *
        from actionlist
        where acl_object_kind = :obj_kind
          and acl_object_id = :c_id
          and acl_from_rd_id = :rd_id)
      and
      exists(
        select *
        from actiongroup
          inner join objectprops
            on opr_object_kind = :obj_kind
              and opr_object_id = :c_id
              and opr_subject_kind = -1
              and opr_subject_id = acg_from_rd_id
        where acg_group_rd_id = :rd_id
          and acg_parent_id is null
          and acg_is_delete = 0))
    then begin
      is_main = iif(rd_code = rd_code1, 1, 0);
      is_targ = iif(rd_code = rd_code2, 1, 0);
      is_delq = iif(rd_code = rd_code3, 1, 0);
      bln_id =
        case rd_code
          when rd_code1 then 1637
          when rd_code2 then 1638
          when rd_code3 then 1639
        else 0 end;
      suspend;
    end
  end
end
