CREATE PROCEDURE AAREP_ACTIVE_PORTFOLIO_SVZ(
    REP_C_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(60),
    KF_OFFICER_NAME VARCHAR(100),
    KF_NAME VARCHAR(50),
    K_CIRCLE INTEGER,
    K_DISB_DATE DATE,
    K_KREDLINE_END DATE,
    K_PERIOD DECIMAL(3,1),
    OKS_BALANCE CURR_TYPE,
    CATEGORY VARCHAR(20),
    PAR1_BALANCE CURR_TYPE,
    PAR7_BALANCE CURR_TYPE,
    PAR30_BALANCE CURR_TYPE,
    PAR1 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    OKS_BALANCE_LAST CURR_TYPE,
    BALANCE_PRINYAT CURR_TYPE,
    BALANCE_PEREDAN CURR_TYPE,
    C_PLASTCARD_NO VARCHAR(20),
    CASHIER_NAME VARCHAR(100),
    DATE_REPAY DATE,
    ZAD_OS_1 CURR_TYPE,
    ZAD_PERC_1 CURR_TYPE,
    ZAD_TOTAL_1 CURR_TYPE,
    DELINQ_ZAD_OS CURR_TYPE,
    DELINQ_ZAD_PERC CURR_TYPE,
    DELINQ_ZAD_FINE CURR_TYPE,
    DELINQ_ZAD_TOTAL CURR_TYPE,
    IS_CLOSED VARCHAR(1),
    DELINQ_DAY INTEGER,
    IS_DELINQ SMALLINT,
    PR_KIND_STR VARCHAR(3))
AS
declare variable oks_region_id integer;
declare variable oks_kc_id id_type;
declare variable oks_delinq_date date;
declare variable rep_date_begin date;
declare variable oks_from_k_id id_type;
declare variable aa_category_id smallint;
declare variable k_from_kf_id decimal(18,0);
declare variable begin_kf_officer decimal(18,0);
declare variable curr_kf_officer decimal(18,0);
declare variable cashier_id id_type;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable sh_calc_fine curr_type;
declare variable oks_region_id_prior integer;
declare variable oks_kc_id_prior id_type;
declare variable sh_calc_perc curr_type;
declare variable pr_product_kind integer;
declare variable is_suspend smallint;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end);
  oks_region_id_prior = 0;
  oks_kc_id_prior = 0;

  for with tabl1
      as (select oks_region_id, oks_kc_id, oks_delinq_date, oks_balance, oks_from_k_id, null oks_balance_last,
                 k_disb_date, k_circle, k_kredline_end, k_period, k_from_kf_id, k_clients_cashier_id, pr_product_kind
          from oksummary, kontract, product
          where oks_kc_id = :rep_c_id and
                :rep_date_end between oks_date_begin and oks_date_end and
                oks_balance > 0 and
                oks_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_product_kind in (1, 2, 3) and
                oks_close_state = 0
          union
          select oks_region_id, oks_kc_id, null, null, oks_from_k_id, oks_balance, k_disb_date, k_circle,
                 k_kredline_end, k_period, k_from_kf_id, k_clients_cashier_id, pr_product_kind
          from oksummary, kontract, product
          where oks_kc_id = :rep_c_id and
                :rep_date_begin between oks_date_begin and oks_date_end and
                oks_balance > 0 and
                oks_from_k_id = k_id and
                k_from_pr_id = pr_id and
                pr_product_kind in (1, 2, 3) and
                oks_close_state = 0)

      select oks_region_id, oks_kc_id, max(oks_delinq_date), max(oks_balance), oks_from_k_id, max(oks_balance_last),
             k_disb_date, k_circle, k_kredline_end, k_period, k_from_kf_id, k_clients_cashier_id, pr_product_kind

      from tabl1
      group by oks_region_id, oks_kc_id, oks_from_k_id, k_disb_date, k_circle, k_kredline_end, k_period, k_from_kf_id, k_clients_cashier_id, pr_product_kind

      order by oks_region_id, oks_kc_id
      into oks_region_id, oks_kc_id, oks_delinq_date, oks_balance, oks_from_k_id, oks_balance_last, k_disb_date,
           k_circle, k_kredline_end, k_period, k_from_kf_id, cashier_id, pr_product_kind
  do
  begin
    pr_kind_str = case pr_product_kind
                    when 1 then 'Гр.'
                    when 2 then 'Ин.'
                    when 3 then 'Лг.'
                  end;

    if (oks_delinq_date is null) then
      oks_delinq_date = '31.12.9999';
    if (oks_balance is null) then
      oks_balance = 0;
    if (oks_balance_last is null) then
      oks_balance_last = 0;
    if (:rep_date_end > :oks_delinq_date) then
      delinq_day = :rep_date_end - :oks_delinq_date;
    else
      delinq_day = 0;

    aa_category_id = (select aa_category_id
                      from aarep_get_category_id(:oks_from_k_id, :rep_date_end, 810, :oks_kc_id, null, null, null, :delinq_day));
    if (aa_category_id <= 2) then
    begin

      is_delinq = 0;
      if (delinq_day > 0) then
      begin
        par1_balance = :oks_balance;
        is_delinq = 1;
      end
      else
        par1_balance = 0;
      if (delinq_day > 7) then
        par7_balance = :oks_balance;
      else
        par7_balance = 0;
      if (delinq_day > 30) then
        par30_balance = :oks_balance;
      else
        par30_balance = 0;

      if (par1_balance > 0) then
      begin
        if (oks_balance > 0) then
          par1 = par1_balance / oks_balance * 100;
        else
          par1 = 100;
      end
      else
        par1 = 0;

      if (par7_balance > 0) then
      begin
        if (oks_balance > 0) then
          par7 = par7_balance / oks_balance * 100;
        else
          par7 = 100;
      end
      else
        par7 = 0;

      if (par30_balance > 0) then
      begin
        if (oks_balance > 0) then
          par30 = par30_balance / oks_balance * 100;
        else
          par30 = 100;
      end
      else
        par30 = 0;

      category = iif(aa_category_id = 1, '1', '2');

      kf_name = null;
      select kal_new_kf_name, kal_new_kf_officer
      from knaktlist
      where knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal_from_kf_id = :k_from_kf_id
      into kf_name, curr_kf_officer;

      if (kf_name is null) then
        kf_name = '';
      if (oks_kc_id_prior <> oks_kc_id) then
      begin
        oks_kc_id_prior = oks_kc_id;
        kf_officer_name = (select c_name_fio
                           from clients
                           where c_id = :oks_kc_id);
      end

      begin_kf_officer = null;
      select kal_new_kf_officer
      from knaktlist
      where knaktlist.is_delete = 0 and
            iif(:rep_date_begin > :k_disb_date, :rep_date_begin, :k_disb_date) between kal_date and kal_date_end and
            kal_from_kf_id = :k_from_kf_id
      into begin_kf_officer;
      if (begin_kf_officer is null) then
        begin_kf_officer = 0;

      balance_peredan = 0;
      balance_prinyat = 0;

      if (begin_kf_officer <> curr_kf_officer) then
      begin
        if (oks_kc_id = begin_kf_officer) then
        begin
          select oks_balance
          from oksummary
          where oks_from_k_id = :oks_from_k_id and
                :rep_date_end between oks_date_begin and oks_date_end and
                oks_close_state = 0
          into balance_peredan;
          if (balance_peredan is null) then
            balance_peredan = 0;
        end
        else
          balance_prinyat = oks_balance;
      end

      if (oks_region_id_prior <> oks_region_id) then
      begin
        oks_region_id_prior = oks_region_id;
        reg_name = null;
        select reg_code || ' ' || reg_name
        from region
        where reg_unicode = :oks_region_id
        into reg_name;
      end

      select c_name_fio, '+' ||(select cont_value
                                from hp_contdata_get_cell_main(0, :cashier_id))
      from clients
      where c_id = :cashier_id
      into cashier_name, c_plastcard_no;

      delinq_zad_os = 0;
      delinq_zad_perc = 0;
      delinq_zad_fine = 0;
      delinq_zad_total = 0;
      zad_os_1 = 0;
      zad_perc_1 = 0;
      zad_total_1 = 0;
      date_repay = null;
      is_closed = '';
      is_suspend = 0;
      for select iif(sh_date < :rep_date_end, null, sh_date) as sh_date, sum(zad_os), sum(zad_perc), sum(sh_calc_fine),
                 sum(sh_calc_perc)
          from aarep_make_zad_list(:oks_from_k_id, :rep_date_end, 1)
          where sh_date is not null and
                sh_date <= :rep_date_end + 10
          group by sh_date
          into date_repay, zad_os, zad_perc, sh_calc_fine, sh_calc_perc
      do
      begin
        if (zad_perc < sh_calc_perc) then
          zad_perc = sh_calc_perc;
        if (date_repay is null) then
        begin
          delinq_zad_os = zad_os;
          delinq_zad_perc = zad_perc;
          delinq_zad_fine = sh_calc_fine;
          delinq_zad_total = zad_os + zad_perc + sh_calc_fine;
        end
        else
        begin
          zad_os_1 = zad_os;
          zad_perc_1 = zad_perc;
          zad_total_1 = zad_os + zad_perc;
        end
        if (date_repay = k_kredline_end) then
          is_closed = 'a';
        else
          is_closed = '';

        if (date_repay is not null) then
        begin
          suspend;
          is_suspend = 1;

          delinq_zad_os = 0;
          delinq_zad_perc = 0;
          delinq_zad_fine = 0;
          delinq_zad_total = 0;
        end
      end

      if (is_suspend = 0) then
        suspend;
    end
  end
end
