CREATE PROCEDURE HP_ACTIONS_CREATE_SOLVENCY(
    KBM_ID D_ID)
AS
declare variable ma_id d_integer;
declare variable c_id d_id;
declare variable un_id d_id;
declare variable lo_id d_id;
declare variable reg_id d_integer;
declare variable disb_date d_date;
declare variable obj_kind d_integer;
declare variable obj_id d_integer;
declare variable rd_id d_integer;
declare variable cr_date d_date;
declare variable datex d_date;
begin
  exit;

  rd_id = f_get_rd_id('Anketa.Solvency');
  if (rd_id is null) then
    exit;

  cr_date = current_date;
  dateX = '25.03.2022';
  obj_kind = f_objkind('MiniAppClient', 'id');
  un_id = 1000000000001;

  select
    ma_id,
    kbm_from_c_id,
    ma_date_oper,
    ma_office_id,
    ma_expert_id
  from kibminiapp
    inner join miniapp
      on ma_id = kbm_ma_id
        and ma_date_end = '31.12.9999'
        and ma_is_active = 1
  where kbm_id = :kbm_id
    and kbm_is_checked = 1
  into
    :ma_id,
    :c_id,
    :disb_date,
    :reg_id,
    :lo_id;

  if (ma_id > 0 and disb_date >= dateX) then
    insert into actionlist (
      acl_date,
      acl_from_reg_id,
      acl_object_kind,
      acl_object_id,
      acl_from_rd_id,
      acl_status,
      acl_respons_c_id,
      acl_from_un_id)
    select
      :cr_date,
      :reg_id,
      :obj_kind,
      mac_id,
      :rd_id,
      0,
      :lo_id,
      :un_id
    from miniappclient
    where mac_from_ma_id = :ma_id
      and mac_from_c_id = :c_id
      and mac_amount > 0;
end
