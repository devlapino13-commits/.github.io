CREATE PROCEDURE HP_AFM_GET_STATUS(
    CATEGORY D_STR_CODE,
    C_ID D_ID,
    MA_ID D_ID)
RETURNS (
    STATUS D_SMALLINT,
    RES_TEXT D_STR_255,
    ANSWER_DATE D_DATE)
AS
declare variable ERR_TYPE D_STR_100;
begin
    -- RFCKZ-91
    status = f_enum('afm_status', 'not sent yet', 'id');
    if (:category not in ('afm', 'pdl', 'high_risk')) then
        exception e_default 'Unknown category code';

    -- RFCKZ-386
    select first 1
        case
            when eil.eil_status = 2 and eil.eil_from_bls_id is not null then null
            when eil.eil_status in (0, 1) then 'sent'
            when eil.eil_status = -2 and eir.eir_answer containing 'Daily limit reached' then 'daily_limit_reached'
            when eil.eil_status = -2 and eir.eir_answer containing 'Subscription expired' then 'subscription_expired'
            else 'afm_error'
        end
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
    left join extinfoerr eir on eir.eir_from_eil_id = eil.eil_id
    where eil.eil_from_c_id = :c_id
        and eil.eil_object_kind = f_objkind('miniapp', 'id')
        and eil.eil_object_id = :ma_id
        and eic.eic_code in ('afm_iin_check', 'afm_name_check')
        and eil.eil_status in (0, 1, -2, 2)
    order by eil.eil_status_datetime desc
    into :err_type;

    if (err_type is not null) then
    begin
        status = f_enum('afm_status', :err_type, 'id');
        res_text = f_enum('afm_status', :status, 'name');
        suspend;
        exit;
    end

    select
        case :category
            when 'afm' then
                coalesce
                (
                    nullif(ac.ac_category_1, 1),
                    nullif(ac.ac_category_4, 1),
                    nullif(ac.ac_category_5, 1),
                    nullif(ac.ac_category_6, 1),
                    nullif(ac.ac_category_7, 1),
                    nullif(ac.ac_category_8, 1),
                    nullif(ac.ac_category_9, 1),
                    nullif(ac.ac_category_10, 1),
                    nullif(ac.ac_category_11, 1),
                    nullif(ac.ac_category_12, 1),
                    nullif(ac.ac_category_14, 1),
                    nullif(ac.ac_category_15, 1),
                    1
                )
            when 'pdl' then ac.ac_category_13
        end,
        ac.ac_date
    from afm_check ac
    where ac.ac_from_c_id = :c_id
        and ac.ac_from_ma_id = :ma_id
        and ac.ac_is_actual = 1
    into
        :status,
        :answer_date;


    :res_text = f_enum('afm_status', :status, 'name');
    suspend;
end
