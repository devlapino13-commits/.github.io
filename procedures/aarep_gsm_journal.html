    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_DEL_STATUS D_INTEGER = 1)
RETURNS (
    OL_ID DECIMAL(18,0),
    OL_FROM_REG_ID INTEGER,
    OL_OPER_TYPE INTEGER,
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_DOC_NO_STR VARCHAR(20),
    OL_DOC_KIND INTEGER,
    OL_MEMO VARCHAR(1024),
    OL_STATUS SMALLINT,
    IS_DELETE SMALLINT,
    OL_DATETIME TIMESTAMP,
    OL_USER_ID DECIMAL(18,0),
    OL_DOCUMENT_CLIENT DECIMAL(18,0),
    OL_FROM_MO_ID DECIMAL(18,0),
    OL_IS_MO_MODIFIED SMALLINT,
    OL_DOCUMENT_OFFICE INTEGER,
    OL_ACC_NO_CLIENT INTEGER,
    OL_ACC_NO_OFFICE INTEGER,
    OL_OPER_SUBTYPE INTEGER,
    OL_DATE_OPER_DAY DATE,
    OL_POCKET INTEGER,
    OL_DOC_CLIENT_NAME VARCHAR(255),
    OL_IS_CLIENT SMALLINT,
    OL_YEAR INTEGER,
    OL_MONTH INTEGER,
    RES_OPER_NAME VARCHAR(100),
    RES_MATOTVET VARCHAR(100),
    RES_COST DECIMAL(18,4),
    RES_PEN_NAME VARCHAR(100),
    RES_KOLICH DECIMAL(18,4),
    RES_LITER DECIMAL(18,4),
    RES_SUMMA DECIMAL(18,4),
    RES_TRANS_NAME VARCHAR(255),
    RES_TRANS_LABEL VARCHAR(50),
    RES_TRANS_LITER DECIMAL(18,4),
    RES_TRANS_SUMMA DECIMAL(18,4),
    RES_PROBEG_LITER DECIMAL(18,4),
    RES_PROBEG_KM DECIMAL(18,4),
    RES_OFFICE VARCHAR(100),
    RES_OFFICE_BALANCE VARCHAR(100),
    RES_TALON_LITER DECIMAL(18,4),
    OL_USERNAME VARCHAR(30),
    OL_PARENT_ID DECIMAL(18,0))
AS
declare variable peo_oper_type integer;
declare variable peo_from_pet_id integer;
declare variable peo_from_pen_id integer;
declare variable peo_from_c_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select distinct ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                      ol_status, operlist.is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id,
                      ol_is_mo_modified, ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype,
                      ol_date_oper_day, ol_pocket, ol_doc_client_name, ol_is_client, ol_year, ol_month, ol_parent_id,
                      peo_oper_type, c_name_fio, peo_cost, peo_kolich, peo_liter, peo_summa, peo_trans_liter,
                      peo_trans_summa, peo_probeg_fact_liter, peo_probeg_fact_km, r1.reg_code || ' ' || r1.reg_name,
                      r3.reg_name, peo_talon_liter, un_name, peo_from_pet_id, peo_from_pen_id
      from operlist, region_list(:report_region_id, 1, -1, 1), username, region r3, clients, region r1, petopers
      where (peo_from_reg_id = id_list) and
            r3.reg_unicode = ol_from_reg_id and
            operlist.is_delete = iif(:report_del_status = -1, operlist.is_delete, 1 - :report_del_status) and
            --operlist.is_delete = 0 and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            ol_user_id = un_id and
            peo_from_ol_id = ol_id and
            peo_from_c_id = c_id and
            r1.reg_unicode = peo_from_reg_id
      order by ol_date_oper_day, ol_from_reg_id, ol_id
      into :ol_id, :ol_from_reg_id, :ol_oper_type, :ol_date, :ol_doc_no, :ol_doc_no_str, :ol_doc_kind, :ol_memo,
           :ol_status, :is_delete, :ol_datetime, :ol_user_id, :ol_document_client, :ol_from_mo_id, :ol_is_mo_modified,
           :ol_document_office, :ol_acc_no_client, :ol_acc_no_office, :ol_oper_subtype, :ol_date_oper_day, :ol_pocket,
           :ol_doc_client_name, :ol_is_client, :ol_year, :ol_month, :ol_parent_id, :peo_oper_type, :res_matotvet,
           :res_cost, :res_kolich, :res_liter, :res_summa, :res_trans_liter, :res_trans_summa, :res_probeg_liter,
           :res_probeg_km, :res_office, :res_office_balance, :res_talon_liter, :ol_username, :peo_from_pet_id,
           :peo_from_pen_id
  do
  begin
    res_trans_name = null;
    res_trans_label = null;
    if (peo_from_pet_id is not null) then
    begin
      select pet_name, pet_label
      from pettrans
      where pet_id = :peo_from_pet_id
      into :res_trans_name, :res_trans_label;
    end

    select pen_name
    from petname
    where pen_id = :peo_from_pen_id
    into :res_pen_name;

    if (peo_oper_type = 0) then
      res_oper_name = 'Первич.остатки';
    if (peo_oper_type = 1) then
      res_oper_name = 'Приход на склад';
    if (peo_oper_type = 2) then
      res_oper_name = 'Залито через чек';
    if (peo_oper_type = 3) then
      res_oper_name = 'Передан в другой офис';
    if (peo_oper_type = 4) then
      res_oper_name = 'Принят из другого офиса';
    if (peo_oper_type = 5) then
      res_oper_name = 'Залито в бак со склада';
    if (peo_oper_type = 6) then
      res_oper_name = 'Возвращен из бака';
    if (peo_oper_type = 7) then
      res_oper_name = 'Списан из бака';
    if (peo_oper_type = 8) then
      res_oper_name = 'Заменен.мат.ответ';
    if (peo_oper_type = 9) then
      res_oper_name = 'Передача между объектами';
    suspend;
  end
end
