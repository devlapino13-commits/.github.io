CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_EXTINFO(
    EXA_ID D_INTEGER)
RETURNS (
    WAIT_STATUS D_INTEGER)
AS
declare variable ma_id d_integer;
declare variable k_id d_id;
declare variable app_date d_date;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable eic_id d_integer;
declare variable fssp_id d_integer;
begin
  -- Проверка статусов всех проверок по внешним источника
  select eic_id
  from extinfocfg
  where eic_code = 'fssp'
  into fssp_id;

  if (fssp_id is null) then
    fssp_id = 0;

  wait_status = 0;

  select exa_from_ma_id, exa_from_k_id
  from expressapp
  where exa_id = :exa_id
  into ma_id, k_id;

  select min(ma_date_app)
  from miniapp
  where ma_id = :ma_id
  into app_date;

  for select f_objkind('miniapp', 'id'), :ma_id
  from rdb$database
  union
  select f_objkind('kontract', 'id'), :k_id
  from rdb$database
  union
  select f_objkind('clients', 'id'), kc_from_c_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  union
  select f_objkind('clients', 'id'), mac_from_c_id
  from miniappclient
  where mac_from_ma_id = :ma_id and
        mac_amount > 0
  into obj_kind, obj_id
  do
  begin
    if (obj_id is null) then
      wait_status = -1;

    if (wait_status <> 0) then
      break;

    -- Получаем все конфигурации, по которым создавался запрос
    for select distinct eil_from_eic_id
    from extinfolist
    where eil_date >= :app_date and
          eil_object_kind = :obj_kind and
          eil_object_id = :obj_id and
          eil_from_eic_id <> :fssp_id
    into eic_id
    do
    begin
      -- если нет заверешенного запроса по данной конфигурации, возвращаем режим ожидания
      if (not exists(select * from extinfolist
      where eil_date >= :app_date and
            eil_object_kind = :obj_kind and
            eil_object_id = :obj_id and
            eil_from_eic_id = :eic_id and
            eil_status = 2)) then
      begin
        wait_status = 2; -- до этого было 1
        break;
      end
    end
  end

  suspend;
end
