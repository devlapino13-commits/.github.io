CREATE PROCEDURE AAREP_CALC_PERC_ONE_TEMP(
    KONT_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE,
    INCLUDE_LAST_DAY_PERC INTEGER,
    K_STOP_PERC DATE,
    DB_ID INTEGER)
RETURNS (
    RESULT_CALCED DECIMAL(18,4),
    RESULT_PAYED DECIMAL(18,4),
    RESULT_WRITEOFF DECIMAL(18,4),
    RESULT_PERC_END DECIMAL(18,4),
    RESULT_BALANCE_END DECIMAL(18,4),
    RESULT_CALCED_ALL DECIMAL(18,4),
    RESULT_PAYED_ALL DECIMAL(18,4),
    RESULT_WRITEOFF_ALL DECIMAL(18,4),
    RESULT_DISB_AMOUNT DECIMAL(18,4),
    RESULT_DISB_DATE DATE,
    RESULT_OUT_PERC_END DECIMAL(18,4),
    RESULT_OUT_CALCED DECIMAL(18,4),
    DD_CHANGE_BEGIN DATE,
    DD_CHANGE_END DATE,
    DD_CHANGE_DAYS INTEGER)
AS
declare variable date_end_orig date;
declare variable payed_perc decimal(18,4);
declare variable balance_new decimal(18,4);
declare variable ok_t integer;
declare variable dd2 date;
declare variable pr_is_360 integer;
declare variable total_balance decimal(18,4);
declare variable perc_a decimal(18,4);
declare variable dd_orig date;
declare variable ks_date_end date;
declare variable max_perc decimal(18,4);
declare variable disb_amount decimal(18,4);
declare variable perc_per_day decimal(18,7);
declare variable calc_method integer;
declare variable date_end_final date;
declare variable kol_days integer;
declare variable dd_prior date;
declare variable dd date;
declare variable round_perc integer;
declare variable rep_date_0 date;
declare variable dd_tmp date;
declare variable sort_no integer;
declare variable perc_per_day_last decimal(18,7);
declare variable loan_plan_close_date date;
declare variable loan_perc_base decimal(18,4) = 0;
declare variable last_day_balance decimal(18,4) = 0;
declare variable dd_out date;
declare variable dd_prior_out date;
declare variable kol_days_out integer;
declare variable perc_a_out decimal(18,4) = 0;
declare variable pr_percent_kind d_smallint;
declare variable sh_perc_all d_currency;
declare variable max_diff d_currency;
declare variable res_psk_limit d_currency;
declare variable max_perc_amount d_currency;
declare variable payed_fine d_currency;
begin
  if (coalesce(db_id, 0) = 0) then
    db_id = f_get_db_id();

  max_diff = 5;
  if (db_id = 94401) then
    max_diff = 1;

  -- Include_last_day_perc равно 0 при приеме погашений, равно 1 при начислении процентов
  rep_date_0 = cast('01.01.1990' as date);
  max_perc = 0;
  round_perc = 0;
  dd_prior = null;
  dd_prior_out = null;
  kol_days = 0;
  kol_days_out = 0;
  result_calced = 0;
  result_out_calced = 0;
  result_perc_end = 0;
  result_out_perc_end = 0;
  total_balance = 0;
  result_payed = 0;
  result_writeoff = 0;
  result_balance_end = 0;
  result_payed_all = 0;
  result_writeoff_all = 0;
  result_calced_all = 0;
  date_end_orig = date_end;
  result_disb_amount = 0;
  result_disb_date = null;
  dd_change_days = 0;
  sh_perc_all = 0;

  if (date_begin > date_end) then
  begin
    suspend;
    exit;
  end

  select ks_date_begin, ks_date_end, k_amount_approved, pr_is_360, pr_round_perc, k_percent_per_day,
         pr_interest_method, k_kredline_end, pr_percent_kind
  from kontstate, kontract, product
  where ks_from_k_id = :kont_id and
        ks_date_begin between :rep_date_0 and :date_end_orig and
        ks_ol_is_delete = 0 and
        ks_state = 1 and
        k_id = :kont_id and
        k_from_pr_id = pr_id
  into :result_disb_date, :ks_date_end, :result_disb_amount, :pr_is_360, :round_perc, :perc_per_day,
       calc_method, loan_plan_close_date, pr_percent_kind;

  if (pr_percent_kind = 2 and f_tmp_const('ipf_percent_mode_active') = 1) then
  begin
    -- Новый вариант начисления процентов по ИФП продуктам
    -- Вместо использования ставки процента применяется метод амортизации процента
    select result_calced, result_payed, result_writeoff, result_perc_end, result_balance_end,
           result_calced_all, result_payed_all, result_writeoff_all, result_disb_amount, result_disb_date,
           result_out_perc_end, result_out_calced
    from aarep_calc_perc_one_ipf(:kont_id, :date_begin, :date_end)
    into :result_calced, :result_payed, :result_writeoff, :result_perc_end, :result_balance_end,
         :result_calced_all, :result_payed_all, :result_writeoff_all, :result_disb_amount, :result_disb_date,
         :result_out_perc_end, :result_out_calced;

    suspend;
    exit;
  end

  if (date_end_orig >= '01.04.2015') then -- С апреля 2015 года лишние проценты из-за переноса даты не начисляем
  begin
    -- Проверям, нет ли переноса изза праздников
    dd_change_begin = null;
    dd_change_end = null;
    select min(ok_sh_date), max(ok_sh_date)
    from operkred, operlist
    where ok_from_k_id = :kont_id and
          ok_ol_is_active = 1 and
          ok_type = 2 and
          ok_from_ol_id = ol_id and
          ol_oper_type = 1 and
          ol_oper_subtype = 9 and -- Перенос даты погашения
          ok_sh_os <> 0
    group by ok_sh_id
    having max(ok_sh_date) >= :date_end_orig and min(ok_sh_date) <= :date_end_orig
    order by ok_sh_id
    rows 1
    into :dd_change_begin, :dd_change_end;
    if ((dd_change_begin is not null) and
        (dd_change_end is not null)) then
      dd_change_days = date_end - dd_change_begin;
  end

  disb_amount = result_disb_amount;
  if (k_stop_perc > ks_date_end) then
    k_stop_perc = ks_date_end;

  if (include_last_day_perc = 1) then
    date_end = :date_end + :include_last_day_perc;

  if (date_end < k_stop_perc) then
    k_stop_perc = null;

  if (calc_method <= 0) then
  begin
    kol_days = f_agein360(:result_disb_date, :k_stop_perc, :pr_is_360);
    max_perc = disb_amount * perc_per_day / 100 * kol_days;
  end

  perc_per_day_last = perc_per_day;
  -- Переход между двумя новым подходом и старым. 28.07.2013 г.
  if (date_end_orig < '01.07.2013') then
    date_end_final = date_end;
  else
    date_end_final = date_end_orig;

  -- Изменение с 24.03.2014 г.
  if ((loan_plan_close_date <= date_end) and
      (db_id = 81001) and
      (result_disb_date >= '01.10.2013')) then
  begin
    -- Если РФ и выдачи после 1 октября 2013 года и срок договора истек,
    -- то дальше будем начислять проценты на остаток на последнюю дату по графику
    select oks_balance
    from oksummary
    where oks_from_k_id = :kont_id and
          oks_date_begin <= :loan_plan_close_date and
          oks_date_end >= :loan_plan_close_date and
          oks_close_state = 0
    into :last_day_balance;
    if (last_day_balance is null) then
      last_day_balance = 0;
  end
  else
    last_day_balance = disb_amount;
  -- Конец изменения с 24.03.2014 г.

  for select pdd, pok_t, psh_dd, 0, sum(pbalance), sum(ppayed_perc), pdd_orig, kd_percent_per_day
      from aarep_calc_perc_one_temp2(:kont_id, :date_end_final), kontdata
      where kd_from_k_id = :kont_id and
            kd_ol_date <= pdd and
            kd_date_end >= pdd and
            kd_ol_is_delete = 0
      group by pdd, pok_t, psh_dd, pdd_orig, kd_percent_per_day
      union
      select kd_ol_date, 1, kd_ol_date, 1, 0, 0, kd_ol_date, kd_percent_per_day
      from kontdata
      where kd_from_k_id = :kont_id and
            kd_is_primary = 0 and
            kd_ol_is_delete = 0 and
            kd_ol_date <= :date_end
      order by 1, 3, 2, 4
      into :dd, :ok_t, :dd_tmp, :sort_no, :balance_new, :payed_perc, :dd_orig, :perc_per_day_last
  do
  begin
    dd_out = dd;

    -- Изменение с 24.03.2014 г.
    if (db_id = 81001) then
    begin
      if (loan_plan_close_date < dd) then
        loan_perc_base = last_day_balance;
      else
        loan_perc_base = disb_amount;
    end
    else
      loan_perc_base = disb_amount;
    -- Конец изменения с 24.03.2014 г.

    if (balance_new is null) then
      balance_new = 0;
    if (payed_perc is null) then
      payed_perc = 0;
    dd2 = dd_orig;
    if ((dd_orig >= date_begin) and
        (dd_orig <= date_end_orig)) then
    begin
      if (ok_t = 4) then
        result_payed = result_payed - payed_perc;

      if (ok_t in (5, 6)) then
        result_writeoff = result_writeoff - payed_perc;
    end
    if (dd_orig <= date_end_orig) then
    begin
      if (ok_t = 4) then
        result_payed_all = result_payed_all - payed_perc;

      if (ok_t in (5, 6)) then
        result_writeoff_all = result_writeoff_all - payed_perc;
    end
    if ((dd_orig <= date_end_orig) and
        (ok_t <> 0)) then
      result_balance_end = result_balance_end + balance_new;

    if (dd_prior is null) then
    begin
      kol_days = 0;
      dd_prior = dd;
    end
    if (dd_prior_out is null) then
    begin
      kol_days_out = 0;
      dd_prior_out = dd_out;
    end
    if (dd_orig <= date_end_orig) then
    begin
      if (((date_end > '01.01.2013') and
          (db_id = 81001)) or
          ((date_end > '01.06.2013') and
          (db_id in (39801, 41701))) or
          ((date_end > '01.06.2013') and
          (db_id = 94401))) then
      begin
        if (ok_t <> 7) then
        begin
          result_perc_end = result_perc_end + payed_perc;
          result_out_perc_end = result_out_perc_end + payed_perc;
        end
        else
        begin
          if (date_end >= '17.08.2013') then
          begin
            if ((payed_perc < 0) and
                ((result_disb_date = dd_orig) or
                (db_id = 81001))) then -- 18.08.2013
            begin
              result_perc_end = result_perc_end + payed_perc;
              result_out_perc_end = result_out_perc_end + payed_perc;
            end
            else
            begin
              if ((payed_perc > 0) and
                  (date_end >= '01.09.2015') and
                  (dd >= '01.09.2015')) then -- 26/09/2015
              begin
                result_perc_end = result_perc_end + payed_perc;
                result_out_perc_end = result_out_perc_end + payed_perc;
              end
            end
          end
          else
          begin
            if (payed_perc < 0) then
            begin
              result_perc_end = result_perc_end + payed_perc;
              result_out_perc_end = result_out_perc_end + payed_perc;
            end
          end
        end
      end
      else
      begin
        result_perc_end = result_perc_end + payed_perc;
        result_out_perc_end = result_out_perc_end + payed_perc;
      end
    end
    if ((total_balance <> 0) or
        (date_end > '01.01.2013')) then
    begin
      if (date_end_orig <= '01.08.2013') then
      begin
        if ((k_stop_perc is null) or
            (k_stop_perc >= dd)) then
          kol_days = f_agein360(:dd_prior, :dd, -:pr_is_360);
        else
        begin
          kol_days = f_agein360(:dd_prior, :k_stop_perc, -:pr_is_360);

          dd = k_stop_perc;
        end
        kol_days_out = f_agein360(:dd_prior_out, :dd_out, -:pr_is_360);
      end
      else
      begin
        if (date_end_orig > '31.03.2018') then
        begin
          if (dd > date_end_orig) then
            dd = date_end_orig;
          if (dd_out > date_end_orig) then
            dd_out = date_end_orig;
        end

        if ((k_stop_perc is null) or
            (k_stop_perc >= dd)) then
        begin
          -- Если дата приема погашений попадает на новую перенесенную дату, то уменьшаем дни
          if (date_end_orig <= '31.03.2020') then
          begin
            if ((dd > dd_change_begin) and
                (dd_change_days <> 0)) then
            begin
              dd = dd - dd_change_days;
            end
          end
          kol_days = f_agein360(:dd_prior, :dd, :pr_is_360);
        end
        else
        begin
          dd = k_stop_perc;
          if (date_end_orig <= '31.03.2020') then
          begin
            if ((dd > dd_change_begin) and
                (dd_change_days <> 0)) then
            begin
              dd = dd - dd_change_days;
              k_stop_perc = k_stop_perc - dd_change_days;
            end
          end
          kol_days = f_agein360(:dd_prior, :dd, :pr_is_360);
        end
        kol_days_out = f_agein360(:dd_prior_out, :dd_out, :pr_is_360);
      end
      if (kol_days < 0) then
        kol_days = 0;
      if (kol_days_out < 0) then
        kol_days_out = 0;
      if (kol_days > 0) then
      begin
        if (calc_method > 0) then
          perc_a = total_balance * perc_per_day / 100 * kol_days;
        else
          perc_a = loan_perc_base * perc_per_day / 100 * kol_days;

        result_perc_end = result_perc_end + perc_a;
      end

      if (kol_days_out > 0) then
      begin
        if (calc_method > 0) then
          perc_a_out = total_balance * perc_per_day / 100 * kol_days_out;
        else
          perc_a_out = loan_perc_base * perc_per_day / 100 * kol_days_out;

        result_out_perc_end = result_out_perc_end + perc_a_out;
      end

    end
    if (dd2 <= date_end_orig) then
    begin
      total_balance = total_balance + balance_new;

      -- исправление алгоритма начиная с 01.04.2018.
      -- До этой даты все должно работать по старому
      if (date_end <= '31.03.2018' or date_end >= '01.05.2020') then
        if (total_balance < 0) then
          total_balance = 0;
    end
    dd_prior = dd;
    dd_prior_out = dd_out;

    perc_per_day = perc_per_day_last;
  end
  -- 06.05.2013
  if (date_end > '01.01.2013') then
  begin
    -- Доначислять если 2013 год и не наступил срок приостановления
    if (k_stop_perc >= date_end) then
    begin
      if (k_stop_perc - dd_change_days = dd) then
        k_stop_perc = k_stop_perc - dd_change_days;
    end
    kol_days = 0;
    kol_days_out = 0;
    if (k_stop_perc > dd) then
    begin
      if (date_end_orig <= '01.08.2013') then
      begin
        if (k_stop_perc <= :date_end_final) then
        begin
          if (db_id = 81001) then
          begin
            if (loan_plan_close_date < k_stop_perc) then
              loan_perc_base = last_day_balance;
            else
              loan_perc_base = disb_amount;
          end
          else
            loan_perc_base = disb_amount;
            kol_days = f_agein360(:dd_prior, :k_stop_perc, -:pr_is_360);
        end
        else
        begin
          if (date_end_final > dd) then
          begin
            if (db_id = 81001) then
            begin
              if (loan_plan_close_date < date_end) then
                loan_perc_base = last_day_balance;
              else
                loan_perc_base = disb_amount;
            end
            else
              loan_perc_base = disb_amount;

            kol_days = f_agein360(:dd_prior, :date_end, -:pr_is_360);
          end
        end
      end
      else
      begin
        if (k_stop_perc <= :date_end_final) then
        begin
          if (db_id = 81001) then
          begin
            if (loan_plan_close_date < k_stop_perc) then
              loan_perc_base = last_day_balance;
            else
              loan_perc_base = disb_amount;
          end
          else
            loan_perc_base = disb_amount;

          -- Если дата приема погашений попадает на новую перенесенную дату, то уменьшаем дни
          if ((dd > dd_change_begin) and
              (dd_change_days <> 0))
          then kol_days = f_agein360(:dd_prior, :k_stop_perc - dd_change_days, :pr_is_360);
          else kol_days = f_agein360(:dd_prior, :k_stop_perc, :pr_is_360);
        end
        else
        begin
          if (date_end_final > dd) then
          begin
            if (db_id = 81001) then
            begin
              if (loan_plan_close_date < date_end) then
                loan_perc_base = last_day_balance;
              else
                loan_perc_base = disb_amount;
            end
            else
              loan_perc_base = disb_amount;

            -- Если дата приема погашений попадает на новую перенесенную дату, то уменьшаем дни
            if ((dd > dd_change_begin) and
                (dd_change_days <> 0))
            then kol_days = f_agein360(:dd_prior, :date_end - :dd_change_days, :pr_is_360);
            else kol_days = f_agein360(:dd_prior, :date_end, :pr_is_360);
          end
        end
      end
    end

    if (date_end_orig <= '01.08.2013') then
    begin
      if (date_end_final > dd_out) then
      begin
        if (db_id = 81001) then
        begin
          if (loan_plan_close_date < date_end) then
            loan_perc_base = last_day_balance;
          else
            loan_perc_base = disb_amount;
        end
        else
          loan_perc_base = disb_amount;

        kol_days_out = f_agein360(:dd_prior_out, :date_end, -:pr_is_360);
      end
    end
    else
    begin
      if (date_end_final > dd_out) then
      begin
        if (db_id = 81001) then
        begin
          if (loan_plan_close_date < date_end) then
            loan_perc_base = last_day_balance;
          else
            loan_perc_base = disb_amount;
        end
        else
          loan_perc_base = disb_amount;
        if ((dd > dd_change_begin) and
            (dd_change_days <> 0)) then
        begin
          if (coalesce(k_stop_perc, '31.12.9999') >= (:date_end - dd_change_days))
          then kol_days_out = f_agein360(:dd_prior_out, :date_end - dd_change_days, :pr_is_360);
          else kol_days_out = f_agein360(:dd_prior_out, :k_stop_perc, :pr_is_360);
        end
        else
        begin
          if (coalesce(k_stop_perc, '31.12.9999') >= :date_end)
          then kol_days_out = f_agein360(:dd_prior_out, :date_end, :pr_is_360);
          else kol_days_out = f_agein360(:dd_prior_out, :k_stop_perc, :pr_is_360);
        end
      end
    end

    if (kol_days < 0) then
      kol_days = 0;
    if (kol_days > 0) then
    begin
      if (calc_method > 0) then
        perc_a = total_balance * perc_per_day / 100 * kol_days;
      else
        perc_a = loan_perc_base * perc_per_day / 100 * kol_days;

      result_perc_end = result_perc_end + perc_a;
    end

    if (kol_days_out < 0) then
      kol_days_out = 0;
    if (kol_days_out > 0) then
    begin
      if (calc_method > 0) then
        perc_a_out = total_balance * perc_per_day / 100 * kol_days_out;
      else
        perc_a_out = loan_perc_base * perc_per_day / 100 * kol_days_out;

      result_out_perc_end = result_out_perc_end + perc_a_out;
    end
  end

  if ((calc_method <= 0) and
      (max_perc > 0) and
      (date_end > '01.01.2013')) then
  begin
    if (result_perc_end > max_perc) then
      result_perc_end = :max_perc;
  end
  if (result_perc_end < 0) then
    result_perc_end = 0;

  if (result_out_perc_end < 0) then
    result_out_perc_end = 0;

  if ((date_end <= '01.01.2013')) then
  begin
    -- Обнулять проценты как раньше если это не 2013
    if ((result_perc_end > 0) and
        (result_balance_end <= 0)) then
      result_perc_end = 0;

    if ((result_out_perc_end > 0) and
        (result_balance_end <= 0)) then
      result_out_perc_end = 0;
  end
  if ((result_perc_end > 0) and
      (date_begin > ks_date_end)) then
    result_perc_end = 0;

  if ((result_perc_end > 0) and
      (date_end_orig > ks_date_end)) then
    result_perc_end = 0;

  if ((result_out_perc_end > 0) and
      (date_begin > ks_date_end)) then
    result_out_perc_end = 0;
  if ((result_out_perc_end > 0) and
      (date_end_orig > ks_date_end)) then
    result_out_perc_end = 0;

  if (round_perc = 0) then
  begin
    result_perc_end = cast(result_perc_end as decimal(18,0));
    result_out_perc_end = cast(result_out_perc_end as decimal(18,0));
  end
  else
  begin
    result_perc_end = cast(result_perc_end as decimal(18,2));
    result_out_perc_end = cast(result_out_perc_end as decimal(18,2));
  end

  -- если на дату отчета имеется платеж и разница меньше допускаемого, то накопленные подгоняем до суммы в графике
  sh_perc_all = 0;

  select oks_delinq_perc
  from oksummary
  where oks_from_k_id = :kont_id and
        oks_date_begin <= :date_end and
        oks_date_end >= :date_end
  into sh_perc_all;
  if (sh_perc_all is null) then
    sh_perc_all = 0;

  if (abs(sh_perc_all - result_perc_end) <= max_diff) then
    result_perc_end = sh_perc_all;

  select res_psk_limit
  from hp_calcperc_psk_limit(:date_end)
  into :res_psk_limit;

  if (res_psk_limit > 0) then
  begin
    payed_fine = 0;

    select sum(ok_fine)
    from operkred
    where ok_from_k_id = :kont_id and
          ok_type = 4 and
          ok_fine <> 0 and
          ok_ol_date <= :date_end and
          ok_ol_is_active = 1
    into payed_fine;

    if (payed_fine is null) then
      payed_fine = 0;

    max_perc_amount = res_psk_limit * result_disb_amount;
    if (result_perc_end + result_payed_all + result_writeoff_all - payed_fine > max_perc_amount) then
    begin
      result_perc_end = max_perc_amount - (result_payed_all + result_writeoff_all - payed_fine);
      if (result_perc_end < 0) then
        result_perc_end = 0;

      result_out_perc_end = result_perc_end;
    end
  end

  result_out_perc_end = result_out_perc_end - result_perc_end;
  result_calced = result_perc_end + (result_payed + result_writeoff);
  result_out_calced = (result_perc_end + result_out_perc_end) + (result_payed + result_writeoff) - result_calced;

  result_calced_all = result_perc_end + result_payed_all + result_writeoff_all;
  if (result_calced is null) then
    result_calced = 0;
  if (result_out_calced is null) then
    result_out_calced = 0;

  suspend;
end
