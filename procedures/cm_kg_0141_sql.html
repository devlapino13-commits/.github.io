CREATE PROCEDURE CM_KG_0141_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kc_circle d_integer;
declare variable max_amount d_integer;
declare variable max_period d_integer;
declare variable k_period d_integer;
declare variable c_id_approve d_id;
begin
  /* Блок по суммам и срокам для всех индивидуальных займов
     проверка только для изменяющихся величин. Н-р сумма в зависимости от цикла.
     Неизменные величины (н-р % ставка) проверяются в общих блоках
     т.к. наличие или отсутвие консультации влияет только на % ставку => на блок консультация не влияет
  */
  for select k_amount_approved,
             kc_circle,
             kont.k_period
  from kontract kont
  inner join product pr on k_from_pr_id = pr_id and
        pr_code containing '.IND.' and
        ((pr_code containing '.STANDART.') or (pr_code containing '.BUDGET.')) and
        pr_code not containing '.Actiya_Ishenim_Refin.' and
        kont.k_refund = 0 -- < -- не рефинансирование
  inner join knclient kc on k_id = kc_from_k_id and
        kc_amount_approved > 0
  where k_id = :k_id
  into :kc_amount, :kc_circle, :k_period
  do
  begin
    res_error_is_critical = 1;
    max_amount = 0; -- 0 т.к. если цикл 4 и более тогда максимально допустимая сумма = максимально допустимой сумме в параметрах продукта (т.е. 200 000), а проверка на min и max показатели в соотвествии с параметрами продукта реализована в другом кондишене. Цель подобной реалзации: чтобы одни и те же проверки не дублровались в разныз блоках
    max_period = 0; -- смотри пояснение выше
    if (kc_circle <= 1) then
    begin
         max_amount = 80000;
         max_period = 18;
    end
    if (kc_circle between 2 and 3) then
         max_amount = 150000;
    if ((max_amount <> 0) and (kc_amount >  max_amount)) then
    begin
        res_error_text = 'У клиента цикл = '||kc_circle||'. Для клиента, максиамальная допустимая сумма, по данному продукту, = '||max_amount;
        suspend;
    end
    else if ((max_period <> 0) and (k_period > max_period)) then
    begin
        res_error_text = 'У клиента цикл = '||kc_circle||'. Для клиента, максимальный допустимый срок, по даннному продукту, = '||max_period;
        suspend;
    end
    if ((k_period > 18) and (kc_amount < 100001)) then
    begin
        select un.un_from_c_id
        from username un
        where un.un_id = :un_id
        into c_id_approve;
        
        if (exists (select *
                   from hr_pstatus  hps
                   inner join joblist jl on jl.jl_id =  ps_from_jl_id
                   inner join jobkind jk on jk.jk_id = jl.jl_kind
                   where current_date between ps_date_begin and ps_date_end
                   and ps_is_working = 1
                   and jk.jk_system_code = 'JobSystem.Region_Director'
                   and hps.ps_from_c_id  = :c_id_approve)) then
        res_error_is_critical = 0;

        res_error_text = 'Для индив. займа, сумма менее 100001 сом не может быть взата на срок более 18 месяцев.';
        suspend;        
    end
  end
end
