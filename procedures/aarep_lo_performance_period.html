CREATE PROCEDURE AAREP_LO_PERFORMANCE_PERIOD(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    PS_FROM_C_ID D_ID,
    REG_CODE_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    LO_PERIOD_DISB D_INTEGER,
    LO_STAGE_MONTH_FULL D_CURRENCY,
    OTPUSK_DAYS INTEGER,
    LO_WORK_DAYS INTEGER,
    HIRED_DATE_CHANGED D_DATE,
    FIRST_ACTION_DATE_CHANGED D_DATE,
    LAST_EXP_CHANGED D_INTEGER,
    LAST_DEKR_CHANGED D_INTEGER,
    TOTAL_STAJE_CHANGED D_INTEGER,
    MGR_NAME D_STR_100,
    LO_PERIOD_NEW_CLIENT D_INTEGER)
AS
declare variable N integer;
declare variable DD_TMP date;
declare variable REG_ID D_INTEGER;
begin
  /* Аналог AAREP_LO_PERFORMANCE, только работает за период. Некоторые формулы расчитываются в самом эксель файле */
  --  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1; --01.MM.YYYY
  lo_work_days = 5;
  if (extract(day from rep_date_begin) = 1 and
      extract(day from rep_date_end) > 27) then
  begin
    lo_work_days = null;
    select sawd_kolich
    from sa_workdays
    where sawd_year = extract(year from :rep_date_end) and
          sawd_month = extract(month from :rep_date_end)
    into :lo_work_days;
    lo_work_days = coalesce(lo_work_days, 22);
  end

  for select reg_code_name, c_name_fio, ps_from_c_id, ps_from_reg_id
      from aarep_personal_list(:rep_region_id, :rep_date_end, 1) pl
      where jobkind_name in ('Старший КС (СВЗ)', 'Кредитный сотрудник', 'СВЗ', 'СВЗ 0,5') and
            ps_curr_status <> 'Приказ_отпуск_по_уходу_за_ребенком' and
            ps_curr_status <> 'Отпуск без содерж.по уходу за ребенком' and
            ps_curr_status <> 'Декретный отпуск' and
            ps_curr_status <> 'Отпуск по беременности и родам'
      into reg_code_name, c_name_fio, ps_from_c_id, reg_id
  do
  begin
    lo_period_disb = 0;
    lo_period_new_client = 0;

    --Кредиты за период
    select count(ks_from_k_id) count_lo_disb
    from kontstate ks
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks.ks_kc_id = :ps_from_c_id
    into lo_period_disb;
    lo_period_disb = coalesce(lo_period_disb, 0);

    --RFCRU-3017 Новые клиенты за период
    select count(ks_from_k_id) count_lo_disb
    from kontstate ks
    inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks.ks_kc_id = :ps_from_c_id and
          kc.kc_circle = 1
    into lo_period_new_client;
    lo_period_new_client = coalesce(lo_period_new_client, 0);

    -- Проверяем статус отпуска на каждый день недели:
    -- если все 5 дней был в отпуске, исключаем из списка
    otpusk_days = 0;
    n = 0;
    while (n <= lo_work_days) do
    begin
      dd_tmp = dateadd(day, n, rep_date_begin);
      if (exists(select *
                 from hr_voctravel, hr_orders, sa_worktype
                 where hrv_from_ho_id = ho_id and
                       ho_order_type = sawt_id and
                       hrv_from_c_id = :ps_from_c_id and
                       sawt_base_type in (4, 5) and -- отпуск, декретный отпуск
                       sawt_id <> 20 and
                       sawt_hours_real >= 0 and
                       hrv_date_begin <= :dd_tmp and
                       hrv_date_end_fact >= :dd_tmp and
                       hrv_status = 1)) then
        otpusk_days = otpusk_days + 1;
      n = n + 1;
    end

    lo_stage_month_full = null;
    select lo_stage_month_h
    from hp_get_lo_stage_alter(:ps_from_c_id, :rep_date_end)
    into lo_stage_month_full; --Стаж

    select hired_date_changed,
           first_action_date_changed,
           last_exp_changed,
           last_dekr_changed,
           total_staje_changed
    from hp_ks_exp_calc(:rep_date_end, :ps_from_c_id)
    into hired_date_changed,
         first_action_date_changed,
         last_exp_changed,
         last_dekr_changed,
         total_staje_changed;

    mgr_name = null;
    if (exists(select * from region
        where reg_unicode = :reg_id
        and reg_logic_label containing '.mega_office.')) then
    begin
      mgr_name = f_lo_mgr_name(f_lo_mgr_id(null, :rep_date_end, :ps_from_c_id, :reg_id));
    end 

    suspend;
  end
end
