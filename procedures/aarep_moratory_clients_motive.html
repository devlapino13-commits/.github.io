      into :cc_id, :cc_date, :cc_state, :cc_k_id, :m_from_reg_id, :kf_officer_id, :kf_name, :kf_officer_name,
           :kf_reg_name
  do
  begin
    if (cc_id <> :cc_prior_id) then
    begin
      if (cc_state = 2) then
      begin
        if (((cc_date + 30) >= :report_date_begin) and
            ((cc_date + 30) <= :report_date_end)) then
        begin
          cc_m_id = -1;
          select distinct m_id
          from moratory
          where m_from_c_id = :cc_id and
                m_is_active = 1 and
                m_date_begin >= :cc_date rows 1
          into :cc_m_id;
          if ((cc_m_id = -1) or (cc_m_id is null)) then
          begin
            for select k_id, k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no, k_kontract_date,

                       c_name_fio, c_sex, c_birth_date, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire,
                       c_from_reg_id,
                       (select cont_value
                        from hp_contdata_get_email_main(0, :cc_id)), c_kol_family, c_kol_child, c_memo, c_reg_no,
                       c_okpo, c_inn, c_account, c_hired_date, c_tabel,
                       (select cont_value
                        from hp_contdata_get_cell_main(0, :cc_id)), c_others, adl_address_str, kc_from_k_id, kc_number,
                       kc_from_c_id, kc_amount_applicate, kc_from_pt_id, kc_business_type, kc_is_new_business,
                       kc_is_registered, kc_business_name, kc_monthly_oborot, kc_kapital, kc_month_profit,
                       kc_amount_approved, kc_work_kolich, kc_work_kolich_family, kc_is_business_full_time,
                       kc_busuness_subtype, kc_is_new_added, kc_status, kc_circle, kc_patent_no, kc_patent_date,
                       bt_name, bs_name, pr_name, pr_loan_type

                from product, bustype, bussubtype, knclient, clients, kontract, addrclient, addrlist
                where k_id = :cc_k_id and
                      kc_from_k_id = :cc_k_id and
                      kc_from_c_id = c_id and
                      c_id = :cc_id and
                      kc_busuness_subtype = bs_id and
                      kc_business_type = bt_id and
                      k_from_pr_id = pr_id and
                      adc_address_type = 1 and
                      adc_from_c_id = c_id and
                      adc_from_adl_id = adl_id
                into :k_id, :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no,
                     :k_kontract_date, :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period,
                     :k_kolich_lgota, :k_percent, :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id,
                     :k_status, :k_close_date, :k_circle, :k_circle_internal, :k_percent_per_day, :k_memo,
                     :k_amount_approved, :k_percent_for_days, :k_kolich_client_applicate, :k_kolich_client_approved,
                     :k_repay_kolich_closed, :k_class, :k_kommis_perc, :k_from_cs_id, :k_clients_leader_id,
                     :k_clients_cashier_id, :k_credit_target, :k_from_pac_id, :k_is_kredline, :k_kredline_end,
                     :k_stop_perc, :k_stop_fine, :k_index_curr_code, :k_index_curr_rate, :k_from_ma_reg_id,
                     :k_from_ma_app_id, :k_percent_input, :k_percent_category, :c_id, :c_name_fio, :c_sex,
                     :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_from_reg_id,

            begin

              m_last_k_id = :cc_k_id;

              m_client_pass_no = :c_pass_no;

              m_client_birth_date = :c_birth_date;

              m_motive = 'Не заполнена форма моратория';
              m_is_force_text = 'По собств. желанию';
              if (pr_loan_type = 0) then
                pr_loan_type_text = 'Групповой кредит';

                pr_loan_type_text = 'Индив. кредит';
              m_morat_year = extract(year from m_date_begin);
              m_morat_month = extract(month from m_date_begin);

              if (m_last_k_id_prior <> m_last_k_id) then
              begin
                m_last_k_id_prior = m_last_k_id;

                select max(ok_ol_date - ok_sh_date)
                from operkred
                where ok_from_k_id = :m_last_k_id and
                      ok_ol_is_active = 1 and
                      ok_ol_date <= :report_date_end and
                      ok_type in (4, 5) and
                      ok_os + ok_perc + ok_fine < 0 and
                      ok_ol_date > ok_sh_date and
                      ok_from_c_id = 0
                into max_delinq_day;
              end
              suspend;
            end
          end
        end
      end
    end
    cc_prior_id = :cc_id;
  end
