CREATE PROCEDURE MC_CLIENT_BY_GRS_UPD(
    MCF_ID D_ID)
AS
declare variable MCL_ID D_ID;
declare variable C_ID D_ID;
declare variable ADL_ID D_ID;
declare variable UN_VIRT D_ID;
declare variable PARENT_ID D_ID;
declare variable PL_STREET_ID D_ID;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
declare variable ADDRESS_ARRAY D_BLOB_TEXT;
declare variable POST D_INTEGER;
declare variable ADDRESS_STR D_BLOB_TEXT;
declare variable ADDRESS_BUILDING D_BLOB_TEXT;
declare variable ADDRESS_APARTMENT D_BLOB_TEXT;
declare variable SURNAME D_STR_255;
declare variable NAME D_STR_255;
declare variable PATRONYMIC D_STR_255;
declare variable FAMILY_STATUS D_INTEGER;
declare variable GENDER D_STR_SYM;
declare variable DATE_OF_BIRTH D_DATE;
declare variable PASSPORT_SERIES D_STR_10;
declare variable PASSPORT_NUMBER D_STR_20;
declare variable VOID_STATUS D_STR_SYM;
declare variable PIN D_STR_20;
declare variable ISSUED_DATE D_DATE;
declare variable PASSPORT_AUTHORITY D_STR_64;
declare variable PASSPORT_AUTHORITY_CODE D_STR_64;
declare variable EXPIRED_DATE D_DATE;
declare variable ERR_TEXT D_STR_255;
declare variable EIL_ID D_ID;
declare variable MCL_LOGIN D_STR_CODE;
declare variable C_NAME_FIO D_STR_255;
declare variable RD_MARITAL_STATUS D_INTEGER;
declare variable BLS_PASS1 D_ID;
declare variable BLS_PASS2 D_ID;
declare variable DOC_TYPE D_INTEGER;
declare variable MCL_IS_DELETE D_SMALLINT;
declare variable FL_ID D_ID;
declare variable FL_ID_ENG D_ID;
begin
  un_virt = 1000000000000;
  if (exists(select *
             from mc_login mcl
             where mcl.mcl_id = :mcl_id and
                   ((mcl.mcl_os = 1 and
                   cast(mcl.mcl_app_version as float) <= 6.9) or
                   (mcl.mcl_os = 2 and
                   cast(mcl.mcl_app_version as float) <= 3.5)))) then
    select mcf.mcf_from_mcl_id
    from mc_facerecog mcf
    where mcf.mcf_status = 1 and
          mcf.mcf_id = :mcf_id
    into mcl_id;
  else
    mcl_id = mcf_id;

  select eil.eil_from_bls_id, eil.eil_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
        eic.eic_code = 'Tunduk'
  where eil.eil_from_mcl_id = :mcl_id and
        eil.eil_status = 2
  order by eil.eil_status_datetime desc
  rows 1
  into bls_id, eil_id;

  select tg.json_text
  from temp_grs tg
  where tg.bls_id = :bls_id
  into json_text;

  if (coalesce(json_text, '') = '') then
  begin
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into json_text;
  end

  select voidstatus, surname, name, patronymic, family_status, date_of_birth, passport_series, passport_number, pin,
         issued_date, passport_authority, passport_authority_code, expired_date, gender, post, parent_id, pl_street_id,
         address_str, address_building, address_apartment, address_array, rd_marital_status, doc_type
  from hp_parse_json_grs(:json_text)
  into :void_status, :surname, :name, :patronymic, :family_status, :date_of_birth, :passport_series, :passport_number,
       :pin, :issued_date, :passport_authority, :passport_authority_code, :expired_date, :gender, :post, :parent_id,
       :pl_street_id, :address_str, :address_building, :address_apartment, :address_array, :rd_marital_status, :doc_type;

  if (void_status = 1) then
    exit;

  -- address ----------------------------------------------------------- <
  if (coalesce(address_array, '') = '' or address_str containing 'ликвидирован') then
  begin
    update grs_clients gc
    set gc.gc_status = 3
    where gc.gc_from_c_id = :c_id and
          gc.gc_status = 2 and
          gc.gc_is_last = 1 and
          gc.gc_from_eil_id = :eil_id;

    update addrclient adc
    set adc.adc_from_adl_id = 41701000000000
    where adc.adc_address_type = 1 and
          adc.adc_from_c_id = :c_id;
  end
  if (coalesce(:parent_id, 0) > 0) then
    insert into addrlist (adl_from_pl_id, adl_depart_no, adl_depart_pref, adl_flat_no, adl_building_no, adl_section_no,
                          adl_cross_street_id, adl_address_str, adl_comment, adl_post_index, adl_is_edited,
                          adl_from_un_id, adl_street_id, adl_is_without_street, adl_is_filled, adl_long, adl_lat,
                          adl_geo_count, adl_place_pls_id, adl_street_pls_id, adl_lang_kind)
    values (:parent_id, 0, :address_building, :address_apartment, 0, 0, null, :address_str, null, :post, 0, :un_virt,
            :pl_street_id, iif(:pl_street_id is null, 1, 0), 1, null, null, -1, null, null, 0)
    returning adl_id
    into :adl_id;
  else
    adl_id = 41701000000000;

  select c.c_id
  from clients c
  where c.c_inn = :pin
  into c_id;

  select mcl.mcl_login, mcl.mcl_is_delete
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into mcl_login, :mcl_is_delete;

  select drl.drl_from_bls_1, drl.drl_from_bls_2
    from docrecoglist drl
    inner join passkind pk on pk.pk_id = drl.drl_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id and
                                  pdt.pdt_is_main_doc = 1
    where drl.drl_from_mcl_id = :mcl_id and
          drl.drl_is_main_doc = 1 and
          drl.drl_from_bls_1 > 0
          order by drl.drl_saved_datetime desc
          rows 1
  into bls_pass1, bls_pass2;

  if (coalesce(:c_id, 0) = 0) then
  begin
    execute procedure fm_add_client(f_translit_fio_grs('cyr', 'cyr', coalesce(:surname, '') || ' ' || coalesce(:name, '') || ' ' || coalesce(:patronymic, ''), 'kg'), --
        iif(:gender = 'M', 1, 0), --
        :date_of_birth, --
        null, --
        null, --
        null, --
        f_get_rd_id('Citizen.KG'), --
        :doc_type, -- тип документа (8 - пластик КГ, 296 - биометрический паспорт, 372 - новый паспорт КР)
        :passport_series || :passport_number, --
        :issued_date, --
        :pin, --
        1000000000000,
        :surname,
        :name,
        :patronymic,
         null,
         null,
         null)
        returning_values :c_id, :err_text;

    update clients c
    set c.c_blob_pass1_id = :bls_pass1,
        c.c_blob_pass2_id = :bls_pass2
    where c.c_id = :c_id;

    update clientdocs cds
    set cds.cds_doc_blob_1 = :bls_pass1,
        cds.cds_doc_blob_2 = :bls_pass2
    where cds.cds_from_c_id = :c_id and
          cds.cds_is_delete = 0 and
          cds.cds_is_main_doc = 1;

    if (:mcl_is_delete = 0 and :mcl_login not containing '_') then -- RFCKG-5025
        execute procedure hp_continfo_update(:c_id, 'cell', :mcl_login, 0, :un_virt, current_timestamp, 1);
  end
  else
  begin
    c_name_fio = f_translit_fio_grs('cyr', 'cyr', coalesce(:surname, '') || ' ' || coalesce(:name, '') || ' ' || coalesce(:patronymic, ''), 'kg');

    select fl_id, fl_id_eng
    from hp_prepare_fio(:surname, :name, :patronymic, null, null, null)
    into :fl_id, fl_id_eng;

    update clients c
    set c.c_name_fio = :c_name_fio,
        c.c_sex = iif(:gender = 'M', 1, 0),
        c.c_birth_date = :date_of_birth,
        c.c_pass_no = :passport_series || :passport_number,
        c.c_doc_type = :doc_type,
        c.c_pass_date = :issued_date,
        c.c_pass_date_expire = :expired_date,
        c.c_family = :family_status,
        c.c_index_c_pass_no = :passport_series || :passport_number,
        c.c_verify_status = 2,
        c.c_from_fl_id = :fl_id,
        c.c_name_eng_fl_id = :fl_id_eng
    where c.c_id = :c_id and
          c.c_inn = :pin;

    update clientdocs cds
    set cds.cds_doc_blob_1 = :bls_pass1,
        cds.cds_doc_blob_2 = :bls_pass2
    where cds.cds_from_c_id = :c_id and
          cds.cds_is_delete = 0 and
          cds.cds_is_main_doc = 1;


    if ((:mcl_is_delete = 0 and :mcl_login not containing '_' -- RFCKG-5025
        and coalesce(f_cont_info('clients', :c_id, 'cell_main', 'ok'), '') = '')
    ) then
        execute procedure hp_continfo_update(:c_id, 'cell', :mcl_login, 0, :un_virt, current_timestamp, 1);
  end
  if (:c_id > 0) then
  begin
    if (exists(select *
               from mc_login mcl
               where mcl.mcl_id = :mcl_id and
                     ((mcl.mcl_os = 1 and
                     cast(mcl.mcl_app_version as float) <= 6.9) or
                     (mcl.mcl_os = 2 and
                     cast(mcl.mcl_app_version as float) <= 3.5)))) then
    begin
      update mc_login mcl
      set mcl.mcl_from_c_id = :c_id
      where mcl.mcl_id = :mcl_id;

      update mc_facerecog mcf
      set mcf.mcf_from_c_id = :c_id
      where mcf.mcf_from_mcl_id = :mcl_id;
    end

    update extinfolist eil
    set eil.eil_from_c_id = :c_id
    where eil.eil_from_mcl_id = :mcl_id and
          eil.eil_status = 2;

    update grs_clients gc
    set gc.gc_from_c_id = :c_id
    where gc.gc_from_eil_id = :eil_id and
          gc.gc_from_mcl_id = :mcl_id;

    update or insert into addrclient (adc_from_c_id, adc_address_type, adc_from_adl_id, adc_from_un_id)
    values (:c_id, 1, :adl_id, :un_virt)
    matching (adc_from_c_id, adc_address_type);

    update or insert into grs_addrlist (ga_from_eil_id, ga_from_adl_id)
    values (:eil_id, :adl_id);
  end

  execute procedure hp_objectprops_set_with_rd(f_objkind('clients', 'id'), :c_id, -1,
      f_get_rd_id('AnketaQuestions.Family'), current_date, '', 0, :family_status, :un_virt);

  execute procedure hp_objectprops_set_with_rd(f_objkind('clients', 'id'), :c_id, -1,
      f_get_rd_id('AnketaQuestions.Family_tunduk'), current_date, '', 0, :rd_marital_status, :un_virt);
end
