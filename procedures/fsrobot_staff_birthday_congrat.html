CREATE PROCEDURE FSROBOT_STAFF_BIRTHDAY_CONGRAT(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable C_NAME_FIO varchar(100);
declare variable RES_TMP_BODY_1 blob sub_type 1 segment size 80;
declare variable REG_NAME varchar(100);
declare variable RES_TMP_BODY_2 blob sub_type 1 segment size 80;
declare variable DD_MONTH integer;
declare variable DD_DAY integer;
declare variable JL_NAME varchar(100);
declare variable CM_DATABASE_ID integer;
declare variable HAS_REGION integer;
declare variable REG_ID integer;
declare variable HAS_HQ integer;
declare variable NO_LEAP_YEAR integer;
begin
-- Процедура вызывается из FSROBOT_SH_MAIN
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды, ID-СВЗ
-- Rep_UN_ID - пользователь

-- STAFF_BIRTHDAY_CONGRAT - поздравление сотрудников

  res_maillist = rep_email;
  res_tmp_body_1 = '';
  res_tmp_body_2 = '';
  rep_params = '';

  dd_month = extract(month from current_date);
  dd_day = extract(day from current_date);
  no_leap_year = 0;
  if (dd_month * 100 + dd_day = 301 and
      extract(month from dateadd(day, -1, current_date)) * 100 + extract(day from dateadd(day, -1, current_date)) = 228) then
    no_leap_year = 1; -- не весокосный год

  :cm_database_id = f_get_db_id();

  res_maillist = '';
  has_region = 0;
  has_hq = 0;
  res_c_id = 0;

  for select c_name_fio, jl_name, reg_code || ' ' || reg_name, ps_from_reg_id
      from hr_pstatus, region, clients, joblist, jobkind
      where ps_from_c_id = c_id and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            ((:cm_database_id not in (39801, 41701) and
            ps_from_reg_id = 1010100) or
            (:cm_database_id = 41701 and
            ps_from_reg_id in (1090100, 1060100)) or
            (:cm_database_id = 39801 and
            ps_from_reg_id in (1010100, 1020100))) and
            (extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = :dd_month * 100 + :dd_day
                or (:no_leap_year = 1
                  and extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = 229))and
            jl_kind = jk_id and
            c_id <> 1010100015257 -- Kərimov Səbuhi Abgül oğlu
            and
            c_id <> 1090100079709 -- Ким Венера Анатольевна
            and
            c_id <> 1090100839631 -- Ирода
            and
            c_id <> 1090100224546 -- Ирода
            and
            c_id <> 1010100029828 -- Ирода
      order by jl_order_no, c_birth_date, c_name_fio
      into :c_name_fio, jl_name, reg_name, reg_id
  do
  begin
    res_tmp_body_1 = BLOB_APPEND(res_tmp_body_1, '<b>' || :c_name_fio || '</b><br>');
    res_tmp_body_1 = BLOB_APPEND(res_tmp_body_1, reg_name || ', ' || jl_name || '<br>');
    res_tmp_body_1 = BLOB_APPEND(res_tmp_body_1, '<br>');

    has_hq = 1;
  end

  for select c_name_fio, jl_name, reg_code || ' ' || reg_name, ps_from_reg_id
      from hr_pstatus, region, clients, joblist, jobkind
      where ps_from_c_id = c_id and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            not((:cm_database_id not in (39801, 41701) and
            ps_from_reg_id = 1010100) or
            (:cm_database_id = 41701 and
            ps_from_reg_id in (1090100, 1060100)) or
            (:cm_database_id = 39801 and
            ps_from_reg_id in (1010100, 1020100))) and
            (extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = :dd_month * 100 + :dd_day
            or (:no_leap_year = 1
                  and extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = 229))and
            jl_kind = jk_id and
            c_id <> 1010100015257 -- Kərimov Səbuhi Abgül oğlu
            and
            c_id <> 1090100079709 -- Ким Венера Анатольевна
            and
            c_id <> 1090100839631 -- Ирода
            and
            c_id <> 1090100224546 -- Ирода
            and
            c_id <> 1010100029828 -- Ирода
      order by c_birth_date, c_name_fio
      into :c_name_fio, jl_name, reg_name, reg_id
  do
  begin
    res_tmp_body_1 = BLOB_APPEND(res_tmp_body_1, '<b>' || :c_name_fio || '</b><br>');
    res_tmp_body_1 = BLOB_APPEND(res_tmp_body_1, reg_name || ', ' || jl_name || '<br>');
    res_tmp_body_1 = BLOB_APPEND(res_tmp_body_1, '<br>');

    has_region = 1;
  end

  dd_month = extract(month from (current_date + 1));
  dd_day = extract(day from (current_date + 1));
  no_leap_year = 0;
  if (dd_month * 100 + dd_day = 301 and
      extract(month from dateadd(day, -1, current_date + 1)) * 100 + extract(day from dateadd(day, -1, current_date + 1)) = 228) then
    no_leap_year = 1; -- не високосный год

  for select c_name_fio, jl_name, reg_code || ' ' || reg_name, ps_from_reg_id
      from hr_pstatus, region, clients, joblist, jobkind
      where ps_from_c_id = c_id and
            ps_date_begin <= (current_date + 1) and
            ps_date_end >= (current_date + 1) and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            (extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = :dd_month  * 100 + :dd_day
              or (:no_leap_year = 1
                  and extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = 229))and
            jl_kind = jk_id and
            c_id <> 1010100015257 -- Kərimov Səbuhi Abgül oğlu
            and
            c_id <> 1090100079709 -- Ким Венера Анатольевна
            and
            c_id <> 1090100839631 -- Ирода
            and
            c_id <> 1090100224546 -- Ирода
            and
            c_id <> 1010100029828 -- Ирода
      order by c_birth_date, c_name_fio
      into :c_name_fio, jl_name, reg_name, reg_id
  do
  begin
    res_tmp_body_2 = BLOB_APPEND(res_tmp_body_2, '<b>' || :c_name_fio || '</b><br>');
    res_tmp_body_2 = BLOB_APPEND(res_tmp_body_2, reg_name || ', ' || jl_name || '<br>');
    res_tmp_body_2 = BLOB_APPEND(res_tmp_body_2, '<br>');

    if ((:cm_database_id not in (39801, 41701) and reg_id = 1010100) or
        (:cm_database_id = 39801 and reg_id in (1010100, 1020100)) or
        (:cm_database_id = 41701 and reg_id in (1090100, 1060100))) then
      has_hq = 1;
    else
      has_region = 1;
  end
  if ((res_tmp_body_1 = '') and
      (res_tmp_body_2 = '')) then
    exit;

  if (cm_database_id = 81001) then
  begin
    if ((has_hq = 1) and
        (has_region = 1)) then
      res_maillist = 'ru.birthday@molbulak.com; ru.birthday2@molbulak.com';
    if ((has_hq = 1) and
        (has_region = 0)) then
      res_maillist = 'ru.birthday@molbulak.com';
    if ((has_hq = 0) and
        (has_region = 1)) then
      res_maillist = 'ru.birthday2@molbulak.com';
  end
  if (cm_database_id = 94401) then
  begin
    if ((has_hq = 1) and
        (has_region = 1)) then
      res_maillist = 'az.hq@mbulak.az; az.fm@mbulak.az'; -- RFCAZ-435. Почта изменена с az.hq@molbulak.com на az.hq@mbulak.az
    if ((has_hq = 1) and
        (has_region = 0)) then
      res_maillist = 'az.hq@mbulak.az'; -- RFCAZ-435. Почта заменена с az.hq@molbulak.com на az.hq@mbulak.az
    if ((has_hq = 0) and
        (has_region = 1)) then
      res_maillist = 'az.fm@mbulak.az';
  end
  if (cm_database_id = 39801) then
  begin
    -- RFCKZ-327
    if ((has_hq = 1) and (has_region = 1)) then
      res_maillist = 'kz.go@mbulak.kz; kz.mnm@mbulak.kz';
    if ((has_hq = 1) and (has_region = 0)) then
      res_maillist = 'kz.go@mbulak.kz';
    if ((has_hq = 0) and (has_region = 1)) then
      res_maillist = 'kz.mnm@mbulak.kz';
  end
  if (cm_database_id = 41701) then
  begin
    if ((has_hq = 1) and (has_region = 1)) then
      res_maillist = 'hqa@mbulak.kg; kg.mnm@molbulakfinance.kg';
    if ((has_hq = 1) and (has_region = 0)) then
      res_maillist = 'hqa@mbulak.kg';
    if ((has_hq = 0) and (has_region = 1)) then
      res_maillist = 'kg.mnm@molbulakfinance.kg';
  end

    res_subject = iif(cm_database_id <> 94401, 'С Днем Рождения!', 'Ad günü təbriki!');

    if ((res_tmp_body_1 <> '') and
        (res_tmp_body_2 <> '')) then
    begin
      res_body = iif(cm_database_id <> 94401, f_format('<b><font color="blue">%1</font></b><br>', 'Коллеги, поздравляем сегодня наших именинников!'),
                                              f_format('<b><font color="blue">%1<br>%2</font></b><br>', 'Salam!', 'Ad günü münasibəti ilə həmkarlarımızı təbrik edirik!'));
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, res_tmp_body_1);
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, f_format('<b><font color="green">%1</font></b><br>', iif(cm_database_id <> 94401, 'Желаем крепкого здоровья и успехов в работе!',
                                                                                                               'Can sağlığı və işinizdə uğurlar arzulayırıq!')));
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, f_format('<b><font color="darkblue">%1:</font></b><br>', iif(cm_database_id <> 94401, 'Завтра будем поздравлять следующих сотрудников',
                                                                                                                   'Sabah təbrik edəcəyimiz həmkarlar')));
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, res_tmp_body_2);
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, '<br>');

      suspend;
    end

    if ((res_tmp_body_1 = '') and
        (res_tmp_body_2 <> '')) then
    begin
      res_body = f_format('<b><font color="darkblue">%1:</font></b><br>', iif(cm_database_id <> 94401, 'Завтра будем поздравлять следующих сотрудников',
                                                                                                       'Sabah təbrik edəcəyimiz həmkarlar'));
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, res_tmp_body_2);
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, '<br>');

      suspend;
    end

    if ((res_tmp_body_1 <> '') and
        (res_tmp_body_2 = '')) then
    begin
      res_body = iif(cm_database_id <> 94401, f_format('<b><font color="blue">%1</font></b><br>', 'Коллеги, поздравляем сегодня наших именинников!'),
                                              f_format('<b><font color="blue">%1<br>%2</font></b><br>', 'Salam!', 'Ad günü münasibəti ilə həmkarlarımızı təbrik edirik.'));
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, res_tmp_body_1);
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, f_format('<b><font color="green">%1</font></b><br>', iif(cm_database_id <> 94401, 'Желаем крепкого здоровья и успехов в работе!',
                                                                                                               'Can sağlığı və işinizdə uğurlar arzulayırıq!')));
      res_body = BLOB_APPEND(res_body, '<br>');
      res_body = BLOB_APPEND(res_body, '<br>');
      suspend;
    end

end
