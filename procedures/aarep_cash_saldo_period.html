CREATE PROCEDURE AAREP_CASH_SALDO_PERIOD(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REG_BRANCH VARCHAR(100),
    REG_OKRUG VARCHAR(100),
    REG_SALDO_35 DECIMAL(18,4),
    REG_AUDIT_COUNT INTEGER,
    REG_PAST_DAYS INTEGER,
    REG_AUDIT_ERR_COUNT INTEGER,
    REG_AUDIT_ALL_COUNT INTEGER,
    REG_CASH_LIMIT DECIMAL(18,4),
    REG_DISTANCE_TO_BANK DECIMAL(18,4),
    COUNTER D_INTEGER)
AS
declare variable dd d_date;
declare variable res_saldo_cash decimal(18,4);
declare variable last_dd date;
begin
  /* отображает остатки кассы за последние 5 недель */
  for select reg_unicode, reg_code, reg_name
      from region_list(:rep_region_id, 1, 3, 1), region
      where id_list = reg_unicode and
            reg_date_close is null
      into :reg_id, :reg_code, :reg_name
  do
  begin
    counter = 0;
    select aa_curr_branch_name, aa_curr_level3_name
    from aarep_get_office_categ(:reg_id, :rep_date_end)
    into :reg_branch, :reg_okrug;

    reg_cash_limit = 0;
    reg_distance_to_bank = 0;
    select rlm_limit_cash_max, rlm_bank_distance
    from reglimits
    where rlm_from_reg_id = :reg_id and
          rlm_date <= :rep_date_end
    order by rlm_date desc, rlm_id desc
    rows 1
    into :reg_cash_limit, reg_distance_to_bank;
    if (reg_cash_limit is null) then
      reg_cash_limit = 0;
    if (reg_distance_to_bank is null) then
      reg_distance_to_bank = 0;
    dd = :rep_date_begin;
    while (:dd <= :rep_date_end) do
    begin
      res_saldo_cash = 0;
      select res_saldo_cash
      from aarep_money_cash_saldo(:reg_id, :dd)
      into :res_saldo_cash;

      reg_saldo_35 = :res_saldo_cash;
      if (reg_saldo_35 > reg_cash_limit) then
        counter = counter + 1;
      dd = dateadd(day, 1, :dd);
    end
    suspend;
  end
end
