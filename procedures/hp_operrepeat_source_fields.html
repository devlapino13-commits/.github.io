CREATE PROCEDURE HP_OPERREPEAT_SOURCE_FIELDS(
    RES_STORED_PROC D_STR_CODE)
RETURNS (
    RES_FIELDS D_STR_8100)
AS
begin
  res_fields = '';
  select list(f_format('{"name":"%1","caption":%2,"type":"%3","hidden":%4, "width":%5}',
                   trim(lower(proc_field)),
                   f_json_escape(trim(proc_desc), 'str'),
                   trim(lower(proc_field_source)),
                   iif(trim(lower(proc_field)) = 'rec_uniq', 1, 0),
                   case trim(lower(proc_field_source))
                   when 'd_currency' then 90
                   when 'd_integer' then 60
                   when 'd_date' then 80
                   when 'd_datetime' then 120
                   when 'd_id' then 100
                   when 'd_str_code' then 150
                   when 'd_str_255' then 150
                   when 'd_str_long' then 150
                   else 100
                   end))
  from print_procedure_all_fields(:res_stored_proc)
  where proc_field_type = 1
  into res_fields;

  res_fields = '{"fields":[' || res_fields || ']}';

  suspend;
end
