CREATE PROCEDURE PRINT_SMEVPASS_NOTIFY(
    SPM_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_255,
    APP_SUMM_STR D_STR_CODE,
    FAILURE_DATE_STR D_STR_CODE,
    SPM_KIND D_SMALLINT,
    C_SEX D_SMALLINT)
AS
declare variable sbc_id d_id;
  declare variable c_id d_id;
  declare variable ma_id d_id;
  declare variable k_id d_id;
begin
  select c_id, upper(c_name_fio), spm_from_ma_id, sbc_id, spm_kind, c_sex
  from smevpass_msg, smevpass_blockclients, clients
  where spm_id = :spm_id 
    and sbc_id = spm_from_sbc_id
    and c_id = sbc_from_c_id
  into c_id, c_name_fio, ma_id, sbc_id, spm_kind, c_sex;

  if (ma_id is null) then
    select spm_from_ma_id
    from smevpass_msg, smevpass_blockclients
    where spm_from_sbc_id = :sbc_id
      and spm_kind = 1
      and spm_from_ma_id is not null
    rows 1
    into ma_id;

  select k_id
  from kontract
  where k_from_ma_app_id = :ma_id
  order by k_status desc
  rows 1
  into k_id;

  select f_datetostr(krl_datetime)
  from kontreslist
  where krl_ma_reg_id = 0
    and (krl_ma_id = :ma_id or krl_from_k_id = coalesce(:k_id, 0))
  order by krl_datetime desc
  rows 1
  into failure_date_str;

  select f_numtostr(mac_amount, 2)
  from miniappclient
  where mac_from_ma_id = :ma_id
    and mac_from_c_id = :c_id
  into app_summ_str;

  suspend;
end
