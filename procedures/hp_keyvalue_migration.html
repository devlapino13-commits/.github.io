CREATE PROCEDURE HP_KEYVALUE_MIGRATION
RETURNS (
    SQL_SCRIPT VARCHAR(1024))
AS
declare variable sql_text varchar(1024);
declare variable t_id integer;
declare variable t_name varchar(30);
declare variable t_field varchar(20);
declare variable kv_id decimal(18,0);
begin
  for select t_id, t_name, t_field
      from hp_keyvalue_migration_tmp
      into :t_id, :t_name, :t_field
  do
  begin
    kv_id = null;
    sql_text = 'select max(' || :t_field || ') from ' || :t_name;
    execute statement(:sql_text)
        into :kv_id;
    if (kv_id is null) then
      kv_id = 0;
    else
      kv_id = kv_id + 1;

    sql_script = 'ALTER SEQUENCE GEN_' || :t_name || '_ID RESTART WITH ' || :kv_id || ';';
    suspend;
  end
end
