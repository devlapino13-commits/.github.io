CREATE PROCEDURE AAREP_CUSTOMER_REVIEW(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE = current_date,
    REP_DATE_END DATE = current_date)
RETURNS (
    RES_ACL_ID ID_TYPE,
    RES_ACL_FROM_REG_ID INTEGER,
    RES_C_NAME_FIO STR_100,
    RES_KC_CIRCLE INTEGER,
    RES_K_CURR_STATUS INTEGER,
    RES_ACL_DATE DATE,
    RES_ACL_DATETIME TIMESTAMP,
    RES_ACL_LAST_DATETIME TIMESTAMP,
    RES_TIME_PROC STR_20,
    RES_SENDER STR_100,
    RES_INFO_AMOUNT_DUE STR_20,
    RES_INFO_DAYS_OVERDUE STR_100,
    RES_INFO_FSSP STR_255,
    RES_INFO_FMS STR_255,
    RES_INFO_DOFIN STR_255,
    RES_INFO_EFRSB STR_255,
    RES_INFO_COMMENT STR_255,
    RES_EX_OFFICER STR_100,
    RES_EX_KOFFICER STR_100,
    RES_STATUS STR_20,
    RES_KSTATUS STR_20)
AS
declare variable acl_from_reg_id integer;
declare variable acl_status integer;
declare variable acl_kstatus integer;
declare variable rd_id_check_info integer;
declare variable rd_id_check_info_err integer;
declare variable reg_list d_blob_text;
declare variable c_id id_type;
declare variable k_id id_type;
declare variable un_id id_type;
declare variable k_disb_date date;
declare variable acm_memo d_blob_text;
declare variable json_key str_255;
declare variable json_value str_255;
begin
  rd_id_check_info = f_get_rd_id('ActionGroup.Check_Info');
  rd_id_check_info_err = f_get_rd_id('ActionGroup.Check_Info_Error');

  if (rep_region_id = 1000000) then reg_list = '';
    else reg_list = (select list(id_list,',') from region_list(:rep_region_id, 1, 3, 1));
  for
    select
      acl_id,
      acl_from_reg_id,
      acl_respons_c_id,
      acl_from_un_id,
      acl_status,
      c_name_fio,
      acl_date,
      acl_datetime,
      acl_last_datetime,
      k_id,
      k_disb_date,
      kc_circle,
      acm_memo
    from actionlist
      left join actionmemo
        on acm_from_acl_id = acl_id
      inner join knclient
        on kc_from_c_id = acl_object_id
      inner join kontract
        on k_id = kc_from_k_id
          and k_applicate_date = acl_date
      inner join clients
        on c_id = acl_object_id
    where acl_date between :rep_date_begin and :rep_date_end
      and acl_from_rd_id in (:rd_id_check_info, :rd_id_check_info_err)
  into
    :res_acl_id,
    :res_acl_from_reg_id,
    :c_id,
    :un_id,
    :acl_status,
    :res_c_name_fio,
    :res_acl_date,
    :res_acl_datetime,
    :res_acl_last_datetime,
    :k_id,
    :k_disb_date,
    :res_kc_circle,
    :acm_memo
  do
  begin
    if ((reg_list <> '')
      and (reg_list not containing :res_acl_from_reg_id))
    then continue;
    -- отв.исполнитель
    res_ex_officer = (select c_name_fio from clients where c_id = :c_id);
    -- отправитель
    res_sender = (select un_fio from username where un_id = :un_id);
    -- звёздность
    res_k_curr_status = (
      select k_curr_status
      from hp_get_kontract_status(:k_id, iif(:k_disb_date > :rep_date_end, :rep_date_end, :k_disb_date)));
    res_time_proc = cast(datediff(minute from :res_acl_datetime to :res_acl_last_datetime) as str_20);

    if (acm_memo is not null) then
    begin
      for select json_key, trim(json_value)
          from fm_json_parse(:acm_memo, 0, 'verify')
          into :json_key, :json_value
      do
      begin
        if (json_value in ('0.0000', 'нет данных')) then json_value = '';
        if (json_key = 'sumdebt') then
          res_info_amount_due = json_value;
        else
        if (json_key = 'daysdelay') then
          res_info_days_overdue = json_value;
        else
        if (json_key = 'fsspdata') then
          res_info_fssp = json_value;
        else
        if (json_key = 'extbdata') then
          res_info_dofin = json_value;
        else
        if (json_key = 'fmsdata') then
          res_info_fms = json_value;
        else
        if (json_key = 'efrsbdata') then
          res_info_efrsb = json_value;
        else
        if (json_key = 'comment') then
          res_info_comment = json_value;
        --else
        --if (json_key = 'level') then
          --res_status = iif(json_value=1, 'отказан', 'подтверждён');
      end
    end else begin
      res_info_amount_due = '';
      res_info_days_overdue = '';
      res_info_fssp = '';
      res_info_fms = '';
      res_info_dofin = '';
      res_info_efrsb = '';
      res_info_comment = '';
      --res_status = '';
    end

    -- решение по проверке инфы клиента
    res_status =
      case :acl_status
        when -3 then 'oтмена заполнения'
        when -2 then 'отказан'
        when -1 then 'на доработку'
        when  0 then 'в ожидании'
        when  1 then 'принят на обработку'
        when  2 then 'подтвежден'
       else '' end;
    if (acl_status = -3) then res_time_proc = '';
    -- решение по займу
    select first 1
      acl_status,
      acl_respons_c_id
    from actionlist
    where acl_object_kind = 9
      and acl_object_id = :k_id
    order by acl_id desc
    into
      :acl_kstatus,
      :c_id;
    res_ex_kofficer = (select c_name_fio from clients where c_id = :c_id);

    res_kstatus =
      case :acl_kstatus
        when -3 then 'oтмена заполнения'
        when -2 then 'отказан'
        when -1 then 'на доработку'
        when  0 then 'в ожидании'
        when  1 then 'принят на обработку'
        when  2 then 'подтвежден'
       else '' end;
    suspend;
  end
end
