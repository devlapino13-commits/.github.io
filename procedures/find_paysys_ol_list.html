CREATE PROCEDURE FIND_PAYSYS_OL_LIST(
    REGION_ID INTEGER,
    DATE_BEGIN DATE,
    DATE_END DATE,
    DEL_STATUS INTEGER,
    MEMO_TEXT VARCHAR(254),
    USER_ID DECIMAL(18,0),
    PSP_ID INTEGER)
RETURNS (
    OB_CURRENCY VARCHAR(3),
    OB_SUMMA_VAL DECIMAL(18,4),
    OB_SUMMA DECIMAL(18,4),
    OL_ID ID_TYPE,
    OL_FROM_REG_ID INTEGER,
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_CALC_MEMO VARCHAR(1024),
    IS_DELETE SMALLINT,
    IS_DELETE_NAME VARCHAR(20),
    UN_NAME VARCHAR(30),
    DN_NAME VARCHAR(255),
    OB_ACC_SHORT_DT VARCHAR(20),
    OB_ACC_SHORT_KT VARCHAR(20),
    OL_REGION_NAME VARCHAR(100),
    OL_DOC_KIND INTEGER,
    OB_NUMBER INTEGER,
    PSM_ID ID_TYPE,
    PSL_ID INTEGER,
    PSI_ID ID_TYPE,
    PSI_CODE VARCHAR(20),
    PSP_NAME VARCHAR(255))
AS
declare variable old_ol_id id_type;
begin
  old_ol_id = -1;
  for select ob_currency, ob_summa_val, ob_summa, ol_id, ol_from_reg_id, ol_date, ol_doc_no, is_delete, is_delete_name,
             un_name, dn_name, ol_calc_memo, ob_acc_short_dt, ob_acc_short_kt, ol_region_name, ol_doc_kind, ob_number
      from aarep_aa_bookview(:region_id, :date_begin, :date_end, :del_status, '', '', :memo_text, '', :user_id)
      where (ob_acc_short_dt starting '10001' or
            ob_acc_short_kt starting '10001' or
            --ob_acc_short_dt starting '96001' or
            --ob_acc_short_kt starting '96001' or
            ob_acc_short_dt starting '10150' or
            ob_acc_short_kt starting '10150' or
            ob_acc_short_dt starting '10101' or
            ob_acc_short_kt starting '10101' or
            ob_acc_short_dt starting '10003' or
            ob_acc_short_kt starting '10003')
      into ob_currency, ob_summa_val, ob_summa, ol_id, ol_from_reg_id, ol_date, ol_doc_no, is_delete, is_delete_name,
           un_name, dn_name, ol_calc_memo, ob_acc_short_dt, ob_acc_short_kt, ol_region_name, ol_doc_kind, ob_number
  do
  begin
    if (old_ol_id <> ol_id) then
    begin
-- При смене ol_id получаем информацию о закрепленном бланке
      old_ol_id = ol_id;
      psm_id = null;
      psl_id = null;
      psi_id = null;
      psi_code = null;
      select first 1 psi_id, psi_code
      from paysys_main, paysys_list, paysys_item
      where psl_from_psm_id = psm_id and
            psl_from_psi_id = psi_id and
            psl_object_kind = 4 -- Operlist
            and
            psi_from_psp_id = :psp_id and
            psm_is_delete = 0 and
            psl_is_delete = 0 and
            psl_object_id = :ol_id
      group by psi_id, psi_code
      having sum(psl_sign) = 1
      into psi_id, psi_code;
      if (psi_id is not null) then
      begin
        select psl_id, psl_from_psm_id
        from paysys_list
        where psl_from_psi_id = :psi_id and
              psl_sign = 1
        order by psl_from_psm_id desc
        rows 1
        into psl_id, psm_id;

        select psp_name
        from paysys_project
        where psp_id = :psp_id
        into psp_name;
      end
    end
    suspend;
  end
end
