CREATE PROCEDURE AAREP_PLANINDICATORS_NEW(
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    MO_FIO_LO D_STR_100,
    FACT_DEL_BAL_31_360 CURR_TYPE,
    PLAN_DEL_BAL_31_360 CURR_TYPE,
    FACT_KOL_SVZ CURR_TYPE,
    PLAN_KOL_SVZ CURR_TYPE,
    FACT_DISB_AMOUNT CURR_TYPE,
    PLAN_DISB_AMOUNT CURR_TYPE,
    OSTATOK_DEL_BAL_31_360 CURR_TYPE,
    OSTATOK_KOL_SVZ CURR_TYPE,
    OSTATOK_DISB_AMOUNT CURR_TYPE,
    FACT_BALANCE CURR_TYPE,
    PLAN_BALANCE CURR_TYPE,
    OSTATOK_BALANCE CURR_TYPE)
AS
declare variable REG_ID D_INTEGER;
declare variable LML_ID D_INTEGER;
declare variable MGR_C_ID D_ID;
declare variable REG_LOGIC_LABEL D_STR_100;
declare variable REP_PERIOD D_INTEGER;
declare variable MO_ID D_ID;
declare variable REP_DATE_BEGIN D_DATE;
begin
  mo_id = -1;
  if (rep_date_end >= '01.06.2021') then mo_id = 1010100066847; --текущего менеджера нужно добавить в текущу ОИ
  rep_period = extract(year from :rep_date_end) || right((extract(month from :rep_date_end) + 100), 2);
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;

  for
  select reg_unicode,
         reg_name,
         lml_id,
         coalesce(lml_from_c_id, -1),
         lml_name,
         reg_logic_label
  from region left join loanmgrlist on (reg_unicode = lml_from_reg_id and
                                       :REP_DATE_END between lml_date_begin and lml_date_end)
  where reg_status = 3
  and is_delete = 0
  and cast(:rep_date_end as date) < coalesce(reg_date_close, '31.12.9999')
  and reg_unicode <> 1010100
  order by reg_name
  into reg_id,
       reg_name,
       lml_id,
       mgr_c_id,
       mo_fio_lo,
       reg_logic_label
    do
    begin
      if (reg_logic_label containing '.mega_office.' and mgr_c_id = -1) then continue;

      --  Фактический баланс просрочек 31-360 дней
      fact_del_bal_31_360 = null;
      select sum(oks_balance)
      from oksummary
      where oks_region_id = :reg_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            :rep_date_end - oks_delinq_date between 31 and 360 and
            oks_close_state = 0
            and (:mgr_c_id = -1 or
            :lml_id = f_lo_mgr_id(oks_from_k_id,:rep_date_end,oks_kc_id,oks_region_id))
      into fact_del_bal_31_360;
      fact_del_bal_31_360 = coalesce(fact_del_bal_31_360, 0);
      fact_del_bal_31_360 = fact_del_bal_31_360 / 1000;

      --  Плановый баланс просрочек
      plan_del_bal_31_360 = null;

      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :mgr_c_id , 'PlanIndicators.BalancePar31_360', :rep_date_end, :rep_period)
      into plan_del_bal_31_360;

      plan_del_bal_31_360 = coalesce(plan_del_bal_31_360, 0);
      plan_del_bal_31_360 = plan_del_bal_31_360 / 1000;

      --  фактическое кол-во Випс на дату
      fact_kol_svz = null;
      select sum(iif(jk_name = 'СВЗ 0,5',0.5,1))
      from hr_pstatus, joblist, jobkind
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_id and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.LO_Standart' and
            (exists(select *
                   from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                   where is_decret = 0)
             or ps_from_c_id = :mo_id)
             and (:mgr_c_id = -1 or
                  :lml_id = f_lo_mgr_id(null, :rep_date_end, ps_from_c_id, ps_from_reg_id))
      into fact_kol_svz;
      fact_kol_svz = coalesce(fact_kol_svz, 0);

      --  План кол-во СВЗ за указанный период
      plan_kol_svz = null;
      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :mgr_c_id , 'PlanIndicators.LO_Count', :rep_date_end, :rep_period)
      into plan_kol_svz;
      plan_kol_svz = coalesce(plan_kol_svz, 0);

      --  фактическая сумма выдачи за указанные период
      fact_disb_amount = null;
      select sum(k_amount_approved)
      from kontstate, kontract, knaktlist, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = :reg_id and
            :rep_date_end between kal_date and kal_date_end and
            k_from_pr_id = pr_id and
            pr_code not containing '.express_online.' and
            (:mgr_c_id = -1 or
             :lml_id = f_lo_mgr_id(k_id, :rep_date_end, kal_new_kf_officer, kal_new_reg_id))
      into fact_disb_amount;
      fact_disb_amount = coalesce(fact_disb_amount, 0);
      fact_disb_amount = fact_disb_amount / 1000;

      --  Плановые выдачи (в целях бонуса)
      plan_disb_amount = null;
      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :mgr_c_id , 'PlanIndicators.Disb_amount_bonus', :rep_date_end, :rep_period)
      into plan_disb_amount;
      plan_disb_amount = coalesce(plan_disb_amount, 0);
      plan_disb_amount = plan_disb_amount / 1000;


      ostatok_del_bal_31_360 = fact_del_bal_31_360 - plan_del_bal_31_360;
      if (ostatok_del_bal_31_360 < 0) then ostatok_del_bal_31_360 = 0;

      ostatok_kol_svz = plan_kol_svz - fact_kol_svz;
      if (ostatok_kol_svz < 0) then ostatok_kol_svz = 0;

      ostatok_disb_amount = plan_disb_amount - fact_disb_amount;
      if (ostatok_disb_amount < 0) then ostatok_disb_amount = 0;


      -- фактический портфель
      fact_balance = 0;
      select sum(oks_balance) from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end
      and oks_close_state = 0
      and :rep_date_end - oks_delinq_date < 361
      and oks_region_id = :reg_id
      and (:mgr_c_id = -1 or
             :lml_id = f_lo_mgr_id(oks_from_k_id, :rep_date_end, oks_kc_id, oks_region_id))
      into fact_balance;
      fact_balance = coalesce(fact_balance, 0);
      fact_balance = fact_balance / 1000;

      -- план портфель
      plan_balance = null;
      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :mgr_c_id , 'PlanIndicators.Portfolio_amount', :rep_date_end, :rep_period)
      into plan_balance;
      plan_balance = coalesce(plan_balance, 0);
      plan_balance = plan_balance / 1000;

      ostatok_balance = plan_balance - fact_balance;
      if (ostatok_balance < 0) then ostatok_balance = 0;

      suspend;
    end
end
