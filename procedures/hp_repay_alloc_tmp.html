CREATE PROCEDURE HP_REPAY_ALLOC_TMP(
    ALLOC_DIRECTION VARCHAR(20),
    REP_DATE DATE,
    ORDER_OS INTEGER,
    ORDER_PERC INTEGER,
    ORDER_FINE INTEGER,
    ORDER_KOMMIS INTEGER,
    AMOUNT DECIMAL(18,4),
    REPAY_MODE INTEGER,
    SPECIAL_MODE INTEGER)
RETURNS (
    REP_SALDO DECIMAL(18,4))
AS
declare variable order_no double precision;
declare variable repay_os decimal(18,4) = 0;
declare variable repay_perc decimal(18,4) = 0;
declare variable repay_fine decimal(18,4) = 0;
declare variable zad_os decimal(18,4) = 0;
declare variable calcing_perc decimal(18,4) = 0;
declare variable zad_fine decimal(18,4);
declare variable order_kind integer;
declare variable repay_tax decimal(18,4);
declare variable repay_kommis decimal(18,4) = 0;
declare variable tax_tmp decimal(18,4) = 0;
declare variable zad_kommis decimal(18,4) = 0;
declare variable pr_pay_pros_more integer;
declare variable tax_perc decimal(18,7); /* Ставка НсП */
declare variable pr_round_perc integer;
declare variable pr_percent_method integer;
declare variable is_grace integer;
declare variable diff_os decimal(18,4) = 0;
declare variable zad_perc decimal(18,4) = 0;
declare variable local_saldo decimal(18,4) = 0;
declare variable zad_extra decimal(18,4) = 0;
declare variable repay_extra decimal(18,4) = 0;
declare variable zad_tax decimal(18,4) = 0;
declare variable total_zad decimal(18,4) = 0;
declare variable repay_koef double precision;
declare variable s_rec_no integer;
declare variable s_date date;
declare variable db_id integer;
declare variable max_kind d_integer;
begin
  if (coalesce(amount, 0) = 0) then
    exit;
  db_id = f_get_db_id();

  max_kind = 999;
  if (alloc_direction = 'Proportional_ipf') then
  begin
    alloc_direction = 'Proportional';
    max_kind = 2;
  end

  if (not(alloc_direction in ('Horizontal', 'Vertical', 'Extra', 'Proportional'))) then
    exception e_default 'Illegal repay allocation method!';

-- Special_Mode=1 - Особое распределение просрочки с начала
-- Special_Mode=0 - Обычное распределение с начала
-- Special_Mode=-1 - Обычное распределение с конца
  if (special_mode = -1) then
  begin
    special_mode = 0;
    order_kind = -1;
  end
  else
    order_kind = 1;

  select pr_pay_pros_more, pr_round_perc, tax_perc, pr_percent_method
  from temp_params
  into :pr_pay_pros_more, :pr_round_perc, :tax_perc, :pr_percent_method;

  if (repay_mode > 1) then
    pr_pay_pros_more = 0;

  if (db_id = 39801) then
    pr_pay_pros_more = 999999;

-- Вычисляем остаточный баланс, чтоб распределение на ОС не было больше этой суммы
  select sum(zad_os - pay_os)
  from temp_vrepay
  into :local_saldo;

  rep_saldo = amount; -- Сумма для распределения

  if (alloc_direction = 'Extra') then
  begin
    for select s_rec_no, zad_extra
        from temp_vrepay
        where zad_extra <> 0
        order by s_id
        into s_rec_no, :zad_extra
-- as cursor cur
    do
    begin
      repay_extra = 0;

-- Распределение доп.обязательств
      if (rep_saldo > 0) then
      begin
        repay_extra = minvalue(rep_saldo, zad_extra);
        rep_saldo = rep_saldo - repay_extra;
        update temp_vrepay
        set pay_extra = :repay_extra
        where s_rec_no = :s_rec_no;
      end
    end
  end

  if (alloc_direction = 'Proportional') then
  begin
    update temp_vrepay
    set pay_os = 0,
        pay_perc = 0,
        pay_fine = 0,
        pay_tax = 0
    where s_date is not null;

    -- Распределение погашений пропорционально на ОС, наценку, НсП и пеня
    for select s_rec_no, zad_os, calcing_perc, zad_fine
        from temp_vrepay
        where repay_mode <= :max_kind
        order by s_id
        into s_rec_no, zad_os, calcing_perc, zad_fine
    do
    begin
      if (rep_saldo > 0) then
      begin
        zad_tax = (calcing_perc * tax_perc) / 100.00; -- вычисляем НсП от процента
        zad_tax = round(zad_tax, pr_round_perc);
        total_zad = zad_os + calcing_perc + zad_fine + zad_tax;

        repay_os = 0;
        repay_perc = 0;
        repay_fine = 0;
        repay_tax = 0;

        if (total_zad <= rep_saldo) then
        begin
          -- если общая сумма в платеже не больше чем распределяемая сумма, то берем все суммы
          repay_os = zad_os;
          repay_perc = calcing_perc;
          repay_fine = zad_fine;
          repay_tax = zad_tax;
        end
        else
        begin
          -- Иначе берем пропорциоанально
          repay_koef = rep_saldo / total_zad;
          repay_perc = round(calcing_perc * repay_koef, pr_round_perc);
          if (abs(repay_perc - calcing_perc) <= 1) then
            repay_perc = calcing_perc;

          repay_tax = round(zad_tax * repay_koef, pr_round_perc);
          if (abs(repay_tax - zad_tax) <= 1) then
            repay_tax = zad_tax;

          repay_fine = round(zad_fine * repay_koef, pr_round_perc);
          if (abs(repay_fine - zad_fine) <= 1) then
            repay_fine = zad_fine;

          repay_os = rep_saldo - (repay_perc + repay_fine + repay_tax);
        end

        update temp_vrepay
        set pay_os = :repay_os,
            pay_perc = :repay_perc,
            pay_fine = :repay_fine,
            pay_tax = :repay_tax
        where s_rec_no = :s_rec_no;
        rep_saldo = rep_saldo - (repay_os + repay_perc + repay_fine + repay_tax);
      end
    end

    if (max_kind = 2 and rep_saldo > 0) then
    begin
      -- На будущие платежи распределяем только на ОС
      calcing_perc = 0;
      zad_fine = 0;
      zad_tax = 0;
      repay_perc = 0;
      repay_fine = 0;
      repay_tax = 0;
  
      for select s_rec_no, zad_os
          from temp_vrepay
          where repay_mode = 3
          order by s_id desc
          into s_rec_no, zad_os
      do
      begin
        if (rep_saldo > 0) then
        begin
          repay_os = 0;
  
          if (zad_os <= rep_saldo) then
            repay_os = zad_os;
          else
            repay_os = rep_saldo;
  
          update temp_vrepay
          set pay_os = :repay_os
          where s_rec_no = :s_rec_no;
  
          rep_saldo = rep_saldo - repay_os;
        end
      end
    end
  end

  if (alloc_direction = 'Horizontal') then
  begin
    for select s_rec_no, zad_os, calcing_perc, zad_fine, zad_kommis, is_grace, zad_perc
        from temp_vrepay
        where ((datediff(day, coalesce(s_date, cast('01.01.1900' as date)), :rep_date) > :pr_pay_pros_more and
              :special_mode = 1) -- igusev 22.09.2018 добавил coalesce(s_date, cast('01.01.1900' as date)) #operkroff_pay
              or
              (datediff(day, coalesce(s_date, cast('01.01.1900' as date)), :rep_date) <= :pr_pay_pros_more and
              :special_mode = 0)) -- igusev 22.09.2018 добавил coalesce(s_date, cast('01.01.1900' as date)) #operkroff_pay
              and
              repay_mode = :repay_mode
        order by s_rec_no * :order_kind
        into s_rec_no, zad_os, calcing_perc, zad_fine, zad_kommis, is_grace, zad_perc
-- as cursor cur
    do
    begin
      order_no = 0;
      repay_tax = 0;
      repay_kommis = 0;
      repay_os = 0;
      repay_perc = 0;
      repay_fine = 0;
      while (order_no < 4) do
      begin
        order_no = order_no + 1;

-- Распределение основной суммы
        if ((rep_saldo > 0) and
            (order_no = order_os)) then
        begin
-- контролируем чтоб на ОС не распределилось больше чем есть в балансе
          repay_os = minvalue(minvalue(rep_saldo, local_saldo), zad_os);
          rep_saldo = rep_saldo - repay_os;
          update temp_vrepay
          set pay_os = :repay_os
          where s_rec_no = :s_rec_no;

-- уменьшаем баланс на уже распределенную часть ОС
          local_saldo = local_saldo - repay_os;

-- Если метод аннуитета и ОС распределяется после %%,
-- или льготное погашение
-- то разницу между накопленным %% и по графику перекидываем на ОС
          if (((pr_percent_method = 2) and
              (order_os > order_perc)) or ((is_grace = 1) and
              (order_os > order_perc))) then
          begin
            diff_os = zad_perc - calcing_perc;
            if ((diff_os > 0) and
                (rep_saldo > 0)) then
            begin
              repay_os = minvalue(minvalue(rep_saldo, local_saldo), diff_os);
              rep_saldo = rep_saldo - repay_os;
              update temp_vrepay
              set pay_os = pay_os + :repay_os
              where s_rec_no = :s_rec_no;
              local_saldo = local_saldo - repay_os;
            end
          end
        end

-- Распределение процентов
        if ((rep_saldo > 0) and
            (order_no = order_perc)) then
        begin
          repay_tax = 0;
          if (tax_perc > 0) then -- если есть НсП, вычисляем
          begin
            tax_tmp = (calcing_perc * tax_perc) / 100.00; -- вычисляем НсП от процента
            tax_tmp = round(tax_tmp, pr_round_perc);
            if (calcing_perc + tax_tmp <= rep_saldo) then -- если %%+НсП больше остатка, берем эту НсП
            begin
              repay_tax = tax_tmp;
              repay_perc = calcing_perc;
            end
            else
            begin
-- Иначе от суммы остатка выделяем НсП, остальное на %%
              tax_tmp = (rep_saldo / (cast((100.00 + tax_perc) as double precision))) * tax_perc;
              repay_tax = round(tax_tmp, pr_round_perc);
              repay_perc = rep_saldo - repay_tax;
            end
          end
          else
            repay_perc = minvalue(rep_saldo, calcing_perc);

          rep_saldo = rep_saldo - repay_perc - repay_tax;

          update temp_vrepay
          set pay_perc = :repay_perc,
              pay_tax = :repay_tax
          where s_rec_no = :s_rec_no;

          -- Если метод аннуитета и ОС распределяется ДО %%,
          -- или льготное погашение
          -- то разницу между накопленным %% и по графику перекидываем на ОС
          if (pr_percent_method = 2 and order_os < order_perc and db_id = 39801) then
          begin
            diff_os = zad_perc - calcing_perc;
            if ((diff_os > 0) and
                (rep_saldo > 0)) then
            begin
              repay_os = minvalue(minvalue(rep_saldo, local_saldo), diff_os);
              rep_saldo = rep_saldo - repay_os;
              update temp_vrepay
              set pay_os = pay_os + :repay_os
              where s_rec_no = :s_rec_no;
              local_saldo = local_saldo - repay_os;
            end
          end
        end

-- Распределение комиссии за досрочное погашение
        if ((rep_saldo > 0) and
            (order_no = order_kommis)) then
        begin
          repay_tax = 0;
          repay_kommis = 0;
          if (tax_perc > 0) then -- если есть НсП, вычисляем
          begin
            tax_tmp = (zad_kommis * tax_perc) / 100.00; -- вычисляем НсП на комиссию за досрочное погашение
            tax_tmp = round(tax_tmp, pr_round_perc);
            if (zad_kommis + tax_tmp <= rep_saldo) then -- если %%+НсП больше остатка, берем эту НсП
            begin
              repay_tax = tax_tmp;
              repay_kommis = zad_kommis;
            end
            else
            begin
-- Иначе от суммы остатка выделяем НсП, остальное на %%
              tax_tmp = (rep_saldo / (cast((100.00 + tax_perc) as double precision))) * tax_perc;
              repay_tax = round(tax_tmp, pr_round_perc);
              repay_kommis = rep_saldo - repay_tax;
            end
          end
          else
            repay_kommis = minvalue(rep_saldo, zad_kommis);

          rep_saldo = rep_saldo - repay_kommis - repay_tax;
          update temp_vrepay
          set pay_kommis = :repay_kommis,
              pay_tax = pay_tax + :repay_tax
          where s_rec_no = :s_rec_no;
        end

-- Распределение пени
        if ((rep_saldo > 0) and
            (order_no = order_fine)) then
        begin
          repay_fine = minvalue(rep_saldo, zad_fine);
          rep_saldo = rep_saldo - repay_fine;
          update temp_vrepay
          set pay_fine = :repay_fine
          where s_rec_no = :s_rec_no;
        end

      end
    end
  end

  if (alloc_direction = 'Vertical') then
  begin
    order_no = 0;
    while (order_no < 4) do
    begin
      order_no = order_no + 1;
      for select s_rec_no, zad_os, calcing_perc, zad_fine, zad_kommis, is_grace, zad_perc, s_date
          from temp_vrepay
          where ((datediff(day, coalesce(s_date, cast('01.01.1900' as date)), :rep_date) > :pr_pay_pros_more and
                :special_mode = 1) -- igusev 22.09.2018 добавил coalesce(s_date, cast('01.01.1900' as date)) #operkroff_pay
                or
                (datediff(day, coalesce(s_date, cast('01.01.1900' as date)), :rep_date) <= :pr_pay_pros_more and
                :special_mode = 0)) -- igusev 22.09.2018 добавил coalesce(s_date, cast('01.01.1900' as date)) #operkroff_pay
                and
                repay_mode = :repay_mode
          order by s_rec_no * :order_kind
          into s_rec_no, :zad_os, calcing_perc, zad_fine, zad_kommis, is_grace, zad_perc, s_date
-- as cursor cur
      do
      begin
        repay_tax = 0;
        repay_kommis = 0;
        repay_os = 0;
        repay_perc = 0;
        repay_fine = 0;

-- Распределение основной суммы
        if ((rep_saldo > 0) and
            (order_no = order_os)) then
        begin
-- контролируем чтоб на ОС не распределилось больше чем есть в балансе
          repay_os = minvalue(minvalue(rep_saldo, local_saldo), zad_os);
          rep_saldo = rep_saldo - repay_os;
          update temp_vrepay
          set pay_os = :repay_os
          where s_rec_no = :s_rec_no;

-- уменьшаем баланс на уже распределенную часть ОС
          local_saldo = local_saldo - repay_os;

-- Если метод аннуитета и ОС распределяется после %%,
-- или льготное погашение
-- то разницу между накопленным %% и по графику перекидываем на ОС
          if (((pr_percent_method = 2) and
              (order_os > order_perc)) or ((is_grace = 1) and
              (order_os > order_perc))) then
          begin
            diff_os = zad_perc - calcing_perc;
            if ((diff_os > 0) and
                (rep_saldo > 0)) then
            begin
              repay_os = minvalue(minvalue(rep_saldo, local_saldo), diff_os);
              rep_saldo = rep_saldo - repay_os;
              update temp_vrepay
              set pay_os = pay_os + :repay_os
              where s_rec_no = :s_rec_no;
              local_saldo = local_saldo - repay_os;
            end
          end
        end

-- Распределение процентов
        if ((rep_saldo > 0) and
            (order_no = order_perc)) then
        begin
          if (tax_perc > 0) then -- если есть НсП, вычисляем
          begin
            tax_tmp = 0;
            if (not(db_id in (39801, 41701) and s_date is null)) then  -- Если внесистемный учет и КГ база то на погашенные проценты НСП не будет, добавлено от 03.06.2019
            tax_tmp = (calcing_perc * tax_perc) / 100.00; -- вычисляем НсП от процента
            tax_tmp = round(tax_tmp, pr_round_perc);
            if (calcing_perc + tax_tmp <= rep_saldo) then -- если %%+НсП больше остатка, берем эту НсП
            begin
              repay_tax = tax_tmp;
              repay_perc = calcing_perc;
            end
            else
            begin
-- Иначе от суммы остатка выделяем НсП, остальное на %%
              if (not(db_id in (39801, 41701) and s_date is null)) then -- Если внесистемный учет и КГ база то на погашенные проценты НСП не будет, добавлено от 03.06.2019
              tax_tmp = (rep_saldo / (cast((100.00 + tax_perc) as double precision))) * tax_perc;
              repay_tax = round(tax_tmp, pr_round_perc);
              repay_perc = rep_saldo - repay_tax;
            end
          end
          else
            repay_perc = minvalue(rep_saldo, calcing_perc);

          rep_saldo = rep_saldo - repay_perc - repay_tax;
--          repay_tax = 0; -- 4
          update temp_vrepay
          set pay_perc = :repay_perc,
              pay_tax = :repay_tax
          where s_rec_no = :s_rec_no;
        end

-- Распределение комиссии за досрочное погашение
        if ((rep_saldo > 0) and
            (order_no = order_kommis)) then
        begin
          repay_tax = 0;
          repay_kommis = 0;
          if (tax_perc > 0) then -- если есть НсП, вычисляем
          begin
            tax_tmp = (zad_kommis * tax_perc) / 100.00; -- вычисляем НсП на комиссию за досрочное погашение
            tax_tmp = round(tax_tmp, pr_round_perc);
            if (zad_kommis + tax_tmp <= rep_saldo) then -- если %%+НсП больше остатка, берем эту НсП
            begin
              repay_tax = tax_tmp;
              repay_kommis = zad_kommis;
            end
            else
            begin
-- Иначе от суммы остатка выделяем НсП, остальное на %%
              tax_tmp = (rep_saldo / (cast((100.00 + tax_perc) as double precision))) * tax_perc;

              repay_tax = round(tax_tmp, pr_round_perc);
              repay_kommis = rep_saldo - repay_tax;
            end
          end
          else
            repay_kommis = minvalue(rep_saldo, zad_kommis);

          rep_saldo = rep_saldo - repay_kommis - repay_tax;
          update temp_vrepay
          set pay_kommis = :repay_kommis,
              pay_tax = pay_tax + :repay_tax
          where s_rec_no = :s_rec_no;
        end

-- Распределение пени
        if ((rep_saldo > 0) and
            (order_no = order_fine)) then
        begin
          repay_fine = minvalue(rep_saldo, zad_fine);
          rep_saldo = rep_saldo - repay_fine;
          update temp_vrepay
          set pay_fine = :repay_fine
          where s_rec_no = :s_rec_no;
        end
      end
    end
  end

  suspend;
end
