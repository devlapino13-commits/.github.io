CREATE PROCEDURE FP_CLIENT_QUESTIONNAIRE(
    KC_ID ID_TYPE)
RETURNS (
    QU_KC_ID ID_TYPE,
    QU_K_ID ID_TYPE,
    QU_C_ID ID_TYPE,
    QU_ID INTEGER,
    QU_NAME D_BLOB_TEXT,
    QU_TYPE INTEGER,
    AN_ONE STR_100,
    AN_TWO STR_100,
    AN_ID INTEGER,
    AN_OTHER STR_255)
AS
declare variable nr char(1);
declare variable cm_name str_255;
begin
  select cm.cm_name
  from company cm
  into :cm_name;

  qu_kc_id = kc_id;
  if (kc_id > 0) then begin
    select
      kc_from_k_id,
      kc_from_c_id
    from knclient
    where kc_id = :kc_id
    into
      qu_k_id,
      qu_c_id;
  end else begin
    select
      z_from_k_id,
      z_from_c_id
    from zalog
    where z_id = - :kc_id
    into
      qu_k_id,
      qu_c_id;
  end

  nr = ascii_char(13);
  for with lst
      as (select 1 as qu_id,
                 'Являетесь ли Вы :' || :nr || '     Российским публичным должностным лицом ' || 'лицом, замещающих (занимающих) государственные ' || 'должности РФ, должности членов Совета директоров ЦБ ' || 'РФ, должности федеральной государственной службы, ' || 'назначение на которые и освобождение от которых ' || 'осуществляются Президентом РФ или Правительством ' || 'РФ, должности в ЦБ РФ, государственных корпорациях и ' || 'иных организациях, созданных на основании ' || 'федеральных законов, включенные в перечни ' || 'должностей, определяемые Президентом Российской ' || 'Федерации (далее - РПДЛ) ?' as qu_name,
                 0 as qu_type, 'Не являюсь' as an_one, 'Являюсь' as an_two
          from rdb$database
          union
          select 2 as qu_id,
                 'Являетесь ли Вы :' || :nr || '     Международным публичным должностным ' || 'лицом – международным гражданским служащим или ' || 'любым лицом, которое уполномочено международной ' || 'организацией действовать от ее имени (далее - МПДЛ) ?' as qu_name,
                 0 as qu_type, 'Не являюсь' as an_one, 'Являюсь' as an_two
          from rdb$database
          union
          select 3 as qu_id,
                 'Являетесь ли Вы :' || :nr || '     Иностранным публичным должностным лицом – ' || 'лицом назначаемым или избираемым, занимающим ' || 'какую-либо должность в законодательном, ' || 'исполнительном, административном или судебном ' || 'органе иностранного государства, или лицом, ' || 'выполняющим какую-либо публичную функцию для ' || 'иностранного государства, в том числе для публичного ' || 'ведомства или государственного предприятия, либо ' || 'лицом, ранее занимавшим публичную должность, с ' || 'момента сложения полномочий которого прошло менее ' || '1 года (далее - ИПДЛ) ?' as qu_name,
                 0 as qu_type, 'Не являюсь' as an_one, 'Являюсь' as an_two
          from rdb$database
          union
          select 4 as qu_id,
                 'Являетесь ли Вы родственником лица ,принадлежащего ' || 'к одной из перечисленных категорий (РПДЛ, МПДЛ, ' || 'ИПДЛ) ?' as qu_name,
                 0 as qu_type, 'Не являюсь' as an_one, 'Являюсь' as an_two
          from rdb$database
          union
          select 5 as qu_id,
                 'Действуете ли Вы от имени лица, принадлежащего к ' || 'одной из вышеперечисленных категорий (РПДЛ, МПДЛ, ' || 'ИПДЛ) ?' as qu_name,
                 0 as qu_type, 'Нет' as an_one, 'Да' as an_two
          from rdb$database
          union
          select 6 as qu_id,
                 'Имеется ли у Вас выгодоприобретатель – лицо, к ' || 'выгоде которого Вы действуете при проведении ' || 'операций с денежными средствами или иным ' || 'имуществом ?' as qu_name,
                 0 as qu_type, 'Нет' as an_one, 'Да' as an_two
          from rdb$database
          union
          select 7 as qu_id,
                 'Имеются ли у Вас бенефициарные владельцы – ' || 'физические лица, которые в конечном счете прямо или ' || 'косвенно (через третьих лиц) имеют возможность ' || 'контролировать Ваши  действия ?' as qu_name,
                 0 as qu_type, 'Нет' as an_one, 'Да' as an_two
          from rdb$database
          union
          select 8 as qu_id,
                 'Цели установления отношений с '||:cm_name as qu_name,
                 1 as qu_type, 'Получение микрозайма' as an_one, 'Иное:' as an_two
          from rdb$database
          union
          select 9 as qu_id,
                 'Предполагаемый характер деловых отношений с ' || :cm_name as qu_name,
                 0 as qu_type,
                 'Долгосрочное сотрудничество для ' || 'периодического получения заемных средств' as an_one,
                 'Разовое обращение за микрозаймом' as an_two
          from rdb$database
          union
          select 10 as qu_id,
                 'Цели финансово-хозяйственной деятельности' as qu_name,
                 1 as qu_type, 'Отсутствуют' as an_one, 'Имеются. Какие:' as an_two
          from rdb$database
          union
          select 11 as qu_id, 'Финансовое положение' as qu_name, 1 as qu_type,
                 'Устойчивое' as an_one, 'Иное:' as an_two
          from rdb$database
          union
          select 12 as qu_id,
                 'Деловая репутация: Имеются ли в отношении Вас ' || 'факторы, негативно влияющие на вашу деловую ' || 'репутацию. Например, участие в противоправной ' || 'деятельности, просроченные обязательства, ' || 'возбужденные против Вас административные, ' || 'следственные либо судебные процедуры, и т.п.' as qu_name,
                 1 as qu_type, 'Нет' as an_one, 'Да. Указать подробнее:' as an_two
          from rdb$database
          union
          select 13 as qu_id,
                 'Источник происхождения денежных средств и(или) ' || 'иного имущества' as qu_name,
                 1 as qu_type, 'Заработная плата' as an_one, 'Иное:' as an_two
          from rdb$database)
      select qu_id, qu_name, qu_type, an_one, an_two, iif(fqa_id is null, 0, fqa_an_id),
             iif(fqa_id is null, '', fqa_an_other)
      from lst
      left join fp_question_anketa on fqa_from_k_id = :qu_k_id and
            fqa_from_c_id = :qu_c_id and
            fqa_qu_id = qu_id
      into :qu_id, :qu_name, :qu_type, :an_one, :an_two, :an_id, :an_other
  do
  begin
    suspend;
  end
end
