CREATE PROCEDURE AAREP_MATURITY_REPORT_PR_KIND(
    REG_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    YEAR_NO INTEGER,
    MONTH_NO INTEGER,
    DIFF_NO INTEGER,
    PLAN_OS DECIMAL(18,4),
    K_BALANCE DECIMAL(18,4),
    PLAN_OS_1 DECIMAL(18,4),
    K_BALANCE_1 DECIMAL(18,4),
    PLAN_OS_2 DECIMAL(18,4),
    K_BALANCE_2 DECIMAL(18,4),
    PLAN_OS_3 DECIMAL(18,4),
    K_BALANCE_3 DECIMAL(18,4),
    PLAN_OS_4 DECIMAL(18,4),
    K_BALANCE_4 DECIMAL(18,4),
    REPAY_NAME VARCHAR(50),
    MONTH_END DATE)
AS
declare variable tmp_llc decimal(18,4);
declare variable tmp_balance decimal(18,4);
declare variable tmp_balance_1 decimal(18,4);
declare variable tmp_balance_2 decimal(18,4);
declare variable tmp_balance_3 decimal(18,4);
declare variable tmp_balance_4 decimal(18,4);
begin
  for select rep_year, rep_month, rep_line, sum(sh_os), sum(kont_balance), sum(
             case pr_product_kind
               when 1 then sh_os
               else 0
             end),
             sum(
             case pr_product_kind
               when 1 then kont_balance
               else 0
             end),
             sum(
             case pr_product_kind
               when 2 then sh_os
               else 0
             end),
             sum(
             case pr_product_kind
               when 2 then kont_balance
               else 0
             end),
             sum(
             case pr_product_kind
               when 3 then sh_os
               else 0
             end),
             sum(
             case pr_product_kind
               when 3 then kont_balance
               else 0
             end),
             sum(
             case pr_product_kind
               when 4 then sh_os
               else 0
             end),
             sum(
             case pr_product_kind
               when 4 then kont_balance
               else 0
             end)

      from aarep_maturity_rep_tmp(:rep_date)
      group by rep_year, rep_month, rep_line
      order by rep_line
      into :year_no, :month_no, :diff_no, :plan_os, :tmp_balance, :plan_os_1, :tmp_balance_1, :plan_os_2,
           :tmp_balance_2, :plan_os_3, :tmp_balance_3, :plan_os_4, :tmp_balance_4

  do
  begin
    if (diff_no = -1) then
    begin
      k_balance = tmp_balance;
      k_balance_1 = tmp_balance_1;
      k_balance_2 = tmp_balance_2;
      k_balance_3 = tmp_balance_3;
      k_balance_4 = tmp_balance_4;
    end
    else
    begin
      k_balance = k_balance - plan_os;
      k_balance_1 = k_balance_1 - plan_os_1;
      k_balance_2 = k_balance_2 - plan_os_2;
      k_balance_3 = k_balance_3 - plan_os_3;
      k_balance_4 = k_balance_4 - plan_os_4;
    end

    if (diff_no = -1) then
      repay_name = 'Показатели на начало';
    if (diff_no = 0) then
      repay_name = 'Погашение просроченных';
    if (diff_no > 0) then
      repay_name = 'Через ' || diff_no || ' мес.';

    if (month_no < 12) then
      month_end = cast(('01.' ||(month_no + 1) || '.' || year_no) as date) - 1;
    else
      month_end = cast(('31.12.' || year_no) as date);
    suspend;
  end
end
