CREATE PROCEDURE HP_FUELGENREG_LOAD(
    ID D_INTEGER,
    UN_ID D_ID = null,
    REG_ID D_INTEGER = null)
RETURNS (
    FG_ID D_INTEGER,
    FG_DTWRITEOFF D_DATE,
    FG_DTBEG D_DATE,
    FG_DTEND D_DATE,
    FG_OFFICE_ID D_INTEGER,
    FG_PET_ID D_INTEGER,
    PET_NAME D_STR_255,
    PET_LABEL D_STR_CODE,
    PET_IS_RENT D_INTEGER,
    FG_START D_CURRENCY,
    FG_START_SUM D_CURRENCY,
    FG_FILLUP D_CURRENCY,
    FG_FILLUP_SUM D_CURRENCY,
    FG_WHOURS D_CURRENCY,
    FG_WRATE D_CURRENCY,
    FG_USEDUP D_CURRENCY,
    FG_USEDUP_SUM D_CURRENCY,
    FG_END D_CURRENCY,
    FG_END_SUM D_CURRENCY,
    FG_DRIVER_ID D_ID,
    DRIVER_NAME D_STR_100,
    FG_MANAGER_ID D_ID,
    MANAGER_NAME D_STR_100,
    FG_BOOKER_ID D_ID,
    BOOKER_NAME D_STR_100,
    FG_MANAGER_DTSIGN D_DATETIME,
    FG_BOOKER_DTSIGN D_DATETIME,
    FG_DRIVER_EDL_ID D_ID,
    FG_MANAGER_EDL_ID D_ID,
    FG_BLS_ID D_ID,
    FG_STATUS D_INTEGER,
    FG_STATE D_STR_20)
AS
begin
  if (id > 0) then
  begin
    select
      fg_id,
      fg_dtwriteoff,
      fg_dtbeg,
      fg_dtend,
      fg_office_id,
      fg_pet_id,
      pet_name,
      pet_label,
      pet_is_rent,
      fg_start,
      fg_start_sum,
      fg_fillup,
      fg_fillup_sum,
      fg_whours,
      fg_wrate,
      fg_usedup,
      fg_usedup_sum,
      fg_end,
      fg_end_sum,
      fg_driver_id,
      fg_manager_id,
      fg_booker_id,
      fg_manager_dtsign,
      fg_booker_dtsign,
      fg_driver_edl_id,
      fg_manager_edl_id,
      fg_bls_id,
      fg_status
    from fuelgenreg
      inner join pettrans
        on pet_id = fg_pet_id
    where fg_id = :id
    into
      :fg_id,
      :fg_dtwriteoff,
      :fg_dtbeg,
      :fg_dtend,
      :fg_office_id,
      :fg_pet_id,
      :pet_name,
      :pet_label,
      :pet_is_rent,
      :fg_start,
      :fg_start_sum,
      :fg_fillup,
      :fg_fillup_sum,
      :fg_whours,
      :fg_wrate,
      :fg_usedup,
      :fg_usedup_sum,
      :fg_end,
      :fg_end_sum,
      :fg_driver_id,
      :fg_manager_id,
      :fg_booker_id,
      :fg_manager_dtsign,
      :fg_booker_dtsign,
      :fg_driver_edl_id,
      :fg_manager_edl_id,
      :fg_bls_id,
      :fg_status;
    driver_name = coalesce((select c_name_fio from clients where c_id = :fg_driver_id), '');
    manager_name = coalesce((select c_name_fio from clients where c_id = :fg_manager_id), '');
  end else
  begin
    fg_id = 0;
    fg_dtend = current_date - extract(day from current_date);
    fg_dtbeg = fg_dtend - extract(day from fg_dtend) + 1;
    fg_dtwriteoff = fg_dtend;
    fg_office_id = reg_id;
    fg_pet_id = 0;
    pet_name = '';
    pet_label = '';
    fg_start = 0;
    fg_start_sum = 0;
    fg_fillup = 0;
    fg_fillup_sum = 0;
    fg_whours = 0;
    fg_wrate = 0;
    fg_usedup = 0;
    fg_usedup_sum = 0;
    fg_end = 0;
    fg_end_sum = 0;
    fg_booker_id = (select un_from_c_id from username where un_id = :un_id);
    fg_driver_id = 0;
    fg_manager_id = 0;
    fg_driver_edl_id = null;
    fg_manager_edl_id = null;
    fg_status = 0;
    driver_name = '';
    manager_name = '';
  end
  booker_name = coalesce((select c_name_fio from clients where c_id = :fg_booker_id), '');
  fg_state =
    case fg_status
      when -1 then 'аннулирован'
      when  0 then 'создан'
      when  1 then 'подписан'
    end;
  suspend;
end
