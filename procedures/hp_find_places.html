CREATE PROCEDURE HP_FIND_PLACES(
    PARENT_ID DECIMAL(18,0),
    FIND_PLACE VARCHAR(255),
    FIND_STREET VARCHAR(255))
RETURNS (
    PL_ID DECIMAL(18,0),
    FULL_TEXT VARCHAR(255),
    PL_NAME VARCHAR(255),
    PL_NAME_KOMITET VARCHAR(255),
    PL_STATUS_NAME VARCHAR(100),
    PL_CAN_SELECT INTEGER,
    STREET_ID DECIMAL(18,0),
    STREET_NAME VARCHAR(255),
    STREET_ALTERNATE_NAME VARCHAR(255),
    STREE_KIND VARCHAR(100),
    STREET_NAME_ORIG VARCHAR(255),
    PL_POST_INDEX VARCHAR(20))
AS
declare variable id_list decimal(18,0);
declare variable pl_parent_id decimal(18,0);
declare variable pl_post_index_tmp varchar(20);
begin
-- Ищет населенный пункты и улицы - зависит от входящего параметра

  pl_post_index = '';
  street_name_orig = '';
  for select id_list
      from hp_find_places_tmp(:parent_id, :find_place, :find_street)
      rows 300
      into :id_list
  do
  begin
    for select pl_id,
               (
               case plt_short_name_side
                 when -1 then plt_short_name || ' ' || pl_name
                 when 0 then pl_name
                 when 1 then pl_name || ' ' || plt_short_name
               end),
               pl_name_komitet, plt_name, parent, plt_can_select, pl_name, pl_post_index
        from places, placetype
        where plt_id = pl_status and
              pl_id = :id_list
        into :street_id, street_name, street_alternate_name, stree_kind, pl_parent_id, pl_can_select, street_name_orig,
             pl_post_index
    do
    begin
      full_text = '';
      if (pl_post_index is null) then
        pl_post_index = '';
      if (pl_parent_id is null) then
        pl_can_select = -1;
      if (pl_can_select = 2) then
      begin
        select pl_id,
               (
               case plt_short_name_side
                 when -1 then plt_short_name || ' ' || pl_name
                 when 0 then pl_name
                 when 1 then pl_name || ' ' || plt_short_name
               end),
               pl_name_komitet, plt_name, pl_post_index
        from places, placetype
        where pl_id = :pl_parent_id and
              pl_status = plt_id
        into :pl_id, :pl_name, :pl_name_komitet, :pl_status_name, pl_post_index_tmp;
        if (pl_post_index = '') then
          pl_post_index = pl_post_index_tmp;
        if (pl_post_index is null) then
          pl_post_index = '';
      end
      else
      begin
        street_name_orig = '';
        pl_id = :street_id;
        pl_name = street_name;
        pl_name_komitet = street_alternate_name;
        pl_status_name = stree_kind;
        street_id = 0;
        street_name = '';
        street_alternate_name = '';
        stree_kind = '';
      end
      pl_name = replace(pl_name, '. ', '.');
      street_name = replace(street_name, '. ', '.');
      select place_full_region
      from aarep_places_link(:pl_id)
      into :full_text;

      suspend;
    end
  end
end
