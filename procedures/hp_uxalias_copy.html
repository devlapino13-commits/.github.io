CREATE PROCEDURE HP_UXALIAS_COPY(
    ORIG_UXA_CODE D_STR_CODE,
    NEW_UXA_CODE D_STR_CODE,
    NEW_UXA_NAME D_STR_CODE,
    UXA_KIND D_SMALLINT,
    INCLUDE_ACTIONS D_BOOL)
RETURNS (
    UXA_CODE D_STR_100)
AS
begin
  insert into uxalias (uxa_code, uxa_ux_code, uxa_name, uxa_kind)
  select :new_uxa_code, uxa_ux_code, :new_uxa_name, :uxa_kind
  from uxalias
  where uxa_code = :orig_uxa_code
  returning uxa_code
  into uxa_code;

  if (include_actions = 1) then
  begin
    insert into uxaliasaction (uaa_uxa_code, uaa_ua_code)
    select :uxa_code, uaa_ua_code
    from uxaliasaction
    where uaa_uxa_code = :orig_uxa_code;
  end

  suspend;
end
