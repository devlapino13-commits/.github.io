CREATE PROCEDURE AAREP_NBKR_NOTIF_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    KONT_STATUS D_STR_100,
    CATEG D_STR_100,
    NNL_AMOUNT D_CURRENCY,
    PREDPISANIYE_STATUS D_STR_100,
    HAS_ASP SMALLINT,
    KS_FIO D_STR_100,
    CLIENT_STATUS D_STR_100,
    INDEXED_ADR D_STR_255,
    CREATE_DATETIME D_DATETIME)
AS
declare variable config d_str_100;
declare variable edc_id d_integer;
declare variable c_id d_id;
declare variable kc_id d_id;
declare variable kal_new_kf_officer d_id;
declare variable edoc_status d_smallint;
begin
  predpisaniye_status = null;
  edoc_status = null;

  for select reg.reg_name, office.reg_name, kal.kal_new_kf_name, kont.k_kontract_no, kont.k_disb_date, cl.c_name_fio,
             iif(ks.ks_state = 1, 'Активный', 'Закрытый'), t.nnl_amount, nnl_from_edc_id, cl.c_id,
             kc_id, kal_new_kf_officer
      from nbkr_notif_list t
      inner join kontract kont on kont.k_id = t.nnl_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      inner join region_list(:rep_region_id, 1, 3, 1) rl on cast(rl.id_list as d_id) = kal.kal_new_reg_id
      inner join region office on office.reg_unicode = rl.id_list
      inner join region reg on reg.reg_unicode = office.parent
      inner join knclient kncl on kncl.kc_id = t.nnl_from_kc_id
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      inner join kontstate ks on ks.ks_from_k_id = kont.k_id
      where kal.is_delete = 0 and
            :rep_date between kal.kal_date and kal.kal_date_end and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0
      into :reg_name, :office_name, :kf_name, :k_kontract_no, :k_disb_date, :c_name_fio, :kont_status, :nnl_amount,
           :edc_id, :c_id, :kc_id, :kal_new_kf_officer
  do
  begin
    ks_fio = '';

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :kal_new_kf_officer
    into :ks_fio;

    has_asp = 0;
    if (exists(select *
               from hp_edoc_status('asp.agreement', 0, :c_id)
               where client_sign_status = 1)) then
      has_asp = 1;

    categ = 'не определено';
    config = '';

    if (edc_id = 21) then
    begin
      categ = 'Закрытый; Нам должны';
      config = 'nbkr.cli_close_we_owe';
    end
    else
    if (edc_id = 22) then
    begin
      categ = 'Активный; Нам должны';
      config = 'nbkr.cli_active_we_owe';
    end
    else
    if (edc_id = 23) then
    begin
      categ = 'Закрытый; Мы должны';
      config = 'nbkr.cli_close_we_must';
    end
    else
    if (edc_id = 24) then
    begin
      categ = 'Активный; Мы должны';
      config = 'nbkr.cli_active_we_must';
    end

    predpisaniye_status = 'СМС еще отправлялся';

    if (edc_id is not null) then
    begin
      select edoc_status_str, edoc_status
      from hp_edoc_status(:config, 8, :kc_id)
      into :predpisaniye_status, :edoc_status;
      create_datetime = null;
      if (:edoc_status <> 1) then
      begin
        select cds.cds_create_datetime
        from clientdocs cds, passkind, passdoctype
        where cds_from_c_id = :c_id and
              cds_from_pk_id = pk_id and
              pk_from_pdt_id = pdt_id and
              cds_is_delete = 0 and
              pdt_code containing 'Dop_Sog_NBKR'
        order by cds.cds_create_datetime
        rows 1
        into :create_datetime;
        if (create_datetime is not null) then
          predpisaniye_status = 'Подписан вручную';
      end
    end

    client_status = '';
    if (exists(select *
               from clientdocs cds, passkind
               where cds_from_c_id = :c_id and
                     cds_from_pk_id = pk_id and
                     cds_is_delete = 0 and
                     pk_id = 301 and ---прописано специально, ТП подвязала один pdt_code на смерть клиента  и на смерть родственника
                     :rep_date between cds.cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.9999'))) then
      client_status = 'смерть клиента/умер';

    if (exists(select *
               from clientdocs cds
               inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
               inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
               where cds.cds_from_c_id = :c_id and
                     cds_is_delete = 0 and
                     pdt.pdt_id = 68 and
                     :rep_date between cds.cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.9999')) and client_status = '') then
      client_status = 'Не проживает';

    indexed_adr = null;
    select adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1
    into indexed_adr;

    suspend;
  end
end
