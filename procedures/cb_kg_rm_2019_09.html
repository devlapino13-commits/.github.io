CREATE PROCEDURE CB_KG_RM_2019_09(
    REP_DATE_END DATE)
RETURNS (
    REGION_NAME VARCHAR(100),
    TOTAL_BONUS_OKLAD_SPF CURR_TYPE,
    KOL_KC INTEGER,
    BONUS_OKLAD_SPF CURR_TYPE,
    BALANCE_END CURR_TYPE,
    BALANCE_END_IPF CURR_TYPE,
    DELINQ_BALANCE_30 CURR_TYPE,
    BALANCE_BEGIN CURR_TYPE,
    BALANCE_BEGIN_IPF CURR_TYPE,
    BASIC_BONUS CURR_TYPE,
    PRIROST CURR_TYPE,
    PRIROST_IPF CURR_TYPE,
    KPI1 CURR_TYPE,
    KPI2 CURR_TYPE,
    KPI3 CURR_TYPE,
    BONUS_RM CURR_TYPE,
    BONUS_RM_COMMENT INTEGER,
    PLAN_GROWTH DECIMAL(18,4),
    PLAN_GROWTH_IPF DECIMAL(18,4),
    PLAN_SPF INTEGER,
    CUMULATIVE_KPI CURR_TYPE,
    AAREP_BONUS_MEMO VARCHAR(50))
AS
declare variable rep_date_begin date;
declare variable region_id integer;
declare variable rep_period integer;
declare variable c_id decimal(18,0); /* declare variable ps_from_c_id id_type; */
declare variable bonus_koef curr_type;
declare variable bonus_sum curr_type;
declare variable bonus_memo varchar(255);
begin

  rep_date_begin = rep_date_end - extract(day from rep_date_end);
  rep_period = extract(year from rep_date_end) || right((extract(month from rep_date_end) + 100), 2);

  -- Сумма заработка всех СПФ компании
  --  delete from temp_cb_rm;
  total_bonus_oklad_spf = 0;

  for select parent, sum(aarep_bonus_total)
      from cb_kg_lo_from_km(1000000, :rep_date_end), region
      where reg_id = reg_unicode
      group by parent
      into region_id, bonus_oklad_spf
  do
  begin
    insert into temp_cb_rm (tcr_region_id, tcr_bonus_oklad_spf)
    values (:region_id, :bonus_oklad_spf);
    total_bonus_oklad_spf = total_bonus_oklad_spf + bonus_oklad_spf;
  end

  for select prs_region, prs_plan_growth, prs_plan_growth_ipf, prs_plan_spf
      from planregionspf
      where prs_period = :rep_period and
            prs_kind = 0
      into region_id, plan_growth, plan_growth_ipf, plan_spf
  do
  begin
    select reg_name
    from region
    where reg_unicode = :region_id
    into region_name;

    select tcr_bonus_oklad_spf
    from temp_cb_rm
    where tcr_region_id = :region_id
    into bonus_oklad_spf; -- бонус КС до вычета оклада

    --  Базовый бонус РМ
    basic_bonus = round(bonus_oklad_spf / total_bonus_oklad_spf * 440000, 0);
    --  портфель на конец, баланс просрочки >30 дней
    select sum(oks_balance), sum(iif(pr_code containing '.ipf.', oks_balance, 0)), sum(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0)) --, count(distinct oks_kc_id)
    from oksummary, region_list(:region_id, 1, 3, 1), kontract, product
    where oks_region_id = id_list and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          oks_balance > 0 and
          oks_close_state = 0
    into balance_end, balance_end_ipf, delinq_balance_30; --, kol_k


    for select c_id
        from hr_pstatus, region_list(:region_id, 1, 3, 1), clients, joblist, jobsystem
        where ps_from_reg_id = id_list and
              :rep_date_end between ps_date_begin and ps_date_end and
              ps_is_working = 1 and
              ps_status = 1 and
              ps_from_c_id = c_id and
              ps_from_jl_id = jl_id and
              js_id = jl_kind and
              js_memo = 'JobSystem.Region_Director'
        into c_id

    do
    begin

      kol_kc = 0;
      select count(*)
      from aarep_personal_list(:region_id, :rep_date_end, 1)
      where jl_name in ('Специалист по финансированию', 'Старший специалист по финансированию') and
            ps_curr_status not in ('Отпуск без содерж.по уходу за ребенком', 'Отпуск по беременности и родам', 'Трудовой отпуск декретницы')
      into kol_kc;
      kol_kc = coalesce(kol_kc, 1);

      --  портфель на начало
      select sum(oks_balance), sum(iif(pr_code containing '.ipf.', oks_balance, 0))
      from oksummary, region_list(:region_id, 1, 3, 1), kontract, product
      where oks_region_id = id_list and
            :rep_date_begin between oks_date_begin and oks_date_end and
            oks_from_k_id = k_id and
            oks_close_state = 0 and
            k_from_pr_id = pr_id
      into balance_begin, balance_begin_ipf;
      balance_begin = balance_begin - balance_begin_ipf;

      prirost = balance_end - balance_begin;
      prirost_ipf = balance_end_ipf - balance_begin_ipf;

      --  KPI1
      kpi1 = iif(prirost / plan_growth > 0 and prirost_ipf / plan_growth_ipf > 0, (prirost / plan_growth) * (prirost_ipf / plan_growth_ipf), 0);
      if (kpi1 > 1.1) then
        kpi1 = 1.1;
      --  KPI2
      kpi2 = 1 - (delinq_balance_30 / 1000000) * 7 / 100;
      --  KPI3
      kpi3 = 1 - (cast(plan_spf as double precision) - kol_kc) * 10 / 100;
      if (kpi3 > 1.0) then
        kpi3 = 1.0;

      --  Совокупный KPI
      cumulative_kpi = kpi1 * 0.2 + kpi2 * 0.7 + kpi3 * 0.1;

      --  Бонус РМ к начислению
      --    bonus_rm = round(cast(basic_bonus as double precision) * cumulative_kpi, 0);
      if (basic_bonus < 0) then
        bonus_rm = 0;
      else
        bonus_rm = round(cast(basic_bonus as double precision) * (kpi1 * 0.2 + kpi2 * 0.7 + kpi3 * 0.1), 0);

      if (bonus_rm < 0) then
        bonus_rm = 0;
      ----------------------------------------------------------------------------------
      bonus_rm_comment = 0;
      bonus_memo = '';
      bonus_koef = 0;
      bonus_sum = 0;

      select bonus_koef, bonus_sum, bonus_memo
      from hp_get_vigovor_koef(:c_id, :rep_date_end)
      where bonus_memo <> 'Нет замечаний и выговоров'
      into bonus_koef, bonus_sum, bonus_memo;

      if (bonus_rm > 0 and
          bonus_memo <> '') then
      begin
        bonus_rm_comment = bonus_rm * bonus_koef / 100 - bonus_sum;
        if (bonus_rm_comment < 0) then
          bonus_rm_comment = 0;
      end
      aarep_bonus_memo = bonus_memo;
      if (bonus_memo = '') then
       bonus_rm_comment = bonus_rm;
      suspend;
    end
  end
end
