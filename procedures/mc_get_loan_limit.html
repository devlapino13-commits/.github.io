CREATE PROCEDURE MC_GET_LOAN_LIMIT(
    MCL_ID D_ID)
RETURNS (
    AMOUNT D_CURRENCY,
    CURRENCY D_STR_100,
    INFORM_TXT D_STR_255)
AS
declare variable C_ID D_ID;
begin
  currency = 'сом';
  amount = 0;
  inform_txt = 'txt_1';

  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  if (coalesce(:c_id, 0) <> 0) then
  begin
    amount = (select credit_limit
              from hp_calc_credit_limit(:c_id, 'installment_goods'));

    if (amount is null) then
      inform_txt = 'txt_2';
  end

  suspend;
end
