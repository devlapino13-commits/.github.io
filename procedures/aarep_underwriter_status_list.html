CREATE PROCEDURE AAREP_UNDERWRITER_STATUS_LIST(
    REP_C_ID ID_TYPE,
    REP_KIND STR_100,
    REP_UNIT STR_255)
RETURNS (
    RES_ACL_ID ID_TYPE,
    RES_OBJECT_KIND INTEGER,
    RES_OBJECT_ID ID_TYPE,
    RES_LAST_DATETIME TIMESTAMP,
    RES_LINE_NAME STR_255,
    RES_STATUS INTEGER,
    RES_QUOTA_USED INTEGER,
    RES_K_STATUS INTEGER,
    RES_PROCTIME INTEGER,
    RES_AUTOLOAN INTEGER,
    MARK STR_255,
    RES_SCORES INTEGER)
AS
declare variable RD_ID integer;
declare variable K_DISB_DATE date;
declare variable DB_ID integer;
declare variable SCORES integer;
begin
  db_id = f_get_db_id();
  res_k_status = null;
  scores = 0;
  res_scores = 0;

  if (rep_unit = 'underwriter_manager') then
  begin
    mark = 'yes';
    rep_unit = 'underwriter';
  end
  else
    mark = 'no';
  if (db_id not in (39801, 41701) and rep_unit = 'UnderWriter.HighPriority') then
    rep_unit = 'underwriter';

  if (rep_unit <> 'underwriter' and
      rep_unit <> 'UnderWriter.HighPriority' and
      rep_unit <> 'passport_verify' and
      rep_unit <> 'docrecog_client' and
      rep_unit not starting 'simpleverify_' and
      rep_unit <> 'check_info' and
      rep_unit <> 'Manual_Inn_Check') then -- RFCRU-3399
    exit;

  res_status = case :rep_kind
                 when 'In_Process' then 1
                 when 'New' then 0
                 when 'Approved' then 2
                 when 'Rejected' then -2
                 when 'Revision' then -1
               end;

  if (coalesce(rep_c_id, 0) <> 0) then
  begin
    for select rd_id
        from refdata
        where rd_main_code = 'ActionGroup' and
              rd_str_id = :rep_unit
        into :rd_id
    do
    begin
      if (rep_unit in ('UnderWriter', 'UnderWriter.HighPriority') and
          db_id in (39801, 41701)) then
        continue;
      for select acl_id, acl_object_kind, acl_object_id, acl_last_datetime, acl_quota_used
          from actionlist
          where acl_respons_c_id = :rep_c_id and
                acl_status = :res_status and
                acl_from_rd_id = :rd_id and
                acl_last_datetime >= current_date and
                acl_dont_show_in_order = 0
          order by 1
          into :res_acl_id, :res_object_kind, :res_object_id, :res_last_datetime, :res_quota_used
      do
      begin
        if (db_id = 81001 and
            rep_unit = 'underwriter') then
        begin
          if (exists(select *
                     from actionmark
                     where atm_from_acl_id = :res_acl_id)) then
          begin
            if (mark = 'no') then
              continue;
          end
          else
          begin
            if (mark = 'yes') then
              continue;
          end
        end

        if (res_quota_used <> 0) then
          res_quota_used = 1;
        res_line_name = '';
        if (res_object_kind = 0) then
        begin
          select c_name_fio
          from clients
          where c_id = :res_object_id
          into res_line_name;
        end
        if (res_object_kind = 9) then
        begin
          select kal_new_kf_name || ', ' || k_kontract_no || ' от ' || k_disb_date, k_disb_date
          from kontract, knaktlist
          where k_id = :res_object_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                kal_date <= k_disb_date and
                kal_date_end >= k_disb_date
          into res_line_name, k_disb_date;

          res_k_status = 0;
          if ((db_id = 81001) and
              (k_disb_date >= '16.10.2017')) then
          begin
            select k_curr_status
            from hp_get_kontract_status(:res_object_id, iif(:k_disb_date > current_date, current_date, :k_disb_date))
            into :res_k_status;
          end
        end
        if (res_object_kind = 10) then
        begin
          select c_name_fio || ', ' || pk_name
          from clientdocs, passkind, clients
          where cds_from_pk_id = pk_id and
                cds_from_c_id = c_id and
                cds_id = :res_object_id
          into res_line_name;
        end
        if (res_object_kind = 31) then
        begin
          select c_name_fio || ', ' || pk_name
          from clientdocs, passkind, clients
          where cds_from_pk_id = pk_id and
                cds_from_c_id = c_id and
                cds_is_delete = 0 and
                cds_from_drl_id = :res_object_id
          into res_line_name;
        end
        if (res_object_kind = 58) then
        begin
          select c_name_fio
          from qr_list, clients
          where qrl_id = :res_object_id and
                qrl_from_c_id = c_id
          into res_line_name;
        end
        suspend;
      end
    end
  end
  else
  begin
    for select rd_id
        from refdata
        where rd_main_code = 'ActionGroup' and
              rd_str_id = :rep_unit
        into :rd_id
    do
    begin
      if (rep_unit in ('UnderWriter', 'UnderWriter.HighPriority') and
          db_id in (39801, 41701)) then
        continue;
      for select aco_id, aco_object_kind, aco_object_id, acl_last_datetime, acl_quota_used,
                 coalesce(datediff(second, acl_datetime, current_timestamp), 0)
          from actionorder, actionlist
          where aco_respons_c_id is null and
                aco_status = :res_status and
                aco_from_rd_id = :rd_id and
                acl_dont_show_in_order = 0 and
                aco_id = acl_id
          order by aco_orderno, aco_id
          into :res_acl_id, :res_object_kind, :res_object_id, :res_last_datetime, :res_quota_used, :res_proctime
      do
      begin
        res_autoloan = 0;
        if (db_id = 81001 and
            rep_unit = 'underwriter') then
        begin
          if (exists(select *
                     from actionmark
                     where atm_from_acl_id = :res_acl_id)) then
          begin
            if (mark = 'no') then
              continue;
          end
          else
          begin
            if (mark = 'yes') then
              continue;
            if (exists(select *
                       from kontract
                       inner join product on pr_id = k_from_pr_id and
                             (pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
                       where k_id = :res_object_id)) then
              res_autoloan = 1;
          end
        end

        if (res_quota_used <> 0) then
          res_quota_used = 1;
        res_line_name = '';
        if (res_object_kind = 0) then
        begin
          select c_name_fio
          from clients
          where c_id = :res_object_id
          into res_line_name;
        end
        if (res_object_kind = 9) then
        begin
          select kal_new_kf_name || ', ' || k_kontract_no || ' от ' || k_disb_date, k_disb_date
          from kontract, knaktlist
          where k_id = :res_object_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                kal_date <= k_disb_date and
                kal_date_end >= k_disb_date
          into res_line_name, k_disb_date;

          res_k_status = 0;
          if (db_id = 81001) then
          begin
            select k_curr_status
            from hp_get_kontract_status(:res_object_id, iif(:k_disb_date > current_date, current_date, :k_disb_date))
            into :res_k_status;
          end
        end
        if (res_object_kind = 10) then
        begin
          select c_name_fio || ', ' || pk_name
          from clientdocs, passkind, clients
          where cds_from_pk_id = pk_id and
                cds_from_c_id = c_id and
                cds_id = :res_object_id
          into res_line_name;
        end
        if (res_object_kind = 31) then
        begin
          select c_name_fio || ', ' || pk_name
          from clientdocs, passkind, clients
          where cds_from_pk_id = pk_id and
                cds_from_c_id = c_id and
                cds_is_delete = 0 and
                cds_from_drl_id = :res_object_id
          into res_line_name;
        end
        if (res_object_kind = 58) then
        begin
          select c_name_fio
          from qr_list, clients
          where qrl_id = :res_object_id and
                qrl_from_c_id = c_id
          into res_line_name;
        end
        suspend;
      end
    end
  end
  /*Вывод приориетной очереди для underwriter*/

  if (rep_unit in ('UnderWriter', 'UnderWriter.HighPriority') and
      db_id in (39801, 41701)) then
  begin
    scores = 0;

    select rd.rd_orderno
    from refdata rd
    where rd.rd_code = 'PrioritySertif.ScoresLimit'
    into :scores;

    if (coalesce(rep_c_id, 0) <> 0) then
    begin
      for select rd_id
          from refdata
          where rd_main_code = 'ActionGroup' and
                rd_str_id = 'UnderWriter'
          into :rd_id
      do
      begin
        for select acl_id, acl_object_kind, acl_object_id, acl_last_datetime, acl_quota_used
            from actionlist
            where acl_respons_c_id = :rep_c_id and
                  acl_status = :res_status and
                  acl_from_rd_id = :rd_id and
                  acl_last_datetime >= current_date and
                  acl_dont_show_in_order = 0
            order by 1
            into :res_acl_id, :res_object_kind, :res_object_id, :res_last_datetime, :res_quota_used
        do
        begin
          if (db_id = 81001) then
          begin
            if (exists(select *
                       from actionmark
                       where atm_from_acl_id = :res_acl_id)) then
            begin
              if (mark = 'no') then
                continue;
            end
            else
            begin
              if (mark = 'yes') then
                continue;
            end
          end

          if (res_quota_used <> 0) then
            res_quota_used = 1;
          res_line_name = '';
          if (res_object_kind = 0) then
          begin
            select c_name_fio
            from clients
            where c_id = :res_object_id
            into res_line_name;
          end
          if (res_object_kind = 9) then
          begin
            select kal_new_kf_name || ', ' || k_kontract_no || ' от ' || k_disb_date, k_disb_date
            from kontract, knaktlist
            where k_id = :res_object_id and
                  k_from_kf_id = kal_from_kf_id and
                  knaktlist.is_delete = 0 and
                  kal_date <= k_disb_date and
                  kal_date_end >= k_disb_date
            into res_line_name, k_disb_date;

            res_k_status = 0;
            if ((db_id = 81001) and
                (k_disb_date >= '16.10.2017')) then
            begin
              select k_curr_status
              from hp_get_kontract_status(:res_object_id, iif(:k_disb_date > current_date, current_date, :k_disb_date))
              into :res_k_status;
            end
          end
          if (res_object_kind = 10) then
          begin
            select c_name_fio || ', ' || pk_name
            from clientdocs, passkind, clients
            where cds_from_pk_id = pk_id and
                  cds_from_c_id = c_id and
                  cds_id = :res_object_id
            into res_line_name;
          end
          if (res_object_kind = 31) then
          begin
            select c_name_fio || ', ' || pk_name
            from clientdocs, passkind, clients
            where cds_from_pk_id = pk_id and
                  cds_from_c_id = c_id and
                  cds_is_delete = 0 and
                  cds_from_drl_id = :res_object_id
            into res_line_name;
          end
          if (res_object_kind = 58) then
          begin
            select c_name_fio
            from qr_list, clients
            where qrl_id = :res_object_id and
                  qrl_from_c_id = c_id
            into res_line_name;
          end
          suspend;
        end
      end
    end
    else
    begin
      for select rd_id
          from refdata
          where rd_main_code = 'ActionGroup' and
                rd_str_id = 'UnderWriter'
          into :rd_id
      do
      begin
        for select aco_id, aco_object_kind, aco_object_id, acl_last_datetime, acl_quota_used,
                   coalesce(datediff(second, acl_datetime, current_timestamp), 0)
            from actionorder, actionlist
            where aco_respons_c_id is null and
                  aco_status = :res_status and
                  aco_from_rd_id = :rd_id and
                  acl_dont_show_in_order = 0 and
                  aco_id = acl_id
            order by aco_orderno, aco_id
            into :res_acl_id, :res_object_kind, :res_object_id, :res_last_datetime, :res_quota_used, :res_proctime
        do
        begin
          res_autoloan = 0;
          if (db_id = 81001) then
          begin
            if (exists(select *
                       from actionmark
                       where atm_from_acl_id = :res_acl_id)) then
            begin
              if (mark = 'no') then
                continue;
            end
            else
            begin
              if (mark = 'yes') then
                continue;
              if (exists(select *
                         from kontract
                         inner join product on pr_id = k_from_pr_id and
                               (pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
                         where k_id = :res_object_id)) then
                res_autoloan = 1;
            end
          end

          if (res_quota_used <> 0) then
            res_quota_used = 1;
          res_line_name = '';
          if (res_object_kind = 0) then
          begin
            select c_name_fio
            from clients
            where c_id = :res_object_id
            into res_line_name;
          end
          if (res_object_kind = 9) then
          begin
            res_scores = 0;
            select kal_new_kf_name || ', ' || k_kontract_no || ' от ' || k_disb_date, k_disb_date,
                   f_point(k_clients_cashier_id, current_date)
            from kontract, knaktlist
            where k_id = :res_object_id and
                  k_from_kf_id = kal_from_kf_id and
                  knaktlist.is_delete = 0 and
                  kal_date <= k_disb_date and
                  kal_date_end >= k_disb_date
            into res_line_name, k_disb_date, res_scores;

            if (rep_unit = 'UnderWriter' and
                res_scores >= :scores) then
              continue;
            if (rep_unit = 'UnderWriter.HighPriority' and
                res_scores < :scores) then
              continue;

            res_k_status = 0;
            if (db_id = 81001) then
            begin
              select k_curr_status
              from hp_get_kontract_status(:res_object_id, iif(:k_disb_date > current_date, current_date, :k_disb_date))
              into :res_k_status;
            end
          end
          if (res_object_kind = 10) then
          begin
            select c_name_fio || ', ' || pk_name
            from clientdocs, passkind, clients
            where cds_from_pk_id = pk_id and
                  cds_from_c_id = c_id and
                  cds_id = :res_object_id
            into res_line_name;
          end
          if (res_object_kind = 31) then
          begin
            select c_name_fio || ', ' || pk_name
            from clientdocs, passkind, clients
            where cds_from_pk_id = pk_id and
                  cds_from_c_id = c_id and
                  cds_is_delete = 0 and
                  cds_from_drl_id = :res_object_id
            into res_line_name;
          end
          if (res_object_kind = 58) then
          begin
            select c_name_fio
            from qr_list, clients
            where qrl_id = :res_object_id and
                  qrl_from_c_id = c_id
            into res_line_name;
          end
          suspend;
        end
      end
    end
  end
end
