CREATE PROCEDURE HP_DEPEND_LIST(
    TABL VARCHAR(31),
    FLD VARCHAR(31))
RETURNS (
    OUT_NAME VARCHAR(31),
    OUT_TYPE_NAME VARCHAR(31))
AS
begin
/************************************************************************
*                Процедура возвращает список объектов,
*                в которых участвуют выбранные элементы
*                tabl - таблица,
*                fld - поле этой таблицы
*           Далее надо выполнить запрос вида:
*             select list(h.out_name) "процедуры"
*             from hp_depend_list(CLIENTS, C_NAME_FIO) h
*             where h.out_type_name = 'PROCEDURE'
*           Далее выполнить выгрузку этих процедур в скрипт
*           и произвести нужную замену.
************************************************************************/
    tabl = upper(coalesce(tabl,'CLIENTS'));
    fld = upper(coalesce(fld,'C_NAME_FIO'));

  for select distinct trim(dep.rdb$dependent_name), trim(typ.rdb$type_name)
      from rdb$dependencies dep
      left join rdb$types typ on typ.rdb$type = dep.rdb$dependent_type and
            typ.rdb$field_name = 'RDB$OBJECT_TYPE'
      where dep.rdb$depended_on_name = :tabl and
            (dep.rdb$field_name = :fld or :fld = 'all')
      into :out_name, :out_type_name
  do
  begin
    suspend;
  end
end
