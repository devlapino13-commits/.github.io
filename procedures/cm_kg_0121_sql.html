CREATE PROCEDURE CM_KG_0121_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id id_type;
declare variable pr_id d_integer;
declare variable pr_name d_str_100;
declare variable prl_amount_cl_min d_currency;
declare variable prl_amount_cl_max d_currency;
declare variable pr_code d_str_100;
begin
  res_error_is_critical = 1;
  select k_disb_date, pr_id, pr_name, min(pl.prl_amount_cl_min), max(pl.prl_amount_cl_max), pr.pr_code
  from kontract
  inner join product pr on k_from_pr_id = pr_id
  inner join prodlogic pl on pr_id = pl.prl_from_pr_id and
        k_disb_date between pl.prl_date_begin and pl.prl_date_end and
        k_kolich_repay between pl.prl_repay_min and pl.prl_repay_max
  where k_id = :k_id
  group by 1, 2, 3, 6
  into kf_disb_date, pr_id, pr_name, prl_amount_cl_min, prl_amount_cl_max, pr_code;

  for select kc_from_c_id, c_name_fio, kc_amount_approved
      from knclient mcl
      inner join clients on kc_from_c_id = c_id
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into c_id, res_info_text, kc_amount
  do
  begin
    if (kc_amount < prl_amount_cl_min) then
    begin
      res_error_text = pr_name || ' - Сумма клиента не может быть ниже: ' || round(prl_amount_cl_min, 2);
      suspend;
    end
    if (kc_amount > prl_amount_cl_max) then
    begin
      res_error_text = pr_name || ' - Сумма клиента не может быть больше: ' || round(prl_amount_cl_max, 2);
      suspend;
    end
  end
end
