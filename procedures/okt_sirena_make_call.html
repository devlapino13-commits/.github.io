CREATE PROCEDURE OKT_SIRENA_MAKE_CALL(
    IGNORE_BLOCK D_SMALLINT,
    CALL_DATE D_DATE,
    DO_ACTION STR_100,
    TRANSFER_REASON_CODE STR_100,
    PROJECT_TYPE_CODE STR_100,
    PANT_ID ID_TYPE,
    K_ID ID_TYPE,
    NUM_ID D_INTEGER,
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE,
    SUBJECT_KIND STR_100,
    UN_ID ID_TYPE,
    OPERATOR_LOGIN STR_100,
    CALL_RESULT D_INTEGER,
    MEMO_STR D_STR_LONG,
    NEXT_CALL_DATETIME D_DATETIME,
    ASKED_CALLBACK D_SMALLINT,
    CONT_CALL_IS_ACTIVE D_SMALLINT = -1,
    OTL_ID D_INTEGER = -1,
    PNC_ID D_ID = -1)
RETURNS (
    MESS_STR STR_255)
AS
declare variable PROJECT_TYPE D_INTEGER;
declare variable CC_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable PORUCH_ID ID_TYPE;
declare variable DO_BLOCK D_SMALLINT;
declare variable NEW_PANT_ID ID_TYPE;
declare variable OLD_PANT_ID ID_TYPE;
declare variable DB_ID D_INTEGER;
declare variable SUBJECT_KIND_ID D_INTEGER;
declare variable PCL_ID D_ID;
declare variable PCON_ID D_ID;
declare variable CALL_CENTER_PANTERA_COMMANDS D_INTEGER;
declare variable COMMAND_ID D_INTEGER;
declare variable SBL_KS_FINANCE D_INTEGER;
begin
  /*
    DO_ACTION:
        MAKE_CALL - регистрация звонка
        BLOCK     - блокировка группы
        UNBLOCK   - разблокировка группы
  */
  ignore_block = coalesce(ignore_block, 0);
  db_id = f_get_db_id();
  call_center_pantera_commands = 0;
  pnc_id = iif(pnc_id = -1, null, pnc_id);

  if (f_get_const('call_center_pantera_commands') = 1) then
    call_center_pantera_commands = 1; -- 1 - перешли на командную работу

  if (call_date is null) then call_date = current_date;

  if (un_id is not null) then
  begin
    select un.un_name
    from username un
    where un.un_id = :un_id
    into :operator_login;
  end

  if (coalesce(project_type_code, '') = '') then
  begin
    select pantk.pant_from_k_id,
           rd.rd_id,
           rd.rd_code
    from pantera_kontracts pantk
    inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
    where pantk.pant_id = :pant_id
    into :k_id,
         :project_type,
         :project_type_code;
  end
  else
  begin
    select rd.rd_id
    from refdata rd
    where rd.rd_code = :project_type_code
    into :project_type;
  end

  if (call_center_pantera_commands = 1) then
  begin
      command_id = null;

      select command_id
      from get_oper_command(:un_id, :project_type, :call_date)
      into :command_id;

      if (do_action <> 'unblock') then
      begin
          if (un_id not in (1010100000278, 1000000000000)) then
          begin
              if (un_id is null) then
                mess_str = 'Оператор не определен [UN_ID = null]';
              else if (command_id is null) then
                mess_str = 'Команда оператора не определена. UN_ID = '||:un_id;
          end
      end

      if (mess_str <> '') then
      begin
        suspend;
        exit;
      end
  end

  if (do_action = 'make_call') then
  begin
    select pantk.pant_from_k_id
    from pantera_kontracts pantk
    where pantk.pant_id = :pant_id and
          :call_date between pantk.pant_date_begin and pantk.pant_date_end
    into :k_id;

    if (k_id is null) then
      exit;

    if (object_kind = 'contacts') then
    begin
      cc_id = object_id;

      select cl.c_id
      from clients cl
      where cl.c_id = :cc_id
      into :c_id;
    end
    else
    if (object_kind = 'clients') then
    begin
      c_id = null;

      select kncl.kc_from_c_id
      from knclient kncl
      where kncl.kc_from_c_id = :object_id and
            kncl.kc_from_k_id = :k_id and
            kncl.kc_amount_approved > 0
      into :c_id;

      if (c_id is null) then
      begin
        select zal.z_from_c_id
        from zalog zal
        where zal.z_from_c_id = :object_id and
              zal.z_type = 0
        into :poruch_id;
      end
    end
  end
  cc_id = coalesce(cc_id, 0);
  c_id = coalesce(c_id, 0);
  poruch_id = coalesce(poruch_id, 0);
  mess_str = '';

  do_block = 0;
  new_pant_id = null;

  -- > == BLOCK ================================================================
  if (do_action = 'block') then -- оператор бронирует за собой группу, выбирая ее для обзвона
  begin
    old_pant_id = null;

    select pk.pant_id
    from pantera_kontracts pk
    where pk.pant_from_k_id = :k_id and
          pk.pant_rd_project_type = :project_type and
          pk.pant_date_end = cast('30.12.9999' as date)
    into :old_pant_id;

    if (old_pant_id is null) then
      do_block = 1;
    else
    begin
      if (ignore_block = 1) then
        do_block = 1;
      else
        mess_str = 'Группа уже заблокирована';
    end

    if (do_block = 1) then
    begin
      if ((project_type_code = 'OKTELL.Projects.Pantera') and (call_center_pantera_commands = 0)) then
      begin
        -- если ранее группа уже числилась за этим оператором по Пантере
        if (exists(select *
                   from pantera_kontracts pantk
                   where pantk.pant_oper_called = :operator_login and
                         pantk.pant_from_k_id = :k_id and
                         pantk.pant_date_begin <= :call_date and
                         pantk.pant_rd_project_type = :project_type)) then
        begin
          do_block = 0;
          mess_str = 'Данная группа ранее уже за числилась за ' || operator_login || '. Блокировка запрещена';
        end
      end
    end

    if (do_block = 1) then
    begin
      insert into pantera_kontracts (pant_from_k_id, pant_from_num_id, pant_from_un_id, pant_date_begin, pant_date_end,
                                     pant_rd_project_type, pant_oper_called)
      values (:k_id, :num_id, :un_id, :call_date, cast('30.12.9999' as date), :project_type, :operator_login)
      returning pant_id
      into :new_pant_id;

      -- фиксирую факт передачи
      execute procedure hp_pantera_transfer_insert(:call_date, :old_pant_id, :new_pant_id, :transfer_reason_code);
    end
  end
  -- < =========================================================================

  -- > == UNBLOCK ==============================================================
  if (do_action = 'unblock') then -- разблокировка группы
  begin
    -- фиксирую факт разблокировки
    execute procedure hp_pantera_transfer_insert(:call_date, :pant_id, null, :transfer_reason_code);

    if ((db_id = 81001) and
       (project_type_code = 'OKTELL.Projects.Pantera') and
       (transfer_reason_code in ('SirenaTransferReason.Unlock40days'))) then
    begin
        select pantk.pant_from_k_id
        from pantera_kontracts pantk
        where pantk.pant_id = :pant_id
        into :k_id;

        sbl_ks_finance = (select rd.rd_str_id from refdata rd where rd.rd_code = 'OKTELL.Projects.ks_finance_2');

        insert into sb_akt (sba_from_k_id, sba_number, sba_date_begin, sba_from_sbl_id, sba_status, sba_from_un_id, sba_is_delete)
        values (:k_id, -1, :call_date + 1, :sbl_ks_finance, 2, :un_id, 0);
    end

    update pantera_kontracts pantk
    set pantk.pant_date_end = :call_date
    where pantk.pant_id = :pant_id;
  end
  -- < =========================================================================

  -- > ==  MAKE_CALL ===========================================================
  if (do_action = 'make_call') then -- оператор совершает звонок
  begin
    select rd.rd_id
    from refdata rd
    where rd.rd_code = :subject_kind
    into :subject_kind_id;

    if ((cc_id = 0) and
        (poruch_id = 0)) then --  если оператор позвонил участнику группы
    begin
      if (cont_call_is_active = -1) then -- если -1 значит нужно брать значение из последнего результат звонка, но только в рамках текущие блокировки pant_id
      begin
        cont_call_is_active = null;

        select pds.pds_cont_call_is_active
        from pantera_data_svod pds
        where pds.pds_from_pan_id = :pant_id and
              pds.pds_from_c_id = :c_id and
              coalesce(pds.pds_from_cc_id, 0) = 0 and
              coalesce(pds.pds_from_poruch_id, 0) = 0
        rows 1
        into :cont_call_is_active;
      end

      if (coalesce(cont_call_is_active, -1) = -1)  then
          cont_call_is_active = 1;

      insert into pantera_clients (pcl_from_pan_id, pcl_from_c_id, pcl_rd_res, pcl_memo, pcl_next_call,
                                   pcl_asked_callback, pcl_subject_kind, pcl_from_pnc_id)
      values (:pant_id, :c_id, :call_result, :memo_str, :next_call_datetime, :asked_callback, :subject_kind_id, :pnc_id)
      returning pcl_id
      into :pcl_id;

      if (call_center_pantera_commands = 1) then
      begin
        insert into pantera_clients_opers (pco_from_pcl_id, pco_from_un_id)
        values (:pcl_id, :un_id);
      end

      if (otl_id <> -1) then
      begin
         insert into okt_income_call_link (icl_object_kind, icl_object_id, icl_from_otl_id)
         values (f_objkind('pantera_clients', 'id') , :pcl_id, :otl_id);
      end

      update pantera_data_svod pds
      set pds.pds_cont_call_is_active = :cont_call_is_active
      where pds.pds_from_pan_id = :pant_id and
            pds.pds_from_c_id = :c_id and
            coalesce(pds.pds_from_cc_id, 0) = 0 and
            coalesce(pds.pds_from_poruch_id, 0) = 0;

      object_kind = 'clients';
    end
    else
    if (cc_id <> 0) then --  если оператор позвонил контактному лицу
    begin
      if (coalesce(cont_call_is_active, -1) = -1)  then
          cont_call_is_active = 1;

      insert into pantera_contacts (pcon_from_pan_id, pcon_from_c_id, pcon_from_cc_id, pcon_rd_res, pcon_memo,
                                    pcon_next_call, pcon_asked_callback, pcon_subject_id, pcon_next_cont_call, pcon_from_pnc_id)
      values (:pant_id, :c_id, :cc_id, :call_result, :memo_str, :next_call_datetime, :asked_callback, :subject_kind_id, :cont_call_is_active, :pnc_id)
      returning pcon_id
      into :pcon_id;

      if (call_center_pantera_commands = 1) then
      begin
          insert into pantera_contacts_opers (pco_from_pcon_id, pco_from_un_id)
          values (:pcon_id, :un_id);
      end

      if (otl_id <> -1) then
      begin
         insert into okt_income_call_link (icl_object_kind, icl_object_id, icl_from_otl_id)
         values (f_objkind('pantera_contacts', 'id') , :pcon_id, :otl_id);
      end

      object_kind = 'contacts';
    end
    else
    if (poruch_id <> 0) then --  если оператор позвонил поручителю
    begin
      if (coalesce(cont_call_is_active, -1) = -1)  then
        cont_call_is_active = 1;

      insert into pantera_contacts (pcon_from_pan_id, pcon_from_c_id, pcon_from_poruch_id, pcon_rd_res, pcon_memo,
                                    pcon_next_call, pcon_asked_callback, pcon_subject_id, pcon_next_cont_call, pcon_from_pnc_id)
      values (:pant_id, :c_id, :poruch_id, :call_result, :memo_str, :next_call_datetime, :asked_callback,
              :subject_kind_id, :cont_call_is_active, :pnc_id)
      returning pcon_id
      into :pcon_id;

       if (call_center_pantera_commands = 1) then
      begin
          insert into pantera_contacts_opers (pco_from_pcon_id, pco_from_un_id)
          values (:pcon_id, :un_id);
      end

      if (otl_id <> -1) then
      begin
         insert into okt_income_call_link (icl_object_kind, icl_object_id, icl_from_otl_id)
         values (f_objkind('pantera_contacts', 'id') , :pcon_id, :otl_id);
      end
      object_kind = 'poruch';
    end
  end
  -- < =========================================================================
  suspend;
end
