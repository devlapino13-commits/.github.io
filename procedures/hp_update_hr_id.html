CREATE PROCEDURE HP_UPDATE_HR_ID(
    HO_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    HO_ORDER_NO VARCHAR(20),
    PS_TABEL_NO VARCHAR(10))
AS
begin
  update hr_orders
  set ho_order_no = :ho_order_no
  where ho_id = :ho_id and
        ((ho_order_no <> :ho_order_no) or
        (ho_order_no is null));

  update hr_pstatus
  set ps_tabel_no = :ps_tabel_no
  where ps_from_c_id = :c_id and
        ps_from_ho_id = :ho_id and
        ((ps_tabel_no <> :ps_tabel_no) or
        (ps_tabel_no is null));
end
