CREATE PROCEDURE KFOLDER_KONTRACTS_LIST(
    KONT_ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_FROM_KF_ID DECIMAL(18,0),
    K_CURRENCY VARCHAR(3),
    K_KONTRACT_NO VARCHAR(20),
    K_KONTRACT_DATE DATE,
    K_DISB_DATE DATE,
    K_PERIOD DECIMAL(15,2),
    K_PERCENT DECIMAL(18,7),
    K_PERCENT_LGOTA DECIMAL(18,7),
    K_KOLICH_REPAY INTEGER,
    K_FROM_F_ID VARCHAR(20),
    K_FROM_PT_ID VARCHAR(20),
    K_STATUS INTEGER,
    K_STATUS_NAME VARCHAR(30),
    K_CLOSE_DATE DATE,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_REPAY_KOLICH_CLOSED INTEGER,
    KFI_NAME VARCHAR(100),
    KFI_OFFICER_CODE VARCHAR(20),
    KFI_OFFICER_NAME VARCHAR(100),
    KFI_REGION_NAME VARCHAR(50),
    KFI_REGION_ID INTEGER,
    KFI_DATE DATE,
    KFI_DATE_END DATE,
    PR_CODE VARCHAR(100),
    PR_NAME VARCHAR(50),
    PR_IS_SECURED_TEXT VARCHAR(50),
    PR_LOAN_TYPE_TEXT VARCHAR(50),
    K_CIRCLE_TEXT VARCHAR(10),
    K_BALANCE DECIMAL(18,4),
    K_PAYED_FINE DECIMAL(18,4),
    KS_STATE INTEGER,
    KS_STATE_TEXT VARCHAR(50))
AS
declare variable pr_is_secured integer;
declare variable pr_loan_type integer;
declare variable kf_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select distinct k_from_kf_id
  from kontract
  where k_id = :kont_id
  into :kf_id;

  for select k_id, k_from_kf_id, k_currency, k_kontract_no, k_kontract_date, k_disb_date, k_period, k_percent,
             k_percent_lgota, k_kolich_repay, k_from_f_id, k_from_pt_id, kontract.k_status, k_close_date, k_circle,
             k_circle_internal, k_amount_approved, k_kolich_client_approved, k_repay_kolich_closed, kal_new_kf_name,
             kal_new_officer_code, c_name_fio, reg_name, reg_unicode, kal_date, kal_date_end, pr_code, pr_name,
             pr_is_secured, pr_loan_type, ks_state
      from kontract, knaktlist, clients, region, product, kontstate
      where k_from_kf_id = :kf_id and
            k_disb_date <= :dd and
            kal_from_kf_id = k_from_kf_id and
            kal_new_kf_officer = c_id and
            k_from_pr_id = pr_id and
            reg_unicode = kal_new_reg_id and
            kal_date <= k_disb_date and
            kal_date_end >= k_disb_date and
            k_repay_kolich_closed >= 0 and
            knaktlist.is_delete = 0 and
            kontract.k_status >= 0 and
            cast(ks_date_begin as date) <= :dd and
            ks_date_end >= :dd and
            ks_from_k_id = k_id and
            ks_ol_is_delete = 0
      order by k_disb_date, k_circle, k_circle_internal
      into :k_id, :k_from_kf_id, :k_currency, :k_kontract_no, :k_kontract_date, :k_disb_date, :k_period, :k_percent,
           :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id, :k_status, :k_close_date, :k_circle,
           :k_circle_internal, :k_amount_approved, :k_kolich_client_approved, :k_repay_kolich_closed, :kfi_name,
           :kfi_officer_code, :kfi_officer_name, :kfi_region_name, :kfi_region_id, :kfi_date, :kfi_date_end, :pr_code,
           :pr_name, :pr_is_secured, :pr_loan_type, :ks_state
  do
  begin
    select sum(ok_os), -1 * sum(ok_fine), max(ok_ol_date)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type <= 8 and
          ok_ol_date <= :dd and
          ok_from_c_id = 0
    into :k_balance, :k_payed_fine, :k_close_date;
    if (k_balance is null) then
      k_balance = 0;
    if (k_payed_fine is null) then
      k_payed_fine = 0;

    k_status_name = '';
    if (k_status = -1) then
      k_status_name = 'Аннулирован';
    if (k_status = 0) then
      k_status_name = 'Заявка';
    if (k_status = 1) then
      k_status_name = 'Контракт';

    ks_state_text = '';
    if ((ks_state = 0) or (ks_state = 1)) then
      ks_state_text = 'В процессе';
    if (ks_state = 2) then
      ks_state_text = 'Активный';
    if (ks_state = 3) then
      ks_state_text = 'Закрытый';

    if (pr_is_secured = 1) then
      pr_is_secured_text = 'С залогом';
    else
      pr_is_secured_text = 'Без залога';

    if (pr_loan_type = 1) then
      pr_loan_type_text = 'Индивидуальный кредит';
    else
      pr_loan_type_text = 'Групповой кредит';

    k_percent = k_percent - k_percent_lgota;
    if (k_circle_internal > 1) then
      k_circle_text = cast(k_circle as varchar(3)) || ' / ' || cast(k_circle_internal as varchar(3));
    else
      k_circle_text = cast(k_circle as varchar(5));
    suspend;
  end
end
