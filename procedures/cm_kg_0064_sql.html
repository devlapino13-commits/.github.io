CREATE PROCEDURE CM_KG_0064_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable c_id id_type;
declare variable kc_circle integer;
declare variable kol_cli integer;
declare variable pr_code varchar(100);
declare variable kol_actiya integer;
begin
  for
  select k_kolich_client_approved, pr_code
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code not containing '.ipf.'
  and pr_code not containing '.express.'
  into kol_cli, pr_code
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;

    kol_actiya = 0;
    for
    select kc_from_c_id, kc_amount_approved
    from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    into c_id, kc_amount
    do
    begin
      kc_circle = null;
      select sum(1) from knclient, kontstate
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      into kc_circle;
      if (kc_circle is null) then kc_circle = 0;
      if (kc_circle >= 1) then
      begin
        kol_actiya = kol_actiya + 1;
      end
    end
    if (pr_code containing '.Actiya_11.' and kol_cli - kol_actiya > kol_actiya) then
    begin
      res_error_text = 'Состав группы не соответствует акцие 11%, выберите другой продукт.';
      res_error_is_critical = 1;
      suspend;
    end /* else
    if (pr_code not containing '.Actiya_11.' and kol_cli - kol_actiya <= kol_actiya) then
    begin
      res_error_text = 'Состав группы соответствует акцие 11%, выберите соответствующий продукт.';
      res_error_is_critical = 1;
      suspend;
    end */
  end
end
