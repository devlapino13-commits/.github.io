CREATE PROCEDURE CM_RU_0248_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable pr_code d_str_100;
declare variable mac_amount d_currency;
declare variable had_shans d_integer;
begin
  res_error_is_critical = 2;
  res_error_text = 'Займы наличными запрещены для иностранных граждан.';

  for select mac.mac_from_ma_id, cl.c_id, cl.c_name_fio, pr.pr_code, mac.mac_amount
      from miniappclient mac
      inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
      inner join clients cl on cl.c_id = mac.mac_from_c_id
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      where mac.mac_from_ma_id = :ma_id and
            mac.mac_amount > 0 and
            ma.ma_disb_kind = 1 and
            cl.c_type = 0 and
            cl.c_citizen <> 13 and
            ma.ma_date_end = '31.12.9999'
      into :ma_id, :c_id, :res_info_text, :pr_code, :mac_amount
  do
  begin
    if (pr_code containing '.shans.') then
    begin
      if (mac_amount > 30000) then
        suspend;
      else
      begin
        had_shans = iif(exists(select *
                               from knclient
                               inner join kontract on k_id = kc_from_k_id
                               inner join product on pr_id = k_from_pr_id
                               where kc_from_c_id = :c_id and
                                     kc_amount_approved > 0 and
                                     pr_code containing '.shans.' and
                                     k_disb_kind = 1 and
                                     k_fact_disb_date between current_date - 30 and current_date and
                                     k_status = 1), 1, 0);
        if (had_shans = 1 and current_date >= '16.06.2025') then
        begin
          res_error_text = 'Прошло менее 30 дней со дня прошлой выдачи';
          suspend;
        end
        if (had_shans = 0 and
            mac_amount > 30000 and current_date >= '16.06.2025') then
        begin
          res_error_text = 'Сумма должна быть до 30 тыс.';
          suspend;
        end
      end
    end
    else
    begin
      suspend;
    end
  end
end
