CREATE PROCEDURE HP_APPMERCHANT_VIEW(
    DTBEG D_DATE = null,
    DTEND D_DATE = null)
RETURNS (
    IQA_ID D_ID,
    IQA_DATE D_DATE,
    IQA_STATUS D_SMALLINT,
    IQA_EDITOR_UN_ID D_ID,
    UE_FIO D_STR_100,
    IQA_DATETIME D_DATETIME,
    IQA_DELETED_UN_ID D_ID,
    UD_FIO D_STR_100,
    IQA_DELETED_DATETIME D_DATETIME,
    IQA_SEND_TO_IPC_UN_ID D_ID,
    US_FIO D_STR_100,
    IQA_SEND_TO_IPC_DATETIME D_DATETIME)
AS
begin
  dtbeg = coalesce(dtbeg, '30.12.1899');
  dtend = coalesce(dtend, current_date);

  for
    select
      iqa_id,
      iqa_date,
      iqa_status,
      iqa_editor_un_id,
      ue.un_fio,
      iqa_datetime,
      iqa_deleted_un_id,
      ud.un_fio,
      iqa_deleted_datetime,
      iqa_send_to_ipc_un_id,
      us.un_fio,
      iqa_send_to_ipc_datetime
    from ipc_qr_app
      inner join username as ue
        on ue.un_id = iqa_editor_un_id
      left join username as ud
        on ud.un_id = iqa_deleted_un_id
      left join username as us
        on us.un_id = iqa_send_to_ipc_un_id
    where iqa_date between :dtbeg and :dtend
  into
    :iqa_id,
    :iqa_date,
    :iqa_status,
    :iqa_editor_un_id,
    :ue_fio,
    :iqa_datetime,
    :iqa_deleted_un_id,
    :ud_fio,
    :iqa_deleted_datetime,
    :iqa_send_to_ipc_un_id,
    :us_fio,
    :iqa_send_to_ipc_datetime
  do
  begin
    suspend;
  end
end
