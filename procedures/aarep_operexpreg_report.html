CREATE PROCEDURE AAREP_OPEREXPREG_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OB_BALANCE VARCHAR(20),
    OB_ACC_SHORT VARCHAR(20),
    REG_ID INTEGER,
    OB_AMOUNT DECIMAL(18,4),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_LEVEL4_NAME VARCHAR(100),
    AA_CURR_LEVEL3_NAME VARCHAR(100),
    BG_KIND_STR VARCHAR(20),
    CC_NAME VARCHAR(255))
AS
declare variable reg_prior integer;
declare variable aa_curr_level1_name varchar(100);
declare variable aa_curr_level2_name varchar(100);
declare variable aa_curr_branch_name varchar(100);
declare variable cc_id integer;
begin
  reg_prior = -1;
  for select reg_id, ob_balance, ob_acc_short, bg_kind_str, cc_name, cc_id, sum(ob_amount)
      from aarep_operexpreg_report_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by reg_id, ob_balance, ob_acc_short, bg_kind_str, cc_name, cc_id
      into :reg_id, :ob_balance, :ob_acc_short, bg_kind_str, cc_name, :cc_id, :ob_amount
  do
  begin
    if (reg_id <> reg_prior) then
    begin
      if (reg_id <> 0) then
      begin
        select aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_level1_name, aa_curr_level2_name,
               aa_curr_level3_name, aa_curr_level4_name
        from aarep_get_office_categ(:reg_id, :rep_date_end)
        into :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name, :aa_curr_level1_name,
             :aa_curr_level2_name, :aa_curr_level3_name, :aa_curr_level4_name;
      end
      else
      begin
        aa_curr_office_name = 'Не распределен';
        aa_curr_branch_name = 'Не распределен';
        aa_curr_level_name = 'Не распределен';
        aa_curr_level1_name = 'Не распределен';
        aa_curr_level2_name = 'Не распределен';
        aa_curr_level3_name = 'Не распределен';
        aa_curr_level4_name = 'Не распределен';
      end
      reg_prior = :reg_id;
    end
    if (cc_id = 1) then
      cc_name = :aa_curr_office_name;
    suspend;
  end
end
