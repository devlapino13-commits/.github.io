CREATE PROCEDURE HP_PVNOPER_CASHING_OLD(
    CARD_ID D_INTEGER NOT NULL,
    AMOUNT D_CURRENCY NOT NULL,
    OPER_DATE D_DATE NOT NULL,
    TICKET_NO D_STR_CODE NOT NULL,
    ACC_KT D_STR_20 NOT NULL,
    OFFICE_ID D_INTEGER NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    OL_ID D_ID)
AS
declare variable acc_dt d_str_20;
declare variable id_no d_id;
declare variable c_id d_id;
declare variable is_cashing d_bool;
declare variable ol_doc_no d_integer;
declare variable ol_memo d_str_long;
begin
  -- acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = 417 and al_account = '10001/1');

  if (coalesce(acc_kt, '') = '') then
    exception e_message 'Счет Кт не определен';

  if (amount <= 0) then
    exception e_message 'Сумма обналичивания меньше или равна 0';

  acc_dt = (select al_id from acc_list where al_from_reg_id = :office_id and al_currency = 417 and al_account = '10024/2'); -- Операции по ПВН

  if (coalesce(acc_dt, '') = '') then
    exception e_message 'Счет Дт не определен';

  select c_id,
         is_cashing
  from find_client_by_mb_card(:card_id)
  into :c_id,
       :is_cashing;

  if (c_id is null) then
    exception e_message 'Клиент не определен';

  if (is_cashing <> 1) then
    exception e_message 'С данной карты запрещено обналичивание';

  select kv
  from get_kv(:office_id, 5)
  into :id_no;

  ol_id = :id_no;

  -- очередной номер документа
  select num
  from last_doc_number(:oper_date, 0 /* 0  т.к. нас интерсует расходный кассовый ордер*/, :office_id)
  into :ol_doc_no;

  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                        ol_user_id, ol_document_client, ol_doc_no, ol_doc_no_str)
  values (:ol_id, :office_id, 2, :oper_date, 0,
          f_format('Обналичивание средств по ПВН согласно № чека: %1, ID карты: %2', :ticket_no, :card_id),
          0, 0, :un_id, :c_id, :ol_doc_no, :ol_doc_no)
  returning ol_memo into :ol_memo;

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                        ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                        ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
  values (:ol_id, 1, '417', 1, null, :acc_dt, :amount, :amount, null, null, null, :acc_kt, :c_id,
          :ol_memo, null, :oper_date, 1, :office_id);

  insert into pvnopers (po_date, po_from_ol_id, po_oper_type, po_card_id, po_from_c_id, po_from_k_id, po_amount,
                        po_ticket_no, po_un_id, po_office_id)
  values (:oper_date, :ol_id, 0, :card_id, :c_id, null, :amount, :ticket_no, :un_id, :office_id);
  suspend;
end
