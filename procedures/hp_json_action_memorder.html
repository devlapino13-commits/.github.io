CREATE PROCEDURE HP_JSON_ACTION_MEMORDER(
    ID D_STR_CODE,
    IN_CLASS_NAME D_STR_CODE,
    IN_COMMAND D_STR_CODE,
    IN_JSON D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_ID D_STR_CODE,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable ja_json d_str_8100;
declare variable ja_name d_str_255;
declare variable reg_id d_id;
declare variable sync_mode d_smallint;
declare variable ob_number d_integer;
declare variable is_active d_smallint;
declare variable kob_id d_id;
declare variable ol_date d_date;
declare variable ol_from_reg_id d_integer;
declare variable summa_val d_currency;
declare variable has_amount d_smallint;
declare variable kc_id d_id;
declare variable kob_summa d_currency;
declare variable kob_summa_val d_currency;
declare variable ob_summa d_currency;
declare variable ob_curr_rate d_currency;
begin
  if (coalesce(id, '') = '') then
    id = '0';
  res_id = id;
  res_error_text = '';

  if (coalesce(in_class_name, '') = '') then
    exit;

  if (in_command = 'doLoad') then -- загрузка новой (пустой) записи
  begin
    select un_curr_reg_id
    from username
    where un_id = :un_id
    into reg_id;

    ja_name = 'vOperlist';
    ja_json = f_json_make_pair('ol_from_reg_id', reg_id, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('ol_oper_type', 2, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('ol_oper_subtype', 11, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('ol_date', current_date, 'date');
    ja_json = ja_json || ',' || f_json_make_pair('ol_doc_kind', 9, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('ol_document_client', null, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('ol_document_client_name', '', 'str');
    ja_json = ja_json || ',' || f_json_make_pair('ol_doc_client_name', '', 'str');

    ja_json = '{' || ja_json || '}';
    insert into temp_json_action (ja_name, ja_json)
    values (:ja_name, :ja_json);

    ja_name = 'vOperbook';
    ja_json = f_json_make_pair('ob_currency', f_get_const('cm_default_currency'), 'str');
    ja_json = ja_json || ',' || f_json_make_pair('ob_curr_rate', 1, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('ob_summa_val', 0, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('ob_summa', 0, 'num');

    ja_json = '{' || ja_json || '}';
    insert into temp_json_action (ja_name, ja_json)
    values (:ja_name, :ja_json);

    ja_name = 'vKnOperbook';
    ja_json = f_json_make_pair('kob_currency', f_get_const('cm_default_currency'), 'str');
    ja_json = ja_json || ',' || f_json_make_pair('kob_curr_rate', 1, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('kob_summa_val', 0, 'num');
    ja_json = ja_json || ',' || f_json_make_pair('kob_summa', 0, 'num');

    ja_json = '{' || ja_json || '}';
    insert into temp_json_action (ja_name, ja_json)
    values (:ja_name, :ja_json);

    suspend;
  end

  if (in_command = 'doSave') then
  begin
    sync_mode = 0;
    if (id <> 0 and is_active = 1) then sync_mode = 1;

    select res_id, res_error_text, ol_date, ol_from_reg_id, ol_is_active
    from hp_json_action_save_operlist('vOperlist', :id, :un_id)
    into :res_id, :res_error_text, :ol_date, :ol_from_reg_id, :is_active;

    if (res_error_text <> '') then
    begin
      Suspend;
      Exit;
    end

    has_amount = 0;
    if (sync_mode = 1) then
    begin
      update operbook
      set ob_ol_is_active = -1 -- временно устанавливаем -1
      where ob_from_ol_id = :res_id;

      update kn_operbook
      set kob_ol_is_active = -1 -- временно устанавливаем -1
      where kob_from_ol_id = :res_id;
    end
    else
    begin
      delete from operbook
      where ob_from_ol_id = :res_id;

      delete from kn_operbook
      where kob_from_ol_id = :res_id;
    end

    ja_name = 'vOperbook';
    for select ja_json
        from temp_json_action
        where ja_name = :ja_name
        into :ja_json
    do
    begin
      for select f_json_str_get_value(:ja_json, 'ob_number',null) as ob_number,
                 f_json_str_get_value(:ja_json, 'ob_currency',null) as ob_currency,
                 f_json_str_get_value(:ja_json, 'ob_curr_rate',null) as ob_curr_rate,
                 f_json_str_get_value(:ja_json, 'ob_client_dt',null) as ob_client_dt,
                 f_json_str_get_value(:ja_json, 'ob_account_dt',null) as ob_account_dt,
                 f_json_str_get_value(:ja_json, 'ob_summa_val',0) as ob_summa_val,
                 f_json_str_get_value(:ja_json, 'ob_summa',0) as ob_summa,
                 f_json_str_get_value(:ja_json, 'ob_from_pc_id',null) as ob_from_pc_id,
                 f_json_str_get_value(:ja_json, 'ob_from_kp_id',null) as ob_from_kp_id,
                 f_json_str_get_value(:ja_json, 'ob_account_kt',null) as ob_account_kt,
                 f_json_str_get_value(:ja_json, 'ob_client_kt',null) as ob_client_kt,
                 f_json_str_get_value(:ja_json, 'ob_detail','') as ob_detail,
                 f_json_str_get_value(:ja_json, 'ob_from_sac_id',null) as ob_from_sac_id
          from rdb$database as cursor ob
      do
      begin
        res_error_text = '';

        ob_summa = ob.ob_summa;
        ob_curr_rate = ob.ob_curr_rate;
        summa_val = ob.ob_summa_val;
        if (ob_curr_rate = 1) then
          summa_val = ob_summa;


        if (coalesce(ob.ob_account_dt, '') = '' and
            coalesce(ob.ob_account_kt, '') <> '') then
          res_error_text = 'Дебет счет не заполнен';
        else
        if (coalesce(ob.ob_account_dt, '') <> '' and
            coalesce(ob.ob_account_kt, '') = '') then
          res_error_text = 'Кредит счет не заполнен';
        if (coalesce(ob.ob_account_dt, '') = '' and
            coalesce(ob.ob_account_kt, '') = '' and
            (coalesce(ob_summa, 0) <> 0 or
            coalesce(summa_val, 0) <> 0)) then
          res_error_text = 'Счета не заполнены';
        else
        if (coalesce(ob.ob_account_dt, '') <> '' and
            coalesce(ob.ob_account_kt, '') <> '' and
            (coalesce(ob_summa, 0) <> 0 or
            coalesce(summa_val, 0) <> 0)) then
        begin
          ob_number = ob.ob_number;
          if (coalesce(ob_number, 0) = 0) then
            ob_number = -1;

          has_amount = 1;
          update or insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt,
                                          ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id,
                                          ob_account_kt, ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_is_active)
          values (:res_id, :ob_number, ob.ob_currency, :ob_curr_rate, ob.ob_client_dt, ob.ob_account_dt, :summa_val,
                  :ob_summa, ob.ob_from_pc_id, ob.ob_from_kp_id, ob.ob_account_kt, ob.ob_client_kt, ob.ob_detail,
                  ob.ob_from_sac_id, :is_active)
          matching (ob_from_ol_id, ob_number);
        end

        if (res_error_text <> '') then
        begin
          suspend;
          exit;
        end
      end
    end

    ja_name = 'vKnOperbook';
    for select ja_json
        from temp_json_action
        where ja_name = :ja_name
        into :ja_json
    do
    begin
      for select f_json_str_get_value(:ja_json, 'kob_id',null) as kob_id,
                 f_json_str_get_value(:ja_json, 'kob_currency','') as kob_currency,
                 f_json_str_get_value(:ja_json, 'kob_curr_rate',1) as kob_curr_rate,
                 f_json_str_get_value(:ja_json, 'kob_acc_dt',null) as kob_acc_dt,
                 f_json_str_get_value(:ja_json, 'kob_acc_kt',null) as kob_acc_kt,
                 f_json_str_get_value(:ja_json, 'kob_summa',0) as kob_summa,
                 f_json_str_get_value(:ja_json, 'kob_summa_val',0) as kob_summa_val,
                 f_json_str_get_value(:ja_json, 'kob_detail_txt','') as kob_detail_txt,
                 f_json_str_get_value(:ja_json, 'kob_client_dt',null) as kob_client_dt,
                 f_json_str_get_value(:ja_json, 'kob_client_kt',null) as kob_client_kt,
                 f_json_str_get_value(:ja_json, 'kob_from_ol_id',null) as kob_from_ol_id,
                 f_json_str_get_value(:ja_json, 'kob_from_kc_id',null) as kob_from_kc_id
          from rdb$database as cursor kob
      do
      begin
        res_error_text = '';
        kob_summa = kob.kob_summa;
        kob_summa_val = kob.kob_summa_val;
        ob_curr_rate = kob.kob_curr_rate;

        if (kob.kob_acc_dt is null and
            kob.kob_acc_kt is not null) then
          res_error_text = 'Дебет счет ЕПС не заполнен';
        else
        if (kob.kob_acc_dt is not null and
            kob.kob_acc_kt is null) then
          res_error_text = 'Кредит счет ЕПС не заполнен';
        if (kob.kob_acc_dt is null and
            kob.kob_acc_kt is null and
            (coalesce(kob_summa, 0) <> 0 or
             coalesce(kob_summa_val, 0) <> 0)) then
          res_error_text = 'Не заполнены счета ЕПС';
        else
        if (kob.kob_acc_dt is not null and
            kob.kob_acc_kt is not null and
            (coalesce(kob_summa, 0) <> 0 or
            coalesce(kob_summa_val, 0) <> 0)) then
        begin
          kob_id = kob.kob_id;
          if (coalesce(kob_id, 0) = 0) then
            kob_id = null;

          kc_id = kob.kob_from_kc_id;
          if (sync_mode = 0 and kc_id is null) then
          begin
            select acc_object_id
            from ref_account
            where acc_id = kob.kob_acc_dt and
                  acc_object_kind = 8 and
                  acc_object_id >= 9
            into kc_id;

            if (kc_id is null) then
            begin
              select acc_object_id
              from ref_account
              where acc_id = kob.kob_acc_kt and
                    acc_object_kind = 8 and
                    acc_object_id >= 9
              into kc_id;
            end
          end

          has_amount = 1;
          execute procedure hp_do_kn_operbook_ins_simple(:kob_id, :res_id, :kc_id, kob.kob_acc_dt,
              kob.kob_acc_kt, :kob_summa, :kob_summa_val, :ol_date, :ol_from_reg_id, :is_active, kob.kob_detail_txt,
              kob.kob_client_dt, kob.kob_client_kt, kob.kob_currency, :ob_curr_rate);
        end
        if (res_error_text <> '') then
        begin
          suspend;
          exit;
        end
      end
    end

    delete from operbook
    where ob_from_ol_id = :res_id and
          ob_ol_is_active = -1;

    delete from kn_operbook
    where kob_from_ol_id = :res_id and
          kob_ol_is_active = -1;

    if (has_amount = 0) then
      res_error_text = 'Нет проводок для сохранения';

    suspend;
  end
end
