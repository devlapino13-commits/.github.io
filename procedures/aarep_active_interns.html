CREATE PROCEDURE AAREP_ACTIVE_INTERNS(
    REP_REG_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    PARENT_NAME VARCHAR(50),
    REG_NAME D_STR_100,
    C_FIO D_STR_100,
    JL_NAME VARCHAR(50),
    SD_DATE_BEGIN D_DATE,
    SD_DATE_END D_DATE,
    BIRTH_DATE D_DATE,
    PHONE_NUM D_STR_100,
    PASS_NO VARCHAR(30),
    PASS_DATE D_DATE,
    PASS_WHO D_STR_255,
    INN D_STR_20,
    ADDRESS D_STR_255,
    ADDRESS_FACT D_STR_255,
    NATIONALITY D_STR_255,
    AGE D_INTEGER,
    SEX D_STR_10)
AS
declare variable i_id_tmp d_integer;
declare variable tmp_res d_smallint;
declare variable c_id d_id;
declare variable i_id d_integer;
declare variable jl_id d_integer;
declare variable jl_id_tmp d_integer;
declare variable last_result_date d_date;
declare variable reg_id d_integer;
declare variable reg_id_tmp d_integer;
declare variable sd_id d_integer;
declare variable nation_tmp d_integer;
declare variable nation d_integer;
declare variable address_type d_smallint;
declare variable addr_str_tmp d_str_255;
declare variable addr_empty d_str_255;
declare variable adl_id d_id;
begin
  select adl.adl_address_str
  from addrlist adl
  where adl.adl_id = 41701000000000
  into :addr_empty;

  reg_id = null;
  for select i.i_id, i.i_from_c_id, i.i_from_jl_id, i.i_from_reg_id, rl.parent_name
      from interns i
      inner join region_list(:rep_reg_id, 1, 3, 1) rl on i.i_from_reg_id = rl.id_list
      where i.i_is_delete = 0 and
            ((:rep_date between coalesce(i.i_date_begin, :rep_date) and coalesce(i.i_date_end, :rep_date)) or
            :rep_date <= coalesce(i.i_date_end_fact, :rep_date))
      into :i_id, :c_id, :jl_id_tmp, :reg_id_tmp, :parent_name
  do
  begin
    i_id_tmp = null;
    last_result_date = null;
    sd_id = null;
    select icr.icr_from_i_id, coalesce(icr.icr_personnel_com_res, 0) + coalesce(icr.icr_educational_com_res, 0),
           cast(icr.icr_educational_com_timestamp as date), icr.icr_from_sd_id
    from intern_committee_result icr
    where icr.icr_from_i_id = :i_id and
          ((icr.icr_personnel_com_res = 1 and
          icr.icr_educational_com_res is null) or
          (icr.icr_personnel_com_res = 1 and
          icr.icr_educational_com_res = 1))
    into :i_id_tmp, :tmp_res, :last_result_date, :sd_id;
    if (:i_id_tmp is null and
        not exists(select *
                   from intern_committee_result icr
                   where icr.icr_from_i_id = :i_id)) then
      i_id_tmp = i_id;
    if (:i_id_tmp is null) then
      continue;

    if (:tmp_res = 2 and
        not exists(select *
                   from hr_pstatus ps
                   where ps.ps_from_c_id = :c_id and
                         ps.ps_status = 1 and
                         ps.ps_from_jl_id = :jl_id_tmp and
                         ps.ps_date_begin >= :last_result_date) and
        datediff(month, :last_result_date, :rep_date) > 0) then
      continue;

    if (:reg_id_tmp <> :reg_id or :reg_id is null) then
    begin
      reg_id = :reg_id_tmp;
      select reg.reg_code || ' ' || reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_name;
    end

    if (:jl_id_tmp <> :jl_id or :jl_id is null) then
    begin
      jl_id = :jl_id_tmp;
      select jl.jl_name
      from joblist jl
      where jl.jl_id = :jl_id
      into :jl_name;
    end

    select c.c_name_fio, c.c_birth_date, c.c_pass_no, c.c_pass_date, c.c_pass_who, c.c_inn, c.c_nation,
           f_age(c.c_birth_date, :rep_date),
           case c.c_sex
             when 0 then 'женский'
             when 1 then 'мужской'
           end,
           f_cont_info('clients', c.c_id, 'cell', 'last')
    from clients c
    where c.c_id = :c_id
    into :c_fio, :birth_date, :pass_no, :pass_date, :pass_who, :inn, :nation_tmp, :age, :sex, :phone_num;

    if (:nation_tmp is null) then
      nationality = null;
    else
    if (:nation_tmp <> :nation or :nation is null) then
    begin
      nation = :nation_tmp;
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :nation
      into :nationality;
    end

    sd_date_begin = null;
    sd_date_end = null;

    if (:sd_id is not null) then
      select sd.sd_date_begin, sd.sd_date_end
      from servicedog sd
      where sd.sd_id = :sd_id
      into :sd_date_begin, :sd_date_end;

    address = null;
    address_fact = null;

    for select adc.adc_address_type, adc.adc_from_adl_id
        from addrclient adc
        where adc.adc_from_c_id = :c_id and
              adc.adc_address_type in (1, 2)
        into :address_type, :adl_id

    do
    begin
      if (:adl_id <> 41701000000000) then
        select adl.adl_address_str
        from addrlist adl
        where adl.adl_id = :adl_id
        into :addr_str_tmp;
      else
        addr_str_tmp = :addr_empty;

      if (:address_type = 1) then
        address = :addr_str_tmp;
      if (:address_type = 2) then
        address_fact = :addr_str_tmp;
    end

    suspend;
  end
end
