CREATE PROCEDURE AAREP_DELIQ_DEBTORS_LIST(
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    KF_BRANCH_ID INTEGER,
    DELIQ_OS DECIMAL(18,4),
    DELIQ_PERC DECIMAL(18,4),
    DELIQ_DAYS INTEGER,
    KAL_NEW_KF_OFFICER DECIMAL(18,0),
    OK_SH_DATE_MIN DATE,
    K_KONT_BALANCE DECIMAL(18,4),
    DEBTOR_ID DECIMAL(18,0))
AS
declare variable ok_from_k_id decimal(18,0);
declare variable ok_sh_date date;
declare variable ok_os_deliq decimal(18,4);
declare variable ok_perc_deliq decimal(18,4);
declare variable is_deliq smallint; /* 0 - неактивный, 1 - активный */
declare variable deliq_os_temp decimal(18,4);
declare variable deliq_perc_temp decimal(18,4);
declare variable deliq_days_temp integer;
declare variable is_deliq1 smallint;
declare variable cr_from_cu_id varchar(3);
declare variable cr_rate decimal(18,7);
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate decimal(18,7);
begin
-- knclient_fix

  /*
CREATED BY: IGUSEV, UBAKACHIEV 06.06.2012

PURPOSE: Процедура позволяет на определенную дату вытащить список клиентов-должников, у которых количество дней просрочки от 14 дней и более.
         Разаработана для проекта "Обзвон должников (Колл-Центр)".

LOGIC:   Сначала вытаскивается список активнх кредитов,
         затем в разрезе этого списка осуществляется поиск задолжностей по кредтам.
         Если задолжность есть и количество дней просрочки более или = 14 дней,
         тогда вытаскиваются клиенты относящиеся к этому кредиту, у которых 'выданная сумма' > 0

MEMO:    OK_SH_DATE_MIN  -  дата начала просрочек
         K_KONT_BALANCE  -  активный портфель просроченного кредита  (На данный момент NUll, т.к. вычисление закоменчено)

*/

  select cr_from_cu_id, cr_rate
  from opercurr
  where :rep_date_end between cr_date and cr_date_end and
        cr_from_cu_id = '840'
  into :cr_from_cu_id, :cr_rate;

  /*Вытаскиваю активные контракты*/
  for select k_id, k_kontract_no, k_disb_date, k_amount_approved,
             /*reg_code||' '||reg_name,  */
             kal_new_reg_id, /* ID офиса кредитной папки*/
             kal_new_kf_officer, k_index_curr_code, k_index_curr_rate
      from kontstate, kontract, knaktlist /*, region reg */
      where :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            k_id = ks_from_k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end
            /*and kal_new_reg_id=reg_unicode  */
      into :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :kf_branch_id, :kal_new_kf_officer,
           :k_index_curr_code, :k_index_curr_rate
  do
  begin
    ok_sh_date = null;
    ok_os_deliq = null;
    ok_perc_deliq = null;

    deliq_os_temp = 0;
    deliq_perc_temp = 0;
    deliq_days_temp = 0;

    ok_sh_date_min = :rep_date_end;
    is_deliq = 0;

    deliq_os = null;
    deliq_perc = null;
    deliq_days = null;

    is_deliq1 = 0;

    /*В разрезе активного кредита выявляю просрочку (Вытаскиваю просрочнные кредиты)*/
    for select ok_from_k_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_ol_date <= :rep_date_end and
              ok_sh_date <= :rep_date_end and
              ok_from_k_id = :k_id and
              ok_from_c_id = 0
        group by ok_from_k_id, ok_sh_date
        having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
        into :ok_from_k_id, :ok_sh_date, :ok_os_deliq, :ok_perc_deliq
    do
    begin
      if (:ok_os_deliq + :ok_perc_deliq > 0) then
      begin
        is_deliq = 1;
        if (ok_sh_date_min > :ok_sh_date) then
          ok_sh_date_min = :ok_sh_date;
      end
      deliq_os_temp = deliq_os_temp + ok_os_deliq;
      deliq_perc_temp = deliq_perc_temp + ok_perc_deliq;
    end

    if (is_deliq = 1) then
    begin
      deliq_days_temp = cast((:rep_date_end - ok_sh_date_min) as integer);

      k_kont_balance = null;

      deliq_os = deliq_os_temp;
      deliq_perc = deliq_perc_temp;
      deliq_days = deliq_days_temp;

      is_deliq1 = is_deliq;
    end
    if (is_deliq1 = 1) then
    begin
      if (deliq_days >= 14) then
      /*
          Исходя из ТЗ Кыштобаевой Эльвиры (ТЗ от 01.06.2012)
          Выдержка из ТЗ:
          "Информация автоматически высвечивается в БД.
          Она переходит из БД Финстарта, как только период просроченного кредита превышает 14 дней."
      */
      begin
        /* Вытаскиваю клиентов относящихся к просроченным кредитам   */
        for select kc_from_c_id
            from knclient
            where kc_from_k_id = :k_id and
                  kc_amount_approved > 0
            into :debtor_id
        do
        begin
          suspend;
        end
      end
    end

  end
end
