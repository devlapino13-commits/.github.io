CREATE PROCEDURE HP_ANALITICA_FILL_FIELDS(
    FIELD_CODE VARCHAR(100),
    FIELD_SCRIPT VARCHAR(1024))
RETURNS (
    RES_FIELD_NAME VARCHAR(100))
AS
declare variable field_id integer;
declare variable sql_text varchar(1200);
begin
-- Процедура заполняет нужные поля по скрипту и возвращает названия поля где сохранены данные
  select taf_field_id
  from temp_analitica_fields
  where taf_field_code = :field_code
  into :field_id;

  if (field_id is not null) then
  begin
    res_field_name = 'ad_' || field_id;
    suspend;
    exit;
  end

  select max(taf_field_id)
  from temp_analitica_fields
  into :field_id;
  if (field_id is null) then
    field_id = 0;

  field_id = field_id + 1;
  insert into temp_analitica_fields (taf_field_id, taf_field_code)
  values (:field_id, :field_code);
  res_field_name = 'ad_' || field_id;

  sql_text = 'update temp_analitica_data set ' || :res_field_name || '=left((' || :field_script || '),100)';
  execute statement sql_text;
  suspend;
end
