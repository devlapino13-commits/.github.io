CREATE PROCEDURE HP_EXTSYSMAIN_TM_PERIOD(
    REG_ID D_INTEGER NOT NULL,
    DATETIME_START_INC D_DATETIME,
    DATETIME_FINISH D_DATETIME)
RETURNS (
    REG_NAME D_STR_CODE,
    TERMINAL_CODE D_STR_100,
    REP_AMOUNT_ALL D_CURRENCY,
    REP_AMOUNT_CLS D_CURRENCY,
    REP_AMOUNT_IPF D_CURRENCY,
    REP_PERIOD_INFO D_STR_255)
AS
declare variable TM_DATE_BEGIN D_DATE;
declare variable LAST_DATETIME D_DATETIME;
declare variable TERMINAL_ID D_INTEGER;
declare variable DATETIME_START D_DATETIME;
begin
  --RFCKG-4675
  if (datetime_finish is null) then
    datetime_finish = current_timestamp;

  for select reg.reg_name, tm.tm_id, tm.tm_terminal_id, tm_date_begin
  from region_list(:reg_id, 1, 3, 1)
  inner join region reg on id_list = reg.reg_unicode
  inner join terminals tm on reg.reg_unicode = tm.tm_office
  where tm.tm_date_end > cast(:datetime_finish as date) or
       tm.tm_date_end is null
  into reg_name, terminal_id, terminal_code, tm_date_begin
  do
  begin
    rep_amount_all = 0;
    rep_amount_cls = 0;
    rep_amount_ipf = 0;
    rep_period_info = '';

    if (tm_date_begin > cast(datetime_finish as date)) then
    begin
      suspend;
      continue;
    end

    last_datetime = null;
    select tml_last_datetime
    from termilast
    where tml_from_tm_id = :terminal_id
    into last_datetime;

    if (last_datetime is null) then
      continue;

    datetime_start = datetime_start_inc;
    if (datetime_start is null or datetime_start < :last_datetime) then
      datetime_start = :last_datetime;

    select rep_amount_all, rep_amount_cls, rep_amount_ipf, rep_period_info
    from hp_extsysmain_terminal_info(:terminal_id, :datetime_start, :datetime_finish)
    into :rep_amount_all, rep_amount_cls, rep_amount_ipf, rep_period_info;

    suspend;
  end
end
