CREATE PROCEDURE HR_PSTATUS_LIST(
    HO_NEED_ID DECIMAL(18,0))
RETURNS (
    HO_ID DECIMAL(18,0),
    HO_DATE DATE,
    HO_FROM_REG_ID INTEGER,
    HO_TEXT VARCHAR(1024),
    HO_REASON VARCHAR(255),
    HO_ORDER_TYPE INTEGER,
    HO_ORDER_NO VARCHAR(20),
    PS_ID INTEGER,
    PS_FROM_HO_ID DECIMAL(18,0),
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_FROM_REG_ID INTEGER,
    PS_FROM_DEP_ID INTEGER,
    PS_FROM_JL_ID INTEGER,
    PS_IS_ON_STAFF SMALLINT,
    PS_IS_WORKING SMALLINT,
    PS_FROM_C_ID DECIMAL(18,0),
    PS_SALARY DECIMAL(18,4),
    PS_TRANSFER_OFFICE INTEGER,
    PS_TABEL_NO VARCHAR(10),
    PS_CLASS INTEGER,
    C_NAME_FIO VARCHAR(100),
    C_OFFICE_NAME VARCHAR(100),
    C_JOBLIST VARCHAR(100),
    C_IS_WORK VARCHAR(50),
    C_DEPART_NAME VARCHAR(100),
    HO_COME_BACK_DATE DATE,
    PSA_AGREEMENT_NO VARCHAR(10),
    HO_CAUSE_ID INTEGER,
    C_AGE INTEGER)
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ps_id, ps_from_ho_id, ps_date_begin, ps_date_end, ps_period_month, ps_date_end_in_order, ps_from_reg_id,
             ps_from_dep_id, ps_from_jl_id, ps_is_on_staff, ps_is_working, ps_from_c_id, ps_salary, ps_transfer_office,
             ps_tabel_no, ps_class, ho_id, ho_date, ho_from_reg_id, ho_text, ho_reason, ho_order_type, ho_order_no,
             c_name_fio, reg_name, jl_name, dep_name, ho_come_back_date,
             ho_cause_id, datediff(day,c_birth_date,coalesce(ho_date,current_date))/365
      from hr_pstatus, hr_orders, clients, region, joblist, depart
      where ps_from_ho_id = ho_id and
            ho_id = :ho_need_id and
            ps_from_c_id = c_id and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            ps_from_dep_id = dep_id
      into :ps_id, :ps_from_ho_id, :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order,
           :ps_from_reg_id, :ps_from_dep_id, :ps_from_jl_id, :ps_is_on_staff, :ps_is_working, :ps_from_c_id, :ps_salary,
           :ps_transfer_office, :ps_tabel_no, :ps_class, :ho_id, :ho_date, :ho_from_reg_id, :ho_text, :ho_reason,
           :ho_order_type, :ho_order_no, :c_name_fio, :c_office_name, :c_joblist, :c_depart_name, :ho_come_back_date,
           :ho_cause_id, :c_age
  do
  begin
    if (ps_is_working = 0) then
      c_is_work = 'Уволен';
    else
      c_is_work = 'Работает';

    psa_agreement_no = null;

    select psa.psa_agreement_no
    from hr_pstatus_agreement psa
    where psa.psa_from_ho_id = :ps_from_ho_id and
          psa.psa_from_ps_id = :ps_id
    into :psa_agreement_no;

    suspend;
  end
end
