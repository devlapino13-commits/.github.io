Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXEVENT_CONTROLS
RETURNS (
    CONTROL_ID D_STR_CODE,
    CONTROL_NAME D_STR_CODE,
    CONTROL_MEMO D_STR_255,
    CONTROL_IMAGE D_INTEGER)
AS
declare variable control_kind d_str_code;
begin
  control_id = 'editor';
  control_name = 'Редактор';
  control_memo = 'Визуальные редакторы в окне';
  control_image = -1;
  suspend;

  for select editor_id, editor_kind, editor_name, editor_memo, image_index
  from uxp$control_kind
  into :control_id, :control_kind, :control_name, :control_memo, control_image
  do
  begin
    if (control_kind = 'editor') then
      Continue;

    if (control_kind = 'aux' and control_id not in ('label', 'button')) then
      Continue;

    suspend;
  end

  control_id = 'source';
  control_name = 'Источник данных';
  control_memo = 'Источник данных, Dataset';
  control_image = -1;
  suspend;

  control_id = 'field';
  control_name = 'Поле источника';
  control_memo = 'Поля источника в БД';
  control_image = -1;
  suspend;

  control_id = 'param';
  control_name = 'Параметр источника';
  control_memo = 'Параметр хранимой процедуры';
  control_image = -1;
  suspend;

  control_id = 'column';
  control_name = 'Колонка грида';
  control_memo = 'Колонка грида или дерева';
  control_image = -1;
  suspend;
end
