CREATE PROCEDURE HP_EDOC_WEB_UNIQACC_CHECK(
    CL_ACCOUNT D_STR_10,
    CL_BIRTHDATE D_DATE)
RETURNS (
    RES_CLIENT_ID D_ID,
    RES_TEXT D_STR_255,
    RES_CODE D_INTEGER)
AS
declare variable cell_no d_str_code;
declare variable client_name d_str_100;
declare variable db_id d_integer;
declare variable k_id d_id;
begin
  res_code = 0;
  res_text = '';
  db_id = f_get_db_id();

  if (cl_account is null) then
  begin
    res_code = 1;
    res_text = 'Лицевой счет не указан';
  end
  if (res_code = 0 and cl_birthdate is null) then
  begin
    res_code = 1;
    res_text = 'Дата рождения не указана';
  end
  if (res_code = 0) then
  begin
    select c_id
    from hp_check_clientuniq_acc_no(:cl_birthdate, :cl_account)
    into res_client_id;
    if (res_client_id = -1) then
    begin
      res_code = 1;
      res_text = 'Не найден клиент с такой датой рождения и лицевым счетом, обратитесь в офис компании';
    end
  end

  if (res_code = 0 and db_id = 81001) then
  begin
    -- Если займ утвержден, но еще не выдан, запрещаем изменение карты
    if (exists(
        select 1
        from kontract, knclient, kontstate
        where kc_from_c_id = :res_client_id
          and kc_from_k_id = k_id
          and kc_amount_approved > 0
          and k_status = 1
          and ks_from_k_id = k_id
          and ks_ol_is_delete = 0
          and current_date between ks_date_begin and ks_date_end
          and (ks_state = 0 or (ks_state = 1 and ks_date_begin = current_date))))
    then
    begin
      res_code = 1;
      res_text = 'Невозможно заменить карту, так как ваш заем в процессе выдачи.';
    end
  end

  if (res_code = 0) then
  begin
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :res_client_id
    into client_name;

    select short_fio
    from hp_fio_clear(:client_name)
    into client_name;

    cell_no = f_cont_info('clients', res_client_id, 'cell_main', 'ok');
    if (cell_no = '') then
    begin
      res_code = 1;
      res_text = 'У вас нет подтвержденного номера. Обратитесь в офис компании.';
    end
    else
    begin
      cell_no = f_cont_format(cell_no, 'cell', 'secured');
      res_text = f_format('%1, на ваш телефон %2 будет выслана ссылка и код для ввода данных банковской карты. Подтверждаете?', client_name, cell_no);
    end
  end
  suspend;
end
