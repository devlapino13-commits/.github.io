CREATE PROCEDURE CM_KG_0060_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable KAL_NEW_KF_OFFICER ID_TYPE;
declare variable UN_FROM_C_ID ID_TYPE;
declare variable UN_NAME varchar(30);
begin
 if (k_id = 109010045525100101) then exit;
  select KAL_NEW_KF_OFFICER, K.K_AMOUNT_APPROVED
  from KNAKTLIST, KONTRACT K
  where KNAKTLIST.IS_DELETE = 0 and
        KAL_NEW_KF_OFFICER in (--
                           1090100584537, 1090100705444, 1090100061797, --
                           1090100242975, 1090100686545, 1090100791526, --
                           1090100595202, 1090100741327, 1090100801562,--
                           1090100845364 ) and
        K_DISB_DATE between KAL_DATE and KAL_DATE_END and
        KAL_FROM_KF_ID = K_FROM_KF_ID and
        K_AMOUNT_APPROVED > 20000 and
        K_ID = :K_ID
  into KAL_NEW_KF_OFFICER, KC_AMOUNT;
  if (coalesce(KAL_NEW_KF_OFFICER, 0) <> 0) then
  begin
    select UN_NAME, UN.UN_FROM_C_ID
    from USERNAME UN
    where UN_NAME = RDB$GET_CONTEXT('USER_SESSION', 'FINSTART_USER_LOGIN') and
          --UN_NAME in ('ADYIKANOVA', 'ISMAILOVA', 'EISKANDEROVA', 'IRYSALIEVA')
          UN_FROM_C_ID = :KAL_NEW_KF_OFFICER
    into UN_NAME, UN_FROM_C_ID;
    if (KAL_NEW_KF_OFFICER = coalesce(UN_FROM_C_ID, 0)) then
    begin
      if (KC_AMOUNT > 20000 and
          UN_NAME = 'ISMAILOVA') then
      begin
        res_error_text = 'Разрешено утверждать максимум 20000 с.';
        res_error_is_critical = 1;
        suspend;
      end
      if (KC_AMOUNT > 50000 and
          UN_NAME not in( 'ISMAILOVA')) then
      begin
        res_error_text = 'Разрешено утверждать максимум 50000 с.';
        res_error_is_critical = 1;
        suspend;
      end
    end
  end
end
