CREATE PROCEDURE CM_KG_0027_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable kf_id id_type;
declare variable reg_id integer;
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_percent_per_day perc_type;
declare variable pr_product_kind integer;
declare variable k_circle integer;
declare variable k_kommis_perc perc_type;
declare variable k_amount_approved curr_type;
declare variable pr_name varchar(50);
declare variable pr_code varchar(100);
declare variable kolich_poruch integer;
declare variable re_k_id id_type;
declare variable re_kol_cli integer;
declare variable re_percent_day perc_type;
declare variable kd_ol_date date;
declare variable kd_ol_date1 date;
declare variable re_pr_code varchar(100);
declare variable percent_snijen curr_type;
declare variable percent_snijen_plan curr_type;
declare variable re_k_id_cli_prior id_type;
declare variable max_delinq_day_cli integer;
declare variable c_id_delinq id_type;
declare variable max_kol_delinq integer;
declare variable re_ks_state integer;
declare variable c_id id_type;
declare variable re_k_id_cli id_type;
declare variable re_k_disb_date date;
declare variable re_k_close_date date;
declare variable re_ks_date_begin date;
declare variable is_repeat integer;
declare variable re_disb_date_prior date;
declare variable kol_delinq integer;
declare variable delinq_day_cli integer;
declare variable re_close_days integer;
begin
  for
  select k_disb_date, k_from_kf_id, k_last_reg_id, k_kolich_repay, k_kolich_client_approved, K_Percent_Per_Day,
           pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code
  from kontract, product
  where k_id=:k_id
  and k_from_pr_id = pr_id
  into :kf_disb_date, :kf_id, :reg_id, :k_kolich_repay, :k_kolich_client_approved, K_Percent_Per_Day,
                 :pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;
    --Определяем группа повторная или новая
    re_k_id = null;
    re_kol_cli = null;
    re_percent_day = 0;
    if (pr_code containing '.BIRGE.' and pr_code containing '.IND.') then
    begin
    select k_id, k_percent_per_day, ks_date_end - ks_date_begin as re_close_days
    from product, kontract, kontstate
    where pr_code containing '.BIRGE.' and pr_code containing '.IND.'
    and pr_id = k_from_pr_id
    and k_from_kf_id = :kf_id
    and k_id <> :k_id
    and k_id = ks_from_k_id
    and ks_state = 1
    and ks_date_begin < :kf_disb_date
    order by ks_date_begin desc
    rows 1
    into re_k_id, re_percent_day, re_close_days;
    end else
    begin
    --групповой
    select k_id, k_percent_per_day, k.k_kolich_client_approved, ks_date_begin, ks_state, ks_date_end - ks_date_begin as re_close_days
    from kontstate ks, kontract k, product
    where ks_ol_is_delete = 0
    and ks_state > 0
    and ks_date_begin <= :kf_disb_date
    and ks_from_k_id = k_id
    and k_from_kf_id = :kf_id
    and k_from_pr_id = pr_id
    and pr_code containing '.GR.'
    and pr_code not containing '.VYGOD.'
    and pr_code not containing '.PAR.'
    order by ks_date_begin desc
    rows 1
    into re_k_id, re_percent_day, re_kol_cli, re_ks_date_begin, re_ks_state, re_close_days;
    end
    
    if (re_k_id is null) then
    begin
    re_k_id = 0;
    re_kol_cli = 0;
    re_percent_day = 0;
    end
    
    percent_snijen = round(K_Percent_Per_Day * 365, 1);
    percent_snijen_plan = 0;
    --если кол-во повторных клиентов не менее 50% то данная группа считается повторной
    if ((cast(re_kol_cli as curr_type) >= cast(k_kolich_client_approved as curr_type) / 2 and pr_product_kind = 1)
     or (pr_code containing '.BIRGE.' and pr_code containing '.IND.' and re_k_id <> 0)
      ) then
    begin
    --             Проверка каждого клиента на предыдущий цикл
    re_k_id_cli_prior = 1;
    max_delinq_day_cli = 0;
    c_id_delinq = 0;
    res_info_text = null;
    KF_CLIENT_PASS_NO = null;
    max_kol_delinq = 0;
    for
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    into c_id
    do
    begin
     re_k_id_cli = null;
     is_repeat = 1;
     re_disb_date_prior = null;
     for
     select ks_from_k_id, ks_date_begin, ks_date_end, pr_code, ks_date_end - ks_date_begin as re_close_days
     from knclient, kontstate, kontract, product
     where kc_from_c_id = :c_id
     and kc_amount_approved > 0
     and kc_from_k_id = ks_from_k_id
     and ks_state = 1
     and ks_date_begin < :kf_disb_date
     and ks_from_k_id <> :k_id
     and ks_ol_is_delete = 0
     and ks_from_k_id = k_id
     and k_from_pr_id = pr_id
     order by ks_date_begin desc
     rows 2  -- последние 2 цикла вытаскивается
     into re_k_id_cli, re_k_disb_date, re_k_close_date, re_pr_code, re_close_days
     do
     begin
    --                   если последний цикл активный
    --                   то проверят следующий до этого бравший цикл
    --                   если до этого бравший цикл был закрыт после получения последнего цикла то тогда проверяется
         if (is_repeat = 1 and (re_disb_date_prior is null or re_disb_date_prior < re_k_close_date)) then
         begin
             is_repeat = 0;
             kol_delinq = 0;
             if (re_k_id_cli_prior <> re_k_id_cli and re_k_id_cli <> 0) then
             begin
               re_k_id_cli_prior = re_k_id_cli;
               delinq_day_cli = null;
               select max(ok_ol_date - ok_sh_date), count(distinct ok_sh_date) from operkred
               where ok_from_k_id = :re_k_id_cli
               and ok_ol_is_active = 1
               and ok_type in (4,5)
               and ok_os + ok_perc < 0
               and ok_ol_date > ok_sh_date
               into delinq_day_cli, kol_delinq;
               if (delinq_day_cli is null) then delinq_day_cli = 0;
               if (kol_delinq is null) then kol_delinq = 0;
               if (max_delinq_day_cli < delinq_day_cli) then
               begin
                 max_delinq_day_cli = delinq_day_cli;
                 c_id_delinq = :c_id;
               end
    
               if (max_kol_delinq < kol_delinq) then
               begin
                 max_kol_delinq = kol_delinq;
               end
             end
         end
    
         if (re_k_close_date > current_date) then -- значить предыдущий цикл активный, тогда позапредыдущий цикл тоже будет проверяться
         begin
           is_repeat = 1;
           re_disb_date_prior = re_k_disb_date;
         end 
     end
    end
    
    if (c_id_delinq <> 0) then
    begin
     select c_name_fio, c_pass_no
     from clients
     where c_id = :c_id_delinq
     into res_info_text, KF_CLIENT_PASS_NO;
    end
    if (re_close_days >= 80) then
    percent_snijen_plan = case
                           when max_delinq_day_cli > 30 then 200
                           when max_delinq_day_cli between 16 and 30 then 3.0
                           when max_delinq_day_cli between 8 and 15 then 2.0
                           when max_delinq_day_cli between 1 and 7 then 1.0
                           else - 0.2
                         end; else percent_snijen_plan = 0;
    if (re_pr_code = '.GR.' and round(re_percent_day * 365, 1) < 35) then percent_snijen_plan = round(re_percent_day * 365, 1);
    else percent_snijen_plan = round(re_percent_day * 365, 1) + percent_snijen_plan;
    if (percent_snijen_plan > 43.5) then percent_snijen_plan = 43.5;
    
    if (pr_code containing '.IND.' and pr_code containing '.BIRGE.'
       and pr_code not containing '.ECO.' and pr_code not containing '.PAR.'
       and percent_snijen_plan < 37) then percent_snijen_plan = 37;
    if (pr_code containing '.IND.' and pr_code containing '.BIRGE.'
       and pr_code containing '.ECO.' and pr_code not containing '.PAR.'
       and percent_snijen_plan <> 34.9) then percent_snijen_plan = 34.9;
    if (pr_code containing '.IND.' and pr_code containing '.BIRGE.'
       and pr_code not containing '.ECO.' and pr_code containing '.PAR.'
       and percent_snijen_plan < 35) then percent_snijen_plan = 35;
    
    if (pr_code = '.GR.') then
    begin
     if (percent_snijen_plan > 42.5) then percent_snijen_plan = 42.5;
     if (percent_snijen_plan < 33) then percent_snijen_plan = 33;
    
     if (:k_circle <= 1 and percent_snijen_plan < 35) then percent_snijen_plan = 35;
     if (re_ks_state = 1) then 
      begin
        res_error_is_critical = 2;
        res_error_text='Предыдущий цикл кредита активный!';
        suspend;
      end
    end
    if (pr_code = '.GR.VYGOD.') then
    begin
     if (percent_snijen_plan <> 33.9) then percent_snijen_plan = 33.9;
    end 
    if (pr_code = '.GR.PAR.') then
    begin
     if (percent_snijen_plan > 42.5) then percent_snijen_plan = 42.5;
    
     if (percent_snijen_plan < 33) then percent_snijen_plan = 33;
    end
    
    if (max_delinq_day_cli > 30 and percent_snijen < percent_snijen_plan
        and pr_code not containing '.vygod.' and pr_code not containing '.EC.') then
    begin
      res_error_is_critical = 2;
      res_error_text='В предыдущем цикле были просрочки более 30-дней, годовая процентная ставка должно быть: '||percent_snijen_plan;
      suspend;
    end
    if (max_delinq_day_cli between 16 and 30 and percent_snijen < percent_snijen_plan
        and pr_code not containing '.vygod.' and pr_code not containing '.EC.') then
    begin
     res_error_is_critical = 2;
     res_error_text='В предыдущем цикле были просрочки более 15-дней, годовая процентная ставка должно быть: '||percent_snijen_plan;
     suspend;
    end
    if (max_delinq_day_cli between 8 and 15 and percent_snijen < percent_snijen_plan
        and pr_code not containing '.vygod.' and pr_code not containing '.EC.') then
    begin
     res_error_is_critical = 2;
     res_error_text='В предыдущем цикле были просрочки более 8-дней, годовая процентная ставка должно быть: '||percent_snijen_plan;
     suspend;
    end
    if (max_delinq_day_cli between 1 and 7 and percent_snijen < percent_snijen_plan
        and pr_code not containing '.vygod.' and pr_code not containing '.EC.') then
    begin
     res_error_is_critical = 2;
     res_error_text='В предыдущем цикле были просрочки до 8-дней, годовая процентная ставка должно быть: '||percent_snijen_plan;
     suspend;
    end
    if (percent_snijen > percent_snijen_plan
        and pr_code not containing '.vygod.' and pr_code not containing '.EC.') then
    begin
     res_error_is_critical = 0;
     res_error_text='Вы указали годовую процентную ставку больше чем положено, должно быть: '||percent_snijen_plan;
     suspend;
    end
    end
    begin
    if (:k_circle = 1 and pr_product_kind = 1 and round(K_Percent_Per_Day * 365, 1) < 42.5
        and pr_code not containing '.vygod.' and pr_code not containing '.EC.') then
    begin
      res_error_is_critical = 0;
      res_error_text='Контракт новый, годовая процентная ставка должно быть 42,5%.';
      suspend;
    end
    if (:k_circle = 1
       and pr_code containing '.BIRGE.'
       and pr_code containing '.IND.'
       and pr_code not containing '.EC.'
       and re_k_id = 0
       and round(K_Percent_Per_Day * 365, 1) < 43.5) then
    begin
     res_error_is_critical = 0;
     res_error_text='Для новых индив. кредитов, годовая процентная ставка должно быть 43.5%.';
     suspend;
    end
    end
  end
end
