CREATE PROCEDURE AAREP_INVOICE_PRINT(
    REP_REGION_ID INTEGER,
    REP_IM_ID INTEGER,
    REP_PERIOD INTEGER)
RETURNS (
    AAREP_REGION_KPP VARCHAR(20),
    AAREP_REGION_ADDRESS VARCHAR(255),
    AAREP_INVOICE_NO VARCHAR(15),
    AAREP_INVOICE_DATE DATE,
    AAREP_CUSTOMER VARCHAR(100),
    AAREP_CUSTOMER_ADDRESS VARCHAR(255),
    AAREP_INV_AMOUNT DECIMAL(18,4),
    AAREP_INV_KOMMIS DECIMAL(18,4),
    AAREP_INV_PERC DECIMAL(18,4),
    AAREP_KONTRACT_NO VARCHAR(20),
    AAREP_DISB_DATE DATE)
AS
declare variable reg_code varchar(10);
declare variable k_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable reg_id integer;
declare variable im_id decimal(18,0);
begin
-- Формируем дату
  if (mod(rep_period, 100) < 12) then
  begin
    aarep_invoice_date = '01.' || right(rep_period + 1, 2) || '.' || left(rep_period, 4);
    aarep_invoice_date = aarep_invoice_date - 1;
  end
  else
    aarep_invoice_date = '31.12.' || left(rep_period, 4);

  if (coalesce(rep_im_id, 0) = 0) then
  begin -- Возвращаем список счетов-фактур
    for select reg_unicode, reg_code, im_id, im_invoice_no
        from region, region_list(:rep_region_id, 1, 3, 1), invoicemain
        where reg_unicode = id_list and
              im_from_reg_id = reg_unicode and
              im_period = :rep_period
        order by reg_unicode
        into reg_id, reg_code, :im_id, :aarep_invoice_no
    do
    begin
      aarep_region_kpp = '';
      aarep_region_address = '';
      select rp_kpp_no, adl_address_str
      from regprops, addrlist
      where :aarep_invoice_date between rp_date_begin and rp_date_end and
            rp_nalog_adl_id = adl_id and
            rp_from_reg_id = :reg_id
      into aarep_region_kpp, aarep_region_address;

      for select min(k_kontract_no), min(k_disb_date), sum(kcp_perc_calced - kcp_kommis_calced), sum(kcp_perc_calced),
                 sum(-kcp_kommis_calced), min(kcp_from_k_id)
          from kontcalcperc, invoicekont, kontract
          where ik_from_im_id = :im_id and
                kcp_period = :rep_period and
                ik_from_k_id = kcp_from_k_id and
                k_id = ik_from_k_id
          into aarep_kontract_no, aarep_disb_date, aarep_inv_amount, aarep_inv_perc, aarep_inv_kommis, k_id
      do
      begin
        c_id = null;
        select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        order by kc_status desc
        rows 1
        into :c_id;
        aarep_customer = '';
        aarep_customer_address = '';

        if (c_id is not null) then
        begin
          select c_name_fio, adl_address_str
          from clients, addrclient, addrlist
          where adc_from_c_id = c_id and
                adc_address_type = 1 and
                adc_from_adl_id = adl_id and
                c_id = :c_id
          into aarep_customer, aarep_customer_address;
        end

        aarep_invoice_no = right('000000000' || aarep_invoice_no, 9) || '/' || left(reg_code, 2) || '-' || substring(reg_code from 3 for 2);
        suspend;
      end
    end
  end
  else
  begin -- Возвращаем одну счет-фактуру

    for select reg_unicode, reg_code, im_id, im_invoice_no
        from region, invoicemain
        where im_from_reg_id = reg_unicode and
              im_period starting(:rep_period / 100) and
              im_id = :rep_im_id
        into reg_id, reg_code, :im_id, :aarep_invoice_no
    do
    begin
      aarep_region_kpp = '';
      aarep_region_address = '';
      select rp_kpp_no, adl_address_str
      from regprops, addrlist
      where :aarep_invoice_date between rp_date_begin and rp_date_end and
            rp_nalog_adl_id = adl_id and
            rp_from_reg_id = :reg_id
      into aarep_region_kpp, aarep_region_address;

      for select min(k_kontract_no), min(k_disb_date), sum(kcp_perc_calced - kcp_kommis_calced), sum(kcp_perc_calced),
                 sum(-kcp_kommis_calced), min(kcp_from_k_id)
          from kontcalcperc, invoicekont, kontract, invoicemain
          where ik_from_im_id = im_id and
                im_id = :im_id and
                kcp_period = im_period and
                ik_from_k_id = kcp_from_k_id and
                k_id = ik_from_k_id
          into aarep_kontract_no, aarep_disb_date, aarep_inv_amount, aarep_inv_perc, aarep_inv_kommis, k_id
      do
      begin
        c_id = null;
        select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        order by kc_status desc
        rows 1
        into :c_id;
        aarep_customer = '';
        aarep_customer_address = '';

        if (c_id is not null) then
        begin
          select c_name_fio, adl_address_str
          from clients, addrclient, addrlist
          where adc_from_c_id = c_id and
                adc_address_type = 1 and
                adc_from_adl_id = adl_id and
                c_id = :c_id
          into aarep_customer, aarep_customer_address;
        end

        aarep_invoice_no = right('000000000' || aarep_invoice_no, 9) || '/' || left(reg_code, 2) || '-' || substring(reg_code from 3 for 2);
        suspend;
      end
    end

  end
end
