CREATE PROCEDURE FIND_KFOLDER(
    FOLDER_STR VARCHAR(20),
    CLIENT_STR VARCHAR(255),
    OFFICER_STR VARCHAR(255),
    DD DATE)
RETURNS (
    KFOL_ID DECIMAL(18,0),
    KFOL_NAME VARCHAR(100),
    KFOL_MEMO VARCHAR(1024),
    KFOL_REG_ID INTEGER,
    KFOL_OFF_ID DECIMAL(18,0),
    KFOL_OFF_CODE VARCHAR(20),
    KFOL_NUMBER INTEGER,
    KFOL_OFF_NAME VARCHAR(255),
    KFOL_IS_NEW INTEGER)
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  folder_str = upper(folder_str);
  client_str = upper(client_str);
  officer_str = upper(officer_str);
  if (client_str = '') then
  begin
    for select kf_id, kf_memo, kal_new_kf_officer, kal_new_kf_name, kal_new_no, kal_new_reg_id, kal_new_officer_code,
               c1.c_name_fio, kf_is_new
        from kfolder, knaktlist, clients c1
        where kfolder.is_delete = 0 and
              kf_id = kal_from_kf_id and
              kal_new_kf_officer = c1.c_id and
              kal_date <= :dd and
              kal_date_end >= :dd and
              knaktlist.is_delete = 0 and
              ((:folder_str = '') or
              (cast(kal_from_kf_id as varchar(20)) like '%' || :folder_str || '%') or
              (cast(kal_new_no as varchar(20)) like '%' || :folder_str || '%') or
              (upper(kf_memo) like '%' || :folder_str || '%') or
              (upper(kal_new_kf_name) like '%' || :folder_str || '%') or
              (upper(kal_new_officer_code) like '%' || :folder_str || '%')) and
              ((:officer_str = '') or
              (upper(c1.c_name_fio) like '%' || :officer_str || '%'))
        order by kal_new_kf_name
        into :kfol_id, :kfol_memo, :kfol_off_id, :kfol_name, :kfol_number, :kfol_reg_id, :kfol_off_code, :kfol_off_name,
             :kfol_is_new
    do
    begin
      suspend;
    end
  end
  else
  begin
    for select kf_id, kf_memo, kal_new_kf_officer, kal_new_kf_name, kal_new_no, kal_new_reg_id, kal_new_officer_code,
               c1.c_name_fio, kf_is_new
        from kfolder, knaktlist, clients c1, get_kf_from_client(:client_str)
        where kfolder.is_delete = 0 and
              kf_id = kal_from_kf_id and
              kal_new_kf_officer = c1.c_id and
              kf_id = kfc_id and
              kal_date <= :dd and
              kal_date_end >= :dd and
              knaktlist.is_delete = 0 and
              ((:folder_str = '') or
              (cast(kal_from_kf_id as varchar(20)) like '%' || :folder_str || '%') or
              (upper(kf_memo) like '%' || :folder_str || '%') or
              (upper(kal_new_kf_name) like '%' || :folder_str || '%') or
              (upper(kal_new_officer_code) like '%' || :folder_str || '%')) and
              ((:officer_str = '') or
              (upper(c1.c_name_fio) like '%' || :officer_str || '%'))
        order by kal_new_kf_name
        into :kfol_id, :kfol_memo, :kfol_off_id, :kfol_name, :kfol_number, :kfol_reg_id, :kfol_off_code, :kfol_off_name,
             :kfol_is_new
    do
    begin
      suspend;
    end
  end
end
