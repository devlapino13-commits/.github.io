CREATE PROCEDURE GET_CLI_CONTACT_DATA(
    C_ID ID_TYPE)
RETURNS (
    CC_ID INTEGER,
    CC_FROM_C_ID ID_TYPE,
    CC_NAME STR_100,
    CC_STATUS STR_100,
    CC_ADDRESS STR_255,
    CC_CHECK_RESULT INTEGER,
    CC_IN_PROCESS INTEGER,
    CC_TRY_KOL INTEGER,
    CC_CALL_USER STR_255,
    CC_TIME_BEGIN TIMESTAMP,
    CC_TIME_END TIMESTAMP,
    CC_FROM_UN_ID ID_TYPE,
    CC_CHECK_MEMO STR_255,
    CC_TYPE INTEGER,
    CC_CLIENT_OK SMALLINT,
    CC_NAME_OK SMALLINT,
    CC_STATUS_OK SMALLINT,
    CC_ADDRESS_OK SMALLINT,
    CC_CREATE_DATE TIMESTAMP,
    CC_CHECKED_DATETIME TIMESTAMP,
    CC_EDIT_DATE TIMESTAMP,
    CC_OPER_INFO STR_255,
    CC_LANGUAGE INTEGER,
    CC_FROM_FL_ID ID_TYPE,
    CC_FROM_ADL_ID ID_TYPE,
    CC_SHORTNAME STR_100,
    CC_DATE DATE,
    CC_TOTAL_RES INTEGER,
    CC_CHECKED_DATE DATE,
    CC_IS_OLD_CONTACT SMALLINT,
    CC_BLOCKED_USER VARCHAR(255),
    CC_CELL_NO STR_255,
    ADDRESS STR_255,
    COMMENT STR_255,
    LANGUAGE STR_100,
    CC_ADL_ADRESS STR_255,
    LANGUAGE_STR STR_100,
    TOTAL_RES STR_100)
AS
begin
  for select cc_id, cc_from_c_id, cc_name, cc_status, cc_address, cont.cc_cell_no, cc_check_result, cc_in_process,
             cc_try_kol, cc_call_user, cc_time_begin, cc_time_end, cc_from_un_id, cc_check_memo, cc_type, cc_client_ok,
             cc_name_ok, cc_status_ok, cc_address_ok, cc_create_date, cc_checked_datetime, cc_edit_date, cc_oper_info,
             cc_language, cc_from_fl_id, cc_from_adl_id, cc_shortname, cc_date, cc_total_res, cc_checked_date,
             cc_is_old_contact, cc_blocked_user
      from contacts cont
      where cc_from_c_id = :c_id
      order by cont.cc_id
      into cc_id, cc_from_c_id, cc_name, cc_status, cc_address, cc_cell_no, cc_check_result, cc_in_process, cc_try_kol,
           cc_call_user, cc_time_begin, cc_time_end, cc_from_un_id, cc_check_memo, cc_type, cc_client_ok, cc_name_ok,
           cc_status_ok, cc_address_ok, cc_create_date, cc_checked_datetime, cc_edit_date, cc_oper_info, cc_language,
           cc_from_fl_id, cc_from_adl_id, cc_shortname, cc_date, cc_total_res, cc_checked_date, cc_is_old_contact,
           cc_blocked_user
  do
  begin
    address = '';
    comment = '';

    select adl_address_str, adl_comment
    from addrlist adl
    where adl.adl_id = :cc_from_adl_id
    into :address, :comment;

    if (address is null) then
      address = '';
    if (comment is null) then
      comment = '';

    language = '';

    select ddr.rd_name
    from refdata ddr
    where ddr.rd_id = :cc_language
    into :language;

    if (language is null) then
      language = '';

    cc_adl_adress = :address;
    language_str = :language;

    total_res = (case cc_total_res
                   when 0 then 'В процессе'
                   when 1 then 'Подтвержден'
                   when -1 then 'Не дозвонились'
                   when -2 then 'Не подтвержден'
                 end);
    suspend;
  end
end
