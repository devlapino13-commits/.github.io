CREATE PROCEDURE HP_DO_KONT_FOR_RESTRUCT_SAVE(
    K_ID D_ID)
AS
declare variable kfr_id d_integer;
declare variable res_is_error  d_smallint;
begin
  select kf.kfr_id
  from kont_for_restruct_kz kf
  where kf.kfr_from_k_id = :k_id
    and kf.kfr_is_delete = 0
  order by kfr_id desc
  rows 1
  into kfr_id;

  select res_is_error
  from hp_do_reschedule_kz(:k_id, current_date)
  into res_is_error;
  if (res_is_error = 0) then
    update kont_for_restruct_kz_dtl kfd
    set kfd.krd_is_done = 1
    where kfd.krd_from_kfr_id = :kfr_id and
          kfd.krd_is_done = 0;
end
