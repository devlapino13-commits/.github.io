CREATE PROCEDURE HP_BKI_CHECK_SENT_UIDS(
    DATE_BEGIN D_DATE)
RETURNS (
    MA_ID D_ID,
    C_NAME_FIO D_STR_255,
    SIGN_DATE D_DATE,
    BKK_NAME D_STR_CODE,
    CUR_EVENT D_STR_CODE,
    RES_TEXT D_STR_255)
AS
declare variable bkk_id d_id;
  declare variable c_id d_id;
  declare variable app_date d_date;
  declare variable max_event d_integer;
  declare variable max_event_date d_date;
  declare variable uid_cnt d_integer;
  declare variable ev_new_app d_integer;
  declare variable ev_approve d_integer;
  declare variable ev_conclud d_integer;
  declare variable ev_annul d_integer;
  declare variable ev_disb d_integer;
  declare variable events d_str_255;
begin
  ev_new_app  = f_enum('bki_event', '1.1_new_app',      'id');
  ev_approve  = f_enum('bki_event', '1.2_app_approved', 'id');
  ev_annul    = f_enum('bki_event', '1.3_app_annul',    'id');
  ev_conclud  = f_enum('bki_event', '1.4_deal_concluded', 'id');
  ev_disb     = f_enum('bki_event', '2.2_disb',         'id');

  for with t1(bkk_id, c_id, object_kind, object_id, event_id, event_date) as (
    select distinct bkp_from_bkk_id, bkc_from_c_id, bkl_object_kind, bkl_object_id, bev_event_id, bev_event_date
    from bki_events, bki_clients, bki_loans, bki_packet, bki_packet_type
    where bkc_id = bev_from_bkc_id
      and bkl_id = bkc_from_bkl_id
      and bkp_id = bkl_from_bkp_id
      and bkt_id = bkp_from_bkt_id
      and bkt_format = 0
      and bkp_date >= :date_begin
      and bev_event_date >= :date_begin - 10
      and bev_status > 0
      and bkp_status > 0
      and coalesce(bev_delete_bev_id, 0) = 0
      and bev_event_id in (:ev_approve, :ev_annul, :ev_disb)
  ), t2 as(
    select bkk_id, c_id, object_id ma_id, event_id, event_date
    from t1
    where object_kind = 11
    union 
    select bkk_id, c_id, k_from_ma_app_id, event_id, event_date
    from t1, kontract
    where object_kind = 9
      and k_id = object_id
  )
  select bkk_id, c_id, ma_id, max(event_id), max(event_date)
  from t2
  group by 1,2,3
  into bkk_id, c_id, ma_id, max_event, max_event_date
  do
  begin
    sign_date = null;
    res_text = '';

    with t_objects(object_kind, object_id) as(
      select 11, :ma_id
      from rdb$database
    union
      select 9, k_id
      from kontract k
      where k_from_ma_app_id = :ma_id
    )
    select list(distinct '|' || bev_event_id || '|'), count(distinct bkc_uid_bkd_id)
    from t_objects, bki_packet_type, bki_packet, bki_loans, bki_clients, bki_events
    where bkp_from_bkt_id = bkt_id
      and bkl_from_bkp_id = bkp_id
      and bkc_from_bkl_id = bkl_id
      and bev_from_bkc_id = bkc_id
      and bkt_format = 0
      and bev_status > 0
      and bkp_status > 0
      and bkp_from_bkk_id = :bkk_id
      and coalesce(bev_delete_bev_id, 0) = 0
      and bkl_object_kind = object_kind
      and bkl_object_id = object_id
      and bev_event_date <= :max_event_date
    into events, uid_cnt;

    if (uid_cnt > 1) then
      res_text = res_text || 'Отправлены различные UID.  ';

    if (events not containing '|' || ev_new_app || '|') then
      res_text = res_text || 'Событие 1.1 не найдено.  ';

    if (max_event >= ev_disb) then
    begin
      if (events not containing '|' || ev_approve || '|') then
        res_text = res_text || 'Событие 1.2 не найдено.  ';

      if (events not containing '|' || ev_conclud || '|') then
        res_text = res_text || 'Событие 1.4 не найдено.  ';
    end

    if (res_text <> '') then
    begin
      select min(eds_status_datetime)
      from edoc_sign, edoc_list, edoc_blank edb
      where eds_from_edl_id = edl_id
        and edb_from_edl_id = edl_id
        and edl_from_edc_id = f_edoc_kind('loan.application','id')
        and edl_object_kind = 11
        and edl_object_id = :ma_id
        and eds_status = 1
      into sign_date;

      if (sign_date is null) then
      begin
        app_date = (select min(ma_date_app) from miniapp ma where ma.ma_id = :ma_id);
        if (app_date < '01.01.2021') then
          sign_date = app_date;
      end 

      if (sign_date is null) then
        res_text = res_text || 'Нет подписанной заявки.  ';

      bkk_name = (select bkk_name from bki_kind where bkk_id = :bkk_id);
      c_name_fio = (select c_name_fio from clients where c_id = :c_id);
      cur_event = f_enum('bki_event', max_event, 'code');

      suspend;
    end
  end
end
