CREATE PROCEDURE FM_HTML_EXTINFO_INN_ENTITY(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable lang d_str_lang;
declare variable eil_date d_date;
declare variable bls_id d_id;
declare variable c_name_fio d_str_255;
declare variable calc_status d_str_255;
declare variable json d_blob_text;
declare variable db_id d_integer;
declare variable response d_blob_text;
begin
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);
  select eil_date, eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id and
        eil_from_eic_id = eic_id and
        eic_code = 'INN_entity' and
        eil_is_json = 1 and
        eil_status = 2
  order by eil_date desc, eil_id
  rows 1
  into eil_date, bls_id;
  c_name_fio = f_lang(lang, 'Клиент не найден');
  if (row_count = 0) then
    calc_status = f_lang(lang, 'Проверка юр.лица по ИНН не получена');
  else
  begin
    if (coalesce(bls_id, 0) <= 0) then
      calc_status = f_lang(lang, 'Проверка юр.лица по ИНН по клиенту отсутствует.');
    else
    begin
      calc_status = f_format('<br>%1: %2', f_lang(lang, 'Дата проверки'), f_datetostr(eil_date));
    end
  end
  select c_name_fio
  from clients
  where c_id = :c_id
  into c_name_fio;
  res_html = (select res_style
              from fm_html_styles) || f_format('<body><a name="top"></a><h2>%1</h2>%2<br>', c_name_fio, calc_status);
  select res_blob
  from fm_blob_get(:bls_id)
  into :json;

  if (db_id = 94401 and (:json containing 'Информация о клиенте не найдена' or :json containing 'Информация не найдена')) then
  begin
    res_html = BLOB_APPEND(res_html, '<br> Информация о клиенте не найдена </br>');
    suspend;
    exit;
  end

  if ((coalesce(:json, '') <> '') and (coalesce(:json, '') <>'Информация не найдена')) then
  begin
  response = f_json_str_get_value(:json, 'Response', '');
  if (coalesce(response, '') = '') then
  begin
    res_html = BLOB_APPEND(res_html, 'Информация не найдена');
    suspend;
    exit;
  end
  res_html = BLOB_APPEND(res_html, f_format('<h1>%1</h1><table width="100%"><col width="40%"><col width="60%">', f_lang(lang, 'Проверка юр.лица по ИНН')));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Имя') || '</b>', f_json_str_get_value(:json, 'Response.Chief.Name', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Фамилия') || '</b>', f_json_str_get_value(:json, 'Response.Chief.Surname', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Отчество') || '</b>', f_json_str_get_value(:json, 'Response.Chief.Patronymic', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Адрес по прописке') || '</b>', f_json_str_get_value(:json, 'Response.NaturalPerson.LegalAddress', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Тип плательщика') || '</b>', f_json_str_get_value(:json, 'Response.PayerType.Description', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Дата создания') || '</b>', f_json_str_get_value(:json, 'Response.VOENInfo.CreateDate', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Дата последнего обновления') || '</b>', f_json_str_get_value(:json, 'Response.VOENInfo.LastUpdateDate', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'Статус') || '</b>', f_json_str_get_value(:json, 'Response.VOENInfo.StateDescription', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, f_html_table_row('<b>' || f_lang(lang, 'ИНН') || '</b>', f_json_str_get_value(:json, 'Response.VOENInfo.VOEN', ''), '', '', -1, -1, -1, -1, 2, ''));
  res_html = BLOB_APPEND(res_html, '</table>');
  end
  suspend;
end
