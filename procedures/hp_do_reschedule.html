CREATE PROCEDURE HP_DO_RESCHEDULE(
    K_ID D_ID,
    REP_DATE D_DATE,
    ACTION_KIND D_STR_CODE,
    ADD_PARAMS D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255,
    OL_ID D_ID)
AS
declare variable sh_rec_id                  d_integer;
declare variable sh_is_grace_perc           d_smallint;
declare variable tmp_amount                 d_currency;
declare variable add_month                  d_integer;
declare variable count_prolog               d_integer;
declare variable count_grace_os             d_integer;
declare variable count_grace_perc           d_integer;
declare variable reg_id                     d_integer;
declare variable reg_name                   d_str_100;
declare variable last_date                  d_date;
declare variable ol_memo                    d_str_long;
declare variable protocol_date              d_date;
declare variable reason                     d_integer;
declare variable rest_kind                  d_integer;
declare variable lgota_perc_date            d_date;
declare variable kd_percent                 d_percent;
declare variable kd_percent_per_day         d_percent;
declare variable kd_percent_effect          d_percent;
declare variable kd_fine_percent            d_percent;
declare variable pr_percent_method          d_integer;
declare variable first_perc_kind            d_smallint;
declare variable cm_database_id             d_integer;
declare variable perc_lgota_value           d_currency;
declare variable new_kd_percent             d_percent;
declare variable new_kd_percent_per_day     d_percent;
declare variable new_kd_percent_effect      d_percent;
declare variable new_kd_fine_percent        d_percent;
declare variable kd_percent_effect_fifd     d_percent;
declare variable new_kd_percent_effect_fifd d_percent;
declare variable kps_id                     d_id;
declare variable stop_perc                  boolean;
begin
  res_is_error = 0;
  res_text = '';

  if (coalesce(k_id, 0) = 0) then
    exception e_default 'Input parameter K_ID is null';

  if (action_kind is null) then
    exception e_default 'action_kind is null';

  if (action_kind not in ('load', 'auto', 'calc', 'new_annu', 'grace_perc', 'save', 'add_month')) then
    exception e_default 'Unknown action_kind: ' || action_kind;

  cm_database_id = f_get_db_id();
  stop_perc = (cm_database_id = 39801)
    and exists(select * from kontract where k_id = :k_id and coalesce(k_stop_perc, '31.12.9999') <= :rep_date);

  first_perc_kind = 1;
  perc_lgota_value = 0;

  if (add_params = '') then
    add_params = '{}';

  if (add_params <> '') then
  begin
    perc_lgota_value = f_json_str_get_value(add_params, 'perc_lgota_value', 0); -- значение скидки или новой %% ставки

    first_perc_kind = f_json_str_get_value(add_params, 'first_perc_kind', null);
    if (first_perc_kind is null) then
    begin
      first_perc_kind = 1;
      if (cm_database_id = 81001) then first_perc_kind = 0;

    end
  end
  add_params = f_json_str_set_value(add_params, 'first_perc_kind', first_perc_kind);

  if (action_kind in ('load', 'auto')) then
  begin
    select iif(s_running_total <> 0, 1, 0)
    from hp_repay_fill(:k_id, 0, :rep_date, 1, 0, 0, 0, 0, 0)
    into :sh_rec_id;

    select pr_percent_method from temp_params
    into pr_percent_method;

    delete from temp_sh
    where 1 = 1;

    insert into temp_sh (sh_id, sh_date, sh_date_orig, sh_repay_mode,
                         sh_zad_os, sh_zad_perc, sh_zad_fine,
                         sh_new_os, sh_new_perc, sh_new_fine,
                         sh_is_grace_os, sh_is_grace_perc, sh_is_grace_os_orig, sh_is_grace_perc_orig,
                         sh_annu, sh_annu_orig, sh_calcing_perc, sh_prior_percent_per_day)
    select s_id, s_date, s_date, repay_mode,
           zad_os, zad_perc, zad_fine,
           zad_os, iif(:stop_perc, 0, zad_perc), zad_fine,
           is_grace, 0, is_grace, 0,
           iif(:pr_percent_method in (-1, 2), zad_os + zad_perc, zad_os),
           iif(:pr_percent_method in (-1, 2), zad_os + zad_perc, zad_os),
           calcing_perc_orig, sh_prior_percent_per_day
    from temp_vrepay
    order by 2, 1;

    delete from temp_vrepay
    where 1 = 1;

    if (action_kind = 'auto') then
    begin
      select res_is_error, res_text
      from hp_do_reschedule_auto(:k_id, :rep_date, :action_kind, :add_params, :un_id)
      into :res_is_error, :res_text;
    end

    if (res_is_error = 0) then
    begin
      select 'Реструктуризация графика погашений займа ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date)
      from kontract
      where k_id = :k_id
      into res_text;
    end
  end

  if (action_kind = 'grace_perc') then
  begin
    sh_rec_id = f_json_str_get_value(add_params, 'sh_rec_id', '0');

    if (sh_rec_id = 0) then exit;

    select sh_is_grace_perc
    from temp_sh
    where sh_rec_id = :sh_rec_id
    into sh_is_grace_perc;

    if (sh_is_grace_perc = 1) then
    begin
      update temp_sh
      set sh_is_grace_os = iif(sh_repay_mode <> 1, 1, sh_is_grace_os_orig),
          sh_is_grace_perc = iif(sh_repay_mode <> 1, 1, sh_is_grace_perc_orig)
      where sh_rec_id <= :sh_rec_id;
    end else
    begin
      update temp_sh
      set sh_is_grace_os = iif(sh_repay_mode <> 1, 0, sh_is_grace_os_orig),
          sh_is_grace_perc = iif(sh_repay_mode <> 1, 0, sh_is_grace_perc_orig)
      where sh_rec_id >= :sh_rec_id and
            sh_repay_mode <> 1;
    end

    update temp_sh
    set sh_is_grace_os = 0,
        sh_is_grace_perc = 0
    where 1 = 1
    order by sh_id desc
    rows 1;

    execute procedure hp_do_reschedule_calc(:k_id, :rep_date, 1, :first_perc_kind, :perc_lgota_value);
  end

  if (action_kind = 'new_annu') then
  begin
    update temp_sh
    set sh_is_grace_os = 0,
        sh_is_grace_perc = 0
    where 1 = 1
    order by sh_id desc
    rows 1;

    tmp_amount = f_json_str_get_value(add_params, 'amount', '0');
    if (tmp_amount = 0)
    then
      execute procedure hp_do_reschedule_calc(:k_id, :rep_date, 1, :first_perc_kind, :perc_lgota_value);
    else
    begin
      update temp_sh
      set sh_annu = iif(sh_is_grace_os = 1 or sh_is_grace_perc = 1, 0, :tmp_amount)
      where sh_repay_mode <> 1;

      execute procedure hp_do_reschedule_calc(:k_id, :rep_date, 0, :first_perc_kind, :perc_lgota_value);
    end
  end

  if (action_kind = 'calc') then
    execute procedure hp_do_reschedule_calc(:k_id, :rep_date, 0, :first_perc_kind, :perc_lgota_value);

  if (action_kind = 'add_month') then
  begin
    add_month = f_json_str_get_value(add_params, 'count', '0');
    if (add_month = 0) then Exit;

    execute procedure hp_do_reschedule_add_period(:add_month, 0);
    execute procedure hp_do_reschedule_calc(:k_id, :rep_date, 1, :first_perc_kind, :perc_lgota_value);
  end

  if (action_kind = 'save') then
  begin
    if (res_text = '') then
    begin
      select max(ok_ol_date)
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_from_c_id = 0
      into last_date;

      if (last_date > rep_date) then
      begin
        res_is_error = 1;
        res_text = 'Есть операции по данному займу более поздней датой: ' || f_datetostr(last_date);
      end
    end

    if (res_text = '') then
    begin
      if (not exists(
      select * from temp_sh
      where sh_date_orig is null or
            sh_zad_os <> sh_new_os or
            sh_zad_perc <> sh_new_perc or
            sh_date <> sh_date_orig)) then
      begin
        res_is_error = 1;
        res_text = 'Не найдены изменения в графике погашений';
      end
    end

    if (res_text = '') then
    begin
      select a.kal_new_reg_id, reg_code || ' ' || reg_name
      from knaktlist a, kontract, region
      where k_id = :k_id and
            k_from_kf_id = a.kal_from_kf_id and
            a.is_delete = 0 and
            a.kal_date <= :rep_date and
            a.kal_date_end >= :rep_date and
            a.kal_new_reg_id = reg_unicode
      into reg_id, reg_name;
  
      if (exists(select *
                 from hp_is_day_closed(:reg_id, :rep_date, 0)
                 where day_closed = 1)) then
      begin
        res_is_error = 1;
        res_text = 'Операционный день закрыт, офис ' || reg_name;
      end
    end

    if (res_text = '') then -- Сохраняем операции
    begin
      -- Значение OperRest.OR_Kind:
  
      -- 0 Обычная реструктуризация
      -- 1 Обычная с льгот.период. по ОС
      -- 2 Обычная с льгот.период. по ОС и %%
      -- 3 Пролонгация без льгот. периода
      -- 4 Пролонгация с льгот.ОС
      -- 5 Пролонгация с льгот.ОС и %%
  

      lgota_perc_date = null;
      select sum(iif(sh_date_orig is null, 1, 0)),
             sum(iif(sh_is_grace_os <> sh_is_grace_os_orig, 1, 0)),
             sum(iif(sh_is_grace_perc <> sh_is_grace_perc_orig, 1, 0))
      from temp_sh
      into count_prolog, count_grace_os, count_grace_perc;
      if (count_prolog is null) then count_prolog = 0;
      if (count_grace_os is null) then count_grace_os = 0;
      if (count_grace_perc is null) then count_grace_perc = 0;

      select sh_date
      from temp_sh
      where sh_is_grace_perc = 0 and
            sh_date >= :rep_date
      order by sh_date
      rows 1
      into lgota_perc_date;

      rest_kind = 0;
      if (count_prolog = 0) then
      begin
        if (count_grace_os <> 0 and count_grace_perc <> 0)
        then rest_kind = 2;
        else
          if (count_grace_os <> 0)
          then rest_kind = 1;
      end else
      begin
        rest_kind = 3;
        if (count_grace_os <> 0 and count_grace_perc <> 0)
        then rest_kind = 4;
        else
          if (count_grace_os <> 0)
          then rest_kind = 4;
      end

      ol_memo = f_json_str_get_value(add_params, 'memo', '');
      protocol_date = f_json_str_get_value(add_params, 'protocol_date', null);
      reason = f_json_str_get_value(add_params, 'reason', 0);

      insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                            ol_memo, ol_parent_id, is_delete, ol_user_id, ol_from_rd_id)
      values (:reg_id, 1, 4, :rep_date, 9, :ol_memo, null, 0, :un_id, :reason)
      returning ol_id
      into :ol_id;

      if (lgota_perc_date is not null) then
      begin
        kd_percent = 0;
        kd_percent_per_day = 0;
        kd_percent_effect = 0;
        kd_percent_effect_fifd = 0;
        kd_fine_percent = 0;
  
        select kd_percent, kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_fine_percent
        from kontdata
        where kd_from_k_id = :k_id
        and kd_ol_is_delete = 0
        and :rep_date between kd_ol_date and kd_date_end
        into kd_percent, kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_fine_percent;

        new_kd_percent = kd_percent;
        new_kd_percent_per_day = kd_percent_per_day;
        new_kd_percent_effect = kd_percent_effect;
        new_kd_percent_effect_fifd = kd_percent_effect_fifd;

        new_kd_fine_percent = kd_fine_percent;

        if (perc_lgota_value <> 0) then
        begin
          if (perc_lgota_value = 100) then
          begin
            new_kd_percent = 0;
            new_kd_percent_per_day = 0;
            new_kd_fine_percent = 0;
          end else
          begin
            new_kd_percent = kd_percent * ((100.0000 - perc_lgota_value) / 100.0000);
            new_kd_percent_per_day = kd_percent_per_day * ((100.0000 - perc_lgota_value) / 100.0000);
          end

          insert into kontdata (kd_from_k_id, kd_from_ol_id, kd_ol_date, kd_ol_is_delete,
                                kd_percent, kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_fine_percent)
          values (:k_id, :ol_id, :rep_date, 0,
                  :new_kd_percent, :new_kd_percent_per_day, :new_kd_percent_effect, :new_kd_percent_effect_fifd, :new_kd_fine_percent);
    
          insert into kontdata (kd_from_k_id, kd_from_ol_id, kd_ol_date, kd_ol_is_delete,
                                kd_percent, kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_fine_percent)
          values (:k_id, :ol_id, :lgota_perc_date, 0,
                  :kd_percent, :kd_percent_per_day, :kd_percent_effect, :kd_percent_effect_fifd, :kd_fine_percent);
        end
      end

      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_pay_type,
                            ok_fine, ok_sh_os, ok_sh_perc,
                            ok_ol_date, ok_ol_reg_id, ok_ol_is_active)
      -- изменения в сумме
      select :ol_id, -1, :k_id, sh_id, sh_date, 2, 3,
             0.0, sh_new_os - sh_zad_os, sh_new_perc - sh_zad_perc,
             :rep_date, :reg_id, 1
      from temp_sh
      where sh_date = sh_date_orig and
            (sh_zad_os <> sh_new_os or sh_zad_perc <> sh_new_perc)
      union all
      -- новые даты
      select :ol_id, -1, :k_id, sh_id, sh_date, 2, 3,
             0.0, sh_new_os, sh_new_perc,
             :rep_date, :reg_id, 1
      from temp_sh
      where sh_date_orig is null
      union all
      -- если дата менялась, убираем старые суммы из старой даты
      select :ol_id, -1, :k_id, sh_id, sh_date_orig, 2, 3,
             0.0, -sh_zad_os, -sh_zad_perc,
             :rep_date, :reg_id, 1
      from temp_sh
      where sh_date_orig is not null and
            sh_date <> sh_date_orig
      union all
      -- если дата менялась, добавляем новые суммы на новые даты
      select :ol_id, -1, :k_id, sh_id, sh_date, 2, 3,
             0.0, sh_new_os, sh_new_perc,
             :rep_date, :reg_id, 1
      from temp_sh
      where sh_date_orig is not null and
            sh_date <> sh_date_orig
--      union
--      -- списываем пени если есть и передан в параметре
--      select :ol_id, -1, :k_id, sh_id, sh_date, 5, 3,
--             -sh_zad_fine, 0.0, 0.0,
--             :rep_date, :reg_id, 1
--      from temp_sh
--      where sh_zad_fine <> 0 and
--            :fine_writeoff = 1
      order by 4, 5, 6;

      insert into operrest (or_from_ol_id, or_from_k_id, or_kind, or_ol_date, or_ol_is_active, or_from_rd_id, or_from_reg_id, or_date_prot)
      values (:ol_id, :k_id, :rest_kind, :rep_date, 1, :reason, :reg_id, :protocol_date);


      kd_percent_effect_fifd = null;
      select kps_id, kps_psk
      from hp_edoc_ins_kontpsk(:k_id, :ol_id, :un_id)
      into kps_id, kd_percent_effect_fifd;

      if (kps_id is not null) then
      begin
        update kontdata
        set kd_percent_effect_fifd = :kd_percent_effect_fifd
        where kd_from_k_id = :k_id and
              kd_ol_date >= :rep_date and
              kd_ol_is_delete = 0;
      end

      if (cm_database_id = 81001 and rep_date >= '01.01.2024') then
      begin
        -- После реструктуризации не повышаем ставку
        if (not exists (
              select * from kontdata
              where kd_from_k_id = :k_id and
              kd_ol_date = :rep_date and
              kd_ol_is_delete = 0)) then
        begin
         -- Получаем обновленные значения ставки, раз уж вызвали, берем обе
          select effect_rate, effect_rate_common
          from hp_effect_rate_common(:k_id, :ol_id, 1)
          into kd_percent_effect_fifd, kd_percent_effect;

          -- Добавляем новую запись, текущей датой
          -- Оригинал взят из HP_KONTVAC_CREATE, там тоже kd_float_perc_level = 999
          insert into kontdata (kd_from_k_id, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent,
                                kd_percent_per_day, kd_percent_effect, kd_percent_effect_fifd, kd_memo,
                                kd_fine_percent, kd_float_perc_level, kd_vac_status)
          select kd_from_k_id, :ol_id, :rep_date, 0, kd_percent, kd_percent_per_day,
                 :kd_percent_effect, :kd_percent_effect_fifd, 'Остановка повышения плавающей ставки при реструктуризации',
                 kd_fine_percent, 999, kd_vac_status
          from kontdata
          where kd_from_k_id = :k_id and
                kd_ol_date <= :rep_date and
                kd_ol_is_delete = 0
          order by kd_ol_date desc
          rows 1;
        end

        -- Контрольно обновляем float_perc_level текущую и будущую даты
        -- т.к. например, каникулы по %% добавляют 2 записи в kontdata
        update kontdata
        set kd_float_perc_level = 999
        where kd_from_k_id = :k_id and
              kd_ol_date >= :rep_date and
              kd_ol_is_delete = 0;
      end
    end
  end

  suspend;
end
