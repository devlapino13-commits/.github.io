CREATE PROCEDURE HP_BKI_GET_PACKET_CONTENT(
    BKP_ID D_ID)
RETURNS (
    REC_NO D_INTEGER,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    EVENT_CODE D_STR_CODE,
    EVENT_DATE D_DATE,
    EVENT_TIME D_TIME,
    TAG D_INTEGER,
    ERROR_CODE D_STR_CODE,
    ERROR_TEXT D_STR_LONG,
    FIXED_DATETIME D_DATETIME,
    REC_STATUS D_INTEGER,
    CRED_STATUS D_STR_CODE,
    UID D_STR_CODE,
    REC_ID D_ID)
AS
begin
  for select bkl_object_kind, bkl_object_id, bev_tag, bev_event_date, bev_event_time,
         bev_status, f_enum('bki_event', bev_event_id, 'code', 0), bev_tag, bev_rec_no,
         bke_error_code, f_memo_text(bke_error_id), bke_fixed_datetime,
         f_bki_data_text(bkc_uid_bkd_id), bev_id, bev_loan_status
  from bki_loans
  left join bki_clients on bkc_from_bkl_id = bkl_id
  left join bki_events on bev_from_bkc_id = bkc_id
  left join bki_errors on bke_from_bev_id = bev_id
  where bkl_from_bkp_id = :bkp_id
  order by bev_rec_no, bkl_id, bkc_id, bev_event_date
  into object_kind, object_id, tag, event_date, event_time,
    rec_status, event_code, tag, rec_no,
    error_code, error_text, fixed_datetime, uid, rec_id, cred_status
  do
    suspend;
end
