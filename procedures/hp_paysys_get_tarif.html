CREATE PROCEDURE HP_PAYSYS_GET_TARIF(
    K_ID DECIMAL(18,0),
    PSP_ID INTEGER,
    INSURED_AMOUNT INTEGER)
RETURNS (
    POLICY_AMOUNT DECIMAL(18,4),
    POLICY_AMOUNT_TAX DECIMAL(18,4),
    POLICY_PERC DECIMAL(15,2),
    MONTH_COUNT INTEGER,
    RES_TEXT VARCHAR(255))
AS
declare variable dd date;
declare variable date_end date; /* Дата окончания действия полиса */
declare variable date_begin date; /* Дата начала действия полиса */
begin
  /* 11.03.14 changed by Gusev Igor
     month_count - теперь тянется из контракта (поле K_Kolich_Repay) а не рассчитывается в скрипте
  */

  policy_amount = 0;
  policy_amount = 0;
  month_count = 0;
  policy_amount = 0;
  policy_perc = 0;
  policy_amount_tax = 0;
  date_begin = null;
  date_end = null;
  select k_disb_date, k_kredline_end, k_kolich_repay
  from kontract
  where k_id = :k_id
  into :date_begin, :date_end, :month_count;
  if (date_begin is null or date_end is null) then
  begin
    res_text = 'Контракт не найден. Или не указаны даты начала окончания договора';
    suspend;
    exit;
  end
  res_text = 'Тариф по проекту не определен';
  if (psp_id = 2) then
  begin
    res_text = '';
    /*
      Расчет количества месяцев страхования.
      В случае, не полного месяца - до 10 дней в расчет не включаются,
      а от 10 дней и выше рассчитывается как за полный  месяц.
    */

--month_count = datediff(month from date_begin to date_end);
    if (date_begin > date_end) then
    begin
--month_count = 0;
      res_text = 'Дата начала должна быть раньше даты окончания';
      suspend;
      exit;
    end

    if (month_count = 0) then
      month_count = 1;
    else
    begin
--month_count = month_count - 1;
      if (extract(month from date_begin) < 12) then
      begin
        dd = '01.' ||(extract(month from date_begin) + 1) || '.' || extract(year from date_begin);
        dd = dd - 1;
      end
      else
        dd = '31.12.' || extract(year from date_end);

      /*
      if (extract(day from dd) - extract(day from date_begin) > 9) then
        month_count = month_count + 1;
  
      if (extract(day from date_end) > 9) then
        month_count = month_count + 1;
      */
    end
    if (month_count > 0) then
    begin
      policy_perc = case
                      when month_count between 1 and 9 then 2.61
                      when month_count between 10 and 12 then 2.90
                      when month_count between 13 and 15 then 3.63
                      when month_count between 16 and 18 then 4.35
                      else 0
                    end;
      if (policy_perc > 0) then
      begin
        policy_amount = round(insured_amount * policy_perc / 100.0, 0);
--        policy_amount_tax = policy_amount - round(policy_amount / 1.02, 0);
        policy_amount_tax = 0;
      end
      else
        res_text = 'Для срока кредита: ' || month_count || ' мес. тариф не определен';
    end
    else
      res_text = 'Слишком короткий срок страхования, тариф не может быть применен';
  end
  suspend;
end
