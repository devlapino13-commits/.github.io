CREATE PROCEDURE HP_OPERCURR_DEL(
    CU_ID D_STR_LANG,
    CR_DATE D_DATE)
RETURNS (
    IS_ERROR D_INTEGER,
    RES_TEXT D_STR_255)
AS
begin
  -- удаляет курс валюты в opercurr

  is_error = 1;
  res_text = '';

  if (cr_date < current_date) then
    res_text = 'Ошибка: Нельзя удалить курс на прошедшую дату';

  if (res_text = '') then
  begin
    delete from opercurr
    where cr_from_cu_id = :cu_id and
          cr_date = :cr_date;

    if (row_count = 1) then
    begin
      is_error = 0;
      res_text = 'Курс валюты успешно удален';
    end
  end

  suspend;
end
