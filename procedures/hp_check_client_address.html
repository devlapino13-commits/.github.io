CREATE PROCEDURE HP_CHECK_CLIENT_ADDRESS(
    C_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER,
    ADL_ADDRESS_STR D_STR_255,
    ADL_CORRECT_ADDRESS D_STR_255)
AS
declare variable place_id d_id;
declare variable street_id d_id;
declare variable post_index d_str_20;
declare variable address_str d_str_255;
begin
  -- Процедура проверяет текст адреса регистрации с выбранным населенным пунктом и улицей.

  res_info_text = '';
  res_error_text = '';
  res_error_is_critical = 0;

  select c_name_fio
  from clients
  where c_id = :c_id
  into res_info_text;

  select adl_from_pl_id, adl_street_id, adl_post_index, adl_address_str
  from addrclient, addrlist
  where adc_from_c_id = :c_id and
        adc_address_type = 1 and
        adc_from_adl_id = adl_id
  into place_id, street_id, post_index, adl_address_str;

  address_str = '';

  if (place_id is null) then
  begin
    res_error_text = 'Адрес регистрации не заполнен';
    res_error_is_critical = 2;
  end else
  begin
    select address_str
    from hp_make_addr_place(0, :place_id, :street_id, null, null, :post_index)
    into :address_str;
  
    if (address_str <> '') then
    begin
      if (position(replace(address_str, ' ', ''), replace(adl_address_str, ' ', '')) = 0) then
      begin
        adl_correct_address = address_str;
        address_str = '';
        select address_str
        from hp_make_addr_place(1, :place_id, :street_id, null, null, :post_index)
        into :address_str;
  
        if (address_str <> '') then
        begin
          if (position(replace(address_str, ' ', ''), replace(adl_address_str, ' ', '')) = 0) then
          begin
            res_error_text = 'Адрес регистрации не совпадает с выбранным населенным пунктом или улицей в адресе. Заполните адрес повторно.';
            res_error_is_critical = 2;
          end
        end
      end
    end
  end

  suspend;
end
