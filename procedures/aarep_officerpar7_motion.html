CREATE PROCEDURE AAREP_OFFICERPAR7_MOTION(
    REP_DATE DATE)
RETURNS (
    REGION_NAME VARCHAR(100),
    OFFICE_NAME VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    REPAY_SUM CURR_TYPE,
    CURR_PAR7 DECIMAL(5,5),
    CURR_BALANCE CURR_TYPE,
    LAST_PAR7 DECIMAL(5,5),
    LAST_BALANCE DECIMAL(18,0),
    RAZN_PAR7 CURR_TYPE,
    OP_DATE_BEGIN DATE,
    LAST_KOL_CLI INTEGER,
    CURR_KOL_CLI INTEGER,
    MEMO VARCHAR(50),
    CURR_DELINQ_BALANCE7 DECIMAL(18,4),
    LAST_DELINQ_BALANCE7 DECIMAL(18,4))
AS
declare variable op_kc_id decimal(18,0);
declare variable op_reg_id integer;
declare variable op_reg_id_prior integer;
begin
  op_reg_id_prior = 0;
  for select op_kc_id, op_date_begin,
--         OP_BALANCE,
--         OP_PAR8_180,
             op_reg_id
      from officerpar7
      where :rep_date between op_date_begin and op_date_end and
            op_par8_180 >= 7.0000
      order by op_reg_id
      into :op_kc_id, :op_date_begin,
--       :last_balance,
--       :last_par7,
           :op_reg_id
  do
  begin
    if (op_reg_id_prior <> op_reg_id) then
    begin
      op_reg_id_prior = op_reg_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :op_reg_id and
            r1.parent = r2.reg_unicode
      into office_name, region_name;
    end
    select c_name_fio
    from clients
    where c_id = :op_kc_id
    into officer_name;
--  сумма выплат за день, в руб
    repay_sum = null;
    select -sum(ok_os + ok_perc + ok_fine)
    from knaktlist, kontract, operkred
    where kal_new_kf_officer = :op_kc_id and
          knaktlist.is_delete = 0 and
          :rep_date between kal_date and kal_date_end and
          kal_from_kf_id = k_from_kf_id and
          k_id = ok_from_k_id and
          ok_ol_is_active = 1 and
          ok_ol_date = :rep_date and
          ok_type in (4, 9) and
          ok_os + ok_perc + ok_fine < 0
    into repay_sum;
    repay_sum = coalesce(repay_sum, 0);

--  общая сумма задолженности по портфелю СВЗ в руб на дату приостановления
    last_balance = null;
    last_delinq_balance7 = null;
    last_kol_cli = null;
    select sum(oks_balance), sum(iif(:op_date_begin - oks_delinq_date > 7, oks_balance, 0)), sum((select res_just_client_count
                                                                                                  from hp_get_just_client_in_loan(:op_date_begin, k_id, k_disb_date)))
    from oksummary, kontract
    where :op_date_begin between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :op_kc_id and
          oks_from_k_id = k_id
          and oks_close_state = 0
    into last_balance, last_delinq_balance7, last_kol_cli;
    last_balance = coalesce(last_balance, 0);
    last_delinq_balance7 = coalesce(last_delinq_balance7, 0);
    last_kol_cli = coalesce(last_kol_cli, 0);
    if (last_balance > 0) then
      last_par7 = cast(last_delinq_balance7 as double precision) / last_balance * 100;
    else
      last_par7 = 0;

--  общая сумма задолженности по портфелю СВЗ в руб на данный день
    curr_balance = null;
    curr_delinq_balance7 = null;
    select sum(oks_balance), sum(iif(:rep_date - oks_delinq_date > 7, oks_balance, 0)), sum((select res_just_client_count
                                                                                             from hp_get_just_client_in_loan(:rep_date, k_id, k_disb_date)))
    from oksummary, kontract
    where :rep_date between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :op_kc_id and
          oks_from_k_id = k_id
          and oks_close_state = 0
    into curr_balance, curr_delinq_balance7, curr_kol_cli;
    curr_balance = coalesce(curr_balance, 0);
    curr_delinq_balance7 = coalesce(curr_delinq_balance7, 0);
    curr_kol_cli = coalesce(curr_kol_cli, 0);
    if (curr_balance > 0) then
      curr_par7 = cast(curr_delinq_balance7 as double precision) / curr_balance * 100;
    else
      curr_par7 = 0;

--  уровень PAR7 на день введения (23 апреля)
--  общая сумма задолженности по портфелю СВЗ в руб на день введения (23 апреля)
    razn_par7 = curr_par7 - last_par7;

    memo = 'в кадровых приказах не регистрирован';
    select iif(ps_is_working = 0, 'уволен от ' || ps_date_begin, '')
    from hr_pstatus
    where :rep_date between ps_date_begin and ps_date_end and
          ps_status = 1 and
          ps_from_c_id = :op_kc_id
    into memo;

    suspend;
  end
end
