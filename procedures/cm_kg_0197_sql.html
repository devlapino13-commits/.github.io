CREATE PROCEDURE CM_KG_0197_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable state              d_smallint;
declare variable c_id               d_id;
declare variable pr_code            d_str_255;
declare variable gc_status_datetime d_datetime;
begin
  if (f_grs_active(:un_id) = 0) then exit;
  if (ma_id = 3912776) then exit;
  if (ma_id = 3690517) then exit;

  for select mac.mac_from_c_id, pr.pr_code
      from miniapp ma
      inner join miniappclient mac on ma.ma_id = mac.mac_from_ma_id and
            mac.mac_amount > 0
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      where ma.ma_is_active = 1 and
            ma.ma_id = :ma_id and
            ma.ma_date_end = '31.12.9999' and
            pr.pr_code not containing '.jonokoi_cards.' and
            pr.pr_code not containing '.installment_goods.' and
            pr.pr_code not containing '.sugarmill.' and
            pr.pr_code not containing '.online_loan.'          -- RFCKG-2691
      into :c_id, :pr_code
  do
  begin
   -- if (current_date <= '05.07.2024' and pr_code containing '.online_loan.') then
   --   exit;

    select state
    from hp_check_pasport_actual_grs(:c_id)
    into state;

    if (state = 0) then
    begin
      select gc.gc_status_datetime
      from grs_clients gc
      where gc.gc_from_c_id = :c_id and
            gc.gc_is_last = 1
            rows 1
      into gc_status_datetime;
      -- если нет ответа от ГРС, но была ручная верификация после последней отправки в ГРС, пропускаем через блок
      if (exists(select *
                 from actionlist acl
                 where acl.acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO')) and
                       acl.acl_object_kind = f_objkind('clients', 'id') and
                       acl.acl_object_id = :c_id and
                       acl.acl_status = 2 and
                       acl.acl_last_datetime > :gc_status_datetime)) then
        state = 1;
    end

    if (state = 0) then
    begin
      res_error_text = 'Информация с ГРС не получена или просроченная, прошу отправить запрос в ГРС!';
      res_error_is_critical = 1;
      suspend;
    end
  end
end
