CREATE PROCEDURE HP_ADM_DASHB_CONFIRM_CODE(
    ADC_USERNAME VARCHAR(50) NOT NULL,
    ADC_PASSWORD VARCHAR(50) NOT NULL,
    ADC_CODE INTEGER NOT NULL)
RETURNS (
    ADC_TOKEN VARCHAR(45),
    ADC_ROLE VARCHAR(10))
AS
declare variable v_adu_id integer;
begin
  select adu_role, adu_id
  from adm_dashb_users
  where adu_active = 1 and
        adu_username = :adc_username and
        adu_password = :adc_password and
        adu_code = :adc_code
        /*  and adu_code_expired_at < current_timestamp   */
  into :adc_role, v_adu_id;

  if (v_adu_id is not null) then
  begin
    adc_token = (select uuid_to_char(gen_uuid())
                 from rdb$database);
    update adm_dashb_users
    set adu_session_token = :adc_token,
        adu_session_token_expired_at = dateadd(10 minute to current_timestamp)
    where adu_id = :v_adu_id;

    suspend;
  end

end
