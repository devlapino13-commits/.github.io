CREATE PROCEDURE HP_CLONE_BLANKBLOB(
    ORIG_BLN_ID D_INTEGER,
    ORIG_BLB_NUMBER D_INTEGER,
    NEW_DATE D_DATE,
    UN_ID D_ID)
RETURNS (
    BLB_NUMBER D_INTEGER)
AS
begin
  insert into blankblob (blb_from_bln_id, blb_date_begin, blb_blob, blb_params,
                         blb_frx_blob, blb_json_blob, blb_frx_status, blb_font_style, blb_font_size, blb_from_un_id)
  select blb_from_bln_id, :new_date, blb_blob, blb_params, blb_frx_blob,
         blb_json_blob, blb_frx_status, blb_font_style, blb_font_size, :un_id
  from blankblob
  where blb_from_bln_id = :orig_bln_id and
        blb_number = :orig_blb_number
  returning blb_number
  into blb_number;

  insert into blankjsonlink (bjl_from_bln_id, bjl_from_blb_number, bjl_from_bjf_id)
  select bjl_from_bln_id, :blb_number, bjl_from_bjf_id
  from blankjsonlink
  where bjl_from_bln_id = :orig_bln_id and
        bjl_from_blb_number = :orig_blb_number;


  Suspend;
end
