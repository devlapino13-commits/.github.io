CREATE PROCEDURE HP_BKI_XML_LIST_UX(
    XML_VERSION_ID D_ID)
RETURNS (
    BFV_ID D_ID,
    POS D_INTEGER,
    CODE D_STR_CODE,
    NAME D_STR_255,
    DB_FIELD D_STR_255,
    XML_FIELD D_STR_CODE,
    FORCED_VALUE D_STR_255,
    XML_ID D_ID,
    PARENT_ID D_ID,
    MAIN_CODE D_STR_255)
AS
begin
  if (xml_version_id is null) then
    exit;

  for with recursive tab as(
    select bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bfv_forced_value, bxf_id, bfv_parent_id
    from bki_xml_fields_ver, bki_xml_fields, bki_indicators
    where bxf_id = bfv_from_bxf_id
      and bki_code = bxf_from_bki_code
      and bfv_from_bxv_id = :xml_version_id
    order by bfv_position
  ), t1 (lvl, id, pos, code, name, db, xml, def, xml_id, parent_id, main_code) as(
    select 0, bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bfv_forced_value, bxf_id, bfv_parent_id, bki_code
    from tab
    where bfv_parent_id is null
    union all
    select lvl + 1, bfv_id, bfv_position, bki_code, bki_name, bki_db_field, bxf_xml_node, bfv_forced_value, bxf_id, bfv_parent_id, main_code
    from t1, tab
    where bfv_parent_id = id
  )
  select id, pos, code, name, db, rpad('', 2*lvl, '| ')|| xml, def, xml_id, parent_id, main_code
  from t1
  into bfv_id, pos, code, name, db_field, xml_field, forced_value, xml_id, parent_id, main_code
  do
    suspend;
end
