CREATE PROCEDURE HP_MORPRONES_SEARCH(
    FIND_C_ID ID_TYPE,
    FIND_CELL_NO STR_100,
    FIND_FIO STR_100,
    FIND_CODE VARCHAR(20))
RETURNS (
    RES_DESCRIPTION STR_255)
AS
declare variable last_cell_main str_100;
declare variable last_cell_add str_100;
declare variable ok_cell_main str_100;
declare variable ok_cell_add str_100;
begin
  res_description = '';

  find_cell_no = coalesce(find_cell_no, '');
  find_fio = coalesce(find_fio, '');
  find_code = coalesce(find_code, '');
  find_c_id = coalesce(find_c_id, 0);

  if (find_c_id = 0) then
  begin
    if (find_cell_no <> '') then
    begin
      select mp.mp_description
      from morphones mp
      where mp.mp_cell_no = :find_cell_no rows 1
      into :res_description;

      if (res_description <> '') then
        suspend;
    end

    if ((find_fio <> '') and
        (res_description = '')) then
    begin
      for select f_cont_info('clients', c_id, 'cell_main', 'last'),
                 f_cont_info('clients', c_id, 'cell_add', 'last'),
                 f_cont_info('clients', c_id, 'cell_main', 'ok'),
                 f_cont_info('clients', c_id, 'cell_add', 'ok')
          from clients
          where c_name_fio = :find_fio
          into :last_cell_main, :last_cell_add, :ok_cell_main, :ok_cell_add
      do
      begin
        select mp.mp_description
        from morphones mp
        where (mp.mp_cell_no = :last_cell_main) or
              (mp.mp_cell_no = :last_cell_add) or
              (mp.mp_cell_no = :ok_cell_main) or
              (mp.mp_cell_no = :ok_cell_add)
        rows 1
        into :res_description;

        if (res_description <> '') then
          suspend;
      end
    end

    if ((find_code <> '') and
        (res_description = '')) then
    begin
      for select f_cont_info('clients', c_id, 'cell_main', 'last'),
                 f_cont_info('clients', c_id, 'cell_add', 'last'),
                 f_cont_info('clients', c_id, 'cell_main', 'ok'),
                 f_cont_info('clients', c_id, 'cell_add', 'ok')
          from clients
          where c_inn = :find_code
          into :last_cell_main, :last_cell_add, :ok_cell_main, :ok_cell_add
      do
      begin
        select mp.mp_description
        from morphones mp
        where (mp.mp_cell_no = :last_cell_main) or
              (mp.mp_cell_no = :last_cell_add) or
              (mp.mp_cell_no = :ok_cell_main) or
              (mp.mp_cell_no = :ok_cell_add) rows 1
        into :res_description;

        if (res_description <> '') then
          suspend;
      end
    end
  end
  else
  begin
    last_cell_main = f_cont_info('clients', find_c_id, 'cell_main', 'last');
    last_cell_add = f_cont_info('clients', find_c_id, 'cell_add', 'last');
    ok_cell_main = f_cont_info('clients', find_c_id, 'cell_main', 'ok');
    ok_cell_add = f_cont_info('clients', find_c_id, 'cell_add', 'ok');

    select mp.mp_description
    from morphones mp
    where (mp.mp_cell_no = :last_cell_main) or
          (mp.mp_cell_no = :last_cell_add) or
          (mp.mp_cell_no = :ok_cell_main) or
          (mp.mp_cell_no = :ok_cell_add) rows 1
    into :res_description;

    if (res_description <> '') then
      suspend;
  end
end
