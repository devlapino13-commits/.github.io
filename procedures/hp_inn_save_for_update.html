CREATE PROCEDURE HP_INN_SAVE_FOR_UPDATE
AS
declare variable C_ID D_ID;
declare variable C_INN D_STR_20;
declare variable IS_SUCCESS_UPDATE D_BOOL;
begin
    -- RFCRU-3596 Доработка механизма сохранения ИНН
    -- Процедура делает обновление в карточке клиента из временной таблицы
    -- Если обновление успешное то запись удаляется (is_success_update)
    for select
            ifu.ifu_from_c_id,
            ifu.ifu_inn
    from inn_for_update ifu
    into c_id,
         c_inn
    do
    begin
        is_success_update = 1;

        begin
            update clients
               set c_inn = :c_inn
            where c_id = :c_id
            and c_inn is distinct from :c_inn;
            when any do
                is_success_update = 0;
        end

        if (is_success_update = 1) then
            delete from inn_for_update ifu where ifu.ifu_from_c_id = :c_id;
    end
end
