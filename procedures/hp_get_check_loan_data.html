CREATE PROCEDURE HP_GET_CHECK_LOAN_DATA(
    K_ID D_ID)
RETURNS (
    RD_ID D_INTEGER,
    RD_CODE D_STR_255)
AS
declare variable pr_code d_str_100;
declare variable db_id d_integer;
begin
  pr_code = '';
  db_id = f_get_db_id();
  if (db_id = 81001) then
  begin
    select p.pr_code
    from kontract k
    inner join product p on p.pr_id = k.k_from_pr_id
    where k.k_id = :k_id
    into :pr_code;
  end
  rd_code = 'ActionGroup.Check_Loan_Data';
  if (pr_code containing '.kk_special.') then
    rd_code = 'ActionGroup.Check_Loan_Data_Special';

  rd_id = 0;
  select r.rd_id
  from refdata r
  where r.rd_code = :rd_code
  into :rd_id;
  suspend;
end
