CREATE PROCEDURE HP_PERMISSION_REG_LIST(
    UN_ID DECIMAL(18,0))
RETURNS (
    REG_ID INTEGER,
    REG_PARENT_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REG_R_CHECKED INTEGER,
    REG_R_DATE_BEGIN DATE,
    REG_R_DATE_END DATE,
    REG_R_PM_ID INTEGER,
    REG_FULL_PATH VARCHAR(255))
AS
declare variable parent_tmp integer;
declare variable uniqtab_username integer;
declare variable uniqtab_region integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Region')
  into :uniqtab_region;

  for select reg_unicode, parent, reg_code, reg_name
      from region
      into :reg_id, :reg_parent_id, :reg_code, :reg_name
  do
  begin
    parent_tmp = reg_id;
    reg_full_path = '[' || parent_tmp || ']';
    while (parent_tmp is not null) do
    begin
      select parent
      from region
      where reg_unicode = :parent_tmp
      into :parent_tmp;
      if (parent_tmp is not null) then
        reg_full_path = '[' || parent_tmp || ']' || reg_full_path;
    end
    reg_r_checked = 0;
    reg_r_pm_id = -1;
    reg_r_date_begin = null;
    reg_r_date_end = null;

    select pm_id, pm_date_begin, pm_date_end
    from permissions
    where pm_object_kind = :uniqtab_username and
          pm_object_id = :un_id and
          pm_subject_kind = :uniqtab_region -- Офис для чтения
          and
          pm_subject_id = :reg_id and
          pm_is_delete = 0 and
          coalesce(pm_date_begin, '01.01.2000') <= current_date and
          coalesce(pm_date_end, '31.12.2999') >= current_date rows 1
    into :reg_r_pm_id, :reg_r_date_begin, :reg_r_date_end;
    if (reg_r_pm_id is null) then
      reg_r_pm_id = -1;
    if (reg_r_pm_id <> -1) then
      reg_r_checked = 1;

    suspend;
  end
end
