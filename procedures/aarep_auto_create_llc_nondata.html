CREATE PROCEDURE AAREP_AUTO_CREATE_LLC_NONDATA(
    REP_DATE DATE,
    USER_ID ID_TYPE)
RETURNS (
    ERROR_TEXT STR_255)
AS
declare variable QTY integer;
declare variable REP_DATE_BEGIN date;
declare variable REP_DATE_END date;
declare variable DB_ID integer;
begin
  rep_date = dateadd(-1 month to coalesce(rep_date, current_date));
  rep_date_begin = dateadd(1 - extract(day from rep_date) day to rep_date);
  rep_date_end = dateadd(1 month to rep_date_begin) - 1;

  db_id = f_get_db_id();

  insert into llc_create_log (lcl_un_id, lcl_date)
  values (:user_id, :rep_date);

  if (db_id = 94401) then -- если АЗ
  begin
    execute procedure aarep_auto_create_az_llc(:rep_date_begin, :rep_date_end);
  end
  else
  if (db_id = 81001) then -- если РФ
  begin
    select count(*)
    from hp_llc_create_ru_2014_09(:rep_date_begin, :rep_date_end)
    into :qty;
  end
  else
  if (db_id in (39801, 41701)) then -- если это КГ
  begin
    select count(*)
    from aarep_auto_create_llc(:rep_date_begin, :rep_date_end)
    into :qty;

    select count(*)
    from hp_llc_create_cli(1000000, :rep_date_end)
    into :qty;
  end
  error_text = '';
  suspend;
end
