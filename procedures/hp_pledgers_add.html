CREATE PROCEDURE HP_PLEDGERS_ADD(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    C_ID D_ID)
RETURNS (
    RES_TEXT D_STR_LONG)
AS
declare variable rd_id d_integer;
declare variable acl_id d_id;
declare variable un_id d_id;
declare variable obj_kind_id d_integer;
begin
  -- RFCRU-2633

  select un_id
  from username
  where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN')
  into un_id;

  obj_kind_id = f_objkind(obj_kind, 'id');
  --rd_id = f_get_rd_id('ActionGroup.Check_Loan_Data');

  if (obj_kind = 9) then
  begin
    select rd_id
    from hp_get_check_loan_data(:obj_id)
    into :rd_id;
  end

  -- проверка в КК - на этапе обработки
  if (exists(select *
             from actionlist
             where acl_object_kind = :obj_kind_id and
                   acl_object_id = :obj_id and
                   acl_status = 1 and
                   acl_from_rd_id = :rd_id)) then
  begin
    res_text = 'Договор находится на рассмотрении КК. Изменения запрещены.';
    suspend;
    exit;
  end

  for select acl_id
      from actionlist
      where acl_object_kind = :obj_kind_id and
            acl_object_id = :obj_id and
            acl_status = 2 and
            acl_from_rd_id = :rd_id
      into :acl_id
  do
  begin
    update actionlist
    set acl_status = 0,
        acl_from_un_id = :un_id
    where acl_id = :acl_id;

    res_text = 'Повторно отправлен на КК в связи с добавлением или обновлением залогодателя чз МП';
    execute procedure hp_objectprops_memo_set(9, :obj_id, -1, :rd_id, res_text, un_id);
  end

  update or insert into zalog (z_from_k_id, z_from_c_id, z_type, z_subtype)
  values (:obj_id, :c_id, 1, 4)
  matching (z_from_k_id, z_type, z_subtype);

  res_text = 'Залогодатель был добавлен успешно';
  suspend;
end
