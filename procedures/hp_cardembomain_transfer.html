CREATE PROCEDURE HP_CARDEMBOMAIN_TRANSFER(
    CEM_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    OL_ID D_ID,
    ERR_CODE D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable card_count d_integer;
declare variable acc_dt1 d_str_20;
declare variable acc_kt1 d_str_20;
declare variable acc_dt2 d_str_20;
declare variable acc_kt2 d_str_20;
declare variable ol_memo d_str_255;
begin
  err_code = 0;
  err_text = '';

  select
    cem_app_count,
    cem_from_ol_id
  from cardembomain
  where cem_id = :cem_id
    and cem_from_ces_id = 3
  into
    :card_count,
    :ol_id;
  if (row_count = 0) then
  begin
    err_code = 1;
    err_text = 'Ошибка изменения статуса заявки №' || cem_id;
    suspend;
    exit;
  end

  if (ol_id is null or
    exists(
      select *
      from operlist
      where ol_id = :ol_id
        and is_delete = 1)) then
  begin
    select
      cpl_from_reg_id,
      c_id,
      c_name_fio
    from cardplace
      inner join clients
        on c_id = cpl_from_c_id
    where cpl_id = 0
    into
      :reg_id,
      :c_id,
      :c_name_fio;

    acc_dt1 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_account = '96101/12');
    acc_kt1 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_account = '96101/10');
    acc_dt2 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_account = '96102/10');
    acc_kt2 = (select al_id from acc_list where al_from_reg_id = :reg_id and al_account = '96102/12');
    ol_memo = 'Передача заготовок карт в "' || c_name_fio || '"; заявка № ' || cem_id || ', количествo ' || card_count || 'шт.';

    insert into operlist (
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id,
      ol_document_client
     )
    values (
      :reg_id,
      2,
      2,
      current_date,
      :cem_id,
      0,
      :ol_memo,
      :un_id,
      :c_id
    )
    returning ol_id
    into :ol_id;

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      '417',
      1,
      :acc_dt1,
      :acc_kt1,
      :card_count,
      :card_count
    );

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      '417',
      1,
      :acc_dt2,
      :acc_kt2,
      :card_count,
      :card_count
    );

    update cardembomain set
      cem_from_ces_id = 4,
      cem_from_ol_id = :ol_id,
      cem_from_un_id = :un_id
    where cem_id = :cem_id;

    err_code = 0;
    err_text = 'Заявка №' || cem_id || ' получила статус "Заготовки переданы"';
  end else
  begin
    err_code = 1;
    err_text = 'Ошибка создания операции по передаче заготовок карт в МПЦ; заявка № ' || cem_id;
  end

  suspend;
end
