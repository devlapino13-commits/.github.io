CREATE PROCEDURE AAREP_TOP_SVZ_EX(
    REP_DATE D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    REG_NAME D_STR_100,
    PLACE D_STR_255,
    CELL_NO D_STR_20,
    K_KOL D_INTEGER,
    DATE_BEGIN DATE,
    DATE_END DATE)
AS
declare variable kc_id d_id;
declare variable reg_id d_integer;
declare variable ps_from_reg_id d_id;
declare variable otpusk_days d_smallint;
declare variable n d_smallint;
declare variable dd_tmp date;
declare variable lo_work_days d_smallint;
declare variable lo_kol integer;
begin
  lo_kol = 0;
  --отчет будет формироваться в субботу роботом, за отчетный период: прошлая суббота - текущая пятница
  date_begin = dateadd(-7 day to :rep_date); --прошлая суббота
  date_end = dateadd(-1 day to :rep_date); --текущая пятница

  for select distinct(hps.ps_from_c_id), hps.ps_from_reg_id, count(ks_from_k_id) k_kol
      from hr_pstatus hps
      inner join joblist jl on jl.jl_id = ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind and
            coalesce(jk.jk_memo, '') <> 'оригинал: JobSystem.Director_General' and
            jk.jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior') --берем все свз
      left join kontstate ks on ps_from_c_id = ks_kc_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :date_begin and :date_end
      where :rep_date between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            hps.ps_status = 1
      group by 1, 2
      order by k_kol
      into :kc_id, :ps_from_reg_id, k_kol
  do
  begin
    -- Проверяем статус отпуска на каждый день недели:
    -- если все 7 дней был в отпуске, исключаем из списка
    otpusk_days = 0;
    n = 0;
    while (n <= 6) do
    begin
      dd_tmp = dateadd(day, n, date_begin);
      if (exists(select *
                 from hr_voctravel, hr_orders, sa_worktype
                 where hrv_from_ho_id = ho_id and
                       ho_order_type = sawt_id and
                       hrv_from_c_id = :kc_id and
                       sawt_base_type in (4, 5) and -- отпуск, декретный отпуск
                       sawt_id <> 20 and
                       sawt_hours_real >= 0 and
                       hrv_date_begin <= :dd_tmp and
                       hrv_date_end_fact >= :dd_tmp and
                       hrv_status = 1)) then
        otpusk_days = otpusk_days + 1;
      n = n + 1;
    end
    lo_work_days = 7 - otpusk_days;

    if (lo_work_days > 0) then
    begin
      lo_kol = lo_kol + 1;

      c_name_fio = null;
      cell_no = null;
      place = null;

      select pl.pl_name, r.reg_name
      from region r
      inner join addrlist al on adl_id = reg_address_fact
      inner join places pl on al.adl_from_pl_id = pl.pl_id
      where r.reg_unicode = :ps_from_reg_id
      into :place, :reg_name;

      select cl.c_name_fio
      from clients cl
      where cl.c_id = :kc_id
      into :c_name_fio;

      cell_no = f_cont_info('Clients', kc_id, 'cell_main', 'ok');
      if (cell_no is null) then
        cell_no = f_cont_info('Clients', kc_id, 'cell_main', 'last');

      suspend;
    end

    if (lo_kol = 50) then
      exit;
  end
end
