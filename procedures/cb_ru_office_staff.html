CREATE PROCEDURE CB_RU_OFFICE_STAFF(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    JL_NAME VARCHAR(255),
    OFFICER_NAME VARCHAR(100),
    AAREP_BONUS_PLUS CURR_TYPE,
    AAREP_BONUS_MINUS CURR_TYPE,
    AAREP_BONUS_TOTAL CURR_TYPE,
    AAREP_BONUS_TOTAL_FACT CURR_TYPE,
    AAREP_BONUS_OFFICER CURR_TYPE,
    SUM_MINUS CURR_TYPE,
    AAREP_DELINQ_OS CURR_TYPE,
    AAREP_DELINQ_PERC CURR_TYPE,
    AAREP_FACT_OS CURR_TYPE,
    AAREP_FACT_PERC CURR_TYPE,
    AAREP_FACT_FINE CURR_TYPE,
    AAREP_FACT_DELINQ_OS CURR_TYPE,
    AAREP_FACT_DELINQ_PERC CURR_TYPE,
    AAREP_FACT_ADV_OS CURR_TYPE,
    AAREP_FACT_ADV_PERC CURR_TYPE,
    AAREP_BONUS_OS CURR_TYPE,
    AAREP_BONUS_OS_ADV CURR_TYPE,
    REG_LOGIC_LABEL VARCHAR(255),
    KOL_A INTEGER,
    KOL_B INTEGER,
    KOL_C INTEGER,
    RES_QUAL_KOEF_AVG PERC_TYPE,
    AAREP_BONUS_OFFICER_BEFORE CURR_TYPE,
    COEF_CORRECT PERC_TYPE,
    STAGE_MONTH DECIMAL(4,2),
    AAREP_BONUS_OS_ZAL_CAR CURR_TYPE)
AS
declare variable reg_id integer;
declare variable rep_date_begin date;
declare variable officer_id id_type;
declare variable koef_bonus curr_type;
declare variable koef_penalty curr_type;
declare variable lo_stage_month integer;
declare variable aarep_plan_all decimal(18,4);
declare variable aarep_fact_all decimal(18,4);
declare variable aarep_fact_planned decimal(18,4);
declare variable aarep_fact_other decimal(18,4);
declare variable aarep_month_plan_os decimal(18,4);
declare variable aarep_month_plan_perc decimal(18,4);
declare variable aarep_bonus_total_svz curr_type;
declare variable jk_system_code varchar(100);
declare variable kol integer;
declare variable kol_mo integer;
declare variable kol_zam_mo integer;
declare variable kol_kobd integer;
declare variable kol_admin integer;
declare variable perc_bonus_mo curr_type;
declare variable perc_bonus_zam_mo curr_type;
declare variable perc_bonus_kobd curr_type;
declare variable perc_bonus_admin curr_type;
declare variable aarep_bonus_total_fact_tmp curr_type;
declare variable aarep_fact_all_zalog curr_type;
declare variable koef_bonus_zalog curr_type;
declare variable aarep_bonus_os_zal_car_tmp curr_type;
declare variable aarep_fact_all_tmp curr_type;
declare variable aarep_bonus_plus_tmp curr_type;
declare variable aarep_bonus_os_tmp curr_type;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  koef_bonus=5.0;
  koef_penalty=25.0;
  koef_bonus_zalog = 2.5;
  for
  select id_list
  from region_list(:rep_region_id, 1, 3, 1), region
  where id_list = reg_unicode
  and coalesce(reg_logic_label,'') not containing '.mini_office.'
  into reg_id
  do
  begin

    select sum(aarep_plan_all),
           sum(aarep_fact_all),
           sum(aarep_fact_planned),
           sum(aarep_fact_other), -- не учасвтует
           sum(aarep_month_plan_os), -- не учасвтует
           sum(aarep_month_plan_perc), -- не учасвтует
           sum(aarep_delinq_os), -- не учасвтует
           sum(aarep_delinq_perc), -- не учасвтует
           sum(aarep_fact_os), -- не учасвтует
           sum(aarep_fact_perc), -- не учасвтует
           sum(aarep_fact_fine), -- не учасвтует
           sum(aarep_fact_delinq_os), -- не учасвтует
           sum(aarep_fact_delinq_perc), -- не учасвтует
           sum(aarep_fact_adv_os), -- не учасвтует
           sum(aarep_fact_adv_perc), -- не учасвтует
           sum(aarep_bonus_os), -- не учасвтует
           sum(aarep_bonus_os_adv), -- не учасвтует
           sum(iif(pr_code containing '.zalog_car.', aarep_fact_all, 0))
           , sum(aarep_bonus_os_zal_car)
    from cb_ru_ssvz_201604_one_tmp(:reg_id, :rep_date_end)
--    from cb_ru_lo_201410_one_tmp(:reg_id, 0, :rep_date_begin, :rep_date_end)
    into:  aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os, aarep_month_plan_perc,
           aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine, aarep_fact_delinq_os,
           aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc,
           aarep_bonus_os, aarep_bonus_os_adv, aarep_fact_all_zalog
           , aarep_bonus_os_zal_car
           ;

    if (aarep_plan_all is null) then aarep_plan_all=0; -- Сумма новых просрочек + выплаченные по графику за период
    if (aarep_fact_all is null) then aarep_fact_all=0; -- Сумма новых просрочек + выплаченные по графику за период
    if (aarep_fact_planned is null) then aarep_fact_planned=0; -- выплаченные по графику за период
    if (aarep_fact_other is null) then aarep_fact_other=0;     -- выплаченные по графику за период
    if (aarep_month_plan_os is null) then aarep_month_plan_os=0;
    if (aarep_month_plan_perc is null) then aarep_month_plan_perc=0;
    if (aarep_delinq_os is null) then aarep_delinq_os=0;
    if (aarep_delinq_perc is null) then aarep_delinq_perc=0;
    if (aarep_fact_os is null) then aarep_fact_os=0;
    if (aarep_fact_perc is null) then aarep_fact_perc=0;
    if (aarep_fact_fine is null) then aarep_fact_fine=0;
    if (aarep_fact_delinq_os is null) then aarep_fact_delinq_os=0;
    if (aarep_fact_delinq_perc is null) then aarep_fact_delinq_perc=0;
    if (aarep_fact_adv_os is null) then aarep_fact_adv_os=0;
    if (aarep_fact_adv_perc is null) then aarep_fact_adv_perc=0;
    if (aarep_bonus_os is null) then aarep_bonus_os=0;
    if (aarep_bonus_os_adv is null) then aarep_bonus_os_adv=0;
    if (aarep_fact_all_zalog is null) then aarep_fact_all_zalog = 0;

    if (:rep_date_end < '01.05.2019') then
    begin
      aarep_bonus_plus=round((aarep_fact_all/1000.0)*koef_bonus,0);
    end else
    begin
      aarep_fact_all = aarep_fact_all - aarep_fact_all_zalog;
      aarep_bonus_plus=round((aarep_fact_all/1000.0)*koef_bonus,0) + round((aarep_fact_all_zalog/1000.0)*koef_bonus_zalog,0);
    end

    if (aarep_plan_all>aarep_fact_planned)
    then aarep_bonus_minus=round(((aarep_plan_all-aarep_fact_planned)/1000.0)*koef_penalty,0);
    else aarep_bonus_minus=0;
    aarep_bonus_total=aarep_bonus_plus-aarep_bonus_minus; -- Базовый заработок

    sum_minus = 0;
    kol_a = 0;
    kol_b = 0;
    kol_c = 0;
    aarep_bonus_os_zal_car_tmp = 0;
--    aarep_bonus_os_zal_car = 0;

--    aarep_fact_all = 0;
--    aarep_bonus_plus = 0;
--    aarep_bonus_os = 0;

--    aarep_fact_all_tmp = 0;
    aarep_bonus_plus_tmp = 0;
    aarep_bonus_os_tmp = 0;

    for
    select c_id, aarep_bonus_total--, aarep_bonus_os_zal_car --aarep_fact_all, aarep_bonus_plus, aarep_bonus_os
    from cb_ru_lo_201410_one_lo(:reg_id, 0, :rep_date_begin, :rep_date_end, 0)
    into officer_id, :aarep_bonus_total_svz--, aarep_bonus_os_zal_car_tmp --aarep_fact_all_tmp, aarep_bonus_plus_tmp, aarep_bonus_os_tmp
    do
    begin
      lo_stage_month = 0;
      select lo_stage_month
      from HP_GET_LO_STAGE(:officer_id, :rep_date_end)
      into lo_stage_month;

      if (lo_stage_month >= 12 and aarep_bonus_total_svz < 45000.0) then
      begin
        sum_minus = sum_minus + 2250;
        kol_a = kol_a + 1;
      end --else
      if (lo_stage_month >= 6 and aarep_bonus_total_svz < 25000.0) then
      begin
        sum_minus = sum_minus + 1500;
        kol_b = kol_b + 1;
      end --else
      if (lo_stage_month >= 3 and aarep_bonus_total_svz < 10000.0) then
      begin
        sum_minus = sum_minus + 750;
        kol_c = kol_c + 1;
      end
--      aarep_bonus_os_zal_car = aarep_bonus_os_zal_car + aarep_bonus_os_zal_car_tmp;
--      aarep_fact_all = aarep_fact_all + aarep_fact_all_tmp;
--      aarep_bonus_plus = aarep_bonus_plus + aarep_bonus_plus_tmp;
--      aarep_bonus_os = aarep_bonus_os + aarep_bonus_os_tmp;

    end 
    aarep_bonus_total_fact = aarep_bonus_total - sum_minus;

    if (aarep_plan_all+ aarep_fact_all+ aarep_fact_planned+ aarep_fact_other+ aarep_month_plan_os+ aarep_month_plan_perc+
         aarep_delinq_os+ aarep_delinq_perc+ aarep_fact_os+ aarep_fact_perc+ aarep_fact_fine+ aarep_fact_delinq_os+
         aarep_fact_delinq_perc+ aarep_fact_adv_os+ aarep_fact_adv_perc + aarep_fact_all_zalog <> 0) then
    begin
      select r1.reg_code||' '||r1.reg_name,
             r2.reg_code||' '||r2.reg_name,
             r1.reg_logic_label
      from region r1, region r2
      where r1.reg_unicode = :reg_id
      and r1.parent = r2.reg_unicode
      into reg_name, branch_name, reg_logic_label;

      suspend;
      aarep_bonus_total_fact_tmp = aarep_bonus_total_fact;
      if (aarep_bonus_total_fact_tmp < 0) then aarep_bonus_total_fact_tmp = 0;

      aarep_delinq_os = 0;
      aarep_delinq_perc = 0;
      aarep_fact_os = 0;
      aarep_fact_perc = 0;
      aarep_fact_fine = 0;
      aarep_fact_delinq_os = 0;
      aarep_fact_delinq_perc = 0;
      aarep_fact_adv_os = 0;
      aarep_fact_adv_perc = 0;
      aarep_bonus_os = 0;
      aarep_bonus_os_adv = 0;
      sum_minus = 0;
      aarep_bonus_plus = 0;
      aarep_bonus_minus = 0;
      aarep_bonus_total = 0;
      aarep_bonus_total_fact = 0;
      aarep_bonus_os_zal_car = 0;

      kol_a = 0;
      kol_b = 0;
      kol_c = 0;

--    вычисляем кол-во сотрудников по должностям
      kol_mo = 0;
      kol_zam_mo = 0;
      kol_kobd = 0;
      kol_admin = 0;
      for
      select jk_system_code,
             count(*),
             max(iif(jl_name <> 'Заместитель менеджера офиса' and jk_system_code = 'JobSystem.Region_KM',1,0)) -- если результат 1 то имеется менеджер офиса, если 0 то нету
      from hr_pstatus, joblist, jobkind, clients
      where :rep_date_end between ps_date_begin and ps_date_end
      and ps_status = 1
      and ps_is_working = 1
      and ps_from_c_id > 1000000000003
      and ps_from_jl_id = jl_id
      and jl_kind = jk_id
      and jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Specialist_KOBD', 'JobSystem.Administrator')
      and ps_from_reg_id = :reg_id
      and exists(select * from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                 where is_decret = 0)
      and jl_name <> 'И.о. заместителя менеджера офиса'
      and jl_name <> 'Выездной кассир-оператор'
      and ps_from_c_id = c_id
      group by jk_system_code
      into jk_system_code, kol, kol_mo
      do
      begin
        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
--          kol_mo = 1;
          kol_zam_mo = kol - kol_mo;
        end
        if (jk_system_code = 'JobSystem.Specialist_KOBD') then
        begin
          kol_kobd = kol;
        end
        if (jk_system_code = 'JobSystem.Administrator') then
        begin
          kol_admin = kol;
        end
      end

      perc_bonus_mo = 0;
      perc_bonus_zam_mo = 0;
      perc_bonus_kobd = 0;
      perc_bonus_admin = 0;
--    нужно раскидать
      if (reg_logic_label = '.mega_office.') then
      begin
        perc_bonus_mo = 30.0;
        if (kol_zam_mo > 0) then
          perc_bonus_zam_mo = cast(46.0 as double precision) / kol_zam_mo;
        if (perc_bonus_zam_mo > 23.0) then perc_bonus_zam_mo = 23.0;
        if (kol_kobd > 0) then
          perc_bonus_kobd = cast(10.0 as double precision) / kol_kobd;
        if (kol_admin > 0) then
          perc_bonus_admin = cast(14.0 as double precision) / kol_admin;
      end else
      if (kol_zam_mo = 0) then
      begin
        perc_bonus_mo = 85.0;
        if (kol_kobd > 0) then
          perc_bonus_kobd = cast(15.0 as double precision) / kol_kobd;
      end else
      if (kol_zam_mo >= 1) then
      begin
        perc_bonus_mo = 45.0;
        if (kol_zam_mo > 0) then
          perc_bonus_zam_mo = cast(40.0 as double precision) / kol_zam_mo;
        if (kol_kobd > 0) then
          perc_bonus_kobd = cast(15.0 as double precision) / kol_kobd;
      end

      for
      select jl_name, c_name_fio, c_id, jk_system_code
      from hr_pstatus, joblist, jobkind, clients
      where :rep_date_end between ps_date_begin and ps_date_end
      and ps_status = 1
      and ps_is_working = 1
      and ps_from_c_id > 1000000000003
      and ps_from_jl_id = jl_id
      and jl_kind = jk_id
      and jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Specialist_KOBD', 'JobSystem.Administrator')
      and ps_from_reg_id = :reg_id
      and exists(select * from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                 where is_decret = 0)
      and jl_name <> 'И.о. заместителя менеджера офиса'
      and jl_name <> 'Выездной кассир-оператор'
      and ps_from_c_id = c_id
      order by jk_system_id, jl_name desc
      into jl_name, officer_name, officer_id, jk_system_code
      do
      begin


        res_qual_koef_avg = null;
        aarep_bonus_officer_before = null;
        coef_correct = null;
        aarep_bonus_officer = 0;
        stage_month = 0;
--      Расчет совокупного стажа сотрудинка, если менее 3х месяцев то бонус не получает
        select lo_stage_month_full from hp_get_lo_stage_full(:officer_id, :rep_date_end, :jk_system_code)
        into stage_month;

--        if (stage_month >= 3.5) then
--        begin
            if (jk_system_code = 'JobSystem.Region_KM') then
            begin
              if (stage_month >= 3.5) then
              begin
                if (jl_name <> 'Заместитель менеджера офиса') then aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_mo;
                else aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_zam_mo;
              end
              res_qual_koef_avg = 0;
              select (res_qual_koef1_stage_12 + res_qual_koef2_stage_12 + res_qual_koef3_stage_12) /
                     ((iif(res_qual_koef1_stage_12 > 0,1,0) + iif(res_qual_koef2_stage_12 > 0,1,0) + iif(res_qual_koef3_stage_12 > 0,1,0)) * 100)
              from aarep_disb_manager_quality(:reg_id,:rep_date_begin,:rep_date_end,:officer_id,1)
              where res_qual_koef1_stage_12 + res_qual_koef2_stage_12 + res_qual_koef3_stage_12 > 0
              into res_qual_koef_avg;
              if (res_qual_koef_avg is null) then res_qual_koef_avg = 0;
    
              if (stage_month >= 3.5) then
              begin
                coef_correct = case
                               when res_qual_koef_avg >= 0.97 then 1.1
                               when res_qual_koef_avg >= 0.95 then 1.0
                               when res_qual_koef_avg >= 0.90 then res_qual_koef_avg
                               when res_qual_koef_avg >= 0.875 then 0.8
                               when res_qual_koef_avg >= 0.85 then 0.7
                               else 0.6
                             end;
                aarep_bonus_officer_before = aarep_bonus_officer;
                aarep_bonus_officer = aarep_bonus_officer * coef_correct;
              end

            end else
            if (stage_month >= 3.5) then
            begin
              if (jk_system_code = 'JobSystem.Specialist_KOBD') then
              begin
                aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_kobd;
              end else
              if (jk_system_code = 'JobSystem.Administrator') then
              begin
                aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_admin;
              end
            end
--        end

        aarep_bonus_officer = (select res from fhp_round_bank(:aarep_bonus_officer,0));
        suspend;
      end
      aarep_bonus_officer = null;
      jl_name = null;
      officer_name = null;
      officer_id = null;

    end
  end
end
