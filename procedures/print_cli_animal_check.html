CREATE PROCEDURE PRINT_CLI_ANIMAL_CHECK(
    C_ID D_ID,
    ADD_PARAMS D_ID)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_INN D_STR_100,
    EIL_DATE D_DATE,
    ANIMAL_TYPE_TXT D_STR_255,
    ANIMAL_COUNT D_INTEGER,
    ORG_NAME D_STR_255)
AS
declare variable bls_id d_id;
declare variable json d_blob_text;
declare variable obj_count d_integer;
declare variable animal_type d_integer;
declare variable temp_animal_type d_integer;
declare variable counter d_integer;
declare variable k_disb_date d_date;
begin
  org_name = 'Ветеринарная служба при МСХ КР';

  select k.k_disb_date
  from kontract k
  where k.k_id = :add_params
  into k_disb_date;

  k_disb_date = coalesce(k_disb_date, current_date);

  select eil_date, eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id and
        eil_from_eic_id = eic_id and
        eic_code = 'animals_check' and
        eil_is_json = 1 and
        eil_status = 2 and
        eil_date <= :k_disb_date --RFCKG-3426
  order by eil_date desc, eil_id
  rows 1
  into eil_date, bls_id;

  if (bls_id is null) then -- Если не удалось найти бланк берем последний
  begin
     select eil_date, eil_from_bls_id
     from extinfolist, extinfocfg
     where eil_from_c_id = :c_id and
           eil_from_eic_id = eic_id and
           eic_code = 'animals_check' and
           eil_is_json = 1 and
           eil_status = 2
     order by eil_date desc, eil_id
     rows 1
     into eil_date, bls_id;

     eil_date = k_disb_date;
  end

  select c_name_fio, c_inn
  from clients
  where c_id = :c_id
  into c_name_fio, c_inn;

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json;

  :json = '{"animal_list" :' || :json || '}';
  obj_count = f_json_blob_get_count(:json, 'animal_list');
  animal_type = 1;

  while (animal_type < 7) do
  begin
    counter = 0;
    animal_count = 0;
    while (counter < obj_count) do
    begin
      temp_animal_type = f_json_blob_get_value(:json, 'animal_list[' || counter || '].Type', 0);

      if (temp_animal_type = animal_type) then
        animal_count = animal_count + 1;
      counter = counter + 1;
    end
    animal_type_txt = case animal_type
                        when 1 then 'Крупно-рогатый скот и яки'
                        when 2 then 'Лошади'
                        when 3 then 'Мелко-рогатый скот (овцы и козы)'
                        when 4 then 'Свиньи'
                        when 5 then 'Домашние птицы'
                        when 6 then 'Пчелы'
                      end;

    animal_type = animal_type + 1;
    suspend;
  end

end
