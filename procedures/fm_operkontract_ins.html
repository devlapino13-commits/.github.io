CREATE PROCEDURE FM_OPERKONTRACT_INS(
    K_ID D_ID,
    AMOUNT CURR_TYPE)
RETURNS (
    OL_ID D_ID,
    ERROR_TEXT D_STR_255)
AS
declare variable ol_doc_kind d_integer;
declare variable k_status d_integer;
declare variable ol_doc_no d_integer;
declare variable ol_doc_no_str d_str_100;
declare variable ol_memo d_str_255;
declare variable res_acc_kt d_str_100;
declare variable res_acc_dt d_str_100;
declare variable un_id d_id;
declare variable c_id d_id;
declare variable res_currency d_str_100;
declare variable reg_id d_id;
declare variable k_disb_date d_date;
declare variable disb_ol_id id_type;
declare variable k_prepayment_summa decimal(15,2);
declare variable payed_val decimal(15,2);
declare variable operation_type d_integer;
declare variable k_kontract_no d_str_20;
declare variable pr_code d_str_255;
declare variable ob_kt d_str_100;
declare variable pr_id d_id;
begin
  operation_type = f_get_rd_id('CashTransactionIncom.PartialPrice');
  ol_doc_kind = 9; -- мемориальный ордер
  un_id = 1000000000000;
  select k.k_id, k.k_kontract_no, k.k_status, k.k_fact_disb_ol_id, coalesce(k.k_prepayment_summa, 0), k.k_last_reg_id,
         k.k_disb_date, pr_code, pr_id
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where k.k_id = :k_id
  into :k_id, :k_kontract_no, :k_status, :disb_ol_id, :k_prepayment_summa, reg_id, k_disb_date, pr_code, pr_id;

  if (k_status <> 1) then
  begin
    error_text = 'Ошибка! Договор должен быть утвержден';
    suspend;
    exit;
  end

  if (disb_ol_id is not null) then
  begin
    if (exists(select *
               from operlist ol
               where ol.ol_id = :disb_ol_id and
                     ol.is_delete = 0)) then
    begin
      error_text = 'Договор уже выдан. Операция не может быть осуществлена';
      suspend;
      exit;
    end
  end

  if ((k_prepayment_summa = 0) and
      (amount <> 0)) then
  begin
    error_text = 'Ошибка! Для данного кредита не указана сумма частичной оплаты стоймости товара';
    suspend;
    exit;
  end
  if ((amount > k_prepayment_summa) and
      (k_prepayment_summa <> 0)) then
  begin
    error_text = 'Ошибка! Для данного кредита cумма частичной оплаты стоймости товара составляет: ' || k_prepayment_summa || ' сом, а вносится: ' || cast(amount as decimal(15,2)) || ' сом';
    suspend;
    exit;
  end

  payed_val = 0;
  select sum(opk.ok_amount)
  from operkontract opk
  inner join operlist ol on ol.ol_id = opk.ok_from_ol_id and
        ol.is_delete = 0
  where opk.ok_from_k_id = :k_id and
        opk.ok_operation_type = :operation_type
  into :payed_val;
  payed_val = coalesce(payed_val, 0);

  if ((payed_val = k_prepayment_summa) and
      (k_prepayment_summa > 0)) then
  begin
    error_text = 'Ошибка! Сумма оплаты, ранее уже была внесена в полном объеме';
    suspend;
    exit;
  end
  else
  if (amount > (k_prepayment_summa - payed_val)) then
  begin
    error_text = 'Ошибка! Сумма оплаты, частично уже была внесена, оставшаяся часть: ' ||(k_prepayment_summa - payed_val) || '. Вы пытаетесь внести: ' || cast(amount as decimal(15,2));
    suspend;
    exit;
  end

  res_currency = f_get_const('cm_default_currency');

  select kc.kc_from_c_id
  from knclient kc
  where kc.kc_from_k_id = :k_id and
        kc.kc_amount_approved > 0
  into c_id;

  select kv
  from get_kv(:reg_id, 5)
  into :ol_id;

  select num
  from last_doc_number(:k_disb_date, :ol_doc_kind, :reg_id)
  into :ol_doc_no;

  ol_doc_no_str = :ol_doc_no;
  ol_memo = 'Оплата частичной стоимости товара по договору финансирования ' || k_kontract_no;

  insert into operlist (ol_id, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str, ol_oper_type, ol_oper_subtype,
                        ol_doc_kind, ol_memo, ol_user_id, ol_document_client, ol_doc_client_name, ol_is_client,
                        ol_parent_id, ol_manager_c_id, ol_from_rd_id)
  values (:ol_id, :reg_id, :k_disb_date, :ol_doc_no, :ol_doc_no_str, 2, 1, :ol_doc_kind, :ol_memo, :un_id, :c_id, null,
          iif(:c_id is not null, 1, 0), null, null, 0);

  insert into ipf_operkontract (io_id, io_from_ol_id, io_from_k_id, io_is_active)
  values (null, :ol_id, :k_id, 1);

  select res_account
  from hp_make_kredit_acclist(:pr_id, :reg_id, 46)
  into :res_acc_kt;
  res_acc_dt = (select al_id
                from acc_list a
                where a.al_from_reg_id = :reg_id and
                      a.al_from_bg_id = 10024 and
                      a.al_number = 108 and
                      a.al_currency = :res_currency);

  insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val, ob_summa,
                        ob_detail, ob_client_kt, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
  values (:ol_id, :res_currency, 1, :res_acc_dt, :res_acc_kt, :k_prepayment_summa, :k_prepayment_summa, '', :c_id,
          :k_disb_date, 1, :reg_id);

  insert into operkontract (ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
  values (:k_id, :ol_id, :amount, :operation_type);
  suspend;
end
