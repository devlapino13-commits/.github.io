CREATE PROCEDURE HP_BANK_OBOR_RESTORE2
RETURNS (
    OL_ID D_ID,
    OL_DATE D_DATE,
    OB_SUMMA D_CURRENCY,
    AL_ID D_STR_CODE,
    OL_MEMO D_STR_LONG,
    K_KONTRACT_NO D_STR_CODE,
    K_ID D_ID,
    NEW_OL_ID D_ID,
    OB_NUMBER D_INTEGER,
    MAIN_OL_ID D_ID,
    RES D_STR_CODE)
AS
declare variable ol_memo_orig d_str_long;
begin
  for select tol_object_id, tol_add_info_1, ol_date, ol_memo
  from temp_objlist, operlist O
  where tol_kind = 'fix_51_01_ol_id' and
        tol_object_kind = 9 and
        coalesce(tol_add_info_1, '') = '' and
        tol_status = 0 and
        tol_object_id = O.ol_id and
        O.is_delete = 1 and
        O.ol_id  = 1010100018917606
  into ol_id, k_kontract_no, ol_date, ol_memo
  do
  begin
    ob_summa = 0;
    select ob_account_dt, sum(ob_summa)
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_account_dt containing '51-01' and
          ob_ol_is_active = 0
    group by 1
    into al_id, ob_summa;
    if (ob_summa is null) then
      ob_summa = 0;

    if (ob_summa = 0) then
      Continue;

    select ob_from_ol_id
    from operbook, operlist o1, operlist o2
    where ob_summa = :ob_summa and
          ob_ol_date = :ol_date and
          ob_account_dt containing '57-04' and
          o2.ol_memo = :ol_memo and
          ob_ol_is_active = 1 and
          ob_from_ol_id = o1.ol_id and
          o1.ol_parent_id = o2.ol_id
    into new_ol_id;

    select ol_parent_id
    from operlist
    where ol_id = :new_ol_id
    into main_ol_id;


    update operbook
    set ob_from_ol_id = :main_ol_id,
        ob_ol_is_active = 1
    where ob_from_ol_id = :ol_id and
          ob_account_dt = :al_id and
          ob_summa = :ob_summa;

    update temp_objlist
    set tol_status = 1
    where tol_kind = 'fix_51_01_ol_id' and
          tol_object_kind = 9 and
          tol_object_id = :ol_id;

    suspend;
  end
end
