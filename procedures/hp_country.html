CREATE PROCEDURE HP_COUNTRY
RETURNS (
    CNT_ID VARCHAR(3),
    CNT_COUNTRY_NAME VARCHAR(100),
    CNT_COUNTRY_CODE VARCHAR(10),
    CNT_CELL_MASK VARCHAR(50),
    CNT_CELL_START VARCHAR(10))
AS
begin
  for select rd_str_id, rd_name, replace(rd_code, 'Citizen.', ''), rd_memo, rd_tag
      from refdata
      where rd_parent_id is not null and
            rd_main_code = 'Citizen' and
            coalesce(rd_str_id, '') <> ''
      into :cnt_id, :cnt_country_name, :cnt_country_code, :cnt_cell_mask, :cnt_cell_start
  do
  begin
    if (cnt_cell_start = '0') then
      cnt_cell_start = '';
    suspend;
  end
end
