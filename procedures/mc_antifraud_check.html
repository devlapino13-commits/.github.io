CREATE PROCEDURE MC_ANTIFRAUD_CHECK(
    INN D_STR_20)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255,
    AC_ID D_ID)
AS
declare variable IS_WORKING D_ID;
begin
    -- RFCKG-302

    -- Процедура нужна для того, чтобы КМП знало, какому пути придерживаться
    -- proceed - механизм антифрода не работает, продолжаем процесс без него (старая реализация)
    -- antifrod - осуществляем проверку через антифрод
    if (coalesce(:inn, '') = '' or char_length(:inn) <> 12) then
    begin
        res_code = 'error';
        res_text = 'Не удалось обнаружить ИИН. Пожалуйста проверьте, что вы прикрепляли лицевую сторону документа';     
        suspend;
        exit;
    end

    is_working = (select hp.is_working from antifraud_api_working_status hp);
    if (:is_working = 0) then
    begin
        -- если механизм антифрода не работает, тогда мы должны идти по обычному сценарию
        res_code = 'proceed';
        res_text = '';
        suspend;
        exit;
    end

    -- все проверки закончены, делаем запрос
    ac_id = (select ac_id from hp_extinfo_insert_antifraud(:inn));
    res_code = 'antifraud';
    res_text = '';

    suspend;
end
