CREATE PROCEDURE HP_GET_KONTRACT_STATUS(
    REP_K_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    K_LAST_STATUS INTEGER,
    K_CURR_STATUS INTEGER,
    K_LAST_STATUS_STR VARCHAR(100),
    K_CURR_STATUS_STR VARCHAR(100))
AS
declare variable C_ID ID_TYPE;
declare variable C_LAST_STATUS integer;
declare variable C_CURR_STATUS integer;
declare variable C_LAST_STATUS_STR varchar(100);
declare variable C_CURR_STATUS_STR varchar(100);
begin
  k_last_status = 999;
  k_curr_status = 999;
  k_last_status_str = 'Балл: *****';
  k_curr_status_str = 'Балл: *****';
  c_id = null;
  for select kc_from_c_id
      from knclient
      where kc_from_k_id = :rep_k_id and
            kc_amount_approved > 0
      into c_id
  do
  begin
    --    если в группе несколько участвников то берем статус самого минимального пункт 3.2.
    c_last_status = null;
    c_curr_status = null;

    select c_curr_status, c_curr_status_str
    from hp_get_client_status_str(:c_id, :rep_date_end)
    into c_curr_status, c_curr_status_str;
    if (c_curr_status is null) then
      c_curr_status = 0;
    if (c_curr_status < k_curr_status) then
    begin
      k_curr_status = c_curr_status;
      k_curr_status_str = c_curr_status_str;
    end

    select c_last_status, c_last_status_str
    from hp_get_client_status_str(:c_id, :rep_date_end)
    into c_last_status, c_last_status_str;
    if (c_last_status is null) then
      c_last_status = 0;
    if (c_last_status < k_last_status) then
    begin
      k_last_status = c_last_status;
      k_last_status_str = c_last_status_str;
    end
  end
  if (c_id is null) then
  begin
    k_last_status = 0;
    k_curr_status = 0;
    k_last_status_str = 'Без статуса';
    k_curr_status_str = 'Без статуса';
  end
  suspend;
end
