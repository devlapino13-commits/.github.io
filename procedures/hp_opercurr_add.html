CREATE PROCEDURE HP_OPERCURR_ADD(
    CU_ID D_STR_LANG,
    CR_DATE D_DATE,
    CR_RATE D_CURRENCY)
RETURNS (
    IS_ERROR D_INTEGER,
    RES_TEXT D_STR_255)
AS
declare variable CU_CURR_UNIT D_INTEGER;
begin
  -- добавляет курс валюты в opercurr

  is_error = 1;
  res_text = '';

  if (exists(select *
             from opercurr
             where cr_from_cu_id = :cu_id and
                   cr_date = :cr_date)) then
    res_text = 'Ошибка: курс на указанный день уже добавлен';

  --if (cr_date < current_date) then
  --  res_text = 'Ошибка: нельзя добавить курс на прошедшую дату';

  if (coalesce(cr_rate, 0) <= 0) then
    res_text = 'Ошибка: не указан курс валюты';

  select cu_curr_unit
  from currlist
  where cu_id = :cu_id
  into cu_curr_unit;

  if (coalesce(cu_curr_unit, 0) < 1) then
    res_text = 'Ошибка: не указано соотношение осн. валюты к текущей';

  if (res_text = '') then
  begin
    insert into opercurr (cr_from_cu_id, cr_date, cr_rate, cr_curr_unit)
    values (:cu_id, :cr_date, :cr_rate, :cu_curr_unit);

    is_error = 0;
    res_text = 'Курс валюты успешно добавлен';
  end

  suspend;
end
