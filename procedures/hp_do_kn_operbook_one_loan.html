CREATE PROCEDURE HP_DO_KN_OPERBOOK_ONE_LOAN(
    K_ID ID_TYPE,
    K_PERIOD INTEGER = null,
    IS_FULL_PERIOD INTEGER = 0)
AS
declare variable ol_id id_type;
declare variable ol_date date;
declare variable ol_date_prv date;
declare variable kc_id id_type;
declare variable saldo_date date;
declare variable need_saldo integer;
declare variable off_saldo curr_type;
declare variable ks_date_begin date;
declare variable ks_date_end date;
declare variable first_oper_date date;
declare variable last_oper_date date;
declare variable close_month_end date;
declare variable oek_rd_id integer;
declare variable kob_id d_id;
declare variable ol_oper_subtype d_integer;
begin
-- данная процедура заменена HP_DO_KN_OPERBOOK_ONE_ONLINE.
-- после успешного перехода на онлайн построения проводок, данная процедура должна быть удалена
-- Sanjari, 12/04/2019
  exception e_default 'Процедура кандидат на удаление.';

  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

--  execute procedure hp_fill_kn_oksummary(:k_id, 1);
  if (k_period is null) then
    exception e_default 'Period is empty> input like YYYYMM';

  saldo_date = '31.12.2017';

  first_oper_date = cast('01.' || right(k_period, 2) || '.' || left(k_period, 4) as date);
  last_oper_date = dateadd(month, 1, first_oper_date) - 1;

  if (is_full_period = 1) then
    first_oper_date = saldo_date + 1;

  if (first_oper_date <> (saldo_date + 1)) then
    saldo_date = null;

  if (k_period >= 201903) then -- начиная с марта 2019 года создаем единый OL_ID
  begin
    for select distinct kob_from_ol_id, kob_from_kc_id, ol_oper_subtype
        from kn_operbook, knclient, operlist
        where kc_from_k_id = :k_id and
              kc_id = kob_from_kc_id and
              kob_from_ol_id = ol_id and
              kob_ol_date >= coalesce(:saldo_date, :first_oper_date) and
              ol_oper_type = 2 and
              ol_oper_subtype in (31, 32, 33, 34)
        into ol_id, :kc_id, ol_oper_subtype
    do
    begin
      if (ol_oper_subtype = 31) then
      begin
-- проводки по смене резиденства
        delete from operlist
        where ol_id = :ol_id;
      end
      else
      begin
-- начисления: процент, резерв и корректировка ВБС
        delete from kn_operbook
        where kob_from_ol_id = :ol_id and
              kob_from_kc_id = :kc_id;
      end
    end

    for select kc_id
        from knclient
        where kc_from_k_id = :k_id
        into :kc_id
    do
    begin
      for select kob_id
          from kn_operbook, operlist
          where kob_from_ol_id = ol_id and
                kob_from_kc_id = :kc_id and
                not(ol_oper_type = 2 and
                ol_oper_subtype in (32, 33, 34)) and
                kob_ol_date >= coalesce(:saldo_date, :first_oper_date)
          order by kob_from_ol_id desc, kob_id desc
          into kob_id
      do
      begin
        delete from kn_operbook
        where kob_id = :kob_id;
      end
    end
  end
  else
  begin
    for select distinct kob_from_ol_id
        from kn_operbook, knclient
        where kc_from_k_id = :k_id and
              kc_id = kob_from_kc_id and
              kob_ol_date >= coalesce(:saldo_date, :first_oper_date)
        into :ol_id
    do
    begin
      if (exists(select *
                 from operlist
                 where ol_id = :ol_id and
                       ol_oper_type = 2 and
                       ol_oper_subtype in (30, 31, 32))) then
      begin
        delete from operlist
        where ol_id = :ol_id;
      end
    end

    for select kc_id
        from knclient
        where kc_from_k_id = :k_id
        into :kc_id
    do
    begin
      delete from kn_operbook
      where kob_from_kc_id = :kc_id and
            kob_ol_date >= coalesce(:saldo_date, :first_oper_date);
    end
  end
  ol_date_prv = null;
  need_saldo = 0;
  select ks_date_begin, ks_date_end
  from kontstate
  where ks_from_k_id = :k_id and
        ks_ol_is_delete = 0 and
        ks_state = 1 rows 1
  into :ks_date_begin, ks_date_end;

  if (saldo_date is not null and
      ks_date_begin <= cast('31.12.2017' as date)) then
  begin
    if (ks_date_begin <= :saldo_date and
        ks_date_end >= :saldo_date) then
      need_saldo = 1;

    if (need_saldo = 0) then
    begin
      off_saldo = 0;
      select sum(oko_sh_os + oko_os + oko_sh_perc + oko_perc)
      from operkroff
      where oko_from_k_id = :k_id and
            oko_ol_is_active = 1 and
            oko_ol_date <= :saldo_date
      into :off_saldo;
      if (off_saldo is null) then
        off_saldo = 0;
      if (off_saldo <> 0) then
        need_saldo = 1;
    end

    if (need_saldo = 0) then
    begin
      off_saldo = 0;
      select sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff)
      from operextra
      where ox_from_rd_id = :oek_rd_id and
            ox_object_id = :k_id and
            ox_ol_is_active = 1 and
            ox_ol_date <= :saldo_date
      into :off_saldo;
      if (off_saldo is null) then
        off_saldo = 0;
      if (off_saldo <> 0) then
        need_saldo = 1;
    end

    if (need_saldo = 0) then
    begin
      off_saldo = 0;
      select sum(ok_os)
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_ol_date <= :saldo_date and
            ok_type = 9
      into :off_saldo;
      if (off_saldo is null) then
        off_saldo = 0;
      if (off_saldo <> 0) then
        need_saldo = 1;
    end

    if (need_saldo = 1) then
    begin
      ol_date_prv = saldo_date;
      execute procedure hp_do_ref_account_loan_ob_saldo(:k_id);
    end
  end

  for select distinct ok_ol_date
      from operkred
      where ok_from_k_id = :k_id and
            (ok_os <> 0 or
            ok_perc <> 0 or
            ok_fine <> 0 or
            ok_kommis <> 0 or
            ok_kommis_adv <> 0) and
            ok_type in (4, 5, 6, 8, 9) and
            ok_ol_date between :first_oper_date and :last_oper_date
      union
      select distinct oko_ol_date
      from operkroff
      where oko_from_k_id = :k_id and
            (oko_os <> 0 or
            oko_perc <> 0 or
            oko_sh_os <> 0 or
            oko_sh_perc <> 0) and
            oko_type in (4, 5) and
            oko_ol_date between :first_oper_date and :last_oper_date
      union
      select distinct ox_ol_date
      from operextra
      where ox_from_rd_id = :oek_rd_id and
            ox_object_id = :k_id and
            ox_object_kind = 9 and
            (ox_amount_plus <> 0 or
            ox_amount_minus <> 0 or
            ox_amount_writeoff <> 0) and
            ox_ol_date between :first_oper_date and :last_oper_date
      into :ol_date
  do
  begin
    if (ol_date_prv is null) then
    begin
      if (saldo_date is null) then
        ol_date_prv = first_oper_date - 1;
      else
        ol_date_prv = ol_date;
    end
    execute procedure hp_do_kn_operbook_one_loan_date(:k_id, :ol_date, :ol_date_prv, null);
    ol_date_prv = ol_date;
  end

  if (saldo_date is not null and
      ks_date_end < cast('31.12.2999' as date)) then
  begin
    close_month_end = ks_date_end + 1;

    if (close_month_end < ol_date_prv) then
      close_month_end = ol_date_prv;

    close_month_end = close_month_end - extract(day from close_month_end) + 1;
    close_month_end = dateadd(month, 1, close_month_end) - 1;

    if (close_month_end > last_oper_date) then
      close_month_end = last_oper_date;

    if (close_month_end < first_oper_date) then
      exit;
  end
  else
    close_month_end = last_oper_date; -- если займ еще активный, формируем проводки до 30.09.2018.

  if (ol_date_prv is null) then
    ol_date_prv = first_oper_date - 1;

  if ((close_month_end > saldo_date or saldo_date is null) and
      (close_month_end > ol_date_prv or ol_date_prv is null)) then
    execute procedure hp_do_kn_operbook_one_loan_date(:k_id, :close_month_end, :ol_date_prv, null);

-- проводки по переносу сальдо из-за смены резидентства
  if (last_oper_date >= '31.12.2018' and
      ks_date_end >= last_oper_date) then
    execute procedure hp_do_kn_resident_update(:k_id, :last_oper_date, 13, 39, 1000000000001);
end
