CREATE PROCEDURE HP_CALC_KDD(
    C_ID D_ID,
    UN_ID D_ID,
    MA_ID D_ID,
    MAC_ID D_ID,
    SALARY_INCOME D_CURRENCY,
    PENSION_INCOME D_CURRENCY,
    ASP_INCOME D_CURRENCY)
RETURNS (
    KDD D_CURRENCY)
AS
declare variable MA_SUMM D_CURRENCY;
declare variable JSON_DOCUMENT D_BLOB_TEXT;
declare variable ARR_LENGTH D_INTEGER;
declare variable I D_INTEGER;
declare variable CURRENT_PAYMENT D_CURRENCY;
declare variable MOUNT_PAYMENTS D_CURRENCY;
declare variable OVERDUE_PAYMENT D_CURRENCY;
declare variable ALL_INCOME D_CURRENCY;
declare variable PP_SUMM D_CURRENCY;
declare variable BLS_ID D_ID;
declare variable TMPSTR D_STR_100;
declare variable SZNZ D_CURRENCY;
declare variable SNZ D_CURRENCY;
declare variable RESIDUAL_AMOUNT D_CURRENCY;
declare variable CREDIT_LIMIT D_CURRENCY;
declare variable TEN_PERC_OF_CREDIT_LIMIT D_CURRENCY;
begin
  -- RFCKZ-56 расчет КДД
  kdd = 0;
  overdue_payment = 0;
  sznz = 0; -- СЗНЗ - сумма задолженности по всем непогашенным займам и микрокредитам заемщика

  -- СГДЗ Сумма всех доходов (ЗП согласно договору, доход от пенсии, доход от АСП) * 12
  salary_income = coalesce(salary_income, 0);
  pension_income = coalesce(pension_income, 0);
  asp_income = coalesce(asp_income, 0);

  all_income = (salary_income + pension_income + asp_income) * 12;

  -- Получаем json_document от PKB в КЗ
  select blx_blob,
         eil.eil_from_bls_id
  from extinfolist eil, extinfocfg eic, hp_get_blobex_blob(eil.eil_from_bls_id)
  where eil.eil_from_c_id = :c_id and
        eil.eil_status = 2 and
        eil.eil_from_eic_id = eic.eic_id and
        eic.eic_code = 'pkb_cib_report'
  order by eil.eil_status_datetime desc
  rows 1
  into :json_document,
       :bls_id;

  -- СНЗ  Сумма по новой заявке
  select ma.ma_summa_minus
  from miniapp ma
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end = '31.12.9999'
  order by ma.ma_number desc
  rows 1
  into snz;
    
  snz = coalesce(snz, 0);

  if (json_document <> '') then
  begin
    --------------------------- СЗНЗ --------------------------------------- (сумма задолженности по всем непогашенным займам и микрокредитам заемщика.)

    -- ежемесячный платеж по каждому непогашенному микрокредиту/займу клиента
    arr_length = f_json_blob_get_count(:json_document, 'Report.Root.ExistingContracts.Contract');


    if (arr_length <> -1) then
    begin
      i = 0;

      while (arr_length > i) do
      begin
        tmpstr = f_json_blob_get_value(:json_document, f_format('Report.Root.ExistingContracts.Contract[%1].TypeOfFounding.id', i), '');
        residual_amount = 0;
        credit_limit = 0;
        ten_perc_of_credit_limit = 0;
        mount_payments = 0;

        if (tmpstr = '9') then
        begin
          -- использованная сумма кредитного лимита
          residual_amount = iif(replace(replace(replace(f_json_blob_get_value(:json_document,f_format('Report.Root.ExistingContracts.Contract[%1].ResidualAmount.value', i), 0), 'KZT',''), ' ',''), ',', '.') = '', 0,
                                replace(replace(replace(f_json_blob_get_value(:json_document,f_format('Report.Root.ExistingContracts.Contract[%1].ResidualAmount.value', i), 0), 'KZT',''), ' ',''), ',', '.'));

          -- кредитный лимит
          credit_limit = iif(replace(replace(replace(f_json_blob_get_value(:json_document,f_format('Report.Root.ExistingContracts.Contract[%1].CreditLimit.value', i), 0), 'KZT',''), ' ',''), ',', '.') = '', 0,
                             replace(replace(replace(f_json_blob_get_value(:json_document,f_format('Report.Root.ExistingContracts.Contract[%1].CreditLimit.value', i), 0), 'KZT',''), ' ',''), ',', '.'));

          -- 10% от оставшееся суммы кредитного лимита
          ten_perc_of_credit_limit = (credit_limit - residual_amount) * 0.1;
        end

        mount_payments = iif(replace(replace(replace(f_json_blob_get_value(:json_document,f_format('Report.Root.ExistingContracts.Contract[%1].MonthlyInstalmentAmount.value', i), 0), 'KZT',''), ' ',''), ',', '.') = '', 0,
                             replace(replace(replace(f_json_blob_get_value(:json_document,f_format('Report.Root.ExistingContracts.Contract[%1].MonthlyInstalmentAmount.value', i), 0), 'KZT',''), ' ',''), ',', '.'));

        sznz = sznz + (residual_amount + ten_perc_of_credit_limit + mount_payments);
        i = i + 1;
      end

    end
    else
    begin
      tmpstr = f_json_blob_get_value(:json_document, 'Report.Root.ExistingContracts.Contract.TypeOfFounding.id', '');
      residual_amount = 0;
      credit_limit = 0;
      ten_perc_of_credit_limit = 0;
      mount_payments = 0;

      if (tmpstr = '9') then
      begin
        -- использованная сумма кредитного лимита
        residual_amount = iif(replace(replace(replace(f_json_blob_get_value(:json_document, 'Report.Root.ExistingContracts.Contract.ResidualAmount.value', 0), 'KZT',''), ' ',''), ',', '.') = '', 0,
                              replace(replace(replace(f_json_blob_get_value(:json_document, 'Report.Root.ExistingContracts.Contract.ResidualAmount.value', 0), 'KZT',''), ' ',''), ',', '.'));

        -- кредитный лимит
        credit_limit = iif(replace(replace(replace(f_json_blob_get_value(:json_document, 'Report.Root.ExistingContracts.Contract.CreditLimit.value', 0), 'KZT',''), ' ',''), ',', '.') = '', 0,
                           replace(replace(replace(f_json_blob_get_value(:json_document, 'Report.Root.ExistingContracts.Contract.CreditLimit.value', 0), 'KZT',''), ' ',''), ',', '.'));

        -- 10% от оставшееся суммы кредитного лимита
        ten_perc_of_credit_limit = (credit_limit - residual_amount) * 0.1;
      end

      mount_payments = iif(replace(replace(replace(f_json_blob_get_value(:json_document, 'Report.Root.ExistingContracts.Contract.MonthlyInstalmentAmount.value', ''), 'KZT',0), ' ',''), ',', '.') = '', 0,
                           replace(replace(replace(f_json_blob_get_value(:json_document, 'Report.Root.ExistingContracts.Contract.MonthlyInstalmentAmount.value', ''), 'KZT',0), ' ',''), ',', '.'));

      sznz = residual_amount + ten_perc_of_credit_limit + mount_payments;
    end


    -- сумма просроченных платежей по каждому микрокредиту/займу клиента
    arr_length = f_json_blob_get_count(:json_document, 'Report.Root.SummaryInformation.ExistingContracts.SubjectRole');

    if (arr_length > 0) then
    begin
      i = 0;
      current_payment = 0;

      while (i < arr_length) do
      begin
        tmpstr = f_json_blob_get_value(cast(f_json_blob_get_value(:json_document,'Report.Root.SummaryInformation.ExistingContracts.SubjectRole[' || i || ']', '') as d_blob_text),'id', '');

        if (tmpstr = '1') then
        begin
          current_payment = 0;

          tmpstr = replace(replace(replace(f_json_blob_get_value(:json_document,'Report.Root.SummaryInformation.ExistingContracts.SubjectRole[' || i || '].TotalDebtOverdue.value', 0), 'KZT',''), ' ',''), ',', '.');
          current_payment = case when tmpstr <> '' then cast(tmpstr as decimal(18,2)) else 0 end ;
          overdue_payment = overdue_payment + current_payment;
        end

          i = i + 1;
       end
     end
     else
     begin
       if (f_json_blob_get_value(:json_document, 'Report.Root.SummaryInformation.ExistingContracts.SubjectRole.id', '') = '1') then
       begin
         tmpstr = replace(replace(replace(f_json_blob_get_value(:json_document,'Report.Root.SummaryInformation.ExistingContracts.SubjectRole.TotalDebtOverdue.value', ''), 'KZT',''), ' ',''), ',', '.');
         overdue_payment = case when tmpstr <> '' then cast(tmpstr as decimal(18,2)) else 0 end;
       end
     end

     sznz = sznz + coalesce(overdue_payment, 0);
    --------------------------- СЗНЗ ---------------------------------КОНЕЦ
  end

  -- Рассчет КДД
  if (all_income > 0 and snz > 0) then  -- RFCKZ-215
  begin
    kdd = (sznz + snz) / all_income;
  end

  suspend;
end
