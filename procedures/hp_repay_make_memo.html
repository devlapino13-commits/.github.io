CREATE PROCEDURE HP_REPAY_MAKE_MEMO(
    K_ID DECIMAL(18,0),
    REP_DATE DATE,
    IS_WRITEOFF INTEGER,
    REPAY_CLIENT_FIO_OR_REASON VARCHAR(255),
    REPAY_TO_OS INTEGER,
    REPAY_TO_PERC INTEGER,
    REPAY_TO_FINE INTEGER,
    REPAY_CURR_RATE DECIMAL(18,4) = 0)
RETURNS (
    RES_MEMO VARCHAR(1024))
AS
declare variable s_text varchar(1024);
declare variable curr_currency varchar(3);
declare variable dm_kontract_no varchar(20);
declare variable dm_disb_date_str varchar(20);
declare variable client_koef double precision;
declare variable client_fio varchar(100);
declare variable dm_kal_new_kf_name str_100;
declare variable dm_currency varchar(3);
declare variable pr_code varchar(255);
begin
  res_memo = '';
  select cm_default_currency
  from company
  into curr_currency;

  select client_koef, client_fio
  from temp_params
  into :client_koef, client_fio;
  if (client_koef is null) then
    client_koef = 1;
  if (client_fio is null) then
    client_fio = '';

  if (repay_client_fio_or_reason is null) then
    repay_client_fio_or_reason = '';

  s_text = '';
  if (repay_to_os = 0) then
    s_text = s_text || ', основная сумма отложена';
  if (repay_to_perc = 0) then
    s_text = s_text || ', проценты отложены';
  if (repay_to_fine = 0) then
    s_text = s_text || ', накопленные штрафы отложены';
-- .MURABAHA.

  select k_kontract_no, k_kontract_date, kal_new_kf_name, k_index_curr_code, pr_code
  from kontract, knaktlist, product
  where k_id = :k_id and
        kal_from_kf_id = k_from_kf_id and
        knaktlist.is_delete = 0 and
        kal_date <= :rep_date and
        kal_date_end >= :rep_date and
        k_from_pr_id = pr_id
  into :dm_kontract_no, dm_disb_date_str, dm_kal_new_kf_name, dm_currency, pr_code;

  if (is_writeoff = 0) then
  begin
    res_memo = (case
                  when (curr_currency = '810') then 'Погашение займа согл. договора займа ' || dm_kontract_no || ' от ' || dm_disb_date_str || s_text
                  when (curr_currency = '944') then dm_kontract_no || ' nomreli ' || dm_disb_date_str || ' tarixli muqavileye esasen kredit odenishi'
                  when (curr_currency in ('398', '417') and not(pr_code containing '.IPF.')) then 'Погашение кредита согл. договора ' || dm_kontract_no || ' от ' || dm_disb_date_str || s_text -- было pr_code containing '.MURABAHA.' изменен от 02.04.2018 Улан
                  when (curr_currency in ('398', '417') and pr_code containing '.IPF.') then 'Погашение согл. договора финансирования ' || dm_kontract_no || ' от ' || dm_disb_date_str || s_text -- было pr_code containing '.MURABAHA.' изменен от 02.04.2018 Улан
                end);
  end
  else
  begin
    res_memo = (case
                  when (curr_currency = '810') then 'Списание займа согл. договора займа ' || dm_kontract_no || ' от ' || dm_disb_date_str || s_text
                  when (curr_currency = '944') then dm_kontract_no || ' nomreli ' || dm_disb_date_str || ' tarixli muqavileye esasen kredit silinmesi'
                  when (curr_currency in ('398', '417') and not(pr_code containing '.IPF.')) then 'Списание кредита по договору ' || dm_kontract_no || ' от ' || dm_disb_date_str || s_text -- было pr_code containing '.MURABAHA.' изменен от 02.04.2018 Улан
                  when (curr_currency in ('398', '417') and pr_code containing '.IPF.') then 'Списание, договор финансирования ' || dm_kontract_no || ' от ' || dm_disb_date_str || s_text -- было pr_code containing '.MURABAHA.' изменен от 02.04.2018 Улан
                end);
    if (repay_client_fio_or_reason <> '') then
      res_memo = res_memo || '    Основание: ' || repay_client_fio_or_reason;
  end
  if (curr_currency <> '944') then
  begin
    if (client_koef <> 1) then
      res_memo = res_memo || '. Клиент: ' || client_fio;
    if (dm_kal_new_kf_name <> '') then
      res_memo = res_memo || ' ' || trim(iif(curr_currency <> '417', 'Кр.папка №', 'Папка №')) || ' ' || dm_kal_new_kf_name;
    if (dm_currency <> curr_currency) then
      res_memo = res_memo || ' (Курс валюты: ' || repay_curr_rate || ')';
  end
  else
  begin
    if (client_koef <> 1) then
      res_memo = res_memo || '. Client: ' || client_fio;
    if (dm_kal_new_kf_name <> '') then
      res_memo = res_memo || ' kredit qovluq № ' || dm_kal_new_kf_name;
    if (dm_currency <> curr_currency) then
      res_memo = res_memo || ' (məzənnə: ' || repay_curr_rate || ')';
  end
  suspend;
end
