CREATE PROCEDURE AAREP_KONTRACT_LIST_PAR30_3(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    K_ID VARCHAR(18),
    KF_REGION_NAME D_STR_100,
    K_KONTRACT_NO VARCHAR(20),
    KF_NAME VARCHAR(100),
    KF_OFFICER_NAME VARCHAR(100),
    CL_CASHIER_NAME VARCHAR(100),
    K_DISB_DATE DATE,
    REP_DELIQ_BEGIN_DATE DATE,
    REP_DELIQ_DAYS_KOLICH INTEGER,
    REP_ACTIVE_BALANCE DECIMAL(18,4),
    REP_DELIQ_OS DECIMAL(18,4),
    REP_DELIQ_PERC DECIMAL(18,4),
    KAL_DATE_PRIN D_DATE,
    C_NAME_FIO_NEW D_STR_100,
    C_NAME_FIO_OLD D_STR_100,
    RD_NAME D_STR_255,
    KAL_DATE_PERED D_DATE,
    MANAGER_BEFORE_TRANSFER D_STR_100,
    MANAGER_LAST D_STR_100)
AS
declare variable KAL_DATE D_DATE;
declare variable OK_SH_ID ID_TYPE;
declare variable D_BEG D_DATE;
declare variable LY D_INTEGER;
declare variable LM D_INTEGER;
declare variable K_FROM_KF_ID decimal(18,0);
declare variable KAL_NEW_KF_OFFICER decimal(18,0);
declare variable D_END D_DATE;
declare variable C_NATION D_INTEGER;
declare variable C_ID D_ID;
begin
  lm = extract(month from :report_date_end) - 1;
  if (lm = 0) then
    lm = 12;
  ly = extract(year from :report_date_end);
  d_beg = '01.08.2024';
  d_end = '31.08.2024';

  for select ak.aarep_k_id, k_kontract_no, r1.reg_code || ' ' || r1.reg_name, akt.kal_new_kf_name, c1.c_name_fio,
             c3.c_name_fio, k_disb_date, ak.aarep_k_deliq_begin_date, ak.aarep_k_deliq_days, ak.aarep_k_balance,
             ak.aarep_k_deliq_os, ak.aarep_k_deliq_perc, k_from_kf_id, akt.kal_new_kf_officer, akt.kal_date, c3.c_nation, c3.c_id

      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 5) ak
      left join kontract kon on ak.aarep_k_id = kon.k_id
      left join product pr on kon.k_from_pr_id = pr.pr_id
      left join knaktlist akt on (ak.aarep_kf_id = akt.kal_from_kf_id and
            akt.kal_date <= ak.aarep_date_action and
            akt.kal_date_end >= ak.aarep_date_action and
            akt.is_delete = 0)
      left join clients c1 on akt.kal_new_kf_officer = c1.c_id
      left join clients c3 on kon.k_clients_cashier_id = c3.c_id
      left join region r1 on akt.kal_new_reg_id = r1.reg_unicode
--      left join clanguage cl on cl.cl_from_c_id = c3.c_id
--      left join refdata rd on rd.rd_id = coalesce(cl.cl_lang_no, c3.c_nation)
      where not exists(select * from knclient, bankrupt
            where kc_from_k_id = ak.aarep_k_id
            and kc_amount_approved > 0
            and kc_from_c_id = brt_from_c_id
            and :report_date_end between brt_date_begin and brt_date_end)

      -- kon.k_kontract_no='1401-373392-07'
      into :k_id, :k_kontract_no, :kf_region_name, :kf_name, :kf_officer_name, :cl_cashier_name, :k_disb_date,
           :rep_deliq_begin_date, :rep_deliq_days_kolich, :rep_active_balance, :rep_deliq_os, :rep_deliq_perc,
           :k_from_kf_id, :kal_new_kf_officer, :kal_date, :c_nation, :c_id
  do
  begin
    rd_name = null;
    select list(rd_name) from clanguage, refdata
    where cl_from_c_id = :c_id
    and rd_id = cl_lang_no
    into rd_name;
    if (rd_name is null) then
    begin
      select rd_name from refdata
      where rd_id = :c_nation
      into rd_name;
    end 

    ok_sh_id = null;
    select first 1 ok_sh_id
    from operkred
    where ok_type = 4 and
          ok_ol_date between :d_beg and :d_end and
          ok_ol_is_active = 1 and
          ok_os + ok_perc < 0 and
          ok_from_k_id = :k_id
    into :ok_sh_id;

    c_name_fio_old = '';
    kal_date_pered = null;

    select first 1 c.c_name_fio, kal.kal_date
    from knaktlist kal
    inner join clients c on c.c_id = kal.kal_prior_lo_id
    where kal.kal_new_kf_officer = :kal_new_kf_officer and
          kal.is_delete = 0 and
          kal.kal_from_kf_id = :k_from_kf_id and
          kal.kal_date >= :kal_date and
          kal.kal_prior_lo_id <> :kal_new_kf_officer
    order by kal.kal_date
    into :c_name_fio_old, :kal_date_pered;

    c_name_fio_new = '';
    kal_date_prin = null;
    select first 1 c.c_name_fio, kal.kal_date
    from knaktlist kal
    inner join clients c on c.c_id = kal.kal_new_kf_officer
    where kal.kal_prior_lo_id = :kal_new_kf_officer and
          kal.is_delete = 0 and
          kal.kal_from_kf_id = :k_from_kf_id and
          kal.kal_date >= :kal_date and
          kal.kal_new_kf_officer <> :kal_new_kf_officer
          -- kal.kal_date >= '01.09.2022'
    order by kal.kal_date
    into :c_name_fio_new, :kal_date_prin;
    --    if ((:ok_sh_id is null) and
    --        (:kf_officer_name <> 'КЦ Сирена') and
    --        (:rep_deliq_days_kolich >= 121) and
    --        (:rep_deliq_days_kolich <= 325)) then
    --      suspend;

    manager_before_transfer = null;
    if (kal_date_prin is not null) then
    begin
      manager_before_transfer = f_lo_mgr_name(f_lo_mgr_id(k_id, (kal_date_prin - 1), null, null));
    end 
    manager_last = f_lo_mgr_name(f_lo_mgr_id(:k_id, :report_date_end, null, null));




    if ((:ok_sh_id is null) and
        (:kf_officer_name = 'КЦ Сирена') and
        (:rep_deliq_days_kolich >= 60) and
        (:rep_deliq_days_kolich <= 300)) then
      suspend;
  end ----end of 'first for"
end
