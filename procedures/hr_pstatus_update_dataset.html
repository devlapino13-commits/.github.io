CREATE PROCEDURE HR_PSTATUS_UPDATE_DATASET(
    PS_ID INTEGER,
    PS_FROM_HO_ID ID_TYPE,
    PS_DATE_BEGIN DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_FROM_REG_ID INTEGER,
    PS_FROM_DEP_ID INTEGER,
    PS_FROM_JL_ID INTEGER,
    PS_IS_ON_STAFF SMALLINT,
    PS_IS_WORKING SMALLINT,
    PS_FROM_C_ID ID_TYPE,
    PS_SALARY CURR_TYPE,
    PS_TRANSFER_OFFICE INTEGER,
    PS_TABEL_NO VARCHAR(10),
    PS_CLASS INTEGER,
    PSA_AGREEMENT_NO VARCHAR(10))
AS
begin

  update hr_pstatus ps
  set ps.ps_date_begin = :ps_date_begin,
      ps.ps_period_month = :ps_period_month,
      ps.ps_date_end_in_order = :ps_date_end_in_order,
      ps.ps_from_reg_id = :ps_from_reg_id,
      ps.ps_from_dep_id = :ps_from_dep_id,
      ps.ps_from_jl_id = :ps_from_jl_id,
      ps.ps_is_on_staff = :ps_is_on_staff,
      ps.ps_is_working = :ps_is_working,
      ps.ps_from_c_id = :ps_from_c_id,
      ps.ps_salary = :ps_salary,
      ps.ps_transfer_office = :ps_transfer_office,
      ps.ps_tabel_no = :ps_tabel_no,
      ps.ps_class = :ps_class
  where ps.ps_id = :ps_id and
        ps.ps_from_ho_id = :ps_from_ho_id;

  update hr_pstatus_agreement psa
  set psa.psa_agreement_no = :psa_agreement_no
  where psa.psa_from_ho_id = :ps_from_ho_id and
        psa.psa_from_ps_id = :ps_id;
end
