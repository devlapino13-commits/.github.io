CREATE PROCEDURE UXP$UXIMPORT_PAIRS_PREPARE
AS
declare variable up_old_uuid d_uuid;
declare variable up_new_uuid d_uuid;
declare variable mm_uuid d_uuid;
begin
  delete from ux$pairuuid
  where 1 = 1;

  mm_uuid = uxf$sys_id();
  for select ufm_uuid
  from ux$code, ux$form
  where ux_us_uuid = :mm_uuid and
        ux_uuid = ufm_ux_uuid
  into up_new_uuid
  do
  begin
    update or insert into ux$pairuuid (up_old_uuid, up_new_uuid)
    values (:up_new_uuid, :up_new_uuid);
  
    for select ufr_uuid
    from ux$frame
    where ufr_owner = :up_new_uuid
    into up_old_uuid
    do
    begin
      update or insert into ux$pairuuid (up_old_uuid, up_new_uuid)
      values (:up_old_uuid, :up_old_uuid);
    end

    for select ufp_uuid
    from ux$frparam, ux$frame
    where ufr_owner = :up_new_uuid and
          ufr_uuid = ufp_ufr_uuid
    into up_old_uuid
    do
    begin
      update or insert into ux$pairuuid (up_old_uuid, up_new_uuid)
      values (:up_old_uuid, :up_old_uuid);
    end
  end

  mm_uuid = uxf$mainmenu();
  update or insert into ux$pairuuid (up_old_uuid, up_new_uuid)
  values (:mm_uuid, :mm_uuid);

  for select ufr_uuid
  from ux$frame
  where ufr_owner = :mm_uuid
  into up_old_uuid
  do
  begin
    update or insert into ux$pairuuid (up_old_uuid, up_new_uuid)
    values (:up_old_uuid, :up_old_uuid);
  end
end
