CREATE PROCEDURE AAREP_REPAYMENT_STAT_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    APL_DATE D_DATE,
    APL_REGION D_STR_100,
    APL_PAY_KIND D_STR_CODE,
    APL_BALANCE D_STR_CODE,
    APL_AMOUNT D_CURRENCY,
    APL_OS D_CURRENCY,
    APL_PERC D_CURRENCY,
    APL_FINE D_CURRENCY,
    APL_COUNT D_INTEGER)
AS
declare variable reg_id d_integer;
declare variable ok_pay_type integer;
declare variable ol_id_prv d_id;
declare variable project_code d_str_code;
declare variable main_ol_id d_id;
begin
  ol_id_prv = null;
  for select id_list, reg_code || ' ' || reg_name
  from region, region_list(:rep_region_id, 1, 3, 1)
  where id_list = reg_unicode
  into reg_id, apl_region
  do
  begin
    for select iif(ol_parent_id is not null, ol_parent_id, ol_id), ok_ol_date, ok_pay_type,
               -ok_os - ok_perc - ok_fine - ok_kommis - ok_kommis_adv, -ok_os, -ok_perc, -ok_fine,
               'Баланс'
    from operkred, operlist
    where ok_type = 4 and
          ok_ol_date between :rep_date_begin and :rep_date_end and
          ok_ol_is_active = 1 and
          ok_ol_reg_id = :reg_id and
          ok_from_ol_id = ol_id
    union all
    select iif(ol_parent_id is not null, ol_parent_id, ol_id), oko_ol_date,
           iif(ol_doc_kind = 1, 1, iif(ol_doc_kind = 7, 2, 3)),
           -oko_os - oko_perc - oko_other, -oko_os, -oko_perc, -oko_other,
           'Забаланс'
    from operkroff, operlist
    where oko_type = 4 and
          oko_ol_date between :rep_date_begin and :rep_date_end and
          oko_from_ol_id = ol_id and
          ol_from_reg_id = :reg_id
    order by 1
    into main_ol_id, apl_date, ok_pay_type, :apl_amount, :apl_os, :apl_perc, :apl_fine, :apl_balance
    do
    begin
      apl_count = 0;
      if (ol_id_prv is null or ol_id_prv <> main_ol_id) then
      begin
        apl_count = 1;
        apl_pay_kind = 'Касса компании';
        if (ok_pay_type = 2)
        then apl_pay_kind = 'Расчетный счет';
        else
        if (ok_pay_type = 3) then
        begin
          project_code = '';
          select esm_project
          from extsysoper, extsysmain
          where eso_from_ol_id = :main_ol_id and
                eso_from_esm_id = esm_id
          rows 1
          into project_code;

          if (project_code = '')
          then apl_pay_kind = 'Внутренний бух.счет';
          else apl_pay_kind = project_code;

          if (apl_balance = 'Забаланс' and apl_pay_kind = 'Внутренний бух.счет')
          then apl_pay_kind = 'Другие счета';
        end

        ol_id_prv = main_ol_id;
      end
      suspend;
    end
  end

end
