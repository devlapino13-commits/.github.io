CREATE PROCEDURE MC_BARCODE_SAVE(
    MCL_ID BIGINT,
    BARCODE D_INTEGER,
    GOODS_NAME D_STR_LONG)
RETURNS (
    RESULT D_STR_100)
AS
declare variable c_id id_type;
declare variable cs_id id_type;
declare variable gll_id id_type;
begin
  --Антон вызывает процедуру что бы добавить товар
  -- предворительно проверяем нет ли в таблице товара с таким штрих кодом, если нет то добовляем
  select mc.mcl_from_c_id
  from mc_login mc
  where mc.mcl_id = :mcl_id
  into :c_id;

  select cs.cs_id
  from clientshops cs
  where cs.cs_from_c_id = :c_id
  into :cs_id;

  if (not exists(select *
                 from gi_local_list cll
                 where cll.gll_barcode = :barcode)) then
  begin
    insert into gi_local_list (gll_from_mcl_id, gll_from_cs_id, gll_name, gll_barcode)
    values (:mcl_id, :cs_id, :goods_name, :barcode);
  end
  suspend;
end
