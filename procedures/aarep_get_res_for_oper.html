CREATE PROCEDURE AAREP_GET_RES_FOR_OPER(
    REP_K_ID ID_TYPE,
    REP_DATE_END DATE,
    REP_DB_ID INTEGER)
RETURNS (
    IS_RESTRUCT SMALLINT,
    RESTRUCT_MEMO VARCHAR(255),
    RESTRUCT_STR VARCHAR(255),
    RESTRUCT_DATE DATE,
    RESTRUCT_KOL INTEGER,
    RESTRUCT_DATE_LAST DATE)
AS
declare variable OL_FROM_RD_ID integer;
declare variable KOL_ROWS integer;
declare variable OL_DATE date;
declare variable OL_USER_ID ID_TYPE;
declare variable RD_NAME varchar(255);
declare variable RD_MEMO varchar(255);
declare variable IS_FIRST smallint;
declare variable MAX_DATE_GRAPH date;
declare variable MAX_DATE_REST date;
declare variable OR_FROM_OL_ID ID_TYPE;
declare variable MAX_DAY_IN_LAST_PAYMENT D_INTEGER;
begin
  -- данная процедура аналог AAREP_GET_RESTRUCTURING только не имеется ограничение на 31 дней
  ol_from_rd_id = 0;
  is_restruct = 0;
  restruct_date = null;
  restruct_kol = 0;
  is_first = 0;
  rd_name = '';
  rd_memo = '';
  restruct_memo = null;
  restruct_str = null;
  if (rep_db_id is null) then
    rep_db_id = f_get_db_id();

  max_day_in_last_payment = 0;

  -- с 31.03.2020 в РФ увеличение срока не более на 31 дней не считаем реструктуризацией, Sanjari, 13.05.2020
--  if (rep_db_id = 81001 and rep_date_end >= '01.03.2020') then
--    max_day_in_last_payment = 31;

  for select
        coalesce(or_from_rd_id, 1),
        or_ol_date,
        or_from_ol_id,
        ol_user_id
      from operrest
        inner join operlist
          on ol_id = or_from_ol_id
      where or_from_k_id = :rep_k_id and
            or_ol_is_active = 1 and
            :rep_date_end between or_ol_date and or_date_end
      into
        :ol_from_rd_id,
        :ol_date,
        :or_from_ol_id,
        :ol_user_id
  do
  begin
    is_restruct = 0;

    if (rep_date_end < '01.12.2016' or rep_db_id not in (39801, 41701)) then
    begin
      is_restruct = 1;
      if (rep_db_id = 81001) then
      begin
        select max(iif(ok_type = 1, ok_sh_date, cast('01.01.1900' as date))),
               max(iif(ok_type = 2 and ok_from_ol_id = :or_from_ol_id, ok_sh_date, cast('01.01.1900' as date)))
        from operkred
        where ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date_end and
              ok_from_k_id = :rep_k_id and
              ok_type in (1, 2) and
              ok_from_c_id = 0
        into max_date_graph, max_date_rest;

        if (datediff(day, max_date_graph, max_date_rest) <= max_day_in_last_payment) then
          is_restruct = 0;
      end

      if (is_restruct = 1) then
      begin
        if (restruct_date is null) then
          restruct_date = ol_date;
        restruct_date_last = ol_date;
      end

    end
    else
    begin
      rd_name = '';
      rd_memo = '';
      if (ol_from_rd_id > 0) then
      begin
        select rd_name, rd_memo
        from refdata
        where rd_id = :ol_from_rd_id
        into rd_name, rd_memo;

        if (rd_memo = 'Реструктуризация')
          then is_restruct = 1;
      end else begin
        if (ol_user_id = 1000000000000)
          then is_restruct = 1;
      end

      if (is_restruct = 1) then
      begin
        if (is_first = 0) then
        begin
          is_first = 1;
          restruct_memo = rd_name;
          restruct_str = rd_memo;
          restruct_date = ol_date;
        end
        restruct_date_last = ol_date;
      end
    end
    restruct_kol = restruct_kol + is_restruct;
  end
-- нижнее условие сделано для того чтобы если было несколько реструктуризации и из них последний не реструктуризация
  if (restruct_kol > 0) then
    is_restruct = 1;
  else
  if (rd_name <> '' or rd_memo <> '') then
  begin
    restruct_memo = rd_name;
    restruct_str = rd_memo;
  end

  suspend;
end
