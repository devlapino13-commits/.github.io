CREATE PROCEDURE HP_GET_CLIENT_TYPE_OLD(
    C_ID D_ID,
    CURR_CITIZEN_ID D_INTEGER,
    DOC_VNJ_ID D_INTEGER,
    REP_DATE D_DATE = null)
RETURNS (
    RES_CLIENT_TYPE INTEGER,
    RES_NON_RESIDENT INTEGER,
    RES_FROM_CDS_ID INTEGER)
AS
declare variable db_id integer;
declare variable citizen_str d_str_code;
begin
  -- устаревшая процедура

  -- определяет резидентсво контрагента и тип (ЮЛ, ФЛ или ИП)

  if (rep_date is null) then
    rep_date = current_date;
  res_from_cds_id = null;

  if (curr_citizen_id is null) then
  begin
    db_id = f_get_db_id();
    citizen_str = trim((case db_id
                          when 81001 then 'citizen.ru'
                          when 41701 then 'citizen.kg'
                          when 94401 then 'citizen.az'
                        end));

    select rd_id
    from refdata
    where rd_code = :citizen_str
    into :curr_citizen_id;
  end

  if (doc_vnj_id is null) then
  begin
    select pk_id
    from passkind, passdoctype
    where pdt_id = pk_from_pdt_id and
          pk_is_delete = 0 and
          pdt_code = 'registration_vnj' rows 1
    into :doc_vnj_id;
  end

  select iif(c_citizen = :curr_citizen_id, 0, 1),
         iif(coalesce(c_type, 0) = 1, iif(c_name_fio like 'ИП %' or c_name_fio like '% ИП', 2, 1), 3)
  from clients
  where c_id = :c_id
  into :res_non_resident, res_client_type;
  if (res_non_resident is null) then
    res_non_resident = 0;

  if (res_non_resident = 1 and
      res_client_type = 3) then
  begin
    if (doc_vnj_id is not null) then
    begin
      -- если имеется действующий документ ВНЖ, то резидент
      -- начиная с апреля 2019 добавлено новое условие:
      -- смотрим на время добавления документа, чтобы исторические отчеты не изменились, Sanjari 28.04.2019
      select cds_id
      from clientdocs
      where cds_from_c_id = :c_id and
            cds_from_pk_id = :doc_vnj_id and
            cds_doc_date_begin <= :rep_date and
            cds_doc_date_end >= :rep_date and
            cds_is_delete = 0 and
            coalesce(cds_create_datetime, cds_datetime) < (:rep_date + 1) -- новое условие
      order by cds_doc_date_begin desc, cds_datetime desc
      rows 1
      into :res_from_cds_id;

      if (res_from_cds_id is not null) then
        res_non_resident = 0;
    end
  end

  suspend;
end
