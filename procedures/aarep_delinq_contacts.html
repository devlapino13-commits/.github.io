CREATE PROCEDURE AAREP_DELINQ_CONTACTS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    K_KONTRACT_NO VARCHAR(20),
    KONT_AMOUNT DECIMAL(18,4),
    DELIQ_BALANCE DECIMAL(18,4),
    CLIENT_KOLICH INTEGER,
    K_DISB_DATE DATE,
    CLIENT_AMOUNT DECIMAL(18,4),
    AA_DELIQ_DAYS INTEGER,
    AA_OFFICE VARCHAR(100),
    AA_KC VARCHAR(100),
    CLIENT_ADDRESS VARCHAR(1024),
    CONTACT_ADDRESS_1 VARCHAR(1024),
    CONTACT_ADDRESS_2 VARCHAR(1024),
    CONTACT_ADDRESS_3 VARCHAR(1024),
    KNC_STR_CITIZEN VARCHAR(100),
    KNC_DOC_TYPE_NAME VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_BIRTH_COUNTRY VARCHAR(100),
    C_BIRTH_CITY VARCHAR(100))
AS
declare variable CA_ADD varchar(255);
declare variable CA_STR varchar(255);
declare variable CC_NAME varchar(100);
declare variable CC_ADDRESS varchar(255);
declare variable CC_CELL_NO varchar(255);
declare variable CC_STATUS varchar(255);
declare variable CON_NO integer;
declare variable CC_REGION varchar(100);
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select distinct c_id, c_name_fio, c_pass_no, kontract.k_kontract_no, k_amount_approved, -- aa_loan_amount
                      aa_deliq_balance, k_kolich_client_approved, -- aa_kolich_cl_active_end
                      k_disb_date, -- dd_disb_date
                      kc_amount_approved, aa_deliq_days, aa_curr_office_name, aa_curr_kc_fio, rd_name,
                      pk_country || ' ' || pk_name, cl.c_birth_date, cl.c_birth_country, cl.c_birth_city
      from knclient, clients cl, kontract, refdata, passkind, aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 0, 0, 0, 0, 0)
      where kc_from_c_id = c_id and
            k_id = kc_from_k_id --
            and
            kc_amount_approved > 0 and
            aa_k_id = k_id --
            and
            aa_deliq_days > 30 and
            rd_id = c_citizen and
            pk_id = c_doc_type
      into :c_id, :c_name_fio, :c_pass_no, :k_kontract_no, :kont_amount, :deliq_balance, :client_kolich, :k_disb_date,
           :client_amount, :aa_deliq_days, :aa_office, :aa_kc, :knc_str_citizen, :knc_doc_type_name, :c_birth_date,
           c_birth_country, c_birth_city
  do
  begin
    client_address = '';
    cc_region = '';

    for select ca_address, ca_street
        from caddress
        where ca_from_c_id = :c_id and
              ca_is_main = 1 rows 1
        into :ca_add, :ca_str

    do
    begin
      if (ca_add is null) then
        ca_add = '';
      if (ca_str is null) then
        ca_str = '';

      client_address = client_address || ' ' || ca_add || ' ' || ca_str;
    end

    contact_address_1 = '';
    contact_address_2 = '';
    contact_address_3 = '';
    con_no = 0;

    for select ct.cc_name, ct.cc_address, ct.cc_cell_no, ct.cc_status
        from contacts ct
        where cc_from_c_id = :c_id
        into :cc_name, :cc_address, :cc_cell_no , :cc_status



    do
    begin

      con_no = con_no + 1;
      if (con_no = 1) then
        contact_address_1 = cc_name || ', ' || cc_status || ', ' || cc_cell_no || '. ';
      if (con_no = 2) then
        contact_address_2 = cc_name || ', ' || cc_status|| ', ' || cc_cell_no || '. ';
      if (con_no = 3) then
        contact_address_3 = cc_name || ', ' || cc_status || ', ' || cc_cell_no || '. ';
    end

    suspend;
  end

end
