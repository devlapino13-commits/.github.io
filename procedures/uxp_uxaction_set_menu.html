CREATE PROCEDURE UXP$UXACTION_SET_MENU(
    UX_UUID D_UUID,
    OLD_ITEM_CODE D_STR_100,
    NEW_ITEM_CODE D_STR_CODE,
    UX_UUID_CODE D_STR_CODE = null)
AS
declare variable len d_integer;
declare variable ua_uuid d_uuid;
declare variable old_is_root d_bool;
begin
  if (ux_uuid is null) then
    ux_uuid = uxf$uniq(ux_uuid_code);

  len = char_length(old_item_code);

  if (right(old_item_code, 2) = '00' and new_item_code = left(old_item_code, len - 2)) then
  begin
    -- Удаление ветки
    if (exists(select * from ux$submenu
    where usm_ufm_uuid = :ux_uuid and
          usm_menu_index starting :new_item_code and
          char_length(usm_menu_index) >= :len)) then
      exception e_message 'Сначала удалите дочерние группы меню';
  end

  if (new_item_code = old_item_code) then
    exit;

  len = char_length(new_item_code);
  if (right(new_item_code, 2) = '00') then
  begin
    select ua_menu_index
    from ux$action
    where ua_owner = :ux_uuid and
          ua_menu_index starting left(:new_item_code, :len - 2) and
          char_length(ua_menu_index) = :len
    order by ua_menu_index desc
    rows 1
    into new_item_code;

    new_item_code = left(new_item_code, len - 2) || right('0' || (cast(right(new_item_code, 2) as integer) + 1), 2);

    if (row_count = 0) then
    begin
      -- Создается новая ветка
      update ux$action
      set ua_menu_index = :new_item_code
      where ua_owner = :ux_uuid and
            ua_menu_index = :old_item_code;

      exit;
    end
  end

  len = char_length(old_item_code);

  old_is_root = 0;
  if (right(old_item_code, 2) = '00') then
    old_is_root = 1;

  for select ua_uuid
  from ux$action
  where ua_owner = :ux_uuid and
        ua_menu_index starting left(:old_item_code, :len - 2) and
        char_length(ua_menu_index) = :len and
        ((ua_menu_index = :old_item_code and :old_is_root = 0) or
         (ua_menu_index >= :old_item_code and :old_is_root = 1))
  order by ua_menu_index desc
  into ua_uuid
  do
  begin
    execute procedure uxp$uxaction_menu_move(:ux_uuid, :ua_uuid, -1);

    update ux$action
    set ua_menu_index = :new_item_code
    where ua_uuid = :ua_uuid;

    execute procedure uxp$uxaction_menu_move(:ux_uuid, :ua_uuid, 1);
  end
end
