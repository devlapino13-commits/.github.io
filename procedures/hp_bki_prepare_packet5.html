CREATE PROCEDURE HP_BKI_PREPARE_PACKET5(
    DEBUG_MODE D_BOOL = 0,
    BKT_FORMAT D_SMALLINT = null)
AS
declare variable packet_id d_id;
declare variable event_date d_date;
declare variable object_kind d_integer;
declare variable object_id d_id;
declare variable bkk_id d_id;
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable bkc_id d_id;
declare variable bkl_id d_id;
declare variable bev_id d_id;
declare variable next_status_date d_date;
declare variable packet_date d_date;
declare variable tag d_integer;
declare variable bkt_action_code d_str_code;
declare variable c_id d_id;
declare variable time_start d_datetime;
declare variable cnt d_integer;
declare variable start_date d_date;
declare variable op_next_payout_date d_date;
declare variable op_next_payout_date_tmp d_date;
declare variable debt_calc_date d_date;
declare variable loan_is_close boolean;
declare variable jso d_integer;
declare variable jsa d_integer;
declare variable jso_clients d_integer;
declare variable xml_version_id d_integer;
declare variable bkc_add d_str_long;
declare variable need_find_next_event boolean;
declare variable i d_smallint;
begin
  time_start = current_timestamp;
  xml_version_id = null;

  if (bkt_format is null) then  -- enum  bki_packet_format
    bkt_format = 0;

  select bxv_id
  from bki_xml_version
  where bxv_date_begin <= current_date
  order by bxv_date_begin desc
  rows 1
  into xml_version_id;

  if (xml_version_id is null) then
    exception e_default 'Not found xml_version_id';

  if (bkt_format = 0) then
    execute procedure hp_bki_corr_prepare_packet(1, null);

  for select bkp_id, bkk_id, bkp_date, f_enum('bki_packet_action', bkt_action, 'code')
  from bki_packet, bki_kind, bki_packet_type
  where bkp_status = 0
    and bkk_id = bkp_from_bkk_id
    and bkt_id = bkp_from_bkt_id
    and bkt_format = :bkt_format
  into packet_id, bkk_id, packet_date, bkt_action_code
  do
  begin
    for select distinct bkl_id, bkl_object_id, bkl_object_kind
    from bki_loans, bki_clients, bki_events
    where bkl_from_bkp_id = :packet_id
      and bkc_from_bkl_id = bkl_id
      and bev_from_bkc_id = bkc_id
      and bev_status = 0
      and bev_event_id is null
    order by bkl_object_id
    into bkl_id, object_id, object_kind
    do
    in autonomous transaction do
    begin
      jso_clients = json.obj_create();

      while (true)
      do
      begin
        select bev_event_date, bev_tag, bev_id, bkc_from_c_id, bkc_id, bkc_add
        from bki_clients, bki_events
        where bkc_from_bkl_id = :bkl_id
          and bev_from_bkc_id = bkc_id
          and bev_status = 0
          and bev_event_id is null
        order by bev_event_date
        rows 1
        into event_date, tag, bev_id, c_id, bkc_id, bkc_add;
    
        if (row_count = 0) then
           leave;              -- выход из цикла

        -- уберем дубли (могут возникнуть при очистке пакета, либо при ручной вставке и т.д.)
        delete from bki_events
        where bev_from_bkc_id = :bkc_id
          and bev_status = 0
          and bev_event_id is null
          and bev_event_date = :event_date
          and bev_id <> :bev_id;

        k_id = null;
        ma_id = null;
        next_status_date = null;

        if (object_kind = f_objkind('miniapp', 'id')) then
        begin
          ma_id = object_id;
      
          select k_id
          from kontract k
          where k.k_from_ma_app_id = :ma_id
            and k.k_status >= 0
          order by 1 desc
          rows 1
          into k_id;

          if (k_id is not null) then
          begin
            object_kind = f_objkind('kontract', 'id');
            object_id = k_id;
          end
        end
        else if (object_kind = f_objkind('kontract', 'id')) then
        begin
          k_id = object_id;
      
          select k.k_from_ma_app_id
          from kontract k
          where k.k_id = :k_id
          into ma_id;
        end
  
        if (object_kind = 9
            and bkt_format = 0
            and exists(select bsl_id from bki_stoplist_loans where bsl_from_k_id = :object_id) -- если контракт в стоплисте
            and (bkt_action_code <> 'delete') -- и это не удаление (из стоплиста позволяем только удалять)
        ) then
        begin
          update bki_events
          set bev_status = -3
          where bev_id = :bev_id;

          continue;
        end

        -- если переотправка, то данные "ручного" Блока 23 надо отправить тоже
        if (bkt_format = 0 and bkt_action_code in ('recreate', 'reset')) then
        begin
          start_date = iif(bkt_action_code = 'reset', json.obj_get_val(json.parse(bkc_add), 'start_date', '01.01.1900'), '01.01.1900');

          update bki_once_data
          set bod_from_bkc_id = null
          where bod_kind = 3
            and bod_object_kind = 9
            and bod_object_id = :k_id
            and bod_add is not null
            and json.obj_get_val(json.parse(bod_add), 'applyDate', '01.01.1900') >= :start_date;
        end

        jsa = json.arr_create();

        execute procedure hp_bki_calculate_loan5(:jsa, :jso_clients, :xml_version_id, :bev_id);

        -- если строка не была обработана то поставим статус "пустая строка"
        update bki_events
        set bev_status = -3
        where bev_id = :bev_id
          and bev_status = 0;

        need_find_next_event = (bkt_format = 0);

        if (need_find_next_event and bkt_action_code = 'delete') then
          need_find_next_event = false;

        if (need_find_next_event and tag in (8,9,30,55)) then
          need_find_next_event = false;

        if (need_find_next_event and tag = 22) then -- банкротство
        begin
          -- если не было других событий в этот день
          if ((select bev_event_id = f_enum('bki_event', '1.12_change_bankruptcy', 'id')
              from bki_events
              where bev_id = :bev_id))
          then
            need_find_next_event = false;
        end 

        if (need_find_next_event) then
        begin
          -- поиск следующего события
          loan_is_close = false;
          op_next_payout_date = null;
          debt_calc_date = null;

          for select jso
          from hp_bki_json_get_list(:jsa, 1)
          where json.obj_get_val(jso, 'is_poruch', '0') = '0'
          into jso
          do
          begin
            if (json.obj_get_val(jso, 'status') = 'close') then
              loan_is_close = true;
            else
            begin
              op_next_payout_date_tmp = json.obj_get_val(jso, 'op_next_payout_date');
  
              if (op_next_payout_date_tmp < coalesce(op_next_payout_date, '31.12.9999')
                and json.obj_get_val(jso, 'is_poruch', '0') = '0')
              then
                op_next_payout_date = op_next_payout_date_tmp;
  
              if (debt_calc_date is null) then
                debt_calc_date = json.obj_get_val(:jso, 'debt_calc_date');
            end
          end

          if (not loan_is_close) then
          begin
            select min(ma_date_app)
            from miniapp
            where ma_id = :ma_id
              and ma_date_app > :event_date
            into next_status_date;

            select eds_status_datetime
            from hp_edoc_client_sign(:c_id)
            where eds_status = 1
              and edl_from_edc_id = f_edoc_kind('loan.application','id')
              and edl_object_kind = 11
              and edl_object_id = :ma_id
              and cast(eds_status_datetime as date) between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
            order by 1
            rows 1
            into next_status_date;

            select msg_datetime
            from edoc_sign, edoc_list, edoc_msg, fsrobot_msgbox, kontract
            where eds_from_edl_id = edl_id
              and edm_from_eds_id = eds_id
              and msg_id = edm_from_msg_id
              and edl_object_id = k_id
              and edl_from_edc_id = f_edoc_kind('loan.agreement','id')
              and edl_object_kind = 9
              and k_id = :k_id
              and cast(msg_datetime as date) between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
            into next_status_date;

            select ks_date_begin
            from kontstate ks
            where ks_from_k_id = :k_id
              and ks_ol_is_delete = 0
              and ks_date_begin between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
            order by 1
            rows 1
            into next_status_date;
            if (event_date = op_next_payout_date) then -- если следующий платеж сегодня, то значит его сегодня не было, надо отправить на завтра образование просрочки
              next_status_date = event_date + 1;
            else
            begin
              if (op_next_payout_date between event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')) then
                next_status_date = op_next_payout_date;

              if (coalesce(next_status_date, '31.12.9999') > debt_calc_date + iif(xml_version_id < 2, 30, 31)) then --через 30 дней после даты расчета задолженности
                next_status_date = debt_calc_date + iif(xml_version_id < 2, 30, 31);
        
              select ok_ol_date
              from operkred ok
              where ok_ol_is_active = 1
                and ok_from_k_id = :k_id
                and ok_ol_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select ok_sh_date
              from operkred ok
              where ok_ol_is_active = 1
                and ok_type in (1,2,3,4,6)
                and ok_from_k_id = :k_id
                and ok_sh_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select oko_ol_date
              from operkroff
              where oko_ol_is_active = 1
                and oko_from_k_id = :k_id
                and oko_ol_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select lam_date_begin
              from lawkredit, lawmain
              where lak_from_lam_id = lam_id
                and lak_from_k_id = :k_id
                and lam_is_active = 1
                and lam_date_begin is not null
                and lam_date_begin between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select ker_date
              from konteffrate ker
              where ker.ker_from_k_id = :k_id
                and ker.ker_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
                and ker.ker_state = 0
              order by 1
              rows 1
              into next_status_date;

              select kd_ol_date
              from kontdata kd
              where kd_ol_is_delete = 0
                and kd_from_k_id = :k_id
                and kd_ol_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select ol_date
              from kontpsk, operlist
              where kps_from_k_id = :k_id
                and ol_id = kps_from_ol_id
                and is_delete = 0
                and ol_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select z_date_begin
              from zalog
              where z_from_k_id = :k_id
                and z_date_begin between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select zap_date
              from zalogautoprof
              inner join zalogauto on zap_from_za_id = za_id
              where za_from_k_id = :k_id
              and zap_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select cast(f_json_str_get_value(bod_add, 'applyDate', null) as date)
              from bki_once_data bod
              where bod_kind = 3
                and bod_object_kind = 9
                and bod_object_id = :k_id
                and bod_from_bkc_id is null
                and bod_add is not null
                and cast(f_json_str_get_value(bod_add, 'applyDate', null) as date) between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select bbt_date
              from bki_bankrupt
              where bbt_from_k_id = :k_id
                and bbt_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select brt_date_fact
              from bankrupt
              where brt_from_c_id = :c_id
              and brt_date_fact between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select bbt_date
              from kontract, bki_bankrupt
              where bbt_from_k_id = k_id
                and k_base_k_id = :k_id
                and bbt_date between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;

              select k_stop_perc
              from kontract
              where k_id = :k_id
                and k_stop_perc between :event_date + 1 and coalesce(:next_status_date - 1, '31.12.9999')
              order by 1
              rows 1
              into next_status_date;
           end
          end
          else if (json.obj_get_val(jso, 'is_poruch', '0') = '0') then
            delete from bki_buffer
            where bkb_from_bkk_id = :bkk_id
              and bkb_object_kind = 9
              and bkb_object_id = :k_id
              and bkb_date >= :event_date
              and bkb_type = 0;

          if (next_status_date < packet_date) then
          begin
            if (not exists (select 1 from bki_loans, bki_clients, bki_events
                            where bkl_from_bkp_id = :packet_id
                              and bkc_from_bkl_id = bkl_id
                              and bev_from_bkc_id = bkc_id
                              and bev_event_date = :next_status_date
                              and bkl_object_kind = :object_kind
                              and bkl_object_id = :object_id))
            then
              insert into bki_events (bev_from_bkc_id, bev_event_date, bev_tag)
              values (:bkc_id, :next_status_date, 11);
          end
          else if (next_status_date is not null) then
            update or insert into bki_buffer (bkb_from_bkk_id, bkb_object_kind, bkb_object_id, bkb_date, bkb_tag, bkb_type)
            values(:bkk_id, :object_kind, :object_id, :next_status_date, 11, 0)
            matching(bkb_from_bkk_id, bkb_object_kind, bkb_object_id, bkb_type, bkb_date);
        end

        -- что бы не тормозить весь пакет, запишем ошибку, если возникнет
        when any do
        begin
          if (coalesce(debug_mode, 0) <> 0) then
            exception;
  
          update bki_events
          set bev_status = -1
          where bev_id = :bev_id;
    
          insert into bki_errors (bke_from_bev_id, bke_date, bke_error_code, bke_error_id)
          values (:bev_id, current_date, 'DB', f_memo_id('DB'));
        end
      end

--      if (datediff(minute, time_start, cast('now' as timestamp)) > 27) then
--      begin
--        i = (select coalesce(max(tol_object_kind), 0) from temp_objlist where tol_kind = 'skv_proc_time_100726' and tol_object_id = :packet_id);
--      
--        insert into temp_objlist (tol_kind, tol_object_kind, tol_object_id, tol_add_info_1, tol_add_info_2)
--        values ('skv_proc_time_100726', :i + 1, :packet_id, 'HP_BKI_PREPARE_PACKET5', datediff(second, :time_start, cast('now' as timestamp)));
--
--        exit;
--      end
    end

    in autonomous transaction do
    begin
      -- пакет подготовлен
      select count(bkc_id)
      from bki_loans, bki_clients, bki_events
      where bkc_from_bkl_id = bkl_id
        and bev_from_bkc_id = bkc_id
        and bkl_from_bkp_id = :packet_id
        and bev_status = 1
      into cnt;

      update bki_packet
      set bkp_status = iif(:cnt > 0, 1, -1),
          bkp_result_status = iif(:cnt > 0, 'Prepared', 'EmptyPacket')
      where bkp_id = :packet_id;
    end
  end

  i = (select coalesce(max(tol_object_kind), 0) from temp_objlist where tol_kind = 'skv_proc_time_100726' and tol_object_id = :packet_id);

  if (packet_id is not null) then
    insert into temp_objlist (tol_kind, tol_object_kind, tol_object_id, tol_add_info_1, tol_add_info_2, tol_add_info_3)
    values ('skv_proc_time_100726', :i + 1, :packet_id, 'HP_BKI_PREPARE_PACKET5', datediff(second, :time_start, cast('now' as timestamp)), f_datetimetostr(:time_start));

  if (bkt_format = 0 and packet_id is not null and coalesce(debug_mode, 0) = 0) then
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind)
    values (0, 's.kravtsov@mbfplanet.com', 'БКИ. Подготовка пакетов', 'Обработка завершена.', 1, 1000000000001, '');
end
