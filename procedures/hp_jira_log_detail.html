CREATE PROCEDURE HP_JIRA_LOG_DETAIL(
    ISSUE_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE = null,
    REP_DATE_END D_DATE = null)
RETURNS (
    JRS_CODE D_STR_CODE,
    JRS_TEXT D_STR_CODE,
    JRS_NEW_CODE D_STR_CODE,
    JRS_NEW_TEXT D_STR_CODE,
    JRS_CAPTION D_STR_CODE,
    JRS_NEW_CAPTION D_STR_CODE,
    MIN_DATETIME D_DATETIME,
    MAX_DATETIME D_DATETIME,
    SECOND_TOTAL D_INTEGER,
    SECOND_PERIOD D_INTEGER,
    TIME_TOTAL D_STR_CODE,
    TIME_PERIOD D_STR_CODE,
    JRS_NEW_RESULT D_SMALLINT,
    JRS_IS_LAST D_BOOL,
    GMT_TIMEZONE D_STR_10,
    JRD_FIO D_STR_100)
AS
declare variable rep_datetime_begin d_datetime;
declare variable rep_datetime_end d_datetime;
declare variable total_min d_datetime;
declare variable total_max d_datetime;
declare variable jrl_kpi_timezone d_smallint;
begin
  if (rep_date_begin is not null) then
    rep_datetime_begin = rep_date_begin;
  else
    rep_datetime_begin = cast('01.01.1990' as d_datetime);

  if (rep_date_end is not null) then
    rep_datetime_end = rep_date_end + 1;
  else
    rep_datetime_end = current_timestamp;

  total_min = null;
  total_max = null;
  jrs_text = '';
  jrs_new_code = '';
  jrs_new_text = '';
  jrs_caption = '';
  jrs_new_caption = '';

  jrs_is_last = 0;
  for select js_old.jrs_code,
             js_old.jrs_text,
             js_new.jrs_code,
             js_new.jrs_text,
             js_new.jrs_result,
             jrl_old_datetime,
             jrl_new_datetime,
             iif(jrl_new_datetime > :rep_datetime_end, 1, 0),
             jrl_kpi_timezone,
             jrd_fio
  from jira_log jrl
          inner join jira_states js_old on (jrl_old_state = js_old.jrs_id)
          inner join jira_states js_new on (jrl_new_state = js_new.jrs_id)
          left join jira_dev on (jrl_dev_id = jrd_id)
  where jrl_issue_id = :issue_id and
        jrl_old_datetime < :rep_datetime_end and
        jrl_old_datetime <= jrl_new_datetime
  order by jrl_new_datetime, jrl.rdb$db_key
  into jrs_code, jrs_text, jrs_new_code, jrs_new_text, jrs_new_result, min_datetime, max_datetime,
       jrs_is_last, jrl_kpi_timezone, jrd_fio
  do
  begin
    if (jrl_kpi_timezone = 0) then
      gmt_timezone = 'GMT БД';
    else
    if (jrl_kpi_timezone > 0) then
      gmt_timezone = 'GMT+' || jrl_kpi_timezone;
    else
      gmt_timezone = 'GMT' || jrl_kpi_timezone;

    second_total = f_jira_diff(min_datetime, minvalue(max_datetime, rep_datetime_end), jrl_kpi_timezone);
    second_period = f_jira_diff(maxvalue(min_datetime, rep_datetime_begin), minvalue(max_datetime, rep_datetime_end), jrl_kpi_timezone);

    if (second_total is null) then
      second_total = 0;

    if (second_period is null) then
      second_period = 0;

    time_total = f_jira_period(second_total);
    time_period = f_jira_period(second_period);

    jrs_caption = jrs_code || '   / ' || jrs_text;
    jrs_new_caption = jrs_new_code || '   / ' || jrs_new_text;

    suspend;

    if (total_min is null or total_min > min_datetime) then
      total_min = min_datetime;

    if (total_max is null or total_max < max_datetime) then
      total_max = max_datetime;
  end

  if (max_datetime < rep_datetime_end and jrs_new_result = 0) then
  begin
    jrs_is_last = 1;
    jrs_code = jrs_new_code;
    jrs_text = jrs_new_text;
    min_datetime = max_datetime;
    jrs_new_code = '';
    jrs_new_text = '';

    jrs_caption = jrs_code || '   / ' || jrs_text;
    jrs_new_caption = '';

    if (jrs_new_result = 0) then
      max_datetime = rep_datetime_end;


    second_total = f_jira_diff(min_datetime, max_datetime, jrl_kpi_timezone);
    second_period = f_jira_diff(maxvalue(min_datetime, rep_datetime_begin), max_datetime, jrl_kpi_timezone);

    time_total = f_jira_period(second_total);
    time_period = f_jira_period(second_period);

    suspend;
  end
end
