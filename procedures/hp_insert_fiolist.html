CREATE PROCEDURE HP_INSERT_FIOLIST(
    FIO_TXT VARCHAR(100))
RETURNS (
    FL_ID DECIMAL(18,0),
    TXT1 VARCHAR(100),
    TXT2 VARCHAR(100),
    TXT3 VARCHAR(100))
AS
declare variable db_id integer;
declare variable orig1 d_str_100;
declare variable orig2 d_str_100;
declare variable orig3 d_str_100;
declare variable iter d_integer;
begin
  select full_text, txt1, txt2, txt3
  from hp_fio_clear(:fio_txt)
  into :fio_txt, :txt1, :txt2, :txt3;

  if (txt2 <> '') then
  begin
    orig1 = txt1;
    orig2 = txt2;
    orig3 = txt3;
    iter = 1;

    while (iter <= 6 and fl_id is null) do
    begin
      if (iter = 2) then
      begin
        txt1 = orig1;
        txt2 = orig3;
        txt3 = orig2;
      end
      if (iter = 3) then
      begin
        txt1 = orig2;
        txt2 = orig1;
        txt3 = orig3;
      end
      if (iter = 4) then
      begin
        txt1 = orig2;
        txt2 = orig3;
        txt3 = orig1;
      end
      if (iter = 5) then
      begin
        txt1 = orig3;
        txt2 = orig1;
        txt3 = orig2;
      end
      if (iter = 6) then
      begin
        txt1 = orig3;
        txt2 = orig2;
        txt3 = orig1;
      end

      select fl_id
      from fiolist
      where fl_name1 = :txt1 and
            fl_name2 = :txt2 and
            fl_name3 = :txt3
      plan(fiolist index(fiolist_name1, fiolist_name2, fiolist_name3))
      rows 1
      into :fl_id;

      iter = iter + 1;
    end


    if (fl_id is null) then
    begin
      fl_id = gen_id(gen_fiolist_id, 1);

      db_id = f_get_db_id();
      fl_id = db_id * 1000000000 + fl_id;

      insert into fiolist (fl_id, fl_name1, fl_name2, fl_name3)
      values (:fl_id, :orig1, :orig2, :orig3);
    end
  end

  suspend;
end
