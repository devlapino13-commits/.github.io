CREATE PROCEDURE AAREP_CLIENTS_STAT_LOANS(
    REP_DATE_END DATE)
RETURNS (
    CLIENT_IN_GROUP INTEGER,
    CLIENT_KOLICH INTEGER,
    LOAN_KOLICH INTEGER)
AS
begin
  for with client_list
      as (select kc_from_c_id, count(*) as kol
          from knclient, kontstate
          where kc_from_k_id = ks_from_k_id and
                ks_ol_is_delete = 0 and
                cast(ks_date_begin as date) <= :rep_date_end and
                ks_date_end >= :rep_date_end and
                ks_state = 1 and
                kc_amount_approved > 0
          group by kc_from_c_id)
      select kol, count(kol), kol * count(kol)
      from client_list
      group by kol
      into :client_in_group, :client_kolich, :loan_kolich

  do
  begin
    suspend;
  end
end
