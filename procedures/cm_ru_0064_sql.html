CREATE PROCEDURE CM_RU_0064_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable pr_code varchar(100);
declare variable kc_from_c_id id_type;
declare variable c_citizen integer;
declare variable rd_code varchar(255);
declare variable rd_name varchar(255);
declare variable kc_circle integer;
declare variable k_kredline_end date;
declare variable pdt_code varchar(100);
declare variable CDS_DOC_DATE_BEGIN date;
declare variable CDS_DOC_DATE_END date;
declare variable pdt_name varchar(100);
begin
  for select kc_from_c_id, kc_amount_approved, kc_circle, pr_code, k_disb_date
  from knclient, kontract, product
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_from_k_id = k_id
  and k_from_pr_id = pr_id       
  and pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.)%'
  into kc_from_c_id, kc_amount, kc_circle, pr_code, KF_DISB_DATE 
  do
  begin
    select c_name_fio, c_citizen
    from clients
    where c_id = :kc_from_c_id
    into res_info_text, c_citizen;
    select rd_code, rd_name from refdata
    where rd_id = :c_citizen
    into rd_code, rd_name;
    if (pr_code containing 'BARAKAT' and rd_code not in ('Citizen.KG', 'Citizen.UZ', 'Citizen.TJ', 'Citizen.RU')) then
    begin
      res_error_text = 'По выбранному продукту гражданины '||rd_name||' не могут брать.';
      res_error_is_critical = 1;
      suspend;
    end
    
    if (pr_code containing 'BARAKAT' and :rd_code = 'Citizen.KG' and
            not exists(select *
                        from CLIENTDOCS cd, PASSKIND, PASSDOCTYPE
                        where CDS_FROM_C_ID = :kc_from_c_id
                        and cds_from_pk_id = pk_id
                        and pk_from_pdt_id = pdt_id
                        and cds_is_delete = 0
                        and coalesce(cds_doc_date_end, cast('31.12.99' as date))>= dateadd(month, 1, :kf_disb_date)
                        and (pdt_code in ('magration_card', 'registration_rvp', -- Разрешение на временное проживание (РВП)
                                     'registration_vnj', 'job_agreement') -- Вид на жительство (ВНЖ) и труд.договор
                                        or pk_id = 87 ))) then
    begin
      res_error_text = 'Для гражд. КГ не имеется документ миграционного учета.';
      res_error_is_critical = 1;
      suspend;
    end
     
    if (pr_code containing '.BARAKAT.' and rd_code <> 'Citizen.RU' and rd_code <> 'Citizen.KG' and
            not exists(select *
                        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                        where CDS_FROM_C_ID = :kc_from_c_id
                        and cds_from_pk_id = pk_id
                        and pk_from_pdt_id = pdt_id
                        and cds_is_delete = 0
                        and pdt_code in ('work_patent','work_permit')
                                         )) then
    begin
      res_error_text = 'У клиента должно быть патент или разрешения на работу!';
      res_error_is_critical = 1;
      suspend;
    end
    if (pr_code containing '.BARAKAT PLUS.') then
    begin
      if (not exists(select *
                    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                    where CDS_FROM_C_ID = :kc_from_c_id
                    and cds_from_pk_id = pk_id
                    and pk_from_pdt_id = pdt_id
                    and cds_is_delete = 0
                    and pdt_code in ('registration_rvp', -- Разрешение на временное проживание (РВП)
                                     'registration_vnj') -- Вид на жительство (ВНЖ)
                                     )) then
      begin
        res_error_text = 'У клиента должно быть Разрешения на временное проживание или Вид на жительство';
        res_error_is_critical = 1;
        suspend;
      end 
    end
--    Обязательно наличие у клиента Патента или Разрешения на работу
--    Разработан по пункту 5
    for
    select pdt_code,
           min(CDS_DOC_DATE_BEGIN),
           max(CDS_DOC_DATE_END),
           pdt_name
    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
    where CDS_FROM_C_ID = :kc_from_c_id
    and cds_from_pk_id = pk_id
    and pk_from_pdt_id = pdt_id
    and cds_is_delete = 0
    and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
    group by pdt_code, pdt_name
    into pdt_code,
         CDS_DOC_DATE_BEGIN,
         CDS_DOC_DATE_END,
         pdt_name
    do
    begin
      if (pdt_code similar to 'registration_yur|registration_fiz' and
            CDS_DOC_DATE_END < k_kredline_end and :rd_code = 'Citizen.KG' and
            exists (select *
                    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                    where CDS_FROM_C_ID = :kc_from_c_id
                    and cds_from_pk_id = pk_id
                    and pk_from_pdt_id = pdt_id
                    and cds_is_delete = 0
                    and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                    and pdt_code similar to 'registration_yur|registration_fiz'
                    and CDS_DOC_DATE_END >= :k_kredline_end)) then
        CDS_DOC_DATE_END = k_kredline_end +10;
      if (pdt_code containing 'registration_rvp' and :rd_code = 'Citizen.KG' and
            exists (select *
                    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                    where CDS_FROM_C_ID = :kc_from_c_id
                    and cds_from_pk_id = pk_id
                    and pk_from_pdt_id = pdt_id
                    and cds_is_delete = 0
                    and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                    and pdt_code containing 'registration_vnj')) then CDS_DOC_DATE_END = :k_kredline_end + 10;
      if (:rd_code in ('Citizen.TJ', 'Citizen.UZ')) then
      begin
        if (pr_code containing '.BARAKAT.') then
        begin
          if (pdt_code in ('work_permit', -- Разрешение на работу
                           'work_patent')  -- Патент) then
                and not exists (select *
                    from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                    where CDS_FROM_C_ID = :kc_from_c_id
                    and cds_from_pk_id = pk_id
                    and pk_from_pdt_id = pdt_id
                    and cds_is_delete = 0
                    and pdt_code similar to 'certificate_approving_doc|application_citizenship_rassian_lang_speaker')) then
          begin 
              if (dateadd(month, 2, :kf_disb_date) > CDS_DOC_DATE_END and pdt_code in ('work_permit')) then  -- Разрешение на работу
              begin
                res_error_text = 'До конца срока действия разрешения на работу должно оставаться не менее 2месяцев со дня получения микрозайма';
                res_error_is_critical = 1;
                suspend;
              end 
              if (dateadd(month, 11, CDS_DOC_DATE_BEGIN) < kf_disb_date and
                   pdt_code in ('work_patent')and
                        not exists (select *
                                from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                                where CDS_FROM_C_ID = :kc_from_c_id
                                and cds_from_pk_id = pk_id
                                and pk_from_pdt_id = pdt_id
                                and cds_is_delete = 0
                                and pdt_code containing 'work_patent'
                                and CDS_DOC_DATE_END >= :kf_disb_date+30))  then
              begin
                res_error_text = 'Со дня выдачи патента должно пройти не более 11 месяцев на момент получения микрозайма';
                res_error_is_critical = 1;
                suspend;
              end
          end 
        end else
        if (pr_code containing '.BARAKAT PLUS.') then
        begin
          if (pdt_code in ('registration_rvp')-- Разрешение на временное проживание (РВП)
                and not exists( select *
                                from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                                where CDS_FROM_C_ID = :kc_from_c_id
                                and cds_from_pk_id = pk_id
                                and pk_from_pdt_id = pdt_id
                                and cds_is_delete = 0
                                and CDS_DOC_DATE_END > :CDS_DOC_DATE_END
                                and pdt_code containing 'registration_vnj')) then
          begin
            if (not exists(select *
                        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                        where CDS_FROM_C_ID = :kc_from_c_id
                        and cds_from_pk_id = pk_id
                        and pk_from_pdt_id = pdt_id
                        and cds_is_delete = 0
                        and  pdt_code containing 'certificate_approving_doc') and
                        dateadd(month, 3, :kf_disb_date) > CDS_DOC_DATE_END) then
            begin
              res_error_text = 'Для гражд. ТЖ и УЗ '||pdt_name||' должны быть действительны не менее 3 месяцев после получения микрозайма';
              res_error_is_critical = 1;
              suspend;
            end
          end
          if (pdt_code in ('registration_vnj') -- Вид на жительство (ВНЖ)
              ) then
          begin
            if (not exists(select *
                        from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                        where CDS_FROM_C_ID = :kc_from_c_id
                        and cds_from_pk_id = pk_id
                        and pk_from_pdt_id = pdt_id
                        and cds_is_delete = 0
                        and  pdt_code containing 'certificate_approving_doc')
                and dateadd(month, 3, :kf_disb_date) > CDS_DOC_DATE_END) then
            begin
              res_error_text = 'Для гражд. ТЖ и УЗ '||pdt_name||' должны быть действительны не менее 3 месяцев после получения микрозайма';
              res_error_is_critical = 1;
              suspend;
            end
          end
        end
      end
    end
  end
end
