CREATE PROCEDURE HP_ACTIONS_LIST_IN_ORDER(
    UN_ID DECIMAL(18,0),
    ONLY_OWN INTEGER)
RETURNS (
    ACL_ID DECIMAL(18,0),
    ACL_OBJECT_KIND INTEGER,
    ACL_OBJECT_ID DECIMAL(18,0),
    ACL_OBJECT_NAME VARCHAR(100),
    ACL_STATUS INTEGER,
    RD_ACTION_ID INTEGER,
    RD_ACTION_CODE VARCHAR(100),
    RD_ACTION_NAME VARCHAR(255),
    RD_SYSTEM_CODE VARCHAR(255))
AS
declare variable c_id decimal(18,0);
begin
-- Процедура возвращает список задач в очереди

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  for select aco_id, aco_object_kind, aco_object_id, aco_status, rd_id, rd_code, rd_name, rd_str_id
      from refdata, actionorder
      where aco_from_rd_id = rd_id and
            rd_main_code = 'ActionGroup' and
            rd_str_id in ('Anketa', 'CheckAnketa') and
            rd_code <> rd_main_code and
            (aco_respons_c_id = :c_id or
            :only_own <> 1) and
            aco_object_kind = 0
      order by aco_id
      into :acl_id, acl_object_kind, acl_object_id, acl_status, rd_action_id, rd_action_code, rd_action_name,
           rd_system_code
  do
  begin
    acl_object_name = '';
    select c_name_fio
    from clients
    where c_id = :acl_object_id
    into :acl_object_name;
    suspend;
  end
end
