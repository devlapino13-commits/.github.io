CREATE PROCEDURE MC_LAST_READ_UPDATE(
    MCL_ID D_ID)
AS
declare variable c_id d_id;
begin
  /*
   процедура отвечает за обновление даты/времени последного просмотра сообщений пользователем МП
   А так же за обнуление числа непрочтенных push-уведомлений
  */
  select ml.mcl_from_c_id
  from mc_login ml
  where ml.mcl_id = :mcl_id
  into :c_id;

  if (coalesce(mcl_id, 0) = 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  update mc_login ml
  set ml.mcl_message_last_read = current_timestamp
  where ml.mcl_id = :mcl_id;

  update mc_message_counter mco
  set mco.mco_message_qty = 0
  where mco.mco_object_kind = f_objkind('mc_login', 'id')
  and mco.mco_object_id = :mcl_id;

  if (c_id is not null) then
  begin
    update mc_message_counter mco
    set mco.mco_message_qty = 0
    where mco.mco_object_kind = f_objkind('clients', 'id')
    and mco.mco_object_id = :c_id;
  end
end
