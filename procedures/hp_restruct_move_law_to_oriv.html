CREATE PROCEDURE HP_RESTRUCT_MOVE_LAW_TO_ORIV(
    REP_DATE D_DATE,
    PROCESS_MODE D_SMALLINT)
AS
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable rbc_id d_id;
declare variable res_k_id d_id;
begin
    /*
      RFCKZ-705. Процедура проводит автоматическую передачу контрактов с баланса Юр. отдела
      на баланс ОРиВ

      process_mode:
        1 - обработка всех записей на rep_date (стандартная обработка)
        2 - обработка только ошибочных записей
    */

    if (process_mode not in (1, 2)) then
        exception e_message 'Некорректный process_mode';

    for
        select rbc.rbc_id, rbc.rbc_from_k_id
        from restruct_block_cancel rbc
        where rbc.rbc_date_begin = :rep_date and
              rbc.rbc_moved_law_to_oriv = 0 and
              (
                (:process_mode = 1 and coalesce(rbc.rbc_move_law_to_oriv_error, '') = '') or
                (:process_mode = 2 and coalesce(rbc.rbc_move_law_to_oriv_error, '') <> '')
              )
        into :rbc_id, :k_id
    do
    begin
        begin
            k_kontract_no = (select k.k_kontract_no from kontract k where k.k_id = :k_id);
        
            -- Передаем кредит из юр. отдела в ОРиВ
            select k_id from insert_sb_akt(:k_kontract_no, :rep_date, 3, 2, 1000000000000)
            into :res_k_id;
    
            -- Перемещение прошло успешно, обновляем запись
            update restruct_block_cancel rbc
            set rbc.rbc_moved_law_to_oriv = 1,
                rbc.rbc_move_law_to_oriv_datetime = current_timestamp,
                rbc.rbc_move_law_to_oriv_error = null
            where rbc.rbc_id = :rbc_id;
    
            rbc_id = null;
            k_id = null;
    
            when any do
            begin
                update restruct_block_cancel rbc
                set rbc.rbc_move_law_to_oriv_error = left(rdb$error(message), 255)
                where rbc.rbc_id = :rbc_id;

                execute procedure hp_send_quick_message('email', 't.mavlyanov@mbulak.kg',
                    'Ошибка при переводе кредита Юр. отдел -> ОРиВ',
                    f_format('Возникла ошибка в контракте: %1, rbc_id: %2, текст ошибки: %3', :k_id, :rbc_id, left(rdb$error(message), 2048)),
                    '', 7, 1000000000000
                );

                rbc_id = null;
                k_id = null;
            end
        end
    end
end
