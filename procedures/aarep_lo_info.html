CREATE PROCEDURE AAREP_LO_INFO(
    UN_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    RES_TEXT VARCHAR(255),
    RES_AMOUNT_1 DECIMAL(18,4),
    RES_AMOUNT_2 DECIMAL(18,4),
    RES_STR_1 VARCHAR(100),
    RES_STR_2 VARCHAR(100),
    RES_ROUND_1 INTEGER,
    RES_ROUND_2 INTEGER)
AS
declare variable c_id decimal(18,0);
declare variable reg_id integer;
declare variable loan_count decimal(18,4) = 0;
declare variable loan_balance decimal(18,4) = 0;
declare variable loan_count7 decimal(18,4) = 0;
declare variable loan_balance7 decimal(18,4) = 0;
declare variable loan_count30 decimal(18,4) = 0;
declare variable loan_balance30 decimal(18,4) = 0;
declare variable dd_begin date;
declare variable res_money_str varchar(3);
declare variable cm_database_id integer;
declare variable loan_tmp decimal(18,4) = 0;
declare variable rep_date_begin date;
declare variable loan_begin_count integer;
declare variable loan_begin_balance decimal(18,4) = 0;
declare variable loan_count_end integer;
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;
  if (c_id is null) then
    exit;

  rep_date_begin = :rep_date_end - extract(day from :rep_date_end);

  :cm_database_id = f_get_db_id();

  res_money_str = (case cm_database_id
                     when 81001 then 'р.'
                     when 41701 then 'с.'
                     when 94401 then 'AZN'
                   end);

  res_round_1 = 0;
  res_round_2 = 0;

  select ps_from_reg_id
  from hr_pstatus, joblist, jobkind
  where ps_from_c_id = :c_id and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_status = 1 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id and
        jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
        ps_is_working = 1
  into :reg_id;
  if (reg_id is null) then
    reg_id = 0;
  if (reg_id <> 0) then
    c_id = reg_id;

  res_text = 'Активные кредиты';
  res_amount_1 = 0;
  res_amount_2 = 0;
  select sum(k_kolich_client_approved), sum(oks_balance),
         sum(iif(oks_delinq_date < (:rep_date_end - 7), k_kolich_client_approved, 0.0)),
         sum(iif(oks_delinq_date < (:rep_date_end - 7), oks_balance, 0.0)),
         sum(iif(oks_delinq_date < (:rep_date_end - 30), k_kolich_client_approved, 0.0)),
         sum(iif(oks_delinq_date < (:rep_date_end - 30), oks_balance, 0.0))
  from oksummary, kontract
  where (oks_kc_id = :c_id or
        oks_region_id = :c_id) and
        oks_date_begin <= :rep_date_end and
        oks_date_end >= :rep_date_end and
        oks_balance > 0 and
        oks_from_k_id = k_id and
        oks_kc_id > 1000000000003
        and oks_close_state = 0
  into :loan_count, :loan_balance, loan_count7, loan_balance7, loan_count30, loan_balance30;

  select sum(k_kolich_client_approved), sum(oks_balance)
  from oksummary, kontract
  where (oks_kc_id = :c_id or
        oks_region_id = :c_id) and
        oks_date_begin <= :rep_date_begin and
        oks_date_end >= :rep_date_begin and
        oks_balance > 0 and
        oks_from_k_id = k_id and
        oks_kc_id > 1000000000003
        and oks_close_state = 0
  into :loan_begin_count, :loan_begin_balance;

  if (loan_count is null) then
    loan_count = 0;
  if (loan_balance is null) then
    loan_balance = 0;
  if (loan_begin_count is null) then
    loan_begin_count = 0;
  if (loan_begin_balance is null) then
    loan_begin_balance = 0;
  if (loan_count7 is null) then
    loan_count7 = 0;
  if (loan_balance7 is null) then
    loan_balance7 = 0;
  if (loan_count30 is null) then
    loan_count30 = 0;
  if (loan_balance30 is null) then
    loan_balance30 = 0;
  loan_count_end = loan_count;

  res_amount_1 = loan_count;
  res_amount_2 = loan_balance;
  res_str_1 = '';
  res_str_2 = res_money_str;
  res_round_1 = 0;
  res_round_2 = 0;
  suspend;
  res_amount_1 = 0;
  res_amount_2 = 0;

  res_text = 'Просроч.кредиты > 7 дней';
  res_amount_1 = loan_count7;
  if (loan_count <> 0) then
    res_amount_2 = round(loan_count7 * 100.0 / loan_count, 1);
  res_str_1 = '';
  res_str_2 = '%';
  res_round_1 = 0;
  res_round_2 = 1;
  suspend;
  res_amount_1 = 0;
  res_amount_2 = 0;

  res_text = 'Баланс просроч. > 7 дней';
  res_amount_1 = loan_balance7;
  if (loan_balance <> 0) then
    res_amount_2 = round(loan_balance7 * 100.0 / loan_balance, 1);
  res_str_1 = res_money_str;
  res_str_2 = '%';
  res_round_1 = 0;
  res_round_2 = 1;
  suspend;
  res_amount_1 = 0;
  res_amount_2 = 0;

  res_text = 'Просроч.кредиты > 30 дней';
  res_amount_1 = loan_count30;
  if (loan_count <> 0) then
    res_amount_2 = round(loan_count30 * 100.0 / loan_count, 1);
  res_str_1 = '';
  res_str_2 = '%';
  res_round_1 = 0;
  res_round_2 = 1;
  suspend;
  res_amount_1 = 0;
  res_amount_2 = 0;

  res_text = 'Баланс просроч. > 30 дней';
  res_amount_1 = loan_balance30;
  if (loan_balance <> 0) then
    res_amount_2 = round(loan_balance30 * 100.0 / loan_balance, 1);
  res_str_1 = res_money_str;
  res_str_2 = '%';
  res_round_1 = 0;
  res_round_2 = 1;
  suspend;
  res_amount_1 = 0;
  res_amount_2 = 0;

  dd_begin = dateadd(day, -(extract(day from :rep_date_end) - 1), :rep_date_end);
  select count(*), sum(kc_amount_approved), sum(iif(coalesce(kc_circle, 0) <= 1, 1.0, 0.0)),
         sum(iif(coalesce(kc_circle, 0) <= 1, kc_amount_approved, 0.0))
  from knclient, kontstate
  where ks_from_k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        ks_ol_is_delete = 0 and
        ks_state = 1 and
        (ks_kc_id = :c_id or
        ks_region_id = :c_id) and
        ks_date_begin between :dd_begin and :rep_date_end
  into :res_amount_1, :res_amount_2, loan_count, loan_tmp;
  if (res_amount_1 is null) then
    res_amount_1 = 0;
  if (res_amount_2 is null) then
    res_amount_2 = 0;
  if (loan_count is null) then
    loan_count = 0;
  if (loan_tmp is null) then
    loan_tmp = 0;

  res_text = 'Всего выдач с начала месяца';
  res_str_1 = '';
  res_str_2 = res_money_str;
  res_round_1 = 0;
  res_round_2 = 0;
  suspend;
  res_text = 'Выдачи новым клиентам';
  res_amount_1 = loan_count;
  res_amount_2 = loan_tmp;
  res_str_1 = '';
  res_str_2 = res_money_str;
  res_round_1 = 0;
  res_round_2 = 0;
  suspend;
  res_amount_1 = 0;
  res_amount_2 = 0;

  res_text = 'Активные на начало месяца';
  res_amount_1 = loan_begin_count;
  res_amount_2 = loan_begin_balance;
  res_str_1 = '';
  res_str_2 = res_money_str;
  res_round_1 = 0;
  res_round_2 = 0;
  suspend;

  res_text = 'Прирост с начала месяца';
  res_amount_1 = loan_count_end - loan_begin_count;
  res_amount_2 = loan_balance - loan_begin_balance;
  res_str_1 = '';
  res_str_2 = res_money_str;
  res_round_1 = 0;
  res_round_2 = 0;
  suspend;

  select sum(oks_balance)
  from oksummary
  where (oks_kc_id = :c_id or
        oks_region_id = :c_id) and
        oks_date_begin <= (:dd_begin - 1) and
        oks_date_end >= (:dd_begin - 1) and
        oks_delinq_date < (:dd_begin - 8) and
        oks_balance > 0 and
        oks_kc_id > 1000000000003
        and oks_close_state = 0
  into :loan_balance30; -- Просрочка 7 дн.
  if (loan_balance30 is null) then
    loan_balance30 = 0;
  if (loan_balance30 < loan_balance7) then
    res_text = 'Баланс PAR7 вырос';
  if (loan_balance30 > loan_balance7) then
    res_text = 'Баланс PAR7 снизился';
  if (loan_balance30 = loan_balance7) then
    res_text = 'Баланс PAR7 не изменился';
  res_amount_1 = loan_balance7 - loan_balance30;
  if (loan_balance <> 0) then
    res_amount_2 = res_amount_1 * 100.0 / loan_balance;
  res_str_1 = res_money_str;
  res_str_2 = '%';
  res_round_1 = 0;
  res_round_2 = 1;
  suspend;

end
