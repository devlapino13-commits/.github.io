CREATE PROCEDURE HP_IPF_INSURANCE_OPERBOOK(
    EDL_ID D_ID,
    K_ID D_ID)
AS
declare variable kf_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable pr_code d_str_100;
declare variable office_id d_integer;
declare variable office_name d_str_100;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable ag_id d_integer;
declare variable insurance_c_id d_id;
declare variable insurance_al_account d_str_20;
declare variable ik_commission d_currency;
declare variable ik_mbulak_share d_currency;
declare variable ol_id d_id;
declare variable ol_date d_date;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ob_summa d_currency;
declare variable ol_doc_kind d_integer;
declare variable ol_doc_client_name d_str_255;
declare variable ol_doc_no d_integer;
declare variable ol_is_client d_integer;
declare variable ol_status d_smallint;
declare variable ol_memo d_str_long;
declare variable ob_detail d_str_long;
declare variable ob_client_dt d_id;
declare variable ob_client_kt d_id;
declare variable ob_account_kt d_str_20;
declare variable ob_account_dt d_str_20;
declare variable acc_short_dt d_str_20;
declare variable acc_short_kt d_str_20;
declare variable acc_not_exist d_str_100;
begin
  -- создание бух. проводок по страхованию ИПФ займа (RFCKG-5390):
  -- 1) признание счета к получению от клиента и задолженности по сумме страхового полиса при подписании заявки на страхование
  -- 2) признание комиссии МБулак

  ol_date = current_date; -- дата операции

  select k_kontract_no,
         k_disb_date,
         k_from_kf_id,
         pr_code
  from kontract
  inner join product on pr_id = k_from_pr_id
  where k_id = :k_id
  into k_kontract_no,
       k_disb_date,
       kf_id,
       pr_code;

  if (pr_code not containing '.ipf.' or
      pr_code containing '.ignore_insurance.') then
    exception e_message 'Займ не подлежит страхованию';

  select kal_new_reg_id
  from knaktlist
  where is_delete = 0 and
        kal_from_kf_id = :kf_id and
        :ol_date between kal_date and kal_date_end
  into office_id; -- офис контракта

  if (office_id is null) then
    exception e_message 'Офис займа не указан';

  select ik_from_ag_id,
         ik_commission,
         ik_mbulak_share
  from insurance_kontract
  where ik_from_k_id = :k_id
  into ag_id, -- id договора со страховой компанией
       ik_commission, -- сумма страховой премии
       ik_mbulak_share; -- сумма комиссии МБулак

  if (ag_id is null or
      ik_commission is null or
      ik_mbulak_share is null) then
    exception e_message 'Договор со страховой компанией не найден';

  select al_account,
         ag_object_id
  from acc_list
  inner join agreements on ag_from_al_id = al_id
  where ag_id = :ag_id
  into insurance_al_account, -- короткий код счета (например 21122/311)
       insurance_c_id; -- c_id страховой компании

  select c_id,
         c_name_fio
  from clients
  inner join knclient on kc_from_c_id = c_id
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into c_id,
       c_name_fio; -- клиент

  select num
  from last_doc_number(current_date, 9, :office_id)
  into ol_doc_no;

  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :office_id
  into office_name;

  acc_not_exist = 'Для офиса ' || office_name || ' не определен счет ';

  ol_doc_client_name = 'Оплата за страхование финансирования';
  ol_memo = ol_doc_client_name || ' / ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date) || ' / ' || c_name_fio;
  ol_oper_type = 2; -- бухгалтерская операция
  ol_oper_subtype = 11; -- мемориальный ордер
  ol_doc_kind = 9; -- мемориальный ордер
  ol_is_client = 1; -- операция по клиенту
  ol_status = 1; -- статус операции утверджен

  insert into operlist (ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no, ol_doc_no_str, ol_document_client, ol_is_client, ol_memo, ol_doc_client_name, ol_user_id, ol_date, ol_status)
  values (:office_id, :office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :ol_doc_no, :k_id, :c_id, :ol_is_client, :ol_memo, :ol_doc_client_name, 1000000000000, :ol_date, :ol_status)
  returning (ol_id)
  into ol_id;

  --***-- 1.1
  ob_detail = 'Признание счета к получению от клиента и задолженности по сумме страхового полиса';
  ob_summa = ik_commission;

  ob_client_dt = c_id;
  ob_client_kt = insurance_c_id;

  acc_short_dt = '11531/311';
  acc_short_kt = insurance_al_account;

  ob_account_dt = '';
  select al_id
  from acc_list
  where is_delete = 0 and
        al_is_active = 1 and
        al_currency = '417' and
        al_from_reg_id = :office_id and
        al_account = :acc_short_dt
  into ob_account_dt;

  if (coalesce(ob_account_dt, '') = '') then
    exception e_message acc_not_exist || acc_short_dt;

  ob_account_kt = '';
  select al_id
  from acc_list
  where is_delete = 0 and
        al_is_active = 1 and
        al_currency = '417' and
        al_from_reg_id = :office_id and
        al_account = :acc_short_kt
  into ob_account_kt;

  if (coalesce(ob_account_kt, '') = '') then
    exception e_message acc_not_exist || acc_short_kt;

  insert into operbook (ob_from_ol_id, ob_client_dt, ob_client_kt, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:ol_id, :ob_client_dt, :ob_client_kt, :ob_account_dt, :ob_account_kt, :ob_summa, :ob_summa, :ol_date, :office_id, :ob_detail);

  --***-- 1.2
  ob_detail = 'Признание комиссии МБулак';
  ob_summa = ik_mbulak_share;

  ob_client_dt = insurance_c_id;
  ob_client_kt = c_id;

  acc_short_dt = insurance_al_account;
  acc_short_kt = '21114/201';

  ob_account_dt = '';
  select al_id
  from acc_list
  where is_delete = 0 and
        al_is_active = 1 and
        al_currency = '417' and
        al_from_reg_id = :office_id and
        al_account = :acc_short_dt
  into ob_account_dt;

  if (coalesce(ob_account_dt, '') = '') then
    exception e_message acc_not_exist || acc_short_dt;

  ob_account_kt = '';
  select al_id
  from acc_list
  where is_delete = 0 and
        al_is_active = 1 and
        al_currency = '417' and
        al_from_reg_id = :office_id and
        al_account = :acc_short_kt
  into ob_account_kt;

  if (coalesce(ob_account_kt, '') = '') then
    exception e_message acc_not_exist || acc_short_kt;

  insert into operbook (ob_from_ol_id, ob_client_dt, ob_client_kt, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:ol_id, :ob_client_dt, :ob_client_kt, :ob_account_dt, :ob_account_kt, :ob_summa, :ob_summa, :ol_date, :office_id, :ob_detail);

  -- сохранение связки эл. документ + бух. операция
  if (not exists(select *
                 from ipf_insurance_operbook where iio_edl_id = :edl_id)) then
    insert into ipf_insurance_operbook (iio_edl_id, iio_ol_id)
    values (:edl_id, :ol_id);
end
