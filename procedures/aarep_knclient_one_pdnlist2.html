          begin
            credpartnertype_str = 'ССП';
            credtype_str = 'ССП';
            credactive_str = 'ССП';
            credcategory_str = 'ССП';
            credconsumer_str = 'ССП';
            credratio_str = 'ССП';
            credpurpose_str = 'ССП';
            zalog_str = 'ССП';
            creddate = null;
            credenddate = null;
            credenddatefact = null;
            credupdate = null;
            credsumdebt = null;
            credsumoverdue = null;
            creddayoverdue = null;
            credfullcost = null;
            credsum = null;
            poruch_str = 'ССП';
            zalog_str = 'ССП';
            calc_koef_period = null;
            calc_period_month = null;
            calc_is_credit_card = null;

            kc_anketa_income = null;
