CREATE PROCEDURE CB24_RFC1302_RESCHEDULED_LOANS(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REP_LINE_ID D_INTEGER,
    REP_KC_UUID D_STR_CODE,
    REP_KONT_NUMBER_DATE D_STR_CODE,
    REP_KONT_OFFICE D_STR_100,
    REP_CLIENT_FIO D_STR_100,
    REP_KONT_AMOUNT D_CURRENCY,
    REP_RESH_DATE D_DATE,
    REP_RESH_REASON D_STR_255,
    REP_KC_ID D_ID,
    REP_KONTRACT_NO D_STR_20,
    REP_DISB_DATE D_DATE)
AS
declare variable k_id d_id;
declare variable ol_id d_id;
begin
  rep_line_id = 0;

  for select distinct kc_id, k_kontract_no, k_disb_date,
             reg_code || ' ' || reg_name, c_name_fio, kc_amount_approved, or_ol_date,
             or_from_k_id, or_from_ol_id
  from operrest
       inner join region_list(:rep_region_id, 1, 3, 1) on (or_from_reg_id = id_list)
       inner join kontract on (or_from_k_id = k_id)
       inner join knclient on (kc_from_k_id = k_id and kc_amount_approved > 0)
       inner join clients on (kc_from_c_id = c_id)
       left join region on (or_from_reg_id = reg_unicode)
  where or_ol_date between :rep_date_begin and :rep_date_end and
        or_ol_is_active = 1
  union
  select kc_id, k_kontract_no, k_disb_date,
         reg_code || ' ' || reg_name, c_name_fio, kc_amount_approved, ol_date,
         kd_from_k_id, kd_from_ol_id
  from kontdata
       inner join kontract on (kd_from_k_id = k_id)
       inner join operlist ol on (k_stopped_perc_ol_id = ol.ol_id and ol.is_delete = 0)
       inner join region_list(:rep_region_id, 1, 3, 1) on (ol.ol_from_reg_id = id_list)
       inner join knclient on (kc_from_k_id = k_id and kc_amount_approved > 0)
       inner join clients on (kc_from_c_id = c_id)
       inner join region on (ol_from_reg_id = reg_unicode)
  where kd_ol_date between :rep_date_begin and :rep_date_end and
        kd_ol_is_delete = 0 and
        kd_is_primary = 0 and
        kd_from_ol_id is not null
--  union
--  select kc_id, k_kontract_no, k_disb_date,
--         reg_code || ' ' || reg_name, c_name_fio, kc_amount_approved, ol_date,
--         ksp_from_k_id, k_stopped_perc_ol_id
--  from kontstopperc
--       inner join kontract on (ksp_from_k_id = k_id)
--       inner join operlist ol on (k_stopped_perc_ol_id = ol.ol_id and ol.is_delete = 0)
--       inner join region_list(:rep_region_id, 1, 3, 1) on (ol.ol_from_reg_id = id_list)
--       inner join knclient on (kc_from_k_id = k_id and kc_amount_approved > 0)
--       inner join clients on (kc_from_c_id = c_id)
--       inner join region on (ol_from_reg_id = reg_unicode)
--  where ksp_date between :rep_date_begin and :rep_date_end
  into rep_kc_id, rep_kontract_no, rep_disb_date,
       rep_kont_office, rep_client_fio, rep_kont_amount, rep_resh_date,
       k_id, ol_id
  do
  begin
    rep_line_id = rep_line_id + 1;

    rep_kont_number_date = rep_kontract_no || ' от ' || f_datetostr(rep_disb_date);
    rep_kc_uuid = f_uuid_cbr(f_uuid('knclient', rep_kc_id));

    rep_resh_reason = '';
    if (exists(select * from opervac
               where ov_from_k_id = :k_id and
                     ov_from_ol_id = :ol_id)) then
      rep_resh_reason = 'Кредитные каникулы';
    else
    if (exists(select * from operrest
               where or_from_k_id = :k_id and
                     or_from_ol_id = :ol_id)) then
      rep_resh_reason = 'Реструктуризация';
    else
    if (exists(select * from kontdata, operlist
               where kd_from_k_id = :k_id and
                     kd_from_ol_id = :ol_id and
                     kd_from_ol_id = ol_id and
                     not(ol_oper_type = 1 and ol_oper_subtype in (1, 12)))) then
      rep_resh_reason = 'Изменения %% ставки';

    suspend;
  end
end
