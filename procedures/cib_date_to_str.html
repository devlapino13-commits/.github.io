CREATE PROCEDURE CIB_DATE_TO_STR(
    DATE_FOR_CONVERT DATE)
RETURNS (
    RESULT_DATE_STR VARCHAR(10))
AS
begin
  if (date_for_convert is null) then
    result_date_str = '';
  else
    result_date_str = replace(extract(year from date_for_convert) || '-' || iif(extract(month from date_for_convert) < 10, '0', '') || extract(month from date_for_convert) || '-' || iif(extract(day from date_for_convert) < 10, '0', '') || extract(day from date_for_convert), ' ', '');
  suspend;
end
