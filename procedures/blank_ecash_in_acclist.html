CREATE PROCEDURE BLANK_ECASH_IN_ACCLIST(
    OL DECIMAL(18,0))
RETURNS (
    BL_SUMMA D_CURRENCY,
    BL_CORR_ACCOUNT D_STR_20,
    BL_BALANCE_ID D_STR_20,
    BL_REGION_CODE D_STR_CODE,
    BL_ACCOUNT_NO D_INTEGER)
AS
declare variable cash_acc d_str_20;
declare variable line d_integer;
declare variable k_id d_id;
declare variable opc_from_ol_id d_id;
declare variable opc_corr_reg_id d_integer;
declare variable opc_office_id d_integer;
declare variable ol_id d_id;
declare variable opc_sign d_smallint;
declare variable in_opc_id d_integer;
begin
  select opc_sign
  from opercash
  where opc_id = :ol
  into opc_sign;

  if (opc_sign = -1) then
  begin
    -- готовим счета приходного ордера, но входящий ol (opc_id) расходника,
    -- ищем opc_id приходника, операция может быть дочерней расходной операции
    -- например: расходник - выдача займа, приходник - прием комиссии за кассовое обслуживание (RFCKG-4162)

    select opc_id
    from opercash
    where opc_sign = 1 and -- приходный ордер
          opc_parent_id = :ol
    order by opc_is_delete, opc_status desc
    rows 1
    into in_opc_id;

    if (row_count = 0) then
      exception e_default 'in_opc_id not found';

    ol = in_opc_id; -- меняем opc_id расходника на opc_id приходника
  end
  
  select opc_amount, opc_corr_al_id, opc_from_k_id, opc_from_ol_id, reg_code,
         opc_corr_reg_id, opc_office_id, iif(opc_is_ipf = 1, csp_ipf_al_id, csp_cls_al_id)
  from opercash, region, cashplace
  where opc_id = :ol and
        opc_office_id = reg_unicode and
        opc_from_csp_id = csp_id
  into bl_summa, bl_corr_account, k_id, opc_from_ol_id, bl_region_code,
       opc_corr_reg_id, opc_office_id, cash_acc;

  bl_balance_id = '';
  bl_account_no = null;
  line = 0;

  if (bl_corr_account is not null) then
  begin
    select al_from_bg_id, al_number
    from acc_list
    where al_id = :bl_corr_account
    into bl_balance_id, bl_account_no;

    suspend;
    line = 1;
  end else
  begin
    if (opc_from_ol_id is null) then
    begin
      if (bl_corr_account is not null) then
      begin
        select al_from_bg_id, al_number
        from acc_list
        where al_id = :bl_corr_account
        into bl_balance_id, bl_account_no;
      end else
      if (k_id is not null) then
      begin
        select k_kontract_no
        from kontract
        where k_id = :k_id
        into bl_corr_account;
      end
    
      suspend;
      line = 1;
    end else
    begin
      if (k_id is not null and opc_corr_reg_id <> opc_office_id) then
      begin
        select ol_id
        from operlist
        where ol_parent_id = :opc_from_ol_id and
              ol_from_reg_id = :opc_corr_reg_id
        order by ol_id desc
        rows 1
        into ol_id;
      end
  
      if (ol_id is null and opc_corr_reg_id <> opc_office_id) then
      begin
        select ol_id
        from operlist, operbook
        where ol_parent_id = :opc_from_ol_id and
              ol_from_reg_id = :opc_corr_reg_id and
              ol_id > :opc_from_ol_id and
              ob_from_ol_id = ol_id and
              ob_summa = :bl_summa
        rows 1
        into ol_id;
      end
  
      if (ol_id is not null) then
      begin
        for select ob_account_kt, al_from_bg_id, al_number, ob_summa
        from operbook, acc_list
        where ob_from_ol_id = :ol_id and
              ob_account_kt = al_id
        into bl_corr_account, bl_balance_id, bl_account_no, bl_summa
        do
        begin
          suspend;
          line = line + 1;
        end
      end
  
      if (line = 0) then
      begin
        for select ob_account_kt, al_from_bg_id, al_number, ob_summa
        from operbook, acc_list
        where ob_from_ol_id = :opc_from_ol_id and
              ob_account_dt = :cash_acc and
              ob_account_kt = al_id
        into bl_corr_account, bl_balance_id, bl_account_no, bl_summa
        do
        begin
          suspend;
          line = line + 1;
        end
      end
    end
  end
  
  bl_summa = null;
  bl_corr_account = '';
  bl_balance_id = '';
  bl_account_no = null;
  bl_region_code = '';

  if (line = 1) then
    suspend;

  if (line <= 2) then
    suspend;
end
