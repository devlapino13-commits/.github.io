CREATE PROCEDURE HP_BLANK_PRODUCTS(
    BLN_ID INTEGER)
RETURNS (
    PRTL_ID INTEGER,
    PRTL_FROM_PRT_ID INTEGER,
    PRTL_FROM_PR_ID INTEGER,
    PRTL_DATE_BEGIN DATE,
    PRTL_DATE_END DATE,
    PRTL_FROM_UN_ID DECIMAL(18,0),
    PRTL_DATETIME TIMESTAMP,
    PRT_NAME VARCHAR(255),
    PR_NAME VARCHAR(255),
    PR_MEMO VARCHAR(255),
    PR_COMMON_NAME VARCHAR(255),
    PR_CODE VARCHAR(255),
    UN_NAME VARCHAR(100))
AS
begin
  for select prt_id, prt_name
      from prodtype, prodtypeblank
      where prt_id = prtb_from_prt_id and
            prtb_from_bln_id = :bln_id
      into :prtl_from_prt_id, prt_name
  do
  begin
    for select prtl_id, prtl_from_pr_id, prtl_date_begin, prtl_date_end, prtl_from_un_id, prtl_datetime, pr_name,
               pr_memo, pr_common_name, pr_code, un_name
        from prodtypelink, product, username
        where prtl_from_pr_id = pr_id and
              (prtl_date_begin <= current_date or
              prtl_date_begin is null) and
              (prtl_date_end > current_date or
              prtl_date_end is null) and
              pr_date_begin <= current_date and
              pr_date_end > current_date and
              prtl_from_prt_id = :prtl_from_prt_id and
              prtl_from_un_id = un_id
        into prtl_id, prtl_from_pr_id, prtl_date_begin, prtl_date_end, prtl_from_un_id, prtl_datetime, pr_name, pr_memo,
             pr_common_name, pr_code, un_name
    do
    begin
      suspend;
    end
  end
end
