CREATE PROCEDURE HP_DO_RESCHEDULE_AUTO(
    K_ID D_ID,
    REP_DATE D_DATE,
    ACTION_KIND D_STR_CODE,
    ADD_PARAMS D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable add_month_count   d_integer;
declare variable add_month_first   d_integer;
declare variable new_annu          d_currency;
declare variable new_pay_day       d_integer;
declare variable perc_lgota_value  d_percent;
declare variable grace_os_list     d_str_long;
declare variable last_sh_date      d_date;
declare variable sh_rec_id         d_integer;
declare variable sh_date           d_date;
declare variable sh_month          d_integer;
declare variable sh_year           d_integer;
declare variable pay_day           d_integer;
declare variable new_sh_date       d_date;
declare variable sh_id             d_integer;
declare variable first_perc_kind   d_smallint;
declare variable curr_period_count d_integer;
declare variable pr_percent_method d_integer;
declare variable is_debug          d_smallint;
declare variable cml_id            d_id;
declare variable first_date        d_date;
declare variable last_rec_id       d_integer;
declare variable month_no          d_integer;
declare variable date_prv          d_date;
declare variable db_id             d_integer;
declare variable stop_perc         boolean;
begin
  res_is_error = 0;
  res_text = '';

  if (coalesce(k_id, 0) = 0) then
    exception e_default 'Input parameter K_ID is null';

  if (action_kind is null) then
    exception e_default 'action_kind is null';

  if (action_kind <> 'auto') then
    exception e_default 'Unknown action_kind: ' || action_kind;

  if (add_params = '') then
    add_params = '{}';

  db_id = f_get_db_id();
  stop_perc = (db_id = 39801)
    and exists(select * from kontract where k_id = :k_id and coalesce(k_stop_perc, '31.12.9999') <= :rep_date);

  is_debug = 0;
  if (action_kind = 'auto' and add_params = 'debug') then
  begin
    is_debug = 1;

    select cml_id, cml_add_params
    from com_list
    where cml_object_kind = 9 and
          cml_object_id = :k_id and
          cml_date = :rep_date and
          cml_is_closed = 0
    into cml_id, add_params;

    select iif(s_running_total <> 0, 1, 0)
    from hp_repay_fill(:k_id, 0, :rep_date, 1, 0, 0, 0, 0, 0)
    into :sh_rec_id;

    select pr_percent_method from temp_params
    into pr_percent_method;

    delete from temp_sh
    where 1 = 1;

    insert into temp_sh (sh_id, sh_date, sh_date_orig, sh_repay_mode,
                         sh_zad_os, sh_zad_perc, sh_zad_fine,
                         sh_new_os, sh_new_perc, sh_new_fine,
                         sh_is_grace_os, sh_is_grace_perc, sh_is_grace_os_orig, sh_is_grace_perc_orig,
                         sh_annu, sh_annu_orig, sh_calcing_perc, sh_prior_percent_per_day)
    select s_id, s_date, s_date, repay_mode,
           zad_os, zad_perc, zad_fine,
           zad_os, iif(:stop_perc, 0, zad_perc), zad_fine,
           is_grace, 0, is_grace, 0,
           iif(:pr_percent_method in (-2, -1, 2), zad_os + zad_perc, zad_os),
           iif(:pr_percent_method in (-2, -1, 2), zad_os + zad_perc, zad_os),
           calcing_perc_orig, sh_prior_percent_per_day
    from temp_vrepay
    order by 2, 1;

    delete from temp_vrepay
    where 1 = 1;
  end

  first_perc_kind = f_json_str_get_value(add_params, 'first_perc_kind', 1);  -- разницу по процентам на последний платеж (1) или на первый
  add_month_count = f_json_str_get_value(add_params, 'add_month_count', 0);  -- кол-во доп.месяцев
  add_month_first = f_json_str_get_value(add_params, 'add_month_kind', 0);  -- кол-во сдвигов первого месяца
  new_annu = f_json_str_get_value(add_params, 'new_annu', 0);         -- сумма аннуитета. 0 - вычислять автоматом, иначе установить эту сумму
  new_pay_day = f_json_str_get_value(add_params, 'new_pay_day', 0);      -- новое число месяца платежа, если 0 то даты не меняются
  perc_lgota_value = f_json_str_get_value(add_params, 'perc_lgota_value', 0); -- значение скидки или новой %% ставки
  grace_os_list = f_json_str_get_value(add_params, 'grace_os_list', '');   -- список льготных погашений. В массиве передаются значения SH_ID через запятую

  last_sh_date = null;

  -- добавляем новый срок
  if (add_month_count <> 0 or add_month_first <> 0) then
  begin
    select count(*)
    from temp_sh
    where sh_repay_mode <> 1 and
          (sh_zad_os <> 0 or sh_zad_perc <> 0)
    into curr_period_count;
    if (curr_period_count is null) then
      curr_period_count = 0;

    if ((curr_period_count + add_month_count) <= add_month_first) then
    begin
      res_is_error = 1;
      res_text = 'Количество непросроченных платежей должно быть больше чем сдвиг первого платежа, увеличьте срок.';
      suspend;
      exit;
    end

    if (new_pay_day = 0) then
    begin
      select sh_date
      from temp_sh
      order by sh_date desc
      rows 1
      into last_sh_date;

      select iif(k_repay_day <= 0, extract(day from k_disb_date), k_repay_day)
      from kontract
      where k_id = :k_id
      into new_pay_day;
    end

    execute procedure hp_do_reschedule_add_period(:add_month_count, :add_month_first, :new_pay_day);

    if (last_sh_date is not null) then
    begin
      last_sh_date = dateadd(month, add_month_count, last_sh_date);
      last_sh_date = f_date_work_day(last_sh_date);

      update temp_sh
      set sh_date = :last_sh_date
      where 1 = 1
      order by sh_rec_id desc
      rows 1;
    end
  end

  if (new_pay_day <> 0) then -- установка новой даты платежа
  begin
    select sh_rec_id
    from temp_sh
    where sh_date >= :rep_date
    order by sh_rec_id desc
    rows 1
    into last_rec_id;

    first_date = null;
    date_prv = null;
    month_no = 0;
    for select sh_rec_id, sh_date
    from temp_sh
    where sh_date >= :rep_date
    order by sh_rec_id
    into sh_rec_id, sh_date
    do
    begin
      if (last_rec_id = sh_rec_id and last_sh_date is not null) then
      begin
        if (date_prv is null or date_prv < last_sh_date) then
          Continue;
      end

      if (first_date is null) then
      begin
        pay_day = new_pay_day;
        sh_month = extract(month from sh_date);
        sh_year = extract(year from sh_date);
        begin
          new_sh_date = right('0' || pay_day, 2) || '.' || right('0' || sh_month, 2) || '.' || sh_year;
          when any do
            new_sh_date = f_date_month_end('01.' || right('0' || sh_month, 2) || '.' || sh_year);
        end

        first_date = new_sh_date;
        new_sh_date = f_date_work_day(new_sh_date);
      end else
      begin
        month_no = month_no + 1;
        new_sh_date = dateadd(month, month_no, first_date);
        new_sh_date = f_date_work_day(new_sh_date);
      end

      update temp_sh
      set sh_date = :new_sh_date
      where sh_rec_id = :sh_rec_id;

      date_prv = new_sh_date;
    end
  end

  if (grace_os_list <> '') then -- устанавливаем льготные периоды по основной сумме
  begin
    for select res_str
    from hp_split_str(:grace_os_list, ',')
    into sh_id
    do
    begin
      update temp_sh
      set sh_is_grace_os = 1
      where sh_id = :sh_id and
            coalesce(sh_is_grace_os, 0) <> 1;
    end
  end

  if (new_annu = 0) then
    execute procedure hp_do_reschedule_calc(:k_id, :rep_date, 1, :first_perc_kind, :perc_lgota_value);
  else
  begin
    update temp_sh
    set sh_annu = iif(sh_is_grace_os = 1, 0, :new_annu)
    where sh_repay_mode <> 1;

    execute procedure hp_do_reschedule_calc(:k_id, :rep_date, 0, :first_perc_kind, :perc_lgota_value);
  end

  select res_is_error, res_text
  from hp_do_reschedule_can_delinq
  into :res_is_error, :res_text;

  if (res_text = '' and
      exists(select * from temp_sh
             where sh_new_os < 0 or
                   sh_new_perc < 0)) then
  begin
    res_is_error = 1;
    res_text = 'При построении нового графика появляются отрицательные суммы. Обратитесь в тех.поддержку.';
  end

  if (res_text = '' and perc_lgota_value > 0 and rep_date <= '31.05.2020') then
  begin
    select sh_date from temp_sh
    where sh_new_os > 0 or
          sh_new_perc > 0
    order by sh_date
    rows 1
    into sh_date;

    if (sh_date > '01.09.2020') then
    begin
      res_is_error = 1;
      res_text = 'Первый платеж слишком далеко, измените параметры чтобы попадал в август.';
    end
  end

  if (res_text = '' and is_debug = 1 and action_kind = 'auto') then
  begin
    delete from com_list_resh
    where cmr_from_cml_id = :cml_id;

    insert into com_list_resh (cmr_from_cml_id, cmr_sh_id, cmr_sh_date, cmr_sh_os, cmr_sh_perc)
    select :cml_id, sh_id, sh_date, sh_new_os, sh_new_perc
    from temp_sh
    where sh_new_os <> 0 or sh_new_perc <> 0
    order by sh_date, sh_id;
  end

  suspend;
end
