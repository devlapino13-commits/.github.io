CREATE PROCEDURE AAREP_PLAN_INDICATOR(
    UN_ID ID_TYPE,
    DT_CALC DATE)
RETURNS (
    HTML_TXT D_BLOB_TEXT)
AS
declare variable dt_beg date;
declare variable dt_end date;
declare variable yr_calc integer;
declare variable mn_calc integer;
declare variable rd_id integer;
declare variable rd_code str_255;
declare variable rd_name str_255;
declare variable is_ipf str_255;
declare variable c_id id_type;
declare variable k_id id_type;
declare variable k_amount_approved curr_type;
declare variable sum_plan curr_type;
declare variable sum_fact curr_type;
declare variable sum_ahead curr_type;
declare variable perc_perf curr_type;
declare variable html_head d_blob_text;
declare variable html_body d_blob_text;
declare variable delim str_10;
declare variable ok_os curr_type;
declare variable ol_date date;
declare variable kf_officer id_type;
declare variable pr_code str_100;
declare variable kf_id id_type;
declare variable reg_id integer;
declare variable jk_system_code varchar(100);
declare variable is_km smallint;
begin
  html_txt = '';
  c_id = (select un_from_c_id
          from username
          where un_id = :un_id);
  if (c_id is null) then
  begin
    suspend;
    exit;
  end

  select ps_from_reg_id, jk_system_code
  from hr_pstatus, joblist, jobkind
  where :dt_calc between ps_date_begin and ps_date_end and
        ps_from_c_id = :c_id and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id
  into :reg_id, jk_system_code;

  if (jk_system_code = 'JobSystem.Region_KM') then
  begin
    is_km = 1;
  end
  else
  begin
--    reg_id = 1000000;
    is_km = 0;
  end

  dt_beg = dateadd(1 - extract(day from dt_calc) day to dt_calc);
  dt_end = dateadd(month, 1, dt_beg) - 1;

  yr_calc = extract(year from dt_calc);
  mn_calc = extract(month from dt_calc);
  delim = ascii_char(13) || ascii_char(10);

  html_head = '<br><h1 class="green_text">Планы по показателям</h1>' || delim || '<font color="black">На месяц: ' || lower(f_monthname(:dt_calc)) || '</font><br>' || delim;
  html_body = '';
  for select rd_id, rd_code, rd_name
      from refdata
      where rd_code in ('PlanIndicators.Disb_Amount', 'PlanIndicators.Disb_Amount_IPF')
      order by rd_id
      into :rd_id, :rd_code, :rd_name
  do
  begin
    is_ipf = iif(rd_code = 'PlanIndicators.Disb_Amount', 0, 1);

    -- Новый вариант получения плановых показателей, Sanjari
    select po_amount
    from hp_get_planoper_info(:reg_id, :c_id, :rd_code, :dt_end, null)
    into sum_plan;

--  Отключен из-за медленного выполнения, Sanjari
--    sum_plan = coalesce((select sum(pd_data_1)
--                         from hp_tmp_plandata_show(:reg_id, :dt_calc, :rd_id, 100 * :yr_calc + :mn_calc, 1, :un_id) -- было  1000000, :dt_calc, :rd_id, 100 * :yr_calc + :mn_calc, 1, :un_id)
--                         where pd_from_c_id = :c_id or
--                               (:is_km = 1 and
--                               pd_from_c_id = 0)), 0); -- было  WHERE pd_from_c_id = :c_id), 0);


    if (sum_plan = 0) then
      continue;
    sum_fact = 0;
    sum_ahead = 0;
    for select k_id, k_amount_approved
        from kontract
        inner join product on pr_id = k_from_pr_id and
              iif(pr_code containing '.ipf.', 1, 0) = :is_ipf
        inner join kontstate on ks_from_k_id = k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :dt_beg and :dt_end and
              (ks_kc_id = :c_id or (:is_km = 1 and
              ks_region_id = :reg_id))
        into :k_id, :k_amount_approved
    do
    begin
      sum_fact = sum_fact + k_amount_approved;
    end
    /*
-- << -- Вычисляю дострочные погашения -------------------------------------
-- выборка групп, с которыми был связан СВЗ за период
    for select kal.kal_from_kf_id
    from knaktlist kal -- все контракты которые передавались от СВЗ в рамках периода
    where kal.is_delete = 0
    and (kal.kal_new_kf_officer = :c_id --1090100478215
         or (:is_km = 1 and kal_new_reg_id = :reg_id))
    and kal.kal_date_end >= :dt_beg --cast('01.09.2018'  as date)
    and kal.kal_date_end <= :dt_end --cast('30.09.2018'  as date)
    and kal.kal_date <= kal.kal_date_end
    union
    select kal.kal_from_kf_id
    from knaktlist kal -- все которые остаются на руках у свз на конец периода.
    where kal.is_delete = 0
    and (kal.kal_new_kf_officer = :c_id --1090100478215
         or (:is_km = 1 and kal_new_reg_id = :reg_id))
    and kal.kal_date_end >= :dt_end --cast('30.09.2018'  as date)
    and kal.kal_date <= :dt_end --cast('30.09.2018'  as date)
    and kal.kal_date <= kal.kal_date_end
    into :kf_id
    do
    begin
-- определяю продукт и кредит
      for select pr.pr_code, kon.k_id
      from kontract kon
      inner join product pr on pr.pr_id = kon.k_from_pr_id
      where kon.k_from_kf_id = :kf_id
      into :pr_code, :k_id
      do
      begin
          if(IIF(pr_code CONTAINING '.ipf.', 1, 0) = :is_ipf) then
          begin
--  Если есть погашение, которое считается досрочным. (Нас интересует сумма ОС)
              for select -sum(ok_os), ok_ol_date
              from operkred
              where ok_from_k_id = :k_id
              and ok_sh_date > :dt_end
              and ok_ol_date between :dt_beg and :dt_end
              and ok_ol_is_active = 1
              and ok_type in (4,9)
              and ok_os < 0
              group by ok_ol_date
              into :ok_os, :ol_date
              do
              begin
-- если на момент погашения группа числится за нашим СВЗ, тогда + его сумму к итоговой по досрочным
                 if (exists (select *
                             from knaktlist kal
                             where kal.kal_from_kf_id =:kf_id
                             and :ol_date between kal.kal_date and kal.kal_date_end
                             and (kal.kal_new_kf_officer = :c_id
                                  or (:is_km = 1 and kal_new_reg_id = :reg_id))
                             and kal.is_delete = 0)) then
                    sum_ahead = sum_ahead + :ok_os;
              end
          end
       end
    end
-- << --
    */
    select -sum(ok_os)
    from operkred, kontract, knaktlist, product
    where ok_ol_is_active = 1 and
          ok_ol_date between :dt_beg and :dt_end and
          ok_os < 0 and
          ok_type in (4, 9) and
          ok_sh_date > :dt_end and
          ok_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          ok_ol_date between kal_date and kal_date_end and
          k_from_pr_id = pr_id and
          ((:is_ipf = 1 and
          pr_code containing '.ipf.') or
          (:is_ipf = 0 and
          pr_code not containing '.ipf.')) and
          ((:is_km = 1 and
          kal_new_reg_id = :reg_id) or
          kal_new_kf_officer = :c_id)
    into sum_ahead;
    if (sum_ahead is null) then
      sum_ahead = 0;

    sum_plan = round(sum_plan);
    sum_fact = round(sum_fact);
--perc_perf = 100 * sum_fact / (sum_plan + sum_ahead);
    html_body = blob_append(html_body, '<h2>' || rd_name || '</h2>' || delim || '<table width="50%"><col width="70%"><col width="30%">' || delim || '<tr><td>' || ' План                ' || '</td>' || '<td align="right">' || cast(sum_plan /* + sum_ahead*/ as id_type) || '</td></tr>' || delim || --  /* + sum_ahead*/ убрал igusev 06.10.2018 в соответсивии с запросом Бахтияра
    '<tr><td>' || ' Осталось ' || '</td>' || '<td align="right">' || cast((sum_plan + sum_ahead - sum_fact) as id_type) || '</td></tr>' || delim || '</table>' || delim || '<br>' || delim);
  end
  html_txt = blob_append(html_head, html_body);
  suspend;
end
