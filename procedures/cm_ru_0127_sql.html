CREATE PROCEDURE CM_RU_0127_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KF_DISB_DATE D_DATE;
declare variable KC_FROM_C_ID ID_TYPE;
declare variable K_KONTRACT_NO varchar(50);
declare variable RE_PR_CODE varchar(100);
declare variable PR_CODE D_STR_100;
begin
  for
  select k_disb_date, kc_from_c_id, pr_code
  from kontract, product, knclient
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_product_kind <> 2
  and pr_code not containing '.patent plus.'
  and pr_code not containing '.food.'
  and pr_code not containing '.float_percent.'
  and pr_code not containing '.express_online.'
  and pr_code not containing '.support.'
  and pr_code not containing '.summer_plus.'
  and pr_code not containing '.comfortable.'
  and pr_code not containing '.zalog.'
  and pr_code not containing '.selfemployed.'
  and pr_code not containing '.business.'
  and pr_code not containing '.loyal.'
  and pr_code not containing '.reliable.'
  and pr_code not containing '.universal.'
  and pr_code not containing '.privilege.'
  and pr_code not containing '.privilege_plus.'
  and pr_code not containing '.oivo.'
  and pr_code not containing '.plus_60.'
  and pr_code not containing '.plus_100.'
  and pr_code not containing '.plus_100_b.'
  and k_refund = 0
  and kc_from_k_id = k_id
  and kc_amount_approved > 0
  into kf_disb_date, kc_from_c_id, pr_code
  do
  begin
    for
    select k_kontract_no, pr_code
    from kontstate, knclient, kontract, product pr
    where ks_ol_is_delete = 0
    and ks_state = 1
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kc_from_c_id = :kc_from_c_id
    and kc_from_k_id = k_id
    and kc_from_k_id <> :k_id
    and k_from_pr_id = pr_id
    and pr_product_kind <> 2
    and current_date between ks_date_begin and ks_date_end
    and pr.pr_code not containing '.float_percent.'
    and pr_code not containing '.summer_plus.'
    and pr_code not containing '.comfortable.'
    and pr_code not containing '.zalog.'
    and pr_code not containing '.selfemployed.'
    and pr_code not containing '.business.'
    and pr_code not containing '.privilege.'
    and pr_code not containing '.privilege_plus.'
    and pr_code not containing '.universal.'
    and pr_code not containing '.oivo.'
    and pr_code not containing '.plus_60.'
    and pr_code not containing '.plus_100.'
    and pr_code not containing '.plus_100_b.'
    into k_kontract_no, re_pr_code
    do
    begin
      if ( pr_code = re_pr_code and  --RFCRU-4953
      not((pr_code containing '.loyal.' and re_pr_code containing '.loyal.') or
          (pr_code containing '.reliable.' and re_pr_code containing '.reliable.') or
          (pr_code containing '.shans.' and re_pr_code not containing '.shans.') or
          (pr_code containing '.express.' and re_pr_code containing '.express.')   --RFCRU-4953
          ))
      then
      begin
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into res_info_text;
        res_error_text = 'У клиента имеется активный кредит в следующем контракте: '||k_kontract_no;
        res_error_is_critical = 1;
        --По просьбе Светланы 11.05.20
        if (re_pr_code containing '.food.' and
              not exists(select * from oksummary oks
                         where :kf_disb_date between oks_date_begin and oks_date_end and
                         oks_delinq_date < :kf_disb_date and 
                         oks_close_state = 0 and
                         oks_from_k_id = :k_id)) then
        res_error_is_critical = 0;
        suspend;
      end
    end
  end
end
