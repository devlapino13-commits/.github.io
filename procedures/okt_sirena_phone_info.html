CREATE PROCEDURE OKT_SIRENA_PHONE_INFO(
    CELL_NO D_STR_20)
RETURNS (
    HTML D_BLOB_TEXT)
AS
declare variable c_id d_id;
declare variable c_birth_date d_date;
declare variable c_name_fio d_str_100;
declare variable citizen d_str_255;
declare variable nation d_str_255;
declare variable adc_address_type d_integer;
declare variable address_str d_str_255;
declare variable address_jur d_str_255;
declare variable address_fact d_str_255;
declare variable address_rodina_jur d_str_255;
declare variable address_rodina_fact d_str_255;
declare variable timezone_ru d_integer;
declare variable cl_time d_str_100;
declare variable paysys_acc_no d_str_10;
declare variable ks_state d_smallint;
declare variable k_kontract_no d_str_20;
declare variable k_id d_id;
declare variable status varchar(50);
declare variable age d_integer;
declare variable timezone d_integer;
declare variable loan_counter d_integer;
begin

  html = '';

  select c_id, c_birth_date, c_name_fio
  from hp_find_client_by_continfo(null, :cell_no)
  into c_id, c_birth_date, c_name_fio;

  if (coalesce(c_id, 0) = 0) then
  begin
    html = 'Информация о клиенте не найдена';
    suspend;
    exit;
  end

  select rd.rd_name,
         rd2.rd_name
  from clients cl
  inner join refdata rd on rd.rd_id = cl.c_citizen
  inner join refdata rd2 on rd2.rd_id = cl.c_nation
  where cl.c_id = :c_id
  into :citizen,
       :nation;

  age = f_age(:c_birth_date, current_date);

  for select adcl.adc_address_type, trim(coalesce(adl.adl_address_str, ''))
      from addrclient adcl
      inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
      where adcl.adc_from_c_id = :c_id and
            adcl.adc_address_type in (1, 3, 4, 5)
      into :adc_address_type,
           :address_str
  do
  begin
    if (adc_address_type = 1) then
      address_jur = :address_str;
    if (adc_address_type = 2) then
      address_fact = :address_str;
    if (adc_address_type = 4) then
      address_rodina_jur = :address_str;
    if (adc_address_type = 5) then
      address_rodina_fact = :address_str;
  end
  address_jur = coalesce(address_jur,'');
  address_fact = coalesce(address_fact,'');
  address_rodina_jur = coalesce(address_rodina_jur,'');
  address_rodina_fact = coalesce(address_rodina_fact,'');

  timezone_ru = coalesce(timezone_ru, 3);

  if (cell_no starting('996')) then
    timezone = 6;
  else
  if (cell_no starting('998')) then
    timezone = 5;
  else
  if (cell_no starting('992')) then
    timezone = 5;
  else
    timezone = timezone_ru;

  cl_time = trim(substring((dateadd(hour, :timezone - 3, current_timestamp)) from 12 for 5));

  select uniq_acc_str
  from hp_show_clientuniq_acc_no(:c_id)
  into :paysys_acc_no;

  html = '<br><b>' || :c_name_fio || '</b></br>' ||
         '<br><b>' || 'Общая информация по клиенту' || '</b></br>' ||
         '<br>' || 'Гражданство..............' || :citizen || '</br>' ||
         '<br>' || 'Национальность.......' || :nation || '</br>' ||
         '<br>' || 'Дата рождения..........' || :c_birth_date || '</br>' ||
         '<br>' || 'Возраст (лет).............' || :age || '</br>' ||
         '<br>' || 'Адрес прописки........' || iif(:address_jur <> '', :address_jur, 'Незаполнено') || '</br>' ||
         '<br>' || 'Адрес проживания....' || iif(:address_fact <> '', :address_fact, 'Незаполнено')  || '</br>' ||
         '<br>' || 'Адрес прописки (Родина)..' || iif(:address_rodina_jur <> '', :address_rodina_jur, 'Незаполнено') || '</br>' ||
         '<br>' || 'Адрес проживания (Родина)..' || iif(:address_rodina_fact <> '', :address_rodina_fact, 'Незаполнено') || '</br>' ||
         '<br>' || 'Часовой пояс.............' || :timezone || '</br>' ||
         '<br>' || 'Сейчас у абонента....' || :cl_time || '</br>' ||
         '<br><b>' || 'Лицевой счёт клиента ' || :paysys_acc_no || '</b></br>';

  html = BLOB_APPEND(html, '<br><b> Кредитные договора </b></br>');
  html = BLOB_APPEND(html, '<table border="1">');

  loan_counter = 0;
  for select ks.ks_state,
             kt.k_kontract_no,
             kt.k_id
      from knclient kc
      inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
      inner join kontract kt on kt.k_id = ks.ks_from_k_id
      where kc.kc_from_c_id = :c_id and
            kc.kc_amount_approved > 0 and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end
      into :ks_state,
           :k_kontract_no,
           :k_id
  do
  begin
    if (exists(select *
               from oksummary oks
               where oks.oks_from_k_id = :k_id and
                     oks.oks_close_state = 1 and
                     current_date between oks.oks_date_begin and oks.oks_date_end)) then
      status = 'Внесистемный учёт';
    else
      status = (case
                  when :ks_state = 0 then 'В процессе'
                  when :ks_state = 1 then 'Активный'
                  when :ks_state = 2 then 'Закрытый'
                end);
    html = BLOB_APPEND(html, '<tr><th>' ||:k_kontract_no||'</th><th>'||:status||'</th></tr>');
    loan_counter = loan_counter + 1;
  end

  if (loan_counter = 0) then
    html = BLOB_APPEND(html, '<br><b>Кредитные договора отсуствуют</b></br>');


  html = BLOB_APPEND(html, '</table>');
  suspend;
end
