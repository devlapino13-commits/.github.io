CREATE PROCEDURE UXP$UXIMAGE_GET_ITEM(
    IMG_UUID_CODE D_STR_CODE = null)
RETURNS (
    IMG_NAME D_STR_100,
    IMG_SVG D_BLOB_TEXT)
AS
declare variable img_uuid d_uuid;
begin
  img_uuid = char_to_uuid(img_uuid_code);

  img_svg = '';
  select img_svg, replace(img_name, '.svg', '') || '   / ' || img_category
  from ux$image
  where img_uuid = :img_uuid
  into img_svg, img_name;

  suspend;
end
