CREATE PROCEDURE HP_IDENTICAL_CLIENT_ADDRESS(
    ADL_ID D_ID,
    C_ID D_ID,
    ADL_ADDRESS_TYPES D_STR_100)
RETURNS (
    RES_C_ID D_ID,
    RES_ADDRESS_STR D_STR_255)
AS
declare variable adl_street_id d_id;
declare variable adl_depart_no d_str_20;
declare variable adl_depart_pref d_str_20;
declare variable adl_flat_no d_str_20;
declare variable adl_building_no d_str_20;
declare variable adl_section_no d_str_20;
declare variable adl_indexed_address d_str_255;
declare variable adl_from_pl_id d_id;
declare variable pl_parent_id d_id;
begin
  -- по переданному ADL_ID ищет других клиентов с таким адресом

  if (coalesce(adl_id, 0) = 0) then
    exit;

  -- если адрес не заполнен, завершаем работу
  if (adl_id in (81001000000000, 41701000000000, 94401000000000)) then
    exit;

  select adl_street_id, adl_depart_no, adl_depart_pref, adl_flat_no,
    adl_building_no, adl_section_no, adl_address_str, adl_indexed_address,
    adl_from_pl_id, pl.parent
  from addrlist, places pl
  where adl_id = :adl_id
    and adl_from_pl_id = pl.pl_id
  into adl_street_id, adl_depart_no, adl_depart_pref, adl_flat_no,
    adl_building_no, adl_section_no, res_address_str, adl_indexed_address,
    adl_from_pl_id, pl_parent_id;

  -- если улица не указана (выыбрана главная ветка), тоже завершаем работу
  if (adl_street_id in (810000000001, 417000000001, 944000000001)) then
    exit;

  -- Если населенный пункт (столица) подчиняется стране, и нет улицы, завершаем работу
  if (pl_parent_id in (810000000001, 417000000001, 944000000001) and adl_from_pl_id = adl_street_id) then
    exit;

  if (coalesce(adl_address_types, '') = '') then
    adl_address_types = '1,2,3,4,5';

  if (adl_depart_no is null) then  adl_depart_no = 0;
  if (adl_depart_pref is null) then  adl_depart_pref = '';
  if (adl_flat_no is null) then  adl_flat_no = 0;
  if (adl_building_no is null) then  adl_building_no = 0;
  if (adl_section_no is null) then  adl_section_no = 0;

  for select distinct adc_from_c_id
  from addrlist, addrclient
  where adc_from_c_id <> :c_id and
        hash(adl_indexed_address) = hash(:adl_indexed_address) and
        adl_indexed_address = :adl_indexed_address and
        adl_id = adc_from_adl_id and
        :adl_address_types containing adc_address_type
  into res_c_id
  do
    suspend;
end
