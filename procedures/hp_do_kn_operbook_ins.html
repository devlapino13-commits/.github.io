CREATE PROCEDURE HP_DO_KN_OPERBOOK_INS(
    KOB_FROM_OL_ID D_ID,
    KOB_FROM_KC_ID D_ID,
    KOB_ACC_DT D_ID,
    KOB_ACC_KT D_ID,
    KOB_MINUS_ACC_DT D_ID,
    KOB_MINUS_ACC_KT D_ID,
    KOB_SUMMA D_CURRENCY,
    KOB_SUMMA_VAL D_CURRENCY,
    KOB_SUMMA_TAX D_CURRENCY,
    KOB_OL_DATE D_DATE,
    KOB_OL_REG_ID D_INTEGER,
    KOB_OL_IS_ACTIVE D_INTEGER,
    KOB_DETAIL_STR D_STR_255,
    KOB_SUMMA_KIND D_SMALLINT = 0,
    KOB_ALLOW_MINUS D_INTEGER = 1,
    KOB_CLIENT_DT D_ID = null,
    KOB_CLIENT_KT D_ID = null,
    KOB_CURRENCY D_STR_LANG = null,
    KOB_CURR_RATE D_CURRENCY = null)
AS
declare variable kob_detail_id decimal(18,0);
declare variable saldo_sum curr_type;
declare variable saldo_val curr_type;
declare variable saldo_tax curr_type;
declare variable tmp_sum curr_type;
declare variable tmp_val curr_type;
declare variable tmp_tax curr_type;
declare variable tmp_id id_type;
declare variable acc_type integer;
declare variable dt_pair_dt_acc id_type;
declare variable dt_pair_kt_acc id_type;
declare variable dt_amount_sum curr_type;
declare variable dt_amount_val curr_type;
declare variable dt_amount_tax curr_type;
declare variable kt_amount_tax curr_type;
declare variable kt_pair_dt_acc id_type;
declare variable kt_pair_kt_acc id_type;
declare variable kt_amount_sum curr_type;
declare variable kt_amount_val curr_type;
declare variable tmp_acc_id id_type;
declare variable pair_acc id_type;
declare variable tax_kind_dt d_smallint;
declare variable tax_kind_kt d_smallint;
declare variable kob_summa_tax_dt curr_type;
declare variable kob_summa_tax_kt curr_type;
declare variable tmp_tax_dt curr_type;
declare variable tmp_tax_kt curr_type;
declare variable saldo_tax_dt curr_type;
declare variable saldo_tax_kt curr_type;
begin
  -- процедура для сохранения одной строки проводки в новом плане счетов
  -- если значения параметра kob_summa_kind:
  -- 0, значит это сумма проводки.
  -- 1 - значит это сумма остатка ДТ счета после сохранения проводки.
  -- -1 - значит это сумма остатка КТ счета после сохранения проводки.
  -- 2 - значит общий оборот за день должен быть равен указанной сумме
  -- при значении <> 0, вычисляется текущий остаток на дату и на разницу строится проводка

  dt_pair_dt_acc = null;
  dt_pair_kt_acc = null;

  kob_summa_tax_dt = kob_summa_tax;
  kob_summa_tax_kt = kob_summa_tax;

  if (kob_detail_str is null) then
    kob_detail_str = '';
  kob_detail_str = trim(kob_detail_str);

  if (kob_curr_rate is null) then
    kob_curr_rate = 1;

  if (coalesce(kob_currency, '') = '') then
  begin
    select cm_default_currency
    from company
    into kob_currency;
    kob_curr_rate = 1;
  end

  tax_kind_dt = 2;
  tax_kind_kt = 2;

  if (kob_curr_rate = 0) then
    kob_curr_rate = 1;
  if (kob_curr_rate = 1) then
    kob_summa_val = 0;

  if (kob_ol_is_active is null or
      kob_ol_date is null) then
  begin
    select ol_date, 1 - coalesce(is_delete, 0)
    from operlist
    where ol_id = :kob_from_ol_id
    into kob_ol_date, kob_ol_is_active;
  end

  if (kob_summa_kind is null) then
    kob_summa_kind = 0;
  if (kob_summa is null) then
    kob_summa = 0;
  if (kob_summa_val is null) then
    kob_summa_val = 0;
  if (kob_summa_tax_dt is null) then
    kob_summa_tax_dt = 0;
  if (kob_summa_tax_kt is null) then
    kob_summa_tax_kt = 0;

  -- нулевые проводки не сохраняем
  if (kob_summa_kind = 0 and
      kob_summa = 0 and
      kob_summa_val = 0 and
      kob_summa_tax_dt = 0 and
      kob_summa_tax_kt = 0) then
    exit;

  if (kob_allow_minus is null) then
    kob_allow_minus = 1;

  pair_acc = null;
  if (kob_summa_kind = -1) then
  begin
    select acc_type, iif(ab_pair_kind = 1, null, acc_pair_acc_id)
    from ref_account, ref_accbalance
    where acc_id = :kob_acc_kt and
          acc_from_ab_id = ab_id
    into :acc_type, :pair_acc;

    saldo_sum = 0;
    saldo_val = 0;
    saldo_tax = 0;
    select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
    from kn_obsummary
    where kobs_acc_id = :kob_acc_kt and
          kobs_date <= :kob_ol_date
    order by kobs_acc_id desc, kobs_date desc
    rows 1
    into saldo_sum, saldo_val, saldo_tax;

    if (saldo_sum is null) then
      saldo_sum = 0;
    if (saldo_val is null) then
      saldo_val = 0;
    if (saldo_tax is null) then
      saldo_tax = 0;

    if (pair_acc is not null) then
    begin
      tmp_sum = 0;
      tmp_val = 0;
      tmp_tax = 0;

      select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
      from kn_obsummary
      where kobs_acc_id = :pair_acc and
            kobs_date <= :kob_ol_date
      order by kobs_acc_id desc, kobs_date desc
      rows 1
      into tmp_sum, tmp_val, tmp_tax;

      if (tmp_sum is null) then
        tmp_sum = 0;
      if (tmp_val is null) then
        tmp_val = 0;
      if (tmp_tax is null) then
        tmp_tax = 0;

      saldo_sum = saldo_sum + tmp_sum;
      saldo_val = saldo_val + tmp_val;
      saldo_tax = saldo_tax + tmp_tax;
    end
    -- меняем знак текущего остатка, так как остаток по КТ идет с минусом
    saldo_sum = -saldo_sum;
    saldo_val = -saldo_val;
    saldo_tax = -saldo_tax;

    kob_summa = kob_summa - saldo_sum;
    kob_summa_val = kob_summa_val - saldo_val;
    kob_summa_tax_dt = kob_summa_tax_dt - saldo_tax;
    kob_summa_tax_kt = kob_summa_tax_kt - saldo_tax;
  end

  if (kob_summa_kind = 1) then
  begin
    pair_acc = null;
    select acc_type, iif(ab_pair_kind = 1, null, acc_pair_acc_id)
    from ref_account, ref_accbalance
    where acc_id = :kob_acc_dt and
          acc_from_ab_id = ab_id
    into :acc_type, :pair_acc;

    saldo_sum = 0;
    saldo_val = 0;
    saldo_tax = 0;
    select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
    from kn_obsummary
    where kobs_acc_id = :kob_acc_dt and
          kobs_date <= :kob_ol_date
    order by kobs_acc_id desc, kobs_date desc
    rows 1
    into saldo_sum, saldo_val, saldo_tax;

    if (saldo_sum is null) then
      saldo_sum = 0;
    if (saldo_val is null) then
      saldo_val = 0;
    if (saldo_tax is null) then
      saldo_tax = 0;

    if (pair_acc is not null) then
    begin
      tmp_sum = 0;
      tmp_val = 0;
      tmp_tax = 0;
      select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
      from kn_obsummary
      where kobs_acc_id = :pair_acc and
            kobs_date <= :kob_ol_date
      order by kobs_acc_id desc, kobs_date desc
      rows 1
      into tmp_sum, tmp_val, tmp_tax;

      if (tmp_sum is null) then
        tmp_sum = 0;
      if (tmp_val is null) then
        tmp_val = 0;
      if (tmp_tax is null) then
        tmp_tax = 0;

      saldo_sum = saldo_sum + tmp_sum;
      saldo_val = saldo_val + tmp_val;
      saldo_tax = saldo_tax + tmp_tax;
    end

    kob_summa = kob_summa - saldo_sum;
    kob_summa_val = kob_summa_val - saldo_val;
    kob_summa_tax_dt = kob_summa_tax_dt - saldo_tax;
    kob_summa_tax_kt = kob_summa_tax_kt - saldo_tax;
  end

  if (kob_summa_kind = 2) then
  begin
    saldo_sum = 0;
    saldo_val = 0;
    saldo_tax_dt = 0;
    saldo_tax_kt = 0;

    tmp_sum = 0;
    tmp_val = 0;
    tmp_tax = 0;
    select sum(kob_summa), sum(kob_summa_val), sum(kob_summa_tax_dt), sum(kob_summa_tax_kt)
    from kn_operbook
    where kob_acc_dt = :kob_acc_dt and
          kob_acc_kt = :kob_acc_kt and
          kob_ol_date = :kob_ol_date and
          kob_ol_is_active = 1
    into tmp_sum, tmp_val, tmp_tax_dt, tmp_tax_kt;

    if (tmp_sum is null) then
      tmp_sum = 0;
    if (tmp_val is null) then
      tmp_val = 0;
    if (tmp_tax_dt is null) then
      tmp_tax_dt = 0;
    if (tmp_tax_kt is null) then
      tmp_tax_kt = 0;
    saldo_sum = saldo_sum + tmp_sum;
    saldo_val = saldo_val + tmp_val;
    saldo_tax_dt = saldo_tax_dt + tmp_tax_dt;
    saldo_tax_kt = saldo_tax_kt + tmp_tax_kt;

    tmp_sum = 0;
    tmp_val = 0;
    tmp_tax_dt = 0;
    tmp_tax_kt = 0;
    select sum(kob_summa), sum(kob_summa_val), sum(kob_summa_tax_dt), sum(kob_summa_tax_kt)
    from kn_operbook
    where kob_acc_dt = iif(:kob_minus_acc_kt is not null, :kob_minus_acc_kt, :kob_acc_kt) and
          kob_acc_kt = iif(:kob_minus_acc_dt is not null, :kob_minus_acc_dt, :kob_acc_dt) and
          kob_ol_date = :kob_ol_date and
          kob_ol_is_active = 1
    into tmp_sum, tmp_val, tmp_tax_dt, tmp_tax_kt;

    if (tmp_sum is null) then
      tmp_sum = 0;
    if (tmp_val is null) then
      tmp_val = 0;
    if (tmp_tax_dt is null) then
      tmp_tax_dt = 0;
    if (tmp_tax_kt is null) then
      tmp_tax_kt = 0;
    saldo_sum = saldo_sum - tmp_sum;
    saldo_val = saldo_val - tmp_val;
    saldo_tax_dt = saldo_tax_dt - tmp_tax_dt;
    saldo_tax_kt = saldo_tax_kt - tmp_tax_kt;

    kob_summa = kob_summa - saldo_sum;
    kob_summa_val = kob_summa_val - saldo_val;
    kob_summa_tax_dt = kob_summa_tax_dt - saldo_tax_dt;
    kob_summa_tax_kt = kob_summa_tax_kt - saldo_tax_kt;

    -- нулевые проводки не сохраняем
    if (kob_summa = 0 and
        kob_summa_val = 0 and
        kob_summa_tax_dt = 0 and
        kob_summa_tax_kt = 0) then
      exit;
  end

  if (kob_allow_minus = 0 and
      kob_summa < 0 and
      kob_summa_kind in (-1, 0, 1)) then
  begin
    if (kob_summa_kind = -1 and
        kob_minus_acc_dt is not null) then
      kob_acc_dt = kob_minus_acc_dt;
    if (kob_summa_kind = 1 and
        kob_minus_acc_kt is not null) then
      kob_acc_kt = kob_minus_acc_kt;

    tmp_id = kob_acc_dt;
    kob_acc_dt = kob_acc_kt;
    kob_acc_kt = tmp_id;

    tmp_id = kob_client_dt;
    kob_client_dt = kob_client_kt;
    kob_client_kt = tmp_id;

    kob_summa_kind = -kob_summa_kind;

    kob_summa = -kob_summa;
    kob_summa_val = -kob_summa_val;
    kob_summa_tax_dt = -kob_summa_tax_dt;
    kob_summa_tax_kt = -kob_summa_tax_kt;
  end

  dt_amount_sum = 0;
  dt_amount_val = 0;
  dt_amount_tax = 0;

  kt_amount_sum = 0;
  kt_amount_val = 0;
  kt_amount_tax = 0;

  pair_acc = null;
  select acc_type, iif(ab_pair_kind = 1, null, acc_pair_acc_id),
         coalesce(sab_tax_kind, coalesce(ab_tax_kind, 0))
  from ref_account, ref_accbalance, sys_accbalance
  where acc_id = :kob_acc_dt and
        acc_from_ab_id = ab_id and
        acc_from_sab_id = sab_id
  into :acc_type, :pair_acc, :tax_kind_dt;

  -- Правила для парных счетов:
  -- В проводках приоритетно берем тот счет, где имеется остаток
  -- Если в обоих счетах нет остатка, то:
  --   Если счет участвует в Дт, берем активный счет
  --   Иначе берем пассивный счет

  -- если по Дебету идет парный счет, проверяем
  if (pair_acc is not null) then
  begin
    tmp_sum = 0;
    tmp_val = 0;
    tmp_tax = 0;
    select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
    from kn_obsummary
    where kobs_acc_id = :kob_acc_dt and
          kobs_date <= :kob_ol_date
    order by kobs_acc_id desc, kobs_date desc
    rows 1
    into tmp_sum, tmp_val, tmp_tax;

    -- если по этому счету нет остатка, проверяем дальше
    if (tmp_sum = 0 and
        tmp_val = 0 and
        tmp_tax = 0) then
    begin
      select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
      from kn_obsummary
      where kobs_acc_id = :pair_acc and
            kobs_date <= :kob_ol_date
      order by kobs_acc_id desc, kobs_date desc
      rows 1
      into tmp_sum, tmp_val, tmp_tax;

      -- если прилетел пассивный счет, меняем его на активный (парный)
      if (acc_type = 1) then
      begin
        tmp_acc_id = kob_acc_dt;
        kob_acc_dt = pair_acc;
        pair_acc = tmp_acc_id;
        acc_type = 0;
      end
      else
      begin
        -- иначе проверяем, если у парного есть остаток, меняем Дт счет на парный.
        if (tmp_sum <> 0 or
            tmp_val <> 0 or
            tmp_tax <> 0) then
        begin
          tmp_acc_id = kob_acc_dt;
          kob_acc_dt = pair_acc;
          pair_acc = tmp_acc_id;
          acc_type = 1 - acc_type;
        end
      end
    end

    if (acc_type = 0) then
    begin
      if (tmp_sum + kob_summa < 0) then
      begin
        dt_pair_dt_acc = kob_acc_dt;
        dt_pair_kt_acc = pair_acc;
        dt_amount_sum = -(tmp_sum + kob_summa);
        dt_amount_val = -(tmp_val + kob_summa_val);
        dt_amount_tax = -(tmp_tax + kob_summa_tax_dt);
      end
    end

    if (acc_type = 1) then
    begin
      if (tmp_sum + kob_summa > 0) then
      begin
        dt_pair_dt_acc = pair_acc;
        dt_pair_kt_acc = kob_acc_dt;
        dt_amount_sum = (tmp_sum + kob_summa);
        dt_amount_val = (tmp_val + kob_summa_val);
        dt_amount_tax = (tmp_tax + kob_summa_tax_dt);
      end
    end
  end

  pair_acc = null;
  select acc_type, iif(ab_pair_kind = 1, null, acc_pair_acc_id),
         coalesce(sab_tax_kind, coalesce(ab_tax_kind, 0))
  from ref_account, ref_accbalance, sys_accbalance
  where acc_id = :kob_acc_kt and
        acc_from_ab_id = ab_id and
        acc_from_sab_id = sab_id
  into :acc_type, :pair_acc, :tax_kind_kt;

  -- если по Кредит идет парный счет, проверяем
  if (pair_acc is not null) then
  begin
    tmp_sum = 0;
    tmp_val = 0;
    tmp_tax = 0;
    select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
    from kn_obsummary
    where kobs_acc_id = :kob_acc_kt and
          kobs_date <= :kob_ol_date
    order by kobs_acc_id desc, kobs_date desc
    rows 1
    into tmp_sum, tmp_val, tmp_tax;

    -- если по этому счету нет остатка, проверяем дальше
    if (tmp_sum = 0 and
        tmp_val = 0 and
        tmp_tax = 0) then
    begin
      select kobs_bal_sum, kobs_bal_val, kobs_bal_tax
      from kn_obsummary
      where kobs_acc_id = :pair_acc and
            kobs_date <= :kob_ol_date
      order by kobs_acc_id desc, kobs_date desc
      rows 1
      into tmp_sum, tmp_val, tmp_tax;

      -- если прилетел активный счет, меняем его на пассивный (парный)
      if (acc_type = 0) then
      begin
        tmp_acc_id = kob_acc_kt;
        kob_acc_kt = pair_acc;
        pair_acc = tmp_acc_id;
        acc_type = 1;
      end
      else
      begin
        -- иначе проверяем, если у парного есть остаток, меняем Дт счет на парный.
        if (tmp_sum <> 0 or
            tmp_val <> 0 or
            tmp_tax <> 0) then
        begin
          tmp_acc_id = kob_acc_kt;
          kob_acc_kt = pair_acc;
          pair_acc = tmp_acc_id;
          acc_type = 1 - acc_type;
        end
      end
    end

    if (acc_type = 0) then
    begin
      if (tmp_sum - kob_summa < 0) then
      begin
        kt_pair_dt_acc = kob_acc_kt;
        kt_pair_kt_acc = pair_acc;
        kt_amount_sum = -(tmp_sum - kob_summa);
        kt_amount_val = -(tmp_val - kob_summa_val);
        kt_amount_tax = -(tmp_tax - kob_summa_tax_kt);
      end
    end

    if (acc_type = 1) then
    begin
      if (tmp_sum - kob_summa > 0) then
      begin
        kt_pair_dt_acc = pair_acc;
        kt_pair_kt_acc = kob_acc_kt;
        kt_amount_sum = (tmp_sum - kob_summa);
        kt_amount_val = (tmp_val - kob_summa_val);
        kt_amount_tax = (tmp_tax - kob_summa_tax_kt);
      end
    end
  end

  if (kob_detail_str <> '' and
      kob_detail_str similar to '[-]?[[:DIGIT:]]+') then
  begin
    kob_detail_id = kob_detail_str;
  end
  else
  begin
    kob_detail_id = null;
    select obd_id
    from kn_obdetail
    where obd_detail = :kob_detail_str
    into :kob_detail_id;

    if (kob_detail_id is null) then
    begin
      insert into kn_obdetail (obd_detail)
      values (:kob_detail_str)
      returning obd_id
      into :kob_detail_id;
    end
  end

  if (tax_kind_dt = 0) then
    kob_summa_tax_dt = kob_summa;
  if (tax_kind_dt = 1) then
    kob_summa_tax_dt = 0;

  if (tax_kind_kt = 0) then
    kob_summa_tax_kt = kob_summa;
  if (tax_kind_kt = 1) then
    kob_summa_tax_kt = 0;

--  if ((tax_kind_dt = 0 and
--      tax_kind_kt = 1) or
--      (tax_kind_dt = 1 and
--      tax_kind_kt = 0)) then
--    exception e_default 'Incorrect Dt and Kt tax amount kind. Dt: ' || tax_kind_dt || ', Kt: ' || tax_kind_kt;

  if (tax_kind_dt = 2 and
      tax_kind_kt in (0, 1)) then
    kob_summa_tax_dt = kob_summa_tax_kt;

  if (tax_kind_kt = 2 and
      tax_kind_dt in (0, 1)) then
    kob_summa_tax_kt = kob_summa_tax_dt;

  if (tax_kind_dt = 2 and
      tax_kind_kt = 2 and
      kob_summa <> 0) then
  begin
    if (kob_summa_tax_dt <> 0 and
        kob_summa_tax_dt <> kob_summa) then
      kob_summa_tax_dt = kob_summa;
    if (kob_summa_tax_kt <> 0 and
        kob_summa_tax_kt <> kob_summa) then
      kob_summa_tax_kt = kob_summa;
  end

  if (kob_summa <> 0 or
      kob_summa_val <> 0 or
      kob_summa_tax_dt <> 0 or
      kob_summa_tax_kt <> 0) then
  begin
    insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa,
                             kob_summa_val, kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id,
                             kob_ol_is_active, kob_detail_id, kob_client_dt, kob_client_kt, kob_from_kc_id)
    values (:kob_from_ol_id, :kob_currency, :kob_curr_rate, :kob_acc_dt, :kob_acc_kt, :kob_summa, :kob_summa_val,
            :kob_summa_tax_dt, :kob_summa_tax_kt, :kob_ol_date, :kob_ol_reg_id, :kob_ol_is_active, :kob_detail_id,
            :kob_client_dt, :kob_client_kt, :kob_from_kc_id);
  end

  if (dt_pair_dt_acc is not null and
      dt_pair_kt_acc is not null) then
  begin
    kob_detail_id = null;
    kob_detail_str = 'Кор-ка остатка парных счетов';
    select obd_id
    from kn_obdetail
    where obd_detail = :kob_detail_str
    into :kob_detail_id;

    if (kob_detail_id is null) then
    begin
      insert into kn_obdetail (obd_detail)
      values (:kob_detail_str)
      returning obd_id
      into :kob_detail_id;
    end

    if (tax_kind_dt = 0) then
      dt_amount_tax = dt_amount_sum;
    if (tax_kind_dt = 1) then
      dt_amount_tax = 0;

    if (dt_amount_sum <> 0 or
        dt_amount_val <> 0 or
        dt_amount_tax <> 0) then
    begin
      insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa,
                               kob_summa_val, kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id,
                               kob_ol_is_active, kob_detail_id, kob_client_dt, kob_client_kt, kob_from_kc_id)
      values (:kob_from_ol_id, :kob_currency, :kob_curr_rate, :dt_pair_dt_acc, :dt_pair_kt_acc, :dt_amount_sum,
              :dt_amount_val, :dt_amount_tax, :dt_amount_tax, -- это не ошибка, строится проводка между парными
              :kob_ol_date, :kob_ol_reg_id, :kob_ol_is_active, :kob_detail_id, null, null, :kob_from_kc_id);
    end

    if (tax_kind_kt = 0) then
      kt_amount_tax = kt_amount_sum;
    if (tax_kind_kt = 1) then
      kt_amount_tax = 0;

    if (kt_amount_sum <> 0 or
        kt_amount_val <> 0 or
        kt_amount_tax <> 0) then
    begin
      insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa,
                               kob_summa_val, kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id,
                               kob_ol_is_active, kob_detail_id, kob_client_dt, kob_client_kt, kob_from_kc_id)
      values (:kob_from_ol_id, :kob_currency, :kob_curr_rate, :kt_pair_dt_acc, :kt_pair_kt_acc, :kt_amount_sum,
              :kt_amount_val, :kt_amount_tax, -- это не ошибка, строится проводка между парными
              :kt_amount_tax, :kob_ol_date, :kob_ol_reg_id, :kob_ol_is_active, :kob_detail_id, null, null,
              :kob_from_kc_id);
    end
  end
