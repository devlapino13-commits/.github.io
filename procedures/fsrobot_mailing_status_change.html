CREATE PROCEDURE FSROBOT_MAILING_STATUS_CHANGE(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable cm_database_id integer;
declare variable kf_id decimal(18,0);
declare variable kf_name varchar(100);
declare variable lo_id decimal(18,0);
declare variable google_id varchar(255);
declare variable k_status integer;
declare variable k_id decimal(18,0);
declare variable str varchar(255);
declare variable k_status_2 integer;
declare variable rd_code d_str_100;
begin
-- Отправка PUSH СВЗ после регистрации заявки

-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь

-- СМС после выдачи кредита
  res_c_id = 0;
  if (user_fio is null) then
    user_fio = '';
  res_maillist = :rep_email;

  :cm_database_id = f_get_db_id();
  if (not(exists(select *
                 from refdata
                 where rd_code = 'Actiongroup.Check_Loan_Data' and
                       rd_is_active = 1))) then
    exit;

  res_body = '';

  select k_id, k_from_kf_id, k_status
  from kontract
  where k_id = :rep_params
  into :k_id, :kf_id, :k_status;
  if (kf_id is null) then
    exit;

  select kal_new_kf_name, kal_new_kf_officer
  from knaktlist
  where knaktlist.is_delete = 0 and
        kal_from_kf_id = :kf_id and
        kal_date_end >= '31.12.2999' and
        kal_new_kf_officer > 1000000000004
  into :kf_name, :lo_id;
  if (lo_id is null) then
    exit;
  res_c_id = lo_id;

  res_body = '';
  res_subject = '';

  select un_google_id
  from username
  where un_from_c_id = :lo_id rows 1
  into :google_id;
  if (coalesce(google_id, '') = '') then
    exit;
  res_maillist = google_id;

  str = (case k_status
           when -1 then iif(cm_database_id <> 94401, 'Отказан', 'İmtina olunub')
           when 0 then iif(cm_database_id <> 94401, 'В процессе', 'Prossesdə')
           when 1 then iif(cm_database_id <> 94401, 'Утвержден', 'Təsdiqlənib')
           when -2 then iif(cm_database_id <> 94401, 'Заявка', 'Ərizə')
         end);

  if (k_status = 0) then
  begin

    select rd_code
    from hp_get_check_loan_data(:k_id)
    into:rd_code;


    k_status_2 = null;
    select round(ovs_add_value, 0)
    from hp_objectprops_get('Kontract', :k_id, -1, :rd_code, null)
    into :k_status_2;

    if (k_status_2 is not null) then
    begin
      str = (case k_status_2
               when -2 then f_format('КК: %1', iif(cm_database_id <> 94401, 'Отказан', 'İmtina olunub'))
               when -1 then f_format('КК: %1', iif(cm_database_id <> 94401, 'Доработка', 'Tamamlanma'))
               when 0 then f_format('КК: %1', iif(cm_database_id <> 94401, 'В очереди', 'Növbədədir'))
               when 1 then f_format('КК: %1', iif(cm_database_id <> 94401, 'На рассмотрении', 'Yoxlanılır'))
               when 2 then f_format('КК: %1', iif(cm_database_id <> 94401, 'Подтвержден', 'Təsdiqlənib'))
             end);
    end
  end
  res_subject = iif(cm_database_id <> 94401, 'Статус ', 'Status') || kf_name || ': ' || str;
  res_body = iif(cm_database_id <> 94401, 'Новый статус заявления от группы ', 'Qrup ərizəsi üzrə yeni status ') || kf_name || ': ' || str;
  if (k_status = 1) then
    res_body = BLOB_APPEND(res_body, iif(cm_database_id <> 94401, '<br>Готов к выдаче.', '<br>Buraxılışa hazır'));
  suspend;
end
