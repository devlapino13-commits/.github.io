CREATE PROCEDURE AAREP_REGLIMITS(
    REP_REGION_ID INTEGER)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REG_BRANCH VARCHAR(100),
    RLM_DATE TYPE OF COLUMN REGLIMITS.RLM_DATE,
    RLM_LIMIT_CASH_MIN TYPE OF COLUMN REGLIMITS.RLM_LIMIT_CASH_MIN,
    RLM_LIMIT_CASH_MAX TYPE OF COLUMN REGLIMITS.RLM_LIMIT_CASH_MAX,
    RLM_LIMIT_BANK_MIN TYPE OF COLUMN REGLIMITS.RLM_LIMIT_BANK_MIN,
    RLM_LIMIT_BANK_MAX TYPE OF COLUMN REGLIMITS.RLM_LIMIT_BANK_MAX,
    RLM_BANK_DISTANCE TYPE OF COLUMN REGLIMITS.RLM_BANK_DISTANCE)
AS
begin
  for select id_list, r1.reg_code, r1.reg_name, r2.reg_name
      from region_list(:rep_region_id, 1, 3, 1), region r1, region r2
      where id_list = r1.reg_unicode and
            r1.parent = r2.reg_unicode
      into :reg_id, :reg_code, :reg_name, :reg_branch
  do
  begin
    rlm_date = null;
    rlm_limit_cash_min = 0;
    rlm_limit_cash_max = 0;
    rlm_limit_bank_min = 0;
    rlm_limit_bank_max = 0;
    rlm_bank_distance = 0;
    select rlm_date, rlm_limit_cash_min, rlm_limit_cash_max, rlm_limit_bank_min, rlm_limit_bank_max, rlm_bank_distance
    from reglimits
    where rlm_from_reg_id = :reg_id
    order by rlm_date desc, rlm_id desc
    rows 1
    into rlm_date, rlm_limit_cash_min, rlm_limit_cash_max, rlm_limit_bank_min, rlm_limit_bank_max, rlm_bank_distance;

    suspend;
  end
end
