CREATE PROCEDURE PRINT_FREELANCER_AKT(
    SSHE_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG)
RETURNS (
    NAME_FIO D_STR_100,
    PASS_NO VARCHAR(30),
    PASS_DATE D_DATE,
    PASS_WHO D_STR_255,
    FULL_ADRESS D_STR_255,
    INN D_STR_20,
    ADRESS_FACT D_STR_255,
    DATE_SIGN D_STR_20,
    QR_CODE D_STR_LONG,
    BC_ACCOUNT_NO VARCHAR(50),
    B_ID D_STR_10,
    B_NAME D_STR_255,
    DOGOVOR_NO D_STR_20,
    IS_COMPLITED D_SMALLINT,
    MEMO D_STR_255,
    AMOUNT_AKT D_CURRENCY,
    DOGOVOR_DATE D_STR_20,
    REG_NAME D_STR_100,
    ACT_DATE D_DATE,
    ACT_DATE_STR D_STR_20)
AS
declare variable SD_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable SIGN_STATUS D_SMALLINT;
declare variable EDL_ID D_ID;
declare variable DATE_SIGN_TMP D_DATE;
declare variable ACT_DATE_TMP D_STR_10;
begin
  select sshe.sshe_is_complited, sshe.sshe_memo, ssh.ssh_amount_dogovor, sd.sd_id, sd.sd_from_c_id, reg.reg_name, ssh.ssh_date, extract(month from ssh.ssh_date)
  from servicesh_edoc sshe
  inner join servicedog sd on sshe.sshe_from_sd_id = sd.sd_id
  inner join servicesh ssh on ssh.ssh_from_sd_id = sshe.sshe_from_sd_id and
        ssh.ssh_id = sshe.sshe_from_ssh_id
  inner join region reg on reg.reg_unicode = sd.sd_from_reg_id
  where sshe.sshe_id = :sshe_id
  into :is_complited, :memo, :amount_akt, :sd_id, :c_id, :reg_name, :act_date, act_date_tmp;

  if (is_complited is null) then
    is_complited = 0;
  if (memo is null) then
    memo = '';

  select name_fio, pass_no, pass_date, pass_who, full_adress, inn, adress_fact, bc_account_no, b_id, b_name, dogovor_no,
         date_sign
  from print_freelancer_kontract(:sd_id, null)
  into :name_fio, :pass_no, :pass_date, :pass_who, :full_adress, :inn, :adress_fact, :bc_account_no, :b_id, :b_name,
       :dogovor_no, :dogovor_date;

  select cast(eds.eds_status_datetime as date), eds.eds_status
  from edoc_list edl
  inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id and
        edc.edc_code = 'freelancer_akt'
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
  where edl.edl_object_kind = 0 and
        edl.edl_object_id = :c_id
  order by edl_status desc, edl_id desc
  rows 1
  into :date_sign_tmp, :sign_status;

  if (sign_status is null) then
    date_sign_tmp = null;

  if (date_sign_tmp is null) then
    date_sign = '';
  else
    date_sign = f_datetostr(date_sign_tmp);

  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(:add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(:edl_id, :c_id);
  end

  act_date_str = '';
  act_date_str = case
                    when act_date_tmp = '1' then 'Январь'
                    when act_date_tmp = '2' then 'Февраль'
                    when act_date_tmp = '3' then 'Март'
                    when act_date_tmp = '4' then 'Апрель'
                    when act_date_tmp = '5' then 'Май'
                    when act_date_tmp = '6' then 'Июнь'
                    when act_date_tmp = '7' then 'Июль'
                    when act_date_tmp = '8' then 'Август'
                    when act_date_tmp = '9' then 'Сентябрь'
                    when act_date_tmp = '10' then 'Октябрь'
                    when act_date_tmp = '11' then 'Ноябрь'
                    when act_date_tmp = '12' then 'Декабрь'
                    else ''
                 end;
  suspend;
end
