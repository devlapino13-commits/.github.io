CREATE PROCEDURE AAREP_COUNT_CC_KMP(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    COUNT_BLOCKED_VIPS_KMP D_INTEGER,
    COUNT_UNBLOCKED_VIPS_KMP D_INTEGER,
    COUNT_UNBLOCK_VIPS_CC D_INTEGER,
    COUNT_BLOCK_VIPS_CC D_INTEGER)
AS
begin
  --не заблокировано випсом
  count_unblocked_vips_kmp = null;
  select count(*)
  from extminiapp ex
  inner join ticket tk on tk.tkt_id = ex.ema_from_tkt_id
  inner join region rg on rg.reg_unicode = ex.ema_office_id
  inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(rg.reg_unicode as d_integer)
  where ex.ema_from_c_id is not null and
        ex.ema_date between :rep_date_begin and :rep_date_end and
        ex.ema_block_datetime is null and
        tk.tkt_source = 2
        and ex.ema_result_rd_id = 8786
  into count_unblocked_vips_kmp;

  --Заблокированные заявки
  count_blocked_vips_kmp = null;
  select count(*)
  from extminiapp ex
  inner join ticket tk on tk.tkt_id = ex.ema_from_tkt_id
  inner join region rg on rg.reg_unicode = ex.ema_office_id
  inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(rg.reg_unicode as d_integer)
  where ex.ema_from_c_id is not null and
        ex.ema_date between :rep_date_begin and :rep_date_end and
        ex.ema_block_datetime is not null and
        ex.ema_lo_id is not null  and
        tk.tkt_source = 2
        and ex.ema_result_rd_id = 8787
  into count_blocked_vips_kmp;


   ---    незаблокированные КЦ заявки
   select count(*)
   from CC_WEBSITE_MINIAPP CWA
   inner join ticket tk on tk.tkt_id = cwa.wma_from_tkt_id
   inner join extminiapp ex on ex.ema_from_tkt_id = tk.tkt_id
   inner join region rg on rg.reg_unicode = ex.ema_office_id
   inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(rg.reg_unicode as d_integer)
   where --cwa.wma_status = 2
          tk.tkt_source = 3
         and ex.ema_block_datetime is  null
         and cwa.wma_date between :rep_date_begin and :rep_date_end
         and ex.ema_result_rd_id = 8786
   into count_unblock_vips_cc;

   --- Заблокированные КЦ заявки
   select count(*)
   from CC_WEBSITE_MINIAPP CWA
   inner join ticket tk on tk.tkt_id = cwa.wma_from_tkt_id
   inner join extminiapp ex on ex.ema_from_tkt_id = tk.tkt_id
   inner join region rg on rg.reg_unicode = ex.ema_office_id
   inner join region_list(:rep_region_id, 1, 3, 1) t on t.id_list = cast(rg.reg_unicode as d_integer)
   where cwa.wma_status = 2
         and  tk.tkt_source = 3
         and ex.ema_block_datetime is not null
         and ex.ema_lo_id is not null
         and cwa.wma_date between :rep_date_begin and :rep_date_end
         and ex.ema_result_rd_id = 8787
   into count_block_vips_cc;



suspend;
end
