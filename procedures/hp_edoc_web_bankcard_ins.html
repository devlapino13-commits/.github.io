CREATE PROCEDURE HP_EDOC_WEB_BANKCARD_INS(
    CLIENT_ID D_ID)
RETURNS (
    RES_CODE D_INTEGER,
    RES_TEXT D_STR_255)
AS
declare variable edl_id d_id;
begin
  -- Создает edoc_list и отправляет клиенту ссылку, для ввода карточных данных
  res_code = 0;
  select edl_id, res_text
  from hp_edoc_ins_bankcard('bankcard.moneta', :client_id, 1000000000000)
  into edl_id, res_text;
  if (res_text <> '') then
    res_code = 1;
  else
  if (edl_id is null) then
  begin
    res_code = 1;
    res_text = 'Неизвестная ошибка';
  end
  else
    res_text = 'Мы отправили вам СМС/PUSH, в котором содержится ссылка и пароль для ввода данных карты';
  suspend;
end
