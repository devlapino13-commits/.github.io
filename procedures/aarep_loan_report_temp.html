CREATE PROCEDURE AAREP_LOAN_REPORT_TEMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    DISB_SUM_ALL CURR_TYPE,
    PERCENT_EFFECT_ALL CURR_TYPE,
    DISB_SUM_KLAS CURR_TYPE,
    PERCENT_EFFECT_KLAS CURR_TYPE,
    DISB_SUM_IPF CURR_TYPE,
    PERCENT_EFFECT_IPF CURR_TYPE,
    DISB_KOL_ALL INTEGER,
    DISB_KOL_KLASS INTEGER,
    DISB_KOL_IPF INTEGER,
    BALANCE_ALL CURR_TYPE,
    BAL_EFFECT_ALL CURR_TYPE,
    BALANCE_KLASS CURR_TYPE,
    BAL_EFFECT_KLASS CURR_TYPE,
    BALANCE_IPF CURR_TYPE,
    BAL_EFFECT_IPF CURR_TYPE)
AS
declare variable K integer;
declare variable ED_IZMER CURR_TYPE;
begin
  ed_izmer = 1000;
  if (rep_date_begin = rep_date_end) then
    k = 0;
  else
    k = 1;
  for select sum(k_amount_approved),
             sum(cast(k_amount_approved as double precision) * k_percent) / sum(k_amount_approved),
             sum(iif(pr_code not containing '.ipf.', k_amount_approved, 0)),
             sum(iif(pr_code not containing '.ipf.', cast(k_amount_approved as double precision) * k_percent, 0)) / sum(iif(pr_code not containing '.ipf.', k_amount_approved, 1)),
             sum(iif(pr_code containing '.ipf.', k_amount_approved, 0)),
             sum(iif(pr_code containing '.ipf.', cast(k_amount_approved as double precision) * k_percent, 0)) / sum(iif(pr_code containing '.ipf.', k_amount_approved, 1)),
             sum(k_kolich_client_approved), sum(iif(pr_code not containing '.ipf.', k_kolich_client_approved, 0)),
             sum(iif(pr_code containing '.ipf.', k_kolich_client_approved, 0))
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id
      into disb_sum_all, percent_effect_all, disb_sum_klas, percent_effect_klas, disb_sum_ipf, percent_effect_ipf,
           disb_kol_all, disb_kol_klass, disb_kol_ipf
  do
  begin
    disb_sum_all = disb_sum_all / ed_izmer;
    disb_sum_klas = disb_sum_klas / ed_izmer;
    disb_sum_ipf = disb_sum_ipf / ed_izmer;

    rep_date_begin = :rep_date_begin - :k;

    select sum(oks_balance),
           sum(cast(oks_balance as double precision) * f_get_interest_rate(k_id, :rep_date_begin, 'yearly')) / sum(oks_balance),
           sum(iif(pr_code not containing '.ipf.', oks_balance, 0)),
           sum(iif(pr_code not containing '.ipf.', cast(oks_balance as double precision) * f_get_interest_rate(k_id, :rep_date_begin, 'yearly'), 0)) / sum(iif(pr_code not containing '.ipf.', oks_balance, 0)),
           sum(iif(pr_code containing '.ipf.', oks_balance, 0)),
           sum(iif(pr_code containing '.ipf.', cast(oks_balance as double precision) * f_get_interest_rate(k_id, :rep_date_begin, 'yearly'), 0)) / sum(iif(pr_code containing '.ipf.', oks_balance, 0))
    from oksummary, kontract, product
    where :rep_date_begin between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_from_k_id = k_id and
          k_from_pr_id = pr_id
          and oks_close_state = 0
    into balance_all, bal_effect_all, balance_klass, bal_effect_klass, balance_ipf, bal_effect_ipf;

    balance_all = balance_all / ed_izmer;
    balance_klass = balance_klass / ed_izmer;
    balance_ipf = balance_ipf / ed_izmer;

    suspend;
  end
end
