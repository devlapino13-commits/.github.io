CREATE PROCEDURE HP_DOCPRLOAN_EDIT(
    PR_ID D_INTEGER,
    DP_ID D_INTEGER,
    CLIENT_KIND D_SMALLINT,
    DPL_SET_ID D_INTEGER,
    DPL_CHECKED D_INTEGER,
    DPL_DAYS D_INTEGER,
    UN_ID D_ID)
AS
declare variable dp_is_base d_bool;
begin
  if (dpl_set_id is null) then
    dpl_set_id = 0;

  if (dpl_days is null) then
    dpl_days = 0;

  select dp_is_base
  from docprop
  where dp_id = :dp_id
  into dp_is_base;

  if ((dp_is_base = 0 and dpl_checked = -1) or
      (dp_is_base = 1 and dpl_checked = 0)) then -- Аннулирование связи
  begin
    update docprloan
    set dpl_is_delete = 1,
        dpl_delete_un_id = :un_id
    where dpl_from_pr_id = :pr_id and
          dpl_from_dp_id = :dp_id and
          dpl_client_kind = :client_kind and
          dpl_set_id = :dpl_set_id and
          dpl_is_delete = 0;
  end else
  begin
    -- Добавление связи
    if (dpl_checked = 0) then
      dpl_days = 0;

    if (not exists(select * from docprloan
    where dpl_from_pr_id = :pr_id and
         dpl_from_dp_id = :dp_id and
         dpl_client_kind = :client_kind and
         dpl_set_id = :dpl_set_id and
         dpl_date_kind = :dpl_checked and
         dpl_date_days = :dpl_days and
         dpl_is_delete = 0)) then
    begin
      update docprloan
      set dpl_date_kind = :dpl_checked,
          dpl_date_days = :dpl_days
      where dpl_from_pr_id = :pr_id and
           dpl_from_dp_id = :dp_id and
           dpl_client_kind = :client_kind and
           dpl_set_id = :dpl_set_id and
           dpl_is_delete = 0;
  
      if (row_count = 0) then
      begin
        insert into docprloan (dpl_from_pr_id, dpl_from_dp_id, dpl_client_kind, dpl_set_id, dpl_date_kind, dpl_date_days, dpl_create_un_id)
        values (:pr_id, :dp_id, :client_kind, :dpl_set_id, :dpl_checked, :dpl_days, :un_id);
      end
    end
  end
end
