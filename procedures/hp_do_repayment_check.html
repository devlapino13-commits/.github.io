CREATE PROCEDURE HP_DO_REPAYMENT_CHECK(
    K_ID D_ID,
    PAY_DATE D_DATE,
    PAY_AMOUNT D_CURRENCY)
RETURNS (
    RES_ERROR_TEXT D_STR_255)
AS
declare variable CURR_NAME D_STR_CODE;
declare variable K_DISB_DATE D_DATE;
declare variable ZAD_KOMMIS D_CURRENCY;
declare variable NSP_PERC D_PERCENT;
begin
  -- Процедура проверяет возможность оплаты по договору
  -- Если дата погашения и выдачи совпадает, сравнивает сумму с отложенным доп.процентом

  res_error_text = '';

  if (f_get_const('repayment_on_disb_date') = 1) then
  begin
    suspend;
    exit;
  end

  select k_disb_date
  from kontract
  where k_id = :k_id
  into k_disb_date;

  if (pay_date = k_disb_date) then
  begin
    zad_kommis = 0;
    select sum(ok_kommis)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type <= 7
    into zad_kommis;
    if (zad_kommis is null) then
      zad_kommis = 0;

    if (zad_kommis > 0) then
    begin
      nsp_perc = 0;
      select nsp_percent_for_extra_int
      from hp_get_kontract_nsp_acc(1010100, :k_disb_date)
      into :nsp_perc;
  
      if (nsp_perc <> 0.0) then
      begin
        zad_kommis = zad_kommis / (100.00 - nsp_perc);
        zad_kommis = round(zad_kommis * 100.00, 0);
      end
    end

    if (pay_amount > zad_kommis) then
    begin
      select cm_currency_name
      from company
      into curr_name;

      if (zad_kommis = 0)
      then res_error_text = 'Запрещается погашение в день выдачи';
      else res_error_text = 'Запрещается погашение в день выдачи. Можно оплатить только отложенный доп.процент: ' || round(zad_kommis, 2) || ' ' || curr_name;
    end
  end

  suspend;
end
