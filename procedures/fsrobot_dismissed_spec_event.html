CREATE PROCEDURE FSROBOT_DISMISSED_SPEC_EVENT(
    UN_ID DECIMAL(18,0))
RETURNS (
    LO_ID DECIMAL(18,0),
    REG_ID INTEGER,
    LO_FIO VARCHAR(100),
    REG_NAME VARCHAR(100),
    PS_DATE_BEGIN DATE)
AS
declare variable rep_date_begin date;
declare variable rep_date_end3 date;
declare variable rep_date_end4 date;
declare variable ka_id decimal(18,0);
begin
  rep_date_begin = dateadd(month, -4, current_date);

  for with tab
      as (select distinct oks_region_id, oks_kc_id
          from oksummary
          where oks_date_end >= current_date and
                oks_date_begin <= current_date and
                oks_balance > 0 and
                oks_close_state = 0)

      select oks_region_id, oks_kc_id, c_name_fio, reg_code || ' ' || reg_name, ps_date_begin
      from tab, hr_pstatus, clients, region
      where ps_from_c_id = oks_kc_id and
            ps_from_c_id = c_id and
            oks_region_id = reg_unicode and
            ps_status = 1 and
            ps_date_begin between :rep_date_begin and current_date and
            ps_date_end >= current_date and
            ps_is_working = 0 and
            (not(ps_from_c_id in ('1010100004163', '1010100231589', '1010100073400', '1010100071855', '1010100222482', '1010100015996', '1010100003405') and
            ps_date_begin <= '30.06.2015'))
      into :reg_id, :lo_id, :lo_fio, :reg_name, :ps_date_begin
  do
  begin
    rep_date_end3 = ps_date_begin + 2;
    if (extract(weekday from rep_date_end3) = 6) then
      rep_date_end3 = dateadd(day, 2, rep_date_end3);
    if (extract(weekday from rep_date_end3) = 0) then
      rep_date_end3 = dateadd(day, 1, rep_date_end3);

    rep_date_end4 = ps_date_begin + 3;
    if (extract(weekday from rep_date_end4) = 6) then
      rep_date_end4 = dateadd(day, 2, rep_date_end4);
    if (extract(weekday from rep_date_end4) = 0) then
      rep_date_end4 = dateadd(day, 1, rep_date_end4);

    if (rep_date_end3 <= current_date) then -- На третий день отправляем уведомление коллегам что завтра будет перекинута
    begin
      execute procedure fsrobot_insert_event('DoMailing', 'LP_Dismissed_LO_3_day', lo_id, :un_id);
-- связка к процедуре FSROBOT_Dismissed_LO_3_day
    end
    if (rep_date_end4 <= current_date) then -- На четвертый день перекидываем, отправляем СМС и уведомляем коллег что перекинута
    begin
      for select ka_id
          from hp_transfer_lp_to_other(current_date, :lo_id, 1000000000004, 'Передача КП уволенного сотрудника ' || :lo_fio || ' (' || :reg_name || ') на КЦ "Сирена"')
          into :ka_id
      do
      begin
        execute procedure fsrobot_insert_event('DoMailing', 'LP_Transfer_LO_4_day', ka_id, :un_id);
        execute procedure fsrobot_insert_event('DoMailing', 'LP_Transfer_LO_Client_SMS', ka_id, :un_id);
      end
    end
    suspend;
  end
end
