CREATE PROCEDURE AAREP_CLIENTDOCSCHECK_HEADER(
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    NUMBER_OF_ORDER D_INTEGER NOT NULL,
    EMPLOYEE_DIGIT_NAME TYPE OF COLUMN USERNAME.UN_FIO NOT NULL,
    EMPLOYEE_DIGIT_ID TYPE OF COLUMN USERNAME.UN_ID NOT NULL,
    NUMBER_OF_CLIENTS_DOCUMENTS D_INTEGER NOT NULL,
    NUMBER_OF_CLIENTS D_INTEGER NOT NULL,
    NUMBER_OF_EDITED_DOCUMENTS D_INTEGER NOT NULL,
    AVERAGE_CLIENT_SEC D_INTEGER NOT NULL,
    AVERAGE_DOCUMENT_SEC D_INTEGER NOT NULL,
    AVERAGE_EDITED_DOCUMENT_SEC D_INTEGER NOT NULL,
    AVERAGE_CLIENT D_DOUBLE NOT NULL,
    AVERAGE_DOCUMENT D_DOUBLE NOT NULL,
    AVERAGE_EDITED_DOCUMENT D_DOUBLE NOT NULL)
AS
begin
  for
    select
        row_number() over ()                                                                                                as number_of_order,
        (select first 1 u.un_fio from username u where (u.un_id = q.employee_digit_id))                                     as employee_digit_name,
        q.employee_digit_id,
        q.number_of_clients,
        q.number_of_client_documents,
        q.number_of_edited_documents,

        iif(q.number_of_clients=0, 0,          round(q.time_processing_client_sec/q.number_of_clients))                     as average_client_sec,
        iif(q.number_of_client_documents=0, 0, round(q.time_processing_client_sec/q.number_of_client_documents))            as average_document_sec,
        iif(q.number_of_edited_documents=0, 0, round(q.time_processing_client_sec/q.number_of_edited_documents))            as average_edited_document_sec,

        iif(q.number_of_clients=0, 0,          (q.time_processing_client_sec/q.number_of_clients)/86400.000000000)          as average_client,
        iif(q.number_of_client_documents=0, 0, (q.time_processing_client_sec/q.number_of_client_documents)/86400.000000000) as average_document,
        iif(q.number_of_edited_documents=0, 0, (q.time_processing_client_sec/q.number_of_edited_documents)/86400.000000000) as average_edited_document
      from
        (
          select
              edc.employee_digit_id,
              coalesce(count(edc.client_id), 0)                as number_of_clients,                -- количество досье
              coalesce(sum(edc.number_of_client_documents), 0) as number_of_client_documents,
              coalesce(sum(edc.number_of_edited_documents), 0) as number_of_edited_documents,
              coalesce(sum(edc.time_processing_client_sec), 0) as time_processing_client_sec
            from
              (
                with client_doc_check_count as (
                  select
                      cdc.dcl_c_id as client_id,
                      count(*)     as doc_check_count,
                      (select count(*) from clientdocs cd where cd.cds_from_c_id = cdc.dcl_c_id) as doc_count
                    from
                      clientdocscheck_log cdc
                    where
                      (cdc.dcl_date_check between :report_date_begin and :report_date_end)
                    group by
                      cdc.dcl_c_id
                )
                select
                    cqc.cq_un_id                                                                                  as employee_digit_id,
                    cqc.cq_c_id                                                                                   as client_id,
                    cdcc.doc_count                                                                                as number_of_client_documents,
                    cdcc.doc_check_count                                                                          as number_of_edited_documents,
                    datediff(second, cqc.cq_date_start_check, coalesce(cqc.cq_date_end_check, current_timestamp)) as time_processing_client_sec
                  from
                    clientqueuecheck   as cqc
                    left join client_doc_check_count as cdcc
                      on (cdcc.client_id = cqc.cq_c_id)
                  where
                   (    (cqc.cq_date_start_check between :report_date_begin and :report_date_end)
                     or (coalesce(cqc.cq_date_end_check, current_timestamp) between :report_date_begin and :report_date_end))
                  group by
                    cqc.cq_un_id,
                    cqc.cq_c_id,
                    cqc.cq_date_start_check,
                    cqc.cq_date_end_check,
                    cdcc.doc_count,
                    cdcc.doc_check_count
               ) as edc
            group by
              edc.employee_digit_id
        ) as q
      into
        :number_of_order,
        :employee_digit_name,
        :employee_digit_id,
        :number_of_clients,
        :number_of_clients_documents,
        :number_of_edited_documents,
        :average_client_sec,
        :average_document_sec,
        :average_edited_document_sec,
        :average_client,
        :average_document,
        :average_edited_document
    do
      suspend;
end
