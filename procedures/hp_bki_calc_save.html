CREATE PROCEDURE HP_BKI_CALC_SAVE(
    MAIN_NODE D_STR_CODE,
    XML_VERSION_ID D_ID,
    OBJ_KIND D_SMALLINT,
    OBJ_ID D_ID,
    JSO D_INTEGER,
    ACTION_CODE D_STR_10 = null,
    EVENT_DATETIME D_DATETIME = null,
    EVENT_TIMEZONE D_INTEGER = null,
    DONT_USE_DEF D_BOOL = 0)
AS
declare variable var_value d_str_8100;
declare variable tmp_value d_str_8100;
declare variable bfv_id d_id;
declare variable db_field d_str_code;
declare variable i d_integer;
begin
  -- если  event_timezone = 777, то используется временная зона 00 и сдвиг относительно МСК не производится

  dont_use_def = coalesce(dont_use_def, 0);

  for select bfv_id, db_field, i, var_value
  from hp_bki_get_valuelist4event(:main_node, :xml_version_id, :jso)
  into bfv_id, db_field, i, tmp_value
  do
  begin
    if (dont_use_def = 1) then
      var_value = iif(db_field is null or db_field = 'deal_sign_credit', null, tmp_value);
    else if (db_field = 'action_code') then
      var_value = action_code;
    else if (db_field = 'event_date') then
      var_value = event_datetime;
    else if (db_field = 'event_datetime') then
      var_value = f_bki_datetime_to_str(event_datetime, iif(event_timezone = 777, 0, event_timezone), iif(event_timezone = 777, 0, 1));
    else
      var_value = tmp_value;

    if (var_value is not null) then
      insert into bki_send_data_oper (bsd_from_bfv_id, bsd_obj_kind, bsd_obj_id, bsd_from_bkd_id, bsd_number)
      values (:bfv_id, :obj_kind, :obj_id, f_bki_data_id(:var_value), :i);
  end
end
