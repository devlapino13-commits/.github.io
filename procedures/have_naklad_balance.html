CREATE PROCEDURE HAVE_NAKLAD_BALANCE
RETURNS (
    MBAL_OL_ID DECIMAL(18,0))
AS
declare variable mbal_yediz varchar(20);
declare variable mbal_memo varchar(1024);
declare variable mbal_docno integer;
declare variable mbal_date date;
declare variable mbal_ml_name varchar(255);
declare variable mbal_summa decimal(18,4);
declare variable mbal_kolich decimal(18,4);
declare variable mbal_ml_id decimal(18,0);
declare variable mbal_line_no integer;
begin
  mbal_ol_id = 0;
  mbal_summa = 0;
  mbal_kolich = 0;
  mbal_line_no = 0;
  mbal_ml_id = 0;
  mbal_ml_name = '';

  for select on_naklad_ol_id, on_naklad_number, on_from_ml_id, ml_name, yl_code, o2.ol_date, o2.ol_doc_no, o2.ol_memo,
             sum(on_kolich_plus - on_kolich_minus), sum(on_summa_plus - on_summa_minus)
      from operlist o1, matlist, yedizlist, opernak
      left join operlist o2 on (on_naklad_ol_id = o2.ol_id)
      where o1.ol_id = on_from_ol_id and
            ml_from_yl_id = yl_id and
            o1.is_delete = 0 and
            on_from_ml_id = ml_id
      group by on_naklad_ol_id, on_naklad_number, on_from_ml_id, ml_name, o2.ol_date, o2.ol_doc_no, o2.ol_memo, yl_code
      having sum(on_kolich_plus - on_kolich_minus) <> 0 or sum(on_summa_plus - on_summa_minus) <> 0
      order by on_naklad_ol_id, ml_name
      rows 1
      into :mbal_ol_id, :mbal_line_no, :mbal_ml_id, :mbal_ml_name, :mbal_yediz, :mbal_date, :mbal_docno, :mbal_memo,
           :mbal_kolich, :mbal_summa
  do
  begin
    suspend;
  end
end
