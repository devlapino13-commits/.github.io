CREATE PROCEDURE HP_EXTINFO_STOP_CR_PROCESS(
    EIL_ID D_ID)
AS
declare variable OBJ_KIND D_ID;
declare variable OBJ_ID D_ID;
declare variable MA_ID D_ID;
declare variable C_ID D_ID;
declare variable STATUS D_INTEGER;
begin
  select eil.eil_object_kind, eil.eil_object_id, eil.eil_from_c_id
  from extinfolist eil
  where eil.eil_id = :eil_id
  into :obj_kind, :obj_id, :c_id;

  if (obj_kind <> f_objkind('kontract', 'id')) then  -- если запрос был не на уровне контракта, то не отпраляем в АФМ
    exit;

  select status
  from hp_stop_credit_query_status(:obj_kind, :obj_id)
  into :status;

  if (status = 0 or cast(f_is_stopcred_doc_loaded(:obj_kind, :obj_id) as d_integer) = 1) then  -- если нет самозапрета или приереплен документ
  begin
    select kn.k_from_ma_app_id
    from kontract kn
    where kn.k_id = :obj_id
    into :ma_id;

    execute procedure hp_extinfo_insert_afm(:c_id, :ma_id);
  end
end
