CREATE PROCEDURE HP_BLOBSTORE_INSERT(
    BLS_FILENAME TYPE OF COLUMN BLOBSTORE.BLS_FILENAME,
    BLS_CAPTION TYPE OF COLUMN BLOBSTORE.BLS_CAPTION,
    BLS_BLOB_SIZE TYPE OF COLUMN BLOBSTORE.BLS_BLOB_SIZE,
    BLS_FROM_UN_ID TYPE OF COLUMN BLOBSTORE.BLS_FROM_UN_ID,
    BLS_KIND INTEGER,
    BLS_KEY_ID DECIMAL(18,0),
    BLS_KEY_2_ID INTEGER,
    BLS_BLC_ID INTEGER)
RETURNS (
    BLS_ID DECIMAL(18,0),
    KEYFIELD1_VALUE DECIMAL(18,0),
    KEYFIELD2_VALUE DECIMAL(18,0),
    KEYFIELD3_VALUE DECIMAL(18,0),
    BLOB_BLS_ID DECIMAL(18,0),
    KEYFIELD1_NAME VARCHAR(100),
    KEYFIELD2_NAME VARCHAR(100),
    KEYFIELD3_NAME VARCHAR(100),
    BLC_TABLE_NAME VARCHAR(100),
    BLC_BLOB_FIELDNAME VARCHAR(100),
    BLC_LINK_FIELD VARCHAR(100),
    SQL_TEXT VARCHAR(1024))
AS
declare variable key_fields_count integer;
begin
  if (bls_key_2_id is null) then
    bls_key_2_id = -1;

  bls_id = null;
  insert into blobstore (bls_filename, bls_caption, bls_blob_size, bls_from_un_id, bls_kind)
  values (:bls_filename, :bls_caption, :bls_blob_size, :bls_from_un_id, :bls_kind)
  returning bls_id
  into :bls_id;

  if (bls_key_id <> 0) then
  begin
    select keyfield1_value, keyfield2_value, keyfield3_value, blob_bls_id, upper(keyfield1_name), upper(keyfield2_name),
           upper(keyfield3_name), upper(blc_table_name), upper(blc_blob_fieldname), upper(blc_link_field)
    from aarep_blob_list(:bls_key_id, :bls_blc_id)
    rows 1
    into :keyfield1_value, :keyfield2_value, :keyfield3_value, :blob_bls_id, :keyfield1_name, :keyfield2_name,
         :keyfield3_name, :blc_table_name, :blc_blob_fieldname, :blc_link_field;
    key_fields_count = 0;
    if (keyfield3_name <> '') then
      key_fields_count = 3;
    else
    if (keyfield2_name <> '') then
      key_fields_count = 2;
    else
    if (keyfield1_name <> '') then
      key_fields_count = 1;
    if ((keyfield2_name = '') and
        (blc_link_field = keyfield1_name)) then
    begin
      sql_text = 'update ' || :blc_table_name || ' set ' || :blc_blob_fieldname || '=' || :bls_id || ' where ' || :blc_link_field || '=' || :bls_key_id;
    end
    else
    begin
      if (key_fields_count = 3) then
      begin
        if (blc_link_field = keyfield1_name) then
        begin
          sql_text = 'insert into ' || :blc_table_name || ' (' || :keyfield1_name || ', ' || :keyfield2_name || ', ' || :keyfield3_name || ', ' || :blc_blob_fieldname || ')' || ' values (' || :bls_key_id || ', -1, -1,' || :bls_id || ')';
        end
        else
        begin
          sql_text = 'insert into ' || :blc_table_name || ' (' || :keyfield1_name || ', ' || :keyfield2_name || ', ' || :keyfield3_name || ', ' || :blc_link_field || ', ' || :blc_blob_fieldname || ')' || ' values (-1, -1, -1,' || :bls_key_id || ', ' || :bls_id || ')';
        end
      end
      if (key_fields_count = 2) then
      begin
        if (blc_link_field = keyfield1_name) then
        begin
          sql_text = 'update or insert into ' || :blc_table_name || ' (' || :keyfield1_name || ', ' || :keyfield2_name || ', ' || :blc_blob_fieldname || ')' || ' values (' || :bls_key_id || ', ' || :bls_key_2_id || ', ' || :bls_id || ') matching (' || :keyfield1_name || ', ' || :keyfield2_name || ')';
        end
        else
        begin
          sql_text = 'update or insert into ' || :blc_table_name || ' (' || :keyfield1_name || ', ' || :keyfield2_name || ', ' || :blc_link_field || ', ' || :blc_blob_fieldname || ')' || ' values (-1, -1, ' || :bls_key_id || ', ' || :bls_id || ') matching (' || :keyfield1_name || ', ' || :keyfield2_name || ')';
        end
      end
      if (key_fields_count = 1) then
      begin
        if (blc_link_field <> keyfield1_name) then
        begin
          sql_text = 'insert into ' || :blc_table_name || ' (' || :keyfield1_name || ', ' || :blc_link_field || ', ' || :blc_blob_fieldname || ')' || ' values (-1, ' || :bls_key_id || ', ' || :bls_id || ')';
        end
      end
    end
    execute statement :sql_text;
  end
  suspend;
end
