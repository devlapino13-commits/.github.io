CREATE PROCEDURE CM_KG_0019_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_client_pass_no varchar(30);
declare variable kc_from_c_id id_type;
declare variable c_citizen integer;
declare variable rd_code varchar(255);
declare variable c_pass_date date;
declare variable c_pass_date_expire date;
declare variable k_percent_for_days integer;
declare variable pr_percent_for_days integer;
declare variable db_id integer;
declare variable pr_code d_str_100;
begin
  select com.cm_database_id
  from company com
  into :db_id;
  for select kc_from_c_id,
             kc_amount_approved,
             coalesce(kont.k_percent_for_days, -777),
             coalesce(pr.pr_percent_for_days,-777),
             pr.pr_code
  from knclient kncl
  inner join kontract kont on kont.k_id = kncl.kc_from_k_id
  inner join product pr    on pr.pr_id  = kont.k_from_pr_id
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  into kc_from_c_id, kc_amount, k_percent_for_days, pr_percent_for_days, pr_code
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;

    if (k_percent_for_days <> pr_percent_for_days) then
    begin
        res_error_text = 'Неправильное значение ежедневной ставки. Просим заново перестроить график';
        res_error_is_critical = 1;
        suspend;
    end
    c_citizen = null;
    rd_code   = null;
    select c_name_fio,
           c_citizen,
           c_pass_no,
           C_PASS_DATE,
           C_PASS_DATE_EXPIRE
    from clients
    where c_id = :kc_from_c_id
    into res_info_text,
         c_citizen,
         KF_CLIENT_PASS_NO,
         C_PASS_DATE,
         C_PASS_DATE_EXPIRE;

    select rd_code from refdata
    where rd_id = :c_citizen
    into rd_code;
    if (db_id <> 94401) then
    begin
        if (rd_code not in ('Citizen.RU','Citizen.KG','Citizen.TJ','Citizen.UZ')) then
        begin
          res_error_text = 'Клиент не является гражданином след. республик: РФ, КГ, ТЖ и УЗ';
          res_error_is_critical = 1;
          suspend;
        end
    end
    if (C_PASS_DATE < cast('01.01.1990' as date) or C_PASS_DATE > current_date) then
    begin
      res_error_text = 'Дата выдачи паспорта, указана не верно!';
      res_error_is_critical = 1;
      suspend;
    end
    if (db_id = 81001) then
    begin
        if (current_date > dateadd(month, -6, :C_PASS_DATE_EXPIRE)) then
        begin
          res_error_text = 'Срок паспорта заканчивается в течении 6 месяцев';
          res_error_is_critical = 1;
          suspend;
        end
    end
  end
end
