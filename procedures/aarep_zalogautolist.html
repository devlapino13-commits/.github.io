CREATE PROCEDURE AAREP_ZALOGAUTOLIST(
    IN_ZAP_ID D_INTEGER)
RETURNS (
    ZAL_ID D_INTEGER,
    ZAL_FROM_ZAP_ID D_INTEGER,
    ZAL_COST D_CURRENCY,
    ZAL_WEBPAGE D_STR_LONG,
    ZAL_STATUS_STR D_STR_CODE,
    ZAL_ADD_PARAMS D_STR_8100,
    ZAL_IS_DELETE D_SMALLINT,
    ZAL_FROM_UN_ID D_ID,
    ZAL_DATETIME D_DATETIME,
    ZAL_HAS_PHOTO D_BOOL,
    ZAB_FROM_BLS_ID D_ID,
    USER_NAME D_STR_CODE)
AS
declare variable zal_status d_smallint;
begin
  for select zal_id,
             zal_from_zap_id,
             zal_cost,
             zal_webpage,
             zal_status,
             zal_add_params,
             zal_is_delete,
             zal_from_un_id,
             zal_datetime
      from zalogautolist
      where zal_from_zap_id = :in_zap_id and
            zal_is_delete = 0
      order by zal_id
      into :zal_id,
           :zal_from_zap_id,
           :zal_cost,
           :zal_webpage,
           :zal_status,
           :zal_add_params,
           :zal_is_delete,
           :zal_from_un_id,
           :zal_datetime
  do
  begin
    zal_status_str = trim(case zal_status
      when -2 then 'Ошибка'
      when 0 then 'Новый'
      when 1 then 'В очереди'
      when 2 then 'Получен'
      else '-'
    end);

    if (coalesce(zal_webpage, '') = '' and zal_status = 2) then
      zal_status_str = '-';

    zal_has_photo = 0;
    zab_from_bls_id = null;

    select zab_from_bls_id
    from zalogautoblob
    where zab_from_zal_id = :zal_id and
          zab_from_bls_id is not null
    order by zab_id desc
    rows 1
    into :zab_from_bls_id;

    if (zab_from_bls_id is not null) then
      zal_has_photo = 1;

    user_name = '';

    if (zal_from_un_id is not null) then
    begin
      select un_name
      from username
      where un_id = :zal_from_un_id
      into user_name;
    end

    suspend;
  end
end
