CREATE PROCEDURE HP_DO_KN_OKSUMMARY_FIX_ZERO(
    K_ID ID_TYPE)
AS
declare variable kc_id decimal(18,0);
declare variable kc_amount curr_type;
declare variable last_ol_id id_type;
declare variable last_ol_number integer;
declare variable last_date date;
declare variable oks_date_begin date;
declare variable gr_amount curr_type;
declare variable cl_count integer;
declare variable cl_balance curr_type;
declare variable new_balance curr_type;
declare variable koef double precision;
declare variable saldo curr_type;
declare variable line_no integer;
declare variable gr_balance curr_type;
declare variable try_count integer;
declare variable is_changed integer;
begin
  try_count = 0;
  is_changed = 1;

--  if (exists(select oks_balance
--  from oksummary
--  where oks_date_end >= '31.12.2999'
--  and oks_from_k_id = :k_id
--  and oks_balance >0)) then Exit;
--
  while (is_changed = 1 and try_count <= 10) do
  begin
    is_changed = 0;
    try_count = try_count + 1;

    oks_date_begin = null;

    select oks_date_begin, sum(oks_balance)
    from kn_oksummary
    where oks_from_k_id = :k_id
    group by oks_date_begin
    having((sum(oks_balance) > 0 and min(oks_balance) <= 0) or (sum(oks_balance) = 0 and min(oks_balance) < 0))
    order by 1
    rows 1
    into oks_date_begin, gr_balance;

    if (oks_date_begin is null) then
      exit;

    last_ol_id = null;
    select ok_from_ol_id, ok_number, ok_ol_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_date = :oks_date_begin and
          ok_os <> 0 and
          ok_ol_is_active = 1 and
          ok_type in (4, 5, 6)
    order by ok_ol_date desc, ok_from_ol_id desc
    rows 1
    into last_ol_id, last_ol_number, last_date;

    if (last_ol_id is not null) then
    begin
      select sum(kc_amount_approved), count(*)
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into gr_amount, cl_count;

      if (coalesce(gr_amount, 0) = 0) then
        exit;

      saldo = gr_balance;
      line_no = 0;
      for select oks_from_kc_id, oks_balance, kc_amount_approved
          from kn_oksummary, knclient
          where oks_from_k_id = :k_id and
                oks_date_begin = :last_date and
                oks_from_kc_id = kc_id
          into :kc_id, :cl_balance, :kc_amount
      do
      begin
        line_no = line_no + 1;
        if (line_no < cl_count) then
        begin
          koef = kc_amount / gr_amount;
          new_balance = round(gr_balance * koef, 2);
          saldo = saldo - new_balance;
        end
        else
          new_balance = saldo;

        update kn_operkred
        set ok_os = ok_os - (:cl_balance - :new_balance)
        where ok_from_ol_id = :last_ol_id and
              ok_number = :last_ol_number and
              ok_from_kc_id = :kc_id;

        is_changed = 1;
      end
    end
  end
end
