CREATE PROCEDURE HP_SET_SESSION_USER(
    USER_NAME D_STR_CODE)
RETURNS (
    UN_ID D_ID,
    DB_IS_ACTUAL D_BOOL,
    DB_ID D_INTEGER,
    REPORT_DB_PATH D_STR_255,
    UN_LANG_CODE D_STR_10)
AS
begin
  if (coalesce(user_name, '') = '') then
    Exit;

  execute procedure uxp$set_session_username(:user_name);

  db_id = uxf$db_id();
  un_id = uxf$user_id(user_name, false);

  un_lang_code = '';
  if (un_id is not null) then
  begin
    select un_lang_code
    from username
    where un_id = :un_id
    into un_lang_code;
  end

  report_db_path = '';
  if (uxf$db_actual()) then
  begin
    db_is_actual = 1;
    report_db_path = uxf$rep_db();
  end
  else
    db_is_actual = 0;

  suspend;
end
