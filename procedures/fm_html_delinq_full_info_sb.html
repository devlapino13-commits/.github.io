CREATE PROCEDURE FM_HTML_DELINQ_FULL_INFO_SB(
    UN_ID DECIMAL(18,0),
    DATE_END DATE,
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0))
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable oks_close_state d_smallint;
declare variable kroff d_currency;
declare variable extra d_currency;
declare variable delinq_sum_all decimal(18,2);
declare variable k_kontract_no d_str_20;
declare variable k_fact_disb_date date;
declare variable c_name_fio d_str_100;
declare variable cm_database_id d_id;
declare variable anketa_code d_str_100;
declare variable db_id d_id;
declare variable res_field_name varchar(255);
declare variable c_id d_id;
begin

  db_id = f_get_db_id();

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  res_html = (select res_style
              from fm_html_styles) || '<body>';
  anketa_code = 'anketa.delinq.1';

  if (obj_kind = 'clients') then
  begin

  end
  if (obj_kind = 'kontract') then
  begin
    delinq_sum_all = 0;

    select k_kontract_no, k.k_fact_disb_date
    from kontract k
    where k.k_id = :obj_id rows 1
    into :k_kontract_no, k_fact_disb_date;

    --res_text = BLOB_APPEND(res_text, f_format('<font color="red">%1.</font><br><br>', iif(:db_id <> 94401, 'Ограниченный режим просмотра', 'Ограниченный режим просмотра')));

    select oks.oks_close_state,
           (oks.oks_out_os + oks.oks_out_perc + oks.oks_out_other), oks.oks_extra
    from oksummary oks
    where current_date between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_from_k_id = :obj_id
    into :oks_close_state, kroff, extra;

    if (oks_close_state = 0) then
    begin

      select coalesce(delinq_sum_all, 0)
      from print_kont_delinq_zadol(:obj_id)
      into :delinq_sum_all;
    end

    if (oks_close_state = 1) then
    begin
      delinq_sum_all = kroff + extra;
      delinq_sum_all = round(delinq_sum_all, 0);

    end
    delinq_sum_all = round(delinq_sum_all, 0);

    res_html = BLOB_APPEND(res_html, '<h1 class="green_text">' || coalesce(k_kontract_no, '') || ' от ' || f_datetostr(k_fact_disb_date) || '</h1><br>');

    for select cl.c_name_fio, cl.c_id
        from knclient knc
        inner join clients cl on cl.c_id = knc.kc_from_c_id
        where knc.kc_from_k_id = :obj_id and
              knc.kc_amount_approved > 0
        into :c_name_fio, :c_id
    do
    begin
      res_field_name = f_uri_open(left(c_name_fio, 50), 'client', 'Clients', :c_id, left(c_name_fio, 50), '', :un_id);
      res_html = BLOB_APPEND(res_html, f_format('Участник: <b> %1', res_field_name) || '</b><br>');
    end

    res_html = BLOB_APPEND(res_html, f_format('Общий долг по контракту: <b> %1', delinq_sum_all) || '</b><br>');
  end

  res_html = BLOB_APPEND(res_html, '</body>');
  suspend;
end
