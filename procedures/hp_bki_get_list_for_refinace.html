CREATE PROCEDURE HP_BKI_GET_LIST_FOR_REFINACE(
    C_ID D_ID,
    START_DATE_IN D_DATE = null,
    IS_NO_EXIT D_BOOL = 0,
    INCLUDE_NBKI_KO D_BOOL = null)
RETURNS (
    UUID D_STR_CODE,
    CRED_DATE D_DATE,
    CRED_SUM D_CURRENCY,
    CRED_PARTNER_TYPE_STR D_STR_CODE,
    CRED_DEBT_SUM D_CURRENCY,
    SSP_SUMM D_CURRENCY,
    ERROR_TEXT D_STR_100,
    CRED_SUM_OVERDUE D_CURRENCY)
AS
begin
  if (is_no_exit is null) then
    is_no_exit = 0;

  for select uuid, cred_date, cred_sum, cred_partner_type_str, cred_debt_sum, ssp_summ, error_text, cred_sum_overdue
      from hp_bki_get_list_from_ko(:c_id, current_date, :start_date_in, :include_nbki_ko)
      where is_our = 0 and
            is_in_ssp = 1 and
            is_in_ko + :is_no_exit > 0 and
            ssp_summ > 0 and
            coalesce(cred_debt_sum, 1) > 0 and
            cred_partner_type_str in ('МФО', 'MKK', 'MFK', 'MFO', 'NO_MFO', 'KPK')
      into uuid, cred_date, cred_sum, cred_partner_type_str, cred_debt_sum, ssp_summ, error_text, cred_sum_overdue
  do
    suspend;
end
