CREATE PROCEDURE HP_CONTINFO_CAN_CHANGE(
    C_ID D_ID,
    CONT_KIND D_STR_CODE,
    CONT_VALUE D_STR_100,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255,
    FOUND_C_NAME D_STR_255,
    CDA_DATE D_DATE,
    FOUND_C_ID D_ID,
    C_BIRTH_DATE D_DATE,
    CLIENT_IS_ACTIVE D_SMALLINT)
AS
declare variable CLT_ID D_INTEGER;
declare variable CONT_IS_MAIN D_SMALLINT;
declare variable CONT_KIND_CODE D_STR_CODE;
declare variable HAS_MINIAPP D_SMALLINT;
declare variable ACCESS_ALLOWED D_SMALLINT;
declare variable DB_ID D_INTEGER;
begin
  res_is_error = 0;
  res_text = '';

  select cont_kind_code, cont_is_main
  from hp_contkind_info(:cont_kind)
  into :cont_kind_code, :cont_is_main;

  if (not(cont_kind_code = 'cell' and
      cont_is_main = 1)) then
    exit;

  if (f_cont(c_id, cont_kind_code, 'last', 'main') = '') then
    exit;

  access_allowed = 0;
  if (exists(select has_access
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
             where has_access = 1)) then
    access_allowed = 1;

  client_is_active = 0;
  db_id = f_get_db_id();

  if (db_id <> 81001) then
  begin
    if (exists(select *
               from kontstate, knclient, kontract
               where kc_from_c_id = :c_id and
                     kc_amount_approved > 0 and
                     kc_from_k_id = ks_from_k_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0 and
                     ks_date_end >= '31.12.2999' and
                     kc_from_k_id = k_id and
                     k_stop_perc is null)) then
      client_is_active = 1;
  end

  if (cont_value = '') then
  begin
    if (access_allowed = 0 and
        exists(select has_access
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Access_Delete_Active_Cell')
               where has_access = 1)) then
      access_allowed = 1;

    if (client_is_active = 1) then
    begin
      res_text = 'У текущего клиента имеется активный займ';

      if (access_allowed = 0) then
      begin
        res_text = 'У вас нет прав на удаление номера активного клиента';
        res_is_error = 1;
      end

      suspend;
    end
    exit;
  end

  if (db_id not in (39801, 41701) and
      access_allowed = 0 and
      exists(select has_access
             from hp_permission_access_check(:un_id, 'MenuList', 0, 'mEDocScan')
             where has_access = 1)) then
    access_allowed = 1;

  if (db_id in (39801, 41701) and
      access_allowed = 0 and
      exists(select has_access
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'kontinfoChange')
             where has_access = 1)) then
    access_allowed = 1;

  has_miniapp = 0;
  if (client_is_active = 1) then
  begin
    if (db_id <> 81001) then
    begin
      select has_miniapp
      from fm_client_has_miniapp(:un_id, :c_id)
      into :has_miniapp;
    end

    if (has_miniapp = 0) then
    begin
      res_text = 'У текущего клиента имеется активный займ';

      if (access_allowed = 0) then
      begin
        res_text = 'У вас нет прав на редактирование номера активного клиента';
        res_is_error = 1;
      end
      suspend;
    end
  end

  cont_kind = f_enum('cont_kind', cont_kind_code, 'id');
  clt_id = f_cont_id(cont_value, 'null');

  if (clt_id is null) then
    exit;

  for select cda_from_c_id, cda_date, c_birth_date
      from contdata, clients
      where cda_from_c_id <> :c_id and
            cda_from_clt_id = :clt_id and
            cda_cont_kind = :cont_kind and
            cda_status = 1 and
            cda_is_main = 1 and
            cda_from_c_id = c_id and
            cda_is_delete = 0
      into found_c_id, cda_date, c_birth_date
  do
  begin
    res_is_error = 0;
    res_text = '';

    select c_name_fio
    from clients
    where c_id = :found_c_id
    into found_c_name;

    if (exists(select *
               from kontstate, knclient, kontract
               where kc_from_c_id = :found_c_id and
                     kc_amount_approved > 0 and
                     kc_from_k_id = ks_from_k_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0 and
                     ks_date_end >= '31.12.2999' and
                     kc_from_k_id = k_id and
                     k_stop_perc is null)) then
    begin
      res_text = 'Есть активный займ';
      if (access_allowed = 0) then
        res_is_error = 1;

      if (datediff(day, cda_date, current_date) <= 90) then
      begin
        res_is_error = 1;
        res_text = 'Подтвержден менее 90 дней назад';
      end
    end
    suspend;
  end
end
