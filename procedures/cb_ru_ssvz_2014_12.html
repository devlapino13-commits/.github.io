CREATE PROCEDURE CB_RU_SSVZ_2014_12(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    OKRUG_NAME VARCHAR(100),
    OKS_BALANCE CURR_TYPE,
    OKS_DELINQ_BALANCE CURR_TYPE,
    PAR7 CURR_TYPE,
    COEF_PAR7 CURR_TYPE,
    KOL_SVOEVREMEN INTEGER,
    KOL_SVZ INTEGER,
    KOL_LUCH INTEGER,
    KOL_SRED INTEGER,
    KOL_HUDSH INTEGER,
    BONUS_LUCH CURR_TYPE,
    BONUS_SRED CURR_TYPE,
    BONUS_HUDSH CURR_TYPE,
    SREDNEVZESH_ZP CURR_TYPE,
    KOL_SSVZ INTEGER,
    BONUS CURR_TYPE,
    FIO_SSVZ VARCHAR(1024),
    COEF_SVZ CURR_TYPE,
    KOL_SVZ_FROM_KOEF INTEGER)
AS
declare variable oks_region_id integer;
declare variable rep_date_begin date;
declare variable reg_id integer;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin
    for select sum(oks_balance), sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0))
        from oksummary
        where oks_region_id = :reg_id and
              :rep_date_end between oks_date_begin and oks_date_end and
              oks_kc_id <> 1000000000003 and
              oks_close_state = 0
        group by oks_region_id
        into oks_balance, oks_delinq_balance
    do
    begin
      par7 = oks_delinq_balance / oks_balance * 100;
--    расчет коэффицента пара
      coef_par7 = case
                    when par7 <= 5.0000 then 1.1
                    when par7 <= 7.0000 then 1
                    when par7 <= 10.0000 then 0.5
                    else 0.1
                  end;
--    кол-во своевременных закрытых кредитов за указанный период
      kol_svoevremen = null;
      select
--             count(*)
             sum(k_kolich_client_approved)
      from kontstate, kontract
      where ks_ol_is_delete = 0 and
            ks_state = 2 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_region_id = :reg_id and
            ks_from_k_id = k_id and
            extract(month from ks_date_begin) = extract(month from k_kredline_end) and
            extract(year from ks_date_begin) = extract(year from k_kredline_end)
      into kol_svoevremen;
      if (kol_svoevremen is null) then
        kol_svoevremen = 0;

--    Вычисляет средневзеванный оклад СВЗ
      kol_svz = null;
      kol_luch = null;
      kol_sred = null;
      kol_hudsh = null;
      srednevzesh_zp = null;
      kol_svz_from_koef = null;
      select kol_svz, kol_luch, kol_sred, kol_hudsh, srednevzesh_zp, bonus_luch, bonus_sred, bonus_hudsh,
             kol_svz_from_koef
      from cb_ru_ssvz_2014_12_temp(:reg_id, :rep_date_end)
      into :kol_svz, :kol_luch, :kol_sred, :kol_hudsh, :srednevzesh_zp, :bonus_luch, :bonus_sred, :bonus_hudsh,
           :kol_svz_from_koef;

--    Вычисление кол-во ССВЗ
      kol_ssvz = 0;
      fio_ssvz = '';
      select count(*), list(c_name_fio, '; ')
      from hr_pstatus, joblist, jobkind, clients
      where ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.Region_KM' and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_id and
            ps_from_c_id = c_id
      into kol_ssvz, fio_ssvz;
      if (kol_ssvz is null) then
        kol_ssvz = 0;

--     если в одном офисе несколько ССВЗ то для вычисления коэффициента СВЗ, кол-во СВЗ делятся на кол-во ССВЗ и округляется
--     в сторону увеличения
      if (kol_ssvz > 1) then
      begin
        kol_svz_from_koef = kol_svz_from_koef / kol_ssvz;
      end

--     Добавлено от 6го апреля 2015 по письму Ироды
      if (kol_svz_from_koef is null) then
        kol_svz_from_koef = 0;
      coef_svz = case
                   when kol_svz_from_koef = 0 then 0
                   when kol_svz_from_koef <= 2 then 30
                   when kol_svz_from_koef <= 5 then 35
                   when kol_svz_from_koef <= 7 then 45
                   when kol_svz_from_koef <= 9 then 55
                   else 60
                 end;

--       bonus = coef_par7 * (SREDNEVZESH_ZP * 30 / 100 + 100 * cast(kol_svoevremen as curr_type) / iif(kol_ssvz = 0, 1, kol_ssvz));
      bonus = coef_par7 * (srednevzesh_zp * coef_svz / 100 + 100 * cast(kol_svoevremen as curr_type) / iif(kol_ssvz = 0, 1, kol_ssvz));
      if (bonus < 0) then
        bonus = 0; -- Добавлен от 07.04.2015

      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
      from region r1, region r2, region r3
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode and
            r2.parent = r3.reg_unicode
      into reg_name, branch_name, okrug_name;

      suspend;
    end
  end
end
