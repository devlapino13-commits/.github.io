CREATE PROCEDURE CARD_WRITEOFF_ON_CLOSING_LOAN(
    CS_ID D_INTEGER,
    AMOUNT D_CURRENCY)
AS
declare variable packet_no d_str_10;
declare variable ticket_no d_integer;
begin
  if (f_get_const('cardprjclose') = 1) then
    exit;
  select
    packet_no,
    ticket_no
  from cardsynctrn_create
  into
    :packet_no,
    :ticket_no;

  insert into cardsync (
    cs_parent_id,
    cs_from_csc_id,
    cs_card_id,
    cs_from_cca_id,
    cs_object_kind,
    cs_object_id,
    cs_amount,
    cs_request_json,
    cs_office_id,
    cs_from_un_id,
    cs_from_k_id,
    cs_from_c_id
  )
  select
    cs_id,
    f_get_cardsynccfg_id('write-off_on_closing'),
    cs_card_id,
    cs_from_cca_id,
    cs_object_kind,
    cs_object_id,
    :amount,
    f_format(
      '{"card_acct":"%1","card_acct_ccy":"KGS","tran_type":"12c","tran_ccy":"KGS","tran_amount":%2,' ||
      '%3,"tran_comment":"%4","bank_c":"21","groupc":"01","payment_mode":2,"tran_date_time":"%5"}',
        cs_from_cca_id,
        round(:amount, 2),
        f_format('"tran_pocket":"%1","tran_ticket":"%2"', :packet_no, :ticket_no),
        coalesce(f_json_str_get_value(cs_request_json, 'info', null), 'переплата') || ':' || round(:amount, 2) || ';',
        f_datetimetostr(current_timestamp)),
    cs_office_id,
    cs_from_un_id,
    cs_from_k_id,
    cs_from_c_id
  from cardsync
  where cs_id = :cs_id;
end
