CREATE PROCEDURE MC_EXTINFO_STATUS(
    MCL_ID D_ID NOT NULL,
    EIL_ID D_ID NOT NULL)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255)
AS
declare variable eil_status d_smallint;
declare variable eil_from_bls_id d_id;
declare variable mcl_lang d_str_lang;
begin
    /*
      Процедура возвращает статус запроса в extinfolist
      Используется в КМП

      Возможные значения res_code:
      1. success - ответ был успешно получен
      2. in_process - ожидание ответа
      3. error - сервис ответил с ошибкой
    */

    mcl_lang = f_get_lang_code(0, :mcl_id);

    select eil.eil_status, eil.eil_from_bls_id
    from extinfolist eil
    where eil.eil_id = :eil_id
    into :eil_status, :eil_from_bls_id;

    if (eil_status is null) then
        res_text = f_lang(mcl_lang, 'Не удалось начать проверку. Попробуйте ещё раз');
    else if (eil_status in (-1, 0, 1) and eil_from_bls_id is null) then
        res_code = 'in_process';
    else if (eil_status = 2 and eil_from_bls_id is not null) then
        res_code = 'success';
    else if (eil_status = -2) then
        res_text = f_lang(mcl_lang, 'Не удалось получить данные. Попробуйте повторить попытку позже');
    else
        res_text = f_lang(mcl_lang, 'Что-то пошло не так. Попробуйте повторить попытку позже');

    if (coalesce(res_text, '') <> '') then
        res_code = 'error';

    suspend;
end
