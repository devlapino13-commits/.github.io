CREATE PROCEDURE FIND_CLIENT_IN_DELETE_BUFFER(
    INPUT_C_ID D_ID = null,
    INPUT_C_NAME_FIO D_STR_100 = null,
    INPUT_C_INN D_STR_20 = null,
    INPUT_C_PASS_NO D_STR_20 = null)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_CODE,
    C_INN D_STR_20,
    C_EXT_INN D_STR_20)
AS
begin
    -- RFCRU-5595
    -- Возвращаем абсолютно все записи, если ни один из входных параметров не был передан
    -- Или конкретную запись по одному из параметров
    -- Примечание: если было передано больше одного параметра, то есть вариант получить больше одной записи
    for select
        c.cdb_from_c_id,
        c.c_name_fio,
        c.c_birth_date,
        c.c_pass_no,
        c.c_inn,
        c.c_ext_inn
    from client_delete_buffer c
    where (:input_c_id is null and :input_c_name_fio is null and :input_c_inn is null and :input_c_pass_no is null and c.cdb_is_restored = 0)
        or (:input_c_id is not null and c.cdb_from_c_id = :input_c_id)
        or (:input_c_name_fio is not null and c.c_name_fio containing :input_c_name_fio)
        or (:input_c_inn is not null and (c.c_inn = :input_c_inn or c.c_ext_inn = :input_c_inn))
        or (:input_c_pass_no is not null and c.c_pass_no = :input_c_pass_no)
    into
        c_id,
        c_name_fio,
        c_birth_date,
        c_pass_no,
        c_inn,
        c_ext_inn
    do
        suspend;

end
