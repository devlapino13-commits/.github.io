CREATE PROCEDURE UXP$UI_UXGRITEMS(
    UA_UUID D_UUID,
    VER_KIND D_STR_CODE = null,
    GRID_UUID D_UUID = null)
RETURNS (
    UFI_UUID D_UUID,
    UFI_CODE D_STR_CODE,
    UFI_PARENT_UUID D_UUID,
    UFI_PARENT_CODE D_STR_CODE,
    UFI_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_NAME D_STR_100,
    UFI_CAPTION_ALIGN D_INTEGER,
    UFI_HINT D_STR_LONG,
    UFI_HIDDEN D_SMALLINT,
    UFP_REQUIRED D_BOOL,
    UFI_REFRESH D_BOOL,
    UFI_READONLY D_BOOL,
    UFI_GROUP_DIRECTION D_INTEGER,
    UFI_H_ALIGN D_INTEGER,
    UFI_V_ALIGN D_INTEGER,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_SIZABLE D_BOOL,
    UFI_ORDER_INDEX D_SMALLINT,
    UFI_GROUP_INDEX D_SMALLINT,
    UFI_FIXED_KIND D_SMALLINT,
    UFI_SUMMARY_KIND D_SMALLINT,
    UFI_DATASET_FIELD D_STR_100,
    UFI_LOOKUP_SQL D_STR_LONG,
    UFI_UFR_CODE D_STR_100,
    UFI_UFR_UUID D_UUID,
    UFI_IS_PARAM D_SMALLINT,
    UFI_COMBO_UF_UUID D_UUID,
    FR_CODE D_STR_100,
    FR_UUID D_UUID,
    UFI_UA_UUID D_UUID,
    USR_ID_FIELD D_STR_CODE,
    USR_TEXT_FIELD D_STR_CODE,
    UA_USR_ID_FIELD D_STR_CODE,
    UA_USR_TEXT_FIELD D_STR_CODE,
    USR_EDITOR_PARAM_FIELD D_STR_CODE,
    USR_EDITOR_READONLY D_BOOL,
    UFP_EDIT_MASK D_STR_LONG,
    UFP_MASK_MESSAGE D_STR_255,
    UFI_LIST_KIND D_SMALLINT,
    UFP_PUBLIC D_BOOL,
    UFI_GLOBAL_PARAM D_STR_CODE,
    UFI_IMG_UUID_CODE D_STR_CODE,
    UFI_UF_UUID D_UUID,
    UFI_EVENT_CONTROL D_STR_CODE,
    EDI_NAME D_STR_CODE,
    USR_MULTISELECT_DELIMITER D_STR_CODE,
    UFI_BAND_INDEX D_INTEGER,
    UFI_HEADER_ALIGN D_SMALLINT,
    UFI_VALUE_ALIGN D_SMALLINT,
    UFP_UUID D_UUID,
    USR_IS_FORM_EDITOR D_BOOL,
    UFI_FONT_SIZE D_SMALLINT,
    UFI_ADD_EXPR_BUTTON D_BOOL,
    UFI_ADD_ACTION_BUTTON D_BOOL,
    UFI_ACTION_NAME D_STR_255)
AS
begin
  for select ufi_uuid, ufi_code, ufi_parent_uuid, ufi_parent_code, ufi_kind, ufi_control, ufi_position, ufi_name,
             ufi_caption_align, ufi_hint, ufi_hidden, ufp_required, ufi_refresh, ufi_readonly, ufi_group_direction,
             ufi_h_align, ufi_v_align, ufi_width, ufi_height, ufi_sizable, ufi_order_index, ufi_group_index,
             ufi_fixed_kind, ufi_summary_kind, ufi_dataset_field, ufi_lookup_sql, ufi_ufr_code, ufi_ufr_uuid,
             ufi_is_param, ufi_combo_uf_uuid, fr_code, fr_uuid, ufi_ua_uuid, usr_id_field, usr_text_field,
             ua_usr_id_field, ua_usr_text_field, usr_editor_param_field, usr_editor_readonly, ufp_edit_mask,
             ufp_mask_message, ufi_list_kind, ufp_public, ufi_global_param, ufi_img_uuid_code, ufi_uf_uuid,
             ufi_event_control, edi_name, usr_multiselect_delimiter, ufi_band_index,
             ufi_header_align, ufi_value_align, ufp_uuid, usr_is_form_editor, ufi_font_size,
             ufi_add_expr_button, ufi_add_action_button, ufi_action_name
      from uxp$ui_uxfritems(:ua_uuid, 'uxgrid', :ver_kind, :grid_uuid)
      where ufi_uuid <> ufi_parent_uuid
      into :ufi_uuid, :ufi_code, :ufi_parent_uuid, :ufi_parent_code, :ufi_kind, :ufi_control, :ufi_position, :ufi_name,
           :ufi_caption_align, :ufi_hint, :ufi_hidden, :ufp_required, :ufi_refresh, :ufi_readonly, :ufi_group_direction,
           :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height, :ufi_sizable, :ufi_order_index, :ufi_group_index,
           :ufi_fixed_kind, :ufi_summary_kind, :ufi_dataset_field, :ufi_lookup_sql, :ufi_ufr_code, :ufi_ufr_uuid,
           :ufi_is_param, :ufi_combo_uf_uuid, :fr_code, :fr_uuid, :ufi_ua_uuid, :usr_id_field, :usr_text_field,
           :ua_usr_id_field, :ua_usr_text_field, :usr_editor_param_field, :usr_editor_readonly, :ufp_edit_mask,
           :ufp_mask_message, :ufi_list_kind, :ufp_public, :ufi_global_param, :ufi_img_uuid_code, :ufi_uf_uuid,
           :ufi_event_control, :edi_name, :usr_multiselect_delimiter, :ufi_band_index,
           :ufi_header_align, :ufi_value_align, ufp_uuid, usr_is_form_editor, ufi_font_size,
           ufi_add_expr_button, ufi_add_action_button, ufi_action_name
  do
    suspend;
end
