CREATE PROCEDURE HP_CLIENT_CONTDATA_DELETED(
    REG_ID D_INTEGER,
    LO_ID D_ID)
RETURNS (
    C_ID D_ID,
    C_NAME D_STR_100,
    C_BIRTH_DATE D_STR_CODE,
    KF_NAME D_STR_100,
    REG_NAME D_STR_100,
    DEL_DATE D_DATE,
    LO_NAME D_STR_100,
    RES_TEXT_1 D_STR_LONG,
    RES_TEXT_2 D_STR_LONG)
AS
declare variable moved_id d_integer;
declare variable cell_main_id d_integer;
declare variable db_id d_integer;
declare variable need_suspend d_smallint;
begin
  if (reg_id is null) then
    reg_id = 0;

  if (lo_id is null) then
    lo_id = 0;

  if (lo_id = 0 and reg_id = 0) then
    exit;

  if (not exists(select ps_from_reg_id
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :lo_id and
                       ps_date_begin <= current_date and
                       ps_date_end >= current_date and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
                       ps_is_working = 1)) then
    reg_id = 0;

  if (reg_id <> 0) then
    lo_id = 0;

  db_id = f_get_db_id();

  cell_main_id = f_enum('cont_kind', 'cell', 'id');
  moved_id = f_enum('cont_delete_kind', 'moved', 'id');

  if (lo_id <> 0) then
  begin
    for select cs_from_c_id, cs_kf_name, reg_code || ' ' || reg_name, c_name_fio
    from clientstate, region, clients
    where reg_unicode = cs_region_id and
          cs_lo_id = :lo_id and
          cs_client_state = 1 and
          cs_lo_id = c_id
    into c_id, kf_name, reg_name, lo_name
    do
    begin
      if (not exists(select * from contdata
      where cda_from_c_id = :c_id and
            cda_cont_kind = :cell_main_id and
            cda_is_main = 1 and
            cda_is_delete = 0 and
            cda_status = 1 and
            cda_from_clt_id <> 0)) then
      begin
        del_date = null;
        select max(cdl_date)
        from contdata, contdelete
        where cda_from_c_id = :c_id and
              cda_cont_kind = :cell_main_id and
              cda_is_main = 1 and
              cda_is_delete = 1 and
              cda_id = cdl_from_cda_id and
              cdl_cont_delete_kind = :moved_id and
              cda_from_clt_id <> 0
        into del_date;

        if (del_date is not null) then
        begin
          need_suspend = 1;
          if (db_id = 81001) then
          begin
            need_suspend = 0;
            if (exists(select *
            from knclient, kontstate, kontract
            where kc_from_c_id = :c_id and
                  kc_from_k_id = ks_from_k_id and
                  ks_state = 1 and
                  ks_ol_is_delete = 0 and
                  ks_date_end >= '31.12.2999' and
                  ks_from_k_id = k_id and
                  k_stop_perc is null)) then
              need_suspend = 1;
          end
          if (need_suspend = 1) then
          begin
            select c_name_fio, f_datetostr(c_birth_date)
            from clients
            where c_id = :c_id
            into c_name, c_birth_date;

            res_text_1 = c_name || ' (' || c_birth_date || ') <br>' || kf_name || iif(:reg_id <> 0, ' ' || lo_name, '');
            res_text_2 = f_datetostr(del_date);
            suspend;
          end
        end
      end
    end
  end else
  begin
    for select cs_from_c_id, cs_kf_name, reg_code || ' ' || reg_name, c_name_fio
    from clientstate, region, region_list(:reg_id, 1, 3, 1), clients
    where cs_region_id = id_list and
          reg_unicode = cs_region_id and
          cs_client_state = 1 and
          cs_lo_id = c_id
    into c_id, kf_name, reg_name, lo_name
    do
    begin
      if (not exists(select * from contdata
      where cda_from_c_id = :c_id and
            cda_cont_kind = :cell_main_id and
            cda_is_main = 1 and
            cda_is_delete = 0 and
            cda_status = 1 and
            cda_from_clt_id <> 0)) then
      begin
        del_date = null;
        select max(cdl_date)
        from contdata, contdelete
        where cda_from_c_id = :c_id and
              cda_cont_kind = :cell_main_id and
              cda_is_main = 1 and
              cda_is_delete = 1 and
              cda_id = cdl_from_cda_id and
              cdl_cont_delete_kind = :moved_id and
              cda_from_clt_id <> 0
        into del_date;

        if (del_date is not null) then
        begin
          need_suspend = 1;
          if (db_id = 81001) then
          begin
            need_suspend = 0;
            if (exists(select *
            from knclient, kontstate, kontract
            where kc_from_c_id = :c_id and
                  kc_from_k_id = ks_from_k_id and
                  ks_state = 1 and
                  ks_ol_is_delete = 0 and
                  ks_date_end >= '31.12.2999' and
                  ks_from_k_id = k_id and
                  k_stop_perc is null)) then
              need_suspend = 1;
          end
          if (need_suspend = 1) then
          begin
            select c_name_fio, f_datetostr(c_birth_date)
            from clients
            where c_id = :c_id
            into c_name, c_birth_date;

            res_text_1 = c_name || ' (' || c_birth_date || ') <br>' || kf_name || iif(:reg_id <> 0, ' ' || lo_name, '');
            res_text_2 = f_datetostr(del_date);
            suspend;
          end
        end
      end
    end
  end
end
