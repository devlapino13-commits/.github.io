CREATE PROCEDURE MC_CREDIT_LIMIT(
    MCL_ID ID_TYPE)
RETURNS (
    LIMIT_AMOUNT D_STR_255,
    AMOUNT D_CURRENCY,
    NEED_CALC_LIMIT D_INTEGER,
    HAS_ACCESS_TO_SCAN D_INTEGER)
AS
declare variable c_id id_type;
begin
  has_access_to_scan = 0;
  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  if (c_id is null or f_get_db_id() not in (39801, 41701)) then
  begin
    limit_amount = 'Ваш кредитный лимит равен нулю';
    amount = 0;
    need_calc_limit = 0;
    suspend;
    exit;
  end

  if (not exists(select kbm.kbm_id
                 from kibminiapp kbm
                 where kbm.kbm_from_c_id = :c_id and
                       kbm.kbm_is_checked = 1 and
                       datediff(day, kbm.kbm_checked_datetime, current_date) <= 30)) then
  begin
    limit_amount = 'Ваш кредитный лимит еще не определен';
    amount = 0;
    need_calc_limit = 1;
    suspend;
    exit;
  end
  else
  begin
    amount = (select credit_limit
              from hp_calc_credit_limit(:c_id, 'installment_goods'));
    amount = coalesce(amount, 0);
    need_calc_limit = 0;
    if (amount > 0) then
    begin
      limit_amount = 'Вам доступен кредитный лимит';
      has_access_to_scan = 1;
    end
    else
      limit_amount = 'Ваш кредитный лимит равен нулю';

  end

  suspend;
end
