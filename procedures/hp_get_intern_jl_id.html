CREATE PROCEDURE HP_GET_INTERN_JL_ID
RETURNS (
    JL_ID D_INTEGER,
    JL_NAME D_STR_100)
AS
begin
  for select jl.jl_id, jl.jl_name
      from joblist jl
      inner join jobkind jk on jk.jk_id = jl.jl_kind and
            jk.jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.Specialist_KOBD')
      where jl.jl_title = 1 and
            jl.jl_name <> 'Стажер'
      into :jl_id, :jl_name
  do
    suspend;
end
