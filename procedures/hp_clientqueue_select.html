CREATE PROCEDURE HP_CLIENTQUEUE_SELECT(
    UN_ID ID_TYPE)
RETURNS (
    C_ID ID_TYPE,
    C_NAME D_STR_100,
    C_NAME_ISCORRECT D_INTEGER,
    C_COMMENT D_STR_4096,
    ADL_ID ID_TYPE,
    ADL_ADDRESS D_STR_255,
    K_STATUS D_STR_20,
    K_DATE D_DATE)
AS
begin
  select first 1
    cq_c_id,
    cq_c_name_iscorrect,
    cq_comment
  from clientqueuecheck
  where cq_un_id = :un_id and cq_status_check = 1
  into
    c_id,
    c_name_iscorrect,
    c_comment;
  if (c_id is null) then begin
    update clientqueuecheck set
      cq_status_check = 1,
      cq_un_id = :un_id
    where cq_id = (select first 1 cq_id from clientqueuecheck
      where cq_status_check = 0 order by cq_orderno, cq_id )
    returning cq_c_id, cq_c_name_iscorrect
    into :c_id, :c_name_iscorrect;
  end
  if (c_id is not null) then begin
    c_name = (select c_name_fio from clients where c_id = :c_id);
    select first 1
      adl_id,
      adl_address_str
    from addrclient
      inner join addrlist
        on adl_id = adc_from_adl_id
    where adc_from_c_id = :c_id
      and adc_address_type = 1
    into
      :adl_id,
      :adl_address;
  end

  select
    iif(ks_state=1, 'Активный', 'Закрытый'),
    max(iif(ks_state=1,k_disb_date,k_kredline_end))
  from kontract
    inner join kontstate
      on ks_from_k_id = k_id
        and ks_date_end = '31.12.9999'
        and ks_state in (1,2)
        and ks_ol_is_delete = 0
    inner join knclient
      on kc_from_k_id = k_id
        and kc_from_c_id = :c_id
        and kc_amount_approved > 0
  group by ks_state
  order by ks_state
  rows 1
  into
    :k_status,
    :k_date;

  suspend;
end
