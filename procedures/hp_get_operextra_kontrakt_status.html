CREATE PROCEDURE HP_GET_OPEREXTRA_KONTRAKT_STATUS(
    K_ID D_ID,
    RD_ID D_INTEGER,
    SUMMA D_CURRENCY,
    IN_DATE D_DATE)
RETURNS (
    RES_TEXT D_STR_255,
    CAN_SAVE D_SMALLINT)
AS
declare variable RD_ID_CANCEL D_INTEGER;
declare variable RD_ID_RESTORATION D_INTEGER;
declare variable ACCRUALS D_CURRENCY;
declare variable ol_date d_date;
begin
  -- RFCKZ-639 проверка перед сохранением для модуля Отмена/Восстановление исполнительной надписи OperExtraCancelAndRestorationForm
  rd_id_cancel = f_get_rd_id('OperExtraKind.LawClientExpenseCancel');
  rd_id_restoration = f_get_rd_id('OperExtraKind.LawClientExpenseRestoration');
  res_text = '';
  can_save = 1;

  if (not exists(select *
                 from kontract k
                 where k.k_id = :k_id)) then
  begin
    res_text = 'не найден договор';
    can_save = 0;
  end

  if (rd_id not in(rd_id_cancel, rd_id_restoration)) then
  begin
    res_text = 'некорректный вид начисления';
    can_save = 0;
  end

  if (can_save = 1) then
  begin
    select
          sum(ox_amount_plus - ox_amount_minus)
    from operextra ox
    inner join refdata rd on rd.rd_id = ox.ox_from_rd_id
    where ox_object_kind = 9 and
          ox_object_id = :k_id and
          ox_ol_is_active = 1 and
          rd.rd_code in ('OperExtraKind.ExecutiveInscription')
    group by ox_from_rd_id
    having sum(ox_amount_plus - ox_amount_minus) > 0
    into accruals;

    accruals = coalesce(accruals, 0);

    -- проверка последовательности
    -- Пример правильной последовательности:
    -- Первоначальное начисление -> Отмена первоначального начисления -> Восстановление первоначального начисления
    if (can_save = 1) then
    begin
      if (rd_id = rd_id_restoration) then
      begin
        if (not exists (select *
                        from operextra_execution_note oen
                        inner join operextra ox on ox.ox_from_ol_id = oen.oen_from_ol_id_cancel and ox.ox_id = oen.oen_from_ox_id_cancel
                        where ox.ox_ol_is_active = 1 and
                              oen.oen_object_kind = 9 and
                              oen.oen_object_id = :k_id and
                              oen.oen_from_rd_id = :rd_id_cancel)) then
        begin
          res_text = 'некорректная последовательность';   -- Если восстановление И нет записи отмены Делаем: ошибочный статаус
          can_save = 0;
        end

        if (exists (select *
                    from operextra_execution_note oen
                    inner join operextra ox on ox.ox_from_ol_id = oen.oen_from_ol_id_restoration and ox.ox_id = oen.oen_from_ox_id_restoration
                    where ox.ox_ol_is_active = 1 and
                          oen.oen_object_kind = 9 and
                          oen.oen_object_id = :k_id and
                          oen.oen_from_rd_id = :rd_id_restoration)) then
        begin
          res_text = 'некорректная последовательность';   -- Если восстановление И восстановление производилась ранее Делаем: ошибочный статаус
          can_save = 0;
        end
      end

      if (rd_id = rd_id_cancel) then
      begin
        if (exists (select *
                    from operextra_execution_note oen
                    inner join operextra ox on ox.ox_from_ol_id = oen.oen_from_ol_id_cancel and ox.ox_id = oen.oen_from_ox_id_cancel
                    where ox.ox_ol_is_active = 1 and
                          oen.oen_object_kind = 9 and
                          oen.oen_object_id = :k_id and
                          oen.oen_from_rd_id = :rd_id_cancel)) then
        begin
          res_text = 'некорректная последовательность';     -- Если отмена И отмена производилась ранее Делаем: ошибочный статаус
          can_save = 0;
        end

      end
    end

    -- Проверка даты начисления
    if (can_save = 1) then
    begin
      if (rd_id = rd_id_cancel) then
      begin

        select
          ox.ox_ol_date
        from operextra ox
        inner join refdata rd on rd.rd_id = ox.ox_from_rd_id
        where ox_object_kind = 9 and
              ox_object_id = :k_id and
              ox_ol_is_active = 1 and
              rd.rd_code in ('OperExtraKind.ExecutiveInscription') and
              ox.ox_amount_plus > 0
       rows 1
       into ol_date;

       if (ol_date > in_date) then
       begin
         res_text = 'дата выписки исполнительной надписи больше даты отмены';
         can_save = 0;
       end

      end
      if (rd_id = rd_id_restoration) then
      begin
        select ox.ox_ol_date
        from operextra_execution_note oen
        inner join operextra ox on ox.ox_from_ol_id = oen.oen_from_ol_id_cancel and ox.ox_id = oen.oen_from_ox_id_cancel
        where ox.ox_ol_is_active = 1 and
              oen.oen_object_kind = 9 and
              oen.oen_object_id = :k_id and
              oen.oen_from_rd_id = :rd_id_cancel
        rows 1
        into ol_date;

        if (ol_date > in_date) then
        begin
          res_text = 'дата отмены исполнительной надписи больше даты восстановления';
         can_save = 0;
        end
      end
    end

    -- Проверка Суммы начисления
    if (can_save = 1) then
    begin
      if (rd_id = rd_id_cancel) then
      begin
        if ((accruals <> summa) and (can_save = 1)) then
        begin
          res_text = 'несоответствие суммы';
          can_save = 0;
        end
      end
      if (rd_id = rd_id_restoration) then
      begin
        select
              sum(ox_amount_plus - ox_amount_minus)
        from operextra ox
        inner join refdata rd on rd.rd_id = ox.ox_from_rd_id
        where ox_object_kind = 9
          and ox_object_id = :k_id
          and ox_ol_is_active = 1
          and rd.rd_code in ('OperExtraKind.ExecutiveInscription')
          and ox.ox_amount_minus <= 0
       group by ox_from_rd_id
       having sum(ox_amount_plus - ox_amount_minus) > 0
       into accruals;

        if ((accruals <> summa) and (can_save = 1)) then
        begin
          res_text = 'несоответствие суммы';
          can_save = 0;
        end
      end
    end
  end

  if (can_save = 1) then
    res_text = 'успешно';
  else if (can_save = 0 and res_text = '') then
    res_text = 'ошибка';

  suspend;
end
