CREATE PROCEDURE HP_SIGNMEMBER_COUNT(
    SGN_ID D_INTEGER,
    DT_CURR D_DATE = null)
RETURNS (
    NUM_COUNT D_INTEGER)
AS
declare variable numb d_integer;
declare variable c_id d_id;
declare variable sm_kind d_smallint;
declare variable sm_obj_id d_id;
begin
  numb = 0;
  delete from temp_cli where 1 = 1;

  if (dt_curr is null) then
    dt_curr = (select sgn_datetime from signdoc where sgn_id = :sgn_id);

  for
    select
      sm_kind,
      sm_obj_id
    from signmember
    where sm_sgn_id = :sgn_id
  into
    :sm_kind,
    :sm_obj_id
  do
  begin
    -- сотрудник
    if (sm_kind = 0) then
    for
      select
        c_id
      from clients
        inner join hr_pstatus
          on ps_from_c_id = c_id
            and :dt_curr between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
      where c_id = :sm_obj_id
    into
      :c_id
    do
    begin
      numb = numb + 1;
      insert into temp_cli (tc_number, tc_from_c_id)
      values (:numb, :c_id);
    end

    -- должность
    if (sm_kind = 1) then
    for
      select
        c_id
      from joblist
        inner join hr_pstatus
          on ps_from_jl_id = jl_id
            and :dt_curr between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
        inner join clients
          on c_id = ps_from_c_id
      where jl_id = :sm_obj_id
    into
      :c_id
    do
    begin
      numb = numb + 1;
      insert into temp_cli (tc_number, tc_from_c_id)
      values (:numb, :c_id);
    end

    -- отдел
    if (sm_kind = 2) then
    for
      select
        c_id
      from depart
        inner join hr_pstatus
          on ps_from_dep_id = dep_id
            and :dt_curr between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
        inner join clients
          on c_id = ps_from_c_id
      where dep_id = :sm_obj_id
    into
      :c_id
    do
    begin
      numb = numb + 1;
      insert into temp_cli (tc_number, tc_from_c_id)
      values (:numb, :c_id);
    end

    -- регион
    if (sm_kind = 3) then
    for
      select
        c_id
      from region_list(:sm_obj_id, 1, 3, 1)
        inner join hr_pstatus
          on ps_from_reg_id = id_list
            and :dt_curr between ps_date_begin and ps_date_end
            and ps_status = 1
            and ps_is_working = 1
        inner join clients
          on c_id = ps_from_c_id
    into
      :c_id
    do
    begin
      numb = numb + 1;
      insert into temp_cli (tc_number, tc_from_c_id)
      values (:numb, :c_id);
    end
  end

  num_count = coalesce((
    select
      count(*)
    from (
      select
        tc_from_c_id
      from temp_cli
      group by tc_from_c_id)), 0);

  suspend;
end
