CREATE PROCEDURE AAREP_BALANCE_OBOROT_ACC(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3),
    REPORT_ACCOUNT VARCHAR(20))
RETURNS (
    AJ_CURRENCY VARCHAR(3),
    AJ_BALANCE_DT VARCHAR(20),
    AJ_BALANCE_KT VARCHAR(20),
    AJ_SUMMA DECIMAL(18,4),
    AJ_KOLICH INTEGER,
    AJ_SUMMA_EKV DECIMAL(18,4))
AS
begin
  for select count(*), ob_currency, a1.al_from_bg_id, a2.al_from_bg_id, sum(ob_summa), sum(ob_summa)
      from acc_list a1, acc_list a2, operbook, operlist, region_list(:report_region_id, 1, -1, 1)
      where a1.al_id = ob_account_dt and
            a2.al_id = ob_account_kt and
            (a1.al_from_bg_id = :report_account or
            a2.al_from_bg_id = :report_account) and
            ob_from_ol_id = ol_id and
            ((ob_currency = :report_currency) or
            (:report_currency = '000') or
            (:report_currency = '---')) and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            operlist.is_delete = 0 and
            (ol_from_reg_id = id_list)
      group by ob_currency, a1.al_from_bg_id, a2.al_from_bg_id
      order by 3, 4
      into :aj_kolich, :aj_currency, :aj_balance_dt, :aj_balance_kt, :aj_summa, :aj_summa_ekv
  do
  begin
    suspend;
  end
end
