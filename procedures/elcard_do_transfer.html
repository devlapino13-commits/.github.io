CREATE PROCEDURE ELCARD_DO_TRANSFER(
    K_ID ID_TYPE)
AS
declare variable C_ID D_ID;
declare variable DISB_AMOUNT CURR_TYPE;
declare variable PAN_MASKED D_STR_20;
declare variable OA_ID D_INTEGER;
begin
    -- если уже есть активная или обработанная задача на перевод денег, тогда новую задачу не создаю
    if (exists(select *
             from extsysmain em
             where em.esm_project = 'elcard_disb' and
                   em.esm_action = 'do_transfer' and
                   em.esm_from_k_id = :k_id and
                   em.esm_status in (0, 1))) then
        exit;

    -- онлайн заем всегда индив, поэтому делаю так, не опасаясь поймать multiselect
    select kncl.kc_from_c_id,
           kncl.kc_amount_approved
    from knclient kncl
    where kncl.kc_from_k_id = :k_id
    and kncl.kc_amount_approved > 0
    into :c_id,
         :disb_amount;

    select oa.oa_pan_masked,
           oa.oa_id
    from online_app oa
    where oa.oa_from_k_id = :k_id
    into :pan_masked,
         :oa_id;
    
    -- создаю задачу на перевод денег на карту
    insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                          esm_full_source, esm_amount, esm_from_k_id, esm_account, esm_result_id)
    values ('elcard_disb', 'do_transfer', current_date, 0, :c_id, 0, '', :disb_amount, :k_id, :pan_masked, :oa_id);
    
    post_event 'elcard_disb_do_transfer';
end
