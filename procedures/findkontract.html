CREATE PROCEDURE FINDKONTRACT(
    EDIT1 VARCHAR(100),
    EDIT2 VARCHAR(100),
    EDIT3 VARCHAR(100),
    COMBO1 INTEGER,
    CHECKED INTEGER,
    REG INTEGER,
    PARAM_K_ID DECIMAL(18,0) = 0)
RETURNS (
    K_ID DECIMAL(18,0),
    K_ID_STR VARCHAR(20),
    K_FROM_KF_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_KOLICH_REPAY INTEGER,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_FROM_F_ID VARCHAR(20),
    K_STATUS INTEGER,
    K_REPAY_KOLICH_CLOSED INTEGER,
    KAL_NEW_KF_NAME VARCHAR(100),
    KAL_NEW_REG_ID INTEGER,
    PR_NAME VARCHAR(50),
    KS_STATE INTEGER,
    K_APPLICATE_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    K_FROM_PR_ID INTEGER,
    K_PERCENT DECIMAL(18,7),
    K_CURRENCY VARCHAR(3),
    PR_IS_PERSONAL INTEGER,
    K_CONTACT_RES VARCHAR(20),
    K_PAYSYS_LINKED INTEGER,
    K_HAS_VIDEO_STR VARCHAR(20),
    K_VERIFY_STATUS INTEGER,
    ACL_ID DECIMAL(18,0),
    ACL_STATUS INTEGER,
    K_CERT_TIME VARCHAR(10),
    K_REFUND INTEGER,
    PR_CODE VARCHAR(100),
    HAS_WRITEOFF_DEBT INTEGER,
    K_IS_ONLINE D_SMALLINT,
    METHOD_ISSUE D_STR_100,
    VID_STATUS D_INTEGER,
    BY_RATING D_SMALLINT,
    IS_CESSION D_SMALLINT)
AS
declare variable cm_check_contact integer;
declare variable cc_id decimal(18,0);
declare variable kol_client integer;
declare variable kol_contact integer;
declare variable cc_client_ok smallint;
declare variable cc_name_ok smallint;
declare variable cc_status_ok smallint;
declare variable cc_address_ok smallint;
declare variable kol_region integer;
declare variable kol_rodina integer;
declare variable cc_type integer;
declare variable cc_check_result integer;
declare variable need_paysys integer;
declare variable k_has_video integer;
declare variable c_verify_status integer;
declare variable cm_database_id integer;
declare variable ma_id integer;
declare variable cert_time_tmp timestamp;
declare variable rd_id integer;
declare variable combo1_old integer;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable k_disb_kind d_integer;
declare variable need_kk d_integer;
declare variable un_id d_id;
declare variable vwp_type d_bool;
declare variable vwp_status d_integer;
declare variable acl_id1 d_id;
declare variable acl_status1 d_integer;
begin
  select cm_check_contact
  from company
  into :cm_check_contact;

  :cm_database_id = f_get_db_id();

  if (cm_database_id in (39801, 41701)) then
  begin
    execute procedure findkontract_check('folder', :edit1);
    execute procedure findkontract_check('loan', :edit2);
  end

  if (cm_check_contact is null) then
    cm_check_contact = 0;
  k_contact_res = ' ';

  if (exists(select *
             from paysys_project)) then
    need_paysys = 1;
  else
    need_paysys = 0;

  if (coalesce(param_k_id, 0) <> 0) then
    edit3 = param_k_id;

  /*select rd_id
  from refdata
  where rd_code = 'ActionGroup.Check_Loan_Data' and
        rd_is_active = 1
  into :rd_id;*/

  acl_id = null;
  acl_status = null;

  combo1_old = -1;

  -- << добавил igusev 23.09.2018 #operkroff_pay
  if ((:combo1 = 2) /* and (cm_database_id in (81001, 94401))*/ ) then
  begin
    combo1 = 0;
    combo1_old = 2;
  end
  -- <<

  for select distinct k_id, ks_state, kal_new_kf_name, kal_new_reg_id, k_id_str, k_from_kf_id, k_kontract_no,
                      k_disb_date, k_kolich_repay, k_circle, k_circle_internal, k_kolich_client_approved, k_from_f_id,
                      k_status, k_repay_kolich_closed, k_applicate_date, k_amount_approved, k_from_pr_id, k_percent,
                      k_currency, k_clients_cashier_id, ma_id, k_refund, k_is_online, k_disb_kind, coalesce(by_rating, 0)
      from findkontract_new_tmp(:reg, :edit1, :edit2, :edit3, :combo1, :checked)
      rows 100
      into :k_id, :ks_state, :kal_new_kf_name, :kal_new_reg_id, :k_id_str, :k_from_kf_id, :k_kontract_no, :k_disb_date,
           :k_kolich_repay, :k_circle, :k_circle_internal, :k_kolich_client_approved, :k_from_f_id, :k_status,
           :k_repay_kolich_closed, :k_applicate_date, :k_amount_approved, :k_from_pr_id, :k_percent, :k_currency, :c_id,
           :ma_id, :k_refund, :k_is_online, :k_disb_kind, :by_rating
  do
  begin
    /* -- 1 >> -- закомментировал igusev 16.11.2018 вместо этого срипта, создан скрипт идущий ниже закомментированного

-- 2 << добавил igusev 23.09.2018 #operkroff_pay  --
    if ((COMBO1_OLD = 2) and (cm_database_id = 81001)) then
    begin
        if (KS_STATE not in(1, 2)) then continue;

        if (KS_STATE = 2) then
        begin
            zad_os = 0;
            zad_perc = 0;
    
            select sum(oko_os + oko_sh_os),
                   sum(oko_perc + oko_sh_perc)
            from operkroff okf
            where oko_from_k_id = :k_id
            and okf.oko_ol_is_active = 1
            into :zad_os, :zad_perc;

            zad_os = coalesce(zad_os, 0);
            zad_perc = coalesce(zad_perc, 0);
        
            if ((zad_os <= 0) and (zad_perc <= 0)) then -- если кредит закрыт и по нему нет долга на внесистемном учете
                continue;
        end
    end
-- 2 << ------------------------------------------ */
    -- 2 >> --- добавил igusev 16.11.2018 вместо скрипта идущего выше
    is_cession = 0;

    if (cm_database_id = 81001 and ks_state = 2 and
        exists(select 1
               from sb_akt sba
               where sba.sba_from_k_id = :k_id
                 and current_date between sba.sba_date_begin and sba.sba_date_end
                 and sba.sba_is_delete = 0
                 and sba.sba_from_sbl_id in (224, 225))) then  -- БКГ ЦЕССИЯ
    is_cession = 1;

    has_writeoff_debt = 0;

    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id;

    -- #operkroff_pay
    if ((ks_state = 2) /*and (cm_database_id in (81001, 94401))*/ ) then
    begin
      zad_os = 0;
      zad_perc = 0;

      select sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc)
      from operkroff okf
      where oko_from_k_id = :k_id and
            okf.oko_ol_is_active = 1
      into :zad_os, :zad_perc;

      zad_os = coalesce(zad_os, 0);
      zad_perc = coalesce(zad_perc, 0);

      has_writeoff_debt = iif((zad_os > 0) or (zad_perc > 0), 1, 0);

      -- если займ закрыт и нет внебалансовой задоженности, проверяем наличие судебных издежек
      if (has_writeoff_debt = 0) then
      begin
        zad_os = 0;
        select sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff)
        from operextra
        inner join refdata on rd_id = ox_from_rd_id and
              rd_main_code = 'OperExtraKind' and
              rd_parent_id > 0
        where ox_ol_is_active = 1 and
              --ox_from_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense') and
              ox_object_kind = f_objkind('kontract', 'id') and
              ox_object_id = :k_id
        into zad_os;
        if (zad_os is null) then
          zad_os = 0;
        if (zad_os > 0) then
          has_writeoff_debt = 1;
      end
    end
    -- #operkroff_pay
    if ((combo1_old = 2) /*and (cm_database_id in (81001, 94401))*/ ) then
    begin
      if (ks_state not in (1, 2)) then
        continue;

      if ((ks_state = 2) and
          (has_writeoff_debt = 0)) then -- если это РФ, тогда: если в окне поиска контрактов в фильтре указано "Активные", тогда отображаю активные контракты + закрытые, у которых есть долг на внесистемке
        continue;
    end
    -- << ------------------------------------------

    -- 2 <<-------------------------------------------
    -- 1 <<-------------------------------------------
    k_cert_time = '';
    if (coalesce(ma_id, 0) > 0) then
    begin
      select cast(ma_plan_disb_time as time), ma_from_un_id
      from miniapp
      where ma_id = :ma_id and
            ma_date_end >= '31.12.2999'
      into :cert_time_tmp, :un_id;
      if (cert_time_tmp is not null) then
        k_cert_time = right('0' || extract(hour from cert_time_tmp), 2) || ':' || right('0' || extract(minute from cert_time_tmp), 2);
    end

    k_has_video = null;
    select fs_status
    from filestore
    where fs_object_kind = 0 and
          fs_object_id = :k_id rows 1
    into :k_has_video;
    if (k_has_video is null) then
      k_has_video_str = 'Нет видео';
    if (k_has_video = 0) then
      k_has_video_str = 'Загружается';
    if (k_has_video = 1) then
      k_has_video_str = 'Готово';

    c_name_fio = '';
    if (c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :c_id
      into :c_name_fio;
    end
    if (k_from_pr_id is not null) then
    begin
      pr_name = '';
      pr_is_personal = 0;
      select pr_name, pr_is_employer, pr_code
      from product
      where pr_id = :k_from_pr_id
      into :pr_name, :pr_is_personal, pr_code;
    end
    kol_client = 0;
    kol_contact = 0;
    k_verify_status = 2;
    for select c_verify_status
        from knclient, clients
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = c_id
        into :c_verify_status
    do
    begin
      if (c_verify_status <> -1) then
      begin
        if (c_verify_status < k_verify_status) then
          k_verify_status = c_verify_status;
      end
    end

    if (cm_check_contact >= 1) then
    begin
      for select kc_from_c_id
          from knclient
          where kc_from_k_id = :k_id and
                kc_amount_approved is not null and
                kc_amount_approved <> 0
          into :cc_id
      do
      begin
        kol_client = kol_client + 1;
        kol_region = 0;
        kol_rodina = 0;
        for select cc_client_ok, cc_name_ok, cc_status_ok, cc_address_ok, cc_type, cc_check_result
            from contacts
            where cc_from_c_id = :cc_id
            into :cc_client_ok, :cc_name_ok, :cc_status_ok, :cc_address_ok, :cc_type, :cc_check_result
        do
        begin
          if ((cc_client_ok = 1) and
              (cc_name_ok = 1) and
              (cc_status_ok = 1) and
              (cc_address_ok = 1) and
              (cc_check_result = 1)) then
          begin
            if (cc_type = 0) then
              kol_region = kol_region + 1;
            if (cc_type = 1) then
              kol_rodina = kol_rodina + 1;
          end
        end
        if ((kol_region >= 1) and
            (kol_rodina >= 1)) then
          kol_contact = kol_contact + 1;
      end
      k_contact_res = cast(kol_contact as varchar(2)) || ' / ' || cast(kol_client as varchar(2));
    end
    k_paysys_linked = 0;
    if (need_paysys = 1) then
    begin
      select sum(psl_sign)
      from paysys_list
      where psl_object_kind = 3 -- контракт
            and
            psl_object_id = :k_id
      into :k_paysys_linked;
      if ((k_paysys_linked is null) or (k_paysys_linked < 0)) then
        k_paysys_linked = 0;
      if (k_paysys_linked > 1) then
        k_paysys_linked = 1;
    end

    acl_id = null;
    acl_status = null;

    need_kk = 0;

    if (rd_id is not null) then
      need_kk = 1;

    if (cm_database_id in (39801, 41701) and
        exists(select *
               from expressapp exa
               where exa.exa_from_k_id = :k_id)) then
    begin
      need_kk = 0;
    end

    if (cm_database_id in (39801, 41701)) then
    begin
      if (:pr_code containing '.express_similar.' or :pr_code containing '.fitness.') then
        need_kk = 0;

     if (exists(select *
                from kontract kt
                where kt.k_id = :k_id and
                      kt.k_by_rating = 1)) then
        need_kk = 0;
    end 

    if (cm_database_id in (39801, 41701) and
        not exists(select *
                   from username un
                   inner join hr_pstatus ps on ps.ps_from_c_id = un.un_from_c_id and
                         current_date between ps.ps_date_begin and ps.ps_date_end
                   inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
                   inner join jobkind jk on jl.jl_kind = jk.jk_id and
                         jk.jk_system_code in ('JobSystem.LO_Senior', 'JobSystem.LO_Standart')
                   where un.un_id = :un_id)) then
      select un.un_id
      from kontract k
      inner join knaktlist kal on k.k_from_kf_id = kal.kal_from_kf_id and
            kal.is_delete = 0 and
            current_date between kal.kal_date and coalesce(kal.kal_date_end, '31.12.2999')
      inner join username un on un.un_from_c_id = kal.kal_new_kf_officer
      where k.k_id = :k_id
      into :un_id;

    -- если договор не из списка офисов, который должн быть обработаны андеррайтерами, тогда сертифкация не труебуется (вообще она выставляется автоматом).
    -- согласование было по telegram, whatsup с Раминой и Бактыгуль
    if ((need_kk = 1) and
        (cm_database_id in (39801, 41701)) and
        (not exists(select *
                    from underwritingcertif usf
                    inner join underwritingcertifspf usfs on usf.ucf_from_reg_id = usfs.ucfs_from_reg_id and
                          current_date between usfs.ucfs_date_begin and usfs.ucfs_date_end and
                          usfs.ucfs_active = 1
                    inner join username un on usfs.ucfs_from_c_id = un.un_from_c_id and
                          un.un_id = :un_id
                    where usf.ucf_from_reg_id = :kal_new_reg_id and
                          current_date between usf.ucf_date_begin and usf.ucf_date_end and
                          usf.ucf_system = f_enum_id('access_sytem', 'undervriter_certif') and
                          usf.ucf_active = 1))) then
    begin
      select acl_id, acl_status
      from hp_actionlist_kontract_get(:k_id)
      into :acl_id, :acl_status;
      /*
      Требование сертификаторов:
      если сотрудник / офис отключены от сертификации и статус КК:
      КК: Отказан
      КК: Отправлен на доработку
      КК: В ожидании
      КК: На рассмотрении
      Подтвержден
      То статус КК должен оставаться актуальный.
      */
      --RFCKG-2462
      if (acl_status between -3 and 2) then
        need_kk = 1;
      else
        need_kk = 0;
    end

    if (need_kk = 1) then -- для тех у кого не запущен кк
    begin
      if (acl_id is null and
          acl_status is null) then
        select acl_id, acl_status
        from hp_actionlist_kontract_get(:k_id)
        into :acl_id, :acl_status;
    end
    else
    begin
      acl_status = -4; -- Не требуется расмотрение КК
    end

    method_issue = case k_disb_kind
                     when 1 then 'Касса'
                     when 5 then 'Банк.карта ' || iif(k_is_online = 1, 'онлайн', 'офлайн')
                     when 6 then 'Банк.карта МБулак'
                     when 7 then 'СБП по № тел.'
                     when 8 then 'На расч. счет'
                     else ''
                   end;

    -- Видеоидентификация
    if (cm_database_id in (39801, 41701)) then
    begin
      vwp_status = null;
      vid_status = -2;
      select
        vwp_type,
        vwp_status,
        acl_id,
        acl_status
      from verify_webcam_photo
        left join actionlist
          on acl_id = vwp_acl_id
      where vwp_k_id = :k_id
      order by vwp_id desc
      rows 1
      into
        :vwp_type,
        :vwp_status,
        :acl_id1,
        :acl_status1;
     if (vwp_type = 0) then
       vid_status =
         case
           when vwp_status = 0 and acl_status1 < 0 then -1
           when vwp_status = 0 and acl_status1 in (0, 1) then 0
           when vwp_status = 1 and acl_id1 is null then 1
           when vwp_status = 1 and acl_status1 = 2 then 2
         end;
    end

    suspend;
  end
end
