CREATE PROCEDURE CM_KG_0238_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable commission d_currency;
declare variable ok_amount d_currency;
declare variable c_id d_id;
declare variable insurance_min_amount d_integer;
declare variable ma_insur_kind d_smallint;
begin
    if (f_get_const('new_loan_conditions') = 1) then
      insurance_min_amount = cast(f_get_const('kont_new_insurance_min_amount') as d_integer);
    else
      insurance_min_amount = cast(f_get_const('kont_insurance_min_amount') as d_integer);

    if (exists(select 1
               from kontract k
               inner join product p on p.pr_id = k.k_from_pr_id
               where k.k_id = :k_id and
                     p.pr_code containing '.ipf.')) then
    insurance_min_amount = cast(f_get_const('kont_insurance_ipf_min_amount') as d_integer);

    res_error_is_critical = 1;
    if (f_get_const('kont_insurance_active') = 1) then
    begin
        select first 1 ma.ma_insur_kind
        from kontract k
        inner join miniapp ma on k.k_from_ma_app_id = ma.ma_id
        where k.k_id = :k_id and
              ma.ma_is_active = 1 and
              ma.ma_date_end >= '21.12.2999'
        into :ma_insur_kind;

        for select cl.c_name_fio,
                   cl.c_id,
                   ik.ik_commission
        from kontract k
        inner join knclient kc on k.k_id = kc.kc_from_k_id
        inner join product p on k.k_from_pr_id = p.pr_id
        inner join clients cl on cl.c_id = kc.kc_from_c_id
        inner join insurance_kontract ik on ik.ik_from_k_id = k.k_id
        where k.k_id = :k_id
        and k.k_amount_approved >= :insurance_min_amount
        and kc.kc_amount_approved > 0
        and p.pr_code containing '.ind.'
        and p.pr_code not containing '.consultation.'
        and p.pr_code not containing '.insur_kontrakt.'   -- RFCKG-4831
        and p.pr_code not containing '.insur_base.'
        and (
             (p.pr_code containing '.standart.') or -- RFCKG-5388
             (p.pr_code containing '.ipf.' and -- RFCKG-5442
              p.pr_code not containing '.ignore_insurance.' and
              :ma_insur_kind = f_enum_id('ma_insur_payment_kind', 'now')
             )
           )
        and p.pr_code not containing '.ignore_insurance.'
        into :res_info_text, :c_id , :commission
        do
        begin
            ok_amount = 0;
            select sum(ok.ok_amount) from operkontract ok
            inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
            where ok.ok_from_k_id = :k_id
            and ok.ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance')     
            and ol.is_delete = 0  
            and ol_document_client = :c_id
            into :ok_amount;
    
            ok_amount = coalesce(ok_amount, 0);
        
            res_error_text='';
    
            if (ok_amount = 0) then
            begin
              res_error_text = 'Оплата за страхование не произведена';
              suspend;
            end
            else if (ok_amount < commission) then
            begin
              res_error_text = 'Оплата за страхование произведена не в полном объеме. Цена за страхование = '||round(commission)||' сом . Оплачено: '||round(ok_amount)||' сом';
              suspend;
            end
            else if (ok_amount > commission) then
            begin
              res_error_text = 'Оплата за страхование произведена в большем объеме. Цена за страхование = '||round(commission)||' сом . Оплачено: '||round(ok_amount)||' сом';
              suspend;
            end
        end
    end
end
