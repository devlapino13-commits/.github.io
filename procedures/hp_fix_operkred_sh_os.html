CREATE PROCEDURE HP_FIX_OPERKRED_SH_OS(
    K_ID ID_TYPE)
AS
declare variable main_ol_id id_type;
declare variable main_sh_date date;
declare variable main_sh_os curr_type;
declare variable fixed integer;
declare variable fix_ol_id id_type;
declare variable has_date integer;
declare variable circle_count integer;
declare variable tmp_ol_id id_type;
declare variable rec_count integer;
declare variable min_count integer;
begin
  circle_count = 0;
  fixed = 1;
  while (fixed = 1) do
  begin
    fixed = 0;
    main_ol_id = null;
    circle_count = circle_count + 1;
    min_count = 999;

    for select ok_from_ol_id
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_type in (2, 3) and
              ok_sh_os <> 0
        group by 1
        having sum(ok_sh_os) <> 0
        into :tmp_ol_id
    do
    begin
      rec_count = 0;
      with tab
      as (select ok_sh_date
          from operkred
          where ok_from_ol_id = :tmp_ol_id and
                ok_from_k_id = :k_id and
                ok_ol_is_active = 1 and
                ok_sh_os <> 0 and
                ok_type in (2, 3)
          group by 1
          having sum(ok_sh_os) <> 0)
      select count(*)
      from tab
      into rec_count;

      if (rec_count < min_count) then
      begin
        min_count = rec_count;
        main_ol_id = tmp_ol_id;
      end
    end
    if (main_ol_id is null) then
      exit;

    for select ok_sh_date, sum(ok_sh_os)
        from operkred
        where ok_from_ol_id = :main_ol_id and
              ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_type in (2, 3)
        group by 1
        having sum(ok_sh_os) <> 0
        order by ok_sh_date desc
        rows 1
        into :main_sh_date, :main_sh_os
    do
    begin
      fixed = 0;
      fix_ol_id = null;
      has_date = 0;

      update operkred
      set ok_sh_os = ok_sh_os - :main_sh_os
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_type in (2, 3) and
            ok_from_ol_id = :main_ol_id and
            ok_sh_date = :main_sh_date rows 1;

      select ok_from_ol_id, max(iif(ok_sh_date = :main_sh_date, 1, 0))
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_type in (2, 3) and
            ok_from_ol_id <> :main_ol_id
      group by 1
      having sum(ok_sh_os) <> 0
      order by 2 desc, 1 desc
      rows 1
      into :fix_ol_id, :has_date;

      if (fix_ol_id is not null) then
      begin
        if (has_date = 0) then
        begin
          insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                                ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax,
                                ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
          select :fix_ol_id, -1, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, 0, 0, 0, ok_pay_type, 0, 0, 0, 0, 0,
                 ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_from_c_id
          from operkred
          where ok_from_k_id = :k_id and
                ok_ol_is_active = 1 and
                ok_type in (2, 3) and
                ok_from_ol_id = :main_ol_id and
                ok_sh_date = :main_sh_date rows 1;
        end

        update operkred
        set ok_sh_os = ok_sh_os + :main_sh_os
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_type in (2, 3) and
              ok_from_ol_id = :fix_ol_id and
              ok_sh_date = :main_sh_date rows 1;
        if (row_count = 1) then
          fixed = 1;
      end
    end

    if (fixed = 0 or circle_count = 150) then
      exception e_default 'Cannot fix';
  end
end
