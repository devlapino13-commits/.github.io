CREATE PROCEDURE AAREP_CLIENTDATAAPP_LIST
RETURNS (
    CDAP_ID TYPE OF COLUMN CLIENTDATAAPP.CDAP_ID,
    CDAP_FROM_C_ID TYPE OF COLUMN CLIENTDATAAPP.CDAP_FROM_C_ID,
    CDAP_CELL_NO TYPE OF COLUMN CLIENTDATAAPP.CDAP_CELL_NO,
    CDAP_EMAIL TYPE OF COLUMN CLIENTDATAAPP.CDAP_EMAIL,
    CDAP_STATUS TYPE OF COLUMN CLIENTDATAAPP.CDAP_STATUS,
    CDAP_FROM_UN_ID TYPE OF COLUMN CLIENTDATAAPP.CDAP_FROM_UN_ID,
    CDAP_DATETIME TYPE OF COLUMN CLIENTDATAAPP.CDAP_DATETIME,
    CDAP_STATUS_UN_ID TYPE OF COLUMN CLIENTDATAAPP.CDAP_STATUS_UN_ID,
    CDAP_STATUS_DATETIME TYPE OF COLUMN CLIENTDATAAPP.CDAP_STATUS_DATETIME,
    CDAP_FROM_BLS_ID TYPE OF COLUMN CLIENTDATAAPP.CDAP_FROM_BLS_ID,
    C_NAME_FIO VARCHAR(100),
    C_CELL_NO VARCHAR(20),
    C_EMAIL VARCHAR(100),
    UN_NAME VARCHAR(30),
    UN_FIO VARCHAR(100))
AS
begin
  for select cdap_id, cdap_from_c_id, cdap_cell_no, cdap_email, cdap_status, cdap_from_un_id, cdap_datetime,
             cdap_status_un_id, cdap_status_datetime, cdap_from_bls_id, c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_valid(0, c_id)),
             (select cont_value
              from hp_contdata_get_email_valid(0, c_id))
      from clientdataapp, clients
      where cdap_is_waiting = 1 and
            c_id = cdap_from_c_id
      into :cdap_id, :cdap_from_c_id, :cdap_cell_no, :cdap_email, :cdap_status, :cdap_from_un_id, :cdap_datetime,
           :cdap_status_un_id, :cdap_status_datetime, :cdap_from_bls_id, :c_name_fio, :c_cell_no, :c_email
  do
  begin
    select un_name, c_name_fio
    from username, clients
    where un_id = :cdap_from_un_id and
          un_from_c_id = c_id
    into :un_name, :un_fio;

    suspend;
  end
end
