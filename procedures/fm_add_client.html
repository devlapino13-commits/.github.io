CREATE PROCEDURE FM_ADD_CLIENT(
    C_NAME_FIO D_STR_100,
    C_SEX D_SMALLINT,
    C_BIRTH_DATE D_DATE,
    C_BIRTH_COUNTRY D_STR_100,
    C_BIRTH_CITY D_STR_100,
    C_NATION D_INTEGER,
    C_CITIZEN D_INTEGER,
    C_DOC_TYPE D_INTEGER,
    C_PASS_NO D_STR_100,
    C_PASS_DATE D_DATE,
    C_INN D_STR_100,
    UN_ID D_ID,
    SURNAME D_STR_100,
    NAME D_STR_100,
    PATRONYMIC D_STR_100,
    SURNAME_ENG D_STR_100,
    NAME_ENG D_STR_100,
    PATRONYMIC_ENG D_STR_100)
RETURNS (
    C_ID D_ID,
    ERR_TEXT D_STR_255)
AS
declare variable DB_ID D_INTEGER;
declare variable LO_ID D_ID;
declare variable C_NAME_FIO_TMP D_STR_100 collate UNICODE;
declare variable C_EXT_INN D_STR_CODE;
declare variable FL_ID D_ID;
declare variable FL_ID_ENG D_ID;
declare variable C_NAME_FIO_ENG D_STR_100;
begin
  err_text = '';
  uxf$set_user(un_id);
  begin
    -- Генерируем c_id
    select result
    from get_kv_id(2)
    into :c_id;

    select un_from_c_id
    from username
    where un_id = :un_id
    into :lo_id;

    db_id = f_get_db_id();

    -- Исправляем баг клавиатуры Xiaomi с латинской буквой ё
    c_name_fio_tmp = c_name_fio;
    c_name_fio_tmp = replace(c_name_fio_tmp, 'ë', 'ё');
    c_name_fio_tmp = replace(c_name_fio_tmp, 'Ë', 'Ё');
    c_name_fio = c_name_fio_tmp;

    -- Корректировки по запросу Алмаза
    c_birth_country = trim(c_birth_country);
    c_birth_country = upper(left(c_birth_country, 1)) || lower(substring(c_birth_country from 2));

    if (surname is not null and name is not null) then
    begin
      select fl_id, fl_id_eng, c_name_fio, c_name_fio_eng
      from hp_prepare_fio(:surname, :name, :patronymic, :surname_eng, :name_eng, :patronymic_eng)
      into :fl_id, :fl_id_eng, :c_name_fio, :c_name_fio_eng;
    end

    if (db_id = 94401) then
    begin
      -- Для AZ вставка C_INN -> C_PINCODE
      insert into clients (c_id, c_type, c_name_fio, c_sex, c_birth_date, c_birth_country, c_birth_city, c_nation,
                           c_citizen, c_doc_type, c_pass_no, c_pass_date, c_pincode, c_is_personal, c_from_reg_id,
                           c_from_un_id, c_source_system, c_from_fl_id, c_name_eng, c_name_eng_fl_id)
      values (:c_id, 0, :c_name_fio, :c_sex, :c_birth_date, :c_birth_country, :c_birth_city, :c_nation, :c_citizen,
              :c_doc_type, left(:c_pass_no, 30), :c_pass_date, left(:c_inn, 20), 1, 1000000, :un_id, 1, :fl_id,
              :c_name_fio_eng, :fl_id_eng)
      returning c_id
      into c_id;
    end
    else
    begin

      -- В РФ, для граждан КР, РК (Citizen.KG, Citizen.KZ) складываем ИНН в c_ext_inn

      c_ext_inn = '';
      if (db_id = 81001 and c_citizen in (8, 10)) then
      begin
        c_ext_inn = c_inn;
        c_inn = '';
      end

      -- Общая вставка для РФ, КР
      insert into clients (c_id, c_type, c_name_fio, c_sex, c_birth_date, c_birth_country, c_birth_city, c_nation,
                           c_citizen, c_doc_type, c_pass_no, c_pass_date, c_inn, c_ext_inn, c_is_personal, c_from_reg_id,
                           c_from_un_id, c_source_system, c_from_fl_id, c_name_eng, c_name_eng_fl_id)
      values (:c_id, 0, :c_name_fio, :c_sex, :c_birth_date, :c_birth_country, :c_birth_city, :c_nation, :c_citizen,
              :c_doc_type, left(:c_pass_no, 30), :c_pass_date, left(:c_inn, 20), left(:c_ext_inn, 20), 1, 1000000, :un_id, 1, :fl_id,
              :c_name_fio_eng, :fl_id_eng)
      returning c_id
      into c_id;
    end

    execute procedure hp_actions_create_anketa(:c_id, :lo_id);
  end
  suspend;
end
