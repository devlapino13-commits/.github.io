CREATE PROCEDURE FM_CLIENTLINK_LIST(
    UN_ID ID_TYPE,
    FIND_TEXT D_STR_255,
    MCL_ID D_ID = 0)
RETURNS (
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    RES_TEXT_1 D_STR_255,
    RES_TEXT_2 D_STR_10)
AS
declare variable C_ID ID_TYPE;
declare variable LO_ID D_ID;
declare variable COMMENT_STR D_STR_255;
declare variable WB_TXT D_STR_100;
begin
  obj_kind = f_objkind('clientlink', 'id');

  if (find_text is null) then find_text = '';

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  mcl_id = coalesce(mcl_id,0);
  if (mcl_id > 0) then
  begin
    select mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into :c_id;
  end 

  if (c_id is null) then
    exit;

  lo_id = c_id;

  for select clk_id, clk_link_name, clk_comment, clk_status,
             (
             case clk_status
               when 0 then 1
               when -1 then 2
               when 1 then 3
               when 2 then 4
               when -2 then 5
             end) as orderno,
             clk_add_datetime, c_name_fio, clk_link_cell_no
      from clientlink, clients
      where (clk_from_un_id = :un_id or clk_from_c_id = :c_id) and
            clk_status <> -2 and
            clk_from_c_id = c_id
      union
      select clk_id, clk_link_name, clk_comment, clk_status,
             (
             case clk_status
               when 0 then 1
               when -1 then 2
               when 1 then 3
               when 2 then 4
               when -2 then 5
             end) as orderno,
             clk_add_datetime, c_name_fio, clk_link_cell_no
      from clientlink, oksummary, knclient, clients
      where oks_date_begin <= current_date and
            oks_date_end >= current_date and
            oks_kc_id = :lo_id and
            oks_from_k_id = kc_from_k_id and
            oks_close_state = 0 and
            kc_amount_approved > 0 and
            kc_from_c_id = clk_from_c_id and
            clk_status <> -2 and
            clk_from_c_id = c_id and
            clk_from_un_id <> 1000000000000
      order by 5, 6 desc as cursor clk
  do
  begin
    obj_id = clk.clk_id;

    res_text_1 = '';
    res_text_2 = '';

    comment_str = '';
    if (coalesce(clk.clk_comment, '') <> '') then
    begin
      comment_str = '<br><font color="blue">' || clk.clk_comment || '</font>';
    end


    wb_txt = '';
    if (exists(select * from extminiweb where emw_phone = clk.clk_link_cell_no)) then -- если через веб сайт то дбавляем "ВБ"
    begin
      wb_txt = '(WS)';
      wb_txt = '<b>' ||:wb_txt||'</b>';
      wb_txt = '<font color="blue">' || :wb_txt || '</font>';
    end 

    res_text_1 = '<br><b>' || clk.clk_link_name || '</b> (от ' || clk.c_name_fio || ')'||:wb_txt||comment_str;

    res_text_2 = (case
                    when clk.clk_status = -2 then '&#9785;' -- хмурое лицо, отказался
                    when clk.clk_status = -1 then '&#9872;' -- флажок, перезвонить
                    when clk.clk_status = 0 then '&#9990;' -- телефон, новый контакт
                    when clk.clk_status = 1 then '&#9786;' -- улыбка, заинтересован
                    when clk.clk_status = 2 then '&#10004;' -- галочка, выдан
                  end);

    if (find_text = '' or res_text_1 containing find_text) then
      suspend;
  end
end
