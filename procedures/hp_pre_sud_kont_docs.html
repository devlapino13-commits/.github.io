CREATE PROCEDURE HP_PRE_SUD_KONT_DOCS(
    REP_K_ID ID_TYPE)
RETURNS (
    EDL_ID D_INTEGER,
    EDC_NAME STR_255,
    DOC_DATE D_DATE,
    SCD_NAME STR_255,
    SLB_FROM_BLS_ID ID_TYPE)
AS
declare variable ma_id d_integer;
declare variable c_id id_type;
begin

  select k_from_ma_app_id
  from kontract
  where k_id = :rep_k_id
  into ma_id;

  for select kc_from_c_id
      from knclient
      where kc_from_k_id = :rep_k_id and
            kc_amount_approved > 0
      into c_id
  do
  begin
    for select edl_id, edc_name, edl_blank_date
        from edoc_list
        inner join edoc_cfg on edl_from_edc_id = edc_id
        where edl_object_id = :c_id and
              edl_object_kind = 0 and
              edl_status = 1
        into edl_id, edc_name, doc_date
    do
    begin
      suspend;
    end
  end

  for select edl_id, edc_name, edl_blank_date
      from edoc_list
      inner join edoc_cfg on edl_from_edc_id = edc_id
      where edl_object_id = :ma_id and
            edl_object_kind = 11 and
            edl_status = 1
      into edl_id, edc_name, doc_date
  do
  begin
    suspend;
  end

  for select edl_id, edc_name, edl_blank_date
      from edoc_list
      inner join edoc_cfg on edl_from_edc_id = edc_id
      where edl_object_id = :rep_k_id and
            edl_object_kind = 9 and
            edl_status = 1
      into edl_id, edc_name, doc_date
  do
  begin
    suspend;
  end

  edl_id = null;
  edc_name = '';

  -- судебные документы
  for select scd_name, slb_from_bls_id, smr_date
      from sud_mainreq, sud_loandocs, sud_cfgdockind, sud_loanbls
      where smr_from_k_id = :rep_k_id and
            sld_doc_kind = scd_id and
            sld_from_smd_id = smr_id and
            sld_id = slb_from_sld_id
      into scd_name, slb_from_bls_id, doc_date
  do
  begin
    suspend;
  end
end
