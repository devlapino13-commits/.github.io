CREATE PROCEDURE AAREP_LO_PORTFOLIO_STATUS(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    REG_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    JK_NAME VARCHAR(100),
    JS_NAME VARCHAR(100),
    PS_IS_WORKING INTEGER,
    PS_WARNING INTEGER,
    PS_WARNING_MEMO VARCHAR(100),
    RES_BALANCE INTEGER,
    PS_DATE_BEGIN DATE,
    RES_DEKRET INTEGER)
AS
declare variable js_code varchar(100);
begin
  for select c_id, c_name_fio, reg_code || ' ' || reg_name, jl_name, jk_name, rd_name, jk_system_code, ps_is_working,
             ps_date_begin
      from hr_pstatus, clients, region, joblist, jobkind, refdata
      where ps_from_c_id = c_id and
            ps_date_begin <= :rep_date and
            ps_date_end >= :rep_date and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            ps_from_c_id > 1000000000003 and
            jl_kind = jk_id and
            jk_system_id = rd_id and
            ps_status = 1
      into c_id, c_name_fio, reg_name, jl_name, jk_name, js_name, js_code, ps_is_working, ps_date_begin
  do
  begin
    ps_warning = 0;
    res_balance = 0;
    ps_warning_memo = '';
    res_dekret = 0;
    if (exists(select *
               from oksummary
               where oks_kc_id = :c_id and
                     oks_date_begin <= :rep_date and
                     oks_balance > 0 and
                     oks_close_state = 0 and
                     oks_date_end >= :rep_date)) then
      res_balance = 1;

    if (exists(select *
               from hr_voctravel, hr_orders, sa_worktype
               where hrv_from_c_id = :c_id and
                     hrv_date_begin <= :rep_date and
                     hrv_date_end_fact >= :rep_date and
                     hrv_from_ho_id = ho_id and
                     ho_order_type = sawt_id and
                     sawt_base_type = 5 and
                     sawt_include_holiday = 1 and
                     sawt_is_paid = 0 and
                     hrv_status = 1)) then
      res_dekret = 1;

    if ((ps_is_working = 0) and
        (res_balance = 1)) then
    begin
      ps_warning = 1;
      ps_warning_memo = 'Есть КП у уволенного сотрудника';
    end
    if ((js_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) and
        (res_balance = 0) and
        (ps_is_working = 1) and
        (res_dekret = 0)) then
    begin
      if ((:rep_date - ps_date_begin) > 30) then
      begin
        ps_warning = 1;
        ps_warning_memo = 'Сотрудник без КП';
      end
    end
    if ((not(js_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner'))) and
        (res_balance = 1) and
        (ps_is_working = 1)) then
    begin
      ps_warning = 1;
      ps_warning_memo = 'Есть КП, но не СВЗ';
    end
    if (ps_warning_memo <> '') then
      suspend;
  end
end
