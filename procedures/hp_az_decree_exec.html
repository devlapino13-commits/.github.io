CREATE PROCEDURE HP_AZ_DECREE_EXEC(
    REP_K_ID ID_TYPE,
    REP_DATE DATE,
    REP_UN_ID ID_TYPE)
RETURNS (
    ERROR_TEXT STR_255)
AS
declare variable sh_id smallint;
declare variable sh_date date;
declare variable zad_os curr_type;
declare variable sh_calc_perc curr_type;
declare variable sh_calc_fine curr_type;
declare variable rd_id integer;
declare variable k_kontract_no str_20;
declare variable ol_memo varchar(1024);
declare variable new_ol_id id_type;
declare variable k_last_reg_id integer;
declare variable os_sum curr_type;
declare variable perc_sum curr_type;
declare variable at_id smallint;
declare variable al_id str_20;
declare variable acc_llc_perc str_20;
declare variable acc_perc str_20;
declare variable acc_llc_os str_20;
declare variable acc_balance str_20;
declare variable acc_dt_perc str_20;
declare variable acc_dt_os str_20;
declare variable list_fio varchar(1024);
declare variable kol_cli smallint;
declare variable c_id id_type;
declare variable zad_os_operkroff curr_type;
declare variable oks_balance curr_type;
declare variable acc_kt_os str_20;
declare variable acc_kt_perc str_20;
declare variable zad_perc_operkroff curr_type;
declare variable zad_other_operkroff curr_type;
begin
  error_text = '';
-- Если договора в списке нет, тогда ошибка
  if (not exists(select * from azrepaykont --kont_for_writeoff
  where ark_from_k_id = :rep_k_id)) then
  begin
    error_text = 'Договор в предоставленном списке не найден.';
    suspend;
    exit;
  end 

  select k_kontract_no, k_last_reg_id, k_kolich_client_approved from kontract
  where k_id = :rep_k_id
  into k_kontract_no, k_last_reg_id, kol_cli;

  if (kol_cli > 1) then
  begin
    error_text = 'В договоре более одного клиента, необходимо амнистировать клиентов.';
    suspend;
    exit;
  end

  select c_name_fio, c_id from knclient, clients
  where kc_from_k_id = :rep_k_id
  and kc_amount_approved > 0
  and kc_from_c_id = c_id
  into list_fio, c_id;

-- Поверка, имеется ли задолженнос на внесистемном учете
  zad_os_operkroff = null;
  select sum(oko_os + oko_sh_os),
         sum(oko_perc + oko_sh_perc),
         sum(oko_other + oko_sh_other)
  from operkroff
  where oko_from_k_id = :rep_k_id
  and oko_ol_date <= :rep_date
  and oko_ol_is_active = 1
  and oko_from_c_id = :c_id
  into zad_os_operkroff, zad_perc_operkroff, zad_other_operkroff;
  zad_os_operkroff = coalesce(zad_os_operkroff,0);
  zad_perc_operkroff = coalesce(zad_perc_operkroff,0);
  zad_other_operkroff = coalesce(zad_other_operkroff,0);

--проверка имеется ли активный баланс
  oks_balance = null;
  select oks_balance from oksummary
  where :rep_date between oks_date_begin and oks_date_end
  and oks_from_k_id = :rep_k_id and
  oks_close_state = 0
  into oks_balance;
  oks_balance = coalesce(oks_balance, 0);

  if (oks_balance > 0 and zad_os_operkroff > 0) then
  begin
    error_text = 'В текущем контракте имеется задолженность на внесистемном учете и в активном балансе.';
    suspend;
    exit;
  end else
  if (oks_balance = 0 and zad_os_operkroff = 0) then
  begin
    error_text = 'В текущем контракте не имеется сумма для рефинансирование.';
    suspend;
    exit;
  end

  if (oks_balance > 0) then -- Реализация для активного кредита
  begin
  --1)  Проверяется, нет ли операций после переданной даты, если есть  - ошибка
    if (exists(select * from operkred
    where ok_from_k_id = :rep_k_id
    and ok_ol_is_active = 1
    and ok_ol_date > :rep_date)) then
    begin
      error_text = 'Имеется операция поздней датой.';
      suspend;
      exit;
    end 
  --2)  Проверка на предмет активности займа – если на текущую дату кредит не активный – ошибка
    if (not exists(select * from kontstate
    where ks_from_k_id = :rep_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    and :rep_date between ks_date_begin and ks_date_end)) then
    begin
      error_text = 'Контракт не активный.';
      suspend;
      exit;
    end 
  
    acc_balance = null;
    acc_perc = null;
    for
    select at_id, al_id
    from kontract, product, prodacclist, acctype, acc_list
    where k_id = :rep_k_id
    and k_from_pr_id = pr_id
    and pr_from_pat_id = pal_from_pat_id
    and pal_from_at_id = at_id
    and at_kind = 1 -- выборка только по кредитным счетам
    and pal_main_acc is not null
    and pal_main_acc = al_from_bg_id
    and pal_main_acc_no = al_number
    and k_last_reg_id = al_from_reg_id
    and at_id in (1, 2)
    into at_id, al_id
    do
    begin
      if (at_id = 1) then acc_balance = :al_id; -- Кредитный портфель
      if (at_id = 2) then acc_perc = :al_id; -- Начисленные проценты
    end
    if (acc_balance is null) then
    begin
      error_text = 'Не настроен счет "Портфеля", прошу настроить.';
      suspend;
      exit;
    end
    if (acc_perc is null) then
    begin
      error_text = 'Не настроен счет "Начисленные проценты", прошу настроить.';
      suspend;
      exit;
    end
  
    acc_dt_os = null;
    select al_id from acc_list
    where al_from_reg_id = :k_last_reg_id
    and al_account = '25270/1'
    into acc_dt_os;
  
    if (acc_dt_os is null) then
    begin
      error_text = 'Не настроен счет "25270/1", прошу настроить.';
      suspend;
      exit;
    end
  
    acc_dt_perc = null;
    select al_id from acc_list
    where al_from_reg_id = :k_last_reg_id
    and al_account = '64012/1' --'89150/1'
    into acc_dt_perc;
  
    if (acc_dt_perc is null) then
    begin
      error_text = 'Не настроен счет "64012/1", прошу настроить.';
      suspend;
      exit;
    end
  end else
  if (zad_os_operkroff > 0) then -- Реализация для внесистемного учета
  begin
    if (exists(select * from operkroff
    where oko_from_k_id = :rep_k_id
    and oko_ol_is_active = 1
    and oko_ol_date > :rep_date)) then
    begin
      error_text = 'Имеется операция поздней датой.';
      suspend;
      exit;
    end 

    acc_dt_os = null;
    select al_id from acc_list
    where al_from_reg_id = :k_last_reg_id
    and al_account = '99990/1'
    into acc_dt_os;
  
    if (acc_dt_os is null) then
    begin
      error_text = 'Не настроен счет "99990/1", прошу настроить.';
      suspend;
      exit;
    end

    acc_dt_perc = acc_dt_os;

    acc_kt_os = null;
    select al_id from acc_list
    where al_from_reg_id = :k_last_reg_id
    and al_account = '99300/1'
    into acc_kt_os;
  
    if (acc_kt_os is null) then
    begin
      error_text = 'Не настроен счет "99300/1", прошу настроить.';
      suspend;
      exit;
    end
    acc_balance = acc_kt_os;

    acc_kt_perc = null;
    select al_id from acc_list
    where al_from_reg_id = :k_last_reg_id
    and al_account = '99301/1'
    into acc_kt_perc;
  
    if (acc_kt_perc is null) then
    begin
      error_text = 'Не настроен счет "99301/1", прошу настроить.';
      suspend;
      exit;
    end
    acc_perc = acc_kt_perc;
  end

  rd_id = (select rd_id from refdata where rd_code = 'Reason_WriteOff.PresidentDecree');

  ol_memo = k_kontract_no||', '||list_fio||'. Prezidentinin 28.02.2019-cu il tarixli Fərmanına əsasən.';
  select kv from Get_kv(:k_last_reg_id, 5)
  into :new_ol_id;
  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete, ol_user_id, ol_from_rd_id)
  values (:new_ol_id, :k_last_reg_id, 1, :rep_date, 8, :ol_memo, 0, 0, :rep_un_id, :rd_id);

  os_sum = 0;
  perc_sum = 0;
  if (oks_balance > 0) then -- закрытия активного кредита
  begin
    for
    select sh_id, sh_date, zad_os, sh_calc_perc, sh_calc_fine from aarep_make_zad_list(:rep_k_id,:rep_date,1)
    where (zad_os <> 0) or (sh_calc_perc <> 0) or (sh_calc_fine <> 0)
    into sh_id, sh_date, zad_os, sh_calc_perc, sh_calc_fine
    do
    begin
       insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os,
                        ok_perc, ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
       values (:new_ol_id, -1, :rep_k_id, :sh_id, :sh_date, 5, 0, -:sh_calc_perc, -:sh_calc_fine, 3, 0, 0, 0, 0, 0);
  
       insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os,
                        ok_perc, ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
       values (:new_ol_id, -1, :rep_k_id, :sh_id, :sh_date, 4, -:zad_os, 0, 0, 3, 0, 0, 0, 0, 0);

       os_sum = os_sum  + zad_os;
       perc_sum = perc_sum + sh_calc_perc;
    end
  end else
  if (zad_os_operkroff > 0) then -- закрытия внесистемного учета
  begin
    insert into operkroff (oko_from_ol_id, oko_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc, oko_other)
    values (:new_ol_id, -1, :rep_k_id, :c_id, 5, -:zad_os_operkroff, -:zad_perc_operkroff, -:zad_other_operkroff);
    os_sum = zad_os_operkroff;
    perc_sum = zad_perc_operkroff;
  end 

  if (os_sum > 0) then
  begin
    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
                   ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date)
    values (:new_ol_id, -1, 944, 1, null, :acc_dt_os, :os_sum, :os_sum, null, null, null, :acc_balance, null, 'Основная сумма займа', null, :rep_date);

    update azrepaykont set ark_fact_repay_sum = :os_sum, ark_from_c_id = :c_id
    where ark_from_k_id = :rep_k_id;

  end 
  if (perc_sum > 0) then
  begin
    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val, ob_summa, ob_from_pc_id,
                   ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail, ob_from_sac_id, ob_ol_date)
    values (:new_ol_id, -1, 944, 1, null, :acc_dt_perc, :perc_sum, :perc_sum, null, null, null, :acc_perc, null, 'Проценты по кредиту',  null, :rep_date);
  end 
  if (oks_balance > 0) then
  begin
    insert into kontstate (ks_from_ol_id, ks_from_k_id, ks_date_begin, ks_date_end, ks_ol_is_delete, ks_state)
    values (:new_ol_id, :rep_k_id, :rep_date, '31.12.9999', 0, 2);
  end
  suspend;
end
