CREATE PROCEDURE HP_INVOICEIPF_PRINT(
    ID D_INTEGER)
RETURNS (
    INV_ID D_INTEGER,
    INV_NUMBER D_STR_CODE,
    INV_DATE_SUPPLY D_DATE,
    PLACE_SUPPLY D_STR_255,
    INV_SUPPLER_ID D_ID,
    SUPPLER_NAME D_STR_100,
    SUPPLER_INN D_STR_20,
    SUPPLER_TAX D_STR_255,
    BUYER_NAME D_STR_100,
    BUYER_INN D_STR_20,
    BUYER_TAX D_STR_255,
    INV_REG_ID D_ID,
    REG_NAME D_STR_CODE,
    INV_AMOUNT D_CURRENCY,
    INV_STATUS D_SMALLINT,
    STATUS D_STR_20,
    INV_NOTE D_STR_LONG,
    INV_UN_ID D_ID,
    UN_NAME D_STR_100,
    INV_DATETIME D_DATETIME)
AS
begin
  for
    select
      inv_id,
      inv_number,
      inv_date_supply,
      inv_suppler_id,
      c_name_fio,
      c_inn,
      inv_reg_id,
      reg_name,
      inv_amount,
      inv_status,
      case inv_status
        when -1 then 'аннулирован'
        when 0 then  'создан'
        when 1 then  'номер внесен'
        when 2 then  'принят'
      end,
      inv_note,
      inv_un_id,
      un_fio,
      inv_datetime
    from invoiceipf
      inner join clients
        on c_id = inv_suppler_id
      inner join region
        on reg_unicode = inv_reg_id
      inner join username
        on un_id = inv_un_id
    where inv_id = :id
  into
    :inv_id,
    :inv_number,
    :inv_date_supply,
    :inv_suppler_id,
    :suppler_name,
    :suppler_inn,
    :inv_reg_id,
    :reg_name,
    :inv_amount,
    :inv_status,
    :status,
    :inv_note,
    :inv_un_id,
    :un_name,
    :inv_datetime
  do
  begin
    suppler_tax = (
      select first 1
        cds_memo_3
      from clientdocs
        inner join passkind
          on pk_id = cds_from_pk_id
        inner join passdoctype
          on pdt_id = pk_from_pdt_id
            and pdt_code = 'certificate_ip'
      where cds_from_c_id = :inv_suppler_id
      order by cds_id desc);

    place_supply = (
      select first 1
        adl_address_str
      from addrclient
        inner join addrlist
          on adl_id = adc_from_adl_id
      where adc_from_c_id = :inv_suppler_id
        and adc_address_type = 3
      order by adc_datetime desc);

    buyer_name = 'Филиал «Центр» ОсОО МКК «М Булак»';
    buyer_inn = '40605200910226';
    buyer_tax = '004 - Первомайский УГНС';

    suspend;
  end
end
