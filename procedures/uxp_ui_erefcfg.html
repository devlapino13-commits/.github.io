CREATE PROCEDURE UXP$UI_EREFCFG(
    EREF_UUID_CODE D_STR_CODE)
RETURNS (
    ERL_NAME D_STR_255,
    ERL_CODE D_STR_255,
    ERC_SHOW_INACTIVE D_BOOL,
    ERC_INT_TAG_CAPTION D_STR_CODE,
    ERC_STR_TAG_CAPTION D_STR_CODE,
    ERC_VISIBLE_ID D_BOOL,
    ERC_VISIBLE_CODE D_BOOL,
    ERC_VISIBLE_NAME D_BOOL,
    ERC_VISIBLE_MEMO D_BOOL,
    ERC_VISIBLE_ORDERNO D_BOOL,
    ERC_VISIBLE_IS_ACTIVE D_BOOL,
    ERC_VISIBLE_STR_ID D_BOOL,
    ERC_VISIBLE_TAG D_BOOL,
    ERC_LOOKUP_FIELD D_STR_CODE,
    USER_ACCESS D_BOOL)
AS
declare variable erl_kind d_smallint;
declare variable erl_lookup d_smallint;
declare variable un_id d_id;
begin
  select erl_kind, erl_code, erl_name, erl_lookup
  from ux$ereflist
  where erl_uuid = char_to_uuid(:eref_uuid_code)
  into erl_kind, erl_code, erl_name, erl_lookup;

  erc_lookup_field = trim(case erl_lookup
    when 0 then 'name'
    when 1 then 'code'
    else 'caption'
  end);

  un_id = uxf$user();

  user_access = 0;
  if (erl_kind = 1) then
  begin
    if (exists(select has_access
               from hp_permission_access_check(:un_id, 'PermActions', 0, :erl_code)
               where has_access = 1)) then
      user_access = 1;
  end

  erc_show_inactive = 0;
  erc_int_tag_caption = '';
  erc_str_tag_caption = '';
  erc_visible_id = 1;
  erc_visible_code = 1;
  erc_visible_name = 1;
  erc_visible_memo = 0;
  erc_visible_orderno = 0;
  erc_visible_is_active = 0;
  erc_visible_str_id = 0;
  erc_visible_tag = 0;

  if (erl_kind = 0) then
  begin
    erc_visible_id = 0;
    erc_visible_code = 0;
    erc_visible_name = 0;

    if (erl_lookup <> 1) then
      erc_visible_name = 1;
    if (erl_lookup <> 0) then
      erc_visible_code = 1;
  end else
  if (erl_kind = 1) then
  begin
    select erc_show_inactive, erc_int_tag_caption, erc_str_tag_caption, erc_visible_id, erc_visible_code,
               erc_visible_name, erc_visible_memo, erc_visible_orderno, erc_visible_is_active, erc_visible_str_id,
               erc_visible_tag
    from ux$erefcfg
    where erc_erl_uuid = char_to_uuid(:eref_uuid_code)
    into :erc_show_inactive, :erc_int_tag_caption, :erc_str_tag_caption, :erc_visible_id,
         :erc_visible_code, :erc_visible_name, :erc_visible_memo, :erc_visible_orderno, :erc_visible_is_active,
         :erc_visible_str_id, :erc_visible_tag;
  end

  suspend;
end
