CREATE PROCEDURE HP_DOCRECOG_WEB_SAVE(
    DOC_DATA D_BLOB_TEXT,
    DOC_BLOB1 D_BLOB_BINARY = null,
    DOC_BLOB2 D_BLOB_BINARY = null)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable tmp_json d_str_long;
declare variable drm_has_page1 d_smallint;
declare variable drm_has_page2 d_smallint;
declare variable pk_id d_integer;
declare variable drl_id d_id;
declare variable c_id d_id;
declare variable bls_caption d_str_100;
declare variable bls_id1 d_id;
declare variable bls_id2 d_id;
declare variable un_id d_id;
declare variable acl_id d_id;
begin
  -- Процедура сохранения документа вместе со сканами с сайта edoc
  /*
    doc_data ожидаем в следующем формате:
    {"c_id": 1010100106656, "drm_code":"kgz.id.type1",
      "fields":[
        {"code":"name", "value":"ИВАН", "is_accepted":"0"},
        {"code":"surname", "value":"ИВАНОВ", "is_accepted":"0"},
        {"code":"birth_date", "value":"01.01.1990", "is_accepted":"0"},
        {"code":"issue_date", "value":"01.01.2010", "is_accepted":"0"}
      ]
    }
  */
  un_id = 1000000000000;
  c_id = f_json_blob_get_value(doc_data, 'c_id', null);
  if (c_id is null) then
    exception e_default 'c_id not found';


  select res_data, res_id, res_text
  from fm_docrecog_action(:un_id, null, 'draft_create_web', :doc_data)
  into :tmp_json, :res_code, :res_text;

  if (res_code = 1) then
  begin
    suspend;
    exit;
  end

  drl_id = f_json_str_get_value(tmp_json, 'drl_id', null);
  if (res_text = '' and drl_id is null) then
    exception e_default 'drl_id not found';

  select pk_id, pk_name
  from docrecoglist, passkind
  where drl_id = :drl_id
    and drl_from_pk_id = pk_id
  into pk_id, bls_caption;

  select drm_has_page1, drm_has_page2
  from hp_get_drm_has_pages(:pk_id)
  into :drm_has_page1, :drm_has_page2;

  if (drm_has_page1 = 0 and doc_blob1 is not null) then
    exception e_default 'doc_blob1 is not needed';
  if (drm_has_page2 = 0 and doc_blob2 is not null) then
    exception e_default 'doc_blob2 is not needed';

  -- Вставка первой страницы
  if (drm_has_page1 = 1) then
  begin
    if (doc_blob1 is null) then
      exception e_default 'doc_blob1 is required';

    bls_caption = 'Стр.1'|| coalesce(': ' || bls_caption, '');

    execute procedure hp_insert_into_blobstore ('file.jpg', :bls_caption, :doc_blob1,
      :un_id, 3, 3) -- bls_kind: Личный кабинет, bls_source_system: Сервис
    returning_values :bls_id1;
  end

  -- Вставка второй страницы
  if (drm_has_page2 = 1) then
  begin
    if (doc_blob2 is null) then
      exception e_default 'doc_blob2 is required';

    bls_caption = 'Стр.2'|| coalesce(': ' || bls_caption, '');

    execute procedure hp_insert_into_blobstore ('file.jpg', :bls_caption, :doc_blob2,
      :un_id, 3, 3) -- bls_kind: Личный кабинет, bls_source_system: Сервис
    returning_values :bls_id2;
  end

  select res_acl_id
  from hp_actionlist_insert(null, 1010100, 'DocRecogList', :drl_id, 'ActionGroup.DocRecog_Client', 0, :un_id)
  into acl_id;

  -- Закрепляем блобы к документу
  update docrecoglist
  set
    drl_from_bls_1 = :bls_id1,
    drl_from_bls_2 = :bls_id2,
    drl_from_acl_id = :acl_id,
    drl_from_un_id = :un_id,
    drl_status = 0
  where drl_id = :drl_id;

  res_code = 0;
  res_text = 'Документ успешно сохранен';

  suspend;
end
