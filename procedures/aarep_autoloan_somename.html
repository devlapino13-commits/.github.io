CREATE PROCEDURE AAREP_AUTOLOAN_SOMENAME(
    K_ID D_ID,
    REP_DATE D_DATE,
    IS_OWNER INTEGER = 1)
RETURNS (
    RES_NAME STR_255)
AS
declare variable k_disb_date date;
declare variable c_id id_type;
begin
  rep_date = coalesce(rep_date,(select k_disb_date from kontract where k_id = :k_id));
  for
    select z_from_c_id
    from zalog where z_from_k_id = :k_id and z_type = 1
    union
    select kc_from_c_id
    from knclient where kc_from_k_id = :k_id
  into
    :c_id
  do
  begin
    if (is_owner=1) then begin
      select
        f_docrecog_value(cds_from_drl_id, 'calc_name_full')
      from clientdocs
      where cds_from_c_id = :c_id
        and cds_from_pk_id = 310
        and cds_is_delete = 0
        and cds_from_drl_id > 0
        and datediff(day, :rep_date, cds_create_datetime) <= 0
      order by cds_create_datetime desc rows 1
      into :res_name;
    end else begin
      res_name = (select c_name_fio from clients where c_id = :c_id);
    end
    if (res_name<>'') then suspend;
  end
end
