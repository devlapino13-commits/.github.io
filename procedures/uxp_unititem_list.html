CREATE PROCEDURE UXP$UNITITEM_LIST(
    UNIT_UUID D_UUID)
RETURNS (
    UNI_UUID_CODE D_STR_CODE,
    UNI_PARENT_UUID_CODE D_STR_CODE,
    UNI_UUID D_UUID,
    UNI_UNT_UUID D_UUID,
    UNI_PARENT_UUID D_UUID,
    RES_CAPTION D_STR_255,
    UNI_CODE D_STR_CODE,
    UNI_NAME D_STR_100,
    UNI_JSON_PATH D_STR_LONG,
    UNI_POSITION D_INTEGER,
    UNI_INACTIVE D_BOOL,
    UNI_KIND D_SMALLINT,
    UNI_CAPTION D_STR_100,
    UNI_CONTROL D_STR_CODE,
    UNI_HINT D_STR_LONG,
    UNI_ADD_INFO D_STR_8100,
    UNI_CAPTION_ALIGN D_SMALLINT,
    UNI_HIDDEN D_BOOL,
    UNI_READONLY D_BOOL,
    UNI_GROUP_DIRECTION D_SMALLINT,
    UNI_H_ALIGN D_SMALLINT,
    UNI_V_ALIGN D_SMALLINT,
    UNI_LEFT D_INTEGER,
    UNI_TOP D_INTEGER,
    UNI_WIDTH D_INTEGER,
    UNI_HEIGHT D_INTEGER,
    UNI_SIZABLE D_BOOL,
    UNI_ORDER_INDEX D_SMALLINT,
    UNI_GROUP_INDEX D_SMALLINT,
    UNI_FIXED_KIND D_SMALLINT,
    UNI_SUMMARY_KIND D_SMALLINT,
    RES_IMAGE_INDEX D_INTEGER,
    UNI_DATASOURCE D_STR_CODE,
    UNI_FIELDNAME D_STR_CODE,
    UNI_LINKED_LABEL D_STR_CODE,
    UNI_LOOKUP_SOURCE D_STR_CODE,
    UNI_LOOKUP_KEYFIELDS D_STR_255,
    UNI_LOOKUP_FIELDS D_STR_255,
    RES_VISUAL D_SMALLINT,
    RES_FRAME_LINKED D_BOOL,
    RES_FRAME_UUID D_UUID,
    RES_STATE D_SMALLINT)
AS
declare variable unf_code d_str_code;
begin
  for select uni_uuid,
             uni_unt_uuid,
             uni_parent_uuid,
             uni_code,
             uni_name,
             uni_json_path,
             uni_position,
             uni_inactive,
             uni_kind, 
             uni_caption, 
             uni_control, 
             uni_hint, 
             uni_add_info, 
             uni_caption_align, 
             uni_hidden, 
             uni_readonly, 
             uni_group_direction, 
             uni_h_align, 
             uni_v_align, 
             uni_left,
             uni_top,
             uni_width, 
             uni_height, 
             uni_sizable, 
             uni_order_index, 
             uni_group_index, 
             uni_fixed_kind, 
             uni_summary_kind,
             uni_datasource,
             uni_fieldname,
             uni_linked_label,
             uni_lookup_source,
             uni_lookup_keyfields, 
             uni_lookup_fields, 
             cls_image_id,
             cls_visual,
             unf_uuid,
             unf_code
      from ux$unititem inner join ux$class on (uni_name = cls_name)
                       left join ux$unitframe on (unf_owner = uni_unt_uuid and unf_uni_uuid = uni_uuid)
      where uni_unt_uuid = :unit_uuid
      into :uni_uuid,
           :uni_unt_uuid,
           :uni_parent_uuid,
           :uni_code,
           :uni_name,
           :uni_json_path,
           :uni_position,
           :uni_inactive,
           :uni_kind, 
           :uni_caption, 
           :uni_control, 
           :uni_hint, 
           :uni_add_info, 
           :uni_caption_align, 
           :uni_hidden, 
           :uni_readonly, 
           :uni_group_direction, 
           :uni_h_align, 
           :uni_v_align, 
           :uni_left,
           :uni_top,
           :uni_width, 
           :uni_height, 
           :uni_sizable, 
           :uni_order_index, 
           :uni_group_index, 
           :uni_fixed_kind, 
           :uni_summary_kind,
           :uni_datasource,
           :uni_fieldname,
           :uni_linked_label,
           :uni_lookup_source,
           :uni_lookup_keyfields,
           :uni_lookup_fields,
           :res_image_index,
           :res_visual,
           :res_frame_uuid,
           :unf_code
  do
  begin
    res_caption = uni_code;
    if (res_caption not containing uni_name) then
      res_caption = res_caption || '    / ' || uni_name;

    uni_uuid_code = uuid_to_char(uni_uuid);
    uni_parent_uuid_code = uuid_to_char(uni_parent_uuid);

    res_frame_linked = 0;
    if (res_frame_uuid is not null) then
    begin
      res_frame_linked = 1;
      if (res_caption not containing unf_code) then
        res_caption = res_caption || '    / ' || unf_code;
    end

    res_state = uni_group_direction;

    if (uni_inactive = 1) then
      res_state = -2;

    suspend;
  end
end
