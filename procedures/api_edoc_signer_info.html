CREATE PROCEDURE API_EDOC_SIGNER_INFO(
    EDOC_ID D_ID,
    EDOC_SIGNER D_STR_CODE)
RETURNS (
    RES_COMPANY_NAME D_STR_255,
    RES_TITLE D_STR_255,
    RES_NAME D_STR_255,
    RES_JOBNAME D_STR_255,
    RES_OFFICE D_STR_255,
    RES_DOC_KIND D_STR_255,
    RES_DOC_INFO D_STR_255,
    RES_SIGN_DATETIME D_STR_CODE,
    RES_SIGN_GUID D_STR_CODE)
AS
declare variable edoc_signer_guid d_str_code;
declare variable need_exit d_bool;
declare variable edl_object_kind d_integer;
declare variable edl_object_id d_id;
declare variable obj_kind_code d_str_code;
declare variable edl_blank_date d_date;
declare variable c_id d_id;
begin
  res_company_name = '';
  res_title = 'Неверная ссылка, документ не найден';
  res_name = '';
  res_jobname = '';
  res_office = '';
  res_doc_kind = '';
  res_doc_info = '';
  res_sign_datetime = null;
  res_sign_guid = '';

  need_exit = 0;
  if (coalesce(edoc_id, 0) = 0 or coalesce(edoc_signer, '') = '') then
    need_exit = 1;
  else
  begin
    edoc_signer_guid = f_edoc_signer(edoc_id);
    if (edoc_signer <> edoc_signer_guid) then
      need_exit = 1;
  end

  if (need_exit = 0) then
  begin
    select edl_object_kind, edl_object_id, edl_blank_date
    from edoc_list
    where edl_id = :edoc_id
    into edl_object_kind, edl_object_id, edl_blank_date;

    if (row_count = 0) then
      need_exit = 0;
    else
    begin
      obj_kind_code = f_objkind(edl_object_kind, 'code');
      if (obj_kind_code <> 'hr_orders') then
        need_exit = 1;
    end
  end

  if (need_exit = 0) then
  begin
    select reg_code || ' ' || reg_name, ksm_from_c_id,
           trim(iif(coalesce(ho_order_no, '') = '', 'б/н', ho_order_no)) || ' от ' || f_datetostr(ho_date),
           sawt_name,
           f_datetimetostr(coalesce(ho_status_datetime, ho_save_time)),
           ksr_kont_postion
    from hr_orders, kontsignmen, kontsignreg, region, sa_worktype
    where ho_id = :edl_object_id and
          ho_signer_id = ksr_id and
          ksr_from_ksm_id = ksm_id and
          ho_order_type = sawt_id and
          ho_from_reg_id = reg_unicode
    into res_office, c_id, res_doc_info, res_doc_kind, res_sign_datetime, res_jobname;

    if (c_id is null) then
      need_exit = 1;
  end

  if (need_exit = 0) then
  begin
    select cm_name
    from company
    into res_company_name;

    res_title = 'Расшифровка электронной подписи';

    select c_name_fio
    from clients
    where c_id = :c_id
    into res_name;

    select jl_name
    from hr_pstatus, joblist
    where ps_from_c_id = :c_id and
          ps_date_begin <= :edl_blank_date and
          ps_date_end >= :edl_blank_date and
          ps_status = 1 and
          ps_from_jl_id = jl_id
    into res_jobname;

    res_sign_guid = lower(edoc_signer);
  end

  suspend;
end
