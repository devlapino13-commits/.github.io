CREATE PROCEDURE AAREP_LAST_MAN_PERIOD_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    REGION_NAME VARCHAR(100),
    REPORT_MAX_DATE DATE,
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_DATE_OPER DATE,
    MA_PLAN DECIMAL(18,4))
AS
declare variable reg_id integer;
declare variable ma_id integer;
declare variable ma_number integer;
declare variable ma_is_active integer;
declare variable ma_kf_id decimal(18,0);
declare variable kol integer;
declare variable reg_id_prior integer;
declare variable ma_id_prior integer;
declare variable ma_is_new integer;
begin
  ma_plan = 0;
  for select reg_unicode, reg_code || ' ' || reg_name, max(man_disb_date)
      from miniappnew, region, region_list(:report_region_id, 1, -1, 1)
      where man_date <= :report_date_end and
            man_office = reg_unicode and
            (reg_unicode = id_list)
      group by reg_unicode, reg_code, reg_name
      into :reg_id, :region_name, :report_max_date
  do
  begin
    kol = 0;
    ma_summa_minus = 0;
    ma_id_prior = -1;
    reg_id_prior = -1;
    ma_plan = 0;
    if (report_max_date > report_date_end) then
    begin
      for select reg_unicode, reg_code || ' ' || reg_name, man_disb_date, max(man_amount)
          from miniappnew, region
          where man_date <= :report_date_end and
                man_disb_date between :report_date_end and :report_max_date and
                man_office = :reg_id and
                reg_unicode = man_office
          group by reg_unicode, reg_code, reg_name, man_disb_date
          into :reg_id, :region_name, :ma_date_oper, :ma_plan
      do
      begin
        if (ma_plan is null) then
          ma_plan = 0;
        suspend;
      end
      ma_plan = 0;
      for select ma_id, ma_number, ma_date_oper, ma_is_active, ma_summa_minus, ma_from_kf_id
          from miniapp
          where ma_date_app <= :report_date_end and
                ma_date_end >= :report_date_end and
                ma_date_oper between :report_date_end - 60 and :report_max_date and
                ma_office_id = :reg_id
          order by ma_id desc, ma_number desc
          into :ma_id, :ma_number, :ma_date_oper, :ma_is_active, :ma_summa_minus, :ma_kf_id
      do
      begin
        if (ma_id_prior <> ma_id) then
        begin
          if (ma_is_active <= 0) then
            ma_summa_minus = 0;
          if ((ma_date_oper >= :report_date_end and
              ma_date_oper <= :report_max_date)) then
          begin
            /* Проверка цикла */
            select distinct ks_state
            from kontstate, kontract
            where k_id = ks_from_k_id and
                  cast(ks_date_begin as date) < :report_date_end and
                  ks_state = 1 and
                  ks_ol_is_delete = 0 and
                  k_from_kf_id = :ma_kf_id rows 1
            into :ma_is_new;
            if ((ma_is_new is null) or (ma_is_new = 0)) then
              ma_is_new = 1;
            else
              ma_is_new = 0;
            if (ma_is_new = 1) then
            begin
              kol = 1;
              if (ma_plan is null) then
                ma_plan = 0;
              suspend;
            end
          end
        end
      end
    end
    if (kol = 0) then
    begin
      ma_date_oper = report_max_date;
      suspend;
    end
  end
end
