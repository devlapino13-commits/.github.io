CREATE PROCEDURE AAREP_CONTACTS_LO_PENALTY(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KF_BRANCH_NAME STR_100,
    REG_NAME STR_100,
    SVZ_FIO STR_100,
    DATE_APP DATE,
    KF_NAME STR_100,
    C_NAME_FIO STR_100,
    CC_NAME STR_100,
    CC_CREATE_DATE TIMESTAMP,
    TOTAL_RES STR_100)
AS
declare variable ma_id integer;
declare variable kf_id id_type;
declare variable ma_expert_id id_type;
declare variable c_id id_type;
declare variable cc_cell_no str_100;
declare variable cc_total_res integer;
declare variable cc_cell_no_old str_100;
begin
  for select maa.ma_id, maa.ma_from_kf_id, maa.ma_expert_id, maa.ma_date_app
      from miniapp maa
      where maa.ma_number = 0 and
            maa.ma_date_app between :rep_date_begin and :rep_date_end
      into :ma_id, :kf_id, :ma_expert_id, date_app
  do
  begin
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :ma_expert_id
    into :svz_fio;

    select kal.kal_new_kf_name, reg.reg_name, reg2.reg_name
    from knaktlist kal
    inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
    inner join region reg2 on reg2.reg_unicode = reg.parent
    where :rep_date_end between kal.kal_date and kal.kal_date_end and
          kal.kal_from_kf_id = :kf_id and
          kal.is_delete = 0
    into :kf_name, :reg_name, kf_branch_name;

    cc_cell_no_old = '';

    for select macl.mac_from_c_id, cl.c_name_fio
        from miniappclient macl
        inner join clients cl on cl.c_id = macl.mac_from_c_id
        where macl.mac_from_ma_id = :ma_id and
              macl.mac_amount > 0
        into :c_id, c_name_fio
    do
    begin
      for select cont.cc_cell_no, cont.cc_name, cont.cc_total_res, cont.cc_create_date
          from contacts cont
          where cont.cc_from_c_id = :c_id and
                cont.cc_create_date < :rep_date_end + 1 and
                cont.cc_create_date >= :date_app - 5
          order by cc_create_date, cont.cc_cell_no, cont.cc_id desc
          into cc_cell_no, cc_name, cc_total_res, cc_create_date
      do
      begin
        if (cc_cell_no <> cc_cell_no_old) then
        begin
          cc_cell_no_old = cc_cell_no;

          total_res = (case cc_total_res
                         when 0 then 'не звонили'
                         when 1 then 'информация подтверждена'
                         when -1 then 'не дозвон'
                         when -2 then 'не подтверждена'
                       end);

          suspend;
        end
      end
    end
  end
end
