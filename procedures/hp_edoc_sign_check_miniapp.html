CREATE PROCEDURE HP_EDOC_SIGN_CHECK_MINIAPP(
    MA_ID INTEGER)
RETURNS (
    KF_CLIENT_FIO VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    KF_ERROR_TEXT VARCHAR(255),
    KF_ERROR_IS_CRITICAL INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable CHECK_DATE D_DATE;
declare variable DB_ID D_INTEGER;
begin
  -- если ИПФ и по ИФП eDoc отключен, тогда дальше не проверяем
  if (f_tmp_const('edoc_ipf_inactive') = 1) then
  begin
      if (exists(select *
                  from miniapp maa
                  inner join product pr on pr.pr_id = maa.ma_from_pr_id
                  where maa.ma_id = :ma_id
                  and current_date between ma_date_app and ma_date_end
                  and ma_is_active = 1
                  and pr.pr_code like '%.ipf.%')) then
      begin
        exit;
      end
  end

  -- Заявка
  select ma_date_app
  from miniapp
  where ma_id = :ma_id
  order by ma_id desc, ma_number desc
  rows 1
  into check_date;

  db_id = f_get_db_id();
  if (db_id not in (39801, 41701)) then
  begin
    kf_error_text = f_edoc_check('loan.application', ma_id, check_date);

    if (kf_error_text <> '') then
    begin
      kf_error_is_critical = 2;
      suspend;
    end
  end

  -- Договор АСП
  for select c_id, c_name_fio, c_pass_no, mac_amount, ma_date_oper
      from miniapp, miniappclient, clients
      where ma_id = :ma_id and
            current_date between ma_date_app and ma_date_end and
            ma_is_active = 1 and
            ma_id = mac_from_ma_id and
            mac_amount > 0 and
            mac_from_c_id = c_id
      into c_id, kf_client_fio, kf_client_pass_no, kc_amount, kf_disb_date
  do
  begin
    if (kf_disb_date >= '10.01.2020') then
    begin
      kf_error_text = f_edoc_check('asp.agreement', c_id, kf_disb_date);
      if (kf_error_text <> '') then
      begin
        kf_error_is_critical = 2;
        suspend;
      end
    end
  end
end
