CREATE PROCEDURE HP_ASP_VALID(
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID)
RETURNS (
    ASP_VALID D_SMALLINT,
    INVALID_FIO_LIST D_STR_2048,
    OFFLINE_FIO_LIST D_STR_2048)
AS
declare variable c_id d_id;
declare variable asp_valid_tmp d_smallint;
declare variable c_name_fio d_str_100;
begin
  invalid_fio_list = '';
  offline_fio_list = '';
  asp_valid = -777;

  if (obj_kind = 'miniapp') then
  begin
      if (obj_id is null) then  -- если заявка еще не создана, обращаемся к временной таблице - temp_miniapp_client
      begin
          for select mac_from_c_id,
                     f_asp_valid(mac_from_c_id, current_date)
          from temp_miniapp_client
          where mac_amount > 0
          into :c_id, :asp_valid_tmp
          do
          begin
            if (asp_valid < asp_valid_tmp) then
                asp_valid = asp_valid_tmp;
    
            if (asp_valid < 2) then
            begin
                select cl.c_name_fio
                from clients cl
                where cl.c_id = :c_id
                into :c_name_fio;
            end

            if (asp_valid = 0) then
                invalid_fio_list = invalid_fio_list || c_name_fio || ';';
            if (asp_valid = 1) then
                offline_fio_list = offline_fio_list || c_name_fio || ';';

          end
      end
      else    -- если заявка создана, тогда обращаемся к реальной таблице  - miniappclient
      begin
         for select mac_from_c_id,
                    f_asp_valid(mac_from_c_id, current_date)
          from miniappclient mac
          where mac_amount > 0
          and mac.mac_from_ma_id = :obj_id
          into :c_id, :asp_valid_tmp
          do
          begin
            if (asp_valid < asp_valid_tmp) then
                asp_valid = asp_valid_tmp;
    
            if (asp_valid < 2) then
            begin
                select cl.c_name_fio
                from clients cl
                where cl.c_id = :c_id
                into :c_name_fio;
            end

            if (asp_valid = 0) then
                invalid_fio_list = invalid_fio_list || c_name_fio || ';';
            if (asp_valid = 1) then
                offline_fio_list = offline_fio_list || c_name_fio || ';';
          end
      end
  end

  if (obj_kind = 'kontract') then
  begin
         for select kc_from_c_id,
                    f_asp_valid(kc_from_c_id, current_date)
          from knclient
          where kc_amount_approved > 0
          and kc_from_k_id = :obj_id
          into :c_id, :asp_valid_tmp
          do
          begin
            if (asp_valid < asp_valid_tmp) then
                asp_valid = asp_valid_tmp;
    
            if (asp_valid < 2) then
            begin
                select cl.c_name_fio
                from clients cl
                where cl.c_id = :c_id
                into :c_name_fio;
            end

            if (asp_valid = 0) then
                invalid_fio_list = invalid_fio_list || c_name_fio || ';';
            if (asp_valid = 1) then
                offline_fio_list = offline_fio_list || c_name_fio || ';';
          end
  end 

  if (invalid_fio_list <> '') then
   invalid_fio_list = substring(invalid_fio_list from 1 for character_length(invalid_fio_list) -1);

  if (offline_fio_list <> '') then
   offline_fio_list = substring(offline_fio_list from 1 for character_length(offline_fio_list) -1);

  suspend;
end
