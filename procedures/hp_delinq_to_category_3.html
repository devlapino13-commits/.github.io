CREATE PROCEDURE HP_DELINQ_TO_CATEGORY_3(
    REP_DATE_END DATE)
RETURNS (
    K_ID ID_TYPE,
    REG_ID INTEGER,
    KC_ID ID_TYPE,
    KA_ID ID_TYPE,
    KA_AKT_NO INTEGER)
AS
declare variable reg_id_prior integer;
declare variable ka_memo varchar(255);
declare variable kf_id id_type;
declare variable kal_new_kf_name varchar(100);
declare variable kal_new_no integer;
declare variable kal_new_officer_code varchar(20);
begin
  /*    if (not exists(select * from company
                   where cm_database_id = 81001)
        or :rep_date_end >= '01.01.2016' -- добавлено от 02.01.2016 Улан
        ) then exit;  */
  if ((f_get_db_id() <> 81001) or (:rep_date_end >= '01.01.2016')) -- добавлено от 02.01.2016 Улан
  then
    exit;

  ka_memo = 'Авто-Передача кред. папок с просроченными днями более 180 дней "Виртуальному сотруднику (Категория 3)"';
  reg_id_prior = 0;
  for select oks_from_k_id, oks_region_id, oks_kc_id
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            cast(:rep_date_end as date) - oks_delinq_date > 180 and
            oks_balance > 0 and
            oks_kc_id <> 1000000000003 and
            oks_close_state = 0
      order by oks_region_id
      into k_id, reg_id, kc_id
  do
  begin
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      ka_id = (select kv
               from get_kv(:reg_id, 7));
      ka_akt_no = coalesce((select max(cast(ka_akt_no as integer))
                            from knakt
                            where ka_from_reg_id = :reg_id), 0) + 1;
      insert into knakt (ka_id, ka_date, ka_from_reg_id, ka_akt_no, ka_memo, is_delete, ka_from_ol_id)
      values (:ka_id, :rep_date_end, :reg_id, :ka_akt_no, :ka_memo, 0, null);
    end
    kf_id = (select k_from_kf_id
             from kontract
             where k_id = :k_id);
    kal_new_kf_name = null;
    kal_new_no = null;
    kal_new_officer_code = null;
    select kal_new_kf_name, kal_new_no, kal_new_officer_code
    from knaktlist
    where knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_from_kf_id = :kf_id
    into kal_new_kf_name, kal_new_no, kal_new_officer_code;
    kal_new_kf_name = kal_new_kf_name || '. ВСК-3';

    insert into knaktlist (kal_from_ka_id, kal_number, kal_from_kf_id, kal_from_k_id, kal_date, kal_date_end,
                           kal_new_kf_officer, kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no,
                           kal_new_officer_code)
    values (:ka_id, -1, :kf_id, :kf_id, :rep_date_end, '31.12.9999', 1000000000003, :kal_new_kf_name, 0, :reg_id,
            :kal_new_no, :kal_new_officer_code);

--      suspend;
  end
end
