CREATE PROCEDURE EXPORT_1C_CLIENTS
RETURNS (
    C_ID VARCHAR(20),
    C_TYPE INTEGER,
    C_NAME_FIO VARCHAR(100),
    C_SEX SMALLINT,
    C_BIRTH_DATE DATE,
    C_NATION INTEGER,
    C_CITIZEN INTEGER,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_FROM_REG_ID INTEGER,
    C_EMAIL VARCHAR(100),
    C_FAMILY INTEGER,
    C_EDUCATION INTEGER,
    C_KOL_FAMILY INTEGER,
    C_KOL_CHILD INTEGER,
    C_MEMO VARCHAR(1024),
    C_REG_NO VARCHAR(20),
    C_OKPO VARCHAR(20),
    C_INN VARCHAR(20),
    C_ACCOUNT VARCHAR(20),
    C_ORG_FROM_B_ID VARCHAR(10),
    C_HIRED_DATE DATE,
    C_IS_RESIDENT SMALLINT,
    C_ARMY INTEGER,
    C_TABEL VARCHAR(10),
    C_PLASTCARD_NO VARCHAR(20),
    C_OTHERS VARCHAR(20),
    C_IS_PERSONAL SMALLINT,
    C_DOC_TYPE INTEGER,
    C_FULL_ADDRESS VARCHAR(255),
    C_FROM_PL_ID DECIMAL(18,0),
    C_FORM_STATUS_LONG VARCHAR(255),
    C_FORM_STATUS_SHORT VARCHAR(20),
    C_ORG_TYPE_LONG VARCHAR(255),
    C_ORG_TYPE_SHORT VARCHAR(20),
    C_CELL_OPERATOR INTEGER,
    C_ORIGINAL_ID DECIMAL(18,0),
    C_IS_EMPLOYER INTEGER,
    C_CLIENT_CIRCLE INTEGER,
    PK_NAME VARCHAR(100),
    DD_NAME VARCHAR(100),
    DD_NAME1 VARCHAR(100))
AS
declare variable c_id_tmp decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable cs_staff_state integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  c_email = '';
  c_plastcard_no = '';
  c_cell_operator = 0;

  for select pk_name, d1.rd_name, d2.rd_name, c_id, c_type, c_name_fio, c_sex, c_birth_date, c_nation, c_citizen,
             c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_from_reg_id, c_family, c_education, c_kol_family,
             c_kol_child, c_memo, c_reg_no, c_okpo, c_inn, c_account, c_org_from_b_id, c_hired_date, c_is_resident,
             c_army, c_tabel, c_others, c_is_personal, c_doc_type, adl_address_str, adl_from_pl_id, c_form_status_long,
             c_form_status_short, c_org_type_long, c_org_type_short, c_original_id
      from clients, passkind, refdata d1, refdata d2, addrclient, addrlist
      where c_doc_type = pk_id and
            c_nation = d1.rd_id and
            c_citizen = d2.rd_id and
            c_id = adc_from_c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      order by c_id desc
      rows 30000
      into :pk_name, :dd_name, :dd_name1, :c_id_tmp, :c_type, :c_name_fio, :c_sex, :c_birth_date, :c_nation, :c_citizen,
           :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_from_reg_id, :c_family, :c_education,
           :c_kol_family, :c_kol_child, :c_memo, :c_reg_no, :c_okpo, :c_inn, :c_account, :c_org_from_b_id,
           :c_hired_date, :c_is_resident, :c_army, :c_tabel, :c_others, :c_is_personal, :c_doc_type, :c_full_address,
           :c_from_pl_id, :c_form_status_long, :c_form_status_short, :c_org_type_long, :c_org_type_short,
           :c_original_id
  do
  begin
    cs_staff_state = -1;
    c_client_circle = 0;
    select cs_staff_state, cs_client_circle
    from clientstate
    where cs_from_c_id = :c_id_tmp
    into cs_staff_state, c_client_circle;
    if (cs_staff_state is null) then
      cs_staff_state = -1;
    if (c_client_circle is null) then
      c_client_circle = 0;

    c_is_employer = case
                      when cs_staff_state = -1 then 0
                      when cs_staff_state = 1 then 1
                      when cs_staff_state = 0 then 2
                    end;

-- cs_staff_state:
-- -1 нет кадровых приказов
-- 0 бывший сотрудник
-- 1 действующий сотрудник.

--  Раньше c_is_employer был:
-- 0 - не сотрудник
-- 1 - сотрудник
-- 2 - бывш.сотруд.

    c_id = 'N' || cast(c_id_tmp as varchar(15));
    suspend;
  end
end
