CREATE PROCEDURE AAREP_LAW_EXECUTORY_PROCESS(
    REP_DATE DATE)
RETURNS (
    LAM_ID INTEGER,
    LAM_DELO_NO STR_100,
    LAM_DATE_BEGIN DATE,
    LAM_FINAL_RESULT_STR STR_255,
    LAM_FINAL_RESULT_DATE DATE,
    LAL_NAME STR_255,
    KF_NAME STR_100,
    KONTRACT_NO STR_100,
    CLIENT_FIO STR_100,
    C_BIRTH_DATE DATE,
    C_CITIZEN STR_100,
    C_PASS_NO VARCHAR(30),
    C_PASS_WHO STR_255,
    C_PASS_DATE DATE,
    C_ADDRESS_YUR STR_255,
    C_ADDRESS_FACT STR_255,
    C_ADDRESS_RODINA_YUR STR_255,
    C_ADDRESS_RODINA_FACT STR_255,
    IN_BLACK_LIST VARCHAR(10),
    CL_ZAD_TOTAL CURR_TYPE,
    OWN_PAYED_AMOUNT CURR_TYPE,
    MAX_PAYED_DATE DATE,
    LAK_IL_ZADOL CURR_TYPE,
    LAK_NATIONAL_DUTY CURR_TYPE,
    LAK_FROM_PL_NAME STR_255,
    LAK_EB STR_255,
    LAK_EL_FILING_DATE DATE,
    LAK_FILING_TYPE STR_100,
    LAK_ENF_NUMBER STR_255,
    LAK_ENF_PROC_DATE DATE,
    LAK_BAILIFFS_CONTACTS STR_255,
    LAK_REST_MOV_FROM DATE,
    LAK_REST_MOV_IN DATE,
    LAK_FMC STR_255,
    LAK_SEARCH STR_255,
    LAK_HL_PENALTY CURR_TYPE,
    LAK_PROC_STOP_DATE DATE,
    LAK_STOP_REASON STR_255,
    LAK_MEMO STR_255)
AS
declare variable lam_from_lal_id integer;
begin
  for select lm.lam_id, lm.lam_delo_no, lm.lam_date_begin, lm.lam_final_result_str, lm.lam_final_result_date,
             lm.lam_from_lal_id
      from lawmain lm
      where lm.lam_is_active = 1 and
            lm.lam_date_begin <= :rep_date
      into :lam_id, :lam_delo_no, :lam_date_begin, :lam_final_result_str, :lam_final_result_date, :lam_from_lal_id
  do
  begin
    lal_name = '';

    select ll.lal_name
    from lawlist ll
    where ll.lal_id = :lam_from_lal_id
    into :lal_name;

    lal_name = coalesce(lal_name, '');

    for select kf_name, kontract_no, client_fio, c_birth_date, c_citizen, c_pass_no, c_pass_who, c_pass_date,
               c_address_yur, c_address_fact, c_address_rodina_yur, c_address_rodina_fact,
               iif(in_black_list = 1, 'Да', 'Нет') as in_black_list, cl_zad_total, own_payed_amount,
               max_payed_date, lak_il_zadol, lak_national_duty, lak_from_pl_name, lak_eb, lak_el_filing_date,
               lak_filing_type, lak_enf_number, lak_enf_proc_date, lak_bailiffs_contacts, lak_rest_mov_from,
               lak_rest_mov_in, lak_fmc, lak_search, lak_hl_penalty, lak_proc_stop_date, lak_stop_reason, lak_memo
        from law_executory_process(:lam_id, :rep_date, 1)
        into kf_name, kontract_no, client_fio, c_birth_date, c_citizen, c_pass_no, c_pass_who, c_pass_date,
             c_address_yur, c_address_fact, c_address_rodina_yur, c_address_rodina_fact, in_black_list, cl_zad_total,
             own_payed_amount, max_payed_date, lak_il_zadol, lak_national_duty, lak_from_pl_name, lak_eb,
             lak_el_filing_date, lak_filing_type, lak_enf_number, lak_enf_proc_date, lak_bailiffs_contacts,
             lak_rest_mov_from, lak_rest_mov_in, lak_fmc, lak_search, lak_hl_penalty, lak_proc_stop_date,
             lak_stop_reason, lak_memo
    do
    begin
      suspend;
    end
  end
end
