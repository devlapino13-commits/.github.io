CREATE PROCEDURE HP_CASHTERMINAL_UPDATE(
    TM_CIL_ID D_INTEGER,
    TM_CASH_ID D_INTEGER,
    TM_DATE D_DATE,
    TM_MANAGER_ID D_ID,
    TM_AMOUNT D_CURRENCY,
    TM_AMOUNT$BOND D_STR_LONG,
    TM_FACT_AMOUNT D_CURRENCY,
    TM_FACT_AMOUNT$BOND D_STR_LONG,
    TM_ID D_INTEGER,
    TM_DOC_NO D_INTEGER,
    TM_PERIOD_BEGIN D_DATETIME,
    TM_PERIOD_END D_DATETIME,
    TM_FROM_BLS_ID D_ID,
    PERSON_ID D_ID,
    IN_DIFF_KIND D_SMALLINT,
    UN_ID D_ID)
AS
declare variable cashier_id d_id;
declare variable oper_kind d_integer;
declare variable res_cil_id d_integer;
declare variable cfl_id d_integer;
begin
  if (tm_amount$bond is null) then
    tm_amount$bond = '';

  if (un_id is null) then
    un_id = uxf$user();

  oper_kind = f_enum_id('cashoperkind', 'plus.inkassa');

  select csp_cashier_id
  from cashplace
  where csp_id = :tm_cash_id
  into cashier_id;

  if (cashier_id is null) then
    exception e_message 'Кассир не определен';

  if (tm_manager_id is null) then
    exception e_message 'Менеджер/контролер не указан';

  if (coalesce(tm_amount, 0) = 0) then
    exception e_message 'Сумма по чеку не заполнена';

  if (coalesce(tm_cil_id, 0) <> 0) then
  begin
    res_cil_id = tm_cil_id;

    update cashinvlist
    set cil_inspector_id = :tm_manager_id,
        cil_inv_amount = :tm_amount,
        cil_from_un_id = :un_id,
        cil_from_bls_id = coalesce(:tm_from_bls_id, cil_from_bls_id)
    where cil_id = :tm_cil_id;

    execute procedure hp_cashbond_update(:tm_cil_id, :tm_amount$bond);

    update cashterminal
    set ctm_doc_no = :tm_doc_no,
        ctm_period_begin = :tm_period_begin,
        ctm_period_end = :tm_period_end
    where ctm_from_cil_id = :tm_cil_id;
  end else
  begin
    insert into cashinvlist (cil_from_csp_id, cil_date, cil_oper_kind, cil_from_bls_id,
                             cil_cashier_id, cil_inspector_id, cil_inv_amount, cil_from_un_id)
    values (:tm_cash_id, :tm_date, :oper_kind, :tm_from_bls_id,
            :cashier_id, :tm_manager_id, :tm_amount, :un_id)
    returning cil_id
    into :res_cil_id;
  
    if (tm_amount$bond <> '') then
      execute procedure hp_cashbond_update(:res_cil_id, :tm_amount$bond);

    insert into cashterminal (ctm_from_cil_id, ctm_from_tm_id, ctm_doc_no, ctm_period_begin, ctm_period_end)
    values (:res_cil_id, :tm_id, :tm_doc_no, :tm_period_begin, :tm_period_end);
  end

  if (person_id is not null) then
    execute procedure hp_cashinvmembers_update(:res_cil_id, '-1', :person_id, 0, :un_id);

  if (tm_fact_amount is null) then
    tm_fact_amount = 0;

  if (tm_fact_amount$bond is null) then
    tm_fact_amount$bond = '';

  if (in_diff_kind = 0) then
  begin
    tm_fact_amount = tm_amount;
    tm_fact_amount$bond = tm_amount$bond;
  end

  if (in_diff_kind <> 2) then
  begin
    update or insert into cashinvlist (cil_parent_id, cil_from_csp_id, cil_date, cil_oper_kind,
                             cil_cashier_id, cil_inspector_id, cil_inv_amount, cil_is_delete, cil_from_un_id)
    values (:res_cil_id, :tm_cash_id, :tm_date, :oper_kind,
            :cashier_id, :tm_manager_id, :tm_fact_amount, 0, :un_id)
    matching (cil_parent_id, cil_is_delete)
    returning cil_id
    into :cfl_id;

    execute procedure hp_cashbond_update(:cfl_id, :tm_fact_amount$bond);
    execute procedure hp_cashinvmembers_update(:cfl_id, '-1', :person_id, 1, :un_id);
    execute procedure hp_cashterminal_book_save(:cfl_id, :un_id);
  end
end
