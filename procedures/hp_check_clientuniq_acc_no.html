CREATE PROCEDURE HP_CHECK_CLIENTUNIQ_ACC_NO(
    UNIQ_BIRTH_DATE DATE,
    UNIQ_ACC INTEGER)
RETURNS (
    C_ID DECIMAL(18,0))
AS
declare variable control_digit integer;
declare variable db_inc integer;
declare variable total integer;
declare variable c_ext_id integer;
begin
-- Входящий параметр: дата рождение и лицевой счет.
-- Лицевой счет уже содержит в конце контрольный разряд.
-- Данная процедура проверяет правильность контрольного разряда
-- Если все нормально, C_ID
-- Если нет клиента с таким лицевым счетом или разряд неправильный, результат -1

  c_id = -1;
  if ((uniq_birth_date is null) or (coalesce(uniq_acc, 0) < 10)) then
  begin
    suspend;
    exit;
  end

  control_digit = mod(uniq_acc, 10);
  c_ext_id = left(uniq_acc, char_length(uniq_acc) - 1);

  :db_inc = f_get_db_id();
  db_inc = (case db_inc
              when 41701 then 1
              when 81001 then 2
              when 94401 then 3
            end);

-- Формула расчета контрольного разряда: [День+Месяц+Год+(Порядковый номер * 97)] остаток от деления на 10 + КОД БД;
  total = extract(day from uniq_birth_date) + extract(month from uniq_birth_date) + extract(year from uniq_birth_date) + c_ext_id * 97 + db_inc;
  total = mod(total, 10);

  if (control_digit = mod(total, 10)) then -- если контрольный разряд совпадает
  begin
    select cunq_from_c_id
    from clientuniq
    where cunq_birth_date = :uniq_birth_date and
          cunq_ext_id = :c_ext_id
    order by cunq_datetime desc
    rows 1
    into :c_id;
  end
  suspend;
end
