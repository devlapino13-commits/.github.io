CREATE PROCEDURE OKT_OPER_COM_ACT(
    UN_ID D_ID,
    DATE_BEGIN D_DATE,
    COMMAND_TYPE D_STR_10,
    COMMAND_ID D_INTEGER)
RETURNS (
    ERROR_TXT D_STR_255)
AS
begin
  -- Процедура для закрепления/открепления команды сотрудника кол-центра
  -- Закрепить - fix
  -- Открепить - unfix
  error_txt = '';
  if (command_type = 'fix') then
  begin
    insert into pantera_commands_operators(pco_from_pcm_id, pco_from_un_id, pco_date_begin)
    values(:command_id, :un_id, :date_begin);
  end
  else if (command_type = 'unfix') then
  begin
    update pantera_commands_operators
    set pco_date_end = current_date
    where pco_from_un_id = :un_id and
          pco_from_pcm_id = :command_id and
          pco_date_end = '31.12.9999';
  end
  else
    error_txt = 'Неизвестный тип команды';
  suspend;
end
