CREATE PROCEDURE HP_EDOC_BLANK_IN_PROCESS(
    EDB_STATUS D_SMALLINT = 0,
    THREAD_ID D_INTEGER = 0,
    THREAD_COUNT D_INTEGER = 1)
RETURNS (
    EDB_ID D_ID,
    REG_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    UN_ID D_ID,
    BLN_ID D_INTEGER,
    BLB_FONT_SIZE D_INTEGER,
    BLB_FONT_STYLE D_STR_100,
    BLB_BLOB D_BLOB_BINARY,
    MANAGER_KSR_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG,
    HASH_KIND D_SMALLINT,
    EDL_ID D_ID)
AS
declare variable manager_c_id d_id;
declare variable edb_from_c_id d_id;
declare variable blob_date d_date;
declare variable tmp_obj_kind d_integer;
declare variable tmp_obj_id d_id;
declare variable edf_date_kind d_smallint;
declare variable edc_id d_integer;
begin
  -- Процедура вызывается из сервиса FsService подсистемой (потоками) подготовки
  -- бланков эл. документов
  if (edb_status not in (0, -1)) then
    exception e_default 'Invalid param edb_status, should be -1 or 0';

  -- Параметры thread_id, thread_count предназначены для разделения бланков
  -- электронных документов между несколькими потоками.
  -- thread_count - общее количество потоков, обрабатывающих бланки, 1 или более
  -- thread_id - номер текущего потока, от 0 до thread_count-1
  -- Таким образом если mod(bln_id, thread_count) = thread_id, то поток забирает текущий бланк.
  -- Если поток только один (thread_count = 1), возвращаются все бланки без разделения.
  -- Функция mod возвращает остаток от деления первого аргумента на второй.
  -- Пример: Допустим у нас два потока, т.е. thread_count = 2
  -- bln_id   mod(bln_id, thread_count)
  --   1        1
  --   2        0
  --   3        1
  --   4        0
  -- Т.е. для thread_id=0 процедура вернет бланки с bln_id 2 и 4,
  -- А для thread_id=1 процедура вернет бланки 1 и 3

  thread_id = coalesce(thread_id, 0);
  thread_count = coalesce(thread_count, 1);
  if (thread_count < 1) then
    exception e_default 'Invalid thread_count. Should be 1 or greater';

  if (thread_id >= thread_count) then
    exception e_default 'Invalid thread_id. Should be less than thread_count';

  for select edb_id, edb_from_bln_id, edl_object_kind, edl_object_id,
             edl_create_un_id, edl_blank_date, edb_from_c_id,
             edl_id, edl_from_reg_id, edl_from_edc_id, edl_manager_c_id,
             edf_date_kind,
             iif(edl_status = 1, edf_hash_kind, 0) -- Пока клиент не подписал, hash не нужен
      from edoc_blank
        inner join edoc_list on edb_from_edl_id = edl_id
        inner join edoc_frx on edb_from_edf_id = edf_id
      where edb_status = :edb_status
        and (edl_status <> -1)
        and (:thread_count = 1 or mod(edb_id, :thread_count) = :thread_id)
      order by edb_id desc -- Последние бланки в первую очередь
      rows 200 -- Перезапуск каждые 200 бланков
      into edb_id, bln_id, obj_kind, obj_id, un_id, date_begin,
            edb_from_c_id, edl_id, reg_id, edc_id, manager_c_id,
            edf_date_kind, hash_kind
  do
  begin
    obj_id = abs(obj_id);
    if (manager_c_id is null) then
    begin
      select edc_default_manager_id
      from edoc_cfg
      where edc_id = :edc_id
      into manager_c_id;
    end

    add_params = '';

    -- Поля для вычисления
    if (edb_from_c_id is not null) then
      add_params = f_format('{"edl_id":%1,"c_id":%2}', edl_id, edb_from_c_id);
    else
      add_params = f_format('{"edl_id":%1}', edl_id);

    blob_date = date_begin;
    if (edf_date_kind = 1) then -- версию бланка нужно брать на дату выдачи займа
    begin
      tmp_obj_kind = obj_kind;
      tmp_obj_id = obj_id;
      if (f_objkind(obj_kind, 'name') = 'com_list') then
      begin
        select cml_object_kind, cml_object_id
        from com_list
        where cml_id = :obj_id
        into tmp_obj_kind, tmp_obj_id;
      end

      select blank_date
      from hp_edoc_object_info(:tmp_obj_kind, :tmp_obj_id, :un_id, null)
      into :blob_date;
    end

    if (date_begin is null) then
      date_begin = current_date;
    date_end = date_begin;

    if (reg_id is null) then
    begin
      select reg_id
      from hp_edoc_object_info(:obj_kind, :obj_id, :un_id, :date_begin)
      into :reg_id;
    end

    -- Получаем подписанта, приоритет у активных, потом у последних добавленных
    -- Т.е. допускаем что подписант может стать неактивным, когда клиент подписал
    select ksr_id
    from kontsignmen, kontsignreg, region_list(:reg_id, 1, -1, 0)
    where ksm_id = ksr_from_ksm_id and
          ksm_from_c_id = :manager_c_id and
          ksr_from_reg_id = id_list and
          coalesce(ksr_date_begin, '01.01.1990') <= :date_begin and
          coalesce(ksr_date_end, '31.12.9999') >= :date_begin
    order by (ksr_is_active + ksm_is_active) desc, ksr_id desc
    rows 1
    into manager_ksr_id;

    select iif(bln_show_mode = 0, blb_blob, blb_frx_blob), blb_font_style, blb_font_size
    from blankblob, blankname
    where blb_from_bln_id = bln_id and
          blb_date_begin <= :blob_date and
          blb_date_end >= :blob_date and
          blb_from_bln_id = :bln_id
    into blb_blob, blb_font_style, blb_font_size;

    if (blb_blob is not null) then
      suspend;
  end
end
