CREATE PROCEDURE MC_OFFICE_LIST
RETURNS (
    OFFICE_ID D_INTEGER,
    OFFICE_NAME D_STR_100,
    OFFICE_CITY_ID D_ID,
    ADDRESS D_STR_255,
    LONGITUDE PERC_TYPE,
    LATITUDE PERC_TYPE)
AS
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  for select reg.reg_unicode,
             reg.reg_name,
             pl.pl_id,
             adl.adl_address_str,
             coalesce(adl.adl_long,0),
             coalesce(adl.adl_lat,0)
  from places pl
  inner join addrlist adl on adl.adl_from_pl_id = pl.pl_id
  inner join region reg on reg.reg_address_fact = adl.adl_id
  where reg.reg_date_close is null and
        reg.is_delete = 0 and
        reg.reg_status = 3
  order by pl.pl_name
  into office_id,
       office_name,
       office_city_id,
       address,
       longitude,
       latitude
  do
  begin
    if  ((db_id = 81001) and
         (office_id in (1012100, 1012300, 1010100, 1012700, 1490700, 1280300, 1012800))) then
            continue;

        suspend;
  end
end
