CREATE PROCEDURE HP_SUD_LIGRES_LIST(
    MODE D_SMALLINT,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    SMR_ID D_ID,
    SLC_ID D_ID,
    K_KONTRACT_NO D_STR_CODE,
    C_NAME_FIO D_STR_255,
    STATUS D_STR_255)
AS
declare variable c_id d_id;
declare variable eil_status d_smallint;
declare variable eil_date d_date;
declare variable collect_begin_date d_date;
declare variable obj_kind d_integer;
begin
  -- mode: 0 - готовы к отправке запроса, 1 - отправленные запросы, 2 - нет информации
  mode = coalesce(mode, 0);
  obj_kind = f_objkind('sud_loanclient', 'id');

  if (mode = 0) then
  begin
    -- стадии C1-C4
    -- «Дата подачи» во вкладке «ФССП» больше > 3 месяцев от текущей даты
    -- последний удачный запрос старше 3 месяцев

    for select slc_id, c_id, c_name_fio, smc_begin_date, slc_from_smr_id, k_kontract_no
    from sud_maincollect, sud_loanclient, sud_mainreq, knclient, clients, kontract, oksummary
    where slc_id = smc_from_slc_id
      and kc_id = slc_from_kc_id
      and smr_id = slc_from_smr_id
      and c_id = kc_from_c_id
      and k_id = kc_from_k_id
      and smc_begin_date < dateadd(month, -1, current_date)
      and smc_status < 2
      and oks_from_k_id = k_id
      and current_date between oks_date_begin and oks_date_end
      and (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc + oks_out_other + oks_extra) > 0
    into slc_id, c_id, c_name_fio, collect_begin_date, smr_id, k_kontract_no
    do
    begin
      eil_status = null;
      eil_date = null;

      select eil_status, eil_date
      from extinfolist, extinfocfg
      where eic_id = eil_from_eic_id
        and eic_code = 'ligres_fssp_cli_check'
        and eil_from_c_id = :c_id
        and eil_date >= :collect_begin_date
      order by eil_date desc, eil_id desc
      rows 1
      into eil_status, eil_date;

      if (eil_status = 2 and eil_date > dateadd(month, -1, current_date)) then
        continue;

      if (eil_date is null) then
        status = '-';
      else
        status = f_datetostr(eil_date) || ' ' || decode(eil_status, -2, 'Ошибка', -1, 'Ошибка', 0, 'В очереди', 1, 'Отправлен', 2, 'Ответ получен', 'Неизвестно');

      suspend;
    end
  end
  ---------------------------------
  else if (mode = 1) then
  begin
    for select slc_id, max(c_id), max(c_name_fio), max(smc_begin_date), max(slc_from_smr_id), max(k_kontract_no), max(eil_date)
    from sud_maincollect, sud_loanclient, knclient, clients, kontract, extinfolist, extinfocfg
    where slc_id = smc_from_slc_id
      and kc_id = slc_from_kc_id
      and c_id = kc_from_c_id
      and k_id = kc_from_k_id
      and eil_from_c_id = c_id
      and eic_id = eil_from_eic_id
      and eic_code = 'ligres_fssp_cli_check'
      and eil_date between :rep_date_begin and :rep_date_end
      and eil_status = 2
      and not exists(select 1 from sud_object_list where sol_kind = 1 and sol_object_kind = :obj_kind and sol_object_id = slc_id)
    group by 1
    into slc_id, c_id, c_name_fio, collect_begin_date, smr_id, k_kontract_no, eil_date
    do
    begin
      status = f_datetostr(eil_date);
      suspend;
    end
  end
  ---------------------------------
  else if (mode = 2) then
  begin
    for select slc_id, max(c_id), max(c_name_fio), max(smc_begin_date), max(slc_from_smr_id), max(k_kontract_no), max(eil_date)
    from sud_object_list, sud_maincollect, sud_loanclient, knclient, clients, kontract, extinfolist, extinfocfg
    where sol_kind = 1
      and sol_object_kind = :obj_kind
      and sol_object_id = slc_id
      and slc_id = smc_from_slc_id
      and kc_id = slc_from_kc_id
      and c_id = kc_from_c_id
      and k_id = kc_from_k_id
      and eil_from_c_id = c_id
      and eic_id = eil_from_eic_id
      and eic_code = 'ligres_fssp_cli_check'
      and eil_status = 2
    group by 1
    into slc_id, c_id, c_name_fio, collect_begin_date, smr_id, k_kontract_no, eil_date
    do
    begin
      status = f_datetostr(eil_date);
      suspend;
    end
  end
end
