CREATE PROCEDURE HP_BKI_XML_LINK_NODES(
    XML_VERSION_ID D_ID,
    TARGET_BFV_ID D_ID,
    BXF_ID D_ID,
    MODE D_SMALLINT,
    BFV_FORCED_VALUE D_STR_255 = null)
AS
declare variable bfv_id d_id;
declare variable parent_id d_id;
declare variable old_pos_in_block d_integer;
declare variable bki_db_field d_str_code;
declare variable pos_in_block d_integer;
begin
  -- процедура создает, удаляет подчиненности между узлами XML
  -- mode: 1 - добавить внутрь target_bxf_id, 2 - добавить до target_bxf_id, 3 - удалить связь
  select bki_db_field, iif(:mode = 1, bfv_id, bfv_parent_id), bfv_position
  from bki_xml_fields_ver, bki_xml_fields, bki_indicators
  where bfv_id = :target_bfv_id
    and bxf_id = bfv_from_bxf_id
    and bki_code = bxf_from_bki_code
  into bki_db_field, parent_id, pos_in_block;

  if (row_count = 0) then
    exception e_message 'Target not found';

  select bfv_id, bfv_position
  from bki_xml_fields_ver
  where bfv_from_bxf_id = :bxf_id
    and bfv_from_bxv_id = :xml_version_id
    and bfv_parent_id is not distinct from :parent_id
  into bfv_id, old_pos_in_block;

  if (mode in (1,2)) then
  begin
    if (mode = 1) then
    begin
      if (coalesce(bki_db_field, '') not in ('<OBJ>', '<ARR>')) then
        exception e_message 'Target is not group';

      pos_in_block = 0;

      select max(bfv_position)
      from bki_xml_fields_ver
      where bfv_from_bxv_id = :xml_version_id
        and bfv_parent_id is not distinct from :parent_id
      into pos_in_block;

      pos_in_block = coalesce(pos_in_block, 0) + 1;
    end 

    if (old_pos_in_block = pos_in_block) then
      exit;

    -- надо выделить место
    update bki_xml_fields_ver
    set bfv_position = bfv_position + 1
    where bfv_from_bxv_id = :xml_version_id
      and bfv_parent_id is not distinct from :parent_id
      and bfv_position >= :pos_in_block
    order by bfv_position desc;

    if (old_pos_in_block is not null) then
    begin
      -- перемещаем
      update bki_xml_fields_ver
      set bfv_position = :pos_in_block
      where bfv_id = :bfv_id;
      -- закроем дыру после перемещения
      update bki_xml_fields_ver
      set bfv_position = bfv_position - 1
      where bfv_from_bxv_id = :xml_version_id
        and bfv_parent_id is not distinct from :parent_id
        and bfv_position > :old_pos_in_block
      order by bfv_position;
    end
    else
      insert into bki_xml_fields_ver (bfv_from_bxv_id, bfv_position, bfv_from_bxf_id, bfv_parent_id, bfv_forced_value)
      values (:xml_version_id, :pos_in_block, :bxf_id, :parent_id, :bfv_forced_value);
  end 
  else if (mode = 3) then
  begin
    if (target_bfv_id is null) then
      exception e_message 'Not exists';

    delete from bki_xml_fields_ver
    where bfv_id = :target_bfv_id;

    -- закроем дыру после удаления
    update bki_xml_fields_ver
    set bfv_position = bfv_position - 1
    where bfv_from_bxv_id = :xml_version_id
      and bfv_parent_id is not distinct from :parent_id
      and bfv_position > :pos_in_block
    order by bfv_position;
  end
  else
    exception e_message 'Unknown mode ' || coalesce(mode, '<NULL>');
end
