CREATE PROCEDURE CM_KG_0220_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable ma_date_oper d_date;
declare variable k_disb_date d_date;
declare variable kc_amount_approved d_currency;
declare variable mac_amount d_currency;
begin
  --JIRA 1497
  for select mac.mac_from_c_id, ma.ma_date_oper, cl.c_name_fio, mac.mac_amount
      from miniapp ma
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
      inner join clients cl on cl.c_id = mac.mac_from_c_id
      where ma.ma_id = :ma_id and
            ma.ma_is_active = 1 and
            ma.ma_date_end >= '31.12.2999' and
            mac.mac_amount > 0 and
            (pr.pr_code containing '.par.' and
             pr.pr_code not containing '.standart.') --связка этих pr_code - ишеним плюс
      into c_id, ma_date_oper, res_info_text, mac_amount
  do
  begin
    k_disb_date = null;
    kc_amount_approved = 0;
    select max(ks.ks_date_begin), max(kc.kc_amount_approved)
    from kontract kt
    inner join kontstate ks on ks.ks_from_k_id = kt.k_id
    inner join knclient kc on kc.kc_from_k_id = kt.k_id
    inner join product pr on pr.pr_id = kt.k_from_pr_id
    where kc.kc_from_c_id = :c_id and
          kc.kc_amount_approved > 0 and
          pr.pr_code containing '.consultation.' and
          ks.ks_state = 1 and
          ks.ks_ol_is_delete = 0 and
          ks.ks_date_begin <= :ma_date_oper and
          :ma_date_oper between ks.ks_date_begin and ks.ks_date_end  --активный на дату заявки
    into k_disb_date, kc_amount_approved;

    if (datediff(day, :k_disb_date, :ma_date_oper) <= 90) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Со дня получения последнего займа по продукту Ишеним не прошло 90 дней. Клиент: ' || :res_info_text ;
      suspend;
    end
    if (mac_amount > kc_amount_approved) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Сумма основного займа по "Ишеним" меньше , суммы указанной в заявке. Клиент: ' || :res_info_text ;
      suspend;
    end 

    if (exists(select *
               from kontract kt
               inner join kontstate ks on ks.ks_from_k_id = kt.k_id
               inner join knclient kc on kc.kc_from_k_id = kt.k_id
               inner join product pr on pr.pr_id = kt.k_from_pr_id
               where kc.kc_from_c_id = :c_id and
                     kc.kc_amount_approved > 0 and
                     ks.ks_state = 1 and
                     ks.ks_ol_is_delete = 0 and
                     :ma_date_oper between ks.ks_date_begin and ks.ks_date_end and
                     (pr.pr_code containing '.par.' and
                      pr.pr_code not containing '.standart.'))) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Клиент не может иметь больше одного займа по "Ишеним плюс". Клиент: ' || :res_info_text ;
      suspend;
    end
  end
end
