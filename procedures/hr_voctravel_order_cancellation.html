CREATE PROCEDURE HR_VOCTRAVEL_ORDER_CANCELLATION(
    HO_ID ID_TYPE,
    NUMBER INTEGER,
    HO_FROM_UN_ID ID_TYPE)
AS
declare variable subject_kind integer;
declare variable subject_id id_type;
declare variable un_id id_type;
declare variable pm_id integer;
declare variable hrv_date_begin date;
declare variable hrv_date_end date;
begin
  for select un.un_id, hrv.hrv_date_begin, hrv.hrv_date_end_fact
      from hr_voctravel hrv
      inner join username un on un.un_from_c_id = hrv.hrv_from_c_id
      where hrv.hrv_from_ho_id = :ho_id and
            hrv.hrv_number = :number
      into :un_id, :hrv_date_begin, :hrv_date_end

  do
  begin
    for select pm.pm_id, pm.pm_subject_kind, pm.pm_subject_id
        from permissions pm
        where current_date between :hrv_date_begin and :hrv_date_end and
              pm.pm_object_id = :un_id and
              pm.pm_is_delete = 0 and
              pm.pm_access_kind = -1
        into :pm_id, :subject_kind, :subject_id
    do
    begin
      update permissions
      set pm_is_delete = 1
      where (pm_id = :pm_id);

      insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_date_begin,
                               pm_date_end, pm_access_kind, pm_is_delete, pm_from_un_id, pm_datetime, pm_delete_un_id,
                               pm_delete_datetime)
      values (null, 0, :un_id, :subject_kind, :subject_id, current_date, null, 1, 0, :ho_from_un_id, current_timestamp,
              null, null);
    end
  end
end
