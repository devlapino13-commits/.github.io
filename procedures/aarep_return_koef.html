CREATE PROCEDURE AAREP_RETURN_KOEF(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PR_COMMON_NAME VARCHAR(255),
    AAREP_PLAN_OS DECIMAL(18,4),
    AAREP_PLAN_PERC DECIMAL(18,4),
    AAREP_REG_ID INTEGER,
    AAREP_LO_ID DECIMAL(18,0),
    AAREP_REPAY_YEAR_MONTH INTEGER,
    AAREP_FACT_OS DECIMAL(18,4),
    AAREP_FACT_PERC DECIMAL(18,4),
    AAREP_REPAY_WEEK INTEGER,
    AAREP_LO_FIO VARCHAR(100),
    REG_NAME VARCHAR(100),
    AAREP_LOAN_AMOUNT DECIMAL(18,4),
    AAREP_BALANCE_BEGIN DECIMAL(18,4),
    AAREP_BALANCE_END DECIMAL(18,4),
    AAREP_DELINQ60_BEGIN DECIMAL(18,4),
    AAREP_DELINQ180_BEGIN DECIMAL(18,4),
    AAREP_DELINQ60_END DECIMAL(18,4),
    AAREP_DELINQ180_END DECIMAL(18,4))
AS
begin
  for select pr_common_name, aarep_reg_id, aarep_lo_id, aarep_repay_year_month, sum(aarep_plan_os),
             sum(aarep_plan_perc), sum(aarep_fact_os), sum(aarep_fact_perc), sum(aarep_loan_amount),
             sum(aarep_balance_begin), sum(aarep_balance_end), sum(aarep_delinq60_begin), sum(aarep_delinq180_begin),
             sum(aarep_delinq60_end), sum(aarep_delinq180_end)
      from aarep_return_koef_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by pr_common_name, aarep_reg_id, aarep_lo_id, aarep_repay_year_month
      into pr_common_name, aarep_reg_id, aarep_lo_id, aarep_repay_year_month, aarep_plan_os, aarep_plan_perc,
           aarep_fact_os, aarep_fact_perc, :aarep_loan_amount, :aarep_balance_begin, :aarep_balance_end,
           :aarep_delinq60_begin, :aarep_delinq180_begin, :aarep_delinq60_end, :aarep_delinq180_end
  do
  begin
    select c_name_fio
    from clients
    where c_id = :aarep_lo_id
    into :aarep_lo_fio;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :aarep_reg_id
    into :reg_name;
    suspend;
  end
end
