CREATE PROCEDURE UXP$UXFRPARAM_DOUBLE_LIST(
    UFM_UUID D_UUID,
    FIELD_NAME D_STR_CODE)
RETURNS (
    UFP_UUID D_UUID,
    UFP_CODE D_STR_CODE,
    UFP_UFR_UUID D_UUID,
    UFP_UF_UUID D_UUID,
    UF_IS_PARAM D_BOOL,
    UF_PK_FIELD D_SMALLINT,
    UF_POSITION D_INTEGER,
    UF_DOMAIN D_STR_CODE,
    UF_DATA_TYPE D_STR_CODE,
    UF_NOT_NULL D_BOOL,
    UF_DEFAULT D_STR_100,
    UFP_REFRESH D_BOOL,
    UFP_NOT_NULL D_BOOL,
    UFP_EDITOR D_STR_CODE,
    UFP_DEFAULT D_STR_100,
    UFP_LOCKED D_BOOL,
    RES_FIELD_NAME D_STR_100,
    RES_IN_USE D_BOOL,
    RES_EDITOR_INFO D_STR_LONG,
    RES_CAPTION D_STR_100,
    RES_UCM_CODE D_STR_100,
    RES_UCM_IS_OWN D_BOOL,
    RES_BUTTONEDIT_UXA D_STR_100,
    RES_INDEX_TXT D_STR_CODE,
    RES_FIELD_KIND D_STR_10,
    RES_IN_UI D_SMALLINT,
    UEX_EXPRESSION D_STR_4096,
    RES_CATEGORY D_STR_255,
    RES_IS_FK D_BOOL,
    UFP_PUBLIC D_BOOL)
AS
declare variable fr_uuid d_uuid;
declare variable fr_name d_str_100;
begin
  if (coalesce(field_name, '') = '') then
    exit;

  for select ufr_uuid, ufr_name
  from ux$frame
  where ufr_owner = :ufm_uuid
  into fr_uuid, fr_name
  do
  begin
    if (not exists(select * from ux$frparam, ux$field
    where ufp_ufr_uuid = :fr_uuid and
          ufp_parent is null and
          ufp_uf_uuid = uf_uuid and
          uf_field_name = :field_name and
          (uf_is_param = 1 or uf_pk_field <> 0))) then
     Continue;

     for select ufp_uuid, ufp_code, ufp_ufr_uuid, ufp_uf_uuid, uf_is_param, uf_pk_field, uf_position, uf_domain, uf_data_type,
                uf_not_null, uf_default, ufp_refresh, ufp_not_null, ufp_editor, ufp_default, ufp_locked,
                res_field_name, res_in_use, res_editor_info, res_caption, res_ucm_code, res_ucm_is_own,
                res_buttonedit_uxa, res_index_txt, res_field_kind, res_in_ui, uex_expression,
                res_category, res_is_fk, ufp_public
     from uxp$uxfrparam_list(:fr_uuid, null, :field_name)
     into :ufp_uuid, :ufp_code, :ufp_ufr_uuid, :ufp_uf_uuid, :uf_is_param, :uf_pk_field, :uf_position, :uf_domain, :uf_data_type,
          :uf_not_null, :uf_default, :ufp_refresh, :ufp_not_null, :ufp_editor, :ufp_default, :ufp_locked,
          :res_field_name, :res_in_use, :res_editor_info, :res_caption, :res_ucm_code, :res_ucm_is_own,
          :res_buttonedit_uxa, :res_index_txt, :res_field_kind, :res_in_ui, :uex_expression,
          :res_category, :res_is_fk, ufp_public
     do
     begin
       res_field_name = fr_name;
       res_caption = res_category || ': ' || res_caption;
       suspend;
     end
  end
end
