CREATE PROCEDURE AAREP_AEIR(
    REP_REGION_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    ID_PARENT ID_TYPE,
    REG_NAME STR_100,
    AEIR_ACTIV_DAY PERC_TYPE,
    AEIR_ACTIV_MONTH PERC_TYPE,
    AEIR_ACTIV_YEAR PERC_TYPE,
    AEIR_BALANCE_DAY CURR_TYPE,
    AEIR_BALANCE_MONTH CURR_TYPE,
    AEIR_BALANCE_YEAR CURR_TYPE,
    AEIR_KS_DAY PERC_TYPE,
    AEIR_KS_MONTH PERC_TYPE,
    AEIR_KS_YEAR PERC_TYPE,
    AEIR_KS_DAY_BALANCE CURR_TYPE,
    AEIR_KS_MONTH_BALANCE CURR_TYPE,
    AEIR_KS_YEAR_BALANCE CURR_TYPE)
AS
declare variable rep_date_month date;
declare variable rep_date_year date;
declare variable aeir_activ_day_tmp perc_type;
declare variable aeir_activ_month_tmp perc_type;
declare variable aeir_activ_year_tmp perc_type;
declare variable aeir_ks_day_tmp perc_type;
declare variable aeir_ks_month_tmp perc_type;
declare variable aeir_ks_year_tmp perc_type;
begin
--Report on the annual effective interest rate
  for select distinct id_parent, parent_name
      from region_list(:rep_region_id, 0, 3, 1)
      where id_parent not in (1060000, 1090006, 1120000)
      into id_parent, reg_name
  do
  begin

    rep_date_month = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date);
    rep_date_year = cast('01.01.' || extract(year from :rep_date_end) as date);

--Баланс на дату отчета по активным
    select sum(oks_balance) balance
    from oksummary oks, region_list(:id_parent, 1, -1, 1)
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_region_id = id_list
    into aeir_balance_day;

    aeir_activ_day = 0;
    aeir_activ_month = 0;
    aeir_activ_year = 0;
    aeir_ks_day = 0;
    aeir_ks_month = 0;
    aeir_ks_year = 0;

--ГЭПС на дату отчета по активным
    for select oks_balance / :aeir_balance_day * kd_percent_effect
        from oksummary oks, kontdata, region_list(:id_parent, 1, -1, 1)
        where :rep_date_end between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_from_k_id = kd_from_k_id and
              kd_ol_is_delete = 0 and
              :rep_date_end between kd_ol_date and kd_date_end and
              oks_close_state = 0 and
              oks_region_id = id_list
        into aeir_activ_day_tmp
    do
    begin
      aeir_activ_day = aeir_activ_day + aeir_activ_day_tmp;
    end

--Баланс на начало месяца по активным
    select sum(oks_balance) balance
    from oksummary oks, region_list(:id_parent, 1, -1, 1)
    where :rep_date_month between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_region_id = id_list
    into aeir_balance_month;

--ГЭПС по активным на начала месяца
    for select oks_balance / :aeir_balance_month * kd_percent_effect
        from oksummary oks, kontdata, region_list(:id_parent, 1, -1, 1)
        where :rep_date_month between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_from_k_id = kd_from_k_id and
              kd_ol_is_delete = 0 and
              :rep_date_month between kd_ol_date and kd_date_end and
              oks_close_state = 0 and
              oks_region_id = id_list
        into aeir_activ_month_tmp
    do
    begin
      aeir_activ_month = aeir_activ_month + aeir_activ_month_tmp;
    end

--Баланс на начало года по активным
    select sum(oks_balance) balance
    from oksummary oks, region_list(:id_parent, 1, -1, 1)
    where :rep_date_year between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_region_id = id_list
    into aeir_balance_year;

--ГЭПС по активным на начала года
    for select oks_balance / :aeir_balance_year * kd_percent_effect
        from oksummary oks, kontdata, region_list(:id_parent, 1, -1, 1)
        where :rep_date_year between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_from_k_id = kd_from_k_id and
              kd_ol_is_delete = 0 and
              :rep_date_year between kd_ol_date and kd_date_end and
              oks_close_state = 0 and
              oks_region_id = id_list
        into aeir_activ_year_tmp
    do
    begin
      aeir_activ_year = aeir_activ_year + aeir_activ_year_tmp;
    end

--Сумма выданных на дату отчета
    select sum(k_amount_approved)
    from kontract k, kontstate ks, region_list(:id_parent, 1, -1, 1)
    where k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_date_begin = :rep_date_end and
          ks_region_id = id_list and
          ks_state = 1
    into aeir_ks_day_balance;

--ГЭПС на дату отчета по выданным
    for select k_amount_approved / :aeir_ks_day_balance * k.k_percent_effect
        from kontract k, kontstate ks, region_list(:id_parent, 1, -1, 1)
        where k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_date_begin = :rep_date_end and
              ks_region_id = id_list and
              ks_state = 1
        into aeir_ks_day_tmp
    do
    begin
      aeir_ks_day = aeir_ks_day + aeir_ks_day_tmp;
    end

--Общая сумма выданных на начало месяца отчета
    select sum(k_amount_approved)
    from kontract k, kontstate ks, region_list(:id_parent, 1, -1, 1)
    where k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_date_begin between :rep_date_month and :rep_date_end and
          ks_region_id = id_list and
          ks_state = 1
    into aeir_ks_month_balance;

--ГЭПС по выданным кредитам с начала месяца
    for select k_amount_approved / :aeir_ks_month_balance * k.k_percent_effect
        from kontract k, kontstate ks, region_list(:id_parent, 1, -1, 1)
        where k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_date_begin between :rep_date_month and :rep_date_end and
              ks_region_id = id_list and
              ks_state = 1
        into aeir_ks_month_tmp
    do
    begin
      aeir_ks_month = aeir_ks_month + aeir_ks_month_tmp;
    end

--Общая сумма выданых за один год
    select sum(k_amount_approved)
    from kontract k, kontstate ks, region_list(:id_parent, 1, -1, 1)
    where k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_date_begin between :rep_date_year and :rep_date_end and
          ks_region_id = id_list and
          ks_state = 1
    into aeir_ks_year_balance;

--ГЭПС по выданным кредитам с начала года
    for select k_amount_approved / :aeir_ks_year_balance * k.k_percent_effect
        from kontract k, kontstate ks, region_list(:id_parent, 1, -1, 1)
        where k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_date_begin between :rep_date_year and :rep_date_end and
              ks_region_id = id_list and
              ks_state = 1
        into aeir_ks_year_tmp
    do
    begin
      aeir_ks_year = aeir_ks_year + aeir_ks_year_tmp;
    end

    suspend;
  end
end
