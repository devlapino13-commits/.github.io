CREATE PROCEDURE MC_DOCRECOG_UPDATE_BLOB(
    DRL_ID D_ID,
    BLS_BLOB D_BLOB_BINARY,
    BLS_FILENAME D_STR_100,
    BLS_BLOB_SIZE D_FLOAT,
    PAGE_NO D_SMALLINT,
    SRC_KIND D_SMALLINT,
    MCL_ID D_ID)
AS
declare variable bls_id d_id;
declare variable bls_caption d_str_100;
declare variable pk_id d_integer;
declare variable need_photo2 d_bool;
declare variable un_id d_id;
declare variable acl_id d_id;
declare variable pask_name d_str_100;
declare variable subject d_str_255;
declare variable msg_body d_str_255;
declare variable reg_id d_integer;
declare variable db_id d_integer;
declare variable pdt_code d_str_code;
declare variable c_id d_id;
declare variable err_text d_str_255;
declare variable pdt_is_main_doc d_smallint;
declare variable edl_id d_id;
declare variable res_text d_str_255;
declare variable mpc_id d_id; /* ID записи mc_phone_change */
declare variable mpc_status d_integer;
declare variable is_need_digitization d_integer;
declare variable drl_project d_smallint;
declare variable lang_code d_str_lang;
begin
    bls_id = 0;
    is_need_digitization = 1; -- Нужно ли отправлять документ на оцифровку
    un_id = 0; -- Вставка из МПКлиентов, пользователь прилетает в mcl_id
    db_id = f_get_db_id();
    reg_id = f_hq_id(db_id);
    lang_code = f_get_lang_code(null, mcl_id);

    select
        pk_id,
        pk_name,
        pdt_code,
        drl_from_c_id,
        drl_project,
        pdt_is_main_doc
    from docrecoglist, passkind, passdoctype
    where drl_id = :drl_id
    and drl_from_pk_id = pk_id
    and pk_from_pdt_id = pdt_id
    into
        :pk_id,
        :bls_caption,
        :pdt_code,
        :c_id,
        :drl_project,
        :pdt_is_main_doc;

    pask_name = bls_caption;
    bls_caption = 'Стр.' || :page_no || coalesce(': ' || bls_caption, '');

    select drm_has_page2
    from hp_get_drm_has_pages(:pk_id)
    into :need_photo2;

    execute procedure hp_insert_into_blobstore(:bls_filename, :bls_caption, :bls_blob, :un_id, :src_kind, 2, :mcl_id)
    returning_values :bls_id;
    
    -- к документу прикрепляем блобы
    update docrecoglist drl
    set drl.drl_from_bls_1 = iif(:page_no = 1, :bls_id, drl.drl_from_bls_1),
        drl.drl_from_bls_2 = iif(:page_no <> 1, :bls_id, drl.drl_from_bls_2),
        drl_from_mcl_id = :mcl_id,
        drl_from_un_id = :un_id
    where drl.drl_id = :drl_id;

    -- Если документ еще не собран полностью, выходим
    if (not exists(select *
                   from docrecoglist
                   where drl_id = :drl_id and
                         drl_from_bls_1 is not null and
                         (drl_from_bls_2 is not null or :need_photo2 = 0))) then
        exit;

    ---------------------------------------------------------------------------
    -- Специальный кейс: фото клиента
    -- Создание документа мимо оцифровки RFCRU-405
    -- Пока один документ, если будет несколько нужно добавить поле в docrecogmain
    ---------------------------------------------------------------------------
    if (pdt_code = 'selfie_photo' and c_id is not null) then
    begin
        is_need_digitization = 0; -- В специальных кейсах на оцифровку не отправляем

        update docrecoglist
        set drl_status = 1
        where drl_id = :drl_id;
        
        select err_text
        from fm_docrecog_add_clientdoc(:drl_id, :un_id)
        into :err_text;
        
        if (err_text <> '') then
            exception e_default err_text;
    end

    ---------------------------------------------------------------------------
    -- Специальный кейс: паспорт для смены номера телефона
    -- Документ не сохраняется в карточку клиента и не отправляется на оцифровку
    ---------------------------------------------------------------------------
    if (db_id = 81001 and pdt_is_main_doc = 1 and f_enum_code('docrecog_project', :drl_project, 0) = 'phone_change_no_access') then
    begin
        is_need_digitization = 0;

        update docrecoglist
        set drl_status = 1
        where drl_id = :drl_id;
        
        -- Определяем ID записи для изменения. Берем последнюю запись по клиенту
        select first 1 mpc.mpc_id
        from mc_phone_change mpc
        where mpc.mpc_from_mcl_id = :mcl_id and
              mpc.mpc_status = f_enum_id('mc_phone_change', 'not_yet_created')
        order by mpc.mpc_id desc
        into :mpc_id;
        
        update mc_phone_change mpc
        set mpc.mpc_pass_drl_id = :drl_id
        where mpc.mpc_id = :mpc_id;
    end

    ---------------------------------------------------------------------------
    -- Специальный кейс: фото клиента для смены номера телефона
    -- Документ не сохраняется в карточку клиента и не отправляется на оцифровку
    ---------------------------------------------------------------------------
    if (pdt_code = 'selfie_photo_phone_change') then
    begin
        is_need_digitization = 0;

        update docrecoglist
        set drl_status = 1
        where drl_id = :drl_id;
        
        -- Ищем последнюю запись по клиенту, где уже сохранен паспорт, но еще не сохранено фото
        select first 1 mpc.mpc_id
        from mc_phone_change mpc
        where mpc.mpc_from_mcl_id = :mcl_id and
              mpc.mpc_status = f_enum_id('mc_phone_change', 'not_yet_created') and
              mpc.mpc_pass_drl_id is not null and
              mpc.mpc_client_photo_drl_id is null
        order by mpc.mpc_id desc
        into :mpc_id;
        
        select edl_id, res_text
        from hp_edoc_insert('cell.application.no_access', 'clients', :c_id, :un_id)
        into :edl_id, :res_text;
        
        mpc_status = iif(
            res_text = '',
            f_enum_id('mc_phone_change', 'created'),
            f_enum_id('mc_phone_change', 'doc_creation_error')
        );
        
        msg_body = iif(
            res_text = '',
            'Уважаемый клиент, Ваше заявление на смену номера телефона отправлено в обработку.',
            'Уважаемый клиент, при обработке заявления на смену номера телефона произошла ошибка, попробуйте позже или обратитесь к тех. поддержке'
        );
        
        update mc_phone_change
        set
            mpc_client_photo_drl_id = :drl_id,
            mpc_from_edl_id = :edl_id,
            mpc_status = :mpc_status
        where mpc_id = :mpc_id;
        
        execute procedure mc_send_push('mc_login', :mcl_id, 'Смена номера', :msg_body, -1);
    end


    -- Если оцифровка нужна, отправляем документ (стандартная логика выполнения)
    if (is_need_digitization = 1) then
    begin
        if (:db_id in (39801, 41701) and
            c_id is not null and
            f_grs_active(:un_id) = 1 and
            :pdt_is_main_doc = 1) then
        begin
            execute procedure hp_extinfo_insert('Tunduk', f_objkind('clients', 'id'), :c_id, 1);
            exit;
        end  

        if (:db_id in (39801, 41701) and
            f_get_const('mc_facerecog') = 1 and
            c_id is null) then
        begin
            execute procedure mc_send_to_grs(:mcl_id);
            exit;
        end  
            
        if (not exists(select *
                       from hp_docrecogrequest_link(:drl_id)
                       where drl_is_linked = 1)) then
        begin  
            -- Добавляем в очередь на оцифровку
            select res_acl_id
            from hp_actionlist_insert(null, :reg_id, 'DocRecogList', :drl_id, 'ActionGroup.DocRecog_Client', 0, :un_id)
            into :acl_id;
            
            update docrecoglist
            set drl_from_acl_id = :acl_id
            where drl_id = :drl_id;
        end
        
        -- Отправляем push-уведомление пользователю КМП о том, что на оцифровку отправлен документ
        pask_name = coalesce(pask_name, f_lang(:lang_code, 'Не определен'));
        subject = f_lang(lang_code, 'Отправлен документ на оцифровку');
        msg_body = pask_name;
        
        execute procedure mc_send_push('mc_login', :mcl_id, :subject, :msg_body);
    end
end
