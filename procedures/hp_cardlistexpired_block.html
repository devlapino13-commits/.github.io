CREATE PROCEDURE HP_CARDLISTEXPIRED_BLOCK(
    ON_DATE D_DATE = null)
AS
declare variable card_id d_integer;
declare variable cem_date d_date;
declare variable last_date d_date;
begin
  on_date = coalesce(on_date, current_date);
  last_date = f_date_month_end(on_date);

  if (on_date = last_date) then
  begin
    for
      select
        cal_card_id,
        cem_date
      from cardlist
        inner join cardembomain
          on cem_id = cal_from_cem_id
            and f_date_month_end(dateadd(month, 38, cem_date)) <= :on_date
      where cal_status = f_enum_id('cardlist_status', 'ready_to_use')
        and not exists(select * from cardlistexpired where cle_card_id = cal_card_id)
    into
      :card_id,
      :cem_date
    do
    begin
      insert into cardlistexpired(cle_card_id, cle_cem_date, cle_date_expired)
      values (:card_id, :cem_date, :on_date);

      update cardlist set
        cal_status = f_enum_id('cardlist_status', 'hard_stoplist_in')
      where cal_card_id = :card_id;
    end
  end
end
