CREATE PROCEDURE HP_OPERCASH_PVN_SYNC(
    IN_OFFICE_ID D_INTEGER,
    IN_DATE D_DATE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable res_count d_integer;
declare variable rec_count d_integer;
declare variable oper_kind d_integer;
declare variable po_id d_integer;
declare variable po_from_ol_id d_id;
begin
  res_is_error = 0;
  res_text = '';

  rec_count = 0;
  res_count = 0;
  oper_kind = f_enum_id('cashoperkind', 'minus.mcard');

  for select po_id, po_from_ol_id
  from pvnopers
  where po_office_id = :in_office_id and
        po_date = :in_date and
        po_from_k_id is null and
        po_oper_type = 0 and
        po_from_ol_id is not null and
        coalesce(po_ol_is_active, 1) = 1 and
        coalesce(po_is_delete, 0) = 0
  into :po_id, :po_from_ol_id
  do
  begin
    if (exists(select * from opercash
               where opc_date = :in_date and
                     opc_from_ol_id = :po_from_ol_id and
                     opc_oper_kind = :oper_kind)) then
      Continue;

    rec_count = rec_count + 1;
    execute procedure hp_opercash_pvnoper_save(:po_id);

    if (exists(select * from opercash
               where opc_date = :in_date and
                     opc_from_ol_id = :po_from_ol_id and
                     opc_oper_kind = :oper_kind)) then
      res_count = res_count + 1;
  end

  if (rec_count > 0) then
  begin
    res_text = 'Синхронизированны ПВН операции: ' || res_count || ' из ' || rec_count;
    if (res_count = 0) then
      res_is_error = 1;
  end else
  begin
    res_text = 'Не найдены новые ПВН операции';
    res_is_error = 1;
  end

  suspend;
end
