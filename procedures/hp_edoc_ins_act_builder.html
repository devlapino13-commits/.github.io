CREATE PROCEDURE HP_EDOC_INS_ACT_BUILDER(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable pr_code d_str_100;
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable k_status d_smallint;
declare variable ks_state d_smallint;
declare variable app_date d_date;
declare variable k_disb_date date;
declare variable kma_id d_id;
declare variable kml_purchase_datetime d_datetime;
begin
  --p_act = purchase act  акт прием передачи
  -- Процедура создает электронный документ по займу.
  -- Готовит бланки и отправляет СМС.
  -- Создан на основе HP_EDOC_INS_LOAN_AGREEMENT
  res_text = '';

  select k_from_ma_app_id, pr_code, k_status, k_disb_date
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into ma_id, pr_code, k_status, k_disb_date;

  if (k_status <> 1) then
    res_text = 'Сначала утвердите договор займа';

  if (res_text = '') then
  begin
    select ks_state
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999'
    into ks_state;

    if (ks_state = 1) then
      res_text = 'Займ уже действующий';

    if (ks_state = 2) then
      res_text = 'Займ уже закрыт';

  end

  if (res_text = '') then
  begin
    for select ag_object_id, kma.kma_id
        from kontmatag kma
        inner join agreements ag on kma.kma_from_ag_id = ag.ag_id
        where kma.kma_from_k_id = :k_id
        into :c_id, :kma_id
    do
    begin
      if (coalesce(:c_id, 0) <> 0) then
      begin
        select kml_purchase_datetime
        from print_kontmat(:k_id, :c_id)
        order by kml_purchase_datetime desc
        rows 1
        into kml_purchase_datetime;
        if (coalesce(kml_purchase_datetime, '31.12.9999') <> '31.12.9999') then
        begin
          if (f_edoc_status('act_from_builder.agreement', :kma_id, current_date) <> 'signed') then
            select edl_id, res_text
            from hp_edoc_insert('act_from_builder.agreement', f_objkind('kontmatag', 'id'), :kma_id, :un_id)
            into edl_id, res_text;
        end
        else
          res_text = 'Fatal error! don`t worry, be happy';
      end
    end
  end
  suspend;
end
