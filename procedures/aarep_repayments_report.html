CREATE PROCEDURE AAREP_REPAYMENTS_REPORT(
    REP_REG_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    K_NO D_STR_20,
    IS_IPF D_STR_10,
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    ACC1 D_STR_20,
    SUMM1 D_CURRENCY,
    OL_DATE1 D_DATE,
    ACC2 D_STR_20,
    SUMM2 D_CURRENCY,
    OL_DATE2 D_DATE,
    TOTAL_SUMM D_CURRENCY)
AS
declare variable k_id d_id;
declare variable reg_id d_integer;
declare variable k_id_tmp d_id;
declare variable summ_tmp d_currency;
declare variable ob_date_tmp d_date;
declare variable acc_dt d_str_10;
declare variable acc_kt d_str_10;
begin
  for select id_list
      from region_list(:rep_reg_id, 1, 3, 1)
      into :reg_id
  do
  begin
    k_id = 0;

    select reg.reg_code || ' ' || reg.reg_name
    from region reg
    where reg.reg_unicode = :reg_id
    into :reg_name;

    for with tmp
        as (select ob.ob_from_ol_id as ol_id, ob.ob_short_acc_dt as acc_dt, ob.ob_short_acc_kt as acc_kt,
                   ob.ob_summa as summa, ob.ob_ol_date as ol_date
            from operbook ob
            where (ob.ob_short_acc_dt in ('21199/1', '21199/101', '11531/7', '11531/107', '21199/1101') or
                  ob.ob_short_acc_kt in ('21199/1', '21199/101', '11531/7', '11531/107', '21199/1101')) and
                  ob.ob_ol_is_active = 1 and
                  ob.ob_ol_date between :rep_date_begin and :rep_date_end and
                  ob.ob_ol_reg_id = :reg_id)
        select distinct(ok.ok_from_k_id), acc_dt, acc_kt, summa, ol_date
        from operkred ok
        inner join tmp on ok.ok_from_ol_id = ol_id
        into :k_id_tmp, :acc_dt, :acc_kt, :summ_tmp, :ob_date_tmp
    do
    begin
      summ1 = 0;
      summ2 = 0;
      ol_date1 = null;
      ol_date2 = null;
      acc1 = '';
      acc2 = '';
      if (k_id <> k_id_tmp) then
      begin
        k_id = k_id_tmp;
        select k.k_kontract_no, iif(pr.pr_code containing '.ipf.', 'ИПФ', 'Классика'), c.c_name_fio,
               c.c_inn
        from kontract k
        inner join knclient kc on k.k_id = kc.kc_from_k_id and
              kc.kc_status = 2
        inner join clients c on kc.kc_from_c_id = c.c_id
        inner join product pr on k.k_from_pr_id = pr.pr_id
        where k.k_id = :k_id
        into :k_no, :is_ipf, :c_name_fio, :c_inn;
      end
      if (acc_kt in ('21199/1', '21199/101', '21199/1101')) then
        summ1 = summ1 + :summ_tmp;
      if (acc_dt in ('21199/1', '21199/101', '21199/1101')) then
        summ1 = summ1 - :summ_tmp;
      if (acc_kt in ('11531/7', '11531/107')) then
        summ2 = summ2 + :summ_tmp;
      if (acc_dt in ('11531/7', '11531/107')) then
        summ2 = summ2 - :summ_tmp;
      if (summ1 <> 0) then
      begin
        acc1 = iif(acc_dt in ('21199/1', '21199/101', '21199/1101'), acc_dt, acc_kt);
        ol_date1 = :ob_date_tmp;
      end
      if (summ2 <> 0) then
      begin
        acc2 = iif(acc_dt in ('11531/7', '11531/107'), acc_dt, acc_kt);
        ol_date2 = :ob_date_tmp;
      end
      total_summ = summ1 + summ2;
      suspend;
    end
  end
end
