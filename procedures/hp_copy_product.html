CREATE PROCEDURE HP_COPY_PRODUCT(
    PRODUCT_ORIG_ID INTEGER,
    PRODUCT_DATE DATE,
    UN_ID DECIMAL(18,0))
RETURNS (
    PRODUCT_NEW_ID INTEGER)
AS
declare variable prl_date_begin_orig date;
declare variable prl_date_end_orig date;
begin
  select max(pr_id) + 1
  from product
  into :product_new_id;
  if ((product_new_id is null) or (product_new_id <= 0)) then
    product_new_id = 1;

  insert into product (pr_id, pr_date_begin, pr_active, pr_category, pr_loan_type, pr_interest_method, pr_fine_method,
                       pr_fine_percent, pr_fine_max_day, pr_repay_type, pr_week_or_month, pr_percent_for_days,
                       pr_repay_go_back, pr_round_fine, pr_round_admin, pr_round, pr_is_secured, pr_komiss_type,
                       pr_pay_pros, pr_pay_curr, pr_pay_pred, pr_is_fine_weekend, is_delete, pr_code, pr_wm_kolich,
                       pr_name, pr_round_perc, pr_kommis_back_name, pr_is_kommis_back, pr_path_form, pr_kontract_format,
                       pr_adv_pay_percent, pr_adv_close_percent, pr_can_deposit, pr_need_kommis_adv_percent,
                       pr_from_pac_id, pr_kommis_is_percent, pr_need_reschedule, pr_fine_for_percent, pr_date_end,
                       pr_currency, pr_index_curr_code, pr_set_rate_in_kont, pr_calc_perc_tax, pr_is_employer,
                       pr_perc_input_kind, pr_is_360, pr_is_local_currency, pr_product_kind, pr_is_weekend_work,
                       pr_category_perc, pr_calc_perc_for_deliq, pr_min_fine, pr_send_to_kib, pr_min_fine_base,
                       pr_fine_like_perc, pr_common_name, pr_two_repay_date, pr_pay_pros_more, pr_can_disb_paral,
                       pr_from_un_id, pr_datetime, pr_fine_linked_to_perc, pr_need_sms_approve,
                       pr_from_pat_id, pr_memo, pr_ipf_commission, pr_percent_kind, pr_mbulak_cards_kind,
                       pr_add_month_period, pr_add_month_lgota)
  select :product_new_id, :product_date, 1, pr_category, pr_loan_type, pr_interest_method, pr_fine_method,
         pr_fine_percent, pr_fine_max_day, pr_repay_type, pr_week_or_month, pr_percent_for_days, pr_repay_go_back,
         pr_round_fine, pr_round_admin, pr_round, pr_is_secured, pr_komiss_type, pr_pay_pros, pr_pay_curr, pr_pay_pred,
         pr_is_fine_weekend, is_delete, pr_code, pr_wm_kolich, pr_name, pr_round_perc, pr_kommis_back_name,
         pr_is_kommis_back, pr_path_form, pr_kontract_format, pr_adv_pay_percent, pr_adv_close_percent, pr_can_deposit,
         pr_need_kommis_adv_percent, pr_from_pac_id, pr_kommis_is_percent, pr_need_reschedule, pr_fine_for_percent,
         pr_date_end, pr_currency, pr_index_curr_code, pr_set_rate_in_kont, pr_calc_perc_tax, pr_is_employer,
         pr_perc_input_kind, pr_is_360, pr_is_local_currency, pr_product_kind, pr_is_weekend_work, pr_category_perc,
         pr_calc_perc_for_deliq, pr_min_fine, pr_send_to_kib, pr_min_fine_base, pr_fine_like_perc, pr_common_name,
         pr_two_repay_date, pr_pay_pros_more, pr_can_disb_paral, :un_id, pr_datetime, pr_fine_linked_to_perc,
         pr_need_sms_approve, pr_from_pat_id, pr_memo, pr_ipf_commission, pr_percent_kind, pr_mbulak_cards_kind,
         pr_add_month_period, pr_add_month_lgota
  from product
  where pr_id = :product_orig_id;

  update product
  set pr_name = pr_name || ' копия'
  where pr_id = :product_new_id;

  prl_date_begin_orig = null;
  prl_date_end_orig = null;

  select distinct prl_date_begin, prl_date_end
  from prodlogic
  where prl_from_pr_id = :product_orig_id
  order by prl_date_begin desc, prl_date_end desc
  rows 1
  into :prl_date_begin_orig, prl_date_end_orig;

  insert into prodtypelink (prtl_from_pr_id, prtl_from_prt_id, prtl_date_begin, prtl_date_end, prtl_from_un_id)
  select :product_new_id, prtl_from_prt_id, prtl_date_begin, prtl_date_end, :un_id
  from prodtypelink
  where prtl_from_pr_id = :product_orig_id and
        coalesce(prtl_date_begin, '01.01.1990') <= :product_date and
        coalesce(prtl_date_end, '31.12.2999') <= :product_date;

  insert into productblank (prb_from_pr_id, prb_from_bln_id, prb_date_begin, prb_date_end, prb_order_no, prb_from_un_id)
  select :product_new_id, prb_from_bln_id, prb_date_begin, prb_date_end, prb_order_no, :un_id
  from productblank
  where prb_from_pr_id = :product_orig_id
  order by prb_id;

  execute procedure hp_copy_product_logic(:product_orig_id, :product_new_id, :prl_date_begin_orig, :prl_date_end_orig,
      :product_date, '31.12.9999');
  suspend;
end
