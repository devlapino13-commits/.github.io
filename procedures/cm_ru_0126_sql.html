CREATE PROCEDURE CM_RU_0126_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable PR_CODE varchar(100);
declare variable K_KOLICH_LGOTA integer;
begin
  for select pr_code, k_kolich_lgota
      from kontract, product
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            k_kolich_lgota > 0
      into pr_code, k_kolich_lgota
  do
  begin
    if (pr_code containing '.express.'
        -- or (pr_code containing '.par.' and pr_code not containing '.patent plus.')
        or pr_code containing '.status.' or pr_code containing '.shans plus.' or pr_code containing '.education.' or pr_code containing '.food.') then
    begin
      res_error_text = 'В текущем продукте не разрешается льготный период';
      res_error_is_critical = 2;
      suspend;
    end
    if (pr_code containing '.zalog_car.' /*or pr_code containing '.privilege_plus.'*/) then
    begin
      if (k_kolich_lgota > 3) then
      begin
        res_error_text = 'Кол-во льготных периодов не должно превышать 3.';
        res_error_is_critical = 2;
        suspend;
      end
      /*
      if (pr_code containing '.privilege_plus.' and
          exists(select *
                 from schedule
                 where sh_from_k_id = :k_id and
                       sh_id >= 4 and
                       sh_grace = 1)) then
      begin
        res_error_text = 'По продукту  "Привилегия + " льготные периоды должны быть в первых 3х платежах';
        res_error_is_critical = 2;
        suspend;
      end      */
    end
    else
    if ((pr_code containing '.zalog.' or pr_code containing '.privilege.') and
        k_kolich_lgota > 4) then
    begin
      res_error_text = 'Кол-во льготных периодов не должно превышать 4.';
      res_error_is_critical = 2;
      suspend;
    end
    else
    if ((pr_code containing '.reliable_client.' or pr_code containing '.universal.' or pr_code containing '.oivo.' or pr_code containing '.selfemployed.' or
         pr_code containing '.business.') and
        k_kolich_lgota > 4) then
    begin
      res_error_text = 'Кол-во льготных периодов не должно превышать 2.';
      res_error_is_critical = 2;
      suspend;
    end

    else
    if (k_kolich_lgota > 2 and
        pr_code not containing '.privilege.' and
        pr_code not containing '.privilege_plus.' and
        pr_code not containing '.reliable_client.' and
        pr_code not containing '.universal.' and
        pr_code not containing '.oivo.' and  
        pr_code not containing '.par.' and
        pr_code not containing '.selfemployed.'and
        pr_code not containing '.business.'and
        pr_code not containing '.refinance.' and
        pr_code not containing '.refinance_oivo.' and
        pr_code not containing '.refinance_plus.' and
        pr_code not containing '.b_refinance.' and
        pr_code not containing '.plus_100.' and
        pr_code not containing '.plus_100_b.'

        ) then
    begin
      res_error_text = 'Кол-во льготных периодов не должно превышать 2.';
      res_error_is_critical = 2;
      suspend;
    end
    if (pr_code containing '.shans.' or pr_code containing '.patent plus.') then
    begin
      if (k_kolich_lgota > 1) then
      begin
        res_error_text = 'В текущем продукте разрешено 1 льготный период.';
        res_error_is_critical = 2;
        suspend;
      end
      if (k_kolich_lgota = 1 and
          exists(select *
                 from schedule sh
                 where sh.sh_from_k_id = :k_id and
                       sh.sh_id > 0 and
                       sh.sh_os = 0 and
                       sh.sh_id not in (1))) then
      begin
        res_error_text = 'В текущем продукте льготный период разрешается только на первом платеже.';
        res_error_is_critical = 2;
        suspend;
      end
    end
    if (pr_code containing '.switch_plus.' or pr_code containing '.blago.' or pr_code containing '.uspeh.' or pr_code containing '.privilege_plus.' or pr_code containing '.refinance.' or pr_code containing '.refinance_oivo.') then
    begin
      if (exists(select *
                 from schedule sh
                 where sh.sh_from_k_id = :k_id and
                       sh.sh_id > 0 and
                       sh.sh_os = 0 and
                       extract(month from sh_date) not in (1, 2))) then
      begin
        res_error_text = 'Льготный период можно получить только на январь и/или февраль.';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end

  /*
  if (exists(select pr_code, k_kolich_lgota
             from kontract, product
             where k_id = :k_id and
                   k_from_pr_id = pr_id and
                   k_kolich_lgota < 3 and
                   pr_code containing '.privilege_plus.')) then
  begin
    res_error_text = 'По текущему продукту можете выбрать до 3х льготных периодов.';
    res_error_is_critical = 2;
    suspend;
  end   */
end
