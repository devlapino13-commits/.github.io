CREATE PROCEDURE CB_KG_KOBD_2022_06(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_C_ID D_BLOB_TEXT = 0)
RETURNS (
    REG_NAME VARCHAR(100),
    REPAY_SUM_NOT_OS CURR_TYPE,
    REPAY_KOMMIS_IPF CURR_TYPE,
    DELINQ_OS_PERIOD CURR_TYPE,
    AAREP_CONSULT_SUMMA CURR_TYPE,
    SUM_VOZNOG_OFFICE CURR_TYPE,
    KOL_KOBD SMALLINT,
    FIO_KOBD VARCHAR(1024),
    BONUS_KOBD CURR_TYPE,
    MEMO VARCHAR(2048),
    USING_SCORES D_INTEGER,
    BONUS_KOBD_FINAL D_CURRENCY,
    C_ID D_ID,
    KOL_TRANS D_INTEGER,
    BONUS_TRANS CURR_TYPE)
AS
declare variable REP_DATE_BEGIN date;
declare variable REG_ID integer;
declare variable BONUS_MEMO varchar(255);
declare variable PS_DATE_BEGIN date;
declare variable IS_KOBD smallint;
declare variable DATE_DISMISS date;
declare variable RD_ID_IPF D_INTEGER;
declare variable RD_ID_CONS D_INTEGER;
declare variable TMP_SUM D_CURRENCY;
declare variable OK_OPERATION_TYPE D_INTEGER;
declare variable JL_ID_LIST varchar(1024);
declare variable C_ID_LIST varchar(1024);
declare variable BONUS_KOBD_TOTAL D_CURRENCY;
declare variable SQL D_BLOB_TEXT;
declare variable JL_ID_LIST_SEP D_STR_LONG;
declare variable OFFICER_LIST D_STR_LONG;
declare variable OFFICER_LIST_DOTS D_STR_LONG;
declare variable KOL_TRANS_1 D_INTEGER;
declare variable INSURANCE_SUM CURR_TYPE;
declare variable RD_ID_INSURANCE D_INTEGER;
declare variable C_ID_LIST_ALL D_STR_LONG;
begin
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  if (rep_c_id not containing '.') then
    rep_c_id = rep_c_id || '.';

  select list(jl.jl_id, ', '), '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jl.jl_kind = jk.jk_id and
        jk_system_code = 'JobSystem.Specialist_KOBD'
  where jl.jl_name <> 'Выездной кассир'
  into :jl_id_list, :jl_id_list_sep;

  select list(ps.ps_from_c_id, ', '), '.' || list(ps.ps_from_c_id, '.') || '.'
  from hr_pstatus ps
  inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
  inner join jobkind jk on jk.jk_id = jl.jl_kind and
        jk.jk_system_code = 'Jobsystem.Operational_manager'
  where ps.ps_status = 1 and
        ps.ps_is_on_staff = 1 and
        :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_is_working = 1
  into :officer_list, :officer_list_dots;

  officer_list = officer_list || ', 1090101820206';
  officer_list_dots = officer_list_dots || '1090101820206.';


  rd_id_ipf = f_get_rd_id('cashtransactionincom.ipfcommission');
  rd_id_cons = f_get_rd_id('cashtransactionincom.consultation');
  -- RFCKG-4045
  rd_id_insurance = f_get_rd_id('cashtransactionincom.insurance');

  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
      into reg_id
  do
  begin
    --  Суммы погашении процентов пеня и админ сбор
    repay_sum_not_os = null;
    repay_kommis_ipf = 0;
    aarep_consult_summa = 0;
    bonus_kobd_total = 0;
    sql = '';
    sql = f_format('select -sum(ok.ok_perc + ok.ok_fine + ok.ok_kommis)
                    from operkred ok
                    inner join kontract k on k.k_id = ok.ok_from_k_id
                    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id and
                          ''%1'' between kal.kal_date and kal.kal_date_end and
                          kal_new_kf_officer not in (1000000000102, 1090101933446)and
                          kal.is_delete = 0  %4
                    where ok.ok_ol_is_active = 1 and
                          ok.ok_type = 4 and
                          ok.ok_ol_reg_id = %3 and
                          ok.ok_ol_date between ''%2'' and ''%1''', :rep_date_end, :rep_date_begin, :reg_id, iif(:officer_list <> '', ' and kal.kal_new_kf_officer not in (' || :officer_list || ') ', ''));
    execute statement sql
        into repay_sum_not_os;
    repay_sum_not_os = coalesce(repay_sum_not_os, 0);

    repay_kommis_ipf     = 0;
    aarep_consult_summa  = 0;
    insurance_sum        = 0;

    select 
        coalesce(sum(iif(ok.ok_operation_type = :rd_id_ipf,  ok.ok_amount, 0)), 0),
        coalesce(sum(iif(ok.ok_operation_type = :rd_id_cons, ok.ok_amount, 0)), 0)
    from operkontract ok
    inner join operlist ol 
            on ol.ol_id = ok.ok_from_ol_id
    inner join kontract kon 
            on kon.k_id = ok.ok_from_k_id
    inner join knaktlist kal 
            on kal.kal_from_kf_id = kon.k_from_kf_id
           and kal.is_delete = 0
           and :rep_date_end between kal.kal_date and kal.kal_date_end
           and kal.kal_new_kf_officer not in (1000000000102, 1090101933446)
           and :officer_list_dots not containing '.' || kal.kal_new_kf_officer || '.'
    where ol.is_delete = 0
      and ol.ol_from_reg_id = :reg_id
      and ol.ol_date between :rep_date_begin and :rep_date_end
    into :repay_kommis_ipf, :aarep_consult_summa;

    select coalesce(sum(ik.ik_commission / 102 * 100), 0)
    from kontstate ks
    inner join knclient kc 
            on kc.kc_from_k_id = ks.ks_from_k_id
    inner join insurance_kontract ik 
            on ik.ik_from_k_id = ks.ks_from_k_id
           and (ik.ik_enum_status in (
                    f_enum_id('insurance_kontract', 'insurance_policy_obtained'),
                    f_enum_id('insurance_kontract', 'insurance_policy_app_was_sent'))
                or ik.ik_from_ag_id in (7414,7570))
    inner join kontract k 
            on k.k_id = ks.ks_from_k_id
    inner join product pr 
            on k.k_from_pr_id = pr.pr_id
    inner join knaktlist kal 
            on kal.kal_from_kf_id = k.k_from_kf_id
           and kal.is_delete = 0
           and :rep_date_end between kal.kal_date and kal.kal_date_end
           and kal.kal_new_kf_officer not in (1000000000102, 1090101933446)
           and :officer_list_dots not containing '.' || kal.kal_new_kf_officer || '.'
    inner join region_list(:reg_id, 1, 3, 1) t 
            on t.id_list = ks.ks_region_id
    where ks.ks_ol_is_delete = 0
      and ks.ks_state = 1
      and ks.ks_date_begin between :rep_date_begin and :rep_date_end
      --and pr.pr_code not containing '.ipf.'
      and (
            (k.k_fact_disb_date < '10.11.2025'
             and kc.kc_amount_approved >= 70000
             and pr.pr_code containing '.ind.'
             and (pr.pr_code containing '.STANDART.' or pr.pr_code containing '.IPF.')
             and pr.pr_code not containing '.consultation.')
         or (k.k_fact_disb_date >= '10.11.2025')
          )
    into :insurance_sum;

    --  Сумма просроченной задолженности по ОС за отчетный период
    delinq_os_period = null;
    select sum(ok_os + ok_sh_os)
    from oksummary, operkred
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :reg_id and
          :rep_date_end >= oks_delinq_date and
          oks_from_k_id = ok_from_k_id and
          oks_close_state = 0 and
          ok_ol_is_active = 1 and
          ok_type < 7 and
          ok_sh_date between :rep_date_begin and :rep_date_end and
          ok_ol_date <= :rep_date_end and
          oks_kc_id not in (1000000000102, 1090101933446) and
          :officer_list_dots not containing '.' || oks_kc_id || '.'
    into delinq_os_period;
    delinq_os_period = coalesce(delinq_os_period, 0);

    --  Сумма вознаграждения подотчетного офиса
    sum_voznog_office = (repay_sum_not_os + repay_kommis_ipf) * 10 / 100 + (aarep_consult_summa) * 10 / 100 + insurance_sum * 9 / 100 - delinq_os_period * 50 / 100;

    if (repay_sum_not_os <> 0 or repay_kommis_ipf <> 0 or delinq_os_period <> 0) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;

      kol_kobd = 0;
      fio_kobd = '';
      c_id_list = '.';
      c_id_list_all = '.';
      memo = '';
      sql = f_format('select ps_from_c_id, ps_date_begin
                    from hr_pstatus
                    where ps_status = 1 and
                          ps_is_working = 1 and
                          ''%1'' between ps_date_begin and ps_date_end and
                          ps_from_reg_id = %2 and
                          ps_from_jl_id in (%3)
                          %4', :rep_date_end, :reg_id, :jl_id_list, iif(rep_c_id <> '0.', ' and ps_from_c_id in (' ||(select list(value_out, ', ')
                                                                                                                      from str_with_separator_parser(:rep_c_id, '.')) || ')', ''));
      for execute statement sql
              into :c_id, :ps_date_begin
      do
      begin
        if (coalesce((select ho_order_type
                      from hr_voctravel, hr_orders
                      where hrv_from_ho_id = ho_id and
                            hrv_from_c_id = :c_id and
                            :rep_date_end between hrv_date_begin and hrv_date_end_fact and
                            hrv_status = 1
                      order by hrv_date_begin desc
                      rows 1), 0) not in (16, 17, 18, 24)) then --исключила условия "неоплачиваемый отпуск" по просьбе Назгуль RFCKG-372
        begin
          if (ps_date_begin <= dateadd(month, -3, :rep_date_end)) then
            is_kobd = 1;
          else
          begin
            --        вычисляем дату последнего увольнения
            date_dismiss = null;
            select max(ps_date_begin)
            from hr_pstatus
            where ps_from_c_id = :c_id and
                  ps_status = 1 and
                  ps_is_working = 0 and
                  ps_date_begin <= :rep_date_end
            into date_dismiss;
            if (date_dismiss is null) then
              date_dismiss = '01.01.1990';

            select min(ps_date_begin)
            from hr_pstatus
            where ps_status = 1 and
                  ps_is_working = 1 and
                  ps_from_c_id = :c_id and
                  ps_date_begin > :date_dismiss and
                  :jl_id_list_sep containing '.' || ps_from_jl_id || '.'
            into ps_date_begin;
            if (ps_date_begin <= dateadd(month, -3, :rep_date_end)) then
              is_kobd = 1;
            else
              is_kobd = 0;
          end
          -- RFCKG-2170
          if (:c_id = 1090101662818) then
             is_kobd = 1;

          kol_kobd = kol_kobd + 1;
          c_id_list_all = c_id_list_all || :c_id || '.';
            
          -- только >3 месяцев
          if (is_kobd = 1) then
          begin
             c_id_list = c_id_list || :c_id || '.';
          end
        end
      end
      if (kol_kobd is null) then
        kol_kobd = 0;
      if (kol_kobd > 0) then
        bonus_kobd_total = sum_voznog_office / (50 * kol_kobd);
      else
        bonus_kobd_total = 0;
      if (sum_voznog_office < 0) then
        bonus_kobd_total = 0;

      for select value_out
          from str_with_separator_parser(:c_id_list_all, '.')
          into :c_id
      do
      begin
        select c.c_name_fio
        from clients c
        where c.c_id = :c_id
        into :fio_kobd;
        bonus_trans = 0;
        kol_trans = 0;
        kol_trans_1 = 0;
        bonus_kobd_final = 0;
        using_scores = 0;

        select count(*) / 3
        from operbook ob
        inner join operlist ol on ol.ol_id = ob.ob_from_ol_id and
             ol.is_delete = 0 and
              cast(ol.ol_datetime as date) between :rep_date_begin and :rep_date_end
        inner join username un on un.un_id = ol.ol_user_id and
              un.un_from_c_id = :c_id
        where (ob.ob_short_acc_dt starting '10001' and
               ob.ob_short_acc_kt in ('10931/1', '10941/1', '10941/11', '10977/1', '11510/2', '11531/4', '11531/6', '21141/1', '21141/111', '21141/2',
                                      '21141/201', '21122/300', '61099/10', '61099/3'))
        into kol_trans;

        kol_trans_1 = 0;
    
        select count(*)
        from operbook ob
        inner join operlist ol on ol.ol_id = ob.ob_from_ol_id and
              ol.is_delete = 0 and
              ol.ol_date between :rep_date_begin and :rep_date_end
        inner join username un on un.un_id = ol.ol_user_id and
              un.un_from_c_id = :c_id
        where ob.ob_short_acc_dt in ('10941/1') and
              ob.ob_short_acc_kt starting '10001' and
              ob.ob_ol_is_active = 1
        into kol_trans_1;

        kol_trans = :kol_trans + :kol_trans_1;
        kol_trans_1 = 0;
        select count(*)
        from operbook ob
        inner join operlist ol on ol.ol_id = ob.ob_from_ol_id and
              ol.is_delete = 0 and
              ol.ol_date between :rep_date_begin and :rep_date_end
        inner join username un on un.un_id = ol.ol_user_id and
              un.un_from_c_id = :c_id
        where ob.ob_short_acc_dt starting '10001' and
              ob.ob_short_acc_kt in ('21141/102', '21114/10') and
              ob.ob_ol_is_active = 1
        into kol_trans_1;

        kol_trans = :kol_trans + :kol_trans_1;
        bonus_trans =  :kol_trans * 0.2;
        bonus_kobd = :bonus_kobd_total + :bonus_trans;

        if (:c_id_list containing '.' || :c_id || '.') then
        begin
           -- стаж более 3 месяцев
           select bonus_fact, using_scores
           from hp_calc_rebuke_bonus(:bonus_kobd, :c_id, current_date)
           into :bonus_kobd_final, :using_scores;
        
           memo = null;
        end
        else
        begin
           -- стаж менее 3 месяцев
           memo = 'Стаж менее 3 мес.';
        end

        suspend;
      end
    end
  end
end
