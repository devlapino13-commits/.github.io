CREATE PROCEDURE EQUIFAX_LOANS_FOR_DELETE(
    STEP D_INTEGER)
RETURNS (
    K_ID D_ID,
    K_ORIG_ID D_ID,
    K_KONTRACT_NO D_STR_100,
    KONTRACT_NO_ORIG D_STR_100,
    C_ID D_ID,
    KC_NUMBER D_INTEGER,
    IS_PORUCH D_ID,
    CRED_NO D_STR_100,
    CRED_TYPE D_INTEGER,
    PASS_NO D_STR_100,
    K_DISB_DATE D_DATE,
    K_KREDLINE_END D_DATE,
    RES_STR D_STR_255,
    KC_NUMBER_G D_INTEGER,
    REASON D_STR_255)
AS
declare variable C_ID_TMP D_ID;
declare variable CRED_NO_TMP D_STR_100;
begin
    -- Процедура отвечает за формирование файла на удаление займов в Эквифакс
    if (step = 0) then
    begin
        for select t.ekd_k_id, t.ekd_reason
        from equifax_kont_for_del t
        where t.ekd_id is not null
        into :k_id, :reason
        do
        begin
            suspend;
        end
    end

    if (step = 1) then
    begin
        for select distinct k_id, reason
        from equifax_loans_for_delete(0)
        into :k_id, :reason
        do
        begin
            for select kont.k_base_k_id,
                       kont.k_kontract_no,
                       kont.k_disb_date,
                       kont.k_kredline_end
            from kontract kont
            where kont.k_id = :k_id
            into :k_orig_id, :k_kontract_no, :k_disb_date, :k_kredline_end
            do
            begin
                kontract_no_orig = null;
        
                select kont.k_kontract_no
                from kontract kont
                where kont.k_id = :k_orig_id
                into :kontract_no_orig;
        
                for select kc_from_c_id,
                           kc_number
                  from knclient kncl
                  inner join clients cl on cl.c_id = kncl.kc_from_c_id
                  where kncl.kc_from_k_id = :k_id and
                        kncl.kc_amount_approved > 0
                into :c_id_tmp, :kc_number
                do
                begin
                      for select kc_from_c_id,
                                 iif(kncl.kc_from_c_id = :c_id_tmp, 0, 1),
                                 kc_number
                          from knclient kncl
                          where kncl.kc_from_k_id = :k_id and
                                kncl.kc_amount_approved > 0
                          into :c_id,
                               :is_poruch,
                               :kc_number_g
                      do
                      begin
                        suspend;
                      end
            
                      if (k_orig_id is not null) then
                      begin
                            for select kc_from_c_id,
                                       kc_number
                            from kontract kont
                            inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
                            where kncl.kc_amount_approved > 0
                            and kont.k_base_k_id = :k_orig_id
                            and kont.k_id <> :k_id
                            into :c_id,
                                 :kc_number_g
                            do
                            begin
                                is_poruch = 1;
                                suspend;
                            end
                      end
                end
           end
       end
   end

   if (step = 2) then
   begin
        for select K_ID, K_ORIG_ID, K_KONTRACT_NO, KONTRACT_NO_ORIG, C_ID, KC_NUMBER, IS_PORUCH, k_disb_date, k_kredline_end, kc_number_g, reason
        from equifax_loans_for_delete(1)
        order by k_id, kc_number, is_poruch -- именно такая последоательность важна !!!
        into k_id, k_orig_id, k_kontract_no, kontract_no_orig, c_id, kc_number, is_poruch, k_disb_date, k_kredline_end, kc_number_g, reason
        do
        begin
          if (is_poruch = 0) then -- рессчитываю только для заемщиков
          begin
            cred_type = '19';
            cred_no =  iif(kontract_no_orig is null, k_kontract_no, kontract_no_orig)|| '(' || cast(kc_number as varchar(10)) || ')';
            suspend;
          end
          else
          begin
            cred_type = '90';
            cred_no = iif(kontract_no_orig is null, k_kontract_no, kontract_no_orig)|| '(' || cast(kc_number as varchar(10)) || ')g'||:kc_number_g;
            suspend;
          end
        end
   end

   if (step = 3) then
   begin
        for select K_ID, K_ORIG_ID, K_KONTRACT_NO, KONTRACT_NO_ORIG, C_ID, KC_NUMBER, IS_PORUCH, CRED_NO, CRED_TYPE, k_disb_date, k_kredline_end, reason
        from equifax_loans_for_delete(2)
        order by K_ORIG_ID, C_ID
        into :K_ID, K_ORIG_ID, K_KONTRACT_NO, KONTRACT_NO_ORIG, C_ID, KC_NUMBER, IS_PORUCH, CRED_NO, CRED_TYPE, k_disb_date, k_kredline_end, reason
        do
        begin
            for select distinct cds_doc_no
            from
            (
                select distinct cds.cds_doc_no as cds_doc_no
                from clientdocs cds
                where cds.cds_from_c_id = :c_id
                and cds.cds_is_main_doc = 1
                union
                select cl.c_pass_no
                from clients cl
                where cl.c_id = :c_id
            ) A
            into :pass_no
            do
            begin
                -- cred_no;cred_currency;cred_date;cred_enddate;is_person;doc_or_egrn;reason;cred_type
                res_str = CRED_NO||';810;'||f_datetostr(k_disb_date)||';'||f_datetostr(k_kredline_end)||';1;'||pass_no||';'||reason||';'||CRED_TYPE;
                suspend;
            end
        end

        K_ID = null;
        K_ORIG_ID  = null;
        K_KONTRACT_NO  = null;
        KONTRACT_NO_ORIG  = null;
        C_ID  = null;
        KC_NUMBER  = null;
        IS_PORUCH  = null;
        CRED_NO  = null;
        CRED_TYPE  = null;
        PASS_NO  = null;
        K_DISB_DATE  = null;
        K_KREDLINE_END  = null;

        --'0302-040945-03(1);810;21.08.2014;23.02.2015;1;AA2502138;Некорректная выгрузка;19'
        for select t.ekd_res_str
        from equifax_kont_for_del t
        where coalesce(t.ekd_res_str, '') <> ''
        into :res_str
        do
        begin
            suspend;
        end
   end
end
