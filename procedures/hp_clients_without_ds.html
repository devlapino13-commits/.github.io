CREATE PROCEDURE HP_CLIENTS_WITHOUT_DS(
    MA_ID D_INTEGER)
RETURNS (
    CL_LIST D_STR_255)
AS
declare variable DB_ID D_ID;
declare variable C_ID D_ID;
declare variable MAC_AMOUNT D_CURRENCY;
declare variable C_NAME_FIO D_STR_100;
begin
  db_id = f_get_db_id();
  -- клиенты в заявке без ЭЦП;
  if (db_id in (39801, 41701)) then -- В БД КГ у сотрудников и клиентов разный договор ЭЦП
  begin
  cl_list = '';
    if (f_get_const('asp_const') = 1) then
    begin
      for select c_id, c_name_fio
          from miniappclient mac
          inner join clients on c_id = mac_from_c_id
          where mac.mac_from_ma_id = :ma_id and
              (mac.mac_amount > 0 or
              mac_is_poruch in (-1, 1))
          into c_id, c_name_fio
      do
      begin
        if (f_asp_signed(c_id, current_date) <> 1) then
          cl_list = cl_list || '    ' || c_name_fio ||  ';' || ascii_char(13);
      end
      --Убираем ";"
      if (cl_list <> '') then
        cl_list = left(cl_list, character_length(cl_list) - 2);
    end
    else
    begin
      select coalesce(list('    ' || c_name_fio, ';' || ascii_char(13)), '')
      from miniappclient mac
      inner join clients on c_id = mac_from_c_id
      where mac.mac_from_ma_id = :ma_id and
            (mac.mac_amount > 0 or
            mac_is_poruch in (-1, 1)) and
            iif(f_edoc_status('asp.agreement', mac_from_c_id, current_date) <> 'signed', f_edoc_status('asp.agreement_employee', mac_from_c_id, current_date) <> 'signed', false)
            --Проверяем есть ли неподписанный ЭЦП, если не подписан, то проверяем ЭЦП сотрудника
      into cl_list;

      if (exists(select *
                 from miniapp ma
                 inner join product pr on pr.pr_id = ma.ma_from_pr_id
                 where ma.ma_id = :ma_id and
                       ma.ma_is_active = 1 and
                       ma.ma_date_end >= '31.12.2999' and
                       pr.pr_code containing '.sugarmill.')) then
      begin
        cl_list = '';
        for select mac.mac_from_c_id, mac.mac_amount, cl.c_name_fio
            from miniappclient mac
            inner join clients cl on cl.c_id = mac.mac_from_c_id
            where mac.mac_from_ma_id = :ma_id
            into c_id, mac_amount, c_name_fio
        do
        begin
          if (mac_amount > 0) then
            cl_list = iif(f_edoc_status('asp.agreement', :c_id, current_date) <> 'signed', trim(:c_name_fio), trim(''));
          else
          begin
            if (f_edoc_status('asp.agreement', :c_id, current_date) <> 'signed') then
              continue;
          end
        end
      end
    end
  end
  else
  begin
    -- Для других баз отрабатывает по старому алгоритму
    select coalesce(list('    ' || c_name_fio, ';' || ascii_char(13)), '')
    from miniappclient
    inner join clients on c_id = mac_from_c_id
    where mac_from_ma_id = :ma_id and
          (mac_amount > 0 or
          mac_is_poruch in (-1, 1)) and
          (f_edoc_status('asp.agreement', mac_from_c_id, current_date) <> 'signed')
    into :cl_list;
  end
  if (cl_list <> '') then
    cl_list = 'Проверка по КИБ невозможна.' || ascii_char(13) || 'Договор ЭЦП не добавлен:' || ascii_char(13) || cl_list;
  suspend;
end
