CREATE PROCEDURE HP_TICKET_LINK(
    TABLE_NAME D_STR_CODE,
    TABLE_KEY_ID D_ID,
    TKT_ID D_ID,
    LINK_OBJECT_ID D_STR_CODE,
    TKT_KIND D_SMALLINT,
    TKT_SOURCE D_SMALLINT)
RETURNS (
    RES_TKT_ID D_ID)
AS
declare variable tkt_group d_id;
begin
  if (table_name not in ('extminiapp', 'cc_website_miniapp', 'miniapp', 'kontract', 'extsysmain')) then
    exception e_default 'Invalid table name: '||coalesce(table_name, '<NULL>');

  if (tkt_id is null) then
  begin
    if (link_object_id is not null) then
    begin
      if (table_name = 'extminiapp') then
      begin
        -- если от этого клиента была заявка за последний 7 дней,
        -- объединяем их в одну группу
        select tkt_group
        from extminiapp, ticket
        where ema_from_c_id = :link_object_id and
              ema_date > (current_date - 7) and
              ema_from_tkt_id = tkt_id and
              tkt_is_closed = 1 and
              tkt_status < 0
        order by ema_id desc
        rows 1
        into :tkt_group;
      end

      if (table_name = 'cc_website_miniapp') then
      begin
        select tkt_group
        from cc_website_miniapp, ticket
        where wma_phone_no = :link_object_id and
              wma_date > (current_date - 7) and
              wma_from_tkt_id = tkt_id and
              tkt_is_closed = 1 and
              tkt_status < 0
        order by wma_id desc
        rows 1
        into :tkt_group;
      end

      if (table_name = 'extsysmain') then
      begin
      end

      if (table_name = 'miniapp') then
      begin
      end

      if (table_name = 'kontract') then
      begin
      end
    end
  
    select tkt_id
    from hp_ticket_ins(:tkt_group, :tkt_kind, :tkt_source, f_objkind(:table_name, 'id'), :table_key_id)
    into res_tkt_id;
  end else
  begin
    res_tkt_id = tkt_id;
    execute procedure hp_ticket_upd(:tkt_id, 0, f_objkind(:table_name, 'id'), :table_key_id);
  end

  Suspend;
end
