CREATE PROCEDURE HP_ACTIONLIST_GET_PASSDATA(
    REP_C_ID DECIMAL(18,0))
RETURNS (
    RES_OBJECT_DETAIL BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_BLOB_IDS VARCHAR(255),
    RES_OBJECT_BLOB_NAMES BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable c_name_fio varchar(100);
declare variable c_pass_no varchar(100);
declare variable c_alternat_fio varchar(100);
declare variable c_birth_date date;
declare variable c_pass_who varchar(255);
declare variable c_pass_date date;
declare variable c_pass_date_expire date;
declare variable c_citizen integer;
declare variable c_citizen_str varchar(255);
declare variable c_birth_country varchar(255);
declare variable c_birth_city varchar(255);
declare variable c_pass_who_code varchar(100);
declare variable c_inn varchar(20);
declare variable c_ext_inn varchar(20);
declare variable null_str varchar(255);
declare variable fms_check_result integer;
declare variable db_id integer;
declare variable c_pincode varchar(100);
declare variable c_name_eng d_str_255;
begin
-- Базовая процедура для получения паспортных данных в виде HTML
  res_object_blob_names = '';
  res_blob_ids = '';
  res_object_detail = '';
  null_str = '<font color="red">Не указан</font>';

  res_object_detail = '';

  :db_id = f_get_db_id();

  select c_name_fio, coalesce(c_alternat_fio, ''), c_pass_no, c_pass_who, c_pass_date, c_pass_date_expire, c_birth_date,
         c_citizen, coalesce(c_birth_country, ''), coalesce(c_birth_city, ''), coalesce(c_pass_who_code, ''),
         coalesce(c_inn, ''), coalesce(c_ext_inn, ''), coalesce(c_pincode, ''),
         coalesce(c_name_eng, '')
  from clients
  where c_id = :rep_c_id
  into :c_name_fio, :c_alternat_fio, :c_pass_no, c_pass_who, c_pass_date, c_pass_date_expire, c_birth_date, c_citizen,
       c_birth_country, c_birth_city, c_pass_who_code, c_inn, c_ext_inn, c_pincode, c_name_eng;

  c_citizen_str = '';
  select rd_name
  from refdata
  where rd_id = :c_citizen
  into :c_citizen_str;

  if (db_id <> 94401) then
  begin
-- кроме АЗ базы

    res_object_detail = BLOB_APPEND(res_object_detail, 'ФИО (кириллица): <b>' || c_name_fio || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'ФИО (латиница):  <b>' || c_alternat_fio || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'ФИО (английский):  <b>' || c_name_eng || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'День рождения:   <b>' || iif(c_birth_date is not null, right('0' || extract(day from c_birth_date), 2) || '.' || right('0' || extract(month from c_birth_date), 2) || '.' || extract(year from c_birth_date), 'Не указан') || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Гражданство:     <b>' || c_citizen_str || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'ИНН:             <b>' || c_inn || '</b><br>');
    if (db_id = 81001) then
      res_object_detail = BLOB_APPEND(res_object_detail, 'ИНН на родине:   <b>' || c_ext_inn || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Номер паспорта:  <b>' || c_pass_no || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Срок действия:   <b>' ||

    iif(c_pass_date is not null, right('0' || extract(day from c_pass_date), 2) || '.' || right('0' || extract(month from c_pass_date), 2) || '.' || extract(year from c_pass_date), 'Не указан') || '</b> до <b>' || iif(c_pass_date_expire is not null, right('0' || extract(day from c_pass_date_expire), 2) || '.' || right('0' || extract(month from c_pass_date_expire), 2) || '.' || extract(year from c_pass_date_expire), 'Не указан') || '</b><br>');

    res_object_detail = BLOB_APPEND(res_object_detail, 'Код орг.выдачи:  <b>' || c_pass_who_code || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Место выдачи:    <b>' || c_pass_who || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Страна рождения: <b>' || c_birth_country || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Место рождения:  <b>' || c_birth_city || '</b><br>');

-- Проверка в ФМС для граждан РФ
    if (exists(select rd_name
               from refdata
               where rd_id = :c_citizen and
                     rd_code = 'Citizen.RU')) then
    begin
      if (c_pass_no is not null) then
      begin
        select check_result
        from hp_check_fms_passport(:c_pass_no)
        into fms_check_result;
        if (fms_check_result = 1) then
          res_object_detail = BLOB_APPEND(res_object_detail, 'Проверка ФМС:    <font color="red"><b>НАЙДЕН</b></font><br>');
        else
          res_object_detail = BLOB_APPEND(res_object_detail, 'Проверка ФМС:    <b>не найден</b><br>');
      end else
        res_object_detail = BLOB_APPEND(res_object_detail, 'Проверка ФМС:    <b>паспорт не указан</b><br>');
    end
  end
  else
  begin
-- для АЗ базы

    res_object_detail = BLOB_APPEND(res_object_detail, 'ФИО: <b>' || c_name_fio || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'ФИО (альтернатива):  <b>' || c_alternat_fio || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'День рождения:   <b>' || iif(c_birth_date is not null, right('0' || extract(day from c_birth_date), 2) || '.' || right('0' || extract(month from c_birth_date), 2) || '.' || extract(year from c_birth_date), 'Не указан') || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Гражданство:     <b>' || c_citizen_str || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'ИНН:             <b>' || c_inn || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'PIN:             <b>' || c_pincode || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Номер паспорта:  <b>' || c_pass_no || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Срок действия:   <b>' ||

    iif(c_pass_date is not null, right('0' || extract(day from c_pass_date), 2) || '.' || right('0' || extract(month from c_pass_date), 2) || '.' || extract(year from c_pass_date), 'Не указан') || '</b> до <b>' || iif(c_pass_date_expire is not null, right('0' || extract(day from c_pass_date_expire), 2) || '.' || right('0' || extract(month from c_pass_date_expire), 2) || '.' || extract(year from c_pass_date_expire), 'Не указан') || '</b><br>');

    res_object_detail = BLOB_APPEND(res_object_detail, 'Код орг.выдачи:  <b>' || c_pass_who_code || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Место выдачи:    <b>' || c_pass_who || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Страна рождения: <b>' || c_birth_country || '</b><br>');
    res_object_detail = BLOB_APPEND(res_object_detail, 'Место рождения:  <b>' || c_birth_city || '</b><br>');
  end

  suspend;

end
