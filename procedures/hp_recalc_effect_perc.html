CREATE PROCEDURE HP_RECALC_EFFECT_PERC(
    K_ID DECIMAL(18,0))
RETURNS (
    DISB_DATE DATE,
    DISB_AMOUNT DECIMAL(18,4),
    KOMMIS_AMOUNT DECIMAL(18,4),
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4))
AS
declare variable rep_dd date;
begin
  rep_dd = '18.09.2013';
  select ks_date_begin, k_amount_approved
  from kontstate, kontract
  where ks_from_k_id = k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_from_k_id = :k_id
  into :disb_date, disb_amount;
  if (disb_amount is null) then
    disb_amount = 0;

  select sum(-ok_kommis)
  from operkred
  where ok_type = 4 and
        ok_ol_is_active = 1 and
        ok_from_k_id = :k_id and
        ok_from_c_id = 0
  into :kommis_amount;
  if (kommis_amount is null) then
    kommis_amount = 0;

  sh_id = 0;
  for with tab
      as (select ok_ol_date as sh_date, sum(-ok_os) as sh_os, sum(-ok_perc) as sh_perc
          from operkred
          where ok_from_k_id = :k_id and
                ok_type in (4, 5) and
                ok_ol_is_active = 1 and
                ok_ol_date <= :rep_dd and
                ok_from_c_id = 0
          group by ok_ol_date
          having sum(ok_os) <> 0 or sum(ok_perc) <> 0
          union all
          select ok_sh_date as sh_date, sum(ok_os + ok_sh_os) as sh_os, sum(ok_perc + ok_sh_perc) as sh_perc
          from operkred
          where ok_from_k_id = :k_id and
                ok_type < 7 and
                ok_ol_is_active = 1 and
                ok_ol_date <= :rep_dd and
                ok_from_c_id = 0
          group by ok_sh_date
          having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0)
      select sh_date, sum(sh_os), sum(sh_perc)
      from tab
      group by sh_date
      order by sh_date
      into :sh_date, :sh_os, :sh_perc
  do
  begin
    sh_id = sh_id + 1;
    suspend;
  end

end
