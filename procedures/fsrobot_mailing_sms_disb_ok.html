CREATE PROCEDURE FSROBOT_MAILING_SMS_DISB_OK(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable C_NAME_FIO varchar(100);
declare variable OL_DATE date;
declare variable C_CELL_NO varchar(100);
declare variable C_ID decimal(18,0);
declare variable KOL integer;
declare variable DATE_FORMAT_STR varchar(20);
declare variable K_ID decimal(18,0);
declare variable AMOUNT decimal(18,4) = 0;
declare variable UNIQ_ACC_STR varchar(10);
declare variable DOC_NO varchar(20);
declare variable AMOUNT_STR varchar(20);
declare variable DB_ID integer;
declare variable AMOUNT_SYM varchar(10);
declare variable CELL_ALL varchar(255);
declare variable K_REFUND smallint;
declare variable KC_CIRCLE integer;
declare variable POINT integer;
declare variable MCL_PROJECT integer;
begin
  -- Процедура вызывается из FSROBOT_ON_MAILING
  -- Процедура готовит список уведомлений для отправки роботом по почте или sms
  -- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
  -- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
  -- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

  -- RefData.RD_Memo    - Список получателей
  -- RefData.RD_Tag     - Срок жизни сообщения
  -- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

  -- Rep_Params - входящий параметры в зависимости от команды
  -- Rep_UN_ID - пользователь

  -- СМС после выдачи кредита
  db_id = f_get_db_id();

  --RFCKG-2987
  if (db_id in (39801, 41701)) then
    exit;

  if (user_fio is null) then
    user_fio = '';
  res_maillist = :rep_email;

  amount_sym = (case
                  when (db_id = 81001) then 'r.'
                  when (db_id in (39801, 41701)) then 'c.'
                  when (db_id = 94401) then 'AZN'
                end);

  res_body = '';
  kol = 0;
  for select distinct ok_from_k_id, ok_ol_date, iif(coalesce(ol_doc_no, '0') = '0', 'Б/Н', ol_doc_no)
      from operkred, operlist
      where ok_from_ol_id = :rep_params and
            ol_id = :rep_params and
            ol_oper_type = 1 -- Выдача
            and
            ol_oper_subtype = 1 -- Выдача
            and
            ok_ol_is_active = 1 -- Действующая проводка
            and
            ok_type = 8 -- Выдача
      into :k_id, :ol_date, doc_no
  do
  begin
    date_format_str = f_datetostr(ol_date);

    cell_all = '';
    for select c_id, kc_amount_approved,
               (select cont_value
                from hp_contdata_get_cell_main(0, c_id)), c_name_fio, kont.k_refund
        from knclient kncl, clients, kontract kont
        where kc_amount_approved > 0 and
              kc_from_k_id = :k_id and
              kc_from_c_id = c_id and
              kont.k_id = kncl.kc_from_k_id
        into :c_id, :amount, c_cell_no, c_name_fio, k_refund
    do
    begin
      res_c_id = c_id;

      if ((right(coalesce(c_cell_no, '000000'), 6) <> '000000') and
          (position(c_cell_no, cell_all) <= 0) and
          (not((db_id in (39801, 41701)) and
          (k_refund = 1)))) then -- igusev 26.04.2020 добавил условие, на основании письма от Бахтияра "Subject: Сервисный смс" Sent: Saturday, April 25, 2020 12:03 PM
      begin
        cell_all = cell_all || c_cell_no || ';';
        c_name_fio = (select short_fio
                      from hp_fio_clear(:c_name_fio));
        uniq_acc_str = (select uniq_acc_str
                        from hp_show_clientuniq_acc_no(:c_id));

        amount_str = round(amount, 0);
        if (char_length(amount_str) > 6) then
          amount_str = left(amount_str, char_length(amount_str) - 6) || ',' || left(right(amount_str, 6), 3) || ',' || right(amount_str, 3);
        else
        if (char_length(amount_str) > 3) then
          amount_str = left(amount_str, char_length(amount_str) - 3) || ',' || right(amount_str, 3);
        amount_str = amount_str || amount_sym;
        kol = kol + 1;

        res_body = '';
        res_maillist = :c_cell_no;
        res_subject = 'Выдача кредита: СМС';
        mcl_project = f_get_mcl_project(f_objkind('kontract', 'id'), :k_id); -- функция определяет проект (М Булак или OiVO)

        if (db_id in (39801, 41701)) then -- КГ
          res_body = 'Л/счет:' || uniq_acc_str || ';Профинансировано ' || amount_str || ';квит:' || doc_no || ';' || :date_format_str;
        else
        /*---------
        if (db_id = 81001) then -- РФ
        begin
          res_body = 'Lisevoi chet:' || uniq_acc_str || '. Vydacha zaima proshla uspeshno. Summa Vashego zayma: ' || amount_str || ' Vasha kvit: ' || doc_no || '; ' || :date_format_str || '. Spasibo, chto vybiraete M Bulak:)';

          -- отправляем в КМП пуш-уведомление
          if (f_clientcom_allowed(c_id, 0, 0, 1) = 1) then
            execute procedure mc_send_push('clients', :c_id, 'Выдача кредита',
              'Лицевой счет:' || uniq_acc_str || '. Выдача займа прошла успешно. Сумма Вашего займа составляет: ' || amount_str || '. Ваша квитанция:' || doc_no || '; ' || :date_format_str || '. Спасибо, что выбираете М Булак. Вы всегда можете позвонить Вашему специалисту по вопросам займа или в колл-центр по номеру 8 800 555 71 71. Мы всегда рады Вам:)');
        end*/
        ------------RFCRU-4257
        if (db_id = 81001) then -- РФ
        begin
        -- СМС для М Булак или OiVO
        if (mcl_project = 1) then -- М Булак
        res_body = 'Lisevoi chet:' || uniq_acc_str || '. Vydacha zaima proshla uspeshno. Summa Vashego zayma: ' || amount_str || ' Vasha kvit: ' || doc_no || '; ' || :date_format_str || '. Spasibo, chto vybiraete M Bulak:)';
        else
        if (mcl_project = 2) then -- OiVO
        res_body = 'Lisevoi chet:' || uniq_acc_str || '. Vydacha zaima proshla uspeshno. Summa Vashego zayma: ' || amount_str || ' Vasha kvit: ' || doc_no || '; ' || :date_format_str || '. Spasibo, chto vybiraete OiVO Kredit:)';

        -- Отправка пуш-уведомлений, только если разрешено
        if (f_clientcom_allowed(c_id, 0, 0, 1) = 1) then
        begin
        if (mcl_project = 1) then -- М Булак
            execute procedure mc_send_push('clients', :c_id, 'Выдача кредита',
                'Лицевой счет:' || uniq_acc_str || '. Выдача займа прошла успешно. Сумма Вашего займа составляет: ' || amount_str || '. Ваша квитанция:' || doc_no || '; ' || :date_format_str || '. Спасибо, что выбираете М Булак. Вы всегда можете позвонить Вашему специалисту по вопросам займа или в колл-центр по номеру 8 800 555 71 71. Мы всегда рады Вам:)');
        else
        if (mcl_project = 2) then -- OiVO
            execute procedure mc_send_push('clients', :c_id, 'Выдача кредита',
                'Лицевой счет:' || uniq_acc_str || '. Выдача займа прошла успешно. Сумма Вашего займа составляет: ' || amount_str || '. Ваша квитанция:' || doc_no || '; ' || :date_format_str || '. Спасибо, что выбираете OiVO Кредит. Вы всегда можете позвонить Вашему специалисту по вопросам займа или в колл-центр по номеру 8 800 555 93 48. Мы всегда рады Вам:)');
        end
        end
        -----------
        else
        if (db_id = 94401) then -- AZ
          res_body = 'Hesab N:' || uniq_acc_str || '; MBulakdan kredit verildi: ' || amount_str || ' / tarix ' || :date_format_str;

        kc_circle = coalesce(kc_circle, 1);
        select count(kc_circle)
        from knclient kc, clients c
        where c.c_id = :c_id and
              kc.kc_amount_approved > 0 and
              kc_from_c_id = c_id
        into kc_circle;

        point = f_point(:c_id, current_date);

        if (db_id <> 81001 or f_clientcom_allowed(c_id, 1, 1, 1) = 1) then
        begin
          if (kc_circle <= 1) then
            execute procedure mc_send_push('clients', :c_id, 'Поздравляем!',
              'Теперь вы участвуете в программе “Надежный клиент”. Вносите платежи без опозданий и получайте баллы.');
          else if (point > 0) then
            execute procedure mc_send_push('clients', :c_id, 'С возвращением!',
              'Мы с хорошими новостями! На вашем счету ' || :point || ' баллов. Платите вовремя, снижайте ставку и копите баллы!');
        end

        if (db_id <> 81001 or f_clientcom_allowed(c_id, 0, 0, 1) = 1) then
          suspend;
      end
    end
  end
end
