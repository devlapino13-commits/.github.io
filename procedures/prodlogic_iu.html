CREATE PROCEDURE PRODLOGIC_IU(
    PRL_FROM_PR_ID INTEGER,
    PRL_ID INTEGER,
    PRL_DATE_BEGIN DATE,
    PRL_DATE_END DATE,
    PRL_CIRCLE_MIN INTEGER,
    PRL_CIRCLE_MAX INTEGER,
    PRL_KOLICH_MIN INTEGER,
    PRL_KOLICH_MAX INTEGER,
    PRL_AMOUNT_GR_MIN DECIMAL(18,4),
    PRL_AMOUNT_GR_MAX DECIMAL(18,4),
    PRL_AMOUNT_CL_MIN DECIMAL(18,4),
    PRL_AMOUNT_CL_MAX DECIMAL(18,4),
    PRL_REPAY_MIN INTEGER,
    PRL_REPAY_MAX INTEGER,
    PRL_PERCENT_MIN DECIMAL(18,7),
    PRL_PERCENT_MAX DECIMAL(18,7),
    PRL_ADMIN_PERC_MIN DECIMAL(18,7),
    PRL_ADMIN_PERC_MAX DECIMAL(18,7),
    PRL_MIN_MAX_DIFF DECIMAL(18,4),
    PRL_MALE_MAX_PERC DECIMAL(18,7))
AS
begin
  if (exists(select prl_from_pr_id
             from prodlogic
             where (prl_from_pr_id = :prl_from_pr_id) and
                   (prl_id = :prl_id))) then
    update prodlogic
    set prl_date_begin = :prl_date_begin,
        prl_date_end = :prl_date_end,
        prl_circle_min = :prl_circle_min,
        prl_circle_max = :prl_circle_max,
        prl_kolich_min = :prl_kolich_min,
        prl_kolich_max = :prl_kolich_max,
        prl_amount_gr_min = :prl_amount_gr_min,
        prl_amount_gr_max = :prl_amount_gr_max,
        prl_amount_cl_min = :prl_amount_cl_min,
        prl_amount_cl_max = :prl_amount_cl_max,
        prl_repay_min = :prl_repay_min,
        prl_repay_max = :prl_repay_max,
        prl_percent_min = :prl_percent_min,
        prl_percent_max = :prl_percent_max,
        prl_admin_perc_min = :prl_admin_perc_min,
        prl_admin_perc_max = :prl_admin_perc_max,
        prl_min_max_diff = :prl_min_max_diff,
        prl_male_max_perc = :prl_male_max_perc
    where (prl_from_pr_id = :prl_from_pr_id) and
          (prl_id = :prl_id);
  else
    insert into prodlogic (prl_from_pr_id, prl_id, prl_date_begin, prl_date_end, prl_circle_min, prl_circle_max,
                           prl_kolich_min, prl_kolich_max, prl_amount_gr_min, prl_amount_gr_max, prl_amount_cl_min,
                           prl_amount_cl_max, prl_repay_min, prl_repay_max, prl_percent_min, prl_percent_max,
                           prl_admin_perc_min, prl_admin_perc_max, prl_min_max_diff, prl_male_max_perc)
    values (:prl_from_pr_id, :prl_id, :prl_date_begin, :prl_date_end, :prl_circle_min, :prl_circle_max, :prl_kolich_min,
            :prl_kolich_max, :prl_amount_gr_min, :prl_amount_gr_max, :prl_amount_cl_min, :prl_amount_cl_max,
            :prl_repay_min, :prl_repay_max, :prl_percent_min, :prl_percent_max, :prl_admin_perc_min,
            :prl_admin_perc_max, :prl_min_max_diff, :prl_male_max_perc);
end
