CREATE PROCEDURE HP_BLANKBLOB_JSON_FIELDS(
    BLB_FROM_BLN_ID D_INTEGER,
    BLB_NUMBER D_INTEGER)
RETURNS (
    BJL_FROM_BLN_ID D_INTEGER,
    BJL_FROM_BLB_NUMBER D_INTEGER,
    BJF_ID D_INTEGER,
    BJF_TABLE_NAME D_STR_CODE,
    BJF_FIELD_NAME D_STR_255,
    BJF_CAPTION D_STR_255)
AS
begin
  for select bjl_from_bln_id, bjl_from_blb_number, bjf_id,
             bjf_table_name, bjf_field_name, bjf_caption
      from blankjsonlink, blankjsonfields
      where bjl_from_bjf_id = bjf_id and
            bjl_from_bln_id = :blb_from_bln_id and
            bjl_from_blb_number = :blb_number
   order by bjf_table_name, bjf_field_name
      into :bjl_from_bln_id, :bjl_from_blb_number, :bjf_id,
           :bjf_table_name, :bjf_field_name, :bjf_caption
   do
     Suspend;
end
