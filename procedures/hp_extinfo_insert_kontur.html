CREATE PROCEDURE HP_EXTINFO_INSERT_KONTUR(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    FORCE_INSERT D_SMALLINT,
    UN_ID D_ID = null,
    CONFIG_KIND D_SMALLINT = 0)
AS
declare variable obj_kind_str d_str_code;
declare variable c_id d_id;
declare variable is_pledgor d_bool;
declare variable pk_id_cnc d_integer;
declare variable pk_id_ok d_integer;
declare variable drl_id d_id;
declare variable actual_days d_integer;
declare variable obj_kind_drl d_integer;
declare variable eil_date d_date;
declare variable eic_id d_integer;
declare variable config d_str_code;
begin
  obj_kind_str = f_objkind(obj_kind, 'name');

  config = (case config_kind
              when 0 then 'kontur'
              when 1 then 'kontur_b2'
            end);

  -- Отправляем клиента/ов договора/заявки
  execute procedure hp_extinfo_insert(:config, :obj_kind, :obj_id, :force_insert, :un_id);

  -- Далее по заемщикам/клиентам отправляем смену ФИО, если было
  for select kc_from_c_id, 0
      from knclient
      where :obj_kind_str = 'Kontract' and
            kc_from_k_id = :obj_id and
            kc_amount_approved > 0
      union
      select mac_from_c_id, 0
      from miniappclient
      where :obj_kind_str = 'Miniapp' and
            mac_from_ma_id = :obj_id and
            mac_amount > 0
      union
      select c_id, 0
      from clients
      where :obj_kind_str = 'Clients' and
            c_id = :obj_id
      union
      -- Для автозаймов добавляем залогодателей
      select c_id, 1
      from kontract, product, zalog, clients
      where :obj_kind_str = 'Kontract' and
        k_id = :obj_id and
        z_from_k_id = k_id and
        k_from_pr_id = pr_id and
        z_from_c_id = c_id and
        c_type = 0 and
        pr_code containing '.zalog_car.' and
        z_type = 1 and
        z_from_c_id is not null
        into c_id, is_pledgor
  do
  begin
    -- Добавляем залогодателей
    if (is_pledgor = 1) then
      execute procedure hp_extinfo_insert(:config, 'clients', :c_id, :force_insert, :un_id);

    -- Далее добавляем старое ФИО для клиентов и залогодателей, если есть документ о смене

    -- Получаем pk_id документа "Свидетельство о перемене Фамилии, Имени, Отчества"
    if (pk_id_cnc is null) then
      select pk_id
      from docrecogmain, passkind
      where drm_code = 'rus.cert.name_change'
        and pk_from_drm_id = drm_id
      into pk_id_cnc;

    -- Получаем pk_id документа "Свидетельство о перемене Фамилии, Имени, Отчества"
    if (pk_id_ok is null) then
      select pk_id
      from docrecogmain, passkind
      where drm_code = 'rus.other.kontur'
        and pk_from_drm_id = drm_id
      into pk_id_ok;

    if (obj_kind_drl is null) then
      obj_kind_drl = f_objkind('docrecoglist', 'id');

    -- Получаем id конфигурации и количество дней хранения
    if (actual_days is null) then
    select eic_id, eic_info_actual_days
    from extinfocfg
    where eic_code = :config
    into eic_id, actual_days;

    -- Обходим список документов:
    --   "Свидетельство о перемене Фамилии, Имени, Отчества"
    --   "Данные для проверки Контур/Федресурс"

    for select cds_from_drl_id
    from clientdocs
    where cds_from_c_id = :c_id
      and cds_from_pk_id in (:pk_id_cnc, :pk_id_ok)
      and cds_is_delete = 0
    into drl_id
    do
    begin
      -- Проверяем, возможно документ добавлен и не устарел
      -- Данный поиск учитывает object_kind, object_id в отличие от
      -- HP_EXTINFO_INSERT которая ищет на уровне клиента (edl_from_c_id)
      eil_date = null;
      select eil_date
      from extinfolist
      where eil_from_c_id = :c_id and
            eil_object_kind = :obj_kind_drl and
            eil_object_id = :drl_id and
            eil_from_eic_id = :eic_id and
            eil_status in (-1, 0, 1, 2) and
            datediff(day, eil_date, current_date) <= :actual_days
      order by eil_date desc, eil_create_datetime desc
      rows 1
      into eil_date;

      -- Добавляем, если не найден действующий или force_insert = 1 и сегодня не отправляли
      if (eil_date is null or (eil_date < current_date and force_insert = 1)) then
        execute procedure hp_extinfo_insert(:config, 'docrecoglist', :drl_id, :force_insert, :un_id);
    end
  end
end
