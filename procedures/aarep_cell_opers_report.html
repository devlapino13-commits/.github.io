CREATE PROCEDURE AAREP_CELL_OPERS_REPORT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    CB_AMOUNT DECIMAL(18,4),
    CAL_REGION_NAME VARCHAR(100),
    CAL_LIMIT_AMOUNT DECIMAL(18,4),
    CB_AMOUNT_WITH_LIMIT DECIMAL(18,4),
    CAL_AMOUNT_TO_SALARY DECIMAL(18,4),
    CAL_AMOUNT_TAX_TO_SALARY DECIMAL(18,4),
    CAL_BRANCH_NAME VARCHAR(100))
AS
declare variable salary_tax_koef decimal(18,7);
declare variable report_month integer;
declare variable report_year integer;
begin
  report_month = extract(month from :report_date_end);
  report_year = extract(year from :report_date_end);
  salary_tax_koef = 3.5;
  for select sum(cb_amount), cal_region_name, sum(cal_limit_amount), sum(cb_amount_with_limit),
             sum(cal_amount_to_salary), sum(cal_amount_tax_to_salary), reg_code || ' ' || reg_name
      from aarep_cell_opers_personal(:report_year, :report_month, :report_date_end, :report_region_id, 0, 0, :salary_tax_koef, :report_date_end), region
      where reg_unicode = cal_branch_id
      group by cal_region_name, reg_code, reg_name
      order by reg_code, cal_region_name
      into :cb_amount, :cal_region_name, :cal_limit_amount, :cb_amount_with_limit, :cal_amount_to_salary,
           :cal_amount_tax_to_salary, :cal_branch_name
  do
  begin
    suspend;
  end
end
