CREATE PROCEDURE HP_INVENTORY_SIGN(
    IM_ID D_ID,
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    SIGN_STATUS D_SMALLINT)
AS
declare variable imem_id d_id;
declare variable c_id_tmp d_id;
declare variable im_status d_smallint;
begin
  select un.un_from_c_id
  from username un
  where un.un_id = :un_id
  into c_id_tmp;

  if (c_id_tmp <> c_id) then
  begin
    res_text = 'Вы не можете подписать акт за другого участника';
    sign_status = 0;
    suspend;
    exit;
  end 

  select im.im_status
  from invmain im
  where im.im_id = :im_id
  into im_status;

  if (coalesce(im_status, -777) <> 2) then
  begin
    res_text = 'Подпись возможна только при статусе "Проверен".';
    suspend;
    exit;
  end 

  select coalesce(imem.imem_sign_status, 0), imem.imem_id
  from invmember imem
  where imem.imem_from_im_id = :im_id and
        imem.imem_from_c_id = :c_id
  into sign_status, imem_id;

  --Если запрос выше не дал результат значит
  if (coalesce(imem_id, -1) = -1) then
  begin
    res_text = 'Вы не являетесь участником комиссии в данном акте. Подписание отменяется.';
    suspend;
    exit;
  end 

  if (sign_status = 0) then
  begin
    update invmember imem
    set imem.imem_sign_status = 1,
        imem.imem_sign_type = 0 --подпись через FinStart
    where imem.imem_id = :imem_id;

    res_text = 'Акт успешно подписан.';
    sign_status = 1;
  end
  else
  if (sign_status = 1) then
    res_text = 'Акт уже подписан вами.';

  suspend;
end
