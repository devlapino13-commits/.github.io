CREATE PROCEDURE HP_EXTSYS_OPER_ELSOM(
    PARAMS BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    ESM_ID DECIMAL(18,0))
AS
declare variable project varchar(20);
declare variable teg_name str_100;
declare variable index_start integer;
declare variable ch_qty integer;
declare variable index_finish integer;
declare variable params_2 blob sub_type 1 segment size 80;
declare variable esf_id integer;
declare variable protocol_version str_100;
declare variable request_type varchar(10);
declare variable terminal_id str_100;
declare variable transaction_number str_100;
declare variable currency_code_from varchar(3);
declare variable amount_from str_100;
declare variable service_id str_100;
declare variable currency_code_to varchar(3);
declare variable amount_to str_100;
declare variable account_number str_100;
declare variable datetime str_100;
declare variable receipt_number str_100;
begin
--Процедура создает extsysmain и заполняет значения extsysvalues по проекту ЭЛ-СОМ
  project = 'EL-SOM';

  /* Создание записи в extsysmain, далее она будет обновляться */
  insert into extsysmain (esm_id, esm_project, esm_date, esm_datetime, esm_full_source)
  values (-1, :project, current_date, current_timestamp, :params)
  returning esm_id
  into esm_id;

  teg_name = 'protocol-version';
  index_start = position('<' || teg_name || '>' in params) + character_length(teg_name) + 2;
  ch_qty = position('</' || teg_name || '>' in params) - index_start;
  protocol_version = substring(params from index_start for ch_qty);

  select esf_id
  from extsysfields
  where esf_field_code = :teg_name and
        esf_project = :project
  into esf_id;

  insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
  values (:esm_id, :esf_id, :protocol_version);

--<<----------
  teg_name = 'request-type';
  index_start = position('<' || teg_name || '>' in params) + character_length(teg_name) + 2;
  ch_qty = position('</' || teg_name || '>' in params) - index_start;
  request_type = substring(params from index_start for ch_qty);

  select esf_id
  from extsysfields
  where esf_field_code = :teg_name and
        esf_project = :project
  into esf_id;

  insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
  values (:esm_id, :esf_id, :request_type);

--<<----------
  teg_name = 'terminal-id';
  index_start = position('<' || teg_name || '>' in params) + character_length(teg_name) + 2;
  ch_qty = position('</' || teg_name || '>' in params) - index_start;
  terminal_id = substring(params from index_start for ch_qty);

  select esf_id
  from extsysfields
  where esf_field_code = :teg_name and
        esf_project = :project
  into esf_id;

  insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
  values (:esm_id, :esf_id, :terminal_id);

--<<----------
  teg_name = 'transaction-number';
  index_start = position('<' || teg_name || '>' in params) + character_length(teg_name) + 2;
  ch_qty = position('</' || teg_name || '>' in params) - index_start;
  transaction_number = substring(params from index_start for ch_qty);

  select esf_id
  from extsysfields
  where esf_field_code = :teg_name and
        esf_project = :project
  into esf_id;

  insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
  values (:esm_id, :esf_id, :transaction_number);

--<<----------
-- <from>...<amount currency-code="417">10</amount>....</from>
  if (position('<from>' in params) > 0) then
  begin
    index_start = position('<amount currency-code=' in params);
    currency_code_from = substring(params from index_start + 23 for 3);

    select esf_id
    from extsysfields
    where esf_field_code = 'currency_code_from' and
          esf_project = :project
    into esf_id;

    insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
    values (:esm_id, :esf_id, :currency_code_from);

    index_finish = position('</amount>' in params);
    amount_from = substring(params from index_start + 28 for index_finish - (index_start + 28));

    select esf_id
    from extsysfields
    where esf_field_code = 'amount_from' and
          esf_project = :project
    into esf_id;

    insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
    values (:esm_id, :esf_id, :amount_from);
--
    params_2 = substring(params from index_start + 28 + (index_finish - (index_start + 28)) + 9);
  end
  else
  begin
    params_2 = params;
  end

-- <to>...<amount currency-code="417">10</amount>....</to>
  if (position('<to>' in params) > 0) then
  begin
    teg_name = 'service-id';
    index_start = position('<' || teg_name || '>' in params) + character_length(teg_name) + 2;
    ch_qty = position('</' || teg_name || '>' in params) - index_start;
    service_id = substring(params from index_start for ch_qty);

    select esf_id
    from extsysfields
    where esf_field_code = :teg_name and
          esf_project = :project
    into esf_id;

    insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
    values (:esm_id, :esf_id, :service_id);

--<<----------
    teg_name = 'account-number';
    index_start = position('<' || teg_name || '>' in params) + character_length(teg_name) + 2;
    ch_qty = position('</' || teg_name || '>' in params) - index_start;
    account_number = substring(params from index_start for ch_qty);

    select esf_id
    from extsysfields
    where esf_field_code = :teg_name and
          esf_project = :project
    into esf_id;

    insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
    values (:esm_id, :esf_id, :account_number);

--<<----------
    index_start = position('<amount currency-code=' in params_2);
    currency_code_to = substring(params_2 from index_start + 23 for 3);

    select esf_id
    from extsysfields
    where esf_field_code = 'currency_code_to' and
          esf_project = :project
    into esf_id;

    insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
    values (:esm_id, :esf_id, :currency_code_to);

    index_finish = position('</amount>' in params_2);
    amount_to = substring(params_2 from index_start + 28 for index_finish - (index_start + 28));

    select esf_id
    from extsysfields
    where esf_field_code = 'amount_to' and
          esf_project = :project
    into esf_id;

    insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
    values (:esm_id, :esf_id, :amount_to);
  end

--20150804000113 преобразовываем в '2015.08.04:00:01.13'
  teg_name = 'datetime';
  index_start = position('<' || teg_name || '>' in params) + character_length(teg_name) + 2;
  ch_qty = position('</' || teg_name || '>' in params) - index_start;
  datetime = substring(params from index_start for ch_qty);

  /*
     datetime_date =  substring(datetime from 7 for 2)||'.'||
                      substring(datetime from 5 for 2)||'.'||
                      substring(datetime from 1 for 4);
     */

  select esf_id
  from extsysfields
  where esf_field_code = :teg_name and
        esf_project = :project
  into esf_id;

  insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
  values (:esm_id, :esf_id, :datetime);

--<<----------
  teg_name = 'receipt-number';
  index_start = position('<' || teg_name || '>' in params) + character_length(teg_name) + 2;
  ch_qty = position('</' || teg_name || '>' in params) - index_start;
  receipt_number = substring(params from index_start for ch_qty);

  select esf_id
  from extsysfields
  where esf_field_code = :teg_name and
        esf_project = :project
  into esf_id;

  insert into extsysvalues (esv_from_esm_id, esv_from_esf_id, esv_value_txt)
  values (:esm_id, :esf_id, :receipt_number);
--<<----------
--end

  suspend;
end
