CREATE PROCEDURE CIB_POCKET_SET_STATUS(
    CIB_NAME STR_100,
    POCKET_ID INTEGER,
    POCKET_EXTERNAL_STATUS STR_100)
RETURNS (
    POCKET_INTERNAL_STATUS INTEGER)
AS
declare variable k_id id_type;
begin
  -- Процедура используется для Эквифакс и НБКИ
  if (cib_name in ('EQUIFAX', 'NBKI')) then
  begin
    pocket_internal_status = (case pocket_external_status
                                when 'PocketRejected' then -2
                                when 'TechnicalErrorNotSent' then -1
                                when 'Sended' then 1
                                when 'XSDChecked' then 2
                                when 'Processed' then 3
                                else -999
                              end);
    update cib_pocket cp
    set cp_pocket_status = :pocket_internal_status,
        cp.cp_pocket_kib_status = :pocket_external_status,
        cp.cp_status_datetime = current_timestamp
    where cp.cp_id = :pocket_id;
  end

  suspend;
end
