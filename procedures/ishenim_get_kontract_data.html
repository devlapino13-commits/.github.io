CREATE PROCEDURE ISHENIM_GET_KONTRACT_DATA(
    POCKET_ID INTEGER,
    START_RECORD INTEGER,
    LAST_RECORD INTEGER,
    DO_UPDATE SMALLINT = 1)
RETURNS (
    K_ID ID_TYPE,
    CONTRACTCODE ID_TYPE,
    PHASEOFCONTRACT STR_100,
    CONTRACTSTATUS STR_100,
    TYPEOFCONTRACT STR_100,
    CONTRACTSUBTYPE STR_100,
    OWNERSHIPTYPE STR_100,
    CURRENCYOFCONTRACT VARCHAR(3),
    TOTALTAKENAMOUNT CURR_TYPE,
    AMOUNTWRITTENOFF CURR_TYPE,
    INTERESTWRITTENOFF CURR_TYPE,
    NUMBEROFINSTALLMENTS INTEGER,
    RESTRUCTURINGDATE VARCHAR(10),
    ADDITIONALFEESWRITTENOFF CURR_TYPE,
    STARTDATE VARCHAR(10),
    EXPECTEDENDDATE VARCHAR(10),
    REALENDDATE VARCHAR(10),
    INSTALLMENTTYPE STR_100,
    INSTALLMENTAMOUNT CURR_TYPE,
    PRINCIPALOUTSTANDINGAMOUNT CURR_TYPE,
    INTERESTOUTSTANDINGAMOUNT CURR_TYPE,
    PASTDUEAMOUNT CURR_TYPE,
    PASTDUEINTEREST CURR_TYPE,
    ADDITIONALFEESSUM CURR_TYPE,
    NEXTPAYMENTDATE VARCHAR(10),
    TOTALMONTHLYPAYMENT CURR_TYPE,
    PAYMENTPERIODICITY STR_100,
    NUMBEROFDUEINSTALLMENTS INTEGER,
    NEGATIVESTATUSOFTHECONTRACT STR_100,
    PROLONGATIONAMOUNT CURR_TYPE,
    PROLONGATIONDATE VARCHAR(10),
    PASTDUEDAYS INTEGER,
    OFFICE_ID INTEGER,
    REGION_ID STR_100,
    TOTALAMOUNT CURR_TYPE,
    PURPOSEOFFINANCING STR_100)
AS
declare variable rep_date_end date;
declare variable k_kolich_client_approved integer;
declare variable k_currency_id_old varchar(3);
declare variable k_currency_id varchar(3);
declare variable closed_ol_id id_type;
declare variable write_off_os curr_type;
declare variable pr_wm_kolich integer;
declare variable pr_week_or_month integer;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable sh_calc_fine curr_type;
declare variable sh_date date;
declare variable k_kredline_end date;
declare variable office_id_old integer;
declare variable pl_name str_100;
declare variable kf_id id_type;
declare variable fact_closed_date date;
declare variable pocket_type varchar(10);
declare variable rep_date_begin date;
declare variable do_suspend smallint;
declare variable reg_ishenim_region integer;
declare variable kc_busuness_subtype integer;
declare variable bt_name varchar(50);
declare variable kc_loan_target integer;
declare variable loan_target str_255;
declare variable is_fixed smallint;
declare variable is_restruct integer;
begin
  select cp.cp_date_begin, cp.cp_date_end, cp.cp_pocket_type
  from cib_pocket cp
  where cp.cp_id = :pocket_id
  into :rep_date_begin, :rep_date_end, :pocket_type;

  if (rep_date_end is null) then
    exit;

  do_suspend = 0;
  if (pocket_type = 'm') then
    do_suspend = 1;

  k_currency_id_old = '';

  additionalfeeswrittenoff = 0;
  prolongationamount = 0;
  prolongationdate = '';
  restructuringdate = '';
  office_id_old = 0;

  for select k_id, phaseofcontract, is_fixed
      from ishenim_get_kontract_data_tmp(:pocket_id)
      rows :start_record to :last_record
      into :k_id, :phaseofcontract, :is_fixed
  do
  begin
    pastduedays = 0;

    select :rep_date_end - oks.oks_delinq_date
    from oksummary oks
    where oks.oks_from_k_id = :k_id and
          oks.oks_close_state = 0 and
          :rep_date_end between oks.oks_date_begin and oks.oks_date_end
    into :pastduedays;

    pastduedays = coalesce(pastduedays, 0);

    select kont.k_kolich_client_approved, kont.k_currency, kont.k_amount_approved,
           trim(iif(pr_interest_method = 2, 'Installment', 'NonInstallment')), kont.k_kolich_repay, pr.pr_wm_kolich,
           pr.pr_week_or_month,
           (select result_date_str
            from cib_date_to_str(kont.k_fact_disb_date)) as startdate, kont.k_kredline_end,
           iif(:phaseofcontract = 'Closed', kont.k_fact_closed_ol_id, null), kont.k_from_kf_id,
           kont.k_fact_closed_date
    from kontract kont
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kont.k_id = :k_id
    order by kont.k_currency
    into :k_kolich_client_approved, :k_currency_id, :totaltakenamount, :typeofcontract, :numberofinstallments,
         :pr_wm_kolich, :pr_week_or_month, :startdate, :k_kredline_end, :closed_ol_id, :kf_id, :fact_closed_date;

    /* если тип пакета - ежеденвный тогда в рамках пакета отправляю кредиты. Если:
        а) закрыт за период
        б) находится в просрочке
        в) выдан за период 
        г) был платеж за период
    */

    if (pocket_type = 'd')  then
    begin
      do_suspend = 0;

      if (is_fixed = 1) then
        do_suspend = 1;
      else if (phaseofcontract = 'Closed') then
        do_suspend = 1;
      else if (pastduedays > 0) then
        do_suspend = 1;
      else if (startdate between :rep_date_begin and :rep_date_end) then
        do_suspend = 1;
      else if (do_suspend = 0) then
      begin
        if (exists(select *
                   from operkred ok
                   where ok.ok_type in (2, 4, 9) and
                         ok.ok_ol_is_active = 1 and
                         ok.ok_from_k_id = :k_id and
                         ok.ok_ol_date between :rep_date_begin and :rep_date_end)) then
          do_suspend = 1;
      end
    end


    if (do_suspend = 1) then
    begin
      totalamount = :totaltakenamount;

      select kal.kal_new_reg_id
      from knaktlist kal
      where kal.kal_from_kf_id = :kf_id and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into :office_id;

      realenddate = null;

      if (closed_ol_id is not null) then
      begin
        select ol.ol_date
        from operlist ol
        where ol.ol_id = :closed_ol_id and
              ol.is_delete = 0
        into :realenddate;
      end

      if (office_id <> office_id_old) then
      begin
        office_id_old = office_id;

        select reg_ishenim_region
        from region
        where reg_unicode = :office_id
        into reg_ishenim_region;

        region_id = null;
        select rd_str_id
        from refdata
        where rd_id = :reg_ishenim_region
        into region_id;
        region_id = coalesce(region_id, 'Other');
      end

      expectedenddate = (select result_date_str
                         from cib_date_to_str(:k_kredline_end));

      if (k_currency_id <> k_currency_id_old) then
      begin
        select cul.cu_symbol
        from currlist cul
        where cul.cu_id = :k_currency_id
        into :currencyofcontract;

        k_currency_id_old = k_currency_id;
      end

      contractcode = k_id;
      ownershiptype = iif(:k_kolich_client_approved > 1, 'GroupLoan', 'Individual');

      --  Если групповой кредит то передаем NotSpecified, а если в группе один участник то передаем по определению типа бизнеса
      if (ownershiptype = 'GroupLoan') then
      begin
        contractsubtype = 'NotSpecified'; -- Тип кредита
        purposeoffinancing = 'NotSpecified'; -- Цель финансирования
      end
      else
      begin
        kc_busuness_subtype = null;
        select kc_busuness_subtype, kc_loan_target
        from knclient
        where kc_amount_approved > 0 and
              kc_from_k_id = :k_id rows 1
        into kc_busuness_subtype, kc_loan_target;

        loan_target = null;
        select rd_code
        from refdata
        where rd_id = :kc_loan_target
        into loan_target;
        loan_target = coalesce(loan_target, '');
        if (loan_target = 'Loan_Target.Consumer') then
          contractsubtype = 'ConsumerLoan';
        else
        if (loan_target in ('Loan_Target.BusinessGrowth', 'Loan_Target.BusinessCreation')) then
          contractsubtype = 'BusinessLoan';
        else
          contractsubtype = 'NotSpecified';

        bt_name = null;
        select bt_name
        from bussubtype, bustype
        where bs_id = :kc_busuness_subtype and
              bs_from_bt_id = bt_id
        into bt_name;
        bt_name = coalesce(bt_name, '');

        purposeoffinancing = case bt_name
                               when 'Торговля' then 'TradeFinanceFacility'
                               when 'Услуги' then 'SocialServices'
                               when 'Производство' then 'Industry'
                               when 'Сельское хозяйство' then 'Agriculture'
                               when 'Животноводство' then 'Agriculture'
                               when 'Микрокредитование' then 'TradeFinanceFacility'
                               when 'Физические лица Зарплата' then 'PersonalLoan'
                               else 'NotSpecified'
                             end;
      end

      write_off_os = 0;
      interestwrittenoff = 0;
      amountwrittenoff = 0;

      select -sum(ok.ok_os), -sum(ok.ok_perc)
      from operkred ok
      where ok.ok_from_ol_id = :closed_ol_id and
            ok.ok_from_k_id = :k_id and
            ok.ok_type = 5 and
            ok_from_c_id = 0
      into :write_off_os, :interestwrittenoff;

      write_off_os = coalesce(write_off_os, 0);
      interestwrittenoff = coalesce(interestwrittenoff, 0);
      amountwrittenoff = (write_off_os + interestwrittenoff);

      if (typeofcontract = 'Installment') then
        installmenttype = 'Fixed';
      else
        installmenttype = 'Variable';

      select (sh.sh_os + sh.sh_perc)
      from schedule sh
      where sh.sh_from_k_id = :k_id and
            sh.sh_kommis = 0 and
            ((sh.sh_os <> 0) or
            (sh.sh_perc <> 0))
      order by sh.sh_date
      rows 1
      into :installmentamount;

      principaloutstandingamount = 0;
      interestoutstandingamount = 0;
      pastdueamount = 0; -- deliq_os
      pastdueinterest = 0; -- delinq %
      additionalfeessum = 0; --UNPAIDFINE
      nextpaymentdate = '';
      totalmonthlypayment = 0; --FACT_AMOUNT
      numberofdueinstallments = 0; -- число активных просроченных платежей
      negativestatusofthecontract = 'NoNegativeStatus';

      if (phaseofcontract = 'Open') then
      begin
        contractstatus = 'GrantedAndActivated';
        if (pastduedays < 0) then
          pastduedays = 0;

        if (pastduedays < 30) then
          negativestatusofthecontract = 'UpToOneMonthOverdue'; --до 30 дней просрочки
        else
        if ((pastduedays >= 30) and
            (pastduedays < 60)) then
          negativestatusofthecontract = 'TwoMonthsOverdue'; --30 дней до 59 дней просрочки
        else
        if ((pastduedays >= 60) and
            (pastduedays < 90)) then
          negativestatusofthecontract = 'ThreeMonthsOverDue'; --60 дней до 89 дней просрочки
        else
        if ((pastduedays >= 90) and
            (pastduedays < 120)) then
          negativestatusofthecontract = 'FourMonthsOverdue'; --90 дней  до 119 дней просрочки
        else
        if (pastduedays >= 120) then
          negativestatusofthecontract = 'FourMonthsOverdue'; --120 дней просрочки и более
        else
          negativestatusofthecontract = 'NotSpecified';

        for select zad_os, zad_perc, sh_calc_fine, sh_date
            from aarep_make_zad_list(:k_id, :rep_date_end, 1)
            where sh_date is not null
            into :zad_os, :zad_perc, :sh_calc_fine, :sh_date
        do
        begin
          if (sh_date < rep_date_end) then
          begin
            pastdueamount = pastdueamount + zad_os;
            pastdueinterest = pastdueinterest + zad_perc;
            additionalfeessum = additionalfeessum + sh_calc_fine;

            numberofdueinstallments = numberofdueinstallments + 1;
          end
          else
          begin
            if (nextpaymentdate = '') then
              nextpaymentdate = (select result_date_str
                                 from cib_date_to_str(:sh_date));
          end

          principaloutstandingamount = principaloutstandingamount + zad_os;
          interestoutstandingamount = interestoutstandingamount + zad_perc;
        end

        select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_tax)
        from operkred ok
        where ok.ok_type in (4, 9) and
              ok.ok_ol_is_active = 1 and
              ok.ok_from_k_id = :k_id and
              ok.ok_ol_date <= :rep_date_end and
              ok.ok_kommis = 0 /*не комиссия*/
              and
              ok_from_c_id = 0
        group by ok.ok_ol_date
        order by ok.ok_ol_date desc
        rows 1
        into :totalmonthlypayment;
        totalmonthlypayment = coalesce(totalmonthlypayment, 0);

         if (totalmonthlypayment <0) then
        begin
           totalmonthlypayment = 0;
        end
      end

      ----------- частота выплат -------------------------------------------------
      paymentperiodicity = 'Other';
      if (pr_wm_kolich = 1 and
          pr_week_or_month = 1) then    -- еженедельно
        paymentperiodicity = 'Days7';
      else
      if (pr_wm_kolich = 2 and
          pr_week_or_month = 1) then    -- раз в две недели
        paymentperiodicity = 'Days15';
      else
      if (pr_wm_kolich = 1 and
          pr_week_or_month = 2) then    -- ежемесячно
        paymentperiodicity = 'Days30';
      else
      if (pr_wm_kolich = 2 and
          pr_week_or_month = 2) then    -- раз в два месяца
        paymentperiodicity = 'Days60';
      else
      if (pr_wm_kolich = 3 and
          pr_week_or_month = 2) then    -- поквартально
        paymentperiodicity = 'Days90';
      else
      if (pr_wm_kolich = 4 and
          pr_week_or_month = 2) then     -- три раза год
        paymentperiodicity = 'Days120';
      else
      if (pr_wm_kolich = 6 and           -- раз в полгода
          pr_week_or_month = 2) then
        paymentperiodicity = 'Days180';
      else
      if (pr_wm_kolich = 12 and         -- ежегодно
          pr_week_or_month = 2) then
        paymentperiodicity = 'Days360';

      ----------------------------------------------------------------------------

      /*
      if (do_update = 1) then
      begin
        ...
      end
      */

      if (phaseofcontract = 'Closed') then
      begin
        contractstatus = 'Settled';

        if (fact_closed_date < k_kredline_end) then
          contractstatus = 'SettledInAdvance'; -- погашен досрочно
        else
        if (fact_closed_date = k_kredline_end) then
          contractstatus = 'SettledOnTime'; -- погашен вовремя
        else
        if (fact_closed_date > k_kredline_end) then
          contractstatus = 'SettledLate'; -- погашен c опозданием

        if (amountwrittenoff > 0) then
        begin
            if (write_off_os = 0) then         ---Если списан контракт не по ОС, то статус закрытый
                contractstatus = 'SettledOnTime';
            else
                contractstatus = 'WrittenOff';
        end
      end

      if (contractstatus = 'SettledOnTime') then    --Если списан по % , то сумма списание =0
      begin
        amountwrittenoff = 0;
        interestwrittenoff = 0;
      end

      if (phaseofcontract = 'Open') then
      begin
        select is_restruct
        from aarep_get_restructuring(:k_id, :rep_date_end, 41701)
        into :is_restruct ;

        if (is_restruct = 1) then
          contractstatus = 'Restructured';
      end 

      if (is_fixed = 1) then
      begin
        select iif(ca.phaseofcontract = '---', :phaseofcontract, ca.phaseofcontract),
               iif(ca.contractstatus = '---', :contractstatus, ca.contractstatus),
               iif(ca.amountwrittenoff = -777, :amountwrittenoff, ca.amountwrittenoff),
               iif(ca.interestwrittenoff = -777, :interestwrittenoff, ca.interestwrittenoff)
        from cib_actuality ca where ca.k_id = :k_id
        into :phaseofcontract, :contractstatus, :amountwrittenoff, :interestwrittenoff;
      end

      suspend;
    end
  end
end
