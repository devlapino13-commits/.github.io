CREATE PROCEDURE AAREP_GET_OFFICE_PAR(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    BALANCE_OFFICE CURR_TYPE,
    DELINQ_BALANCE8_180 CURR_TYPE,
    PAR8_180 DECIMAL(5,5))
AS
declare variable k_id id_type;
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate curr_type;
declare variable cr_rate curr_type;
declare variable ok_os curr_type;
declare variable def_curr varchar(3);
declare variable dd date;
begin
--  Вычисляет долларовый курс на дату
  def_curr = (select cm_default_currency
              from company);

-- Вытаскивает активные кредиты для расчета баланса и баланса просрочки
  balance_office = 0;
  delinq_balance8_180 = 0;

  for select oks_from_k_id, k_index_curr_code, k_index_curr_rate, oks_balance, oks_delinq_date
      from kontract, oksummary
      where oks_from_k_id = k_id and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_region_id = :rep_region_id and
            oks_close_state = 0
      into k_id, k_index_curr_code, k_index_curr_rate, :ok_os, :dd
  do
  begin
    if (ok_os is null) then
      ok_os = 0;

    if (k_index_curr_code <> def_curr) then
    begin
      cr_rate = null;
      select cr_rate
      from opercurr
      where cr_from_cu_id = :k_index_curr_code and
            :rep_date_end between cr_date and cr_date_end
      into cr_rate;
      if (cr_rate is null) then
        cr_rate = 1;

      if (cr_rate > k_index_curr_rate) then
        ok_os = ok_os * cr_rate;
      else
        ok_os = ok_os * k_index_curr_rate;
    end
--  Расчет просроченного баланса от 8 - 180 дней
    if ((:rep_date_end - dd) between 8 and 180) then
      delinq_balance8_180 = delinq_balance8_180 + ok_os;
--  Расчет баланса
    balance_office = balance_office + ok_os;
  end
  par8_180 = 0;
  if (balance_office > 0) then
    par8_180 = cast(delinq_balance8_180 as double precision) / balance_office * 100;
  suspend;
end
