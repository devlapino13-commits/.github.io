CREATE PROCEDURE AAREP_OPERKROFF_EMPTY_LIST
RETURNS (
    C_NAME_FIO VARCHAR(100),
    OL_ID DECIMAL(18,0),
    OL_DATE DATE,
    OL_FROM_REG_ID INTEGER,
    OB_SUMMA DECIMAL(18,4),
    OB_ACCOUNT_DT VARCHAR(20),
    OB_ACCOUNT_KT VARCHAR(20),
    C_ID DECIMAL(18,0),
    OL_DOC_CLIENT_NAME VARCHAR(100),
    OL_MEMO VARCHAR(1024),
    OB_DETAIL VARCHAR(1024),
    PAYED_AMOUNT DECIMAL(18,4),
    K_ID DECIMAL(18,0))
AS
declare variable oko_id integer;
declare variable is_delete integer;
begin
  for select ob_from_ol_id, ob_summa, ob_account_dt, ob_account_kt, ob_detail
      from operbook, acc_list
      where ob_account_kt = al_id and
            al_from_bg_id = '61099' and
            al_number = 3
      into :ol_id, :ob_summa, :ob_account_dt, :ob_account_kt, :ob_detail
  do
  begin
    select is_delete, ol_date, ol_from_reg_id, ol_document_client, ol_doc_client_name, ol_memo
    from operlist
    where ol_id = :ol_id
    into is_delete, ol_date, ol_from_reg_id, :c_id, ol_doc_client_name, ol_memo;
    if (is_delete = 0) then
    begin
      c_name_fio = null;
      if (c_id is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :c_id
        into :c_name_fio;
      end
      suspend;
    end
  end
end
