CREATE PROCEDURE HP_GET_CLIENT_TYPE(
    C_ID D_ID,
    CURR_CITIZEN_ID D_INTEGER,
    DOC_VNJ_ID D_INTEGER,
    REP_DATE D_DATE = null)
RETURNS (
    RES_CLIENT_TYPE D_INTEGER,
    RES_NON_RESIDENT D_INTEGER,
    RES_FROM_CDS_ID D_INTEGER,
    RES_COUNTRY_CODE D_STR_20,
    RES_COUNTRY_ID D_INTEGER,
    PDT_MEMO D_STR_255)
AS
declare variable db_id integer;
declare variable citizen_str d_str_code;
declare variable pk_id d_integer;
declare variable pdt_is_main_doc d_integer;
declare variable has_expired_citizen_passport d_smallint;
declare variable c_citizen d_integer;
begin
  -- новый вариант процедуры, разработан 19.09.2019 г.
  -- старый вариант переименован на HP_GET_CLIENT_TYPE_OLD
  -- определяет резидентсво контрагента и тип (ЮЛ, ФЛ или ИП)

  if (rep_date is null) then
    rep_date = current_date;
  res_from_cds_id = null;

  res_country_id = null;

  if (curr_citizen_id is null) then
  begin
    db_id = f_get_db_id();
    citizen_str = trim((case db_id
                          when 81001 then 'citizen.ru'
                          when 41701 then 'citizen.kg'
                          when 94401 then 'citizen.az'
                        end));

    select rd_id
    from refdata
    where rd_code = :citizen_str
    into :curr_citizen_id;
  end

  res_country_code = replace(citizen_str, 'citizen.', '');

  -- новый код проверки резидентства
  select iif(c_citizen = :curr_citizen_id, 0, 1),
         iif(coalesce(c_type, 0) = 1, iif(c_name_fio like 'ИП %' or c_name_fio like '% ИП', 2, 1), 3), c_citizen
  from clients
  where c_id = :c_id
  into :res_non_resident, res_client_type, c_citizen;
  if (res_non_resident is null) then
    res_non_resident = 0;

  if (res_client_type = 1) then
  begin
    select rd_code
    from refdata
    where rd_id = :c_citizen
    into res_country_code;
    res_country_code = replace(res_country_code, 'citizen.', '');

  end

  if (res_client_type = 3) then -- если физ.лицо, проверяем наличие документов дающие статус резидента
  begin
    res_non_resident = 1;
    has_expired_citizen_passport = 0;
    pdt_memo = null;
    -- статус резиденства дает либо местный паспорт либо ВНЖ
    for select pk_id, pdt_is_main_doc
        from passkind, passdoctype
        where pdt_id = pk_from_pdt_id and
              pk_is_delete = 0 and
              pk_country_id = :curr_citizen_id and
              -- (pdt_code = 'registration_vnj' or pdt_is_main_doc = 1) старое условие, до 18.10.2019
              pdt_code in ('registration_vnj', 'passport_local', 'passport_card') -- 18.10.2019 новое условие, отличается от старого тем что загранпаспорт исключаем
        order by pdt_is_main_doc desc
        into :pk_id, pdt_is_main_doc
    do
    begin
      select cds_id ,pk_name
      from clientdocs
       inner join passkind
          on pk_id = cds_from_pk_id
      where cds_from_c_id = :c_id and
            cds_from_pk_id = :pk_id and
            cds_doc_date_begin <= :rep_date and
            coalesce(cds_doc_date_end, :rep_date) >= :rep_date and -- 01.06.2020 Sanjari
            -- cds_is_delete = 0 and вместо этого добавляется cds_delete_datetime, 30.03.2020 Sanjari
            cds_is_main_doc = :pdt_is_main_doc and
            coalesce(cds_create_datetime, cds_datetime) < (:rep_date + 1) and
            (cds_delete_datetime is null or
            cds_delete_datetime > :rep_date) -- 30.03.2020 Sanjari
      order by cds_doc_date_begin desc, cds_datetime desc
      rows 1
      into res_from_cds_id,PDT_MEMO;
      if (res_from_cds_id is not null) then
      begin
        res_country_id = curr_citizen_id;
        res_non_resident = 0;
        select rd_code
        from refdata
        where rd_id = :res_country_id
        into res_country_code;

        res_country_code = replace(res_country_code, 'citizen.', '');
        suspend;
        exit;
      end

      -- 01.06.2020 Sanjari
      --      if (pdt_is_main_doc = 1 and has_expired_citizen_passport = 0) then
      --      begin
      --        if (exists(select *
      --        from clientdocs
      --        where cds_from_c_id = :c_id and
      --              cds_from_pk_id = :pk_id and
      --              cds_doc_date_begin <= :rep_date and
      --              coalesce(cds_doc_date_end, :rep_date) < :rep_date and
      --              cds_is_main_doc = 1 and
      --              coalesce(cds_create_datetime, cds_datetime) < (:rep_date + 1) and
      --              (cds_delete_datetime is null or cds_delete_datetime > :rep_date)))
      --        then has_expired_citizen_passport = 1; -- есть просроченный паспорт гражданина
      --      end
    end

    select cds_id, pk_country_id , pk_name
    from clientdocs, passkind
    where cds_from_c_id = :c_id and
          cds_doc_date_begin <= :rep_date and
          coalesce(cds_doc_date_end, :rep_date) >= :rep_date and -- 01.06.2020 Sanjari
          -- cds_is_delete = 0 and вместо этого добавляется cds_delete_datetime, 30.03.2020 Sanjari
          cds_is_main_doc = 1 and
          cds_from_pk_id = pk_id and
          pk_country_id <> :curr_citizen_id and
          coalesce(cds_create_datetime, cds_datetime) < (:rep_date + 1) and
          (cds_delete_datetime is null or
          cds_delete_datetime > :rep_date) -- 30.03.2020 Sanjari
    order by cds_doc_date_begin desc
    rows 1
    into res_from_cds_id, res_country_id, :pdt_memo;

    if (res_country_id is null) then
    begin
      select cds_id, pk_country_id, pk_name
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_doc_date_begin <= :rep_date and
            cds_is_main_doc = 1 and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = pdt_id and
            pdt_code in ('passport_local', 'passport_card') and
            coalesce(cds_create_datetime, cds_datetime) < (:rep_date + 1) and
            (cds_delete_datetime is null or
            cds_delete_datetime > :rep_date) -- 30.03.2020 Sanjari
      order by cds_create_datetime desc, cds_doc_date_begin desc
      rows 1
      into res_from_cds_id, res_country_id, :pdt_memo;
      if (res_country_id is not null) then
      begin
        if (res_country_id = :curr_citizen_id) then
          res_non_resident = 0;
        else
          res_non_resident = 1;
      end
      else
      begin
        res_country_id = c_citizen;
        res_non_resident = 1;
        if ((c_citizen = :curr_citizen_id) and
            (current_date = :rep_date)) then
          res_non_resident = 0;
      end
    end
    else
      res_non_resident = 1;

    select rd_code
    from refdata
    where rd_id = :res_country_id
    into res_country_code;

    res_country_code = replace(res_country_code, 'citizen.', '');
  end

  suspend;
end
