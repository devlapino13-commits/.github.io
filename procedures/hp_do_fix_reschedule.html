CREATE PROCEDURE HP_DO_FIX_RESCHEDULE(
    K_ID D_ID,
    RECALC_DATE D_DATE,
    RECALC_START_DATE D_DATE,
    NEED_EDOC D_BOOL)
AS
declare variable ol_id d_id;
declare variable reg_id d_integer;
declare variable ol_memo d_str_long;
declare variable un_id d_id;
declare variable kd_percent_effect d_percent;
declare variable curr_sh_date d_date;
declare variable curr_sh_id d_integer;
declare variable sh_date d_date;
declare variable sh_id d_id;
declare variable sum_os d_currency;
declare variable sum_perc d_currency;
declare variable fix_delinq d_bool = 0;
declare variable oks_delinq_date d_date;
declare variable ok_sh_os d_currency;
declare variable ok_sh_perc d_currency;
declare variable oks_balance d_currency;
declare variable oks_close_state d_currency;
declare variable sh_count d_integer;
begin
  -- Процедура исправляет график создав новую операцию
  -- Является аналогом принятия нулевого платежа

  un_id = 1000000000000;
  if (need_edoc is null) then
    need_edoc = 0;

  if (recalc_start_date > recalc_date) then
    exception e_message 'Incorrect_date';

  select oks_region_id, oks_delinq_date, oks_balance, oks_close_state, k_kontract_no || ' от ' || f_datetostr(k_disb_date) || 'г. '
  from oksummary, kontract
  where oks_from_k_id = :k_id and
        oks_from_k_id = k_id
  order by oks_from_k_id desc, oks_date_begin desc
  rows 1
  into reg_id, curr_sh_date, oks_balance, oks_close_state, ol_memo;

  if (coalesce(oks_balance, 0) <= 500) then
    exit;

  if (coalesce(oks_close_state, 1) > 0) then
    exit;

  if (reg_id is null) then
    exception e_default 'reg_id not found';


  sh_count = 0;
  for select ok_sh_date, ok_sh_id
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type <= 7 and
        ok_ol_date <= :recalc_date
  group by ok_sh_date, ok_sh_id
  having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
  order by ok_sh_date, ok_sh_id
  into sh_date, sh_id
  do
  begin
    if (sh_date = curr_sh_date and curr_sh_id is null) then
      curr_sh_id = sh_id;

    sh_count = sh_count + 1;
  end

  if (sh_count <= 1) then
    exit;

  if (curr_sh_id is null) then
    exception e_message 'current sh_id not found';

  ol_memo = 'Корректировка графика погашений по договору ' || ol_memo;

  insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                        ol_memo, ol_parent_id, is_delete, ol_user_id)
  values (:reg_id, 1, 4, :recalc_date, 9, :ol_memo, null, 0, :un_id)
  returning ol_id
  into :ol_id;

  fix_delinq = 0;

  select ok_sh_date
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type <= 7 and
        ok_ol_date <= :recalc_date and
        not(ok_ol_date between :recalc_start_date and :recalc_date and ok_type = 3)
  group by ok_sh_date
  having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
  order by ok_sh_date
  rows 1
  into oks_delinq_date;

  if (oks_delinq_date < curr_sh_date) then
    fix_delinq = 1;

  for select ok_sh_id, ok_sh_date, sum(ok_sh_os), sum(ok_sh_perc)
  from operkred
  where ok_from_k_id = :k_id and
        ok_type = 3 and
        ok_ol_is_active = 1 and
        ok_ol_date between :recalc_start_date and :recalc_date and
        (ok_sh_os <> 0 or ok_sh_perc <> 0)
  group by ok_sh_id, ok_sh_date
  into sh_id, sh_date, ok_sh_os, ok_sh_perc
  do
  begin
    if (ok_sh_os is null) then
      ok_sh_os = 0;

    if (ok_sh_perc is null) then
      ok_sh_perc = 0;

    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                          ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc)
    values (:ol_id, -1, :k_id, :sh_id, :sh_date, 3, 0, 0, 0, 3, 0, -:ok_sh_os, -:ok_sh_perc);
  end

  if (fix_delinq = 1) then
  begin
    for select ok_sh_date, ok_sh_id, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type <= 7 and
          ok_ol_date <= :recalc_date and
          ok_sh_date < :curr_sh_date
    group by ok_sh_date, ok_sh_id
    having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
    into sh_date, sh_id, sum_os, sum_perc
    do
    begin
      if (sum_os is null) then
        sum_os = 0;

      if (sum_perc is null) then
        sum_perc = 0;

      -- Переносим просроченные платежи на ближайший
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                            ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc)
      values (:ol_id, -1, :k_id, :sh_id, :sh_date, 3, 0, 0, 0, 3, 0, -:sum_os, -:sum_perc);

      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                            ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc)
      values (:ol_id, -1, :k_id, :curr_sh_id, :curr_sh_date, 3, 0, 0, 0, 3, 0, :sum_os, :sum_perc);
    end
  end

  execute procedure hp_do_repayment_recalc_sh(:ol_id, :k_id);

  if (need_edoc = 1) then
  begin
    select kps_psk
    from hp_edoc_ins_kontpsk(:k_id, :ol_id, :un_id)
    into :kd_percent_effect;
  end
end
