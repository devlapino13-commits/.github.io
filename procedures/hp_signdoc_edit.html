CREATE PROCEDURE HP_SIGNDOC_EDIT(
    ID D_INTEGER)
RETURNS (
    SGN_ID D_INTEGER,
    SGN_NAME D_STR_255,
    SGN_SGG_ID D_INTEGER,
    SGN_BLS_ID D_ID,
    SGN_EDL_ID D_ID,
    SGN_STATUS D_SMALLINT,
    STATUS_STR D_STR_CODE,
    SGN_UN_ID D_ID,
    UN_FIO D_STR_100,
    SGN_DATETIME D_DATETIME)
AS
begin
  for
    select
      sgn_id,
      sgn_name,
      sgn_sgg_id,
      sgn_bls_id,
      sgn_edl_id,
      sgn_status,
      case sgn_status
        when -1 then 'запрос аннулирован'
        when  0 then 'запрос создан'
        when  1 then 'ЭД создан'
        when  3 then 'ЭД подписан'
      end,
      sgn_un_id,
      un_fio,
      sgn_datetime
    from signdoc
      inner join username
        on un_id = sgn_un_id
    where sgn_id = :id
  into
    :sgn_id,
    :sgn_name,
    :sgn_sgg_id,
    :sgn_bls_id,
    :sgn_edl_id,
    :sgn_status,
    :status_str,
    :sgn_un_id,
    :un_fio,
    :sgn_datetime
  do
  begin
    suspend;
  end
end
