CREATE PROCEDURE CHECK_MINIAPP_ANNUL(
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_ID,
    UN_ID D_ID)
RETURNS (
    ERRMSG D_STR_255)
AS
declare variable ksr_id d_integer;
declare variable k_status d_integer;
declare variable k_id d_id;
declare variable db_id d_integer;
declare variable is_manager d_smallint;
declare variable rd_id d_integer;
begin
  errmsg = '';
  db_id = f_get_db_id();
  if (object_kind = 'miniapp') then
  begin
    select k_id, k_status
    from kontract
    where k_from_ma_app_id = :object_id rows 1
    into k_id, k_status;
  end
  else -- if (object_kind = 'kontract') then
  begin
    select k_id, k_status
    from kontract
    where k_id = :object_id
    into k_id, k_status;
  end

  if (k_status >= 0 and
      exists(select kl_datetime
             from kontlog
             where kl_from_k_id = :k_id and
                   kl_state = 1)) then
  begin
    is_manager = 0; -- Пока нужно только в РФ
    if (db_id = 81001 and
        exists(select *
               from hr_pstatus, joblist, jobkind, username
               where un_id = :un_id and
                     ps_from_c_id = un_from_c_id and
                     ps_date_begin <= current_date and
                     ps_date_end >= current_date and
                     ps_is_working = 1 and
                     ps_status = 1 and
                     ps_from_jl_id = jl_id and
                     jl_kind = jk_id and
                     jk_system_code in ('JobSystem.LO_Senior', 'JobSystem.Region_KM', 'JobSystem.Region_Director'))) then
      is_manager = 1;

    if (is_manager = 1) then
      errmsg = 'Сначала подпишите у клиента Заявление на отказ от займа. Размещено в верхнем левом углу экрана';
    else
      errmsg = 'Заявка уже была переведена в договор. Верните договор в статус заявки или откажите по договору.';
    suspend;
    exit;
  end

  select rd_id
  from hp_get_check_loan_data(:k_id)
  into :rd_id;

  if (k_id is not null and
      exists(select *
             from actionlist al
             inner join refdata rd on al.acl_from_rd_id = rd.rd_id
             where rd.rd_id = :rd_id and
                   al.acl_object_kind = 9 and
                   al.acl_object_id = :k_id)) then
  begin
    select ksr_id
    from aarep_kontsign_access(:k_id, :un_id, 0, '')
    into ksr_id;
    if (ksr_id = 0) then
      errmsg = 'Кредит уже был отправлен на КК, отказать может сотрудник, имеющий право утверждать договора';
  end

  if (errmsg = '' and
      db_id = 81001 and
      k_id is not null) then
  begin
    -- договор нельзя отказ делать после утверждения не прошли 5 рабочих дней
    select res_text
    from hp_cancel_loan_status(:k_id, :un_id, 1)
    into errmsg;
  end
  suspend;
end
