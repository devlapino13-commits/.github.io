CREATE PROCEDURE AAREP_VAC_REQ_STAFF(
    REPORT_REGION_ID INTEGER,
    REPORT_YEAR INTEGER)
RETURNS (
    C_ID TYPE OF COLUMN CLIENTS.C_ID,
    C_NAME_FIO TYPE OF COLUMN CLIENTS.C_NAME_FIO,
    JL_NAME TYPE OF COLUMN JOBLIST.JL_NAME,
    REG_NAME TYPE OF COLUMN REGION.REG_NAME,
    DAYS_NORMA DECIMAL(18,4),
    DAYS_USED DECIMAL(18,4),
    DAYS_BALANCE DECIMAL(18,4),
    REP_YEAR INTEGER,
    C_HIRED_DATE D_DATE)
AS
declare variable year_begin date;
declare variable year_end date;
declare variable i integer;
declare variable days_tmp decimal(18,4);
declare variable ps_from_reg_id integer;
declare variable c_id_tmp id_type;
declare variable ps_from_jl_id integer;
begin
  c_id_tmp = -1;
  rep_year = report_year;
  year_begin = cast('01.01.' || :report_year as date);
  year_end = cast('31.12.' || :report_year as date);
  for select ps.ps_from_c_id, ps.ps_from_reg_id, ps.ps_from_jl_id
      from hr_pstatus ps
      inner join region_list(:report_region_id, 1, -1, 1) rl on ps.ps_from_reg_id = rl.id_list
      where ps.ps_date_begin <= :year_end and
            ps.ps_date_end >= :year_begin and
            ps.ps_status = 1 and
            ps.ps_is_working = 1
      order by ps.ps_from_c_id, ps.ps_date_begin desc
      into :c_id, :ps_from_reg_id, :ps_from_jl_id
  do
  begin
    if (c_id <> c_id_tmp) then
    begin
      c_id_tmp = :c_id;
      days_norma = 0;
      days_used = 0;
      days_balance = 0;

      select c_name_fio
      from clients
      where c_id = :c_id
      into :c_name_fio;

      select ps.ps_date_begin
      from hr_pstatus ps
      inner join hr_orders ho on ps.ps_from_ho_id = ho.ho_id and
            ho.ho_order_type = 1 and
            ho.ho_status = 1
      where ps.ps_from_c_id = :c_id and
            extract(year from ps.ps_date_begin) <= :report_year
      order by 1 desc
      rows 1
      into :c_hired_date;

      select jl_name
      from joblist
      where jl_id = :ps_from_jl_id
      into :jl_name;

      select reg_name
      from region
      where reg_unicode = :ps_from_reg_id
      into :reg_name;

      select sum(vr.vr_days)
      from vacationrequest vr
      where vr_from_c_id = :c_id and
            vr_date_begin >= :year_begin and
            vr_date_end <= :year_end
      into :days_used;
      days_used = coalesce(days_used, 0);
      days_tmp = 0;
      i = 1;
      while (i <= 12) do
      begin
        for select sat_vocation_curr
            from aarep_sa_tabel_one_voc(:ps_from_reg_id, :c_id, :report_year, :i)
            into :days_tmp
        do
        begin
          days_norma = days_norma + days_tmp;
        end
        i = i + 1;
      end

      days_norma = round(days_norma);
      days_balance = days_norma - days_used;
      suspend;
    end
  end
end
