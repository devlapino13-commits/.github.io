CREATE PROCEDURE AAREP_KONTRACT_ACTIVE_CURR_TEMP(
    REG_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    AC_K_ID DECIMAL(18,0),
    AC_K_BALANCE DECIMAL(18,4),
    AC_LOAN_TYPE SMALLINT,
    AC_KF_ID DECIMAL(18,0),
    AC_REGION_ID INTEGER,
    AC_CLIENT_KOLICH INTEGER,
    AC_DISB DECIMAL(18,4),
    K_CURR_RATE DECIMAL(18,4),
    CURR_CODE VARCHAR(3),
    AC_NO VARCHAR(20),
    CR_RATE DECIMAL(18,4),
    AMOUNT_DIFF DECIMAL(18,4))
AS
declare variable report_currency varchar(3);
declare variable pr_id_prior integer;
declare variable pr_id integer;
begin
  select cm_default_currency
  from company
  into :report_currency;

  pr_id_prior = 0;
  for select cr_from_cu_id, cr_rate
      from opercurr
      where cr_date <= :rep_date and
            cr_date_end >= :rep_date
      into :curr_code, :cr_rate
  do
  begin
    if (curr_code <> :report_currency) then
    begin
      for select k_amount_approved, k_id, k_from_pr_id, k_from_kf_id, k_kolich_client_approved, k_index_curr_rate
          from kontract
          where k_index_curr_code = :curr_code and
                k_index_curr_rate between(:cr_rate + 0.00001) and 99999999
          into :ac_disb, :ac_k_id, :pr_id, :ac_kf_id, :ac_client_kolich, :k_curr_rate
      do
      begin
        if (pr_id_prior <> pr_id) then
        begin
          ac_loan_type = -1;
          select pr_loan_type
          from product
          where pr_id = :pr_id and
                pr_set_rate_in_kont = 1
          into :ac_loan_type;
          if (ac_loan_type is null) then
            ac_loan_type = -1;
          pr_id_prior = :pr_id;
        end
        if (ac_loan_type <> -1) then
        begin
          for select oks_region_id, oks_balance
              from oksummary
              where oks_date_begin <= :rep_date and
                    oks_date_end >= :rep_date and
                    oks_from_k_id = :ac_k_id and
                    oks_close_state = 0
              into :ac_region_id, :ac_k_balance
          do
          begin
            if (exists(select *
                       from region_list(:ac_region_id, 1, -1, 0)
                       where id_list = :reg_id)) then
            begin
              amount_diff = cast(:ac_k_balance * (:k_curr_rate - :cr_rate) as decimal(15,2));
              if (amount_diff <> 0) then
              begin
                ac_no = '';
                select ac_no
                from kontr_acc_list(:ac_k_id, :rep_date)
                where ac_type = 1 rows 1
                into :ac_no;

                suspend;
              end
            end
          end
        end
      end
    end
  end
end
