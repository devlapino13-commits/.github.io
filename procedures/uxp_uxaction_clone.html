CREATE PROCEDURE UXP$UXACTION_CLONE(
    OWNER_UUID D_UUID,
    NEW_OWNER_UUID D_UUID)
AS
declare variable ua_uuid d_uuid;
declare variable new_ufp_uuid d_uuid;
begin
  for select ua_uuid, ua_owner, ua_uxa_uuid, ua_uxa_fcode, ua_name, ua_hint, ua_disable_on_edit,
             ua_readonly, ua_inactive,
             ua_menu_index, ua_usr_id_field, ua_usr_text_field,
             ua_img_uuid, ua_ext_component, ua_ext_dataset, ua_ext_action, ua_ext_replace_kind,
             ua_hotkey, ua_add_delimiter, ua_auto_refresh_seconds, ua_is_group, ua_alias_group,
             ua_trigger_kind, ua_preblank_kind, ua_preblank_code, ua_preblank_save
  from ux$action
  where ua_owner = :owner_uuid
  as cursor cur
  do
  begin
    update or insert into ux$action (ua_owner, ua_uxa_uuid, ua_uxa_fcode, ua_name, ua_hint, ua_disable_on_edit, ua_readonly,
                                     ua_inactive, ua_menu_index, ua_usr_id_field, ua_usr_text_field,
                                     ua_img_uuid, ua_ext_component, ua_ext_dataset, ua_ext_action, ua_ext_replace_kind,
                                     ua_hotkey, ua_add_delimiter, ua_auto_refresh_seconds, ua_is_group, ua_alias_group,
                                     ua_trigger_kind, ua_preblank_kind, ua_preblank_code, ua_preblank_save)
    values (:new_owner_uuid, cur.ua_uxa_uuid, cur.ua_uxa_fcode, cur.ua_name, cur.ua_hint, cur.ua_disable_on_edit, cur.ua_readonly,
            cur.ua_inactive, cur.ua_menu_index, cur.ua_usr_id_field, cur.ua_usr_text_field,
            cur.ua_img_uuid, cur.ua_ext_component, cur.ua_ext_dataset, cur.ua_ext_action, cur.ua_ext_replace_kind,
            cur.ua_hotkey, cur.ua_add_delimiter, cur.ua_auto_refresh_seconds, cur.ua_is_group, cur.ua_alias_group,
            cur.ua_trigger_kind, cur.ua_preblank_kind, cur.ua_preblank_code, cur.ua_preblank_save)
    matching (ua_owner, ua_uxa_uuid, ua_usr_id_field)
    returning (ua_uuid)
    into ua_uuid;

    execute procedure uxp$uxpairuuid_add(cur.ua_uuid, :ua_uuid);
    execute procedure uxp$uxexpression_clone(cur.ua_uuid, :ua_uuid);

    for select uap_ua_uuid, uap_param_name, uap_owner_ufp_uuid, uap_owner_ext_field,
               uap_default, uap_use_default, uap_locked, uap_exec_param_id
    from ux$actionparam
    where uap_ua_uuid = cur.ua_uuid
    as cursor act
    do
    begin
      new_ufp_uuid = null;

      if (act.uap_owner_ufp_uuid is not null) then
        new_ufp_uuid = uxf$pair(act.uap_owner_ufp_uuid);

      insert into ux$actionparam (uap_ua_uuid, uap_param_name, uap_owner_ufp_uuid, uap_owner_ext_field,
                                  uap_default, uap_use_default, uap_locked, uap_exec_param_id)
      values (:ua_uuid, act.uap_param_name, :new_ufp_uuid, act.uap_owner_ext_field,
              act.uap_default, act.uap_use_default, act.uap_locked, act.uap_exec_param_id);
    end
  end

end
