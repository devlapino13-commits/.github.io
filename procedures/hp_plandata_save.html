CREATE PROCEDURE HP_PLANDATA_SAVE(
    PP_ID INTEGER,
    PD_DATE DATE,
    PD_PERIOD_BEGIN INTEGER,
    PD_PERIOD_COUNT INTEGER,
    PD_FROM_RD_ID INTEGER,
    PD_FROM_REG_ID INTEGER,
    PD_FROM_C_ID DECIMAL(18,0),
    PD_FROM_PM_ID INTEGER,
    PD_DATA_1 DECIMAL(18,4),
    PD_DATA_2 DECIMAL(18,4),
    PD_DATA_3 DECIMAL(18,4),
    PD_DATA_4 DECIMAL(18,4),
    PD_DATA_5 DECIMAL(18,4),
    PD_DATA_6 DECIMAL(18,4),
    PD_DATA_7 DECIMAL(18,4),
    PD_DATA_8 DECIMAL(18,4),
    PD_DATA_9 DECIMAL(18,4),
    PD_DATA_10 DECIMAL(18,4),
    PD_DATA_11 DECIMAL(18,4),
    PD_DATA_12 DECIMAL(18,4))
AS
declare variable period_no integer;
declare variable mm integer;
declare variable yy integer;
declare variable po_amount decimal(18,4) = 0;
declare variable delta_amount decimal(18,4);
declare variable curr_data decimal(18,4);
declare variable curr_period integer;
declare variable pp_is_mgr d_bool;
begin
-- Процедура сохраняет плановые данные
  if (pd_from_c_id = 1) then
    pd_from_c_id = null;

  select pp_is_mgr
  from planpocket
  where pp_id = :pp_id
  into :pp_is_mgr;

  if (pp_is_mgr is null) then
    pp_is_mgr = 0;

  -- Если данные на уровне менеджеров, кроме C_ID другие строки не сохраняем
  if (pp_is_mgr = 1 and coalesce(pd_from_c_id, 0) = 0) then
    exit;

  if (pd_from_pm_id = -1) then
  begin
    if (pd_from_c_id is not null) then
    begin
      select pm_id
      from planmain
      where pm_from_rd_id = :pd_from_rd_id and
            pm_from_reg_id = :pd_from_reg_id and
            pm_from_c_id = :pd_from_c_id and
            pm_is_mgr = :pp_is_mgr
      into :pd_from_pm_id;
    end
    else
    begin
      select pm_id
      from planmain
      where pm_from_rd_id = :pd_from_rd_id and
            pm_from_reg_id = :pd_from_reg_id and
            pm_from_c_id is null
      into :pd_from_pm_id;
    end
    if (pd_from_pm_id is null or pd_from_pm_id = -1) then
    begin
      insert into planmain (pm_from_rd_id, pm_from_reg_id, pm_from_c_id, pm_date_begin, pm_date_end, pm_is_mgr)
      values (:pd_from_rd_id, :pd_from_reg_id, :pd_from_c_id, null, null, :pp_is_mgr)
      returning pm_id
      into :pd_from_pm_id;
    end
  end
  period_no = 0;
  mm = cast(right(pd_period_begin, 2) as integer);
  yy = cast(left(pd_period_begin, 4) as integer);

-- На каждый месяц выполняем цикл
  while (period_no < pd_period_count) do
  begin
    period_no = period_no + 1;
    if ((mm + period_no - 1) <= 12) then
      curr_period = yy * 100 + (mm + period_no - 1);
    else
      curr_period = (yy + 1) * 100 + (mm + (period_no - 1) - 12);

    curr_data = 0; -- новое значение для сохранения
    curr_data = (case
                   when (:period_no = 1) then :pd_data_1
                   when (:period_no = 2) then :pd_data_2
                   when (:period_no = 3) then :pd_data_3
                   when (:period_no = 4) then :pd_data_4
                   when (:period_no = 5) then :pd_data_5
                   when (:period_no = 6) then :pd_data_6
                   when (:period_no = 7) then :pd_data_7
                   when (:period_no = 8) then :pd_data_8
                   when (:period_no = 9) then :pd_data_9
                   when (:period_no = 10) then :pd_data_10
                   when (:period_no = 11) then :pd_data_11
                   when (:period_no = 12) then :pd_data_12
                 end);
    if (curr_data is null) then
      curr_data = 0;

-- получаем текущее значение в БД
    po_amount = 0;
    select sum(po_calc - po_minus)
    from planoper
    where po_from_pm_id = :pd_from_pm_id and
          po_period = :curr_period and
          po_date <= :pd_date
    into :po_amount;
    if (po_amount is null) then
      po_amount = 0;

    delta_amount = curr_data - po_amount;
    if (delta_amount <> 0) then
    begin
      insert into planoper (po_from_pp_id, po_id, po_from_pm_id, po_date, po_period, po_calc, po_minus)
      values (:pp_id, -1, :pd_from_pm_id, :pd_date, :curr_period, :delta_amount, 0);
    end
  end
end
