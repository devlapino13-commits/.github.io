CREATE PROCEDURE HP_SUD_SEND_RECEIPT_REQ(
    K_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_LONG)
AS
declare variable c_id d_id;
declare variable eil_id d_id;
begin
  -- процедура создает запрос на получение платежной квитанции от Монеты

  if (f_get_db_id() = 81001) then
  begin
    -- проверка на наличие номера квитанции
    if (exists(select esm_id
              from extsysmain em, extsysanswer ea
              where ea.esa_from_esm_id = esm_id
                and em.esm_from_k_id = :k_id
                and em.esm_project = 'moneta_disb'
                and em.esm_action = 'CardDisb'
                and esm_status = 1
                and ea.esa_message starting with 'answer:'
                and f_json_blob_get_value(replace(ea.esa_message, 'answer:', ''), 'attrs.statusid', '') = 'SUCCEED'
    )) then
    begin
      c_id = (select kc_from_c_id from knclient where kc_from_k_id = :k_id rows 1);

      select eil_id
      from hp_sud_extinfo_insert('moneta_doc', 'kontract', :k_id, :c_id)
      into eil_id;
    end
    else
      error_text = 'esm_id not found';
  end
  suspend;
end
