begin

  for select reg_unicode, reg_code, reg_name
      from region_list(:rep_region_id, 1, 3, 1), region
      where id_list = reg_unicode and

    rows 1

      reg_cash_limit = 0;

    from cashcheckmain

          ccm_from_reg_id = :reg_id and
          ccm_order_kind <= 2

    from cashcheckmain

          ccm_from_reg_id = :reg_id
    into :reg_audit_all_count;
    if (reg_audit_all_count is null) then
      reg_audit_all_count = 0;

    select count(*)
    from cashcheckmain
    where ccm_date between(:rep_date_end - 35) and :rep_date_end and
          ccm_from_reg_id = :reg_id and
          abs(ccm_cash_saldo_mannual - ccm_cash_saldo_fact) >= 100
    into :reg_audit_err_count;
    if (reg_audit_err_count is null) then
      reg_audit_err_count = 0;

    select max(ccm_date)
    from cashcheckmain
    where ccm_date <= :rep_date_end and
          ccm_from_reg_id = :reg_id and

    into :last_dd;
    if (last_dd is not null) then
