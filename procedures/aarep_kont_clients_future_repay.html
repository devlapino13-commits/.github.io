CREATE PROCEDURE AAREP_KONT_CLIENTS_FUTURE_REPAY(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KF_BRANCH_NAME D_STR_100,
    KF_REGION_NAME D_STR_100,
    KF_OFFICER_NAME D_STR_100,
    PR_NAME D_STR_100,
    KF_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_100,
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    REP_PLAN_REPAY_IS_LAST VARCHAR(10),
    K_PERIOD DECIMAL(15,2),
    REP_PLAN_REPAY_DATE DATE,
    REP_PLAN_OS_LOC DECIMAL(18,4),
    REP_PLAN_PERC_LOC DECIMAL(18,4),
    REP_PLAN_ALL_LOC DECIMAL(18,4),
    REP_ACTIVE_BALANCE_LOC DECIMAL(18,4),
    KF_DATE_IN_TO_KRS DATE,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_KOLICH_REPAY INTEGER,
    K_PERCENT_REAL DECIMAL(18,7),
    K_PERCENT DECIMAL(18,7),
    K_PERCENT_INPUT DECIMAL(18,7),
    REP_DATE_ACTION DATE,
    K_KREDLINE_END DATE,
    C_INN D_STR_20)
AS
declare variable k_id d_id;
declare variable dolya decimal(4,2);
declare variable k_amount decimal(18,4);
declare variable kc_amount decimal(18,4);
declare variable kc_amount_fact decimal(18,4);
declare variable k decimal(10,8);
declare variable rep_plan_os decimal(18,4);
declare variable rep_plan_perc decimal(18,4);
declare variable rep_plan_all decimal(18,4);
declare variable rep_active_balance decimal(18,4);
begin
  for select k_id, kf_branch_name, kf_region_name, kf_officer_name, pr_name, kf_name, k_kontract_no, k_disb_date,
             k_amount_approved, rep_plan_repay_is_last, k_period, rep_plan_repay_date, rep_plan_os, rep_plan_perc,
             rep_plan_all, rep_active_balance, kf_date_in_to_krs, k_kolich_client_approved, k_kolich_repay,
             k_percent_real, k_percent, k_percent_input, rep_date_action, k_kredline_end, k_period
      from aarep_kontract_list(:REP_REGION_ID, :REP_DATE_BEGIN, :REP_DATE_END, 0, '000', 10)
      into k_id, kf_branch_name, kf_region_name, kf_officer_name, pr_name, kf_name, k_kontract_no, k_disb_date,
           k_amount_approved, rep_plan_repay_is_last, k_period, rep_plan_repay_date, rep_plan_os, rep_plan_perc,
           rep_plan_all, rep_active_balance, kf_date_in_to_krs, k_kolich_client_approved, k_kolich_repay,
           k_percent_real, k_percent, k_percent_input, rep_date_action, k_kredline_end, k_period
  do
  begin
    for select cl.c_name_fio, k.k_amount_approved, knc.kc_amount_approved, cl.c_inn
        from knclient knc
        inner join kontract k on k.k_id = knc.kc_from_k_id
        inner join clients cl on cl.c_id = knc.kc_from_c_id
        where knc.kc_from_k_id = :k_id and
              knc.kc_amount_approved > 0
        into :c_name_fio, k_amount, kc_amount, c_inn
    do
    begin
      k_amount_approved = kc_amount;
      rep_plan_os_loc = 0;
      rep_plan_perc_loc = 0;
      rep_plan_all_loc = 0;
      rep_active_balance_loc = 0;
      kc_amount_fact = 0;
      k = 1;
      select sum(kc_amount_approved)
      from hp_client_short_list(:k_id)
      into :kc_amount_fact;

      if (exists(select *
                 from hp_client_short_list(:k_id)
                 where client_name = :c_name_fio)) then


      begin
        k = kc_amount_fact / kc_amount;

        rep_plan_os_loc = round(rep_plan_os / k, 0);
        rep_plan_perc_loc = round(rep_plan_perc / k, 0);
        rep_plan_all_loc = rep_plan_perc_loc + rep_plan_os_loc;
        rep_active_balance_loc = round(rep_active_balance / k, 0);
      end
      suspend;
    end
  end
end
