CREATE PROCEDURE HP_IMESSAGE_SIGN(
    UN_ID D_ID,
    IM_LST D_STR_8100,
    IM_TYPE D_INTEGER = 0)
RETURNS (
    RES_TXT D_STR_255)
AS
declare variable dt_sign d_datetime;
declare variable asign_resp d_integer;
declare variable asign_auth d_integer;
declare variable im_id d_id;
declare variable num d_integer;
declare variable posb d_integer;
declare variable pose d_integer;
begin
  res_txt = '';
  dt_sign = current_timestamp;

  -- право подписи ответственным лицом
  asign_resp =
    iif(exists(
        select *
        from permissions
        where pm_object_kind = 0
          and pm_object_id = :un_id
          and pm_subject_kind = 3
          and pm_subject_id = f_get_pa_id('FinStart.IM_EarlyRepay_Responsible')
          and pm_is_delete = 0
          and current_date between coalesce(pm_date_begin, '01.01.1900')
            and coalesce(pm_date_end,'31.12.9999')
          and pm_access_kind = 1), 1, 0);
  -- право подписи уполномоченным лицом
  asign_auth =
    iif(exists(
        select *
        from permissions
        where pm_object_kind = 0
          and pm_object_id = :un_id
          and pm_subject_kind = 3
          and pm_subject_id = f_get_pa_id('FinStart.IM_EarlyRepay_Authorized')
          and pm_is_delete = 0
          and current_date between coalesce(pm_date_begin, '01.01.1900')
            and coalesce(pm_date_end, '31.12.9999')
          and pm_access_kind = 1), 1, 0);

  num = 0;
  posb = 1;
  while (1=1) do begin
    pose = position(';', im_lst, posb);
    if (pose > posb) then begin
      im_id = substring(im_lst from posb for pose - posb);

      if (exists(select * from internal_message where im_id = :im_id and im_status = 0)
        or (asign_resp = 0 and asign_auth = 0)) then
        update internal_message set
          im_cashier_sign_datetime = :dt_sign,
          im_status = 1
        where im_id = :im_id
          and im_cashier_id = :un_id
          and im_status = 0;
      else
      if (asign_resp = 1 and asign_auth = 0) then
        update internal_message set
          im_controller_1_id = :un_id,
          im_controller_1_sign_datetime = :dt_sign,
          im_status = 2
        where im_id = :im_id
          and im_controller_1_id is null
          and im_status = 1;
      else
      if (asign_resp = 0 and asign_auth = 1) then
        update internal_message set
          im_controller_2_id = :un_id,
          im_controller_2_sign_datetime = :dt_sign,
          im_status = 3
        where im_id = :im_id
          and im_controller_2_id is null
          and im_status = 2;
      else
      if (asign_resp = 1 and asign_auth = 1) then
        update internal_message set
          im_controller_1_id = :un_id,
          im_controller_1_sign_datetime = :dt_sign,
          im_controller_2_id = :un_id,
          im_controller_2_sign_datetime = :dt_sign,
          im_status = 3
        where im_id = :im_id
          and im_controller_1_id is null
          and im_status = 1;
      num = num + row_count;
      posb = pose + 1;
    end else leave;
  end
  res_txt = 'Успешно подписано - ' || num || ' ВС.';
  suspend;
end
