CREATE PROCEDURE HP_ACTIONLIST_KK_GET(
    IN_ACL_ID D_ID,
    IN_STATUS D_INTEGER,
    IN_IS_OWN D_INTEGER,
    IN_RD_CODE D_STR_255,
    UN_ID D_ID)
RETURNS (
    REC_ID DECIMAL(18,0),
    REC_PARENT_ID DECIMAL(18,0),
    ACL_ID DECIMAL(18,0),
    ACL_PARENT_ID DECIMAL(18,0),
    ACL_DATE DATE,
    ACL_FROM_REG_ID INTEGER,
    ACL_OBJECT_KIND INTEGER,
    ACL_OBJECT_ID DECIMAL(18,0),
    ACL_FROM_RD_ID INTEGER,
    ACL_RESPONS_C_ID DECIMAL(18,0),
    RES_OBJECT_NAME VARCHAR(255),
    RES_RD_NAME VARCHAR(255),
    RES_RD_MEMO VARCHAR(255),
    RES_RD_ORDERNO INTEGER,
    RES_OPR_ID DECIMAL(18,0),
    RES_OBJECT_STATUS INTEGER,
    RES_OBJECT_DETAIL BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_OBJECT_MEMO BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_OBJECT_BLOB_IDS D_STR_2048,
    RDL_MAIN_RD_ID INTEGER,
    RES_HAS_BLOB INTEGER,
    RES_HAS_MEMO INTEGER,
    RES_OBJECT_BLOB_NAMES VARCHAR(8100),
    ACL_RESPONS_C_FIO VARCHAR(100),
    KONT_STAR_STATUS INTEGER,
    RES_HAS_BLOCK D_BOOL,
    RES_NEW_STATUS D_INTEGER,
    RES_NEW_COMMENT_TEXT D_STR_255,
    RES_NEW_COMMENT_CODE D_STR_255,
    RES_NEW_COMMENT_ID D_ID,
    RES_NEW_COMMENT_MEMO D_STR_255,
    MAIN_ACL_ID D_ID,
    RES_JSON_DOCS D_STR_8100,
    RES_HAS_JSON_DOCS D_BOOL,
    RES_BLS_DOCS D_STR_8100,
    RES_HAS_BLS_DOCS D_BOOL)
AS
declare variable json_item d_str_8100;
declare variable res_str d_str_255;
declare variable blob_id_index d_integer;
declare variable blob_name_index d_integer;
declare variable blob_caption d_str_255;
declare variable v_delimiter d_str_code;
begin
-- Основая процедура для отображения окна кредитного комитета (андеррайтинг)

-- В качестве параметра необходимо передавать ACL_ID из ActionList либо RD_Code из RefData: ActionGroup
-- Данная процедура обращается в процедуры hp_actionlist_rd_status - hp_actionlist_get_1.
-- процедура hp_actionlist_get_1 получает в качестве параметра поле RD_Str_ID из строки справочника в RefData

  if (coalesce(in_acl_id, 0) = 0) then
  begin
    if (in_rd_code <> 'ActionGroup.Check_Loan_Data' or  in_rd_code <> 'ActionGroup.Check_Loan_Data_Special') then
      exception e_message 'Задача пока не поддерживается в данном модуле';

    select acl_id
    from aarep_actionlist_get_next(:un_id, :in_acl_id, :in_status, :in_is_own, :in_rd_code)
    into :in_acl_id;
  end

  if (coalesce(in_acl_id, 0) = 0) then
    exception e_message 'Нет задач в очереди';

  main_acl_id = in_acl_id;

  res_new_comment_text = '';
  res_new_comment_code = '';
  res_new_comment_memo = '';
  res_new_comment_id = null;

  for select rec_id, rec_parent_id, acl_id, acl_parent_id, acl_date, acl_from_reg_id, acl_object_kind, acl_object_id,
             acl_from_rd_id, acl_respons_c_id, res_object_name, res_rd_name, res_rd_memo, res_rd_orderno, res_opr_id,
             res_object_status, res_object_detail, res_object_memo, res_object_blob_ids, rdl_main_rd_id, res_has_blob,
             res_has_memo, res_object_blob_names, acl_respons_c_fio, kont_star_status, res_has_block
  from aarep_actionlist_full(:in_acl_id, '', 0, '', 0, 'yes')
  into :rec_id, :rec_parent_id, :acl_id, :acl_parent_id, :acl_date, :acl_from_reg_id, :acl_object_kind,
       :acl_object_id, :acl_from_rd_id, :acl_respons_c_id, :res_object_name, :res_rd_name, :res_rd_memo,
       :res_rd_orderno, :res_opr_id, :res_object_status, :res_object_detail, :res_object_memo, :res_object_blob_ids,
       :rdl_main_rd_id, :res_has_blob, :res_has_memo, :res_object_blob_names, :acl_respons_c_fio, :kont_star_status, :res_has_block
  do
  begin
    res_new_status = res_object_status;

    res_json_docs = '';
    res_has_json_docs = 0;
    if (acl_object_kind = 9) then
    begin
      select left(res_data, 8100)
      from hp_json_kontract_docs(:acl_object_id, :un_id)
      into :res_json_docs;

      if (res_json_docs = '[') then
        res_json_docs = '';

      if (res_json_docs <> '') then
        res_has_json_docs = 1;
    end

    res_bls_docs = '';
    res_has_bls_docs = 0;

    if (res_object_blob_ids <> '') then
    begin
      blob_id_index = 0;
      v_delimiter = '';
  
      for select res_str
      from hp_split_str(:res_object_blob_ids, ';')
      where res_str <> '0'
      into :res_str
      do
      begin
        blob_id_index = blob_id_index + 1;
  
        res_object_blob_names = replace(res_object_blob_names, '][', '];[');
        blob_caption = '';
        blob_name_index = 0;
        for select res_str
        from hp_split_str(:res_object_blob_names, ';')
        where res_str <> '0'
        into :blob_caption
        do
        begin
          blob_caption = replace(blob_caption, '[', '');
          blob_caption = replace(blob_caption, ']', '');

          blob_name_index = blob_name_index + 1;

          if (blob_name_index = blob_id_index) then
            break;
        end

        if (blob_name_index < blob_id_index) then
          blob_caption = 'Файл';
  
        json_item = f_image_item('blobstore', res_str, blob_caption, un_id, 0, 0, 0);
        res_bls_docs = res_bls_docs || v_delimiter || json_item;
        v_delimiter = ',';
      end

      if (res_bls_docs <> '') then
      begin
        res_bls_docs = '{"caption":"Документы", "auto_load":1, "blobs":[' || res_bls_docs || ']}';
        res_has_bls_docs = 1;
      end
    end

    suspend;
  end
end
