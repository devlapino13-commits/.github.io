CREATE PROCEDURE AAREP_PLANNING_SALDOS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7),
    REP_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    MA_BRANCH_ID INTEGER,
    MA_BRANCH_NAME VARCHAR(100),
    MA_REGION_ID INTEGER,
    MA_REGION_NAME VARCHAR(100),
    MA_SALDO_0 DECIMAL(18,4),
    MA_SALDO_1 DECIMAL(18,4),
    MA_SALDO_2 DECIMAL(18,4),
    MA_SALDO_3 DECIMAL(18,4),
    MA_SALDO_4 DECIMAL(18,4),
    MA_SALDO_5 DECIMAL(18,4),
    MA_SALDO_6 DECIMAL(18,4),
    MA_SALDO_7 DECIMAL(18,4),
    MA_SALDO_8 DECIMAL(18,4),
    MA_SALDO_9 DECIMAL(18,4),
    MA_SALDO_10 DECIMAL(18,4),
    MA_SALDO_11 DECIMAL(18,4),
    MA_SALDO_12 DECIMAL(18,4),
    MA_SALDO_13 DECIMAL(18,4),
    MA_SALDO_14 DECIMAL(18,4),
    MA_DATE_OPER DATE,
    MA_HQ_NAME VARCHAR(100))
AS
declare variable res_hq integer;
begin
  ma_date_oper = :rep_date_begin - 1 + rep_include_first_day;
  for select ma_region_id, ma_region_name, sum(ma_saldo_0), sum(ma_saldo_0 + ma_saldo_1),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2), sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6 + ma_saldo_7),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6 + ma_saldo_7 + ma_saldo_8),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6 + ma_saldo_7 + ma_saldo_8 + ma_saldo_9),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6 + ma_saldo_7 + ma_saldo_8 + ma_saldo_9 + ma_saldo_10),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6 + ma_saldo_7 + ma_saldo_8 + ma_saldo_9 + ma_saldo_10 + ma_saldo_11),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6 + ma_saldo_7 + ma_saldo_8 + ma_saldo_9 + ma_saldo_10 + ma_saldo_11 + ma_saldo_12),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6 + ma_saldo_7 + ma_saldo_8 + ma_saldo_9 + ma_saldo_10 + ma_saldo_11 + ma_saldo_12 + ma_saldo_13),
             sum(ma_saldo_0 + ma_saldo_1 + ma_saldo_2 + ma_saldo_3 + ma_saldo_4 + ma_saldo_5 + ma_saldo_6 + ma_saldo_7 + ma_saldo_8 + ma_saldo_9 + ma_saldo_10 + ma_saldo_11 + ma_saldo_12 + ma_saldo_13 + ma_saldo_14)
      from aarep_planning_saldos_2(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_auto_period, :rep_auto_koef, :rep_include_first_day)
      group by ma_region_id, ma_region_name
      into :ma_region_id, :ma_region_name, :ma_saldo_0, :ma_saldo_1, :ma_saldo_2, :ma_saldo_3, :ma_saldo_4, :ma_saldo_5,
           :ma_saldo_6, :ma_saldo_7, :ma_saldo_8, :ma_saldo_9, :ma_saldo_10, :ma_saldo_11, :ma_saldo_12, :ma_saldo_13,
           :ma_saldo_14
  do
  begin
    select r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name, r3.reg_unicode
    from region r1, region r2, region r3
    where r1.reg_unicode = :ma_region_id and
          r2.reg_unicode = r1.parent and
          r3.reg_unicode = r2.parent
    into :ma_branch_id, :ma_branch_name, :ma_hq_name, :res_hq;
    if (res_hq = 1000000) then
    begin
      ma_hq_name = ma_branch_name;
    end
    suspend;
  end

end
