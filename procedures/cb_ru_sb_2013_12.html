CREATE PROCEDURE CB_RU_SB_2013_12(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OKRUG_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    REG_NAME VARCHAR(50),
    PERSONAL_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    REG_BALANCE_LAST CURR_TYPE,
    REG_BALANCE_DELINQ CURR_TYPE,
    PAR7 DECIMAL(18,2),
    PAR7_LAST DECIMAL(18,2),
    BONUS_PAR CURR_TYPE,
    SUM_VZYSKANIE CURR_TYPE,
    BONUS_VZYSKANIE CURR_TYPE,
    CB_BONUS_SB_TOTAL CURR_TYPE,
    SRED_PLATEJ_GROUP CURR_TYPE,
    PAR_SNIJENIE CURR_TYPE,
    KOL_CLI_CATEG_1_2 INTEGER,
    BONUS_ACTION CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable rep_date_end_2 date;
declare variable ps_from_c_id curr_type;
declare variable reg_balance_curr decimal(18,4);
declare variable reg_kolich_sb integer;
declare variable id_list integer;
declare variable kol_group integer;
declare variable sum_platej_2000 curr_type;
declare variable rep_date_end_3 date;
begin
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end) + 1; -- Начало периода
  rep_date_end_2 = :rep_date_begin - 1; -- Конец предыдущего месяца

  rep_date_end_3 = :rep_date_end_2 - extract(day from :rep_date_end_2);

  for select id_list, r3.reg_code || ' ' || r3.reg_name, r2.reg_code || ' ' || r2.reg_name,
             r1.reg_code || ' ' || r1.reg_name
      from region_list(:rep_region_id, 1, 3, 1), region r1, region r2, region r3
      where r1.reg_unicode = id_list and
            r2.reg_unicode = r1.parent and
            r3.reg_unicode = r2.parent
      order by r3.reg_code, r2.reg_code, r1.reg_code
      into id_list, okrug_name, branch_name, reg_name
  do
  begin
-- Вычисление ПАР, за предыдущий отчетный период
-----------------------------------------------------------------------------------------------
--  Портфель займа всего офиса (1 и 2 категории) на предыдущую отчетную дату
    reg_balance_last = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(oks_from_k_id, :rep_date_end_3, null, null, null, null, null, null)) in (1, 2), oks_balance, 0))
    from oksummary
    where :rep_date_end_3 between oks_date_begin and oks_date_end and
          oks_region_id = :id_list and
          oks_close_state = 0
    into reg_balance_last;
    if (reg_balance_last is null) then
      reg_balance_last = 0;

--  Портфель займов всего офиса (с просрочками от 8 до 180 дней) на отчетную дату
    reg_balance_delinq = 0;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end_2 between oks_date_begin and oks_date_end and
          oks_region_id = :id_list and
          oks_delinq_date <= :rep_date_end_2 and
          :rep_date_end_2 - oks_delinq_date between 8 and 180 and
          oks_close_state = 0
    into reg_balance_delinq;
    if (reg_balance_delinq is null) then
      reg_balance_delinq = 0;

--  Расчет PAR,
    if (reg_balance_last > 0) then
      par7_last = reg_balance_delinq / reg_balance_last * 100;
    else
    if (reg_balance_delinq > 0) then
      par7_last = 100;
    else
      par7_last = 0;
-----------------------------------------------------------------------------------------------

--  Портфель займа всего офиса (1 и 2 категории) на предыдущую отчетную дату
    reg_balance_last = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(oks_from_k_id, :rep_date_end_2, null, null, null, null, null, null)) in (1, 2), oks_balance, 0))
    from oksummary
    where :rep_date_end_2 between oks_date_begin and oks_date_end and
          oks_region_id = :id_list and
          oks_close_state = 0
    into reg_balance_last;
    if (reg_balance_last is null) then
      reg_balance_last = 0;

--  Портфель займов всего офиса (с просрочками от 8 до 180 дней) на отчетную дату
    reg_balance_delinq = 0;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_region_id = :id_list and
          oks_delinq_date <= :rep_date_end and
          :rep_date_end - oks_delinq_date between 8 and 180 and
          oks_close_state = 0
    into reg_balance_delinq;
    if (reg_balance_delinq is null) then
      reg_balance_delinq = 0;

--  Расчет PAR,
    if (reg_balance_last > 0) then
      par7 = reg_balance_delinq / reg_balance_last * 100;
    else
    if (reg_balance_delinq > 0) then
      par7 = 100;
    else
      par7 = 0;
    if (par7 <= 3.5) then
      bonus_par = 8000;
    else
    if (par7 <= 7) then
      bonus_par = 6000;
    else
      bonus_par = 0;

    par_snijenie = par7_last - par7;
--  Разработка акция до 31го декабря
----------------------------------------------------------------------------------------------------------------
-- кол-во активных клиентов по категории 1 и 2, данного офиса
    with tabl1
    as (select oks_from_k_id
        from oksummary
        where :rep_date_end between oks_date_begin and oks_date_end and
              oks_balance > 0 and
              oks_close_state = 0 and
              (select aa_category_id
               from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
              oks_region_id = :id_list)
    select count(distinct kc_from_c_id)
    from tabl1, knclient
    where oks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0
    into kol_cli_categ_1_2;

    if (kol_cli_categ_1_2 >= 1000) then
      bonus_action = case
                       when par_snijenie between 1 and 1.49 then 3000
                       when par_snijenie between 1.5 and 1.99 then 5000
                       when par_snijenie between 2 and 2.49 then 7000
                       when par_snijenie between 2.5 and 2.99 then 9000
                       when par_snijenie between 3 and 3.49 then 11000
                       when par_snijenie between 3.5 and 3.99 then 13000
                       when par_snijenie between 4 and 4.49 then 15000
                       when par_snijenie between 4.5 and 4.99 then 17000
                       when par_snijenie between 5 and 5.49 then 19000
                       when par_snijenie between 5.5 and 5.99 then 21000
                       when par_snijenie >= 6 then 25000
                       else 0
                     end;
    else
      bonus_action = case
                       when par_snijenie between 1 and 1.49 then 2000
                       when par_snijenie between 1.5 and 1.99 then 3000
                       when par_snijenie between 2 and 2.49 then 5000
                       when par_snijenie between 2.5 and 2.99 then 6000
                       when par_snijenie between 3 and 3.49 then 8000
                       when par_snijenie between 3.5 and 3.99 then 9000
                       when par_snijenie between 4 and 4.49 then 10000
                       when par_snijenie between 4.5 and 4.99 then 11000
                       when par_snijenie between 5 and 5.49 then 12000
                       when par_snijenie between 5.5 and 5.99 then 13000
                       when par_snijenie >= 6 then 15000
                       else 0
                     end;

----------------------------------------------------------------------------------------------------------------

--  Расчет кол-во СБ данного офиса
    select count(ps_from_c_id)
    from hr_pstatus, joblist
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :id_list and
          ps_status > -1 -- не аннулированные приказы
          and
          ps_is_working = 1 -- все активные сотрдуники
          and
          ps_from_jl_id = jl_id and
          jl_kind = 8
    into reg_kolich_sb;
--  Если кол-во ССВЗ > 1 данного офиса то все суммы расчетов делит на кол-во ССВЗ
    if (reg_kolich_sb > 1) then
    begin
      reg_balance_last = reg_balance_last / reg_kolich_sb;
      reg_balance_curr = reg_balance_curr / reg_kolich_sb;
      reg_balance_delinq = reg_balance_delinq / reg_kolich_sb;
      bonus_action = bonus_action / reg_kolich_sb;
--      cb_bonus_sb_total = cb_bonus_sb_total / reg_kolich_sb;
    end
    else
    if (reg_kolich_sb < 1) then
      bonus_par = 0;
--    if (cb_bonus_sb_total > 40000.0) then cb_bonus_sb_total = 40000.0;
--  Расчет суммы бонусов
    cb_bonus_sb_total = 0;
--  Вытаскивает ФИО и должность СБ
    personal_name = null;
    jl_name = null;
    sum_vzyskanie = 0;
    bonus_vzyskanie = 0;

    kol_group = 0;
    sum_platej_2000 = 0;
    sred_platej_group = 0;
    for select c_name_fio, jl_name, ps_from_c_id
        from hr_pstatus, joblist, clients
        where :rep_date_end between ps_date_begin and ps_date_end and
              ps_from_reg_id = :id_list and
              ps_status > -1 -- не аннулированные приказы
              and
              ps_is_working = 1 -- все активные сотрдуники
              and
              ps_from_jl_id = jl_id and
              jl_kind = 8 -- Выборка только по СБ
              and
              ps_from_c_id = c_id
        into personal_name, jl_name, ps_from_c_id
    do
    begin
--    Сумма взыскании по портфелю катекории 2 и кол-во групп.
      sum_vzyskanie = 0;
      kol_group = 0;
      sum_platej_2000 = 0;
      sred_platej_group = 0;
      with tabl1
      as (select k_id, -sum(ok_os + ok_perc + ok_fine) sum_vzyskan
          from operkred, kontract, knaktlist
          where ok_ol_is_active = 1 and
                ok_ol_date between :rep_date_begin and :rep_date_end and
                ok_type = 4 and
                ok_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                ok_ol_date between kal_date and kal_date_end and
                kal_new_kf_officer = :ps_from_c_id and
                ok_from_c_id = 0 and
                (select aa_category_id
                 from aarep_get_category_id(k_id, ok_ol_date - 1, null, null, null, null, null, null)) in (2, 3) and
                not exists(select *
                           from sb_akt, sb_list -- добавлено по приказу №302-06-13, пункт 2, не включать закрытые по СБ
                           where sba_is_delete = 0 and
                                 sba_from_sbl_id <> 0 and
                                 sba_status = 2 and
                                 ok_ol_date between sba_date_begin and sba_date_end and
                                 sba_from_k_id = k_id and
                                 sba_from_sbl_id = sbl_id and
                                 sbl_type = 1)
          group by k_id)
      select sum(sum_vzyskan), count(iif(sum_vzyskan >= 2000, k_id, null)),
             sum(iif(sum_vzyskan >= 2000, sum_vzyskan, 0))
      from tabl1
      into sum_vzyskanie, kol_group, sum_platej_2000;

      if (sum_vzyskanie is null) then
        sum_vzyskanie = 0;
      if (sum_platej_2000 is null) then
        sum_platej_2000 = 0;
      if (kol_group > 0) then
        sred_platej_group = sum_platej_2000 / kol_group;
      else
        sred_platej_group = 0;

      if (par7 <= 7) then
        bonus_vzyskanie = sum_vzyskanie * 7 / 100;
      else
      if (sum_vzyskanie < 210000.00) then
        bonus_vzyskanie = sum_vzyskanie * 5 / 100;
      else
      if (par7 - par7_last >= 0.5) then
        bonus_vzyskanie = sum_vzyskanie * 7 / 100;
      else
      if (sum_vzyskanie < 300000.00) then
      begin
        if (sred_platej_group <= 50000.00) then
          bonus_vzyskanie = sum_vzyskanie * 12 / 100;
        else
          bonus_vzyskanie = sum_vzyskanie * 8 / 100;
      end
      else
      if (sred_platej_group <= 50000.00) then
        bonus_vzyskanie = sum_vzyskanie * 15 / 100;
      else
        bonus_vzyskanie = sum_vzyskanie * 9 / 100;

      cb_bonus_sb_total = bonus_par + bonus_vzyskanie;

      suspend;
    end
    if (jl_name is null) then
      suspend; -- если в данном офисе нету СБ то возвращает данные
  end
end
