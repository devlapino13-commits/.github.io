CREATE PROCEDURE FM_INCOME_EXPENSE_FORM_DATA(
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable ACL_ID D_ID;
declare variable RESPONS_C_ID D_ID;
declare variable KONTRACT_NO D_STR_255;
declare variable C_NAME_FIO D_STR_255;
declare variable ACTION_LINK D_STR_LONG;
declare variable ROW_HTML D_STR_LONG;
declare variable HEADER_HTML D_STR_LONG;
declare variable LANG D_STR_LANG;
declare variable HAS_ROWS D_SMALLINT;
begin
  res_html = coalesce((select res_style from fm_html_styles), '') || '<body>';
  lang = f_get_lang_code(:un_id, null);
  respons_c_id = (select un.un_from_c_id from username un where un.un_id = :un_id);

  header_html = f_format('<h1 class="green_text">%1</h1>', 'Анкета доходов/расходов') ||
                         '<p>Список контрактов, ожидающих заполнения анкеты:</p>' ||
                         '<table width="100%"><col width="30%"><col width="30%"><col width="20%">' ||
                         '<tr><th>Контракт</th><th>Клиент</th><th>Действие</th></tr>';

  res_html = blob_append(res_html, :header_html);
  has_rows = 0;

  for select acl.acl_id,
             kn.k_kontract_no,
             cl.c_name_fio
  from actionlist acl
  inner join miniappclient mac on mac.mac_id = acl.acl_object_id
  inner join kontract kn on kn.k_from_ma_app_id = mac.mac_from_ma_id
  inner join clients cl on cl.c_id = mac.mac_from_c_id
  where acl.acl_from_rd_id = f_get_rd_id('ActionGroup.IncomeExpenses') and
        acl.acl_respons_c_id = :respons_c_id and
        acl.acl_status <> 2 and
        mac.mac_amount > 0
  into :acl_id, :kontract_no, :c_name_fio
  do
  begin
    action_link = f_uri_open(f_lang(:lang, 'Заполнить'), 'cli_financial_profile',
                              f_objkind('miniappclient', 'id'), -:acl_id, null, null, :un_id);

    row_html = '<tr>' ||
               '<td align="center">' || coalesce(:kontract_no, '-') || '</td>' ||
               '<td align="center">' || coalesce(:c_name_fio, 'Неизвестный клиент') || '</td>' ||
               '<td align="center"><b>' || :action_link || '</b></td>' ||
               '</tr>';

    res_html = blob_append(res_html, :row_html);
    has_rows = 1;
  end

  res_html = blob_append(res_html, '</table><br><br>');

  if (has_rows = 0) then
    res_html = '<p>Нет задач, ожидающих заполнения анкеты.</p>';

  res_html = blob_append(res_html, '</body></html>');
  suspend;
end
