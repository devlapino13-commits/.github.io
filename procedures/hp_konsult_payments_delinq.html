CREATE PROCEDURE HP_KONSULT_PAYMENTS_DELINQ
AS
declare variable k_id d_id;
declare variable kf_id d_id;
declare variable k_fact_disb_date d_date;
declare variable office_id d_integer;
declare variable office_name d_str_100;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable ol_id d_id;
declare variable ol_date d_date;
declare variable ol_memo d_str_long;
declare variable ob_detail d_str_long;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_doc_kind d_integer;
declare variable ol_is_client d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable sh_date d_date;
declare variable delinq_ol_id d_id;
declare variable delinq_amount d_currency;
begin
  -- Ежедневная проверка просрочки графика платежей конс. услуг в рассрочку (RFCKG-4900)
  -- При нахождении просрочки сумма просрочки фиксируется в operlist/operbook
  -- Фиксируется сумма просрочки только последнего планового платежа (последнего до текущего дня),
  -- предыдущие просрочки должны были быть зафиксированы в свои даты (когда они были последними плановыми платежами).
  -- Иначе говоря, проводки по просрочкам показывают просроченные суммы плановых платежей каждый отдельно.

  for select oc_from_k_id,
             max(oc_schedule_date)
      from opercons
      inner join oksummary on oks_from_k_id = oc_from_k_id and current_date between oks_date_begin and oks_date_end and oks_close_state = 0
      where oc_ol_is_active = 1 and
            oc_schedule_date < current_date
      group by oc_from_k_id
      having sum(oc_schedule_principal + oc_paid_principal) > 0
      into k_id, -- активный контракт, у которго есть просрочка по предыдущим платежам
           sh_date -- дата последнего планового платежа до текущего дня
  do
  begin
    select sum(oc_schedule_principal + oc_paid_principal)
    from opercons
    where oc_ol_is_active = 1 and
          oc_from_k_id = :k_id and
          oc_schedule_date = :sh_date
    into delinq_amount; -- сумма просрочки последнего планового платежа

    if (delinq_amount is null) then
      delinq_amount = 0;

    if (delinq_amount <= 0) then
      continue; -- последний плановый платеж без просрочки

    select oc_delinq_ol_id
    from opercons
    where oc_ol_is_active = 1 and
          oc_from_k_id = :k_id and
          oc_operation_type_id = 1 and -- график оплаты
          oc_schedule_date = :sh_date
    into delinq_ol_id; -- id операции фиксации просрочки

    if (delinq_ol_id is not null) then
      continue; -- просрочка уже зафиксирована в operlist/operbook

    -- создаем записи в operlist/operbook
    ol_date = current_date;
    ol_oper_type = 2; -- Чисто бухгалтерская
    ol_oper_subtype = 11; -- Мемориальный ордер
    ol_doc_kind = 9; -- Мемориальный ордер
    ol_is_client = 1;

    select k_from_kf_id,
           k_fact_disb_date
    from kontract
    where k_id = :k_id
    into kf_id,
         k_fact_disb_date;

    select kal_new_reg_id
    from knaktlist
    where is_delete = 0 and
          kal_from_kf_id = :kf_id and
          :ol_date between kal_date and kal_date_end
    into office_id;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :office_id
    into office_name;

    select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 rows 1
    into c_id;

    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    ol_memo = 'Просроченная часть суммы к получению по договору Консультационных услуг / ' || f_datetostr(k_fact_disb_date) || ' / ' || c_name_fio;
    ob_detail = 'Просрочка конс. услуг в рассрочку';
    ol_id = null;

    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = '99911/2'
    into acc_dt;

    if (acc_dt = '') then
      exception e_message 'Дт. счет просрочки платежей конс. услуг в рассрочку не определен, офис ' || office_name;

    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = '99911/1'
    into acc_kt;

    if (acc_kt = '') then
      exception e_message 'Кт. счет просрочки платежей конс. услуг в рассрочку не определен, офис ' || office_name;

    insert into operlist (ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no_str, ol_document_client, ol_is_client, ol_memo, ol_status, ol_user_id, ol_date)
    values (:office_id, :office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :k_id, :c_id, :ol_is_client, :ol_memo, 1, 1000000000000, :ol_date)
    returning (ol_id)
    into ol_id;

    insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_client_dt, ob_detail)
    values (:ol_id, :acc_dt, :acc_kt, :delinq_amount, :delinq_amount, :ol_date, :office_id, :c_id, :ob_detail);

    -- процедура запускается роботом ежедневно, поэтому по одной и той же просрочке в operlist/operbook создавались бы одни те же записи (дубликаты)
    -- во избежание этого ol_id фиксируем в opercons в строке графика оплаты, а при каждом запуске процедуры в начале проверяем наличие ol_id
    -- если ol_id (oc_delinq_ol_id) есть, то новые записи в operlist/operbook не создаем
    update opercons
    set oc_delinq_ol_id = :ol_id
    where oc_from_k_id = :k_id and
          oc_schedule_date = :sh_date and
          oc_ol_is_active = 1 and
          oc_operation_type_id = 1; -- график оплаты
  end
end
