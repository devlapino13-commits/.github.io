CREATE PROCEDURE HP_PRETRIAL_CLAIM_SEND(
    SLD_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable smr_id d_id;
declare variable scd_code d_str_code;
declare variable scd_name d_str_255;
declare variable edl_id d_id;
declare variable edb_id d_id;
begin
  if (sld_id is null) then
    exit;

  res_text = '';

  select smr_id,
         scd_code,
         scd_name
  from sud_mainreq
  inner join sud_maindocs on smd_id = smr_id
  inner join sud_loandocs on sld_from_smd_id = smd_id
  inner join sud_cfgdockind on scd_id = sld_doc_kind
  where sld_id = :sld_id
  into smr_id,
       scd_code,
       scd_name;

  if (smr_id is null) then
  begin
    res_text = f_format('Не найдена заявка для sld_id = ', sld_id);
    suspend;
    exit;
  end

  if (scd_code <> 'pretrial_claim') then
  begin
    res_text = f_format('Для документа "%1" не предусмотрена отправка электронного документа.', scd_name);
    suspend;
    exit;
  end

  select edl_id,
         res_text
  from hp_edoc_ins(:scd_code, 'sud_mainreq', :smr_id, :un_id)
  into edl_id,
       res_text;

  if (coalesce(res_text, '') = '') then
  begin
    -- найдем бланк досудебной претензии
    select edb_id
    from edoc_blank
    inner join edoc_frx on edf_id = edb_from_edf_id
    where edf_code = :scd_code and
          edb_from_edl_id = :edl_id
    into edb_id;

    if (edb_id is null) then
      exception e_default 'Не найден бланк досудебной претензии (edoc = ' || edl_id || ')';

    insert into sud_loanbls (slb_from_sld_id, slb_object_kind, slb_object_id, slb_from_un_id)
    values (:sld_id, f_objkind('edoc_blank', 'id'), :edb_id, :un_id);
  end

  suspend;
end
