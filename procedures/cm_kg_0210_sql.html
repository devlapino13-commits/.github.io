CREATE PROCEDURE CM_KG_0210_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KMI_IMEI D_ID;
declare variable KONTRACT_OBJ_KIND D_INTEGER;
declare variable EXISTING_KML_ID D_INTEGER;
declare variable KML_ID D_INTEGER;
begin
  -- блок 210 проверяет контракт на этапах: сохранения, проверки на условия;
  -- утверждения; отправки на кредитный комитет; выдаче займа.
  --.loan.create_loan.check_preview.loan_approve.send_kk.loan_disb.
  -- срабатывает только тогда, когда есть другой активный или закрытый контракт с этим imei кодом

  check_kind = ''; -- заглушка, не используется
  un_id = 0; -- заглушка, не используется
  kontract_obj_kind = f_objkind('kontract', 'id');

  for select kmi.kmi_imei, km.kml_id
      from kontmat km
      inner join kontmat_imei kmi on kmi.kmi_kml_id = km.kml_id
      where km.kml_object_kind = :kontract_obj_kind and
            km.kml_object_id = :k_id and
            km.kml_status = 1
      into kmi_imei, kml_id
  do
  begin
    select km.kml_id
    from kontmat_imei kmi
    inner join kontmat km on km.kml_id = kmi.kmi_kml_id
    inner join kontstate ks on ks.ks_from_k_id = km.kml_object_id
    where km.kml_object_kind = :kontract_obj_kind and
      km.kml_object_id <> :k_id and -- другие займы
      km.kml_status = 1 and
      ks.ks_ol_is_delete = 0 and -- не удаленные
      ks.ks_state in (1, 2) and  -- только активные и закрытые
      current_date between ks.ks_date_begin and ks.ks_date_end and  -- добавил RFCKG-4871
      kmi.kmi_imei = :kmi_imei -- с таким же imei кодом
    into :existing_kml_id;

    if (:existing_kml_id is not null and :existing_kml_id <> 0) then                  --RFCKG-4854
    begin
      if (not exists (select 1
                   from duplicated_imei di
                   where di.di_from_k_id = :k_id and
                     di.di_from_kml_id = :kml_id)) then
        execute procedure duplicated_imei_actions(:kmi_imei, :k_id, :existing_kml_id, :un_id);

      res_info_text = '';
      res_error_text = 'Есть другой активный или закрытый контракт с таким же IMEI кодом';
      res_error_is_critical = 1;
      suspend;
      leave;
    end
  end
end
