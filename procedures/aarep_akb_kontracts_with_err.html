CREATE PROCEDURE AAREP_AKB_KONTRACTS_WITH_ERR(
    REP_DATE D_DATE)
RETURNS (
    K_NO D_STR_20,
    POCKET_ID D_ID,
    MP_DATE_BEGIN D_DATE,
    MP_DATE_END D_DATE,
    ERR_TEXT D_STR_2048)
AS
begin
  rep_date = rep_date - 1;
  for select k.k_kontract_no, mp.mp_id, melc.lc_error_code, mp.mp_date_begin, mp.mp_date_end
      from mkr_poket mp
      inner join mkr_credit mc on mp.mp_id = mc.mc_from_mp_id
      inner join mkr_err_log_credits melc on melc.lc_k_id = mc.mc_from_k_id
      inner join kontract k on mc.mc_from_k_id = k.k_id
      where mc.mc_is_err = 1 and
            mp.mp_date_end = :rep_date
      into :k_no, :pocket_id, :err_text, :mp_date_begin, :mp_date_end
      do
      begin
        suspend;
      end
end
