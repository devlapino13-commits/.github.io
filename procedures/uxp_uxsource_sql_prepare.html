CREATE PROCEDURE UXP$UXSOURCE_SQL_PREPARE(
    US_UUID D_UUID,
    UF_FIELD_NAME D_STR_CODE,
    USR_UUID_CODE D_STR_CODE = null)
RETURNS (
    LOOKUP_SQL D_STR_LONG,
    KEYNAME_SQL D_STR_LONG,
    ERROR_TEXT D_STR_255)
AS
declare variable us_main_table d_str_code;
declare variable tmp_uuid d_uuid;
begin
  lookup_sql = '';
  keyname_sql = '';
  error_text = '';

  if (uf_field_name is null) then
    uf_field_name = '';

  if (coalesce(usr_uuid_code, '') <> '' and uf_field_name <> '') then
  begin
    if (exists(select * from ux$search
    where usr_uuid = char_to_uuid(:usr_uuid_code) and
          usr_id_field = usr_text_field)) then
    begin
      lookup_sql = 'cast(' || uf_field_name || ' as d_str_255)';
      keyname_sql = 'cast(' || uf_field_name || ' as d_str_255)';
      suspend;
      exit;
    end
  end

  select us_main_table
  from ux$source
  where us_uuid = :us_uuid
  into us_main_table;

  if (us_main_table is null) then
    us_main_table = '';

  if (us_main_table <> '') then
  begin
    select keyname_sql
    from uxp$uxsource_sql_keyname(null, :us_main_table)
    where error_text = ''
    into :keyname_sql;
  end

  if (keyname_sql = '') then
  begin
    select keyname_sql, error_text
    from uxp$uxsource_sql_keyname(:us_uuid)
    into :keyname_sql, error_text;
  end

  if (keyname_sql = '' and coalesce(usr_uuid_code, '') <> '') then
  begin
    select us_uuid
    from ux$alias, ux$code, ux$form, ux$frame, ux$source
    where uxa_uuid = char_to_uuid(:usr_uuid_code) and
          uxa_owner = ux_uuid and
          coalesce(ux_release_uuid, ux_draft_uuid) = ufm_uuid and
          ufm_uuid = ufr_owner and
          ufr_is_mainframe = 1 and
          ufr_add_table_fields = us_code
    rows 1
    into tmp_uuid;

    if (tmp_uuid is not null) then
    begin
      select keyname_sql, error_text
      from uxp$uxsource_sql_keyname(:tmp_uuid)
      into :keyname_sql, error_text;
    end
  end

  select lookup_sql, iif(error_text <> '', error_text, :error_text)
  from uxp$uxsource_sql_lookup(:us_uuid)
  into :lookup_sql, error_text;

  suspend;
end
