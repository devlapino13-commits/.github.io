CREATE PROCEDURE FM_HTML_COMPET_DETAIL_FOR_MO(
    UN_ID DECIMAL(18,0),
    OBJ_ID DECIMAL(18,0))
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable color_green d_str_20;
declare variable reg_id d_id;
declare variable cpd_add_data_1 d_currency;
declare variable cpd_add_data_2 d_currency;
declare variable cpd_add_data_3 d_currency;
declare variable cpd_add_data_4 d_currency;
declare variable cpd_add_data_5 d_currency;
declare variable cpd_add_data_6 d_currency;
declare variable cpd_add_data_7 d_currency;
declare variable cpd_add_data_8 d_currency;
declare variable cpd_add_data_9 d_currency;
declare variable cpr_id d_id;
declare variable mo_id d_id;
declare variable cpd_points_1 d_integer;
declare variable cpd_points_2 d_integer;
declare variable cpd_points_3 d_integer;
declare variable cpd_points_4 d_integer;
declare variable cpd_points_5 d_integer;
declare variable cpp_stage d_integer;
declare variable cpr_final_orderno d_integer;
declare variable cpr_final_points d_integer;
declare variable cpr_group_name d_str_255;
declare variable rd_id d_id;
declare variable cpp_date_end d_date;
declare variable res_team_id d_integer;
declare variable cpd_add_data_10 d_currency;
begin
  res_html = (select res_style
              from fm_html_styles) || '<body>';
  color_green = '#02B316';

  select ps.ps_from_reg_id, ps.ps_from_c_id
  from hr_pstatus ps
  inner join username un on un.un_from_c_id = ps.ps_from_c_id and
        un.un_id = :un_id
  where current_date between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_is_working = 1
  into reg_id, mo_id;

  select cpp.cpp_date_end, cpp.cpp_stage
  from competperiod cpp
  where cpp.cpp_from_rd_id = f_get_rd_id('Competitions.Olimpic_2026_spf') and
        cpp.cpp_status = 1 and
        cpp.cpp_id = :obj_id
  into cpp_date_end, cpp_stage;

  select res_team_id
  from aarep_staffteam_list(:reg_id, :cpp_date_end)
  where res_team_id <> 0
  rows 1
  into res_team_id;

  for select distinct cpr.cpr_id, -- lo_id
                      cpr.cpr_object_id, -- название группы
                      cpd.cpd_add_data_1, -- Рост портфеля без просрочек
                      cpd.cpd_add_data_2, -- Возвратность
                      cpd.cpd_add_data_3, -- Кол-во уволенных ВИПС
                      cpd.cpd_add_data_4, -- Супер-показатель
                      cpd.cpd_add_data_5, -- Выдачи СПФ новым клиентам
                      cpd.cpd_add_data_6, -- Выдачи СПФ c 1-15
                      cpd.cpd_add_data_8, -- Супер-показатель 2
                      cpd.cpd_add_data_9,
                      cpd.cpd_points_1,
                      cpd.cpd_points_2,
                      cpd.cpd_points_3,
                      cpd.cpd_points_4,
                      cpd.cpd_points_5,
                      cpr.cpr_final_points,
                      cpr.cpr_final_orderno
      from competresult cpr
      inner join competdetails cpd on cpd.cpd_from_cpr_id = cpr.cpr_id
      where cpr.cpr_object_kind = -1 and
            cpr.cpr_object_id = :res_team_id and
            cpr.cpr_from_cpp_id = :obj_id and
            cpd_add_data_12 = :reg_id
      into cpr_id, rd_id, cpd_add_data_1, cpd_add_data_2, cpd_add_data_3, cpd_add_data_4, cpd_add_data_5,
           cpd_add_data_6,cpd_add_data_8, cpd_add_data_9,  cpd_points_1, cpd_points_2, cpd_points_3, cpd_points_4, cpd_points_5, cpr_final_points, cpr_final_orderno
  do
  begin
    select rd.rd_name
    from refdata rd
    where rd.rd_id = :rd_id
    into cpr_group_name;

    res_html = blob_append(res_html, f_format('<br>Название команды: <b><font color="%1" style="font-size: 100%">%2</font></b><br><br>', :color_green, :cpr_group_name));
    res_html = blob_append(res_html, '<table width="100%">
       <col width="40%">
       <col width="30%">
       <col width="30%">
       <tr>' || f_format('<th><b>Показатель</b></th><th><b>Значение</b></th><th><b>Балл</b></th></tr><tr>') --
    || f_format('<th>Выдачи СПФ новым клиентам</th><th>%1</th><th>%2</th>', coalesce(:cpd_add_data_4, '-'), iif(coalesce(:cpd_add_data_4, 0) > 0 and cpp_stage = 0, :cpr_final_points, 0)) || '</tr><tr>' --
    || f_format('<th>Выдачи СПФ c 1-15</th><th>%1</th><th>%2</th>', coalesce(:cpd_add_data_5, '-'), iif(coalesce(:cpd_add_data_5, 0) > 0 and cpp_stage = 0, :cpr_final_points, 0)) || '</tr><tr>' --
    || f_format('<th>Рост портфеля без просрочек</th><th>%1</th><th>%2</th>', coalesce(:cpd_add_data_1, '-'), coalesce(:cpd_points_1, '-')) || '</tr><tr>' --
    || f_format('<th>Возвратность</th><th>%1</th><th>%2</th>', coalesce(:cpd_add_data_2, '-'), coalesce(:cpd_points_2, '-')) || '</tr><tr>' --
    || f_format('<th>Выполнение плана по портфелю</th><th>%1</th><th>%2</th>', coalesce(:cpd_add_data_7, '-'), coalesce(:cpd_points_4, '-')) || '</tr><tr>' --
    || f_format('<th>1й Супер-показатель</th><th>%1</th><th>%2</th>', coalesce(:cpd_add_data_3, '-'), coalesce(:cpd_points_3, '-')) || '</tr>'
    || f_format('<th>2й Супер-показатель</th><th>%1</th><th>%2</th>', coalesce(:cpd_add_data_8, '-'), coalesce(:cpd_points_5, '-')) || '</tr>');
    --|| f_format('<th>Ранг по супер-показателю </th><th>%1</th><th>%2</th>', coalesce(:cpd_add_data_4, '-'), coalesce(:office_rank, '-')) || '</tr>');



    res_html = blob_append(res_html, '</table>');
    res_html = blob_append(res_html, f_format('<br><b>Ранг команды: %1<br> Общий балл команды: %2<br></b>', :cpr_final_orderno, :cpr_final_points));
  end
  suspend;
end
