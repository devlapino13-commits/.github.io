CREATE PROCEDURE EPASS_INS_OR_UPD_CLIENT(
    ER_ID D_INTEGER)
RETURNS (
    IS_ERROR D_SMALLINT,
    ERR_TEXT D_STR_100)
AS
declare variable FIO D_STR_100;
declare variable SEX D_SMALLINT;
declare variable INN D_STR_20;
declare variable BIRTH_DATE D_DATE;
declare variable CITIZEN D_INTEGER;
declare variable NATION D_INTEGER;
declare variable PASS_NO D_STR_100;
declare variable OLD_PASS_NO D_STR_100;
declare variable PASS_DATE D_DATE;
declare variable PASS_DATE_END D_DATE;
declare variable DOC_TYPE D_INTEGER;
declare variable AUTHORITY D_STR_255;
declare variable AUTHORITY_CODE D_STR_20;
declare variable UN_ID D_ID;
declare variable CDS_ID D_ID;
declare variable C_ID D_ID;
declare variable BLS1 D_ID;
declare variable INN_TMP D_STR_20;
declare variable SURNAME D_STR_100;
declare variable NAME D_STR_100;
declare variable PATRONYMIC D_STR_100;
begin
  is_error = 0;
  err_text = '';
  old_pass_no = '';

  select erp.erp_pdf_blob, epr.er_from_c_id
  from epass_response erp
  inner join epass_req epr on epr.er_id = erp.erp_from_er_id
  where erp.erp_from_er_id = :er_id
  into :bls1, :c_id;

  select fio, sex, inn, birth_date, citizen, nation, pass_no, pass_date, pass_date_end, doc_type, authority,
         un_id, surname, name, patronymic
  from epass_parse_res_json(:er_id)
  into :fio, :sex, :inn, :birth_date, :citizen, :nation, :pass_no, :pass_date, pass_date_end, :doc_type, authority,
       :un_id, :surname, :name, :patronymic;

  if (coalesce(c_id, 0) > 0) then
    select c.c_id, c.c_pass_no
    from clients c
    where c.c_id = :c_id
    into :c_id, old_pass_no;

  if (coalesce(c_id, 0) = 0) then
  begin
    select c_id
    from fm_add_client(:fio, :sex, :birth_date, null, null, :nation, :citizen, :doc_type, :pass_no, :pass_date, :inn, :un_id, :surname, :name, :patronymic, null, null, null)
    into :c_id;
    if (coalesce(c_id, 0) = 0) then
    begin
      is_error = 1;
      err_text = 'Произошла ошибка при регистрации нового клиента.';
      suspend;
      exit;
    end
  end
  else
  begin
    select inn
    from fm_get_client_inn(:c_id)
    into :inn_tmp;

    if (:inn_tmp = '' and
        coalesce(:inn, '') <> '') then
    begin
      update clients
      set c_inn = :inn
      where (c_id = :c_id);
    end
  end

  select value_out
  from str_with_separator_parser(replace(:authority, ' ', ';'))
  where value_out similar to '[-]?[[:DIGIT:]]+' = true
  into :authority_code;

  select cds_id, err_text, 1
  from fm_insert_cldoc(:c_id, :doc_type, :un_id, 1, :pass_no, :pass_date, :pass_date_end, :authority, :authority_code, '', '', '', '', '', '', '')
  into :cds_id, :err_text, is_error;

  if (coalesce(:err_text, '') = '') then
  begin
    update clientdocs
    set cds_doc_blob_1 = :bls1
    where cds_id = :cds_id;

    update clients c
    set c.c_blob_pass1_id = :bls1,
        c.c_blob_pass2_id = null
    where c.c_id = :c_id;

    if (:old_pass_no <> :pass_no) then
    begin
      select res_text
      from fm_send_passport_grs(:c_id, :un_id)
      into err_text;

      is_error = iif(:err_text = 'Данные успешно отправлены в ГРС', 0, 1);
    end
  end

  update epass_response
  set erp_from_c_id = :c_id
  where erp_from_er_id = :er_id;

  suspend;
end
