CREATE PROCEDURE HP_HISTORY_EDIT_CLIENT(
    C_ID D_ID)
RETURNS (
    CLL_DATETIME TIMESTAMP,
    EDIT_C_NAME_FIO D_STR_100,
    COMMENT_TEXT D_STR_2048,
    CLL_NAME_FIO D_STR_255,
    CLL_BIRTH_DATE DATE,
    CLL_PASS_NO D_STR_100,
    CLL_PASS_DATE DATE,
    CLL_BLOB_PHOTO_ID D_ID,
    CLL_BLOB_PASS1_ID D_ID,
    CLL_BLOB_PASS2_ID D_ID,
    CLL_INN D_STR_20)
AS
declare variable i d_integer;
declare variable old_cll_name_fio d_str_100;
declare variable old_cll_birth_date date;
declare variable old_cll_pass_no d_str_100;
declare variable old_cll_pass_date date;
declare variable old_cll_blob_photo_id d_id;
declare variable old_cll_blob_pass1_id d_id;
declare variable old_cll_blob_pass2_id d_id;
declare variable cll_from_un_id d_id;
declare variable old_cll_inn d_str_20;
declare variable db_id d_id;
begin
  db_id = f_get_db_id();
  i = 1;
  for with q
      as (select cll_name_fio, cll_from_un_id, cll_birth_date, cll_pass_no, cll_pass_date, cll_datetime,
                 cll_blob_photo_id, cll_blob_pass1_id, cll_blob_pass2_id, cll_inn
          from clientlog cll
          where cll.cll_from_c_id = :c_id
          order by cll.cll_datetime)
      select cll_name_fio, cll_from_un_id, cll_birth_date, cll_pass_no, cll_pass_date, cll_datetime, cll_blob_photo_id,
             cll_blob_pass1_id, cll_blob_pass2_id, cll_inn
      from q
      union all
      select c_name_fio, c_from_un_id, c_birth_date, c_pass_no, c_pass_date, cl.c_datetime, c_blob_photo_id,
             c_blob_pass1_id, c_blob_pass2_id, c_inn
      from clients cl
      where cl.c_id = :c_id
      into :cll_name_fio, cll_from_un_id, :cll_birth_date, :cll_pass_no, :cll_pass_date, cll_datetime,
           cll_blob_photo_id, cll_blob_pass1_id, cll_blob_pass2_id, :cll_inn
  do
  begin

    comment_text = '';

    if (i = 1) then
    begin
      comment_text = 'Создание клиента в БД';
      if (exists(select *
                 from docrecoglist d
                 where d.drl_from_c_id = :c_id and
                       drl_client_created = 1 and
                       d.drl_from_mcl_id is not null)) then
        comment_text = comment_text || ' через оцифровку КМП';
    end

    if (i > 1) then
    begin
      if (coalesce(old_cll_name_fio, '') <> coalesce(cll_name_fio, '')) then
        comment_text = comment_text || ' ФИО,';

      if (coalesce(old_cll_birth_date, '') <> coalesce(cll_birth_date, '')) then
        comment_text = comment_text || ' дата рождения,';

      if (coalesce(old_cll_pass_no, '') <> coalesce(cll_pass_no, '')) then
        comment_text = comment_text || ' номер паспорта,';

      if (coalesce(old_cll_pass_date, '') <> coalesce(cll_pass_date, '')) then
        comment_text = comment_text || ' дата выдачи паспорта,';

      if (coalesce(old_cll_blob_photo_id, '') <> coalesce(cll_blob_photo_id, '')) then
        comment_text = comment_text || ' фото клиента,';

      if (coalesce(old_cll_blob_pass1_id, '') <> coalesce(cll_blob_pass1_id, '')) then
        comment_text = comment_text || ' Лицевая сторона паспорта,';

      if (coalesce(old_cll_blob_pass2_id, '') <> coalesce(cll_blob_pass2_id, '')) then
        comment_text = comment_text || ' Обратная сторона паспорта,';

      if (db_id = 41701 and cll_datetime >= '16.10.2025' and coalesce(old_cll_inn, '') <> coalesce(cll_inn, '')) then
        comment_text = comment_text || ' ИНН,';

      if (comment_text <> '') then
        comment_text = substring(comment_text from 1 for character_length(comment_text) - 1); -- чтобы убрать лишнюю ','

    end
    old_cll_name_fio = cll_name_fio;
    old_cll_birth_date = cll_birth_date;
    old_cll_pass_no = cll_pass_no;
    old_cll_pass_date = cll_pass_date;
    old_cll_blob_photo_id = cll_blob_photo_id;
    old_cll_blob_pass1_id = cll_blob_pass1_id;
    old_cll_blob_pass2_id = cll_blob_pass2_id;
    old_cll_inn = cll_inn;

    edit_c_name_fio = '';
    select cl.c_name_fio
    from username u
    inner join clients cl on cl.c_id = u.un_from_c_id
    where u.un_id = :cll_from_un_id
    into :edit_c_name_fio;
    i = i + 1;

    if (comment_text <> '') then
      suspend;
  end
end
