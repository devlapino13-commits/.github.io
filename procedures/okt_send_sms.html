CREATE PROCEDURE OKT_SEND_SMS(
    CELLNO STR_20,
    SMS_TEXT STR_255)
AS
begin
  cellno = trim(coalesce(cellno, ''));
  sms_text = trim(coalesce(sms_text, ''));
  sms_text = replace(sms_text, '~', '''');

  if ((cellno = '') or (sms_text = '')) then
    exit;

  execute procedure fsrobot_insert_event('domailing', 'okt_sms', cellno || ';' || sms_text, 1000000000000);
end
