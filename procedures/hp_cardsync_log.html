CREATE PROCEDURE HP_CARDSYNC_LOG(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    CS_ID D_INTEGER,
    CS_PARENT_ID D_INTEGER,
    CS_DATE D_DATE,
    REG_NAME D_STR_100,
    K_KONTRACT_NO D_STR_100,
    C_NAME_FIO D_STR_100,
    CS_CARD_ID D_INTEGER,
    CCA_CODE D_STR_CODE,
    CSC_NAME D_STR_100,
    CSC_CODE D_STR_100,
    CS_STATUS_STR D_STR_100,
    CS_DATETIME D_DATETIME,
    CS_UPD_DATETIME D_DATETIME,
    CS_REQUEST_JSON D_STR_8100,
    CS_ANSWER_JSON D_STR_8100,
    IPC_ERR_TXT D_STR_LONG,
    FS_ERR_TXT D_STR_LONG,
    CSA_ID D_INTEGER,
    CS_AMOUNT CURR_TYPE,
    CS_STATUS D_INTEGER,
    IMG_ID D_SMALLINT,
    UN_NAME VARCHAR(100))
AS
begin
    -- Отчет по запросам в МПЦ
    
    for select cs.cs_id, -- id запроса
               cs.cs_parent_id,
               cs.cs_date, -- дата запроса
               reg.reg_name,  -- офис займа
               kont.k_kontract_no,  -- № договора
               cl.c_name_fio, -- фио клиента
               cs.cs_card_id, -- id карты
               caa.cca_code, -- карт счет
               cf.csc_name, -- задача
               cf.csc_code,
               cs.cs_status,
               cs.cs_datetime,  -- дата/время задачи
               cs.cs_upd_datetime, -- дата/время обработки
               cs.cs_request_json,  -- детали запроса
               cs.cs_answer_json,  -- детали ответа
               cse.csr_err_txt, -- ошибка от МПЦ
               csae.cse_error_txt, -- ошибка в FS после ответа МПЦ
               csa.csa_id, -- id обработчика в FS,
               cs.cs_amount,
               un.un_fio
    from cardsync cs
    inner join cardclientacc caa on caa.cca_id = cs.cs_from_cca_id
    inner join cardsynccfg cf on cf.csc_id + 0 = cs.cs_from_csc_id
    inner join region_list(:rep_region_id, 1, 3, 1) rl on rl.id_list + 0 = cs.cs_office_id
    inner join region reg on reg.reg_unicode = cs.cs_office_id
    inner join username un on cs.cs_from_un_id = un.un_id
    left join kontract kont on kont.k_id = cs.cs_from_k_id
    left join clients cl on cl.c_id = cs.cs_from_c_id
    left join cardsync_error cse on cse.csr_from_cs_id = cs.cs_id
    left join cardsync_action csa on csa.csa_from_cs_id = cs.cs_id
    left join cardsync_action_error csae on csae.cse_from_csa_id = csa.csa_id
    where cs.cs_date between :rep_date_begin and :rep_date_end
    order by cs.cs_card_id, cs_id
    into cs_id,
         cs_parent_id,
         cs_date,
         reg_name,
         k_kontract_no,
         c_name_fio,
         cs_card_id,
         cca_code,
         csc_name,
         csc_code,
         cs_status,
         cs_datetime,
         cs_upd_datetime,
         cs_request_json,
         cs_answer_json,
         ipc_err_txt,
         fs_err_txt,
         csa_id,
         cs_amount,
         un_name
    do
    begin
        -- статус задачи
        cs_status_str = trim((case cs_status when -2 then 'Окончательная ошибка МПЦ'
                                             when -1 then 'Временная ошибка МПЦ'
                                             when 0 then 'В процессе'
                                             when 1 then 'В процессе (заблокирована сервисом)'
                                             when 2 then 'Успешно обработан МПЦ'
                                             else 'Статус не определен' end));

        img_id = case
                   when cs_status < 0 then 2
                   when cs_status = 2 then 0
                   when cs_status = 0 then 1
                   else 2
                 end;


        suspend;
    end
end
