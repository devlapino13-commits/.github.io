CREATE PROCEDURE AAREP_CALC_BONUS_KG_KM(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_BRANCH VARCHAR(100),
    AA_OFFICE VARCHAR(100),
    KM_NAME VARCHAR(100),
    KM_JOBNAME VARCHAR(100),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_AMOUNT_DISB DECIMAL(18,4),
    AA_APPROVE_KOL INTEGER,
    AA_ANNUL_KOL INTEGER,
    AA_PAR DECIMAL(15,2),
    AA_BONUS_APPROVE DECIMAL(18,4),
    AA_BONUS_ANNUL DECIMAL(18,4),
    AA_BONUS_TOTAL DECIMAL(18,4),
    AA_BONUS_PAR DECIMAL(18,4))
AS
declare variable aa_deliq_days integer;
declare variable k_id_prior decimal(18,0);
declare variable kl_datetime timestamp;
declare variable aa_k_id decimal(18,0);
declare variable aa_curr_region_id integer;
declare variable km_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  aa_par = 0;
  for select aa_curr_region_id, sum(aa_amount_disb), sum(aa_amount_active_end), sum(aa_deliq_balance), km_id,
             sum(aa_approve_kol), sum(aa_annul_kol)
      from aarep_calc_bonus_kg_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by aa_curr_region_id, km_id
      order by aa_curr_region_id, km_id
      into :aa_curr_region_id, :aa_amount_disb, :aa_amount_active_end, :aa_deliq_balance, :km_id, :aa_approve_kol,
           :aa_annul_kol
  do
  begin
    aa_bonus_approve = 0;
    aa_bonus_annul = 0;
    aa_bonus_total = 0;
    aa_bonus_par = 0;

    select r1.reg_code || ' ' || r1.reg_name, r2.reg_name
    from region r1, region r2
    where r1.reg_unicode = :aa_curr_region_id and
          r1.parent = r2.reg_unicode
    into :aa_office, :aa_branch;

    km_name = '';
    km_jobname = '';
    if (km_id = 0) then
    begin
      if (aa_amount_active_end <> 0) then
      begin
        aa_deliq_balance = aa_deliq_balance * 100;
        aa_par = cast(aa_deliq_balance / aa_amount_active_end as decimal(15,2));
      end
    end
    if (km_id <> 0) then
    begin
      select c_name_fio, jl_name
      from hr_pstatus, clients, joblist
      where ps_from_c_id = :km_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_is_working = 1 and
            c_id = ps_from_c_id and
            ps_from_jl_id = jl_id rows 1
      into :km_name, :km_jobname;
    end
    aa_bonus_approve = aa_approve_kol * 300;
    aa_bonus_annul = aa_annul_kol * 200;
    aa_bonus_total = aa_bonus_approve + aa_bonus_annul;
    if (aa_par > 2.0) then
      aa_bonus_par = 0;
    else
      aa_bonus_par = aa_bonus_total;

    suspend;
  end

end
