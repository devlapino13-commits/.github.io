CREATE PROCEDURE CM_RU_0128_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable reg_id integer;
declare variable pr_code varchar(100);
declare variable k_disb_date date;
begin
  select k_last_reg_id, pr_code, k.k_disb_date
  from kontract k, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  into reg_id, pr_code, :k_disb_date;
  /*В соответствии с Приказом № 29-02-19 от 13.02.2019г . необходимо:
    с 15 февраля 2019 г.  
    - открыть доступ к продукту «Шанс»  для ОП № 13-02 г. Омск
    - закрыть доступ к продукту «Патент» для ОП № 13-02 г. Омск 
    - закрыть доступ в к продуктам «Патент» и «Патент плюс» для ОП46-01 Владимир
    
    с 20 февраля 2019г. для ОП № 12-03 "Благородный" г. Сургут
    - открыть доступ по продукту «Шанс» 
    - закрыть доступ по продукту «Патент» 
    */
  if (k_disb_date >= '15.02.2019') then
  begin   
    if (k_disb_date >= '20.02.2019') then
    begin  
        if (reg_id = 1120300 and
              pr_code containing '.PATENT.') then
        begin
          res_error_text = 'По указанному продукту запрещено выдавать в вашем офисе!';
          res_error_is_critical = 2;
          suspend;
        end
    end
    if (reg_id = 1130200 and
          pr_code containing '.PATENT.') then
    begin
      res_error_text = 'По указанному продукту запрещено выдавать в вашем офисе!';
      res_error_is_critical = 2;
      suspend;
    end
/*
    if (reg_id = 1460100 and
          pr_code containing '.PATENT') then
    begin
      res_error_text = 'По указанному продукту запрещено выдавать в вашем офисе!';
      res_error_is_critical = 2;
      suspend;
    end*/
  end  else
                            
 --added by Aleksei on 31-12-2018 -------------------------------------------->>
  if ((pr_code containing '.PATENT.')
       and (k_disb_date >= '09.01.2019')
       and (reg_id in (1400200,1500100)))  then   --- Воронеж и ОП 50-01 Ульяновск --ОП46-01 Владимир 1460100,убрал от 09.01.19 Музаффар
  begin
      res_error_text = 'По указанному продукту запрещено выдавать в вашем офисе!';
      res_error_is_critical = 2;
      suspend;
  end else
  ----------------------------------------------------------------------------<<
  if (pr_code similar to '%(.patent plus.|.status.)%') then
  begin
    if (reg_id not in (1120300,1130200,1400200,1420200,1460100,1500100,1120400,1490200,1490300,1510100,1490400)) then
    begin
      res_error_text = 'По указанному продукту запрещено выдавать в вашем офисе!';
      res_error_is_critical = 2;
      suspend;
    end 
  end
  else
  if (pr_code similar to '%(.express.|.blago.|.uspeh.|.loyal.)%') then
  begin
    if (reg_id in (1120300,1130200,1400200,1420200,1460100,1500100,1120400,1490200,1490300,1510100,1490400)) then
    begin
      res_error_text = 'По указанному продукту запрещено выдавать в вашем офисе!';
      res_error_is_critical = 2;
      suspend;
    end 
  end else
  if (pr_code not containing '.shans.') then
  begin
    res_error_text = 'Выбран не правильный продукт!';
    res_error_is_critical = 2;
    suspend;
  end
end
