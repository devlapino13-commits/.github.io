CREATE PROCEDURE AAREP_CANDIDATES_DISMISSAL(
    REP_REGION_ID INTEGER,
    REP_BONUS_ID INTEGER,
    REP_USER_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    CB_PERSONAL_ID ID_TYPE,
    CB_REGION_ID INTEGER,
    CB_DISB_KOLICH INTEGER,
    CB_DISB_KOLICH_CLI INTEGER,
    CB_ACTIV_KOLICH_CLI INTEGER,
    CB_DELINQ_BALANCE CURR_TYPE,
    CB_ACTIV_BALANCE CURR_TYPE,
    CB_PERSONAL_STAGE DATE,
    CB_PERSONAL_STAGE_MONTH DECIMAL(15,1),
    PAR30 DECIMAL(15,2),
    CB_MEMO_DISB VARCHAR(100),
    CB_MEMO_PAR VARCHAR(100),
    CB_PORTFEL_C CURR_TYPE,
    CB_PORTFEL_D CURR_TYPE,
    CB_IS_WORKING_STR VARCHAR(20),
    OFFICE_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    OKRUG_NAME VARCHAR(50),
    KF_OFFICER_NAME VARCHAR(100))
AS
declare variable rep_date_begin date;
declare variable rep_date_end_2 date;
declare variable rep_date_begin_3 date;
declare variable cb_is_working integer;
declare variable rep_is_hr smallint;
begin

  rep_date_begin = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date); -- Начало периода
  rep_date_end_2 = rep_date_begin - 1; -- Конец предыдущего месяца
  if (:rep_date_end between '02.01.2013' and '31.01.2013') then -- Для того чтобы если за январь2013 считать то за пред-й портфель брал 25.12.2012
  begin
    rep_date_end_2 = '25.12.2012';
    rep_date_begin = rep_date_end_2 + 1;
  end
  rep_date_begin_3 = :rep_date_begin - 32;
  rep_date_begin_3 = cast('01.' || extract(month from :rep_date_begin_3) || '.' || extract(year from :rep_date_begin_3) as date); -- Начало 3 месяц назад
-- Вытаскивает всех СВЗ которые за текущий месяц какое либо операцию делал или на конец даты есть активный портфель
  for with tabl1
      as (select distinct kal_new_kf_officer, kal_new_reg_id
          from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1)
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :rep_date_end between ks_date_begin and ks_date_end and
                ks_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                :rep_date_end between kal_date and kal_date_end and
                kal_new_reg_id = id_list
          union
          select distinct ks_kc_id, ks_region_id
          from kontstate, region_list(:rep_region_id, 1, 3, 1)
          where ks_ol_is_delete = 0 and
                cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
                ks_state in (1, 2) and
                ks_region_id = id_list)
      select distinct kal_new_kf_officer --, kal_new_reg_id
      from tabl1
      into cb_personal_id --, cb_region_id
  do
  begin
--  Просроченный баланс категории 1 от 30 до 180 дней
    select sum(
           case
             when (coalesce((select rep_delinq_days
                             from aarep_kont_delinq_status_new(k_id, :rep_date_end)), 0) between 31 and 180 and (select aa_category_id
                                                                                                                 from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1) then (select sum(ok_os)
                                                                                                                                                                                                                 from operkred
                                                                                                                                                                                                                 where ok_from_k_id = k_id and
                                                                                                                                                                                                                       ok_type in (4, 5, 6, 8) and
                                                                                                                                                                                                                       ok_ol_is_active = 1 and
                                                                                                                                                                                                                       ok_from_c_id = 0 and
                                                                                                                                                                                                                       ok_ol_date <= :rep_date_end)
             else 0
           end)
    from kontstate, kontract, knaktlist
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_new_kf_officer = :cb_personal_id
--    and kal_new_reg_id = :cb_region_id
    into cb_delinq_balance;
    if (cb_delinq_balance is null) then
      cb_delinq_balance = 0;

--  Расчет баланса КС на конец предыдущего месяца по Категории 1; B - общий портфель катег.1 на предыдущую отчетную дату
    select sum(
           case
             when (select aa_category_id
                   from aarep_get_category_id(k_id, :rep_date_end_2, null, null, null, null, null, null)) = 1 then coalesce((select sum(ok_os)
                                                                                                                             from operkred
                                                                                                                             where ok_from_k_id = k_id and
                                                                                                                                   ok_type in (4, 5, 6, 8) and
                                                                                                                                   ok_ol_is_active = 1 and
                                                                                                                                   ok_from_c_id = 0 and
                                                                                                                                   ok_ol_date <= :rep_date_end_2), 0)
             else 0
           end)
    from kontstate, kontract, knaktlist
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end_2 between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end_2 between kal_date and kal_date_end and
          kal_new_kf_officer = :cb_personal_id
--    and kal_new_reg_id = :cb_region_id
    into :cb_activ_balance;
    if (:cb_activ_balance is null) then
      cb_activ_balance = 0;
-- С - Портфель займов, переданный в категорию 1 от другого СВЗ в течение отчетного месяца
    cb_portfel_c = null;
    select sum(
           case
             when (select aa_category_id
                   from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1 then coalesce((select sum(ok_os)
                                                                                                                           from operkred
                                                                                                                           where ok_from_k_id = k_id and
                                                                                                                                 ok_type in (4, 5, 6, 8) and
                                                                                                                                 ok_ol_is_active = 1 and
                                                                                                                                 ok_from_c_id = 0 and
                                                                                                                                 ok_ol_date <= :rep_date_end), 0)
             else 0
           end)
    from kontstate, kontract, knaktlist kal1, knaktlist kal2
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal1.kal_from_kf_id and
          kal1.is_delete = 0 and
          :rep_date_end between kal1.kal_date and kal1.kal_date_end and
          kal1.kal_new_kf_officer = :cb_personal_id
--   and kal1.kal_new_reg_id = :cb_region_id
          and
          k_from_kf_id = kal2.kal_from_kf_id and
          kal2.is_delete = 0 and
          :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
          kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer
    into cb_portfel_c;
    if (cb_portfel_c is null) then
      cb_portfel_c = 0;

-- D - Портфель займов, переданный в категорию 1 от данного СВЗ другому СВЗ в течение отчетного месяца
    cb_portfel_d = null;
    select sum(
           case
             when (select aa_category_id
                   from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) = 1 then coalesce((select sum(ok_os)
                                                                                                                           from operkred
                                                                                                                           where ok_from_k_id = k_id and
                                                                                                                                 ok_type in (4, 5, 6, 8) and
                                                                                                                                 ok_ol_is_active = 1 and
                                                                                                                                 ok_from_c_id = 0 and
                                                                                                                                 ok_ol_date <= :rep_date_end), 0)
             else 0
           end)
    from kontstate, kontract, knaktlist kal1, knaktlist kal2
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal1.kal_from_kf_id and
          kal1.is_delete = 0 and
          :rep_date_end between kal1.kal_date and kal1.kal_date_end
--   and kal1.kal_new_kf_officer = :cb_personal_id
          and
          k_from_kf_id = kal2.kal_from_kf_id and
          kal2.is_delete = 0 and
          :rep_date_begin between kal2.kal_date and kal2.kal_date_end and
          kal2.kal_new_kf_officer <> kal1.kal_new_kf_officer and
          kal2.kal_new_kf_officer = :cb_personal_id
--   and kal2.kal_new_reg_id = :cb_region_id
    into cb_portfel_d;
    if (cb_portfel_d is null) then
      cb_portfel_d = 0;

--  Вычисляет кол-во активных клиентво по категории 1 и 2
    select sum(
           case
             when ((select aa_category_id
                    from aarep_get_category_id(k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2))
--          and (coalesce((select rep_delinq_days from aarep_kont_delinq_status(k_id,:rep_date_end)),0)<=30)) -- (пункт 3.1.1) -- убран после обсуждения с Илоной по скайпу от 13.05.2013
               then k_kolich_client_approved
             else 0
           end)
    from kontstate, kontract, knaktlist
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_new_kf_officer = :cb_personal_id
--    and kal_new_reg_id = :cb_region_id
    into :cb_activ_kolich_cli;
    if (cb_activ_kolich_cli is null) then
      cb_activ_kolich_cli = 0;

--  Вычисляет кол-во выдачи данного СВЗ
    select coalesce(sum(k_kolich_client_approved), 0), count(k_id)
    from kontstate, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :cb_personal_id and
          ks_from_k_id = k_id and
          k_circle <= 1 -- новые выданные кредиты
    into cb_disb_kolich_cli, cb_disb_kolich;

-- Вычисляет стаж КС     .
    cb_personal_stage = null;
    cb_is_working = null;

-- Cтатус на конец даты,  0 - не работает; 1 - работает;
    select ps_is_working, ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :cb_personal_id and
          ps_date_begin is not null and
          ps_status = 1 and
          :rep_date_end between ps_date_begin and ps_date_end
    into cb_is_working, cb_region_id;
    if (cb_is_working is null) then
      cb_is_working = 0;
    if (cb_is_working = 0) then
      cb_is_working_str = 'Уволен.';
    else
      cb_is_working_str = 'Работает.';

-- is_hr = 1; -- 0 - стаж считает с первой выдачи; 1 - стаж берет из трудового договора
    rep_is_hr = 0;
    if (rep_is_hr is null or rep_is_hr not in (0, 1)) then
      rep_is_hr = 0;
    if (rep_is_hr = 0) then
    begin
      select cb_personal_stage
      from cb_personal_data(:cb_region_id, :rep_date_end, :cb_personal_id) -- перед тем приготовить нужно проверить процедуру что он берет стаж с первой выдачи по всей компании
      into cb_personal_stage;
    end
    else
    begin
--    Дата поступления на работу
      select ps_date_begin
      from hr_pstatus --, hr_orders , joblist
      where ps_from_c_id = :cb_personal_id and
            ps_date_begin is not null and
            ps_status = 1
      order by ps_date_begin
      rows 1
      into :cb_personal_stage;
    end

    if (cb_personal_stage is null) then
      cb_personal_stage = :rep_date_end;

    cb_personal_stage_month = datediff(day, :cb_personal_stage, :rep_date_end) / 30;

    if (cb_delinq_balance = 0) then
    begin
      par30 = 0;
    end
    else
    begin
      if (cb_activ_balance + cb_portfel_c - cb_portfel_d > 0) then
      begin
        par30 = cb_delinq_balance / (cb_activ_balance + cb_portfel_c - cb_portfel_d) * 100;
      end
      else
      begin
        par30 = 100;
      end
    end

    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
    from region r1, region r2, region r3
    where r1.reg_unicode = :cb_region_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into office_name, branch_name, okrug_name;

    select c_name_fio
    from clients
    where c_id = :cb_personal_id
    into kf_officer_name;

    cb_memo_disb = null;
    if (cb_personal_stage_month > 1 and
        cb_activ_kolich_cli < 150) then
    begin
      if (cb_disb_kolich < 2) then
        cb_memo_disb = 'Кандидат на увольнение, кол-во новых выданных групп < 2';
    end
    cb_memo_par = null;
    if (par30 > 7.00) then
    begin
      cb_memo_par = 'PAR более 7%, проверьте за предыдущий месяц.';
    end

    suspend;
  end
end
