    UN_ID DECIMAL(18,0))

AS
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer;
declare variable rd_id integer;
declare variable res_period_id integer;
declare variable res_lo_id decimal(18,0);

declare variable ad_active_balance decimal(18,4) = 0;
declare variable ad_diff_client integer;
declare variable ad_delinq_balance_30 decimal(18,4) = 0;
declare variable ad_delinq_balance_diff_30 decimal(18,4) = 0;
declare variable points_1 decimal(18,4) = 0;
declare variable points_2 decimal(18,4) = 0;
declare variable points_all decimal(18,4) = 0;

declare variable cpr_id integer;
declare variable cpr_group_name varchar(255);

declare variable lo_panelty_count integer;
declare variable avg_points decimal(18,4) = 0;
declare variable total_points decimal(18,4) = 0;

declare variable max_points decimal(18,4) = 0;

declare variable kind integer;

declare variable res_qual_koef2 perc_type;

declare variable points_5 curr_type = 0;

declare variable ad_diff_balance curr_type = 0;
declare variable reg_id integer;
declare variable res_team_id decimal(18,0);
declare variable res_total1 integer;
declare variable res_total2 integer;

declare variable ad_disb_loan curr_type;
declare variable svz_portfel_begin curr_type;
declare variable svz_portfel_end curr_type;
declare variable prinyat_portfel curr_type;
begin
-- Расчет итогов олимпиады РФ 2016
-- Stage:
-- 0 - период (месяц)
-- 1 - Промежуточные итоги
-- 2 - Итоги
-- 3 - Финальные результаты

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  if (stage = 0) then -- Если уровень МЕСЯЦ
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

-- Формируем в памяти данные по РФ
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

-- Группируем данные из памяти по ID Региона
    res_team_id_prior = -1;
    for select res_field_1, res_field_2, sum(ad_active_client) ad_active_client,
               sum(ad_active_balance) ad_active_balance, sum(ad_diff_client) ad_diff_client,
               sum(ad_delinq_balance_end3) ad_delinq_balance_end3, sum(ad_delinq_balance_diff3) ad_delinq_balance_diff3,
--           sum(ad_diff_balance) ad_diff_balance,
               sum(ad_delinq_balance_end2) ad_delinq_balance_end2, sum(ad_disb_loan) ad_disb_loan
        from hp_analitica_report('reg_id, reg_name', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360,
--                             -1
        9, 'standart')
        group by 1, 2
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :reg_id, -- ID офиса
             :reg_name, -- наименование офиса
             :ad_active_client, -- Клиенты на конец периода
             :ad_active_balance, -- Баланс на конец периода
             :ad_diff_client, -- Прирост клиентов
             :ad_delinq_balance_30, -- Баланс просрочек 30 дней
             :ad_delinq_balance_diff_30, -- Изменение баланс 30 дней
--         :ad_diff_balance,             -- Изменение активного баланса
             :ad_delinq_balance_7, -- Баланс просрочек 30 дней
             :ad_disb_loan
    do
    begin
--    кол-во СВЗ
      kol_svz = null;
      select count(*)
      from hr_pstatus, joblist, jobkind, clients
      where ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.LO_Standart' -- , 'JobSystem.LO_Partner')
            and
            :res_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_id and
            ps_from_c_id = c_id and
            coalesce((select sawt_id
                      from hr_voctravel, hr_orders, sa_worktype -- убираем декретников и неоплачиваемых отпускников
                      where hrv_from_ho_id = ho_id and
                            ho_order_type = sawt_id and
                            hrv_from_c_id = ps_from_c_id and
                            :res_date_end between hrv_date_begin and hrv_date_end_fact
                      order by hrv_date_begin desc
                      rows 1), 0) not in (16, 17, 18, 21) -- Убрал 8 - неоплачиваемый отпуск
            and
            exists(select *
                   from oksummary -- берем только тех у кого на конец имеется активный портфель
                   where oks_kc_id = ps_from_c_id and
                         oks_region_id = ps_from_reg_id and
                         :res_date_end between oks_date_begin and oks_date_end and
                         oks_balance > 0 and
                         oks_close_state = 0)
      into kol_svz;
      kol_svz = coalesce(kol_svz, 0);

      if (kol_svz = 0) then
        kol_svz = 1; -- если в офисе не имеется СВЗ то будем считать 1 СВЗ
--    Разработка по рейтингу
--    Вычисляем балы
      points_1 = 0.0;
      points_2 = 0.0;
      points_3 = 0.0;
      points_4 = 0.0;
      points_5 = 0.0;
      points_6 = 0.0;

      /*
--    4. Изменение баланса просрочек в среднем на 1 СВЗ
--    при условии баланс просрочки cнижен не менее чем 5000 руб
      if (ad_delinq_balance_diff_30 <= -5000.0
          and cast(ad_delinq_balance_diff_30 as double precision) / kol_svz <= -1000.0) then
      begin
--      4.1 снижен баланс просрочки на 1000 руб и более на 1 СВЗ за отчетный месяц
        points_4 = 10.0;
      end else
      if (ad_delinq_balance_7 * 100.00 / ad_active_balance < 7.00
          or
          ad_delinq_balance_diff_30 <= 0.0 -- еслин не было увеличение то бал должен = 0; добавлено от 11.04.2017 письмо от Ироды
           ) then
      begin
--      4.2 если пункт 4.1 не выполнено но PAR7 < 7%
        points_4 = 0;
      end else
      begin
--      4.3 если не выполнились пункты 4.1 и 4.2
        points_4 = -10.0;
      end 
      */

--    4. Изменение баланса просрочек в среднем на 1 СВЗ
--    4.1 Снижен баланс просрочек продолжительностью более 30 дней  на 10 000 руб. и более на 1 СВЗ за отчетный месяц
      if (cast(ad_delinq_balance_diff_30 as double precision) / kol_svz <= -10000.0) then
      begin
        points_4 = 15.0;
      end
      else

--      4.2 Снижен баланс просрочек продолжительностью более 30 дней  на 1000 и более рублей, но менее 10 000 руб.
--      на 1 СВЗ за отчетный месяц (баллы начисляются при условии, что за отчетный период баланс просрочек офиса снижен не менее чем на 5000 руб.)
      if (cast(ad_delinq_balance_diff_30 as double precision) / kol_svz <= -1000.0 and
          ad_delinq_balance_diff_30 <= -5000.0) then
      begin
        points_4 = 7.0;
      end
      else
      if (ad_delinq_balance_7 * 100.00 / ad_active_balance < 7.00 -- но PAR7 офиса < 7 %;
          and
          (cast(ad_delinq_balance_diff_30 as double precision) / kol_svz > -1000.0 -- баланс просрочек в среднем на 1 СВЗ снизился менее чем на 1000 руб.;
          or cast(ad_delinq_balance_diff_30 as double precision) / kol_svz > 0.0 -- Баланс просрочек в среднем на 1 СВЗ увеличился на любую сумму.;
          or ad_delinq_balance_diff_30 <= -5000.0 -- Общий баланс просрочек офиса снизился менее чем на 5000 руб
          )) then
      begin
        points_4 = 0;
      end
      else
      if (cast(ad_delinq_balance_diff_30 as double precision) / kol_svz <= 30000.0) then
      begin
--      4.4 баланс просрочек продолжительностью более 30 дней увеличился, но не более чем на 30 000  в среднем на 1 СВЗ за отчетный месяц
        points_4 = -7.0;
      end
      else
      begin
--      4.5 баланс просрочек продолжительностью более 30 дней увеличился более чем на 30 000  в среднем на 1 СВЗ за отчетный месяц
        points_4 = -15.0;
      end
      /*
--    5. Прирост активных клиентов в среднем на 1 СВЗ
      if (ad_diff_client >= 20 and cast(ad_diff_client as double precision) / kol_svz >= 4.0) then
      begin
--      5.1 прирост на 4 кл. на 1 СВЗ
        points_5 = 20.0;
      end else
      if (ad_diff_client >= 5 and cast(ad_diff_client as double precision) / kol_svz >= 1.0) then
      begin
--      5.2 прирост на 1 кл. на 1 СВЗ
        points_5 = 8.0;
      end else
      if (ad_diff_client >= 0) then
      begin
--      5.3 если условии 5.1 и 5.2 не выполнено но сделано прирост
        points_5 = 0.0;
      end else
      begin
--      5.4 если не выполнились условии 5.1-5.3 и снижен хотябы на одного клиента
        points_5 = -8.0;
      end 
      */
--    5. Среднее количество выдач на 1 СВЗ офиса
      if (cast(ad_disb_loan as double precision) / kol_svz >= 35.0) then
      begin
--      5.1 Среднее количество выдач на 1 СВЗ ? 35 займов
        points_5 = 20.0;
      end
      else
      if (cast(ad_disb_loan as double precision) / kol_svz >= 30.0) then
      begin
--      5.2 Среднее количество выдач на 1 СВЗ ? 30, но < 35
        points_5 = 8.0;
      end
      else
      if (cast(ad_disb_loan as double precision) / kol_svz >= 25.0) then
      begin
--      5.3 периодСреднее количество выдач на 1 СВЗ ? 25, но < 30
        points_5 = 0.0;
      end
      else
      begin
--      5.4 Среднее количество выдач < 25
        points_5 = -8.0;
      end

--------------  Добавлено от 09.02.2018 По запросу от Ироды, расчет прироста такой же как и в олим играх
      svz_portfel_begin = null;
      select sum(oks_balance)
      from oksummary
      where :res_date_begin - 1 between oks_date_begin and oks_date_end and
            :res_date_begin - 1 - oks_delinq_date <= 360 and
            oks_region_id = :reg_id and
            oks_close_state = 0
      into svz_portfel_begin;
      if (svz_portfel_begin is null) then
        svz_portfel_begin = 0;

      svz_portfel_end = null;
      select sum(oks_balance)
      from oksummary
      where :res_date_end between oks_date_begin and oks_date_end and
            :res_date_end - oks_delinq_date <= 360 and
            oks_region_id = :reg_id and
            oks_close_state = 0
      into svz_portfel_end;
      if (svz_portfel_end is null) then
        svz_portfel_end = 0;

      prinyat_portfel = null;
      select sum(oks_balance)
      from knaktlist, kontract, oksummary
      where knaktlist.is_delete = 0 and
            kal_date between :res_date_begin and :res_date_end and
            kal_from_ka_id > 0 and
            kal_new_reg_id <> kal_prior_reg_id and
            kal_new_reg_id = :reg_id and
            kal_from_kf_id = k_from_kf_id and
            k_id = oks_from_k_id and
            :res_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_region_id = :reg_id and
            oks_close_state = 0
      into prinyat_portfel;
      if (prinyat_portfel is null) then
        prinyat_portfel = 0;

      ad_diff_balance = svz_portfel_end - svz_portfel_begin - prinyat_portfel;

--------------

--    6. Прирост портфеля в среднем на 1 СВЗ
      if (ad_diff_balance >= 500000.0 and
          cast(ad_diff_balance as double precision) / kol_svz >= 100000.0) then
      begin
--      6.1 прирост на 100 тыс. руб на 1 СВЗ
        points_6 = 25.0;
      end
      else
      if (ad_diff_balance >= 200000.0 and
          cast(ad_diff_balance as double precision) / kol_svz >= 40000.0) then
      begin
--      6.2 прирост на 40 тыс. руб на 1 СВЗ
        points_6 = 12.0;
      end
      else
      if (ad_diff_balance >= 0.0) then
      begin
--      6.3 Если условие 6.1-6.2 не выполнено но сделан общий прирост
        points_6 = 0;
      end
      else
      begin
--      6.4 Если условии 6.1-6.3 не выполнены
        points_6 = -12.00;
      end

      group_name = '';
      if (reg_id in (1010300, 1010800, 1011900, 1020200, 1031300, 1031500, 1040100, 1040900, 1041100, 1050100,
                     1050300, 1060100, 1190100, 1200100, 1012400, 1012500, 1041500)) then
      begin
--        cpr_group_name = 'A '||reg_name;
        group_name = 'A';
      end
      else
      if (reg_id in (1041400, 1080100, 1090100, 1110100, 1120100, 1120200, 1130100, 1140100, 1150100, 1160100,
                     1170100, 1180100, 1220100, 1230100, 1240100, 1260100, 1270100, 1280100, 1300100, 1360100,
                     1380100, 1400100, 1420100, 1430100, 1440100, 1450100, 1060200, 1470100, 1480100, 1490100)) then
      begin
--        cpr_group_name = 'B '||reg_name;
        group_name = 'B';
      end
      cpr_group_name = group_name || ' ' || reg_name;

--    Вытаскиваем МО данного офиса
      res_team_id = null;
      for select ps_from_c_id
          from hr_pstatus, joblist, jobkind
          where ps_status = 1 and
                ps_is_working = 1 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                (jk_system_code = 'JobSystem.Region_KM')
--        or ps_from_c_id=1010100148288) -- Добавлен Артиков Самад как испольняющий МО
                and
                :res_date_end between ps_date_begin and ps_date_end and
                ps_from_reg_id = :reg_id and
                coalesce((select sawt_id
                          from hr_voctravel, hr_orders, sa_worktype
                          where hrv_from_ho_id = ho_id and
                                ho_order_type = sawt_id and
                                hrv_from_c_id = ps_from_c_id and
                                :res_date_end between hrv_date_begin and hrv_date_end_fact
                          order by hrv_date_begin desc
                          rows 1), 0) not in (16, 17, 18, 21) -- Декретников и не оплачиваемых отпускников не включает  -- убрал 8 - неоплачиваемый отпус от 11 мая 2018 Улан по запросу Ироды
                and
                jl_name <> 'И.о. заместителя менеджера офиса'
          into :res_team_id -- ID МО
--        , :cpr_group_name  -- ФИО МО
      do
      begin
        if (exists(select *
                   from hr_pstatus
                   where ps_status = 1 and
                         ps_is_working = 1 and
                         :res_date_end + 8 between ps_date_begin and ps_date_end -- добавлено +8 дней чтобы уволенных на момент формирование не показывало письмо от 10.10.2016 Ирода
                         and
                         ps_from_c_id = :res_team_id)) then
        begin
          cpr_id = null;
          select cpr_id
          from competresult
          where cpr_from_cpp_id = :cpp_id and
                cpr_object_kind = 0 and
                cpr_object_id = :res_team_id
          into :cpr_id;

          if (cpr_id is null) then -- если такой строки нет, создаем новую
          begin
            insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                      cpr_from_un_id)
            values (:cpp_id, :cpr_group_name, 0, :res_team_id, 0, :un_id)
            returning cpr_id
            into :cpr_id;
          end

--      вычисляем коэффициент Качество принемаемых решений МО
          res_qual_koef1 = null;
          res_qual_koef2 = null;
          res_qual_koef3 = null;
          res_total1 = 0;
          res_total2 = 0;
          res_total3 = 0;
          select
--                 round(res1_qual_koef1,2), -- Качество принемаемых решений МО 1
                 round(res_qual_koef1, 2), -- Качество принемаемых решений МО 1
                 round(res_qual_koef2, 2), -- Качество принемаемых решений МО 2
                 round(res_qual_koef3, 2), -- Качество принемаемых решений МО 3
--                 res1_total1,
                 res_total1, res_total2, res_total3
          from aarep_disb_manager_quality(:reg_id, :res_date_begin, :res_date_end, :res_team_id)
          into res_qual_koef1, res_qual_koef2, res_qual_koef3, res_total1, res_total2, res_total3;
          res_qual_koef1 = coalesce(res_qual_koef1, 100);
          res_qual_koef2 = coalesce(res_qual_koef2, 100);
          res_qual_koef3 = coalesce(res_qual_koef3, 100);
          res_total1 = coalesce(res_total1, 0);
          res_total2 = coalesce(res_total2, 0);
          res_total3 = coalesce(res_total3, 0);

          cpd_comment = '';
          cpd_comment = cpd_comment || 'Актив.кл: <b>' || round(ad_active_client) || '</b><br>';
--          cpd_comment=cpd_comment||'Прирост кл: <b>'||round(ad_diff_client)||'</b><br>';
          cpd_comment = cpd_comment || 'Кол-во выдач: <b>' || round(ad_disb_loan) || '</b><br>';

          cpd_comment = cpd_comment || 'PAR30: <b>' || round(ad_delinq_balance_30 * 100.00 / ad_active_balance, 2) || '%</b><br>';
          cpd_comment = cpd_comment || 'Изменение PAR30: <b>' || round(ad_delinq_balance_diff_30, 2) || ' р.</b><br>';

          cpd_comment = cpd_comment || 'PAR7: <b>' || round(ad_delinq_balance_7 * 100.00 / ad_active_balance, 2) || '%</b><br>';

          cpd_comment = cpd_comment || 'Актив. портфель: <b>' || round(ad_active_balance, 2) || ' р.</b><br>';
          cpd_comment = cpd_comment || 'Изменение портфель: <b>' || round(ad_diff_balance, 2) || ' р.</b><br>';
          cpd_comment = cpd_comment || 'Кол-во СВЗ: <b>' || round(kol_svz) || '</b><br>';

          if (res_total1 > 0) then
            cpd_comment = cpd_comment || 'Качество по 1му погашению: <b>' || round(res_qual_koef1, 2) || '</b><br>';
          else
            cpd_comment = cpd_comment || 'Качество по 1 му погашению: <b>' || 'Кредитов не утверждал.' || '</b><br>';
          if (res_total2 > 0) then
            cpd_comment = cpd_comment || 'Качество по 2му погашению: <b>' || round(res_qual_koef2, 2) || '</b><br>';
          else
            cpd_comment = cpd_comment || 'Качество по 2му погашению: <b>' || 'Кредитов не утверждал.' || '</b><br>';
          if (res_total3 > 0) then
            cpd_comment = cpd_comment || 'Качество по 3му погашению: <b>' || round(res_qual_koef3, 2) || '</b><br>';
          else
            cpd_comment = cpd_comment || 'Качество по 3му погашению: <b>' || 'Кредитов не утверждал.' || '</b><br>';

          cpd_comment = cpd_comment || 'Кол-во принятых реш. с 1 пог.: <b>' || res_total1 || '</b><br>';
          cpd_comment = cpd_comment || 'Кол-во принятых реш. с 2 пог.: <b>' || res_total2 || '</b><br>';
          cpd_comment = cpd_comment || 'Кол-во принятых реш. с 3 пог.: <b>' || res_total3 || '</b><br>';

--     Вычисляем балы
--     1. Качество принемаемых решений МО по первому погашению
          if (res_total1 > 0) then
          begin
            points_1 = case
                         when res_qual_koef1 > 99.0 then 20
                         when res_qual_koef1 > 98.0 then 10
                         when res_qual_koef1 > 97.0 then 5
                         when res_qual_koef1 >= 95.0 then 0
                         else -5
                       end;
          end
          else
          begin
            points_1 = 0;
            res_qual_koef1 = 0;
          end

--     2. Качество принемаемых решений МО по второму погашению
          if (res_total2 > 0) then
          begin
            points_2 = case
                         when res_qual_koef2 > 99.0 then 25
                         when res_qual_koef2 > 98.0 then 15
                         when res_qual_koef2 > 96.0 then 5
                         when res_qual_koef2 >= 94.0 then 0
                         else -5
                       end;
          end
          else
          begin
            points_2 = 0;
            res_qual_koef2 = 0;
          end

--     3. Качество принемаемых решений МО по первому погашению
          if (res_total3 > 0) then
          begin
            points_3 = case
                         when res_qual_koef3 > 98.0 then 30
                         when res_qual_koef3 > 97.0 then 20
                         when res_qual_koef3 > 95.0 then 5
                         when res_qual_koef3 >= 93.0 then 0
                         else -10
                       end;
          end
          else
          begin
            points_3 = 0;
            res_qual_koef3 = 0;
          end

--       Начисленные положительные балы за качество МО увеличивается на 25%
--       если кол-во принятых решений за отчетный месяц:
--       более 250 для групп А
--       более 100 для групп Б
          if (group_name = 'A') then
          begin
            if (res_total1 > 250 and
                points_1 > 0) then
              points_1 = points_1 * 1.25;
            if (res_total2 > 250 and
                points_2 > 0) then
              points_2 = points_2 * 1.25;
            if (res_total3 > 250 and
                points_3 > 0) then
              points_3 = points_3 * 1.25;
          end
          else
          if (group_name = 'B') then
          begin
            if (res_total1 > 100 and
                points_1 > 0) then
              points_1 = points_1 * 1.25;
            if (res_total2 > 100 and
                points_2 > 0) then
              points_2 = points_2 * 1.25;
            if (res_total3 > 100 and
                points_3 > 0) then
              points_3 = points_3 * 1.25;
          end

          points_all = :points_1 + :points_2 + :points_3 + :points_4 + :points_5 + :points_6;
--      Вставляем данные по МО

          insert into competdetails (cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, cpd_points_1,
                                     cpd_points_2, cpd_points_3, cpd_points_4, cpd_points_5, cpd_points_6, cpd_comment,
                                     cpd_add_data_1, cpd_add_data_2, cpd_add_data_3, cpd_add_data_4, cpd_add_data_5,
                                     cpd_add_data_6, cpd_add_data_7, cpd_add_data_8, cpd_add_data_9)
          values (:cpr_id, 0, :res_team_id, :points_all, :points_1, :points_2, :points_3, :points_4, :points_5,
                  :points_6, left(:cpd_comment, 4032),
--                  round(:ad_active_client),
                  round(:ad_disb_loan), round(:ad_delinq_balance_7 * 100.00 / :ad_active_balance, 2),
                  round(:ad_delinq_balance_diff_30, 2), round(:ad_active_balance, 2), round(:ad_diff_balance, 2),
                  round(:kol_svz), round(:res_qual_koef1, 2), round(:res_qual_koef2, 2), round(:res_qual_koef3, 2));
        end
      end
    end

-- Пересчитываем итоги по командам
    for select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id
        into :cpr_id
    do
    begin
      total_points = 0.0;
      select round(cpd_points_all, 2)
      from competdetails
      where cpd_from_cpr_id = :cpr_id
      into total_points;

      if (total_points is null) then
        total_points = 0;

      update competresult
      set cpr_calced_points = :total_points
      where (cpr_id = :cpr_id);
    end

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  suspend;
end
