CREATE PROCEDURE HP_BIOGUARD_SAVE_ADDRESS(
    ADL_TYPE D_INTEGER,
    C_ID D_ID,
    COUNTRY_CODE D_STR_100,
    COUNTRY_TXT D_STR_255,
    DISTRICT_CODE D_STR_20,
    DISTRICT_TXT D_STR_255,
    REGION_CODE D_STR_100,
    REGION_TXT D_STR_255,
    CITY_TXT D_STR_255,
    STREET_TXT D_STR_255,
    BUILDING_TXT D_STR_255,
    FLAT_TXT D_STR_255)
AS
declare variable ADL_ID D_ID;
declare variable PL_COUNTRY_ID D_ID;
declare variable PL_DISTRICT_ID D_ID;
declare variable PL_REGION_ID D_ID;
declare variable PL_CITY_ID D_ID;
declare variable PL_STREET_ID D_ID;
begin
    country_txt = coalesce(:country_txt, 'Казахстан');
    pl_country_id = 398000000001; -- ID Казахстана

    -- Вытаскиваем информацию об области
    -- Не создаём новых областей или городов республиканского значения
    -- 398002 - Область         398006 - Города (Если родительский объект - Казахстан, тогда это город республиканского значения. К примеру Алматы, Астана, Шымкент)
    pl_district_id = 0;

    -- RFCKZ-127
    select pl.pl_id
    from places pl
    where pl.pl_status in ('398002', '398006')
        and pl.parent = :pl_country_id
        and pl.pl_name = :district_txt
        and pl.is_delete = 0
    into :pl_district_id;



    -- Вытаскиваем информацию о районе
    -- 398003 - Район         398004 - Областной акимат         0 - Объект добавленный в процессе (в данном случае критичен parent и pl_name)
    pl_region_id = 0;

    select first 1 pl.pl_id
    from places pl
    where pl.pl_status in ('398003', '398004', '0')
        and pl.parent = :pl_district_id
        and pl.pl_name = :region_txt
        and pl.is_delete = 0
    order by pl.pl_id asc
    into pl_region_id;

    -- Создаю новую запись
    if (pl_region_id = 0 and coalesce(:region_txt, '') <> '') then
        select pl_id
        from hp_insert_place_node(:pl_district_id, :region_txt, 0, 1000000000000, null, 1)
        into pl_region_id;



    -- Вытаскиваем информацию о городе
    -- 398006 - Города          0 - Объект добавленный в процессе (в данном случае критичен parent и pl_name)
    pl_city_id = 0;

    select first 1 pl.pl_id
    from places pl
    where pl.pl_status in ('398006', '0')
        and pl.parent = :pl_region_id
        and pl.pl_name = :city_txt
        and pl.is_delete = 0
    order by pl.pl_id asc
    into :pl_city_id;

    if (pl_city_id = 0 and coalesce(:city_txt, '') <> '') then
        select pl_id
        from hp_insert_place_node(:pl_region_id, :city_txt, 0, 1000000000000, null, 1)
        into pl_city_id;
    


    -- Вытаскиваем информацию об улице
    -- 0 - Объект добавленный в процессе (в данном случае критичен parent и pl_name)
    pl_street_id = 0;

    select first 1 pl.pl_id
    from places pl
    where pl.pl_status in ('0')
        and pl.parent = coalesce(nullif(:pl_city_id, 0), :pl_district_id)
        and pl.pl_name = :street_txt
        and pl.is_delete = 0
    order by pl.pl_id asc
    into pl_street_id;

    if (pl_street_id = 0 and coalesce(:street_txt, '') <> '') then
        select pl_id
        from hp_insert_place_node(coalesce(nullif(:pl_city_id, 0), :pl_district_id), :street_txt, 0, 1000000000000, null, 1)
        into pl_street_id;



    -- Сохраняем адрес клиента
    street_txt = f_format('%1 %2 %3 %4 %5', :country_txt, :district_txt, :region_txt, :city_txt, :street_txt);
    street_txt = f_format('%1 %2 %3', :street_txt, :building_txt, :flat_txt);
    insert into addrlist
    (
        adl_from_pl_id,
        adl_flat_no,
        adl_building_no,
        adl_address_str,
        adl_street_id
    )
    values
    (
        :pl_district_id,
        :flat_txt,
        :building_txt,
        :street_txt,
        :pl_street_id
    )
    returning adl_id
    into :adl_id;
    
    update or insert into addrclient
    (
        adc_from_c_id,
        adc_address_type,
        adc_from_adl_id,
        adc_from_un_id
    )
    values
    (
        :c_id,
        :adl_type,
        :adl_id,
        1000000000000
    )
    matching (adc_from_c_id, adc_address_type);

end
