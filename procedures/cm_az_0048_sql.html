CREATE PROCEDURE CM_AZ_0048_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable c_id id_type;
declare variable kc_circle integer;
declare variable kol_cli integer;
declare variable pr_code varchar(100);
declare variable kol_actiya integer;
declare variable ma_percent perc_type;
begin
  for select ma_percent
      from miniapp, product
      where ma_id = :ma_id and
            ma_is_active = 1 and
            ma_date_end = '31.12.9999' and
            ma_from_pr_id = pr_id and
            pr_code containing '.akcia.'
      into :ma_percent
  do
  begin
    for select mac_from_c_id, mac_amount
        from miniappclient
        where mac_from_ma_id = :ma_id and
              mac_amount > 0
        into :c_id, :kc_amount
    do
    begin
      kc_circle = 0;
      select count(*)
      from knclient, kontstate
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1
      into kc_circle;
      kc_circle = coalesce(kc_circle, 0) + 1; -- +1 tekushiy
    end
    if ((kc_circle = 2 and kc_amount > 1000) or
        (kc_circle = 2 and ma_percent <> 45)) then
    begin
      /*Комментарий: “Для клиентов в 2-ом цикле % ставка должна быть 37%.”                            
      Сообщение: “2-ci dövr müştərilər üçün illik % dərəcəsi 37% olmalidir.”*/
      res_error_text = 'İkinci dövrdə kreditin faizi 45% olmalidir, maksimal məbləğ 1000 manat.';
      res_error_is_critical = 2;
      suspend;
    end
    if ((kc_circle = 3 and kc_amount > 1500) or
        (kc_circle = 3 and ma_percent <> 42)) then
    begin
      /*Комментарий: “Для клиентов в 3-ем цикле % ставка должна быть 36%.”
        Сообщение: “3-ci dövr müştərilər üçün illik % dərəcəsi 36% olmalidir.”*/
      res_error_text = 'Üçüncü dövrdə kreditin faizi 42% olmalidir, maksimal məbləğ 1500 manat.';
      res_error_is_critical = 2;
      suspend;
    end
    if ((kc_circle >= 4 and kc_amount > 2000) or
        (kc_circle >= 4 and ma_percent <> 39)) then
    begin
      /*Комментарий: “Для клиентов в 4-ом или большем цикле % ставка должна быть 35%.”
        Сообщение: “4-cü və daha çox dövr müştərilər üçün illik % dərəcəsi 35% olmalidir.”*/
      res_error_text = 'dörd və daha artiq dövrlərdə kreditin faizi 39% olmalidir, maksimal məbləğ 2000 manat.';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
