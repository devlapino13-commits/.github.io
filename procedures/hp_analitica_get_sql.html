CREATE PROCEDURE HP_ANALITICA_GET_SQL(
    REP_SELECTED_FIELDS VARCHAR(4096),
    REP_REPORT_KIND VARCHAR(100),
    MAX_DELINQ_DAYS INTEGER,
    MAX_DELINQ_DEVIDE INTEGER,
    REP_KIND VARCHAR(100) = '')
RETURNS (
    SQL_SELECT BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    SQL_FROM BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    SQL_WHERE BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    SQL_GROUP BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable n integer;
declare variable field_count integer;
declare variable f_name varchar(255);
declare variable af_sql_field varchar(2048);
declare variable af_sql_from varchar(2048);
declare variable af_sql_where varchar(2048);
declare variable af_name varchar(255);
declare variable union_count integer;
declare variable k integer;
declare variable sfrom varchar(255);
declare variable sql_sub_where varchar(2048);
declare variable sql_sub_select varchar(2048);
declare variable group_sub_count integer;
declare variable period_txt varchar(100);
declare variable rep_selected_orig varchar(4096);
declare variable ap_id integer;
begin
--  rep_report_kind:
--    tree_group - вложенные группировки
--    flat_group - группировка по каждому полю отдельно, объединяются через Union

  if (rep_selected_fields is null) then
    rep_selected_fields = '';
  if (rep_report_kind = '') then
    rep_report_kind = 'tree_group';
  if (rep_kind = '') then
    rep_kind = 'standart';
  rep_selected_orig = rep_selected_fields;

  sql_sub_where = '';
  sql_sub_select = '';
  group_sub_count = 0;

--  max_delinq_devide: 1  Показать выше
--  max_delinq_devide: -1 Показать ниже
--  max_delinq_devide: 0 Показать без делений

  for select ap_id,
             '''' || right(100 + ap_orderno, 2) || ') ' || right(100 + extract(day from ap_date_end), 2) || '.' || right(100 + extract(month from ap_date_end), 2) || '.' || extract(year from ap_date_end) || ''''
      from temp_analitica_period
      order by ap_orderno desc
      into ap_id, :period_txt
  do
  begin
    rep_selected_fields = rep_selected_orig;
    sql_sub_where = '';
    group_sub_count = 1;
    if (max_delinq_devide = 1) then
      sql_sub_where = ' where ((ad_begin_delinq_days>' || :max_delinq_days || ') or (ad_delinq_days>' || :max_delinq_days || '))';
    if (max_delinq_devide = -1) then
      sql_sub_where = ' where ((ad_begin_delinq_days<=' || :max_delinq_days || ') or (ad_delinq_days<=' || :max_delinq_days || '))';
    sql_sub_select = 'cast(' || period_txt || ' as str_100)' || ', ';

    if (sql_sub_where = '') then
      sql_sub_where = ' where (ad_from_ap_id=' || ap_id || ')';
    else
      sql_sub_where = sql_sub_where || ' and (ad_from_ap_id=' || ap_id || ')';

    if (rep_kind = 'delinq') then
    begin
      if (sql_sub_where = '') then
        sql_sub_where = ' where (ad_id=aq_from_ad_id)';
      else
        sql_sub_where = sql_sub_where || ' and (ad_id=aq_from_ad_id)';
    end

    af_name = '';
    union_count = 0;
    sql_select = sql_sub_select;
    field_count = group_sub_count;

    if (rep_report_kind = 'tree_group' or rep_selected_fields = '') then
    begin
      if (rep_kind = 'delinq') then
        sql_from = 'from [temp_analitica_data],[temp_analitica_delinq]';
      if (rep_kind = 'standart') then
        sql_from = 'from [temp_analitica_data]';
      sql_where = sql_sub_where;
      sql_group = 'group by 1';

      if (rep_selected_fields <> '') then
      begin
        while (rep_selected_fields <> '') do
        begin
          f_name = '';
          n = position(',', rep_selected_fields, 1);
          if (n > 0) then
          begin
            f_name = left(rep_selected_fields, n - 1);
            rep_selected_fields = trim(substring(rep_selected_fields from n + 1));
          end
          else
          begin
            f_name = rep_selected_fields;
            rep_selected_fields = '';
          end
          if (f_name <> '') then
          begin
            af_sql_field = '';
            af_sql_from = '';
            af_sql_where = '';

            select af_sql_field, af_sql_from, af_sql_where
            from hp_analitica_get_fields
            where af_code = :f_name
            into af_sql_field, :af_sql_from, :af_sql_where;

-- Если f_name начинается на select, то заполняем дополнительные поля
            if ((trim(af_sql_field) starting 'select ') or (trim(af_sql_field) starting '(select ')) then
            begin
              select res_field_name
              from hp_analitica_fill_fields(:f_name, :af_sql_field)
              into af_sql_field;
            end

            if (af_sql_field <> '') then
            begin
              field_count = field_count + 1;
              sql_select = BLOB_APPEND(sql_select, ' ' || af_sql_field || ', ');
            end
            if (af_sql_from <> '') then
            begin
              k = position(',', af_sql_from);
              while (k > 0) do
              begin
                sfrom = trim(left(af_sql_from, k - 1));
                af_sql_from = trim(substring(af_sql_from from k + 1 for 255));
                if (position('[' || sfrom || ']', sql_from) = 0) then
                begin
                  sql_from = BLOB_APPEND(sql_from, ', [' || sfrom || ']');
                end
                k = position(',', af_sql_from);
              end
              if ((af_sql_from <> '') and
                  (position('[' || af_sql_from || ']', sql_from) = 0)) then
              begin
                sql_from = BLOB_APPEND(sql_from, ', [' || af_sql_from || ']');
              end
            end
            if (af_sql_where <> '') then
            begin
              if (sql_where = '') then
                sql_where = 'where (' || af_sql_where || ') ';
              else
              begin
                if (position('(' || af_sql_where || ')', sql_where) = 0) then
                  sql_where = BLOB_APPEND(sql_where, ' and (' || af_sql_where || ') ');
              end
            end
            if (af_sql_field <> '') then
            begin
              if (sql_group = '') then
                sql_group = 'group by ' || field_count;
              else
                sql_group = BLOB_APPEND(sql_group, ', ' || field_count);
            end
          end
        end
      end
      else
      begin
        if (rep_kind = 'delinq') then
          sql_from = 'from [temp_analitica_data],[temp_analitica_delinq]';
        if (rep_kind = 'standart') then
          sql_from = 'from [temp_analitica_data]';
        sql_select = BLOB_APPEND(sql_select, ' ''Всего'',');
        sql_where = sql_sub_where;
        sql_group = 'group by 1, 2';
        field_count = 2;
      end
      while (field_count < 8) do
      begin
        field_count = field_count + 1;
        sql_select = BLOB_APPEND(sql_select, ' '''',');
      end

      sql_from = replace(sql_from, '[', '');
      sql_from = replace(sql_from, ']', '');
      suspend;
    end
    if (rep_report_kind = 'flat_group') then
    begin
      while (rep_selected_fields <> '') do
      begin
        union_count = union_count + 1;
        if (rep_kind = 'delinq') then
          sql_from = 'from [temp_analitica_data],[temp_analitica_delinq]';
        if (rep_kind = 'standart') then
          sql_from = 'from [temp_analitica_data]';
        sql_where = sql_sub_where;
        sql_group = '';
        field_count = 0;

        f_name = '';
        n = position(',', rep_selected_fields, 1);
        if (n > 0) then
        begin
          f_name = left(rep_selected_fields, n - 1);
          rep_selected_fields = trim(substring(rep_selected_fields from n + 1));
        end
        else
        begin
          f_name = rep_selected_fields;
          rep_selected_fields = '';
        end
        if (f_name <> '') then
        begin
          af_sql_field = '';
          af_sql_from = '';
          af_sql_where = '';
          sql_select = sql_sub_select;

          select af_sql_field, af_sql_from, af_sql_where, af_name
          from hp_analitica_get_fields
          where af_code = :f_name
          into af_sql_field, :af_sql_from, :af_sql_where, af_name;

          field_count = 2;
          sql_select = BLOB_APPEND(sql_select, ' ''' || right(100 + union_count, 2) || '. ' || af_name || '''' || ', ');
          sql_group = 'group by 1, 2';

-- Если f_name начинается на select, то заполняем дополнительные поля
          if ((trim(af_sql_field) starting 'select ') or (trim(af_sql_field) starting '(select ')) then
          begin
            select res_field_name
            from hp_analitica_fill_fields(:f_name, :af_sql_field)
            into af_sql_field;
          end

          if (af_sql_field <> '') then
          begin
            field_count = field_count + 1;
            sql_select = BLOB_APPEND(sql_select, ' ' || af_sql_field || ', ');
          end
          if (af_sql_from <> '') then
          begin
            if (position('[' || af_sql_from || ']', sql_from) = 0) then
            begin
              sql_from = BLOB_APPEND(sql_from, ', [' || af_sql_from || ']');
            end
          end
          if (af_sql_where <> '') then
          begin
            if (sql_where = '') then
              sql_where = 'where (' || af_sql_where || ') ';
            else
            begin
              if (position('(' || af_sql_where || ')', sql_where) = 0) then
                sql_where = BLOB_APPEND(sql_where, ' and (' || af_sql_where || ') ');
            end
          end
          if (af_sql_field <> '') then
          begin
            if (sql_group = '') then
              sql_group = 'group by ' || field_count;
            else
              sql_group = BLOB_APPEND(sql_group, ', ' || field_count);
          end
        end
        while (field_count < 8) do
        begin
          field_count = field_count + 1;
          sql_select = BLOB_APPEND(sql_select, ' '''',');
        end

        sql_from = replace(sql_from, '[', '');
        sql_from = replace(sql_from, ']', '');
        suspend;
      end
    end
  end
end
