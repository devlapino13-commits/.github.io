CREATE PROCEDURE FM_HTML_CONDITIONS_LIST(
    OBJ_ID D_ID,
    OBJ_KIND D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable k_id d_id;
declare variable err d_integer;
declare variable tmp_str d_str_100;
declare variable res_field_name d_str_255;
declare variable res_field_values blob sub_type 1 segment size 80;
declare variable res_order d_integer;
declare variable db_id d_id;
declare variable color_green d_str_100;
declare variable color_red d_str_100;
declare variable lang d_str_lang;
declare variable ma_id d_id;
begin
  db_id = f_get_db_id();
  color_green = '#02b316';
  color_red = 'red';
  lang = f_get_lang_code(:un_id, null);

  if (obj_kind = f_objkind('miniapp', 'id')) then
  begin
    ma_id = :obj_id;
  end 
  else
  if (obj_kind = f_objkind('kontract', 'id')) then
  begin
    k_id = :obj_id;
  end

  res_html = (select res_style
              from fm_html_styles) || '<body>';
  res_html = BLOB_APPEND(res_html, f_format('<h2>%1</h2>', f_lang(:lang, 'Результат проверки')));
  res_html = BLOB_APPEND(res_html, '<table width="100%"><col width="40%"><col width="60%">');

  err = 0;
  tmp_str = '.check_preview.';

  if (ma_id > 0) then
    tmp_str = '.miniapp.';

  for select kf_client_fio,
             kf_error_text
      from hp_check_loan_konditions(:k_id, 1000000, :ma_id, :tmp_str, :un_id)
      where kf_error_is_critical >= 1
      into res_field_name,
           res_field_values
  do
  begin
    if (res_field_name is null) then
      res_field_name = '';

    if (res_field_values is null) then
      res_field_values = '';

    if (res_field_values <> '') then
    begin
      res_field_values = coalesce(res_field_values, '');
      res_order = res_order + 1;

      select coalesce(short_fio, iif(:db_id <> 94401, 'Договор', 'Müqavilə'))
      from hp_fio_clear(:res_field_name)
      into :res_field_name;
      err = err + 1;

      res_html = BLOB_APPEND(res_html, f_html_table_row(res_field_name, '<font color="' || color_red || '">' || res_field_values || '</font>', '', '', -1, -1, -1, -1, 2, ''));
    end
  end

  if (k_id > 0 and
      err = 0) then
  begin
    res_order = res_order + 1;
    res_html = BLOB_APPEND(res_html, f_html_table_row('Результат', '<font color="' || color_green || f_format('">%1</font>', iif(db_id <> 94401, 'Ошибок нет, готов к выдаче', 'Səhv yoxdur, buraxılış üçün hazırdı')), '', '', -1, -1, -1, -1, 2, ''));
  end
  res_html = BLOB_APPEND(res_html, '</table></body>');

  suspend;
end
