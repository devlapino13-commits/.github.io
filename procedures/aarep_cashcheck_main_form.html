declare variable diff_amount decimal(18,4);
declare variable month_name varchar(15);
declare variable dtsign d_datetime;
declare variable ccm_cash_saldo_delta d_currency;
declare variable db_id d_integer;
begin 
  db_id = f_get_db_id();

  select cm_currency_name, cm_currency_cent
  from company
  into :curr_name, curr_cent;

  diff_amount = 100;
  for select ccm_id, ccm_date, ccm_from_reg_id, ccm_order_kind, ccm_order_no, ccm_order_date, ccm_is_delete,
             ccm_cashier_curr, ccm_cashier_curr_datesign, iif(ccm_cashier_curr_datesign is null, 0, 1),
             ccm_cashier_new, ccm_cashier_new_datesign, iif(ccm_cashier_new_datesign is null, 0, 1),
             ccm_auditor, ccm_auditor_datesign, iif(ccm_auditor_datesign is null, 0, 1),
             ccm_cash_saldo_auto, ccm_cash_saldo_mannual, ccm_cash_saldo_fact, ccm_last_doc_no_income, ccm_last_doc_no_outcome, ccm_is_video_working,
             ccm_result_memo, ccm_from_bls_id, ccm_from_un_id, ccm_datetime, ccm_delete_user_id, ccm_delete_datetime,
             r1.reg_code, r1.reg_name, r2.reg_name, un_name,
             ccm_sh_bank_check, ccm_sh_cash_order, ccm_sh_pled_docum, ccm_is_stamp_paid, ccm_is_seal_cont, ccm_is_fuel_card,
             ccm_is_key_cash, ccm_is_key_safe, ccm_is_key_term,
             ccm_pled_docum_saldo_auto, ccm_cash_order_saldo_auto, ccm_bank_check_saldo_auto, ccm_num_cash,
             ccm_mbcard_fact, ccm_mbcard_calc, ccm_mbpinc_fact, ccm_mbpinc_calc, ccm_edl_id
      from cashcheckmain, region r1, region r2, username
      where ccm_from_reg_id = r1.reg_unicode and
            r1.parent = r2.reg_unicode and
            ccm_from_un_id = un_id and
            ccm_id = :rep_id
      into :ccm_id, :ccm_date, :ccm_from_reg_id, :ccm_order_kind, :ccm_order_no, :ccm_order_date, :ccm_is_delete,
           :ccm_cashier_curr, :ccm_cashier_curr_datesign, cashier_curr_sign,
           :ccm_cashier_new, :ccm_cashier_new_datesign, cashier_new_sign,
           :ccm_auditor, :ccm_auditor_datesign, :auditor_sign,
           :ccm_cash_saldo_auto, :ccm_cash_saldo_mannual, :ccm_cash_saldo_fact, :ccm_last_doc_no_income, :ccm_last_doc_no_outcome, :ccm_is_video_working,
           :ccm_result_memo, :ccm_from_bls_id, :ccm_from_un_id, :ccm_datetime, :ccm_delete_user_id,
           :ccm_delete_datetime, :reg_code, :reg_name, :branch_name, :un_name,
           :ccm_sh_bank_check, :ccm_sh_cash_order, :ccm_sh_pled_docum, :ccm_is_stamp_paid, :ccm_is_seal_cont, :ccm_is_fuel_card,
           :ccm_is_key_cash, :ccm_is_key_safe, :ccm_is_key_term,
           :ccm_pled_docum_saldo_auto, :ccm_cash_order_saldo_auto, :ccm_bank_check_saldo_auto, :ccm_num_cash,
           :ccm_mbcard_fact, :ccm_mbcard_calc, :ccm_mbpinc_fact, :ccm_mbpinc_calc, :ccm_edl_id
  do
  begin
    auditor_jobname = '';
    cashier_curr_fio = '';
    cashier_curr_jobname = '';
    cashier_new_jobname = '';
    cashier_new_fio = '';
    auditor_fio = '';

    ccm_cash_saldo_delta = ccm_cash_saldo_auto - ccm_cash_saldo_fact;
    if (db_id = 41701 and ccm_cash_saldo_delta > 0 and ccm_cash_saldo_delta < 0.01) then
      ccm_cash_saldo_auto = ccm_cash_saldo_auto - ccm_cash_saldo_delta; -- RFCKG-4558

    if (ccm_cashier_curr is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ccm_cashier_curr
      into :cashier_curr_fio;

      select jl_name
      from hr_pstatus, joblist
      where ps_from_c_id = :ccm_cashier_curr and
            ps_date_begin <= :ccm_date and
            ps_date_end >= :ccm_date and
            ps_status = 1 and
            ps_from_jl_id = jl_id
      order by ps_is_working desc
      rows 1
      into :cashier_curr_jobname;
    end
    if (ccm_cashier_new is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ccm_cashier_new
      into :cashier_new_fio;

      select jl_name
      from hr_pstatus, joblist
      where ps_from_c_id = :ccm_cashier_new and
            ps_date_begin <= :ccm_date and
            ps_date_end >= :ccm_date and
            ps_status = 1 and
            ps_from_jl_id = jl_id
      order by ps_is_working desc
      rows 1
      into :cashier_new_jobname;
    end
    if (ccm_auditor is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ccm_auditor
      into :auditor_fio;

      select jl_name
      from hr_pstatus, joblist
      where ps_from_c_id = :ccm_auditor and
            ps_date_begin <= :ccm_date and
            ps_date_end >= :ccm_date and
            ps_status = 1 and
            ps_from_jl_id = jl_id
      order by ps_is_working desc
      rows 1
      into :auditor_jobname;

    end
    if (ccm_from_bls_id > 0) then
      has_scan = 1;
    else
      has_scan = 0;

    order_name = '';
    if (ccm_order_kind = 0) then
      order_name = 'Приказ';
    if (ccm_order_kind = 1) then
      order_name = 'Распоряжение';
    if (ccm_order_kind = 2) then
      order_name = 'Внезапная проверка';
    if (ccm_order_kind = 3) then
      order_name = 'Прием-передача';
    if (ccm_order_kind = 4) then
      order_name = 'Закрытие кассы';
    if (ccm_order_kind = 5) then
      order_name = 'Превышение установленного лимита';

    if (ccm_cash_saldo_mannual = ccm_cash_saldo_fact) then
      ccm_result_str = 'Расхождений нет';
    if (ccm_cash_saldo_mannual > ccm_cash_saldo_fact) then
      ccm_result_str = 'Недостача ' || trunc((ccm_cash_saldo_mannual - ccm_cash_saldo_fact), 0) || ' ' || curr_name || ' ' || mod((ccm_cash_saldo_mannual - ccm_cash_saldo_fact) * 100, 100) || ' ' || curr_cent;
    if (ccm_cash_saldo_mannual < ccm_cash_saldo_fact) then
      ccm_result_str = 'Излишек ' || trunc((ccm_cash_saldo_fact - ccm_cash_saldo_mannual), 0) || ' ' || curr_name || ' ' || mod((ccm_cash_saldo_fact - ccm_cash_saldo_mannual) * 100, 100) || ' ' || curr_cent;

    ccm_is_video_str = '';
    if (ccm_is_video_working = 0) then
      ccm_is_video_str = 'Работает';
    if (ccm_is_video_working = 1) then
      ccm_is_video_str = 'Выключено';
    if (ccm_is_video_working = 2) then
      ccm_is_video_str = 'Неисправно';
    if (ccm_is_video_working = 3) then
      ccm_is_video_str = 'Отсутствует';

    ccm_order_text = '';
    if (ccm_order_kind = 0) then
      ccm_order_text = 'приказа';
    if (ccm_order_kind = 1) then
      ccm_order_text = 'распоряжения';
    if (ccm_order_kind = 2) then
      ccm_order_text = 'внезапной проверки';
    if (ccm_order_kind = 3) then
      ccm_order_text = 'приема-передачи';
    if (ccm_order_kind = 4) then
      ccm_order_text = 'закрытия кассы';
    if (ccm_order_kind = 5) then
      ccm_order_text = 'превышения установленного лимита';
    if (ccm_order_date is not null) then
    begin
      month_name = case extract(month from ccm_order_date)
                     when 1 then 'января'
                     when 2 then 'февраля'
                     when 3 then 'марта'
                     when 4 then 'апреля'
                     when 5 then 'мая'
                     when 6 then 'июня'
                     when 7 then 'июля'
                     when 8 then 'августа'
                     when 9 then 'сентября'
                     when 10 then 'октября'
                     when 11 then 'ноября'
                     when 12 then 'декабря'
                   end;
      ccm_order_text = ccm_order_text || ' от ' || extract(day from ccm_order_date) || ' ' || month_name || ' ' || extract(year from ccm_order_date) || ' года';
    end
    if ((ccm_order_no is not null) and
        (ccm_order_no <> '')) then
      ccm_order_text = ccm_order_text || ' № ' || ccm_order_no;

    if (ccm_date is not null) then
    begin
      month_name = case extract(month from ccm_date)
                     when 1 then 'января'
                     when 2 then 'февраля'
                     when 3 then 'марта'
                     when 4 then 'апреля'
                     when 5 then 'мая'
                     when 6 then 'июня'
                     when 7 then 'июля'
                     when 8 then 'августа'
                     when 9 then 'сентября'
                     when 10 then 'октября'
                     when 11 then 'ноября'
                     when 12 then 'декабря'
                   end;
      ccm_date_txt = extract(day from ccm_date) || ' ' || month_name || ' ' || extract(year from ccm_date) || ' г.';
    end
    ccm_saldo_fact_1 = trunc(ccm_cash_saldo_fact, 0);
    ccm_saldo_fact_2 = mod((ccm_cash_saldo_fact * 100), 100);

    ccm_saldo_manual_1 = trunc(ccm_cash_saldo_mannual, 0);
    ccm_saldo_manual_2 = mod((ccm_cash_saldo_mannual * 100), 100);

    auditor_login = (
      select first 1 un_name
      from username
      where un_from_c_id = :ccm_auditor
      order by un_id desc);
    cashier_curr_login = (
      select first 1 un_name
      from username
      where un_from_c_id = :ccm_cashier_curr
      order by un_id desc);
    cashier_new_login = (
      select first 1 un_name
      from username
      where un_from_c_id = :ccm_cashier_new
      order by un_id desc);

    if (ccm_edl_id > 0) then
    begin
      if (ccm_auditor_datesign is null) then
      begin
        dtsign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :ccm_edl_id
            and eds_from_c_id = :ccm_auditor
            and eds_status = 1
            and eds_enum_edoc_sign_kind <= 1);
        if (dtsign is not null) then
        begin
          ccm_auditor_datesign = dtsign;
          auditor_sign = 2;
          auditor_qr = f_edoc_qr_by_edl(:ccm_edl_id, :ccm_auditor);
        end
      end

      if (ccm_cashier_curr_datesign is null) then
      begin
        dtsign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :ccm_edl_id
            and eds_from_c_id = :ccm_cashier_curr
            and eds_status = 1
            and eds_enum_edoc_sign_kind <= 1);
        if (dtsign is not null) then
        begin
          ccm_cashier_curr_datesign = dtsign;
          cashier_curr_sign = 2;
          cashier_curr_qr = f_edoc_qr_by_edl(:ccm_edl_id, :ccm_cashier_curr);
        end
      end

      if (ccm_cashier_new_datesign is null) then
      begin
        dtsign = (
          select eds_status_datetime
          from edoc_sign
          where eds_from_edl_id = :ccm_edl_id
            and eds_from_c_id = :ccm_cashier_new
            and eds_status = 1
            and eds_enum_edoc_sign_kind <= 1);
        if (dtsign is not null) then
        begin
          ccm_cashier_new_datesign = dtsign;
          cashier_new_sign = 2;
          cashier_new_qr = f_edoc_qr_by_edl(:ccm_edl_id, :ccm_cashier_new);
        end
      end
    end

    suspend;
  end
end
