CREATE PROCEDURE HP_E_HR_REBUKE_LOAD(
    IN_RB_ID D_INTEGER,
    IN_C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RB_ID D_INTEGER,
    RB_NUMBER D_STR_20,
    RB_DATE D_DATE,
    RB_FROM_REG_ID D_INTEGER,
    RB_FROM_C_ID D_ID,
    RB_TYPE D_SMALLINT,
    RB_BODY D_STR_255,
    RB_REASON D_STR_255,
    RB_BONUS_CHANGE D_SMALLINT,
    RB_PERIOD_START D_DATE,
    RB_PERIOD_END D_DATE,
    RB_BONUS_KOEF D_CURRENCY,
    RB_SUB_NUMBER D_SMALLINT,
    RB_NUMBER_N D_INTEGER,
    RB_INITIATOR_ID D_ID,
    RB_NO_NUMBER D_SMALLINT,
    RB_SIGNER_ID D_INTEGER,
    RB_NOTE_DATE D_DATE,
    RS_PROTOCOL_NO D_INTEGER,
    RS_PROTOCOL_DATE D_DATE,
    RS_EXPLANATORY_DATE D_DATE,
    RS_FROM_RBK_ID D_INTEGER,
    RS_SCORES D_INTEGER,
    RES_ACTUAL_OFFICE D_STR_100,
    RES_ACTUAL_JOBLIST D_STR_100,
    RES_IS_SCORE_MODE D_SMALLINT,
    RES_TOTAL_ARTICLES D_STR_255,
    RES_TOTAL_SCORE D_INTEGER,
    RES_SCORE_LIMITS D_STR_20,
    RES_FROM_UN_ID D_ID)
AS
declare variable db_id d_integer;
begin 
  db_id = f_get_db_id();

  rb_date = current_date;
  rb_from_c_id = in_c_id;
  rb_bonus_change = 0;
  rb_bonus_koef = 0;
  rb_type = 1;
  rb_no_number = 0;
  res_total_score = 0;
  res_total_articles = '';
  res_from_un_id = un_id;

  res_score_limits = '';
  rs_from_rbk_id = 0;
  rs_scores = 0;

  res_is_score_mode = 1;

  if (coalesce(in_rb_id, 0) > 0) then
  begin
    res_is_score_mode = 2;

    select rb_id,
           rb_number,
           rb_date,
           rb_from_reg_id,
           rb_from_c_id,
           rb_type,
           rb_body,
           rb_reason,
           rb_bonus_change,
           rb_period_start,
           rb_period_end,
           rb_bonus_koef,
           rb_sub_number,
           rb_number_n,
           rb_initiator_id,
           rb_no_number,
           rb_signer_id,
           rb_note_date,
           rb_explanatory_date
        from rebuke
        where rb_id = :in_rb_id
        into :rb_id,
             :rb_number,
             :rb_date,
             :rb_from_reg_id,
             :rb_from_c_id,
             :rb_type,
             :rb_body,
             :rb_reason,
             :rb_bonus_change,
             :rb_period_start,
             :rb_period_end,
             :rb_bonus_koef,
             :rb_sub_number,
             :rb_number_n,
             :rb_initiator_id,
             :rb_no_number,
             :rb_signer_id,
             :rb_note_date,
             :rs_explanatory_date;

    if (not(db_id = 41701 and rb_from_reg_id in(1090100, 1060100))) then
      select rs_protocol_no, rs_protocol_date, rs_explanatory_date
      from rebuke_scores
      where rs_from_rb_id = :rb_id
      rows 1
      into rs_protocol_no, rs_protocol_date, rs_explanatory_date;

    select sum(rs_scores)
    from rebuke_scores
    where rs_from_rb_id = :rb_id and
          rs_is_delete = 0
    into res_total_score;

    if (res_total_score is null) then
      res_total_score = 0;

    select list(distinct rbk_code, ', ')
    from rebuke_scores, rebuke_kind
    where rs_from_rb_id = :rb_id and
          rs_is_delete = 0 and
          rs_from_rbk_id = rbk_id
    into res_total_articles;

    if (res_total_articles is null) then
      res_total_articles = '';
  end

  if (rb_from_c_id is not null) then
  begin
    select res_actual_office, res_actual_joblist, coalesce(:rb_from_reg_id, res_actual_reg_id)
    from hp_e_hr_orders_pers_info(:rb_from_c_id, :rb_date)
    into :res_actual_office, :res_actual_joblist, :rb_from_reg_id;
  end else
  if (rb_from_reg_id is null) then
  begin
    select un_curr_reg_id
    from username
    where un_id = :un_id
    into rb_from_reg_id;
  end

  suspend;
end
