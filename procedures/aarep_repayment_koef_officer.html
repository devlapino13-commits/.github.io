CREATE PROCEDURE AAREP_REPAYMENT_KOEF_OFFICER(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    REP_OKRUG VARCHAR(100),
    REP_OFFICE VARCHAR(100),
    REP_KC_FIO VARCHAR(100),
    REP_DISB DECIMAL(18,4),
    REP_ACTIVE_BEGIN DECIMAL(18,4),
    REP_ACTIVE_END DECIMAL(18,4),
    REP_PLAN_OS DECIMAL(18,4),
    REP_REPAY_ALL DECIMAL(18,4),
    REP_REPAY_NORMAL DECIMAL(18,4),
    REP_REPAY_ADV DECIMAL(18,4),
    REP_REPAY_DELINQ DECIMAL(18,4),
    REP_WRITE_OFF DECIMAL(18,4),
    REP_DISB_GR DECIMAL(18,4),
    REP_ACTIVE_BEGIN_GR DECIMAL(18,4),
    REP_ACTIVE_END_GR DECIMAL(18,4),
    REP_PLAN_OS_GR DECIMAL(18,4),
    REP_REPAY_ALL_GR DECIMAL(18,4),
    REP_REPAY_NORMAL_GR DECIMAL(18,4),
    REP_REPAY_ADV_GR DECIMAL(18,4),
    REP_REPAY_DELINQ_GR DECIMAL(18,4),
    REP_WRITE_OFF_GR DECIMAL(18,4),
    REP_DISB_IND DECIMAL(18,4),
    REP_ACTIVE_BEGIN_IND DECIMAL(18,4),
    REP_ACTIVE_END_IND DECIMAL(18,4),
    REP_PLAN_OS_IND DECIMAL(18,4),
    REP_REPAY_ALL_IND DECIMAL(18,4),
    REP_REPAY_NORMAL_IND DECIMAL(18,4),
    REP_REPAY_ADV_IND DECIMAL(18,4),
    REP_REPAY_DELINQ_IND DECIMAL(18,4),
    REP_WRITE_OFF_IND DECIMAL(18,4))
AS
begin
  for select rep_okrug, rep_office, rep_kc_fio, sum(rep_disb), sum(rep_active_begin), sum(rep_active_end),
             sum(rep_plan_os), sum(rep_repay_all), sum(rep_repay_normal), sum(rep_repay_adv), sum(rep_repay_delinq),
             sum(rep_write_off), sum(rep_disb_gr), sum(rep_active_begin_gr), sum(rep_active_end_gr),
             sum(rep_plan_os_gr), sum(rep_repay_all_gr), sum(rep_repay_normal_gr), sum(rep_repay_adv_gr),
             sum(rep_repay_delinq_gr), sum(rep_write_off_gr), sum(rep_disb_ind), sum(rep_active_begin_ind),
             sum(rep_active_end_ind), sum(rep_plan_os_ind), sum(rep_repay_all_ind), sum(rep_repay_normal_ind),
             sum(rep_repay_adv_ind), sum(rep_repay_delinq_ind), sum(rep_write_off_ind)
      from aarep_repayment_koef_temp(:report_region_id, :report_date_begin, :report_date_end)
      where rep_kc_fio is not null
      group by rep_okrug, rep_office, rep_kc_fio
      into :rep_okrug, :rep_office, :rep_kc_fio, :rep_disb, :rep_active_begin, :rep_active_end, :rep_plan_os,
           :rep_repay_all, :rep_repay_normal, :rep_repay_adv, :rep_repay_delinq, :rep_write_off, :rep_disb_gr,
           :rep_active_begin_gr, :rep_active_end_gr, :rep_plan_os_gr, :rep_repay_all_gr, :rep_repay_normal_gr,
           :rep_repay_adv_gr, :rep_repay_delinq_gr, :rep_write_off_gr, :rep_disb_ind, :rep_active_begin_ind,
           :rep_active_end_ind, :rep_plan_os_ind, :rep_repay_all_ind, :rep_repay_normal_ind, :rep_repay_adv_ind,
           :rep_repay_delinq_ind, :rep_write_off_ind
  do
  begin
    suspend;
  end

end
