CREATE PROCEDURE HP_GC_TRANSFER_ACCOUNTING_ENTRY
AS
declare variable reg_id d_integer;
declare variable un_id d_id;
declare variable igt_id d_integer;
declare variable iga_id d_integer;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable amount d_currency;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable ol_id d_id;
declare variable ol_parent_id d_id;
declare variable ol_date d_date;
declare variable ob_detail d_str_long;
declare variable obj_kind d_integer;
begin
  reg_id = 1090100;
  un_id = 1000000000000;
  obj_kind = f_objkind('ipc_gc_transfer', 'id');
  acc_kt = '090141710150000100';

  for
    select
      igt_id,
      igt_from_iga_id,
      c_id,
      c_name_fio,
      ag_from_al_id,
      igt_amount
    from ipc_gc_transfer
      inner join clients
        on c_id = igt_from_c_id
      inner join agreements
        on ag_id = igt_from_ag_id
    where igt_status = 2
      and igt_from_ol_id is null
  into
    :igt_id,
    :iga_id,
    :c_id,
    :c_name_fio,
    :acc_dt,
    :amount
  do
  begin
    ol_parent_id = (
      select
        ifa_from_ol_id
      from ipc_gc_analiz
        inner join ipc_file_analiz
          on ifa_id = iga_from_ifa_id
      where iga_id = :iga_id);
    ol_date = coalesce((
      select first 1
        cast(etj_pay_datetime as date)
      from elsom_transfer_job
      where etj_object_id = :igt_id
        and etj_object_kind = :obj_kind
        and etj_is_last = 1), current_date);

    insert into operlist (
      ol_parent_id,
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_document_client,
      ol_memo,
      ol_user_id
     )
    values (
      :ol_parent_id,
      :reg_id,
      2,
      11,
      :ol_date,
      :igt_id,
      9,
      :c_id,
      'Комиссия эквайера',
      :un_id
    )
    returning ol_id
    into :ol_id;

    ob_detail = f_format('Комиссия эквайера, возмещение по проведенным операциям; контрагент: %1', c_name_fio);
    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_detail,
      ob_summa,
      ob_summa_val
    )
    values (
      :ol_id,
      '417',
      1,
      :acc_dt,
      :acc_kt,
      :ob_detail,
      :amount,
      :amount
    );

    update ipc_gc_transfer set
      igt_status = 3,
      igt_from_ol_id = :ol_id
    where igt_id = :igt_id;
  end
end
