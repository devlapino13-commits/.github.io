CREATE PROCEDURE UXP$UXEVENT_ITEMS(
    UX_UUID D_UUID,
    LIST_KIND D_STR_CODE,
    WITH_EVENTS D_BOOL,
    CLONE_EVT_UUID D_UUID = null)
RETURNS (
    UFI_UUID D_UUID,
    UFI_EVENT_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_HIDDEN D_BOOL,
    UFI_READONLY D_BOOL,
    IMAGE_INDEX D_INTEGER,
    RES_CAPTION D_STR_255,
    RES_COMMENT D_STR_LONG,
    UF_FIELD_NAME D_STR_CODE,
    UF_SOURCE D_STR_CODE,
    UFI_UUID_STR D_STR_CODE,
    UFI_PARENT_UUID_STR D_STR_CODE,
    UFI_EVENT_CONTROL D_STR_CODE,
    UFI_EVENT_STATUS D_SMALLINT,
    UFI_UFR_UUID D_UUID,
    UFI_CHECKED_STATE D_SMALLINT,
    UFI_LINKED_EVT_UUID D_UUID)
AS
declare variable tmp_root d_uuid;
declare variable rec_count d_integer;
begin
  if (with_events is null) then
    with_events = 1;

  for select ufi_uuid, ufi_control, ufi_position, ufi_hidden, ufi_readonly, image_index, res_caption, res_comment,
             uf_field_name, uf_source, ufi_uuid_str, ufi_parent_uuid_str, ufi_event_control, ufi_event_status,
             ufi_ufr_uuid
  from uxp$uxevent_items_tmp(:ux_uuid, :list_kind)
  into :ufi_uuid, :ufi_control, :ufi_position, :ufi_hidden, :ufi_readonly, :image_index, :res_caption, :res_comment,
       :uf_field_name, :uf_source, :ufi_uuid_str, :ufi_parent_uuid_str, :ufi_event_control, :ufi_event_status,
       ufi_ufr_uuid
  do
  begin
    ufi_event_kind = 0;
    ufi_checked_state = null;
    ufi_linked_evt_uuid = null;

    if (clone_evt_uuid is not null) then
      ufi_event_status = -1;
    suspend;

    if (with_events = 1 and clone_evt_uuid is null and ufi_event_status >= 1) then
    begin
      tmp_root = :ufi_uuid;
      ufi_parent_uuid_str = uuid_to_char(ufi_uuid);
      ufi_hidden = 0;
      ufi_readonly = 0;
      ufi_position = 0;

      if (list_kind = 'sender') then
      begin
        image_index = 107; -- Самолетик

        for select evt_event_kind, count(*)
        from ux$event
        where evt_sender_uuid = :tmp_root
        group by 1
        into :ufi_event_kind, :rec_count
        do
        begin
          if (rec_count = 1) then
            ufi_event_status = 1;
          else
            ufi_event_status = 2;
  
          ufi_checked_state = 0;
          ufi_linked_evt_uuid = null;

          if (clone_evt_uuid is not null) then
          begin
            select evt_uuid
            from ux$event
            where evt_sender_uuid = :tmp_root and
                  evt_event_kind = :ufi_event_kind and
                  evt_target_uuid = :clone_evt_uuid and
                  evt_handler_kind = 0
            into ufi_linked_evt_uuid;

            if (ufi_linked_evt_uuid is not null) then
              ufi_checked_state = 1;
          end

          for select event_code || '   - ' || event_name, event_memo
          from uxp$uxevent_kinds('On')
          where event_id = :ufi_event_kind
          into res_caption, res_comment
          do
          begin
            ufi_uuid_str = uuid_to_char(ufi_uuid) || '$' || right('00000' || ufi_event_kind, 5);
            suspend;
          end
        end
      end else
      begin
        image_index = 108; -- Fx

        for select evt_event_kind, count(*)
        from ux$event
        where evt_target_uuid = :tmp_root
        group by 1
        into :ufi_event_kind, :rec_count
        do
        begin
          if (rec_count = 1) then
            ufi_event_status = 1;
          else
            ufi_event_status = 2;
  
          for select event_code || '   - ' || event_name, event_memo
          from uxp$uxevent_kinds('Do')
          where event_id = :ufi_event_kind
          into res_caption, res_comment
          do
          begin
            ufi_uuid_str = uuid_to_char(ufi_uuid) || '$' || right('00000' || ufi_event_kind, 5);
  
            suspend;
          end
        end
      end
    end else
    if (list_kind = 'sender' and with_events = 1 and clone_evt_uuid is not null) then
    begin
      tmp_root = :ufi_uuid;
      ufi_parent_uuid_str = uuid_to_char(ufi_uuid);
      ufi_hidden = 0;
      ufi_readonly = 0;
      ufi_position = 0;
      image_index = 107; -- Самолетик

      for select event_id, event_caption
      from uxp$uxevent_control_links(:ufi_event_control, 'On')
      into :ufi_event_kind, :res_caption
      do
      begin
        select count(*)
        from ux$event
        where evt_sender_uuid = :tmp_root and
              evt_event_kind = :ufi_event_kind
        into :rec_count;

        ufi_event_status = 0;
        if (rec_count = 1) then
          ufi_event_status = 1;
        else
        if (rec_count > 1) then
          ufi_event_status = 2;

        ufi_checked_state = 0;
        ufi_linked_evt_uuid = null;

        if (clone_evt_uuid is not null) then
        begin
          select evt_uuid
          from ux$event
          where evt_sender_uuid = :tmp_root and
                evt_event_kind = :ufi_event_kind and
                evt_target_uuid = :clone_evt_uuid and
                evt_handler_kind = 0
          into ufi_linked_evt_uuid;

          if (clone_evt_uuid = ufi_linked_evt_uuid) then
          begin
            ufi_checked_state = null;
            ufi_event_status = -1;
          end
          else
          if (ufi_linked_evt_uuid is not null) then
            ufi_checked_state = 1;
        end

        ufi_uuid_str = uuid_to_char(ufi_uuid) || '$' || right('00000' || ufi_event_kind, 5);
        suspend;
      end
    end
  end
end
