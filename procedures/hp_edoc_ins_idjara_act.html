CREATE PROCEDURE HP_EDOC_INS_IDJARA_ACT(
    KMA_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable ma_id d_id;
declare variable k_id d_id;
declare variable prepay_sum curr_type;
declare variable ag_id d_integer;
begin
  res_text = '';
  select kma.kma_from_k_id, kt.k_from_ma_app_id, kma.kma_from_ag_id
  from kontmatag kma
  inner join kontract kt on kt.k_id = kma.kma_from_k_id
  where kma.kma_id = :kma_id
  into k_id, ma_id, ag_id;

  select ma.ma_prepayment_summa
  from miniapp ma
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  into prepay_sum;

  if (coalesce(prepay_sum, 0) > 0) then
  begin
    if (not exists(select *
                   from operkontract ok
                   where ok.ok_from_k_id = :k_id and
                         ok.ok_operation_type = f_get_rd_id('CashTransactionIncom.PartialPrice') and
                         ok.ok_amount = :prepay_sum)) then
      res_text = 'Частичная сумма не оприходована в кассу. Оприходуйте сумму.';
  end

  if (current_date > (select ag.ag_date_end
                      from agreements ag
                      where ag.ag_id = :ag_id)) then
     res_text = 'Договор с контрагентом устарел';

  if (res_text = '' and f_edoc_status('idjara_goods.contragent_act', :kma_id, current_date) <> 'signed') then
  begin
    select res_text
    from hp_edoc_insert('idjara_goods.contragent_act', 'kontmatag', :kma_id, :un_id)
    into res_text;
  end
  suspend;
end
