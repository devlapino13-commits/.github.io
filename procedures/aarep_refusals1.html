CREATE PROCEDURE AAREP_REFUSALS1(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_FULL SMALLINT)
RETURNS (
    K_ID ID_TYPE,
    MA_ID ID_TYPE,
    UN_ID ID_TYPE,
    REF_STAGE STR_20,
    REF_EMPLOYEE STR_100,
    REF_POSITION STR_100,
    REF_REASON STR_255,
    REFUSAL_DATE DATE,
    REG_NAME STR_100,
    OFFICE_NAME STR_100,
    LO_NAME STR_100,
    C_NAME STR_100,
    KAL_NEW_KF_NAME STR_100,
    K_KONTRACT_NO STR_20)
AS
declare variable un_robot_id id_type = 1000000000000;
declare variable do_continue smallint;
begin
  if (rep_date_begin < '01.06.2018') then
    exit;
-- До 01.06.2018 данные по отказам были не корректные
  for select krl_from_k_id, krl_ma_id, krl_from_un_id, iif(krl_from_un_id = :un_robot_id, 'Робот', 'KC'), un_fio,
             jl_name, kr_text, cast(krl_datetime as date)
      from kontreslist
      inner join kontres on kr_id = krl_from_kr_id
      inner join username on un_id = krl_from_un_id
      left join hr_pstatus on ps_from_c_id = un_from_c_id and
            ps_status = 1 and
            ps_is_working = 1 and
            cast(krl_datetime as date) between ps_date_begin and ps_date_end
      left join joblist on jl_id = ps_from_jl_id
      where krl_datetime >= :rep_date_begin and
            krl_datetime < :rep_date_end + 1 and
            (krl_from_k_id > 0 or
            krl_ma_id > 0)
      into :k_id, :ma_id, :un_id, :ref_stage, :ref_employee, :ref_position, :ref_reason, :refusal_date
  do
  begin
    do_continue = 1;
    if ((ref_stage = 'робот') and
        (is_full = 0)) then
      do_continue = 0;

    if (do_continue = 1) then
    begin
      if (k_id > 0) then
      begin
-- kontract
        for select regp.reg_name, reg.reg_code || ' ' || reg.reg_name, lo.c_name_fio, cl.c_name_fio, kal_new_kf_name,
                   k_kontract_no
            from kontract
            inner join knaktlist on kal_from_kf_id = k_from_kf_id and
                  kal_date <= :rep_date_end and
                  kal_date_end >= :rep_date_end and
                  is_delete = 0
            inner join clients lo on lo.c_id = kal_new_kf_officer
            inner join knclient on kc_from_k_id = k_id and
                  kc_amount_approved > 0
            inner join clients cl on cl.c_id = kc_from_c_id
            inner join region as reg on reg.reg_unicode = kal_new_reg_id and exists(select *
                                                                                    from region_list(:rep_region_id, 1, 3, 1)
                                                                                    where id_list = reg.reg_unicode)
            inner join region as regp on regp.reg_unicode = reg.parent
            where k_id = :k_id
            into :reg_name, :office_name, :lo_name, :c_name, :kal_new_kf_name, :k_kontract_no
        do
          suspend;
      end
      else
      begin
-- miniapp
        if (exists(select *
                   from kontract
                   where k_from_ma_app_id = :ma_id)) then
          continue;
        for select regp.reg_name, reg.reg_code || ' ' || reg.reg_name, lo.c_name_fio, cl.c_name_fio, kal_new_kf_name,
                   k_kontract_no
            from miniapp
            left join kontract on k_from_ma_app_id = ma_id
            inner join knaktlist on kal_from_kf_id = ma_from_kf_id and
                  kal_date <= :rep_date_end and
                  kal_date_end >= :rep_date_end and
                  is_delete = 0
            inner join clients lo on lo.c_id = kal_new_kf_officer
            inner join miniappclient on mac_from_ma_id = ma_id and
                  mac_is_poruch = 0
            inner join clients cl on cl.c_id = mac_from_c_id
            inner join region as reg on reg.reg_unicode = kal_new_reg_id and exists(select *
                                                                                    from region_list(:rep_region_id, 1, 3, 1)
                                                                                    where id_list = reg.reg_unicode)
            inner join region as regp on regp.reg_unicode = reg.parent
            where ma_id = :ma_id and
                  ma_date_end = '31.12.9999'
            into :reg_name, :office_name, :lo_name, :c_name, :kal_new_kf_name, :k_kontract_no
        do
          suspend;
      end
    end
  end

  if (is_full = 1) then
  begin
-- actionlist
    ma_id = null;
    ref_stage = 'КК';
    for select acl_object_id, acl_from_un_id, un_fio, jl_name, al.acl_last_datetime, regp.reg_name,
               reg.reg_code || ' ' || reg.reg_name, lo.c_name_fio, cl.c_name_fio, kal_new_kf_name, k_kontract_no
        from actionlist al
        inner join refdata on rd_id = acl_from_rd_id and
              rd_str_id = 'underwriter' and
              rd_main_code = 'actiongroup'
        inner join kontract on k_id = acl_object_id
        inner join knaktlist on kal_from_kf_id = k_from_kf_id and
              kal_date <= :rep_date_end and
              kal_date_end >= :rep_date_end and
              is_delete = 0
        inner join clients lo on lo.c_id = kal_new_kf_officer
        inner join knclient on kc_from_k_id = k_id and
              kc_amount_approved > 0
        inner join clients as cl on cl.c_id = kc_from_c_id
        inner join region as reg on reg.reg_unicode = acl_from_reg_id and exists(select *
                                                                                 from region_list(:rep_region_id, 1, 3, 1)
                                                                                 where id_list = reg.reg_unicode)
        inner join region as regp on regp.reg_unicode = reg.parent
        inner join username on un_id = acl_from_un_id
        left join hr_pstatus on ps_from_c_id = un_from_c_id and
              ps_status = 1 and
              ps_is_working = 1 and
              acl_date between ps_date_begin and ps_date_end
        left join joblist on jl_id = ps_from_jl_id
        where acl_date between :rep_date_begin and :rep_date_end and
              acl_status = -2 and
              acl_object_kind = 9 and
              acl_dont_show_in_order = 0
        into :k_id, :un_id, :ref_employee, :ref_position, :refusal_date, :reg_name, :office_name, :lo_name, :c_name,
             :kal_new_kf_name, :k_kontract_no
    do
    begin
      select f_get_verify_memo(left(obm_memo, 2048), 'reason_memo')
      from objectmemo
      inner join objectprops on opr_object_id = :k_id and
            opr_object_kind = 9 and
            opr_subject_kind = -1 and
            opr_id = obm_from_opr_id
      order by obm_id desc
      rows 1
      into ref_reason;
      suspend;
    end
  end
end
