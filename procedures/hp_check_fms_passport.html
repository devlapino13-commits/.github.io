CREATE PROCEDURE HP_CHECK_FMS_PASSPORT(
    PASS_NO VARCHAR(20))
RETURNS (
    CHECK_RESULT INTEGER)
AS
declare variable db_path varchar(255);
declare variable found_no varchar(20);
declare variable db_connect varchar(255);
begin
  check_result = 0; -- все нормально

  if (f_get_db_id() = 81001 and pass_no is not null) then -- если база РФ
  begin
    -- находим путь к БД
    db_path = rdb$get_context('SYSTEM','DB_NAME');
    select dbc_connection_path
    from dbconnects
    where dbc_connection_path like '%:' || :db_path rows 1
    into :db_connect;

    if (db_connect is not null) then
    begin
      found_no = null;
      -- через путь к БД насраиваем путь подключения к внешней базе
      db_connect = replace(db_connect, 'DB_MBFRU.IB', 'DB_FMS.IB');

      -- если номер паспорта находится на внешней БД, значит он недействителен
      pass_no = replace(pass_no, ' ', '');
      execute statement('select pv_number from passvoid where pv_number = cast(:pass_no as varchar(20)) ') (pass_no := pass_no)
          on external data source db_connect
          as user 'SYSDBA' password 'F!ora80X'
          into :found_no;

      if (found_no is not null) then
        check_result = 1; -- Паспорт недействителен
    end
  end
  suspend;
end
