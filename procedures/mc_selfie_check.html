CREATE PROCEDURE MC_SELFIE_CHECK(
    MCL_ID D_ID)
RETURNS (
    HAS_SELFIE D_SMALLINT)
AS
declare variable photo_id d_id;
declare variable c_id d_id;
begin
  if (f_get_db_id() not in (39801, 41701)) then
  begin
    has_selfie = 1;
    suspend;
    exit;
  end
  has_selfie = 0;
  photo_id = 0;
  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id;

  select c.c_blob_photo_id
  from clients c
  where c.c_id = :c_id
  into photo_id;

  if (coalesce(photo_id, 0) > 0) then
    has_selfie = 1;

  suspend;
end
