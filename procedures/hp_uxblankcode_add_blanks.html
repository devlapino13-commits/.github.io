CREATE PROCEDURE HP_UXBLANKCODE_ADD_BLANKS(
    IN_UBC_ID D_INTEGER,
    IN_BLANK_LIST D_STR_LONG)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable blank_id d_str_code;
declare variable add_count d_integer;
begin
  res_is_error = 0;
  res_text = '';

  if (trim(coalesce(in_blank_list, '')) = '') then
  begin
    res_is_error = 1;
    res_text = 'Код не указан';
    suspend;
    exit;
  end

  add_count = 0;
  for select res_str
  from hp_split_str(:in_blank_list, ',')
  into :blank_id
  do
  begin
    if (not exists(select * from ux$blanklink
                   where ubl_from_ubc_id = :in_ubc_id and
                         ubl_blank_id = :blank_id)) then
    begin
      insert into ux$blanklink (ubl_from_ubc_id, ubl_blank_id)
      values (:in_ubc_id, :blank_id);

      add_count = add_count + 1;
    end
  end

  res_text = 'Бланки добавлены в пакет: ' || add_count;

  suspend;
end
