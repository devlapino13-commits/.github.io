CREATE PROCEDURE AAREP_EFFECTIVENESS_WEEK_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(50),
    WEEK_1_DISB_SVZ_ONE CURR_TYPE,
    WEEK_2_DISB_SVZ_ONE CURR_TYPE,
    WEEK_3_DISB_SVZ_ONE CURR_TYPE,
    WEEK_4_DISB_SVZ_ONE CURR_TYPE,
    WEEK_5_DISB_SVZ_ONE CURR_TYPE,
    WEEK_6_DISB_SVZ_ONE CURR_TYPE,
    WEEK_1_SVZ_CLI_AVG CURR_TYPE,
    WEEK_2_SVZ_CLI_AVG CURR_TYPE,
    WEEK_3_SVZ_CLI_AVG CURR_TYPE,
    WEEK_4_SVZ_CLI_AVG CURR_TYPE,
    WEEK_5_SVZ_CLI_AVG CURR_TYPE,
    WEEK_6_SVZ_CLI_AVG CURR_TYPE,
    WEEK_1_SVZ_KOL INTEGER,
    WEEK_2_SVZ_KOL INTEGER,
    WEEK_3_SVZ_KOL INTEGER,
    WEEK_4_SVZ_KOL INTEGER,
    WEEK_5_SVZ_KOL INTEGER,
    WEEK_6_SVZ_KOL INTEGER)
AS
declare variable reg_unicode integer;
declare variable i smallint;
declare variable rep_date date;
declare variable svz_kol_act integer;
declare variable disb_kred_week integer;
declare variable rep_date_begin date;
declare variable kol_cli integer;
begin
  i = 0;
  while (i < 6) do
  begin
    i = i + 1;
    rep_date = rep_date_end - (6 - i) * 7;
    rep_date_begin = rep_date - 6; -- начало недели

-- Обнуление значений исходящих параметров
    svz_kol_act = 0;
    disb_kred_week = 0;
    kol_cli = 0;
    week_1_disb_svz_one = 0;
    week_1_svz_cli_avg = 0;
    week_2_disb_svz_one = 0;
    week_2_svz_cli_avg = 0;
    week_3_disb_svz_one = 0;
    week_3_svz_cli_avg = 0;
    week_4_disb_svz_one = 0;
    week_4_svz_cli_avg = 0;
    week_5_disb_svz_one = 0;
    week_5_svz_cli_avg = 0;
    week_6_disb_svz_one = 0;
    week_6_svz_cli_avg = 0;
    week_1_svz_kol = 0;
    week_2_svz_kol = 0;
    week_3_svz_kol = 0;
    week_4_svz_kol = 0;
    week_5_svz_kol = 0;
    week_6_svz_kol = 0;

-- Кол-во СВЗ
    for select ps_from_reg_id, count(ps_from_c_id)
        from jobkind, joblist, hr_pstatus
        where jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
              jk_id = jl_kind and
              jl_id = ps_from_jl_id and
              ps_status = 1 and
              ps_is_working = 1 and
              :rep_date between ps_date_begin and ps_date_end
-- Исключаем декретников
              and
              not coalesce((select sawt_id
                            from hr_voctravel, hr_orders, sa_worktype
                            where hrv_from_ho_id = ho_id and
                                  ho_order_type = sawt_id and
                                  hrv_from_c_id = ps_from_c_id and
                                  :rep_date between hrv_date_begin and hrv_date_end_fact
                            order by hrv_date_begin desc
                            rows 1), 0) in (16, 17, 18)
        group by ps_from_reg_id
        into reg_unicode, svz_kol_act
    do
    begin
      if (svz_kol_act is null) then
        svz_kol_act = 0;
      if (i = 1) then
        week_1_svz_kol = svz_kol_act;
      if (i = 2) then
        week_2_svz_kol = svz_kol_act;
      if (i = 3) then
        week_3_svz_kol = svz_kol_act;
      if (i = 4) then
        week_4_svz_kol = svz_kol_act;
      if (i = 5) then
        week_5_svz_kol = svz_kol_act;
      if (i = 6) then
        week_6_svz_kol = svz_kol_act;
      select reg_name
      from region
      where reg_unicode = :reg_unicode
      into :office_name;
      suspend;
    end

-- Обнуление значений исходящих параметров
    svz_kol_act = 0;
    disb_kred_week = 0;
    kol_cli = 0;
    week_1_disb_svz_one = 0;
    week_1_svz_cli_avg = 0;
    week_2_disb_svz_one = 0;
    week_2_svz_cli_avg = 0;
    week_3_disb_svz_one = 0;
    week_3_svz_cli_avg = 0;
    week_4_disb_svz_one = 0;
    week_4_svz_cli_avg = 0;
    week_5_disb_svz_one = 0;
    week_5_svz_cli_avg = 0;
    week_6_disb_svz_one = 0;
    week_6_svz_cli_avg = 0;
    week_1_svz_kol = 0;
    week_2_svz_kol = 0;
    week_3_svz_kol = 0;
    week_4_svz_kol = 0;
    week_5_svz_kol = 0;
    week_6_svz_kol = 0;

-- Кол-во выдачи
    for select ks_region_id, sum(k_kolich_client_approved)
        from kontstate, kontract, product
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :rep_date_begin and :rep_date and
              ks_from_k_id = k_id and
              k_from_pr_id = pr_id and
              pr_product_kind <> 2
        group by ks_region_id
        into reg_unicode, disb_kred_week
    do
    begin
      if (disb_kred_week is null) then
        disb_kred_week = 0;
      if (i = 1) then
        week_1_disb_svz_one = disb_kred_week;
      if (i = 2) then
        week_2_disb_svz_one = disb_kred_week;
      if (i = 3) then
        week_3_disb_svz_one = disb_kred_week;
      if (i = 4) then
        week_4_disb_svz_one = disb_kred_week;
      if (i = 5) then
        week_5_disb_svz_one = disb_kred_week;
      if (i = 6) then
        week_6_disb_svz_one = disb_kred_week;
      select reg_name
      from region
      where reg_unicode = :reg_unicode
      into :office_name;
      suspend;
    end

-- Обнуление значений исходящих параметров
    svz_kol_act = 0;
    disb_kred_week = 0;
    kol_cli = 0;
    week_1_disb_svz_one = 0;
    week_1_svz_cli_avg = 0;
    week_2_disb_svz_one = 0;
    week_2_svz_cli_avg = 0;
    week_3_disb_svz_one = 0;
    week_3_svz_cli_avg = 0;
    week_4_disb_svz_one = 0;
    week_4_svz_cli_avg = 0;
    week_5_disb_svz_one = 0;
    week_5_svz_cli_avg = 0;
    week_6_disb_svz_one = 0;
    week_6_svz_cli_avg = 0;
    week_1_svz_kol = 0;
    week_2_svz_kol = 0;
    week_3_svz_kol = 0;
    week_4_svz_kol = 0;
    week_5_svz_kol = 0;
    week_6_svz_kol = 0;

-- Кол-во активных клиентов
    for select ac_region_id, sum(ac_is_client)
        from aarep_cli_active_list(:rep_date)
        where ac_is_client = 1 and
              (select aa_category_id
               from aarep_get_category_id(ac_kontract_id, :rep_date, null, null, null, null, null, null)) in (1, 2)
        group by ac_region_id
        into reg_unicode, kol_cli
    do
    begin
      if (kol_cli is null) then
        kol_cli = 0;
      if (i = 1) then
        week_1_svz_cli_avg = kol_cli;
      if (i = 2) then
        week_2_svz_cli_avg = kol_cli;
      if (i = 3) then
        week_3_svz_cli_avg = kol_cli;
      if (i = 4) then
        week_4_svz_cli_avg = kol_cli;
      if (i = 5) then
        week_5_svz_cli_avg = kol_cli;
      if (i = 6) then
        week_6_svz_cli_avg = kol_cli;
      select reg_name
      from region
      where reg_unicode = :reg_unicode
      into :office_name;
      suspend;
    end
  end

end
