CREATE PROCEDURE AAREP_GET_MONEYOPER_CODE(
    ACC_DT VARCHAR(20),
    ACC_KT VARCHAR(20),
    IS_RESIDENT INTEGER,
    NUMBER_DT INTEGER,
    NUMBER_KT INTEGER)
RETURNS (
    RES_OPERCODE_RD_ID INTEGER,
    RES_OPERCODE_STR VARCHAR(20),
    RES_ERROR INTEGER)
AS
declare variable res_mop_id integer;
declare variable opercode_resident integer;
declare variable opercode_noresident integer;
declare variable is_sud integer;
begin
  is_sud = 0;
  if (is_resident >= 100) then
  begin
    is_sud = 1;
    is_resident = mod(is_resident, 100);
  end
  res_error = 0;
  res_mop_id = 0;
  res_opercode_rd_id = 0;
  res_opercode_str = '';
  opercode_resident = null;
  opercode_noresident = null;
  if (is_resident is null) then
    is_resident = 1;

  if ((is_sud = 1) and
      (acc_kt containing '58-03' or acc_kt containing '76-09' or acc_kt containing '76-02') and
      (acc_dt containing '50-' or acc_dt containing '51-')) then
  begin
    res_mop_id = 57; -- Настройки по судебным решениям
    opercode_resident = 5455;
    opercode_noresident = 5465;
    if (acc_dt containing '50-') then
      opercode_noresident = opercode_resident;
  end
  else
  begin
    select mop_id, mop_rd_id_resident, mop_rd_id_noresident
    from moneyoper
    where mop_mask_dt = :acc_dt and
          mop_mask_kt = :acc_kt and
          mop_is_active = 1 and
          mop_dt_min <= :number_dt and
          mop_dt_max >= :number_dt and
          mop_kt_min <= :number_kt and
          mop_kt_max >= :number_kt
    order by mop_id desc
    rows 1
    into :res_mop_id, :opercode_resident, :opercode_noresident;
  end

  if (res_mop_id is null) then
    res_mop_id = 0;
  if (res_mop_id = 0) then
    res_error = 1;
  if ((opercode_resident is not null) and
      (opercode_noresident is not null)) then
  begin
    if (is_resident = 1) then
      res_opercode_rd_id = opercode_resident;
    if (is_resident = 0) then
      res_opercode_rd_id = opercode_noresident;
  end
  else
  begin
    if (opercode_resident is not null) then
      res_opercode_rd_id = opercode_resident;
    if (opercode_noresident is not null) then
      res_opercode_rd_id = opercode_noresident;
  end
  if (res_opercode_rd_id <> 0) then
  begin
    select rd_str_id
    from refdata
    where rd_id = :res_opercode_rd_id
    into :res_opercode_str;
  end
  suspend;
end
