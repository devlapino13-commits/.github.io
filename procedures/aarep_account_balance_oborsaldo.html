CREATE PROCEDURE AAREP_ACCOUNT_BALANCE_OBORSALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    ACC_BALANCE VARCHAR(255),
    ACC_BALANCE_NO VARCHAR(255),
    ACC_GROUP VARCHAR(255),
    ACC_CLASS VARCHAR(255),
    CURR_CODE VARCHAR(3),
    BEG_DT DECIMAL(18,4),
    BEG_KT DECIMAL(18,4),
    OB_DT DECIMAL(18,4),
    OB_KT DECIMAL(18,4),
    END_DT DECIMAL(18,4),
    END_KT DECIMAL(18,4),
    LAST_OPER_DATE DATE,
    ACC_GROUP_NO VARCHAR(20),
    ACC_KIND INTEGER,
    ACC_KIND_NAME VARCHAR(255),
    ACCOUNT_IS_ERROR INTEGER)
AS
begin
  for select acc_balance, acc_balance_no, acc_group, acc_class, curr_code, sum(beg_dt), sum(beg_kt), sum(ob_dt),
             sum(ob_kt), sum(end_dt), sum(end_kt), max(last_oper_date), sum(account_is_error), acc_group_no, acc_kind,
             acc_kind_name
      from aarep_account_oborsaldo(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency)
      group by acc_balance, acc_balance_no, acc_group, acc_class, curr_code, acc_group_no, acc_kind, acc_kind_name
      into :acc_balance, :acc_balance_no, :acc_group, :acc_class, :curr_code, :beg_dt, :beg_kt, :ob_dt, :ob_kt, :end_dt,
           :end_kt, :last_oper_date, :account_is_error, :acc_group_no, :acc_kind, :acc_kind_name
  do
  begin
    suspend;
  end
end
