CREATE PROCEDURE HP_ANTIFROD_GET_RES(
    K_ID D_ID)
RETURNS (
    REP_KND_DATETIME D_DATETIME,
    JSO D_INTEGER)
AS
declare variable bls_id d_id;
begin
  select afl_datetime, iif(afl_content is null, null, json.parse(afl_content)), afl_content_id
  from antifrod_loan
  where afl_from_k_id = :k_id
  into rep_knd_datetime, jso, bls_id;

  if (jso is null) then
  begin
    if (rep_knd_datetime is null or bls_id is null) then -- для старых, у кого нет расчета в таблице
    begin
      select jso
      from hp_antifrod_get_calc(:k_id)
      into jso;

      rep_knd_datetime = json.obj_get_val(:jso, 'calc_datetime');
    end
    else
    begin
      select json.parse(blx_blob)
      from hp_get_blobex_blob(:bls_id)
      into jso;
    end
  end
  suspend;
end
