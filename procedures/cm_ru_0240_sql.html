CREATE PROCEDURE CM_RU_0240_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable pr_code d_str_100;
declare variable c_id id_type;
declare variable drl_vin d_str_255;
declare variable z_cost_market curr_type;
declare variable z_cost_likvid curr_type;
declare variable z_cost_kontract curr_type;
declare variable z_document d_str_255;
begin
  for select pr_code, kc_from_c_id
      from kontract
      inner join product on pr_id = k_from_pr_id
      inner join knclient on kc_from_k_id = k_id
      where k_id = :k_id and
            kc_amount_approved > 0 and
            (pr_code containing '.zalog.' or pr_code containing '.zalog_car.')
      into pr_code, c_id
  do
  begin
    select f_docrecog_value(cds_from_drl_id, 'vin', '')
    from clientdocs cds
    where cds_from_c_id = :c_id and
          cds_is_delete = 0 and
          cds_from_pk_id = 315
    order by cds_create_datetime desc rows 1
    into drl_vin;

    drl_vin = coalesce(drl_vin, '');

    select z_cost_market, z_cost_likvid, z_cost_kontract, z_document
    from zalog
    where z_from_k_id = :k_id
    into z_cost_market, z_cost_likvid, z_cost_kontract, z_document;

    if (drl_vin not similar to '[а-я]+' and drl_vin <> '' and coalesce(z_document, '') = '') then  --RFCRU-3518
    begin
      res_error_text = 'Не указан VIN';
      res_error_is_critical = 2;
      suspend;
    end 
    if (coalesce(z_cost_market, 0) = 0) then
    begin
      res_error_text = 'Не указана рыночная стоимость';
      res_error_is_critical = 2;
      suspend;
    end 
    if (coalesce(z_cost_likvid, 0) = 0) then
    begin
      res_error_text = 'Не указана залоговая стоимость';
      res_error_is_critical = 2;
      suspend;
    end 
    if (coalesce(z_cost_kontract, 0) = 0 and pr_code containing '.ZALOG_CAR.') then
    begin
      res_error_text = 'Не указана контрактная стоимость';
      res_error_is_critical = 2;
      suspend;
    end 
  end
end
