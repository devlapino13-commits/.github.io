CREATE PROCEDURE AAREP_APR_LAWYER(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    OFFICER_NAME STR_100,
    DEL_BAL_BEGIN_7 CURR_TYPE,
    DEL_BAL_END_7 CURR_TYPE,
    CHANGES_BALANCE_7 CURR_TYPE,
    DEL_BAL_BEGIN_30 CURR_TYPE,
    DEL_BAL_END_30 CURR_TYPE,
    CHANGES_BALANCE_30 CURR_TYPE,
    KOL_CL INTEGER,
    WORK_DAYS_QTY INTEGER,
    FIRST_DATE_MONTH DATE,
    DISB_SUM DECIMAL(18,4),
    JL_NAME D_STR_100,
    PARENT_NAME D_STR_100)
AS
declare variable OKS_KC_ID ID_TYPE;
declare variable REG_UNICODE D_INTEGER;
begin
  for select apr.officer_name,
             apr.reg_name,
             apr.parent_name,
             apr.del_bal_begin_7,
             apr.del_bal_end_7,
             apr.changes_balance_7,
             apr.del_bal_begin_30,
             apr.del_bal_end_30,
             apr.changes_balance_30,
             apr.oks_kc_id,
             apr.reg_unicode,
             apr.first_date_month,
             apr.work_days_qty
      from AAREP_REPORT_APR(:REP_REGION_ID, :REP_DATE_END) APR
      where APR.OFFICER_NAME in ('Quliyev Rəşad Elşad oğlu',
                                 'Ağamedov Nadir Nizamoviç',
                                 'Süleymanov Məhəmməd Şahabudin oğlu',
                                 'Quliyev Kamil Hikmət oğlu',
                                 'Nadirov Rafiq Şura oğlu',
                                 'Quliyev Kamil Hikmət oğlu',
                                 'Məmmədov Ramin Hafis oğlu',
                                 'Ağayev Zakir Niyaz oğlu',
                                 'Sultanov Saleh Vaqif oğlu',
                                 'Əliyev Murad Ələsgər oğlu',
                                 'İsmayılov Sənan Musa oğlu',
                                 'Abdullayev Əli Allahbaxış oğlu',
                                 'Məmmədova Aytac Teymur qızı')
      into officer_name,
           reg_name,
           parent_name,
           del_bal_begin_7,
           del_bal_end_7,
           changes_balance_7,
           del_bal_begin_30,
           del_bal_end_30,
           changes_balance_30,
           oks_kc_id,
           reg_unicode,
           first_date_month,
           work_days_qty
  do
  begin
    jl_name = '';
    select jl.jl_name
    from hr_pstatus ps
    inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
    where ps.ps_from_c_id = :oks_kc_id and
          jl.jl_name in ('Problemli kreditlər şöbəsinin rəhbəri',
                         'Problemli kreditlər üzrə hüquqşünas',
                         'Problemli kreditlər üzrə mütəxəssis',
                         'Hüquqşunas')
    rows 1
    into jl_name;

    kol_cl = 0;
    select count(*)
    from oksummary oks
    where oks.oks_kc_id = :oks_kc_id and
          :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_close_state = 0 and
          oks.oks_balance > 0 and
          oks.oks_region_id = :reg_unicode 
    into kol_cl;
    suspend;
  end
end
