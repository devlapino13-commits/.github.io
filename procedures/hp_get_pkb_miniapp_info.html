CREATE PROCEDURE HP_GET_PKB_MINIAPP_INFO
RETURNS (
    JSON_VALUE D_BLOB_TEXT,
    PMI_ID D_ID,
    PROJECT D_STR_10)
AS
declare variable MA_ID D_ID;
declare variable C_ID D_ID;
declare variable STATUS D_SMALLINT;
declare variable REP_TYPE D_SMALLINT;
declare variable PMI_SUM D_CURRENCY;
declare variable MA_SH_PAYMENT D_CURRENCY;
declare variable CELL_NO D_STR_100;
declare variable INN D_STR_20;
declare variable PMI_PROJECT D_INTEGER;
declare variable APPLICATION_TIMESTAMP varchar(30);
declare variable TEMP_DATETIME D_STR_20;
declare variable LOAN_APP_SUB_DATE varchar(30);
declare variable PMI_CREATE_DATETIME varchar(30);
declare variable KRL_DATETIME varchar(30);
begin
  for select pmi.pmi_id, pmi.pmi_from_ma_id, pmi.pmi_from_c_id, pmi.pmi_status,
             pmi.pmi_sum, pmi.pmi_rep_type, pmi.pmi_project, f_datetimetostr(pmi.pmi_create_datetime),
             iif(pmi_project =f_get_rd_id('cib.pkb'),
                 substring(replace(cast(pmi.pmi_create_datetime as varchar(50)), ' ', 'T') from 1 for 23)|| '+05:00',
                 substring(replace(cast(pmi.pmi_create_datetime as varchar(50)), ' ', 'T') from 1 for 19))
  from pkb_miniapp_info pmi
  where pmi_is_sent in (0, -1)
        and not (
            pmi.pmi_project = f_get_rd_id('cib.gkb')
            and f_datetimetostr(pmi.pmi_create_datetime) = '28.04.2026 13:42'
        )
  into :pmi_id, :ma_id, :c_id, :status,
       :pmi_sum, :rep_type, :pmi_project, :temp_datetime, :pmi_create_datetime
  do
  begin
    select cl.c_inn
    from clients cl
    where cl.c_id = :c_id
    into :inn;

    if (status in (3, 5)) then
      select sum(sh_all_with_tax) / max(sh_id)
      from fm_schedule_list(11, :ma_id)
      where sh_id <> 0
      into :ma_sh_payment;

    if (pmi_project = f_get_rd_id('cib.pkb')) then
    begin
      project = 'pkb';

      if (:rep_type = 2 and :status = 1) then  -- Этап №2.1 Первая передача данных о заявке на кредит
      begin
        cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');
        select substring(replace(cast(eds.eds_status_datetime as varchar(50)), ' ', 'T') from 1 for 23)|| '+05:00'
        from edoc_list edl
        inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
        where edl.edl_object_kind = f_objkind('miniapp', 'id') and
              edl.edl_object_id = :ma_id and
              edl.edl_from_edc_id = 2
        into :loan_app_sub_date;

        json_value = '{' || f_json_make_pair('iin', :inn, 'str') || ', '
                         || f_json_make_pair('rep_type', :rep_type, 'num') || ', '
                         || f_json_make_pair('phone_number', :cell_no, 'str') || ', '
                         || f_json_make_pair('sum', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('status', :status, 'num')  || ', '
                         || f_json_make_pair('loan_object', 10, 'num') || ', '
                         || f_json_make_pair('loan_purpose', 9, 'num') || ', '
                         || f_json_make_pair('consent_confirmed', 1, 'num') || ', '
                         || f_json_make_pair('id_app', :ma_id, 'str') || ', '
                         || f_json_make_pair('loan_app_sub_date', :loan_app_sub_date, 'str') || '}';
      end
      else if (:rep_type = 1 and :status = 1) then  -- Этап №2.4 Обновление данных о заявке на кредит при редактировании заявки
      begin
        json_value = '{' || f_json_make_pair('id_app', :ma_id, 'str') || ', '
                         || f_json_make_pair('status', :status, 'num')  || ', '
                         || f_json_make_pair('sum', :pmi_sum, 'num') || '}';
      end
      else if (:status = 3) then  -- Этап №2.2 Обновление данных о заявке на кредит после утверждения договора менеджером
      begin
        json_value = '{' || f_json_make_pair('id_app', :ma_id, 'str') || ', '
                         || f_json_make_pair('status', :status, 'num')  || ', '
                         || f_json_make_pair('loan_app_decision_date', :pmi_create_datetime, 'str') || ', '
                         || f_json_make_pair('loan_app_approved_amount', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('loan_app_approved_instalment_amount', :ma_sh_payment, 'num') || '}';
      end
      else if (:status = 5) then  -- Этап №2.5 Обновление данных о заявке на кредит после выдачи
      begin
        json_value = '{' || f_json_make_pair('id_app', :ma_id, 'str') || ', '
                         || f_json_make_pair('status', :status, 'num')  || ', '
                         || f_json_make_pair('loan_app_issue_date', :pmi_create_datetime, 'str') || ', '
                         || f_json_make_pair('loan_app_issue_amount', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('loan_app_issue_instalment_amount', :ma_sh_payment, 'num') || '}';
      end
      else if (:status = 2) then  -- Этап №2.3 Обновление данных о заявке на кредит при отказе
      begin
        select substring(replace(cast(ktl.krl_datetime as varchar(50)), ' ', 'T') from 1 for 23)|| '+05:00'
        from kontreslist ktl
        where ktl.krl_ma_id = :ma_id
          and ktl.krl_from_kr_id <> 1
        into :krl_datetime;

        json_value = '{' || f_json_make_pair('id_app', :ma_id, 'str') || ', '
                         || f_json_make_pair('status', :status, 'num')  || ', '
                         || f_json_make_pair('loan_app_decision_date', :krl_datetime, 'str') || '}';

      end  
      else if (:status = 4) then  -- Этап №2.4 Обновление данных о заявке на кредит при отказе
      begin
        json_value = '{' || f_json_make_pair('id_app', :ma_id, 'str') || ', '
                         || f_json_make_pair('status', :status, 'num') || '}';
      end
    end
    else
    begin
      project = 'gkb';

      select substring(replace(cast(ma.ma_date_save as varchar(50)), ' ', 'T') from 1 for 19)
      from miniapp ma
      where ma.ma_id = :ma_id and
            ma.ma_date_end = '31.12.9999' rows 1
      into :application_timestamp;

      if (:rep_type = 2 and :status = 1) then  -- Этап №2.1 Первая передача данных о заявке на кредит и Этап №2.1 Обновление данных о заявке на кредит при редактировании заявки
      begin
        json_value = '{' || f_json_make_pair('iin', :inn, 'str') || ', '
                         || f_json_make_pair('sum', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('loan_purpose', 9, 'num') || ', '
                         || f_json_make_pair('loan_object', 10, 'num') || ', '
                         || f_json_make_pair('creditor_application_id', :ma_id, 'str')  || ', '
                         || f_json_make_pair('application_timestamp', :application_timestamp, 'str') || ', '
                         || f_json_make_pair('application_status', :status, 'num') || ', '
                         || f_json_make_pair('credit_type_id', :rep_type, 'num') || '}';
      end
      else if (:status = 3) then  -- Этап №2.2 Обновление данных о заявке на кредит после утверждения договора менеджером
      begin
        json_value = '{' || f_json_make_pair('iin', :inn, 'str') || ', '
                         || f_json_make_pair('sum', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('creditor_application_id', :ma_id, 'str') || ', '
                         || f_json_make_pair('application_status', :status, 'num')  || ', '
                         || f_json_make_pair('application_timestamp', :application_timestamp, 'str') || ', '
                         || f_json_make_pair('approved_amount', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('approved_pay', :ma_sh_payment, 'num') || ', '
                         || f_json_make_pair('denied_or_approved_timestamp', :pmi_create_datetime, 'str') || '}';
      end
      else if (:status = 5) then  -- Этап №2.5 Обновление данных о заявке на кредит после выдачи
      begin
        json_value = '{' || f_json_make_pair('iin', :inn, 'str') || ', '
                         || f_json_make_pair('sum', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('creditor_application_id', :ma_id, 'str') || ', '
                         || f_json_make_pair('application_status', :status, 'num')  || ', '
                         || f_json_make_pair('application_timestamp', :application_timestamp, 'str') || ', '
                         || f_json_make_pair('approved_amount', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('approved_pay', :ma_sh_payment, 'num') || ', '
                         || f_json_make_pair('issued_amount', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('issued_pay', :ma_sh_payment, 'num') || ', '
                         || f_json_make_pair('issue_timestamp', :pmi_create_datetime, 'str') || '}';
      end
      else if (:status = 2) then  -- Этап №2.3 Обновление данных о заявке на кредит при отказе
      begin
        select substring(replace(cast(ktl.krl_datetime as varchar(50)), ' ', 'T') from 1 for 19)
        from kontreslist ktl
        where ktl.krl_ma_id = :ma_id
          and ktl.krl_from_kr_id <> 1
        into :krl_datetime;

        json_value = '{' || f_json_make_pair('iin', :inn, 'str') || ', '
                         || f_json_make_pair('sum', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('creditor_application_id', :ma_id, 'str')  || ', '
                         || f_json_make_pair('application_status', :status, 'num') || ', '
                         || f_json_make_pair('application_timestamp', :application_timestamp, 'str') || ', '
                         || f_json_make_pair('denied_or_approved_timestamp', :krl_datetime, 'str') || '}';
      end  
      else if (:status = 4) then  -- Этап №2.3 Обновление данных о заявке на кредит при отказе
      begin
        json_value = '{' || f_json_make_pair('iin', :inn, 'str') || ', '
                         || f_json_make_pair('sum', :pmi_sum, 'num') || ', '
                         || f_json_make_pair('creditor_application_id', :ma_id, 'str')  || ', '
                         || f_json_make_pair('application_timestamp', :application_timestamp, 'str') || ', '
                         || f_json_make_pair('application_status', :status, 'num') || '}';
      end
    end

    suspend;
  end
end
/*
applicationStatus - Статус заявки (статус = 3, "Одобрено")
approved_amount - Одобренная сумма
approved_pay - Ежемесячный платеж по одобренной сумме
deniedOrApprovedTimestamp - Дата и время одобрения (статус = 3)
deniedOrApprovedTimestamp - Дата и время отказа (статус = 2)  */
