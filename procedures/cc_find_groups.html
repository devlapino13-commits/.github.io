CREATE PROCEDURE CC_FIND_GROUPS(
    PROJECT_CC_CODE STR_100,
    OPERATOR_LOGIN STR_100)
RETURNS (
    KAL_NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_CODE,
    SBA_FROM_K_ID D_ID,
    SBA_NUMBER D_INTEGER,
    PANT_ID D_ID)
AS
declare variable PROJECT_RD_CODE D_STR_100;
declare variable SBL_ID D_INTEGER;
begin
  select rd.rd_code
  from refdata rd
  where rd.rd_main_code = 'oktell.projects'
  and rd.rd_memo = :project_cc_code
  into :project_rd_code;

  if (project_rd_code is null) then
    exception e_default 'project_rd_code is null';

  if (project_cc_code in ('ks_finance_1', 'ks_finance_2', 'pantera', 'pantera_1', 'pantera_2', 'drpv')) then
  begin
     if (OPERATOR_LOGIN = '---') then
     begin
         for select distinct kf_name as kal_new_kf_name,
             k_kontract_no,
             k_id as sba_from_k_id,
             0 as sba_number,
             pant_id
         from okt_pantera_list(current_date, :project_rd_code, '', '', '', '', 1, 0, 0, 0, 0, 1)
         where called_oper_str =  ''
         into :kal_new_kf_name, :k_kontract_no, :sba_from_k_id, :sba_number, :pant_id
         do
         begin
            suspend;
         end
     end
     else
     begin
        for select distinct kf_name as kal_new_kf_name,
               k_kontract_no,
               k_id as sba_from_k_id,
               0 as sba_number,
               pant_id
        from okt_pantera_list(current_date, :project_rd_code, :operator_login, '', '', '', 1, 0, 0, 0, 0, 1)
        into :kal_new_kf_name, :k_kontract_no, :sba_from_k_id, :sba_number, :pant_id
        do
        begin
           suspend;
        end
     end
  end
  else if (project_cc_code in ('sirena', 'sirena_1', 'sirena_2')) then
  begin
     if (OPERATOR_LOGIN = '---') then
     begin
        for select distinct kf_name as kal_new_kf_name,
                   k_kontract_no,
                   k_id as sba_from_k_id,
                   0 as sba_number,
                   pant_id
        from okt_sirena_list(current_date, :project_rd_code, 'OKTELL.SubProjects.Sirena.Delinq','', '', '', '', 0, 0, 0, 0, 0, 1)
        where called_oper_str = ''
        into :kal_new_kf_name, :k_kontract_no, :sba_from_k_id, :sba_number, :pant_id
        do
        begin
           suspend;
        end
        if (project_cc_code in ('sirena_1', 'sirena_2')) then
        begin
            for select distinct kf_name as kal_new_kf_name,
                                k_kontract_no,
                                k_id as sba_from_k_id,
                                0 as sba_number,
                                pant_id
            from okt_sirena_list(current_date, :project_rd_code, 'OKTELL.SubProjects.Sirena.VE','', '', '', '', 0, 0, 0, 0, 0, 1)
            where called_oper_str = ''
            into :kal_new_kf_name, :k_kontract_no, :sba_from_k_id, :sba_number, :pant_id
            do
            begin
               suspend;
            end
        end
     end
     else
     begin
        for select distinct kf_name as kal_new_kf_name,
                    k_kontract_no,
                    k_id as sba_from_k_id,
                    0 as sba_number,
                    pant_id
        from okt_sirena_list(current_date, :project_rd_code, 'OKTELL.SubProjects.Sirena.Delinq', :operator_login, '', '', '', 0, 0, 0, 0, 0, 1)
        into :kal_new_kf_name, :k_kontract_no, :sba_from_k_id, :sba_number, :pant_id
        do
        begin
           suspend;
        end
        if (project_cc_code in ('sirena_1', 'sirena_2')) then
        begin
            for select distinct kf_name as kal_new_kf_name,
                                k_kontract_no,
                                k_id as sba_from_k_id,
                                0 as sba_number,
                                pant_id
            from OKT_SIRENA_LIST(current_date, :project_rd_code, 'OKTELL.SubProjects.Sirena.VE', :operator_login, '', '', '', 0, 0, 0, 0, 0, 1)
            into :kal_new_kf_name, :k_kontract_no, :sba_from_k_id, :sba_number, :pant_id
            do
            begin
               suspend;
            end
        end
     end
  end
end
