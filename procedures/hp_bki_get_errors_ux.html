CREATE PROCEDURE HP_BKI_GET_ERRORS_UX(
    SHOW_FIXED D_BOOL,
    SHOW_REPEAT D_BOOL,
    BKP_ID_IN D_ID,
    REP_DATE_BEGIN D_DATE)
RETURNS (
    BKE_ID D_ID,
    REC_NO D_INTEGER,
    IS_FIXED D_BOOL,
    KONTRACT_NO D_STR_20,
    K_ID D_ID,
    ERROR_CODE D_STR_CODE,
    ERROR_TEXT D_STR_LONG,
    FIXED_DATETIME D_DATETIME,
    BKP_DATE D_DATE,
    BKP_CODE D_STR_CODE,
    BKP_STATUS D_INTEGER,
    BKP_DATETIME D_DATETIME,
    EVENT_DATE D_DATE,
    EVENT_CODE D_STR_CODE,
    ACTION_CODE D_STR_CODE,
    CRED_STATUS D_STR_CODE,
    EVENT_GOOD_SEND_DATE D_DATE,
    BKK_ID D_ID,
    BKP_ID D_ID,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    REC_ID D_ID)
AS
declare variable event_id d_integer;
declare variable action_id d_integer;
declare variable error_id d_id;
declare variable ma_kind d_integer;
declare variable k_kind d_integer;
declare variable ma_id d_id;
declare variable bkc_uid_bkd_id d_id;
begin
  ma_kind = f_objkind('miniapp', 'id');
  k_kind = f_objkind('kontract', 'id');

  if (show_fixed is null) then show_fixed = 0;
  if (show_repeat is null) then show_repeat = 0;
  if (rep_date_begin is null) then rep_date_begin = '01.11.2022';

  for select bkp_id, bkp_date, bkp_code, bkp_status, bkp_datetime, bkp_from_bkk_id,
             bev_event_date, bkl_object_kind, bkl_object_id,
             bev_id, bev_rec_no, bev_event_id, bev_loan_status,
             bke_id, bke_error_code, bke_fixed, bke_fixed_datetime, bke_error_id,
             bkt_action, bkc_uid_bkd_id
  from bki_errors, bki_events, bki_clients, bki_loans, bki_packet, bki_packet_type
  where bev_id = bke_from_bev_id
    and bkc_id = bev_from_bkc_id
    and bkl_id = bkc_from_bkl_id
    and bkp_id = bkl_from_bkp_id
    and bkt_id = bkp_from_bkt_id
    and bkp_id = coalesce(:bkp_id_in, bkp_id)
    and bke_date >= :rep_date_begin
    and (:show_repeat = 1 or bke_error_code not in ('2011', '2010'))
    and (:show_fixed = 1 or bke_fixed = 0)
  into bkp_id, bkp_date, bkp_code, bkp_status, bkp_datetime, bkk_id,
       event_date, object_kind, object_id,
       rec_id, rec_no, event_id, cred_status,
       bke_id, error_code, is_fixed, fixed_datetime, error_id,
       action_id, bkc_uid_bkd_id
  do
  begin
    event_code = f_enum('bki_event', event_id, 'code', 0);
    action_code = f_enum('bki_packet_action', action_id, 'code', 0);
    error_text = f_memo_text(error_id);

    kontract_no = null;
    k_id = null;

    if (object_kind = ma_kind) then
    begin
      ma_id = object_id;

      select k_id, k_kontract_no
      from kontract k
      where k.k_from_ma_app_id = :object_id
        and k.k_status >= 0
      order by k_status desc, 1 desc
      rows 1
      into k_id, kontract_no;
    end
    else if (object_kind = k_kind) then
    begin
      k_id = object_id;
  
      select k_kontract_no, k_from_ma_app_id
      from kontract k
      where k.k_id = :k_id
      into kontract_no, ma_id;
    end

    event_good_send_date = null;

    with t_objects(object_kind, object_id) as(
      select :ma_kind, :ma_id
      from rdb$database
    union
      select :k_kind, k_id
      from kontract k
      where k_from_ma_app_id = :ma_id
    )
    select max(bkp_date)
    from t_objects, bki_events, bki_clients, bki_loans, bki_packet
    where bkc_id = bev_from_bkc_id
      and bkl_id = bkc_from_bkl_id
      and bkp_id = bkl_from_bkp_id
      and bkl_object_kind = object_kind
      and bkl_object_id = object_id
      and bev_event_id = :event_id
      and bev_event_date = :event_date
      and bev_status = 3
      and bkp_status = 3
      and coalesce(bev_delete_bev_id, 0) = 0
      and bkc_uid_bkd_id = :bkc_uid_bkd_id
    into event_good_send_date;

    suspend;
  end 
end
