CREATE PROCEDURE HP_SUD_LOANDOCS_AUTOCREATE(
    SMD_ID D_ID,
    UN_ID D_ID)
AS
declare variable doc_kind d_integer;
declare variable doc_code d_str_code;
declare variable k_id d_id;
declare variable disb_kind d_integer;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable edoc_obj_kind d_integer;
declare variable edoc_kind d_integer;
declare variable bls_id d_id;
declare variable res_sld_id d_id;
declare variable slc_id d_id;
declare variable edoc_blank_code d_str_code;
declare variable disb_date d_date;
declare variable c_id d_id;
declare variable bls_id_2 d_id;
declare variable err_text d_str_255;
begin
  -- Процедура создает список необходимых документов по судебному делу

  if (un_id is null) then
    un_id = 1000000000000;

  select smr_from_k_id, k_disb_date
  from sud_mainreq, kontract
  where smr_id = :smd_id and
        smr_from_k_id = k_id
  into k_id, disb_date;

  select smd_disb_kind
  from sud_maindocs
  where smd_id = :smd_id
  into disb_kind;

  for select scd_id, scd_code
  from sud_cfgdockind
  where scd_on_request > 0
  order by  scd_sort, scd_id
  into :doc_kind, :doc_code
  do
  begin
    -- Если выдан не через кассу, кассовый ордер игнорируем
    if (disb_kind <> 1 and doc_code = 'cash_order')  then
      Continue;

    -- Если выдан через кассу, квитанцию от Монеты игнорируем
    if (disb_kind = 1 and doc_code = 'transfer_receipt')  then
      Continue;

    res_sld_id = null;
    obj_id = null;
    bls_id = null;
    edoc_blank_code = '';

    -- договор займа
    if (doc_code in ('kontract', 'payment_schedule')) then
    begin
      if (doc_code = 'kontract') then
        edoc_blank_code = 'loan.agreement';

      if (doc_code = 'payment_schedule') then
        edoc_blank_code = 'loan.schedule';

      obj_kind = f_objkind('edoc_blank', 'id');

      edoc_obj_kind = f_objkind('kontract', 'id');
      edoc_kind = f_edoc_kind('loan.agreement', 'id');

      select edb_id, coalesce(edb_scan_bls_id, edb_from_bls_id)
      from edoc_blank edb, edoc_list edl, edoc_frx frx
      where edb.edb_from_edl_id = edl_id and
            edb.edb_from_edf_id = frx.edf_id and
            edl.edl_object_kind = :edoc_obj_kind and
            edl.edl_object_id = :k_id and
            edl.edl_status = 1 and
            edl.edl_blank_date <= :disb_date and
            coalesce(edl.edl_date_end, :disb_date) >= :disb_date and
            edl.edl_from_edc_id = :edoc_kind and
            edb.edb_status >= 1 and
            frx.edf_add_code = :edoc_blank_code
      into obj_id, bls_id;
    end

    if (doc_code = 'transfer_receipt') then
    begin
      -- Отправляем запрос на получение квитанции от Монеты
      err_text = '';

      select error_text
      from hp_sud_send_receipt_req(:k_id)
      into err_text;
    end

    if (doc_code = 'pretrial_claim') then -- претензия досудебная (RFCRU-3114)
    begin
      obj_kind = f_objkind('edoc_blank', 'id');
      edoc_obj_kind = f_objkind('clients', 'id');
      edoc_kind = f_edoc_kind(doc_code, 'id');

      select edb_id, coalesce(edb_scan_bls_id, edb_from_bls_id)
      from edoc_blank edb, edoc_list edl, edoc_frx frx
      where edb.edb_from_edl_id = edl_id and
            edb.edb_from_edf_id = frx.edf_id and
            edl.edl_object_kind = :edoc_obj_kind and
            edl.edl_object_id = :c_id and
            edl.edl_status = 1 and
            edl.edl_blank_date <= :disb_date and
            coalesce(edl.edl_date_end, :disb_date) >= :disb_date and
            edl.edl_from_edc_id = :edoc_kind and
            edb.edb_status >= 1
      into obj_id, bls_id;
    end

    if (obj_id is not null and bls_id is not null) then
    begin
      select res_sld_id
      from hp_sud_loanbls_add(:smd_id, null, :doc_kind, :obj_kind, :obj_id, :bls_id, :un_id)
      into res_sld_id;
    end else
    begin
      select res_sld_id
      from hp_sud_loandocs_add(:smd_id, null, :doc_kind, :un_id)
      into :res_sld_id;
    end

    if (res_sld_id is null) then
      exception e_default 'Sud_LoanBLS not added';
  end

  for select scd_id, scd_code
  from sud_cfgdockind
  where scd_on_client > 0
  into :doc_kind, :doc_code
  do
  begin
    -- Если выдан не через кассу, кассовый ордер игнорируем
    if (disb_kind <> 1 and doc_code = 'cash_order')  then
      Continue;

    -- Если выдан через кассу, квитанцию от Монеты игнорируем
    if (disb_kind = 1 and doc_code = 'transfer_receipt')  then
      Continue;

    for select slc_id, kc_from_c_id
    from sud_loanclient, knclient
    where slc_from_smr_id = :smd_id and
          slc_from_kc_id = kc_id
    into :slc_id, c_id
    do
    begin
      res_sld_id = null;

      obj_id = null;
      bls_id = null;
      bls_id_2 = null;
      edoc_blank_code = '';
  
      -- договор займа
      if (doc_code = 'asp_agreement') then
      begin
        obj_kind = f_objkind('edoc_blank', 'id');
  
        edoc_obj_kind = f_objkind('clients', 'id');
        edoc_kind = f_edoc_kind('asp.agreement', 'id');
  
        select edb_id, coalesce(edb_scan_bls_id, edb_from_bls_id)
        from edoc_blank edb, edoc_list edl, edoc_frx frx
        where edb.edb_from_edl_id = edl_id and
              edb.edb_from_edf_id = frx.edf_id and
              edl.edl_object_kind = :edoc_obj_kind and
              edl.edl_object_id = :c_id and
              edl.edl_status = 1 and
              edl.edl_blank_date <= :disb_date and
              coalesce(edl.edl_date_end, :disb_date) >= :disb_date and
              edl.edl_from_edc_id = :edoc_kind and
              edb.edb_status >= 1
        order by edb_id desc
        rows 1
        into obj_id, bls_id;
      end

      if (doc_code = 'passport') then
      begin
        obj_kind = f_objkind('clientdocs', 'id');
  
        select cds_id, cds_doc_blob_1, cds_doc_blob_2
        from clientdocs cds, passkind pk, passdoctype pdt
        where pk_id = cds_from_pk_id and
          pdt_id = pk_from_pdt_id and
          cds_from_c_id = :c_id and
          cds_doc_date_begin <= :disb_date and -- выдан до даты выдачи
          coalesce(cds_doc_date_end, '31.12.2999') >= :disb_date and -- со сроком действия после даты выдачи
          cds_create_datetime <= (:disb_date + 1) and -- Паспорт должен быть прикреплен до выдачи
          cds_is_delete = 0 and -- до сих пор актуальный
          cds_verify_status = 2 and -- подтврежленный
          pdt.pdt_is_main_doc = 1 and
          (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
        order by cds_create_datetime desc
        rows 1
        into obj_id, bls_id, bls_id_2;
      end

      if (doc_code = 'registration') then
      begin
        obj_kind = f_objkind('clientdocs', 'id');

        select cds_id, cds_doc_blob_1, cds_doc_blob_2
        from clientdocs cds, passkind pk, passdoctype pdt
        where pk_id = cds_from_pk_id and
          pdt_id = pk_from_pdt_id and
          cds_from_c_id = :c_id and
          cds_doc_date_begin <= :disb_date and -- выдан до даты выдачи
          coalesce(cds_doc_date_end, '31.12.2999') >= :disb_date and -- со сроком действия после даты выдачи
          cds_create_datetime <= (:disb_date + 1) and -- Паспорт должен быть прикреплен до выдачи
          coalesce(cds_delete_datetime, current_timestamp) >= :disb_date and -- Не аннулирован до выдачи
          cds_verify_status = 2 and -- подтврежленный
          pdt.pdt_code in ('registration_permanent', 'registration_fiz', 'registration_yur') and
          (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
        order by cds_create_datetime desc
        rows 1
        into obj_id, bls_id, bls_id_2;
      end

      if (obj_id is not null and bls_id is not null) then
      begin
        select res_slb_id
        from hp_sud_loanbls_add(:smd_id, :slc_id, :doc_kind, :obj_kind, :obj_id, :bls_id, :un_id)
        into res_sld_id;
      end else
      begin
        select res_sld_id
        from hp_sud_loandocs_add(:smd_id, :slc_id, :doc_kind, :un_id)
        into :res_sld_id;
      end

      if (res_sld_id is null) then
        exception e_default 'Sud_LoanBLS not added';

      if (bls_id_2 is not null and bls_id_2 <> coalesce(bls_id, 0)) then
      begin
        select res_slb_id
        from hp_sud_loanbls_add(:smd_id, :slc_id, :doc_kind, :obj_kind, :obj_id, :bls_id_2, :un_id)
        into res_sld_id;
      end
    end
  end
end
