CREATE PROCEDURE AAREP_PRINT_CELLAKT(
    AKT_ID DECIMAL(18,0))
RETURNS (
    CAL_FROM_CAM_ID DECIMAL(18,0),
    CAL_ID INTEGER,
    CAL_DATE_BEGIN DATE,
    CAL_DATE_END DATE,
    CAL_FROM_C_ID DECIMAL(18,0),
    CAL_CELL_NO VARCHAR(20),
    CAL_STATE INTEGER,
    CAL_FROM_COP_ID INTEGER,
    CAL_PRIOR_C_ID DECIMAL(18,0),
    CAM_DATE DATE,
    CAM_FROM_REG_ID INTEGER,
    CAM_MEMO VARCHAR(255),
    CAM_DATETIME TIMESTAMP,
    CAM_USER_ID DECIMAL(18,0),
    CAM_OFFICE VARCHAR(100),
    CAL_CLIENT_FIO VARCHAR(100),
    CAL_FIO_FROM VARCHAR(100),
    CAM_TYPE_STR VARCHAR(100),
    CAM_USER_NAME VARCHAR(100),
    CAL_IS_CONTACT_STR VARCHAR(20),
    CAL_IS_CORPORATE_STR VARCHAR(20),
    CAL_HAS_LIMIT_STR VARCHAR(100),
    CAL_CELL_NO_STR VARCHAR(20))
AS
declare variable cam_type integer;
declare variable cal_is_contact integer;
declare variable cal_is_corporate integer;
declare variable cal_has_limit integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable clk_limit_kind varchar(100);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select cal_from_cam_id, cal_id, cal_date_begin, cal_date_end, cal_from_c_id, cal_cell_no, cal_state,
             cal_is_contact, cal_is_corporate, cal_has_limit, cal_from_cop_id, cal_prior_c_id, cam_date, cam_type,
             cam_from_reg_id, cam_memo, cam_datetime, cam_user_id, reg_code || ' ' || reg_name, c1.c_name_fio,
             c3.c_name_fio, clk_name
      from cellaktmain, region, clients c1, username, clients c3, cellaktlist, cellimitkind
      where cal_from_cam_id = cam_id and
            cam_from_reg_id = reg_unicode and
            c1.c_id = cal_from_c_id and
            cam_type <> -1 and
            cam_user_id = un_id and
            un_from_c_id = c3.c_id and
            cal_from_cam_id = :akt_id and
            clk_id = cal_has_limit
      order by cal_id
      into :cal_from_cam_id, :cal_id, :cal_date_begin, :cal_date_end, :cal_from_c_id, :cal_cell_no, :cal_state,
           :cal_is_contact, :cal_is_corporate, :cal_has_limit, :cal_from_cop_id, :cal_prior_c_id, :cam_date, :cam_type,
           :cam_from_reg_id, :cam_memo, :cam_datetime, :cam_user_id, :cam_office, :cal_client_fio, :cam_user_name,
           :clk_limit_kind
  do
  begin
    cal_fio_from = '';
    if (cal_prior_c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :cal_prior_c_id
      into :cal_fio_from;
    end
    if (cam_type = 0) then
      cam_type_str = 'Прием на склад';
    if (cam_type = 1) then
      cam_type_str = 'Личную симку на контакты';
    if (cam_type = 2) then
      cam_type_str = 'Личную симку на корпоратив';
    if (cam_type = 3) then
      cam_type_str = 'Со склада на пользование';
    if (cam_type = 4) then
      cam_type_str = 'Передача между сотрудниками';
    if (cam_type = 5) then
      cam_type_str = 'Изменение статуса';
    if (cam_type = 6) then
      cam_type_str = 'Аннулирование действ.симки';

    if (cal_is_contact = 1) then
      cal_is_contact_str = 'Основн.';
    else
      cal_is_contact_str = 'Дополн.';

    if (cal_is_corporate = 1) then
      cal_is_corporate_str = 'Да';
    else
      cal_is_corporate_str = 'Нет';

--      if (CAL_HAS_LIMIT=1)
--      then CAL_HAS_LIMIT_STR='Есть';
--      else CAL_HAS_LIMIT_STR='Нет';
    cal_has_limit_str = clk_limit_kind;

    cal_cell_no_str = left(:cal_cell_no, 3) || ' ' || substring(:cal_cell_no from 4 for 3) || ' ' || right(:cal_cell_no, 6);
    suspend;
  end
end
