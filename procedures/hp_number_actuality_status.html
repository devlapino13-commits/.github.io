CREATE PROCEDURE HP_NUMBER_ACTUALITY_STATUS(
    C_ID D_ID)
RETURNS (
    RES_TEXT_LINK D_STR_8100,
    STATUS D_SMALLINT)
AS
declare variable cell_no d_str_100;
declare variable const_minus_days d_datetime;
declare variable eds_status_datetime d_datetime;
declare variable cnc_is_sucsess d_smallint;
declare variable mcl_google_id d_str_255;
begin
  -- RFCRU-3522 возвращает ссылку и статус актуальности номера телефона клиента
  res_text_link = '';
  status = 0;
  cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');

  select mcl.mcl_google_id
  from mc_login mcl
  where mcl.mcl_from_c_id = :c_id
    and mcl.mcl_login = :cell_no
    and mcl.mcl_is_delete = 0
  order by mcl.mcl_last_login_datetime desc
  rows 1
  into mcl_google_id;

  const_minus_days = dateadd(f_get_const('const_minus_days') day to current_timestamp); -- константа для сколько времени прошло с даты подписания посл. документа

  select eds.eds_status_datetime
  from edoc_sign eds
  inner join edoc_msg edm on edm.edm_from_eds_id = eds.eds_id
  inner join fsrobot_msgbox msg on msg.msg_id = edm.edm_from_msg_id
  where eds.eds_from_c_id = :c_id
   and eds.eds_status = 1                                       -- последний подписанный документ через СМС
   and msg.msg_type in (1, 2, 3)
   and msg.msg_mail_to in (:cell_no, :mcl_google_id)
  order by eds.eds_status_datetime desc
  rows 1
  into eds_status_datetime;

  eds_status_datetime = coalesce(eds_status_datetime, current_timestamp); -- Если клиент ничего не подписал значит кл. новый

  select cnc.cnc_is_sucsess
  from check_number_code cnc
  where cnc.cnc_from_c_id = :c_id
    and cnc.cnc_is_active = 1
    and cnc.cnc_cell_main = :cell_no
    and cnc.cnc_create_datetime between :eds_status_datetime and current_timestamp -- между посл. док и сегодня
  order by cnc.cnc_create_datetime
  rows 1
  into cnc_is_sucsess;

  cnc_is_sucsess = coalesce(cnc_is_sucsess, -1);

  if (const_minus_days >= eds_status_datetime and cnc_is_sucsess not in (1, 0)) then
  begin
    res_text_link = BLOB_APPEND(res_text_link, f_uri_sql('Отправить запрос на проверку доступа клиента к верифицированному номеру', 'send_check_code', 'Clients', c_id, 'none', 'Подтверждаете', '', 1, 4, '', '', 'dialog', 'null'));
  end
  else if (cnc_is_sucsess = 0) then
  begin
    res_text_link = BLOB_APPEND(res_text_link, f_uri_sql('Отправить смс повторно', 'send_check_code', 'Clients', c_id, 'none', 'Подтверждаете', '', 1, 4, '', '', 'dialog', 'null'));
    res_text_link = BLOB_APPEND(res_text_link, '<br>');
    res_text_link = BLOB_APPEND(res_text_link, f_uri_sql('Ввести код подтверждения', 'Check_number_actuality', 'Clients', c_id, 'text', 'Введите код', '', 1, 4, '', '', 'dialog', 'null'));
  end
  else if (cnc_is_sucsess = 1 or cnc_is_sucsess = -1) then
    status = 1;

  suspend;
end
