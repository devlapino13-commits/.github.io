CREATE PROCEDURE AAREP_ISSUED_CARDS(
    REP_REGION D_INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PARENT_NAME D_STR_100,
    REG_NAME D_STR_100,
    FIO D_STR_100,
    CARD_ID D_INTEGER,
    CCA_CODE D_STR_CODE,
    CAC_DATE DATE,
    KONTRACT_NO VARCHAR(20),
    LIMIT_DATE DATE,
    LIMIT DECIMAL(18,2),
    PINCONVERT VARCHAR(3),
    K_DISB_DATE DATE,
    C_NAME_FIO VARCHAR(255),
    PR_NAME VARCHAR(50))
AS
declare variable reg_id d_integer;
declare variable cca_id d_integer;
declare variable limit_tmp d_currency;
declare variable limit_date_tmp d_date;
declare variable status_issued d_integer;
declare variable status_pin d_integer;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable k_from_pr_id d_integer;
declare variable reg_parent d_integer;
declare variable k_fact_disb_lo_id id_type;
begin
  status_issued = f_enum('cardaktclient', 'issued', 'id');
  status_pin = f_enum('cardaktclient', 'pin_assigned', 'id');

  for select rl.id_list, reg.reg_name, reg.parent
      from region_list(:rep_region, 1, 3, 1) rl
      inner join region reg on reg.reg_unicode = rl.id_list
      into :reg_id, :reg_name, reg_parent
  do
  begin
    select r.reg_name
    from region r
    where r.reg_unicode = :reg_parent
    into parent_name;
    
    for select c.c_name_fio, cal.cal_card_id, cca.cca_code, cac.cac_date, cca.cca_id, cca.cca_credit_limit,
               cast(cca.cca_datetime as date), iif(cem.cem_is_pincover = 0, 'Нет', 'Да'), c.c_id
        from cardaktclient cac
        inner join cardclientacc cca on cca.cca_id = cac.cac_from_cca_id
        inner join cardlist cal on cca.cca_id = cal.cal_from_cca_id
        inner join clients c on c.c_id = cac.cac_from_c_id
        inner join cardembomain cem on cal.cal_from_cem_id = cem.cem_id
        inner join cardembostatus ces on ces.ces_id = cem.cem_from_ces_id
        where cac.cac_status in (:status_issued, :status_pin) and
              cac.cac_date between :rep_date_begin and :rep_date_end and
              cac.cac_from_reg_id = :reg_id
        order by 1
        into :fio, :card_id, :cca_code, :cac_date, :cca_id, :limit_tmp, :limit_date_tmp, :pinconvert, :c_id
    do
    begin
      limit = 0;
      --Если после выдачи лимит не менялся, то вывожу лимит, присвоенный привыдачи.
      select k.k_kontract_no, ccl.ccl_ol_date, ccl.ccl_new_limit, k.k_id, k.k_disb_date, k.k_fact_disb_lo_id,
             k.k_from_pr_id
      from cardclientacc_limit ccl
      inner join kontract k on k.k_id = ccl.ccl_from_k_id
      where ccl.ccl_from_cca_id = :cca_id and
            ccl.ccl_ol_date between :rep_date_begin and :rep_date_end
      order by 2 desc
      rows 1
      into :kontract_no, :limit_date, :limit, :k_id, :k_disb_date, :k_fact_disb_lo_id, k_from_pr_id;

      if (:limit = 0) then
      begin
        limit = :limit_tmp;
        limit_date = :limit_date_tmp;
      end
      select cl.c_name_fio
      from clients cl
      where cl.c_id=:k_fact_disb_lo_id
      into c_name_fio;
      
      select pr.pr_name
      from product pr
      where pr.pr_id = :k_from_pr_id
      into pr_name;

      suspend;

    end
  end
end
