CREATE PROCEDURE CM_RU_0144_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable Z_COST_KONTRACT CURR_TYPE;
declare variable Z_COST_KONTRACT_80 CURR_TYPE;
declare variable Z_COST_MARKET CURR_TYPE;
declare variable Z_COST_LIKVID CURR_TYPE;
declare variable KOL_ZALOG integer;
declare variable Z_SUBTYPE smallint;
declare variable Z_ADDRESS STR_255;
declare variable Z_FROM_C_ID ID_TYPE;
declare variable Z_FROM_ADL_ID ID_TYPE;
declare variable ADL_ADDRESS_STR STR_255;
declare variable REGISTRATION_PERMANENT smallint;
declare variable REGISTRATION_FIZ_DATE_END date;
declare variable PDT_CODE varchar(100);
declare variable CDS_DOC_DATE_END date;
declare variable REG_ID D_INTEGER;
begin
  for
  select k_disb_date, k_amount_approved, k_last_reg_id
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code containing '.zalog_car.'
  into kf_disb_date, kc_amount, reg_id
  do
  begin
    kol_zalog = null;
    select count(*)
    from zalog
    where z_from_k_id = :k_id
    and z_type = 1
    into kol_zalog;
    kol_zalog = coalesce(kol_zalog,0);
    if (kol_zalog = 0) then
    begin
      res_error_text = 'В базе залог не зарегистрирован, прошу зарегистрировать.';
      res_error_is_critical = 2;
      suspend;
      exit;
    end 
    if (kol_zalog > 1) then
    begin
      res_error_text = 'В базе зарегистрирован несколько залогов, прошу оставить только один';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
    z_cost_kontract = null;
    z_cost_market = null;
    z_cost_likvid = null;
    z_subtype = null;
    z_address = null;
    z_from_c_id = null;
    select z_cost_kontract, z_cost_market, z_cost_likvid, z_subtype, z_address, z_from_c_id, z_from_adl_id
    from zalog
    where z_from_k_id = :k_id
    and z_type = 1
    into z_cost_kontract, z_cost_market, z_cost_likvid, z_subtype, z_address, z_from_c_id, z_from_adl_id;
    z_cost_kontract = coalesce(z_cost_kontract,0);
    z_cost_market = coalesce(z_cost_market,0);
    z_cost_likvid = coalesce(z_cost_likvid,0);
    z_subtype = coalesce(z_subtype,0);
    z_address = coalesce(z_address,'');
    z_address = trim(z_address);
    if (z_from_c_id is null) then
    begin
      res_error_text = 'Не заполнен поле "Владелец", прошу заполнить.';
      res_error_is_critical = 2;
      suspend;
    end
    adl_address_str = '';
    select adl_address_str from addrlist
    where adl_id = :z_from_adl_id
    into adl_address_str;
    if (adl_address_str = '' or adl_address_str = 'не указан') then
    begin
      res_error_text = 'Не заполнен АДРЕС залога, прошу заполнить.';
      res_error_is_critical = 2;
      suspend;
    end 
    if (z_cost_kontract = 0) then
    begin
      res_error_text = 'Стоимость залога не введен, прошу ввести КОНТРАКТНУЮ стоимость залога.';
      res_error_is_critical = 2;
      suspend;
    end
    if (z_cost_market = 0) then
    begin
      res_error_text = 'Стоимость залога не введен, прошу ввести РЫНОЧНУЮ стоимость залога.';
      res_error_is_critical = 2;
      suspend;
    end
    if (z_cost_likvid = 0) then
    begin
      res_error_text = 'Стоимость залога не введен, прошу ввести ЛИКВИДАЦИОННУЮ стоимость залога.';
      res_error_is_critical = 2;
      suspend;
    end
    if (z_subtype <> 4) then
    begin
      res_error_text = 'Не правильно выбран вид залога, прошу выбрать "Транспортные средства"';
      res_error_is_critical = 2;
      suspend;
    end
    if (z_cost_kontract > 0) then
    begin
      if (exists(select * from region
        where reg_unicode = :reg_id
        and parent in (1010003, 1010005))) then
      begin
        z_cost_kontract_80 = round(cast(z_cost_kontract as double precision) * 90 / 100,0);
        res_error_text = '90';
      end else
      begin
        z_cost_kontract_80 = round(cast(z_cost_kontract as double precision) * 70 / 100,0);
        res_error_text = '70';
      end

      if (kc_amount > z_cost_kontract_80) then
      begin
        res_error_text = 'Сумма кредита не должно превышать '||res_error_text||'% от стоимости залога.';
        res_error_is_critical = 2;
        suspend;
      end
    end
--  если залогодатель не является клиентом
    if (exists(
    select * from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    and kc_from_c_id <> :z_from_c_id)) then
    begin
      registration_permanent = 0;
      registration_fiz_date_end = '01.01.1900';
      for
      select pdt_code,
--             pk_name,
--             cds_doc_date_begin,
             cds_doc_date_end
      from clientdocs, passkind, passdoctype
      where cds_from_c_id = :z_from_c_id
      and cds_is_delete = 0
      and coalesce(cds_doc_blob_1, 0) > 0
      and cds_from_pk_id = pk_id
      and pk_from_pdt_id = pdt_id
      and :kf_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.2999')
      into pdt_code,
--           pk_name,
--           cds_doc_date_begin,
           cds_doc_date_end
      do
      begin
        if (pdt_code = 'registration_permanent') then registration_permanent = 1;
        if (pdt_code = 'registration_fiz') then
        begin
--          registration_fiz = 1;
--          registration_fiz_date_begin = cds_doc_date_begin;
          registration_fiz_date_end = cds_doc_date_end;
        end
        if (pdt_code = 'registration_yur') then
        begin
--          registration_yur = 1;
--          registration_fiz_date_begin = cds_doc_date_begin;
          registration_fiz_date_end = cds_doc_date_end;
        end
      end
      if (registration_permanent = 0 and dateadd(month, 3, kf_disb_date) > registration_fiz_date_end) then
      begin
        res_error_text = 'Регистрация, должно быть не менее 3 месяцев от планируемой даты получения займа.';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end
  for
  select k_disb_date, k_amount_approved
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code not containing '.zalog_car.'
  and pr_code not containing '.zalog.'
  into kf_disb_date, kc_amount
  do
  begin
    kol_zalog = null;
    select count(*)
    from zalog
    where z_from_k_id = :k_id
    and z_type = 1
    into kol_zalog;
    kol_zalog = coalesce(kol_zalog,0);
    if (kol_zalog > 0) then
    begin
      res_error_text = 'По продукту не предусмотрено залог: очистите вкладку «Залог»';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
