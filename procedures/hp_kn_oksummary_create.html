CREATE PROCEDURE HP_KN_OKSUMMARY_CREATE(
    K_ID D_ID NOT NULL)
AS
declare variable ol_date d_date;
declare variable ks_date_begin d_date;
declare variable ks_date_end d_date;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable tmp_date d_date;
begin
  -- Создает полную историю данных в таблице kn_oksummary.
  if ((select is_kn_mode
       from vbs_is_kn_mode(:k_id)) = 0) then
    Exit;

  delete from kn_oksummary
  where oks_from_k_id = :k_id
  order by oks_from_kc_id desc, oks_date_begin desc;

  select ks_date_begin, ks_date_end
  from kontstate
  where ks_from_k_id = :k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0
  into ks_date_begin, ks_date_end;

  if (ks_date_begin is null) then
    exit;

  if (ks_date_end < '31.12.2999') then
    ks_date_end = ks_date_end + 1;

  for select kc_id, kc_from_c_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into kc_id, c_id
  do
  begin
    ol_date = null;

    for select distinct ok_ol_date
    from kn_operkred
    where ok_from_kc_id = :kc_id and
          not(ok_type = 9 and ok_ol_date > :ks_date_end) and
          ok_ol_is_active = 1
    union
    select ok_sh_date
    from kn_operkred
    where ok_from_kc_id = :kc_id and
          ok_ol_is_active = 1
    union
    select oko_ol_date
    from operkroff
    where oko_from_k_id = :k_id and
          oko_from_c_id = :c_id and
          oko_ol_is_active = 1
    union
    select ox_ol_date
    from kn_operextra
    where ox_from_kc_id = :kc_id and
          ox_ol_is_active = 1
    union
    select or_ol_date
    from operrest
    where or_from_k_id = :k_id and
          or_ol_is_active = 1
    union
    select kd_ol_date
    from kontdata
    where kd_from_k_id = :k_id and
          kd_ol_is_delete = 0
    union
    select ks_date_begin
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          ks_state = 2
    union
    select f_date_month_end(cast('01.' || right(kcp_period, 2) || '.' || left(kcp_period, 4) as date))
    from kn_kontcalcperc
    where kcp_from_kc_id = :kc_id
    into ol_date
    do
    begin
      if (ol_date is null) then
        continue;
  
      execute procedure hp_kn_oksummary_insert(:kc_id, :ol_date, 0);
    end

    if (ol_date is not null) then
    begin
      tmp_date = f_date_month_end(ol_date);
      execute procedure hp_kn_oksummary_insert(:kc_id, :tmp_date, 0);
    end

    execute procedure hp_kn_oksummary_recalc(:kc_id, :ks_date_begin);
  end
end
