CREATE PROCEDURE HP_EDOC_INS_INSURANCE_POLICY(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable k_status d_smallint;
declare variable k_disb_date d_date;
declare variable enum_status d_smallint;
declare variable ag_id d_integer;
declare variable edc_code d_str_255;
declare variable pr_code d_str_100;
declare variable commission_amount d_currency;
declare variable mbulak_share d_currency;
declare variable policy_company_share d_currency;
declare variable tax d_currency;
declare variable is_checked d_smallint;
declare variable c_id d_id;
declare variable tarif perc_type;
declare variable k_score d_integer;
declare variable pts_date_begin d_date;
declare variable pts_date_end d_date;
begin
    res_text = '';
    res_is_error = 0;

    select k.k_status, k.k_disb_date, pr.pr_code
    from kontract k
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :k_id
    into :k_status, :k_disb_date, :pr_code;

    if (res_text = '' and pr_code containing '.ignore_insurance.') then
        res_text = 'По продукту не предусмотрено страхование';

    if (res_text = '' and k_status <> 1) then
      res_text = 'Сначала утвердите договор займа';

   -- RFCKG-5481
   if (res_text = ''
        and exists
        (
            select *
            from edoc_list edl
            inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
                and edc.edc_code in ('insurance.application', 'insurance.kyrgyz', 'insurance.kyrgyz.ipf')
            where edl.edl_object_kind = f_objkind('kontract', 'id')
                and edl.edl_object_id = :k_id
                and edl.edl_status = 1
                and edl.edl_blank_date = current_date
        )
    ) then
        res_text = 'Документ уже подписан';
    
    if (res_text = ''
        and exists
        (
            select *
            from edoc_list edl
            inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
                and edc.edc_code in ('insurance.application', 'insurance.kyrgyz', 'insurance.kyrgyz.ipf')
            where edl.edl_object_kind = f_objkind('kontract', 'id')
                and edl.edl_object_id = :k_id
                and edl.edl_status = 0
                and edl.edl_blank_date = current_date
        )
    ) then
        res_text = 'Документ уже отправлен';

    if (res_text = '') then
    begin
        select
            c_id, k_score, tarif,
            commission_amount, mbulak_share, policy_company_share,
            tax, ag_id, edc_code,
            pts_date_begin, pts_date_end
        from hp_get_insurance_params(:k_id)
        into
            :c_id, :k_score, :tarif,
            :commission_amount, :mbulak_share, :policy_company_share,
            :tax, :ag_id, :edc_code,
            :pts_date_begin, :pts_date_end;

        is_checked = 1;


        select edl_id, res_text
        from hp_edoc_insert(:edc_code, f_objkind('kontract', 'id'), :k_id, :un_id)
        into edl_id, res_text;

        if (res_text = '') then
        begin
            enum_status = f_enum_id('insurance_kontract','edoc_created');

            insert into insur_logs (il_from_k_id, il_tarif, il_k_score, il_pts_date_begin, il_pts_date_end, il_create_datetime)
            values (:k_id, :tarif, :k_score, :pts_date_begin, :pts_date_end, current_timestamp);

            -- RFCKG-3872
            update or insert into insurance_kontract (ik_from_k_id, ik_from_ag_id, ik_enum_status, ik_policy_no, ik_commission, ik_mbulak_share, ik_policy_company_share, ik_tax, ik_is_checked)
            values (:k_id, :ag_id, :enum_status, '', coalesce(:commission_amount, 0), coalesce(:mbulak_share, 0), coalesce(:policy_company_share, 0), coalesce(:tax, 0), coalesce(:is_checked, 0))
            matching (ik_from_k_id);
        end
    end

    suspend;
end
