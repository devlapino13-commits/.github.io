CREATE PROCEDURE AAREP_HIGH_RISK_CLIENTS(
    REP_DATE D_DATE)
RETURNS (
    OFFICE_NAME D_STR_100,
    CL_FIO D_STR_100,
    KONT_NO D_STR_100,
    KF_FIO D_STR_100)
AS
declare variable c_id d_id;
declare variable kf_id d_id;
declare variable temp_reg_id d_id;
declare variable reg_id d_id;
declare variable k_id d_id;
declare variable ma_id d_id;
begin
  for select kc.kc_from_c_id, k.k_kontract_no, k.k_from_kf_id, c.c_name_fio
      from kontstate ks
      inner join kontract k on k.k_id = ks.ks_from_k_id
      inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
      inner join clients c on c.c_id = kc.kc_from_c_id and
            c.c_pft_risk_koef = 2
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            kc.kc_amount_approved > 0
      into :c_id, kont_no, :kf_id, :cl_fio
  do
  begin
    ma_id = null;
    select ma.ma_id
    from miniappclient mac
    inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
    where mac.mac_from_c_id = :c_id and
          ma.ma_is_active = 1 and
          ma.ma_date_end = '31.12.9999'
    order by ma.ma_date_app desc
    rows 1
    into :ma_id;
    if (not exists(select *
                   from operkroff oko
                   where oko.oko_from_k_id = :k_id and
                         oko.oko_ol_is_active = 1) and
        exists(select *
               from edoc_list edl
               where edl.edl_object_kind = 11 and
                     edl.edl_object_id = :ma_id and
                     edl.edl_status = 1 and
                     edl.edl_from_edc_id = 2 and
                     datediff(month, edl.edl_blank_date, :rep_date) >= 11)

        ) then
    begin
      office_name = null;
      kf_fio = null;
      select c.c_name_fio, reg.reg_name
      from knaktlist kal
      inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
      inner join clients c on c.c_id = kal.kal_new_kf_officer
      where kal.kal_from_kf_id = :kf_id and
            kal.is_delete = 0 and
            :rep_date between kal.kal_date and kal.kal_date_end
      into :kf_fio, :office_name;
      suspend;
    end
  end
end
