CREATE PROCEDURE CM_RU_0023_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kf_disb_date d_date;
declare variable K_PERCENT_EFFECT PERC_TYPE;
declare variable K_PERIOD integer;
declare variable K_FROM_PR_ID integer;
declare variable PR_CODE varchar(100);
declare variable PR_NAME varchar(50);
declare variable MIN_PSK PERC_TYPE;
declare variable MAX_PSK PERC_TYPE;
declare variable K_KREDLINE_END date;
begin
  K_PERCENT_EFFECT = null;
  select K_PERCENT_EFFECT_FIFD, K_PERIOD, K_FROM_PR_ID, K_DISB_DATE, K_KREDLINE_END
  from KONTRACT
  where K_ID = :K_ID
  into K_PERCENT_EFFECT, K_PERIOD, K_FROM_PR_ID, KF_DISB_DATE, K_KREDLINE_END;
  K_PERCENT_EFFECT = coalesce(K_PERCENT_EFFECT, 0);
  if (K_KREDLINE_END - KF_DISB_DATE > 365) then
  begin
    res_error_is_critical = 1;
    res_error_text = 'Общий срок займа не должен превышать 365 дней';
    suspend;
  end
  select PR_CODE, PR_NAME
  from PRODUCT
  where PR_ID = :K_FROM_PR_ID
  into PR_CODE, PR_NAME;
  if (K_PERCENT_EFFECT is null) then
    K_PERCENT_EFFECT = 0;
  if (K_PERCENT_EFFECT <= 0) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'ПСК не правильно посчитан, заново перестройте график и попробуйте утвердить!';
    suspend;
  end
  if (KF_DISB_DATE >= cast('01.07.2015' as date) and
      KF_DISB_DATE <= cast('22.10.2015' as date) -- Добавлено по письму Ироды от 22.10.2015
      ) then
  begin
    if (K_PERCENT_EFFECT > 244.00) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'ПСК не должно превышать от 244%';
      suspend;
    end
  end
  else
  if (KF_DISB_DATE <= cast('01.01.2016' as date)) then
  begin
    select PR_CODE, PR_NAME
    from PRODUCT
    where PR_ID = :K_FROM_PR_ID
    into PR_CODE, PR_NAME;
    if (PR_CODE containing '.BARAKAT.') then
    begin
      MIN_PSK = 69.598;
      MAX_PSK = 121.232;
    end
    else
    if (PR_CODE containing '.BARAKAT PLUS.') then
    begin
      MIN_PSK = 59.502;
      MAX_PSK = 109.230;
    end
    else
    if (PR_CODE containing '.FAMILY.') then
    begin
      MIN_PSK = 35.944;
      MAX_PSK = 85.366;
    end
    else
    if (PR_CODE containing '.FAMILY PLUS.') then
    begin
      MIN_PSK = 50.358;
      MAX_PSK = 99.322;
    end
    else
    if (PR_CODE containing '.ONE.') then
    begin
      MIN_PSK = 76.053;
      MAX_PSK = 226.870;
    end
    else
    if (PR_CODE containing '.PAR.') then
    begin
      MIN_PSK = 30.377;
      MAX_PSK = 107.945;
    end
    else
    if (PR_CODE containing '.EMP.') then
    begin
      MIN_PSK = 35.649;
      MAX_PSK = 44.159;
    end
    else
    if (PR_CODE containing '.GR.') then
    begin
      MIN_PSK = 61.464;
      MAX_PSK = 200.387;
    end
    else
    begin
      MIN_PSK = 0;
      MAX_PSK = 244.00;
    end
    if (K_PERCENT_EFFECT > MAX_PSK) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Для кр. продукта:' || :PR_NAME || ' ПСК не должно превышать ' || :MAX_PSK || '%';
      suspend;
    end
    else
    if (K_PERCENT_EFFECT < MIN_PSK) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Для кр. продукта:' || :PR_NAME || ' ПСК не должно быть ниже ' || :MIN_PSK || '%';
      suspend;
    end
  end
  else
  if (:KF_DISB_DATE < cast('01.04.2017' as date)) then
  begin
    if (PR_CODE containing '.BARAKAT.') then
    begin
      MIN_PSK = 85.958;
      MAX_PSK = 124.765;
    end
    else
    if (PR_CODE containing '.BARAKAT PLUS.') then
    begin
      MIN_PSK = 76.366;
      MAX_PSK = 112.257;
    end
    else
    if (PR_CODE containing '.FAMILY.') then
    begin
      MIN_PSK = 54.674;
      MAX_PSK = 85.331;
    end
    else
    if (PR_CODE containing '.FAMILY PLUS.') then
    begin
      MIN_PSK = 67.849;
      MAX_PSK = 103.701;
    end
    else
    if (PR_CODE containing '.ONE.') then
    begin
      MIN_PSK = 76.053;
      MAX_PSK = 226.870;
    end
    else
    if (PR_CODE containing '.PAR.') then
    begin
      MIN_PSK = 54.693;
      MAX_PSK = 82.945;
    end
    else
    if (PR_CODE containing '.PATENT PLUS.') then
    begin
      MIN_PSK = 85.870;
      MAX_PSK = 88.618;
    end
  end
  else
  if (:KF_DISB_DATE between cast('01.04.2017' as date) and cast('30.04.2017' as date)) then
  begin
    MIN_PSK = 0;
    MAX_PSK = 244.00;
    if (PR_CODE containing '.GR.') then
    begin
      if (PR_CODE containing '.FAMILY.') then --«Семья»
      begin
        MIN_PSK = 54.482;
        MAX_PSK = 85.3;
      end
      else
      if (PR_CODE containing '.FAMILY PLUS.') then --«Семья Плюс»
      begin
        MIN_PSK = 65.486;
        MAX_PSK = 103.676;
      end
      else
      if (PR_CODE containing '.PATENT.') then --«Групповой микрозаем на патент»
      begin
        MIN_PSK = 101.585;
        MAX_PSK = 108.416;
      end
    end
    if (PR_CODE containing '.IND.') then
    begin
      if (PR_CODE containing '.OB.') then --«Микрозаем сотруднику»
      begin
        MIN_PSK = 28.336;
        MAX_PSK = 34.407;
      end
      else
      if (PR_CODE containing '.FOREIGN.') then --«Микрозаем  иностранному сотруднику»
      begin
        MIN_PSK = 23.252;
        MAX_PSK = 29.344;
      end
      else
      if (PR_CODE containing '.PAR.') then --Микрозаем по программе лояльности
      begin
        MIN_PSK = 54.681;
        MAX_PSK = 82.945;
      end
      else
      if (PR_CODE containing '.BARAKAT.') then --«Баракат»
      begin
        MIN_PSK = 82.294;
        MAX_PSK = 124.748;
      end
      else
      if (PR_CODE containing '.BARAKAT PLUS.') then --«Баракат Плюс»
      begin
        MIN_PSK = 73.888;
        MAX_PSK = 112.295;
      end
      else
      if (PR_CODE containing '.PATENT.') then --«Микрозаем на патент»
      begin
        MIN_PSK = 85.9;
        MAX_PSK = 125.449;
      end
      else
      if (PR_CODE containing '.PATENT PLUS.') then --«Микрозаем на патент Плюс»
      begin
        MIN_PSK = 78.713;
        MAX_PSK = 81.757;
      end
      else
      if (PR_CODE containing '.WINNER.') then --«Микрозаем для победителей»
      begin
        MIN_PSK = 19.251;
        MAX_PSK = 25.277;
      end
    end
  end
  else
  if (:KF_DISB_DATE >= cast('01.05.2017' as date)) then
  begin
    MIN_PSK = 0;
    MAX_PSK = 244.00;
    if (PR_CODE containing '.GR.') then
    begin
      if (PR_CODE containing '.FAMILY.') then --«Семья»
      begin
        if (KF_DISB_DATE < '16.10.2017') then
        begin
          MIN_PSK = 54.482;
          MAX_PSK = iif(KF_DISB_DATE < '07.06.17', 85.305, 106.460);
          if (KF_DISB_DATE between '05.10.17' and '31.10.17') then
            MIN_PSK = 53.482;
        end
        else
        begin
          MIN_PSK = 73.188;
          MAX_PSK = 106.460;
        end
      end
      else
      if (PR_CODE containing '.FAMILY PLUS.') then --«Семья Плюс»
      begin
        if (KF_DISB_DATE < '16.10.2017') then
        begin
          MIN_PSK = 65.486;
          MAX_PSK = iif(KF_DISB_DATE < '07.06.17', 104.933, 120.890);
          if (KF_DISB_DATE >= '03.10.17') then
            MAX_PSK = 120.130;
          if (KF_DISB_DATE between '05.10.17' and '31.10.17') then
            MIN_PSK = 64.486;
        end
        else
        begin
          MIN_PSK = 88.761;
          MAX_PSK = 120.130;
        end
      end
      else
      if (PR_CODE containing '.PATENT.') then --«Групповой микрозаем на патент»
      begin
        MIN_PSK = 101.585;
        MAX_PSK = 108.426;
      end
    end
    if (PR_CODE containing '.IND.') then
    begin
      if (PR_CODE containing '.BARAKAT.') then --«Баракат»
      begin
        if (KF_DISB_DATE < '16.10.2017') then
        begin
          MIN_PSK = 82.294;
          MAX_PSK = iif(KF_DISB_DATE < '07.06.17', 126.719, 128.040);
          if (KF_DISB_DATE between '05.10.17' and '31.10.17') then
            MIN_PSK = 81.294;
        end
        else
        begin
          MIN_PSK = 95.188;
          MAX_PSK = 128.040;
        end
      end
      else
      if (PR_CODE containing '.BARAKAT PLUS.') then --«Баракат Плюс»
      begin
        if (KF_DISB_DATE < '16.10.2017') then
        begin
          MIN_PSK = 73.888;
          MAX_PSK = iif(KF_DISB_DATE < '07.06.17', 113.526, 117.195);
          if (KF_DISB_DATE between '05.10.17' and '31.10.17') then
            MIN_PSK = 72.888;
        end
        else
        begin
          MIN_PSK = 85.147;
          MAX_PSK = 117.195;
        end
      end
      else
      if (PR_CODE containing '.PATENT.') then --«Микрозаем на патент»
      begin
        MIN_PSK = 85.9;
        MAX_PSK = 125.462;
      end
      else
      if (PR_CODE containing '.FOREIGN.') then --«Микрозаем  иностранному сотруднику»
      begin
        MIN_PSK = 23.252;
        MAX_PSK = 29.344;
      end
      else
      if (PR_CODE containing '.OB.') then --«Микрозаем сотруднику»
      begin
        MIN_PSK = 28.336;
        MAX_PSK = 34.407;
      end
      else
      if (PR_CODE containing '.PATENT PLUS.') then --«Микрозаем на патент Плюс»
      begin
        if (KF_DISB_DATE < '01.10.17') then
        begin
          MIN_PSK = 78.713;
          if (:KF_DISB_DATE < '01.07.17') then
            MAX_PSK = 81.757;
          else
          if (:KF_DISB_DATE < '01.01.18') then
            MAX_PSK = 69.990;
          else
            MAX_PSK = 80.145;
        end
        else   
        if (KF_DISB_DATE < '01.04.18') then
        begin
          --макс ПСК  75,342%,  мин. ПСК 72,436 %.
          MIN_PSK = 72.436;
          MAX_PSK = 75.342;
        end else          
        if (KF_DISB_DATE < '01.07.18') then
        begin
          /*«Микрозаем на патент Плюс» и «Микрозаем на патент Плюс параллельный» 5.7 % в месяц
            Мин. ПСК 67,850%, Макс. ПСК 69,675% */
          MIN_PSK = 67.850;
          MAX_PSK = 69.675;
        end else
        begin
          /*с 1 июля 2018г. по продуктам «Микрозаем на патент Плюс» и «Микрозаем на патент плюс параллельный»:
            Ставка – 4,95 % в месяц
            Минимальный ПСК – 58.820% годовых
            Максимальный ПСК – 60.410% годовых
             */
          MIN_PSK = 58.820;
          MAX_PSK = 60.410;
        end
      end
      else
      if (PR_CODE containing '.PAR.') then --Микрозаем по программе лояльности
      begin
        if (KF_DISB_DATE < '16.10.17') then
        begin
          MIN_PSK = 54.681;
          MAX_PSK = 83.945;
        end
        else
        begin
          if (:KF_DISB_DATE between '01.01.18' and '31.03.18' and
              PR_CODE containing '.PATENT PLUS.') then --«Микрозаем на патент Плюс»
            MAX_PSK = 69.990;
          else
          if (PR_CODE not containing '.PATENT PLUS.') then --«Микрозаем на патент Плюс» условия ПСК по данному продукту совпадают с основным
          begin
            MIN_PSK = 73.188;
            MAX_PSK = 128.040;
          end
        end
      end
      else
      if (PR_CODE containing '.WINNER.') then --«Микрозаем для победителей»
      begin
        MIN_PSK = 19.251;
        MAX_PSK = 25.277;
      end
    end
  end
  if (K_PERCENT_EFFECT > MAX_PSK) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Для кр. продукта: "' || :PR_NAME || '" ПСК не должно превышать ' || :MAX_PSK || '%';
    suspend;
  end
  else
  if (K_PERCENT_EFFECT < MIN_PSK) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Для кр. продукта: "' || :PR_NAME || '" ПСК не должно быть ниже ' || :MIN_PSK || '%';
    suspend;
  end
end
