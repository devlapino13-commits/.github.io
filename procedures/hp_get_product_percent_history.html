CREATE PROCEDURE HP_GET_PRODUCT_PERCENT_HISTORY(
    PR_ID D_INTEGER,
    PRL_ID D_INTEGER)
RETURNS (
    C_NAME_FIO D_STR_100,
    PRR_DATETIME D_DATETIME,
    RES_TEXT D_STR_8100)
AS
declare variable prr_id d_id;
declare variable prr_enum_status d_integer;
declare variable prr_parent d_id;
declare variable prr_min_point d_integer;
declare variable prr_max_point d_integer;
declare variable prr_min_period d_integer;
declare variable prr_max_period d_integer;
declare variable prr_percent d_currency;
declare variable prr_percent_2 d_currency;
declare variable prr_percent_7 d_currency;
declare variable tmp_prr_min_point d_integer;
declare variable tmp_prr_max_point d_integer;
declare variable tmp_prr_min_period d_integer;
declare variable tmp_prr_max_period d_integer;
declare variable tmp_prr_percent d_currency;
declare variable tmp_prr_percent_2 d_currency;
declare variable tmp_prr_percent_7 d_currency;
declare variable prr_is_actual d_smallint;
begin
  -- RFCKG-3640 История изменений для модуля «Процентная ставка»
  res_text = '';

  for select prr.prr_id,
             c.c_name_fio,
             prr.prr_datetime,
             prr.prr_enum_status,
             prr.prr_parent,
             prr.prr_min_point,
             prr.prr_max_point,
             prr.prr_min_period,
             prr.prr_max_period,
             prr.prr_percent,
             prr.prr_percent_2,
             prr.prr_percent_7,
             prr.prr_is_actual
  from product_rates prr
  inner join username un on un.un_id = prr.prr_from_un_id
  inner join clients c on c.c_id = un.un_from_c_id
  where prr.prr_from_pr_id = :pr_id
    and prr.prr_from_prl_id = :prl_id
  into prr_id,
       c_name_fio,
       prr_datetime,
       prr_enum_status,
       prr_parent,
       prr_min_point,
       prr_max_point,
       prr_min_period,
       prr_max_period,
       prr_percent,
       prr_percent_2,
       prr_percent_7,
       prr_is_actual
  do
  begin
    if (prr_is_actual = 0) then
    begin
      if (prr_enum_status = 1) then
        res_text = f_format('Запись %1 удалена пользователем', prr_id);
  
      if (prr_enum_status = 3) then
        res_text = f_format('Запись %1 удалена при импортировании', prr_id);

      if (prr_enum_status = 4) then
        res_text = f_format('Запись %1 удалена при импортировании из Excel', prr_id);  -- RFCKG-4737
  
      if (prr_enum_status = 2) then
      begin
        select prr.prr_min_point,
               prr.prr_max_point,
               prr.prr_min_period,
               prr.prr_max_period,
               prr.prr_percent,
               prr.prr_percent_2, 
               prr.prr_percent_7
        from product_rates prr
        where prr.prr_parent = :prr_id
        into tmp_prr_min_point,
             tmp_prr_max_point,
             tmp_prr_min_period,
             tmp_prr_max_period,
             tmp_prr_percent,
             tmp_prr_percent_2,
             tmp_prr_percent_7;

        if(tmp_prr_min_point <> prr_min_point or
           tmp_prr_max_point <> prr_max_point or
           tmp_prr_min_period <> prr_min_period or
           tmp_prr_max_period <> prr_max_period or
           tmp_prr_percent <> prr_percent or
           tmp_prr_percent_2 <> prr_percent_2 or
           tmp_prr_percent_7 <> prr_percent_7) then
          res_text = res_text || f_format('Запись %1 ', prr_id) || ASCII_CHAR(13) || ASCII_CHAR(10);
  
        if (tmp_prr_min_point <> prr_min_point) then
          res_text = res_text || f_format('Изменен %1. Старое значение (%2) Новое значение (%3)', 'мин балл клиента', prr_min_point, tmp_prr_min_point) || ASCII_CHAR(13) || ASCII_CHAR(10);
  
        if (tmp_prr_max_point <> prr_max_point) then
          res_text = res_text || f_format('Изменен %1. Старое значение (%2) Новое значение (%3)', 'макс балл клиента', prr_max_point, tmp_prr_max_point) || ASCII_CHAR(13) || ASCII_CHAR(10);
  
        if (tmp_prr_min_period <> prr_min_period) then
          res_text = res_text || f_format('Изменен %1. Старое значение (%2) Новое значение (%3)', 'мин период', prr_min_period, tmp_prr_min_period) || ASCII_CHAR(13) || ASCII_CHAR(10);
  
        if (tmp_prr_max_period <> prr_max_period) then
          res_text = res_text || f_format('Изменен %1. Старое значение (%2) Новое значение (%3)', 'макс период', prr_max_period, tmp_prr_max_period) || ASCII_CHAR(13) || ASCII_CHAR(10);
  
        if (tmp_prr_percent <> prr_percent) then
          res_text = res_text || f_format('Изменен %1. Старое значение (%2) Новое значение (%3)', 'процентная ставка', round(prr_percent, 2), round(tmp_prr_percent, 2)) || ASCII_CHAR(13) || ASCII_CHAR(10);

        if (tmp_prr_percent_2 <> prr_percent_2) then
          res_text = res_text || f_format('Изменен %1. Старое значение (%2) Новое значение (%3)', 'процентная ставка', round(prr_percent_2, 2), round(tmp_prr_percent_2, 2)) || ASCII_CHAR(13) || ASCII_CHAR(10);

        if (tmp_prr_percent_7 <> prr_percent_7) then
          res_text = res_text || f_format('Изменен %1. Старое значение (%2) Новое значение (%3)', 'процентная ставка', round(prr_percent_7, 2), round(tmp_prr_percent_7, 2)) || ASCII_CHAR(13) || ASCII_CHAR(10);
      end
    end

    if (prr_is_actual = 1) then
    begin
      res_text = f_format('Запись %1 создана пользователем', prr_id);
    end

    suspend;
  end
end
