CREATE PROCEDURE HP_GDFL_PASS_CHECK(
    C_ID D_ID)
RETURNS (
    DESCRIPTION D_STR_255)
AS
declare variable JSON_STR D_BLOB_TEXT;
declare variable C_PASS_NO D_STR_CODE;
declare variable ARR_LENGTH D_INTEGER;
declare variable I D_INTEGER;
declare variable DOCUMENT D_STR_4096;
declare variable TYPE_CODE D_INTEGER;
declare variable STATUS_CODE D_INTEGER;
declare variable DOCUMENT_END_DATE D_DATE;
declare variable DOCUMENT_PASS_NO D_STR_CODE;
declare variable bls_id D_ID;
begin
  -- RFCKG-4180 Процедура сверяет паспорт записанный в clients с ответом полученным от ГДФЛ
  c_pass_no = trim((select c.c_pass_no from clients c where c.c_id = :c_id));
  i = 0;
  description = '';

  select eil.eil_from_bls_id
  from extinfolist eil
  join extinfocfg ext on ext.eic_id = eil.eil_from_eic_id
     and ext.eic_code in ('gbdfl_get_cli_data', 'bioguard_info')
  where eil.eil_status = 2
    and eil.eil_from_c_id = :c_id
    and eil.eil_date = current_date
    and coalesce(eil.eil_from_bls_id, '') <> ''
  order by eil.eil_status_datetime desc
  rows 1
  into :bls_id;

  if (:bls_id is not null) then
  begin
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into json_str;
  end

  if (coalesce(json_str, '') = '') then
  begin
    description = 'Запрос в ГДФЛ не отправлен.';
    suspend;
    exit;
  end

  arr_length = f_json_blob_get_count(json_str, 'documents.document');

  if (arr_length = -1) then
    exit;

  while (arr_length <> i)
  do
  begin

    document = f_json_blob_get_value(json_str, f_format('documents.document[%1]', i), null);  -- получаем паспорт клиента
    i = i + 1;

    document_pass_no = trim(f_json_blob_get_value(document, 'number', null));

    if(document_pass_no <> c_pass_no) then
      continue;

    document_end_date = cast(f_json_blob_get_value(document, 'end_date', null) as d_date);
    type_code = cast(f_json_blob_get_value(document, 'type.code', null) as d_integer);
    status_code = cast(f_json_blob_get_value(document, 'status.code', null) as d_integer);

    if (status_code in (1, 5, 7, 8, 9, 10, 11, 12, 13, 14, 16, 17, 18, 19, 20, 21)) then
      description = cast(f_json_blob_get_value(document, 'status.name_ru', '') as d_str_255);

    if (current_date >= (document_end_date - 14)) then  -- Если срок годности по удостоверению личности истекает через 14 дней или раньше
      description = f_format('Срок годности по удостоверения личности истекает через %1 дней', (document_end_date - current_date));

    break;
  end

  suspend;  
end
