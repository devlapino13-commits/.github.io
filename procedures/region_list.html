CREATE PROCEDURE REGION_LIST(
    ID INTEGER,
    SELF_INCLUDE INTEGER,
    KIND INTEGER,
    IS_DOWN INTEGER)
RETURNS (
    ID_LIST INTEGER,
    ID_PARENT INTEGER,
    PARENT_NAME VARCHAR(50))
AS
begin
  if (((id = 0) or (id = 1000000)) and
      (is_down = 1)) then
  begin
    for select r1.reg_unicode, coalesce(r1.parent, 1000000), coalesce(r2.reg_name, '')
        from region r1
        left join region r2 on (r1.parent = r2.reg_unicode)
        where ((:kind = -1) or
              (:kind <> -1 and
              r1.reg_status = :kind))
        into :id_list, :id_parent, parent_name
    do
    begin
      suspend;
    end
  end
  else
  begin
    if (is_down = 1) then
    begin
      for with recursive reg_list
          as (select reg_unicode, parent, reg_status
              from region
              where reg_unicode = :id
              union all
              select r.reg_unicode, r.parent, r.reg_status
              from region r
              join reg_list on (reg_list.reg_unicode = r.parent))
          select reg_unicode, parent
          from reg_list
          where ((:self_include = 0 and
                reg_unicode <> :id) or
                (:self_include = 1)) and
                ((:kind = -1) or
                (:kind <> -1 and
                reg_status = :kind))
          into :id_list, :id_parent
      do
      begin
        if (id_parent is not null) then
        begin
          select distinct reg_name
          from region
          where reg_unicode = :id_parent rows 1
          into :parent_name;
        end
        else
        begin
          id_parent = 1000000;
          parent_name = '';
        end
        suspend;
      end
    end
    else
    begin
      for with recursive reg_list
          as (select reg_unicode, parent, reg_status
              from region
              where reg_unicode = :id
              union all
              select r.reg_unicode, r.parent, r.reg_status
              from region r
              join reg_list on (reg_list.parent = r.reg_unicode))
          select reg_unicode, parent
          from reg_list
          where ((:self_include = 0 and
                reg_unicode <> :id) or
                (:self_include = 1)) and
                ((:kind = -1) or
                (:kind <> -1 and
                reg_status = :kind))
          into :id_list, :id_parent
      do
      begin
        if (id_parent is not null) then
        begin
          select distinct reg_name
          from region
          where reg_unicode = :id_parent rows 1
          into :parent_name;
        end
        else
        begin
          id_parent = 1000000;
          parent_name = '';
        end
        suspend;
      end
    end
  end
end
