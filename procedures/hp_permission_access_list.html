CREATE PROCEDURE HP_PERMISSION_ACCESS_LIST(
    UN_ID DECIMAL(18,0),
    SUBJECT_KIND VARCHAR(100))
RETURNS (
    RES_SUBJECT_ID DECIMAL(18,0))
AS
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
declare variable subject_kind_id integer;
begin
-- Возвращает список субьектов доступных пользователю
  if (subject_kind = '2') then
    subject_kind = 'Menulist';
  if (subject_kind = '3') then
    subject_kind = 'PermActions';
  if (subject_kind = '4') then
    subject_kind = 'Region';

  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id(:subject_kind)
  into :subject_kind_id;

  for with tab
      as (select distinct un.pm_subject_id as sub_kind, un.pm_access_kind as access_kind
          from permissions un
          where un.pm_subject_kind = :subject_kind_id and
                un.pm_object_kind = :uniqtab_username and
                un.pm_object_id = :un_id and
                un.pm_is_delete = 0 and
                (coalesce(un.pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(un.pm_date_end, '31.12.2999') >= current_date)
          union
          select distinct ro.pm_subject_id as sub_kind, ro.pm_access_kind as access_kind
          from permissions un, permissions ro
          where ro.pm_subject_kind = :subject_kind_id and
                ro.pm_is_delete = 0 and
                (coalesce(ro.pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(ro.pm_date_end, '31.12.2999') >= current_date) and
                ro.pm_object_kind = :uniqtab_rolename and
                ro.pm_object_id = un.pm_subject_id and
                un.pm_subject_kind = :uniqtab_rolename and
                un.pm_object_kind = :uniqtab_username and
                un.pm_object_id = :un_id and
                un.pm_is_delete = 0 and
                (coalesce(un.pm_date_begin, '01.01.2000') <= current_date) and
                (coalesce(un.pm_date_end, '31.12.2999') >= current_date))
      select sub_kind
      from tab
      group by sub_kind
      having min(access_kind) = 1
      into :res_subject_id
  do
  begin
    suspend;
  end
end
