CREATE PROCEDURE FM_EDOC_LIST_TMP(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_EDL_ID D_ID)
AS
declare variable obj_kind_id d_integer;
declare variable can_suspend d_smallint;
declare variable edl_object_kind d_integer;
declare variable edl_object_id d_id;
declare variable k_id d_id;
begin
  obj_kind = f_objkind(obj_kind, 'name');
  obj_kind_id = f_objkind(obj_kind, 'id');

  if (obj_kind = 'edoc_list') then
  begin
    res_edl_id = obj_id;
    suspend;
  end
  else if (obj_kind = 'clients') then
  begin
    for select edl_id, edl_object_kind, edl_object_id
    from edoc_sign, edoc_list
    where eds_from_c_id = :obj_id and
          eds_from_edl_id = edl_id and
          edl_status >= 0
    order by edl_create_datetime desc
    into res_edl_id, edl_object_kind, edl_object_id
    do
    begin
      can_suspend = 1;
      k_id = null;
      -- Для договоров займа отображаем только активные
      if (edl_object_kind = 9) then
        k_id = edl_object_id;

      if (edl_object_kind = 11) then
      begin
        select k_id from kontract
        where k_from_ma_app_id = :edl_object_id
        order by k_kontract_date desc
        rows 1
        into k_id;
      end

      if (k_id is not null and not exists(
        select * from kontstate
        where ks_from_k_id = :k_id and
              ks_state in (0, 1) and
              ks_date_end >= '31.12.2999' and
              ks_ol_is_delete = 0)) then
        can_suspend = 0;

      if (can_suspend = 1) then
        suspend;
    end
  end else
  begin
    for select edl_id
    from edoc_list
    where edl_object_kind = :obj_kind_id and
          edl_object_id = :obj_id and
          edl_status >= 0
    order by edl_create_datetime desc
    into res_edl_id
    do
      suspend;
  end
end
