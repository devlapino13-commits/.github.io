CREATE PROCEDURE HP_EDOC_BLANK_SIGN(
    EDS_ID D_ID,
    BLANK_LIST D_STR_LONG)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable edb_id d_id;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  res_code = 0;
  res_text = '';
  -- Проверка, что все бланки, обязательные для подписания есть в списке blank_list
  if (exists(
    select *
    from edoc_sign
        left join edoc_blank on edb_from_edl_id = eds_from_edl_id
        left join edoc_frx on edf_id = edb_from_edf_id
        left join hp_split_str(:blank_list, ',') on res_str = edb_id
    where eds_id = :eds_id and
          edf_optional_sign = 0 and
          (edb_from_c_id is null or edb_from_c_id = eds_from_c_id) and
          res_str is null)) then
  begin
    res_code = 1;
    res_text = 'Не все обязательные для подписания документы были отмечены.';
  end
  -- RFCRU-5900 для РФ (db_id = 81001) блок "После окончания периода охлаждения отказаться нельзя" отключён
  if (res_text = '' and db_id <> 81001) then
  begin
    if (exists(select edl_id
               from edoc_list
               inner join edoc_sign on eds_from_edl_id = edl_id
               inner join edoc_cfg on edc_id = edl_from_edc_id
               where eds_id = :eds_id and
                     edc_code = 'loan.cancel' and
                     exists(select freeze_time_h
                            from hp_get_freeze_data(edl_object_id)
                            where current_timestamp > freeze_datetime_end))) then
    begin
      res_code = 1;
      res_text = 'После окончания периода охлаждения отказаться нельзя.';
    end
  end
  if (res_code = 0) then
  begin
    -- Если ошибок нет, подписываем собственные бланки клиента
    -- Бланки уровня документа будут подписаны в edoc_list_au0
    for select res_str
    from hp_split_str(:blank_list,  ','), edoc_blank, edoc_sign
    where edb_id = res_str
      and edb_from_edl_id = eds_from_edl_id
      and edb_from_c_id = eds_from_c_id
    into edb_id
    do
      update edoc_blank
      set edb_sign_status = 1
      where edb_id = :edb_id;
  end
  suspend;
end
