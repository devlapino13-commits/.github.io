CREATE PROCEDURE HP_CHECK_CLIENT_INFO_STATE(
    C_ID D_ID,
    CHECK_KIND D_STR_CODE)
RETURNS (
    RES_IS_OK D_SMALLINT)
AS
declare variable last_disb_datetime d_datetime;
declare variable db_id d_integer;
declare variable tmp_id d_integer;
declare variable k_id d_id;
declare variable rd_id d_integer;
begin
  -- Процедура проверяет, после последней выдачи займа были ли изменения данных
  -- Например, менялись документы, или менялись персональные данные и т.д.
  -- Возвращает RES_IS_OK:
  -- 0 - Да, менялись
  -- 1 - После последней выдачи данные не менялись, все ОК

  -- Данная процедура также вызывается для освежения статуса из ClientState
  -- Для обновления статуса устанавливается значение ClientState.cs_info_state = -1 и далее триггер перевычислет

  res_is_ok = 1;

  if (coalesce(check_kind, '') = '') then
    check_kind = 'full';

  db_id = f_get_db_id();

  /*select rd_id
  from refdata
  where rd_code = 'ActionGroup.Check_Loan_Data'
  into tmp_id;*/

  -- Берем время выдачи последнего займа
  for select ks_from_k_id, ol_datetime
      from kontstate, knclient, operlist
      where kc_from_c_id = :c_id and
            kc_from_k_id = ks_from_k_id and
            kc_amount_approved > 0 and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_from_ol_id = ol_id
      order by kc_from_c_id desc, kc_date_disb desc, kc_from_k_id desc
      into k_id, last_disb_datetime
  do
  begin
    if (db_id <> 81001) then
      break;

    select rd_id
    from hp_get_check_loan_data(:k_id)
    into :rd_id;

    -- Если займ утвержден роботом, не берем эту дату
    if (exists(select *
               from actionlist
               where acl_object_kind = 9 and
                     acl_object_id = :k_id and
                     acl_from_rd_id = :rd_id and
                     acl_status = 2 and
                     acl_from_un_id in (1000000000001, 1000000000000))) then
      last_disb_datetime = null;

    if (last_disb_datetime is not null) then
      break;
  end

  -- Если выдач не было вообще, завершаем работу
  if (last_disb_datetime is null) then
    res_is_ok = 0;

  -- Проверка состояние верификации паспортных данных
  if (res_is_ok = 1) then
  begin
    if (exists(select *
               from clients
               where c_id = :c_id and
                     c_verify_status <> 2)) then
      res_is_ok = 0;
  end

  -- Проверка изменения документов
  if (res_is_ok = 1) then
  begin
    if (exists(select *
               from clientdocs, passkind
               where cds_from_c_id = :c_id and
                     cds_from_pk_id = pk_id and
                     pk_is_other_category = 0 and
                     (cds_create_datetime > :last_disb_datetime or
                     cds_delete_datetime > :last_disb_datetime or
                     cds_datetime > :last_disb_datetime))) then
      res_is_ok = 0;
  end

  -- Проверка действующего основного документа
  if (res_is_ok = 1) then
  begin
    if (not exists(select *
                   from clientdocs
                   where cds_from_c_id = :c_id and
                         cds_is_main_doc = 1 and
                         cds_delete_datetime is null and
                         cds_doc_date_begin <= current_date and
                         (cds_doc_date_end >= current_date or
                         cds_doc_date_end is null))) then
      res_is_ok = 0;
  end

  if (res_is_ok = 1) then
  begin
    -- Проверка объединения клиентов
    if (exists(select *
               from joinedclients
               where jc_stay_c_id = :c_id and
                     jc_datetime >= :last_disb_datetime)) then
      res_is_ok = 0;
  end

  if (res_is_ok = 1) then
  begin
    -- Проверка изменения ФИО, ДР и т.д.

    if (exists(select *
               from clientlog
               where cll_from_c_id = :c_id and
                     cll_curr_datetime >= :last_disb_datetime)) then
      res_is_ok = 0;
  end

  if (res_is_ok = 1) then
  begin
    -- Если договор АСП подписан после выдачи последнего займа
    tmp_id = null;
    for select edc_id
        from edoc_cfg
        where edc_code in ('asp.agreement', 'cell.application') and
              edc_state = 3
        into tmp_id
    do
    begin
      if (exists(select *
                 from edoc_list
                 where edl_from_edc_id = :tmp_id and
                       edl_object_kind = 0 and
                       edl_object_id = :c_id and
                       edl_create_datetime >= :last_disb_datetime)) then
        res_is_ok = 0;
    end
  end

  if (res_is_ok = 1) then
  begin
    -- Если менялся адрес клиента
    if (exists(select *
               from addrclient
               where adc_from_c_id = :c_id and
                     adc_address_type in (1, 2) and
                     adc_datetime >= :last_disb_datetime)) then
      res_is_ok = 0;
  end

  suspend;
end
