CREATE PROCEDURE HP_PERSTEMP_VERIFY_SET(
    ACL_ID DECIMAL(18,0),
    ACL_STATUS_STR VARCHAR(255),
    ACL_MEMO VARCHAR(255),
    UN_ID DECIMAL(18,0))
AS
declare variable cv_status integer;
declare variable c_id decimal(18,0);
declare variable acm_memo varchar(4000);
declare variable check_kind varchar(20);
declare variable is_confirm d_bool;
declare variable err_text d_str_255;
begin
-- данные подтверждаются
  acm_memo = '';
  cv_status = 2;
  if (acl_status_str <> '') then
    cv_status = -1;

  if (un_id is null) then
  begin
    un_id = uxf$user();
    when any do
      un_id = 1000000000000;
  end

  select acl_object_id
  from actionlist
  where acl_id = :acl_id and
        acl_object_kind = 0
  into :c_id;

  update actionlist
  set acl_status = :cv_status,
      acl_from_un_id = :un_id
  where acl_id = :acl_id and
        coalesce(acl_status, -777) <> :cv_status;

  update clients
  set c_verify_status = iif(:cv_status = -1, -2, :cv_status)
  where c_id = :c_id and
        coalesce(c_verify_status, -777) <> :cv_status;
  is_confirm = iif(row_count = 1 and cv_status = 2, 1, 0);

  if (coalesce(acl_memo, '') <> '') then
  begin
    acl_memo = replace(acl_memo, '"', '\"');
    acm_memo = acm_memo || iif(acm_memo <> '', ',', '') || '"comment":"' || acl_memo || '"';
  end

  if (cv_status = -1) then
  begin
    for select res_str
    from hp_split_str(:acl_status_str, ';')
    into :check_kind
    do
      acm_memo = acm_memo || iif(acm_memo <> '', ',', '') || '"' || check_kind || '":"1"';
  end

  if (acm_memo <> '') then
  begin
    acm_memo = '{"verify":{' || trim(acm_memo) || '}}';

    update or insert into actionmemo (acm_from_acl_id, acm_memo)
    values (:acl_id, :acm_memo)
    matching (acm_from_acl_id);
  end

  if (f_get_db_id() in (39801, 41701) and is_confirm = 1) then
  begin
    if (exists(
      select * from cardclientid where cci_from_c_id = :c_id)) then
    err_text = (select err_text from hp_cardclient_update_customer(:c_id, null, 0, :un_id));
  end

end
