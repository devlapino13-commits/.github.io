CREATE PROCEDURE HR_PSTATUS_LAST_ROW_DATA(
    PER_ID ID_TYPE,
    DD DATE,
    ORDER_TYPE D_INTEGER = null)
RETURNS (
    REG_NAME STR_100,
    JL_NAME STR_100,
    C_NAME_FIO STR_100,
    DEP_NAME STR_100,
    PS_FROM_JL_ID INTEGER,
    PS_FROM_REG_ID INTEGER,
    PS_FROM_DEP_ID INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PSA_AGREEMENT_NO VARCHAR(10),
    C_AGE INTEGER)
AS
declare variable ps_id integer;
declare variable ps_from_ho_id id_type;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  for select reg_name, jl_name, c_name_fio, dep.dep_name, ps.ps_from_jl_id, ps.ps_from_reg_id, ps.ps_from_dep_id,
             ps.ps_date_end_in_order, ps.ps_id, ps.ps_from_ho_id, datediff(day,c_birth_date,coalesce(:dd,current_date))/365
      from hr_pstatus ps, clients, region, joblist, depart dep, sa_worktype, hr_orders
      where ps_from_c_id = :per_id and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id and
            ps_from_dep_id = dep_id and
            c_id = :per_id and
            ps_date_begin <= :dd and
            ps_date_end >= :dd and
            ps_from_ho_id = ho_id and
            sawt_id = ho_order_type and
            sawt_base_type <> 3
      order by ps_date_begin desc, ps_date_end desc
      rows 1
      into :reg_name, :jl_name, :c_name_fio, :dep_name, :ps_from_jl_id, :ps_from_reg_id, :ps_from_dep_id,
           :ps_date_end_in_order, :ps_id, :ps_from_ho_id, :c_age
  do
  begin
    if (db_id in (39801, 41701) and order_type in (3, 4)
      and exists(
        select *
        from cardplace
          inner join cardlist
            on cal_last_cpl_id = cpl_id
              and cal_status = 1
        where cpl_from_c_id = :per_id)) then
    exception e_message f_format(
      'Сотруднику %1 необходимо передать, ' || ascii_char(10) ||
      'имеющиеся в мат.ответственности, карты MBulak', c_name_fio);

    psa_agreement_no = null;
    select psa.psa_agreement_no
    from hr_pstatus_agreement psa
    where psa.psa_from_ho_id = :ps_from_ho_id and
          psa.psa_from_ps_id = :ps_id
    into :psa_agreement_no;

    suspend;
  end
end
