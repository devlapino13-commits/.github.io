CREATE PROCEDURE HP_PERMISSIONS_CHECKED_MENU(
    USER_ROLES VARCHAR(255))
RETURNS (
    ML_ID INTEGER,
    PM_ID INTEGER)
AS
declare variable sql_text varchar(2048);
declare variable is_one_role integer;
declare variable rn_id decimal(18,0);
declare variable uniqtab_rolename integer;
declare variable uniqtab_menulist integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;

  if (position(',', user_roles) <= 0) then
  begin
    is_one_role = 1;
    rn_id = cast(user_roles as id_type);
  end
  else
    is_one_role = 0;

  sql_text = 'select distinct pm_subject_id from permissions
  where pm_object_kind=' || :uniqtab_rolename || ' and pm_object_id in (' || :user_roles || ') and pm_subject_kind=' || :uniqtab_menulist || ' and pm_is_delete=0
  and (coalesce(pm_date_begin, ''01.01.2000'')<=current_date)
  and (coalesce(pm_date_end, ''31.12.2999'')>=current_date)';
  for execute statement(:sql_text)
          into :ml_id
  do
  begin
    pm_id = -1;
    if (is_one_role = 1) then
    begin
      select pm_id
      from permissions
      where pm_object_kind = :uniqtab_rolename and
            pm_object_id = :rn_id and
            pm_subject_kind = :uniqtab_menulist and
            pm_subject_id = :ml_id and
            pm_is_delete = 0 and
            (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
            (coalesce(pm_date_end, '31.12.2999') >= current_date) rows 1
      into :pm_id;
    end
    suspend;
  end
end
