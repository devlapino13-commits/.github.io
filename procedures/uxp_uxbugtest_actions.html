CREATE PROCEDURE UXP$UXBUGTEST_ACTIONS(
    UA_UUID D_UUID,
    VER_KIND D_STR_CODE)
RETURNS (
    RES_UA_UUID D_UUID,
    RES_UX_CAPTION D_STR_255,
    RES_UX_KIND D_STR_CODE,
    RES_UX_ITEM D_STR_CODE,
    RES_PLACE D_STR_100)
AS
declare variable uxa_uuid d_uuid;
declare variable ua_usr_id_field d_str_code;
declare variable ufm_uuid d_uuid;
declare variable is_editor d_bool;
begin
  for select ua_uxa_uuid, uxf$ver(uxa_owner, :ver_kind), uxa_name, uxf$kind(uu_kind, 'code')
  from ux$action, ux$alias, ux$uniq
  where ua_uuid = :ua_uuid and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = uu_id
  into uxa_uuid, ufm_uuid, res_ux_caption, res_ux_kind
  do
  begin
    res_place = '';
    res_ux_item = 'Текущая форма';
    res_ua_uuid = ua_uuid;
    suspend;

    res_place = 'Форма';
    for select ua_uuid, ua_name || '    / ' || uxf$kind(uu_kind, 'caption'),
               uxf$kind(uu_kind, 'code'), ua_usr_id_field
    from ux$action, ux$alias, ux$uniq
    where ua_owner = :ufm_uuid and
          ua_uxa_uuid = uxa_uuid and
          uxa_owner = uu_id
    into res_ua_uuid, res_ux_caption, res_ux_kind, ua_usr_id_field
    do
    begin
      res_ux_item = 'Меню формы';

      is_editor = 0;
      if (exists(select * from ux$frbutton where ufb_ua_uuid = :res_ua_uuid)) then
      begin
        is_editor = 1;
        res_ux_item = 'Кнопка в форме';
      end

      if (coalesce(ua_usr_id_field, '') <> '') then
      begin
        is_editor = 1;
        res_ux_item = 'Редактор в форме';
      end

      if (is_editor = 0) then
      begin
        if (not exists(select * from ux$aliasaction
        where uaa_ua_uuid = :res_ua_uuid and
              uaa_uxa_uuid = :uxa_uuid)) then
          Continue;
      end

      suspend;
    end

    for select ua_uuid, ua_name || '    / ' || uxf$kind(uu_kind, 'caption'),
               uxf$kind(uu_kind, 'code'), ua_usr_id_field, ufr_name
    from ux$frame, ux$action, ux$alias, ux$uniq
    where ufr_owner = :ufm_uuid and
          ua_owner = ufr_uuid and
          ua_uxa_uuid = uxa_uuid and
          uxa_owner = uu_id
    order by ufr_is_mainframe desc
    into res_ua_uuid, res_ux_caption, res_ux_kind, ua_usr_id_field, res_place
    do
    begin
      res_ux_item = 'Меню';

      is_editor = 0;
      if (exists(select * from ux$frbutton where ufb_ua_uuid = :res_ua_uuid)) then
      begin
        is_editor = 1;
        res_ux_item = 'Кнопка';
      end

      if (coalesce(ua_usr_id_field, '') <> '') then
      begin
        is_editor = 1;
        res_ux_item = 'Редактор';
      end

      if (is_editor = 0) then
      begin
        if (not exists(select * from ux$aliasaction
        where uaa_ua_uuid = :res_ua_uuid and
              uaa_uxa_uuid = :uxa_uuid)) then
          Continue;
      end

      suspend;
    end
  end
end
