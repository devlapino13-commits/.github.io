CREATE PROCEDURE HP_EDOC_WEB_CHECK_CODE(
    UUID D_STR_CODE,
    CODE D_STR_CODE)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable obj_id d_id;
declare variable edm_status d_smallint;
declare variable edl_status d_integer;
declare variable hash_check d_smallint;
declare variable obj_code d_str_code;
declare variable cv_status d_integer;
declare variable edc_code d_str_code;
declare variable c_id d_id;
begin
  res_text = '';
  if (code is null) then code = '';

  obj_code = '';
  if (res_text = '' and char_length(:uuid) <> 36) then
    res_text = 'Неверная ссылка';

  if (res_text = '') then
  begin
    select obj_kind, obj_id
    from hp_uuid(:uuid)
    into obj_code, :obj_id;
    if (obj_id is null)
    then res_text = 'Неверная ссылка';
    else obj_code = f_objkind(obj_code, 'name');
  end

  if (obj_code = 'edoc_sign') then
  begin
    if (res_text = '') then
    begin
      select iif(coalesce(edl_date_end, current_date) < current_date, -1, edl_status),
        edc_code, eds_from_c_id
      from edoc_sign, edoc_list, edoc_cfg
      where eds_id = :obj_id
        and eds_from_edl_id = edl_id
        and edl_from_edc_id = edc_id
      into edl_status, edc_code, c_id;
      if (res_text = '' and edl_status = -1) then res_text = 'Данные устарели';
      if (res_text = '' and edc_code in ('bankcard.moneta', 'bankcard.elcart') and edl_status = 1) then
        res_text = 'Данные карты уже успешно привязаны. ' ||
          'Если хотите заменить карту, выберите "Ввести реквизиты банковской карты" на сайте Компании.';
    end
  
    if (res_text = '') then
    begin
      select edm_status, iif(:code = '', 0, iif(edm_hash_pin in (:code, f_md5(:code)), 1, 0))
      from edoc_msg
      where edm_from_eds_id = :obj_id
      order by iif(:code = '', 0, iif(edm_hash_pin in (:code, f_md5(:code)), 1, 0)) desc
      rows 1 -- берем наиболее успешный статус
      into edm_status, hash_check;
  
      if (res_text = '' and edm_status is null) then res_text = 'Неверная ссылка';
      if (res_text = '' and hash_check = 0) then
        res_text = 'Неверный код';

      if (res_text = '' and edc_code = 'bankcard.moneta') then
        insert into extsys_token (et_from_esm_id, et_from_c_id, et_code)
        values (null, :c_id, :code);
    end
  end
  else if (obj_code = 'contveri') then
  begin
    -- Проверка ContVeri
    select cvr_status, iif(cvr_hash_pin in (f_coder_pass(:code), f_md5(:code)), 1, 0)
    from contveri
    where cvr_id = :obj_id
    into cv_status, hash_check;

    if (res_text = '' and cv_status = 1) then
      res_text = 'Данные уже подтверждены';
    if (res_text = '' and hash_check = 0) then
      res_text = 'Неверный код';
  end
  else if (obj_code = 'contverify') then
  begin
    -- Проверка ContVerify
    select cv_status, iif(cv_password in (f_coder_pass(:code), f_md5(:code)), 1, 0)
    from contverify
    where cv_id = :obj_id
    into cv_status, hash_check;

    if (res_text = '' and cv_status = 1) then
      res_text = 'Данные уже подтверждены';
    if (res_text = '' and hash_check = 0) then
      res_text = 'Неверный код';
  end
  else
  begin
    res_text = 'Неверная ссылка';
  end

  Suspend;
end
