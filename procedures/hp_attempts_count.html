CREATE PROCEDURE HP_ATTEMPTS_COUNT(
    PHONE_NO D_STR_100,
    C_ID D_ID,
    C_INN D_STR_20)
RETURNS (
    MQL_DATETIME D_DATETIME,
    MQL_CONT_CODE D_STR_CODE)
AS
declare variable CELL_NO D_STR_100;
declare variable CL_ID D_ID;
declare variable CNT integer = 0;
declare variable SEARCH_NUM D_STR_100;
begin
    if (coalesce(phone_no, '') <> '') then cnt = cnt + 1;
    if (coalesce(c_id, 0) <> 0)       then cnt = cnt + 1;
    if (coalesce(c_inn, '') <> '')    then cnt = cnt + 1;

    if (cnt <> 1) then
        exception e_message('Укажите один параметр для поиска');

    if (coalesce(phone_no, '') <> '') then
        search_num = phone_no;

    else if (coalesce(c_id, 0) <> 0) then
    begin
        cell_no = f_cont_info(f_objkind('clients', 'id'), c_id, 'cell', 'ok');

        if (coalesce(cell_no, '') = '') then
            exception e_message('У клиента нет верифицированного мобильного номера');

        search_num = cell_no;
    end

    else if (coalesce(c_inn, '') <> '') then
    begin
        select c.c_id
        from clients c
        where c.c_inn = :c_inn
        into cl_id;

        if (coalesce(cl_id, 0) = 0) then
            exception e_message('Клиент с указанным ИНН не найден');

        cell_no = f_cont_info(f_objkind('clients', 'id'), cl_id, 'cell', 'ok');

        if (coalesce(cell_no, '') = '') then
            exception e_message('У клиента нет верифицированного мобильного номера');

        search_num = cell_no;
    end

    for select mql.mql_datetime,
               mql.mql_cont_code
    from msgquotalist mql
    where mql.mql_cont_code = :search_num
    order by mql.mql_datetime desc
    into mql_datetime,
         mql_cont_code
    do
    suspend;
end
