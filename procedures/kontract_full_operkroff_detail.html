CREATE PROCEDURE KONTRACT_FULL_OPERKROFF_DETAIL(
    KONTRACT_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    OK_TYPE_TEXT VARCHAR(100),
    OL_DATE DATE,
    OL_DOC_NO VARCHAR(10),
    OK_SH_NUMBER INTEGER,
    SUM_OF_ALL DECIMAL(18,4),
    SUM_OF_OK_OS DECIMAL(18,4),
    SUM_OF_OK_PERC DECIMAL(18,4),
    SUM_OF_OK_FINE DECIMAL(18,4),
    OL_ID VARCHAR(20),
    SUM_OF_OK_KOMMIS DECIMAL(18,4),
    KOL_DELIQ INTEGER,
    OL_CLIENT_NAME VARCHAR(255),
    OL_DATETIME TIMESTAMP)
AS
declare variable fact_fio varchar(255);
begin
  ok_sh_number = 0;
  sum_of_ok_kommis = 0;
  kol_deliq = 0;

  ok_type_text = 'Спец.учет';
  for select ol.ol_id, ol.ol_date, ol.ol_datetime, ol_doc_no,
             sum(oko.oko_os + oko.oko_perc + oko.oko_other + oko.oko_sh_os + oko.oko_sh_perc + oko.oko_sh_other),
             sum(oko.oko_os + oko.oko_sh_os), sum(oko.oko_perc + oko.oko_sh_perc),
             sum(oko.oko_other + oko.oko_sh_other)
      from operkroff oko
      inner join operlist ol on ol.ol_id = oko.oko_from_ol_id
      where oko.oko_from_k_id = :kontract_id and
            oko.oko_ol_is_active = 1 and
            ol.ol_date <= :rep_date_end
      group by ol.ol_id, ol.ol_date, ol.ol_datetime, ol_doc_no
      order by ol.ol_date
      into ol_id, ol_date, ol_datetime, ol_doc_no, sum_of_all, sum_of_ok_os, sum_of_ok_perc, sum_of_ok_fine
  do
  begin

    ol_client_name = '';
    select c_name_fio, left(ol_doc_client_name, 50)
    from operlist, clients
    where c_id = ol_document_client and
          ol_id = :ol_id
    into :ol_client_name, :fact_fio;
    if (fact_fio is null) then
      fact_fio = '';
    if (ol_client_name is null) then
      ol_client_name = '';
    if ((fact_fio <> '') and
        (ol_client_name <> '')) then
      ol_client_name = ol_client_name || ' через ' || fact_fio;

    if (ol_client_name = '') then
    begin
      select ol_doc_client_name
      from operlist
      where ol_id = :ol_id
      into :ol_client_name;
    end

    suspend;
  end
end
