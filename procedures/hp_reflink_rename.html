CREATE PROCEDURE HP_REFLINK_RENAME(
    REF_CATEGORY VARCHAR(255),
    REF_OLD_VALUE VARCHAR(255),
    REF_NEW_VALUE VARCHAR(255))
AS
declare variable ref_table_name varchar(255);
declare variable ref_label_field varchar(255);
declare variable sql_text varchar(2000);
begin
  if (ref_category is null) then
    exit;
  if (coalesce(ref_old_value, '') = '') then
    exit;
  if (coalesce(ref_new_value, '') = '') then
    exit;

  for select ref_table_name, ref_label_field
      from hp_reflink_category(0)
      where ref_name = :ref_category and
            ref_label_field <> ''
      into :ref_table_name, ref_label_field
  do
  begin
    if (ref_table_name is null) then
      exit;
    if (ref_label_field is null) then
      exit;

    sql_text = 'update ' || ref_table_name || ' set ' || ref_label_field || '=replace(' || ref_label_field || ', ''.' || ref_old_value || '.'', ''.' || ref_new_value || '.'')';
    execute statement sql_text;
  end
end
