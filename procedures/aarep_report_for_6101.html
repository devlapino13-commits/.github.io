CREATE PROCEDURE AAREP_REPORT_FOR_6101(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    C_NAME_FIO STR_100,
    C_BIRTH_DATE D_DATE,
    CITIZEN STR_100,
    AAREP_K_DELIQ_DAYS D_INTEGER,
    KF_REGION_NAME STR_100,
    KF_OFFICER_NAME STR_100,
    COUNTER D_INTEGER,
    KF_FOLDER_NAME D_STR_100,
    C_POINT D_INTEGER,
    LAST_DATE_CLOSE D_DATE,
    IS_STOPLIST D_STR_10,
    IS_CESSION D_STR_10,
    DEPORT_DATE_BEGIN D_DATE,
    DEPORT_DATE_END D_DATE,
    DEPORT_STATUS D_STR_255)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable kf_id d_id;
declare variable doc_no str_100;
declare variable doc_date d_date;
begin
  counter = 0;
  for select aarep_k_id --, aarep_k_deliq_days -- aarep_k_balance
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 4)
      into k_id --, aarep_k_deliq_days -- rep_active_balance
  do
  begin
    select k.k_from_kf_id
    from kontract k
    where k.k_id = :k_id
    into kf_id;

    is_cession = iif(exists (select scl_id
                          from sud_cession_packet
                          join sud_cession_loans on scl_from_scp_id = scp_id
                          where scl_from_k_id = :k_id and
                                scp_status > 0 and
                                scp_date <= :report_date_end
                          ), 'да', 'нет');

    select reg_name, c_name_fio, kal.kal_new_kf_name
    from knaktlist kal
    join region on reg_unicode = kal_new_reg_id
    join clients on c_id = kal_new_kf_officer
    where kal_from_kf_id = :kf_id and
          kal_date_end = '31.12.9999' and
          kal.is_delete = 0
    into kf_region_name, kf_officer_name, kf_folder_name;

   -- doc_no = null; doc_date = null;

    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and kc_amount_approved > 0
        into c_id
    do
    begin
      select c.c_name_fio, c.c_birth_date, rd.rd_name, f_point(c_id, :report_date_begin)
      from clients c
      join refdata rd on rd.rd_id = c.c_citizen
      where c.c_id = :c_id
      into c_name_fio, c_birth_date, citizen, c_point;

      -- дата закрытия последнего займа клиента
      select ks.ks_date_begin
      from kontstate ks
      join knclient kn on kn.kc_from_k_id = ks.ks_from_k_id
      where ks.ks_state = 2
            and ks.ks_ol_is_delete = 0
            and :report_date_end >= ks.ks_date_begin
            and kn.kc_from_c_id = :c_id
            and kn.kc_from_k_id = :k_id
      order by ks.ks_date_begin desc
      rows 1
      into last_date_close;

      is_stoplist = iif(exists (
          select m.m_is_active
          from moratory m
          where m.m_from_c_id = :c_id and m.m_is_active = 1
                and :report_date_end between m.m_date_begin and m.m_date_end
                          ), 'да', 'нет');

      deport_date_begin = null;
      deport_date_end = null;

      select cd.cds_doc_date_begin, cd.cds_doc_date_end
      from clientdocs cd
      join passkind pk on pk.pk_id = cd.cds_from_pk_id
      join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cd.cds_from_c_id = :c_id and
            pdt.pdt_code = 'deport_document' and
            cd.cds_is_delete = 0 and
            :report_date_end between cd.cds_doc_date_begin and cd.cds_doc_date_end
      order by cd.cds_doc_date_begin desc
      rows 1
      into :deport_date_begin, :deport_date_end;

      select iif(status_full = '', 'Нет', coalesce(status_full, 'Нет'))
      from hp_deprot_status(:c_id)
      into deport_status;

      for select cds_doc_no, cds_doc_date_begin
          from clientdocs
          where cds_from_c_id = :c_id and
                cds_is_main_doc = 1 and
                cds_is_delete = 0
          into doc_no, doc_date
      do
      begin
--        if (aarep_k_deliq_days <= 250) then
--        begin
          counter = counter + 1;
          suspend;
--        end
      end

    end
  end
end
