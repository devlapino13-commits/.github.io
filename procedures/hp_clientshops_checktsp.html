CREATE PROCEDURE HP_CLIENTSHOPS_CHECKTSP(
    C_ID D_ID NOT NULL,
    AGREEMENT_KIND D_STR_255,
    DT_CHECK D_DATE = null)
RETURNS (
    ERR_TEXT D_STR_255,
    REG_ID D_INTEGER)
AS
declare variable ag_id d_integer;
declare variable c_name_fio d_str_100;
declare variable dogovor_tsp d_integer;
declare variable tsp_signed d_str_100;
declare variable edoc_kind d_str_code;
declare variable pdt_code d_str_100;
begin
  err_text = '';
  dt_check = cast(coalesce(dt_check, current_date) as date);
  if (:agreement_kind = 'AgreementKinds.tsp') then
  begin
    edoc_kind = 'tsp.agreement';
    pdt_code = 'tsp_agreement';
  end
  if (:agreement_kind = 'AgreementKinds.installment_goods') then
  begin
    edoc_kind = 'installment_goods.agreement';
    pdt_code = 'installment_goods_agreement';
  end

  select ag_id, ag_from_init_reg_id, c_name_fio
  from agreements
  inner join clients on c_id = ag_object_id
  where ag_object_kind = 0 and
        ag_object_id = :c_id and
        ag_from_rd_id = f_get_rd_id(:agreement_kind) and
        ag_status = 1 and
        :dt_check between coalesce(ag_date_begin, cast('01.01.1900' as date)) and coalesce(ag_date_end, cast('31.12.9999' as date))
  order by ag_id desc
  rows 1
  into :ag_id, :reg_id, :c_name_fio;
  tsp_signed = f_edoc_status(:edoc_kind, ag_id, :dt_check);
  if (tsp_signed <> 'signed') then
  begin
    dogovor_tsp = 0;
    select cdc.cds_id
    from clientdocs cdc
    inner join passkind pask on pask.pk_id = cdc.cds_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pask.pk_from_pdt_id
    where cdc.cds_from_c_id = :c_id and
          cdc.cds_is_delete = 0 and
          pdt.pdt_code = :pdt_code and
          :dt_check between coalesce(cdc.cds_doc_date_begin, cast('01.01.1900' as date)) and coalesce(cdc.cds_doc_date_end, cast('31.12.9999' as date))
    order by cdc.cds_datetime desc
    rows 1
    into dogovor_tsp;
    if (coalesce(dogovor_tsp, 0) = 0) then
    begin
      err_text = f_format('Выбранный контрагент %1 не имеет активного %2', c_name_fio, iif(agreement_kind = 'AgreementKinds.tsp', 'договора ТСП', 'соглашения'));
    end
  end
  else
  begin
    if (agreement_kind = 'AgreementKinds.tsp' and
        not exists(select *
                   from agreement_feac
                   inner join refdata on rd_id = afe_from_rd_id and
                         trim(rd_str_id) <> ''
                   where afe_from_ag_id = :ag_id)) then
      err_text = f_format('Для контрагента %1 не определен МСС-код по виду деятельности', c_name_fio);
  end
  suspend;
end
