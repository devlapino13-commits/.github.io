CREATE PROCEDURE FM_USER_LOGIN(
    UN_LOGIN D_STR_255,
    UN_PASS_HASH D_STR_255,
    UN_IS_LOGIN D_SMALLINT,
    UN_SYSTEM_NAME D_STR_100,
    UN_APP_VER D_INTEGER,
    UN_ANDROID_ID D_STR_CODE)
RETURNS (
    UN_ID D_ID,
    UN_MESSAGE D_STR_255,
    UN_CAN_WORK D_INTEGER,
    UN_LANG_CODE D_STR_LANG,
    UN_SETTINGS D_STR_8100)
AS
declare variable db_android_id d_str_code;
declare variable db_app_ver d_integer;
declare variable db_id d_integer;
declare variable dont_check_device d_smallint;
declare variable db_is_actual d_bool;
declare variable access_test_only d_bool;
declare variable access_test d_bool;
declare variable quota_exceeded d_bool;
begin
  -- Надстройка над hp_check_user_password (проверка логина)

  -- Результаты hp_check_user_password:
  -- UN_CAN_WORK = -3 Требуется восстановление пароля, без этого не может работать
  -- UN_CAN_WORK = -2 Требуется смена пароля, без этого не может работать
  -- UN_CAN_WORK = -1 Не может работать дальше
  -- UN_CAN_WORK = 0  Пароль неверный, повторная попытка
  -- UN_CAN_WORK = 1  Все ок, пускаем в систему
  -- Дополнительно:
  -- UN_CAN_WORK = -4 Необходимо обновить приложение

  --  FsMobile.Login - Android
  --  FsMobileIOS.Login - IOS

  db_id = f_get_db_id();
  un_settings = '{}';

  if (un_system_name = 'FsMobileIOS.Login') then
    db_app_ver = f_get_const('fsm_stuff_ios_ver');
  else
    db_app_ver = f_get_const('fsm_stuff_android_ver');

  -- Подмена, т.к. FsMobileIOS.Login в PermActions не существует,
  -- но используется для проверки версии выше
  if (un_system_name = 'FsMobileIOS.Login') then
    un_system_name = 'FsMobile.Login';

  -- Пока не выяснили пользователя используем язык по-умолчанию
  un_lang_code = f_get_lang_code(null, null);

  -- Проверка версии приложения
  -- Версия в БД должна быть меньше или равна версии приложения
  -- Но разница не должна быть 5 и более версий
  if (un_app_ver < db_app_ver or un_app_ver - db_app_ver >= 5) then
  begin
    un_can_work = -4; -- Необходимо обновить приложение
    suspend;
    exit;
  end

  -- Проверка Android ID
  un_android_id = coalesce(un_android_id, '');
  if (un_android_id = '') then
  begin
    un_can_work = -1; -- закрытие программы
    un_message = iif(db_id = 94401, 'Namelum qurgu. Dostup qadagandir.', 'Неизвестное устройство, доступ запрещен.');
    suspend;
    exit;
  end

  select quota_exceeded
  from hp_msgquota_check('mobile.staff.user_login', :un_login)
  into quota_exceeded;

  if (quota_exceeded = 1) then
  begin
    un_can_work = -1; -- закрытие программы
    un_message = iif(db_id = 94401, 'Giriş cəhdlərinin sayı aşıldı. Dostup qadagandir.', 'Превышено количество попыток входа, доступ запрещен.');
    suspend;
    exit;
  end

  if (exists(select * from hp_db_is_actual)) then
    db_is_actual = 1;

  db_android_id = '';
  select coalesce(un_android_id, ''), un_lang_code
  from username
  where un_name = :un_login
  into db_android_id, un_lang_code;

  if (un_can_work = 0) then
  begin
    -- Тут просто увеличиваем счетчик, проверка происходит выше
    select quota_exceeded
    from hp_msgquota_process('mobile.staff.user_login', :un_login)
    into quota_exceeded;
  end

  select un_id, un_message, un_can_work
  from hp_check_user_password(:un_login, :un_pass_hash, :un_is_login, :un_system_name)
  into :un_id, :un_message, :un_can_work;

  access_test_only = 0;
  if (exists(
      select has_access
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.TestOnlyAccess')
      where has_access = 1)) then
    access_test_only = 1;

  access_test = 0;
  if (exists(
      select has_access
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.TestAccess')
      where has_access = 1)) then
    access_test = 1;

  if (un_can_work in (-2, 1)) then
  begin
    -- Если пользователь пытался восстановить пароль, но потом вспомнил текущий,
    -- то сносим un_password_reset
    update username
    set un_password_reset = ''
    where un_id = :un_id and
          un_password_reset <> '';

    -- Если логин успешный,  дополнительно проверяем android_id
    -- Обновляем, если это первый вход
    if (db_android_id = '') then
    begin
      update username
      set un_android_id = :un_android_id
      where un_name = :un_login;
      db_android_id = un_android_id;
    end

    -- Проверяем, совпадают ли android_id на устройстве и в БД
    dont_check_device = (select has_access
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.MultiDeviceAccess'));

    if (dont_check_device = 1) then
    begin
      -- Без проверки устройства только обновляем android_id
      update username
      set un_android_id = :un_android_id
      where un_name = :un_login;
      db_android_id = un_android_id;
    end
    else
    begin
      -- Для остальных пользователей проверяем совпадение идентификатора устройства
      if (un_android_id <> db_android_id) then
      begin
        un_can_work = -1; -- закрытие программы
        un_message = iif(db_id = 94401, 'Siz qurgunu deyishdiniz, dostupun berpasi ucun "Parolun atilmasi/Telefonun deyishirilmesi" duymesini sixin.', 'Вы сменили устройство, для получения доступа нажмите "Сбросить пароль/Сменить телефон".');
        suspend;
        exit;
      end
    end

    if (db_is_actual = 0 and access_test = 0 and access_test_only = 0) then
    begin
      un_can_work = 0; -- закрытие программы
      un_message = iif(db_id = 94401, 'Sizin test bazaya dostupunuz mövcüd deyil.', 'У вас нет доступа к тестовой базе');
      suspend;
      exit;
    end

    if (db_is_actual = 1 and access_test_only = 1) then
    begin
      un_can_work = 0; -- закрытие программы
      un_message = iif(db_id = 94401, 'Sizin əsas bazaya dostupunuz mövcüd deyil.', 'У вас нет доступа к боевой базе');
      suspend;
      exit;
    end
  end
  else
  if (un_can_work = 0) then
  begin

    if (not(exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.Login')
                   where has_access = 1)) and :un_login <> 'fmtest') then
    begin
      un_message = iif(db_id = 94401, 'Sizin carı sistemə dostupunuz mövcüd deyil.', 'У вас нет доступа к данной системе');
      suspend;
      exit;
    end

    if (db_is_actual = 0 and access_test = 0 and access_test_only = 0) then
    begin
      un_can_work = 0; -- закрытие программы
      un_message = iif(db_id = 94401, 'Sizin test bazaya dostupunuz mövcüd deyil.', 'У вас нет доступа к тестовой базе');
      suspend;
      exit;
    end

    if (db_is_actual = 1 and access_test_only = 1) then
    begin
      un_can_work = 0; -- закрытие программы
      un_message = iif(db_id = 94401, 'Sizin əsas bazaya dostupunuz mövcüd deyil.', 'У вас нет доступа к боевой базе');
      suspend;
      exit;
    end

    -- Если логин/пароль неверный, пробуем залониниться через un_password_reset
    select -2
    from username
    where un_name = :un_login and
          un_password_reset = :un_pass_hash and
          coalesce(un_password_reset, '') <> ''
    into un_can_work;
    if (un_can_work = -2) then
    begin
      -- Если un_pass_hash подойдет, пользователь сразу должен будет сменить пароль
      un_message = iif(db_id = 94401, 'Parol deyishdirilmelidir.', 'Вам необходимо сменить пароль.');

      update username
      set un_password_new = :un_pass_hash,
          un_password_reset = '',
          un_android_id = :un_android_id,
          un_google_id = ''
      where un_name = :un_login and
            un_password_reset = :un_pass_hash;

      -- Заставляем сменить пароль, если пользователь не сменит сразу
      -- Отдельный запрос, из-за того что в триггере при смене пароля
      -- un_password_change переписывается
      update username
      set un_password_change = dateadd(year, -1, current_date)
      where un_name = :un_login and
            un_password_new = :un_pass_hash;
    end
  end

  if (un_can_work in (-2, 1)) then
  begin
    select res_data
    from fm_user_settings(:un_id)
    into un_settings;
  end

  suspend;
end
