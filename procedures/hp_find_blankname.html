CREATE PROCEDURE HP_FIND_BLANKNAME(
    FIND_TEXT D_STR_100,
    BLN_CATEGORY D_INTEGER = -1)
RETURNS (
    BLN_ID D_INTEGER,
    BLN_NAME D_STR_255,
    BLN_LANG D_STR_20,
    RES_DISABLE_DATE D_DATE)
AS
begin
  if (bln_category is null) then
    bln_category = -1;

  if (find_text is null) then
    find_text = '';

  for select bln_id, replace(bln_name, '_', ' '), bln_lang
  from blankname
  where (:bln_category = -1 or bln_kind = :bln_category) and
        bln_name containing :find_text
  order by bln_id
  into :bln_id, :bln_name, bln_lang
  do
  begin
    select max(blb_date_end)
    from blankblob
    where blb_from_bln_id = :bln_id and
          (blb_blob is not null or blb_frx_blob is not null)
    into res_disable_date;

    if (res_disable_date >= current_date) then
      res_disable_date = null;

    suspend;
  end
end
