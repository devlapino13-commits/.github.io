CREATE PROCEDURE AAREP_LO_RETENTION_KOEF_DETAIL(
    LO_ID DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    TXT D_STR_100,
    K_KONTRACT_NO D_STR_100,
    RES_CLOSED INTEGER,
    RES_REPEAT INTEGER,
    RES_KOEF DECIMAL(18,7),
    RES_ACTIVE_BEGIN INTEGER)
AS
declare variable res_closed_tmp integer;
declare variable kf_id decimal(18,0);
declare variable dd_tmp date;
declare variable kc_when_prior_loan_closed date;
declare variable k_refund integer;
declare variable c_id id_type;
declare variable k_disb_date date;
declare variable kc_id id_type;
declare variable ol_id id_type;
declare variable min_sh_date date;
declare variable kf_id_prior id_type;
declare variable delinq_day_closed integer;
declare variable c_id_prior id_type;
begin
  res_closed = 0;
  res_repeat = 0;
  res_active_begin = 0;
  kf_id_prior = 0;
  delinq_day_closed = 0;
  c_id_prior = 0;
  for select k_from_kf_id, ks_date_begin, ks_from_ol_id, kc_from_c_id, k_kontract_no
      from kontract, kontstate, knclient
      where ks_state = 2 and
            ks_kc_id = :lo_id and
            ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = k_id and
            k_id = kc_from_k_id and
            kc_amount_approved > 0
      order by kc_from_c_id, kc_id
      into :kf_id, :dd_tmp, :ol_id, :c_id, k_kontract_no
  do
  begin

    kf_id_prior = kf_id;
    res_closed_tmp = 1;
    -- если по одному клиенту было 2 выдачи то считать только одну, это касается если выдача в один и тот же день
    if (c_id_prior = c_id) then
      res_closed_tmp = 0;
    c_id_prior = c_id;

    if (res_closed_tmp = 1) then
    begin
      select min(ok_sh_date)
      from operkred
      where ok_from_ol_id = :ol_id and
            ok_os + ok_perc < 0
      into min_sh_date;

      delinq_day_closed = dd_tmp - min_sh_date; -- просрочки считать на момент закрытия кредита
      -- не включать закрытые с просроченными более 360 дней
      if (delinq_day_closed > 45) then
        res_closed_tmp = 0;
    end

    -- если в предыдущий день закрытия папка находилась в другом СВЗ то не будет считаться
    if (res_closed_tmp = 1) then
    begin
      if (not exists(select *
                     from knaktlist
                     where kal_from_kf_id = :kf_id and
                           knaktlist.is_delete = 0 and
                           kal_new_kf_officer = :lo_id and
                           kal_date <= (:dd_tmp - 1) and
                           kal_date_end >= (:dd_tmp - 1))) then
        res_closed_tmp = 0;
    end

    -- если на момент закрытия имеется активный кредит то не будет считаться что закрытым
    if (res_closed_tmp = 1) then
    begin
      if (exists(select *
                 from kontstate, knclient
                 where ks_ol_is_delete = 0 and
                       ks_state = 1 and
                       :dd_tmp between ks_date_begin and ks_date_end and
                       :dd_tmp > ks_date_begin -- если дата выдачи совпадает с датой закрытия предыдущего займа то будет считаться удержанием
                       and
                       ks_from_k_id = kc_from_k_id and
                       kc_amount_approved > 0 and
                       kc_from_c_id = :c_id)) then
        res_closed_tmp = 0;
    end

    res_closed = res_closed + res_closed_tmp;
    if (res_closed_tmp = 1) then
    begin
      txt = 'Закрыт за период';
      suspend;
    end
  end
    if (res_closed is null) then
  res_closed = 0;

  for select coalesce(kc_when_prior_loan_closed, '31.12.2999'), coalesce(k_refund, 0), k_disb_date, kc_from_c_id, kc_id,
             k_kontract_no
      from knclient, kontstate, kontract
      where ks_state = 1 and
            ks_kc_id = :lo_id and
            ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id
      into kc_when_prior_loan_closed, k_refund, k_disb_date, c_id, kc_id, k_kontract_no
  do
  begin
    --  Если контракт рефинансируется то сперва новый контракт выдается потом старый закрывается
    --  тем самым поле kc_when_prior_loan_closed не заполняется так как при выдаче старый заем активный из-за этого сделаны ниже следюущие дейстивя
    if (k_refund = 1 and
        kc_when_prior_loan_closed = '31.12.2999') then
    begin
      select ks_date_end
      from kontstate, knclient
      where ks_from_k_id = kc_from_k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_date_begin <= :k_disb_date and
            kc_id <> :kc_id
      order by ks_date_end desc
      rows 1
      into kc_when_prior_loan_closed;
      kc_when_prior_loan_closed = coalesce(kc_when_prior_loan_closed, '31.12.2999');
      if (kc_when_prior_loan_closed < '31.12.2999') then
        kc_when_prior_loan_closed = kc_when_prior_loan_closed + 1;
    end

    if (kc_when_prior_loan_closed between :rep_date_begin and :rep_date_end) then
    begin
      res_repeat = res_repeat + 1;
      txt = 'Повторно выданный';
      suspend;
    end

  end
  
  txt = '';
  res_koef = 100.0;
  if (res_closed <> 0) then
    res_koef = round(cast(res_repeat as double precision) / cast(res_closed as double precision) * 100.0, 1);

  select count(distinct kc_from_c_id)
  from oksummary, knclient
  where oks_date_begin <= (:rep_date_begin - 1) and
        oks_date_end >= (:rep_date_begin - 1) and
        oks_kc_id = :lo_id and
        oks_from_k_id = kc_from_k_id and
        kc_amount_approved > 0 and
        oks_balance > 0
        and oks_close_state = 0
  into :res_active_begin;

  if (res_active_begin is null) then
    res_active_begin = 0;

  txt = 'Итог';
  k_kontract_no = '';
  suspend;
end
