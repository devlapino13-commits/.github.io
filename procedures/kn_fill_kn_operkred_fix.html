CREATE PROCEDURE KN_FILL_KN_OPERKRED_FIX(
    K_ID D_ID,
    REP_DATE D_DATE,
    NEED_RECALC_OPERBOOK D_SMALLINT)
AS
declare variable need_close_loan d_smallint;
declare variable kc_id d_id;
declare variable ol_id d_id;
begin
  for select kc_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into kc_id
  do
  begin
    delete from kn_operkred
    where ok_from_kc_id = :kc_id and
          ok_ol_date >= :rep_date
    order by ok_from_kc_id desc, ok_ol_date desc;
  end

  for select distinct ok_from_ol_id
  from operkred
  where ok_from_k_id = :k_id and      
        ok_ol_date >= :rep_date and
        ok_ol_is_active = 1
  into ol_id
  do
  begin
    need_close_loan = 0;
    if (exists(select * from kontstate
    where ks_from_k_id = :k_id and
          ks_from_ol_id = :ol_id and
          ks_state = 2 and
          ks_ol_is_delete = 0)) then
      need_close_loan = 1;

    execute procedure hp_fill_kn_oksummary_one(:k_id, :ol_id, :need_close_loan);
  end

  if (need_recalc_operbook = 1) then
    execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :rep_date - 1);
end
