CREATE PROCEDURE AARE_AFFILIATED_EMPLOYEES(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    HIRED_DATE DATE,
    JL_NAME VARCHAR(255),
    LAST_CELL_MAIN VARCHAR(100),
    FIO_RODSTVENNIKA VARCHAR(100),
    REL_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    CLI_FIO VARCHAR(100),
    KC_AMOUNT_APPROVED CURR_TYPE,
    KC_BALANCE CURR_TYPE,
    CC_NAME VARCHAR(100),
    CC_STATUS VARCHAR(100),
    DELINQ_DAY INTEGER,
    ADC_ADDRESS_TYPE_STR VARCHAR(50),
    ADL_ADDRESS_STR VARCHAR(255),
    MEMO VARCHAR(255))
AS
declare variable C_ID ID_TYPE;
declare variable REG_ID integer;
declare variable REG_ID_PRIOR integer;
declare variable CREL_OTHER_C_ID ID_TYPE;
declare variable K_ID ID_TYPE;
declare variable OKS_BALANCE CURR_TYPE;
declare variable KC_FROM_C_ID ID_TYPE;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable DOLYA double precision;
declare variable ADC_ADDRESS_TYPE integer;
declare variable ADC_FROM_ADL_ID ID_TYPE;
begin
  memo = '';
  delete from temp_cli_contacts
  where 1 = 1;

  for select oks_from_k_id, oks_balance, :rep_date_end - oks_delinq_date
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            :rep_date_end - oks_delinq_date > 0 and
            oks_region_id = id_list and
            oks_close_state = 0 
      union
      select oko_from_k_id, sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc), 0
      from operkroff okf
      inner join kontract k on k_id = oko_from_k_id
      inner join kontstate on ks_from_k_id = k_id and
            ks_state = 2 and
            ks_ol_is_delete = 0 and
            :rep_date_end between ks_date_begin and ks_date_end
      inner join knaktlist on kal_from_kf_id = k_from_kf_id and
            :rep_date_end between kal_date and kal_date_end and
            is_delete = 0
      inner join region_list(:rep_region_id, 1, 3, 1) rl on rl.id_list = kal_new_reg_id
      where oko_ol_is_active = 1 and
            oko_ol_date <= :rep_date_end
      group by oko_from_k_id
      having sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc) > 0
      into :k_id, :oks_balance, :delinq_day
  do
  begin
    if (delinq_day < 0) then
      delinq_day = null;

    select k_kontract_no, k_disb_date, k_amount_approved
    from kontract
    where k_id = :k_id
    into k_kontract_no, k_disb_date, k_amount_approved;

    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_from_c_id, kc_amount_approved
    do
    begin

      dolya = cast(kc_amount_approved as double precision) / k_amount_approved;

      kc_balance = dolya * oks_balance;

      last_cell_main = f_cont_info('clients', kc_from_c_id, 'cell_main', 'last');
      if (last_cell_main = '') then
        last_cell_main = null;

      cc_name = null;
      cc_status = null;
      adc_address_type = null;
      adc_from_adl_id = null;

      insert into temp_cli_contacts (k_kontract_no, k_disb_date, kc_from_c_id, kc_amount, kc_balance, kc_cell_main,
                                     cc_name, cc_status, delinq_day)
      values (:k_kontract_no, :k_disb_date, :kc_from_c_id, :kc_amount_approved, :kc_balance, :last_cell_main, :cc_name,
              :cc_status, :delinq_day);

      last_cell_main = null;
      for select adc_address_type, adc_from_adl_id
          from addrclient
          where adc_from_c_id = :kc_from_c_id and
                adc_from_adl_id <> 81001000000000 -- не включать "Не указан"
                and
                adc_address_type <> 3 -- не включать адрес работы
          into adc_address_type, adc_from_adl_id
      do
      begin
        insert into temp_cli_contacts (k_kontract_no, k_disb_date, kc_from_c_id, kc_amount, kc_balance, kc_cell_main,
                                       cc_name, cc_status, delinq_day, adc_address_type, adc_from_adl_id)
        values (:k_kontract_no, :k_disb_date, :kc_from_c_id, :kc_amount_approved, :kc_balance, :last_cell_main,
                :cc_name, :cc_status, :delinq_day, :adc_address_type, :adc_from_adl_id);
      end

      for select cc_cell_no, cc_name, cc_status, cc_from_adl_id
          from contacts
          where cc_from_c_id = :kc_from_c_id and
                cc_total_res = 1
          into last_cell_main, cc_name, cc_status, adc_from_adl_id
      do
      begin
        insert into temp_cli_contacts (k_kontract_no, k_disb_date, kc_from_c_id, kc_amount, kc_balance, kc_cell_main,
                                       cc_name, cc_status, delinq_day, adc_from_adl_id)
        values (:k_kontract_no, :k_disb_date, :kc_from_c_id, :kc_amount_approved, :kc_balance, :last_cell_main,
                :cc_name, :cc_status, :delinq_day, :adc_from_adl_id);
      end
    end
  end

  reg_id_prior = 0;
  for select ps_from_c_id, jl_name, ps_from_reg_id
      from hr_pstatus, joblist
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id
      order by ps_from_reg_id
      into c_id, jl_name, reg_id
  do
  begin
    fio_rodstvennika = null;
    crel_other_c_id = null;
    rel_name = null;

    select c_name_fio, c_hired_date
    from clients
    where c_id = :c_id
    into officer_name, hired_date;

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;

      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end

    last_cell_main = f_cont_info('clients', c_id, 'cell_main', 'last');
    if (last_cell_main = '') then
      last_cell_main = null;

    cli_fio = null;
    for select distinct k_kontract_no, k_disb_date, kc_from_c_id, kc_amount, kc_balance, kc_cell_main, cc_name,
                        cc_status, delinq_day
        from temp_cli_contacts
        where kc_cell_main = :last_cell_main
        into :k_kontract_no, :k_disb_date, :kc_from_c_id, :kc_amount_approved, :kc_balance, :last_cell_main, :cc_name,
             :cc_status, :delinq_day
    do
    begin
      select c_name_fio
      from clients
      where c_id = :kc_from_c_id
      into cli_fio;

      if (cc_name is null) then
      begin
        memo = '№ тел. сотруд. совпадает с № клиента.';
      end
      else
      begin
        memo = '№ тел. сотруд. совпадает с № контакта клиента.';
      end

      suspend;
    end

    for select adc_address_type, adc_from_adl_id
        from addrclient
        where adc_from_c_id = :c_id and
              adc_from_adl_id <> 81001000000000 and
              adc_address_type <> 3 -- не включать адрес работы
        into adc_address_type, adc_from_adl_id
    do
    begin
      cli_fio = null;
      adc_address_type_str = null;
      adl_address_str = null;

      for select distinct k_kontract_no, k_disb_date, kc_from_c_id, kc_amount, kc_balance, kc_cell_main, cc_name,
                          cc_status, delinq_day
          from temp_cli_contacts
          where adc_from_adl_id = :adc_from_adl_id
          into :k_kontract_no, :k_disb_date, :kc_from_c_id, :kc_amount_approved, :kc_balance, :last_cell_main, :cc_name,
               :cc_status, :delinq_day
      do
      begin
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into cli_fio;

        adc_address_type_str = case adc_address_type
                                 when 1 then 'Юридический'
                                 when 2 then 'Фактический'
                                 when 4 then 'Юр. на родине'
                                 when 5 then 'Факт. на родине'
                                 else null
                               end;
        adl_address_str = null;
        select adl_address_str
        from addrlist
        where adl_id = :adc_from_adl_id
        into adl_address_str;

        if (cc_name is null) then
        begin
          memo = 'Адрес сотруд. совпадает с адресом клиента.';
        end
        else
        begin
          memo = 'Адрес сотруд. совпадает с адресом контакта клиента.';
        end

        suspend;
      end
    end

    last_cell_main = null;
    for select c_name_fio, crel_other_c_id, rel_name
        from hp_client_relation_list(:c_id, 0)
        where char_length(crel_from_rel_id) = 1
        into fio_rodstvennika, crel_other_c_id, rel_name
    do
    begin
      adc_address_type = null;
      adc_from_adl_id = null;
      adc_address_type_str = null;

      last_cell_main = f_cont_info('clients', crel_other_c_id, 'cell_main', 'last');
      if (last_cell_main = '') then
        last_cell_main = null;

      cli_fio = null;
      for select distinct k_kontract_no, k_disb_date, kc_from_c_id, kc_amount, kc_balance, kc_cell_main, cc_name,
                          cc_status, delinq_day
          from temp_cli_contacts
          where kc_cell_main = :last_cell_main
          into :k_kontract_no, :k_disb_date, :kc_from_c_id, :kc_amount_approved, :kc_balance, :last_cell_main, :cc_name,
               :cc_status, :delinq_day
      do
      begin
        select c_name_fio
        from clients
        where c_id = :kc_from_c_id
        into cli_fio;

        if (cc_name is null) then
        begin
          memo = '№ тел. родствен. сотруд. совпадает с № клиента.';
        end
        else
        begin
          memo = '№ тел. родствен. сотруд. совпадает с № контакта клиента.';
        end

        suspend;
      end

      for select adc_address_type, adc_from_adl_id
          from addrclient
          where adc_from_c_id = :crel_other_c_id and
                adc_from_adl_id <> 81001000000000 and
                adc_address_type <> 3 -- не включать адрес работы
          into adc_address_type, adc_from_adl_id
      do
      begin
        cli_fio = null;

        for select distinct k_kontract_no, k_disb_date, kc_from_c_id, kc_amount, kc_balance, kc_cell_main, cc_name,
                            cc_status, delinq_day
            from temp_cli_contacts
            where adc_from_adl_id = :adc_from_adl_id
            into :k_kontract_no, :k_disb_date, :kc_from_c_id, :kc_amount_approved, :kc_balance, :last_cell_main,
                 :cc_name, :cc_status, :delinq_day
        do
        begin
          select c_name_fio
          from clients
          where c_id = :kc_from_c_id
          into cli_fio;

          adc_address_type_str = case adc_address_type
                                   when 1 then 'Юридический'
                                   when 2 then 'Фактический'
                                   when 4 then 'Юр. на родине'
                                   when 5 then 'Факт. на родине'
                                   else null
                                 end;

          adl_address_str = null;
          select adl_address_str
          from addrlist
          where adl_id = :adc_from_adl_id
          into adl_address_str;

          if (cc_name is null) then
          begin
            memo = 'Адрес родственника сотруд. совпадает с адресом клиента.';
          end
          else
          begin
            memo = 'Адрес родственника сотруд. совпадает с адресом контакта клиента.';
          end
          suspend;
        end
      end
    end
  end
end
