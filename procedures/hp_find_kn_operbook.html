CREATE PROCEDURE HP_FIND_KN_OPERBOOK(
    REP_DATE D_DATE,
    IN_CFX_ID D_ID,
    SHOW_ALL D_BOOL)
RETURNS (
    RES_KOB_ID D_ID,
    RES_ACC_DT D_STR_CODE,
    RES_ACC_KT D_STR_CODE,
    RES_SUMMA D_CURRENCY,
    RES_LOAN_ID D_ID,
    RES_KONTRACT_NO D_STR_20,
    RES_CLIENT_FIO D_STR_255,
    RES_OL_MEMO D_STR_LONG,
    RES_OL_IS_ACTIVE D_BOOL)
AS
declare variable ol_document_client d_id;
declare variable in_ol_id d_id;
begin
  if (coalesce(show_all, 0) = 0) then
  begin
    select cfx_ol_id
    from cashflowexcel
    where cfx_id = :in_cfx_id
    into in_ol_id;
  end

  if (in_ol_id is not null) then
  begin
    for select kob_id,
               f_fmt_acc(kob_acc_dt),
               f_fmt_acc(kob_acc_kt),
               kob_summa,
               kob_from_kc_id,
               k_kontract_no,
               c_name_fio,
               ol_memo,
               kob_ol_is_active,
               ol_document_client
    from kn_operbook
         inner join operlist on (kob_from_ol_id = ol_id)
         left join knclient on (kob_from_kc_id = kc_id)
         left join kontract on (kc_from_k_id = k_id)
         left join clients on (kc_from_c_id = c_id)
    where kob_from_ol_id = :in_ol_id and
          kob_from_rml_id > 0
    into res_kob_id, res_acc_dt, res_acc_kt, res_summa, res_loan_id, res_kontract_no,
         res_client_fio, res_ol_memo, res_ol_is_active, ol_document_client
    do
    begin
      if (coalesce(res_client_fio, '') = '' and ol_document_client is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :ol_document_client
        into res_client_fio;
      end

      suspend;
    end
  end else
  begin
    for select kob_id,
               f_fmt_acc(kob_acc_dt),
               f_fmt_acc(kob_acc_kt),
               kob_summa,
               kob_from_kc_id,
               k_kontract_no,
               c_name_fio,
               ol_memo,
               kob_ol_is_active,
               ol_document_client
    from kn_operbook
         inner join operlist on (kob_from_ol_id = ol_id)
         left join knclient on (kob_from_kc_id = kc_id)
         left join kontract on (kc_from_k_id = k_id)
         left join clients on (kc_from_c_id = c_id)
    where kob_ol_date = :rep_date and
          kob_from_rml_id > 0 and
          kob_ol_is_active = 1
    into res_kob_id, res_acc_dt, res_acc_kt, res_summa, res_loan_id, res_kontract_no,
         res_client_fio, res_ol_memo, res_ol_is_active, ol_document_client
    do
    begin
      if (coalesce(res_client_fio, '') = '' and ol_document_client is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :ol_document_client
        into res_client_fio;
      end

      suspend;
    end
  end
end
