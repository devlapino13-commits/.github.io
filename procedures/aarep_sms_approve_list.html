CREATE PROCEDURE AAREP_SMS_APPROVE_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REGION_CODE VARCHAR(20),
    REGION_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    OKRUG_NAME VARCHAR(100),
    SV_KOLICH INTEGER,
    SV_VERIFIED_STR VARCHAR(100))
AS
begin
  for select region_code, region_name, branch_name, okrug_name, sv_verified_str, count(*)
      from aarep_sms_approve_list_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by region_code, region_name, branch_name, okrug_name, sv_verified_str
      into :region_code, :region_name, :branch_name, :okrug_name, :sv_verified_str, sv_kolich
  do
  begin
    suspend;
  end
end
