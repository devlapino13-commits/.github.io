CREATE PROCEDURE AAREP_SIRENA_BONUS_REP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_TYPE SMALLINT,
    DAYS_QTY INTEGER)
RETURNS (
    K_ID ID_TYPE,
    OPER_CALLED STR_100,
    PAYED_SUM CURR_TYPE,
    OK_OL_DATE DATE,
    KF_NAME STR_100,
    CC_PROJECT STR_100,
    K_KONTRACT_NO VARCHAR(20),
    DELINQ_DAY INTEGER)
AS
declare variable pant_date_begin date;
declare variable pant_date_end date;
declare variable kf_id id_type;
declare variable pant_last_call_date date;
declare variable is_ok smallint;
declare variable pant_id id_type;
declare variable ok_delinq_day integer;
declare variable oks_delinq_day integer;
declare variable is_ok2 smallint;
begin
  /*
    REP_TYPE = 1 Бонусы операторам звонившим по Сирене     (Расчет по старому)
    REP_TYPE = 2 Бонусы операторам звонившим просрочникам  (Расчет по сновому)

    REP_TYPE = 3 Бонусы операторам звонившим по Сирене     (Расчет по новому)
    REP_TYPE = 4 Бонусы операторам звонившим просрочникам  (Расчет по новому)


    Краткое описание логики:
    Беру контракты которые сейчас висят на операторах Сирены

    (старая логика 1,2)
    {
        Смотрю, были ли успешные звонки до даты конца отчета. Если да, по таким контрактам проверяю были ли погашения.
        Если погашение было,
    }
    (новая логика 3,4)
    {
        Смотрю,погашения за период. Если есть,
        по таким контрактам проверяю когда были успешные звонки.
        Если до погашения оператор совершил успешный звонок в рамках 2 недель, то
    }

    REP_TYPE = 1, 3 проверяю, висел ли контракт на вирт. сотруднике КЦ Сирена, в момент погашения
    REP_TYPE = 2, 4 проверяю, НЕ висел ли контракт на вирт. сотруднике КЦ Сирена, в момент погашения

    и не висел ли он на СБ/Коллекторах.
    Т.е. если висел на СБ/Коллекторах, то это уже их бонус.
  */

  for select pantk.pant_oper_called, pantk.pant_date_begin, pantk.pant_date_end, pantk.pant_from_k_id,
             kont.k_from_kf_id, pant_last_call_date, pant_id, rd.rd_name, kont.k_kontract_no
      from pantera_kontracts pantk
      inner join kontract kont on pantk.pant_from_k_id = kont.k_id
      inner join refdata rd on pantk.pant_rd_project_type = rd.rd_id
      where --pantk.pant_project_type in (4, 5, 6)
            rd.rd_code in ('OKTELL.Projects.Sirena', 'OKTELL.Projects.Sirena1', 'OKTELL.Projects.Sirena2') and
            pantk.pant_date_end >= :rep_date_begin and
            ((pantk.pant_cl_call_qty > 0) or
            (pantk.pant_cont_call_qty > 0)) and
            pantk.pant_date_end >= pant_date_begin -- исключаю передачи в один и тот же день
      order by pant_oper_called
      into :oper_called, :pant_date_begin, :pant_date_end, k_id, kf_id, :pant_last_call_date, :pant_id, :cc_project,
           :k_kontract_no
  do
  begin
    is_ok = 0;

    if (rep_type in (1, 2)) then -- старая логика
    begin
      /* Если был хотя бы 1 успешный звонок группе  -- >> */
      if (pant_last_call_date <= :rep_date_end) then
        is_ok = 1;

      if (is_ok = 0) then
      begin
        if (exists(select *
                   from pantera_clients pc
                   where pc.pcl_from_pan_id = :pant_id and
                         pc.pcl_res > 0 and
                         pc.pcl_called_date <= :rep_date_end rows 1)) then
          is_ok = 1;
      end
      if (is_ok = 0) then
      begin
        if (exists(select *
                   from pantera_contacts pcon
                   where pcon.pcon_from_pan_id = :pant_id and
                         pcon.pcon_res > 0 and
                         pcon.pcon_called_date <= :rep_date_end rows 1)) then
          is_ok = 1;
      end
-- <<---
      if (is_ok = 1) then
      begin
        if (pant_date_begin < rep_date_begin) then
          pant_date_begin = rep_date_begin;
        if (pant_date_end > rep_date_end) then
          pant_date_end = rep_date_end;

        payed_sum = 0;
        ok_ol_date = null;

        for select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine), ok.ok_ol_date
            from operkred ok
            where ok.ok_type = 4 and
                  ok.ok_ol_is_active = 1 and
                  ok.ok_from_c_id = 0 and
                  ok.ok_from_k_id = :k_id and
                  ok.ok_ol_date between :pant_date_begin and :pant_date_end
            group by ok_ol_date
            into :payed_sum, ok_ol_date
        do
        begin
          is_ok = 0;

-- проверяю висела ли группа на Сирене на момент погашения или нет

          if (rep_type = 1) then
          begin
            if (exists(select kal.kal_new_kf_name
                       from knaktlist kal
                       where kal.kal_from_kf_id = :kf_id and
                             :ok_ol_date between kal.kal_date and kal.kal_date_end and
                             kal.is_delete = 0 and
                             kal.kal_new_kf_officer = 1000000000004)) then
              is_ok = 1;
          end

          if (rep_type = 2) then
          begin
            if (not exists(select kal.kal_new_kf_name
                           from knaktlist kal
                           where kal.kal_from_kf_id = :kf_id and
                                 :ok_ol_date between kal.kal_date and kal.kal_date_end and
                                 kal.is_delete = 0 and
                                 kal.kal_new_kf_officer = 1000000000004)) then
              is_ok = 1;
          end

          if (is_ok = 1) then
          begin
            if (not exists( -- проверяю, висела ли группа на СБ в момент принятия погашения или нет
            select *
            from sb_akt sba
            where sba.sba_is_delete = 0 and
                  sba.sba_from_k_id = :k_id and
                  :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                  sba.sba_from_sbl_id <> 0)) then
            begin

-- проверяю, была ли группа в просрочке в день принятия погашения или нет
              delinq_day = null;
              ok_delinq_day = null;
              oks_delinq_day = 0;

              select iif(oks_date_end > cast(:ok_ol_date as date), iif(cast(:ok_ol_date as date) - oks_delinq_date < 0, 0, cast(:ok_ol_date as date) - oks_delinq_date), iif(oks_date_end - oks_delinq_date < 0, 0, oks_date_end - oks_delinq_date)) as delinq_days
              from oksummary oks
              where :ok_ol_date between oks_date_begin and oks_date_end and
                    oks_from_k_id = :k_id and
                    oks.oks_close_state = 0
              into oks_delinq_day;

              select max(ok_ol_date - ok_sh_date)
              from operkred ok
              where ok_from_k_id = :k_id and
                    ok_ol_date = :ok_ol_date and
                    ok.ok_ol_is_active = 1 and
                    ok.ok_from_c_id = 0
              into ok_delinq_day;

              if (ok_delinq_day is null) then
                ok_delinq_day = 0;

              if (ok_delinq_day <= oks_delinq_day) then
                delinq_day = oks_delinq_day;
              if (ok_delinq_day > oks_delinq_day) then
                delinq_day = ok_delinq_day;

              if (delinq_day > 0) then -- если на утро дня принятия погашения, группа была в просрочке
              begin
                kf_name = null;

-- вытаскиваю текущий номер группы
                select kal.kal_new_kf_name
                from knaktlist kal
                where kal.kal_from_kf_id = :kf_id and
                      current_date between kal.kal_date and kal.kal_date_end and
                      kal.is_delete = 0
                into :kf_name;

                suspend;
              end
            end
          end
        end
      end
    end
    else
    if (rep_type in (3, 4)) then
    begin
      if (pant_date_begin < rep_date_begin) then
        pant_date_begin = rep_date_begin;
      if (pant_date_end > rep_date_end) then
        pant_date_end = rep_date_end;

      payed_sum = 0;
      ok_ol_date = null;

      for select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine), ok.ok_ol_date
          from operkred ok
          where ok.ok_type = 4 and
                ok.ok_ol_is_active = 1 and
                ok.ok_from_c_id = 0 and
                ok.ok_from_k_id = :k_id and
                ok.ok_ol_date between :pant_date_begin and :pant_date_end
          group by ok_ol_date
          into :payed_sum, ok_ol_date
      do
      begin
        is_ok = 0;
        is_ok2 = 0;
-- проверяю висела ли группа на Сирене на момент погашения или нет

        if (rep_type = 3) then
        begin
          if (exists(select kal.kal_new_kf_name
                     from knaktlist kal
                     where kal.kal_from_kf_id = :kf_id and
                           :ok_ol_date between kal.kal_date and kal.kal_date_end and
                           kal.is_delete = 0 and
                           kal.kal_new_kf_officer = 1000000000004)) then
            is_ok = 1;
        end

        if (rep_type = 4) then
        begin
          if (not exists(select kal.kal_new_kf_name
                         from knaktlist kal
                         where kal.kal_from_kf_id = :kf_id and
                               :ok_ol_date between kal.kal_date and kal.kal_date_end and
                               kal.is_delete = 0 and
                               kal.kal_new_kf_officer = 1000000000004)) then
            is_ok = 1;
        end

        if (is_ok2 = 0) then
        begin
          if (exists(select *
                     from pantera_clients pc
                     where pc.pcl_from_pan_id = :pant_id and
                           pc.pcl_res > 0 and
                           pc.pcl_called_date between :ok_ol_date - :days_qty and :ok_ol_date rows 1)) then
            is_ok2 = 1;
        end
        if (is_ok2 = 0) then
        begin
          if (exists(select *
                     from pantera_contacts pcon
                     where pcon.pcon_from_pan_id = :pant_id and
                           pcon.pcon_res > 0 and
                           pcon.pcon_called_date between :ok_ol_date - :days_qty and :ok_ol_date rows 1)) then
            is_ok2 = 1;
        end

        if ((is_ok = 1) and
            (is_ok2 = 1)) then
        begin
          if (not exists( -- проверяю, висела ли группа на СБ в момент принятия погашения или нет
          select *
          from sb_akt sba
          where sba.sba_is_delete = 0 and
                sba.sba_from_k_id = :k_id and
                :ok_ol_date between sba.sba_date_begin and sba.sba_date_end and
                sba.sba_from_sbl_id <> 0)) then
          begin

-- проверяю, была ли группа в просрочке в день принятия погашения или нет
            delinq_day = null;
            ok_delinq_day = null;
            oks_delinq_day = 0;

            select iif(oks_date_end > cast(:ok_ol_date as date), iif(cast(:ok_ol_date as date) - oks_delinq_date < 0, 0, cast(:ok_ol_date as date) - oks_delinq_date), iif(oks_date_end - oks_delinq_date < 0, 0, oks_date_end - oks_delinq_date)) as delinq_days
            from oksummary oks
            where :ok_ol_date between oks_date_begin and oks_date_end and
                  oks_from_k_id = :k_id and
                  oks.oks_close_state = 0
            into oks_delinq_day;

            select max(ok_ol_date - ok_sh_date)
            from operkred ok
            where ok_from_k_id = :k_id and
                  ok_ol_date = :ok_ol_date and
                  ok.ok_ol_is_active = 1 and
                  ok.ok_from_c_id = 0
            into ok_delinq_day;

            if (ok_delinq_day is null) then
              ok_delinq_day = 0;

            if (ok_delinq_day <= oks_delinq_day) then
              delinq_day = oks_delinq_day;
            if (ok_delinq_day > oks_delinq_day) then
              delinq_day = ok_delinq_day;

            if (delinq_day > 0) then -- если на утро дня принятия погашения, группа была в просрочке
            begin
              select (:ok_ol_date - 1) - oks_delinq_date
              from oksummary oks
              where oks.oks_from_k_id = :k_id and
                    (:ok_ol_date - 1) between oks.oks_date_begin and oks.oks_date_end and
                    oks.oks_close_state = 0
              into :delinq_day;

              if (delinq_day >= 0) then
              begin
                kf_name = null;

-- вытаскиваю текущий номер группы
                select kal.kal_new_kf_name
                from knaktlist kal
                where kal.kal_from_kf_id = :kf_id and
                      current_date between kal.kal_date and kal.kal_date_end and
                      kal.is_delete = 0
                into :kf_name;

                suspend;
              end
            end
          end
        end
      end
    end
  end
end
