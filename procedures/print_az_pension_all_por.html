CREATE PROCEDURE PRINT_AZ_PENSION_ALL_POR(
    K_ID D_ID)
RETURNS (
    ALLOWANCE_TYPE_DESCR D_STR_255,
    ALLOWANCE_GROUP_DESCR VARCHAR(512),
    ALLOWANCE_AMOUNT D_CURRENCY)
AS
declare variable COUNTER D_INTEGER;
declare variable ALLOWANCE_COUNTER D_INTEGER;
declare variable C_ID D_ID;
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
begin
  allowance_type_descr = '';
  allowance_group_descr = '';
  allowance_amount = 0;

  select z.z_from_c_id
  from zalog z
  where z.z_from_k_id = :k_id and
        z.z_type = 0
  into :c_id;

  select eil.eil_from_bls_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
        eic.eic_code = 'pension'
  where eil.eil_from_c_id = :c_id and
        eil.eil_status = 2
  order by eil.eil_id desc
  rows 1
  into :bls_id;

  if (bls_id is not null) then
  begin
    select cast(blx_blob as d_blob_text)
    from hp_get_blobex_blob(:bls_id)
    into :json_text;

    allowance_counter = f_json_str_get_count(:json_text, 'Allowance');
    counter = 0;

    while (allowance_counter > counter) do
    begin
      allowance_type_descr = f_json_str_get_value(:json_text, 'Allowance[' || cast(counter as varchar(25)) || '].Type.Description', '');
      allowance_group_descr = f_json_str_get_value(:json_text, 'Allowance[' || cast(counter as varchar(25)) || '].Group.Description', '');
      allowance_amount = f_json_str_get_value(:json_text, 'Allowance[' || cast(counter as varchar(25)) || '].Amount', 0);
      counter = counter + 1;
      suspend;
    end
  end
  else
    suspend;
end
