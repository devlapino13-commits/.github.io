CREATE PROCEDURE HP_KN_OPERBOOK_SYS_LIST(
    KC_ID ID_TYPE,
    REP_DATE DATE,
    K_ID ID_TYPE = null)
RETURNS (
    OL_DATE DATE,
    DT_ACC_NAME D_STR_LONG,
    KT_ACC_NAME D_STR_LONG,
    OB_SUMMA CURR_TYPE,
    OB_SUMMA_TAX CURR_TYPE,
    OB_SUMMA_TAX_DT CURR_TYPE,
    OB_SUMMA_TAX_KT CURR_TYPE,
    OB_DETAIL D_STR_LONG)
AS
begin
  if (kc_id = 0) then
    kc_id = null;
  if (kc_id is null and
      k_id is null) then
    exit;
  if (kc_id is not null and
      k_id is null) then
  begin
    select kc_from_k_id
    from knclient
    where kc_id = :kc_id
    into :k_id;
  end

  for select kob_ol_date, sdt.sab_full_code, skt.sab_full_code, sum(kob_summa), sum(kob_summa_tax_dt),
             sum(kob_summa_tax_kt), obd_detail
      from operlist
      inner join kn_operbook on (kob_from_ol_id = ol_id and kob_ol_is_active = 1)
      inner join knclient on (kc_id = kob_from_kc_id)
      inner join ref_account dt on (kob_acc_dt = dt.acc_id)
      inner join ref_account kt on (kob_acc_kt = kt.acc_id)
      inner join kn_obdetail on (kob_detail_id = obd_id)
      left join sys_accbalance sdt on (dt.acc_from_sab_id = sdt.sab_id)
      left join sys_accbalance skt on (kt.acc_from_sab_id = skt.sab_id)
      where (:rep_date is null or
            kob_ol_date = :rep_date) and
            kc_amount_approved > 0 and
            kc_from_k_id = :k_id and
            (:kc_id is null or
            kc_id = :kc_id)
      group by kob_ol_date, sdt.sab_full_code, skt.sab_full_code, obd_detail
      order by kob_ol_date, min(ol_id)
      into :ol_date, dt_acc_name, kt_acc_name, ob_summa, ob_summa_tax_dt, ob_summa_tax_kt, ob_detail
  do
  begin
    ob_summa_tax = ob_summa_tax_dt;
    suspend;
  end
end
