CREATE PROCEDURE PKB_GET_KONT_DATA(
    POCKET_ID D_INTEGER)
RETURNS (
    CONTRACT_CODE D_STR_20,
    AGREEMENT_NUMBER D_STR_20,
    PROV_GROUP_EXP D_BOOL,
    FUNDING_TYPE D_INTEGER,
    CREDIT_PURPOSE D_STR_255,
    CREDIT_OBJECT D_STR_255,
    FUNDING_SOURCE D_STR_255,
    CONTRACT_PHASE D_INTEGER,
    CONTRACT_STATUS D_INTEGER,
    APPLICATION_DATE D_DATE,
    START_DATE D_DATE,
    END_DATE D_DATE,
    ACTUAL_DATE D_DATE,
    REAL_PAYMENT_DATE D_DATE,
    ANNUAL_EFFECT_RATE D_PERCENT,
    NOMINAL_RATE D_PERCENT,
    CLASSIFICATION D_INTEGER,
    TOTAL_AMOUNT CURR_TYPE,
    INSTALMENT_AMOUNT CURR_TYPE,
    INSTALMENT_COUNT D_INTEGER,
    OUTSTAND_COUNT D_INTEGER,
    OUTSTAND_AMOUNT CURR_TYPE,
    OVERDUE_COUNT D_INTEGER,
    OVERDUE_AMOUNT CURR_TYPE,
    FINE CURR_TYPE,
    PENALTY CURR_TYPE,
    LAST_PAYMENT_DATE D_DATE,
    LAST_PAYMENT_SUM CURR_TYPE,
    ACCOUNTING_DATE D_DATE,
    INS D_SMALLINT,
    DEBT CURR_TYPE,
    DEBT_PASTDUE CURR_TYPE,
    DEBT_WRITEOFF CURR_TYPE,
    DEBT_FORGIVEN CURR_TYPE,
    INTEREST CURR_TYPE,
    INTEREST_PASTDUE CURR_TYPE,
    INTEREST_WRITEOFF CURR_TYPE,
    INTEREST_FORGIVEN CURR_TYPE,
    ACCRUED_PENALTY CURR_TYPE,
    ACCRUED_PENALTY_WRITEOFF CURR_TYPE,
    ACCRUED_PENALTY_FORGIVEN CURR_TYPE)
AS
declare variable REP_DATE_END D_DATE;
declare variable K_ID D_ID;
declare variable MA_ID D_ID;
declare variable OL_ID D_ID;
declare variable S_OS CURR_TYPE;
declare variable S_PERC CURR_TYPE;
declare variable KOL_OUTSTANDING_PAYMENTS D_INTEGER;
declare variable DELINQ_OS CURR_TYPE;
declare variable DELINQ_PERC CURR_TYPE;
declare variable CIB_CODE D_STR_20;
declare variable STAVKA_RO PERC_TYPE;
declare variable IS_RESTRUCT D_INTEGER;
declare variable MCQ_STATUS D_INTEGER;
declare variable MCQ_DATE D_DATE;
begin
  select cp.cp_date_end, f_get_rd_code(cp.cp_cib_id)
  from cib_pocket cp
  where cp.cp_id = :pocket_id
  into :rep_date_end, cib_code;
  if (rep_date_end is null) then
    exit;

   accounting_date = :rep_date_end;
  for select k_id, --
             phaseofcontract, --
             days_past_due_domain
      from pkb_get_kontract_data_tmp(:pocket_id)
      into :k_id, --
           :contract_phase, --
           :contract_status
  do
  begin
    ins = 0;
    select k.k_kontract_no, --
           k.k_disb_date, --
           k.k_kredline_end, --
           k.k_fact_disb_date, --
           k.k_fact_closed_date, --
           k.k_percent_effect, --
           k.k_percent, --
           k.k_amount_approved, --
           k.k_kolich_repay, --
           k.k_from_ma_app_id
    from kontract k
    where k.k_id = :k_id
    into agreement_number, --
         start_date, --
         end_date, --
         actual_date, --
         real_payment_date, --
         annual_effect_rate, --
         nominal_rate, --
         total_amount, --
         instalment_count,
         ma_id;

    if (not
        exists(select 1
               from kontstate ks
               where ks.ks_from_k_id = :k_id and
                     ks.ks_ol_is_delete = 0 and
                     ks.ks_state = 2 and
                     current_date between ks.ks_date_begin and ks.ks_date_end)) then
      real_payment_date = null;

    if (:start_date = :rep_date_end) then
    begin
       ins = 1;
       contract_status = 27;
    end
   if (real_payment_date < end_date and contract_phase = 5) then
     contract_phase = 6;

    contract_code = :k_id;
    prov_group_exp = 1;
    funding_type =  2;
    credit_purpose =  '14';
    credit_object =  '10';
    funding_source = '01';
    classification = 1;
    outstand_count = 0;
    overdue_amount = 0;
    outstand_amount = 0;
    overdue_count = 0;
    /*                          RFCKZ-287
    DEBT CURR_TYPE,          -- Остаток непросроченного основного долга
    DEBT_PASTDUE CURR_TYPE,  -- Остаток просроченного основного долга 
    INTEREST CURR_TYPE,           -- Остаток непогашенного вознаграждения 
    INTEREST_PASTDUE CURR_TYPE,   -- Остаток просроченного вознаграждения
    DEBT_WRITEOFF CURR_TYPE, -- Остаток списанного основного долга
    INTEREST_WRITEOFF CURR_TYPE,  -- Остаток списанного вознаграждения
    ACCRUED_PENALTY CURR_TYPE,          -- Остаток непогашенной неустойки
    ACCRUED_PENALTY_WRITEOFF CURR_TYPE, -- Остаток списанной неустойки
    ACCRUED_PENALTY_FORGIVEN CURR_TYPE) -- Остаток прощенной неустойки
    DEBT_FORGIVEN CURR_TYPE, -- Остаток прощенного основного долга
    INTEREST_FORGIVEN CURR_TYPE,  -- Остаток прощенного вознаграждения 
    */
    debt = 0;
    debt_pastdue = 0;
    debt_writeoff = 0;
    debt_forgiven = 0;
    interest = 0;
    interest_pastdue = 0;
    interest_writeoff = 0;
    interest_forgiven = 0;
    accrued_penalty = 0;
    accrued_penalty_writeoff = 0;
    accrued_penalty_forgiven = 0;
    select ma.ma_date_app from  miniapp ma
    where ma.ma_id = :ma_id and ma.ma_is_active = 1 and ma.ma_date_end >= '31.12.9999'
    into  application_date;


   for select oksn.oks_ecl_percent
        from kn_oksummary oksn
        where oksn.oks_from_k_id = :k_id
        and :rep_date_end between oksn.oks_date_begin and oksn.oks_date_end
        into :stavka_ro
    do
    begin
        if (:stavka_ro between 0.01 and 5) then
            classification = 2;
        else if (:stavka_ro between 5.01 and 10) then
            classification = 3;
        else if (:stavka_ro between 10.01 and 20) then
            classification = 4;
        else if (:stavka_ro between 20.01 and 25) then
            classification = 5;
        else if (:stavka_ro between 25.01 and 50) then
            classification = 6;
        else if (:stavka_ro > 50.01) then
            classification = 7;


   select (sh.sh_os + sh.sh_perc)
    from schedule sh
    where sh.sh_from_k_id = :k_id and
        sh.sh_kommis = 0 and
        ((sh.sh_os <> 0) or
        (sh.sh_perc <> 0))
    order by sh.sh_date
    rows 1
    into :instalment_amount;


    select max(ok.ok_ol_date), ok.ok_from_ol_id
    from operkred ok
    where ok.ok_from_k_id = :k_id and
        ok.ok_type = 4 and
        ok.ok_ol_is_active = 1 and
        -(ok.ok_os + ok.ok_perc) <> 0 and
        ok.ok_ol_date <= :rep_date_end
        group by ok.ok_from_ol_id
        rows 1
    into last_payment_date, ol_id;

    select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_kommis)
    from operkred ok
    where ok.ok_from_ol_id = :ol_id
    into last_payment_sum;

    if (contract_phase = 4) then
    begin
       select iif(:rep_date_end - oks.oks_delinq_date <= 0, 0, :rep_date_end - oks.oks_delinq_date)
       from oksummary oks
       where oks.oks_from_k_id = :k_id and
             oks.oks_close_state = 0 and
             :rep_date_end between oks.oks_date_begin and oks.oks_date_end
       into :overdue_count;

        select sum(zad_os), --
        sum(sh_calc_perc),   --
        sum(zad_os + sh_calc_perc + sh_calc_fine)
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date is not null
        into :s_os, --
             :s_perc, --
             :outstand_amount;

        debt = s_os;  -- Остаток непросроченного основного долга
        interest = s_perc;  -- Остаток непогашенного вознаграждения

        select coalesce(-sum(ok_os), 0),
               coalesce(-sum(ok_perc), 0)
        from operkred
        where ok_ol_is_active = 1
          and ok_ol_date <= :rep_date_end
          and ok_from_k_id = :k_id
          and ok_from_c_id = 0
          and ok_type in (5, 6)
        into :debt_writeoff,     -- Остаток списанного основного долга
             :interest_writeoff; -- Остаток списанного вознаграждения

        accrued_penalty_forgiven = 0; -- Остаток прощенной неустойки
        accrued_penalty = 0; -- Остаток непогашенной неустойки
        accrued_penalty_writeoff = 0; -- Остаток списанной неустойки

        select coalesce(sum(-oko.oko_os), 0),
               coalesce(sum(-oko.oko_perc), 0)
        from operkroff oko
        inner join operlist ol on ol.ol_id = oko.oko_from_ol_id and
              ol.is_delete = 0
        inner join refdata rd on rd.rd_id = ol.ol_from_rd_id and
              rd.rd_main_code = 'Reason_WriteOff' and
              rd.rd_tag = 1
        where oko.oko_from_k_id = :k_id
        into: debt_forgiven,      -- Остаток прощенного основного долга
              interest_forgiven;  -- Остаток прощенного вознаграждения

        select count(sh_id)
        from aarep_make_zad_list(:k_id, :rep_date_end, 0, 0)
        into outstand_count;
        
        if (outstand_count <= 0) then
        begin
          overdue_amount = 0;
        end
        if (outstand_count > 0) then
        begin
          select count(sh_id),  --
                 sum(zad_os),  --
                 sum(zad_perc)
          from aarep_make_zad_list(:k_id, :rep_date_end, 0, 0)
          where sh_date < :rep_date_end and
                (zad_os > 0 or
                zad_perc > 0)
          into kol_outstanding_payments, --
               delinq_os, --
               delinq_perc;

          debt_pastdue = coalesce(delinq_os, 0);  -- Остаток просроченного основного долга
          interest_pastdue = coalesce(delinq_perc,0);  -- Остаток просроченного вознаграждения

          outstand_count = cast(:outstand_count as integer) - cast(:kol_outstanding_payments as integer);
          overdue_amount = coalesce(delinq_os + delinq_perc, 0);

          if (overdue_amount = 0) then
             outstand_count = 0;
          fine = 0;
          penalty  = 0;
        end
    end

    --------Добавлено для ПКБ ----------
    if(cib_code = 'CIB.PKB'
       and contract_phase in (4, 6)
       and overdue_count = 0
       and overdue_amount = 0
       and ins = 0) then
    begin
        contract_status = 1;
    end
    ------------------------------------

    -----------Добавлено, если контракт погашен в день выдачи -----------------
    if (contract_phase = 6
        and exists(select 1
                   from cib_to_send_kontracts cts
                   where cts.cts_from_k_id = :k_id)
        and ins = 1
        and contract_status = 27
        and overdue_count = 0
        and overdue_amount = 0) then
        begin
            ins = 0;
            contract_status = 1;
        end

    if (ins = 1) then
    begin
      last_payment_date = null;
      last_payment_sum = null;
    end
    if (ins = 1 and current_date > :rep_date_end) then
    begin
      ins = 0;
      contract_status = 1;
    end
    if (real_payment_date > :rep_date_end) then
      real_payment_date = null;

    is_restruct = 0;
    select is_restruct
    from aarep_get_restructuring(:k_id, current_date, 39801)
    into :is_restruct;

    mcq_status = 0;
    mcq_date = null;
    select mcq.mcq_status, mcq.mcq_answer_date
    from knclient kc
    inner join military_check_query mcq on mcq.mcq_from_c_id = kc.kc_from_c_id
    where kc.kc_from_k_id = :k_id and
          kc.kc_amount_approved > 0 and
          mcq.mcq_status = 1
    order by mcq.mcq_status_datetime desc
    rows 1
    into :mcq_status, :mcq_date;

    if (:mcq_status = 1 and
        :mcq_date >= dateadd(-14 month to current_date)) then
        contract_status = 30;

    if (:is_restruct = 1) then
    begin
        interest_writeoff = 0;
        interest_forgiven = 0;
    end

    suspend;
  end
 end
end
