CREATE PROCEDURE UXP$UXFILTER_LIST(
    OWNER_UUID_CODE D_STR_CODE,
    IS_SAVING D_BOOL,
    UN_ID D_ID)
RETURNS (
    UFL_UUID D_UUID,
    UFL_UUID_CODE D_STR_CODE,
    UFL_NAME D_STR_100,
    UFL_MEMO D_STR_255,
    UFL_KIND D_SMALLINT,
    UFL_FILTER_TEXT D_STR_4096,
    UFL_FILTER_CAPTION D_STR_4096,
    UFL_DATETIME D_DATETIME)
AS
declare variable un_name d_str_code;
begin
  if (coalesce(owner_uuid_code, '') = '') then
    exit;

  if (un_id is null) then
    un_id = uxf$user();

  un_name = uxf$user_login(un_id);

  if (coalesce(un_name, '') = '') then
    exit;

  if (is_saving is null) then
    is_saving = 0;

  ufl_kind = 0;
  if (is_saving = 1) then
  begin
    ufl_uuid_code = '';
    ufl_name = 'Сохранить как новый';
    ufl_memo = '';
    ufl_filter_text = '';
    ufl_filter_caption = '';
    ufl_datetime = null;
    suspend;
  end

  for select ufl_uuid, ufl_name, ufl_memo, ufl_kind, ufl_filter_text, ufl_filter_caption, ufl_datetime
  from ux$filter
  where ufl_owner = char_to_uuid(:owner_uuid_code) and
        (ufl_user = :un_name or (:is_saving = 0 and ufl_kind <> 0))
  order by ufl_kind, ufl_position
  into ufl_uuid, ufl_name, ufl_memo, ufl_kind, ufl_filter_text, ufl_filter_caption, ufl_datetime
  do
  begin
    if (coalesce(ufl_filter_caption, '') = '') then
      ufl_filter_caption = ufl_filter_text;

    ufl_uuid_code = uuid_to_char(ufl_uuid);
    suspend;
  end
end
