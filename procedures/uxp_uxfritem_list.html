CREATE PROCEDURE UXP$UXFRITEM_LIST(
    UX_UUID D_UUID,
    FULL_UI_ITEMS D_SMALLINT,
    UFM_UUID D_UUID = null,
    RES_KIND D_STR_CODE = null,
    UX_UUID_CODE D_STR_100 = null,
    UFM_UUID_CODE D_STR_CODE = null)
RETURNS (
    UFI_UUID D_UUID,
    UFI_PARENT_UUID D_UUID,
    UFI_UX_CODE D_STR_100,
    UFI_UFR_UUID D_UUID,
    UFI_UFR_CODE D_STR_100,
    UFI_UF_UUID D_UUID,
    UFI_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_NAME D_STR_100,
    UFI_CAPTION_ALIGN D_INTEGER,
    UFI_HINT D_STR_LONG,
    UFI_HIDDEN D_SMALLINT,
    UFI_READONLY D_BOOL,
    UFI_GROUP_DIRECTION D_INTEGER,
    UFI_H_ALIGN D_INTEGER,
    UFI_V_ALIGN D_INTEGER,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_SIZABLE D_BOOL,
    UFI_ORDER_INDEX D_SMALLINT,
    UFI_GROUP_INDEX D_SMALLINT,
    UFI_FIXED_KIND D_SMALLINT,
    UFI_SUMMARY_KIND D_SMALLINT,
    IMAGE_DETAIL D_INTEGER,
    IMAGE_INDEX D_INTEGER,
    RES_CAPTION D_STR_255,
    RES_COMMENT D_STR_LONG,
    ALLOW_CHILD D_BOOL,
    UF_IS_PARAM D_SMALLINT,
    UF_FIELD_NAME D_STR_CODE,
    UF_SOURCE D_STR_CODE,
    UF_NOT_NULL D_BOOL,
    UF_SECURED D_BOOL,
    UFI_REFRESH D_BOOL,
    UFI_UUID_STR D_STR_CODE,
    UFI_PARENT_UUID_STR D_STR_CODE,
    UFP_PUBLIC D_BOOL,
    UFI_EVENT_CONTROL D_STR_CODE,
    UFI_EXPRESSION D_STR_4096,
    RES_EXPR_KINDS D_SMALLINT,
    UFI_HEADER_ALIGN D_SMALLINT,
    UFI_VALUE_ALIGN D_SMALLINT,
    UFI_FONT_SIZE D_SMALLINT,
    UFI_ACTION_CAN_LINK D_STR_CODE,
    UFI_ACTION_NAME D_STR_255,
    UFI_IMG_CODE D_STR_CODE,
    UFI_HIDE_MENU D_BOOL)
AS
declare variable tmp_not_null d_bool;
declare variable ufp_parent d_uuid;
declare variable parent_field d_str_code;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code, 0);

  if (coalesce(ufm_uuid, '') = '') then
    ufm_uuid = uxf$uniq(ufm_uuid_code, 0);

  if (coalesce(ux_uuid, '') = '') then
    exit;

  if (full_ui_items is null) then
    full_ui_items = 0;

  ufi_ux_code = uxf$code(ux_uuid, 'fcode');

  if (coalesce(res_kind, '') = '') then
    res_kind = trim(coalesce(res_kind, ''));

  res_expr_kinds = -1;
  ufi_expression = '';

  for select ufi_uuid, ufi_parent_uuid, ufi_ux_code, ufi_ufr_uuid, ufi_ufr_code, ufi_uf_uuid, ufi_kind, ufi_control,
             ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_hidden, ufi_readonly, ufi_group_direction,
             ufi_h_align, ufi_v_align, ufi_width, ufi_height, ufi_sizable, ufi_order_index, ufi_group_index,
             ufi_fixed_kind, ufi_summary_kind, image_index, res_caption, res_comment, allow_child, uf_is_param,
             uf_field_name, uf_source, uf_not_null, uf_secured, ufi_refresh, ufi_uuid_str, ufi_parent_uuid_str,
             ufp_public, ufi_event_control, ufi_header_align, ufi_value_align, ufi_font_size
      from uxp$uxfritem_temp(:ux_uuid, :full_ui_items, :ufm_uuid, :res_kind)
      into :ufi_uuid, :ufi_parent_uuid, :ufi_ux_code, :ufi_ufr_uuid, :ufi_ufr_code, :ufi_uf_uuid, :ufi_kind, :ufi_control,
           :ufi_position, :ufi_name, :ufi_caption_align, :ufi_hint, :ufi_hidden, :ufi_readonly, :ufi_group_direction,
           :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height, :ufi_sizable, :ufi_order_index, :ufi_group_index,
           :ufi_fixed_kind, :ufi_summary_kind, :image_index, :res_caption, :res_comment, :allow_child, :uf_is_param,
           :uf_field_name, :uf_source, :uf_not_null, :uf_secured, :ufi_refresh, :ufi_uuid_str, :ufi_parent_uuid_str,
           :ufp_public, :ufi_event_control, :ufi_header_align, :ufi_value_align, :ufi_font_size
  do
  begin
    if (uf_is_param is null) then
      uf_is_param = 0;

    if (uf_field_name is null) then
      uf_field_name = '';

    if (uf_source is null) then
      uf_source = '';

    ufi_refresh = 0;
    ufp_public = 0;

    ufi_action_can_link = 1;
    ufi_action_name = '';

    if (ufi_control in ('buttonedit', 'dialogedit')) then
      ufi_action_can_link = 0;

    if (ufi_kind <> 0) then
      ufi_action_can_link = 0;

    if (ufi_control in ('button', 'label', 'dblabel')) then
      ufi_action_can_link = 1;

    if (ufi_action_can_link = 1) then
    begin
      select ua_name || '   - ' || uxa_fcode
      from ux$frbutton, ux$action, ux$alias
      where ufb_ufi_uuid = :ufi_uuid and
            ufb_ua_uuid = ua_uuid and
            ua_uxa_uuid = uxa_uuid
      into ufi_action_name;
    end

    ufi_img_code = '';
    ufi_hide_menu = null;
    if (ufi_control in ('button', 'expressbtn', 'mediabutton', 'blob', 'blob_json', 'blob_bls', 'blob_edoc')) then
    begin
      select uuid_to_char(ufb_img_uuid), ufb_hide_menu
      from ux$frbutton
      where ufb_ufi_uuid = :ufi_uuid
      into ufi_img_code, ufi_hide_menu;

      if (ufi_control <> 'button') then
        ufi_hide_menu = null;
    end

    if (ufi_uf_uuid is not null) then
    begin
      tmp_not_null = 0;
      ufp_parent = null;

      select ufp_parent, ufp_not_null, ufp_public, ufp_refresh
      from ux$frparam
      where ufp_ufr_uuid = :ufi_ufr_uuid and
            ufp_uf_uuid = :ufi_uf_uuid
      into ufp_parent, tmp_not_null, ufp_public, ufi_refresh;

      if (uf_is_param <> 1) then
        ufi_refresh = 0;

      if (full_ui_items = 1) then
        uf_not_null = tmp_not_null;

      if (ufp_parent is not null) then
      begin
        parent_field = '';
        select uf_field_name
        from ux$frparam, ux$field
        where ufp_uuid = :ufp_parent and
              ufp_uf_uuid = uf_uuid
        into parent_field;

        uf_field_name = parent_field || '$' || uf_field_name;
      end
    end

    if (uf_not_null is null) then
      uf_not_null = 0;

    allow_child = iif(ufi_kind in (-1, -2, -3), 1, 0);

    if (ufi_parent_uuid is null) then
    begin
      if (ufi_kind = -2) then -- Форма
      begin
        image_detail = 19;
        image_index = 19;
      end else
      begin
        image_detail = 17;
        image_index = 17;
      end
  
      allow_child = 1;
      uf_source = '';

      res_comment = ufi_ux_code;
      if (coalesce(ufi_hint, '') <> '')
      then res_comment = res_comment || '  - ' || ufi_hint;
  
      res_caption = 'Элементы интерфейса';
    end else
    begin
      image_detail = 1;
      if (ufi_kind = 0) then
      begin
        select editor_image
        from uxp$uxfrparam_editors
        where editor_id = :ufi_control
        into image_detail;
      end else
      begin
        image_detail = case ufi_control
          when 'group' then 0
          when 'band' then 129
          when 'button' then 103
          when 'expressbtn' then 140
          when 'label' then 2
          when 'space' then 3
          when 'grid' then 12
          when 'vgrid' then 13
          when 'tree' then 14
          when 'separator' then 15
          when 'splitter' then 16
          when 'frame' then 17
          when 'form' then 19
          else 1
        end;
      end

      image_index = case ufi_control
        when 'group' then 0
        when 'band' then 129
        when 'button' then 103
        when 'expressbtn' then 140
        when 'label' then 2
        when 'space' then 3
        when 'grid' then 12
        when 'vgrid' then 13
        when 'tree' then 14
        when 'separator' then 15
        when 'splitter' then 16
        when 'frame' then 17
        when 'form' then 19
        else 1
      end;

      if (ufi_kind <> 0) then
        uf_is_param = -1;
  
      res_comment = ufi_name;
      if (coalesce(ufi_hint, '') <> '')
      then res_comment = res_comment || '  - ' || ufi_hint;
  
      res_caption = ufi_name;

      if (res_caption = '') then
      begin
        res_caption = '<Пусто> ';

        if (ufi_kind = 1 and ufi_ufr_uuid is not null) then
          res_caption = res_caption || uxf$code(ufi_ufr_uuid, 'name');
        else
        if (ufi_parent_uuid is null) then
          res_caption = res_caption || uxf$code(ux_uuid, 'name');
        else
        if (coalesce(uf_field_name, '') <> '') then
          res_caption = res_caption || uf_field_name;
        else
          res_caption = res_caption || ufi_control || ufi_position;
      end else
      if (coalesce(uf_field_name, '') <> '' and uf_field_name <> res_caption) then
        res_caption = res_caption || '  (' || uf_field_name || ')';
    end

    ufi_uuid_str = uuid_to_char(ufi_uuid);

    res_expr_kinds = -1;
    ufi_expression = '';

    select exp_expression_all, exp_kinds
    from uxp$uxexpression_text(:ufi_uuid)
    into :ufi_expression, res_expr_kinds;

    ufi_parent_uuid_str = '';
    if (ufi_parent_uuid is not null) then
      ufi_parent_uuid_str = uuid_to_char(ufi_parent_uuid);

    suspend;
  end
end
