CREATE PROCEDURE HP_KK_CLI_ABSENT_FIX(
    K_ID ID_TYPE)
RETURNS (
    ACL_ID ID_TYPE,
    SYSTEM_TEXT STR_255,
    C_ID ID_TYPE,
    UN_ID ID_TYPE)
AS
declare variable reg_id id_type;
declare variable rd_id integer;
begin
--Процедур исправляет некоторые проблему - добавляет клиента на рассмотрение КК
--Если не указали айди контракта или не правильно ввели айди

  system_text = '';
  if (not exists(select *
                 from kontract kont
                 where k_id = :k_id)) then
  begin
    system_text = 'Введите ID-контракта правильно!';
    suspend;
    exit;
  end

  /*select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ActionGroup.Check_Loan_Data'
  into :rd_id;*/

  select rd_id
  from hp_get_check_loan_data(:k_id)
  into:rd_id;

  select al.acl_id, -- получаете родительский id (parent_id)
         al.acl_from_reg_id, al.acl_from_un_id
  from actionlist al
  where al.acl_object_kind = 9 and
        al.acl_object_id = :k_id and
        al.acl_from_rd_id = :rd_id
  order by al.acl_datetime desc
  rows 1 --Условие на всякий случаи, обычно больше одного не бывает
  into acl_id, reg_id, un_id;

  if (acl_id is null) then
  begin
    system_text = 'Контракт еще не отправлен на рассмотрение в КК!';
    suspend;
    exit;
  end

--Ошибка №1 "Не отображает клиента из группы
  for select kc_from_c_id
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into c_id
  do
  begin
    if (not exists(select *
                   from actionlist al
                   where al.acl_object_kind = 0 and
                         al.acl_parent_id = :acl_id and
                         al.acl_object_kind = 0 and
                         al.acl_object_id = :c_id)) then
    begin
      insert into actionlist (acl_parent_id, acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id,
                              acl_status, acl_respons_c_id, acl_datetime, acl_from_un_id, acl_lifetime_in_second,
                              acl_result_rd_id, acl_last_datetime, acl_dont_show_in_order, acl_quota_used)
      values (:acl_id, current_date, :reg_id, 0, :c_id, 5782, 0, null, current_timestamp, :un_id, null, 0,
              current_timestamp, 1, 0)
      returning acl_object_id
      into c_id;

      system_text = (select c_name_fio
                     from clients
                     where c_id = :c_id) || ' - добавлен для проверки';
      system_text = coalesce(system_text, 'Системная ошибка'); --Если нечего показать, значит есть ошибка
      suspend;
    end
  end
end
