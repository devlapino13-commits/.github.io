CREATE PROCEDURE HP_KONTRACT_INFO_MAIN(
    IN_K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_STR_100,
    K_KOLICH_CLIENT D_INTEGER,
    PR_NAME D_STR_255,
    LOAN_FOLDER D_STR_255,
    LOAN_FIO D_STR_100,
    LOAN_OFFICE D_STR_100)
AS
declare variable kf_id d_id;
begin
  select k_from_kf_id, k_kontract_no, k_disb_date, k_amount_approved, k_kolich_client_approved,
         pr_name
  from kontract, product
  where k_id = :in_k_id and
        k_from_pr_id = pr_id
  into kf_id, k_kontract_no, k_disb_date, k_amount_approved, k_kolich_client,
       pr_name;

  select ka.kal_new_kf_name, reg_code || ' ' || reg_name, c_name_fio
  from knaktlist ka, region, clients
  where ka.kal_from_kf_id = :kf_id and
        ka.is_delete = 0 and
        ka.kal_date <= :rep_date and
        ka.kal_new_kf_officer = c_id and
        ka.kal_new_reg_id = reg_unicode
  order by ka.kal_date desc
  rows 1
  into loan_folder, loan_office, loan_fio;

  suspend;
end
