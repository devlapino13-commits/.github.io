CREATE PROCEDURE HP_DO_KN_PEREPLAT_TO_LAW_OFF(
    KC_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    OL_ID D_ID,
    IN_KC_ID D_ID,
    NEW_KC_ID D_ID)
AS
declare variable sab_id d_integer;
declare variable bank_acc_id d_id;
declare variable bank_al_id d_str_20;
declare variable corr_sab_id d_integer;
declare variable ob_amount_sum d_currency;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable new_k_id d_id;
declare variable new_k_kontract_no d_str_20;
declare variable rd_id d_id;
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable kob_id d_id;
declare variable ok_number d_integer;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable off_sab_id_income d_integer;
declare variable off_sab_id_os d_integer;
declare variable off_sab_id_perc d_integer;
declare variable income_acc_id d_id;
declare variable acc_99999_id d_id;
declare variable reserv_sab_id d_integer;
declare variable reserv_acc_id d_id;
declare variable acc_id d_id;
declare variable amount_sum d_currency;
declare variable corr_acc_id d_id;
declare variable corr_amount_sum d_currency;
declare variable ok_pereplat d_currency;
declare variable pair_acc_id d_id;
declare variable reserv3_sab_id d_integer;
declare variable kf_id d_id;
declare variable reg_code d_str_code;
declare variable reg_id d_integer;
declare variable al_id d_str_20;
declare variable pr_id d_integer;
declare variable pereplat_al_id d_str_20;
begin
  in_kc_id = kc_id;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Погашено.Переплата'
  into :sab_id;

  select acc_id
  from ref_account
  where acc_code = '20501810000000000047'
  into bank_acc_id;

  if (bank_acc_id is null) then
    exception e_default 'Bank account not found';

  select al_id
  from acc_list
  where al_from_acc_id = :bank_acc_id
  into bank_al_id;

  rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  acc_id = null;
  amount_sum = 0;

  select acc_id, -kobs_bal_sum
  from ref_account, kn_obsummary
  where acc_from_sab_id = :sab_id and
        acc_object_kind = 8 and
        acc_object_id = :kc_id and
        kobs_acc_id = acc_id and
        kobs_date <= :rep_date and
        kobs_date_end >= :rep_date and
        kobs_bal_sum <> 0
  order by iif(kobs_bal_sum <> 0, 0, 1)
  rows 1
  into acc_id, amount_sum;

  if (acc_id is null) then
    exception e_default 'Acc_ID not found for KC_ID: ' || kc_id;

  if (amount_sum = 0) then
    exit;

  ok_pereplat = 0;
  select -sum(ok_os), max(iif(ok_sh_id is not null, ok_sh_id, 1)), max(ok_sh_date)
  from kn_operkred
  where ok_from_kc_id = :kc_id and
        ok_ol_date <= :rep_date and
        ok_ol_is_active = 1 and
        ok_type = 9
  into ok_pereplat, sh_id, sh_date;

  if (ok_pereplat is null) then
    ok_pereplat = 0;

  if (ok_pereplat <> amount_sum) then
    exception e_default f_format('Kn_Operkred (%1) and Accounting (%2) amount is different', ok_pereplat, amount_sum);

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Погашено.Госпошлина'
  into :corr_sab_id;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Резерв.Госпошлина3'
  into :reserv3_sab_id;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Доход.ИзмРезерваГоспошлина'
  into :reserv_sab_id;

  select kc_from_c_id, c_name_fio, k_id, k_kontract_no, k_from_pr_id, k_from_kf_id
  from knclient, clients, kontract
  where kc_id = :kc_id and
        kc_from_c_id = c_id and
        kc_from_k_id = k_id
  into c_id, c_name_fio, k_id, k_kontract_no, pr_id, kf_id;

  reg_code = null;
  reg_id = null;
  select reg_unicode, reg_code
  from knaktlist k, region
  where k.kal_from_kf_id = :kf_id and
        k.is_delete = 0 and
        k.kal_date <= :rep_date and
        k.kal_date_end >= :rep_date and
        k.kal_new_reg_id = reg_unicode
  into reg_id, reg_code;

  if (row_count = 0) then
  begin
    reg_id = 1010100;
    reg_code = '0101';
  end

  reserv_acc_id = null;
  select acc_id
  from ref_account
  where acc_from_sab_id = :reserv_sab_id and
        acc_from_reg_id = :reg_id
  into reserv_acc_id;

  pereplat_al_id = '';
  select res_account
  from hp_make_kredit_acclist(:pr_id, :reg_id, 8) -- Переплата
  into :pereplat_al_id;

  for select kc_id, k_id, k_kontract_no
  from knclient, kontract
  where kc_id = :kc_id and
        kc_from_k_id = k_id
  order by iif(kc_id = :kc_id, 0, 1), k_disb_date
  into new_kc_id, new_k_id, new_k_kontract_no
  do
  begin
    corr_acc_id = null;
    corr_amount_sum = 0;

    select acc_id, kobs_bal_sum
    from ref_account, kn_obsummary
    where acc_from_sab_id = :corr_sab_id and
          acc_object_kind = 8 and
          acc_object_id = :new_kc_id and
          kobs_acc_id = acc_id and
          kobs_date <= :rep_date and
          kobs_date_end >= :rep_date and
          kobs_bal_sum <> 0
    order by iif(kobs_bal_sum <> 0, 0, 1)
    rows 1
    into corr_acc_id, corr_amount_sum;

    if (corr_amount_sum = 0) then
      Continue;

    pair_acc_id = null;
    select acc_id
    from ref_account, kn_obsummary
    where acc_from_sab_id = :reserv3_sab_id and
          acc_object_kind = 8 and
          acc_object_id = :new_kc_id and
          kobs_acc_id = acc_id and
          kobs_date <= :rep_date and
          kobs_date_end >= :rep_date and
          kobs_bal_sum <> 0
    order by iif(kobs_bal_sum <> 0, 0, 1)
    rows 1
    into pair_acc_id;

    ob_amount_sum = minvalue(corr_amount_sum, amount_sum);

    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no,
                          ol_doc_no_str, ol_memo, ol_parent_id, is_delete, ol_user_id)
    values (1010100, 2, 11, :rep_date, 9, 0, '', 'Погашение госпошлины по ' || :new_k_kontract_no || ' за счет сторно переплаты по ' || :k_kontract_no || ', ' || :c_name_fio,
            null, 0, 1000000000001)
    returning ol_id
    into :ol_id;

    insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id, opl_from_acc_id)
    values (:ol_id, 2, :bank_al_id, :bank_acc_id);

    suspend;

    kob_id = null;
    select kob_id
    from hp_do_kn_operbook_memorder(:ol_id, :kc_id, null, null, -:ob_amount_sum, 0, -:ob_amount_sum, -:ob_amount_sum, 'Погашение госпошлины за счет сторно переплаты', :bank_acc_id, :acc_id, :rep_date, '810')
    into :kob_id;
  
    if (kob_id is not null) then
    begin
      corr_amount_sum = corr_amount_sum - ob_amount_sum;
      amount_sum = amount_sum - ob_amount_sum;
    end

    ok_number = null;
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                          ok_ol_reg_id, ok_ol_is_active)
    values (:ol_id, -1, :k_id, :sh_id, :sh_date, 9, :ob_amount_sum, 0, 0, 3, 0, 0, 0, 0, 0, :rep_date, 1010100, 1)
    returning ok_number
    into :ok_number;
  
    if (ok_number is null) then
      exception e_default 'Error when insert Operkred, KC_ID: ' || kc_id;
  
    delete from kn_operkred
    where ok_from_ol_id = :ol_id and
          ok_number = :ok_number;
  
    insert into kn_operkred (ok_from_ol_id, ok_number, ok_from_kc_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                             ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                             ok_ol_reg_id, ok_ol_is_active, ok_is_actual)
    values (:ol_id, :ok_number, :kc_id, :sh_id, :sh_date, 9, :ob_amount_sum, 0, 0, 3, 0, 0, 0, 0, 0, :rep_date, 1010100, 1, 1);

    kob_id = null;
    select kob_id
    from hp_do_kn_operbook_memorder(:ol_id, :new_kc_id, null, null, :ob_amount_sum, 0, :ob_amount_sum, :ob_amount_sum, 'Погашение госпошлины за счет сторно переплаты', :bank_acc_id, :corr_acc_id, :rep_date, '810')
    into :kob_id;

    kob_id = null;
    select kob_id
    from hp_do_kn_operbook_memorder(:ol_id, :new_kc_id, null, null, :ob_amount_sum - 1.00, 0, 0, 0, 'Погашение госпошлины за счет сторно переплаты', :pair_acc_id, :reserv_acc_id, :rep_date, '810')
    into :kob_id;

    ok_number = null;
    insert into operextra (ox_from_ol_id, ox_ol_date, ox_ol_is_active, ox_object_kind, ox_object_id, ox_from_rd_id,
                           ox_amount_plus, ox_amount_minus, ox_detail, ox_from_reg_id, ox_amount_writeoff)
    values (:ol_id, :rep_date, 1, 9, :new_k_id, :rd_id,
            0, :ob_amount_sum, 'Погашение госпошлины за счет сторно переплаты', 1010100, 0)
    returning (ox_id)
    into ok_number;

    delete from kn_operextra
    where ox_from_ol_id = :ol_id;

    insert into kn_operextra (ox_from_ol_id, ox_id, ox_ol_date, ox_ol_is_active, ox_from_kc_id, ox_from_rd_id,
                           ox_amount_plus, ox_amount_minus, ox_detail, ox_from_reg_id, ox_amount_writeoff)
    values (:ol_id, :ok_number, :rep_date, 1, :new_kc_id, :rd_id,
            0, :ob_amount_sum, 'Погашение госпошлины за счет сторно переплаты', 1010100, 0);


    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:ol_id, -1, '810', 1, null, :bank_al_id, -:ob_amount_sum,
            -:ob_amount_sum, :pereplat_al_id, :c_id, 'Погашение госпошлины за счет сторно переплаты',
            :rep_date, 1, 1010100);

    al_id = reg_code || '81076-02000005';

    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_account_kt, ob_client_kt, ob_detail,
                          ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
    values (:ol_id, -1, '810', 1, null, :bank_al_id, :ob_amount_sum,
            :ob_amount_sum, :al_id, :c_id, 'Погашение госпошлины за счет сторно переплаты',
            :rep_date, 1, 1010100);

    if (amount_sum <= 0) then
      exit;
  end

  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Доход.Госпошлина'
  into :off_sab_id_income;

  select acc_id
  from ref_account
  where acc_from_sab_id = :off_sab_id_income and
        acc_from_reg_id = 1010100
  into income_acc_id;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.ОС'
  into :off_sab_id_os;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Процент'
  into :off_sab_id_perc;

  select acc_id
  from ref_account
  where acc_code = '99999810000000000001'
  into acc_99999_id;

  for select kc_id, k_id, k_kontract_no
  from knclient, kontract
  where kc_from_c_id = :c_id and
        kc_from_k_id = k_id and
        kc_amount_approved > 0
  order by iif(kc_id = :kc_id, 0, 1), k_disb_date
  into new_kc_id, new_k_id, new_k_kontract_no
  do
  begin
    corr_acc_id = null;
    corr_amount_sum = 0;

    for select acc_id, kobs_bal_sum
    from ref_account, kn_obsummary
    where acc_from_sab_id = :off_sab_id_perc and
          acc_object_kind = 8 and
          acc_object_id = :new_kc_id and
          kobs_acc_id = acc_id and
          kobs_date <= :rep_date and
          kobs_date_end >= :rep_date and
          kobs_bal_sum <> 0
    order by iif(kobs_bal_sum <> 0, 0, 1)
    rows 1
    into corr_acc_id, corr_amount_sum
    do
    begin
      ob_amount_sum = minvalue(corr_amount_sum, amount_sum);
  
      insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no,
                            ol_doc_no_str, ol_memo, ol_parent_id, is_delete, ol_user_id)
      values (1010100, 2, 11, :rep_date, 9, 0, '', 'Погашение списанного по ' || :new_k_kontract_no || ' за счет сторно переплаты по ' || :k_kontract_no || ', ' || :c_name_fio,
              null, 0, 1000000000001)
      returning ol_id
      into :ol_id;

      insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id, opl_from_acc_id)
      values (:ol_id, 2, :bank_al_id, :bank_acc_id);

      suspend;
  
      kob_id = null;
      select kob_id
      from hp_do_kn_operbook_memorder(:ol_id, :kc_id, null, null, -:ob_amount_sum, 0, -:ob_amount_sum, -:ob_amount_sum, 'Погашение списанного за счет сторно переплаты', :bank_acc_id, :acc_id, :rep_date, '810')

      begin
        corr_amount_sum = corr_amount_sum - ob_amount_sum;
        amount_sum = amount_sum - ob_amount_sum;
      end

      ok_number = null;
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,

                            ok_ol_reg_id, ok_ol_is_active)
      values (:ol_id, -1, :k_id, :sh_id, :sh_date, 9, :ob_amount_sum, 0, 0, 3, 0, 0, 0, 0, 0, :rep_date, 1010100, 1)
      returning ok_number

      if (ok_number is null) then

      delete from kn_operkred
      where ok_from_ol_id = :ol_id and

      insert into kn_operkred (ok_from_ol_id, ok_number, ok_from_kc_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                               ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,

      if (income_acc_id is null) then
        exception e_default 'Income account not found';

      select kob_id
      from hp_do_kn_operbook_memorder(:ol_id, :new_kc_id, null, null, :ob_amount_sum, 0, :ob_amount_sum, :ob_amount_sum, 'Погашение списанного за счет сторно переплаты', :bank_acc_id, :income_acc_id, :rep_date, '810')
      into :kob_id;

      select kob_id
      from hp_do_kn_operbook_memorder(:ol_id, :new_kc_id, null, null, :ob_amount_sum, 0, :ob_amount_sum, :ob_amount_sum, 'Погашение списанного за счет сторно переплаты', :acc_99999_id, :corr_acc_id, :rep_date, '810')
      into :kob_id;
  
      insert into operkroff (oko_from_ol_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc, oko_other,
                             oko_sh_os, oko_sh_perc, oko_sh_other)
      values (:ol_id, :new_k_id, :c_id, 4, 0, -:ob_amount_sum, 0, 0, 0, 0);
    
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_account_kt, ob_client_kt, ob_detail,
                            ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id, -1, '810', 1, null, :bank_al_id, -:ob_amount_sum,
              -:ob_amount_sum, :pereplat_al_id, :c_id, 'Погашение списанного за счет сторно переплаты',
              :rep_date, 1, 1010100);
  
      al_id = reg_code || '81091-01000005';
  
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_account_kt, ob_client_kt, ob_detail,

      values (:ol_id, -1, '810', 1, null, :bank_al_id, :ob_amount_sum,
              :ob_amount_sum, :al_id, :c_id, 'Погашение списанного за счет сторно переплаты',
