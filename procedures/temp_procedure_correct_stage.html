CREATE PROCEDURE TEMP_PROCEDURE_CORRECT_STAGE(
    K_ID ID_TYPE)
AS
declare variable ks_state smallint;
declare variable ks_kc_id id_type;
declare variable ks_date_begin date;
declare variable ks_number integer;
declare variable lo_stage_month_h decimal(4,2);
declare variable lo_stage_month_full decimal(4,2);
begin
  for select ks_state, ks_kc_id, ks_date_begin, ks_number
      from kontstate
      where ks_from_k_id = :k_id
      into ks_state, ks_kc_id, ks_date_begin, ks_number
  do
  begin
    if (ks_state in (0, 1)) then
    begin
      lo_stage_month_h = 0;
      select lo_stage_month_h
      from hp_get_lo_stage(:ks_kc_id, :ks_date_begin)
      into lo_stage_month_h;
      lo_stage_month_full = 0;
      select lo_stage_month_full
      from hp_get_lo_stage_full(:ks_kc_id, :ks_date_begin, 'JobSystem.LO_Standart')
      into lo_stage_month_full;

      update kontstate
      set ks_lo_experience = :lo_stage_month_h,
          ks_lo_experience_full = :lo_stage_month_full
      where ks_from_k_id = :k_id and
            ks_number = :ks_number;
    end
    else
    begin
      update kontstate
      set ks_lo_experience = 0,
          ks_lo_experience_full = 0
      where ks_from_k_id = :k_id and
            ks_number = :ks_number;
    end
  end
end
