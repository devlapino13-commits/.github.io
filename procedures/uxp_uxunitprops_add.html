CREATE PROCEDURE UXP$UXUNITPROPS_ADD(
    UNIT_CODE D_STR_CODE NOT NULL,
    ITEM_UUID D_UUID NOT NULL,
    CLASS_NAME D_STR_CODE NOT NULL,
    PROP_NAME D_STR_100 NOT NULL,
    PROP_VALUE D_STR_255)
AS
declare variable unt_uuid d_uuid;
declare variable unc_uuid d_uuid;
declare variable unt_name d_str_100;
declare variable unc_is_event d_bool;
begin
  select unt_uuid, unt_name
  from ux$unit
  where unt_code = :unit_code
  into unt_uuid, unt_name;

  if (unt_uuid is null) then
    exception e_message f_format('Unit not found: %1', unit_code);

  if (class_name = unt_name) then
   class_name = 'TForm';

  unc_is_event = 0;
  select unc_uuid, unc_is_event
  from ux$unitclass
  where unc_class_name = :class_name and
        unc_prop_name = :prop_name
  into :unc_uuid, :unc_is_event;

  if (unc_uuid is null) then
  begin
    insert into ux$unitclass (unc_class_name, unc_prop_name)
    values (:class_name, :prop_name)
    returning unc_uuid, unc_is_event
    into :unc_uuid, unc_is_event;
  end

  if (prop_value is null) then
    prop_value = '';

  if (prop_value = '-') then
    prop_value = '';

  if (coalesce(unc_is_event, 0) = 0) then
    prop_value = '';

  if (exists(select * from ux$unitprops
  where unp_uni_uuid = :item_uuid and
        unp_unc_uuid = :unc_uuid)) then
  begin
    update ux$unitprops
    set unp_value = :prop_value
    where unp_uni_uuid = :item_uuid and
          unp_unc_uuid = :unc_uuid and
          unp_value is distinct from :prop_value;
  end else
  begin
    insert into ux$unitprops (unp_uni_uuid, unp_unc_uuid, unp_value)
    values (:item_uuid, :unc_uuid, :prop_value);
  end
end
