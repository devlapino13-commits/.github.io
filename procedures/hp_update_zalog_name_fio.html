CREATE PROCEDURE HP_UPDATE_ZALOG_NAME_FIO(
    C_ID D_ID,
    C_NAME_FIO D_STR_100)
AS
declare variable Z_ID D_INTEGER;
begin
    --RFCRU-3102
    if (f_get_db_id() <> 81001) then
        exit;

    for select z.z_id
    from zalog z
    inner join kontract k on k.k_id = z.z_from_k_id
    inner join kontstate ks on ks.ks_from_k_id = z.z_from_k_id
    where z.z_from_c_id = :c_id
      and k.k_status = 0
      and ks.ks_state = 0
      and current_date between ks.ks_date_begin and ks.ks_date_end
    into z_id
    do
    begin
        update zalog z
        set z.z_name_fio = :c_name_fio
        where z.z_id = :z_id
          and z.z_from_c_id = :c_id;
    end
end
