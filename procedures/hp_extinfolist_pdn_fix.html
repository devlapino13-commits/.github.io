CREATE PROCEDURE HP_EXTINFOLIST_PDN_FIX(
    KC_ID D_ID)
AS
declare variable json_full d_blob_text;
declare variable bls_id d_id;
declare variable poruch_json d_str_8100;
declare variable zalog_json d_str_8100;
declare variable is_poruch d_integer;
declare variable is_zalog d_integer;
declare variable kpl_rec_no d_id;
declare variable credid d_str_code;
declare variable kcp_id d_id;
begin
  select kcp_id
  from knclient_pdn
  where kc_id = :kc_id
    and kcp_kind = 0
  into kcp_id;

  if (not exists(select * from knclient_pdnlist
  where kpl_from_kcp_id = :kcp_id and
        (kpl_poruch is null or kpl_zalog is null))) then
    exit;

  update knclient_pdnlist
  set kpl_poruch = 0,
      kpl_zalog = 0
  where kpl_from_kcp_id = :kcp_id;

  select eil_from_bls_id
  from knclient_pdn, extinfolist
  where kcp_id = :kcp_id and
        kc_from_eil_id = eil_id
  into bls_id;

  if (bls_id is null) then
    exit;

  begin
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into :json_full;
  
    when any do
      json_full = '';
  end

  if (json_full = '') then
    exit;

  poruch_json = '';
  zalog_json = '';
  poruch_json = f_json_blob_get_value(json_full, 'response.basePart.credGuarantee', '');
  zalog_json = f_json_blob_get_value(json_full, 'response.basePart.collateral', '');

  if (poruch_json = '' and zalog_json = '') then
    Exit;

  for select kpl_rec_no, kpl_credid
  from knclient_pdnlist
  where kpl_from_kcp_id = :kcp_id and
        kpl_rec_no > 0
  into kpl_rec_no, credId
  do
  begin
    is_poruch = 0;
    is_zalog = 0;
    credId = coalesce(credId, '');
    if (credId = '') then
      Continue;

    if (poruch_json <> '' and position('"' || credId || '"', poruch_json) > 0) then
      is_poruch = 1;

    if (zalog_json <> '' and position('"' || credId || '"', zalog_json) > 0) then
      is_zalog = 1;

    if (is_poruch = 1 or is_zalog = 1) then
    begin
      update knclient_pdnlist
      set kpl_poruch = :is_poruch,
          kpl_zalog = :is_zalog
      where kpl_from_kcp_id = :kcp_id and
            kpl_rec_no = :kpl_rec_no;
    end
  end
end
