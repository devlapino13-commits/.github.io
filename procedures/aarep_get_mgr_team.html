CREATE PROCEDURE AAREP_GET_MGR_TEAM(
    RD_ID D_INTEGER,
    UN_ID D_ID,
    REP_DATE D_DATE = null)
RETURNS (
    RES_TEAM_ID D_INTEGER)
AS
declare variable lml_id d_integer;
begin
  res_team_id = null;
  if (rep_date is null) then
    rep_date = current_date;

  lml_id = f_un_mgr_id(:un_id);

  select cps.cps_team_id
  from competstructure cps
  where cps.cps_from_rd_id = :rd_id and
        cps.cps_from_lml_id = :lml_id and
        :rep_date between cps.cps_date_begin and cps.cps_date_end
  into :res_team_id;
  suspend;

end
