Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXACTIONPARAM_UXA(
    UX_UUID D_UUID,
    PARAM_KIND D_SMALLINT,
    UFM_UUID D_UUID = null,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    UAP_UA_UUID D_UUID,
    UAP_UXA_UFP_UUID D_UUID,
    UAP_UXA_UFP_CODE D_STR_CODE,
    UAP_OWNER_UFP_UUID D_UUID,
    UAP_OWNER_UFP_CODE D_STR_CODE,
    UAP_KIND D_SMALLINT,
    UAP_DEFAULT D_STR_255,
    UAP_USE_DEFAULT D_SMALLINT,
    UAP_UXA_UFP_CAPTION D_STR_255,
    UAP_LOCKED D_BOOL,
    RES_EDITOR D_STR_CODE,
    RES_STATE D_INTEGER,
    RES_READONLY D_BOOL,
    RES_UXA_UUID D_UUID,
    RES_CATEGORY_1 D_STR_255)
AS
declare variable ux_kind d_str_code;
declare variable uxa_uuid d_uuid;
declare variable ua_uuid d_uuid;
declare variable ufr_uuid d_uuid;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  if (ux_uuid is null) then
    exit;

  ux_kind = uxf$code(ux_uuid, 'kind');

  if (ux_kind not in ('uxaction', 'uxalias')) then
    exception e_default f_format('Incorrect object kind: %1', ux_kind);

  if (ux_kind = 'uxaction') then
    ua_uuid = ux_uuid;
  else
    uxa_uuid = ux_uuid;

  uap_ua_uuid = ua_uuid;
  res_state = 0;

  if (ua_uuid is not null) then
  begin
    select uxa_uuid
    from uxp$uxalias_info(:ua_uuid)
    into uxa_uuid;
  end

  res_uxa_uuid = uxa_uuid;

  if (ufm_uuid is null) then
    ufm_uuid = uxf$ver(uxa_uuid, 'draft');

  for select ufr_uuid
  from ux$frame
  where ufr_owner = :ufm_uuid
  order by ufr_is_mainframe desc, ufr_code
  into ufr_uuid
  do
  begin
    for select uap_ua_uuid, uap_uxa_ufp_uuid, uap_uxa_ufp_code,
               uap_owner_ufp_uuid, uap_owner_ufp_code,
               uap_kind, uap_default, uap_use_default, uap_uxa_ufp_caption, uap_locked, res_editor,
               res_state, res_readonly, res_uxa_uuid, res_category_1
    from uxp$uxactionparam_uxa_frame(:ux_uuid, :ufr_uuid, :param_kind)
    into :uap_ua_uuid, :uap_uxa_ufp_uuid, :uap_uxa_ufp_code,
         :uap_owner_ufp_uuid, :uap_owner_ufp_code,
         :uap_kind, :uap_default, :uap_use_default, :uap_uxa_ufp_caption, :uap_locked, :res_editor,
         :res_state, :res_readonly, :res_uxa_uuid, :res_category_1
    do
      suspend;
  end
end
