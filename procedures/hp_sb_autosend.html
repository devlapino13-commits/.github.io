CREATE PROCEDURE HP_SB_AUTOSEND(
    REP_DATE DATE)
AS
declare variable un_id id_type;
declare variable k_id id_type;
declare variable is_ok smallint;
declare variable zad_total curr_type;
declare variable k_amount_approved curr_type;
declare variable kc_amount_approved curr_type;
declare variable perc perc_type;
declare variable cl_zadol curr_type;
begin
  if (rep_date < cast('26.02.2018' as date)) then
    exit;
  if (rep_date >= cast('13.04.2018' as date)) then
    exit;

  select un_id
  from username un
  where un.un_name = 'IGUSEV'
  into :un_id;

--  беру все контракты с просрочкой 421 день и более
  for select oks_from_k_id
      from oksummary, kontract
      where :rep_date between oks_date_begin and oks_date_end and
            :rep_date - oks_delinq_date >= 421 and
            oks_from_k_id = k_id and
            oks_close_state = 0
      into :k_id
  do
  begin
--  проверяю чтобы группа не висела на СБ
    if (not exists(select *
                   from sb_akt sa
                   where sa.sba_is_delete = 0 and
                         sa.sba_from_k_id = :k_id and
                         sa.sba_status >= 0 and
                         :rep_date between sa.sba_date_begin and sa.sba_date_end)) then
    begin
      is_ok = 1;
      zad_total = 0;
-- если группа не висит не за одним из СБ,

      select sum(zad_os + sh_calc_perc + sh_calc_fine)
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      where sh_date is not null
      into :zad_total;

      select kont.k_amount_approved
      from kontract kont
      where kont.k_id = :k_id
      into :k_amount_approved;

      for select kncl.kc_amount_approved
          from knclient kncl
          where kncl.kc_from_k_id = :k_id and
                kncl.kc_amount_approved > 0
          into :kc_amount_approved
      do
      begin
        perc = (kc_amount_approved * 100) / k_amount_approved;

        cl_zadol = (zad_total * perc) / 100;

        if (cl_zadol >= 900001) then
        begin
          is_ok = 0;
        end
      end
-- если сумма задолженности на одного заемщика меньше 900 001,
-- сажаю группу на Шерматова Э. Важно ID = 25 (Есть еще одна запись Шерматов Э. с ID = 29. Нужен именно 25)
      if (is_ok = 1) then
      begin
        insert into sb_akt (sba_from_k_id, sba_number, sba_date_begin, sba_from_sbl_id, sba_status, sba_from_un_id,
                            sba_is_delete)
        values (:k_id, -1, :rep_date, 25, 2, :un_id, 0);
      end
    end
  end
end
