CREATE PROCEDURE HP_SURVEY_VIEW_ACT(
    C_ID D_ID,
    RD_ID D_INTEGER)
RETURNS (
    ACL_ID D_ID,
    ACL_STATUS D_INTEGER,
    ACL_DATE D_DATE,
    UN_FIO D_STR_100)
AS
begin
  for
    select
      acl_id,
      acl_status,
      acl_date, --acl_last_datetime,
      un_fio
    from actionlist
      inner join username
        on un_id = acl_from_un_id
    where acl_object_kind = 0
      and acl_object_id = :c_id
      and acl_from_rd_id = :rd_id
  into
    :acl_id,
    :acl_status,
    :acl_date,
    :un_fio
  do
  begin
    suspend;
  end
end
