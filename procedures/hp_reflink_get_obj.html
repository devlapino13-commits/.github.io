CREATE PROCEDURE HP_REFLINK_GET_OBJ(
    RD_STR_ID VARCHAR(255),
    REF_NAME VARCHAR(255))
RETURNS (
    RES_OBJ_KIND INTEGER,
    RES_OBJ_ID DECIMAL(18,0))
AS
declare variable RD_ID integer;
begin
  select rd_id
  from refdata
  where rd_str_id = :rd_str_id and
        rd_main_code = 'reflink' and
        rd_parent_id is not null
  into :rd_id;

  if (rd_id is null) then
  begin
    if (rd_str_id similar to '[-]?[[:DIGIT:]]+') then
    begin
      select rd_id
      from refdata
      where rd_id = :rd_str_id and
            rd_main_code = 'reflink' and
            rd_parent_id is not null
      into :rd_id;
    end
  end
  if (rd_id is null) then
    exception e_default 'Unknown refdata value in reflink: ' || :rd_str_id;

  if (coalesce(ref_name, '') = '') then
  begin
    for select rlk_object_kind, rlk_object_id
        from reflink
        where rlk_from_rd_id = :rd_id
        into :res_obj_kind, :res_obj_id
    do
    begin
      suspend;
    end
  end
  else
  begin
    select f_objkind(ref_table_name, 'id')
    from hp_reflink_category(0)
    where ref_name = :ref_name
    into :res_obj_kind;
    if (res_obj_kind is null) then
      exception e_default 'Unknown reflink category: ' || :ref_name;

    for select rlk_object_id
        from reflink
        where rlk_from_rd_id = :rd_id and
              rlk_object_kind = :res_obj_kind
        into :res_obj_id
    do
    begin
      suspend;
    end
  end
end
