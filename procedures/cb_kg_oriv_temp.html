CREATE PROCEDURE CB_KG_ORIV_TEMP(
    REP_DATE_END DATE,
    REP_DATE_BEGIN DATE = null)
RETURNS (
    K_ID ID_TYPE,
    SBL_ID INTEGER,
    REPAY_ACTIVE CURR_TYPE,
    REPAY_CLOSE CURR_TYPE,
    IS_COURT SMALLINT)
AS
begin
  if (rep_date_begin is null) then
    rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  repay_active = 0;
  repay_close = 0;
  --  проверка, были ли операции по operkred за выбранный период
  for select sba_from_k_id, sba_from_sbl_id, -sum(ok_os + ok_perc)
      from sb_akt, operkred
      where sba_is_delete = 0 and
            sba_status = 2 and
            sba_from_k_id = ok_from_k_id and
            sba_from_sbl_id > 0 and
            ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
--            sba_date_begin <= :rep_date_end and
--            sba_date_end >= :rep_date_begin and
            ok_ol_date between sba_date_begin and sba_date_end and
--            :rep_date_end between sba_date_begin and sba_date_end and
            ok_from_c_id = 0 and
            ok_type = 4 and
            sba_from_sbl_id <> 55
      group by sba_from_k_id, sba_from_sbl_id
      having sum(ok_os + ok_perc) <> 0
      into k_id, sbl_id, repay_active
  do
  begin
    if (:k_id in (109010107649000201, 109010114207900201, 109010114207900301, 109010111341700101, 109010111337500201, 109010072951100201, 109010072951100301, 109010072951100401, 109010093060500301, 109010115136100101,
                  109010116846500101, 109010108614000101)) then
      continue;
    is_court = 0;
    if (exists(select *
               from lawmain, lawkredit
               where lam_is_active = 1 and
                     lam_id = lak_from_lam_id and
                     lak_from_k_id = :k_id)) then
      is_court = 1;

    suspend;
  end
  --  проверка, были ли операции по operkroff т.е. погашении из внесистемного учета
  repay_active = 0;
  repay_close = 0;
  for select sba_from_k_id, sba_from_sbl_id, -sum(oko_os + oko_perc + oko_other)  --  RFCKG-3465
      from sb_akt, operkroff
      where sba_is_delete = 0 and
            sba_status = 2 and
            sba_from_k_id = oko_from_k_id and
            sba_from_sbl_id > 0 and
            oko_ol_is_active = 1 and
            oko_ol_date between :rep_date_begin and :rep_date_end and
            oko_ol_date between sba_date_begin and sba_date_end and
--            :rep_date_end between sba_date_begin and sba_date_end and
            oko_type = 4 and
            sba_from_sbl_id <> 55
      group by sba_from_k_id, sba_from_sbl_id
      having sum(oko_os + oko_perc) <> 0
      into k_id, sbl_id, repay_close
  do
  begin
    if (:k_id in (109010107649000201, 109010114207900201, 109010114207900301, 109010111341700101, 109010111337500201, 109010072951100201, 109010072951100301, 109010072951100401, 109010093060500301, 109010115136100101,
                  109010116846500101, 109010108614000101)) then
      continue;
    is_court = 0;
    if (exists(select *
               from lawmain, lawkredit
               where lam_is_active = 1 and
                     lam_id = lak_from_lam_id and
                     lak_from_k_id = :k_id)) then
      is_court = 1;
    suspend;
  end
end
