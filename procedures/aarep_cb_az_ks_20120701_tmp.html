RETURNS (

    AA_DELIQ_DAYS INTEGER,
    AA_CURR_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_CURR_KOLICH_CL_ACTIVE_END INTEGER,
    AA_CURR_DELIQ_CL_KOLICH INTEGER,
    AA_CURR_DELIQ_BALANCE DECIMAL(18,4),
    AA_AMOUNT_DISB DECIMAL(18,4),
    AA_KOLICH_CL_DISB INTEGER,
    AA_IS_OWN INTEGER,
    AA_CURR_KC_ID DECIMAL(18,0),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100))
AS
declare variable aa_disb_kc_id decimal(18,0);
declare variable aa_lo_region_id integer;
declare variable ps_is_on_staff integer;
declare variable ps_is_working integer;
declare variable lev_id integer;
declare variable s_level varchar(100);
declare variable jl_id_prior integer;
declare variable ps_from_jl_id integer;
declare variable aa_lo_id decimal(18,0);
declare variable aa_own_amount_active_end decimal(18,4);
declare variable aa_own_kolich_cl_active_end integer;
declare variable aa_own_deliq_cl_kolich integer;
declare variable aa_own_deliq_balance decimal(18,4);
declare variable aa_in_amount_active_end decimal(18,4);
declare variable aa_in_kolich_cl_active_end integer;
declare variable aa_in_deliq_cl_kolich integer;
declare variable aa_in_deliq_balance decimal(18,4);
declare variable aa_out_amount_active_end decimal(18,4);
declare variable aa_out_kolich_cl_active_end integer;

declare variable aa_out_deliq_balance decimal(18,4);
declare variable aa_disb_lo_fio varchar(100);
declare variable aa_lo_office_name varchar(100);
declare variable aa_lo_jobname varchar(100);
declare variable ps_is_on_staff_str varchar(100);
declare variable ps_is_working_str varchar(100);
declare variable aa_lo_branch_name varchar(100);
declare variable aa_lo_level_name varchar(100);
declare variable aa_lo_level2_name varchar(100);
declare variable aa_lo_level3_name varchar(100);
begin
  jl_id_prior = -1;
  for select aa_k_id, aa_amount_active_end, aa_kolich_cl_active_end, aa_deliq_cl_kolich, aa_deliq_balance,
             aa_disb_kc_id, /*Выдавший КС*/
             aa_curr_kc_id, /*Текущий КС*/
             aa_is_own, /*Свои*/
             aa_deliq_days, aa_amount_disb, aa_kolich_cl_disb, aa_curr_kc_fio, aa_curr_office_name, aa_curr_branch_name,
             aa_curr_level_name
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_project, :rep_period_kind, 0, 1, 0, 0, 0, 0), kontract, product
      where aa_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_loan_type = 0 and
            k_disb_date >= '01.04.2012' and
            aa_amount_active_end <> 0
      into :aa_k_id, :aa_curr_amount_active_end, :aa_curr_kolich_cl_active_end, :aa_curr_deliq_cl_kolich,
           :aa_curr_deliq_balance, :aa_disb_kc_id, :aa_curr_kc_id, :aa_is_own, :aa_deliq_days, :aa_amount_disb,

  do

  end
