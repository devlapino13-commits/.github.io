CREATE PROCEDURE UXP$USERLANG_LIST(
    USER_LANG D_STR_10)
RETURNS (
    ORIGINAL D_STR_255,
    TRANSLATED D_STR_255)
AS
begin
  if (coalesce(user_lang, '') = '') then
    user_lang = uxf$userlang();

  for select trm_orig_text, coalesce(trl_actual_text, trl_translate_text)
  from ux$transmain
       left join ux$translate on (trl_from_trm_id = trm_id and trl_lang_code = :user_lang)
  where trm_is_delphi = 1
  into original, translated
  do
    suspend;
end
