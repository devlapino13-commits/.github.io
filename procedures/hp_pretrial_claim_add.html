CREATE PROCEDURE HP_PRETRIAL_CLAIM_ADD(
    SMD_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable stage_id d_integer;
declare variable doc_kind d_integer;
declare variable doc_code d_str_code;
declare variable res_sld_id d_id;
declare variable err_text d_str_255;
begin
  -- Процедура добавляет в список документов по судебному делу документ "Претензия досудебная" (RFCRU-3114)
  -- при переходе заявки из стадии "Заявка на утверждение" в стадию "Определение подсудности"
  -- претензия добавляется автоматически через sud_maindocs_ai0 -> hp_sud_loandocs_autocreate,
  -- если же претензию необходимо добавить в заявках, которые уже прошли стадию "Заявка на утверждение",
  -- т.е. в заявках которые в работе до внедрения эл. документа претензии, то используется данная процедура.

  res_sld_id = null;
  doc_code = 'pretrial_claim';

  if (un_id is null) then
    un_id = 1000000000000;

  select sul_object_kind
  from sud_mainreq
  inner join sud_mainlog on sul_id = smr_last_sul_id
  where smr_id = :smd_id
  into stage_id;

  if (stage_id <> f_objkind('sud_maindocs', 'id')) then
    exception e_message 'Претензию можно добавить, если заявка на этапе Документы';

  select scd_id
  from sud_cfgdockind
  where scd_code = :doc_code
  into doc_kind;

  select res_sld_id
  from hp_sud_loandocs_add(:smd_id, null, :doc_kind, :un_id)
  into :res_sld_id;

  if (res_sld_id is null) then
    exception e_message 'Бланк эл. документа претензии не добавлен';

  suspend;
end
