CREATE PROCEDURE CM_KZ_0013_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable MA_ID D_INTEGER;
declare variable SUBJECT_ID D_ID;
declare variable RD_CODE D_STR_255;
declare variable STATUS D_SMALLINT;
declare variable MAC_ID D_ID;
begin
  -- RFCKG-4255 Справки о доходах (ЗП, пенсия).
  -- RFCRU-3940 Справка АСП

  for select c.c_id,
             c.c_name_fio,
             k.k_from_ma_app_id,
             mac.mac_id
      from knclient kc
      inner join kontract k on k.k_id = kc.kc_from_k_id
      inner join clients c on c.c_id = kc.kc_from_c_id
      inner join miniappclient mac on mac.mac_from_ma_id = k.k_from_ma_app_id
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      into c_id,
           c_name_fio,
           ma_id,
           mac_id
  do
  begin
    for select rd_code,
               opr_subject_id
        from objectvalues, objectprops, objectvalres, refdata
        where opr_object_kind = 0 and
              opr_object_id = :c_id and
              opr_subject_kind = '-1' and
              opr_subject_id in (9264, 9262, 9265) and -- 9264 - Является ли клиент пенсионером (Да/Нет) см. refdata
              ovs_from_opr_id = opr_id and -- 9262 - Является ли клиент трудоустроенным в настоящее время (Да/Нет) см. refdata
              ovr_result_rd_id = rd_id and -- 9265 - Является получателем АСП (да)
              ovs_id = opr_actual_ovs_id and
              ovr_from_opr_id = ovs_from_opr_id and
              ovr_from_ovs_id = ovs_id
        into rd_code,
             subject_id
    do
    begin
      if (subject_id = 9262 and rd_code containing 'Yes') then
      begin
        select coalesce(icq.icq_status, -2)
        from income_check_querys icq
        where icq.icq_from_c_id = :c_id and
              (icq.icq_from_ma_id = :ma_id or
              icq.icq_from_mac_id = :mac_id) and
              icq.icq_config = 'salary_check'
        order by icq.icq_status_datetime desc rows 1
        into status;

        if (status <> 2) then
          res_error_text = c_name_fio || ': нет информации о доходах';

        if (exists(select *
                   from clientdocs cds
                   where cds.cds_from_c_id = :c_id and
                         cds.cds_is_delete = 0 and
                         cds.cds_from_pk_id in (370))) then  -- (PASSKIND 370 - Выписка по карте)
          res_error_text = '';
      end

      if (subject_id = 9264 and rd_code containing 'Yes') then
      begin
        select coalesce(icq.icq_status, -2)
        from income_check_querys icq
        where icq.icq_from_c_id = :c_id and
              (icq.icq_from_ma_id = :ma_id or
              icq.icq_from_mac_id = :mac_id) and
              icq.icq_config = 'pension_check'
        order by icq.icq_status_datetime desc rows 1
        into status;

        if (status <> 2) then
          res_error_text = c_name_fio || ': нет информации о доходах';

        if (exists(select *
                   from clientdocs cds
                   where cds.cds_from_c_id = :c_id and
                         cds.cds_is_delete = 0 and
                         cds.cds_from_pk_id in (370))) then   -- (PASSKIND 370 - Выписка по карте)
          res_error_text = '';
      end
      
      if (subject_id = 9265 and rd_code containing 'Yes') then
      begin
        select coalesce(icq.icq_status, -2)
        from income_check_querys icq
        where icq.icq_from_c_id = :c_id and
              (icq.icq_from_ma_id = :ma_id or
              icq.icq_from_mac_id = :mac_id) and
              icq.icq_config = 'asp_check'
        order by icq.icq_status_datetime desc rows 1
        into status;

        if (status is null or status  <> 2) then
          res_error_text = 'Справка АСП не получена. Измените данные в анкете «Анализ клиента».';
      end
    end
  end

  if (coalesce(res_error_text, '') <> '') then
  begin
    res_error_is_critical = 2;
    suspend;
  end

end
