CREATE PROCEDURE UXP$UXRELATION_MODIFY(
    UF_UUID D_UUID NOT NULL,
    ACTION_KIND D_STR_CODE NOT NULL,
    TABLE_NAME D_STR_CODE NOT NULL,
    UF_UUID_CODE D_STR_CODE = null)
AS
declare variable uix_index d_str_100;
declare variable uix_virtual d_bool;
declare variable field_name d_str_code;
declare variable source_name d_str_code;
begin
  if (action_kind not in ('delete', 'insert')) then
    exception e_default 'Undefined action_kind: ' || action_kind;

  if (coalesce(uf_uuid, '') = '') then
    uf_uuid = uxf$uniq(uf_uuid_code);

  field_name = uxf$code(uf_uuid, 'code');
  source_name = uxf$owner(uf_uuid, 'code');

  if (action_kind = 'insert') then
  begin
    execute procedure uxp$uxindex_insert_virtual(:source_name, :field_name, :table_name);
  end else
  begin
    select uix_index, uix_virtual
    from ux$index
    where uix_source = :source_name and
          uix_fields = :field_name and
          uix_fk = 1 and
          uix_virtual = 1 and
          uix_fk_source = :table_name
    order by uix_virtual desc
    rows 1
    into uix_index, uix_virtual;

    if (uix_virtual = 1) then
    begin
      delete from ux$index
      where uix_index = :uix_index;
    end else
      exception e_message 'Можно удалить только виртуальные внешние ключи. Настоящие удаляются через метаданные БД.';
  end
end
