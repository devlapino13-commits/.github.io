AS
declare variable KNR_NON_RESIDENT D_INTEGER;
declare variable CLI_IS_RESIDENT_ON_DISB D_INTEGER;
declare variable K_ID ID_TYPE;
declare variable PR_CODE D_STR_100;
declare variable KC_CLIENT_TYPE D_INTEGER;
declare variable KC_IS_MSP D_SMALLINT;

begin

  for select oks_from_kc_id, k_id, max(reg_code || ' ' || reg_name), max(oks_from_k_id), max(k_kontract_no), max(c_id),
             max(k_fact_disb_online), max(c_name_fio), max(iif(oks_date_begin = :rep_date_end, oks_is_flat, 0)),
             max(iif(oks_date_begin <= :rep_date_end and oks_date_begin >= :rep_date_begin, oks_disb_date, null)),
             max(iif(oks_date_begin = :rep_date_end, oks_date_stop_perc, null)),
             min(iif(oks_date_begin <= :rep_date_end and oks_date_begin >= :rep_date_begin, oks_percent_norm, 0)),
             max(iif(oks_date_begin <= :rep_date_end and oks_date_begin >= :rep_date_begin, oks_percent_esp, 0)),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, 1, 0)), sum(oks_disb_amount),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_balance, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_kommis_amount, 0)),
             max(iif(oks_date_begin <= :rep_date_end and oks_date_begin >= :rep_date_begin, oks_kont_last_date, null)),
             max(iif(oks_date_begin <= :rep_date_end and oks_date_begin >= :rep_date_begin, oks_last_sh_date, null)),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_bal_zad_os, 0)),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_bal_zad_perc, 0)),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_vbs_end, 0)),
             max(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_delinq_date, null)),
             sum(iif(oks_date_begin = :rep_date_end, oks_delinq_days, 0)), sum(oks_pmt_os_norm), sum(oks_pmt_perc_norm),
             sum(oks_pmt_kommis), sum(oks_pmt_other_adv), sum(oks_pmt_os_delinq), sum(oks_pmt_perc_delinq),
             sum(oks_pmt_os_adv), sum(oks_pmt_perc_adv), sum(oks_pmt_is_adv), sum(oks_acr_perc),
             sum(iif(oks_date_begin = :rep_date_end, oks_bal_perc_fixed, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_bal_perc, 0)), sum(oks_acr_kommis),
             sum(iif(oks_date_begin = :rep_date_end, oks_bal_kommis, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_bal_kommis_fixed, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_is_reschedule_llr, 0)), sum(oks_acr_esp_perc),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state <> 2, oks_bal_esp_end - oks_off_esp_bal, 0)),
             sum(oks_acr_kommis_adv), sum(oks_diff_esp_perc), sum(oks_diff_esp_norm),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_diff_esp_bal, 0)),
             sum(oks_off_esp_bal), sum(iif(oks_date_begin = :rep_date_end, oks_llr_kind, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_llr_percent, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_ecl_kind, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_ecl_percent, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_ecl_period_koef, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_ecl_acc_kind, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_llr_os, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_llr_perc, 0)),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_ecl_vbs, 0)),
             sum(iif(oks_date_begin = :rep_date_end and oks_close_state = 0, oks_diff_ecl_llr, 0)), sum(oks_off_os),
             sum(oks_off_perc), sum(oks_pmt_os_off), sum(oks_pmt_perc_off), sum(oks_acr_law), sum(oks_pmt_law),
             sum(oks_off_law), sum(iif(oks_date_begin = :rep_date_end, oks_bal_law, 0)),
             sum(iif(oks_date_begin = :rep_date_end, oks_reserv_law, 0)),

             max(iif(oks_date_begin = :rep_date_end, oks_ead_future_vbs, null)),
             max(iif(oks_date_begin = :rep_date_end, oks_ead_future_perc, null)),
             max(iif(oks_date_begin = :rep_date_end, oks_ead_estimate_delinq_date, null)),
             max(iif(oks_date_begin = :rep_date_end, oks_ead_estimate_def_date, null)),
             max(iif(oks_date_begin = :rep_date_end, oks_ead_amount, null)), pr.pr_name, kc_amount_approved,
             1 - coalesce(kc_non_resident, 0),
             max(kc_client_type), max(kc_is_msp)
      from kn_oksummary, knclient, kontract, clients, knaktlist akt, region, product pr
      where oks_date_begin between :rep_date_begin and :rep_date_end and
            oks_from_kc_id = kc_id and
            kc_from_k_id = k_id and
            kc_from_c_id = c_id and
            akt.kal_from_kf_id = k_from_kf_id and
            akt.kal_date <= :rep_date_end and
            akt.kal_date_end >= :rep_date_end and
            akt.is_delete = 0 and
            akt.kal_new_reg_id = reg_unicode and
            pr.pr_id = k_from_pr_id
            --and k_kontract_no = '2201-675374-04'








      group by oks_from_kc_id, k_id, pr.pr_name, kc_amount_approved, 1 - coalesce(kc_non_resident, 0)
      into oks_from_kc_id, k_id, reg_name, oks_from_k_id, k_kontract_no, c_id, k_is_online, c_name_fio, oks_is_flat,
           oks_disb_date, oks_date_stop_perc, oks_percent_norm, oks_percent_esp, oks_is_active, oks_disb_amount,
           oks_balance, oks_kommis_amount, oks_kont_last_date, oks_last_sh_date, oks_bal_zad_os, oks_bal_zad_perc,
           oks_vbs_end, oks_delinq_date, oks_delinq_days, oks_pmt_os_norm, oks_pmt_perc_norm, oks_pmt_kommis,
           oks_pmt_other_adv, oks_pmt_os_delinq, oks_pmt_perc_delinq, oks_pmt_os_adv, oks_pmt_perc_adv, oks_pmt_is_adv,
           oks_acr_perc, oks_bal_perc_fixed, oks_bal_perc, oks_acr_kommis, oks_bal_kommis, oks_bal_kommis_fixed,
           oks_is_reschedule, oks_acr_esp_perc, oks_bal_esp_end, oks_acr_kommis_adv, oks_diff_esp_perc,
           oks_diff_esp_norm, oks_diff_esp_bal, oks_off_esp_bal, oks_llr_kind, oks_llr_percent, oks_ecl_kind,
           oks_ecl_percent, oks_ecl_period_koef, oks_ecl_acc_kind, oks_llr_os, oks_llr_perc, oks_ecl_vbs,
           oks_diff_ecl_llr, oks_off_os, oks_off_perc, oks_pmt_os_off, oks_pmt_perc_off, oks_acr_law, oks_pmt_law,
           oks_off_law, oks_bal_law, oks_reserv_law, oks_ead_future_vbs, oks_ead_future_perc,
           oks_ead_estimate_delinq_date, oks_ead_estimate_def_date, oks_ead_amount, pr_name, kc_amount,
           cli_is_resident_on_disb,
           kc_client_type, kc_is_msp
  do
  begin
    if (coalesce(oks_vbs_end, 0) = 0) then
      perc_proviz = 0;
    else
      perc_proviz = oks_ecl_vbs / oks_vbs_end * 100;

    select round(k_percent_per_day * 365.00, 3),pr.pr_code
    from kontract kon, product pr
    where kon.k_id = :k_id and
    pr.pr_id = kon.k_from_pr_id
    into :k_percent_flat_annual, pr_code;

    if (pr_code containing '.selfemployed.') then
    begin
      client_type = 'СЗ';
    end else
    begin
      kc_is_msp = coalesce(kc_is_msp, 0);

      if (kc_client_type = 1) then
      begin
        client_type = 'ЮЛ';
      end else
      if (kc_client_type = 2) then
      begin
        if (kc_is_msp = 1) then client_type = 'ИП-МСП';
          else client_type = 'ИП-неМСП';
      end else
      begin
        client_type = 'ФЛ';
      end
    end

    oks_ecl_acc_kind_beg = 0;
    select oks_ecl_acc_kind
    from kn_oksummary
    where oks_from_kc_id = :oks_from_kc_id and
          oks_date_begin < :rep_date_begin
    order by oks_date_begin desc
    rows 1
    into :oks_ecl_acc_kind_beg;

    oks_pmt_all = oks_pmt_os_norm + oks_pmt_perc_norm + oks_pmt_kommis + oks_pmt_other_adv + oks_pmt_os_delinq + oks_pmt_perc_delinq + oks_pmt_os_adv + oks_pmt_perc_adv;

    oks_pmt_other_all = oks_pmt_kommis + oks_pmt_other_adv;

    oks_pmt_os_all = oks_pmt_os_norm + oks_pmt_os_delinq + oks_pmt_os_adv;

    oks_pmt_perc_all = oks_pmt_perc_norm + oks_pmt_perc_delinq + oks_pmt_perc_adv;

    r_od = 0;
    r_proc = 0;
    r_kor = 0;
    r_ro = 0;
    diff_acr_perc = 0;
    acr_perc_prov = 0;

    select coalesce(sum(
           case
             when s_dt.sab_full_code = 'Микрозаймы.Расход.ИзмРезерва' and s_kt.sab_full_code in ('Микрозаймы.Резерв.ОС1', 'Микрозаймы.Резерв.ОС2', 'Микрозаймы.Резерв.ОС3') then ob.kob_summa
             else 0
           end), 0) -

           coalesce(sum(
           case
             when s_dt.sab_full_code in ('Микрозаймы.Резерв.ОС1', 'Микрозаймы.Резерв.ОС2', 'Микрозаймы.Резерв.ОС3') and s_kt.sab_full_code = 'Микрозаймы.Доход.ИзмРезерва' then ob.kob_summa
             else 0
           end), 0) r_od,

           coalesce(sum(
           case
             when s_dt.sab_full_code = 'Микрозаймы.Расход.ИзмРезерва' and s_kt.sab_full_code in ('Микрозаймы.Резерв.Процент1', 'Микрозаймы.Резерв.Процент2', 'Микрозаймы.Резерв.Процент3') then ob.kob_summa
             else 0
           end), 0) -

           coalesce(sum(
           case
             when s_dt.sab_full_code in ('Микрозаймы.Резерв.Процент1', 'Микрозаймы.Резерв.Процент2', 'Микрозаймы.Резерв.Процент3') and s_kt.sab_full_code = 'Микрозаймы.Доход.ИзмРезерва' then ob.kob_summa
             else 0
           end), 0) r_proc,

           coalesce(sum(
           case
             when s_dt.sab_full_code = 'Микрозаймы.Расход.ИзмРезерваРО' and s_kt.sab_full_code in ('Микрозаймы.Резерв.Корректировка1', 'Микрозаймы.Резерв.Корректировка2', 'Микрозаймы.Резерв.Корректировка3') then ob.kob_summa
             else 0
           end), 0) -

           coalesce(sum(
           case
             when s_dt.sab_full_code in ('Микрозаймы.Резерв.Корректировка1', 'Микрозаймы.Резерв.Корректировка2', 'Микрозаймы.Резерв.Корректировка3') and s_kt.sab_full_code = 'Микрозаймы.Доход.ИзмРезерваРО' then ob.kob_summa
             else 0
           end), 0) r_kor,

           coalesce(sum(
           case
             when (s_dt.sab_full_code = 'Микрозаймы.Начислено.Процент' and s_kt.sab_full_code = 'Микрозаймы.Доход.Процент') or (s_dt.sab_full_code = 'Микрозаймы.Начислено.ДопПроцент' and s_kt.sab_full_code = 'Микрозаймы.Доход.ДопПроцент') or (s_dt.sab_full_code = 'Микрозаймы.Баланс.ИзмВБС' and s_kt.sab_full_code = 'Микрозаймы.Доход.КорРазницаСтавок') then ob.kob_summa
             when s_dt.sab_full_code = 'Микрозаймы.Доход.КорРазницаСтавок' and s_kt.sab_full_code = 'Микрозаймы.Баланс.ИзмВБС' then -ob.kob_summa
             else 0
           end), 0) acr_perc_prov

    from kn_operbook ob
    left join ref_account a_kt on a_kt.acc_id = ob.kob_acc_kt
    left join ref_account a_dt on a_dt.acc_id = ob.kob_acc_dt
    left join sys_accbalance s_dt on a_dt.acc_from_sab_id = s_dt.sab_id
    left join sys_accbalance s_kt on a_kt.acc_from_sab_id = s_kt.sab_id
    where (a_kt.acc_object_kind = 8 or
          a_dt.acc_object_kind = 8) and
          ob.kob_ol_is_active = 1 and
          ob.kob_from_kc_id = :oks_from_kc_id and
          s_dt.sab_create_kind <> 0 and
          s_kt.sab_create_kind <> 0 and
          ob.kob_ol_date between :rep_date_begin and :rep_date_end and
          s_dt.sab_full_code in ('Микрозаймы.Расход.ИзмРезерва', 'Микрозаймы.Резерв.ОС1', 'Микрозаймы.Резерв.ОС2', 'Микрозаймы.Резерв.ОС3', 'Микрозаймы.Резерв.Процент1', 'Микрозаймы.Резерв.Процент2', 'Микрозаймы.Резерв.Процент3', 'Микрозаймы.Расход.ИзмРезерваРО', 'Микрозаймы.Резерв.Корректировка1', 'Микрозаймы.Резерв.Корректировка2',
                                 'Микрозаймы.Резерв.Корректировка3', 'Микрозаймы.Начислено.Процент', 'Микрозаймы.Начислено.ДопПроцент', 'Микрозаймы.Баланс.ИзмВБС', 'Микрозаймы.Доход.КорРазницаСтавок') and
          s_kt.sab_full_code in ('Микрозаймы.Резерв.ОС1', 'Микрозаймы.Резерв.ОС2', 'Микрозаймы.Резерв.ОС3', 'Микрозаймы.Доход.ИзмРезерва', 'Микрозаймы.Резерв.Процент1', 'Микрозаймы.Резерв.Процент2', 'Микрозаймы.Резерв.Процент3', 'Микрозаймы.Резерв.Корректировка1', 'Микрозаймы.Резерв.Корректировка2', 'Микрозаймы.Резерв.Корректировка3',
                                 'Микрозаймы.Доход.ИзмРезерваРО', 'Микрозаймы.Доход.Процент', 'Микрозаймы.Доход.ДопПроцент', 'Микрозаймы.Доход.КорРазницаСтавок', 'Микрозаймы.Баланс.ИзмВБС')
    into r_od, r_proc, r_kor, acr_perc_prov;

    r_ro = r_od + r_proc + r_kor;
    diff_acr_perc = oks_acr_esp_perc - acr_perc_prov;

    knr_non_resident = 1 - cli_is_resident_on_disb;
    select knr_non_resident
    from kn_resident
    where knr_from_kc_id = :oks_from_kc_id and
          :rep_date_end between knr_date_begin and knr_date_end
    into knr_non_resident;

    acc_no = null;
    if (knr_non_resident = 1) then
      acc_no = '492';
    else
    if (knr_non_resident = 0) then
      acc_no = '488';

    suspend;
  end
