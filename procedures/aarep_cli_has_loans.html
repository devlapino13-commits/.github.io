CREATE PROCEDURE AAREP_CLI_HAS_LOANS(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(61),
    PARENT_NAME VARCHAR(61),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_BIRTH_DATE DATE,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KF_OFFICER VARCHAR(100),
    KAL_NEW_KF_NAME VARCHAR(100),
    K_KONT_BALANCE DECIMAL(18,4),
    K_CLI_BALANCE DECIMAL(18,4),
    K_INDEX_CURR_CODE VARCHAR(3),
    K_INDEX_CURR_RATE DECIMAL(18,4),
    KC_HAS_LOANS_STR VARCHAR(100),
    KC_HAS_FAMILY_LOANS_STR VARCHAR(100),
    LK_DATE DATE,
    LLC_BEGIN_RESERVE VARCHAR(50),
    LLC_CURRENT_STATUS_STR VARCHAR(50),
    LLC_CURRENT_CLASS VARCHAR(50),
    REG_STAT_REP_REGION VARCHAR(255))
AS
declare variable k_id decimal(18,0);
declare variable kc_has_loans integer;
declare variable kc_has_family_loans integer;
declare variable old_k_id decimal(18,0);
declare variable cr_from_cu_id varchar(3);
declare variable cr_rate decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  old_k_id = 0;
  k_cli_balance = 0;

  /*Вытаскивает курс доллара на указанную дату*/
  select cr_from_cu_id, cr_rate
  from opercurr
  where :rep_date_end between cr_date and cr_date_end and
        cr_from_cu_id = '840'
  into :cr_from_cu_id, :cr_rate;

  for select r1.reg_code || ' ' || r1.reg_name as reg_name, r2.reg_code || ' ' || r2.reg_name as parent_name, k_id,
             k_kontract_no, k_disb_date, k_amount_approved, k_index_curr_code, k_index_curr_rate, cli.c_name_fio,
             cli.c_pass_no, cli.c_birth_date, kc3.kc_amount_approved, kc3.kc_has_loans, kc3.kc_has_family_loans,
             off.c_name_fio as kf_officer, kal_new_kf_name, r1.reg_stat_rep_region

      from knclient kc3, kontract, kontstate, clients cli, knaktlist, clients off, region r1, region r2, region_list(:rep_region_id, 1, 3, 1)
      where kc3.kc_amount_approved > 0 and
            (kc3.kc_has_loans <> 0 or
            kc3.kc_has_family_loans <> 0 or
            (exists(select 1
                    from kontstate ks1, kontstate ks2, knclient kc1, knclient kc2
                    where ks1.ks_ol_is_delete = 0 and
                          ks1.ks_state = 1 and
                          :rep_date_end between ks1.ks_date_begin and ks1.ks_date_end and
                          ks1.ks_from_k_id = kc1.kc_from_k_id and
                          ks2.ks_ol_is_delete = 0 and
                          ks2.ks_state = 1 and
                          :rep_date_end between ks2.ks_date_begin and ks2.ks_date_end and
                          ks2.ks_from_k_id = kc2.kc_from_k_id and
                          kc1.kc_from_c_id = kc2.kc_from_c_id and
                          kc1.kc_from_c_id = kc3.kc_from_c_id -- Привязка с внешней таблицей
                          and
                          kc1.kc_from_k_id <> kc2.kc_from_k_id and
                          kc1.kc_amount_approved > 0 and
                          kc2.kc_amount_approved > 0 and
                          kc1.kc_has_loans = 0 and
                          cast(ks1.ks_date_begin as date) > cast(ks2.ks_date_begin as date) and
                          ks1.ks_date_begin <= :rep_date_end and
                          kc1.kc_from_k_id = k_id))) and
            kc3.kc_from_k_id = k_id and
            ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_state = 1 and
            cli.c_id = kc3.kc_from_c_id and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            off.c_id = kal_new_kf_officer and
            r1.reg_unicode = kal_new_reg_id and
            r2.reg_unicode = r1.parent and
            kal_new_reg_id = id_list
      order by k_id /*Сортировка для того чтоб, у контракта повторный раз не пересчитывал активный портфель*/
      into :reg_name, :parent_name, :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :k_index_curr_code,
           :k_index_curr_rate, :c_name_fio, :c_pass_no, :c_birth_date, :kc_amount_approved, :kc_has_loans,
           :kc_has_family_loans, :kf_officer, :kal_new_kf_name, :reg_stat_rep_region
  do
  begin
    k_cli_balance = null;

    /*Вычисления активного портфеля контракта :KS_FROM_K_ID*/
    if (old_k_id <> :k_id) then
    /*Проверяет на повторность контракта, если контракт повторяеться то не будет повторно считать*/
    begin
      k_kont_balance = 0;
      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_close_state = 0
      into :k_kont_balance;

      if (:k_index_curr_code = :cr_from_cu_id) then
      /*Если контракт долларовый то переводит в эквивалент местной валюты*/
      begin
        if (:k_index_curr_rate > :cr_rate) then
          k_kont_balance = :k_kont_balance * :k_index_curr_rate;
        else
          k_kont_balance = :k_kont_balance * :cr_rate;
      end
      /*Вытаскивает классификацию данного клиента*/
      lk_date = null;
      llc_begin_reserve = null;
      llc_current_status_str = null;
      llc_current_class = null;
      select lk_date, cs_llr_kind_text, cs_name, cs_class_text
      from llc_kont, classific, llc_main
      where lk_status = cs_id and
            lk_from_lm_id = lm_id and
            lm_is_signed = 1 and
            lk_from_k_id = :k_id
      order by lk_date desc
      rows 1
      into :lk_date, :llc_begin_reserve, :llc_current_status_str, :llc_current_class;

    end

    k_cli_balance = :kc_amount_approved / :k_amount_approved * :k_kont_balance;/*Вычисляет долю портфеля клиента*/
    old_k_id = :k_id;

    if (kc_has_loans = 1) then
      kc_has_loans_str = 'Клиент имеет кредит в других ФКУ';
    else
      kc_has_loans_str = null;
    if (kc_has_family_loans = 1) then
      kc_has_family_loans_str = 'У родственника есть кредит, имеют единый источник дохода';
    else
      kc_has_family_loans_str = null;

    suspend;
  end
end
