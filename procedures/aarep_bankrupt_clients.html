CREATE PROCEDURE AAREP_BANKRUPT_CLIENTS(
    REP_REGION_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    EIL_DATE D_DATE,
    K_KONTRACT_NO D_STR_CODE,
    C_NAME_FIO D_STR_100,
    K_DISB_DATE D_DATE,
    BFE_BANKRUPTCY_DATE D_DATE)
AS
begin
  for select eil.eil_date,
             k.k_kontract_no,
             c.c_name_fio,
             k.k_disb_date,
             bfe.bfe_bankruptcy_date
      from bankruptcy_for_exec bfe
      inner join extinfolist eil on eil.eil_id = bfe.bfe_from_eil_id
      inner join clients c on c.c_id = eil.eil_from_c_id
      inner join kontract k on k.k_id = bfe.bfe_from_k_id
      where bfe.bfe_status = 1 and
            eil.eil_date between :rep_date_begin and :rep_date_end
      order by c.c_name_fio
      into :eil_date,
           :k_kontract_no,
           :c_name_fio,
           :k_disb_date,
           :bfe_bankruptcy_date
  do
  begin
    suspend;
  end
end
