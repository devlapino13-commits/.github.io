CREATE PROCEDURE HP_GET_CLIENT_ADDRESS_DISTANCE(
    C_ID DECIMAL(18,0),
    ADDR_TYPE INTEGER,
    REGION_TO_DISTANCE INTEGER)
RETURNS (
    RES_DISTANCE DOUBLE PRECISION)
AS
declare variable from_lat double precision;
declare variable from_lon double precision;
declare variable res_lat double precision;
declare variable res_lon double precision;
begin
  res_distance = -1;

  select adl_lat, adl_long
  from addrlist, addrclient
  where adc_from_c_id = :c_id and
        adc_address_type = :addr_type and
        adc_from_adl_id = adl_id
  into :res_lat, res_lon;

  if (res_lat is null) then
    res_lat = 0;
  if (res_lon is null) then
    res_lon = 0;
  if (res_lat = 0 and
      res_lon = 0) then
  begin
    suspend;
    exit;
  end

  if (region_to_distance > 0) then
  begin
    select adl_lat, adl_long
    from addrlist, region
    where reg_unicode = :region_to_distance and
          reg_address_fact = adl_id
    into :from_lat, from_lon;
  end

  if (from_lat > 0 or from_lon > 0) then
    res_distance = f_gps_distance(:from_lat, :from_lon, :res_lat, :res_lon) / 1000.000;

  suspend;
end
