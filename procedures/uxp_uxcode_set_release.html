CREATE PROCEDURE UXP$UXCODE_SET_RELEASE(
    UFM_UUID D_UUID,
    IS_RELEASE D_BOOL = 0,
    UFM_UUID_CODE D_STR_CODE = null)
AS
declare variable ux_uuid d_uuid;
begin
  if (ufm_uuid is null) then
    ufm_uuid = uxf$uniq(ufm_uuid_code);

  select ufm_ux_uuid
  from ux$form
  where ufm_uuid = :ufm_uuid
  into ux_uuid;

  if (row_count = 0) then
    exception e_message f_format('Форма не найдена: %1', uuid_to_char(ufm_uuid));

  if (is_release = 0) then
    ufm_uuid = null;

  update ux$code
  set ux_release_uuid = :ufm_uuid
  where ux_uuid = :ux_uuid and
        ux_release_uuid is distinct from :ufm_uuid;
end
