CREATE PROCEDURE AAREP_CONTACTS_ACTIVE_CLI(
    REP_DATE DATE,
    REPORT_REGION_ID INTEGER)
RETURNS (
    C_NAME_FIO STR_100,
    K_KONTRACT_NO STR_100,
    KF_NAME STR_100,
    OFFICER_FIO STR_100,
    CC_NAME STR_100,
    CC_STATUS STR_100,
    CC_CELL_NO STR_100,
    IS_OLD_CONTACT INTEGER)
AS
declare variable k_id id_type;
declare variable kf_officer_id_old id_type;
declare variable kf_officer_id id_type;
declare variable c_id id_type;
begin
  officer_fio = '';
  kf_officer_id_old = 0;

  for select k_id, kont.k_kontract_no, kal.kal_new_kf_name, kal.kal_new_kf_officer
      from kontstate ks
      inner join kontract kont on kont.k_id = ks.ks_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      inner join region_list(:report_region_id, 1, -1, 1) on id_list = kal.kal_new_reg_id
      where ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            kal.is_delete = 0 and
            :rep_date between kal.kal_date and kal.kal_date_end
      order by kal.kal_new_kf_officer
      into :k_id, :k_kontract_no, :kf_name, :kf_officer_id
  do
  begin
    if (kf_officer_id <> kf_officer_id_old) then
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :kf_officer_id
      into :officer_fio;
    end

    for select cl.c_name_fio, cl.c_id
        from knclient kncl
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        where kncl.kc_from_k_id = :k_id and
              kncl.kc_amount_approved > 0
        into :c_name_fio, :c_id
    do
    begin
      for select cont.cc_name, cont.cc_status, cont.cc_cell_no, coalesce(cont.cc_is_old_contact, 0)
          from contacts cont
          where cont.cc_from_c_id = :c_id
          order by cc_is_old_contact
          into :cc_name, :cc_status, :cc_cell_no, :is_old_contact
      do
      begin
        suspend;
      end
    end
  end
end
