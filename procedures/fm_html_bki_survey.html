CREATE PROCEDURE FM_HTML_BKI_SURVEY(
    K_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable html_text d_str_8100;
declare variable html_row d_str_8100;
declare variable kc_id d_id;
declare variable survey_res d_str_code;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
begin
  res_text = (select res_style from fm_html_styles);

  for select kc_id, c_id, c_name_fio
  from knclient, clients
  where c_id = kc_from_c_id
    and kc_from_k_id = :k_id
    and kc_amount_approved > 0
  into kc_id, c_id, c_name_fio
  do
  begin
    html_text = '<body><h1>' || c_name_fio || '</h1><br>';
    html_text = html_text || '<table width="100%"><col width="5%"><col width="60%"><col width="35%"><th>№</th><th>Вопрос</th><th>Ответ</th>';
  
    for select f_format('<tr><td>%1</td><td>%2</td><td>%3</td></tr>', str_no, question_str, answer_str)
    from hp_bki_get_survey_content(:c_id)
    into html_row
    do
      html_text = html_text || html_row;
  
    html_text = html_text || '<br><b>Сохраненный результат: </b>';

    survey_res =  f_knclient_data(kc_id, 'bki_survey_res');

    survey_res =  case survey_res
                    when '1' then 'Положительный'
                    when '-1' then 'Отрицательный'
                    else coalesce('Не определен ('|| survey_res ||')', 'Нет сохраненного результата')
                  end;

    html_text = html_text || survey_res || ' ' ||
        f_uri_sql('Изменить', 'set_bki_survey_res', 'knclient', kc_id, 'combo', 'Укажите результат опроса клиента по БКИ:', '', 0, 0, '1', '', 'dialog', '') ||
        '<br>';
  end

  html_text = html_text || '</body>';

  res_text = BLOB_APPEND(res_text, html_text);


  suspend;
end
