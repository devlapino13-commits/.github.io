CREATE PROCEDURE HP_ACTIONLIST_OBJECT_INSERT(
    OBJECT_KIND_STR VARCHAR(100),
    OBJECT_ID DECIMAL(18,0),
    ACTION_CODE VARCHAR(255),
    UN_ID DECIMAL(18,0))
RETURNS (
    ACL_ID DECIMAL(18,0),
    RES_MESSAGE VARCHAR(255))
AS
declare variable reg_id integer;
declare variable c_id decimal(18,0);
declare variable tmp_acl_id decimal(18,0);
declare variable acl_from_rd_id integer;
declare variable db_id d_id;
declare variable k_disb_date d_date;
declare variable kf_id d_id;
declare variable lo_id d_id;
declare variable rd_code d_str_100;
begin
  -- Добавляет объект в список действий (ActionList)
  -- Здесь можно произвезти проверку готовности, например утвержден ли кредит

  res_message = ''; -- если ошибка (или условия не позволяют создать объект, Res_Message будет возвращать текст
  db_id = f_get_db_id();

  acl_id = 0;
  if (not(action_code starting 'ActionGroup.')) then
    action_code = 'ActionGroup.' || action_code;

  if (object_kind_str = 'Kontract') then
  begin
    select rd_code
    from hp_get_check_loan_data(:object_id)
    into :rd_code;

    if (rd_code = 'ActionGroup.Check_Loan_Data_Special') then
      action_code = 'ActionGroup.Check_Loan_Data_Special';

  end

  if (action_code = 'ActionGroup.Check_Loan_Data' or action_code = 'ActionGroup.Check_Loan_Data_Special') then
  begin
    if (not(exists(select *
                   from refdata
                   where rd_code = :action_code and
                         rd_is_active = 1))) then
    begin
      res_message = 'Данная опция пока отключена';
      suspend;
      exit;
    end
    if (exists(select *
               from kontstate
               where ks_from_k_id = :object_id and
                     ks_state <> 0 and
                     ks_ol_is_delete = 0 and
                     ks_date_end >= '31.12.2999')) then
    begin
      res_message = 'На проверку необходимо отправлять только кредиты в процессе';
      suspend;
      exit;
    end
  end
  else
  begin
    res_message = 'Данная опция пока отключена';
    suspend;
    exit;
  end

  reg_id = null;
  if (object_kind_str = 'Kontract' and
      (action_code = 'ActionGroup.Check_Loan_Data' or action_code = 'ActionGroup.Check_Loan_Data_Special')) then
  begin
    select k_last_reg_id, kt.k_disb_date, kt.k_from_kf_id
    from kontract kt
    inner join product pr on pr.pr_id = kt.k_from_pr_id
    where k_id = :object_id
    into :reg_id, k_disb_date, kf_id;

    if (db_id in (39801, 41701)) then
    begin
      --Для КР кредиты проходят только пост-проверку
      execute procedure hp_actionlist_insert(null, :reg_id, :object_kind_str, :object_id, :action_code, 0, :un_id)
          returning_values :acl_id;

      select kal.kal_new_kf_officer
      from knaktlist kal
      where kal.kal_from_kf_id = :kf_id and
            kal.is_delete = 0 and
            :k_disb_date between kal.kal_date and kal.kal_date_end
      into lo_id;

      --RFCKG-2462
      if (lo_id = 1090101820206 or un_id = 1090100006287) then
        exit;

      update actionlist acl
      set acl.acl_status = 2,
          acl.acl_from_un_id = 1000000000000,
          acl.acl_respons_c_id = 1000000000000
      where acl.acl_id = :acl_id;
      suspend;
      exit;
    end
  end

  execute procedure hp_actionlist_insert(null, :reg_id, :object_kind_str, :object_id, :action_code, 0, :un_id)
      returning_values :acl_id;

  if (action_code = 'ActionGroup.Check_Loan_Data' or action_code = 'ActionGroup.Check_Loan_Data_Special') then
  begin
    -- если это проверка кредита, добавляем клиентов тоже на проверку
    select rd_id
    from refdata
    where rd_code = 'ActionGroup.Check_Client_Data'
    into :acl_from_rd_id;

    for select acl_id, acl_object_id
        from actionlist
        where acl_object_kind = 0 and
              acl_from_rd_id = :acl_from_rd_id and
              acl_parent_id = :acl_id
        into :tmp_acl_id, :c_id
    do
    begin
      if (not(exists(select *
                     from knclient
                     where kc_from_k_id = :object_id and
                           kc_amount_approved > 0 and
                           kc_from_c_id = :c_id))) then
      begin
        delete from actionlist
        where acl_id = :tmp_acl_id;
      end
    end

    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :object_id and
              kc_amount_approved > 0
        union
        select z_from_c_id
        from zalog
        where z_from_k_id = :object_id and
              z_type = 0 and
              :db_id = 94401 -- поручитель (RFCAZ-175)
        into :c_id
    do
    begin
      tmp_acl_id = null;
      select acl_id
      from actionlist
      where acl_object_kind = 0 and
            acl_object_id = :c_id and
            acl_from_rd_id = :acl_from_rd_id and
            acl_parent_id = :acl_id
      order by acl_id desc
      rows 1
      into :tmp_acl_id;

      if (tmp_acl_id is null) then
      begin
        execute procedure hp_actionlist_insert(:acl_id, :reg_id, 'Clients', :c_id, 'ActionGroup.Check_Client_Data', 0,
            :un_id)
            returning_values :tmp_acl_id;
      end
      else
      begin
        update actionlist
        set acl_status = 0,
            acl_from_un_id = :un_id
        where acl_id = :tmp_acl_id and
              coalesce(acl_status, -777) <> 0;
      end
    end
  end

  suspend;
end
