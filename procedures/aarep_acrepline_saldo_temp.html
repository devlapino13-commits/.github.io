CREATE PROCEDURE AAREP_ACREPLINE_SALDO_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    ACC_BALANCE VARCHAR(255),
    ACC_NUMBER INTEGER,
    ACC_SUMMA DECIMAL(18,4))
AS
declare variable al_id varchar(20);
declare variable acc_balance_prior varchar(20);
declare variable acc_number_prior integer;
declare variable acc_balance_prior2 varchar(20);
declare variable acc_number_prior2 integer;
declare variable summa_one_ekv decimal(18,4);
declare variable suspended smallint;
begin
  acc_balance_prior = '';
  acc_number_prior = -1;
  acc_summa = 0;
  for select al_id, al_from_bg_id, al_number
      from acc_list, region_list(:report_region_id, 1, 3, 1)
      where al_from_reg_id = id_list
      order by al_from_bg_id, al_number
      into :al_id, :acc_balance, acc_number
  do
  begin
    summa_one_ekv = f_al_bal(al_id, report_date_end);

    if (((acc_balance_prior <> acc_balance) or (acc_number_prior <> acc_number)) and
        (acc_number_prior > -1)) then
    begin
      acc_balance_prior2 = acc_balance;
      acc_number_prior2 = acc_number;
      acc_balance = acc_balance_prior;
      acc_number = acc_number_prior;
      suspended = 1;
      if (acc_summa <> 0) then
        suspend;
      acc_balance = acc_balance_prior2;
      acc_number = acc_number_prior2;
      acc_summa = summa_one_ekv;
    end
    else
    begin
      acc_summa = acc_summa + summa_one_ekv;
      suspended = 0;
    end

    acc_balance_prior = acc_balance;
    acc_number_prior = acc_number;
  end
  if ((suspended = 0) and
      (acc_summa <> 0)) then
    suspend;
end
