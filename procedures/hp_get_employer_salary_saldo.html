CREATE PROCEDURE HP_GET_EMPLOYER_SALARY_SALDO(
    REP_C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    REP_SALARY_SALDO DECIMAL(18,4))
AS
begin
  rep_salary_saldo = 0;
  select oss_saldo
  from opersalsaldo
  where oss_from_c_id = :rep_c_id and
        oss_date_begin <= :rep_date and
        oss_date_end >= :rep_date
  order by oss_from_c_id desc, oss_date_begin desc
  rows 1
  into :rep_salary_saldo;
  if (rep_salary_saldo is null) then
    rep_salary_saldo = 0;
  suspend;
end
