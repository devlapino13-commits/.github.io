CREATE PROCEDURE HP_PANTERA_IGNORE_DAYS_LIMIT(
    REP_DATE DATE,
    DAYS INTEGER)
AS
declare variable PI_ID integer;
declare variable PI_RD_PROJECT_ID integer;
declare variable PI_OBJECT_KIND integer;
declare variable PI_OBJECT_ID ID_TYPE;
declare variable PI_SUBJECT_KIND integer;
begin
  for select pi.pi_id, pi.pi_rd_project_id, pi.pi_object_kind, pi.pi_object_id, pi.pi_subject_kind
      from pantera_ignore pi
      inner join refdata rd on rd.rd_id = pi.pi_reason_rd_id
      where pi.pi_date_begin <= :rep_date - :days and
            pi.pi_date_end = '31.12.9999' and
            pi.pi_is_active = 1 and
            rd.rd_code in ('OKTELL.SirenaResults.CellNoNotServed',
                           'OKTELL.SirenaResults.CantCalled',
                           'OKTELL.SirenaResults.CellNoTempUnavailable',
                           'OKTELL.SirenaResults.CalledToOneCantact')
      into :pi_id, :pi_rd_project_id, :pi_object_kind, :pi_object_id, :pi_subject_kind
  do
  begin
    /*
    -- если ранее абонента не было в игнор-листе по причинам недозвона, тогда  вывожу его, иначе он там остается
    if (not exists(select pi.pi_id
                   from pantera_ignore pi
                   inner join refdata rd on rd.rd_id = pi.pi_reason_rd_id
                   where pi.pi_id < :pi_id and
                         pi.pi_is_active = 1 and
                         pi.pi_rd_project_id = :pi_rd_project_id and
                         pi.pi_object_kind = :pi_object_kind and
                         pi.pi_object_id = :pi_object_id and
                         pi.pi_object_kind = :pi_subject_kind and
                         rd.rd_code in ('OKTELL.SirenaResults.CellNoNotServed', 'OKTELL.SirenaResults.CantCalled'))) then
    begin
    */
      update pantera_ignore pi
      set pi.pi_date_end = :rep_date
      where pi.pi_id = :pi_id;
    --end
  end
end
