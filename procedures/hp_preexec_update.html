CREATE PROCEDURE HP_PREEXEC_UPDATE
AS
declare variable SIRENA1_RD_ID D_INTEGER;
declare variable PANTERA_RD_ID D_INTEGER;
declare variable INV_AND_REC_RD_ID D_INTEGER;
declare variable KS2_RD_ID D_INTEGER;
declare variable K_ID D_ID;
declare variable PROJECT_RD_ID D_INTEGER;
declare variable PROJECT_CODE D_STR_100;
begin
    sirena1_rd_id = f_get_rd_id('OKTELL.Projects.Sirena1');
    pantera_rd_id = f_get_rd_id('OKTELL.Projects.Pantera');
    inv_and_rec_rd_id = f_get_rd_id('OKTELL.Projects.Investigation_And_Recovery');
    ks2_rd_id = f_get_rd_id('OKTELL.Projects.Ks_finance_2');

    delete from  pantera_preexec where 0 = 0;

    for select k_id, :sirena1_rd_id, ''
    from okt_sirena_list_tmp(current_date, 'OKTELL.Projects.Sirena1', '', '', '', 21, 60)
    union all
    select k_id, :pantera_rd_id, ''
    from okt_pantera_list_tmp(current_date, 'OKTELL.Projects.Pantera', '', '', 1, 9999)
    union all
    select k_id, :inv_and_rec_rd_id, ''
    from okt_pantera_list_tmp(current_date, 'OKTELL.Projects.Investigation_And_Recovery', '', '', 1, 9999)
    union all
    select k_id, :ks2_rd_id, 'OKTELL.Projects.Ks_finance_2'
    from okt_pantera_list_tmp(current_date, 'OKTELL.Projects.Ks_finance_2', '', '', 1, 9999)
    into :k_id, :project_rd_id, :project_code
    do
    begin
        if (project_code = 'OKTELL.Projects.Ks_finance_2') then
        begin
            if (exists(
                select *
                from pantera_kontracts pantk
                left join pantera_kontracts_com  pantkc on pantk.pant_id = pantkc.pkc_from_pant_id
                where pantk.pant_date_end = '30.12.9999' and
                      pantk.pant_oper_called = 'VOKSFINANCE' and
                      pantk.pant_from_k_id = :k_id)) then
                continue;
        end

        update or insert into pantera_preexec (ppe_from_k_id, ppe_project_rd_id)
        values (:k_id, :project_rd_id)
        matching (ppe_from_k_id);
    end
end
