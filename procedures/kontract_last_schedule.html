CREATE PROCEDURE KONTRACT_LAST_SCHEDULE(
    KONT DECIMAL(18,0))
RETURNS (
    S_ID INTEGER,
    S_DATE DATE,
    S_OS DECIMAL(18,4),
    S_PERC DECIMAL(18,4),
    S_ALL DECIMAL(18,4),
    V_OS DECIMAL(18,4),
    V_PERC DECIMAL(18,4),
    V_FINE DECIMAL(18,4),
    DISB_AMOUNT DECIMAL(18,4))
AS
declare variable r_os decimal(18,4);
declare variable r_perc decimal(18,4);
begin
  /* Procedure Text */
  s_os = 0;
  s_perc = 0;
  v_os = 0;
  v_perc = 0;
  v_fine = 0;
  s_all = 0;
  disb_amount = 0;

  select ok_os
  from operkred
  where ok_from_k_id = :kont and
        ok_type = 8 and
        ok_ol_is_active = 1 and
        ok_sh_id = 0 and
        ok_os <> 0 and
        ok_from_c_id = 0
  into :disb_amount;
  if (disb_amount is null) then
    disb_amount = 0;

  for select ok_sh_id, ok_sh_date, sum(ok_sh_os), sum(ok_sh_perc), sum(-1 * ok_os), sum(-1 * ok_perc),
             sum(-1 * ok_fine)
      from operkred
      where ok_from_k_id = :kont and
            ok_ol_is_active = 1 and
            ok_type < 7 and
            ok_from_c_id = 0
      group by ok_sh_id, ok_sh_date
      having(sum(ok_sh_os) + sum(-1 * ok_os) <> 0) or (sum(ok_sh_perc) + sum(-1 * ok_perc) <> 0)
      order by 1, 2
      into :s_id, :s_date, :s_os, :s_perc, v_os, v_perc, v_fine
  do
  begin
    s_all = s_os + s_perc;
    if (s_all is null) then
      s_all = 0;
    suspend;
  end

end
