CREATE PROCEDURE AAREP_ANALITICA_LO(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_PROJECT VARCHAR(100),
    REP_PERIOD_KIND INTEGER)
RETURNS (
    AA_LO_ID DECIMAL(18,0),
    AA_DISB_LO_FIO VARCHAR(100),
    AA_LO_OFFICE_NAME VARCHAR(100),
    AA_OWN_KOLICH_CL_ACTIVE_END INTEGER,
    AA_OWN_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_OWN_DELIQ_CL_KOLICH INTEGER,
    AA_OWN_DELIQ_BALANCE DECIMAL(18,4),
    AA_IN_KOLICH_CL_ACTIVE_END INTEGER,
    AA_IN_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_IN_DELIQ_CL_KOLICH INTEGER,
    AA_IN_DELIQ_BALANCE DECIMAL(18,4),
    AA_OUT_KOLICH_CL_ACTIVE_END INTEGER,
    AA_OUT_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_OUT_DELIQ_CL_KOLICH INTEGER,
    AA_OUT_DELIQ_BALANCE DECIMAL(18,4),
    AA_CURR_KOLICH_CL_ACTIVE_END INTEGER,
    AA_CURR_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_CURR_DELIQ_CL_KOLICH INTEGER,
    AA_CURR_DELIQ_BALANCE DECIMAL(18,4),
    AA_LO_JOBNAME VARCHAR(100),
    PS_IS_ON_STAFF_STR VARCHAR(100),
    PS_IS_WORKING_STR VARCHAR(100),
    AA_LO_BRANCH_NAME VARCHAR(100),
    AA_LO_LEVEL_NAME VARCHAR(100),
    AA_LO_LEVEL2_NAME VARCHAR(100),
    AA_LO_LEVEL3_NAME VARCHAR(100),
    AA_LO_STAGE_DATE DATE,
    AA_LO_STAGE DECIMAL(18,4))
AS
begin
  for select aa_lo_id, aa_disb_lo_fio, aa_lo_office_name, sum(aa_own_amount_active_end),
             sum(aa_own_kolich_cl_active_end), sum(aa_own_deliq_cl_kolich), sum(aa_own_deliq_balance),
             sum(aa_in_amount_active_end), sum(aa_in_kolich_cl_active_end), sum(aa_in_deliq_cl_kolich),
             sum(aa_in_deliq_balance), sum(aa_out_amount_active_end), sum(aa_out_kolich_cl_active_end),
             sum(aa_out_deliq_cl_kolich), sum(aa_out_deliq_balance), sum(aa_curr_amount_active_end),
             sum(aa_curr_kolich_cl_active_end), sum(aa_curr_deliq_cl_kolich), sum(aa_curr_deliq_balance), aa_lo_jobname,
             ps_is_on_staff_str, ps_is_working_str, aa_lo_branch_name, aa_lo_level_name, aa_lo_level2_name,
             aa_lo_level3_name
      from aarep_analitica_lo_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_project, :rep_period_kind)
      group by aa_lo_id, aa_disb_lo_fio, aa_lo_office_name, aa_lo_jobname, ps_is_on_staff_str, ps_is_working_str, aa_lo_branch_name, aa_lo_level_name, aa_lo_level2_name, aa_lo_level3_name
      into :aa_lo_id, :aa_disb_lo_fio, :aa_lo_office_name, :aa_own_amount_active_end, :aa_own_kolich_cl_active_end,
           :aa_own_deliq_cl_kolich, :aa_own_deliq_balance, :aa_in_amount_active_end, :aa_in_kolich_cl_active_end,
           :aa_in_deliq_cl_kolich, :aa_in_deliq_balance, :aa_out_amount_active_end, :aa_out_kolich_cl_active_end,
           :aa_out_deliq_cl_kolich, :aa_out_deliq_balance, :aa_curr_amount_active_end, :aa_curr_kolich_cl_active_end,
           :aa_curr_deliq_cl_kolich, :aa_curr_deliq_balance, :aa_lo_jobname, :ps_is_on_staff_str, :ps_is_working_str,
           :aa_lo_branch_name, :aa_lo_level_name, :aa_lo_level2_name, :aa_lo_level3_name
  do
  begin
    select min(ks_date_begin)
    from kontstate
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_kc_id = :aa_lo_id
    into :aa_lo_stage_date;
    if (aa_lo_stage_date is null) then
      aa_lo_stage_date = rep_date_end;
    aa_lo_stage = datediff(day from aa_lo_stage_date to :rep_date_end) / 30;

    suspend;
  end

end
