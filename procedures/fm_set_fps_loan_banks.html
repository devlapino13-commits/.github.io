CREATE PROCEDURE FM_SET_FPS_LOAN_BANKS(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    FPB_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable clt_id d_integer;
declare variable clt_code d_str_code;
declare variable fcb_id d_integer;
declare variable obj_code d_str_code;
declare variable c_id d_id;
declare variable k_status d_smallint;
declare variable c_status d_smallint;
declare variable k_id d_id;
declare variable need_name d_bool;
declare variable check_count d_integer;
declare variable rd_id d_integer;
begin
  res_is_error = 0;
  res_text = 'Банк успешно изменен';

  obj_code = f_objkind(obj_kind, 'code');
  if (obj_code not in ('knclient', 'miniappclient')) then
    exception e_default f_format('Incorrect obj_kind. KnClient/ miniapp expected, actually: %1', obj_code);

  if (obj_code = 'knclient') then
  begin
    select k_id, kc_from_c_id, k_status, c_verify_status
    from knclient, kontract, clients
    where kc_id = :obj_id and
      kc_from_k_id = k_id and
      kc_from_c_id = c_id
    into k_id, c_id, k_status, c_status;

    if (k_status = 1) then
    begin
      res_is_error = 1;
      res_text = 'Договор уже утвержден. Необходимо вернуть в процесс.';
      suspend;
      exit;
    end


    select rd_id
    from hp_get_check_loan_data(:k_id)
    into:rd_id;

    if (exists(
      select *
      from actionlist
      where acl_object_kind = 9
        and acl_object_id = :k_id
        and acl_status = 1
        and acl_from_rd_id = :rd_id
      )) then
    begin
      res_is_error = 1;
      res_text = 'Договор находится на рассмотрении КК. Изменения запрещены.';
      suspend;
      exit;
    end
  end
  else if (obj_code = 'miniappclient') then
    select c.c_id, c.c_verify_status
    from miniappclient mac
    inner join clients c on mac.mac_from_c_id = c.c_id
    where mac.mac_id = :obj_id
    into c_id, c_status;

  clt_code = f_cont(c_id, 'cell', 'ok', 'main');
  if (clt_code = '') then
  begin
    res_is_error = 1;
    res_text = 'Подтвержденный основной сотовый номер отсутствует';
    suspend;
    exit;
  end

  if (c_status <> 2) then
  begin
    res_is_error = 1;
    res_text = 'Клиент не верифицирован. Необходимо дождаться успешной верификации.';
    suspend;
    exit;
  end

  clt_id = f_cont_id(clt_code, 'error');

  if (fpb_id is null) then
  begin
    res_is_error = 1;
    res_text = 'Банк не выбран. Выберите банк из списка.';
    suspend;
    exit;
  end

  -- Далее банк будет отправлен на проверку, для случая fcb_status = -1

  select fcb_id
  from fps_client_banks
  where fcb_from_c_id = :c_id and
        fcb_from_clt_id = :clt_id and
        fcb_from_fpb_id = :fpb_id
  into fcb_id;

  if (fcb_id is null) then
  begin
    insert into fps_client_banks (fcb_from_c_id, fcb_from_clt_id, fcb_from_fpb_id, fcb_from_un_id)
    values (:c_id, :clt_id, :fpb_id, :un_id)
    returning fcb_id
    into fcb_id;
  end
  else if (f_get_const('fps_manual_disb') <> 1) then
  begin
    -- При необходимости клиент будет отправлен в банк на проверку
    -- Контрольно проверяем превышение лимита проверок в сутки
    -- За последние сутки не должно быть более 5 проверок
    check_count = 0;
    select count(*)
    from fps_client_banks_checks
    where fcc_from_fcb_id = :fcb_id
      and datediff(hour, fcc_datetime, current_timestamp) <= 24
    into check_count;

    if (check_count >= cast(f_get_const('max_disb_attempts_number') as d_integer)) then  -- RFCRU-4714
    begin
      res_is_error = 1;
      res_text = 'Превышен лимит проверок клиента в СБП. Клиент не найден. Выдача через СБП невозможна. Используйте другие способы выдачи займа.';
      suspend;
      exit;
    end

    -- Более двух суток данные о проверках не храним
    delete from fps_client_banks_checks
    where fcc_from_fcb_id = :fcb_id
      and fcc_datetime < current_timestamp - 2;

    -- Контрольно отправляем на проверку, если не проверялся
    -- Или если нет подтвержденного имени в fps_client_names (раньше при автоподтверждении оно не добавлялось)
    need_name = 0;

    if (not exists(
        select * from fps_client_names
        where fcn_from_fcb_id = :fcb_id
          and fcn_status = 2)) then
      need_name = 1;

    update fps_client_banks
    set fcb_status = 0
    where fcb_id = :fcb_id
      and (fcb_status < 0 or :need_name = 1);
  end

  if (obj_code = 'knclient') then
    update or insert into fps_loan_banks (flb_from_kc_id, fbl_from_fcb_id)
    values (:obj_id, :fcb_id)
    matching (flb_from_kc_id);

  else if (obj_code = 'miniappclient') then
    update or insert into fps_application_banks (fab_from_mac_id, fab_from_fcb_id)
    values (:obj_id, :fcb_id)
    matching (fab_from_mac_id);

  suspend;
end
