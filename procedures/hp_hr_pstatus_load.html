CREATE PROCEDURE HP_HR_PSTATUS_LOAD(
    IN_ORDER_ID D_ID,
    IN_C_ID D_ID,
    IN_ORDER_TYPE D_INTEGER,
    UN_ID D_ID)
RETURNS (
    HO_ID D_ID,
    HO_DATE D_DATE,
    HO_FROM_REG_ID D_INTEGER,
    HO_TEXT D_STR_LONG,
    HO_REASON D_STR_255,
    HO_SAVE_TIME D_DATETIME,
    HO_ORDER_TYPE D_INTEGER,
    HO_ORDER_SUBTYPE D_INTEGER,
    HO_ORDER_NO D_STR_20,
    HO_STATUS D_INTEGER,
    HO_COME_BACK_DATE D_DATE,
    HO_CAUSE_ID D_INTEGER,
    HO_SIGNER_ID D_INTEGER,
    HO_FROM_EDL_ID D_ID,
    PS_ID D_INTEGER,
    PS_FROM_HO_ID ID_TYPE,
    PS_DATE_BEGIN D_DATE,
    PS_DATE_END D_DATE,
    PS_PERIOD_MONTH D_INTEGER,
    PS_DATE_END_IN_ORDER D_DATE,
    PS_FROM_REG_ID D_INTEGER,
    PS_FROM_DEP_ID D_INTEGER,
    PS_FROM_JL_ID D_INTEGER,
    PS_IS_ON_STAFF D_SMALLINT,
    PS_IS_WORKING D_SMALLINT,
    PS_FROM_C_ID ID_TYPE,
    PS_SALARY CURR_TYPE,
    PS_TRANSFER_OFFICE D_INTEGER,
    PS_TABEL_NO D_STR_10,
    PS_CLASS D_INTEGER,
    PS_STATUS D_INTEGER,
    PS_IS_MOUNT D_BOOL,
    PS_IS_PART D_BOOL,
    PS_IS_TIMELESS D_BOOL,
    PS_MOUNT_KOEF D_CURRENCY,
    PS_IS_WORK_BOOK D_BOOL,
    RES_IS_EDIT D_BOOL,
    RES_WORK_KIND D_SMALLINT,
    PSA_AGREEMENT_NO D_STR_10,
    HM_MOUNT D_INTEGER,
    RES_FULL_INFO D_STR_LONG,
    PS_NOTIFIC_NO D_STR_CODE,
    PS_NOTIFIC_ADDRESS D_INTEGER,
    HO_TRANSFER_OFFICE D_INTEGER)
AS
declare variable ps_id_tmp d_integer;
declare variable ho_id_tmp d_id;
declare variable head_office d_integer;
begin
  res_is_edit = 0;
  hm_mount = 0;
  res_full_info = '';
  if (coalesce(in_order_id, 0) <> 0) then
  begin
    for select ho_id,
               ho_date,
               ho_from_reg_id,
               ho_text,
               ho_reason,
               ho_save_time,
               ho_order_type,
               ho_order_subtype,
               ho_order_no,
               ho_status,
               ho_come_back_date,
               ho_cause_id,
               ho_signer_id,
               ho_from_edl_id,
               ps_id, 
               ps_from_ho_id, 
               ps_date_begin, 
               ps_date_end, 
               ps_period_month, 
               ps_date_end_in_order, 
               ps_from_reg_id, 
               ps_from_dep_id, 
               ps_from_jl_id, 
               ps_is_on_staff, 
               ps_is_working, 
               ps_from_c_id, 
               ps_salary, 
               ps_transfer_office, 
               ps_tabel_no, 
               ps_class, 
               ps_status,
               ps_is_mount,
               ps_is_part,
               ps_is_timeless,
               ps_mount_koef,
               ps_is_work_book,
               ps_notific_no,  -- RFCKG-4655
               ps_notific_address
        from hr_orders
        inner join hr_pstatus on ps_from_ho_id = ho_id
        where ho_id = :in_order_id
        into :ho_id,
             :ho_date,
             :ho_from_reg_id,
             :ho_text,
             :ho_reason,
             :ho_save_time,
             :ho_order_type,
             :ho_order_subtype,
             :ho_order_no,
             :ho_status,
             :ho_come_back_date,
             :ho_cause_id,
             :ho_signer_id,
             :ho_from_edl_id,
             :ps_id, 
             :ps_from_ho_id, 
             :ps_date_begin, 
             :ps_date_end, 
             :ps_period_month, 
             :ps_date_end_in_order, 
             :ps_from_reg_id, 
             :ps_from_dep_id, 
             :ps_from_jl_id, 
             :ps_is_on_staff, 
             :ps_is_working, 
             :ps_from_c_id, 
             :ps_salary, 
             :ps_transfer_office, 
             :ps_tabel_no, 
             :ps_class, 
             :ps_status,
             :ps_is_mount,
             :ps_is_part,
             :ps_is_timeless,
             :ps_mount_koef,
             :ps_is_work_book,
             :ps_notific_no,
             :ps_notific_address
    do
    begin
      res_is_edit = 1;
      hm_mount = 0;
      res_full_info = '';

      if (ho_order_type = 6) then
      begin
        ps_date_begin = ps_date_begin - 1;
        ps_date_end_in_order = ps_date_begin;
      end

      res_work_kind = ps_is_on_staff;
      if (ps_is_part = 1) then
        res_work_kind = 2;

      psa_agreement_no = '';

      select psa_agreement_no
      from hr_pstatus_agreement
      where psa_from_ps_id = :ps_id and
            psa_from_ho_id = :ho_id
      into psa_agreement_no;

      select hm_mount
      from hr_mount
      where hm_from_ps_id = :ps_id and
            hm_ps_from_ho_id = :ho_id
      into hm_mount;

      if (row_count = 0) then
      begin
        select hm.hm_mount
        from hr_pstatus ps
        inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id and
              ho.ho_order_type = 1 and
              ho.ho_status = 1
        left join hr_mount hm on hm.hm_ps_from_ho_id = ps.ps_from_ho_id and
              hm.hm_from_ps_id = ps.ps_id
        where ps.ps_from_c_id = :ps_from_c_id and
              ps.ps_status = 1 and
              ps.ps_date_begin <= :ho_date
        order by ps.ps_date_begin desc
        rows 1
        into :hm_mount;
      end

      select res_full_info
      from hp_e_hr_orders_pers_info(:ps_from_c_id, :ho_date)
      into res_full_info;

      suspend;
    end

    if (row_count = 0) then
      exception e_message 'Не найден приказ по Id: ' || in_order_id;
  end else
  begin
    select un_curr_reg_id
    from username
    where un_id = :un_id
    into ho_from_reg_id;

    if (coalesce(in_order_type, 0) > 0) then
      ho_order_type = in_order_type;
    else
      ho_order_type = 1;

    ho_order_subtype = 0;
    ps_mount_koef = 0;

    if (coalesce(in_c_id, 0) <> 0) then
    begin
      select ps_id, ps_from_ho_id, ps_from_reg_id, ps_from_dep_id, ps_from_jl_id, ps_is_mount, ps_tabel_no,
             ps_from_reg_id as ps_transfer_office, ps_mount_koef,
             iif(ps_is_part = 1, 2, ps_is_on_staff) as res_work_kind
      from hr_pstatus
      where ps_from_c_id = :in_c_id and
            ps_date_begin <= current_date and
            ps_is_working = 1 and
            ps_date_end >= ps_date_begin
      order by ps_date_begin desc, ps_from_ho_id desc
      rows 1
      into ps_id_tmp, ho_id_tmp, ps_from_reg_id, ps_from_dep_id, ps_from_jl_id, ps_is_mount, ps_tabel_no,
           ps_transfer_office, ps_mount_koef,
           res_work_kind;

      if (ps_id_tmp is not null) then
      begin
        psa_agreement_no = '';
        ho_from_reg_id = ps_from_reg_id;
  
        select psa_agreement_no
        from hr_pstatus_agreement
        where psa_from_ps_id = :ps_id_tmp and
              psa_from_ho_id = :ho_id_tmp
        into psa_agreement_no;

        select hm.hm_mount
        from hr_pstatus ps
        inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id and
              ho.ho_order_type = 1 and
              ho.ho_status = 1
        left join hr_mount hm on hm.hm_ps_from_ho_id = ps.ps_from_ho_id and
              hm.hm_from_ps_id = ps.ps_id
        where ps.ps_from_c_id = :in_c_id and
              ps.ps_status = 1 and
              ps.ps_date_begin <= current_date
        order by ps.ps_date_begin desc
        rows 1
        into :hm_mount;
      end

      select res_full_info
      from hp_e_hr_orders_pers_info(:in_c_id, current_date)
      into res_full_info;
    end

    ho_date = current_date;
    ps_date_begin = current_date;
    ps_period_month = 1000;
    ps_date_end = cast('31.12.9999' as date);
    ps_is_mount = 0;
    ps_is_work_book = 0;
    ps_is_part = 0;
    ps_is_timeless = 1;
    ps_is_on_staff = 1;
    ps_is_working = 1;
    ps_from_c_id = :in_c_id;

    res_work_kind = ps_is_on_staff;

    if (ho_signer_id is null) then
    begin
      head_office = f_hq_id();

      if (ho_from_reg_id = head_office) then
      begin
        select ho_signer_id
        from hr_orders
        where ho_signer_id is not null and
              ho_from_reg_id = :ho_from_reg_id
        order by ho_id desc
        rows 1
        into ho_signer_id;
      end else
      begin
        select ho_signer_id
        from hr_orders
        where ho_signer_id is not null and
              ho_from_reg_id <> :ho_from_reg_id
        order by ho_id desc
        rows 1
        into ho_signer_id;
      end
    end

    suspend;
  end
end
