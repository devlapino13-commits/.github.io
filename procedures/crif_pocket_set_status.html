CREATE PROCEDURE CRIF_POCKET_SET_STATUS(
    POCKET_ID D_INTEGER,
    POCKET_STATUS STR_100)
RETURNS (
    POCKET_INTERNAL_STATUS INTEGER)
AS
begin
  pocket_internal_status = (case pocket_status
                              when 'New' then 0  -- пакет у сервиса Антона
                              when 'Compiled'  then 1
                              when 'Compiling' then 1
                              when 'Sending' then 1
                              when 'Sent' then 1 -- пакет отправлен
                              when 'ResultReceivedOK' then 2 -- принят, и в пакете нет ошибок
                              when 'ResultReceivedAcceptedWithErrors' then 2
                              -- негативные статусы -->
                              when 'SendingFailed' then -1
                              when 'ResultReceivedDeclinedPre' then - 1
                              when 'ResultReceivedDeclinedLowQuality' then -1
                              when 'Empty' then -1
                              ------------------------<
                              else -1
                            end);
  update cib_pocket cp
  set cp_pocket_status = :pocket_internal_status,
      cp.cp_pocket_kib_status = :pocket_status,
      cp.cp_status_datetime = current_timestamp
  where cp.cp_id = :pocket_id;
end
