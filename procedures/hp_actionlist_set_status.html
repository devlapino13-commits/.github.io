CREATE PROCEDURE HP_ACTIONLIST_SET_STATUS(
    OBJECT_KIND INTEGER,
    OBJECT_ID DECIMAL(18,0),
    RD_ID_MAIN INTEGER,
    RD_ID_DETAIL INTEGER,
    NEW_STATUS DECIMAL(18,4),
    NEW_MEMO D_STR_8100,
    UN_ID D_ID,
    RD_ID_REASON D_INTEGER = null,
    PROOF_INCOME D_INTEGER = 0)
AS
declare variable ACL_ID decimal(18,0);
declare variable ROUND_STATUS integer;
declare variable DB_ID D_INTEGER;
declare variable EDL_ID D_ID;
declare variable DATE_SE D_DATE;
declare variable PR_CODE D_STR_100;
declare variable KC_ID D_ID;
declare variable MAC_ID D_ID;
declare variable IS_PROOF boolean;
declare variable ERR_TXT D_STR_255;
declare variable ERR_CODE D_INTEGER;
declare variable C_ID D_ID;
begin
  db_id = f_get_db_id();

  -- устанавливаем значение (состояние) ActionList
  round_status = round(new_status);
  rd_id_reason = coalesce(rd_id_reason, 0);

  if (new_memo is null) then
    new_memo = ''; -- Комментарии при смене статуса

  if (rd_id_main = rd_id_detail) then
  begin
    select acl_id
    from actionlist
    where acl_object_kind = :object_kind and
          acl_object_id = :object_id and
          acl_from_rd_id = :rd_id_main
    order by acl_id desc
    rows 1
    into :acl_id;

    if (acl_id is not null) then
    begin
      -- RFCRU-5572 Проверка кода регистрации перед подтверждением КК
      if (db_id = 81001 and
          object_kind = 9 and
          round_status = 2) then
      begin
        c_id = null;

        select kc.kc_from_c_id
        from knclient kc
        where kc.kc_from_k_id = :object_id and
              kc.kc_amount_approved > 0
        rows 1
        into :c_id;

        if (c_id is not null) then
        begin
          if (not exists(
            select *
            from clientdocs
              inner join addresstype_registration
                on atr_from_cds_id = cds_id
            where cds_from_c_id = :c_id
              and cds_is_delete = 0))
          then
            exception e_message 'не выбран код регистрации - подтверждение запрещено';
        end
      end

      if (exists(select *
                 from refdata
                 where rd_id = :rd_id_main and
                       rd_code starting 'ActionGroup.Passport_Verify') and
          :round_status = 1) then
      begin
        update actionlist
        set acl_status = :round_status
        where acl_id = :acl_id;
      end
      else
      begin
        update actionlist
        set acl_status = :round_status,
            acl_from_un_id = :un_id
        where acl_id = :acl_id;

        if (db_id = 81001 and object_kind = 9) then
        begin
          pr_code = (select pr_code
                     from kontract
                     join product on pr_id = k_from_pr_id
                     where k_id = :object_id);

          -- ПДН - расчет от документально подтвержденного дохода
          -- RFCRU-2289
          if (f_get_const('documented_income_active') = '1') then
          begin
            is_proof = (new_status = 2) and (proof_income = 1);

            for select kc_id,
                       mac_id
            from kontract
            join knclient on kc_from_k_id = k_id and kc_amount_approved > 0
            join miniappclient on mac_from_ma_id = k_from_ma_app_id and mac_from_c_id = kc_from_c_id
            where k_id = :object_id
            into :kc_id,
                 :mac_id
            do
            begin
              if ((new_status = 2) or
                 ((new_status = -1) and
                 exists(select *
                        from verifycomment
                        where vcm_code = '24'
                          and :new_memo containing vcm_caption))) then
              begin
                if (is_proof) then
                begin
                  execute procedure knclient_pdn_limit_check(:kc_id)
                    returning_values :err_txt, :err_code;
                  if (coalesce(err_txt, '') <> '') then
                    exception e_message err_txt;
                end

                update or insert into miniappclient_income (
                  maci_from_mac_id,
                  maci_is_documented,
                  maci_datetime,
                  maci_from_un_id,
                  maci_is_blocked)
                values (
                  :mac_id,
                  iif(:is_proof, 1, 0),
                  current_timestamp,
                  :un_id,
                  1)
                matching (maci_from_mac_id);
              end
              else if (:new_status = -1 and
                      exists(select*
                             from miniappclient_income maci
                             where maci.maci_from_mac_id = :mac_id
                               and maci.maci_is_documented = 1)) then
              begin
                if (exists(select *
                          from verifycomment
                          where vcm_code = '25'
                           and :new_memo containing vcm_caption)) then
                  execute procedure pdn_save_json(:mac_id, :un_id, null, 'send_to_revision');
                else
                begin
                  update miniappclient_income maci
                  set maci.maci_is_blocked = 1
                  where maci.maci_from_mac_id = :mac_id;
                end
              end
            end
          end

          -- Доработка по самозанятым
          if (pr_code containing '.selfemployed.') then
          begin
            -- RFCRU-2544
            if ((new_status = -1) and exists(
              select *
              from verifycomment
              where vcm_code = '7.11'
                and :new_memo containing vcm_caption) and (f_tmp_const('use_perstemp') <> 1)) then
            begin
              date_se = f_get_const('selfemployed_start_date');
              for
                select
                  edl_id
                from kontract
                  inner join knclient
                    on kc_from_k_id = k_id
                      and kc_amount_approved > 0
                  inner join ft_anketa_list
                    on fal_from_c_id = kc_from_c_id
                      and fal_date >= :date_se
                  inner join edoc_list
                    on edl_object_kind = f_objkind('ft_anketa_list', 'id')
                      and edl_object_id = fal_id
                      and edl_status = 1
                where k_id = :object_id
              into
                :edl_id
              do
              begin
                update edoc_list set
                  edl_status = -1
                where edl_id = :edl_id;
              end
            end
            -- RFCRU-2562
            if ((new_status = 2) and exists(
              select *
              from knclient
                left join kk_selfemployed
                  on kse_from_acl_id = :acl_id
                    and kse_from_k_id = kc_from_k_id
                    and kse_from_c_id = kc_from_c_id
              where kc_from_k_id = :object_id
                and kc_amount_approved > 0
                and kse_id is null)) then
              exception e_message('Предварительно не заполнена инфа по самозанятому.');
          end
        end
        --

      end
    end
    else
    begin
      execute procedure hp_objectprops_set(:object_kind, :object_id, -1,
          :rd_id_detail, current_date, :round_status, :round_status, :un_id);
    end
  end
  else
  begin
    execute procedure hp_objectprops_set(:object_kind, :object_id, -1,
        :rd_id_detail, current_date, :round_status, :round_status, :un_id);
  end
  if (new_memo <> '') then
  begin
    execute procedure hp_objectprops_memo_set(:object_kind, :object_id, -1,
        :rd_id_detail, :new_memo, :un_id);
  end
end
