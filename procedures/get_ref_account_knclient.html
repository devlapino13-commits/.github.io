CREATE PROCEDURE GET_REF_ACCOUNT_KNCLIENT(
    SAB_FULL_NAME D_STR_255,
    SAB_INVERT_NAME D_STR_255,
    KC_ID ID_TYPE,
    KC_NON_RESIDENT SMALLINT = 0,
    KC_CLIENT_TYPE SMALLINT = 0,
    REP_DATE DATE = null,
    REG_ID INTEGER = null)
RETURNS (
    ACC_ID D_ID)
AS
declare variable kf_id id_type;
declare variable k_disb_date date;
begin
  if (kc_id is null) then
    exception e_default 'KnClient ID is emptry';

  if (rep_date is null) then
    rep_date = current_date;

  if (reg_id is null) then
  begin
    select k_from_kf_id, k_disb_date, iif(:kc_client_type = 0, kc_client_type, :kc_client_type)
    from kontract, knclient
    where kc_id = :kc_id and
          kc_from_k_id = k_id
    into :kf_id, :k_disb_date, :kc_client_type;

    select kal_new_reg_id
    from knaktlist
    where kal_from_kf_id = :kf_id and
          kal_date <= :k_disb_date and
          kal_date_end >= :k_disb_date and
          is_delete = 0
    into :reg_id;

    if (coalesce(reg_id, 0) = 0) then
    begin
      reg_id = f_hq_id();
    end
  end

  select acc_id
  from get_ref_account_any(:sab_full_name, :sab_invert_name, 'knclient', :kc_id, '', :reg_id, -1, :kc_non_resident, :kc_client_type, :rep_date)
  into :acc_id;
  suspend;
end
