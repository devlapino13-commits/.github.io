CREATE PROCEDURE HP_FIND_PERSON_SEND_EDOC(
    EDOC_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID,
    SIGNER_C_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    EDL_ID D_ID)
AS
declare variable signer_is_needed d_smallint;
begin
  /*т.к. процедура рассчитана на отправку эл документов, за исключение ЭЦП из формы поиска клиентов, то obj_kind = 0*/
  /*Процедура, в которой делаются пред проверки*/
  signer_is_needed = 0;
  res_text = '';

  select res_text, signer_c_id
  from hp_edoc_agreements_check(:edoc_kind, :obj_id, :un_id)
  into :res_text, :signer_c_id;

  if (signer_is_needed = 1 and
      signer_c_id = 0) then
  begin
    res_text = 'Необходимо указать подписанта';
    suspend;
    exit;
  end

  if (res_text = '') then
  begin
    /*Процедура отправки СМС*/
    select edl_id, res_text
    from hp_edoc_insert(:edoc_kind, 0, :obj_id, :un_id)
    into :edl_id, :res_text;

    /*Если необходимо, апдейтим подписанта*/

    if (:edl_id is null) then
      edl_id = 0;

    if (res_text = '' and
        edl_id > 0 and
        signer_is_needed = 1) then
      update edoc_list
      set edl_manager_c_id = :signer_c_id
      where edl_id = :edl_id;
  end
  suspend;
end
