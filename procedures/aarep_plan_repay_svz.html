CREATE PROCEDURE AAREP_PLAN_REPAY_SVZ(
    REP_C_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    KF_NAME VARCHAR(30),
    CASHIER_NAME VARCHAR(100),
    OK_OS CURR_TYPE,
    OK_PERC CURR_TYPE,
    SUM_TOTAL CURR_TYPE,
    C_PLASTCARD_NO VARCHAR(30),
    STATUS VARCHAR(20),
    KF_OFFICER_NAME VARCHAR(100),
    OK_SH_DATE DATE)
AS
declare variable k_id id_type;
declare variable cashier_id id_type;
declare variable kf_id id_type;
begin
--  rep_c_id = 1010100001577;
--  select c_id from clients
--  where c_name_fio = :rep_c_name
--  into rep_c_id;
--  KF_OFFICER_NAME = (select c_name_fio from clients where c_id = :REP_C_ID);
  for select oks_from_k_id
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_kc_id = :rep_c_id and
            oks_close_state = 0
      into k_id
  do
  begin
    ok_os = null;
    ok_perc = null;
    for select ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_sh_date between :rep_date_end and :rep_date_end + 2 and
              ok_ol_date < :rep_date_end and
              ok_from_c_id = 0
        group by ok_sh_date
        into ok_sh_date, ok_os, ok_perc
    do
    begin
      if (ok_os is null) then
        ok_os = 0;
      if (ok_perc is null) then
        ok_perc = 0;

      sum_total = ok_os + ok_perc;

      if (sum_total > 0) then
      begin
        cashier_id = null;
        kf_id = null;
        select k_clients_cashier_id, k_from_kf_id
        from kontract
        where k_id = :k_id
        into cashier_id, kf_id;

        kf_name = null;
        select kal_new_kf_name
        from knaktlist
        where kal_from_kf_id = :kf_id and
              :rep_date_end between kal_date and kal_date_end and
              knaktlist.is_delete = 0
        into kf_name;

        select c_name_fio,
               (select cont_value
                from hp_contdata_get_cell_valid(0, :cashier_id))
        from clients
        where c_id = :cashier_id
        into cashier_name, c_plastcard_no;

--    Проверка статуса
        if ((select sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc)
             from operkred
             where ok_from_k_id = :k_id and
                   ok_ol_is_active = 1 and
                   ok_sh_date between :rep_date_end and :rep_date_end + 2 and
                   ok_ol_date <= :rep_date_end and
                   ok_from_c_id = 0) = 0) then
          status = 'Погашен';
        else
          status = '';

        suspend;
      end
    end
  end
end
