CREATE PROCEDURE CM_RU_0124_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable pr_code varchar(100);
declare variable c_id id_type;
declare variable kc_balance curr_type;
declare variable re_k_id id_type;
declare variable kol_parallel integer;
declare variable re_pr_code varchar(100);
declare variable c_curr_status integer;
declare variable re_disb_date date;
declare variable kol_vyplat integer;
begin
  for
  select pr_code, k_disb_date
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_product_kind = 2
  and (pr_code not containing '.shans plus.' and pr_code not containing '.status_plus.')
  into pr_code, kf_disb_date
  do
  begin
    if (kf_disb_date >= '01.11.2020') then -- igusev 30.10.20 блок отключаем т.к. вместо него начинает работать блок 173. Созданный Фатимой
        exit;
    for
    select kc_from_c_id, kc_amount_approved
    from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    into c_id, kc_amount
    do
    begin
      if (c_id = 1010100059428) then
        exit;
      select c_name_fio, c_pass_no
      from clients
      where c_id = :c_id
      into res_info_text, kf_client_pass_no;
      select c_curr_status
      from hp_get_client_status_str(:c_id, :kf_disb_date)
      into c_curr_status;
      if (c_curr_status < 3) then
      begin
        res_error_is_critical = 2;
        res_error_text = 'У клиента статус менее 3х звезд, не может получить параллельный кредит.';
        suspend;
        exit;
      end
      if (c_curr_status = 3 and kc_amount > 20000.0) then
      begin
        res_error_is_critical = 2;
        res_error_text = 'Статус клиента 3 звезды, сумма не должна быть более 20 тыс.';
        suspend;
      end 
      if (c_curr_status = 4 and kc_amount > 30000.0) then
      begin
        res_error_is_critical = 2;
        res_error_text = 'Статус клиента 3 звезды, сумма не должна быть более 30 тыс.';
        suspend;
      end 
      if (c_curr_status = 5 and kc_amount > 40000.0) then
      begin
        res_error_is_critical = 2;
        res_error_text = 'Статус клиента 5 звезды, сумма не должна быть более 40 тыс.';
        suspend;
      end
--    вычисляем всех активных кредитов данного клиента
      kc_balance = null;
      kol_parallel = null;
      select sum(cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance),
             sum(iif(pr_product_kind = 2,1,0))
      from knclient, oksummary, kontract, product
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id = oks_from_k_id
      and kc_from_k_id = k_id
      and :kf_disb_date between oks_date_begin and oks_date_end
      and oks_balance > 0
      and oks_close_state = 0
      and k_from_pr_id = pr_id
      into kc_balance, kol_parallel;
      kc_balance = coalesce(kc_balance,0);
      kol_parallel = coalesce(kol_parallel,0);
      if (kol_parallel >= 1) then
      begin
        res_error_text = 'Уже имеется активный параллельный кредит!';
        res_error_is_critical = 2;
        suspend;
        exit;
      end 
      if (kc_balance = 0) then
      begin
        res_error_text = 'Не имеется основной активный кредит';
        res_error_is_critical = 2;
        suspend;
        exit;
      end 
      if (kc_balance + kc_amount > 90000.00) then
      begin
        res_error_text = 'Баланс активных кредитов + сумма параллельного кредита не должно превышать 90 тыс.';
        res_error_is_critical = 2;
        suspend;
      end
--    вычисляем основной кредит
      re_k_id = null;
      select ks_from_k_id, pr_code, ks_date_begin
      from kontstate, knclient, kontract, product
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin < :kf_disb_date
      and :kf_disb_date between ks_date_begin and ks_date_end
      and ks_from_k_id = k_id
      and k_from_pr_id = pr_id
      and pr_product_kind <> 2 
      and pr_code not containing '.food.'
      order by ks_date_begin desc
      rows 1
      into re_k_id, re_pr_code, re_disb_date;
      re_k_id = coalesce(re_k_id,0);
      if (re_k_id = 0) then
      begin
        res_error_text = 'Не имеется основной активный кредит';
        res_error_is_critical = 2;
        suspend;
        exit;
      end 
      if (pr_code containing '.patent plus.') then
      begin
        if (re_pr_code not containing '.patent plus.') then
        begin
          res_error_text = 'Для продукта "патент + параллельный" должен быть основной кредит "патент +".';
          res_error_is_critical = 2;
          suspend;
        end 
      end else
      begin
        if (re_pr_code not similar to '%(.shans.|.blago.|.uspeh.|.barakat|.family)%') then
        begin
          res_error_text = 'Не имеется основной активный кредит(Шанс, Благо или Успех)';
          res_error_is_critical = 2;
          suspend;
          exit;
        end 
      end
      if (kf_disb_date - re_disb_date < 65) then
      begin
        res_error_text = 'Параллельный кредит выдается ранее чем 65 дней от основного кредита.';
        res_error_is_critical = 2;
        suspend;
        exit;
      end 
      kol_vyplat = null;
      with tabl1 as (
          select ok_sh_date, sum(ok_os + ok_sh_os) zad_os, sum(ok_perc + ok_sh_perc) zad_perc
          from operkred
          where ok_from_k_id = :re_k_id
          and ok_ol_is_active = 1
          and ok_type < 7
          and ok_ol_date <= :kf_disb_date
          and ok_sh_id > 0
          group by ok_sh_date
          having sum(ok_os + ok_sh_os) = 0 and sum(ok_perc + ok_sh_perc) = 0
          )
      select count(*) from tabl1
      into kol_vyplat;
      kol_vyplat = coalesce(kol_vyplat,0);
      if (kol_vyplat < 2) then
      begin
        res_error_text = 'Еще не произведена 2 плановых платежа основного активного кредита';
        res_error_is_critical = 2;
        suspend;
      end 
    end
  end
end
