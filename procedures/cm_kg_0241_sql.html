CREATE PROCEDURE CM_KG_0241_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable K_DISB_DATE D_DATE;
declare variable MIN_AMOUNT D_CURRENCY;
begin
    if (f_get_const('kont_insurance_active') <> 1) then
        exit;

    min_amount = cast(f_get_const('kont_insurance_min_amount') as d_currency);
    
    select
        coalesce(k.k_clients_cashier_id, 0),
        k.k_disb_date
    from kontract k
    inner join knclient kc on kc.kc_from_k_id = k.k_id
        and kc.kc_from_c_id = :c_id
        and kc.kc_amount_approved < :min_amount
        and kc.kc_amount_approved > 0
    inner join product pr on k.k_from_pr_id = pr.pr_id
        and current_date between pr.pr_date_begin and pr.pr_date_end
        and pr_code containing '.IND.'
        and pr_code containing '.STANDART.'
        and pr_code not containing '.consultation.'
    where k.k_id = :k_id
    into
        :c_id,
        :k_disb_date;

    if (exists
        (
            select 1
            from kontract k
            inner join knclient kc on kc.kc_from_k_id = k.k_id
                and kc.kc_from_c_id = :c_id
                and kc.kc_amount_approved < :min_amount
                and kc.kc_amount_approved > 0
            inner join product pr on k.k_from_pr_id = pr.pr_id
                and current_date between pr.pr_date_begin and pr.pr_date_end
                and pr_code containing '.IND.'
                and pr_code containing '.STANDART.'
                and pr_code not containing '.consultation.'
            where k.k_id <> :k_id
                and k.k_disb_date >= :k_disb_date - 30
        )
    ) then
    begin
        res_error_is_critical = 2;
        res_error_text = 'Индивидуальный стандарт на сумму меньше '||:min_amount||' сом можно выдать только 1 раз в 30 дней';
        suspend;
        exit;
    end
end
