CREATE PROCEDURE HP_REPAY_SAVE_OPERBOOK(
    K_ID D_ID,
    REP_DATE D_DATE,
    IS_WRITEOFF D_INTEGER,
    REPAY_FACT_OFFICE_ID D_INTEGER,
    REPAY_DT_ACCOUNT_KIND D_INTEGER,
    REPAY_DT_ACCOUNT_NO D_STR_20,
    REPAY_CURR_RATE D_CURRENCY,
    REPAY_USER_ID D_ID,
    REPAY_DOCUMENT_CLIENT D_ID,
    REPAY_DOC_CLIENT_NAME_REASON D_STR_255,
    REPAY_MANAGER_C_ID D_ID,
    REPAY_FROM_RD_ID D_INTEGER,
    REPAY_TO_OS D_INTEGER,
    REPAY_TO_PERC D_INTEGER,
    REPAY_TO_FINE D_INTEGER,
    NEED_WRITEOFF_OPEREXTRA D_INTEGER = 1)
RETURNS (
    OL_ID D_ID,
    OL_MAIN_ID D_ID,
    OL_DOCUMENT_KIND D_INTEGER)
AS
declare variable res_reg_id d_integer;
declare variable res_currency d_str_lang;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable total_line d_integer;
declare variable res_line_no d_integer;
declare variable res_amount d_currency;
declare variable res_memo d_str_255;
declare variable ol_id_prior d_id;
declare variable new_ol_id d_id;
declare variable curr_region_id d_integer;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_doc_no d_integer;
declare variable ol_doc_no_str d_str_20;
declare variable ol_memo d_str_long;
declare variable res_amount_ekv d_currency;
declare variable z_os d_currency;
declare variable z_perc d_currency;
declare variable z_fine d_currency;
declare variable ol_parent_id d_id;
declare variable curr_currency d_str_lang;
declare variable loan_currency d_str_lang;
declare variable res_curr_rate d_currency = 0;
declare variable ol_doc_kind d_integer;
declare variable db_id d_integer;
declare variable bg_id d_str_20;
declare variable pal_from_at_id d_integer;
declare variable res_acc_dt_vs d_str_20;
declare variable bg_id_vs d_str_20;
declare variable res_acc_kt_vs d_str_20;
declare variable pay_os curr_type;
declare variable pay_perc curr_type;
declare variable pay_os_ekv curr_type;
declare variable pay_perc_ekv curr_type;
declare variable kc_id d_id;
declare variable acc_id d_id;
declare variable opl_from_acc_id d_id;
declare variable operrepay_ol_id d_id;
declare variable operrepay_main_ol_id d_id;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable add_acc_dt d_str_20;
declare variable add_acc_kt d_str_20;
declare variable pr_id d_integer;
declare variable pr_code d_str_100;
declare variable perc_calced d_currency;
declare variable perc_delayed d_currency;
declare variable amount1 d_currency;
declare variable amount2 d_currency;
declare variable k_amount_approved d_currency;
declare variable k_percent_per_day d_percent;
declare variable pr_is_360 d_integer;
declare variable dd1 d_date;
declare variable calced d_currency;
declare variable tmp_amount d_currency;
declare variable days d_integer;
begin
  select topr_from_ol_id
  from temp_operrepay
  where topr_from_k_id = :k_id
  into operrepay_ol_id;

  if (operrepay_ol_id is not null) then
  begin
    select iif(ol_parent_id is not null, ol_parent_id, ol_id)
    from operlist
    where ol_id = :operrepay_ol_id
    into operrepay_main_ol_id;
  end

  db_id = f_get_db_id();
  curr_region_id = f_hq_id(db_id);

  if (repay_to_os is null) then
    repay_to_os = 1;
  if (repay_to_perc is null) then
    repay_to_perc = 1;
  if (repay_to_fine is null) then
    repay_to_fine = 1;

  select sum(zad_os), sum(calcing_perc), sum(zad_fine)
  from temp_vrepay
  into :z_os, :z_perc, :z_fine;

  if (z_os = 0 and
      repay_to_os = 0) then
    repay_to_os = 1;
  if (z_perc = 0 and
      repay_to_perc = 0) then
    repay_to_perc = 1;
  if (z_fine = 0 and
      repay_to_fine = 0) then
    repay_to_fine = 1;

  select k_index_curr_code, k_amount_approved, k_percent_per_day, pr_is_360
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into :loan_currency, k_amount_approved, k_percent_per_day, pr_is_360;

  if (repay_curr_rate = 0) then
  begin
    select curr_rate
    from get_curr_rate(:loan_currency, :rep_date)
    into :repay_curr_rate;
  end

  if (repay_dt_account_kind = 3 and
      repay_dt_account_no containing '01-01000007') then
  begin
    select kc_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into kc_id;

    select acc_id
    from get_ref_account_knclient('Микрозаймы.Погашено.ЗалогСТДО', '', :kc_id, 0, 0, :rep_date)
    into acc_id;
  end

  select res_memo
  from hp_repay_make_memo(:k_id, :rep_date, :is_writeoff, :repay_doc_client_name_reason, :repay_to_os, :repay_to_perc, :repay_to_fine, :repay_curr_rate)
  into :ol_memo;
  if (ol_memo is null) then
    ol_memo = '';

  ol_id_prior = 0;
  for
    select
      new_ol_id,
      res_reg_id,
      res_currency,
      res_acc_dt,
      res_acc_kt,
      total_line,
      res_line_no,
      res_amount,
      res_memo,
      curr_currency,
      pr_id,
      pr_code
    from hp_repay_save_operbook_tmp(:k_id, :rep_date, :is_writeoff, :repay_fact_office_id, :repay_dt_account_no, :need_writeoff_operextra, :repay_from_rd_id)
    order by new_ol_id, res_record_no
  into
    :new_ol_id,
    :res_reg_id,
    :res_currency,
    :res_acc_dt,
    :res_acc_kt,
    :total_line,
    :res_line_no,
    :res_amount,
    :res_memo,
    :curr_currency,
    :pr_id,
    :pr_code
  do
  begin
    if (operrepay_ol_id is not null and
        total_line <> res_line_no) then
      continue;

    if (res_currency <> curr_currency) then
      res_curr_rate = repay_curr_rate;
    else
      res_curr_rate = 1;

    if (new_ol_id <> ol_id_prior) then
    begin
      if (operrepay_ol_id is null) then
      begin
        -- Формирование очередного номера OL_ID
        select kv
        from get_kv(:curr_region_id, 5)
        into :ol_id;
  
        if (ol_main_id is null) then
          ol_main_id = ol_id;
        if (res_line_no = 1) then
          ol_parent_id = null;
        else
          ol_parent_id = ol_main_id;
      end else
      begin
        ol_id = operrepay_ol_id;
        ol_main_id = operrepay_main_ol_id;
      end

      ol_id_prior = new_ol_id;

      -- определение параметров вставляемой операции

      -- тип документа
      if (res_line_no = 1) then
      begin
        ol_doc_kind = (case
                         when :repay_dt_account_kind = 1 then 1
                         when :repay_dt_account_kind = 2 then 7
                         when :repay_dt_account_kind = 3 then 9
                         when :repay_dt_account_kind = 18 then 18  --Чек о приеме платежа по ПВН
                       end);
        if (ol_document_kind is null) then
          ol_document_kind = ol_doc_kind;

        if (operrepay_ol_id is null) then
        begin
          -- очередной номер документа
          select num
          from last_doc_number(:rep_date, :ol_doc_kind, :repay_fact_office_id)
          into :ol_doc_no;
          ol_doc_no_str = :ol_doc_no;
        end
      end
      else
        ol_doc_kind = 9;

      -- тип операции
      if (res_line_no = total_line) then
      begin
        ol_oper_type = 1;
        ol_oper_subtype = 2 + is_writeoff;
      end
      else
      begin
        ol_oper_type = 2;
        ol_oper_subtype = 20;
      end

      if (operrepay_ol_id is null) then
      begin
        -- Вставка новой операции в Operlist
        insert into operlist (ol_id, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str, ol_oper_type, ol_oper_subtype,
                              ol_doc_kind, ol_memo, ol_user_id, ol_document_client, ol_doc_client_name, ol_is_client,
                              ol_parent_id, ol_manager_c_id, ol_from_rd_id)
        values (:ol_id, :res_reg_id, :rep_date, :ol_doc_no, :ol_doc_no_str, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind,
                :ol_memo, :repay_user_id, :repay_document_client, :repay_doc_client_name_reason,
                iif(:repay_document_client is not null, 1, 0), :ol_parent_id, :repay_manager_c_id, :repay_from_rd_id);
  
        if (res_line_no = total_line) then
        begin
          update operloan
          set opl_from_al_id = :repay_dt_account_no,
              opl_acc_kind = :repay_dt_account_kind,
              opl_from_acc_id = iif(opl_from_acc_id is null, :acc_id, opl_from_acc_id)
          where opl_from_ol_id = :ol_main_id
          returning opl_from_acc_id
          into opl_from_acc_id;
  
          if (opl_from_acc_id is null) then
          begin
            update operloan
            set opl_from_acc_id = :acc_id
            where opl_from_ol_id = :ol_main_id;
          end
        end
      end
    end

    if (res_curr_rate = 1) then
      res_amount_ekv = res_amount;
    else
      res_amount_ekv = round((res_amount * res_curr_rate), 2);

    if (is_writeoff = 1
      and res_memo starting 'Проценты по'
      and db_id in (39801, 41701)
      and rep_date >= '01.08.2023'
      and pr_code containing '.murabaha.'
      /*and f_new_markup_mbh(k_id)*/) then
    begin
      acc_dt = '11031/2';
      acc_kt = '11941/1';
      res_acc_kt = (
        select al_id
        from acc_list
        where al_from_reg_id = :res_reg_id
          and al_currency = '417'
          and al_account = :acc_kt);

      -- начисленные проценты
      if (f_tmp_const('ipf_percent_mode_active') = 1) then
      begin
        tmp_amount = 0;
        dd1 = rep_date - extract(day from rep_date) + 1;

        select -sum(ok_sh_perc)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_ol_date between :dd1 and :rep_date and
              ok_type = 3
        into tmp_amount;
      
        if (tmp_amount is null) then
          tmp_amount = 0;

        days = f_agein360(dd1, rep_date, pr_is_360) + 1;
        calced = round(cast(k_amount_approved * k_percent_per_day as double precision) * days / 100.00, 0);
        perc_calced = calced;
      end else
      begin
        perc_calced = coalesce((
          select
            sh_perc
          from schedule
          where sh_from_k_id = :k_id
            and sh_date >= :rep_date
          order by sh_date rows 1), 0);
      end


      -- отсроченный доход
      perc_delayed = coalesce((
        select
          kcp_delayed_perc_end
        from kontcalcperc
        where kcp_from_k_id = :k_id
        order by kcp_period desc rows 1), 0);

      amount1 = iif(res_amount > perc_delayed, perc_delayed - perc_calced, res_amount);
      amount2 = iif(res_amount > perc_delayed, res_amount + perc_calced - perc_delayed,
                                               perc_delayed - perc_calced - res_amount - tmp_amount);

      if (amount2 < 0) then
      begin
        amount1 = amount1 - amount2;
        amount2 = 0;
      end

      update or insert into temp_operkroff_mbh (tom_k_id, tom_perc, tom_other)
      values (:k_id, iif(:res_amount > :perc_delayed, :amount2, 0), :amount1) matching (tom_k_id);

      res_acc_dt = (select res_account from hp_make_kredit_acclist(:pr_id, :res_reg_id, 30));
      if (amount1 <> 0) then
      begin
        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_summa_val,
          ob_summa,
          ob_detail,
          ob_ol_date,
          ob_ol_is_active,
          ob_ol_reg_id)
        values (
          :ol_id,
          :res_currency,
          :res_curr_rate,
          :res_acc_dt,
          :res_acc_kt,
          :amount1,
          :amount1,
          :res_memo,
          :rep_date,
          1,
          :res_reg_id);
      end

      if (amount2 <> 0) then
      begin
        res_acc_dt = (select res_account from hp_make_kredit_acclist(:pr_id, :res_reg_id, iif(:res_amount > :perc_delayed, 15, 30)));
        if (res_amount <= perc_delayed) then
          res_acc_kt = (select res_account from hp_make_kredit_acclist(:pr_id, :res_reg_id, 30));
        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_summa_val,
          ob_summa,
          ob_detail,
          ob_ol_date,
          ob_ol_is_active,
          ob_ol_reg_id)
        values (
          :ol_id,
          :res_currency,
          :res_curr_rate,
          :res_acc_dt,
          :res_acc_kt,
          :amount2,
          :amount2,
          :res_memo,
          :rep_date,
          1,
          :res_reg_id);
      end
    end else
      insert into operbook (
        ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
        ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (
        :ol_id, :res_currency, :res_curr_rate, :res_acc_dt, :res_acc_kt, :res_amount,
        :res_amount_ekv, :res_memo, :rep_date, 1, :res_reg_id)
      returning ob_short_acc_dt, ob_short_acc_kt
      into :acc_dt, :acc_kt;

    if (db_id = 41701 and is_writeoff = 1) then
    begin
      if (acc_dt = '11001/1' and acc_kt in ('10931/1', '10941/1', '10941/11')) then
      begin
        acc_dt = '95001/1';
        acc_kt = '95002/1';
      end else
      if (acc_dt = '11001/2' and acc_kt in ('11437/1', '11437/11', '11437/2')) then
      begin
        acc_dt = '95003/1';
        acc_kt = '95004/1';
      end else
      if (acc_dt = '11031/1' and acc_kt in ('10972/1', '10972/2', '10974/1', '10976/1', '10977/1')) then
      begin
        acc_dt = '95001/2';
        acc_kt = '95002/2';
      end else
      if (acc_dt = '11031/2' and acc_kt in ('11941/1', '11942/1', '11942/2', '11943/1')) then
      begin
        acc_dt = '95003/2';
        acc_kt = '95004/2';
      end else
      begin
        acc_dt = '';
        acc_kt = '';
      end
      if (acc_dt <> '' and acc_kt <> '') then
      begin
        add_acc_dt = (select al_id from acc_list
          where al_from_reg_id = :res_reg_id and al_currency = :res_currency and al_account = :acc_dt);
        add_acc_kt = (select al_id from acc_list
          where al_from_reg_id = :res_reg_id and al_currency = :res_currency and al_account = :acc_kt);
        insert into operbook (
          ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
          ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
        values (
          :ol_id, :res_currency, :res_curr_rate, :add_acc_dt, :add_acc_kt, :res_amount,
          :res_amount_ekv, :res_memo, :rep_date, 1, :res_reg_id);
      end
    end

    if (is_writeoff = 1 and
        db_id = 94401 and
        not exists(select *
                   from temp_params
                   where coalesce(rd_exclude_from_operkroff, 0) = 2)) then -- добавил Улан от 18.11.2019
    begin
      select al_from_bg_id
      from acc_list
      where al_id = :res_acc_kt -- из счета кредита вычисляем счет ОС или процента через bg_id
      into bg_id;

      pal_from_at_id = null;
      select pal_from_at_id
      from kontract, product, prodacclist
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            pr_from_pat_id = pal_from_pat_id and
            (pal_main_acc = :bg_id or
            pal_delinq_acc = :bg_id)
      into pal_from_at_id;

      bg_id_vs = null;
      if (pal_from_at_id = 1) then -- основная сумма
      begin
        bg_id_vs = '99300';
      end
      else
      if (pal_from_at_id = 2) then -- проценты
      begin
        bg_id_vs = '99301';
      end

      res_acc_dt_vs = null;
      if (bg_id_vs is not null) then
      begin
        select al_id
        from acc_list
        where is_delete = 0 and
              al_is_active = 1 and
              al_from_reg_id = :res_reg_id and
              al_from_bg_id = :bg_id_vs and
              al_currency = :res_currency
        into res_acc_dt_vs;
      end
      --    res_acc_dt_vs = null то нужно эксепт вернуть что не создан счет
      if (res_acc_dt_vs is null) then
      begin
        exception e_default 'в текущем офисе не создан счет:' || coalesce(bg_id_vs, '');
      end
      res_acc_kt_vs = null;
      select al_id
      from acc_list
      where is_delete = 0 and
            al_is_active = 1 and
            al_from_reg_id = :res_reg_id and
            al_from_bg_id = '99990' and
            al_currency = :res_currency
      into res_acc_kt_vs;
      --    res_acc_kt_vs = null то нужно эксепт вернуть что не создан счет
      if (res_acc_kt_vs is null) then
      begin
        exception e_default 'в текущем офисе не создан счет:99990';
      end
      if (res_acc_dt_vs is not null and
          res_acc_kt_vs is not null) then
      begin
        insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                              ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
        values (:ol_id, :res_currency, :res_curr_rate, :res_acc_dt_vs, :res_acc_kt_vs, :res_amount, :res_amount_ekv,
                :res_memo, :rep_date, 1, :res_reg_id);
      end
    end
    if (res_acc_kt containing '89150' and
        db_id = 94401) then
    begin
      for select sum(pay_os), sum(pay_perc)
          from temp_vrepay
          where s_date is null having sum(pay_os) <> 0 or
                sum(pay_perc) <> 0
          into pay_os, pay_perc
      do
      begin
        res_acc_dt_vs = null;
        select al_id
        from acc_list
        where is_delete = 0 and
              al_is_active = 1 and
              al_from_reg_id = :res_reg_id and
              al_from_bg_id = '99990' and
              al_currency = :res_currency
        into res_acc_dt_vs;

        if (res_acc_dt_vs is null) then
        begin
          exception e_default 'в текущем офисе не создан счет:99990';
        end

        if (pay_os <> 0) then
        begin
          bg_id_vs = '99300';
          res_acc_kt_vs = null;
          select al_id
          from acc_list
          where is_delete = 0 and
                al_is_active = 1 and
                al_from_reg_id = :res_reg_id and
                al_from_bg_id = :bg_id_vs and
                al_currency = :res_currency
          into res_acc_kt_vs;

          if (res_acc_kt_vs is null) then
          begin
            exception e_default 'в текущем офисе не создан счет:' || coalesce(bg_id_vs, '');
          end

          if (res_acc_dt_vs is not null and
              res_acc_kt_vs is not null) then
          begin
            if (res_curr_rate = 1) then
              pay_os_ekv = pay_os;
            else
              pay_os_ekv = round((pay_os * res_curr_rate), 2);

            insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                                  ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
            values (:ol_id, :res_currency, :res_curr_rate, :res_acc_dt_vs, :res_acc_kt_vs, :pay_os, :pay_os_ekv,
                    :res_memo, :rep_date, 1, :res_reg_id);
          end
        end

        if (pay_perc <> 0) then
        begin
          bg_id_vs = '99301';
          res_acc_kt_vs = null;
          select al_id
          from acc_list
          where is_delete = 0 and
                al_is_active = 1 and
                al_from_reg_id = :res_reg_id and
                al_from_bg_id = :bg_id_vs and
                al_currency = :res_currency
          into res_acc_kt_vs;

          if (res_acc_kt_vs is null) then
          begin
            exception e_default 'в текущем офисе не создан счет:' || coalesce(bg_id_vs, '');
          end

          if (res_acc_dt_vs is not null and
              res_acc_kt_vs is not null) then
          begin
            if (res_curr_rate = 1) then
              pay_perc_ekv = pay_perc;
            else
              pay_perc_ekv = round((pay_perc * res_curr_rate), 2);

            insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                                  ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
            values (:ol_id, :res_currency, :res_curr_rate, :res_acc_dt_vs, :res_acc_kt_vs, :pay_perc, :pay_perc_ekv,
                    :res_memo, :rep_date, 1, :res_reg_id);
          end
        end
      end
    end
  end

  if (operrepay_ol_id is null and ol_id is null) then
  begin
    -- Формирование очередного номера OL_ID
    select kv
    from get_kv(:curr_region_id, 5)
    into :ol_id;

    if (ol_main_id is null) then
      ol_main_id = ol_id;
    ol_parent_id = null;

    -- определение параметров вставляемой операции
    -- тип документа
    ol_doc_kind = (case
                     when :repay_dt_account_kind = 1 then 1
                     when :repay_dt_account_kind = 2 then 7
                     when :repay_dt_account_kind = 3 then 9
                     when :repay_dt_account_kind = 18 then 18 --Чек о приеме платежа по ПВН
                   end);
    if (ol_document_kind is null) then
      ol_document_kind = ol_doc_kind;

    -- очередной номер документа
    ol_doc_no = 0;
    ol_doc_no_str = '';

    -- тип операции
    ol_oper_type = 1;
    ol_oper_subtype = 2 + is_writeoff;

    select kal_new_reg_id
    from knaktlist, kontract
    where k_id = :k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_date <= :rep_date and
          kal_date_end >= :rep_date
    into :res_reg_id;

    -- Вставка новой операции в Operlist
    insert into operlist (ol_id, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str, ol_oper_type, ol_oper_subtype,
                          ol_doc_kind, ol_memo, ol_user_id, ol_document_client, ol_doc_client_name, ol_is_client,
                          ol_parent_id, ol_manager_c_id, ol_from_rd_id)
    values (:ol_id, :res_reg_id, :rep_date, :ol_doc_no, :ol_doc_no_str, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind,
            :ol_memo, :repay_user_id, :repay_document_client, :repay_doc_client_name_reason,
            iif(:repay_document_client is not null, 1, 0), :ol_parent_id, :repay_manager_c_id, :repay_from_rd_id);
  end

  suspend;
end
