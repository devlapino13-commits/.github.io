CREATE PROCEDURE CM_RU_0004_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_client_pass_no varchar(30);
declare variable cc_is_active smallint;
declare variable type_0 smallint;
declare variable type_1 smallint;
declare variable pr_code varchar(20);
declare variable c_id id_type;
begin
  select cm_cc_is_active from company
  into cc_is_active;
  if (cc_is_active = 1) then  -- если блок по КЦ активный, тогда выполняется условие
  begin
    for
    select kc_from_c_id
    from kontract, product, knclient
    where k_id = :k_id
    and k_from_pr_id = pr_id
    and pr_product_kind <> 2
    and pr_code not containing '.week.'
    and k_id = kc_from_k_id
    and kc_amount_approved > 0
    into c_id
    do
    begin
      type_0 = null;
      type_1 = null;
      select sum(iif(cc_type = 0, 1,0)) type_0,
             sum(iif(cc_type = 1, 1,0)) type_1
      from contacts
      where cc_from_c_id = :c_id
      and cc_check_result = 1
      and cc_is_old_contact = 0
      into type_0, type_1;
      if (type_0 is null) then type_0 = 0;
      if (type_1 is null) then type_1 = 0;
      if ((type_0 + type_1) < 3) then
      begin
         select c_name_fio, c_pass_no
         from clients
         where c_id = :c_id
         into res_info_text, KF_CLIENT_PASS_NO;
        res_error_text = 'Минимум 3 контактных лица должны быть подтверждены для клиента '||res_info_text;
        res_error_is_critical = 2;
        suspend;
      end
    end
  end
end
