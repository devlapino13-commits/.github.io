CREATE PROCEDURE AAREP_CALC_BONUS_RU_TMP_NEW(
    REP_REGION INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_CIRCLE_CLIENT INTEGER)
RETURNS (
    REP_REGION_ID INTEGER,
    REP_FROM_K_ID DECIMAL(18,0),
    REP_FROM_C_ID DECIMAL(18,0),
    REP_KOLICH_BEGIN INTEGER,
    REP_KOLICH_DISB_NEW INTEGER,
    REP_KOLICH_DISB_REPEAT INTEGER,
    REP_KOLICH_CLOSE INTEGER,
    REP_KOLICH_TRANSFER INTEGER,
    REP_KOLICH_END INTEGER,
    REP_DELIQ_IN_MONTH INTEGER,
    REP_DELIQ_AT_END INTEGER,
    REP_OFFICE_NAME VARCHAR(100),
    REP_KC_NAME VARCHAR(100),
    REP_FOLDER_NAME VARCHAR(100))
AS
declare variable kc_circle integer;
declare variable kol integer;
declare variable k_id_prior decimal(18,0);
declare variable k_circle integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  rep_kolich_begin = 0;
  rep_kolich_close = 0;
  rep_kolich_transfer = 0;
  rep_kolich_end = 0;
  rep_deliq_in_month = 0;
  rep_deliq_at_end = 0;

  for select ks_from_k_id, ks_region_id, ks_kc_id, reg_code || ' ' || reg_name, c_name_fio, ks_kfolder_no, k_circle
      from kontstate, kontract, region_list(:rep_region, 1, -1, 1), region, clients
      where ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_region_id = id_list and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_region_id = reg_unicode and
            ks_kc_id = c_id
      into :rep_from_k_id, :rep_region_id, :rep_from_c_id, :rep_office_name, :rep_kc_name, :rep_folder_name, :k_circle
  do
  begin
    if (is_circle_client = 1) then
    begin
      rep_kolich_disb_repeat = 0;
      rep_kolich_disb_new = 0;
      for select kc_circle
          from knclient
          where kc_from_k_id = :rep_from_k_id and
                kc_amount_approved <> 0 and
                kc_amount_approved is not null
          into :kc_circle
      do
      begin
        if (kc_circle is null) then
          kc_circle = 1;
        if (kc_circle = 1) then
          rep_kolich_disb_new = rep_kolich_disb_new + 1;
        else
          rep_kolich_disb_repeat = rep_kolich_disb_repeat + 1;
      end
      suspend;
    end
    else
    begin
      rep_kolich_disb_repeat = 0;
      rep_kolich_disb_new = 0;
      for select kc_circle
          from knclient
          where kc_from_k_id = :rep_from_k_id and
                kc_amount_approved <> 0 and
                kc_amount_approved is not null
          into :kc_circle
      do
      begin
        if (k_circle is null) then
          k_circle = 1;
        if (k_circle = 1) then
          rep_kolich_disb_new = rep_kolich_disb_new + 1;
        else
          rep_kolich_disb_repeat = rep_kolich_disb_repeat + 1;
      end
      suspend;
    end
  end
  rep_kolich_disb_repeat = 0;
  rep_kolich_disb_new = 0;

  for select ks_from_k_id, kal_new_reg_id, kal_new_kf_officer, k_kolich_client_approved, reg_code || ' ' || reg_name,
             c_name_fio, kal_new_kf_name
      from kontstate, kontract, knaktlist, region_list(:rep_region, 1, -1, 1), region, clients
      where ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            knaktlist.is_delete = 0 and
            k_from_kf_id = kal_from_kf_id and
            (kal_new_reg_id = id_list) and
            ks_state = 1 and
            cast(ks_date_begin as date) <= (:rep_date_begin - 1) and
            ks_date_end >= (:rep_date_begin - 1) and
            kal_date <= :rep_date_begin and
            kal_date_end >= :rep_date_begin and
            kal_new_reg_id = reg_unicode and
            kal_new_kf_officer = c_id
      into :rep_from_k_id, :rep_region_id, :rep_from_c_id, rep_kolich_begin, :rep_office_name, :rep_kc_name,
           :rep_folder_name
  do
  begin
    suspend;
  end
  rep_kolich_begin = 0;

  for select ks_from_k_id, kal_new_reg_id, kal_new_kf_officer, k_kolich_client_approved, reg_code || ' ' || reg_name,
             c_name_fio, kal_new_kf_name
      from kontstate, kontract, knaktlist, region_list(:rep_region, 1, -1, 1), region, clients
      where ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            knaktlist.is_delete = 0 and
            k_from_kf_id = kal_from_kf_id and
            (kal_new_reg_id = id_list) and
            ks_state = 1 and
            cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end >= :rep_date_end and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            kal_new_reg_id = reg_unicode and
            kal_new_kf_officer = c_id
      into :rep_from_k_id, :rep_region_id, :rep_from_c_id, rep_kolich_end, :rep_office_name, :rep_kc_name,
           :rep_folder_name
  do
  begin
    suspend;
  end
  rep_kolich_end = 0;

  for select ks_from_k_id, kal_new_reg_id, kal_new_kf_officer, k_kolich_client_approved, reg_code || ' ' || reg_name,
             c_name_fio, kal_new_kf_name
      from kontstate, kontract, knaktlist, region_list(:rep_region, 1, -1, 1), region, clients
      where ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            knaktlist.is_delete = 0 and
            k_from_kf_id = kal_from_kf_id and
            (kal_new_reg_id = id_list) and
            ks_state = 2 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            kal_date <= ks_date_begin and
            kal_date_end >= ks_date_begin and
            kal_new_reg_id = reg_unicode and
            kal_new_kf_officer = c_id
      into :rep_from_k_id, :rep_region_id, :rep_from_c_id, rep_kolich_close, :rep_office_name, :rep_kc_name,
           :rep_folder_name
  do
  begin
    suspend;
  end
  rep_kolich_close = 0;
  for select ok_from_k_id, ok_ol_reg_id, kal_new_kf_officer, max(cast((ok_ol_date - ok_sh_date) as integer)),
             reg_code || ' ' || reg_name, c_name_fio, kal_new_kf_name
      from operkred, kontract, knaktlist, region, clients
      where ok_ol_is_active = 1 and
            ok_from_k_id = k_id and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= ok_ol_date and
            kal_date_end >= ok_ol_date and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_ol_date > ok_sh_date and
            (ok_os <> 0 and
            ok_perc <> 0) and
            (ok_type >= 4 or
            ok_type <= 6) and
            ok_sh_date >= :rep_date_begin and
            ok_ol_reg_id = reg_unicode and
            kal_new_kf_officer = c_id and
            ok_from_c_id = 0
            /* and (((ol_date-kal_date>60) and (kal_from_ka_id>0)) or (kal_from_ka_id=0)) */
      group by ok_from_k_id, ok_ol_reg_id, kal_new_kf_officer, reg_code, reg_name, c_name_fio, kal_new_kf_name
      into :rep_from_k_id, :rep_region_id, :rep_from_c_id, :rep_deliq_in_month, :rep_office_name, :rep_kc_name,
           :rep_folder_name
  do
  begin
    suspend;
  end
  rep_deliq_in_month = 0;
end
