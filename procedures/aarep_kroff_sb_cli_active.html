CREATE PROCEDURE AAREP_KROFF_SB_CLI_ACTIVE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_KONTRACT_NO VARCHAR(20),
    AA_SB_TEAM_NAME VARCHAR(255),
    AA_SB_NAME VARCHAR(255),
    SBA_DATE_BEGIN DATE,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_KOLICH_CL_ACTIVE_BEGIN INTEGER,
    AA_PRODUCT_NAME VARCHAR(100),
    AA_LOAN_CURRENCY VARCHAR(30),
    AA_LOAN_KIND VARCHAR(30),
    AA_DELIQ_REPAY_KOLICH INTEGER,
    AA_DELIQ_DAYS INTEGER,
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_DELIQ_PERC_ALL DECIMAL(18,4),
    AA_DELIQ_OS_ALL DECIMAL(18,4),
    AA_REPAY_WRITEOFF_FINE DECIMAL(18,4),
    AA_REPAY_WRITEOFF_PERC DECIMAL(18,4),
    AA_REPAY_WRITEOFF_OS DECIMAL(18,4),
    AA_REPAY_ALL_KOMMIS_ADV DECIMAL(18,4),
    AA_REPAY_ALL_KOMMIS DECIMAL(18,4),
    AA_REPAY_ALL_FINE DECIMAL(18,4),
    AA_REPAY_ALL_PERC DECIMAL(18,4),
    AA_REPAY_ALL_OS DECIMAL(18,4),
    AA_STATUS_TEXT VARCHAR(20),
    AA_KOLICH_CLOSED INTEGER,
    AA_LOAN_AMOUNT DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_BEGIN DECIMAL(18,4),
    AA_CURR_KC_SEX VARCHAR(20),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_DELIQ_BEG_CL_KOLICH INTEGER,
    AA_DELIQ_BEG_BALANCE DECIMAL(18,4),
    K_CASHIER_FIO VARCHAR(100),
    SBA_CLIENT_NAME VARCHAR(100),
    SBA_CL_BIRTH_DATE DATE,
    SBA_CL_TEL_NO VARCHAR(20),
    C_ADDRESS_YUR VARCHAR(255),
    C_ADDRESS_FACT VARCHAR(255))
AS
declare variable kc_from_c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable dolya double precision;
declare variable oko_payed_perc1 curr_type;
declare variable oko_payed_os1 curr_type;
declare variable oko_balance_total1 curr_type;
declare variable oko_balance_perc1 curr_type;
declare variable oko_balance_other1 curr_type;
declare variable oko_balance_os1 curr_type;
declare variable oko_payed_other1 curr_type;
declare variable oko_payed_os_per1 curr_type;
declare variable oko_payed_perc_per1 curr_type;
declare variable oko_payed_other_per1 curr_type;
declare variable oko_payed_total_per1 curr_type;
declare variable oko_payed_total1 curr_type;
declare variable aa_k_id id_type;
declare variable aa_deliq_balance1 curr_type;
declare variable aa_deliq_perc_all1 curr_type;
declare variable aa_deliq_os_all1 curr_type;
declare variable aa_repay_writeoff_fine1 curr_type;
declare variable aa_repay_writeoff_perc1 curr_type;
declare variable aa_repay_writeoff_os1 curr_type;
declare variable aa_repay_all_kommis_adv1 curr_type;
declare variable aa_repay_all_kommis1 curr_type;
declare variable aa_repay_all_fine1 curr_type;
declare variable aa_repay_all_perc1 curr_type;
declare variable aa_repay_all_os1 curr_type;
declare variable aa_loan_amount1 curr_type;
declare variable aa_amount_active_end1 curr_type;
declare variable aa_amount_active_begin1 curr_type;
begin
-- knclient_fix

  for select aa_curr_office_name, aa_kontract_no, aa_sb_team_name, aa_sb_name, sba_date_begin, aa_kolich_cl_active_end,
             aa_kolich_cl_active_begin, aa_product_name, aa_loan_currency, aa_loan_kind, aa_deliq_repay_kolich,
             aa_deliq_days, aa_deliq_balance, aa_deliq_perc_all, aa_deliq_os_all, aa_repay_writeoff_fine,
             aa_repay_writeoff_perc, aa_repay_writeoff_os, aa_repay_all_kommis_adv, aa_repay_all_kommis,
             aa_repay_all_fine, aa_repay_all_perc, aa_repay_all_os, aa_status_text, aa_kolich_closed, aa_loan_amount,
             aa_amount_active_end, aa_amount_active_begin, aa_curr_kc_sex, aa_curr_kc_fio, aa_curr_level_name,
             aa_curr_branch_name, aa_deliq_beg_cl_kolich, aa_deliq_beg_balance, k_cashier_fio, aa_k_id
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, 'PROBLEM2', 0, 0, 1, 1, 0, 0, 0)
      into aa_curr_office_name, aa_kontract_no, aa_sb_team_name, aa_sb_name, sba_date_begin, aa_kolich_cl_active_end,
           aa_kolich_cl_active_begin, aa_product_name, aa_loan_currency, aa_loan_kind, aa_deliq_repay_kolich,
           aa_deliq_days, aa_deliq_balance, aa_deliq_perc_all, aa_deliq_os_all, aa_repay_writeoff_fine,
           aa_repay_writeoff_perc, aa_repay_writeoff_os, aa_repay_all_kommis_adv, aa_repay_all_kommis,
           aa_repay_all_fine, aa_repay_all_perc, aa_repay_all_os, aa_status_text, aa_kolich_closed, aa_loan_amount,
           aa_amount_active_end, aa_amount_active_begin, aa_curr_kc_sex, aa_curr_kc_fio, aa_curr_level_name,
           aa_curr_branch_name, aa_deliq_beg_cl_kolich, aa_deliq_beg_balance, k_cashier_fio, aa_k_id

  do
  begin
    if (aa_kolich_cl_active_end > 0) then
      aa_kolich_cl_active_end = 1;
    if (aa_kolich_cl_active_begin > 0) then
      aa_kolich_cl_active_begin = 1;

    aa_deliq_balance1 = aa_deliq_balance;
    aa_deliq_perc_all1 = aa_deliq_perc_all;
    aa_deliq_os_all1 = aa_deliq_os_all;
    aa_repay_writeoff_fine1 = aa_repay_writeoff_fine;
    aa_repay_writeoff_perc1 = aa_repay_writeoff_perc;
    aa_repay_writeoff_os1 = aa_repay_writeoff_os;
    aa_repay_all_kommis_adv1 = aa_repay_all_kommis_adv;
    aa_repay_all_kommis1 = aa_repay_all_kommis;
    aa_repay_all_fine1 = aa_repay_all_fine;
    aa_repay_all_perc1 = aa_repay_all_perc;
    aa_repay_all_os1 = aa_repay_all_os;
    aa_loan_amount1 = aa_loan_amount;
    aa_amount_active_end1 = aa_amount_active_end;
    aa_amount_active_begin1 = aa_amount_active_begin;

    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :aa_k_id and
              kc_amount_approved > 0
--     and KC_CLOSED_OL_ID is null -- не будет включать тех которые закрыли свою долю
        into kc_from_c_id, kc_amount_approved
    do
    begin
      dolya = kc_amount_approved / aa_loan_amount1;

      aa_deliq_balance = dolya * aa_deliq_balance1;
      aa_deliq_perc_all = dolya * aa_deliq_perc_all1;
      aa_deliq_os_all = dolya * aa_deliq_os_all1;
      aa_repay_writeoff_fine = dolya * aa_repay_writeoff_fine1;
      aa_repay_writeoff_perc = dolya * aa_repay_writeoff_perc1;
      aa_repay_writeoff_os = dolya * aa_repay_writeoff_os1;
      aa_repay_all_kommis_adv = dolya * aa_repay_all_kommis_adv1;
      aa_repay_all_kommis = dolya * aa_repay_all_kommis1;
      aa_repay_all_fine = dolya * aa_repay_all_fine1;
      aa_repay_all_perc = dolya * aa_repay_all_perc1;
      aa_repay_all_os = dolya * aa_repay_all_os1;
      aa_loan_amount = kc_amount_approved;
      aa_amount_active_end = dolya * aa_amount_active_end1;
      aa_amount_active_begin = dolya * aa_amount_active_begin1;

      select c_name_fio, c_birth_date,
             (select cont_value
              from hp_contdata_get_cell_main(0, :kc_from_c_id))
      from clients
      where c_id = :kc_from_c_id
      into sba_client_name, sba_cl_birth_date, sba_cl_tel_no;

      c_address_yur = null;
      c_address_fact = null;

      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :kc_from_c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      into :c_address_yur;
      if (c_address_yur is null) then
        c_address_yur = 'Не указан';

      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :kc_from_c_id and
            adc_address_type = 2 and
            adc_from_adl_id = adl_id
      into :c_address_fact;
      if (c_address_fact is null) then
        c_address_fact = 'Не указан';

      suspend;
    end
  end

end
