CREATE PROCEDURE HP_RESET_ATTEMPTS_COUNT(
    PHONE_NO D_STR_100,
    C_ID D_ID,
    C_INN D_STR_20)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable CELL_NO D_STR_100;
declare variable CL_ID D_ID;
declare variable CNT integer = 0;
declare variable SEARCH_NUM D_STR_100;
begin
    res_text = '';

    if (coalesce(phone_no, '') <> '') then cnt = cnt + 1;
    if (coalesce(c_id, 0) <> 0)       then cnt = cnt + 1;
    if (coalesce(c_inn, '') <> '')    then cnt = cnt + 1;

    if (cnt <> 1) then
    begin
        res_text = 'Укажите один параметр для поиска';
        suspend;
        exit;
    end

    if (coalesce(phone_no, '') <> '') then
        search_num = phone_no;

    else if (coalesce(c_id, 0) <> 0) then
    begin
        cell_no = f_cont_info(f_objkind('clients', 'id'), c_id, 'cell', 'ok');

        if (coalesce(cell_no, '') = '') then
        begin
            res_text = 'У клиента нет верифицированного мобильного номера';
            suspend;
            exit;
        end

        search_num = cell_no;
    end

    else if (coalesce(c_inn, '') <> '') then
    begin
        select c.c_id
        from clients c
        where c.c_inn = :c_inn
        into cl_id;

        if (coalesce(cl_id, 0) = 0) then
        begin
            res_text = 'Клиент с указанным ИНН не найден';
            suspend;
            exit;
        end

        cell_no = f_cont_info(f_objkind('clients', 'id'), cl_id, 'cell', 'ok');

        if (coalesce(cell_no, '') = '') then
        begin
            res_text = 'У клиента нет верифицированного мобильного номера';
            suspend;
            exit;
        end

        search_num = cell_no;
    end

    delete from msgquotalist
    where mql_cont_code = :search_num;

    suspend;
end
