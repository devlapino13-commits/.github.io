CREATE PROCEDURE GET_KF_FROM_CLIENT(
    CL_STR VARCHAR(255))
RETURNS (
    KFC_ID DECIMAL(18,0))
AS
declare variable pass_no varchar(255);
begin
  /* Procedure Text */
  pass_no = replace(cl_str, ' ', '');

  if (cl_str <> '') then
  begin
    cl_str = upper(cl_str);
    for select distinct k_from_kf_id
        from knclient, kontract, hp_find_from_namelist(0, 0, :cl_str)
        where kc_from_c_id = object_id and
              kc_amount_approved > 0 and
              k_id = kc_from_k_id
        union
        select distinct k_from_kf_id
        from clients, knclient, kontract
        where c_index_c_pass_no starting with :pass_no and
              kc_from_c_id = c_id and
              kc_amount_approved > 0 and
              k_id = kc_from_k_id
        union
        select distinct k_from_kf_id
        from kontract
        where k_kontract_no starting with :cl_str
        into :kfc_id
    do
    begin
      suspend;
    end
  end
end
