CREATE PROCEDURE AAREP_CASH_JOURNAL_SUBTMP_2(
    OL_DOC_CLIENT_NAME STR_255,
    OL_DOCUMENT_CLIENT ID_TYPE,
    OL_DOCUMENT_CLIENT2 ID_TYPE,
    CM_DATABASE_ID INTEGER,
    OL_DATE DATE,
    OL_OPER_TYPE INTEGER,
    OL_OPER_SUBTYPE INTEGER,
    OL_ID ID_TYPE,
    OL_KOL_MINUS_INPUT INTEGER,
    HAS_LINE_INPUT INTEGER,
    OL_DOC_NO_STR_INPUT VARCHAR(50),
    REG_CODE VARCHAR(20),
    OL_DOC_NO INTEGER,
    OL_ACCOUNT STR_255,
    OL_SUMMA_MINUS_INPUT CURR_TYPE)
RETURNS (
    OL_SUMMA_MINUS CURR_TYPE,
    OL_CLIENT_NAME STR_255,
    OL_KOL_MINUS INTEGER,
    OL_DOC_NO_STR VARCHAR(50),
    HAS_LINE INTEGER,
    OL_ACCOUNT_2 STR_255,
    C_IS_RESIDENT INTEGER,
    KC_FIO STR_100)
AS
declare variable sub_no integer;
declare variable kk_id id_type;
declare variable oek_rd_id integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  ol_doc_no_str = ol_doc_no_str_input;
  ol_kol_minus = ol_kol_minus_input;
  ol_summa_minus = ol_summa_minus_input;
  has_line = has_line_input;

  sub_no = 0;
  if ((cm_database_id = 81001 and
      ol_oper_type = 1 and
      ol_oper_subtype = 1 and
      ol_date >= '01.01.2014')) then
  begin
    kk_id = null;
    for select distinct ok_from_k_id
        from operkred
        where ok_from_ol_id = :ol_id and
              ok_ol_is_active = 1
        union
        select distinct ox_object_id
        from operextra
        where ox_from_rd_id = :oek_rd_id and
              ox_from_ol_id = :ol_id and
              ox_object_kind = 9 and
              ox_ol_is_active = 1
--rows 1
        into :kk_id
    do
    begin
      for select kc_amount_approved, c_name_fio, iif(kncl.kc_non_resident = 1, 0, 1), cl.c_name_fio
          from knclient kncl, clients cl
          where kc_from_k_id = :kk_id and
                kc_from_c_id = c_id and
                kc_amount_approved > 0 and
                cl.c_id = kncl.kc_from_c_id
          into :ol_summa_minus, :ol_client_name, :c_is_resident, kc_fio
      do
      begin
        sub_no = sub_no + 1;
        ol_kol_minus = ol_kol_minus + 1;
        ol_doc_no_str = reg_code || '-' || right('000000' || ol_doc_no, 6) || '/' || sub_no;
        has_line = 1;

        ol_account_2 = ol_account;

        if ((cm_database_id = 81001) and
            (ol_date >= '01.01.2018')) then
        begin
          ol_account_2 = left(ol_account_2, 5);

          ol_account_2 = (case ol_account_2
                            when '50-01' then '20202'
                            when '57-01' then '20209'
                            when '76-02' then '60323'
                            when '76-06' then '47422'
                            when '91-01' then '71801'
                            when '51-01' then '20501'
                            when '70-01' then '60305'
                            when '73-02' then '60308'
                            when '94-01' then '60308'
                            when '91-01' then '71801'
                            else ol_account_2
                          end);

          if (c_is_resident = 1) then
          begin
            ol_account_2 = (case ol_account_2
                              when '58-03' then '48801'
                              when '76-09' then '48809'
                              when '98-01' then '48809'
                              else ol_account_2
                            end);
          end
          else
          begin
            ol_account_2 = (case ol_account_2
                              when '58-03' then '49201'
                              when '76-09' then '49209'
                              when '98-01' then '49209'
                              else ol_account_2
                            end);
          end
        end

        suspend;
      end
    end
  end
  else
  begin
    c_is_resident = null;

    if (ol_doc_client_name is null) then
      ol_doc_client_name = '';
    if (ol_document_client is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :ol_document_client
      into :ol_client_name;
    end
    else
    begin
      if (ol_document_client2 is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :ol_document_client2
        into :ol_client_name;
      end
      else
        ol_client_name = :ol_doc_client_name;
    end

    ol_kol_minus = ol_kol_minus + 1;
    ol_doc_no_str = reg_code || '-' || right('000000' || ol_doc_no, 6);
    has_line = 1;

    ol_account_2 = ol_account;

    if ((cm_database_id = 81001) and
        (ol_date >= '01.01.2018')) then
    begin
      ol_account_2 = left(ol_account_2, 5);

      ol_account_2 = (case ol_account_2
                        when '50-01' then '20202'
                        when '57-01' then '20209'
                        when '76-02' then '60323'
                        when '76-06' then '47422'
                        when '91-01' then '71801'
                        when '51-01' then '20501'
                        when '70-01' then '60305'
                        when '73-02' then '60308'
                        when '94-01' then '60308'
                        else ol_account_2
                      end);
    end
    suspend;
  end
end
