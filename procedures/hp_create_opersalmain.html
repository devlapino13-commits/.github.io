CREATE PROCEDURE HP_CREATE_OPERSALMAIN(
    C_ID DECIMAL(18,0))
RETURNS (
    OSM_ID INTEGER)
AS
declare variable os_from_c_id decimal(18,0);
declare variable os_from_ol_id decimal(18,0);
declare variable ol_date date;
declare variable is_active integer;
declare variable os_from_sac_id integer;
declare variable sac_sign integer;
declare variable os_value decimal(18,4) = 0;
declare variable ol_year_month integer;
declare variable c_id_prior decimal(18,0);
declare variable ol_id_prior decimal(18,0);
declare variable is_delete integer;
declare variable sac_id_prior integer;
begin
  c_id_prior = 0;
  ol_id_prior = 0;
  sac_id_prior = 0;
  osm_id = 0;
  if (c_id is null) then
    c_id = 0;
  if (c_id = 0) then
  begin
    delete from opersalmain;
    delete from opersalsaldo;
  end
  else
  begin
    delete from opersalmain
    where osm_from_c_id = :c_id;
    delete from opersalsaldo
    where oss_from_c_id = :c_id;
  end

  if (c_id = 0) then
  begin
    for select os_from_ol_id, os_from_c_id, os_from_sac_id, ol_date, is_delete, sum(os_value)
        from opersalary, operlist
        where os_from_ol_id = ol_id
        group by os_from_ol_id, os_from_c_id, os_from_sac_id, ol_date, is_delete
        into :os_from_ol_id, :os_from_c_id, :os_from_sac_id, :ol_date, :is_delete, :os_value
    do
    begin
      if (sac_id_prior <> os_from_sac_id) then
      begin
        select sac_sign
        from sa_config
        where sac_id = :os_from_sac_id
        into :sac_sign;
        sac_id_prior = os_from_sac_id;
      end
      is_active = 1 - is_delete;

      if ((ol_id_prior <> os_from_ol_id) or (c_id_prior <> os_from_c_id)) then
      begin
        osm_id = osm_id + 1;
        ol_id_prior = os_from_ol_id;
        c_id_prior = os_from_c_id;

        insert into opersalmain (osm_id, osm_from_ol_id, osm_from_c_id, osm_ol_date, osm_ol_is_active, osm_oborot)
        values (:osm_id, :os_from_ol_id, :os_from_c_id, :ol_date, :is_active, 0);
      end

      insert into opersaldetail (osd_from_osm_id, osd_from_sac_id, osd_value, osd_sign)
      values (:osm_id, :os_from_sac_id, :os_value, :sac_sign);
    end
  end
  else
  begin
    for select os_from_ol_id, os_from_c_id, os_from_sac_id, ol_date, is_delete, sum(os_value)
        from opersalary, operlist
        where os_from_ol_id = ol_id and
              os_from_c_id = :c_id
        group by os_from_ol_id, os_from_c_id, os_from_sac_id, ol_date, is_delete
        into :os_from_ol_id, :os_from_c_id, :os_from_sac_id, :ol_date, :is_delete, :os_value
    do
    begin
      if (sac_id_prior <> os_from_sac_id) then
      begin
        select sac_sign
        from sa_config
        where sac_id = :os_from_sac_id
        into :sac_sign;
        sac_id_prior = os_from_sac_id;
      end
      is_active = 1 - is_delete;

      if ((ol_id_prior <> os_from_ol_id) or (c_id_prior <> os_from_c_id)) then
      begin
        osm_id = -1;
        ol_id_prior = os_from_ol_id;
        c_id_prior = os_from_c_id;

        insert into opersalmain (osm_id, osm_from_ol_id, osm_from_c_id, osm_ol_date, osm_ol_is_active, osm_oborot)
        values (:osm_id, :os_from_ol_id, :os_from_c_id, :ol_date, :is_active, 0)
        returning osm_id
        into :osm_id;

      end

      insert into opersaldetail (osd_from_osm_id, osd_from_sac_id, osd_value, osd_sign)
      values (:osm_id, :os_from_sac_id, :os_value, :sac_sign);
    end
  end
  suspend;
end
