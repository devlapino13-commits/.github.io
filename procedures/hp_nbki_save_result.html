CREATE PROCEDURE HP_NBKI_SAVE_RESULT(
    NR_ID INTEGER,
    XML_BLOB BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    BLS_BLOB_SIZE TYPE OF COLUMN BLOBSTORE.BLS_BLOB_SIZE,
    ERROR_TXT STR_255)
RETURNS (
    BLS_ID ID_TYPE)
AS
declare variable un_id id_type;
declare variable bls_caption str_100;
declare variable bls_filename type of column blobstore.bls_filename;
begin
  error_txt = coalesce(error_txt, '');
  bls_blob_size = coalesce(bls_blob_size, 0);
  nr_id = coalesce(nr_id, 0);

  if (error_txt <> '') then
  begin
    update nbki_request nr
    set nr_status = -1,
        nr_result_datetime = current_timestamp,
        nr_result_date = current_date,
        nr_error_txt = ''
    where (nr_id = :nr_id);

    exit;
  end

  if ((xml_blob is null) or (bls_blob_size = 0) or (nr_id = 0)) then
    exit;

  select nr.nr_from_un_id, 'НБКИ отчет по (' || cl.c_name_fio || ')'
  from nbki_request nr
  inner join clients cl on cl.c_id = nr.nr_from_c_id
  where nr.nr_id = :nr_id
  into :un_id, bls_filename;

  bls_caption = coalesce(bls_filename, '');
  if (bls_filename <> '') then
    bls_filename = bls_filename || '.xml';

  execute procedure hp_insert_into_blobstore (:bls_filename, :bls_caption, :xml_blob, :un_id, 2, 0)
  returning_values :bls_id;

  update nbki_request nr
  set nr_status = 1,
      nr_result_datetime = current_timestamp,
      nr_result_date = current_date,
      nr.nr_bls_id = :bls_id
  where (nr_id = :nr_id);

  suspend;
end
