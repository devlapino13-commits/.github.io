CREATE PROCEDURE PRINT_KK_MAIN(
    CMP_ID D_INTEGER)
RETURNS (
    CMP_STR_NO D_STR_CODE,
    CMP_DATE D_DATE,
    CMP_STATUS D_SMALLINT,
    CMP_IS_IPF D_SMALLINT,
    CMP_MEMO D_STR_LONG,
    KK_REGION_NAME D_STR_100,
    KK_NAME_SHORT D_STR_255,
    KK_NAME_FULL D_STR_255,
    KK_REASON D_STR_LONG,
    KK_DECISION D_STR_LONG,
    CHAIRMAN_FIO D_STR_100,
    CHAIRMAN_JOBNAME D_STR_100,
    SECRETARY_FIO D_STR_100,
    SECRETARY_JOBNAME D_STR_100)
AS
declare variable cmp_from_reg_id d_integer;
declare variable cmp_from_cmk_id d_integer;
declare variable cmp_from_cmd_id d_integer;
declare variable c_id d_id;
begin
  select cmp_str_no, cmp_date, cmp_from_reg_id,
         cmp_from_cmk_id, cmp_from_cmd_id, cmp_status, cmp_is_ipf,
         cmp_memo
  from com_packet
  where cmp_id = :cmp_id
  into :cmp_str_no, :cmp_date, :cmp_from_reg_id,
       :cmp_from_cmk_id, :cmp_from_cmd_id, :cmp_status, :cmp_is_ipf, :cmp_memo;

  kk_region_name = '';
  kk_name_short = '';
  kk_name_full = '';
  kk_reason = '';
  kk_decision = '';

  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :cmp_from_reg_id
  into kk_region_name;

  if (cmp_from_cmk_id is not null) then
  begin
    select cmk_code, cmk_name, cmk_reason
    from com_kind
    where cmk_id = :cmp_from_cmk_id
    into :kk_name_short, :kk_name_full, :kk_reason;
  end

  if (cmp_from_cmd_id is not null) then
  begin
    select cmd_memo
    from com_decision
    where cmd_id = :cmp_from_cmd_id
    into kk_decision;
  end

  chairman_fio = '';
  chairman_jobname = '';
  secretary_fio = '';
  secretary_jobname = '';

  for select cms_from_c_id
  from com_staff
  where cms_from_cmp_id = :cmp_id and
        cms_status = 1 -- секретарь кк
  rows 1
  into c_id
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into secretary_fio;

    select jl_name
    from joblist, hr_pstatus
    where ps_from_c_id =: c_id and
          ps_date_begin <= :cmp_date and
          ps_date_end >= :cmp_date and
          ps_from_jl_id = jl_id
    into secretary_jobname;
  end

  for select cms_from_c_id
  from com_staff
  where cms_from_cmp_id = :cmp_id and
        cms_status = 2 -- Председатель кк
  rows 1
  into c_id
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into chairman_fio;

    select jl_name
    from joblist, hr_pstatus
    where ps_from_c_id =: c_id and
          ps_date_begin <= :cmp_date and
          ps_date_end >= :cmp_date and
          ps_from_jl_id = jl_id
    into chairman_jobname;
  end


  suspend;
end
