CREATE PROCEDURE PAYSYS_PRINT_AKT_ITEMS(
    PSM_ID DECIMAL(18,0))
RETURNS (
    PSI_CODE VARCHAR(45),
    PSI_COUNT INTEGER,
    FIRST_PSI_CODE VARCHAR(20),
    LAST_PSI_CODE VARCHAR(20))
AS
declare variable is_card smallint;
declare variable psp_is_card smallint;
declare variable tmp_psi_code varchar(20);
begin
  is_card = 0;
  psi_count = 1;
  for select distinct psp_is_card
      from paysys_list, paysys_item, paysys_project
      where psl_from_psm_id = :psm_id and
            psl_from_psi_id = psi_id and
            psi_from_psp_id = psp_id
      into :psp_is_card
  do
  begin
    if (psp_is_card = 1) then
    begin
      is_card = 1;
      leave;
    end
  end
  first_psi_code = null;
  for select distinct psi_code
      from paysys_list, paysys_item
      where psl_from_psm_id = :psm_id and
            psl_from_psi_id = psi_id and
            psl_sign = -1
      order by psi_code
      into psi_code
  do
  begin
    if (is_card = 1) then -- Номера карт, каждая выводится отдельным suspend
    begin
-- форматирование кода для повышения читаемости: XXXX-XXXX-XXXX-XXXX
      if (char_length(psi_code) = 16) then
        psi_code = left(psi_code, 4) || '-' || substring(psi_code from 5 for 4) || '-' || substring(psi_code from 9 for 4) || '-' || right(psi_code, 4);
      suspend;
    end -- Номера бланков выводим диапазонами XXXXXXX-XXXXXXX
    else -- или по одной, если не входит в диапазон XXXXXXX
    begin
      tmp_psi_code = psi_code;
      if (first_psi_code is null) then
      begin -- Первый встреченный номер - инициализация
        first_psi_code = psi_code;
        last_psi_code = psi_code;
      end
      else
      begin
        if (cast(psi_code as decimal(18,0)) <> cast(last_psi_code as decimal(18,0)) + 1) then
        begin -- Встретили разрыв
          if (first_psi_code = last_psi_code) then
          begin -- Возвращаем один номер
            psi_code = last_psi_code;
            psi_count = cast(last_psi_code as decimal(18,0)) - cast(first_psi_code as decimal(18,0)) + 1;
            suspend;
          end
          else
          begin -- Возвращаем диапазон
            psi_code = first_psi_code || ' - ' || last_psi_code;
            psi_count = cast(last_psi_code as decimal(18,0)) - cast(first_psi_code as decimal(18,0)) + 1;
            suspend;
          end
          first_psi_code = tmp_psi_code;
        end
        last_psi_code = tmp_psi_code;
      end
    end
  end
  if (first_psi_code is not null) then
  begin
    if (first_psi_code = last_psi_code) then
    begin -- Возвращаем один номер
      psi_code = last_psi_code;
      psi_count = 1;
      suspend;
    end
    else
    begin -- Возвращаем диапазон
      psi_code = first_psi_code || ' - ' || last_psi_code;
      psi_count = cast(last_psi_code as decimal(18,0)) - cast(first_psi_code as decimal(18,0)) + 1;
      suspend;
    end
  end
end
