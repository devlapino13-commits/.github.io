CREATE PROCEDURE AAREP_CALC_BONUS_RU_NEW(
    REP_REGION INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    IS_CIRCLE_CLIENT INTEGER)
RETURNS (
    REP_REGION_ID INTEGER,
    REP_FROM_K_ID DECIMAL(18,0),
    REP_FROM_C_ID DECIMAL(18,0),
    REP_KOLICH_BEGIN INTEGER,
    REP_KOLICH_DISB_NEW INTEGER,
    REP_KOLICH_DISB_REPEAT INTEGER,
    REP_KOLICH_CLOSE INTEGER,
    REP_KOLICH_TRANSFER INTEGER,
    REP_KOLICH_END INTEGER,
    REP_DELIQ_IN_MONTH INTEGER,
    REP_DELIQ_AT_END INTEGER,
    REP_BONUS_DISB DECIMAL(18,4),
    REP_BONUS_DELIQ DECIMAL(18,4),
    REP_OFFICE_NAME VARCHAR(100),
    REP_KC_NAME VARCHAR(100),
    REP_FOLDER_NAME VARCHAR(100))
AS
declare variable kol integer;
declare variable sh_date date;
declare variable del_os decimal(18,4);
declare variable del_perc decimal(18,4);
begin
  rep_kolich_begin = 0;
  rep_kolich_close = 0;
  rep_kolich_transfer = 0;
  rep_kolich_end = 0;
  rep_deliq_in_month = 0;
  rep_deliq_at_end = 0;
  for select rep_region_id, rep_from_k_id, rep_from_c_id, rep_office_name, rep_kc_name, rep_folder_name,
             sum(rep_kolich_begin), sum(rep_kolich_disb_new), sum(rep_kolich_disb_repeat), sum(rep_kolich_close),
             sum(rep_kolich_transfer), sum(rep_kolich_end), max(rep_deliq_in_month), max(rep_deliq_at_end)
      from aarep_calc_bonus_ru_tmp_new(:rep_region, :rep_date_begin, :rep_date_end, :is_circle_client)
      where rep_region_id <> 1010100
      group by rep_region_id, rep_from_k_id, rep_from_c_id, rep_office_name, rep_kc_name, rep_folder_name
      into :rep_region_id, :rep_from_k_id, :rep_from_c_id, :rep_office_name, :rep_kc_name, :rep_folder_name,
           :rep_kolich_begin, :rep_kolich_disb_new, :rep_kolich_disb_repeat, :rep_kolich_close, :rep_kolich_transfer,
           :rep_kolich_end, :rep_deliq_in_month, :rep_deliq_at_end
  do
  begin
    rep_kolich_transfer = rep_kolich_end - rep_kolich_begin - rep_kolich_disb_new - rep_kolich_disb_repeat + rep_kolich_close;
    rep_deliq_at_end = 0;
    rep_bonus_deliq = 0;
    rep_bonus_deliq = 0;
    if (rep_kolich_end > 0) then
    begin
      kol = 0;
      for select ok_sh_date, sum(ok_sh_os + ok_os), sum(ok_sh_perc + ok_perc)
          from operkred
          where ok_from_k_id = :rep_from_k_id and
                ok_ol_is_active = 1 and
                ok_ol_date <= :rep_date_end and
                ok_sh_date <= :rep_date_end and
                ok_type < 7 and
                ok_from_c_id = 0
          group by ok_sh_date
          having sum(ok_sh_os + ok_os) > 0 or sum(ok_sh_perc + ok_perc) > 0
          into :sh_date, :del_os, :del_perc
      do
      begin
        kol = cast((:rep_date_end - :sh_date) as integer);
        if (kol > rep_deliq_at_end) then
          rep_deliq_at_end = :kol;
      end
    end
    kol = 0;
    rep_bonus_disb = cast(rep_kolich_disb_new * 250 as decimal(15,2)) + cast(rep_kolich_disb_repeat * 150 as decimal(15,2));
    if (rep_deliq_in_month > kol) then
      kol = rep_deliq_in_month;
    if (rep_deliq_at_end > kol) then
      kol = rep_deliq_at_end;
    if (kol = 1) then
      rep_bonus_deliq = -500.00;
    if (kol = 2) then
      rep_bonus_deliq = -1500.00;
    if (kol >= 3) then
      rep_bonus_deliq = -2500.00;
    suspend;
  end
end
