CREATE PROCEDURE HP_PERSVALUE_DELETE(
    TABLE_NAME D_STR_CODE NOT NULL,
    FIELD_NAME D_STR_CODE NOT NULL,
    TABLE_KEY_ID D_ID NOT NULL)
AS
declare variable pif_uuid d_uuid;
declare variable pda_id d_id;
declare variable pva_is_delete d_bool;
declare variable pin_id d_id;
begin
  if (f_get_db_id() <> 81001) then
    exit;

  select pin_id
  from persinfo
  where pin_object_kind = 0 and
        pin_object_id = :table_key_id
  into pin_id;

  if (pin_id is null) then
    exit;

  for select pif_uuid
  from ux$persfield
  where pif_source = :table_name and
        pif_inactive = 0 and
        (:field_name = '' or pif_field_name = :field_name)
  into pif_uuid
  do
  begin
    pda_id = null;
    pva_is_delete = null;

    select pda_id, pva_is_delete
    from persdata, persvalue
    where pda_from_pif_uuid = :pif_uuid and
          pda_from_pin_id = :pin_id and
          pda_value_id = pva_id
    into pda_id, pva_is_delete;

    if (pva_is_delete = 0 and pda_id is not null) then
    begin
      insert into persvalue (pva_from_pda_id, pva_is_delete)
      values (:pda_id, 1);
    end
  end
end
