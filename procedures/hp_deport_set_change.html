CREATE PROCEDURE HP_DEPORT_SET_CHANGE(
    C_ID D_ID,
    UN_ID D_ID,
    ACTION_CODE D_STR_CODE,
    M_PARAM D_STR_LONG)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable dsl_id d_id;
begin
  -- RFCRU-5875 Редактирование
  res_text = '';

  if (exists(select 1
             from deport_status_list dsl
             where dsl.dsl_from_c_id = :c_id and
                   dsl.dsl_record_status = 1)) then
  begin
  -- Если есть запись в процессе редактирования
    select dsl.dsl_id
    from deport_status_list dsl
    where dsl.dsl_from_c_id = :c_id and
          dsl.dsl_record_status = 1
    into dsl_id;
  end
  else if (exists(select 1
             from deport_status_list dsl
             where dsl.dsl_from_c_id = :c_id and
                   dsl.dsl_record_status = 2)) then
  begin
  -- Если есть сохраненная (актуальная) запись
    select dsl.dsl_id
    from deport_status_list dsl
    where dsl.dsl_from_c_id = :c_id and
          dsl.dsl_record_status = 2
    into dsl_id;

    insert into deport_status_list (dsl_from_c_id, dsl_record_status, dsl_deport_reason, dsl_deport_start_date, dsl_deport_finish_date, dsl_deport_comment, dsl_from_un_id)
    select :c_id, 1, dsl.dsl_deport_reason, dsl.dsl_deport_start_date, dsl.dsl_deport_finish_date, dsl.dsl_deport_comment, :un_id
    from deport_status_list dsl
    where dsl.dsl_id = :dsl_id
    returning (dsl_id)
    into dsl_id;

  end
  else
    insert into deport_status_list (dsl_from_c_id, dsl_record_status, dsl_deport_reason, dsl_deport_start_date, dsl_deport_finish_date, dsl_deport_comment, dsl_from_un_id)
    values (:c_id, 1, coalesce(f_enum_id('deport_reason', 'deport'), 1), current_date, current_date, '', :un_id)
    returning (dsl_id)
    into dsl_id;

  if (coalesce(dsl_id, 0) <> 0) then
  begin
    if (action_code = 'change_deport_reason') then
    begin
      update deport_status_list dsl
         set dsl.dsl_deport_reason = :m_param,
             dsl.dsl_from_un_id = :un_id
      where dsl.dsl_id = :dsl_id;
    end
    if (action_code = 'change_deport_start_date') then
    begin
      update deport_status_list dsl
        set dsl.dsl_deport_start_date = :m_param,
            dsl.dsl_from_un_id = :un_id
      where dsl.dsl_id = :dsl_id;
    end
    if (action_code = 'change_deport_finish_date') then
    begin
      update deport_status_list dsl
        set dsl.dsl_deport_finish_date = :m_param,
            dsl.dsl_from_un_id = :un_id
      where dsl.dsl_id = :dsl_id;
    end
    if (action_code = 'change_deport_comment') then
    begin
      update deport_status_list dsl
        set dsl.dsl_deport_comment = :m_param,
            dsl.dsl_from_un_id = :un_id
      where dsl.dsl_id = :dsl_id;
    end
    res_text = 'Запись изменена';
  end

  suspend;
end
