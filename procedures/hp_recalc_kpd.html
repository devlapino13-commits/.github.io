CREATE PROCEDURE HP_RECALC_KPD
AS
declare variable i d_integer;
declare variable ca_id d_id;
declare variable repayment_obligations decimal(15,2);
declare variable current_repayment_obligations decimal(15,2);
declare variable k_id d_id;
declare variable mac_id d_id;
declare variable ma_id d_id;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable kpd_err d_str_255;
declare variable const_max_index d_integer;
begin
  -- RFCKG-4535 Робот делает пересчет КПД
  -- если repayment_obligations (Погашение других долговых обязательств) из анкеты не совпадает с расчетом по текущей формуле
  i = 0;
  const_max_index = 300; -- количество контрактов для расчета

  for select k.k_from_ma_app_id,
             k.k_id
  from kontstate ks
  inner join kontract k on ks.ks_from_k_id = k.k_id
  where ks.ks_ol_is_delete = 0 and
        ks.ks_state = 1 and
        ks.ks_date_end = '31.12.9999' and
        ks.ks_date_begin >= '24.11.2024' and
        ks.ks_date_begin <= '19.06.2025'
      into ma_id,
           k_id
  do
  begin
    if (i = const_max_index) then    -- так как расчет КПД тяжелый есть ограничение по количеству контрактов
      exit;

    select mac.mac_id,
           mac.mac_from_c_id
    from miniappclient mac
    where mac.mac_from_ma_id = :ma_id and
          mac.mac_amount > 0 rows 1
    into mac_id,
         c_id;

    select ca.ca_id
    from cfp_anketa ca
    where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
          ca.ca_object_id = :mac_id and
          ca.ca_from_c_id = :c_id rows 1
    into ca_id;

    ca_id = coalesce(:ca_id, -1);

    if (ca_id <= 0) then
      continue;

    select kc.kc_id
    from knclient kc
    where kc.kc_from_k_id = :k_id and
          kc.kc_amount_approved > 0 rows 1
    into kc_id;

    select repayment_obligations
    from print_client_analiz(:kc_id)
    into current_repayment_obligations;

    select repayment_obligations
    from hp_calc_repayment_obligations(:ma_id)
    into :repayment_obligations;

    if (current_repayment_obligations <> repayment_obligations) then
    begin

      select kpd_err
      from save_cfp_kpd(:mac_id)       -- Делаем расчет
      into :kpd_err;

      if (coalesce(kpd_err, '') = '') then
        i = i + 1;
    end

  end
end
