  -- rep_command_kind. настройки сообщения хранятся в таблице refdata:
  -- справочник notifymail || '.' || rep_command_kind - строка настроек email
  -- справочник notifysms  || '.' || rep_command_kind - строка настроек sms

  -- refdata.rd_memo    - список получателей
  -- refdata.rd_tag     - срок жизни сообщения
  -- refdata.rd_str_id  - название хранимой процедуры, которая готовит текст сообщения

  -- rep_params - входящий параметры в зависимости от команды
  -- rep_un_id - пользователь

  -- смс после погашения кредита
  db_id = f_get_db_id();

  --RFCKG-2987
  if (db_id = 41701) then

                      k_kredline_end, k_disb_date, k_kontract_no

      where ok_from_ol_id = :rep_params and

      into k_id, ol_date, doc_no, kont_last_date, k_disb_date, k_kontract_no

  -- Определяем проект: М Булак или OiVO  RFCRU-4257
        mcl_project = F_GET_MCL_PROJECT(F_OBJKIND('kontract','id'), :k_id);

      push_body_1 = 'Вы успешно погасили заём. сумма погашения: ' || amount_str || '. ваша квитанция:' || doc_no || '; ' || :date_format_str;

                kc_from_k_id = :k_id
          into c_id
      do

                        main_payment_str || ' сомасында ай сайынғы т�©лем т�©ленді. Артық т�©ленген сома ' ||

      point = f_point(:c_id, current_date);

      if (db_id <> 81001 or f_clientcom_allowed(c_id, 1, 1, 1) = 1) then

          execute procedure mc_send_push('clients', :c_id, 'поздравляем!', :body_mc, 35);

          push_body_2 = 'Заём полностью погашен. Спасибо, что выбираете М Булак:) Удобно и легко оплатить заём можно на сайте компании по ссылке: www.molbulak.ru/p или еще проще в мобильном приложении М Булак. с заботой о Вас М Булак!';
