CREATE PROCEDURE HP_PHONE_CHANGE_DETAILS(
    MPC_ID D_ID,
    UN_ID D_ID)
RETURNS (
    BLANK_JSON D_STR_4096)
AS
declare variable edl_id d_id;
declare variable edb_id d_id;
declare variable mpc_pass1_blob d_id;
declare variable mpc_pass2_blob d_id;
declare variable mpc_client_photo_blob d_id;
declare variable pass_name d_str_100;
declare variable photo_name d_str_100;
declare variable edl_object_name d_str_255;
declare variable edl_object_kind d_integer;
declare variable edl_object_id d_id;
declare variable edl_blank_date d_date;
declare variable edf_print_name d_str_100;
begin
  -- RFCRU-4763
  select mpc.mpc_from_edl_id,
         drlPass.drl_from_bls_1,
         drlPass.drl_from_bls_2,
         drlPhoto.drl_from_bls_1,
         pkPass.pk_name,
         pkPhoto.pk_name
  from mc_phone_change mpc
  inner join docrecoglist drlPass on drlPass.drl_id = mpc.mpc_pass_drl_id
  inner join docrecoglist drlPhoto on drlPhoto.drl_id = mpc.mpc_client_photo_drl_id
  inner join passkind pkPass on pkPass.pk_id = drlPass.drl_from_pk_id
  inner join passkind pkPhoto on pkPhoto.pk_id = drlPhoto.drl_from_pk_id
  where mpc.mpc_id = :mpc_id
  into edl_id,
       mpc_pass1_blob,
       mpc_pass2_blob,
       mpc_client_photo_blob,
       pass_name,
       photo_name;

  -- Заявление на смену номера
  select edb.edb_id,
         edl.edl_object_kind,
         edl.edl_object_id,
         edl.edl_blank_date,
         edf.edf_print_name
  from edoc_list edl
  inner join edoc_blank edb on edl.edl_id = edb.edb_from_edl_id
  inner join edoc_frx edf on edb.edb_from_edf_id = edf.edf_id
  where edl.edl_id = :edl_id
  into edb_id,
       edl_object_kind,
       edl_object_id,
       edl_blank_date,
       edf_print_name;

  select doc_name
  from hp_edoc_object_info(:edl_object_kind, :edl_object_id, :un_id, :edl_blank_date)
  into edl_object_name;

  blank_json = f_format('[{"caption": "%1", "blobs":[%2, %3, %4, %5]}]',
                        :edl_object_name,
                        f_image_item('edoc_blank', :edb_id, :edf_print_name, :un_id), -- бланк
                        f_image_item('blobstore', :mpc_pass1_blob, :pass_name, :un_id), -- паспорт 1
                        f_image_item('blobstore', :mpc_pass2_blob, :pass_name, :un_id), -- паспорт 2
                        f_image_item('blobstore', :mpc_client_photo_blob, :photo_name, :un_id) -- фото клиента
                        );
  suspend;
end
