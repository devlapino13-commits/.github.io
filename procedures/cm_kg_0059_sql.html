CREATE PROCEDURE CM_KG_0059_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_client_pass_no varchar(30);
declare variable pr_code varchar(100);
declare variable pr_product_kind smallint;
declare variable c_id id_type;
declare variable active_balance curr_type;
declare variable k_disb_date date;
declare variable re_k_id id_type;
declare variable re_close_date date;
declare variable sum_delinq_day integer;
declare variable delinq_day integer;
declare variable sum_limit curr_type;
begin
 
  if  (K_ID= 109010040111400101) then exit;
  for
  select pr_code, pr_product_kind, k_disb_date
  from kontract kont, product
  where k_id =:k_id
  and k_from_pr_id = pr_id
  and pr_loan_type = 1
  and pr_code not containing '.IPF.'
  and pr_code not containing '.BUSINESSMAN.'
  and kont.k_refund = 0
  into pr_code, pr_product_kind, k_disb_date
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;
    for
    select kc_from_c_id, kc_amount_approved, c_name_fio, c_pass_no
    from knclient, clients
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    and kc_from_c_id = c_id
    into c_id, kc_amount, res_info_text, kf_client_pass_no
    do
    begin
--    вычисляется активный баланс текущего клиента
      if (pr_code containing '.PAR.') then  --igusev 05.10.2018 ранее было  (pr_code not containing '.express.'). Основние для имзенения: Письмо от Айзад "Fwd: Блок на продукт Экспресс"
      begin
        sum_limit = 100000;
        if (k_disb_date >= '15.11.2019') then sum_limit = 150000;
          active_balance = 0;
          select sum(cast(kc_amount_approved as double precision)/ k_amount_approved * oks_balance)
          from oksummary, kontract, product, knclient
          where :k_disb_date between oks_date_begin and oks_date_end
          and oks_close_state = 0
          and oks_from_k_id = k_id
          and k_from_pr_id = pr_id
          and k_id = kc_from_k_id
          and kc_from_c_id = :c_id
          and kc_amount_approved > 0
          and k_id <> :k_id
          and pr_code not containing '.IPF.'
          and pr_code not containing '.express.'
          into active_balance;
          active_balance = coalesce(active_balance, 0);
    
          if (active_balance + kc_amount > sum_limit) then
          begin
            res_error_text = 'Общий баланс основного и льготного кредитов не может быть более '||round(:sum_limit)||' сом.';
            res_error_is_critical = 1;
            suspend;
          end
      end
      if (pr_code not containing '.express.') then
      begin
        exit;
      end 

      if (pr_code containing '.BIRGE.' and pr_code containing '.EXPRESS.') then
      exit;        --board 1791
--    вычисляем предыдущий цикл клиента
      re_k_id = null;
      re_close_date = null;
      select kc_from_k_id, ks_date_end
      from knclient, kontstate
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id <> :k_id
      and kc_from_k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin < :k_disb_date
      order by ks_date_begin desc
      rows 1
      into re_k_id, re_close_date;
      if (re_k_id is not null) then
      begin
--      вычисляем суммарный просроченный день
        sum_delinq_day = null;
        select sum(distinct ok_ol_date - ok_sh_date) from operkred
        where ok_from_k_id = :re_k_id
        and ok_ol_is_active = 1
        and ok_type in (4,5)
        and ok_os + ok_perc < 0
        and ok_ol_date > ok_sh_date
--        and ok_ol_date <= :k_disb_date
        into sum_delinq_day;
        sum_delinq_day = coalesce(sum_delinq_day,0);
        if (sum_delinq_day > 7) then
        begin
          res_error_text = 'У заемщика по кредиту в МБФ не должно быть просроченных дней более 7 дней(суммарно)';
          res_error_is_critical = 2;
          if(k_id = 109010036692200101) then 
          res_error_is_critical = 1;
          suspend;
        end else
        if (re_close_date > k_disb_date) then
        begin
--        если активный то текущую просрочку тоже нужно прибавлять
--        если оно имеется
          delinq_day = null;
          select :k_disb_date - oks_delinq_date
          from oksummary
          where oks_from_k_id = :re_k_id
          and :k_disb_date between oks_date_begin and oks_date_end
          and oks_close_state = 0
          into delinq_day;
          delinq_day = coalesce(delinq_day,0);
          if (delinq_day < 0) then delinq_day = 0;
          if (sum_delinq_day + delinq_day > 7) then
          begin
            res_error_text = 'У заемщика по кредиту в МБФ не должно быть просроченных дней более 7 дней(суммарно)';
            res_error_is_critical = 2;
            suspend;
          end
        end
      end
    end
  end
end
