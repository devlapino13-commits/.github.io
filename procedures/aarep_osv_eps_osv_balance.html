CREATE PROCEDURE AAREP_OSV_EPS_OSV_BALANCE(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SHOW_ADD_CODE INTEGER = 0)
RETURNS (
    AB_ID ID_TYPE,
    AB_CODE D_STR_CODE,
    DT_SUM CURR_TYPE,
    DT_VAL CURR_TYPE,
    DT_TAX CURR_TYPE,
    KT_SUM CURR_TYPE,
    KT_VAL CURR_TYPE,
    KT_TAX CURR_TYPE,
    B_DT_SUM CURR_TYPE,
    B_DT_VAL CURR_TYPE,
    B_DT_TAX CURR_TYPE,
    B_KT_SUM CURR_TYPE,
    B_KT_VAL CURR_TYPE,
    B_KT_TAX CURR_TYPE,
    E_DT_SUM CURR_TYPE,
    E_DT_VAL CURR_TYPE,
    E_DT_TAX CURR_TYPE,
    E_KT_SUM CURR_TYPE,
    E_KT_VAL CURR_TYPE,
    E_KT_TAX CURR_TYPE,
    AB_NAME D_STR_255)
AS
declare variable has_filter d_smallint;
declare variable ab_acc_type d_smallint;
declare variable b_saldo_sum d_currency;
declare variable b_saldo_val d_currency;
declare variable b_saldo_tax d_currency;
begin
  has_filter = 0;
  if (exists(select *
             from temp_ref_account_rep
             rows 1)) then
    has_filter = 1;

  if (rep_show_add_code is null) then
    rep_show_add_code = 0;

  if (has_filter = 0) then
  begin
    for select ab_code, iif(:rep_show_add_code = 0, ab_id, ab_parent_id), kbs_acc_type,
               sum(iif(kbs_date between :rep_date_begin and :rep_date_end, kbs_dt_sum, 0)),
               sum(iif(kbs_date between :rep_date_begin and :rep_date_end, kbs_dt_val, 0)),
               sum(iif(kbs_date between :rep_date_begin and :rep_date_end, kbs_dt_tax, 0)),
               sum(iif(kbs_date between :rep_date_begin and :rep_date_end, kbs_kt_sum, 0)),
               sum(iif(kbs_date between :rep_date_begin and :rep_date_end, kbs_kt_val, 0)),
               sum(iif(kbs_date between :rep_date_begin and :rep_date_end, kbs_kt_tax, 0)),
               --sum(kbs_bal_dt_sum), sum(kbs_bal_dt_val), sum(kbs_bal_dt_tax),
               --sum(kbs_bal_kt_sum), sum(kbs_bal_kt_val), sum(kbs_bal_kt_tax)
               sum(iif(:rep_date_end between kbs_date and kbs_date_end, kbs_bal_dt_sum, 0)),
               sum(iif(:rep_date_end between kbs_date and kbs_date_end, kbs_bal_dt_val, 0)),
               sum(iif(:rep_date_end between kbs_date and kbs_date_end, kbs_bal_dt_tax, 0)),
               sum(iif(:rep_date_end between kbs_date and kbs_date_end, kbs_bal_kt_sum, 0)),
               sum(iif(:rep_date_end between kbs_date and kbs_date_end, kbs_bal_kt_val, 0)),
               sum(iif(:rep_date_end between kbs_date and kbs_date_end, kbs_bal_kt_tax, 0))
        from kn_balsummary, ref_accbalance
        where kbs_ab_id = ab_id and
              kbs_date <= :rep_date_end and
              kbs_is_extended = :rep_show_add_code
        group by 1, 2, 3
        into :ab_code, :ab_id, :ab_acc_type,
             :dt_sum, :dt_val, :dt_tax, :kt_sum, :kt_val, :kt_tax,
             :e_dt_sum, :e_dt_val, :e_dt_tax, :e_kt_sum, :e_kt_val, :e_kt_tax
        do
    begin
      b_dt_sum = 0;
      b_dt_val = 0;
      b_dt_tax = 0;
      b_kt_sum = 0;
      b_kt_val = 0;
      b_kt_tax = 0;

      b_saldo_sum = e_dt_sum - e_kt_sum - (dt_sum - kt_sum);
      b_saldo_val = e_dt_val - e_kt_val - (dt_val - kt_val);
      b_saldo_tax = e_dt_tax - e_kt_tax - (dt_tax - kt_tax);

      if (ab_acc_type = 0 or (ab_acc_type = 2 and b_saldo_sum >= 0)) then
      begin
        b_dt_sum = b_saldo_sum;
        b_dt_val = b_saldo_val;
        b_dt_tax = b_saldo_tax;
      end
      if (ab_acc_type = 1 or (ab_acc_type = 2 and b_saldo_sum < 0)) then
      begin
        b_kt_sum = -b_saldo_sum;
        b_kt_val = -b_saldo_val;
        b_kt_tax = -b_saldo_tax;
      end
      suspend;
    end
  end else
  begin
    for select acc_from_ab_id, trim(iif(:rep_show_add_code = 0, acc_ab_code, acc_add_code)), sum(dt_sum), sum(dt_val),
               sum(dt_tax), sum(kt_sum), sum(kt_val), sum(kt_tax), sum(b_dt_sum), sum(b_dt_val), sum(b_dt_tax),
               sum(b_kt_sum), sum(b_kt_val), sum(b_kt_tax), sum(e_dt_sum), sum(e_dt_val), sum(e_dt_tax), sum(e_kt_sum),
               sum(e_kt_val), sum(e_kt_tax)
        from aarep_osv_eps_osv_tmp(:rep_date_begin, :rep_date_end, null, 0), ref_account
        where osv_acc_id = acc_id
        group by 1, 2
        into :ab_id, :ab_code, :dt_sum, :dt_val, :dt_tax, :kt_sum, :kt_val, :kt_tax, :b_dt_sum, :b_dt_val, :b_dt_tax,
             :b_kt_sum, :b_kt_val, :b_kt_tax, :e_dt_sum, :e_dt_val, :e_dt_tax, :e_kt_sum, :e_kt_val, :e_kt_tax
    do
    begin
      ab_name = '';
      select ab_name
      from ref_accbalance
      where ab_id = :ab_id
      into ab_name;
      suspend;
    end
  end
end
