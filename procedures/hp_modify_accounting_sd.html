CREATE PROCEDURE HP_MODIFY_ACCOUNTING_SD(
    OL_ID D_ID)
AS
declare variable ol_id1 d_id;
declare variable reg_id d_integer;
declare variable reg_code d_str_10;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable s_find1 d_str_code;
declare variable s_find2 d_str_code;
declare variable s_find3 d_str_code;
declare variable s_repl1 d_str_code;
declare variable s_repl2 d_str_code;
begin
  if (exists(
    select *
    from operlist
    where ol_id = :ol_id
      and ol_oper_type = 2
      and ol_oper_subtype = 24)) then
  begin
    s_find1 = 'Начисление обяз-ва соцфонд';
    s_repl1 = 'Удержание страховых взносов';

    s_find2 = 'Начисление обяз-ва пд.налог';
    s_repl2 = 'Удержание подоходного налога';

    s_find3 = 'Начисление соцфонд';
    ----------------------------------------------------------------------------

    select
      ol_id, ol_from_reg_id, reg_code
    from operlist
      inner join region
        on reg_unicode = ol_from_reg_id
    where
      ol_parent_id = :ol_id
    into
      :ol_id1, :reg_id, :reg_code;

    select
      ob_account_kt
    from operbook where
      ob_from_ol_id = :ol_id1
        and ob_number = 1
    into
      :acc_kt;

    insert into operbook (
      ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
      ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
      ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
      ob_ol_is_active, ob_ol_reg_id)
    select
      :ol_id1, ob_currency, ob_curr_rate, ob_client_dt, :acc_kt,
      ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, overlay(ob_account_kt placing :reg_code from 1),
      ob_client_kt, replace(replace(ob_detail, :s_find1, :s_repl1), :s_find2, :s_repl2), ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
      ob_ol_is_active, :reg_id --ob_ol_reg_id
    from operbook
    where ob_from_ol_id = :ol_id
      and ob_number > 1;
    ----------------------------------------------------------------------------

    select
      ob_account_dt, ob_account_kt
    from operbook
    where ob_from_ol_id = :ol_id
      and ob_number = 1
    into :acc_dt, :acc_kt;

    update operbook set
      ob_account_kt = :acc_dt
    where ob_from_ol_id = :ol_id
      and ob_account_dt = :acc_kt;

    insert into operbook (
      ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
      ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
      ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
      ob_ol_is_active, ob_ol_reg_id)
    select
      ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_kt,
      ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_dt,
      ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
      ob_ol_is_active, ob_ol_reg_id
    from operbook
    where ob_from_ol_id = :ol_id
      and ob_detail starting :s_find3;
  end
end
