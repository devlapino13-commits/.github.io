CREATE PROCEDURE CM_RU_0017_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable k_from_kf_id id_type;
declare variable k_id_last_group id_type;
declare variable disb_date_last_group date;
declare variable pr_code_last_group varchar(20);
declare variable close_date_last_group date;
declare variable kol_cli_last_group integer;
declare variable kol_repay_last_group integer;
declare variable close_date_plan_last_group date;
declare variable KOMMIS_PERC_last_group perc_type;
declare variable is_dosroch_last_group smallint;
declare variable re_kol_cli integer;
declare variable k_kolich_client_approved integer;
declare variable re_k_id id_type;
declare variable kol_last_cli integer;
declare variable kc_from_c_id id_type;
declare variable last_k_id_cli id_type;
declare variable max_delinq_day_last_group integer;
declare variable kol_delinq_last_group integer;
declare variable is_rest_last_group integer;
declare variable pr_code varchar(20);
declare variable not_dosroch smallint;
declare variable k_kommis_perc perc_type;
declare variable kol_doc_cli integer;
begin
  for
  select k_disb_date,
         k_from_kf_id,
         k_kolich_client_approved,
         pr_code,
         k_kommis_perc
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_loan_type = 0        -- только для групповых
  and pr_product_kind <> 2    -- и не параллельных
  and pr_code not containing '.FAMILY'
  into kf_disb_date,
       k_from_kf_id,
       k_kolich_client_approved,
       pr_code,
       k_kommis_perc
  do
  begin
--------------------------------------------------Проверка новым условиям которые работают от 13.04.2015, на уровне группы --------------------
-- вычисляется предыдущий цикл группы основного группового кредита
      k_id_last_group = null;
      disb_date_last_group = null;
      pr_code_last_group = null;
      KOMMIS_PERC_last_group = null;
      select k_id,
             ks_date_begin,
             pr_code,
             ks_date_end,
             k_kolich_client_approved,
             k_kolich_repay,
             k_kredline_end,
             K_KOMMIS_PERC
      from kontstate, kontract, product
      where k_from_kf_id = :k_from_kf_id
      and k_id <> :k_id
      and k_id = ks_from_k_id
      and ks_ol_is_delete = 0
      and ks_state = 1
      and ks_date_begin < :KF_DISB_DATE
      and k_from_pr_id = pr_id
      and pr_product_kind = 1
      order by ks_date_begin desc
      rows 1
      into k_id_last_group, -- ID контракты предыдущего цикла группового кредита, не параллельный и не индив выборка по групповым
           disb_date_last_group,
           pr_code_last_group,
           close_date_last_group,
           kol_cli_last_group,
           kol_repay_last_group,
           close_date_plan_last_group,
           KOMMIS_PERC_last_group
           ;
      if (close_date_last_group < '31.12.9999') then close_date_last_group = close_date_last_group + 1;
    
      if (disb_date_last_group >= '14.04.2015') then
      begin
        if (close_date_plan_last_group - 10 > close_date_last_group) then is_dosroch_last_group = 1;
        else is_dosroch_last_group = 0;
      end else
      begin
        if (dateadd(month, 3, disb_date_last_group) > close_date_last_group) then is_dosroch_last_group = 1;
        else is_dosroch_last_group = 0;
      end
    
      if (KOMMIS_PERC_last_group is null) then KOMMIS_PERC_last_group = 0;
    
    -- Проверка группа повторный или нет
      if (k_id_last_group is null) then
      begin
        k_id_last_group = 0; -- группа новая
    
        -- Кол-во клиентов в группе которые получает повторные займы
        select count(*) from knclient
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
        and kc_circle > 1
        into re_kol_cli;
        -- Если кол-во повторных клиентов не менее 50% группы то проверяет следующее
        if (cast(re_kol_cli as curr_type) >= cast(k_kolich_client_approved as curr_type) / 2 ) then
        begin
          re_kol_cli = null; -- обнуляем чтобы обратно вычислить
    --  Если не менее повторных то проверяет что клиенты не были в одной группе в предыдущем контракте
          with
    --  Вытаскивает всех клиентов у которых были повторные займы
          tabl1 as (
            select kc_from_c_id as c_id,
                   kc_circle as kc_circle_curr
            from knclient
            where kc_from_k_id = :k_id
            and kc_amount_approved > 0
            and kc_circle > 1),
    --  кол-в повторных клиентов в предыдущих контрактов, т.е. группировка по контракту
          tabl2 as (
            select ks_from_k_id,
                   count(*) kol_cli
            from kontstate, knclient, tabl1
            where ks_ol_is_delete = 0
            and ks_state = 1
            and ks_from_k_id = kc_from_k_id
            and kc_amount_approved > 0
            and kc_from_c_id = c_id
            and kc_circle = kc_circle_curr - 1
            group by ks_from_k_id)
          select ks_from_k_id, kol_cli from tabl2
          order by kol_cli desc
          rows 1
          into re_k_id, re_kol_cli;
          if (re_kol_cli is null) then re_kol_cli = 0;
    --    если клиенты в предыдущем цикле были в одном контракте и кол-во таких клиентов не менне 50% текущей группы
    --    то группа считется повторной
          if (cast(re_kol_cli as curr_type) >= cast(k_kolich_client_approved as curr_type)/2) then
          begin
    --          k_id_last_group = re_k_id;
              KOMMIS_PERC_last_group = null;
              select k_id,
                     ks_date_begin,
                     pr_code,
                     ks_date_end,
                     k_kolich_client_approved,
                     k_kolich_repay,
                     k_kredline_end,
                     K_KOMMIS_PERC
              from kontstate, kontract, product
              where k_id = :re_k_id
              and k_id = ks_from_k_id
              and ks_ol_is_delete = 0
              and ks_state = 1
              and k_from_pr_id = pr_id
              and pr_product_kind = 1
              into k_id_last_group, -- ID контракты предыдущего цикла группового кредита, не параллельный и не индив выборка по групповым
                   disb_date_last_group,
                   pr_code_last_group,
                   close_date_last_group,
                   kol_cli_last_group,
                   kol_repay_last_group,
                   close_date_plan_last_group,
                   KOMMIS_PERC_last_group
                   ;
    
            if (close_date_last_group < '31.12.9999') then close_date_last_group = close_date_last_group + 1;
    
            if (disb_date_last_group >= '14.04.2015') then
            begin
                if (close_date_plan_last_group - 10 > close_date_last_group) then is_dosroch_last_group = 1;
                else is_dosroch_last_group = 0;
            end else
            begin
                if (dateadd(month, 3, disb_date_last_group) > close_date_last_group) then is_dosroch_last_group = 1;
                else is_dosroch_last_group = 0;
            end
    
            if (KOMMIS_PERC_last_group is null) then KOMMIS_PERC_last_group = 0;
          end else k_id_last_group = 0;
        end
    
      end else
      begin
    --  вычисляет из старого состава группы сколько клиентов остались
    --  у каждого клиента проверяется предыдущий групповой цикл кредита
        kol_last_cli = 0;
        for
        select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
        into kc_from_c_id
        do
        begin
          last_k_id_cli = null;
          select k_id
          from kontstate, kontract, product, knclient
          where k_id <> :k_id
          and k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and ks_date_begin < :KF_DISB_DATE
          and k_from_pr_id = pr_id
          and pr_product_kind = 1
          and k_id = kc_from_k_id
          and kc_from_c_id = :kc_from_c_id
          and kc_amount_approved > 0 -- добавлено от 07.05.2015 Улан
          order by ks_date_begin desc
          rows 1
          into last_k_id_cli;
          if (last_k_id_cli is null) then last_k_id_cli = 0;
          if (last_k_id_cli = k_id_last_group) then kol_last_cli = kol_last_cli + 1;
        end
    --  если кол-во клиентов с предыдущим циклом не совпадает то проверяется на следующее
        if (kol_last_cli <> k_kolich_client_approved) then
        begin
    --  из старого состава не менее 50% и после закрытия не должно пройти 6 месяцев
          if (cast(kol_last_cli as curr_type) < cast(k_kolich_client_approved as curr_type) / 2
              or close_date_last_group < dateadd(year, -6, :kf_disb_date)
          ) then
          begin
            k_id_last_group = 0; -- значить новый контракт будет а не повторный
          end 
        end
      end
    -- вычисляется просроченные дни предыдущего контракта
      max_delinq_day_last_group = 0;
      kol_delinq_last_group = 0;
      is_rest_last_group = 0;
      if (k_id_last_group > 0) then
      begin
        select max(ok_ol_date - ok_sh_date),
               count(*)
        from operkred
        where ok_from_k_id = :k_id_last_group
        and ok_ol_is_active = 1
        and ok_from_c_id = 0
        and ok_ol_date > ok_sh_date
        and ok_type in (4,5,9)
        and ok_os + ok_perc < 0
        into max_delinq_day_last_group, kol_delinq_last_group;
        if (max_delinq_day_last_group is null) then max_delinq_day_last_group = 0;
        if (kol_delinq_last_group is null) then kol_delinq_last_group = 0;
    
        if (disb_date_last_group >= cast('13.04.2015' as date)) then -- применяется новым группам
        begin
            if (exists(select * from operrest, operlist
                       where or_from_k_id = :k_id_last_group
                       and or_from_ol_id = ol_id
                       and operlist.is_delete = 0
                       and current_date between ol_date and or_date_end
                       and or_kind > 0
                       )) then is_rest_last_group = 1;
        end
      end
    
      if (pr_code containing '.PARTNER.') then
      begin
        if (k_id_last_group = 0) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Группа новая и не подлежит по тарифу Партнер';
          suspend;
        end 
    
        if (pr_code_last_group containing '.KEL.') then
        begin
          if (is_dosroch_last_group = 1) then
          begin
            res_error_is_critical = 1;
            if ((select ma_date_app
                from miniapp, kontract
                where ma_id = k_from_ma_app_id
                and k_id = :k_id
                order by ma_date_app
                rows 1) between cast('10.04.2015' as date) and cast('13.04.2015' as date)) then res_error_is_critical = 0;
    
            res_error_text = 'Предыдущий цикл закрыли досрочно и не имеет право перейти на следующий тариф "Партнер"';
            suspend;
          end
        end
        if (pr_code_last_group containing '.PARTNER.') then
        begin
            if ((max_delinq_day_last_group between 1 and 3) and (kol_delinq_last_group < 2)) then
            begin
              res_error_is_critical = 1;
              res_error_text = 'В предыдущем цикле имется просрочка до 3х дней, группа подлежит по "Кел Кел"';
              suspend;
            end else
            if (max_delinq_day_last_group > 3 or (kol_delinq_last_group > 1)) then
            begin
              res_error_is_critical = 1;
              res_error_text = 'В предыдущем цикле имется просрочка более 3х дней, группа подлежит по "Ассалам"';
              suspend;
            end 
        end else
        if (max_delinq_day_last_group > 0) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'В предыдущем цикле имется просрочка, группа не может получить по тарифу "Партнер"';
          suspend;
        end else -- если в предыдущем цикле брал по продукту ассалам то проверяется на досрочные погашении
        if (pr_code_last_group containing '.ASALAM.') then
        begin
    --  проверка на досрочное погашение
          not_dosroch = 0;
          if (exists(select ok_ol_date, min(ok_sh_date)
              from operkred
              where ok_from_k_id = :k_id_last_group
              and ok_ol_is_active = 1
              and ok_type = 4
              and ok_os + ok_perc <> 0
              group by ok_ol_date
              having (ok_ol_date >= min(ok_sh_date)
    --          or ok_ol_date <  min(ok_sh_date) - 10 -- досрочно погашенные заранее свыше 10ти дней не будут включаться
                    ))) then not_dosroch = 1;
    
          if (not_dosroch = 0) then
          begin
    --       если контракт закрылся досрочно за 10 дней то не должно перейти с тарифа Асалам на Партнер
             if (exists(
                select * from kontstate
                where ks_ol_is_delete = 0
                and ks_state = 2
                and ks_from_k_id = :k_id_last_group
                and ks_date_begin < (select max(ok_sh_date) from operkred
                                     where ok_ol_is_active = 1
                                     and ok_from_k_id = :k_id_last_group
                                     and ok_type in (1,2,3)) - 10
                )) then
             begin
               res_error_is_critical = 1;
               res_error_text = 'Предыдущий цикл закрылся досрочно за более 10 дней, и не переходит к тарифу "Партнер"';
               suspend;
             end 
          end
    
          if (not_dosroch = 1) then
          begin
            res_error_is_critical = 1;
            res_error_text = 'В предыдущем цикле ни все платежы досрочно погашены и не подлежит к тарифу "Партнер"';
            suspend;
          end else
          if (kol_repay_last_group not in (6,9)) then
          begin
            res_error_is_critical = 1;
            res_error_text = 'Кол-во погашений предыдущего цикла Ассалам не позволяет брать по тарифу "Партнер"';
            suspend;
          end
        end
    
        if (is_rest_last_group = 1) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'В предыдущем цикле была реструктуризация, не имеют доступ к тарифу "Партнер"';
          suspend;
        end 
      end
    
      if (pr_code containing '.KEL.') then
      begin
        if (k_id_last_group = 0) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Группа новая и не подлежит по тарифу "Кел Кел"';
          suspend;
        end
        if (max_delinq_day_last_group > 0 and (pr_code_last_group containing '.ASALAM.' or pr_code_last_group containing '.KEL.')) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'В предыдущем цикле имеется просрочка, группа не может получить по тарифу "Кел Кел"';
          suspend;
        end
        if (is_dosroch_last_group = 1 and pr_code_last_group containing '.ASALAM.') then
        begin
          res_error_is_critical = 1;
          if ((select ma_date_app
                from miniapp, kontract
                where ma_id = k_from_ma_app_id
                and k_id = :k_id
                order by ma_date_app
                rows 1) between cast('10.04.2015' as date) and cast('13.04.2015' as date)) then res_error_is_critical = 0;
          res_error_text = 'Предыдущий цикл закрыли досрочно и не имеет право перейти на следующий тариф "Кел Кел"';
          suspend;
        end
        if (is_rest_last_group = 1 and pr_code_last_group not containing '.PARTNER.') then
        begin
          res_error_is_critical = 1;
          res_error_text = 'В предыдущем цикле была реструктуризация, не имеют доступ к тарифу "Кел Кел"';
          suspend;
        end 
      end
    
      if (max_delinq_day_last_group > 7 and K_KOMMIS_PERC < 3.0000 and disb_date_last_group >= '10.06.2014') then
      begin
        res_error_is_critical = 1;
        res_error_text = 'В предыдущем цикле были просрочки более 7 дней, доп.% должно быть = 3%';
        suspend;
      end else
      if (max_delinq_day_last_group > 0 and K_KOMMIS_PERC < 2.0000 and disb_date_last_group >= '10.06.2014') then
      begin
        res_error_is_critical = 1;
        res_error_text = 'В предыдущем цикле были просрочки более 1 дня, доп.% должно быть = 2%';
        suspend;
      end else
    -- если кредит без коммисии то проверяется каждый клиент на следующие документы
      if (K_KOMMIS_PERC = 0.0000) then
      begin
    --  проверка доп процента предыдущего цикла
        if (KOMMIS_PERC_last_group >= 2) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'В предыдущем цикле доп% был повышен из-за просрочки и не могут брать под нулевым доп.% ';
          suspend;
        end else
    --  проверка каждого клиента, имеется ли три необходимые документы. Пункт 4,2
        begin
          for
          select kc_from_c_id
          from knclient
          where kc_from_k_id = :k_id
          and kc_amount_approved > 0
          into kc_from_c_id
          do
          begin
            select count(distinct case
                                     when pdt_code = 'certificate_annulment' then 'certificate_marriage'
                                     when pdt_code = 'diplom_suz' then 'diplom_vuz'
                                     else pdt_code
                                   end)
            from CLIENTDOCS, PASSKIND, PASSDOCTYPE
            where CDS_FROM_C_ID = :kc_from_c_id
            and cds_from_pk_id = pk_id
            and pk_from_pdt_id = pdt_id
            and cds_is_delete = 0
    --        and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
            and (pdt_code in ('certificate_driver', -- Водительское удостоверение
                             'certificate_birth', -- Свидетельство о рождении
                             'certificate_marriage', -- Свидетельство о заключении брака
                             'certificate_annulment', -- Свидетельство о расторжении брака
                             'diplom_vuz', -- Диплом ВУЗ
                             'diplom_suz', -- Диплом СУЗ
                             'diplom_school', -- Аттестат
    --                         'work_permit',  -- Разрешение на работу
                             'military_id', -- Военный билет
                             'snils',  --  Страховое свидетельство обязательного пенсионного страхования (СНИЛС) 
                             'certificate_tax' -- Свидетельство о постановке на учет в налоговом органе физического лица (ИНН)
                              )
                  or (pdt_code = 'work_permit' -- Разрешение на работу
                      and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END)
                      )
            into kol_doc_cli;
            if (kol_doc_cli < 3) then
            begin
              select c_name_fio, c_pass_no from clients
              where c_id = :kc_from_c_id
              into res_info_text, KF_CLIENT_PASS_NO;
              res_error_is_critical = 1;
              res_error_text = 'У клиента не имеются трех дополнительных документов, не может получить под нулевым доп.% ';
              suspend;
            end 
          end
        end
      end
    
  end
end
