CREATE PROCEDURE HP_PERSTEMP_RESEND_STAFF(
    IN_C_ID D_ID)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable un_id d_id;
declare variable c_name_fio d_str_100;
declare variable usr_id d_id;
begin
  res_is_error = 0;
  res_text = 'Тест: Уведомление сотруднику успешно отправлено:';

  select pte_usr_added
  from perstemp
  where pte_from_c_id = :in_c_id
  into usr_id;

  un_id = uxf$usr_un(usr_id, 0);

  if (un_id is null) then
  begin
    res_is_error = 1;
    res_text = 'Инициатор не сотрудник, отправка уведомления отменена';
  end else
  begin
    select un_fio
    from username
    where un_id = :un_id
    into c_name_fio;

    res_text = res_text || ascii_char(13) || c_name_fio;
  end

  suspend;
end
