CREATE PROCEDURE HP_UXDEPEND_FILL_TREE(
    UX_CODE D_STR_CODE,
    OWNER_CODE D_STR_CODE,
    PARENT_CODE D_STR_255,
    CURR_LEVEL D_INTEGER,
    MAX_LEVEL D_INTEGER)
AS
declare variable res_kind d_str_code;
declare variable res_image d_integer;
declare variable res_code d_str_100;
declare variable tmp_ux_code d_str_code;
declare variable owner_table d_str_code;
declare variable ua_code d_str_100;
declare variable lock_delete d_bool;
declare variable tmp_parent_code d_str_255;
declare variable uxa_kind d_smallint;
begin
  if (curr_level is null) then
    curr_level = 0;

  if (max_level is null) then
    max_level = 1;

  curr_level = curr_level + 1;

  if (curr_level > max_level) then
    exit;

  if (coalesce(parent_code, '') = '') then
    parent_code = null;

  if (coalesce(owner_code, '') = '') then
    owner_code = ux_code;

  select ux_table_name
  from uxcode
  where ux_code = :owner_code
  into :owner_table;

  for select iif(:owner_table <> ux_table_name, 1, 0), uxf$kind(ux_kind, 'code')
  from uxcode
  where ux_code = :ux_code
  into lock_delete, res_kind
  do
  begin
    select res_image_id
    from hp_uxkind_image_id(:res_kind, :lock_delete)
    into :res_image;

    res_code = null;

    select udp_code
    from uxdepend
    where udp_owner_code = :owner_code and
          udp_ux_code = :ux_code
    into res_code;

    if (res_code is null) then
    begin
      insert into uxdepend (udp_owner_code, udp_parent_code, udp_ux_code, udp_ux_kind, udp_image_id, udp_lock_delete)
      values (:owner_code, :parent_code, :ux_code, :res_kind, :res_image, :lock_delete)
      returning udp_code
      into :res_code;
    end else
    begin
      if (lock_delete = 1) then
      begin
        update uxdepend
        set udp_lock_delete = 1
        where udp_owner_code = :owner_code and
              udp_ux_code = :ux_code;
      end

      exit;
    end

    if (res_kind = 'uxframe') then
    begin
      for select ucf_ux_code
      from uxcodeframe
      where ucf_ufr_code = :ux_code
      into tmp_ux_code
      do
      begin
        execute procedure hp_uxdepend_fill_tree(:tmp_ux_code, :owner_code, :res_code, :curr_level, :max_level);
      end
    end else
    begin
      lock_delete = 1;
      for select ua_code, uxa_ux_code, uxa_kind,
                 iif(ua_ux_code = 'fm_mainmenu', 1, iif(ua_usr_id_field <> '', 1, 0))
      from uxaction, uxalias
      where uxa_ux_code = :ux_code and
            uxa_code = ua_uxa_code and
            ua_ux_code = 'fm_mainmenu'
      into :ua_code, :tmp_ux_code, :uxa_kind, :lock_delete
      do
      begin
        if (not exists(select udp_code
        from uxdepend
        where udp_owner_code = :owner_code and
              udp_ux_code = :ua_code)) then
        begin
          select res_image_id
          from hp_uxkind_image_id('uxaction', 'mainmenu')
          into :res_image;

          insert into uxdepend (udp_owner_code, udp_parent_code, udp_ux_code, udp_ux_kind, udp_image_id, udp_lock_delete)
          values (:owner_code, :res_code, :ua_code, 'uxaction', :res_image, :lock_delete);
        end
      end

      lock_delete = 0;
      for select distinct ua_ux_code
      from uxaction, uxalias
      where uxa_ux_code = :ux_code and
            uxa_code = ua_uxa_code and
            ua_ux_code <> 'fm_mainmenu'
      into :tmp_ux_code
      do
      begin
        execute procedure hp_uxdepend_fill_tree(:tmp_ux_code, :owner_code, :res_code, :curr_level, :max_level);

        tmp_parent_code = null;
        select udp_code
        from uxdepend
        where udp_owner_code = :owner_code and
              udp_ux_code = :tmp_ux_code
        into tmp_parent_code;

        if (tmp_parent_code is null) then
          tmp_parent_code = res_code;

        for select ua_code, iif(ua_usr_id_field <> '', -1, uxa_kind)
        from uxaction, uxalias
        where uxa_ux_code = :ux_code and
              uxa_code = ua_uxa_code and
              ua_ux_code = :tmp_ux_code
        into :ua_code, :uxa_kind
        do
        begin
          if (not exists(select udp_code
          from uxdepend
          where udp_owner_code = :owner_code and
                udp_ux_code = :ua_code)) then
          begin
            select res_image_id
            from hp_uxkind_image_id('uxaction', :uxa_kind)
            into :res_image;

            insert into uxdepend (udp_owner_code, udp_parent_code, udp_ux_code, udp_ux_kind, udp_image_id, udp_lock_delete)
            values (:owner_code, :tmp_parent_code, :ua_code, 'uxaction', :res_image, :lock_delete);
          end
        end
      end

      for select ucf_ufr_code
      from uxcodeframe
      where ucf_ux_code = :ux_code
      into tmp_ux_code
      do
        execute procedure hp_uxdepend_fill_tree(:tmp_ux_code, :owner_code, :res_code, :curr_level, :max_level);
    end
  end
end
