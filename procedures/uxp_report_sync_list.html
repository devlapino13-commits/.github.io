CREATE PROCEDURE UXP$REPORT_SYNC_LIST
RETURNS (
    RES_ID D_INTEGER,
    RES_PARENT_ID D_INTEGER,
    RES_REP_ID D_INTEGER,
    RES_CAPTION D_STR_255,
    RES_HINT D_STR_255,
    RES_IMPORT_KIND D_SMALLINT,
    RES_REPORT_KIND D_STR_CODE,
    RES_POSITION D_INTEGER,
    RES_URT_UUID_STR D_STR_CODE,
    RES_US_UUID_STR D_STR_CODE,
    RES_UX_UUID_STR D_STR_CODE,
    RES_UA_UUID D_UUID,
    RES_IMPORT_STATUS D_SMALLINT,
    RES_IMAGE_INDEX D_INTEGER,
    RES_SOURCE_LIST D_STR_LONG,
    RES_RL_SQL D_STR_LONG,
    RES_IS_ERROR D_BOOL,
    RES_ERROR_TEXT D_STR_LONG)
AS
declare variable urt_uuid d_uuid;
declare variable ux_uuid d_uuid;
declare variable mm_uuid d_uuid;
declare variable uxa_fcode d_str_100;
declare variable rl_report_kind d_smallint;
declare variable res_is_folder d_bool;
declare variable us_code d_str_code;
declare variable urt_import_datetime d_datetime;
declare variable rl_datetime d_datetime;
declare variable rl_is_excel d_bool;
declare variable is_frx d_bool;
begin
  mm_uuid = uxf$mainmenu();

  res_import_kind = 1;
  for select rl_id, parent, iif(coalesce(rl_type, 1) = 1, 1, 0),
             rl_name_rus, rl_hint_rus, rl_report_kind, rl_order_no, rl_sql,
             iif(rl_report_kind in (1, 2) and rl_data_rus is not null, 1, 0), rl_datetime
  from reportlist
  order by parent, rl_order_no, rl_id
  into :res_id, :res_parent_id, :res_is_folder,
       :res_caption, :res_hint, :rl_report_kind, :res_position, :res_rl_sql,
       :rl_is_excel, :rl_datetime
  do
  begin
    res_urt_uuid_str = '';
    res_ux_uuid_str = '';
    res_us_uuid_str = '';
    res_ua_uuid = null;
    res_report_kind = '';

    res_is_error = 0;
    res_error_text = '';

    select rim_import_text
    from ux$repimport
    where rim_kind = :res_import_kind and
          rim_id = :res_id
    into res_error_text;

    if (res_error_text <> '') then
      res_is_error = 1;

    res_rep_id = res_id;

    if (res_is_folder = 1) then
      res_image_index = 119;  -- папка
    else
      res_image_index = 101; -- маленький квадрат

    res_import_status = 0;
    if (res_is_folder = 0) then
      res_import_status = -1;

    if (rl_report_kind in (1, 2)) then
    begin
      res_report_kind = 'Таблица';
      res_image_index = 85;

      if (rl_is_excel = 1) then
      begin
        res_report_kind = 'Excel';
        res_image_index = 123;
      end

      res_import_status = 1;
    end
    else
    if (rl_report_kind in (3, 4)) then
    begin
      res_report_kind = 'FastReport';
      res_image_index = 29;
      res_import_status = 1;
    end

    urt_uuid = null;
    urt_import_datetime = null;
    select urt_uuid, urt_import_datetime
    from ux$report
    where urt_import_kind = :res_import_kind and
          urt_import_id = :res_rep_id
    into :urt_uuid, urt_import_datetime;

    res_source_list = '';
    us_code = '';

    if (urt_uuid is not null) then
    begin
      res_import_status = 2;
      res_urt_uuid_str = uuid_to_char(urt_uuid);

      if (rl_datetime <> urt_import_datetime) then
        res_import_status = 3; -- Изменен

      select ufm_ux_uuid, uuid_to_char(ux_us_uuid), iif(us_kind <> -1, us_code, '<SQL>')
      from ux$form, ux$code, ux$source
      where ufm_uuid = :urt_uuid and
            ufm_ux_uuid = ux_uuid and
            ux_us_uuid = us_uuid
      into ux_uuid, res_us_uuid_str, us_code;

      res_ux_uuid_str = uuid_to_char(ux_uuid);

      if (us_code <> 'reportlist' and position('[' || us_code || ']', res_source_list) = 0) then
        res_source_list = res_source_list || '[' || us_code || ']';

      for select trim(iif(us_kind <> -1, us_code, '<SQL>'))
      from ux$frame, ux$source
      where ufr_owner = :urt_uuid and
            ufr_us_uuid = us_uuid
      into us_code
      do
      begin
        if (position('[' || us_code || ']', res_source_list) = 0) then
          res_source_list = res_source_list || '[' || us_code || ']';
      end

      res_source_list = replace(res_source_list, '[', '');
      res_source_list = trim(replace(res_source_list, ']', ', '));
      if (right(res_source_list, 1) = ',') then
        res_source_list = left(res_source_list, char_length(res_source_list) - 1);

      select uxa_fcode
      from ux$alias
      where uxa_owner = :ux_uuid
      into uxa_fcode;

      select ua_uuid
      from ux$action
      where ua_uxa_fcode = :uxa_fcode and
            ua_owner = :mm_uuid and
            ua_usr_id_field = ''
      into res_ua_uuid;
    end

    suspend;
  end

  res_import_kind = 2;
  res_is_error = 0;
  res_error_text = '';

  res_source_list = '';
  res_rl_sql = '';

  res_rep_id = null;

  res_id = -1;
  res_parent_id = null;
  res_is_folder = 1;
  res_caption = 'Служебные бланки';
  res_hint = '';
  res_urt_uuid_str = '';
  res_ux_uuid_str = '';
  res_us_uuid_str = '';
  res_ua_uuid = null;
  res_report_kind = '';
  res_position = 1;
  res_import_status = 0;
  res_image_index = 119;
  suspend;

  res_id = -2;
  res_parent_id = null;
  res_is_folder = 1;
  res_caption = 'Бланки';
  res_hint = '';
  res_urt_uuid_str = '';
  res_ux_uuid_str = '';
  res_us_uuid_str = '';
  res_ua_uuid = null;
  res_report_kind = '';
  res_position = 1;
  res_import_status = 0;
  res_image_index = 119;
  suspend;

  res_is_folder = 0;
  res_hint = '';

  for select bln_id, bln_name, bln_show_mode
  from blankname
  order by bln_id
  into :res_rep_id, :res_caption, rl_report_kind
  do
  begin
    res_id = res_rep_id + 1000000;

    if (rl_report_kind <> 0) then
    begin
      is_frx = 0;
      select iif(blb_frx_blob is not null, 1, 0)
      from blankblob
      where blb_from_bln_id = :res_rep_id
      order by blb_date_begin desc
      rows 1
      into is_frx;

      if (is_frx = 1) then
        rl_report_kind = 0;
    end

    if (res_rep_id < 1000) then
      res_parent_id = -1;
    else
      res_parent_id = -2;

    res_position = res_id;

    res_urt_uuid_str = '';
    res_ux_uuid_str = '';
    res_us_uuid_str = '';
    res_ua_uuid = null;
    res_report_kind = '';

    res_is_error = 0;
    res_error_text = '';
    rl_datetime = null;

    select rim_import_text
    from ux$repimport
    where rim_kind = :res_import_kind and
          rim_id = :res_rep_id
    into res_error_text;

    if (res_error_text <> '') then
      res_is_error = 1;

    res_image_index = 101;
    res_import_status = -1;

    if (rl_report_kind = 0) then
    begin
      res_report_kind = 'FastReport';
      res_image_index = 29;
      res_import_status = 1;
    end else
    if (rl_report_kind = 1) then
      res_import_status = 0; -- "Не поддерживается"
    else
    if (rl_report_kind = 2) then
    begin
      res_report_kind = 'Excel';
      res_image_index = 123;
      res_import_status = 0;
    end

    urt_uuid = null;
    select urt_uuid, urt_import_datetime
    from ux$report
    where urt_import_kind = :res_import_kind and
          urt_import_id = :res_rep_id
    into :urt_uuid, urt_import_datetime;

    res_source_list = '';
    us_code = '';

    if (urt_uuid is not null) then
    begin
      res_import_status = 2;
      res_urt_uuid_str = uuid_to_char(urt_uuid);

      rl_datetime = null;
      select blb_datetime
      from blankblob
      where blb_from_bln_id = :res_rep_id and
           (blb_frx_blob is not null or blb_blob is not null)
      order by blb_date_begin desc
      rows 1
      into rl_datetime;

      if (rl_datetime <> urt_import_datetime) then
        res_import_status = 3; -- Изменен

      select ufm_ux_uuid, uuid_to_char(ux_us_uuid), iif(us_kind <> -1, us_code, '<SQL>')
      from ux$form, ux$code, ux$source
      where ufm_uuid = :urt_uuid and
            ufm_ux_uuid = ux_uuid and
            ux_us_uuid = us_uuid
      into ux_uuid, res_us_uuid_str, us_code;

      if (us_code <> 'reportlist' and position('[' || us_code || ']', res_source_list) = 0) then
        res_source_list = res_source_list || '[' || us_code || ']';

      for select iif(us_kind <> -1, us_code, '<SQL>')
      from ux$frame, ux$source
      where ufr_owner = :urt_uuid and
            ufr_us_uuid = us_uuid
      into us_code
      do
      begin
        if (position('[' || us_code || ']', res_source_list) = 0) then
          res_source_list = res_source_list || '[' || us_code || ']';
      end

      res_source_list = replace(res_source_list, '[', '');
      res_source_list = trim(replace(res_source_list, ']', ', '));
      if (right(res_source_list, 1) = ',') then
        res_source_list = left(res_source_list, char_length(res_source_list) - 1);

      res_ux_uuid_str = uuid_to_char(ux_uuid);

      select uxa_fcode
      from ux$alias
      where uxa_owner = :ux_uuid
      into uxa_fcode;

      select ua_uuid
      from ux$action
      where ua_uxa_fcode = :uxa_fcode and
            ua_owner = :mm_uuid and
            ua_usr_id_field = ''
      into res_ua_uuid;
    end

    suspend;
  end
end
