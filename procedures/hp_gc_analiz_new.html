CREATE PROCEDURE HP_GC_ANALIZ_NEW(
    GC_FNAME D_STR_100,
    GC_DATE D_DATE,
    GC_DATE_BEGIN D_DATE,
    GC_DATE_END D_DATE,
    GC_BLOB D_BLOB_BINARY,
    GC_SUMM D_CURRENCY,
    DB_SUMM D_CURRENCY,
    UN_ID D_ID)
RETURNS (
    IFA_ID D_INTEGER,
    IGA_ID D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable bls_id d_id;
declare variable iqp_id d_id;
declare variable cs_id d_id;
declare variable c_id d_id;
declare variable ag_id d_integer;
declare variable iqp_amount d_currency;
declare variable commis_perc d_percent;
declare variable commis_amount d_currency;
declare variable rd_id_tsp d_integer;
begin
  err_text = '';
  rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');

  execute procedure hp_insert_into_blobstore (
    :gc_fname,
    'GC-файл',
    :gc_blob,
    :un_id,
    2,
    0
  )
  returning_values :bls_id;

  insert into ipc_file_analiz(
    ifa_file_name,
    ifa_file_type,
    ifa_date,
    ifa_from_bls_id,
    ifa_from_un_id
  )
  values (
    :gc_fname,
    1,
    :gc_date,
    :bls_id,
    :un_id
  )
  returning ifa_id
  into :ifa_id;

  insert into ipc_gc_analiz(
    iga_from_ifa_id,
    iga_date_begin,
    iga_date_end,
    iga_gc_qr_summ,
    iga_db_qr_summ
  )
  values (
    :ifa_id,
    :gc_date_begin,
    :gc_date_end,
    :gc_summ,
    :db_summ
  )
  returning iga_id
  into :iga_id;
  for
    select
      iqp_id,
      iqp_from_cs_id,
      cs_from_c_id,
      iqp_amount
    from ipc_qr_payments
      inner join clientshops
        on cs_id = iqp_from_cs_id and
           cs_from_rd_id = :rd_id_tsp
    where iqp_date between :gc_date_begin and :gc_date_end
  into
    :iqp_id,
    :cs_id,
    :c_id,
    :iqp_amount
  do
  begin
    commis_perc = 0;
    commis_amount = 0;

    select first 1
      ag_id,
      ag_commis_cards_other
    from agreements
    where ag_object_kind = 0
      and ag_object_id = :c_id
      and ag_from_rd_id = f_get_rd_id('AgreementKinds.tsp')
      and ag_status = 1
      and :gc_date between coalesce(ag_date_begin, '01.01.1900') and coalesce(ag_date_end, '31.12.9999')
    order by ag_id desc
    into
      :ag_id,
      :commis_perc;
    commis_amount = trunc(commis_perc * iqp_amount / 100.0, 2);

    insert into ipc_gc_analiz_qr_payments(
      igqp_from_iga_id,
      igqp_from_iqp_id,
      igqp_from_cs_id,
      igqp_from_c_id,
      igqp_from_ag_id,
      igqp_amount,
      igqp_tsp_commis_perc,
      igqp_tsp_commis_amount
    )
    values (
      :iga_id,
      :iqp_id,
      :cs_id,
      :c_id,
      :ag_id,
      :iqp_amount,
      :commis_perc,
      :commis_amount
    );
  end

  suspend;
end
