CREATE PROCEDURE HP_CHECK_ALL_KONTRACT(
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    CHECK_MEMO VARCHAR(255),
    OK_OL_DATE DATE,
    CHECK_MEMO_ALL VARCHAR(8000),
    K_ID VARCHAR(18),
    OK_FROM_OL_ID VARCHAR(18),
    ESM_ID ID_TYPE,
    LINK VARCHAR(2048))
AS
declare variable rep_date_begin date;
declare variable db_id integer;
declare variable i smallint;
declare variable oks_balance decimal(18,4);
begin
  :db_id = f_get_db_id();
  i = 0;

  check_memo_all = '';
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end) + 1;
  for with tabl1
      as (select ks_from_k_id, 1 no_devides
          from kontstate
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_end >= :rep_date_begin - 1 and -- чтобы закрытых на первый день месяца тоже проверял
                ks_date_begin <= :rep_date_end
          union
          select kd_k_new, -1
          from kontdevides)
      select ks_from_k_id
      from tabl1
      group by ks_from_k_id
      having sum(no_devides) = 1
      into k_id
  do
  begin
    if (exists (select dci_k_id from daily_check_ignore where dci_k_id = :k_id)) then
      continue; --не проверяем контракты из игнор-листа

    ok_ol_date = null;
    if (:db_id = 81001) then
    begin
      for select k_kontract_no, check_memo, ok_ol_date, ok_from_ol_id
          from hp_check_kontract(:k_id, :rep_date_end)
          order by ok_ol_date desc, k_kontract_no
          into :k_kontract_no, :check_memo, :ok_ol_date, ok_from_ol_id
      do
      begin
        i = 1;
        if (ok_ol_date is null) then
          select ol_date
          from operlist
          where ol_id = :ok_from_ol_id rows 1
          into ok_ol_date;

        suspend;
      end
    end
    if (:db_id = 94401) then
    begin
      for select k_kontract_no, check_memo, ok_ol_date, ok_from_ol_id, link
          from hp_check_kontract_az(:k_id, :rep_date_end)
          order by ok_ol_date desc, k_kontract_no
          into :k_kontract_no, :check_memo, :ok_ol_date, ok_from_ol_id, link
      do
      begin
        i = 1;
        if (ok_ol_date is null) then
          select ol_date
          from operlist
          where ol_id = :ok_from_ol_id rows 1
          into ok_ol_date;

        suspend;
      end
    end
    if (:db_id in (39801, 41701)) then
    begin
      for select k_kontract_no, check_memo, ok_ol_date, ok_from_ol_id, link
          from hp_check_kontract_kg(:k_id, :rep_date_end)
          order by ok_ol_date desc, k_kontract_no
          into :k_kontract_no, :check_memo, :ok_ol_date, ok_from_ol_id, link
      do
      begin
        i = 1;
        if (ok_ol_date is null) then
          select ol_date
          from operlist
          where ol_id = :ok_from_ol_id rows 1
          into ok_ol_date;

        suspend;
      end
    end
  end

  for select k_kontract_no, oks.oks_from_k_id, oks_balance
      from oksummary oks
      inner join kontract kont on kont.k_id = oks.oks_from_k_id
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance < 0 and
            oks_close_state = 0
      into :k_kontract_no, :k_id, :oks_balance
  do
  begin
    check_memo = 'У контракта баланс по oksummary < 0: ' || cast(oks_balance as varchar(18));
    i = 1;
    suspend;
  end

  for select em.esm_id
      from extsysoper eo
      inner join operlist ol on ol.ol_id = eo.eso_from_ol_id
      inner join extsysmain em on em.esm_id = eo.eso_from_esm_id and
            ol.is_delete = 0 and
            em.esm_date = :rep_date_end and
            em.esm_project = 'europlat'
      group by em.esm_id
      having count(eo.eso_from_ol_id) > 1
      into :esm_id
  do
  begin
    i = 1;
    check_memo = 'По одному платежу от Европлат, несколько операции в БД. esm_id = ' || :esm_id;
    suspend;
  end

  if (i = 0) then
  begin
    k_id = '';
    check_memo = 'Ошибок не обнаружено';
    suspend;
  end
end
