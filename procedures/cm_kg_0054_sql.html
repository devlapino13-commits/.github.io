CREATE PROCEDURE CM_KG_0054_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable re_k_id id_type;
declare variable k_from_kf_id id_type;
declare variable r integer;
declare variable oks_balance curr_type;
declare variable delinq_day integer;
declare variable max_delinq_day integer;
declare variable oks_delinq_date date;
declare variable repay_date_2 date;
declare variable pr_code varchar(100);
begin
  -- Раздел 4. Условия выдачи кредитного продукта "Бирге - Льготный"
  --  пункт 4.2
  for select k_disb_date, kc_from_c_id, kc_amount_approved, c_name_fio, c_pass_no, k_from_kf_id, pr_code
      from kontract, product, knclient, clients
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            pr_code not containing '.IPF.' and
            pr_code containing '.PAR.' and
            k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into kf_disb_date, c_id, kc_amount, res_info_text, kf_client_pass_no, k_from_kf_id, pr_code
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;

    if (pr_code containing '.gr.') then
      pr_code = '.gr.';
    else
    if (pr_code containing '.ind.') then
      pr_code = '.ind.';
    else
      pr_code = null;
    --  Отсутвие просрочек на предыдущем и текущем циклах
    re_k_id = null;
    select k_id
    from kontstate, knclient, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :kf_disb_date between ks_date_begin and ks_date_end and
          ks_date_begin <= :kf_disb_date and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = :c_id and
          kc_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
          pr_code not containing '.express.' and
          pr_code containing :pr_code
    order by ks_date_begin desc
    rows 1
    into re_k_id;
    oks_balance = null;
    delinq_day = null;
    select oks_balance,
           (:kf_disb_date - oks_delinq_date), oks_delinq_date
    from oksummary
    where :kf_disb_date between oks_date_begin and oks_date_end and
          oks_close_state = 0 and
          oks_from_k_id = :re_k_id
    into oks_balance, delinq_day, oks_delinq_date;
    oks_balance = coalesce(oks_balance, 0);
    delinq_day = coalesce(delinq_day, 0);
    if (oks_balance = 0) then
    begin
      res_error_text = 'Не имеется основного кредита, выдача льготного кредита запрещается.';
      res_error_is_critical = 2;
      suspend;
    end
    if (delinq_day > 3 and
        not(oks_delinq_date between '01.12.20' and '31.01.21' and
        (pr_code containing '.ind.' or pr_code containing '.gr.' or pr_code not containing '.ipf.') and
        exists(select *
               from knclient_compens kcc
               inner join knclient kncl on kncl.kc_id = kcc.kcc_from_kc_id and
                     kc_from_c_id = :c_id
               where kcc.kcc_date_end > kcc.kcc_date_begin))) then
    begin
      res_error_text = 'Текущий цикл просрочен, выдача льготного кредита запрещается.';
      res_error_is_critical = 2;
      suspend;
    end
    -- текущий цикл произвел 2 и более погашений
    select sh_date
    from schedule
    where sh_from_k_id = :re_k_id and
          sh_id = 2
    into repay_date_2;
    if (repay_date_2 >= oks_delinq_date) then
    begin
      res_error_text = 'Основной кредит не произвел 2 и более погашений, выдача льготного кредита запрещается';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
    max_delinq_day = null;
    select max(ok_ol_date - ok_sh_date)
    from operkred
    where ok_from_k_id = :re_k_id and
          ok_ol_is_active = 1 and
          ok_type in (4, 5) and
          ok_ol_date > ok_sh_date and
          ok_os + ok_perc < -100 and
          not(ok_sh_date between '01.12.20' and '31.01.21' and
          exists(select *
                 from knclient_compens kcc
                 inner join knclient kncl on kncl.kc_id = kcc.kcc_from_kc_id and
                       kc_from_c_id = :c_id
                 where kcc.kcc_date_end > kcc.kcc_date_begin))
    into max_delinq_day;
    max_delinq_day = coalesce(max_delinq_day, 0);
    if (max_delinq_day > 3) then
    begin
      res_error_text = 'В предыдущем цикле имеется просрочка, выдача льготного кредита запрещается';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
