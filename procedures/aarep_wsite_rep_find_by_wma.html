CREATE PROCEDURE AAREP_WSITE_REP_FIND_BY_WMA(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ESM_ID INTEGER,
    ESM_DATE DATE,
    CL_NAME STR_255,
    CL_CITY STR_100,
    CL_PHONE_NO STR_100,
    CL_BIRTHDAY DATE,
    CC_CHECKED_DATE DATE,
    CC_CHECKED_OPER STR_100,
    CC_CALL_STATUS STR_100,
    CC_MEMO VARCHAR(1000),
    CL_ADDITIONAL_CELL_NO STR_100,
    LOAN_STASUS STR_100,
    KF_NAME STR_100,
    SEARCH_METHOD STR_100,
    SVZ_FIO STR_100,
    SVZ_CELL_NO STR_100,
    DISB_DATE DATE,
    KF_REG_NAME STR_100,
    WMA_TO_SVZ STR_100,
    WMA_SVZ_CELL_NO STR_100,
    WMA_REGION STR_100)
AS
declare variable params varchar(8000);
declare variable project_rd_id integer;
declare variable cc_fio str_100;
declare variable ma_is_active integer;
declare variable kf_id id_type;
declare variable k_id id_type;
declare variable ol_id id_type;
declare variable k_status integer;
declare variable cc_birth_date date;
declare variable c_id id_type;
declare variable ma_id integer;
declare variable k_disb_date date;
declare variable clt_id d_integer;
declare variable is_new_contdata d_smallint;
begin
  project_rd_id = f_get_rd_id('OKTELL.Projects.WebsiteMiniapp');
  delete from website_miniapp_cli_check_tmp
  where 1 = 1;

  is_new_contdata = f_get_const('new_contdata');
  is_new_contdata = 1; -- процедуру переводим на новый механизм, Sanjari, 21.12.2019

  -- подтягиваем заявки с сайта для мигранатов за период
  for select em.esm_id, em.esm_date, em.esm_full_source
      from extsysmain em
      where em.esm_project = 'website' and
            em.esm_status = 0 -- если = 0 значит успешная команда
            and
            em.esm_date between :rep_date_begin and :rep_date_end and
            em.esm_action = 'save_cl_info'
      into :esm_id, :esm_date, :params
  do
  begin
    cl_name = (select field_value
               from hp_extsys_parcer(:params, 'cl_name'));
    cl_city = (select field_value
               from hp_extsys_parcer(:params, 'cl_city'));
    cl_phone_no = (select field_value
                   from hp_extsys_parcer(:params, 'phone_no'));

    cl_phone_no = coalesce(cl_phone_no, '');

    begin
      cl_birthday = (select field_value
                     from hp_extsys_parcer(:params, 'cl_birthday'));
      when sqlcode -413 do
      -- convert_error
      begin
        cl_birthday = null;
      end
    end

    cc_checked_date = null;
    cc_checked_oper = null;
    cc_call_status = null;
    cc_birth_date = null;
    cc_fio = null;
    cc_memo = null;

    -- получаем результаты звонков с конечными статусами
    select chk.ck_checked_date, -- дата звонка
           chk.ck_checked_oper,
           -- оператор кц.
           (
           case ck_status
             when 1 then 'Интресуется (перезвонить)'
             when 2 then 'Интресуется (конеч. статус)'
             when -2 then 'Не интересуется'
           end) as ck_status,
           chk.ck_birth_date, chk.ck_fio, -- фио по заявке
           chk.ck_memo,
           -- комментарий оператора
           --ckt.ckt_cell_no,
           (select cell_no
            from cc_secondary_data_parse(chk.ck_secondary_data)),
            cl.c_name_fio, f_cont_info('clients', cl.c_id, 'cell_main', 'ok'), reg.reg_name
    from cc_check_knclient chk
    left join cc_website_lo_sms wls on wls.wls_from_ck_id = chk.ck_id
    left join clients cl on cl.c_id = wls.wls_lo_id
    left join region reg on reg.reg_unicode = ck_from_reg_id
          --left join cc_check_knclient_tmp  ckt on ckt.ckt_from_ck_id = chk.ck_id
    where chk.ck_project_rd_id = :project_rd_id and
          chk.ck_from_esm_id = :esm_id and
          ck_status in (1, 2, -2) -- берем только конечные статусы
    order by chk.ck_id desc
    rows 1
    into :cc_checked_date, :cc_checked_oper, :cc_call_status, :cc_birth_date, :cc_fio, :cc_memo, :cl_additional_cell_no,
         :wma_to_svz, :wma_svz_cell_no, :wma_region;

    cc_fio = coalesce(cc_fio, '');
    cl_additional_cell_no = coalesce(cl_additional_cell_no, '');

    if (cc_fio <> '') then
      cl_name = cc_fio;
    if (cc_birth_date is not null) then
      cl_birthday = cc_birth_date;

    -- осуществляем поиск контрагента по ФИО+дата рождения, номеру телефона -->>
    -- поиск по фио и дате рождения
    for select cl.c_id
        from clients cl
        where cl.c_birth_date = :cl_birthday and
              cl.c_name_fio = :cl_name
        into :c_id
    do
    begin
      update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method)
      values (:esm_id, :c_id, 'Найден по фио и дате рождения')
      matching (wmc_esm_id, wmc_c_id);
    end

    -- поиск по телефону из  заявки
    if (cl_phone_no <> '') then
    begin
      if (f_get_const('new_conttdata') = 0) then
      begin
        for select distinct opr.opr_object_id
            from objectprops opr, objectvalues ov, refdata
            where opr_subject_kind = -1 and
                  opr_subject_id = rd_id and
                  rd_code in ('ContactInfo.Cell_Main', 'ContactInfo.Cell_Add') and
                  opr_id = ovs_from_opr_id and
                  hash(lower(ovs_value)) = hash(lower(:cl_phone_no)) and
                  ovs_value = :cl_phone_no and
                  ov.ovs_date_begin >= '01.01.1900'
            into :c_id
        do
        begin
          update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method)
          values (:esm_id, :c_id, 'Найден по телефону, из заявки с сайта')
          matching (wmc_esm_id, wmc_c_id);
        end
      end else
      begin
        clt_id = f_cont_id(cl_phone_no, 'null');
        if (clt_id is not null) then
        begin
          for select distinct cda_from_c_id
          from contdata
          where cda_is_delete = 0 and       
                cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
                cda_from_clt_id = :clt_id
          into c_id
          do
          begin
            update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method)
            values (:esm_id, :c_id, 'Найден по телефону, из заявки с сайта')
            matching (wmc_esm_id, wmc_c_id);
          end
        end
      end
    end
    -- поиск по доп телефону, который внес оператор
    if (cl_additional_cell_no <> '') then
    begin
      if (is_new_contdata = 0) then
      begin
        for select distinct opr.opr_object_id
            from objectprops opr, objectvalues ov, refdata
            where opr_subject_kind = -1 and
                  opr_subject_id = rd_id and
                  rd_code in ('ContactInfo.Cell_Main', 'ContactInfo.Cell_Add') and
                  opr_id = ovs_from_opr_id and
                  hash(lower(ovs_value)) = hash(lower(:cl_additional_cell_no)) and
                  ovs_value = :cl_additional_cell_no and
                  ov.ovs_date_begin >= '01.01.1900'
            into :c_id
        do
        begin
          update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method)
          values (:esm_id, :c_id, 'Найден по телефону, который внес оператор')
          matching (wmc_esm_id, wmc_c_id);
        end
      end else
      begin
        clt_id = f_cont_id(cl_additional_cell_no, 'null');
        if (clt_id is not null) then
        begin
          for select distinct cda_from_c_id
          from contdata
          where cda_is_delete = 0 and       
                cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
                cda_from_clt_id = :clt_id
          into c_id
          do
          begin
            update or insert into website_miniapp_cli_check_tmp (wmc_esm_id, wmc_c_id, wmc_method)
            values (:esm_id, :c_id, 'Найден по телефону, который внес оператор')
            matching (wmc_esm_id, wmc_c_id);
          end
        end
      end
    end
    ----------- << ----------

    loan_stasus = '';
    ma_id = null;
    ma_is_active = null;
    kf_id = null;
    kf_name = null;
    k_id = null;
    k_status = null;
    ol_id = null;
    search_method = '';
    svz_fio = null;
    svz_cell_no = null;
    disb_date = null;
    kf_reg_name = null;

    -- по найденным контрагентам осузествляем поиск на предмет получения займа после оформления заявки с сайта
    for select mcc.wmc_c_id, mcc.wmc_method
        from website_miniapp_cli_check_tmp mcc
        where mcc.wmc_esm_id = :esm_id
        into :c_id, :search_method
    do
    begin
      loan_stasus = '';
      ma_id = null;
      ma_is_active = null;
      kf_id = null;
      kf_name = null;
      k_id = null;
      k_status = null;
      ol_id = null;
      svz_fio = null;
      svz_cell_no = null;
      disb_date = null;
      kf_reg_name = null;

      for select maa.ma_id, maa.ma_is_active, maa.ma_from_kf_id, maa.ma_date_oper
          from miniapp maa
          inner join miniappclient macl on maa.ma_id = macl.mac_from_ma_id
          where maa.ma_date_app >= :esm_date and
                maa.ma_date_end = '31.12.9999' and
                macl.mac_amount > 0 and
                macl.mac_from_c_id = :c_id
          into :ma_id, :ma_is_active, :kf_id, :disb_date
      do
      begin
        loan_stasus = 'Кредитная заявка';

        select kal.kal_new_kf_name, cl.c_name_fio, f_cont_info('clients', cl.c_id, 'cell_main', 'ok'), reg_name
        from knaktlist kal
        inner join clients cl on kal.kal_new_kf_officer = cl.c_id
        inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
        where kal.kal_from_kf_id = :kf_id and
              kal.is_delete = 0 and
              current_date between kal.kal_date and kal.kal_date_end -- беру именно на текущую дату
        into :kf_name, :svz_fio, :svz_cell_no, :kf_reg_name;

        if (ma_is_active = 0) then
          loan_stasus = 'Отказ';
        else
        begin
          k_id = null;
          ol_id = null;
          k_disb_date = null;

          select kont.k_id, kont.k_status, kont.k_disb_date, ol.ol_id
          from kontract kont
          left join operlist ol on ol.ol_id = kont.k_fact_disb_ol_id and
                ol.is_delete = 0
          where kont.k_from_ma_app_id = :ma_id rows 1
          into :k_id, k_status, :k_disb_date, :ol_id;

          if (k_id is not null) then
          begin
            loan_stasus = 'Договор в процессе';

            if (k_status = -1) then
              loan_stasus = 'Отказ';
            else
            begin
              if (ol_id is not null) then
                loan_stasus = 'Выдан';
            end
            disb_date = k_disb_date;
          end
        end

        suspend;
      end
    end

    if (ma_id is null) then
    begin
      c_id = null;
      search_method = '';
      suspend;
    end
  end
end
