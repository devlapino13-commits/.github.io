CREATE PROCEDURE CM_RU_0096_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable ma_expert_fio varchar(100);
declare variable ma_expert_id id_type;
declare variable rd_id integer;
begin

 if(ma_id = 504225) then exit;
  select ma.ma_expert_id
  from miniapp ma
  where ma_id = :ma_id 
  and current_date between ma.ma_date_app and ma_date_end
  and ma.ma_is_active = 1
  into ma_expert_id;
  if (ma_expert_id is not null) then
  begin
    select rd_id
    from refdata rd
    where rd.rd_code = 'Anketa.Delinq.1'
    into rd_id;
    if (exists(
                select *
                from actionorder ao
                where ao.aco_date <= current_date-5
                and ao.aco_respons_c_id = :ma_expert_id
                and ao.aco_from_rd_id = :rd_id
               )
        ) then
    begin
      select c_name_fio
      from clients
      where c_id = :ma_expert_id
      into ma_expert_fio;
      res_error_text = 'У '||ma_expert_fio||' есть активные задачи по «Анкета по просрочке» которые числятся необработанными 5 дней и более';
      res_error_is_critical = 1;
      suspend;
    end
  end
end
