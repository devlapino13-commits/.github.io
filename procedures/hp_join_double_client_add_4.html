CREATE PROCEDURE HP_JOIN_DOUBLE_CLIENT_ADD_4(
    C_STAY D_ID,
    C_REMOVE D_ID)
AS
declare variable ccr_stay d_integer;
declare variable ccr_remove d_integer;
begin
  if (not exists(select * from clients where c_id = :c_stay) or
      not exists(select * from clients where c_id = :c_remove))
  then exit;

  update cardclientid set
    cci_from_c_id = :c_stay
  where cci_from_c_id = :c_remove;

  update cardaktclient set
    cac_from_c_id = :c_stay
  where cac_from_c_id = :c_remove;

  update cardlistlnk set
    cll_from_c_id = :c_stay
  where cll_from_c_id = :c_remove;

  update cardclientacc_limit set
    ccl_from_c_id = :c_stay
  where ccl_from_c_id = :c_remove;

  update cardsync set
    cs_from_c_id = :c_stay
  where cs_from_c_id = :c_remove;

  update cardblockopers set
    cbo_from_c_id = :c_stay
  where cbo_from_c_id = :c_remove;

  update finerefund set
    fr_from_c_id = :c_stay
  where fr_from_c_id = :c_remove;

  update edoc_face
  set edf_from_c_id = :c_stay
  where edf_from_c_id = :c_remove;

  update epass_response
  set erp_from_c_id = :c_stay
  where erp_from_c_id = :c_remove;

  update kommisrefund_cards
  set kmr_from_c_id = :c_stay
  where kmr_from_c_id = :c_remove;

  update overpayment
  set ovp_from_c_id = :c_stay
  where ovp_from_c_id = :c_remove;

  update cardlistloan
  set clo_from_c_id = :c_stay
  where clo_from_c_id = :c_remove;

  update ipc_exd_detail
  set ied_from_c_id = :c_stay
  where ied_from_c_id = :c_remove;

  update epass_response
  set erp_from_c_id = :c_stay
  where erp_from_c_id = :c_remove;

  -- Перенос ограничений Реестра с приоритетом для запретов (ccr_limit_* = 1)
  if (f_get_db_id() = 81001) then
  begin
    merge into clientcomreg ccr_stay
    using (
      select
        :c_stay as ccr_from_c_id,
        max(ccr_limit_38) as ccr_limit_38,
        max(ccr_limit_152) as ccr_limit_152,
        max(ccr_limit_230) as ccr_limit_230
      from clientcomreg
      where ccr_from_c_id in (:c_stay, :c_remove)) ccr_actual
    on ccr_stay.ccr_from_c_id = :c_stay
    when matched then
     update set
       ccr_limit_38 = ccr_actual.ccr_limit_38,
       ccr_limit_152 = ccr_actual.ccr_limit_152,
       ccr_limit_230 = ccr_actual.ccr_limit_230
    when not matched then
      insert (
        ccr_from_c_id,
        ccr_limit_38,
        ccr_limit_152,
        ccr_limit_230)
      values (
        ccr_actual.ccr_from_c_id,
        ccr_actual.ccr_limit_38,
        ccr_actual.ccr_limit_152,
        ccr_actual.ccr_limit_230);
  
    select ccr_id
    from clientcomreg
    where ccr_from_c_id = :c_stay
    into ccr_stay;
  
    select ccr_id
    from clientcomreg
    where ccr_from_c_id = :c_remove
    into ccr_remove;
  
    update clientcomlog
    set ccl_from_ccr_id = :ccr_stay
    where ccl_from_ccr_id = :ccr_remove;
  
    -- Снимаем ограничения если есть подписанные документы
    if (f_edoc_blank_signed(c_stay, 'ad.agreement', current_date) = 1) then
      execute procedure hp_clientcomreg_set(:c_stay, null, 'remove', 1, 0, 0, 'edoc_signed');
    if (f_edoc_blank_signed(c_stay, 'pdata.agreement', current_date) = 1) then
      execute procedure hp_clientcomreg_set(:c_stay, null, 'remove', 0, 1, 0, 'edoc_signed');
  end
end
