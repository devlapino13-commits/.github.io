CREATE PROCEDURE HP_TIMESHEET_LIST_HOURS(
    YEAR_INP D_INTEGER,
    MONTH_INP D_INTEGER,
    C_ID D_ID,
    REG_ID D_INTEGER)
RETURNS (
    C_ID_OUT D_ID,
    SAWT_ID D_INTEGER,
    SAWT_NAME D_STR_100,
    SAT_DAY_1 D_INTEGER,
    SAT_DAY_2 D_INTEGER,
    SAT_DAY_3 D_INTEGER,
    SAT_DAY_4 D_INTEGER,
    SAT_DAY_5 D_INTEGER,
    SAT_DAY_6 D_INTEGER,
    SAT_DAY_7 D_INTEGER,
    SAT_DAY_8 D_INTEGER,
    SAT_DAY_9 D_INTEGER,
    SAT_DAY_10 D_INTEGER,
    SAT_DAY_11 D_INTEGER,
    SAT_DAY_12 D_INTEGER,
    SAT_DAY_13 D_INTEGER,
    SAT_DAY_14 D_INTEGER,
    SAT_DAY_15 D_INTEGER,
    SAT_DAY_16 D_INTEGER,
    SAT_DAY_17 D_INTEGER,
    SAT_DAY_18 D_INTEGER,
    SAT_DAY_19 D_INTEGER,
    SAT_DAY_20 D_INTEGER,
    SAT_DAY_21 D_INTEGER,
    SAT_DAY_22 D_INTEGER,
    SAT_DAY_23 D_INTEGER,
    SAT_DAY_24 D_INTEGER,
    SAT_DAY_25 D_INTEGER,
    SAT_DAY_26 D_INTEGER,
    SAT_DAY_27 D_INTEGER,
    SAT_DAY_28 D_INTEGER,
    SAT_DAY_29 D_INTEGER,
    SAT_DAY_30 D_INTEGER,
    SAT_DAY_31 D_INTEGER,
    SAT_IS_READY D_INTEGER,
    SAT_TOTAL_HOURS D_INTEGER,
    SAT_TOTAL_CALC D_INTEGER,
    SAT_DAYS_TOTAL D_CURRENCY,
    SAT_DAYS_CALC D_CURRENCY)
AS
declare variable year_month varchar(10);
begin
  year_month = :year_inp || iif(:month_inp < 10, '0' || :month_inp, :month_inp);
  c_id_out = :c_id;
  for select sat_day_1, sat_day_2, sat_day_3, sat_day_4, sat_day_5, sat_day_6, sat_day_7, sat_day_8, sat_day_9,
             sat_day_10, sat_day_11, sat_day_12, sat_day_13, sat_day_14, sat_day_15, sat_day_16, sat_day_17, sat_day_18,
             sat_day_19, sat_day_20, sat_day_21, sat_day_22, sat_day_23, sat_day_24, sat_day_25, sat_day_26, sat_day_27,
             sat_day_28, sat_day_29, sat_day_30, sat_day_31, sat_from_sawt_id, sat_is_ready, sat_total_hours,
             sat_total_calc, sat_days_total, sat_days_calc
      from sa_tabelo sat
      where sat.sat_year_month = :year_month and
            sat.sat_from_c_id = :c_id and
            sat.sat_office_id = :reg_id
      order by sat_from_sawt_id
      into :sat_day_1, :sat_day_2, :sat_day_3, :sat_day_4, :sat_day_5, :sat_day_6, :sat_day_7, :sat_day_8, :sat_day_9,
           :sat_day_10, :sat_day_11, :sat_day_12, :sat_day_13, :sat_day_14, :sat_day_15, :sat_day_16, :sat_day_17,
           :sat_day_18, :sat_day_19, :sat_day_20, :sat_day_21, :sat_day_22, :sat_day_23, :sat_day_24, :sat_day_25,
           :sat_day_26, :sat_day_27, :sat_day_28, :sat_day_29, :sat_day_30, :sat_day_31, :sawt_id, :sat_is_ready,
           :sat_total_hours, :sat_total_calc, :sat_days_total, :sat_days_calc
  do
  begin
    sawt_name = (select sawt_name from sa_worktype where sawt_id = :sawt_id);
    suspend;
  end
end
