CREATE PROCEDURE CM_KG_0170_SQL(
    MA_ID D_ID,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable point d_integer;
declare variable mac_amount curr_type;
declare variable ma_kolich_repay d_integer;
declare variable ma_date_oper d_date;
begin
  select
    ma_kolich_repay,
    ma_date_oper
  from miniapp
    inner join product
      on pr_id = ma_from_pr_id
        and pr_code containing '.mbulak_cards.'
  where ma_id = :ma_id
    and ma_date_end = '31.12.9999'
    and ma_is_active = 1
  into
    :ma_kolich_repay,
    :ma_date_oper;

  if (row_count = 1) then
  begin
    res_error_is_critical = 1;
    for
      select
        f_point(mac_from_c_id, :ma_date_oper),
        mac_amount
      from miniappclient
      where mac_from_ma_id = :ma_id
        and mac_amount > 0
    into
      :point,
      :mac_amount
    do
    begin
      if (point <= 3) then
      begin
        if (mac_amount > 80000) then
        begin
          res_error_text = 'У клиента балл меньше 3, поэтому он не может получить сумму больше 80 000 сом';
          suspend;
        end

        if (ma_kolich_repay > 18) then
        begin
          res_error_text = 'У клиента балл меньше 3, поэтому он не может получить заем на срок больше 18 месяцев';
          suspend;
        end
      end
      else
      if (point between 4 and 9) then
      begin
        if (mac_amount > 100000) then
        begin
          res_error_text = 'По данному продукту максимально допустимая сумма 100 000 сом, поэтому клиент не может получить сумму больше 100 000 сом';
          suspend;
        end

        if (ma_kolich_repay > 18) then
        begin
          res_error_text = 'По данному продукту максимальный период 18 месяцев';
          suspend;
        end
      end
      else
       if (point between 10 and 19) then
      begin
        if (mac_amount > 150000) then
        begin
          res_error_text = 'По данному продукту максимально допустимая сумма 150 000 сом, поэтому клиент не может получить сумму больше 100 000 сом';
          suspend;
        end

        if (ma_kolich_repay > 18) then
        begin
          res_error_text = 'По данному продукту максимальный период 18 месяцев';
          suspend;
        end
      end
      else
      if (point >= 20) then
      begin
        if (mac_amount > 200000) then
        begin
          res_error_text = 'Максимальная сумма займа 200 000 сом';
          suspend;
        end

        if (ma_kolich_repay > 24) then
        begin
          res_error_text = 'Срок займа не может превышать 24 месяцев';
          suspend;
        end
      end
    end
  end
end
