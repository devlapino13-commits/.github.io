CREATE PROCEDURE UXP$UXUNITSOURCE_LIST(
    UNIT_UUID D_UUID)
RETURNS (
    USR_UUID D_UUID,
    USR_CODE D_STR_100,
    UDS_UUID D_UUID,
    UDS_DATASET D_STR_CODE,
    UDS_SQL_SELECT D_STR_4096,
    UDS_SQL_UPDATE D_STR_4096,
    UDS_SQL_INSERT D_STR_4096,
    UDS_SQL_DELETE D_STR_LONG,
    UDS_READONLY D_BOOL,
    RES_STATE D_SMALLINT,
    RES_UNI_LINKED D_BOOL,
    RES_UNI_UUID D_UUID)
AS
declare variable unt_code d_str_code;
begin
  select unt_code
  from ux$unit
  where unt_uuid = :unit_uuid
  into unt_code;

  if (coalesce(unt_code, '') = '') then
   Exit;

  res_state = 0;
  for select distinct uni_datasource, uni_usr_uuid
  from ux$unititem
  where uni_unt_uuid = :unit_uuid and
        uni_usr_uuid is not null
  into usr_code, usr_uuid
  do
  begin
    for select uds_uuid,
               iif(uds_unit_code = :unt_code, '', uds_unit_code || '.') || uds_dataset,
               uds_sql_select, uds_sql_update, uds_sql_insert, uds_sql_delete, uds_readonly
    from ux$unitsource, ux$unitdata
    where usr_uuid = :usr_uuid and
          usr_uds_uuid = uds_uuid
    into :uds_uuid, :uds_dataset,
         :uds_sql_select, :uds_sql_update, :uds_sql_insert, :uds_sql_delete, :uds_readonly
    do
    begin
      res_uni_uuid = null;
      res_uni_linked = 0;

      select unf_uni_uuid
      from ux$unitframe, ux$unititem
      where unf_owner = :unit_uuid and
            unf_usr_uuid = :usr_uuid and
            unf_uni_uuid = uni_uuid
      into :res_uni_uuid;

      if (res_uni_uuid is not null) then
        res_uni_linked = 1;

      suspend;
    end
  end
end
