CREATE PROCEDURE HP_TSERVICESH(
    PAR1 D_INTEGER)
RETURNS (
    SSH_FROM_SD_ID D_INTEGER,
    SSH_ID D_INTEGER,
    SSH_DATE D_DATE,
    SSH_AMOUNT_DOGOVOR D_CURRENCY,
    SSH_DATE_REGIST D_DATE,
    SSHE_IS_COMPLITED D_SMALLINT,
    SSHE_MEMO D_STR_255)
AS
begin
  for select ssh.ssh_from_sd_id, ssh.ssh_id, ssh.ssh_date, ssh.ssh_amount_dogovor, ssh.ssh_date_regist,
             sshe.sshe_is_complited, sshe.sshe_memo
      from servicesh ssh
      left join servicesh_edoc sshe on ssh.ssh_from_sd_id = sshe.sshe_from_sd_id and
            ssh.ssh_id = sshe.sshe_id
      where ssh_from_sd_id = :par1
      into :ssh_from_sd_id, :ssh_id, :ssh_date, :ssh_amount_dogovor, :ssh_date_regist, :sshe_is_complited, :sshe_memo

  do
  begin
    suspend;
  end
end
