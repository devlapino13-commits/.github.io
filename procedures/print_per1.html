CREATE PROCEDURE PRINT_PER1(
    C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    HO_DATE DATE,
    HO_TEXT VARCHAR(1024),
    HO_ORDER VARCHAR(100),
    HO_ORDER_NO VARCHAR(20),
    HO_REGION VARCHAR(100),
    HO_DATE_BEGIN DATE,
    HO_DATE_END DATE,
    HO_JOBNAME VARCHAR(255),
    HO_IS_ORDER INTEGER,
    HO_FIO VARCHAR(100),
    HO_TABEL_NO VARCHAR(10),
    HO_CURR_JOBNAME VARCHAR(255),
    HO_REP_DATE DATE,
    HO_IS_WORK VARCHAR(20))
AS
declare variable is_work integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  ho_rep_date = :rep_date;
  select c_name_fio
  from clients
  where c_id = :c_id
  into :ho_fio;

  select ps_tabel_no
  from hr_pstatus
  where ps_tabel_no is not null and
        ps_from_c_id = :c_id and
        ps_status in (0, 1) -- включает только аннулированные и утвержденные
  order by ps_date_begin desc
  rows 1
  into :ho_tabel_no;

  select reg_code || ' ' || reg_name || ', ' || jl_name, ps_is_working
  from hr_pstatus, region, depart, joblist
  where ps_from_c_id = :c_id and
        ps_from_reg_id = reg_unicode and
        ps_from_dep_id = dep_id and
        ps_from_jl_id = jl_id and
        ps_date_begin <= :rep_date and
        ps_date_end >= :rep_date and
        ps_status in (0, 1) -- включает только аннулированные и утвержденные
  order by ps_date_begin
  rows 1
  into :ho_curr_jobname, :is_work;
  if (is_work is null) then
    is_work = 0;
  if (is_work = 1) then
    ho_is_work = 'Работает';
  else
    ho_is_work = 'Уволен';

  for select ho_date, ho_text, ho_order, ho_order_no, ho_region, ho_date_begin, ho_date_end, ho_jobname, ho_tabel_no,
             ho_is_order
      from print_per1_tmp(:c_id, :rep_date)
      order by ho_date_begin desc, ho_date desc, ho_is_order desc
      into :ho_date, :ho_text, :ho_order, :ho_order_no, :ho_region, :ho_date_begin, :ho_date_end, :ho_jobname,
           :ho_tabel_no, :ho_is_order
  do
  begin
    suspend;
  end
end
