CREATE PROCEDURE OKT_PANTERA_KF_CHECK(
    KF_NAME STR_100,
    KONTRACT_NO STR_100,
    PROJEC_TYPE D_SMALLINT)
RETURNS (
    MESSAGE_TEXT D_STR_LONG)
AS
declare variable REP_DATE D_DATE;
declare variable SBL_NAME STR_255;
declare variable PANT_OPER_CALLED STR_255;
declare variable SBL_ID D_INTEGER;
declare variable SBA_K_ID ID_TYPE;
declare variable K_ID D_ID;
declare variable REG_NAME STR_255;
declare variable C_NAME_FIO STR_100;
declare variable C_CITIZEN STR_100;
declare variable SQL_TEXT_1 D_BLOB_TEXT;
begin
  kf_name = trim(coalesce(kf_name, ''));
  kontract_no = trim(coalesce(kontract_no, ''));

  rep_date = current_date;
  sba_k_id = null;
  sbl_name = '';
  pant_oper_called = '';
  message_text = '';

  if ((kontract_no = '') and (kf_name = '')) then
  begin
    message_text = '№ договора или № группы не заполнены';
    suspend;
    exit;
  end

  if ((kontract_no <> '') and (kf_name <> '')) then
  begin
    message_text = 'Для поиска укажите либо № группы либо № договора';
    suspend;
    exit;
  end

  if (projec_type is null)  then
  begin
    message_text = 'Проект не передан (projec_type is null)';
    suspend;
    exit;
  end

  select cast(rd.rd_str_id as d_integer)
  from refdata rd
  where rd.rd_main_code = 'OKTELL.Projects'
  and rd.rd_tag = :projec_type
  into :sbl_id;

  sbl_name = '';

  select sbl.sbl_name
  from sb_list sbl
  where sbl.sbl_id = :sbl_id
  into :sbl_name;

  if (sbl_name = '')  then
  begin
    message_text = 'Проект не определен';
    suspend;
    exit;
  end

  k_id = null;

  if (kontract_no <> '') then
  begin
    select kont.k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id;    
  end

  if ((k_id is null) and (kf_name = '')) then
  begin
    message_text = 'Контракт закрыт или не числится за ' || sbl_name;
    suspend;
    exit;
  end

  sql_text_1 = 'select kont.k_id, reg.reg_name ';
  sql_text_1 = blob_append(sql_text_1,'from sb_akt sba ');
  sql_text_1 = blob_append(sql_text_1,'inner join kontract kont on sba.sba_from_k_id = kont.k_id ');
  sql_text_1 = blob_append(sql_text_1,'inner join knaktlist kal on kont.k_from_kf_id = kal.kal_from_kf_id ');
  sql_text_1 = blob_append(sql_text_1,'inner join oksummary oks on oks.oks_from_k_id = kont.k_id ');
  sql_text_1 = blob_append(sql_text_1,'inner join region reg on reg.reg_unicode = kont.k_last_reg_id ');
  sql_text_1 = blob_append(sql_text_1,'where kal.is_delete = 0 and ');
  sql_text_1 = blob_append(sql_text_1,''''||f_datetostr(:rep_date)||''' between kal_date and kal_date_end and ');
  sql_text_1 = blob_append(sql_text_1,'sba.sba_date_end >= '''||f_datetostr(:rep_date)||''' and ');
  sql_text_1 = blob_append(sql_text_1,'sba.sba_date_begin <= sba.sba_date_end and ');
  sql_text_1 = blob_append(sql_text_1,'sba_is_delete = 0 and ');
  sql_text_1 = blob_append(sql_text_1,''''||f_datetostr(:rep_date)||''' between oks.oks_date_begin and oks.oks_date_end ');
  sql_text_1 = blob_append(sql_text_1,'and sba.sba_from_sbl_id = '||:sbl_id||' ');

  if ((kf_name <> '') and (k_id is null)) then
    sql_text_1 = blob_append(sql_text_1,'and kal.kal_new_kf_name = '''||kf_name||''' ');

  if ((k_id is not null) and (kf_name = '')) then
    sql_text_1 = blob_append(sql_text_1,'and kont.k_id = '''||K_id||''' ');

  sql_text_1 = blob_append(sql_text_1,'rows 1 ');

  execute statement(sql_text_1)
  into :sba_k_id, :reg_name;

  if (sba_k_id is null) then
  begin
    message_text = 'Контракт закрыт или не числится за ' || sbl_name;
  end
  else
  begin
    select pant.pant_oper_called
    from pantera_kontracts pant
    where pant.pant_from_k_id = :sba_k_id and
          :rep_date between pant.pant_date_begin and pant.pant_date_end and
          pant.pant_project_type = :projec_type
    into :pant_oper_called;

    if (pant_oper_called is null) then
      pant_oper_called = '';

    if (pant_oper_called = '') then
      message_text = 'Группа свободна для блокировки. Офис: ' || reg_name;
    else
    begin
      message_text = 'Группа числится за ' || pant_oper_called || '. Офис: ' || reg_name || ';  ';

      for select c_name_fio, ddr.rd_name
          from (select cl.c_name_fio, coalesce(cl.c_citizen, 0) as c_citizen
                from knclient kncl
                inner join clients cl on cl.c_id = kncl.kc_from_c_id
                where kncl.kc_from_k_id = :sba_k_id and
                      kncl.kc_amount_approved > 0) t
          inner join refdata ddr on ddr.rd_id = t.c_citizen
          into :c_name_fio, :c_citizen
      do
        message_text = message_text || :c_name_fio || '(' || :c_citizen || ');  ';
    end
  end
  suspend;
end
