CREATE PROCEDURE HP_MAKE_ADDR_PLACE(
    LANG_KIND D_SMALLINT,
    ADL_FROM_PL_ID D_ID,
    ADL_STREET_ID D_ID,
    ADL_PLACE_PLS_ID D_ID,
    ADL_STREET_PLS_ID D_ID,
    ADL_POST_INDEX D_STR_20)
RETURNS (
    ADDRESS_STR D_STR_255)
AS
declare variable str_tmp d_str_255;
declare variable pl_name d_str_100;
declare variable plt_short_name d_str_100;
declare variable plt_short_name_side d_smallint;
declare variable pl_parent_id d_id;
declare variable plt_show_in_address d_smallint;
begin
  address_str = '';
  if (adl_street_id is null and
      adl_from_pl_id is null) then
  begin
    suspend;
    exit;
  end

  pl_name = '';

  str_tmp = '';
  if (adl_street_pls_id is not null) then
  begin
    select pls_name
    from placesyn
    where pls_id = :adl_street_pls_id
    into :str_tmp;
  end

  if (adl_street_id is not null and
      adl_street_id <> adl_from_pl_id) then
  begin
    select trim(iif(:str_tmp <> '', :str_tmp, pl_name)), trim(iif(:lang_kind <> 2, plt_short_name, plt_alt_short_name)),
           iif(:lang_kind <> 2, plt_short_name_side, plt_alt_short_name_side)
    from places, placetype
    where pl_status = plt_id and
          pl_id = :adl_street_id
    into :pl_name, plt_short_name, plt_short_name_side;
    if (plt_short_name_side = 1) then
      pl_name = pl_name || ' ' || plt_short_name;
    if (plt_short_name_side = -1) then
      pl_name = plt_short_name || ' ' || pl_name;
  end

  address_str = pl_name;

  str_tmp = '';
  if (adl_place_pls_id is not null) then
  begin
    select trim(pls_name)
    from placesyn
    where pls_id = :adl_place_pls_id
    into :str_tmp;
  end

  pl_parent_id = adl_from_pl_id;
  while (pl_parent_id is not null) do
  begin
    for select trim(iif(:str_tmp <> '', :str_tmp, pl_name)), parent, plt_show_in_address,
               trim(iif(:lang_kind <> 2, plt_short_name, plt_alt_short_name)),
               iif(:lang_kind <> 2, plt_short_name_side, plt_alt_short_name_side)
        from places, placetype
        where pl_status = plt_id and
              pl_id = :pl_parent_id
        into :pl_name, :pl_parent_id, plt_show_in_address, plt_short_name, plt_short_name_side
    do
    begin
      if (plt_show_in_address = 1) then
      begin
        if (plt_short_name_side = 1) then
          pl_name = pl_name || ' ' || plt_short_name;
        if (plt_short_name_side = -1) then
        begin
          if (right(plt_short_name, 1) = '.') then
            pl_name = plt_short_name || pl_name;
          else
            pl_name = plt_short_name || ' ' || pl_name;
        end

        if (address_str <> '') then
          address_str = pl_name || ', ' || address_str;
        else
          address_str = pl_name;
      end
    end
  end


  if (coalesce(adl_post_index, '') = '') then
    adl_post_index = '';
  if (char_length(adl_post_index) <= 3) then
    adl_post_index = '';

  if (adl_post_index <> '') then
    address_str = adl_post_index || ', ' || address_str;

  address_str = f_replace_all(trim(address_str), '  ', ' ');
  address_str = replace(address_str, '. ', '.');

  suspend;
end
