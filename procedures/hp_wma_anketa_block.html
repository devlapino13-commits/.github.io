CREATE PROCEDURE HP_WMA_ANKETA_BLOCK(
    WMA_ID INTEGER,
    USER_ID ID_TYPE)
RETURNS (
    ERROR_MESSAGE STR_255)
AS
declare variable UN_ID ID_TYPE;
declare variable WMA_STATUS D_INTEGER;
begin
  error_message = '';
  select t.wma_blocked_un_id, t.wma_status
  from cc_website_miniapp t
  where t.wma_id = :wma_id
  into :un_id, :wma_status;

  if ((un_id is not null) and (user_id <> un_id)) then
    error_message = 'Заявка заблокирована';
  else if (wma_status not in (0, 1)) then
    error_message = 'Заявка уже переведена в конечный статус';
  else
  begin
    update cc_website_miniapp t
    set t.wma_blocked_un_id = :user_id
    where t.wma_id = :wma_id
    and t.wma_blocked_un_id is null;
  end

  suspend;
end
