CREATE PROCEDURE AAREP_NEW_SCHEDULE_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    PR_NAME VARCHAR(100),
    KAL_NEW_KF_NAME VARCHAR(100),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_KOLICH_REPAY INTEGER,
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    IS_ZERO_PERC INTEGER,
    BY_CLIENT_BLANK_NO INTEGER,
    BY_GROUP_RU_BLANK_NO INTEGER,
    BY_GROUP_KG_BLANK_NO INTEGER)
AS
declare variable kd_percent decimal(18,7);
begin
  for select k_id, k_kontract_no, k_disb_date, k_amount_approved, pr_name, kal_new_kf_name, k_kolich_client_approved,
             k_kolich_repay, k_clients_cashier_id, coalesce(kd_percent, 0.0)
      from kontract, kontdata, knaktlist, product
      where kd_from_k_id = k_id and
            kd_is_primary = 0 and
            kd_ol_is_delete = 0 and
            kd_ol_date = :rep_date and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = :rep_region_id and
            k_from_pr_id = pr_id
      into k_id, k_kontract_no, k_disb_date, k_amount_approved, pr_name, kal_new_kf_name, k_kolich_client_approved,
           k_kolich_repay, c_id, kd_percent
  do
  begin
    is_zero_perc = 0;
    if (kd_percent = 0) then
      is_zero_perc = 1;
    c_name_fio = '';
    if (c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :c_id
      into :c_name_fio;
    end

    if (is_zero_perc = 1) then -- если это обнуление
    begin
      by_client_blank_no = 49;

      if (k_kolich_client_approved = 1) then --  если индив. кредит
      begin
        by_group_ru_blank_no = 67; -- на русском
        by_group_kg_blank_no = 68; -- на кыргызском
      end
      else
      begin
        by_group_ru_blank_no = 50; -- на русском
        by_group_kg_blank_no = 51; -- на кыргызском
      end
    end
    else -- если это снижение %
    begin
      by_client_blank_no = 57;
      if (k_kolich_client_approved = 1) then --  если индив. кредит
      begin
        by_group_ru_blank_no = 63; -- на русском
        by_group_kg_blank_no = 64; -- на кыргызском
      end
      else -- если групповой кредит
      begin
        by_group_ru_blank_no = 58; -- на русском
        by_group_kg_blank_no = 59; -- на кыргызском
      end
    end

    suspend;
  end
end
