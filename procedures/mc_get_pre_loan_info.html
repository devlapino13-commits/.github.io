CREATE PROCEDURE MC_GET_PRE_LOAN_INFO(
    MCL_ID D_ID,
    LOAN_AMOUNT D_INTEGER,
    LOAN_PERIOD D_INTEGER,
    FIRST_PAYMENT_DATE D_DATE)
RETURNS (
    JSON_RES D_BLOB_TEXT)
AS
declare variable PR_ID D_ID;
declare variable LOAN_DISB_DATE D_DATE;
declare variable LOAN_REPAY_DAY D_INTEGER;
declare variable MIN_AMOUNT D_INTEGER;
declare variable MAX_AMOUNT D_INTEGER;
declare variable MIN_PERIOD D_INTEGER;
declare variable MAX_PERIOD D_INTEGER;
declare variable ERR_TEXT D_STR_255;
declare variable C_ID D_ID;
declare variable K_NSP_PERCENT D_PERCENT;
declare variable LOAN_PERCENT D_INTEGER;
declare variable JSON_TMP D_BLOB_TEXT;
declare variable SH_ID D_INTEGER;
declare variable SH_DATE D_DATE;
declare variable SH_OS D_INTEGER;
declare variable SH_PERC D_INTEGER;
declare variable SH_TOTAL D_INTEGER;
declare variable SH_KOMMIS D_INTEGER;
declare variable SH_PERC_TAX D_INTEGER;
declare variable SH_TOTAL_WITH_NSP D_INTEGER;
declare variable MONTHLY_PAYMENT D_INTEGER;
declare variable SUM_SH_TOTAL D_INTEGER;
declare variable IS_LOYAL D_SMALLINT;
declare variable BASE_PERCENT D_INTEGER;
declare variable LOYAL_IS_0_TEXT D_STR_LONG;
declare variable LOYAL_IS_1_TEXT D_STR_LONG;
declare variable PERCENT_RATE_INFO D_STR_LONG;
declare variable MCL_LANG_CODE D_STR_LANG;
begin
  pr_id = (select pr_id
           from product pr
           where pr.pr_code containing '.online_loan.'
           order by pr_id
           rows 1);

  loan_disb_date = current_date;
  loan_repay_day = extract(day from :first_payment_date);

  select mcl.mcl_from_c_id,
         mcl.mcl_lang_code
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id,
       mcl_lang_code;

  select
    t.perc_max,  --Макс. и мин. % ставка равны
    period_max,
    period_min
  from hp_get_percent(:pr_id, current_date, 1, :loan_amount, :loan_period, :loan_amount, :loan_amount, :c_id, 0, 0, 6) t
  into
    loan_percent,
    max_period,
    min_period;

  select prl.prl_amount_cl_min, prl.prl_amount_cl_max, prl.prl_percent_max
  from prodlogic prl
  where prl.prl_from_pr_id = :pr_id and
        current_date between prl.prl_date_begin and prl.prl_date_end
  into min_amount, max_amount, base_percent;

  err_text = '';
  if (datediff(day, current_date, :first_payment_date) < 21 or datediff(day, current_date, :first_payment_date) > 45) then
    err_text = f_lang(:mcl_lang_code, 'Первая дата погашения займа должна быть не меньше 21 дней и не больше 45 дней.');

  if (loan_period < min_period or loan_period > max_period) then
    err_text = f_lang(:mcl_lang_code, 'Некорректный срок займа. Минимальный срок займа: ') || :min_period || f_lang(:mcl_lang_code, ', максимальный срок займа: ') || :max_period || '.';

  if (loan_amount < min_amount or loan_amount > max_amount) then
    err_text = f_lang(:mcl_lang_code, 'Некорректная сумма займа. Минимальная сумма займа: ') || :min_amount || f_lang(:mcl_lang_code, ', максимальный сумма займа: ') || :max_amount || '.';

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :loan_disb_date and
                   nsd_date_end >= :loan_disb_date)) then
  begin
    k_nsp_percent = 0;
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 and -- нсп от процента
          nsd_date_begin <= :loan_disb_date and
          nsd_date_end >= :loan_disb_date
    into :k_nsp_percent;
  end

  is_loyal = iif(loan_percent = base_percent, 0, 1);

  loyal_is_1_text = f_lang(:mcl_lang_code, 'Вы надежный клиент, и для вас доступна выгодная базовая ставка; Окончательная эффективная ставка, включая все комиссии, будет указана в договоре.');
  loyal_is_0_text = f_lang(:mcl_lang_code, 'Показанная ставка - базовая, окончательная эффективная ставка по кредиту, включая все комиссии, будет указана в договоре на заём.');
  percent_rate_info = iif(is_loyal = 0, trim(loyal_is_0_text), trim(loyal_is_1_text));

  json_tmp = '';
  sum_sh_total = 0;
  for select sh_id, sh_date, sh_os, sh_perc, sh_total, sh_kommis,
             (select res
              from fhp_round_bank(((:sh_perc + :sh_kommis) * :k_nsp_percent / 100.0), 0)) as sh_perc_tax
      from hp_make_schedule_by_param(:loan_disb_date, :pr_id, :loan_amount, :loan_percent, 0, :loan_period, :loan_repay_day)
      into :sh_id, :sh_date, :sh_os, :sh_perc, :sh_total, :sh_kommis, :sh_perc_tax
  do
  begin
    if (sh_id = 0) then
      continue;

    sh_total_with_nsp = sh_total + coalesce(sh_perc_tax, 0);
    sum_sh_total = sum_sh_total + sh_total_with_nsp;

    json_tmp = json_tmp || '{ "number": ' || sh_id || ', ' || '"date": "' || f_datetostr(sh_date) || '", ' ||
                           '"os": ' || sh_os || ', ' || '"percent": ' || sh_perc || ', '  ||
                           '"nsp": ' || coalesce(sh_perc_tax, 0) || ',' || '"total": ' || sh_total_with_nsp || ' },';
  end

  json_tmp = left(json_tmp, character_length(json_tmp) - 1);

  monthly_payment = sum_sh_total / loan_period;

  json_res = '{' ||
                '"amount": ' || :loan_amount || ',' ||
                '"period": ' || :loan_period || ',' ||
                '"first_payment_date": "' || f_datetostr(:first_payment_date) || '",' ||
                '"pay_day": ' || :loan_repay_day || ',' ||
                '"disb_date": "' || f_datetostr(current_date) || '",' ||
                '"percent_rate": ' || :loan_percent || ',' ||
                '"base_percent_rate": ' || :base_percent || ',' ||
                '"percent_rate_info": "' || :percent_rate_info || '",' ||
                '"is_loyal": ' || :is_loyal || ',' ||
                '"monthly_payment": ' || :monthly_payment || ',' ||
                '"kommission": ' || :sh_kommis || ',' ||
                '"error_text": "' || :err_text || '",' ||
                '"payments": [' || :json_tmp || ']' ||
             '}';
  suspend;
end
