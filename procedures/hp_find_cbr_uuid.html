CREATE PROCEDURE HP_FIND_CBR_UUID(
    UUID_STR D_STR_CODE)
RETURNS (
    RES_UUID D_UUID,
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
AS
declare variable num d_integer;
begin
  uuid_str = left(uuid_str, 36);
  res_uuid = char_to_uuid(uuid_str);

  num = 0;
  while (num <= 15) do
  begin
    if (num <= 9) then
      uuid_str = left(uuid_str, 14) || num || substring(uuid_str from 16);
    else
      uuid_str = left(uuid_str, 14) || (ascii_char(55 + num)) || substring(uuid_str from 16);

    res_uuid = char_to_uuid(uuid_str);

    num = num + 1;

    select udl_object_kind, udl_object_id
    from uuidlist
    where udl_uuid = :res_uuid
    into :obj_kind, :obj_id;

    if (row_count = 1) then
      break;
  end

  suspend;
end
