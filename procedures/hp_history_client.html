CREATE PROCEDURE HP_HISTORY_CLIENT(
    C_ID ID_TYPE,
    POINT_ID INTEGER,
    MODE STR_20 = 'anketa')
RETURNS (
    RES_DATE D_DATETIME,
    RES_USER STR_100,
    RES_VALUE STR_255)
AS
begin
  if (mode = 'anketa' and point_id > 0) then
    for select ovs_date_begin, un_fio, iif(ovr_value <> '', rd_name || ': ' || ovr_value, rd_name)
        from objectprops
        inner join objectvalues on ovs_from_opr_id = opr_id
        inner join objectvalres on ovr_from_opr_id = opr_id and
              ovr_from_ovs_id = ovs_id
        inner join refdata on rd_id = ovr_result_rd_id
        inner join username on un_id = ovs_from_un_id
        where opr_object_kind = 0 and
              opr_object_id = :c_id and
              opr_subject_kind = -1 and
              opr_subject_id = :point_id
        order by ovs_id desc
        into :res_date, :res_user, :res_value
    do
      suspend;

  if (mode = 'address') then
    for with
    -- Сначала собираем текущую запись addrclient, и записи из логов по этому типу адреса
    adl_base as (
      -- нельзя брать adl_datetime меньше чем adc_date, т.к. это дата вставки записи в addrclient
      select adc_from_adl_id as adl_base_id, adc_date as min_date, timestamp'31.12.9999' as max_date from addrclient
      where adc_from_c_id = :c_id and
            adc_address_type = :point_id
      union
      -- нельзя брать adl_datetime больше чем adcl_datetime, т.к. с этой даты актуальный adl уже другой
      select adcl_from_adl_id, date'01.01.1900', adcl_datetime from addrclog
      where adcl_from_c_id = :c_id and
            adcl_address_type = :point_id
    ),
    -- Для найденных выше adl берем данные из addrlist, и логов addrloglist
    adl_ext as (
      select adl_address_str, adl_datetime, adl_from_un_id
      from addrlist, adl_base
      where adl_id = adl_base_id
        and adl_datetime between min_date and max_date
      union
      select adl_address_str, adl_datetime, adl_from_un_id
      from addrloglist, adl_base
      where adl_id = adl_base_id
        and adl_datetime between min_date and max_date
    )
    -- Собираем итоговые данные
    select adl_datetime, un_fio, adl_address_str
    from adl_ext, username
    where un_id = adl_from_un_id
    order by adl_datetime desc
    into :res_date, :res_user, :res_value
    do
      suspend;

    if (mode = 'anketa' and point_id < 0) then
      for select res_date, res_user, res_value
      from hp_deport_status_history (:c_id)
      into :res_date, :res_user, :res_value
      do
        suspend;
end
