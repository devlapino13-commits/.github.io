CREATE PROCEDURE AAREP_CHECK_CLIENT_FIO(
    FIND_C_ID DECIMAL(18,0),
    FIND_FIO VARCHAR(255),
    FIND_PASS_NO VARCHAR(30),
    FIND_INN VARCHAR(20),
    CHECK_BLACK_LIST INTEGER,
    FIND_BIRTH_DATE DATE,
    FIND_CELL_NO VARCHAR(20),
    IS_CRITICAL_FIO SMALLINT,
    FIND_DRL_ID D_ID = null,
    FIND_CITIZEN INTEGER = 0,
    IS_ALL_CHECK D_BOOL = 0)
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(255),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_INN VARCHAR(20),
    C_COMMENT VARCHAR(100),
    C_COMMENT_ID INTEGER,
    C_ADDRESS VARCHAR(255),
    C_PLASTCARD_NO VARCHAR(20),
    C_SEX INTEGER)
AS
declare variable find_c_birth_date date;
declare variable find_fio_orig varchar(100);
declare variable find_pass_no_orig varchar(30);
declare variable txt1 varchar(20);
declare variable txt2 varchar(20);
declare variable txt3 varchar(20);
declare variable ss_text varchar(100);
declare variable suspended integer;
declare variable fp_birth_date date;
declare variable found_fp integer;
declare variable c_comment_tmp varchar(255);
declare variable find_fio2 varchar(255);
declare variable find_fio3 varchar(255);
declare variable finpolis_comment varchar(100);
declare variable pos d_integer;
begin

  pos = position(ascii_char(0), find_fio);
  if (pos > 0) then find_fio = left(find_fio, pos-1);

  if (coalesce(find_drl_id, 0) <> 0) then
  begin
    find_fio = f_docrecog_value(:find_drl_id, 'calc_name_full', '');
    find_birth_date = f_docrecog_value(:find_drl_id, 'birth_date', null);
    find_inn = f_docrecog_value(:find_drl_id, 'inn', '');
    find_pass_no = f_docrecog_value(:find_drl_id, 'doc_no', '');
    select pk_country_id
    from docrecoglist, passkind
    where drl_id = :find_drl_id and
          pk_id = drl_from_pk_id
    into find_citizen;
  end

  if (find_pass_no is null) then
    find_pass_no = '';
  if (find_fio is null) then
    find_fio = '';
  if (find_cell_no is null) then
    find_cell_no = '';

  find_fio_orig = :find_fio;

  c_comment_id = 0;
  is_critical_fio = coalesce(is_critical_fio, 0);

  if (find_c_id > 0) then
  begin
    select iif(:find_fio = '', c_name_fio, :find_fio), iif(:find_pass_no = '', c_pass_no, :find_pass_no),
           iif(:find_inn = '', c_inn, :find_inn), iif(:find_cell_no = '',
           (select cont_value
            from hp_contdata_get_cell_main(0, :find_c_id)), :find_cell_no),
           iif(:find_c_birth_date is null, c_birth_date, :find_c_birth_date)
    from clients
    where c_id = :find_c_id
    into :find_fio, find_pass_no, find_inn, find_cell_no, find_c_birth_date;

    find_fio_orig = :find_fio;
  end
  else
    find_c_birth_date = :find_birth_date;

  if (right(find_cell_no, 6) = '000000') then
    find_cell_no = '';

  if (find_pass_no is null) then
    find_pass_no = '';
  if (find_inn is null) then
    find_inn = '';

  find_pass_no_orig = :find_pass_no;
  while (position(' ', find_pass_no) > 0) do
    find_pass_no = replace(:find_pass_no, ' ', '');
  find_fio = upper(:find_fio);
  while (position(' ', find_inn) > 0) do
    find_inn = replace(:find_inn, ' ', '');
  suspended = 0;

-- Сначала проверяем по СРФ
  c_comment = '';
  c_comment_id = 0;
  find_fio2 = find_fio;
  find_fio3 = find_fio;
  find_fio2 = replace(find_fio, '-', ' ');
  find_fio2 = f_replace_all(find_fio2, '  ', ' ');
  find_fio3 = replace(find_fio, ' уулу', '-уулу');
  find_fio3 = replace(find_fio3, ' кызы', '-кызы');
  find_fio3 = f_replace_all(find_fio3, '  ', ' ');
  for select distinct fpp_fio, sd, rg, nd, gr
      from finpolisparced, finpolis
      where fpp_from_fp_id = fp_id and
            (fpp_fio = :find_fio or
            fpp_fio = :find_fio2 or
            fpp_fio = :find_fio3)
       and not exists(
                       select 1
                       from refdata rd
                       where rd.rd_id = finpolis.terr_categ_id
                         and rd.rd_main_code = 'FionpolisDirectory'
                         and coalesce(rd.rd_memo,'') = iif(:is_all_check = 1, '#all#', 'soft_check')
                     )

      into :c_name_fio, :txt1, :txt2, :txt3, :fp_birth_date
  do
  begin
    if (c_comment_id = 0) then
    begin
      if (is_critical_fio = 1) then
        c_comment_id = c_comment_id + 1;
      c_comment = 'РФМ (СФР): ФИО';
      if (txt1 is null) then
        txt1 = '';
      if (txt2 is null) then
        txt2 = '';
      if (txt3 is null) then
        txt3 = '';
      ss_text = txt1 || txt2;
      ss_text = replace(ss_text, ' ', '');
      if (find_pass_no <> '') then
      begin
        if (ss_text = find_pass_no) then
        begin
          c_comment_id = c_comment_id + 1;
          c_pass_no = find_pass_no;
          c_comment = c_comment || ', Паспорт';
        end
      end
      if (find_inn <> '') then
      begin
        ss_text = replace(txt3, ' ', '');
        if (ss_text = find_inn) then
        begin
          c_comment_id = c_comment_id + 1;
          c_inn = find_inn;
          c_comment = c_comment || ', ИНН';
        end
      end
      if (find_c_birth_date = fp_birth_date) then
      begin
        c_comment_id = c_comment_id + 1;
        c_birth_date = fp_birth_date;
        c_comment = c_comment || ', д.р.';
      end
      if (c_comment_id > 0) then
      begin
        c_comment_id = 1;
        suspend;
      end
      suspended = 1;
      c_pass_no = :find_pass_no_orig;
      c_inn = :find_inn;
    end
  end
  if (suspended = 0) then
  begin
    for select distinct nameu, sd, rg, nd, gr
        from finpolis
        where not exists(
                           select 1
                           from refdata rd
                           where rd.rd_id = finpolis.terr_categ_id
                             and rd.rd_main_code = 'FionpolisDirectory'
                             and coalesce(rd.rd_memo,'') = iif(:is_all_check = 1, '#all#', 'soft_check')
                        )

        into c_name_fio, :txt1, :txt2, :txt3, :fp_birth_date
    do
    begin
      if (c_comment_id = 0) then
      begin
        found_fp = 0;
        c_comment = '';
        if (txt1 is null) then
          txt1 = '';
        if (txt2 is null) then
          txt2 = '';
        if (txt3 is null) then
          txt3 = '';
        ss_text = txt1 || txt2;
        ss_text = replace(ss_text, ' ', '');
        if (find_pass_no <> '') then
        begin
          if (ss_text = find_pass_no) then
          begin
            if (c_comment = '') then
              c_comment = 'РФМ (СФР):';
            found_fp = found_fp + 1;
            c_pass_no = find_pass_no;
            c_comment = c_comment || ', Паспорт';
          end
        end
        ss_text = replace(txt3, ' ', '');
        if (find_inn <> '') then
        begin
          if (ss_text = find_inn) then
          begin
            if (c_comment = '') then
              c_comment = 'РФМ (СФР):';
            found_fp = found_fp + 1;
            c_inn = find_inn;
            c_comment = c_comment || ', ИНН';
          end
        end
        if (found_fp > 0) then
        begin
          if (find_c_birth_date = fp_birth_date) then
          begin
            if (c_comment = '') then
              c_comment = 'РФМ (СФР):';
            found_fp = found_fp + 1;
            c_birth_date = fp_birth_date;
            c_comment = c_comment || ', д.р.';
          end
        end
        if (found_fp > 1) then
        begin
          c_comment_id = 1;
          c_pass_no = :find_pass_no_orig;
          c_inn = :find_inn;
        end
      end
    end
  end
  finpolis_comment = c_comment;
  c_comment = '';

  if (check_black_list = 1) then
  begin
    c_comment = '';
    c_comment_id = null;
    select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_comment, c_comment_id, c_address, c_plastcard_no, c_sex
    from hp_blacklist_search(0, :find_fio, :find_c_birth_date, :find_pass_no, :find_inn, :find_cell_no, 1, 1, 0)
    order by c_comment_id desc
    rows 1
    into :c_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn, :c_comment_tmp, :c_comment_id, :c_address,
         :c_plastcard_no, c_sex;
    if (c_comment_id is not null) then
    begin
      if (c_comment = '') then
        c_comment = 'СТОП ЛИСТ' || c_comment_tmp;
      else
        c_comment = c_comment ||'. СТОП ЛИСТ '|| c_comment_tmp;
      suspend;
    end
    else
    begin
      if ((find_pass_no_orig <> '')) then
      begin
        if (exists(select *
                   from moratory
                   where m_client_pass_no starting with :find_pass_no_orig and
                         m_is_active = 1 and
                         m_is_black > 0)) then
        begin
          c_name_fio = :find_fio_orig;
          c_pass_no = :find_pass_no_orig;
          c_inn = :find_inn;
          if (c_comment = '') then
            c_comment ='паспорт в СТОП ЛИСТЕ';
          else
            c_comment ='паспорт в СТОП ЛИСТЕ'|| ' ' || c_comment;
          c_comment_id = 1;
          suspend;
        end
      end
    end
    if (c_comment_id is null) then
      c_comment_id = 0;
  end
  if ((c_comment <> '') and
      (c_comment_id = 1)) then
    finpolis_comment = finpolis_comment || ', ' || c_comment;

  for select c_id, is_critical, res_comment, c_name_fio, c_pass_no, c_birth_date, c_inn, c_full_address,
             c_plastcard_no
      from hp_find_clients(:find_c_id, :find_fio, :find_c_birth_date, :find_inn, null, :find_pass_no, :find_cell_no, 0, :find_citizen)
      into :c_id, :c_comment_id, :c_comment, :c_name_fio, :c_pass_no, :c_birth_date, :c_inn, :c_address,
           :c_plastcard_no
  do
  begin
    suspend;
  end

  c_id = null;
  if (check_black_list = 1) then
  begin
    c_comment = '';
    c_comment_id = null;
    select c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_comment, c_comment_id, c_address, c_plastcard_no, c_sex
    from hp_blacklist_search(0, :find_fio, :find_c_birth_date, :find_pass_no, :find_inn, :find_cell_no, 1, 1, 0)
    order by c_comment_id desc
    rows 1
    into :c_id, :c_name_fio, :c_birth_date, :c_pass_no, :c_inn, :c_comment_tmp, :c_comment_id, :c_address,
         :c_plastcard_no, c_sex;
    if (c_comment_id is not null) then
    begin
      if (c_comment = '') then
        c_comment = 'СТОП ЛИСТ ' || c_comment_tmp;
      else
        c_comment = c_comment || '. СТОП ЛИСТ ' || c_comment_tmp;
      suspend;
    end
    else
    begin
      if ((find_pass_no_orig <> '')) then
      begin
        if (exists(select *
                   from moratory
                   where m_client_pass_no starting with :find_pass_no_orig and
                         m_is_active = 1 and
                         m_is_black > 0)) then
        begin
          c_name_fio = :find_fio_orig;
          c_pass_no = :find_pass_no_orig;
          c_inn = :find_inn;
          if (c_comment = '') then
            c_comment = 'паспорт в СТОП ЛИСТЕ';
          else
            c_comment = 'паспорт в СТОП ЛИСТЕ' || ' ' || c_comment;
          c_comment_id = 1;
          suspend;
        end
      end
    end
    if (c_comment_id is null) then
      c_comment_id = 0;
  end
  c_id = null;
  c_sex = 0;

  if (c_comment_id = 0) then
  begin
    c_comment = '';
    c_comment_id = 0;
    find_fio2 = find_fio;
    find_fio3 = find_fio;
    find_fio2 = replace(find_fio, '-', ' ');
    find_fio2 = f_replace_all(find_fio2, '  ', ' ');
    find_fio3 = replace(find_fio, ' уулу', '-уулу');
    find_fio3 = replace(find_fio3, ' кызы', '-кызы');
    find_fio3 = f_replace_all(find_fio3, '  ', ' ');
    for select distinct fpp_fio, sd, rg, nd, gr
        from finpolisparced, finpolis
        where fpp_from_fp_id = fp_id and
              (fpp_fio = :find_fio or
              fpp_fio = :find_fio2 or
              fpp_fio = :find_fio3)
         and not exists(
                           select 1
                           from refdata rd
                           where rd.rd_id = finpolis.terr_categ_id
                             and rd.rd_main_code = 'FionpolisDirectory'
                             and coalesce(rd.rd_memo,'') = iif(:is_all_check = 1, '#all#', 'soft_check')
                        )

        into :c_name_fio, :txt1, :txt2, :txt3, :fp_birth_date
    do
    begin
      if (c_comment_id = 0) then
      begin
        c_comment = 'РФМ (СФР): ФИО';
        if (txt1 is null) then
          txt1 = '';
        if (txt2 is null) then
          txt2 = '';
        if (txt3 is null) then
          txt3 = '';
        ss_text = txt1 || txt2;
        ss_text = replace(ss_text, ' ', '');
        if (find_pass_no <> '') then
        begin
          if (ss_text = find_pass_no) then
          begin
            c_comment_id = c_comment_id + 1;
            c_pass_no = find_pass_no;
            c_comment = c_comment || ', Паспорт';
          end
        end
        if (find_inn <> '') then
        begin
          ss_text = replace(txt3, ' ', '');
          if (ss_text = find_inn) then
          begin
            c_comment_id = c_comment_id + 1;
            c_inn = find_inn;
            c_comment = c_comment || ', ИНН';
          end
        end
        if (find_c_birth_date = fp_birth_date) then
        begin
          c_comment_id = c_comment_id + 1;
          c_birth_date = fp_birth_date;
          c_comment = c_comment || ', д.р.';
        end
        if (c_comment_id > 0) then
        begin
          c_comment_id = 1;
        end
        suspended = 1;
        c_pass_no = :find_pass_no_orig;
        c_inn = :find_inn;
        suspend;
      end
    end
    if (suspended = 0) then
    begin
      for select distinct nameu, sd, rg, nd, gr
          from finpolis
          where not exists(
                             select 1
                             from refdata rd
                             where rd.rd_id = finpolis.terr_categ_id
                               and rd.rd_main_code = 'FionpolisDirectory'
                               and coalesce(rd.rd_memo,'') = iif(:is_all_check = 1, '#all#', 'soft_check')
                          )

          into c_name_fio, :txt1, :txt2, :txt3, :fp_birth_date
      do
      begin
        if (c_comment_id = 0) then
        begin
          found_fp = 0;
          c_comment = '';
          if (txt1 is null) then
            txt1 = '';
          if (txt2 is null) then
            txt2 = '';
          if (txt3 is null) then
            txt3 = '';
          ss_text = txt1 || txt2;
          ss_text = replace(ss_text, ' ', '');
          if (find_pass_no <> '') then
          begin
            if (ss_text = find_pass_no) then
            begin
              if (c_comment = '') then
                c_comment = 'РФМ (СФР):';
              found_fp = found_fp + 1;
              c_pass_no = find_pass_no;
              c_comment = c_comment || ', Паспорт';
            end
          end
          ss_text = replace(txt3, ' ', '');
          if (find_inn <> '') then
          begin
            if (ss_text = find_inn) then
            begin
              if (c_comment = '') then
                c_comment = 'РФМ (СФР):';
              found_fp = found_fp + 1;
              c_inn = find_inn;
              c_comment = c_comment || ', ИНН';
            end
          end
          if (found_fp > 0) then
          begin
            if (find_c_birth_date = fp_birth_date) then
            begin
              if (c_comment = '') then
                c_comment = 'РФМ (СФР):';
              found_fp = found_fp + 1;
              c_birth_date = fp_birth_date;
              c_comment = c_comment || ', д.р.';
            end
          end
          if (found_fp > 1) then
          begin
            c_comment_id = 1;
            c_pass_no = :find_pass_no_orig;
            c_inn = :find_inn;
            if (found_fp > 1) then
              suspend;
          end
        end
      end
    end
  end
end
