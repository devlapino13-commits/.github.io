CREATE PROCEDURE HP_UPDATE_BLOBSTORE(
    BLS_ID D_ID,
    BLS_FILENAME D_STR_100,
    BLS_CAPTION D_STR_100,
    BLS_BLOB D_BLOB_BINARY,
    BLS_FROM_UN_ID ID_TYPE,
    BLS_KIND D_INTEGER,
    BLS_SOURCE_SYSTEM D_INTEGER,
    BLS_FROM_MCL_ID D_ID = 0)
AS
declare variable bls_blob_size d_float;
declare variable db_path d_str_255;
declare variable v_bls_id d_id;
begin

  bls_blob_size = round(octet_length(bls_blob) / 1024.000, 3);

  bls_from_mcl_id = coalesce(bls_from_mcl_id, 0);

  update blobstore
  set bls_filename = :bls_filename,
      bls_caption = :bls_caption,
      bls_blob = null,
      bls_blob_size = :bls_blob_size,
      bls_from_un_id = :bls_from_un_id,
      bls_from_mcl_id = :bls_from_mcl_id,
      bls_kind = :bls_kind,
      bls_source_system = :bls_source_system
  where (bls_id = :bls_id)
  returning bls_id
  into :v_bls_id;
  bls_id = abs(bls_id);

  if (v_bls_id is null) then
    exception e_default ('Record with "bls_id '||coalesce('= '||bls_id,'is <NULL>')||'" not found');

  -- Обновление в EXT BLOBSTORE
  select db_path
  from hp_db_is_actual
  into db_path;
  if (db_path is not null) then
  begin
    execute statement('update blobextore set bls_blob = :bls_blob where bls_id = :bls_id')
                      (bls_id := :bls_id, bls_blob := :bls_blob)
        on external data source db_path
        as user 'UBLOBSTORE' password 'qJdCv55GcYpnwqBf';

    if (:bls_blob is not null) then --если Blob не пустой, то поменять знак в bls_id
      update blobstore
      set bls_id = :bls_id
      where (bls_id = -:bls_id);

    when any do
    begin
      exception e_default 'Blobstore database error: update';
    end
  end
  else
    update blobstore
    set bls_id = :bls_id,
        bls_blob = :bls_blob
    where bls_id = :bls_id;

end
