CREATE PROCEDURE HP_NBKR_NOTIF_LIST_SENT
AS
declare variable NUM_PROCESSING D_INTEGER;
declare variable NUM_ADDED D_INTEGER;
declare variable RES_TEXT D_STR_255;
declare variable C_ID D_ID;
declare variable NNL_ID D_INTEGER;
declare variable KC_ID D_ID;
declare variable K_ID D_ID;
declare variable NNL_AMOUNT D_CURRENCY;
declare variable KS_STATE D_INTEGER;
declare variable EDC_ID D_INTEGER;
declare variable EDC_CODE D_STR_100;
begin
   if (extract(hour from current_timestamp) < 9 or
     extract(hour from current_timestamp) > 21) then
   exit;

  if (f_get_db_id() not in (39801, 41701)) then
    exit;

  num_processing = (
    select count(*)
    from edoc_blank, edoc_list
    where edb_from_edl_id = edl_id
      and edb_status = 0
      and edl_status <> -1);

  num_added = 0;

  if (num_processing < 50) then
  begin
    for
      select first 250
          t.nnl_id,
          nnl_from_kc_id,
          nnl_k_id,
          t.nnl_amount
      from nbkr_notif_list t
      where t.nnl_mark = 0
    into
      :nnl_id,
      :kc_id,
      :k_id,
      :nnl_amount
    do
    begin
         ks_state = 0;
         edc_id = -1;
    
         select ks.ks_state
         from kontstate ks
         where ks.ks_from_k_id = :k_id
         and current_date between ks.ks_date_begin and ks.ks_date_end
         and ks.ks_ol_is_delete = 0
         into :ks_state;
    
         if (ks_state = 1) then
         begin
           if (nnl_amount > 0) then
                edc_id = 24;
           else if (nnl_amount < 0) then
                edc_id = 22;
         end
        
         if (ks_state = 2) then
         begin
           if (nnl_amount > 0) then
                edc_id = 23;
           else if (nnl_amount < 0) then
                edc_id = 21;
         end
    
         update nbkr_notif_list t
         set t.nnl_from_edc_id = :edc_id
         where t.nnl_id =:nnl_id;

         res_text = '';

         if (edc_id <> -1) then
         begin
             select edc.edc_code
             from edoc_cfg edc
             where edc.edc_id = :edc_id
             into :edc_code;

             select res_text
             from hp_edoc_ins(:edc_code, '8', :KC_ID, 1000000000000)
             into :res_text;
         end

         update nbkr_notif_list t
         set t.nnl_mark = iif(:res_text = '', 1, -1)
         where t.nnl_from_kc_id = :kc_id;

         num_added = num_added + 1;
    end
    if (num_added = 0) then
      update fsrobot_sh_main set fsm_is_active = 0 where fsm_id = 68;
  end
end
