CREATE PROCEDURE HP_REPEAT_REPAYMENT_SAVE(
    ORP_ID D_INTEGER,
    ORP_CHANGE_LOG D_BLOB_TEXT,
    UN_ID D_ID)
AS
declare variable k_id d_id;
declare variable start_date d_date;
declare variable reverse_date d_date;
declare variable loan_reverse d_smallint;
declare variable book_reverse d_smallint;
declare variable sql_text d_str_8100;
declare variable orp_state d_smallint;
declare variable orp_state_str d_str_code;
begin
  -- Процедура пере-принимает погашения по займу начиная с даты START_DATE
  -- Операции сторнирования будут сохранены датой REVERSE_DATE

  select orp_from_k_id, orp_ol_date, orp_begin_date, orp_book_reverse, orp_loan_reverse, orp_sql, orp_state
  from operrepeat
  where orp_id = :orp_id
  into :k_id, :reverse_date, :start_date, :book_reverse, :loan_reverse, :sql_text, orp_state;

  if (row_count = 0) then
    exception e_default f_format('В базе не найден запрос: %1', coalesce(orp_id, 'Null'));

  orp_state_str = f_enum('operrepeat_state', orp_state, 'code');
  if (orp_state_str <> 'checked') then
  begin
    if (coalesce(orp_change_log, '') = '') then
      exception e_message f_format('Для сохранения запрос должен иметь статус "Проверен", сейчас статус: "%1"',
                                    f_enum('operrepeat_state', orp_state, 'name'));
    else
      execute procedure hp_operrepeat_set_status(:orp_id, f_enum('operrepeat_state', 'checked', 'id'), :un_id);
  end

  execute procedure hp_repeat_repayment_tmp(:orp_id, :k_id, :start_date, :reverse_date, :loan_reverse, :book_reverse, :sql_text, :un_id);

  update operrepeat
  set orp_change_log = :orp_change_log
  where orp_id = :orp_id;

  execute procedure hp_operrepeat_set_status(:orp_id, f_enum('operrepeat_state', 'done', 'id'), :un_id);
end
