CREATE PROCEDURE AAREP_CLIENTS_ANALIZ(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    LOAN_TYPE_TXT VARCHAR(20),
    BARANCH_NAME VARCHAR(50),
    OFFICE_NAME VARCHAR(50),
    CLOSE_CL_NOT_DELINQ INTEGER,
    CLOSE_CL_DELINQ_1_7 INTEGER,
    CLOSE_CL_DELINQ_8_15 INTEGER,
    CLOSE_CL_DELINQ_16_30 INTEGER,
    CLOSE_CL_DELINQ_30_ INTEGER,
    CLOSE_CL_WOFF_OS INTEGER,
    CLOSE_CL_WOFF_PERC INTEGER,
    CLOSE_CL_WOFF_FINE INTEGER,
    CLOSE_CL_NOT_DELINQ_DISB INTEGER,
    CLOSE_CL_DELINQ_1_7_DISB INTEGER,
    CLOSE_CL_DELINQ_8_15_DISB INTEGER,
    CLOSE_CL_DELINQ_16_30_DISB INTEGER,
    CLOSE_CL_DELINQ_30__DISB INTEGER,
    CLOSE_CL_WOFF_OS_DISB INTEGER,
    CLOSE_CL_WOFF_PERC_DISB INTEGER,
    CLOSE_CL_WOFF_FINE_DISB INTEGER,
    CLOSE_CL_BEFORE_PERIOD_DISB INTEGER)
AS
begin

  for select loan_type_txt, baranch_name, office_name,
             sum(iif(is_woff = 0 and before_period_disb = 0 and delinq_day = 0, 1, 0)), --Закрыт. кл. без проср., без спис.
             sum(iif(is_woff = 0 and before_period_disb = 0 and delinq_day between 1 and 7, 1, 0)), --до 7 дней
             sum(iif(is_woff = 0 and before_period_disb = 0 and delinq_day between 8 and 15, 1, 0)), --от 8 до 15 дней
             sum(iif(is_woff = 0 and before_period_disb = 0 and delinq_day between 16 and 30, 1, 0)), --от 16 до 30 дней
             sum(iif(is_woff = 0 and before_period_disb = 0 and delinq_day > 30, 1, 0)), --более 30 дней
--       Закрыт. кл. со списанием
             sum(iif(is_woff = 1, is_os, 0)), -- ОС
             sum(iif(is_woff = 1, is_perc, 0)), -- Проценты
             sum(iif(is_woff = 1, is_fine, 0)), -- Пеня
--       Закрытые и повтороно выданные за указанный период
             sum(iif(in_period_disb = 1 and is_woff = 0 and delinq_day = 0, 1, 0)), -- без просрочек
             sum(iif(in_period_disb = 1 and is_woff = 0 and delinq_day between 1 and 7, 1, 0)), --до 7 дней
             sum(iif(in_period_disb = 1 and is_woff = 0 and delinq_day between 8 and 15, 1, 0)), --от 8 до 15 дней
             sum(iif(in_period_disb = 1 and is_woff = 0 and delinq_day between 16 and 30, 1, 0)), --от 16 до 30 дней
             sum(iif(in_period_disb = 1 and is_woff = 0 and delinq_day > 30, 1, 0)), --более 30 дней
--       Закрытые со списанием и повтороно выданные за указанный период
             sum(iif(in_period_disb = 1 and is_woff = 1, is_os, 0)), -- ОС
             sum(iif(in_period_disb = 1 and is_woff = 1, is_perc, 0)), -- Проценты
             sum(iif(in_period_disb = 1 and is_woff = 1, is_fine, 0)), -- Пеня
--       Повторн. закр. кред. до нач. периода, т.е. повторно выданные за указанные период которые закрылись до указанного пеирода
             sum(before_period_disb)
      from aarep_clients_analiz_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by baranch_name, office_name, loan_type_txt
      into loan_type_txt, baranch_name, office_name, close_cl_not_delinq, close_cl_delinq_1_7, close_cl_delinq_8_15,
           close_cl_delinq_16_30, close_cl_delinq_30_, close_cl_woff_os, close_cl_woff_perc, close_cl_woff_fine,
           close_cl_not_delinq_disb, close_cl_delinq_1_7_disb, close_cl_delinq_8_15_disb, close_cl_delinq_16_30_disb,
           close_cl_delinq_30__disb, close_cl_woff_os_disb, close_cl_woff_perc_disb, close_cl_woff_fine_disb,
           close_cl_before_period_disb
  do
  begin
    suspend;
  end
end
