CREATE PROCEDURE AAREP_RATING_OFFICE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_REGION_ID INTEGER,
    AA_CURR_OFFICE_NAME VARCHAR(50),
    AA_KOLICH_CL_ACTIVE_BEGIN INTEGER,
    AA_AMOUNT_ACTIVE_BEGIN CURR_TYPE,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_AMOUNT_ACTIVE_END CURR_TYPE,
    ROST_CLI INTEGER,
    ROST_BALANCE CURR_TYPE,
    DELINQ_BALANCE7_BEGIN CURR_TYPE,
    DELINQ_BALANCE7_END CURR_TYPE,
    PAR7_BEGIN DECIMAL(15,2),
    PAR7_END DECIMAL(15,2),
    PAR7_IZMEN DECIMAL(15,2))
AS
begin
  for select aa_curr_region_id, aa_curr_office_name, sum(aa_kolich_cl_active_begin), sum(aa_amount_active_begin),
             sum(aa_kolich_cl_active_end), sum(aa_amount_active_end)
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 1, 1, 0, 0, 0)
      where aa_amount_active_begin <> 0 or
            aa_amount_active_end <> 0
      group by aa_curr_region_id, aa_curr_office_name
      into aa_curr_region_id, aa_curr_office_name, aa_kolich_cl_active_begin, aa_amount_active_begin,
           aa_kolich_cl_active_end, aa_amount_active_end
  do
  begin
    if (aa_kolich_cl_active_begin is null) then
      aa_kolich_cl_active_begin = 0;
    if (aa_amount_active_begin is null) then
      aa_amount_active_begin = 0;
    if (aa_kolich_cl_active_end is null) then
      aa_kolich_cl_active_end = 0;
    if (aa_amount_active_end is null) then
      aa_amount_active_end = 0;

    rost_cli = aa_kolich_cl_active_end - aa_kolich_cl_active_begin;
    rost_balance = aa_amount_active_end - aa_amount_active_begin;

    select coalesce(sum(aarep_k_balance), 0)
    from aarep_kontract_id(:aa_curr_region_id, :rep_date_begin - 1, :rep_date_begin - 1, 0, '000', 5)
    where aarep_k_deliq_days > 7
    into delinq_balance7_begin;

    select coalesce(sum(aarep_k_balance), 0)
    from aarep_kontract_id(:aa_curr_region_id, :rep_date_end, :rep_date_end, 0, '000', 5)
    where aarep_k_deliq_days > 7
    into delinq_balance7_end;

    par7_begin = 0;
    par7_end = 0;
    if (aa_amount_active_begin <> 0) then
      par7_begin = round(delinq_balance7_begin / aa_amount_active_begin * 100, 2);
    if (aa_amount_active_end <> 0) then
      par7_end = round(delinq_balance7_end / aa_amount_active_end * 100, 2);

    par7_izmen = par7_end - par7_begin;

    suspend;
  end

end
