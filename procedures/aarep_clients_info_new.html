CREATE PROCEDURE AAREP_CLIENTS_INFO_NEW(
    REP_DATE_END DATE)
RETURNS (
    KS_FROM_K_ID DECIMAL(18,0),
    KS_DATE_BEGIN DATE,
    KS_DATE_END DATE,
    K_APPLICATE_DATE DATE,
    K_DISB_DATE DATE,
    K_KREDLINE_END DATE,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    PR_ID INTEGER,
    PR_LOAN_TYPE SMALLINT,
    PR_CODE VARCHAR(100),
    KC_STATUS SMALLINT,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_CIRCLE INTEGER,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_REPAY INTEGER,
    C_ID DECIMAL(18,0),
    C_SEX SMALLINT,
    C_BIRTH_DATE DATE,
    C_KOL_FAMILY INTEGER,
    C_KOL_CHILD INTEGER,
    C_NATION INTEGER,
    C_CITIZEN INTEGER,
    C_EDUCATION INTEGER,
    C_FAMMILY INTEGER,
    KAL_NEW_KF_OFFICER DECIMAL(18,0),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_CIRCLE INTEGER,
    BT_ID INTEGER,
    BS_ID INTEGER,
    REG_ID INTEGER,
    BRANCHE_ID INTEGER,
    DISTRICT_ID INTEGER,
    K_KONT_BALANCE DECIMAL(18,4),
    K_CLI_BALANCE DECIMAL(18,4),
    DELIQ_OS DECIMAL(18,4),
    DELIQ_PERC DECIMAL(18,4),
    DELIQ_DAY INTEGER,
    DELIQ_CLI_OS DECIMAL(18,4),
    DELIQ_CLI_PERC DECIMAL(18,4),
    STATUS_KONT VARCHAR(20),
    STATUS_DELINQ VARCHAR(20),
    KOL_CHECK_RESULT SMALLINT,
    DELIQ_1_3 INTEGER,
    DELIQ_4_10 INTEGER,
    DELIQ_10_ INTEGER,
    DELIQ_DAY_ALL_OLD INTEGER,
    DELIQ_DAY_ALL INTEGER,
    DELIQ_DAY_MAX INTEGER)
AS
declare variable old_k_id decimal(18,0);
declare variable ok_from_k_id decimal(18,0);
declare variable ok_sh_date date;
declare variable ok_os_deliq decimal(18,4);
declare variable ok_perc_deliq integer;
declare variable is_deliq smallint;
declare variable cli_dolya decimal(18,4);
declare variable ok_sh_date_min date;
declare variable deliq_day_old integer;
declare variable old_k_id_1 decimal(18,0);
declare variable old_officer_id decimal(18,0);
declare variable ks_date_end_fact date;
begin
-- oksummary_fix

  old_k_id = 0;
  old_k_id_1 = 0;
  k_cli_balance = 0;
  old_officer_id = 0;
  for select ks_from_k_id, ks_date_begin,
             case
               when ks_date_end < '31.12.9999' then ks_date_end + 1
             end,
             ks_date_end, k_applicate_date, k_disb_date, k_kredline_end, k_kolich_client_approved, k_amount_approved,
             k_circle, k_period, k_kolich_repay, pr_id, pr_loan_type, pr_code, kc_status, cli.c_id, cli.c_sex,
             cli.c_birth_date, cli.c_kol_family, cli.c_kol_child, cli.c_nation, cli.c_citizen, cli.c_education,
             cli.c_family, kal_new_kf_officer, kc_amount_approved, kc_circle, kc_business_type, kc_busuness_subtype,
             reg1.reg_unicode, reg2.reg_unicode, reg3.reg_unicode

      from kontract, product, kontstate, knaktlist, knclient, clients cli, region reg1
      left join region reg2 on (reg1.parent = reg2.reg_unicode)
      left join region reg3 on (reg2.parent = reg3.reg_unicode)
      where k_from_pr_id = pr_id and
            ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0
            /* and ks_date_end-1 between kal_date and kal_date_end */
            and
            (case
               when ks_date_end < '31.12.9999' then ks_date_end
               else :rep_date_end
             end) between kal_date and kal_date_end and
            kc_from_k_id = k_id and
            kc_amount_approved <> 0 and
            cli.c_id = kc_from_c_id and
            kal_new_reg_id = reg1.reg_unicode
      order by kal_new_kf_officer, ks_from_k_id

      into :ks_from_k_id, :ks_date_begin, :ks_date_end, :ks_date_end_fact, :k_applicate_date, :k_disb_date,
           :k_kredline_end, :k_kolich_client_approved, :k_amount_approved, :k_circle, :k_period, :k_kolich_repay,
           :pr_id, :pr_loan_type, :pr_code, :kc_status, :c_id, :c_sex, :c_birth_date, :c_kol_family, :c_kol_child,
           :c_nation, :c_citizen, :c_education, :c_fammily, :kal_new_kf_officer, :kc_amount_approved, :kc_circle,
           :bt_id, :bs_id, :reg_id, :branche_id, :district_id

  do
  begin

    if (:rep_date_end between :ks_date_begin and :ks_date_end_fact) then
    /*Если контракт активный делает следующее*/
    begin
      k_cli_balance = null;
      status_kont = 'Активный';
      /*Вычисления активного портфеля контракта :KS_FROM_K_ID*/
      if (old_k_id <> :ks_from_k_id) then
      /*Проверяет на повторность контракта, если контракт повторяеться то не будет повторно считать*/
      begin
        /*Вычисляет баланс контракта*/
        k_kont_balance = 0;
        select oks_balance
        from oksummary
        where oks_from_k_id = :ks_from_k_id and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_close_state = 0
        into :k_kont_balance;

        ok_sh_date = null;
        ok_os_deliq = null;
        ok_perc_deliq = null;

        deliq_os = 0;
        deliq_perc = 0;
        deliq_day = 0;
        deliq_day_max = 0;
        deliq_cli_os = null;
        deliq_cli_perc = null;

        status_delinq = '';
        ok_sh_date_min = :rep_date_end;
        is_deliq = 0;

        for /*Вычисляет просроченные контракты*/
            select ok_from_k_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
            from operkred
            where ok_ol_is_active = 1 and
                  ok_type < 7 and
                  ok_ol_date <= :rep_date_end and
                  ok_sh_date <= :rep_date_end and
                  ok_from_k_id = :ks_from_k_id and
                  ok_from_c_id = 0
            group by ok_from_k_id, ok_sh_date
            having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
            into :ok_from_k_id, :ok_sh_date, :ok_os_deliq, :ok_perc_deliq
        do
        begin
          if (:ok_os_deliq + :ok_perc_deliq > 0) then
          begin
            is_deliq = 1;
            if (ok_sh_date_min > :ok_sh_date) then
              ok_sh_date_min = :ok_sh_date;
          end
          deliq_os = deliq_os + ok_os_deliq;
          deliq_perc = deliq_perc + ok_perc_deliq;
        end

        if (is_deliq = 1) then
        begin
          deliq_day = cast((:rep_date_end - ok_sh_date_min) as integer);
          status_delinq = 'Просрочка';
          deliq_day_max = deliq_day;
        end
      end
      cli_dolya = :kc_amount_approved / :k_amount_approved;/*коэффицент доли клиента*/
      k_cli_balance = cli_dolya * :k_kont_balance;/*Вычисляет долю портфеля клиента*/
      deliq_cli_os = cli_dolya * :deliq_os;
      deliq_cli_perc = cli_dolya * deliq_perc;
      old_k_id = :ks_from_k_id;
    end
    else
    begin
      k_cli_balance = null;
      k_kont_balance = null;
      deliq_os = null;
      deliq_perc = null;
      deliq_day = null;
      ok_sh_date_min = null;
      deliq_cli_os = null;
      deliq_cli_perc = null;
      status_kont = 'Закрытый';
      status_delinq = null;
--              DELIQ_DAY_MAX = 0;
    end

    if (old_k_id_1 <> :ks_from_k_id) then
    /*Вычисляет дни просроченных выплат*/
    begin
      deliq_day_old = 0;
      deliq_1_3 = 0;
      deliq_4_10 = 0;
      deliq_10_ = 0;
      deliq_day_all_old = 0;
      deliq_day_max = 0;
      for select max(ok_ol_date) - ok_sh_date
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :ks_from_k_id and
                ok_type in (4, 5) and
                ok_from_c_id = 0
          group by ok_sh_date
          having max(ok_ol_date) - ok_sh_date > 0
          into :deliq_day_old
      do
      begin
        if (:deliq_day_old between 1 and 3) then
          deliq_1_3 = deliq_1_3 + 1;
        if (:deliq_day_old between 4 and 10) then
          deliq_4_10 = deliq_4_10 + 1;
        if (:deliq_day_old > 10) then
          deliq_10_ = deliq_10_ + 1;
        deliq_day_all_old = deliq_day_all_old + :deliq_day_old;
        if (:deliq_day_max < :deliq_day_old) then
          deliq_day_max = :deliq_day_old;
      end
    end
    old_k_id_1 = :ks_from_k_id;

    deliq_day_all = deliq_day_all_old + deliq_day;

    suspend;
  end
end
