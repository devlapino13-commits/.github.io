CREATE PROCEDURE AAREP_AFR_CLIENTS(
    REP_REG_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_64,
    KAL_NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    KC_AMOUNT_APPROVED D_CURRENCY,
    KC_CIRCLE D_INTEGER,
    IS_FREEZE D_STR_10,
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    C_INN D_STR_20,
    C_AGE D_SMALLINT)
AS
declare variable reg_unicode d_integer;
declare variable kf_id d_id;
declare variable k_id d_id;
declare variable eds_status_datetime d_datetime;
declare variable disb_ol_datetime d_datetime;
begin
  -- данные отчета "Клиенты выданных кредитов" (RFCKZ-370)

  for select id_list
      from region_list(:rep_reg_id, 1, 3, 1)
      into reg_unicode
  do
  begin
    k_id = null;

    for select ks_from_k_id
        from kontstate
        where ks_region_id = :reg_unicode and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :rep_date_begin and :rep_date_end
        into k_id
    do
    begin
      k_kontract_no = null;
      k_fact_disb_date = null;
      kf_id = null;
      kal_new_kf_name = null;
      reg_name = null;
      c_id = null;
      kc_circle = null;
      kc_amount_approved = null;
      is_freeze = null;

      select k_from_kf_id,
             k_kontract_no,
             k_fact_disb_date
      from kontract
      where k_id = :k_id
      into kf_id,
           k_kontract_no,
           k_fact_disb_date;

      select kal_new_kf_name
      from knaktlist
      where is_delete = 0 and
            kal_from_kf_id = :kf_id and
            :k_fact_disb_date between kal_date and kal_date_end
      into kal_new_kf_name;

      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_unicode
      into reg_name;

      select max(eds_status_datetime)
      from edoc_list
      inner join edoc_sign on eds_from_edl_id = edl_id
      where edl_from_edc_id = f_edoc_kind('loan.agreement', 'id') and
            edl_object_kind = 9 and
            edl_object_id = :k_id and
            eds_status = 1 and
            edl_status = 1
      into eds_status_datetime;

      select ol_datetime
      from operlist
      inner join kontract on k_fact_disb_ol_id = ol_id
      where k_id = :k_id
      into disb_ol_datetime;

      if (datediff(minute from eds_status_datetime to disb_ol_datetime) > 1440) then
        is_freeze = 'Да'; -- с момента подписания клиентом КД и выдачей займа прошло 24 часа
      else
        is_freeze = 'Нет';

      for select kc_from_c_id,
                 kc_circle,
                 kc_amount_approved
          from knclient
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0
          into c_id,
               kc_circle,
               kc_amount_approved
      do
      begin
        c_name_fio = null;
        c_inn = null;
        c_age = null;

        select c_name_fio,
               c_inn,
               f_age(c_birth_date, :k_fact_disb_date)
        from clients
        where c_id = :c_id
        into c_name_fio,
             c_inn,
             c_age;

        suspend;
      end
    end
  end
end
