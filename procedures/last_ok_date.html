CREATE PROCEDURE LAST_OK_DATE(
    KONT DECIMAL(18,0))
RETURNS (
    LAST_DATE DATE)
AS
declare variable operdel_date d_date;
begin
  operdel_date = null;

  if (f_get_db_id() = 81001) then
  begin
    -- Есть ли подтвержленная заявка на удаление
    select min(odm_oper_date)
    from operdelkont, operdelmain
    where odk_from_k_id = :kont and
          odk_from_odm_id = odm_id and
          odm_state = 1 and
          odm_oper_date >= (current_date - 7)
    into operdel_date;
  end

  if (operdel_date is null) then
    operdel_date = cast('31.12.2999' as date);

  select ok_ol_date
  from operkred
  where ok_ol_is_active = 1 and
        ok_from_k_id = :kont and
        ok_from_c_id = 0 and
        ok_ol_date < :operdel_date -- проверяем до даты заявки на удаление
  order by ok_ol_is_active desc,ok_ol_date desc
  rows 1
  into last_date;

  if (last_date is null) then
    last_date = cast('01.01.0100' as date);

  suspend;
end
