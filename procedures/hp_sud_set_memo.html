CREATE PROCEDURE HP_SUD_SET_MEMO(
    ID D_ID NOT NULL,
    IS_CLIENT D_BOOL NOT NULL,
    IS_EDIT D_BOOL NOT NULL,
    MEMO D_STR_LONG NOT NULL,
    UN_ID D_ID NOT NULL)
AS
declare variable sul_id d_id;
declare variable smm_id d_id;
declare variable smm_from_un_id d_id;
begin
  if(coalesce(memo,'') <> '' or is_edit = 1) then
  begin
    if (is_client = 0) then
    begin
      select smr_last_sul_id
      from sud_mainreq
      where smr_id = :id
      into sul_id;
    end else
    begin
      select slc_last_sul_id
      from sud_loanclient
      where slc_id = :id
      into sul_id;
    end

    if (is_edit = 1) then
    begin
      select smm_id, smm_from_un_id
      from sud_mainmemo
      where smm_from_sul_id = :sul_id
      order by smm_datetime desc, smm_id desc
      rows 1
      into smm_id, smm_from_un_id;

      if (smm_id is null) then
        is_edit = 0;
    end

    if (is_edit = 0) then
    begin
      insert into sud_mainmemo (smm_from_sul_id, smm_memo, smm_from_un_id)
      values (:sul_id, :memo, :un_id);
    end else
    begin
      if (smm_from_un_id <> un_id) then
        exception e_message 'Редактирование доступно если ваш комментарий ясляется последним.';

      if (trim(coalesce(memo, '')) <> '') then
      begin
        update sud_mainmemo
        set smm_memo = :memo,
            smm_from_un_id = :un_id,
            smm_datetime = current_timestamp
        where smm_id = :smm_id;
      end else
      begin
        delete from sud_mainmemo
        where smm_id = :smm_id;
      end
    end
  end
end
