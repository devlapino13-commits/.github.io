CREATE PROCEDURE FM_INSURANCE_PREMIUM_LIST(
    UN_ID D_ID,
    REG_ID D_INTEGER = null)
RETURNS (
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    RES_TEXT_1 D_STR_255,
    RES_TEXT_2 D_STR_255,
    RES_TEXT_3 D_STR_255,
    RES_TEXT_4 D_STR_255)
AS
declare variable lang d_str_lang;
declare variable c_id d_id;
declare variable is_manager d_smallint;
declare variable k_id d_id;
declare variable cli_id d_id;
declare variable c_name_fio d_str_100;
declare variable k_kontract_no d_str_20;
declare variable insurance_amount d_currency;
declare variable payed_amount d_currency;
declare variable pay_due_date d_date;
begin
  -- Список клиентов, которым необходимо оплатить страховую премию.
  -- Только ИПФ-продукты (как в HP_IPF_CHECK_INSURANCE_PAY).
  -- ВИПС видит свои кредитные папки, МО - папки офиса, выбранного в МП (reg_id).
  lang = f_get_lang_code(un_id, null);
  obj_kind = 'clients';

  select un.un_from_c_id
  from username un
  where un.un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exit;

  -- Признак менеджера офиса по действующим кадровым приказам.
  -- МО может быть приписан сразу к нескольким офисам, поэтому сам офис
  -- берем не отсюда, а из входящего reg_id (выбор офиса в МП).
  is_manager = 0;

  if (exists(select *
             from hr_pstatus ps
             inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
             inner join jobkind jk on jk.jk_id = jl.jl_kind
             where ps.ps_from_c_id = :c_id and
                   ps.ps_status = 1 and
                   ps.ps_is_working = 1 and
                   current_date between ps.ps_date_begin and ps.ps_date_end and
                   jk.jk_system_code = 'JobSystem.Region_KM')) then
    is_manager = 1;

  if (is_manager = 1) then
  begin
    -- Офис не выбран - показывать нечего
    if (reg_id is null) then
      exit;

    -- Офис должен быть среди тех, к которым приписан менеджер
    if (not exists(select *
                   from hr_pstatus ps
                   inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
                   inner join jobkind jk on jk.jk_id = jl.jl_kind
                   where ps.ps_from_c_id = :c_id and
                         ps.ps_status = 1 and
                         ps.ps_is_working = 1 and
                         current_date between ps.ps_date_begin and ps.ps_date_end and
                         jk.jk_system_code = 'JobSystem.Region_KM' and
                         ps.ps_from_reg_id = :reg_id)) then
      exit;
  end

  for select k.k_id,
             kc.kc_from_c_id,
             c.c_name_fio,
             k.k_kontract_no,
             ik.ik_commission
  from knaktlist kal
  inner join kontract k on k.k_from_kf_id = kal.kal_from_kf_id
  inner join product pr on pr.pr_id = k.k_from_pr_id and
        pr.pr_code containing '.ipf.' and
        pr.pr_code not containing '.ignore_insurance.'
  inner join kontstate ks on ks.ks_from_k_id = k.k_id and
        ks.ks_state = 1 and
        ks.ks_ol_is_delete = 0 and
        current_date between ks.ks_date_begin and ks.ks_date_end
  inner join insurance_kontract ik on ik.ik_from_k_id = k.k_id and
        (ik.ik_enum_status in (f_enum_id('insurance_kontract', 'insurance_policy_obtained'),
                               f_enum_id('insurance_kontract', 'insurance_policy_app_was_sent')) or
         ik.ik_from_ag_id in (7414, 7570))
  inner join knclient kc on kc.kc_from_k_id = k.k_id and
        kc.kc_amount_approved > 0
  inner join clients c on c.c_id = kc.kc_from_c_id
  where kal.is_delete = 0 and
        current_date between kal.kal_date and kal.kal_date_end and
        ((:is_manager = 1 and kal.kal_new_reg_id = :reg_id) or
         (:is_manager = 0 and kal.kal_new_kf_officer = :c_id))
  order by c.c_name_fio
  into :k_id,
       :cli_id,
       :c_name_fio,
       :k_kontract_no,
       :insurance_amount
  do
  begin
    -- Дата, до которой нужно оплатить премию - первый платеж по графику
    pay_due_date = null;
    select sh.sh_date
    from schedule sh
    where sh.sh_from_k_id = :k_id and
          sh.sh_id = 1
    into :pay_due_date;

    -- Оплаченная сумма премии.
    -- Оплата - это приход денег извне: по кредиту 11531/xxx (закрытие
    -- требования) или 21122/xxx (прямая оплата в кассу), при этом по
    -- дебету НЕ эти счета. Проводка Дт 11531/xxx -> Кт 21122/xxx это
    -- начисление в день выдачи, оплатой не является.
    -- Договор определяем по клиенту и уточняем номером в OL_MEMO.
    payed_amount = 0;
    select coalesce(sum(ob.ob_summa), 0)
    from operbook ob
    inner join operlist ol on ol.ol_id = ob.ob_from_ol_id and
          ol.is_delete = 0
    where (ob.ob_client_dt = :cli_id or
           ob.ob_client_kt = :cli_id) and
          ob.ob_ol_is_active = 1 and
          ob.ob_short_acc_dt not starting '11531/' and
          ob.ob_short_acc_dt not starting '21122/' and
          (ob.ob_short_acc_kt starting '11531/' or
           ob.ob_short_acc_kt starting '21122/') and
          ol.ol_memo containing :k_kontract_no || ' от '
    into :payed_amount;

    payed_amount = coalesce(payed_amount, 0);
    insurance_amount = coalesce(insurance_amount, 0);

    obj_id = cli_id;
    res_text_1 = c_name_fio || '<br><small>' || coalesce(k_kontract_no, '') || '</small>';
    res_text_2 = f_numtostr(insurance_amount, 2);
    res_text_3 = iif(pay_due_date is null, '-', f_datetostr(pay_due_date));
    res_text_4 = iif(payed_amount >= insurance_amount,
                     f_lang(lang, 'Оплачено'),
                     f_lang(lang, 'Не оплачено'));
    suspend;
  end
end
