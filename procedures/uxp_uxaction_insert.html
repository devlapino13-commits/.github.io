CREATE PROCEDURE UXP$UXACTION_INSERT(
    UX_UUID D_UUID,
    UXA_CODE D_STR_100,
    ADD_TO_MENU D_BOOL = 0)
RETURNS (
    UA_UUID_STR D_STR_CODE)
AS
declare variable ua_text d_str_255;
declare variable ua_uuid d_uuid;
declare variable uxa_uuid d_uuid;
declare variable ufm_uuid d_uuid;
declare variable field_name d_str_code;
declare variable ua_is_group d_bool;
declare variable owner_uxa_uuid d_uuid;
declare variable uxa_kind d_smallint;
declare variable uf_is_param d_bool;
begin
  if (coalesce(ux_uuid, '') = '') then
    exception e_message 'Empty ux_uuid';

  ua_is_group = 0;
  if (uxa_code = 'ActionGroup') then
  begin
    uxa_code = '';
    ua_is_group = 1;
  end else
  begin
    select ua_name
    from ux$action
    where ua_owner = :ux_uuid and
          ua_uxa_fcode = :uxa_code and
          ua_usr_id_field = ''
    into ua_text;
  
    if (row_count > 0) then
      exception e_message f_format('Данный объект ранее добавлен, действие: %1', ua_text);
  end

  insert into ux$action (ua_owner, ua_uxa_fcode, ua_is_group)
  values (:ux_uuid, :uxa_code, :ua_is_group)
  returning ua_uuid, ua_uxa_uuid
  into :ua_uuid, uxa_uuid;

  ua_uuid_str = uuid_to_char(ua_uuid);

  if (ua_is_group = 0) then
  begin
    select uxa_uuid, uxa_kind
    from ux$alias
    where uxa_fcode = :uxa_code
    into uxa_uuid, uxa_kind;
  
    ufm_uuid = uxf$ver(uxa_uuid, 'draft');
  
    for select distinct uf_field_name, uf_is_param
    from ux$frame, ux$frparam, ux$field
    where ufr_owner = :ufm_uuid and
          ufr_uuid = ufp_ufr_uuid and
          ufp_uf_uuid = uf_uuid and
          (uf_is_param = 1 or uf_pk_field <> 0)
    into field_name, uf_is_param
    do
    begin
      for select uap_ua_uuid, uap_param_name, uap_owner_ufp_uuid, uap_owner_ext_field, uap_default, uap_use_default,
                 uap_locked
      from ux$actionparam, ux$action, ux$uniq
      where ua_owner = :ux_uuid and
            uap_param_name = :field_name and
            uap_is_output = 0 and
            uap_ua_uuid = ua_uuid and
            ua_uuid = uu_id
      order by uu_datetime desc
      rows 1
      as cursor cur
      do
      begin
        if (not exists(select * from ux$actionparam
                       where uap_ua_uuid = :ua_uuid and
                             uap_param_name = :field_name)) then
        begin
          if (uxa_kind = 2 and uf_is_param = 0) then
          begin
            insert into ux$actionparam (uap_ua_uuid, uap_param_name, uap_owner_ext_field)
            values (:ua_uuid, :field_name, cur.uap_owner_ext_field);
          end else
          begin
            insert into ux$actionparam (uap_ua_uuid, uap_param_name, uap_owner_ufp_uuid, uap_owner_ext_field,
                                        uap_default, uap_use_default, uap_locked)
            values (:ua_uuid, :field_name, cur.uap_owner_ufp_uuid, cur.uap_owner_ext_field,
                    cur.uap_default, cur.uap_use_default,  cur.uap_locked);
          end
        end
      end
    end

    if (coalesce(add_to_menu, 0) = 1) then
    begin
      for select uxa_uuid
      from ux$form, ux$alias
      where ufm_uuid = :ux_uuid and
            ufm_ux_uuid = uxa_owner
      into owner_uxa_uuid
      do
      begin
        update or insert into ux$aliasaction (uaa_uxa_uuid, uaa_ua_uuid)
        values (:owner_uxa_uuid, :ua_uuid)
        matching (uaa_uxa_uuid, uaa_ua_uuid);
      end
    end
  end

  suspend;
end
