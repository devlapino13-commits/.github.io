CREATE PROCEDURE HP_NF_ANALIZ_DETAIL_VIEW(
    IFA_ID D_INTEGER)
RETURNS (
    IND_ID D_INTEGER,
    IND_REC_NO D_INTEGER,
    IND_MAIN_BANK_ID D_INTEGER,
    MAIN_BANK_CODE D_STR_CODE,
    IND_SUB_BANK_ID D_INTEGER,
    SUB_BANK_CODE D_STR_CODE,
    IND_RECIEVED_TRAN D_INTEGER,
    IND_MUST_PAY_TRAN D_CURRENCY,
    IND_NETTO_COUNT_TRANS D_INTEGER,
    IND_NETTO_AMOUNT D_CURRENCY,
    IS_ERROR D_BOOL,
    IND_ERROR_ID D_INTEGER,
    ERROR_NAME D_STR_255)
AS
begin
  for
    select
      ind_id,
      ind_rec_no,
      ind_main_bank_id,
      b1.ib_bank_code,
      ind_sub_bank_id,
      b2.ib_bank_code,
      ind_recieved_tran,
      ind_must_pay_tran,
      ind_netto_count_trans,
      ind_netto_amount,
      iif(ind_error_id > 0, 1, 0),
      ind_error_id,
      rd_name
    from ipc_nonfin_detail
      left join ipc_banks as b1
        on b1.ib_id = ind_main_bank_id
      left join ipc_banks as b2
        on b2.ib_id = ind_sub_bank_id
      left join refdata
        on rd_id = ind_error_id
    where ind_from_ifa_id = :ifa_id
  into
    :ind_id,
    :ind_rec_no,
    :ind_main_bank_id,
    :main_bank_code,
    :ind_sub_bank_id,
    :sub_bank_code,
    :ind_recieved_tran,
    :ind_must_pay_tran,
    :ind_netto_count_trans,
    :ind_netto_amount,
    :is_error,
    :ind_error_id,
    :error_name
  do
  begin
    suspend;
  end
end
