CREATE PROCEDURE AAREP_SIRENA_ACTIVE_KONT(
    REP_REGION_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(50),
    K_DISB_DATE DATE,
    KAL_NEW_KF_NAME VARCHAR(100),
    K_CICLE INTEGER,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_REPAY INTEGER,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_AMOUNT_APPROVED CURR_TYPE,
    K_BALANCE CURR_TYPE,
    K_DELINQ_DAYS INTEGER,
    K_KC_FIO VARCHAR(100),
    K_PR_NAME VARCHAR(50),
    K_OFFICE_NAME VARCHAR(100),
    K_CASHIER_FIO VARCHAR(100),
    K_KREDLINE_END DATE,
    K_KONTRACT_CATEGORY VARCHAR(50),
    K_FIRST_KC VARCHAR(100),
    K_DATE_AKT DATE,
    SBL_NAME VARCHAR(100),
    IS_SB VARCHAR(100))
AS
declare variable k_id id_type;
declare variable pr_id id_type;
declare variable k_reg_id id_type;
declare variable k_kc_id id_type;
declare variable k_clients_cashier_id id_type;
declare variable k_fact_disb_ol_id id_type;
declare variable kf_id id_type;
declare variable sbl_type integer;
declare variable k_is_problem integer;
declare variable def_curr varchar(3);
declare variable sba_date_begin date;
declare variable kal_date date;
begin
-- все активные кредиты КС Сирена
  for select oks_kc_id, oks_from_k_id, oks_region_id, oks_balance,
             iif(cast(:rep_date_end as date) - oks_delinq_date <= 0, 0, cast(:rep_date_end as date) - oks_delinq_date)
      from oksummary oks, region_list(:rep_region_id, 1, -1, 1)
      where oks_kc_id = 1000000000004 and
            :rep_date_end between oks_date_begin and oks_date_end and
            (oks_region_id = id_list)
            and oks_close_state = 0
--      and oks_from_k_id = 101010000243500701
      order by oks_from_k_id
      into k_kc_id, k_id, k_reg_id, k_balance, k_delinq_days
  do
  begin
    select k_kontract_no, k_disb_date, k.k_period, k_kolich_repay, k.k_kolich_client_approved, k.k_amount_approved,
           k.k_from_pr_id, k.k_clients_cashier_id, k.k_kredline_end, k.k_circle, k.k_fact_disb_lo_id, k.k_currency,
           k.k_is_problem, k_from_kf_id
    from kontract k
    where k_id = :k_id
    into k_kontract_no, k_disb_date, k_period, k_kolich_repay, k_kolich_client_approved, k_amount_approved, pr_id,
         k_clients_cashier_id, k_kredline_end, k_cicle, k_fact_disb_ol_id, def_curr, k_is_problem, kf_id;

    select p.pr_name
    from product p
    where p.pr_id = :pr_id
    into k_pr_name;

    select reg_name
    from region
    where reg_unicode = :k_reg_id
    into k_office_name;

    select c_name_fio
    from clients
    where c_id = :k_kc_id
    into k_kc_fio;

    select c_name_fio
    from clients
    where c_id = :k_clients_cashier_id
    into k_cashier_fio;

    select cl3.c_name_fio
    from clients cl3
    where c_id = :k_fact_disb_ol_id
    into k_first_kc;

--  Наименование СБ (скопирован с отчета АКтивные кредиты) >>>>
    sbl_name = null;
    sbl_type = null;
    sba_date_begin = null;
    select sbl_name, sbl_type, sba_date_begin
    from sb_akt, sb_list
    where sba_from_k_id = :k_id and
          sba_status = 2 and
          :rep_date_end between sba_date_begin and sba_date_end and
          sba_is_delete = 0 and
          sba_from_sbl_id <> 0 and
          sba_from_sbl_id = sbl_id
    into sbl_name, sbl_type, k_date_akt;

    is_sb = '';
    if (sbl_type is not null) then
    begin
      is_sb = case sbl_type
                when 0 then 'в учете СБ'
                when 1 then 'в учете коллеторов'
                else ''
              end;
    end

    k_kontract_category = null;
    if ((rep_date_end < '01.12.2012') or (def_curr <> '810')) then
    begin
      k_kontract_category = 'Категория A';
      if (k_is_problem = 1) then
      begin
        if (sbl_type = 1) then
          k_kontract_category = 'Категория C';
      end
      if ((k_kontract_category = 'Категория A') and
          (def_curr = '810') and
          (:pr_id >= 18) and
          (:pr_id <= 30)) then
        k_kontract_category = 'Категория B';
    end
    else
    begin
      select kont_category
      from aarep_get_category_id(:k_id, :rep_date_end, null, null, null, null, null, null)
      into k_kontract_category;
    end
--<<<<<<Конец копии
    select kal_date, kal.kal_new_kf_name
    from knaktlist kal
    where kal.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal.kal_from_kf_id = :kf_id
    into kal_date, kal_new_kf_name;

    k_date_akt = iif(:kal_date is null, :sba_date_begin, :kal_date);

    suspend;
  end
end
