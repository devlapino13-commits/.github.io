CREATE PROCEDURE AAREP_APADVANCED_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_DATE_KIND INTEGER,
    FIND_KF_NAME VARCHAR(100),
    FIND_K_NO VARCHAR(20),
    FIND_C_FIO VARCHAR(100),
    REP_IS_DELETED SMALLINT)
RETURNS (
    AA_FROM_K_ID ID_TYPE,
    AA_ID INTEGER,
    AA_DATE_APP DATE,
    AA_DATE_REPAY DATE,
    KAL_NEW_KF_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    REG_NAME VARCHAR(50),
    FIO VARCHAR(100),
    STATUS VARCHAR(20),
    REASON VARCHAR(50),
    UN_NAME VARCHAR(30),
    DATECHANGED TIMESTAMP,
    AA_IS_DELETED SMALLINT,
    COMMENTS VARCHAR(512))
AS
declare variable can_suspend integer;
declare variable group_name varchar(20);
declare variable aa_from_reg_id integer;
declare variable aa_from_un_id id_type;
declare variable aa_reason_id smallint;
declare variable aa_from_c_id id_type;
declare variable maxdate date;
declare variable aa_is_one_client integer;
begin
  group_name = 'Вся группа';
  maxdate = cast('31.12.2999' as date);
  if ((find_kf_name <> '') or (find_c_fio <> '') or (find_k_no <> '')) then
  begin
    /* Поиск по критериям */
    for select distinct k_id, kal_new_kf_name, k_kontract_no
        from findkontract_new_tmp(:rep_region_id, :find_kf_name, :find_k_no, :find_c_fio, 2, 0) k
        where exists(select 1
                     from apadvanced a
                     where k.k_id = a.aa_from_k_id)
        into :aa_from_k_id, :kal_new_kf_name, :k_kontract_no
    do
    begin
      for select aa_id, aa_date_app, aa_date_repay, aa_datetime, aa_is_deleted, comments, aa_from_reg_id, aa_from_un_id,
                 aa_is_deleted, aa_reason_id, aa_from_c_id, aa_is_one_client
          from apadvanced
          where aa_from_k_id = :aa_from_k_id and
                (:rep_is_deleted = 0 or
                (:rep_is_deleted = 1 and
                aa_is_deleted = 0) or
                (:rep_is_deleted = 2 and
                aa_is_deleted = 1))
          into :aa_id, :aa_date_app, :aa_date_repay, :datechanged, :aa_is_deleted, :comments, :aa_from_reg_id,
               :aa_from_un_id, :aa_is_deleted, :aa_reason_id, :aa_from_c_id, :aa_is_one_client
      do
      begin
        if (aa_is_one_client is null) then
          aa_is_one_client = 0;
        can_suspend = 0;
        if (rep_date_kind = 0) then
          can_suspend = 1;
        else
        if ((rep_date_kind = 1) and
            (aa_date_app between :rep_date_begin and :rep_date_end)) then
          can_suspend = 1;
        else
        if ((rep_date_kind = 2) and
            (aa_date_repay between :rep_date_begin and :rep_date_end)) then
          can_suspend = 1;
        if (can_suspend = 1) then
        begin
          select reg_name
          from region
          where reg_unicode = :aa_from_reg_id
          into :reg_name;

          select un_name
          from username
          where un_id = :aa_from_un_id
          into :un_name;

          if (aa_is_deleted = 0) then
            status = 'Активный';
          else
            status = 'Аннулированный';

          select rd_name
          from refdata
          where rd_id = :aa_reason_id
          into :reason;

          if (:aa_is_one_client = 0) then
            fio = group_name;
          else
            select c_name_fio
            from clients
            where c_id = :aa_from_c_id
            into :fio;

          suspend;
        end
      end
    end
  end
  else
  if (:rep_date_kind > 0) then
  begin
    for select aa_id, aa_date_app, aa_date_repay, aa_datetime, aa_is_deleted, comments, aa_from_reg_id, aa_from_un_id,
               aa_is_deleted, aa_reason_id, aa_from_c_id, aa_from_k_id, aa_is_one_client
        from apadvanced
        inner join region_list(:rep_region_id, 1, -1, 1) on aa_from_reg_id = id_list
        where (:rep_is_deleted = 0 or
              (:rep_is_deleted = 1 and
              aa_is_deleted = 0) or
              (:rep_is_deleted = 2 and
              aa_is_deleted = 1)) and
              ((:rep_date_kind = 1 and
              aa_date_app between :rep_date_begin and :rep_date_end) or
              (:rep_date_kind = 2 and
              aa_date_repay between :rep_date_begin and :rep_date_end))
        into :aa_id, :aa_date_app, :aa_date_repay, :datechanged, :aa_is_deleted, :comments, :aa_from_reg_id,
             :aa_from_un_id, :aa_is_deleted, :aa_reason_id, :aa_from_c_id, :aa_from_k_id, aa_is_one_client
    do
    begin
      if (aa_is_one_client is null) then
        aa_is_one_client = 0;
      select kl.kal_new_kf_name, k.k_kontract_no
      from kontract k
      inner join knaktlist kl on k.k_from_kf_id = kl.kal_from_kf_id
      where k.k_id = :aa_from_k_id and
            kal_date_end >= :maxdate
      into :kal_new_kf_name, :k_kontract_no;

      select reg_name
      from region
      where reg_unicode = :aa_from_reg_id
      into :reg_name;

      select un_name
      from username
      where un_id = :aa_from_un_id
      into :un_name;

      if (aa_is_deleted = 0) then
        status = 'Активный';
      else
        status = 'Аннулированный';

      select rd_name
      from refdata
      where rd_id = :aa_reason_id
      into :reason;

      if (:aa_is_one_client = 0) then
        fio = group_name;
      else
        select c_name_fio
        from clients
        where c_id = :aa_from_c_id
        into :fio;

      suspend;
    end
  end
end
