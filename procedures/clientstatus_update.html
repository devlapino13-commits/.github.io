CREATE PROCEDURE CLIENTSTATUS_UPDATE(
    C_ID ID_TYPE)
AS
declare variable cs_id id_type;
declare variable cs_date_begin date;
declare variable chet integer;
declare variable d_prior date;
begin
  chet = 0;
  for select cs_id, cs_date_begin
      from clientstatus
      where cs_from_c_id = :c_id and
            cs_ol_is_delete = 0
      order by cs_date_begin desc, cs_id desc
      into cs_id, cs_date_begin
  do
  begin
    chet = chet + 1;
    if (chet > 1) then
    begin
      update clientstatus
      set cs_date_end = :d_prior - 1
      where cs_id = :cs_id and
            cs_date_end <> :d_prior - 1;
    end
    else
    begin
      update clientstatus
      set cs_date_end = '31.12.9999'
      where cs_id = :cs_id and
            cs_date_end <> '31.12.9999';
    end
    d_prior = cs_date_begin;
  end
end
