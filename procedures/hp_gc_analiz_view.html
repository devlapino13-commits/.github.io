CREATE PROCEDURE HP_GC_ANALIZ_VIEW(
    DTBEG D_DATE = null,
    DTEND D_DATE = null)
RETURNS (
    IGA_ID D_INTEGER,
    IGA_GC_QR_SUMM D_CURRENCY,
    IGA_DB_QR_SUMM D_CURRENCY,
    IGA_IS_QR_PROBLEM D_BOOL,
    IFA_ID D_INTEGER,
    IFA_FILE_NAME D_STR_100,
    IFA_DATE D_DATE,
    IFA_STATUS D_SMALLINT,
    STATUS_STR D_STR_100,
    IFA_FROM_BLS_ID D_ID,
    IFA_IS_DETAIL_PROBLEM D_BOOL,
    IFA_FROM_UN_ID D_ID,
    UN_NAME D_STR_100,
    IFA_DATETIME D_DATETIME,
    IFA_FROM_OL_ID D_ID)
AS
begin
  dtbeg = coalesce(dtbeg, '30.12.1899');
  dtend = coalesce(dtend, current_date);
  for
    select
      iga_id,
      iga_gc_qr_summ,
      iga_db_qr_summ,
      iga_is_qr_problem,
      ifa_id,
      ifa_file_name,
      ifa_date,
      ifa_status,
      case ifa_status
        when 0  then 'документ создан'
        when -1 then 'бух.проводки по операциям не нужны'
        when 1  then 'построены бух.проводки по операциям'
        when 2  then 'заполнена таблица для перевода денег контрагентам'
        when 3  then 'cформированы задачи на отправку денег контрагентам'
        when 4  then 'не всем выбранным контрагентам деньги ушли из-за ошибки'
        when 5  then 'всем выбранным контрагентам деньги ушли'
      end,
      ifa_from_bls_id,
      ifa_is_detail_problem,
      ifa_from_un_id,
      un_fio,
      ifa_datetime,
      ifa_from_ol_id
    from ipc_file_analiz
      inner join ipc_gc_analiz
        on iga_from_ifa_id = ifa_id
      inner join username
        on un_id = ifa_from_un_id
    where ifa_date between :dtbeg and :dtend
      and ifa_file_type = 1
  into
    :iga_id,
    :iga_gc_qr_summ,
    :iga_db_qr_summ,
    :iga_is_qr_problem,
    :ifa_id,
    :ifa_file_name,
    :ifa_date,
    :ifa_status,
    :status_str,
    :ifa_from_bls_id,
    :ifa_is_detail_problem,
    :ifa_from_un_id,
    :un_name,
    :ifa_datetime,
    :ifa_from_ol_id
  do
  begin
    suspend;
  end
end
