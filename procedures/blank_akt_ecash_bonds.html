CREATE PROCEDURE BLANK_AKT_ECASH_BONDS(
    ID D_INTEGER)
RETURNS (
    BND_ID D_INTEGER,
    BND_BOND D_CURRENCY,
    BND_NAME D_STR_100,
    BND_INV_COUNT D_INTEGER,
    BND_INV_SUMMA D_CURRENCY,
    BND_FACT_COUNT D_INTEGER,
    BND_FACT_SUMMA D_CURRENCY,
    BND_DIFF_COUNT D_INTEGER,
    BND_DIFF_SUMMA D_CURRENCY,
    BND_PLUS_COUNT D_INTEGER,
    BND_PLUS_SUMMA D_CURRENCY,
    BND_MINUS_COUNT D_INTEGER,
    BND_MINUS_SUMMA D_CURRENCY,
    BND_HAS_BONDS D_BOOL)
AS
declare variable cm_currency_name d_str_code;
declare variable tmp_id d_integer;
declare variable main_akt_id d_integer;
declare variable fact_akt_id d_integer;
declare variable main_absent d_bool;
declare variable main_amount d_currency;
declare variable fact_amount d_currency;
declare variable fact_absent d_bool;
begin
  select cm_currency_name
  from company
  into cm_currency_name;

  if (char_length(cm_currency_name) > 3) then
    cm_currency_name = left(cm_currency_name, 3) || '.';

  select opc_from_cil_id
  from opercash
  where opc_id = :id
  into tmp_id;

  if (row_count = 0) then
  begin
    select cil_id
    from cashinvlist
    where cil_id = :id
    into tmp_id;
  end

  if (tmp_id is null) then
    exit;

  main_amount = 0;
  fact_amount = 0;

  select cil_parent_id, cil_inv_amount
  from cashinvlist
  where cil_id = :tmp_id
  into main_akt_id, main_amount;

  if (main_akt_id is not null) then
  begin
    fact_akt_id = tmp_id;
    fact_amount = main_amount;
    main_amount = 0;

    select cil_inv_amount
    from cashinvlist
    where cil_id = :main_akt_id
    into main_amount;
  end
  else
  begin
    main_akt_id = tmp_id;

    select cil_id, cil_inv_amount
    from cashinvlist
    where cil_parent_id = :main_akt_id
    order by cil_id desc
    rows 1
    into fact_akt_id, fact_amount;
  end

  main_absent = 1;
  if (exists (select * from cashinvbonds
              where cib_from_cil_id = :main_akt_id)) then
    main_absent = 0;

  fact_absent = 1;
  if (fact_akt_id is not null) then
  begin
    if (exists (select * from cashinvbonds
                where cib_from_cil_id = :fact_akt_id)) then
      fact_absent = 0;
  end

  bnd_id = 0;
  for select bnl_id
  from bondlist
  where bnl_id >= 1
  order by bnl_id desc
  into :bnd_bond
  do
  begin
    bnd_id = bnd_id + 1;
    bnd_name = f_numtostr(bnd_bond, 0) || ' ' || cm_currency_name;

    bnd_inv_count = 0;
    bnd_inv_summa = 0;
    bnd_fact_count = 0;
    bnd_fact_summa = 0;

    if (main_absent = 0) then
    begin
      select cib_count
      from cashinvbonds
      where cib_from_cil_id = :main_akt_id and
            cib_bond_id = :bnd_bond
      into :bnd_inv_count;
    end else
    if (bnd_bond = 1) then
      bnd_inv_count = main_amount;

    if (fact_akt_id is not null and fact_absent = 0) then
    begin
      select cib_count
      from cashinvbonds
      where cib_from_cil_id = :fact_akt_id and
            cib_bond_id = :bnd_bond
      into :bnd_fact_count;
    end

    if (fact_absent = 1 and bnd_bond = 1) then
      bnd_fact_count = fact_amount;

    if (fact_absent = 1 and main_absent = 0) then
      bnd_fact_count = bnd_inv_count;

    bnd_inv_summa = bnd_bond * bnd_inv_count;
    bnd_fact_summa = bnd_bond * bnd_fact_count;

    -- Разница Факт - Опись
    bnd_diff_count = bnd_fact_count - bnd_inv_count;
    bnd_diff_summa = bnd_fact_summa - bnd_inv_summa;

    bnd_plus_count = 0;
    bnd_plus_summa = 0;
    bnd_minus_count = 0;
    bnd_minus_summa = 0;

    if (bnd_diff_count > 0) then
    begin
      -- Излишек
      bnd_plus_count = bnd_diff_count;
      bnd_plus_summa = bnd_diff_summa;
    end else
    if (bnd_diff_count < 0) then
    begin
      -- Недостача
      bnd_minus_count = -bnd_diff_count;
      bnd_minus_summa = -bnd_diff_summa;
    end

    bnd_has_bonds = 0;
    if (bnd_inv_count <> 0 or bnd_fact_count <> 0) then
      bnd_has_bonds = 1;

    suspend;
  end
end
