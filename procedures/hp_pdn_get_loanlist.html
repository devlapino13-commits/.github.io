CREATE PROCEDURE HP_PDN_GET_LOANLIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_K_ID D_ID,
    REPEATED D_INTEGER)
RETURNS (
    OPER_DATETIME D_DATETIME,
    NEW_KC_ID D_ID,
    CML_ID D_ID)
AS
begin
  if (repeated is null) then
    repeated = 0;

  if (repeated in (0,3)) then -- выдача
  begin
    if (rep_k_id is null) then
    begin
      for select ol_datetime, kc_id
      from kontstate, kontract, operlist, knclient
      where k_id = ks_from_k_id
        and ol_id = k_fact_disb_ol_id
        and kc_from_k_id = ks_from_k_id
        and ks_state = 1
        and ks_ol_is_delete = 0
        and ks_date_begin between :rep_date_begin and :rep_date_end
        and kc_amount_approved > 0
      order by ks_date_begin, ol_datetime
      into oper_datetime, new_kc_id
      do
        suspend;
    end
    else
    begin
      for select kc_id
      from knclient
      where kc_from_k_id = :rep_k_id
        and kc_amount_approved > 0
      into new_kc_id
      do
      begin
        select ol_datetime
        from kontract, operlist
        where ol_id = k_fact_disb_ol_id
          and k_id = :rep_k_id
        into oper_datetime;
        suspend;
      end
    end
  end
  else if (repeated = 1) then -- повторные
  begin
    if (rep_k_id is null) then
    begin
      for select cml_date, kc_id, cml_id
        from com_list, com_cfg, com_action, com_operlist, knclient
        where cml_date between :rep_date_begin and :rep_date_end
          and cml_from_cmc_id = cmc_id
          and cmc_code in ('rest_standart', 'rastor_dog_sogl_storon')
          and cma_from_cml_id = cml_id
          and cmo_from_cma_id = cma_id
          and cmo_ol_is_delete = 0
          and kc_from_k_id = cml_object_id
      union
        select kc_date, kc_id, null
        from knclient_pdn
        where kcp_kind = 2
          and kc_date between :rep_date_begin and :rep_date_end
      order by 1
      into oper_datetime, new_kc_id, cml_id
      do
        suspend;
    end
    else
    begin
      for select cml_date, kc_id, cml_id
        from com_list, com_cfg, com_action, com_operlist, knclient
        where cml_object_kind = 9
          and cml_object_id = :rep_k_id
          and cml_from_cmc_id = cmc_id
          and cmc_code in ('rest_standart', 'rastor_dog_sogl_storon')
          and cma_from_cml_id = cml_id
          and cmo_from_cma_id = cma_id
          and cmo_ol_is_delete = 0
          and kc_from_k_id = cml_object_id
      union
        select kc_date, kc.kc_id, null
        from knclient_pdn kcp, knclient kc
        where kc.kc_id = kcp.kc_id
          and kc_from_k_id = :rep_k_id
          and kcp_kind = 2
      order by 1
      into oper_datetime, new_kc_id, cml_id
      do
        suspend;
    end
  end
  else if (repeated = 2) then  -- рефинанс
  begin
    if (rep_k_id is null) then
    begin
      for select distinct ol_datetime, kc_id
      from kontstate, kontract, operlist, knclient, kontrefinance
      where k_id = ks_from_k_id
        and ol_id = k_fact_disb_ol_id
        and kc_from_k_id = ks_from_k_id
        and kr_from_ma_id = k_from_ma_app_id
        and ks_state = 1
        and ks_ol_is_delete = 0
        and ks_date_begin + 0 between :rep_date_begin and :rep_date_end
        and kc_amount_approved > 0
      order by ks_date_begin, ol_datetime
      into oper_datetime, new_kc_id
      do
        suspend;
    end
    else
    begin
      for select distinct ol_datetime, kc_id
      from kontstate, kontract, operlist, knclient, kontrefinance
      where k_id = ks_from_k_id
        and ol_id = k_fact_disb_ol_id
        and kc_from_k_id = ks_from_k_id
        and kr_from_ma_id = k_from_ma_app_id
        and ks_state = 1
        and ks_ol_is_delete = 0
        and kc_amount_approved > 0
        and k_id = :rep_k_id
      order by ks_date_begin, ol_datetime
      into oper_datetime, new_kc_id
      do
        suspend;
    end
  end
end
