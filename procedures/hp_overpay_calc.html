CREATE PROCEDURE HP_OVERPAY_CALC(
    OPM_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_LONG)
AS
declare variable overpay_sum_oper d_currency;
declare variable repay_sum d_currency;
declare variable debt_sum d_currency;
declare variable debt_sum_oper d_currency;
declare variable active_k_id d_id;
declare variable overpay_k_id d_id;
declare variable opl_id d_id;
declare variable jso d_integer;
declare variable repay_date d_date;
declare variable min_days d_integer;
begin
  min_days = 7;  -- менять так же в HP_OVERPAY_PREP
  error_text = '';
  jso = json.obj_create();

  for select opl_from_k_id, opl_res_sum
  from overpay_loans
  where opl_from_opm_id = :opm_id
    and opl_type = 2
    and opl_res_sum > 0
  order by opl_no nulls last
  into active_k_id, debt_sum
  do
  begin
    select coalesce(sum(zad_os), 0) + coalesce(sum(sh_calc_perc), 0) + coalesce(sum(sh_calc_fine), 0)
    from aarep_make_zad_list(:active_k_id, current_date, 1)
    into debt_sum_oper;

    if (debt_sum_oper < debt_sum) then
    begin
      error_text = f_format('Изменилась сумма задолженности по дог. %1', (select k_kontract_no from kontract where k_id = :active_k_id));
      break;
    end

    while (debt_sum > 0)
    do
    begin
      overpay_k_id = null;

      select opl_id, opl_from_k_id, opl_date, opl_sum - cast(json.obj_get_val(:jso, opl_id, 0) as d_currency)
      from overpay_loans
      where opl_from_opm_id = :opm_id
        and opl_type = 1
        and cast(json.obj_get_val(:jso, opl_id, 0) as d_currency) < opl_sum
      order by opl_no 
      rows 1
      into opl_id, overpay_k_id, repay_date, repay_sum;
    
      if (overpay_k_id is null) then
      begin
        error_text = 'overpay_k_id is null';
        break;
      end 

      select coalesce(-sum(ok_os + ok_perc), 0)
      from operkred
      where ok_from_k_id = :overpay_k_id
        and ok_ol_is_active = 1
        and ok_type = 9
        and (ok_os + ok_perc > 0 or ok_ol_date <= current_date - :min_days)
      into overpay_sum_oper;
  
      if (repay_sum > overpay_sum_oper) then
      begin
        error_text = 'Изменилась сумма переплаты';
        break;
      end

      repay_sum = minvalue(repay_sum, debt_sum);

      begin
        select error_text
        from hp_overpay_transfer_pay(:overpay_k_id, :repay_date, :active_k_id, :repay_sum, 1000000000000)
        into error_text;

        when any do
          error_text = f_format('select error_text from hp_overpay_transfer_pay(%1, ''%2'', %3, %4, 1000000000000)', overpay_k_id, repay_date, active_k_id, repay_sum);
      end

      if (error_text <> '') then
        break;

      debt_sum = debt_sum - repay_sum;
      json.obj_set_s(jso, opl_id, cast(json.obj_get_val(jso, opl_id, 0) as d_currency) + repay_sum);
    end 

    if (error_text <> '') then
      break;
  end

  if (error_text = '') then
    update overpay_main
    set opm_status = 3
    where opm_id = :opm_id;

  suspend;
end
