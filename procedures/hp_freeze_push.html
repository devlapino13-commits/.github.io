CREATE PROCEDURE HP_FREEZE_PUSH
AS
declare variable K_ID D_ID;
declare variable DISB_DATE D_DATETIME;
declare variable FREEZE_TIME_H D_INTEGER;
declare variable K_KONTRACT_NO STR_20;
declare variable C_NAME_FIO STR_100;
declare variable Z_NAME_FIO STR_100;
declare variable EDOC_KIND STR_100;
declare variable UN_ID D_ID;
declare variable RES_TEXT STR_255;
declare variable ANNUL_TIME STR_10;
declare variable KS_KC_ID D_ID;
begin
  for select ks.ks_from_k_id, k.k_disb_date, k.k_kontract_no, ks.ks_kc_id
  from kontstate ks
  join kontract k on k.k_id = ks.ks_from_k_id
  where current_date between ks.ks_date_begin and ks.ks_date_end
    and ks.ks_ol_is_delete = 0
    and ks.ks_state = 0
    and k.k_status = 1
  into k_id, disb_date, k_kontract_no, ks_kc_id
  do
  begin
    select freeze_time_h
    from hp_get_freeze_data(:k_id)
    where freeze_datetime_end is null
    into freeze_time_h;

    if (freeze_time_h is null or freeze_time_h = 0) then
      continue;

    if ((freeze_time_h = 4 and extract(hour from cast(current_time as time)) <> 16) or
        (freeze_time_h = 48 and extract(hour from cast(current_time as time)) <> 20)) then
      continue;

    annul_time = case freeze_time_h
                   when 4 then '16:30'
                   when 48 then '20:30'
                 end;

    select u.un_id
    from username u
    where u.un_from_c_id = :ks_kc_id
    into un_id;

    for select edc.edc_code
    from edoc_list edl
    join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
    where edl.edl_object_id = :k_id
      and edl.edl_object_kind = 9
      and edl.edl_status = 0
      and edc.edc_code in ('loan.agreement', 'loan.collateral')
    into edoc_kind
    do
    begin
      if (edoc_kind = 'loan.agreement') then
      begin
        select c.c_name_fio
        from clients c
        join knclient kc on kc.kc_from_c_id = c.c_id
        where kc.kc_from_k_id = :k_id
        into c_name_fio;

        res_text = 'Договор займа № ' || k_kontract_no || ' от ' ||
                   cast(disb_date as date) || ' ' || coalesce(c_name_fio,'') ||
                   ' не подписан. В ' || annul_time || ' мск он будет аннулирован';
      end
      else
      begin
        select z_name_fio
        from zalog
        where z_from_k_id = :k_id
          and z_type = 1
        into z_name_fio;
  
        res_text = 'Договор залога № ' || k_kontract_no || ' от ' ||
                   cast(disb_date as date) || ' ' || coalesce(z_name_fio,'') ||
                   ' не подписан. В ' || annul_time || ' мск он будет аннулирован';
      end

      if (un_id is not null) then
        f_send_push('username',un_id,'Предупреждение о неподписанном договоре',res_text,'HP_FREEZE_PUSH',7,1000000000000);
    end
  end
end
