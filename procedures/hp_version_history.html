Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_VERSION_HISTORY(
    SHOW_ONLY_CHECK D_BOOL,
    ITEM_NAME D_STR_CODE = null)
RETURNS (
    OBJECT_NAME D_STR_CODE,
    RESPONSE_USER D_STR_CODE,
    KG_CHANGE_DATE D_DATETIME,
    KG_CHANGE_USER D_STR_CODE,
    KG_STATE D_STR_CODE,
    RU_CHANGE_DATE D_DATETIME,
    RU_CHANGE_USER D_STR_CODE,
    RU_STATE D_STR_CODE,
    AZ_CHANGE_DATE D_DATETIME,
    AZ_CHANGE_USER D_STR_CODE,
    AZ_STATE D_STR_CODE,
    KZ_CHANGE_DATE D_DATETIME,
    KZ_CHANGE_USER D_STR_CODE,
    KZ_STATE D_STR_CODE,
    OBJECT_STATE D_STR_CODE,
    KG_CHANGE_BODY D_STR_CODE,
    RU_CHANGE_BODY D_STR_CODE,
    AZ_CHANGE_BODY D_STR_CODE,
    KZ_CHANGE_BODY D_STR_CODE)
AS
begin
  -- перед запуском этой процедуры, необходимо освежить данные через HP_VERSIONLOG_CHECK_DIFF

  item_name = upper(trim(coalesce(item_name, '')));

  if (show_only_check is null) then
    show_only_check = 1;

  for select obj_name,
             max(iif(obj_db_id = 41701, change_date, null)) as kg_change_date,
             max(iif(obj_db_id = 41701, change_user, '')) as kg_change_user,
             max(iif(obj_db_id = 41701, obj_body, '')) as kg_change_body,
             max(iif(obj_db_id = 81001, change_date, null)) as ru_change_date,
             max(iif(obj_db_id = 81001, change_user, '')) as ru_change_user,
             max(iif(obj_db_id = 81001, obj_body, '')) as ru_change_body,
             max(iif(obj_db_id = 94401, change_date, null)) as az_change_date,
             max(iif(obj_db_id = 94401, change_user, '')) as az_change_user,
             max(iif(obj_db_id = 94401, obj_body, '')) as az_change_body,
             max(iif(obj_db_id = 39801, change_date, null)) as kz_change_date,
             max(iif(obj_db_id = 39801, change_user, '')) as kz_change_user,
             max(iif(obj_db_id = 39801, obj_body, '')) as kz_change_body
  from hp_version_history_all(:item_name)
  group by obj_name
  into object_name,
       kg_change_date, kg_change_user, kg_change_body,
       ru_change_date, ru_change_user, ru_change_body,
       az_change_date, az_change_user, az_change_body,
       kz_change_date, kz_change_user, kz_change_body
  do
  begin
    kg_state = '';
    ru_state = '';
    az_state = '';
    kz_state = '';

    response_user = kg_change_user;
    if (ru_change_date > kg_change_date and ru_change_date > az_change_date and ru_change_user <> 'sysdba') then
      response_user = ru_change_user;
    else
    if (az_change_date > kg_change_date and az_change_date > ru_change_date and az_change_user <> 'sysdba') then
      response_user = az_change_user;
    else
    if (ru_change_date > kg_change_date and ru_change_user <> 'sysdba') then
      response_user = ru_change_user;
    else
    if (az_change_date > kg_change_date and az_change_user <> 'sysdba') then
      response_user = az_change_user;

    if (coalesce(kg_change_body, '') = '') then
      kg_change_body = 'deleted';

    if (coalesce(ru_change_body, '') = '') then
      ru_change_body = 'deleted';

    if (coalesce(az_change_body, '') = '') then
      az_change_body = 'deleted';

    if (coalesce(kz_change_body, '') = '') then
      kz_change_body = 'deleted';

    object_state = '';
    if (kg_change_body <> ru_change_body) then
    begin
      if (kg_change_date < ru_change_date) then
      begin
        ru_state = 'ru new';
        object_state = 'check';
      end
      else
        ru_state = 'ru diff';

      if (kg_change_body = 'deleted') then
        kg_state = kg_change_body;
      else
        kg_state = 'base';
    end else
      ru_state = 'ru ok';

    if (kg_change_body <> az_change_body) then
    begin
      if (kg_change_date < az_change_date) then
      begin
        az_state = 'az new';
        object_state = 'check';
      end
      else
        az_state = 'az diff';

      if (kg_state = '') then
      begin
        if (kg_change_body = 'deleted') then
          kg_state = kg_change_body;
        else
          kg_state = 'base';
      end
    end else
      az_state = 'az ok';

    if (kg_change_body <> kz_change_body) then
    begin
      if (kg_change_date < kz_change_date) then
      begin
        kz_state = 'kz new';
        object_state = 'check';
      end
      else
        kz_state = 'kz diff';

      if (kg_state = '') then
      begin
        if (kg_change_body = 'deleted') then
          kg_state = kg_change_body;
        else
          kg_state = 'base';
      end
    end else
      kz_state = 'kz ok';

    if (not(kg_change_body = 'deleted' and ru_change_body = 'deleted' and az_change_body = 'deleted' and kz_change_body = 'deleted')) then
      if (kg_state <> '' and (show_only_check = 0 or object_state = 'check')) then
        suspend;
  end
end
