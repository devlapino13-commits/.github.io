CREATE PROCEDURE HP_KIBHISTORY_INSERT(
    KBM_ID DECIMAL(18,0),
    KH_AMOUNT DECIMAL(18,2),
    KH_LIMIT DECIMAL(18,2),
    KH_DATEAWARD DATE,
    KH_DATESTOP DATE,
    KH_DATEPROLONGATION DATE,
    KH_AMONTHOFPROLONGATION DECIMAL(18,2),
    KH_UNPAIDAMOUNT DECIMAL(18,2),
    KH_UNPAIDPERCENAMOUNT DECIMAL(18,2),
    KH_COUNTOFDELINQUENCY INTEGER,
    KH_CREDITSTATUS VARCHAR(255),
    KH_CREDITTYPE VARCHAR(255),
    KH_CURRENCY VARCHAR(50),
    KH_REPAYSTATUS VARCHAR(50),
    KH_IS_OUR SMALLINT,
    KH_CREATE_DATE TIMESTAMP,
    KH_PERIOD_IN_MONTH DECIMAL(15,2),
    KH_CLIENTS_KOL INTEGER,
    KH_CLIENTS_FIO VARCHAR(1024),
    KH_PORUCH_FIO VARCHAR(1024),
    KH_DELIQDAYSCOUNT INTEGER,
    KH_OWNAMOUNT DECIMAL(18,2),
    KH_IS_DEBTOR SMALLINT,
    KH_DELREPAYMENT VARCHAR(255),
    KH_ACTIVE_DELINQ_DAYS_COUNT INTEGER = 0,
    KH_MAX_DELINQ_DAYS_COUNT INTEGER = 0,
    KH_OWN_UNPAIDAMOUNT DECIMAL(18,2) = 0,
    KH_COMPANY_NAME D_STR_255 = '',
    KH_LOAN_PERCENT DECIMAL(18,2) = 0)
AS
begin
  insert into kibhistory (kh_from_kbm_id, kh_amount, kh_limit, kh_dateaward, kh_datestop, kh_dateprolongation,
                          kh_amonthofprolongation, kh_unpaidamount, kh_unpaidpercenamount, kh_countofdelinquency,
                          kh_creditstatus, kh_credittype, kh_currency, kh_repaystatus, kh_is_our, kh_create_date,
                          kh_period_in_month, kh_clients_kol, kh_clients_fio, kh_poruch_fio, kh_deliqdayscount,
                          kh_ownamount, kh_is_debtor, kh_active_delinq_days_count, kh_max_delinq_days_count,
                          kh_own_unpaidamount, kh_company_name, kh_loan_percent)
  values (:kbm_id, :kh_amount, :kh_limit, :kh_dateaward, :kh_datestop, :kh_dateprolongation, :kh_amonthofprolongation,
          :kh_unpaidamount, :kh_unpaidpercenamount, :kh_countofdelinquency, :kh_creditstatus, :kh_credittype,
          :kh_currency, :kh_repaystatus, :kh_is_our, :kh_create_date, :kh_period_in_month, :kh_clients_kol,
          :kh_clients_fio, :kh_poruch_fio, :kh_deliqdayscount, :kh_ownamount, :kh_is_debtor,
          :kh_active_delinq_days_count, :kh_max_delinq_days_count, :kh_own_unpaidamount, :kh_company_name, :kh_loan_percent);
end
