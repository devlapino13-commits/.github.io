CREATE PROCEDURE HP_CLIENTSTATE_FIX_ONE(
    C_ID D_ID,
    NEED_FIX D_BOOL)
RETURNS (
    IS_CHANGED D_BOOL)
AS
declare variable res_date d_date;
declare variable res_client_state d_integer;
declare variable res_region_id d_integer;
declare variable res_lo_id d_id;
declare variable res_kf_name d_str_100;
declare variable res_circle d_integer;
declare variable res_staff_state d_integer;
begin
-- Процедура обновляет статус клиента
  select res_date, res_client_state, res_region_id, res_lo_id, res_kf_name, res_circle, res_staff_state
  from hp_clientstate_get(:c_id)
  into :res_date, :res_client_state, :res_region_id, :res_lo_id, :res_kf_name, res_circle, res_staff_state;

  is_changed = 0;
  -- Обновляет статус контрагента если что-то изменилось
  if (need_fix = 1) then
  begin
    update clientstate
    set cs_client_state = :res_client_state,
        cs_date = maxvalue(cs_date, :res_date),
        cs_region_id = coalesce(:res_region_id, cs_region_id),
        cs_lo_id = :res_lo_id,
        cs_kf_name = coalesce(:res_kf_name, ''),
        cs_client_circle = :res_circle,
        cs_staff_state = :res_staff_state
    where cs_from_c_id = :c_id and
          (cs_client_state is distinct from :res_client_state or
           cs_date is distinct from maxvalue(cs_date, :res_date) or
           cs_region_id is distinct from :res_region_id or
           cs_lo_id is distinct from :res_lo_id or
           coalesce(cs_kf_name, '') is distinct from coalesce(:res_kf_name, '') or
           cs_client_circle is distinct from :res_circle or
           cs_staff_state is distinct from :res_staff_state);
  
    if (row_count = 1) then
      is_changed = 1;
  end else
  begin
    if (exists(select * from clientstate
               where cs_from_c_id = :c_id and
                    (cs_client_state is distinct from :res_client_state or
                     cs_date is distinct from maxvalue(cs_date, :res_date) or
                     cs_region_id is distinct from :res_region_id or
                     cs_lo_id is distinct from :res_lo_id or
                     coalesce(cs_kf_name, '') is distinct from coalesce(:res_kf_name, '') or
                     cs_client_circle is distinct from :res_circle or
                     cs_staff_state is distinct from :res_staff_state))) then
      is_changed = 1;
  end

  suspend;
end
