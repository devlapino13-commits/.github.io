CREATE PROCEDURE HP_DO_DISBURSEMENT_CARD_MB(
    K_ID D_ID NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable kontract_no d_str_20;
declare variable disb_kind d_integer;
declare variable reg_id d_integer;
declare variable kolich_client_approved d_integer;
declare variable cashier_id d_id;
declare variable c_name_fio d_str_100;
declare variable manager_id d_id;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable card_id d_integer;
declare variable amount d_currency;
declare variable k_prev_id d_id;
begin
  err_text = '';

  select
    k_kontract_no,
    k_disb_kind,
    k_last_reg_id,
    k_kolich_client_approved,
    coalesce(k_approved_manager,0)
  from kontract kont
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where k_id = :k_id
  into
    :kontract_no,
    :disb_kind,
    :reg_id,
    :kolich_client_approved,
    :manager_id;

  if (f_get_db_id() not in (39801, 41701)) then
    err_text = 'БД не соответствует.';
  else
  if (kontract_no is null) then
    err_text = 'Договор не определён.';
  else
  if (disb_kind <> 6) then
    err_text = f_format('Способ получения по договору %1 не соответствует "выдаче на банковскую карту МБулак".', kontract_no);
  else
  if (kolich_client_approved > 1) then
    err_text = f_format('В договоре %1 более одного участника.', kontract_no);

  if (err_text = '') then
  begin
    select
      kc_from_c_id,
      c_name_fio
    from knclient
      inner join clients
        on c_id = kc_from_c_id
    where kc_from_k_id = :k_id
      and kc_amount_approved > 0
    into
      :cashier_id,
      :c_name_fio;

    select
      card_id
    from hp_client_bank_card_mb(:cashier_id, current_date)
    into
      :card_id;

    if (card_id is null) then
      err_text = f_format('Клиент %1 не имеет активную банковскую карту МБулак', c_name_fio);
  end

  if (err_text = '') then
  begin
    -- бух.проводка
    acc_dt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = 417 and al_account = '10941/1');
    acc_kt = (select al_id from acc_list where al_from_reg_id = :reg_id and al_currency = 417 and al_account = '20001/1');
    select
      left(res_text, 255)
    from hp_do_disbursement(
      :un_id,
      :k_id,
      :reg_id,
      :disb_kind,
      :acc_kt,
      :reg_id,
      :disb_kind,
      :acc_dt,
      :cashier_id,
      :manager_id)
    into
      :err_text;
  end

  if (err_text = '') then
  begin
    execute procedure card_set_risk_group (f_objkind('kontract', 'id') , :k_id);
    update cardlistlnk set
        cll_from_k_id = :k_id
    where cll_card_id = :card_id; 
  end

  suspend;
end
