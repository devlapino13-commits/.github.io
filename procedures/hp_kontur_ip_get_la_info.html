CREATE PROCEDURE HP_KONTUR_IP_GET_LA_INFO(
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_ID)
RETURNS (
    REQ_STATUS D_SMALLINT,
    REQ_DATE D_DATE,
    ACTUALITY_DATE D_DATE,
    OGRN D_STR_CODE,
    FNS_DISQUALIFIED D_BOOL,
    IS_MSP D_BOOL,
    MSP_INCLUDE_DATE D_DATE,
    MSP_DELETE_DATE D_DATE)
AS
declare variable jso d_integer;
declare variable fns_disqualified_jso d_integer;
declare variable include_jso d_integer;
declare variable delete_jso d_integer;
declare variable fns_is_actual d_boolean;
declare variable is_actual_include d_boolean;
declare variable is_actual_delete d_boolean;
begin
  -- RFCRU-5304 получение ответа из Контур (Юридические признаки)
  -- входной object_kind может быть miniapp, kontract, knclient, clients.
/*
  req_status -- eil_status
  req_date -- Дата запроса
  actuality_date -- Дата актуальности (Дата запроса + eic_info_actual_days
*/

  select jso, req_status, req_date, actuality_date
  from hp_extinfo_get_jso('kontur_ip_la', :object_kind, :object_id)
  into jso, req_status, req_date, actuality_date;

/*
  ogrn -- Регистрационный номер индивидуального предпринимателя (ОГРНИП)  Api3ReqResponse[0].ogrn
  fns_disqualified  -- Сведения из реестра дисквалифицированных лиц фнс  LegalAnalytics[0].legalAnalyticsData.fnsDisqualified.data boolean
  is_msp -- МСП   LegalAnalytics[0].legalAnalyticsData.enterpriseIncludeDate.data (дата включения)
                        LegalAnalytics[0].legalAnalyticsData.enterpriseDeleteDate.data (дата исключения)
                        если ИМЕЕТСЯ запись о дате включения в Реестр, а записи о дате исключения при этом НЕ ИМЕЕТСЯ или она имеется,
                        но дата исключения предшествует дате включения - 1, иначе - 0
  msp_include_date --  Дата внесения МСП LegalAnalytics[0].legalAnalyticsData.enterpriseIncludeDate.data
*/

  if (coalesce(req_status, 0) < 2) then   -- запрос не найден
  begin
    suspend;
    exit;
  end

  if (jso is null) then
    exception e_default 'json пустой. ' || iif(exists(select 1 from hp_db_is_actual), '', 'Необходимо синхронизировать blobfiles');

  select ogrn
  from hp_kontur_ip_get_base_info(:object_kind, :object_id)
  into ogrn;

  for select json.obj_get_val(jso, 'ogrn'),
             json.obj_get_obj(jso, 'legalAnalyticsData.fnsDisqualified'),
             json.obj_get_obj(jso, 'legalAnalyticsData.enterpriseIncludeDate'),
             json.obj_get_obj(jso, 'legalAnalyticsData.enterpriseDeleteDate')
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'data'), 1)
  where (coalesce(:ogrn, '') = '' or json.obj_get_val(jso, 'ogrn') = :ogrn)
  rows 1
  into ogrn, fns_disqualified_jso, include_jso, delete_jso
  do
  begin
    if (fns_disqualified_jso is not null) then
    begin
      -- Сведения из реестра дисквалифицированных лиц фнс
      fns_is_actual = json.obj_get_val(fns_disqualified_jso, 'isActual');

      if (fns_is_actual) then
        fns_disqualified = iif(cast(json.obj_get_val(fns_disqualified_jso, 'data', false) as d_boolean), 1, 0);
    end

    if (include_jso is not null) then
    begin
      -- Дата внесения МСП
      is_actual_include = json.obj_get_val(include_jso, 'isActual');

      if (is_actual_include) then
        msp_include_date = cast(nullif(json.obj_get_val(include_jso, 'data', ''), '') as d_date);
    end

    if (delete_jso is not null) then
    begin
      -- Дата исключения МСП
      is_actual_delete = json.obj_get_val(delete_jso, 'isActual');

      if (is_actual_delete) then
        msp_delete_date = cast(nullif(json.obj_get_val(delete_jso, 'data', ''), '') as d_date);
    end

    is_msp = iif(is_actual_include and msp_include_date is not null
                 and (is_actual_delete = false
                      or (iif(msp_delete_date is not null, msp_delete_date < msp_include_date, true))
                      ),
                  1, 0);
  end

  suspend;
end
