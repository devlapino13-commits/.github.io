CREATE PROCEDURE HP_RECALC_REBUKE_SCORES(
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    ERR_TEXT D_STR_100)
AS
declare variable KS_SKS_JL_ID_LIST varchar(1024);
declare variable KM_JL_ID_LIST varchar(1024);
declare variable RM_JL_ID_LIST varchar(1024);
declare variable KOBD_JL_ID_LIST varchar(1024);
declare variable REP_DATE_END D_DATE;
declare variable C_ID D_ID;
declare variable USING_SCORES D_INTEGER;
declare variable C_ID_LIST D_BLOB_TEXT;
declare variable JOB_KIND D_STR_10;
declare variable REG_ID D_INTEGER;
declare variable KS_SKS_COUNTER D_INTEGER;
declare variable KM_COUNTER D_INTEGER;
declare variable RM_COUNTER D_INTEGER;
declare variable KOBD_COUNTER D_INTEGER;
declare variable DELIMETR D_STR_10;
declare variable LAST_DATE date;
begin
  res_text = '';
  err_text = '';
  delimetr = ascii_char(13) || ascii_char(10);
  ks_sks_counter = 0;
  km_counter = 0;
  rm_counter = 0;
  kobd_counter = 0;
  rep_date_end = cast('01.' || extract(month from current_date) || '.' || extract(year from current_date) as date) - 1;

  select rs.rs_date
  from rebuke_scores rs
  where rs.rs_id = -1
  into :last_date;

  if (:rep_date_end = coalesce(:last_date, '01.01.1900')) then
  begin
    err_text = 'Рассчет баллов за прошлый месяц уже был произведен.';
    suspend;
    exit;
  end

  select '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jl.jl_kind = jk.jk_id
  where jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')
  into :ks_sks_jl_id_list;

  select '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jl.jl_kind = jk.jk_id
  where jk_system_code = 'jobsystem.region_km'
  into :km_jl_id_list;

  select '.' || list(jl.jl_id, '.') || '.'
  from jobsystem js
  inner join joblist jl on js.js_id = jl.jl_kind
  where js.js_memo = 'JobSystem.Region_Director' and
        js.js_id >= 0
  into :rm_jl_id_list;

  select '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jl.jl_kind = jk.jk_id and
        jk_system_code = 'JobSystem.Specialist_KOBD'
  where jl.jl_name <> 'Выездной кассир'
  into :kobd_jl_id_list;

  for with tmp
      as (select c_id, jl_id, sum_scores, reg_id
          from aarep_rebuke_scores(1000000, current_date)
          where sum_scores > 0)
      select '.' || list(c_id, '.') || '.', 'ks_sks', 1000000
      from tmp
      where :ks_sks_jl_id_list containing '.' || jl_id || '.'
      union all
      select '.' || list(c_id, '.') || '.', 'km', reg_id
      from tmp
      where :km_jl_id_list containing '.' || jl_id || '.'
      group by 3
      union all
      select '.' || list(c_id, '.') || '.', 'rm', reg_id
      from tmp
      where :rm_jl_id_list containing '.' || jl_id || '.'
      group by 3
      union all
      select '.' || list(c_id, '.') || '.', 'kobd', reg_id
      from tmp
      where :kobd_jl_id_list containing '.' || jl_id || '.'
      group by 3
      into :c_id_list, :job_kind, :reg_id

  do
  begin
    if (c_id_list is null) then
      continue;
    if (job_kind = 'ks_sks') then
    begin
      for select c_id, using_scores
          from cb_kg_lo_220106_one_lo(:reg_id, :c_id_list, :rep_date_end, 0)
          where using_scores > 0
          into :c_id, :using_scores
      do
      begin
        insert into rebuke_scores (rs_from_c_id, rs_scores, rs_from_un_id, rs_date)
        values (:c_id, (0 - :using_scores), :un_id, current_date);
        ks_sks_counter = :ks_sks_counter + 1;
      end
    end
    if (job_kind = 'km') then
    begin
      for select using_scores, c_id
          from cb_kg_km_2022_06(:reg_id, :rep_date_end, :c_id_list)
          where using_scores > 0
          into :using_scores, :c_id
      do
      begin
        insert into rebuke_scores (rs_from_c_id, rs_scores, rs_from_un_id, rs_date)
        values (:c_id, (0 - :using_scores), :un_id, current_date);
        km_counter = :km_counter + 1;
      end
    end
    if (job_kind = 'rm') then
    begin
      for select using_scores, c_id
          from cb_kg_rm_2022_06(:rep_date_end, :c_id_list)
          where using_scores > 0
          into :using_scores, :c_id
      do
      begin
        insert into rebuke_scores (rs_from_c_id, rs_scores, rs_from_un_id, rs_date)
        values (:c_id, (0 - :using_scores), :un_id, current_date);
        rm_counter = :rm_counter + 1;
      end
    end
    if (job_kind = 'kobd') then
    begin
      for select using_scores, c_id
          from cb_kg_kobd_2022_06(:reg_id, :rep_date_end, :c_id_list)
          where using_scores > 0
          into :using_scores, :c_id
      do
      begin
        insert into rebuke_scores (rs_from_c_id, rs_scores, rs_from_un_id, rs_date)
        values (:c_id, (0 - :using_scores), :un_id, current_date);
        kobd_counter = :kobd_counter + 1;
      end
    end
  end

  update or insert into rebuke_scores (rs_id, rs_from_un_id, rs_datetime, rs_date)
  values (-1, :un_id, current_timestamp, :rep_date_end)
  matching (rs_id);

  if ((:ks_sks_counter + :km_counter + :rm_counter + kobd_counter) = 0) then
    res_text = res_text || 'За прошлый месяц баллы не были списаны.' || :delimetr;
  else
  begin
    res_text = res_text || 'За прошлый месяц штрафные баллы были списаны:' || :delimetr;
    res_text = res_text || 'КС и СКС: ' || ks_sks_counter || ' сотрудник(ов)' || :delimetr;
    res_text = res_text || 'КМ: ' || km_counter || ' сотрудник(ов)' || :delimetr;
    res_text = res_text || 'РМ: ' || rm_counter || ' сотрудник(ов)' || :delimetr;
    res_text = res_text || 'КОБД: ' || kobd_counter || ' сотрудник(ов)' || :delimetr;
  end
  res_text = res_text || 'Редактирование/удаление приказов дисциплинарных взысканий заблокировано от ' || f_datetostr(current_date);
  suspend;
end
