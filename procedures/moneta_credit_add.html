CREATE PROCEDURE MONETA_CREDIT_ADD(
    ESM_ID ID_TYPE,
    FULL_SOURCE D_BLOB_TEXT,
    ANSWER D_BLOB_TEXT,
    ERROR_CODE D_BLOB_TEXT)
AS
declare variable k_id d_id;
declare variable esm_date d_date;
declare variable edoc_pmt_project_disb d_str_255;
begin
  -- Процедура производит сохранение результата выдачи на карту/СБП/счет
  -- Вызывается из сервиса выдач по Монете.

  error_code = coalesce(error_code, '');
  answer = coalesce(answer, '');
  full_source = coalesce(full_source, '');

  if (esm_id is null) then
    exception e_default 'ESM_ID is null';

  select esm_from_k_id, esm_date, esm_project
  from extsysmain
  where esm_id = :esm_id
  and esm_project in ('moneta_disb', 'wirebank_disb')
  and esm_action in ('CardDisb', 'PhoneDisb', 'AccDisb')
  into k_id, esm_date, edoc_pmt_project_disb;

  -- исхожу из того, что не может быть такого, чтобы  у extsysmain не был
  -- заполнен esm_date
  if (esm_date is null) then
    exception e_default 'ESM_DATE is not set for esm_id: ' || coalesce(esm_id, 'NULL');

  delete from extsysanswer
  where esa_from_esm_id = :esm_id;

  if (error_code <> '') then
  begin
    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'error: ' || :error_code);
  end
  if (answer <> '') then
  begin
    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'answer: ' || :answer);
  end

  update extsysmain
  set esm_status = iif(:error_code = '', 1, -1),
      esm_full_source = iif(:full_source <> '', :full_source, esm_full_source)
  where esm_id = :esm_id;

  if (error_code = '') then -- если нет ошибки, тогда формирую задачу на создание проводки - комиссия за перевод денежных стредств
  begin
    -- Важно! В качестве object_kind, object_id указываю extsysmain записи, на основании которой был осуществлен перевод денег чз Монету клиенту
    insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                            esm_full_source, esm_amount, esm_from_k_id, esm_account)
    values (:edoc_pmt_project_disb, 'OperbookCreate', :esm_date, f_objkind('extsysmain', 'id'), :esm_id, 0, '', 0, :k_id, '');
  end
end
