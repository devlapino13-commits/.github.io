CREATE PROCEDURE FM_GET_PROVIDER_SWITCHER_DATA(
    UN_ID D_ID)
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable UN_NAME STR_100;
declare variable CURR_PAYMENTS STR_255;
declare variable CURR_DISBURSEMENTS STR_100;
declare variable LOG_DATETIME timestamp;
declare variable UN_FIO STR_100;
declare variable LOG_OLD_VAL STR_100;
declare variable LOG_NEW_VAL STR_100;
declare variable COLOR_GREEN STR_20;
begin

  select un_name
  from  username
  where un_id = :un_id 
  into :un_name;

  color_green = '#02b316';

  if (:un_name <> 'zgitinova') then
  begin
    res_html = '<html><body><font color="red"><b>Ошибка: доступ запрещен.</b></font></body></html>';
    suspend;
    exit;
  end

  res_html = (select res_style from fm_html_styles) || '<body>';

  res_html = blob_append(res_html, f_format('<h1 class="green_text">%1</h1>', 'Переключатель поставщика'));
  res_html = blob_append(res_html, '<b>Пользователь:</b> ' || :un_name || '<br><br>');

  select coalesce(pmt_project, 'не определен')
  from hp_edoc_web_get_pmt_project
  into :curr_payments;

  res_html = blob_append(res_html, '<table width="100%"><col width="60%"><col width="40%">');
  
  res_html = blob_append(res_html,
    '<tr><td class="filled" align="left"><b>Текущее значение (погашения)</b></td>' ||
    '<td class="filled" align="right"><font color="' || :color_green || '"><b>' || upper(:curr_payments) || '</b></font></td></tr>' ||
    '<tr><td align="left">&nbsp;&nbsp;Изменить оператора погашений на:</td>' ||
    '<td align="right">' ||
      '<b>' || f_uri_sql('Т-Банк', 'set_pmt_proj', 'provider_switcher', 0, 'none', 'Переключить на Т-Банк?', '', 0, 0, '', '', 'dialog', 'tinkoff') || '</b>' || '&nbsp;&nbsp;' ||
      '<b>' || f_uri_sql('Монета', 'set_pmt_proj', 'provider_switcher', 0, 'none', 'Переключить на Монету?', '', 0, 0, '', '', 'dialog', 'moneta') || '</b>' || '&nbsp;&nbsp;' ||
      '<b>' || f_uri_sql('ТКБ', 'set_pmt_proj', 'provider_switcher', 0, 'none', 'Переключить на ТКБ?', '', 0, 0, '', '', 'dialog', 'wirebank') || '</b>' ||
    '</td></tr>' ||
    '</table><br>');

  res_html = blob_append(res_html, '<h3>История изменений (погашения):</h3>');
  res_html = blob_append(res_html, '<table width="100%"><tr><th>Дата/время</th><th>Пользователь</th><th>Старое значение</th><th>Новое значение</th></tr>');
  
  log_new_val = :curr_payments;
  for select first 5 
             cl.cvl_value,
             cl.cvl_datetime, 
             u.un_fio
      from const_var_log cl
      inner join username u on u.un_id = cl.cvl_from_un_id
      where cl.cvl_code = 'edoc_pmt_project'
      order by cl.cvl_datetime desc
      into :log_old_val, :log_datetime, :un_fio
  do
  begin
    res_html = blob_append(res_html, '<tr><td>' || f_datetimetostr(:log_datetime) || '</td><td>' || :un_fio || '</td><td>' || upper(coalesce(:log_old_val, '-')) || '</td><td>' || upper(coalesce(:log_new_val, '-')) || '</td></tr>');
    log_new_val = log_old_val; 
  end
  res_html = blob_append(res_html, '</table><br><br>'); 
    
  select coalesce(cv_value, 'не определен')
  from const_var
  where cv_code = 'edoc_pmt_project_disb'
  into :curr_disbursements;

  res_html = blob_append(res_html, '<table width="100%"><col width="60%"><col width="40%">');
  
  res_html = blob_append(res_html,
    '<tr><td class="filled" align="left"><b>Текущее значение (выдачи)</b></td>' ||
    '<td class="filled" align="right"><font color="' || :color_green || '"><b>' || upper(:curr_disbursements) || '</b></font></td></tr>' ||
    '<tr><td align="left">&nbsp;&nbsp;Изменить оператора выдач на:</td>' ||
    '<td align="right">' ||
      '<b>' || f_uri_sql('Монета', 'set_disb_proj', 'provider_switcher', 0, 'none', 'Переключить выдачи на Монету?', '', 0, 0, '', '', 'dialog', 'moneta') || '</b>' || '&nbsp;&nbsp;' ||
      '<b>' || f_uri_sql('ТКБ', 'set_disb_proj', 'provider_switcher', 0, 'none', 'Переключить выдачи на ТКБ?', '', 0, 0, '', '', 'dialog', 'wirebank') || '</b>' ||
    '</td></tr>' ||
    '</table><br>');
  res_html = blob_append(res_html, '<h3>История изменений (выдачи):</h3>');
  res_html = blob_append(res_html, '<table width="100%"><tr><th>Дата/время</th><th>Пользователь</th><th>Старое значение</th><th>Новое значение</th></tr>');
  
  log_new_val = :curr_disbursements;
  for select first 5 
             cl.cvl_value,
             cl.cvl_datetime, 
             u.un_fio
      from const_var_log cl
      inner join username u on u.un_id = cl.cvl_from_un_id
      where cl.cvl_code = 'edoc_pmt_project_disb'
      order by cl.cvl_datetime desc
      into :log_old_val, :log_datetime, :un_fio
  do
  begin
    res_html = blob_append(res_html, '<tr><td>' || f_datetimetostr(:log_datetime) || '</td><td>' || :un_fio || '</td><td>' || upper(coalesce(:log_old_val, '-')) || '</td><td>' || upper(coalesce(:log_new_val, '-')) || '</td></tr>');
    log_new_val = log_old_val;
  end
  res_html = blob_append(res_html, '</table><br>'); 

  res_html = blob_append(res_html, '</body></html>');

  suspend;
end
