CREATE PROCEDURE SA_MAKE_WEEKENDS(
    D1 DATE,
    D2 DATE,
    ADD_SATURDAY SMALLINT,
    ADD_SUNDAY SMALLINT)
RETURNS (
    HOLIDAY_DATE DATE)
AS
declare variable i integer;
declare variable y1 integer;
declare variable y2 integer;
declare variable mon integer;
declare variable days integer;
declare variable k integer;
begin
  /* Procedure Text */
  holiday_date = null;
  y1 = extract(year from d1);
  y2 = extract(year from d2);
  i = y1;
  while (i <= y2) do
  begin
    for select scal_month, scal_day
        from sa_calendar
        where scal_active = 1 and
              scal_begin_date <= :d1
        into :mon, :days
    do
    begin
      holiday_date = cast((cast(days as varchar(2)) || '.' || cast(mon as varchar(2)) || '.' || cast(i as varchar(4))) as date);
      if ((holiday_date >= d1) and
          (holiday_date <= d2)) then
        suspend;
    end
    i = i + 1;
  end
  if ((add_saturday = 1) or (add_sunday = 1)) then
  begin
    holiday_date = d1;
    while (holiday_date <= d2) do
    begin
      k = extract(weekday from holiday_date);
      if ((add_sunday = 1) and
          (k = 0)) then
        suspend;
      if ((add_saturday = 1) and
          (k = 6)) then
        suspend;
      holiday_date = holiday_date + 1;
    end
  end
end
