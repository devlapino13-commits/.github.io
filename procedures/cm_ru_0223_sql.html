CREATE PROCEDURE CM_RU_0223_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable cds_doc_blob_1 d_id;
declare variable cds_doc_blob_2 d_id;
declare variable c_id d_id;
declare variable k_disb_kind d_integer;
declare variable kc_circle d_integer;
declare variable k_disb_date d_date;
declare variable cds_id d_id;
declare variable pr_code d_str_100;
declare variable cds_doc_date_begin d_date;
declare variable cds_doc_date_end d_date;
declare variable sign_edoc_zalog d_smallint;
declare variable cds_create_datetime d_datetime;
declare variable approve_datetime d_datetime;
begin
  sign_edoc_zalog = f_get_const('sign_edoc_zalog');

  select kl_datetime
  from kontlog
  where kl_from_k_id = :k_id and
        kl_state = 1
  order by kl_datetime rows 1
  into approve_datetime;

  for select knc.kc_from_c_id, k.k_disb_kind, knc.kc_circle, k_disb_date, p.pr_code
      from kontract k, knclient knc, product p
      where k_id = knc.kc_from_k_id and
            p.pr_id = k.k_from_pr_id and
            knc.kc_amount_approved > 0 and
            k_id = :k_id
      into :c_id, k_disb_kind, kc_circle, k_disb_date, pr_code
  do
  begin

    if (check_kind containing '.loan_approve.' or check_kind containing '.loan_disb.' or check_kind containing '.Check_Preview.') then
    begin
      cds_id = 0;
      cds_doc_blob_1 = 0;
      cds_doc_blob_2 = 0;

      select cds.cds_id, cds.cds_doc_blob_1, cds.cds_doc_blob_2, cds_doc_date_begin, cds_doc_date_end, cds_create_datetime
      from clientdocs cds, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk_id and
            cds_is_delete = 0 and
            pk_from_pdt_id = pdt_id and
            --:k_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, cds_doc_date_begin + 2) and
            pdt_code in ('video_sertif')
      order by cds.cds_create_datetime desc
      rows 1
      into :cds_id, cds_doc_blob_1, cds_doc_blob_2, cds_doc_date_begin, cds_doc_date_end, cds_create_datetime;

      if (k_disb_kind in (7,8)) then
      begin
        if (cds_id = 0) then
        begin
          res_error_text = 'Не закреплен фото документ "Видео-сертификация"';
          res_error_is_critical = 2;
          suspend;
        end
        else
        begin
          if (coalesce(cds_doc_blob_1, 0) + coalesce(cds_doc_blob_2, 0) = 0) then
          begin
            res_error_text = 'Не закреплено фото в слоте "Видео-сертификация"';
            res_error_is_critical = 2;
            suspend;
          end

          if (cast(cds_create_datetime as date) < coalesce(cast(:approve_datetime as date), current_date)) then
          begin
            res_error_text = 'Нет документа "Видео-сертификация" либо срок документа устарел';
            res_error_is_critical = 2;
            suspend;
          end
        end
      end
    end

    if (check_kind containing '.loan_disb.') then
    begin
      if ((pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.') and
          sign_edoc_zalog = 0) then
      begin
        cds_id = 0;
        cds_doc_blob_1 = 0;
        cds_doc_blob_2 = 0;

        select coalesce(cds.cds_id, 0), cds.cds_doc_blob_1, cds.cds_doc_blob_2
        from clientdocs cds, passkind pk, passdoctype
        where cds_from_c_id = :c_id and
              cds_from_pk_id = pk_id and
              cds_is_delete = 0 and
              pk_from_pdt_id = pdt_id and
              :k_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.2999') and
              pk.pk_id = 364
        order by cds.cds_create_datetime desc
        rows 1
        into :cds_id, cds_doc_blob_1, cds_doc_blob_2;
        if (cds_id = 0) then
        begin
          res_error_text = 'Не закреплен фото документ "Договор залога"';
          res_error_is_critical = 2;
          suspend;
        end
        else
        if (cds_doc_blob_1 + cds_doc_blob_2 = 0) then
        begin
          res_error_text = 'Не закреплено фото в слоте "Договор залога"';
          res_error_is_critical = 2;
          suspend;
        end
      end
    end
  end
end
