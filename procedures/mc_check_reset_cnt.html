CREATE PROCEDURE MC_CHECK_RESET_CNT
AS
declare variable mcc_id d_id;
declare variable mcc_check_kind d_integer;
begin
    -- RFCKG-3246
    for select mcc_id, mcc_check_kind
        from mc_check
        where mcc_to_reset_cnt = 1
        into :mcc_id, :mcc_check_kind
    do
    begin
        if (mcc_check_kind in (0, 1)) then -- Логин и смена номера
        begin
            update mc_check mcc
            set
              mcc_login_try_cnt = iif(mcc_login_try_cnt > 0, mcc_login_try_cnt - 1, 0),
              mcc_send_code_cnt = iif(mcc_send_code_cnt > 0, mcc_send_code_cnt - 1, 0),
              mcc_check_code_cnt = iif(mcc_check_code_cnt > 0, mcc_check_code_cnt - 1, 0),
              mcc_last_reset_time = current_timestamp
            where mcc_id = :mcc_id;
        end
        else if (mcc_check_kind = 2) then -- Смена номера без наличия доступа к старому номеру
        begin
            update mc_check mcc
            set
              mcc_login_try_cnt = iif(mcc_login_try_cnt > 0, mcc_login_try_cnt - 1, 0),
              mcc_send_code_cnt = iif(mcc_send_code_cnt > 0, mcc_send_code_cnt - 1, 0),
              mcc_check_code_cnt = iif(mcc_check_code_cnt > 0, mcc_check_code_cnt - 1, 0),
              mcc_last_reset_time = current_timestamp
            where mcc_id = :mcc_id and
                  cast(datediff(second, mcc_last_reset_time, current_timestamp) / 3600 as integer) >= 12; -- Очищаем раз в 12 часов
        end

        when any do
        begin
        end
    end
end
