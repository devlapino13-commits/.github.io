CREATE PROCEDURE DEBTORS_LIST_FOR_OPERATORS(
    REP_DATE TIMESTAMP,
    REP_DELIQ_DAYS_FROM INTEGER,
    REP_DELIQ_DAYS_TO INTEGER,
    REP_REGION_ID INTEGER,
    KONT_NO VARCHAR(255))
RETURNS (
    CCDL_ID DECIMAL(18,0),
    KONTRACT_ID DECIMAL(18,0),
    KONTRACT_NO VARCHAR(20),
    DEBTOR_FIO VARCHAR(255),
    DEBTOR_PHONE VARCHAR(255),
    CONTACT_FIO VARCHAR(255),
    CONTACT_PHONE VARCHAR(255),
    KS_FIO VARCHAR(255),
    KS_PHONE VARCHAR(255),
    KF_BRANCH_NAME VARCHAR(255),
    DELIQ_DAYS INTEGER,
    DELIQ_SUMM DECIMAL(18,2),
    LOCKED_OPERATOR VARCHAR(255),
    DAY_CALLS_COUNT INTEGER,
    FULL_CALLS_COUNT INTEGER,
    LAST_CALL TIMESTAMP,
    LAST_CALL_OPER_FIO VARCHAR(255),
    IS_LOCKED SMALLINT,
    IS_CONTACT SMALLINT,
    LAST_RESULT SMALLINT,
    RESULT_STRING VARCHAR(1000),
    CONTACT_STATUS VARCHAR(255),
    PAY_DATE DATE,
    IS_DEBITOR SMALLINT,
    FIO VARCHAR(255))
AS
declare variable ccdr_call_datetime timestamp;
declare variable ccdr_oper_fio char(255);
declare variable rep_date_buf date;
declare variable ccdl_from_contacts_id integer;
declare variable ccdr_result smallint;
declare variable ccdr_call_datetime_buf date;
declare variable ccdr_pay_date date;
declare variable rep_date_end date;
declare variable is_auto smallint;
begin
  /*
    CREATED BY IGUSEV 17.07.2012

    PURPOSE: Предоставляет список респондентов (должники + конт. лица) для обзвона

    CHANGED BY IGUSEV
    CHANGES: добавлено условие  nd cdbl.ccdl_exclude_date is null - только не убранные оператором
  */

  if (rep_deliq_days_from is null) then
    rep_deliq_days_from = 0;

  if ((rep_deliq_days_to is null) or (rep_deliq_days_to = 0)) then
    rep_deliq_days_to = 9999;

  if (kont_no is null) then
    kont_no = '';

  if ((rep_region_id = 0) or (rep_region_id is null)) then
  begin
    rep_region_id = 1000000;
  end

  rep_date_end = cast('now' as date);

  rep_date_buf = cast(rep_date as date);/* REP_DATE - TIMESTAMP т.к. в asp нельзя передать в DataSet автоматом тип Date*/
  result_string = '';
  for select ccdl_id, coalesce(ccdl_from_contacts_id, 0) contacts_id, cl.c_name_fio,
             /* ФИО должника*/
             (select cont_value
              from hp_contdata_get_cell_valid(0, cdbl.ccdl_from_c_id)), /* номер телефона должника*/
             cont.cc_name, /* ФИО конт лица*/
             cont.cc_cell_no, /* тел конт лица*/
             cl2.c_name_fio,
             /* ФИО КС*/
             (select cont_value
              from hp_contdata_get_cell_valid(0, cdbl.ccdl_officer_id)), /* номер телефона КС*/
             reg.reg_name, /* наименование офиса кред. папки*/
             cdbl.ccdl_deliq_days, /* число дней просрочки*/
             cdbl.ccdl_deliq_os + cdbl.ccdl_deliq_perc, /* сумма задолжности*/
             cdbl.ccdl_locked_operator, /* опертор заблокировавший запись на время звонка*/
             kont.k_kontract_no, /* № кредитного договора*/
             cont.cc_status, /* статус конт. лица (брат, сестра,....)*/
             cdbl.ccdl_from_k_id,
             /* k_id*/
             (
             case
               when ccdl_from_contacts_id is null then cl.c_name_fio
               else cont.cc_name
             end) as fio
      from callcenter_debtors_list cdbl
      inner join kontract kont on kont.k_id = cdbl.ccdl_from_k_id
      inner join clients cl on cl.c_id = cdbl.ccdl_from_c_id
      inner join clients cl2 on cl2.c_id = cdbl.ccdl_officer_id

      left join contacts cont on cont.cc_id = cdbl.ccdl_from_contacts_id
      inner join region_list(:rep_region_id, 1, -1, 1) regl on regl.id_list = ccdl_kf_branch_id
      inner join region reg on reg.reg_unicode = regl.id_list
      inner join kontstate ks on k_id = ks_from_k_id
      where cdbl.ccdl_deliq_closed_date is null /* не погашенные*/
            and
            cdbl.ccdl_exclude_date is null /* только не убранные оператором*/
            and
            cdbl.ccdl_timeout_date is null /* только должники/конт. лица которые не исчесчезли по timeout-у т.к. им звонили на протяжении 3 дней */
            and
            ccdl_deliq_days >= :rep_deliq_days_from /* число дней задолжности начиная с */
            and
            ccdl_deliq_days <= :rep_deliq_days_to /* число дней задолжности начиная с */
            and
            ((kont.k_kontract_no = :kont_no) or
            (:kont_no = '')) and
            ks_ol_is_delete = 0 and
            ks_state = 1 /*только активные*/
            and
            :rep_date_end between ks_date_begin and ks_date_end
      order by reg.reg_name, k_kontract_no, cl.c_name_fio, contacts_id
      into :ccdl_id, :ccdl_from_contacts_id, :debtor_fio, /* ФИО должника*/
           :debtor_phone, /* номер телефона должника*/
           :contact_fio, /* ФИО конт лица*/
           :contact_phone, /* тел конт лица*/
           :ks_fio, /* ФИО КС*/
           :ks_phone, /* номер телефона КС*/
           :kf_branch_name, /* наименование офиса кред. папки*/
           :deliq_days, /* число дней просрочки*/
           :deliq_summ, /* сумма задолжности*/
           :locked_operator, /* опертор заблокировавший запись на время звонка*/
           :kontract_no, :contact_status, :kontract_id, :fio
  do
  begin
    result_string = kontract_no || ';   Дней просроч.: ' || cast(deliq_days as varchar(255)) || ';   Сумма задолжн.: ' || cast(deliq_summ as varchar(255));

    if (locked_operator is null) then
      is_locked = 0;
    else
      is_locked = 1;

    if (ccdl_from_contacts_id = 0) then
      /* опеределяю контактное лицо это или заемщик*/
      is_contact = 0;
    else
      is_contact = 1;

    day_calls_count = 0;
    full_calls_count = 0;

    pay_date = null;

    last_call = null;
    last_call_oper_fio = null;
    last_result = null;
    for select ccdr_call_datetime, ccdr_oper_fio, cdr.ccdr_result, cdr.ccdr_pay_date, cdr.ccdr_is_auto
        from callcenter_debtors_result cdr
        /* where  cdr.ccdr_is_auto = 0 */ /* запись сформирокана оператором, а не автоматически*/
        where cdr.ccdr_from_ccdl_id = :ccdl_id
        order by cdr.ccdr_call_datetime
        into :ccdr_call_datetime, :ccdr_oper_fio, :ccdr_result, :ccdr_pay_date, :is_auto
    do
    begin
      ccdr_call_datetime_buf = cast(ccdr_call_datetime as date);

      if (is_auto = 0) then
      begin
        if (ccdr_call_datetime_buf = rep_date_buf) then
        begin
          day_calls_count = day_calls_count + 1;
        end
        full_calls_count = full_calls_count + 1;
      end
      last_call = ccdr_call_datetime;
      last_call_oper_fio = ccdr_oper_fio;
      last_result = ccdr_result;
      pay_date = :ccdr_pay_date;
    end
    /*  if (LAST_RESULT != 1) then */ /* если последий результат звонка клиенту/конт лицу <> положительный тогда в списке текущих*/
    suspend;
  end
end
