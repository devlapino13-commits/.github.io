CREATE PROCEDURE CM_AZ_0059_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable CLIENT_POINT D_INTEGER;
declare variable PR_ID D_ID;
declare variable MA_AMOUNT D_CURRENCY;
declare variable ACTIVE_CNT D_INTEGER;
declare variable MDS D_CURRENCY;
declare variable DELINQ_EPISODES D_INTEGER;
declare variable MAX_DELINQ_DAYS D_INTEGER;
declare variable MAX_AMOUNT_BY_POINTS D_CURRENCY;
declare variable ACTIVE_BALANCE D_CURRENCY;
begin
  select
    mac.mac_amount,
    ma.ma_from_pr_id,
    mac.mac_from_c_id
  from miniapp ma
  inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
    and mac.mac_amount > 0
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end = '31.12.9999'
  into
    :ma_amount,
    :pr_id,
    :c_id;

  if (pr_id not in (170,171,172,173,174,175,176,177)) then
    exit;


  /* 1. Не более двух активных кредитов */
  select count(distinct ks.ks_from_k_id)
  from knclient kc
  inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
    and current_date between ks.ks_date_begin and ks.ks_date_end
    and ks.ks_ol_is_delete = 0
    and ks.ks_state = 1
  where kc.kc_from_c_id = :c_id
    and kc.kc_amount_approved > 0
  into :active_cnt;

  if (active_cnt >= 2) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Müştərinin 2 aktiv krediti mövcuddur, 3-cü kreditin verilişi qadağan olunur';
    suspend;
  end

  /* 2. Минимум 20 баллов */
  client_point = f_point(:c_id, current_date);
  if (client_point < 20) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Cari məhsul üzrə müştərinin minimum 20 balı olmalıdır.';
    suspend;
  end

  /* 2a. Активная просрочка */
  if (exists
    (
      select 1
      from knclient kc
      inner join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id
      where kc.kc_from_c_id = :c_id
        and oks.oks_close_state = 0
        and current_date between oks.oks_date_begin and oks.oks_date_end
        and current_date - oks.oks_delinq_date > 0
    )
  ) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Müştərinin aktiv gecikməsi mövcuddur.';
    suspend;
  end

  /* 2b. Не более 1 просрочки за год, и она ≤3 дня */
  select count(distinct ok.ok_sh_date), coalesce(max(ok.ok_ol_date - ok.ok_sh_date), 0)
  from operkred ok
  inner join knclient kc on kc.kc_from_k_id = ok.ok_from_k_id
  where kc.kc_from_c_id = :c_id
    and ok.ok_ol_is_active = 1
    and ok.ok_ol_date <= current_date
    and ok.ok_ol_date > ok.ok_sh_date
    and ok.ok_type < 7
    and ok.ok_os + ok.ok_perc < 0
    and ok.ok_from_c_id = 0
    and ok.ok_ol_date >= dateadd(year, -1, current_date)
  into :delinq_episodes, :max_delinq_days;

  if (:delinq_episodes > 1 or coalesce(:max_delinq_days, 0) > 3) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Müştərinin son 1 il ərzində 3 gündən artıq gecikməsi olmamalıdır.';
    suspend;
  end
  -- RFCAZ-611
  max_amount_by_points = case
    when client_point between 0 and 2 then 4500
    when client_point between 3 and 5 then 4500
    when client_point between 6 and 11 then 5000
    when client_point between 12 and 17 then 5500
    when client_point between 18 and 23 then 6000
    when client_point between 24 and 29 then 6000
    when client_point between 30 and 35 then 6500
    when client_point between 36 and 41 then 7500
    when client_point between 42 and 47 then 8000
    else 8500 -- 48+
  end;

  /* 4. Проверка лимита по баллам */
  select coalesce(sum(oks.oks_balance), 0)
  from knclient kc
  inner join oksummary oks on oks.oks_from_k_id = kc.kc_from_k_id
    and oks.oks_close_state = 0
    and current_date between oks.oks_date_begin and oks.oks_date_end
  where kc.kc_from_c_id = :c_id
  into :active_balance;

  if (coalesce(:active_balance, 0) + coalesce(:ma_amount, 0) > :max_amount_by_points) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Aktiv kreditin və aksiya çərçivəsində alınan kreditin ümumi məbləği toplanmış ballara görə maksimum kredit məbləğindən çox olmamalıdır';
    suspend;
  end
end
