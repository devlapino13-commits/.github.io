CREATE PROCEDURE AAREP_CENTERBANK_REPORT_DATA(
    KC_ID D_ID,
    REP_DATE D_DATE,
    REP_QUERY_DATE D_DATE)
RETURNS (
    OKS_BALANCE D_CURRENCY,
    OKS_BAL_PERC D_CURRENCY,
    OKS_BAL_ZAD_OS D_CURRENCY,
    OKS_BAL_ZAD_PERC D_CURRENCY,
    OKS_DELINQ_DAYS D_INTEGER,
    OKS_FIRST_DELINQ_90 D_DATE,
    OKS_PMT_OS D_CURRENCY,
    OKS_PMT_PERC D_CURRENCY,
    SELL_DATE D_DATE,
    SELL_PMT_AMOUNT D_CURRENCY,
    WRITEOFF_DATE D_DATE,
    LAST_DATE D_DATE)
AS
declare variable oks_delinq_date d_date;
declare variable rep_date_begin d_date;
declare variable ok_sh_date d_date;
declare variable oks_close_state d_smallint;
declare variable oks_from_k_id d_id;
declare variable oks_date_begin d_date;
begin
  oks_balance = 0;
  oks_bal_perc = 0;
  oks_bal_zad_os = 0;
  oks_bal_zad_perc = 0;
  oks_delinq_days = 0;
  oks_first_delinq_90 = null;
  oks_pmt_os = 0;
  oks_pmt_perc = 0;
  sell_date = null;     -- всегда пусто
  sell_pmt_amount = 0;  -- всегда 0
  writeoff_date = null;
  last_date = null;

  rep_date_begin = rep_date - extract(day from rep_date) + 1;

  select oks_from_k_id, oks_date_begin, oks_balance, oks_bal_perc, oks_bal_zad_os, oks_bal_zad_perc, oks_delinq_date, oks_restruct_date, oks_close_state
  from kn_oksummary
  where oks_from_kc_id = :kc_id and
        oks_date_begin <= :rep_date
  order by oks_from_kc_id desc, oks_date_begin desc
  rows 1
  into oks_from_k_id, oks_date_begin, oks_balance, oks_bal_perc, oks_bal_zad_os, oks_bal_zad_perc, oks_delinq_date, last_date, oks_close_state;

  if (oks_close_state <> 0 or oks_date_begin < rep_date) then
  begin
    oks_balance = 0;
    oks_bal_zad_os = 0;
    oks_bal_zad_perc = 0;
  end else
  begin
    if (oks_balance <> oks_bal_zad_os and abs(oks_balance - oks_bal_zad_os) < 10) then
    begin
      if (f_tmp_kc(oks_from_k_id) = 1) then
        oks_bal_zad_os = oks_balance;
    end
  end

  if (oks_delinq_date is not null) then
    oks_delinq_days = datediff(day, oks_delinq_date, rep_date);
  if (oks_delinq_days < 0) then
    oks_delinq_days = 0;

  if (last_date > rep_date) then
    last_date = null;

  select sum(-ok_os), sum(-ok_perc)
  from kn_operkred
  where ok_from_kc_id = :kc_id and
        ok_ol_is_active = 1 and
        ok_ol_date between :rep_date_begin and :rep_date and
        ok_type = 4
  into oks_pmt_os, oks_pmt_perc;
  if (oks_pmt_os is null) then
    oks_pmt_os = 0;
  if (oks_pmt_perc is null) then
    oks_pmt_perc = 0;

  oks_first_delinq_90 = null;
  writeoff_date = null;

  if (rep_query_date is not null) then
  begin
    select ok_ol_date
    from kn_operkred
    where ok_from_kc_id = :kc_id and
          ok_ol_date <= :rep_query_date and
          ok_ol_is_active = 1 and
          ok_type = 5 and
          (ok_os < 0 or ok_perc < 0)
    order by ok_ol_date
    rows 1
    into writeoff_date;

    ok_sh_date = null;

    if (oks_delinq_date is not null and datediff(day, oks_delinq_date, rep_date) > 90) then
      oks_first_delinq_90 = oks_delinq_date;

    select ok_sh_date
    from kn_operkred
    where ok_from_kc_id = :kc_id and
          ok_ol_date <= :rep_query_date and
          ok_ol_is_active = 1 and
          datediff(day, ok_sh_date, ok_ol_date) > 90 and
          ok_type in (4, 5, 6) and
          (ok_os < 0 or ok_perc < 0)
    order by ok_sh_date
    rows 1
    into ok_sh_date;

    oks_first_delinq_90 = minvalue(coalesce(oks_first_delinq_90, '31.12.9999'), coalesce(ok_sh_date, '31.12.9999'));

    if (oks_first_delinq_90 = '31.12.9999') then
      oks_first_delinq_90 = null;

    last_date = minvalue(coalesce(last_date, '31.12.9999'), coalesce(oks_first_delinq_90 + 90, '31.12.9999'));
    last_date = minvalue(coalesce(last_date, '31.12.9999'), coalesce(sell_date, '31.12.9999'));

    if (last_date = '31.12.9999') then
      last_date = null;
  end

  suspend;
end
