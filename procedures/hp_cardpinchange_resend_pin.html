CREATE PROCEDURE HP_CARDPINCHANGE_RESEND_PIN(
    CPH_ID D_INTEGER NOT NULL,
    CSC_CODE D_STR_100 NOT NULL,
    UN_ID D_ID)
RETURNS (
    CS_ID D_INTEGER,
    ERR_TEXT D_STR_255)
AS
declare variable card_id d_integer;
declare variable reg_id d_integer;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable cca_id d_integer;
declare variable cac_status d_integer;
declare variable date_prev d_datetime;
declare variable retry_period d_integer;
declare variable cell_no d_str_20;
declare variable cs_request_json d_str_8100;
declare variable obj_kind d_integer;
declare variable edl_id d_id;
begin
  cs_id = -1;
  err_text = '';
  obj_kind = f_objkind('cardpinchange', 'id');
  if (csc_code not in ('reset_card_pin_counter', 'resend_pin')) then
  begin
    err_text = 'Недопустимый вх.параметр процедуры';
    suspend;
    exit;
  end
  
  if (csc_code = 'reset_card_pin_counter') then
  begin
    select
      cph_card_id,
      cph_region_id,
      cph_from_c_id,
      cll_from_k_id,
      cal_from_cca_id,
      cph_status
    from cardpinchange
      inner join cardlist
        on cal_card_id = cph_card_id
      inner join cardlistlnk
        on cll_card_id = cph_card_id
    where cph_id = :cph_id
    into
      :card_id,
      :reg_id,
      :c_id,
      :k_id,
      :cca_id,
      :cac_status;
  
    -- проверка статуса
    if (exists(
      select *
      from cardsync
      where cs_object_id = :cph_id
        and cs_object_kind = :obj_kind
        and cs_answer_json is null )) then
    begin
      err_text = 'Предыдущий запрос на переотправку PIN-кода еще не обработан.';
      suspend;
      exit;
    end

    -- новая задача по отправке/переотправке PIN-кода карты клиенту
    cell_no = coalesce(f_cont_info('clients', :c_id, 'cell_main', 'ok'), '');
    if (cell_no = '') then
    begin
      err_text = 'Верифицированный номер телефона клиента не определен.';
      suspend;
      exit;
    end

    cs_request_json = f_format('{"card_id":%1, "phone":"%2", "reason_comment":"Смена пин-кода"}', card_id, cell_no);
    insert into cardsync(
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_k_id,
      cs_from_c_id,
      cs_from_un_id)
    values(
      f_get_cardsynccfg_id(:csc_code),
      :card_id,
      :cca_id,
      :obj_kind,
      :cph_id,
      :cs_request_json,
      :reg_id,
      :k_id,
      :c_id,
      :un_id)
    returning
      cs_id
    into
      :cs_id;
  end else
  begin
    insert into cardsync(
      cs_parent_id,
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_k_id,
      cs_from_c_id,
      cs_from_un_id)
    select
      cs_id,
      f_get_cardsynccfg_id(:csc_code),
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_k_id,
      cs_from_c_id,
      cs_from_un_id
    from cardsync
    where cs_id = -:cph_id
    returning
      cs_id, cs_object_id
    into
      :cs_id, :cph_id;

    select
      edl_id, res_text
    from hp_edoc_insert('change_pincode_transfer', f_objkind('cardpinchange', 'id'), :cph_id, :un_id)
    into :edl_id, :err_text;

    update cardpinchange set
      cph_res_edl_id = :edl_id
    where cph_id = :cph_id;
  end

  suspend;
end
