CREATE PROCEDURE MC_CHECK_ONLINE_LOAN_AVAILABILITY(
    MCL_ID D_ID,
    STAGE D_STR_CODE,
    EXTRA_DATA D_STR_4096 = '')
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255)
AS
declare variable mcl_lang d_str_10;
declare variable c_id d_id;
declare variable ool_id d_id;
declare variable ma_id d_integer;
declare variable un_id d_id;
declare variable phone d_str_20;
declare variable fpb_id d_integer;
declare variable fps_res d_str_code;
declare variable fps_text d_str_255;
declare variable smev_status_code d_integer;
declare variable smev_error_text d_str_255;
declare variable brt_critical d_smallint;
declare variable brt_text d_str_255;
declare variable client_pass_cds_id d_id;
declare variable smevpass_answer_bls_id d_id;
declare variable smevpass_answer_eil_status d_integer;
declare variable smevpass_answer_json d_blob_text;
declare variable ma_date_oper d_date;
declare variable ssp_result_id d_smallint;
declare variable ki_result_id d_smallint;
declare variable kc_id d_id;
declare variable final_pdn d_currency;
declare variable pdn_limit d_currency;
declare variable k_id d_id;
begin
    res_code = 'success';
    res_text = '';
    un_id = 1000000000000;
    

    if (stage = 'check_smev_n_anketa') then
    begin
        -- RFCRU-5817: Проверка паспорта клиента по СМЭВ и проверка заполнения анкетных данных

        c_id = (select mcl.mcl_from_c_id from mc_login mcl where mcl.mcl_id = :mcl_id);
        if (c_id is null) then
        begin
            res_code = 'error';
            res_text = 'Не удалось загрузить данные профиля. Попробуйте войти в приложение заново';
            suspend;
            exit;
        end

        -- Проверка паспорта по СМЭВ
        select first 1 cds_id
        from clientdocs cds
        where cds.cds_from_c_id = :c_id
            and cds_is_main_doc = 1
            and cds_is_delete = 0
            and cds_verify_status = 2
            and cds_index_doc_no <> ''
        order by cds_id desc
        into :client_pass_cds_id;

        select first 1 eil_from_bls_id, eil_status
        from extinfolist, extinfocfg
        where eil_from_c_id = :c_id
            and eil_from_eic_id = eic_id
            and eic_code = 'passport_smev'
            and eil_object_kind = f_objkind('clientdocs', 'id')
            and eil_object_id = :client_pass_cds_id
            and eil_date = current_date
        order by eil_status desc, eil_date desc, eil_id desc
        into :smevpass_answer_bls_id, :smevpass_answer_eil_status;

        smevpass_answer_json = (select cast(blx_blob as d_blob_text) from hp_get_blobex_blob(:smevpass_answer_bls_id));

        -- По идее, такая ситуация на данном этапе невозможна. До того, как попасть сюда, запрос
        -- по СМЭВ отправляется в mc_extinfo_request и проверяется в mc_extinfo_status, но на всякий
        -- случай контрольно проверяем и здесь
        if (smevpass_answer_eil_status <> 2 or smevpass_answer_bls_id is null or smevpass_answer_json is null) then
        begin
            res_code = 'reject';
            res_text = 'Не удалось проверить подлинность паспорта. Попробуйте начать процесс заново';
            suspend;
            exit;
        end

        smev_status_code = json.obj_get_val(smevpass_answer_json, 'status_code');

        -- Паспорт клиента недействителен, отказываем
        if (coalesce(smev_status_code, 0) <> 300) then
        begin
            res_code = 'reject';
            res_text = 'Паспорт недействителен';
            suspend;
            exit;
        end

        -- Проверка заполненности анкетных данных
        if (exists
            (
                select *
                from hp_print_oivo_online_loan_anketa(:c_id, current_date)
                where is_filled = 0
            )
        ) then
        begin
            res_code = 'profile_incomplete';
            res_text = '';
            suspend;
            exit;
        end

        res_code = 'success';
        suspend;
    end

    else if (stage in ('check_fps_bank1', 'check_fps_bank2')) then
    begin
        -- RFCRU-5828

        /*

            Проверка принадлежности номера СБП клиенту (п.16).
            check_fps_bank1 - проверяется номер из mc_login,
            check_fps_bank2 - номер, введенный клиентом (mc_oivo_online_loan_tmp)

        */

        -- актуальная онлайн-заявка пользователя

        c_id = (select mcl.mcl_from_c_id from mc_login mcl where mcl.mcl_id = :mcl_id);
        if (c_id is null) then
        begin
            res_code = 'error';
            res_text = 'Не удалось загрузить данные профиля. Попробуйте войти в приложение заново';
            suspend;
            exit;
        end

        ma_id = f_json_str_get_value(extra_data, 'ma_id', 0);
        if (coalesce(:ma_id, 0) = 0) then
        begin
            res_code = 'error';
            res_text = f_lang(:mcl_lang, 'Не удалось начать проверку. Попробуйте ещё раз');
            suspend;
            exit;
        end

        select ool_id
        from mc_oivo_online_loan ool
        where ool.ool_from_mcl_id = :mcl_id
            and ool.ool_from_ma_id = :ma_id
            and ool.ool_from_k_id is null
        order by ool_id desc
        into ool_id;

        if (ool_id is null) then
        begin
            res_code = 'error';
            res_text = 'Не удалось начать проверку. Попробуйте ещё раз';
            suspend;
            exit;
        end

        -- банк, выбранный клиентом на этапе выбора способа получения
        select fcb.fcb_from_fpb_id
        from miniappclient mac
        inner join fps_application_banks fab on fab.fab_from_mac_id = mac.mac_id
        inner join fps_client_banks fcb on fcb.fcb_id = fab.fab_from_fcb_id
        where mac.mac_from_ma_id = :ma_id
            and mac.mac_from_c_id = :c_id
        into fpb_id;

        if (fpb_id is null) then
        begin
            res_code = 'error';
            res_text = 'Банк СБП не выбран';
            suspend;
            exit;
        end

        if (stage = 'check_fps_bank1') then
            phone = f_cont_info('clients', :c_id, 'cell', 'ok');

        if (stage = 'check_fps_bank2') then
        begin
            phone = null;
            select oolt_phone
            from mc_oivo_online_loan_tmp
            where oolt_from_ool_id = :ool_id
            into phone;

            if (coalesce(phone, '') = '') then
            begin
                res_code = 'error';
                res_text = 'Номер телефона для проверки не указан';
                suspend;
                exit;
            end
        end

        select res_code, res_text
        from hp_oivo_fps_bank_check(:c_id, :phone, :fpb_id, :un_id)
        into fps_res, fps_text;

        if (fps_res = 'success') then
        begin
            -- фиксируем прохождение этапа
            res_code = 'success';

            update mc_oivo_online_loan
            set ool_stage = f_enum('oivo_online_loan', :stage, 'id')
            where ool_id = :ool_id;
        end
        else if (fps_res in ('fail', 'limit')) then
        begin
            if (stage = 'check_fps_bank1') then
            begin
                -- клиент может сменить банк или ввести другой номер
                res_code = 'retry';
                res_text = 'Номер Вашего телефона не найден в СБП банка - проверьте настройки СБП или поменяйте банк';
            end
            else
            begin
                -- отказ в заявке по причине несоответствия номера СБП
                res_code = 'reject';
                res_text = 'Вам отказано в займе по причине несоответствия номера СБП';

                update mc_oivo_online_loan
                set ool_stage = f_enum('oivo_online_loan', 'rejected', 'id')
                where ool_id = :ool_id;
            end
        end
        else
        begin
            -- результат из банка еще не получен либо некорректное состояние
            res_code = 'error';
            res_text = fps_text;
        end

        suspend;
    end

    else if (stage = 'final_check') then
    begin
        /*
            Проверки, выполняемые на данном этапе:

            1. Использование номера
            2. Использование устройства
            3. Банкротство +
            4. БКИ: +
                a. ССП
                b. Проверка КИ
            5. ФСПП ?
            6. Самозапрет +
            7. ПДН +
            8. Финцерт антифрод: +
                a. проверку хэша паспорта
                b. Проверку хэша снилс
                c. проверку номера телефона
                d. Проверку номера СБП
            9. паспорт в СМЭВ +
            10. Расчет скорбалла по антифроду
            11. Отсутствие клиента в Перечнях ЭТ, Решениях МВК и Перечнях Совбеза ООН
        */

        c_id = (select mcl.mcl_from_c_id from mc_login mcl where mcl.mcl_id = :mcl_id);
        if (c_id is null) then
        begin
            res_code = 'error';
            res_text = 'Не удалось загрузить данные профиля. Попробуйте войти в приложение заново';
            suspend;
            exit;
        end

        ma_id = f_json_str_get_value(extra_data, 'ma_id', 0);
        if (coalesce(:ma_id, 0) = 0) then
        begin
            res_code = 'error';
            res_text = f_lang(:mcl_lang, 'Не удалось начать проверку. Попробуйте ещё раз');
            suspend;
            exit;
        end

        -- Проверка 3. Банкротство
        brt_critical = null;
        select first 1 kf_error_is_critical, kf_error_text
        from hp_bankrupt_check('miniapp', :ma_id, :un_id)
        into brt_critical, brt_text;

        if (brt_critical > 0) then
        begin
            res_code = 'reject';
            res_text = 'К сожалению выдача невозможна. Пожалуйста, обратитесь в офис за повторным рассмотрением заявки';
            suspend;
            exit;
        end

        /*
            Проверка 4. БКИ:
                     a. ССП
                     b. Проверка КИ
        */
        select ki_result_id, ssp_result_id
        from hp_pdn_get_request_status(:c_id, null, current_date, :ma_id)
        into :ki_result_id, :ssp_result_id;

        if (coalesce(ki_result_id, 0) not in (5,6,9) or coalesce(ssp_result_id, 0) not in (5,6,9)) then
        begin
            res_code = 'reject';
            res_text = 'К сожалению выдача не возможна. Пожалуйста, обратитесь в офис за повторным рассмотрением заявки';
            suspend;
            exit;
        end

        -- Проверка 6. Самозапрет
        -- Планируемая дата выдачи, самозапрет проверяется именно на нее
        select first 1 ma.ma_date_oper
        from miniapp ma
        where ma.ma_id = :ma_id
            and ma.ma_date_end >= '31.12.2999'
            and ma.ma_is_active = 1
        order by ma.ma_date_app desc
        into :ma_date_oper;

        if (f_is_selfblock(:c_id, :ma_date_oper) = 1) then
        begin
            res_code = 'reject';
            res_text = 'Выдача невозможна. У вас установлен самозапрет на получение кредита';
            suspend;
            exit;
        end

        -- Проверка 7. ПДН
        select kc.kc_id, k.k_id
        from kontract k
        inner join knclient kc on kc.kc_from_k_id = k.k_id
            and kc.kc_from_c_id = :c_id
        where k.k_from_ma_app_id = :ma_id
        into :kc_id, :k_id;

        final_pdn = 0;
        pdn_limit = 0;
        
        select coalesce(kc_pdn_koef, 0)
        from knclient_pdn
        where kc_id = :kc_id
            and kcp_kind = 0
        into final_pdn;
        
        select pdn_limit
        from hp_pdn_get_limit(:kc_id)
        into pdn_limit;

        if (pdn_limit <> 0 and final_pdn > pdn_limit) then
        begin
            res_code = 'reject';
            res_text = 'К сожалению выдача не возможна. Пожалуйста, обратитесь в офис за повторным рассмотрением заявки';
            suspend;
            exit;
        end

        /*
          Проверка
          8. Финцерт антифрод:
            a. проверку хэша паспорта
            b. Проверку хэша снилс
            c. проверку номера телефона
            d. Проверку номера СБП
        */
        if (exists
            (
                select *
                from hp_fincert_get_result(:c_id, null)
                where res = 1
            )
        ) then
        begin
            res_code = 'reject';
            res_text = 'К сожалению выдача не возможна. Пожалуйста, обратитесь в офис за повторным рассмотрением заявки';
            suspend;
            exit;
        end  

        -- Проверка 9. паспорт в СМЭВ
        smev_error_text = '';
        smev_status_code = 0;

        select status_code, error_text
        from hp_smevpass_get_result(:c_id, :ma_id, current_date)
        into :smev_status_code, :smev_error_text;

        if (coalesce(smev_error_text, '') <> '' or coalesce(smev_status_code, 0) <> 300) then
        begin
            res_code = 'reject';
            res_text = 'К сожалению выдача невозможна. Пожалуйста, обратитесь в офис за повторным рассмотрением заявки';
            suspend;
            exit;
        end

        -- Проверка 10. Расчет скорбалла по антифроду
        if (exists
            (
                select *
                from print_result_antifrod(f_objkind('kontract', 'id'), :k_id)
                where risk_level > 0
            )
        ) then
        begin
            res_code = 'reject';
            res_text = 'К сожалению выдача невозможна. Пожалуйста, обратитесь в офис за повторным рассмотрением заявки';
            suspend;
            exit;
        end  
    end

    else
    begin
        res_code = 'error';
        res_text = 'Что-то пошло не так. Пожалуйста, обратитесь к технической поддержке';
        suspend;
    end

    when any do
    begin
        res_code = 'error';
        res_text = 'Техническая ошибка. Попробуйте позже (' || gdscode || ')';
        suspend;
    end
end
