CREATE PROCEDURE FM_EDOC_SIGN(
    UN_ID D_ID,
    EDS_ID D_ID,
    VERIFY_UUID D_STR_CODE = null)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable lang d_str_lang;
declare variable edl_id d_id;
declare variable edl_status d_integer;
declare variable eds_status d_integer;
declare variable edf_id d_id;
begin
  uxf$set_user(un_id);
  res_text = '';
  res_code = 0;
  lang = f_get_lang_code(un_id, null);

  select edl_id, eds_status, iif(coalesce(edl_date_end, current_date) < current_date, -1, edl_status)
  from edoc_sign, edoc_list
  where eds_id = :eds_id and
        eds_from_edl_id = edl_id
  into edl_id, eds_status, edl_status;

  if (res_text = '' and
      edl_status = -1) then
    res_text = f_lang(lang, 'Данные устарели');
  if (res_text = '' and
      eds_status = 1) then
    res_text = f_lang(lang, 'Документы уже подписаны');

  if (res_text = '') then
  begin

    -- Подписываем все отправленные сообщения
    update edoc_sign
    set eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'face', 'id'),
        eds_status = 1
    where eds_id = :eds_id and
          eds_enum_edoc_sign_kind <> f_enum('edoc_sign_kind', 'face', 'id');

    if (verify_uuid is not null) then
    begin
      select udl.udl_object_id
      from uuidlist udl
      where udl.udl_uuid = char_to_uuid(:verify_uuid) and
            udl.udl_object_kind = f_objkind('edoc_face', 'id')
      into edf_id;

      insert into edoc_sign_recog (esr_from_eds_id, esr_from_edf_id)
      values (:eds_id, :edf_id);
    end

    select res_is_error, res_text
    from hp_edoc_com_action(:edl_id, :un_id)
    into res_code, res_text;

    if (res_code = 0) then
    begin
      res_text = 'Документы успешно подписаны';
    end
    else
      res_code = 1;
  end

  suspend;
end
