CREATE PROCEDURE AAREP_LOAN_OFFICERS(
    REG_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    KE_FROM_P_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    KE_CODE VARCHAR(20),
    KE_FROM_REG_ID INTEGER,
    KE_OPER_DAYS INTEGER)
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable db_id integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  ke_from_reg_id = :reg_id;

  for select distinct ke_from_p_id, c_name_fio, ke_code, ke_oper_days
      from krexpert, clients
      where ke_from_p_id = c_id and
            ke_status = 1 and
            ke_date <= :rep_date and
            ke_date_end >= :rep_date and
            ke_is_krs <= 2 and
            ke_from_reg_id = :reg_id
      order by c_name_fio
      into :ke_from_p_id, :c_name_fio, :ke_code, :ke_oper_days
  do
  begin
    suspend;
  end

  -- виртуальный сотрудник
  db_id = f_get_db_id();
  if (db_id in (39801, 41701)) then
  begin
    ke_from_p_id = null;
    for select c_id, c_name_fio
        from clients
        where c_id in (1000000000005, 1000000000100)
        into :ke_from_p_id, :c_name_fio
    do
    begin
      if (ke_from_p_id = 1000000000100) then
        ke_code = cast(substring(ke_from_reg_id from 2 for 2) as integer) || '-' || cast(substring(ke_from_reg_id from 4 for 2) as integer) || '-0';
      else
        ke_code = '';
      ke_oper_days = 127;
      suspend;
    end
  end

  if (db_id = 81001) then
  begin
    ke_from_p_id = null;
    for select c_id, c_name_fio
        from clients
        where c_id in (1000000000000, 1000000000002, 1000000000004)
        into :ke_from_p_id, :c_name_fio
    do
    begin
      ke_code = '.ВС';
      ke_oper_days = 127;
      suspend;
    end
  end
end
