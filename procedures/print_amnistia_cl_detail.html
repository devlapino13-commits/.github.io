CREATE PROCEDURE PRINT_AMNISTIA_CL_DETAIL(
    AAREP_K_ID ID_TYPE)
RETURNS (
    KD_K_NEW ID_TYPE,
    C_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    KC_AMOUNT_APPROVED CURR_TYPE,
    K_AMOUNT_APPROVED CURR_TYPE,
    S_DATE DATE,
    S_NUMBER INTEGER,
    S_OS CURR_TYPE,
    S_PERC CURR_TYPE,
    P_OS CURR_TYPE,
    P_PERC CURR_TYPE,
    P_FINE CURR_TYPE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    K_KREDLINE_END DATE,
    IS_PAY_DIFF INTEGER)
AS
begin
  for select distinct kc_from_k_id, kc_from_c_id, kc_amount_approved
      from kontstate ks, knclient kc
      where ks_from_k_id starting left(:aarep_k_id, 17) and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0
      into kd_k_new, c_id, kc_amount_approved
  do
  begin
    for select k_kontract_no, k_amount_approved, s_date, s_number, s_os, s_perc, p_os, p_perc, p_fine, zad_os, zad_perc,
               k_kredline_end, is_pay_diff
        from print_amnistia_detail(:aarep_k_id)
        where kd_k_new = :kd_k_new
        into k_kontract_no, k_amount_approved, s_date, s_number, s_os, s_perc, p_os, p_perc, p_fine, zad_os, zad_perc,
             k_kredline_end, is_pay_diff
    do
    begin

      s_os = kc_amount_approved / k_amount_approved * s_os;
      s_perc = kc_amount_approved / k_amount_approved * s_perc;
      p_os = kc_amount_approved / k_amount_approved * p_os;
      p_perc = kc_amount_approved / k_amount_approved * p_perc;
      p_fine = kc_amount_approved / k_amount_approved * p_fine;
      zad_os = kc_amount_approved / k_amount_approved * zad_os;
      zad_perc = kc_amount_approved / k_amount_approved * zad_perc;

      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name_fio;

      suspend;
    end
  end
end
