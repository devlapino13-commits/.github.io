CREATE PROCEDURE AAREP_CLIENT_CALC_PERC(
    REPORT_REGION_ID ID_TYPE,
    REPORT_DATE_END DATE)
RETURNS (
    OFFICE_NAME D_STR_100,
    K_FACT_DISB_REG_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    KONT_ID D_STR_20,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    KC_AMOUNT_APPROVED D_CURRENCY,
    SH_PLAN_CLOSE_DATE D_DATE,
    K_KREDLINE_END D_DATE,
    PKS D_PERCENT,
    ZAD_OS_ALL_CL D_CURRENCY,
    SH_CALC_PERC_CL D_CURRENCY,
    KOL_PROSR_OS D_INTEGER,
    KOL_PROSR_PERC D_INTEGER,
    MAX_DELINQ_DAYS D_INTEGER,
    K_PERCENT D_PERCENT,
    MAX_OL_DATE D_DATE,
    C_PASS_NO D_STR_100,
    IS_AZREPAYKONT D_SMALLINT,
    LAST_OL_DATE_OS D_DATE,
    LAST_PAY_OS_CL D_CURRENCY,
    LAST_OL_DATE_PERC D_DATE,
    LAST_PAY_PERC_CL D_CURRENCY,
    K_KOMMIS_PERC D_PERCENT,
    K_BASE_K_ID D_STR_20,
    KC_NUMBER D_INTEGER,
    PR_NAME VARCHAR(50),
    C_PINCODE VARCHAR(20))
AS
declare variable k_id d_id;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable sh_date_os d_date;
declare variable sh_date_perc d_date;
declare variable zad_os d_currency;
declare variable zad_perc d_currency;
declare variable sh_calc_perc d_currency;
declare variable zad_os_all d_currency;
declare variable sh_calc_perc_all d_currency;
declare variable dolya d_percent;
declare variable db_id d_integer;
declare variable oks_region_id_tmp d_integer;
declare variable oks_region_id d_integer;
declare variable k_fact_disb_reg_id d_integer;
declare variable last_pay_os d_currency;
declare variable last_pay_perc d_currency;
declare variable pr_id d_integer;
begin
  :db_id = f_get_db_id();

  for select oks_region_id, oks_from_k_id
      from oksummary oks
      inner join region_list(:report_region_id, 1, -1, 1) on oks_region_id = id_list
      where :report_date_end between oks_date_begin and oks_date_end and
      oks_close_state = 0
      order by oks_region_id
      into oks_region_id, k_id
  do
  begin
    kont_id = cast(k_id as d_str_20);

    if (coalesce(oks_region_id_tmp, 0) <> oks_region_id) then
    begin
      oks_region_id_tmp = oks_region_id;
      select reg_name
      from region r
      where reg_unicode = :oks_region_id
      into office_name;
    end

    pks = null;
    k_kredline_end = null;
    k_percent = null;
    k_kommis_perc = null;
    select k.k_kontract_no, k_disb_date, k_kredline_end, k_percent_effect, k_amount_approved, k.k_percent_per_day * 365,
           k.k_kommis_perc, k.k_fact_disb_reg_id,
           cast(iif(k_base_k_id is null, k_id, k_base_k_id) as d_str_20) k_base_k_id, k.k_from_pr_id
    from kontract k
    where k_id = :k_id
    into k_kontract_no, k_disb_date, k_kredline_end, pks, k_amount_approved, k_percent, k_kommis_perc,
         k_fact_disb_reg_id, k_base_k_id, pr_id;

    k_fact_disb_reg_name = '';
    select reg_name
    from region
    where reg_unicode = :k_fact_disb_reg_id
    into k_fact_disb_reg_name;

    sh_plan_close_date = null;
    select max(sh_date)
    from schedule sh
    where sh.sh_from_k_id = :k_id
    into sh_plan_close_date;

    sh_id = null;
    sh_date = null;
    zad_os = 0;
    zad_perc = 0;
    sh_calc_perc = 0;
    zad_os_all = 0;
    sh_calc_perc_all = 0;
    sh_date_os = null;
    sh_date_perc = null;
    kol_prosr_os = 0;
    kol_prosr_perc = 0;
    for select sh_id, sh_date, zad_os, zad_perc, sh_calc_perc
        from aarep_make_zad_list(:k_id, :report_date_end, 1)
        where sh_date is not null
        into sh_id, sh_date, zad_os, zad_perc, sh_calc_perc
    do
    begin
      zad_os_all = coalesce(zad_os_all, 0) + zad_os;
      sh_calc_perc_all = coalesce(sh_calc_perc_all, 0) + sh_calc_perc;

      if (coalesce(sh_date_os, report_date_end) > sh_date and
          zad_os > 0) then
      begin
        kol_prosr_os = report_date_end - sh_date;

        sh_date_os = sh_date;
      end
      if (coalesce(sh_date_perc, report_date_end) > sh_date and
          sh_calc_perc > 0) then
      begin
        kol_prosr_perc = report_date_end - sh_date;

        sh_date_perc = sh_date;
      end
    end

    max_delinq_days = null;
    select max(iif(oks_date_end <= :report_date_end, oks_date_end + 1, :report_date_end) - oks_delinq_date) as max_del_days
    from oksummary
    where oks_from_k_id = :k_id and
          oks_close_state = 0 and
          oks_delinq_date <= :report_date_end and
          oks_date_begin <= :report_date_end
    into max_delinq_days;
    if (coalesce(max_delinq_days, 0) < 0) then
      max_delinq_days = 0;

    max_ol_date = null;
    last_ol_date_os = null;
    last_pay_os = null;
    last_ol_date_perc = null;
    last_pay_perc = null;
    with t1
    as (select oko.oko_ol_date, oko_os, oko_perc
        from operkroff oko
        where oko_from_k_id = :k_id and
              oko_type = 4 and
              oko.oko_ol_is_active = 1 and
              oko.oko_ol_date <= :report_date_end
        union
        select ok_ol_date, sum(ok_os), sum(ok_perc)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_ol_date <= :report_date_end
        group by 1),
    t2
    as (select max(oko_ol_date) max_ol_date, max(iif(oko_os <> 0, oko_ol_date, null)) last_ol_date_os,
               max(iif(oko_perc <> 0, oko_ol_date, null)) last_ol_date_perc
        from t1)
    select max_ol_date, last_ol_date_os, sum(iif(last_ol_date_os = oko_ol_date, oko_os, 0)) last_pay_os,
           last_ol_date_perc, sum(iif(last_ol_date_perc = oko_ol_date, oko_perc, 0)) last_pay_perc
    from t1, t2
    group by max_ol_date, last_ol_date_os, last_ol_date_perc
    into max_ol_date, last_ol_date_os, last_pay_os, last_ol_date_perc, last_pay_perc;

    dolya = null;
    zad_os_all_cl = null;
    sh_calc_perc_cl = null;
    for select kc_from_c_id, kc_amount_approved, kc_number
        from knclient kc
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, --ID клиента
             kc_amount_approved,
             kc_number
    do
    begin
      dolya = coalesce(kc_amount_approved, 1) / coalesce(k_amount_approved, 1);
      zad_os_all_cl = coalesce(zad_os_all, 0) * dolya;
      sh_calc_perc_cl = coalesce(sh_calc_perc_all, 0) * dolya;

      /*
          MRAKHMANOV
        № паспорта
        Номинальная % ставка
        Последнее погашение (как по внебалансу)
      */
      select c_name_fio, c.c_pass_no, c.c_pincode
      from clients c
      where c_id = :c_id
      into c_name_fio, c_pass_no, c_pincode; --ФИО клиента

      if (db_id = 94401) then
      begin
        if (exists(select *
                   from azrepaykont ar
                   where ar.ark_from_k_id = :k_id)) then
          is_azrepaykont = 1;
        else
          is_azrepaykont = 0;
      end

      if (zad_os_all_cl >= 0) then
      begin
        last_pay_os_cl = -coalesce(last_pay_os, 0) * dolya;
        last_pay_perc_cl = -coalesce(last_pay_perc, 0) * dolya;

        pr_name = null;

        select pr.pr_name
        from product pr
        where pr.pr_id = :pr_id
        rows 1
        into :pr_name;

        suspend;
      end
    end
  end
end
