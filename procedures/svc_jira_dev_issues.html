CREATE PROCEDURE SVC_JIRA_DEV_ISSUES(
    DEV_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    JRI_ISSUE_ID D_INTEGER,
    JRI_JIRA_CODE D_STR_CODE,
    JRI_TITLE D_STR_LONG,
    JRI_LEVEL D_SMALLINT,
    JRI_ASSIGNED D_DATETIME,
    JRI_FINISHED D_DATETIME,
    JRI_STATE_DATETIME D_DATETIME,
    JRD_LOGIN D_STR_CODE,
    JRD_FIO D_STR_100,
    JRT_TEXT D_STR_CODE,
    TASK_LAST_STATE D_STR_CODE,
    TASK_STACK D_STR_CODE)
AS
begin
  for select jri_issue_id,
             jri_jira_code,
             jri_title,
             jri_level,
             jri_created,
             jri_finished,
             jri_state_datetime,
             jd.jrd_login,
             jd.jrd_fio,
             jt.jrt_text,
             js.jrs_text
      from jira_issues ji
      left join jira_dev jd on jd.jrd_id = ji.jri_dev_id
      left join jira_types jt on jt.jrt_id = ji.jri_type_id
      left join jira_states js on js.jrs_id = ji.jri_state_id
      where ((jri_created between :rep_date_begin and :rep_date_end)
         or (ji.jri_finished between :rep_date_begin and :rep_date_end))
         and jt.jrt_id in (1, 2, 5, 7, 8, 9)
         and js.jrs_result >= 0
         and jri_dev_id = :dev_id
      into jri_issue_id,
           jri_jira_code,
           jri_title,
           jri_level,
           jri_assigned,
           jri_finished,
           jri_state_datetime,
           jrd_login,
           jrd_fio,
           jrt_text,
           task_last_state
  do
  begin
    select task_stack, task_level
    from jira_update_params(:jrd_fio, :jrt_text, :jri_level, :jri_issue_id)
    into :task_stack, :jri_level;

    suspend;
  end
end
