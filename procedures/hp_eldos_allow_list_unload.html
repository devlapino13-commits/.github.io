CREATE PROCEDURE HP_ELDOS_ALLOW_LIST_UNLOAD(
    K_ID D_ID)
RETURNS (
    BLS_ID D_ID,
    IS_BLN D_BOOL,
    FD_NAME D_STR_100,
    FL_NAME D_STR_100,
    JS_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable kl_date d_date;
declare variable cds_id d_id;
declare variable c_name d_str_100;
declare variable ma_id d_id;
declare variable c_kind d_smallint;
declare variable edb_id d_id;
declare variable pdt_code d_str_code;
declare variable kc_id d_id;
declare variable pr_code d_str_100;
declare variable db_id d_integer;
declare variable blank_first_date d_date;
declare variable exa_id d_integer;
declare variable zap_id d_integer;
declare variable status d_integer;
declare variable vwp_type d_bool;
declare variable kc_status d_smallint;
declare variable mp_id d_id;
declare variable by_rating d_smallint;
declare variable edc_code d_str_code;
declare variable edl_status d_smallint;
declare variable k_disb_date d_date;
declare variable k_disb_kind d_integer;
declare variable use_cardst boolean;
declare variable bls_id1 d_id;
declare variable bls_id2 d_id;
declare variable fl_name1 d_str_100;
declare variable num_cli d_integer;
declare variable is_pdf boolean;
begin
  db_id = f_get_db_id();

  select max(kl.kl_datetime)
  from kontlog kl
  where kl.kl_from_k_id = :k_id and
        kl.kl_state = 1
  into :kl_date;

  select k_from_ma_app_id, pr_code, k_by_rating, k_disb_date, k_disb_kind, k_kolich_client_approved
  from kontract kont
  inner join product pr on pr_id = k_from_pr_id
  where k_id = :k_id
  into ma_id, pr_code, by_rating, k_disb_date, k_disb_kind, num_cli;
  use_cardst = (db_id in (39801, 41701)) and (k_disb_kind = 6) and (k_disb_date >= f_get_const('card_change_date'));

  --Предложение по займу для старых экспрессов
  if (db_id in (39801, 41701) and
      pr_code containing '.express_online.' and
      pr_code not containing '.sugarmill.' and
      pr_code not containing '.installment_goods.') then
  begin
    select exa.exa_id
    from expressapp exa
    where exa.exa_id > 0 and
          exa.exa_from_k_id = :k_id
    into exa_id;

    for select edb_from_bls_id, replace(edf_print_name, '/', '~') || '.pdf'
    from edoc_list, edoc_cfg, edoc_blank, edoc_frx
    where edl_object_kind = f_objkind('expressapp', 'id') and
          edl_object_id = :exa_id and
          edl_status = 1 and
          edl_from_edc_id = edc_id and
          edc_code = 'loan.express' and
          edb_from_edl_id = edl_id and
          edb_from_edf_id = edf_id and
          edb_from_bls_id > 0
    order by edb_from_c_id
    into :bls_id, :fl_name
    do
    begin
      fd_name = '';
      is_bln = 0;
      suspend;
    end
  end

  if (db_id in (39801, 41701) and pr_code containing '.installment_goods.') then
  begin
    select mp.mp_id
    from expressapp exa
    inner join mc_purchase mp on mp.mp_from_exa_id = exa.exa_id
    where exa.exa_id > 0 and
          exa.exa_from_k_id = :k_id and
          exa.exa_from_exp_id = -2  --Пакет по товарам в рассрочку
    into mp_id;

    for select edb_from_bls_id, replace(edf_print_name, '/', '~') || '.pdf'
    from edoc_list, edoc_cfg, edoc_blank, edoc_frx
    where edl_object_kind = f_objkind('mc_purchase', 'id') and
          edl_object_id = :mp_id and
          edl_status = 1 and
          edl_from_edc_id = edc_id and
          edc_code = 'mc_purchase.agreement_autosign' and
          edb_from_edl_id = edl_id and
          edb_from_edf_id = edf_id and
          edb_from_bls_id > 0
    order by edb_from_c_id
    into :bls_id, :fl_name
    do
    begin
      fd_name = '';
      is_bln = 0;
      suspend;
    end
  end

  -- Заявка
  for select edb_from_bls_id, replace(edf_print_name, '/', '~') || '.pdf'
  from edoc_list, edoc_cfg, edoc_blank, edoc_frx
  where edl_object_kind = 11
    and edl_object_id = :ma_id
    and edl_status = 1
    and edl_from_edc_id = edc_id
    and edc_code in ('loan.application', 'loan.application_autosign', 'loan.spouses_consent')
    and edb_from_edl_id = edl_id
    and edb_from_edf_id = edf_id
    and edb_from_bls_id > 0
  order by edb_from_c_id
  into :bls_id, :fl_name
  do
  begin
    fd_name = '';
    is_bln = 0;
    suspend;
  end

  -- Договор
  for select edb_from_bls_id, replace(edf_print_name, '/', '~') || '.pdf'
  from edoc_list, edoc_cfg, edoc_blank, edoc_frx
  where edl_object_kind = 9
    and edl_object_id = :k_id
    and edl_status = 1
    and edl_from_edc_id = edc_id
    and edc_code in ('loan.agreement', 'loan.agreement_autosign')
    and edb_from_edl_id = edl_id
    and edb_from_edf_id = edf_id
    and edb_from_bls_id > 0
    into :bls_id, :fl_name
  do
  begin
    fd_name = '';
    is_bln = 0;
    suspend;
  end

   -- RFCKZ-122
  -- Пакет с согласиями в МП
  -- 1001 - Согласие на сбор и обработку перс. данных
  -- 1002 - Согласие для ГКБ и ПКБ на получение перс. данных от АО ЦРТР
  -- 1003 - Согласие субъекта кредитной истории на выдачу кред. отчета получателю из КБ
  -- 1004 - Согласие субъекта кредитной истории на предоставление инфо о нем в КБ
  if (db_id = 39801 and
      exists(select * from edoc_list edl
             where edl.edl_object_kind = 11 and
                   edl.edl_object_id = :ma_id and
                   edl.edl_status = 1)) then
  begin
    for select edb_from_bls_id, replace(edf_print_name, '/', '~') || '.pdf'
    from edoc_list edl
    inner join edoc_blank edb on edb.edb_from_edl_id = edl.edl_id
    inner join edoc_frx edf on edf.edf_id = edb.edb_from_edf_id
    inner join blankname bln on bln.bln_id = edb.edb_from_bln_id
    inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
    where edl.edl_object_kind = 11
      and edl.edl_object_id = :ma_id
      and edl.edl_status = 1
      and edb.edb_from_bln_id in (1001, 1002, 1003, 1004)
      and edb.edb_from_bls_id > 0
    into :bls_id, :fl_name
    do
    begin
      fd_name = '';
      is_bln = 0;
     suspend;
    end
  end


  -- Бланки для контракта.
  -- Добавляются в его вкладки
  -- 1635 Решение по заявке Классика
  -- 1636 Решение по заявке ИПФ
  -- 1730 Гарантийное письмо для сах. завода
  -- 1766 Решение по заявке Классика (выдачи по кредитному рейтингу)
  -- 1767 Решение по заявке ИПФ (выдачи по кредитному рейтингу)
  for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)

  inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
  where bln.bln_id in (1635, 1636, 1730, 1766, 1767)
  group by 1,2
  into :bls_id, :fl_name, :blank_first_date
  do
  begin
    if (blank_first_date > kl_date) then continue;
    if ((bls_id in (1635, 1766)) and (pr_code containing '.ipf.')) then continue;
    if ((bls_id in (1636, 1767)) and (pr_code not containing '.ipf.')) then continue;
    if (db_id in (39801, 41701)) then
    begin
      if (pr_code containing '.express_online.' and pr_code not containing '.sugarmill.' and pr_code not containing '.installment_goods.') then
        continue;

      if (pr_code not containing '.sugarmill.' and bls_id = 1730) then
        continue;

      if (coalesce(by_rating, 0) = 0 and bls_id in (1766, 1767)) then
        continue;

      if (coalesce(by_rating, 0) = 1 and bls_id in (1635, 1636)) then
        continue;
    end 
    is_bln = 1;
    js_text = f_format('{"obj_kind":9,"obj_id":%1}', k_id);
    suspend;
  end

  if (db_id in (39801, 41701)) then
  begin
   for select edb_from_bls_id, replace(edf_print_name, '/', '~') || '.pdf'
        from edoc_list edl
        inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id and
             edc.edc_code = 'loan.specification'
        inner join kontract_istinaa_prepay kip on kip.kip_id = edl.edl_object_id and
                                                  kip.kip_from_k_id = :k_id
        inner join edoc_frx edf on edf.edf_from_edc_id = edc.edc_id
        inner join edoc_blank edb on edb.edb_from_edl_id = edl.edl_id and edb.edb_from_bls_id > 0
        where edl.edl_object_kind = f_objkind('kontract_istinaa_prepay', 'id') and
              edl.edl_status = 1
        order by edl_create_datetime desc
        rows 1
      into :bls_id, :fl_name
    do
    begin
      fd_name = '';
      is_bln = 0;
      suspend;
    end

    for select edb_from_bls_id, replace(edf_print_name, '/', '~') || '.pdf', edc_code, edl_status
        from edoc_list edl
        inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id and
             edc.edc_code in ('act_from_builder.agreement', 'act_from_company.agreement',
                              'idjara_goods.client_notif', 'idjara_goods.contragent_notif')
        inner join kontmatag kma on kma.kma_id = edl.edl_object_id and
                                    kma.kma_from_k_id = :k_id
        inner join edoc_frx edf on edf.edf_from_edc_id = edc.edc_id
        inner join edoc_blank edb on edb.edb_from_edl_id = edl.edl_id and edb.edb_from_bls_id > 0
        where edl.edl_object_kind = f_objkind('kontmatag', 'id')
        order by edl_create_datetime desc
      into
        :bls_id,
        :fl_name,
        :edc_code,
        :edl_status
    do
    begin
      fd_name = '';
      is_bln = 0;
      suspend;
    end
  end

  for with tab as (
    select 0 as c_kind, c_id, c_name_fio, c_blob_photo_id, kc_id, kc_status
    from knclient, clients
    where kc_from_k_id = :k_id
      and kc_from_c_id = c_id
      and kc_amount_approved > 0
    union select mac_is_poruch, c_id, c_name_fio, c_blob_photo_id, 0, 0
    from miniappclient, clients
    where mac_from_ma_id = :ma_id
      and c_id = mac_from_c_id
      and mac_is_poruch in (-1, 1)
    )
  select c_id, c_kind, c_name_fio, c_blob_photo_id, kc_id, kc_status
  from tab
  order by abs(c_kind) -- клиент будет первым
  into :c_id, :c_kind, :fd_name, :bls_id, :kc_id, :kc_status
  do
  begin
    -- Заголовок узла (корня) дерева
    fd_name = fd_name || ' ' ||
      trim(case c_kind
        when -1 then 'Лицо, дающее согласие'
        when 1 then 'Поручитель'
        else 'Клиент'
      end);
    fl_name = 'Фото.jpeg';
    is_bln = 0;
    suspend;

    -- Тестовые бланки для клиента. Добавляются в его вкладки
    if (db_id in (39801, 41701) and kc_id > 0) then
    begin
      -- 1637 - Анкета анализ клиента
      -- 1638 - Целевой мониторинг
      -- 1639 - Анкета по просрочке


        for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)
        from blankname bln
        inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
        where bln.bln_id in (1637,1638,1639)
        group by 1,2
        into :bls_id, :fl_name, :blank_first_date
        do
        begin
          if (blank_first_date > kl_date) then continue;
          if (bls_id = 1638) then
          begin
              if (not exists(select * from print_target_monitoring(:kc_id) where has_anketa = 1) or use_cardst) then continue;
          end
          else if (bls_id = 1639) then
          begin
              if (not exists(select * from print_delinq_monitoring(:kc_id) where has_anketa = 1)) then continue;
          end

          if (bls_id = 1637 and coalesce(by_rating, 0) = 1) then
            continue;

          is_bln = 1;
          js_text = f_format('{"obj_kind":8,"obj_id":%1}', kc_id);
          suspend;
        end
    end

    -- отчет КИБ
    is_bln = 1;
    fl_name = 'Отчет КИБ.pdf';
    bls_id = 41;
    js_text = f_format('{"add_params":{"ma_id":%1,"c_id":%2,"region_id":0}}', ma_id, c_id);
    suspend;

    --Анализ платёжного мониторинга
    if (coalesce(by_rating, 0) = 1) then
    begin
      is_bln = 1;
      fl_name = 'Анализ платежного мониторинга.pdf';
      bls_id = 84;
      js_text = f_format('{"obj_kind":9,"obj_id":%1}', k_id);
      suspend;
    end

    -- Договор ЭЦП
    for
      select
        edb_from_bls_id,
        replace(edf_print_name, '/', '~') || '.pdf'
      from edoc_list, edoc_cfg, edoc_blank, edoc_frx
      where edl_object_kind = 0
        and edl_object_id = :c_id
        and edl_status = 1
        and edl_from_edc_id = edc_id
        and edc_code = 'asp.agreement'
        and edb_from_edl_id = edl_id
        and edb_from_edf_id = edf_id
        and edb_from_bls_id > 0
        and cast(edl_create_datetime as date) <= :kl_date
      order by edl_create_datetime desc
      rows 1
    into :bls_id, :fl_name
    do
    begin
      is_bln = 0;
      suspend;
    end

    -- Документы клиента
    for select cds.cds_id, pdt.pdt_code, c_name_fio, cds_doc_blob_1, cds_doc_blob_2, replace(pk_name, '/', '~')
        from clientdocs cds
        inner join clients on c_id = cds_from_c_id
        inner join passkind pk on cds.cds_from_pk_id = pk.pk_id
        inner join passdoctype pdt on pk.pk_from_pdt_id = pdt.pdt_id
        where cds.cds_from_c_id = :c_id and
            cds.cds_is_delete = 0 and
            (cds.cds_doc_date_end >= current_date or
            cds.cds_doc_date_end is null) and
            cds.cds_datetime = (select max(cds.cds_datetime)
                                from clientdocs cds
                                where cds.cds_from_c_id = :c_id and
                                        cds.cds_is_delete = 0 and
                                        cds.cds_from_pk_id = pk.pk_id)
    order by pdt_is_main_doc desc, pdt_name
    into :cds_id, :pdt_code, :c_name, :bls_id1, :bls_id2, :fl_name1

    do
    begin
      if (c_kind = -1 and pdt_code = 'place_of_business' and db_id in (39801, 41701)) then --Не выводить фото с места бизнеса для ЛДС
        continue;

      if (pr_code containing '.express_online.' and (pdt_code <> 'passport_card') and db_id in (39801, 41701)) then continue;

      if (bls_id1 > 0) then
      begin
        bls_id = bls_id1;
        is_pdf = (select bls_filename containing '.pdf' from blobstore where bls_id = :bls_id);
        fl_name = fl_name1 || iif(:num_cli > 1, f_format(' (%1)', :c_name), '') || ' Стр1.';

        if (pdt_code = 'insurance_certificate' and db_id = 41701) then -- RFCKG-4411
          fl_name = fl_name1 || iif(:num_cli > 1, f_format(' (%1)', :c_name), '');

        fl_name = fl_name || iif(is_pdf, 'pdf', 'jpeg');
        is_bln = 0;
        suspend;
      end
      if (bls_id2 > 0) then
      begin
        bls_id = bls_id2;
        is_pdf = (select bls_filename containing '.pdf' from blobstore where bls_id = :bls_id);
        fl_name = fl_name1 || iif(:num_cli > 1, f_format(' (%1)', :c_name), '') || ' Стр2.' || iif(is_pdf, 'pdf', 'jpeg');
        is_bln = 0;
        suspend;
      end
    end

     -- Тестовые бланки для клиента. Добавляются в его вкладки
    if (db_id in (39801, 41701)) then
    begin
      -- 1637 - Анкета анализ клиента
      -- 1638 - Целевой мониторинг
      -- 1639 - Анкета по просрочке


        for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)
        from blankname bln
        inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
        where bln.bln_id in (1769,1770,1771)
        group by 1,2
        into :bls_id, :fl_name, :blank_first_date
        do
        begin
          is_bln = 1;
          js_text = f_format('{"add_params":{"k_id":%1,"c_id":%2}}', k_id, c_id);
          suspend;
        end

        if (exists(select * from extinfolist eil
        where eil.eil_from_eic_id = 10 and
              eil.eil_from_c_id = :c_id and
              eil.eil_is_json = 1 and
              eil.eil_status = 2 and
              eil.eil_from_bls_id > 0)) then    -- Cправка с соц фонда
        begin
           for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)
           from blankname bln
           inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
           where bln.bln_id = 1812
           group by 1,2
           into :bls_id, :fl_name, :blank_first_date
           do
           begin
             is_bln = 1;
             js_text = f_format('{"obj_kind":0,"obj_id":%1}', c_id);
             suspend;
           end
        end
     if (db_id = 39801) then
     begin
      if (exists(select * from extinfolist eil
        where eil.eil_from_eic_id = 9 and
              eil.eil_from_c_id = :c_id and
              eil.eil_is_json = 1 and
              eil.eil_status = 2 and
              eil.eil_from_bls_id > 0)) then
        begin
           for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)
           from blankname bln
           inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
           where bln.bln_id = 1010
           group by 1,2
           into :bls_id, :fl_name, :blank_first_date
           do
           begin
             is_bln = 1;
             js_text = f_format('{"obj_kind":0,"obj_id":%1}', c_id);
             suspend;
            end
        end
        if (exists(select * from extinfolist eil
        where eil.eil_from_eic_id = 10 and
              eil.eil_from_c_id = :c_id and
              eil.eil_is_json = 1 and
              eil.eil_status = 2 and
              eil.eil_from_bls_id > 0)) then
        begin
           for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)
           from blankname bln
           inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
           where bln.bln_id = 1011
           group by 1,2
           into :bls_id, :fl_name, :blank_first_date
           do
           begin
             is_bln = 1;
             js_text = f_format('{"obj_kind":0,"obj_id":%1}', c_id);
             suspend;
            end
        end
        if (exists(select * from extinfolist eil
        where eil.eil_from_eic_id = 12 and
              eil.eil_from_c_id = :c_id and
              eil.eil_is_json = 1 and
              eil.eil_status = 2 and
              eil.eil_from_bls_id > 0)) then    -- Cправка АСП 
        begin
           for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)
           from blankname bln
           inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
           where bln.bln_id = 1012
           group by 1,2
           into :bls_id, :fl_name, :blank_first_date
           do
           begin
             is_bln = 1;
             js_text = f_format('{"obj_kind":0,"obj_id":%1}', c_id);
             suspend;
            end
        end
    end

    -- RFCKZ-280
    if (db_id = 39801) then
    begin
       for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)
        from blankname bln
        join blankblob blb on bln.bln_id = blb.blb_from_bln_id
        where bln.bln_id = 1014
        group by 1,2
        into :bls_id, :fl_name, :blank_first_date
        do
        begin
          is_bln = 1;
          js_text = f_format('{"add_params":{"k_id":%1,"c_id":%2}}', k_id, c_id);
          suspend;
        end
        for select bln_id, replace(bln_name, '/', '~') || '.pdf', min(blb.blb_date_begin)
            from blankname bln
            inner join blankblob blb on bln.bln_id = blb.blb_from_bln_id
            inner join extinfolist eil on eil.eil_object_id = :ma_id 
                  and eil.eil_object_kind = f_objkind('miniapp', 'id')
                  and eil.eil_status = 2 and eil.eil_from_bls_id > 0
            inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id
            where 
                (eic.eic_code = 'afm_iin_check' and bln.bln_id = 1017) or
                (eic.eic_code = 'afm_name_check' and bln.bln_id = 1018)
            group by 1, 2
            into :bls_id, :fl_name, :blank_first_date
        do
        begin
          is_bln = 1;
          js_text = f_format('{"add_params":{"k_id":%1,"c_id":%2}}', k_id, c_id);
          suspend;
        end
    end
    end

    -- Фото на момент выдачи (webcam)
    if (db_id in (39801, 41701) and kc_status = 2) then
    begin
      bls_id = null;
      select
        vwp_type,
        vwp_wcam_bls_id,
        vwp_status
      from verify_webcam_photo
      where vwp_k_id = :k_id
        and (vwp_type = 0 or vwp_card_id > 0)
      order by vwp_id desc
      rows 1
      into
        :vwp_type,
        :bls_id,
        :status;
      if (bls_id > 0 and status = 1) then
      begin
        fl_name = 'Фото на момент выдачи.jpeg';
        is_bln = 0;
        suspend;
      end
    end
  end

  -- RFCKZ-775
  if (db_id = 39801) then
  for
    select
      edb_from_bls_id,
      replace(edf_print_name, '/', '~') || '.pdf'
    from edoc_list, edoc_blank, edoc_frx
    where edl_object_kind = 9
      and edl_object_id = :k_id
      and edl_status = 1
      and edl_from_edc_id = 10
      and edb_from_edl_id = edl_id
      and edb_from_edf_id = edf_id
      and edb_from_bls_id > 0
    order by edb_from_c_id
  into
    :bls_id,
    :fl_name
  do
  begin
    fd_name = '';
    is_bln = 0;
    suspend;
  end

  if (use_cardst) then
  begin
    is_bln = 1;
    fl_name = 'Выписка по карте';
    js_text = f_format('{"obj_id":%1}', -k_id);
    suspend;
  end
end
