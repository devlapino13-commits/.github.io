CREATE PROCEDURE HP_GET_EXTSYS_PAY_LOAN_DATA(
    PROJECT D_STR_100,
    C_ID ID_TYPE,
    COMMAND STR_100,
    KONTRACT_NO STR_100,
    IS_JSON SMALLINT,
    CL_INN D_STR_20 = null)
RETURNS (
    RESULT_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable K_ID ID_TYPE;
declare variable K_NO STR_20;
declare variable K_DATE D_DATE;
declare variable C_NAME_FIO STR_100;
declare variable SUM_REPAY CURR_TYPE;
declare variable SUM_NAME D_STR_255;
declare variable CUR_NAME STR_20;
declare variable NUM_COUNT D_INTEGER;
declare variable CRLF varchar(2);
declare variable REPEAT_STR STR_255;
declare variable DB_ID D_INTEGER;
declare variable NEXT_PAYMENT_DATE STR_100;
declare variable SH_DATE D_DATE;
declare variable SUM_TMP CURR_TYPE;
declare variable ROWS_QTY D_INTEGER;
declare variable C_BIRTH_DATE_STR D_STR_100;
declare variable C_BIRTH_DATE_CONVERTED D_STR_100;
declare variable TEXT D_STR_255;
declare variable FL_ID D_ID;
declare variable RIGHT_K_ID D_ID;
declare variable SUM_REPAY_TOTAL CURR_TYPE;
declare variable K_CURR_OTHER CURR_TYPE;
begin
  if (kontract_no = '') then
    kontract_no = null;

  rows_qty = 1;
  result_txt = '';
  text = '; рекомендуемая сумма платежа';

  if (command = 'loan_list') then
  begin
    if (is_json = 0) then
      result_txt = BLOB_APPEND(result_txt, '  <loan_list>' || crlf);
    else
      result_txt = BLOB_APPEND(result_txt, ', "loan_list": [');

    rows_qty = 10;
  end

  db_id = f_get_db_id();

  cur_name = (select first 1 cm_currency_name
              from company);

  crlf = ascii_char(13) || ascii_char(10);
  num_count = 0;
  /*
  if (db_id = 81001) then
  begin
    right_k_id = 0;
    select k_id
    from hp_get_contract_for_pay(:c_id)
    into right_k_id; -- "правильный" контракт для оплаты (RFCRU-2129)
    right_k_id = coalesce(right_k_id, 0);
  end       */

  for select k_id, k_kontract_no, k_fact_disb_date, c_name_fio, f_datetostr(cl.c_birth_date), c_from_fl_id
      from oksummary
      inner join knclient on kc_from_k_id = oks_from_k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0
      inner join clients cl on c_id = kc_from_c_id
      inner join kontract kont on k_id = oks_from_k_id
      where oks_date_end >= '31.12.2999' and
            iif(:db_id in (39801, 41701), oks_close_state in (0, 1), oks_close_state = 0) and
            iif(:kontract_no is null, true, kont.k_kontract_no = :kontract_no)
      order by oks_delinq_date, oks_balance
      rows :rows_qty
      into :k_id, :k_no, :k_date, :c_name_fio, :c_birth_date_str, :fl_id
  do
  begin
  /*  if (db_id = 81001 and right_k_id > 0 and right_k_id <> k_id) then
      continue; -- если есть "правильный" контракт, то другие контракты не показываем   */
/*
    if (db_id = 41701 and exists(select oko_from_k_id
                                 from operkroff
                                 where oko_from_k_id = :k_id
                                 group by oko_from_k_id
                                 having sum(oko_os + oko_perc + oko_other + oko_sh_os + oko_sh_perc + oko_sh_other) > 0)) then
      continue; -- погашения по внесистемному кредиту принимаются только через кассу МБулак (RFCKG: 4105, 4269)
*/
    num_count = num_count + 1;
    c_birth_date_converted = substring(c_birth_date_str from 7) || substring(c_birth_date_str from 4 for 2) || substring(c_birth_date_str from 1 for 2);

    if (command in ('check', 'loan_list')) then
    begin

      if (db_id in (39801, 41701)) then
      begin
        if (command = 'check') then
          text =  ' по №:'|| :k_no || '; рекомендуемая сумма платежа';

        if (command = 'loan_list' and k_date = current_date) then
          continue;
      end
      sum_repay_total = 0;

      sum_repay = (select recom_pay_sum
                   from get_recommended_pay_sum(:k_id, current_date));

      if (db_id = 39801 and project = 'kaspi') then
      begin
        sum_repay_total = (select sum(zad_os + sh_calc_perc + sh_calc_fine)
                           from aarep_make_zad_list(:k_id, current_date, 1));

        k_curr_other = (select coalesce(sum(res_amount_saldo), 0)
                        from hp_repay_extra_saldo(:k_id, 0, current_date));

        if (k_curr_other > 0) then -- Для Каспи в рекомендуемую сумму платежа необходимо включать начисленные суммы госпошлины и расходов по исполнительной надписи
        begin
          sum_repay = sum_repay + k_curr_other;
          sum_repay_total = sum_repay_total + k_curr_other;
        end 
      end

      if (db_id = 39801 and project = 'kaspi' and sum_repay = 0 and sum_repay_total > 0) then
      begin
        sum_repay = 10; -- Для Каспий устанавливаем минимальную рекомендуемую сумму = 10 тг согласно сообщению от Маржан в общей группе с Каспий (TP_MBulak_AGWS). Это вынужденная мера, связанная с тем, что при передаче значения 0 механизм Каспий ломается и отображает пустой экран.
        if (sum_repay_total < sum_repay) then
            sum_repay = sum_repay_total;
      end
      sum_name = iif(db_id = 94401, '; məsləhət olunan ödəniş məbləği', iif(db_id = 81001, '; рекомендуемая сумма платежа', :text)) || iif(project in ('doscredo', 'doscredo_test', 'doscredoacq', 'doscredoacq_test', 'PayBoxAcq', 'PayBoxAcq_test', 'IPC', 'IPC_text', 'IPCacq', 'PaySoft'), ' ', ':') || round(sum_repay, 2) || ' ' || cur_name; -- igusev 19.11.2019 убрал двоеточение, т.к. из-за него у Доскредо проблема с автоматизацией
    end
    else
    if (command = 'check_auto') then
    begin
      sum_repay = 0;
      next_payment_date = '';

      for select sh_date,
                 round(sum(zad_os + sh_calc_perc + sh_calc_fine + zad_kommis + iif(:db_id in (39801, 41701), round(zad_perc * 0.02), 0)), 2)
          from aarep_make_zad_list(:k_id, current_date, 1)
          where sh_date is not null
          group by sh_date
          order by sh_date
          into :sh_date, :sum_tmp
      do
      begin
        if (sh_date <= current_date) then
          sum_repay = sum_repay + sum_tmp;
        else
        begin
          if (next_payment_date = '') then
            next_payment_date = f_datetostr(:sh_date);
        end
      end
      sum_name = '; сумма к оплате: ' || round(sum_repay, 2) || ' ' || cur_name;
    end

    if (command in ('check', 'check_auto')) then
    begin
      select short_fio
      from hp_fio_clear(:c_name_fio)
      into :c_name_fio;

      if (f_get_db_id() in (39801, 41701) and coalesce(:cl_inn, '') <> '') then
      begin
        c_name_fio = (select left(fl.fl_name1, 1) || '.' || left(fl.fl_name2, 1) || '.' || iif(coalesce(fl.fl_name3, '') = '', trim(''), trim(left(fl.fl_name3, 1) || '.'))
                        from fiolist fl
                        where fl.fl_id = :fl_id);
      end 

      if (is_json = 0) then
      begin
        if (project <> 'qiwi') then
          result_txt = BLOB_APPEND(result_txt, '  <fields name="info">' || c_name_fio || sum_name || '</fields>' || crlf);
        else
        begin
          result_txt = BLOB_APPEND(result_txt, '    <fields>' || crlf);
          result_txt = BLOB_APPEND(result_txt, '    <field1 name="info" type="disp">' || c_name_fio || sum_name || '</field1>' || crlf);
          result_txt = BLOB_APPEND(result_txt, '    <field2 name="cl_date" type="disp">' || c_birth_date_converted || '</field2>' || crlf);
          result_txt = BLOB_APPEND(result_txt, '    </fields>' || crlf);
        end
      end
      else
      begin
        result_txt = BLOB_APPEND(result_txt, ', ' || f_json_make_pair('name', :c_name_fio || sum_name, 'str'));

        if ((command = 'check') and
            (project in ('doscredoacq', 'doscredoacq_test', 'PayBoxAcq', 'PayBoxAcq_test', 'IPC', 'IPC_text', 'IPCacq', 'PaySoft'))) then
          result_txt = BLOB_APPEND(result_txt, ', ' || f_json_make_pair('c_id', :c_id, 'str'));

        if (command = 'check' and project = 'kaspi') then
        begin
          result_txt = BLOB_APPEND(result_txt, ', ' || f_json_make_pair('sum', :sum_repay, 'str'));
          result_txt = BLOB_APPEND(result_txt, ', ' || f_json_make_pair('sum_total', :sum_repay_total, 'str'));
        end
        if (command = 'check_auto') then
        begin
          result_txt = BLOB_APPEND(result_txt, ', ' || f_json_make_pair('balance', round(:sum_repay, 2), 'str'));
          result_txt = BLOB_APPEND(result_txt, ', ' || f_json_make_pair('next_pay_date', :next_payment_date, 'str'));
        end
      end
    end
    if (command = 'loan_list') then
    begin
      repeat_str = iif(num_count = 1, '', ', ') || '{ "loan" : {' || f_json_make_pair('loan_number', :k_no, 'str') || ', ' || f_json_make_pair('loan_data', 'договор № ' || k_no || ' от ' || f_datetostr(k_date) || sum_name, 'str') || iif(:project = 'kaspi', ', ' || f_json_make_pair('loan_sum', :sum_repay, 'str') || ', ' || f_json_make_pair('sum_total', :sum_repay_total, 'str'), '') || ' } }';
      result_txt = BLOB_APPEND(result_txt, repeat_str);
    end
  end
  if (command = 'loan_list') then
  begin
    result_txt = BLOB_APPEND(result_txt, ']');
  end

  suspend;
end
