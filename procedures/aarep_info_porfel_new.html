CREATE PROCEDURE AAREP_INFO_PORFEL_NEW(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    PORTFEL_END CURR_TYPE,
    DELINQ_BALANCE_END CURR_TYPE,
    DELINQ_BALANE7_END CURR_TYPE,
    DELINQ_BALANE30_END CURR_TYPE,
    PORTFEL_LAST CURR_TYPE,
    DELINQ_BALANCE_LAST CURR_TYPE,
    DELINQ_BALANE7_LAST CURR_TYPE,
    DELINQ_BALANE30_LAST CURR_TYPE,
    PORTFEL_BEGIN CURR_TYPE,
    DELINQ_BALANCE_BEGIN CURR_TYPE,
    DELINQ_BALANE7_BEGIN CURR_TYPE,
    DELINQ_BALANE30_BEGIN CURR_TYPE,
    KOL_CLI_END INTEGER,
    KOL_CLI_LAST INTEGER,
    KOL_CLI_BEGIN INTEGER,
    KOL_KRED_END INTEGER,
    KOL_KRED_LAST INTEGER,
    KOL_KRED_BEGIN INTEGER,
    KOL_PARAL_END INTEGER,
    KOL_PARAL_LAST INTEGER,
    KOL_PARAL_BEGIN INTEGER,
    REST_BALANCE_END CURR_TYPE,
    REST_BALANCE_LAST CURR_TYPE,
    REST_BALANCE_BEGIN CURR_TYPE,
    FLAT_SUM_END CURR_TYPE,
    FLAT_SUM_LAST CURR_TYPE,
    FLAT_SUM_BEGIN CURR_TYPE,
    FLAT_PERC_END CURR_TYPE,
    FLAT_PERC_LAST CURR_TYPE,
    FLAT_PERC_BEGIN CURR_TYPE,
    PORTFEL_PARAL_END CURR_TYPE,
    PORTFEL_PARAL_LAST CURR_TYPE,
    PORTFEL_PARAL_BEGIN CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable rep_date_last date;
declare variable def_curr varchar(3);
declare variable cr_rate_end curr_type;
declare variable cr_rate_last curr_type;
declare variable cr_rate_begin curr_type;
declare variable n smallint;
begin
  select cm_default_currency
  from company
  into def_curr;
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end);
  rep_date_last = rep_date_end - 1;
-- вычисление день недели
  n = extract(weekday from rep_date_last);
-- если попадает на выходные предыдущий день, то передвигает на пятьницу
--  if (n=0) then rep_date_last=rep_date_last - 2;
  if (n = 6) then
    rep_date_last = rep_date_last - 2;
-- Валюта на конец
  select cr_rate
  from opercurr
  where :rep_date_end between cr_date and cr_date_end and
        cr_from_cu_id = '840'
  into cr_rate_end;
-- Валюта на предыдущий день
  select cr_rate
  from opercurr
  where :rep_date_last between cr_date and cr_date_end and
        cr_from_cu_id = '840'
  into cr_rate_last;
-- Валюта на начало
  select cr_rate
  from opercurr
  where :rep_date_begin between cr_date and cr_date_end and
        cr_from_cu_id = '840'
  into cr_rate_begin;
-- Расчет кредитного партфеля на конец
  with tabl1
  as (select oks_date_begin, oks_date_end, oks_delinq_date,
             iif(k_index_curr_code = :def_curr, oks_balance, iif(:cr_rate_end > k_index_curr_rate, oks_balance * :cr_rate_end, oks_balance * k_index_curr_rate)) as oks_balance,
             kd_percent_per_day * 365 as flat_percent
      from oksummary, kontract, kontdata, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_from_k_id = k_id and
            k_id = kd_from_k_id and
            kd_ol_is_delete = 0 and
            :rep_date_end between kd_ol_date and kd_date_end and
            oks_close_state = 0 and
            oks_region_id = id_list)
  select sum(oks_balance) portfel_end, sum(iif(oks_delinq_date <= :rep_date_end, oks_balance, 0)),
         sum(iif(oks_delinq_date < :rep_date_end - 7, oks_balance, 0)),
         sum(iif(oks_delinq_date < :rep_date_end - 30, oks_balance, 0)),
         sum(cast(oks_balance as double precision) / 100 * flat_percent)
  from tabl1
  into portfel_end, delinq_balance_end, delinq_balane7_end, delinq_balane30_end, flat_sum_end;
-- Расчет кредитного портфеля на предыдущий день
  with tabl1
  as (select oks_date_begin, oks_date_end, oks_delinq_date,
             iif(k_index_curr_code = :def_curr, oks_balance, iif(:cr_rate_last > k_index_curr_rate, oks_balance * :cr_rate_last, oks_balance * k_index_curr_rate)) as oks_balance,
             kd_percent_per_day * 365 as flat_percent
      from oksummary, kontract, kontdata, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_last between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_from_k_id = k_id and
            k_id = kd_from_k_id and
            kd_ol_is_delete = 0 and
            oks_close_state = 0 and
            :rep_date_end between kd_ol_date and kd_date_end and
            oks_region_id = id_list)
  select sum(oks_balance) portfel_end, sum(iif(oks_delinq_date <= :rep_date_last, oks_balance, 0)),
         sum(iif(oks_delinq_date < :rep_date_last - 7, oks_balance, 0)),
         sum(iif(oks_delinq_date < :rep_date_last - 30, oks_balance, 0)),
         sum(cast(oks_balance as double precision) / 100 * flat_percent)
  from tabl1
  into portfel_last, delinq_balance_last, delinq_balane7_last, delinq_balane30_last, flat_sum_last;
-- Расчет кредитного портфеля на начало месяца
  with tabl1
  as (select oks_date_begin, oks_date_end, oks_delinq_date,
             iif(k_index_curr_code = :def_curr, oks_balance, iif(:cr_rate_begin > k_index_curr_rate, oks_balance * :cr_rate_begin, oks_balance * k_index_curr_rate)) as oks_balance,
             kd_percent_per_day * 365 as flat_percent
      from oksummary, kontract, kontdata, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_begin between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_from_k_id = k_id and
            k_id = kd_from_k_id and
            kd_ol_is_delete = 0 and
            oks_close_state = 0 and
            :rep_date_end between kd_ol_date and kd_date_end and
            oks_region_id = id_list)
  select sum(oks_balance) portfel_end, sum(iif(oks_delinq_date <= :rep_date_begin, oks_balance, 0)),
         sum(iif(oks_delinq_date < :rep_date_begin - 7, oks_balance, 0)),
         sum(iif(oks_delinq_date < :rep_date_begin - 30, oks_balance, 0)),
         sum(cast(oks_balance as double precision) / 100 * flat_percent)
  from tabl1
  into portfel_begin, delinq_balance_begin, delinq_balane7_begin, delinq_balane30_begin, flat_sum_begin;
-- кол-во клиентов на конец
  select sum(ac_is_client), count(ac_is_client), sum(iif(ac_is_client = 0, ac_cli_balance, 0))
  from aarep_cli_active_list(:rep_date_end), region_list(:rep_region_id, 1, 3, 1)
  where ac_region_id = id_list
  into kol_cli_end, kol_kred_end, portfel_paral_end;
-- кол-во клиентов на предыдущий день
  select sum(ac_is_client), count(ac_is_client), sum(iif(ac_is_client = 0, ac_cli_balance, 0))
  from aarep_cli_active_list(:rep_date_last), region_list(:rep_region_id, 1, 3, 1)
  where ac_region_id = id_list
  into kol_cli_last, kol_kred_last, portfel_paral_last;
-- кол-во клиентов на начало
  select sum(ac_is_client), count(ac_is_client), sum(iif(ac_is_client = 0, ac_cli_balance, 0))
  from aarep_cli_active_list(:rep_date_begin), region_list(:rep_region_id, 1, 3, 1)
  where ac_region_id = id_list
  into kol_cli_begin, kol_kred_begin, portfel_paral_begin;
-- Вычисление параллельных клиентов
  kol_paral_end = kol_kred_end - kol_cli_end;
  kol_paral_last = kol_kred_last - kol_cli_last;
  kol_paral_begin = kol_kred_begin - kol_cli_begin;
-- Баланс реструктуризированных кредитов с просроченными днями менее 30 дней, отчет на конец
  with tabl1
  as (select distinct or_from_k_id
      from kontstate, operrest, operlist
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = or_from_k_id and
            or_from_ol_id = ol_id and
            or_date_end >= :rep_date_end and
            operlist.is_delete = 0 and
            ol_date <= :rep_date_end)
  select sum(oks_balance)
  from tabl1, oksummary, region_list(:rep_region_id, 1, 3, 1)
  where or_from_k_id = oks_from_k_id and
        :rep_date_end between oks_date_begin and oks_date_end and
        :rep_date_end - oks_delinq_date < 30 and
        oks_region_id = id_list and
        oks_close_state = 0
  into rest_balance_end;
-- Баланс реструктуризированных кредитов с просроченными днями менее 30 дней, отчет на предыдущий день
  with tabl1
  as (select distinct or_from_k_id
      from kontstate, operrest, operlist
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_last between ks_date_begin and ks_date_end and
            ks_from_k_id = or_from_k_id and
            or_from_ol_id = ol_id and
            or_date_end >= :rep_date_last and
            operlist.is_delete = 0 and
            ol_date <= :rep_date_last)
  select sum(oks_balance)
  from tabl1, oksummary, region_list(:rep_region_id, 1, 3, 1)
  where or_from_k_id = oks_from_k_id and
        :rep_date_last between oks_date_begin and oks_date_end and
        :rep_date_last - oks_delinq_date < 30 and
        oks_region_id = id_list and
        oks_close_state = 0
  into rest_balance_last;
-- Баланс реструктуризированных кредитов с просроченными днями менее 30 дней, отчет на начало
  with tabl1
  as (select distinct or_from_k_id
      from kontstate, operrest, operlist
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_begin between ks_date_begin and ks_date_end and
            ks_from_k_id = or_from_k_id and
            or_from_ol_id = ol_id and
            or_date_end >= :rep_date_begin and
            operlist.is_delete = 0 and
            ol_date <= :rep_date_begin)
  select sum(oks_balance)
  from tabl1, oksummary, region_list(:rep_region_id, 1, 3, 1)
  where or_from_k_id = oks_from_k_id and
        :rep_date_begin between oks_date_begin and oks_date_end and
        :rep_date_begin - oks_delinq_date < 30 and
        oks_region_id = id_list and
        oks_close_state = 0
  into rest_balance_begin;
-- Расчет средневзвешанной процентной ставки
  flat_perc_end = flat_sum_end / portfel_end;
  flat_perc_last = flat_sum_last / portfel_last;
  flat_perc_begin = flat_sum_begin / portfel_begin;

  suspend;
end
