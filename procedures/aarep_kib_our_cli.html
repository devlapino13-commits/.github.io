CREATE PROCEDURE AAREP_KIB_OUR_CLI(
    REP_DATE_END DATE)
RETURNS (
    KS_FROM_K_ID DECIMAL(18,0),
    KC_FROM_C_ID DECIMAL(18,0),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    K_KONTRACT_NO VARCHAR(20),
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_CLI_BALANCE DECIMAL(18,4),
    K_INDEX_CURR_CODE VARCHAR(10),
    K_INDEX_CURR_RATE DECIMAL(18,4),
    K_KONT_BALANCE DECIMAL(18,4),
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30))
AS
declare variable old_k_id decimal(18,0);
declare variable cr_from_cu_id varchar(3);
declare variable cr_rate decimal(18,4);
begin
  old_k_id = 0;
  k_cli_balance = 0;
  select cr_from_cu_id, cr_rate
  from opercurr
  where :rep_date_end between cr_date and cr_date_end and
        cr_from_cu_id = '840'
  into :cr_from_cu_id, :cr_rate;

  for select distinct ks_from_k_id, kc_from_c_id, kc_amount_approved, k_kontract_no, k_amount_approved,
                      k_index_curr_code, k_index_curr_rate, c_name_fio, c_birth_date, c_pass_no
      from kontstate, knclient, kibminiapp, kibhistory, kontract, clients
      where :rep_date_end between ks_date_begin and ks_date_end and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            kc_from_k_id = ks_from_k_id and
            kc_amount_approved <> 0 and
            kbm_from_c_id = kc_from_c_id and
            kh_from_kbm_id = kbm_id and
            kh_is_our = 0 and
            k_id = ks_from_k_id and
            kc_from_c_id = c_id
      order by k_id
      into :ks_from_k_id, :kc_from_c_id, :kc_amount_approved, :k_kontract_no, :k_amount_approved, :k_index_curr_code,
           :k_index_curr_rate, :c_name_fio, :c_birth_date, :c_pass_no
  do
  begin
    k_cli_balance = null;
    if (old_k_id <> :ks_from_k_id) then
    begin
      k_kont_balance = 0;
      select oks_balance
      from oksummary
      where oks_from_k_id = :ks_from_k_id and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_close_state = 0
      into :k_kont_balance;
      if (k_kont_balance < 0) then
        k_kont_balance = 0;

      if (:k_index_curr_code = :cr_from_cu_id) then
      begin
        if (:k_index_curr_rate > :cr_rate) then
          k_kont_balance = :k_kont_balance * :k_index_curr_rate;
        else
          k_kont_balance = :k_kont_balance * :cr_rate;
      end
    end

    k_cli_balance = :kc_amount_approved / :k_amount_approved * :k_kont_balance;
    old_k_id = :ks_from_k_id;

    suspend;
  end
end
