CREATE PROCEDURE HP_OBJECTPROPS_SET(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID VARCHAR(255),
    SUBJECT_KIND VARCHAR(100),
    SUBJECT_ID VARCHAR(255),
    DATE_BEGIN DATE,
    SET_VALUE D_STR_2048,
    SET_ADD_VALUE DECIMAL(18,4),
    UN_ID DECIMAL(18,0))
AS
declare variable UNIQ_OBJECT_ID decimal(18,0);
declare variable UNIQ_SUBJECT_ID decimal(18,0);
declare variable UNIQ_OBJECT_KIND integer;
declare variable UNIQ_SUBJECT_KIND integer;
declare variable OVS_FROM_OPR_ID decimal(18,0);
begin
-- Процедура устанавливает новое значение для пары Object/Subject
  if (object_kind similar to '[-]?[[:DIGIT:]]+' and
      object_id similar to '[-]?[[:DIGIT:]]+' and
      subject_kind similar to '[-]?[[:DIGIT:]]+' and
      subject_id similar to '[-]?[[:DIGIT:]]+') then
  begin
    uniq_object_kind = object_kind;
    uniq_object_id = object_id;
    uniq_subject_kind = subject_kind;
    uniq_subject_id = subject_id;
  end
  else
  begin
    select opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id
    from hp_get_propsparam(:object_kind, :object_id, :subject_kind, :subject_id)
    into :uniq_object_kind, :uniq_object_id, :uniq_subject_kind, :uniq_subject_id;
  end

  if (set_add_value is null) then
    set_add_value = 0;
  if (date_begin is null) then
    date_begin = current_date;

  select opr_id
  from objectprops
  where opr_object_kind = :uniq_object_kind and
        opr_object_id = :uniq_object_id and
        opr_subject_kind = :uniq_subject_kind and
        opr_subject_id = :uniq_subject_id
  into :ovs_from_opr_id;

  if (ovs_from_opr_id is null) then
  begin
    insert into objectprops (opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id)
    values (:uniq_object_kind, :uniq_object_id, :uniq_subject_kind, :uniq_subject_id)
    returning (opr_id)
    into :ovs_from_opr_id;
  end

  insert into objectvalues (ovs_from_opr_id, ovs_date_begin, ovs_value, ovs_add_value, ovs_from_un_id)
  values (:ovs_from_opr_id, :date_begin, :set_value, :set_add_value, :un_id);

  if (uniq_object_kind = 0 and
      uniq_subject_kind = -1 and
      coalesce(set_add_value, 0) = 0) then
  begin
    if (exists(
        select * from refdata
        where rd_id = :uniq_subject_id and
        rd_main_code = 'ContactInfo')) then
      exception e_default 'Error inserting ContactInfo in objectvalues. Use HP_CONTINFO_UPDATE';
  end
end
