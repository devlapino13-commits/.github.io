CREATE PROCEDURE ADDR_CORR
RETURNS (
    C_ID DECIMAL(18,0),
    ADL_ID DECIMAL(18,0),
    ADC_PHONE VARCHAR(255),
    ADC_MEMO VARCHAR(255))
AS
begin
  for select adc_from_c_id
      from addrclient
      where adc_from_adl_id = 81001000000000 and
            adc_address_type = 1
      into :c_id
  do
  begin
    adl_id = null;
    select adc_from_adl_id, adc_phone, adc_memo
    from addrclient
    where adc_from_adl_id <> 81001000000000 and
          adc_from_c_id = :c_id rows 1
    into :adl_id, adc_phone, adc_memo;
    if (adl_id is not null) then
    begin
      update addrclient
      set adc_from_adl_id = :adl_id,
          adc_phone = :adc_phone,
          adc_memo = :adc_memo
      where adc_from_c_id = :c_id and
            adc_address_type = 1;
      if (exists(select *
                 from addrclient
                 where adc_from_adl_id = 81001000000000 and
                       adc_address_type = 2)) then
      begin
        update addrclient
        set adc_from_adl_id = :adl_id,
            adc_phone = :adc_phone,
            adc_memo = :adc_memo
        where adc_from_c_id = :c_id and
              adc_address_type = 2;
      end
    end

  end

end
