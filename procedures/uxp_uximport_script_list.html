CREATE PROCEDURE UXP$UXIMPORT_SCRIPT_LIST
RETURNS (
    TABLE_NAME D_STR_CODE,
    SQL_BLOCK_EXTERNAL VARCHAR(8000) CHARACTER SET WIN1251,
    SQL_BLOCK_LOCAL VARCHAR(8000) CHARACTER SET WIN1251,
    SQL_SELECT VARCHAR(8000) CHARACTER SET WIN1251,
    SQL_INSERT VARCHAR(8000) CHARACTER SET WIN1251,
    SQL_EMPTY_SELECT VARCHAR(8000))
AS
begin
  for select table_name
  from uxp$uximport_cfg
  order by table_id
  into :table_name
  do
  begin
    select sql_block_external, sql_block_local, sql_select, sql_insert, sql_empty_select
    from uxp$uximport_script_cfg(:table_name)
    into :sql_block_external, sql_block_local, sql_select, sql_insert, sql_empty_select;

    suspend;
  end
end
