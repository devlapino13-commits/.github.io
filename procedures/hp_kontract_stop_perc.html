CREATE PROCEDURE HP_KONTRACT_STOP_PERC(
    K_ID DECIMAL(18,0),
    K_STOP_PERC DATE,
    K_STOP_FINE DATE)
AS
begin
  if (k_stop_perc is not null) then
  begin
    update kontract
    set k_stop_perc = :k_stop_perc
    where k_id = :k_id and
          k_stop_perc is null;
  end
  if (k_stop_fine is not null) then
  begin
    update kontract
    set k_stop_fine = :k_stop_fine
    where k_id = :k_id and
          k_stop_fine is null;
  end
end
