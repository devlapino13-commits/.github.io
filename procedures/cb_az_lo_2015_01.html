CREATE PROCEDURE CB_AZ_LO_2015_01(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    BALANCE CURR_TYPE,
    DELINQ_BALANCE_15 CURR_TYPE,
    DELINQ_BALANCE_30 CURR_TYPE,
    PAR_15 CURR_TYPE,
    PAR_30 CURR_TYPE,
    BALANCE_LAST CURR_TYPE,
    BALANCE_AVG CURR_TYPE,
    BONUS_PORTFEL CURR_TYPE,
    PERC_UDERJANIE INTEGER,
    BONUS_FINAL CURR_TYPE)
AS
declare variable rep_date_end_last date;
declare variable reg_id_prior integer;
declare variable c_id decimal(18,0);
declare variable reg_id integer;
begin
  rep_date_end_last = rep_date_end - extract(day from :rep_date_end);
  reg_id_prior = 0;
  for select ps_from_c_id, id_list
      from hr_pstatus hr1, joblist, jobkind, region_list(:rep_region_id, 1, 3, 1)
      where ps_status = 1 and
            ps_is_working = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            ps_from_reg_id = id_list

            and
            (dateadd(month, 3, ps_date_begin) < :rep_date_end -- проверка стажа
            or
            (select min(ps_date_begin)
             from hr_pstatus hr2, joblist, jobkind -- если на текущем должности менее 3 месяцев, то дополнительно проверяет стаж
             where hr2.ps_from_c_id = hr1.ps_from_c_id and
                   ps_status = 1 and
                   ps_is_working = 1 and
                   ps_from_jl_id = jl_id and
                   jl_kind = jk_id and
                   jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) < dateadd(month, -3, :rep_date_end))
      order by id_list
      into c_id, reg_id
  do
  begin
--  вычисления портфеля и пар
    balance = null;
    delinq_balance_15 = null;
    delinq_balance_30 = null;
    select sum(oks_balance), sum(iif(:rep_date_end - oks_delinq_date > 15, oks_balance, 0)),
           sum(iif(:rep_date_end - oks_delinq_date > 30, oks_balance, 0))
    from oksummary
    where oks_kc_id = :c_id and
          oks_region_id = :reg_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0
    into balance, delinq_balance_15, delinq_balance_30;

    if (balance is null) then
      balance = 0;
    if (delinq_balance_15 is null) then
      delinq_balance_15 = 0;
    if (delinq_balance_30 is null) then
      delinq_balance_30 = 0;

    if (balance > 0) then
    begin
      par_15 = delinq_balance_15 / balance * 100;
      par_30 = delinq_balance_30 / balance * 100;
    end
    else
    begin
      par_15 = 0;
      par_30 = 0;
    end

-- Вычисления баланса на предыдущую дату
    balance_last = null;
    select sum(oks_balance)
    from oksummary
    where oks_kc_id = :c_id and
          oks_region_id = :reg_id and
          :rep_date_end_last between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0
    into balance_last;
    if (balance_last is null) then
      balance_last = 0;

    balance_avg = (balance + balance_last) / 2;
--  Пункт 3,1 после достижения портфеля 100тыс манат
    bonus_portfel = 0;
    if (balance_avg >= 100000.000) then
    begin
      bonus_portfel = case
                        when balance_avg >= 600000.0000 then 1200
                        when balance_avg >= 400000.0000 then 900
                        when balance_avg >= 300000.0000 then 700
                        when balance_avg >= 200000.0000 then 500
                        else 300
                      end;
    end
--  Пункт 5,2. процент удержание бонусов
    perc_uderjanie = maxvalue(case
                                when par_15 < 1.0000 then 0
                                when par_15 < 2.0000 then 15
                                when par_15 < 3.0000 then 30
                                else 45
                              end, case
                                     when par_30 < 1.0000 then 0
                                     when par_30 < 2.0000 then 30
                                     when par_30 < 3.0000 then 60
                                     else 90
                                   end);

    bonus_final = bonus_portfel * (100 - perc_uderjanie) / 100;

    officer_name = (select c_name_fio
                    from clients
                    where c_id = :c_id);
    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end

    suspend;
  end
end
