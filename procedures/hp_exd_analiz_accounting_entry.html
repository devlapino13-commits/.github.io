CREATE PROCEDURE HP_EXD_ANALIZ_ACCOUNTING_ENTRY(
    IFA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable ifa_date d_date;
declare variable ifa_file_name d_str_100;
declare variable ifa_is_detail_problem d_bool;
declare variable ied_id d_integer;
declare variable ied_rec_no d_integer;
declare variable ied_trans_datetime d_datetime;
declare variable ied_ipc_trans_processed d_date;
declare variable ied_bank_code d_str_lang;
declare variable ied_filial_code d_str_lang;
declare variable cal_card_id d_integer;
declare variable ied_from_cca_id d_integer;
declare variable ied_pan d_str_20;
declare variable ied_from_c_id d_id;
declare variable ied_from_k_id d_id;
declare variable binstr d_str_10;
declare variable card_is_our d_bool;
declare variable ied_trans_currency d_str_lang;
declare variable ied_trans_amount d_currency;
declare variable ied_oper_type d_str_sym;
declare variable ied_from_itt_id d_integer;
declare variable itt_code d_str_10;
declare variable itt_name_en d_str_255;
declare variable ied_terminal_type d_str_sym;
declare variable ied_trade_point_code d_str_20;
declare variable ied_trade_point_name d_str_code;
declare variable ied_trade_point_mcc d_str_10;
declare variable ied_city d_str_20;
declare variable ied_emitent_comission d_currency;
declare variable ied_acquirer_cmi d_str_20;
declare variable ied_acquirer_bank_id d_integer;
declare variable ib_bank_code d_str_code;
declare variable ied_acquirer_comission d_currency;
declare variable curr_code d_str_lang;
declare variable trm_type d_smallint;
declare variable trm_from_object_id d_id;
declare variable c_id d_id;
declare variable ol_parent_id d_id;
declare variable ol_id d_id;
declare variable tmp_ol_id d_id;
declare variable tmp1_ol_id d_id;
declare variable tmp2_ol_id d_id;
declare variable ob_detail d_str_long;
declare variable ob_detail1 d_str_long;
declare variable reg_id d_integer;
declare variable card_reg_id d_integer;
declare variable object_id d_id;
declare variable num_row d_integer;
declare variable ib_subaccount d_integer;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable total_line d_integer;
declare variable res_line_no d_integer;
declare variable acc_10004 d_str_20;
declare variable acc_10102_1 d_str_20;
declare variable acc_10024_1 d_str_20;
declare variable acc_10024_2 d_str_20;
declare variable acc_20001_1 d_str_20;
declare variable acc_21114_11 d_str_20;
declare variable acc_21122 d_str_20;
declare variable acc_80509_10 d_str_20;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable rd_id_tsp d_integer;
begin
  err_text = '';
  curr_code = '417';
  rd_id_tsp = f_get_rd_id('AgreementKinds.tsp');

  select
    ifa_date,
    ifa_file_name,
    ifa_is_detail_problem
  from ipc_file_analiz
  where ifa_id = :ifa_id
    and ifa_file_type = 2
    and ifa_status = 0
  into
    :ifa_date,
    :ifa_file_name,
    :ifa_is_detail_problem;
  if (row_count = 0 or not exists(select * from ipc_exd_detail where ied_from_ifa_id = :ifa_id and ied_error_id is null)) then
    err_text = 'Данные для построения бухгалтерских проводок не определены.';

  if (err_text = '') then
  begin
    reg_id = 1090100;
    num_row = 0;
    for
      select
        ied_id,
        ied_rec_no,
        ied_trans_datetime,
        ied_ipc_trans_processed,
        ied_bank_code,
        ied_filial_code,
        ied_from_cca_id,
        ied_from_card_id,
        ied_pan,
        ied_from_c_id,
        ied_from_k_id,
        ied_trans_currency,
        ied_trans_amount,
        ied_oper_type,
        ied_from_itt_id,
        itt_code,
        itt_name_en,
        ied_terminal_type,
        ied_trade_point_code,
        ied_trade_point_name,
        ied_trade_point_mcc,
        ied_city,
        ied_emitent_comission,
        ied_acquirer_cmi,
        ied_acquirer_bank_id,
        ib_bank_code,
        abs(ied_acquirer_commission)
      from ipc_exd_detail
        left join ipc_transaction_types
          on itt_id = ied_from_itt_id
        left join ipc_banks
          on ib_id = ied_acquirer_bank_id
      where ied_from_ifa_id = :ifa_id
        and ied_error_id is null
        and ied_trade_point_code <> ''
        and itt_code not in ('110', '11V', '120', '12V')
    into
      :ied_id,
      :ied_rec_no,
      :ied_trans_datetime,
      :ied_ipc_trans_processed,
      :ied_bank_code,
      :ied_filial_code,
      :ied_from_cca_id,
      :cal_card_id,
      :ied_pan,
      :ied_from_c_id,
      :ied_from_k_id,
      :ied_trans_currency,
      :ied_trans_amount,
      :ied_oper_type,
      :ied_from_itt_id,
      :itt_code,
      :itt_name_en,
      :ied_terminal_type,
      :ied_trade_point_code,
      :ied_trade_point_name,
      :ied_trade_point_mcc,
      :ied_city,
      :ied_emitent_comission,
      :ied_acquirer_cmi,
      :ied_acquirer_bank_id,
      :ib_bank_code,
      :ied_acquirer_comission
    do
    begin
      num_row = num_row + 1;
      if (num_row = 1) then
        insert into operlist (
          ol_from_reg_id,
          ol_oper_type,
          ol_oper_subtype,
          ol_date,
          ol_doc_no,
          ol_doc_kind,
          ol_memo,
          ol_user_id
        )
        values (
          :reg_id,
          2,
          11,
          :ifa_date,
          :ifa_id,
          9,
          f_format('EXD-файл "%1" от %2 г.', :ifa_file_name, f_datetostr(:ifa_date)),
          :un_id
        )
        returning ol_id
        into :ol_parent_id;

      -- тип устройства (0-чужое, 1-ТСП, 2-ПВН, 3-QR, 4-P2P, 5-WEB)
      trm_type = 0;
      trm_from_object_id = null;
      select
        trm_type,
        ito_object_id
      from ipc_terminals
        inner join ipc_terminal_ownership
          on ito_trm_id = trm_id
            and :ifa_date between ito_date_begin and ito_date_end
      where trm_ipc_id = :ied_trade_point_code
      into
        :trm_type,
        :trm_from_object_id;
      if (trm_type = 0) then
      begin
        if (ied_trade_point_code = 'IPC21001') then trm_type = 3;
        else
        if (ied_trade_point_code = 'IPCP2P21') then trm_type = 4;
        else
        if (ied_trade_point_code = 'WSCH21') then trm_type = 5;
      end
      object_id = iif(trm_type in (1,2), trm_from_object_id, null);

      if (trm_type in (5)) then
        exception e_message f_format('Проводки по: "%1" еще не реализованы.', ied_trade_point_code);

      -- Принадлежность карты (наша/не наша)
      binstr = left(ied_pan, 6);
      if (exists(
        select *
        from ipc_bank_cards
          inner join ipc_banks
            on ib_id = ibc_from_ib_id
              and ib_bank_code = 'MBULAK'
        where ibc_bin = :binstr))
      then card_is_our = 1;
        else card_is_our = 0;

      -- Офис карты
      card_reg_id = null;
      select
        cpl_from_reg_id
      from cardlist
        inner join cardplace
          on cpl_id = cal_last_cpl_id
      where cal_card_id = :cal_card_id
      into
        :card_reg_id;

      -- Контрагент ТСП
      if (trm_type = 1) then
        c_id = (
          select
            cs_from_c_id
          from clientshops
          where cs_id = :object_id and
                cs_from_rd_id = :rd_id_tsp);
      else c_id = null;

      -- Субсчет эмитента/эквайера
      ib_subaccount = (
        select first 1
          ib_subaccount
        from ipc_bank_cards
          inner join ipc_banks
            on ib_id = ibc_from_ib_id
        where ibc_cmi = :ied_acquirer_cmi);

      acc_10004  = (
        select
          al_id
        from acc_list
        where al_from_reg_id = :reg_id
          and al_currency = :curr_code
          and al_account = '10004/' || :ib_subaccount);

      acc_10102_1  = (
        select
          al_id
        from acc_list
        where al_from_reg_id = :reg_id
          and al_currency = :curr_code
          and al_account = '10150/101');

      acc_10024_1  = (
        select
          al_id
        from acc_list
        where al_from_reg_id = :reg_id
          and al_currency = :curr_code
          and al_account = '10024/1');

      acc_10024_2  = (
        select
          al_id
        from acc_list
        where al_from_reg_id = :object_id
          and al_currency = :curr_code
          and al_account = '10024/2');

      acc_20001_1  = (
        select
          al_id
        from acc_list
        where al_from_reg_id = :card_reg_id
          and al_currency = :curr_code
          and al_account = '20001/1');

      acc_21114_11 = (
        select
          al_id
        from acc_list
        where al_from_reg_id = :card_reg_id
          and al_currency = :curr_code
          and al_account = '21114/11');

      acc_21122    = (
        select first 1
          ag_from_al_id
        from agreements
        where ag_object_kind = 0
          and ag_object_id = :c_id
          and ag_from_rd_id = f_get_rd_id('AgreementKinds.tsp')
          and ag_status = 1
          and :ifa_date between coalesce(ag_date_begin, '01.01.1900') and coalesce(ag_date_end, '31.12.9999')
        order by ag_id desc);

      acc_80509_10 = (
        select
          al_id
        from acc_list
        where al_from_reg_id = :card_reg_id
          and al_currency = :curr_code
          and al_account = '80509/10');

      ob_detail = f_format('Транзакция%1: %2 %3; ',
        iif(cal_card_id > 0, ' по ID ' || cal_card_id || ' (сч.' || ied_from_cca_id || ')', ''),
        itt_code,
        itt_name_en);
      ob_detail = ob_detail || f_format('Устройство: "%1"; Пункт: %2 %3; MCC-код: %4; ' ,
        case ied_terminal_type
          when 'P' then 'POS'
          when 'A' then 'ATM'
          when 'N' then 'импринтер'
        else '-' end,
        ied_trade_point_code,
        ied_trade_point_name,
        ied_trade_point_mcc);
      ob_detail = ob_detail || f_format('Дата: %1; Bank Name: %2; Город: %3',
        f_datetimetostr(ied_trans_datetime),
        coalesce(iif(trm_type = 4, ied_trade_point_code, ib_bank_code), ''),
        ied_city);

      -- Основная
      acc_dt = acc_20001_1;
      acc_kt =
        case trm_type
          when 0 then :acc_10004   -- чужое устройство
          when 1 then :acc_21122   -- ТСП
          when 2 then :acc_10024_2 -- ПВН
          when 3 then :acc_10024_1 -- QR
          when 4 then :acc_10004   -- P2P
        end;

      for
        select
          res_reg_id,
          res_acc_dt,
          res_acc_kt,
          total_line,
          res_line_no
        from aarep_loronostro_link(:card_reg_id, :acc_dt, :acc_kt)
      into
        :res_reg_id,
        :res_acc_dt,
        :res_acc_kt,
        :total_line,
        :res_line_no
      do
      begin
        if (res_acc_dt is null) then
          exception e_message(f_format('Не определен счет %1 по DT, офис %2.', acc_dt, res_reg_id));
        if (res_acc_kt is null) then
          exception e_message(f_format('Не определен счет %1 по KT, офис %2.', acc_kt, res_reg_id));

        if (res_line_no = 1) then
          tmp_ol_id = ol_parent_id;

        insert into operlist (
          ol_parent_id,
          ol_from_reg_id,
          ol_oper_type,
          ol_oper_subtype,
          ol_date,
          ol_doc_no,
          ol_doc_kind,
          ol_memo,
          ol_document_client,
          ol_user_id
        )
        values (
          :tmp_ol_id,
          :res_reg_id,
          2,
          11,
          :ifa_date,
          :ifa_id,
          9,
          f_format('EXD-файл "%1" от %2 г. строка %3', :ifa_file_name, f_datetostr(:ifa_date), :ied_rec_no),
          :ied_from_c_id,
          :un_id
        )
        returning ol_id
        into :ol_id;

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val,
          ob_client_kt
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :res_acc_dt,
          :res_acc_kt,
          :ob_detail,
          :ied_trans_amount,
          :ied_trans_amount,
          :c_id
        );

        if (res_line_no = 1) then
        begin
          tmp_ol_id = ol_id;

          insert into cardclientacc_balance (
            ccb_from_cca_id,
            ccb_from_ol_id,
            ccb_ol_date,
            ccb_amount_plus,
            ccb_amount_minus,
            ccb_ol_is_active,
            ccb_oper_type
          )
          values (
            :ied_from_cca_id,
            :tmp_ol_id,
            :ifa_date,
            iif(:ied_oper_type = 'C', :ied_trans_amount, 0),
            iif(:ied_oper_type = 'D', :ied_trans_amount, 0),
            1,
            iif(:ied_oper_type = 'D', 1, 2)
          );
        end
      end
      if (trm_type in (0, 4)) then
      begin
        insert into operlist (
          ol_parent_id,
          ol_from_reg_id,
          ol_oper_type,
          ol_oper_subtype,
          ol_date,
          ol_doc_no,
          ol_doc_kind,
          ol_memo,
          ol_document_client,
          ol_user_id
        )
        values (
          :tmp_ol_id,
          :reg_id,
          2,
          11,
          :ifa_date,
          :ifa_id,
          9,
          f_format('EXD-файл "%1" от %2 г. строка %3', :ifa_file_name, f_datetostr(:ifa_date), :ied_rec_no),
          :ied_from_c_id,
          :un_id
        )
        returning ol_id
        into :ol_id;

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val,
          ob_client_kt
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :acc_10004,
          :acc_10102_1,
          :ob_detail,
          :ied_trans_amount,
          :ied_trans_amount,
          :c_id
        );
      end

      -- Комиссия Эквайера
      if (ied_acquirer_comission > 0) then
      begin
        ob_detail1 = 'Комиссия Эквайера; ' || ob_detail;
        for
          select
            res_reg_id,
            res_acc_dt,
            res_acc_kt,
            total_line,
            res_line_no
          from aarep_loronostro_link(:card_reg_id, :acc_80509_10, :acc_10004)
        into
          :res_reg_id,
          :res_acc_dt,
          :res_acc_kt,
          :total_line,
          :res_line_no
        do
        begin
          if (res_line_no = 1) then
            tmp1_ol_id = ol_parent_id;

          insert into operlist (
            ol_parent_id,
            ol_from_reg_id,
            ol_oper_type,
            ol_oper_subtype,
            ol_date,
            ol_doc_no,
            ol_doc_kind,
            ol_memo,
            ol_document_client,
            ol_user_id
          )
          values (
            :tmp1_ol_id,
            :res_reg_id,
            2,
            11,
            :ifa_date,
            :ifa_id,
            9,
            f_format('EXD-файл(комиссия эквайера) "%1" от %2 г. строка %3', :ifa_file_name, f_datetostr(:ifa_date), :ied_rec_no),
            :ied_from_c_id,
            :un_id
          )
          returning ol_id
          into :ol_id;

          if (res_line_no = 1) then
            tmp1_ol_id = ol_id;

          insert into operbook (
            ob_from_ol_id,
            ob_currency,
            ob_curr_rate,
            ob_account_dt,
            ob_account_kt,
            ob_detail,
            ob_summa,
            ob_summa_val,
            ob_client_kt
          )
          values (
            :ol_id,
            :curr_code,
            1,
            :res_acc_dt,
            :res_acc_kt,
            :ob_detail1,
            :ied_acquirer_comission,
            :ied_acquirer_comission,
            :c_id
          );
        end
        insert into operlist (
          ol_parent_id,
          ol_from_reg_id,
          ol_oper_type,
          ol_oper_subtype,
          ol_date,
          ol_doc_no,
          ol_doc_kind,
          ol_memo,
          ol_document_client,
          ol_user_id
        )
        values (
          :tmp1_ol_id,
          :reg_id,
          2,
          11,
          :ifa_date,
          :ifa_id,
          9,
          f_format('EXD-файл(комиссия эквайера) "%1" от %2 г. строка %3', :ifa_file_name, f_datetostr(:ifa_date), :ied_rec_no),
          :ied_from_c_id,
          :un_id
        )
        returning ol_id
        into :ol_id;

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val,
          ob_client_kt
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :acc_10004,
          :acc_10102_1,
          :ob_detail1,
          :ied_acquirer_comission,
          :ied_acquirer_comission,
          :c_id
        );
      end  ---

      -- Комиссия Эмитента
      if (ied_emitent_comission > 0) then
      begin
        ob_detail1 = 'Комиссия Эмитента; ' || ob_detail;
        for
          select
            res_reg_id,
            res_acc_dt,
            res_acc_kt,
            total_line,
            res_line_no
          from aarep_loronostro_link(:reg_id, :acc_10004, :acc_21114_11)
        into
          :res_reg_id,
          :res_acc_dt,
          :res_acc_kt,
          :total_line,
          :res_line_no
        do
        begin
          if (res_line_no = 1) then
            tmp2_ol_id = ol_parent_id;

          insert into operlist (
            ol_parent_id,
            ol_from_reg_id,
            ol_oper_type,
            ol_oper_subtype,
            ol_date,
            ol_doc_no,
            ol_doc_kind,
            ol_memo,
            ol_document_client,
            ol_user_id
          )
          values (
            :tmp2_ol_id,
            :res_reg_id,
            2,
            11,
            :ifa_date,
            :ifa_id,
            9,
            f_format('EXD-файл(комиссия эмитента) "%1" от %2 г. строка %3', :ifa_file_name, f_datetostr(:ifa_date), :ied_rec_no),
            :ied_from_c_id,
            :un_id
          )
          returning ol_id
          into :ol_id;

          if (res_line_no = 1) then
            tmp2_ol_id = ol_id;

          insert into operbook (
            ob_from_ol_id,
            ob_currency,
            ob_curr_rate,
            ob_account_dt,
            ob_account_kt,
            ob_detail,
            ob_summa,
            ob_summa_val,
            ob_client_kt
          )
          values (
            :ol_id,
            :curr_code,
            1,
            :res_acc_dt,
            :res_acc_kt,
            :ob_detail1,
            :ied_emitent_comission,
            :ied_emitent_comission,
            :c_id
          );
        end
        insert into operlist (
          ol_parent_id,
          ol_from_reg_id,
          ol_oper_type,
          ol_oper_subtype,
          ol_date,
          ol_doc_no,
          ol_doc_kind,
          ol_memo,
          ol_document_client,
          ol_user_id
        )
        values (
          :tmp2_ol_id,
          :reg_id,
          2,
          11,
          :ifa_date,
          :ifa_id,
          9,
          f_format('EXD-файл(комиссия эмитента) "%1" от %2 г. строка %3', :ifa_file_name, f_datetostr(:ifa_date), :ied_rec_no),
          :ied_from_c_id,
          :un_id
        )
        returning ol_id
        into :ol_id;

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val,
          ob_client_kt
        )
        values (
          :ol_id,
          :curr_code,
          1,
          :acc_10102_1,
          :acc_10004,
          :ob_detail1,
          :ied_emitent_comission,
          :ied_emitent_comission,
          :c_id
        );
      end ---

      update ipc_exd_detail set
        ied_from_ol_id = :tmp_ol_id
      where ied_id = :ied_id;
    end
    if (num_row > 0) then
      update ipc_file_analiz set
        ifa_status = 1,
        ifa_from_un_id = :un_id,
        ifa_datetime = current_timestamp,
        ifa_from_ol_id = :ol_parent_id
      where ifa_id = :ifa_id;
  end

  if (err_text = '' and num_row = 0) then
    err_text = 'Не обработано ни одной строки. Проводки не построены.';

  suspend;
end
