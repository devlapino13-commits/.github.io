CREATE PROCEDURE HP_EXTINFO_INSERT_SMEVPASS(
    C_ID D_ID,
    FORCE_INSERT D_SMALLINT,
    UN_ID D_ID)
AS
declare variable CDS_ID D_ID;
declare variable PK_ID D_INTEGER;
begin
  select cds_id, cds_from_pk_id
  from clientdocs cds
  where cds.cds_from_c_id = :c_id
    and cds_is_main_doc = 1
    and cds_is_delete = 0
    and cds_verify_status = 2
    and cds_index_doc_no <> ''
  order by cds_id desc
  rows 1 -- RFCRU-4832
  into cds_id, pk_id;

  if (pk_id is distinct from 12) then
    exit;

  execute procedure hp_extinfo_insert_bki_cfg('passport_smev', :c_id, f_objkind('clientdocs', 'id'), :cds_id, :force_insert, :un_id);
end
