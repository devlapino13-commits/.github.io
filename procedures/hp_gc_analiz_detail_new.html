CREATE PROCEDURE HP_GC_ANALIZ_DETAIL_NEW(
    IGA_ID D_INTEGER,
    IFA_DATE D_DATE,
    LINE_NO D_INTEGER,
    CARD_NO D_STR_20,
    TRM_IPC_ID D_STR_20,
    AMOUNT D_CURRENCY,
    CURRNM D_STR_LANG,
    TRANS_NO D_STR_10,
    TRANS_DT D_DATE)
RETURNS (
    IGD_ID D_ID)
AS
declare variable c_id d_id;
declare variable ag_id d_integer;
declare variable trm_id d_integer;
declare variable trm_type d_smallint;
declare variable trm_from_object_kind d_integer;
declare variable trm_from_object_id d_id;
declare variable ib_id d_integer;
declare variable bin_str d_str_10;
declare variable is_our_card d_bool;
declare variable commis_perc d_percent;
declare variable commis_amount d_currency;
declare variable ecommis_perc d_percent;
declare variable ecommis_amount d_currency;
begin
  bin_str = left(card_no, 6);
  card_no = bin_str || lpad(right(card_no, 4),10,'*');
  select first 1
    ib_id,
    iif(ib_bic = '106001', 1, 0)
  from ipc_bank_cards
    inner join ipc_banks
      on ib_id = ibc_from_ib_id
  where ibc_bin starting :bin_str
  into
    :ib_id,
    :is_our_card;
  ecommis_perc = iif(is_our_card = 1, 0, 0.3);

  if (trm_ipc_id <> 'IPC21001') then
  begin
    select
      trm_id,
      trm_type,
      ito_object_kind,
      ito_object_id
    from ipc_terminals
      inner join ipc_terminal_ownership
        on ito_trm_id = trm_id
          and :ifa_date between ito_date_begin and ito_date_end
    where trm_ipc_id = :trm_ipc_id
    into
      :trm_id,
      :trm_type,
      :trm_from_object_kind,
      :trm_from_object_id;

    if (trm_id > 0) then
    begin
      commis_perc = 0;
      if (trm_type = 1) then
      begin
        select
          cs_from_c_id
        from clientshops
        where cs_id = :trm_from_object_id
        into
          :c_id;
        select first 1
          ag_id,
          iif(:is_our_card = 1, ag_commis_cards_mb, ag_commis_cards_other)
        from agreements
        where ag_object_kind = 0
          and ag_object_id = :c_id
          and ag_from_rd_id = f_get_rd_id('AgreementKinds.tsp')
          and ag_status = 1
          and :ifa_date between coalesce(ag_date_begin, '01.01.1900') and coalesce(ag_date_end, '31.12.9999')
        order by ag_id desc
        into
          :ag_id,
          :commis_perc;
      end
    end
  end

  commis_amount = round(commis_perc * amount / 100.0, 2);
  ecommis_amount = round(ecommis_perc * amount / 100.0, 2);

  if (trm_ipc_id <> 'IPC21001') then
    insert into ipc_gc_analiz_detail_pos (
      igdp_from_iga_id,
      igdp_rec_no,
      igdp_object_kind,
      igdp_object_id,
      igdp_from_c_id,
      igdp_from_ag_id,
      igdp_terminal_id,
      igdp_type,
      igdp_from_ib_id,
      igdp_is_our_card,
      igdp_card_no,
      igdp_trans_no,
      igdp_trans_date,
      igdp_inet_acquiring,
      igdp_amount,
      igdp_currency,
      igdp_tsp_commis_perc,
      igdp_tsp_commis_amount,
      igdp_ipc_emitent_commis_perc,
      igdp_ipc_emitent_commis_amount
    )
    values (
      :iga_id,
      :line_no,
      :trm_from_object_kind,
      :trm_from_object_id,
      :c_id,
      :ag_id,
      :trm_id,
      :trm_type,
      :ib_id,
      :is_our_card,
      :card_no,
      :trans_no,
      :trans_dt,
      case :trm_ipc_id
        when 'IA210001' then 2
        when 'TS210001' then 1
      else 0 end,
      :amount,
      :currnm,
      :commis_perc,
      :commis_amount,
      :ecommis_perc,
      :ecommis_amount
    )
    returning igdp_id
    into :igd_id;
  else
    insert into ipc_gc_analiz_detail_qr (
      igdq_from_iga_id,
      igdq_rec_no,
      igdq_from_ib_id,
      igdq_is_our_card,
      igdq_card_no,
      igdq_trans_no,
      igdq_trans_date,
      igdq_amount,
      igdq_currency,
      igdq_ipc_emitent_commis_perc,
      igdq_ipc_emitent_commis_amount
    )
    values (
      :iga_id,
      :line_no,
      :ib_id,
      :is_our_card,
      :card_no,
      :trans_no,
      :trans_dt,
      :amount,
      :currnm,
      :ecommis_perc,
      :ecommis_amount
    )
    returning igdq_id
    into :igd_id;
end
