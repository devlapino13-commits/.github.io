CREATE PROCEDURE HP_FIND_AGREEMENTS(
    REG_ID INTEGER)
RETURNS (
    REC_ID D_INTEGER,
    AG_ID D_INTEGER,
    AG_FROM_RD_ID D_INTEGER,
    AG_OBJECT_KIND D_INTEGER,
    AG_OBJECT_ID D_ID,
    AG_FROM_REG_ID D_INTEGER,
    AG_CAPTION D_STR_255,
    AG_AGREEMENT_NO VARCHAR(30),
    AG_AGREEMENT_DATE D_DATE,
    AG_DATE_BEGIN D_DATE,
    AG_DATE_END D_DATE,
    AG_MEMO D_STR_255,
    AG_STATUS D_INTEGER,
    REG_NAME D_STR_100,
    RD_NAME D_STR_255,
    OBJECT_NAME D_STR_255,
    KML_COUNT D_CURRENCY,
    KML_AMOUNT D_CURRENCY)
AS
declare variable reg_lst d_str_8100;
declare variable reg_lst2 d_str_8100;
begin
  rec_id = 0;
  reg_lst = (select list(id_list, ';') from region_list(:reg_id, 1, -1, 1));
  reg_lst2 = (select list(id_list, ';') from region_list(:reg_id, 1, 3, 1));

  for
    select
      ag_id,
      ag_from_rd_id,
      ag_object_kind,
      ag_object_id,
      ag_caption,
      ag_agreement_no,
      ag_agreement_date,
      ag_date_begin,
      ag_date_end,
      ag_memo,
      ag_status,
      ag_from_reg_id,
      rd_name,
      c_name_fio
    from agreements ag
      inner join refdata
        on ag_from_rd_id = rd_id
          and rd_code in ('AgreementKinds.IPF_Sellers', 'AgreementKinds.tsp_and_ipf', 'AgreementKinds.IPF_WithCommission')
      inner join clients
        on c_id = ag_object_id
    where ag_status = 1
        and (ag_date_end is null or ag_date_end >= current_date)
  into
    :ag_id,
    :ag_from_rd_id,
    :ag_object_kind,
    :ag_object_id,
    :ag_caption,
    :ag_agreement_no,
    :ag_agreement_date,
    :ag_date_begin,
    :ag_date_end,
    :ag_memo,
    :ag_status,
    :ag_from_reg_id,
    :rd_name,
    :object_name
  do
  begin
    if (not (reg_lst containing :ag_from_reg_id or
      exists(select * from region_list(:ag_from_reg_id, 1, -1, 1) where id_list = :reg_id)))
    then continue;

    for
      select
        kal_new_reg_id,
        sum(kml_count),
        sum(kml_total)
      from kontmat
        inner join kontract
          on k_id = kml_object_id
            and k_status >= 0
        inner join kontstate ks
          on ks_from_k_id = k_id
            and ks_ol_is_delete = 0
            and current_date between ks_date_begin and ks_date_end
            and ks_state = 0
        inner join product
          on pr_id = k_from_pr_id
            and pr_code not like '%.IDJARA.%'
        inner join knaktlist
          on kal_from_kf_id = k_from_kf_id
            and k_disb_date between kal_date and kal_date_end
            and :reg_lst2 containing kal_new_reg_id
            and knaktlist.is_delete = 0
      where kml_from_ag_id = :ag_id
        and (kml_bill_id is null or exists(select * from operlist where ol_id = kml_bill_id and operlist.is_delete = 1))
        and not exists(select * from kontmat_own where kmo_kml_id = kml_id)
      group by kal_new_reg_id
      having sum(kml_count) > 0
    into
      :ag_from_reg_id,
      :kml_count,
      :kml_amount
    do
    begin
      rec_id = rec_id + 1;
      reg_name = (
        select
          reg_code || ' ' || reg_name
        from region
        where reg_unicode = :ag_from_reg_id);
      suspend;
    end
  end
end
