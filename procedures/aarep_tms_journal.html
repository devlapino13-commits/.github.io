CREATE PROCEDURE AAREP_TMS_JOURNAL(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_ML_TEXT VARCHAR(100),
    REPORT_DEL_STATUS D_INTEGER = 1)
RETURNS (
    ML_NAME VARCHAR(255),
    OL_ID DECIMAL(18,0),
    OL_FROM_REG_ID INTEGER,
    OL_OPER_TYPE INTEGER,
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_DOC_NO_STR VARCHAR(20),
    OL_DOC_KIND INTEGER,
    OL_MEMO VARCHAR(1024),
    OL_STATUS SMALLINT,
    IS_DELETE SMALLINT,
    OL_DATETIME TIMESTAMP,
    OL_USER_ID DECIMAL(18,0),
    OL_DOCUMENT_CLIENT DECIMAL(18,0),
    OL_FROM_MO_ID DECIMAL(18,0),
    OL_IS_MO_MODIFIED SMALLINT,
    OL_DOCUMENT_OFFICE INTEGER,
    OL_ACC_NO_CLIENT INTEGER,
    OL_ACC_NO_OFFICE INTEGER,
    OL_OPER_SUBTYPE INTEGER,
    OL_DATE_OPER_DAY DATE,
    OL_POCKET INTEGER,
    OL_DOC_CLIENT_NAME VARCHAR(255),
    OL_IS_CLIENT SMALLINT,
    OL_YEAR INTEGER,
    OL_MONTH INTEGER,
    OM_FROM_OL_ID DECIMAL(18,0),
    OM_NUMBER INTEGER,
    OM_BILL_ID DECIMAL(18,0),
    OM_BILL_LINE_NO INTEGER,
    OM_OPER_TYPE INTEGER,
    OM_WORKPLACE_ID DECIMAL(18,0),
    OM_MATOTVET_ID DECIMAL(18,0),
    OM_SUMMA_PLUS DECIMAL(18,4),
    OM_SUMMA_MINUS DECIMAL(18,4),
    OM_KOLICH_PLUS DECIMAL(18,4),
    OM_KOLICH_MINUS DECIMAL(18,4),
    OM_NDS_PERCENT DECIMAL(15,2),
    OM_ACCOUNT VARCHAR(20),
    OM_IS_WRITTEN_OFF SMALLINT,
    OM_WORKPLACE_INFO DECIMAL(18,0),
    OM_MATOTVET_INFO DECIMAL(18,0),
    OM_ACCOUNT_INFO VARCHAR(20),
    OM_SUMMA_TOTAL DECIMAL(18,4),
    OM_MAIN_LINE SMALLINT,
    UN_NAME VARCHAR(30),
    DN_NAME VARCHAR(255),
    MO_NAME VARCHAR(100),
    RES_WP_REGION_PARENT VARCHAR(100),
    RES_WP_REGION VARCHAR(100),
    RES_WP_REGION_INFO VARCHAR(100),
    RES_OL_REGION VARCHAR(100),
    RES_WP_NAME VARCHAR(100),
    RES_WP_NAME_INFO VARCHAR(100),
    RES_MO_NAME VARCHAR(100),
    RES_MO_NAME_INFO VARCHAR(100),
    RES_OPER_TYPE VARCHAR(255),
    OL_PARENT_ID DECIMAL(18,0))
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable reg_parent integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* тестовый вариант для ускорения */
  /* 23.07.2012, Sanjari */
  for select ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo, ol_status,
             operlist.is_delete, ol_datetime, ol_user_id, ol_document_client, ol_from_mo_id, ol_is_mo_modified,
             ol_document_office, ol_acc_no_client, ol_acc_no_office, ol_oper_subtype, ol_date_oper_day, ol_pocket,
             ol_doc_client_name, ol_is_client, ol_year, ol_month, ol_parent_id, om_from_ol_id, om_number, om_bill_id,
             om_bill_line_no, om_oper_type, om_workplace_id, om_matotvet_id, om_summa_plus, om_summa_minus,
             om_kolich_plus, om_kolich_minus, om_nds_percent, om_account, om_is_written_off, om_workplace_info,
             om_matotvet_info, om_account_info, om_summa_total, om_main_line, un_name, dn_name, mo_name, r3.reg_name,
             opl_name, r1.reg_code || ' ' || r1.reg_name, wp_name, c_name_fio, ml_name, r1.parent
      from opermat, operlist, region_list(:report_region_id, 1, 3, 1), username, docname, mopocket, operbill, matlist, region r3, optlist, clients, workplace, region r1
      where om_from_ol_id = ol_id and
            ol_from_reg_id = id_list and
            mo_id = ol_pocket and
            operlist.is_delete = iif(:report_del_status = -1, operlist.is_delete, 1 - :report_del_status) and
            --operlist.is_delete = 0 and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            ol_user_id = un_id and
            ol_doc_kind = dn_id and
            om_main_line = 1 and
            om_bill_id = ob_bill_id and
            om_bill_line_no = ob_bill_line_no and
            ob_date_begin <= ol_date and
            ob_date_end >= ol_date and
            --ob_is_delete = 0 and
            ob_from_ml_id = ml_id and
            upper(ml_name) like '%' || :report_ml_text || '%' and
            r3.reg_status = 3 and
            r3.reg_unicode = ol_from_reg_id and
            ol_oper_type = opl_main and
            ol_oper_subtype = opl_detail and
            om_matotvet_id = c_id and
            om_workplace_id = wp_id and
            r1.reg_unicode = wp_from_reg_id
      order by ol_date, ol_from_reg_id, ol_id, om_number
      into :ol_id, :ol_from_reg_id, :ol_oper_type, :ol_date, :ol_doc_no, :ol_doc_no_str, :ol_doc_kind, :ol_memo,
           :ol_status, :is_delete, :ol_datetime, :ol_user_id, :ol_document_client, :ol_from_mo_id, :ol_is_mo_modified,
           :ol_document_office, :ol_acc_no_client, :ol_acc_no_office, :ol_oper_subtype, :ol_date_oper_day, :ol_pocket,
           :ol_doc_client_name, :ol_is_client, :ol_year, :ol_month, :ol_parent_id, :om_from_ol_id, :om_number,
           :om_bill_id, :om_bill_line_no, :om_oper_type, :om_workplace_id, :om_matotvet_id, :om_summa_plus,
           :om_summa_minus, :om_kolich_plus, :om_kolich_minus, :om_nds_percent, :om_account, :om_is_written_off,
           :om_workplace_info, :om_matotvet_info, :om_account_info, :om_summa_total, :om_main_line, :un_name, :dn_name,
           :mo_name, :res_ol_region, :res_oper_type, :res_wp_region, :res_wp_name, :res_mo_name, :ml_name, :reg_parent
  do
  begin
    if (om_workplace_info is not null) then
    begin
      select reg_code || ' ' || reg_name, wp_name
      from workplace, region
      where wp_id = :om_workplace_info and
            wp_from_reg_id = reg_unicode
      into :res_wp_region_info, :res_wp_name_info;
    end
    else
    begin
      res_wp_region_info = null;
      res_wp_name_info = null;
    end
    if (om_matotvet_info is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :om_matotvet_info
      into :res_mo_name_info;
    end
    else
      res_mo_name_info = null;

    if (res_wp_region_info is not null) then
    begin
      if (res_wp_region_info <> res_wp_region) then
      begin
        if (om_summa_plus <> 0) then
          res_wp_region_info = 'Откуда: ' || :res_wp_region_info;
        if (om_summa_minus <> 0) then
          res_wp_region_info = 'Куда: ' || :res_wp_region_info;
      end
      else
        res_wp_region_info = 'Тот же офис';
    end
    else
      res_wp_region_info = 'Тот же офис';

    if (res_wp_name_info is not null) then
    begin
      if (res_wp_name_info <> res_wp_name) then
      begin
        if (om_summa_plus <> 0) then
          res_wp_name_info = 'Откуда: ' || :res_wp_name_info;
        if (om_summa_minus <> 0) then
          res_wp_name_info = 'Куда: ' || :res_wp_name_info;
      end
      else
        res_wp_name_info = 'Тот же рабочее место';
    end
    else
      res_wp_name_info = 'Тот же рабочее место';
    if (res_mo_name_info is not null) then
    begin
      if (res_mo_name_info <> res_mo_name) then
      begin
        if (om_summa_plus <> 0) then
          res_mo_name_info = 'От кого: ' || :res_mo_name_info;
        if (om_summa_minus <> 0) then
          res_mo_name_info = 'Кому: ' || :res_mo_name_info;
      end
      else
        res_mo_name_info = 'Тот же сотрудник';
    end
    else
      res_mo_name_info = 'Тот же сотрудник';
    res_wp_region_parent = (select reg_name from region where reg_unicode = :reg_parent);
    suspend;
  end
end
