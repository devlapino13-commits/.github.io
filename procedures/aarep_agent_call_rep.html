CREATE PROCEDURE AAREP_AGENT_CALL_REP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    CK_CHECKED_DATE DATE,
    C_NAME_FIO STR_100,
    CALL_RESULT STR_100,
    CK_CHECKED_OPER STR_100,
    C_PLASTCARD_NO VARCHAR(20),
    CK_CHECKED_DATETIME TIMESTAMP)
AS
begin
  for select reg.reg_name, chk.ck_checked_date, chk.ck_checked_oper,

             cl.c_name_fio, rd.rd_name,
             (select cont_value
              from hp_contdata_get_cell_valid(0, chk.ck_from_c_id)),

             chk.ck_checked_datetime
      from cc_check_knclient chk
      inner join kontract kont on kont.k_id = chk.ck_from_k_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = kont.k_last_reg_id
      inner join region reg on reg.reg_unicode = id_list
      inner join clients cl on cl.c_id = chk.ck_from_c_id
      inner join refdata rd on rd.rd_id = chk.ck_status
      where chk.ck_type = 4 and
            chk.ck_checked_date between :rep_date_begin and :rep_date_end
      into :reg_name, :ck_checked_date, :ck_checked_oper, :c_name_fio, :call_result, :c_plastcard_no,
           :ck_checked_datetime
  do
  begin
    suspend;
  end
end
