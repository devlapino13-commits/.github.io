CREATE PROCEDURE HP_ADD_FP_VERIFY(
    KML_ID D_ID)
AS
declare variable MA_ID D_ID;
declare variable FPV_C_ID D_ID;
declare variable FPV_FP_ID D_ID;
declare variable FPV_FIO D_STR_255;
declare variable FPV_BIRTHDATE D_DATE;
declare variable FPV_PASS_NO D_STR_CODE;
declare variable FPV_TERR_CATEG_ID D_ID;
begin
  -- RFCKG-3647  
  select k.k_from_ma_app_id, ag.ag_object_id
  from kontmat kml
  inner join agreements ag on ag.ag_id = kml.kml_from_ag_id
  inner join kontract k on k.k_id = kml.kml_object_id
  where kml.kml_id = :kml_id and
        kml.kml_object_kind = 9 and
        ag.ag_object_kind = 0
  into ma_id, fpv_c_id;

  for select fs.fp_id,
             fs.c_name_fio,
             fs.c_birth_date,
             fs.c_pass_no,
             fs.terr_categ_id
      from hp_check_client_fio_ft_fuzzy(:fpv_c_id, '', '', '', null, '', 1) as fs
      where c_comment_id = 1
      into :fpv_fp_id,
           :fpv_fio,
           :fpv_birthdate,
           :fpv_pass_no,
           :fpv_terr_categ_id
  do
  begin
      insert into fp_verify (fpv_ma_id, fpv_c_id, fpv_fp_id, fpv_fio, fpv_birthdate, fpv_pass_no, fpv_terr_categ_id)
      values (:ma_id, :fpv_c_id, :fpv_fp_id, :fpv_fio, :fpv_birthdate, :fpv_pass_no, :fpv_terr_categ_id);
  end
end
