CREATE PROCEDURE UXP$UI_CLASS_PROPS_TMP(
    PROC_NAME D_STR_CODE)
RETURNS (
    FIELD_NAME D_STR_CODE,
    FIELD_SOURCE D_STR_CODE,
    PROP_NAME D_STR_CODE,
    PROP_TYPE D_STR_CODE)
AS
declare variable p d_integer;
declare variable sym d_str_sym;
declare variable len d_integer;
declare variable need_upper d_bool;
begin
  proc_name = upper(proc_name);

  for select lower(trim(p.rdb$parameter_name)), lower(trim(p.rdb$field_source))
  from rdb$procedure_parameters p
  where p.rdb$procedure_name = :proc_name and
        p.rdb$parameter_type = 1
  order by p.rdb$parameter_number
  into field_name, field_source
  do
  begin
    prop_name = '';
    len = char_length(field_name);
    p = 0;
    need_upper = 0;
    while (p < len) do
    begin
      p = p + 1;

      sym = substring(field_name from p for 1);
      if (need_upper = 1) then
        sym = upper(sym);

      need_upper = 0;
      if (sym = '_') then
        need_upper = 1;
      else
        prop_name = prop_name || sym;
    end

    prop_type = case field_source
      when 'd_bool' then 'Boolean'
      when 'd_currency' then 'Float'
      when 'd_date' then 'TDate'
      when 'd_datetime' then 'TDateTime'
      when 'd_double' then 'Float'
      when 'd_float' then 'Float'
      when 'd_id' then 'LargeInt'
      when 'd_integer' then 'Integer'
      when 'd_percent' then 'Float'
      when 'd_smallint' then 'Integer'
      when 'd_time' then 'TTime'
      when 'd_uuid' then 'TGUID'
      else 'String'
    end;

    suspend;
  end
end
