CREATE PROCEDURE HP_ANTIFROD_CHECK(
    K_ID D_ID)
AS
declare variable afc_id d_id;
declare variable jso d_integer;
declare variable jso_cat d_integer;
begin
  if (f_loan_is_online(k_id) = 0) then
    exit;  

  delete from antifrod_loanlist
  where all_from_afl_id = :k_id;

  update or insert into antifrod_loan (afl_from_k_id, afl_datetime, afl_risk_level, afl_content, afl_content_id)
  values (:k_id, current_timestamp, 0, null, null);

  select jso
  from hp_antifrod_get_calc(:k_id)
  into jso;

  update antifrod_loan
  set afl_risk_level = json.obj_get_val(:jso, 'risk_level', 0),
      afl_content = json.obj_to_json(:jso)
  where afl_from_k_id = :k_id;

  for select afc_id
  from antifrod_categories
  where afc_id > 0
  into afc_id
  do
  begin
    jso_cat = coalesce(json.obj_get_obj(jso, afc_id), json.obj_create(jso, afc_id));

    insert into antifrod_loanlist (all_from_afl_id, all_from_afc_id, all_result)
    values (:k_id, :afc_id, json.obj_get_val(:jso_cat, 'category_res', 0));
  end
end
