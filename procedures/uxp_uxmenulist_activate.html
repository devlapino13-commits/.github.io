CREATE PROCEDURE UXP$UXMENULIST_ACTIVATE(
    ML_ID D_INTEGER,
    ACTION_KIND D_SMALLINT)
AS
declare variable ml_ua_uuid d_uuid;
declare variable new_fs_disabled d_bool;
declare variable new_web_disabled d_bool;
declare variable uml_fs_disabled d_bool;
begin
  -- Активирует или отключает пункт меню из FinStart/Web

  if (action_kind not between -3 and 3) then
    exit;

  ml_ua_uuid = null;
  select uml_ua_uuid, uml_fs_disabled
  from ux$menulist
  where uml_ml_id = :ml_id
  into ml_ua_uuid, uml_fs_disabled;

  if (ml_ua_uuid is null) then
    exit;

  new_fs_disabled = null;
  new_web_disabled = null;

  if (action_kind = -1 or action_kind = -3) then
    new_fs_disabled = 1;
  else
  if (action_kind = 1 or action_kind = 3) then
    new_fs_disabled = 0;

  if (action_kind = -2 or action_kind = -3) then
    new_web_disabled = 1;
  else
  if (action_kind = 2 or action_kind = 3) then
    new_web_disabled = 0;
  else
  if (action_kind = 0) then
    new_web_disabled = uml_fs_disabled;

  update ux$menulist
  set uml_fs_disabled = coalesce(:new_fs_disabled, uml_fs_disabled),
      uml_web_disabled = coalesce(:new_web_disabled, uml_web_disabled)
  where uml_ml_id = :ml_id and
        (uml_fs_disabled is distinct from coalesce(:new_fs_disabled, uml_fs_disabled) or
         uml_web_disabled is distinct from coalesce(:new_web_disabled, uml_web_disabled));
end
