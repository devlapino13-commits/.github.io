CREATE PROCEDURE HP_DO_ANKETASYNC(
    RD_CODE VARCHAR(100),
    OBJECT_KIND INTEGER,
    OBJECT_ID DECIMAL(18,0),
    RES_RD_ID INTEGER,
    RES_VALUE D_STR_2048,
    UN_ID DECIMAL(18,0))
AS
declare variable sql_text varchar(255);
declare variable object_kind_name varchar(100);
begin
   if (rd_code = 'AnketaQuestions.Family' and
      object_kind = 0 and (not exists(select * from grs_clients gc
                                               where gc.gc_from_c_id = :object_id and
                                                     gc.gc_status in (2,3) and
                                                     gc.gc_is_last = 1) or
                               exists(select * from clients c
                                     where c.c_id = :object_id and
                                           c.c_family = 0))) then
  begin
    sql_text = 'update clients set c_family=' || :res_rd_id || ' where c_id=' || :object_id || ' and coalesce(c_family,-777)<>' || :res_rd_id;
    execute statement :sql_text;
    when any do
    begin
      insert into fsrobot_eventlist (fel_event_name, fel_command_kind, fel_params, fel_from_un_id)
      values ('Exec_SQL', 'ClientSync', :sql_text, :un_id);
    end
  end

  if (rd_code = 'AnketaQuestions.Education' and
      object_kind = 0) then
  begin
    sql_text = 'update clients set c_education=' || :res_rd_id || ' where c_id=' || :object_id || ' and coalesce(c_education,-777)<>' || :res_rd_id;
    execute statement :sql_text;
    when any do
    begin
      insert into fsrobot_eventlist (fel_event_name, fel_command_kind, fel_params, fel_from_un_id)
      values ('Exec_SQL', 'ClientSync', :sql_text, :un_id);
    end
  end

  if (rd_code = 'AnketaQuestions.Nation' and
      object_kind = 0) then
  begin
    sql_text = 'update clients set c_nation=' || :res_rd_id || ' where c_id=' || :object_id || ' and coalesce(c_nation,-777)<>' || :res_rd_id;
    execute statement :sql_text;
    when any do
    begin
      insert into fsrobot_eventlist (fel_event_name, fel_command_kind, fel_params, fel_from_un_id)
      values ('Exec_SQL', 'ClientSync', :sql_text, :un_id);
    end
  end

  if (rd_code = 'AnketaQuestions.InfoSource' and
      object_kind = 0) then
  begin
    sql_text = 'update clients set c_from_is_id=' || :res_rd_id || ' where c_id=' || :object_id || ' and coalesce(c_from_is_id,-777)<>' || :res_rd_id;
    execute statement :sql_text;
    when any do
    begin
      insert into fsrobot_eventlist (fel_event_name, fel_command_kind, fel_params, fel_from_un_id)
      values ('Exec_SQL', 'ClientSync', :sql_text, :un_id);
    end
  end

end
