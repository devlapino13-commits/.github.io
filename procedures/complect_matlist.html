Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE COMPLECT_MATLIST(
    CPL_ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    MBAL_BILL_ID DECIMAL(18,0),
    MBAL_BILL_LINE_NO INTEGER,
    MBAL_WP_ID DECIMAL(18,0),
    MBAL_MO_ID DECIMAL(18,0),
    MBAL_ML_ID DECIMAL(18,0),
    MBAL_KOLICH DECIMAL(18,4),
    MBAL_SUMMA DECIMAL(18,4),
    MBAL_BILL_COUNT INTEGER,
    MBAL_CPL_ID DECIMAL(18,0),
    MBAL_MC_ID DECIMAL(18,0),
    MBAL_INVENT_NO INTEGER,
    MBAL_INVENT_STR VARCHAR(50),
    MBAL_FULL_DETAIL VARCHAR(2048),
    MBAL_ACCOUNT VARCHAR(20))
AS
declare variable reg integer;
begin
  if (cpl_id is not null) then
  begin
    for select ob_complect_id, om_bill_id, om_bill_line_no, om_workplace_id, om_matotvet_id, ob_from_ml_id,
               ob_from_mc_id, ob_from_reg_id, ob_invent_no, ob_invent_str, ob_full_detail, om_account,
               sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) as kol_balance,
               sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) as sum_balance
        from opermat, operlist, operbill
        where om_from_ol_id = ol_id and
              ob_complect_id = :cpl_id and
              ob_bill_id = om_bill_id and
              ob_bill_line_no = om_bill_line_no and
              ob_date_begin <= :dd and
              ob_date_end >= :dd and
              is_delete = 0 and
              ol_date <= :dd and
              ob_is_delete = 0
        group by ob_complect_id, om_bill_id, om_bill_line_no, om_workplace_id, om_matotvet_id, ob_from_ml_id, ob_from_mc_id, ob_from_reg_id, ob_invent_no, ob_invent_str, ob_full_detail, om_account
        having sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0 or sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0
        order by om_workplace_id, om_matotvet_id, ob_from_ml_id, om_bill_id, om_bill_line_no
        into :mbal_cpl_id, :mbal_bill_id, :mbal_bill_line_no, :mbal_wp_id, :mbal_mo_id, :mbal_ml_id, :mbal_mc_id, :reg,
             :mbal_invent_no, :mbal_invent_str, :mbal_full_detail, :mbal_account, :mbal_kolich, :mbal_summa
    do
    begin
      if (mbal_account is null) then
        mbal_account = '';
      if (mbal_kolich is null) then
        mbal_kolich = 0;
      if (mbal_summa is null) then
        mbal_summa = 0;
      if (mbal_cpl_id is null) then
        mbal_cpl_id = 0;
      mbal_bill_count = 1;
      if ((mbal_kolich <> 0) or (mbal_summa <> 0)) then
        suspend;
    end
  end
end
