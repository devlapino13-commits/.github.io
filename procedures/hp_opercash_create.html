CREATE PROCEDURE HP_OPERCASH_CREATE(
    IN_OFFICE_ID D_INTEGER,
    IN_CASH_ID D_INTEGER,
    IN_DATE D_DATE,
    IN_OPER_SIGN D_SMALLINT,
    IN_OPER_KIND D_STR_100,
    IN_K_ID D_ID,
    IN_MANAGER_ID D_ID,
    IN_UN_ID D_ID)
RETURNS (
    OPC_ID D_INTEGER,
    OPC_DATE D_DATE,
    OPC_OFFICE_ID D_INTEGER,
    OPC_FROM_CSP_ID D_INTEGER,
    OPC_OPER_KIND D_INTEGER,
    OPC_SIGN D_SMALLINT,
    OPC_AMOUNT D_CURRENCY,
    OPC_AMOUNT$BOND D_STR_LONG,
    OPC_FROM_C_ID D_ID,
    OPC_IS_IPF D_BOOL,
    OPC_CORR_AL_ID D_STR_20,
    OPC_FROM_K_ID D_ID,
    OPC_CORR_REG_ID D_INTEGER,
    OPC_CARD_NO D_STR_20,
    OPC_FROM_AG_ID D_INTEGER,
    OPC_MEMO D_STR_LONG,
    OPC_FROM_UN_ID D_ID,
    OPC_ADD_INFO D_STR_255,
    RES_ADD_AMOUNT D_CURRENCY,
    RES_ADD_AMOUNT$BOND D_STR_LONG,
    RES_IPF_LOCKED D_BOOL,
    RES_OPER_KIND_LOCKED D_BOOL,
    RES_COMMENT D_STR_LONG,
    RES_KONTRACT_NAME D_STR_100,
    RES_K_CONS_KIND D_SMALLINT,
    RES_IS_DEFERRED_PAYMENT D_BOOL,
    CIL_FROM_BLS_ID D_ID,
    RES_IS_EDIT D_BOOL,
    OPC_OTHER_CLIENT D_STR_255,
    OPC_OTHER_CLIENT_ID D_ID)
AS
declare variable oper_kind_code d_str_code;
declare variable opc_parent_id d_integer;
declare variable opc_from_cil_id d_integer;
declare variable opc_from_edl_id d_id;
declare variable opc_from_ol_id d_id;
declare variable opc_is_delete d_bool;
declare variable opc_orig_memo d_str_long;
declare variable cfl_id d_integer;
declare variable opc_manager_id d_id;
begin
  opc_id = null;
  res_is_edit = 0;
  opc_other_client = '';
  if (in_oper_kind = 'edit') then
  begin
    res_is_edit = 1;

    select opc_id, opc_parent_id, opc_date, opc_office_id, opc_from_csp_id, opc_oper_kind, opc_sign, opc_amount,
           opc_from_c_id, opc_is_ipf, opc_corr_al_id, opc_from_k_id, opc_corr_reg_id,
           opc_card_no, opc_from_ag_id, opc_from_cil_id, opc_memo, opc_from_edl_id,
           opc_from_ol_id, opc_is_delete, opc_add_info, opc_manager_id, opc_other_client, opc_other_client_id
    from opercash
    where opc_id = :in_cash_id
    into :opc_id, :opc_parent_id, :opc_date, :opc_office_id, :opc_from_csp_id, :opc_oper_kind, :opc_sign, :opc_amount,
         :opc_from_c_id, :opc_is_ipf, :opc_corr_al_id, :opc_from_k_id, :opc_corr_reg_id,
         :opc_card_no, :opc_from_ag_id, :opc_from_cil_id, :opc_memo, :opc_from_edl_id,
         :opc_from_ol_id, :opc_is_delete, :opc_add_info, opc_manager_id, opc_other_client, opc_other_client_id;

    if (in_manager_id is null) then
      in_manager_id = opc_manager_id;

    if (row_count = 0) then
      exception e_message 'Документ не найден, Id: ' || coalesce(in_cash_id, '<Null>');

    opc_from_un_id = in_un_id;

    if (opc_parent_id is not null) then
      exception e_message 'Редактировать необходимо главный документ';

    if (opc_from_ol_id is not null) then
      exception e_message 'Документ ранее проведен. Запрет на редактирование';

    if (opc_is_delete = 1) then
      exception e_message 'Документ ранее аннулирован. Запрет на редактирование';

    if (opc_from_edl_id is not null) then
    begin
      -- Электронный документ
      if (exists(select * from edoc_list
                 where edl_id = :opc_from_edl_id and
                       edl_status = 1)) then
        exception e_message 'Электронный документ уже подписан, сначала аннулируйте его';
    end

    opc_orig_memo = opc_memo;
    oper_kind_code = f_enum_code('cashoperkind', opc_oper_kind, 0);

    res_add_amount = 0;

    if (opc_from_cil_id is not null) then
    begin
      -- загрузка купью
      select res_bond_list
      from hp_cashbond_load(:opc_from_cil_id)
      into :opc_amount$bond;

      select cil_from_bls_id
      from cashinvlist
      where cil_id = :opc_from_cil_id
      into cil_from_bls_id;

      if (oper_kind_code = 'plus.from_bank') then
      begin
        select cil_id, cil_inv_amount, coalesce(:cil_from_bls_id, cil_from_bls_id)
        from cashinvlist
        where cil_parent_id = :opc_from_cil_id
        rows 1
        into cfl_id, res_add_amount, cil_from_bls_id;

        if (cfl_id is not null and res_add_amount <> 0) then
        begin
          select res_bond_list
          from hp_cashbond_load(:cfl_id)
          into :res_add_amount$bond;
        end
      end
    end

--    , res_add_amount, res_add_amount$bond, res_ipf_locked, res_oper_kind_locked, res_comment, res_kontract_name, cil_from_bls_id

    in_office_id = opc_office_id;
    in_cash_id = opc_from_csp_id;
    in_date = opc_date;
    in_oper_sign = opc_sign;
    in_oper_kind = opc_oper_kind;
    in_k_id = opc_from_k_id;
    res_oper_kind_locked = 1;
    res_is_deferred_payment = 0;

    select k_cons_kind
    from kontract
    where k_id = :in_k_id
    into res_k_cons_kind; -- RFCKG-4900

    if (res_k_cons_kind is null) then
      res_k_cons_kind = 0;
  end else
  begin
    if (in_oper_sign not in (-1, 1)) then
      exception e_message 'Неверный признак операции';
  
    if (in_cash_id is null) then
      exception e_message 'Касса не указана';
  
    if (in_manager_id is null) then
      exception e_message 'Контролер не указан';
  
    if (in_date is null) then
      in_date = current_date;
  
    if (in_office_id is not null) then
      opc_office_id = in_office_id;
    else
    begin
      select csp_office_id
      from cashplace
      where csp_id = :in_cash_id
      into opc_office_id;
    end
  
    opc_date = in_date;
    opc_from_csp_id = in_cash_id;
    opc_sign = in_oper_sign;
    opc_from_c_id = null;
    opc_corr_al_id = null;
    opc_from_k_id = in_k_id;
    opc_corr_reg_id = opc_office_id;
    opc_card_no = '';
    opc_from_ag_id = null;
    opc_memo = '';
    opc_amount$bond = '';
    opc_add_info = null;
    cil_from_bls_id = null;
    res_add_amount$bond = '';
    opc_from_un_id = in_un_id;
  
    if (position(',', in_oper_kind) = 0) then
      opc_oper_kind = f_enum_id('cashoperkind', in_oper_kind, 0);
    else
    begin
      if (in_oper_kind containing 'plus.loan.repay') then
        opc_oper_kind = f_enum_id('cashoperkind', 'plus.loan.repay', 0);
    end
  
    if (opc_oper_kind is null) then
      opc_oper_kind = 0;
  
    res_oper_kind_locked = 0;
  
    opc_is_ipf = 0;
    opc_amount = 0;
    res_add_amount = 0;
    res_ipf_locked = 0;
    res_k_cons_kind = 0;
    res_is_deferred_payment = 0;
    res_kontract_name = '';
  
    res_comment = '';
  
    if (opc_oper_kind = 0) then
    begin
      res_comment = 'Выберите вид операции';
      suspend;
      exit;
    end

    oper_kind_code = f_enum_code('cashoperkind', opc_oper_kind, 0);
  end

  if (in_k_id is not null) then
  begin
    select k_cons_kind
    from kontract
    where k_id = :in_k_id
    into res_k_cons_kind;

    if (res_k_cons_kind is null) then
      res_k_cons_kind = 0;

    if (oper_kind_code = 'minus.loan.disb') then
    begin
      for select k_disb_date, k_amount_approved, k_status, k_kontract_no, k_clients_cashier_id
      from kontract k
      where k_id = :in_k_id
      as cursor cur
      do
      begin
        if (cur.k_disb_date is distinct from in_date) then
          exception e_message 'Договор займа ' || cur.k_kontract_no || ', установленная дата выдачи: ' || f_datetostr(cur.k_disb_date);

        if (cur.k_status <> 1) then
          exception e_message 'Договор займа ' || cur.k_kontract_no || ' не еще утвержден';

        opc_amount = cur.k_amount_approved;
        res_kontract_name = cur.k_kontract_no || ' от ' || f_datetostr(cur.k_disb_date);
      end

      select sh_kommis
      from schedule
      where sh_from_k_id = :in_k_id and
            sh_id = 0
      into res_add_amount;

      if (res_add_amount > 0) then
        res_comment = 'Будет сформирован расходный ордер и приходный ордер для доп.процента';
    end
  end else
  if (oper_kind_code = 'minus.to_bank') then
  begin
    select res_def_amount, res_kont_comment, res_def_memo
    from hp_opercash_default_info(:opc_from_csp_id, :in_date, :oper_kind_code, :opc_is_ipf)
    into :opc_amount, :res_comment, :opc_memo;
  end else
  if (oper_kind_code = 'plus.from_bank') then
  begin
    if (in_manager_id is null) then
      exception e_message 'Контролер не указан';

    select res_def_memo
    from hp_opercash_default_info(:opc_from_csp_id, :in_date, :oper_kind_code, :opc_is_ipf)
    into :opc_memo;
  end

  if (res_is_edit = 1 and coalesce(opc_orig_memo, '') <> '') then
    opc_memo = opc_orig_memo;

  if (exists(select *
             from hp_is_day_closed(:opc_office_id, :opc_date, 1)
             where day_closed = 1)) then
    exception e_message 'Опер.день кассы закрыт: ' || f_datetostr(opc_date);

  suspend;
end
