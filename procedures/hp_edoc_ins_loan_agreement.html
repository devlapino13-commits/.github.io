CREATE PROCEDURE HP_EDOC_INS_LOAN_AGREEMENT(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable PR_CODE D_STR_100;
declare variable MA_ID D_ID;
declare variable K_STATUS D_SMALLINT;
declare variable KS_STATE D_SMALLINT;
declare variable APP_DATE D_DATE;
declare variable DB_ID D_INTEGER;
declare variable KMA_ID D_ID;
declare variable NEED_CHECK D_SMALLINT;
declare variable AGENT_C_ID D_ID;
declare variable BENEFIC_C_ID D_ID;
declare variable AG_ID D_ID;
declare variable CDS_DOC_BLOB_1 D_ID;
declare variable CDS_DOC_BLOB_2 D_ID;
declare variable K_DISB_KIND D_INTEGER;
declare variable C_ID D_ID;
declare variable K_DISB_DATE D_DATE;
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable REG_ID D_INTEGER;
declare variable SIGN_EDOC_ZALOG D_SMALLINT;
declare variable KC_PDN_KOEFF D_PERCENT;
declare variable APPROVE_DATETIME D_DATETIME;
declare variable INSURANCE_MIN_AMOUNT D_INTEGER;
declare variable PROHIBITION_STATUS D_INTEGER;
declare variable POINT D_INTEGER;
declare variable INSURANCE_REQUIRED D_SMALLINT;
declare variable INSURANCE_COMMISSION D_CURRENCY;
declare variable INSURANCE_OK_AMOUNT D_CURRENCY;
declare variable INSURANCE_C_ID D_ID;
declare variable MA_INSUR_KIND D_SMALLINT;
declare variable K_KONTRACT_DATE D_DATE;
begin
  -- Процедура создает электронный документ по займу.
  -- Готовит бланки и отправляет СМС.
  -- Пследние изменения были внесены согласно задаче RFCRU-3027. Доработали блок на видео-сертификацию.
  res_text = '';
  db_id = f_get_db_id();
  sign_edoc_zalog = f_get_const('sign_edoc_zalog'); -- подписание edoc залогодателя проект RFCRU -1182
  select k_from_ma_app_id, pr_code, k_status, k_disb_kind, k_disb_date, k_last_reg_id, k_amount_approved, k_clients_cashier_id, k_kontract_date
  from kontract k, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into ma_id, pr_code, k_status, k_disb_kind, k_disb_date, reg_id, k_amount_approved, c_id, k_kontract_date;

  if (f_tmp_const('edoc_ipf_inactive') = 1) then -- если по ИФП eDoc отключен
  begin
    -- по ИФП пока не формируем эл.документ
    if (position('.ipf.', pr_code) > 0) then
      exit;
  end

  if (db_id = 41701 and
      f_get_const('self_prohibition') = 1) then
  begin
    for select knc.kc_from_c_id
          from knclient knc
          where knc.kc_from_k_id = :k_id and
                knc.kc_amount_approved > 0
          into :c_id
    do
    begin
      select res
      from hp_self_prohibition_get_status(:c_id, 'kontract', :k_id)
      into :prohibition_status;

      res_text = case prohibition_status
                   when 0  then 'Формирование документа невозможно, запрос в сервис Самозапрет в обработке'
                   when -1 then 'Формирование документа невозможно, ошибка в запросе в сервис Самозапрет'
                   when -2 then 'Формирование документа невозможно, требуется прикрепить скриншот с Тундука по отсутствию самозапрета'
                   when 1  then 'Формирование документа невозможно, у клиента действует Самозапрет'
                   when 2 then ''
                   else 'Формирование документа невозможно, запрос не осуществлялся'
                 end;
    end
  end

  if (db_id = 41701 and
      pr_code not containing '.ignore_insurance.') then
  begin
    if (f_get_const('new_loan_conditions') = 1) then
      insurance_min_amount = cast(f_get_const('kont_new_insurance_min_amount') as d_integer);
    else
      insurance_min_amount = cast(f_get_const('kont_insurance_min_amount') as d_integer);

    if (pr_code containing '.ipf.') then
      insurance_min_amount = cast(f_get_const('kont_insurance_ipf_min_amount') as d_integer);

    -- RFCKG-3653
    if (f_get_const('kont_insurance_active') = 1
    and pr_code containing '.IND.'
    and (pr_code containing '.STANDART.' or
         pr_code containing '.IPF.') -- RFCKG-5348
    and pr_code not containing '.consultation.'
    and pr_code not containing '.insur_kontrakt.'   -- RFCKG-4831
    and k_amount_approved >= :insurance_min_amount
    and not( f_edoc_status('insurance.application', :k_id, k_kontract_date) = 'signed' or -- RFCKG-5466
             f_edoc_status('insurance.kyrgyz', :k_id, k_kontract_date) = 'signed' or
             f_edoc_status('insurance.kyrgyz.ipf', :k_id, k_kontract_date) = 'signed')
    ) then
    begin
      res_text = 'Пакет документов по страхованию не подписан';
      suspend;
      exit;
    end

    if (f_get_const('kont_insurance_active') = 1
    and f_get_const('new_loan_conditions') = 1
    and pr_code containing '.IND.'
    and (pr_code containing '.STANDART.' or
         pr_code containing '.IPF.') -- RFCKG-5348
    and pr_code containing '.consultation.'
    and pr_code containing '.INSUR.'
    and pr_code not containing '.insur_kontrakt.'   -- RFCKG-4831
    and k_amount_approved >= :insurance_min_amount
    and not( f_edoc_status('insurance.application', :k_id, k_kontract_date) = 'signed' or
          f_edoc_status('insurance.kyrgyz', :k_id, k_kontract_date) = 'signed' or
          f_edoc_status('insurance.kyrgyz.ipf', :k_id, k_kontract_date) = 'signed')
    ) then
    begin
      for select kc.kc_from_c_id, kn.k_disb_date, kn.k_amount_approved
      from kontract kn
      inner join knclient kc on kc.kc_from_k_id = kn.k_id
      where kn.k_id = :k_id and
            kc.kc_amount_approved > 0
      into :c_id,  k_disb_date, k_amount_approved
      do
      begin
        point = coalesce(f_point(:c_id, :k_disb_date), 0);

        if (point < 20) then
          -- Баллы 0–19 страховка ОБЯЗАТЕЛЬНО
          insurance_required = 1;
        else
          -- Баллы 20 и выше > смотрим сумму кредита
          insurance_required = iif(k_amount_approved >= 250000, 1, 0);

        if (insurance_required = 1) then
        begin
          res_text = 'Пакет документов по страхованию не подписан';
          suspend;
          exit;
        end
      end
    end

    select coalesce(res_text, '')  -- RFCKG-4988
    from hp_check_ins_commission_amount(:k_id)
    into :res_text;

    -- RFCKG-5348
    -- Проверка оплаты страхования перед формированием договора финансирования ИПФ
    if (res_text = '') then
    begin
      if (f_get_const('kont_insurance_active') = 1) then
      begin
        select first 1 ma.ma_insur_kind
        from kontract k
        inner join miniapp ma on k.k_from_ma_app_id = ma.ma_id
        where k.k_id = :k_id and
              ma.ma_is_active = 1 and
              ma.ma_date_end >= '21.12.2999'
        into :ma_insur_kind;

        for select cl.c_id,
                   ik.ik_commission
            from kontract k
            inner join knclient kc on k.k_id = kc.kc_from_k_id
            inner join product p on k.k_from_pr_id = p.pr_id
            inner join clients cl on cl.c_id = kc.kc_from_c_id
            inner join insurance_kontract ik on ik.ik_from_k_id = k.k_id
            where k.k_id = :k_id and
                  k.k_amount_approved >= :insurance_min_amount and
                  kc.kc_amount_approved > 0 and
                  p.pr_code containing '.ind.' and
                  (
                    (p.pr_code containing '.standart.') or -- RFCKG-5388
                    (p.pr_code containing '.ipf.' and -- RFCKG-5442
                     p.pr_code not containing '.ignore_insurance.' and
                     :ma_insur_kind = f_enum_id('ma_insur_payment_kind', 'now')
                    )
                  ) and -- RFCKG-5388
                  p.pr_code not containing '.consultation.' and
                  p.pr_code not containing '.insur_kontrakt.' and -- RFCKG-4831
                  p.pr_code not containing '.insur_base.'
            into :insurance_c_id,
                 :insurance_commission
        do
        begin
          insurance_ok_amount = 0;

          select sum(ok.ok_amount)
          from operkontract ok
          inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
          where ok.ok_from_k_id = :k_id and
                ok.ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance') and
                ol.is_delete = 0 and
                ol.ol_document_client = :insurance_c_id
          into :insurance_ok_amount;

          insurance_ok_amount = coalesce(insurance_ok_amount, 0);

          if (insurance_ok_amount = 0) then
          begin
            res_text = 'Оплата за страхование не произведена';
            suspend;
            exit;
          end
          else
          if (insurance_ok_amount < insurance_commission) then
          begin
            res_text = 'Оплата за страхование произведена не в полном объеме. Цена за страхование = ' || round(insurance_commission) || ' сом . Оплачено: ' || round(insurance_ok_amount) || ' сом';
            suspend;
            exit;
          end
          else
          if (insurance_ok_amount > insurance_commission) then
          begin
            res_text = 'Оплата за страхование произведена в большем объеме. Цена за страхование = ' || round(insurance_commission) || ' сом . Оплачено: ' || round(insurance_ok_amount) || ' сом';
            suspend;
            exit;
          end
        end
      end
    end
  end


  if (pr_code containing '.consultation.' and
      db_id in (39801, 41701)) then
    select res_text
    from fm_consult_link_check(:k_id)
    into :res_text;

  if (k_status <> 1) then
    res_text = 'Сначала утвердите договор займа';

  if (res_text = '') then
  begin
    select ks_state
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999'
    into ks_state;

    if (ks_state = 1) then
      res_text = 'Займ уже действующий';

    if (ks_state = 2) then
      res_text = 'Займ уже закрыт';
  end

  if (db_id = 39801 and res_text = '') then
    select res_text
    from hp_afm_check_status(:k_id)
    into :res_text;

  -- До формирование договора займа, проверяем наличе подписанной заявки
  if (res_text = '') then
  begin
    select ma_date_app
    from miniapp
    where ma_id = :ma_id
    order by ma_id desc, ma_number desc
    rows 1
    into app_date;

    res_text = f_edoc_check('loan.application', ma_id, app_date);
  end

  -- RFCKZ-805
  if (res_text = '' and
      db_id = 39801 and
      f_get_const('kid_active') = 1) then
  begin
    if (f_edoc_status('loan.kid', k_id, k_disb_date) <> 'signed') then
      res_text = 'Клиент не подписал документы КИД и проект договора.';
  end

  --Если отправляется кредитный договор по мурабаха, то проверяем подписан ли акт приёма-передачи
  --Если акт приёма-передачи не подписан, то выдаём ошибку о том что нужно сперва подписать акт приёма-передачи
  if (db_id in (39801, 41701) and
      res_text = '' and
      (pr_code containing '.murabaha.' or
       pr_code containing '.idjara_goods.')) then
  begin
    for select kma.kma_id, kma.kma_from_ag_id
        from kontmatag kma
        where kma.kma_from_k_id = :k_id
        into kma_id, ag_id
    do
    begin
      need_check = 1;

      agent_c_id = null;
      select ag.ag_object_id
      from agreements ag
      where ag.ag_id = :ag_id
      into agent_c_id;

      if ((select c_type
           from clients
           where c_id = :agent_c_id) = 1) then
      begin
        benefic_c_id = null;
        select cb.cb_ben_from_c_id
        from clientbenefic cb
        where cb.cb_from_c_id = :agent_c_id and
              current_date between cb.cb_date_begin and coalesce(cb.cb_date_end, '31.12.9999')
        into benefic_c_id;

        if (coalesce(benefic_c_id, 0) <> 0) then
        begin
          if (f_asp_signed(:benefic_c_id, current_date) <> 1) then
            need_check = 0;
        end
        else
        begin
          if (f_asp_signed(:agent_c_id, current_date) <> 1) then
            need_check = 0;
        end
      end

      if (exists(
        select *
        from agreements
          inner join agreement_esign_access
            on aea_from_ag_id = ag_id
              and not aea_sign
        where ag_object_id = :agent_c_id
          and ag_from_reg_id = :reg_id
          and current_date between ag_date_begin and coalesce(ag_date_end, '31.12.9999')
          and ag_status = 1))
      then
        need_check = 0;

      if ((need_check = 1)
        and exists(
          select *
          from agreements
            inner join agreement_esign_access
              on aea_from_ag_id = ag_id
                and aea_sign
          where ag_object_id = :agent_c_id
            and current_date between ag_date_begin and coalesce(ag_date_end, '31.12.9999')
            and ag_status = 1)
        and not exists(
          select *
          from edoc_list edl
          where edl.edl_object_kind = f_objkind('kontmatag', 'id')
            and edl.edl_object_id = :kma_id
            and edl.edl_from_edc_id = (
              case
                when :pr_code containing '.murabaha.' then f_edoc_kind('purchase_act.agreement', 'id')
                when :pr_code containing '.idjara_goods.' then f_edoc_kind('idjara_goods.contragent_act', 'id')
              else null end)
            and edl.edl_status = 1)) then
      begin
        res_text = 'Отправка эл. кредитного договора невозможна. Сначала подпишите акт приёма-передачи товара.';
        break;
      end
    end
  end

  if (res_text = '') then
  begin
    select res_text
    from hp_miniapp_check_goods_on_stock(:ma_id)
    into :res_text;
  end

  if (db_id = 81001) then
  begin
    if (res_text = '' and k_disb_kind in (7, 8)) then
    begin
      for select knc.kc_from_c_id
          from knclient knc
          where knc.kc_from_k_id = :k_id and
                knc.kc_amount_approved > 0
          into :c_id
      do
      begin
  
        cds_doc_blob_1 = 0;
        cds_doc_blob_2 = 0;

        select kl_datetime
        from kontlog
        where kl_from_k_id = :k_id and
              kl_state = 1
        order by kl_datetime rows 1
        into approve_datetime;
  
        select cds.cds_doc_blob_1, cds.cds_doc_blob_2
        from clientdocs cds, passkind, passdoctype
        where cds_from_c_id = :c_id and
              cds_from_pk_id = pk_id and
              cds_is_delete = 0 and
              pk_from_pdt_id = pdt_id and
              cast(cds_create_datetime as date) >= coalesce(cast(:approve_datetime as date), current_date) and
              pdt_code in ('video_sertif')
        order by cds.cds_create_datetime desc
        rows 1
        into :cds_doc_blob_1, cds_doc_blob_2;
  
        if ((coalesce(cds_doc_blob_1, 0) + coalesce(cds_doc_blob_2, 0) = 0)) then
        begin
          res_text = 'Не закреплено фото в слоте "Видео-сертификация"';
          break;
        end
      end
    end

    select kcp.kc_pdn_koef
    from knclient kc, knclient_pdn kcp
    where kc.kc_from_k_id = :k_id and
          kc.kc_amount_approved > 0 and
          kc.kc_id = kcp.kc_id and
          kcp_kind = 0
    into kc_pdn_koeff;

    if (coalesce(kc_pdn_koeff, 0) > 50) then -- уведомление о ПДН проверяется только если ПДН > 50 (RFCRU-1784)
      if (res_text = '' and f_edoc_state2('loan.pdn', k_disb_date, un_id) = 'active' and
          f_edoc_status('loan.pdn', k_id, k_disb_date) <> 'signed') then
         res_text = 'Сначала нужно чтобы клиент подписал: Уведомление о рассчитанном ПДН';
  end

  if (res_text = '') then
  begin
    -- PM-294.  Дата договора это дата создания бланка (договор должен быть подписан этим числом)   RFCKZ-179 для всех стран
    update kontract
    set k_kontract_date = f_get_kontract_date(:k_id, current_date)
    where k_id = :k_id;

    select edl_id, res_text
    from hp_edoc_insert('loan.agreement', f_objkind('kontract', 'id'), :k_id, :un_id)
    into edl_id, res_text;

    if (res_text = '') then
    begin
      -- если есть залогодатель, тогда создаем
      if ((sign_edoc_zalog = 1) and (exists(select *

                 from zalog
                 where z_from_k_id = :k_id and
                       z_from_c_id is not null and
                       z_type = 1))) then
      begin
        select edl_id, res_text
        from hp_edoc_insert('loan.collateral', f_objkind('kontract', 'id'), :k_id, :un_id)
        into :edl_id, :res_text;
      end
    end
    if (res_text = '' and
        db_id in (39801, 41701)) then
    begin
      if (exists(select *
                 from kontract k
                 inner join product pr on k.k_from_pr_id = pr.pr_id and
                       pr.pr_code containing '.ipf.'
                 where k.k_id = :k_id)) then
      begin
        if (exists(select *
                   from zalog
                   where z_from_k_id = :k_id and
                         z_from_c_id is not null)) then
        begin
          select edl_id, res_text
          from hp_edoc_insert('loan.poruch.agreement', f_objkind('kontract', 'id'), :k_id, :un_id)
          into :edl_id, :res_text;
        end
      end
    end

    if (res_text = '' and
        db_id = 81001) then
    begin
      begin
        execute procedure hp_bki_add_to_buffer('.hp_edoc_ins_loan_agreement.', 'kontract', :k_id, current_date);
      end
    end 

  end
  suspend;
end
