CREATE PROCEDURE UXP$UXFRITEM_UPDATE(
    UFI_UUID_STR D_STR_CODE,
    UFI_PARENT_UUID_STR D_STR_CODE,
    UFI_OWNER D_UUID,
    UFI_KIND_STR D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_NAME D_STR_100,
    UFI_CAPTION_ALIGN D_SMALLINT,
    UFI_CAPTION_VISIBLE D_BOOL,
    UFI_HIDDEN D_BOOL,
    UFI_GROUP_DIRECTION D_SMALLINT,
    UFI_GROUP_BORDER_VISIBLE D_BOOL,
    UFI_H_ALIGN D_SMALLINT,
    UFI_V_ALIGN D_SMALLINT,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_IS_UNIT D_BOOL)
RETURNS (
    NEW_UFI_UUID_STR D_STR_CODE)
AS
declare variable ufi_parent_uuid d_uuid;
declare variable ufi_uuid d_uuid;
declare variable fr_uuid d_uuid;
declare variable code d_str_code;
declare variable gr_no d_integer;
begin
  if (ufi_is_unit is null) then
    ufi_is_unit = 0;

  if (coalesce(ufi_uuid_str, '') <> '') then
  begin
    if (right(ufi_uuid_str, 5) in ('-text', '-file', '-blob', '-json')) then
      ufi_uuid_str = left(ufi_uuid_str, char_length(ufi_uuid_str) - 5);

    ufi_uuid = char_to_uuid(ufi_uuid_str);
  end

  if (coalesce(ufi_parent_uuid_str, '') <> '') then
    ufi_parent_uuid = char_to_uuid(ufi_parent_uuid_str);

  if (ufi_kind_str <> 'button') then
  begin
    if (ufi_caption_visible = 0) then
      ufi_caption_align = -1;
  
    if (ufi_group_border_visible = 0) then
      ufi_caption_align = -1;
  end

  if (ufi_is_unit = 0) then -- Сохранение стандартное
  begin
    if (ufi_uuid is null) then
    begin
      if (ufi_kind_str = 'group') then
      begin
        select ufi_uuid_str
        from uxp$uxfritem_add_aux(:ufi_owner, :ufi_kind_str, :ufi_kind_str, null)
        into :new_ufi_uuid_str;
      end else
      begin
        select ufi_uuid_str
        from uxp$uxfritem_add_aux(:ufi_owner, 'aux', :ufi_kind_str, null)
        into :new_ufi_uuid_str;
      end
  
      ufi_uuid = char_to_uuid(new_ufi_uuid_str);
    end else
    begin
      if (ufi_parent_uuid_str <> '') then
      begin
        fr_uuid = null;
        select ufi_owner
        from ux$fritem
        where ufi_uuid = :ufi_uuid and
              ufi_parent_uuid is null
        into fr_uuid;
  
        if (fr_uuid is not null) then
        begin
          select ufi_uuid
          from ux$fritem
          where ufi_owner = :ufi_owner and      
                ufi_ufr_uuid = :fr_uuid
          into ufi_uuid;
        end
      end
  
      new_ufi_uuid_str = uuid_to_char(ufi_uuid);
    end
  
    update ux$fritem
    set ufi_parent_uuid = :ufi_parent_uuid,
        ufi_position = :ufi_position,
        ufi_name = :ufi_name,
        ufi_caption_align = :ufi_caption_align,
        ufi_hidden = :ufi_hidden,
        ufi_group_direction = :ufi_group_direction,
        ufi_h_align = :ufi_h_align,
        ufi_v_align = :ufi_v_align,
        ufi_width = coalesce(:ufi_width, ufi_width),
        ufi_height = coalesce(:ufi_height, ufi_height)
    where ufi_uuid = :ufi_uuid;
  end else -- Сохранение UxUnitItem
  begin
    if (ufi_uuid is null) then
    begin
      if (ufi_kind_str <> 'group') then
        exception e_message 'Разрешается создать только группы (контейнер)';

      -- Ищем свободный номер (кодовое имя)
      gr_no = 0;
      while (1 = 1) do
      begin
        gr_no = gr_no + 1;
        code = 'NewGroup' || gr_no;

        if (not exists(select * from ux$unititem
        where uni_unt_uuid = :ufi_owner and
              uni_code = :code)) then
          Break;
      end

      if (coalesce(ufi_width, 0) = 0) then
        ufi_width = 10;
      if (coalesce(ufi_height, 0) = 0) then
        ufi_height = 10;

      insert into ux$unititem (uni_unt_uuid, uni_parent_uuid, uni_code, uni_name, uni_json_path,
                               uni_kind, uni_caption, uni_control, uni_add_info, uni_position,
                               uni_caption_align, uni_hidden,
                               uni_group_direction, uni_h_align, uni_v_align,
                               uni_left, uni_top, uni_width, uni_height)
      values (:ufi_owner, :ufi_parent_uuid, :code, 'TPanel', '',
              -1, :ufi_name, :ufi_kind_str, 'manual added', :ufi_position,
              :ufi_caption_align, :ufi_hidden,
              :ufi_group_direction, :ufi_h_align, :ufi_v_align,
              0, 0, :ufi_width, :ufi_height)
      returning uni_uuid
      into :ufi_uuid;

      new_ufi_uuid_str = uuid_to_char(ufi_uuid);
    end else
    begin
      update ux$unititem
      set uni_parent_uuid = :ufi_parent_uuid,
          uni_position = :ufi_position,
          uni_caption = :ufi_name,
          uni_caption_align = :ufi_caption_align,
          uni_hidden = :ufi_hidden,
          uni_group_direction = :ufi_group_direction,
          uni_h_align = :ufi_h_align,
          uni_v_align = :ufi_v_align,
          uni_width = :ufi_width,
          uni_height = :ufi_height
      where uni_uuid = :ufi_uuid;
    end
  end

  suspend;
end
