CREATE PROCEDURE ADD_MARKUP_DISCOUNT(
    K_ID D_ID,
    OL_ID D_ID)
AS
declare variable dt_oper d_date;
declare variable reg_id d_integer;
declare variable pr_id d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable amount d_currency;
declare variable k_kredline_end d_date;
begin
  -- RFCKG-5358
  select
    pr_id,
    k_kredline_end
  from kontract
    inner join product
      on pr_id = k_from_pr_id
        and pr_code containing '.murabaha.'
        and pr_percent_kind = 0
  where k_id = :k_id
  into
    :pr_id,
    :k_kredline_end;

  select
    ol_date,
    ol_from_reg_id
  from operlist
  where ol_id = :ol_id
  into
    :dt_oper,
    :reg_id;

  if (f_get_db_id() = 41701
    and pr_id > 0
    and exists(
      select * from knclient
      where kc_from_k_id = :k_id
        --and kc_loan_target = 63
        and kc_amount_approved > 0)
    and (
      -- досрочное частичное погашение
      exists(
        select * from oksummary
          where oks_from_k_id = :k_id
            and :dt_oper between oks_date_begin and oks_date_end
           and :dt_oper < oks_delinq_date)
      or
      -- досрочное полное погашение RFCKG-5474
      exists(
        select *
        from kontstate
        where ks_from_ol_id = :ol_id
          and ks_date_begin < :k_kredline_end
          and ks_ol_is_delete = 0
          and ks_state = 2))) then
  begin
    amount =
      coalesce((
        select abs(sum(ok_sh_perc))
        from operkred
        where ok_from_ol_id = :ol_id
          and ok_type in (2,3)), 0);
    if (amount > 0) then
    begin
      acc_dt = (
        select res_account
        from hp_make_kredit_acclist(:pr_id, :reg_id, 2));
      acc_kt = (
        select res_account
        from hp_make_kredit_acclist(:pr_id, :reg_id, 3));

      insert into operbook(
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val
      )
      select
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        :acc_dt,
        :acc_kt,
        'Уступка по наценке в связи с досрочным / частично досрочным погашением по кредиту',
        :amount,
        :amount
      from operbook
      where ob_from_ol_id = :ol_id
        and ob_account_kt = :acc_kt;
    end
  end
end
