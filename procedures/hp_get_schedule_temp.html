CREATE PROCEDURE HP_GET_SCHEDULE_TEMP(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    JSON_TEXT D_BLOB_TEXT,
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
declare variable sh_id d_integer;
declare variable sh_calc_fine curr_type;
declare variable sh_calc_perc curr_type;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable s_perc curr_type;
declare variable s_os curr_type;
declare variable s_all curr_type;
declare variable sh_repay_type d_integer;
declare variable sh_date d_date;
declare variable is_first_row d_smallint;
declare variable sh_date_before d_date;
declare variable delinq_start_date d_date;
declare variable fine curr_type;
declare variable month_add d_integer;
declare variable sh_balance_before curr_type;
declare variable k_period d_integer;
declare variable kd_percent_effect_fifd d_percent;
begin
  json_text = '{';

    select sum(ok_os)
    from operkred
    where ok_ol_is_active = 1
      and ok_from_k_id = :k_id
      and ok_from_c_id = 0
    into :sh_balance_before;


  select res_is_error, res_text, delinq_start_date, sh_date, fine
  from hp_do_reschedule_kz(:k_id, :rep_date)
  into :res_is_error, :res_text, delinq_start_date, sh_date_before, fine;

  delinq_start_date = dateadd(1 day to delinq_start_date);

  select iif(kn.k_period - kn.k_kolich_repay < 0, 0, kn.k_period - kn.k_kolich_repay), kn.k_period -- тоже, что и в балансе кредита. Если k_period = 22,6 то округляется до 23. В FS также  
  from kontract kn
  where kn.k_id = :k_id
  into :month_add, k_period;

  
  select kd_percent_effect_fifd
  from kontdata
  where kd_from_k_id = :k_id
        and kd_ol_is_delete = 0
  order by kd_ol_Date desc
  rows 1
  into kd_percent_effect_fifd;

  json_text = blob_append(json_text, f_format('"sh_date_before":"%1", "delinq_start_date":"%2" , "fine":%3, "month_add":%4, "sh_balance_before":%5 ,', f_datetostr(:sh_date_before), f_datetostr(:delinq_start_date), :fine,:month_add, :sh_balance_before));
  json_text = blob_append(json_text, f_format(' "k_period":"%1", "k_effect_rate":"%2",',:k_period, :kd_percent_effect_fifd));



  json_text = blob_append(json_text, ' "schedule": [');
  is_first_row = 1;
  for select sh_id, sh_date, sh_calc_fine, sh_calc_perc, zad_os, zad_perc, sh_repay_type, s_os, s_perc, s_all
      from aarep_make_zad_list(:k_id, current_date, 1)
      into sh_id, sh_date, sh_calc_fine, sh_calc_perc, zad_os, zad_perc, sh_repay_type, s_os, s_perc, s_all
  do
  begin
    if (is_first_row = 0) then
      json_text = blob_append(json_text,',');
    json_text = blob_append(json_text, '{' || f_json_make_pair('sh_id', :sh_id, 'str') || ', ' || --
    f_json_make_pair('sh_date', :sh_date, 'str') || ', ' || --
    f_json_make_pair('sh_calc_fine', :sh_calc_fine, 'str') || ', ' || --
    f_json_make_pair('sh_calc_perc', :sh_calc_perc, 'str') || ', ' || --
    f_json_make_pair('zad_os', :zad_os, 'str') || ', ' || --
    f_json_make_pair('zad_perc', :zad_perc, 'str') || ', ' || --
    f_json_make_pair('sh_repay_type', :sh_repay_type, 'str') || ', ' || --
    f_json_make_pair('s_os', :s_os, 'str') || ', ' || --
    f_json_make_pair('s_perc', :s_perc, 'str') || ', ' || --
    f_json_make_pair('s_all', :s_all, 'str') || '}'); --

    if (is_first_row = 1) then
      is_first_row = 0;
  end
  json_text = blob_append(json_text, ']}');
  suspend;
end
