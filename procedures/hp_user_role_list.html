CREATE PROCEDURE HP_USER_ROLE_LIST(
    FILTER_COMMAND VARCHAR(255),
    IN_UN_ID BIGINT = 0)
RETURNS (
    UN_ID DECIMAL(18,0),
    UN_NAME VARCHAR(30),
    RN_ID DECIMAL(18,0),
    RN_NAME VARCHAR(100),
    UN_IS_ACTIVE INTEGER,
    UN_CURR_REG_ID INTEGER,
    UN_FIO VARCHAR(100),
    UN_FROM_C_ID DECIMAL(18,0),
    UN_ANDROID_ID VARCHAR(50),
    REG_NAME D_STR_4096,
    UN_READONLY D_BOOL)
AS
declare variable kol integer;
declare variable m_kind varchar(100);
declare variable m_id decimal(18,0);
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;

  if (filter_command is null) then
    filter_command = '';
  m_kind = '';
  m_id = 0;
  if (filter_command <> '') then
  begin
    if (filter_command starting with '[menu]') then
    begin
      m_kind = 'Menulist';
      m_id = replace(filter_command, '[menu]', '');
    end
    if (filter_command starting with '[permission]') then
    begin
      m_kind = 'Permactions';
      m_id = replace(filter_command, '[permission]', '');
    end
  end
  for select un_id, un_name, coalesce(un_curr_reg_id, 0), un_fio, un_from_c_id, un_android_id, un_readonly
      from username
      where un_id = :in_un_id or :in_un_id = 0
      into :un_id, :un_name, :un_curr_reg_id, :un_fio, :un_from_c_id, :un_android_id, :un_readonly
  do
  begin
    select un_is_active
    from hp_un_is_active(:un_id)
    into :un_is_active;

    rn_id = 0;
    rn_name = '';
    kol = 0;
    for select pm_subject_id
        from permissions
        where pm_object_kind = :uniqtab_username and
              pm_object_id = :un_id and
              pm_subject_kind = :uniqtab_rolename and
              pm_is_delete = 0 and
              (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
              (coalesce(pm_date_end, '31.12.2999') >= current_date)
        into :rn_id
    do
    begin
      kol = kol + 1;
      if (rn_name = '') then
      begin
        select rn_name
        from rolename
        where rn_id = :rn_id
        into :rn_name;
      end
    end
    if (kol > 1) then
      rn_name = rn_name || ' и ' ||(kol - 1) || ' других';

    if ((kol = 0) and
        (filter_command = '')) then
    begin
      for select pm_subject_id
          from permissions
          where pm_object_kind = :uniqtab_username and
                pm_object_id = :un_id and
                pm_subject_kind = :uniqtab_rolename and
                pm_is_delete = 0
          into :rn_id
      do
      begin
        kol = kol + 1;
        if (rn_name = '') then
        begin
          select rn_name
          from rolename
          where rn_id = :rn_id
          into :rn_name;
        end
      end
      if (kol > 1) then
        rn_name = rn_name || ' и ' ||(kol - 1) || ' других старых';
      if (kol = 1) then
        rn_name = rn_name || ' (старый)';
      if (kol = 0) then
      begin
        for select pm_subject_id
            from permissions
            where pm_object_kind = :uniqtab_username and
                  pm_object_id = :un_id and
                  pm_subject_kind = :uniqtab_rolename
            into :rn_id
        do
        begin
          kol = kol + 1;
          if (rn_name = '') then
          begin
            select rn_name
            from rolename
            where rn_id = :rn_id
            into :rn_name;
          end
        end
        if (kol > 1) then
          rn_name = rn_name || ' и ' ||(kol - 1) || ' других аннулированных';
        if (kol = 1) then
          rn_name = rn_name || ' (аннулир.)';
        if (kol = 0) then
        begin
          rn_name = 'Без роли';
          kol = 1;
        end
      end
    end
    if ((kol > 0) and
        (un_is_active = 0) and
        (filter_command <> '')) then
      kol = 0;

    if ((kol > 0) and
        (filter_command <> '')) then
    begin
-- Проверка доп.условий
      if (exists(select has_access
                 from hp_permission_access_check(:un_id, :m_kind, :m_id, '')
                 where has_access = 0)) then
      begin
        kol = 0;
      end
    end
    if (kol > 0) then
    begin
--S.K. 24.12.2018 г.
      reg_name = '';
      reg_name = (select list(reg_name, '; ')
                  from permissions
                  inner join region on reg_unicode = pm_subject_id
                  where pm_object_kind = 0 and
                        pm_object_id = :un_id and
                        pm_subject_kind = 4 and
                        pm_access_kind = 1 and
                        pm_is_delete = 0 and
                        current_date between coalesce(pm_date_begin, current_date) and coalesce(pm_date_end, current_date));
--
      suspend;
    end
  end
end
