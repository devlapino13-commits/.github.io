CREATE PROCEDURE AAREP_STAT_CYCLE(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    STEP SMALLINT)
RETURNS (
    C_ID ID_TYPE,
    C_NAME_FIO VARCHAR(100),
    CYCLE_CL INTEGER,
    CYCLE_1 INTEGER,
    CYCLE_2_MORE INTEGER,
    MA_FROM_PR_ID INTEGER,
    PR_NAME VARCHAR(50),
    CYCLE_ALL INTEGER,
    MA_ID INTEGER)
AS
declare variable ma_date_app date;
begin
  if (step = 1) then
  begin
    for select mac_from_c_id, ma_date_app, ma.ma_from_pr_id, ma.ma_id
        from miniapp ma, miniappclient mac, region_list(1000000, 1, -1, 1)
        where ma.ma_number = 0 and
              ma.ma_date_app between :rep_date_begin and :rep_date_end and
              ma.ma_id = mac.mac_from_ma_id and
              mac.mac_amount > 0 and
              mac.mac_is_poruch = 0 and
              ma.ma_office_id = id_list
        into c_id, ma_date_app, ma_from_pr_id, :ma_id
    do
    begin
      cycle_cl = null;

      select count(distinct ks_from_k_id)
      from kontstate, knclient
      where ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_from_k_id = kc_from_k_id and
            ks_date_begin < :ma_date_app and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0
      into cycle_cl;

      select c_name_fio
      from clients
      where c_id = :c_id
      into c_name_fio;

      if (cycle_cl is null) then
        cycle_cl = 0;

--select distinct iif(pr.pr_common_name starting 'ПП', 'Параллельный займ',
--                    iif(pr_common_name = 'Займ для сотрудников', 'Прочие', pr_common_name) )
      select pr.pr_name
      from product pr
      where pr.pr_id = :ma_from_pr_id
      into pr_name;

      suspend;
    end
  end
  if (step = 2) then
  begin
    for select sum(cycle_1), sum(cycle_2_more)
        from (select iif(cycle_cl <= 1, 1, 0) cycle_1, iif(cycle_cl > 1, 1, 0) cycle_2_more
              from aarep_stat_cycle(:rep_date_begin, :rep_date_end, 1)) a
        into cycle_1, cycle_2_more
    do
    begin
      suspend;
    end
  end

  if (step = 3) then
  begin
    for select pr_name, sum(cycle_1), sum(cycle_2_more)
        from (select pr_name, iif(cycle_cl <= 1, 1, 0) cycle_1, iif(cycle_cl > 1, 1, 0) cycle_2_more
              from aarep_stat_cycle(:rep_date_begin, :rep_date_end, 1)) a
        group by pr_name
        into pr_name, cycle_1, cycle_2_more
    do
    begin
      cycle_all = cycle_1 + cycle_2_more;
      suspend;
    end
  end
end
