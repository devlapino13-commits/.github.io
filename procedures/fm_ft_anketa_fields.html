CREATE PROCEDURE FM_FT_ANKETA_FIELDS(
    C_ID D_ID,
    UN_ID D_ID,
    FAK_CODE D_STR_20 = 'client')
RETURNS (
    RES_TEXT D_STR_255,
    RES_DATA D_STR_8100,
    RES_CHANGE_CONFIRMATION D_STR_255,
    RES_SAVE_CONFIRMATION D_STR_255)
AS
declare variable LANG D_STR_LANG;
declare variable JSON_ITEM D_STR_2048;
declare variable COMPANY_NAME D_STR_255;
declare variable FAK_ID D_INTEGER;
declare variable NEED_SHOW_FT D_SMALLINT;
declare variable SELFEMPLOYED_START_DATE D_DATE;
begin
  -- Процедура проверяет, что для клиента можно создавать анкету ФТ
  -- Если создание разрешено, возвращает поля анкеты
  if ((:fak_code is null) or (coalesce (:fak_code, '') = '')) then
    fak_code = 'client';
  lang = f_get_lang_code(un_id, null);
  res_text = '';
  res_data = '';
  res_change_confirmation = 'Вы уверены, что хотите изменить ответ?';
  res_save_confirmation = 'Есть нестандартные ответы. Анкета будет отправлена на проверку. Подтверждаете?';

  company_name = f_company_name('short', 'local');
  --RFCRU-2543
  selfemployed_start_date = cast(f_get_const('selfemployed_start_date') as date);
  need_show_ft = 0;

  if (selfemployed_start_date <> '31.12.9999' and
      exists(select *
             from miniappclient mac
             inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id and
                    ma.ma_is_active = 1 and
                    ma.ma_date_end >= '31.12.9999'
             inner join product pr on pr.pr_id = ma.ma_from_pr_id and
                    pr.pr_code containing '.selfemployed.'
             where mac.mac_from_c_id = :c_id and
                   not exists(select *
                              from kontract kt
                              inner join kontstate ks on ks.ks_from_k_id = kt.k_id and
                                    ks.ks_state = 1 and
                                    ks.ks_ol_is_delete = 0 and
                                    ks.ks_date_begin <= current_date
                              where kt.k_from_ma_app_id = ma.ma_id))) then
    need_show_ft = 1;

  if (f_tmp_const('use_perstemp') = 1) then
    need_show_ft = 0;

  if (f_meet_valid(c_id, 120) = 0 and (select is_oivo from hp_check_oivo_assignment(:c_id, :un_id)) <> 1) then
    res_text = f_lang(lang, 'Требуется верификация клиента по QR-коду.');
  else if (f_asp_valid(c_id, current_date) = 0) then
    res_text =  f_lang(lang, 'Требуется подписание договора АСП с клиентом.');
  else if (f_ft_valid(c_id, current_date) = 1 and need_show_ft = 0) then
    res_text =  f_lang(lang, 'Создание Анкеты-опросника не требуется');


  select fak.fak_id
  from ft_anketa_kind fak
  where fak.fak_code = :fak_code
  into :fak_id;

  if (res_text = '') then
  begin
    for select '{' ||
      f_json_make_pair('id', faf_id, 'str') || ',' ||
      f_json_make_pair('que', trim(replace(faf_short_caption, '[company_name]', :company_name)), 'str') || ',' ||
      f_json_make_pair('ans_0', trim(faf_false_caption), 'str') || ',' ||
      f_json_make_pair('ans_1', trim(faf_true_caption), 'str') || ',' ||
      f_json_make_pair('ans_def', faf_default_value, 'str') || ',' ||
      f_json_make_pair('has_text', faf_need_comment, 'str') || '}'
    from ft_anketa_field
    where faf_inactive = 0 and
    faf_from_fak_id = :fak_id
    order by faf_position
    into json_item
    do
      res_data = res_data || iif(res_data = '', '[', ',') || json_item;
    res_data = res_data || ']';
  end

  suspend;
end
