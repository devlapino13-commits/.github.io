CREATE PROCEDURE RCIB_CRED_FIP_LIST_TEMP(
    BASE_RCP_ID ID_TYPE,
    REP_K_ID ID_TYPE,
    REP_MA_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REP_DATE_BEGIN DATE,
    K_ID ID_TYPE,
    CRED_MISSEDPAY_120 SMALLINT,
    CRED_FULL_PAY SMALLINT,
    ORDER_NO SMALLINT,
    MA_ID INTEGER,
    FAILURE_REASON INTEGER,
    FAILURE_REASON_TEXT STR_100,
    PR_ID INTEGER,
    STATUS VARCHAR(20),
    K_ID_REAL ID_TYPE,
    STATUS_DATE DATE,
    STATUS_CODE SMALLINT,
    K_KONTRACT_NO VARCHAR(50),
    K_KREDLINE_END DATE,
    K_CURRENCY VARCHAR(10),
    CRED_DURATION INTEGER)
AS
declare variable active_date date;
declare variable ma_is_active integer;
begin
  cred_missedpay_120 = 0;
  cred_full_pay = 0;
  failure_reason = null;
  failure_reason_text = '';
  pr_id = 0;
  if (base_rcp_id = 0) then
    base_rcp_id = null;
  if (rep_k_id = 0) then
    rep_k_id = null;
  if (rep_ma_id = 0) then
    rep_ma_id = null;

  active_date = cast('01.03.2015' as date); -- дата запуска проекта

  if (base_rcp_id is null) then
  begin
    if ((rep_k_id is null) and
        (:rep_ma_id is null)) then
    begin
      select max(rcp_date) + 1
      from rcib_pocket rcp
      where rcp.rcp_sended = 1 and
            rcp.rcp_is_kredit = 0 --! не кредитный а ФИЧ-пакет.
            and
            rcp.rcp_type = 1 -- групповой
            and
            rcp.rcp_date <= :rep_date_end
      into :rep_date_begin;
    end
    else
    begin
      rep_date_begin = active_date;
    end
  end
  else --  если пакет по ошибкам
  begin
    select rcp.rcp_date_begin, rcp.rcp_date
    from rcib_pocket rcp
    where rcp.rcp_id = :base_rcp_id and
          rcp.rcp_is_kredit = 0 -- << -- ! не кредитный а ФИЧ-пакет.
    into :rep_date_begin, :rep_date_end;
  end

  if (rep_date_begin is null) then
    rep_date_begin = active_date;

-- 1) контракты у которых в течении 120 дней с момента выдачи была непрерывная просрочка более двух платежей

  /*
    График платежей по договору:
    01.01.2015     01.02.2015   01.03.2015   01.04.2015   01.05.2015   01.06.2015   01.07.2015   01.08.2015   01.09.2015   ***
    Дата договора  1-ый платеж  2-ой платеж  3-ий платеж  4-ый платеж  5-ый платеж  6-ой платеж  7-ой платеж  8-ой платеж  *** 
                                                                                    ~ 120 дней с 
                                                                                    первого платежа
     
    1.  1) Если субъект КИ  не погасил подряд  1-ый и 2-ой  платеж –  пропущено 2 подряд платежа  в течение 120 календарных дней с момента 
    первого платежа, – то необходимо передать cred_missedpay_120 = 1. 

    2.  Если субъект КИ погасил 1-ый и 2-ой платеж, но не погасил подряд 3-ий и 4-ий платеж (или 4-ый и 5-ый) – пропущено 2 подряд платежа в 
    течение 120 календарных дней с момента первого платежа, – то необходимо передать cred_missedpay_120 = 1. 

    3.  Если субъект КИ  не погасил 5-ый и 6-ой платеж, то необходимо передать cred_missedpay_120  =  0,  т.к. в течение 120 календарных дней с 
    момента первого платежа был только один пропущенный платеж (5-ый платеж). 
  */
  if (base_rcp_id is null) then
  begin
    if (:rep_ma_id is null) then
    begin
      for select ks_from_k_id, ks_date_begin
          from kontstate ks
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :rep_date_end between ks_date_begin and ks_date_end and
                ks.ks_date_begin >= cast('01.03.2015' as date) and
                ((ks.ks_from_k_id = :rep_k_id) or
                (:rep_k_id is null))
          union all
          select ks_from_k_id, ks_date_begin
          from kontstate ks
          inner join kontdevides kd on kd.kd_k_new = ks.ks_from_k_id
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :rep_date_end between ks_date_begin and ks_date_end and
                ks.ks_date_begin >= cast('01.03.2015' as date) and
                (kd.kd_k_orig = :rep_k_id)

--  rows 100
          into :k_id, :status_date
      do
      begin
-- если ранее не отправлялся
        if (not exists(select *
                       from rcib_fip rcf
                       inner join rcib_pocket rcp on rcp.rcp_id = rcf.rcf_from_rcp_id
                       where rcf.rcf_from_k_id = :k_id and
                             rcf.rcf_cred_missedpay_120 = 1 and
                             rcp.rcp_sended = 1)) then
        begin
          if (exists(select *
                     from rcib_cred_missedpay_120(:k_id, :status_date, :rep_date_end))) then
          begin
            cred_missedpay_120 = 1; -- была непрерывная просрочка в размере более 2-ух платежей, в первые 120 дней с момента выдачи займа

            order_no = 1;
            status_code = 4; -- контракт выдан
            status = 'cont_120';
            suspend;
          end
        end
      end
    end
  end
  else
  begin
    for select rcf.rcf_from_k_id, rcf.rcf_cred_date
        from rcib_pocket rcp
        inner join rcib_kont_errors rke on rke.rce_pocket_id = rcp.rcp_id
        inner join rcib_fip rcf on rcf.rcf_from_rcp_id = rcp.rcp_id and
              rcf.rcf_rec_no = rke.rce_from_rec_no
        where rcp.rcp_id = :base_rcp_id and
              rcf.rcf_data_type = 'cont_120'
        into :k_id, :status_date
    do
    begin
      cred_missedpay_120 = 1; -- была непрерывная просрочка в размере более 2-ух платежей, в первые 120 дней с момента выдачи займа

      order_no = 1;
      status_code = 4; -- контракт выдан
      status = 'cont_120';
      suspend;
    end
  end

  cred_missedpay_120 = 0;
  cred_full_pay = 0;
  failure_reason = null;
  failure_reason_text = '';
  pr_id = 0;
  cred_duration = null;

-- 2) Закрытые контракты за период, по которым осуществлено полное исполнение обязательств
--    по договору займа (кредита)
  if (base_rcp_id is null) then
  begin
    if (:rep_ma_id is null) then
    begin
      for select ks.ks_from_k_id, k_fact_disb_date,

                 kont.k_kontract_no, kont.k_kredline_end, kont.k_currency,

                 round(kont.k_period)
          from kontstate ks
          inner join kontract kont on kont.k_id = ks.ks_from_k_id
          where ks.ks_state = 2 and
                ks.ks_ol_is_delete = 0 and
                ks.ks_date_begin between :rep_date_begin and :rep_date_end and
                kont.k_fact_disb_date >= cast('01.03.2015' as date) and
                ((ks.ks_from_k_id = :rep_k_id) or
                (:rep_k_id is null))
          union all
          select ks.ks_from_k_id, k_fact_disb_date,

                 kont.k_kontract_no, kont.k_kredline_end, kont.k_currency,

                 round(kont.k_period)
          from kontstate ks
          inner join kontract kont on kont.k_id = ks.ks_from_k_id
          inner join kontdevides kd on kd.kd_k_new = kont.k_id
          where ks.ks_state = 2 and
                ks.ks_ol_is_delete = 0 and
                ks.ks_date_begin between :rep_date_begin and :rep_date_end and
                kont.k_fact_disb_date >= cast('01.03.2015' as date) and
                (kd.kd_k_orig = :rep_k_id)
--      rows 100
          into :k_id, :status_date, :k_kontract_no, :k_kredline_end, :k_currency, :cred_duration
      do
      begin
        if ((select sum(ok.ok_sh_os + ok.ok_os)
             from operkred ok
             where ok.ok_type in (1, 4) and
                   ok.ok_from_k_id = :k_id and
                   ok.ok_ol_is_active = 1) <= 0) then
        begin
          cred_full_pay = 1;
          cred_missedpay_120 = 0;

          if (exists(select *
                     from rcib_cred_missedpay_120(:k_id, :status_date, :rep_date_end))) then
          begin
            cred_missedpay_120 = 1;
          end

          order_no = 2;
          status_code = 4; -- контракт выдан
          status = 'closed';
          suspend;
        end
      end
    end
  end
  else
  begin
    for select rcf.rcf_from_k_id, rcf.rcf_cred_date, rcf.rcf_cred_missedpay_120
        from rcib_pocket rcp
        inner join rcib_kont_errors rke on rke.rce_pocket_id = rcp.rcp_id
        inner join rcib_fip rcf on rcf.rcf_from_rcp_id = rcp.rcp_id and
              rcf.rcf_rec_no = rke.rce_from_rec_no
        where rcp.rcp_id = :base_rcp_id and
              rcf.rcf_data_type = 'closed'
        into :k_id, :status_date, :cred_missedpay_120
    do
    begin
      order_no = 2;
      status_code = 4; -- контракт выдан
      status = 'closed';
      suspend;
    end
  end
  k_kontract_no = null;
  k_kredline_end = null;
  k_currency = null;

  cred_missedpay_120 = 0;
  cred_full_pay = 0;
  failure_reason = null;
  failure_reason_text = '';
  pr_id = 0;

  cred_duration = null;

-- 3) Получаю все выданные кредиты за период
  if (base_rcp_id is null) then
  begin
    if (:rep_ma_id is null) then
    begin
      for select ks.ks_from_k_id, ks.ks_date_begin
          from kontstate ks
          where ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1 and
                ks.ks_date_begin between :rep_date_begin and :rep_date_end and
                ((ks.ks_from_k_id = :rep_k_id) or
                (:rep_k_id is null))
          union all
          select ks.ks_from_k_id, ks.ks_date_begin
          from kontstate ks
          inner join kontdevides kd on kd.kd_k_new = ks.ks_from_k_id
          where ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1 and
                ks.ks_date_begin between :rep_date_begin and :rep_date_end and
                (kd.kd_k_orig = :rep_k_id)
--   rows 100
          into :k_id, :status_date
      do
      begin
        order_no = 3;
        status_code = 4; -- контракт выдан
        status = 'disb';
        suspend;
      end
    end
  end
  else
  begin
    for select rcf.rcf_from_k_id, rcf.rcf_cred_date
        from rcib_pocket rcp
        inner join rcib_kont_errors rke on rke.rce_pocket_id = rcp.rcp_id
        inner join rcib_fip rcf on rcf.rcf_from_rcp_id = rcp.rcp_id and
              rcf.rcf_rec_no = rke.rce_from_rec_no
        where rcp.rcp_id = :base_rcp_id and
              rcf.rcf_data_type = 'disb'
        into :k_id, :status_date
    do
    begin
      order_no = 3;
      status_code = 4; -- контракт выдан
      status = 'disb';
      suspend;
    end
  end

  cred_missedpay_120 = 0;
  cred_full_pay = 0;
  failure_reason = null;
  failure_reason_text = '';
  pr_id = 0;
  cred_duration = null;
-- 5) получаю все аннулированные заявки за период
  if (base_rcp_id is null) then
  begin
    if (rep_k_id is null) then
    begin
-- если по заявкам за период
      for select maa.ma_from_pr_id, maa.ma_id, cast(maa.ma_date_save as date) as ma_date_save
          from miniapp maa
          where maa.ma_date_oper between iif(:rep_ma_id is null, :rep_date_begin - 30, :rep_date_end + 1) and :rep_date_end and
                maa.ma_date_end = '31.12.9999' and
                maa.ma_is_active = 0
          union all
-- если по одной заявке
          select maa.ma_from_pr_id, maa.ma_id, cast(maa.ma_date_save as date) as ma_date_save
          from miniapp maa
          where maa.ma_date_oper between :active_date and :rep_date_end and
                maa.ma_date_end = '31.12.9999' and
                maa.ma_is_active = 0 and
                maa.ma_id = :rep_ma_id
-- rows 100
          into :pr_id, :ma_id, :status_date
      do
      begin
        k_id = null;
        k_id_real = null;

        select k_id
        from kontract kont
        where kont.k_from_ma_app_id = :ma_id rows 1
        into :k_id;

        status_code = 3; -- Отказ Партнера
        select (case krl.krl_from_kr_id
                  when 1 then 6 -- Несоответствие информации, указанной заемщиком в кредитной заявке, и информации, имеющейся у Партнера
                  when 2 then 4 -- Избыточная долговая нагрузка на заемщика
                  when 3 then 2
                  when 4 then 99
                end) -- Кредитная история заемщика
        from kontreslist krl
        where krl.krl_ma_reg_id = 0 and
              ((krl.krl_ma_id = :ma_id and
              krl.krl_from_k_id is null) or
              (krl.krl_from_k_id = :k_id and
              krl.krl_ma_id is null)) rows 1
        into :failure_reason;

        if (failure_reason = 99) then
        begin
          failure_reason_text = 'Аннулирование для перерегистрации заявки';
          status_code = 6; -- Отмена
        end

        if (failure_reason is null) then
        begin
          failure_reason = 99;
          failure_reason_text = 'Аннулирование для перерегистрации заявки'; --'Кредитная политика заимодавца (кредитора)';
          status_code = 6; -- Отмена
        end
        status = 'annul';
        order_no = 4;

        if ((status_code = 3) and
            (failure_reason <> 99)) then
          failure_reason_text = '';

        if (k_id is not null) then
          k_id_real = k_id;
        else
        begin
          k_id = ma_id;
          k_id_real = null;
        end

        suspend;
      end
    end
  end
  else
  begin
    for select rcf.rcf_from_ma_id, rcf.rcf_failure_reason, rcf.rcf_failure_reason_text
        from rcib_pocket rcp
        inner join rcib_kont_errors rke on rke.rce_pocket_id = rcp.rcp_id
        inner join rcib_fip rcf on rcf.rcf_from_rcp_id = rcp.rcp_id and
              rcf.rcf_rec_no = rke.rce_from_rec_no
        where rcp.rcp_id = :base_rcp_id and
              rcf.rcf_data_type = 'annul'
        into :ma_id, :failure_reason, :failure_reason_text
    do
    begin
      select maa.ma_from_pr_id, cast(maa.ma_date_save as date)
      from miniapp maa
      where maa.ma_date_oper between :rep_date_begin and :rep_date_end and
            maa.ma_date_end = '31.12.9999' and
            maa.ma_id = :ma_id rows 1
      into :pr_id, :status_date;

      status_code = 3; -- Отказ Партнера
      order_no = 4;
      status = 'annul';
      suspend;
    end
  end

  cred_missedpay_120 = 0;
  cred_full_pay = 0;
  failure_reason = null;
  failure_reason_text = '';
  pr_id = 0;
  cred_duration = null;
-- 4) получаю договора в процессе за период
  if (base_rcp_id is null) then
  begin
    if (:rep_ma_id is null) then
    begin
      for select ks.ks_from_k_id, ks.ks_date_begin
          from kontstate ks
          where ks.ks_state = 0 and
                ks.ks_date_begin between :rep_date_begin and :rep_date_end and
                ks.ks_ol_is_delete = 0 and
                ((ks.ks_from_k_id = :rep_k_id) or
                (:rep_k_id is null))
--      rows 100
          into :k_id, :status_date
      do
      begin
        order_no = 5;

        status_code = 1; -- На рассмотрении
        if (exists(select *
                   from kontract kont
                   where kont.k_id = :k_id and
                         kont.k_status = 1)) then
          status_code = 2; -- Одобрен

        status = 'process';
        suspend;
      end
    end
  end
  else
  begin
    for select rcf.rcf_from_k_id, rcf.rcf_ma_status
        from rcib_pocket rcp
        inner join rcib_kont_errors rke on rke.rce_pocket_id = rcp.rcp_id
        inner join rcib_fip rcf on rcf.rcf_from_rcp_id = rcp.rcp_id and
              rcf.rcf_rec_no = rke.rce_from_rec_no
        where rcp.rcp_id = :base_rcp_id and
              rcf.rcf_data_type = 'process'
        into :k_id, :status_code
    do
    begin
      order_no = 5;
      status = 'process';
      suspend;
    end
  end
end
