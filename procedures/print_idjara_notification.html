CREATE PROCEDURE PRINT_IDJARA_NOTIFICATION(
    KMA_ID D_ID)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    SIGN_DATE D_DATE,
    K_DISB_DATE D_DATE,
    C_NAME_FIO D_STR_100)
AS
declare variable k_id d_id;
declare variable ag_id d_id;
declare variable ag_obj_kind d_integer;
declare variable edc_id d_integer;
begin
  select kma.kma_from_k_id, kma.kma_from_ag_id
  from kontmatag kma
  where kma.kma_id = :kma_id
  into k_id, ag_id;

  select kt.k_kontract_no, kt.k_disb_date
  from kontract kt
  where kt.k_id = :k_id
  into k_kontract_no, k_disb_date;

  ag_obj_kind = f_objkind('agreements', 'id');
  edc_id = f_edoc_kind('gpsi.agreement', 'id');

  select cast(eds.eds_status_datetime as date)
  from edoc_list edl
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
  where edl.edl_object_kind = :ag_obj_kind and
        edl.edl_object_id = :ag_id and
        edl.edl_from_edc_id = :edc_id and
        edl.edl_status = 1 and
        eds.eds_status = 1
  into sign_date;

  for select cl.c_name_fio
      from knclient kc
      inner join clients cl on cl.c_id = kc.kc_from_c_id
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      into c_name_fio
  do
  begin
    suspend;
  end
end
