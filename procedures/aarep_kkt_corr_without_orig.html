CREATE PROCEDURE AAREP_KKT_CORR_WITHOUT_ORIG(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    DOC_NO D_STR_CODE,
    KC_ID D_ID,
    FISC_DATETIME D_DATETIME,
    OL_DATETIME D_DATETIME,
    FPD D_ID,
    AMOUNT DECIMAL(18,2),
    SUMM DECIMAL(18,2),
    ORIG_DOC_NO D_STR_CODE,
    ORIG_OL_DATE D_DATE,
    OFD_LINK D_STR_255)
AS
declare variable kkm_id d_id;
declare variable k_id d_id;
declare variable ol_id d_id;
declare variable corr_ol_id d_id;
declare variable k_clients_cashier_id d_id;
begin
  for select kkm_id, kkm_from_ol_id, kkm_corr_ol_id, kkm_from_k_id, kkm_fpd, kkm_fisc_datetime
  from kkt_main kkm
  where kkm.kkm_corr_ol_id is not null
    and kkm.kkm_status = 2
    and kkm_duplicated_kkm_id is null
    and kkm.kkm_fisc_datetime between :rep_date_begin and :rep_date_end
    and not exists(select 1 from kkt_main kkm1 where kkm1.kkm_from_ol_id = kkm.kkm_corr_ol_id and kkm_duplicated_kkm_id is null)
  into kkm_id, ol_id, corr_ol_id, k_id, fpd, fisc_datetime
  do
  begin
    kc_id = null;
    amount = 0;
    ofd_link = null;

    select k_clients_cashier_id, k_kontract_no
    from kontract
    where k_id = :k_id
    into k_clients_cashier_id, doc_no;

    select kc_id
    from knclient
    where kc_from_k_id = :k_id
      and kc_from_c_id = :k_clients_cashier_id
    rows 1
    into :kc_id;

    if (kc_id is null) then
      select min(kc_id)
      from knclient
      where kc_from_k_id = :k_id
      into :kc_id;

    select ol_datetime
    from operlist
    where ol_id = :ol_id
    into ol_datetime;

    select sum(kki_amount)
    from kkt_items 
    where kki_from_kkm_id = :kkm_id
    into amount;

    select max(ol_date), max(k_kontract_no), -sum(ok_os + ok_perc + ok_fine)
    from operlist, operkred, kontract
    where ok_from_ol_id = ol_id
      and k_id = ok_from_k_id
      and (ol_parent_id = :corr_ol_id
          or ol_id = :corr_ol_id)
    rows 1
    into orig_ol_date, orig_doc_no, summ;

    select link
    from hp_kkt_get_ofd_link(:kkm_id)
    into ofd_link;

    suspend;
  end
end
