CREATE PROCEDURE AAREP_MAKE_FIRST_DELINQ(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    POB_AMOUNT DECIMAL(18,4),
    POB_REGION_NAME VARCHAR(100),
    POB_REGION_ID INTEGER,
    POB_BRANCH_ID INTEGER,
    POB_AMOUNT_BALANCE DECIMAL(18,4),
    POB_AMOUNT_DIFF DECIMAL(18,4),
    POB_ACC_DT VARCHAR(20),
    POB_ACC_KT VARCHAR(20),
    POB_BOOK_MEMO VARCHAR(1024))
AS
declare variable acc_main varchar(20);
declare variable acc_delinq varchar(20);
declare variable k_id decimal(18,0);
declare variable tmp_amount decimal(18,4) = 0;
begin
-- SQL_TEXT должен выдавать поля в порядке:
-- 1. Сумма в аналитической части
-- 2. Название офиса
-- 3. ID офиса, за которую формируются данная проводка
-- 4. ID офиса где будет формироваться проводки (обычно это тот же самый офис)
-- 5. Сумму в балансе (бух.учете)
-- 6. Сумму разницы (аналитика - баланс). Это и есть сумма проводки, если сумма минусовая, система поменяет Дт и Кт счета местами
-- 7. Дт счет
-- 8. Кт счет
-- 9. Примечание в проводке, краткое описание

-- получаем группу счетов, где счет учета нормальных кредитов отличается от счета просроченных

  for select distinct reg_unicode, reg_code || ' ' || reg_name,
                      reg_code || pal_currency || pal_main_acc || right('000000' || pal_main_acc_no, 6),
                      reg_code || pal_currency || pal_delinq_acc || right('000000' || pal_delinq_acc_no, 6),
                      sum(dk_os)
      from aarep_kontract_deliq_temp(:report_region_id, null, :report_date_end, '---'), product, kontract, prodacclist, knaktlist, region
      where (dk_os + dk_perc) >= 0.01 and
            dk_id = k_id and
            k_from_pr_id = pr_id and
            pr_from_pat_id = pal_from_pat_id and
            pal_from_at_id = 1 and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :report_date_end and
            kal_date_end >= :report_date_end and
            reg_unicode = kal_new_reg_id
      group by reg_unicode, reg_code, reg_name, pal_currency, pal_main_acc, pal_main_acc_no, pal_delinq_acc, pal_delinq_acc_no
      into :pob_region_id, :pob_region_name, :acc_main, :acc_delinq, :pob_amount
  do
  begin
    pob_amount_balance = 0;
    pob_amount_diff = pob_amount;
    pob_book_memo = 'Перенос основную сумму просроченных кредитов';

-- находим просроченные кредиты, возникщие сегодня
    pob_acc_dt = acc_delinq;
    pob_acc_kt = acc_main;
    pob_branch_id = pob_region_id;
    suspend;
  end

end
