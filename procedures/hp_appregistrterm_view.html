CREATE PROCEDURE HP_APPREGISTRTERM_VIEW(
    DTBEG D_DATE = null,
    DTEND D_DATE = null)
RETURNS (
    ITA_ID D_ID,
    ITA_DOC_NO D_STR_20,
    ITA_DOC_DATE D_DATE,
    ITA_TYPE D_SMALLINT,
    REG_NAME D_STR_CODE,
    ITA_TERM_NAME D_STR_255,
    TRM_IPC_ID D_STR_20,
    ITA_LAST_UN_ID D_ID,
    ITA_STATUS D_SMALLINT)
AS
begin
  dtbeg = coalesce(dtbeg, '30.12.1899');
  dtend = coalesce(dtend, current_date);
  for
    select
      ita_id,
      ita_doc_no,
      ita_doc_date,
      ita_type,
      ita_term_name,
      reg_name,
      trm_ipc_id,
      ita_last_un_id,
      ita_status
    from ipc_terminal_app
      inner join ipc_terminals
        on trm_id = ita_from_trm_id
      left join region
        on reg_unicode = ita_from_reg_id
    where ita_doc_date between :dtbeg and :dtend
  into
    :ita_id,
    :ita_doc_no,
    :ita_doc_date,
    :ita_type,
    :ita_term_name,
    :reg_name,
    :trm_ipc_id,
    ita_last_un_id,
    :ita_status
  do
  begin
    suspend;
  end
end
