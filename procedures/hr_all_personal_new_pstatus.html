CREATE PROCEDURE HR_ALL_PERSONAL_NEW_PSTATUS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PS_ID INTEGER,
    PS_FROM_HO_ID DECIMAL(18,0),
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_FROM_REG_ID INTEGER,
    PS_FROM_DEP_ID INTEGER,
    PS_FROM_JL_ID INTEGER,
    PS_FROM_C_ID DECIMAL(18,0),
    PS_SALARY DECIMAL(18,4),
    PS_REG_NAME VARCHAR(100),
    PS_DEP_NAME VARCHAR(100),
    PS_JL_NAME VARCHAR(100),
    PS_C_NAME_FIO VARCHAR(100),
    PS_ORDER_TYPE INTEGER,
    PS_ORDER_BASE_TYPE INTEGER,
    PS_TABEL_NO VARCHAR(20),
    PS_DATE_HIRE DATE,
    PS_C_SEX_TEXT VARCHAR(20),
    PS_IS_ON_STAFF_TEXT VARCHAR(20),
    PS_IS_WORKING_TEXT VARCHAR(20),
    PS_IS_ON_STAFF SMALLINT,
    PS_IS_WORKING SMALLINT,
    PS_JL_CATEGORY VARCHAR(50),
    PS_JL_IS_KREDIT_TEXT VARCHAR(50),
    PS_JL_IS_FRONT_TEXT VARCHAR(50),
    PS_NORMA_HOURS D_SMALLINT,
    C_INN VARCHAR(20),
    REG_CODE_NAME VARCHAR(100),
    PS_CURR_STATUS VARCHAR(255),
    PS_TABEL_STATUS VARCHAR(100),
    PS_DATE_HIRE_OFFICE DATE,
    PS_DATE_GONE_OFFICE DATE,
    C_SEX INTEGER,
    C_BIRTH_DATE DATE,
    C_NEED_TABEL INTEGER,
    PS_DATE_LEAVE DATE,
    C_PASS_NO VARCHAR(20))
AS
declare variable jl_is_front integer;
declare variable jk_system_code varchar(100);
declare variable cm_database_id integer;
declare variable need_suspend integer;
declare variable period_begin integer;
declare variable period_end integer;
declare variable sal_saldo decimal(18,4) = 0;
declare variable ps_last_date_begin date;
declare variable period_date_begin date;
begin

  :cm_database_id = f_get_db_id();

  period_begin = extract(year from :rep_date_begin) * 100 + extract(month from :rep_date_begin);
  period_end = extract(year from :rep_date_end) * 100 + extract(month from :rep_date_end);

  for select ps_id, ps_from_ho_id, ps_date_begin, ps_date_end, ps_period_month, ps_date_end_in_order, ps_from_reg_id,
             ps_from_dep_id, ps_from_jl_id, ps_from_c_id, ps_is_on_staff, ps_is_working, ps_salary,
             iif(:cm_database_id <> 94401, cl.c_inn, cl.c_pincode), iif(ps_part_hours is null, 8, ps_part_hours)
      from hr_pstatus, region_list(:rep_region_id, 1, -1, 1), clients cl
      where ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_from_reg_id = id_list and
            ps_status = 1 and
            ps_from_c_id = c_id
      order by ps_from_c_id desc, ps_date_begin desc
      into :ps_id, :ps_from_ho_id, :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order,
           :ps_from_reg_id, :ps_from_dep_id, :ps_from_jl_id, :ps_from_c_id, :ps_is_on_staff, :ps_is_working, :ps_salary,
           :ps_tabel_no, ps_norma_hours
  do
  begin
    need_suspend = 0;
    sal_saldo = 0;
    c_need_tabel = 0;
    ps_tabel_status = '';
    if (ps_is_working = 1) then
    begin
      need_suspend = 1; -- Если действующий сотрудник, показываем
      c_need_tabel = 1;
      ps_tabel_status = 'Работает';
    end

    ps_date_leave = null;
    if (need_suspend = 0) then
    begin
      period_date_begin = dateadd(month, -1, :rep_date_begin);

      if (cm_database_id = 94401) then
        period_date_begin = rep_date_begin; -- отчетный период в АЗ только тот месяц, который выбран при открытии табеля (RFCAZ-212)

      if (ps_date_begin between period_date_begin and :rep_date_end) then
      begin
        need_suspend = 1; -- Если уволен, но за отчетный период, показываем
        c_need_tabel = 1;
        ps_tabel_status = 'Уволен за отчетный период';
      end
      ps_date_leave = ps_date_begin;
    end

    if (need_suspend = 0) then
    begin
      select oss_saldo
      from opersalsaldo
      where oss_from_c_id = :ps_from_c_id and
            oss_date_begin <= :rep_date_end and
            oss_date_end >= :rep_date_end and
            oss_saldo <> 0
      order by oss_from_c_id desc, oss_date_begin desc
      rows 1
      into :sal_saldo;
      if (sal_saldo is null) then
        sal_saldo = 0;
      if (sal_saldo <> 0) then
      begin
        need_suspend = 1; -- Если уволен, но есть сальдо по ЗП, показываем
        ps_tabel_status = 'Уволен, оборот или сальдо ЗП';
      end
    end

    if (need_suspend = 0) then
    begin
      if (exists(select *
                 from sa_tabel
                 where sat_from_c_id = :ps_from_c_id and
                       sat_is_ready = 1 and
                       sat_year_month between :period_begin and :period_end)) then
      begin
        need_suspend = 1; -- Если уволен, но есть табель за период, показываем
        ps_tabel_status = 'Уволен, создан табель';
      end
    end
    if (need_suspend = 0) then
    begin
      if (exists(select *
                 from opersalmain
                 where osm_ol_date between :rep_date_begin and :rep_date_end and
                       osm_from_c_id = :ps_from_c_id and
                       osm_ol_is_active = 1)) then
      begin
        need_suspend = 1; -- Если уволен, но есть движение по ЗП за период, показываем
        ps_tabel_status = 'Уволен, оборот или сальдо ЗП';
      end
    end
    if (need_suspend = 1) then -- если попадает в наш список, вытаскиваем остальные данные
    begin
      select reg_name, dep_name, jl_name, c_name_fio, ho_order_type, sawt_base_type, c_hired_date, c_sex, jl_category,
             jl_is_front, c_inn, reg_code || ' ' || reg_name, jk_system_code, c_birth_date, c_pass_no
      from hr_pstatus, region, depart, joblist, clients, hr_orders, sa_worktype, jobkind
      where ps_id = :ps_id and
            ps_from_ho_id = :ps_from_ho_id and
            ho_order_type = sawt_id and
            ps_from_ho_id = ho_id and
            ps_from_reg_id = reg_unicode and
            ps_from_dep_id = dep_id and
            ps_from_jl_id = jl_id and
            ps_from_c_id = c_id and
            jl_kind = jk_id
      into :ps_reg_name, :ps_dep_name, :ps_jl_name, :ps_c_name_fio, :ps_order_type, :ps_order_base_type, :ps_date_hire,
           :c_sex, :ps_jl_category, :jl_is_front, :c_inn, :reg_code_name, :jk_system_code, :c_birth_date, :c_pass_no;

      select ps_date_begin
      from hr_pstatus
      where ps_from_c_id = :ps_from_c_id and
            ps_is_working = 1 and
            ps_status = 1
      order by ps_date_begin
      rows 1
      into :ps_date_hire;

      ps_date_hire_office = null;
      ps_date_gone_office = null;
      if (ps_order_base_type = 1) then
      begin
        ps_last_date_begin = null;
        select ps_date_begin
        from hr_pstatus, hr_orders
        where ps_from_c_id = :ps_from_c_id and
              ho_id = ps_from_ho_id and
              ps_date_begin <= (:rep_date_begin - 1) and
              ps_date_end >= (:rep_date_begin - 1) and
              ps_status = 1 and
              ps_is_working = 1
        union
        select ps_date_begin
        from hr_pstatus, hr_orders
        where ps_from_c_id = :ps_from_c_id and
              ho_id = ps_from_ho_id and
              ps_date_begin >= :rep_date_begin and
              ps_status = 1 and
              ps_is_working = 1 and
              ho_order_type = 1
        order by 1
        rows 1
        into :ps_last_date_begin;

        if ((ps_date_begin >= :rep_date_begin) and
            (ps_date_begin <= :rep_date_end)) then
        begin
          if (ps_last_date_begin is null) then
            ps_date_hire_office = ps_date_begin;
          else
          begin
            if (ps_last_date_begin >= :rep_date_begin) then
              ps_date_hire_office = ps_last_date_begin;
          end
        end
      end
      else
      if (ps_date_hire is not null and
          ps_date_hire between rep_date_begin and rep_date_end) then
      begin
        ps_date_hire_office = ps_date_hire;
      end

      if (ps_order_base_type = 3) then
      begin
        if ((ps_date_begin >= :rep_date_begin) and
            (ps_date_begin <= :rep_date_end)) then
          ps_date_gone_office = ps_date_begin;
      end

      if (ps_salary is null) then
        ps_salary = 0;
      if (ps_tabel_no is null) then
        ps_tabel_no = '';

      if (c_sex = 0) then
        ps_c_sex_text = 'Женский';
      else
      if (c_sex = 1) then
        ps_c_sex_text = 'Мужской';
      else
        ps_c_sex_text = 'Юр.лицо';

      if (ps_is_on_staff = 1) then
        ps_is_on_staff_text = 'Штатный';
      else
        ps_is_on_staff_text = 'Внештатный';

      if (jl_is_front = 1) then
        ps_jl_is_front_text = 'Фронт-офис';
      else
        ps_jl_is_front_text = 'Бэк-офис';

      if (jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
        ps_jl_is_kredit_text = 'Кред.персонал';
      else
        ps_jl_is_kredit_text = 'Админ.персонал';

      if (ps_is_working = 1) then
        ps_is_working_text = 'Работает';
      else
        ps_is_working_text = 'Уволен';

      ps_curr_status = '';
      select sawt_name
      from hr_voctravel, hr_orders, sa_worktype
      where hrv_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            hrv_from_c_id = :ps_from_c_id and
            hrv_status = 1 and
            hrv_date_begin <= :rep_date_end and
            hrv_date_end_fact >= :rep_date_end
      order by hrv_from_c_id desc, hrv_date_begin desc
      rows 1
      into :ps_curr_status;

      suspend;
    end
  end
end
