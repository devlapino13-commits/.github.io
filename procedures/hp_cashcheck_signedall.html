CREATE PROCEDURE HP_CASHCHECK_SIGNEDALL(
    CCM_ID D_INTEGER)
RETURNS (
    SIGNED_ALL BOOLEAN)
AS
declare variable c_id d_id;
declare variable dt_sign d_datetime;
declare variable edl_id d_id;
begin
  signed_all = true;
  edl_id = (select ccm_edl_id
    from cashcheckmain where ccm_id = :ccm_id);

  for
    select
      ccm_auditor as c_id,
      ccm_auditor_datesign as dt_sign
    from cashcheckmain
    where ccm_id = :ccm_id
      and ccm_auditor > 0
    union
    select
      ccm_cashier_curr as c_id,
      ccm_cashier_curr_datesign as dt_sign
    from cashcheckmain
    where ccm_id = :ccm_id
      and ccm_cashier_curr > 0
    union
    select
      ccm_cashier_new as c_id,
      ccm_cashier_new_datesign as dt_sign
    from cashcheckmain
    where ccm_id = :ccm_id
      and ccm_cashier_new > 0
    union
    select
      ccp_from_c_id as c_id,
      ccp_datesign as dt_sign
    from cashcheckpart
    where ccp_from_ccm_id = :ccm_id
  into
    :c_id,
    :dt_sign
  do
  begin
    if (dt_sign is null and
      not exists(
        select * from edoc_sign
        where eds_from_edl_id = :edl_id
          and eds_from_c_id = :c_id
          and eds_status = 1)) then
    begin
      signed_all = false;
      break;
    end
  end
  suspend;
end
