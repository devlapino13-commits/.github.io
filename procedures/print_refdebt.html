CREATE PROCEDURE PRINT_REFDEBT(
    PRC_ID INTEGER)
RETURNS (
    REF_NO INTEGER,
    REF_DATE DATE,
    REF_K_KONTRACT_NO STR_20,
    REF_K_DISB_DATE DATE,
    REF_C_NAME_FIO STR_100,
    REF_C_PASS_NO VARCHAR(30),
    REF_C_PASS_WHO STR_255,
    REF_C_PASS_DATE DATE,
    REF_C_INN STR_20,
    REF_OVERDAY_CUR INTEGER,
    REF_OVERDAY_MAX INTEGER,
    REF_OVER_OS CURR_TYPE,
    REF_OVER_PERC CURR_TYPE,
    REF_OVER_FINE CURR_TYPE,
    REF_OVER_TAX CURR_TYPE,
    REF_NDEBT_OS CURR_TYPE,
    REF_NDEBT_PERC CURR_TYPE,
    REF_NDEBT_TAX CURR_TYPE,
    REF_SH_PERC CURR_TYPE,
    REF_SH_TAX CURR_TYPE,
    REF_IND_OVER_OS CURR_TYPE,
    REF_IND_OVER_PERC CURR_TYPE,
    REF_IND_OVER_FINE CURR_TYPE,
    REF_IND_OVER_TAX CURR_TYPE,
    REF_IND_NDEBT_OS CURR_TYPE,
    REF_IND_NDEBT_PERC CURR_TYPE,
    REF_IND_NDEBT_TAX CURR_TYPE,
    REF_IND_SH_PERC CURR_TYPE,
    REF_IND_SH_TAX CURR_TYPE,
    REF_VSYS_IND_OVER_OS CURR_TYPE,
    REF_VSYS_IND_OVER_PERC CURR_TYPE,
    REF_VSYS_OVER_OS CURR_TYPE,
    REF_VSYS_OVER_PERC CURR_TYPE,
    REF_SIGN_BY_FIO STR_100,
    REF_SIGN_BY_JL_NAME STR_100,
    REF_SIGN_BY_REG_NAME STR_100,
    REF_UN_FIO STR_100,
    REF_UN_JL_NAME STR_100,
    PORUCH_NAME_FIO D_STR_100,
    PORUCH_PASS_NO D_STR_100,
    PORUCH_PASS_DATE D_DATE,
    PORUCH_PASS_WHO D_STR_255,
    PORUCH_PASS_WHO_CODE D_STR_100,
    K_AMOUNT_APPROVED CURR_TYPE,
    K_PERIOD DECIMAL(15,2))
AS
declare variable K_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable CLOSED_OL_ID ID_TYPE;
declare variable PR_PAY_PRED smallint;
declare variable PR_ROUND_PERC smallint;
declare variable PR_IS_IPF smallint;
declare variable NSD_PERCENT PERC_TYPE;
declare variable KOEF PERC_TYPE;
declare variable VSYS_IND_OVER_OS CURR_TYPE;
declare variable VSYS_IND_OVER_PERC CURR_TYPE;
declare variable VSYS_OVER_OS CURR_TYPE;
declare variable VSYS_OVER_PERC CURR_TYPE;
declare variable REF_CURR_OS CURR_TYPE;
declare variable REF_CURR_PERC CURR_TYPE;
declare variable REF_CURR_TAX CURR_TYPE;
declare variable REF_IND_CURR_OS CURR_TYPE;
declare variable REF_IND_CURR_PERC CURR_TYPE;
declare variable REF_IND_CURR_TAX CURR_TYPE;
declare variable PRC_SIGN_BY_C_ID ID_TYPE;
declare variable PRC_FROM_REG_ID ID_TYPE;
declare variable PRC_FROM_UN_ID ID_TYPE;
declare variable DB_ID D_ID;
begin
  db_id = f_get_db_id();
  select prc_no, prc_date, prc_from_k_id, prc_from_c_id, prc_sign_by_c_id, prc_from_un_id, prc_from_reg_id
  from printcert pt
  where prc_id = :prc_id
  into :ref_no, :ref_date, :k_id, :c_id, :prc_sign_by_c_id, :prc_from_un_id, :prc_from_reg_id;

  select cl.c_name_fio, jl.jl_name
  from hr_pstatus h
  inner join clients cl on cl.c_id = h.ps_from_c_id
  inner join joblist jl on jl.jl_id = h.ps_from_jl_id
  where :ref_date between h.ps_date_begin and h.ps_date_end and
        h.ps_is_working = 1 and
        h.ps_status = 1 and
        h.ps_from_c_id = :prc_sign_by_c_id
  into :ref_sign_by_fio, :ref_sign_by_jl_name;

  select cl.c_name_fio, j.jl_name
  from username u
  inner join hr_pstatus h on h.ps_from_c_id = u.un_from_c_id
  inner join clients cl on cl.c_id = u.un_from_c_id
  inner join joblist j on j.jl_id = h.ps_from_jl_id
  where u.un_id = :prc_from_un_id and
        :ref_date between h.ps_date_begin and h.ps_date_end and
        h.ps_is_working = 1 and
        h.ps_status = 1
  into :ref_un_fio, :ref_un_jl_name;

  select r.reg_name
  from region r
  where r.reg_unicode = :prc_from_reg_id
  into :ref_sign_by_reg_name;

  select max(iif(kc_from_c_id = :c_id, kc_closed_ol_id, null)),
         sum(iif(kc_from_c_id = :c_id, kc_amount_approved, 0)) / sum(kc_amount_approved) -- sum(iif(kc_closed_ol_id > 0, 0, kc_amount_approved))
  from knclient
  where kc_from_k_id = :k_id
  into :closed_ol_id, :koef;
  if ((closed_ol_id > 0) and
      exists(select *
             from operlist
             where ol_id = :closed_ol_id and
                   :ref_date >= ol_date and
                   is_delete = 0)) then
    koef = 0;

  if (k_id is not null) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 and
          :ref_date between nsd_date_begin and nsd_date_end
    into :nsd_percent;

    select c_name_fio, c_pass_no, c_pass_who, c_pass_date, c_inn
    from clients
    where c_id = :c_id
    into :ref_c_name_fio, :ref_c_pass_no, :ref_c_pass_who, :ref_c_pass_date, :ref_c_inn;

    select k_kontract_no, k_disb_date, pr_pay_pred, pr_round_perc, iif(pr_code containing '.ipf.', 1, 0),
           kont.k_amount_approved, kont.k_period
    from kontract kont
    inner join product on pr_id = k_from_pr_id
    where k_id = :k_id
    into :ref_k_kontract_no, :ref_k_disb_date, :pr_pay_pred, :pr_round_perc, :pr_is_ipf, :k_amount_approved, :k_period;

    -- Дни просрочки
    select sum(iif(:ref_date - 1 between oks_date_begin and oks_date_end, :ref_date - oks_delinq_date, 0)),
           max(iif(oks_date_end > :ref_date - 1, :ref_date, oks_date_end + 1) - oks_delinq_date)
    from oksummary
    where oks_from_k_id = :k_id and
          oks_delinq_date < iif(oks_date_end > :ref_date, :ref_date, oks_date_end + 1)
          and oks_close_state = 0
    into :ref_overday_cur, :ref_overday_max;

    if (not(exists(select *
                    from operkroff oko
                    where oko.oko_from_k_id = :k_id and
                          oko.oko_from_c_id = :c_id)) or db_id not in (39801, 41701)) then
    begin
      select
           -- Просроченная задолженность
           sum(iif(sh_repay_type = 1, zad_os, 0)),
           -- накопленные проценты
           sum(iif(sh_repay_type = 1, iif(:pr_is_ipf = 1, zad_perc, sh_calc_perc), 0)),
           -- пеня за просрочку
           sum(iif(sh_repay_type = 1, sh_calc_fine, 0)),
           -- НсП
           sum(round(iif(sh_repay_type = 1, iif(:pr_is_ipf = 1, zad_perc, sh_calc_perc), 0) * :nsd_percent / 100, :pr_round_perc)),

           -- Текущая задолженность
           sum(iif((sh_repay_type = 2) and (sh_date = :ref_date), zad_os, 0)),
           -- %
           sum(iif(sh_repay_type = 2, sh_calc_perc, 0)),
           -- НсП
           sum(round(iif(sh_repay_type = 2, sh_calc_perc, 0) * :nsd_percent / 100, :pr_round_perc)),

           -- Незадолженный баланс
           sum(iif(sh_date > :ref_date, zad_os, 0)),
           -- %
           sum(iif(sh_repay_type > 1, iif(:pr_pay_pred = 1, zad_perc, sh_calc_perc), 0)) - sum(iif(sh_repay_type = 2, sh_calc_perc, 0)),
           -- НсП
           sum(round(iif(sh_repay_type > 1, iif(:pr_pay_pred = 1, zad_perc, sh_calc_perc), 0) * :nsd_percent / 100, :pr_round_perc)) - sum(round(iif(sh_repay_type = 2, sh_calc_perc, 0) * :nsd_percent / 100, :pr_round_perc)),

           -- Процент по графику
           -- OS
           sum(zad_perc),
           -- %
           sum(round(zad_perc * :nsd_percent / 100, :pr_round_perc))
      from aarep_make_zad_list(:k_id, :ref_date, 1)
      into :ref_over_os, :ref_over_perc, :ref_over_fine, :ref_over_tax, :ref_curr_os, :ref_curr_perc, :ref_curr_tax,
         :ref_ndebt_os, :ref_ndebt_perc, :ref_ndebt_tax, :ref_sh_perc, :ref_sh_tax;
    end
    ref_ind_over_os = round(koef * ref_over_os, 0);
    ref_ind_over_perc = round(koef * ref_over_perc, 0);
    ref_ind_over_fine = round(koef * ref_over_fine, 0);
    ref_ind_curr_os = round(koef * ref_curr_os, 0);
    ref_ind_curr_perc = round(koef * ref_curr_perc, 0);
    ref_ind_ndebt_os = round(koef * ref_ndebt_os, 0);
    ref_ind_ndebt_perc = round(koef * ref_ndebt_perc, 0);
    ref_ind_sh_perc = round(koef * ref_sh_perc, 0);

    ref_ind_over_tax = round(koef * ref_over_tax, 0);
    ref_ind_curr_tax = round(koef * ref_curr_tax, 0);
    ref_ind_ndebt_tax = round(koef * ref_ndebt_tax, 0);
    ref_ind_sh_tax = round(koef * ref_sh_tax, 0);

    ref_overday_cur = coalesce(iif(ref_overday_cur < 0, 0, ref_overday_cur), 0);
    ref_overday_max = coalesce(iif(ref_overday_max < 0, 0, ref_overday_max), 0);
    ref_over_os = coalesce(ref_over_os, 0);
    ref_over_perc = coalesce(ref_over_perc, 0);
    ref_over_fine = coalesce(ref_over_fine, 0);
    ref_over_tax = coalesce(ref_over_tax, 0);
    ref_curr_os = coalesce(ref_curr_os, 0);
    ref_curr_perc = coalesce(ref_curr_perc, 0);
    ref_curr_tax = coalesce(ref_curr_tax, 0);
    ref_ndebt_os = coalesce(ref_ndebt_os, 0);
    ref_ndebt_perc = coalesce(ref_ndebt_perc, 0);
    ref_ndebt_tax = coalesce(ref_ndebt_tax, 0);
    ref_sh_perc = coalesce(ref_sh_perc, 0);
    ref_sh_tax = coalesce(ref_sh_tax, 0);
    ref_ind_over_os = coalesce(ref_ind_over_os, 0);
    ref_ind_over_perc = coalesce(ref_ind_over_perc, 0);
    ref_ind_over_fine = coalesce(ref_ind_over_fine, 0);
    ref_ind_over_tax = coalesce(ref_ind_over_tax, 0);
    ref_ind_curr_os = coalesce(ref_ind_curr_os, 0);
    ref_ind_curr_perc = coalesce(ref_ind_curr_perc, 0);
    ref_ind_curr_tax = coalesce(ref_ind_curr_tax, 0);
    ref_ind_ndebt_os = coalesce(ref_ind_ndebt_os, 0);
    ref_ind_ndebt_perc = coalesce(ref_ind_ndebt_perc, 0);
    ref_ind_ndebt_tax = coalesce(ref_ind_ndebt_tax, 0);
    ref_ind_sh_perc = coalesce(ref_ind_sh_perc, 0);
    ref_ind_sh_tax = coalesce(ref_ind_sh_tax, 0);

    ref_ndebt_os = ref_ndebt_os + ref_curr_os;
    ref_ndebt_perc = ref_ndebt_perc + ref_curr_perc;
    ref_ndebt_tax = ref_ndebt_tax + ref_curr_tax;
    ref_ind_ndebt_os = ref_ind_ndebt_os + ref_ind_curr_os;
    ref_ind_ndebt_perc = ref_ind_ndebt_perc + ref_ind_curr_perc;
    ref_ind_ndebt_tax = ref_ind_ndebt_tax + ref_ind_curr_tax;

    -- Внесистемная задолженность ... потом убрать
    select coalesce(sum(iif(oko_from_c_id = :c_id, oko_os + oko_sh_os, 0)), 0),
           coalesce(sum(iif(oko_from_c_id = :c_id, oko_perc + oko_sh_perc, 0)), 0),
           coalesce(sum(oko_os + oko_sh_os), 0), coalesce(sum(oko_perc + oko_sh_perc), 0)
    from operkroff
    where oko_from_k_id = :k_id and
          oko_ol_is_active = 1 and
          oko_ol_date <= :ref_date
    into :ref_vsys_ind_over_os, :ref_vsys_ind_over_perc, :ref_vsys_over_os, :ref_vsys_over_perc;

    -- добовляем данные подучителя применив процедуру print_poruch
    select c_name_fio, c_pass_no, c_pass_date, c_pass_who, c_pass_who_code
    from print_poruch(:k_id, 9) rows 1
    into :poruch_name_fio, :poruch_pass_no, :poruch_pass_date, :poruch_pass_who, :poruch_pass_who_code;
    suspend;
  end
end
