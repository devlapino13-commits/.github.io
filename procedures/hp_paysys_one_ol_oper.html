CREATE PROCEDURE HP_PAYSYS_ONE_OL_OPER(
    PSI_ID DECIMAL(18,0),
    DESTIN_KIND INTEGER,
    DESTIN_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0),
    OPER_MEMO VARCHAR(255),
    OPER_DATE DATE)
RETURNS (
    PSM_ID DECIMAL(18,0),
    RES_TEXT VARCHAR(100))
AS
declare variable max_psl_date date;
begin
  /*
    Автор: Анатолий �©егольский
    Дата: 13.01.2015
    Назначение: Надстройка над hp_paysys_one_card_oper
      Дополнительно проверяется и обновляется дата закрепления из даты операции
  */

-- Выясняем дату последнего перемещения psm_id
  select max(psl_date)
  from paysys_list
  where psl_from_psi_id = :psi_id and
        psl_is_delete = 0
  into max_psl_date;

  if (max_psl_date > oper_date) then
  begin
    res_text = 'Для данного БСО существуют операции c более поздней датой (' || right('00' || extract(day from max_psl_date), 2) || '.' || right('00' || extract(month from max_psl_date), 2) || '.' || extract(year from max_psl_date) || '). Закрепление невозможно.';
    suspend;
    exit;
  end

  select psm_id, res_text
  from hp_paysys_one_card_oper(:psi_id, :destin_kind, :destin_id, :un_id, :oper_memo)
  into psm_id, res_text;

-- Обновляем дату закрепления, чтобы она совпадала с датой операции (OL_DATE)
  update paysys_main
  set psm_date = :oper_date
  where psm_id = :psm_id;

  suspend;

end
