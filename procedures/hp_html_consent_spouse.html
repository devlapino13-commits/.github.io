CREATE PROCEDURE HP_HTML_CONSENT_SPOUSE(
    C_ID D_ID,
    MA_ID_TMP D_ID,
    MA_DATE_APP D_DATE)
RETURNS (
    CONSENT D_STR_10)
AS
declare variable crel_other_c_id d_id;
declare variable family d_str_20;
begin
  consent = 'Нет';
  crel_other_c_id = null; --Супруг\супруга
  select crt.rel_other_c_id
  from hp_client_relation_tmp(:c_id, 2) crt
  where crt.rel_code in ('1', '2') and
        rel_is_active = 1 rows 1
  into :crel_other_c_id;

    if (consent = 'Нет' and
        exists(select *
               from edoc_list edl
               inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
               inner join edoc_frx edf on edf.edf_from_edc_id = edc.edc_id
               inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
               where edl.edl_object_kind = f_objkind('miniapp', 'id') and
                     edl.edl_object_id = :ma_id_tmp and
                     edl.edl_status = 1 and
                     edf.edf_from_blt_id = 6 and
                     eds.eds_from_c_id = :crel_other_c_id)) then --согласие супруга в электроном виде
      consent = 'да';
    if (consent = 'Нет' and
        exists(select *
               from clientdocs cds
               inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
               inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id and
                     pdt.pdt_code = 'consent_to_a_loan'
               where cds.cds_from_c_id = :crel_other_c_id and
                     cds.cds_is_delete = 0 and
                     :ma_date_app >= cds.cds_doc_date_begin)) then --согласие супруга в бумажном виде
      consent = 'да';
  suspend;
end
