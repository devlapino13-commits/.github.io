CREATE PROCEDURE AAREP_LOAN_DISB_LIST_TMP_DEMO(
    UN_ID DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    RES_DATE DATE,
    RES_APP_AMOUNT DECIMAL(18,4),
    RES_DISB_AMOUNT DECIMAL(18,4),
    RES_MEMO VARCHAR(255),
    RES_KIND VARCHAR(100),
    RES_LO_ID DECIMAL(18,0),
    K_CLIENT_COUNT INTEGER)
AS
declare variable k_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable ma_id integer;
declare variable ks_state integer;
declare variable amount decimal(18,4) = 0;
declare variable k_status integer;
declare variable rep_plan_disb_date date;
declare variable ma_is_active integer;
declare variable need_suspend integer;
declare variable res_object_status integer;
declare variable kf_name varchar(100);
declare variable lo_id decimal(18,0);
declare variable reg_id integer;
declare variable jk_system_code varchar(100);
declare variable un_name varchar(100);
declare variable cm_database_id integer;
declare variable lang_code d_str_lang;
declare variable rd_code d_str_100;
begin
  select un_lang_code
  from username
  where un_id = :un_id
  into :lang_code;

  :cm_database_id = f_get_db_id();

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id;
  if (lo_id is null) then
    exit;

  reg_id = null;
  select ps_from_reg_id, jk_system_code
  from hr_pstatus, joblist, jobkind
  where ps_date_begin <= :rep_date_end and
        ps_date_end >= :rep_date_end and
        ps_from_c_id = :lo_id and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id
  into :reg_id, jk_system_code;

  if (not(jk_system_code in ('JobSystem.LO_Partner', 'JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.Specialist_KOBD', 'JobSystem.Specialist_Expert'))) then
    reg_id = null;

  if (reg_id is null) then
    exit;

  if (jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Specialist_KOBD')) then
    lo_id = 0;

  for select res_c_id
      from hp_staff_changer_list(:un_id, 1)
      where :lo_id <> 0
      union
      select 0
      from company
      where :lo_id = 0
      into :lo_id
  do
  begin
    obj_kind = 'Kontract';
    res_kind = f_lang(lang_code, 'выдан');
-- Фактически выданные кредиты
    for select k_id, k_from_kf_id, ks_date_begin, k_status, k_amount_approved, k_kolich_client_approved,
               k_from_ma_app_id, ks_kfolder_no, ks_kc_id
        from kontstate, kontract
        where ks_from_k_id = k_id and
              ks_state = 1 and
              ks_ol_is_delete = 0 and
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_region_id = :reg_id and
              (:lo_id = 0 or
              ks_kc_id = :lo_id) and
              ks_date_end > :rep_date_end
        into :k_id, :kf_id, :res_date, :k_status, :amount, :k_client_count, ma_id, :kf_name, :res_lo_id
    do
    begin
      obj_id = k_id;
      res_app_amount = 0;
      res_disb_amount = amount;
      rep_plan_disb_date = null;
      if (coalesce(ma_id, 0) <> 0) then
      begin
        select ma_date_oper
        from miniapp
        where ma_date_app <= :rep_date_end and
              ma_date_end >= '31.12.2999' and
              ma_id = :ma_id and
              ma_is_active = 1
        order by ma_date_app desc
        rows 1
        into :rep_plan_disb_date;

        select sum(mac_amount)
        from miniappclient
        where mac_from_ma_id = :ma_id
        into :res_app_amount;
      end
      res_memo = f_lang(lang_code, 'выдан в срок');
      if (coalesce(rep_plan_disb_date, '01.01.1990') <> res_date) then
      begin
        if (rep_plan_disb_date is not null) then
          res_memo = f_lang(lang_code, 'по плану') || ' ' || extract(day from rep_plan_disb_date) || ' ' ||(case
                                                                                                                     when extract(month from rep_plan_disb_date) = 1 then f_lang(lang_code, 'янв.')
                                                                                                                     when extract(month from rep_plan_disb_date) = 2 then f_lang(lang_code, 'фев.')
                                                                                                                     when extract(month from rep_plan_disb_date) = 3 then f_lang(lang_code, 'март')
                                                                                                                     when extract(month from rep_plan_disb_date) = 4 then f_lang(lang_code, 'апр.')
                                                                                                                     when extract(month from rep_plan_disb_date) = 5 then f_lang(lang_code, 'май')
                                                                                                                     when extract(month from rep_plan_disb_date) = 6 then f_lang(lang_code, 'июнь')
                                                                                                                     when extract(month from rep_plan_disb_date) = 7 then f_lang(lang_code, 'июль')
                                                                                                                     when extract(month from rep_plan_disb_date) = 8 then f_lang(lang_code, 'авг.')
                                                                                                                     when extract(month from rep_plan_disb_date) = 9 then f_lang(lang_code, 'сен.')
                                                                                                                     when extract(month from rep_plan_disb_date) = 10 then f_lang(lang_code, 'окт.')
                                                                                                                     when extract(month from rep_plan_disb_date) = 11 then f_lang(lang_code, 'ноя.')
                                                                                                                     when extract(month from rep_plan_disb_date) = 12 then f_lang(lang_code, 'дек.')
                                                                                                                   end);
        else
          res_memo = f_lang(lang_code, 'выдан без заявки');
      end
      res_memo = kf_name || ': ' || res_memo;
      suspend;
    end

-- Все заявки
    ks_state = 0;
    for select ma_id, ma_from_kf_id, ma_date_oper, ma_is_active, ma_summa_minus, ma_kolich_client, kal_new_kf_name,
               kal_new_kf_officer
        from miniapp, knaktlist
        where ma_date_app <= :rep_date_end and
              ma_date_end >= :rep_date_end and
              ma_date_oper between :rep_date_begin and :rep_date_end and
              ma_from_kf_id = kal_from_kf_id and
              kal_date <= :rep_date_end and
              kal_date_end >= :rep_date_end and
              knaktlist.is_delete = 0 and
              ma_office_id = :reg_id and
              (:lo_id = 0 or
              kal_new_kf_officer = :lo_id)
        into :ma_id, :kf_id, :res_date, :ma_is_active, :amount, :k_client_count, kf_name, res_lo_id
    do
    begin
      need_suspend = 1;
      res_kind = f_lang(lang_code, 'в процессе');

-- исключаем заявку если уже выдан (по ID заявки)
      if (exists(select *
                 from kontract, kontstate
                 where k_from_ma_app_id = :ma_id and
                       ks_from_k_id = k_id and
                       ks_ol_is_delete = 0 and
                       ks_state = 1 and
                       ks_date_end >= '31.12.2999')) then
        need_suspend = 0;

-- исключаем если выдан (по номер папки)
      if (exists(select *
                 from kontract, kontstate
                 where k_from_kf_id = :kf_id and
                       ks_from_k_id = k_id and
                       ks_ol_is_delete = 0 and
                       ks_date_begin = :res_date and
                       ks_state = 1 and
                       ks_date_end >= '31.12.2999')) then
        need_suspend = 0;

      if (not(res_date between :rep_date_begin and :rep_date_end)) then
        need_suspend = 0;

      if (need_suspend = 1) then
      begin
        obj_kind = 'Miniapp';
        obj_id = ma_id;
        res_app_amount = amount;
        res_disb_amount = 0;
        k_status = 0;

        res_memo = f_lang(lang_code, 'Заявка');
        k_id = null;
        select k_id, k_status, k_amount_approved, k_kolich_client_approved
        from kontract, kontstate
        where k_from_ma_app_id = :ma_id and
              ks_from_k_id = k_id and
              ks_ol_is_delete = 0 and
              ks_state <> 1 and
              ks_date_end >= '31.12.2999' rows 1
        into :k_id, k_status, :amount, :k_client_count;
        if (k_id is not null) then
        begin
          obj_kind = 'Kontract';
          obj_id = k_id;
          res_app_amount = amount;
          res_memo = '';
          if (k_status = -1) then
          begin
            res_memo = f_lang(lang_code, 'Отказано');
            res_kind = f_lang(lang_code, 'отказано');
          end
          else
          begin
            res_kind = f_lang(lang_code, 'в процессе');

            select rd_code
            from hp_get_check_loan_data(:k_id)
            into :rd_code;

            res_object_status = null;
            select ovs_add_value
            from hp_objectprops_get('Kontract', :k_id, -1, :rd_code, null)
            into :res_object_status;

            /*select ovs_add_value
            from hp_objectprops_get('Kontract', :k_id, -1, 'ActionGroup.Check_Loan_Data', null)
            into :res_object_status;*/
            if (res_object_status is null) then
            begin
              res_memo = f_lang(lang_code, 'в процессе');
            end
            else
            begin
              res_memo = (case res_object_status
                            when -2 then f_lang(lang_code, 'КК: Отказано')
                            when -1 then f_lang(lang_code, 'КК: Доработка')
                            when 0 then f_lang(lang_code, 'КК: В очереди')
                            when 1 then f_lang(lang_code, 'КК: Рассматр-ся')
                            when 2 then f_lang(lang_code, 'КК: Подтверждено')
                          end);
            end
          end
        end
        res_memo = kf_name || ': ' || res_memo;

        suspend;
      end
    end
  end
end
