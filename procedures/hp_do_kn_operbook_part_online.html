CREATE PROCEDURE HP_DO_KN_OPERBOOK_PART_ONLINE(
    K_ID D_ID)
RETURNS (
    RES_K_ID D_ID)
AS
declare variable cnt d_integer;
declare variable rows_kol d_integer;
begin
  cnt = 0;
  rows_kol = 100;

  if (k_id = 0 and
      current_date = cast('27.05.2019' as date)) then
  begin
    -- 27.05.2019 создаем записи по займам где нет записи как is_pmt = 1
    -- действите связано с подготовлением баланса на 27.05.2019

    insert into kn_oksummary (oks_from_kc_id, oks_date_begin, oks_from_k_id, oks_is_pmt)
    select kc_id, current_date, ks_from_k_id, 1
    from kontstate
    inner join knclient on (ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0)
    left join kn_oksummary on (kc_id = oks_from_kc_id and
          oks_date_begin = current_date)
    where ks_date_begin <= current_date and
          ks_date_end >= current_date and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          oks_from_kc_id is null;
  end

  for select ks_from_k_id
      from kontstate
      where ks_state = 1 and
            ks_date_begin <= current_date and
            ks_date_end >= current_date and
            ks_ol_is_delete = 0 and
            ks_from_k_id > :k_id
      order by ks_from_k_id
      rows :rows_kol
      into :res_k_id
  do
  begin
    cnt = cnt + 1;
    execute procedure hp_do_kn_operbook_one_online(:res_k_id, current_date, null, null);
  end
  if (cnt < rows_kol) then
    res_k_id = null;
  suspend;
end
