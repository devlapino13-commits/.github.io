Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_UXMAINMENU_LIST(
    UN_ID D_ID,
    PARENT_ID D_INTEGER = null)
RETURNS (
    ML_ID D_INTEGER,
    ML_CODE D_STR_100,
    ML_PARENT_CODE D_STR_100,
    ML_CAPTION D_STR_255,
    ML_HINT D_STR_LONG,
    ML_UA_CODE D_STR_100)
AS
declare variable ux_action_found d_bool;
declare variable tmp_ml_id d_integer;
begin
  for select ml_id, ml_code, ml_parent_code, ml_caption, ml_description, ml_ua_code
  from menulist
  where ml_parent_id is not distinct from :parent_id
  into ml_id, ml_code, ml_parent_code, ml_caption, ml_hint, ml_ua_code
  do
  begin
    if (ml_ua_code is not null) then
      suspend;
    else
    begin
      ux_action_found = 0;
      select ux_action_found
      from hp_uxmainmenu_find_ux(:ml_id, :un_id)
      into ux_action_found;

      if (ux_action_found = 1) then
      begin
        suspend;

        tmp_ml_id = ml_id;
        for select ml_id, ml_code, ml_parent_code, ml_caption, ml_hint, ml_ua_code
        from hp_uxmainmenu_list(:un_id, :tmp_ml_id)
        into :ml_id, :ml_code, ml_parent_code, :ml_caption, :ml_hint, :ml_ua_code
        do
          suspend;
      end
    end
  end
end
