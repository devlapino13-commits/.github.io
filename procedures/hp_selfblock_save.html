CREATE PROCEDURE HP_SELFBLOCK_SAVE(
    EIL_ID D_ID)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable bls_id d_id;
declare variable sbk_id d_id;
declare variable ma_id d_id;
declare variable un_id d_id;
declare variable json_full_str d_blob_text;
declare variable jso d_integer;
declare variable kbki_jso d_integer;
declare variable conditinon_jso d_integer;
declare variable conditinon_doc_date d_datetime;
declare variable conditinon_doc_date_tmp d_datetime;
declare variable conditinon_begin_date d_date;
declare variable conditinon_code d_str_10;
declare variable conditinon_code_tmp d_str_10;
declare variable block_active boolean;
declare variable sbk_date_begin d_date;
declare variable ks_state d_integer;
declare variable ma_is_active d_integer;
declare variable pr_code d_str_100;
declare variable ok_cell_main d_str_100;
declare variable c_name_fio d_str_100;
declare variable c_birth_date d_date;
declare variable c_full_address d_str_255;
declare variable c_from_reg_id d_integer;
declare variable c_pass_no d_str_code;
begin
  -- процедура обновления информации по самозапрету СКИ по методу "одного окна" из полученного от сервиса КБКИ json
  select eil_from_c_id, eil_from_bls_id, eil_object_id
  from extinfolist
  where eil_id = :eil_id
  into c_id, bls_id, ma_id;

  if (bls_id is null) then
    exception e_message 'Ошибка получения КБКИ: не получен отчет СП';

  select cast(blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id)
  into json_full_str;

  if (json_full_str is null) then
    exception e_default 'Ошибка получения КБКИ: json пустой';

  -- разбор json
  jso = json.parse(json_full_str);

  conditinon_doc_date = '01.01.1900';
  conditinon_begin_date = current_date;
  conditinon_code = 0;

  for select jso
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'КБКИ'), 1)
  into kbki_jso
  do
  begin
    for select jso
    from hp_bki_json_get_list(json.obj_get_arr(:kbki_jso, 'УсловияЗапрета.Условие'), 1)
    into conditinon_jso
    do
    begin
      conditinon_code_tmp = json.obj_get_val(conditinon_jso, 'value');
      conditinon_doc_date_tmp = json.obj_get_val(conditinon_jso, 'ДатаЗаявления') || ' ' || json.obj_get_val(conditinon_jso, 'ВремяЗаявления');

      if (conditinon_code_tmp in ('0','1','3','5')
        and conditinon_doc_date_tmp >= conditinon_doc_date)
      then
        if (conditinon_doc_date_tmp > conditinon_doc_date or conditinon_code_tmp <> '0') then   -- RFCRU-3288 -- при одинаковой дате, приоритет за запретом
        begin
          conditinon_doc_date = conditinon_doc_date_tmp;
          conditinon_code = conditinon_code_tmp;
          conditinon_begin_date = json.obj_get_val(conditinon_jso, 'НачалоДействия');
        end
    end
  end

  block_active = not conditinon_code = '0';

  select sbk_id, sbk_date_begin
  from selfblock
  where sbk_from_c_id = :c_id
    and current_date between sbk_date_begin and sbk_date_end
  into sbk_id, sbk_date_begin;

  if (sbk_id is null) then
  begin
    if (block_active) then
      insert into selfblock (sbk_from_c_id, sbk_date_begin)
      values (:c_id, :conditinon_begin_date);
  end
  else
  begin
    if (not block_active and sbk_date_begin <= conditinon_begin_date) then
      update selfblock
      set sbk_date_end = :conditinon_begin_date - 1
      where sbk_id = :sbk_id;
  end


  if (f_is_selfblock(c_id, current_date) = 1) then -- есть действующий самозапрет
  begin
    select pr_code, ma_is_active
    from miniapp, product
    where ma_id = :ma_id
      and pr_id = ma_from_pr_id
    order by ma_number desc
    rows 1
    into pr_code, ma_is_active;

    if (pr_code not containing '.zalog' and ma_is_active > 0) then -- если залоговый, то самозапрет ингонорируем
    begin
      un_id = coalesce(uxf$usr_un(null, 0),1000000000000);

      -- проверим контракт
      select ks_state, k_id
      from kontract, kontstate
      where ks_from_k_id = k_id
        and k_from_ma_app_id = :ma_id
        and current_date between ks_date_begin and ks_date_end
        and ks_ol_is_delete = 0
        and ks_state >= 0
      into ks_state, k_id;

      if (:k_id is not null) then
      begin
        if (ks_state = 1) then  -- убедимся что не выдан
          exit;

        insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                                 krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
        values (23, 1000000000000, :k_id, 0, null, current_timestamp, current_date, current_date, 0);
    
        update kontract
        set k_status = -1,
            k_user_id = 1000000000000
        where k_id = :k_id;
      end
      else -- отказ по заявке
      begin
        insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                                 krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
        values (23, 1000000000000, null, 0, :ma_id, current_timestamp, current_date, current_date, 0);

        execute procedure hp_miniapp_status_change(:ma_id, 1000000000000, 0);
      end

      --список Стоп-лист
      if (not exists(select 1 from moratory
                    where m_from_c_id = :c_id
                      and m_is_active = 1
                      and m_is_black = 2 --RFCRU-4840
                      and current_date between m_date_begin and m_date_end))
      then
      begin
        select c_name_fio, c_pass_no, c_birth_date, adl_address_str, c_birth_date, c_from_reg_id
        from clients c, addrclient, addrlist
        where c.c_id = :c_id
          and c_type = 0
          and adc_from_c_id = c_id
          and adc_address_type = 1
          and adc_from_adl_id = adl_id
        into c_name_fio, c_pass_no, c_birth_date, c_full_address, c_birth_date, c_from_reg_id;

        insert into moratory (m_id, m_number, m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end, m_month,
                              m_is_active, m_comment_group, m_comment_officer, m_comment_manager, m_client_fio,
                              m_client_pass_no, m_client_address, m_client_birth_date, m_our_clients, m_from_reg_id,
                              m_no, m_datetime, m_datetime_edit, m_user_create, m_user_edit, m_db_code, m_original_id)
        values (-255, 1, :c_id, null, 2, current_date, '31.12.9999', 0, 1, null, 'Установлен самозапрет на кредитование',
                null, :c_name_fio, :c_pass_no, :c_full_address, :c_birth_date, 1, :c_from_reg_id, null, current_timestamp,
                current_timestamp, null, null, null, null);
      end

      select ok_cell_main
      from continfo
      where cci_object_kind = 0
      and cci_object_id = :c_id
      rows 1
      into ok_cell_main;

      if (coalesce(ok_cell_main, '') <> '') then
        execute procedure hp_send_quick_message('sms', :ok_cell_main, '', 'В займе отказано в связи с наличием самозапрета на кредитование', '', -1, 1000000000000);

      execute procedure fsrobot_insert_event('DoMailing', 'Push_selfblock_miniapp_cancel', f_format('%1;%2;', :ma_id, :c_id), :un_id); -- RFCRU-3453
    end
  end
end
