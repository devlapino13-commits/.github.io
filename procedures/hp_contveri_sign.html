CREATE PROCEDURE HP_CONTVERI_SIGN(
    CONT_KIND D_STR_CODE,
    CONT_VALUE D_STR_100,
    SIGN_KIND D_STR_CODE,
    PIN_CODE D_STR_100)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255,
    C_ID D_ID,
    NEED_SEND_SMS D_SMALLINT)
AS
declare variable cv_id d_id;
declare variable cv_status d_smallint;
declare variable db_id d_integer;
declare variable clt_id d_integer;
declare variable rd_id d_integer;
declare variable rd_id_2 d_integer;
declare variable sign_kind_code d_str_code;
declare variable cv_status_new d_integer;
declare variable is_new_contdata d_smallint;
begin
  -- Процедура подтверждения контактных данных

  need_send_sms = 0;
  res_is_error = 0;
  res_text = '';

  db_id = f_get_db_id();
  is_new_contdata =  f_get_const('new_contdata');

  -- если пин-код 111 прилетел через входящий СМС, меняем его тип подписания
  if (pin_code = '111' and sign_kind = 'sms') then sign_kind = 'sms_111';

  sign_kind_code = f_enum('edoc_sign_kind', :sign_kind, 'code');

  if (f_enum('cont_kind', cont_kind, 'code', 0) = 'cell') then
  begin
    select rd_id
    from refdata
    where rd_code = 'contactinfo.cell_add'
    into rd_id_2;
  end

  -- cont_kind могут передаваться как старый, так и новый вариант кодовых названий
  -- поэтому конверетируем их между собой
  select rd_id, cont_kind_id
  from hp_contkind_info(:cont_kind)
  into :rd_id, :cont_kind;

  if (rd_id_2 is null) then rd_id_2 = rd_id;

  if (is_new_contdata = 0) then
  begin
    -- получаем последнюю запись где совпал номер телефона и пароль
    select cv_id, cv_status, cv_object_id
    from contverify
    where cv_contact_value = :cont_value and
          cv_from_rd_id in (:rd_id, :rd_id_2) and
          (:sign_kind_code in ('sms_111', 'paper') or cv_password in (f_coder_pass(:pin_code), f_md5(:pin_code))) and
          cv_object_kind = f_objkind('clients', 'id') and
          cv_status >= 0
    order by cv_id desc, cv_status
    rows 1
    into cv_id, cv_status, c_id;

    if (cv_id is null) then
      res_text = 'Неверный пин-код';

    if (res_text = '') then
    begin
      if (cv_status = 0) then
      begin
        cv_status_new = case sign_kind_code
                        when 'paper' then 2
                        when 'sms' then 3
                        when 'sms_111' then 3
                        else 1
        end;

        update contverify
        set cv_status = :cv_status_new
        where cv_id = :cv_id;
  
        execute procedure hp_contveri_sign_sync('contverify', :cv_id, null, :sign_kind_code);

        need_send_sms = 1;
      end else
        res_text = iif(db_id <> 94401, 'Данные уже подтверждены', 'Məlumatlar artıq təstiq olunub');
     end
  end

  if (is_new_contdata = 1) then
  begin
    cv_status = null;
    cv_id = null;

    clt_id = f_cont_id(cont_value, 'null');

    -- получаем последнюю запись где совпал номер телефона и пароль
    if (clt_id is not null) then
    begin
      select cvr_id, cvr_status, cda_from_c_id
      from contveri, contdata
      where cda_from_clt_id = :clt_id and
            cda_cont_kind = :cont_kind and
            cda_id = cvr_from_cda_id and
            (:sign_kind_code in ('sms_111', 'paper') or cvr_hash_pin in (f_coder_pass(:pin_code), f_md5(:pin_code))) and
            cvr_status >= 0
      order by cvr_id desc, cvr_status
      rows 1
      into cv_id, cv_status, c_id;
    end

    if (cv_id is null) then
      res_text = 'Неверный пин-код';

    if (res_text = '' and cv_id is not null) then
    begin
      if (cv_status = 0) then
      begin
        update contveri
        set cvr_status = 1,
            cvr_sign_kind = f_enum('edoc_sign_kind', :sign_kind, 'id')
        where cvr_id = :cv_id;

        need_send_sms = 1;

        execute procedure hp_contveri_sign_sync('contveri', :cv_id, null, :sign_kind);
      end else
        res_text = iif(db_id <> 94401, 'Данные уже подтверждены', 'Məlumatlar artıq təstiq olunub');
    end
  end
  if (need_send_sms = 1) then
    res_text = iif(db_id <> 94401, 'Успешно подтвержден', 'Uğurla təsdiq olundu');

  res_is_error = 1 - need_send_sms;
  suspend;
end
