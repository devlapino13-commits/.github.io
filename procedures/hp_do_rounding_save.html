CREATE PROCEDURE HP_DO_ROUNDING_SAVE(
    OPER_DATE DATE,
    REG_ID INTEGER,
    BALACCOUNT INTEGER,
    ACCNUMBER INTEGER,
    SUMMA_VAL D_CURRENCY,
    C_ID ID_TYPE,
    UN_ID ID_TYPE)
RETURNS (
    OL_ID ID_TYPE,
    OL_ID_STR D_STR_255,
    RES_TEXT D_STR_LONG)
AS
declare variable sac_month_minus curr_type;
declare variable dt_acc d_str_20;
declare variable kt_acc d_str_20;
declare variable var_acc d_str_20;
declare variable text_comment d_str_255;
declare variable summa_val_tail d_currency;
declare variable summa_val_min d_currency;
declare variable doc_kind d_integer;
declare variable oper_subtype d_integer;
declare variable doc_no d_integer;
begin
  dt_acc = '';
  kt_acc = '';
  text_comment = 'Округление тыйыновых операций, расход с кассы';
    doc_kind = 0;
    oper_subtype = 2;
    doc_no = 0;
  select cast(acl.al_id as str_20)
  from acc_list acl
  where acl.al_from_bg_id = 80501 and
        acl.al_number = 3 and
        acl.al_from_reg_id = :reg_id
  into :dt_acc;

  select cast(acl.al_id as str_20)
  from acc_list acl
  where acl.al_from_bg_id = :balaccount and
        acl.al_number = :accnumber and
        acl.al_from_reg_id = :reg_id
  into :kt_acc;


  summa_val_min = 0;
  summa_val_tail = 0;
  summa_val_min = trunc(summa_val, 0);
  summa_val_tail = summa_val - summa_val_min;

  if (summa_val_tail > 0.5) then
  begin
    text_comment = 'Округление тыйыновых операций, приход в кассу';
    var_acc = dt_acc;
    dt_acc = kt_acc;
    kt_acc = var_acc;
    summa_val_tail = 1 - summa_val_tail;
    doc_kind = 1;
    oper_subtype = 1;
    doc_no = 1;
  end

  res_text = '';
  select ol_id, res_text, ol_id_str
  from hp_do_new_money_oper_save(:oper_date, :reg_id, :doc_kind, :oper_subtype, 0, :doc_no, null, :text_comment, 3, :c_id, 1, 0, null, :dt_acc, :kt_acc, '417', 1, :summa_val_tail, '', '', :text_comment, null, null, '', 0, null, null, :un_id)
  into :ol_id, res_text, ol_id_str;

  suspend;
end
