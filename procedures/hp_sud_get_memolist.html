Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_SUD_GET_MEMOLIST(
    REQUEST_ID D_ID,
    LAST_ON_TOP D_BOOL,
    UN_ID D_ID)
RETURNS (
    SMM_ID D_ID,
    SMM_IS_NEW D_BOOL,
    SMM_FROM_SUL_ID D_ID,
    SMM_MEMO D_STR_LONG,
    UN_FIO D_STR_100,
    SMM_DATETIME D_DATETIME,
    SMM_AUTHOR D_STR_255)
AS
begin
  if (last_on_top is null) then
    last_on_top = 1;

  smm_is_new = 1;

  if (last_on_top = 1) then
  begin
    for select smm_id, smm_from_sul_id, smm_memo, un_fio, smm_datetime,
               f_datetimetostr(smm_datetime) || ', ' || un_fio
    from sud_mainmemo, sud_mainlog, username
    where sul_id = smm_from_sul_id and
          smm_from_un_id = un_id and
          sul_from_smr_id = :request_id
    order by smm_datetime desc, smm_id desc
    into :smm_id, smm_from_sul_id, :smm_memo, :un_fio, :smm_datetime,
         :smm_author
    do
      suspend;
  end else
  begin
    for select smm_id, smm_from_sul_id, smm_memo, un_fio, smm_datetime,
               f_datetimetostr(smm_datetime) || ', ' || un_fio
    from sud_mainmemo, sud_mainlog, username
    where sul_id = smm_from_sul_id and
          smm_from_un_id = un_id and
          sul_from_smr_id = :request_id
    order by smm_datetime, smm_id
    into :smm_id, smm_from_sul_id, :smm_memo, :un_fio, :smm_datetime,
         :smm_author
    do
      suspend;
  end
end
