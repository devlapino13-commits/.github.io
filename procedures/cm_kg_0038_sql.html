CREATE PROCEDURE CM_KG_0038_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable PR_CODE varchar(100);
declare variable IS_DR smallint;
begin
  if (k_id = 109010294303800101) then exit; --RFCKG-2905

  select distinct pr.pr_code
  from product pr
  inner join kontract kont on kont.k_from_pr_id = pr.pr_id
  where kont.k_id = :k_id
  into :pr_code;

  if (pr_code containing '.sugarmill.') then
    exit;
  if (pr_code containing '.installment_goods.') then
    exit;
  if (pr_code containing '.online_loan.') then
    exit;

  if (exists(select count(*)
             from knclient kncl
             where kncl.kc_amount_approved > 0 and
                   kncl.kc_from_k_id = :k_id having count(*) > 1) and
      pr_code containing '.IND.') then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Количество клиентов больше одного. Проверьте детали контракта ';
    suspend;
  end

  is_dr = 0;
  if (un_id = 1090100001337) then -- если Шавкат
    is_dr = 1;

  -- если хоть у одного участника группы сумма займа более 100 000 сом, тогда данный заем может утвердить ДР
  -- 31.10.2022 Здравствуйте,  согласовано с ГД:
  -- С 01.11.22 все суммы до 250к будут утверждаться самими МО (сейчас только до 100к).
  if (exists(select kncl.kc_amount_approved
             from knclient kncl
             where kncl.kc_amount_approved > 100000 and
                   kncl.kc_from_k_id = :k_id)) then
  begin
    if ((is_dr = 0) and
        (not exists(select *
                    from hp_get_user_roles(:un_id)
                    where lst_roles containing 'kred_manager' or
                          lst_roles containing 'reg_manager' or
                          lst_roles containing 'direktor_deputy_executiv'))) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Данный заем должен быть утвержден менеджером офиса, т.к. сумма заема одно или нескольких участников группы превышает 100 000 сом. [UN_ID =' || coalesce(un_id, 0) || ']';
      suspend;
    end
  end

  if (exists(select kncl.kc_amount_approved
             from knclient kncl
             where kncl.kc_amount_approved > 500000 and
                   kncl.kc_from_k_id = :k_id)) then
  begin
    if ((is_dr = 0) and
        (not exists(select *
                    from hp_get_user_roles(:un_id)
                    where lst_roles containing 'reg_manager' or
                          lst_roles containing 'direktor_deputy_executiv' or
                          lst_roles containing 'kred_manager'))) then        -- RFCKG-3752
    begin
      res_error_is_critical = 1;
      res_error_text = 'Данный заем должен быть утвержден региональным менеджером, т.к. сумма заема одно или нескольких участников группы превышает 500 000 сом. [UN_ID =' || coalesce(un_id, 0) || ']';
      suspend;
    end
  end

end
