CREATE PROCEDURE HP_OVERPAYMENT(
    K_ID ID_TYPE,
    REP_DATE DATE,
    REPAY_AMOUNT CURR_TYPE,
    REPAY_FACT_OFFICE_ID INTEGER,
    REPAY_DT_ACCOUNT_NO STR_20,
    REPAY_MEMO VARCHAR(1024),
    REPAY_USER_ID ID_TYPE = 1000000000001)
RETURNS (
    OL_ID ID_TYPE,
    ERROR_TEXT STR_255)
AS
begin
  ol_id = 0;
  error_text = '';

  if (not exists(select *
                 from acc_list
                 where al_id = :repay_dt_account_no)) then
  begin
    error_text = 'Счёт № ' || repay_dt_account_no || ' - не определён.';
    suspend;
  end

  for select ol_id, error_text
      from hp_do_repayment(:k_id, 0, :rep_date, :repay_amount, 1, -- :repay_to_os,
      0, -- :repay_to_perc,
      0, -- :repay_to_fine,
      0, -- :repay_to_kommis,
      1, -- :need_save,
      0, -- :is_full_repay,
      0, -- :is_writeoff,
      :repay_fact_office_id, -- :repay_fact_office_id, -- головной офис в рф
      3, -- :repay_dt_account_kind, -- внутренний бух.счет европлата
      :repay_dt_account_no, -- :repay_dt_account_no,
      1, -- :repay_curr_rate,
      :repay_user_id, -- :repay_user_id,
      null, -- :repay_document_client,
      null, -- :repay_doc_client_name_reason,
      null, -- :repay_manager_c_id,
      0, -- :repay_from_rd_id,
      0) -- контрольная сумма)
      into :ol_id, :error_text
  do
  begin
    if ((error_text = '') and
        coalesce(repay_memo, '') <> '') then
      update operlist
      set ol_memo = :repay_memo
      where ol_id = :ol_id;
    suspend;
  end
end
