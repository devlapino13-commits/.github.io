CREATE PROCEDURE HP_OPERDELMAIN_ADD(
    K_ID D_ID,
    OL_ID D_ID,
    DEL_MEMO D_STR_LONG,
    UN_ID D_ID)
AS
declare variable ol_is_delete d_bool;
declare variable odm_id d_integer;
begin
  if (k_id is null) then
    exception e_message 'Id договора не указан';

  if (ol_id is null) then
    exception e_message 'Id операции не указан';

  select coalesce(ol_parent_id, ol_id), is_delete
  from operlist
  where ol_id = :ol_id
  into ol_id, ol_is_delete;

  if (row_count = 0) then
    exception e_message 'Операция не найдена, Id: ' || ol_id;

  if (ol_is_delete = 1) then
    exception e_message 'Операция уже аннулирована, Id: ' || ol_id;

  if (exists(select * from operdelmain
  where odm_main_ol_id = :ol_id and
        odm_state >= 0)) then
    exception e_message 'Заявка на удаление уже существует в БД, id: ' || ol_id;

  select odm_id
  from operdelmain, operdelkont
  where odm_id = odk_from_odm_id and
        odm_state in (0, 1) and
        odk_from_k_id = :k_id
  rows 1
  into odm_id;

  if (odm_id is not null) then
    exception e_message 'Договор уже участвует в заявке на удаление №: ' || odm_id ||ascii_char(13) ||
                        'Сначала нужно закрыть эту заявку';

  insert into operdelmain (odm_main_ol_id, odm_memo, odm_create_un_id)
  values (:ol_id, :del_memo, :un_id)
  returning odm_id
  into :odm_id;

  insert into operdelkont (odk_from_odm_id, odk_loan_ol_id, odk_from_k_id)
  values (:odm_id, :ol_id, :k_id);
end
