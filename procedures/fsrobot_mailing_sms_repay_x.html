CREATE PROCEDURE FSROBOT_MAILING_SMS_REPAY_X(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable ol_date date;
declare variable c_cell_no varchar(100);
declare variable c_id decimal(18,0);
declare variable kol integer;
declare variable date_format_str varchar(20);
declare variable k_id decimal(18,0);
declare variable amount decimal(18,4) = 0;
declare variable uniq_acc_str varchar(10);
declare variable amount_str varchar(20);
declare variable db_id integer;
declare variable amount_sym varchar(10);
declare variable cell_all varchar(255);
declare variable push_body d_str_long;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь

-- СМС после погашения кредита
  if (user_fio is null) then
    user_fio = '';
  res_maillist = :rep_email;

  :db_id = f_get_db_id();
  amount_sym = (case
                  when (db_id = 81001) then ' rub.'
                  when (db_id in (39801, 41701)) then 'c.'
                  when (db_id = 94401) then 'AZN'
                end);

  res_body = '';
  kol = 0;
  for select distinct ok_from_k_id, ok_ol_date
      from operkred, operlist
      where ok_from_ol_id = :rep_params and
            ol_id = :rep_params and
            ol_oper_type = 1 -- Погашение
            and
            ol_oper_subtype = 2 -- Погашение
            and
            ok_ol_is_active = 0 -- Аннулированное погашение
            and
            ok_type in (4, 9) -- Погашение
      into :k_id, :ol_date
  do
  begin

    date_format_str = f_datetostr(ol_date);

    select -sum(ok_os + ok_perc + ok_fine + ok_kommis_adv + ok_tax)
    from operkred
    where ok_from_ol_id = :rep_params and
          ok_type in (4, 9)
    into :amount;
    if (amount is null) then
      amount = 0;
    amount_str = round(amount, 0);
    if (char_length(amount_str) > 6) then
      amount_str = left(amount_str, char_length(amount_str) - 6) || ',' || left(right(amount_str, 6), 3) || ',' || right(amount_str, 3);
    else
    if (char_length(amount_str) > 3) then
      amount_str = left(amount_str, char_length(amount_str) - 3) || ',' || right(amount_str, 3);
    amount_str = amount_str || amount_sym;
    kol = kol + 1;

    cell_all = '';
    for select c_id,
               (select cont_value
                from hp_contdata_get_cell_main(0, c_id))
        from knclient, clients
        where kc_amount_approved > 0 and
              kc_from_k_id = :k_id and
              kc_from_c_id = c_id
        into c_id, c_cell_no
    do
    begin
      if (exists(select *
                 from sms_ignore sms
                 where sms.si_from_c_id = :c_id)) then exit;

      res_c_id = c_id;
      if ((right(coalesce(c_cell_no, '000000'), 6) <> '000000') and
          (position(c_cell_no, cell_all) <= 0)) then
      begin
        cell_all = cell_all || c_cell_no || ';';
        res_maillist = :c_cell_no;

        uniq_acc_str = (select uniq_acc_str
                        from hp_show_clientuniq_acc_no(:c_id));

        -- Первый СМС, о транзакции
        res_body = '';
        push_body = '';
        res_subject = 'Погашение кредита, аннулирование';
        res_body = f_format('%1: %2; %3%4 %5', iif(db_id <> 94401, 'Lisevoi chet', 'Hesab N'),
                                                uniq_acc_str,
                                                iif(db_id <> 94401, 'Vashe pogashenie annulirovano.', 'MolBulakdan kredit legv olundu: '),
                                                amount_str,
                                                :date_format_str);

        if (db_id = 81001 and f_clientcom_allowed(c_id, 0, 0, 1) = 1) then
        begin
            push_body = f_format('%1:%2; %3 %4; %5', iif(db_id <> 94401, 'Лицевой счет', 'Hesab N'),
                                                uniq_acc_str,
                                                iif(db_id <> 94401, 'Ваше погашение аннулировано. ', 'MolBulakdan kredit legv olundu: '),
                                                amount_str,
                                                :date_format_str);
            -- отправляем в КМП пуш-уведомление
            execute procedure mc_send_push('clients', :c_id, :res_subject, :push_body);
        end
        if (db_id <> 81001 or f_clientcom_allowed(c_id, 0, 0, 1) = 1) then
          suspend;
      end
    end
  end
end
