CREATE PROCEDURE HP_COPY_REPORT(
    RL_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RL_ID_NEW INTEGER)
AS
declare variable PM_OBJECT_KIND integer;
declare variable PM_OBJECT_ID decimal(18,0);
declare variable PM_SUBJECT_KIND integer;
declare variable PM_DATE_BEGIN date;
declare variable PM_DATE_END date;
declare variable PM_ACCESS_KIND integer;
declare variable UNIQTAB_ROLENAME integer;
declare variable UNIQTAB_MENULIST integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;

  select max(rl_id) + 1
  from reportlist
  into :rl_id_new;
  if (rl_id_new is null) then
    rl_id_new = 1;

  insert into reportlist (rl_id, parent, lb, rb, rl_name_rus, rl_hint_rus, rl_name_eng, rl_hint_eng, rl_type, rl_hotkey,
                          rl_data_rus, rl_data_eng, rl_is_ready, rl_user_id, rl_date_kind, rl_report_kind, rl_sql,
                          rl_filter, rl_order_no)
  select :rl_id_new, parent, lb, rb, rl_name_rus || ' копия', rl_hint_rus, rl_name_eng || ' copy', rl_hint_eng,
         rl_type, rl_hotkey, rl_data_rus, rl_data_eng, rl_is_ready, :un_id, rl_date_kind, rl_report_kind, rl_sql,
         rl_filter, rl_order_no
  from reportlist
  where rl_id = :rl_id;

  insert into reptemplates (rt_from_rl_id, rt_number, rt_name, rt_name_eng, rt_user_id, rt_date_created, rt_for_all,
                            rt_default, rt_header, rt_footer, rt_autowidth, rt_page_orientation)
  select :rl_id_new, rt_number, rt_name, rt_name_eng, rt_user_id, rt_date_created, 0, 0, rt_header,
         rt_footer, rt_autowidth, rt_page_orientation
  from reptemplates
  where rt_from_rl_id = :rl_id;

  insert into repfields (rf_from_rl_id, rf_number, rf_field_name, rf_name_rus, rf_hint_rus, rf_name_eng, rf_hint_eng,
                         rf_type, rf_width, rf_can_group, rf_agregate, rf_lookup_sql)
  select :rl_id_new, rf_number, rf_field_name, rf_name_rus, rf_hint_rus, rf_name_eng, rf_hint_eng, rf_type, rf_width,
         rf_can_group, rf_agregate, rf_lookup_sql
  from repfields
  where rf_from_rl_id = :rl_id;

  insert into reptempfields (rtf_from_rl_id, rtf_from_rt_number, rtf_from_rf_number, rtf_order_no, rtf_group_number,
                             rtf_agr_function, rtf_width, rtf_sort_index, rtf_sort_order)
  select :rl_id_new, rtf_from_rt_number, rtf_from_rf_number, rtf_order_no, rtf_group_number, rtf_agr_function,
         rtf_width, rtf_sort_index, rtf_sort_order
  from reptempfields
  where rtf_from_rl_id = :rl_id;

  insert into acrepline (arl_from_arm_id, arl_number, parent, lb, rb, arl_name, arl_need_child_sum, arl_order_no,
                         arl_formula)
  select :rl_id_new, arl_number, parent, lb, rb, arl_name, arl_need_child_sum, arl_order_no, arl_formula
  from acrepline
  where arl_from_arm_id = :rl_id;

  insert into acrepcalc (arc_from_arm_id, arc_from_arl_number, arc_number, arc_calc_type, arc_plus_minus, arc_mask_dt,
                         arc_mask_kt, arc_action, arc_action_koef, arc_dt_no, arc_kt_no)
  select :rl_id_new, arc_from_arl_number, arc_number, arc_calc_type, arc_plus_minus, arc_mask_dt, arc_mask_kt,
         arc_action, arc_action_koef, arc_dt_no, arc_kt_no
  from acrepcalc
  where arc_from_arm_id = :rl_id;
