CREATE PROCEDURE HP_EXTINFO_TO_SEND_EQUIFAX_40(
    C_ID ID_TYPE,
    OBJECT_KIND INTEGER,
    OBJECT_ID ID_TYPE)
RETURNS (
    JSON_VALUE D_STR_8100)
AS
declare variable lastname d_str_100;
declare variable firstname d_str_100;
declare variable middlename d_str_100;
declare variable birthday d_str_20;
declare variable consentdate d_str_20;
declare variable admcodeinform d_str_10;
declare variable json_docs d_str_8100;
declare variable json_fio d_str_8100;
declare variable doc_country d_str_10;
declare variable doc_type d_str_10;
declare variable doc_serial d_str_10;
declare variable doc_number d_str_20;
declare variable doc_date d_str_20;
declare variable doc_who d_str_255;
declare variable request_reason_code d_str_10;
declare variable consent_reason_code d_str_10;
declare variable app_sum decimal(15,2);
declare variable consenthash d_str_64;
declare variable consentperiod d_str_10;
declare variable app_currency d_str_10;
declare variable contractdate d_str_20;
declare variable doc_enddate d_str_20;
declare variable c_inn d_str_20;
declare variable inn_checked d_bool;
declare variable req_type d_str_code;
declare variable ogrn d_str_code;
begin
  json_docs = '';
  json_fio = '';

  for select distinct doc_country, doc_type, doc_serial, doc_number, doc_date, doc_who, doc_enddate
  from hp_bki_get_info4req_doc(:c_id)
  into :doc_country, :doc_type, :doc_serial, :doc_number, :doc_date, :doc_who, :doc_enddate
  do
  begin
      json_docs = json_docs || iif(json_docs <> '', ',', '') ||
          '{' ||
              f_json_make_pair('country', doc_country,  'str',  1) ||
              f_json_make_pair('type',    doc_type,     'str',  1) ||
              f_json_make_pair('serial',  doc_serial,   'str',  1) ||
              f_json_make_pair('number',  doc_number,   'str',  1) ||
              f_json_make_pair('who',     doc_who,      'str',  1) ||
              f_json_make_pair('date',    doc_date,     'date', 1) ||
              f_json_make_pair('endDate', doc_enddate,  'date') ||
          '}';
  end

  for select lastname, firstname, middlename
  from hp_bki_get_info4req_fio(:c_id)
  into :lastname, :firstname, :middlename
  do
  begin
    json_fio = json_fio || iif(json_fio <> '', ',', '') ||
          '{' ||
              f_json_make_pair('last',  lastname,  'str', 1) ||
              f_json_make_pair('first', firstname, 'str', 1) ||
              f_json_make_pair('middle', middlename, 'str') ||
          '}';
  end 

  select birthday, consentdate, admcodeinform, request_reason_code,
         consent_reason_code, app_sum, consenthash, consentperiod, app_currency, contractdate,
         c_inn, inn_checked, req_type, ogrn
  from hp_bki_get_info4req(:c_id, :object_kind, :object_id)
  into :birthday, :consentdate, :admcodeinform, :request_reason_code,
       :consent_reason_code, :app_sum, :consenthash, :consentperiod, :app_currency, :contractdate,
       :c_inn, :inn_checked, :req_type, ogrn;

  json_value = '{"request":{' ||
                    '"_private":{' ||
                        '"name":[' || json_fio || '],' ||
                        f_json_make_pair('birthday', birthday, 'str', 1) ||
                        trim(iif(coalesce(c_inn, '') <> '',
                            '"inn":{' ||
                              f_json_make_pair('code', 1, 'str' ,1) ||
                              f_json_make_pair('no', c_inn, 'str' ,1) ||
                              f_json_make_pair('checked', inn_checked, 'str') ||
                            '},',
                        ''))||
                        trim(iif(coalesce(ogrn, '') <> '', f_json_make_pair('ogrnip', ogrn, 'str', 1), '')) ||
                        '"doc":[' || json_docs || ']' ||
                    '},' ||
                    '"reason":{' ||
                        f_json_make_pair('code', request_reason_code, 'str') ||
                    '},' ||
                    '"consent":{' ||
                      '"reason":[{' ||
                          f_json_make_pair('code', consent_reason_code, 'str') ||
                      '}],' ||
                      f_json_make_pair('date', consentdate, 'str', 1) ||
                      f_json_make_pair('contractDate', contractdate, 'str', 1) ||
                      f_json_make_pair('period', consentperiod, 'str', 1) ||
                      f_json_make_pair('hash', consenthash, 'str', 1) ||
                      f_json_make_pair('admcodeInform', admcodeInform, 'str') ||
                    '},' ||
                    '"application":{' ||
                        f_json_make_pair('credCurrency', app_currency, 'str', 1) ||
                        f_json_make_pair('credSum', app_sum, 'str') ||
                    '},' ||
                    f_json_make_pair('type', req_type, 'str') ||
                  '}' ||
               '}';

  if (consenthash is not null) then
    suspend;
end
