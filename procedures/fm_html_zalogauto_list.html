CREATE PROCEDURE FM_HTML_ZALOGAUTO_LIST(
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable lang d_str_lang;
declare variable can_add d_bool;
declare variable can_edit d_bool;
declare variable tmp_text d_str_8100;
declare variable tmp_str d_str_100;
declare variable tmp_id d_id;
declare variable tmp_int d_integer;
declare variable zap_status d_smallint;
declare variable zap_status_str d_str_code;
declare variable status_color d_str_code;
begin
  lang = f_get_lang_code(un_id, null);
  can_add = 1;
  object_kind = f_objkind(object_kind, 'name');

  res_text = (select res_style from fm_html_styles) || '<body>';

  if (object_kind = 'miniapp') then
  begin
    select ma_no
    from miniapp
    where ma_id = :object_id and
      ma_is_active = 1 and
      ma_date_end >= '31.12.2999'
    into tmp_str;
    res_text = BLOB_APPEND(res_text, f_format('<h1>%1 №%2</h1><hr>', f_lang(lang, 'Автозалоги по заявке'), tmp_str));
  end

  if (object_kind = 'kontract') then
  begin
    select k_kontract_no
    from kontract
    where k_id = :object_id
    into tmp_str;
    res_text = BLOB_APPEND(res_text, f_format('<h1>%1 №%2</h1><hr>', f_lang(lang, 'Автозалоги по договору'), tmp_str));
  end

  for select za_id, za_from_k_id, za_from_c_id, za_from_z_id, za_orig_sts_id,
    za_orig_pts_id, za_new_sts_id, za_new_pts_id, za_from_zal_id
  from zalogauto
  where
    :object_id = za_from_ma_id or
    :object_id = za_from_k_id
  as cursor za do
  begin
    tmp_text = '';

    -- Статус запроса
    zap_status = -3;
    select zap_status
    from zalogautoprof
    where zap_from_za_id = za.za_id
    into zap_status;
    zap_status_str =
      case zap_status
        when -3 then 'Новый (не отправлен)'
        when -2 then 'Отказ'
        when -1 then 'Доработка'
        when  0 then 'В очереди'
        when  1 then 'На рассмотрении'
        when  2 then 'Одобрен'
        else 'Ошибка (отсутствует)'
      end;

    status_color = 'Black';
    if (zap_status = -2) then
      status_color = 'Red';
    else if (zap_status = -1) then
      status_color = 'Orange';
    else if (zap_status = 2) then
      status_color = 'Green';

    tmp_text = tmp_text || f_format('%1: <font color="%2"><b>%3</b></font><br>',
      f_lang(lang, 'Статус'), status_color, zap_status_str);

    can_edit = 0;
    if (zap_status in (-3, -1)) then
      can_edit = 1;

    if (can_edit = 1) then
    begin
      tmp_text = tmp_text || f_uri_sql(f_lang(lang, 'Отправить на рассмотрение'),
        'zalogauto_send', 'zalogauto', za.za_id, 'none',
        f_lang(lang, 'Подтвердите отправку залога на рассмотрение.' ||
        ' После этого редактирование будет запрещено.'),
        null, null, null, null, null, 'dialog', null) || '<br><br>';
    end

    -- Если есть любой залог кроме отказанных, запрещаем добавлять новый
    if (zap_status <> -2) then
      can_add = 0;

    -- Залогодатель
    tmp_str = f_lang(lang, 'Не указан');
    if (za.za_from_c_id is not null) then
      select c_name_fio from clients
      where c_id = za.za_from_c_id
      into tmp_str;
    tmp_text = tmp_text || f_format('%1: <b>%2</b><br>',
      f_lang(lang, 'Залогодатель'), tmp_str);

    -- Стоимость
    tmp_str = f_lang(lang, 'Не указана');
    if (za.za_from_zal_id is not null) then
    begin
      select f_numtostr(zal_cost, 2)
      from zalogautolist
      where zal_id = za.za_from_zal_id
      into tmp_str;
      tmp_text = tmp_text || f_format('%1: <b>%2</b><br>', f_lang(lang, 'Стоимость'), tmp_str);
    end

    tmp_text = tmp_text || f_format('<b>%1</b><br>',
      f_lang(lang, 'Документы до регистрации'));

    -- ПТС до регистрации и залог
    tmp_str = f_lang(lang, 'Не указан');
    tmp_id = null;
    if (za.za_orig_pts_id is not null) then
      select cds_from_drl_id, f_format('%1 №%2 от %3', pk_name, cds_doc_no,
        f_datetostr(cds_doc_date_begin))
      from clientdocs, passkind
      where cds_id = za.za_orig_pts_id
        and cds_from_pk_id = pk_id
      into tmp_id, tmp_str;
    tmp_text = tmp_text || f_format('%1: %2<br>', f_lang(lang, 'ПТС'), tmp_str);

    tmp_str = f_lang(lang, 'Не указано');
    if (za.za_from_z_id is not null) then
    begin
      -- Наименование залога берем из Zalog
      select z_name from zalog
      where z_id = za.za_from_z_id
      into tmp_str;
    end
    else if (tmp_id is not null) then
    begin
      -- Наименование залога, берем из Docrecog (через ClientDocs)
      tmp_str = f_docrecog_value(tmp_id, 'model', f_lang(lang, 'Не указано'));
    end
    tmp_text =
      f_format('%1: <b>%2</b><br>', f_lang(lang, 'Наименование залога'), tmp_str) || tmp_text;

    -- СТС до регистрации
    tmp_str = f_lang(lang, 'Не указан');
    if (za.za_orig_sts_id is not null) then
      select f_format('%1 №%2 от %3', pk_name, cds_doc_no,
        f_datetostr(cds_doc_date_begin))
      from clientdocs, passkind
      where cds_id = za.za_orig_sts_id
        and cds_from_pk_id = pk_id
      into tmp_str;
      tmp_text = tmp_text || f_format('%1: %2<br>', f_lang(lang, 'СТС'), tmp_str);

    if (za.za_new_sts_id is not null or za.za_new_pts_id is not null) then
    begin
      tmp_text = tmp_text || f_format('<b>%1</b><br>', f_lang(lang, 'Документы после регистрации'));
      -- ПТС после регистрации
      tmp_str = f_lang(lang, 'Не указан');
      if (za.za_new_pts_id is not null) then
        select f_format('%1 №%2 от %3', pk_name, cds_doc_no,
          f_datetostr(cds_doc_date_begin))
        from clientdocs, passkind
        where cds_id = za.za_new_pts_id
          and cds_from_pk_id = pk_id
        into tmp_str;
      tmp_text = tmp_text || tmp_str || '<br>';

      -- СТС после регистрации
      tmp_str = f_lang(lang, 'Не указан');
      if (za.za_new_sts_id is not null) then
        select f_format('%1 №%2 от %3', pk_name, cds_doc_no,
          f_datetostr(cds_doc_date_begin))
        from clientdocs, passkind
        where cds_id = za.za_new_sts_id
          and cds_from_pk_id = pk_id
        into tmp_str;
      tmp_text = tmp_text || tmp_str || '<br>';
    end

    if (can_edit = 1) then
    begin
      tmp_text = tmp_text ||
        f_uri_open(f_lang(lang, 'Редактировать залог'),'zalogauto_edit',
          object_kind, object_id, f_lang(lang, 'Редактирование автозалога'),
          'za_id=' || za.za_id, un_id) || '<br>';
    end

    -- Фото залога
    tmp_int = 0;
    select count(*)
    from zalogautoblob, blobstore
    where
      zab_from_zal_id = za.za_from_zal_id and
      zab_from_bls_id = bls_id and
      bls_source_system = 1 -- Закрепленные в МП
    into tmp_int;

    status_color = 'Green';
    if (tmp_int < 6 or tmp_int > 10) then
      status_color = 'Red';

    tmp_text = tmp_text ||
      f_format('<br><b>%1</b><br>', f_lang(lang, 'Фотографии залога')) ||
      f_format('%1: <font color="%2"><b>%3</b></font> %4<br>',
        f_lang(lang, 'Закреплено фото'), status_color, tmp_int,
        f_lang(lang, '(минимум 6, максимум 10)'));

    if (can_edit = 1) then
    begin
      tmp_text = tmp_text ||
        f_uri_open(f_lang(lang, 'Редактировать фото'),'zalogautoblob_edit',
        'zalogautoblob', :za.za_id, f_lang(lang, 'Редактирование фото'),
        '', un_id) || '<br>';
    end
    res_text = BLOB_APPEND(res_text, tmp_text || '<hr>');

  end

  if (can_add = 1) then
  begin
    res_text = BLOB_APPEND(res_text, f_format('%1. %2',
      f_lang(lang, 'Автозалог еще не добавлен'),
      f_uri_open(f_lang(lang, 'Добавить новый'), 'zalogauto_edit', object_kind,
      object_id, f_lang(lang, 'Добавление автозалога'), 'za_id=', un_id)));
  end

  res_text = BLOB_APPEND(res_text, '</body>');

  suspend;
end
