CREATE PROCEDURE HP_PAYMET_MONITORING(
    KC_ID D_ID)
RETURNS (
    K_KONTRACT_NO_1 D_STR_100,
    K_DISB_DATE D_DATE,
    K_PERIOD D_INTEGER,
    K_AMOUNT_APPROVED D_INTEGER,
    KS_STATE D_STR_10,
    PR_NAME D_STR_100,
    DELINQ_DAYS_QTY D_INTEGER,
    BALANCE_OS D_INTEGER,
    BALANCE_PERC D_INTEGER,
    PTC_SALDO D_INTEGER,
    SH_CALC_FINE D_INTEGER)
AS
declare variable k_id_2 d_id;
declare variable c_id d_id;
declare variable ptc_date d_date;
declare variable ptc_number d_integer;
declare variable ptc_sh_date d_date;
declare variable ptc_kind d_integer;
declare variable ptc_calc d_integer;
begin
   ptc_saldo = 0;
      select kc.kc_from_c_id
      from knclient kc
      where kc.kc_id= :kc_id
      into  :c_id;

  for select kt.k_id, kt.k_kontract_no, kt.k_disb_date, kt.k_period,
             iif(ks.ks_state = 1, 'Активный', 'Закрытый'), kt.k_amount_approved, pr.pr_name
      from kontract kt
      inner join kontstate ks on ks.ks_from_k_id = kt.k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      inner join knclient kc on kc.kc_from_k_id = kt.k_id
      where current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_state in (1, 2) and
            ks.ks_ol_is_delete = 0 and
            kt.k_amount_approved > 0 and
            kc.kc_from_c_id = :c_id
      order by k_disb_date asc
      into  :k_id_2 ,:k_kontract_no_1, :k_disb_date, :k_period, :ks_state, :k_amount_approved, :pr_name
  do
  begin
    delinq_days_qty = 0;
    balance_os = 0;
    balance_perc = 0;

    select iif((cast(current_date as date) - oks.oks_delinq_date < 0), 0, cast(current_date as date) - oks.oks_delinq_date) -- числоднейактивнойпросрочки
    from oksummary oks
    where current_date between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_from_k_id = :k_id_2 and
          oks_close_state = 0
    into :delinq_days_qty;

    select sum(zad_os), sum(zad_perc) , sum(sh_calc_fine)
    from aarep_make_zad_list(:k_id_2, current_date, 0)
    where sh_date is not null
    into :balance_os, :balance_perc, :sh_calc_fine;


    for select ptc_date, ptc_number, ptc_sh_date, ptc_kind, ptc_calc
        from pointcalc
        inner join kontract on ptc_from_k_id = k_id
        where --ptc_from_c_id = :c_id and
              ptc_from_k_id = :k_id_2 and
              ptc_date <= current_date and
              ptc_type = 0
         order by k_disb_date asc
        into ptc_date, ptc_number, ptc_sh_date, ptc_kind, ptc_calc
    do
    begin
      ptc_saldo = ptc_saldo + ptc_calc;
    end

     suspend;
  end
  end
