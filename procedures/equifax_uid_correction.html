CREATE PROCEDURE EQUIFAX_UID_CORRECTION(
    FILE_NAME D_STR_100,
    SHOW_ONLY_PROBLEMS D_SMALLINT = 0)
RETURNS (
    CRED_NO D_STR_100,
    CRED_DATE D_DATE,
    CRED_TYPE D_INTEGER,
    "update" D_DATE,
    CRED_ACTIVE D_INTEGER,
    CRED_ACTIVE_DATE D_DATE,
    UID D_STR_100,
    PARENT_UID D_STR_100,
    GUARANTEE_UID D_STR_100,
    GUARANTEE_NO D_STR_100,
    GUARANTEE_DATE D_DATE)
AS
declare variable PROBLEM_TYPE D_STR_100;
declare variable K_ID D_ID;
declare variable CLIENT_KC_ID D_ID;
declare variable PORUCH_KC_ID D_ID;
declare variable EPR_ID D_ID;
declare variable IS_PROBLEM D_SMALLINT;
declare variable KONTRACT_NO D_STR_100;
declare variable KC_CLIENT_NUMBER D_INTEGER;
declare variable KC_PORUCH_NUMBER D_INTEGER;
begin
  if (file_name containing 'no_uid') then
    problem_type = 'no_uid';
  else if (file_name containing 'no_parent_uid') then
    problem_type = 'no_parent_uid';

  for select eup_cred_no, eup_cred_date, eup_cred_type, eup_update, eup_cred_active, eup_cred_active_date, eup_uid,
             eup_parent_uid, eup_guarantee_uid, eup_guarantee_no, eup_guarantee_date
      from equifax_uid_problem
      order by eup_id
      --rows 100
      into :cred_no, :cred_date, :cred_type, :"update", :cred_active, :cred_active_date, :uid,
           :parent_uid, :guarantee_uid, :guarantee_no, :guarantee_date
  do
  begin
    is_problem = 0;

    kontract_no = '';
    kc_client_number = 0;
    kc_poruch_number = 0;
    uid = '';
    client_kc_id = 0;
    poruch_kc_id = 0;
    epr_id = 0;

    if ((cred_type = 90) and (cred_no not containing 'G')) then continue;

    kontract_no = substring(cred_no from 1 for 14);
    kc_client_number = substring(cred_no from 16 for 1);

    if (cred_type = 90) then
        kc_poruch_number = substring(cred_no from 19 for 1);

    select kont.k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id;

    select kncl.kc_id
    from kontract kont
    inner join knclient kncl on kncl.kc_from_k_id = kont.k_id and kncl.kc_amount_approved > 0
    where ((kont.k_base_k_id = :k_id) or (kont.k_id = :k_id))
    and kncl.kc_number = :kc_client_number
    into :client_kc_id;

    parent_uid = '';

    --if (problem_type = 'no_uid') then
    --begin
        if (cred_type = 90) then
        begin
            parent_uid = f_uuid_cbr(f_uuid('knclient', :client_kc_id));

            select kncl.kc_id
            from kontract kont
            inner join knclient kncl on kncl.kc_from_k_id = kont.k_id and kncl.kc_amount_approved > 0
            where ((kont.k_base_k_id = :k_id) or (kont.k_id = :k_id))
            and kncl.kc_number = :kc_poruch_number
            into :poruch_kc_id;
        
            select ep.epr_id
            from equifax_poruch ep
            where ep.epr_client_kc_id = :client_kc_id
            and ep.epr_poruch_kc_id = :poruch_kc_id
            into :epr_id;
        
            if (epr_id <> 0) then
                uid = f_uuid_cbr(f_uuid('equifax_poruch', :epr_id));
            else
                is_problem = 1;
        end
        else if (cred_type = 19) then
        begin
            if (client_kc_id <> 0) then
                uid = f_uuid_cbr(f_uuid('knclient', :client_kc_id));
            else
                is_problem = 1;
        end
    --end
    --else if (problem_type = 'no_parent_uid') then
    --begin
    --   if (client_kc_id <> 0) then
    --        parent_uid = f_uuid_cbr(f_uuid('knclient', :client_kc_id));
    --    else
    --        is_problem = 1;
    --end


    if ((is_problem = 0) and (show_only_problems = 0)) then
        suspend;
    else if ((is_problem = 1) and (show_only_problems = 1)) then
        suspend;
  end
end
