CREATE PROCEDURE AAREP_TABEL_CHECK(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    SAT_FROM_C_ID DECIMAL(18,0),
    SAT_C_NAME_FIO VARCHAR(100),
    SAT_YEAR INTEGER,
    SAT_MONTH INTEGER,
    SAT_TABEL_DAYS DECIMAL(18,4),
    SAT_ORDER_DAYS DECIMAL(18,4),
    SAT_REGION_NAME VARCHAR(100),
    SAT_JOBNAME VARCHAR(100),
    SAT_STATUS VARCHAR(100))
AS
declare variable dd date;
begin
  for select sat_from_c_id, sat_year, sat_month, sum(sat_tabel_days), sum(sat_order_days)
      from aarep_tabel_check_tmp(:rep_region_id, :rep_date_end)
      group by sat_from_c_id, sat_year, sat_month
      having sum(sat_tabel_days) <> sum(sat_order_days)
      order by sat_from_c_id, sat_year, sat_month
      into :sat_from_c_id, :sat_year, :sat_month, :sat_tabel_days, :sat_order_days
  do
  begin
    select c_name_fio
    from clients
    where c_id = :sat_from_c_id
    into :sat_c_name_fio;

    if (sat_month < 12) then
      dd = cast('01.' || lpad((sat_month + 1), 2, '0') || '.' || sat_year as date);
    else
      dd = cast('31.' || sat_month || '.' || sat_year as date);
    dd = dd - 1;

    sat_region_name = '';
    sat_jobname = '';
    sat_status = '';
    select reg_code || ' ' || reg_name, jl_name, iif(ps_is_working = 0, 'Уволен', '')
    from hr_pstatus, region, joblist
    where ps_from_c_id = :sat_from_c_id and
          ps_date_begin <= :dd and
          ps_date_end >= :dd and
          ps_status = 1 and
          ps_from_reg_id = reg_unicode and
          ps_from_jl_id = jl_id
    order by ps_is_working desc
    rows 1
    into :sat_region_name, :sat_jobname, :sat_status;

    suspend;
  end
end
