CREATE PROCEDURE HP_EDOC_WEB_UNSIGN_LIST(
    C_ID D_ID)
RETURNS (
    RES_JSON D_STR_8100)
AS
declare variable EDL_ID D_ID;
declare variable EDC_UUID D_STR_LONG;
declare variable EDC_NAME D_STR_255;
declare variable EDB_ID D_ID;
declare variable BLN_NAME D_STR_255;
declare variable C_NAME D_STR_100;
declare variable MSG_TEXT D_STR_100;
declare variable DOC_LIST D_STR_8100;
declare variable BLANK_LIST D_STR_8100;
declare variable BLANK_QTY D_INTEGER;
declare variable SIGN_BUTTON_CAPTION d_str_100;
begin
  c_name = '';
  msg_text = 'Документы для подписания';

  select c_name_fio
  from clients 
  where c_id = :c_id
  into c_name;

  doc_list = '';
  for select f_uuid('edoc_sign', eds_id),
             edc.edc_name,
             edl.edl_id
  from edoc_sign eds
  inner join edoc_list edl on edl.edl_id = eds.eds_from_edl_id
  inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
  where eds.eds_from_c_id = :c_id and
        eds.eds_status = 0 and
        edl.edl_status <> -1 and
        edc.edc_process_kind = 1
  into edc_uuid,
       edc_name,
       edl_id
  do
  begin
    if (doc_list <> '') then doc_list = doc_list || ',';
        doc_list = doc_list || '{' || f_json_make_pair('edc_uuid', edc_uuid, 'str') || ',' ||
                                      f_json_make_pair('name', edc_name, 'str') || ',';
    blank_list = '';
    blank_qty = 0;
    for select
            bln.bln_name,
            edb.edb_id
        from edoc_blank edb
        inner join blankname bln on bln.bln_id = edb.edb_from_bln_id
        where edb.edb_from_edl_id = :edl_id and
              edb.edb_status = 1
        into  bln_name,
              edb_id
    do
    begin
        if (blank_list <> '') then blank_list = blank_list || ',';
        blank_list = blank_list || '{' || f_json_make_pair('id', edb_id, 'num') || ',' ||
                                          f_json_make_pair('name', bln_name, 'str') || '}';
        blank_qty = blank_qty + 1;
    end
    doc_list = doc_list || '"blank_list":['||blank_list||'],';
    sign_button_caption = trim(iif(blank_qty > 1, 'Подписать документы', 'Подписать документ'));
    doc_list = doc_list || f_json_make_pair('sign_button_caption', sign_button_caption, 'str') || '}';
  end

  res_json = '{' ||
             f_json_make_pair('message', msg_text, 'str') || ',' ||
             f_json_make_pair('fio', c_name, 'str') || ',' ||
             '"doc_list":['||doc_list||']}';
  suspend;
end
