CREATE PROCEDURE UXP$UXACTION_LIST(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    UA_UUID D_UUID,
    UA_OWNER D_UUID,
    UA_UXA_UUID D_UUID,
    UA_USR_ID_FIELD D_STR_CODE,
    UA_KIND D_SMALLINT,
    UA_NAME D_STR_100,
    UA_HINT D_STR_255,
    UA_DISABLE_ON_EDIT D_BOOL,
    UA_READONLY D_BOOL,
    UA_SAFE_MODE D_BOOL,
    UA_INACTIVE D_BOOL,
    RES_UX_KIND D_INTEGER,
    RES_IS_OWN D_BOOL,
    RES_CAPTION D_STR_255,
    RES_MENU_COUNT D_INTEGER,
    RES_OWNER_CODE D_STR_100,
    RES_UXA_CODE D_STR_CODE,
    UA_UUID_STR D_STR_CODE,
    RES_OWNER_CAPTION D_STR_255,
    UA_IMG_UUID D_UUID,
    UA_IMG_UUID_CODE D_STR_CODE,
    RES_OWNER_KIND D_STR_CODE,
    UA_EXT_COMPONENT D_STR_CODE,
    UA_EXT_DATASET D_STR_CODE,
    UA_EXT_REPLACE_KIND D_SMALLINT,
    UA_HOTKEY D_STR_10,
    UA_ADD_DELIMITER D_SMALLINT,
    RES_HAS_EXPRESSION D_SMALLINT,
    RES_EXPRESSION D_STR_LONG,
    UA_IS_GROUP D_BOOL,
    RES_GROUP_MEMO D_STR_8100,
    RES_GROUP_WARNING D_BOOL,
    UA_ALIAS_GROUP D_STR_8100,
    UA_AUTO_REFRESH_SECONDS D_INTEGER,
    UA_TRIGGER_KIND D_INTEGER,
    UA_PREBLANK_KIND D_INTEGER,
    UA_PREBLANK_CODE D_STR_CODE,
    UA_PREBLANK_SAVE D_SMALLINT,
    RES_IS_PREBLANKS D_BOOL,
    RES_PREBLANK_KIND D_STR_CODE,
    RES_PREBLANK_SAVE D_STR_CODE,
    RES_UX_KIND_STR D_STR_CODE)
AS
declare variable enter_sym d_str_lang;
declare variable jsm d_integer;
declare variable jsa d_integer;
declare variable arr_count d_integer;
declare variable arr_index d_integer;
declare variable jso d_integer;
declare variable exp_value d_str_code;
declare variable exp_alias_code d_str_code;
declare variable exp_alias_name d_str_255;
begin
  enter_sym = ascii_char(13) || ascii_char(10);

  for select ua_uuid, ua_owner, ua_uxa_uuid, ua_usr_id_field, ua_kind, ua_name, ua_hint, ua_disable_on_edit, ua_readonly,
             ua_safe_mode, ua_inactive, res_ux_kind, res_is_own, res_caption, res_menu_count, res_owner_code,
             res_uxa_code, ua_uuid_str, res_owner_caption, ua_img_uuid, ua_img_uuid_code, res_owner_kind,
             ua_ext_component, ua_ext_dataset, ua_ext_replace_kind, ua_hotkey, ua_add_delimiter, res_has_expression,
             res_expression, ua_is_group, ua_alias_group, ua_auto_refresh_seconds,
             ua_trigger_kind, ua_preblank_kind, ua_preblank_code, ua_preblank_save
  from uxp$uxaction_tmp(:ux_uuid, :ux_uuid_code)
  into :ua_uuid, :ua_owner, :ua_uxa_uuid, :ua_usr_id_field, :ua_kind, :ua_name, :ua_hint, :ua_disable_on_edit,
       :ua_readonly, :ua_safe_mode, :ua_inactive, :res_ux_kind, :res_is_own, :res_caption, :res_menu_count,
       :res_owner_code, :res_uxa_code, :ua_uuid_str, :res_owner_caption, :ua_img_uuid, :ua_img_uuid_code,
       :res_owner_kind, :ua_ext_component, :ua_ext_dataset, :ua_ext_replace_kind, :ua_hotkey, :ua_add_delimiter,
       :res_has_expression, :res_expression, :ua_is_group, :ua_alias_group, :ua_auto_refresh_seconds,
       :ua_trigger_kind, :ua_preblank_kind, :ua_preblank_code, :ua_preblank_save
  do
  begin
    res_group_memo = '';
    res_group_warning = 0;
    res_is_preblanks = 0;
    if (res_uxa_code containing 'ux$preblanks') then
      res_is_preblanks = 1;

    res_preblank_kind = '';
    res_preblank_save = '';

    res_ux_kind_str = 'group';

    if (res_is_preblanks = 1) then
      res_ux_kind_str = 'preblank';
    else
    begin
      if (coalesce(res_ux_kind, '') <> '') then
        res_ux_kind_str = uxf$kind(res_ux_kind, 'code');
  
      if (res_owner_kind = 'uxframe' and res_ux_kind_str = 'uxform' and ua_usr_id_field <> '') then
        res_ux_kind_str = 'search';
    end

    if (ua_preblank_kind <> 0) then
      res_preblank_kind = uxf$enum_name('uxaction_preblank_kind', ua_preblank_kind);

    if (res_preblank_kind = '') then
      res_preblank_kind = '--  Не настроен  --';

    if (ua_preblank_save <> 0) then
      res_preblank_save = uxf$enum_name('uxaction_preblank_save', ua_preblank_save);

    if (ua_is_group = 1) then
    begin
      res_has_expression = 2;

      select exp_expression
      from ux$expression
      where exp_ux_uuid = :ua_uuid and
            exp_kind = 1
      into res_group_memo;

      if (res_group_memo = '') then
      begin
        res_group_memo = 'Выражение: Отсутствует' || enter_sym;
        res_group_warning = 1;
      end
      else
        res_group_memo = 'Выражение для выбора формы: ' || enter_sym || '---------- Начало ----------' || enter_sym || res_group_memo || enter_sym || '---------- Конец ----------' || enter_sym || enter_sym;

      if (coalesce(ua_alias_group, '') = '') then
      begin
        res_group_memo = res_group_memo || 'Список алиасов не указан';
        res_group_warning = 1;
      end
      else
      begin
        res_group_memo = res_group_memo || 'Условия вызова:' || enter_sym;

        -- Парсинг массива
        jsm = json.parse(ua_alias_group);
        jsa = json.obj_get_arr(jsm, 'actions');
        arr_count = json.count_get(jsa);

        arr_index = 0;
        while (arr_index < arr_count) do
        begin
          exp_value = '';
          exp_alias_code = '';
          jso = json.arr_get_obj(jsa, arr_index);
    
          -- Получение значения из объекта
          exp_value = json.obj_get_val(jso, 'v', exp_value);
          exp_alias_code = json.obj_get_val(jso, 'a', exp_alias_code);
          exp_alias_name = '';

          if (exp_alias_code <> '') then
          begin
            select uxa_fcode || '   - ' || uxa_name
            from ux$alias
            where uxa_fcode = :exp_alias_code
            into exp_alias_name;

            if (row_count = 0) then
            begin
              exp_alias_name = exp_alias_code || '   - Отсутствует';
              res_group_warning = 1;
            end
          end else
          begin
            exp_alias_name = '<Алиас не определен>';
            res_group_warning = 1;
          end

          if (exp_value = '') then
          begin
            exp_value = '<Не указано>';
            res_group_warning = 1;
          end

          res_group_memo = res_group_memo || rpad(exp_value, 20, ' ') || exp_alias_name || enter_sym;

          arr_index = arr_index + 1;
        end
    
        json.destroy(jsm);
      end
    end

    suspend;
  end
end
