CREATE PROCEDURE EDOC_PKB_SIGNER_DATA(
    EDS_ID D_ID)
RETURNS (
    INN D_STR_20,
    PHONE D_STR_20)
AS
begin
    for select c.c_inn, f_cont_info('clients', c.c_id, 'cell_main', 'ok')
    from edoc_sign eds
    inner join clients c on eds.eds_from_c_id = c.c_id
    where eds.eds_id = :eds_id
    into :inn, :phone
    do
        suspend;

end
