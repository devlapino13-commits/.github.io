CREATE PROCEDURE MC_ONLINE_APP_TO_EXPRESS(
    MCL_ID D_ID,
    ONLINE_APP_ID D_INTEGER)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable OA_STATUS D_SMALLINT;
declare variable OA_FROM_MCL_ID D_ID;
declare variable JSON_STR D_BLOB_TEXT;
declare variable REG_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable PR_ID D_INTEGER;
declare variable EXC_ID_BEGIN D_INTEGER;
declare variable AMOUNT D_CURRENCY;
declare variable PERIOD D_INTEGER;
declare variable PERCENT D_PERCENT;
declare variable REPAY_DAY D_INTEGER;
declare variable EXA_ID D_ID;
declare variable KBM_ID D_ID;
declare variable GROUP_ID D_INTEGER;
declare variable RES_STATUS D_SMALLINT;
declare variable MA_ID D_ID;
declare variable SEND_KOL D_INTEGER;
declare variable OA_DATE D_DATE;
declare variable DO_CANCEL D_SMALLINT;
declare variable ERROR_TEXT_TMP D_STR_LONG;
declare variable OA_FROM_KBM_ID D_ID;
declare variable MCL_LANG_CODE D_STR_LANG;
declare variable MINIAPP_OBJ_KIND D_INTEGER;
declare variable EIC_ID D_INTEGER;
declare variable OA_CARD_TYPE D_SMALLINT;
declare variable PR_CODE D_STR_100;
declare variable DISB_KIND D_INTEGER;
declare variable OA_PAN_MASKED D_STR_20;
declare variable OA_PAN_ENCRYPTED D_STR_LONG;
begin
  error_text = '';
  do_cancel = 0;

  select oa.oa_status,
         oa.oa_from_mcl_id,
         oa.oa_json,
         oa.oa_last_client_office_id,
         oa.oa_date,
         oa.oa_from_kbm_id,
         oa.oa_card_type,
         coalesce(oa.oa_pan_masked, ''),
         coalesce(oa.oa_pan_encrypted, '')
  from online_app oa
  where oa.oa_id = :online_app_id
  into :oa_status,
       :oa_from_mcl_id,
       :json_str,
       :reg_id,
       :oa_date,
       :oa_from_kbm_id,
       :oa_card_type,
       :oa_pan_masked,
       :oa_pan_encrypted;

  select mcl.mcl_from_c_id, mcl.mcl_lang_code
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into c_id,
       mcl_lang_code;

  if (oa_card_type = 1) then
  begin
    if (oa_pan_masked = '') then
    begin
        error_text = 'Маскированный номер карты не передан';
        do_cancel = 1;
    end
    else if (oa_pan_encrypted = '') then
    begin
        error_text = 'Шифрованный номер карты не передан';
        do_cancel = 1;
    end
  end

  if (oa_status not between 0 and 2) then
    error_text = f_lang(:mcl_lang_code, 'Заявка не находится в процессе. Она не может быть взята в обработку');

  if (oa_from_mcl_id <> mcl_id) then
  begin
    do_cancel = 1;
    error_text = f_lang(:mcl_lang_code, 'Заявка зарегистрирована другим пользователем');
  end

  if (oa_date <> current_date) then
  begin
    do_cancel = 1;
    error_text = f_lang(:mcl_lang_code, 'Заявка была содана не сегодня. Она не может быть взята в обработку');
  end

  if (do_cancel = 1) then
  begin
     select error_text
     from mc_online_app_cancel(:mcl_id, :online_app_id)
     into :error_text_tmp;
  end

  if (oa_status <> 0) then
  begin
    suspend;
    exit;
  end

  miniapp_obj_kind = f_objkind('miniapp', 'id');

  -- создаем экспресс займ и его id сохраняем в переменную :exa_id
  select pr.pr_id, pr.pr_code
  from product pr
  where pr.pr_code containing '.online_loan.' and
        iif(coalesce(:oa_card_type, 0) = 1,
            pr.pr_code containing '.elcard_disb.',
            pr.pr_code not containing '.elcard_disb.')
  into pr_id, pr_code;

  select exc.exc_id
  from expresscfg exc
  where exc.exc_action_code = 'miniapp.create'
  into exc_id_begin; --Создание кредитной заявки

  amount = f_json_blob_get_value(:json_str, 'amount', '');
  period = f_json_blob_get_value(:json_str, 'period', '');
--  percent = f_json_blob_get_value(:json_str, 'percent_rate', '');
  repay_day = f_json_blob_get_value(:json_str, 'pay_day', '');

  --RFCKG-2557
  disb_kind = 6;
  if (pr_code containing '.elcard_disb.') then
    disb_kind = 5;

  select t.perc_max  --Макс. и мин. % ставка равны
  from hp_get_percent(:pr_id, current_date, 1, :amount, :period, :amount, :amount, :c_id, 0, 0, :disb_kind) t
  into percent;

  group_id = null;
  select exa.exa_group_id
  from expressapp exa
  where exa.exa_from_exp_id = -3 and --пакет по онлайн займам
        exa.exa_from_exc_id <> 200 --незавершенные экспрессы
  order by exa.exa_id desc
  rows 1
  into group_id;

  group_id = coalesce(group_id, 0) + 1;

  if (group_id >= 3) then
    group_id = 0;

  if (coalesce(reg_id, 0) = 0) then
    reg_id = 1180300;  --Борбор

  insert into expressapp (exa_from_exp_id, exa_date, exa_group_id, exa_done, exa_from_c_id, exa_base_k_id,
                          exa_from_exc_id, exa_final_status, exa_wait_status, exa_from_ma_id, exa_from_k_id,
                          exa_from_edl_id, exa_disb_date, exa_from_pr_id, exa_amount, exa_period, exa_percent,
                          exa_repay_day, exa_lo_id, exa_from_reg_id, exa_try_count, exa_from_un_id, exa_datetime,
                          exa_reshipping_count)
  values (-3, current_date, :group_id, 0, :c_id, null, :exc_id_begin, 0, 0, null, null, null, current_date, :pr_id, :amount,
          :period, :percent, :repay_day, 1000000000102, :reg_id, 0, 1000000000000, current_timestamp, 0)
  returning exa_id
  into :exa_id;

  
  update online_app oa
  set oa.oa_from_exa_id = :exa_id
  where oa.oa_id = :online_app_id;

  select res_status
  from hp_do_expressapp_action_macreat(:exa_id)
  into res_status;

  if (res_status = 0) then
  begin
    select exa.exa_from_ma_id
    from expressapp exa
    where exa.exa_id = :exa_id
    into ma_id;

    if (coalesce(oa_from_kbm_id, 0) = 0) then
    begin
      select send_kol
      from insert_kibminiapp(:reg_id, :ma_id, 1000000000000, 1)
      into send_kol;
    
      if (send_kol = 0) then
      begin
        update online_app oa
        set oa.oa_status = -1
        where oa.oa_id = :online_app_id;
    
        error_text = f_lang(:mcl_lang_code, 'Ошибка при отправке запроса в КИБ');
        suspend;
        exit;
      end

      select kbm_id
      from kibminiapp
      inner join kibminiapp_link on kbl_from_kbm_id = kbm_id
      where kbl_object_kind = :miniapp_obj_kind and
            kbl_object_id = :ma_id
      order by kbm_id desc
      rows 1
      into kbm_id;

      oa_from_kbm_id = kbm_id;
    end
    else
      insert into kibminiapp_link(kbl_from_kbm_id, kbl_object_kind, kbl_object_id)
      values (:oa_from_kbm_id, :miniapp_obj_kind, :ma_id);

    eic_id = (select eic_id from extinfocfg where eic_code = 'check_in_ru_db');

    if (not exists(select *
                   from extinfolist eil
                   where eil.eil_from_c_id = :c_id and
                         eil.eil_object_kind = :miniapp_obj_kind and
                         eil.eil_object_id = :ma_id and
                         eil.eil_from_eic_id = :eic_id)) then
      insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json)
      values (:c_id, :eic_id, :miniapp_obj_kind, :ma_id, 1);

    update online_app oa
    set oa.oa_status = 1,
      oa.oa_approve_datetime = current_timestamp,
      oa.oa_from_kbm_id = :oa_from_kbm_id
    where oa.oa_id = :online_app_id;

    update expressapp exa
    set exa.exa_from_exc_id = 60  --Робот начнёт процесс с создания эл. документа (заявка уже создана выше)
    where exa.exa_id = :exa_id;
  end
  suspend;
end
