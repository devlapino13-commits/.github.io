CREATE PROCEDURE RCIB_KONTRACT_LIST(
    RCP_ID ID_TYPE)
RETURNS (
    RCK_FROM_CP_ID ID_TYPE,
    RCK_REC_NO INTEGER,
    RCP_VERSION INTEGER,
    RCP_FILENAME STR_255,
    RCK_FROM_K_ID ID_TYPE,
    RCK_FROM_C_ID ID_TYPE,
    RCK_CRED_STATUS SMALLINT,
    RCK_CRED_UPDATE DATE,
    RCK_CRED_NO VARCHAR(50),
    RCK_CRED_ACCOUNT_NO VARCHAR(50),
    RCK_CRED_ID VARCHAR(50),
    RCK_CRED_ACTIVE SMALLINT,
    RCK_CDER_ACTIVE_DATE DATE,
    RCK_CRED_PURPOSE VARCHAR(2),
    RCK_CRED_JOIN SMALLINT,
    RCK_CRED_SUM DECIMAL(15,2),
    RCK_CRED_CURRENCY VARCHAR(3),
    RCK_CRED_PERCENT DECIMAL(3,2),
    RCK_CRED_DATE DATE,
    RCK_CRED_ENDDATE DATE,
    RCK_CRED_AMOUNT_FREQUENCY VARCHAR(2),
    RCK_CRED_SECURITY VARCHAR(2),
    RCK_CRED_DATE_MISSEDPAYOUT DATE,
    RCK_OP_CRED_SUM_PAYOUT DECIMAL(15,2),
    RCK_OP_CRED_DATE_PAYOUT DATE,
    RCK_OP_CRED_SUM_DEBT DECIMAL(15,2),
    RCK_OP_CRED_SUM_PAID DECIMAL(15,2),
    RCK_OP_CRED_SUM_OVERDUE DECIMAL(15,2),
    RCK_OP_CRED_DATE_OVERDUE DATE,
    RCK_OP_CRED_SUM_NEXTPAYOUT DECIMAL(15,2),
    RCK_OP_CRED_DATE_NEXTPAYOUT DATE,
    RCK_TA_CRED_SUM_PAYOUT DECIMAL(15,2),
    RCK_TA_CRED_DATE_PAYOUT DATE,
    RCK_TA_CRED_SUM_DEBT DECIMAL(15,2),
    RCK_TA_CRED_SUM_PAID DECIMAL(15,2),
    RCK_TA_CRED_SUM_OVERDUE DECIMAL(15,2),
    RCK_TA_CRED_DATE_OVERDUE DATE,
    RCK_TA_CRED_SUM_NEXTPAYOUT DECIMAL(15,2),
    RCK_TA_CRED_DATE_NEXTPAYOUT DATE,
    CRED_TYPE VARCHAR(10),
    CRED_FACILITY SMALLINT,
    CONSENTDATE DATE,
    CRED_SUM_LIMIT DECIMAL(15,2),
    CRED_INSURED SMALLINT,
    RETURN_COURT_DATA SMALLINT,
    COURT_NO STR_100,
    COURT_NUMBER VARCHAR(2),
    COURT_SOURCE STR_255,
    COURT_PLAINTIFF SMALLINT,
    COURT_RESOLUTION VARCHAR(1500),
    CRED_JOIN SMALLINT,
    RCK_CRED_PURCHASER STR_255,
    RCK_STATE_SOURCE STR_255,
    RCK_STATE_INFO STR_255,
    RCK_TYPE INTEGER,
    RCK_IS_PORUCH SMALLINT,
    RCK_IS_OBESP SMALLINT,
    CONSENT SMALLINT,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    GUARANTEE_VOLUME SMALLINT,
    CRED_COLLATERAL SMALLINT,
    CRED_FULL_COST DECIMAL(15,3),
    CRED_WAY SMALLINT)
AS
declare variable null_date date;
declare variable rcp_sended smallint;
declare variable law_main_id integer;
declare variable lam_from_lal_id integer;
declare variable rck_from_k_id_old id_type;
begin
  rck_from_k_id_old = 0;
  return_court_data = 0;
  cred_join = 0;

  select rcp.rcp_id, rcp_version, rcp_filename, rcp_sended
  from rcib_pocket rcp
  where rcp.rcp_id = :rcp_id and
        rcp.rcp_is_kredit = 1
  into :rcp_id, :rcp_version, :rcp_filename, :rcp_sended;

  null_date = cast('01.02.1900' as date);

--CRED_TYPE        = '19'; -- микрозайм
  cred_facility = 8; -- Статус кредитной линии (8 - Кредит выдан не в форме кредитной линии)
  consent = 1; -- Флаг согласия на передачу данных в БКИ
  cred_sum_limit = 0;
  cred_insured = 0; -- Флаг страховки кредита
  guarantee_volume = 1; -- поручительство в полном объеме займа (кредита)
  cred_collateral = 0; -- Отметка о погашении за счет обеспечения (не было погашения за счет обеспечения)
  cred_way = 3; --  Прямой (Кредит выдан при обращении Заёмщика в офис кредитной организации.)

  if (rcp_sended = 0) then
  begin
    for select rck_rec_no, rck_from_cp_id, rck_from_k_id, rck_from_c_id, rck_cred_status, rck_cred_update, rck_cred_no,
               rck_cred_account_no, rck_cred_id, rck_cred_active, rck_cder_active_date, rck_cred_purpose, rck_cred_join,
               rck_cred_sum, rck_cred_currency, rck_cred_percent, rck_cred_date, coalesce(rck_cred_enddate, :null_date),
               rck_cred_amount_frequency, rck_cred_security, coalesce(rck_cred_date_missedpayout, :null_date),
               rck_op_cred_sum_payout, coalesce(rck_op_cred_date_payout, :null_date), rck_op_cred_sum_debt,
               rck_op_cred_sum_paid, rck_op_cred_sum_overdue, coalesce(rck_op_cred_date_overdue, :null_date),
               rck_op_cred_sum_nextpayout, coalesce(rck_op_cred_date_nextpayout, :null_date), rck_ta_cred_sum_payout,
               coalesce(rck_ta_cred_date_payout, :null_date), rck_ta_cred_sum_debt, rck_ta_cred_sum_paid,
               rck_ta_cred_sum_overdue, coalesce(rck_ta_cred_date_overdue, :null_date), rck_ta_cred_sum_nextpayout,
               coalesce(rck_ta_cred_date_nextpayout, :null_date), rk.rck_cred_purchaser, rk.rck_state_source,
               rk.rck_state_info, rk.rck_type, rk.rck_is_poruch, rk.rck_is_obesp
        from rcib_kontract rk
        where rk.rck_from_cp_id = :rcp_id
        order by rck_rec_no
        into :rck_rec_no, :rck_from_cp_id, :rck_from_k_id, :rck_from_c_id, :rck_cred_status, :rck_cred_update,
             :rck_cred_no, :rck_cred_account_no, :rck_cred_id, :rck_cred_active, :rck_cder_active_date,
             :rck_cred_purpose, :rck_cred_join, :rck_cred_sum, :rck_cred_currency, :rck_cred_percent, :rck_cred_date,
             :rck_cred_enddate, :rck_cred_amount_frequency, :rck_cred_security, :rck_cred_date_missedpayout,
             :rck_op_cred_sum_payout, :rck_op_cred_date_payout, :rck_op_cred_sum_debt, :rck_op_cred_sum_paid,
             :rck_op_cred_sum_overdue, :rck_op_cred_date_overdue, :rck_op_cred_sum_nextpayout,
             :rck_op_cred_date_nextpayout, :rck_ta_cred_sum_payout, :rck_ta_cred_date_payout, :rck_ta_cred_sum_debt,
             :rck_ta_cred_sum_paid, :rck_ta_cred_sum_overdue, :rck_ta_cred_date_overdue, :rck_ta_cred_sum_nextpayout,
             :rck_ta_cred_date_nextpayout, :rck_cred_purchaser, :rck_state_source, :rck_state_info, :rck_type,
             :rck_is_poruch, :rck_is_obesp
    do
    begin
      if (rck_from_k_id <> rck_from_k_id_old) then
      begin
        rck_from_k_id_old = rck_from_k_id;

        select k_applicate_date, k_kolich_client_approved, cast(kont.k_percent_effect_fifd as decimal(15,3))
        from kontract kont
        where kont.k_id = :rck_from_k_id
        into :consentdate, -- Дата согласия на передачу данных в БКИ
             :k_kolich_client_approved, :cred_full_cost; -- ПСК

        /*-----------  Судебные дела -----------------------------------------*/
        return_court_data = 0;
        law_main_id = (select lk.lak_from_lam_id
                       from lawkredit lk
                       where lk.lak_from_k_id = :rck_from_k_id rows 1);
        law_main_id = coalesce(law_main_id, 0);

        court_no = '';
        lam_from_lal_id = null;
        court_resolution = '';
        court_source = '';

        if (law_main_id <> 0) then
        begin
          select lam.lam_delo_no, lam.lam_from_lal_id, left(lam_resolution_text, 1500)
          from lawmain lam
          where lam.lam_id = :law_main_id
          into court_no, :lam_from_lal_id, :court_resolution;

          court_no = coalesce(court_no, '');
          court_resolution = coalesce(court_resolution, '');
          court_number = '00';
          court_plaintiff = 0; -- ответчик

          select lal_name
          from lawlist lal
          where lal.lal_id = :lam_from_lal_id
          into :court_source;
          court_source = coalesce(court_source, '');

          if ((court_no <> '') and
              (court_source <> '') and
              (court_resolution <> '')) then
            return_court_data = 1;
        end
      end
      /*--------------------------------------------------------------------*/

      if (rck_is_poruch = 1) then
        cred_type = '90';
      else
        cred_type = '19';

      suspend;
    end
  end
end
