CREATE PROCEDURE UXP$PROCEDURE_MAIN_TABLES(
    PROC_NAME D_STR_CODE)
RETURNS (
    TABLE_NAME D_STR_CODE)
AS
declare variable pk_fields d_str_255;
declare variable pk_count d_integer;
begin
  proc_name = trim(upper(coalesce(proc_name, '')));
  if (proc_name = '') then
    exit;

  for select rdb$depended_on_name
  from rdb$dependencies
  where rdb$dependent_name = :proc_name and
        rdb$depended_on_type = 0 and
        rdb$field_name is null
  into table_name
  do
  begin
    pk_fields = '';
    pk_count = 0;
    select list(trim(i.rdb$field_name)), count(*)
    from rdb$relation_constraints r
    inner join rdb$index_segments i on i.rdb$index_name = r.rdb$index_name
    inner join rdb$relation_fields rf on rf.rdb$field_name = i.rdb$field_name and
          rf.rdb$relation_name = r.rdb$relation_name
    inner join rdb$fields f on f.rdb$field_name = rf.rdb$field_source
    where r.rdb$constraint_type = 'PRIMARY KEY' and
         r.rdb$relation_name = :table_name
    into pk_fields, pk_count;

    if (pk_count = 1) then
    begin
      if (exists(select * from rdb$dependencies
      where rdb$dependent_name = :proc_name and
            rdb$field_name = :pk_fields and
            rdb$depended_on_type = 0)) then
      begin
        table_name = lower(trim(table_name));
        suspend;
      end
    end
  end
end
