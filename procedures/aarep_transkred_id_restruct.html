CREATE PROCEDURE AAREP_TRANSKRED_ID_RESTRUCT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    TRANS_DATE DATE,
    TRANS_DOC_NO INTEGER,
    TRANS_REGION_ID INTEGER,
    TRANS_KONTRACT_ID DECIMAL(18,0),
    TRANS_TYPE INTEGER,
    TRANS_TYPE_TEXT VARCHAR(50),
    TRANS_DISB_OS DECIMAL(18,4),
    TRANS_OS DECIMAL(18,4),
    TRANS_PERC DECIMAL(18,4),
    TRANS_FINE DECIMAL(18,4),
    TRANS_KOMMIS DECIMAL(18,4),
    TRANS_KOMMIS_ADV DECIMAL(18,4),
    TRANS_PAY_TYPE INTEGER,
    TRANS_PAY_TYPE_TEXT VARCHAR(20),
    TRANS_OPERLIST_ID DECIMAL(18,0),
    TRANS_PAY_ALL DECIMAL(18,4),
    TRANS_OTHER_CLIENT VARCHAR(255),
    TRANS_DATETIME TIMESTAMP,
    TRANS_OTHER_DAY VARCHAR(20),
    TRANS_KF_ID DECIMAL(18,0))
AS
declare variable db_id integer;
begin

  :db_id = f_get_db_id();

  if (:report_currency is null) then
    report_currency = '000';
  if (:report_currency = '000') then
  begin
    for select ol_date, ol_doc_no, ok_from_k_id, k_from_kf_id, ok_type, ok_pay_type, ol_from_reg_id, ol_id,
               ol_doc_client_name, ol_datetime, sum(ok_os * cr_rate), cast(0 as decimal(18,4)),
               cast(0 as decimal(18,4)), cast(0 as decimal(18,4)), cast(0 as decimal(18,4)), cast(0 as decimal(18,4))
        from kontract, opercurr, operkred, operlist, region_list(:report_region_id, 1, -1, 1)
        where ok_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ok_type = 8 and
              ol_date_oper_day between :report_date_begin and :report_date_end and
              (ol_from_reg_id = id_list) and
              ol_date <= cr_date_end and
              ol_date >= cr_date and
              ok_from_k_id = k_id and
              cr_from_cu_id = k_index_curr_code and
              ok_from_c_id = 0 and
              ok_type <> 7 and
              exists(select *
                     from aarep_get_restructuring(ok_from_k_id, ol_date, :db_id)
                     where is_restruct = 1)

              /*выбрать реструктуризированные кредиты*/
        group by ol_date, ol_doc_no, ok_from_k_id, k_from_kf_id, ok_type, ok_pay_type, ol_from_reg_id, ol_id, ol_doc_client_name, ol_datetime
        union
        select ol_date, ol_doc_no, ok_from_k_id, k_from_kf_id, ok_type, ok_pay_type, ol_from_reg_id, ol_id,
               ol_doc_client_name, ol_datetime, cast(0 as decimal(18,4)), sum(ok_os * cr_rate), sum(ok_perc * cr_rate),
               sum(ok_fine * cr_rate), sum(ok_kommis * cr_rate), sum(ok_kommis_adv * cr_rate)
        from kontract, opercurr, operkred /* LEFT JOIN kroper ON (ok_type=ot_id) - убрал, из-за этого выдавалось ошибка*/ , operlist, region_list(:report_region_id, 1, -1, 1)
        where ok_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ok_type >= 4 and
              ok_type <> 8 and
              ol_date_oper_day between :report_date_begin and :report_date_end and
              (ol_from_reg_id = id_list) and
              ol_date_oper_day between cr_date and cr_date_end /*Выборку сдел по OL_Date_Oper_Day - так как поле индексированный*/
              /*and ol_date<=cr_date_end
  and ol_date>=cr_date */
              and
              ok_from_k_id = k_id and
              cr_from_cu_id = k_index_curr_code and
              ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
              and
              ok_from_c_id = 0 and
              exists(select *
                     from aarep_get_restructuring(ok_from_k_id, ol_date, :db_id)
                     where is_restruct = 1)
              /*выбрать реструктуризированные кредиты*/

        group by ol_date, ol_doc_no, ok_from_k_id, k_from_kf_id, ok_type, ok_pay_type, ol_from_reg_id, ol_id, ol_doc_client_name, ol_datetime
        into :trans_date, :trans_doc_no, :trans_kontract_id, :trans_kf_id, :trans_type, :trans_pay_type,
             :trans_region_id, :trans_operlist_id, :trans_other_client, :trans_datetime, :trans_disb_os, :trans_os,
             :trans_perc, :trans_fine, :trans_kommis, :trans_kommis_adv
    do
    begin
      if (cast(:trans_datetime as date) <> trans_date) then
        trans_other_day = 'Другой день';
      else
        trans_other_day = '';

      if (trans_other_client is null) then
        trans_other_client = '';

      if (trans_type = 4) then
        trans_type_text = 'Погашение';
      if (trans_type = 5) then
        trans_type_text = 'Списание';
      if (trans_type = 6) then
        trans_type_text = 'Недоплата (списан.)';
      if (trans_type = 7) then
        trans_type_text = 'Начисление';
      if (trans_type = 8) then
        trans_type_text = 'Выдача';
      if (trans_type = 9) then
        trans_type_text = 'Переплата';

      trans_pay_type_text = '';
      if (trans_pay_type = 1) then
        trans_pay_type_text = 'Наличная';
      if (trans_pay_type = 2) then
        trans_pay_type_text = 'Безналичная';
      if (trans_pay_type = 3) then
        trans_pay_type_text = 'Внутренняя';

      trans_pay_all = trans_disb_os + trans_os + trans_perc + trans_fine + trans_kommis + trans_kommis_adv;
      suspend;
    end
  end
  else
  begin
    for select ol_date, ol_doc_no, ok_from_k_id, k_from_kf_id, ok_type, ok_pay_type, ol_from_reg_id, ol_id,
               ol_doc_client_name, ol_datetime, sum(ok_os), cast(0 as decimal(18,4)), cast(0 as decimal(18,4)),
               cast(0 as decimal(18,4)), cast(0 as decimal(18,4)), cast(0 as decimal(18,4))
        from operkred, operlist, region_list(:report_region_id, 1, -1, 1), kontract
        where ok_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ok_type = 8 and
              ol_date_oper_day between :report_date_begin and :report_date_end and
              ok_from_c_id = 0 and
              ok_from_k_id = k_id and
              (ol_from_reg_id = id_list)
              /* and exists (select min(a.ol_date), or_from_k_id from operrest, operlist a
             where or_from_ol_id=a.ol_id
             and or_date_end >= :report_date_end
             and a.is_delete=0
             and a.ol_date<=:report_date_end
             and operkred.ok_from_k_id=or_from_k_id
             and operkred.ok_from_c_id=0
             group by or_from_k_id
             having min(a.ol_date)<=operlist.ol_date
             ) */
              and
              exists(select *
                     from aarep_get_restructuring(ok_from_k_id, ol_date, :db_id)
                     where is_restruct = 1)
              /*выбрать реструктуризированные кредиты*/

        group by ol_date, ol_doc_no, ok_from_k_id, k_from_kf_id, ok_type, ok_pay_type, ol_from_reg_id, ol_id, ol_doc_client_name, ol_datetime
        union
        select ol_date, ol_doc_no, ok_from_k_id, k_from_kf_id, ok_type, ok_pay_type, ol_from_reg_id, ol_id,
               ol_doc_client_name, ol_datetime, cast(0 as decimal(18,4)), sum(ok_os), sum(ok_perc), sum(ok_fine),
               sum(ok_kommis), sum(ok_kommis_adv)
        from operkred, operlist, region_list(:report_region_id, 1, -1, 1), kontract
        where ok_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              ok_type >= 4 and
              ok_type <> 8 and
              ol_date_oper_day between :report_date_begin and :report_date_end and
              (ol_from_reg_id = id_list) and
              ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
              and
              ok_from_c_id = 0 and
              ok_from_k_id = k_id
              /*
  and exists (select min(a.ol_date), or_from_k_id from operrest, operlist a
             where or_from_ol_id=a.ol_id
             and or_date_end >= :report_date_end
             and a.is_delete=0
             and a.ol_date<=:report_date_end
             and operkred.ok_from_k_id=or_from_k_id
             and operkred.ok_from_c_id=0
             group by or_from_k_id
             having min(a.ol_date)<=operlist.ol_date
             )   */
              and
              exists(select *
                     from aarep_get_restructuring(ok_from_k_id, ol_date, :db_id)
                     where is_restruct = 1)
              /*выбрать реструктуризированные кредиты*/

        group by ol_date, ol_doc_no, ok_from_k_id, k_from_kf_id, ok_type, ok_pay_type, ol_from_reg_id, ol_id, ol_doc_client_name, ol_datetime
        into :trans_date, :trans_doc_no, :trans_kontract_id, :trans_kf_id, :trans_type, :trans_pay_type,
             :trans_region_id, :trans_operlist_id, :trans_other_client, :trans_datetime, :trans_disb_os, :trans_os,
             :trans_perc, :trans_fine, :trans_kommis, :trans_kommis_adv
    do
    begin
      if (cast(:trans_datetime as date) <> trans_date) then
        trans_other_day = 'Другой день';
      else
        trans_other_day = '';

      if (trans_other_client is null) then
        trans_other_client = '';

      if (trans_type = 4) then
        trans_type_text = 'Погашение';
      if (trans_type = 5) then
        trans_type_text = 'Списание';
      if (trans_type = 6) then
        trans_type_text = 'Недоплата (списан.)';
      if (trans_type = 7) then
        trans_type_text = 'Начисление';
      if (trans_type = 8) then
        trans_type_text = 'Выдача';
      if (trans_type = 9) then
        trans_type_text = 'Переплата';

      trans_pay_type_text = '';
      if (trans_pay_type = 1) then
        trans_pay_type_text = 'Наличная';
      if (trans_pay_type = 2) then
        trans_pay_type_text = 'Безналичная';
      if (trans_pay_type = 3) then
        trans_pay_type_text = 'Внутренняя';

      trans_pay_all = trans_disb_os + trans_os + trans_perc + trans_fine + trans_kommis + trans_kommis_adv;
      suspend;
    end
  end
end
