CREATE PROCEDURE AAREP_PAYSYS_DISB(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(60),
    OKRUG_NAME VARCHAR(60),
    DISB_LGOT INTEGER,
    DISB_IS_MORE_70 INTEGER,
    DISB_NOT_MORE_70 INTEGER,
    DISB_NOT_LGOT INTEGER,
    DISB_TOTAL INTEGER,
    DISB_POLYS INTEGER)
AS
declare variable ks_region_id integer;
begin
  for with tabl1
      as (select ks_region_id, iif(pr_product_kind <> 2, 0, 1) is_lgot,
                 iif(c_birth_date <= dateadd(year, -70, k_disb_date), 1, 0) is_more_70,
                 iif(psi_code is null, 0, 1) is_polys
          from kontract, product, region_list(:rep_region_id, 1, 3, 1), kontstate
          left join hp_paysys_get_kont_cl_info(ks_from_k_id) on (1 = 1)
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
                ks_from_k_id = k_id and
                k_from_pr_id = pr_id and
                ks_region_id = id_list)
      select ks_region_id, sum(is_lgot) is_lgot, sum(iif(is_lgot = 0 and is_more_70 = 1, 1, 0)) is_more_70,
             sum(iif(is_lgot = 0 and is_more_70 = 0, 1, 0)) not_more_70, sum(iif(is_lgot = 0, 1, 0)) not_lgot,
             count(ks_region_id) total_disb, sum(is_polys) is_polys
      from tabl1
      group by ks_region_id
      into ks_region_id, disb_lgot, disb_is_more_70, disb_not_more_70, disb_not_lgot, disb_total, disb_polys
  do
  begin
    select aa_curr_office_name, aa_curr_level3_name
    from aarep_get_office_categ(:ks_region_id, :rep_date_end)
    into reg_name, okrug_name;
    suspend;
  end
end
