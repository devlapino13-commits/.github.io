CREATE PROCEDURE AARE_SFR_AMOUNT_CONTROL(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OL_DATE DATE,
    OL_MEMO VARCHAR(1024),
    OB_DETAIL VARCHAR(1024),
    OB_SUMMA CURR_TYPE,
    AL_NAME_KT VARCHAR(200),
    AL_NAME_DT VARCHAR(200),
    CL_NAME VARCHAR(100))
AS
declare variable ob_account_kt varchar(20);
declare variable ob_account_dt varchar(20);
declare variable ol_document_client id_type;
begin
  for select ol_date, ol_memo, ob_detail, ob_summa, ob_account_kt, ob_account_dt, ol_document_client
      from operbook, operlist, region_list(:rep_region_id, 1, 3, 1)
      where ob_ol_is_active = 1 and
            ol_date between :rep_date_begin and :rep_date_end and
            ob_summa > 500000 and
            ob_from_ol_id = ol_id and
            ol_doc_kind in (0, 1, 4, 6, 7) -- в таблице DOCNAME написаны описании
            and
            ol_from_reg_id = id_list

            and
            (not(ob_detail starting 'Перевод средств между офисами' or
            ob_detail starting 'Подкрепление оборотной кассы' or
            ob_detail starting 'Инкассация' or
            ob_detail starting 'Подкрепление расчетного счета' or
            ob_detail starting 'Инкассция' -- забили в БД с ошбиками, изза этого добавлено данное доп условия
            or
            ob_detail starting 'Денежные активы в расчетах') or
            ob_detail is null)

      order by ol_date
      into ol_date, ol_memo, ob_detail, ob_summa, ob_account_kt, ob_account_dt, ol_document_client
  do
  begin
--  наименование кредит
    select al_name
    from acc_list
    where al_id = :ob_account_kt
    into al_name_kt;
--  наименование дебит
    select al_name
    from acc_list
    where al_id = :ob_account_dt
    into al_name_dt;

    cl_name = null;
    select c_name_fio
    from clients
    where c_id = :ol_document_client
    into cl_name;

    suspend;
  end
end
