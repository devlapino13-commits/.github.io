CREATE PROCEDURE AAREP_EFFECTIVENESS_NEW(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_KOL_STAFF INTEGER,
    AA_KOL_CURR_KC INTEGER,
    AA_KOL_DISB_KC INTEGER,
    AA_KOLICH_CL_ACTIVE_BEGIN INTEGER,
    AA_AMOUNT_ACTIVE_BEGIN DECIMAL(18,4),
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_ROST DECIMAL(18,4),
    AA_AMOUNT_ACTIVE_ROST_PERC DECIMAL(15,1),
    AA_KOLICH_CL_ACTIVE_ROST INTEGER,
    AA_KOLICH_CL_ROST_PERC DECIMAL(15,1),
    AA_KOLICH_CL_DISB INTEGER,
    AA_AMOUNT_DISB DECIMAL(18,4),
    AA_REPAY_NORMONTH_OS DECIMAL(18,4),
    AA_REPAY_FACT_PERIOD_OS DECIMAL(18,4),
    AA_REPAY_OS_PERC DECIMAL(15,1),
    AA_DELIQ_CL_KOLICH INTEGER,
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_PAR DECIMAL(15,1),
    AA_DELIQ_CL_KOLICH7 INTEGER,
    AA_DELIQ_BALANCE7 DECIMAL(18,4),
    AA_PAR7 DECIMAL(15,1),
    RESULT_CALCED DECIMAL(18,4),
    RESULT_PAYED DECIMAL(18,4),
    RES_AMOUNT_PERIOD DECIMAL(18,4),
    AA_REPAY_ALL_FINE DECIMAL(18,4),
    LLC_RPPU DECIMAL(18,4),
    REG_DATE DATE,
    AA_CURR_LEVEL2_NAME VARCHAR(100),
    AA_CURR_LEVEL3_NAME VARCHAR(100),
    PRODUCTIVITY_DISB_KC DECIMAL(18,4),
    PRODUCTIVITY_ACTIVE_KC DECIMAL(18,4),
    AA_REPAY_ALL_OS CURR_TYPE,
    AA_DELIQ_BALANCE1 CURR_TYPE,
    AA_DELIQ_BALANCE30 CURR_TYPE,
    AA_KOL_SKC INTEGER,
    AA_PAR1 DECIMAL(15,1),
    AA_PAR30 DECIMAL(15,1),
    AA_REPAY_ALL_KOMMIS CURR_TYPE,
    AA_KOL_CURR_SKC INTEGER,
    AA_LOAN_KIND VARCHAR(100))
AS
declare variable aa_curr_region_id integer;
declare variable c_id_tmp id_type;
begin
  for select aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_level2_name, aa_curr_level3_name,
             aa_curr_region_id, aa_loan_kind, sum(aa_amount_active_begin), sum(aa_kolich_cl_active_begin),
             sum(aa_amount_active_end), sum(aa_kolich_cl_active_end), sum(aa_kolich_cl_disb), sum(aa_amount_disb),
             sum(aa_repay_normonth_os), sum(aa_repay_fact_period_os), sum(aa_deliq_cl_kolich), sum(aa_deliq_balance),
             sum(aa_deliq_cl_kolich7), sum(aa_deliq_balance7), sum(result_calced), sum(result_payed),
             sum(res_amount_period), sum(aa_repay_all_fine), sum(aa_repay_all_os), sum(aa_deliq_balance1),
             sum(aa_deliq_balance30), sum(aa_repay_all_kommis), sum(aa_kol_kc_in_region), sum(aa_kol_skc_in_region)

      from aarep_effectiveness_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_level2_name, aa_curr_level3_name, aa_curr_region_id, aa_loan_kind
      into :aa_curr_office_name, :aa_curr_branch_name, :aa_curr_level_name, :aa_curr_level2_name, :aa_curr_level3_name,
           :aa_curr_region_id, :aa_loan_kind, :aa_amount_active_begin, :aa_kolich_cl_active_begin,
           :aa_amount_active_end, :aa_kolich_cl_active_end, :aa_kolich_cl_disb, :aa_amount_disb, :aa_repay_normonth_os,
           :aa_repay_fact_period_os, :aa_deliq_cl_kolich, :aa_deliq_balance, :aa_deliq_cl_kolich7, :aa_deliq_balance7,
           :result_calced, :result_payed, :res_amount_period, :aa_repay_all_fine, aa_repay_all_os, aa_deliq_balance1,
           aa_deliq_balance30, aa_repay_all_kommis, aa_kol_curr_kc, aa_kol_curr_skc
  do
  begin
-- Кол-во штатных КС
    /*
--  было до 26.07.2016 закомментировал igusev
-- Кол-во штатных КС

    aa_kol_staff = 0;
    select count(ps_from_c_id) from hr_pstatus, joblist
    where ps_is_working = 1
    and ps_from_jl_id = jl_id
    and :rep_date_end between ps_date_begin and ps_date_end
    and ps_from_reg_id = :aa_curr_region_id
    and jl_kind = 5  -- Выбор только по КС
    and ps_is_on_staff = 1
    into :aa_kol_staff;

--измененный скрипт ниже
--Измнения связаны с писмом Отабека от 12.07.2016

-->> В колонке «Кол-во КС работающих» нужно указывать всех КС и СКС, которые работают по приказу (кроме тех, кто в «отпуске без содержания по уходу за ребенком» и  «отпуске по беременности и родам») на дату отчета.
    */

    aa_kol_staff = 0;

    for select ps_from_c_id
        from hr_pstatus, joblist, jobkind jk
        where ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              :rep_date_end between ps_date_begin and ps_date_end and
              ps_from_reg_id = :aa_curr_region_id and
              jl_kind = jk.jk_id and
              jk.jk_system_code in ('JobSystem.LO_Senior', 'JobSystem.LO_Standart') -- КС и СКС
              and
              ps_is_on_staff = 1
        into :c_id_tmp
    do
    begin
      if (not exists(select hv.hrv_from_c_id
                     from hr_voctravel hv, sa_worktype swt, hr_orders
                     where hrv_date_begin <= hrv_date_end_fact and
                           hrv_status = 1 and
                           ho_order_type = sawt_id and
                           hrv_from_ho_id = ho_id and
                           :rep_date_end between hv.hrv_date_begin and hv.hrv_date_end_fact and
                           hrv_from_c_id = :c_id_tmp and
                           swt.sawt_id in (16, 17, 18) --Декретный отпуск, Отпуск по беременности и родам, Отпуск без содержания по уходу за ребенком
          )) then
      begin
        aa_kol_staff = aa_kol_staff + 1;
      end
    end

-- Кол-во КС имеющие активный портфель на конец периода
    /*
    aa_kol_curr_kc = 0;
    select count(distinct kal_new_kf_officer)
    from kontstate, kontract, knaktlist
    where ks_ol_is_delete = 0
    and ks_state = 1
    and :rep_date_end between ks_date_begin and ks_date_end
    and ks_from_k_id = k_id
    and k_from_kf_id = kal_from_kf_id
    and knaktlist.is_delete = 0
    and :rep_date_end between kal_date and kal_date_end
    and kal_new_reg_id = :aa_curr_region_id
    into :aa_kol_curr_kc;
*/
-- Кол-во КС которые производили выдачу за отчетный период
    aa_kol_disb_kc = 0;
    select count(distinct ks_kc_id)
    from kontstate
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_region_id = :aa_curr_region_id
    into :aa_kol_disb_kc;
    if (aa_kol_disb_kc is null) then
      aa_kol_disb_kc = 0;

--  Кол-во ССВЗ (Кредитый менеджер)
    aa_kol_skc = 0;
    select count(ps_from_c_id)
    from hr_pstatus, joblist
    where ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :aa_curr_region_id and
          jl_kind = 6 -- выбор только по ССВЗ
          and
          ps_is_on_staff = 1
    into :aa_kol_skc;
    if (aa_kol_skc is null) then
      aa_kol_skc = 0;

-- Продуктивность по выдаче, кол-во выданных клиентов делим на кол-во выдавших КС
    productivity_disb_kc = 0;
    if (:aa_kol_disb_kc <> 0) then
      productivity_disb_kc = round(:aa_kolich_cl_disb / :aa_kol_disb_kc, 0);
-- Продуктивность по портфелю, кол-во активных клиентов делим на кол-во КС у кого имеется активный портфель
    productivity_active_kc = 0;
    if (:aa_kol_curr_kc <> 0) then
      productivity_active_kc = round(:aa_kolich_cl_active_end / :aa_kol_curr_kc, 0);
--  Вычисляет рост портфеля и в процентном соотнощении тоже
    aa_amount_active_rost = :aa_amount_active_end - :aa_amount_active_begin;
    aa_amount_active_rost_perc = 100;
    if (:aa_amount_active_begin <> 0) then
      aa_amount_active_rost_perc = cast(:aa_amount_active_end as double precision) / :aa_amount_active_begin * 100 - 100;
--  Вычисляет рост клиентов и в процентном соотнощении тоже
    aa_kolich_cl_active_rost = :aa_kolich_cl_active_end - :aa_kolich_cl_active_begin;
    aa_kolich_cl_rost_perc = 100;
    if (:aa_kolich_cl_active_begin <> 0) then
      aa_kolich_cl_rost_perc = cast(:aa_kolich_cl_active_end as double precision) / :aa_kolich_cl_active_begin * 100 - 100;
--  Вычисляет в процентном соотнощении возвратность ОС
    aa_repay_os_perc = 0;
    if (:aa_repay_normonth_os <> 0) then
      aa_repay_os_perc = :aa_repay_fact_period_os / :aa_repay_normonth_os * 100;
--  Вычисляет риск портфеля всех и более 7ми дней
    if (:aa_amount_active_end <> 0) then
    begin
      aa_par = :aa_deliq_balance / :aa_amount_active_end * 100;
      aa_par7 = :aa_deliq_balance7 / :aa_amount_active_end * 100;
      aa_par1 = :aa_deliq_balance1 / :aa_amount_active_end * 100;
      aa_par30 = :aa_deliq_balance30 / :aa_amount_active_end * 100;
    end
    else
    begin
      aa_par = 0;
      aa_par7 = 0;
      aa_par1 = 0;
      aa_par30 = 0;
    end
--  Вычисление возраст офиса
    reg_date = null;
    select min(ol_date_oper_day)
    from operlist
    where ol_from_reg_id = :aa_curr_region_id and
          operlist.is_delete = 0
    into :reg_date;

--  Вычисляет сумму РППУ контракта
    /*
    llc_rppu = 0;
    Select sum(rppu) From AAREP_LLC_REPORT(:aa_curr_region_id, :rep_date_end)
    into :llc_rppu;

*/
    select sum(llc_rppu) as rppu
    from aarep_llc_report_temp(:aa_curr_region_id, :rep_date_end)
    into :llc_rppu;

    suspend;
  end

end
