CREATE PROCEDURE HP_EXTINFO_SET_RESULT_BIN(
    ID D_ID,
    BLOB_DATA D_BLOB_BINARY)
AS
declare variable file_name d_str_255;
declare variable caption d_str_255;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable bls_id d_id;
begin
  -- процедура сохраняет результат полученный от внешних источников

  bls_id = null;
  if (blob_data is not null) then
  begin
    file_name = '';
    select eic_code, eic_name, eil_from_c_id
    from extinfocfg, extinfolist
    where eil_id = :id and
          eil_from_eic_id = eic_id
    into file_name, caption, c_id;

    c_name_fio = '';
    if (c_id is not null) then
    begin
      select c_name_fio from clients
      where c_id = :c_id
      into c_name_fio;
    end
    if (c_name_fio <> '') then
      caption = caption ||', '||c_name_fio;

    if (file_name = '') then
      exception e_default 'ID not found: '||coalesce(id, '<NULL>');

    file_name = file_name || '_json.txt';

    execute procedure hp_insert_into_blobstore (:file_name, :caption, :blob_data, 1000000000000, 2, 3)
    returning_values :bls_id;
  end

  update extinfolist
  set eil_status = 2,
      eil_from_bls_id = :bls_id
  where eil_id = :id;
end
