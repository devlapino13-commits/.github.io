CREATE PROCEDURE AAREP_CLI_HAS_LOANS_ANALITIC(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_BIRTH_DATE DATE,
    AA_CURR_OFFICE_NAME VARCHAR(100),
    AA_CURR_BRANCH_NAME VARCHAR(100),
    AA_CURR_LEVEL_NAME VARCHAR(100),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_DELIQ_DAYS INTEGER,
    AA_DELIQ_REPAY_KOLICH INTEGER,
    DD_DISB_DATE DATE,
    AA_DELIQ_KOLICH INTEGER,
    AA_DELIQ_KOLICH7 INTEGER,
    AA_DELIQ_KOLICH30 INTEGER,
    AA_DELIQ_KOLICH60 INTEGER,
    AA_DELIQ_KOLICH90 INTEGER,
    AA_DELIQ_KOLICH180 INTEGER,
    AA_DELIQ_KOLICH360 INTEGER,
    AA_KF_NAME VARCHAR(100),
    KC_HAS_LOANS_STR VARCHAR(100),
    KC_HAS_FAMILY_LOANS_STR VARCHAR(100),
    K_CLI_BALANCE DECIMAL(18,4),
    K_CLI_DELIQ_BALANCE DECIMAL(18,4),
    K_CLI_DELIQ_BALANCE7 DECIMAL(18,4),
    K_CLI_DELIQ_BALANCE30 DECIMAL(18,4),
    K_CLI_DELIQ_BALANCE60 DECIMAL(18,4),
    K_CLI_DELIQ_BALANCE90 DECIMAL(18,4),
    K_CLI_DELIQ_BALANCE180 DECIMAL(18,4),
    K_CLI_DELIQ_BALANCE360 DECIMAL(18,4),
    AA_K_ID DECIMAL(18,0),
    KC_HAS VARCHAR(100),
    AA_REST_NORM_KOL INTEGER,
    AA_REST_DELIQ_KOL INTEGER,
    AA_REST_PERIOD_KOL INTEGER,
    K_CLI_REST_NORM_BAL DECIMAL(18,4),
    K_CLI_REST_DELIQ_BAL DECIMAL(18,4),
    K_CLI_REST_PERIOD_BAL DECIMAL(18,4),
    K_CLI_REPAY_WOFF_OS DECIMAL(18,4),
    K_CLI_REPAY_WOFF_PERC DECIMAL(18,4),
    K_CLI_REPAY_WOFF_FINE DECIMAL(18,4),
    AA_PRODUCT_NAME VARCHAR(50),
    CS_NAME VARCHAR(50),
    LK_PERCENT PERC_TYPE,
    LK_IS_INDIV SMALLINT,
    AA_REST_DELIQ_KOL30 INTEGER,
    AA_REST_DELIQ_BAL30 DECIMAL(18,4),
    AA_REST_DELIQ_BAL_LESS_30 DECIMAL(18,4),
    AA_REST_DELIQ_KOL_LESS_30 INTEGER)
AS
declare variable dolya double precision;
declare variable aa_amount_active_end decimal(18,4);
declare variable aa_loan_amount decimal(18,4);
declare variable aa_deliq_balance decimal(18,4);
declare variable kc_has_loans integer;
declare variable kc_has_family_loans integer;
declare variable aa_deliq_balance7 decimal(18,4);
declare variable aa_deliq_balance30 decimal(18,4);
declare variable aa_deliq_balance60 decimal(18,4);
declare variable aa_deliq_balance90 decimal(18,4);
declare variable aa_deliq_balance180 decimal(18,4);
declare variable aa_deliq_balance360 decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
declare variable aa_rest_norm_bal decimal(18,4);
declare variable aa_rest_deliq_bal decimal(18,4);
declare variable aa_rest_period_bal decimal(18,4);
declare variable k_amount_approved decimal(18,4);
declare variable aa_repay_writeoff_os decimal(18,4);
declare variable aa_repay_writeoff_perc decimal(18,4);
declare variable aa_repay_writeoff_fine decimal(18,4);
declare variable kc_from_c_id id_type;
declare variable k_from_kf_id id_type;
begin
-- knclient_fix

  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for with tab1
      as (select aa_k_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name, aa_curr_kc_fio,
                 aa_amount_active_end, aa_loan_amount, aa_deliq_balance, aa_deliq_days, aa_deliq_repay_kolich,
                 dd_disb_date, aa_deliq_kolich, aa_deliq_kolich7, aa_deliq_kolich30, aa_deliq_kolich60,
                 aa_deliq_kolich90, aa_deliq_kolich180, aa_deliq_kolich360, aa_deliq_balance7, aa_deliq_balance30,
                 aa_deliq_balance60, aa_deliq_balance90, aa_deliq_balance180, aa_deliq_balance360, aa_rest_norm_bal,
                 aa_rest_deliq_bal, aa_rest_norm_kol, aa_rest_deliq_kol, aa_rest_period_kol, aa_rest_period_bal,
                 aa_repay_writeoff_os, aa_repay_writeoff_perc, aa_repay_writeoff_fine, aa_product_name,
                 (select cs_name
                  from llc_main, llc_kont, classific
                  where lm_is_signed = 1 -- выборка только по утвержденным протоколам
                        and
                        lk_from_lm_id = lm_id and
                        :rep_date_end between lk_date and lk_date_end and
                        lk_status = cs_id and
                        lk_from_k_id = aa_k_id and
                        aa_amount_active_end <> 0),
                 (select lk_percent
                  from llc_main, llc_kont, classific
                  where lm_is_signed = 1 -- выборка только по утвержденным протоколам
                        and
                        lk_from_lm_id = lm_id and
                        :rep_date_end between lk_date and lk_date_end and
                        lk_status = cs_id and
                        lk_from_k_id = aa_k_id and
                        aa_amount_active_end <> 0),
                 (select lk_is_indiv
                  from llc_main, llc_kont, classific
                  where lm_is_signed = 1 -- выборка только по утвержденным протоколам
                        and
                        lk_from_lm_id = lm_id and
                        :rep_date_end between lk_date and lk_date_end and
                        lk_status = cs_id and
                        lk_from_k_id = aa_k_id and
                        aa_amount_active_end <> 0), aa_rest_deliq_kol30, aa_rest_deliq_bal30
          from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 0, 0, 0, 0, 0)
          /*where AA_AMOUNT_ACTIVE_END<>0*/ )

      select kc_from_c_id, kc_amount_approved, kc_has_loans, kc_has_family_loans, k_kontract_no, k_amount_approved,
             c_name_fio, c_pass_no, c_birth_date, aa_k_id, aa_curr_office_name, aa_curr_branch_name, aa_curr_level_name,
             aa_curr_kc_fio, aa_amount_active_end, aa_loan_amount, aa_deliq_balance, aa_deliq_days,
             aa_deliq_repay_kolich, dd_disb_date, aa_deliq_kolich, aa_deliq_kolich7, aa_deliq_kolich30,
             aa_deliq_kolich60, aa_deliq_kolich90, aa_deliq_kolich180, aa_deliq_kolich360, aa_deliq_balance7,
             aa_deliq_balance30, aa_deliq_balance60, aa_deliq_balance90, aa_deliq_balance180, aa_deliq_balance360, --AA_KF_NAME,
             aa_rest_norm_bal, aa_rest_deliq_bal, aa_rest_norm_kol, aa_rest_deliq_kol, aa_rest_period_kol,
             aa_rest_period_bal, aa_repay_writeoff_os, aa_repay_writeoff_perc, aa_repay_writeoff_fine, aa_product_name,
             cs_name, lk_percent, lk_is_indiv, aa_rest_deliq_kol30, aa_rest_deliq_bal30, k_from_kf_id
      from knclient, tab1, clients, kontract
      where aa_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id and
            kc_from_k_id = k_id
      into :kc_from_c_id, :kc_amount_approved, :kc_has_loans, :kc_has_family_loans, :k_kontract_no, :k_amount_approved,
           :c_name_fio, :c_pass_no, :c_birth_date, :aa_k_id, :aa_curr_office_name, :aa_curr_branch_name,
           :aa_curr_level_name, :aa_curr_kc_fio, :aa_amount_active_end, :aa_loan_amount, :aa_deliq_balance,
           :aa_deliq_days, :aa_deliq_repay_kolich, :dd_disb_date, :aa_deliq_kolich, :aa_deliq_kolich7,
           :aa_deliq_kolich30, :aa_deliq_kolich60, :aa_deliq_kolich90, :aa_deliq_kolich180, :aa_deliq_kolich360,
           :aa_deliq_balance7, :aa_deliq_balance30, :aa_deliq_balance60, :aa_deliq_balance90, :aa_deliq_balance180,
           :aa_deliq_balance360,
--         :AA_KF_NAME,
           :aa_rest_norm_bal, :aa_rest_deliq_bal, :aa_rest_norm_kol, :aa_rest_deliq_kol, :aa_rest_period_kol,
           :aa_rest_period_bal, :aa_repay_writeoff_os, :aa_repay_writeoff_perc, :aa_repay_writeoff_fine,
           :aa_product_name, :cs_name, :lk_percent, :lk_is_indiv, :aa_rest_deliq_kol30, :aa_rest_deliq_bal30,
           :k_from_kf_id
  do
  begin
    dolya = cast(kc_amount_approved as double precision) / k_amount_approved;

    k_cli_balance = :aa_amount_active_end * :dolya;
    k_cli_deliq_balance = :aa_deliq_balance * :dolya;
    k_cli_deliq_balance7 = :aa_deliq_balance7 * :dolya;
    k_cli_deliq_balance30 = :aa_deliq_balance30 * :dolya;
    k_cli_deliq_balance60 = :aa_deliq_balance60 * :dolya;
    k_cli_deliq_balance90 = :aa_deliq_balance90 * :dolya;
    k_cli_deliq_balance180 = :aa_deliq_balance180 * :dolya;
    k_cli_deliq_balance360 = :aa_deliq_balance360 * :dolya;

    k_cli_rest_norm_bal = :aa_rest_norm_bal * :dolya;
    k_cli_rest_deliq_bal = :aa_rest_deliq_bal * :dolya;

    /*-->> igusev 25.07.13 */
    aa_rest_deliq_bal30 = :aa_rest_deliq_bal30 * :dolya;

    aa_rest_deliq_bal_less_30 = k_cli_rest_deliq_bal - aa_rest_deliq_bal30; -- баланс реструктуризированных с просроч.  до 30 дней
    aa_rest_deliq_kol_less_30 = aa_rest_deliq_kol - aa_rest_deliq_kol30; -- кол-во реструктуризированных с просроч. до 30 дней
    if (:aa_rest_deliq_kol_less_30 > 1) then
      aa_rest_deliq_kol_less_30 = 1;
    if (:aa_rest_deliq_kol30 > 1) then
      aa_rest_deliq_kol30 = 1;
    /*<--*/

    k_cli_rest_period_bal = :aa_rest_period_bal * :dolya;
    if (:aa_rest_norm_kol > 1) then
      aa_rest_norm_kol = 1;
    if (:aa_rest_deliq_kol > 1) then
      aa_rest_deliq_kol = 1;
    if (:aa_rest_period_kol > 1) then
      aa_rest_period_kol = 1;

    k_cli_repay_woff_os = :aa_repay_writeoff_os * :dolya;
    k_cli_repay_woff_perc = :aa_repay_writeoff_perc * :dolya;
    k_cli_repay_woff_fine = :aa_repay_writeoff_fine * :dolya;

    kc_has_loans_str = null;
    kc_has_family_loans_str = null;
    kc_has = null;
    if (kc_has_loans = 1 or (exists(select 1
                                    from kontstate ks1, kontstate ks2, knclient kc1, knclient kc2
                                    where ks1.ks_ol_is_delete = 0 and
                                          ks1.ks_state = 1 and
                                          :rep_date_end between ks1.ks_date_begin and ks1.ks_date_end and
                                          ks1.ks_from_k_id = kc1.kc_from_k_id and
                                          ks2.ks_ol_is_delete = 0 and
                                          ks2.ks_state = 1 and
                                          :rep_date_end between ks2.ks_date_begin and ks2.ks_date_end and
                                          ks2.ks_from_k_id = kc2.kc_from_k_id and
                                          kc1.kc_from_c_id = kc2.kc_from_c_id and
                                          kc1.kc_from_c_id = :kc_from_c_id -- Привязка с внешней таблицей
                                          and
                                          kc1.kc_from_k_id <> kc2.kc_from_k_id and
                                          kc1.kc_amount_approved > 0 and
                                          kc2.kc_amount_approved > 0 and
                                          kc1.kc_has_loans = 0 and
                                          ks1.ks_date_begin <= :rep_date_end and
                                          kc1.kc_from_k_id = :aa_k_id))) then
    begin
      if (kc_has_loans = 1) then
        kc_has_loans_str = 'Клиент имеет кредит в других ФКУ';
      kc_has = '1. имеет параллельный кредит';
    end
    else
      kc_has = '2. не имеет параллельный кредит';
    if (kc_has_family_loans = 1) then
      kc_has_family_loans_str = 'У родственника есть кредит, имеют единный источник дохода';

    aa_kf_name = (select kal_new_kf_name
                  from knaktlist
                  where knaktlist.is_delete = 0 and
                        :rep_date_end between kal_date and kal_date_end and
                        kal_from_kf_id = :k_from_kf_id);

    suspend;
  end

end
