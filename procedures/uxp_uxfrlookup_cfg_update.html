CREATE PROCEDURE UXP$UXFRLOOKUP_CFG_UPDATE(
    UFP_UUID D_UUID,
    ULP_SOURCE D_STR_CODE,
    ULP_LOOKUP_ID D_STR_CODE,
    ULP_LOOKUP_TEXT D_STR_CODE,
    ULP_LINK_KIND D_SMALLINT = 0)
AS
declare variable ufp_ufr_uuid d_uuid;
declare variable ulp_uf_uuid d_uuid;
declare variable us_uuid d_uuid;
declare variable updated d_bool;
begin
  select us_uuid
  from ux$source
  where us_code = :ulp_source
  into us_uuid;

  updated = 0;
  if (not exists(select * from ux$frlookup
                 where ufl_ufp_uuid = :ufp_uuid and
                       ufl_us_uuid = :us_uuid)) then
  begin
    update or insert into ux$frlookup (ufl_ufp_uuid, ufl_us_uuid, ufl_lookup_id, ufl_lookup_text, ufl_list_kind)
    values (:ufp_uuid, :us_uuid, :ulp_lookup_id, :ulp_lookup_text, :ulp_link_kind)
    matching (ufl_ufp_uuid);

    updated = 1;
  end else
  begin
    update ux$frlookup
    set ufl_lookup_id = :ulp_lookup_id,
        ufl_lookup_text = :ulp_lookup_text,
        ufl_list_kind = :ulp_link_kind
    where ufl_ufp_uuid = :ufp_uuid and
          (ufl_lookup_id is distinct from :ulp_lookup_id or
           ufl_lookup_text is distinct from :ulp_lookup_text or
           ufl_list_kind is distinct from :ulp_link_kind);
  end

  if (updated = 1) then
  begin
    delete from ux$frparam
    where ufp_parent = :ufp_uuid;
  
    update ux$frparam
    set ufp_editor = 'lookup'
    where ufp_uuid = :ufp_uuid
    returning ufp_ufr_uuid
    into ufp_ufr_uuid;
  
    for select uf_uuid
    from ux$field
    where uf_owner = :us_uuid and
          uf_is_param = 1
    into ulp_uf_uuid
    do
    begin
      insert into ux$frparam (ufp_ufr_uuid, ufp_uf_uuid, ufp_parent)
      values (:ufp_ufr_uuid, :ulp_uf_uuid, :ufp_uuid);
    end
  end
end
