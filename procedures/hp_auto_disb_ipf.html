CREATE PROCEDURE HP_AUTO_DISB_IPF(
    EDL_OBJECT_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable kma_obj_kind d_integer;
declare variable kma_edoc_kind d_integer;
declare variable pr_code d_str_100;
declare variable can_disb d_smallint;
declare variable kma_id d_id;
declare variable ag_id d_id;
declare variable c_id d_id;
declare variable c_type d_smallint;
declare variable benefic_c_id d_id;
declare variable reg_id d_integer;
declare variable manager_id d_id;
declare variable un_id d_id;
declare variable acc_kt_1 d_str_20;
declare variable acc_dt_1 d_str_20;
begin
  --Автоматическая выдача работает только по продуктам Мурабаха или Иджара товары
  res_text = '';

  select pr_code
  from kontract kt
  inner join product pr on pr.pr_id = kt.k_from_pr_id
  where kt.k_id = :edl_object_id
  into pr_code;

  if (pr_code not containing '.murabaha.' and
      pr_code not containing '.idjara_goods.') then
    exit;

  kma_obj_kind = f_objkind('kontmatag', 'id');
  kma_edoc_kind = f_edoc_kind(iif(:pr_code containing '.murabaha.', trim('purchase_act.agreement'), trim('idjara_goods.contragent_act')), 'id');

  can_disb = 1;

  --Для автоматической выдачи по мурабаха нужны подписанные акты приёма-перадачи
  --Если акт не был подписан или аннулирован, то выдача невозможна
  for select kma.kma_id, kma.kma_from_ag_id
      from kontmatag kma
      where kma.kma_from_k_id = :edl_object_id
      into kma_id, ag_id
  do
  begin
    if (not exists(select *
                   from edoc_list edl
                   where edl.edl_from_edc_id = :kma_edoc_kind and
                         edl.edl_object_kind = :kma_obj_kind and
                         edl.edl_object_id = :kma_id and
                         edl.edl_status = 1)) then
    begin
      --Доработка JIRA-667 KG
      --Проверяем является ли продавец юр лицом и есть ли у продавца активный договор ЭЦП на текущий день
      --Проверка действует только для продавцов юр лиц
      --Если продавец не имеет активного договора ЭЦП то выдать кредит можно (согласованно с Кыял)
      --Если же продавец имееет активный договор ЭЦП, но не подписал акт приёма передачи товара(проверка выше строка 173), то выдачу не делаем
      can_disb = 0;
      c_id = null;
      select ag.ag_object_id
      from agreements ag
      where ag.ag_id = :ag_id
      into c_id;

      c_type = null;
      select cl.c_type
      from clients cl
      where cl.c_id = :c_id
      into c_type;

      if (coalesce(c_type, 0) = 1) then
      begin
        benefic_c_id = null;
        select cb.cb_ben_from_c_id
        from clientbenefic cb
        where cb.cb_from_c_id = :c_id and
              current_date between cb.cb_date_begin and coalesce(cb.cb_date_end, '31.12.9999')
        into benefic_c_id;

        if (coalesce(benefic_c_id, 0) <> 0) then
          can_disb = iif(f_asp_signed(:benefic_c_id, current_date) <> 1, 1, 0);
        else
          can_disb = iif(f_asp_signed(:c_id, current_date) <> 1, 1, 0);

        if (can_disb = 0 and exists(
          select *
          from agreement_esign_access
          where aea_from_ag_id = :ag_id
           and not aea_sign))
        then
          can_disb = 1;
      end
    end
  end

  if (can_disb = 1) then
  begin
    select ks.ks_region_id
    from kontstate ks
    where ks.ks_from_k_id = :edl_object_id and
          ks.ks_state = 0
    into reg_id;

    --Перезаписываю переменную c_id
    select kt.k_clients_cashier_id, kt.k_approved_manager
    from kontract kt
    where kt.k_id = :edl_object_id
    into c_id, manager_id;

    --Выдача "вешается" на сотрудника который утвердил займ
    select kl.kl_user_id
    from kontlog kl
    where kl.kl_from_k_id = :edl_object_id and
          kl.kl_state = 1
    order by kl.kl_datetime desc
    rows 1
    into un_id;

    if (coalesce(un_id, 0) = 0) then
      res_text = 'Для выдачи займа, договор должен быть утверждён';

    acc_kt_1 = (select al_id
                from acc_list
                where al_from_reg_id = :reg_id and
                      al_currency = '417' and
                      al_account = (case
                                      when :pr_code containing '.murabaha.' then '11209/1'
                                      when :pr_code containing '.idjara_goods.' then '11207/3'
                                      else null
                                    end));

    if (coalesce(acc_kt_1, '') = '') then
      res_text = 'счет кт не определен';

    acc_dt_1 = (select al_id
                from acc_list
                where al_from_reg_id = :reg_id and
                      al_currency = '417' and
                      al_account = (case
                                      when :pr_code containing '.murabaha.' then '10977/1'
                                      when :pr_code containing '.idjara_goods.' then '10972/3'
                                      else null
                                    end));

    if (coalesce(acc_dt_1, '') = '') then
      res_text = 'счет дт не определен';

    if (res_text = '') then
    begin
      select res_text
      from hp_do_disbursement(:un_id, :edl_object_id, :reg_id, 3, :acc_kt_1, :reg_id, 1, :acc_dt_1, :c_id, :manager_id)
      into res_text;
    end
  end
  suspend;
end
