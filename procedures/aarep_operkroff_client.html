CREATE PROCEDURE AAREP_OPERKROFF_CLIENT(
    C_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    K_ID DECIMAL(18,0),
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_OTHER DECIMAL(18,4),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    OL_DATE_WRITEOFF DATE,
    WOFF_OS DECIMAL(18,4),
    WOFF_PERC DECIMAL(18,4),
    WOFF_OTHER DECIMAL(18,4),
    KF_NAME_ORIG VARCHAR(100),
    KF_NAME_CURR VARCHAR(100),
    KF_CURR_REGION_ID INTEGER,
    IS_IPF D_BOOL)
AS
declare variable k_from_kf_id decimal(18,0);
begin
  select c_name_fio, c_pass_no, c_pass_date
  from clients
  where c_id = :c_id
  into :c_name_fio, :c_pass_no, c_pass_date;

  for select oko_from_k_id, sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc), sum(oko_other + oko_sh_other)
      from operkroff, operlist
      where oko_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            oko_from_c_id = :c_id
      group by oko_from_k_id
      into :k_id, :sh_os, :sh_perc, :sh_other
  do
  begin
    if (sh_os is null) then
      sh_os = 0;
    if (sh_perc is null) then
      sh_perc = 0;
    if (sh_other is null) then
      sh_other = 0;
    if ((sh_os <> 0) or (sh_perc <> 0) or (sh_other <> 0)) then
    begin
      woff_os = 0;
      woff_perc = 0;
      woff_other = 0;

      select k_kontract_no, k_disb_date, k_from_kf_id, iif(pr_code like '%.ipf.%', 1, 0)
      from kontract
        inner join product
          on pr_id = k_from_pr_id
      where k_id = :k_id
      into :k_kontract_no, :k_disb_date, :k_from_kf_id, :is_ipf;

      select min(ol_date), sum(oko_sh_os), sum(oko_sh_perc), sum(oko_sh_other)
      from operkroff, operlist
      where oko_from_c_id = :c_id and
            oko_from_k_id = :k_id and
            oko_type = 1 and
            oko_from_ol_id = ol_id and
            operlist.is_delete = 0
      into :ol_date_writeoff, :woff_os, :woff_perc, :woff_other;

      select kal_new_reg_id, kal_new_kf_name
      from knaktlist
      where is_delete = 0 and
            kal_from_kf_id = :k_from_kf_id and
            kal_date <= :ol_date_writeoff and
            kal_date_end >= :ol_date_writeoff rows 1
      into :kf_curr_region_id, :kf_name_orig;
      if (kf_name_orig is null) then
        kf_name_orig = '';

      select kal_new_kf_name
      from knaktlist
      where is_delete = 0 and
            kal_from_kf_id = :k_from_kf_id and
            kal_date <= current_date and
            kal_date_end >= current_date rows 1
      into :kf_name_curr;
      if (kf_name_curr is null) then
        kf_name_curr = '';

      if (woff_os is null) then
        woff_os = 0;
      if (woff_perc is null) then
        woff_perc = 0;
      if (woff_other is null) then
        woff_other = 0;

      if ((sh_os + sh_perc + sh_other) > 0) then
        suspend;
    end
  end
end
