CREATE PROCEDURE HP_PRE_SUD_CLIENTS_INFO(
    REP_C_ID ID_TYPE)
RETURNS (
    C_NAME_FIO STR_100,
    C_SEX STR_20,
    C_BIRTH_DATE D_DATE,
    C_BIRTH_PLACE STR_255,
    C_RESIDENT STR_100,
    C_CITIZEN STR_100,
    ADDRESS STR_255,
    CELL_MAIN STR_20,
    CLI_CONTACTS D_STR_LONG,
    CONT_EMAIL STR_100,
    SOCIAL_MEDIA D_STR_LONG,
    WORK_PLACE_NAME STR_255,
    WORK_STATUS_KIND STR_255,
    WORK_MONTHLY_INCOME STR_255,
    OWNERSHIP STR_255,
    PHOTO ID_TYPE)
AS
begin
  select c_name_fio, c_sex_str, c_birth_date, c_resident_str, c_citizen_str, c_birth_country, address_home, cell_main,
         cont_email,
         'Facebook - ' || cont_facebook || ascii_char(13) || 'VK - ' || cont_vkontakte || ascii_char(13) || 'Одноклассники - ' || cont_odnoklassniki || ascii_char(13) || 'Telegram - ' || cont_telegramm || ascii_char(13) || 'Imo - ' || cont_imo || ascii_char(13) || 'Whatsapp - ' || cont_whatsapp || ascii_char(13) || 'Viber - ' || cont_viber || ascii_char(13) || 'Skype - ' || cont_skype || ascii_char(13),
         work_place_name, work_status_kind, work_monthly_income, ownership, coalesce(contact_no_rodina, '') || ', ' || coalesce(contact_no_ru, '')
  from aarep_client_data_print(0, :rep_c_id)
  into c_name_fio, c_sex, c_birth_date, c_resident, c_citizen, c_birth_place, address, cell_main, cont_email,
       social_media, work_place_name, work_status_kind, work_monthly_income, ownership, cli_contacts;

  select c_blob_photo_id
  from clients
  where c_id = :rep_c_id
  into photo;
  suspend;
end
