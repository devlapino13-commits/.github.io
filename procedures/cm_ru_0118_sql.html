CREATE PROCEDURE CM_RU_0118_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable KF_CLIENT_PASS_NO varchar(30);
declare variable KC_FROM_C_ID ID_TYPE;
declare variable C_CITIZEN integer;
declare variable RD_CODE varchar(255);
declare variable PDT_CODE varchar(255);
declare variable C_PASS_DATE date;
declare variable C_PASS_DATE_EXPIRE date;
declare variable K_KOLICH_REPAY integer;
declare variable KC_CIRCLE integer;
declare variable K_KREDLINE_END date;
declare variable KF_CLIENT_PASS_NO_TMP varchar(30);
declare variable DOC_DATE_BEGIN date;
declare variable DOC_DATE_END date;
declare variable C_CURR_STATUS integer;
declare variable REGISTRATION_ADDRESS STR_10;
declare variable DOC_FOREIGNERS STR_10;
declare variable ARRIVAL_DATE_LONGER_90_DAYS STR_10;
declare variable ADDIT_DOC_FOREIGNER STR_10;
declare variable MIGR_CARD STR_10;
declare variable MIGR_DATE date;
declare variable MIGR_DATE_END date;
declare variable PR_CODE D_STR_100;
declare variable V_EXPIRE D_DATE;
begin
  for select KC_FROM_C_ID, KC_AMOUNT_APPROVED, K_DISB_DATE, K_KOLICH_REPAY, KC_CIRCLE, KONT.K_KREDLINE_END,
             pr_code
      from KNCLIENT
      inner join KONTRACT KONT on K_ID = KC_FROM_K_ID
      inner join PRODUCT on PR_ID = K_FROM_PR_ID
      where KC_FROM_K_ID = :K_ID and
            KC_AMOUNT_APPROVED > 0
      into KC_FROM_C_ID, KC_AMOUNT, KF_DISB_DATE, K_KOLICH_REPAY, KC_CIRCLE, K_KREDLINE_END, pr_code
  do
  begin
    C_CITIZEN = null;
    RD_CODE = null;
    select C_NAME_FIO, C_CITIZEN, C_PASS_NO, C_PASS_DATE, C_PASS_DATE_EXPIRE
    from CLIENTS
    where C_ID = :KC_FROM_C_ID
    into res_info_text, C_CITIZEN, KF_CLIENT_PASS_NO, C_PASS_DATE, C_PASS_DATE_EXPIRE;
    select RD_CODE
    from REFDATA
    where RD_ID = :C_CITIZEN
    into RD_CODE;
    if (RD_CODE not in ('Citizen.RU', 'Citizen.KG', 'Citizen.TJ', 'Citizen.UZ', 'Citizen.KZ')) then
    begin
      res_error_text = 'Только граждане РФ, КГ, ТЖ, УЗ и КЗ могут получить займ';
      res_error_is_critical = 1;
      suspend;
      exit;
    end

    v_expire = c_pass_date_expire;
    if (rd_code = 'Citizen.RU') then
      v_expire = dateadd(90 day to v_expire);

    if (KF_DISB_DATE not between C_PASS_DATE and v_expire) then
    begin
      res_error_text = 'Срок паспорта не действителен!';
      res_error_is_critical = 1;
      suspend;
    end
    /*
    SYSDBA
    |чтобы без нельзя было получить заем без этих основных документов:
    паспорт, регистрация (для РФ), + МК/РВП/ВНЖ (для иностранцев).
    А если срок пребывания в РФ более 90 дней должен быть дополнительный документ
    (патент, разрешение на работу, справка о подаче на РВП, ВНЖ, Патент или гражданство,
    трудовой договор и т.п.). 
                */
    REGISTRATION_ADDRESS = 'NO'; --Регистрация, должна быть у всех
    DOC_FOREIGNERS = 'NO'; --Основной документы для иностранца
    MIGR_DATE = cast('01.01.01' as date); --Дата приезда в РФ, значение по умолчанию. Внизу берет дату из миг.карты
    ARRIVAL_DATE_LONGER_90_DAYS = 'NO'; --Срок пребывание не больше 90 дней, если больше то поле равен будет "YES"
    ADDIT_DOC_FOREIGNER = 'NO'; --ДОполнительный документ для иностранца
    for select PDT_CODE, coalesce(CDS_DOC_DATE_BEGIN, '01.01.01'), coalesce(CDS_DOC_DATE_END, '31.12.1999')
        from CLIENTDOCS CDS, PASSKIND, PASSDOCTYPE
        where CDS_FROM_C_ID = :KC_FROM_C_ID and
              CDS_IS_DELETE = 0 and
              ((cast(:KF_DISB_DATE as date) between coalesce(CDS_DOC_DATE_BEGIN, '01.01.01') and coalesce(CDS_DOC_DATE_END, '31.12.9999')) or
               (PDT_CODE in ('registration_fiz', 'registration_yur', 'registration_permanent') and
                :RD_CODE not in ('Citizen.TJ', 'Citizen.UZ') and
                CDS_DOC_DATE_END between cast('15.03.2020' as date) and cast('15.06.2021' as date))) and
              --              coalesce(CDS.CDS_DOC_BLOB_1, 0) > 0 and
              CDS_FROM_PK_ID = PK_ID and
              PK_FROM_PDT_ID = PDT_ID
        into PDT_CODE, DOC_DATE_BEGIN, DOC_DATE_END
    do
    begin
      if (REGISTRATION_ADDRESS = 'NO' and
          PDT_CODE in ('registration_fiz', 'registration_yur', 'registration_permanent')) then
      begin
        REGISTRATION_ADDRESS = 'YES';
      end
      if (PDT_CODE in ('magration_card', 'registration_rvp', 'registration_vnj')) then
      begin
        if (PDT_CODE = 'magration_card') then
        begin
          MIGR_DATE = DOC_DATE_BEGIN; --Дата приезда в РФ
          MIGR_DATE_END = DOC_DATE_END; --Srok
          MIGR_CARD = 'YES';
        end
        else
          DOC_FOREIGNERS = 'YES';
      end
      if (ADDIT_DOC_FOREIGNER = 'NO') then
      begin
        if (RD_CODE in ('Citizen.KG','Citizen.KZ') and
            PDT_CODE = 'job_agreement' -- трудовой договор - только для граждан КР.
            ) then
        begin
          ADDIT_DOC_FOREIGNER = 'YES';
        end
        if (ADDIT_DOC_FOREIGNER = 'NO' and
            PDT_CODE in (--
                     'work_patent', -- патент; Свидетельство участника программы добровольного переселения(один PDT_CODE)
                     'work_permit', -- разрешение на работу;
                     'work_permit_WTO', -- РНР членов Всемирной торговой организации
                     'application_citizenship_rassian_lang_speaker', -- заявление о гражданстве носителя рус. языка
                     'application_rezidence_permit_vnj', -- заявление о выдаче ВНЖ;
                     'request_VKS', -- ходатайство работодателя на привлечение к работе высококвалифицированного специалиста; 
                     'work_permit_application_extension_VKS', -- заявление работодателя о продлении РНР ВКС;
                     'certificate_approving_doc', -- документ о подаче на патент, РВП или ВНЖ;
                     'student_card' -- студенческий билет;
                     )) then
        begin
          ADDIT_DOC_FOREIGNER = 'YES';
        end
      end
    end
    --Блоки:
    if (REGISTRATION_ADDRESS = 'NO' and pr_code not containing '.express_online.') then --Для всех
    begin
      res_error_text = 'Не найдена актуальная регистрация физ./юр. лица или постоянная регистрация в РФ!';
      res_error_is_critical = 1;
      suspend;
    end
    if (RD_CODE <> 'Citizen.RU') then --Для иностранцев
    begin
      if (DOC_FOREIGNERS = 'NO' and
          MIGR_CARD = 'NO') then
      begin
        res_error_text = 'Должен быть прикреплен хотя бы один из документов: миграц. карта, РВП, ВНЖ';
        res_error_is_critical = 1;
        suspend;
      end
      else
      if (KF_DISB_DATE - MIGR_DATE > 90 and
          MIGR_DATE_END < KF_DISB_DATE and
          ADDIT_DOC_FOREIGNER = 'NO' and
          DOC_FOREIGNERS = 'NO') then
      begin
        res_error_text = 'Срок пребывание больше 90-дней. Необходимо добавить дополнительный документ патент, РНР, справка о подаче на РВП или ВНЖ, Патент, гражданство, трудовой договор и т.п.)';
        res_error_is_critical = 1;
        suspend;
      end
    end
  end
end
