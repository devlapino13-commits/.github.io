CREATE PROCEDURE HP_REPAYMENT_DETAIL_PLAN_KG(
    K_ID D_ID,
    REP_DATE D_DATE,
    K_KONTRACT_NO D_STR_CODE = null)
RETURNS (
    SH_DATE D_DATE,
    ORIG_OS D_CURRENCY,
    ORIG_PERC D_CURRENCY,
    ORIG_ALL D_CURRENCY,
    OK_MOVED_OS D_CURRENCY,
    OK_SH_OS D_CURRENCY,
    OK_SH_PERC D_CURRENCY,
    OK_SH_ALL D_CURRENCY,
    OK_ZAD_OS D_CURRENCY,
    OK_ZAD_PERC D_CURRENCY,
    OK_INFO D_STR_CODE)
AS
declare variable ok_sh_id d_integer;
declare variable ol_id d_id;
declare variable k_stop_perc d_date;
declare variable ok_moved_date d_date;
declare variable pr_code d_str_100;
declare variable ol_date d_date;
declare variable sh_last_date d_date;
declare variable is_vu d_smallint;
declare variable oko_zad_os d_currency;
declare variable oko_zad_perc d_currency;
begin
  if (k_id is null and
      k_kontract_no <> '') then
  begin
    select k_id
    from kontract k
    where k_kontract_no = :k_kontract_no
    into k_id;
  end

  select k_stop_perc, pr_code
  from kontract k
  inner join product pr on k.k_from_pr_id = pr.pr_id
  where k_id = :k_id
  into k_stop_perc, pr_code;

  for select sh.sh_date, sh.sh_os, sh.sh_perc
      from schedule sh
      where sh.sh_from_k_id = :k_id
      into :sh_date, :orig_os, :orig_perc
  do
  begin
    orig_all = orig_os + orig_perc;

    ok_sh_os = 0;
    ok_sh_perc = 0;

    sh_last_date = null;
    select first 1 sh_date
    from schedule sh
    where sh.sh_from_k_id = :k_id
    order by sh.sh_date desc
    into :sh_last_date;

    select sum(ok_sh_os), sum(ok_sh_perc)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date and
          ok_sh_date = :sh_date and
          ok_type <= 3
    into :ok_sh_os, ok_sh_perc;

    if (ok_sh_os is null) then
      ok_sh_os = 0;

    if (ok_sh_perc is null) then
      ok_sh_perc = 0;

    ok_sh_all = ok_sh_os + ok_sh_perc;

    ok_moved_os = 0;

    ok_moved_os = ok_sh_os - orig_os;

    is_vu = 0;

    if (exists(select *
               from operkroff oko
               where oko.oko_from_k_id = :k_id and
                     oko.oko_ol_is_active = 1 and
                     oko.oko_ol_date <= :rep_date)) then
      is_vu = 1;

    if (is_vu = 1) then
    begin
      ok_zad_os = null;
      ok_zad_perc = null;

      for select sum(oko.oko_sh_os + oko.oko_os), sum(oko.oko_sh_perc + oko.oko_perc)
          from operkroff oko
          where oko.oko_from_k_id = :k_id and
                oko.oko_ol_is_active = 1 and
                oko.oko_ol_date <= :rep_date
          into :oko_zad_os, :oko_zad_perc
      do
      begin
        oko_zad_os = coalesce(oko_zad_os, 0);
        oko_zad_perc = coalesce(oko_zad_perc, 0);
      end
    end

    ok_zad_os = 0;
    ok_zad_perc = 0;

    select sum(zad.zad_os), sum(zad.zad_perc)
    from aarep_make_zad_list(:k_id, :sh_date, 1, 0) zad
    where zad.sh_date = :sh_date
    into :ok_zad_os, :ok_zad_perc;

    if (ok_zad_os is null) then
      ok_zad_os = 0;

    if (ok_zad_perc is null) then
      ok_zad_perc = 0;

    ok_info = '';
    ok_moved_date = null;
    ok_sh_id = null;
    ol_id = null;

    select ok_sh_id, ok_from_ol_id
    from operkred, operlist
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_sh_date = :sh_date and
          ok_from_ol_id = ol_id and
          ol_oper_type = 1 and
          ol_oper_subtype = 9 and
          (ok_sh_os < 0 or ok_sh_perc < 0)
    into ok_sh_id, ol_id;

    if (ok_sh_id is not null) then
    begin
      select ok_sh_date
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_sh_id = :ok_sh_id and
            ok_from_ol_id = :ol_id and
            ok_sh_date > :sh_date
      into ok_moved_date;

      if (ok_moved_date is not null) then
        ok_info = f_datetostr(ok_moved_date);
    end

    if (k_stop_perc = sh_date) then
    begin
      if (ok_info <> '') then
        ok_info = ok_info || ', договор расторгнут';
      else
        ok_info = 'Договор расторгнут';
    end

    if (is_vu = 1) then
    begin
      ok_zad_os = 0;
      ok_zad_perc = 0;
      if (sh_date = sh_last_date) then
      begin
        ok_zad_os = oko_zad_os;
        ok_zad_perc = oko_zad_perc;
      end
    end
    suspend;
  end
end
