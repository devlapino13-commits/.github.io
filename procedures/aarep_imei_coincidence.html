CREATE PROCEDURE AAREP_IMEI_COINCIDENCE(
    REP_REG_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_CODE,
    KONTRAGENT_FIO D_STR_100,
    KF_OFFICER_FIO D_STR_100,
    FOUND_IMEI D_ID,
    EXISTING_KONTRACT D_STR_20,
    FINAL_IMEI D_STR_255,
    CLIENT_FIO D_STR_100)
AS
declare variable K_ID D_ID;
declare variable KMI_IMEI D_ID;
begin
  ----RFCKG-4855 Отчет по совпадениям IMEI
  for select reg.reg_name, c.c_name_fio, k.k_kontract_no, c2.c_name_fio, kmi.kmi_imei, di.di_from_k_id
  from duplicated_imei di
  inner join region_list(:rep_reg_id, 1, 3, 1) rl on di.di_last_reg_id = rl.id_list
  inner join region reg on reg.reg_unicode = di.di_last_reg_id
  inner join clients c on c.c_id = di.di_kf_officer
  inner join kontmat kml on kml.kml_id = di.di_from_kml_id
  inner join kontmat_imei kmi on kmi.kmi_kml_id = kml.kml_id
  inner join kontract k on k.k_id = kml.kml_object_id
  inner join agreements ag on ag.ag_id = kml.kml_from_ag_id
  inner join clients c2 on c2.c_id = ag.ag_object_id
  where di.di_datetime between :rep_date_begin and (:rep_date_end + 1)
  into: reg_name, kf_officer_fio, existing_kontract, kontragent_fio, found_imei, k_id
  do
  begin
    final_imei = '';
    select c.c_name_fio
    from knclient kc
    inner join clients c on c.c_id = kc.kc_from_c_id
    where kc.kc_amount_approved > 0
      and kc.kc_from_k_id = :k_id
    into :client_fio;

    for select kmi.kmi_imei
    from kontmat km
    inner join kontmat_imei kmi on kmi.kmi_kml_id = km.kml_id
    where km.kml_object_kind = f_objkind('kontract', 'id')
      and km.kml_object_id = :k_id
      and km.kml_status = 1
    into :kmi_imei
    do
    begin
      if (row_count > 0) then
        final_imei = final_imei || kmi_imei || '; ';
    end
    suspend;
  end
end
