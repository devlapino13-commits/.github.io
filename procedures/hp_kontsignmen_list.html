CREATE PROCEDURE HP_KONTSIGNMEN_LIST(
    ONLY_ACTIVE D_BOOL,
    ONLY_HR D_BOOL)
RETURNS (
    KSM_ID D_INTEGER,
    KSM_FROM_C_ID D_ID,
    KSM_IS_ACTIVE D_BOOL,
    KSM_CAN_APPROVE D_BOOL,
    KSM_HR_HQ D_INTEGER,
    KSM_HR_BRANCH D_INTEGER,
    KSM_STATUS_KIND D_STR_CODE)
AS
declare variable reg_count d_integer;
declare variable active_count d_integer;
begin
  if (only_active is null) then
    only_active = 1;

  if (only_hr is null) then
    only_hr = 0;

  for select ksm_id,
             ksm_from_c_id,
             ksm_is_active,
             ksm_can_approve,
             ksm_hr_hq,
             ksm_hr_branch
      from kontsignmen
      where (:only_active = 0 or
            (:only_active = 1 and (ksm_is_active = 1 or
                                   ksm_can_approve = 1 or
                                   ksm_hr_hq > 0 or
                                   ksm_hr_branch > 0))) and
            (:only_hr = 0 or
            (:only_hr = 1 and (ksm_hr_hq > 0 or ksm_hr_branch > 0)))
      order by ksm_id desc
      into :ksm_id,
           :ksm_from_c_id,
           :ksm_is_active,
           :ksm_can_approve,
           :ksm_hr_hq,
           :ksm_hr_branch
  do
  begin
    if (ksm_hr_hq = 0) then
      ksm_hr_hq = null;

    if (ksm_hr_branch = 0) then
      ksm_hr_branch = null;

    ksm_status_kind = 'Неактивный';
    if (ksm_is_active = 1 or ksm_can_approve = 1 or ksm_hr_hq > 0 or ksm_hr_branch > 0) then
    begin
      ksm_status_kind = 'Действующий';

      select count(*),
             sum(iif(ksr_is_active = 1 and
                     coalesce(ksr_date_begin, current_date) <= current_date and
                     coalesce(ksr_date_end, current_date) >= current_date, 1, 0))
      from kontsignreg
      where ksr_from_ksm_id = :ksm_id
      into reg_count, active_count;

      if (active_count is null) then
        active_count = 0;

      if (active_count = 0) then
      begin
        if (reg_count = 0) then
          ksm_status_kind = 'Без регионов';
        else
          ksm_status_kind = 'Срок истек';
      end
    end

    suspend;
  end
end
