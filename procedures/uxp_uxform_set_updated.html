CREATE PROCEDURE UXP$UXFORM_SET_UPDATED(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
AS
declare variable ux_kind d_str_code;
declare variable is_base d_bool;
declare variable ufm_uuid d_uuid;
declare variable ux_release_uuid d_uuid;
begin
  -- Обновляет время изменения формы
  -- Входящий параметр - любой объект, далее процедура получает код формы

  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code, 0);

  if (ux_uuid is null) then
    exit;

  select uxf$kind(uu_kind, 'code'), uu_is_base
  from ux$uniq
  where uu_id = :ux_uuid
  into ux_kind, is_base;

  if (is_base = 1) then
  begin
    select ux_draft_uuid, ux_release_uuid
    from ux$code
    where ux_uuid = :ux_uuid
    into ufm_uuid, ux_release_uuid;
  end else
  begin
    if (ux_kind = 'uxfritem') then
    begin
      select ufi_owner
      from ux$fritem
      where ufi_uuid = :ux_uuid
      into ux_uuid;

      ux_kind = uxf$code(ux_uuid, 'kind', 0);
    end else
    if (ux_kind = 'uxfrparam') then
    begin
      select ufp_ufr_uuid
      from ux$frparam
      where ufp_uuid = :ux_uuid
      into ux_uuid;

      ux_kind = uxf$code(ux_uuid, 'kind', 0);
    end else
    if (ux_kind = 'uxaction') then
    begin
      select ua_owner
      from ux$action
      where ua_uuid = :ux_uuid
      into ux_uuid;

      ux_kind = uxf$code(ux_uuid, 'kind', 0);
    end

    if (ux_kind is null) then
      ux_kind = '';

    if (ux_kind in ('uxform', 'uxcommand', 'uxreport')) then
      ufm_uuid = ux_uuid;
    else
    if (ux_kind = 'uxframe') then
    begin
      select ufr_owner
      from ux$frame
      where ufr_uuid = :ux_uuid
      into ufm_uuid;
    end else
    if (ux_kind = 'uxalias') then
    begin
      select ux_draft_uuid, ux_release_uuid
      from ux$alias, ux$code
      where uxa_uuid = :ux_uuid and
            uxa_owner = ux_uuid
      into ufm_uuid, ux_release_uuid;
    end else
    if (ux_kind = 'uxgrid') then
    begin
      select ufr_owner
      from ux$grid, ux$frame
      where ugr_uuid = :ux_uuid and
            ugr_ufr_uuid = ufr_uuid
      into ufm_uuid;
    end else
    if (ux_kind = 'uxgritem') then
    begin
      select ufr_owner
      from ux$gritem, ux$grid, ux$frame
      where ugi_uuid = :ux_uuid and
            ugi_ugr_uuid = ugr_uuid and
            ugr_ufr_uuid = ufr_uuid
      into ufm_uuid;
    end
  end

  if (ufm_uuid is not null) then
  begin
    update ux$form
    set ufm_updated = current_timestamp
    where ufm_uuid = :ufm_uuid and
          ufm_updated <> current_timestamp;
  end

  if (ux_release_uuid is not null and ux_release_uuid <> ufm_uuid) then
  begin
    update ux$form
    set ufm_updated = current_timestamp
    where ufm_uuid = :ux_release_uuid and
          ufm_updated <> current_timestamp;
  end
end
