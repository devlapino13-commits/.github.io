CREATE PROCEDURE AAREP_REPORT_FOR_TSB_3_DIV(
    REP_DATE_END DATE)
RETURNS (
    KC_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    CLIENT_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_NAME_FIO VARCHAR(100),
    K_DISB_DATE DATE,
    RES_CLIENT_IS_RESIDENT INTEGER,
    IS_SECURED INTEGER,
    OL_DATE DATE,
    IS_RESTRUCT INTEGER,
    RESTRUCT_DATE_LAST DATE,
    CLI_BALANCE CURR_TYPE,
    CLI_PERC_SALDO CURR_TYPE,
    DELINQ_DAY INTEGER,
    LK_PERCENT PERC_TYPE,
    CLI_RPPU_OS CURR_TYPE,
    CLI_RPPU_PERC CURR_TYPE,
    PO_PERCENT PERC_TYPE,
    CLI_PO_SUMM CURR_TYPE,
    TERM_IN_DAYS INTEGER,
    PERCENT_YEAR PERC_TYPE,
    PERCENT_EFFECT PERC_TYPE,
    LOAN_EMP SMALLINT,
    DELINQ_DAY_REST_DATE INTEGER,
    KC_AMOUNT DECIMAL(18,4),
    PR_NAME VARCHAR(50),
    CLI_PERCENT_SCHEDULE CURR_TYPE,
    CLI_IS_RESIDENT_ON_DISB INTEGER,
    CLI_BALANCE_ROUND CURR_TYPE,
    CLI_PERC_SALDO_ROUND CURR_TYPE,
    ACC_NO STR_20)
AS
declare variable knr_non_resident smallint;
declare variable k_id id_type;
declare variable sh_date date;
declare variable k_amount_approved curr_type;
declare variable dolya double precision;
declare variable c_id id_type;
declare variable citizen_ru_id integer;
declare variable doc_vnj_id integer;
declare variable pr_code varchar(100);
declare variable pr_loan_type integer;
declare variable k_kolich_client_approved integer;
declare variable k_id_prior id_type;
declare variable pr_product_kind integer;
declare variable pr_id integer;
declare variable oks_balance curr_type;
declare variable rep_period integer;
declare variable kcp_perc_saldo curr_type;
declare variable lk_balance curr_type;
declare variable lk_calcperc curr_type;
declare variable rppu_os curr_type;
declare variable rppu_perc curr_type;
declare variable po_summ curr_type;
declare variable lk_deliq_days integer;
declare variable max_sh_date date;
declare variable percent_schedule decimal(18,4);
declare variable k_is_devided integer;
declare variable k_base_k_id id_type;
declare variable k_base_k_id_old id_type;
declare variable lk_deliq_days_tmp integer;
declare variable delinq_day_tmp integer;
declare variable lk_percent_tmp perc_type;
declare variable pr_id_old integer;
declare variable lk_deliq_days_old integer;
declare variable period_day integer;
declare variable oks_balance_tmp curr_type;
declare variable kcp_perc_saldo_tmp curr_type;
declare variable k integer;
declare variable kol_cli integer;
declare variable oks_ecl_percent perc_type;
begin
  rep_period = extract(year from cast(:rep_date_end as date)) || right('0' || extract(month from cast(:rep_date_end as date)), 2);

  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  k_id_prior = 0;
  k_base_k_id_old = 0;
  pr_id_old = 0;
  lk_deliq_days_old = 0;

  for select oks_from_k_id, oks_balance, :rep_date_end - oks_delinq_date, k_kontract_no, k_amount_approved, k_disb_date,
             k_kolich_client_approved, coalesce(k_is_devided, 0), k_base_k_id, k_from_pr_id
      from oksummary oks, kontract kont
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks.oks_from_k_id = kont.k_id
            and oks_close_state = 0
      order by k_base_k_id, k_id
      into k_id, oks_balance, delinq_day_tmp, k_kontract_no, k_amount_approved, k_disb_date, k_kolich_client_approved,
           k_is_devided, k_base_k_id, pr_id
  do
  begin
    lk_balance = null;
    lk_calcperc = null;
    lk_percent = null;

    lk_percent_tmp = null;
    lk_deliq_days_tmp = null;

    if (pr_id <> pr_id_old) then
    begin
      select pr_code, pr_loan_type, pr_product_kind, pr_name
      from product
      where pr_id = :pr_id
      into pr_code, pr_loan_type, pr_product_kind, pr_name;

      pr_id_old = pr_id;
    end

    if (delinq_day_tmp < 0) then
      delinq_day_tmp = 0;

    if (k_is_devided = 1) then
    begin
      if (k_base_k_id <> k_base_k_id_old) then
      begin
        delinq_day = 0;
        is_restruct = 0;
        restruct_date_last = null;
        delinq_day_rest_date = null;
        is_secured = 0;

        lk_deliq_days = null;

        select res_deliq_days, res_is_restruct, res_sh_close_date, restruct_date_last, is_secured, delinq_day_rest_date,
               lk_deliq_days
        from hp_devided_kont_result(:rep_date_end, :k_base_k_id)
        into :delinq_day, :is_restruct, max_sh_date, restruct_date_last, is_secured, delinq_day_rest_date,
             lk_deliq_days;

        k_base_k_id_old = k_base_k_id;
      end
      if (k_id <> k_base_k_id) then
      begin
        select kont.k_kontract_no
        from kontract kont
        where kont.k_id = :k_base_k_id
        into :k_kontract_no;
      end
    end
    else
    begin
      delinq_day = delinq_day_tmp;
      lk_deliq_days = null;

      select is_restruct, restruct_date_last
      from aarep_get_restructuring(:k_id, :rep_date_end, 81001)
      into :is_restruct, :restruct_date_last;

      max_sh_date = null;

      select max(sh_date)
      from schedule
      where sh_from_k_id = :k_id
      into max_sh_date;

      is_secured = null;
      select is_secured
      from aarep_get_secured_loan(:pr_code, :pr_loan_type, :pr_product_kind, :pr_id, :k_disb_date, :k_kolich_client_approved, :k_id)
      into is_secured;

      delinq_day_rest_date = null;
      if (is_restruct = 1) then
      begin
        restruct_date_last = restruct_date_last - 1;
        select :restruct_date_last - oks_delinq_date
        from oksummary
        where oks_from_k_id = :k_id and
              :restruct_date_last between oks_date_begin and oks_date_end
              and oks_close_state = 0
        into delinq_day_rest_date;

        if (delinq_day_rest_date < 0) then
          delinq_day_rest_date = null;
        if (delinq_day_rest_date > 0) then
          delinq_day_rest_date = delinq_day_rest_date + 1;
        restruct_date_last = restruct_date_last + 1;
      end
    end

    if (pr_product_kind = 4) then
      loan_emp = 1;
    else
      loan_emp = 0;

    percent_year = null;
    select kd_percent_per_day * 365
    from kontdata
    where :k_disb_date between kd_ol_date and kd_date_end and
          kd_ol_is_delete = 0 and
          kd_from_k_id = :k_id
    into percent_year;

    select sum(sh_perc)
    from schedule
    where sh_from_k_id = :k_id
    into percent_schedule;

    if (max_sh_date is not null) then
      term_in_days = max_sh_date - k_disb_date;
    else
      term_in_days = null;

    kcp_perc_saldo = null;

    select lk_balance, lk_calcperc, lk_percent, lk_deliq_days
    from llc_main, llc_kont
    where lm_is_signed = 1 and
          lm_id = lk_from_lm_id and
          :rep_date_end between lk_date and lk_date_end and
          lk_from_k_id = :k_id
    into lk_balance, lk_calcperc, lk_percent_tmp, lk_deliq_days_tmp;

    if (k_disb_date < '28.02.2012') then
    begin
      /*
          select KCP_PERC_SALDO
          from kontcalcperc
          where kcp_from_k_id = :k_id
          and kcp_period = :rep_period
          into kcp_perc_saldo;
          kcp_perc_saldo = coalesce(kcp_perc_saldo,0);
          */

      -- из kontcalcperc данные тянуть не могу, т.к. для амнистированных займов в kontcalcperc нет записей.
      select sum(sh_calc_perc)
      from aarep_make_zad_list(:k_id, :rep_date_end, 1)
      where sh_date is not null
      into :kcp_perc_saldo;

      kcp_perc_saldo = coalesce(kcp_perc_saldo, 0);
    end
    else
      kcp_perc_saldo = coalesce(lk_calcperc, 0);

    lk_balance = coalesce(lk_balance, 0);
    lk_calcperc = coalesce(lk_calcperc, 0);
    lk_percent_tmp = coalesce(lk_percent_tmp, 0);
    lk_deliq_days_tmp = coalesce(lk_deliq_days_tmp, 0);

    if (k_is_devided = 0) then
    begin
      lk_deliq_days = lk_deliq_days_tmp;
      lk_percent = lk_percent_tmp;
    end

    --rppu_os = cast(lk_balance as double precision) * lk_percent / 100;
    --rppu_perc = cast(lk_calcperc as double precision) * lk_percent / 100;

    if (k_disb_date < '28.02.2012') then
    begin
      rppu_perc = 0;
      if (lk_balance = 0 and
          delinq_day = 0 and
          kcp_perc_saldo > 0 and
          rep_date_end > '30.09.2014') then
      begin
        sh_date = null;
        select sh_date
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_calc_perc > 0 and
              zad_perc = 0 and
              zad_os = 0 and
              sh_repay_type = 1 and
              sh_date is not null
        order by sh_date
        rows 1
        into sh_date;

        if (sh_date is not null) then
        begin
          delinq_day = :rep_date_end - :sh_date;
        end
      end
    end

    if (delinq_day < lk_deliq_days) then
      delinq_day = lk_deliq_days;

    rppu_os = cast(lk_balance as double precision); -- * lk_percent / 100;
    rppu_perc = cast(lk_calcperc as double precision); --* lk_percent / 100;

    if (k_disb_date < '28.02.2012') then
      rppu_perc = 0;

    select po_percent
    from get_percent_po(:rep_date_end, :delinq_day)
    into po_percent;

    po_summ = po_percent / 100 * (oks_balance + kcp_perc_saldo);

    oks_balance_tmp = oks_balance;
    kcp_perc_saldo_tmp = kcp_perc_saldo;
    k = 0;
    kol_cli = k_kolich_client_approved;
    for select kc_id, kc_from_c_id, kc_amount_approved, 1 - coalesce(kc_non_resident, 0)
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_id, c_id, kc_amount, cli_is_resident_on_disb
    do
    begin
      c_name_fio = 'n' || :c_id;
      dolya = cast(kc_amount as double precision) / k_amount_approved;

      select 1 - res_non_resident
      from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :rep_date_end)
      into :res_client_is_resident;

      period_day = max_sh_date - k_disb_date;

      if (k_is_devided = 1) then
      begin
        select percent_rvpz
        from hp_get_rvpz(:kc_amount, :period_day, :is_secured, :is_restruct, :lk_deliq_days)
        into lk_percent;
      end

      cli_balance = dolya * oks_balance;
      cli_perc_saldo = dolya * kcp_perc_saldo;
      /*
        k = k + 1;
        if (k < kol_cli) then
        begin
          cli_balance_round = round(cli_balance, 0);
          cli_perc_saldo_round = round(cli_perc_saldo, 0);

          oks_balance_tmp = oks_balance_tmp - cli_balance_round;
          kcp_perc_saldo_tmp = kcp_perc_saldo_tmp - cli_perc_saldo_round;
        end else
        begin
          cli_balance_round = oks_balance_tmp;
          cli_perc_saldo_round = kcp_perc_saldo_tmp;
        end
        */
      cli_balance_round = null;
      cli_perc_saldo_round = null;
      select oks_balance, oks_bal_perc_fixed, oks_percent_esp, oks_llr_os, oks_llr_perc, oks_ecl_percent
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into cli_balance_round, cli_perc_saldo_round, percent_effect, cli_rppu_os, cli_rppu_perc, oks_ecl_percent;

      if (cli_balance_round is null) then
        cli_balance_round = 0;
      if (cli_perc_saldo_round is null) then
        cli_perc_saldo_round = 0;

      --        if (k_disb_date < '28.02.2012') then  cli_rppu_os = cli_balance_round * oks_ecl_percent / 100;

      --        cli_rppu_os = dolya * rppu_os;
      --        cli_rppu_perc = dolya * rppu_perc;

      --        cli_rppu_os = cast(cli_rppu_os as double precision) * lk_percent / 100;
      --        cli_rppu_perc = cast(cli_rppu_perc as double precision) * lk_percent / 100;

      cli_po_summ = dolya * po_summ;

      cli_percent_schedule = dolya * percent_schedule;

      client_name_fio = null;
      c_birth_date = null;
      select c_name_fio, c_birth_date
      from clients c
      where c_id = :c_id
      into client_name_fio, c_birth_date;

      knr_non_resident = 1 - cli_is_resident_on_disb;
      select knr_non_resident
      from kn_resident
      where knr_from_kc_id = :kc_id and
            :rep_date_end between knr_date_begin and knr_date_end
      into knr_non_resident;

      acc_no = null;
      if (knr_non_resident = 1) then
        acc_no = '492';
      else
      if (knr_non_resident = 0) then
        acc_no = '488';

      suspend;
    end
  end
end
