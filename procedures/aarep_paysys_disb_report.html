CREATE PROCEDURE AAREP_PAYSYS_DISB_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SHOW_JUST_POLIST INTEGER)
RETURNS (
    REP_REGION_NAME VARCHAR(100),
    REP_KFOLDER_NO VARCHAR(100),
    REP_AMOUNT_APPROVED DECIMAL(18,4),
    REP_POLIS_NO VARCHAR(30),
    REP_POLIS_AMOUNT DECIMAL(18,4),
    REP_KONTRACT_PERIOD DECIMAL(18,4),
    REP_DISB_DATE DATE,
    REP_CLIENT_FIO VARCHAR(100),
    REP_LO_FIO VARCHAR(100),
    REP_OKRUG_NAME VARCHAR(100),
    REP_POLIS_BEGIN DATE,
    REP_POLIS_END DATE,
    REP_POLIS_STATUS VARCHAR(20),
    PSM_OPER_KIND INTEGER,
    REP_POLIS_KOLICH INTEGER,
    REP_DISB_CLIENT_KOLICH INTEGER,
    REP_PRODUCT_KIND SMALLINT)
AS
declare variable reg_id integer;
declare variable lo_id decimal(18,0);
declare variable reg_id_prior integer;
declare variable lo_id_prior decimal(18,0);
declare variable rep_k_id decimal(18,0);
declare variable kc_from_c_id decimal(18,0);
begin
  lo_id_prior = 0;
  reg_id_prior = 0;
--  REP_DISB_CLIENT_KOLICH=1;

  for select ks_from_k_id, reg_code || ' ' || reg_name, ks_region_id, ks_kc_id, ks_kfolder_no, kc_from_c_id,
             kc_amount_approved, k_period, ks_date_begin, c_name_fio, pr_product_kind
      from kontstate, region_list(:rep_region_id, 1, 3, 1), kontract, knclient, region, clients, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_region_id = id_list and
            kc_from_k_id = k_id and
            kc_amount_approved > 0 and
            ks_from_k_id = k_id and
            ks_region_id = reg_unicode and
            c_id = kc_from_c_id and
            k_from_pr_id = pr_id
      into :rep_k_id, :rep_region_name, :reg_id, :lo_id, rep_kfolder_no, kc_from_c_id, rep_amount_approved,
           :rep_kontract_period, :rep_disb_date, :rep_client_fio, :rep_product_kind
  do
  begin
    rep_disb_client_kolich = 1;
    if (lo_id <> lo_id_prior) then
    begin
      select c_name_fio
      from clients
      where c_id = :lo_id
      into :rep_lo_fio;
      lo_id_prior = lo_id;
    end

    if (reg_id <> reg_id_prior) then
    begin
      select aa_curr_level3_name
      from aarep_get_office_categ(:reg_id, :rep_date_end)
      into :rep_okrug_name;
      reg_id_prior = reg_id;
    end
    rep_polis_no = '';
    rep_polis_amount = 0;
    rep_polis_begin = null;
    rep_polis_end = null;
    rep_polis_kolich = 0;

    rep_polis_status = 'Без страхования';

    for select psi_code, psip_amount, psi_date_begin, psi_date_end, psm_oper_kind
        from paysys_main, paysys_list, paysys_item, paysys_item_props
        where psm_id = psl_from_psm_id and
              psl_object_kind = 3 and
              psl_object_id = :rep_k_id and
              psl_is_delete = 0 and
              psm_is_delete = 0 and
              psl_from_psi_id = psi_id and
              psip_from_psi_id = psi_id and
              psip_from_c_id = :kc_from_c_id
        into :rep_polis_no, :rep_polis_amount, :rep_polis_begin, rep_polis_end, psm_oper_kind
    do
    begin
      if (psm_oper_kind = 0) then
      begin
        rep_polis_status = 'Страхование';
        rep_polis_kolich = 1;
--        REP_DISB_CLIENT_KOLICH = 1;
        if (rep_polis_amount < 0) then
          rep_polis_amount = -rep_polis_amount;
        if (rep_amount_approved < 0) then
          rep_amount_approved = -rep_amount_approved;
      end
      else
      begin
        rep_polis_status = 'Испорчен';
        rep_polis_kolich = -1;
--        REP_DISB_CLIENT_KOLICH = -1;
        if (rep_polis_amount > 0) then
          rep_polis_amount = -rep_polis_amount;
        if (rep_amount_approved > 0) then
          rep_amount_approved = -rep_amount_approved;
      end
      if (rep_polis_no is null) then
        rep_polis_no = '';
      if (rep_polis_amount is null) then
        rep_polis_amount = 0;
      suspend;
      if (rep_disb_client_kolich = 1) then
        rep_disb_client_kolich = 0;
    end

    if (rep_polis_kolich = 0) then
    begin
      if (rep_show_just_polist = 0) then
        suspend;
    end
  end
end
