CREATE PROCEDURE HP_CONSULT_INSTALLMENT_REPAY(
    ESM_ID D_ID,
    K_ID D_ID,
    KC_ID D_ID,
    CONS_SUM D_CURRENCY)
RETURNS (
    RES_TEXT D_STR_LONG,
    PARENT_OL_ID D_ID)
AS
declare variable ol_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_fact_disb_date d_date;
declare variable go_office_id d_integer;
declare variable bo_office_id d_integer;
declare variable office_id d_integer;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable ol_date d_date;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_doc_kind d_integer;
declare variable ol_doc_client_name d_str_255;
declare variable ol_doc_no d_integer;
declare variable ol_is_client d_integer;
declare variable ol_status d_smallint;
declare variable ol_memo d_str_long;
declare variable ol_user_id d_id;
declare variable ob_summa d_currency;
declare variable ob_detail d_str_long;
declare variable al_go_dt d_str_20;
declare variable al_go_kt d_str_20;
declare variable al_bo_dt d_str_20;
declare variable al_bo_kt d_str_20;
declare variable al_dt d_str_20;
declare variable al_kt d_str_20;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable acc_short_dt d_str_20;
declare variable acc_short_kt d_str_20;
declare variable esm_project d_str_100;
declare variable esm_full_source d_str_long;
declare variable esp_memo d_str_255;
declare variable sh_date d_date;
declare variable sh_id d_integer;
declare variable sh_sum d_currency;
declare variable fix_sum d_currency;
declare variable paid_sum d_currency;
declare variable delta d_currency;
begin
  -- обрабатывает погашение консультационных услуг в рассрочку через сайт (RFCKG-4903)

  parent_ol_id = 0;
  res_text = '';

  go_office_id = 1090100; -- Бишкек ГО
  bo_office_id = 1060100; -- Бишкекский офис

  select k.k_kontract_no,
         k.k_fact_disb_date,
         kal.kal_new_reg_id
  from kontract k
  inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
  where k.k_id = :k_id and
        kal.is_delete = 0 and
        current_date between kal.kal_date and kal.kal_date_end
  into k_kontract_no,
       k_fact_disb_date,
       office_id;

  if (office_id is null) then
  begin
    res_text = 'Ошибка: не найден текущий офис по контракту ' || k_kontract_no;
    suspend;
    exit;
  end

  select c_id,
         c_name_fio
  from clients
  inner join knclient on kc_from_c_id = c_id
  where kc_id = :kc_id and
        kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into :c_id,
       :c_name_fio;

  if (c_id is null) then
  begin
    res_text = 'Ошибка: не найден клиент по контракту ' || k_kontract_no;
    suspend;
    exit;
  end

  -- Получение данных внешней системы
  esm_project = null;
  select esm.esm_project,
         esm.esm_date,
         esm.esm_full_source
  from extsysmain esm
  where esm.esm_id = :esm_id
  into :esm_project,
       :ol_date,
       :esm_full_source;

  if (esm_project is null) then
  begin
    res_text = 'Ошибка: не найден проект';
    suspend;
    exit;
  end

  esp_memo = '';
  select first 1 esp.esp_memo
  from extsyspro esp
  where esp_project_kind = 'payment' and
        esp_project = :esm_project and
        esp_status = 1
  into :esp_memo;

  ob_detail = 'Погашение консультационных услуг через ';

  -- Определение типа платежа (ipc/терминал/QR)
  if (esm_project = 'ipc') then
  begin
    acc_short_dt = '10003/211';
    ob_detail = ob_detail || 'IPC';
  end
  else if (esm_full_source like '%terminal%' or
      esm_full_source like '%терминал%' or
      esp_memo like '%терминал%') then
  begin
    acc_short_dt = '10003/138';
    ob_detail = ob_detail || 'терминал';
  end
  else
  begin
    acc_short_dt = '10024/111';
    ob_detail = ob_detail || 'QR';
  end

  -- 2) Проверка счетов

  -- 2.1 Счета для Бишкек ГО. Оплата приходит в ГО, поэтому DT - это локальный счет (QR или Терминал), KT - перевод в БО
  select al_id
  from find_acc_list(:acc_short_dt)
  where al_from_reg_id = :go_office_id
  into al_go_dt;

  select ln_account
  from loronostro
  where ln_region_loro = :go_office_id and
        ln_region_nostro = :bo_office_id
  into al_go_kt;

  if (al_go_dt is null or
      al_go_kt is null) then
  begin
    res_text = 'Не найден счет ' || acc_short_dt || ' или лоро/ностро для ГО';
    suspend;
    exit;
  end

  -- 2.2 Счета Бишкекского офиса (транзитный узел): DT - прием от ГО, KT - перевод в офис контракта
  select ln_account
  from loronostro
  where ln_region_loro = :bo_office_id and
        ln_region_nostro = :go_office_id
  into al_bo_dt;

  select ln_account
  from loronostro
  where ln_region_loro = :bo_office_id and
        ln_region_nostro = :office_id
  into al_bo_kt;

  if (al_bo_dt is null or
      al_bo_kt is null) then
  begin
    res_text = 'Не найден лоро/ностро для офиса ' || bo_office_id;
    suspend;
    exit;
  end

  -- 2.3 Счета для текущего офиса: DT - прием от БО, KT - итоговый счет (21114/10)
  select ln_account
  from loronostro
  where ln_region_loro = :office_id and
        ln_region_nostro = :bo_office_id
  into al_dt;

  acc_short_kt = '21114/10';
  select al_id
  from find_acc_list(:acc_short_kt)
  where al_from_reg_id = :office_id
  into al_kt;

  if (al_dt is null or
      al_kt is null) then
  begin
    res_text = 'Ошибка: не найден лоро/ностро или счет ' || acc_short_kt || ' для офиса ' || office_id;
    suspend;
    exit;
  end

  select al_id
  from acc_list
  where al_from_reg_id = :office_id and
        al_currency = '417' and
        al_account = '99912/1'
  into acc_dt;

  select al_id
  from acc_list
  where al_from_reg_id = :office_id and
        al_currency = '417' and
        al_account = '99911/1'
  into acc_kt;

  if (coalesce(acc_dt, '') = '' or
      coalesce(acc_kt, '') = '') then
  begin
    res_text = 'Ошибка: не найден забалансовый счет к получению оплаты консультационных услуг, офис ' || office_id;
    suspend;
    exit;
  end

  -- 3) Создание проводок (выполняется, только если все проверки пройдены)
  ol_doc_client_name = 'Оплата консультационных услуг';
  ol_memo = ol_doc_client_name || ' / ' || k_kontract_no || ' от ' || f_datetostr(k_fact_disb_date) || ' / ' || c_name_fio;
  ol_oper_type = 2; -- операция чисто бухгалтерская
  ol_oper_subtype = 11; -- мемориальный ордер
  ol_doc_kind = 9; -- мемориальный ордер
  ol_is_client = 1;
  ol_status = 1;
  ol_user_id = 1000000000000;
  ob_summa = cons_sum;

  select num
  from last_doc_number(:ol_date, 9, :go_office_id)
  into :ol_doc_no;

  -- создание родительской операции (Бишкек ГО)
  insert into operlist (ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no, ol_doc_no_str, ol_document_client, ol_is_client, ol_memo, ol_doc_client_name, ol_user_id, ol_date, ol_status)
  values (:go_office_id, :go_office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :ol_doc_no, :k_id, :c_id, :ol_is_client, :ol_memo, :ol_doc_client_name, :ol_user_id, :ol_date, :ol_status)
  returning ol_id
  into :parent_ol_id;

  ol_id = parent_ol_id;

  insert into operbook (ob_from_ol_id, ob_client_dt, ob_client_kt, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:parent_ol_id, null, null, :al_go_dt, :al_go_kt, :ob_summa, :ob_summa, :ol_date, :go_office_id, :ob_detail);

  -- создание дочерней операции (Бишкекский офис)
  insert into operlist (ol_parent_id, ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no, ol_doc_no_str, ol_document_client, ol_is_client, ol_memo, ol_doc_client_name, ol_user_id, ol_date, ol_status)
  values (:parent_ol_id, :bo_office_id, :bo_office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :ol_doc_no, :k_id, :c_id, :ol_is_client, :ol_memo, :ol_doc_client_name, :ol_user_id, :ol_date, :ol_status)
  returning ol_id
  into :ol_id;

  insert into operbook (ob_from_ol_id, ob_client_dt, ob_client_kt, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:ol_id, null, null, :al_bo_dt, :al_bo_kt, :ob_summa, :ob_summa, :ol_date, :bo_office_id, :ob_detail);

  -- создание дочерней операции (текущий офис контракта)
  insert into operlist (ol_parent_id, ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no, ol_doc_no_str, ol_document_client, ol_is_client, ol_memo, ol_doc_client_name, ol_user_id, ol_date, ol_status)
  values (:parent_ol_id, :office_id, :office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :ol_doc_no, :k_id, :c_id, :ol_is_client, :ol_memo, :ol_doc_client_name, :ol_user_id, :ol_date, :ol_status)
  returning ol_id
  into :ol_id;

  insert into operbook (ob_from_ol_id, ob_client_dt, ob_client_kt, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:ol_id, null, :c_id, :al_dt, :al_kt, :ob_summa, :ob_summa, :ol_date, :office_id, :ob_detail);

  -- 4) Фиксация платежа в operkontract
  insert into operkontract (ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
  values (:k_id, :ol_id, :cons_sum, f_get_rd_id('CashTransactionIncom.Consultation'));

  -- 5.1) -- создание дочерней операции (забалансовый учет, поступление погашения)
  ol_memo = ol_doc_client_name || ' / ' || f_datetostr(k_fact_disb_date) || ' / ' || c_name_fio;
  ob_detail = replace(ob_detail, k_kontract_no || ' от ', '');

  insert into operlist (ol_parent_id, ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no, ol_doc_no_str, ol_document_client, ol_is_client, ol_memo, ol_doc_client_name, ol_user_id, ol_date, ol_status)
  values (:parent_ol_id, :office_id, :office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :ol_doc_no, :k_id, :c_id, :ol_is_client, :ol_memo, :ol_doc_client_name, :ol_user_id, :ol_date, :ol_status)
  returning ol_id
  into :ol_id;

  insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_client_kt, ob_detail)
  values (:ol_id, :acc_dt, :acc_kt, :ob_summa, :ob_summa, :ol_date, :office_id, :c_id, :ob_detail);

  -- 5.2) Фиксация платежа в opercons
  for select oc_sh_id,
             oc_schedule_date,
             oc_schedule_principal
      from opercons
      where oc_from_k_id = :k_id and
            oc_ol_is_active = 1 and
            oc_operation_type_id = 1 -- график оплаты
      order by oc_schedule_date
      into sh_id,
           sh_date,
           sh_sum
  do
  begin
    fix_sum = 0; -- сумма платежа, фиксируемая в новой записи opercons
    paid_sum = 0; -- ранее оплаченная сумма

    select sum(oc_paid_principal)
    from opercons
    where oc_from_k_id = :k_id and
          oc_ol_is_active = 1 and
          oc_operation_type_id = 4 and
          oc_schedule_date = :sh_date
    into paid_sum;

    delta = coalesce(sh_sum, 0) + coalesce(paid_sum, 0); -- разница между суммой по графику и ранее оплаченной суммой (paid_sum равен нулю или отриц. числу)

    if (delta <= 0) then
      continue; -- платеж по графику закрыт
    else
    begin
      if (cons_sum < delta) then
      begin
        -- внесенная сумма меньше суммы платежа по графику
        fix_sum = cons_sum;
        cons_sum = 0;
      end
      else
      begin
        -- внесенная сумма больше или равно сумме платежа по графику
        fix_sum = delta;
        cons_sum = cons_sum - delta;
      end

      insert into opercons (oc_from_ol_id, oc_number, oc_from_k_id, oc_from_c_id, oc_sh_id, oc_schedule_date, oc_schedule_principal, oc_paid_principal, oc_operation_type_id, oc_ol_date, oc_ol_reg_id, oc_ol_is_active, ok_pay_source)
      values (:ol_id, -1, :k_id, :c_id, :sh_id, :sh_date, 0, -:fix_sum, 4, :ol_date, :office_id, 1, 3);
    end

    if (cons_sum = 0) then -- внесенной суммы не осталось для распределения
      leave;
  end

  suspend;
end
