CREATE PROCEDURE AAREP_PAYSYS_TURNOVER(
    OBJ_TYPE INTEGER,
    OBJ_ID ID_TYPE,
    DATE_BEGIN DATE,
    DATE_END DATE,
    TURNOVER_TYPE SMALLINT)
RETURNS (
    PSM_ID ID_TYPE,
    PSM_DATE DATE,
    OBJ_PSL_ID ID_TYPE,
    PSI_ID ID_TYPE,
    PSI_CODE VARCHAR(20),
    PAR_PSL_ID ID_TYPE,
    PAR_ID ID_TYPE,
    PAR_NAME VARCHAR(100),
    PAR_TYPE INTEGER,
    PAR_TYPE_NAME VARCHAR(15),
    OPER_NAME VARCHAR(15),
    CANCEL_DATE DATE,
    CANCEL_PSM_ID ID_TYPE,
    CANCEL_TEXT VARCHAR(15),
    PSL_SIGN SMALLINT)
AS
declare variable psm_oper_kind smallint;
begin
  if (turnover_type in (1, 0)) then
    psl_sign = 1;
  else
    psl_sign = -1;

  while (psl_sign > -2) do
  begin
    for select psm_id, psm_date, psm_oper_kind, psl_id, psi_id, psi_code
        from paysys_main, paysys_list, paysys_item
        where psm_id = psl_from_psm_id and
              psm_status = 1 and
              psl_object_kind = :obj_type and
              psl_object_id = :obj_id and
              psl_is_delete = 0 and
              psm_is_delete = 0 and
              psl_from_psi_id = psi_id and
              psm_date between :date_begin and :date_end and
              psl_sign = :psl_sign
        into psm_id, psm_date, psm_oper_kind, obj_psl_id, psi_id, psi_code
    do
    begin
      if (psm_oper_kind = 1) then --  Аннулирование
      begin
        cancel_date = psm_date;
        oper_name = 'Аннулирование';
        cancel_text = 'Аннулирован';
      end
      else -- Прием-передача
      begin
-- Ищем получателя в той же операции (т.к. это прием-передача)
-- Определяем тип и название получателя
        select psl_id, psl_object_kind, psl_object_id
        from paysys_list
        where psl_from_psm_id = :psm_id and
              psl_sign = :psl_sign * -1 and
              psl_is_delete = 0 and
              psl_from_psi_id = :psi_id
        into par_psl_id, par_type, par_id;
        par_type_name = 'Неизвестный тип';
        par_name = 'Не найден';
        if (par_type = 0) then
        begin
          par_type_name = 'Банк/Эмитент';
          select psp_name
          from paysys_project
          where psp_id = :par_id
          into par_name;
        end
        if (par_type = 1) then
        begin
          par_type_name = 'Офис';
          select reg_name
          from region
          where reg_unicode = :par_id
          into par_name;
        end
        if (par_type in (2, 5)) then
        begin
          if (par_type = 2) then
            par_type_name = 'Сотрудник';
          else
            par_type_name = 'Клиент';
          select c_name_fio
          from clients
          where c_id = :par_id
          into par_name;
        end
        if (par_type = 3) then
        begin
          par_type_name = 'Кр.Договор';
          select k_kontract_no
          from kontract
          where k_id = :par_id
          into par_name;

        end
        if (par_type = 4) then
        begin
          par_type_name = 'Документ/Проводка';
          select dn_name
          from operlist, docname
          where ol_id = :par_id and
                dn_id = ol_doc_kind
          into par_name;
        end
-- Название операции
        if (psl_sign = 1) then
        begin
          oper_name = 'Приход';
          if (par_type in (1, 2, 3) and
              obj_type in (3, 5)) then
            oper_name = 'Выдача';
          else
          if (par_type in (1, 2, 3) and
              obj_type = 4) then
            oper_name = 'Закрепление';
          if (par_type in (3, 4, 5) and
              obj_type in (1, 2, 3)) then
            oper_name = 'Возврат';
        end
        else
        begin
          oper_name = 'Передача';
          if (obj_type in (1, 2, 3) and
              par_type in (3, 5)) then
            oper_name = 'Выдача';
          else
          if (obj_type in (1, 2, 3) and
              par_type = 4) then
            oper_name = 'Закрепление';
          if (obj_type in (3, 4, 5) and
              par_type in (1, 2, 3)) then
            oper_name = 'Возврат';
        end

-- Было ли затем аннулировано, в случае, если закрепление/выдача
        if (psl_sign = -1 and
            par_type > 3) then
        begin
          cancel_psm_id = null;
          cancel_date = null;
          cancel_text = '';
          select psm_id, psm_date
          from paysys_main, paysys_list
          where psm_id > :psm_id and
                psl_from_psm_id = psm_id and
                psm_is_delete = 0 and
                psl_is_delete = 0 and
                psl_from_psi_id = :psi_id and
                psm_oper_kind = 1 and
                psm_status = 1
          into cancel_psm_id, cancel_date;
          if (cancel_psm_id is not null) then
            cancel_text = 'Аннулирован';
        end
      end
      suspend;
    end
    if (turnover_type = 0) then -- Если все обороты, показываем еще и расход
      psl_sign = psl_sign - 2;
    else -- Иначе выходим
      psl_sign = -2;
  end
end
