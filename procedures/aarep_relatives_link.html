CREATE PROCEDURE AAREP_RELATIVES_LINK(
    REP_DATE_END DATE,
    REP_C_ID DECIMAL(18,0))
RETURNS (
    RLL_ID INTEGER,
    REG_ID_OF_1 INTEGER,
    REG_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    JK_NAME VARCHAR(100),
    REL_FIO VARCHAR(100),
    REL_NAME VARCHAR(100),
    REL_JL_NAME VARCHAR(100),
    REL_JK_NAME VARCHAR(100),
    REL_OFFICE VARCHAR(100),
    REL_MEMO VARCHAR(255),
    REL_CODE VARCHAR(50),
    REL_KIND VARCHAR(100),
    RLL_CLIENT_1_STATUS INTEGER,
    RLL_CLIENT_2_STATUS INTEGER,
    RLL_LINK_STATUS INTEGER,
    RLL_LINK_STATUS_STR VARCHAR(50))
AS
declare variable i integer;
declare variable n integer;
declare variable ss varchar(100);
declare variable c1_tmp decimal(18,0);
declare variable c2_tmp decimal(18,0);
begin
-- ХП показывает все связи сотрудников и кандидатов

  if (rep_date_end is null) then
    rep_date_end = current_date;
  for select rll_id, rll_client_1, rll_client_2, rll_link_code, c1.c_name_fio, c2.c_name_fio, rd_name,
             rll_client_1_status, rll_client_2_status, rll_link_status
      from relationlink, clients c1, clients c2, refdata
      where rll_is_active = 1 and
            rll_client_1 = c1.c_id and
            rll_client_2 = c2.c_id and
            rll_link_kind = rd_id and
            (:rep_c_id = 0 or
            rll_client_1 = :rep_c_id)
      into :rll_id, :c1_tmp, :c2_tmp, :rel_code, c_name_fio, rel_fio, rel_kind, rll_client_1_status,
           rll_client_2_status, rll_link_status
  do
  begin
    reg_name = '';
    jl_name = 'На дату отчета не является сотрудником';
    jk_name = '';
    reg_id_of_1 = 1000000;
    rll_link_status_str = (case
                             when rll_link_status = -1 then 'Отказан'
                             when rll_link_status = 0 then 'Новая связь'
                             when rll_link_status = 1 then 'В ожидании'
                             when rll_link_status = 2 then 'Одорено'
                           end);

    select reg_unicode, reg_code || ' ' || reg_name, jl_name, jk_name
    from hr_pstatus, joblist, jobkind, region
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_reg_id = reg_unicode and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          ps_from_c_id = :c1_tmp
    into reg_id_of_1, reg_name, jl_name, jk_name;

    rel_office = '';
    rel_jl_name = 'На дату отчета не является сотрудником';
    rel_jk_name = '';
    select reg_code || ' ' || reg_name, jl_name, jk_name
    from hr_pstatus, joblist, jobkind, region
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_c_id = :c2_tmp and
          ps_is_working = 1 and
          ps_status = 1 and
          ps_from_reg_id = reg_unicode and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id
    into rel_office, rel_jl_name, rel_jk_name;

    rel_name = '';
    rel_memo = '';
    select rd_name
    from refdata
    where rd_str_id = :rel_code and
          rd_main_code = 'Relation'
    into :rel_name;
    if (rel_name = '') then
    begin
      rel_name = 'Дальний родственник';
      i = 0;
      n = char_length(rel_code);
      while (i < n) do
      begin
        i = i + 1;
        select rd_name
        from refdata
        where rd_str_id = substring(:rel_code from :i for 1) and
              rd_main_code = 'Relation'
        into :ss;
        if (rel_memo = '') then
          rel_memo = ss;
        else
          rel_memo = rel_memo || ' -> ' || ss;
      end
    end
    suspend;
  end

end
