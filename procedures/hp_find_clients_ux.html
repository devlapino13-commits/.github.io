CREATE PROCEDURE HP_FIND_CLIENTS_UX(
    FIND_TEXT D_STR_100,
    FIND_KIND D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE_STR D_STR_20,
    C_BIRTH_PLACE D_STR_100,
    C_PASS_NO D_STR_CODE,
    C_INN D_STR_CODE,
    C_STATUS D_STR_100,
    C_VERIFY_STATUS D_SMALLINT,
    C_PERS_STATUS D_SMALLINT,
    RES_PERS_STATUS_CAPTION D_STR_100,
    RES_PERS_STATUS_MEMO D_STR_255,
    RES_PERS_STATUS_COLOR D_INTEGER,
    RES_EDL_ID D_ID)
AS
declare variable c_birth_city d_str_100;
declare variable c_type d_smallint;
declare variable c_birth_date d_date;
declare variable perstemp_used d_smallint;
begin
  -- FIND_KIND:
  --  -1 - все контрагенты
  --   0 - Физ.лица
  --   1 - Юр.лица
  --   2 - Сотрудники

  if (un_id is null) then
    un_id = uxf$user();

  perstemp_used = f_tmp_const('use_perstemp');

  for select res_c_id
  from hp_find_clients_ux_tmp(:find_text, :find_kind, :un_id)
  group by res_c_id
  order by min(res_priority), res_c_id
  into :c_id
  do
  begin
    select c_name_fio, c_type, c_birth_date, c_verify_status,
           trim(coalesce(c_birth_country, '')), trim(coalesce(c_birth_city, '')),
           trim(coalesce(c_pass_no, '')), trim(coalesce(c_inn, ''))
    from clients
    where c_id = :c_id
    into c_name_fio, c_type, c_birth_date, c_verify_status, c_birth_place, c_birth_city,
         c_pass_no, c_inn;

    if (row_count = 0) then
      Continue;

    if (find_kind in (0, 1)) then
    begin
      if (find_kind <> c_type) then
        Continue;
    end else
    if (find_kind = 2) then
    begin
      if (c_type <> 0) then
        Continue;

      if (not exists(select * from hr_pstatus
      where ps_from_c_id = :c_id)) then
        Continue;
    end

    res_edl_id = null;
    c_pers_status = c_verify_status;

    if (perstemp_used = 1) then
    begin
      c_pers_status = f_pers_status(c_id, c_verify_status);

      if (c_pers_status <> 2) then
      begin
        select pte_edoc_for_approve
        from perstemp, edoc_list
        where pte_from_c_id = :c_id and
              pte_edoc_for_approve = edl_id and
              edl_status = 0 and
              edl_blank_status = 1
        into res_edl_id;
      end
    end

    select res_caption, res_memo, res_color_state
    from hp_perstemp_verify_info(:c_pers_status)
    into :res_pers_status_caption, :res_pers_status_memo, :res_pers_status_color;

    c_birth_date_str = '';
    if (c_birth_date is not null) then
    begin
      c_birth_date_str = right('0' || extract(day from c_birth_date), 2) || '.' ||
                         right('0' || extract(month from c_birth_date), 2) || '.' ||
                         left(extract(year from c_birth_date), 2) || '*' || right(extract(year from c_birth_date), 1);
    end

    if (char_length(c_pass_no) >= 5) then
      c_pass_no = left(c_pass_no, char_length(c_pass_no) - 5) || '***' || right(c_pass_no, 2);

    if (char_length(c_inn) >= 7) then
      c_inn = left(c_inn, char_length(c_inn) - 7) || '***' || right(c_inn, 4);

    if (c_birth_city <> '') then
    begin
      if (c_birth_place = '') then
        c_birth_place = c_birth_city;
      else
        c_birth_place = left(c_birth_place || ', ' || c_birth_city, 100);
    end

    c_status = '';
    select c_status
    from hp_find_client_status(:c_id)
    into :c_status;

    suspend;
  end
end
