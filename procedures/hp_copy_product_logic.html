CREATE PROCEDURE HP_COPY_PRODUCT_LOGIC(
    PRODUCT_ID_ORIG INTEGER,
    PRODUCT_ID_NEW INTEGER,
    PRL_DATE_BEGIN_ORIG DATE,
    PRL_DATE_END_ORIG DATE,
    PRL_DATE_BEGIN_NEW DATE,
    PRL_DATE_END_NEW DATE)
AS
begin
  insert into prodlogic (prl_from_pr_id, prl_id, prl_date_begin, prl_date_end, prl_circle_min, prl_circle_max,
                         prl_kolich_min, prl_kolich_max, prl_amount_gr_min, prl_amount_gr_max, prl_amount_cl_min,
                         prl_amount_cl_max, prl_repay_min, prl_repay_max, prl_percent_min, prl_percent_max,
                         prl_admin_perc_min, prl_admin_perc_max, prl_min_max_diff, prl_male_max_perc)
  select :product_id_new, -1, :prl_date_begin_new, :prl_date_end_new, prl_circle_min, prl_circle_max, prl_kolich_min,
         prl_kolich_max, prl_amount_gr_min, prl_amount_gr_max, prl_amount_cl_min, prl_amount_cl_max, prl_repay_min,
         prl_repay_max, prl_percent_min, prl_percent_max, prl_admin_perc_min, prl_admin_perc_max, prl_min_max_diff,
         prl_male_max_perc
  from prodlogic
  where prl_from_pr_id = :product_id_orig and
        prl_date_begin = :prl_date_begin_orig and
        prl_date_end = :prl_date_end_orig;

  if (product_id_orig = product_id_new) then
  begin
    update prodlogic
    set prl_date_end = (:prl_date_begin_new - 1)
    where prl_from_pr_id = :product_id_orig and
          prl_date_begin = :prl_date_begin_orig and
          prl_date_end = :prl_date_end_orig;
  end
end
