CREATE PROCEDURE HP_EDOC_MANUAL_SIGN_LIST(
    REP_REGION_ID D_INTEGER,
    LIST_KIND D_STR_CODE,
    CURR_EDB_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    EDB_ID D_ID,
    EDL_BLANK_DATE D_DATE,
    EDF_PRINT_NAME D_STR_100,
    EDC_NAME D_STR_255,
    REG_NAME D_STR_100,
    EDL_OBJECT_NAME D_STR_255,
    EDS_CLIENT_LIST D_STR_4096,
    EDL_OBJECT_KIND D_INTEGER,
    EDL_OBJECT_ID D_ID,
    EDBS_STATUS D_SMALLINT,
    EDBS_KIND D_SMALLINT,
    EDB_FROM_BLS_ID D_ID,
    EDBS_FROM_BLS_ID D_ID,
    EDBS_FROM_UN_ID D_ID,
    UN_FROM_C_ID D_ID,
    EDBS_STATUS_STR D_STR_CODE,
    EDB_RESPONSE_FIO D_STR_100,
    EDB_BLANK_JSON D_STR_4096,
    EDBS_SIGNED_JSON D_STR_4096,
    EDBS_FROM_RD_ID D_INTEGER,
    EDBS_FROM_RD_NAME D_STR_255)
AS
declare variable REG_ID D_INTEGER;
declare variable UN_NAME D_STR_CODE;
declare variable DB_ID D_INTEGER;
declare variable KOBD_OFFICE D_INTEGER;
begin
  if (trim(coalesce(list_kind, '')) = '') then
    exception e_default 'Empty list_kind';

  if (list_kind not in ('scan', 'check')) then
    exception e_default 'Unknown list_kind: ' || list_kind;

  if (curr_edb_id is null) then
    curr_edb_id = 0;

  db_id = f_get_db_id();

  if (db_id = 81001) then
  begin
    if (exists(select pm_object_id
               from permissions, rolename
               where pm_object_kind = (select ukt_id
                                       from hp_get_uniqtable_id('Username')) and
                     pm_object_id = :un_id and
                     pm_subject_kind = (select ukt_id
                                        from hp_get_uniqtable_id('Rolename')) and
                     pm_is_delete = 0 and
                     current_date between coalesce(pm_date_begin, '01.01.2000') and coalesce(pm_date_end, '31.12.2999') and
                     pm_subject_id = rn_id and
                     rn_name in ('KRED_MANAGER', 'OPERATOR'))) then
    begin

      kobd_office = null;

      select ps_from_reg_id
      from hr_pstatus
      inner join joblist on ps_from_jl_id = jl_id
      inner join jobkind on jk_id = jl_kind
      inner join username on ps_from_c_id = un_from_c_id
      where current_date between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            un_id = :un_id
      into kobd_office;
      --RFCRU-4093
      if (kobd_office is not null and :rep_region_id = 0) then
        rep_region_id = kobd_office;
    end
  end

  for select id_list, reg_code || ' ' || reg_name
      from region, region_list(:rep_region_id, 1, 3, 1)
      where id_list = reg_unicode
      into reg_id, reg_name
  do
  begin
    for select edbp_from_edb_id, edbs_status, edbs_kind, edbs_from_bls_id, edbs_from_un_id, un_from_c_id, un_name, edbs_from_rd_id
        from edoc_blank_pending, edoc_blank_scan, edoc_blank, username, edoc_cfg, edoc_list
        where edbp_from_edb_id = edbs_from_edb_id and
              edbs_from_un_id = un_id and
              edbs_from_edb_id = edb_id and
              edb_from_edl_id = edl_id and
              edl_from_edc_id = edc_id and
              edc_state <> 0 and
              (edb_from_bls_id is not null or (edb_blob is not null)) and
              edbp_from_reg_id = :reg_id and
              (:curr_edb_id = 0 or
              edbp_from_edb_id = :curr_edb_id) and
              (:curr_edb_id = 0 and
              ((:list_kind = 'scan' and
              edbs_status in (-1, 0) and
              edbs_kind in (0, 1)) or
              (:list_kind = 'check' and
              ((edbs_status = 1 and
              edbs_from_bls_id is not null) or
              (edbs_status = 0 and
              edbs_kind = 2)))))
        into edb_id, edbs_status, edbs_kind, edbs_from_bls_id, edbs_from_un_id, un_from_c_id, un_name, edbs_from_rd_id
    do
    begin
      edb_response_fio = '';
      select c_name_fio
      from clients
      where c_id = :un_from_c_id
      into edb_response_fio;

      edbs_from_rd_name = '';
      if (edbs_from_rd_id is not null) then
      begin
        select rd_name
        from refdata
        where rd_id = :edbs_from_rd_id
        into edbs_from_rd_name;
      end

      if (edb_response_fio = '') then
        edb_response_fio = un_name;

      select edl_id, edl_blank_date, edf_print_name, edc_name, edl_object_kind, edl_object_id, edb_from_bls_id
      from edoc_blank, edoc_list, edoc_cfg, edoc_frx
      where edb_id = :edb_id and
            edb_from_edl_id = edl_id and
            edl_from_edc_id = edc_id and
            edb_from_edf_id = edf_id and
            (edb_from_bls_id is not null or edb_blob is not null)
      into edl_id, edl_blank_date, edf_print_name, edc_name, edl_object_kind, edl_object_id, edb_from_bls_id;

      edbs_status_str = trim(case edbs_status
                               when -1 then 'Отказ'
                               when 0 then 'Нет скана'
                               when 1 then 'На проверке'
                               when 2 then 'Одобрен'
                             end);

      if (edbs_status = 0 and
          edbs_from_bls_id is not null) then
        edbs_status_str = 'Прикреплен';

      eds_client_list = '';
      edl_object_name = '';

      select doc_name
      from hp_edoc_object_info(:edl_object_kind, :edl_object_id, :un_id, :edl_blank_date)
      into edl_object_name;

      select list(c_name_fio)
      from edoc_sign, clients
      where eds_from_edl_id = :edl_id and
            eds_from_c_id = c_id
      into eds_client_list;

      edb_blank_json = '';
      if (edb_from_bls_id is not null)
      then edb_blank_json = f_format('[{"caption": "%1", "blobs":[%2]}]', edl_object_name, f_image_item('blobstore', edb_from_bls_id, edf_print_name, un_id));
      else edb_blank_json = f_format('[{"caption": "%1", "blobs":[%2]}]', edl_object_name, f_image_item('edoc_blank', edb_id, edf_print_name, un_id));

      edbs_signed_json = '';
      if (edbs_from_bls_id is not null) then
        edbs_signed_json = f_format('[{"caption": "%1", "blobs":[%2]}]', edl_object_name, f_image_item('blobstore', edbs_from_bls_id, edf_print_name, un_id));

      if (db_id <> 81001) then
        suspend;

      if (db_id = 81001 and
          (kobd_office is null or (kobd_office is not null and
          (edc_name = 'Изменение номера' or reg_id = kobd_office) and
          reg_id <> 1010100))) then
        suspend;
    end
  end
end
