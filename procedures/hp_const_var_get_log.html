CREATE PROCEDURE HP_CONST_VAR_GET_LOG(
    CV_CODE D_STR_CODE)
RETURNS (
    OLD_VALUE D_STR_CODE,
    NEW_VALUE D_STR_CODE,
    CVL_DATETIME D_DATETIME,
    USER_NAME D_STR_255)
AS
begin
  select cv_value
  from const_var
  where cv_code = :cv_code
  into new_value;

  for select cvl_value, cvl_datetime, un_fio
  from const_var_log
  left join username on un_id = cvl_from_un_id
  where cvl_code = :cv_code
  order by cvl_datetime desc
  into old_value, cvl_datetime, user_name
  do
  begin
    suspend;

    new_value = old_value;
  end 
end
