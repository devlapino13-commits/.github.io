CREATE PROCEDURE RUS_STANDART_CARDS_FOR_DISB
RETURNS (
    TARGET STR_100,
    OPERATION STR_100,
    ESM_ID ID_TYPE,
    SEX SMALLINT,
    BIRTHDAY VARCHAR(10),
    CLIENTNAME STR_100,
    CLIENTSURNAME VARCHAR(100),
    CLIENTLASTNAME STR_100,
    BORNPLACE STR_100,
    BORN_COUNTRY INTEGER,
    CITIZENSHIP INTEGER,
    CHANGEFIO SMALLINT,
    PREVIOUSNAME STR_100,
    INN STR_100,
    IDRESIDENT INTEGER,
    ISANOTHERGRAJD INTEGER,
    AGREEPERSDATA SMALLINT,
    MIGRATORY_NUMBER STR_100,
    MIGRATORY_EXP STR_100,
    MIGRATORY_FROM STR_100,
    RESIDENCE_ISSUE STR_100,
    RESIDENCE_ISSUE_EXP STR_100,
    RESIDENCE_ISSUE_FROM STR_100,
    RESIDENCE_NUMBER STR_100,
    RESIDENCE_SERIES STR_100,
    DOCGIVEDATE STR_100,
    DOCGIVE STR_100,
    DEPCODE STR_100,
    DOCSERIES STR_100,
    DOCNUM STR_100,
    COUNTRY_REG VARCHAR(100),
    POSTINDEX_REG VARCHAR(100),
    REGION_REG VARCHAR(100),
    DISTRICT_REG VARCHAR(100),
    CITY_REG VARCHAR(100),
    SETTLEMENT_REG VARCHAR(100),
    STREET_REG VARCHAR(100),
    STREET_REG_EXT VARCHAR(100),
    HOUSENUMBER_REG VARCHAR(100),
    HOUSECASE_REG VARCHAR(100),
    APPARTMENT_REG VARCHAR(100),
    IDTIMEREG INTEGER,
    COUNTRY_POST VARCHAR(100),
    POSTINDEX_POST VARCHAR(100),
    REGION_POST VARCHAR(100),
    DISTRICT_POST VARCHAR(100),
    CITY_POST VARCHAR(100),
    SETTLEMENT_POST VARCHAR(100),
    STREET_POST VARCHAR(100),
    STREET_POST_EXT VARCHAR(100),
    HOUSENUMBER_POST VARCHAR(100),
    HOUSECASE_POST VARCHAR(100),
    APPARTMENT_POST VARCHAR(100),
    FOREIGNADDRESS_POST VARCHAR(100),
    MOBILPHONE VARCHAR(100),
    POFDECLARE VARCHAR(100),
    POF_FIO VARCHAR(100),
    ISPUBLICOFFICER VARCHAR(100),
    CODEWORD VARCHAR(100),
    IDCARD VARCHAR(100),
    CURRENCY VARCHAR(100),
    AMOUNT VARCHAR(100),
    TRANSACTIONID ID_TYPE)
AS
declare variable c_id id_type;
declare variable c_name_fio str_100;
declare variable kk integer;
declare variable c_citizen integer;
declare variable c_birth_country str_100;
declare variable k_disb_date date;
declare variable pdt_code str_100;
declare variable cds_doc_issued_by str_255;
declare variable cds_doc_no str_255;
declare variable doc_date_begin str_10;
declare variable doc_date_end str_10;
declare variable c_pass_no str_100;
declare variable adl_from_pl_id id_type;
declare variable psi_id id_type;
declare variable esm_idcard str_100;
begin
  target = 'client_system'; -- из API Должно равняться значению client_system
  operation = 'AddRevPrePaid';
  changefio = 0;
  previousname = '';
  isanothergrajd = 0;
  agreepersdata = 1;
  foreignaddress_post = ''; --Не заполняется,  так как у наших клиентах у всех есть РФ адрес
  pofdeclare = 0; --Всегда 0 (непубличное лицо и не родственник)
  pof_fio = ''; --ВСегда пусто
  ispublicofficer = 0; --Всегда 0 (непубличное лицо)
  currency = 3810; --Валюта пополнения. Всегда значение 3810.
  idtimereg = 0;

  for select esm_id, cl.c_id,

             cl.c_name_fio, iif(cl.c_sex = 1, 1, 2), f_datetostr(cl.c_birth_date),

             cl.c_birth_country, cl.c_birth_city, cl.c_citizen,

             coalesce(cl.c_inn, ''), kont.k_disb_date, cl.c_pass_no,

             f_datetostr(c_pass_date), coalesce(cl.c_pass_who, ''), coalesce(cl.c_pass_who_code, ''),

             iif(kc_non_resident = 0, 1, 2), coalesce(em.esm_account, ''), em.esm_trans_id,

             em.esm_amount
      from extsysmain em
      inner join clients cl on cl.c_id = em.esm_object_id
      inner join kontract kont on kont.k_id = em.esm_from_k_id
      inner join knclient kncl on kncl.kc_from_k_id = kont.k_id and
            kncl.kc_from_c_id = cl.c_id and
            kncl.kc_amount_approved > 0
      where em.esm_project = 'rus_standart' and
            em.esm_action = 'CardDisb' and
            em.esm_status = 0
      into :esm_id, :c_id, :c_name_fio, :sex, :birthday, :c_birth_country, :bornplace, :c_citizen, :inn, :k_disb_date,
           :c_pass_no, :docgivedate, :docgive, :depcode, :idresident, :esm_idcard, :transactionid, :amount
  do
  begin
    for select psi_id
        from hp_paysys_get_cl_info(:c_id, 6) -- 6 - Русский Стандарт
        order by psm_date desc
        rows 1
        into :psi_id
    do
    begin
      select pys.psi_code, pys.psi_item_status
      from paysys_item pys
      where pys.psi_id = :psi_id
      into :idcard, :codeword;
    end

    if (esm_idcard = idcard) then
    begin
      born_country = (case c_birth_country
                        when 'Кыргызстан' then 115 --из Grajded
                        when 'Узбекистан' then 227
                        when 'Таджикистан' then 211
                        when 'Россия' then 1
                        else 0 --Нет данных
                      end);

      citizenship = (case c_citizen
                       when 8 then 115 --'Кыргызстан' из нашей БД REFDATA из Grajded
                       when 9 then 227 --'Узбекистан'
                       when 12 then 211 --'Таджикистан'
                       when 13 then 1 --'Россия'
                       else 0 --Нет данных
                     end);

      migratory_number = '';
      migratory_exp = null;
      migratory_from = null;
      residence_issue = null;
      residence_issue_exp = null;
      residence_issue_from = null;
      residence_number = null;

      if (citizenship <> 1) then -- если не гражданин рф
      begin
        for select pdt_code, cds.cds_doc_no, cds.cds_doc_issued_by,
                   f_datetostr(coalesce(cds_doc_date_begin, '01.01.01')),
                   f_datetostr(coalesce(cds_doc_date_end, '31.12.1999'))
            from clientdocs cds, passkind, passdoctype
            where cds_from_c_id = :c_id and
                  cds_is_delete = 0 and
                  cast(:k_disb_date as date) between coalesce(cds_doc_date_begin, '01.01.01') and coalesce(cds_doc_date_end, '31.12.9999') and
                  coalesce(cds.cds_doc_blob_1, 0) > 0 and
                  cds_from_pk_id = pk_id and
                  pk_from_pdt_id = pdt_id
            into pdt_code, cds_doc_no, cds_doc_issued_by, doc_date_begin, doc_date_end
        do
        begin
          if (pdt_code = 'magration_card') then
          begin
            migratory_number = cds_doc_no;
            migratory_from = doc_date_begin;
            migratory_exp = doc_date_end;
          end

          if (pdt_code in ('registration_rvp', 'registration_vnj')) then
          begin
            residence_series = '';
            residence_number = cds_doc_no;
            residence_issue = cds_doc_issued_by;
            residence_issue_from = doc_date_begin;
            residence_issue_exp = doc_date_end;
          end

          adl_from_pl_id = null;
          postindex_reg = null;
          housenumber_reg = null;
          housecase_reg = null;
          appartment_reg = null;
          street_reg = null;
          street_reg_ext = null;
          region_reg = null;
          district_reg = null;
          city_reg = null;
          settlement_reg = null;

          select pl_id, adl_post_index, region_name, district, city, street, street_plt, adl_depart_no, adl_section_no,
                 adl_flat_no, locality
          from hp_get_client_address(:c_id, 1)
          into :adl_from_pl_id, :postindex_reg, :region_reg, :district_reg, :city_reg, :street_reg, :street_reg_ext,
               :housenumber_reg, :housecase_reg, :appartment_reg, :settlement_reg;

          country_reg = (case left(adl_from_pl_id, 3) --adl_from_pl_id начинаются на валюту страны
                           when 810 then 1 --'Россия' из Grajded
                           when 417 then 115 --'Кыргызстан'
                           when 860 then 227 --'Узбекистан'
                           when 972 then 211 --'Таджикистан'
                           else 0 --Нет данных
                         end);

          adl_from_pl_id = null;
          postindex_post = null;
          housenumber_post = null;
          housecase_post = null;
          appartment_post = null;
          street_post = null;
          street_post_ext = null;

          select al.adl_from_pl_id, al.adl_post_index, al.adl_depart_no, adl_section_no, adl_flat_no, pl.pl_name,
                 plt.plt_name
          from addrclient ac
          inner join addrlist al on adc_from_adl_id = adl_id
          left join places pl on al.adl_street_id = pl.pl_id
          left join placetype plt on pl.pl_status = plt.plt_id
          where adc_from_c_id = :c_id and
                adc_address_type = 2 --почтового адреса понимаю как адрес по факту место жительства
          into adl_from_pl_id, postindex_post, housenumber_post, housecase_post, appartment_post, street_post,
               street_post_ext;

          country_post = (case left(adl_from_pl_id, 3) --adl_from_pl_id начинаются на валюту страны
                            when 810 then 1 --'россия' из grajded
                            when 417 then 115 --'кыргызстан'
                            when 860 then 227 --'узбекистан'
                            when 972 then 211 --'таджикистан'
                            else 0 --нет данных
                          end);

        end
      end

      docseries = substring(c_pass_no from 1 for 2);
      docnum = substring(c_pass_no from 3);

      clientname = '';
      clientsurname = '';
      clientlastname = '';

      select txt1, txt2, txt3
      from hp_fio_clear(:c_name_fio)
      into :clientsurname, :clientname, :clientlastname;

      if (c_name_fio like '%уулу%') then
          clientsurname = clientsurname || ' уулу';
      if (c_name_fio like '%кызы%') then
          clientsurname = clientsurname || ' кызы';

      clientsurname = left(clientsurname, 1) || lower(substring(clientsurname from 2 for 100));
      clientname = left(clientname, 1) || lower(substring(clientname from 2 for 100));
      clientlastname = left(clientlastname, 1) || lower(substring(clientlastname from 2 for 100));

      mobilphone = f_cont_info('clients', c_id, 'cell_main', 'ok');
      if (mobilphone starting '7')
      then mobilphone = right(mobilphone, 10);
      else mobilphone = '';

      /*Только для теста */
      --idcard = '2000298952164';

      residence_issue_from = f_datetostr(coalesce(residence_issue_from, current_date));
      residence_issue_exp = f_datetostr(coalesce(residence_issue_exp, current_date));

      suspend;
    end
  end
end
