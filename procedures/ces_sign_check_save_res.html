CREATE PROCEDURE CES_SIGN_CHECK_SAVE_RES(
    CSC_ID D_ID,
    REQUEST_JSON D_STR_8100,
    ANSWER_JSON D_STR_8100)
AS
declare variable ESM_ID D_ID;
declare variable EDS_ID D_ID;
declare variable EDL_ID D_ID;
declare variable C_ID D_ID;
begin
  select csc.csc_from_eds_id, edl.edl_id, eds.eds_from_c_id
  from ces_sign_check csc
  inner join edoc_sign eds on eds.eds_id = csc.csc_from_eds_id
  inner join edoc_list edl on edl.edl_id = eds.eds_from_edl_id
  where csc.csc_id = :csc_id
  and csc.csc_status = 0
  into eds_id, edl_id, c_id;

  insert into extsysmain (esm_trans_id, esm_project, esm_action, esm_date, esm_account, esm_object_kind, esm_object_id, esm_datetime,
                          esm_status, esm_full_source)
  values (:eds_id, 'ces', 'sign_check', current_date, :edl_id, 0, :c_id, current_timestamp, 1, :request_json)
  returning esm_id
  into esm_id;

  insert into extsysanswer (esa_from_esm_id, esa_message)
  values (:esm_id, :answer_json);

  -- RFCKG-3160
  if (lower(:answer_json) like '%ошибка%') then
    update ces_sign_check
    set csc_status = -1,
        csc_from_esm_id = :esm_id
    where csc_id = :csc_id;
  else
    update ces_sign_check
    set csc_status = 1,
        csc_from_esm_id = :esm_id
    where csc_id = :csc_id;  

end
