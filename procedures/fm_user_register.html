CREATE PROCEDURE FM_USER_REGISTER(
    UN_ID D_ID,
    GOOGLE_ID D_STR_255,
    DEV_ID D_STR_255)
AS
begin
  -- Процедура обновляет Google_ID для получения увдомлений
  if (un_id is null) then
    exit;

  if (google_id = '-') then
    google_id = '';

  update username
  set un_google_id = coalesce(:google_id, '')
  where un_id = :un_id and
        un_google_id is distinct from :google_id;

  -- Логирование идентификатора устройства
  if (dev_id = '') then
    dev_id = null;

  update or insert into usernameid (unid_from_un_id, unid_dev_id)
  values (:un_id, char_to_uuid(:dev_id))
  matching (unid_from_un_id, unid_dev_id);
end
