CREATE PROCEDURE HP_EFFECT_RATE_RESCHEDULE(
    K_ID D_ID,
    OL_DATE D_DATE,
    CML_ID D_ID)
RETURNS (
    EFFECT_RATE D_PERCENT,
    EFFECT_SUM D_CURRENCY)
AS
declare variable IPF_COMMISSION D_CURRENCY;
declare variable SH_DATE_PREV D_DATE;
declare variable REC_ID D_INTEGER;
declare variable SH_ID D_INTEGER;
declare variable SH_DAYS D_INTEGER;
declare variable SH_DATE D_DATE;
declare variable SH_ALL D_CURRENCY;
declare variable SH_PERC D_CURRENCY;
begin
  delete from temp_schedule
  where 1 = 1;

  ipf_commission = 0;

  rec_id = 0;
  select coalesce(pr.pr_ipf_commission, 0)
  from product pr
  inner join kontract kont on kont.k_from_pr_id = pr.pr_id
  where kont.k_id = :k_id
  into :ipf_commission;

  sh_id = 0;
  sh_days = 0;
  select min(sh_date), -sum(sh_os - sh_kommis) + :ipf_commission
  from schedule
  where sh_from_k_id = :k_id
  into sh_date, sh_all;
  rec_id = rec_id + 1;

  insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
  values (:sh_id, :sh_date, 0, :sh_all);

  sh_date_prev = sh_date;
  effect_sum = 0;
  for with tab as (
  select minvalue(ok_sh_date, ok_ol_date) sh_date, -sum(ok_os + ok_perc) as sh_all,
         -sum(ok_perc + ok_kommis) sh_perc
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type between 4 and 6 and
        ok_ol_date <= :ol_date
  group by 1
  having sum(ok_os + ok_perc) <> 0
  union all
  select cmr_sh_date, sum(cmr_sh_os + cmr_sh_perc),
         sum(cmr_sh_perc)
  from com_list_resh
  where cmr_from_cml_id = :cml_id
  group by 1)
  select sh_date, sum(sh_all), sum(sh_perc)
  from tab
  group by 1
  order by 1
  into sh_date, sh_all, sh_perc
  do
  begin
    sh_id = sh_id + 1;
    sh_days = sh_date - sh_date_prev;
    rec_id = rec_id + 1;

    effect_sum = effect_sum + sh_perc;

    insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
    values (:sh_id, :sh_date, :sh_days, :sh_all);

    sh_date_prev = sh_date;
  end

  select effect_rate
  from hp_effect_rate_common(:k_id, null, 2)
  into :effect_rate;

  suspend;
end
