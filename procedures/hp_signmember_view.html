CREATE PROCEDURE HP_SIGNMEMBER_VIEW(
    ID D_INTEGER)
RETURNS (
    SM_ID D_INTEGER,
    SM_SGN_ID D_INTEGER,
    SM_KIND D_SMALLINT,
    KINDSTR D_STR_CODE,
    SM_OBJ_ID D_ID,
    OBJ_NAME D_STR_100)
AS
begin
  for
    select
      sm_id,
      sm_sgn_id,
      sm_kind,
      case sm_kind
        when 0 then 'сотрудник'
        when 1 then 'должность'
        when 2 then 'отдел'
        when 3 then 'регион'
      end,
      sm_obj_id
    from signmember
    where sm_sgn_id = :id
  into
    :sm_id,
    :sm_sgn_id,
    :sm_kind,
    :kindstr,
    :sm_obj_id
  do
  begin
    obj_name =
      case sm_kind
        when 0 then
          (select c_name_fio from clients where c_id = :sm_obj_id)
        when 1 then
          (select jl_name from joblist where jl_id = :sm_obj_id)
        when 2 then
          (select dep_name from depart where dep_id = :sm_obj_id)
        when 3 then
          (select reg_name from region where reg_unicode = :sm_obj_id)
      end;
    suspend;
  end
end
