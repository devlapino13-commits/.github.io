CREATE PROCEDURE HP_DO_RESCHEDULE_KZ_DYNAMIC(
    K_ID D_ID,
    REP_DATE D_DATE,
    PERC_RATE_LGOTA D_PERCENT,
    DEFER_MONTH_COUNT D_INTEGER = 2,
    PAYMENT_REDUCE_PERCENT D_PERCENT = 10)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255,
    DELINQ_START_DATE D_DATE,
    SH_DATE D_DATE,
    FINE CURR_TYPE,
    MONTH_ADD D_INTEGER)
AS
declare variable REG_ID D_ID;
declare variable K_KONTRACT_NO D_STR_100;
declare variable K_DISB_DATE D_DATE;
declare variable CONTROL_AMOUNT CURR_TYPE;
declare variable OL_ID D_ID;
declare variable DELINQ_MONTH D_INTEGER;
declare variable MONTHLY_PAYMENT CURR_TYPE;
declare variable MONTHLY_PAYMENT_BASE CURR_TYPE;
declare variable CONTROL_AMOUNT_NEW CURR_TYPE;
declare variable C_ID D_ID;
declare variable K_PERIOD D_INTEGER;
declare variable FINE_ACCOUNT D_STR_20;
declare variable REPAY_ACCOUNT D_STR_20;
declare variable ADD_PARAMS D_STR_LONG;
declare variable NON_DELINQ_MONTH D_INTEGER;
declare variable K_STOP_PERC D_DATE;
declare variable ADD_MONTH_90 D_INTEGER;
declare variable LAST_MONTH D_DATE;
declare variable OL_MONTH D_INTEGER;
declare variable SH_MONTH D_DATE;
declare variable EFFECT_RATE_COMMON d_percent;
declare variable PRIMARY_PERC_RATE d_percent;
declare variable TARGET_LGOTA_PERC_RATE d_currency;
declare variable ADD_MONTH_KIND d_integer;
declare variable AUTO_MONTH_ADD d_integer;
declare variable AUTO_ITER d_integer;
declare variable AUTO_ITER_MAX d_integer;
declare variable LAST_PAYMENT_TOTAL curr_type;
declare variable NEW_PAY_DAY d_integer;
declare variable NEED_MORE_MONTH d_bool;
declare variable CORRECTION_APPLIED d_bool;
begin
  res_is_error = 0;
  res_text = '';
  fine = 0;
  month_add = 0;

  if (coalesce(k_id, 0) = 0) then
    exception e_message 'Контракт не указан';

  if (rep_date is null) then
    exception e_message 'Дата реструктуризации не указана';

  select k.k_fact_disb_reg_id,
         k.k_kontract_no,
         k.k_disb_date,
         k.k_period,
         k.k_stop_perc,
         k.k_repay_day,
         kc.kc_from_c_id
  from kontract k
  inner join knclient kc on kc.kc_from_k_id = k.k_id and
        kc.kc_amount_approved > 0
  where k.k_id = :k_id
  into reg_id, k_kontract_no, k_disb_date, k_period, k_stop_perc, new_pay_day, c_id;

  if (coalesce(new_pay_day, 0) = 0) then
    new_pay_day = extract(day from k_disb_date);

  select oks.oks_delinq_date
  from oksummary oks
  where oks.oks_from_k_id = :k_id and
        oks.oks_close_state = 0 and
        :rep_date between oks.oks_date_begin and oks.oks_date_end
  into :delinq_start_date;

  select sum(sh_calc_fine + sh_calc_perc + zad_os),
         sum(sh_calc_fine)
  from aarep_make_zad_list(:k_id, :rep_date, 1)
  into control_amount, fine; -- общая сумма задолженности и пеня

  control_amount = coalesce(control_amount, 0);
  fine = coalesce(fine, 0);

  -- Контракты с не приостановленным начислением % (просрочка меньше 90 дней)
  if (k_stop_perc is null) then
  begin
    if (perc_rate_lgota is null or perc_rate_lgota not in (10, 15, 20)) then
      exception e_message 'Процентная ставка на льготный период может принимать только следующие значения: 10%, 15%, 20%';

    if (payment_reduce_percent is null or payment_reduce_percent not between 0 and 30) then
      exception e_message 'Уменьшить первоначальный платеж по графику можно минимум на 0%, максимум на 30%';

    if (defer_month_count is null or defer_month_count not between 1 and 6) then
      exception e_message 'Количество месяцев отсрочки должно быть от 1 до 6';

    with t_zad
    as (
        select ok_sh_id,
               ok_sh_date,
               sum(ok.ok_os + ok_sh_os) sh_os,
               sum(ok.ok_perc + ok_sh_perc) sh_perc
        from operkred ok
        where ok.ok_from_k_id = :k_id and
              ok.ok_type < 7 and
              ok.ok_ol_is_active = 1 and
              ok_from_c_id = 0
        group by 1, 2
    )
    select count(ok_sh_date)
    from t_zad
    where :rep_date >= ok_sh_date and
          (sh_os <> 0 or sh_perc <> 0)
    into delinq_month; -- количество просроченных платежей

    delinq_month = coalesce(delinq_month, 0);

    with t_zad
    as (
        select ok_sh_id,
               ok_sh_date,
               sum(ok.ok_os + ok_sh_os) sh_os,
               sum(ok.ok_perc + ok_sh_perc) sh_perc
        from operkred ok
        where ok.ok_from_k_id = :k_id and
              ok.ok_type < 7 and
              ok.ok_ol_is_active = 1 and
              ok_from_c_id = 0
        group by 1, 2
    )
    select count(ok_sh_date)
    from t_zad
    where :rep_date < ok_sh_date and
          (sh_os <> 0 or sh_perc <> 0)
    into non_delinq_month; -- количество непросроченных платежей

    non_delinq_month = coalesce(non_delinq_month, 0);

    monthly_payment = 0;
    monthly_payment_base = 0;

    with zad_all
    as (
        select sh_os + sh_perc as payment,
               sh_id
        from schedule
        where sh_from_k_id = :k_id and
              sh_id > 0
    )
    select round(avg(payment), 0)
    from zad_all
    where sh_id < (select max(sh_id)
                   from zad_all)
    into monthly_payment_base;

    monthly_payment_base = coalesce(monthly_payment_base, 0);

    monthly_payment = round(:monthly_payment_base * ((100.0000 - :payment_reduce_percent) / 100.0000), 0);
    monthly_payment = coalesce(monthly_payment, 0);

    if (monthly_payment <= 0) then
      exception e_message 'Не удалось определить ежемесячный платеж';

    select kd.kd_percent
    from kontdata kd
    where kd.kd_from_k_id = :k_id and
          kd.kd_is_primary = 1
    into :primary_perc_rate;

    if (coalesce(primary_perc_rate, 0) <= 0) then
      exception e_message 'Не удалось определить текущую процентную ставку';

    if (perc_rate_lgota > primary_perc_rate) then
      exception e_message 'Льготная процентная ставка не может быть больше текущей ставки';

    -- Считаем какую скидку нужно дать на изначальную % ставку, чтобы получилось ровно :perc_rate_lgota
    target_lgota_perc_rate = cast(100 - (:perc_rate_lgota / primary_perc_rate * 100.0) as d_currency);

    /*
      defer_month_count:
        1 - отсрочка на 1 месяц
        2 - отсрочка на 2 месяца
        3 - отсрочка на 3 месяца
        ... до 6 месяцев включительно

      add_month_kind = количество первых льготных платежей
    */
    add_month_kind = defer_month_count;

    fine_account = (
      select al.al_id
      from acc_list al
      where al.al_from_reg_id = :reg_id and
            al.al_is_active = 1 and
            al.al_account = '1280.09/1'
    );

    select ol_id, error_text
    from hp_do_repayment(:k_id, 0, :rep_date, :fine, 0, 0, 1, 1, 1, 0, 1, :reg_id, 3, :fine_account, 0, 1000000000000, null,
                         'cписание пении', null, f_get_rd_id('Reason_WriteOff.fin_difficulties'), :control_amount)
    into ol_id, res_text; -- списание пении

    -- Сколько месяцев нужно добавить, стартуем с 0
    auto_month_add = 0;

    non_delinq_month = coalesce(non_delinq_month, 0);
    add_month_kind = coalesce(add_month_kind, 0);

    -- Защита от ошибки в hp_do_reschedule_auto:
    -- количество будущих периодов после продления должно быть больше add_month_kind
    if ((non_delinq_month + auto_month_add) <= add_month_kind) then
      auto_month_add = add_month_kind - non_delinq_month + 1;

    if (auto_month_add < 0) then
      auto_month_add = 0;

    auto_iter = 0;
    auto_iter_max = 120;

    while (1 = 1) do
    begin
      add_params =
          '{' ||
          '"protocol_date":"' || cast(:rep_date as varchar(10)) || '",' ||
          '"reason":4088,' ||
          '"memo":"реструктуризация графика погашений займа ' || :k_kontract_no || ' от ' || cast(:k_disb_date as varchar(10)) || '",' ||
          '"first_perc_kind":1,' ||
          '"grace_os_list":"1",' ||
          '"perc_lgota_value":' || cast(:target_lgota_perc_rate as varchar(50)) || ',' ||
          '"add_month_kind":' || cast(:add_month_kind as varchar(10)) || ',' ||
          '"add_month_count":' || cast(:auto_month_add as varchar(50)) || ',' ||
          '"new_pay_day":' || cast(:new_pay_day as varchar(10)) || ',' ||
          '"new_annu":' || cast(:monthly_payment as varchar(50)) ||
          '}';

      -- Строим реструктуризацию во временной таблице temp_sh
      select res_is_error, res_text
      from hp_do_reschedule(:k_id, :rep_date, 'auto', :add_params, 1000000000000)
      into res_is_error, res_text;

      if (coalesce(res_is_error, 0) <> 0) then
      begin
        suspend;
        exit;
      end

      /*
        Базовая защита для обычных кейсов, где корректировка не нужна.
        Если auto сам построил последний платеж больше ежемесячного лимита —
        добавляем месяц и строим заново.
      */
      last_payment_total = 0;

      select coalesce(sh_new_os, 0) + coalesce(sh_new_perc, 0)
      from temp_sh
      where sh_date >= :rep_date
        and sh_repay_mode <> 1
        and (coalesce(sh_new_os, 0) <> 0 or coalesce(sh_new_perc, 0) <> 0)
      order by sh_date desc, sh_id desc
      rows 1
      into :last_payment_total;

      last_payment_total = coalesce(last_payment_total, 0);

      if (last_payment_total > monthly_payment) then
      begin
        auto_iter = auto_iter + 1;

        if (auto_iter >= auto_iter_max) then
        begin
          res_is_error = 1;
          res_text = f_format('Не удалось построить график: последний платеж больше лимита, требуется продление срока больше %1 мес.', :auto_iter_max);
          suspend;
          exit;
        end

        auto_month_add = auto_month_add + 1;
        continue;
      end

      /*
        Корректировка для переноса разницы процентов в график
          - применяется только если первый платеж больше второго;
          - уменьшает первый платеж до второго;
          - переносит срезанный процент дальше;
          - пересчитывает проценты от нового остатка ОС;
          - возвращает need_more_month = 1, если текущего количества месяцев не хватило.

        Если первый платеж не больше второго, процедура ничего не меняет
        и вернет correction_applied = 0
      */
      correction_applied = 0;
      need_more_month = 0;

      select correction_applied, need_more_month
      from hp_kz_correct_temp_sh_percent(:rep_date, :monthly_payment)
      into :correction_applied, :need_more_month;

      if (correction_applied = 1 and need_more_month = 1) then
      begin
        auto_iter = auto_iter + 1;

        if (auto_iter >= auto_iter_max) then
        begin
          res_is_error = 1;
          res_text = f_format('Не удалось построить график: после корректировки требуется продление срока больше %1 мес. Измените параметры реструктуризации', :auto_iter_max);
          suspend;
          exit;
        end

        auto_month_add = auto_month_add + 1;
        continue;
      end

      -- График либо не требовал корректировки, либо корректировка успешно вместилась.
      leave;
    end

    month_add = auto_month_add;

    -- Обновляем add_params финальным month_add.
    add_params =
        '{' ||
        '"protocol_date":"' || cast(:rep_date as varchar(10)) || '",' ||
        '"reason":4088,' ||
        '"memo":"реструктуризация графика погашений займа ' || :k_kontract_no || ' от ' || cast(:k_disb_date as varchar(10)) || '",' ||
        '"first_perc_kind":1,' ||
        '"grace_os_list":"1",' ||
        '"perc_lgota_value":' || cast(:target_lgota_perc_rate as varchar(50)) || ',' ||
        '"add_month_kind":' || cast(:add_month_kind as varchar(10)) || ',' ||
        '"add_month_count":' || cast(:month_add as varchar(50)) || ',' ||
        '"new_pay_day":' || cast(:new_pay_day as varchar(10)) || ',' ||
        '"new_annu":' || cast(:monthly_payment as varchar(50)) ||
        '}';

    -- Сохраняем уже скорректированный temp_sh в operkred
    select res_is_error, res_text
    from hp_do_reschedule(:k_id, :rep_date, 'save', :add_params, 1000000000000)
    into res_is_error, res_text;

    if (coalesce(res_is_error, 0) <> 0) then
    begin
      suspend;
      exit;
    end

    repay_account = (
      select al.al_id
      from acc_list al
      where al.al_from_reg_id = :reg_id and
            al.al_is_active = 1 and
            al.al_account = '1021/1'
    );

    select sum(sh_calc_fine + sh_calc_perc + zad_os)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    into control_amount_new; -- общая сумма задолженности после реструктуризации

    control_amount_new = coalesce(control_amount_new, 0);

    select ol_id, error_text
    from hp_do_repayment(:k_id, 0, :rep_date, 0, 1, 1, 1, 1, 1, 0, 0, :reg_id, 3, :repay_account, 0, 1000000000000, :c_id, '', 0, 0, :control_amount_new)
    into ol_id, res_text; -- делаем нулевое погашение

    effect_rate_common = 0;

    delete from temp_schedule
    where 1 = 1;

    insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
    select sh_id, sh_date, sh_days, sh_all
    from hp_effect_rate_reschedule_kz(:k_id, :rep_date);

    select calc_eff
    from hp_effect_rate_kg(:k_id)
    into effect_rate_common; -- есп в расчете вбс

    update kontdata kd
    set kd.kd_percent_effect_fifd = :effect_rate_common
    where kd.kd_from_k_id = :k_id and
          kd.kd_ol_is_delete = 0 and
          kd.kd_date_end >= '31.12.9999';
  end
  else
  begin
    add_month_90 = 0;
    last_month = null;
    ol_month = extract(month from :rep_date);

    for select sh_date
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    where zad_os <> 0 or zad_perc <> 0
    order by sh_date desc
    rows 2
    into sh_month
    do
    begin
      if (last_month is null) then
        last_month = sh_month;

      if (extract(month from last_month) = ol_month) then
        add_month_90 = 1;
      else if (last_month > :rep_date) then
        add_month_90 = 1;
      else if (last_month <> sh_month) then
      begin
        add_month_90 = datediff(month, last_month, :rep_date);

        if (datediff(month, sh_month, last_month) > 1) then
          add_month_90 = add_month_90 + 1;
      end
    end

    select res_is_error
    from hp_kont_reschedule_move(:k_id, :rep_date, :add_month_90, 1000000000000)
    into res_is_error;
  end

  suspend;
end
