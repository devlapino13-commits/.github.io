CREATE PROCEDURE HP_BLOCKCLIENTS_SEND_MSG
AS
declare variable ma_id d_integer;
declare variable sbc_id d_id;
declare variable k_id d_id;
begin
  -- запускается только по рабочим дням
  if (f_date_is_weekend(current_date, 0) = 0) then
  begin
    -- выборка из таблицы блок клиента
    for
    select sbc_from_ma_id, sbc_id
    from smevpass_blockclients
    where sbc_unblock_datetime is null
    into ma_id, sbc_id
    do
    begin
      -- по тем по которым не были отправлены СМС
      if (not exists(select * from smevpass_msg
      where spm_from_sbc_id = :sbc_id
      and spm_kind = 1
      )) then
      begin
        rdb$set_context('USER_TRANSACTION', 'SMEV_REJECT_MA_ID', ma_id);

        select k_id from kontract
        where k_from_ma_app_id = :ma_id
        into k_id;

        insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                                 krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
        values (1, 1000000000000, :k_id, 0, null, current_timestamp, current_date, current_date, 0);
    
        update kontract
        set k_status = -1,
            k_user_id = 1000000000000
        where k_id = :k_id;

        execute procedure hp_smevpass_send_msg(1, :sbc_id, 1000000000000, :ma_id);
      end 
    end
  end 
end
