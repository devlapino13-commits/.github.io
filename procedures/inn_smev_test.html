CREATE PROCEDURE INN_SMEV_TEST
AS
declare variable COUNTER D_INTEGER;
declare variable C_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable BIRTH_DATE D_DATE;
declare variable PK_ID D_INTEGER;
declare variable DOC_NO D_STR_CODE;
declare variable LAST_NAME D_STR_CODE;
declare variable MIDDLE_NAME D_STR_CODE;
declare variable NAME_TAIL D_STR_CODE;
declare variable DOC_TYPE D_INTEGER;
declare variable C_ALTERNAT_FIO D_STR_100;
declare variable CFG D_STR_20;
declare variable FIRST_NAME d_str_100;
declare variable DOC_DATE d_str_code;
begin
    if (exists(select *
             from hp_db_is_actual)) then
        exit;

    counter = 0;

    for select kncl.kc_from_c_id
    from kontstate ks
    inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
    where ks.ks_state = 1
    and current_date between ks.ks_date_begin and ks.ks_date_end
    and ks.ks_ol_is_delete = 0
    and kncl.kc_amount_approved > 0
    and ks.ks_date_begin >= current_date - 3
    into :c_id
    do
    begin
        select c_name_fio,
             c_birth_date,
             c_alternat_fio
        from clients
        where c_id = :c_id
        into c_name_fio,
           birth_date,
           c_alternat_fio;

        select txt1, txt2, txt3
        from hp_fio_clear(:c_name_fio)
        into last_name, first_name, middle_name;
        name_tail = '';

        if (c_name_fio like '%УУЛУ%') then
          name_tail = 'УУЛУ';
        if (c_name_fio like '%КЫЗЫ%') then
          name_tail = 'КЫЗЫ';
        if (c_name_fio like '%УГЛИ%') then
           name_tail = 'УГЛИ';
        if (c_name_fio like '%КИЗИ%') then
          name_tail = 'КЫЗЫ';
    
        if (name_tail <> '') then
          if (middle_name <> '') then
            middle_name = middle_name || ' ' || name_tail;
          else
            last_name = last_name || ' ' || name_tail;
    
        for select distinct
            pk_id,
            cds_doc_no,
            f_datetostr(cds_doc_date_begin)
        from clientdocs cds
        inner join passkind on cds_from_pk_id = pk_id
        inner join passdoctype on pk_from_pdt_id = pdt_id
        where cds_from_c_id = :c_id
        and pdt_is_main_doc = 1
        and cds_verify_status = 2
        and cds.cds_is_delete = 0
        order by cds.cds_doc_date_begin desc
        rows 1
        into pk_id, doc_no, doc_date
        do
        begin
          -- Полный справочник документов:
          -- Паспорт гражданина СССР - "01"
          -- Свидетельство о рождении - "03"
          -- Паспорт иностранного гражданина - "10"
          -- Вид на жительство в России - "12"
          -- Разрешение на временное проживание в России - "15"
          -- Свидетельство о предоставлении временного убежища на территории России - "19"
          -- Паспорт гражданина России - "21"
          -- Свидетельство о рождении, выданное уполномоченным органом иностранного государства - "23"
          -- Вид на жительство иностранного гражданина - "62"
      
          doc_no = replace(doc_no, ' ', '');
          if (coalesce(doc_no, '') = '') then
            continue;
    
          -- Паспорт/Загранпаспорт РФ
          if (pk_id = 12) then
          begin
            doc_type = 21;
            --Серия и номер паспорта гражданина РФ имеют формат 99 99 9999990, где:
            --9 – десятичная цифра (обязательная к вводу);
            --0 – десятичная цифра (необязательная, может отсутствовать).
            doc_no = left(doc_no, 2) || ' ' || substring(doc_no from 3 for 2) || ' ' || substring(doc_no from 5);
          end
          else
            doc_type = 10;
        end

        if (coalesce(doc_no, '') = '') then
          continue;

        if(coalesce(birth_date, '') = '') then
            continue;
    
        if(coalesce(c_name_fio, '') = '') then
            continue;

        -- создается запрос на получени ИНН
        execute procedure hp_extinfo_insert('inn_smev', 'clients', :c_id, 0, 1010100000278);
        counter = counter + 1;

        if (counter = 10) then
        begin
            post_event 'inn_smev';
            exit;
        end
    end
end
