CREATE PROCEDURE HP_ACTION_ON_RAMADAN
RETURNS (
    K_ID ID_TYPE,
    OL_ID ID_TYPE,
    OL_DATE DATE,
    OK_NUMBER SMALLINT,
    SH_ID SMALLINT,
    SH_DATE DATE,
    SH_CALC_FINE CURR_TYPE,
    OK_OL_REG_ID INTEGER)
AS
declare variable rep_date_begin date;
declare variable rep_date_end date;
declare variable k integer;
begin
  rep_date_begin = '26.08.2014';
  rep_date_end = current_date;

  for select ok_from_k_id, ok_from_ol_id, ok_ol_date, ok_ol_reg_id
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_type = 4 and
            ok_from_c_id = 0
      group by ok_from_k_id, ok_from_ol_id, ok_ol_date, ok_ol_reg_id
      having sum(ok_os + ok_perc + ok_fine) < 0
      into k_id, ol_id, ol_date, ok_ol_reg_id
  do
  begin
--  проверка на дату погашения
    k = 0;
    ok_number = 0;
    if (not exists(select *
                   from oksummary
                   where :ol_date between oks_date_begin and oks_date_end and
                         :ol_date > oks_delinq_date and
                         oks_from_k_id = :k_id and
                         oks_close_state = 0)) then
      for select sh_id, sh_date, sh_calc_fine
          from aarep_make_zad_list(:k_id, :ol_date, 0)
          where sh_calc_fine > 0 and
                sh_date is not null
          into sh_id, sh_date, sh_calc_fine
      do
      begin
        if (ok_number = 0) then
        begin
          ok_number = (select max(ok_number)
                       from operkred
                       where ok_from_ol_id = :ol_id and
                             ok_from_c_id = 0);
        end

        k = k + 1;
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                              ok_ol_reg_id, ok_ol_is_active)
        values (:ol_id, :ok_number + :k, :k_id, :sh_id, :sh_date, 5, 0, 0, -:sh_calc_fine, 3, 0, 0, 0, 0, 0, :ol_date,
                :ok_ol_reg_id, 1);

--      suspend;
      end

    if (rep_date_end = '30.09.2014') then
    begin
-- проверка на дату конец
      k = 0;
      ol_id = 0;
      for select sh_id, sh_date, sh_calc_fine
          from aarep_make_zad_list(:k_id, :rep_date_end, 0)
          where sh_calc_fine > 0
          into sh_id, sh_date, sh_calc_fine
      do
      begin
        if (ol_id = 0) then
        begin
          select kv
          from get_kv(1010100, 5)
          into :ol_id;

          insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status,
                                is_delete, ol_user_id)
          values (:ol_id, :ok_ol_reg_id, 5, :rep_date_end, 4,
                  'Списание пени от 30.09.2014г. Продление акции по Рамазану', 0,
                  0, 1000000000001);
        end

        k = k + 1;
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                              ok_ol_reg_id, ok_ol_is_active)
        values (:ol_id, :k, :k_id, :sh_id, :sh_date, 5, 0, 0, -:sh_calc_fine, 3, 0, 0, 0, 0, 0, :rep_date_end,
                :ok_ol_reg_id, 1);

--        suspend;
      end
    end

  end
end
