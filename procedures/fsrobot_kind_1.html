CREATE PROCEDURE FSROBOT_KIND_1(
    REP_EMAIL_FROM VARCHAR(100),
    REP_SUBJECT VARCHAR(255))
RETURNS (
    RESULT_STR VARCHAR(255),
    RESULT_IS_OK INTEGER,
    RESULT_UN_FIO VARCHAR(100))
AS
declare variable un_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable fsr_text varchar(255);
declare variable len_text integer;
declare variable param_str varchar(255);
declare variable k_id decimal(18,0);
declare variable k_status integer;
declare variable rep_command varchar(255);
declare variable ksr_id integer;
declare variable reg_id integer;
declare variable err_count integer;
declare variable reg_name varchar(100);
declare variable kf_name varchar(100);
declare variable un_name varchar(30);
declare variable n integer;
declare variable kont_no varchar(20);
declare variable dd_action date;
declare variable kont_client_fio varchar(100);
declare variable client_id decimal(18,0);
declare variable k_new_id decimal(18,0);
declare variable k_new_no varchar(20);
declare variable cell_no varchar(20);
declare variable c_name_fio varchar(100);
declare variable sdr_memo varchar(255);
begin
  result_is_ok = 1;
  result_str = '';
  rep_subject = coalesce(rep_subject, '');
  rep_subject = trim(rep_subject);

  select distinct fsr_sys_command
  from fsrobot
  where :rep_subject starting fsr_text rows 1
  into rep_command;

  if (rep_command <> 'agent_register') then
  begin
-- Для регистрации агента, параметры пользователя не определяем, т.к.
-- e-mail может быть не привязан к таблице Clients (н-р: manager, mbf)
-- Для остальных команд определяем
    select user_id, user_name, user_c_id, user_fio, user_result_txt, user_command
    from hp_get_user_by_email(:rep_email_from, :rep_subject)
    into :un_id, :un_name, :c_id, :result_un_fio, :result_str, :rep_command;

    if (result_str <> '') then -- Проблема с пользователем, выходим без выполнения команд
    begin
      result_is_ok = 0;
-- Result_Str уже содержит нужный текст про проблемы пользователя
      suspend;
      exit;
    end
    rdb$set_context('USER_SESSION', 'FINSTART_USER_LOGIN', :un_name);
  end

  if (rep_command = 'HELP') then -- Команда на получение список команд
  begin
    for select '<B>' || fsr_text || '</B>  - ' || fsr_memo
        from fsrobot
        into :result_str
    do
    begin
      result_is_ok = 1;
      suspend;
    end
    exit;
  end

  if (rep_command = 'APPROVE_LOAN') then -- Команда на утверждение договора
  begin
    param_str = '';
    for select fsr_text
        from fsrobot
        where fsr_sys_command = :rep_command and
              fsr_kind = 1
        into :fsr_text
    do
    begin
      len_text = char_length(fsr_text);
      if (left(rep_subject, :len_text) = :fsr_text) then
      begin
        param_str = trim(substring(:rep_subject from :len_text + 1 for 255));
        break;
      end
    end
    if (param_str = '') then
    begin
      result_is_ok = 0;
      result_str = 'Команда не распознана';
      suspend;
      exit;
    end
    select k_id, k_status, kal_new_reg_id, reg_name, kal_new_kf_name
    from kontract, knaktlist, region
    where k_kontract_no = :param_str and
          kal_from_kf_id = k_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_date <= k_disb_date and
          kal_date_end >= k_disb_date and
          kal_new_reg_id = reg_unicode rows 1
    into :k_id, :k_status, :reg_id, :reg_name, :kf_name;

    if (k_id is null) then
    begin
      result_is_ok = 0;
      result_str = 'Договор не найден';
      suspend;
      exit;
    end

    if (k_status <> 0) then
    begin
      if (k_status = 1) then
        result_str = 'Договор уже утвержден';
      if (k_status = -1) then
        result_str = 'Договор аннулирова';
      if (k_status = -2) then
        result_str = 'Договора нет, на стадии заявки';
      result_is_ok = 0;
      suspend;
      exit;
    end

-- Утверждаем договор

    if (un_id <> 1090100000524) then -- Исключением является Бахтияр Тольбаев
    begin
-- Проверяем права
      ksr_id = 0;
      select ksr_id
      from aarep_kontsign_access(:k_id, :un_id, :reg_id, '')
      into :ksr_id;
      if (ksr_id = 0) then
      begin
        result_str = 'У вас нет прав на этот регион';
        result_is_ok = 0;
        suspend;
        exit;
      end
      err_count = 0;
      for select kf_client_fio || ': ' || kf_error_text
          from hp_check_loan_konditions(:k_id, 0, 0, '')
          where kf_error_is_critical >= 1
          into :result_str
      do
      begin
        result_is_ok = 0;
        err_count = err_count + 1;
        suspend;
      end
-- Если были критические сообщения, отказываем
      if (err_count > 0) then
      begin
        result_str = 'Отказано системой. Необходимо устранить вышеуказанные предупреждения.';
        suspend;
        exit;
      end
    end
-- Если дошли до этой строки, значит утверждаем

-- Утверждение отключено на момент тестирования
    update kontract
    set k_status = 1,
        k_user_id = :un_id
    where k_id = :k_id;

    result_str = 'Договор успешно утвержден. Договор: ' || :param_str || ', Офис: ' || :reg_name || ', кр.папка: ' || :kf_name;
    result_is_ok = 1;
    suspend;
  end

  if (rep_command = 'AGENT_REGISTER') then -- Команда на утверждение договора
  begin
    param_str = '';
    for select fsr_text
        from fsrobot
        where fsr_sys_command = :rep_command and
              fsr_kind = 1
        into :fsr_text
    do
    begin
      len_text = char_length(fsr_text);
      if (left(rep_subject, :len_text) = :fsr_text) then
      begin
        param_str = trim(substring(:rep_subject from :len_text + 1 for 255));
        break;
      end
    end

    if (not((rep_email_from like '%@molbulak.com') or (rep_email_from like '%@molbulakfinance.kg') or (rep_email_from like '%@mbf.kg') or (rep_email_from like '%@molbulak.ru') or (rep_email_from like '%@mbfplanet.com'))) then
    begin
      result_is_ok = 0;
      result_str = 'Данная команда доступна только через корпоративную почту';
      suspend;
      exit;
    end

    if (param_str = '') then
    begin
      result_is_ok = 0;
      result_str = 'Команда не распознана';
      suspend;
      exit;
    end

    n = 0;
    cell_no = '';
    while (n < char_length(param_str)) do
    begin
      n = n + 1;
      if (substring(param_str from n for 1) in ('1', '2', '3', '4', '5', '6', '7', '8', '9', '0') and
          (char_length(cell_no) < 19)) then
      begin
        cell_no = cell_no || substring(param_str from n for 1);
      end
    end
    if (char_length(cell_no) <> 11) then
    begin
      result_is_ok = 0;
      result_str = 'Неправильный формат телефона. Регистрация агента отменена.';
      suspend;
      exit;
    end
    if (substring(cell_no from 1 for 1) = '8') then
    begin
      cell_no = '7' || substring(cell_no from 2 for 10);
    end

    c_name_fio = '';
    select res_object_id, c_name_fio
    from hp_contdata_find_object(:cell_no, 'cell'), clients
    where res_object_kind = 0 and
          res_object_id = c_id rows 1
    into :c_id, :c_name_fio;
    if (c_name_fio is null) then
      c_name_fio = '';

    if (c_name_fio = '') then
    begin
      result_is_ok = 0;
      result_str = 'Ошибка! В базе нет клиента с номером: ' || cell_no;
      suspend;
      exit;
    end

-- Имитируем входящий смс
    if (c_id is not null) then
      sdr_memo = c_id;
    else
      sdr_memo = :rep_email_from;
    insert into smsdialogreciever (sdr_id, sdr_cell_no, sdr_text, sdr_from_c_id, sdr_memo)
    values (-1, :cell_no, 'Email_for_Agent', :c_id, :sdr_memo);

    result_str = c_name_fio || ' успешно зарегистрирован как агент.' || ascii_char(13) || ascii_char(10) || 'В течении 2 минут агенту будет выслан СМС с кодом агента.';
    result_is_ok = 1;
    suspend;
    exit;
  end

  if (rep_command = 'AMNESTY_CLIENT') then -- Команда на амнистирирование клиента
  begin
    param_str = '';
    for select fsr_text
        from fsrobot
        where fsr_sys_command = :rep_command and
              fsr_kind = 1
        into :fsr_text
    do
    begin
      len_text = char_length(fsr_text);
      if (left(rep_subject, :len_text) = :fsr_text) then
      begin
        param_str = trim(substring(:rep_subject from :len_text + 1 for 255));
        break;
      end
    end
    if (param_str = '') then
    begin
      result_is_ok = 0;
      result_str = 'Команда не распознана';
      suspend;
      exit;
    end

    kont_client_fio = '';
    kont_no = '';
    n = position(' ', param_str);
    if (n > 0) then
    begin
      kont_no = left(param_str, n - 1);
      param_str = trim(substring(:param_str from :n + 1 for 255));
    end
    else
    begin
      result_is_ok = 0;
      result_str = 'Параметры не определены';
      suspend;
      exit;
    end

    n = position(' ', param_str);
    if (n > 0) then
    begin
      dd_action = cast(left(param_str, n - 1) as date);
      param_str = trim(substring(:param_str from :n + 1 for 255));
    end
    else
    begin
      result_is_ok = 0;
      result_str = 'Дата в параметре не определена';
      suspend;
      exit;
    end
    if (param_str <> '') then
    begin
      kont_client_fio = trim(param_str);
    end

    if (kont_client_fio = '') then
    begin
      result_is_ok = 0;
      result_str = 'ФИО в параметре не определено';
      suspend;
      exit;
    end

    select k_id, k_status, kal_new_reg_id, reg_name, kal_new_kf_name
    from kontract, knaktlist, region, kontstate
    where k_kontract_no = :kont_no and
          kal_from_kf_id = k_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_date <= :dd_action and
          kal_date_end >= :dd_action and
          kal_new_reg_id = reg_unicode and
          k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_date_begin <= :dd_action and
          ks_date_end >= :dd_action and
          ks_state = 1 rows 1
    into :k_id, :k_status, :reg_id, :reg_name, :kf_name;

    if (k_id is null) then
    begin
      result_is_ok = 0;
      result_str = 'Нет активного договора с номером: ' || kont_no;
      suspend;
      exit;
    end

    client_id = null;
    select c_id
    from clients, knclient
    where kc_from_c_id = c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = :k_id and
          c_name_fio starting :kont_client_fio rows 1
    into :client_id;

    if (client_id is null) then
    begin
      result_is_ok = 0;
      result_str = 'В договоре отсутствует клиент с именем: ' || kont_client_fio;
      suspend;
      exit;
    end

-- Проверяем права
    if (not(exists(select *
                   from region
                   where reg_unicode = :reg_id and
                         reg_email_manager = :rep_email_from))) then
    begin
      if (not((rep_email_from = 'l.abdurakhmanova@molbulak.com') or (rep_email_from = 's.danilova@molbulak.com') or (rep_email_from = 'r.ilalov@molbulak.com'))) then
      begin
        result_str = 'У вас нет прав на этот регион. Команда должна быть от руководителя офиса.';
        result_is_ok = 0;
        suspend;
        exit;
      end
    end
    else
    begin
      if ((not(current_date between '23.06.2014' and '03.08.2014')) and
          (not(current_date between '26.08.2014' and '30.09.2014'))) then
      begin
        result_str = 'Данная опция не действительно за этот период.';
        result_is_ok = 0;
        suspend;
        exit;
      end
    end

    if (dd_action < current_date) then
    begin
      result_str = 'Запрешается амнистировать задним числом';
      result_is_ok = 0;
      suspend;
      exit;
    end

-- Если дошли до этой строки, значит амнистируем

-- Утверждение отключено на момент тестирования
    k_new_no = null;
    select k_new_id, k_new_no
    from hp_devide_kontract_clients(:k_id, :client_id, :un_id, :dd_action, :rep_email_from || ': ' || :rep_subject, 0)
    into :k_new_id, :k_new_no;

    if (k_new_no is null) then
      k_new_no = '';
    if ((k_new_id is null) or (k_new_id = 0)) then
    begin
      result_str = 'Произошла ошибка при амнистировании: ' || k_new_no;
      result_is_ok = 0;
      suspend;
      exit;
    end

    result_str = kont_client_fio || ' успешно амнистирован. Номер нового договора клиента: ' || :k_new_no || ', Офис: ' || :reg_name || ', кр.папка: ' || :kf_name;
    result_is_ok = 1;
    suspend;
  end
end
