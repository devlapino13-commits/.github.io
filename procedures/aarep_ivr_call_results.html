CREATE PROCEDURE AAREP_IVR_CALL_RESULTS(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    OAS_ID D_ID,
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_CODE,
    PROJECT D_STR_100,
    OAS_CALL_DATETIME D_DATETIME,
    OAS_RESULT D_STR_100,
    OAS_TIMEZONE D_INTEGER,
    OAS_CALL_DATE D_DATE,
    OAS_DURATION D_INTEGER,
    OAS_ATTEMPTS_REMAINING_QTY D_INTEGER,
    OAS_ABONENT_PHONE D_STR_100,
    OAS_OPERATOR_PHONE D_STR_100,
    OAS_IGNORE_REASON D_STR_100)
AS
begin
  for select t.oas_id,
             cl.c_name_fio,
             k.k_kontract_no,
             rd_proj.rd_name,
             oas_call_datetime,
             rd_res.rd_name,
             oas_timezone,
             oas_call_date,
             oas_duration,
             oas_attempts_remaining_qty,
             cl_abonent.clt_code,
             cl_operator.clt_code,
             rd_i.rd_name
      from okt_abonent_calls t
      inner join pantera_kontracts pak on pak.pant_id = t.oas_from_pant_id
      inner join clients cl on cl.c_id = oas_object_id
      inner join kontract k on k.k_id = pak.pant_from_k_id
      inner join refdata rd_proj on rd_proj.rd_id = pak.pant_rd_project_type
      inner join refdata rd_res on rd_res.rd_id = t.oas_result
      left join contlist cl_abonent on cl_abonent.clt_id = t.oas_abonent_phone
      left join contlist cl_operator on cl_operator.clt_id = t.oas_operator_phone
      left join pantera_ignore pii on pii.pi_id = t.oas_ignore_id
      left join refdata rd_i on rd_i.rd_id = pii.pi_reason_rd_id
      where t.oas_call_date between :rep_date_begin and :rep_date_end
      order by t.oas_id
      into :oas_id,
           :c_name_fio,
           :k_kontract_no,
           :project,
           :oas_call_datetime,
           :oas_result,
           :oas_timezone,
           :oas_call_date,
           :oas_duration,
           :oas_attempts_remaining_qty,
           :oas_abonent_phone,
           :oas_operator_phone,
           :oas_ignore_reason
  do
  begin
    suspend;
  end
end
