CREATE PROCEDURE AARE_AZ_PAYCARD_DISB_CLI(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    DISB_DATE DATE,
    DISB_DATE_TIME TIMESTAMP,
    CUSTOMERID STR_100,
    FIRSTNAME STR_100,
    LASTNAME STR_100,
    MIDDLENAME STR_100,
    BIRTHDATE DATE,
    GENDER STR_100,
    BRANCH STR_100,
    MARITALSTATUS STR_100,
    MOBILE_PHONE STR_100,
    CUSTEMAIL STR_100,
    PASSPORTSERIAL STR_100,
    PASSPORTNUMBER STR_100,
    PINCODE VARCHAR(20),
    TIN STR_100,
    COUNTRY STR_100,
    CITY STR_100,
    REGISTRATION_ADDRESS STR_100,
    STATUS STR_100,
    TYPE_ STR_100,
    AMOUNT_OF_LOAN CURR_TYPE,
    CURRENCY VARCHAR(10),
    PANNUMBER STR_100,
    EXPIRYDATE STR_100,
    SECRETWORD STR_100,
    AMOUNT_OF_LOAN_APPROVED CURR_TYPE,
    PRODUCT_NAME VARCHAR(50),
    K_KOMMIS CURR_TYPE)
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable fl_id id_type;
declare variable c_pass_no_tmp str_100;
declare variable passportserial_tmp str_100;
declare variable k_currency varchar(10);
declare variable k_currency_old varchar(10);
declare variable k_memo varchar(1024);
declare variable ol_id id_type;
declare variable pr_id id_type;
declare variable c_name_fio str_255;
declare variable k_kommis_perc perc_type;
begin
  k_currency_old = '';
  k_memo = '';

  for select kont.k_id, kont.k_kontract_no, kont.k_currency, kont.k_memo, ks.ks_from_ol_id, kont.k_kommis_perc,
             k_from_pr_id, pr_name
      from kontstate ks
      inner join kontract kont on ks.ks_from_k_id = kont.k_id
      inner join product on (k_from_pr_id = pr_id and
            pr_code not containing '.Micro_Biznes.')
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end
      into :k_id, :customerid, :k_currency, :k_memo, :ol_id, :k_kommis_perc, :pr_id, --pr_id добавил Музаффар по запросу Саида, запрос от 23.01.18
           :product_name --добавил Музаффар по запросу Саида, запрос от 23.01.18
  do
  begin
    select ol.ol_date, ol.ol_datetime
    from operlist ol
    where ol.ol_id = :ol_id
    into :disb_date, :disb_date_time;

    if (k_currency_old <> k_currency) then
    begin
      select crl.cu_symbol
      from currlist crl
      where crl.cu_id = :k_currency
      into :currency;

      k_currency_old = k_currency;
    end

    for select cl.c_id, cl.c_birth_date, cl.c_from_fl_id, coalesce(cl.c_pass_no, ''), coalesce(cl.c_pincode, ''),
               kncl.kc_amount_approved, cl.c_name_fio
        from knclient kncl
        inner join clients cl on kncl.kc_from_c_id = cl.c_id
        where kncl.kc_amount_approved > 0 and
              kncl.kc_from_k_id = :k_id
        into :c_id, :birthdate, :fl_id, :c_pass_no_tmp, :pincode, :amount_of_loan_approved, :c_name_fio
    do
    begin
      k_kommis = amount_of_loan_approved / 100 * k_kommis_perc;

      /*1.    Если İD кредитного продукта один из: 64, 65, 66, 67 (это продукты мобильных выдач кредитов) то
        поле “AMOUNT_OF_LOAN” должна вычисляться как указано ниже:
        “AMOUNT_OF_LOAN”= “Сумма Выдачи кредита” – 3 маната (это оплата за проверку клиента в МКР) – “Сумма комиссии”
        74, 75, 76, 77, 78, 79 dobavlen 25.04.18 mrahmanov
        */
      if (pr_id in (64, 65, 66, 67, 74, 75, 76, 77, 78, 79)) then
        amount_of_loan = amount_of_loan_approved - 3 - k_kommis;
      else
        amount_of_loan = amount_of_loan_approved;

      select txt1, txt2, txt3
      from hp_fio_clear(:c_name_fio)
      into :lastname, :firstname, :middlename;

      passportserial = '';
      passportnumber = '';

      c_pass_no_tmp = trim(coalesce(c_pass_no_tmp, ''));

      if (character_length(c_pass_no_tmp) > 3) then
      begin
        passportserial_tmp = substring(c_pass_no_tmp from 1 for 3);

        if (passportserial_tmp starting('MN') or passportserial_tmp starting('SQ') or passportserial_tmp starting('DQ') or passportserial_tmp starting('AD') or passportserial_tmp starting('GH') or passportserial_tmp starting('SX') or passportserial_tmp starting('ZH')) then
        begin
          passportserial = substring(passportserial_tmp from 1 for 2);
          passportnumber = substring(c_pass_no_tmp from 3);

          if (pincode = '') then
          begin
            pincode = replace(iif(extract(day from birthdate) < 10, '0', '') || extract(day from birthdate) || '.' || iif(extract(month from birthdate) < 10, '0', '') || extract(month from birthdate) || '.' || extract(year from birthdate), ' ', '');
          end
        end
        else
        if (passportserial_tmp = 'AZE') then
        begin
          passportserial = passportserial_tmp;
          passportnumber = substring(c_pass_no_tmp from 4);
        end
        else
          passportnumber = c_pass_no_tmp;
      end

      pannumber = '';
      expirydate = '';
      secretword = '';

      k_memo = coalesce(k_memo, '');

      if (char_length(k_memo) = 27) then
      begin
        pannumber = substring(k_memo from 1 for 16);
        expirydate = substring(k_memo from 18 for 5);
        secretword = substring(k_memo from 24);
      end
      suspend;
    end
  end
end
