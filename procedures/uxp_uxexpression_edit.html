CREATE PROCEDURE UXP$UXEXPRESSION_EDIT(
    EXP_UX_UUID D_UUID,
    EXP_KIND D_SMALLINT,
    EXP_EXPRESSION D_STR_LONG,
    EXP_CALC_UFR_UUID D_UUID,
    EXP_MESSAGE D_STR_LONG,
    EXP_IS_SQL D_BOOL,
    EXP_ACTION D_SMALLINT,
    EXP_POSITION D_INTEGER,
    EXP_CALC_KIND D_SMALLINT = null)
AS
begin
  if (trim(coalesce(exp_expression, '')) = '') then
  begin
    delete from ux$expression
    where exp_ux_uuid = :exp_ux_uuid and
          exp_kind = :exp_kind;
  end else
  begin
    update ux$expression
    set exp_expression = :exp_expression,
        exp_calc_ufr_uuid = :exp_calc_ufr_uuid,
        exp_message = :exp_message,
        exp_is_sql = :exp_is_sql,
        exp_action = :exp_action,
        exp_calc_kind = coalesce(:exp_calc_kind, exp_calc_kind),
        exp_position = coalesce(:exp_position, exp_position)
    where exp_ux_uuid = :exp_ux_uuid and
          exp_kind = :exp_kind;

    if (row_count = 0) then
    begin
      insert into ux$expression (exp_ux_uuid, exp_kind, exp_expression, exp_calc_ufr_uuid, exp_message,
                                 exp_is_sql, exp_action, exp_position, exp_calc_kind)
      values (:exp_ux_uuid, :exp_kind, :exp_expression, :exp_calc_ufr_uuid, :exp_message,
              :exp_is_sql, :exp_action, :exp_position, :exp_calc_kind);
    end
  end
end
