CREATE PROCEDURE AAREP_CONTROLLER_DETAIL_REP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    CONTR_TYPE SMALLINT)
RETURNS (
    CONTROLLER_FIO STR_100,
    REG_NAME STR_100,
    KF_NAME STR_100,
    K_KONTRACT_NO STR_100,
    C_NAME_FIO STR_100,
    CONTR_BLOCKED_DATETIME TIMESTAMP,
    CHECKED_DATE DATE,
    CONTR_CHECKED_DATETIME TIMESTAMP,
    ANNUL_STATUS STR_100,
    CHECK_RESULT STR_100,
    CLIENT_APPROVED INTEGER,
    KOMMENT_VALUE STR_255,
    REP_C_ADDRESS_JOB STR_255)
AS
declare variable rd_id integer;
declare variable c_id id_type;
declare variable un_id id_type;
declare variable check_result_id integer;
declare variable un_id_old id_type;
declare variable reg_id_old integer;
declare variable reg_id integer;
begin
  /*
  CONTR_TYPE  = 1 - СВО
                2 - Контролер
  */
  select rd.rd_id
  from refdata rd
  where rd.rd_code = (case :contr_type
                        when 1 then 'ControllerType.Specialist_Visitor'
                        when 2 then 'ControllerType.Controller'
                      end)
  into :rd_id;

  un_id_old = 0;
  reg_id_old = 0;

  for select id_list, kal.kal_new_kf_name, kont.k_kontract_no, cont.contr_from_c_id, cont.contr_blocked_datetime,
             cont.contr_from_un_id, cont.contr_check_result, cont.contr_checked_date, cont.contr_checked_datetime,
             iif(cont.contr_is_delete = 2, 'Аннулирован (Изменили данные по месту работы)', ''),
             kont.k_kolich_client_approved
      from controller cont
      inner join kontract kont on kont.k_id = cont.contr_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = kal.kal_new_reg_id
      inner join region reg on reg.reg_unicode = id_list
      where cont.contr_checked_date between :rep_date_begin and :rep_date_end and
            cont.contr_is_delete in (0, 2) and
            cont.contr_rd_type = :rd_id and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      order by contr_from_un_id, id_list, contr_checked_datetime, contr_from_c_id
      into :reg_id, :kf_name, :k_kontract_no, :c_id, :contr_blocked_datetime, :un_id, :check_result_id, :checked_date,
           :contr_checked_datetime, :annul_status, :client_approved
  do
  begin
    if (:un_id <> :un_id_old) then
    begin
      un_id_old = un_id;

      select cl.c_name_fio
      from username un
      inner join clients cl on un.un_from_c_id = cl.c_id
      where un.un_id = :un_id
      into :controller_fio;
    end

    if (reg_id <> reg_id_old) then
    begin
      reg_id_old = reg_id;

      select reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_name;
    end

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into :c_name_fio;

    select rd.rd_name
    from refdata rd
    where rd.rd_id = :check_result_id
    into :check_result;

    komment_value = '';

    select cdv_value
    from hp_get_customdatavalue(0, :c_id, 'work_komment', :rep_date_end)
    rows 1
    into :komment_value;

    komment_value = coalesce(komment_value, '');

    rep_c_address_job = '';

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :c_id and
          adc_address_type = 3 and
          adc_from_adl_id = adl_id
    into :rep_c_address_job;

    if (coalesce(rep_c_address_job, 'Не указан') = 'Не указан') then
      rep_c_address_job = '';
    if (rep_c_address_job = '') then
    begin
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'work_address_name', :rep_date_end)
      into rep_c_address_job;
    end

--    select CDV_VALUE
--    from HP_GET_CUSTOMDATAVALUE(0, :c_id, 'work_address_name', :REP_DATE_END)
--    into :REP_C_ADDRESS_JOB;
--
    rep_c_address_job = coalesce(rep_c_address_job, '');

    suspend;
  end
end
