CREATE PROCEDURE HP_IDJARA_GOODS_SEND_NOTIF(
    K_ID D_ID)
AS
declare variable kma_id d_id;
declare variable res_text d_str_255;
begin
  --Отправляем уведомления только по закрытым договорам продукта "Иджара товары"
  if (not exists(select *
                 from kontract kt
                 inner join product pr on pr.pr_id = kt.k_from_pr_id
                 where kt.k_id = :k_id and
                       pr.pr_code containing '.idjara_goods.')) then
    exit;

  for select kma.kma_id
      from kontmatag kma
      where kma.kma_from_k_id = :k_id
      into kma_id
  do
  begin
    res_text = '';

    --Отправка уведомления клиенту
    select res_text
    from hp_edoc_insert('idjara_goods.client_notif', 'kontmatag', :kma_id, 1000000000000)
    into res_text;

   /* if (res_text <> '') then
      exception e_default('Error: idjara goods notification send. KMA_ID: ' || :KMA_ID);
     */
    --Отправка уведомления контрагенту
    select res_text
    from hp_edoc_insert('idjara_goods.contragent_notif', 'kontmatag', :kma_id, 1000000000000)
    into res_text;
     /*
    if (res_text <> '') then
      exception e_default('Error: idjara goods notification send. KMA_ID: ' || :KMA_ID);    */
  end
end
