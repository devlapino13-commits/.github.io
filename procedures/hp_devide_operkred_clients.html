CREATE PROCEDURE HP_DEVIDE_OPERKRED_CLIENTS(
    REP_K_ID ID_TYPE,
    NEW_K_ID ID_TYPE,
    REP_C_ID ID_TYPE,
    ROUND_COEF INTEGER)
RETURNS (
    OK_FROM_OL_ID DECIMAL(18,0),
    OK_SH_ID INTEGER,
    OK_SH_DATE DATE,
    OK_TYPE INTEGER,
    OK_OS_CLI DECIMAL(18,4),
    OK_PERC_CLI DECIMAL(18,4),
    OK_FINE_CLI CURR_TYPE,
    OK_PAY_TYPE INTEGER,
    OK_KOMMIS_CLI CURR_TYPE,
    OK_SH_OS_CLI CURR_TYPE,
    OK_SH_PERC_CLI CURR_TYPE,
    OK_KOMMIS_ADV_CLI CURR_TYPE,
    OK_TAX_CLI CURR_TYPE,
    OK_OL_DATE DATE,
    OK_OL_REG_ID INTEGER,
    OK_OL_IS_ACTIVE SMALLINT)
AS
declare variable k_amount curr_type;
declare variable k_kol_cli integer;
declare variable kol_ok_type1 integer;
declare variable kol_ok_type2 integer;
declare variable kol_ok_type3 integer;
declare variable ok_from_k_id decimal(18,0);
declare variable ok_os decimal(18,4);
declare variable ok_perc decimal(18,4);
declare variable ok_fine decimal(18,4);
declare variable ok_kommis decimal(18,4);
declare variable ok_sh_os decimal(18,4);
declare variable ok_sh_perc decimal(18,4);
declare variable ok_kommis_adv decimal(18,4);
declare variable ok_tax decimal(18,4);
declare variable tc_cli_dolya double precision;
declare variable tc_amount curr_type;
declare variable k2 integer;
declare variable tc_amount2 curr_type;
declare variable k3 integer;
declare variable tc_amount3 curr_type;
declare variable ok_sh_date_prior date;
declare variable ok_os_saldo curr_type;
declare variable ok_perc_saldo curr_type;
declare variable ok_os_saldo2 curr_type;
declare variable ok_perc_saldo2 curr_type;
declare variable ok_fine_saldo2 curr_type;
declare variable ok_kommis_saldo2 curr_type;
declare variable ok_sh_os_saldo2 curr_type;
declare variable ok_sh_perc_saldo2 curr_type;
declare variable ok_kommis_adv_saldo2 curr_type;
declare variable ok_tax_saldo2 curr_type;
declare variable k1 integer;
declare variable ok_perc_is_last smallint;
declare variable ok_os_is_last smallint;
declare variable delta_perc decimal(18,4);
declare variable ok_number integer;
declare variable ok_os_total curr_type;
declare variable ok_perc_total curr_type;
declare variable delta_os decimal(18,4);
begin
-- клиент KnClient уже выведен в другой New_K_ID
-- Механизм деления для КР и АЗ, где еще нет учета на уровне клиентов

  if (round_coef is null) then
    round_coef = 2;
  k_amount = null;
  k_kol_cli = null;
  select k_amount_approved, k_kolich_client_approved
  from kontract
  where k_id = :rep_k_id
  into k_amount, k_kol_cli;
  k_amount = coalesce(k_amount, 0);
  k_kol_cli = coalesce(k_kol_cli, 0);

  tc_amount = null; -- сумма выдачи клиента
  select kc_amount_approved
  from knclient
  where kc_from_k_id = :new_k_id and
        kc_from_c_id = :rep_c_id and
        kc_amount_approved > 0
  into tc_amount;
  tc_amount = coalesce(tc_amount, 0);
  k_amount = k_amount + tc_amount;

  tc_cli_dolya = cast(tc_amount as double precision) / k_amount;

  delete from temp_operkred3
  where 1 = 1;

  ok_os_saldo = null;
  ok_perc_saldo = null;
  ok_sh_date_prior = '01.01.1900';
  kol_ok_type1 = 0;
  kol_ok_type2 = 0;
  kol_ok_type3 = 0;

  for select ok_from_ol_id, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, sum(ok_os) ok_os, sum(ok_perc) ok_perc,
             sum(ok_fine) ok_fine, max(ok_pay_type), sum(ok_kommis) ok_kommis, sum(ok_sh_os) ok_sh_os,
             sum(ok_sh_perc) ok_sh_perc, sum(ok_kommis_adv) ok_kommis_adv, sum(ok_tax) ok_tax, ok_ol_date, ok_ol_reg_id,
             ok_ol_is_active
      from operkred
      where ok_from_k_id = :rep_k_id and
            ok_ol_is_active = 1 and
            ok_from_c_id = 0
      group by ok_sh_date, ok_ol_date, ok_from_ol_id, ok_from_k_id, ok_sh_id, ok_type, ok_ol_reg_id, ok_ol_is_active
      into :ok_from_ol_id, :ok_from_k_id, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type,
           :ok_kommis, :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax, :ok_ol_date, :ok_ol_reg_id, :ok_ol_is_active
  do
  begin
    if (ok_sh_date_prior <> ok_sh_date) then
    begin
      if (ok_sh_date_prior <> '01.01.1900') then
      begin
        ok_os_saldo = 0;
        ok_perc_saldo = 0;
      end
      ok_sh_date_prior = ok_sh_date;
    end
    if (ok_type not in (7, 8)) then
    begin
--        добавлено для провкрки
      if (ok_os_saldo = -9990000 and
          ok_os + ok_sh_os <> 0) then
      begin
        ok_os_saldo = 0;
      end
--------------------------------
      ok_os_saldo = ok_os_saldo + ok_os + ok_sh_os;
--        ново добавленно для контракта исправления ошибки контракта 101010003679400101
--        могут быть, что проценты закрываются и заного появляюется по ok_type 2 или 3
      if (ok_perc_saldo = -9990000 and
          ok_perc + ok_sh_perc <> 0) then
      begin
        ok_perc_saldo = 0;
      end
      ok_perc_saldo = ok_perc_saldo + ok_perc + ok_sh_perc;
    end
    insert into temp_operkred3 (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                                ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax,
                                ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_os_is_last, ok_perc_is_last, ok_os_saldo,
                                ok_perc_saldo, ok_fine_saldo, ok_kommis_saldo, ok_sh_os_saldo, ok_sh_perc_saldo,
                                ok_kommis_adv_saldo, ok_tax_saldo, ok_os_total, ok_perc_total)
    values (:ok_from_ol_id, -1, :ok_from_k_id, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine,
            :ok_pay_type, :ok_kommis, :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax, :ok_ol_date, :ok_ol_reg_id,
            :ok_ol_is_active, iif(:ok_os_saldo = 0.0, 1, 0), iif(:ok_perc_saldo = 0.0, 1, 0), :ok_os, :ok_perc,
            :ok_fine, :ok_kommis, :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax, :ok_os_saldo, :ok_perc_saldo);

    if (ok_type not in (7, 8)) then
    begin
      if (ok_os_saldo = 0) then
        ok_os_saldo = -9990000;
      if (ok_perc_saldo = 0) then
        ok_perc_saldo = -9990000;
    end
--      нужно кол-во по типам вычислить, чтобы в следующем запросе выровнять
    if (ok_type = 1) then
      kol_ok_type1 = kol_ok_type1 + 1;
    if (ok_type = 2) then
      kol_ok_type2 = kol_ok_type2 + 1;
    if (ok_type = 3) then
      kol_ok_type3 = kol_ok_type3 + 1;
  end

  delta_os = 0;
  delta_perc = 0;
  k1 = 0;
  k2 = 0;
  k3 = 0;
  tc_amount2 = 0; -- сумма по ok_type = 2
  tc_amount3 = 0; -- сумма по ok_type = 3
  for select ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
             ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id,
             ok_ol_is_active, ok_os_is_last, ok_perc_is_last, ok_os_saldo, ok_perc_saldo, ok_fine_saldo,
             ok_kommis_saldo, ok_sh_os_saldo, ok_sh_perc_saldo, ok_kommis_adv_saldo, ok_tax_saldo, ok_os_total,
             ok_perc_total
      from temp_operkred3
      into :ok_from_ol_id, :ok_number, :ok_from_k_id, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine,
           :ok_pay_type, :ok_kommis, :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax, :ok_ol_date, :ok_ol_reg_id,
           :ok_ol_is_active, ok_os_is_last, ok_perc_is_last, ok_os_saldo2, ok_perc_saldo2, ok_fine_saldo2,
           ok_kommis_saldo2, ok_sh_os_saldo2, ok_sh_perc_saldo2, ok_kommis_adv_saldo2, ok_tax_saldo2, ok_os_total,
           ok_perc_total
  do
  begin
    if (ok_sh_date_prior <> ok_sh_date) then
    begin
      ok_sh_date_prior = ok_sh_date;
      delta_os = 0;
      delta_perc = 0;
    end

    ok_os_cli = round(:tc_cli_dolya * :ok_os, :round_coef);
    ok_perc_cli = round(:tc_cli_dolya * :ok_perc, :round_coef);
    ok_fine_cli = round(:tc_cli_dolya * :ok_fine, :round_coef);
    ok_kommis_cli = round(:tc_cli_dolya * :ok_kommis, :round_coef);
    ok_sh_os_cli = round(:tc_cli_dolya * :ok_sh_os, :round_coef);
    ok_sh_perc_cli = round(:tc_cli_dolya * :ok_sh_perc, :round_coef);
    ok_kommis_adv_cli = round(:tc_cli_dolya * :ok_kommis_adv, :round_coef);
    ok_tax_cli = round(:tc_cli_dolya * :ok_tax, :round_coef);

    if (ok_type not in (7, 8)) then
    begin
      if (ok_type = 1) then
      begin
        k1 = k1 + 1;
        if (k1 = kol_ok_type1) then
        begin
          ok_sh_os_cli = tc_amount;
        end
        else
        begin
          tc_amount = tc_amount - ok_sh_os_cli;
        end
      end
      if (ok_type = 2) then -- собирает все суммы по ok_type = 2 и на последнем минусует эту разницу так как сумма по ok_ype = 2 должно ровняться к нулюу
      begin
        k2 = k2 + 1;
        if (k2 = kol_ok_type2) then
        begin
          ok_sh_os_cli = -tc_amount2;
        end
        else
        begin
          tc_amount2 = tc_amount2 + ok_sh_os_cli;
        end
      end
      if (ok_type = 3) then -- собирает все суммы по ok_type = 3 и на последнем минусует эту разницу так как сумма по ok_ype = 3 должно ровняться к нулюу
      begin
        k3 = k3 + 1;
        if (k3 = kol_ok_type3) then
        begin
          ok_sh_os_cli = -tc_amount3;
        end
        else
        begin
          tc_amount3 = tc_amount3 + ok_sh_os_cli;
        end
      end
--             101010000002000306 нужно вот этот контракт проверить при ok_type = 2 сумма ok_os <> 0
--             после корректировки ok_sh_os_cli нужно вычислить tc_amount2, чтобы старый данный не брать
      delta_os = delta_os + ok_os_cli + ok_sh_os_cli;
      delta_perc = delta_perc + ok_perc_cli + ok_sh_perc_cli;
    end
--            если суммы ос + проценты = 0 то не должно иметь просрочку
    if (ok_os_total + ok_perc_total = 0 and
        ok_os_total <> 0) then
    begin
--              если есть разница в сумме клиента после округления, то выполнять следующее
      if (delta_os + delta_perc <> 0) then
      begin
        if (ok_os <> 0) then --было (ok_os_cli <> 0) изменен от 02.02.2015
        begin
          ok_os_cli = ok_os_cli - (delta_os + delta_perc);
          delta_os = delta_os - (delta_os + delta_perc);
        end
        else
        if (ok_perc <> 0) then -- было if (ok_perc_cli <> 0) then изменен от 28.01.2015
        begin
          ok_perc_cli = ok_perc_cli - (delta_os + delta_perc);
          delta_perc = delta_perc - (delta_os + delta_perc);
        end
        else
        if (ok_sh_os <> 0) then -- было (ok_sh_os_cli <> 0)  измеен от 02.02.2015
        begin
          ok_sh_os_cli = ok_sh_os_cli - (delta_os + delta_perc);
          delta_os = delta_os - (delta_os + delta_perc);
        end
        else
        if (ok_sh_perc <> 0) then -- было  ok_sh_perc_cli
        begin
          ok_sh_perc_cli = ok_sh_perc_cli - (delta_os + delta_perc);
          delta_perc = delta_perc - (delta_os + delta_perc);
        end
      end
    end

    if (ok_os_is_last = 1 and
        delta_os <> 0) then -- текущий момент тчательно протестировать
    begin
      if (ok_os <> 0) then -- было (ok_os_cli <> 0) изменен от 02.02.2015
      begin
        ok_os_cli = ok_os_cli - delta_os;
        delta_os = 0;
      end
    end

    if (ok_perc_is_last = 1 and
        delta_perc <> 0) then -- текущий момент тчательно протестировать
    begin
      if (ok_perc <> 0) then -- было (ok_perc_cli <> 0) изменен от 28.01.2015
      begin
        ok_perc_cli = ok_perc_cli - delta_perc;
        delta_perc = 0;
      end
      else
      if (ok_sh_perc <> 0) then -- было (ok_sh_perc_cli <> 0) изменен от 28.01.2015
      begin
        ok_sh_perc_cli = ok_sh_perc_cli - delta_perc;
        delta_perc = 0;
      end
    end

    suspend;
  end
end
