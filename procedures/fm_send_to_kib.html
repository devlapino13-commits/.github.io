CREATE PROCEDURE FM_SEND_TO_KIB(
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT,
    SHOW_RES D_STR_10)
AS
declare variable DB_ID D_INTEGER;
declare variable TMP_INT D_INTEGER;
declare variable CL_FIO_PRIOR D_STR_100;
declare variable CL_FIO D_STR_100;
declare variable TMP_STR_LONG D_STR_2048;
declare variable C_ID D_ID;
declare variable EDOC_STATUS D_STR_CODE;
begin
  res_text = '';
  show_res = 'window';

  db_id = f_get_db_id();

  tmp_int = 0;

  if (f_objkind(obj_kind, 'code') = 'miniapp') then
  begin

    if (db_id = 39801) then -- RFCKG-4210 Если согласия не подписаны / RFCKG-4486 Если нет кредитной истории
    begin
      if ((select priority_id
         from hp_get_priority_rep_id(f_objkind('miniapp', 'id'), :obj_id)) = -1) then
      begin
         res_text = 'У клиента отсутствует кредитная история';
         show_res = 'dialog';
         suspend;
         exit;
      end

      for select c.c_id, c.c_name_fio
      from miniapp ma
      inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
      inner join clients c on c.c_id = mac.mac_from_c_id
      where ma.ma_id = :obj_id and
            mac_is_poruch = 0 and
            ma_date_end >= '31.12.2999'
      into c_id, cl_fio
      do
      begin
        edoc_status =  f_edoc_status('pers_data_consent', :c_id, current_date);

        if (edoc_status <> 'signed') then
        begin
          res_text = cl_fio || ': Сначала клиент должен подписать согласия';
          show_res = 'dialog';
          suspend;
          exit;
        end
      end
    end

    if (db_id in (39801, 41701)) then
    begin
      res_text = (select cl_list from hp_clients_without_ds(:obj_id));
      if (res_text <> '') then
      begin
        show_res = 'dialog';
        suspend;
        exit;
      end
    end

    cl_fio_prior = '';
    for
      select
        res_client_fio,
        iif(res_is_critical = 1,
          iif(:db_id <> 94401, '<font color="red">критично</font>', '<font color="red">kritik</font>'),
            iif(res_is_critical = 2, iif(:db_id <> 94401, '<font color="red">очень критично</font>', '<font color="red">krtiki hədd</font>'), iif(:db_id <> 94401, 'сигнал', 'Siqnal'))) || ' -> ' || res_txt
      from hp_check_data_complete(:obj_kind, :obj_id)
      where res_is_critical > 0
      order by res_client_fio
      into
        :cl_fio,
        :tmp_str_long
    do
    begin
      tmp_int = tmp_int + 1;
      if (cl_fio <> cl_fio_prior) then
      begin
        res_text = BLOB_APPEND(res_text, iif(cl_fio_prior <> '', '<br><br>', '') || '<b>' || cl_fio || '</b>');
        cl_fio_prior = cl_fio;
      end
      res_text = BLOB_APPEND(res_text, '<br>' || tmp_str_long);
    end
  end

  if (tmp_int = 0)
    then res_text = '';
    else res_text = BLOB_APPEND(res_text, '</body>');

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  if (db_id = 39801 and
      f_objkind(obj_kind, 'code') = 'kontract') then -- RFCKG-4210 Если согласия не подписаны / RFCKG-4486 Если нет кредитной истории
    begin
      if ((select priority_id
         from hp_get_priority_rep_id(f_objkind('kontract', 'id'), :obj_id)) = -1) then
      begin
         res_text = 'У клиента отсутствует кредитная история';
         show_res = 'dialog';
         suspend;
         exit;
      end

      for select c.c_id, c.c_name_fio
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join clients c on c.c_id = kc.kc_from_c_id
      where k.k_id = :obj_id and
            kc.kc_amount_approved > 0
      into c_id, cl_fio
      do
      begin
        edoc_status =  f_edoc_status('pers_data_consent', :c_id, current_date);

        if (edoc_status <> 'signed') then
        begin
          res_text = cl_fio || ': Сначала клиент должен подписать согласия';
          show_res = 'dialog';
          suspend;
          exit;
        end
      end
    end

  select send_kol
  from insert_kibminiapp(0, :obj_id, :un_id)
  into tmp_int;

  if (tmp_int = 0)
    then res_text = iif(db_id <> 94401,
                          'Клиенты НЕ отправлены. Возможно они ранее уже отправлялись',
                          'Müştərilər göndərilmədi.Əvvəl göndərilib');
    else res_text = iif(db_id <> 94401,
                          'Данные успешно отправлены на проверку в КИБ. Количество отправленных клиентов: ',
                          'Məlumatlar MKR-də yoxlama üçün göndərildi. Göndərilən müştərilərin sayı: ') || tmp_int;
  suspend;
end
