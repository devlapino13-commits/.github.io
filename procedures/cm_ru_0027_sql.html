CREATE PROCEDURE CM_RU_0027_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kf_officer id_type;
begin
  for select kal.kal_new_kf_officer
  from kontract kont
  inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
  inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
  where k_id = :k_id
  and reg.reg_logic_label containing('.mega_office.')
  and kal.is_delete = 0
  and current_date between kal.kal_date and kal.kal_date_end
  and K_KOMMIS_LGOTA > 0
  into :kf_officer
  do
  begin
      if  (kf_officer <> 1000000000004) then
      begin
        res_error_text = 'Клиенты от промоутеров должны обслуживаться через "Экспертов-аналитиков"';
        res_error_is_critical = 1;
        suspend;
      end
  end
end
