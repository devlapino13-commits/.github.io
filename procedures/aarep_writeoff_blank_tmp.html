CREATE PROCEDURE AAREP_WRITEOFF_BLANK_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    K_ID ID_TYPE,
    K_MAIN ID_TYPE,
    WRITEOFF_OS CURR_TYPE,
    WRITEOFF_PERC CURR_TYPE,
    WRITEOFF_TOTAL CURR_TYPE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    K_KONTRACT_NO VARCHAR(50),
    K_DISB_DATE DATE)
AS
begin
  for select ok.ok_from_k_id, -sum(ok.ok_os), -sum(ok.ok_perc)
      from operkred ok, kontract kont
      where kont.k_id = ok.ok_from_k_id and
            ok.ok_ol_date = :rep_date and
            ok.ok_ol_is_active = 1 and
            ok.ok_type = 5 and
            ok.ok_from_c_id = 0 and
            ((ok.ok_os < 0) or
            (ok.ok_perc < 0)) and
            kont.k_last_reg_id in (select id_list
                                   from region_list(:rep_region_id, 1, 3, 1))
      group by ok_from_k_id
      order by ok_from_k_id
      into :k_id, :writeoff_os, -- списанные ОС
           :writeoff_perc -- списанные %
  do
  begin
    writeoff_total = writeoff_os + writeoff_perc;

    k_main = null;
    select kdv.kd_k_orig
    from kontdevides kdv
    where kdv.kd_k_new = :k_id
    into k_main;

    if (k_main is null) then
      k_main = k_id;

    select kont.k_kontract_no, ks.ks_date_begin
    from kontract kont
    inner join kontstate ks on ks.ks_from_k_id = kont.k_id
    where kont.k_id = :k_main and
          ks.ks_state = 1 and
          ks.ks_ol_is_delete = 0
    into :k_kontract_no, :k_disb_date;

    zad_os = null;
    zad_perc = null;
    select sum(zad_os), sum(sh_calc_perc)
    from aarep_make_zad_list(:k_id, :rep_date, 1)
    where sh_date is not null
    into :zad_os, :zad_perc;

    if (zad_os is null) then
      zad_os = 0;
    if (zad_perc is null) then
      zad_perc = 0;

    zad_os = zad_os + writeoff_os; -- Сумма задолженности по основному долгу по состоянию на rep_date  составляет
    zad_perc = zad_perc + writeoff_perc; -- задолженность по процентам

    suspend;
  end
end
