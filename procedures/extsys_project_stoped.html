CREATE PROCEDURE EXTSYS_PROJECT_STOPED(
    ESM_PROJECT D_STR_CODE)
RETURNS (
    IS_ACTIVE D_BOOL)
AS
declare variable DB_ID D_INTEGER;
begin
  is_active = 1;
  db_id = f_get_db_id();

  if (db_id in (39801, 41701)) then
  begin
      if (esm_project in ('Pay24', 'QuickPay', 'Keremet', 'KICB',
                          'Umai' /* ночью 20.10.24 включил Умай в стоп-лист согласно задче RFCKG-3171 (включил на основе обращения в вотсап, задача была создана на след день)*/,
                          'Companion', 'Grintelecom', 'kicb_paysoft',
                          'FINCA', 'Onoi', 'megapay', 'Optima')) then
    
        is_active = 0;
  end
  suspend;
end
