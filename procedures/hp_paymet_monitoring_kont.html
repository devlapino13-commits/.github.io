CREATE PROCEDURE HP_PAYMET_MONITORING_KONT(
    K_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_INN D_ID,
    K_KONTRACT_NO D_STR_255,
    K_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_INTEGER,
    K_PERIOD D_INTEGER,
    KS_STATE D_STR_10,
    MAX_OVERDUE_DAYS D_INTEGER,
    OVERDUE_PAYMENTS_COUNT D_INTEGER,
    SCORE_RATE D_INTEGER,
    RISK_LEVEL D_STR_10,
    RISK_DESCRIPTION D_STR_100,
    IS_NEW D_SMALLINT)
AS
declare variable c_id d_id;
declare variable kc_circle d_integer;
declare variable k_id_2 d_id;
declare variable max_k_disb_date d_date;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
                                      -- Если клиент старый то 0, если новый то
  select kc.kc_from_c_id,
         kc.kc_circle,
         cl.c_name_fio,
         cl.c_inn,
         k.k_disb_date
  from knclient kc
  inner join kontract k on k.k_id = kc.kc_from_k_id and k.k_id = :K_ID
  join clients cl on cl.c_id = kc.kc_from_c_id
  where kc.kc_amount_approved > 0
  into  :c_id,
        :kc_circle,
        :c_name_fio,
        :c_inn,
        :max_k_disb_date;

  is_new = iif(kc_circle > 1, 0, 1);

  if(is_new = 0) then
  begin
      K_ID_2 = 0;
      MAX_OVERDUE_DAYS = 0;
      OVERDUE_PAYMENTS_COUNT = 0;

      for select kt.k_id,
                 kt.k_kontract_no,
                 kt.k_disb_date,
                 kt.k_amount_approved,
                 kt.k_period,
                 iif(ks.ks_state = 1, 'Активный', 'Закрытый')
          from kontract kt
          inner join kontstate ks on ks.ks_from_k_id = kt.k_id
          inner join knclient kc on kc.kc_from_k_id = kt.k_id
          inner join clients cl on cl.c_id = kc.kc_from_c_id
          where current_date between ks.ks_date_begin and ks.ks_date_end
                and ks.ks_state in (1, 2)
                and ks.ks_ol_is_delete = 0
                and kt.k_amount_approved > 0
                and kc.kc_from_c_id = :c_id
                and (:db_id <> 41701 or (kt.k_disb_date <= :max_k_disb_date and kt.k_id <> :k_id)) -- RFCKG-4950
          order by kt.k_disb_date desc
          rows 5
          into :k_id_2,
               :k_kontract_no,
               :k_disb_date,
               :k_amount_approved,
               :k_period,
               :ks_state
      do
      begin
        -- Получаем максимальное количество просроченных дней
        select coalesce(max(ok.ok_ol_date - ok.ok_sh_date), 0),
               count(*)
        from operkred ok
        where ok.ok_ol_is_active = 1
              and ok.ok_ol_date <= current_date
              and ok.ok_ol_date > ok.ok_sh_date
              and ok.ok_type < 7
              and ok.ok_from_k_id = :k_id_2
              and ok_os + ok_perc < 0
              and ok_from_c_id = 0
        into :max_overdue_days,
             :overdue_payments_count;
        suspend;
      end
  end
  -----------------------ДЛЯ НОВЫХ КЛИЕНТОВ-----------------------

  else
  begin

      select kbm.kbm_score_rate
      from kibminiapp kbm
      where kbm.kbm_from_c_id = :c_id
      order by kbm.kbm_datetime desc
      rows 1
      into :score_rate;                                               

      risk_level = (case
                   when score_rate between 250 and 250 then 'E3'
                   when score_rate between 251 and 529 then 'E2'
                   when score_rate between 530 and 573 then 'E1'
                   when score_rate between 574 and 603 then 'D3'
                   when score_rate between 604 and 619 then 'D2'
                   when score_rate between 620 and 640 then 'D1'
                   when score_rate between 641 and 652 then 'C3'
                   when score_rate between 653 and 664 then 'C2'
                   when score_rate between 665 and 676 then 'C1'
                   when score_rate between 677 and 688 then 'B3'
                   when score_rate between 689 and 700 then 'B2'
                   when score_rate between 701 and 712 then 'B1'
                   when score_rate between 713 and 730 then 'A3'
                   when score_rate between 731 and 742 then 'A2'
                   when score_rate between 743 and 900 then 'A1'
                   else '--'
                 end);
    
         risk_description = (case
                               when risk_level in ('E3', 'E2', 'E1') then 'Очень высокий риск'
                               when risk_level in ('D3', 'D2', 'D1') then 'Высокий риск'
                               when risk_level in ('C3', 'C2', 'C1') then 'Средний риск'
                               when risk_level in ('B3', 'B2', 'B1') then 'Низкий риск'
                               when risk_level in ('A3', 'A2', 'A1') then 'Очень низкий риск'
                               else '--'
                             end);
          suspend;
  end
end
