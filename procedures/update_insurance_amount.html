CREATE PROCEDURE UPDATE_INSURANCE_AMOUNT(
    K_ID D_ID)
AS
declare variable K_AMOUNT_APPROVED D_CURRENCY;
declare variable K_DISB_DATE D_DATE;
declare variable K_KREDLINE_END D_DATE;
declare variable N_VALUE D_INTEGER;
declare variable PR_CODE D_STR_100;
declare variable TARIF PERC_TYPE;
declare variable TEMP_RES D_CURRENCY;
declare variable AG_COMMIS_CARDS_OTHER D_CURRENCY;
declare variable COMMISSION_AMOUNT D_CURRENCY;
declare variable MBULAK_SHARE D_CURRENCY;
declare variable POLICY_COMPANY_SHARE D_CURRENCY;
declare variable TAX D_CURRENCY;
declare variable K_SCORE D_INTEGER;
declare variable AG_ID D_ID;
declare variable C_ID D_ID;
declare variable PTS_DATE_BEGIN D_DATE;
declare variable PTS_DATE_END D_DATE;
declare variable MBULAK_BASE D_CURRENCY;
begin
    select
        k.k_amount_approved, -- сумма страхования (в контексте страхования конечно же)
        k.k_kredline_end - k.k_disb_date,
        pr.pr_code,
        k.k_disb_date,
        k.k_kredline_end
    from kontract k
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :k_id
    into
        :k_amount_approved,
        :n_value,
        :pr_code,
        :k_disb_date,
        :k_kredline_end;

    if (pr_code containing '.ignore_insurance.') then
      exit;

    /*
    Тариф был очень маленьким (0.00005479452), и при умножении на большие суммы возникало переполнение (Integer overflow).
    Чтобы избежать ошибки и сохранить точность, тариф увеличили на 100000,
    а в формуле расчёта комиссии добавили деление на 100000 для восстановления исходного масштаба.
    */
    --tarif = iif(f_get_const('new_loan_conditions') = 1, 0.0000548, 0.00005);
    tarif = iif(f_get_const('new_loan_conditions') = 1, 5.479452, 5);

    if (n_value <= 365) then
        n_value = 365;

    select kc.kc_from_c_id
    from knclient kc
    where kc.kc_from_k_id = :k_id
      and kc.kc_amount_approved > 0
    rows 1
    into :c_id;

    k_score = coalesce(f_point(:c_id, :k_disb_date, iif(:pr_code containing '.ipf.', 2, 0)), 0);

    if (:k_disb_date >= '12.03.2026') then
    begin
        if (pr_code containing '.sputnik.' or
            pr_code containing '.insur_base.') then
        begin
            tarif = case
                      when :k_score between 0 and 19 then 13.699000  -- 5%
                      when :k_score between 20 and 41 then 12.329000  -- 4.5%
                      else 10.958904                                 -- 4%
                    end;
        end
        else if (pr_code not containing '.consultation.'
                 and pr_code not containing '.ipf.'
                 and pr_code containing '.standart.'
                 and pr_code containing '.ind.') then
        begin
            tarif = case
                      when :k_score between 0 and 41 then 16.438000  -- 6%
                      else 13.699000                                 -- 5%
                    end;
        end
        else if (pr_code containing '.ipf.') then -- RFCKG-5348
        begin
            tarif = case
                      when :k_score between 0 and 19 then 19.178082 -- 7%

                      when :k_score between 20 and 30 and
                           :k_kredline_end <= dateadd(month, 6, :k_disb_date) then 10.958904 -- 4%

                      when :k_score between 20 and 30 then 16.438356 -- 6%

                      when :k_kredline_end <= dateadd(month, 6, :k_disb_date) then 8.219178 -- 3%

                      else 13.698630 -- 5%
                    end;
        end
        if (pr_code containing '.standart.' and
            pr_code containing '.point_ignore.' and
            pr_code containing '.insurance.') then
        begin
            tarif = 6.849315; -- 2.5%
            n_value = 730;
        end
    end
    else
    if (:k_disb_date >= '17.11.2025' and    -- RFCKG-5080
        :k_disb_date <= '12.03.2026') then
    begin
        if (:pr_code containing '.sputnik.') then
        begin
            :tarif = case
                        when :k_score between 0 and 19 then 13.699000 -- 0.05
                        else 10.958904 -- 0.04
                     end;
        end

        else if (:pr_code not containing '.consultation.'
                 and :pr_code not containing '.ipf.'
                 and :pr_code containing '.standart.'
                 and :pr_code containing '.ind.'
        ) then
        begin
            :tarif = case
                        when :k_score between 0 and 12 then 16.438000 -- 0.06
                        else 13.699000 -- 0.05
                     end;
        end
    end

    select
        pts_date_begin,
        pts_date_end
    from pointsaldo
    where pts_from_c_id = :c_id
      and pts_type = iif(:pr_code containing '.ipf.', 2, 0)
      and pts_date_begin <= :k_disb_date
    order by pts_from_c_id desc, pts_date_begin desc
    rows 1
    into :pts_date_begin, :pts_date_end;

    -- сумму комиссии
    --commission_amount = (k_amount_approved * tarif * n_value) * 1.02;
    temp_res = k_amount_approved * tarif;
    temp_res = temp_res * n_value;

    if (pr_code containing '.ipf.') then -- RFCKG-5348
    begin
        temp_res = round(temp_res / 100000, 2);

        -- НСП
        tax = round(temp_res * 0.02, 2);

        commission_amount = temp_res + tax;
    end
    else
    begin
        temp_res = temp_res * 1.02 / 100000;
        commission_amount = temp_res;

        -- НСП
        tax = commission_amount / 102 * 2;
    end

    select ag.ag_commis_cards_other, ag.ag_id
    from agreements ag
    inner join insurance_kontract ik on ik.ik_from_ag_id = ag.ag_id
    where ik.ik_from_k_id = :k_id
    rows  1
    into ag_commis_cards_other, ag_id;

    -- RFCKG-5348
    if (pr_code containing '.ipf.') then
      ag_commis_cards_other = case
                                when tarif <= 13.698630 then 90
                                when tarif <= 16.438356 then 92
                                else 93
                              end;
    else
    if (ag_id = 7414) then  -- ЗАО Страховая компания "Кыргызстан"
      ag_commis_cards_other = case when tarif <= 13.699000 then 90   -- RFCKG-4937
                                   when tarif <= 16.438356 then 92
                                   when tarif > 16.438356 then 93
                                   else 93
                                   end;
    else if (ag_id = 7130) then  -- ЗАО СК "Арсеналъ-Кыргызстан"
    begin
      ag_commis_cards_other = case
        when tarif <= 0.821918  then 30
        when tarif <= 1.095890  then 40
        when tarif <= 1.369863  then 50
        when tarif <= 1.643836  then 55
        when tarif <= 1.917808  then 60
        when tarif <= 2.191781  then 70
        when tarif <= 2.465753  then 75
        when tarif <= 13.424657 then 92
        else 93
      end;
    end

    -- доля Мбулак
    -- RFCKG-5348
    if (pr_code containing '.ipf.') then
        mbulak_base = (commission_amount - tax) * 0.90;
    else
        mbulak_base = commission_amount - tax;

    mbulak_share = mbulak_base * (ag_commis_cards_other / 100);     -- RFCKG-4803

    -- доля сраховой компании
    policy_company_share = commission_amount - mbulak_share;

    insert into insur_logs (
        il_from_k_id,
        il_tarif,
        il_k_score,
        il_pts_date_begin,
        il_pts_date_end,
        il_create_datetime
    )
    values (
        :k_id,
        :tarif,
        :k_score,
        :pts_date_begin,
        :pts_date_end,
        current_timestamp
    );

    update insurance_kontract ik
    set ik.ik_commission = :commission_amount,
        ik.ik_mbulak_share = :mbulak_share,
        ik.ik_policy_company_share = :policy_company_share,
        ik.ik_tax = :tax,
        ik.ik_is_checked = 1
    where ik.ik_from_k_id = :k_id;

end
