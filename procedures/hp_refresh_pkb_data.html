CREATE PROCEDURE HP_REFRESH_PKB_DATA(
    EIL_ID D_ID)
AS
declare variable is_gambler d_smallint;
declare variable is_bankrot d_smallint;
declare variable is_overdue d_smallint;
declare variable is_overdue_90 d_smallint;
declare variable payment_count d_integer;
declare variable json_text d_blob_text;
declare variable temp_str d_str_255;
declare variable job_title d_str_255;
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable un_id d_id;
declare variable fku_type d_integer;
declare variable curr_year d_integer;
declare variable curr_month d_integer;
declare variable from_date d_date;
declare variable from_year d_integer;
declare variable from_month d_integer;
declare variable contract_type d_str_code;
declare variable is_restructured d_smallint;
declare variable settlement_type_id d_str_255;
declare variable settlement_date d_str_255;
declare variable tmp_value d_str_20;
declare variable yy d_integer;
declare variable mm d_integer;
declare variable jso d_integer;
declare variable jso_root d_integer;
declare variable jsa_contracts d_integer;
declare variable jso_contract d_integer;
declare variable jsa_bankrupt_items d_integer;
declare variable jsa_settlement d_integer;
declare variable jso_settlement_item d_integer;
declare variable jsa_years d_integer;
declare variable jso_year d_integer;
declare variable jsa_payments d_integer;
declare variable jso_pay d_integer;
declare variable t_type d_str_20;
declare variable t_sub_type d_str_20;
declare variable t_year_type d_str_20;
declare variable i d_integer;
declare variable k d_integer;
declare variable y d_integer;
declare variable p d_integer;
declare variable cnt d_integer;
declare variable sub_cnt d_integer;
declare variable year_cnt d_integer;
declare variable pay_cnt d_integer;
declare variable ct_idx d_integer;
declare variable contract_path d_str_255;
declare variable query_count_90 d_integer;

declare function get_json_type(p_parent d_integer, p_path d_str_255) returns d_str_20
as
begin
  if (json.contains(p_parent, p_path) = 1) then
    return json.val_get_type_name(p_parent, p_path);
  return 'null';
end

declare function parse_date_ddmmyyyy(p_str d_str_255) returns date
as
begin
  if (p_str is not null and p_str <> '' and p_str <> '-' and char_length(p_str) = 10) then
    return cast(substring(p_str from 7 for 4) || '-' || substring(p_str from 4 for 2) || '-' || substring(p_str from 1 for 2) as date);
  return null;
end

begin
  is_overdue = 0;
  is_overdue_90 = 0;
  is_restructured = 0;
  is_gambler = 0;
  is_bankrot = 1;

  select cast(blx_blob as d_blob_text), ei.eil_from_c_id, ei.eil_from_un_id, ei.eil_object_id
  from extinfolist ei, hp_get_blobex_blob(ei.eil_from_bls_id)
  where ei.eil_id = :eil_id
  into :json_text, :c_id, un_id, ma_id;

  if (trim(json_text) = '') then
    exception e_default f_format('invalid eil_id: %1. no data found', :eil_id);

  jso = json.parse(json_text);
  jso_root = json.obj_get_obj(jso, 'Report.Root');

  if (jso_root is null) then
    exception e_default 'Не удалось десериализовать JSON (jso_root is null)';

  job_title = json.obj_get_val(jso_root, 'ClassificationOfBorrower.SubjectsPosition.value', 'Самозанятый');

  is_gambler = iif((json.obj_get_val(jso_root, 'Gamblers.GamblerFeature')) = 'true', 1, 0);

  query_count_90 = json.obj_get_val(jso_root, 'NumberOfQueries.Days90.value', 0);

  temp_str = json.obj_get_val(jso_root, 'TazalauBankruptInfo.Status.value', '');
  if (temp_str starting 'Данные о применении процедуры банкротства отсутствуют') then
    is_bankrot = 0;
  else
  begin
    if (temp_str = '') then
    begin
      t_type = get_json_type(jso_root, 'TazalauBankruptInfo.Item');

      if (t_type = 'Array') then
      begin
        jsa_bankrupt_items = json.obj_get_arr(jso_root, 'TazalauBankruptInfo.Item');
        cnt = json.count_get(jsa_bankrupt_items);
        if (cnt > 0) then
        begin
          -- забираем самый последний элемент массива
          jso_pay = json.arr_get_obj(jsa_bankrupt_items, cnt - 1);
          temp_str = json.obj_get_val(jso_pay, 'StatementStatus.value');
          if (coalesce(temp_str, '') not containing 'Завершено') then
            is_bankrot = 0;
        end
      end
      else if (t_type = 'Object') then
      begin
        temp_str = json.obj_get_val(jso_root, 'TazalauBankruptInfo.Item.StatementStatus.value', '');
        if (temp_str not containing 'Завершено') then
          is_bankrot = 0;
      end
    end
  end

  -- проверка реструктуризации в нашей бд
  select is_restructured
  from hp_is_restructured_kz(:c_id, 12)
  into :is_restructured;

  -- анализ действующих контрактов
  t_type = get_json_type(jso_root, 'ExistingContracts.Contract');

  -- выставляем границы и дескрипторы для универсального while-обхода
  if (t_type = 'Object') then
  begin
    cnt = 1;
    jso_contract = json.obj_get_obj(jso_root, 'ExistingContracts.Contract');
  end
  else if (t_type = 'Array') then
  begin
    jsa_contracts = json.obj_get_arr(jso_root, 'ExistingContracts.Contract');
    cnt = json.count_get(jsa_contracts);
  end
  else
    cnt = 0;

  i = 0;
  while (i < cnt) do
  begin
    if (t_type = 'Array') then
      jso_contract = json.arr_get_obj(jsa_contracts, i);

    if (jso_contract is not null) then
    begin
      payment_count = coalesce(nullif(json.obj_get_val(jso_contract, 'NumberOfOverdueInstalments.value'), ''), 0);  -- возможны пустые значения (<> null, = '')
      fku_type = coalesce(nullif(json.obj_get_val(jso_contract, 'CreditorType.id'), ''), 0);  -- возможны пустые значения (<> null, = '')


      if ((fku_type = 1 and payment_count > 30) or (fku_type = 2 and payment_count > 1)) then
        is_overdue = 1;

      -- проверка реструктуризации внутри текущего контракта
      if (coalesce(is_restructured, 0) = 0) then
      begin
        if (json.contains(jso_contract, 'SettlementIndicatorList.SettlementIndicator') = 1) then
          t_sub_type = json.val_get_type_name(jso_contract, 'SettlementIndicatorList.SettlementIndicator');
        else
          t_sub_type = 'null';

        if (t_sub_type = 'Array') then
        begin
          jsa_settlement = json.obj_get_arr(jso_contract, 'SettlementIndicatorList.SettlementIndicator');
          sub_cnt = json.count_get(jsa_settlement);
          k = 0;
          while (k < sub_cnt) do
          begin
            jso_settlement_item = json.arr_get_obj(jsa_settlement, k);
            settlement_type_id = json.obj_get_val(jso_settlement_item, 'SettlementIndicator.id');
            settlement_date = json.obj_get_val(jso_settlement_item, 'SettlementIndicatorDate.value');

            if (trim(settlement_type_id) = '1' and settlement_date <> '' and settlement_date <> '-' and char_length(settlement_date) = 10) then
            begin
              if (trim(settlement_type_id) = '1' and parse_date_ddmmyyyy(settlement_date) >= dateadd(-12 month to current_date)) then
              begin
                is_restructured = 1;
                leave; -- выход из внутреннего while по SettlementIndicator
              end
            end
            k = k + 1;
          end
        end
        else if (t_sub_type = 'Object') then
        begin
          settlement_type_id = json.obj_get_val(jso_contract, 'SettlementIndicatorList.SettlementIndicator.SettlementIndicator.id', '');
          settlement_date = json.obj_get_val(jso_contract, 'SettlementIndicatorList.SettlementIndicator.SettlementIndicatorDate.value', '');

          if (trim(settlement_type_id) = '1' and settlement_date <> '' and settlement_date <> '-' and char_length(settlement_date) = 10) then
          begin
          if (trim(settlement_type_id) = '1' and parse_date_ddmmyyyy(settlement_date) >= dateadd(-12 month to current_date)) then
            is_restructured = 1;
          end
        end
      end

      if (is_overdue = 1 and coalesce(is_restructured, 0) = 1) then
        leave; -- выход из главного WHILE по контрактам
    end

    i = i + 1;
  end


  -- расчет просрочки 90+ дней за последние 12 месяцев
  curr_year = extract(year from current_date);
  curr_month = extract(month from current_date);
  from_date = dateadd(-11 month to current_date);
  from_year = extract(year from from_date);
  from_month = extract(month from from_date);

  ct_idx = 1;
  while (ct_idx <= 2) do
  begin
    if (is_overdue_90 = 1) then leave;
    contract_type = iif(ct_idx = 1, 'TerminatedContracts', 'ExistingContracts');
    contract_path = contract_type || '.Contract';

    t_type = get_json_type(jso_root, contract_path);

    if (t_type = 'Object') then
    begin
      cnt = 1;
      jso_contract = json.obj_get_obj(jso_root, contract_path);
    end
    else if (t_type = 'Array') then
    begin
      jsa_contracts = json.obj_get_arr(jso_root, contract_path);
      cnt = json.count_get(jsa_contracts);
    end
    else
      cnt = 0;

    i = 0;
    while (i < cnt) do
    begin
      if (is_overdue_90 = 1) then leave;

      if (t_type = 'Array') then
        jso_contract = json.arr_get_obj(jsa_contracts, i);

      if (jso_contract is null) then
      begin
        i = i + 1;
        continue;
      end

      temp_str = json.obj_get_val(jso_contract, 'CreditorType.id', '');
      if (temp_str not in ('1', '2')) then
      begin
        i = i + 1;
        continue;
      end

      -- Фильтр закрытых договоров по дате окончания
      if (contract_type = 'TerminatedContracts') then
      begin
        temp_str = json.obj_get_val(jso_contract, 'DateOfCreditEnd.value', '');
        if (temp_str <> '' and temp_str <> '-' and char_length(temp_str) = 10) then
        begin
          if (cast(substring(temp_str from 7 for 4) || '-' || substring(temp_str from 4 for 2) || '-' || substring(temp_str from 1 for 2) as date) < :from_date) then
          begin
            i = i + 1;
            continue;
          end
        end
      end

      -- анализ календаря платежей PaymentsCalendar.Year
      if (json.contains(jso_contract, 'PaymentsCalendar.Year') = 1) then
        t_year_type = json.val_get_type_name(jso_contract, 'PaymentsCalendar.Year');
      else
        t_year_type = 'null';

      if (t_year_type = 'Object') then
      begin
        year_cnt = 1;
        jso_year = json.obj_get_obj(jso_contract, 'PaymentsCalendar.Year');
      end
      else if (t_year_type = 'Array') then
      begin
        jsa_years = json.obj_get_arr(jso_contract, 'PaymentsCalendar.Year');
        year_cnt = json.count_get(jsa_years);
      end
      else
        year_cnt = 0;

      -- итерируемся по массиву лет с конца в начало
      y = year_cnt - 1;
      while (y >= 0) do
      begin
        if (is_overdue_90 = 1) then leave;

        if (t_year_type = 'Array') then
          jso_year = json.arr_get_obj(jsa_years, y);

        if (jso_year is not null) then
        begin
          yy = json.obj_get_val(jso_year, 'title', 0);
          if (yy < from_year) then leave; -- если год уже меньше искомого диапазона, прекращаем обход лет

          if (yy between from_year and curr_year) then
          begin
            jsa_payments = json.obj_get_arr(jso_year, 'Payment');
            if (jsa_payments is not null) then
            begin
              pay_cnt = json.count_get(jsa_payments);
              p = 0;
              while (p < pay_cnt) do
              begin
                jso_pay = json.arr_get_obj(jsa_payments, p);
                if (jso_pay is not null) then
                begin
                  mm = json.obj_get_val(jso_pay, 'number', 0);

                  if ((yy = from_year and mm >= from_month) or (yy = curr_year and mm <= curr_month)) then
                  begin
                    tmp_value = json.obj_get_val(jso_pay, 'value', '');
                    if (tmp_value not in ('', '-', 'null') and cast(tmp_value as d_integer) > 90) then
                    begin
                      is_overdue_90 = 1;
                      leave;
                    end
                  end
                end
                p = p + 1;
              end
            end
          end
        end
        y = y - 1;
      end
      i = i + 1;
    end
    ct_idx = ct_idx + 1;
  end

  update or insert into negativ_kib_history (nkh_from_c_id, nkh_is_gambler, nkh_is_overdue, nkh_is_bankrot,
                                             nkh_last, nkh_from_un_id, nkh_from_eil_id, nkh_is_overdue_90, nkh_is_restructured, nkh_query_count_90)
  values (:c_id, :is_gambler, :is_overdue, :is_bankrot, 1, :un_id, :eil_id, :is_overdue_90, :is_restructured, :query_count_90)
  matching (nkh_from_c_id, nkh_last);

  insert into kib_job_info (kji_from_c_id, kji_from_ma_id, kji_job_title, kji_from_eil_id, kji_create_datetime)
  values (:c_id, :ma_id, :job_title, :eil_id, current_timestamp);
end
