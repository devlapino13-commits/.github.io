CREATE PROCEDURE AAREP_KFOLDER_HISTORY(
    KF_ID DECIMAL(18,0))
RETURNS (
    KA_ID DECIMAL(18,0),
    KA_DATE DATE,
    KA_FROM_REG_ID INTEGER,
    KA_AKT_NO VARCHAR(20),
    KA_MEMO VARCHAR(2048),
    KAL_FROM_KA_ID DECIMAL(18,0),
    KAL_NUMBER INTEGER,
    KAL_FROM_KF_ID DECIMAL(18,0),
    KAL_FROM_K_ID DECIMAL(18,0),
    KAL_DATE DATE,
    KAL_DATE_END DATE,
    KAL_NEW_KF_OFFICER DECIMAL(18,0),
    KAL_NEW_KF_NAME VARCHAR(100),
    KAL_NEW_REG_ID INTEGER,
    KAL_NEW_NO INTEGER,
    KAL_NEW_OFFICER_CODE VARCHAR(20),
    KA_REGION_NAME VARCHAR(100),
    KAL_REGION_NAME VARCHAR(100),
    KAL_OFFICE_FIO VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ka_id, ka_date, ka_from_reg_id, ka_akt_no, ka_memo, kal_from_ka_id, kal_number, kal_from_kf_id,
             kal_from_k_id, kal_date, kal_date_end, kal_new_kf_officer, kal_new_kf_name, kal_new_reg_id, kal_new_no,
             kal_new_officer_code, r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, c_name_fio
      from knaktlist
      left join(knakt left join region r1 on (ka_from_reg_id = r1.reg_unicode)) on (kal_from_ka_id = ka_id and
            knakt.is_delete = 0), region r2, clients
      where kal_from_kf_id = :kf_id and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = r2.reg_unicode and
            kal_new_kf_officer = c_id
      order by kal_date
      into :ka_id, :ka_date, :ka_from_reg_id, :ka_akt_no, :ka_memo, :kal_from_ka_id, :kal_number, :kal_from_kf_id,
           :kal_from_k_id, :kal_date, :kal_date_end, :kal_new_kf_officer, :kal_new_kf_name, :kal_new_reg_id,
           :kal_new_no, :kal_new_officer_code, :ka_region_name, :kal_region_name, :kal_office_fio
  do
  begin
    if (ka_id is null) then
    begin
      ka_from_reg_id = kal_new_reg_id;
      ka_date = kal_date;
      ka_memo = 'Создание кред.папки';
      ka_region_name = :kal_region_name;
    end
    suspend;
  end
end
