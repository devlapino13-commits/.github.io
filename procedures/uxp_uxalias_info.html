CREATE PROCEDURE UXP$UXALIAS_INFO(
    UA_UUID D_UUID,
    UA_UUID_CODE D_STR_100 = null)
RETURNS (
    UA_OWNER D_UUID,
    UXA_UUID D_UUID,
    UXA_NAME D_STR_100,
    UXA_KIND D_SMALLINT,
    UA_UXA_OWNER D_UUID,
    UA_UXA_OWNER_KIND D_INTEGER,
    UA_NAME D_STR_100,
    UA_DISABLE_ON_EDIT D_BOOL,
    UA_READONLY D_BOOL,
    UA_SAFE_MODE D_BOOL,
    UA_USR_ID_FIELD D_STR_CODE,
    UA_OWNER_KIND D_STR_CODE,
    UXA_MENU_KIND D_SMALLINT,
    UA_AUTO_REFRESH_SECONDS D_INTEGER,
    UA_TRIGGER_CODE D_STR_CODE,
    UA_PREBLANK_SAVE D_STR_CODE)
AS
declare variable ua_trigger_kind d_integer;
declare variable preblank_save d_integer;
begin
  -- Возвращает информацию об объекте через код действия (UxAction.Ua_Code)
  -- При тестировании возможно действия не созданы и не прикреплен меню
  -- Поэтому случай с fm_Mainmenu обрабатываем дополнительно

  if (coalesce(ua_uuid, '') = '') then
    ua_uuid = uxf$uniq(ua_uuid_code);

  if (coalesce(ua_uuid, '') = '') then
    exit;

  uxa_menu_kind = 0;
  if (uxf$code(ua_uuid, 'kind') <> 'uxaction') then
  begin
    if (uxf$code(ua_uuid, 'kind') <> 'uxalias') then
      exception e_default 'UUID must be action or alias uuid: ' || uuid_to_char(ua_uuid);

    uxa_uuid = ua_uuid;
    ua_uuid = null;

    select ua_uuid
    from ux$action
    where ua_owner = uxf$mainmenu() and
          ua_uxa_uuid = :uxa_uuid
    into ua_uuid;

    if (row_count = 0) then
      exception e_default 'Mainmenu action for alias not found';
  end

  ua_usr_id_field = '';
  uxa_kind = 0;
  ua_trigger_code = '';
  ua_preblank_save = '';
  ua_safe_mode = 0;

  select ua_owner, ua_uxa_uuid, uxa_kind, uxa_owner, uu_kind, ua_name, ua_auto_refresh_seconds,
         ua_usr_id_field, ua_disable_on_edit, ua_readonly, ua_safe_mode, uxa_menu_kind,
         ua_trigger_kind, ua_preblank_save
  from ux$action, ux$alias, ux$code, ux$uniq
  where ua_uuid = :ua_uuid and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = ux_uuid and
        uu_id = ux_uuid
  into ua_owner, uxa_uuid, uxa_kind, ua_uxa_owner, ua_uxa_owner_kind, ua_name, ua_auto_refresh_seconds,
       ua_usr_id_field, ua_disable_on_edit, ua_readonly, ua_safe_mode, uxa_menu_kind,
       ua_trigger_kind, preblank_save;

  if (ua_trigger_kind <> 0) then
    ua_trigger_code = uxf$enum_code('uxaction_trigger_kind', ua_trigger_kind);

  if (preblank_save <> 0) then
    ua_preblank_save = uxf$enum_code('uxaction_preblank_save', preblank_save);

  uxa_name = uxf$code(uxa_uuid, 'name');
  ua_owner_kind = uxf$code(ua_owner, 'kind');

  suspend;
end
