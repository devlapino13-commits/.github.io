CREATE PROCEDURE HP_EFFECT_RATE_KG(
    K_ID D_ID)
RETURNS (
    CALC_EFF PERC_TYPE)
AS
declare variable iter integer;
declare variable calculate_error decimal(18,10) = 0.00000001;
declare variable max_iter integer = 100;
declare variable a_sum double precision;
declare variable d_sum double precision;
declare variable y double precision;
declare variable curr_date date;
declare variable first_date date;
declare variable p double precision;
declare variable curr_amount double precision = 0;
declare variable days_in_year double precision = 365;
declare variable calc_eff_tmp double precision;
begin
  -- Расчет эффективной процентной ставки (Кыргызстан, и все БД)
  -- В Excel XIRR, ЧИСТВНДОХ

  calc_eff_tmp = 0.0;
  select sh_date
  from temp_schedule
  where sh_id = 0
  into first_date;
  if (first_date is null) then
  begin
    suspend;
    exit;
  end

  calc_eff_tmp = 0.1;
  iter = 0;
  while (iter < max_iter) do
  begin
    a_sum = 0;
    d_sum = 0;
    y = 1 + calc_eff_tmp;
    for select sh_date, sh_os
        from temp_schedule
        into curr_date, curr_amount
    do
    begin
      p = (curr_date - first_date) / days_in_year;
      a_sum = a_sum + curr_amount / power(y, p);
      d_sum = d_sum - p * curr_amount / power(y, p + 1);
    end

    if (abs(a_sum) < calculate_error) then
      break;
    if (abs(d_sum) < 0.0001) then
      exception e_default 'Cannot calc effective rate for contract: ' || k_id;
    y = y - a_sum / d_sum;
    calc_eff_tmp = y - 1;
    iter = iter + 1;
  end
  calc_eff = calc_eff_tmp * 100.0;
  suspend;
end
