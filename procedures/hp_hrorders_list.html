CREATE PROCEDURE HP_HRORDERS_LIST(
    REP_REGION_ID D_INTEGER,
    IN_C_ID D_ID,
    IN_FIND_TEXT D_STR_100,
    IN_ORDER_TYPE D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    IN_FIND_TYPE D_INTEGER)
RETURNS (
    ORDER_ID D_ID,
    ORDER_DATE D_DATE,
    ORDER_REG_NAME D_STR_CODE,
    ORDER_TEXT D_STR_LONG,
    ORDER_REASON D_STR_255,
    ORDER_TYPE_ID D_INTEGER,
    ORDER_TYPE_STR D_STR_100,
    ORDER_SUBTYPE_ID D_INTEGER,
    ORDER_SUBTYPE_STR D_STR_255,
    ORDER_NO D_STR_20,
    ORDER_REG_ID D_INTEGER,
    ORDER_SIGNER_ID D_INTEGER,
    ORDER_EDL_ID D_ID,
    PS_STATUS D_INTEGER,
    HO_FROM_UN_ID D_ID,
    UN_FIO D_STR_100,
    HRC_NAME D_STR_255,
    FIO_LIST D_STR_LONG,
    SAWT_IS_ORDER D_INTEGER,
    RES_EDOC_STATUS D_INTEGER,
    RES_SIGNER_FIO D_STR_100)
AS
declare variable un_id_prior d_id;
declare variable hrv_class d_str_code;
declare variable has_id d_id;
declare variable has_is_sign d_bool;
declare variable has_is_delete d_bool;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  for select order_id, fio_list,
             ho_date, reg_code || ' ' || reg_name, ho_text, left(ho_reason, 255), ho_order_type, ho_order_no,
             ho_from_reg_id, ho_status, ho_from_un_id, ho_signer_id, ho_from_edl_id,
             hrv_class, sawt_name, sawt_is_order, ho_order_subtype, saws_name
  from hr_orders, region, sa_worktype, sa_worksubtype,
       hp_hrorders_tmp(:rep_region_id, :in_c_id, :in_find_text, :in_order_type, :rep_date_begin, :rep_date_end, :in_find_type)
  where order_id = ho_id and
        ho_from_reg_id = reg_unicode and
        ho_order_type = sawt_id and
        ho_order_subtype = saws_id
  order by ho_date desc, ho_id desc
  into :order_id, :fio_list,
       :order_date, :order_reg_name, :order_text, :order_reason, :order_type_id, :order_no,
       :order_reg_id, :ps_status, :ho_from_un_id, :order_signer_id, :order_edl_id,
       hrv_class, order_type_str, sawt_is_order, order_subtype_id, order_subtype_str
  do
  begin
    res_edoc_status = null;

    if (order_edl_id is not null) then
    begin
      select edl_status
      from edoc_list
      where edl_id = :order_edl_id
      into res_edoc_status;
    end

    if (db_id = 41701 and
        order_type_id = 22) then -- мат.помощь, RFCKG-4382
    begin
      has_id = null;
      has_is_sign = null;
      has_is_delete = null;

      select has_id, has_is_sign, has_is_delete
      from hr_accountant_sign
      where has_ho_id = :order_id
      order by has_id desc
      rows 1
      into has_id, has_is_sign, has_is_delete;

      if (has_id is not null) then
      begin
        res_edoc_status = has_is_sign;
        if (has_is_delete = 1) then
          res_edoc_status = -1;
      end
    end

    res_signer_fio = '';
    if (coalesce(order_signer_id, 0) <> 0) then
    begin
      select c_name_fio
      from kontsignreg, kontsignmen, clients
      where ksr_id = :order_signer_id and
            ksr_from_ksm_id = ksm_id and
            ksm_from_c_id = c_id
      into res_signer_fio;
    end

    hrc_name = '';
    if (order_type_id = 6) then
    begin
      select hrc_name
      from hr_pstatus, hr_class
      where ps_from_ho_id = :order_id and
            ps_class = hrc_id
      rows 1
      into hrc_name;
    end

    if (coalesce(hrv_class, '') <> '') then
    begin
      select left(trim(coalesce(hrc_code, '') || '  ' || hrc_name), 255)
      from hr_class
      where hrc_id = :hrv_class
      into hrc_name;
    end

    if (un_id_prior is distinct from ho_from_un_id) then
    begin
      un_fio = '';
      select c_name_fio
      from username, clients
      where un_id = :ho_from_un_id and
            un_from_c_id = c_id
      into :un_fio;

      if (un_fio is null) then
        un_fio = '';

      un_id_prior = ho_from_un_id;
    end

    suspend;
  end
end
