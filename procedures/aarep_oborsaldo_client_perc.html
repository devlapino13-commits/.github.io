CREATE PROCEDURE AAREP_OBORSALDO_CLIENT_PERC(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ACC_BALANCE_NO VARCHAR(20),
    ACCOUNT VARCHAR(20),
    REG_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    BEG_DT CURR_TYPE,
    BEG_KT CURR_TYPE,
    OB_DT CURR_TYPE,
    OB_KT CURR_TYPE,
    END_DT CURR_TYPE,
    END_KT CURR_TYPE,
    C_ID_STR VARCHAR(20),
    ACC_NO VARCHAR(20))
AS
declare variable beg_dt_tmp curr_type;
declare variable ob_kt_tmp curr_type;
declare variable ob_dt_tmp curr_type;
declare variable end_dt_tmp curr_type;
declare variable k_id decimal(18,0);
declare variable kol_cli integer;
declare variable dolya double precision;
declare variable kc_from_c_id id_type;
declare variable kc_amount curr_type;
declare variable k_amount decimal(18,4);
declare variable k integer;
declare variable beg_dt_cli_last curr_type;
declare variable ob_kt_cli_last curr_type;
declare variable ob_dt_cli_last curr_type;
declare variable end_dt_cli_last curr_type;
declare variable is_resident_disb smallint;
declare variable kc_id id_type;
begin

  for select acc_balance_no, account, reg_name, k_id, k_kontract_no, k_amount, beg_dt, beg_kt, ob_kt, ob_dt, end_dt,
             end_kt, kol_cli
      from aarep_oborsaldo_kontract_perc(:rep_region_id, :rep_date_begin, :rep_date_end)
      into :acc_balance_no, :account, :reg_name, :k_id, :k_kontract_no, :k_amount, :beg_dt_tmp, :beg_kt, :ob_kt_tmp,
           :ob_dt_tmp, :end_dt_tmp, :end_kt, :kol_cli
  do
  begin
    k = 0;

    beg_dt_cli_last = beg_dt_tmp;
    ob_kt_cli_last = ob_kt_tmp;
    ob_dt_cli_last = ob_dt_tmp;
    end_dt_cli_last = end_dt_tmp;

    for select kc_id, kc_from_c_id, kc_amount_approved, 1 - kc_non_resident
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_id, kc_from_c_id, kc_amount, is_resident_disb
    do
    begin
      k = k + 1;
      if (k < kol_cli) then
      begin
        dolya = cast(kc_amount as double precision) / k_amount;
        beg_dt = round(dolya * beg_dt_tmp, 2);
        ob_kt = round(dolya * ob_kt_tmp, 2);
        ob_dt = round(dolya * ob_dt_tmp, 2);
        end_dt = round(dolya * end_dt_tmp, 2);

        beg_dt_cli_last = beg_dt_cli_last - beg_dt;
        ob_kt_cli_last = ob_kt_cli_last - ob_kt;
        ob_dt_cli_last = ob_dt_cli_last - ob_dt;
        end_dt_cli_last = end_dt_cli_last - end_dt;
      end
      else
      begin
        beg_dt = beg_dt_cli_last;
        ob_kt = ob_kt_cli_last;
        ob_dt = ob_dt_cli_last;
        end_dt = end_dt_cli_last;
      end

      select c_name_fio
      from clients
      where c_id = :kc_from_c_id
      into c_name_fio;
      c_id_str = 'n' || kc_from_c_id;
      acc_no = iif(is_resident_disb = 1, '488028100010', '492028100010') || lpad(kc_id, 8, '0');

      suspend;
    end
  end

end
