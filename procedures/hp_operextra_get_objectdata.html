CREATE PROCEDURE HP_OPEREXTRA_GET_OBJECTDATA(
    REP_RD_ID INTEGER,
    REP_OBJ_KIND VARCHAR(100),
    REP_OBJ_NO VARCHAR(100),
    REP_OBJ_ID DECIMAL(18,0),
    RES_AMOUNT DECIMAL(18,4),
    RES_DETAIL VARCHAR(255))
RETURNS (
    RES_CHECKED INTEGER,
    RES_OBJECT_KIND INTEGER,
    RES_OBJECT_ID DECIMAL(18,0),
    RES_FROM_REG_ID INTEGER,
    RES_ACCOUNT_DT VARCHAR(20),
    RES_ACCOUNT_KT VARCHAR(20),
    RES_STATUS INTEGER,
    RES_COMMENT VARCHAR(255))
AS
declare variable res_date  date;
declare variable oek_rd_id integer;
begin
  oek_rd_id = rep_rd_id;

  res_checked = 0;
  res_object_kind = 0;
  res_object_id = 0;
  res_from_reg_id = 0;
  res_account_dt = '';
  res_account_kt = '';
  res_status = -1;
  res_comment = 'Не найден';

  if (rep_obj_kind = 'Kontract') then
  begin
    res_object_kind = 9;

    if (coalesce(rep_obj_id, 0) <> 0) then
    begin
      select k_id, k_last_reg_id
      from kontract
      where k_id = :rep_obj_id
      into :res_object_id, res_from_reg_id;
    end
    else
    begin
      select k_id, k_last_reg_id
      from kontract
      where k_kontract_no = :rep_obj_no
      into :res_object_id, res_from_reg_id;
    end
    if (coalesce(res_from_reg_id, 0) <> 0) then
    begin
      res_date = null;
      select ox_ol_date
      from operextra
      where ox_from_rd_id = :oek_rd_id and
            ox_from_reg_id = :res_from_reg_id and
            ox_object_kind = :res_object_kind and
            ox_object_id = :res_object_id and
            ox_amount_plus = :res_amount and
            ox_detail = :res_detail and
            ox_ol_is_active = 1 rows 1
      into :res_date;

      if (res_date is not null) then
      begin
        res_comment = 'Ранее сохранен от ' || res_date;
      end
      else
      begin
        select reg_code || ' ' || reg_name
        from region
        where reg_unicode = :res_from_reg_id
        into :res_comment;
        res_status = 1;
        res_checked = 1;

        select res_account, res_acc_delinq
        from hp_make_kredit_acclist(0, :res_from_reg_id, -:rep_rd_id)
        into :res_account_dt, res_account_kt;
      end
    end
  end
  suspend;
end
