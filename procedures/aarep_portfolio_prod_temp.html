CREATE PROCEDURE AAREP_PORTFOLIO_PROD_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_OFFICE VARCHAR(100),
    AAREP_DISB_AMOUNT DECIMAL(18,4),
    AAREP_DISB_KOLICH INTEGER,
    AAREP_LOAN_TYPE_TEXT VARCHAR(100),
    AAREP_ACTIVE_PORT_BEGIN DECIMAL(18,4),
    AAREP_ACTIVE_PORT_END DECIMAL(18,4),
    AAREP_ACTIVE_CLIENT_BEGIN INTEGER,
    AAREP_ACTIVE_CLIENT_END INTEGER,
    AAREP_DELIQ_DAYS_BEGIN INTEGER,
    AAREP_DELIQ_BALANCE_BEGIN DECIMAL(18,4),
    AAREP_DELIQ_DAYS_END INTEGER,
    AAREP_DELIQ_BALANCE_END DECIMAL(18,4))
AS
declare variable pr_loan_type integer;
declare variable aarep_k_id decimal(18,0);
declare variable paral integer;
declare variable emp integer;
declare variable curr varchar(3);
declare variable curr_curr varchar(3);
begin
  select cm_default_currency
  from company
  into :curr_curr;

  /* Активные на начало */
  aarep_disb_amount = 0;
  aarep_disb_kolich = 0;
  aarep_active_port_begin = 0;
  aarep_active_port_end = 0;
  aarep_active_client_begin = 0;
  aarep_active_client_end = 0;
  aarep_deliq_days_begin = 0;
  aarep_deliq_balance_begin = 0;
  aarep_deliq_days_end = 0;
  aarep_deliq_balance_end = 0;

  for select aarep_k_id, aarep_k_balance, aarep_kolich_active, pr_loan_type, pr_is_employer, pr_category,
             reg_code || ' ' || reg_name, k_index_curr_code
      from aarep_kontract_id(:report_region_id, :report_date_begin - 1, :report_date_begin - 1, :report_reverse, :report_currency, 3), kontract, product, region
      where k_id = aarep_k_id and
            k_from_pr_id = pr_id and
            aarep_region_id = reg_unicode
      into :aarep_k_id, :aarep_active_port_begin, :aarep_active_client_begin, :pr_loan_type, :emp, :paral,
           :aarep_office, :curr
  do
  begin
    if (curr = :curr_curr) then
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'A Групповой';
      if (paral = 1) then
        aarep_loan_type_text = 'C Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'E Индивидуальный';
      if (emp = 1) then
        aarep_loan_type_text = 'G Кред.сотрудникам';
    end
    else
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'B Индекс.Групповой';
      if (paral = 1) then
        aarep_loan_type_text = 'D Индекс.Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'F Индекс.Индивидуальный';
      if (emp = 1) then
        aarep_loan_type_text = 'H Индекс.Кред.сотрудникам';
    end
    suspend;
  end

  /* Активные на конец */
  aarep_disb_amount = 0;
  aarep_disb_kolich = 0;
  aarep_active_port_begin = 0;
  aarep_active_port_end = 0;
  aarep_active_client_begin = 0;
  aarep_active_client_end = 0;
  aarep_deliq_days_begin = 0;
  aarep_deliq_balance_begin = 0;
  aarep_deliq_days_end = 0;
  aarep_deliq_balance_end = 0;

  for select aarep_k_id, aarep_k_balance, aarep_kolich_active, pr_loan_type, pr_is_employer, pr_category,
             reg_code || ' ' || reg_name, k_index_curr_code
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 3), kontract, product, region
      where k_id = aarep_k_id and
            k_from_pr_id = pr_id and
            aarep_region_id = reg_unicode
      into :aarep_k_id, :aarep_active_port_end, :aarep_active_client_end, :pr_loan_type, :emp, :paral, :aarep_office,
           :curr
  do
  begin
    if (curr = :curr_curr) then
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'A Групповой';
      if (paral = 1) then
        aarep_loan_type_text = 'C Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'E Индивидуальный';
      if (emp = 1) then
        aarep_loan_type_text = 'G Кред.сотрудникам';
    end
    else
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'B Индекс.Групповой';
      if (paral = 1) then
        aarep_loan_type_text = 'D Индекс.Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'F Индекс.Индивидуальный';
      if (emp = 1) then
        aarep_loan_type_text = 'H Индекс.Кред.сотрудникам';
    end
    suspend;
  end

  /* Просроченные на начало */
  aarep_disb_amount = 0;
  aarep_disb_kolich = 0;
  aarep_active_port_begin = 0;
  aarep_active_port_end = 0;
  aarep_active_client_begin = 0;
  aarep_active_client_end = 0;
  aarep_deliq_days_begin = 0;
  aarep_deliq_balance_begin = 0;
  aarep_deliq_days_end = 0;
  aarep_deliq_balance_end = 0;

  for select aarep_k_id, aarep_k_balance, aarep_k_deliq_days, pr_loan_type, pr_is_employer, pr_category,
             reg_code || ' ' || reg_name, k_index_curr_code
      from aarep_kontract_id(:report_region_id, :report_date_begin - 1, :report_date_begin - 1, :report_reverse, :report_currency, 5), kontract, product, region
      where k_id = aarep_k_id and
            k_from_pr_id = pr_id and
            aarep_region_id = reg_unicode
      into :aarep_k_id, :aarep_deliq_balance_begin, :aarep_deliq_days_begin, :pr_loan_type, :emp, :paral, :aarep_office,
           :curr
  do
  begin
    if (curr = :curr_curr) then
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'A Групповой';
      if (paral = 1) then
        aarep_loan_type_text = 'C Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'E Индивидуальный';
      if (emp = 1) then
        aarep_loan_type_text = 'G Кред.сотрудникам';
    end
    else
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'B Индекс.Групповой';
      if (paral = 1) then
        aarep_loan_type_text = 'D Индекс.Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'F Индекс.Индивидуальный';
      if (emp = 1) then
        aarep_loan_type_text = 'H Индекс.Кред.сотрудникам';
    end
    suspend;
  end

  /* Просроченные на конец */
  aarep_disb_amount = 0;
  aarep_disb_kolich = 0;
  aarep_active_port_begin = 0;
  aarep_active_port_end = 0;
  aarep_active_client_begin = 0;
  aarep_active_client_end = 0;
  aarep_deliq_days_begin = 0;
  aarep_deliq_balance_begin = 0;
  aarep_deliq_days_end = 0;
  aarep_deliq_balance_end = 0;

  for select aarep_k_id, aarep_k_balance, aarep_k_deliq_days, pr_loan_type, pr_is_employer, pr_category,
             reg_code || ' ' || reg_name, k_index_curr_code
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 5), kontract, product, region
      where k_id = aarep_k_id and
            k_from_pr_id = pr_id and
            aarep_region_id = reg_unicode
      into :aarep_k_id, :aarep_deliq_balance_end, :aarep_deliq_days_end, :pr_loan_type, :emp, :paral, :aarep_office,
           :curr
  do
  begin
    if (curr = :curr_curr) then
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'A Групповой';
      if (paral = 1) then
        aarep_loan_type_text = 'C Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'E Индивидуальный';
      if (emp = 1) then
        aarep_loan_type_text = 'G Кред.сотрудникам';
    end
    else
    begin
      if ((pr_loan_type = 0) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'B Индекс.Групповой';
      if (paral = 1) then
        aarep_loan_type_text = 'D Индекс.Параллельный';
      if ((pr_loan_type = 1) and
          (emp = 0) and
          (paral = 0)) then
        aarep_loan_type_text = 'F Индекс.Индивидуальный';
      if (emp = 1) then
        aarep_loan_type_text = 'H Индекс.Кред.сотрудникам';
    end
    suspend;
  end

  suspend;
