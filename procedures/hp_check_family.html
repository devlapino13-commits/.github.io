CREATE PROCEDURE HP_CHECK_FAMILY(
    C_ID D_ID,
    RES_QUESTION_ID D_INTEGER = 0)
RETURNS (
    ACCESS D_SMALLINT)
AS
declare variable bls_id d_id;
declare variable family_status d_integer;
declare variable marital_status d_str_100;
declare variable json_text d_blob_text;
begin
  access = 0;
  select eil.eil_from_bls_id
  from extinfolist eil
  where eil.eil_from_c_id = :c_id and
        eil.eil_from_eic_id = 2 and
        eil.eil_is_json = 1 and
        eil.eil_status = 2
  order by eil.eil_id desc
  rows 1
  into bls_id;

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into json_text;

  family_status = f_json_str_get_value(:json_text, 'familyStatus', -1);
  marital_status = f_json_str_get_value(:json_text, 'maritalStatus', '');

  if (f_get_rd_code(:res_question_id) = 'AnketaQuestions.Family') then
  begin
    if (coalesce(family_status, -1) = -1 and
        coalesce(marital_status, '') = '') then
      access = 1;
    suspend;
    exit;
  end
  if (f_get_rd_code(:res_question_id) = 'AnketaQuestions.Family_tunduk') then
  begin
    if (coalesce(marital_status, '') = '') then
      access = 1;
    suspend;
    exit;
  end
end
