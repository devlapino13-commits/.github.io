CREATE PROCEDURE HP_DO_CHECK_MONEY_ACC(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    AL_ID D_STR_CODE,
    ACC_ID D_ID,
    ACC_CODE D_STR_CODE,
    OL_ID D_ID,
    OB_FROM_OL_ID D_ID,
    OB_SUMMA_DT D_CURRENCY,
    OB_SUMMA_KT D_CURRENCY,
    KOB_SUMMA_DT D_CURRENCY,
    KOB_SUMMA_KT D_CURRENCY)
AS
declare variable dd d_date;
declare variable ob_dt d_currency;
declare variable ob_kt d_currency;
declare variable ob_saldo d_currency;
declare variable kob_dt d_currency;
declare variable kob_kt d_currency;
declare variable kob_saldo d_currency;
begin
  for select al_id, al_from_acc_id
      from acc_list
      where al_from_bg_id in ('50-01', '51-01', '52-02', '55-04')
      into :al_id, acc_id
  do
  begin
    acc_code = f_fmt_acc(acc_id);
    dd = rep_date_begin;

    while (dd <= rep_date_end) do
    begin
      ob_dt = 0;
      ob_kt = 0;
      ob_saldo = 0;

      kob_dt = 0;
      kob_kt = 0;
      kob_saldo = 0;

      select iif(obs_date = :dd, obs_dt_ekv, 0),
             iif(obs_date = :dd, obs_kt_ekv, 0),
             obs_saldo_ekv
      from obsummary
      where obs_from_al_id = :al_id and
            obs_date <= :dd
      order by obs_from_al_id desc, obs_date desc
      rows 1
      into ob_dt, ob_kt, ob_saldo;

      if (acc_id is not null) then
      begin
        select iif(kobs_date = :dd, kobs_dt_sum, 0),
               iif(kobs_date = :dd, kobs_kt_sum, 0),
               kobs_bal_sum
        from kn_obsummary
        where kobs_acc_id = :acc_id and
              kobs_date <= :dd
        order by kobs_acc_id desc, kobs_date desc
        rows 1
        into kob_dt, kob_kt, kob_saldo;
      end

      if (ob_dt <> kob_dt or ob_kt <> kob_kt or ob_saldo <> kob_saldo) then
      begin
        -- Проверка на уровне операций

        for select ob_from_ol_id, ol_parent_id, sum(iif(ob_account_dt = :al_id, ob_summa, 0)),
                   sum(iif(ob_account_kt = :al_id, ob_summa, 0))
            from operbook, operlist
            where ob_ol_date = :dd and
                  ob_ol_is_active = 1 and
                  ob_from_ol_id = ol_id and
                  (ob_account_dt = :al_id or
                  ob_account_kt = :al_id)
            group by 1, 2
            into :ob_from_ol_id, :ol_id, ob_summa_dt, ob_summa_kt
        do
        begin
          kob_summa_dt = 0;
          kob_summa_kt = 0;

          if (ol_id is null) then
            ol_id = ob_from_ol_id;

          if (acc_id is not null) then
          begin
            select sum(iif(kob_acc_dt = :acc_id, kob_summa, 0)), sum(iif(kob_acc_kt = :acc_id, kob_summa, 0))
            from kn_operbook
            where kob_from_ol_id = :ob_from_ol_id and
                  (kob_acc_dt + 0 = :acc_id or
                  kob_acc_kt + 0 = :acc_id)
            into kob_summa_dt, kob_summa_kt;

            if (kob_summa_dt is null) then
              kob_summa_dt = 0;
            if (kob_summa_kt is null) then
              kob_summa_kt = 0;
            if (kob_summa_dt = 0 and
                kob_summa_kt = 0) then
            begin
              kob_summa_dt = 0;
              kob_summa_kt = 0;

              select sum(iif(kob_acc_dt = :acc_id, kob_summa, 0)), sum(iif(kob_acc_kt = :acc_id, kob_summa, 0))
              from kn_operbook
              where kob_from_ol_id = :ol_id and
                    (kob_acc_dt + 0 = :acc_id or
                    kob_acc_kt + 0 = :acc_id)
              into kob_summa_dt, kob_summa_kt;
              if (kob_summa_dt is null) then
                kob_summa_dt = 0;
              if (kob_summa_kt is null) then
                kob_summa_kt = 0;
            end
          end
          if (ob_summa_dt <> kob_summa_dt or ob_summa_kt <> kob_summa_kt) then
            suspend;
        end

        -- конец проверки на уровне операций
      end

      dd = dd + 1;
    end
  end
end
