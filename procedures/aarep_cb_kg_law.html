CREATE PROCEDURE AAREP_CB_KG_LAW(
    REP_DATE_END D_DATE)
RETURNS (
    SBL_NAME D_STR_255,
    K_KONTRACT_NO D_STR_20,
    PAY_DATE D_STR_255,
    PAY_SUM D_CURRENCY,
    PAY_TYPE D_STR_CODE)
AS
declare variable sbl_id d_integer;
declare variable k_id d_id;
declare variable pay_active d_currency;
declare variable pay_close d_currency;
declare variable is_court d_bool;
declare variable rep_date_begin d_date;
begin
  -- данные отчета "Расшифровка бонусов юр. отдела" (RFCKG-2577)

  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  for select sbl_id,
             k_id,
             repay_active,
             repay_close,
             is_court
      from cb_kg_oriv_temp(:rep_date_end)
      into sbl_id,
           k_id,
           pay_active,
           pay_close,
           is_court
  do
  begin
    select sbl_name
    from sb_list
    where sbl_id = :sbl_id
    into sbl_name;

    select k_kontract_no
    from kontract
    where k_id = :k_id
    into k_kontract_no;

    pay_date = null;
    pay_sum = null;
    pay_type = null;

    if (pay_active > 0) then
    begin
      -- активные контракты
      pay_sum = pay_active;
      pay_type = iif(is_court = 1, 'Погаш. активные (через суд)', 'Погаш. активные (досудеб)');

      with t as (select distinct ok_ol_date
                 from operkred
                 where ok_from_k_id = :k_id and
                       ok_ol_date between :rep_date_begin and :rep_date_end and
                       ok_ol_is_active = 1 and
                       ok_from_c_id = 0 and
                       ok_type = 4
                 order by ok_ol_date)
      select left(list(f_datetostr(ok_ol_date), ', '), 255)
      from t
      into pay_date;
    end
    else
    begin
      -- контракты на внесистемке
      pay_sum = pay_close;
      pay_type = iif(is_court = 1, 'Погаш. внесистем (через суд)', 'Погаш. внесистем (досудеб)');

      with t as (select distinct oko_ol_date
                 from operkroff
                 where oko_from_k_id = :k_id and
                       oko_ol_date between :rep_date_begin and :rep_date_end and
                       oko_ol_is_active = 1 and
                       oko_type = 4
                 order by oko_ol_date)
      select left(list(f_datetostr(oko_ol_date), ', '), 255)
      from t
      into pay_date;
    end

    suspend;
  end
end
