CREATE PROCEDURE HP_MOR_SAVE_ONE_CLI(
    REP_C_ID ID_TYPE)
RETURNS (
    IS_SAVE SMALLINT,
    C_MEMO VARCHAR(1024))
AS
declare variable c_pass_no varchar(30);
declare variable c_full_address varchar(255);
declare variable c_birth_date date;
declare variable c_from_reg_id integer;
declare variable k_kontract_no varchar(20);
declare variable k_disb_date date;
declare variable k_amount_approved decimal(18,4);
declare variable kc_amount_approved decimal(18,4);
declare variable ps_date_begin date;
declare variable reg_name varchar(50);
declare variable c_memo1 varchar(1024);
declare variable ol_user_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable c_name_fio varchar(100);
declare variable ps_is_working integer;
begin
  for select c_name_fio, c_pass_no, adl_address_str, c_birth_date, c_from_reg_id
      from clients, addrclient, addrlist
      where c_id = :rep_c_id and
            c_type = 0 and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      into c_name_fio, c_pass_no, c_full_address, c_birth_date, c_from_reg_id
  do
  begin
    is_save = 0;
    k_kontract_no = null;
    k_disb_date = null;
    k_amount_approved = null;
    kc_amount_approved = null;
    select distinct k_kontract_no, k_disb_date, k_amount_approved, kc_amount_approved, ol_user_id, k_id
    from knclient, kontstate, operkred, kontract, operlist
    where kc_from_c_id = :rep_c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = ks_from_k_id and
          kc_from_k_id = ok_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 2 and
          current_date between ks_date_begin and ks_date_end and
          ks_from_ol_id = ok_from_ol_id and
          ok_type = 5 and
          k_id = ks_from_k_id and
          operlist.is_delete = 0 and
          ol_id = ks_from_ol_id and
          ok_from_c_id = 0
    order by ks_date_begin desc
    rows 1
    into k_kontract_no, k_disb_date, k_amount_approved, kc_amount_approved, ol_user_id, k_id;
    c_memo = null;
    if (k_kontract_no is not null) then
    begin
      c_memo = 'Контракт №' || :k_kontract_no || ' закрыт со списанием, дата выдачи ' || :k_disb_date || ', сумма клиента ' || cast(:kc_amount_approved as decimal(15,2));
      is_save = 1;
    end
    ps_date_begin = null;
    reg_name = null;
    c_memo1 = null;

    select ps_date_begin, reg_code || ' ' || reg_name, ho_from_un_id, ps_is_working
    from hr_pstatus, region, hr_orders
    where ps_from_c_id = :rep_c_id
--    and current_date + 1 between ps_date_begin and ps_date_end
--    and ps_is_working = 0
          and
          ps_status = 1 and
          ps_from_reg_id = reg_unicode and
          ps_from_ho_id = ho_id
    order by ps_date_begin desc
    rows 1
    into ps_date_begin, reg_name, ol_user_id, ps_is_working;

    if (ps_is_working = 0) then
    begin
      c_memo1 = 'Сотрудник был уволен, из офиса - ' || :reg_name || '; дата увольнения - ' || :ps_date_begin;
      is_save = 1;
    end
    if (is_save = 1) then
    begin
      if (exists(select 1
                 from moratory
                 where (m_from_c_id = :rep_c_id or
                       (m_client_fio = :c_name_fio and
                       m_client_birth_date = :c_birth_date)) and
                       m_is_active = 1 and
                       m_is_black = 2 rows 1)) then
      begin
        is_save = 0;
        c_memo = 'Контрагент ' || :c_name_fio || ' в стоп листе уже существовал.';
      end
      else
      begin
        c_memo = coalesce(:c_memo || '; ', '') || coalesce(:c_memo1, '');
      end
      if (is_save = 1) then
      begin
        insert into moratory (m_id, m_number, m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end, m_month,
                              m_is_active, m_comment_group, m_comment_officer, m_comment_manager, m_client_fio,
                              m_client_pass_no, m_client_address, m_client_birth_date, m_our_clients, m_from_reg_id,
                              m_no, m_datetime, m_datetime_edit, m_user_create, m_user_edit, m_db_code, m_original_id)
        values (-255, 1, :rep_c_id, :k_id, 2, current_date, current_date, 0, 1, null, :c_memo, null, :c_name_fio,
                :c_pass_no, :c_full_address, :c_birth_date, 1, :c_from_reg_id, null, current_timestamp,
                current_timestamp, :ol_user_id, :ol_user_id, null, null);
      end
    end
    else
    begin
      c_memo = 'Не совпадает критерии контрагента в стоп лист, клиент: ' || :c_name_fio;
    end
    suspend;
  end
end
