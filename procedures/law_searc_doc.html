CREATE PROCEDURE LAW_SEARC_DOC(
    LAF_ID INTEGER,
    LAD_DATE_BEGIN DATE,
    LAD_DATE_END DATE,
    DELO_NO VARCHAR(255),
    KONT_NO VARCHAR(255),
    CL_FIO VARCHAR(255))
RETURNS (
    BLS_ID D_ID,
    DOC_EXIST D_INTEGER,
    LAM_ID INTEGER,
    LAM_DELO_NO VARCHAR(255),
    PARENT_DELO_NO VARCHAR(255),
    LAM_DATE_BEGIN DATE,
    LAWER_FIO VARCHAR(255),
    MANAGER_FIO VARCHAR(255),
    LAM_COST_ALL CURR_TYPE,
    LAM_CLIENTS_COUNT INTEGER,
    LAL_NAME VARCHAR(255),
    LAM_FINAL_RESULT_STR VARCHAR(255),
    LAM_FINAL_RESULT_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    KF_NAME VARCHAR(30),
    KS_STATE_STR VARCHAR(20),
    K_ID DECIMAL(18,0),
    LAD_LAF_NAME VARCHAR(255),
    LAD_DATE DATE,
    LAD_NEXT_LAF_NAME VARCHAR(255),
    LAD_NEXT_DATE DATE,
    PAYMASTER_FIO D_STR_100,
    LAM_MEMO D_STR_255,
    LAM_DATE_RESULT D_DATE,
    PAYED_SUMM DECIMAL(18,2),
    DT_CANCEL D_DATETIME)
AS
declare variable sql_text_1      varchar(5000);
declare variable ks_state        integer;
declare variable laf_order_no    integer;
declare variable laf_curr_id     integer;
declare variable lam_lawer_id    decimal(18,0);
declare variable lam_manager_id  decimal(18,0);
declare variable lam_from_lal_id integer;
declare variable c_id            d_id;
declare variable payed_over_sum  d_currency;
declare variable db_id           d_integer;
declare variable un_id           d_id;
declare variable reg_id          d_integer;
declare variable reg_lst         d_str_8100;
declare variable dt              d_date;
declare variable tm              d_time;
begin
  db_id = f_get_db_id();
  if (db_id = 94401) then
  begin
    un_id = (
      select un_id
      from username
      where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN'));
    reg_lst = (
      select list(reg_unicode, ';')
      from aarep_region_user_list(:un_id));
  end

  parent_delo_no = '';
  lad_date_begin = coalesce(lad_date_begin, cast('01.01.1899' as date));
  lad_date_end = coalesce(lad_date_end, cast('01.01.1899' as date));

  delo_no = trim(coalesce(delo_no, ''));
  kont_no = trim(coalesce(kont_no, ''));
  cl_fio = trim(coalesce(cl_fio, ''));

  if (laf_id is null) then
    laf_id = 0;

  sql_text_1 = 'select lm.lam_id,
                       lm.lam_delo_no,
                       lm.lam_date_begin,
                       lam_lawer_id,
                       lam_manager_id,
                       lm.lam_cost_all,
                       lm.lam_clients_count,
                       lam_from_lal_id,
                       lm.LAM_FINAL_RESULT_STR,
                       lm.LAM_FINAL_RESULT_DATE,
                       lm.lam_memo,
                       lm.lam_date_result
                from lawmain lm ';

  if (laf_id <> 0) then
  begin
    sql_text_1 = sql_text_1 || ' inner join lawdata ld on ld.lad_from_lam_id  = lm.lam_id and ld.lad_from_laf_id = ' || :laf_id || ' and ld.lad_date between ''' || :lad_date_begin || ''' and ''' || :lad_date_end || ''' ';
  end
  if ((kont_no <> '') or (cl_fio <> '')) then
  begin
    sql_text_1 = sql_text_1 || ' inner join lawkredit lk on lk.lak_from_lam_id = lm.lam_id ';

    if (kont_no <> '') then
      sql_text_1 = sql_text_1 || ' inner join kontract kont on lk.lak_from_k_id = kont.k_id and kont.k_kontract_no = ''' || :kont_no || ''' ';

    if (cl_fio <> '') then
      sql_text_1 = sql_text_1 || ' inner join clients cl on cl.c_id = lk.lak_from_c_id and cl.c_name_fio starting ''' || :cl_fio || ''' ';
  end
  sql_text_1 = sql_text_1 || ' where lm.lam_is_active = 1 ';
  if (delo_no <> '') then
  begin
    sql_text_1 = sql_text_1 || ' and lm.lam_delo_no = ''' || delo_no || ''' ';
  end
  sql_text_1 = sql_text_1 || ' group by lam_id, lam_delo_no, lam_date_begin, lam_lawer_id,  lam_manager_id, lam_cost_all, lam_clients_count, lam_from_lal_id, lam_final_result_str, lam_final_result_date, lam_memo, lm.lam_date_result ';

  for execute statement(:sql_text_1)
          with caller privileges
          into :lam_id, :lam_delo_no, :lam_date_begin, :lam_lawer_id, :lam_manager_id, :lam_cost_all,
          :lam_clients_count, :lam_from_lal_id, :lam_final_result_str, :lam_final_result_date, :lam_memo, :lam_date_result
  do
  begin
    lawer_fio = '';
    manager_fio = '';
    lal_name = '';
    paymaster_fio = '';
    payed_summ = 0;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :lam_lawer_id
    into :lawer_fio;

    select cl.c_name_fio
    from clients cl
    where cl.c_id = :lam_manager_id
    into :manager_fio;

    select lal_name
    from lawlist ll
    where ll.lal_id = :lam_from_lal_id
    into :lal_name;

    doc_exist = 0;
    bls_id = null;

    select lab_from_bls_id, iif(lab_from_bls_id is null, 0, 1)
    from lawblob lab
    where lab.lab_from_lam_id = :lam_id and
          lab.lab_is_delete = 0
    into :bls_id, :doc_exist;

    kf_name = '';
    k_kontract_no = '';
    k_id = null;
    reg_id = null;
    ks_state = 0;
    select lak_from_k_id, k_kontract_no, kal_new_kf_name, ks_state, kal_new_reg_id
    from lawkredit, kontract, knaktlist, kontstate
    where lak_from_lam_id = :lam_id and
          lak_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_date <= :lad_date_end and
          kal_date_end >= :lad_date_end and
          ks_from_k_id = k_id and
          ks_ol_is_delete = 0 and
          ks_date_begin <= :lad_date_end and
          ks_date_end >= :lad_date_end rows 1
    into :k_id, :k_kontract_no, :kf_name, :ks_state, :reg_id;
    if (k_id is not null) then
    begin
      if (not (reg_lst is null or reg_lst containing reg_id)) then
        continue;

      select c.c_name_fio, c.c_id
      from kontract k
      inner join clients c on k.k_clients_cashier_id = c.c_id
      where k.k_id = :k_id
      into :paymaster_fio, :c_id;

      if (ks_state = 0) then
        ks_state_str = 'В процессе';
      if (ks_state = 1) then
        ks_state_str = 'Активный';
      if (ks_state = 2) then
      begin
        ks_state_str = --S.K. 15.01.2019г.
        iif(exists(select oko_from_k_id
                   from operkroff
                   where oko_from_k_id = :k_id and
                         oko_ol_is_active = 1 and
                         oko_ol_date <= :lad_date_end
                   group by oko_from_k_id
                   having sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc) > 0), 'Внесистемный учет', 'Закрытый');
        --
      end

      if (ks_state in (1, 2)) then
        select sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_tax) * -1
        from operkred ok
        inner join operlist ol on ol.ol_id = ok.ok_from_ol_id and
              ol.is_delete = 0 and
              ol.ol_document_client = :c_id
        where ok.ok_from_k_id = :k_id and
              ok.ok_ol_date between :lam_date_result and :lad_date_end and
              ok.ok_ol_is_active = 1
        into :payed_summ;

      payed_summ = coalesce(payed_summ, 0);

      payed_over_sum = 0;

      select sum(oko.oko_os + oko.oko_perc) * -1
      from operkroff oko
      where oko.oko_from_k_id = :k_id and
            --oko.oko_from_c_id = :c_id and
            oko.oko_ol_date >= :lam_date_result and
            oko.oko_type = 4 and
            oko.oko_ol_is_active = 1
      into payed_over_sum;

      payed_over_sum = coalesce(payed_over_sum, 0);

      payed_summ = payed_summ + payed_over_sum;
    end

    lad_date = null;
    lad_laf_name = null;
    select lad_date, laf_name, laf_order_no, laf_id
    from lawfields, lawdata
    where lad_from_lam_id = :lam_id and
          lad_from_laf_id = laf_id and
          lad_is_done = 1
    order by laf_order_no desc
    rows 1
    into :lad_date, :lad_laf_name, :laf_order_no, :laf_curr_id;

    lad_next_date = null;
    lad_next_laf_name = null;
    if (laf_order_no is null) then
      laf_order_no = 0;
    if (laf_curr_id is null) then
      laf_curr_id = 0;
    select lad_date, laf_name
    from lawfields, lawdata
    where lad_from_lam_id = :lam_id and
          lad_from_laf_id = laf_id and
          lad_is_done = 0
    order by laf_order_no
    rows 1
    into :lad_next_date, :lad_next_laf_name;
    if (lad_next_laf_name is null) then
    begin
      select laf_name
      from lawfields
      where laf_order_no >= :laf_order_no and
            laf_id <> :laf_curr_id
      order by laf_order_no
      rows 1
      into lad_next_laf_name;
    end

    if (db_id = 39801) then
    begin
      dt = null;
      tm = null;
      dt_cancel = null;

      select
        cast(lad_date as d_datetime),
        cast(lad_time as d_datetime)
      from lawdata
        inner join lawfields
          on laf_id = lad_from_laf_id
            and laf_id = 8
            and laf_is_active = 1
      where lad_from_lam_id = :lam_id
        and lad_is_done = 1
      into
        :dt,
        :tm;
      if (dt is not null) then
      begin
        dt_cancel = dt;
        if (tm is not null) then
           dt_cancel = dateadd(datediff(second, cast('00:00:00' as d_time), tm) second to dt_cancel);
      end
    end

    suspend;
  end
end
