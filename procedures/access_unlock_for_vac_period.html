CREATE PROCEDURE ACCESS_UNLOCK_FOR_VAC_PERIOD
AS
declare variable un_id id_type;
declare variable subject_kind integer;
declare variable subject_id id_type;
declare variable pm_id integer;
declare variable subject_kind_tmp integer;
declare variable subject_id_tmp id_type;
declare variable day_begin d_date;
declare variable c_id d_id;
begin
  /*
    По просьбе Алмаза исключил из автооткрытия доступа:
    24 - Трудовой отпуск декретницы
    16 - Декретный отпуск
    17 - Отпуск по беременности и родам
    18 - Отпуск без содерж.по уходу за ребенком
    */
  for select un.un_id, hrv.hrv_date_begin, hrv.hrv_from_c_id
      from hr_voctravel hrv
      inner join username un on un.un_from_c_id = hrv.hrv_from_c_id
      inner join hr_orders ho on ho.ho_id = hrv.hrv_from_ho_id and
            ho.ho_order_type in (7, 8, 9, 10, 13)
      where hrv.hrv_status = 1 and
            hrv.hrv_day_back = current_date
      into :un_id, :day_begin, :c_id

  do
  begin
    -- Проверяю, не было ли приказа на увольнение за период отпуска.
    if (exists(select *
               from hr_orders ho
               inner join hr_pstatus hp on hp.ps_from_ho_id = ho.ho_id and
                     hp.ps_from_c_id = :c_id
               where ho.ho_order_type = 6 and
                     ho.ho_date between :day_begin and current_date and
                     ho.ho_status = 1) or (exists(select *
                                                  from hr_orders ho
                                                  inner join hr_voctravel hrv on ho.ho_id = hrv.hrv_from_ho_id and
                                                        hrv.hrv_date_begin <= :day_begin and
                                                        hrv.hrv_date_end_fact >= current_date and
                                                        hrv.hrv_from_c_id = :c_id
                                                  where ho.ho_order_type in (16, 17, 18, 24) and
                                                        ho.ho_status = 1)) or (exists(select *
                                                                                      from hr_pstatus ps
                                                                                      where ps.ps_date_begin <= (current_date + 1) and
                                                                                            ps.ps_date_end >= '31.12.2099' and
                                                                                            ps.ps_from_c_id = :c_id and
                                                                                            ps.ps_is_working = 0 and
                                                                                            ps.ps_status = 1))) then
      continue;

    subject_kind_tmp = 0;
    subject_id_tmp = 0;
    for select pm.pm_subject_kind, pm.pm_subject_id, pm.pm_id
        from permissions pm
        where pm.pm_object_id = :un_id and
              pm.pm_date_begin >= :day_begin and
              pm.pm_is_delete = 0
        order by 1, 2
        into :subject_kind, :subject_id, :pm_id

    do
    begin

      update permissions
      set pm_is_delete = 1
      where (pm_id = :pm_id);

      if (:subject_kind_tmp = :subject_kind and
          :subject_id_tmp = :subject_id) then
        continue;
      if (:subject_kind_tmp <> :subject_kind) then
        :subject_kind_tmp = :subject_kind;
      if (:subject_id_tmp <> :subject_id) then
        :subject_id_tmp = :subject_id;

      insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_date_begin,
                               pm_date_end, pm_access_kind, pm_is_delete, pm_from_un_id, pm_datetime, pm_delete_un_id,
                               pm_delete_datetime)
      values (null, 0, :un_id, :subject_kind, :subject_id, current_date, '31.12.2999', 1, 0, 1000000000000,
              current_timestamp, null, null);
    end
  end
end
