CREATE PROCEDURE AAREP_SA_OKLAD_ONE(
    C_ID DECIMAL(18,0))
RETURNS (
    SAO_FROM_C_ID DECIMAL(18,0),
    SAO_ID INTEGER,
    SAO_DATE DATE,
    SAO_SUMMA DECIMAL(18,4),
    SAO_MEMO VARCHAR(255),
    SAO_IS_NET INTEGER,
    SAO_CHILD_KOLICH INTEGER,
    SAO_IS_LGOTA INTEGER,
    SAO_FROM_UN_ID DECIMAL(18,0),
    SAO_DATETIME TIMESTAMP,
    SAO_USERNAME VARCHAR(30),
    SAO_PERSONAL_FIO VARCHAR(100),
    SAO_MOUNTAIN_PERC DECIMAL(18,7),
    SAO_IS_HOUR INTEGER,
    SAO_SF_LGOTA DECIMAL(18,7),
    SAO_PDNAL_LGOTA DECIMAL(18,4),
    SAO_DONATION DECIMAL(18,4),
    SAO_OKLAD_CHANGES DECIMAL(18,4))
AS
declare variable summa decimal(18,4);
declare variable is_net integer;
declare variable child_kolich integer;
declare variable is_lgota integer;
declare variable mountain_perc decimal(18,7);
declare variable is_hour integer;
declare variable sf_lgota decimal(18,7);
declare variable st_staff integer;
declare variable st_client integer;
declare variable sao_pdnal_lg decimal(18,4);
declare variable sao_donation_tmp decimal(18,4);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  sao_summa = 0;
  sao_is_net = 0;
  sao_child_kolich = 0;
  sao_is_lgota = 0;
  sao_mountain_perc = 0;
  sao_is_hour = 0;
  sao_sf_lgota = 0;
  sao_pdnal_lgota = 0;
  sao_donation = 0;
  for select sao_from_c_id, sao_id, sao_date, sao_summa, sao_memo, sao_is_net, sao_child_kolich, sao_is_lgota,
             sao_mountain_perc, sao_is_hour, sao_from_un_id, sao_datetime, sao_sf_lgota, c_name_fio, un_name,
             sao_pdnal_lgota, sao_donation
      from sa_oklad, clients, username
      where sao_from_c_id = :c_id and
            c_id = :c_id and
            sao_from_un_id = un_id
      order by sao_date
      into :sao_from_c_id, :sao_id, :sao_date, :summa, :sao_memo, :is_net, :child_kolich, :is_lgota, :mountain_perc,
           :is_hour, :sao_from_un_id, :sao_datetime, :sao_sf_lgota, :sao_personal_fio, :sao_username, :sao_pdnal_lg,
           :sao_donation_tmp
  do
  begin
    if (summa is null) then
      summa = 0;
    sao_oklad_changes = summa;
    if (sao_pdnal_lg is null) then
      sao_pdnal_lg = 0;
    if (sao_donation_tmp is null) then
      sao_donation_tmp = 0;
    if (sao_mountain_perc is null) then
      sao_mountain_perc = 0;
    sao_summa = sao_summa + summa;
    sao_is_net = sao_is_net + is_net;
    sao_child_kolich = sao_child_kolich + child_kolich;
    sao_is_lgota = sao_is_lgota + is_lgota;
    sao_is_hour = sao_is_hour + is_hour;
    sao_mountain_perc = sao_mountain_perc + mountain_perc;
    sao_sf_lgota = sao_sf_lgota + sf_lgota;
    sao_pdnal_lgota = sao_pdnal_lgota + sao_pdnal_lg;
    sao_donation = sao_donation + sao_donation_tmp;
    suspend;
  end
end
