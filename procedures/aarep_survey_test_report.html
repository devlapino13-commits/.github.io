CREATE PROCEDURE AAREP_SURVEY_TEST_REPORT(
    QUES_LIST VARCHAR(255))
RETURNS (
    OBJ_NAME VARCHAR(255),
    TOTAL_COUNT INTEGER,
    IS_CORRECT INTEGER,
    RES_PERCENT DECIMAL(18,7))
AS
begin
  for select obj_name, count(*), sum(is_correct)
      from aarep_survey_test_tmp(:ques_list)
      group by obj_name
      order by 3 desc
      into :obj_name, :total_count, :is_correct
  do
  begin
    res_percent = 0;
    if (total_count <> 0) then
      res_percent = round((is_correct * 100.0) / total_count, 1);
    suspend;
  end
end
