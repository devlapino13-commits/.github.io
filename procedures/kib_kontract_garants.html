CREATE PROCEDURE KIB_KONTRACT_GARANTS(
    REP_DATE DATE,
    REP_KONT_COUNT INTEGER)
RETURNS (
    UNIQ_CRED_NUM DECIMAL(18,0),
    FNAME VARCHAR(100),
    LNAME VARCHAR(100),
    PATRONYMIC VARCHAR(100),
    SEX_CODE VARCHAR(3),
    BIRTHD DATE,
    BIRTHPLACE VARCHAR(255),
    IDCARDT_ID INTEGER,
    PASSN VARCHAR(30),
    PASSPLACE VARCHAR(50),
    ADDR VARCHAR(255),
    ADDRFACT VARCHAR(255),
    PATENT VARCHAR(50),
    SOCC VARCHAR(50),
    STATCOM VARCHAR(50),
    GUARANTSTATUS_CODE VARCHAR(10),
    C_ID DECIMAL(18,0),
    UNIQ_K_ID DECIMAL(18,0))
AS
declare variable c_name_fio varchar(100);
declare variable c_sex integer;
declare variable kk integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select distinct uniq_cred_num, uniq_k_id
      from kib_kontract_list(:rep_date, :rep_kont_count)
      order by uniq_cred_num
      into :uniq_cred_num, :uniq_k_id
  do
  begin
    for select c_name_fio, c_sex, c_birth_date, 'Кыргызстан', c_doc_type, c_pass_no, c_pass_who, '', '', '',
               c_id
        from clients, knclient, kontract, product
        where kc_from_c_id = c_id and
              kc_from_k_id = :uniq_k_id and
              c_type = 0 and
              (kc_amount_approved = 0 or
              kc_amount_approved is null) and
              k_id = :uniq_k_id and
              k_from_pr_id = pr_id and
              pr_category = 1
        order by c_id
        into :c_name_fio, :c_sex, :birthd, :birthplace, :idcardt_id, :passn, :passplace, :patent, :socc, :statcom,
             :c_id
    do
    begin
      addr = '';
      select adl_address_str
      from addrclient, addrlist
      where adc_address_type = 1 and
            adc_from_c_id = :c_id and
            adc_from_adl_id = adl_id
      into :addr;

      addrfact = '';
      select adl_address_str
      from addrclient, addrlist
      where adc_address_type = 2 and
            adc_from_c_id = :c_id and
            adc_from_adl_id = adl_id
      into :addrfact;

      guarantstatus_code = '004';
      fname = '';
      lname = '';
      patronymic = '';
      kk = position(' ', :c_name_fio) - 1;
      if (kk > 0) then
      begin
        lname = substring(:c_name_fio from 1 for kk);
        c_name_fio = substring(:c_name_fio from kk + 2 for 255);
        kk = position(' ', c_name_fio) - 1;
        if (kk > 0) then
        begin
          fname = substring(:c_name_fio from 1 for kk);
          c_name_fio = substring(:c_name_fio from kk + 2 for 255);
          patronymic = :c_name_fio;
        end
        else
          fname = :c_name_fio;
      end
      if (c_sex is null) then
        sex_code = 'Н';
      if (c_sex = 0) then
        sex_code = 'Ж';
      if (c_sex = 1) then
        sex_code = 'М';
      fname = trim(fname);
      lname = trim(lname);
      patronymic = trim(patronymic);
      suspend;
    end
  end

end
