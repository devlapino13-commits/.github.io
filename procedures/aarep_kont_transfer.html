CREATE PROCEDURE AAREP_KONT_TRANSFER(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OKRUG_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    OLD_REG_NAME VARCHAR(60),
    KAL_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED CURR_TYPE,
    TRANSFER_BALANCE CURR_TYPE,
    TRANSFER_DELINQ_DAYS INTEGER,
    OLD_OFFICER_NAME VARCHAR(100),
    NEW_OFFICER_NAME VARCHAR(100),
    NEW_REG_NAME VARCHAR(60),
    NEW_KF_NAME VARCHAR(100),
    OLD_KF_NAME VARCHAR(100),
    IS_TRANSFER_OFFICE SMALLINT,
    KA_AKT_NO VARCHAR(20),
    IS_OWN SMALLINT,
    K_KOLICH_CLIENT_APPROVED INTEGER)
AS
declare variable kal_from_kf_id decimal(18,0);
declare variable old_reg_id integer;
declare variable old_officer_id decimal(18,0);
declare variable new_reg_id integer;
declare variable new_officer_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate curr_type;
declare variable def_curr varchar(3);
declare variable cr_rate curr_type;
declare variable ks_kc_id id_type;
declare variable st_staff integer;
declare variable st_client integer;
declare variable can_suspend smallint;
declare variable old_reg_id_prior integer;
declare variable new_reg_id_prior integer;
declare variable old_officer_id_prior curr_type;
declare variable new_officer_id_prior curr_type;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select cm_default_currency
  from company
  into def_curr;
  old_reg_id_prior = 0;
  new_reg_id_prior = 0;
  old_officer_id_prior = 0;
  new_officer_id_prior = 0;
  old_reg_name = null;
  branch_name = null;
  okrug_name = null;
  new_reg_name = null;
  old_officer_name = null;
  new_officer_name = null;
-- Вычисляем все переданные папки за период
  for select kal_from_kf_id, -- id папки
             kal_date, -- дата передачи
             kal_prior_reg_id, -- предыдущий офис
             kal_prior_lo_id, -- предыдущий КС
             kal_new_reg_id, -- новый офис
             kal_new_kf_officer, -- новый КС
             kal_new_kf_name, -- № группы на дату передачи
             kal_prior_kf_name, -- предыдущий № группы
             ka_akt_no
      from knaktlist, knakt, region_list(:rep_region_id, 1, 3, 1)
      where knaktlist.is_delete = 0 and
            kal_date between :rep_date_begin and :rep_date_end and
            kal_from_ka_id = ka_id and
            kal_prior_reg_id = id_list
      into kal_from_kf_id, kal_date, old_reg_id, old_officer_id, new_reg_id, new_officer_id, new_kf_name, old_kf_name,
           ka_akt_no
  do
  begin
--  Оперделение, была ли передача между офисами
    if (old_reg_id <> new_reg_id) then
      is_transfer_office = 1;
    else
      is_transfer_office = 0;
--  Параметры передаваемого контракта
    k_id = null;
    k_kontract_no = null;
    k_disb_date = null;
    k_amount_approved = null;
    k_index_curr_code = null;
    k_index_curr_rate = null;
    k_kolich_client_approved = null;

    is_own = 0;
    transfer_delinq_days = 0;
    transfer_balance = 0;

    can_suspend = 1;
    for select k_id, k_kontract_no, k_disb_date, k_amount_approved, k_index_curr_code, k_index_curr_rate,
               k_kolich_client_approved, oks_balance, -- баланс передачи
               iif(:kal_date > oks_delinq_date, :kal_date - oks_delinq_date, 0) -- кол-во просроченных дней на момент передачи
        from kontract, oksummary
        where k_from_kf_id = :kal_from_kf_id and
              k_id = oks_from_k_id and
              :kal_date between oks_date_begin and oks_date_end and
              oks_close_state = 0
        into k_id, k_kontract_no, k_disb_date, k_amount_approved, k_index_curr_code, k_index_curr_rate,
             k_kolich_client_approved, transfer_balance, transfer_delinq_days
    do
    begin
      ks_kc_id = 0;
      select ks_kc_id
      from kontstate
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :kal_date between ks_date_begin and ks_date_end and
            ks_from_k_id = :k_id
      into ks_kc_id;

      if (ks_kc_id = new_officer_id) then
        is_own = 1;
      else
        is_own = 0;

--      Если валютный кредит то вычисляет валюту на дату передачи
      if (k_index_curr_code <> def_curr) then
        if (transfer_balance is not null and
            transfer_balance > 0) then
        begin
          cr_rate = null;
          select cr_rate
          from opercurr
          where :kal_date between cr_date and cr_date_end and
                cr_from_cu_id = :k_index_curr_code
          into cr_rate;
          if (cr_rate is null) then
            cr_rate = 1;
          if (k_index_curr_rate > cr_rate) then
            transfer_balance = transfer_balance * k_index_curr_rate;
          else
            transfer_balance = transfer_balance * cr_rate;
        end

      if (old_reg_id_prior <> old_reg_id) then
      begin
        old_reg_id_prior = old_reg_id;
        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
        from region r1, region r2, region r3
        where r1.reg_unicode = :old_reg_id and
              r1.parent = r2.reg_unicode and
              r2.parent = r3.reg_unicode
        into old_reg_name, branch_name, okrug_name;
      end

      if (new_reg_id_prior <> new_reg_id) then
      begin
        new_reg_id_prior = new_reg_id;
        select reg_code || ' ' || reg_name
        from region
        where reg_unicode = :new_reg_id
        into new_reg_name;
      end

      if (old_officer_id_prior <> old_officer_id) then
      begin
        old_officer_id_prior = old_officer_id;
        select c_name_fio
        from clients
        where c_id = :old_officer_id
        into old_officer_name;
      end

      if (new_officer_id_prior <> new_officer_id) then
      begin
        new_officer_id_prior = new_officer_id;
        select c_name_fio
        from clients
        where c_id = :new_officer_id
        into new_officer_name;
      end

      can_suspend = 0;
      suspend;
    end

    if (can_suspend = 1) then
    begin
      select k_id, k_kontract_no, k_disb_date, k_amount_approved, k_index_curr_code, k_index_curr_rate,
             k_kolich_client_approved
      from kontract
      where k_disb_date <= :kal_date and
            k_from_kf_id = :kal_from_kf_id and
            k_status <> -1 -- кроме аннулированных контрактов
      order by k_disb_date desc
      rows 1
      into k_id, k_kontract_no, k_disb_date, k_amount_approved, k_index_curr_code, k_index_curr_rate,
           k_kolich_client_approved;

      if (k_id is null) then
      begin
--    если утвержденных контрактов не имеются то тогда будет брать не утвержденный контракт
        select k_id, k_kontract_no, k_disb_date, k_amount_approved, k_index_curr_code, k_index_curr_rate,
               k_kolich_client_approved
        from kontract
        where k_disb_date <= :kal_date and
              k_from_kf_id = :kal_from_kf_id
--        and k_status <> -1    -- кроме аннулированных контрактов
        order by k_disb_date desc
        rows 1
        into k_id, k_kontract_no, k_disb_date, k_amount_approved, k_index_curr_code, k_index_curr_rate,
             k_kolich_client_approved;
      end

      if (old_reg_id_prior <> old_reg_id) then
      begin
        old_reg_id_prior = old_reg_id;
        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
        from region r1, region r2, region r3
        where r1.reg_unicode = :old_reg_id and
              r1.parent = r2.reg_unicode and
              r2.parent = r3.reg_unicode
        into old_reg_name, branch_name, okrug_name;
      end

      if (new_reg_id_prior <> new_reg_id) then
      begin
        new_reg_id_prior = new_reg_id;
        select reg_code || ' ' || reg_name
        from region
        where reg_unicode = :new_reg_id
        into new_reg_name;
      end

      if (old_officer_id_prior <> old_officer_id) then
      begin
        old_officer_id_prior = old_officer_id;
        select c_name_fio
        from clients
        where c_id = :old_officer_id
        into old_officer_name;
      end

      if (new_officer_id_prior <> new_officer_id) then
      begin
        new_officer_id_prior = new_officer_id;
        select c_name_fio
        from clients
        where c_id = :new_officer_id
        into new_officer_name;
      end

      suspend;
    end

  end
end
