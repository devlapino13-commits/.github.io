CREATE PROCEDURE AAREP_ACTIVE_LOANS_PRESCRIPTION(
    REP_REGION_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    PARENT_NAME D_STR_255,
    ASP_SIGNED D_STR_100,
    REG_NAME D_STR_255,
    KF_NAME_FIO D_STR_255,
    K_DISB_DATE D_DATE,
    KONT_LAST_DATE D_DATE,
    CLIENT_FIO D_STR_255,
    K_KONTRACT_NO D_STR_100,
    LOAN_TARGET_DESC D_STR_255,
    BS_NAME D_STR_255,
    PR_NAME D_STR_255,
    KC_AMOUNT_APPROVED D_CURRENCY,
    MONITORING_STATUS D_STR_100,
    MONITORING_MEMO D_STR_255,
    FAMILY_STATUS D_STR_20,
    ATTORNEY D_STR_10,
    ATTORNEY_365 D_STR_10,
    STATEMENTS D_STR_10,
    DEATHS D_STR_10,
    DIVORCE D_STR_10,
    DESCR_OF_LOAN D_STR_2048,
    DELINK_DAYS D_INTEGER,
    ANKETA_QELINK D_STR_10,
    BT_NAME D_STR_100,
    WORK_DESCRIPTION D_STR_LONG,
    ACTIV D_STR_10,
    ASP_DATE D_DATE,
    JUDICIAL_ACT D_STR_10,
    WORK_ADDRESS_ORIENTIR D_STR_LONG)
AS
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable pr_id d_id;
declare variable loan_target d_integer;
declare variable k_id d_id;
declare variable kf_officer d_id;
declare variable is_ipf d_smallint;
declare variable analiz_date d_date;
declare variable date_x d_date;
declare variable rd_id d_id;
declare variable rd_memo_id d_id;
declare variable dt_find d_date;
declare variable obj_id d_id;
declare variable obj_kind d_integer;
declare variable crel_other_c_id d_id;
declare variable k_id_temp d_id;
declare variable bs_id d_id;
declare variable c_family d_integer;
declare variable pk_id d_integer;
declare variable ma_id d_id;
declare variable delinq_memo d_str_255;
declare variable bt_id d_id;
declare variable db_id d_integer;
declare variable k_disb_kind d_integer;
declare variable mac_id d_integer;
declare variable kl_date d_date;
begin
  db_id = f_get_db_id();

  date_x = '03.03.2022'; -- Дата перехода на knclient
  rd_id = f_get_rd_id('Anketa.TargetUseLoan');
  rd_memo_id = f_get_rd_id('Monitoring.Memo');
  activ = 'не активен';
  for select k.k_id, k.k_kontract_no, k.k_fact_disb_date, k.k_from_ma_app_id, kal.kal_new_kf_officer,
             kc.kc_amount_approved, k.k_from_pr_id, kc.kc_loan_target, kc.kc_from_c_id, kc.kc_id, parent_name,
             reg.reg_name, iif(k.k_fact_disb_date >= :date_x, kc.kc_id, kc_from_c_id),
             iif(k.k_fact_disb_date >= :date_x, f_objkind('KnClient', 'id'), f_objkind('Clients', 'id')),
             k.k_kont_last_date, kc.kc_busuness_subtype, kc.kc_business_type, k.k_disb_kind
      from kontstate ks
      inner join kontract k on k.k_id = ks.ks_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
      inner join knclient kc on kc.kc_from_k_id = k.k_id and
            kc.kc_amount_approved > 0
      inner join region_list(:rep_region_id, 1, 3, 1) on id_list = k.k_fact_disb_reg_id
      inner join region reg on reg.reg_unicode = k.k_fact_disb_reg_id
      where ks.ks_state = 1 and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0 and
            kal.kal_new_kf_officer not in (1000000000100) and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into k_id, k_kontract_no, k_disb_date, ma_id, kf_officer, kc_amount_approved, pr_id, loan_target, c_id, kc_id,
           parent_name, reg_name, obj_id, obj_kind, kont_last_date, bs_id, bt_id, k_disb_kind
  do
  begin
    monitoring_status = '';
    bs_name = ''; --Отрасль кредитования
    select bst.bs_name
    from bussubtype bst
    where bst.bs_id = :bs_id
    into bs_name;

    bt_name = ''; --Тип бизнеса
    select bt.bt_name
    from bustype bt
    where bt.bt_id = :bt_id
    into :bt_name;

    crel_other_c_id = null; --Супруг\супруга
    select crt.rel_other_c_id
    from hp_client_relation_tmp(:c_id, 2) crt
    where crt.rel_code in ('1', '2') and
          rel_is_active = 1 rows 1
    into crel_other_c_id;

    asp_signed = 'Нет';

    if (exists(select *
               from edoc_list edl
               inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
               where edl.edl_object_kind = 11 and
                     edl.edl_object_id = :ma_id and
                     eds.eds_from_c_id = :crel_other_c_id and
                     eds.eds_status = 1)) then --Наличие согласия супруга электронный вариант
      asp_signed = 'Да';

    if (asp_signed = 'Нет' and
        exists(select *
               from clientdocs cds
               inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
               inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id and
                     pdt.pdt_code = 'consent_to_a_loan'
               where cds.cds_from_c_id = :crel_other_c_id and
                     cds.cds_is_delete = 0 and
                     :k_disb_date between cds.cds_doc_date_begin and cds.cds_doc_date_end)) then
      asp_signed = 'Да';

    if (asp_signed = 'Нет' and
        :kc_amount_approved < 100000 and
        :k_disb_date >= '05.11.2024') then
      asp_signed = 'Не нужно';

    statements = 'Нет'; -- 336 Заявление о семейном положении
    attorney = 'Нет'; -- 343 Доверенность на уполномоченное лицо
    deaths = 'Нет'; -- 352 Свид о смерти близ родст/супруг
    divorce = 'Нет'; -- 53 Свид-во о расторжении брака
    judicial_act = 'Нет'; -- 364 Судебный акт
    attorney_365 = iif(:db_id in (39801, 41701), 'Нет', 'Для КР'); -- 365 Довереность супруга(ги) (RFCKG-2576)

    for select pk.pk_id
        from clientdocs cds
        inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
        where cds.cds_from_c_id = :c_id and
              cds.cds_doc_date_begin <= :rep_date_end and
              cds.cds_is_delete = 0 and
              pk.pk_is_delete = 0
        into :pk_id
    do
    begin
      if (:pk_id = 336) then
        statements = 'Да'; -- 336 Заявление о семейном положении
      if (:pk_id = 343) then
        attorney = 'Да'; -- 343 Доверенность
      if (:pk_id = 352) then
        deaths = 'Да'; -- 352 Свид о смерти близ родст/супруг
      if (:pk_id = 53) then
        divorce = 'Да'; -- 53 Свид-во о расторжении брака
      if (:pk_id = 310) then
        asp_signed = 'Да'; -- 310 Наличие согласия супруга бумажнный вариант
      if (:pk_id = 364) then
        judicial_act = 'Да'; -- 364 Судебный акт
      if (:pk_id = 365 and :db_id in (39801, 41701)) then
        attorney_365 = 'Да'; -- 365 Довереность супруга(ги) (RFCKG-2576)
    end

    pr_name = '';
    is_ipf = null;
    select pr.pr_name, iif(pr.pr_code containing '.ipf.', 1, 0)
    from product pr
    where pr.pr_id = :pr_id
    into pr_name, is_ipf;

    client_fio = '';
    c_family = 0;
    select c.c_name_fio, c.c_family
    from clients c
    where c.c_id = :c_id
    into client_fio, c_family;

    kf_name_fio = '';
    select cl.c_name_fio
    from clients cl
    where cl.c_id = :kf_officer
    into :kf_name_fio; --СПФ на контроле

    loan_target_desc = '';
    select rd.rd_name
    from refdata rd
    where rd.rd_id = :loan_target
    into loan_target_desc;

    analiz_date = null;
    if (k_disb_date < '14.02.2022') then
      dt_find = k_disb_date + iif(k_disb_date >= '22.12.2021' and is_ipf = 1, 1, 8);
    else
      dt_find = k_disb_date + 1;
    analiz_date = null;

    select acl_last_datetime
    from actionlist
    where acl_object_kind = :obj_kind and
          acl_object_id = :obj_id and
          acl_from_rd_id = :rd_id and
          acl_date = :dt_find and
          acl_status = 2 rows 1
    into :analiz_date;

    family_status = null;
    select rd.rd_name
    from refdata rd
    where rd.rd_id = :c_family
    into :family_status;

    if (exists(select *
               from actionlist acl
               where acl.acl_object_kind = :obj_kind and
                     acl.acl_object_id = :obj_id and
                     acl.acl_respons_c_id = :kf_officer and
                     acl.acl_date = :dt_find and
                     acl.acl_from_rd_id = :rd_id and
                     acl.acl_status in (0,1))) then
      activ = 'активен';

    ---Подробное описание цели
    descr_of_loan = null;
    select coalesce(loan_description, '')
    from hp_get_loan_description(:c_id, :ma_id)
    into :descr_of_loan;

    --Комментарии СПФ
    monitoring_memo = null;
    select coalesce(res_str_values, '')
    from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'Monitoring.Memo', :rep_date_end, null)
    into :monitoring_memo;

    -- Описание доходоприносящей деятельности (RFCKG-2802)
    select mac_id
    from miniappclient
    where mac_from_ma_id = :ma_id
        and mac_from_c_id= :c_id
    into :mac_id;

    work_description = null;
    select ca.ca_income_txt_ru
    from cfp_anketa ca
    where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
          ca.ca_object_id = :mac_id and
          ca.ca_from_c_id = :c_id and   --RFCKG-5659
          ca.ca_date <= :rep_date_end
    order by ca.ca_datetime desc
    rows 1
    into :work_description;

    if(work_description is null) then
        select coalesce(res_str_values, '')
        from hp_objectprops_anketa_get('Clients', :c_id, -1, 'work_place_name', :k_disb_date, null)
        into :work_description;

    -- Доп. инфо о бизнесе (RFCKG-3102)
    kl_date = null;
    select cast(max(kl_datetime) as date)
    from kontlog
    where kl_state = 1 and
          kl_from_k_id = :k_id
    into kl_date;

    work_address_orientir = null;
    select res_str_values
    from hp_objectprops_anketa_get('clients', :c_id, -1, 'work_address_orientir', coalesce(:kl_date, current_date), null)
    into work_address_orientir; -- анкета анализ клиента/поручителя -> поле "Доп. инфо о бизнесе"

    --descr_of_loan = coalesce(trim(:descr_of_loan), '');
    monitoring_status = 'Не заполнен';

    -- (RFCKG-4425) Изменена проверка для поля monitoring_status
    if (exists(
        select 1
        from actionlist acl
        where acl.acl_object_kind = :obj_kind and
              acl.acl_object_id = :kc_id and
              acl.acl_from_rd_id = :rd_id and
              acl.acl_status = 2
    )) then
      monitoring_status = 'Заполнен';

    if (:pr_id in (439, 473, 474, 475, 476, 477, 478, 480, 481, 482,
                   483, 484, 491, 492, 494, 495, 493, 494, 495, 235,
                   236, 250, 251, 264, 280, 283, 290, 295, 340, 503,
                   504, 505) or (:k_disb_kind = 6 and :k_disb_date >= '04.07.2024')) then
      monitoring_status = 'Не нужен';

    if (monitoring_status <> 'Заполнен' and (:kc_amount_approved < 100000 and :k_disb_date >= '05.11.2024')) then
      monitoring_status = 'Не нужен';

    -- RFCKG-5257
    if (db_id = 41701
        and :k_disb_date >= '05.11.2025'
        and :is_ipf = 1
        and lower(:loan_target_desc) starting 'потреб') then
      monitoring_status = 'Не нужен';

    select iif((cast(:rep_date_end as date) - oks.oks_delinq_date < 0), 0, cast(:rep_date_end as date) - oks.oks_delinq_date) -- числоднейактивнойпросрочки
    from oksummary oks
    where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_from_k_id = :k_id and
          oks_close_state = 0
    into :delink_days;

    select coalesce(res_str_values, '')
    from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'DelinqReport.Memo', :rep_date_end, null)
    into :delinq_memo;

    if (delinq_memo = '') then
      anketa_qelink = 'нет';
    else
      anketa_qelink = 'да';

    select cast(max(cds.cds_create_datetime) as date)
    from clientdocs cds
    inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where cds_from_c_id = :c_id and
          pdt.pdt_code = 'contact_eds' -- договор ЭЦП
    into asp_date;

    suspend;
  end
end
