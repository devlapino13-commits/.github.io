CREATE PROCEDURE HP_EXTINFO_RESEND(
    EIL_ID D_ID,
    UN_ID D_ID,
    EIC_ID_IN D_ID = null)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable eic_name d_str_255;
declare variable eic_code d_str_code;
declare variable eic_id d_id;
declare variable eic_name_tmp d_str_255;
declare variable eic_code_tmp d_str_code;
declare variable object_kind d_integer;
declare variable object_id d_id;
declare variable date_begin d_date;
declare variable close_date d_date;
declare variable agreement_date d_date;
declare variable ks_state d_integer;
declare variable ma_id d_id;
declare variable edl_status d_integer;
declare variable eic_kind d_integer;
declare variable eic_kind_tmp d_integer;
declare variable eic_date_end d_date;
declare variable eic_date_end_tmp d_date;
declare variable c_id d_id;
begin
  -- Переотправка запроса на внешние системы

  err_text = '';

  select eic_id, eic_code, eic_name, eil_object_kind, eil_object_id, eic_date_end, eic_kind, eil_from_c_id
  from extinfolist, extinfocfg
  where eil_from_eic_id = eic_id
    and eil_id = :eil_id
  into eic_id, eic_code, eic_name, object_kind, object_id, eic_date_end, eic_kind, c_id;

  if (eic_id_in is not null and eic_kind > 0 and f_get_db_id() = 81001) then
  begin
    select eic_code, eic_name, eic_date_end, eic_kind
    from extinfocfg
    where eic_id = :eic_id_in
    into eic_code_tmp, eic_name_tmp, eic_date_end_tmp, eic_kind_tmp;

    if (eic_kind_tmp is distinct from eic_kind) then
      err_text = f_format('Нельзя переотправить запрос "%1" как "%2"', eic_name, eic_name_tmp);
    else
    begin
      eic_id = eic_id_in;
      eic_code = eic_code_tmp;
      eic_name = eic_name_tmp;
      eic_date_end = eic_date_end_tmp;
    end
  end 

  if (err_text = '' and eic_code is null) then
    err_text = 'Не выбран запрос для повторной отправки';

  if (err_text = '' and current_date > eic_date_end) then
    err_text = f_format('Проект "%1" уже не активен.', eic_name);

  -- Проверяем права
  if (err_text = '' and not exists(
      select has_access
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'ExtInfo.Ins.' || :eic_code)
      where has_access = 1)) then
    err_text = 'У вас отсутствуют права для переотправки запросов: ' || eic_name;

  if (err_text = '' and eic_kind = 1 and f_get_db_id() = 81001) then
  begin
    -- проверка есть ли разрешение для отправки запроса в БКИ
    if (object_kind = 9) then
    begin
      ma_id = (select k_from_ma_app_id from kontract where k_id = :object_id);

      select ks_state
      from kontstate
      where ks_ol_is_delete = 0
        and ks_from_k_id = :object_id
        and current_date between ks_date_begin and ks_date_end
      into ks_state;
  
      if (ks_state = 2) then
        err_text = 'Договор уже не активен.';

      if (err_text = '' and ks_state = 1) then
        select ok_sh_date
        from operkred
        where ok_from_k_id = :object_id
          and ok_type < 7
          and ok_ol_is_active = 1
          and ok_ol_date <= current_date
        group by 1
        having sum(ok_os + ok_perc + ok_sh_os + ok_sh_perc) > 0
        order by 1 desc
        rows 1
        into close_date;

      if (err_text = '' and ks_state = 0) then
      begin
        select max(sh_date)
        from schedule
        where sh_from_k_id = :object_id
        into close_date;
      end
    end
    else if (object_kind = 11) then
    begin
      ma_id = object_id;

      select max(sh_date)
      from schedule
      where sh_from_k_id = -:ma_id
      into close_date;
  
      if (close_date is null) then
      begin
        select max(sh_date)
        from schedule, kontract
        where k_id = sh_from_k_id
          and k_from_ma_app_id = :ma_id
        into close_date;
      end
    end 

    if (err_text = '' and close_date is null) then
      err_text = 'Не определен срок разрешения на запрос в БКИ.';

    if (err_text = '' and coalesce(ks_state, 0) = 0) then -- в процессе
    begin
      date_begin = (select ma_date_app from miniapp where ma_id = :ma_id order by ma_number rows 1);

      select edl_status
      from miniappclient mac, edoc_sign eds, edoc_list edl
      where eds_from_c_id = mac_from_c_id
        and edl_object_id = mac_from_ma_id
        and eds_from_edl_id = edl_id
        and edl_object_kind = f_objkind('miniapp', 'id')
        and edl_object_id = :ma_id
        and edl_from_edc_id = f_edoc_kind('loan.application','id') --заявка на займ
        and eds_status = 1
        and eds_status_datetime between :date_begin and current_timestamp
      order by edl_status desc, eds_status_datetime desc
      rows 1
      into edl_status;

      select eds_status_datetime
      from miniappclient mac, edoc_sign eds, edoc_list edl
      where eds_from_c_id = mac_from_c_id
        and edl_object_id = mac_from_ma_id
        and eds_from_edl_id = edl_id
        and edl_object_kind = f_objkind('miniapp', 'id')
        and edl_object_id = :ma_id
        and edl_from_edc_id = f_edoc_kind('bki.agreement','id') --заявка на займ
        and eds_status = 1
        and edl_status = 1
        and eds_status_datetime between :date_begin and current_timestamp
      order by eds_status_datetime desc
      rows 1
      into agreement_date;

      if (coalesce(edl_status, 0) = 0) then
        err_text = 'Нет подписанной заявки.';
      else if (agreement_date is null and f_get_const('doc_optimize_pack2') <> 1) then
        err_text = 'Отсутствует подписанное согласие на запрос кредитного отчета.';
      else if (edl_status = -1) then
        close_date = dateadd(month, 6, agreement_date);  -- если заявка аннулирована, то срок действия согласия 6 месяцев.

    end

    if (err_text = '' and close_date < current_date) then
      err_text = 'Истек срок разрешения на запрос в БКИ.';
  end

  if (err_text = '') then
  begin
    if (eic_code = 'kontur') then
      execute procedure hp_extinfo_insert_kontur(:object_kind, :object_id, 1, :un_id);
    else if (eic_code = 'passport_smev') then
      execute procedure hp_extinfo_insert_smevpass(:c_id, 1, :un_id);
    else
      insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json, eil_from_un_id)
      select eil_from_c_id, :eic_id, current_date, eil_object_kind, eil_object_id, eil_is_json, :un_id
      from extinfolist
      where eil_id = :eil_id;
  end

  suspend;
end
