CREATE PROCEDURE HP_OPERCONS_SCHEDULE(
    K_ID D_ID,
    K_CONS_KIND D_SMALLINT,
    OL_ID D_ID)
AS
declare variable rows_counter d_smallint;
declare variable res_amount d_currency;
declare variable payed_amount d_currency;
declare variable cons_price d_currency;
declare variable current_payment d_currency;
declare variable distributed_amount d_currency;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable k_disb_date d_date;
declare variable k_reg_id d_id;
declare variable rd_id d_id;
begin
    select 
        kncl.kc_id, 
        kncl.kc_from_c_id, 
        k.k_disb_date,
        k.k_fact_disb_reg_id
    from kontract k
    inner join knclient kncl on kncl.kc_from_k_id = k.k_id
    where k.k_id = :k_id
        and kncl.kc_amount_approved > 0
    into :kc_id,
         :c_id,
         :k_disb_date,
         :k_reg_id;
    
    if (kc_id is null or k_cons_kind is null or k_cons_kind <= 0) then exit;
    
    select price
    from clt_get_price_for_consultation(:kc_id)
    into :res_amount;

    if (res_amount > 0) then
    begin
        rd_id = f_get_rd_id('CashTransactionIncom.Consultation');
        
        select sum(opk.ok_amount)
        from operkontract opk
        inner join operlist ol on ol.ol_id = opk.ok_from_ol_id 
        where opk.ok_from_k_id = :k_id
            and opk.ok_operation_type = :rd_id
            and ol.is_delete = 0
        into :payed_amount;
        
        payed_amount = coalesce(:payed_amount, 0);
        res_amount = round(res_amount - payed_amount, 0);
        
        if (res_amount <= 0) then
            exit;
        
        cons_price = round(:res_amount / :k_cons_kind, 0); 
        rows_counter = 1;

        distributed_amount = 0;

        while (rows_counter <= :k_cons_kind)
        do
        begin
            current_payment = iif(rows_counter < k_cons_kind, 
                                  :cons_price, 
                                  :res_amount - :distributed_amount);

            insert into opercons (
                oc_from_ol_id,
                oc_number,
                oc_from_k_id,
                oc_from_c_id, 
                oc_sh_id,
                oc_schedule_date,
                oc_schedule_principal, 
                oc_paid_principal,
                oc_operation_type_id,
                oc_ol_date, 
                oc_ol_reg_id,
                oc_ol_is_active,
                ok_pay_source)
            values (
                :ol_id,
                :rows_counter, 
                :k_id, 
                :c_id, 
                :rows_counter,
                dateadd(:rows_counter month to :k_disb_date), 
                :current_payment, 
                0,
                1,
                :k_disb_date,
                :k_reg_id,
                1,
                3);

            distributed_amount = :distributed_amount + :current_payment;
            rows_counter = :rows_counter + 1;
        end
    end
end
