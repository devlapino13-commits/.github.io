CREATE PROCEDURE AAREP_TRANSFER_INFO_BEELINE(
    REP_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    NUMBER_OF_CONTRACT D_STR_20,
    EMPLOYEE_PHONE_NUMBER D_STR_20,
    EMPLOYEE_ID ID_TYPE,
    EMPLOYEE_SURNAME D_STR_100,
    EMPLOYEE_NAME D_STR_100,
    EMPLOYEE_MIDDLENAME D_STR_100,
    EMPLOYEE_CITIZENSHIP D_STR_10,
    EMPLOYEE_DATE_OF_BIRTH D_DATE,
    EMPLOYEE_PLACE_OF_BIRTH D_STR_100,
    EMPLOYEE_GENDER D_STR_100,
    EMPLOYEE_INN D_STR_100,
    EMPLOYEE_SNILS D_STR_100,
    DOCUMENT_TYPE D_STR_100,
    DOCUMENT_UIN D_STR_100,
    DOCUMENT_DATE_BEGIN D_DATE,
    DOCUMENT_DATE_END D_DATE,
    DOCUMNET_AUTHORITY_ISSUED_BY D_STR_2048,
    DOCUMNET_AUTHORITY_CODE D_STR_100,
    DOCUMENT_NON_RESIDENT D_STR_2048,
    MIGRATION_CARD_NUMBER D_STR_100,
    MIGRATION_CARD_DATE_BEGIN D_DATE,
    MIGRATION_CARD_DATE_END D_DATE,
    ADDRESS_REG_INDEX D_STR_10,
    ADDRESS_REG_COUNTRY D_STR_10,
    ADDRESS_REG_REGION D_STR_100,
    ADDRESS_REG_DISTRICT D_STR_100,
    ADDRESS_REG_LOCALITY_TYPE D_STR_10,
    ADDRESS_REG_LOCALITY_NAME D_STR_100,
    ADDRESS_REG_STREET_TYPE D_STR_100,
    ADDRESS_REG_STREET_NAME D_STR_100,
    ADDRESS_REG_HOUSE_TYPE D_STR_100,
    ADDRESS_REG_HOUSE_NUMBER D_STR_100,
    ADDRESS_REG_BUILDING_TYPE D_STR_100,
    ADDRESS_REG_BUILDING_NUMBER D_STR_100,
    ADDRESS_REG_ROOM_TYPE D_STR_100,
    ADDRESS_REG_ROOM_NUMBER D_STR_100,
    ADDRESS_REG_COMMENT D_STR_100,
    ADDRESS_REG_FIAS_ID D_STR_100,
    ADDRESS_RES_INDEX D_STR_100,
    ADDRESS_RES_COUNTRY D_STR_10,
    ADDRESS_RES_REGION D_STR_100,
    ADDRESS_RES_DISTRICT D_STR_100,
    ADDRESS_RES_LOCALITY_TYPE D_STR_100,
    ADDRESS_RES_LOCALITY_NAME D_STR_100,
    ADDRESS_RES_STREET_TYPE D_STR_100,
    ADDRESS_RES_STREET_NAME D_STR_100,
    ADDRESS_RES_HOUSE_TYPE D_STR_100,
    ADDRESS_RES_HOUSE_NUMBER D_STR_100,
    ADDRESS_RES_BUILDING_TYPE D_STR_100,
    ADDRESS_RES_BUILDING_NUMBER D_STR_100,
    ADDRESS_RES_ROOM_TYPE D_STR_100,
    ADDRESS_RES_ROOM_NUMBER D_STR_100,
    ADDRESS_RES_COMMENT D_STR_100,
    ADDRESS_RES_FIAS_ID D_STR_100)
AS
begin
  report_date_begin = coalesce(:report_date_begin, '2001-01-01');
  report_date_end = coalesce(:report_date_end, current_date);
  for
    with regions as (
      select
         id_list as id
        from
          region_list(:rep_region_id, 1, -1, 1)
    )
    select
        -- Основные данные
          cal.cal_number_with_operator
        , iif(cal.cal_cell_no starting with '7', substring(cal.cal_cell_no from 2 for 11), cal.cal_cell_no)

        -- Персональные данные
        , c.c_id
        , (select surname    from aarep_transfer_beeline_fio(c.c_name_fio))  -- Фамилия
        , (select firstname  from aarep_transfer_beeline_fio(c.c_name_fio))  -- Имя
        , (select middlename from aarep_transfer_beeline_fio(c.c_name_fio))  -- Отчество
        , (select first 1 bc_code from beeline_country where (bc_is_unknown = 1) or (bc_from_c_citizen = c.c_citizen) order by bc_is_unknown asc)
        , c.c_birth_date
        , c.c_birth_city
        , cast(iif(c.c_sex = 1, 'M', 'F') as d_str_10)
        , c.c_inn
        , null

        -- Документы
        , passport.id
        , replace(coalesce(c.c_pass_no, passport.number), ' ', '')
        , coalesce(c.c_pass_date, passport.date_begin)
        , coalesce(c.c_pass_date_expire, passport.date_end)
        , coalesce(c.c_pass_who, passport.issued_by)
        , coalesce(c.c_pass_who_code, passport.authority_code)

        , doc_other.data_all                                                 as document_non_resident

        , replace(doc_migration.number, ' ', '')                             as migration_card_number
        , doc_migration.date_begin                                           as migration_card_date_begin
        , doc_migration.date_end                                             as migration_card_date_end
    
        -- Адрес регистрации
        , reg.address_index
        , reg.address_country
        , reg.address_region
        , reg.address_district
        , reg.address_locality_type
        , reg.address_locality_name
        , reg.address_street_type
        , reg.address_street_name
        , reg.address_house_type
        , reg.address_house_number
        , reg.address_building_type
        , reg.address_building_number
        , reg.address_room_type
        , reg.address_room_number
        , reg.address_comment
        , reg.address_fias_id
    
        -- Адрес фактический
        , res.address_index
        , res.address_country
        , res.address_region
        , res.address_district
        , res.address_locality_type
        , res.address_locality_name
        , res.address_street_type
        , res.address_street_name
        , res.address_house_type
        , res.address_house_number
        , res.address_building_type
        , res.address_building_number
        , res.address_room_type
        , res.address_room_number
        , res.address_comment
        , res.address_fias_id
      from
        cellaktmain                                                                       as cam
        inner join cellaktlist                                                            as cal
          on (cal.cal_from_cam_id = cam.cam_id)
        inner join regions                                                                as r
          on (r.id = cam.cam_from_reg_id)
        inner join clients                                                                as c
          on (c.c_id = cal.cal_from_c_id)
        left join aarep_transfer_beeline_address(c.c_id, 0)                               as reg
          on (1 = 1)
        left join aarep_transfer_beeline_address(c.c_id, 1)                               as res
          on (1 = 1)
        left join aarep_transfer_beeline_docs(c.c_id, :report_date_end, 01 /*ДДЛ*/, 1) as passport
          on (passport.client_id = c.c_id)
        left join aarep_transfer_beeline_docs(c.c_id, :report_date_end, 16 /*МК*/, 1) as doc_migration
          on (passport.client_id = c.c_id)
        left join aarep_transfer_beeline_docs(c.c_id, :report_date_end, 14 /*Патент+РВП+ВНЖ*/, 0) as doc_other
          on (passport.client_id = c.c_id)
      where
            (cal.cal_date_begin >= :report_date_begin)
        and (cal.cal_date_end >= :report_date_end)
        and (cal.cal_is_corporate = 1)
        and (cal.cal_state = 1)
      order by
        c.c_name_fio
        , cal.cal_cell_no
    into
        :number_of_contract
      , :employee_phone_number
      , :employee_id
      , :employee_surname
      , :employee_name
      , :employee_middlename
      , :employee_citizenship
      , :employee_date_of_birth
      , :employee_place_of_birth
      , :employee_gender
      , :employee_inn
      , :employee_snils
      , :document_type
      , :document_uin
      , :document_date_begin
      , :document_date_end
      , :documnet_authority_issued_by
      , :documnet_authority_code
      , :document_non_resident
      , :migration_card_number
      , :migration_card_date_begin
      , :migration_card_date_end
      , :address_reg_index
      , :address_reg_country
      , :address_reg_region
      , :address_reg_district
      , :address_reg_locality_type
      , :address_reg_locality_name
      , :address_reg_street_type
      , :address_reg_street_name
      , :address_reg_house_type
      , :address_reg_house_number
      , :address_reg_building_type
      , :address_reg_building_number
      , :address_reg_room_type
      , :address_reg_room_number
      , :address_reg_comment
      , :address_reg_fias_id
      , :address_res_index
      , :address_res_country
      , :address_res_region
      , :address_res_district
      , :address_res_locality_type
      , :address_res_locality_name
      , :address_res_street_type
      , :address_res_street_name
      , :address_res_house_type
      , :address_res_house_number
      , :address_res_building_type
      , :address_res_building_number
      , :address_res_room_type
      , :address_res_room_number
      , :address_res_comment
      , :address_res_fias_id
    do
      suspend;
end
