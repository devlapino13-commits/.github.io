CREATE PROCEDURE PRINT_CONFIRMED_INCOME_FOOTER(
    K_ID D_ID)
RETURNS (
    NDFL_PERCENT D_STR_10,
    PERIOD_INCOME D_CURRENCY,
    APPROVED_INCOME D_CURRENCY,
    DIVIDER D_SMALLINT,
    MONTHLY_AVG_INCOME D_CURRENCY,
    PKK_FIO D_STR_100)
AS
declare variable ki_survey d_str_8100;
begin
  -- RFCRU-4782
  -- ПДН; Профсуждение;
  for
    select first 1
      ki_survey
    from kk_income
    where ki_from_k_id = :k_id
      and ki_is_manual = 0
    order by ki_id desc
  into
    :ki_survey
  do
  begin
    ndfl_percent = f_json_str_get_value(ki_survey, 'taxrate', '0') || '%';
    period_income = f_json_str_get_value(ki_survey, 'income', '0');
    approved_income = f_json_str_get_value(ki_survey, 'incomewt', '0');
    divider = f_json_str_get_value(ki_survey, 'nmonth', '0');
    monthly_avg_income  = f_json_str_get_value(ki_survey, 'incomeavg', '0');
    pkk_fio = 'Гитинова Зара';
    suspend;
  end
end
