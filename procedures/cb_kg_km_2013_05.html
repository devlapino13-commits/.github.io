CREATE PROCEDURE CB_KG_KM_2013_05(
    REP_REGION_ID INTEGER,
    REP_BONUS_ID INTEGER,
    REP_USER_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    OKRUG_NAME VARCHAR(50),
    KM_NAME VARCHAR(100),
    BALANCE_OFFICE CURR_TYPE,
    DELINQ_BALANCE8_180 CURR_TYPE,
    PAR8_180 DECIMAL(5,5),
    BONUS_KC CURR_TYPE,
    KOLICH_KC INTEGER,
    BONUS_KM CURR_TYPE,
    BONUS_KOEF CURR_TYPE,
    BONUS_SUM CURR_TYPE,
    BONUS_MEMO VARCHAR(255),
    BONUS_KM_TOTAL CURR_TYPE)
AS
declare variable curr_reg_id integer;
declare variable curr_reg_id_prior integer;
declare variable km_id id_type;
begin
  curr_reg_id_prior = 0;
  for select curr_reg_id, km_id, sum(bonus_end), count(iif(ps_is_working = 1, aa_curr_kc_fio, null))
      from cb_kg_lo_2013_05(:rep_region_id, 0, 0, :rep_date_end)
      group by curr_reg_id, km_id
      order by curr_reg_id
      into curr_reg_id, km_id, bonus_kc, kolich_kc
  do
  begin
    if (curr_reg_id_prior <> curr_reg_id) then
    begin
      curr_reg_id_prior = curr_reg_id;
      select aa_curr_office_name, aa_curr_branch_name, aa_curr_level3_name
      from aarep_get_office_categ(:curr_reg_id, :rep_date_end)
      into office_name, branch_name, okrug_name;

      select balance_office, delinq_balance8_180, par8_180
      from aarep_get_office_par(:curr_reg_id, :rep_date_end)
      into balance_office, delinq_balance8_180, par8_180;
    end
    else
    begin
      balance_office = 0;
      delinq_balance8_180 = 0;
    end
    km_name = (select c_name_fio
               from clients
               where c_id = :km_id);
    if (par8_180 <= 2.00000 and
        km_name is not null) then
    begin
      bonus_km = round((bonus_kc / kolich_kc) * 110 / 100, 0);
    end
    else
    begin
      bonus_km = 0;
      if (km_name is null) then
        km_name = 'КРЕДИТНЫЙ МЕНЕДЖЕР НЕ ОПРЕДЕЛЕН!';
    end
--  вытаскивает данные по выговорам и бонус коэффициент
    select bonus_koef, bonus_sum, bonus_memo
    from hp_get_vigovor_koef(:km_id, :rep_date_end)
    into bonus_koef, bonus_sum, bonus_memo;

    bonus_km_total = bonus_km * bonus_koef / 100 - bonus_sum;
    if (bonus_km_total < 0) then
      bonus_km_total = 0;

    suspend;
  end
end
