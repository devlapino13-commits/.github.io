CREATE PROCEDURE HP_CLIENT_DELINQ_CALLS(
    OBJECT_KIND INTEGER,
    OBJECT_ID ID_TYPE)
RETURNS (
    CALLED_DATE DATE,
    MEMO VARCHAR(1000),
    PROJECT VARCHAR(10),
    RESULT STR_100,
    CLIENT_FIO STR_100)
AS
declare variable pcl_res integer;
declare variable object_kind_name str_100;
declare variable pcl_res_old integer;
begin
  object_kind_name = f_objkind(object_kind, 'name');

  pcl_res_old = 0;

  if (object_kind_name = 'Clients') then
  begin
    for select pcl.pcl_res, pcl.pcl_called_date, pcl.pcl_memo,
               (
               case pant_project_type
                 when 1 then 'Пантера'
                 when 4 then 'Сирена'
               end)
        from pantera_clients pcl
        inner join pantera_kontracts pantk on pantk.pant_id = pcl.pcl_from_pan_id
        inner join kontstate ks on ks.ks_from_k_id = pantk.pant_from_k_id
        where pcl.pcl_from_c_id = :object_id and
              pantk.pant_project_type in (1, 4) and
              current_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1
        order by pcl.pcl_called_date desc
        into :pcl_res, :called_date, :memo, :project
    do
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :object_id
      into :client_fio;

      if (pcl_res <> pcl_res_old) then
      begin
        result = null;

        select rd.rd_name
        from refdata rd
        where rd.rd_main_code = 'OKTELL.SirenaResults' and
              rd.rd_tag = :pcl_res and
              rd.rd_parent_id is not null
        into :result;

        pcl_res_old = pcl_res;
      end

      suspend;
    end
  end
  else
  if (object_kind_name = 'Kontract') then
  begin
    for select pcl.pcl_res, pcl.pcl_called_date, pcl.pcl_memo,
               (
               case pant_project_type
                 when 1 then 'Пантера'
                 when 4 then 'Сирена'
               end),
               cl.c_name_fio
        from pantera_clients pcl
        inner join pantera_kontracts pantk on pantk.pant_id = pcl.pcl_from_pan_id
        inner join kontstate ks on ks.ks_from_k_id = pantk.pant_from_k_id
        inner join clients cl on cl.c_id = pcl.pcl_from_c_id
        where pantk.pant_from_k_id = :object_id and
              pantk.pant_project_type in (1, 4) and
              current_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1
        order by cl.c_name_fio, pcl.pcl_called_date desc
        into :pcl_res, :called_date, :memo, :project, :client_fio
    do
    begin
      result = '';

      if (pcl_res <> pcl_res_old) then
      begin
        result = null;

        select rd.rd_name
        from refdata rd
        where rd.rd_main_code = 'OKTELL.SirenaResults' and
              rd.rd_tag = :pcl_res and
              rd.rd_parent_id is not null
        into :result;

        pcl_res_old = pcl_res;
      end

      suspend;
    end
  end
end
