CREATE PROCEDURE AAREP_CALC_AMORT_KOMMIS_T_OLD(
    KONT_ID DECIMAL(18,0),
    DATE_BEGIN DATE,
    DATE_END DATE)
RETURNS (
    RES_AMOUNT_BEGIN DECIMAL(18,4),
    RES_AMOUNT_END DECIMAL(18,4),
    RES_AMOUNT_PERIOD DECIMAL(18,4),
    RES_DISB_DATE DATE,
    RES_LAST_DATE DATE,
    RES_KRED_AMOUNT DECIMAL(18,4),
    RES_KOMMIS_AMOUNT DECIMAL(18,4),
    RES_PERC_DAILY DECIMAL(18,9),
    RES_KOMMIS_PAYED DECIMAL(18,4))
AS
declare variable curr_days1 decimal(18,4);
declare variable curr_days2 decimal(18,4);
declare variable kont_state integer;
declare variable antidays decimal(18,9);
declare variable max_days decimal(18,4);
declare variable is_closed_end integer;
declare variable res_next_date date;
declare variable res_next_amount decimal(18,4);
declare variable day_all integer;
declare variable res_kommis_last decimal(18,4);
declare variable ttt decimal(18,4);
declare variable day_period integer;
declare variable dd_temp date;
declare variable is_closed_begin integer;
declare variable sh_last_date date;
begin
-- Старая версия процедуры до 23.10.2013
  is_closed_end = 0;
  is_closed_begin = 0;
  res_amount_begin = 0;
  res_amount_end = 0;
  res_kommis_payed = 0;
  res_amount_period = 0;
  max_days = 0;
  curr_days1 = 0;
  curr_days2 = 0;
  res_perc_daily = 0;
  select sum(sh_kommis_monthly), min(sh_date), max(k_amount_approved), max(sh_date)
  from schedule, kontract
  where sh_from_k_id = :kont_id and
        k_id = :kont_id
  into :res_kommis_amount, :res_disb_date, :res_kred_amount, :sh_last_date;
  if (res_kommis_amount is null) then
    res_kommis_amount = 0;

  if (res_kommis_amount <> 0) then
  begin
    if ((res_disb_date >= :date_begin) and
        (res_disb_date <= :date_end)) then
      res_kommis_payed = res_kommis_amount;

    /* на начало */
    if (date_begin < res_disb_date) then
    begin
      res_amount_begin = 0;
      is_closed_begin = 1;
    end
    else
    begin
      if (date_begin > sh_last_date) then
      begin
        res_amount_begin = res_kommis_amount;
        is_closed_begin = 1;
      end
    end

    if (is_closed_begin = 0) then
    begin
      dd_temp = date_begin - 1;
      select sum(sh_kommis_monthly), max(sh_date)
      from schedule
      where sh_from_k_id = :kont_id and
            sh_date <= :dd_temp
      into :res_kommis_last, :res_last_date;
      if (res_kommis_last is null) then
        res_kommis_last = 0;

      select sh_date, sh_kommis_monthly
      from schedule
      where sh_from_k_id = :kont_id and
            sh_date > :dd_temp
      order by sh_date
      rows 1
      into :res_next_date, :res_next_amount;
      if (res_next_date is null) then
        res_next_date = res_last_date;
      if (res_next_amount is null) then
        res_next_amount = 0;

      if (res_next_date > res_last_date) then
      begin
        day_all = (res_next_date - res_last_date);
        day_period = (dd_temp - res_last_date);
        res_amount_begin = res_kommis_last;
        ttt = 0;
        if (day_all <> 0) then
        begin
          ttt = cast(cast(day_period as decimal(15,7)) / cast(day_all as decimal(15,7)) as decimal(15,7));
          ttt = cast(res_next_amount * ttt as decimal(15,2));
        end
        res_amount_begin = res_amount_begin + ttt;
      end
    end

    /* на конец */
    select ks_state
    from kontstate
    where ks_from_k_id = :kont_id and
          cast(ks_date_begin as date) <= :date_end and
          ks_date_end >= :date_end and
          ks_ol_is_delete = 0 rows 1
    into :kont_state;
    if (kont_state <> 1) then
    begin
      is_closed_end = 1;
    end
    else
      res_amount_end = 0;

    if (is_closed_end = 0) then
    begin
      select sum(sh_kommis_monthly), max(sh_date)
      from schedule
      where sh_from_k_id = :kont_id and
            sh_date <= :date_end
      into :res_kommis_last, :res_last_date;
      if (res_kommis_last is null) then
        res_kommis_last = 0;

      select sh_date, sh_kommis_monthly
      from schedule
      where sh_from_k_id = :kont_id and
            sh_date > :date_end
      order by sh_date
      rows 1
      into :res_next_date, :res_next_amount;
      if (res_next_date is null) then
        res_next_date = res_last_date;
      if (res_next_amount is null) then
        res_next_amount = 0;

      if (res_next_date > res_last_date) then
      begin
        day_all = (res_next_date - res_last_date);
        day_period = (date_end - res_last_date);
        res_amount_end = res_kommis_last;
        ttt = 0;
        if (day_all <> 0) then
        begin
          ttt = cast(cast(day_period as decimal(15,7)) / cast(day_all as decimal(15,7)) as decimal(15,7));
          ttt = cast(res_next_amount * ttt as decimal(15,2));
        end
        res_amount_end = res_amount_end + ttt;
      end
    end
    else
      res_amount_end = res_kommis_amount;

    if (date_begin < res_disb_date) then
    begin
      res_amount_begin = 0;
      is_closed_begin = 1;
    end
    else
    begin
      if (date_begin > sh_last_date) then
      begin
        res_amount_begin = 0;
        is_closed_begin = 1;
      end
      else
        res_amount_begin = res_kommis_amount - res_amount_begin;
    end

    if (res_kommis_payed <> 0) then
      res_amount_begin = 0;

    if (date_end >= sh_last_date) then
      res_amount_end = 0;
    else
    begin
      if (date_end < res_disb_date) then
        res_amount_end = 0;
      else
        res_amount_end = res_kommis_amount - res_amount_end;
    end
    res_amount_period = (:res_amount_end - :res_amount_begin - :res_kommis_payed);
  end
  suspend;
end
