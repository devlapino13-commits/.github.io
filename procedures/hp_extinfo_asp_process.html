CREATE PROCEDURE HP_EXTINFO_ASP_PROCESS(
    EIL_ID D_ID)
AS
declare variable BLS_ID D_ID;
declare variable EIL_STATUS D_SMALLINT;
declare variable JSON_STR D_BLOB_TEXT;
declare variable ERROR_STATE D_STR_255;
declare variable C_ID D_ID;
declare variable UN_ID D_ID;
declare variable CONFIG D_STR_CODE;
declare variable COUNTER D_INTEGER;
declare variable SUMMA D_CURRENCY;
declare variable MA_ID D_ID;
begin
-- RFCRU-3940 обработка ответа
  select eil.eil_from_bls_id,
         eil.eil_status
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eil.eil_id = :eil_id and
        eic.eic_code = 'asp_check'
  into bls_id,
       eil_status;

  select icq.icq_from_c_id,
         icq.icq_from_un_id,
         icq.icq_config,
         ma.ma_id
  from income_check_querys icq
  inner join miniappclient mac on mac.mac_id = icq.icq_from_mac_id
  inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
  where icq.icq_from_eil_id = :eil_id
  order by icq.icq_create_datetime desc
  rows 1
  into c_id,
       un_id,
       config,
       ma_id;

  if (eil_status = 2) then
  begin

    select cast(blx_blob as d_blob_text)
    from hp_get_blobex_blob(:bls_id)
    into json_str;

    if (row_count is null or coalesce(json_str, '') = '') then
      exit;

    error_state = cast(f_json_str_get_value(json_str, 'responseCode', null) as d_str_255);

    if (error_state = 'PROVIDER_ERROR') then
    begin

      select count(icq.icq_id)
      from income_check_querys icq
      where icq.icq_status = -1 and
            icq.icq_from_c_id = :c_id
      into counter;

      update income_check_querys
      set icq_status = -1
      where icq_from_eil_id = :eil_id;

      if (counter < 1) then    -- Если первый запрос вернул ошибку то автоматически отправляем второй запрос (Пункт 3)
        execute procedure hp_income_new_query(:ma_id, :un_id, :config);

    end
    else if (:config = 'asp_check') then
    begin
      summa = cast(f_json_str_get_value(json_str, f_format('sddFamilyMembers'), 0) as d_currency);

      update income_check_querys icq
         set icq.icq_status = 2,
             icq.icq_payment_sum = :summa
      where icq_from_eil_id = :eil_id;
    end
  end

  if (eil_status in (-2, -3)) then
  begin
    update income_check_querys icq
       set icq.icq_status = -2
    where icq_from_eil_id = :eil_id;
  end
end
