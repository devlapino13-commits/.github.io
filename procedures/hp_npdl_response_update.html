CREATE PROCEDURE HP_NPDL_RESPONSE_UPDATE(
    EIL_ID D_ID)
AS
declare variable JSON_STR D_STR_LONG;
declare variable BLS_ID D_ID;
declare variable C_ID D_ID;
declare variable EIL_STATUS D_INTEGER;
declare variable POSITION_WORK D_STR_255;
declare variable DATE_APPOINTMENT D_DATE;
declare variable DATE_DISMISSAL D_DATE;
declare variable SURNANME D_STR_255;
declare variable NAME D_STR_255;
declare variable PATRONYMIC D_STR_255;
declare variable SURNANME_TUNDUC D_STR_255;
declare variable NAME_TUNDUC D_STR_255;
declare variable PATRONYMIC_TUNDUC D_STR_255;
declare variable C_NAME_FIO D_STR_255;
declare variable ITEMS_COUNT D_INTEGER;
declare variable IDX D_INTEGER;
declare variable BASEPATH D_STR_255;
declare variable UN_ID D_ID;
begin
  select eil_from_bls_id, eil_status, eil_from_c_id, eil_from_un_id
  from extinfolist, extinfocfg
  where eic_code = 'check_npdl' and
        eil_from_eic_id = eic_id and
        eil_id = :eil_id
  into bls_id, eil_status, c_id, un_id;

  if (coalesce(eil_status, 0) in (0, -1)) then
    exit;

  select cast(blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id)
  into json_str;

  select upper(fl.fl_name1),
         upper(fl.fl_name2),
         upper(fl.fl_name3),
         cl.c_name_fio
  from fiolist fl
  inner join clients cl on cl.c_from_fl_id = fl.fl_id
  where cl.c_id = :c_id
  into: surnanme, name, patronymic, c_name_fio;

  if (c_name_fio like '%УУЛУ%') then
     surnanme = surnanme || ' УУЛУ';
  if (c_name_fio like '%КЫЗЫ%') then
     surnanme = surnanme || ' КЫЗЫ';
  if (c_name_fio like '%UULU%') then
     surnanme = surnanme || ' UULU';
  if (c_name_fio like '%KYZY%') then
     surnanme = surnanme || ' KYZY';

  idx = 0;
  items_count = f_json_str_get_count(:json_str, 'result');
  while(idx < items_count)
  do
  begin
    basepath = 'result[' || :idx || ']';

    surnanme_tunduc = upper(f_json_str_get_value(:json_str, basepath || '.surname', ''));
    name_tunduc = upper(f_json_str_get_value(:json_str, basepath || '.name', ''));
    patronymic_tunduc = upper(f_json_str_get_value(:json_str, basepath || '.patronymic', ''));

    if (surnanme = surnanme_tunduc and
        name = name_tunduc and
        patronymic = patronymic_tunduc) then -- если ФИО у нас в бд = полученному ФИО, то считаем что клиент нпдл  RFCKG-4801
    begin
      position_work = f_json_str_get_value(:json_str, basepath || '.position', '');

      date_appointment = iif(f_json_str_get_value(:json_str, basepath || '.dateAppointment', '') <> '',
                             cast(substring(f_json_str_get_value(:json_str, basepath || '.dateAppointment', '') from 1 for 10) as date), null);

      date_dismissal = iif(f_json_str_get_value(:json_str, basepath || '.dateAppointment', '') <> '',
                           cast(substring(f_json_str_get_value(:json_str, basepath || '.dateDismissal', '') from 1 for 10) as date), null);

      update clients cl
      set cl.c_pft_risk_koef = 2, -- устанавливаем высокий уровень
          cl.c_pft_risk_memo = :position_work, -- устанавливаем обоснование
          cl.c_is_public = 1  -- устанавливаем как "публичное лицо"
      where (cl.c_id = :c_id);

      update or insert into public_persons (pp_from_c_id, pp_date_begin, pp_date_end, pp_memo,
                                            pp_is_delete, pp_from_un_id, pp_datetime)
                                    values (:c_id, :date_appointment, :date_dismissal, :position_work,
                                            0, :un_id, current_timestamp)
      matching (pp_from_c_id);

      exit;
    end
    idx = idx + 1;
  end
  -- если нет в списке, то обновляем поле c_is_public
  update clients cl
  set cl.c_is_public = 0
  where (cl.c_id = :c_id);

end
