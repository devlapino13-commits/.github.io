CREATE PROCEDURE FM_CONTPERSON_FIND(
    C_ID DECIMAL(18,0),
    FIND_PHONE VARCHAR(255))
RETURNS (
    ID INTEGER,
    NAME VARCHAR(100),
    RELATION VARCHAR(100),
    CELL_NO VARCHAR(255),
    CELL_NO_ADD VARCHAR(255),
    LANGUAGE_ID INTEGER,
    LANGUAGE VARCHAR(100),
    ADL_ID DECIMAL(18,0),
    ADDRESS VARCHAR(255),
    COMMENTS VARCHAR(255),
    VERIF_IGNORE SMALLINT,
    CREATE_DATE DATE)
AS
begin
  for select cc_id, coalesce(cc_name, ''), coalesce(cc_status, ''), cc_cell_no, cc_cell_no_add, cc_language,
             cc_from_adl_id, cc_svz_comments, cc_verif_ignore, cc_create_date
      from contacts
      where cc_from_c_id = :c_id and
            (cc_cell_no = :find_phone or
            cc_cell_no_add = :find_phone)
      order by cc_create_date desc
      rows 1
      into id, name, relation, cell_no, cell_no_add, language_id, adl_id, comments, verif_ignore, create_date
  do
  begin
    if (name = '') then
      name = 'ФИО не указано';
-- Если попадается неправильное поле "Кем приходится", очищаем
    if (not exists(select 1
                   from fm_uniref_list(null, null, null, 'ContactRelationList', null, null, null)
                   where text1 = :relation)) then
      relation = '';

    language = '';
    if (language_id is not null) then
    begin
      select rd_name
      from refdata
      where rd_id = :language_id
      into language;
    end
    address = 'Адрес не заполнен.';
    if (adl_id is null) then
      adl_id = 0;
    if (adl_id <> 0 and
        right(adl_id, 9) <> '000000000') then
    begin
      select adl_address_str
      from addrlist
      where adl_id = :adl_id
      into :address;
    end
    suspend;
  end
end
