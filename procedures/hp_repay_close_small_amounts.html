CREATE PROCEDURE HP_REPAY_CLOSE_SMALL_AMOUNTS(
    SMALL_AMOUNT DECIMAL(18,4),
    LAST_REC_NO INTEGER)
AS
declare variable next_os decimal(18,4) = 0;
declare variable next_perc decimal(18,4) = 0;
declare variable s_rec_no integer;
declare variable isupdated integer;
declare variable sh_new_os decimal(18,4);
declare variable sh_new_perc decimal(18,4);
declare variable res_amount decimal(18,4) = 0;
declare variable s_running_total decimal(18,4) = 0;
begin
-- Если общая сумма платежа меньше 500 руб., платеж перекидываем на следующий
  if (small_amount = 0) then
    exit;

  if (not(exists(select *
                 from temp_vrepay
                 where s_id >= 0 and
                       (sh_new_os + sh_new_perc) > 0 and
                       (sh_new_os + sh_new_perc) <= :small_amount and
                       s_date is not null))) then
    exit;

  next_os = 0;
  next_perc = 0;
  isupdated = 0;
  for select s_rec_no, sh_new_os, sh_new_perc
      from temp_vrepay
      where s_id >= 0 and
            s_date is not null
      into :s_rec_no, sh_new_os, sh_new_perc as cursor cur
  do
  begin
    if ((s_rec_no < last_rec_no) and
        (sh_new_os + sh_new_perc > 0)) then
    begin
      if (sh_new_os + sh_new_perc + next_os + next_perc <= :small_amount) then
      begin
        next_os = next_os + sh_new_os;
        next_perc = next_perc + sh_new_perc;
        update temp_vrepay
        set sh_new_os = 0,
            sh_new_perc = 0
        where s_rec_no = :s_rec_no;
      end
      else
      begin
        if (next_os + next_perc > 0) then
        begin
          update temp_vrepay
          set sh_new_os = sh_new_os + :next_os,
              sh_new_perc = :sh_new_perc + :next_perc
          where s_rec_no = :s_rec_no;
          exit;
        end
      end
    end
  end
  if (next_os + next_perc > 0) then
  begin
    update temp_vrepay
    set sh_new_os = sh_new_os + :next_os,
        sh_new_perc = :sh_new_perc + :next_perc
    where s_rec_no = :last_rec_no;
  end

end
