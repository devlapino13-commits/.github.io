CREATE PROCEDURE MC_QR_PAYMENT_RESULT_DETAILS(
    MCL_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    CHECK_NUMBER D_STR_100 = null)
RETURNS (
    TSP_NAME D_STR_255,
    IQP_DATETIME D_DATETIME,
    IQP_CHECK_NUMBER D_STR_100,
    IQP_AMOUNT D_CURRENCY,
    IQP_STATE_STR D_STR_100,
    IQP_FEE DECIMAL(18,4),
    TO_PAY D_CURRENCY,
    PAYM_METHOD D_STR_100)
AS
declare variable c_id d_id;
declare variable db_id d_integer;
declare variable cs_id d_integer;
declare variable iqp_state d_integer;
declare variable lang_code d_str_lang;
begin
  db_id = f_get_db_id();
  lang_code = f_get_lang_code(null, :mcl_id);
  iqp_fee = 0;

  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id --18331
  into c_id; --1090101387630

  if (c_id is null or db_id not in (39801, 41701)) then
  begin
    suspend;
    exit;
  end
  for select cs.cs_name, qp.iqp_datetime, qp.iqp_check_number, qp.iqp_amount, qp.iqp_ipc_state, qp.iqp_from_cs_id,
             qp.iqp_fee
      from ipc_qr_payments qp
      inner join clientshops cs on cs.cs_id = qp.iqp_from_cs_id
      inner join clients cl on cl.c_id = cs.cs_from_c_id
      where qp.iqp_date between :rep_date_begin and :rep_date_end and
            cs.cs_from_c_id = :c_id and
            (:check_number is null or
            qp.iqp_check_number = :check_number)
      into tsp_name, iqp_datetime, iqp_check_number, iqp_amount, iqp_state, cs_id, iqp_fee

  do
  begin

    if (iqp_state = 2) then
      iqp_state_str = f_lang(lang_code, 'Успешно');

    if (iqp_state <> 2) then
      iqp_state_str = f_lang(lang_code, 'Ошибка');

    to_pay = iqp_amount + iqp_fee;
    paym_method = f_lang(lang_code, 'Оплата картой Элькарт');

    suspend;
  end
end
