CREATE PROCEDURE HP_CONDCHECKIGNORE_VIEW(
    DTBEG D_DATE,
    DTEND D_DATE,
    KIND D_INTEGER,
    STATE D_SMALLINT)
RETURNS (
    CCI_ID D_INTEGER,
    CCI_OBJECT_KIND D_INTEGER,
    CCI_OBJECT_ID D_ID,
    CCI_FROM_CDM_ID D_INTEGER,
    CCI_COMMENT D_STR_LONG,
    CCI_DATE_END D_DATE,
    CCI_FROM_UN_ID D_ID,
    CCI_DATETIME D_DATETIME,
    CCI_STATE D_SMALLINT,
    CCI_STATE_UN_ID D_ID,
    CCI_STATE_DATETIME D_DATETIME,
    UN_FIO D_STR_100,
    UN_FIO_STATE D_STR_100,
    CDM_CHECK_KIND D_STR_255,
    CDM_MEMO D_STR_255,
    CDM_DATE_BEGIN D_DATE,
    CDM_DATE_END D_DATE,
    CDM_IS_ACTIVE D_INTEGER,
    OBJ_NO D_STR_20,
    OBJ_DT D_DATE,
    OBJ_FLD D_STR_100,
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    OFFICE_C_ID D_ID,
    OFFICE_C_NAME D_STR_100,
    PR_ID D_INTEGER,
    PR_NAME D_STR_CODE,
    PR_IS_IPF D_SMALLINT)
AS
declare variable obj1_kind d_integer;
declare variable obj2_kind d_integer;
declare variable dtinit d_date;
begin
  obj1_kind = f_objkind('kontract', 'id');
  obj2_kind = f_objkind('miniapp', 'id');
  dtbeg = coalesce(dtbeg, current_date);
  dtend = coalesce(dtend, current_date);

  for
    select
      cci_id,
      cci_object_kind,
      cci_object_id,
      cci_from_cdm_id,
      cci_comment,
      cci_date_end,
      cci_from_un_id,
      cci_datetime,
      cci_state,
      cci_state_un_id,
      cci_state_datetime,
      u1.un_fio, 
      u2.un_fio,
      cdm_check_kind,
      '(' || cdm_id || ') ' || cdm_memo,
      cdm_date_begin,
      cdm_date_end,
      cdm_is_active
    from condcheckignore
      inner join conditionmain
        on cdm_id = cci_from_cdm_id
      inner join username as u1
        on u1.un_id = cci_from_un_id
      left join username as u2
        on u2.un_id = cci_state_un_id
    where cci_datetime >= :dtbeg and cci_datetime < (:dtend + 1)
      and cci_object_kind = coalesce(:kind, cci_object_kind)
      and cci_state = coalesce(:state, cci_state)
  into
    :cci_id,
    :cci_object_kind,
    :cci_object_id,
    :cci_from_cdm_id,
    :cci_comment,
    :cci_date_end,
    :cci_from_un_id,
    :cci_datetime,
    :cci_state,
    :cci_state_un_id,
    :cci_state_datetime,
    :un_fio,
    :un_fio_state,
    :cdm_check_kind,
    :cdm_memo,
    :cdm_date_begin,
    :cdm_date_end,
    :cdm_is_active
  do
  begin
    dtinit = cci_datetime;
    if (cci_object_kind = obj1_kind) then
    begin
      select
        k_kontract_no,
        k_disb_date,
        pr_id,
        pr_name,
        iif(pr_code containing '.ipf.', 1, 0),
        kal_new_kf_name,
        kal_new_reg_id,
        reg_name,
        c_id,
        c_name_fio
      from kontract
        inner join product
          on pr_id = k_from_pr_id
        inner join knaktlist
          on kal_from_kf_id = k_from_kf_id
            and :dtinit between kal_date and kal_date_end
            and knaktlist.is_delete = 0
        inner join region
          on reg_unicode = kal_new_reg_id
        inner join clients
          on c_id = kal_new_kf_officer
      where k_id = :cci_object_id
      into
        :obj_no,
        :obj_dt,
        :pr_id,
        :pr_name,
        :pr_is_ipf,
        :obj_fld,
        :reg_id,
        :reg_name,
        :office_c_id,
        :office_c_name;
    end

    if (cci_object_kind = obj2_kind) then
    begin
      select
        ma_no,
        ma_date_oper,
        pr_id,
        pr_name,
        iif(pr_code containing '.ipf.', 1, 0),
        kal_new_kf_name,
        kal_new_reg_id,
        reg_name,
        c_id,
        c_name_fio
      from miniapp
        inner join product
          on pr_id = ma_from_pr_id
        inner join knaktlist
          on kal_from_kf_id = ma_from_kf_id
            and :dtinit between kal_date and kal_date_end
            and knaktlist.is_delete = 0
        inner join region
          on reg_unicode = kal_new_reg_id
        inner join clients
          on c_id = kal_new_kf_officer
      where ma_id = :cci_object_id
        and ma_date_end = '31.12.9999'
      into
        :obj_no,
        :obj_dt,
        :pr_id,
        :pr_name,
        :pr_is_ipf,
        :obj_fld,
        :reg_id,
        :reg_name,
        :office_c_id,
        :office_c_name;
    end

    suspend;
  end
end
