CREATE PROCEDURE HP_KOMMISREFUND_INS_CHECK
AS
declare variable kmr_amount d_currency;
declare variable kmr_ok_cell_no d_str_20;
declare variable k_id id_type;
declare variable c_id id_type;
declare variable kmr_check_esm_id id_type;
declare variable kmr_id id_type;
begin
  for select kmr.kmr_id, kmr.kmr_from_k_id, kmr.kmr_from_c_id, kmr.kmr_amount, kmr.kmr_ok_cell_no
      from kommisrefund_cards kmr
      where kmr.kmr_ok_cell_no <> '' and
            kmr.kmr_ok_cell_no starting '996'
            and kmr.kmr_id <> 20457
      into :kmr_id, :k_id, :c_id, :kmr_amount, :kmr_ok_cell_no
  do
  begin
    insert into extsysmain (esm_project, esm_action, esm_date, esm_account, esm_amount, esm_object_kind, esm_object_id,
                            esm_from_k_id)
    values ('transfer_on_phone', 'check', current_date, :kmr_ok_cell_no, :kmr_amount, 0, :c_id, :k_id);

    for select esm_id
        from extsysmain esm
        where esm.esm_project = 'transfer_on_phone' and
              esm.esm_action = 'check' and
              esm.esm_account = :kmr_ok_cell_no and
              esm.esm_amount = :kmr_amount and
              esm.esm_object_kind = 0 and
              esm.esm_object_id = :c_id and
              esm.esm_from_k_id = :k_id and
              esm.esm_date = current_date
        into :kmr_check_esm_id
    do
    begin
      update kommisrefund_cards kmr
      set kmr.kmr_check_esm_id = :kmr_check_esm_id
      where kmr.kmr_id = :kmr_id and
            kmr.kmr_from_k_id = :k_id and
            kmr.kmr_from_c_id = :c_id;
    end
  end
end
