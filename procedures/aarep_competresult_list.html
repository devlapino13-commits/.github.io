CREATE PROCEDURE AAREP_COMPETRESULT_LIST(
    CPP_ID INTEGER)
RETURNS (
    CPR_ID INTEGER,
    CPR_FROM_CPP_ID INTEGER,
    CPR_GROUP_NAME VARCHAR(255),
    CPR_OBJECT_KIND INTEGER,
    CPR_OBJECT_ID DECIMAL(18,0),
    CPR_CALCED_POINTS DECIMAL(18,4),
    CPR_ADDITIONAL_POINTS DECIMAL(18,4),
    CPR_ORDER_IN_GROUP INTEGER,
    CPR_FINAL_POINTS DECIMAL(18,4),
    CPR_FINAL_ORDERNO INTEGER,
    CPR_RESULT_KIND INTEGER,
    CPR_COMMENT VARCHAR(255),
    CPR_FROM_UN_ID DECIMAL(18,0),
    CPR_DATETIME TIMESTAMP,
    CPR_OBJECT_NAME VARCHAR(255))
AS
begin
  for select cpr_id, cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
             cpr_additional_points, cpr_order_in_group, cpr_final_points, cpr_final_orderno, cpr_result_kind,
             cpr_comment, cpr_from_un_id, cpr_datetime
      from competresult
      where cpr_from_cpp_id = :cpp_id
      into :cpr_id, :cpr_from_cpp_id, :cpr_group_name, :cpr_object_kind, :cpr_object_id, :cpr_calced_points,
           :cpr_additional_points, :cpr_order_in_group, :cpr_final_points, :cpr_final_orderno, :cpr_result_kind,
           :cpr_comment, :cpr_from_un_id, :cpr_datetime
  do
  begin
    cpr_object_name = '';
    if (cpr_object_kind = -1) then
    begin
      select rd_name
      from refdata
      where rd_id = :cpr_object_id
      into :cpr_object_name;
    end
    if (cpr_object_kind = 0) then
    begin
      select c_name_fio
      from clients
      where c_id = :cpr_object_id
      into :cpr_object_name;
    end
    if (cpr_object_kind = 7) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :cpr_object_id
      into :cpr_object_name;
    end
    suspend;
  end
end
