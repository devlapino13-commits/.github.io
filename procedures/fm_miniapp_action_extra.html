CREATE PROCEDURE FM_MINIAPP_ACTION_EXTRA(
    UN_ID DECIMAL(18,0),
    ACTION_KIND VARCHAR(100),
    MINIAPP_DATA BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    LOANMINIAPPID INTEGER)
RETURNS (
    MA_RESULT_ID INTEGER,
    MA_RESULT_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable json_key varchar(100);
declare variable json_value varchar(255);
declare variable txt varchar(255);
declare variable txt_large blob sub_type 1 segment size 80;
declare variable sh_id integer;
declare variable sh_date date;
declare variable sh_grace integer;
declare variable db_id integer;
begin
  /*  select cm_database_id from company
  into :db_id; */
  :db_id = f_get_db_id();

  ma_result_txt = '';
  ma_result_id = 1;

  if (action_kind = 'doSchedule') then
  begin
    for select json_key, json_value
        from fm_json_parse(:miniapp_data, 0, 'sh_update')
        into :json_key, :json_value
    do
    begin
      if (json_key = 'endofobject') then
      begin
        select res_id, res_txt
        from hp_make_schedule(:un_id, 11, :loanminiappid, :sh_id, :sh_date, :sh_grace)
        into ma_result_id, ma_result_txt;

        if (ma_result_id = 1) then
        begin
          select res_json
          from hp_json_from_sql('array', 'schedule', '
          select sh_id as id,
                 sh_date,
                 sh_os as os,
                 sh_perc as perc,
                 sh_grace as grace,
                 sh_perc_tax as tax,
                 sh_all,
                 sh_all_with_tax as sh_all_tax
          from fm_schedule_list(11, ' || :loanminiappid || ')
          where sh_id>0')
          into ma_result_txt;
          ma_result_txt = BLOB_APPEND('{', ma_result_txt, '}');
        end
        suspend;
      end
      else
      begin
        if (json_key = 'sh_id') then
          sh_id = json_value;
        if (json_key = 'sh_date') then
          sh_date = json_value;
        if (json_key = 'sh_grace') then
          sh_grace = json_value;
      end
    end
    exit;
  end

  if (action_kind = 'getSchedule') then
  begin
    select res_json
    from hp_json_from_sql('array', 'schedule', '
    select sh_id as id,
           sh_date,
           sh_os as os,
           sh_perc as perc,
           sh_grace as grace,
           sh_perc_tax as tax,
           sh_all,
           sh_all_with_tax as sh_all_tax
    from fm_schedule_list(11, ' || :loanminiappid || ')
    where sh_id>0')
    into ma_result_txt;
    if (ma_result_txt is null) then
      ma_result_txt = '';
    ma_result_txt = BLOB_APPEND('{', ma_result_txt, '}');
    suspend;
    exit;
  end

  if (action_kind = 'getBusuness_subtype') then
  begin
    ma_result_txt = '';
    txt_large = '';

    txt_large = f_format('{"id":"0","name":"%1"}' , iif(db_id <> 94401, 'Не указан', 'Qeyd olunmayıb'));
    for select '{"id": "' || bs_id || '","name":"' || replace(trim(bs_name || iif(coalesce(bt_name, '') <> '', '(' || bt_name || ')', '')), '"', '\"') || '"}'
        from bustype, bussubtype
        where bs_from_bt_id = bt_id and
              coalesce(bs_is_delete, 0) = 0 and
              bs_id <> 0
        order by bt_name, bs_name
        into txt
    do
    begin
      txt_large = BLOB_APPEND(txt_large, ',' || txt);
    end
    ma_result_txt = BLOB_APPEND(ma_result_txt, '"business_subtypes":[', txt_large, ']');

    ma_result_txt = BLOB_APPEND('{', ma_result_txt, '}');
    suspend;
    exit;
  end
end
