CREATE PROCEDURE UXP$UXGRID_START_UPDATE(
    GRID_UUID D_UUID,
    GRID_UUID_CODE D_STR_CODE)
AS
declare variable pos d_integer;
begin
  if (coalesce(grid_uuid, '') = '') then
  begin
    if (coalesce(grid_uuid_code, '') <> '') then
    begin
      pos = position('__', grid_uuid_code);
      if (pos > 0) then
      begin
        grid_uuid_code = substring(grid_uuid_code from pos + 2);
        grid_uuid_code = replace(grid_uuid_code, '_', '-');
      end

      grid_uuid = uxf$uniq(grid_uuid_code);
    end else
      exception e_default 'Emptry grid uuid';
  end

  uxf$syncmode('on');

  update ux$gritem
  set ugi_bnd_id = -2 -- Временная метка об аннулировании
  where ugi_ugr_uuid = :grid_uuid;

  delete from ux$band
  where bnd_grid_uuid = :grid_uuid;
end
