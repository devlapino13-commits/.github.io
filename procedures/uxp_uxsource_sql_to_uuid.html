CREATE PROCEDURE UXP$UXSOURCE_SQL_TO_UUID(
    SQL_TEXT D_STR_8100)
RETURNS (
    RES_UUID D_UUID)
AS
declare variable res_code d_str_code;
begin
  res_code = '';
  res_uuid = null;

  if (coalesce(sql_text, '') <> '') then
  begin
    sql_text = lower(sql_text);
    sql_text = replace(sql_text, ' ', ''); -- убираем пробелы
    sql_text = replace(sql_text, ascii_char(13), ''); -- убираем enter
    sql_text = replace(sql_text, ascii_char(10), ''); -- убираем перевод строки
    sql_text = replace(sql_text, ascii_char(9), '');  -- убираем табулации
  
    res_code = f_md5_blob(sql_text);
  
    res_uuid = char_to_uuid(left(res_code, 8) ||
               '-' || substring(res_code from 9 for 4) ||
               '-' || substring(res_code from 13 for 4) ||
               '-' || substring(res_code from 17 for 4) ||
               '-' || substring(res_code from 21 for 12));
  end

  suspend;
end
