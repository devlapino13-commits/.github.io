CREATE PROCEDURE FM_CREL_ADD(
    C_ID_1 DECIMAL(18,0),
    C_ID_2 DECIMAL(18,0),
    REL_ID VARCHAR(50),
    MEMO VARCHAR(255),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_ID INTEGER,
    RES_TEXT VARCHAR(100))
AS
declare variable SEX_1 smallint;
declare variable SEX_2 smallint;
declare variable REL_NAME varchar(100);
declare variable LANG D_STR_LANG;
begin
  res_text = '';
  res_id = 0;
  lang = f_get_lang_code(un_id, null);

  if (coalesce(c_id_1, 0) = 0) then
  begin
    res_text = f_lang(lang, 'Клиент не указан');
    res_id = 1;
    suspend;
    exit;
  end

  if (coalesce(c_id_2, 0) = 0) then
  begin
    res_text = f_lang(lang, 'Родственник не указан');
    res_id = 1;
    suspend;
    exit;
  end

  select c_sex
  from clients
  where c_id = :c_id_1
  into sex_1;

  select c_sex
  from clients
  where c_id = :c_id_2
  into sex_2;

  select rel_name
  from relationtype
  where rel_id = :rel_id
  into rel_name;

  -- Сочетания параметров ниже: Запрещены
  if (sex_2 = 1 and
      rel_id not in (1, 3, 5, 7, 9)) then
  begin
    res_text = f_lang(lang, 'Родственник должен быть мужчиной');
    res_id = 1;
    suspend;
    exit;
  end

  if (sex_2 = 0 and
      rel_id not in (2, 4, 6, 8, 9)) then
  begin
    res_text = f_lang(lang, 'Родственница должна быть женщиной');
    res_id = 1;
    suspend;
    exit;
  end

  if ((sex_1 = 1 and
      rel_id = 1) or -- Супруг у мужчины
      (sex_1 = 0 and
      rel_id = 2)) -- Супруга у женщины
  then
  begin
    res_text = f_lang(lang, 'Супруги должны быть разного пола');
    res_id = 1;
    suspend;
    exit;
  end

  -- Нельзя добавлять одну и ту-же пару клиент-родственник
  if (exists(select *
             from clientrelation
             where crel_from_c_id = :c_id_1 and
                   crel_other_c_id = :c_id_2 and
                   crel_is_active = 1)) then
  begin
    res_text = f_lang(lang, 'Данный родственник уже есть в списке, повторое добавление запрещено');
    res_id = 1;
    suspend;
    exit;
  end

  -- Нельзя добавлять еще одного мужа, жену, отца, мать, если уже есть в БД
  if (rel_id in (1, 2, 3, 4, 9) and
      exists(select *
             from clientrelation
             where crel_from_c_id = :c_id_1 and
                   crel_from_rel_id = :rel_id and
                   crel_is_active = 1)) then
  begin
    res_text = f_format(f_lang(lang, '%1 у данного клиента уже есть в списке.'), rel_name);
    res_id = 1;
    suspend;
    exit;
  end

  if (rel_id in (1, 2, 9)) then
  begin
    if (exists(select *
               from hp_client_relation_list(:c_id_1, 0)
               where crel_from_rel_id in ('1', '2', '9'))) then
    begin
        res_text = f_lang(lang, 'Должен быть 1 супруг/супруга');
        res_id = 1;
        suspend;
        exit;
    end

    if (exists(select *
               from hp_client_relation_list(:c_id_2, 0)
               where crel_from_rel_id in ('1', '2', '9'))) then
    begin
        res_text = f_lang(lang, 'Должен быть 1 супруг/супруга');
        res_id = 1;
        suspend;
        exit;
    end
  end

  -- RFCKG-2809 -------------------->
  if (c_id_1 = c_id_2) then
  begin
    res_text = f_lang(lang, 'Клиент не может быть родственником самому себе');
    res_id = 1;
    suspend;
    exit;
  end
  ----------------------------------<

  update or insert into clientrelation (crel_from_c_id, crel_other_c_id, crel_from_rel_id, crel_from_un_id,
                                        crel_datetime, crel_is_active, crel_memo)
  values (:c_id_1, :c_id_2, :rel_id, :un_id, current_timestamp, 1, :memo)
  matching (crel_from_c_id, crel_other_c_id);

  suspend;
end
