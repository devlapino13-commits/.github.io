CREATE PROCEDURE AAREP_REFUND_RATIO(
    REP_DATE D_DATE,
    PLAN_DATE_BEGIN D_DATE,
    PLAN_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_CODE,
    K_KONTRACT_NO D_STR_CODE,
    OL_DATE D_DATE,
    OKS_BALANCE D_CURRENCY,
    OKS_DELINQ_DATE D_DATE,
    CLIENT_COUNT D_INTEGER,
    PR_NAME D_STR_100,
    OPER_KIND D_STR_CODE,
    PLAN_OS D_CURRENCY,
    PLAN_PERC D_CURRENCY,
    DELINQ_OS D_CURRENCY,
    DELINQ_PERC D_CURRENCY,
    OKS_BALANCE_END D_CURRENCY,
    OKS_DATE_END D_DATE,
    PLAN_PERIOD D_STR_CODE,
    KF_NAME D_STR_CODE,
    LO_NAME D_STR_CODE)
AS
declare variable k_id d_id;
declare variable kf_id d_id;
begin
  for select k_id, reg_code || ' ' || reg_name as reg_name, k_kontract_no,
         oks_date_begin, oks_balance, oks_delinq_date, k_kolich_client_approved, pr_name, 'Рефинанс.' as kind,
         oks_date_end, k_from_kf_id
  from oksummary, kontstate, kontract, region, product
  where oks_date_begin = ks_date_begin and
        oks_date_end >= ks_date_begin and
        oks_from_k_id = ks_from_k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_date_begin between '01.03.2020' and '31.07.2020' and
        ks_from_k_id = k_id and
        k_refund = 1 and
        oks_delinq_date between :plan_date_begin and :plan_date_end and
        oks_region_id = reg_unicode and
        k_from_pr_id = pr_id
        and oks_close_state = 0
  union
  select k_id, reg_code || ' ' || reg_name as reg_name, k_kontract_no,
         or_ol_date, oks_balance, oks_delinq_date, k_kolich_client_approved, pr_name, 'Реструкт.' as kind,
         oks_date_end, k_from_kf_id
  from oksummary, operrest, kontract, region, product
  where oks_date_begin = or_ol_date and
        oks_date_end >= or_ol_date and
        oks_from_k_id = or_from_k_id and
        or_ol_is_active = 1 and
        or_ol_date between '01.03.2020' and '31.07.2020' and
        oks_from_k_id = k_id and
        oks_delinq_date between :plan_date_begin and :plan_date_end and
        or_from_reg_id = reg_unicode and
        k_from_pr_id = pr_id
        and oks_close_state = 0
  into k_id, reg_name, k_kontract_no,
       ol_date, oks_balance, oks_delinq_date, client_count, pr_name, oper_kind, oks_date_end, kf_id
  do
  begin
    plan_os = 0;
    plan_perc = 0;
    delinq_os = 0;
    delinq_perc = 0;

    plan_period = extract(year from oks_delinq_date) || '-' || right('0' || extract(month from oks_delinq_date), 2);

    select sum(ok_sh_os), sum(ok_sh_perc), sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date and
--          ok_sh_date = :oks_delinq_date and -- смотреть только первое погашение
          ok_sh_date between :plan_date_begin and :plan_date_end and -- смотреть все погашения до plan_date_end
          ok_type <= 7
    into plan_os, plan_perc, delinq_os, delinq_perc;
    if (plan_os is null) then
      plan_os = 0;
    if (plan_perc is null) then
      plan_perc = 0;
    if (delinq_os is null) then
      delinq_os = 0;
    if (delinq_perc is null) then
      delinq_perc = 0;

    oks_balance_end = 0;
    select oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :ol_date and
          oks_date_end >= :ol_date
          and oks_close_state = 0
    into oks_balance_end;

    kf_name = '';
    lo_name = '';

    select kal_new_kf_name, c_name_fio
    from knaktlist, clients
    where kal_from_kf_id = :kf_id and
          kal_date <= :rep_date and
          kal_date_end >= :rep_date and
          knaktlist.is_delete = 0 and
          kal_new_kf_officer = c_id
    into kf_name, lo_name;

    suspend;
  end
end
