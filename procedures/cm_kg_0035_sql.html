CREATE PROCEDURE CM_KG_0035_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable kc_circle integer;
declare variable ma_app_id integer;
declare variable max_sum curr_type;
declare variable min_sum curr_type;
declare variable max_delinq_day_cli integer;
declare variable kc_amount_approved_old curr_type;
declare variable k_id_old id_type;
declare variable kf_id id_type;
declare variable reg_id integer;
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_percent_per_day perc_type;
declare variable min_perc perc_type;
declare variable pr_product_kind integer;
declare variable k_circle integer;
declare variable k_kommis_perc perc_type;
declare variable k_amount_approved curr_type;
declare variable pr_name varchar(50);
declare variable pr_code varchar(100);
declare variable pr_code_old varchar(100);
declare variable k_percent_per_day_old perc_type;
declare variable ks_date_begin_old date;
declare variable oks_delinq_day_old integer;
declare variable k_percent_per_day_cl_old perc_type;
declare variable k_percent_per_day_cl_old1 perc_type;
declare variable k_ac integer;
declare variable oks_balance_old curr_type;
declare variable activ_balance curr_type;
declare variable k_refund d_smallint;
declare variable oks_delinq_date d_date;
begin
  for select k_disb_date, k_from_kf_id, k_last_reg_id, k_kolich_repay, k_kolich_client_approved,
             round(k_percent_per_day * 365, 1), pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name,
             pr_code, k_from_ma_app_id, kont.k_refund
      from kontract kont, product
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            pr_code containing '.GR.' and
            pr_code containing '.PAR.'
      into :kf_disb_date, :kf_id, :reg_id, :k_kolich_repay, :k_kolich_client_approved, k_percent_per_day,
           :pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code, ma_app_id, k_refund
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;

    --Общие правила для группового стандартного
    --Паспорт пункт 4,2/2
    if (k_kolich_client_approved not between 3 and 5) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Количество клиентов в кредите должен быть от 3 до 5';
      suspend;
    end
    --Паспорт пункт 4,4
    if (k_kolich_repay not between 3 and 18) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Срок кредита должен быть от 3 до 18 мес.';
      suspend;
    end
    /*
    if (pr_name containing 'Мугалим') then min_perc = 30.9;
    else min_perc = 33;
    --Паспорт пункт 4,3 
    if (K_Percent_Per_Day < min_perc) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Процентная ставка не должен быть меньше: 33%';
      suspend;
    end */
    --Общая сумма не может быть меньше 30000=1.5*10000
    if (k_amount_approved < 15000) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Общая сумма для группы не может быть меньше 15000 сомов';
      suspend;
    end
    /*
    --Паспорт пункт 4,5 ДОП%% всегда 3%, с 6 сент.16 года 2.9%
    if (k_kommis_perc <> iif(kf_disb_date < '06.09.16', 3, 2.9)) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Дополнительный процент указан не правильно!';
      suspend;
    end
    */
    --Конец общие правила
    k_ac = 0;
    select k_id, pr_code, round(k_percent_per_day * 365, 1)
    from kontract k, product pr, kontstate
    where k_from_kf_id = :kf_id and
          k_id <> :k_id and
          k_from_pr_id = pr_id and
          k_id = ks_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          :kf_disb_date between ks_date_begin and ks_date_end and
          pr_code containing '.GR.' and
          pr_code not containing '.VYGOD.'
    order by ks_date_begin desc
    rows 1
    into k_id_old, pr_code_old, k_percent_per_day_old;
    if (pr_code_old containing '.PAR.') then
    begin
      res_error_is_critical = iif(k_refund = 1, 0, 2);
      res_error_text = 'Есть активный льготный кредит. Второй льготный кредит не может быть выдан!';
      suspend;
    end
    --Паспорт пункт 4,2 2.Если у ГСО нет активного кредита по ГР-Стандарт
    if (pr_code_old is null or pr_code_old containing '.A.') then
    begin
      select k_id, pr_code, round(k_percent_per_day * 365, 1)
      from kontract k, product pr, kontstate
      where k_from_kf_id = :kf_id and
            k_id <> :k_id and
            k_from_pr_id = pr_id and
            pr_code containing '.GR.' and
            pr_code not containing '.VYGOD.' and
            pr_code not containing '.PAR.' and
            pr_code not containing '.A.' and
            k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin < :kf_disb_date
      order by ks_date_begin desc
      rows 1
      into k_id_old, pr_code_old, k_percent_per_day_old;
    end
    select sum(iif(:kf_disb_date between oks_date_begin and oks_date_end, oks_balance, 0)),
           max(oks_date_begin - oks_delinq_date), max(oks_delinq_date)
    from oksummary oks
    where oks_close_state = 0 and
          oks_from_k_id = :k_id_old
    into oks_balance_old, oks_delinq_day_old, oks_delinq_date;
    --Паспорт пункт 4,2 группа не должена иметь просрочку в активном кредите
    /* if (oks_delinq_day_old > 0) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'В Основном кредите группы есть просроченные дни: '|| oks_delinq_day_old;
      suspend;
    end*/
    k_percent_per_day_old = coalesce(k_percent_per_day_old, 100) - 2;
    if (k_percent_per_day_old > 38.9) then
      k_percent_per_day_old = iif(kf_disb_date < '06.09.16', 42.5, 38.9);
    if (k_percent_per_day_old < 33) then
      k_percent_per_day_old = 33;
    /*
    if (k_percent_per_day <> k_percent_per_day_old) then
    begin 
      res_error_is_critical = 1;
      res_error_text = 'Годовая процентная ставка указано неправильно, должен быть: '||k_percent_per_day_old;
      suspend;
    end
    */
    --Проверка по клиентам группы
    for select kncl.kc_from_c_id, kncl.kc_amount_approved, kncl.kc_circle, cl.c_name_fio
        from knclient kncl, clients cl
        where kncl.kc_amount_approved > 0 and
              cl.c_id = kncl.kc_from_c_id and
              kncl.kc_from_k_id = :k_id
        into :c_id, :kc_amount_approved, :kc_circle, :res_info_text
    do
    begin
      --Паспорт пункт 3,2 Сумма для клиента может быть от 15 000 до 200 000
      min_sum = iif(:kf_disb_date < '09.06.16', 15000, 5000);
      max_sum = 50000;
      if (kc_amount_approved not between min_sum and max_sum) then
      begin
        res_error_is_critical = iif(k_refund = 1, 0, 2);
        res_error_text = 'Сумма для клиента может быть от ' || min_sum || ' до ' || max_sum;
        suspend;
      end
      max_delinq_day_cli = 0;
      if (kc_circle > 1) then
      begin
        for select kncl.kc_amount_approved, kncl.kc_from_k_id, ks_date_begin, round(k_percent_per_day * 365, 1)
            from kontstate ks
            inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
            inner join kontract on ks.ks_from_k_id = k_id
            inner join product on k_from_pr_id = pr_id
            where ks.ks_state = 1 and
                  ks.ks_ol_is_delete = 0 and
                  ks_date_begin < :kf_disb_date and
                  kncl.kc_amount_approved > 0 and
                  kncl.kc_from_c_id = :c_id and
                  pr_code not containing '.IPF.'
            order by ks.ks_date_begin desc
            rows 1
            into :kc_amount_approved_old, :k_id_old, ks_date_begin_old, k_percent_per_day_cl_old
        do
        begin
          if (:k_id_old is null) then
          begin
            res_error_is_critical = 2;
            res_error_text = 'У клиента нет основного кредита, льготный кредит не может быть выдан!';
            suspend;
          end
          -- определяю макимальное число дней прссрочки на предыдущем цикле
          select max(ok.ok_ol_date - ok_sh_date)
          from operkred ok
          where ok.ok_ol_is_active = 1 and
                ok.ok_ol_date <= :kf_disb_date and
                ok.ok_ol_date > ok_sh_date and
                ok_sh_date not between '01.12.20' and '31.01.21' and
                ok_type in (4, 5, 9) and
                ok_from_k_id = :k_id_old and
                ok_os + ok_perc < 0 and
                ok_from_c_id = 0
          into max_delinq_day_cli;
          max_delinq_day_cli = coalesce(max_delinq_day_cli, 0);
          --Паспорт пункт 4,2 группа не должена иметь просрочку в активном кредите
          if (max_delinq_day_cli > 3 and
              not(oks_delinq_date between '01.12.20' and '31.01.21' and
              pr_code containing '.par.' and
              (pr_code containing '.ind.' or pr_code containing '.gr.' or pr_code containing '.birge.') and
              exists(select *
                     from knclient_compens kcc
                     inner join knclient kncl on kncl.kc_id = kcc.kcc_from_kc_id and
                           kc_from_c_id = :c_id
                     where kcc.kcc_date_end > kcc.kcc_date_begin
              ))) then
          begin
            res_error_is_critical = 2;
            res_error_text = 'У клиента есть просрочка в пред.цикле. Льготный кредит не может быть выдан!';
            suspend;
          end
          activ_balance = kc_amount_approved / k_amount_approved * oks_balance_old;
          --Паспорт пункт 4,3 Общий баланс основного и льготного кредитов не может быть более 200 000 сомов
          if (activ_balance > 200000) then
          begin
            res_error_is_critical = 2;
            res_error_text = 'Общий баланс основного и льготного кредитов не может быть более 200 000 сомов!';
            suspend;
          end
        end
      end
    end
  end
end
