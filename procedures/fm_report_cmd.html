      execute procedure hp_self_prohibition_do_query(:m_add_params, :m_obj_kind, :m_obj_id, :un_id);
      res_text = 'Запрос отправлен';
    end
    need_refresh = 1;
    suspend;
  end
  if (m_kind = 'start_antifraud_check') then
  begin
    if (coalesce(:m_add_params, '') = '') then
        exception e_message('У клиента не указан ИИН');

    ac_id = (select ac_id from hp_extinfo_insert_antifraud(:m_add_params, :un_id));
    res_text = 'Запрос отправлен';
    need_refresh = 1;

    select first 1
           c.c_id
    from clients c
    where c.c_inn = :m_add_params
    into :c_id;

    eic_id = (select eic.eic_id from extinfocfg eic where eic.eic_code = 'antifraud_client_check');

    update extinfolist eil
    set
        eil.eil_from_c_id = :c_id
    where eil.eil_from_eic_id = :eic_id
        and eil.eil_object_kind = f_objkind('antifraud_check', 'id')
        and eil.eil_object_id = :ac_id;
    suspend;
  end
  if (m_kind = 'remove_client_from_reject_list') then
  begin
    if (char_length(:m_param) < 10) then
      res_text = 'Необходимо ввести подробное описание';

    else
      update moratory m
      set
        m.m_is_active = 0,
        m.m_comment_officer = :m_param,
        m.m_date_end = current_date
      where m.m_from_c_id = :m_obj_id
        and m.m_is_black = 1
        and m.m_is_active = 1
        and current_date between m.m_date_begin and m.m_date_end;

    need_refresh = 1;
    suspend;
  end
  if (m_kind = 'start_afm_check') then
  begin
    execute procedure hp_extinfo_insert_afm(:m_add_params, :m_obj_id);
    res_text = 'Повторный запрос в ПОД/ФТ отправлен';
    need_refresh = 1;
    suspend;
  end
  if (m_kind = 'send_query_ie_kontur') then
  begin
    -- RFCRU-5314
    need_refresh = 1;
    select res_text
    from hp_send_query_ie_kontur(:m_obj_kind, :m_obj_id, :un_id)
    into res_text;
    suspend;
  end
  if (m_kind = 'cell.application_send_check') then
  begin
    need_refresh = 1;
    select res_text, res_need_rollback
    from fm_cell_application_send_check(:m_obj_id, :un_id)
    into res_text, need_rollback;
    suspend;
  end
  if (m_kind = 'set_pmt_proj') then
  begin
    need_refresh = 1;
    select res_text from hp_extsys_set_pmt_project(:m_add_params)
    into res_text;
    suspend;
  end
  -- Переключение оператора Выдач
  if (m_kind = 'set_disb_proj') then
  begin
    need_refresh = 1;
    select res_text from hp_extsys_set_pmt_project_disb(:m_add_params)
    into res_text;
    suspend;
  end
  if (m_kind = 'send_video_certification_link') then    -- RFCRU-5604
  begin
    need_refresh = 1;
    select res_text
    from hp_send_video_certification_link(:m_obj_kind, :m_obj_id, :m_param, :un_id)
    into res_text;
    suspend;
  end
  if (m_kind in ('change_deport_reason', 'change_deport_start_date', 'change_deport_finish_date', 'change_deport_comment')) then
  begin
    need_refresh = 1;
    select res_text
    from hp_deport_set_change(:m_obj_id, :un_id, :m_kind, :m_param)
    into res_text;
    suspend;
  end
  if (m_kind in ('cancel_deport_info', 'save_deport_info')) then
  begin
    need_refresh = 1;
    select res_text
    from hp_deport_save_or_cancel(:m_obj_id, :un_id, :m_kind)
    into res_text;
    suspend;
  end
