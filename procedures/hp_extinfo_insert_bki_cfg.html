CREATE PROCEDURE HP_EXTINFO_INSERT_BKI_CFG(
    EIC_CODE D_STR_CODE,
    C_ID ID_TYPE,
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    FORCE_INSERT D_SMALLINT,
    UN_ID D_ID)
AS
declare variable eil_id d_id;
declare variable eic_id d_id;
declare variable eic_info_actual_days d_integer;
begin
  select eic_id, eic_info_actual_days
  from extinfocfg
  where eic_code = :eic_code
  into eic_id, eic_info_actual_days;

  select eil_id
  from extinfolist
  where eil_from_eic_id = :eic_id
    and eil_from_c_id = :c_id
    and eil_object_kind = :obj_kind
    and eil_object_id = :obj_id
    and eil_status in (-1, 0, 1, 2)
    and datediff(day, eil_date, current_date) <= :eic_info_actual_days
  order by eil_date desc, eil_create_datetime desc
  rows 1
  into eil_id;

  if (eil_id is null or force_insert = 1) then
  begin
    insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date, eil_object_kind, eil_object_id, eil_is_json, eil_from_un_id)
    values (:c_id, :eic_id, current_date, :obj_kind, :obj_id, 1, :un_id);

    if (eic_code in ('passport_smev', 'kontur_ip_base', 'kontur_ip_la', 'kontur_ip_lic')) then
      post_event :eic_code;
  end
end
