CREATE PROCEDURE FT_RESP_FIO_GET(
    C_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    RES_TEXT D_STR_LONG)
AS
declare variable region_id integer;
declare variable c_id_tmp id_type;
declare variable fio str_100;
declare variable k_id id_type;
begin
    for select ks.ks_from_k_id
    from kontstate ks
    inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
    where ks.ks_state = 1
    and ks.ks_date_begin <= :rep_date
    and ks.ks_ol_is_delete = 0
    and kncl.kc_amount_approved > 0
    and kncl.kc_from_c_id = :c_id
    order by ks.ks_date_begin desc
    rows 1
    into :k_id
    do
    begin
        select kal.kal_new_reg_id
        from kontract kont
        inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
        where kal.is_delete = 0
        and :rep_date between kal.kal_date and kal.kal_date_end
        and kont.k_id = :k_id
        into :region_id;

        -- получаю список действующих КОБД на дату
        for select c_id, cl.c_name_fio
        from hr_pstatus, clients cl, joblist, jobkind
        where ps_from_c_id = c_id and
              :rep_date between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code in ('JobSystem.Specialist_KOBD') and
              ps_from_reg_id = :region_id
        into :c_id_tmp, :fio
        do
        begin
            -- сотрудник не должен быть в отпуске
            if (not exists (
                select *
                from hr_voctravel hv, hr_orders ho, sa_worktype swt
                where hrv_from_ho_id = ho_id and
                    ho_order_type = sawt_id and
                    hrv_from_c_id = :c_id_tmp and
                    :rep_date between hrv_date_begin and hrv_date_end_fact and
                    ho.ho_status = 1 and
                    swt.sawt_symbol in ('Б','Д','НБ','НО','О','ОР','ОУ','От'))) then -- не должен быть в отпуске
            begin
                res_text = fio|| ascii_char(13) || ascii_char(10);
                res_text = res_text||'Должность: кассир-оператор';
                suspend;
                exit;
            end
        end

        select cl.c_name_fio
        from hr_pstatus, clients cl, joblist, jobkind
        where ps_from_c_id = c_id and
              :rep_date between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code in ('JobSystem.Region_KM') and
              ps_from_reg_id = :region_id
        rows 1
        into :fio;

        res_text = fio|| ascii_char(13) || ascii_char(10);
        res_text = res_text||'Должность: менеджер офиса';
        suspend;
    end
end
