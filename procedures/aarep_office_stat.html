CREATE PROCEDURE AAREP_OFFICE_STAT(
    REP_DATE_END DATE)
RETURNS (
    REG_ID INTEGER,
    REG_NAME VARCHAR(100),
    BRANCH_NAME VARCHAR(100),
    REG_KC_HR INTEGER,
    REG_KC_PORT INTEGER,
    REG_DATE DATE,
    REG_IS_ACTIVE INTEGER,
    REG_DATE_CLOSE DATE)
AS
declare variable kc_id decimal(18,0);
declare variable cm_disable_staff_off integer;
begin
  select cm_disable_staff_off
  from company
  into :cm_disable_staff_off;

  for select r1.reg_unicode, r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r1.reg_date_open,
             r1.reg_date_close
      from region r1, region r2
      where r1.parent = r2.reg_unicode and
            r1.reg_status = 3
      order by r1.reg_code
      into :reg_id, reg_name, branch_name, reg_date, reg_date_close
  do
  begin
    reg_kc_hr = 0;
    select count(*)
    from hr_pstatus, joblist, jobkind
    where ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_is_working = 1 and
          (:cm_disable_staff_off = 0 or
          (:cm_disable_staff_off = 1 and
          ps_is_on_staff = 1)) and
          ps_from_reg_id = :reg_id
    into :reg_kc_hr;
    if (reg_kc_hr is null) then
      reg_kc_hr = 0;

    reg_kc_port = 0;
    if (reg_date is null) then
    begin
      select min(ol_date_oper_day)
      from operlist
      where ol_from_reg_id = :reg_id and
            operlist.is_delete = 0
      into :reg_date;
    end
    reg_is_active = 1;
    if (reg_date <= :rep_date_end) then
    begin
      if (reg_date_close is not null) then
      begin
        if (reg_date_close < :rep_date_end) then
          reg_is_active = 0;
      end
    end
    else
      reg_is_active = 0;
    reg_kc_port = 0;
    for select distinct oks_kc_id
        from oksummary
        where oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_region_id = :reg_id and
              oks_close_state = 0
        into :kc_id
    do
    begin
      reg_kc_port = reg_kc_port + 1;
    end

    suspend;
  end
end
