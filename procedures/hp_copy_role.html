CREATE PROCEDURE HP_COPY_ROLE(
    RN_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    RN_ID_NEW DECIMAL(18,0))
AS
declare variable rn_name varchar(30);
declare variable rn_memo varchar(2048);
declare variable uniqtab_rolename integer;
begin
-- Копирование роли
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;

  select rn_name, rn_memo
  from rolename
  where rn_id = :rn_id
  into :rn_name, :rn_memo;
  rn_name = rn_name || ' копия';

-- Вставтяем новую роль
  insert into rolename (rn_id, rn_name, rn_is_active, rn_registered_user, rn_memo)
  values (-1, :rn_name, 1, :un_id, :rn_memo)
  returning rn_id
  into :rn_id_new;

-- Очищаем доступы новой роли по таблицам
  delete from roletab
  where rt_from_rn_id = :rn_id_new;

-- Копируем доступы по таблицам
  insert into roletab (rt_from_rn_id, rt_from_ds_id, rt_can_update, rt_can_insert, rt_can_delete)
  select :rn_id_new, rt_from_ds_id, rt_can_update, rt_can_insert, rt_can_delete
  from roletab
  where rt_from_rn_id = :rn_id;

-- Очищаем доступы новой роли по меню (старая схема)
  delete from rolemenu
  where rm_from_rn_id = :rn_id_new;

-- Копируем доступы по меню (старая схема)
  insert into rolemenu (rm_from_rn_id, rm_menu_name)
  select :rn_id_new, rm_menu_name
  from rolemenu
  where rm_from_rn_id = :rn_id;

  insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_date_begin,
                           pm_date_end, pm_access_kind, pm_is_delete, pm_from_un_id)
  select -1, 1, :rn_id_new, pm_subject_kind, pm_subject_id, pm_date_begin, pm_date_end, pm_access_kind, pm_is_delete,
         :un_id
  from permissions
  where pm_object_kind = :uniqtab_rolename and
        pm_object_id = :rn_id and
        pm_is_delete = 0;

  suspend;

end
