CREATE PROCEDURE FM_CHANGE_DISB_DATE(
    UN_ID DECIMAL(18,0),
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    DISB_DATE DATE)
RETURNS (
    RES_ID INTEGER,
    RES_TXT VARCHAR(255))
AS
declare variable reg_id integer;
declare variable manager_id decimal(18,0);
declare variable ma_plan_disb_time timestamp;
declare variable k_status integer;
declare variable k_id decimal(18,0);
declare variable k_disb_date date;
begin
  k_status = 0;
  if (obj_kind = 'miniapp') then
  begin
    select k_id, k_status, k_disb_date
    from kontract
    where k_from_ma_app_id = :obj_id
    order by k_disb_date
    rows 1
    into :k_id, :k_status, k_disb_date;

    if (k_id is null) then
    begin
      select ma_date_oper
      from miniapp
      where ma_id = :obj_id and
            ma_date_end >= '31.12.2999'
      into :k_disb_date;
    end
  end
  else
  begin
    select k_id, k_status, k_disb_date
    from kontract
    where k_id = :obj_id
    into :k_id, :k_status, k_disb_date;
  end

  if (disb_date = k_disb_date) then
  begin
    res_id = -1;
    res_txt = 'Вы не поменяли дату выдачи.';
    suspend;
    exit;
  end

  if (k_status = 1) then
  begin
    res_id = -1;
    res_txt = 'Договор уже утвержден. Необходимо вернуть в процесс, либо поменять дату через КОБД.';
    suspend;
    exit;
  end

  if (disb_date < current_date) then
  begin
    res_id = -1;
    res_txt = 'Нельзя устанавливать дату выдачи в прошлом.';
    suspend;
    exit;
  end

  if (datediff(day, current_date, disb_date) > 60) then
  begin
    res_id = -1;
    res_txt = 'Слишком далекая дата выдачи.';
    suspend;
    exit;
  end

  if (k_id is not null) then
  begin
    if (exists(select *
               from kontstate
               where ks_from_k_id = :k_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0)) then
    begin
      res_id = -1;
      res_txt = 'По договору уже осуществлена выдача. Нельзя менять дату выдачи.';
      suspend;
      exit;
    end
  end
  res_id = 1;

  if (obj_kind = 'miniapp') then
  begin
    execute procedure hp_update_miniapp_cert_time(:obj_id, 0, null, :un_id, :disb_date);

    execute procedure hp_miniapp_kontract_sync(11, :obj_id);

    select res_id, res_txt
    from hp_make_schedule(:un_id, 11, :obj_id, 0, null, 0)
    into res_id, res_txt;

  end
  else
  begin
    update kontract
    set k_disb_date = :disb_date,
        k_kontract_date = f_get_kontract_date(:obj_id, :disb_date),
        k_user_id = :un_id
    where k_id = :obj_id;

    execute procedure hp_miniapp_kontract_sync(9, :obj_id);

    select res_id, res_txt
    from hp_make_schedule(:un_id, 9, :obj_id, 0, null, 0)
    into res_id, res_txt;
  end
  suspend;
end
