CREATE PROCEDURE AAREP_SSVZ_DAY(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KF_OFFICER VARCHAR(100),
    DAY_1_KOL_DISB_KRED INTEGER,
    DAY_2_KOL_DISB_KRED INTEGER,
    DAY_3_KOL_DISB_KRED INTEGER,
    DAY_4_KOL_DISB_KRED INTEGER,
    KOL_DISB_PARALEL INTEGER,
    DISB_KOL_NEW INTEGER,
    DISB_KOL_OLD INTEGER,
    DISB_SUM CURR_TYPE,
    PAR1 CURR_TYPE,
    PAR7 CURR_TYPE,
    DELINQ_1_KAT_1_GR CURR_TYPE,
    DELINQ_7_KAT_1_GR CURR_TYPE,
    DELINQ_7_KAT_2_GR CURR_TYPE,
    REP_BALANCE_KAT_1_2 CURR_TYPE,
    KOL_CLI_KAT_1_2 INTEGER,
    DISB_KR_BEGIN_MONTH INTEGER)
AS
declare variable i smallint;
declare variable rep_date date;
declare variable rep_date_begin date;
declare variable svz_id id_type;
declare variable kol_disb_kred integer;
declare variable rep_balance curr_type;
declare variable delinq_balance_7 curr_type;
declare variable delinq_balance_1 curr_type;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  for
-- Вытаскивает всех СВЗ кроме декретников
      select ps_from_c_id
      from jobkind, joblist, hr_pstatus, region_list(:rep_region_id, 1, 3, 1)
      where jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            jk_id = jl_kind and
            jl_id = ps_from_jl_id and
            ps_status = 1 and
            ps_is_working = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = id_list
-- Исключаем декретников
            and
            not coalesce((select sawt_id
                          from hr_voctravel, hr_orders, sa_worktype
                          where hrv_from_ho_id = ho_id and
                                ho_order_type = sawt_id and
                                hrv_from_c_id = ps_from_c_id and
                                :rep_date_end between hrv_date_begin and hrv_date_end_fact
                          order by hrv_date_begin desc
                          rows 1), 0) in (16, 17, 18)
      into svz_id
  do
  begin
    rep_date = :rep_date_end;
    i = 0;
    while (i < 4) do
    begin
      rep_date = rep_date - i;
      rep_date_begin = rep_date; -- начало недели
      i = i + 1;
--    Кол-во выдач за неделю
      kol_disb_kred = 0;
      select sum(k_kolich_client_approved)
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            cast(ks_date_begin as date) = :rep_date and
            ks_kc_id = :svz_id and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_product_kind <> 2
      into kol_disb_kred;
      if (kol_disb_kred is null) then
        kol_disb_kred = 0;

      if (i = 1) then
      begin
        day_1_kol_disb_kred = kol_disb_kred;
      end
      if (i = 2) then
      begin
        day_2_kol_disb_kred = kol_disb_kred;
      end
      if (i = 3) then
      begin
        day_3_kol_disb_kred = kol_disb_kred;
      end
      if (i = 4) then
      begin
        day_4_kol_disb_kred = kol_disb_kred;
      end
    end
--    Кол-во выдач за текущий день, параллельные
    kol_disb_paralel = 0;
    select sum(k_kolich_client_approved)
    from kontstate, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_end - 6 and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind = 2
    into kol_disb_paralel;
    if (kol_disb_paralel is null) then
      kol_disb_paralel = 0;
--    Кол-во новых и повторных выдач за текущий день
    disb_kol_new = null;
    disb_kol_old = null;
    disb_sum = null;
    select count(iif(kc_circle <= 1, kc_from_c_id, null)), count(iif(kc_circle > 1, kc_from_c_id, null)),
           sum(kc_amount_approved)
    from kontstate, kontract, product, knclient
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) = :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2 and
          k_id = kc_from_k_id and
          kc_amount_approved > 0
    into disb_kol_new, disb_kol_old, disb_sum;
    if (disb_kol_new is null) then
      disb_kol_new = 0;
    if (disb_kol_old is null) then
      disb_kol_old = 0;
    if (disb_sum is null) then
      disb_sum = 0;
-- Вычисление ПАР
    select sum(oks_balance), sum(iif(:rep_date_end >= oks_delinq_date, oks_balance, 0)),
           sum(iif(:rep_date_end - 7 >= oks_delinq_date, oks_balance, 0))
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :svz_id and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1
    into rep_balance, delinq_balance_1, delinq_balance_7;
    if (rep_balance is null) then
      rep_balance = 0;
    if (delinq_balance_1 is null) then
      delinq_balance_1 = 0;
    if (delinq_balance_7 is null) then
      delinq_balance_7 = 0;

    if (rep_balance > 0) then
    begin
      par1 = delinq_balance_1 / rep_balance;
      par7 = delinq_balance_7 / rep_balance;
    end
    else
    begin
      par1 = 0;
      par7 = 0;
    end

--    Кол-во просроченных групп
    delinq_1_kat_1_gr = null;
    delinq_7_kat_1_gr = null;
    delinq_7_kat_2_gr = null;
    select sum(iif((select aa_category_id
                    from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1, 1, 0)) as delinq_1_kat_1_gr, sum(iif((select aa_category_id
                                                                                                                                                             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 1 and :rep_date_end - 7 >= oks_delinq_date, 1, 0)) as delinq_7_kat_1_gr, sum(iif((select aa_category_id
                                                                                                                                                                                                                                                                                                                                               from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 2 and :rep_date_end - 7 >= oks_delinq_date, 1, 0)) as delinq_7_kat_2_gr
    from oksummary, kontract, product
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :svz_id and
          oks_close_state = 0 and
          :rep_date_end >= oks_delinq_date and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
          oks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          pr_product_kind <> 2
    into delinq_1_kat_1_gr, delinq_7_kat_1_gr, delinq_7_kat_2_gr;
    if (delinq_1_kat_1_gr is null) then
      delinq_1_kat_1_gr = 0;
    if (delinq_7_kat_1_gr is null) then
      delinq_7_kat_1_gr = 0;
    if (delinq_7_kat_2_gr is null) then
      delinq_7_kat_2_gr = 0;

--    Кредитный портфель категории 1 и 2
    rep_balance_kat_1_2 = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_kc_id = :svz_id and
          oks_close_state = 0 and
          (select aa_category_id
           from aarep_get_category_id(oks_from_k_id, :rep_date_begin - 1, null, null, null, null, null, null)) in (1, 2)
    into rep_balance_kat_1_2;
    if (rep_balance_kat_1_2 is null) then
      rep_balance_kat_1_2 = 0;
--    Активные клиенты категории 1 и 2
    kol_cli_kat_1_2 = null;
    select sum(ac_is_client)
    from aarep_cli_active_list(:rep_date_end)
    where ac_kc_id = :svz_id and
          ac_is_client = 1 and
          (select aa_category_id
           from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
    into kol_cli_kat_1_2;
    if (kol_cli_kat_1_2 is null) then
      kol_cli_kat_1_2 = 0;
--    Выдано кредитов с начало месяца
    disb_kr_begin_month = null;
    select sum(k_kolich_client_approved)
    from kontstate, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :svz_id and
          ks_from_k_id = k_id
    into disb_kr_begin_month;
    if (disb_kr_begin_month is null) then
      disb_kr_begin_month = 0;

--    ФИО СВЗ
    kf_officer = (select c_name_fio
                  from clients
                  where c_id = :svz_id);
    suspend;
  end
end
