CREATE PROCEDURE AAREP_FIND_KANSOPER(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_TEXT VARCHAR(255))
RETURNS (
    KOP_ID INTEGER,
    KOP_FROM_REG_ID INTEGER,
    KOP_DATE DATE,
    KOP_FROM_UN_ID DECIMAL(18,0),
    KOP_DATETIME TIMESTAMP,
    KOP_MEMO VARCHAR(255),
    KOP_TO_OFFICE INTEGER,
    KOP_FOR_REGION INTEGER,
    KOP_CORR_OFFICE VARCHAR(100),
    KOP_OFFICE VARCHAR(100),
    KOP_USERNAME VARCHAR(100),
    KOP_IS_DEL_STR VARCHAR(20))
AS
declare variable kop_is_delete integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  rep_text = upper(:rep_text);
  for select distinct kop_id, kop_from_reg_id, kop_date, kop_from_un_id, kop_datetime, kop_memo, kop_to_office,
                      kop_for_region, reg_code || ' ' || reg_name, c_name_fio, kop_is_delete
      from kansoper, region_list(:rep_region_id, 1, -1, 1), kansoperlist, kanslist, region, username, clients
      where (kop_from_reg_id = id_list or
            kop_to_office = id_list or
            kop_for_region = id_list) and
            kop_date >= :rep_date_begin and
            kop_date <= :rep_date_end and
            kopl_from_kop_id = kop_id and
            kopl_from_ksl_id = ksl_id and
            kop_from_reg_id = reg_unicode and
            kop_from_un_id = un_id and
            un_from_c_id = c_id and
            ((upper(kop_memo) like '%' || :rep_text || '%') or
            (upper(ksl_name) like '%' || :rep_text || '%') or
            (upper(ksl_memo) like '%' || :rep_text || '%'))
      into :kop_id, :kop_from_reg_id, :kop_date, :kop_from_un_id, :kop_datetime, :kop_memo, :kop_to_office,
           :kop_for_region, :kop_office, :kop_username, :kop_is_delete
  do
  begin
    kop_corr_office = :kop_office;
    if (kop_to_office is not null) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :kop_to_office
      into :kop_corr_office;
    end
    if (kop_for_region is not null) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :kop_for_region
      into :kop_corr_office;
    end
    if (kop_is_delete = 1) then
      kop_is_del_str = 'Аннулирован';
    else
      kop_is_del_str = '';
    suspend;
  end
end
