CREATE PROCEDURE HP_FT_INFOCHECK_DATE_INFO(
    FIL_ID D_ID,
    FIC_DATE D_DATE)
RETURNS (
    RES_CAPTION D_STR_LONG,
    RES_VALUE D_STR_LONG)
AS
declare variable fib_id d_id;
declare variable c_id d_id;
declare variable fil_date_open d_date;
declare variable tmp_value d_str_8100;
declare variable terror_value d_str_long;
declare variable fif_fieldname d_str_255;
declare variable company_name d_str_100;
begin
  select fil_from_fib_id, fil_from_c_id, fil_date_open
  from ft_infolist
  where fil_id = :fil_id
  into fib_id, c_id, fil_date_open;

  if (fib_id is null) then Exit;

  for select fic_date
  from ft_infocheck
  where fic_from_fil_id = :fil_id and
        fic_date between :fil_date_open + 1 and fic_date
  order by fic_date
  into fic_date
  do
  begin
    res_caption = 'Даты обновления анкеты (досье) клиента';
    res_value = f_datetostr(fic_date);
    suspend;

    res_caption = 'Результаты обновлений анкеты (досье) клиента';
    res_value = '';

    -- нужно доработать чтобы в бланке отображался как надо
    for select fif_fieldname,
               fid.fif_caption ||': '|| fiv.fiv_value   --list(fiv_value)
    from ft_infovalue fiv, ft_infofield fid
    where fiv_from_c_id = :c_id and
          fiv_date_begin = :fic_date and
          fiv_from_fif_id = fif_id and
          fif_from_fib_id = :fib_id and
          fif_non_client_data = 0 and
          not(fif_fieldname = 'exist_in_fp_list' and fiv_value starting 'Отсутствует в перечне на дату') and
          not(fif_fieldname = 'solution_in_fp' and fiv_value starting 'Решение отсутствует на дату') and
          not(fif_fieldname starting 'relation_purpose')and
          not(fif_fieldname starting 'finance_state') and
          not(fif_fieldname starting 'reputation')
    into :fif_fieldname,
         :tmp_value
    do
    begin
        if (fif_fieldname containing 'other_info') then
            tmp_value = replace(tmp_value, 'Представители отсутствуют, клиент не действует к выгоде третьего лица.', '');

        res_value = res_value||' '||tmp_value|| ascii_char(13) || ascii_char(10);
    end
    if (res_value is null) then res_value = '';

    terror_value = 'Клиент отсутствует в перечне Росфинмониторинга террористов/экстремистов и в перечне лиц, в отношении которых приняты решения о применении мер по замораживанию/блокированию денежных средств и иного имущества '||trim(iif(fib_id >= 7, 'и перечне ОНН', ''))||' по состоянию на '||f_datetostr(:fic_date);

    select replace(fisl_for_update_txt, '@date', f_datetostr(:fic_date))
    from ft_info_stoplist fis
    where fis.fisl_from_c_id = :c_id
    and :fic_date between fis.fisl_date_begin and fis.fisl_date_end
    into :terror_value;

    if (res_value = '')
    then res_value = 'Остальная информация не изменилась';
    else res_value = 'Информация изменилась: '|| ascii_char(13) || ascii_char(10) || trim(res_value);

    res_value = upper(terror_value || ascii_char(13) || ascii_char(10) || res_value);
    suspend;

    company_name = f_company_name('short', 'local', fic_date);
    if (fic_date < '21.01.2022') then
    begin
      res_caption = f_format(
        'Фамилия, имя, отчество (при наличии последнего), должность '||
        'сотрудника %1, заполнявшего (обновившего) анкету (досье) клиента',
        company_name);
      res_value = '';
      select res_text
      from ft_resp_fio_get(:c_id, :fic_date)
      into res_value;
    end
    else
    begin
      res_caption = f_format(
        'Фамилия, имя, отчество (при наличии последнего), должность ' ||
        'сотрудника %1, заполнявшего (обновившего) анкету (досье) клиента ' ||
        '(за исключением случая, когда заполнение (обновление) анкеты (досье) ' ||
        'клиента осуществлено с использованием автоматизированных систем без ' ||
        'фактического участия сотрудника %1)',
        company_name);
      res_value =
        'Заполнение (обновление) Анкеты (досье) клиента осуществлено с ' ||
        'использованием автоматизированных систем';
    end
    suspend;
  end
end
