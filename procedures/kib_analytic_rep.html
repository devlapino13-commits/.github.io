CREATE PROCEDURE KIB_ANALYTIC_REP(
    REGION_ID DECIMAL(18,0),
    MA_ID DECIMAL(18,0))
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(255),
    ACTIVE_AMOUNT DECIMAL(18,2),
    COUNTOFDELINQUENCY_10 INTEGER,
    COUNTOFDELINQUENCY_60 INTEGER,
    KOLICH_KREDIT INTEGER,
    KOLICH_ACTIVE INTEGER,
    MONTHLY_REPAY_ALL DECIMAL(18,4),
    MAC_AMOUNT DECIMAL(18,4),
    REP_DATE DATE,
    REP_OFFICE VARCHAR(100),
    REP_KFOLDER VARCHAR(100),
    REP_LO_FIO VARCHAR(100),
    REP_MA_NO INTEGER,
    KBM_ID ID_TYPE)
AS
declare variable KH_COUNTOFDELINQUENCY integer;
declare variable KH_CREDITSTATUS varchar(255);
declare variable KH_REPAYSTATUS varchar(255);
declare variable KH_AMOUNT decimal(18,2);
declare variable C_FIO_PRIOR varchar(100);
declare variable KH_MONTHLY_REPAY decimal(18,4);
declare variable C_ID_ORIG decimal(18,0);
declare variable C_FIO_ORIG varchar(100);
declare variable C_ID_PRIOR decimal(18,0);
declare variable MAC_AMOUNT_ORIG decimal(18,4);
declare variable MAC_AMOUNT_PRIOR decimal(18,4);
declare variable KH_DELIQDAYSCOUNT integer;
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable IS_REAL_HIST smallint;
declare variable KH_IS_ACTIVE integer;
declare variable KH_CURRENCY STR_100;
declare variable KH_CURRENCY_OLD STR_100;
declare variable KBM_CHECKED_DATE date;
declare variable CU_ID varchar(3);
declare variable CR_RATE CURR_TYPE;
declare variable DB_ID D_INTEGER;
declare variable BLS_ID D_ID;
begin
  db_id = f_get_db_id();
  if (db_id <> 39801) then
  begin
    select st_user_client_access, st_user_staff_access
    from get_stored_proc_options
    into :st_client, :st_staff;
    
    /*
    CHANGED BY IGUSEV 26.06.2012
    changes:  Были добавлены статусы погашения кредита, которых ранее не было.
              Добавленные статусы:
              '30 дней до 59 дней просрочки',
              '60 дней до 89 дней просрочки',
              '90 дней  до 119 дней просрочки'
    
    CHANGED BY IGUSEV 27.06.2012
    changes: если у нас имеются данные о количестве дней просрочек, тогда отталкиваюсь от этого числа,
           в противном случае отталкиваюсь от полученного статуса ((KH_DELIQDAYSCOUNT >=10), (KH_DELIQDAYSCOUNT >=60))
    */
    
    c_fio_prior = '';
    c_id_prior = 0;
    mac_amount_prior = 0;
    rep_date = cast('NOW' as date);
    select distinct kal_new_kf_name, c_name_fio, reg_code || ' ' || reg_name, ma_no
    from knaktlist, clients, region, miniapp
    where kal_new_reg_id = reg_unicode and
        knaktlist.is_delete = 0 and
        kal_date <= ma_date_app and
        kal_date_end >= ma_date_oper and
        kal_new_kf_officer = c_id and
        ma_from_kf_id = kal_from_kf_id and
        ma_id = :ma_id and
        ma_date_end = '31.12.9999'
    into :rep_kfolder, :rep_lo_fio, :rep_office, :rep_ma_no;
    
    is_real_hist = 0;
    kh_currency_old = '';
    
    for select kf_client_id, kf_client_name, kh_unpaidamount, /*KH_APROX_BALANCE,*/
             kh_countofdelinquency, kh_creditstatus, kh_repaystatus, kh_monthly_repay, kf_amount, kh_deliqdayscount,
             is_real_hist, kh_is_active, kh_currency, kbm_checked_date, kbm_id
      from aarep_kib_miniapp_report(:region_id, :ma_id, '')
      into :c_id, :c_name_fio, :kh_amount, :kh_countofdelinquency, :kh_creditstatus, :kh_repaystatus, :kh_monthly_repay,
           :mac_amount, :kh_deliqdayscount, :is_real_hist, :kh_is_active, :kh_currency, :kbm_checked_date, :kbm_id
    do
    begin
    if (c_fio_prior <> c_name_fio) then
    begin
      c_fio_orig = :c_name_fio;
      c_id_orig = :c_id;
    
      mac_amount_orig = :mac_amount;
    
      c_name_fio = c_fio_prior;
      c_id = c_id_prior;
    
      mac_amount = :mac_amount_prior;
    
      if (c_fio_prior <> '') then
        suspend;
      c_name_fio = c_fio_orig;
    
      c_id = :c_id_orig;
      c_id_prior = c_id;
    
      c_fio_prior = c_name_fio;
    
      mac_amount = :mac_amount_orig;
      mac_amount_prior = :mac_amount;
    
      countofdelinquency_10 = 0;
      countofdelinquency_60 = 0;
      active_amount = 0;
      kolich_kredit = 0;
      kolich_active = 0;
      monthly_repay_all = 0;
    end
    
    if (kh_currency <> kh_currency_old) then
    begin
      select cl.cu_id
      from currlist cl
      where cl.cu_symbol = (case :kh_currency
                              when 'Сом (Кыргызский)' then 'KGS'
                              when 'Доллар США' then 'USD'
                              when 'Евро' then 'EUR'
                              when 'Российский рубл' then 'RUB'
                              else 'KGS'
                            end)
      into :cu_id;
    
      select opc.cr_rate
      from opercurr opc
      where opc.cr_from_cu_id = :cu_id and
            :kbm_checked_date between opc.cr_date and opc.cr_date_end
      into :cr_rate;
    
      kh_currency_old = kh_currency;
    
      cr_rate = coalesce(cr_rate, 1);
    end
    
    active_amount = active_amount + kh_amount * cr_rate;
    monthly_repay_all = monthly_repay_all + kh_monthly_repay * cr_rate;
    if (is_real_hist = 1) then
    begin
      kolich_kredit = kolich_kredit + 1;
    end
    
    if (kh_deliqdayscount is null) then
      kh_deliqdayscount = 0;
    
    if (kh_is_active = 1) then
    --if ((Trim(Upper(kh_creditstatus)) = 'ВЫДАН') or (Trim(Upper(kh_creditstatus)) = 'СВОЕВРЕМЕННЫЙ ПЛАТЕЖ') or (Trim(Upper(kh_creditstatus)) like '%ПРОСРОЧЕН%'))  then
    begin
      if (kh_amount is null) then
        kh_amount = 0;
      kolich_active = kolich_active + 1;
    
      /* IGUSEV 27.06.2012
           если у нас имеются данные о количестве дней просрочек, тогда отталкиваюсь от этого числа,
           в противном случае отталкиваюсь от полученного статуса
        */
    
      if (kh_deliqdayscount >= 10) then
        countofdelinquency_10 = countofdelinquency_10 + 1;
      else
      begin
        if (trim(kh_repaystatus) in ('от 10 до 20 дней просрочки', 'от 20 до 30 дней просрочки', '30 дней просрочки и более', '60 дней просрочки и более', '90 дней просрочки и более', '120 дней просрочки и более', '30 дней до 59 дней просрочки', '60 дней до 89 дней просрочки', '90 дней  до 119 дней просрочки')) then
          countofdelinquency_10 = countofdelinquency_10 + 1;
      end
    end
    
    if (kh_deliqdayscount >= 60) then
      countofdelinquency_60 = countofdelinquency_60 + 1;
    else
    begin
      if (trim(kh_repaystatus) in ('60 дней просрочки и более', '90 дней просрочки и более', '120 дней просрочки и более', '60 дней до 89 дней просрочки', '90 дней  до 119 дней просрочки')) then
        countofdelinquency_60 = countofdelinquency_60 + 1;
    end
    c_fio_prior = c_name_fio;
    end
    suspend;
  end
  else
  begin

    select cl.c_name_fio, mac.mac_amount, mac.mac_from_c_id
    from miniappclient mac
    inner join clients cl on cl.c_id = mac.mac_from_c_id
    where mac.mac_from_ma_id = :ma_id
        and mac.mac_amount > 0
    into :c_name_fio, mac_amount, c_id;

    select eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eic.eic_code = 'pkb_cib_report'
        and eil.eil_object_kind = f_objkind('miniapp', 'id')
        and eil.eil_from_c_id = :c_id
        and eil.eil_status = 2
    order by eil.eil_status_datetime desc
    rows 1
    into :bls_id;

    select all_credit_history_count, active_kont, principal_amount, total_monthly_ins_amount, cnt_overdue_10d, cnt_overdue_60d
    from hp_parse_pkb_report(:bls_id)
    rows 1
    into :kolich_kredit, kolich_active, active_amount, monthly_repay_all, countofdelinquency_10, countofdelinquency_60;

    suspend;
  end 
end
