CREATE PROCEDURE HP_CARDEMBOMAIN_REGERRORS_VIEW(
    CEM_ID D_INTEGER)
RETURNS (
    CARD_ID D_INTEGER,
    STATUS D_STR_CODE,
    DESCR D_STR_255,
    "ACTION" D_STR_255)
AS
declare variable cal_status_inproc d_integer;
begin
  if (exists(
    select * from cardembomain
    where cem_id = :cem_id
      and cem_from_ces_id = 1
      and cem_app_count <> cem_ready_count)) then
  begin
    cal_status_inproc = f_enum('cardlist_status', 'in_process', 'id');
    for
      select
        cal_card_id,
        f_enum_name('cardlist_status', cal_status),
        left(replace(f_json_str_get_value(cal_answer_json, 'err_desc', ''), ascii_char(10), ' '), 255),
        left(replace(f_json_str_get_value(cal_answer_json, 'err_action', ''), ascii_char(10), ' '), 255)
      from cardlist
      where cal_from_cem_id = :cem_id
        and cal_status < :cal_status_inproc
    into
      :card_id,
      :status,
      :descr,
      :action
    do
    begin
      suspend;
    end
  end
end
