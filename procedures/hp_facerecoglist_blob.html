CREATE PROCEDURE HP_FACERECOGLIST_BLOB
RETURNS (
    ID D_INTEGER,
    PHOTO_BLOB D_BLOB_BINARY,
    DOC_BLOB D_BLOB_BINARY)
AS
declare variable FRL_PHOTO_BLS_ID D_ID;
declare variable FRL_DOC_BLS_ID D_ID;
begin
  -- Процедура передает список картиной для сравнения фотографий на них

  for select frl_id, frl_photo_bls_id, frl_doc_bls_id
  from facerecoglist
  where frl_state = 0
  rows 1
  into id, frl_photo_bls_id, frl_doc_bls_id
  do
  begin
    select blx_blob
    from hp_get_blobex_blob(:frl_photo_bls_id)
    into :photo_blob;

    select blx_blob
    from hp_get_blobex_blob(:frl_doc_bls_id)
    into :doc_blob;

    suspend;
  end
end
