CREATE PROCEDURE HP_JOIN_DOUBLE_CLIENT(
    C_STAY DECIMAL(18,0),
    C_REMOVE DECIMAL(18,0))
AS
declare variable UN_ID decimal(18,0);
declare variable DC_IDENTITY_KOEF integer;
declare variable DC_IS_CRITICAL integer;
declare variable UN_NAME varchar(100);
declare variable DB_ID D_INTEGER;
declare variable ERR_TEXT D_STR_255;
begin
  db_id = f_get_db_id();

  if (not(exists(select *
                 from joinedclients
                 where jc_stay_c_id = :c_stay and
                       jc_remove_c_id = :c_remove))) then
  begin
    dc_identity_koef = -1;
    dc_is_critical = -1;

    select dc_identity_koef, dc_is_critical
    from doubleclients
    where (dc_from_c_id_1 = :c_stay and
          dc_from_c_id_2 = :c_remove) or
          (dc_from_c_id_2 = :c_stay and
          dc_from_c_id_1 = :c_remove) rows 1
    into :dc_identity_koef, :dc_is_critical;

    un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN');

    select un_id
    from username
    where un_name = :un_name
    into :un_id;

    insert into joinedclients (jc_stay_c_id, jc_remove_c_id, jc_from_un_id, jc_stay_fio, jc_stay_birth_date,
                               jc_remove_fio, jc_remove_birth_date, jc_identity_koef, jc_is_critical)
    select :c_stay, :c_remove, :un_id, cstay.c_name_fio, cstay.c_birth_date, cremove.c_name_fio, cremove.c_birth_date,
           :dc_identity_koef, :dc_is_critical
    from clients cstay, clients cremove
    where cstay.c_id = :c_stay and
          cremove.c_id = :c_remove;
  end

  execute procedure hp_join_double_client_add_1(c_stay, c_remove);
  execute procedure hp_join_double_client_add_2(c_stay, c_remove);
  execute procedure hp_join_double_client_add_3(c_stay, c_remove);
  if (db_id in (39801, 41701) and
      exists(select *
             from cardclientid
             where cci_from_c_id = :c_remove)) then
  begin
    err_text = (select err_text
                from hp_cardclient_update_customer(:c_remove, :c_stay, 0, :un_id));
    execute procedure hp_join_double_client_add_4(c_stay, c_remove);
  end

  if (exists(select *
             from edoc_face edf
             where edf.edf_from_c_id = :c_remove)) then
    update edoc_face edf
    set edf.edf_from_c_id = :c_stay
    where edf.edf_from_c_id = :c_remove;

  if (exists(select *
             from epass_response erp
             where erp.erp_from_c_id = :c_remove)) then
    update epass_response erp
    set erp.erp_from_c_id = :c_stay
    where erp.erp_from_c_id = :c_remove;
  
  delete from clientstate_ru
  where csr_from_c_id = :c_remove;

  delete from clients
  where c_id = :c_remove;

  execute procedure hp_clientstate_set(:c_stay);
  execute procedure do_client_circle(:c_stay);
  execute procedure hp_pointcalc_create(:c_stay);
end
