CREATE PROCEDURE HP_EDOC_WEB_EPAY_CONDITION(
    PAY_TYPE D_STR_100)
RETURNS (
    CONDITION_TXT D_STR_8100)
AS
begin
  condition_txt = '';
  condition_txt = case
                    when pay_type = 'bankcard' then 'Оплата займа через Visa и Mastercard'
                    when pay_type = 'wallet' then 'Оплата займа через Элсом; Оплата займа через O! Деньги; Оплата займа через MegaPay'
                    when pay_type = 'internetbank' then 'Оплата займа через Мбанк'
                    when pay_type = 'mobile_commerce' then 'Оплата займа через Balance.kg'
                    when pay_type = 'elcard' then 'При оплате с помощью карт Элкарт, у вас должны быть активированы интернет-платежи. Проверить подключена ли интернет-оплата по карте, можно позвонив на номер колл-центра Межбанковского Процессингового центра. Круглосуточная служба поддержки: +996 312 637 696. Комиссия при оплате составит 0%'
                    when pay_type in ('mir', 'sbp') then 'Сумма к оплате зависит от официального курса российского рубля, установленного Национальным Банком КР по отношению к кыргызскому сому на момент проведения оплаты'
                    when pay_type = 'qr_code' then 'Комиссия М Булак для оплаты по ElQR = 0 (%/сом). Информируем, что при погашении кредитов/финансирований посредством мобильных приложений Банков Кыргызской Республики, могут взиматься комиссии данного(ых) банка(ов)'
                    when pay_type = 'elqr' then 'Комиссия при оплате составит 0%. Информируем, что при погашении кредитов/финансирований посредством мобильных приложений Банков Кыргызской Республики, могут взиматься комиссии данного(ых) банка(ов)'
                  end;
  condition_txt = trim(condition_txt);
  suspend;
end
