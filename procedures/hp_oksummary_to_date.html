CREATE PROCEDURE HP_OKSUMMARY_TO_DATE(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    OKS_DELINQ_DATE D_DATE,
    OKS_BALANCE D_CURRENCY,
    OKS_SH_PERC D_CURRENCY,
    OKS_ADD_PERC D_CURRENCY,
    OKS_CURR_SH_OS D_CURRENCY,
    OKS_CURR_SH_PERC D_CURRENCY,
    OKS_DELINQ_OS D_CURRENCY,
    OKS_DELINQ_PERC D_CURRENCY,
    OKS_DELINQ_DAY D_INTEGER,
    OKS_DELINQ_DAY_PRV D_INTEGER,
    OKS_IS_DELINQ D_SMALLINT,
    OKS_OUT_OS D_CURRENCY,
    OKS_OUT_PERC D_CURRENCY,
    OKS_OUT_OTHER D_CURRENCY,
    OKS_EXTRA D_CURRENCY,
    OKS_IS_RESH D_SMALLINT,
    OKS_RESH_COUNT D_INTEGER,
    OKS_RESH_FIRST D_DATE,
    OKS_RESH_LAST D_DATE,
    OKS_RESH_PROLONG_DAY D_INTEGER,
    OKS_RESH_MAX_PROLONG D_INTEGER,
    OKS_RESH_DELINQ_DAY D_INTEGER,
    OKS_RESH_DELINQ_OS D_CURRENCY,
    OKS_RESH_DELINQ_PERC D_CURRENCY,
    OKS_LAST_PMT_DATE D_DATE,
    OKS_LAST_SH_DATE D_DATE,
    OKS_CLOSE_DATE D_DATE,
    OKS_CLOSE_STATE D_INTEGER,
    OKS_PERCENT_PER_DAY D_PERCENT,
    OKS_PERCENT_YEAR D_PERCENT,
    OKS_FLOAT_PERC_LEVEL D_SMALLINT,
    OKS_REGION_ID D_INTEGER,
    OKS_KC_ID D_ID,
    OKS_CLOSE_OS_DATE D_DATE,
    OKS_ACTUAL_LAST_DATE D_DATE)
AS
declare variable sh_date d_date;
declare variable ok_os d_currency;
declare variable sh_os d_currency;
declare variable sh_perc d_currency;
declare variable has_writeoff d_smallint;
declare variable is_off d_smallint;
declare variable pmt_date d_date;
declare variable min_date d_date;
declare variable is_resh d_smallint;
declare variable has_resh d_smallint;
declare variable last_date d_date;
declare variable kf_id d_id;
declare variable k_last_date d_date;
declare variable curr_sh_date d_date;
declare variable curr_max_sh_date d_date;
declare variable prv_sh_date d_date;
declare variable prv_max_sh_date d_date;
declare variable max_ol_date d_date;
declare variable ol_date d_date;
declare variable ol_id d_id;
declare variable zad_all_prv d_currency;
declare variable oks_delinq_date_prv d_date;
declare variable disb_date d_date;
declare variable curr_line d_smallint;
declare variable ol_os_date d_date;
declare variable ks_date_begin d_date;
begin
  min_date = '01.01.1900';
  oks_delinq_date = '31.12.9999';
  oks_delinq_date_prv = '31.12.9999';
  oks_close_os_date = oks_delinq_date;
  oks_balance = 0;
  oks_sh_perc = 0;
  oks_add_perc = 0;
  oks_curr_sh_os = 0;
  oks_curr_sh_perc = 0;
  oks_delinq_os = 0;
  oks_delinq_perc = 0;
  oks_close_state = 0;
  oks_out_os = 0;
  oks_out_perc = 0;
  oks_out_other = 0;
  oks_extra = 0;
  has_writeoff = 0;
  oks_last_pmt_date = min_date;
  max_ol_date = min_date;
  oks_delinq_day = 0;
  oks_delinq_day_prv = 0;
  oks_is_delinq = 0;
  oks_close_date = null;
  curr_line = 0;

  has_resh = 0;
  oks_is_resh = 0;
  oks_resh_count = 0;
  oks_resh_first = null;
  oks_resh_last = null;
  oks_resh_max_prolong = 0;
  oks_resh_prolong_day = 0;

  oks_resh_delinq_day = 0;
  oks_resh_delinq_os = 0;
  oks_resh_delinq_perc = 0;

  oks_float_perc_level = 0;
  select k_from_kf_id, k_disb_date, k_percent_per_day, round(k_percent_per_day * 365.00000000, 7), k_fact_disb_reg_id,
         k_fact_disb_lo_id, k_kont_last_date
  from kontract
  where k_id = :k_id
  into kf_id, ol_date, oks_percent_per_day, oks_percent_year, oks_region_id, oks_kc_id, k_last_date;

  oks_actual_last_date = k_last_date;

  max_ol_date = ol_date;
  disb_date = ol_date;

  curr_max_sh_date = k_last_date;
  prv_max_sh_date = k_last_date;
  oks_last_sh_date = min_date;

  select kd_percent_per_day, round(kd_percent_per_day * 365.00000000, 7), kd_float_perc_level
  from kontdata
  where kd_from_k_id = :k_id and
        kd_ol_date <= :rep_date and
        kd_date_end >= :rep_date and
        kd_ol_is_delete = 0
  order by kd_from_k_id desc, kd_ol_date desc
  rows 1
  into :oks_percent_per_day, oks_percent_year, oks_float_perc_level;

  select kal_new_reg_id, kal_new_kf_officer
  from knaktlist a
  where a.kal_from_kf_id = :kf_id and
        a.kal_date <= :rep_date and
        a.kal_date_end >= :rep_date and
        a.is_delete = 0
  into oks_region_id, oks_kc_id;

  for select ok_sh_date,
             sum(iif(ok_type <= 8, ok_os, 0)),
             sum(iif(ok_type < 7, ok_os + ok_sh_os, 0)),
             sum(iif(ok_type < 7, ok_perc + ok_sh_perc, 0)),
             sum(iif(ok_type < 7 and ok_ol_date < :rep_date, ok_os + ok_sh_os + ok_perc + ok_sh_perc, 0)),
             max(ok_ol_date),
             max(iif(ok_type < 7 and ok_os <> 0, ok_ol_date, :min_date)),
             max(iif(ok_type in (5, 6), 1, 0)),
             max(iif(ok_type in (4, 8, 9), ok_ol_date, :min_date)),
             max(iif(ok_type = 2, 1, 0)),
             max(iif(ok_type in (1, 2, 3), ok_sh_date, :min_date)),
             max(iif(ok_type = 2 and ok_ol_date = :rep_date and ok_sh_os > 0, ok_sh_date, :min_date)),
             max(iif(ok_type = 2 and ok_ol_date < :rep_date and ok_sh_os > 0, ok_sh_date, :min_date)),
             sum(ok_kommis)
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_from_c_id = 0 and
            ok_ol_date <= :rep_date
      group by ok_sh_date
      into sh_date, ok_os, sh_os, sh_perc, zad_all_prv, ol_date, ol_os_date, is_off, pmt_date, is_resh, last_date, curr_sh_date,
           prv_sh_date, oks_add_perc
  do
  begin
    if (ok_os is null) then
      ok_os = 0;

    if (sh_os is null) then
      sh_os = 0;

    if (sh_perc is null) then
      sh_perc = 0;

    if (is_off is null) then
      is_off = 0;

    if (zad_all_prv is null) then
      zad_all_prv = 0;

    if (is_off = 1 and
        has_writeoff = 0) then
      has_writeoff = 1;

    if (is_resh is null) then
      is_resh = 0;

    if (ol_date > max_ol_date) then
      max_ol_date = ol_date;

    if (sh_os + sh_perc >= 0.01) then
    begin
      if (sh_date < oks_delinq_date) then
        oks_delinq_date = sh_date;

      oks_actual_last_date = sh_date;

      if (sh_date >= rep_date and curr_line = 0) then
      begin
        oks_curr_sh_os = sh_os;
        oks_curr_sh_perc = sh_perc;
        curr_line = 1;
      end
    end

    if (zad_all_prv > 0 and sh_date < rep_date) then
    begin
      if (sh_date < oks_delinq_date_prv) then
      begin
        oks_delinq_date_prv = sh_date;
        oks_delinq_day_prv = datediff(day, oks_delinq_date_prv, rep_date);
      end
    end

    if (oks_last_sh_date < last_date) then
      oks_last_sh_date = last_date;

    if (curr_max_sh_date < curr_sh_date) then
      curr_max_sh_date = curr_sh_date;

    if (prv_max_sh_date < prv_sh_date) then
      prv_max_sh_date = prv_sh_date;

    if (pmt_date > oks_last_pmt_date) then
      oks_last_pmt_date = pmt_date;

    if (is_resh = 1 and
        has_resh = 0) then
      has_resh = 1;

    oks_balance = oks_balance + ok_os;
    oks_sh_perc = oks_sh_perc + sh_perc;

    if (oks_balance <= 0 and ol_os_date <> min_date and ol_os_date < oks_close_os_date) then
      oks_close_os_date = ol_os_date;

    if (sh_date <= rep_date) then
    begin
      oks_delinq_os = oks_delinq_os + sh_os;
      oks_delinq_perc = oks_delinq_perc + sh_perc;
    end
  end

  if (oks_balance <= 0) then
  begin
    select ks_date_begin, ks_date_end
    from kontstate
    where ks_from_k_id = :k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin <= ks_date_end and
          ks_date_begin <= :rep_date
    into ks_date_begin, oks_close_date;

    if (ks_date_begin is null)
    then oks_close_state = 2; -- для последующего удаления записи
    else
    begin
      if (oks_close_date < '31.12.2999') then
        oks_close_date = oks_close_date + 1;
  
      if (oks_close_date <= rep_date)
      then oks_close_state = 1;
      else oks_close_date = null;
    end
  end

  if (has_writeoff = 1) then
  begin
    pmt_date = null;
    select max(oko_ol_date),
           sum(oko_os + oko_sh_os),
           sum(oko_perc + oko_sh_perc),
           sum(oko_other + oko_sh_other),
           max(iif(oko_type = 4 and (oko_os <> 0 or oko_perc <> 0 or oko_other <> 0), oko_ol_date, :min_date))
    from operkroff
    where oko_from_k_id = :k_id and
          oko_ol_is_active = 1 and
          oko_ol_date between :min_date and :rep_date
    into ol_date, oks_out_os, oks_out_perc, oks_out_other, pmt_date;

    if (ol_date > max_ol_date) then
      max_ol_date = ol_date;

    if (pmt_date is not null) then
      if (pmt_date > oks_last_pmt_date) then
        oks_last_pmt_date = pmt_date;

    if (oks_out_os is null) then
      oks_out_os = 0;

    if (oks_out_perc is null) then
      oks_out_perc = 0;

    if (oks_out_other is null) then
      oks_out_other = 0;
  end

  pmt_date = null;
  select max(ox_ol_date),
         sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff),
         max(iif(ox_amount_minus <> 0, ox_ol_date, :min_date))
  from operextra
  where ox_object_kind = 9 and
        ox_object_id = :k_id and
        ox_ol_is_active = 1 and
        ox_ol_date <= :rep_date
  into ol_date, oks_extra, pmt_date;

  if (ol_date > max_ol_date) then
    max_ol_date = ol_date;

  if (pmt_date is not null) then
    if (pmt_date > oks_last_pmt_date) then
      oks_last_pmt_date = pmt_date;

  if (oks_extra is null) then
    oks_extra = 0;

  if (oks_last_pmt_date = min_date) then
    oks_last_pmt_date = null;

  if (oks_delinq_date <= rep_date and
      oks_delinq_os + oks_delinq_perc >= 0.01) then
  begin
    oks_delinq_day = datediff(day, oks_delinq_date, rep_date);
    oks_is_delinq = 1;
  end

  if (oks_last_sh_date = min_date) then
    oks_last_sh_date = k_last_date;

  if (has_resh = 1) then
  begin
    oks_resh_count = 0;
    oks_resh_first = null;
    oks_resh_last = null;

    for select or_from_ol_id, or_ol_date
    from operrest
    where or_from_k_id = :k_id and
          or_ol_is_active = 1 and
          or_ol_date <= :rep_date
    order by or_ol_is_active, or_ol_date
    into ol_id, ol_date
    do
    begin
      oks_is_resh = 0;
      oks_resh_delinq_day = 0;
      oks_resh_delinq_os = 0;
      oks_resh_delinq_perc = 0;

      if (oks_resh_first is null) then
        oks_resh_first = ol_date;

      oks_resh_last = ol_date;
      oks_resh_count = oks_resh_count + 1;

      if (ol_date = rep_date) then
      begin
        oks_is_resh = 1;

        select max(datediff(day, ok_sh_date, ok_ol_date)), sum(-ok_sh_os), sum(-ok_sh_perc)
        from operkred
        where ok_from_ol_id = :ol_id and
              ok_type = 2 and
              ok_sh_date < ok_ol_date and
              (ok_sh_os <> 0 or ok_sh_perc <> 0)
        into oks_resh_delinq_day, oks_resh_delinq_os, oks_resh_delinq_perc;
        if (oks_resh_delinq_day is null) then
          oks_resh_delinq_day = 0;
        if (oks_resh_delinq_os is null) then
          oks_resh_delinq_os = 0;
        if (oks_resh_delinq_perc is null) then
          oks_resh_delinq_perc = 0;
      end
    end

    if (oks_last_sh_date > k_last_date) then
      oks_resh_max_prolong = datediff(day, k_last_date, oks_last_sh_date);

    if (curr_max_sh_date > prv_max_sh_date) then
      oks_resh_prolong_day = datediff(day, prv_max_sh_date, curr_max_sh_date);
  end

  if (oks_close_state = 1 and
      oks_extra = 0 and
      oks_out_os + oks_out_perc = 0) then
    oks_close_state = 2;

  if (oks_close_state = 2 and
      disb_date = oks_close_date and
      oks_close_date = rep_date) then
    oks_close_state = 3;

  suspend;
end
