CREATE PROCEDURE HP_CHECK_LOAN_KONDITION_ONE(
    K_ID D_ID,
    MA_ID D_INTEGER,
    CDM_ID D_INTEGER,
    CHECK_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    KF_CLIENT_FIO D_STR_100,
    KF_ERROR_TEXT D_STR_255,
    KF_ERROR_IS_CRITICAL D_INTEGER,
    SQL_ID D_INTEGER)
AS
declare variable SQL_TEXT blob sub_type 1 segment size 80;
declare variable KF_CLIENT_PASS_NO varchar(100);
declare variable KF_DISB_DATE date;
declare variable KC_AMOUNT decimal(18,4) = 0;
declare variable START_TIME D_DATETIME;
declare variable OBJ_KIND D_INTEGER;
declare variable OBJ_ID D_ID;
declare variable MAX_SECOND D_CURRENCY;
declare variable CAN_LOG D_SMALLINT;
declare variable FINISH_TIME D_DATETIME;
declare variable IS_ERROR D_SMALLINT;
declare variable EMAIL_LIST D_STR_255;
declare variable EMAIL_HEADER_LONG_EXEC D_STR_2048;
declare variable EMAIL_HEADER_EXCEPTION D_STR_2048;
declare variable EMAIL_SUBJECT D_STR_255;
declare variable DIFF_TIME D_CURRENCY;
declare variable CDM_DONT_LOG_TO_MAIL D_SMALLINT;
declare variable DB_CODE D_STR_10;
declare variable PROC_NAME D_STR_CODE;
declare variable CDM_USE_STORED_PROC D_BOOL;
declare variable ERROR_CODE bigint;
declare variable PR_ID D_INTEGER;
begin
  max_second = 1.0; -- секунд, если запрос выполняется дольше чем этот параметр, инфо логируется и отправляется по почте
  -- в случае возникновении ошибки при выполнении, сразу логируется и отправляется по почте

  email_list = 'finstart@mbfplanet.com';

  if (k_id is null) then
    k_id = 0;
  if (ma_id is null) then
    ma_id = 0;

  if (check_kind is null) then
    check_kind = '';

  can_log = iif(rdb$get_context('SYSTEM', 'READ_ONLY') = false, 1, 0);

  email_header_long_exec = 'ID проверки(cdm_id): <b>%1</b>,<br><br>Параметры:<br><br>obj_id: <b>%2</b><br>check_kind: <b>%3</b><br>un_id: <b>%4</b><br>Время выполнения: <b>%5</b> сек.<br><br>Текст запроса:<br><br>';
  email_header_exception = 'ID проверки(cdm_id): <b>%1</b>,<br><br>Параметры:<br><br>obj_id: <b>%2</b><br>check_kind: <b>%3</b><br>un_id: <b>%4</b><br>GdsCode: <b>%5</b><br><br>Текст запроса:<br><br>';

  if (k_id <> 0) then
  begin
    obj_kind = f_objkind('kontract', 'id');
    obj_id = k_id;

    select k.k_from_pr_id
    from kontract k
    where k.k_id = :obj_id
    into pr_id;
  end
  else
  begin
    obj_kind = f_objkind('miniapp', 'id');
    obj_id = ma_id;

    select ma.ma_from_pr_id
    from miniapp ma
    where ma.ma_id = :obj_id
    order by ma.ma_number desc
    rows 1
    into pr_id;
  end

  db_code = case f_get_db_id()
              when 81001 then 'RU'
              when 41701 then 'KG'
              when 39801 then 'KZ'
              when 94401 then 'AZ'
            end;

  for select cdm_sql, cdm_id, coalesce(cdm_dont_log_to_mail, 0), coalesce(cdm_use_stored_proc, 0)
      from conditionmain
      where (cdm_is_active = 1 or :cdm_id = 0) and -- Если проверка № 0, то проверяем если даже он отключен
            cdm_id = :cdm_id
      into sql_text, sql_id, cdm_dont_log_to_mail, cdm_use_stored_proc
  do
  begin
    if (exists(select *
               from product_condition_ignore pci            -- RFCKG-3811
               where pci.pci_from_pr_id = :pr_id
               and pci.pci_from_cdm_id = :sql_id
               and pci.pci_is_delete = 0) and db_code = 'KG') then
      continue;

    if (exists(select * from condition_ignore ci
               where ci.ci_object_kind = :obj_kind      -- RFCKG-3846
               and ci.ci_object_id = :obj_id
               and ci.ci_from_cdm_id = :sql_id) and (db_code = 'KG' or db_code = 'RU')) then -- RFCRU-3697: Модуль для КГ (RFCKG-3846) продублирован для РФ
      continue;

    if (cast(f_get_const('send_conditions_problem_on_email') as d_integer) = 0) then
        cdm_dont_log_to_mail = 1;

    start_time = cast('NOW' as timestamp);
    finish_time = null;
    proc_name = f_format('CM_%1_%2_SQL', :db_code, lpad(:sql_id, 4, '0'));

    -- Блок ловит исключения
    is_error = 0;
    error_code = 0;
    begin
      if (cdm_use_stored_proc = 1 and
          exists(select *
                 from rdb$procedures
                 where rdb$package_name is null and rdb$procedure_name = :proc_name)) then
      begin
        sql_text = 'select res_info_text, res_error_text, res_error_is_critical from ' || proc_name || '(:obj_id, :check_kind, :un_id)';

        for execute statement(sql_text)(obj_id := obj_id, check_kind := check_kind, un_id := un_id)
                into kf_client_fio, kf_error_text, kf_error_is_critical
        do
        begin
          if (coalesce(kf_client_fio, '') = '') then
            kf_client_fio = ' Общие условия';
          suspend;
        end
      end
      else
      begin
        if (k_id <> 0) then
        begin
          for execute statement(sql_text)(k_id := k_id)
                  into kf_client_fio, kf_client_pass_no, kf_disb_date, kc_amount, kf_error_text, kf_error_is_critical
          do
          begin
            if (coalesce(kf_client_fio, '') = '') then
              kf_client_fio = ' Общие условия';
            suspend;
          end
        end
        if (ma_id <> 0) then
        begin
          for execute statement(sql_text)(ma_id := ma_id)
                  into kf_client_fio, kf_client_pass_no, kf_disb_date, kc_amount, kf_error_text, kf_error_is_critical
          do
          begin
            if (coalesce(kf_client_fio, '') = '') then
              kf_client_fio = ' Общие условия';
            suspend;
          end
        end
      end
      when any do
      begin
        is_error = 1;
        error_code = gdscode;
      end
    end

    if (can_log = 1) then
    begin
      diff_time = 0;
      if (is_error = 0) then
      begin
        finish_time = cast('NOW' as timestamp);
        diff_time = round(cast(datediff(millisecond, start_time, finish_time) as d_currency) / 1000.0, 1);
      end

      if (f_get_const('ext_cond_log_enabled') = 1) then
      begin
        insert into conditionlog_ext (
          cde_from_k_id, cde_from_ma_id, cde_from_cdm_id, cde_check_kind,
          cde_from_un_id, cde_start_time, cde_finish_time)
        values (
          :k_id, :ma_id, :cdm_id, :check_kind, :un_id,
          :start_time, cast('NOW' as timestamp));
      end


      if (is_error = 1 or diff_time >= max_second) then
      begin
        in autonomous transaction do
          insert into conditionlog (cdl_from_cdm_id, cdl_param_obj_kind, cdl_param_obj_id, cdl_datetime_start,
                                    cdl_datetime_finish)
          values (:sql_id, :obj_kind, :obj_id, :start_time, :finish_time);

        if (cdm_dont_log_to_mail = 0) then
        begin
          sql_text = replace(sql_text, ascii_char(13), '<br>');
          sql_text = replace(sql_text, '  ', '&nbsp;&nbsp;');
          if (is_error = 1) then
            sql_text = blob_append(f_format(email_header_exception, sql_id, coalesce(obj_id, 0), check_kind, un_id, error_code), sql_text);
          else
            sql_text = blob_append(f_format(email_header_long_exec, sql_id, coalesce(obj_id, 0), check_kind, un_id, diff_time), sql_text);

          if (is_error = 1) then
            email_subject = 'Ошибка выполнения HP_CHECK_LOAN_KONDITIONS';
          else
            email_subject = 'Длительное выполнение HP_CHECK_LOAN_KONDITIONS';
          in autonomous transaction do
            execute procedure hp_send_quick_message('email', email_list, email_subject, sql_text, '', null,
                1000000000000);
        end
      end
    end

    if (is_error = 1) then
    begin
      kf_error_is_critical = 2;
      kf_client_fio = 'Техническая ошибка БД';
      if (can_log = 1) then
        kf_error_text = 'Отчет об ошибке был отправлен разработчикам';
      else
        kf_error_text = 'Нажмите кнопку Отправить e-mail тех. поддержке';
      suspend;
    end
  end
end
