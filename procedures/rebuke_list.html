CREATE PROCEDURE REBUKE_LIST(
    REGION_ID INTEGER,
    DATE_BEGIN DATE,
    DATE_END DATE,
    CL_FIO VARCHAR(255),
    BONUS_CHANGE SMALLINT,
    DOC_TYPE SMALLINT,
    DOC_NUMBER VARCHAR(255))
RETURNS (
    RB_ID INTEGER,
    CURRENT_REG_NAME VARCHAR(255),
    DOC_REG_NAME VARCHAR(255),
    RB_NUMBER VARCHAR(255),
    RB_DATE DATE,
    C_NAME_FIO VARCHAR(255),
    RB_TYPE VARCHAR(255),
    RB_BONUS_CHANGE SMALLINT,
    RB_BONUS_KOEF DECIMAL(18,2),
    RB_PERIOD_START DATE,
    RB_PERIOD_END DATE,
    INIT_FIO VARCHAR(255),
    IS_WORKING SMALLINT,
    RB_REASON VARCHAR(255),
    JL_NAME VARCHAR(255),
    RS_SCORES D_SMALLINT)
AS
declare variable rb_from_c_id id_type;
declare variable rb_type_id d_smallint;
begin
  /*
    CREATED BY IGUSEV 04.08.2012
    PURPOSE : Модуль - "Регистрация выговоров и замечаний"
              Журнал документов. Предоставляет список документов "Выговоров" в соответствии с настройками фильтра

    Выборка осуществляется по последнему офису сотрудника
  */

  if (cl_fio is null) then
    cl_fio = '';
  cl_fio = lower(cl_fio);
  if (bonus_change is null) then
    bonus_change = 2;
  if (doc_number is null) then
    doc_number = '';
  if (doc_type is null) then
    doc_type = 0;

  for select rb_id,
             reg.reg_name,
             reg_doc.reg_name,
             rb_number,
             rb_date,
             cl.c_name_fio,
             rb_type,
             rb_bonus_change,
             rb.rb_bonus_koef,
             rb.rb_period_start,
             rb.rb_period_end,
             cli.c_name_fio as init_fio,
             hrps.ps_is_working,
             rb.rb_reason,
             rb_from_c_id,
             rs.rs_scores
      from rebuke rb
      inner join clients cl on cl.c_id = rb.rb_from_c_id
      left join clients cli on cli.c_id = rb.rb_initiator_id
      inner join hr_pstatus hrps on hrps.ps_from_c_id = cl.c_id -- для выявления конечного офиса работы сотрудника
      inner join region_list(:region_id, 1, -1, 1) rl on hrps.ps_from_reg_id = rl.id_list
      inner join region reg on reg.reg_unicode = rl.id_list /*текущий офис сотрудника*/
      inner join region reg_doc on reg_doc.reg_unicode = rb.rb_from_reg_id /*офис документа*/
      left join rebuke_scores rs on rb.rb_id = rs.rs_from_rb_id and rs.rs_is_delete = 0
      where rb.rb_is_delete = 0 and
            cast('now' as date) between hrps.ps_date_begin and hrps.ps_date_end and -- всегда только опследний офис работы сорудника на текущую дату
            ((lower(cl.c_name_fio) like '%' || :cl_fio || '%') or (:cl_fio = '')) and
            rb.rb_date between :date_begin and :date_end and
            ((rb.rb_type = :doc_type) or (:doc_type = 0)) and
            ((rb.rb_number like '%' || :doc_number || '%') or (:doc_number = '')) and
            ((rb.rb_bonus_change = :bonus_change) or (:bonus_change = 2))
      order by rb.rb_number_n, rb.rb_sub_number
      into rb_id,
           current_reg_name,
           doc_reg_name,
           rb_number,
           rb_date,
           c_name_fio,
           rb_type_id,
           rb_bonus_change,
           rb_bonus_koef,
           rb_period_start,
           rb_period_end,
           init_fio,
           is_working,
           rb_reason,
           rb_from_c_id,
           rs_scores
  do
  begin
    rb_type = '';
    select el_name
    from enum_list
    where el_kind = 'rebuke_type' and
          el_id = :rb_type_id
    into rb_type; -- RFCKG-3685

    select jl.jl_name
    from hr_pstatus hrps
    inner join joblist jl on jl.jl_id = hrps.ps_from_jl_id
    where hrps.ps_from_c_id = :rb_from_c_id and
          :rb_date between hrps.ps_date_begin and hrps.ps_date_end
    into :jl_name;

    suspend;
  end
end
