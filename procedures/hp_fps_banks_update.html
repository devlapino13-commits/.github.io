CREATE PROCEDURE HP_FPS_BANKS_UPDATE(
    BANKS_LIST D_BLOB_TEXT,
    PMT_PROJECT D_STR_CODE = 'moneta')
AS
declare variable bank_count d_integer;
declare variable i d_integer;
declare variable fpb_code d_str_255;
declare variable fpb_name d_str_code;
declare variable jso_bank d_str_long;
declare variable fpb_status d_smallint;
begin
  -- Процедура обновляет список банков Системы Быстрых Платежей
  -- Вызывается через сервис выдач

  -- Сделано для переходного периода, далее можно удалить эту строку
  pmt_project = coalesce(pmt_project, 'moneta');

  if (coalesce(banks_list, '') = '') then
    exception e_default 'Empty banks_list!';

  if (pmt_project not in ('moneta', 'wirebank')) then
    exception e_default 'Unknown project:' || coalesce(pmt_project, 'NULL');

  -- Переводим всех действующих в кандидаты на удаление
  if (pmt_project = 'moneta') then
  begin
    update fps_banks
    set fpb_status_moneta = -999
    where fpb_status_moneta = 1;
  end
  if (pmt_project = 'wirebank') then
  begin
    update fps_banks
    set fpb_status_wirebank = -999
    where fpb_status_wirebank = 1;
  end

  -- Восстанавливаем тех, которые есть в json. При необходимости, создаем
  bank_count = f_json_blob_get_count(banks_list, 'banks');
  i = 0;
  while (i < bank_count) do
  begin
    jso_bank = f_json_blob_get_value(banks_list, f_format('banks[%1]', i), null);
    fpb_code = f_json_str_get_value(jso_bank, 'id', null);
    fpb_name = f_json_str_get_value(jso_bank, 'value', null);
    if (fpb_code is not null and fpb_name is not null) then
    begin
      fpb_status = null;
      select fpb_status from fps_banks
      where fpb_code = :fpb_code
      into fpb_status;

      if (fpb_status <> -1) then
      begin
        -- Если есть в списке, но не в статусе Отключено, восстанавливаем
        update fps_banks
        set
          fpb_status_moneta = iif(:pmt_project='moneta', 1, fpb_status_moneta),
          fpb_status_wirebank = iif(:pmt_project='wirebank', 1, fpb_status_wirebank)
        where fpb_code = :fpb_code;
      end
      else if (fpb_status is null) then
      begin
        -- Добавляем новые
        insert into fps_banks (fpb_code,  fpb_name, fpb_status, fpb_status_moneta, fpb_status_wirebank)
        values (:fpb_code, :fpb_name, 0, iif(:pmt_project='moneta', 1, 0), iif(:pmt_project='wirebank', 1, 0));
      end
    end
    i = i + 1;
  end

  -- Если не обновили в цикле, значит отсутствует в списке, удаляем
  update fps_banks
  set fpb_status_moneta = 0
  where fpb_status_moneta = -999;

  update fps_banks
  set fpb_status_wirebank = 0
  where fpb_status_wirebank = -999;

  -- Считаем удаленным (отключенным), если банк отсутствует в одной из систем
  update fps_banks
  set fpb_status = minvalue(fpb_status_moneta, fpb_status_wirebank)
  where fpb_status > -1;
end
