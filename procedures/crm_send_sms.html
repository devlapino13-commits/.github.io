CREATE PROCEDURE CRM_SEND_SMS(
    CELL_NO D_STR_100,
    SMS_TEXT D_STR_255)
RETURNS (
    ERROR_TEXT D_STR_255,
    RES_CODE D_STR_10)
AS
begin
  res_code = 'ok';
  error_text = '';
  cell_no = trim(coalesce(cell_no, ''));
  sms_text = trim(coalesce(sms_text, ''));

  if (cell_no = '') then
  begin
    res_code = 'error';
    error_text = 'Телефонный номер не указан';
    suspend;
    exit;
  end

  if (sms_text = '') then
  begin
    res_code = 'error';
    error_text = 'Текст сообщения не указан';
    suspend;
    exit;
  end

  execute procedure hp_send_quick_message('sms', :cell_no, '', :sms_text, '', -1, 1000000000000);
  suspend;
end
