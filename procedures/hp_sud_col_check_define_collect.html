CREATE PROCEDURE HP_SUD_COL_CHECK_DEFINE_COLLECT
AS
declare variable slc_id d_id;
declare variable bls_id d_id;
declare variable json_str d_str_8100;
declare variable target d_str_code;
declare variable sag_kind d_integer;
declare variable sag_code d_str_code;
declare variable sag_name d_str_255;
declare variable sag_address d_str_255;
declare variable sag_email d_str_255;
declare variable sag_phone d_str_255;
declare variable sag_id d_id;
begin
  -- процедура ищет ответы на запросы определения подсудности (ФССП)

  -- найдем заявки ожидающие подсудности
  for select distinct smc_from_slc_id
  from sud_maincollect
  where smc_status = 0
    and smc_auto_collect is null
  into slc_id
  do
  begin
    bls_id = null;

    select bls_id
    from hp_sud_get_request_status(:slc_id, 3)
    where eil_status = 2
    into bls_id;

    if (bls_id is null) then
      continue;

    json_str = f_json_blob_get_value('{"items":'||(select blx_blob from hp_get_blobex_blob(:bls_id))||'}', 'items[0]', '');

    target = f_json_str_get_value(json_str, 'TARGET', '');

    if (target <> 'BAILIFF') then
      exception e_default f_format('wrong target "%1". slc_id = %2', target, slc_id);

    sag_kind = f_enum('sud_agent_kind', 'collector', 'id');
    sag_code = f_json_str_get_value(json_str, 'BAILIFF_CODE', '');
    sag_name = f_json_str_get_value(json_str, 'BAILIFF_NAME', '');
    sag_address = f_json_str_get_value(json_str, 'BAILIFF_POSTCODE', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'BAILIFF_REGION_NAME', '') || ' ' || f_json_str_get_value(json_str, 'BAILIFF_REGION_TYPE', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'BAILIFF_DISTRICT_NAME', '') || ' ' || f_json_str_get_value(json_str, 'BAILIFF_DISTRICT_TYPE', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'BAILIFF_CITY_TYPE', '-') || '. ' || f_json_str_get_value(json_str, 'BAILIFF_CITY_NAME', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'BAILIFF_LOCALITY_TYPE', '-') || '. ' || f_json_str_get_value(json_str, 'BAILIFF_LOCALITY_NAME', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'BAILIFF_STREET_TYPE', '-') || '. ' || f_json_str_get_value(json_str, 'BAILIFF_STREET_NAME', '-') || ', ' ||
                  f_json_str_get_value(json_str, 'BAILIFF_HOUSE', '') || ' ' || f_json_str_get_value(json_str, 'BAILIFF_BLOCK', '');

    sag_address = replace(sag_address,'-. ', '');
    sag_address = replace(sag_address,'-, ', '');

    sag_email = f_json_str_get_value(json_str, 'BAILIFF_EMAIL', '');
    sag_phone = f_json_str_get_value(json_str, 'BAILIFF_PHONE', '');

    update or insert into sud_agents (sag_kind, sag_code, sag_name, sag_address, sag_email, sag_phone)
    values (:sag_kind, :sag_code, :sag_name, :sag_address, :sag_email, :sag_phone)
    matching (sag_kind, sag_code)
    returning sag_id
    into sag_id;

    update sud_maincollect
    set smc_auto_collect = :sag_id
    where smc_from_slc_id = :slc_id
      and smc_auto_collect is null;

  end 
end
