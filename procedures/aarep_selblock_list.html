CREATE PROCEDURE AAREP_SELBLOCK_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_C_ID D_ID)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    C_BIRTH_DATE D_DATE,
    REG_NAME D_STR_255,
    MA_ID D_ID,
    MA_DATE D_DATE,
    ENTITY_NUM D_STR_255,
    KI_DATE D_DATE,
    SB_DATE_CREATE D_DATE,
    SB_DATE_BEGIN D_DATE,
    SB_DATE_END D_DATE,
    SB_DATE_UPDATE D_DATE,
    STATUS D_STR_CODE,
    REFUSAL_REASON D_STR_255,
    M_DATETIME D_DATETIME,
    M_DATETIME_EDIT D_DATETIME,
    M_USER_NAME D_STR_255)
AS
declare variable k_id d_id;
declare variable eic_id d_id;
declare variable k_status d_integer;
declare variable ma_is_active d_integer;
declare variable ref_datetime d_datetime;

declare procedure hp_ma_list (reg_id d_integer, rep_date_begin d_date, rep_date_end d_date, rep_c_id d_id)
returns (rep_ma_id d_id)
as
begin
  if (rep_c_id is not null) then
    for select distinct mac_from_ma_id
    from miniappclient, selfblock
    where mac_from_c_id = :rep_c_id
      and sbk_from_c_id = mac_from_c_id
    into rep_ma_id
    do
      suspend;
  else
    for select distinct ma_id
    from region_list(:reg_id, 1, -1, 1), miniapp, miniappclient, selfblock
    where ma_office_id = id_list
      and mac_from_ma_id = ma_id
      and sbk_from_c_id = mac_from_c_id
      and ma_date_end = '31.12.9999'
      and ma_date_app between :rep_date_begin and :rep_date_end
    into rep_ma_id
    do
      suspend;
end

begin
  eic_id = (select eic_id from extinfocfg where eic_code = 'bki_ssp_40');

  for select ma_id, c_id, c_name_fio, c_birth_date, reg_code || ' ' || reg_name, ma_is_active, ma_date_app, ma_no
  from hp_ma_list(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_c_id), miniapp, miniappclient, clients, region
  where ma_id = rep_ma_id
    and ma_date_end = '31.12.9999'
    and mac_from_ma_id = ma_id
    and c_id = mac_from_c_id
    and reg_unicode = ma_office_id
  order by ma_date_app, ma_id
  into ma_id, c_id, c_name_fio, c_birth_date, reg_name, ma_is_active, ma_date, entity_num
  do
  begin
    sb_date_begin = null;
    sb_date_end = null;
    sb_date_create = null;
    sb_date_update = null;
    k_status = null;

    select max(eil_date)
    from extinfolist
    where eil_from_eic_id = :eic_id
      and eil_object_kind = 11
      and eil_object_id = :ma_id
    into ki_date;

    select sbk_date_begin, sbk_date_end, sbk_create_date, sbk_update_date
    from selfblock
    where sbk_from_c_id = :c_id
      and sbk_create_date <= coalesce(:ki_date, :ma_date)
    order by sbk_update_date desc
    rows 1
    into sb_date_begin, sb_date_end, sb_date_create, sb_date_update;

    if (row_count = 0) then
      continue;

    select :entity_num || ' / ' || k_kontract_no, k_status, k_id
    from kontract
    where k_from_ma_app_id = :ma_id
    order by k_status desc, k_id desc
    rows 1
    into entity_num, k_status, k_id;

    if (ma_is_active < 1) then
      status = 'Отказ';
    else
    begin
      status = 'В процессе';

      if (k_status > 0) then
      begin
        status = 'Утвержден';
  
        select 'Выдан'
        from kontstate
        where ks_from_k_id = :k_id
          and ks_state = 1
          and ks_ol_is_delete = 0
        into status;
      end
    end
    --- RFCRU-4739 Список клиентов с самозапретом
    refusal_reason = null;
    ref_datetime   = null;

    select krs.kr_text, krl.krl_datetime
    from kontreslist krl
    left join kontres krs on krs.kr_id = krl.krl_from_kr_id
    where (krl.krl_ma_id = :ma_id or krl.krl_from_k_id = :k_id)
    order by krl.krl_datetime desc
    rows 1
    into refusal_reason, ref_datetime;

    m_datetime = null;
    m_datetime_edit = null;
    m_user_name = null;

    select m.m_datetime, m.m_datetime_edit, u.un_name || ' ' || u.un_fio
    from moratory m
    left join username u on u.un_id = m.m_user_edit
    where m.m_from_c_id = :c_id
      and cast(m.m_datetime as date) = :ma_date
      and m.m_is_active = 0
    order by m.m_datetime desc
    rows 1
    into m_datetime, m_datetime_edit, m_user_name;

    if (m_datetime is null) then
    m_datetime = ref_datetime;

    suspend;
  end
end
