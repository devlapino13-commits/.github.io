CREATE PROCEDURE AAREP_GOODOWN_BALANCE(
    REG_ID D_INTEGER,
    DT_BEGIN D_DATE,
    DT_END D_DATE)
RETURNS (
    SELLER_ID D_ID,
    WP_ID D_ID,
    MO_ID D_ID,
    ML_ID D_ID,
    MC_ID D_CURRENCY,
    REGION_ID D_INTEGER,
    ACCOUNT D_STR_20,
    QTY D_CURRENCY,
    PRICE D_CURRENCY,
    SUMMA D_CURRENCY,
    SELLER_NAME D_STR_100,
    WP_NAME D_STR_100,
    MO_NAME D_STR_100,
    MC_NAME D_STR_100,
    REG_NAME D_STR_CODE,
    ML_NAME D_STR_255,
    YEDIZ D_STR_20)
AS
begin
  for
    select
      ob_from_own_c_id,
      wp_from_reg_id,
      om_workplace_id,
      om_matotvet_id,
      ob_from_ml_id,
      ob_from_mc_id,
      om_account,
      sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) as kol_balance,
      sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) as sum_balance
    from operbill
      inner join opermat
        on om_bill_id = ob_bill_id
          and om_bill_line_no = ob_bill_line_no
      inner join operlist
        on ol_id = om_from_ol_id
          and is_delete = 0
      inner join workplace
        on wp_id = om_workplace_id
          and exists(select * from region_list(:reg_id, 1, -1, 1) where id_list = wp_from_reg_id)
    where ob_from_own_c_id > 0
      and :dt_end between ob_date_begin and ob_date_end
      and ob_is_delete = 0
    group by ob_from_own_c_id, wp_from_reg_id, om_workplace_id, om_matotvet_id, ob_from_ml_id, ob_from_mc_id, om_account
    having (sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0)
      or (sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0)
  into
    :seller_id,
    :region_id,
    :wp_id,
    :mo_id,
    :ml_id,
    :mc_id,
    :account,
    :qty,
    :summa
  do
  begin
    seller_name = (select c_name_fio from clients where c_id = :seller_id);
    wp_name = (select wp_name from workplace where wp_id = :wp_id);
    mo_name = (select c_name_fio from clients where c_id = :mo_id);
    mc_name = (select mc_name from matcateg where mc_id = :mc_id);
    reg_name = (select reg_name from region where reg_unicode = :region_id);
    price = iif(qty = 0, 0, summa / qty);

    select
      ml_name,
      yl_code
    from matlist
      inner join yedizlist
        on yl_id = ml_from_yl_id
    where ml_id = :ml_id
    into
      :ml_name,
      :yediz;

    suspend;
  end
end
