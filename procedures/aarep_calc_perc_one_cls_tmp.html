CREATE PROCEDURE AAREP_CALC_PERC_ONE_CLS_TMP(
    K_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    FACT_DATE D_DATE,
    DELTA_BASE D_CURRENCY,
    PAY_OS D_CURRENCY,
    PAY_PERC D_CURRENCY,
    OFF_PERC D_CURRENCY,
    ACR7_PERC D_CURRENCY,
    VAC_STATUS D_INTEGER,
    VAC_PERC D_CURRENCY)
AS
declare variable prv_plan_date d_date;
declare variable plan_date d_date;
declare variable orig_fact_date d_date;
declare variable total_base d_currency;
declare variable zad_os d_currency;
declare variable orig_pay_os d_currency;
declare variable orig_pay_perc d_currency;
declare variable orig_off_perc d_currency;
declare variable orig_acr7_perc d_currency;
declare variable total_minus d_currency;
declare variable sh_id d_integer;
declare variable is_transfer d_bool;
declare variable transfer_list d_str_255;
declare variable transfer_item d_str_10;
declare variable transfer_saldo d_currency;
declare variable transfer_last_date d_date;
declare variable min_date d_date;
declare variable k_is_vacated d_bool;
declare variable vac_date_begin d_date;
declare variable vac_date_end d_date;
begin
  total_base = 0;
  delta_base = 0;
  total_minus = 0;
  transfer_saldo = 0;
  prv_plan_date = null;
  transfer_list = '';
  pay_os = 0;
  pay_perc = 0;
  off_perc = 0;
  acr7_perc = 0;
  vac_perc = 0;
  vac_status = 0;

  for select kd_ol_date, k_is_vacated
  from kontdata, kontract
  where kd_from_k_id = :k_id and
        kd_ol_date <= :rep_date_end and
        kd_ol_date <= kd_date_end and
        kd_ol_is_delete = 0 and
        kd_from_k_id = k_id
  order by kd_ol_date
  into fact_date, k_is_vacated
  do
  begin
    if (min_date is null) then
      min_date = fact_date;

    suspend;
  end

  if (k_is_vacated = 1) then
  begin
    select vac_date_begin, vac_date_end
    from hp_kont_vac_info(:k_id, :rep_date_end)
    into :vac_date_begin, :vac_date_end;

    if (vac_date_begin is not null) then
    begin
      fact_date = vac_date_begin;
      vac_status = 1;
      suspend;

      fact_date = vac_date_end;
      vac_status = 2;
      suspend;
    end
  end
  vac_status = 0;

  for select ok_sh_date,
             ok_ol_date,
             ok_sh_id,
             iif(ok_type = 2 and ol_oper_type = 1 and ol_oper_subtype = 9, 1, 0),
             iif(ok_ol_date <= ok_sh_date, ok_os + ok_sh_os, 0),
             ok_os,
             iif(ok_type = 4, -ok_perc, 0),
             iif(ok_type in (5, 6), -ok_perc, 0),
             iif(ok_type = 7, ok_perc, 0)
  from operkred, operlist
  where ok_from_k_id = :k_id and
        ok_ol_date <= :rep_date_end and
        ok_ol_is_active = 1 and
        ok_from_ol_id = ol_id and
        ok_type <= 7
  union all
  select cast('31.12.2999' as date), cast('31.12.2999' as date), 0, 0, 0, 0, 0, 0, 0
  from company
  order by 1, 2, 6, 5
  into plan_date, orig_fact_date, sh_id, is_transfer,
       zad_os, orig_pay_os, orig_pay_perc, orig_off_perc, orig_acr7_perc
  do
  begin
    if (plan_date < min_date) then
      plan_date = min_date;

    if (prv_plan_date is null) then
      prv_plan_date = plan_date;

    if (prv_plan_date <> plan_date) then
    begin
      delta_base = -(total_base + total_minus);
      fact_date = prv_plan_date;
      pay_os = 0;
      pay_perc = 0;
      off_perc = 0;
      acr7_perc = 0;
      suspend;

      total_base = 0;
      delta_base = 0;
      total_minus = 0;
    end

    if (is_transfer = 1 and zad_os > 0) then
      zad_os = 0;

    transfer_item = '[' || sh_id || ']';

    if (position(transfer_item, transfer_list) = 0) then
    begin
      if (is_transfer = 1) then
      begin
        transfer_list = transfer_list || transfer_item;
        transfer_saldo = transfer_saldo - zad_os;
        if (transfer_last_date is null)
        then transfer_last_date = orig_fact_date;
        else transfer_last_date = minvalue(transfer_last_date, orig_fact_date);
      end
    end else
    begin
      if (is_transfer = 1) then
        zad_os = 0;

      if (zad_os < 0 and orig_fact_date >= coalesce(transfer_last_date, orig_fact_date)) then
      begin
        if (transfer_saldo + zad_os >= 0) then
        begin
          transfer_saldo = transfer_saldo + zad_os;
          zad_os = 0;
        end else
        begin
          zad_os = zad_os + transfer_saldo;
          transfer_saldo = 0;
          transfer_last_date = null;
        end
      end
    end

    fact_date = orig_fact_date;
    pay_os = orig_pay_os;
    pay_perc = orig_pay_perc;
    off_perc = orig_off_perc;
    acr7_perc = orig_acr7_perc;

    delta_base = zad_os;
    total_base = total_base + zad_os;

    if (total_base < 0) then
    begin
      delta_base = delta_base - total_base;
      total_minus = total_minus + total_base;
      total_base = 0;
    end else
    begin
      if (total_minus + total_base >= 0) then
      begin
        total_base = total_base + total_minus;
        delta_base = delta_base + total_minus;
        total_minus = 0;
      end else
      begin
        total_minus = total_minus + total_base;
        delta_base = delta_base - total_base;
        total_base = 0;
      end
    end

    suspend;

    prv_plan_date = plan_date;
    delta_base = 0;
  end

  delta_base = 0;
  total_base = 0;
  pay_os = 0;
  pay_perc = 0;
  off_perc = 0;
  acr7_perc = 0;

  if (k_is_vacated = 1) then
  begin
    vac_status = 1;
    for select ol_date, -ov_os
    from opervac, operlist ol
    where ov_from_ol_id = ol.ol_id and
          ov_from_k_id = :k_id and
          ol.ol_date <= :rep_date_end and
          ol.is_delete = 0 and
          ov_type = 1 and
          ov_os <> 0
    into fact_date, delta_base
    do
      suspend;

    delta_base = 0;
    vac_status = 0;
    for select ol_date, ov_perc
    from opervac, operlist ol
    where ov_from_ol_id = ol.ol_id and
          ov_from_k_id = :k_id and
          ol.ol_date <= :rep_date_end and
          ol.is_delete = 0 and
          ov_type = 4 and
          ov_perc <> 0
    into fact_date, vac_perc
    do
      suspend;
  end

  vac_status = 0;
  vac_perc = 0;
  fact_date = rep_date_end;
  delta_base = 0;
  total_base = 0;
  pay_os = 0;
  pay_perc = 0;
  off_perc = 0;
  acr7_perc = 0;
  suspend;
end
