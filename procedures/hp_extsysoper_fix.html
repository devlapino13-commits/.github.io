CREATE PROCEDURE HP_EXTSYSOPER_FIX(
    OBJECT_KIND D_STR_100,
    OBJECT_ID D_ID,
    REP_DATE_BEGIN D_DATE = '01.12.2020')
AS
declare variable esm_amount d_currency;
declare variable esm_date d_date;
declare variable extsysoper_ol_id d_id;
declare variable ol_id d_id;
declare variable ol_doc_client_name d_str_255;
declare variable k_id d_id;
declare variable rrl_lst_new_ol_id d_str_2048;
declare variable ol_new d_id;
declare variable ol_parent_id d_id;
declare variable eo_id d_id;
declare variable rrl_lst_new_ol_id_tmp d_str_255;
declare variable esm_id d_id;
declare variable sql_text d_blob_text;
declare variable is_inserted d_smallint;
begin
  sql_text = '';

  if (object_kind = 'extsysmain') then
  begin
    sql_text = blob_append(sql_text,'select em.esm_id, em.esm_amount, em.esm_date ');
    sql_text = blob_append(sql_text,'from extsysmain em ');
    sql_text = blob_append(sql_text,'where em.esm_id = '||:object_id||' ');
    sql_text = blob_append(sql_text,'and em.esm_status = -3 ');
  end
  else if (object_kind = 'kontract') then
  begin
    sql_text = blob_append(sql_text,'select em.esm_id, em.esm_amount, em.esm_date ');
    sql_text = blob_append(sql_text,'from knclient kncl ');
    sql_text = blob_append(sql_text,'inner join extsysmain em on em.esm_object_kind = 0 ');
    sql_text = blob_append(sql_text,'and em.esm_object_id = kncl.kc_from_c_id ');
    sql_text = blob_append(sql_text,'and em.esm_status = -3 ');
    sql_text = blob_append(sql_text,'and em.esm_date >= '''||f_datetostr(:rep_date_begin)||''' ');
    sql_text = blob_append(sql_text,'where kncl.kc_from_k_id = '||:object_id||' ');
    sql_text = blob_append(sql_text,'and kncl.kc_amount_approved > 0 ');
    sql_text = blob_append(sql_text,'order by em.esm_id ');
  end
  else
    exit;

  for execute statement(sql_text)
  into :esm_id, :esm_amount, :esm_date
  do
  begin
        extsysoper_ol_id = null;

        select eo.eso_from_ol_id
        from extsysoper eo
        where eo.eso_from_esm_id = :esm_id
        order by eo.eso_id
        rows 1
        into :extsysoper_ol_id;

        if (extsysoper_ol_id is null) then continue;

        ol_id = null;
        ol_doc_client_name = null;

        select ol.ol_id,
               ol.ol_doc_client_name
        from operlist ol
        where ol.ol_parent_id = :extsysoper_ol_id
        order by ol.ol_id desc
        rows 1
        into :ol_id, :ol_doc_client_name;

        if (ol_id is null) then continue;

        k_id = null;

        select ok.ok_from_k_id
        from operkred ok
        where ok.ok_from_ol_id = :ol_id
        rows 1
        into :k_id;

        if (k_id is null) then continue;

        rrl_lst_new_ol_id = '';

        /*
        select rrl.rrl_id
        from reaccept_repayment_log rrl
        where rrl.rrl_lst_ol_id like '%'||:ol_id||'%'
        and rrl.rrl_from_k_id = :k_id
        into :rrl_id;

        if (rrl_id is null) then continue;  */

        for select rrl.rrl_lst_new_ol_id
        from reaccept_repayment_log rrl
       -- where rrl.rrl_id >= :rrl_id
        where rrl.rrl_from_k_id = :k_id
        order by  rrl.rrl_id desc
        rows 1
        into :rrl_lst_new_ol_id_tmp
        do
        begin
            rrl_lst_new_ol_id = rrl_lst_new_ol_id ||'|'|| rrl_lst_new_ol_id_tmp;
        end

        is_inserted = 0;

        for select cast(res_str as d_id)
        from hp_split_str(:rrl_lst_new_ol_id, '|')
        into :ol_new
        do
        begin
            if (is_inserted = 0) then
            begin
                ol_parent_id = null;
    
                select ol.ol_parent_id
                from operlist ol
                where ol.ol_id = :ol_new
                and ol.ol_doc_client_name = :ol_doc_client_name
                and ol.ol_date = :esm_date
                and ol.is_delete = 0
                into :ol_parent_id;
    
                if (ol_parent_id is not null) then
                begin
                    eo_id = null;
    
                    select ol.ol_id
                    from operlist ol
                    inner join operbook ob on ob.ob_from_ol_id = ol.ol_id
                    where ol.ol_id = :ol_parent_id
                    and ol.ol_doc_client_name = :ol_doc_client_name
                    and ob.ob_summa_val = :esm_amount
                    and ol.is_delete = 0
                    into :eo_id;
    
                    if (eo_id is not null) then
                    begin
                       if (exists(select *
                                  from extsysoper eo
                                  where eo.eso_from_ol_id = :eo_id)) then
                            continue;
    
                       insert into extsysoper (eso_from_esm_id, eso_from_ol_id)
                       values (:esm_id, :eo_id);
    
                       update extsysmain em
                       set em.esm_from_k_id = :k_id
                       where em.esm_id =:esm_id;
    
                       is_inserted = 1;
                    end
               end
            end
        end
    end
end
