CREATE PROCEDURE HP_KK_ACTION_BLOCK_CHECK(
    K_ID ID_TYPE,
    UN_ID ID_TYPE)
RETURNS (
    MSG_ERR STR_255)
AS
declare variable acl_id id_type;
declare variable acl_status integer;
declare variable acl_un_id id_type;
declare variable pr_code d_str_100;
begin
  msg_err = '';

  pr_code = (
    select
      pr_code
    from kontract
      inner join product
        on pr_id = k_from_pr_id
    where k_id = :k_id);

  select first 1 acl_id, acl_status, un_id
  from actionlist
  left join username on un_from_c_id = acl_respons_c_id
  where acl_object_kind = 9 and
        acl_object_id = :k_id
  order by acl_id desc
  into :acl_id, :acl_status, :acl_un_id;
  if (acl_id > 0) then
  begin
    if (acl_status in (-1, 0, 1)) then
    begin
      if (un_id <> acl_un_id)
      then msg_err = 'Задача по проверке кредитным комитетом заблокирована за другим пользователем.';
    end
    else
      msg_err = 'Задача по проверке кредитным комитетом уже исполнена.';
  end
  else
    msg_err = 'Кредит на кредитный комитет еще не был отправлен.';

  if (msg_err = '') then
  begin
    if (pr_code containing '.zalog_car.' or pr_code containing '.education.')
    then msg_err = 'Займ по данному продукту может быть рассмотрен только сотрудником КК';
  end

/*
  if (msg_err = '') then
  begin
    -- отключаем возможность расмотрения онлайн займов наКК через окно поиска контракта
    if (exists(select * from kontract
               where k_id = :k_id and k_is_online = 1))
    then
      msg_err = 'Займы с онлайн выдачей не могут быть рассмотрены менеджером.';
  end
*/
  suspend;
end
