CREATE PROCEDURE AAREP_CLIENTLOG_ONE(
    C_ID DECIMAL(18,0))
RETURNS (
    CCL_REC_NO INTEGER,
    CLL_ID DECIMAL(18,0),
    CLL_NAME_FIO VARCHAR(100),
    CLL_BIRTH_DATE DATE,
    CLL_PASS_NO VARCHAR(30),
    CLL_PASS_DATE DATE,
    CLL_CELL_NO VARCHAR(20),
    CLL_FROM_UN_ID DECIMAL(18,0),
    CLL_DATETIME TIMESTAMP,
    CLL_FRAUD_FIO INTEGER,
    CLL_FRAUD_BIRTH_DATE INTEGER,
    CLL_FRAUD_PASS_NO INTEGER,
    CLL_FRAUD_PASS_DATE INTEGER,
    CLL_FRAUD_CELL_NO INTEGER,
    CLL_FRAUD_AFTER_DISB INTEGER,
    CLL_CURR_DATETIME TIMESTAMP,
    CLL_CURR_USER_ID DECIMAL(18,0),
    CLL_FROM_C_ID DECIMAL(18,0),
    ORIG_UN_FIO VARCHAR(100),
    ORIG_UN_OFFICE VARCHAR(100),
    ORIG_UN_JOBLIST VARCHAR(100),
    CURR_UN_FIO VARCHAR(100),
    CURR_UN_OFFICE VARCHAR(100),
    CURR_UN_JOBLIST VARCHAR(100),
    CLL_STATUS INTEGER,
    CLL_AUDIT_UN_ID DECIMAL(18,0))
AS
declare variable un_c_id decimal(18,0);
begin
  cll_fraud_fio = 0;
  cll_fraud_birth_date = 0;
  cll_fraud_pass_no = 0;
  cll_fraud_pass_date = 0;
  cll_fraud_cell_no = 0;
  cll_from_c_id = :c_id;
  ccl_rec_no = 0;
  for select c_name_fio, c_birth_date, c_pass_no, c_pass_date,
             (select cont_value
              from hp_contdata_get_cell_main(0, :c_id)), c_from_un_id, c_datetime
      from clients
      where c_id = :c_id
      into :cll_name_fio, :cll_birth_date, :cll_pass_no, :cll_pass_date, :cll_cell_no, :cll_from_un_id, :cll_datetime
  do
  begin
    cll_curr_user_id = cll_from_un_id;
    cll_curr_datetime = cll_datetime;
    un_c_id = null;
    orig_un_fio = '';
    orig_un_office = 'Нет приказа';
    orig_un_joblist = 'Нет приказа';

    select c_id, c_name_fio
    from clients, username
    where un_from_c_id = c_id and
          un_id = :cll_from_un_id
    into :un_c_id, :orig_un_fio;

    select jl_name, reg_code || ' ' || reg_name
    from hr_pstatus, joblist, region
    where ps_from_c_id = :un_c_id and
          ps_date_begin <= current_date and
          ps_date_end >= current_date and
          ps_from_jl_id = jl_id and
          ps_from_reg_id = reg_unicode rows 1
    into :orig_un_joblist, :orig_un_office;

    curr_un_fio = orig_un_fio;
    curr_un_office = orig_un_office;
    curr_un_joblist = orig_un_joblist;

    suspend;
  end
  un_c_id = null;
  orig_un_fio = '';
  orig_un_office = 'Нет приказа';
  orig_un_joblist = 'Нет приказа';
  curr_un_fio = '';
  curr_un_office = 'Нет приказа';
  curr_un_joblist = 'Нет приказа';

  for select cll_id, cll_name_fio, cll_birth_date, cll_pass_no, cll_pass_date, cll_cell_no, cll_from_un_id,
             cll_datetime, cll_fraud_fio, cll_fraud_birth_date, cll_fraud_pass_no, cll_fraud_pass_date,
             cll_fraud_cell_no, cll_fraud_after_disb, cll_curr_datetime, cll_curr_user_id, cll_status, cll_audit_un_id
      from clientlog
      where cll_from_c_id = :c_id
      order by cll_id desc
      into :cll_id, :cll_name_fio, :cll_birth_date, :cll_pass_no, :cll_pass_date, :cll_cell_no, :cll_from_un_id,
           :cll_datetime, :cll_fraud_fio, :cll_fraud_birth_date, :cll_fraud_pass_no, :cll_fraud_pass_date,
           :cll_fraud_cell_no, :cll_fraud_after_disb, :cll_curr_datetime, :cll_curr_user_id, :cll_status,
           :cll_audit_un_id
  do
  begin
    ccl_rec_no = ccl_rec_no + 1;

    select c_id, c_name_fio
    from clients, username
    where un_from_c_id = c_id and
          un_id = :cll_from_un_id
    into :un_c_id, :orig_un_fio;

    select jl_name, reg_code || ' ' || reg_name
    from hr_pstatus, joblist, region
    where ps_from_c_id = :un_c_id and
          ps_date_begin <= current_date and
          ps_date_end >= current_date and
          ps_from_jl_id = jl_id and
          ps_from_reg_id = reg_unicode rows 1
    into :orig_un_joblist, :orig_un_office;

    select c_id, c_name_fio
    from clients, username
    where un_from_c_id = c_id and
          un_id = :cll_curr_user_id
    into :un_c_id, :curr_un_fio;

    select jl_name, reg_code || ' ' || reg_name
    from hr_pstatus, joblist, region
    where ps_from_c_id = :un_c_id and
          ps_date_begin <= current_date and
          ps_date_end >= current_date and
          ps_from_jl_id = jl_id and
          ps_from_reg_id = reg_unicode rows 1
    into :curr_un_joblist, :curr_un_office;

    suspend;
  end
end
