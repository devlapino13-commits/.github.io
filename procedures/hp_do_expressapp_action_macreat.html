CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_MACREAT(
    EXA_ID D_INTEGER)
RETURNS (
    RES_STATUS D_SMALLINT)
AS
declare variable exa_from_c_id bigint;
declare variable exa_base_k_id bigint;
declare variable exa_from_ma_id integer;
declare variable exa_disb_date date;
declare variable exa_from_pr_id integer;
declare variable exa_amount decimal(18,4);
declare variable exa_period integer;
declare variable exa_percent decimal(18,7);
declare variable exa_repay_day integer;
declare variable exa_lo_id d_id;
declare variable exa_from_reg_id integer;
declare variable exa_from_un_id d_id;
declare variable currency_id varchar(3);
declare variable loanfolderid id_type;
declare variable loanmanagerid id_type;
declare variable loanrepayday d_integer;
declare variable add_info d_str_255;
declare variable disb_kind d_integer;
declare variable tkt_id d_id;
declare variable is_online smallint;
declare variable ma_id d_integer;
declare variable ma_no d_integer;
declare variable ma_is_active d_smallint;
declare variable k_from_ma_app_id d_integer;
declare variable mac_loan_target d_integer;
declare variable mac_business_subtype d_integer;
declare variable mac_allow_transfer_rights d_smallint;
declare variable res_schedule_id d_integer;
declare variable db_id d_id;
declare variable pr_code d_str_100;
declare variable crel_other_c_id d_id;
declare variable ma_kommis_percent perc_type;
declare variable oa_card_type d_smallint;
begin
  db_id = f_get_db_id();
  res_status = 0;
  exa_from_c_id = null;
  exa_from_ma_id = null;
  select exa_from_c_id, exa_base_k_id, exa_from_ma_id, exa_disb_date, exa_from_pr_id, exa_amount, exa_period,
         exa_percent, exa_repay_day, exa_lo_id, exa_from_reg_id, exa_from_un_id
  from expressapp
  where exa_id = :exa_id
  into :exa_from_c_id, :exa_base_k_id, :exa_from_ma_id, :exa_disb_date, :exa_from_pr_id, :exa_amount, :exa_period,
       :exa_percent, :exa_repay_day, :exa_lo_id, :exa_from_reg_id, :exa_from_un_id;

  if (exa_from_ma_id is null) then
  begin
    select pr.pr_code
    from product pr
    where pr.pr_id = :exa_from_pr_id
    into pr_code;

    select cm_default_currency
    from company
    into :currency_id;

    if (db_id in (39801, 41701) and
        (pr_code containing '.installment_goods.' or
         pr_code containing '.sugarmill.' or
         pr_code containing '.online_loan.')) then
    begin
      loanfolderid = null;

      select kt.k_from_kf_id
      from knclient kc
      inner join kontract kt on kt.k_id = kc.kc_from_k_id
      inner join product pr on pr.pr_id = kt.k_from_pr_id
      inner join kontstate ks on ks.ks_from_k_id = kt.k_id
      where kc.kc_from_c_id = :exa_from_c_id and
            kc.kc_amount_approved > 0 and
            iif(:pr_code containing '.online_loan.', pr_code containing '.online_loan.', (pr.pr_code containing '.installment_goods.' or  pr.pr_code containing '.sugarmill.')) and
            current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state in (1,2) and
            ks.ks_region_id = :exa_from_reg_id
      order by kt.k_disb_date desc
      rows 1
      into loanfolderid;

      if (loanfolderid is null) then
      begin
        select kf_id
        from insert_new_kfolder(:exa_from_reg_id, :exa_lo_id, current_date, 0, '', 0, '')
        into loanfolderid;
      end

      select ps_from_c_id
      from hr_pstatus, joblist, jobkind
      where current_date between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_reg_id = :exa_from_reg_id and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              (jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Region_Director') or
              (jk_system_code = 'JobSystem.LO_Standart' and
              jl_name = 'Ведущий специалист'))
      order by ps_date_begin
      rows 1
      into loanmanagerid;

      loanrepayday = exa_repay_day;
      add_info = null;

      if (pr_code containing '.online_loan.') then
      begin
        select oa.oa_card_type
        from online_app oa
        where oa.oa_from_exa_id = :exa_id
        into oa_card_type;

        if (coalesce(oa_card_type, 0) = 0) then
          disb_kind = 6;
        else
          disb_kind = 5;
      end
      else
        disb_kind = 1;

      ma_kommis_percent = case
                            when disb_kind = 5 then 0.55
                            when disb_kind = 6 then 1.4
                            else 0
                          end;

      tkt_id = null;
      is_online = iif(disb_kind = 5, 1, 0);

      insert into miniapp (ma_id, ma_number, ma_no, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id,
                           ma_kolich_repay, ma_percent, ma_kommis_perc, ma_kolich_client, ma_memo, ma_is_active,
                           ma_is_auto, ma_summa_minus, ma_from_un_id, ma_office_id, ma_currency_id, ma_percent_category,
                           ma_manager_id, ma_repay_day, ma_source_system, ma_refund, ma_add_month, ma_add_info,
                           ma_prepayment_summa, ma_disb_kind, ma_from_tkt_id, ma_is_online)
      values (-1, -1, -1, current_date, :exa_disb_date, :loanfolderid, :exa_from_pr_id, :exa_period, :exa_percent,
              :ma_kommis_percent, 1, '', 1, 0, :exa_amount, :exa_from_un_id, :exa_from_reg_id, :currency_id,
              :exa_percent, :loanmanagerid, :loanrepayday, 1, 0, 0, :add_info, 0, :disb_kind, :tkt_id, :is_online)
      returning ma_id, ma_no
      into :ma_id, :ma_no;

      delete from miniappclient
      where mac_from_ma_id = :ma_id;

      if (pr_code containing '.installment_goods.' or pr_code containing '.online_loan.') then
      begin
        mac_loan_target = (select rd.rd_id
                           from refdata rd
                           where rd.rd_code = 'Loan_Target.Consumer'); --Потребительский

        mac_business_subtype = 21; -- Потребительский
      end 

      if (pr_code containing '.sugarmill.') then
      begin
        mac_loan_target = (select rd.rd_id
                           from refdata rd
                           where rd.rd_code = 'Loan_Target.BusinessGrowth'); --Развитие бизнеса

        mac_business_subtype = 18; --Сельское хозяйство
      end 


      mac_allow_transfer_rights = 1; -- разрешить передачу прав

      insert into miniappclient (mac_from_reg_id, mac_from_ma_id, mac_from_c_id, mac_amount, mac_is_poruch,
                                 mac_loan_target, mac_is_cashier, mac_business_subtype, mac_kommis_lgota,
                                 mac_allow_transfer_rights)
      values (:exa_from_reg_id, :ma_id, :exa_from_c_id, :exa_amount, 0, :mac_loan_target, 1, :mac_business_subtype, 0,
              :mac_allow_transfer_rights);

      if (pr_code containing '.sugarmill.' and exa_amount >= 50000) then
      begin
        crel_other_c_id = null;

        select t.crel_other_c_id
        from hp_client_relation_list(:exa_from_c_id, 0) t
        where t.crel_from_rel_id in ('1', '2')
        into crel_other_c_id;

        if (coalesce(crel_other_c_id, 0) > 0) then
        begin
          insert into miniappclient (mac_from_reg_id, mac_from_ma_id, mac_from_c_id, mac_amount, mac_is_poruch,
                                 mac_loan_target, mac_is_cashier, mac_business_subtype, mac_kommis_lgota,
                                 mac_allow_transfer_rights)
          values (:exa_from_reg_id, :ma_id, :crel_other_c_id, 0, -1, 0, 1, 0, 0,
                  :mac_allow_transfer_rights);
        end 
      end 

      select res_id
      from hp_make_schedule(:exa_from_un_id, 11, :ma_id, 0, null, 0)
      into res_schedule_id;

      update expressapp
      set exa_from_ma_id = :ma_id,
          exa_percent = :exa_percent,
          exa_from_edl_id = null -- обнуляем, чтобы в следующих обработках не мешала
      where exa_id = :exa_id;

      res_status = 0;
    end
    else
    begin
      select k_from_kf_id, k_from_ma_app_id
      from kontract
      where k_id = :exa_base_k_id
      into loanfolderid, k_from_ma_app_id;

      --  вычисляем менеджера
      loanmanagerid = null;
      select ma_manager_id
      from miniapp
      where ma_id = :k_from_ma_app_id and
            ma_date_end = '31.12.9999'
      order by ma_number desc
      rows 1
      into loanmanagerid;

      --  проверка если менеджер не работает, то находим активного мененджера
      if (not exists(select *
                     from hr_pstatus, joblist, jobkind
                     where ps_from_c_id = :loanmanagerid and
                           current_date between ps_date_begin and ps_date_end and
                           ps_status = 1 and
                           ps_is_working = 1 and
                           ps_from_jl_id = jl_id and
                           jl_kind = jk_id and
                           ps_from_reg_id = :exa_from_reg_id and
                           (jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Region_Director') or
                           (jk_system_code = 'JobSystem.LO_Standart' and
                           jl_name = 'Ведущий специалист')))) then
      begin
        select ps_from_c_id
        from hr_pstatus, joblist, jobkind
        where current_date between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_from_reg_id = :exa_from_reg_id and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              (jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Region_Director') or
              (jk_system_code = 'JobSystem.LO_Standart' and
              jl_name = 'Ведущий специалист'))
        order by ps_date_begin
        rows 1
        into loanmanagerid;
      end

      loanrepayday = exa_repay_day;
      add_info = null;
      disb_kind = iif(db_id = 81001, 7, 1);
      tkt_id = null;
      is_online = iif(db_id = 81001, 1, 0);
      -- создаем заявку
      insert into miniapp (ma_id, ma_number, ma_no, ma_date_app, ma_date_oper, ma_from_kf_id, ma_from_pr_id,
                           ma_kolich_repay, ma_percent, ma_kommis_perc, ma_kolich_client, ma_memo, ma_is_active,
                           ma_is_auto, ma_summa_minus, ma_from_un_id, ma_office_id, ma_currency_id, ma_percent_category,
                           ma_manager_id, ma_repay_day, ma_source_system, ma_refund, ma_add_month, ma_add_info,
                           ma_prepayment_summa, ma_disb_kind, ma_from_tkt_id, ma_is_online)
      values (-1, -1, -1, current_date, :exa_disb_date, :loanfolderid, :exa_from_pr_id, :exa_period, :exa_percent,
              iif(:db_id in (39801, 41701), 1.5, 0), 1, '', 1, 0, :exa_amount, :exa_from_un_id, :exa_from_reg_id, :currency_id,
              :exa_percent, :loanmanagerid, :loanrepayday, 1, 0, 0, :add_info, 0, :disb_kind, :tkt_id, :is_online)
      returning ma_id, ma_no
      into :ma_id, :ma_no;

      delete from miniappclient
      where mac_from_ma_id = :ma_id;

      mac_loan_target = iif(db_id = 81001, 73, 63); -- цель кредита потребительский
      mac_business_subtype = 21; -- потребительский
      mac_allow_transfer_rights = 1; -- разрешить передачу прав

      insert into miniappclient (mac_from_reg_id, mac_from_ma_id, mac_from_c_id, mac_amount, mac_is_poruch,
                                 mac_loan_target, mac_is_cashier, mac_business_subtype, mac_kommis_lgota,
                                 mac_allow_transfer_rights)
      values (:exa_from_reg_id, :ma_id, :exa_from_c_id, :exa_amount, 0, :mac_loan_target, 1, :mac_business_subtype, 0,
              :mac_allow_transfer_rights);

      select res_id
      from hp_make_schedule(:exa_from_un_id, 11, :ma_id, 0, null, 0)
      into res_schedule_id;

      update expressapp
      set exa_from_ma_id = :ma_id,
          exa_from_edl_id = null -- обнуляем, чтобы в следующих обработках не мешала
      where exa_id = :exa_id;

      res_status = 0;
    end
  end
  else
  begin
    if (exists(select *
               from miniapp
               where ma_id = :exa_from_ma_id and
                     ma_date_end = '31.12.9999' and
                     ma_is_active in (0, -1))) then
    begin
      -- если заявка аннулирован или отказан
      res_status = -1;

      update expressapp
      set exa_final_status = -1
      where exa_id = :exa_id;
    end
  end
  suspend;
end
