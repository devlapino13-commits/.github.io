CREATE PROCEDURE HP_PROD_CDM_IGNORE_HISTORY(
    PR_ID D_INTEGER,
    PRL_ID D_INTEGER)
RETURNS (
    CDM_ID D_INTEGER,
    C_NAME_FIO D_STR_100,
    PCI_LAST_DATETIME D_DATETIME,
    RES_TEXT D_STR_255)
AS
declare variable pci_is_delete d_smallint;
begin
  -- RFCKG-3811 процедура отображает историю изменений для модуля
  for select pci.pci_from_cdm_id,
             c.c_name_fio,
             pci.pci_last_datetime,
             pci.pci_is_delete
      from product_condition_ignore pci
      inner join username un on un.un_id = pci.pci_from_un_id
      inner join clients c on c.c_id = un.un_from_c_id
      where pci.pci_from_pr_id = :pr_id and
            pci.pci_from_prl_id = :prl_id
      order by pci_last_datetime desc
      into cdm_id,
           c_name_fio,
           pci_last_datetime,
           pci_is_delete
  do
  begin
    res_text = '';
    if (pci_is_delete = 0) then
      res_text = f_format('Пользователь добавил исключение Блок %1 для продукта', cdm_id);
    else
    if (pci_is_delete = 1) then
      res_text = f_format('Пользователь убрал исключение Блок %1 для продукта', cdm_id);

    suspend;
  end
end
