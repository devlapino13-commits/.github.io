CREATE PROCEDURE UXP$UXFRAMESQL_PARAMS(
    UFR_UUID D_UUID,
    UFR_UUID_CODE D_STR_CODE = null)
RETURNS (
    UFS_UFR_UUID D_UUID,
    UFS_SQL_KIND D_INTEGER,
    UFS_UF_UUID D_UUID,
    UFS_UFP_UUID D_UUID,
    UF_POSITION D_INTEGER,
    UF_FIELD_NAME D_STR_CODE,
    UF_NAME D_STR_100,
    UFS_UFP_CODE D_STR_CODE,
    UFS_DEFAULT D_STR_100,
    UFS_CAN_EDIT_DEFAULT D_BOOL)
AS
declare variable ufd_us_uuid d_uuid;
declare variable us_is_table d_bool;
declare variable uf_is_param d_bool;
declare variable uf_pk_field d_smallint;
declare variable ufr_us_uuid d_uuid;
declare variable main_us_uuid d_uuid;
declare variable ufr_add_table_fields d_str_code;
declare variable uf_readonly d_boolean;
begin
  if (coalesce(ufr_uuid, '') = '') then
    ufr_uuid = uxf$uniq(ufr_uuid_code);

  ufs_ufr_uuid = ufr_uuid;

  for select ufd_kind, ufd_us_uuid, us_is_table, ufr_us_uuid, ufr_add_table_fields
  from ux$frdata, ux$frame, ux$source
  where ufr_uuid = :ufr_uuid and
        ufd_ufr_uuid = ufr_uuid and
        ufd_us_uuid = us_uuid
  order by ufd_kind
  into :ufs_sql_kind, :ufd_us_uuid, :us_is_table, :ufr_us_uuid, ufr_add_table_fields
  do
  begin
    main_us_uuid = null;
    if (ufr_add_table_fields <> '') then
    begin
      select us_uuid
      from ux$source
      where us_code = :ufr_add_table_fields and
            coalesce(us_keyfields, '') <> ''
      into main_us_uuid;
    end

    if (main_us_uuid is null) then
      main_us_uuid = ufr_us_uuid;

    for select uf_uuid, uf_position, uf_field_name, uf_name, uf_is_param, uf_pk_field, uf_readonly
    from ux$field
    where uf_owner = :ufd_us_uuid and
          uf_type = 0
    order by uf_is_param desc, uf_position
    into ufs_uf_uuid, uf_position, uf_field_name, uf_name, uf_is_param, uf_pk_field, uf_readonly
    do
    begin
      ufs_can_edit_default = 0;
      if (uf_is_param = 1 or (ufs_sql_kind = 2 and us_is_table = 1)) then
        ufs_can_edit_default = 1;

      if (us_is_table = 0 and uf_is_param = 0) then
        Continue;

      if (us_is_table = 1 and uf_pk_field = 0 and ufs_sql_kind = 3) then
        Continue;

      if (us_is_table = 1 and uf_readonly) then
        Continue;

      ufs_ufp_uuid = null;
      ufs_ufp_code = null;
      if (uf_field_name = uf_name) then
        uf_name = '';

      ufs_default = '';
      select ufs_ufp_uuid, ufs_default
      from ux$framesql
      where ufs_ufr_uuid = :ufr_uuid and
            ufs_sql_kind = :ufs_sql_kind and
            ufs_uf_uuid = :ufs_uf_uuid
      into ufs_ufp_uuid, ufs_default;

      if (row_count = 0 and ufd_us_uuid in (ufr_us_uuid, main_us_uuid)) then
      begin
        select ufp_uuid
        from ux$frparam
        where ufp_ufr_uuid = :ufr_uuid and
              ufp_uf_uuid = :ufs_uf_uuid
        into ufs_ufp_uuid;
      end

      if (ufs_ufp_uuid is not null) then
        ufs_ufp_code = uuid_to_char(ufs_ufp_uuid);

      suspend;
    end
  end
end
