CREATE PROCEDURE AAREP_CEO_PAR_REPORT_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_C_ID DECIMAL(18,0),
    REP_DAYS_IN_PERIOD INTEGER)
RETURNS (
    REG_ID INTEGER,
    LO_COUNT INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    REP_DATE DATE,
    REP_BALANCE_ALL DECIMAL(18,4),
    REP_BALANCE_7 DECIMAL(18,4),
    REP_BALANCE_7_010114 DECIMAL(18,4),
    REP_BALANCE_30 DECIMAL(18,4),
    REP_KOLICH_7 INTEGER,
    REP_KOLICH_7_010114 INTEGER,
    REP_KOLICH_30 INTEGER,
    LO_ID DECIMAL(18,0),
    REP_DISB_AMOUNT DECIMAL(18,4),
    REP_DISB_CLIENT INTEGER,
    REP_LINE_NO INTEGER,
    REP_BALANCE_1 DECIMAL(18,4),
    REP_KOLICH_1 INTEGER,
    REP_KOLICH_ALL INTEGER,
    REP_BALANCE_8_180 CURR_TYPE,
    REP_BALANCE_31_180 CURR_TYPE,
    REP_KOLICH_CLI INTEGER,
    REP_DISB_LGOT_AMOUNT CURR_TYPE,
    REP_DISB_IND_AMOUNT CURR_TYPE,
    REP_BALANCE_LGOT CURR_TYPE,
    REP_BALANCE_IND CURR_TYPE,
    REP_BALANCE_180_ CURR_TYPE,
    REP_KOLICH_180_ INTEGER,
    REP_KOLICH_CLI_180_ INTEGER,
    REP_KOLICH_IND INTEGER,
    REP_TRANSFER_CLI INTEGER,
    REP_TRANSFER_OFFICE_CLI INTEGER,
    REP_DISB_LGOT_KOLICH INTEGER,
    REP_DISB_IND_KOLICH INTEGER,
    REP_KOLICH_LGOT INTEGER,
    K_ID DECIMAL(18,0),
    PARENT INTEGER,
    PARENT2 INTEGER,
    REP_KOLICH_CLI_EX_TRANSFER INTEGER)
AS
declare variable delinq_date date;
declare variable i integer;
declare variable db_id integer;
declare variable category_id smallint;
declare variable def_curr varchar(3);
declare variable k_disb_date date;
declare variable rep_balance_all_180 curr_type;
declare variable rep_kolich_all_180 integer;
declare variable k_from_kf_id curr_type;
declare variable kal_date date;
declare variable kal_new_reg_id integer;
declare variable reg_id_last integer;
declare variable pr_id integer;
declare variable kk_curr_rate decimal(18,4);
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate decimal(18,4);
declare variable pr_id_prior integer;
declare variable pr_set_rate_in_kont integer;
declare variable parent_prior integer;
declare variable transfer_from_jan integer;
declare variable rep_kolich_cli_tmp integer;
declare variable reg_id_tmp integer;
begin
  parent_prior = 0;

  select cm_default_currency
  from company
  into def_curr;
  :db_id = f_get_db_id();

  lo_count = 10;
  if ((rep_days_in_period is null) or (rep_days_in_period = 0)) then
    rep_days_in_period = 1;

  if (not exists(select * from loanmgrlist
      where lml_from_c_id = :rep_c_id
      and :rep_date_end between lml_date_begin and lml_date_end)) then rep_c_id = 0;

-- Если регион, то на уровне офисов
  for select reg_code, reg_code || ' ' || reg_name, reg_unicode, parent
      from region reg, region_list(:rep_region_id, 1, 3, 1)
      where reg.reg_unicode = id_list and
            coalesce(reg.reg_date_close, cast('31.12.9999' as date)) >= :rep_date_end - 6 and
            reg.reg_is_agent = 0
      union
      select reg.reg_code, reg_code || ' ' || reg_name, reg.reg_unicode, parent
      from region reg
      where reg.reg_unicode = 1120300 and
            :db_id in (39801, 41701) -- Time is Money
      into :reg_code, :reg_name, :reg_id_tmp, :parent
  do
  begin
    reg_id = reg_id_tmp;
    if (parent_prior <> parent) then
    begin
      parent_prior = parent;
      select parent
      from region
      where reg_unicode = :parent
      into parent2;
    end

    if (:rep_region_id = 1000000) then
    begin
      select reg_unicode, reg_code, reg_name
      from region
      where reg_unicode = :parent
      into :reg_id, :reg_code, :reg_name;
    end

    rep_balance_all = 0;
    rep_balance_1 = 0;
    rep_kolich_1 = 0;
    rep_kolich_all = 0;
    rep_balance_7 = 0;
    rep_balance_7_010114 = 0;
    rep_kolich_7 = 0;
    rep_kolich_7_010114 = 0;
    rep_balance_30 = 0;
    rep_kolich_30 = 0;
    rep_balance_8_180 = 0;
    rep_balance_31_180 = 0;
    rep_kolich_cli = 0;
    rep_transfer_cli = 0;
    rep_transfer_office_cli = 0;

    rep_disb_amount = 0;
    rep_disb_client = 0;

    rep_disb_lgot_amount = 0;
    rep_disb_ind_amount = 0;
    rep_disb_lgot_kolich = 0;
    rep_disb_ind_kolich = 0;

    rep_balance_lgot = 0;
    rep_balance_ind = 0;
    rep_kolich_ind = 0;

    rep_balance_180_ = 0;
    rep_kolich_180_ = 0;
    rep_kolich_cli_180_ = 0;

    rep_balance_all_180 = 0;
    rep_kolich_all_180 = 0;

    transfer_from_jan = 0;
    rep_kolich_cli_ex_transfer = 0;

    for select distinct oks_kc_id
        from oksummary
        where ((oks_region_id = :reg_id_tmp and :rep_c_id = 0) or
                (:rep_c_id <> 0 and :rep_c_id = f_lo_mgr_c_id(f_lo_mgr_id(oks_from_k_id, :rep_date_end, oks_kc_id,oks_region_id)))) and
              oks_date_begin + 0 <= :rep_date_end and
              oks_date_end >= :rep_date_end - 6 * :rep_days_in_period and
              oks_balance > 0 and
              oks_close_state = 0
        into :lo_id
    do
    begin
      i = 1;
      while (i <= 7) do
      begin
        rep_line_no = i;
        rep_date = rep_date_end - (i - 1) * :rep_days_in_period;
        suspend;
        i = i + 1;
      end
    end

    i = 1;
    while (i <= 7) do
    begin
      rep_line_no = i;
      rep_date = rep_date_end - (i - 1) * :rep_days_in_period;
      rep_balance_all = 0;
      rep_kolich_all = 0;
      rep_balance_1 = 0;
      rep_kolich_1 = 0;
      rep_balance_7 = 0;
      rep_balance_7_010114 = 0;
      rep_kolich_7 = 0;
      rep_kolich_7_010114 = 0;
      rep_balance_30 = 0;
      rep_kolich_30 = 0;
      rep_balance_8_180 = 0;
      rep_balance_31_180 = 0;
      rep_kolich_cli = 0;
      rep_transfer_cli = 0;
      rep_transfer_office_cli = 0;

      rep_disb_amount = 0;
      rep_disb_client = 0;

      rep_disb_lgot_amount = 0;
      rep_disb_ind_amount = 0;
      rep_disb_lgot_kolich = 0;
      rep_disb_ind_kolich = 0;

      rep_balance_lgot = 0;
      rep_balance_ind = 0;
      rep_kolich_ind = 0;

      rep_balance_180_ = 0;
      rep_kolich_180_ = 0;
      rep_kolich_cli_180_ = 0;

      rep_balance_all_180 = 0;
      rep_kolich_all_180 = 0;

      pr_id_prior = 0;

      transfer_from_jan = 0;
      rep_kolich_cli_ex_transfer = 0;

      for select oks_from_k_id, oks_kc_id, oks_balance, oks_delinq_date, k_kolich_client_approved, k_disb_date,
                 iif(pr_product_kind = 2, oks_balance, 0), iif(pr_product_kind = 2, k_kolich_client_approved, 0),
                 iif(pr_loan_type = 1 and (pr_common_name not like '%недел%' or (:db_id <> 81001)), oks_balance, 0),
                 iif(pr_loan_type = 1 and (pr_common_name not like '%недел%' or (:db_id <> 81001)), k_kolich_client_approved, 0),
                 k_from_kf_id, k_index_curr_code, k_index_curr_rate, pr_id
          from oksummary, kontract, product
          where ((oks_region_id = :reg_id_tmp and :rep_c_id = 0)
                or (:rep_c_id <> 0 and :rep_c_id = f_lo_mgr_c_id(f_lo_mgr_id(oks_from_k_id, :rep_date_end, oks_kc_id,oks_region_id)))) and
                oks_date_begin + 0 <= :rep_date and
                oks_date_end >= :rep_date and
                oks_balance > 0 and
                oks_close_state = 0 and
                oks_from_k_id = k_id and
                k_from_pr_id = pr_id
          into :k_id, :lo_id, :rep_balance_all, :delinq_date, :rep_kolich_all, :k_disb_date, rep_balance_lgot,
               rep_kolich_lgot, rep_balance_ind, rep_kolich_ind, k_from_kf_id, k_index_curr_code, k_index_curr_rate,
               pr_id
      do
      begin
        category_id = (select aa_category_id
                       from aarep_get_category_id(:k_id, :rep_date, :def_curr, null, null, null, null, null));
        if (rep_date - delinq_date > 360) then
          category_id = 3;

        rep_kolich_cli_tmp = 0;
        rep_kolich_cli = 0;
        rep_kolich_cli_180_ = 0;

        select res_just_client_count
        from hp_get_just_client_in_loan(:rep_date, :k_id, :k_disb_date)
        into rep_kolich_cli_tmp;

        if ((:db_id <> 81001) or (category_id in (1, 2))) then
          rep_kolich_cli = rep_kolich_cli_tmp;
        if (rep_date - delinq_date > 360) then
          rep_kolich_cli_180_ = rep_kolich_cli_tmp;

        kk_curr_rate = 1;
        if (k_index_curr_code <> def_curr) then
        begin
          select cr.cr_rate
          from opercurr cr
          where cr_date <= :rep_date and
                cr_date_end >= :rep_date and
                cr_from_cu_id = :k_index_curr_code
          into :kk_curr_rate;

          if (kk_curr_rate < k_index_curr_rate) then
          begin
            if (pr_id <> pr_id_prior) then
            begin
              select pr.pr_set_rate_in_kont
              from product pr
              where pr_id = :pr_id
              into :pr_set_rate_in_kont;

              pr_id_prior = pr_id;
            end
            if (pr_set_rate_in_kont = 1) then
              kk_curr_rate = :k_index_curr_rate;
          end
        end

        rep_balance_all = rep_balance_all * kk_curr_rate;
        rep_balance_lgot = rep_balance_lgot * kk_curr_rate;
        rep_balance_ind = rep_balance_ind * kk_curr_rate;

        rep_balance_all_180 = rep_balance_all;
        rep_kolich_all_180 = rep_kolich_all;

        if ((:db_id <> 81001) or (category_id in (1, 2))) then
        begin
          rep_balance_1 = 0;
          rep_kolich_1 = 0;
          rep_balance_7 = 0;
          rep_balance_7_010114 = 0;
          rep_kolich_7 = 0;
          rep_kolich_7_010114 = 0;
          rep_balance_30 = 0;
          rep_kolich_30 = 0;
          rep_balance_8_180 = 0;
          rep_balance_31_180 = 0;
--rep_kolich_cli = 0;
          rep_transfer_cli = 0;
          rep_transfer_office_cli = 0;

          rep_balance_180_ = 0;
          rep_kolich_180_ = 0;
--rep_kolich_cli_180_ = 0;

          if (rep_date - delinq_date > 0) then
          begin
            rep_balance_1 = rep_balance_all;
            rep_kolich_1 = 1;
          end
          if (rep_date - delinq_date > 7) then
          begin
            rep_balance_7 = rep_balance_all;
            rep_kolich_7 = 1;
          end

          if (rep_date - delinq_date > 7 and
              delinq_date >= '01.01.14') then
          begin
            rep_balance_7_010114 = rep_balance_all;
            rep_kolich_7_010114 = 1;
          end

          if (rep_date - delinq_date > 30) then
          begin
            rep_balance_30 = rep_balance_all;
            rep_kolich_30 = 1;
          end
          if (category_id = 1 and
              (rep_date - delinq_date) between 8 and 360) then
          begin
            rep_balance_8_180 = rep_balance_all;
            if (rep_date - delinq_date > 30) then
              rep_balance_31_180 = rep_balance_all;
          end

--        Добавлено от 23.10.2014, расчет кол-во клиентов
          /*
          select RES_JUST_CLIENT_COUNT
          from HP_GET_JUST_CLIENT_IN_LOAN(:rep_date, :k_id, :k_disb_date)
          into rep_kolich_cli;
          */
          rep_kolich_cli_ex_transfer = rep_kolich_cli - coalesce(transfer_from_jan, 0); -- igusev 14.04.2015

          select kal_date, kal_new_reg_id
          from knaktlist
          where knaktlist.is_delete = 0 and
                :rep_date between kal_date and kal_date_end and
                kal_from_kf_id = :k_from_kf_id
          into kal_date, kal_new_reg_id;

          if (kal_date between(:rep_date + 1 - :rep_days_in_period) and :rep_date -- была ли создана строка в knaktlist'е за указанный пеирод
              and
              (:k_disb_date < (:rep_date + 1 - :rep_days_in_period) -- и была ли выдана до текущего периода т.е. передача папки
              or :k_disb_date < kal_date)) then
          begin
            rep_transfer_cli = rep_kolich_cli;
--          проверяет офис до передачи, если офис другой то будет будет передача между офисами
            reg_id_last = null;
            select kal_new_reg_id
            from knaktlist
            where knaktlist.is_delete = 0 and
                  :kal_date - 1 between kal_date and kal_date_end and
                  kal_from_kf_id = :k_from_kf_id
            into reg_id_last;
--            if (reg_id_last is null) then reg_id_last = 0;

            if (kal_new_reg_id <> reg_id_last) then
              rep_transfer_office_cli = rep_kolich_cli;
          end
          else
          begin
            rep_transfer_cli = 0;
            rep_transfer_office_cli = 0;
          end

          suspend;
        end
--      Показатели по просроченным >360 дней
        if (rep_date - delinq_date > 360) then
        begin
          rep_balance_all = 0;
          rep_kolich_all = 0;

          rep_balance_1 = 0;
          rep_kolich_1 = 0;
          rep_balance_7 = 0;
          rep_balance_7_010114 = 0;
          rep_kolich_7 = 0;
          rep_kolich_7_010114 = 0;
          rep_balance_30 = 0;
          rep_kolich_30 = 0;
          rep_balance_8_180 = 0;
          rep_balance_31_180 = 0;
          rep_kolich_cli = 0;
          rep_transfer_cli = 0;
          rep_transfer_office_cli = 0;

          rep_kolich_cli_ex_transfer = 0;

          rep_balance_180_ = rep_balance_all_180;
          rep_kolich_180_ = rep_kolich_all_180;

          /*
          select RES_JUST_CLIENT_COUNT
          from HP_GET_JUST_CLIENT_IN_LOAN(:rep_date, :k_id, :k_disb_date)
          into REP_KOLICH_CLI_180_;
          */

          suspend;
        end
      end

      rep_balance_all = 0;
      rep_kolich_all = 0;
      rep_balance_1 = 0;
      rep_kolich_1 = 0;
      rep_balance_7 = 0;
      rep_balance_7_010114 = 0;
      rep_kolich_7 = 0;
      rep_kolich_7_010114 = 0;
      rep_balance_30 = 0;
      rep_kolich_30 = 0;
      rep_balance_8_180 = 0;
      rep_balance_31_180 = 0;
      rep_kolich_cli = 0;
      rep_transfer_cli = 0;
      rep_transfer_office_cli = 0;

      rep_disb_amount = 0;
      rep_disb_client = 0;

      rep_disb_lgot_amount = 0;
      rep_disb_ind_amount = 0;
      rep_disb_lgot_kolich = 0;
      rep_disb_ind_kolich = 0;

      rep_balance_lgot = 0;
      rep_balance_ind = 0;
      rep_kolich_ind = 0;

      rep_balance_180_ = 0;
      rep_kolich_180_ = 0;
      rep_kolich_cli_180_ = 0;

      rep_balance_all_180 = 0;
      rep_kolich_all_180 = 0;

      rep_kolich_cli_ex_transfer = 0;

      for select ks_kc_id, sum(k_amount_approved), sum(k_kolich_client_approved),
                 sum(iif(pr_product_kind = 2, k_amount_approved, 0)),
                 sum(iif(pr_product_kind = 2, k_kolich_client_approved, 0)),
                 sum(iif(pr_loan_type = 1 and (pr_common_name not like '%недел%' or (:db_id <> 81001)), k_amount_approved, 0)),
                 sum(iif(pr_loan_type = 1 and (pr_common_name not like '%недел%' or (:db_id <> 81001)), k_kolich_client_approved, 0))
          from kontstate, kontract, product
          where ks_from_k_id = k_id and
                ks_date_begin between(:rep_date - :rep_days_in_period + 1) and :rep_date and
                ks_state = 1 and
                ks_ol_is_delete = 0 and
                ((ks_region_id = :reg_id_tmp and :rep_c_id = 0)
                  or (:rep_c_id <> 0 and :rep_c_id = f_lo_mgr_c_id(f_lo_mgr_id(ks_from_k_id, :rep_date_end, ks_kc_id, ks_region_id))))

                and
                k_from_pr_id = pr_id
          group by ks_kc_id
          into lo_id, rep_disb_amount, rep_disb_client, rep_disb_lgot_amount, rep_disb_lgot_kolich, rep_disb_ind_amount,
               rep_disb_ind_kolich
      do
      begin
        suspend;
      end
      i = i + 1;
    end
  end
end
