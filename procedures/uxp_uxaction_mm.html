CREATE PROCEDURE UXP$UXACTION_MM(
    UX_UUID D_UUID)
RETURNS (
    UA_UUID D_UUID,
    UA_OWNER D_UUID,
    UA_UXA_UUID D_UUID,
    UA_USR_ID_FIELD D_STR_CODE,
    UA_KIND D_SMALLINT,
    UA_NAME D_STR_100,
    UA_HINT D_STR_255,
    UA_DISABLE_ON_EDIT D_BOOL,
    UA_READONLY D_BOOL,
    UA_SAFE_MODE D_BOOL,
    UA_INACTIVE D_BOOL,
    RES_UX_KIND D_INTEGER,
    RES_CAPTION D_STR_255,
    RES_OWNER_CODE D_STR_100,
    RES_UXA_CODE D_STR_CODE,
    UA_UUID_STR D_STR_CODE,
    RES_LINKED D_BOOL,
    RES_IMAGE D_INTEGER,
    UA_IMG_UUID D_UUID,
    UA_IMG_UUID_CODE D_STR_CODE,
    ML_CODE D_STR_CODE,
    ML_PATH D_STR_LONG,
    ML_FS_DISABLED D_SMALLINT,
    ML_MENU_STATE D_STR_255,
    ML_ID D_ID)
AS
declare variable tmp_name d_str_100;
declare variable ux_kind d_str_code;
declare variable mm_uuid d_uuid;
declare variable uml_web_disabled d_smallint;
begin
  -- Список форм для прикрепления к главному меню приложения

  mm_uuid = uxf$mainmenu();

  if (uxf$code(ux_uuid, 'kind') = 'uxalias') then
  begin
    select ua_uuid
    from ux$action
    where ua_owner = :mm_uuid and
          ua_uxa_uuid = :ux_uuid
    into ua_uuid;
  end else
    ua_uuid = ux_uuid;

  -- Список действий объекта
  for select ua_owner,
             ua_uxa_uuid,
             ua_usr_id_field,
             uxa_kind,
             ua_name,
             ua_hint,
             ua_disable_on_edit,
             ua_readonly,
             ua_inactive,
             ua_img_uuid,
             uuid_to_char(ua_img_uuid),
             uu_kind,
             uxf$kind(uu_kind, 'code'),
             ux_fcode,
             uxa_fcode,
             uxa_name
      from ux$action, ux$alias, ux$code, ux$uniq
      where ua_uuid = :ua_uuid and
            uxa_uuid = ua_uxa_uuid and
            uxa_owner = ux_uuid and
            ux_uuid = uu_id
      into :ua_owner,
           :ua_uxa_uuid,
           :ua_usr_id_field,
           :ua_kind,
           :ua_name,
           :ua_hint,
           :ua_disable_on_edit,
           :ua_readonly,
           :ua_inactive,
           :ua_img_uuid,
           :ua_img_uuid_code,
           :res_ux_kind,
           :ux_kind,
           :res_owner_code,
           :res_uxa_code,
           :tmp_name
  do
  begin
    res_linked = 0;
    ua_safe_mode = 0;
    ml_id = null;
    ml_code = '';
    ml_path = '';
    ml_fs_disabled = -1;
    uml_web_disabled = -1;

    ml_menu_state = 'Меню не прикреплено';

    select item_id, item_code, item_fs_disabled, item_web_disabled
    from uxp$actionmenu(:ua_uuid)
    into :ml_id, :ml_code, :ml_fs_disabled, :uml_web_disabled;

    if (ml_id is not null) then
    begin
      if (ml_fs_disabled = 1 and uml_web_disabled = 1) then
        ml_menu_state = 'Отключено в FinStart и Web';
      else
      if (ml_fs_disabled = 1 and uml_web_disabled = 0) then
        ml_menu_state = 'FinStart - отключено, Web - доступно';
      else
      if (ml_fs_disabled = 0 and uml_web_disabled = 1) then
        ml_menu_state = 'FinStart - доступно, Web - отключено';
      else
        ml_menu_state = 'Доступно в FinStart и Web';

      res_linked = 1;
      ml_path = uxf$menupath(ml_id);
    end

    res_image = 19;
    res_image = case ux_kind
      when 'uxform' then 19
      when 'uxcommand' then 35
      when 'uxreport' then 26
    end;

    if (exists(select *
                from  ux$search
                where usr_uuid = :ua_uxa_uuid)) then
      res_image = 4;

    res_caption = tmp_name;
    if (tmp_name not containing res_uxa_code) then
      res_caption = res_caption || '   - ' || res_uxa_code;

    ua_uuid_str = lower(uuid_to_char(ua_uuid));
    suspend;
  end
end
