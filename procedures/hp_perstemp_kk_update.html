CREATE PROCEDURE HP_PERSTEMP_KK_UPDATE(
    PTE_EDIT_KIND D_SMALLINT,
    PTE_DATA_STATE D_SMALLINT,
    PTE_FROM_C_ID D_ID,
    PTE_NAME_FIO D_STR_100,
    PTE_ALTERNAT_FIO D_STR_100,
    PTE_NAME_ENG D_STR_100,
    PTE_SEX D_SMALLINT,
    PTE_BIRTH_DATE D_DATE,
    PTE_CITIZEN D_INTEGER,
    PTE_INN D_STR_20,
    PTE_EXT_INN D_STR_20,
    PTE_PINCODE D_STR_20,
    PTE_SNILS D_STR_20,
    PTE_DOC_TYPE D_INTEGER,
    PTE_PASS_NO D_STR_20,
    PTE_PASS_DATE D_DATE,
    PTE_PASS_DATE_EXPIRE D_DATE,
    PTE_PASS_UNLIMIT D_BOOL,
    PTE_PASS_WHO D_STR_255,
    PTE_PASS_WHO_CODE D_STR_20,
    PTE_PASS_DRL_ID D_ID,
    PTE_BIRTH_COUNTRY D_STR_255,
    PTE_BIRTH_CITY D_STR_255,
    PTE_ADL_ID_1 D_ID,
    PTE_ADL_ID_2 D_ID,
    PTE_BLOB_PHOTO_ID D_ID,
    PTE_BLOB_PASS1_ID D_ID,
    PTE_BLOB_PASS2_ID D_ID,
    DIFF_NAME_FIO D_BOOL,
    DIFF_ALTERNAT_FIO D_BOOL,
    DIFF_NAME_ENG D_BOOL,
    DIFF_SEX D_BOOL,
    DIFF_BIRTH_DATE D_BOOL,
    DIFF_CITIZEN D_BOOL,
    DIFF_INN D_BOOL,
    DIFF_EXT_INN D_BOOL,
    DIFF_PINCODE D_BOOL,
    DIFF_SNILS D_BOOL,
    DIFF_DOC_TYPE D_BOOL,
    DIFF_PASS_NO D_BOOL,
    DIFF_PASS_DATE D_BOOL,
    DIFF_PASS_DATE_EXPIRE D_BOOL,
    DIFF_PASS_WHO D_BOOL,
    DIFF_PASS_WHO_CODE D_BOOL,
    DIFF_PASS_DRL_ID D_BOOL,
    DIFF_BIRTH_COUNTRY D_BOOL,
    DIFF_BIRTH_CITY D_BOOL,
    DIFF_ADL_ID_1 D_BOOL,
    DIFF_ADL_ID_2 D_BOOL,
    DIFF_BLOB_PHOTO_ID D_BOOL,
    DIFF_BLOB_PASS1_ID D_BOOL,
    DIFF_BLOB_PASS2_ID D_BOOL,
    RES_ACL_ID D_ID,
    RES_ACL_MEMO D_STR_255,
    PTE_S D_STR_100,
    PTE_F D_STR_100,
    PTE_M D_STR_100,
    PTE_S_ENG D_STR_100,
    PTE_F_ENG D_STR_100,
    PTE_M_ENG D_STR_100,
    F001 D_INTEGER,
    F002 D_INTEGER,
    UN_ID D_ID = null)
AS
declare variable approved_usr_id d_id;
declare variable edited_usr_id d_id;
declare variable usr_id d_id;
declare variable cs_staff_state d_smallint;
declare variable cs_client_state d_smallint;
declare variable rd_code d_str_255;
declare variable q_inn d_str_20;
declare variable c_inn d_str_20;
declare variable q_inn_received d_bool;
declare variable fl_id d_id;
declare variable fl_id_eng d_id;
declare variable is_correct boolean;
declare variable err_txt d_str_8100;
begin
  if (f_get_db_id() = 81001) then
  begin
    -- RFCRU-5644
    select
      is_correct,
      err_txt
    from hp_separate_fio_checkvalid(:pte_from_c_id, :pte_f, :pte_m, :pte_s, :pte_f_eng, :pte_m_eng, :pte_s_eng, :pte_alternat_fio)
    into
      :is_correct,
      :err_txt;
    if (not is_correct) then
      exception e_message :err_txt;

    select
      coalesce(inn, ''),
      inn_received
    from inn_query_status(:pte_from_c_id)
    into
      :q_inn,
      :q_inn_received;
    
    rd_code = '';
    select
      f_get_rd_code(acl.acl_from_rd_id)
    from actionlist acl
      inner join perstemp pte on
        acl.acl_id = pte.pte_acl_id
    where pte.pte_from_c_id = :pte_from_c_id
    rows 1
    into
      rd_code;
    
    c_inn = coalesce(pte_inn, '');
    if (q_inn_received = 1 and q_inn <> c_inn
      and rd_code starting 'ActionGroup.Passport_Verify')
    then
      exception e_message f_format('Сохранить не удалось: ИНН, полученный от ФНС (%1), не совпадает с ИНН, который вы пытаетесь сохранить (%2). Вероятно, ИНН от ФНС был получен во время верификации.', :q_inn, c_inn);
  end

  if (pte_pass_unlimit = 1) then
    pte_pass_date_expire = '31.12.9999';
  else
  if (pte_pass_date_expire is null) then
    exception e_message 'Срок действия не указан';

  if (pte_edit_kind is null) then
  begin
    if (pte_data_state > 0) then
      exception e_message 'Не указана причина изменений';
    else
      pte_edit_kind = 0;
  end

  if (f001 > 0 and coalesce(f002, 0) = 0) then
    exception e_message 'Есть документ регистрации, но не указан код регистрации'; -- RFCRU-5407

  if (coalesce(f001, 0) = 0 and f002 > 0) then
    exception e_message 'Указан код регистрации, но сам документ регистрации отсутствует'; -- RFCRU-5407

  if (coalesce(pte_inn, '') <> '') then
    execute procedure hp_check_ru_inn(:pte_inn, 0);

  if (un_id is null) then
    un_id = uxf$user();

  if (not exists(select * from perstemp
                 where pte_from_c_id = :pte_from_c_id)) then
  begin
    insert into perstemp (pte_from_c_id, pte_acl_id)
    values (:pte_from_c_id, :res_acl_id);
  end

  approved_usr_id = null;
  edited_usr_id = null;

  usr_id = uxf$usr_id();
  if (coalesce(usr_id, 0) = 0 and un_id is not null) then
    usr_id = uxf$usr_by('username', (select un_name from username where un_id = :un_id));

  if (pte_data_state in (-2, -1, 2)) then
    approved_usr_id = usr_id;
  else
    edited_usr_id = usr_id;

  if (pte_data_state = 2) then
  begin
    if (not exists(select *
                   from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.CanEditTerrAndEx')
                   where has_access = 1)) then
    begin
      if (exists(select * from aarep_check_finpolis_one(:pte_from_c_id))) then
      begin
        exception e_message 'ВНИМАНИЕ,' ||
                            'Необходимо связаться с комплайнс менеджером ГО.' ||
                            'Дальнейшие операции с этим контрагентом запрещены.' ||
                            'Доступен только отказ в верификации.' ||
                            'Детали в меню "Проверка по спискам ТиЭ"';
      end
    end
  end

  select cs_staff_state, cs_client_state
  from hp_perstemp_state_info(:pte_from_c_id, :un_id)
  into :cs_staff_state, :cs_client_state;

  if ((res_acl_id is not null) and (:pte_edit_kind is not null)) then
  begin
      update or insert into actioneditkind (aek_from_acl_id, aek_editkind)
      values (:res_acl_id, :pte_edit_kind)
      matching (aek_from_acl_id);
  end

  update perstemp set
    pte_edit_kind = :pte_edit_kind,
    pte_data_state = :pte_data_state,
    pte_staff_state = :cs_staff_state,
    pte_client_state = :cs_client_state,
    pte_kk_editing = 1,
    pte_name_fio = :pte_name_fio,
    pte_alternat_fio = :pte_alternat_fio,
    pte_name_eng = :pte_name_eng,
    pte_sex = :pte_sex,
    pte_birth_date = :pte_birth_date,
    pte_citizen = :pte_citizen,
    pte_inn = :pte_inn,
    pte_ext_inn = :pte_ext_inn,
    pte_pincode = :pte_pincode,
    pte_snils = :pte_snils,
    pte_doc_type = :pte_doc_type,
    pte_pass_no = :pte_pass_no,
    pte_pass_date = :pte_pass_date,
    pte_pass_date_expire = :pte_pass_date_expire,
    pte_pass_unlimit = :pte_pass_unlimit,
    pte_pass_who = :pte_pass_who,
    pte_pass_who_code = :pte_pass_who_code,
    pte_pass_drl_id = :pte_pass_drl_id,
    pte_blob_pass1_id = :pte_blob_pass1_id,
    pte_blob_pass2_id = :pte_blob_pass2_id,
    pte_blob_photo_id = :pte_blob_photo_id,
    pte_birth_country = :pte_birth_country,
    pte_birth_city = :pte_birth_city,
    pte_adl_id_1 = :pte_adl_id_1,
    pte_adl_id_2 = :pte_adl_id_2,
    pte_memo = :res_acl_memo,
    pte_usr_approved = coalesce(:approved_usr_id, pte_usr_approved),
    pte_usr_edited = coalesce(:edited_usr_id, pte_usr_edited),
    diff_name_fio = :diff_name_fio,
    diff_alternat_fio = :diff_alternat_fio,
    diff_name_eng = :diff_name_eng,
    diff_sex = :diff_sex,
    diff_birth_date = :diff_birth_date,
    diff_citizen = :diff_citizen,
    diff_inn = :diff_inn,
    diff_ext_inn = :diff_ext_inn,
    diff_pincode = :diff_pincode,
    diff_snils = :diff_snils,
    diff_doc_type = :diff_doc_type,
    diff_pass_no = :diff_pass_no,
    diff_pass_date = :diff_pass_date,
    diff_pass_date_expire = :diff_pass_date_expire,
    diff_pass_who = :diff_pass_who,
    diff_pass_who_code = :diff_pass_who_code,
    diff_pass_drl_id = :diff_pass_drl_id,
    diff_birth_country = :diff_birth_country,
    diff_birth_city = :diff_birth_city,
    diff_adl_id_1 = :diff_adl_id_1,
    diff_adl_id_2 = :diff_adl_id_2,
    diff_blob_photo_id = :diff_blob_photo_id,
    diff_blob_pass1_id = :diff_blob_pass1_id,
    diff_blob_pass2_id = :diff_blob_pass2_id,
    pte_sname = :pte_s,
    pte_fname = :pte_f,
    pte_mname = :pte_m,
    pte_sname_eng = :pte_s_eng,
    pte_fname_eng = :pte_f_eng,
    pte_mname_eng = :pte_m_eng
  where (pte_from_c_id = :pte_from_c_id);

  select
    fl_id,
    fl_id_eng
  from hp_prepare_fio(
    :pte_s,
    :pte_f,
    :pte_m,
    :pte_s_eng,
    :pte_f_eng,
    :pte_m_eng)
  into
    :fl_id,
    :fl_id_eng;
  update clients set
    c_from_fl_id = :fl_id,
    c_name_eng_fl_id = :fl_id_eng
  where c_id = :pte_from_c_id;

  if (f001 > 0 and f002 > 0) then -- f001 - документ регистрации, f002 - код адреса регистрации (RFCRU-5407)
  begin
     insert into addresstype_registration (atr_from_cds_id, atr_from_rd_id)
     values (:f001, :f002);
  end
end
