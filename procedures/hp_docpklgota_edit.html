CREATE PROCEDURE HP_DOCPKLGOTA_EDIT(
    PK_ID D_INTEGER,
    PK_CITIZEN_KIND D_INTEGER,
    PK_DATE_BEGIN D_DATE,
    PK_DATE_END D_DATE,
    PK_IS_DELETE D_BOOL,
    UN_ID D_ID)
AS
begin
  if (pk_is_delete = 0) then
  begin
    if (pk_date_begin is null) then
      pk_date_begin = current_date;

    if (pk_date_end is null) then
      pk_date_end = maxvalue(pk_date_begin, current_date);

    if (not exists(select * from docpklgota
    where dpg_from_pk_id = :pk_id and
          dpg_citizen_kind = :pk_citizen_kind and
          dpg_date_begin = :pk_date_begin and
          dpg_date_end = :pk_date_end and
          dpg_is_delete = 0)) then
    begin
      update docpklgota
      set dpg_date_begin = :pk_date_begin,
          dpg_date_end = :pk_date_end
      where dpg_from_pk_id = :pk_id and
            dpg_citizen_kind = :pk_citizen_kind and
            dpg_is_delete = 0;

      if (row_count = 0) then
      begin
        insert into docpklgota (dpg_from_pk_id, dpg_citizen_kind, dpg_date_begin, dpg_date_end, dpg_create_un_id)
        values (:pk_id, :pk_citizen_kind, :pk_date_begin, :pk_date_end, :un_id);
      end
    end
  end else
  begin
    update docpklgota
    set dpg_is_delete = 1,
        dpg_delete_un_id = :un_id
    where dpg_from_pk_id = :pk_id and
          dpg_citizen_kind = :pk_citizen_kind and
          dpg_is_delete = 0;
  end
end
