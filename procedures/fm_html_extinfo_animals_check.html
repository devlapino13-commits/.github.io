CREATE PROCEDURE FM_HTML_EXTINFO_ANIMALS_CHECK(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable lang d_str_lang;
declare variable eil_date d_date;
declare variable bls_id d_id;
declare variable c_name_fio d_str_255;
declare variable calc_status d_str_255;
declare variable json d_blob_text;
declare variable c_inn d_str_100;
declare variable obj_count d_integer;
declare variable animal_type d_integer;
declare variable temp_animal_type d_integer;
declare variable counter d_integer;
declare variable animal_count d_integer;
begin
  lang = f_get_lang_code(un_id, null);

  select eil_date, eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id and
        eil_from_eic_id = eic_id and
        eic_code = 'animals_check' and
        eil_is_json = 1 and
        eil_status = 2
  order by eil_date desc, eil_id
  rows 1
  into eil_date, bls_id;

  c_name_fio = f_lang(lang, 'Клиент не найден');
  if (row_count = 0) then
    calc_status = f_lang(lang, 'Информация о количестве скота не получена');
  else
  begin
    if (coalesce(bls_id, 0) <= 0) then
      calc_status = f_lang(lang, 'Информация о количестве скота по клиенту отсутствует.');
    else
    begin
      calc_status = f_format('<br>%1: %2', f_lang(lang, 'Дата проверки'), f_datetostr(eil_date));
    end
  end

  select c_name_fio, c_inn
  from clients
  where c_id = :c_id
  into c_name_fio, c_inn;
  res_html = (select res_style
              from fm_html_styles) || f_format('<body><a name="top"></a><h2>%1</h2>%2<br>', c_name_fio, calc_status);

  select res_blob
  from fm_blob_get(:bls_id)
  into :json;
  if (coalesce(:json, '') <> '' and
      :json not containing 'Данные не найдены') then
  begin
    res_html = blob_append(res_html, '<br><h1>' || f_lang(lang, 'Информация о наличии сельхоз.скота') || '<br></h1>');
    res_html = blob_append(res_html, f_format('<h3>%1</h3><table width="100%"><col width="40%"><col width="60%">', ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Ф.И.О. Заявителя') || '</b>', :c_name_fio, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'ПИН') || '</b>', :c_inn, '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, '</table>');
    res_html = blob_append(res_html, '<br><h1>' || f_lang(lang, 'Результат услуги') || '<br></h1>');
    res_html = blob_append(res_html, f_format('<h3>%1</h3><table width="100%"><col width="40%"><col width="60%">', ''));

    :json = '{"animal_list" :' || :json || '}';
    obj_count = f_json_blob_get_count(:json, 'animal_list');
    animal_type = 1;

    while (animal_type < 7) do
    begin
      counter = 0;
      animal_count = 0;
      while (counter < obj_count) do
      begin
        temp_animal_type = f_json_blob_get_value(:json, 'animal_list[' || counter || '].Type', 0);

        if (temp_animal_type = animal_type) then
          animal_count = animal_count + 1;
        counter = counter + 1;
      end
      res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, case animal_type
                                                                                when 1 then 'Крупно-рогатый скот и яки'
                                                                                when 2 then 'Лошади'
                                                                                when 3 then 'Мелко-рогатый скот (овцы и козы)'
                                                                                when 4 then 'Свиньи'
                                                                                when 5 then 'Домашние птицы'
                                                                                when 6 then 'Пчелы'
                                                                              end) || '</b>', :animal_count, '', '', -1, -1, -1, -1, 2, ''));
      animal_type = animal_type + 1;
    end
    res_html = blob_append(res_html, f_html_table_row('<b>' || f_lang(lang, 'Наименование органа, выдающего справку') || '</b>', 'Ветеринарная служба при МСХ КР', '', '', -1, -1, -1, -1, 2, ''));
    res_html = blob_append(res_html, '</table>');
  end
  else
  begin
    res_html = blob_append(res_html, '<br>' || f_lang(lang, 'Информация о клиенте не найдена') || '</br>');
    suspend;
    exit;
  end
  suspend;
end
