CREATE PROCEDURE AAREP_MINIAPP_DISB_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7),
    REP_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_REGION_ID INTEGER,
    MA_KF_ID DECIMAL(18,0),
    MA_DISB_KOLICH INTEGER,
    MA_FACT_DATE DATE,
    MA_FACT_DISB_AMOUNT DECIMAL(18,4),
    MA_PLAN_DISB_ALL DECIMAL(18,4),
    MA_APP_DATE DATE,
    MA_APP_NO INTEGER,
    MA_REGION_NAME VARCHAR(100),
    MA_DATE_SORT DATE,
    MA_KF_NAME VARCHAR(100),
    MA_KC_NAME VARCHAR(100),
    MA_PRODUCT VARCHAR(100),
    MA_PLAN_FACT DECIMAL(18,4),
    MA_CIRCLE INTEGER)
AS
declare variable ma_app_id integer;
declare variable ma_dd date;
declare variable ma_uniq_id integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select max(ma_date_oper), sum(ma_summa_minus), sum(ma_summa_minus_repeat), sum(ma_summa_minus_new),
             sum(ma_summa_miniappnew), ma_region_id, ma_kf_id, sum(ma_disb_kolich), max(ma_fact_date),
             sum(ma_fact_disb_amount), sum(ma_plan_disb_all), reg_code || ' ' || reg_name, max(ma_product),
             max(ma_circle), ma_uniq_id
      from aarep_miniapp_disb_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_auto_period, :rep_auto_koef, :rep_include_first_day), region
      where ma_region_id = reg_unicode
      group by ma_region_id, ma_kf_id, reg_code, reg_name, ma_uniq_id
      into :ma_date_oper, :ma_summa_minus, :ma_summa_minus_repeat, :ma_summa_minus_new, :ma_summa_miniappnew,
           :ma_region_id, :ma_kf_id, :ma_disb_kolich, :ma_fact_date, :ma_fact_disb_amount, :ma_plan_disb_all,
           :ma_region_name, :ma_product, :ma_circle, :ma_uniq_id
  do
  begin
    ma_app_id = null;
    if (ma_kf_id is not null) then
    begin
      if (ma_fact_date is null) then
      begin
        select distinct ma_id
        from miniapp
        where ma_date_oper = :ma_date_oper and
              ma_from_kf_id = :ma_kf_id rows 1
        into :ma_app_id;
      end
      else
      begin
        select distinct ma_id, ma_date_app
        from miniapp
        where ma_from_kf_id = :ma_kf_id and
              ma_date_app < :rep_date_end and
              ma_date_end >= :rep_date_end
        order by ma_date_app desc
        rows 1
        into :ma_app_id, :ma_dd;
      end
    end
    if (ma_date_oper is not null) then
      ma_date_sort = :ma_date_oper;
    else
      ma_date_sort = :ma_fact_date;
    ma_app_no = null;
    ma_app_date = null;
    if (ma_app_id is not null) then
    begin
      select ma_no, ma_date_app
      from miniapp
      where ma_id = :ma_app_id
      order by ma_date_app
      rows 1
      into :ma_app_no, :ma_app_date;
    end
    ma_kf_name = null;
    ma_kc_name = null;
    if (ma_kf_id is not null) then
    begin
      select kal_new_kf_name, c_name_fio
      from knaktlist, clients
      where kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            knaktlist.is_delete = 0 and
            kal_from_kf_id = :ma_kf_id and
            kal_new_kf_officer = c_id rows 1
      into :ma_kf_name, :ma_kc_name;
    end
    if (ma_kf_name is null) then
    begin
      ma_kf_name = 'Прогноз.выдачи';
      ma_kc_name = :ma_region_name;
    end
    ma_plan_fact = :ma_plan_disb_all - :ma_fact_disb_amount;
    suspend;
  end

end
