CREATE PROCEDURE CM_RU_0318_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable DISB_DATE D_DATE;
declare variable FINAL_PDN D_CURRENCY;
declare variable KC_ID D_ID;
declare variable CLI_BALANCE CURR_TYPE;
declare variable AMOUNT CURR_TYPE;
declare variable COUNTER D_INTEGER;
declare variable DELINQ_BALANCE D_CURRENCY;
declare variable OFFICE_ID D_INTEGER;
declare variable CLIENT_POINT D_INTEGER;
declare variable LAST_DISB_DATE D_DATE;
declare variable MAX_SUM D_CURRENCY;
begin
  res_error_is_critical = 2;

  for select kc_id, kc_from_c_id, k_disb_date, k_fact_disb_reg_id, kc_amount_approved
      from kontract
      inner join knclient on kc_from_k_id = k_id
      inner join product on pr_id = k_from_pr_id
      where k_id = :k_id and
            pr_code containing '.plus_100_b.' and
            kc_amount_approved > 0
      into kc_id, c_id, disb_date, office_id, amount
  do
  begin
    final_pdn = 0;
    client_point = 0;
    client_point = f_point_get(9, k_id, c_id, disb_date);

    select coalesce(kc_pdn_koef, 0)
    from knclient_pdn
    where kc_id = :kc_id and
          kcp_kind = 0
    into final_pdn;

    -- посчитать количество выданных +60
    counter = 0;
    last_disb_date = null;
    select count(k_id), max(k_disb_date)
    from kontstate, knclient, kontract, product
    where ks_ol_is_delete = 0 and
          ks_state >= 1 and
          ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          ks_from_k_id = kc_from_k_id and
          :disb_date between ks_date_begin and ks_date_end and
          kc_amount_approved > 0 and
          kc_from_c_id = :c_id and
          pr_code containing '.plus_100_b.'
    into counter, last_disb_date;
    counter = coalesce(counter, 0);
    if (last_disb_date is null) then last_disb_date = '01.01.1990';
    /*
    --    клиенты с 0 - 29 баллами могут получить только один заем по данному продукту и при условии, что ПДН не превышает 50%.
    if (client_point <= 29 and
        final_pdn > 50) then
    begin
      res_error_text = 'Кредитование запрещено: ПДН более 50%';
      suspend;
    end

    if (client_point <= 29 and
        counter >= 1) then
    begin
      res_error_text = 'Клиент может получить только один заем по данному продукту';
      suspend;
    end
    */
    --    клиенты могут получить, что ПДН не превышает 80%.
    if (final_pdn > 80) then
    begin
      res_error_text = 'Кредитование запрещено: ПДН более 80%';
      suspend;
    end

    if (last_disb_date + 3 > disb_date) then
    begin
      res_error_text = 'Новый заем по продукту Плюс 100 может быть выдан не ранее ' || f_datetostr((last_disb_date + 3));
      suspend;
    end

    /*
    if (client_point >= 30 and
        counter >= 2) then
    begin
      res_error_text = 'Клиент может получить заем по данному продукту не более двух раз';
      suspend;
    end
    */

    --    Заем может быть выдан при условии, что после его выдачи общая задолженность клиента по основному долгу не превысит 500 000 руб.
    cli_balance = 0;
    delinq_balance = 0;
    select sum(oks_balance), sum(iif(:disb_date > oks_delinq_date, oks_delinq_os + oks_delinq_perc, 0))
    from knclient, oksummary
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = oks_from_k_id and
          oks_balance > 0 and
          oks_close_state = 0 and
          :disb_date between oks_date_begin and oks_date_end
    into cli_balance, delinq_balance;
    cli_balance = coalesce(cli_balance, 0);
    delinq_balance = coalesce(delinq_balance, 0);

    max_sum = case
                when client_point <= 6 then 100000
                when client_point <= 11 then 150000
                when client_point <= 17 then 200000
                when client_point <= 24 then 250000
                when client_point <= 35 then 300000
                when client_point <= 45 then 350000
                else 450000
              end;

    if (cli_balance + amount > max_sum) then
    begin
      res_error_text = 'Общий баланс всех займов превышает лимит:'||max_sum;
      suspend;
    end

    if (delinq_balance > 0) then
    begin
      res_error_text = 'У клиента имеется активная просрочка';
      suspend;
    end

    if (exists(select *
               from schedule sh
               where sh.sh_from_k_id = :k_id and
                     sh.sh_id > 0 and
                     sh.sh_os = 0 and 
                     sh.sh_grace = 1 and
                     extract(month from sh_date) not in (1, 2)
        )) then                                                          
    begin
        res_error_is_critical = 2;
        res_error_text = 'Льготный период можно получить только на январь и/или февраль';
        suspend;
    end


    if (not exists(select * from temp_schedule_correct
    where tsc_k_id = :c_id
    and tsc_sh_id = 1)) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'клиенту запрещено получить кредит по продукту "Плюс 100 базовый"';
      suspend;
    end 

  end
end
