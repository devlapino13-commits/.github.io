CREATE PROCEDURE HP_TEMPO_ISTISNAA(
    K_ID D_ID)
RETURNS (
    K_ID_OUT D_ID,
    K_KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    KS_STATE D_INTEGER,
    IS_OLD D_BOOL,
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    ZAD_OS D_CURRENCY,
    ZAD_PERC D_CURRENCY)
AS
declare variable kf_id d_id;
declare variable pr_code d_str_100;
declare variable office_id d_integer;
declare variable office_name d_str_100;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable ol_id d_id;
declare variable ol_date d_date;
declare variable ol_memo d_str_long;
declare variable ol_doc_client_name d_str_255;
declare variable ob_detail d_str_long;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_doc_kind d_integer;
declare variable ol_is_client d_integer;
declare variable ok_type_pay d_integer;
declare variable ok_type_over d_integer;
declare variable ok_type_auto d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable acc_short_dt d_str_20;
declare variable acc_short_kt d_str_20;
declare variable acc_not_exist d_str_100;
declare variable early_perc d_currency;
declare variable fix_os d_currency;
declare variable ob_os d_currency;
declare variable fix_perc d_currency;
declare variable ob_perc d_currency;
declare variable zad_os_total d_currency;
declare variable zad_perc_total d_currency;
declare variable perc_future d_currency;
declare variable perc_get d_currency;
declare variable pay_os_2 d_currency;
declare variable pay_os_3 d_currency;
declare variable nsp_pay_2 d_currency;
declare variable nsp_pay_3 d_currency;
declare variable nsp_total d_currency;
declare variable nsp_as_pay d_currency;
declare variable nsp_as_over d_currency;
declare variable over_os d_currency;
declare variable over_nsp d_currency;
declare variable over_nsp_tmp d_currency;
declare variable date_begin d_date;
declare variable date_end d_date;
declare variable is_suspend d_bool;
declare variable pr_id d_integer;
declare variable pr_name d_str_code;
declare variable kc_amount_approved d_currency;
declare variable kc_from_c_id d_id;
declare variable kc_loan_target d_integer;
declare variable loan_target d_str_255;
declare variable zad_tax_total d_currency;
declare variable delta d_currency;
begin
  -- разовая процедура для возврата ЧДП (RFCKG-5180)

  date_begin = '10.11.2025';
  date_end = '28.02.2026';
  ol_date = '30.04.2026';
  k_id_out = k_id;

  select k_kontract_no,
         k_fact_disb_date,
         k_from_kf_id,
         pr_id,
         pr_code,
         pr_name
  from kontract
  inner join product on pr_id = k_from_pr_id
  where k_id = :k_id
  into k_kontract_no,
       k_fact_disb_date,
       kf_id,
       pr_id,
       pr_code,
       pr_name;

  if (pr_id <> 445) then -- Истиснаа_1 / .IND.IPF.ISTISNAA.ISTISNAA_PROCESS.prepayment.kontmat.
    exception e_message 'Неверно указан контракт/продукт - ' || k_kontract_no || ' / Истиснаа_1';

  select kc_from_c_id,
         kc_amount_approved,
         kc_loan_target
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  order by kc_amount_approved desc
  rows 1
  into kc_from_c_id,
       kc_amount_approved,
       kc_loan_target; -- цель финансирования (id)

  select rd_name
  from refdata
  where rd_id = :kc_loan_target
  into loan_target; -- цель финансирования (название)

  select ks_state
  from kontstate
  where ks_from_k_id = :k_id and
        ks_ol_is_delete = 0 and
        :ol_date between ks_date_begin and ks_date_end
  into ks_state; -- статус контракта до изменений

  zad_os_total = 0;
  zad_perc_total = 0;
  is_old = 1;
  is_suspend = 0;
  for select sh_id,
             sh_date,
             zad_os,
             zad_perc
      from aarep_make_zad_list(:k_id, :ol_date, 1)
      into sh_id,
           sh_date,
           zad_os,
           zad_perc
  do
  begin
    suspend; -- график до изменений
    is_suspend = 1;
    zad_os_total = zad_os_total + coalesce(zad_os, 0);
    zad_perc_total = zad_perc_total + coalesce(zad_perc, 0);
  end

  if (is_suspend = 0) then
    suspend; -- если предыдущий suspend не сработал, то делаем suspend здесь

  select early_perc
  from hp_get_kontract_early_repay(:k_id, :date_begin, :date_end)
  into early_perc; -- уступка по наценке (сумма уступки)

  if (early_perc = 0) then
    exit;

  -- загрушки на элементы, оставшиеся от скрипта по Мурабахе
  perc_get = 0;
  perc_future = 0;
  pay_os_2 = 0;
  pay_os_3 = 0;
  over_os = 0;

  -- *** (1) -- остатки наценки к получению
  -- по Истиснаа 1-го пункта нет

  select sum(ok_tax)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1
  into zad_tax_total; -- баланс НСП до возврата

  if (zad_os_total < 0) then
    delta = zad_os_total + early_perc; -- разница между задолженностью ОС и суммой уступки
  else
    delta = zad_os_total - early_perc; -- разница между задолженностью ОС и суммой уступки

  -- фиксация исходных данных контракта до возврата ЧДП
  insert into test_return (k_id, k_kontract_no, k_fact_disb_date, kc_amount_approved, loan_target, pr_id, pr_code, pr_name, old_ks_state, old_zad_os_total, old_zad_perc_total, old_zad_tax_total, perc_get, perc_future, early_perc, delta, ol_id)
  values (:k_id, :k_kontract_no, :k_fact_disb_date, :kc_amount_approved, :loan_target, :pr_id, :pr_code, :pr_name, :ks_state, :zad_os_total, :zad_perc_total, :zad_tax_total, :perc_get, :perc_future, :early_perc, :delta, 0);

  --***-- начало возврата ЧДП по Истиснаа

  if (ks_state = 1) then -- займ активный
  begin
    if (zad_os_total > early_perc) then
    begin
      -- после возврата останется активный остаток ОС (1-ситуация)
      pay_os_2 = early_perc;
      early_perc = 0;
      over_os = 0;
    end
    else
    begin
      -- займ закроется после возврата, возможен излишек, т.е. переплата (2-ситуация)
      pay_os_2 = zad_os_total;
      over_os = early_perc - zad_os_total;
      early_perc = 0;
    end
  end

  if (ks_state = 2) then -- займ закрытый
  begin
    -- вся сумма возврата идет на переплату
    pay_os_2 = 0;
    over_os = early_perc;
    early_perc = 0;
  end

  -- далее pay_os_2 и over_os фиксируем в operlist/operkred/operbook если они больше нуля

  select kal_new_reg_id
  from knaktlist
  where is_delete = 0 and
        kal_from_kf_id = :kf_id and
        :ol_date between kal_date and kal_date_end
  into office_id;

  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :office_id
  into office_name;

  acc_not_exist = 'Для офиса ' || office_name || ' не определен счет ';

  select kc_from_c_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0 rows 1
  into c_id;

  select c_name_fio
  from clients
  where c_id = :c_id
  into c_name_fio;

  ol_doc_client_name = 'Возврат наценки по досрочным погашениям согласно Приказа';
  ol_memo = ol_doc_client_name || ' / ' || k_kontract_no || ' от ' || f_datetostr(k_fact_disb_date) || ' / ' || c_name_fio;
  ol_oper_type = 1; -- кредитная
  ol_oper_subtype = 2; -- погашение
  ol_doc_kind = 9; -- мемориальный ордер
  ol_is_client = 1; -- операция по клиенту
  ok_type_pay = 4; -- погашение
  ok_type_over = 9; -- переплата
  ok_type_auto = 3; -- автореструктуризация

  insert into operlist (ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no_str, ol_document_client, ol_is_client, ol_memo, ol_doc_client_name, ol_user_id, ol_date)
  values (:office_id, :office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :k_id, :c_id, :ol_is_client, :ol_memo, :ol_doc_client_name, 1000000000000, :ol_date)
  returning (ol_id)
  into ol_id;

  if (coalesce(ol_id, 0) = 0) then
    exit;

  nsp_pay_2 = 0;
  nsp_pay_3 = 0;
  over_nsp = 0;
  over_nsp_tmp = 0;

  -- *** (2) -- погашение ОС со счета "Отсроченные доходы по наценке"
  if (pay_os_2 > 0) then
  begin
    ob_os = 0;

    for select sh_id,
               sh_date,
               zad_os
        from aarep_make_zad_list(:k_id, :ol_date, 1)
        order by sh_date desc
        into sh_id,
             sh_date,
             zad_os
    do
    begin
      fix_os = 0;

      if (pay_os_2 > zad_os) then
      begin
        -- уступка больше платежа по графику
        fix_os = zad_os;
        pay_os_2 = pay_os_2 - fix_os;
      end
      else
      begin
        fix_os = pay_os_2;
        pay_os_2 = 0;
      end

      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
      values (:ol_id, -1, :k_id, :sh_id, :sh_date, :ok_type_pay, -:fix_os, 0, 0, 3, 0, 0, 0, 0, 0, :ol_date, :office_id, 1, 0);

      ob_os = ob_os + fix_os; -- сумма ОС для фиксации в operbook
      nsp_pay_2 = nsp_pay_2 + round(fix_os * 2 / 100, 0); -- НСП общий по погашениям

      if (pay_os_2 = 0) then -- суммы для распределения не осталось
        leave;
    end

    if (ob_os > 0) then
    begin
      ob_detail = 'Основная сумма финансирования за счет уступки по наценке (2)';
      acc_short_dt = '61107/1';
      acc_short_kt = '10976/1';

      acc_dt = '';
      select al_id
      from acc_list
      where al_from_reg_id = :office_id and
            al_currency = '417' and
            al_account = :acc_short_dt
      into acc_dt;

      if (coalesce(acc_dt, '') = '') then
        exception e_message acc_not_exist || acc_short_dt;

      acc_kt = '';
      select al_id
      from acc_list
      where al_from_reg_id = :office_id and
            al_currency = '417' and
            al_account = :acc_short_kt
      into acc_kt;

      if (coalesce(acc_kt, '') = '') then
        exception e_message acc_not_exist || acc_short_kt;

      insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_client_dt, ob_detail)
      values (:ol_id, :acc_dt, :acc_kt, :ob_os, :ob_os, :ol_date, :office_id, :c_id, :ob_detail);

      zad_os_total = zad_os_total - ob_os; -- баланс ОС после pay_os_2 (2)

      update test_return
      set pay_os_2 = :ob_os
      where k_id = :k_id;
    end
  end

  -- *** (5) -- НСП
  nsp_total = nsp_pay_2; -- НСП погашения ОС (переплата НСП)

  if (nsp_total > 0) then
  begin
    -- если после возврата уступки остается задолженность ОС и в то же время есть переплата НСП, то НСП сажаем на ОС как погашение, излишки НСП сажаем как переплату ОС
    -- если же после возврата уступки задолженности ОС нету, то всю сумму НСП сажаем как переплату ОС

    if (nsp_total > zad_os_total) then
    begin
      -- НСП больше задолженности ОС
      nsp_as_pay = zad_os_total; -- НСП как погашение ОС будет посажено только на сумму задолженности ОС
      nsp_as_over = nsp_total - zad_os_total; -- остальная часть переплаты НСП пойдет на переплату ОС
    end
    else
    begin
      -- НСП меньше или равно задолженности ОС
      nsp_as_pay = nsp_total; -- вся сумма НСП идет на погашение ОС
      nsp_as_over = 0; -- переплаты НСП нет
    end

    if (coalesce(nsp_as_pay, 0) <= 0) then
      nsp_as_pay = 0;

    if (coalesce(nsp_as_over, 0) <= 0) then
      nsp_as_over = 0;

    if (nsp_as_pay > 0) then -- есть сумма НСП, которая должна быть посажена как погашение ОС (5.1)
    begin
      ob_os = 0;

      for select sh_id,
                 sh_date,
                 zad_os
          from aarep_make_zad_list(:k_id, :ol_date, 1)
          order by sh_date desc
          into sh_id,
               sh_date,
               zad_os
      do
      begin
        fix_os = 0;

        if (nsp_as_pay > zad_os) then
        begin
          -- НСП больше платежа по графику
          fix_os = zad_os;
          nsp_as_pay = nsp_as_pay - zad_os;
        end
        else
        begin
          fix_os = nsp_as_pay;
          nsp_as_pay = 0;
        end

        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
        values (:ol_id, -1, :k_id, :sh_id, :sh_date, :ok_type_pay, -:fix_os, 0, 0, 3, 0, 0, 0, 0, 0, :ol_date, :office_id, 1, 0);

        ob_os = ob_os + fix_os; -- сумма ОС для фиксации в operbook

        if (nsp_as_pay = 0) then -- суммы для распределения не осталось
          leave;
      end

      if (ob_os > 0) then
      begin
        ob_detail = 'Основная сумма финансирования за счет переплаты НСП (5.1)';
        acc_short_dt = '21131/21';
        acc_short_kt = '10976/1';

        acc_dt = '';
        select al_id
        from acc_list
        where al_from_reg_id = :office_id and
              al_currency = '417' and
              al_account = :acc_short_dt
        into acc_dt;

        if (coalesce(acc_dt, '') = '') then
          exception e_message acc_not_exist || acc_short_dt;

        acc_kt = '';
        select al_id
        from acc_list
        where al_from_reg_id = :office_id and
              al_currency = '417' and
              al_account = :acc_short_kt
        into acc_kt;

        if (coalesce(acc_kt, '') = '') then
          exception e_message acc_not_exist || acc_short_kt;

        insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_client_dt, ob_detail)
        values (:ol_id, :acc_dt, :acc_kt, :ob_os, :ob_os, :ol_date, :office_id, :c_id, :ob_detail);

        zad_os_total = zad_os_total - ob_os; -- баланс ОС после nsp_as_pay (5.1)
      end
    end

    if (nsp_as_over > 0) then -- есть сумма НСП, которая должна быть посажена как переплата НСП (5.2)
    begin
      over_nsp = nsp_as_over; -- общая сумма НСП

      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
      values (:ol_id, -1, :k_id, :sh_id, :sh_date, :ok_type_over, 0, 0, 0, 3, 0, 0, 0, 0, -:nsp_as_over, :ol_date, :office_id, 1, 0);

      -- в журнал проводок переплата НСП записывается одной суммой (over_nsp), которая сформируется после этапов 5.2 и 4.
    end
  end

  -- *** (4) -- переплата ОС (остаток уступки после ее распределения по графику)
  if (over_os > 0) then
  begin
    over_nsp_tmp = round(over_os * 2 / 100, 0);
    over_nsp = over_nsp + over_nsp_tmp; -- общая сумма НСП

    select ok_sh_id,
           ok_sh_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1
    order by ok_ol_date desc, ok_sh_id desc
    rows 1
    into sh_id,
         sh_date; -- дата последней операции по контракту

    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
    values (:ol_id, -1, :k_id, :sh_id, :sh_date, :ok_type_over, -:over_os, 0, 0, 3, 0, 0, 0, 0, -:over_nsp_tmp, :ol_date, :office_id, 1, 0);

    ob_detail = 'Переплата финансирования на ОС (4)';
    acc_short_dt = '61107/1';
    acc_short_kt = '21199/101';

    acc_dt = '';
    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = :acc_short_dt
    into acc_dt;

    if (coalesce(acc_dt, '') = '') then
      exception e_message acc_not_exist || acc_short_dt;

    acc_kt = '';
    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = :acc_short_kt
    into acc_kt;

    if (coalesce(acc_kt, '') = '') then
      exception e_message acc_not_exist || acc_short_kt;

    insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_client_dt, ob_detail)
    values (:ol_id, :acc_dt, :acc_kt, :over_os, :over_os, :ol_date, :office_id, :c_id, :ob_detail);

    zad_os_total = zad_os_total - over_os; -- баланс ОС после over_os (4)
  end

  if (over_nsp > 0) then
  begin
    -- на этапах 5.2 (излишки НСП) и 4 (НСП переплаты ОС) образуется общая сумма НСП
    -- эту сумму требуется записать в журнал проводок одной проводкой

    ob_detail = 'Переплата финансирования на НСП';
    acc_short_dt = '21131/21';
    acc_short_kt = '21199/101';

    acc_dt = '';
    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = :acc_short_dt
    into acc_dt;

    if (coalesce(acc_dt, '') = '') then
      exception e_message acc_not_exist || acc_short_dt;

    acc_kt = '';
    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = :acc_short_kt
    into acc_kt;

    if (coalesce(acc_kt, '') = '') then
      exception e_message acc_not_exist || acc_short_kt;

    insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_client_dt, ob_detail)
    values (:ol_id, :acc_dt, :acc_kt, :over_nsp, :over_nsp, :ol_date, :office_id, :c_id, :ob_detail);

    zad_os_total = zad_os_total - over_nsp; -- баланс ОС после over_nsp
  end

  --*** (6) -- если есть переплата, то создаем запись в temp_minus_loan_overpay
  -- задача разовая и запись является защитой от "случайно лишнего" возврата клиентам переплаты по ЧДП,
  -- т.к. возврат переплаты по процентам и НСП возможен только если контракт есть в temp_minus_loan_overpay
  if (over_os + over_nsp > 0) then
    insert into temp_minus_loan_overpay (k_id, k_kontract_no, k_fact_disb_date, office_id, client_id, client_fio, ol_id, amount_os, amount_perc, amount_nsp, sh_id, sh_date, ol_date)
    values (:k_id, :k_kontract_no, :k_fact_disb_date, :office_id, :c_id, :c_name_fio, :ol_id, :over_os, 0, :over_nsp, :sh_id, :sh_date, :ol_date);

  select sum(zad_os),
         sum(zad_perc)
  from aarep_make_zad_list(:k_id, :ol_date, 1)
  into zad_os_total,
       zad_perc_total;

  --***-- если после возврата ЧДП задолженности нет, но контракт активный, то закрываем контракт
  if (zad_os_total + zad_perc_total <= 0 and
      exists(select *
             from kontstate
             where ks_from_k_id = :k_id and
                   ks_ol_is_delete = 0 and
                   ks_state = 1 and
                   :ol_date between ks_date_begin and ks_date_end)) then
  begin
    insert into kontstate (ks_from_ol_id, ks_from_k_id, ks_date_begin, ks_ol_is_delete, ks_state)
    values (:ol_id, :k_id, :ol_date, 0, 2);
  end

  select ks_state
  from kontstate
  where ks_from_k_id = :k_id and
        ks_ol_is_delete = 0 and
        :ol_date between ks_date_begin and ks_date_end
  into ks_state; -- статус контракта после изменений (возврата ЧДП)

  is_old = 0;
  is_suspend = 0;

  for select sh_id,
             sh_date,
             zad_os,
             zad_perc
      from aarep_make_zad_list(:k_id, :ol_date, 1)
      into sh_id,
           sh_date,
           zad_os,
           zad_perc
  do
  begin
    suspend; -- график после изменений
    is_suspend = 1;
  end

  if (is_suspend = 0) then
    suspend; -- если предыдущий suspend не сработал, то делаем suspend здесь

  select sum(ok_tax)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1
  into zad_tax_total; -- баланс НСП после возврата

  -- фиксация данных контракта после возврата ЧДП
  update test_return
  set over_os = :over_os,
      nsp_pay_2 = :nsp_pay_2,
      nsp_pay_3 = :nsp_pay_3,
      nsp_over = :over_nsp,
      nsp_total = :nsp_total,
      new_ks_state = :ks_state,
      new_zad_os_total = :zad_os_total,
      new_zad_perc_total = :zad_perc_total,
      new_zad_tax_total = :zad_tax_total,
      ol_id = :ol_id
  where k_id = :k_id;

end
