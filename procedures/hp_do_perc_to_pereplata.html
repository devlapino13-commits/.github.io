CREATE PROCEDURE HP_DO_PERC_TO_PEREPLATA(
    K_ID D_ID)
AS
declare variable rep_date d_date;
declare variable db_id d_integer;
declare variable start_date d_date;
declare variable closed_date d_date;
declare variable un_id d_id;
declare variable orp_id d_integer;
declare variable res_is_error d_smallint;
declare variable res_text d_str_255;
declare variable first_ol_date d_date;
declare variable disb_date d_date;
declare variable min_date d_date;
declare variable closed_ol_id d_id;
declare variable cpf_id d_integer;
declare variable cpf_status d_integer;
declare variable cpf_new_bal_perc d_currency;
declare variable cpf_orig_bal_perc d_currency;
declare variable cpf_closed_bal_perc d_currency;
declare variable new_classic_perc_active d_smallint;
declare variable need_exit d_bool;
declare variable oks_date_begin d_date;
declare variable is_closed d_bool;
declare variable is_terminated d_bool;
declare variable need_calc d_bool;
declare variable action_type d_smallint;
declare variable action_subtype d_smallint;
declare variable old_amount d_currency;
declare variable new_amount d_currency;
declare variable k_stop_perc d_date;
declare variable k_kredline_end d_date;
declare variable tmp_amount d_currency;
declare variable just_type_7 d_bool;
begin
  un_id = 1000000000001;

  new_classic_perc_active = f_tmp_const('new_classic_perc_active');
  if (new_classic_perc_active = 0) then
  begin
    update const_tmp
    set ct_value = 1
    where ct_code = 'new_classic_perc_active';
  end

  db_id = f_get_db_id();
  start_date = f_tmp_const('new_classic_perc_start_date');
  is_closed = 0;
  just_type_7 = 0;

  for select cpf_id, cpf_orig_bal_perc, cpf_new_bal_perc, cpf_closed_bal_perc, cpf_status
  from temp_calcperc_fix
  where cpf_from_k_id = :k_id
  into cpf_id, cpf_orig_bal_perc, cpf_new_bal_perc, cpf_closed_bal_perc, cpf_status
  do
  begin
    need_exit = 0;
    if (coalesce(cpf_status, 0) <> 0) then
      need_exit = 1;

    if (need_exit = 0 and maxvalue(abs(cpf_new_bal_perc - cpf_orig_bal_perc), abs(cpf_closed_bal_perc)) <= 5) then
    begin
      update temp_calcperc_fix
      set cpf_status = -1 -- Разница <= 5 руб
      where cpf_id = :cpf_id;
      need_exit = 1;
    end

    if (need_exit = 0) then
    begin
      select ks_from_ol_id
      from kontstate
      where ks_from_k_id = :k_id and
            ks_state = 2 and
            ks_ol_is_delete = 0
      into closed_ol_id;
    
      if (closed_ol_id is not null) then
      begin
        if (exists(select * from operkred
        where ok_from_ol_id = :closed_ol_id and
              ok_from_k_id = :k_id and
              ok_type = 5 and
              ok_os < 0)) then
        begin
          update temp_calcperc_fix
          set cpf_status = -2 -- Закрыт через списание
          where cpf_id = :cpf_id;
  
          need_exit = 1;
        end
      end
    end
  end

  if (need_exit = 1) then
  begin
    if (new_classic_perc_active = 0) then
    begin
      update const_tmp
      set ct_value = 0
      where ct_code = 'new_classic_perc_active';
    end

    Exit;
  end

  select ks_date_begin, ks_date_end, k_stop_perc, k_kredline_end
  from kontstate, kontract
  where ks_from_k_id = :k_id and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_from_k_id = k_id
  into disb_date, closed_date, k_stop_perc, k_kredline_end;

  is_terminated = 0;
  if (k_stop_perc is not null or k_kredline_end < :start_date) then
    is_terminated = 1;

  if (closed_date < cast('31.12.2999' as date)) then
  begin
    is_closed = 1;
    closed_date = closed_date + 1;
  end

  if (k_stop_perc is not null and cpf_new_bal_perc > cpf_orig_bal_perc) then
    just_type_7 = 1;

  if (is_closed = 1) then
  begin
    if (minvalue(cpf_new_bal_perc - cpf_orig_bal_perc, cpf_closed_bal_perc) > 0)
    then just_type_7 = 1;
    else just_type_7 = 0;
  end

  rep_date = start_date;

  if (not exists(select *
                 from kn_oksummary
                 where oks_from_k_id = :k_id and
                       oks_date_begin = :rep_date)) then
    execute procedure hp_kn_oksummary_kont_insert(:k_id, :rep_date, 1);

  if (is_closed = 0 or closed_date > start_date) then
  begin
    old_amount = cpf_orig_bal_perc;
    new_amount = cpf_new_bal_perc;
  end else
  begin
    old_amount = 0;
    new_amount = cpf_closed_bal_perc;
  end

  action_type = 0;
  action_subtype = 0;
  if (is_closed = 1) then
  begin
    -- Закрытые займы
    if (new_amount > old_amount) then
    begin
      action_type = 1;
      if (closed_date < start_date)
      then action_subtype = 1;
      else action_subtype = 2;
    end else
    begin
      -- Алгоритм ранее реализован
      if (closed_date < start_date) then
      begin
        action_type = 2;
        action_subtype = 3;
      end else
      begin
        action_type = 3;
        action_subtype = 3;
      end
    end
  end else
  begin
    -- Активные займы

    if (is_terminated = 1) then -- Расторжен или срок истек
    begin
      if (new_amount < old_amount) then
      begin
        if (new_amount >= 0) then
        begin
          action_type = 2;
          action_subtype = 1;
        end else
        begin
          action_type = 2;
          action_subtype = 2;
        end
      end else
      begin
        action_type = 1;
        action_subtype = 3;
      end
    end else
    begin
      if (new_amount < old_amount) then
      begin
        if (new_amount >= 0) then
        begin
          action_type = 3;
          action_subtype = 1;
        end else
        begin
          action_type = 4;
          action_subtype = 1;
        end
      end else
      begin
        action_type = 3;
        action_subtype = 2;
      end
    end
  end

  if (action_type = 0 or action_subtype = 0) then
    exception e_default f_format('Unknown logic, k_id^ %1', k_id);

  if (db_id <> 81001 or
     (db_id = 81001 and cpf_closed_bal_perc < 0 and (closed_date <= start_date or disb_date >= start_date))) then
  begin
    execute procedure hp_do_closed_perc_to_pereplata(:k_id);

    update temp_calcperc_fix
    set cpf_status = :action_type * 10 + :action_subtype
    where cpf_from_k_id = :k_id;

    if (new_classic_perc_active = 0) then
    begin
      update const_tmp
      set ct_value = 0
      where ct_code = 'new_classic_perc_active';
    end

    exit;
  end

  if (exists(select *
             from operrepeat, operlist ol
             where orp_from_k_id = :k_id and
                   orp_ol_date = :rep_date and
                   orp_from_ol_id = ol.ol_id and
                   ol.is_delete = 0)) then
    rep_date = rep_date + 1;

  first_ol_date = null;
  min_date = maxvalue(rep_date + 1, disb_date + 1);
  select min(ok_ol_date)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_ol_date between :min_date and :closed_date and
        ok_type = 4 and
        (ok_os <> 0 or ok_perc <> 0)
  into first_ol_date;

  if (action_type = 1) then
  begin
    rep_date = start_date;

    if (is_closed = 0) then
    begin
      need_calc = 1;

      if (k_stop_perc is not null and
          k_stop_perc > start_date and
          k_stop_perc <= cast('31.12.2020' as date)) then
        rep_date = k_stop_perc;

      if (rep_date > maxvalue(k_kredline_end, start_date)) then
        rep_date = maxvalue(k_kredline_end, start_date);
    end else
      if (closed_date > start_date) then
        need_calc = 1;

    if (is_closed = 1 and cpf_closed_bal_perc > 0 and closed_date > start_date) then
    begin
      rep_date = closed_date;
      old_amount = 0;
      new_amount = cpf_closed_bal_perc;
      need_calc = 0;

      execute procedure hp_do_perc_to_os(:k_id, :rep_date, :old_amount, :new_amount, 'type_7');
      execute procedure hp_do_perc_set_kn_saldo(:k_id, :rep_date, null);
    end else
    begin
      if (action_subtype = 2) then
      begin
        execute procedure hp_do_perc_to_os(:k_id, :rep_date, :cpf_orig_bal_perc, :cpf_new_bal_perc, 'type_7');
        execute procedure hp_do_perc_set_kn_saldo(:k_id, :rep_date, null);
      end else
      begin
        execute procedure hp_do_perc_to_os(:k_id, :rep_date, :old_amount, :new_amount, 'type_7');
        execute procedure hp_do_perc_set_kn_saldo(:k_id, :rep_date, null);
      end
    end
  end else
  if (action_type = 2) then
  begin
    rep_date = start_date;

    if (k_stop_perc is not null and
        k_stop_perc > start_date and
        k_stop_perc <= cast('31.12.2020' as date)) then
      rep_date = k_stop_perc;

    if (rep_date > maxvalue(k_kredline_end, start_date)) then
      rep_date = maxvalue(k_kredline_end, start_date);

    need_calc = 1;

    execute procedure hp_do_perc_to_os(:k_id, :rep_date, :old_amount, :new_amount, 'repay');

    if (is_terminated = 1) then
      execute procedure hp_do_perc_set_kn_saldo(:k_id, :rep_date, null);
  end else
  if (action_type = 3) then
  begin
    rep_date = start_date;

    need_calc = 1;
    execute procedure hp_do_perc_set_kn_saldo(:k_id, :rep_date, null);

    if (is_closed = 0 and first_ol_date is null) then
      execute procedure hp_do_perc_to_os(:k_id, :rep_date, :old_amount, :new_amount, 'repay');
  end else
  if (action_type = 4) then
  begin
    rep_date = start_date;
    need_calc = 1;

    execute procedure hp_do_perc_to_os(:k_id, :rep_date, :old_amount, :new_amount, 'repay');

    execute procedure hp_do_perc_set_kn_saldo(:k_id, :rep_date, null);
  end else
    exception e_default f_format('Unknown action_kind, k_id: %1', k_id);

  if (just_type_7 = 0 and first_ol_date is not null) then
  begin
    need_calc = 1;
    if (first_ol_date = start_date) then
      first_ol_date = first_ol_date + 1;

    select orp_id, res_is_error, res_text
    from hp_operrepeat_edit(0, :k_id, :first_ol_date, :first_ol_date, 1, 1, '', 'EMPTY', :un_id)
    into :orp_id, :res_is_error, :res_text;

    if (res_is_error = 1) then
      exception e_default res_text;

    execute procedure hp_operrepeat_set_status(:orp_id, f_enum('operrepeat_state', 'checked', 'id'), :un_id);

    execute procedure hp_repeat_repayment_save(:orp_id, '', :un_id);

    delete from operrepeat
    where orp_id = :orp_id and
          orp_from_ol_id is null;
  end

  if (just_type_7 = 0 and is_closed = 1 and closed_date >= start_date) then
  begin
    select result_perc_end
    from aarep_calc_perc_one_cls(:k_id, :closed_date, :closed_date, 0, 1, 1)
    into tmp_amount;

    if (tmp_amount <> 0) then
    begin
      execute procedure hp_do_perc_to_os(:k_id, :closed_date, 0, tmp_amount, 'type_7');
      execute procedure hp_do_perc_set_kn_saldo(:k_id, :closed_date, null);
    end
  end

  if (need_calc = 1) then
  begin
    if (just_type_7 = 0) then
    begin
      if (is_closed = 1)
      then min_date = minvalue(iif(closed_date = f_date_month_end(closed_date), closed_date - 1, closed_date),
                          cast('31.12.2020' as date));
      else min_date = cast('31.12.2020' as date);
  
      for select distinct oks_date_begin
      from kn_oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin between :start_date and :min_date and
            oks_is_month = 1
      into oks_date_begin
      do
        execute procedure hp_do_perc_set_kn_saldo(:k_id, :oks_date_begin, null);
    end
  end

  execute procedure hp_do_kn_operbook_one_online(:k_id, :start_date, null, :start_date - 1);

  update temp_calcperc_fix
  set cpf_status = :action_type * 10 + :action_subtype
  where cpf_from_k_id = :k_id;

--  execute procedure hp_do_closed_perc_to_pereplata(:k_id);

  if (new_classic_perc_active = 0) then
  begin
    update const_tmp
    set ct_value = 0
    where ct_code = 'new_classic_perc_active';
  end
end
