CREATE PROCEDURE MC_DOCS_FOR_LOAN(
    MCL_ID D_ID,
    PR_ID D_INTEGER,
    AMOUNT D_CURRENCY,
    PERIOD D_INTEGER)
RETURNS (
    PASSKIND_ID D_INTEGER,
    DOC_EXISTS D_SMALLINT,
    ORDER_NO D_INTEGER,
    PASSKIND_NAME D_STR_100,
    CDS_ID D_INTEGER,
    IS_MAIN_DOC D_SMALLINT,
    API_CODE D_STR_255,
    BLS_1 D_ID,
    BLS_2 D_ID,
    DRM_NAME D_STR_255,
    DRM_CODE D_STR_255,
    DRM_IS_MANUAL D_BOOL,
    DRM_RECOG_2 D_BOOL,
    DRM_SHOW_FIELDS_CLIENT D_BOOL,
    DRM_HAS_PAGE1 D_BOOL,
    DRM_HAS_PAGE2 D_BOOL)
AS
declare variable c_id d_id;
declare variable c_citizen d_integer;
declare variable drl_id d_id;
declare variable mcl_main_doc d_integer;
declare variable c_citizen_2 d_integer;
begin
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  if (coalesce(pr_id, 0) <= 0) then
    exception e_default 'parametr "pr_id" has invalid value';

  if (coalesce(amount, 0) <= 0) then
    exception e_default 'parametr "amount" has invalid value';

  if (coalesce(period, 0) <= 0) then
    exception e_default 'parametr "period" has invalid value';

  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into c_id;

  select cd.cds_from_pk_id
  from clientdocs cd
  where cd.cds_from_c_id = :c_id and
        cd.cds_is_main_doc = 1 and
        cd.cds_is_delete = 0 and
        cd.cds_verify_status > -1
  order by cd.cds_verify_datetime desc
  rows 1
  into mcl_main_doc;

  -- если пользователю МП еще не присвоен C_ID и соотвественно его гражданство не изветсно, тяну его самый поелдний основной документ
  if (c_id is null) then
  begin
    cds_id = null;
    passkind_id = null;
    api_code = null;
    is_main_doc = null;
    doc_exists = 0;

    select drl.drl_id, drl.drl_from_pk_id, pdt.pdt_is_main_doc, drm.drm_api_code, drl_from_bls_1, drl_from_bls_2,

           drm_name, drm_code, drm_is_manual, drm_recog_2, drm_show_fields_client
    from docrecoglist drl
    inner join passkind pask on pask.pk_id = drl.drl_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pask.pk_from_pdt_id
    inner join actionlist al on al.acl_id = drl.drl_from_acl_id
    inner join docrecogmain drm on drm.drm_id = pask.pk_from_drm_id
    where drl.drl_from_mcl_id = :mcl_id and
          drl.drl_is_main_doc = 1 and
          ((coalesce(drl.drl_from_bls_1, 0) > 0) or
          (coalesce(drl.drl_from_bls_2, 0) > 0)) and
          al.acl_status >= 0 and -- учитываю только документы, по которым статус проверки "в процессе" или "подтвержден"
          drm.drm_show_in_client_mobile = 1
    order by drl.drl_id desc
    rows 1
    into :drl_id, :passkind_id, :is_main_doc, :api_code, :bls_1, :bls_2,

         :drm_name, :drm_code, :drm_is_manual, :drm_recog_2, :drm_show_fields_client;

    passkind_name = drm_name;

    select res_value
    from mc_doc_name_change(:passkind_id, :passkind_name)
    into :passkind_name;

    if (drl_id is not null) then
      doc_exists = 1;

    order_no = 1;

    select drm_has_page1, drm_has_page2
    from hp_get_drm_has_pages(:passkind_id)
    into :drm_has_page1, :drm_has_page2;

    suspend;
  end
  else
  begin
    select c_citizen
    from clients
    where c_id = :c_id
    into c_citizen;
    c_citizen = coalesce(c_citizen, 0);

    select pte_citizen
    from perstemp
    where pte_from_c_id = :c_id
    into c_citizen_2;

    if (c_citizen_2 is null) then
      c_citizen_2 = c_citizen;

    for select pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id,
               pdt.pdt_is_main_doc
        from docrecogmain drm, passkind pask, passdoctype pdt, hp_citizen_passkind(:c_citizen, :c_citizen_2)
        where drm_inactive = 0 and
              res_pk_id = pk_id and
              pdt.pdt_id = pask.pk_from_pdt_id and
              pk_from_drm_id = drm_id and
              pk_id > 0 and
              drm_show_to_client = 1 and
              drm.drm_show_in_client_mobile = 1 and
              iif(:mcl_main_doc is not null, pask.pk_id = :mcl_main_doc, true)
        order by pdt.pdt_is_main_doc desc, pk_name
        into drm_name, drm_code, api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, passkind_id,
             is_main_doc
    do
    begin
      passkind_name = drm_name;

      select res_value
      from mc_doc_name_change(:passkind_id, :passkind_name)
      into :passkind_name;

      select cds_id, doc_exists, bls_1, bls_2
      from mc_docs_for_loan_get(:mcl_id, :c_id, :passkind_id)
      into :cds_id, :doc_exists, :bls_1, :bls_2;

      select drm_has_page1, drm_has_page2
      from hp_get_drm_has_pages(:passkind_id)
      into :drm_has_page1, :drm_has_page2;

      suspend;
    end
  end
end
