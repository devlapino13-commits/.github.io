CREATE PROCEDURE HP_GET_ANTIFROD_DATA(
    K_ID D_ID)
RETURNS (
    CRITERIA D_STR_255,
    CRITERIA_CODE D_STR_CODE,
    CRITICAL_LEVEL D_SMALLINT,
    ANTIFROD_RES D_BOOL,
    RES_TEXT D_STR_255,
    RES_ADD_JSO D_INTEGER,
    CALC_DATETIME D_DATETIME)
AS
declare variable old_ogrn d_str_code;
declare variable ma_id d_id;
declare variable ma_id_tmp d_id;
declare variable lines_jsa d_integer;
declare variable line_jsa d_integer;
declare variable text1 d_str_8100;
declare variable text2 d_str_8100;
declare variable text3 d_str_8100;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable ipr_is_local d_bool;
declare variable max_is_local d_bool;
declare variable last_is_local d_bool;
declare variable last_datime d_datetime;
declare variable country d_str_255;
declare variable device_kind d_integer;
declare variable device_id d_str_code;
declare variable clients_cnt d_integer;
declare variable overdue_clients_cnt d_integer;
declare variable c_name_fio d_str_255;
declare variable kontract_no_list d_str_long;
declare variable device_type d_str_100;
declare variable cdv_create_datetime d_datetime;
declare variable cdv_update_datetime d_datetime;
declare variable is_dif d_smallint;
declare variable is_have_data d_smallint;
declare variable el_id d_integer;
declare variable acl_datetime d_datetime;
declare variable app_sign_datetime d_datetime;
declare variable fincert_datetime d_datetime;
declare variable k_fact_disb_date d_date;
declare variable fincert_res d_smallint;
declare variable rd_id d_id;
declare variable k_fact_disb_ol_id d_id;
declare variable curr_phone_number d_str_code;
declare variable k_kontract_no d_str_code;
declare variable k_kontract_date d_date;
declare variable k_disb_date d_date;
declare variable c_citizen d_integer;
declare variable pr_code d_str_100;
declare procedure create_jso(headers d_str_255, width_str d_str_255)
returns(jso d_integer, lines_jsa d_integer)
as
declare variable res_str d_str_255;
declare variable width_jsa d_integer;
declare variable header_jsa d_integer;
begin
  jso = json.obj_create();
  header_jsa = json.arr_create(jso, 'header');
  width_jsa = json.arr_create(jso, 'width');
  lines_jsa = json.arr_create(jso, 'lines');

  for select res_str
  from hp_split_str(:headers, ',')
  into res_str
  do
    json.arr_set_s(header_jsa, -1, res_str);

  for select res_str
  from hp_split_str(:width_str, ',')
  into res_str
  do
    json.arr_set_s(width_jsa, -1, res_str);

  suspend;
end

begin
  -- antifrod_res - 0 - положительный результат (не считаем), 1 - отрицательный результат (считаем)
  if (f_get_db_id() <> 81001) then
    exit;

  if (f_loan_is_online(k_id) = 0) then
    exit;  

  select k_from_ma_app_id, k_disb_date, kc_id, kc_from_c_id, k_kontract_no, k_kontract_date, k_fact_disb_ol_id, pr.pr_code
  from kontract k, knclient, product pr
  where kc_from_k_id = k_id
    and k_id = :k_id
    and pr.pr_id = k.k_from_pr_id
  into ma_id, k_disb_date, kc_id, c_id, k_kontract_no, k_kontract_date, k_fact_disb_ol_id, pr_code;

  select c.c_citizen
  from clients c
  where c.c_id = :c_id
  into c_citizen;


  select afl_datetime
  from antifrod_loan
  where afl_from_k_id = :k_id
  into calc_datetime;

  -- для старых займов, для которых еще не сохранялся расчет, найдем время последнего события для расчета
  if (calc_datetime is null) then
  begin
    select ol_datetime
    from operlist
    where ol_id = :k_fact_disb_ol_id
    into calc_datetime;

    if (calc_datetime is null) then
      select krl_datetime
      from kontreslist
      where krl_from_k_id = :k_id
      into calc_datetime;
  
    if (calc_datetime is null) then
      select krl_datetime
      from kontract, kontreslist
      where krl_ma_id = k_from_ma_app_id
        and k_id = :k_id
      into calc_datetime;
  
    if (calc_datetime is null) then
      select max(acl_datetime)
      from actionlist a
      where acl_object_kind = 9
        and acl_object_id = :k_id
      into calc_datetime;
  
    if (calc_datetime is null) then
      select max(ma_date_save)
      from kontract, miniapp
      where ma_id = k_from_ma_app_id
        and k_id = :k_id
      into calc_datetime;
  end

  -- дата и время подписания заявки
  select min(eds_status_datetime)
  from edoc_sign, edoc_list, edoc_blank edb
  where eds_from_edl_id = edl_id
    and edb_from_edl_id = edl_id
    and edl_from_edc_id = f_edoc_kind('loan.application','id')
    and edl_object_kind = 11
    and edl_object_id = :ma_id
    and eds_status = 1
  into app_sign_datetime;


  -- БКИ ------------------
  for select criteria, criteria_code, bki_text, db_text, res_text, critical_level, antifrod_res
  from hp_client_metrics_bki(:kc_id, :calc_datetime, '<br>')
  into :criteria, criteria_code, :text2, :text1, :res_text, :critical_level, antifrod_res
  do
  begin
    res_add_jso = null;

    if ((text1 || text2) is not null) then
    begin
      select jso, lines_jsa
      from create_jso('БД,БКИ', '50,50')
      into res_add_jso, lines_jsa;

      line_jsa = json.arr_create();
      json.arr_set_arr(lines_jsa, -1, line_jsa);
      json.arr_set_s(line_jsa, -1, text1);
      json.arr_set_s(line_jsa, -1, text2);
    end
    suspend;
  end

  -- МП -------------------------
  is_have_data = 0;

  select 1
  from clientdevicekont cdk
  where cdk.cdk_from_k_id = :k_id
  rows 1
  into is_have_data;

  if (is_have_data = 0) then
  begin
    criteria = 'Устройство. Местоположение.';
    criteria_code = 'device_location';
    critical_level = 0;
    res_text = 'нет данных';
    res_add_jso = null;
    max_is_local = 0;
    last_is_local = null;
    country = '';

    select jso, lines_jsa
    from create_jso('Дата,Местоположение', '30,70')
    into res_add_jso, lines_jsa;

    for select coalesce(ipr_country, '<Страна неизвестна>'),
               coalesce(ipr_region || ', ', '') || coalesce(ipr_place, ''), coalesce(ipr_is_local, 0),
               max(cdv_update_datetime)
    from clientdevice, hp_get_ip_info(cdv_value)
    where cdv_from_el_id = f_enum('client_device', 'ip_address', 'id') and
          cdv_from_c_id = :c_id and
          cdv_create_datetime < :calc_datetime
    group by 1, 2, 3
    order by 4 desc
    rows 5
    into country, text1, ipr_is_local, last_datime
    do
    begin
      last_is_local = coalesce(last_is_local, ipr_is_local);
      max_is_local = maxvalue(ipr_is_local, max_is_local);
      line_jsa = json.arr_create();
      json.arr_set_arr(lines_jsa, -1, line_jsa);
      json.arr_set_s(line_jsa, -1, f_datetimetostr(last_datime));
      json.arr_set_s(line_jsa, -1, iif(ipr_is_local = 0, country, text1));
    end

    if (last_is_local is not null) then
    begin
      critical_level = case
                         when last_is_local = 1 then 0 -- последнее местоположение у нас
                         when max_is_local = 1 then 1 -- есть наша страна
                         else 2 -- все не наша страна
                       end;

      res_text = case critical_level
                   when 0 then 'Последнее местоположение в стране'
                   when 1 then 'Последнее местоположение не в стране'
                   else 'Местоположение в стране не зафиксировано'
                 end;

      antifrod_res = iif(critical_level = 0, 0, 1);
    end

    suspend;
    -------
    criteria = 'Устройство. Эмулятор.';
    criteria_code = 'device_emulator';
    critical_level = 0;
    res_text = 'нет данных';
    res_add_jso = null;
  
    select coalesce(cast(cdv_value as integer) * 2, 0)                                                                       ---RFCRU-4235
    from clientdevice
    where cdv_from_el_id in (f_enum('client_device', 'andr_is_emul', 'id'), f_enum('client_device', 'ios_is_emul', 'id')) and
          cdv_from_c_id = :c_id and
          cdv_create_datetime < :calc_datetime
    order by cdv_update_datetime desc
    rows 1
    into critical_level;

    res_text = iif(critical_level = 0, 'нет признака', 'есть признак');
    antifrod_res = iif(critical_level = 0, 0, 1);

    suspend;
    -------

    select cdv_from_el_id, cdv_value
    from clientdevice
    where cdv_from_el_id in (f_enum('client_device', 'andr_id', 'id'), f_enum('client_device', 'ios_id', 'id')) and        ---RFCRU-4235 замена на andr_id
          cdv_from_c_id = :c_id and
          cdv_create_datetime < :calc_datetime
    order by cdv_update_datetime desc
    rows 1
    into device_kind, device_id;
  end
  else
  begin
    criteria = 'Устройство. Местоположение.';
    criteria_code = 'device_location';
    critical_level = 0;
    res_text = 'нет данных';
    res_add_jso = null;
    max_is_local = 0;
    last_is_local = null;
    country = '';

    select jso, lines_jsa
    from create_jso('Дата,Местоположение', '30,70')
    into res_add_jso, lines_jsa;

    for select coalesce(ipr_country, '<Страна неизвестна>'),
               coalesce(ipr_region || ', ', '') || coalesce(ipr_place, ''), coalesce(ipr_is_local, 0),
               max(cdk_update_datetime)
    from clientdevice cdv, clientdevicekont cdk, hp_get_ip_info(cdv_value)
    where cdk.cdk_from_cdv_id = cdv.cdv_id and
          cdv_from_el_id = f_enum('client_device', 'ip_address', 'id') and
          cdv_from_c_id = :c_id and
          cdv_create_datetime < :calc_datetime
    group by 1, 2, 3
    order by 4 desc
    rows 5
    into country, text1, ipr_is_local, last_datime
    do
    begin
      last_is_local = coalesce(last_is_local, ipr_is_local);
      max_is_local = maxvalue(ipr_is_local, max_is_local);
      line_jsa = json.arr_create();
      json.arr_set_arr(lines_jsa, -1, line_jsa);
      json.arr_set_s(line_jsa, -1, f_datetimetostr(last_datime));
      json.arr_set_s(line_jsa, -1, iif(ipr_is_local = 0, country, text1));
    end

    if (last_is_local is not null) then
    begin
      critical_level = case
                         when last_is_local = 1 then 0 -- последнее местоположение у нас
                         when max_is_local = 1 then 1 -- есть наша страна
                         else 2 -- все не наша страна
                       end;

      res_text = case critical_level
                   when 0 then 'Последнее местоположение в стране'
                   when 1 then 'Последнее местоположение не в стране'
                   else 'Местоположение в стране не зафиксировано'
                 end;

      antifrod_res = iif(critical_level = 0, 0, 1);
    end

    suspend;
    -------
    criteria = 'Устройство. Эмулятор.';
    criteria_code = 'device_emulator';
    critical_level = 0;
    res_text = 'нет данных';
    res_add_jso = null;

    select coalesce(cast(cdv_value as integer) * 2, 0)                                                                ---RFCRU-4235
    from clientdevice cdv
    inner join clientdevicekont cdk on cdk.cdk_from_cdv_id = cdv.cdv_id
    where cdv_from_el_id in (f_enum('client_device', 'andr_is_emul', 'id'), f_enum('client_device', 'ios_is_emul', 'id')) and
          cdv_from_c_id = :c_id and
          cdv_create_datetime < :calc_datetime
    order by cdk.cdk_update_datetime desc
    rows 1
    into critical_level;

    res_text = iif(critical_level = 0, 'нет признака', 'есть признак');
    antifrod_res = iif(critical_level = 0, 0, 1);

    suspend;
    -------

    select cdv_from_el_id, cdv_value
    from clientdevice cdv
    inner join clientdevicekont cdk on cdk.cdk_from_cdv_id = cdv.cdv_id
    where cdv_from_el_id in (f_enum('client_device', 'andr_id', 'id'), f_enum('client_device', 'ios_id', 'id')) and    ---RFCRU-4235 замена на andr_id
          cdv_from_c_id = :c_id and
          cdv_create_datetime < :calc_datetime
    order by cdv_update_datetime desc
    rows 1
    into device_kind, device_id;
  end

  -------
  criteria = 'Устройство. Клиенты.';
  critical_level = 0;
  criteria_code = 'device_clients';
  res_text = 'нет данных';
  res_add_jso = null;
  clients_cnt = 0;
  overdue_clients_cnt = 0;

  if (device_kind is not null) then
  begin
    select jso, lines_jsa
    from create_jso('Клиент,Договоры', '50,50')
    into res_add_jso, lines_jsa;

    for select c_name_fio, list(distinct iif(oks_delinq_date < cast(:calc_datetime as date), k_kontract_no, ''))
        from oksummary, knclient, clientdevice, kontract, clients
        where :calc_datetime between oks_date_begin and oks_date_end and
              cdv_from_el_id = :device_kind and
              cdv_value = :device_id and
              cdv_from_c_id <> :c_id and
              cdv_from_c_id = kc_from_c_id and
              oks_from_k_id = kc_from_k_id and
              k_id = kc_from_k_id and
              c_id = kc_from_c_id
        group by 1
        into c_name_fio, kontract_no_list
    do
    begin
      clients_cnt = clients_cnt + 1;
      if (kontract_no_list <> '') then
      begin
        overdue_clients_cnt = overdue_clients_cnt + 1;
        line_jsa = json.arr_create();
        json.arr_set_arr(lines_jsa, -1, line_jsa);
        json.arr_set_s(line_jsa, -1, c_name_fio);
        json.arr_set_s(line_jsa, -1, kontract_no_list);
      end
    end

    antifrod_res = iif(clients_cnt > 2, 1, 0);
    critical_level = iif(clients_cnt > 4, 2, 0);
    res_text = f_format('Количество активных клиентов с таким же ID устройства: %1, из них %2 в просрочке', clients_cnt, overdue_clients_cnt);
  end

  suspend;
  ---------------------------------

  criteria = 'Устройство. Данные.';
  critical_level = 0;
  criteria_code = 'device_data';
  res_text = '';
  res_add_jso = null;
  is_dif = 0;
  device_id = '';
  antifrod_res = 0;

  select jso, lines_jsa
  from create_jso('Тип,Значение,Появился,Обновлён', '25,25,25,25')
  into res_add_jso, lines_jsa;

  select k.k_fact_disb_date
  from kontract k
  where k.k_id = :k_id
  into :k_fact_disb_date;

  select rd_id
  from hp_get_check_loan_data(:k_id)
  into rd_id;

  select acl_datetime
  from actionlist a
  where acl_object_kind = 9 and
        acl_object_id = :k_id and
        acl_from_rd_id = :rd_id
  into :acl_datetime;

  select f_enum('client_device', cdv_from_el_id, 'code'), f_enum('client_device', cdv_from_el_id, 'name'), coalesce(cdv_value, ''),
         cdv_create_datetime, cdv_update_datetime
  from clientdevice
  where cdv_from_el_id in (f_enum('client_device', 'andr_id', 'id'), f_enum('client_device', 'ios_id', 'id')) and     ---RFCRU-4235 замена на andr_id
        cdv_from_c_id = :c_id and
        cdv_create_datetime < :calc_datetime
  order by cdv_update_datetime desc
  rows 1
  into device_type, text1, device_id, cdv_create_datetime, cdv_update_datetime;

  if (cast(cdv_update_datetime as d_date) > k_fact_disb_date) then
    cdv_update_datetime = acl_datetime;

  if (text1 <> '') then
  begin
    line_jsa = json.arr_create();
    json.arr_set_arr(lines_jsa, -1, line_jsa);
    json.arr_set_s(line_jsa, -1, text1);
    json.arr_set_s(line_jsa, -1, coalesce(device_id, ''));
    json.arr_set_s(line_jsa, -1, f_datetimetostr(cdv_create_datetime));
    json.arr_set_s(line_jsa, -1, f_datetimetostr(:cdv_update_datetime));
  end 

  if (device_type = 'andr_id') then               ---RFCRU-4235 замена на andr_id
  begin
    for select en.el_id
        from enum_list en
        where en.el_kind = 'client_device' and
              (en.el_code containing 'andr_' or
              el_code = 'ip_address') and
              en.el_code not in ('client_device', 'andr_device_id')
        order by en.el_id
        into :el_id
    do
    begin

      if (is_have_data = 0) then
      begin
        select coalesce(f_enum('client_device', cdv_from_el_id, 'name'), ''), coalesce(cdv_value, ''),
               cdv_create_datetime, cdv_update_datetime
        from clientdevice
        where cdv_from_el_id = :el_id and
              cdv_from_c_id = :c_id and
              cdv_create_datetime < :calc_datetime
        order by cdv_update_datetime desc
        rows 1
        into :text1, text2, cdv_create_datetime, cdv_update_datetime;

        if (text1 <> '') then
        begin
          line_jsa = json.arr_create();
          json.arr_set_arr(lines_jsa, -1, line_jsa);
          json.arr_set_s(line_jsa, -1, text1);
          json.arr_set_s(line_jsa, -1, text2);
          json.arr_set_s(line_jsa, -1, f_datetimetostr(cdv_create_datetime));
          json.arr_set_s(line_jsa, -1, f_datetimetostr(:cdv_update_datetime));
        end 
      end
      else
      begin
        select coalesce(f_enum('client_device', cdv_from_el_id, 'name'), ''), coalesce(cdv_value, ''),
               cdv_create_datetime, max(cdk.cdk_update_datetime)
        from clientdevice cdv
        inner join clientdevicekont cdk on cdk.cdk_from_cdv_id = cdv.cdv_id
        where cdk.cdk_from_k_id = :k_id and
              cdv.cdv_from_el_id = :el_id
        group by 1, 2, 3
        order by 4 desc
        rows 1
        into :text1, text2, cdv_create_datetime, cdv_update_datetime;

        if (text1 <> '') then
        begin
          line_jsa = json.arr_create();
          json.arr_set_arr(lines_jsa, -1, line_jsa);
          json.arr_set_s(line_jsa, -1, text1);
          json.arr_set_s(line_jsa, -1, text2);
          json.arr_set_s(line_jsa, -1, f_datetimetostr(cdv_create_datetime));
          json.arr_set_s(line_jsa, -1, f_datetimetostr(:cdv_update_datetime));
        end 
      end
    end
  end
  else
  if (device_type = 'ios_id') then
  begin
    for select en.el_id
        from enum_list en
        where en.el_kind = 'client_device' and
              (en.el_code containing 'ios_' or
              el_code = 'ip_address') and
              en.el_code not in ('client_device', 'andr_device_id')
        order by en.el_id
        into :el_id
    do
    begin

      if (is_have_data = 0) then
      begin
        select coalesce(f_enum('client_device', cdv_from_el_id, 'name'), ''), coalesce(cdv_value, ''),
               cdv_create_datetime, cdv_update_datetime
        from clientdevice
        where cdv_from_el_id = :el_id and
              cdv_from_c_id = :c_id and
              cdv_create_datetime < :calc_datetime
        order by cdv_update_datetime desc
        rows 1
        into :text1, text2, cdv_create_datetime, cdv_update_datetime;

        if (text1 <> '') then
        begin
          line_jsa = json.arr_create();
          json.arr_set_arr(lines_jsa, -1, line_jsa);
          json.arr_set_s(line_jsa, -1, text1);
          json.arr_set_s(line_jsa, -1, text2);
          json.arr_set_s(line_jsa, -1, f_datetimetostr(cdv_create_datetime));
          json.arr_set_s(line_jsa, -1, f_datetimetostr(:cdv_update_datetime));
        end 
      end
      else
      begin

        select coalesce(f_enum('client_device', cdv_from_el_id, 'name'), ''), coalesce(cdv_value, ''),
               cdv_create_datetime, max(cdk.cdk_update_datetime)
        from clientdevice cdv
        inner join clientdevicekont cdk on cdk.cdk_from_cdv_id = cdv.cdv_id
        where cdk.cdk_from_k_id = :k_id and
              cdv.cdv_from_el_id = :el_id
        group by 1, 2, 3
        order by 4 desc
        rows 1
        into :text1, text2, cdv_create_datetime, cdv_update_datetime;

        if (text1 <> '') then
        begin
          line_jsa = json.arr_create();
          json.arr_set_arr(lines_jsa, -1, line_jsa);
          json.arr_set_s(line_jsa, -1, text1);
          json.arr_set_s(line_jsa, -1, text2);
          json.arr_set_s(line_jsa, -1, f_datetimetostr(cdv_create_datetime));
          json.arr_set_s(line_jsa, -1, f_datetimetostr(:cdv_update_datetime));
        end 
      end
    end
  end

  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'Ранняя идентификация.';
  criteria_code = 'device_early_identification';
  res_add_jso = null;
  critical_level = 0;
  res_text = 'нет данных';

  if (device_type is not null) then
  begin
    for select k.k_fact_disb_date
        from knclient knc
        inner join kontract k on k.k_id = knc.kc_from_k_id
        where knc.kc_from_c_id = :c_id and
              knc.kc_from_k_id <> :k_id and
              knc.kc_amount_approved > 0 and
              k.k_fact_disb_date < cast(:calc_datetime as date)
        into :k_fact_disb_date
    do
    begin
      res_text = '';

      select cdv_value
      from clientdevice
      where cdv_from_el_id in (f_enum('client_device', :device_type, 'id')) and
            cdv_from_c_id = :c_id and
            cdv_create_datetime <= cast(:k_fact_disb_date as d_date) + 1
      order by cdv_update_datetime desc
      rows 1
      into :res_text;
      if (device_id = res_text) then
        is_dif = 1;
    end
  end

  res_text = iif(is_dif = 0, 'Клиент впервые обращается за займом с этого устройства.', 'Клиент получал предыдущий займ с этим же устройством.');
  antifrod_res = iif(is_dif = 0, 1, 0);

  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'ФИНЦЕРТ.';
  criteria_code = 'fincert';
  res_add_jso = null;
  critical_level = 0;
  res_text = '';

  select jso, lines_jsa
  from create_jso('Время проверки,Название ФИДов,Результат,Дата внесения в Стоп-лист ЦБ,Время получения ФИДа', '20,20,20,20,20')
  into res_add_jso, lines_jsa;

  -- последняя проверка
  select max(fcl_datetime)
  from fincert_check_log
  where fcl_object_kind = 11
    and fcl_object_id = :ma_id
  into fincert_datetime;

  if (fincert_datetime is null) then -- если лога нет, то берем информацию на время КК
    fincert_datetime = acl_datetime;

  for select res, res_type, f_datetostr(block_date), f_datetimetostr(ff_create_datetime)
  from hp_fincert_get_result(:c_id, :fincert_datetime)
  into fincert_res, text1, text2, text3
  do
  begin
    critical_level = maxvalue(critical_level, fincert_res * 2);

    line_jsa = json.arr_create();
    json.arr_set_arr(lines_jsa, -1, line_jsa);
    json.arr_set_s(line_jsa, -1, f_datetimetostr(fincert_datetime));
    json.arr_set_s(line_jsa, -1, text1);
    json.arr_set_s(line_jsa, -1, iif(fincert_res = 1, 'Найден в списке', 'Не найден'));
    json.arr_set_s(line_jsa, -1, text2);
    json.arr_set_s(line_jsa, -1, text3);
  end

  res_text = iif(critical_level = 0, 'Не найден', 'Найден в списке');
  antifrod_res = iif(critical_level = 0, 0, 1);

  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'Самозапрет.';
  criteria_code = 'selfblock';
  res_add_jso = null;
  critical_level = f_is_selfblock(c_id, calc_datetime) * 2;
  text1 = '';

  select f_datetostr(max(eil_date))
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'bki_ssp_40'
    and eil_status = 2
    and eil_date < :calc_datetime
  into text1;

  res_text = f_format('%1 (дата отчета КБКИ: %2)', iif(critical_level = 0, 'Самозапрет отсутствует', 'Выявлен самозапрет'), text1);
  antifrod_res = iif(critical_level = 0, 0, 1);

  suspend;
  ------------------------------------------------------------------------------------------------------------------
  criteria = 'Проверка СМЭВ.';        -- RFCRU-5385
  criteria_code = 'passport_smev';
  res_add_jso = null;
  text1 = '';

  if (f_get_rd_code(:c_citizen) <> 'Citizen.RU') then
  begin
    res_text = 'иностранный гражданин';
    antifrod_res = 0;
  end
  else
  begin
    select iif(status_code = 300, 'паспорт найден', 'паспорт найден в списке недействительных'),
           iif(status_code = 300, 0, 1),
           iif(status_code = 300, 0, 2)
    from hp_smevpass_get_result(:c_id, :ma_id, cast(:calc_datetime as d_date))
    into res_text,
         antifrod_res,
         critical_level;
  end

  suspend;
  ------------------------------------------------------------------------------------------------------------------
  criteria = 'Проверка OГРНИП.';      -- RFCRU-5385
  criteria_code = 'ogrn_ip';
  res_add_jso = null;
  critical_level = 0;
  antifrod_res = 0;
  text1 = '';

  if (:pr_code containing '.business.') then
  begin
    select
           (select ogrn
                    from hp_kontur_ip_get_base_info('kontract', k.k_id))
    from knclient kc
    inner join kontstate ks on kc.kc_from_k_id = ks.ks_from_k_id and ks.ks_state = 1
    inner join kontract k on k.k_id = kc.kc_from_k_id
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where kc.kc_from_c_id = :c_id and
          kc.kc_amount_approved > 0 and
          kc.kc_from_k_id <> :k_id and
          k.k_disb_date < :k_disb_date and
          pr.pr_code containing '.business.'
    order by k_disb_date desc
    rows 1
    into old_ogrn;

    old_ogrn = coalesce(:old_ogrn, '');

    select ogrn
    from hp_kontur_ip_get_base_info('kontract', :k_id)
    into text1;

    if (:old_ogrn <> '' and :old_ogrn <> text1) then
    begin
      res_text = 'номер не совпадает с заявленным, является стоп фактором';
      critical_level = 2;
      antifrod_res = 1;
    end
    else if (coalesce(text1, '') = '') then
    begin
      res_text = 'номер ОГРНИП не найден';
      critical_level = 2;
      antifrod_res = 1;
    end
    else
      res_text = f_format('номер предпринимателя %1', text1) ;
  end
  else
  begin
    res_text = 'Не применимо';
  end

  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'Проверка СБП.';
  criteria_code = 'sbp_check';
  res_add_jso = null;
  critical_level = 0;
  res_text = null;
  text1 = null;

  select fcn_name
  from fps_loan_banks, fps_client_names
  where fcn_from_fcb_id = fbl_from_fcb_id
    and flb_from_kc_id = :kc_id
    and fcn_status = 2
    and fcn_datetime <= :calc_datetime
  order by fcn_id desc
  rows 1
  into text1;

  res_text = iif(text1 is null, 'Проверка не проведена', 'Проверка проведена – ФИО совпадает');
  antifrod_res = iif(text1 is null, 1, 0);

  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'Проверка доступа к номеру мобильного телефона.';
  criteria_code = 'phone_access';
  res_add_jso = null;
  critical_level = 1;
  res_text = 'Проверка не проведена';
  curr_phone_number = null;
  text2 = null;
  text3 = null;

  -- по согласованию с Ильёй, проверяем, если заявка подписана через МП, то проверка пройдена, иначе не пройдена.
  select f_cont_code(edm_from_clt_id), f_datetimetostr(msg_datetime, 'dd.mm.yyyy hh:nn:ss'), f_datetimetostr(eds_status_datetime, 'dd.mm.yyyy hh:nn:ss')
  from edoc_list, edoc_sign, edoc_blank, edoc_msg edm, fsrobot_msgbox
  where eds_from_edl_id = edl_id
    and edb_from_edl_id = edl_id
    and edm_from_eds_id = eds_id
    and msg_id = edm_from_msg_id
    and edl_from_edc_id = f_edoc_kind('loan.application','id')
    and edl_object_kind = 11
    and edl_object_id = :ma_id
    and edl_status = 1
    and edm_status = 1
    and eds_status = 1
    and eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'mobile', 'id')
  order by eds_status_datetime desc, msg_datetime desc, msg_id desc
  rows 1
  into curr_phone_number, text2, text3;

  if (curr_phone_number is not null) then
  begin
    res_text = 'Проверка проведена. Заявка подписана через мобильное приложение';
    critical_level = 0;

    select jso, lines_jsa
    from create_jso('Телефон,Время отправки СМС,Время подписания в мобильном приложении', '33,33,33')
    into res_add_jso, lines_jsa;

    line_jsa = json.arr_create();
    json.arr_set_arr(lines_jsa, -1, line_jsa);
    json.arr_set_s(line_jsa, -1, curr_phone_number);
    json.arr_set_s(line_jsa, -1, text2);
    json.arr_set_s(line_jsa, -1, text3);
  end

  antifrod_res = iif(critical_level = 0, 0, 1);
  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'Номер телефона. Повторное использование. Подписание электронных документов.';
  criteria_code = 'phone_first_use';
  res_add_jso = null;
  critical_level = 1;
  res_text = 'Клиент впервые обращается за займом и подписывает электронные документы с данным номером телефона';
  ma_id_tmp = null;

  select jso, lines_jsa
  from create_jso('#,Номер договора,Дата договора,Номер телефона', '25,25,25,25')
  into res_add_jso, lines_jsa;

  line_jsa = json.arr_create();
  json.arr_set_arr(lines_jsa, -1, line_jsa);
  json.arr_set_s(line_jsa, -1, 'Текущий заем');
  json.arr_set_s(line_jsa, -1, k_kontract_no);
  json.arr_set_s(line_jsa, -1, f_datetostr(k_kontract_date));
  json.arr_set_s(line_jsa, -1, curr_phone_number);

  -- найдем предыдущий выданный займ
  select k_kontract_no, f_datetostr(k_kontract_date), k_from_ma_app_id
  from kontract, kontstate, knclient
  where ks_from_k_id = k_id
    and kc_from_k_id = k_id
    and kc_from_c_id = :c_id
    and ks_state = 1
    and ks_date_begin < :k_disb_date
    and ks_ol_is_delete = 0
  order by ks_date_begin desc
  rows 1
  into text1, text2, ma_id_tmp;

  if (ma_id_tmp is not null) then
  begin
    text3 = '';

    select f_cont_code(edm_from_clt_id)
    from edoc_list, edoc_sign, edoc_blank, edoc_msg edm, fsrobot_msgbox
    where eds_from_edl_id = edl_id
      and edb_from_edl_id = edl_id
      and edm_from_eds_id = eds_id
      and msg_id = edm_from_msg_id
      and edl_from_edc_id = f_edoc_kind('loan.application','id')
      and edl_object_kind = 11
      and edl_object_id = :ma_id_tmp
      and edl_status = 1
      and edm_status = 1
      and eds_status = 1
      and eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'mobile', 'id')
    order by eds_status_datetime desc, msg_datetime desc, msg_id desc
    rows 1
    into text3;
  
    line_jsa = json.arr_create();
    json.arr_set_arr(lines_jsa, -1, line_jsa);
    json.arr_set_s(line_jsa, -1, 'Предыдущий заем');
    json.arr_set_s(line_jsa, -1, text1);
    json.arr_set_s(line_jsa, -1, text2);
    json.arr_set_s(line_jsa, -1, text3);

    if (replace(curr_phone_number, ' ', '') is not distinct from replace(text3, ' ', '')) then
    begin
      res_text = 'Клиент предыдущий заём получал и подписывал электронные документы с использованием этого же телефона';
      critical_level = 0;
    end
  end

  antifrod_res = iif(critical_level = 0, 0, 1);
  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'Проведено взаимодействие с клиентом с использованием средств видеосвязи.';
  criteria_code = 'videocertification';
  res_add_jso = null;
  res_text = null;
  text1 = null;

  select f_datetimetostr(cds_create_datetime)
  from clientdocs, passkind, passdoctype
  where pk_id = cds_from_pk_id
    and pdt_id = pk_from_pdt_id
    and cds_from_c_id = :c_id
    and cds_create_datetime between :app_sign_datetime and :calc_datetime
    and pdt_code = 'video_sertif'
  order by cds_id desc
  rows 1
  into text1;

  res_text = iif(text1 is null, 'Проверка не проведена', res_text);
  critical_level = iif(text1 is null, 1, 0);
  antifrod_res = iif(text1 is null, 1, 0);
  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'БКИ. ОПРОС.';
  criteria_code = 'bki_survey';
  res_add_jso = null;
  antifrod_res = null;

  select iif(knd_value = 1, 0, 1)
  from knclient_data knd
  where knd_from_kc_id = :kc_id
    and knd_from_el_id = f_enum_id('knclient_data', 'bki_survey_res')
  order by knd_id desc
  rows 1
  into antifrod_res;

  if (antifrod_res is null) then
  begin
    res_text = 'Не пройден';
    critical_level = 0; -- на этапе КК еще опрос не пройден
    antifrod_res = 1;
  end
  else
  begin
    res_text = iif(antifrod_res = 0, 'Пройден положительно', 'Пройден отрицательно');
    critical_level = antifrod_res;
  end

  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'Вход в мобильное приложение.';
  criteria_code = 'phone_entry';
  res_add_jso = null;
  critical_level = 1;
  text1 = null;

  select list(distinct mcl_login, ', ')
  from mc_login
  where mcl_from_c_id = :c_id
    and mcl_is_delete = 0
    and mcl_datetime < :calc_datetime
  into text1;

  res_text = 'Номер не зарегистрирован в базе данных компании';

  -- есть ли среди номеров клиента
  if (exists(select 1
            from contdata cda, contlog clg
            where clg_from_cda_id = cda_id
              and cda_from_c_id = :c_id
              and cda_cont_kind = f_enum('cont_kind', 'cell', 'id')
              and cda_status = 1
              and clg_datetime < :calc_datetime
              and :text1 containing trim(f_cont_code(cda_from_clt_id))))
  then
  begin
    critical_level = 0;
    res_text = 'Номер зарегистрирован в базе данных компании';
  end

  res_text = f_format('%1 (%2)', res_text, text1);
  antifrod_res = iif(critical_level = 0, 0, 1);
  suspend;
  ------------------------------------------------------------------------------------------------------------------

  criteria = 'Устройство в Стоп-листе Компании.';
  criteria_code = 'device_blacklist';
  res_add_jso = null;
  critical_level = 0;
  res_text = 'Не найдено в Стоп-листе';
-- Этого пока нет, по умолчанию – 0 (позднее будет отдельная задача по сбору информации об этих устройствах, потом будем сюда тянуть)

  antifrod_res = iif(critical_level = 0, 0, 1);
  suspend;
  ------------------------------------------------------------------------------------------------------------------
end
