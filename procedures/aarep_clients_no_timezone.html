CREATE PROCEDURE AAREP_CLIENTS_NO_TIMEZONE(
    REP_DATE D_DATE NOT NULL)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    K_FACT_DISB_DATE D_DATE,
    DELINQ_DAYS D_INTEGER,
    C_NAME_FIO D_STR_100)
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable pl_id id_type;
declare variable reg_timezone d_smallint;
begin
  for select ks.ks_from_k_id,
             kc.kc_from_c_id
      from kontstate as ks
      join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id
      where ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_ol_is_delete = 0
      into :k_id,
           :c_id
  do
  begin
    delinq_days = 0;
    select iif((:rep_date - oks.oks_delinq_date < 0), 0, :rep_date - oks.oks_delinq_date) -- число дней активной просрочки
    from oksummary oks
    where :rep_date between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_from_k_id = :k_id and
          oks_close_state = 0
    into :delinq_days;
    if (delinq_days is null) then
      delinq_days = 0;

    pl_id = null;
    select adl_from_pl_id
    from addrclient, addrlist
    where adl_id = adc_from_adl_id and
          adc_from_c_id = :c_id and
          adc_address_type = 1
    into pl_id;

    reg_timezone = null;
    if (pl_id is not null) then
      select timezone
      from hp_places_get_timezone_pl(:pl_id)
      into reg_timezone;

    if (reg_timezone is null) then


    begin
      select k.k_kontract_no,
             k.k_fact_disb_date,
             c.c_name_fio
      from kontract k
      join knclient kc on k.k_id = kc.kc_from_k_id
      join clients c on c.c_id = kc.kc_from_c_id
      where k.k_id = :k_id and
            kc.kc_amount_approved > 0 and
            c.c_id = :c_id
      into :k_kontract_no,
           :k_fact_disb_date,
           :c_name_fio;

      suspend;
    end
  end
end
