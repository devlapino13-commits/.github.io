CREATE PROCEDURE AAREP_CALC_PERC_ONE_IPF(
    KONT_ID D_ID,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    RESULT_CALCED D_CURRENCY,
    RESULT_PAYED D_CURRENCY,
    RESULT_WRITEOFF D_CURRENCY,
    RESULT_PERC_BEGIN D_CURRENCY,
    RESULT_PERC_END D_CURRENCY,
    RESULT_BALANCE_END D_CURRENCY,
    RESULT_CALCED_ALL D_CURRENCY,
    RESULT_PAYED_ALL D_CURRENCY,
    RESULT_WRITEOFF_ALL D_CURRENCY,
    RESULT_DISB_AMOUNT D_CURRENCY,
    RESULT_DISB_DATE D_DATE,
    RESULT_OUT_PERC_END D_CURRENCY,
    RESULT_OUT_CALCED D_CURRENCY)
AS
declare variable k_fix_percent_amount d_currency;
declare variable prv_ol_date d_date;
declare variable pr_percent_kind d_smallint;
declare variable pr_is_360 d_smallint;
declare variable ok_ol_date d_date;
declare variable last_sh_date d_date;
declare variable fix_perc_saldo d_double;
declare variable dalta_day d_integer;
declare variable perc_per_day d_double;
declare variable tmp_day d_integer;
declare variable perc_amount d_currency;
declare variable tmp_repay d_currency;
declare variable tmp_off d_currency;
declare variable tmp_repay_all d_currency;
declare variable tmp_off_all d_currency;
declare variable delta_day_period d_integer;
declare variable tmp_repay_begin d_currency;
declare variable tmp_os d_currency;
declare variable total_repay_begin d_currency;
declare variable sh_prolong d_smallint;
declare variable next_day d_smallint;
declare variable ks_date_begin d_date;
declare variable ks_date_end d_date;
declare variable pr_round_perc d_smallint;
declare variable sh_perc_begin d_currency;
declare variable sh_perc_end d_currency;
declare variable tmp_perc_begin d_currency;
declare variable tmp_perc_end d_currency;
declare variable tmp_sh_date_begin d_date;
declare variable tmp_sh_date_end d_date;
declare variable sh_date_begin d_date;
declare variable sh_date_end d_date;
declare variable tmp_sh_diff d_currency;
declare variable sh_all_diff_perc_begin d_currency;
declare variable sh_all_diff_perc_end d_currency;
declare variable ipf_new_mode_begin d_bool;
declare variable ipf_new_mode_end d_bool;
begin
  result_calced = 0;
  result_payed = 0;
  result_writeoff = 0;
  result_perc_begin = 0;
  result_perc_end = 0;
  result_balance_end = 0;
  result_calced_all = 0;
  result_payed_all = 0;
  result_writeoff_all = 0;
  result_disb_amount = 0;
  result_out_perc_end = 0;
  result_out_calced = 0;
  total_repay_begin = 0;

  sh_perc_begin = 0;
  sh_perc_end = 0;

  select k_amount_approved, k_disb_date, k_fix_percent_amount, pr_percent_kind, pr_is_360, pr_round_perc
  from kontract, product
  where k_id = :kont_id and
        k_from_pr_id = pr_id
  into result_disb_amount, result_disb_date, k_fix_percent_amount, pr_percent_kind, pr_is_360, pr_round_perc;

  if (1 <> 1 and pr_percent_kind <> 2) then
    exception e_default 'Accrual method does not match: pr_percent_kind <> 2';

  ipf_new_mode_begin = f_ipf_newrepmode(date_begin - 1, kont_id, 0);
  ipf_new_mode_end = f_ipf_newrepmode(date_end, kont_id, 0);

  if (k_fix_percent_amount is null) then
    k_fix_percent_amount = 0;

  fix_perc_saldo = k_fix_percent_amount;
  next_day = 0;

  sh_all_diff_perc_end = 0;
  sh_all_diff_perc_begin = 0;
  prv_ol_date = null;
  for select ok_ol_date, last_sh_date, sh_prolong, ks_date_begin, ks_date_end
  from aarep_calc_perc_one_ipf_tmp(:kont_id, :date_begin, :date_end)
  order by ok_ol_date -- RFCKG-5282
  into :ok_ol_date, :last_sh_date, :sh_prolong, ks_date_begin, ks_date_end
  do
  begin
    if (prv_ol_date is not null) then
    begin
      dalta_day = f_agein360(prv_ol_date, minvalue(ok_ol_date, last_sh_date), pr_is_360);
      if (dalta_day < 0) then
        dalta_day = 0;

      tmp_repay = 0;
      tmp_off = 0;
      tmp_repay_all = 0;
      tmp_off_all = 0;
      tmp_os = 0;
      tmp_repay_begin = 0;
      tmp_perc_begin = 0;
      tmp_perc_end = 0;
      tmp_sh_diff = 0;

      select sum(iif(ok_type = 4, -ok_perc, 0)),
             sum(iif(ok_type in (5, 6), -ok_perc, 0)),
             sum(ok_os),
             sum(iif(ok_ol_date between :date_begin and :date_end and ok_type = 4, -ok_perc, 0)),
             sum(iif(ok_ol_date between :date_begin and :date_end and ok_type in (5, 6), -ok_perc, 0)),
             sum(iif(ok_ol_date < :date_begin and ok_type in (4, 5, 6), -ok_perc, 0)),
             sum(iif(ok_ol_date < :date_begin and ok_type <= 3 and ok_sh_date < :date_begin, ok_sh_perc, 0)),
             sum(iif(ok_type <= 3 and ok_sh_date <= :date_end, ok_sh_perc, 0)),
             max(iif(ok_ol_date < :date_begin and ok_type <= 3 and ok_sh_date < :date_begin, ok_sh_date, null)),
             max(iif(ok_type <= 3 and ok_sh_date <= :date_end, ok_sh_date, null)),
             sum(iif(ok_type = 3, ok_sh_perc, 0))
      from operkred
      where ok_from_k_id = :kont_id and
            ok_ol_is_active = 1 and
            ok_ol_date between (:prv_ol_date + :next_day) and :ok_ol_date and
            ok_type <= 8
      into tmp_repay_all, tmp_off_all, tmp_os, tmp_repay, tmp_off, tmp_repay_begin,
           tmp_perc_begin, tmp_perc_end, tmp_sh_date_begin, tmp_sh_date_end, tmp_sh_diff;

      if (tmp_repay is null) then
        tmp_repay = 0;

      if (tmp_off is null) then
        tmp_off = 0;

      if (tmp_repay_all is null) then
        tmp_repay_all = 0;

      if (tmp_off_all is null) then
        tmp_off_all = 0;

      if (tmp_repay_begin is null) then
        tmp_repay_begin = 0;

      if (tmp_os is null) then
        tmp_os = 0;

      if (tmp_perc_begin is null) then
        tmp_perc_begin = 0;

      if (tmp_perc_end is null) then
        tmp_perc_end = 0;

      if (tmp_sh_diff is null) then
        tmp_sh_diff = 0;

      if (sh_date_begin is null or sh_date_begin < tmp_sh_date_begin) then
        sh_date_begin = tmp_sh_date_begin;

      if (sh_date_end is null or sh_date_end < tmp_sh_date_end) then
        sh_date_end = tmp_sh_date_end;

      sh_perc_begin = sh_perc_begin + tmp_perc_begin;
      sh_perc_end = sh_perc_end + tmp_perc_end;

      result_balance_end = result_balance_end + tmp_os;

      delta_day_period = f_agein360(prv_ol_date,
                                    minvalue(minvalue(ok_ol_date, date_begin - 1), last_sh_date), pr_is_360);
      if (delta_day_period < 0) then
        delta_day_period = 0;

      perc_amount = round(perc_per_day * delta_day_period, pr_round_perc);
      result_perc_begin = result_perc_begin + perc_amount;
      total_repay_begin = total_repay_begin + tmp_repay_begin;

      if (result_perc_begin > (k_fix_percent_amount + sh_all_diff_perc_begin)) then
        result_perc_begin = (k_fix_percent_amount + sh_all_diff_perc_begin);

      result_payed = result_payed + tmp_repay;
      result_writeoff = result_writeoff + tmp_off;

      result_payed_all = result_payed_all + tmp_repay_all;
      result_writeoff_all = result_writeoff_all + tmp_off_all;

      if (ipf_new_mode_begin = 1 and ok_ol_date < date_begin) then
        sh_all_diff_perc_begin = sh_all_diff_perc_begin + tmp_sh_diff;

      if (ipf_new_mode_end = 1) then
        sh_all_diff_perc_end = sh_all_diff_perc_end + tmp_sh_diff;

      perc_amount = round(perc_per_day * dalta_day, pr_round_perc);
      result_calced_all = result_calced_all + perc_amount;
      if (result_calced_all > (k_fix_percent_amount + sh_all_diff_perc_end)) then
        result_calced_all = (k_fix_percent_amount + sh_all_diff_perc_end);

      next_day = 1;
    end

    prv_ol_date = ok_ol_date;

    if (sh_prolong = 1) then
    begin
      fix_perc_saldo = k_fix_percent_amount + sh_all_diff_perc_end - result_calced_all;
      if (fix_perc_saldo < 0) then
        fix_perc_saldo = 0;
    end

    perc_per_day = 0;
    tmp_day = f_agein360(ok_ol_date, last_sh_date, pr_is_360);
    if (tmp_day < 0) then
      tmp_day = 0;

    if (tmp_day > 0) then
      perc_per_day = fix_perc_saldo / tmp_day;
  end

  if (result_perc_begin - total_repay_begin <= 5) then
    result_perc_begin = total_repay_begin;

  if (sh_date_begin = (date_begin - 1) and abs(result_perc_begin - sh_perc_begin) <= 5) then
    result_perc_begin = sh_perc_begin;

  if (result_perc_begin < 0) then
    result_perc_begin = 0;

  if (date_begin < ks_date_begin) then
    result_perc_begin = 0;

  if (date_begin >= ks_date_end) then
    result_perc_begin = k_fix_percent_amount;

  if (date_end < ks_date_begin) then
  begin
    result_calced_all = 0;
    result_payed_all = 0;
    result_writeoff_all = 0;
  end

  if (result_calced_all - result_payed_all - result_writeoff_all <= 5) then
    result_calced_all = result_payed_all + result_writeoff_all;

  if (sh_date_end = date_end and abs(result_calced_all - sh_perc_end) <= 5) then
    result_calced_all = sh_perc_end;

  if (date_end >= ks_date_end) then
    result_calced_all = k_fix_percent_amount + sh_all_diff_perc_end;

  result_calced = result_calced_all - result_perc_begin;
  result_perc_begin = result_perc_begin - total_repay_begin;
  result_perc_end = result_calced_all - result_payed_all - result_writeoff_all;
  suspend;
end
