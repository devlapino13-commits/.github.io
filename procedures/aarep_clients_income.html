CREATE PROCEDURE AAREP_CLIENTS_INCOME(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    C_INN VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    OVS_VALUE VARCHAR(255))
AS
declare variable C_ID ID_TYPE;
declare variable OKS_KC_ID ID_TYPE;
declare variable OKS_REGION_ID integer;
declare variable C_NAME_FIO varchar(100);
declare variable K_ID ID_TYPE;
declare variable REG_ID_PRIOR integer;
declare variable OKS_KC_ID_PRIOR ID_TYPE;
begin
  reg_id_prior = 0;
  oks_kc_id_prior = 0;
  for select oks_kc_id, oks_region_id, oks_from_k_id
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_region_id = id_list
            and oks_close_state = 0 
      order by oks_region_id, oks_kc_id
      into oks_kc_id, oks_region_id, k_id
  do
  begin
    if (reg_id_prior <> oks_region_id) then
    begin
      reg_id_prior = oks_region_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :oks_region_id
      into reg_name;
    end
    if (oks_kc_id_prior <> oks_kc_id) then
    begin
      oks_kc_id_prior = oks_kc_id;
      select c_name_fio
      from clients
      where c_id = :oks_kc_id
      into officer_name;
    end
    select k_kontract_no
    from kontract
    where k_id = :k_id
    into k_kontract_no;

    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id
    do
    begin

      select c_name_fio, c_inn
      from clients
      where c_id = :c_id
      into c_name_fio, c_inn;
      if (c_inn is null) then
        c_inn = c_name_fio;

      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'work_monthly_income', :rep_date_end)
      into ovs_value;
      suspend;
    end
  end
end
