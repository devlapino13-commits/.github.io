CREATE PROCEDURE AAREP_NON_RF_DISB_LOANS(
    REP_REGION_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    KONTRACT_NO VARCHAR(20),
    KC_AMOUNT_APPROVED CURR_TYPE,
    DISB_DATE DATE,
    CLIENT_FIO VARCHAR(100),
    SVZ_FIO VARCHAR(100),
    SVZ_POS VARCHAR(255),
    CONF_FIO VARCHAR(100),
    CONF_POS VARCHAR(255),
    CASHIER_FIO VARCHAR(100),
    CASHIER_POS VARCHAR(100))
AS
declare variable k_id decimal(18,0);
declare variable conf_id decimal(18,0);
begin
  for select kont.k_id, kont.k_kontract_no, kncl.kc_amount_approved, kont.k_disb_date, cl.c_name_fio as client_fio,
             cl2.c_name_fio as svz_fio
      from kontract kont, kontstate ks, knclient kncl, clients cl, clients cl2, region_list(:rep_region_id, 1, -1, 1) rl
      where kont.k_id = ks.ks_from_k_id and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            kont.k_disb_date = :rep_date and
            kncl.kc_from_k_id = kont.k_id and
            kncl.kc_amount_approved > 0 and
            cl.c_id = kncl.kc_from_c_id and
            cl2.c_id = kont.k_fact_disb_lo_id and
            kncl.kc_amount_approved > 0 and
            kont.k_last_reg_id = rl.id_list and
            cl.c_citizen <> 13 /*13 - РФ*/
      into :k_id, :kontract_no, :kc_amount_approved, :disb_date, :client_fio, :svz_fio
  do
  begin
    svz_pos = 'Специалист по выдаче займов';

    select distinct cl.c_name_fio
    from operkred ok
    inner join operlist ol on ok.ok_from_ol_id = ol.ol_id
    inner join username un on un.un_id = ol.ol_user_id
    inner join clients cl on cl.c_id = un.un_from_c_id
    where ok.ok_from_k_id = :k_id and
          ol.is_delete = 0 and
          ok.ok_type = 8 and
          ok.ok_from_c_id = 0
    into :cashier_fio;

    cashier_pos = 'Кассир-оператор';

    select cl.c_id, cl.c_name_fio
    from kontlog kl, username un, clients cl
    where kl.kl_from_k_id = :k_id and
          kl.kl_state = 1 and
          kl.kl_user_id = un.un_id and
          cl.c_id = un.un_from_c_id
    order by kl.kl_datetime desc
    rows 1
    into :conf_id, :conf_fio;

    select jl.jl_name
    from clients cl
    inner join hr_pstatus hrps on hrps.ps_from_c_id = cl.c_id
    inner join joblist jl on jl.jl_id = hrps.ps_from_jl_id
    where cl.c_id = :conf_id and
          :rep_date between hrps.ps_date_begin and hrps.ps_date_end
    into :conf_pos;

    suspend;
  end
end
