CREATE PROCEDURE AAREP_DELIQ_PERIOD(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_KOLICH_REPAY INTEGER,
    PR_NAME VARCHAR(50),
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    KAL_NEW_KF_NAME VARCHAR(50),
    DELIQ_OS_BEGIN DECIMAL(18,4),
    DELIQ_PERC_BEGIN DECIMAL(18,4),
    DELIQ_DAY_BEGIN INTEGER,
    DELIQ_BALANCE_BEGIN DECIMAL(18,4),
    DELIQ_OS_END DECIMAL(18,4),
    DELIQ_PERC_END DECIMAL(18,4),
    DELIQ_DAY_END INTEGER,
    DELIQ_BALANCE_END DECIMAL(18,4),
    PR_LOAN_TYPE_STR VARCHAR(10),
    DISCHARGE_OS DECIMAL(18,4),
    DISCHARGE_PERC DECIMAL(18,4),
    DISCHARGE_FINE DECIMAL(18,4),
    WRITEOFF_OS DECIMAL(18,4),
    WRITEOFF_PERC DECIMAL(18,4),
    WRITEOFF_FINE DECIMAL(18,4))
AS
declare variable ok_from_k_id decimal(18,0);
declare variable ok_sh_date date;
declare variable ok_os_deliq decimal(18,4);
declare variable ok_perc_deliq decimal(18,4);
declare variable ok_sh_date_min date;
declare variable is_deliq smallint; /* 0 - неактивный, 1 - активный */
declare variable deliq_os decimal(18,4);
declare variable deliq_perc decimal(18,4);
declare variable deliq_day integer;
declare variable rep_active_balance_1 decimal(18,4);
declare variable rep_active_balance decimal(18,4);
declare variable pr_loan_type smallint;
declare variable is_deliq1 smallint;
declare variable ks_date_begin date;
declare variable ks_date_end date;
declare variable parent_id integer;
declare variable reg_prior integer;
begin
  reg_prior = 0;
  rep_date_begin = :rep_date_begin - 1;
  for select k_id, k_kontract_no, k_disb_date, k_amount_approved, k_circle, k_circle_internal, k_kolich_repay, pr_name,
             pr_loan_type, reg_code || ' ' || reg_name, parent_name, kal_new_kf_name, ks_date_begin, ks_date_end

      from kontstate, kontract, product, knaktlist, region_list(:rep_region_id, 1, 3, 1), region
      where cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end >= :rep_date_begin and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            k_id = ks_from_k_id and
            k_from_pr_id = pr_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            (kal_new_reg_id = id_list) and
            reg_unicode = id_list

            /*and k_id=109010004070400101*/

      into :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :k_circle, :k_circle_internal, :k_kolich_repay,
           :pr_name, :pr_loan_type, :reg_name, :parent_name, :kal_new_kf_name, :ks_date_begin, :ks_date_end

  do
  begin
    if (parent_id <> reg_prior) then
    begin
      select reg_name
      from region
      where reg_unicode = :parent_id
      into :parent_name;
      reg_prior = parent_id;
    end

    ok_sh_date = null;
    ok_os_deliq = null;
    ok_perc_deliq = null;

    deliq_os = 0;
    deliq_perc = 0;
    deliq_day = 0;

    ok_sh_date_min = :rep_date_begin;
    is_deliq = 0;

    deliq_os_begin = 0;
    deliq_perc_begin = 0;
    deliq_day_begin = 0;
    deliq_balance_begin = 0;
    is_deliq1 = 0;

    for /*Просрочки на первую дату*/
        select ok_from_k_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_ol_date <= :rep_date_begin and
              ok_sh_date <= :rep_date_begin and
              ok_from_k_id = :k_id and
              ok_from_c_id = 0
        group by ok_from_k_id, ok_sh_date
        having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
        into :ok_from_k_id, :ok_sh_date, :ok_os_deliq, :ok_perc_deliq

    do
    begin
      if (:ok_os_deliq + :ok_perc_deliq > 0) then
      begin
        is_deliq = 1;
        if (ok_sh_date_min > :ok_sh_date) then
          ok_sh_date_min = :ok_sh_date;
      end
      deliq_os = deliq_os + ok_os_deliq;
      deliq_perc = deliq_perc + ok_perc_deliq;
    end

    if (is_deliq = 1) then
    begin
      deliq_day = cast((:rep_date_begin - ok_sh_date_min) as integer);

      rep_active_balance = 0;
      rep_active_balance_1 = 0;

      for select sum(
                 case
                   when :rep_currency = '000' then (case
                                                      when cr_rate < k_index_curr_rate then ok_os * k_index_curr_rate /*Вычисляет активный портфель текущего сотрудника*/
                                                      else ok_os * cr_rate
                                                    end)
                   when :rep_currency = '---' then ok_os
                   when :rep_currency = k_index_curr_code then ok_os
                 end)
          from operkred, kontract, opercurr, kontstate, knaktlist
          where ok_ol_is_active = 1 and
                ok_from_k_id = k_id and
                k_index_curr_code = cr_from_cu_id and
                ok_ol_date <= :rep_date_begin and
                :rep_date_begin between cr_date and cr_date_end and
                ok_type not in (1, 2, 3, 9, 10) and
                kontstate.ks_ol_is_delete = 0 and
                ks_from_k_id = k_id and
                :rep_date_begin between ks_date_begin and ks_date_end and
                ks_state = 1 and
                kal_from_kf_id = k_from_kf_id and
                :rep_date_begin between kal_date and kal_date_end and
                ok_from_c_id = 0 and
                ((:rep_currency = '---') or
                (:rep_currency = '000') or
                (:rep_currency = k_index_curr_code)) and
                knaktlist.is_delete = 0

                and
                ok_from_k_id = :k_id

                and
                ok_os <> 0
          into :rep_active_balance_1
      do
      begin
        rep_active_balance = rep_active_balance_1;
      end

      deliq_os_begin = deliq_os;
      deliq_perc_begin = deliq_perc;
      deliq_day_begin = deliq_day;
      deliq_balance_begin = rep_active_balance;
      is_deliq1 = is_deliq;
    end

    ok_sh_date = null;
    ok_os_deliq = null;
    ok_perc_deliq = null;

    deliq_os = 0;
    deliq_perc = 0;
    deliq_day = 0;

    ok_sh_date_min = :rep_date_end;
    is_deliq = 0;

    deliq_os_end = 0;
    deliq_perc_end = 0;
    deliq_day_end = 0;
    deliq_balance_end = 0;

    for /*Просрочки на вторую дату */
        select ok_from_k_id, ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_ol_date <= :rep_date_end and
              ok_sh_date <= :rep_date_end and
              ok_from_k_id = :k_id and
              :rep_date_end between :ks_date_begin and :ks_date_end and
              ok_from_c_id = 0
        group by ok_from_k_id, ok_sh_date
        having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) > 0
        into :ok_from_k_id, :ok_sh_date, :ok_os_deliq, :ok_perc_deliq

    do
    begin
      if (:ok_os_deliq + :ok_perc_deliq > 0) then
      begin
        is_deliq = 1;
        if (ok_sh_date_min > :ok_sh_date) then
          ok_sh_date_min = :ok_sh_date;
      end
      deliq_os = deliq_os + ok_os_deliq;
      deliq_perc = deliq_perc + ok_perc_deliq;
    end

    if (is_deliq = 1) then
    begin
      deliq_day = cast((:rep_date_end - ok_sh_date_min) as integer);

      rep_active_balance = 0;
      rep_active_balance_1 = 0;

      for select sum(
                 case
                   when :rep_currency = '000' then (case
                                                      when cr_rate < k_index_curr_rate then ok_os * k_index_curr_rate /*Вычисляет активный портфель текущего сотрудника*/
                                                      else ok_os * cr_rate
                                                    end)
                   when :rep_currency = '---' then ok_os
                   when :rep_currency = k_index_curr_code then ok_os
                 end)
          from operkred, kontract, opercurr, kontstate, knaktlist
          where ok_ol_is_active = 1 and
                ok_from_k_id = k_id and
                k_index_curr_code = cr_from_cu_id and
                ok_ol_date <= :rep_date_end and
                :rep_date_end between cr_date and cr_date_end and
                ok_type not in (1, 2, 3, 9, 10) and
                kontstate.ks_ol_is_delete = 0 and
                ks_from_k_id = k_id and
                :rep_date_end between ks_date_begin and ks_date_end and
                ks_state = 1 and
                kal_from_kf_id = k_from_kf_id and
                :rep_date_end between kal_date and kal_date_end and
                ok_from_c_id = 0 and
                ((:rep_currency = '---') or
                (:rep_currency = '000') or
                (:rep_currency = k_index_curr_code)) and
                knaktlist.is_delete = 0

                and
                ok_from_k_id = :k_id

                and
                ok_os <> 0
          into :rep_active_balance_1
      do
      begin
        rep_active_balance = rep_active_balance_1;
      end

      deliq_os_end = deliq_os;
      deliq_perc_end = deliq_perc;
      deliq_day_end = deliq_day;
      deliq_balance_end = rep_active_balance;
      is_deliq1 = is_deliq;
    end

    /*Вычисляет между двумя периодами движении по просроченным кредитам*/
    discharge_os = 0;
    discharge_perc = 0;
    discharge_fine = 0;
    writeoff_os = 0;
    writeoff_perc = 0;
    writeoff_fine = 0;
    if (deliq_os_begin + deliq_perc_begin + deliq_day_begin + deliq_balance_begin <> deliq_os_end + deliq_perc_end + deliq_day_end + deliq_balance_end) then
    begin
      select coalesce(sum(
             case
               when ok_type in (4, 9) then ok_os
             end), 0) discharge_os, coalesce(sum(
             case
               when ok_type in (4, 9) then ok_perc
             end), 0) discharge_perc, coalesce(sum(
             case
               when ok_type in (4, 9) then ok_fine
             end), 0) discharge_fine, coalesce(sum(
             case
               when ok_type in (5, 6, 10) then ok_os
             end), 0) writeoff_os, coalesce(sum(
             case
               when ok_type in (5, 6, 10) then ok_perc
             end), 0) writeoff_perc, coalesce(sum(
             case
               when ok_type in (5, 6, 10) then ok_fine
             end), 0) writeoff_fine
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin + 1 and :rep_date_end and
            ok_from_k_id = :k_id and
            ok_type in (4, 5, 6, 9, 10) and
            ok_from_c_id = 0
      into :discharge_os, :discharge_perc, :discharge_fine, :writeoff_os, :writeoff_perc, :writeoff_fine;
    end

    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Групп.';
    else
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';
    if (is_deliq1 = 1) then
      suspend;
  end
end
