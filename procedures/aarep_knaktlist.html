CREATE PROCEDURE AAREP_KNAKTLIST(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE DATE,
    REPORT_MONTH_END DATE,
    REPORT_KIND INTEGER,
    REPORT_OFFICER_ID DECIMAL(18,0),
    REPORT_NEED_BALANCE INTEGER,
    REPORT_NEED_PERCENT INTEGER,
    REPORT_K_ID DECIMAL(18,0))
RETURNS (
    K_ID DECIMAL(18,0),
    K_DISB_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    K_FROM_KF_ID DECIMAL(18,0),
    KF_NAME VARCHAR(100),
    KF_OFFICER_ID DECIMAL(18,0),
    PR_NAME VARCHAR(50),
    CL_CASHIER_NAME VARCHAR(100),
    K_AMOUNT_BALANCE DECIMAL(18,4),
    K_AMOUNT_PERCENT DECIMAL(18,4),
    K_CURRENCY VARCHAR(3),
    PR_ID INTEGER,
    KS_STATE INTEGER,
    KF_FOLDER_NO INTEGER,
    K_ACC_BALANCE VARCHAR(20),
    K_ACC_PERCENT VARCHAR(20),
    K_PERIOD_OS DECIMAL(18,4),
    K_PERIOD_PERC DECIMAL(18,4),
    K_AMOUNT_DEPO DECIMAL(18,4),
    K_ACC_DEPO VARCHAR(20),
    K_ACC_REPAY_PERCENT VARCHAR(20),
    KAL_DATE DATE,
    K_DISB_LO DECIMAL(18,0))
AS
declare variable cash_id decimal(18,0);
declare variable perc_per_day decimal(18,7);
declare variable disb_amount decimal(18,4);
declare variable k_stop_perc date;
declare variable calc_method integer;
declare variable ks_date_begin date;
declare variable is_process integer;
declare variable is_active integer;
declare variable is_closed integer;
declare variable kf_id decimal(18,0);
declare variable kf_id_prior decimal(18,0);
declare variable k_fund varchar(20);
declare variable k_fund_prior varchar(20);
declare variable k_from_pr_id_prior integer;
declare variable st_staff integer;
declare variable st_client integer;
declare variable is_ok smallint;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  k_from_pr_id_prior = 0;
  k_fund_prior = '';
  kf_id_prior = 0;
  k_acc_balance = '';
  k_acc_percent = '';
  k_acc_depo = '';
  k_amount_percent = 0;
  k_amount_balance = 0;
  k_amount_depo = 0;
  k_period_os = 0;
  is_process = -1;
  is_active = -1;
  is_closed = -1;
  if (report_kind = 0) then
  begin
    select distinct k_from_kf_id
    from kontract
    where k_id = :report_k_id
    into :kf_id;
    if (kf_id is null) then
      kf_id = 0;
  end
  if (report_kind = 1) then
  begin
    is_process = 0;
  end
  if (report_kind = 2) then
  begin
    is_active = 1;
  end
  if (report_kind = 3) then
  begin
    is_process = 0;
    is_active = 1;
  end
  if (report_kind = 4) then
  begin
    is_closed = 2;
  end
  if (report_kind = 5) then
  begin
    is_process = 0;
    is_closed = 2;
  end
  if (report_kind = 6) then
  begin
    is_active = 1;
    is_closed = 2;
  end
  if (report_kind = 7) then
  begin
    is_process = 0;
    is_active = 1;
    is_closed = 2;
  end
  for select ks_from_k_id, k_disb_date, k_kontract_no, k_from_kf_id, kal_new_kf_name, kal_new_kf_officer, pr_id,
             pr_name, k_clients_cashier_id, k_percent_per_day, pr_interest_method, k_amount_approved, k_stop_perc,
             ks_date_begin, k_index_curr_code, ks_state, kal_new_no, k_from_f_id, kal_date, ks_kc_id
      from kontstate, kontract, knaktlist, product
      where ks_from_k_id = k_id and
            kal_from_kf_id = k_from_kf_id and
            cast(ks_date_begin as date) <= :report_date and
            ks_date_end >= :report_date and
            ks_ol_is_delete = 0 and
            (((:report_kind <> 0) and
            ((ks_state = :is_process) or
            (ks_state = :is_active) or
            (ks_state = :is_closed))) or
            ((:report_kind = 0) and
            (k_from_kf_id = :kf_id))) and
            kal_date <= :report_date and
            kal_date_end >= :report_date and
            kal_new_reg_id = :report_region_id and
            knaktlist.is_delete = 0 and
            k_from_pr_id = pr_id
      order by k_id desc
      into :k_id, :k_disb_date, :k_kontract_no, :k_from_kf_id, :kf_name, :kf_officer_id, :pr_id, :pr_name, :cash_id,
           :perc_per_day, :calc_method, :disb_amount, :k_stop_perc, :ks_date_begin, :k_currency, :ks_state,
           :kf_folder_no, :k_fund, :kal_date, :k_disb_lo
  do
  begin
    k_amount_balance = 0;
    k_amount_percent = 0;
    k_period_perc = 0;
    k_amount_depo = 0;

    is_ok = 1;

-- igusev 17.09.2015 добавил --- >>
    if (is_closed = 2 and
        is_active = -1) then
    begin
      if (exists(select *
                 from kontract kont
                 inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                 where ks.ks_ol_is_delete = 0 and
                       ks.ks_state = 1 and
                       kont.k_from_kf_id = :k_from_kf_id and
                       :report_date between ks.ks_date_begin and ks.ks_date_end)) then
        is_ok = 0;
    end
--- <<

    if (is_ok = 1) then
    begin
      if (kf_officer_id = :report_officer_id) then
      begin
        cl_cashier_name = '';
        if (cash_id is not null) then
        begin
          select c_name_fio
          from clients
          where c_id = :cash_id
          into :cl_cashier_name;
        end
        if ((ks_state = 1) or ((ks_state = 2) and
            (ks_date_begin >= :report_month_end + 1) and
            (ks_date_begin <= :report_date))) then
        begin
          if ((report_need_percent = 1) or (report_need_balance = 1)) then
          begin
            select oks_balance
            from oksummary
            where oks_from_k_id = :k_id and
                  oks_date_begin <= :report_date and
                  oks_date_end >= :report_date and
                  oks_close_state = 0
            into :k_amount_balance;
            if (k_amount_balance is null) then
              k_amount_balance = 0;

            select sum(okd_summa_plus - okd_summa_minus)
            from operkreddepo, operlist
            where okd_from_ol_id = ol_id and
                  operlist.is_delete = 0 and
                  ol_date <= :report_date and
                  okd_from_k_id = :k_id
            into :k_amount_depo;
            if (k_amount_depo is null) then
              k_amount_depo = 0;
          end
        end
        if ((k_from_kf_id <> kf_id_prior) or (k_amount_depo <> 0) or (k_amount_percent <> 0) or (k_amount_balance <> 0) or (k_period_os <> 0) or (k_period_perc <> 0)) then
        begin
          if ((k_fund_prior <> k_fund) or (k_from_pr_id_prior <> pr_id) or (k_acc_depo = '') or (k_acc_repay_percent = '') or (k_acc_balance = '') or (k_acc_percent = '')) then
          begin
            if (report_need_balance = 1) then
            begin
              k_acc_balance = '';
              select res_account
              from hp_make_kredit_acclist(:pr_id, :report_region_id, 1) -- Кредитный портфель
              into :k_acc_balance;

            end
            if (report_need_percent = 1) then
            begin
              k_acc_percent = '';
              select res_account
              from hp_make_kredit_acclist(:pr_id, :report_region_id, 2) -- Начисленные проценты
              into :k_acc_percent;

              k_acc_repay_percent = '';
              select res_account
              from hp_make_kredit_acclist(:pr_id, :report_region_id, 3) -- Выплаченные проценты
              into :k_acc_repay_percent;
            end
            if (k_amount_depo <> 0) then
            begin
              k_acc_depo = '';
              select res_account
              from hp_make_kredit_acclist(:pr_id, :report_region_id, 3) -- Счет депонирования
              into :k_acc_depo;
            end
          end
          suspend;
        end
        kf_id_prior = k_from_kf_id;
        k_fund_prior = :k_fund;
        k_from_pr_id_prior = :pr_id;

      end
    end
  end

end
