CREATE PROCEDURE UXP$UX_LIST_USERGO(
    USER_NAME D_STR_CODE = null)
RETURNS (
    USG_ID D_INTEGER,
    USG_UA_UUID D_STR_CODE,
    USG_DATETIME D_DATETIME,
    USG_CAPTION D_STR_255,
    UA_IS_MAINMENU D_BOOL,
    UA_OWNER D_STR_CODE,
    UA_SOURCE D_STR_CODE,
    UA_UXA_UUID D_STR_CODE,
    UA_UXA_OWNER D_STR_CODE,
    UA_UXA_SOURCE D_STR_CODE,
    RES_FCODE D_STR_100,
    RES_KIND_CAPTION D_STR_CODE,
    RES_UUID_STR D_STR_CODE,
    RES_MEMO D_STR_LONG)
AS
declare variable uu_kind d_integer;
declare variable mm_uuid d_uuid;
begin
  if (coalesce(user_name, '') = '') then
    user_name = current_user;

  mm_uuid = uxf$mainmenu();

  for select usg_id, uuid_to_char(usg_ua_uuid), usg_datetime, uuid_to_char(ua_owner), uu_kind,
             uuid_to_char(ua_uxa_uuid), uuid_to_char(uxf$ver(uxa_owner, 'draft')), uuid_to_char(uxa.ux_us_uuid),
             iif(ua_owner = :mm_uuid,
                 ua_name || ' из Главного меню',
                 ua_name || ' из "' || own.ux_name || '"'),
             own.ux_fcode,
             uxa.ux_fcode,
             'Запускает "' || uxa.ux_name || '" (' || uxa.ux_fcode || ') из "' || own.ux_name || '" (' || own.ux_fcode || ')'
  from ux$usergo, ux$action, ux$code own, ux$uniq, ux$alias, ux$code uxa, ux$form ownf
  where usg_user = :user_name and
        usg_ua_uuid = ua_uuid and
        ua_owner = ownf.ufm_uuid and
        own.ux_uuid = ownf.ufm_ux_uuid and
        own.ux_uuid = uu_id and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = uxa.ux_uuid
  order by usg_id desc
  into usg_id, usg_ua_uuid, usg_datetime, ua_owner, uu_kind, ua_uxa_uuid, ua_uxa_owner, ua_uxa_source,
       usg_caption, res_fcode, res_uuid_str, res_memo
  do
  begin
    ua_source = null;

    ua_is_mainmenu = 0;
    if (ua_owner = mm_uuid) then
      ua_is_mainmenu = 1;

    res_kind_caption = uxf$kind(uu_kind, 'caption');

    if (uxf$kind(uu_kind, 'code') = 'uxframe') then
    begin
      select uuid_to_char(ux_us_uuid)
      from ux$frame, ux$form, ux$code
      where ufr_uuid = char_to_uuid(:ua_owner) and
            ufr_owner = ufm_uuid and
            ufm_ux_uuid = ux_uuid
      into ua_source;
    end else
    begin
      select uuid_to_char(ux_us_uuid)
      from ux$form, ux$code
      where ufm_uuid = char_to_uuid(:ua_owner) and
            ufm_ux_uuid = ux_uuid
      into ua_source;

      if (ua_source is null) then
      begin
        select uuid_to_char(ux_us_uuid)
        from ux$code
        where ux_uuid = char_to_uuid(:ua_owner)
        into ua_source;
      end
    end

    if (ua_source is null) then
      exception e_default f_format('Unknown owner for usg_id: %1', usg_id);

    suspend;
  end
end
