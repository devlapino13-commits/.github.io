CREATE PROCEDURE OKT_IVR_SET_RESULT(
    PANT_ID D_ID NOT NULL,
    OBJECT_KIND D_INTEGER NOT NULL,
    OBJECT_ID D_ID NOT NULL,
    RESULT_CODE D_STR_100 NOT NULL,
    TIMEZONE D_INTEGER NOT NULL,
    DURATION D_INTEGER = 0)
RETURNS (
    RES D_INTEGER)
AS
declare variable HOURS_ADD D_SMALLINT;
declare variable RES_ID D_INTEGER;
declare variable NEXT_CALL D_DATETIME;
declare variable LOCAL_CALL_HOUR D_SMALLINT;
declare variable OIS_ID D_ID;
declare variable UNSUCCESSFUL_MORNING_QTY D_SMALLINT;
declare variable UNSUCCESSFUL_DINNER_QTY D_INTEGER;
declare variable UNSUCCESSFUL_EVENING_QTY D_INTEGER;
declare variable LAST_CALL_DATE D_DATE;
begin
  hours_add = timezone - 3;
  res = 0;
  result_code = 'IVRCallStatus.'||result_code;
  res_id = f_get_rd_id(result_code);

  unsuccessful_morning_qty = 0;
  unsuccessful_dinner_qty = 0;
  unsuccessful_evening_qty = 0;

  select cv.ois_id,
         ois_unsuccessful_morning_qty,
         ois_unsuccessful_dinner_qty,
         ois_unsuccessful_evening_qty,
         cast(cv.ois_last_call_datetime as date)
  from okt_ivr_calls_svod cv
  where cv.ois_from_pant_id = :pant_id
  and cv.ois_object_kind = :object_kind
  and cv.ois_object_id = :object_id
  into :ois_id,
       :unsuccessful_morning_qty,
       :unsuccessful_dinner_qty,
       :unsuccessful_evening_qty,
       :last_call_date;

  last_call_date =  coalesce(last_call_date, '01.01.1900');
  if (last_call_date < current_date) then
  begin
    update okt_ivr_calls_svod cv
    set cv.ois_unsuccessful_morning_qty = 0,
        cv.ois_unsuccessful_dinner_qty = 0,
        cv.ois_unsuccessful_evening_qty = 0
    where cv.ois_id = :ois_id;

    unsuccessful_morning_qty = 0;
    unsuccessful_dinner_qty = 0;
    unsuccessful_evening_qty = 0;
  end

  if (result_code = 'IVRCallStatus.Successful') then
    next_call = cast(cast(f_datetostr(current_date + 6) as d_str_10) || ' 09:00' as timestamp);
  else
  begin
    local_call_hour = extract(hour from current_timestamp) + hours_add;
    if (local_call_hour < 13) then -- если звонок был совешон до 13:00
    begin
        unsuccessful_morning_qty = unsuccessful_morning_qty + 1;
        if (unsuccessful_morning_qty < 3) then
            next_call = dateadd(hour, 1 + hours_add, current_timestamp);
        else
            next_call = cast(cast(f_datetostr(current_date)as d_str_10) || ' 13:00' as timestamp);
    end
    else if (local_call_hour between 13 and 16) then -- если звонок был совешон в промежутке между 13:00 и 16:59
    begin
        unsuccessful_dinner_qty = unsuccessful_dinner_qty + 1;
        if (unsuccessful_dinner_qty < 3) then
            next_call = dateadd(hour, 1 + hours_add, current_timestamp);
        else
            next_call = cast(cast(f_datetostr(current_date)as d_str_10) || ' 17:00' as timestamp);
    end
    else if (local_call_hour > 16) then -- если звонок был совешон c 17 и далее часов
    begin
        unsuccessful_evening_qty = unsuccessful_evening_qty + 1;
        if (unsuccessful_evening_qty < 3) then
            next_call = dateadd(hour, 1 + hours_add, current_timestamp);
        else
            next_call = cast(cast(f_datetostr(current_date + 6) as d_str_10) || ' 09:00' as timestamp);
    end
  end

  if (ois_id is null) then
  begin
     insert into okt_ivr_calls_svod (ois_object_kind,
                                     ois_object_id,
                                     ois_from_pant_id,
                                     ois_last_call_datetime,
                                     ois_last_result,
                                     ois_next_call_datetime,
                                     ois_timezone,
                                     ois_unsuccessful_morning_qty,
                                     ois_unsuccessful_dinner_qty,
                                     ois_unsuccessful_evening_qty)
     values (:object_kind,
             :object_id,
             :pant_id,
             current_timestamp,
             :res_id,
             :next_call,
             :timezone,
             :unsuccessful_morning_qty,
             :unsuccessful_dinner_qty,
             :unsuccessful_evening_qty)
     returning ois_id
     into :ois_id;
  end
  else
  begin
     update okt_ivr_calls_svod
     set ois_last_call_datetime = current_timestamp,
         ois_last_result = :res_id,
         ois_next_call_datetime = :next_call,
         ois_timezone = :timezone,
         ois_unsuccessful_morning_qty = :unsuccessful_morning_qty,
         ois_unsuccessful_dinner_qty = :unsuccessful_dinner_qty,
         ois_unsuccessful_evening_qty = :unsuccessful_evening_qty
     where (ois_id = :ois_id);
  end

  insert into okt_ivr_calls (oic_from_ois_id, oic_call_result, oic_duration)
  values (:ois_id, :res_id, :duration);

  suspend;
end
