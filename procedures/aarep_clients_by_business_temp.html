CREATE PROCEDURE AAREP_CLIENTS_BY_BUSINESS_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_AMOUNT_ALL_DISB DECIMAL(18,4),
    AAREP_AMOUNT_ALL_BALANCE DECIMAL(18,4),
    AAREP_KOLICH_ALL_DISB INTEGER,
    AAREP_KOLICH_ALL_BALANCE INTEGER,
    AAREP_AMOUNT_FEMAIL_DISB DECIMAL(18,4),
    AAREP_AMOUNT_FEMAIL_BALANCE DECIMAL(18,4),
    AAREP_KOLICH_FEMAIL_DISB INTEGER,
    AAREP_KOLICH_FEMAIL_BALANCE INTEGER,
    AAREP_BST_ID INTEGER)
AS
declare variable c_sex integer;
declare variable kont_amount decimal(18,4);
declare variable cl_amount decimal(18,4);
declare variable disb_amount decimal(18,4);
declare variable one_amount decimal(18,4);
declare variable ss_amount decimal(18,7);
begin
  aarep_amount_all_disb = 0;
  aarep_amount_all_balance = 0;
  aarep_kolich_all_disb = 0;
  aarep_kolich_all_balance = 0;
  aarep_amount_femail_disb = 0;
  aarep_amount_femail_balance = 0;
  aarep_kolich_femail_disb = 0;
  aarep_kolich_femail_balance = 0;
  for select bst_id
      from bussubtypetarget
      into :aarep_bst_id
  do
  begin
    suspend;
  end

  for select c_sex, kc_amount_approved, k_amount_approved, aarep_plan_os, bst_id
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 2), kontract, knclient, clients, bussubtypetarget, bussubtype
      where kc_busuness_subtype = bs_id and
            aarep_k_id = k_id and
            kc_from_c_id = c_id and
            kc_from_k_id = aarep_k_id and
            bs_is_active = bst_id and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null
      order by c_id
      into c_sex, :cl_amount, :kont_amount, :disb_amount, :aarep_bst_id
  do
  begin
    aarep_kolich_all_disb = aarep_kolich_all_disb + 1;
    if (c_sex = 0) then
      aarep_kolich_femail_disb = aarep_kolich_femail_disb + 1;
    if (kont_amount = cl_amount) then
      one_amount = disb_amount;
    else
    begin
      if (kont_amount <> 0) then
      begin
        begin
          ss_amount = cast(cl_amount / kont_amount as decimal(15,7));
          one_amount = cast(ss_amount * disb_amount as decimal(18,2));
        end
      end
      else
        one_amount = 0;
    end
    if (c_sex = 0) then
      aarep_amount_femail_disb = aarep_amount_femail_disb + one_amount;
    aarep_amount_all_disb = aarep_amount_all_disb + one_amount;
    suspend;
    aarep_amount_all_disb = 0;
    aarep_amount_all_balance = 0;
    aarep_kolich_all_disb = 0;
    aarep_kolich_all_balance = 0;
    aarep_amount_femail_disb = 0;
    aarep_amount_femail_balance = 0;
    aarep_kolich_femail_disb = 0;
    aarep_kolich_femail_balance = 0;
  end

  aarep_amount_all_disb = 0;
  aarep_amount_all_balance = 0;
  aarep_kolich_all_disb = 0;
  aarep_kolich_all_balance = 0;
  aarep_amount_femail_disb = 0;
  aarep_amount_femail_balance = 0;
  aarep_kolich_femail_disb = 0;
  aarep_kolich_femail_balance = 0;

  for select c_sex, kc_amount_approved, k_amount_approved, aarep_k_balance, bst_id
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 3), kontract, knclient, clients, bussubtypetarget, bussubtype
      where kc_busuness_subtype = bs_id and
            bs_is_active = bst_id and
            aarep_k_id = k_id and
            kc_from_c_id = c_id and
            kc_from_k_id = aarep_k_id and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null
      order by c_id
      into c_sex, :cl_amount, :kont_amount, :disb_amount, :aarep_bst_id
  do
  begin
    aarep_kolich_all_balance = aarep_kolich_all_balance + 1;
    if (c_sex = 0) then
      aarep_kolich_femail_balance = aarep_kolich_femail_balance + 1;
    if (kont_amount = cl_amount) then
      one_amount = disb_amount;
    else
    begin
      if (kont_amount <> 0) then
      begin
        begin
          ss_amount = cast(cl_amount / kont_amount as decimal(15,7));
          one_amount = cast(ss_amount * disb_amount as decimal(18,2));
        end
      end
      else
        one_amount = 0;
    end
    if (c_sex = 0) then
      aarep_amount_femail_balance = aarep_amount_femail_balance + one_amount;
    aarep_amount_all_balance = aarep_amount_all_balance + one_amount;
    suspend;
    aarep_amount_all_disb = 0;
    aarep_amount_all_balance = 0;
    aarep_kolich_all_disb = 0;
    aarep_kolich_all_balance = 0;
    aarep_amount_femail_disb = 0;
    aarep_amount_femail_balance = 0;
    aarep_kolich_femail_disb = 0;
    aarep_kolich_femail_balance = 0;
  end
end
