CREATE PROCEDURE UXP$UXCODE_INFO_SOURCES(
    IN_UFM_CODE D_STR_CODE)
RETURNS (
    FIS_ID D_STR_100,
    FIS_PARENT_ID D_STR_100,
    FIS_FR_UUID D_UUID,
    FIS_FR_UUID_CODE D_STR_CODE,
    FIS_CODE D_STR_CODE,
    FIS_CAPTION D_STR_100,
    FIS_MASTER_FIELD D_STR_100,
    FIS_DETAIL_FIELD D_STR_100,
    FIS_FRAME_KIND D_SMALLINT,
    FIS_US_SELECT_SQL D_STR_4096,
    FIS_US_KIND D_STR_CODE,
    FIS_US_SELECT D_STR_100,
    FIS_US_UPDATE D_STR_100,
    FIS_US_INSERT D_STR_100,
    FIS_US_DELETE D_STR_100)
AS
declare variable ufm_uuid d_uuid;
declare variable ufr_is_mainframe d_bool;
declare variable fr_is_grid d_bool;
declare variable us_kind d_smallint;
declare variable us_is_table d_smallint;
declare variable us_code d_str_100;
declare variable ufr_tree_parent_field d_str_code;
begin
  ufm_uuid = char_to_uuid(in_ufm_code);
  if (ufm_uuid is null) then
    exit;

  for select fr_uuid,
             uuid_to_char(master_fr_uuid),
             fr_code,
             ufr_name,
             ufr_is_mainframe,
             fr_master_field,
             fr_detail_field,
             fr_source,
             fr_is_grid,
             us_kind,
             us_is_table,
             us_sql,
             ufr_is_mainframe,
             fr.ufr_tree_parent_field
  from ux$frame fr, ux$source, uxp$uxframe_tree(:ufm_uuid)
  where fr_uuid = ufr_uuid and
        ufr_us_uuid = us_uuid
  union
  select ufr_uuid,
         uuid_to_char(ufr_master_ufr_uuid),
         ufr_code,
         ufr_name,
         ufr_is_mainframe,
         ufr_master_field,
         ufr_detail_field,
         us_code,
         ufr_is_grid,
         us_kind,
         us_is_table,
         us_sql,
         ufr_is_mainframe,
         ufr_tree_parent_field
  from ux$frame fr, ux$source
  where ufr_uuid = :ufm_uuid and
        ufr_us_uuid = us_uuid
  into fis_fr_uuid,
       fis_parent_id,
       fis_code,
       fis_caption,
       ufr_is_mainframe,
       fis_master_field,
       fis_detail_field,
       fis_us_select,
       fr_is_grid,
       us_kind,
       us_is_table,
       fis_us_select_sql,
       ufr_is_mainframe,
       ufr_tree_parent_field
  do
  begin
    fis_id = uuid_to_char(fis_fr_uuid);
    fis_fr_uuid_code = uuid_to_char(fis_fr_uuid);

    if (fis_us_select_sql is null) then
      fis_us_select_sql = '';

    if (ufr_is_mainframe = 1) then
    begin
      if (coalesce(fis_code, '') <> '') then
        fis_code = fis_code || '  (основной)';
      else
        fis_code =  '(основной)';
    end

    fis_us_kind = '';

    if (fr_is_grid = 0) then
      fis_frame_kind = 0;
    else
    begin
      fis_frame_kind = 1;

      if (ufr_tree_parent_field <> '') then
      begin
        if (exists(select * from ux$fritem
                   where ufi_ufr_uuid = :fis_fr_uuid and
                         ufi_control = 'tree')) then
          fis_frame_kind = 2;
      end
    end

    if (us_kind = 0) then
    begin
      if (us_is_table = 1) then
        fis_us_kind = 'table';
      else
        fis_us_kind = 'proc';
    end else
    if (us_kind = -1) then
    begin
      fis_us_kind = 'sql';
      fis_us_select = left(fis_us_select_sql, 97);
      if (char_length(fis_us_select) = 97) then
        fis_us_select = fis_us_select || '...';
    end else
    begin
      if (us_is_table = 1) then
        fis_us_kind = 'table';
      else
        fis_us_kind = 'proc';

      fis_us_kind = 'V'  || fis_us_kind;
    end

    fis_us_update = '';
    fis_us_insert = '';
    fis_us_delete = '';

    for select ufd_kind, us_code || trim(iif(us_is_table = 1, '', '()'))
    from ux$frdata, ux$source
    where ufd_ufr_uuid = :fis_fr_uuid and
          ufd_us_uuid = us_uuid
    order by ufd_kind
    into us_kind, us_code
    do
    begin
      if (us_kind = 1) then
        fis_us_update = us_code;
      else
      if (us_kind = 2) then
        fis_us_insert = us_code;
      else
      if (us_kind = 3) then
        fis_us_delete = us_code;
    end

    suspend;
  end
end
