CREATE PROCEDURE CM_RU_0222_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable adl_from_pl_id id_type;
declare variable c_id d_id;
declare variable parent id_type;
declare variable pl_id id_type;
declare variable pl_status d_integer;
begin
  /*
  RFCRU-1241
 Добавить блок на отправку в КК/утверждение/выдачу для клиентов из Луганской, Донецкой, Запорожской и Херсонской областей
 */

  res_error_is_critical = 2;

  select knc.kc_from_c_id
  from knclient knc
  where knc.kc_from_k_id = :k_id and
        knc.kc_amount_approved > 0
  into :c_id;

  adl_from_pl_id = 0;
  parent = 0;
  select adl.adl_from_pl_id, p.parent
  from addrclient, addrlist adl, places p
  where adc_from_c_id = :c_id and
        adl.adl_from_pl_id = p.pl_id and
        adc_address_type in (1) and
        adc_from_adl_id = adl_id and
        adl_address_str is not null
  into :adl_from_pl_id, parent;

  if (parent > 0) then
  begin
    while (parent <> 810000000001) do
    begin
      pl_id = 0;
      parent = 0;
      pl_status = 0;
      select p.pl_id, p.parent, p.pl_status
      from places p
      where p.pl_id = :adl_from_pl_id
      into :pl_id, parent, pl_status;

      if (pl_status in (810002, 810005)) then
      begin
        if (pl_id in (810001224456, 810001222192, 810001225236, 810001215958)) then --Луганской, Донецкой, Запорожской и Херсонской областей
        --if (pl_id in (810000099923)) then
        begin
          res_error_text = 'Регистрация не зафиксирована, обратитесь в опер. отдел';
          suspend;
          exit;
        end
      end

      adl_from_pl_id = parent;

    end
  end
end
