CREATE PROCEDURE HP_ZP_CARD_IMP
RETURNS (
    ZP_ID INTEGER,
    ZP_FIO VARCHAR(100),
    ZP_CARD_NO VARCHAR(20),
    ZP_FROM_C_ID DECIMAL(18,0))
AS
declare variable c_id decimal(18,0);
declare variable kol integer;
begin
  update zp_cardimp
  set zp_from_c_id = null;

  for select zp_id, zp_fio, zp_card_no, zp_from_c_id
      from zp_cardimp
      into :zp_id, :zp_fio, :zp_card_no, :zp_from_c_id
  do
  begin
    kol = 0;
    zp_fio = upper(:zp_fio);
    select count(c_id)
    from clients
--    where C_INDEX_c_name_fio starting with :zp_fio
    where c_name_fio starting with :zp_fio
    into :kol;
    if (kol is null) then
      kol = 0;
    if (kol = 1) then
    begin
      select c_id
      from clients
--      where c_index_c_name_fio starting with :zp_fio
      where c_name_fio starting with :zp_fio
      into :c_id;

      update zp_cardimp
      set zp_from_c_id = :c_id
      where zp_id = :zp_id;
      zp_from_c_id = :c_id;
    end
    if (kol > 1) then
    begin
      for select c_id
          from clients
--      where C_INDEX_c_name_fio starting with :zp_fio
          where c_name_fio starting with :zp_fio
          into :c_id
      do
      begin
        if (exists(select *
                   from hr_pstatus
                   where ps_from_c_id = :c_id)) then
        begin
          update zp_cardimp
          set zp_from_c_id = :c_id
          where zp_id = :zp_id;
          zp_from_c_id = :c_id;
        end
      end
    end
    suspend;
  end

end
