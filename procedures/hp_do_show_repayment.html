CREATE PROCEDURE HP_DO_SHOW_REPAYMENT(
    K_ID DECIMAL(18,0),
    C_ID DECIMAL(18,0),
    REP_DATE DATE,
    REPAY_AMOUNT DECIMAL(18,4),
    REPAY_TO_OS INTEGER,
    REPAY_TO_PERC INTEGER,
    REPAY_TO_FINE INTEGER,
    REPAY_TO_KOMMIS INTEGER,
    IS_FULL_REPAY INTEGER,
    IS_WRITEOFF INTEGER,
    REPAY_FACT_OFFICE_ID INTEGER,
    REPAY_DT_ACCOUNT_KIND INTEGER,
    REPAY_DT_ACCOUNT_NO VARCHAR(20),
    REPAY_CURR_RATE DECIMAL(18,4),
    REPAY_USER_ID DECIMAL(18,0),
    REPAY_DOCUMENT_CLIENT DECIMAL(18,0),
    REPAY_DOC_CLIENT_NAME_REASON VARCHAR(255),
    REPAY_MANAGER_C_ID DECIMAL(18,0),
    REPAY_FROM_RD_ID INTEGER)
RETURNS (
    OL_REPAY_AMOUNT DECIMAL(18,4),
    S_REC_NO INTEGER NOT NULL,
    S_ID_STR VARCHAR(20),
    S_ID INTEGER,
    S_DATE DATE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_FINE CURR_TYPE,
    REPAY_MODE INTEGER,
    CALCING_PERC CURR_TYPE,
    PAY_OS CURR_TYPE,
    PAY_PERC CURR_TYPE,
    PAY_FINE CURR_TYPE,
    SH_NEW_OS CURR_TYPE,
    SH_NEW_PERC CURR_TYPE,
    PAY_KOMMIS CURR_TYPE,
    PAY_TAX CURR_TYPE,
    ZAD_KOMMIS CURR_TYPE,
    PAY_PEREPLAT CURR_TYPE,
    S_RUNNING_TOTAL CURR_TYPE,
    SH_NEW_FINE DECIMAL(18,4),
    SH_NEW_KOMMIS DECIMAL(18,4),
    GRAPH_DIFF_OS DECIMAL(18,4),
    GRAPH_DIFF_PERC DECIMAL(18,4),
    NEW_REPAY_MODE INTEGER,
    IS_LOAN_CLOSING INTEGER,
    ZAD_EXTRA DECIMAL(18,4),
    PAY_EXTRA DECIMAL(18,4),
    IS_HIDDEN_LINE D_SMALLINT)
AS
declare variable is_curr integer;
declare variable cm_database_id integer;
begin
  --  0101-000252-18  кредит для тестирования
  --  k_id=101010000025201801;
  --  C_ID=1010100000759;

  /* -- igusev тестировал 22.09.2018 #operkroff_pay
  K_ID = 101010000054600101;
  C_ID = 1010100001921;
  REP_DATE = current_date;
  REPAY_AMOUNT = -1;
  REPAY_TO_OS = 1;
  REPAY_TO_PERC = 1;
  REPAY_TO_FINE = 1;
  REPAY_TO_KOMMIS = 1;
  IS_FULL_REPAY = 0;
  IS_WRITEOFF = 0;
  REPAY_FACT_OFFICE_ID = 1020200;
  REPAY_DT_ACCOUNT_KIND = 1;
  REPAY_DT_ACCOUNT_NO = '020281050-01000001';
  REPAY_CURR_RATE = 0;
  REPAY_USER_ID = 1010100000278;
  REPAY_DOC_CLIENT_NAME_REASON = '';
  REPAY_DOCUMENT_CLIENT = 1010100000200;
  REPAY_MANAGER_C_ID = null;
  REPAY_FROM_RD_ID = null;
-- */
  cm_database_id = f_get_db_id();

  if (coalesce(k_id, 0) = 0) then
    exception e_default 'input parameter K_ID is null';

  if (not exists(select *
                 from kontstate
                 where ks_from_k_id = :k_id and
                       ks_ol_is_delete = 0 and
                       ks_state = 1 and
                       :rep_date between ks_date_begin and ks_date_end)) then
  begin
    -- << добавил igusev 22.09.2018. если РФ БД, тогда дополнительно проверяю задолженность по внесистемному учету  #operkroff_pay
    --if (cm_database_id in (81001, 94401)) then
    --begin
    if (not (cm_database_id in (39801, 41701) and exists(select * from hp_repay_extra_saldo(:k_id, 0, :rep_date)))) then
    if (not exists(select okf.oko_from_k_id
                   from operkroff okf
                   where okf.oko_from_k_id = :k_id and
                         okf.oko_ol_is_active = 1
                   group by okf.oko_from_k_id
                   having sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc) > 0)) then
    begin
      s_rec_no = 1; -- если это не сделать то возвращает ошибку
      exit;
    end
    --end
    --else -- <<
    --begin
    --    S_REC_NO = 1;  -- если это не сделать то возвращает ошибку
    --    exit;
    --end
  end

  if (k_id = -1) then
    k_id = 101010000025201801; -- только для тестирования
  if (c_id is null) then
    c_id = 0;

  select ol_repay_amount
  from hp_do_repayment(:k_id, :c_id, :rep_date, :repay_amount, :repay_to_os, :repay_to_perc, :repay_to_fine, :repay_to_kommis, 0, -- :NEED_SAVE,
  :is_full_repay, :is_writeoff, :repay_fact_office_id, :repay_dt_account_kind, :repay_dt_account_no, :repay_curr_rate, :repay_user_id, :repay_document_client, :repay_doc_client_name_reason, :repay_manager_c_id, :repay_from_rd_id, 0)
  into :ol_repay_amount;

  select sum(zad_os - pay_os), sum(calcing_perc - pay_perc), sum(zad_fine - pay_fine), sum(zad_kommis - pay_kommis),
         sum(zad_extra - pay_extra)
  from temp_vrepay
  into :zad_os, :zad_perc, :zad_fine, :zad_kommis, :zad_extra;
  if (zad_os is null) then
    zad_os = 0;
  if (zad_perc is null) then
    zad_perc = 0;
  if (zad_fine is null) then
    zad_fine = 0;
  if (zad_kommis is null) then
    zad_kommis = 0;
  if (zad_extra is null) then
    zad_extra = 0;
  pay_extra = 0;

  is_loan_closing = 0;

  if (zad_os <= 0 and
      zad_perc <= 0) then -- если долг по ос и %% закрыт
  begin
    if (zad_fine <= 0) then
      is_loan_closing = 1; -- если долга по пеням нет, закрываем кредит
    else
      -- рф и кр: закрываем если даже имеется долг по пеням. пеню списываем

      cm_database_id = f_get_db_id();
    if (cm_database_id in (81001, 41701, 39801)) then
      is_loan_closing = 1;
  end
  if (is_loan_closing = 1) then
  begin
    if (zad_fine > 0 or
        zad_kommis > 0) then
      is_loan_closing = 2; -- закроется и остальное (пеня, комиссия) спишется
  end

  is_curr = 0;
  for select s_rec_no, s_id, s_date, repay_mode, zad_os, zad_perc, zad_fine, zad_kommis, calcing_perc, pay_os, pay_perc,
             pay_fine, pay_kommis, pay_tax, pay_pereplat, sh_new_os, sh_new_perc, graph_diff_os, graph_diff_perc,
             zad_extra, pay_extra, s_running_total, is_hidden_line
      from temp_vrepay
      order by /*s_date,*/ s_rec_no -- закомментировал s_date igusev 22.09.2018 #operkroff_pay
      into :s_rec_no, s_id, s_date, repay_mode, zad_os, zad_perc, zad_fine, zad_kommis, calcing_perc, pay_os, pay_perc,
           pay_fine, pay_kommis, pay_tax, pay_pereplat, sh_new_os, sh_new_perc, graph_diff_os, graph_diff_perc,
           zad_extra, pay_extra, s_running_total, is_hidden_line
  do
  begin
    -- вычисляем остаточные пени и комиссии
    sh_new_fine = zad_fine - pay_fine;
    sh_new_kommis = zad_kommis - pay_kommis;
    if (zad_extra is null) then
      zad_extra = 0;
    if (pay_extra is null) then
      pay_extra = 0;

    new_repay_mode = 0;
    if (s_date < :rep_date) then
    begin
      if (sh_new_os <> 0 or
          sh_new_perc <> 0 or
          sh_new_fine <> 0 or
          sh_new_kommis <> 0) then
        new_repay_mode = 1;
    end
    else
    begin
      if (sh_new_os <> 0 or
          sh_new_perc <> 0 or
          sh_new_fine <> 0 or
          sh_new_kommis <> 0) then
      begin
        if (is_curr = 0) then
        begin
          -- текущий платеж
          new_repay_mode = 2;
          is_curr = 1;
        end
        else
          new_repay_mode = 3; -- будущий платеж
      end
    end
    if (is_loan_closing = 0 and
        new_repay_mode = 1) then
      is_loan_closing = 3; -- статус: остается просрочка
    if (s_id >= 0) then
      s_id_str = s_id;
    else
      s_id_str = 'Доп.';
    suspend;
  end
end
