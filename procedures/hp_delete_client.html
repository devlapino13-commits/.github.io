CREATE PROCEDURE HP_DELETE_CLIENT(
    C_ID D_ID)
AS
declare variable virtual_c_id d_smallint;
begin
    /*
        RFCRU-5616

        ВАЖНО!!! ТРЕБУЕТ К ПРОЧТЕНИЮ!!!

        ЭТО ПОСЛЕДНИЙ 4 ЭТАП, КОТОРЫЙ ОТДЕЛЯЕТ НАС ОТ УДАЛЕНИЯ КЛИЕНТА
        И НЕЛЬЗЯ ДОПУСТИТЬ, ЧТОБЫ КЛИЕНТ БЫЛ УДАЛЁН ИЛИ ЗАТЁРТ ИЗ СЛЕДУЮ�©ИХ ТАБЛИЦ:
         * MINIAPPCLIENT
         * KNCLIENT
         * OPERBOOK
         * MC_LOGIN
         * EXTINFOLIST
         * HR_PSTATUS
         * USERNAME
         * SB_LIST


        ПРИМЕЧАНИЕ:
        Данная задача была разделена на часть БД + Сервис

        БД:
         1. hp_delete_client_event  - создать событие, которое дёргает сервис
         2. hp_delete_client_list   - получаем список клиентов на удаление (вызывается сервисом)
         3. hp_delete_client        - удаляем клиента (вызывается сервисом)
         4. hp_delete_client_error  - если поймали ошибку, сохраняем её (вызывается сервисом)
         5. hp_delete_client_mail   - отправляем письма (вызывается сервисом)


        Если в механизме есть что-то не понятное, прошу ознакомиться с инструкцией TODO: прикрепить инструкцию

    */

    virtual_c_id = -1;

    -- Сначала удаляем либо обновляем (согласно бизнес требованию) все "дочерние" таблицы
    merge into edoc_msg
    using contdata
    on edm_from_cda_id = cda_id and cda_from_c_id = :c_id
    when matched then delete;

    merge into qr_verify
    using qr_list
    on qrv_from_qrl_id = qrl_id and qrl_from_c_id = :c_id
    when matched then delete;

    merge into cc_website_miniapp_tmp
    using cc_website_miniapp
    on wmat_from_wma_id = wma_id and wma_from_c_id = :c_id
    when matched then
        update set
            wmat_name = 'Данные удалены',
            wmat_phone_add = 'Данные удалены',
            wmat_birthday = null,
            wmat_pass_no = 'Данные удалены',
            wmat_gender = null;


    -- Удаляем клиента из всех связаных таблиц
    delete from objectprops
    where opr_object_kind = 0 and opr_object_id = :c_id;

    delete from continfo
    where cci_object_kind = 0 and cci_object_id = :c_id;

    delete from contdata
    where cda_from_c_id = :c_id;

    delete from clientdoclog
    where cdsl_from_c_id = :c_id;

    delete from clientdocs
    where cds_from_c_id = :c_id;

    delete from docrecoglist
    where drl_from_c_id = :c_id;
     
    delete from edoc_sign
    where eds_from_c_id = :c_id;

    delete from clientdataapp
    where cdap_from_c_id = :c_id;

    delete from clientlog
    where cll_from_c_id = :c_id;

    delete from addrclient
    where adc_from_c_id = :c_id;

    delete from clientlast
    where cul_from_c_id = :c_id;

    delete from contacts
    where cc_from_c_id = :c_id;

    delete from extminiapp
    where ema_from_c_id = :c_id;

    delete from cc_website_miniapp
    where wma_from_c_id = :c_id;

    delete from qr_list
    where qrl_from_c_id = :c_id;

    delete from smsdialogsend
    where sds_from_c_id = :c_id;

    delete from smsdialogreciever
    where sdr_from_c_id = :c_id;

    delete from partner
    where pnr_from_c_id = :c_id;

    delete from krexpert
    where ke_from_p_id = :c_id;

    delete from cashcheckpart
    where ccp_from_c_id = :c_id;

    delete from contacts_for_check
    where cfc_from_c_id = :c_id;

    delete from clientlink tt
    where tt.clk_from_c_id = :c_id;

    delete from kibminiapp
    where kbm_from_c_id = :c_id;

    delete from clientulog
    where cul_from_c_id = :c_id;

    delete from petdata
    where ped_from_c_id = :c_id;

    delete from empmigcard
    where emc_from_c_id = :c_id;

    delete from petopers
    where peo_from_c_id = :c_id;



    -- Записи в таблицах ниже лучше обновить, чем удалить
    update moratory
    set m_from_c_id = :virtual_c_id
    where m_from_c_id = :c_id;

    update miniapp
    set ma_manager_id = :virtual_c_id
    where ma_manager_id = :c_id;

    update miniapp
    set ma_expert_id = :virtual_c_id
    where ma_expert_id = :c_id;

    update knaktlist kal
    set kal.kal_new_kf_officer = :virtual_c_id
    where kal.kal_new_kf_officer = :c_id;

    update knaktlist
    set kal_prior_lo_id = :virtual_c_id
    where kal_prior_lo_id = :c_id;

    update oksummary
    set oks_kc_id = :virtual_c_id
    where oks_kc_id = :c_id;

    update kontract
    set k_fact_disb_lo_id = :virtual_c_id
    where k_fact_disb_lo_id = :c_id;

    update kontract
    set k_fact_closed_lo_id = :virtual_c_id
    where k_fact_closed_lo_id = :c_id;

    update kontstate
    set ks_kc_id = :virtual_c_id
    where ks_kc_id = :c_id;

    update extsys_token
    set et_from_c_id = :virtual_c_id
    where et_from_c_id = :c_id;

    update printcert
    set prc_from_c_id = :virtual_c_id
    where prc_from_c_id = :c_id;

    update tranmembers
    set tb_from_c_id = :virtual_c_id
    where tb_from_c_id = :c_id;

    update lawmain
    set lam_manager_id = :virtual_c_id
    where lam_manager_id = :c_id;

    update cc_website_miniapp wma
    set wma_from_c_id = :virtual_c_id
    where wma_from_c_id = :c_id;




    -- А также удаляем из таблиц, которые отвечают за непосредственное удаление
    delete from clientstate_ru
    where csr_from_c_id = :c_id;

    delete from client_delete_buffer
    where cdb_from_c_id = :c_id;



    -- Последним пунктом удаляем клиента
    -- Но перед тем как удалять клиента, мы должны сначала отключить все триггеры таблицы clients
    rdb$set_context('USER_TRANSACTION', 'IGNORE_C_ID', :c_id);

    delete from clients
    where c_id = :c_id;

    -- После того как удаление прошло успешно, мы обнуляем контекстную переменную
    rdb$set_context('USER_TRANSACTION', 'IGNORE_C_ID', null);


    -- После того как клиент был полностью удалён
    -- Создаём запись о том что вообще такая c_idбыла удалёна
    insert into client_deleted (c_id)
    values (:c_id);

end
