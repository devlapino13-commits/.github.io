CREATE PROCEDURE UXP$UXFORM_EXCEL_FIELDS(
    UFM_UUID D_UUID,
    ALL_FIELDS D_BOOL = 1)
RETURNS (
    FR_ID D_INTEGER,
    FR_UUID_STR D_STR_CODE,
    FR_MASTER_STR D_STR_CODE,
    FR_IS_GRID D_BOOL,
    FR_CAPTION D_STR_255,
    UF_IS_PARAM D_BOOL,
    UF_FIELD_NAME D_STR_CODE,
    UF_DATA_TYPE D_STR_CODE,
    UF_IN_UI D_BOOL,
    UGI_CAPTION D_STR_100,
    UGI_POSITION D_INTEGER,
    UGI_WIDTH D_INTEGER,
    UGI_GROUP_INDEX D_INTEGER,
    UGI_SUMMARY_KIND D_INTEGER,
    UGI_HINT D_STR_LONG)
AS
declare variable edi_name d_str_code;
declare variable fr_uuid d_uuid;
declare variable master_fr_uuid d_uuid;
declare variable ufp_editor d_str_code;
declare variable grid_uuid d_uuid;
declare variable uf_uuid d_uuid;
declare variable orig_caption d_str_100;
declare variable add_text d_boolean;
begin
  if (all_fields is null) then
    all_fields = 1;

  fr_id = -1;
  for select fr_uuid, master_fr_uuid, fr_is_grid, res_caption
  from uxp$uxframe_tree(:ufm_uuid)
  into fr_uuid, master_fr_uuid, fr_is_grid, fr_caption
  do
  begin
    fr_id = fr_id + 1;

    select ufr_name
    from ux$frame
    where ufr_uuid = :fr_uuid
    into fr_caption;

    fr_uuid_str = uuid_to_char(fr_uuid);
    fr_master_str = '';
    if (master_fr_uuid is not null) then
      fr_master_str = uuid_to_char(master_fr_uuid);

    grid_uuid = null;

    select ugr_uuid
    from ux$grid
    where ugr_ufr_uuid = :fr_uuid
    order by ugr_default desc
    rows 1
    into grid_uuid;

    uf_uuid = null;
    uf_field_name = '';
    uf_is_param = 1;
    uf_data_type = 0;
    ugi_caption = '';
    ugi_position = 0;
    ugi_width = 0;
    ugi_group_index = null;
    ugi_summary_kind = 0;
    ugi_hint = '';

    for select uf_uuid, uf_field_name, uf_name, uf_is_param, uf_position, uf_data_type,
               ufp_editor, iif(uf_edi_uuid is not null, uxf$code(uf_edi_uuid, 'code'), null)
    from ux$field, ux$frparam
    where ufp_ufr_uuid = :fr_uuid and
          ufp_parent is null and
          ufp_uf_uuid = uf_uuid
    into uf_uuid, uf_field_name, ugi_caption, uf_is_param, ugi_position, uf_data_type,
         ufp_editor, edi_name
    do
    begin
      if (edi_name in ('current$reverse', 'current$user')) then
        Continue;

      uf_in_ui = 0;
      ugi_width = 0;
      ugi_group_index = null;
      ugi_summary_kind = 0;
      ugi_hint = '';

      if (grid_uuid is not null and uf_is_param = 0) then
      begin
        select ugi_caption, ugi_position, ugi_width, ugi_group_index, ugi_summary_kind, ugi_hint
        from ux$gritem
        where ugi_ugr_uuid = :grid_uuid and
              ugi_uf_uuid = :uf_uuid
        rows 1
        into :ugi_caption, :ugi_position, :ugi_width, ugi_group_index, ugi_summary_kind, ugi_hint;

        if (row_count = 1) then
          uf_in_ui = 1;
      end

      if (uf_in_ui = 0) then
      begin
        select ufi_name, ufi_position, ufi_width, ufi_hint
        from ux$fritem
        where ufi_ufr_uuid = :fr_uuid and
              ufi_uf_uuid = :uf_uuid
        rows 1
        into :ugi_caption, :ugi_position, :ugi_width, ugi_hint;

        if (row_count = 1) then
          uf_in_ui = 1;
      end

      if (all_fields = 0 and uf_in_ui = 0 and uf_is_param = 0) then
        Continue;

      if (ugi_width = 0 and coalesce(ufp_editor, '') <> '') then
      begin
        select default_width
        from uxp$control_kind
        where editor_id = :ufp_editor
        rows 1
        into :ugi_width;
      end

      if (ugi_width = 0) then
        ugi_width = 100;

      add_text = false;
      orig_caption = ugi_caption;
      if (ufp_editor in ('buttonedit', 'dialogedit', 'lookup', 'enum', 'reflist')) then
      begin
        if (uf_in_ui = 1) then
          add_text = true;

        ugi_caption = ugi_caption || ' (id)';
      end else
      if (edi_name in ('current$region_id', 'current$office_id')) then
      begin
        if (uf_in_ui = 1) then
          add_text = true;

        ugi_caption = ugi_caption || ' (id)';
      end

      suspend;

      if (add_text) then
      begin
        uf_field_name = uf_field_name || '$text';
        ugi_caption = orig_caption;
        suspend;
      end
    end
  end
end
