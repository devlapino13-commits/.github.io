CREATE PROCEDURE HP_COMPET_KG_OLIMPIC_2020_MO(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer = 0;
declare variable rd_id integer;
declare variable cpp_orderno integer;
declare variable res_period_id integer;
declare variable reg_id integer;
declare variable cpr_group_name varchar(255) = '';
declare variable cpd_comment varchar(3072) = '';
declare variable points_all curr_type = 0;
declare variable points_1 curr_type = 0;
declare variable points_2 curr_type = 0;
declare variable points_3 curr_type = 0;
declare variable points_4 curr_type = 0;
declare variable points_5 curr_type = 0;
declare variable points_6 curr_type = 0;
declare variable points_7 curr_type = 0;
declare variable cpr_id integer;
declare variable par7_diff_avg d_percent;
declare variable rank_point_1 smallint;
declare variable avg_disb_amount d_currency;
declare variable ad_diff_balance d_currency;
declare variable rank_point_2 smallint;
declare variable par1_end d_percent;
declare variable par1_end_balance d_percent;
declare variable par7_end d_percent;
declare variable par7_begin d_percent;
declare variable par7_diff d_percent;
declare variable rep_indicator integer;
declare variable is_month_end smallint;
declare variable rep_period integer;
declare variable plan_sum decimal(15,2);
declare variable plan_perc d_percent;
declare variable super_pokazatel d_currency;
declare variable super_rank d_currency;
declare variable sum_voznog_office d_currency;
declare variable reg_qty smallint;
declare variable max_count smallint;
declare variable month_date_begin date;
declare variable avg_points d_currency;
declare variable ad_disb_loan decimal(15,2);
declare variable ad_active_client decimal(15,2);
declare variable cpd_points_all decimal(15,2);
declare variable cpr_rank smallint;
declare variable cpr_final_orderno decimal(15,2);
declare variable kol_kred_closed integer;
declare variable in_period_disb integer;
declare variable coeff_uderj d_percent;
declare variable ad_loan_disb integer;
declare variable new_cpr_final_orderno decimal(15,2);
declare variable max_points decimal(15,2);
declare variable cpr_calced_points decimal(15,2);
declare variable cpr_final_points decimal(15,2);
declare variable avg_disb_1_15 decimal(15,2);
declare variable rank_disb_1_15 decimal(15,2);
declare variable point_disb_1_15 decimal(15,2);
declare variable lo_qty decimal(15,2);
declare variable plan_spf d_currency;
declare variable fact_spf d_currency;
declare variable diff_spf d_currency;
declare variable lo_id d_id;
declare variable plan_sum_ipf decimal(15,2);
declare variable plan_sum_kl decimal(15,2);
declare variable disb_sum decimal(15,2);
declare variable dosroch_sum decimal(15,2);
declare variable plan_perc_reg decimal(15,2);
declare variable lo_panelty_count d_integer;
declare variable add_points d_currency;
declare variable cpr_comment d_str_255;
declare variable ad_active_balance d_currency;
declare variable ad_delinq_balance_7 d_currency;
declare variable ad_begin_active_balance d_currency;
declare variable ad_begin_delinq_balance_7 d_currency;
begin
  /* Олимпийские игры МО */

  -- Удаляем существующие расчеты за этот период
  delete from competresult
  where (cpr_from_cpp_id = :cpp_id);

  res_text = '';

  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  -- Stage:
  -- 0 - период (месяц)
  -- 1 - Промежуточные итоги
  -- 2 - Итоги
  -- 3 - Финальные результаты

  rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);
  month_date_begin = res_date_end - extract(day from res_date_end) + 1;

  if (extract(day from res_date_end) not in (10, 20)) then
  begin
    is_month_end = 1;
  end
  else
    is_month_end = 0;

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  if (extract(month from res_date_end) = 3 and
      extract(day from res_date_end) = 20) then
    is_month_end = 1; --Covid begin

  if (cpp_orderno = 0 and
      res_date_begin = cast('31.01.2020' as date)) then
  begin
    res_date_begin = '01.02.2020';
    res_date_end = '01.02.2020';

    -- Формируем в памяти данные по КР
    execute procedure hp_analitica_create_period(1040100, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID команды и ID СВЗ
    for select res_field_1
        from hp_analitica_report('reg_id, reg_name', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
        where ad_active_balance > 0
        into :reg_id
    do
    begin
      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                cpr_from_un_id, cpr_comment)
      values (:cpp_id, '', 7, :reg_id, 0, :un_id, '')
      returning cpr_id
      into :cpr_id;
    end
    suspend;
    exit;
  end

  if (stage = 0) then -- Если уровень декада
  begin

    -- Формируем в памяти данные по компании
    execute procedure hp_analitica_create_period(1000000, iif(:is_month_end = 1, :month_date_begin, :res_date_begin),
        :res_date_end, 1, null, null, 30, 0, 'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID Региона
    for select res_field_1, res_field_2, --
               sum(ad_delinq_balance_diff2) par7_diff_avg, --
               sum(ad_disb_amount) avg_disb_amount, --
               sum(ad_delinq_balance_end1) par1_end_balance, --
               sum(ad_diff_balance), --
               sum(ad_disb_loan), --
               sum(ad_active_client), --
               sum(ad_active_balance) ad_active_balance, --
               sum(ad_delinq_balance_end2) ad_delinq_balance_7, --
               sum(ad_begin_active_balance) ad_begin_active_balance, --
               sum(ad_delinq_balance_begin2) ad_begin_delinq_balance_7

        from hp_analitica_report('reg_id, reg_name, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
        where res_field_3 > 1000000000010 and
              ad_active_balance > 0.0 -- Берем тех у кого имеется портфель на конец периода
        group by 1, 2
        into :reg_id, -- ID офиса
             :cpr_group_name, -- наименование офиса
             :par7_diff_avg, --
             :avg_disb_amount, --
             :par1_end_balance, --
             :ad_diff_balance, --
             :ad_disb_loan, --
             :ad_active_client, --
             :ad_active_balance, --
             :ad_delinq_balance_7, --
             :ad_begin_active_balance, --
             :ad_begin_delinq_balance_7
    do
    begin
      par7_begin = ad_begin_delinq_balance_7 / ad_begin_active_balance * 100;
      par7_end = ad_delinq_balance_7 / ad_active_balance * 100;

      lo_qty = 0.0;
      plan_perc_reg = 0;
      for select ps_from_c_id, count(h.ps_from_c_id) over() --общее количество СВЗ
          from hr_pstatus h
          inner join region r on r.reg_unicode = h.ps_from_reg_id
          inner join joblist jl on jl.jl_id = h.ps_from_jl_id
          inner join jobkind jk on jk.jk_id = jl.jl_kind
          where :res_date_end between h.ps_date_begin and h.ps_date_end and
                h.ps_is_working = 1 and
                jk.jk_system_code in ('JobSystem.LO_Senior', 'JobSystem.LO_Standart') and
                r.reg_unicode = :reg_id and
                ps_from_c_id not in (select hrv.hrv_from_c_id -- ИСКЛЮЧАЕМ
                                     from hr_voctravel hrv, sa_worktype swt, hr_orders hr --  •    Отпуск без содержания по уходу за ребенком
                                     where hrv.hrv_date_begin <= hrv.hrv_date_end_fact and --  •    Приказ_отпуск_по_уходу_за_ребенком
                                           hrv.hrv_status = 1 and --  •    Декретный отпуск.
                                           ho_order_type = sawt_id and
                                           hrv_from_ho_id = ho_id and --     sawt_base_type =5 чтобы исключить 21-ю позицию в кг и  аз
                                           :res_date_end between hrv.hrv_date_begin and hrv.hrv_date_end_fact and
                                           swt.sawt_id in (16, 17, 18, 21) and
                                           swt.sawt_base_type = 5)
          into :lo_id, :lo_qty
      do
      begin
        --реализация план выдач--------------------------------------------->>>>
        plan_sum_ipf = 0;
        select po_amount
        from hp_get_planoper_info(:reg_id, :lo_id, 'planindicators.disb_amount_ipf', :res_date_end, :rep_period)
        into plan_sum_ipf;
        if (plan_sum_ipf is null) then
          plan_sum_ipf = 0;

        plan_sum_kl = 0;
        select po_amount
        from hp_get_planoper_info(:reg_id, :lo_id, 'planindicators.disb_amount', :res_date_end, :rep_period)
        into plan_sum_kl;
        if (plan_sum_kl is null) then
          plan_sum_kl = 0;

        plan_sum = plan_sum_ipf + plan_sum_kl;

        --      обьем выдач
        disb_sum = 0;
        select sum(k_amount_approved)
        from kontstate, kontract
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_region_id = :reg_id and
              ks_date_begin between :month_date_begin and :res_date_end and
              ks_kc_id = :lo_id and
              ks_from_k_id = k_id
        into disb_sum;
        if (disb_sum is null) then
          disb_sum = 0;

        --      Обьем досроных погашений
        dosroch_sum = 0;
        select -sum(ok_os)
        from operkred, kontract, knaktlist
        where ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_os < 0 and
              ok_ol_date between :month_date_begin and :res_date_end and
              ok_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              ok_ol_date between kal_date and kal_date_end and
              kal_new_kf_officer = :lo_id and
              ok_ol_date < ok_sh_date and
              extract(month from ok_ol_date) <> extract(month from ok_sh_date)
        into dosroch_sum;
        if (dosroch_sum is null) then
          dosroch_sum = 0;

        --      Выполнения плана по выдачам СпФ
        if (plan_sum = 0) then
          plan_sum = 700000.00;
        if (plan_sum > 0) then
          plan_perc = (disb_sum - dosroch_sum) / plan_sum * 100;
        else
          plan_perc = 0;

        plan_perc_reg = plan_perc_reg + plan_perc;
        --реализация план выдач---------------------------------------------<<<<
      end

      if (lo_qty is null) then
        lo_qty = 0.0;

      par7_diff_avg = par7_diff_avg / lo_qty;
      avg_disb_amount = avg_disb_amount / lo_qty;
      par1_end = par1_end_balance / ad_active_balance * 100;
      ad_diff_balance = ad_diff_balance / lo_qty;
      ad_disb_loan = ad_disb_loan / lo_qty;
      ad_active_client = ad_active_client / lo_qty;

      cpd_comment = '';
      points_1 = 0.0;
      points_2 = 0.0;
      points_3 = 0.0;
      points_4 = 0.0;
      points_5 = 0.0;
      points_6 = 0.0;
      points_7 = 0.0;

      cpr_id = null;
      select cpr_id
      from competresult
      where cpr_from_cpp_id = :cpp_id and
            cpr_object_kind = 7 and --7 region_id
            cpr_object_id = :reg_id
      into :cpr_id;

      if (is_month_end = 1) then
      begin
        points_3 = iif(par7_end < 0.1, 15, 0);
        par7_diff = (par7_end - par7_begin);

        if (points_3 = 0) then
        begin

          points_3 = case
                       when par7_diff <= -0.4 then 10
                       when par7_diff <= -0.3 then 7
                       when par7_diff <= -0.2 then 5
                       when par7_diff = 0.0 then 2
                       when par7_diff <= 0.1 then -3
                       else -8
                     end;
        end
        cpd_comment = coalesce(cpd_comment, '') || 'PAR7 офиса на начало: <b>' || round(par7_begin, 2) || ' %.</b><br>';
        cpd_comment = coalesce(cpd_comment, '') || 'PAR7 офиса на конец: <b>' || round(par7_end, 2) || ' %.</b><br>';
        cpd_comment = coalesce(cpd_comment, '') || 'Изменение PAR7 офиса: <b>' || round(par7_diff, 2) || ' %.</b><br><br>';

        --    4. Выполнение плана по количеству ВИПС

        --Заработок офиса
        sum_voznog_office = 0;
        with t_reg_zp
        as (select sum_voznog_office --, plan_spf, fact_spf
            from cb_kg_km_2017_02(:reg_id, :res_date_end)
            union
            select -sum_voznog_office --, plan_spf, fact_spf
            from cb_kg_km_2017_02(:reg_id, :month_date_begin - 1))
        select sum(sum_voznog_office)
        from t_reg_zp
        into sum_voznog_office; --, plan_spf, fact_spf;
        sum_voznog_office = coalesce(sum_voznog_office, 0) / lo_qty;

        /*plan_spf = coalesce(plan_spf, 0);
        fact_spf = coalesce(fact_spf, 0);

        diff_spf = plan_spf - fact_spf;
        diff_spf = iif(diff_spf > 0, diff_spf, 0);

        plan_spf = round(plan_spf, 2);
        fact_spf = round(fact_spf, 2);
        diff_spf = round(diff_spf, 2);

        if (plan_spf > 0) then
        begin
          cpd_comment = coalesce(cpd_comment, '') || 'План по штату ВИПС: <b>' || round(plan_spf, 2) || '.</b><br>';
          cpd_comment = coalesce(cpd_comment, '') || 'Выполнение плана по штату ВИПС: <b>' || round(fact_spf, 2) || '.</b><br>';
          cpd_comment = coalesce(cpd_comment, '') || 'Остаток по штату ВИПС: <b>' || round(diff_spf, 2) || '.</b><br><br>';

          points_4 = case
                       when diff_spf = 0 then 2
                       when diff_spf = 1 then -8
                       else -15
                     end;
        end
        else
          cpd_comment = coalesce(cpd_comment, '') || 'План по штату ВИПС: <b>0</b><br><br>';
        --4. Выполнение плана по штату ВИПС (конец) */

        --      реализация план выдач
        if (coalesce(plan_perc_reg, 0) > 0) then
          plan_perc_reg = plan_perc_reg / lo_qty;
        else
          plan_perc_reg = 0.0;

        cpd_comment = cpd_comment || 'Выполнения планов ВИПС по выдачам: <b>' || round(plan_perc_reg, 0) || '%</b><br><br>';

        --      Выполнение плана по выдачам СВЗ за отчетный месяц в процентах
        points_5 = case
                     when plan_perc_reg >= 100 then 8
                     when plan_perc_reg > 90 then 6
                     when plan_perc_reg > 80 then 4
                     when plan_perc_reg > 70 then 0
                     when plan_perc_reg > 60 then -3
                     else -5
                   end;
        --      Выполнение плана по выдачам СВЗ за отчетный месяц в процентах <<<<<<<<<<<<<<<<<<<<<<<<

        super_pokazatel = 0;
        if (is_month_end = 1) then
        begin
          with tabl1
          as (select kc_from_c_id,
                     (select distinct 1
                      from kontstate ks, knclient kc
                      where ks.ks_ol_is_delete = 0 and
                            ks.ks_state = 1 and
                            ks.ks_date_begin between :res_date_begin and :res_date_end and
                            ks.ks_from_k_id = kc.kc_from_k_id and
                            kc.kc_amount_approved > 0 and
                            kc.kc_from_c_id = kc1.kc_from_c_id and
                            kc.kc_when_prior_loan_closed = ks1.ks_date_begin) as in_period_disb
              from kontstate ks1, knclient kc1
              where ks_ol_is_delete = 0 and
                    ks_state = 2 and
                    ks_date_begin between :res_date_begin and :res_date_end and
                    ks_region_id = :reg_id and
                    ks_from_k_id = kc_from_k_id and
                    kc_amount_approved > 0 and
                    not exists(select *
                               from oksummary oks
                               where oks_from_k_id = ks_from_k_id and
                                     oks_date_end < :res_date_end and --На вс.случ.
                                     oks_delinq_date < :res_date_begin and --На вс.случ.
                                     oks_date_end - oks_delinq_date >= 45 and
                                     oks_close_state = 0 --клиенты с просрочкой менее 45 дней или без просрочек (Исключаем если есть просрочка 45 дн и более)
                    ))
          select count(*) kol_closed, sum(in_period_disb) in_period_disb
          from tabl1
          into kol_kred_closed, in_period_disb;

          if (kol_kred_closed is null) then
            kol_kred_closed = 0;
          if (in_period_disb is null) then
            in_period_disb = 0;
          if (kol_kred_closed > 0) then
            coeff_uderj = cast(in_period_disb as double precision) / kol_kred_closed * 100;

          cpd_comment = cpd_comment || 'Всего закрытых за период: <b>' || round(kol_kred_closed, 0) || '</b><br>';
          cpd_comment = cpd_comment || 'Повторно выданные за период: <b>' || round(in_period_disb, 0) || '</b><br>';
          cpd_comment = cpd_comment || 'Коэффициент удержание клиентов: <b>' || round(coeff_uderj, 2) || '</b><br>';

          if (extract(month from res_date_end) = 2) then
          begin
            select sum(km.kml_total)
            from kontmat km
            inner join kontstate ks on (ks_date_begin between dateadd(month, -1, cast(:res_date_end as date) + 1) and :res_date_end and
                  ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  ks.ks_region_id = :reg_id)
            inner join agreements ag on (kml_from_ag_id = ag_id and
                  ag_object_kind = 0)
            inner join clients c on (ag_object_id = c_id and
                  c_name_fio in (--
                             'ОсОО "Глобал Циано Групп Лоджистикс"', --
                             'ОсОО "Кайоглы-Эрташ"', --
                             'ОсОО "ATS-GROUP"', --
                             'ОсОО "Нави-Транс"'))
            where (kml_object_kind = 9 and
                  kml_object_id = ks_from_k_id and
                  kml_from_ml_id is not null)
            into super_pokazatel;
          end
          else
          if (extract(month from res_date_end) in (3, 10)) then
          begin --
            super_pokazatel = ad_diff_balance;
          end
          else
          if (extract(month from res_date_end) = 5) then
          begin --
            --Выдачи повторным
            select count(*)
            from kontstate ks
            inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id and
                  kc.kc_amount_approved > 0 and
                  kc.kc_circle > 1
            where ks_region_id = :reg_id and
                  ks_state = 1 and
                  ks_ol_is_delete = 0 and
                  ks_date_begin between :res_date_begin and :res_date_end and
                  --Неактивный на 17-мая
                  not exists(select *
                             from knclient kc2
                             inner join kontstate ks2 on kc2.kc_from_k_id = ks2.ks_from_k_id and
                                   ks2.ks_state = 1 and
                                   ks2.ks_ol_is_delete = 0 and
                                   '17.05.20' between ks2.ks_date_begin and ks2.ks_date_end
                             where kc.kc_from_c_id = kc2.kc_from_c_id and
                                   kc2.kc_amount_approved > 0)
            into super_pokazatel;
          end
          else
          if (extract(month from res_date_end) = 6) then
          begin --
            select sum(k_amount_approved)
            from kontstate ks
            inner join kontract k on ks_from_k_id = k_id and
                  k.k_refund = 0
            where ks_region_id = :reg_id and
                  ks_state = 1 and
                  ks_ol_is_delete = 0 and
                  ks_date_begin between :month_date_begin and :res_date_end
            into super_pokazatel;
          end
          else
          if (extract(month from res_date_end) in (7, 8)) then
          begin --
            super_pokazatel = par1_end;
          end
          else
          if (extract(month from res_date_end) in (9)) then
          begin --
            select sum(k_amount_approved)
            from kontstate ks
            inner join kontract k on ks_from_k_id = k_id
            where ks.ks_region_id = :reg_id and
                  ks_state = 1 and
                  ks_ol_is_delete = 0 and
                  ks_date_begin between :month_date_begin and :res_date_end
            into super_pokazatel;
          end
        end
        super_pokazatel = coalesce(super_pokazatel, 0);

                    from hr_pstatus h
                    inner join region r on r.reg_unicode = h.ps_from_reg_id

                                               where hrv.hrv_date_begin <= hrv.hrv_date_end_fact and --  •    Приказ_отпуск_по_уходу_за_ребенком
                                                     hrv.hrv_status = 1 and --  •    Декретный отпуск.
                                                     ho_order_type = sawt_id and
                                                     hrv_from_ho_id = ho_id and --     sawt_base_type =5 чтобы исключить 21-ю позицию в кг и  аз
                                                     :res_date_end between hrv.hrv_date_begin and hrv.hrv_date_end_fact and
                                                     swt.sawt_id in (16, 17, 18, 21) and
                                                     swt.sawt_base_type = 5)) lo_qty --

            from hp_analitica_report('reg_id, reg_name, lo_id', 'tree_group', 31, 999999, 0, 999999, 8, 999999, 31, 999999, 181, 999999, 1065, 0, 'standart')
            where res_field_3 > 1000000000010 and
                  ad_active_balance > 0.0 -- Берем тех у кого имеется портфель на конец периода
            group by 1, 2),
        t_info_all
        as (select reg_id, --
                   reg_name, --
                   ad_delinq_balance_7 / ad_active_balance * 100 as par7_end, --
                   ad_diff_client / lo_qty as ad_diff_clients, --
                   ad_disb_loan / lo_qty as ad_disb_loan
            from t_intrument_analitika)
        /*  Дополнительные основания для начисления баллов
        1.1.        МО с наименьшим показателем PAR7 на конец последней недели Олимпиады    5
        1.2.        МО с наибольшим средним приростом активных клиентов (в случае двух МО, берется среднее значение)     5
        1.3.        МО с наибольшим количеством выдач среди МО (в случае двух МО, берется среднее значение)    5              */

        select reg_id, --
                       reg_name, --
                       par7_end, --
                       'Балл за наименьший ПАР7' cpr_comment
        from t_info_all
        where par7_end = 0.0
        union
        select first 1 reg_id, --
                       reg_name, --
                       rank() over(order by ad_diff_clients desc), --
                       'Балл за ср.прирост активных клиентов'
        from t_info_all
        union
        select first 1 reg_id, --
                       reg_name, --
                       rank() over(order by ad_disb_loan desc), --
                       'Балл за наибольший кол.выдач'
        from t_info_all
        into :reg_id, -- ID офиса
             :cpr_group_name, -- наименование офиса
             :par7_end, :cpr_comment

    do
    begin
      update competresult cpr
      set cpr_additional_points = cpr_additional_points + 5,
          cpr_final_points = cpr_final_points + 5,
          cpr_order_in_group = cpr_final_orderno,
          cpr_comment = iif(cpr_comment = '', '', '; ') || :cpr_comment
      where cpr_from_cpp_id = :cpp_id and
            cpr_object_id = :reg_id;
    end
             
    update competresult
    set cpr_calced_points = cpr_final_points / 3,
        cpr_final_points = cpr_final_points / 3 + cpr_additional_points,
        cpr_order_in_group = cpr_final_orderno
    where (cpr_from_cpp_id = :cpp_id);

    execute procedure hp_competition_recalc_order(:cpp_id);
  end
  suspend;
end
