CREATE PROCEDURE HP_DO_KNCLIENT_LAST_DATE(
    REP_DATE D_DATE)
AS
declare variable kc_id id_type;
declare variable ol_date d_date;
declare variable k_id bigint;
declare variable ol_date_max d_date;
begin
-- восстанавливает значение KC_LAST_OL_DATE по всем займам

  for select ks_from_k_id
      from kontstate
      where ks_date_begin = :rep_date and
            ks_ol_is_delete = 0 and
            ks_state = 1
      into :k_id
  do
  begin
    for select kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into kc_id
    do
    begin
      ol_date = null;

      select kob_ol_date
      from kn_operbook
      where kob_ol_is_active = 1 and
            kob_from_kc_id = :kc_id
      order by kob_ol_date desc
      rows 1
      into ol_date;

      if (ol_date is null) then
      begin
        select ok_ol_date
        from kn_operkred
        where ok_from_kc_id = :kc_id and
              ok_ol_date < '31.12.2017' and
              ok_ol_is_active = 1
        order by ok_from_kc_id desc, ok_ol_date desc
        rows 1
        into ol_date;
      end

      if (ol_date is not null) then
      begin
        update knclient
        set kc_last_ol_date = :ol_date
        where kc_id = :kc_id;
      end
    end
  end
end
