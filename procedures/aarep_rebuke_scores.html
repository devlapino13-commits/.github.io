CREATE PROCEDURE AAREP_REBUKE_SCORES(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    REG_NAME D_STR_100,
    ADDED_SCORES D_INTEGER,
    USED_SCORES D_INTEGER,
    SUM_SCORES D_INTEGER,
    JL_NAME D_STR_100,
    C_ID D_ID,
    JL_ID D_INTEGER,
    REG_ID D_INTEGER,
    REG_PAR_NAME D_STR_100,
    ZEROED_SCORES D_INTEGER)
AS
declare variable reg_id_tmp d_integer;
begin
  reg_id = 0;
  for select ps.ps_from_c_id, ps.ps_from_jl_id, ps.ps_from_reg_id
      from hr_pstatus ps
      inner join region_list(:rep_region_id, 1, 3, 1) on id_list = cast(ps.ps_from_reg_id as integer)
      where :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
            ps.ps_is_working = 1 and
            ps.ps_status = 1 and
            (((ps_is_working = 0) and
            ((ps_transfer_office is null) or
            (ps_transfer_office = ps_from_reg_id))) or
            (ps_is_working = 1))
      into :c_id, :jl_id, :reg_id_tmp
  do
  begin
    select scores_was_added, scores_was_used, scores_sum, zeroed_scores
    from hp_calc_rebuke_scores(:rep_date_end, :c_id)
    into :added_scores, :used_scores, :sum_scores, :zeroed_scores;

    if (added_scores = 0) then
      continue;

    if (reg_id <> reg_id_tmp) then
    begin
      reg_id = :reg_id_tmp;
      select reg.reg_code || ' ' || reg.reg_name, par.reg_code || ' ' || par.reg_name
      from region reg
      inner join region par on reg.parent = par.reg_unicode
      where reg.reg_unicode = :reg_id
      into :reg_name, :reg_par_name;
    end

    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into :c_name_fio;

    select jl.jl_name
    from joblist jl
    where jl.jl_id = :jl_id
    into :jl_name;
    suspend;
  end
end
