CREATE PROCEDURE HP_E_HR_ORDERS_ONE_LOAD(
    IN_ORDER_ID D_ID)
RETURNS (
    HO_ID D_ID,
    HO_DATE D_DATE,
    HO_FROM_REG_ID D_INTEGER,
    HO_TEXT D_STR_LONG,
    HO_REASON D_STR_255,
    HO_REASON_DATE D_DATE,
    HO_FROM_UN_ID D_ID,
    HO_ORDER_TYPE D_INTEGER,
    HO_ORDER_SUBTYPE D_INTEGER,
    HO_ORDER_NO D_STR_20,
    HO_STATUS D_INTEGER,
    HO_STATUS_USER_ID D_ID,
    HO_SIGNER_ID D_INTEGER,
    HO_ACCOUNTANT_ID ID_TYPE,
    HO_HELP_AMOUNT D_INTEGER,
    HO_CERT_MARRIAGE D_BOOL,
    HO_CERT_BIRTH_CHILD D_BOOL,
    HO_CERT_BIRTH_EMP D_BOOL,
    HO_CERT_DEATH D_BOOL,
    RES_CAN_EDIT D_BOOL,
    RES_IS_SIGNED D_BOOL)
AS
declare variable ho_from_edl_id d_id;
begin
  res_can_edit = 1;
  res_is_signed = 0;

  for select ho_id, ho_date, ho_from_reg_id, ho_text, ho_reason, ho_reason_date, ho_from_un_id, ho_order_type,
             ho_order_subtype, ho_order_no, ho_status, ho_status_user_id, ho_signer_id, ho_accountant_id, ho_from_edl_id,
             ho_help_amount, ho_cert_marriage, ho_cert_birth_child, ho_cert_birth_emp, ho_cert_death
  from hr_orders
  where ho_id = :in_order_id
  into ho_id, ho_date, ho_from_reg_id, ho_text, ho_reason, ho_reason_date, ho_from_un_id, ho_order_type,
       ho_order_subtype, ho_order_no, ho_status, ho_status_user_id, ho_signer_id, ho_accountant_id, ho_from_edl_id,
       ho_help_amount, ho_cert_marriage, ho_cert_birth_child, ho_cert_birth_emp, ho_cert_death
  do
  begin
    res_is_signed = null;

    if (ho_from_edl_id is not null) then
    begin
      select edl_status
      from edoc_list
      where edl_id = :ho_from_edl_id
      into res_is_signed;
    end

    suspend;
  end
end
