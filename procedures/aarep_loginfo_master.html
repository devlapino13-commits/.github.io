CREATE PROCEDURE AAREP_LOGINFO_MASTER(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID)
RETURNS (
    LGI_ID TYPE OF D_ID,
    LGI_OBJECT_KIND TYPE OF D_INTEGER,
    LGI_OBJECT_ID TYPE OF D_ID,
    LGI_PRV_ID TYPE OF D_ID,
    LGI_ACTION_STR D_STR_CODE,
    LGI_STATUS TYPE OF D_SMALLINT,
    LGI_IS_DELETE TYPE OF D_SMALLINT,
    LGI_FROM_UN_ID TYPE OF D_ID,
    LGI_DATETIME TYPE OF D_DATETIME,
    UN_NAME D_STR_100)
AS
declare variable lgi_action d_smallint;
declare variable prv_id d_id;
declare variable c_id d_id;
begin
  -- Отобржает логи изменений по одному объекту

  obj_kind = f_objkind(obj_kind, 'id');

  for select lgi_id,
             lgi_object_kind,
             lgi_object_id,
             lgi_prv_id,
             lgi_action,
             lgi_status,
             lgi_is_delete,
             lgi_from_un_id,
             lgi_datetime
      from loginfo
      where lgi_object_kind = :obj_kind and
            lgi_object_id = :obj_id
      order by lgi_id desc
      into :lgi_id,
           :lgi_object_kind,
           :lgi_object_id,
           :lgi_prv_id,
           :lgi_action,
           :lgi_status,
           :lgi_is_delete,
           :lgi_from_un_id,
           :lgi_datetime
  do
  begin
    un_name = 'Не определен';

    if (lgi_from_un_id is not null) then
    begin
      select un_name, un_from_c_id
      from username
      where un_id = :lgi_from_un_id
      into un_name, c_id;
  
      if (c_id is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :c_id
        into un_name;
      end
    end

    lgi_action_str = case lgi_action
      when -1 then 'Удаление'
      when 0 then 'Редактирование'
      when 1 then 'Вставки'
    end;
    suspend;

    if (lgi_prv_id is not null and lgi_prv_id <> obj_id) then
    begin
      prv_id = lgi_prv_id;
      for select lgi_id, lgi_object_kind, lgi_object_id, lgi_prv_id, lgi_action_str, lgi_status, lgi_is_delete,
                 lgi_from_un_id, lgi_datetime
      from aarep_loginfo_master(:obj_kind, :prv_id)
      into :lgi_id, :lgi_object_kind, :lgi_object_id, :lgi_prv_id, :lgi_action_str, :lgi_status, :lgi_is_delete,
           :lgi_from_un_id, :lgi_datetime
      do
        suspend;
    end
  end
end
