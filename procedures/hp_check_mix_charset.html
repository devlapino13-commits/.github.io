CREATE PROCEDURE HP_CHECK_MIX_CHARSET(
    S_TEXT VARCHAR(255),
    CAN_CHANGE SMALLINT,
    FORCE_CHANGE SMALLINT)
RETURNS (
    ERR_TXT VARCHAR(255),
    RESULT VARCHAR(255),
    RES_IS_CHANGED INTEGER)
AS
declare variable i smallint;
declare variable cyr smallint;
declare variable lat smallint;
declare variable s_orig varchar(255);
declare variable cyr_ch d_str_100;
begin
  cyr_ch = 'АБВГДЕЁЖЗИЙКЛМНОПРСТУФХЦЧШ�©ЪЬЫЭЮЯабвгдеёжзийклмнопрстуфхцчшщъьыэюяң�©ү' ||
    iif(f_get_db_id() = 39801, 'әғқұhі', '');

  s_orig = s_text;
  res_is_changed = 0;
  result = s_text;
  cyr = 0;
  lat = 0;
  err_txt = '';
  i = 0;
  while (i < char_length(s_text) and (cyr = 0 or lat = 0)) do
  begin
    i = i + 1;
    if (position(substring(s_text from i for 1), ' 0123456789`~!@#$%^&*()_+"№;%:?*{}[].,/-''|\IVX') = 0) then
    begin
      if (position(substring(s_text from i for 1), cyr_ch) > 0) then
        cyr = 1;
      else
        lat = 1;
    end
  end

  if ((lat = 0 and
      cyr = 0) or (force_change in (0, 1) and
      cyr = 0) or (force_change in (0, 2) and
      lat = 0)) then
  begin
    if (result <> s_orig) then
      res_is_changed = 1;
    suspend;
    exit;
  end
  if (can_change = 0) then
  begin
    if (cyr = 1 and
        lat = 1) then
      err_txt = 'В строке смешаны буквы кириллицы и латиницы.';
    if (force_change = 1 and
        cyr = 1) then
      err_txt = 'В строке присутствуют символы кириллицы.';
    if (force_change = 2 and
        lat = 1) then
      err_txt = 'В строке присутствуют символы латиницы.';
    if (result <> s_orig) then
      res_is_changed = 1;
    suspend;
    exit;
  end

  if (force_change in (0, 1)) then
  begin
    s_text = s_orig;
-- Меняем кириллицу на латиницу
    s_text = replace(s_text, 'А', 'A');
    s_text = replace(s_text, 'В', 'B');
    s_text = replace(s_text, 'С', 'C');
    s_text = replace(s_text, 'Е', 'E');
    s_text = replace(s_text, 'Н', 'H');
    s_text = replace(s_text, 'К', 'K');
    s_text = replace(s_text, 'М', 'M');
    s_text = replace(s_text, 'О', 'O');
    s_text = replace(s_text, 'Р', 'P');
    s_text = replace(s_text, 'Т', 'T');
    s_text = replace(s_text, 'Х', 'X');
    s_text = replace(s_text, 'а', 'a');
    s_text = replace(s_text, 'с', 'c');
    s_text = replace(s_text, 'е', 'e');
    s_text = replace(s_text, 'о', 'o');
    s_text = replace(s_text, 'р', 'p');
    s_text = replace(s_text, 'х', 'x');
    s_text = replace(s_text, 'у', 'y');
    cyr = 0;
    lat = 0;
    i = 0;
    while (i < char_length(s_text) and (cyr = 0 or lat = 0)) do
    begin
      i = i + 1;
      if (position(substring(s_text from i for 1), ' 0123456789`~!@#$%^&*()_+"№;%:?*{}[].,/-''|\IVX') = 0) then
      begin
        if (position(substring(s_text from i for 1), cyr_ch) > 0) then
          cyr = 1;
        else
          lat = 1;
      end
    end

    if (cyr = 0 or lat = 0) then
    begin
-- Принимаем то что нашли
      result = s_text;
      if (result <> s_orig) then
        res_is_changed = 1;
      suspend;
      exit;
    end
  end

  if (force_change in (0, 2)) then
  begin
-- Меняем латиницу на кириллицу
    s_text = s_orig;
    s_text = replace(s_text, 'A', 'А');
    s_text = replace(s_text, 'B', 'В');
    s_text = replace(s_text, 'C', 'С');
    s_text = replace(s_text, 'E', 'Е');
    s_text = replace(s_text, 'H', 'Н');
    s_text = replace(s_text, 'K', 'К');
    s_text = replace(s_text, 'M', 'М');
    s_text = replace(s_text, 'O', 'О');
    s_text = replace(s_text, 'P', 'Р');
    s_text = replace(s_text, 'T', 'Т');
    s_text = replace(s_text, 'X', 'Х');
    s_text = replace(s_text, 'a', 'а');
    s_text = replace(s_text, 'c', 'с');
    s_text = replace(s_text, 'e', 'е');
    s_text = replace(s_text, 'o', 'о');
    s_text = replace(s_text, 'p', 'р');
    s_text = replace(s_text, 'x', 'х');
    s_text = replace(s_text, 'y', 'у');

    cyr = 0;
    lat = 0;
    i = 0;
    while (i < char_length(s_text) and (cyr = 0 or lat = 0)) do
    begin
      i = i + 1;
      if (position(substring(s_text from i for 1), ' 0123456789`~!@#$%^&*()_+"№;%:?*{}[].,/-''|\IVX') = 0) then
      begin
        if (position(substring(s_text from i for 1), cyr_ch) > 0) then
          cyr = 1;
        else
          lat = 1;
      end
    end

    if (cyr = 0 or lat = 0) then
    begin
-- Принимаем то что нашли
      result = s_text;
      if (result <> s_orig) then
        res_is_changed = 1;
      suspend;
      exit;
    end
  end
  if (cyr = 1 and
      lat = 1) then
  begin
    result = s_orig;
    err_txt = 'В строке смешаны буквы кириллицы и латиницы.';
    if (result <> s_orig) then
      res_is_changed = 1;
    suspend;
  end
end
