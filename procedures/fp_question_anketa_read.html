CREATE PROCEDURE FP_QUESTION_ANKETA_READ
RETURNS (
    KC_ID ID_TYPE,
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    DT_FILL DATE,
    K_NO STR_20,
    C_NAME STR_100)
AS
begin
  for select
        fqa_from_kc_id,
        fqa_from_k_id,
        fqa_from_c_id,
        max(fqa_date)
      from fp_question_anketa
      where fqa_state = 0
      group by
        fqa_from_kc_id,
        fqa_from_k_id,
        fqa_from_c_id
      into
        :kc_id,
        :k_id,
        :c_id,
        :dt_fill
  do
  begin
    k_no = (select k_kontract_no from kontract where k_id = :k_id);
    c_name = (select c_name_fio from clients where c_id = :c_id);
    suspend;
  end
end
