CREATE PROCEDURE HP_SEPARATE_FIO_SET(
    C_ID D_ID,
    FNAME D_STR_100,
    MNAME D_STR_100,
    SNAME D_STR_100,
    FNAME_ENG D_STR_100,
    MNAME_ENG D_STR_100,
    SNAME_ENG D_STR_100,
    FULL_FIO D_STR_100,
    FULL_FIO_ENG D_STR_100,
    FULL_FIO_LAT D_STR_100)
AS
declare variable fl_id     d_id;
declare variable fl_id_eng d_id;
declare variable use_pt    boolean;
begin
  if (exists(select * from clients
    where c_id = :c_id and c_type = 0)) then
  begin
    use_pt = f_tmp_const('use_perstemp') = '1';
    if (use_pt) then
    begin
      if (not exists(select * from perstemp where pte_from_c_id = :c_id)) then
        update clients set
          c_name_fio = :full_fio,
          c_name_eng = :full_fio_eng,
          c_alternat_fio = :full_fio_lat
        where c_id = :c_id;
      update perstemp set
        pte_sname = :sname,
        pte_fname = :fname,
        pte_mname = :mname,
        pte_sname_eng = :sname_eng,
        pte_fname_eng = :fname_eng,
        pte_mname_eng = :mname_eng
      where pte_from_c_id = :c_id;
    end else
    begin
      select
        fl_id,
        fl_id_eng
      from hp_prepare_fio(
        :sname,
        :fname,
        :mname,
        :sname_eng,
        :fname_eng,
        :mname_eng)
      into
        :fl_id,
        :fl_id_eng;

      update clients set
        c_name_fio = :full_fio,
        c_from_fl_id = :fl_id,
        c_name_eng = :full_fio_eng,
        c_name_eng_fl_id = :fl_id_eng,
        c_alternat_fio = :full_fio_lat
      where c_id = :c_id;
    end
  end
end
