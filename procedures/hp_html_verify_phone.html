CREATE PROCEDURE HP_HTML_VERIFY_PHONE(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    TMP_HTML D_STR_8100,
    RES_IMAGE_LIST D_STR_8100)
AS
declare variable rd_id d_integer;
declare variable res_status_id d_integer;
declare variable c_name_fio d_str_100;
declare variable c_birth_date d_date;
declare variable phone_number d_str_255;
declare variable k_kontract_no d_str_20;
declare variable k_amount_approved d_integer;
declare variable indexed_adr d_str_255;
declare variable fact_adr d_str_255;
declare variable suprug_fio d_str_100;
declare variable old_phone_number d_str_100;
declare variable db_id d_id;
begin
  db_id = f_get_db_id();

  tmp_html = '';

  select c.c_name_fio, c.c_birth_date
  from clients c
  where c.c_id = :c_id
  into :c_name_fio, c_birth_date;

  tmp_html = f_format('Клиент <b>%1</b>', c_name_fio) || '<br>' ||
             f_format('Дата рождения <b>%1</b>', f_datetostr(c_birth_date)) || '<br>';

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ContactInfo.Cell_Main'
  into :rd_id;

  select res_value, res_status_id
  from hp_get_client_continfo(:c_id, :rd_id)
  into :phone_number, :res_status_id;

  select k.k_kontract_no, cast(k.k_amount_approved as integer)
  from knclient kc
  inner join kontstate ks on kc_from_k_id = ks_from_k_id and
        ks_state = 1
  inner join kontract k on ks_from_k_id = k_id
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0
  order by k_fact_disb_date desc
  rows 1
  into :k_kontract_no, :k_amount_approved;

  indexed_adr = '';

  select adl.adl_address_str
  from addrclient adc
  inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
  where adc.adc_from_c_id = :c_id and
        adc.adc_address_type = 1
  into indexed_adr;

  fact_adr = '';

  select adl.adl_address_str
  from addrclient adc
  inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
  where adc.adc_from_c_id = :c_id and
        adc.adc_address_type = 2
  into fact_adr;

  suprug_fio = '';
  select cl.c_name_fio
  from clientrelation crel
  inner join clients cl on cl.c_id = crel.crel_other_c_id
  where crel.crel_from_c_id = :c_id and
        crel_from_rel_id in ('1', '2') and
        crel_is_active = 1
  into suprug_fio;

  old_phone_number = '';
  select clt.clt_code
  from contdata cd
  inner join contlist clt on clt.clt_id = cd.cda_from_clt_id
  where cd.cda_from_c_id = :c_id and
        clt.clt_code <> '' and
        clt.clt_code <> :phone_number
  order by cd.cda_date desc
  rows 1
  into old_phone_number;

  tmp_html = tmp_html ||
             f_format('Новый номер телефона <b>%1</b>', phone_number) || '<br>' ||
             f_format('Последний займ <b>%1</b>', k_kontract_no) || '<br>' ||
             f_format('Сумма займа <b>%1</b>', k_amount_approved) || '<br><br>' ||
             f_format('Юридический адрес клиента <b>%1</b>', indexed_adr) || '<br><br>' ||
             f_format('Фактический адрес клиента <b>%1</b>', fact_adr) || '<br><br>' ||
             f_format('ФИО супруга клиента <b>%1</b>', suprug_fio) || '<br><br>';

  select res_image_list
  from hp_get_client_doc_images(:c_id, 1, 1, '', :un_id)
  into :res_image_list;
  if (db_id in (39801, 41701)) then
  begin
    tmp_html = tmp_html || 'Позвонить клиенту на' || '<b> новый номер </b>' || '<br><a href=sip:' || :phone_number || '>' || :phone_number || '</a>' || '<br>';
    tmp_html = tmp_html || 'Позвонить клиенту на' || '<b> старый номер </b>' || '<br><a href=sip:' || :old_phone_number || '>' || :old_phone_number || '</a>' || '<br>';
  end 

  suspend;
end
