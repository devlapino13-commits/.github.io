CREATE PROCEDURE HP_KNCLIENT_PDN_UPDATE_REF
RETURNS (
    RES_TEXT D_STR_LONG,
    RES_IS_ERROR D_BOOL)
AS
declare variable kc_id d_id;
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_fact_disb_date d_date;
declare variable pdr_kind d_smallint;
declare variable pdn_status_id d_smallint;
declare variable ssp_date d_date;
declare variable ki_date d_date;
declare variable txt d_str_long;
declare variable is_error d_bool;
declare variable cnt d_integer;
begin
  -- запускает пересчет ПДН по выбранным контрактам в форме "ПДН - расчет в связи с неисполнением цели по рефинансированию" (RFCRU-4309)

  res_text = '';
  res_is_error = 0;

  select count(*)
  from pdn_ref_recalclist
  where prr_is_selected = 1 and
        prr_date = current_date
  into cnt;

  if (cnt > 100) then
    exception e_message 'Кол-во выбранных договоров - ' || cnt || ', за раз можно пересчитать ПДН максимум по 100 договорам';

  txt = '';
  cnt = 0;
  pdr_kind = 2; -- пересчет ПДН через 60 дней (enum_list -> pdn_calc_source -> ref_60)

  for select kc_id,
             prr_k_id,
             prr_k_kontract_no,
             prr_k_fact_disb_date,
             prr_pdn_status_id,
             prr_ssp_date,
             prr_ki_date
      from knclient
      inner join pdn_ref_recalclist on prr_k_id = kc_from_k_id
      where prr_is_selected = 1 and
            prr_date = current_date
      into kc_id,
           k_id,
           k_kontract_no,
           k_fact_disb_date,
           pdn_status_id,
           ssp_date,
           ki_date
  do
  begin
    is_error = 0;

    if (pdn_status_id not in (5, 6)) then
    begin
      if (txt <> '') then
        txt = txt || ascii_char(10);
      txt = txt || k_kontract_no || ': не соответствует статус ПДН'; -- статус должен быть 5 - "Кандидат: КИ получена" или 6 - "Кандидат: клиент не имеет КИ"
      is_error = 1;
    end

    if (is_error = 0 and (ssp_date is null or ssp_date < current_date - 1)) then
    begin
      if (txt <> '') then
        txt = txt || ascii_char(10);
      txt = txt || k_kontract_no || ': нет актуального ССП';
      is_error = 1;
    end

    if (is_error = 0 and (ki_date is null or ki_date < current_date - 1)) then
    begin
      if (txt <> '') then
        txt = txt || ascii_char(10);
      txt = txt || k_kontract_no || ': нет актуальной КИ';
      is_error = 1;
    end

    if (is_error = 0 and exists(select *
                                from knclient_pdn
                                where kc_id = :kc_id and
                                      kcp_kind = :pdr_kind)) then
    begin
      if (txt <> '') then
        txt = txt || ascii_char(10);
      txt = txt || k_kontract_no || ': ПДН уже пересчитан';
      is_error = 1;
    end

    if (is_error = 0) then
    begin
      -- ошибок нет, запускаем перерасчет ПДН
      begin
        execute procedure hp_knclient_pdn_update(kc_id, 0, null, current_date, null, 2, 1, 'ref_60');
        cnt = cnt + 1;

        when any do
        begin
          if (txt <> '') then
            txt = txt || ascii_char(10);
          txt = txt || k_kontract_no || ': ошибка при расчете ПДН';
          is_error = 1;
        end
      end
    end
  end

  res_text = 'Кол-во успешно пересчитанных ПДН - ' || cnt;

  if (txt <> '') then
    res_text = res_text || ascii_char(10) || 'Ошибки:' || ascii_char(10) || txt;

  suspend;
end
