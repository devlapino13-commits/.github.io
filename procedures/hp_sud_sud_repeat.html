CREATE PROCEDURE HP_SUD_SUD_REPEAT(
    SMR_ID D_ID NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    NEW_SMR_ID D_ID)
AS
declare variable k_id d_id;
  declare variable kc_id d_id;
  declare variable sld_id d_id;
  declare variable slc_id d_id;
  declare variable new_slc_id d_id;
  declare variable new_sld_id d_id;
  declare variable sld_inactive d_bool;
  declare variable scd_code d_str_code;
begin
  -- новый повторный запрос в случае отказа от суда.
  if (not exists(select * from sud_mainlist
  where sml_id = :smr_id and
        sml_status = -2)) then
    exception e_message 'Данный запрос не подходит к повторной отправке.';

  insert into sud_mainreq (smr_from_k_id, smr_from_smr_id, smr_from_un_id)
  select smr_from_k_id, :smr_id, :un_id
  from sud_mainreq
  where smr_id = :smr_id
  returning smr_id, smr_from_k_id
  into new_smr_id, k_id;

  execute procedure hp_sud_set_memo(:new_smr_id, 0, 0, 'Повторная подача в суд', :un_id);

  --записать клиентов с пересчетом задолженности
  insert into sud_loanclient (slc_from_smr_id, slc_from_kc_id, slc_sum_date, slc_sum_os, slc_sum_perc,
                              slc_sum_fine, slc_sum_other, slc_sum_all, slc_auto_worldcourt, slc_auto_federalcourt)
  select :new_smr_id, slc_from_kc_id, current_date, cur_cl_os, cur_cl_perc, cur_cl_fine, 0, cur_cl_all,
         slc_auto_worldcourt, slc_auto_federalcourt
  from sud_loanclient, knclient kc, print_client_zadol(:k_id)
  where slc_from_smr_id = :smr_id
    and kc_id = slc_from_kc_id
    and kc_from_c_id = cz_from_c_id;

  -- отправить на комплектацию документов
  update sud_mainreq
  set smr_status = 2
  where smr_id = :new_smr_id;

  -- копирование документов
  for select sld_id, sld_from_slc_id, scd_code, sld_inactive
  from sud_loandocs, sud_cfgdockind
  where sld_from_smd_id = :smr_id
    and scd_id = sld_doc_kind
    and sld_inactive = 0
  into sld_id, slc_id, scd_code, sld_inactive
  do
  begin
    kc_id = (select slc_from_kc_id from sud_loanclient where slc_id = :slc_id);
    new_slc_id = (select slc_id from sud_loanclient where slc_from_kc_id = :kc_id and slc_from_smr_id = :new_smr_id);

    insert into sud_loandocs (sld_from_smd_id, sld_from_slc_id, sld_kind, sld_doc_kind, sld_status, sld_from_un_id, sld_datetime)
    select :new_smr_id, :new_slc_id, sld_kind, sld_doc_kind, sld_status, sld_from_un_id, sld_datetime
    from sud_loandocs
    where sld_id = :sld_id
    returning sld_id
    into new_sld_id;

    if (sld_inactive = 0 and scd_code not in ('court_app', 'debt_calculation')) then
      insert into sud_loanbls (slb_from_sld_id, slb_from_bls_id, slb_from_un_id)
      select :new_sld_id, slb_from_bls_id, slb_from_un_id
      from sud_loanbls
      where slb_from_sld_id = :sld_id
        and slb_is_delete = 0;
  end

  -- отправить на комплектацию документов
  update sud_maindocs
  set smd_status = 1
  where smd_id = :new_smr_id;

  suspend;
end
