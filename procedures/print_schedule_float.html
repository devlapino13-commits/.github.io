CREATE PROCEDURE PRINT_SCHEDULE_FLOAT(
    K_ID D_ID,
    LEVEL_NO D_INTEGER)
RETURNS (
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    SH_OS D_CURRENCY,
    SH_PERC D_CURRENCY,
    SH_KOMMIS D_CURRENCY,
    SH_ALL D_CURRENCY,
    SH_TOTAL D_CURRENCY,
    SH_DISB_AMOUNT D_CURRENCY)
AS
declare variable sh_date_prv d_date;
declare variable perc_new1 d_percent;
declare variable last_sh_id d_integer;
declare variable new_annu d_currency;
declare variable k_percent_per_day d_percent;
declare variable calced_perc d_currency;
declare variable level_day1 d_integer;
declare variable new_perc_per_day1 d_percent;
declare variable first_date d_date;
declare variable days_list d_str_long;
declare variable loan_amount d_currency;
declare variable orig_annu d_currency;
declare variable sh_balance d_currency;
declare variable level_day2 d_integer;
declare variable perc_new2 d_percent;
declare variable new_perc_per_day2 d_percent;
declare variable level_day d_integer;
declare variable new_perc_per_day d_percent;
declare variable k_disb_date d_date;
declare variable level_day3 d_integer;
declare variable perc_new3 d_percent;
declare variable new_perc_per_day3 d_percent;
declare variable orig_month_perc d_percent;
declare variable orig_sh_os d_currency;
declare variable sh_date_1 d_date;
declare variable sh_date_2 d_date;
declare variable tmp_dd d_date;
declare variable start_date d_date;
declare variable pr_percent_kind d_integer;
begin
  if (coalesce(k_id, 0) = 0) then
    exception e_default 'K_ID is emptry';

  if (level_no is null) then
    exception e_default 'level_no is emptry';

  if (not (level_no in (1, 2, 3))) then
    exception e_default 'Incorrect level_no: ' || level_no;

  start_date = cast('01.11.2020' as date);

  select k_percent_per_day, k_amount_approved, k_disb_date, pr_percent_kind
  from kontract, product
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into k_percent_per_day, sh_disb_amount, k_disb_date, pr_percent_kind;

  if (k_disb_date < start_date and level_no = 3) then
    exception e_default 'Incorrect level_no: ' || level_no;

  orig_month_perc = round(cast(k_percent_per_day as double precision) * 30.0, 2);

  sh_kommis = 0;

  days_list = '';
  loan_amount = 0;
  sh_date_prv = null;

  -- Если в РФ после указанной даты выдается по плавающей ставке,
  -- в качестве льготного графика отображаем то что сохранено в базе
  if (pr_percent_kind = 1 and k_disb_date >= '08.09.2021' and f_get_db_id() = 81001) then
  begin
    loan_amount = sh_disb_amount;
    sh_balance = sh_disb_amount;

    for select sh_id, sh_date, sh_os, sh_perc, sh_kommis
    from schedule
    where sh_from_k_id = :k_id
    order by sh_id
    into sh_id, sh_date, sh_os, sh_perc, sh_kommis
    do
    begin
      sh_all = sh_os + sh_perc;
      sh_total = sh_os + sh_perc;
      suspend;
    end

    Exit;
  end




  if (k_disb_date < start_date) then
  begin
    level_day1 = 2;
    perc_new1 = 8.2;
    new_perc_per_day1 = perc_new1 / 30.000000;
  
    level_day2 = 12;
    perc_new2 = 9.0;
    new_perc_per_day2 = perc_new2 / 30.000000;
  
    level_day3 = 12;
    perc_new3 = 9.0;
    new_perc_per_day3 = perc_new3 / 30.000000;

    if (level_no = 1) then
    begin
      level_day = level_day1;
      new_perc_per_day = new_perc_per_day1;
    end
  
    if (level_no = 2) then
    begin
      level_day = level_day2;
      new_perc_per_day = new_perc_per_day2;
    end

    if (level_no = 3) then
    begin
      level_day = level_day3;
      new_perc_per_day = new_perc_per_day3;
    end
  end else
  begin
    select sh_date
    from schedule
    where sh_from_k_id = :k_id and
          sh_id = 1
    into sh_date_1;

    select sh_date
    from schedule
    where sh_from_k_id = :k_id and
          sh_id = 2
    into sh_date_2;

    level_day1 = 2;
    perc_new1 = orig_month_perc + 1.0; -- от первоначального повышается на 1.0 %
    new_perc_per_day1 = perc_new1 / 30.000000;
  
    level_day2 = 7;
    perc_new2 = orig_month_perc + 2.0; -- от первоначального повышается на 2.0 %
    new_perc_per_day2 = perc_new2 / 30.000000;

    level_day3 = 16;
    perc_new3 = orig_month_perc + 2.5; -- от первоначального повышается на 2.5 %
    new_perc_per_day3 = perc_new3 / 30.000000;
  
    if (sh_date_1 is not null and sh_date_2 is not null) then
    begin
      -- Вычисляем рабочие дня соответствующие на
      tmp_dd = f_date_work_nday(sh_date_1, level_day1);
      if (tmp_dd >= sh_date_2) then
        tmp_dd = sh_date_2 - 1;
      level_day1 = datediff(day, sh_date_1, tmp_dd);
  
      tmp_dd = f_date_work_nday(sh_date_1, level_day2);
      if (tmp_dd >= sh_date_2) then
        tmp_dd = sh_date_2 - 1;
      level_day2 = datediff(day, sh_date_1, tmp_dd);
  
      tmp_dd = f_date_work_nday(sh_date_1, level_day3);
      if (tmp_dd >= sh_date_2) then
        tmp_dd = sh_date_2 - 1;
      level_day3 = datediff(day, sh_date_1, tmp_dd);
    end

    if (level_no = 1) then
    begin
      level_day = level_day1;
      new_perc_per_day = new_perc_per_day1;
    end
  
    if (level_no = 2) then
    begin
      level_day = level_day2;
      new_perc_per_day = new_perc_per_day2;
    end

    if (level_no = 3) then
    begin
      level_day = level_day3;
      new_perc_per_day = new_perc_per_day3;
    end
  end

  for select sh_id, sh_date, sh_os, sh_perc
  from schedule
  where sh_from_k_id = :k_id and
        sh_id >= 1
  order by sh_id
  into sh_id, sh_date, sh_os, sh_perc
  do
  begin
    if (sh_id > 1) then
      loan_amount = loan_amount + sh_os;

    if (orig_annu is null and sh_os <> 0) then
      orig_annu = sh_os + sh_perc;

    if (sh_date_prv is not null and sh_os <> 0) then
    begin
      if (days_list <> '') then
        days_list = days_list || ',';

      days_list = days_list || datediff(day, sh_date_prv, sh_date);
    end

    if (sh_id = 1) then
    begin
      sh_date_prv = dateadd(day, level_day, sh_date);
      first_date = sh_date_prv;
    end
    else sh_date_prv = sh_date;

    last_sh_id = sh_id;
  end

  sh_id = 1;
  sh_date = null;

  calced_perc = round((loan_amount / 100.0000 * k_percent_per_day) * level_day1, 0);

  if (level_no >= 2) then
    calced_perc = calced_perc + round((loan_amount / 100.0000 * new_perc_per_day1) * (level_day2 - level_day1), 0);

  if (level_no >= 3) then
    calced_perc = calced_perc + round((loan_amount / 100.0000 * new_perc_per_day2) * (level_day3 - level_day2), 0);

  new_annu = f_annu(loan_amount, calced_perc, 1, new_perc_per_day, days_list, 0, 0, 0, 0, 0);

  sh_balance = loan_amount;
  for select sh_id, sh_date, sh_os, sh_perc, sh_kommis
  from schedule
  where sh_from_k_id = :k_id
  order by sh_id
  into sh_id, sh_date, sh_os, sh_perc, sh_kommis
  do
  begin
    orig_sh_os = sh_os;
    if (sh_id >= 2) then
    begin
      if (sh_id = 2) then
        sh_date_prv = first_date;

      sh_perc = sh_balance / cast(100.0000 as float) * new_perc_per_day * datediff(day, sh_date_prv, sh_date);

      if (sh_id = 2) then
        sh_perc = sh_perc + calced_perc;

      sh_perc = round(sh_perc, 0);

      if (orig_sh_os <> 0) then
      begin
        if (sh_id < last_sh_id) then
        begin
          sh_os = new_annu - sh_perc;
          if (sh_os < 0) then
            sh_os = 0;
        end
        else sh_os = sh_balance;

        sh_balance = sh_balance - sh_os;
      end else sh_os = 0;

      if (sh_os < 0) then
        exception e_default 'Отрицательная основная сумма! Дождитесь решения от разработчиков.';
    end

    sh_all = sh_os + sh_perc;
    sh_total = sh_os + sh_perc;
    suspend;

    sh_date_prv = sh_date;
  end
end
