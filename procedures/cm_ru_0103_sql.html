CREATE PROCEDURE CM_RU_0103_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable RD_CODE varchar(255);
declare variable CDS_DOC_DATE_BEGIN date;
declare variable CDS_DOC_DATE_END date;
declare variable KC_CIRCLE integer;
declare variable KC_FROM_C_ID ID_TYPE;
declare variable K_KOLICH_REPAY integer;
declare variable c_curr_status integer;
begin
  for select c_name_fio,
           kc_from_c_id,
           k_disb_date,
           KC_AMOUNT_APPROVED,
           kc_circle,
           c.c_pass_no,
           c.c_pass_date,
           c.c_pass_date_expire,
           rd_code,
           k_kolich_repay
  from clients c, knclient, refdata, kontract, product
  where c_id = kc_from_c_id
  and kc_amount_approved > 0
  and c_citizen = rd_id
  and rd_main_code = 'Citizen'
  and kc_from_k_id = k_id
  and k_from_pr_id = pr_id
  and pr_code not containing '.PATENT.' --Микрозаем на патент не проверяем
  and kc_from_k_id = :k_id
  into res_info_text,
        kc_from_c_id,
        KF_DISB_DATE,
        KC_AMOUNT,
        kc_circle,
        KF_CLIENT_PASS_NO,
        CDS_DOC_DATE_BEGIN,
        CDS_DOC_DATE_END,
        rd_code,
        k_kolich_repay
  do
  begin
    select c_curr_status from hp_get_client_status(:kc_from_c_id,:kf_disb_date)
    into c_curr_status;
    /*3.1.2. до конца срока действия паспорта иностранного гражданина - повторного клиента,
    который получает заем на срок 3 - 5 месяцев, срок действия паспорта должен превышать срок
    займа не менее чем на 15 календарных дней */
    if (:kc_circle > 1
        and :k_kolich_repay between 3 and 5
        and dateadd(day, 15, :kf_disb_date) > CDS_DOC_DATE_END
        and c_curr_status < 4
         ) then
    begin
      res_error_text = 'Для повторных клиентов паспорта должны быть действительны не менее 15-дней после получения микрозайма';
      if(:rd_code in ('Citizen.TJ', 'Citizen.KG') and exists (select distinct 1
                                                                from CLIENTDOCS, PASSKIND, PASSDOCTYPE
                                                                where CDS_FROM_C_ID = :kc_from_c_id
                                                                and cds_from_pk_id = pk_id
                                                                and pk_from_pdt_id = pdt_id
                                                                and cds_is_delete = 0
                                                                and dateadd(day, 15, :kf_disb_date) < CDS_DOC_DATE_END
                                                                and :kf_disb_date between CDS_DOC_DATE_BEGIN and CDS_DOC_DATE_END
                                                                and pdt_code containing 'passport')) then
      res_error_is_critical = 0; else res_error_is_critical = 1;
      suspend;
    end 
  end
end
