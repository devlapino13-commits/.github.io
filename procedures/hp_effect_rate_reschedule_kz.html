CREATE PROCEDURE HP_EFFECT_RATE_RESCHEDULE_KZ(
    K_ID D_ID,
    OL_DATE D_DATE)
RETURNS (
    REC_ID D_INTEGER,
    SH_ID D_INTEGER,
    SH_DAYS D_INTEGER,
    SH_DATE D_DATE,
    SH_ALL D_CURRENCY)
AS
declare variable IPF_COMMISSION D_CURRENCY;
declare variable SH_DATE_PREV D_DATE;
begin
  ipf_commission = 0;

  rec_id = 0;
  select coalesce(pr.pr_ipf_commission, 0)
  from product pr
  inner join kontract kont on kont.k_from_pr_id = pr.pr_id
  where kont.k_id = :k_id
  into :ipf_commission;

  sh_id = 0;
  sh_days = 0;
  select min(sh_date), -sum(sh_os - sh_kommis) + :ipf_commission
  from schedule
  where sh_from_k_id = :k_id
  into sh_date, sh_all;
  rec_id = rec_id + 1;
  suspend;

  sh_date_prev = sh_date;
  for with tab as (
  select minvalue(ok_sh_date, ok_ol_date) sh_date, -sum(ok_os + ok_perc) as sh_all
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type in (4,6) and
        ok_ol_date <= :ol_date
  group by ok_from_k_id, 1     --  ok_from_k_id добавлено специально. Без него на FB5 сильно тормозит
  having sum(ok_os + ok_perc) <> 0
  union all
  select ok_sh_date, sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc)
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type < 7 and
        ok_ol_date <= :ol_date
  group by ok_from_k_id, ok_sh_date   --  ok_from_k_id добавлено специально. Без него на FB5 сильно тормозит
  having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 0)
  select sh_date, sum(sh_all)
  from tab
  group by 1
  order by 1
  into sh_date, sh_all
  do
  begin
    sh_id = sh_id + 1;
    sh_days = sh_date - sh_date_prev;
    rec_id = rec_id + 1;
    suspend;
    sh_date_prev = sh_date;
  end
end
