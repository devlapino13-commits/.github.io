CREATE PROCEDURE AAREP_NPL90(
    REP_DATE_END D_DATE)
RETURNS (
    BALANCE_TOTAL D_CURRENCY,
    BALANCE_PERCENT_CALC D_CURRENCY,
    BALANCE_SUM D_CURRENCY,
    BALANCE_TOTAL_DELINQ90 D_CURRENCY,
    BALANCE_PERCENT_CALC_DELINQ90 D_CURRENCY,
    BALANCE_SUM_DELINQ90 D_CURRENCY,
    NPL90 D_PERCENT)
AS
declare variable k_id d_id;
declare variable rep_date d_date;
declare variable balance_total_temp d_currency;
declare variable balance_percent_calc_temp d_currency;
begin
    rep_date = :rep_date_end;
    balance_total = 0;
    balance_percent_calc = 0;
    balance_sum = 0;
    balance_total_delinq90 = 0;
    balance_percent_calc_delinq90 = 0;
    balance_sum_delinq90 = 0;

    for
        select oks.oks_from_k_id, oks.oks_balance
        from oksummary oks
        where :rep_date between oks.oks_date_begin and oks.oks_date_end and
              oks.oks_close_state = 0
        into :k_id, balance_total_temp
    do
    begin
        select sum(sh_calc_perc)
        from aarep_make_zad_list(:k_id, :rep_date, 1)
        into :balance_percent_calc_temp;

        balance_total = balance_total + coalesce(balance_total_temp, 0);
        balance_percent_calc = balance_percent_calc + coalesce(balance_percent_calc_temp, 0);

        balance_sum = balance_sum + coalesce(balance_total_temp, 0) + coalesce(balance_percent_calc_temp, 0);

        balance_total_temp = 0;
        balance_percent_calc_temp = 0;
    end

    balance_total_temp = 0;
    balance_percent_calc_temp = 0;

    -- С просрочкой более 90 дней
    for
        select oks.oks_from_k_id, oks.oks_balance
        from oksummary oks
        where :rep_date between oks.oks_date_begin and oks.oks_date_end and
              oks.oks_close_state = 0 and
              :rep_date - oks.oks_delinq_date > 90
        into :k_id, balance_total_temp
    do
    begin
        select sum(sh_calc_perc)
        from aarep_make_zad_list(:k_id, :rep_date, 1)
        into :balance_percent_calc_temp;

        balance_total_delinq90 = balance_total_delinq90 + coalesce(balance_total_temp, 0);
        balance_percent_calc_delinq90 = balance_percent_calc_delinq90 + coalesce(balance_percent_calc_temp, 0);

        balance_sum_delinq90 = balance_sum_delinq90 + coalesce(balance_total_temp, 0) + coalesce(balance_percent_calc_temp, 0);

        balance_total_temp = 0;
        balance_percent_calc_temp = 0;
    end

    if (coalesce(balance_sum, 0) = 0) then
    begin
        suspend;
        exit;
    end

    npl90 = (coalesce(balance_sum_delinq90, 0) / balance_sum);

    suspend;
end
