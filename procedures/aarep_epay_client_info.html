CREATE PROCEDURE AAREP_EPAY_CLIENT_INFO(
    REP_C_ID D_ID,
    REP_DATE_END D_DATE)
RETURNS (
    NAME STR_255,
    SURNAME STR_255,
    MIDLENAME STR_255,
    BIRTHDAY D_DATE,
    IIN STR_100,
    CITIZEN STR_100,
    RESIDENT D_SMALLINT,
    POST_INDEX STR_20,
    REGION_ STR_255,
    COUNTRY STR_255,
    OBL STR_255,
    DISTRICT STR_255,
    CITY STR_255,
    STREET STR_255,
    DEPART_NO D_STR_20,
    BUILDING_NO D_STR_20,
    FLAT_NO D_STR_20,
    SECTION_NO D_STR_20,
    PASS_TYPE STR_100,
    PASS_SER STR_10,
    PASS_COUNTRY STR_100,
    PASS_NO D_STR_100,
    PASS_WHO STR_255,
    PASS_DATE D_DATE,
    SEX D_STR_CODE,
    DURATION_FROM DATE,
    DURATION_TO DATE,
    DOCUMENT STR_100,
    MIGRATION_SER STR_255,
    MIGRATION_NO STR_255,
    FULL_ADDRESS_REGISTR D_STR_255,
    FULL_ADDRESS_FACT D_STR_255,
    PASS_ADD_DATE D_DATE)
AS
declare variable c_id id_type;
declare variable id_ru d_integer;
declare variable c_citizen d_integer;
declare variable c_doc_type d_integer;
declare variable country_code d_str_code;
declare variable country_id d_integer;
declare variable c_citizen_code d_str_100;
declare variable doc_vnj_id d_integer;
begin
  -- для ЗУП
  country_code = case f_get_db_id()
                   when 81001 then 'ru'
                   when 41701 then 'kg'
                   when 94401 then 'az'
                 end;

  country_id = f_get_rd_id('Citizen.' || country_code);
  id_ru = f_get_rd_id('Citizen.ru');

  c_id = rep_c_id;

  for select c_id, fl_name1, --Имя
             fl_name2, --Фамилия
             fl_name3, --Отчество
             cl.c_birth_date, --Дата рождения
             coalesce(cl.c_inn, ''),
             --ИИН если имеется
             (select rd.rd_name
              from refdata rd
              where rd.rd_id = cl.c_citizen), --Гражданство
             cl.c_index_c_pass_no, --Номер пасопрта целиком и без пробелов
             cl.c_pass_who, --кем выдан
             cl.c_pass_date, --дата выдачи
             iif(cl.c_sex = 0, 'Ж', 'М'), --Пол
             c_citizen, c_doc_type, rd_code
      from clients cl, fiolist fl, refdata
      where cl.c_id = :c_id and
            fl_id = c_from_fl_id and
            c_citizen = rd_id
      into :c_id, :name, :surname, :midlename, :birthday, :iin, :citizen, :pass_no, :pass_who, :pass_date, :sex,
           :c_citizen, :c_doc_type, :c_citizen_code

  do
  begin
    --не заполненные или не ясные поля
    region_ = ''; --регион
    district = ''; --район
    resident = 0;

    migration_no = '';
    migration_ser = '';
    document = '';
    duration_from = null;
    duration_to = null;
    pass_ser = '';

    doc_vnj_id = null;
    select pk_id
    from passkind, passdoctype
    where pdt_id = pk_from_pdt_id and
          pk_is_delete = 0 and
          pdt_code = 'registration_vnj' rows 1
    into :doc_vnj_id;

    select 1 - res_non_resident
    from hp_get_client_type(:c_id, :country_id, :doc_vnj_id, :rep_date_end)
    into :resident;
    /*
    if (exists(select *
    from passdoctype, passkind, clients
    where pdt_code = 'passport_local' and
          pk_from_pdt_id = pdt_id and
          pk_country_id = :country_id and
          c_id = :c_id and
          c_doc_type = pk_id and
          :rep_date_end between c_pass_date and c_pass_date_expire)) then resident = 1;   */

    if (resident = 0) then
    begin
      if (exists(select *
                 from passdoctype, passkind, clientdocs
                 where pdt_code = 'passport_local' and
                       pk_from_pdt_id = pdt_id and
                       pk_country_id = :country_id and
                       cds_from_c_id = :c_id and
                       cds_from_pk_id = pk_id and
                       cds_is_delete = 0 and
                       :rep_date_end between cds_doc_date_begin and cds_doc_date_begin)) then
        resident = 1;
    end

    select cast(cds_create_datetime as date)
    from clientdocs
    where cds_from_pk_id = :c_doc_type and
          cds_from_c_id = :c_id and
          cds_doc_date_begin = :pass_date and
          cds_is_main_doc = 1
    order by cds_create_datetime desc
    rows 1
    into pass_add_date;

    street = '';
    depart_no = null;
    building_no = null;
    flat_no = null;
    section_no = null;
    post_index = null;
    full_address_registr = '';
    full_address_fact = '';

    country = '';
    city = '';
    obl = '';

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :c_id and
          adl_id = adc_from_adl_id and
          adc_address_type = 2 -- адрес фактического проживания в стране
    into :full_address_fact;

    --Разбивка адреса на составные
    select (select pl.pl_name
            from places pl
            where pl.pl_id = adl.adl_street_id), adl.adl_depart_no, adl.adl_building_no, adl.adl_flat_no,
            adl.adl_section_no, adl.adl_post_index
    from addrclient adc
    left join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1 -- адрес регистрации в стране
    into :street, :depart_no, :building_no, :flat_no, :section_no, :post_index;

    select coalesce(ga.country, ''), coalesce(ga.city, ''), coalesce(ga.region_name, ''), adl_address_str
    from hp_get_client_address(:c_id, 1) ga
    into :country, :city, :obl, full_address_registr; --страна, город, область

    select pk.pk_name, upper(pk.pk_country)
    from passkind pk, passdoctype pdt
    where pk.pk_id = :c_doc_type and
          pk.pk_from_pdt_id = pdt.pdt_id
    into :pass_type, :pass_country;

    --разбиваем на серию и номер паспорта
    pass_ser = '';
    pass_no = replace(pass_no, ' ', '');

    if (c_citizen = id_ru) then
    begin
      pass_ser = left(:pass_no, 4);
      pass_no = substring(:pass_no from 5);
    end
    else
    begin
      pass_ser = left(:pass_no, 2);
      pass_no = substring(:pass_no from 3);
    end

    select replace(cds_doc_no, ' ', '')
    from clientdocs cds, passkind, passdoctype
    where cds_from_c_id = :c_id and
          cds_from_pk_id = pk_id and
          pk_from_pdt_id = pdt_id and
          pdt_code = 'magration_card' and
          cds_is_delete = 0 and
          current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
    order by cds.cds_doc_date_end desc
    rows 1 --ищем самую последнюю, если несколько
    into :migration_no; --Миграционная карта номер

    if (char_length(migration_no) >= 10) then --выделяем серию и номер из полного номера миграционной карты (для 11ти зн номеров)
    begin
      migration_ser = substring(:migration_no from 1 for 4);
      migration_no = trim(substring(:migration_no from 5));
    end

    --получаем номер и страну происхождения актуального паспорта
    select pk.pk_name --, upper(pk.pk_country)
    from clientdocs cds, passkind pk, passdoctype pdt
    where cds_from_c_id = :c_id and
          cds_from_pk_id = pk.pk_id and
          pk.pk_from_pdt_id = pdt.pdt_id and
          cds.cds_is_main_doc = 1 and --основной документ
          cds.cds_is_delete = 0 and --не удаленный
          current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end --на текущую дату
          rows 1
    into :pass_type; --, :pass_country;

    if (c_citizen_code = 'citizen.kg') then
    begin
      select cds.cds_doc_date_begin, cds.cds_doc_date_end, pk.pk_name --трудовой договор (двух- и односторонний)
      from clientdocs cds, passkind pk, passdoctype pdt
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk.pk_id and
            pk.pk_from_pdt_id = pdt.pdt_id and
            ((pdt.pdt_code = 'job_agreement') or
            (pdt.pdt_code = 'job_agreement_2')) and
            cds.cds_is_delete = 0 and
            current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
      order by cds.cds_doc_date_end desc
      rows 1 --в случае нескольких действующих находим самый новый
      into :duration_from, :duration_to, :document;
    end

    if (c_citizen_code = 'citizen.tj') then
    begin
      select cds.cds_doc_date_begin, pk.pk_name --патент, срок начала действия
      from clientdocs cds, passkind pk, passdoctype pdt
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk.pk_id and
            pk.pk_from_pdt_id = pdt.pdt_id and
            pdt.pdt_code = 'work_patent' and
            cds.cds_is_delete = 0 and
            current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
      order by cds.cds_doc_date_end desc
      rows 1 --в случае нескольких действующих находим самый новый
      into :duration_from, :document;

      select cds.cds_doc_date_end --квитанция, срок окончания действия
      from clientdocs cds, passkind pk, passdoctype pdt
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk.pk_id and
            pk.pk_from_pdt_id = pdt.pdt_id and
            pdt.pdt_code = 'kvit_patent' and
            cds.cds_is_delete = 0 and
            current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
      order by cds.cds_doc_date_end desc
      rows 1 --в случае нескольких действующих находим самый новый
      into :duration_to;
    end

    if (c_citizen_code = 'citizen.uz') then
    begin
      select cds.cds_doc_date_begin, pk.pk_name --патент, срок начала действия
      from clientdocs cds, passkind pk, passdoctype pdt
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk.pk_id and
            pk.pk_from_pdt_id = pdt.pdt_id and
            pdt.pdt_code = 'work_patent' and
            cds.cds_is_delete = 0 and
            current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
      order by cds.cds_doc_date_end desc
      rows 1 --в случае нескольких действующих находим самый новый
      into :duration_from, :document;

      select cds.cds_doc_date_end --квитанция, срок окончания действия
      from clientdocs cds, passkind pk, passdoctype pdt
      where cds_from_c_id = :c_id and
            cds_from_pk_id = pk.pk_id and
            pk.pk_from_pdt_id = pdt.pdt_id and
            pdt.pdt_code = 'kvit_patent' and
            cds.cds_is_delete = 0 and
            current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
      order by cds.cds_doc_date_end desc
      rows 1 --в случае нескольких действующих находим самый новый
      into :duration_to;
    end

    suspend;
  end
end
