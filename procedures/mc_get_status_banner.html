CREATE PROCEDURE MC_GET_STATUS_BANNER(
    MCL_ID D_ID)
RETURNS (
    CORRESPOND_STATUS D_STR_100)
AS
declare variable DB_ID D_ID;
declare variable C_ID D_ID;
declare variable RD_ID D_INTEGER;
declare variable AG_ID D_INTEGER;
declare variable MCL_PREV_LOGIN_DATE D_DATE;
declare variable IS_CLIENT_CORRESPONDS_PP D_SMALLINT;
declare variable MCL_PROJECT D_STR_20;
begin
  /* Процедура для партнерской программы, возращается активность всплывающих баннеров */

  /*correspond_status:
        'partner_banner' - подписали эл.документ (Условия ПП) и стали Партнерами компании.
        'candidate_banner' - соответствуют ПП, но которые еще не подписали эл.документ (Условия ПП)
        'hide_banner' - не соответствуют условиям ПП
  */
  db_id = f_get_db_id();

  if (db_id in (41701, 39801)) then
  begin
    correspond_status = 'hide_banner';
    suspend;
    exit;
  end

  correspond_status = 'hide_banner';
  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  select
    mcl.mcl_from_c_id,
    cast(mcl.mcl_prev_login_datetime as d_date),
    f_enum('mcl_project', mcl.mcl_project, 'code', 0)
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into
    :c_id,
    :mcl_prev_login_date,
    :mcl_project;

  -- RFCRU-3766
  if (mcl_prev_login_date = current_date or mcl_project = 'oivo') then
  begin
    suspend;
    exit;
  end

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.loyal_agreement'
  into rd_id;

  ag_id = null;
  select ag_id
  from agreements -- в этом отрезке нужно добавить дополнительное условие на mcl_id
  where ag_object_kind = 0 and
        ag_object_id = :c_id and
        ag_status = 1 and
        ag_from_rd_id = :rd_id and
        current_date between ag_date_begin and coalesce(ag_date_end, '31.12.9999')
  order by ag_id desc
  rows 1
  into ag_id;

  /*
    Важно! временно были перепутаны местами candidate_banner и partner_banner т.к. в ТЗ для разработчиков фронта была ошибка. Т.е. указана не корректные банеры
    После исправления ошибки разработчками фронта, там где <-- (ниже по тексту) нужно сделать замену на корректное значение

  */

  if (ag_id is not null) then
  begin
    correspond_status = 'partner_banner';
  end
  else
  begin
    is_client_corresponds_pp = 0;
    select is_client_corresponds_pp
    from hp_is_client_corresponds_pp(:c_id)
    into :is_client_corresponds_pp;

    if (is_client_corresponds_pp = 1) then
      correspond_status = 'candidate_banner';
  end

  suspend;
end
