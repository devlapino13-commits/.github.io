CREATE PROCEDURE HP_INCOME_NEW_QUERY(
    MAC_ID D_ID,
    UN_ID D_ID,
    CONFIG D_STR_CODE)
AS
declare variable inn d_str_20;
declare variable eic_id d_id;
declare variable eil_id d_id;
declare variable icq_json d_str_255;
declare variable c_id d_id;
declare variable counter d_integer;
declare variable un_fio d_str_100;
declare variable requestId d_str_code;
begin
-- RFCKG-4255 Отправка запроса на получение Справки о доходах (ЗП, пенсия).
-- RFCRU-3940 Справка АСП
  counter = 0;

  select un.un_fio
  from username un
  where un.un_id = :un_id
  into un_fio;

  for select mac.mac_from_c_id, c.c_inn
  from miniappclient mac
  inner join clients c on c.c_id = mac.mac_from_c_id
  where mac.mac_id = :mac_id and
        mac.mac_amount > 0
  into c_id, inn
  do
  begin
    if (coalesce(:inn, '') = '') then
      exception e_message 'у клиента отсутствует ИНН';

    select cfg.eic_id
    from extinfocfg cfg
    where cfg.eic_code = :config
    into :eic_id;
  
    if (coalesce(:eic_id, '') = '') then
      exception e_message f_format('Конфигурация extinfocfg = %1 не найдена', config);
  
    execute procedure hp_extinfo_insert(:config, 'clients', :c_id, 0, :un_id);
  
    select eil.eil_id
    from extinfolist eil
    where eil.eil_from_c_id = :c_id
    and eil.eil_from_eic_id = :eic_id
    order by eil.eil_id desc
    rows 1
    into :eil_id;
  
    requestId = uuid_to_char(f_md5_uuid(:eil_id));
    icq_json = f_format('{"iin": "%1", "employee_name": "%2", "requestId": "%3", "consent": 5, "request_type": "DEDUCTION_6", "consent_token_term": "120"}', inn, un_fio, requestId);


    insert into income_check_querys (icq_from_un_id, icq_from_c_id, icq_from_mac_id, icq_from_eil_id, icq_json, icq_config)
    values (:un_id, :c_id, :mac_id, :eil_id, :icq_json, :config);

    counter = counter + 1;
  end
  if (counter > 0) then
    post_event :config;
end
