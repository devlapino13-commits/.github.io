CREATE PROCEDURE HP_GET_CLIENT_PASSPORT_HISTORY(
    C_ID D_ID)
RETURNS (
    RES_PASS_SERIA D_STR_10,
    RES_PASS_NUMBER D_STR_CODE,
    RES_DOC_TYPE D_SMALLINT,
    RES_DOC_DATE D_DATE,
    RES_DOC_NO D_STR_CODE)
AS
declare variable pk_id d_integer;
begin
    -- RFCRU-3312 Процедура возвращает ВСЕ паспорта клиента В том числе и с истекшим сроком действия
    for select distinct  pk_id, cds_doc_no, f_datetostr(cds_doc_date_begin)
    from clientdocs
    inner join passkind on cds_from_pk_id = pk_id
    inner join passdoctype on pk_from_pdt_id = pdt_id
    where cds_from_c_id = :c_id
    and pdt_is_main_doc = 1
    and cds_verify_status = 2
    and cds_doc_date_begin is not null
    and coalesce(cds_doc_no, '') <> ''
    order by cds_create_datetime desc -- Сначала свежие, чтобы быстрее найти
    into pk_id,
         res_doc_no,
         res_doc_date
    do
    begin
        -- Обработка номера паспорта
        if (pk_id = 12) then
           res_doc_type = 21;
        else
           res_doc_type = 10;

        select pass_seria,
               pass_number
        from split_passport_number(:pk_id, :res_doc_no) -- RFCKG-3780 Разбивка на серию и номер паспорта
        into :res_pass_seria,
             :res_pass_number;
        suspend;
    end
end
