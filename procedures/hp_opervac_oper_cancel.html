CREATE PROCEDURE HP_OPERVAC_OPER_CANCEL(
    K_ID D_ID,
    IN_OL_ID D_ID)
AS
declare variable ol_date d_date;
declare variable ol_id_tmp d_id;
begin
  if (exists(select * from hp_db_is_actual)) then
    exception e_message 'Нельзя удалять на боевой БД';

  uxf$flag('operdel_executing', 'on');
  ol_date = (select ol_date from operlist where ol_id = :in_ol_id);

  for select distinct coalesce(ol_parent_id, ol_id)
  from operkred, operlist
  where ol_id = ok_from_ol_id
    and ok_from_k_id = :k_id
    and ol_id >= :in_ol_id
    and ol_date >= :ol_date
    and ok_type in (3,4)
    and is_delete = 0
  order by ol_date desc, coalesce(ol_parent_id, ol_id) desc
  into ol_id_tmp
  do
    update operlist
    set is_delete = 1
    where ol_id = :ol_id_tmp;

  uxf$flag('operdel_executing', 'off');

  if (ol_id_tmp is null) then
    exception e_message 'Не найдены операции для удаления';
    
end
