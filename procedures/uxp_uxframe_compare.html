CREATE PROCEDURE UXP$UXFRAME_COMPARE(
    FR_UUID_1 D_UUID,
    FR_UUID_2 D_UUID)
RETURNS (
    CLONE_KIND D_STR_CODE)
AS
begin
  clone_kind = '';
  if (fr_uuid_1 is not distinct from fr_uuid_2) then
    exception e_default 'fr_uuid_1 and fr_uuid_2 have to different';

  for select ufr_uuid, ufr_owner, ufr_us_uuid, ufr_master_ufr_uuid,
             ufr_master_field, ufr_detail_field, ufr_sql_where, ufr_sql_orderby
  from ux$frame
  where ufr_uuid = :fr_uuid_1
  as cursor fr1
  do
  begin
    for select ufr_uuid, ufr_owner, ufr_us_uuid, ufr_master_ufr_uuid,
               ufr_master_field, ufr_detail_field, ufr_sql_where, ufr_sql_orderby
    from ux$frame
    where ufr_uuid = :fr_uuid_2
    as cursor fr2
    do
    begin
      if (fr1.ufr_us_uuid = fr2.ufr_us_uuid) then
        clone_kind = 'link';

      if (fr1.ufr_owner is distinct from fr2.ufr_owner) then
        clone_kind = '';

      if (fr1.ufr_master_ufr_uuid is distinct from fr2.ufr_master_ufr_uuid) then
        clone_kind = '';

      if (fr1.ufr_master_field is distinct from fr2.ufr_master_field) then
        clone_kind = '';

      if (fr1.ufr_detail_field is distinct from fr2.ufr_detail_field) then
        clone_kind = '';

      if (clone_kind = 'link' and fr1.ufr_sql_where is distinct from fr2.ufr_sql_where) then
        clone_kind = 'clone';

      if (clone_kind = 'link' and fr1.ufr_sql_orderby is distinct from fr2.ufr_sql_orderby) then
        clone_kind = 'clone';
    end
  end


  if (clone_kind <> '' and (exists(select *
      from ux$frdata fd1, ux$frdata fd2
      where fd1.ufd_ufr_uuid = :fr_uuid_1 and
            fd2.ufd_ufr_uuid = :fr_uuid_2 and
            fd1.ufd_kind = fd2.ufd_kind and
            fd1.ufd_us_uuid <> fd2.ufd_us_uuid))) then
    clone_kind = '';

  if (clone_kind <> '') then
  begin
    for select upd_uxa_uuid, upd_def_value, upd_locked, ufp_uf_uuid, uf_is_param
    from ux$frparam, ux$paramdef, ux$field
    where ufp_ufr_uuid = :fr_uuid_1 and
          ufp_uuid = upd_ufp_uuid and
          ufp_uf_uuid = uf_uuid
    as cursor fp
    do
    begin
      if (not exists(select * from ux$frparam, ux$paramdef
      where ufp_ufr_uuid = :fr_uuid_2 and
            ufp_uuid = upd_ufp_uuid and
            ufp_uf_uuid = fp.ufp_uf_uuid and
            upd_uxa_uuid is not distinct from fp.upd_uxa_uuid and
            upd_def_value is not distinct from fp.upd_def_value and
            upd_locked is not distinct from fp.upd_locked)) then
      begin
        if (fp.uf_is_param = 1) then
        begin
          clone_kind = '';
          break;
        end else
        if (clone_kind <> '') then
          clone_kind = 'clone';
      end
    end
  end

  if (clone_kind <> '') then
  begin
    for select upd_uxa_uuid, upd_def_value, upd_locked, ufp_uf_uuid, uf_is_param
    from ux$frparam, ux$paramdef, ux$field
    where ufp_ufr_uuid = :fr_uuid_2 and
          ufp_uuid = upd_ufp_uuid and
          ufp_uf_uuid = uf_uuid
    as cursor fp
    do
    begin
      if (not exists(select * from ux$frparam, ux$paramdef
      where ufp_ufr_uuid = :fr_uuid_1 and
            ufp_uuid = upd_ufp_uuid and
            ufp_uf_uuid = fp.ufp_uf_uuid and
            upd_uxa_uuid is not distinct from fp.upd_uxa_uuid and
            upd_def_value is not distinct from fp.upd_def_value and
            upd_locked is not distinct from fp.upd_locked)) then
      begin
        if (fp.uf_is_param = 1) then
        begin
          clone_kind = '';
          break;
        end else
        if (clone_kind <> '') then
          clone_kind = 'clone';
      end
    end
  end

  if (clone_kind <> 'link') then
    clone_kind = '';

  suspend;
end
