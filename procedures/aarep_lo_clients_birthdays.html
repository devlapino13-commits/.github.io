CREATE PROCEDURE AAREP_LO_CLIENTS_BIRTHDAYS(
    UN_ID DECIMAL(18,0),
    DATE_END DATE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable lo_id decimal(18,0);
declare variable rep_day integer;
declare variable rep_month integer;
declare variable k_id decimal(18,0);
declare variable c_name_fio varchar(100);
declare variable kf_name varchar(100);
declare variable cell_no varchar(100);
declare variable kol integer;
declare variable c_year integer;
declare variable db_id integer;
declare variable no_leap_year integer;
declare variable c_id d_id;
begin
  db_id = f_get_db_id();

  select un_from_c_id
  from username
  where un_id = :un_id
  into :lo_id;
  if (lo_id is null) then
    exit;

  res_text = (select res_style
              from fm_html_styles) || '<body>';
  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Список клиентов у кого сегодня День рождения', 'Carı günə ad günü olan müştərilərin siyahısı')));
  res_text = BLOB_APPEND(res_text, f_format('<h2>%1 :)</h2>', iif(db_id <> 94401, 'Они будут рады вашему звонку', 'Onlar sizin zənginizə şad olacaqlar')));
  res_text = BLOB_APPEND(res_text, '<table width="100%">
       <col width="65%">
       <col width="35%">
       <tr>' || f_format('<th>%1</th><th>%2</th></tr>', iif(db_id <> 94401, 'Клиенты', 'Müştərilər'),
                                                        iif(db_id <> 94401, 'Телефон', 'Nömrə')));

  rep_day = extract(day from date_end);
  rep_month = extract(month from date_end);
  kol = 0;
  no_leap_year = 0;
  if (rep_month * 100 + rep_day = 301 and
      extract(month from dateadd(day, -1, date_end)) * 100 + extract(day from dateadd(day, -1, date_end)) = 228) then
    no_leap_year = 1; -- не високосный год

  for select oks_from_k_id
      from oksummary
      where oks_kc_id = :lo_id and
            oks_date_begin <= :date_end and
            oks_date_end >= :date_end and
            oks_balance > 0
            and oks_close_state = 0
      into :k_id
  do
  begin
    for select c_id, c_name_fio,
               (select cont_value
                from hp_contdata_get_cell_valid(0, c_id)), extract(year from c_birth_date)
        from clients, knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = c_id and
              (extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = :rep_month * 100 + :rep_day
              or (:no_leap_year = 1
                  and extract(month from c_birth_date) * 100 + extract(day from c_birth_date) = 229))
        into c_id, c_name_fio, cell_no, c_year
    do
    begin
      kol = kol + 1;
      kf_name = '';
      select kal_new_kf_name
      from kontract, knaktlist
      where kal_from_kf_id = k_from_kf_id and
            k_id = :k_id and
            knaktlist.is_delete = 0 and
            kal_date <= :date_end and
            kal_date_end >= :date_end
      into :kf_name;
      res_text = blob_append(res_text, f_format('<tr><td>%1, %2 %3.</td><td align="center">%4</br>%5</td></tr>', c_name_fio, c_year, iif(db_id <> 94401,'г', 'г'), :kf_name, f_uri_phone(cell_no, 0, c_id)));
    end
  end
  if (kol = 0) then
    res_text = blob_append(res_text, f_format('<tr><td>%1</td><td align="center"></td></tr>', iif(db_id <> 94401, 'Сегодня нет именинников', 'Carı günə doğum günü olan yoxdur')));
  res_text = blob_append(res_text, '</table></body>');
  suspend;
end
