CREATE PROCEDURE BLANK_AKT_ECASH_MAIN(
    ID D_INTEGER)
RETURNS (
    AKT_ID D_INTEGER,
    AKT_KIND D_STR_255,
    AKT_OPER_NAME D_STR_100,
    AKT_OFFICE_NAME D_STR_100,
    AKT_DATE D_DATE,
    AKT_PERSON_FIO D_STR_100,
    AKT_PERSON_INFO D_STR_LONG,
    AKT_INV_AMOUNT D_CURRENCY,
    AKT_FACT_AMOUNT D_CURRENCY,
    AKT_DIFF_AMOUNT D_CURRENCY,
    AKT_DIFF_ABS_AMOUNT D_CURRENCY,
    AKT_DIFF_SIGN D_SMALLINT,
    AKT_DIFF_KIND D_STR_CODE,
    AKT_SHOW_FACT D_BOOL,
    AKT_HEADER D_STR_LONG)
AS
declare variable MAIN_AKT_ID D_INTEGER;
declare variable FACT_AKT_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable OPER_KIND D_STR_CODE;
declare variable OPC_DOC_NO D_INTEGER;
declare variable OPC_SIGN D_SMALLINT;
begin
  select opc_from_cil_id, opc_doc_no, opc_sign, coalesce(opc_from_c_id, opc_cashier_id),
         f_enum_code('cashoperkind', opc_oper_kind),
         f_enum_name('cashoperkind', opc_oper_kind)
  from opercash
  where opc_id = :id
  into akt_id, opc_doc_no, opc_sign, c_id,
       oper_kind, akt_oper_name;

  if (row_count = 0) then
  begin
    select cil_id, null, 0, cil_cashier_id,
           f_enum_code('cashoperkind', cil_oper_kind),
           f_enum_name('cashoperkind', cil_oper_kind)
    from cashinvlist
    where cil_id = :id
    into akt_id, opc_doc_no, opc_sign, c_id,
         oper_kind, akt_oper_name;

    select coalesce(opc_from_c_id, opc_cashier_id)
    from opercash
    where opc_from_cil_id = :id
    and opc_parent_id is null
    into :c_id;
  end

  if (akt_id is null) then
    exit;

  akt_inv_amount = 0;
  akt_fact_amount = 0;
  akt_person_fio = '';
  akt_person_info = '';

  if (c_id is not null) then
  begin
    select
      c_name_fio,
      f_format('паспорт %1 выданный %2 от %3',
        c_pass_no,
        c_pass_who,
        f_datetostr(c_pass_date))
    from clients
    where c_id = :c_id
    into
      :akt_person_fio,
      :akt_person_info;
  end

  select cil_parent_id, cil_inv_amount, cil_date, reg_code || ' ' || reg_name
  from cashinvlist
       inner join region on cil_office_id = reg_unicode
  where cil_id = :akt_id
  into main_akt_id, akt_inv_amount, akt_date, akt_office_name;

  if (main_akt_id is not null) then
  begin
    fact_akt_id = akt_id;
    akt_fact_amount = akt_inv_amount;
    akt_inv_amount = 0;

    select cil_inv_amount,
           f_enum_code('cashoperkind', cil_oper_kind),
           f_enum_name('cashoperkind', cil_oper_kind)
    from cashinvlist
    where cil_id = :main_akt_id
    into akt_inv_amount, oper_kind, akt_oper_name;
  end
  else
  begin
    main_akt_id = akt_id;

    select cil_id, cil_inv_amount
    from cashinvlist
    where cil_id = :main_akt_id
    into fact_akt_id, akt_fact_amount;
  end

  akt_show_fact = 0;

  akt_diff_kind = 'Суммы совпадают';
  akt_diff_sign = 0;
  akt_diff_abs_amount = 0;

  akt_diff_amount = akt_fact_amount - akt_inv_amount;
  if (akt_diff_amount > 0) then
  begin
    akt_diff_sign = 1;
    akt_diff_kind = 'Излишка';
    akt_diff_abs_amount = akt_diff_amount;
  end else
  if (akt_diff_amount < 0) then
  begin
    akt_diff_sign = -1;
    akt_diff_kind = 'Недостача';
    akt_diff_abs_amount = -akt_diff_amount;
  end

  akt_show_fact = 0; -- Показывать ФАКТическое кол-во купюр
  akt_header = ''; -- Заголовочная часть описания акта
  akt_kind = ''; -- Краткое описание акта согласно OPER_KIND

  suspend;
end
