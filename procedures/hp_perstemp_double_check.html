CREATE PROCEDURE HP_PERSTEMP_DOUBLE_CHECK(
    FIND_C_ID DECIMAL(18,0),
    FIND_FIO VARCHAR(255),
    FIND_PASS_NO VARCHAR(30),
    FIND_INN VARCHAR(20),
    FIND_BIRTH_DATE DATE,
    FIND_CELL_NO VARCHAR(20))
RETURNS (
    RES_DOUBLES_COMMENT D_STR_100,
    RES_DOUBLE_IS_CRITICAL D_INTEGER)
AS
declare variable found_count d_integer;
begin
  res_doubles_comment = '';

  select count(*), sum(iif(c_comment_id = 1, 1, 0))
  from hp_perstemp_double_list(:find_c_id, :find_fio, :find_pass_no, :find_inn, :find_birth_date, :find_cell_no)
  into :found_count, :res_double_is_critical;

  if (found_count is null) then
    found_count = 0;

  if (res_double_is_critical is null) then
    res_double_is_critical = 0;

  if (found_count > 0) then
    res_doubles_comment = 'Похожие: ' || found_count;

  if (res_double_is_critical > 0) then
    res_doubles_comment = res_doubles_comment || ', критичные: ' || res_double_is_critical;

  suspend;
end
