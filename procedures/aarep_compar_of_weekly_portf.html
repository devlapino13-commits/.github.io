CREATE PROCEDURE AAREP_COMPAR_OF_WEEKLY_PORTF(
    REP_REGION_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    LO_ID ID_TYPE)
RETURNS (
    MSG_TXT STR_100,
    OKS_REGION_ID DECIMAL(18,0),
    OKS_KC_ID DECIMAL(18,0),
    REG_NAME VARCHAR(100),
    KF_OFFICER_NAME VARCHAR(100),
    C_STAGE INTEGER,
    OKS_KC_BALANCE DECIMAL(18,4),
    NOT_OWN_CL_QTY INTEGER,
    IS_OWN_CL_QTY INTEGER,
    NOT_OWN_PAR30 DECIMAL(18,7),
    IS_OWN_PAR30 DECIMAL(18,7),
    OKS_KC_BALANCE_OLD DECIMAL(18,4),
    NOT_OWN_CL_QTY_OLD INTEGER,
    IS_OWN_CL_QTY_OLD INTEGER,
    NOT_OWN_PAR30_OLD DECIMAL(18,7),
    IS_OWN_PAR30_OLD DECIMAL(18,7),
    OKS_KC_BALANCE_DIFF DECIMAL(18,4),
    NOT_OWN_CL_QTY_DIFF INTEGER,
    IS_OWN_CL_QTY_DIFF INTEGER,
    NOT_OWN_PAR30_DIFF DECIMAL(18,7),
    IS_OWN_PAR30_DIFF DECIMAL(18,7))
AS
begin
  /* comparison of weekly portfolio indicators for employees -
  Процедура готовить данные по портфелю в разрезе свои/переданные по одному или по всем
  кредитным специалистам(випс) и сверяет с данными за неделю назад. */

  if (coalesce(rep_date_begin, rep_date_end) >= rep_date_end) then
  begin
    rep_date_begin = dateadd(week, -1, rep_date_end);
  end
  msg_txt = 'Данные за период с ' ||(select sval
                                                    from datetostr(:rep_date_begin)) || ' по ' ||(select sval
                                                                                                    from datetostr(:rep_date_end));

  if (coalesce(lo_id, 0) > 1000000000003) then
  begin
    if (coalesce(rep_region_id, 0) = 0) then
      rep_region_id = 1000000;
  end
  else
    lo_id = 0;

  for select oks_region_id, oks_kc_id, reg_name, kf_officer_name, c_stage, oks_kc_balance, not_own_cl_qty,
             is_own_cl_qty, not_own_par30, is_own_par30
      from aarep_vips_kf_info(:rep_region_id, :rep_date_end, :lo_id)
      into :oks_region_id, :oks_kc_id, :reg_name, :kf_officer_name, :c_stage, :oks_kc_balance, :not_own_cl_qty,
           :is_own_cl_qty, :not_own_par30, :is_own_par30
  do
  begin
    oks_kc_balance_old = null;
    not_own_cl_qty_old = null;
    is_own_cl_qty_old = null;
    not_own_par30_old = null;
    is_own_par30_old = null;
    for select oks_kc_balance, not_own_cl_qty, is_own_cl_qty, not_own_par30, is_own_par30
        from aarep_vips_kf_info(:oks_region_id, :rep_date_begin, :oks_kc_id)
        into :oks_kc_balance_old, :not_own_cl_qty_old, :is_own_cl_qty_old, :not_own_par30_old, :is_own_par30_old
    do
    begin

      oks_kc_balance_diff = oks_kc_balance - oks_kc_balance_old;
      not_own_cl_qty_diff = not_own_cl_qty - not_own_cl_qty_old;
      is_own_cl_qty_diff = is_own_cl_qty - is_own_cl_qty_old;
      not_own_par30_diff = not_own_par30 - not_own_par30_old;
      is_own_par30_diff = is_own_par30 - is_own_par30_old;

      suspend;
    end
  end
end
