CREATE PROCEDURE HP_PARSE_QUOTA_CODE(
    QUOTA_CODE D_STR_CODE)
RETURNS (
    QUOTA_MQC_ID D_INTEGER,
    QUOTA_INTERVAL D_INTEGER,
    QUOTA_MAX_COUNT D_INTEGER,
    QUOTA_CURRENT_COUNT D_INTEGER)
AS
declare variable pos_t d_integer;
declare variable pos_c d_integer;
declare variable pos_f d_integer;
begin
  /*
    Процедура парсит значения из quota_code, возвращаемого процедурой hp_msgquota_process
    quota_code имеет вид i%1t%2c%3f%4, где:
      - %1: ID конфигурации msgquotacfg (mqc_id)
      - %2: Интервал конфигурации
      - %3: Максимально допустимое количество сообщений по квоте
      - %4: Текущее количество сообщений
  */

  quota_mqc_id = -1;
  quota_interval = -1;
  quota_max_count = -1;
  quota_current_count = -1;

  pos_t = position('t' in :quota_code);
  pos_c = position('c' in :quota_code);
  pos_f = position('f' in :quota_code);

  if (left(coalesce(:quota_code, ''), 1) = 'i' and
      pos_t > 1 and
      pos_c > pos_t and
      pos_f > pos_c) then
  begin
    quota_mqc_id =
      cast(substring(:quota_code from 2 for :pos_t - 2) as d_integer);

    quota_interval =
      cast(substring(:quota_code from :pos_t + 1 for :pos_c - :pos_t - 1) as d_integer);

    quota_max_count =
      cast(substring(:quota_code from :pos_c + 1 for :pos_f - :pos_c - 1) as d_integer);

    quota_current_count =
      cast(substring(:quota_code from :pos_f + 1) as d_integer);
  end

  suspend;
end
