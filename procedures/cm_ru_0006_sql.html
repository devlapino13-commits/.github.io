CREATE PROCEDURE CM_RU_0006_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable KF_DISB_DATE D_DATE;
declare variable KF_CLIENT_PASS_NO varchar(30);
declare variable C_CURR_BONUS_STATUS integer;
declare variable KC_CIRCLE integer;
declare variable PR_PRODUCT_KIND smallint;
declare variable C_ID ID_TYPE;
declare variable C_VERIFY_STATUS integer;
declare variable C_SEX D_SMALLINT;
begin
--        проверка верификации для рф базы
  for
  select kc_from_c_id,
         kc_amount_approved,
         kc_circle,
         pr_product_kind,
         k_disb_date,
         c_verify_status,
         c_name_fio,
         c_pass_no,
         c_sex
  from knclient, kontract, product, clients
  where kc_from_k_id = :k_id
  and kc_amount_approved > 0
  and kc_from_k_id = k_id
  and k_from_pr_id = pr_id
  and kc_from_c_id = c_id
  into c_id,
       kc_amount,
       kc_circle,
       pr_product_kind,
       kf_disb_date,
       c_verify_status,
       res_info_text,
       kf_client_pass_no,
       c_sex
  do
  begin
      c_curr_bonus_status = null;
      if (kc_amount < 20000 and kc_circle > 1 and pr_product_kind = 2) then
      begin
        select c_curr_bonus_status  -- новый кредит еще не выдан, поэтому смотрим на текущий статус
        from hp_kontstate_client_status(:c_id, :kf_disb_date)
        into :c_curr_bonus_status;
      end
      if (c_curr_bonus_status is null) then c_curr_bonus_status = 0;
    
      if (c_verify_status not in(-1,2)) then
      if (c_verify_status = -2) then
      begin
        res_error_text = 'верификация: данные не верны.';
        res_error_is_critical = 2;
        if (c_curr_bonus_status > 0) then res_error_is_critical = 0;
        else suspend;
      end else
      if (c_verify_status = 0) then
      begin
        res_error_text = 'верификация: данные не полны.';
        res_error_is_critical = 2;
        if (c_curr_bonus_status > 0) then res_error_is_critical = 0;
        else suspend;
      end else
      if (c_verify_status =1) then
      begin
        res_error_text = 'верификация: верификация не проведена.';
        res_error_is_critical = 2;
        if (c_curr_bonus_status > 0) then res_error_is_critical = 0;
        else suspend;
      end
      if (c_sex not in (0,1) or c_sex is null) then
      begin
        res_error_text = 'Пол клиента не правильно указан, прошу исправить.';
        res_error_is_critical = 2;
        suspend;
      end 
  end
end
