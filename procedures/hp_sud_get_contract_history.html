CREATE PROCEDURE HP_SUD_GET_CONTRACT_HISTORY(
    REP_K_ID D_ID)
RETURNS (
    STAGE_CODE D_STR_CODE,
    STAGE_STR D_STR_255,
    STATUS_USER D_STR_100,
    STATUS_DATE_STR D_STR_CODE)
AS
declare variable REQUEST_ID D_ID;
declare variable STAGE_ID D_INTEGER;
declare variable STATUS D_INTEGER;
declare variable SUL_FROM_UN_ID D_ID;
declare variable MIN_SLD_STATUS D_INTEGER;
declare variable PRESENT_DOC_OFFICE D_BOOL;
declare variable PRESENT_DOC_ARCHIVE D_BOOL;
declare variable DECISION_BLS_ID D_ID;
declare variable LAWSUIT_NO D_STR_100;
declare variable EXDOC_BLS_ID D_ID;
declare variable SLC_ID D_ID;
declare variable SLC_ID_TMP D_ID;
declare variable STAGE_CODE_TMP D_STR_CODE;
declare variable PARAM_JSON D_STR_255;
begin
  -- Цикл по всей истории логов для конкретного контракта
  for select 
      smr.smr_id,
      sul.sul_object_kind,
      sul.sul_status,
      f_datetimetostr(sul.sul_datetime),
      sul.sul_from_un_id
  from sud_mainreq smr
  inner join sud_mainlog sul on sul.sul_from_smr_id = smr.smr_id
  where smr.smr_from_k_id = :rep_k_id
  order by sul.sul_datetime desc, sul.sul_id desc
  into :request_id, :stage_id, :status, :status_date_str, :sul_from_un_id
  do
  begin
    status_user = (select un_fio from username where un_id = :sul_from_un_id);
    
    -- Определение базового кода этапа (как в оригинале)
    stage_code = case stage_id
                    when f_objkind('sud_mainreq',     'id') then 'R'
                    when f_objkind('sud_maindocs',    'id') then 'D'
                    when f_objkind('sud_mainpay',     'id') then 'P'
                    when f_objkind('sud_mainlist',    'id') then 'J'
                    when f_objkind('sud_maincollect', 'id') then 'JC'
                    else 'error'
                  end;
    if (stage_code = 'error') then
      continue;

    -- ВЕТКА R (Заявка)
    if (stage_code = 'R') then
    begin
      stage_code = case status
                     when 0 then '|R1|'
                     when 1 then '|R2|'
                     when -1 then '|R1|'
                     when -2 then '|R3|'
                     else '|R0|'
                   end;
    end

    -- ВЕТКА D (Документы)
    else if (stage_code = 'D') then
    begin
      select
        min(iif(scd_on_request > 1 or scd_on_client > 1, sld_status, 2)),
        max(iif(sld_kind = 1 and sld_status < 1, 1, 0)),
        max(iif(sld_kind = 2 and sld_status < 1, 1, 0))
      from sud_loandocs, sud_cfgdockind
      where sld_from_smd_id = :request_id
        and sld_inactive = 0
        and scd_id = sld_doc_kind
      into min_sld_status, present_doc_office, present_doc_archive;

      stage_code = '';
      if (status = 0) then stage_code = '|D1|';
      else if (status < 0) then stage_code = '|D6|';
      else if (status = 1) then
      begin
        if (min_sld_status < 2) then stage_code = stage_code || '|D2|';
        if (present_doc_office = 1) then stage_code = stage_code || '|D3|';
        if (present_doc_archive = 1) then stage_code = stage_code || '|D4|';
        if (min_sld_status = 2) then stage_code = stage_code || '|D5|';
      end
      if (stage_code = '') then stage_code = '|D0|';
    end

    -- ВЕТКА P (Оплата)
    else if (stage_code = 'P') then
    begin
      stage_code = case status
                     when 0 then '|P1|'
                     when 1 then '|P1|'
                     when -1 then '|P2|'
                     when -2 then '|P2|'
                     else '|P0|'
                   end;
    end

    -- ВЕТКА J (Суд/Иски)
    else if (stage_code = 'J') then
    begin
      select sml_lawsuit_no, sml_decision_bls_id, sml_exdoc_bls_id
      from sud_mainlist where sml_id = :request_id
      into lawsuit_no, decision_bls_id, exdoc_bls_id;

      stage_code = case
                     when status = 0 then '|J1|'
                     when status = 2 then '|J5|'
                     when status = -2 then '|J6|'
                     when decision_bls_id is null then '|J2|'
                     when decision_bls_id is not null and lawsuit_no is null then '|J3|'
                     when decision_bls_id is not null and lawsuit_no is not null and exdoc_bls_id is null then '|J4|'
                     else '|J0|'
                   end;
    end

    -- ВЕТКА JC (Взыскание)
    else if (stage_code = 'JC') then
    begin
      stage_code = '';
      slc_id_tmp = 0;
      param_json = f_json_str_set_value('{}', 'request_id', request_id);

      for select slc_id, stage_code
      from hp_sud_get_clientslist_ux(f_json_str_set_value(f_json_str_set_value(:param_json, 'param', 'C0'), 'stage_code', '|C0|'))
      into slc_id, stage_code_tmp
      do
      begin
        if (slc_id_tmp <> slc_id) then
        begin
          slc_id_tmp = slc_id;
          if (stage_code not containing stage_code_tmp) then
            stage_code = stage_code || stage_code_tmp;
        end
      end 
      if (stage_code = '') then stage_code = '|JC0|'; -- Технический код, если пусто
    end

    -- Формирование итоговой строки этапа (как в оригинале)
    stage_str = (select list(name || trim(iif(:status = -1, '_(Доработка)', ''))) 
                 from hp_sud_get_stages 
                 where :stage_code containing ('|' || code || '|'));

    suspend;
  end
end
