CREATE PROCEDURE AAREP_CALC_PERC_BUS_ALL(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_BALANCE_BEGIN DECIMAL(18,7),
    AAREP_PERC_BEGIN DECIMAL(18,7),
    AAREP_PERC_CALCED DECIMAL(18,7),
    AAREP_PERC_PAYED DECIMAL(18,7),
    AAREP_PERC_WRITEOFF DECIMAL(18,7),
    AAREP_PERC_END DECIMAL(18,7),
    AAREP_BALANCE_END DECIMAL(18,7),
    AAREP_BUSINESS VARCHAR(100),
    AAREP_BUS_TYPE VARCHAR(100),
    KNC_WORK_KOLICH INTEGER)
AS
begin
  /* Procedure Text */
  for select sum(aarep_balance_begin), sum(aarep_perc_begin), sum(aarep_perc_calced), sum(aarep_perc_payed),
             sum(aarep_perc_writeoff), sum(aarep_perc_end), sum(aarep_balance_end), sum(knc_work_kolich),
             aarep_business, aarep_bus_type
      from aarep_calc_perc_business(:report_region_id, :report_date_begin, :report_date_end)
      group by aarep_business, aarep_bus_type
      order by aarep_bus_type, aarep_business
      into :aarep_balance_begin, :aarep_perc_begin, :aarep_perc_calced, :aarep_perc_payed, :aarep_perc_writeoff,
           :aarep_perc_end, :aarep_balance_end, :knc_work_kolich, :aarep_business, :aarep_bus_type
  do
  begin
    suspend;
  end
end
