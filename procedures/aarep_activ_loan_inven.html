CREATE PROCEDURE AAREP_ACTIV_LOAN_INVEN(
    REP_REGION_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    K_REG_NAME_PRIOR STR_100,
    K_REG_NAME_CUR STR_100,
    K_KONTRACT_NO VARCHAR(50),
    K_KF_NO STR_100,
    K_DISB_DATE DATE,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_KC_PRIOR STR_100,
    K_KC_CUR STR_100,
    KAL_PRIOR_LO_NAME STR_100,
    K_AMOUNT CURR_TYPE,
    DELINQ_DAY INTEGER,
    OKS_BALANCE CURR_TYPE,
    CASHIER_FIO D_STR_100,
    K_FACT_DISB_DATE D_DATE,
    LAST_REPAY_DATE D_DATE,
    DEPORT_DATE_BEGIN D_DATE,
    DEPORT_DATE_END D_DATE,
    RKL_BEGIN D_DATE,
    RKL_FINISH D_DATE,
    PATENT_BEGIN D_DATE,
    PATENT_FINISH D_DATE)
AS
declare variable k_from_kf_id id_type;
declare variable k_reg_id_prior integer;
declare variable k_reg_id_cur integer;
declare variable k_kc_id_prior id_type;
declare variable k_kc_id_cur id_type;
declare variable kal_date date;
declare variable kal_prior_lo_id id_type;
declare variable k_id id_type;
declare variable knc_id id_type;
begin
  /*  List of active loans for inventory
  Список активных кредитов для инвентаризации */
  for select ks_region_id, kal.kal_new_reg_id, ks.ks_kc_id, kal.kal_new_kf_officer, k_kontract_no, kal.kal_new_kf_name,
             ks_date_begin, k.k_kolich_client_approved, kal.kal_date, kal.kal_prior_lo_id, k_id, k_amount_approved,
             c_name_fio, k_from_kf_id
      from kontstate ks, kontract k, knaktlist kal, region_list(:rep_region_id, 1, -1, 1), clients
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            kal.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = id_list and
            k_clients_cashier_id = c_id
            --and k.k_kontract_no = '0313-1048027-02'
      into k_reg_id_prior, k_reg_id_cur, k_kc_id_prior, k_kc_id_cur, k_kontract_no, k_kf_no, k_disb_date,
           k_kolich_client_approved, kal_date, kal_prior_lo_id, k_id, k_amount, cashier_fio, k_from_kf_id
  do
  begin
    k_reg_name_prior = null;
    select reg_name
    from region
    where reg_unicode = :k_reg_id_prior
    into k_reg_name_prior;

    k_reg_name_cur = null;
    select reg_name
    from region
    where reg_unicode = :k_reg_id_cur
    into k_reg_name_cur;

    k_kc_prior = null;
    select c_name_fio
    from clients
    where c_id = :k_kc_id_prior
    into k_kc_prior;

    k_kc_cur = null;
    select c_name_fio
    from clients
    where c_id = :k_kc_id_cur
    into k_kc_cur;

    kal_prior_lo_name = '';

    if (kal_date >= k_disb_date) then
    begin
      if (kal_prior_lo_id <> k_kc_id_cur) then
      begin
        if (kal_prior_lo_id = k_kc_id_prior) then
          kal_prior_lo_name = k_kc_prior;
        else
        begin
          select c_name_fio
          from clients
          where c_id = :kal_prior_lo_id
          into kal_prior_lo_name;
        end
      end
    end

    delinq_day = null;
    oks_balance = null;
    select :rep_date_end - oks_delinq_date, oks_balance
    from oksummary
    where oks_from_k_id = :k_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_close_state = 0
    into delinq_day, oks_balance;

    if (delinq_day is null or delinq_day < 0) then
      delinq_day = 0;
    if (oks_balance is null) then
      oks_balance = 0;

    k_fact_disb_date = null;
    if (kal_prior_lo_id <> k_kc_id_cur) then
    begin
      select first 1 kal_date
      from knaktlist
      where kal_new_kf_officer = :k_kc_id_cur and
            is_delete = 0 and
            kal_from_kf_id = :k_from_kf_id and
            kal_prior_lo_id <> kal_new_kf_officer and
            kal_date <= :rep_date_end
      order by kal_date asc
      into :k_fact_disb_date;
    end

    last_repay_date = null;

    select ok_ol_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_from_c_id = 0
    order by ok_ol_date desc
    rows 1
    into :last_repay_date;

    -- RFCRU-5974 Добавление информации о клиентах, выехавших на Родину, в список активных кредитов для инвентаризации (депортация / РКЛ / патент)
    knc_id = null;
    select kc.kc_from_c_id
    from knclient kc
    where kc.kc_from_k_id = :k_id and
          kc.kc_amount_approved > 0
    order by kc.kc_number
    rows 1
    into :knc_id;
 
    deport_date_begin = null;
    deport_date_end = null;
    rkl_begin = null;
    rkl_finish = null;
    patent_begin = null;
    patent_finish = null;
 
    if (knc_id is not null) then
    begin
      select cd.cds_doc_date_begin,
             cd.cds_doc_date_end
      from clientdocs cd
      inner join passkind pk on pk.pk_id = cd.cds_from_pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cd.cds_from_c_id = :knc_id
        and pdt.pdt_code = 'deport_document'
        and cd.cds_is_delete = 0
        and current_date between cd.cds_doc_date_begin and cd.cds_doc_date_end
      order by cd.cds_doc_date_begin desc
      rows 1
      into :deport_date_begin, :deport_date_end;
 
      select cd.cds_doc_date_begin,
             cd.cds_doc_date_end
      from clientdocs cd
      inner join passkind pk on pk.pk_id = cd.cds_from_pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cd.cds_from_c_id = :knc_id
        and pdt.pdt_code = 'depart_info_rkl'
        and cd.cds_is_delete = 0
        and current_date between cd.cds_doc_date_begin and cd.cds_doc_date_end
      order by cd.cds_doc_date_begin desc
      rows 1
      into :rkl_begin, :rkl_finish;
 
      select cd.cds_doc_date_begin,
             cd.cds_doc_date_end
      from clientdocs cd
      inner join passkind pk on pk.pk_id = cd.cds_from_pk_id
      inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
      where cd.cds_from_c_id = :knc_id
        and pdt.pdt_code = 'depart_info_patent'
        and cd.cds_is_delete = 0
        and current_date between cd.cds_doc_date_begin and cd.cds_doc_date_end
      order by cd.cds_doc_date_begin desc
      rows 1
      into :patent_begin, :patent_finish;
    end
 
    suspend;
  end
end
