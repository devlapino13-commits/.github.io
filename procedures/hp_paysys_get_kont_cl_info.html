CREATE PROCEDURE HP_PAYSYS_GET_KONT_CL_INFO(
    K_ID DECIMAL(18,0))
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_AMOUNT DECIMAL(18,4),
    PSM_ID DECIMAL(18,0),
    PSL_ID DECIMAL(18,0),
    PSI_ID DECIMAL(18,0),
    PSI_CODE VARCHAR(20),
    PSIP_AMOUNT DECIMAL(18,4),
    PSIP_TAX_AMOUNT DECIMAL(18,4),
    PSIP_PERC DECIMAL(18,4),
    C_BIRTH_DATE DATE)
AS
begin
  for select c_id, c_name_fio, kc_amount_approved, c_birth_date
      from knclient, clients
      where kc_from_c_id = c_id and
            kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into :c_id, :c_name_fio, :c_amount, :c_birth_date
  do
  begin
-- Для каждого клиента определяем, есть ли полис
    psm_id = null;
    psl_id = null;
    psi_id = null;
    psi_code = null;
    psip_amount = null;
    psip_tax_amount = null;
    psip_perc = null;
    select psi_id, psi_code, psip_amount, psip_tax_amount, psip_perc
    from paysys_main, paysys_list, paysys_item, paysys_item_props
    where psl_from_psm_id = psm_id and
          psl_from_psi_id = psi_id and
          psl_object_kind = 3 -- Контракт
          and
          psi_from_psp_id = 2 -- Проект страхование
          and
          psm_is_delete = 0 and
          psip_from_psi_id = psi_id and
          psip_from_c_id = :c_id and
          psl_object_id = :k_id
    group by psi_id, psi_code, psip_amount, psip_tax_amount, psip_perc
    having sum(psl_sign) = 1
    into :psi_id, :psi_code, :psip_amount, :psip_tax_amount, psip_perc;

    select psl_id, psl_from_psm_id
    from paysys_list
    where psl_from_psi_id = :psi_id and
          psl_sign = 1
    order by psl_from_psm_id desc
    rows 1
    into :psl_id, :psm_id;

    suspend;
  end
end
