CREATE PROCEDURE HP_ANNUL_UNSIGNED_ASP_OEP
AS
declare variable edc_id d_integer;
declare variable rep_date d_date;
declare variable edl_object_id d_id;
begin
  edc_id = f_edoc_kind('asp.agreement_oep', 'id');
  rep_date = current_date - 1;

  for select edl.edl_object_id
      from edoc_list edl
      where edl.edl_from_edc_id = :edc_id and
            edl.edl_blank_date <= :rep_date and
            edl.edl_status = 0
      into edl_object_id
  do
    execute procedure hp_edoc_cancel('asp.agreement_oep', 'clients', :edl_object_id);
end
