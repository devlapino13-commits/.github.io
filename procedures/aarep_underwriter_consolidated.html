CREATE PROCEDURE AAREP_UNDERWRITER_CONSOLIDATED(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_ACL_ID D_ID,
    RES_ACL_DATE DATE,
    RES_MIN_TIME TIMESTAMP,
    RES_MAX_TIME TIMESTAMP,
    RES_LIFETIME_IN_MINUTE DECIMAL(15,2),
    SERTIFIKATOR VARCHAR(100),
    KF_NAME VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    RES_REG_NAME STR_100,
    RES_REG_NAMEP STR_100,
    OFFICER_NAME VARCHAR(100),
    K_CIRCLE INTEGER,
    CL_BALL D_INTEGER,
    PRODUCT_NAME VARCHAR(50),
    K_CLIENT_COUNT INTEGER,
    K_AMOUNT CURR_TYPE,
    RES_STATUS VARCHAR(30),
    RES_MEMO VARCHAR(2048),
    STOP_FACT VARCHAR(2048),
    RES_APPROVED INTEGER,
    RES_REJECTED INTEGER,
    RES_REFUSED INTEGER,
    RES_REFUSED_NEW INTEGER,
    CL_FIO_ALL VARCHAR(1024),
    K_REFUND_TXT VARCHAR(20),
    RES_FIRST_SOLUTION_DATETIME TIMESTAMP,
    RES_FIRST_SOLUTION_IN_MINUTE D_STR_20,
    RES_FIRST_SOLUTION_IN_MIN_TIME TIME,
    RES_ALL_DATETIME STR_20,
    K_FACT_DISB_DATE DATE,
    K_DISB_DATE DATE,
    RES_SENDER STR_100,
    UN_NAME STR_100,
    RES_FMEMO1 D_STR_8100,
    RES_FMEMO2 D_STR_8100,
    K_IS_ONLINE D_STR_CODE,
    K_IS_ONLINE1 D_STR_CODE,
    RES_LIFETIME D_STR_20)
AS
declare variable officer_id_prior curr_type;
declare variable kfolder_id decimal(18,0);
declare variable product_id_prior integer;
declare variable product_id integer;
declare variable rd_id_check_info integer;
declare variable rd_id_check_info_err integer;
declare variable officer_id id_type;
declare variable k_id id_type;
declare variable cl_id id_type;
declare variable cl_fio varchar(100);
declare variable ma_id d_integer;
declare variable k_disb_kind d_integer;
declare variable ma_date_app d_date;
declare variable pk_id d_integer;
begin
  rd_id_check_info = f_get_rd_id('ActionGroup.Check_Info');
  rd_id_check_info_err = f_get_rd_id('ActionGroup.Check_Info_Error');

  select pk_id
  from passkind
  inner join passdoctype on pdt_id = pk_from_pdt_id and
        pdt_code = 'anketa_questionnaire'
  into :pk_id;

  officer_id_prior = 0;
  product_id_prior = 0;
  for select res_acl_id, res_acl_date, res_min_datetime, res_max_datetime, c_name_fio as sertifikator, k_kontract_no,
             res_reg_name, res_reg_namep, kfolder_id, -- для определение СВЗ папки
             product_id, -- для определение название продукта
             k_client_count, k_amount, res_status, res_memo, res_lifetime_in_minute, res_approved, res_rejected,
             res_refused, res_refused_new, k_id, k_refund_txt, stop_fact, res_first_solution_datetime,
             res_first_solution_in_minute, res_all_datetime, k_fact_disb_date, k_disb_date, res_sender,
             res_lifetime
      from aarep_underwriter_report(:rep_region_id, :rep_date_begin, :rep_date_end)
      into :res_acl_id, :res_acl_date, :res_min_time, :res_max_time, :sertifikator, :k_kontract_no, :res_reg_name,
           :res_reg_namep, :kfolder_id, :product_id, :k_client_count, :k_amount, :res_status, :res_memo,
           :res_lifetime_in_minute, :res_approved, :res_rejected, :res_refused, :res_refused_new, :k_id, :k_refund_txt,
           :stop_fact, :res_first_solution_datetime, :res_first_solution_in_minute, :res_all_datetime,
           :k_fact_disb_date, :k_disb_date, :res_sender, res_lifetime
  do
  begin
    res_first_solution_in_min_time = cast(right(res_first_solution_in_minute, 8) as time);

    select kal_new_kf_officer, kal_new_kf_name
    from knaktlist
    where kal_from_kf_id = :kfolder_id and
          knaktlist.is_delete = 0 and
          :res_acl_date between kal_date and kal_date_end
    into officer_id, kf_name;

    if (officer_id_prior <> officer_id) then
    begin
      officer_id_prior = officer_id;

      select c_name_fio
      from clients
      where c_id = :officer_id
      into officer_name;
    end

    if (product_id_prior <> product_id) then
    begin
      product_id_prior = product_id;

      select pr_name
      from product
      where pr_id = :product_id
      into product_name;
    end

    cl_fio_all = null;
    for select kc_from_c_id, kc_circle
        from knclient kc
        where kc.kc_amount_approved > 0 and
              kc.kc_from_k_id = :k_id
        into cl_id, k_circle
    do
    begin

      cl_ball = f_point(:cl_id, :rep_date_end);

      select c_name_fio
      from clients
      where c_id = :cl_id
      into cl_fio;

      cl_fio_all = coalesce(cl_fio_all || ', ', '') || cl_fio;
    end

    un_name = (select first 1 un_name
               from knclient
               inner join actionlist on acl_object_kind = 0 and
                     acl_object_id = kc_from_c_id and
                     acl_from_rd_id in (:rd_id_check_info, :rd_id_check_info_err) and
                     acl_respons_c_id > 0 and
                     acl_date >= :k_disb_date and
                     acl_status <> -3
               inner join username on un_from_c_id = acl_respons_c_id
               where kc_from_k_id = :k_id);

    res_fmemo1 = (select first 1 f_get_verify_memo(obm_memo, 'reason_memo')
                  from objectprops
                  inner join objectmemo on obm_from_opr_id = opr_id
                  where opr_object_kind = 9 and
                        opr_object_id = :k_id and
                        obm_memo starting 'Для сведения:'
                  order by obm_datetime);

    res_fmemo2 = (select first 1  f_get_verify_memo(obm_memo, 'reason')
                  from objectprops
                  inner join objectmemo on obm_from_opr_id = opr_id
                  where opr_object_kind = 9 and
                        opr_object_id = :k_id and
                        (obm_memo like '%Доработка: %' or obm_memo like '%Отказ: %') and
                        obm_memo containing ']'
                  order by obm_datetime);

    select k_from_ma_app_id, k_disb_kind, iif(k_is_online = 1, 'Online', 'Offline')
    from kontract
    where k_id = :k_id
    into :ma_id, :k_disb_kind, :k_is_online;

    k_is_online1 = k_is_online;
    if (k_disb_kind = 5) then
    begin
      ma_date_app = (select first 1 ma_date_app
                     from miniapp
                     where ma_id = :ma_id
                     order by ma_number);
      if (exists(select *
                 from knclient
                 inner join clientdocs on cds_from_c_id = kc_from_c_id and
                       cds_from_pk_id = :pk_id and
                       cds_doc_date_begin between :ma_date_app and :k_disb_date and
                       (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
                 where kc_from_k_id = :k_id and
                       kc_amount_approved > 0)) then
        k_is_online1 = k_is_online1 || ' c анкетой';
    end
    else
      k_is_online1 = 'Наличными';

    suspend;
  end
end
