CREATE PROCEDURE UXP$UXFRPARAM_SOURCES(
    UFP_UUID D_UUID)
RETURNS (
    SOURCE_NAME D_STR_CODE)
AS
declare variable tmp_source d_str_code;
declare variable tmp_is_view d_bool;
declare variable add_table d_str_code;
declare variable us_main_table d_str_code;
begin
  select us_code, us_is_view, ufr_add_table_fields, us_main_table
  from ux$frparam, ux$field, ux$frame, ux$source
  where ufp_uuid = :ufp_uuid and
        ufp_ufr_uuid = ufr_uuid and
        ufp_uf_uuid = uf_uuid and
        uf_owner = us_uuid
  into tmp_source, tmp_is_view, add_table, us_main_table;

  if (row_count = 0) then
    exit;

  if (add_table is null) then
    add_table = '';

  if (us_main_table is null) then
    us_main_table = '';

  if (tmp_is_view is null) then
    tmp_is_view = 0;

  source_name = tmp_source;
  suspend;

  if (add_table <> '' and add_table <> tmp_source) then
  begin
    source_name = add_table;
    suspend;
  end

  if (us_main_table <> '' and us_main_table <> add_table and us_main_table <> tmp_source) then
  begin
    source_name = us_main_table;
    suspend;
  end

  tmp_source = upper(tmp_source);
  if (tmp_is_view = 1) then
  begin
    for select lower(trim(rdb$relation_name))
    from rdb$view_relations r
    where rdb$view_name = :tmp_source
    into :source_name
    do
      suspend;
  end else
  begin
    for select lower(trim(rdb$view_name))
    from rdb$view_relations r
    where rdb$relation_name = :tmp_source
    into :source_name
    do
      suspend;
  end

  tmp_is_view = 0;
  if (add_table <> '' and add_table <> tmp_source) then
  begin
    select us_is_view
    from ux$source
    where us_code = :add_table
    into tmp_is_view;

    add_table = upper(add_table);
    if (tmp_is_view = 1) then
    begin
      for select lower(trim(rdb$relation_name))
      from rdb$view_relations r
      where rdb$view_name = :add_table
      into :source_name
      do
        suspend;
    end else
    begin
      for select lower(trim(rdb$view_name))
      from rdb$view_relations r
      where rdb$relation_name = :add_table
      into :source_name
      do
        suspend;
    end
  end
end
