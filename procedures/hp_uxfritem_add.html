CREATE PROCEDURE HP_UXFRITEM_ADD(
    UX_CODE D_STR_CODE,
    ITEM_KIND D_STR_CODE,
    ITEM_NAME D_STR_CODE)
RETURNS (
    UFI_CODE D_STR_CODE)
AS
declare variable ufp_name d_str_code;
begin
  -- Добавляет один элемент
  if (item_kind = 'editor') then
  begin
    select ufp_name
    from uxfrparam
    where ufp_code = :item_name
    into ufp_name;

    if (ufp_name is not null) then
    begin
      if (not exists(select * from uxfritem
      where ufi_ux_code = :ux_code and
            ufi_ufp_code = :item_name)) then
      begin
        insert into uxfritem (ufi_ux_code, ufi_ufp_code, ufi_kind)
        values (:ux_code, :item_name, 0)
        returning ufi_code
        into ufi_code;
      end
    end
  end else
  if (item_kind = 'frame') then
  begin
    if (not exists(select * from uxcodeframe
    where ucf_ux_code = :ux_code and
          ucf_ufr_code = :item_name)) then
    begin
      insert into uxcodeframe (ucf_ux_code, ucf_ufr_code)
      values (:ux_code, :item_name);
    end

    if (not exists(select * from uxfritem
    where ufi_ux_code = :ux_code and
          ufi_ufr_code = :item_name)) then
    begin
      insert into uxfritem (ufi_ux_code, ufi_ufr_code, ufi_kind)
      values (:ux_code, :item_name, 1)
      returning ufi_code
      into ufi_code;
    end
  end else
  if (item_kind = 'form') then
  begin
    if (not exists(select * from uxfritem
    where ufi_ux_code = :ux_code)) then
    begin
      insert into uxfritem (ufi_ux_code, ufi_control, ufi_kind)
      values (:ux_code, :item_kind, -2)
      returning ufi_code
      into ufi_code;
    end
  end else
    exception e_default f_format('Unknown item_kind: %1', item_kind);

  if (ufi_code is not null) then
    suspend;
end
