CREATE PROCEDURE HP_EDOC_SMS_TEXT_REPLACE(
    DB_ID D_INTEGER,
    C_ID D_ID,
    EDC_ID D_ID,
    EDC_CODE D_STR_CODE,
    EDL_OBJECT_KIND D_INTEGER,
    EDL_OBJECT_ID D_ID,
    SMS_BODY D_STR_LONG,
    PUSH_TITLE D_STR_100,
    PUSH_BODY D_STR_2048)
RETURNS (
    RES_SMS_BODY D_STR_LONG,
    RES_PUSH_TITLE D_STR_100,
    RES_PUSH_BODY D_STR_2048)
AS
declare variable disb_kind d_integer;
begin
  -- Процедура замены текста СМС/ПУШ в зависимости от различных параметров
  res_sms_body = sms_body;
  res_push_title = push_title;
  res_push_body = push_body;

  if (db_id = 81001) then
  begin
    if (edc_code = 'loan.application' and edl_object_kind = 11 and coalesce(sms_body, '') <> '') then
    begin
      -- RFCRU-3797
      select ma_disb_kind
      from miniapp
      where ma_id = :edl_object_id
      order by ma_number desc
      rows 1
      into disb_kind;

      if (disb_kind in (7, 8)) then
        res_sms_body = 'Podpishite zayavku na zaem v mobilnom prilojenii kod [pin]';

      suspend;
    end 
  end
end
