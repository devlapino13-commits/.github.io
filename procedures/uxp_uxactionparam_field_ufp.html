CREATE PROCEDURE UXP$UXACTIONPARAM_FIELD_UFP(
    UA_UUID D_UUID,
    UA_UUID_CODE D_STR_CODE = null)
RETURNS (
    ID D_STR_CODE,
    TEXT D_STR_255)
AS
declare variable ua_owner d_uuid;
declare variable uf_name d_str_100;
declare variable ux_kind d_str_code;
declare variable uf_is_param d_bool;
begin
  if (coalesce(ua_uuid, '') = '') then
    ua_uuid = uxf$uniq(ua_uuid_code);

  id = null;
  text = '';
  suspend;

  if (coalesce(ua_uuid, '') = '') then
    exit;

  select ua_owner
  from uxp$uxalias_info(:ua_uuid)
  into :ua_owner;

  ux_kind = uxf$code(ua_owner, 'kind');

  if (ux_kind <> 'uxframe') then
  begin
    select ufr_uuid
    from ux$frame
    where ufr_owner = :ua_owner and
          ufr_is_mainframe = 1
    into ua_owner;
  end

  for select uuid_to_char(ufp_uuid), uf_name, uf_field_name, uf_is_param
  from ux$frparam, ux$field
  where ufp_ufr_uuid = :ua_owner and
        ufp_uf_uuid = uf_uuid and
        ufp_parent is null
--        uf_is_param = 0
  order by uf_field_name
  into id, uf_name, text, uf_is_param
  do
  begin
    if (uf_is_param = 1) then
      text = text || '  /P';
    else
      text = text || '  /F';

    if (text not containing uf_name) then
      text = text || '   - ' || uf_name;

    suspend;
  end
end
