CREATE PROCEDURE HP_COM_LIST_MEMO(
    CML_ID D_ID,
    SHOW_KIND D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    RES_ID D_INTEGER,
    RES_USER D_STR_100,
    RES_DATETIME D_DATETIME,
    RES_MEMO D_STR_LONG,
    RES_IS_ACTION D_SMALLINT)
AS
declare variable cmm_memo_kind d_integer;
declare variable cma_status d_integer;
declare variable cma_from_un_id d_id;
declare variable cma_id d_id;
declare variable cmm_from_un_id d_id;
declare variable user_kind d_str_code;
declare variable cma_status_str d_str_code;
declare variable cma_datetime d_datetime;
begin
  -- Процедура отображает все действия и комментарии по зпросу
  -- SHOW_KIND:
  --    0 смотрит заявитель      не отображаем ФИО КК и исполнителей
  --    1 смотрит член комитета  отображаем всё
  --    2 смотрит исполнитель    отображаем всё

  res_id = 0;
  for select cma_id, cma_status, cma_from_un_id, cma_datetime
      from com_action
      where cma_from_cml_id = :cml_id
      order by cma_id desc
      into :cma_id, :cma_status, :cma_from_un_id, :cma_datetime
  do
  begin
    res_is_action = 0;
    for select cmm_memo_kind, cmm_memo, cmm_datetime, cmm_from_un_id
        from com_memo
        where cmm_from_cma_id = :cma_id and
              cmm_is_delete = 0
        order by cmm_id desc
        into :cmm_memo_kind, :res_memo, :res_datetime, cmm_from_un_id
    do
    begin
      res_id = res_id + 1;
      user_kind = trim(case cmm_memo_kind
                         when 0 then 'Заявитель'
                         when 1 then 'Коммитет'
                         when 2 then 'Исполнитель'
                       end);

      res_user = '';
      if (cmm_from_un_id = un_id) then
        res_user = 'Вы';
      else
      begin
        select iif(:cmm_from_un_id = 1000000000000, un_fio, c_name_fio)
        from username, clients
        where un_id = :cmm_from_un_id and
              un_from_c_id = c_id
        into res_user;
      end

      if (show_kind = 0) then
      begin
        if (cmm_memo_kind <> 0) then
          res_user = user_kind;
      end
      else
      begin
        if (cmm_memo_kind = 1) then
          res_user = 'КК - ' || res_user;

        if (cmm_memo_kind = 2) then
          res_user = 'Исп. ' || res_user;
      end

      suspend;
    end

    res_is_action = 1;
    res_id = res_id + 1;
    res_datetime = cma_datetime;
    cma_status_str = f_enum('com_action_status', cma_status, 'code');
    res_memo = f_enum('com_action_status', cma_status, 'name');
    res_is_action = 1;

    cmm_memo_kind = 1;
    if (cma_status_str in ('new', 'sent')) then
      cmm_memo_kind = 0;
    else
    if (cma_status_str = 'done') then
      cmm_memo_kind = 2;

    user_kind = trim(case cmm_memo_kind
                       when 0 then 'Заявитель'
                       when 1 then 'Коммитет'
                       when 2 then 'Исполнитель'
                     end);

    res_user = '';
    if (cma_from_un_id = un_id) then
      res_user = 'Вы';
    else
    begin
      select iif(:cma_from_un_id = 1000000000000, un_fio, c_name_fio)
      from username, clients
      where un_id = :cma_from_un_id and
            un_from_c_id = c_id
      into res_user;
    end

    if (show_kind = 0) then
    begin
      if (cmm_memo_kind <> 0) then
        res_user = user_kind;
    end
    else
    begin
      if (cmm_memo_kind = 1) then
        res_user = 'КК - ' || res_user;

      if (cmm_memo_kind = 2) then
        res_user = 'Исп. ' || res_user;
    end

    suspend;
  end
end
