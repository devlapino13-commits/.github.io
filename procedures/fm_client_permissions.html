CREATE PROCEDURE FM_CLIENT_PERMISSIONS(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_DATA D_STR_8100)
AS
declare variable ACCESS_LEVEL D_SMALLINT;
declare variable LANG D_STR_LANG;
declare variable CLIENT_NAME D_STR_100;
declare variable PHOTO_CAN_LOAD D_SMALLINT;
declare variable PHOTO_ENABLED D_SMALLINT;
declare variable DOCUMENTS_ENABLED D_SMALLINT;
declare variable DOCUMENTS_CAN_EDIT D_SMALLINT;
declare variable CONNECTIONS_ENABLED D_SMALLINT;
declare variable CONTACTS_ENABLED D_SMALLINT;
declare variable ADDRESSES_ENABLED D_SMALLINT;
declare variable MENU_ENABLED D_SMALLINT;
declare variable DB_ID D_ID;
declare variable PHOTO_HINT D_STR_255;
declare variable CITIZEN_CODE D_STR_CODE;
declare variable PHONECALL_ENABLED D_SMALLINT;
begin
  -- Процедура, возвращающая права доступа пользователя к элементам окна клиента
  lang = f_get_lang_code(un_id, null);

  db_id = f_get_db_id();

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  select c_name_fio
  from clients
  where c_id = :c_id
  into client_name;
  if (client_name is null) then
    client_name = f_lang(lang, 'Клиент не найден');

  select access_lev
  from fm_client_status(:un_id, :c_id)
  into access_level;

  -- Разрешение на закрепление фото клиента из галереи
  photo_can_load = 0;
  -- Кнопка Фото
  photo_enabled = 0;
  -- Кнопка Документы
  documents_enabled = 0;
  -- Кнопка Связь
  connections_enabled = 0;
  -- Кнопка Контакты
  contacts_enabled = 0;
  -- Кнопка Адреса
  addresses_enabled = 0;
  -- Кнопка Меню (правый верний угол)
  menu_enabled = 0;
  -- Разрешение на добавление документов в окне документов
  documents_can_edit = 0;
  -- Сообщение об ошибке, когда фотографирование запрещено, если не пустое - блок
  photo_hint = '';
  -- Звонки контактным лицам клиента через ip-телефонию (Asterisk, пока только РФ и только НЕ просрочникам)
  phonecall_enabled = 0;

  --  if (access_level = -1) then
  --  begin
  --    -- Усиленные ограничения. Меню не активно, панель не активна, информация минимальная
  --    -- Для сотрудников, их родственников и не своих активных клиентов
  --  end

  if (access_level = 0) then
  begin
    -- Ограниченный. Меню активно, панель не активна, полная информация
    -- Доступ для менеджеров или аудиторов. Если есть доступ к данному офису
    menu_enabled = 1;
    documents_enabled = 1;
  end

  if (access_level = 2) then
  begin
    -- Ограниченный. Меню не активно, информация сокращенная, на панели активна только "Связь"
    -- Обычно это старые клиенты не обязательно свои
    connections_enabled = 1;
  end

  if (access_level = 1) then
  begin
    -- Полный доступ. Меню доступно, панель активна, полная информация
    -- Свой активный клиент или сотрудник (если есть права)

    photo_enabled = 1;
    documents_enabled = 1;
    connections_enabled = 1;
    contacts_enabled = 1;
    addresses_enabled = 1;
    menu_enabled = 1;
    documents_can_edit = 1;

    --Для повторных клиентов разрешаем загрузку фото из гелереи  RFCRU-221
    if (db_id = 81001) then
    begin
      if (exists(select *
                 from kontstate ks
                 inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
                 where kc.kc_from_c_id = :c_id and
                       kc.kc_amount_approved > 0 and
                       ks.ks_state = 1 and
                       ks.ks_date_begin <= current_date and
                       ks.ks_ol_is_delete = 0)) then
        photo_can_load = 1;
    end

    if (exists(select *
           from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessOriv')
           where has_access = 1)) then
    begin
      photo_can_load = 0;
      photo_enabled = 0;
      connections_enabled = 0;
      documents_can_edit = 0;
      addresses_enabled = 0;
    end
  end

  if (db_id in (39801, 41701, 94401)) then
  begin
  select rd_code
  from clients, refdata
  where c_id = :c_id
    and c_citizen = rd_id
  into citizen_code;

  end
    if (((db_id = 41701 and citizen_code = 'Citizen.KG') or
        (db_id = 39801 and citizen_code = 'Citizen.KZ') or
        (db_id = 94401 and citizen_code = 'Citizen.AZ')) and
        not exists(select *
                   from grs_clients gc
                   inner join enum_list on el_id = gc_status and el_kind = 'grs_status'
                   where gc_from_c_id = :c_id and
                         gc_is_last = 1 and
                         el_code in ('successful', 'successful_adr_empty', 'manual_check')))
    then
      photo_hint = f_lang(lang, f_format('Сначала надо отправить запрос в %1, а после сфотографировать клиента', iif(db_id = 94401, 'Asan', 'ГРС')));

  if (contacts_enabled = 1 and f_get_const('fsm_phonecall_enabled') = 1) then
  begin
    select 1 - is_delinq
    from hp_is_client_delinqed(:c_id)
    into phonecall_enabled;
  end

  res_data = '{' ||
    f_json_make_pair('client_name', client_name, 'str') || ',' ||
    f_json_make_pair('photo_can_load', photo_can_load, 'str') || ',' ||
    f_json_make_pair('photo_enabled', photo_enabled, 'str') || ',' ||
    f_json_make_pair('phonecall_enabled', phonecall_enabled, 'str') || ',' ||
    f_json_make_pair('photo_hint', photo_hint, 'str') || ',' ||
    f_json_make_pair('documents_enabled', documents_enabled, 'str') || ',' ||
    f_json_make_pair('documents_can_edit', documents_can_edit, 'str') || ',' ||
    f_json_make_pair('connections_enabled', connections_enabled, 'str') || ',' ||
    f_json_make_pair('contacts_enabled', contacts_enabled, 'str') || ',' ||
    f_json_make_pair('addresses_enabled', addresses_enabled, 'str') || ',' ||
    f_json_make_pair('menu_enabled', menu_enabled, 'str') ||
    '}';

  suspend;
end
