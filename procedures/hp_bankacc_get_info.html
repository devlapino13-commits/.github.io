CREATE PROCEDURE HP_BANKACC_GET_INFO(
    K_ID D_ID,
    C_ID D_ID)
RETURNS (
    MNT_ID D_STR_10,
    CLIENT_INN D_STR_20,
    CLIENT_NAME D_STR_100,
    CLIENT_ACCOUNT D_STR_CODE,
    BANK_BIK D_STR_10,
    BANK_NAME D_STR_255,
    BANK_KS D_STR_CODE,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE)
AS
begin
  -- Возвращает банковские реквизиты для указанного контракта и клиента
  -- Для выдачи займа на расчетный счет через Монету

  if (c_id is not null) then
  begin
    select b_id, b_name, b_corr_acc, bc_account_no, c_inn, trim(c_name_fio)
    from clients, bankclient, banks
    where c_id = :c_id
      and b_id = bc_from_b_id
      and bc_from_c_id = c_id
      and bc_is_main = 1
      and bc_is_active = 1
      and is_delete = 0
      and b_corr_acc is not null
    into bank_bik, bank_name, bank_ks, client_account, client_inn, client_name;
  end

  if (not client_name starting 'ИП ') then -- Требования бухгалтерии и Монеты
    client_name = 'ИП ' || client_name;

  if (k_id is not null) then
  begin
    select k_kontract_no, k_disb_date,
      iif(pr_code containing '.oivo_pack.', f_get_const('moneta_oivo_acc'), f_get_const('moneta_acc'))
    from kontract
    inner join product on pr_id = k_from_pr_id
    where k_id = :k_id
    into k_kontract_no, k_disb_date, mnt_id;
  end
  suspend;
end
