CREATE PROCEDURE HP_DOCRECOG_IMPORT(
    DO_IMPORT D_BOOL,
    IMP_STR D_STR_8100)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable p d_integer;
declare variable drm_id d_integer;
declare variable drm_api_code d_str_code;
declare variable drf_api_code d_str_code;
declare variable renamed_fields d_str_8100;

declare function new_name (
    drf_old_api_code d_str_code)
returns d_str_code
as
begin
  -- если в docrecogfield есть старое название, и импортируется новое название, считаем это переименованием
  if (drf_old_api_code = 'surname') then
    return 'last_name';
  if (drf_old_api_code = 'birth_date_mrz') then
    return 'mrz_birth_date';
  if (drf_old_api_code = 'expiry_date_mrz') then
    return 'mrz_expiry_date';
  if (drf_old_api_code = 'first_name_mrz') then
    return 'mrz_name';
  if (drf_old_api_code = 'gender_mrz') then
    return 'mrz_gender';
  if (drf_old_api_code = 'last_name_mrz') then
    return 'mrz_last_name';
  if (drf_old_api_code = 'number_mrz') then
    return 'mrz_number';
  if (drf_old_api_code = 'opt_data_1_mrz') then
    return 'mrz_opt_data_1';
  if (drf_old_api_code = 'opt_data_2_mrz') then
    return 'mrz_opt_data_2';
  if (drf_old_api_code = 'surname_eng') then
    return 'last_name_eng';
  if (drf_old_api_code = 'mrz_line_1') then
    return 'mrz_line1';
  if (drf_old_api_code = 'mrz_line_2') then
    return 'mrz_line2';
  if (drf_old_api_code = 'nationality_mrz') then
    return 'mrz_nationality';
  if (drf_old_api_code = 'family_status') then
    return 'marital_status';
  if (drf_old_api_code = 'patornymic') then
    return 'patronymic';
  if (drf_old_api_code = 'id_passport') then
    return 'number';
  if (drf_old_api_code = 'plate') then
    return 'vehicle_plate';
  if (drf_old_api_code = 'model') then
    return 'vehicle_model';
  if (drf_old_api_code = 'type') then
    return 'vehicle_type';
  if (drf_old_api_code = 'year') then
    return 'release_year';
  if (drf_old_api_code = 'colour') then
    return 'body_color';
  if (drf_old_api_code = 'category') then
    return 'vehicle_category';
  return '';
end

declare procedure rename_field (
    drm_id d_integer,
    drm_api_code d_str_code,
    drf_api_code_old d_str_code,
    drf_api_code_new d_str_code)
as
declare variable drf_id d_integer;
begin
  drf_id = null;
  select drf_id
  from docrecogfield
  where drf_from_drm_id = :drm_id and
        drf_api_code = :drf_api_code_old
  into drf_id;
  if (drf_id is null) then
    exception e_default f_format('Field not found for document: %1, api_field: %2', drm_api_code, drf_api_code_old);
  update docrecogfield
  set drf_api_code = :drf_api_code_new
  where drf_id = :drf_id;
end

declare procedure drop_field (
    drm_id d_integer,
    drm_api_code d_str_code,
    drf_api_code d_str_code)
as
declare variable drf_id d_integer;
begin
  drf_id = null;
  select drf_id
  from docrecogfield
  where drf_from_drm_id = :drm_id and
        drf_api_code = :drf_api_code
  into drf_id;
  if (drf_id is null) then
    exception e_default f_format('Field not found for document: %1, api_field: %2', drm_api_code, drf_api_code);
  delete from docrecogvalue
  where drv_from_drf_id = :drf_id;
  delete from docrecogfield
  where drf_id = :drf_id;
end

declare procedure add_field (
    drm_id d_integer,
    drm_api_code d_str_code,
    drf_api_code d_str_code)
as
declare variable drf_id d_integer;
begin
  drf_id = null;
  select drf_id
  from docrecogfield
  where drf_from_drm_id = :drm_id and
        drf_api_code = :drf_api_code
  into drf_id;

  if (drf_id is not null) then
    exception e_default f_format('Field already exists for document: %1, api_field: %2', drm_api_code, drf_api_code);

  insert into docrecogfield (drf_from_drm_id, drf_api_code, drf_name,
                             drf_inactive, drf_not_null, drf_is_extra,
                             drf_data_type)
  values (:drm_id, :drf_api_code, :drf_api_code, 1, 0, 0, 0);
end

begin
  -- Импорт/синхронизация полей библиотеки распознавания (SmartEngine)
  -- На вход передается строчка вида:
  -- rus.inn.entity.type2 [inn, kpp, number, ogrn, series]

  -- Переименованые поля
  -- Инициализируем исключаемыми из импорта (поля изображений)
  renamed_fields = '.photo.signature.stamp.rf_label.authority_signature.stamp_authority_code.ir_rf_label.signature_authority.';

  imp_str = trim(imp_str);
  p = position(': [', imp_str);
  if (p = 0) then
    exception e_default 'Invalid input string.';
  drm_api_code = left(imp_str, p - 1); -- api-имя документа

  select drm_id
  from docrecogmain
  where drm_api_code = :drm_api_code rows 1
  into drm_id;

  if (drm_id is null) then
    exception e_default 'drm_api_code not found: ' || coalesce(drm_api_code, 'null');

  imp_str = substring(imp_str from p + 3 for char_length(imp_str) - p - 3); -- api-список полей

  for select drm_id
      from docrecogmain
      where drm_api_code = :drm_api_code
      into drm_id
  do
  begin

    -- Кандидаты на переименование
    for select drf_api_code
        from docrecogfield, hp_split_str(:imp_str, ',')
        where drf_from_drm_id = :drm_id and
              drf_api_code <> '' and
              new_name(drf_api_code) = trim(res_str)
        into drf_api_code
    do
    begin
      -- в drf_api_code приходят отсутствующие в но
      if (do_import = 1) then
        execute procedure rename_field(drm_id, drm_api_code, drf_api_code,
            new_name(drf_api_code));
      else
      begin
        res_text = f_format('Renamed: %1.%2 -> %3', drm_api_code, drf_api_code, new_name(drf_api_code));
        suspend;
      end
      -- Исключаем найденные из добавления и удаления
      renamed_fields = renamed_fields || f_format('.%1.%2.', drf_api_code, new_name(drf_api_code));
    end

    -- Кандидаты на добавление
    for select trim(res_str)
        from hp_split_str(:imp_str, ',')
        left join docrecogfield on drf_from_drm_id = :drm_id and
              drf_api_code = trim(res_str)
        where drf_id is null
        into drf_api_code
    do
    begin
      if (position('.' || drf_api_code || '.', renamed_fields) = 0 and
          drf_api_code not starting 'rfid_') then
      begin
        if (do_import = 1) then
          execute procedure add_field(drm_id, drm_api_code, drf_api_code);
        else
        begin
          res_text = f_format('Added: %1.%2', drm_api_code, drf_api_code);
          suspend;
        end
      end
    end

    -- Кандидаты на удаление
    for select drf_api_code
        from docrecogfield
        left join hp_split_str(:imp_str, ',') on drf_api_code = trim(res_str)
        where drf_from_drm_id = :drm_id and
              res_str is null and
              drf_api_code <> ''
        into drf_api_code
    do
    begin
      if (position('.' || drf_api_code || '.', renamed_fields) = 0) then
      begin
        if (do_import = 1) then
          execute procedure drop_field(drm_id, drm_api_code, drf_api_code);
        else
        begin
          res_text = f_format('Deleted: %1.%2', drm_api_code, drf_api_code);
          suspend;
        end
      end
    end
  end
end
