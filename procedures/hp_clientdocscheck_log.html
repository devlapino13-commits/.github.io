CREATE PROCEDURE HP_CLIENTDOCSCHECK_LOG(
    DOC_ID D_INTEGER,
    DOC_ISCORRECT D_INTEGER,
    DOC_NO VARCHAR(30),
    DOC_DATE_BEGIN D_DATE,
    DOC_DATE_END D_DATE,
    DOC_ISSUED_BY D_STR_255,
    DOC_CREATE D_DATETIME,
    DOC_PK_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable c_id id_type;
declare variable dcl_id d_integer;
declare variable dcl_pk_name_iscorrect d_integer;
declare variable doc_no1 varchar(30);
declare variable doc_date_begin1 d_date;
declare variable doc_date_end1 d_date;
declare variable doc_issued_by1 d_str_255;
declare variable doc_create1 d_datetime;
declare variable doc_isupdate d_integer;
declare variable cds_from_pk_id d_integer;
begin
  if (doc_date_begin = '30.12.1899') then doc_date_begin = null;
  if (doc_date_end = '30.12.1899') then doc_date_end = null;

  select dcl_id, dcl_pk_name_iscorrect
    from clientdocscheck_log
    where dcl_cds_id = :doc_id
    into :dcl_id, :dcl_pk_name_iscorrect;

  if ((dcl_id > 0) and (doc_iscorrect <> dcl_pk_name_iscorrect)) then
    update
        clientdocscheck_log cds
      set
        cds.dcl_pk_name_iscorrect = :doc_iscorrect,
        cds.dcl_date_check        = current_timestamp
    where
      (cds.dcl_id = :dcl_id);

  select cds_from_c_id, trim(cds_doc_no), cds_doc_date_begin, cds_doc_date_end, trim(cds_doc_issued_by), cds_create_datetime, cds_from_pk_id
    from clientdocs
    where cds_id = :doc_id
    into :c_id, :doc_no1, :doc_date_begin1, :doc_date_end1, :doc_issued_by1, :doc_create1, :cds_from_pk_id;

  if (row_count = 1) then
  begin
    doc_isupdate = 0;
    doc_issued_by = trim(doc_issued_by);
    doc_no = trim(doc_no);
    if  ((:doc_no         is distinct from :doc_no1)
      or (:doc_date_begin is distinct from :doc_date_begin1)
      or (:doc_date_end   is distinct from :doc_date_end1)
      or (:doc_create     is distinct from :doc_create1)
      or (:doc_issued_by  is distinct from :doc_issued_by1)
      or ((:doc_pk_id is not null) and (:doc_pk_id is distinct from :cds_from_pk_id))) then
    begin
      res_text = '';

      if ((:doc_pk_id is not null) and (:doc_pk_id is distinct from :cds_from_pk_id)) then
        res_text = (select res_text from hp_docrecog_change_type(:doc_id, :doc_pk_id, :un_id));

      if (res_text = '') then
      begin
        cds_from_pk_id = coalesce(:doc_pk_id, :cds_from_pk_id);
        update
            clientdocs cd
          set
            cd.cds_doc_no          = :doc_no,
            cd.cds_doc_date_begin  = :doc_date_begin,
            cd.cds_doc_date_end    = :doc_date_end,
            cd.cds_create_datetime = :doc_create,
            cd.cds_doc_issued_by   = :doc_issued_by
          where
            (cd.cds_id = :doc_id);
        doc_isupdate = 1;
      end
      if (dcl_id is null and (doc_isupdate = 1 or doc_iscorrect = 0)) then
      begin
        insert into clientdocscheck_log (dcl_cds_id, dcl_c_id, dcl_pk_name_iscorrect, dcl_doc_no, dcl_doc_date_begin,
                                         dcl_doc_date_end, dcl_doc_issued_by, dcl_doc_create, dcl_from_pk_id)
        values (:doc_id, :c_id, :doc_iscorrect, :doc_no1, :doc_date_begin1, :doc_date_end1, :doc_issued_by1, :doc_create1, :cds_from_pk_id);
      end
    end
  end
  else
    res_text = 'Ошибка обновления данных';
  suspend;
end
