CREATE PROCEDURE HP_PVNOPERS_CHECK_BALANCE(
    PO_ID D_INTEGER,
    OPER_TYPE D_SMALLINT,
    C_ID D_ID,
    CARD_ID D_INTEGER,
    AMOUNT D_CURRENCY,
    OFFICE_ID D_INTEGER,
    ACC_KT D_STR_20,
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    NEW_PO_ID D_INTEGER)
AS
declare variable STATUS D_INTEGER;
declare variable CS_REQUEST_JSON D_STR_8100;
declare variable CS_ID D_INTEGER;
declare variable ACC_DT D_STR_20;
declare variable IS_CASHING D_BOOL;
declare variable IS_MB_CARD D_BOOL;
declare variable CS_FROM_CSC_ID D_INTEGER;
declare variable CS_OBJECT_KIND D_INTEGER;
declare variable CS_UPD_DATETIME timestamp;
declare variable PO_CARD_OFFICE_ID D_INTEGER;
begin
  amount = coalesce(amount, 0);
  is_mb_card = 1;
  cs_from_csc_id = f_get_cardsynccfg('check_balance', 'id');
  cs_object_kind = f_objkind('pvnopers', 'id');

  -- > --------- валидация данных ----------------------------------------------

  -- Проверка на утвержденный кредит
  if (not exists(
                    select 1
                    from cardlistlnk cll
                      inner join kontract k on cll.cll_from_k_id = k.k_id
                    where cll.cll_card_id = :card_id
                      and current_date between cll.cll_date_begin and cll.cll_date_end
                      and k.k_status = 1
                 )) then
    exception e_message 'Кредит по карте не утвержден';

  -- Ограничение проверки баланса за последние 30 сек
  select first 1 cs.cs_upd_datetime
  from cardsync cs
  where
    cs.cs_from_csc_id = :cs_from_csc_id
    and cs.cs_object_kind = :cs_object_kind
    and cs.cs_card_id = :card_id
  order by cs.cs_upd_datetime desc
  into :cs_upd_datetime;

  if (datediff(second, coalesce(:cs_upd_datetime,cast('01.01.1900' as timestamp)), current_timestamp) <= 30) then
    exception e_message 'С момента последней проверки баланса не прошло 30 секунд, повторите попытку позже';

  if (oper_type is null) then
    exception e_message 'Не передан тип операции';

  if (c_id is null) then
    exception e_message 'Не передан id клиента';

  if (card_id is null) then
    exception e_message 'Не передан id карты';

  if (amount <= 0) then
    exception e_message 'Сумма операции должна быть больше 0';

  if (office_id <= 0) then
    exception e_message 'Не передан офис операции';

  if (office_id <= 0) then
    exception e_message 'не передан пользователь - автор записи';

  if (oper_type = 0) then
  begin
      if (coalesce(acc_kt, '') = '') then
        exception e_message 'Счет Кт не определен';
    
      acc_dt = (select al_id from acc_list where al_from_reg_id = :office_id and al_currency = 417 and al_account = '10024/2'); -- Операции по ПВН
    
      if (coalesce(acc_dt, '') = '') then
        exception e_message 'Счет Дт не определен';
    
      select c_id,
             is_cashing
      from find_client_by_mb_card(:card_id)
      into :c_id,
           :is_cashing;
    
      if (c_id is null) then
        exception e_message 'Клиент (владелец) карты не определен';
    
      if (is_cashing <> 1) then
        exception e_message 'С данной карты запрещено обналичивание';

      if (k_id is not null) then
        exception e_message 'Передан id кредита, хотя не должен был предаваться';
  end
  else if (oper_type = 1) then
  begin
      if (k_id is null) then
        exception e_message 'id кредита не предано';

      if (coalesce(acc_kt, '') <> '') then
        exception e_message 'Счет Кт не должен передваться';
  end
  -- < -------------------------------------------------------------------------

-- Опеределяем филиал карты

  select cpl.cpl_from_reg_id
  from cardlist cal
  inner join cardplace cpl on cal.cal_last_cpl_id = cpl.cpl_id
  where cal.cal_card_id = :card_id
  into :po_card_office_id;


  if (po_id is null) then
  begin
      insert into pvnopers (po_date, po_oper_type, po_card_id, po_from_c_id, po_amount, po_un_id, po_office_id, po_status, po_is_mb_card, po_acc_kt, po_from_k_id, po_card_office_id)
      values (current_date, :oper_type, :card_id, :c_id, :amount, :un_id, :office_id, 0, :is_mb_card, :acc_kt, :k_id, :po_card_office_id)
      returning po_id
      into :new_po_id;
  end
  else
  begin
    new_po_id = po_id;

    select po.po_status
    from pvnopers po
    where po.po_id = :po_id
    into :status;

    if (status > 2) then -- если уже отправлен запрос на списание с карты
        exception e_message 'Вы не можете осуществить проверку баланса карты, после того как осуществлен запрос на списание с карты';

    update pvnopers po
    set po.po_date = current_date,
        po.po_oper_type = :oper_type,
        po.po_from_c_id = :c_id,
        po.po_card_id = :card_id,
        po.po_amount = :amount,
        po.po_un_id = :un_id,
        po.po_office_id = :office_id,
        po.po_status = 0,
        po_is_mb_card = :is_mb_card,
        po.po_acc_kt = :acc_kt,
        po.po_from_k_id = :k_id,
        po.po_card_office_id = :po_card_office_id
    where po.po_id = :po_id;
  end

  cs_request_json = f_format('{"card_id": %1}', :card_id);

  insert into cardsync(
      cs_from_csc_id,
      cs_card_id,
      cs_object_kind,
      cs_object_id,
      cs_request_json,
      cs_office_id,
      cs_from_un_id
      )
  values(
      :cs_from_csc_id,
      :card_id,
      :cs_object_kind,
      :new_po_id,
      :cs_request_json,
      :office_id,
      :un_id
  )
  returning cs_id
  into :cs_id;

  update pvnopers po
  set po.po_balance_cs_id = :cs_id
  where po.po_id = :new_po_id;

  suspend;
end
