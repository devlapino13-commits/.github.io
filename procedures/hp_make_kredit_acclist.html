CREATE PROCEDURE HP_MAKE_KREDIT_ACCLIST(
    REP_PR_ID INTEGER,
    REP_OFFICE_ID INTEGER,
    REP_ACC_TYPE INTEGER)
RETURNS (
    RES_ACCOUNT VARCHAR(100),
    RES_ACC_DELINQ VARCHAR(100),
    RES_ACC_OUTBALANCE VARCHAR(100),
    RES_ACC_WRITEOFF VARCHAR(100))
AS
declare variable reg_code varchar(20);
declare variable db_id integer;
declare variable def_curr varchar(20);
declare variable rd_code varchar(255);
begin
  res_account = '';
  res_acc_delinq = '';
  res_acc_outbalance = '';
  res_acc_writeoff = '';
  db_id = f_get_db_id();

  if (rep_acc_type >= 0) then
  begin
    if (db_id = 39801) then
    begin
      select reg_code || pal_currency || pal_main_acc || right('000000' || pal_main_acc_no, 11 - char_length(pal_main_acc)),
             reg_code || pal_currency || pal_delinq_acc || right('000000' || pal_delinq_acc_no, 11 - char_length(pal_delinq_acc)),
             reg_code || pal_currency || pal_outbalance_acc || right('000000' || pal_outbalance_acc_no, 11 - char_length(pal_outbalance_acc)),
             reg_code || pal_currency || pal_writeoff_acc || right('000000' || pal_writeoff_acc_no, 11 - char_length(pal_writeoff_acc))
      from prodacclist, region, product
      where pr_id = :rep_pr_id and
            reg_unicode = :rep_office_id and
            pr_from_pat_id = pal_from_pat_id and
            pal_from_at_id = :rep_acc_type
      into :res_account, res_acc_delinq, res_acc_outbalance, res_acc_writeoff;
    end else
    begin
      select reg_code || pal_currency || pal_main_acc || right('000000' || pal_main_acc_no, 6),
             reg_code || pal_currency || pal_delinq_acc || right('000000' || pal_delinq_acc_no, 6),
             reg_code || pal_currency || pal_outbalance_acc || right('000000' || pal_outbalance_acc_no, 6),
             reg_code || pal_currency || pal_writeoff_acc || right('000000' || pal_writeoff_acc_no, 6)
      from prodacclist, region, product
      where pr_id = :rep_pr_id and
            reg_unicode = :rep_office_id and
            pr_from_pat_id = pal_from_pat_id and
            pal_from_at_id = :rep_acc_type
      into :res_account, res_acc_delinq, res_acc_outbalance, res_acc_writeoff;
    end

    if (res_account is null) then
      res_account = '';
    if (res_acc_delinq is null) then
      res_acc_delinq = '';
    if (res_acc_delinq = '') then
      res_acc_delinq = res_account;
    if (res_acc_outbalance is null) then
      res_acc_outbalance = '';
    if (res_acc_outbalance = '') then
      res_acc_outbalance = res_account;
    if (res_acc_writeoff is null) then
      res_acc_writeoff = '';
    if (res_acc_writeoff = '') then
      res_acc_writeoff = res_account;
  end
  else
  begin
-- Счета доп.обязательств
    rd_code = '';
    select rd_code
    from refdata
    where rd_id = -:rep_acc_type
    into :rd_code;
    if (rd_code <> '') then
    begin
      select reg_code
      from region
      where reg_unicode = :rep_office_id
      into :reg_code;

      /*      select cm_database_id, cm_default_currency from company
      into db_id, def_curr; */
      select cm_default_currency
      from company
      into def_curr;

      if (rd_code = 'OperExtraKind.LawClientExpense') then -- Счета для приема судебных издержек
      begin
        if (db_id = 81001) then
        begin
          res_account = '76-02000005';
          res_acc_delinq = '91-01000006';
        end
        if (db_id in (39801, 41701)) then
        begin
          res_account = '11531001001';
          res_acc_delinq = '21199001001';
        end
        if (db_id = 94401) then
        begin
          res_account = '';
          res_acc_delinq = '';
        end
      end

      --S.K. Задолженность после исключения из реестра субсидирования
      if (rd_code = 'OperExtraKind.ExceptionRegister') then
      begin
        res_account = (
          select
            iif(pr_code like '%.IPF.%', '11531000204', '11531000203')
          from product
          where pr_id = :rep_pr_id);
        res_acc_delinq = '';
      end

      if (res_account <> '') then
        res_account = reg_code || def_curr || res_account;
      if (res_acc_delinq <> '') then
        res_acc_delinq = reg_code || def_curr || res_acc_delinq;
    end
  end
  suspend;
end
