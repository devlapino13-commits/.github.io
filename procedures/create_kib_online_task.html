CREATE PROCEDURE CREATE_KIB_ONLINE_TASK(
    C_ID D_ID)
RETURNS (
    KBM_ID D_ID,
    NAME D_STR_100,
    FNAME D_STR_100,
    LNAME D_STR_100,
    PATRONYMIC D_STR_100,
    SEX_CODE D_STR_10,
    BIRTHD D_DATE,
    BIRTHD_STR D_STR_10,
    COUNTRY_CODE D_STR_10,
    PASSN D_STR_100,
    SOCC D_STR_100,
    OKPO D_STR_100,
    C_DOC_TYPE D_INTEGER,
    C_TYPE D_SMALLINT,
    KBM_PERSON_ID D_INTEGER)
AS
declare variable c_name_fio varchar(255);
declare variable c_sex integer;
declare variable kk integer;
begin
  country_code = 'kgz';
  insert into kibminiapp (kbm_id, kbm_from_c_id, kbm_ma_from_reg_id, kbm_ma_id, kbm_pr_id, kbm_is_checked,
                          kbm_pr_id_datetime, kbm_checked_datetime, kbm_ver_info, kbm_err_text, kbm_from_un_id)
  values (-1, :c_id, 0, 0, '', 0, null, null, 0, '', 1000000000000)
  returning kbm_id
  into :kbm_id;

  insert into kibminiapp_online (kbm_id)
  values (:kbm_id);

  select c.c_name_fio, c.c_sex, c.c_birth_date, c_inn, c_okpo, c_doc_type, c_pass_no, c.c_type, coalesce(c_kib_id, 0)
  from clients c
  where c.c_id = :c_id
  into :c_name_fio, :c_sex, :birthd, :socc, :okpo, :c_doc_type, :passn, :c_type, :kbm_person_id;

  birthd_str = f_datetostr(birthd);

  c_name_fio = replace(c_name_fio, 'ң', 'н');
  c_name_fio = replace(c_name_fio, '�©', 'о');
  c_name_fio = replace(c_name_fio, 'ү', 'у');

  name = c_name_fio;
  if (socc is null) then
    socc = '';
  fname = '';
  lname = '';
  patronymic = '';
  c_name_fio = (:c_name_fio);
  kk = position(' ', :c_name_fio) - 1;
  if (kk > 0) then
  begin
    lname = substring(:c_name_fio from 1 for kk);
    c_name_fio = substring(:c_name_fio from kk + 2 for 255);
    kk = position(' ', c_name_fio) - 1;
    if (kk > 0) then
    begin
      fname = substring(:c_name_fio from 1 for kk);
      c_name_fio = substring(:c_name_fio from kk + 2 for 255);
      patronymic = :c_name_fio;
    end
    else
      fname = :c_name_fio;
  end
  if (c_sex is null) then
    sex_code = 'Н';
  if (c_sex = 0) then
    sex_code = 'Ж';
  if (c_sex = 1) then
    sex_code = 'М';
  fname = trim(fname);
  lname = trim(lname);
  patronymic = trim(patronymic);
  suspend;
end
