CREATE PROCEDURE AAREP_RATING_PROROST_CL(
    REP_DATE_END DATE)
RETURNS (
    OFFICER_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    PRIROST_CLI INTEGER,
    PRIROST_PORTFEL CURR_TYPE,
    KOL_CLI_END INTEGER,
    PAR7 CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable oks_region_id integer;
declare variable oks_kc_id decimal(18,0);
declare variable oks_balance decimal(18,4);
declare variable oks_delinq_balane7 decimal(18,4);
declare variable kol_cli_begin integer;
declare variable kc_balance_begin curr_type;
declare variable kc_balance_end curr_type;
begin
  rep_date_begin = rep_date_end - 7;
  delete from temp_kc_reg_info;
-- на начало
  insert into temp_kc_reg_info (kc_region_id, kc_officer_id, kc_kol_cli, kc_balance, kc_kol_cli_sks)
  select ac_region_id, ac_kc_id, sum(ac_is_client), sum(ac_cli_balance), 0 -- KC_KOL_CLI_SKS - 0 - на дату начало; 1 - на дату конец
  from aarep_cli_active_list(:rep_date_begin)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_begin, null, null, null, null, null, null)) in (1, 2) and
        not exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_begin between sba_date_begin and sba_date_end and
                         sba_from_k_id = ac_kontract_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1 -- находящийся по внешним коллектором не будет считатся
        )
  group by ac_region_id, ac_kc_id;
-- на конец
  insert into temp_kc_reg_info (kc_region_id, kc_officer_id, kc_kol_cli, kc_balance, kc_kol_cli_sks)

  select ac_region_id, ac_kc_id, sum(ac_is_client), sum(ac_cli_balance), 1 -- KC_KOL_CLI_SKS - 0 - на дату начало; 1 - на дату конец
  from aarep_cli_active_list(:rep_date_end)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2) and
        not exists(select *
                   from sb_akt, sb_list
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_end between sba_date_begin and sba_date_end and
                         sba_from_k_id = ac_kontract_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1 -- находящийся по внешним коллектором не будет считатся
        ) and
        exists(select *
               from knaktlist
               where knaktlist.is_delete = 0 and
                     :rep_date_end between kal_date and kal_date_end and
                     kal_from_kf_id = ac_kf_id and
                     kal_new_kf_officer = ac_kc_id and
                     (kal_date <= :rep_date_begin -- считать только не переданным за текущую неделю
                     or
                     kal_date <= ac_disb_date))
  group by ac_region_id, ac_kc_id;

--  select * from TEMP_KC_REG_INFO

  for select oks_kc_id, oks_region_id, sum(oks_balance),
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0))
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_kc_id > 1000000000003 and
            oks_close_state = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
      group by oks_kc_id, oks_region_id
      into oks_kc_id, oks_region_id, oks_balance, oks_delinq_balane7
  do
  begin
    par7 = oks_delinq_balane7 / oks_balance * 100;

    officer_name = (select c_name_fio
                    from clients
                    where c_id = :oks_kc_id);
    reg_name = (select reg_code || ' ' || reg_name
                from region
                where reg_unicode = :oks_region_id);
--  клиенты на начало
    kol_cli_begin = null;
    kc_balance_begin = null;
    select kc_kol_cli, kc_balance
    from temp_kc_reg_info
    where kc_region_id = :oks_region_id and
          kc_officer_id = :oks_kc_id and
          kc_kol_cli_sks = 0 -- выборка начало
    into kol_cli_begin, kc_balance_begin;
    if (kol_cli_begin is null) then
      kol_cli_begin = 0;
    if (kc_balance_begin is null) then
      kc_balance_begin = 0;
--  клиенты на конец
    kol_cli_end = null;
    kc_balance_end = null;
    select kc_kol_cli, kc_balance
    from temp_kc_reg_info
    where kc_region_id = :oks_region_id and
          kc_officer_id = :oks_kc_id and
          kc_kol_cli_sks = 1 -- выборка конец
    into kol_cli_end, kc_balance_end;
    if (kol_cli_end is null) then
      kol_cli_end = 0;
    if (kc_balance_end is null) then
      kc_balance_end = 0;

    prirost_cli = kol_cli_end - kol_cli_begin;
    prirost_portfel = kc_balance_end - kc_balance_begin;

    suspend;
  end

end
