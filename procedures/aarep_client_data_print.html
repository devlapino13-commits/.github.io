CREATE PROCEDURE AAREP_CLIENT_DATA_PRINT(
    OBJECT_KIND INTEGER,
    OBJECT_ID DECIMAL(18,0),
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_ALTERNAT_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_SEX_STR VARCHAR(100),
    C_RESIDENT_STR VARCHAR(100),
    C_CITIZEN_STR VARCHAR(100),
    C_BIRTH_COUNTRY VARCHAR(255),
    CELL_MAIN VARCHAR(100),
    CELL_ADD VARCHAR(100),
    ADDRESS_MAIN VARCHAR(255),
    ADDRESS_HOME VARCHAR(255),
    ADDRESS_WORK VARCHAR(255),
    ADDRESS_RODINA_MAIN VARCHAR(255),
    ADDRESS_RODINE_HOME VARCHAR(255),
    CONTACT_NO_RU VARCHAR(255),
    CONTACT_NO_RODINA VARCHAR(255),
    WORK_PLACE_NAME VARCHAR(255),
    WORK_STATUS_KIND VARCHAR(255),
    LO_FIO VARCHAR(255),
    LOAN_PURPOSE VARCHAR(255),
    CONTACT_NAME_RU VARCHAR(1024),
    CONTACT_NAME_RODINA VARCHAR(1024),
    CONT_FACEBOOK VARCHAR(255),
    CONT_VKONTAKTE VARCHAR(255),
    CONT_ODNOKLASSNIKI VARCHAR(255),
    CONT_TELEGRAMM VARCHAR(255),
    CONT_IMO VARCHAR(255),
    CONT_WHATSAPP VARCHAR(100),
    CONT_VIBER VARCHAR(100),
    CONT_PHONE_WORK VARCHAR(100),
    CONT_SKYPE VARCHAR(100),
    CONT_EMAIL VARCHAR(100),
    WORK_MONTHLY_INCOME VARCHAR(255),
    WORK_EXTRA_AMOUNT VARCHAR(255),
    MONETARY_OBLIGATIONS VARCHAR(255),
    COMMITMENTS_SUM_MONTH VARCHAR(255),
    PER_PAYMENTS_OBLIG_MONTH VARCHAR(255),
    CAN_BE_SURETY VARCHAR(255),
    DEFENDANT_INCOURT VARCHAR(255),
    MA_DATE_APP DATE,
    PREVIOUS_FIO STR_255,
    OWNERSHIP STR_255,
    QR_CODE D_STR_LONG,
    K_KONTRACT_DATE DATE,
    COMPANY_NAME D_STR_100,
    C_INN D_STR_20,
    EDOC_DATE_CREATE D_DATE,
    CONT_MAX VARCHAR(100),
    PR_NAME D_STR_64,
    IP_REGISTR_DATE D_DATE,
    OGRNIP D_STR_CODE)
AS
declare variable c_birth_country_tmp varchar(100);
declare variable c_birth_city_tmp varchar(100);
declare variable res_has_code integer;
declare variable db_id integer;
declare variable edl_id d_id;
declare variable last_cc_date_rf date;
declare variable last_cc_date_rodina date;
declare variable object_kind_str d_str_code;
declare variable kc_anketa_income_calc d_currency;
declare variable kc_id d_id;
declare variable extra_amount d_currency;
declare variable is_documented d_bool;
declare variable k_id d_id;
begin
  lo_fio = '';
  db_id = f_get_db_id();
  company_name = f_company_name('short', 'local');
  object_kind_str = f_objkind(object_kind, 'code');

  if (object_kind_str = 'clients') then
  begin
    select min(ma_date_app), pr_name
    from kontract, knclient, miniapp, product
    where kc_from_k_id = k_id and
          k_from_ma_app_id = ma_id and
          pr_id = k_from_pr_id and
          kc_from_c_id = :object_id and
          kc_amount_approved > 0 and
          ma_date_app >= current_date - 30
    group by 2
    into ma_date_app, pr_name;

    if (ma_date_app is null) then
    begin
      select min(ma_date_app), pr_name
      from miniappclient, miniapp, product
      where ma_is_active = 1 and
            ma_id = mac_from_ma_id and
            pr_id = ma_from_pr_id and
            mac_from_c_id = :object_id and
            mac_amount > 0 and
            ma_date_app >= current_date - 30
      group by 2
      order by 2
      rows 1
      into ma_date_app, pr_name;
    end
  end
  else
  if (object_kind_str = 'kontract') then
  begin
    select min(ma_date_app), k_kontract_date, pr_name
    from miniapp, kontract k, product
    where k_from_ma_app_id = ma_id and
          k_from_pr_id = pr_id and
          k_id = :object_id
    group by 2,3
    into ma_date_app, k_kontract_date, pr_name;

    k_id = object_id;
  end
  else
  if (object_kind_str = 'miniapp') then
  begin
    select min(ma_date_app), pr_name
    from miniapp, product
    where ma_id = :object_id and    
          ma_from_pr_id = pr_id
    group by 2
    into ma_date_app, pr_name;
  end
  else
  if (object_kind_str = 'com_list') then
  begin
    -- !!! Внимание, замена object_kind, object_id, берем из КК
    select cml_date, cml_object_kind, cml_object_id
    from com_list
    where cml_id = :object_id
    into ma_date_app, object_kind, object_id;
    object_kind_str = f_objkind(object_kind, 'code');
    -- Дата в бланке должна быть взята из даты заседания КК
    k_kontract_date = ma_date_app;
  end

  for select c_id, c_name_fio, c_name_eng, c_birth_date, iif(c_sex = 0, 'Женский', 'Мужской'),
             iif(c_is_resident = 0, 'Нет, нерезидент', 'Да, резидент.'), cit.rd_name,
             coalesce(c_birth_country, ''), coalesce(c_birth_city, ''), c_inn
      from clients, refdata cit
      where c_id = :object_id and
            :object_kind = 0 and
            c_citizen = rd_id

      union
      select c_id, c_name_fio, c_name_eng, c_birth_date, iif(c_sex = 0, 'Женский', 'Мужской'),
             iif(c_is_resident = 0, 'Нет, нерезидент', 'Да, резидент.'), cit.rd_name,
             coalesce(c_birth_country, ''), coalesce(c_birth_city, ''), c_inn
      from clients, refdata cit, knclient
      where c_id = kc_from_c_id and
            :object_kind = 9 and
            kc_from_k_id = :object_id and
            kc_amount_approved > 0 and
            c_citizen = rd_id

      union
      select c_id, c_name_fio, c_name_eng, c_birth_date, iif(c_sex = 0, 'Женский', 'Мужской'),
             iif(c_is_resident = 0, 'Да, резидент.', 'Нет, нерезидент'), cit.rd_name,
             coalesce(c_birth_country, ''), coalesce(c_birth_city, ''), c_inn
      from clients, refdata cit, miniappclient
      where c_id = mac_from_c_id and
            mac_from_ma_id = :object_id and
            :object_kind = 11 and
            mac_amount > 0 and
            c_citizen = rd_id

      into c_id, c_name_fio, c_alternat_fio, c_birth_date, c_sex_str, c_resident_str, c_citizen_str,
           c_birth_country_tmp, c_birth_city_tmp, c_inn
  do
  begin
    c_birth_country = '';
    if (c_birth_country_tmp <> '' and
        c_birth_city_tmp <> '') then
    begin
      if (position(c_birth_city_tmp, c_birth_country_tmp) <= 0) then
        c_birth_country = c_birth_country_tmp || ', ' || c_birth_city_tmp;
      else
        c_birth_country = c_birth_country_tmp;
    end
    if (c_birth_country_tmp <> '' and
        c_birth_city_tmp = '') then
      c_birth_country = c_birth_country_tmp;
    if (c_birth_country_tmp = '' and
        c_birth_city_tmp <> '') then
      c_birth_country = c_birth_city_tmp;

    cell_main = '';
    cell_add = '';
    cont_email = '';
    cont_skype = '';
    cont_whatsapp = '';
    cont_viber = '';
    cont_phone_work = '';
    cont_max = '';

    cell_main = f_cont_info('clients', c_id, 'cell_main', 'ok');
    cell_add = f_cont_info('clients', c_id, 'cell_add', 'last');
    cont_email = f_cont_info('clients', c_id, 'email', 'last');

    cont_skype = f_cont_info('clients', c_id, 'skype', 'last');
    cont_whatsapp = f_cont_info('clients', c_id, 'whatsapp', 'last');
    cont_viber = f_cont_info('clients', c_id, 'viber', 'last');
    cont_facebook = f_cont_info('clients', c_id, 'facebook', 'last');
    cont_vkontakte = f_cont_info('clients', c_id, 'vk', 'last');
    cont_odnoklassniki = f_cont_info('clients', c_id, 'ok', 'last');
    cont_telegramm = f_cont_info('clients', c_id, 'telegram', 'last');
    cont_imo = f_cont_info('clients', c_id, 'imo', 'last');
    cont_phone_work = f_cont_info('clients', c_id, 'phone_work', 'last');
    cont_max = f_cont(c_id, 'max', 'last', 'any');
    if (cell_main = '') then
      cell_main = '---';
    if (cell_add = '') then
      cell_add = '---';
    if (cont_email = '') then
      cont_email = '---';
    if (cont_skype = '') then
      cont_skype = '---';
    if (cont_whatsapp = '') then
      cont_whatsapp = '---';
    if (cont_phone_work = '') then
      cont_phone_work = '---';
    if (cont_max = '') then
      cont_max = '---';

    address_main = '';
    address_home = '';
    address_work = '';
    address_rodina_main = '';
    address_rodine_home = '';

    select adl_address_str
    from addrlist, addrclient
    where adl_id = adc_from_adl_id and
          adc_from_c_id = :c_id and
          adc_address_type = 1
    into :address_main;
    if (address_main = '') then
      address_main = '---';

    select adl_address_str
    from addrlist, addrclient
    where adl_id = adc_from_adl_id and
          adc_from_c_id = :c_id and
          adc_address_type = 2
    into :address_home;
    if (address_home = '') then
      address_home = '---';

    select adl_address_str
    from addrlist, addrclient
    where adl_id = adc_from_adl_id and
          adc_from_c_id = :c_id and
          adc_address_type = 3
    into :address_work;
    if (address_work = '') then
      address_work = '---';

    select adl_address_str
    from addrlist, addrclient
    where adl_id = adc_from_adl_id and
          adc_from_c_id = :c_id and
          adc_address_type = 4
    into :address_rodina_main;
    if (address_rodina_main = '') then
      address_rodina_main = '---';

    select adl_address_str
    from addrlist, addrclient
    where adl_id = adc_from_adl_id and
          adc_from_c_id = :c_id and
          adc_address_type = 5
    into :address_rodine_home;
    if (address_rodine_home = '') then
      address_rodine_home = '---';

    contact_no_ru = '';
    contact_name_ru = '';

    select list(cc.cc_name, ', '), list(cc_cell_no, ', ')
    from contacts cc
    where cc_from_c_id = :c_id and
          cc_is_old_contact = 0 and
          coalesce(cc_cell_no, '') <> '' and
          cc_type = 0 -- Контакты в РФ
    into contact_name_ru, contact_no_ru;
    if (contact_name_ru is null) then
      contact_no_ru = '';
    if (contact_no_ru is null) then
      contact_no_ru = '';
    if (contact_no_ru = '') then
      contact_no_ru = '---';

    contact_name_rodina = '';
    contact_no_rodina = '';
    select list(cc_name, ', '), left(list(cc_cell_no, ', '), 255)
    from contacts
    where cc_from_c_id = :c_id and
          cc_is_old_contact = 0 and
          coalesce(cc_cell_no, '') <> '' and
          cc_type = 1 -- Контакты в РФ
    into contact_name_rodina, contact_no_rodina;
    if (contact_name_rodina is null) then
      contact_no_rodina = '';
    if (contact_no_rodina is null) then
      contact_no_rodina = '';
    if (contact_no_rodina = '') then
      contact_no_rodina = '---';

    if (db_id = 81001 and
        (contact_no_ru = '---' or contact_no_rodina = '---')) then
    begin
      if (exists(select ks.ks_state
                 from knclient knc
                 inner join kontract k on k.k_id = knc.kc_from_k_id
                 inner join kontstate ks on ks.ks_from_k_id = k.k_id
                 inner join product p on p.pr_id = k.k_from_pr_id
                 where knc.kc_from_c_id = :c_id and
                       knc.kc_amount_approved > 0 and
                       current_date between ks.ks_date_begin and ks.ks_date_end and
                       ks.ks_ol_is_delete = 0 and
                       p.pr_code containing '.par.' and
                       ks.ks_state in (0, 1))) then
      begin

        if (contact_no_rodina = '---') then
        begin
          last_cc_date_rf = null;
          contact_no_ru = '';
          contact_name_ru = '';

          select max(cc.cc_date)
          from contacts cc
          where cc_from_c_id = :c_id and
                -- cc_is_old_contact = 0 and
                coalesce(cc_cell_no, '') <> '' and
                cc_type = 0
          into :last_cc_date_rf;

          select list(cc.cc_name, ', '), list(cc_cell_no, ', ')
          from contacts cc
          where cc_from_c_id = :c_id and
                cc.cc_date = :last_cc_date_rf and
                coalesce(cc_cell_no, '') <> '' and
                cc_type = 0 -- Контакты в РФ
          into contact_name_ru, contact_no_ru;
          if (contact_name_ru is null) then
            contact_no_ru = '';
          if (contact_no_ru is null) then
            contact_no_ru = '';
          if (contact_no_ru = '') then
            contact_no_ru = '---';
        end

        if (contact_no_rodina = '---') then
        begin
          last_cc_date_rodina = null;
          contact_name_rodina = '';
          contact_no_rodina = '';

          select max(cc.cc_date)
          from contacts cc
          where cc_from_c_id = :c_id and
                coalesce(cc_cell_no, '') <> '' and
                cc_type = 1
          into :last_cc_date_rodina;

          select list(cc_name, ', '), left(list(cc_cell_no, ', '), 255)
          from contacts
          where cc_from_c_id = :c_id and
                cc_date = :last_cc_date_rodina and
                coalesce(cc_cell_no, '') <> '' and
                cc_type = 1 -- Контакты в Родина
          into contact_name_rodina, contact_no_rodina;
          if (contact_name_rodina is null) then
            contact_no_rodina = '';
          if (contact_no_rodina is null) then
            contact_no_rodina = '';
          if (contact_no_rodina = '') then
            contact_no_rodina = '---';
        end
      end
    end

    work_status_kind = '';
    res_has_code = 0;
    work_place_name = '';
    loan_purpose = '';
    ownership = '';

    begin
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'work_status', null)
      into :work_status_kind;
      when any do
      begin
      end
    end

    begin
      select res_has_code
      from hp_objectprops_anketa_get(0, :c_id, -1, 'work_status', null, 'work_status.dont_work')
      into :res_has_code;
      when any do
      begin
      end
    end

    begin
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'work_place_name', null)
      into :work_place_name;
      when any do
      begin
      end
    end

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'AnketaQuestions.LoanPurpose', current_date, '')
      into :loan_purpose;
      when any do
      begin
      end
    end

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'AnketaQuestions.Ownership', current_date, '')
      into :ownership;
      when any do
      begin
      end
    end

    ownership = coalesce(ownership, '');
    ownership = iif(ownership = '', 'Нет', ownership);

    -----------
    work_monthly_income = '';
    work_extra_amount = '';
    monetary_obligations = '';
    commitments_sum_month = '';
    per_payments_oblig_month = '';
    can_be_surety = '';
    defendant_incourt = '';

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'work_monthly_income', current_date, '')
      into :work_monthly_income;
      when any do
      begin
      end
    end

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'work_extra_amount', current_date, '')
      into :work_extra_amount;
      when any do
      begin
      end
    end

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'AnketaQuestions.MonetaryObligations', current_date, '')
      into :monetary_obligations;
      when any do
      begin
      end
    end

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'AnketaQuestions.CommitmentsSumMonth', current_date, '')
      into :commitments_sum_month;
      when any do
      begin
      end
    end

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'AnketaQuestions.PerPaymentsObligMonth', current_date, '')
      into :per_payments_oblig_month;
      when any do
      begin
      end
    end

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'AnketaQuestions.CanBeSurety', current_date, '')
      into :can_be_surety;
      when any do
      begin
      end
    end

    begin
      select res_str_values
      from hp_objectprops_anketa_get(0, :c_id, -1, 'AnketaQuestions.DefendantInCourt', current_date, '')
      into :defendant_incourt;
      when any do
      begin
      end
    end

    if (res_has_code = 1 or work_place_name = '') then
      work_place_name = 'Собственные накопления';

    -- «Прежние ФИО»,
    previous_fio = null;
    if (db_id = 81001) then
    begin
      select res_str_values
      from hp_objectprops_anketa_get('Clients', :c_id, -1, 'AnketaQuestions.previous_fio', current_date, null)
      into :previous_fio;
      previous_fio = coalesce(previous_fio, '');
    end

    -- QR-код электронной подписи клиента. Текст пустой если не подписан
    if (coalesce(add_params, '') <> '') then
    begin
      edl_id = f_json_str_get_value(add_params, 'edl_id', null);
      qr_code = f_edoc_qr_by_edl(edl_id, c_id);
    end

    if (f_get_db_id() = 81001 and k_id is not null) then
    begin
      -- RFCRU-1970 ПДН: меры в целях понижения ПДН клиента до <=80%
      kc_anketa_income_calc = null;
      extra_amount = 0;

      begin
        extra_amount = work_extra_amount;
        when any do
        begin
        end 
      end
       
      select kc_id
      from knclient
      where kc_from_k_id = :k_id
        and kc_from_c_id = :c_id
      into kc_id;

      select kc_anketa_income_calc
      from knclient_pdn
      where kc_id = :kc_id
        and kcp_kind = 0
      into kc_anketa_income_calc;

      if (kc_anketa_income_calc is not null) then
        work_monthly_income = cast(kc_anketa_income_calc - extra_amount as integer);

      -- RFCRU-2566 (Документально подтвержденный доход)
      if (f_get_const('documented_income_active') = 1) then
      begin
        select maci_is_documented
        from knclient, kontract, miniappclient, miniappclient_income
        where kc_id = :kc_id
          and k_id = kc_from_k_id
          and mac_from_ma_id = k_from_ma_app_id
          and mac_from_c_id = kc_from_c_id
          and maci_from_mac_id = mac_id
        into is_documented;

            and ki_from_c_id = kc_from_c_id
