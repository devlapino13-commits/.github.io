CREATE PROCEDURE HP_KKT_GET_OFD_LINK(
    KKM_ID D_ID)
RETURNS (
    LINK D_STR_255)
AS
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  if (db_id = 81001) then
    select f_format('https://consumer.1-ofd.ru/ticket?fn=%1&fp=%2&t=%3&s=%4&i=%5&n=2',
      kkf_code,
      kkm_fpd,
          lpad(extract(year   from kkm_fisc_datetime), 4, '0') ||
          lpad(extract(month  from kkm_fisc_datetime), 2, '0') ||
          lpad(extract(day    from kkm_fisc_datetime), 2, '0') || 'T' ||
          lpad(extract(hour   from kkm_fisc_datetime), 2, '0') ||
          lpad(extract(minute from kkm_fisc_datetime), 2, '0'),
      cast((select coalesce(sum(kki_amount), 0) from kkt_items where kki_from_kkm_id = kkm_id) as decimal(15,2)),
      kkm_fd)
    from kkt_main, kkt_fn kf
    where kkf_id = kkm_from_kkf_id
      and kkm_id = :kkm_id
      and kkm_status = 2
    into link;
  else if (db_id = 39801) then
  begin
    link = trim(iif(exists(select 1 from hp_db_is_actual), 'https://kkm.webkassa.kz', 'https://devkkm.webkassa.kz')); -- убрать после ввода в бой в KZ
    select :link || f_format('/spa-ui/ticket?chb=%1&sh=%2&extnum=%3', kkd_code, kkm_shift_num, uuid_to_char(kkm_uuid))
    from kkt_main, kkt_fn, kkt_devices
    where kkf_id = kkm_from_kkf_id
      and kkd_id = kkf_from_kkd_id
      and kkm_id = :kkm_id
      and kkm_status = 2
    into link;
  end

  suspend;
end
