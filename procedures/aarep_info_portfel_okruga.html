CREATE PROCEDURE AAREP_INFO_PORTFEL_OKRUGA(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    PORTFEL_END CURR_TYPE,
    DELINQ_BALANCE_END CURR_TYPE,
    DELINQ_BALANE7_END CURR_TYPE,
    DELINQ_BALANE30_END CURR_TYPE,
    PORTFEL_LAST CURR_TYPE,
    DELINQ_BALANCE_LAST CURR_TYPE,
    DELINQ_BALANE7_LAST CURR_TYPE,
    DELINQ_BALANE30_LAST CURR_TYPE,
    PORTFEL_BEGIN CURR_TYPE,
    DELINQ_BALANCE_BEGIN CURR_TYPE,
    DELINQ_BALANE7_BEGIN CURR_TYPE,
    DELINQ_BALANE30_BEGIN CURR_TYPE,
    KOL_CLI_END INTEGER,
    KOL_CLI_LAST INTEGER,
    KOL_CLI_BEGIN INTEGER,
    KOL_KRED_END INTEGER,
    KOL_KRED_LAST INTEGER,
    KOL_KRED_BEGIN INTEGER,
    KOL_PARAL_END INTEGER,
    KOL_PARAL_LAST INTEGER,
    KOL_PARAL_BEGIN INTEGER,
    REST_BALANCE_END CURR_TYPE,
    REST_BALANCE_LAST CURR_TYPE,
    REST_BALANCE_BEGIN CURR_TYPE,
    FLAT_SUM_END CURR_TYPE,
    FLAT_SUM_LAST CURR_TYPE,
    FLAT_SUM_BEGIN CURR_TYPE,
    FLAT_PERC_END CURR_TYPE,
    FLAT_PERC_LAST CURR_TYPE,
    FLAT_PERC_BEGIN CURR_TYPE,
    PORTFEL_PARAL_END CURR_TYPE,
    PORTFEL_PARAL_LAST CURR_TYPE,
    PORTFEL_PARAL_BEGIN CURR_TYPE,
    REG_NAME VARCHAR(50))
AS
declare variable reg_unicode integer;
begin
  for select reg_unicode, reg_name
      from region, region_list(:rep_region_id, 1, -1, 1)
      where reg_unicode in (1000000, 1090106, 1090206, 1090406, 1090506) and
            reg_unicode = id_list
      into reg_unicode, reg_name
  do
  begin
    for select portfel_end, delinq_balance_end, delinq_balane7_end, delinq_balane30_end, portfel_last,
               delinq_balance_last, delinq_balane7_last, delinq_balane30_last, portfel_begin, delinq_balance_begin,
               delinq_balane7_begin, delinq_balane30_begin, kol_cli_end, kol_cli_last, kol_cli_begin, kol_kred_end,
               kol_kred_last, kol_kred_begin, kol_paral_end, kol_paral_last, kol_paral_begin, rest_balance_end,
               rest_balance_last, rest_balance_begin, flat_sum_end, flat_sum_last, flat_sum_begin, flat_perc_end,
               flat_perc_last, flat_perc_begin, portfel_paral_end, portfel_paral_last, portfel_paral_begin
        from aarep_info_porfel_new(:reg_unicode, :rep_date_end)
        into :portfel_end, :delinq_balance_end, :delinq_balane7_end, :delinq_balane30_end, :portfel_last,
             :delinq_balance_last, :delinq_balane7_last, :delinq_balane30_last, :portfel_begin, :delinq_balance_begin,
             :delinq_balane7_begin, :delinq_balane30_begin, :kol_cli_end, :kol_cli_last, :kol_cli_begin, :kol_kred_end,
             :kol_kred_last, :kol_kred_begin, :kol_paral_end, :kol_paral_last, :kol_paral_begin, :rest_balance_end,
             :rest_balance_last, :rest_balance_begin, :flat_sum_end, :flat_sum_last, :flat_sum_begin, :flat_perc_end,
             :flat_perc_last, :flat_perc_begin, :portfel_paral_end, :portfel_paral_last, :portfel_paral_begin
    do
    begin
      suspend;
    end
  end
end
