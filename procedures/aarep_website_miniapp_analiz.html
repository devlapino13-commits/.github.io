CREATE PROCEDURE AAREP_WEBSITE_MINIAPP_ANALIZ(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_TYPE SMALLINT,
    STEP SMALLINT,
    SHOW_PHONE SMALLINT)
RETURNS (
    ESM_ID ID_TYPE,
    ESM_DATE DATE,
    ESM_DATETIME TIMESTAMP,
    CITY STR_100,
    CL_NAME STR_100,
    CL_BIRTHDAY DATE,
    CL_CITIZENSHIP STR_100,
    CL_GENDER STR_100,
    CL_LOAN_TERM STR_100,
    HAS_LOAN SMALLINT,
    LOAN_DATE DATE,
    IS_EMPLOYEE SMALLINT,
    EMPLOYEE_DATE DATE,
    PHONE_NO STR_100,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    KC_NAME_FIO STR_100,
    CL_LOAN_AMOUNT CURR_TYPE,
    PASS_NO_OPER VARCHAR(30),
    CELL_NO_OPER VARCHAR(20),
    CK_FIO STR_100,
    CK_BIRTH_DATE DATE,
    FIND_METHOD VARCHAR(50),
    C_FIND_NAME_FIO STR_100,
    K_KONTRACT_NO VARCHAR(20),
    KF_NAME STR_100,
    CL_NEXT_CALL_VALUE TIMESTAMP,
    CL_NEXT_CALL_TYPE STR_100,
    CL_KOLICH_REPAY INTEGER,
    CL_LANGUAGE VARCHAR(20),
    REG_NAME STR_100,
    MEETING_POINT STR_255,
    MEETING_ADDRESS STR_255,
    DATA_SOURCE STR_100)
AS
declare variable esm_id_old integer;
declare variable esf_field_code str_100;
declare variable esv_value_txt str_100;
declare variable c_id id_type;
declare variable esm_id_tmp id_type;
declare variable ck_secondary_data str_255;
declare variable c_pass_no varchar(30);
declare variable esm_id_called id_type;
declare variable ck_from_esm_id id_type;
declare variable fuzzy_search smallint;
declare variable pass_no_oper_tmp str_100;
declare variable cell_no_oper_tmp str_100;
declare variable ck_from_reg_id integer;
declare variable start_date date;
begin
  if (rep_type = 1) then --  если рассматривается число заявок, с сайта, за период. нужно опередилть получили ли по ним займы, или устроился ли "мигрант" в мбф.ру
  begin
    if (step = 1) then --  получаю список заявок с сайта, за период
    begin
      esm_id_old = 0;

      for select em.esm_id, ef.esf_field_code, left(ev.esv_value_txt, 100), em.esm_date, em.esm_datetime
          from extsysmain em, extsysfields ef, extsysvalues ev
          where em.esm_project = ef.esf_project and
                em.esm_project = 'website' and
                ev.esv_from_esm_id = em.esm_id and
                ev.esv_from_esf_id = ef.esf_id and
                em.esm_status = 0 -- если = 0 значит успешная команда
                and
                em.esm_date between :rep_date_begin and :rep_date_end and
                em.esm_action = 'save_cl_info'
          order by em.esm_id, ef.esf_field_code
          into :esm_id, :esf_field_code, :esv_value_txt, :esm_date, :esm_datetime
      do
      begin

        if (esm_id <> esm_id_old) then
        begin
          if (esm_id_old <> 0) then
          begin
            esm_id_tmp = esm_id;
            esm_id = esm_id_old;

            pass_no_oper = '';
            cell_no_oper = '';
            ck_fio = '';
            ck_secondary_data = '';
            ck_birth_date = null;
            ck_from_reg_id = null;
            reg_name = null;

            select chk.ck_secondary_data, chk.ck_fio, chk.ck_birth_date, chk.ck_from_reg_id
            from cc_check_knclient chk
            where chk.ck_from_esm_id = :esm_id and
                  chk.ck_type = 6
            order by chk.ck_checked_datetime desc
            rows 1
            into :ck_secondary_data, :ck_fio, :ck_birth_date, :ck_from_reg_id;

            if (ck_from_reg_id is not null) then
            begin
              select reg.reg_name
              from region reg
              where reg.reg_unicode = :ck_from_reg_id
              into :reg_name;
            end

            select pass_no, cell_no
            from cc_secondary_data_parse(:ck_secondary_data)
            into pass_no_oper, cell_no_oper;

            suspend;
            esm_id = esm_id_tmp;
          end
          cl_name = '';
          cl_birthday = null;
          cl_citizenship = null;
          cl_gender = '';
          cl_loan_term = '';
          cl_loan_amount = 0;
          phone_no = '';
          city = '';
          cl_next_call_value = null;
          cl_next_call_type = '';
          cl_kolich_repay = 0;
          cl_language = '';
          meeting_point = '';
          meeting_address = '';
          data_source = 'www.molbulak.ru';

          esm_id_old = esm_id;
        end

        if (esm_id_old = esm_id) then
        begin
          if (esf_field_code = 'cl_name') then
            cl_name = esv_value_txt;
          else
          if (esf_field_code = 'cl_birthday') then
            cl_birthday = cast(esv_value_txt as date);
          else
          if (esf_field_code = 'cl_citizenship') then
            cl_citizenship = esv_value_txt;
          else
          if (esf_field_code = 'cl_gender') then
            cl_gender = esv_value_txt;
          else
          if (esf_field_code = 'cl_loan_term') then
            cl_loan_term = esv_value_txt;
          else
          if (esf_field_code = 'cl_loan_amount') then
            cl_loan_amount = iif(esv_value_txt = '', 0, coalesce(esv_value_txt, 0));
          else
          if (esf_field_code = 'phone_no') then
            phone_no = esv_value_txt;
          else
          if (esf_field_code = 'cl_city') then
            city = esv_value_txt;
          else
          if (esf_field_code = 'cl_next_call_value') then
            cl_next_call_value = iif(esv_value_txt <> '', cast(esv_value_txt as timestamp), null);
          else
          if (esf_field_code = 'cl_next_call_type') then
            cl_next_call_type = esv_value_txt;
          else
          if (esf_field_code = 'cl_kolich_repay') then
            cl_kolich_repay = iif(esv_value_txt <> '', cast(esv_value_txt as integer), 0);
          else
          if (esf_field_code = 'cl_language') then
            cl_language = esv_value_txt;
          else
          if (esf_field_code = 'meeting_point') then
            meeting_point = esv_value_txt;
          else
          if (esf_field_code = 'meeting_address') then
            meeting_address = esv_value_txt;
          else
          if (esf_field_code = 'data_source') then
            data_source = esv_value_txt;
        end
      end

      if (esm_id_old <> 0) then
      begin
        pass_no_oper = '';
        cell_no_oper = '';
        ck_fio = '';
        ck_secondary_data = '';
        ck_birth_date = null;
        ck_from_reg_id = null;
        reg_name = null;

        select chk.ck_secondary_data, chk.ck_fio, chk.ck_birth_date, chk.ck_from_reg_id
        from cc_check_knclient chk
        where chk.ck_from_esm_id = :esm_id and
              chk.ck_type = 6 and
              chk.ck_status >= 1
        order by chk.ck_id desc
        rows 1
        into :ck_secondary_data, :ck_fio, :ck_birth_date, :ck_from_reg_id;

        if (ck_from_reg_id is not null) then
        begin
          select reg.reg_name
          from region reg
          where reg.reg_unicode = :ck_from_reg_id
          into :reg_name;
        end

        select pass_no, cell_no
        from cc_secondary_data_parse(:ck_secondary_data)
        into pass_no_oper, cell_no_oper;
        suspend;
      end
    end
    else
    if (step = 2) then -- выявляю, получиал ли заем, стал ли сотрудником
    begin
      show_phone = coalesce(show_phone, 0);

      for select esm_id, esm_date, cl_name, cl_birthday, cl_citizenship, cl_gender, cl_loan_term, has_loan, loan_date,
                 is_employee, employee_date, phone_no, esm_datetime, city, pass_no_oper, cell_no_oper, ck_fio,
                 ck_birth_date, find_method, cl_next_call_value, cl_next_call_type, cl_kolich_repay, cl_language,
                 reg_name, meeting_point, meeting_address, data_source
          from aarep_website_miniapp_analiz(:rep_date_begin, :rep_date_end, 1, 1, 1)
          where (trim(coalesce(phone_no, '')) <> '')
          into :esm_id, :esm_date, :cl_name, :cl_birthday, :cl_citizenship, :cl_gender, :cl_loan_term, :has_loan,
               :loan_date, :is_employee, :employee_date, :phone_no, :esm_datetime, :city, :pass_no_oper, :cell_no_oper,
               :ck_fio, :ck_birth_date, :find_method, :cl_next_call_value, :cl_next_call_type, :cl_kolich_repay,
               :cl_language, :reg_name, :meeting_point, :meeting_address, :data_source
      do
      begin
        c_id = null;
        find_method = '';

        select res_object_id
        from hp_contdata_find_object(:phone_no, 'cell')
        where res_object_kind = 0 rows 1
        into :c_id;

        if (c_id is not null) then
          find_method = 'найден по номеру телефона';

        if ((c_id is null) and
            (pass_no_oper <> '')) then
        begin
          select cl.c_id
          from clients cl
          where cl.c_pass_no = :pass_no_oper
          into :c_id;

          if (c_id is not null) then
            find_method = 'найден паспорту';
        end

        if ((c_id is null) and
            (cell_no_oper <> '')) then
        begin
          select res_object_id
          from hp_contdata_find_object(:cell_no_oper, 'cell')
          where res_object_kind = 0 rows 1
          into :c_id;

          if (c_id is not null) then
            find_method = 'найден по доп. телефону';
        end

        fuzzy_search = 0;

        if (c_id is null) then -- поиск по фио, дате рождения клиента
        begin
          for select c_id
              from hp_find_clients_tmp(null, :ck_fio, :ck_birth_date, null, null, null, '', 1)
              where c_id <> 0
              into :c_id
          do
          begin
            if (fuzzy_search = 0) then
            begin
              has_loan = 0;
              loan_date = null;
              employee_date = null;
              is_employee = 0;
              k_kolich_client_approved = 0;
              k_kontract_no = '';
              kf_name = '';

              select ks.ks_date_begin, kont.k_kolich_client_approved, kont.k_kontract_no, kal.kal_new_kf_name
              from knclient kncl
              inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
              inner join kontract kont on kont.k_id = ks.ks_from_k_id
              inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
              where ks.ks_ol_is_delete = 0 and
                    kncl.kc_from_c_id = :c_id and
                    ks.ks_state = 1 and
                    ks.ks_date_begin >= :esm_date and
                    current_date between kal.kal_date and kal.kal_date_end and
                    kal.is_delete = 0
              order by ks.ks_date_begin
              rows 1
              into :loan_date, :k_kolich_client_approved, :k_kontract_no, :kf_name;

              if (loan_date is not null) then
              begin
                fuzzy_search = 1;
                find_method = 'нечеткий поиск по фио, дате рождения';
              end
              else
              begin
                select hps.ps_date_begin
                from hr_pstatus hps
                inner join joblist jl on jl.jl_id = ps_from_jl_id
                inner join jobkind jk on jk.jk_id = jl.jl_kind
                where :esm_date between ps_date_begin and ps_date_end and
                      ps_is_working = 1 and
                      hps.ps_status = 1 and
                      hps.ps_from_c_id = :c_id and
                      ps_date_begin >= :esm_date
                order by hps.ps_date_begin
                rows 1
                into :employee_date;

                if (employee_date is not null) then
                begin
                  fuzzy_search = 1;
                  find_method = 'нечеткий поиск по фио, дате рождения';
                end
              end

              if (find_method = '') then
                c_id = null;
            end
          end
        end

        if (fuzzy_search = 0) then
        begin
          has_loan = 0;
          loan_date = null;
          employee_date = null;
          is_employee = 0;
          k_kolich_client_approved = 0;
          k_kontract_no = null;
          kf_name = null;
        end

        c_find_name_fio = '';

        if (c_id is not null) then
        begin
          select cl.c_name_fio
          from clients cl
          where cl.c_id = :c_id
          into :c_find_name_fio;

          if (fuzzy_search = 0) then
          begin
            select ks.ks_date_begin, kont.k_kolich_client_approved, kont.k_kontract_no, kal.kal_new_kf_name
            from knclient kncl
            inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id
            inner join kontract kont on kont.k_id = ks.ks_from_k_id
            inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
            where ks.ks_ol_is_delete = 0 and
                  kncl.kc_from_c_id = :c_id and
                  ks.ks_state = 1 and
                  ks.ks_date_begin >= :esm_date and
                  current_date between kal.kal_date and kal.kal_date_end and
                  kal.is_delete = 0
            order by ks.ks_date_begin
            rows 1
            into :loan_date, :k_kolich_client_approved, :k_kontract_no, :kf_name;
          end

          if (loan_date is not null) then
            has_loan = 1;

          if (fuzzy_search = 0) then
          begin
            select hps.ps_date_begin
            from hr_pstatus hps
            inner join joblist jl on jl.jl_id = ps_from_jl_id
            inner join jobkind jk on jk.jk_id = jl.jl_kind
            where :esm_date between ps_date_begin and ps_date_end and
                  ps_is_working = 1 and
                  hps.ps_status = 1 and
                  hps.ps_from_c_id = :c_id and
                  ps_date_begin >= :esm_date
            order by hps.ps_date_begin
            rows 1
            into :employee_date;
          end

          if (employee_date is not null) then
            is_employee = 1;
        end
        if (show_phone = 0) then
          phone_no = '';
        cl_name = replace(cl_name, ascii_char(13), '');
        cl_name = replace(cl_name, ascii_char(10), '');

        cl_loan_term = replace(cl_loan_term, ascii_char(13), '');
        cl_loan_term = replace(cl_loan_term, ascii_char(10), '');

        phone_no = replace(phone_no, ascii_char(13), '');
        phone_no = replace(phone_no, ascii_char(10), '');

        city = replace(city, ascii_char(13), '');
        city = replace(city, ascii_char(10), '');

        cl_next_call_type = replace(cl_next_call_type, ascii_char(13), '');
        cl_next_call_type = replace(cl_next_call_type, ascii_char(10), '');

        cl_language = replace(cl_language, ascii_char(13), '');
        cl_language = replace(cl_language, ascii_char(10), '');

        meeting_point = replace(meeting_point, ascii_char(13), '');
        meeting_point = replace(meeting_point, ascii_char(10), '');

        meeting_address = replace(meeting_address, ascii_char(13), '');
        meeting_address = replace(meeting_address, ascii_char(10), '');
        suspend;
      end
    end
  end
  if (rep_type = 2) then --  если рассматривается число выданных кредитов за период. нужно определить сколько клиентв до получения займа, оставили заявки на сайте мигранта
  begin
    has_loan = 1;

    for select ks.ks_date_begin, kont.k_kolich_client_approved,
               f_cont_info('clients', kncl.kc_from_c_id, 'cell_main', 'last'), cl.c_name_fio, cl.c_pass_no,
               reg.reg_name
        from kontstate ks
        inner join kontract kont on kont.k_id = ks.ks_from_k_id
        inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
        inner join clients cl on cl.c_id = kncl.kc_from_c_id
        inner join region reg on reg.reg_unicode = ks.ks_region_id
        where ks.ks_state = 1 and
              ks.ks_date_begin between :rep_date_begin and :rep_date_end and
              ks.ks_ol_is_delete = 0 and
              kncl.kc_amount_approved > 0
        order by ks_date_begin
        into :loan_date, :k_kolich_client_approved, phone_no, kc_name_fio, c_pass_no, reg_name
    do
    begin
      esm_id_called = null;
      ck_secondary_data = '';
      pass_no_oper = '';
      cell_no_oper = '';

      /*
         if (dateadd(month, -6, :loan_date) > cast('01.04.2016' as date)) then
           start_date = '01.04.2016';
        else
           start_date = dateadd(month, -6, :loan_date);
           */

      start_date = dateadd(month, -2, :loan_date);

      for select chk.ck_secondary_data, chk.ck_from_esm_id
          from cc_check_knclient chk
          where chk.ck_type = 6 and
                :loan_date between :start_date /*chk.ck_checked_date -14*/ and chk.ck_checked_date and
                coalesce(chk.ck_secondary_data, '') <> '' and
                chk.ck_status >= 1 and
                ck_checked_date is not null
          order by chk.ck_id desc
          into :ck_secondary_data, :ck_from_esm_id
      do
      begin
        if (esm_id_called is null) then
        begin
          pass_no_oper_tmp = '';
          cell_no_oper_tmp = '';

          select pass_no, cell_no
          from cc_secondary_data_parse(:ck_secondary_data)
          into pass_no_oper_tmp, cell_no_oper_tmp;

          if ((c_pass_no = pass_no_oper_tmp) or (phone_no = cell_no_oper_tmp)) then
          begin
            pass_no_oper = pass_no_oper_tmp;
            cell_no_oper = cell_no_oper_tmp;
            esm_id_called = :ck_from_esm_id;
          end
        end
      end

      for select em.esm_id, em.esm_date, em.esm_datetime
          from extsysmain em, extsysfields ef, extsysvalues ev
          where em.esm_project = ef.esf_project and
                em.esm_project = 'website' and
                ev.esv_from_esm_id = em.esm_id and
                ev.esv_from_esf_id = ef.esf_id and
                em.esm_status = 0 -- если = 0 значит успешная команда
                and
                em.esm_date <= :loan_date and
                em.esm_date >= :start_date and
                ef.esf_field_code = 'phone_no' and
                (('7' || esv_value_txt = :phone_no) or
                (esv_value_txt = :phone_no) or
                (em.esm_id = :esm_id_called)) and
                esm_action = 'save_cl_info'
          order by em.esm_date desc
          into :esm_id, :esm_date, :esm_datetime
      do
      begin
        esm_id_old = 0;
        if (show_phone = 0) then
          phone_no = '';

        for select ef.esf_field_code, left(ev.esv_value_txt, 100)
            from extsysfields ef, extsysvalues ev
            where ev.esv_from_esf_id = ef.esf_id and
                  esv_from_esm_id = :esm_id
            into :esf_field_code, :esv_value_txt
        do
        begin
          if (esm_id <> esm_id_old) then
          begin
            if (esm_id_old <> 0) then
            begin
              cl_name = replace(cl_name, ascii_char(13), '');
              cl_name = replace(cl_name, ascii_char(10), '');

              cl_loan_term = replace(cl_loan_term, ascii_char(13), '');
              cl_loan_term = replace(cl_loan_term, ascii_char(10), '');

              city = replace(city, ascii_char(13), '');
              city = replace(city, ascii_char(10), '');

              cl_next_call_type = replace(cl_next_call_type, ascii_char(13), '');
              cl_next_call_type = replace(cl_next_call_type, ascii_char(10), '');

              cl_language = replace(cl_language, ascii_char(13), '');
              cl_language = replace(cl_language, ascii_char(10), '');

              meeting_point = replace(meeting_point, ascii_char(13), '');
              meeting_point = replace(meeting_point, ascii_char(10), '');

              meeting_address = replace(meeting_address, ascii_char(13), '');
              meeting_address = replace(meeting_address, ascii_char(10), '');

              suspend;
            end
            cl_name = '';
            cl_birthday = null;
            cl_citizenship = null;
            cl_gender = '';
            cl_loan_term = '';
            city = '';
            cl_next_call_value = null;
            cl_next_call_type = '';
            cl_kolich_repay = 0;
            cl_language = '';
            meeting_point = '';
            meeting_address = '';
            data_source = 'www.molbulak.ru';

            esm_id_old = esm_id;
          end

          if (esm_id_old = esm_id) then
          begin
            if (esf_field_code = 'cl_name') then
              cl_name = esv_value_txt;
            else
            if (esf_field_code = 'cl_birthday') then
              cl_birthday = cast(esv_value_txt as date);
            else
            if (esf_field_code = 'cl_citizenship') then
              cl_citizenship = esv_value_txt;
            else
            if (esf_field_code = 'cl_gender') then
              cl_gender = esv_value_txt;
            else
            if (esf_field_code = 'cl_loan_term') then
              cl_loan_term = esv_value_txt;
            else
            if (esf_field_code = 'cl_city') then
              city = esv_value_txt;
            else
            if (esf_field_code = 'cl_next_call_value') then
              cl_next_call_value = iif(esv_value_txt <> '', cast(esv_value_txt as timestamp), null);
            else
            if (esf_field_code = 'cl_next_call_type') then
              cl_next_call_type = esv_value_txt;
            else
            if (esf_field_code = 'cl_kolich_repay') then
              cl_kolich_repay = iif(esv_value_txt <> '', cast(esv_value_txt as integer), 0);
            else
            if (esf_field_code = 'cl_language') then
              cl_language = esv_value_txt;
            else
            if (esf_field_code = 'meeting_point') then
              meeting_point = esv_value_txt;
            else
            if (esf_field_code = 'meeting_address') then
              meeting_address = esv_value_txt;
            else
            if (esf_field_code = 'data_source') then
              data_source = esv_value_txt;
          end
        end

        if (esm_id_old <> 0) then
        begin
          cl_name = replace(cl_name, ascii_char(13), '');
          cl_name = replace(cl_name, ascii_char(10), '');

          cl_loan_term = replace(cl_loan_term, ascii_char(13), '');
          cl_loan_term = replace(cl_loan_term, ascii_char(10), '');

          city = replace(city, ascii_char(13), '');
          city = replace(city, ascii_char(10), '');

          cl_next_call_type = replace(cl_next_call_type, ascii_char(13), '');
          cl_next_call_type = replace(cl_next_call_type, ascii_char(10), '');

          cl_language = replace(cl_language, ascii_char(13), '');
          cl_language = replace(cl_language, ascii_char(10), '');

          meeting_point = replace(meeting_point, ascii_char(13), '');
          meeting_point = replace(meeting_point, ascii_char(10), '');

          meeting_address = replace(meeting_address, ascii_char(13), '');
          meeting_address = replace(meeting_address, ascii_char(10), '');

          suspend;
        end
      end
    end
  end
