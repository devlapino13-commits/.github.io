    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    CB_DISB_KOLICH_CLI INTEGER,
    KPI1_1 CURR_TYPE,
    CB_DELINQ_BALANCE CURR_TYPE,
    CB_ACTIV_BALANCE CURR_TYPE,
    PAR_7 PERC_TYPE,
    KPI1_2 CURR_TYPE,
    CB_PERSONAL_STAGE DATE,
    CB_PERSONAL_STAGE_MONTH INTEGER,
    CB_ACTIV_KOLICH_CLI INTEGER,
    KPI1_3 CURR_TYPE,
    RESULT CURR_TYPE)
AS
begin
  for select reg_code || ' ' || reg_name, c_name_fio, cb_disb_kolich_cli, cb_activ_kolich_cli, cb_delinq_balance,
             cb_activ_balance, cb_personal_stage, cb_personal_stage_month
      from cb_ru_lo_2012_12_new2(:rep_region_id, 0, 0, :rep_date_end, 0), region, clients
      where cb_region_id = reg_unicode and
            cb_personal_id = c_id
      into office_name, c_name_fio, cb_disb_kolich_cli, cb_activ_kolich_cli, cb_delinq_balance, cb_activ_balance,
           cb_personal_stage, cb_personal_stage_month
  do
  begin
    kpi1_1 = iif(cb_disb_kolich_cli >= 30, 0.3, cb_disb_kolich_cli * 0.3000 / 30) * 100;

                                                                 end
               when cb_personal_stage_month = 5 then case

               when cb_personal_stage_month = 6 then case

                                                       else 30
                                                     end
               when cb_personal_stage_month = 7 then case
                                                       when cb_activ_kolich_cli < 49 then 0
                                                       when cb_activ_kolich_cli < 52 then 10
                                                       when cb_activ_kolich_cli < 55 then 15
                                                       when cb_activ_kolich_cli < 58 then 20
                                                       else 30
                                                     end
               when cb_personal_stage_month = 8 then case
                                                       when cb_activ_kolich_cli < 56 then 0
                                                       when cb_activ_kolich_cli < 59 then 10
                                                       when cb_activ_kolich_cli < 63 then 15
                                                       when cb_activ_kolich_cli < 66 then 20
                                                       else 30
                                                     end
               when cb_personal_stage_month = 9 then case
                                                       when cb_activ_kolich_cli < 64 then 0
                                                       when cb_activ_kolich_cli < 68 then 10
                                                       when cb_activ_kolich_cli < 71 then 15

                                                       else 30
                                                     end
               when cb_personal_stage_month = 10 then case
                                                        when cb_activ_kolich_cli < 71 then 0
                                                        when cb_activ_kolich_cli < 75 then 10
                                                        when cb_activ_kolich_cli < 79 then 15
                                                        when cb_activ_kolich_cli < 83 then 20
                                                        else 30
                                                      end
               when cb_personal_stage_month = 11 then case

               when cb_personal_stage_month = 17 then case

                                                        when cb_activ_kolich_cli < 136 then 15

               when cb_personal_stage_month > 17 then case

             end;
    result = kpi1_1 + kpi1_2 + kpi1_3;
