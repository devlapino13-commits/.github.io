CREATE PROCEDURE AAREP_SB_MAKE_AUTO_AKT(
    REP_DATE_END DATE,
    REP_EXECUTE_KIND INTEGER)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    SBL_NAME VARCHAR(255),
    REG_NAME VARCHAR(100),
    KF_NAME VARCHAR(100),
    LO_NAME VARCHAR(100),
    DD DATE,
    REPAY_OS DECIMAL(18,4),
    REPAY_PERC DECIMAL(18,4),
    REPAY_FINE DECIMAL(18,4),
    REPAY_ALL DECIMAL(18,4))
AS
declare variable sba_number integer;
declare variable delinq_date date;
declare variable sba_from_sbl_id integer;
declare variable sba_from_k_id decimal(18,0);
declare variable sba_date_begin date;
declare variable sba_date_end date;
declare variable rep_balance decimal(18,4);
declare variable kf_id decimal(18,0);
declare variable dd1 date;
declare variable dd2 date;
declare variable oks_date_begin date;
begin
  if (rep_execute_kind is null) then
    rep_execute_kind = 0;
  for select sba_from_k_id, sba_number, sba_date_begin, sba_date_end, sba_from_sbl_id, sbl_name, k_kontract_no,
             k_disb_date, k_from_kf_id
      from sb_akt, sb_list, kontract
      where sba_is_delete = 0 and
            sba_status = 2 and
            sba_date_begin <= (:rep_date_end + 1) and
            sba_date_end >= (:rep_date_end + 1) and
            sba_date_begin <= sba_date_end and
            sba_from_k_id = k_id and
            sba_from_sbl_id = sbl_id and
            sbl_id <> 0 and
            sbl_type = 0
      into :sba_from_k_id, :sba_number, :sba_date_begin, :sba_date_end, :sba_from_sbl_id, :sbl_name, :k_kontract_no,
           :k_disb_date, :kf_id

  do
  begin
    delinq_date = null;
    dd = null;
    select oks_balance, oks_date_begin, oks_delinq_date
    from oksummary
    where oks_from_k_id = :sba_from_k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :rep_balance, :oks_date_begin, :delinq_date;
    if (rep_balance is null) then
      rep_balance = 0;

    if (rep_balance = 0) then
    begin
      if (not(exists(select *
                     from kontstate
                     where ks_from_k_id = :sba_from_k_id and
                           ks_ol_is_delete = 0 and
                           ks_date_begin <= :rep_date_end and
                           ks_date_end >= :rep_date_end and
                           ks_state = 1))) then
        delinq_date = null; -- если кредит закрыт, не смотрим на него
    end
    if ((delinq_date is not null) and
        (delinq_date > :rep_date_end)) then
    begin
-- Кредит не просрочен, снимаем с учета СБ. Дата: следующий за отчетным
      dd = :oks_date_begin + 1;
      repay_os = 0;
      repay_perc = 0;
      repay_fine = 0;
      repay_all = 0;

      if (rep_balance > 0) then
      begin
        for select oks_date_begin, oks_date_end, oks_delinq_date
            from oksummary
            where oks_from_k_id = :sba_from_k_id and
                  oks_date_begin <= :rep_date_end and
                  oks_close_state = 0
            order by oks_date_begin desc
            into :dd1, :dd2, :delinq_date
        do
        begin
          if (dd2 >= '31.12.2199') then
            dd2 = :rep_date_end;
          if ((delinq_date <= dd1) or (delinq_date <= dd2)) then
            break;
          else
            dd = :dd1 + 1;
          if (dd < sba_date_begin) then
            dd = sba_date_begin;
          if (dd < '01.05.2014') then
            dd = '01.05.2014';
        end
      end
      if (rep_execute_kind <> 0) then
      begin
        if (rep_execute_kind = 1) then
          dd = :rep_date_end + 1;
        insert into sb_akt (sba_from_k_id, sba_number, sba_date_begin, sba_from_sbl_id, sba_status, sba_from_un_id,
                            sba_is_delete)
        values (:sba_from_k_id, -1, :dd, 0, 2, 1000000000001, 0);
      end
      else
      begin
        reg_name = '';
        kf_name = '';
        select reg_code || ' ' || reg_name, kal_new_kf_name, c_name_fio
        from knaktlist, region, clients
        where kal_from_kf_id = :kf_id and
              knaktlist.is_delete = 0 and
              kal_new_reg_id = reg_unicode and
              kal_date <= :rep_date_end and
              kal_date_end >= :rep_date_end and
              kal_new_kf_officer = c_id
        into :reg_name, :kf_name, :lo_name;

        select sum(-ok_os), sum(-ok_perc), sum(-ok_fine)
        from operkred
        where ok_from_k_id = :sba_from_k_id and
              ok_ol_date between :dd and :rep_date_end and
              ok_ol_is_active = 1 and
              ok_from_c_id = 0 and
              ok_type = 4
        into :repay_os, repay_perc, repay_fine;
        if (repay_os is null) then
          repay_os = 0;
        if (repay_perc is null) then
          repay_perc = 0;
        if (repay_fine is null) then
          repay_fine = 0;
        repay_all = repay_os + repay_perc + repay_fine;

        suspend;
      end
    end
  end
end
