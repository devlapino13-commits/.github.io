CREATE PROCEDURE HP_SEND_PUSH_IPF_CHDP
AS
declare variable k_id d_id;
declare variable edl_id d_id;
declare variable k_fact_disb_lo_id d_id;
declare variable un_id d_id;
declare variable res_text d_str_255;
declare variable i d_integer;
declare variable c_id d_id;
begin
  i = 0;
  for select kn.k_id, kn.k_fact_disb_lo_id, kc.kc_from_c_id
      from kontstate ks
      inner join kontract kn on kn.k_id = ks.ks_from_k_id
      inner join product pr on pr.pr_id = kn.k_from_pr_id
      inner join knclient kc on kc.kc_from_k_id = kn.k_id and
            kc.kc_amount_approved > 0
      where ks.ks_date_begin between '10.11.2025' and '28.02.2026' and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            current_date between ks.ks_date_begin and ks.ks_date_end and
            ks.ks_date_begin between '10.11.2025' and '28.02.2026' and
            pr.pr_code containing '.ipf.' and
            f_asp_signed(kc.kc_from_c_id, current_date) <> 0 and
            (f_edoc_status('ipf.chdp.agreement', ks.ks_from_k_id, current_date) in ('empty', 'cancelled'))
      into :k_id, k_fact_disb_lo_id , c_id
  do
  begin
    if (i = 22) then
      exit;

    select un.un_id
    from username un
    where un.un_from_c_id = :k_fact_disb_lo_id
    into un_id;

    select res_text
    from hp_edoc_client_continfo('ipf.chdp.agreement', :c_id, :un_id)
    into res_text;
    if (res_text = '') then
    begin
      select edl_id, res_text
      from hp_edoc_insert('ipf.chdp.agreement', f_objkind('kontract', 'id'), :k_id, :un_id)
      into :edl_id, :res_text;

      if (res_text <> '') then
      begin
        insert into err_log (erl_id, edl_text) -- на первое время, для быстрого анализа
        values (:edl_id, :res_text);
      end
      else
        i = i + 1;
    end
  end
end
