CREATE PROCEDURE HP_PDN_KONTRACT_INFO(
    KC_ID D_ID,
    REP_DATE D_DATE,
    TILL_OL_ID D_ID = null)
RETURNS (
    RES_DELINQ_DATE D_DATE,
    RES_DELINQ_OS D_CURRENCY,
    RES_DELINQ_PERC D_CURRENCY,
    RES_FUTURE_OS D_CURRENCY,
    RES_FUTURE_PERC_WITH_CALC D_CURRENCY,
    RES_FUTURE_CALC_PERC D_CURRENCY,
    RES_FUTURE_PERC_WITHOUT_CALC D_CURRENCY)
AS
declare variable sh_date d_date;
declare variable zad_os d_currency;
declare variable zad_perc d_currency;
declare variable calc_perc d_currency;
declare variable payed_calc_perc d_currency;
declare variable new_os d_currency;
declare variable new_perc d_currency;
declare variable k_id d_id;
declare variable c_id d_id;
begin
  res_delinq_date = null;
  res_delinq_os = 0;
  res_delinq_perc = 0;
  res_future_os = 0;
  res_future_perc_with_calc = 0;
  res_future_calc_perc = 0;
  res_future_perc_without_calc = 0;

  select kc_from_k_id, kc_from_c_id
  from knclient
  where kc_id = :kc_id
  into k_id, c_id;

  calc_perc = f_perc_saldo(k_id, rep_date);

  for select ok_sh_date,
             sum(ok_os + ok_sh_os),
             sum(ok_perc + ok_sh_perc),
             sum(iif(ok_ol_date = :rep_date and ok_from_ol_id > :till_ol_id, ok_os + ok_sh_os, 0)),
             sum(iif(ok_ol_date = :rep_date and ok_from_ol_id > :till_ol_id, ok_perc + ok_sh_perc, 0)),
             -sum(iif(ok_ol_date = :rep_date and ok_from_ol_id > :till_ol_id and ok_type in (4, 5), ok_perc, 0))
  from kn_operkred
  where ok_from_kc_id = :kc_id and
        ok_type < 7 and
        ok_ol_is_active = 1 and
        ok_ol_date <= :rep_date
  group by ok_sh_date
  union all
  select '01.02.1900',
             sum(oko_os + oko_sh_os),
             sum(oko_perc + oko_sh_perc),
             sum(iif(oko_ol_date = :rep_date and oko_from_ol_id > :till_ol_id, oko_os + oko_sh_os, 0)),
             sum(iif(oko_ol_date = :rep_date and oko_from_ol_id > :till_ol_id, oko_perc + oko_sh_perc, 0)),
             -sum(iif(oko_ol_date = :rep_date and oko_from_ol_id > :till_ol_id and oko_type in (4, 5), oko_perc, 0))
  from operkroff
  where oko_from_k_id = :k_id and
        oko_from_c_id = :c_id and
        oko_ol_is_active = 1 and
        oko_ol_date <= :rep_date
  group by 1
  into sh_date, zad_os, zad_perc, new_os, new_perc, payed_calc_perc
  do
  begin
    zad_os = zad_os - new_os;
    zad_perc = zad_perc - new_perc;
    calc_perc = calc_perc + payed_calc_perc;

    if (sh_date < rep_date) then
    begin
      res_delinq_os = res_delinq_os + zad_os;                 -- Сумма ТЕКУ�©ЕЙ просроченной задолженности по основному долгу
      res_delinq_perc = res_delinq_perc + zad_perc; -- Сумма просроченной задолженности по процентам

      if (res_delinq_date is null and (zad_os + zad_perc) > 0) then
        res_delinq_date = sh_date;
    end
    else
    begin
      res_future_os = res_future_os + zad_os;             -- Сумма срочной задолженности по основному долгу
      res_future_perc_with_calc = res_future_perc_with_calc + zad_perc;  -- Будущие %% по графику
    end
  end

  if (res_future_os < 0) then res_future_os = 0;
  if (res_future_perc_with_calc < 0) then res_future_perc_with_calc = 0;
  if (res_delinq_os < 0) then res_delinq_os = 0;
  if (res_delinq_perc < 0) then res_delinq_perc = 0;
  if (res_delinq_perc < 0) then res_delinq_perc = 0;

  if (res_delinq_date is not null and res_delinq_perc + res_delinq_os = 0) then
    res_delinq_date = null;

  res_future_calc_perc = calc_perc - res_delinq_perc;     -- Срочные начисленные %%
  if (res_future_calc_perc < 0) then res_future_calc_perc = 0;

  res_future_perc_without_calc = res_future_perc_with_calc - res_future_calc_perc; -- Сумма процентов по графику БЕЗ начисленных %%
  if (res_future_perc_without_calc < 0) then res_future_perc_without_calc = 0;

  suspend;
end
