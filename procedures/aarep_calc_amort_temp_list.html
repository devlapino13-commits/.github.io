CREATE PROCEDURE AAREP_CALC_AMORT_TEMP_LIST(
    REG_ID INTEGER,
    YEAR_NO INTEGER,
    MONTH_NO INTEGER)
RETURNS (
    PCA_BILL_ID DECIMAL(18,0),
    PCA_BILL_LINE_NO INTEGER,
    PCA_START_YEAR INTEGER,
    PCA_START_MONTH INTEGER,
    PCA_YEAR_NO INTEGER,
    PCA_MONTH_NO INTEGER,
    PCA_AGE INTEGER,
    PCA_TOTAL_OS DECIMAL(18,4),
    PCA_TOTAL_AMORT DECIMAL(18,4),
    PCA_LIFETIME INTEGER,
    PCA_LIFETIME_SALDO INTEGER,
    PCA_CURR_OS DECIMAL(18,4),
    PCA_CURR_AMORT DECIMAL(18,4),
    PCA_CURR_AMORT_OTHER DECIMAL(18,4),
    PCA_REGION_ID INTEGER)
AS
declare variable pca_koef integer;
declare variable pca_lt_old integer;
declare variable pca_id_old decimal(18,0);
declare variable pca_id_line_no integer;
declare variable dont_calc smallint;
begin
  /* Procedure Text */
  pca_id_old = 0;
  pca_id_line_no = 0;
  pca_lt_old = 0;
  for select oam_bill_id, oam_bill_line_no, oam_koef, oam_start_year, oam_start_month, ol_from_reg_id,
             sum(oal_lifetime_month), cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4)), sum(oal_dont_calc),
             cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4))
      from operamormain, operamorlist, operlist
      where oam_is_delete = 0 and
            operlist.is_delete = 0 and
            oal_bill_id = oam_bill_id and
            oal_bill_line_no = oam_bill_line_no and
            oal_is_delete = 0 and
            ol_id = oal_from_ol_id and
            (oal_year * 12 + oal_month) <= (:year_no * 12 + :month_no) and
            (oam_start_year * 12 + oam_start_month) <= (:year_no * 12 + :month_no)
      group by oam_bill_id, oam_bill_line_no, oam_calc_type, oam_koef, oam_start_year, oam_start_month, ol_from_reg_id
      union
      select oam_bill_id, oam_bill_line_no, oam_koef, oam_start_year, oam_start_month, ol_from_reg_id,
             cast(sum(0) as decimal(18,4)), sum(oal_summa_os), sum(oal_summa_amort), cast(sum(0) as decimal(18,4)),
             cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4))
      from operamormain, operamorlist, operlist, region_list(:reg_id, 1, -1, 1)
      where oam_is_delete = 0 and
            operlist.is_delete = 0 and
            oal_bill_id = oam_bill_id and
            oal_bill_line_no = oam_bill_line_no and
            oal_is_delete = 0 and
            ol_id = oal_from_ol_id and
            (ol_from_reg_id = id_list) and
            (oal_year * 12 + oal_month) <= (:year_no * 12 + :month_no) and
            (oam_start_year * 12 + oam_start_month) <= (:year_no * 12 + :month_no)
      group by oam_bill_id, oam_bill_line_no, oam_calc_type, oam_koef, oam_start_year, oam_start_month, ol_from_reg_id
      union
      select oam_bill_id, oam_bill_line_no, oam_koef, oam_start_year, oam_start_month, ol_from_reg_id,
             cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4)),
             cast(sum(0) as decimal(18,4)), sum(oal_summa_os), sum(oal_summa_amort), cast(sum(0) as decimal(18,4))
      from operamormain, operamorlist, operlist, region_list(:reg_id, 1, -1, 1)
      where oam_is_delete = 0 and
            oal_bill_id = oam_bill_id and
            oal_bill_line_no = oam_bill_line_no and
            oal_is_delete = 0 and
            oal_oper_type = 1 and
            ol_id = oal_from_ol_id and
            operlist.is_delete = 0 and
            (ol_from_reg_id = id_list) and
            (oal_year * 12 + oal_month) = (:year_no * 12 + :month_no)
      group by oam_bill_id, oam_bill_line_no, oam_calc_type, oam_koef, oam_start_year, oam_start_month, ol_from_reg_id
      union
      select oam_bill_id, oam_bill_line_no, oam_koef, oam_start_year, oam_start_month, ol_from_reg_id,
             cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4)), cast(sum(0) as decimal(18,4)),
             cast(sum(0) as decimal(18,4)), sum(oal_summa_os), cast(sum(0) as decimal(18,4)), sum(oal_summa_amort)
      from operamormain, operamorlist, operlist, region_list(:reg_id, 1, -1, 1)
      where oam_is_delete = 0 and
            oal_bill_id = oam_bill_id and
            oal_bill_line_no = oam_bill_line_no and
            oal_is_delete = 0 and
            oal_oper_type <> 1 and
            ol_id = oal_from_ol_id and
            operlist.is_delete = 0 and
            (ol_from_reg_id = id_list) and
            (oal_year * 12 + oal_month) = (:year_no * 12 + :month_no)
      group by oam_bill_id, oam_bill_line_no, oam_koef, oam_start_year, oam_start_month, ol_from_reg_id
      order by 1, 2, 7 desc
      into :pca_bill_id, :pca_bill_line_no, :pca_koef, :pca_start_year, :pca_start_month, :pca_region_id, :pca_lifetime,
           :pca_total_os, :pca_total_amort, :dont_calc, :pca_curr_os, :pca_curr_amort, :pca_curr_amort_other
  do
  begin
    if (pca_lifetime = 0) then
    begin
      if ((pca_id_old = pca_bill_id) and
          (pca_id_line_no = pca_bill_line_no) and
          (pca_lt_old <> 0)) then
        pca_lifetime = :pca_lt_old;
    end
    pca_year_no = :year_no;
    pca_month_no = :month_no;
    pca_age = (:year_no * 12 + :month_no) - (:pca_start_year * 12 + :pca_start_month) + 1;
    pca_age = :pca_age - :dont_calc;
    if (pca_age < 0) then
      pca_age = 0;
    pca_lifetime_saldo = :pca_lifetime - :pca_age + 1;
    if (pca_lifetime_saldo < 0) then
      pca_lifetime_saldo = 0;

    if (pca_bill_id is null) then
      pca_bill_id = 0;
    if (pca_bill_line_no is null) then
      pca_bill_line_no = 0;

    if ((pca_curr_amort <> 0) or (pca_curr_os <> 0) or (pca_curr_amort_other <> 0) or (pca_total_amort <> 0) or (pca_total_os <> 0)) then
    begin
      suspend;
    end
    pca_id_old = :pca_bill_id;
    pca_id_line_no = :pca_bill_line_no;
    pca_lt_old = :pca_lifetime;
  end
end
