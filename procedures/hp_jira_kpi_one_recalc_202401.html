CREATE PROCEDURE HP_JIRA_KPI_ONE_RECALC_202401(
    DEV_ID D_ID,
    CALC_PERIOD D_INTEGER,
    ADD_NORMA_HOUR D_INTEGER = 0,
    NEED_SAVE_KPI D_BOOL = 0)
RETURNS (
    JRC_KPI_1 D_CURRENCY,
    JRC_KPI_2 D_CURRENCY,
    JRC_KPI_3 D_CURRENCY,
    JRC_KPI_4 D_CURRENCY,
    JRC_KPI_5 D_CURRENCY,
    JRC_NORM_HOURS D_INTEGER,
    JRC_DONE_TASKS D_INTEGER,
    JRC_DONE_IN_TIME D_INTEGER,
    JRC_WORK_HOURS D_CURRENCY,
    JRC_AVG_WORK_EXPECT D_CURRENCY,
    JRC_FACT_WORK_HOURS D_INTEGER,
    JRC_TASKS D_INTEGER,
    JRC_TOTAL_DIFFICULT D_INTEGER,
    JRC_DISTINCT_DIFFICULT D_INTEGER,
    JRC_AVG_DIFFICULT D_CURRENCY,
    JRC_REWORKED D_INTEGER,
    JRC_KPI_PERC_1 D_CURRENCY,
    JRC_KPI_PERC_2 D_CURRENCY,
    JRC_KPI_PERC_3 D_CURRENCY,
    JRC_KPI_PERC_4 D_CURRENCY,
    JRC_KPI_PERC_5 D_CURRENCY,
    JRC_KPI_FINAL D_CURRENCY)
AS
declare variable rep_date_begin d_datetime;
declare variable rep_date_end d_datetime;
declare variable kpi_id d_integer;
declare variable res_value d_currency;
declare variable res_info d_smallint;
declare variable res_comment d_str_255;
declare variable tmp_date d_datetime;
declare variable rep_month d_integer;
declare variable delta_hour d_integer;
declare variable tmp_second d_integer;
declare variable tmp_start d_datetime;
declare variable tmp_finish d_datetime;
declare variable prv_level d_integer;
declare variable dev_level d_integer;
declare variable tmp_issue_id d_integer;
declare variable tmp_level d_integer;
declare variable jrd_country d_str_10;
declare variable db_id d_integer;
declare variable jrd_inactive d_bool;
declare variable c_id d_id;
begin
  rep_date_begin = cast('01.' || right(calc_period, 2) || '.' || left(calc_period, 4) as date);
  rep_date_end = dateadd(month, 1, rep_date_begin);
  rep_date_end = dateadd(second, -1, rep_date_end);

  rep_month = mod(calc_period, 100);

  if (add_norma_hour is null) then
    add_norma_hour = 0;

  if (need_save_kpi is null) then
    need_save_kpi = 0;

  dev_level = 1;
  select jrd_level, jrd_from_c_id, jrd_country, jrd_inactive
  from jira_dev jd
  where jrd_id = :dev_id
  into dev_level, c_id, jrd_country, jrd_inactive;

  if (jrd_inactive = 1) then
    exit;

  db_id = f_get_db_id();

  if ((db_id = 81001 and jrd_country <> 'ru') or (db_id <> 81001 and jrd_country = 'ru')) then
    exit;

  if (need_save_kpi = 1) then
  begin
    delete from jira_kpi
    where jrk_period = :calc_period and
          jrk_dev_id = :dev_id;
  end

  jrc_kpi_1 = 0;
  jrc_kpi_2 = 0;
  jrc_kpi_3 = 0;
  jrc_kpi_4 = 0;
  jrc_kpi_5 = 0;
  jrc_norm_hours = 0;
  jrc_done_tasks = 0;
  jrc_done_in_time = 0;
  jrc_work_hours = 0;
  jrc_avg_work_expect = 0;
  jrc_fact_work_hours = 0;
  jrc_tasks = 0;
  jrc_total_difficult = 0;
  jrc_distinct_difficult = 0;
  jrc_avg_difficult = 0;
  jrc_reworked = 0;
  jrc_kpi_perc_1 = 0;
  jrc_kpi_perc_2 = 0;
  jrc_kpi_perc_3 = 0;
  jrc_kpi_perc_4 = 0;
  jrc_kpi_perc_5 = 0;
  jrc_kpi_final = 0;

  -- Начало расчета KPI-1
  kpi_id = 1;

  for select jri_issue_id, jrt_kpi_kind, jri_estimate_secs, jri_worked_secs
  from jira_issues, jira_types, jira_states
  where jri_dev_id = :dev_id and
        jri_type_id = jrt_id and
        jri_state_id = jrs_id and
        jrs_result >= 0 and
        jrt_kpi_kind <> 0 and
        jri_finished between :rep_date_begin and :rep_date_end
  as cursor cur
  do
  begin
    res_value = 0;
    res_info = 0;
    res_comment = '';

    if (cur.jri_estimate_secs > 0) then
    begin
      if (cur.jri_worked_secs <= cur.jri_estimate_secs) then
      begin
        res_value = 1; -- Выполнен в срок
        res_info = 1;
      end else
      begin
        res_value = 0; -- Выполнен с опозданием
        res_info = -1;
        res_comment = 'Выполнена с опозданием';
      end
    end else
    begin
      res_value = 0; -- Выполнен с опозданием
      res_info = -1;
      res_comment = 'Ожидаемый срок не указан';
    end

    if (cur.jrt_kpi_kind = 2) then -- Тип задачи где не штрафуется за задержку (Например: Идеи и планы)
      res_value = 1;

    jrc_done_in_time = jrc_done_in_time + res_value;
    jrc_done_tasks = jrc_done_tasks + 1;

    if (need_save_kpi = 1) then
    begin
      insert into jira_kpi (jrk_period, jrk_dev_id, jrk_kpi, jrk_issues_id, jrk_value, jrk_info, jrk_comment)
      values (:calc_period, :dev_id, :kpi_id, cur.jri_issue_id, :res_value, :res_info, :res_comment);
    end
  end

  if (jrc_done_tasks <> 0) then
  begin
    jrc_kpi_1 = round(cast(jrc_done_in_time as float) / jrc_done_tasks, 2);
    jrc_kpi_perc_1 = round(jrc_kpi_1 * 100, 2);
  end

  -- Начало расчета KPI-2
  kpi_id = 2;
  res_info = 0;
  res_value = 0;
  res_comment = '';

  tmp_date = rep_date_begin;
  while (extract(month from tmp_date) = rep_month) do
  begin
    if (f_jira_workday(tmp_date) = 1) then
    begin
      jrc_norm_hours = jrc_norm_hours + 8;

      if (c_id is not null) then
      begin
        if (exists(select * from hr_orders, hr_voctravel, sa_worktype
                   where hrv_from_c_id = :c_id and
                         hrv_date_begin <= :tmp_date and
                         hrv_date_end_fact >= :tmp_date and
                         hrv_from_ho_id = ho_id and
                         sawt_id = ho_order_type and
                         sawt_base_type in (4, 5) and
                         sawt_hours_real >= 0)) then
          jrc_norm_hours = jrc_norm_hours - 8;
      end
    end

    tmp_date = tmp_date + 1;
  end

  tmp_date = cast(rep_date_begin as date);
  while (extract(month from tmp_date) = rep_month) do
  begin
    res_value = 0;
    res_info = 0;
    res_comment = f_datetostr(tmp_date);

    if (c_id is not null) then
    begin
      if (exists(select * from hr_orders, hr_voctravel, sa_worktype
                 where hrv_from_c_id = :c_id and
                       hrv_date_begin <= :tmp_date and
                       hrv_date_end_fact >= :tmp_date and
                       hrv_from_ho_id = ho_id and
                       sawt_id = ho_order_type and
                       sawt_base_type in (4, 5) and
                       sawt_hours_real >= 0)) then
        res_comment = res_comment || '  - dayoff';
    end

    if (f_jira_workday(tmp_date) = 0) then
    begin
      res_comment = f_datetostr(tmp_date);

      if (extract(weekday from tmp_date) in (0, 6)) then
        res_comment = res_comment || '  - weekend';
      else
        res_comment = res_comment || '  - holiday';
    end
    else
    begin
      delta_hour = 5;
      res_value = 0;

      if (exists(select * from jira_issues, jira_types, jira_states
                 where jri_dev_id = :dev_id and
                       jri_state_id = jrs_id and
                       jri_type_id = jrt_id and
                       jrt_kpi_kind <> 0 and
                       jrs_code = 'work' and
                       jri_state_datetime < :tmp_date)) then
        res_value = 8;
      else
      begin
        while (delta_hour < 21) do
        begin
          tmp_start = dateadd(hour, :delta_hour, :tmp_date);
          tmp_finish = dateadd(hour, :delta_hour + 1, :tmp_date);
  
          tmp_second = 0;
  
          select sum(f_jira_diff(maxvalue(jrl_old_datetime, :tmp_start), minvalue(jrl_new_datetime, :tmp_finish), jrl_kpi_timezone))
          from jira_log, jira_issues, jira_types, jira_states
          where jrl_dev_id = :dev_id and
                jrl_issue_id = jri_issue_id and
                jri_type_id = jrt_id and
                jrt_kpi_kind <> 0 and
                jrl_old_state = jrs_id and
                jrs_code = 'work' and
                jrl_old_datetime <= :tmp_finish and
                jrl_new_datetime >= :tmp_start
          into tmp_second;
  
          if (tmp_second is null) then
            tmp_second = 0;
  
          if (tmp_second > 60 * 60) then
            tmp_second = 60 * 60;

          res_value = res_value + tmp_second;
  
          delta_hour = delta_hour + 1;
        end
  
        res_value = round(res_value / 60 / 60, 2);
        if (trunc(res_value) + 0.1 <= res_value) then
          res_value = trunc(res_value) + 1;
        else
          res_value = trunc(res_value);
  
        if (res_value > 8) then
          res_value = 8;
  
        if (res_value <= 7) then
          res_info = -1;
      end

      jrc_work_hours = jrc_work_hours + res_value;
    end

    if (need_save_kpi = 1) then
    begin
      insert into jira_kpi (jrk_period, jrk_dev_id, jrk_kpi, jrk_issues_id, jrk_value, jrk_info, jrk_comment)
      values (:calc_period, :dev_id, :kpi_id, null, :res_value, :res_info, :res_comment);
    end

    tmp_date = tmp_date + 1;
  end

  if (jrc_norm_hours <> 0) then
  begin
    jrc_kpi_2 = round(cast(jrc_work_hours as float) / (jrc_norm_hours * 0.900), 2);
    jrc_kpi_perc_2 = minvalue(round(jrc_kpi_2 * 100.00, 2), 100.00);
  end

  -- Начало расчета KPI-3
  kpi_id = 3;
  res_info = 0;
  res_value = 0;
  res_comment = '';

  if (dev_level = 0) then
    dev_level = 1;

  prv_level = -1;
  for select distinct jri_level, jri_issue_id
  from jira_log jl, jira_issues, jira_types, jira_states js_old, jira_states js_new
  where jrl_dev_id = :dev_id and
        jrl_issue_id = jri_issue_id and
        jri_type_id = jrt_id and
        jrt_kpi_kind > 0 and
        jrl_old_state = js_old.jrs_id and
        jrl_new_state = js_new.jrs_id and
        ((js_old.jrs_code = 'work' and jrl_old_datetime <= :rep_date_end and jrl_new_datetime >= :rep_date_begin) or
         (js_new.jrs_code = 'work' and jrl_new_datetime between :rep_date_begin and :rep_date_end))
  into tmp_level, tmp_issue_id
  do
  begin
    if (prv_level <> tmp_level) then
    begin
      jrc_distinct_difficult = jrc_distinct_difficult + tmp_level;
      prv_level = tmp_level;
    end

    jrc_tasks = jrc_tasks + 1;
    jrc_total_difficult = jrc_total_difficult + tmp_level;

    res_comment = '';
    if (need_save_kpi = 1) then
    begin
      res_value = tmp_level;

      res_info = 0;
      if (tmp_level < dev_level) then
      begin
        res_info = -1;
        res_comment = 'Сложность ниже';
      end
      else
      if (tmp_level > dev_level) then
      begin
        res_info = 1;
        res_comment = 'Сложность выше';
      end

      insert into jira_kpi (jrk_period, jrk_dev_id, jrk_kpi, jrk_issues_id, jrk_value, jrk_info, jrk_comment)
      values (:calc_period, :dev_id, :kpi_id, :tmp_issue_id, :res_value, :res_info, :res_comment);
    end
  end

  if (jrc_tasks <> 0) then
    jrc_avg_difficult = round(cast(jrc_total_difficult as float) / cast(jrc_tasks as float), 2);

  jrc_kpi_3 = round((cast(jrc_tasks as float) * jrc_avg_difficult)/(10.00 * dev_level), 2);
  jrc_kpi_perc_3 = minvalue(round(jrc_kpi_3 * 100.00, 2), 100.00);

  -- Начало расчета KPI-4
  kpi_id = 4;
  res_info = 0;
  res_value = 0;
  res_comment = '';

  jrc_fact_work_hours = minvalue(jrc_work_hours, jrc_norm_hours);

  if (jrc_tasks * jrc_avg_difficult <> 0) then
    jrc_avg_work_expect = cast(jrc_fact_work_hours as float) / (jrc_tasks * jrc_avg_difficult);

  jrc_kpi_4 = iif(jrc_avg_work_expect > 30, 0, iif(jrc_avg_work_expect >= 20, 0.5, 1.0));
  jrc_kpi_perc_4 = round(jrc_kpi_4 * 100.00, 2);

  -- Начало расчета KPI-5
  kpi_id = 5;
  res_info = 0;
  res_value = 0;
  res_comment = 'Доработка';

  for select distinct jri_issue_id
  from jira_log jl, jira_issues, jira_types, jira_states
  where jrl_dev_id = :dev_id and
        jrl_issue_id = jri_issue_id and
        jri_type_id = jrt_id and
        jrt_kpi_kind > 0 and
        jrl_new_state = jrs_id and
        jrs_code = 'rework' and
        jrl_new_datetime between :rep_date_begin and :rep_date_end
  into tmp_issue_id
  do
  begin
    jrc_reworked = jrc_reworked + 1;

    if (need_save_kpi = 1) then
    begin
      res_value = 0;

      res_info = -1;

      insert into jira_kpi (jrk_period, jrk_dev_id, jrk_kpi, jrk_issues_id, jrk_value, jrk_info, jrk_comment)
      values (:calc_period, :dev_id, :kpi_id, :tmp_issue_id, :res_value, :res_info, :res_comment);
    end
  end

  res_value = 1;
  if (jrc_tasks <> 0) then
    res_value = round(cast(jrc_reworked as float) / jrc_tasks, 2);

  jrc_kpi_5 = iif(res_value > 0.5, 0, iif(res_value > 0.35, 0.5, iif(res_value > 0.2, 0.8, 1)));
  jrc_kpi_perc_5 = round(jrc_kpi_5 * 100.00, 2);
  res_value = 0;

  jrc_kpi_final = ((jrc_kpi_perc_1 + jrc_kpi_perc_3 + jrc_kpi_perc_4) / 100.00 * jrc_kpi_perc_2) / 100.0* jrc_kpi_perc_5;
  jrc_kpi_final = round(jrc_kpi_final / 3.00, 1);

  suspend;
end
