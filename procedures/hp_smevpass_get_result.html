CREATE PROCEDURE HP_SMEVPASS_GET_RESULT(
    C_ID D_ID,
    MA_ID D_ID,
    REP_DATE D_DATE = null)
RETURNS (
    STATUS_DESC D_STR_255,
    STATUS_CODE D_INTEGER,
    INVALIDITY_REASON_DESC D_STR_255,
    INVALIDITY_REASON_CODE D_INTEGER,
    INVALIDITY_SINCE D_DATE,
    ERROR_TEXT D_STR_255,
    CDS_ID D_ID)
AS
declare variable bls_id d_id;
declare variable json_full_str d_str_8100;
declare variable eil_status d_integer;
declare variable jso d_integer;
declare variable app_sign_date d_date;
begin
  error_text = '';

  if (rep_date is null) then
    rep_date = current_date;

  select cds_id
  from clientdocs cds
  where cds.cds_from_c_id = :c_id
    and cds_is_main_doc = 1
    and cds_is_delete = 0
    and cds_verify_status = 2
    and cds_index_doc_no <> ''
  order by cds_id
  rows 1
  into cds_id;
  
  select min(eds_status_datetime)
  from edoc_sign, edoc_list, edoc_blank edb
  where eds_from_edl_id = edl_id
    and edb_from_edl_id = edl_id
    and edl_from_edc_id = f_edoc_kind('loan.application','id')
    and edl_object_kind = 11
    and edl_object_id = :ma_id
    and eds_status = 1
  into app_sign_date;

  if (app_sign_date is null) then
  begin
    error_text = 'Отсутствует подписанная заявка';
    suspend;
    exit;
  end

  select eil_from_bls_id, eil_status
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id
    and eil_from_eic_id = eic_id
    and eic_code = 'passport_smev'
    and eil_object_kind = f_objkind('clientdocs', 'id')
    and eil_object_id = :cds_id
    and eil_date between :app_sign_date and :rep_date
  order by eil_status desc, eil_date desc, eil_id desc
  rows 1
  into bls_id, eil_status;

  if (bls_id is null) then
  begin
    error_text = 'Отсутствует проверка паспорта по СМЭВ';
    suspend;
    exit;
  end

  if (eil_status < 0) then
  begin
    error_text = 'Ошибка проверки паспорта по СМЭВ';
    suspend;
    exit;
  end

  if (eil_status < 2) then
  begin
    error_text = 'Паспорт клиента в процессе проверки по СМЭВ. Дождитесь результатов.';
    suspend;
    exit;
  end

  select cast(blx_blob as d_blob_text)
  from hp_get_blobex_blob(:bls_id)
  into :json_full_str;

  if (json_full_str is null) then
  begin
    error_text = 'Ошибка проверки паспорта по СМЭВ: json пустой';
    suspend;
    exit;
  end

  -- разбор json
  jso = json.parse(json_full_str);

  status_desc = json.obj_get_val(jso, 'status_desc');
  status_code = json.obj_get_val(jso, 'status_code');
  invalidity_reason_desc = json.obj_get_val(jso, 'invalidity_reason_desc');
  invalidity_reason_code = json.obj_get_val(jso, 'invalidity_reason_code');
  invalidity_since = json.obj_get_val(jso, 'invalidity_since');

  suspend;
end
