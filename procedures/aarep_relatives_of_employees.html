CREATE PROCEDURE AAREP_RELATIVES_OF_EMPLOYEES(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    RLL_ID INTEGER,
    REG_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    JK_NAME VARCHAR(100),
    REL_FIO VARCHAR(100),
    REL_NAME VARCHAR(100),
    REL_JL_NAME VARCHAR(100),
    REL_JK_NAME VARCHAR(100),
    REL_OFFICE VARCHAR(100),
    REL_MEMO VARCHAR(255),
    REL_CODE VARCHAR(50),
    REL_KIND VARCHAR(100),
    RLL_LINK_STATUS INTEGER,
    RLL_LINK_STATUS_STR VARCHAR(50))
AS
begin
  if (rep_date_end is null) then
    rep_date_end = current_date;

  for select rll_id, reg_name, c_name_fio, jl_name, jk_name, rel_fio, rel_name, rel_jl_name, rel_jk_name, rel_office,
             rel_memo, rel_code, rel_kind, rll_link_status, rll_link_status_str
      from aarep_relatives_link(:rep_date_end, 0), region_list(:rep_region_id, 1, -1, 1)
      where reg_id_of_1 = id_list
      into :rll_id, :reg_name, :c_name_fio, :jl_name, :jk_name, :rel_fio, :rel_name, :rel_jl_name, :rel_jk_name,
           :rel_office, :rel_memo, :rel_code, :rel_kind, rll_link_status, rll_link_status_str
  do
  begin
    suspend;
  end
end
