CREATE PROCEDURE CHECK_KONTRACT_CONDITIONS(
    K_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255),
    RES_ID INTEGER)
AS
declare variable KC_BUSUNESS_SUBTYPE integer;
declare variable C_SEX integer;
declare variable C_BIRTH_DATE date;
declare variable C_PASS_NO varchar(30);
declare variable C_PASS_DATE date;
declare variable C_PASS_WHO varchar(255);
declare variable C_NAME_FIO varchar(100);
declare variable C_TEMP varchar(100);
declare variable K_DISB_DATE date;
declare variable C_PASS_DATE_EXPIRE date;
declare variable PR_IS_EMPLOYER integer;
declare variable C_ID decimal(18,0);
declare variable C_CITIZEN integer;
declare variable C_NATION integer;
declare variable C_DOC_TYPE integer;
declare variable PR_DATE_END date;
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  /* Procedure Text */
  res_id = 0;
  res_text = '';
  for select c_id, c_name_fio, kc_busuness_subtype, c_sex, c_birth_date, c_pass_no, c_pass_date, c_pass_who,
             k_disb_date, c_pass_date_expire, pr_is_employer, c_citizen, c_nation, c_doc_type, pr_date_end,
             k_disb_date
      from knclient, clients, kontract, product
      where kc_from_k_id = :k_id and
            kc_from_c_id = c_id and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null and
            c_type = 0 and
            kc_from_k_id = k_id and
            k_from_pr_id = pr_id
      into :c_id, :c_name_fio, :kc_busuness_subtype, :c_sex, :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who,
           :k_disb_date, :c_pass_date_expire, :pr_is_employer, :c_citizen, :c_nation, :c_doc_type, :pr_date_end,
           k_disb_date
  do
  begin
    if (pr_date_end < k_disb_date) then
    begin
      res_id = res_id + 1;
      res_text = 'Кредитный продукт недействителен на дату выдачи';
      suspend;
    end
    if ((kc_busuness_subtype is null) or (kc_busuness_subtype = 0)) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' тип бизнеса не указана';
      suspend;
    end
    if (c_sex is null) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' пол не указан';
      suspend;
    end
    if ((c_citizen is null) or (c_citizen = 0)) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' гражданство не указано';
      suspend;
    end
    if ((c_nation is null) or (c_nation = 0)) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' национальность не указана';
      suspend;
    end
    if ((c_doc_type is null) or (c_doc_type = 0)) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' тип документа не указан';
      suspend;
    end
    if (c_birth_date is null) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' дата рождение не указано';
      suspend;
    end

    if (c_pass_date_expire < k_disb_date) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' срок действия паспорта (документа) истек';
      suspend;
    end

    if (c_birth_date is not null) then
    begin
      if (dateadd(18 year to :c_birth_date) > :k_disb_date) then
      begin
        res_id = res_id + 1;
        res_text = c_name_fio || ': ' || ' клиент младше 18 лет';
        suspend;
      end
      if (dateadd(100 year to :c_birth_date) < :k_disb_date) then
      begin
        res_id = res_id + 1;
        res_text = c_name_fio || ': ' || ' возраст клиента не правильно указан';
        suspend;
      end
    end
    if ((c_pass_no is null) or (c_pass_no = '')) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' номер удостоверения не указан';
      suspend;
    end
    if (c_pass_date is null) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' дата выдачи удостоверения не указана';
      suspend;
    end
    if ((c_pass_who is null) or (c_pass_who = '')) then
    begin
      res_id = res_id + 1;
      res_text = c_name_fio || ': ' || ' орган выдавщее удостоверение не указано';
      suspend;
    end
    if (pr_is_employer = 0) then
    begin
      -- #employee_disb -->  12.06.2020 igusev добавил условие для боевых выдач этим МиРМ, в рамках боевого текста КМП
      if (c_id not in (
            1010100005782, -- Азаев Азамат Бактыбекович
            1010100020434, -- Илалов Руслан Асхатович
            1010100059428, -- Мирзаева Эркин
            1010100007176)) then -- Усманов Ахад --------------------- <
      begin
          if (exists(select *
                     from hr_pstatus
                     where ps_from_c_id = :c_id and
                           ps_date_begin <= :k_disb_date and
                           ps_is_working = 1 and
                           ps_date_end >= :k_disb_date)) then
          begin
            res_id = res_id + 1;
            res_text = c_name_fio || ': ' || ' Сотрудник не может получить кредит по этому продукту';
            suspend;
          end
      end

    end
  end
end
