CREATE PROCEDURE INN_KK_PRINT(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    ACL_DATE D_DATE,
    ACL_DATETIME D_DATETIME,
    ACL_LAST_DATETIME D_DATETIME,
    NUM_SEC D_INTEGER,
    STATUS D_STR_CODE,
    MEMO D_BLOB_TEXT,
    CL_NAME D_STR_100,
    EMP_NAME D_STR_100,
    UN_FIO D_STR_100,
    ICQ_DOC_NO D_STR_20,
    ICQ_DOC_DATE D_DATE,
    INN_STATUS D_STR_CODE,
    REG_NAME D_STR_CODE)
AS
declare variable rd_id d_integer;
begin
  rd_id = f_get_rd_id('ActionGroup.Manual_Inn_Check');
  for
    select
      acl_date,
      acl_datetime,
      acl_last_datetime,
      datediff(second, acl_datetime, acl_last_datetime),
      iif(acl_status = -1, 'доработка', 'подтвержден'),
      iif(acl_status = -1, (select obm_memo from hp_objectprops_memo_get('0', acl_object_id, -1, :rd_id, 1)), ''),
      cl1.c_name_fio,
      cl2.c_name_fio,
      un_fio,
      icq_doc_no,
      icq_doc_date,
      iif(icq_inn_received = 1, 'найден', 'не найден'),
      reg_name
    from actionlist
      inner join region_list(:rep_region_id, -1, 3, 1)
        on id_list = acl_from_reg_id
      inner join region
        on reg_unicode = id_list
      inner join clients as cl1
        on cl1.c_id = acl_object_id
      inner join clients as cl2
        on cl2.c_id = acl_respons_c_id
      inner join username
        on un_id = acl_from_un_id
      inner join inn_check_queries
        on icq_acl_id = acl_id
    where acl_date between :rep_date_begin and :rep_date_end
      and acl_status in (-1, 2)
      and acl_object_kind = 0
      and acl_from_rd_id = :rd_id
  into
    :acl_date,
    :acl_datetime,
    :acl_last_datetime,
    :num_sec,
    :status,
    :memo,
    :cl_name,
    :emp_name,
    :un_fio,
    :icq_doc_no,
    :icq_doc_date,
    :inn_status,
    :reg_name
  do
  begin
    suspend;
  end
end
