CREATE PROCEDURE MC_REJECTION_EDOC_STATUS(
    MCL_ID D_ID,
    EDS_ID D_ID)
RETURNS (
    RES_CODE D_STR_255,
    REJECTION_EDOC_STATUS D_SMALLINT,
    ERROR_TEXT D_STR_255)
AS
declare variable edb_status d_integer;
declare variable edl_status d_integer;
begin
  -- RFCKG-4445
  -- Если не добавить это поле, то КМП считает error_text фатальной ошибкой и ломается
  res_code = 'success';
  rejection_edoc_status = -1;
  error_text = '';

  select edb_status, edl_status
  from edoc_sign, edoc_list, edoc_blank
  where edl_id = eds_from_edl_id
    and edb_from_edl_id = edl_id
    and eds_id = :eds_id
  into edb_status, edl_status;

  if (edb_status is null) then
    error_text = 'Документ не найден';
  else if (edl_status < 0) then
    error_text = 'Документ недействителен';
  else if (edb_status < 0) then
    error_text = 'Ошибка приготовления документа';
  else if (edb_status = 0) then
    rejection_edoc_status = 0;
  else if (edb_status > 0) then
    rejection_edoc_status = 1;

  suspend;
end
