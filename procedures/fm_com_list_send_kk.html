CREATE PROCEDURE FM_COM_LIST_SEND_KK(
    CML_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable cmc_com_kind d_str_code;
declare variable obj_id d_id;
declare variable curr_balance d_currency;
declare variable new_balance d_currency;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable asp_status d_str_code;
declare variable cml_date d_date;
declare variable lang d_str_lang;
declare variable cmc_auto_approve d_smallint;
declare variable db_id d_integer;
declare variable cml_add_params d_str_long;
declare variable monthly_income d_integer;
declare variable cmc_id d_integer;
begin
  -- Процедура отправляет запрос на кредитный комитет
  res_text = '';
  res_is_error = 0;
  lang = f_get_lang_code(un_id, null);
  db_id = f_get_db_id();

  select cmc_id, cml_object_id, cml_date, cmc_auto_approve, f_enum('com_kind', cmc_com_kind, 'code'), cml_add_params
  from com_list, com_cfg
  where cml_id = :cml_id and
        cml_from_cmc_id = cmc_id
  into cmc_id, obj_id, cml_date, cmc_auto_approve, cmc_com_kind, cml_add_params;

  if (exists(select * from com_cfg_params
             where cmcp_from_cmc_id = :cmc_id and cmcp_param_name = 'reason')) then
  begin
    if (f_json_str_get_value(cml_add_params, 'reason', '0') = '0') then
    begin
      res_is_error = 1;
      res_text = f_lang(lang, 'Не определена причина реструктуризации');
    end
  end

  if (cmc_com_kind = 'reschedule') then
  begin
    if (cml_date <> current_date) then
    begin
      res_is_error = 1;
      res_text = f_lang(lang, 'Обновите дату запроса на текущую');
    end

    if (res_text = '' and not exists(select *
                   from com_list_resh
                   where cmr_from_cml_id = :cml_id)) then
    begin
      res_is_error = 1;
      res_text = f_lang(lang, 'Новый график погашения не построен');
    end

    if (res_text = '') then
    begin
      if (exists(select *
                 from com_list_resh
                 where cmr_from_cml_id = :cml_id and
                       (cmr_sh_os < 0 or
                       cmr_sh_perc < 0))) then
      begin
        res_is_error = 1;
        res_text = f_lang(lang, 'В графике погашений имеются отрицательные суммы');
      end
    end

    if (res_text = '') then
    begin
      curr_balance = 0;
      new_balance = 0;
      select sum(ok_os + ok_sh_os)
      from operkred
      where ok_from_k_id = :obj_id and
            ok_ol_is_active = 1 and
            ok_type <= 6
      into curr_balance;
      if (curr_balance is null) then
        curr_balance = 0;

      select sum(cmr_sh_os)
      from com_list_resh
      where cmr_from_cml_id = :cml_id
      into new_balance;
      if (new_balance is null) then
        new_balance = 0;

      if (new_balance <> curr_balance) then
      begin
        res_is_error = 1;
        res_text = f_lang(lang, 'Имеется расхождение в сумме основного долга');
      end
    end

    if (res_text = '') then
    begin
      for select kc_from_c_id, c_name_fio
          from knclient, clients
          where kc_from_k_id = :obj_id and
                kc_amount_approved > 0 and
                kc_from_c_id = c_id
          into c_id, c_name_fio
      do
      begin
        if (db_id not in (39801, 41701)) then
        begin
          asp_status = f_edoc_status('asp.agreement', c_id, current_date);
          if (asp_status not in ('signed', 'inactive')) then
            res_is_error = 1;
        end
        else
        begin
          if (f_asp_signed(:c_id, current_date) = 0) then
            res_is_error = 1;
        end

        if (res_is_error = 1) then
        begin
          if (res_text <> '') then
            res_text = res_text || ', ';
          res_text = res_text || c_name_fio;
        end
      end
      if (res_text <> '') then
        res_text = f_lang(lang, f_format('Нет договора %1: ', iif(db_id = 81001, 'АСП', 'ЭЦП'))) || res_text;
    end
  end

  if (res_text = '' and exists(select * from com_cfg_params
             where cmcp_from_cmc_id = :cmc_id and cmcp_param_name = 'monthly_income')) then
  begin
    -- Для РФ ежемесячный доход клиента попадает в эл. док. Уведомление о рассчитанном ПДН (реструктуризация)
    -- Поэтому обязателен к заполнению
    begin
      monthly_income = f_json_str_get_value(cml_add_params, 'monthly_income', '0');
    when any do
      monthly_income = 0;
    end
    if (monthly_income = 0) then
    begin
      res_is_error = 1;
      res_text = f_lang(lang, 'Неверно указан Доход клиента в месяц');
    end
    else
    begin
      select kc_from_c_id
      from knclient
      where kc_from_k_id = :obj_id and
            kc_amount_approved > 0
      into c_id;
      execute procedure hp_objectprops_set_with_rd(0, c_id, -1, 'work_monthly_income', cml_date, '', 0, f_get_rd_id('Results.StrValue'), un_id, monthly_income);
      -- Доп. доход обнуляем, т.к. в окне заявки на КК вводится только доход в месяц
      execute procedure hp_objectprops_set_with_rd(0, c_id, -1, 'work_extra_amount', cml_date, '', 0, f_get_rd_id('Results.StrValue'), un_id, 0);
    end
  end
/*
  if (res_text = '' and db_id = 81001) then
  begin
    -- Законом 353-ФЗ установлен ограничитель по соотношению суммы доходов (процентов и пеней)
    -- к сумме займа, при превышении которого МФО обязана прекратить начисление дохода
    -- Также проверяется превышение лимита по ПСК
    if (cmc_com_kind <> 'termination') then -- для рассторжение не проверяем ПСК
    begin
      select res_text
      from hp_check_psk_limit('com_list', :cml_id, :cml_date)
      into res_text;
    end
  end
*/

  if (res_text = '') then
  begin
    select res_text, res_is_error
    from hp_com_action_add(:cml_id, null, f_enum('com_action_status', 'sent', 'id'), :un_id)
    into res_text, res_is_error;

    if (res_is_error = 0 and cmc_auto_approve = 1) then
    begin
      if (db_id = 81001) then
      begin
        res_text = 'Автоподтверждение отключено, т.к. подписание ПДН в автоматическом режиме не реализовано.';
        res_is_error = 1;
      end
      else
      begin
        -- автоподтверждение
        select res_is_error, res_text
        from hp_com_list_auto_approve(:cml_id)
        into :res_is_error, :res_text;
      end
    end
  end
  suspend;
end
