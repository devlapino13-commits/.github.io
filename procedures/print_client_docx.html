CREATE PROCEDURE PRINT_CLIENT_DOCX(
    AAREP_K_ID ID_TYPE)
RETURNS (
    MIGR_CARD_NO VARCHAR(30),
    MIGR_CARD_BEGIN DATE,
    MIGR_CARD_END DATE,
    TYPE_DOCUMENT VARCHAR(100),
    DOC_SERIES VARCHAR(10),
    DOC_NO VARCHAR(10),
    DOC_DATE_BEGIN DATE,
    DOC_DATE_END DATE,
    LEGAL_COUNTRY VARCHAR(100),
    LEGAL_INDEX INTEGER,
    LEGAL_REGION VARCHAR(100),
    LEGAL_DISTRICT VARCHAR(100),
    LEGAL_CITY VARCHAR(100),
    LEGAL_LOCALITY VARCHAR(100),
    LEGAL_STREET VARCHAR(100),
    LEGAL_DEPART_NO D_STR_20,
    LEGAL_BUILDING_NO D_STR_20,
    LEGAL_SECTION_NO D_STR_20,
    LEGAL_FLAT_NO D_STR_20,
    ACTUAL_COUNTRY VARCHAR(100),
    ACTUAL_INDEX INTEGER,
    ACTUAL_REGION VARCHAR(100),
    ACTUAL_DISTRICT VARCHAR(100),
    ACTUAL_CITY VARCHAR(100),
    ACTUAL_LOCALITY VARCHAR(100),
    ACTUAL_STREET VARCHAR(100),
    ACTUAL_DEPART_NO D_STR_20,
    ACTUAL_BUILDING_NO D_STR_20,
    ACTUAL_SECTION_NO D_STR_20,
    ACTUAL_FLAT_NO D_STR_20,
    REG_YUR_NAME VARCHAR(100),
    REG_YUR_DATE_BEGIN DATE,
    REG_YUR_DATE_END DATE)
AS
declare variable pdt_id integer;
declare variable pdt_code varchar(100);
declare variable n smallint;
declare variable cds_doc_no varchar(30);
declare variable cds_doc_date_begin date;
declare variable cds_doc_date_end date;
declare variable rvp_date_begin date;
declare variable rvp_date_end date;
declare variable vnj_series varchar(30);
declare variable vnj_no varchar(30);
declare variable vnj_date_begin date;
declare variable vnj_date_end date;
declare variable pdt_name varchar(100);
declare variable rvp_name varchar(100);
declare variable vnj_name varchar(100);
declare variable cds_datetime timestamp;
declare variable adc_address_type smallint;
declare variable pl_id id_type;
declare variable country varchar(100);
declare variable adl_post_index integer;
declare variable parent id_type;
declare variable pl_name varchar(100);
declare variable plt_name varchar(255);
declare variable region varchar(100);
declare variable district varchar(100);
declare variable c_id decimal(18,0);
declare variable city varchar(100);
declare variable locality varchar(100);
declare variable street varchar(100);
declare variable adl_depart_no d_str_20;
declare variable adl_building_no d_str_20;
declare variable adl_section_no d_str_20;
declare variable adl_flat_no d_str_20;
declare variable adl_street_id id_type;
begin
  for select kc_from_c_id
      from knclient
      where kc_from_k_id = :aarep_k_id and
            kc_amount_approved > 0
      into c_id
  do
  begin
    migr_card_no = null;
    migr_card_begin = null;
    migr_card_end = null;
    rvp_date_begin = null;
    rvp_date_end = null;
    vnj_series = null;
    vnj_no = null;
    vnj_date_begin = null;
    vnj_date_end = null;
    rvp_name = null;
    vnj_name = null;
    country = null;
    legal_country = null;
    for select pdt_id, pdt_code, pdt_name
        from passdoctype
        where pdt_code in ('magration_card', 'registration_rvp', 'registration_vnj', 'registration_yur')
        into pdt_id, pdt_code, pdt_name
    do
    begin
      cds_doc_no = null;
      cds_doc_date_begin = null;
      cds_doc_date_end = null;
      select cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_datetime
      from clientdocs, passkind
      where cds_from_c_id = :c_id and
            cds_is_delete = 0 and
            coalesce(cds_doc_blob_1, 0) > 0 and
            cds_from_pk_id = pk_id and
            pk_from_pdt_id = :pdt_id and
            current_date between cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.2999')
      order by cds_datetime desc
      rows 1
      into cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_datetime;

      if (cds_datetime is not null) then
      begin
        if (pdt_code = 'magration_card') then
        begin
          migr_card_no = cds_doc_no;
          migr_card_begin = cds_doc_date_begin;
          migr_card_end = cds_doc_date_end;
        end
        else
        if (pdt_code = 'registration_rvp') then
        begin
          rvp_name = pdt_name;
          rvp_date_begin = cds_doc_date_begin;
          rvp_date_end = cds_doc_date_end;
        end
        else
        if (pdt_code = 'registration_vnj') then
        begin
          vnj_name = pdt_name;
          cds_doc_no = trim(cds_doc_no);
          if (cds_doc_no <> '') then
            n = char_length(cds_doc_no);
          if (n > 2) then
          begin
            vnj_series = left(cds_doc_no, 2);
            vnj_no = right(cds_doc_no, n - 2);
          end
          vnj_date_begin = cds_doc_date_begin;
          vnj_date_end = cds_doc_date_end;
        end
        else
        if (pdt_code = 'registration_yur') then
        begin
          reg_yur_name = pdt_name;
          reg_yur_date_begin = cds_doc_date_begin;
          reg_yur_date_end = cds_doc_date_end;
        end

      end
    end

    if (vnj_series is not null) then
    begin
      type_document = vnj_name;
      doc_series = vnj_series;
      doc_no = vnj_no;
      doc_date_begin = vnj_date_begin;
      doc_date_end = vnj_date_end;
    end
    else
    if (rvp_name is not null) then
    begin
      type_document = rvp_name;
      doc_date_begin = rvp_date_begin;
      doc_date_end = rvp_date_end;
    end

    legal_country = null;
    legal_index = null;
    legal_region = null;
    legal_district = null;
    legal_city = null;
    legal_locality = null;
    legal_street = null;
    legal_depart_no = null;
    legal_building_no = null;
    legal_section_no = null;
    legal_flat_no = null;

    actual_country = null;
    actual_index = null;
    actual_region = null;
    actual_district = null;
    actual_city = null;
    actual_locality = null;
    actual_street = null;
    actual_depart_no = null;
    actual_building_no = null;
    actual_section_no = null;
    actual_flat_no = null;

    adl_street_id = null;
    for select adc_address_type, adl_from_pl_id, adl_post_index, adl_depart_no, adl_building_no, adl_section_no,
               adl_flat_no, adl_street_id
        from addrclient, addrlist, places
        where adc_from_c_id = :c_id and
              adc_from_adl_id = adl_id and
              adc_address_type in (1, 2) and
              adl_from_pl_id = pl_id
        into adc_address_type, pl_id, adl_post_index, adl_depart_no, adl_building_no, adl_section_no, adl_flat_no,
             adl_street_id
    do
    begin
      if (adl_street_id is not null) then
        pl_id = adl_street_id;
      country = null;
      :region = null;
      district = null;
      city = null;
      locality = null;
      street = null;

      for with recursive tabl1
          as (select pl_id, parent, pl_name, pl_status
              from places
              where pl_id = :pl_id
              union all
              select pl.pl_id, pl.parent, pl.pl_name, pl.pl_status
              from places pl
              join tabl1 on (tabl1.parent = pl.pl_id))
          select parent, pl_name, plt_name
          from tabl1, placetype
          where pl_status = plt_id
          into parent, pl_name, plt_name
      do
      begin
        if (coalesce(parent, '') = '') then
          country = pl_name;
        if (plt_name = 'Область') then
          :region = pl_name;
        if (plt_name = 'Район') then
          district = pl_name;
        if (plt_name = 'Город') then
          city = pl_name;
        if (plt_name in ('Населенный пункт', 'Деревня')) then
          locality = pl_name;
        if (plt_name = 'Улица') then
          street = pl_name;
      end

      if (adc_address_type = 1) then
      begin
        legal_country = country;
        legal_index = adl_post_index;
        legal_region = :region;
        legal_district = district;
        legal_city = city;
        legal_locality = locality;
        legal_street = street;
        legal_depart_no = adl_depart_no;
        legal_building_no = adl_building_no;
        legal_section_no = adl_section_no;
        legal_flat_no = adl_flat_no;
      end
      else
      begin
        actual_country = country;
        actual_index = adl_post_index;
        actual_region = :region;
        actual_district = district;
        actual_city = city;
        actual_locality = locality;
        actual_street = street;
        actual_depart_no = adl_depart_no;
        actual_building_no = adl_building_no;
        actual_section_no = adl_section_no;
        actual_flat_no = adl_flat_no;
      end
    end

    suspend;
  end
end
