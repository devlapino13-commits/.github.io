CREATE PROCEDURE AAREP_FPANK_REP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    FPA_ID ID_TYPE,
    FPA_CLIENT_FIO STR_255,
    FPA_FROM_UN_ID ID_TYPE,
    FPA_CITY STR_255,
    FPA_CITIZEN STR_255,
    FPA_DP_BIRTH STR_255,
    FPA_PASS_KIND STR_255,
    FPA_PASS_PROPS STR_255,
    FPA_MCARD_PROPS STR_255,
    FPA_ACARD_PROPS STR_255,
    FPA_INN STR_255,
    FPA_PBUL_PROPS STR_255,
    FPA_ADDRESS_LEGAL STR_255,
    FPA_ADDRESS_FACT STR_255,
    FPA_REL_START DATE,
    FPA_RESP_FIO STR_255,
    FPA_CELL_NO STR_20,
    FPA_EMAIL STR_100,
    EXIST_IN_FP_LIST STR_255,
    SOLUTION_IN_FP STR_255,
    RISK_LEVEL STR_255,
    FPA_SNILS STR_100)
AS
begin
  for select kncl.kc_from_c_id
      from kontstate ks, kontract kont, knclient kncl, region_list(:rep_region_id, 1, -1, 1)
      where ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date_end between ks.ks_date_begin and ks.ks_date_end and
            kont.k_id = ks.ks_from_k_id and
            ks.ks_from_k_id = kncl.kc_from_k_id and
            id_list = kont.k_last_reg_id and
            kncl.kc_amount_approved > 0
      into :c_id
  do
  begin
    for select fpa_id, fpa_client_fio, fpa_from_un_id, fpa_city, fpa_citizen, fpa_dp_birth, fpa_pass_kind,
               fpa_pass_props, fpa_mcard_props, fpa_acard_props, fpa_inn, fpa_address_legal, fpa_rel_start,
               fpa_resp_fio, fpa_cell_no, fpa_email, exist_in_fp_list, solution_in_fp, risk_level, fpa_snils
        from print_fp_ank(:c_id)
        into :fpa_id, :fpa_client_fio, :fpa_from_un_id, :fpa_city, :fpa_citizen, :fpa_dp_birth, :fpa_pass_kind,
             :fpa_pass_props, :fpa_mcard_props, :fpa_acard_props, :fpa_inn, :fpa_address_legal, :fpa_rel_start,
             :fpa_resp_fio, :fpa_cell_no, :fpa_email, :exist_in_fp_list, :solution_in_fp, :risk_level, :fpa_snils
    do
    begin
      suspend;
    end
  end
end
