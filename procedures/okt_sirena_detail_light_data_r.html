CREATE PROCEDURE OKT_SIRENA_DETAIL_LIGHT_DATA_R(
    K_ID ID_TYPE,
    OBJECT_KIND D_STR_100,
    OBJECT_ID ID_TYPE,
    REP_DATE D_DATE,
    IS_INCOME D_BOOL)
RETURNS (
    HTML D_STR_8100)
AS
declare variable cl_fio d_str_100;
declare variable con_fio d_str_100;
declare variable status d_str_100;
declare variable c_id d_id;
declare variable zad_summ d_integer;
declare variable delinq_summ d_integer;
declare variable sh_date d_date;
declare variable zad_summ_temp d_integer;
declare variable delinq_days d_integer;
declare variable last_payed_date d_date;
declare variable last_payed_sum d_integer;
declare variable paysys_acc_no d_str_100;
declare variable k_kontract_no d_str_100;
declare variable svz_client_memo d_str_255;
declare variable svz_client_date_str d_str_100;
declare variable svz_kontract_memo d_str_255;
declare variable svz_kontract_memo_date_str d_str_100;
declare variable ch_project str_100;
declare variable ch_result str_100;
declare variable ch_memo varchar(1000);
declare variable ch_called_datetime_str d_str_100;
declare variable ch_pant_oper_called str_100;
declare variable person_txt d_str_255;
declare variable hist_qty d_integer;
declare variable c_birth_date d_date;
declare variable address_str d_str_255;
declare variable cc_cell_no d_str_100;
declare variable client_cell_no_add d_str_100;
declare variable k_disb_date_str d_str_100;
declare variable k_amount_approved d_integer;
declare variable region_id d_integer;
begin
  html = '<h4>' || iif(is_income = 0, 'Исходящий звонок', 'Входящий звонок') || '</h4>';
  suspend;

  if (object_kind = 'Clients') then
  begin
    select cl.c_name_fio,
           cl.c_birth_date
    from clients cl
    where cl.c_id = :object_id
    into :cl_fio,
         :c_birth_date;

    c_id = :object_id;
  end
  else
  begin
    select coalesce(con.cc_name, ''),
           coalesce(con.cc_status, ''),
           cl.c_name_fio,
           cl.c_id,
           cl.c_birth_date,
           con.cc_cell_no
    from contacts con
    inner join clients cl on cl.c_id = con.cc_from_c_id
    where con.cc_id = :object_id
    into :con_fio,
         :status,
         :cl_fio,
         :c_id,
         :c_birth_date,
         :cc_cell_no;
  end

  zad_summ = 0;
  delinq_summ = 0;

  for select coalesce(sh_date, '01.01.1900'),
             round(sum(zad_os + sh_calc_perc + sh_calc_fine + zad_kommis))
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      group by sh_date
      order by sh_date
      into :sh_date,
           :zad_summ_temp
  do
  begin
    if (sh_date < :rep_date) then
      delinq_summ = delinq_summ + zad_summ_temp;
    zad_summ = zad_summ + zad_summ_temp;
  end

  delinq_days = 0;

  select current_date - oks.oks_delinq_date,
         oks.oks_region_id
  from oksummary oks
  where oks.oks_from_k_id = :k_id and
        current_date between oks.oks_date_begin and oks.oks_date_end  and
        oks_close_state = 0
  into :delinq_days,
       :region_id;

  delinq_days = coalesce(delinq_days, 0);
  if (delinq_days < 0) then
    delinq_days = 0;

  select ok.ok_ol_date,
         round(-sum(ok.ok_os + ok.ok_perc + ok.ok_fine))
  from operkred ok
  where ok.ok_from_k_id = :k_id and
        ok.ok_type = 4 and
        ok.ok_ol_is_active = 1 and
        ok.ok_from_c_id = 0
  group by ok.ok_ol_date
  having sum(ok.ok_os + ok.ok_perc + ok.ok_fine) < 0
  order by ok.ok_ol_date desc
  rows 1
  into :last_payed_date, :last_payed_sum;

  last_payed_sum = coalesce(last_payed_sum, 0);
  paysys_acc_no = '';

  select uniq_acc_str
  from hp_show_clientuniq_acc_no(:c_id)
  into :paysys_acc_no; -- № Лицевого счета

  paysys_acc_no = coalesce(paysys_acc_no, '');

  select kont.k_kontract_no,
         f_datetostr(kont.k_disb_date),
         round(kont.k_amount_approved, 0)
  from kontract kont
  where kont.k_id = :k_id
  into :k_kontract_no,
       :k_disb_date_str,
       :k_amount_approved;

  if (object_kind = 'Clients') then
    person_txt = 'Клиент - ' || :cl_fio;
  else
  begin
    person_txt = 'Контактное лицо ('||:status||') - '|| :con_fio || '</br>';
    person_txt  = person_txt || 'ФИО клиента - '|| :cl_fio;
  end

  html = person_txt || '</br>';
  suspend;

  html = 'Дата рождения клиента - '|| f_datetostr(:c_birth_date) || '</br>';
  suspend;

  for select trim(coalesce(adl.adl_address_str, ''))
    from addrclient adcl
    inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
    where adcl.adc_from_c_id = :c_id and
          adcl.adc_address_type = 5
    into :address_str
  do
  begin
    html = 'Фактический адрес проживания клиента на родине - ' || address_str || '</br>';
    suspend;
  end

  html = 'Номер договора - ' || :k_kontract_no ||' </br>';
  suspend;

  html = 'Договор закреплен за офисом - ' || (select reg.reg_name from region reg where reg.reg_unicode = :region_id) ||' </br>';
  suspend;

  html = 'Дата выдачи займа - ' || :k_disb_date_str ||' </br>';
  suspend;

  html = 'Сумма кредита - ' || :k_amount_approved ||' рублей </br>';
  suspend;

  html = 'Просроченная сумма - ' || :delinq_summ || ' рублей </br>';
  suspend;

  html =  'Дней просрочки - ' || :delinq_days || ' дней' || '</br>';
  suspend;

  html = 'Последний раз платил ' || f_datetostr(last_payed_date) || ', сумма ' || :last_payed_sum || ' рублей </br>';
  suspend;

  html = 'Осталось погасить - ' || :zad_summ|| ' рублей </br>';
  suspend;

  html = 'Лицевой счет - ' || :paysys_acc_no|| ' </br>';
  suspend;

  -----
  if (object_kind = 'Contacts') then
  begin
    html = 'Телефонный номер контактного лица - ' || cc_cell_no || '</br>';
    suspend;
  end


  html = 'Основной телефонный номер клиента - ' || f_cont_info('clients', c_id, 'cell_main', 'last') || '</br>';
  suspend;

  client_cell_no_add = coalesce(f_cont_info('clients', object_id, 'cell_add', 'last'), '');
  client_cell_no_add = iif(client_cell_no_add = '', 'Отсутствует', client_cell_no_add);

  html = 'Дополнительный телефонный номер клиента - ' || client_cell_no_add || '</br>';
  suspend;
  ----------------- примечание специалиста (ВИПС-а) по клиенту -- >
  svz_client_memo = '';
  svz_client_date_str = '';

  select obm_memo, f_datetostr(obm_datetime)
  from hp_objectprops_memo_get('Clients', :c_id, 'Refdata', 'ObjectProps.Client_memo', 1)
  order by obm_datetime desc
  rows 1
  into :svz_client_memo, :svz_client_date_str;

  svz_client_memo = coalesce(svz_client_memo, '');
  svz_client_date_str = coalesce(svz_client_date_str, '');

  html = 'Примечание ВИПC по клиенту ' || iif(svz_client_date_str = '', ': ', 'на дату ' || svz_client_date_str || ': ')||iif(svz_client_memo = '', 'Отсутствует', svz_client_memo) || '</br>';
  suspend;
  --------------------------------------------------------------- <
  ----------------- примечание специалиста (ВИПС-а) по кредиту -- >
  svz_kontract_memo = '';
  svz_kontract_memo_date_str = '';

  select obm_memo, f_datetostr(obm_datetime)
  from hp_objectprops_memo_get('Kontract', :k_id, 'Refdata', 'ObjectProps.Kontract_memo', 1)
  order by obm_datetime desc
  rows 1
  into :svz_kontract_memo, :svz_kontract_memo_date_str;

  svz_kontract_memo = coalesce(svz_kontract_memo, '');
  svz_kontract_memo_date_str = coalesce(svz_kontract_memo_date_str, '');

  html = 'Примечание ВИПC по контракту ' || iif(svz_kontract_memo_date_str = '', ': ', 'на дату ' || svz_kontract_memo_date_str||': ') || iif(svz_kontract_memo = '', 'Отсутствует', svz_kontract_memo) || '</br>';
  suspend;
  -----------------------------------------------------------------<

  ----------------------------------------------------------------->
  if (object_kind = 'Clients') then
  begin
      html = 'ИСТОРИЯ ЗВОНКОВ КЛИЕНТУ (3 последних звонка) ' || '</br>';
      suspend;

      hist_qty = 0;
      for select rd.rd_name as project,
                 rd_res.rd_name as result,
                 pantc.pcl_memo,
                 f_datetimetostr(pantc.pcl_called_datetime),
                 iif(un.un_name is null, pantk.pant_oper_called, un.un_name)
            from pantera_kontracts pantk
            inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
            inner join pantera_clients pantc on pantc.pcl_from_pan_id = pantk.pant_id
            inner join refdata rd_res on rd_res.rd_id = pantc.pcl_rd_res
            left join pantera_clients_opers pco on pco.pco_from_pcl_id = pantc.pcl_id
            left join username un on un.un_id = pco.pco_from_un_id
            where pantk.pant_from_k_id = :k_id and
                  pantc.pcl_from_c_id = :c_id
            order by pcl_called_datetime desc
            --rows 3
            into :ch_project, :ch_result, ch_memo, ch_called_datetime_str, ch_pant_oper_called
      do
      begin
          if ((ch_pant_oper_called = 'FINSTART_ROBOT') or (hist_qty = 3)) then
            continue;

          hist_qty = hist_qty + 1;
          html = '';
          html = html || '' || ascii_char(10) || ascii_char(13);
          html = html || '№. ' || cast(hist_qty as str_10) || ascii_char(10);
          html = html || 'Проект: ' || coalesce(ch_project, '') || ascii_char(10);
          html = html || 'Оператор: ' || coalesce(ch_pant_oper_called, '') || ascii_char(10);
          html = html || 'Результат: ' || coalesce(ch_result,'') || ascii_char(10);
          html = html || 'Примечание: ' || coalesce(ch_memo, '') || ascii_char(10);
          html = html || 'Время звонка: ' || coalesce(ch_called_datetime_str, '') || '</br>';
          suspend;
      end
    
      if (hist_qty = 0) then
      begin
        html = 'Звонков не было' || '</br>';
        suspend;
      end
  end
  else if (object_kind = 'Contacts') then
  begin
      html = 'ИСТОРИЯ ЗВОНКОВ КОНТАКТНОМУ ЛИЦУ(3 последних звонка) ' || '</br>';
      hist_qty = 0;

      for select rd.rd_name as project,
                 rd_res.rd_name as result,
                 pantcont.pcon_memo,
                 f_datetimetostr(pantcont.pcon_called_datetime),
                 iif(un.un_name is null, pantk.pant_oper_called, un.un_name)
      from pantera_kontracts pantk
      inner join refdata rd on rd.rd_id = pantk.pant_rd_project_type
      inner join pantera_contacts pantcont on pantcont.pcon_from_pan_id = pantk.pant_id
      inner join refdata rd_res on rd_res.rd_id = pantcont.pcon_rd_res
      left join pantera_contacts_opers pcco on pcco.pco_from_pcon_id = pantcont.pcon_id
      left join username un on un.un_id = pcco.pco_from_un_id
      where pantk.pant_from_k_id = :k_id and
            pantcont.pcon_from_cc_id = :object_id
      order by pcon_called_datetime desc
      --rows 3
      into :ch_project, :ch_result, ch_memo, ch_called_datetime_str, ch_pant_oper_called
      do
      begin
          if ((ch_pant_oper_called = 'FINSTART_ROBOT') or (hist_qty = 3)) then
            continue;

          hist_qty = hist_qty + 1;
          html = '';
          html = html || '№. ' || cast(hist_qty as str_10) || '. ';
          html = html || 'Проект: ' || coalesce(ch_project, '') || '. ';
          html = html || 'Оператор: ' || coalesce(ch_pant_oper_called, '') || '. ';
          html = html || 'Результат: ' || coalesce(ch_result,'') || '. ';
          html = html || 'Примечание: ' || iif(coalesce(ch_memo, '') = '', 'Отсутствует', ch_memo) || '; ';
          html = html || 'Время звонка: ' || coalesce(ch_called_datetime_str, '')  || '</br>';
          suspend;
      end
    
      if (hist_qty = 0) then
      begin
        html = 'Звонков не было' || '</br>';
        suspend;
      end
  end
end
