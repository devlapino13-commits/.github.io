CREATE PROCEDURE HP_GET_KONTRACT_NSP_ACC(
    REG_ID INTEGER,
    REP_DATE DATE,
    ITS_IPF STR_10 NOT NULL = 'no')
RETURNS (
    NSP_PERCENT_FOR_INT DECIMAL(18,7),
    NSP_PERCENT_FOR_EXTRA_INT DECIMAL(18,7),
    NSP_ACC_INT_MAIN VARCHAR(20),
    NSP_ACC_INT_EXPENSE VARCHAR(20),
    NSP_ACC_INT_REPAY VARCHAR(20),
    NSP_ACC_EXTRA_INT_MAIN VARCHAR(20),
    NSP_ACC_EXTRA_INT_EXPENSE VARCHAR(20),
    NSP_ACC_EXTRA_INT_REPAY VARCHAR(20))
AS
declare variable nsd_acc_main_balance varchar(20);
declare variable nsd_acc_main_no integer;
declare variable nsd_acc_expense_balance varchar(20);
declare variable nsd_acc_expense_no integer;
declare variable def_curr varchar(3);
declare variable reg_code varchar(20);
declare variable nsd_acc_repay_balance varchar(20);
declare variable nsd_acc_repay_no integer;
begin
  nsp_percent_for_int = 0;
  nsp_percent_for_extra_int = 0;
  nsp_acc_int_main = '';
  nsp_acc_int_expense = '';
  nsp_acc_int_repay = '';

  nsp_acc_extra_int_main = '';
  nsp_acc_extra_int_expense = '';
  nsp_acc_extra_int_repay = '';

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   :rep_date between nsd_date_begin and nsd_date_end)) then
  begin
    select nsd_percent, nsd_acc_main_balance, nsd_acc_main_no, nsd_acc_expense_balance, nsd_acc_expense_no,
           nsd_acc_repay_balance, iif(:its_ipf = 'yes', nsd_acc_repay_ipf_no, nsd_acc_repay_no)
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процентного дохода
          and
          :rep_date between nsd_date_begin and nsd_date_end
    into :nsp_percent_for_int, :nsd_acc_main_balance, :nsd_acc_main_no, :nsd_acc_expense_balance, :nsd_acc_expense_no,
         nsd_acc_repay_balance, nsd_acc_repay_no;

    if (nsp_percent_for_int is null) then
      nsp_percent_for_int = 0;
    if (nsp_percent_for_int > 0) then
    begin
      select cm_default_currency
      from company
      into :def_curr;
      select reg_code
      from region
      where reg_unicode = :reg_id
      into :reg_code;
      nsp_acc_int_main = :reg_code || :def_curr || nsd_acc_main_balance || right('000000' || nsd_acc_main_no, 6);
      nsp_acc_int_expense = :reg_code || :def_curr || nsd_acc_expense_balance || right('000000' || nsd_acc_expense_no, 6);
      nsp_acc_int_repay = :reg_code || :def_curr || nsd_acc_repay_balance || right('000000' || nsd_acc_repay_no, 6);
    end

    select nsd_percent, nsd_acc_main_balance, nsd_acc_main_no, nsd_acc_expense_balance, nsd_acc_expense_no,
           nsd_acc_repay_balance, nsd_acc_repay_no
    from nspconfig_data
    where nsd_from_nsm_id = -2 -- нсп от доп.процентного дохода
          and
          nsd_date_begin <= :rep_date and
          nsd_date_end >= :rep_date
    into :nsp_percent_for_extra_int, :nsd_acc_main_balance, :nsd_acc_main_no, :nsd_acc_expense_balance,
         :nsd_acc_expense_no, nsd_acc_repay_balance, nsd_acc_repay_no;

    if (nsp_percent_for_extra_int is null) then
      nsp_percent_for_extra_int = 0;
    if (nsp_percent_for_extra_int > 0) then
    begin
      select cm_default_currency
      from company
      into :def_curr;
      select reg_code
      from region
      where reg_unicode = :reg_id
      into :reg_code;
      nsp_acc_extra_int_main = :reg_code || :def_curr || nsd_acc_main_balance || right('000000' || nsd_acc_main_no, 6);
      nsp_acc_extra_int_expense = :reg_code || :def_curr || nsd_acc_expense_balance || right('000000' || nsd_acc_expense_no, 6);
      nsp_acc_extra_int_repay = :reg_code || :def_curr || nsd_acc_repay_balance || right('000000' || nsd_acc_repay_no, 6);
    end
  end

  suspend;
end
