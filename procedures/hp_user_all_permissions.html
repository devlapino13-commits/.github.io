CREATE PROCEDURE HP_USER_ALL_PERMISSIONS(
    UN_ID DECIMAL(18,0))
RETURNS (
    PM_ID TYPE OF COLUMN PERMISSIONS.PM_ID,
    PM_SUBJECT_KIND TYPE OF COLUMN PERMISSIONS.PM_SUBJECT_KIND,
    PM_SUBJECT_ID TYPE OF COLUMN PERMISSIONS.PM_SUBJECT_ID,
    PM_DATE_BEGIN TYPE OF COLUMN PERMISSIONS.PM_DATE_BEGIN,
    PM_DATE_END TYPE OF COLUMN PERMISSIONS.PM_DATE_END,
    PM_ACCESS_KIND TYPE OF COLUMN PERMISSIONS.PM_ACCESS_KIND,
    PM_IS_DELETE TYPE OF COLUMN PERMISSIONS.PM_IS_DELETE,
    PM_FROM_UN_ID TYPE OF COLUMN PERMISSIONS.PM_FROM_UN_ID,
    PM_DATETIME TYPE OF COLUMN PERMISSIONS.PM_DATETIME,
    PM_DELETE_UN_ID TYPE OF COLUMN PERMISSIONS.PM_DELETE_UN_ID,
    PM_DELETE_DATETIME TYPE OF COLUMN PERMISSIONS.PM_DELETE_DATETIME,
    REP_SUBJECT_NAME VARCHAR(255),
    REP_SUBJECT_MEMO VARCHAR(255),
    REP_IS_ACTIVE INTEGER)
AS
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
declare variable uniqtab_menulist integer;
declare variable uniqtab_permactions integer;
declare variable uniqtab_region integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;
  select ukt_id
  from hp_get_uniqtable_id('PermActions')
  into :uniqtab_permactions;
  select ukt_id
  from hp_get_uniqtable_id('Region')
  into :uniqtab_region;

  for select pm_id, pm_subject_kind, pm_subject_id, pm_date_begin, pm_date_end, pm_access_kind, pm_is_delete,
             pm_from_un_id, pm_datetime, pm_delete_un_id, pm_delete_datetime
      from permissions
      where pm_object_kind = :uniqtab_username and
            pm_object_id = :un_id
      order by pm_subject_kind, pm_id
      into :pm_id, :pm_subject_kind, :pm_subject_id, :pm_date_begin, :pm_date_end, :pm_access_kind, :pm_is_delete,
           :pm_from_un_id, :pm_datetime, :pm_delete_un_id, :pm_delete_datetime
  do
  begin
    rep_subject_name = '';
    rep_subject_memo = '';
    rep_is_active = (1 - pm_is_delete);
    if (pm_is_delete = 0) then
    begin
      if (pm_access_kind = 1) then
      begin
        if (not((coalesce(pm_date_begin, '01.01.2000') <= current_date) and
            (coalesce(pm_date_end, '31.12.2999') >= current_date))) then
        begin
          rep_is_active = 0;
          pm_is_delete = 1;
        end
      end
      if (pm_access_kind = -1) then
      begin
        if ((pm_date_begin is null or pm_date_end is null) and    ----<
            (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
            (coalesce(pm_date_end, '31.12.2999') >= current_date)) then
        begin
          rep_is_active = 0;
          pm_is_delete = 1;
        end

        ----------------------

        if (current_date between pm_date_begin and pm_date_end) then
        begin
          rep_is_active = 1;
          pm_is_delete = 0;
        end
        ----------------
      end
    end
    if (pm_subject_kind = :uniqtab_rolename) then
    begin
      select rn_name, rn_memo
      from rolename
      where rn_id = :pm_subject_id
      into :rep_subject_name, :rep_subject_memo;
    end
    if (pm_subject_kind = :uniqtab_menulist) then
    begin
      select ml_caption, ml_description
      from menulist
      where ml_id = :pm_subject_id
      into :rep_subject_name, :rep_subject_memo;
    end
    if (pm_subject_kind = :uniqtab_permactions) then
    begin
      select pa_name, pa_memo
      from permactions
      where pa_id = :pm_subject_id
      into :rep_subject_name, :rep_subject_memo;
    end
    if (pm_subject_kind = :uniqtab_region) then
    begin
      select reg_code || ' ' || reg_name, reg_address
      from region
      where reg_unicode = :pm_subject_id
      into :rep_subject_name, :rep_subject_memo;
    end
    suspend;
  end
end
