CREATE PROCEDURE PRINT_RESCHEDULE_KZ_EXT(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    SH_DATE_BEFORE D_DATE,
    DELINQ_START_DATE D_DATE,
    WRITEOFF_AMOUNT CURR_TYPE,
    MONTH_ADD D_INTEGER,
    WRITEOFF_DATE D_DATE,
    EDL_ID D_ID,
    K_PERIOD D_INTEGER,
    K_EFFECT_RATE D_PERCENT)
AS
declare variable BLS_ID D_ID;
declare variable JSON_TEXT D_BLOB_TEXT;
begin
  edl_id = iif(add_params starting '{', f_json_str_get_value(add_params, 'edl_id', null), add_params);

  select edl.edl_blank_date
  from edoc_list edl
  where edl.edl_id = :edl_id
  into writeoff_date;

  for select eil.eil_from_bls_id
      from extinfolist eil
      inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
            eic.eic_code = 'restruct_schedule'
      where eil.eil_status = 2 and
            eil.eil_object_kind = f_objkind('kontract', 'id') and
            eil.eil_object_id = :k_id and
            eil.eil_is_json = 1 and
            eil.eil_from_bls_id > 0
      order by eil.eil_id desc rows 1
      into bls_id
  do
  begin

    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into json_text;

    sh_date_before = f_json_str_get_value(:json_text, 'schedule[0].sh_date', '');
    delinq_start_date = f_json_str_get_value(:json_text, 'delinq_start_date', '');
    writeoff_amount = f_json_str_get_value(:json_text, 'fine', '');
    month_add = f_json_str_get_value(:json_text, 'month_add', '');
    k_period = f_json_str_get_value(:json_text, 'k_period', '');
    k_effect_rate = f_json_str_get_value(:json_text, 'k_effect_rate', 0);
    delinq_start_date = delinq_start_date + 1;
    suspend;
  end
end
