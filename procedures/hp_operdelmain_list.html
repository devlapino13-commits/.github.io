CREATE PROCEDURE HP_OPERDELMAIN_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_K_ID D_ID)
RETURNS (
    ODM_ID D_INTEGER,
    ODM_APP_DATE D_DATE,
    ODM_OPER_DATE D_DATE,
    ODM_FROM_REG_ID D_INTEGER,
    ODM_MAIN_OL_ID D_ID,
    ODM_PAYMENT_KIND D_SMALLINT,
    ODM_NEED_REVERSE D_BOOL,
    ODM_REASON_ID D_SMALLINT,
    ODM_ACTION_KIND D_SMALLINT,
    ODM_STATE D_SMALLINT,
    ODM_FROM_EDL_ID D_ID,
    ODM_MEMO D_STR_LONG,
    ODM_CREATE_UN_ID D_ID,
    ODM_CREATE_DATETIME D_DATETIME,
    ODM_STATE_UN_ID D_ID,
    ODM_STATE_DATETIME D_DATETIME,
    RES_REGION_NAME D_STR_100,
    RES_MAIN_OL_DATE D_DATE,
    RES_MAIN_OL_MEMO D_STR_LONG,
    RES_CREATE_USER D_STR_CODE,
    RES_STATE_USER D_STR_CODE,
    RES_PAYMENT_KIND_STR D_STR_CODE,
    RES_REASON_ID_STR D_STR_CODE,
    RES_ACTION_KIND_STR D_STR_CODE,
    RES_MAIN_KONTRACTS D_STR_LONG,
    RES_NEW_KONTRACTS D_STR_LONG,
    RES_MAIN_AMOUNT D_CURRENCY,
    RES_NEW_AMOUNT D_CURRENCY,
    RES_STATUS D_STR_CODE)
AS
begin
  if (rep_region_id is null and rep_k_id is null) then
    exit;

  if (rep_date_begin is null) then
    rep_date_begin = current_date;

  if (rep_date_end is null) then
    rep_date_end = current_date;

  for select res_odm_id
  from hp_operdelmain_list_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_k_id)
  into odm_id
  do
  begin
    for select odm_app_date,
               odm_oper_date,
               odm_from_reg_id,
               odm_main_ol_id,
               odm_payment_kind,
               odm_need_reverse,
               odm_reason_id,
               odm_action_kind,
               odm_state,
               odm_from_edl_id,
               odm_memo,
               odm_create_un_id,
               odm_create_datetime,
               odm_state_un_id,
               odm_state_datetime,
               reg_code || ' ' || reg_name,
               ol_date,
               ol_memo,
               cr.un_name,
               st.un_name
        from operdelmain
             inner join region on (odm_from_reg_id = reg_unicode)
             inner join operlist on (odm_main_ol_id = ol_id)
             inner join username cr on (odm_create_un_id = cr.un_id)
             left join username st on (odm_create_un_id = st.un_id)
        where odm_id = :odm_id
        into :odm_app_date,
             :odm_oper_date,
             :odm_from_reg_id,
             :odm_main_ol_id,
             :odm_payment_kind,
             :odm_need_reverse,
             :odm_reason_id,
             :odm_action_kind,
             :odm_state,
             :odm_from_edl_id,
             :odm_memo,
             :odm_create_un_id,
             :odm_create_datetime,
             :odm_state_un_id,
             :odm_state_datetime,
             :res_region_name,
             :res_main_ol_date,
             :res_main_ol_memo,
             :res_create_user,
             :res_state_user
    do
    begin
      res_payment_kind_str = trim(case odm_payment_kind
        when 1 then 'Касса'
        when 2 then 'Банк'
        when 3 then 'Онлайн'
        else '-'
      end);

      res_status = trim(case odm_state
        when -2 then 'Отказана'
        when -1 then 'Отменена'
        when 0 then 'Новая'
        when 1 then 'Проверена'
        when 2 then 'Готов'
      end);

      res_reason_id_str = f_enum('OperDelReason', odm_reason_id, 'name', 0);
      res_action_kind_str = f_enum('OperDelActionKind', odm_action_kind, 'name', 0);

      if (coalesce(res_reason_id_str, '') = '') then
        res_reason_id_str = '-';

      if (coalesce(res_action_kind_str, '') = '') then
        res_action_kind_str = '-';

      res_main_kontracts = '';
      res_new_kontracts = '';
      res_main_amount = 0;
      res_new_amount = 0;

      select list(k_kontract_no), sum(odk_amount)
      from operdelkont, kontract
      where odk_from_odm_id = :odm_id and
            odk_loan_ol_id is not null and
            odk_from_k_id = k_id
      into res_main_kontracts, res_main_amount;

      if (res_main_kontracts is null) then
        res_main_kontracts = '';

      if (res_main_amount is null) then
        res_main_amount = 0;

      select list(k_kontract_no), sum(abs(odk_amount))
      from operdelkont, kontract
      where odk_from_odm_id = :odm_id and
            odk_loan_ol_id is null and
            odk_from_k_id = k_id
      into res_new_kontracts, res_new_amount;

      if (res_new_kontracts is null) then
        res_new_kontracts = '';

      if (res_new_amount is null) then
        res_new_amount = 0;

      suspend;
    end
  end
end
