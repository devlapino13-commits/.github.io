CREATE PROCEDURE HP_FIND_PLACES_TMP(
    PARENT_ID DECIMAL(18,0),
    PLACE_STR VARCHAR(100),
    STREET_STR VARCHAR(100))
RETURNS (
    ID_LIST DECIMAL(18,0))
AS
declare variable id decimal(18,0);
declare variable need_find integer;
declare variable place_str_orig varchar(100);
declare variable street_str_orig varchar(100);
declare variable txt1 varchar(255);
declare variable txt2 varchar(255);
declare variable txt3 varchar(255);
declare variable txt4 varchar(255);
declare variable tmps varchar(255);
declare variable n integer;
declare variable word_count integer;
declare variable find_code_1 varchar(100);
declare variable find_code_2 varchar(100);
declare variable find_code_3 varchar(100);
declare variable find_code_4 varchar(100);
declare variable pl_name varchar(100);
declare variable pl_name_komitet varchar(100);
begin
-- Ищет населенный пункты и улицы - зависит от входящего параметра
  if (place_str is null) then
    place_str = '';
  if (street_str is null) then
    street_str = '';
  if (parent_id = 0) then
    parent_id = null;

-- если параметры поиска не указаны (первоначальный вызов процедуры)
  if (parent_id is null and
      place_str = '' and
      street_str = '') then
    exit;

  if (place_str = '' and
      street_str = '') then -- если нет критериев поиска
  begin
-- Если показываем все дочерние элементы
    for select pl_id
        from places
        where parent = :parent_id
        into id_list
    do
      suspend;
    exit;
  end
-- Если процедура дошла в это место, значит либо не страна либо есть критерии поиска
  place_str_orig = place_str;
  place_str = (select res_text
               from hp_txt_clear(:place_str));

  street_str_orig = street_str;
  street_str = (select res_text
                from hp_txt_clear(:street_str));
  if (street_str <> '') then
  begin
    txt1 = '';
    txt2 = '';
    txt3 = '';
    txt4 = '';

    tmps = street_str;
    n = position(' ', tmps);
    if (n = 0) then
    begin
      txt1 = tmps;
      word_count = 1;
    end
    else
    begin
      while (tmps <> '') do
      begin
        n = position(' ', tmps);
        if (n > 0) then
        begin
          word_count = word_count + 1;
          if (word_count = 1) then
            txt1 = left(tmps, n - 1);
          if (word_count = 2) then
            txt2 = left(tmps, n - 1);
          if (word_count = 3) then
            txt3 = left(tmps, n - 1);
          if (word_count = 4) then
            txt4 = left(tmps, n - 1);
          tmps = substring(tmps from n + 1);
          if (word_count = 4 and
              tmps <> '') then
            tmps = '';
        end
        else
        begin
          word_count = word_count + 1;
          if (word_count = 1) then
            txt1 = tmps;
          if (word_count = 2) then
            txt2 = tmps;
          if (word_count = 3) then
            txt3 = tmps;
          if (word_count = 4) then
            txt4 = tmps;
          tmps = '';
        end
      end
    end
    find_code_1 = f_metaphone(:txt1);
    find_code_2 = f_metaphone(:txt2);
    find_code_3 = f_metaphone(:txt3);
    find_code_4 = f_metaphone(:txt4);
  end

  if (char_length(find_code_1) = 1 or char_length(find_code_2) = 1 or char_length(find_code_3) = 1 or char_length(find_code_4) = 1) then
    exit;

  if (find_code_1 <> 'Ош' and
      char_length(find_code_1) = 2) then
    exit;
  if (find_code_2 <> 'Ош' and
      char_length(find_code_2) = 2) then
    exit;
  if (find_code_3 <> 'Ош' and
      char_length(find_code_3) = 2) then
    exit;
  if (find_code_4 <> 'Ош' and
      char_length(find_code_4) = 2) then
    exit;

  if (place_str <> '' and
      (street_str <> '' or street_str_orig <> '')) then
  begin
    for select pl_id
        from places, placetype
        where (pl_name starting :place_str_orig or
              pl_name_komitet starting :place_str_orig) and
              plt_id = pl_status and
              plt_can_select in (0, 1)
        union
        select object_id
        from hp_find_from_placelist(0, :place_str, '', '', '', ''), places, placetype
        where object_id = pl_id and
              plt_id = pl_status and
              plt_can_select in (0, 1)
        into :id
    do
    begin
      need_find = 0;
      if (parent_id is null) then
        need_find = 1;
      else
      if (exists(select *
                 from places_list(:id, 1, -1, 0)
                 where id_list = :parent_id)) then
        need_find = 1;

      if (street_str = '') then
      begin
        id_list = id;
        suspend;
      end
      else
      if (need_find = 1) then
      begin
        for with recursive pl_list
            as (select pl_id, parent, pl_status, pl_name, pl_name_komitet, plt_can_select
                from places, placetype
                where pl_id = :id and
                      is_delete = 0 and
                      plt_id = pl_status
                union all
                select p.pl_id, p.parent, p.pl_status, p.pl_name, p.pl_name_komitet, pl.plt_can_select
                from placetype pl, places p
                join pl_list on (pl_list.pl_id = p.parent) -- все дочерние
                where p.is_delete = 0 and
                      p.pl_status = pl.plt_id)
            select pl_id, pl_name, pl_name_komitet
            from pl_list
            where plt_can_select = 2
            into :id_list, pl_name, pl_name_komitet
        do
        begin
          if (pl_name starting street_str_orig or pl_name_komitet starting street_str_orig) then
            suspend;
          else
          begin
            if (exists(select *
                       from hp_find_from_placelist(:id_list, :street_str, :find_code_1, :find_code_2, :find_code_3, :find_code_4))) then
              suspend;
          end
        end
      end
    end
  end

  if (place_str <> '' and
      street_str = '' and
      street_str_orig = '') then
  begin
    for select pl_id
        from places, placetype
        where (pl_name starting :place_str_orig or
              pl_name_komitet starting :place_str_orig) and
              plt_id = pl_status and
              plt_can_select in (0, 1)
        union
        select object_id
        from hp_find_from_placelist(0, :place_str, '', '', '', ''), places, placetype
        where object_id = pl_id and
              plt_id = pl_status and
              plt_can_select in (0, 1)
        into :id
    do
    begin
      need_find = 0;
      if (parent_id is null) then
        need_find = 1;
      else
      if (exists(select *
                 from places_list(:id, 1, -1, 0)
                 where id_list = :parent_id)) then
        need_find = 1;

      if (need_find = 1) then
      begin
        id_list = id;
        suspend;
      end
    end
  end

  if (place_str = '' and
      (street_str <> '' or street_str_orig <> '')) then
  begin
    for with recursive pl_list
        as (select pl_id, parent, pl_status, pl_name, pl_name_komitet, plt_can_select, 2 as ordr
            from places, placetype
            where pl_id = :parent_id and
                  is_delete = 0 and
                  plt_id = pl_status
            union all
            select p.pl_id, p.parent, p.pl_status, p.pl_name, p.pl_name_komitet, pl.plt_can_select,
                   iif(p.parent = :parent_id, 0, 1)
            from placetype pl, places p
            join pl_list on (pl_list.pl_id = p.parent) -- все дочерние
            where p.is_delete = 0 and
                  p.pl_status = pl.plt_id)
        select pl_id, pl_name, pl_name_komitet
        from pl_list
        where plt_can_select = 2
        order by ordr
        into :id_list, pl_name, pl_name_komitet
    do
    begin
      if (pl_name starting street_str_orig or pl_name_komitet starting street_str_orig) then
        suspend;
      else
      begin
        if (exists(select *
                   from hp_find_from_placelist(:id_list, :street_str, :find_code_1, :find_code_2, :find_code_3, :find_code_4))) then
          suspend;
      end
    end
  end

end
