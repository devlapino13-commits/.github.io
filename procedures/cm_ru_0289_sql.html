CREATE PROCEDURE CM_RU_0289_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable ma_date_app d_date;
declare variable is_new_contdata d_smallint;
declare variable rd_id d_integer;
declare variable cont_kind_code d_str_code;
declare variable cont_kind_id d_integer;
declare variable has_verified d_smallint;
begin
  -- Пропускаем контракты заявки которых созданы до 17го марта включительно
  if (exists(select 1
             from miniapp ma
             join kontract k on k.k_from_ma_app_id = ma.ma_id
             where k.k_id = :k_id
               and ma.ma_date_app <= '17.03.2026'
               and ma.ma_is_active = 1
               and ma.ma_date_end = '31.12.9999')) then exit;

  -- Получаем ID клиента по контракту
  select kc_from_c_id
  from knclient
  where kc_from_k_id = :k_id
    and kc_amount_approved > 0
  into :c_id;

  -- Получаем информацию о типе контакта 'email'
  select rd_id, cont_kind_code, cont_kind_id
  from hp_contkind_info('email')
  into :rd_id, :cont_kind_code, :cont_kind_id;

  -- Определяем режим работы с контактными данными
  is_new_contdata = f_get_const('new_contdata');

  -- Проверяем наличие верифицированной почты
  has_verified = 0;
  if (is_new_contdata = 0) then -- старый способ
  begin
    select count(*)
    from contverify
    where cv_object_kind = f_objkind('clients', 'id') and
          cv_object_id = :c_id and
          cv_from_rd_id = :rd_id and
          cv_status > 0
    into :has_verified;
  end
  else -- новый способ
  begin
    select count(*)
    from contdata
    where cda_from_c_id = :c_id and
          cda_cont_kind = :cont_kind_id and
          cda_status = 1 and
          cda_is_delete = 0
    into :has_verified;
  end

  if (has_verified = 0) then
  begin
    res_error_is_critical = 2;
    res_error_text = 'Клиент не подтвердил свой e-mail';
    suspend;
  end
end
