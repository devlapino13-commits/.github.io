CREATE PROCEDURE AAREP_INSURANCE_IPF_PAYMENTS_REPORT(
    REP_REG_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_64,
    KAL_NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_255,
    K_FACT_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_KREDLINE_END D_DATE,
    INSURANCE_COMP_NAME D_STR_100,
    POLICY_NO D_STR_64,
    COMMISSION D_CURRENCY,
    NSP D_CURRENCY,
    MBULAK_SHARE D_CURRENCY,
    PAYED_AMOUNT D_CURRENCY,
    K_FACT_CLOSED_DATE D_DATE,
    STATUS D_STR_64,
    INSURER_SETTLEMENT_AMOUNT D_CURRENCY,
    INSURANCE_SUM_DUE D_CURRENCY)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable ag_id d_id;
declare variable reg_unicode d_integer;
declare variable ik_commission d_currency;
begin
    for select distinct
        t.id_list,
        kal.kal_new_kf_name,
        k.k_kontract_no,
        k.k_fact_disb_date,
        k.k_amount_approved,
        k.k_kredline_end,
        ik.ik_policy_no,
        ik.ik_commission,
        ik.ik_commission / 102 * 100,
        ik.ik_tax,
        ik.ik_mbulak_share,
        ik.ik_from_ag_id,
        k.k_fact_closed_date,
        iif(k.k_fact_closed_date is null, 'активный', 'закрыт') as status,
        kc.kc_from_c_id,
        k.k_id
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
    inner join insurance_kontract ik on ik.ik_from_k_id = ks.ks_from_k_id
        and ((ik.ik_enum_status = f_enum_id('insurance_kontract', 'insurance_policy_obtained')
             or ik.ik_enum_status = f_enum_id('insurance_kontract', 'insurance_policy_app_was_sent')) or
             ik.ik_from_ag_id = 7570)
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join product pr on k.k_from_pr_id = pr.pr_id
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
        and kal.is_delete = 0
        and :rep_date_end between kal.kal_date and kal.kal_date_end
    inner join region_list(:rep_reg_id, 1, 3, 1) t on t.id_list = ks.ks_region_id
    where ks.ks_ol_is_delete = 0
        and ks.ks_state = 1
        and ks.ks_date_begin between :rep_date_begin and :rep_date_end
         and pr.pr_code containing '.ipf.'
    into
        :reg_unicode,
        :kal_new_kf_name,
        :k_kontract_no,
        :k_fact_disb_date,
        :k_amount_approved,
        :k_kredline_end,
        :policy_no,
        :ik_commission,
        :commission,
        :nsp,
        :mbulak_share,
        :ag_id,
        :k_fact_closed_date,
        :status,
        :c_id,
        :k_id
    do
    begin
        select reg.reg_name
        from region reg
        where reg.reg_unicode = :reg_unicode
        into :reg_name;

        select c.c_name_fio
        from clients c
        where c.c_id = :c_id
        into :c_name_fio;

        select c_insur.c_name_fio
        from agreements ag
        inner join clients c_insur on c_insur.c_id = ag.ag_object_id
        where ag.ag_id = :ag_id
        into :insurance_comp_name;

        insurer_settlement_amount = commission + nsp - mbulak_share;

        select coalesce(sum(ok_amount), 0)
        from operkontract
        inner join operlist on ol_id = ok_from_ol_id
        where ok_from_k_id = :k_id and
              ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance') and
              is_delete = 0 and
              ol_date <= :rep_date_end
        into :payed_amount;

        insurance_sum_due = coalesce(:ik_commission, 0) - coalesce(:payed_amount, 0);

        suspend;
    end
end
/*
reg_name -- офис
kal_new_kf_name -- номер папки
k_kontract_no -- номер договора
c_name_fio -- фио клиента
k_fact_disb_date -- дата выдачи
k_amount_approved -- сумма кредита
k_kredline_end -- плановая дата закрытия
insurance_comp_name -- наименование поставщика
policy_no --  номер страховой премии
commission -- сумма страховой премии
nsp -- сумма нсп страховой премии
mbulak_share -- комисся Мбулак
k_fact_closed_date -- фактическая дата закрытия
status -- статус
payed_amount -- оплаченная сумма страховой премии за период
insurance_sum_due -- остаток задолженности по страховой премии
insurer_settlement_amount -- сумма расчета со Страховой Компанией  = гр 10+ гр 11- гр 12
*/
