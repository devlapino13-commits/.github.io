CREATE PROCEDURE GET_REF_ACCOUNT_CLIENTS(
    SAB_FULL_NAME D_STR_255,
    SAB_INVERT_NAME D_STR_255,
    C_ID ID_TYPE,
    C_NON_RESIDENT SMALLINT = 0,
    C_CLIENT_TYPE SMALLINT = 0,
    REP_DATE DATE = null,
    REG_ID INTEGER = null)
RETURNS (
    ACC_ID D_ID)
AS
begin
  if (c_id is null) then
    exception e_default 'Client ID is emptry';

  if (rep_date is null) then
    rep_date = current_date;

  if (reg_id is null) then
  begin
    reg_id = f_hq_id();
  end

  select acc_id
  from get_ref_account_any(:sab_full_name, :sab_invert_name, 'clients', :c_id, '', :reg_id, -1, :c_non_resident, :c_client_type, :rep_date)
  into :acc_id;
  suspend;
end
