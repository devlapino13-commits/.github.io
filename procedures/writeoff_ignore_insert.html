CREATE PROCEDURE WRITEOFF_IGNORE_INSERT(
    PERIOD INTEGER,
    KONTRACT_NO VARCHAR(20),
    SH_DATE DATE,
    REASON_STR STR_255)
RETURNS (
    MESSAGE_STR VARCHAR(255))
AS
declare variable k_id id_type;
begin
  kontract_no = trim(coalesce(kontract_no, ''));
  reason_str = trim(coalesce(reason_str, ''));
  period = coalesce(period, 0);

  if ((kontract_no = '') or (reason_str = '') or (period = 0)) then
    message_str = 'не все данные заполнены';
  else
  begin
    select k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id;

    if (k_id is not null) then
      update or insert into writeoff_ignore (wfi_period, wfi_from_k_id, wfi_sh_date, wfi_reason)
      values (:period, :k_id, :sh_date, :reason_str)
      matching (wfi_period, wfi_from_k_id, wfi_sh_date);
    else
      message_str = 'Контракта не существует';
  end
  suspend;
end
