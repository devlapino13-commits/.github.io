CREATE PROCEDURE CM_RU_0024_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable KF_OFFICER ID_TYPE;
declare variable PARENT_REG_ID integer;
declare variable C_ID ID_TYPE;
declare variable CONTR_CHECK_RESULT STR_255;
declare variable CONTR_TYPE STR_255;
declare variable TEMP_STR STR_255;
declare variable TEMP_STR2 STR_255;
declare variable CDV_VALUE STR_255;
declare variable K_CIRCLE integer;
declare variable JK_SYSTEM_CODE STR_255;
declare variable KOMMENT_VALUE STR_255;
declare variable REG_LOGIC_LABEL STR_255;
declare variable CONTR_CHECKED_DATE date;
declare variable KC_CIRCLE integer;
declare variable rd_tag integer;
begin
  res_error_text = '';
  res_error_is_critical = 0;
  select kal.kal_new_kf_officer,
         reg.reg_logic_label,
         kont.k_disb_date
  from knaktlist kal
  inner join kontract kont on kal.kal_from_kf_id = kont.k_from_kf_id
  inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where kont.k_disb_date between kal.kal_date and kal.kal_date_end
  and kal.is_delete = 0
  and kont.k_id = :k_id
  and pr.pr_code not containing('.BARAKAT PLUS.')
  and pr.pr_code not containing('.PATENT.')
  into :kf_officer, :reg_logic_label, :KF_DISB_DATE;
  reg_logic_label = coalesce(reg_logic_label, '');
  if (reg_logic_label containing ('.mega_office.')) then -- Мега офисы
  begin
    for select kncl.kc_from_c_id,
               cl.c_name_fio,
               cl.c_pass_no,
               kncl.kc_circle
    from knclient kncl
    inner join clients cl on cl.c_id = kncl.kc_from_c_id
    where kncl.kc_amount_approved > 0
    and kncl.kc_from_k_id = :k_id
    and kncl.kc_amount_approved >0
    into :c_id, :res_info_text, KF_CLIENT_PASS_NO, kc_circle
    do
    begin
      res_error_text = '';
      res_error_is_critical = 0;
      contr_check_result = null;
      contr_type = null;
      /*Беру результаты проверки ВО за последнюю неделю*/
      for select rd_res.rd_code,
                 rd_type.rd_code,
                 cont.contr_checked_date,
                 '"'||rd_res.rd_name||'"',
                 rd_res.rd_tag
      from controller cont, refdata rd_res, refdata rd_type
      where cont.contr_from_c_id = :c_id
      and cont.contr_is_delete = 0
      and rd_res.rd_id  = cont.contr_check_result
      and rd_type.rd_id = cont.contr_rd_type
      and cont.contr_checked_date between :KF_DISB_DATE - 7 and :KF_DISB_DATE
      and cont.contr_type in (1,2)
      into :contr_check_result,
           :contr_type,
           :contr_checked_date,
           :temp_str,
           :rd_tag
      do
      begin
            contr_check_result = coalesce(contr_check_result,'ControllerCheckStatus.NotChecked');
            if (contr_type = 'ControllerType.Specialist_Visitor') then temp_str2 = 'CВО, указал ';
            if (contr_type = 'ControllerType.LO') then temp_str2 = 'СВЗ, указал ';
            select CDV_VALUE
            from HP_GET_CUSTOMDATAVALUE(0, :c_id, 'work_komment',:contr_checked_date)
            rows 1
            into :KOMMENT_VALUE;
            
            KOMMENT_VALUE = coalesce(KOMMENT_VALUE,'');
            CDV_VALUE = '';
            res_error_is_critical = 0;
            if (rd_tag = 0) then CDV_VALUE = ' (Выдача не рекомендована)';
            if (rd_tag = -1) then
            begin
                res_error_is_critical = 1;
                CDV_VALUE = ' (Выдача запрещена)';
            end
            if (KOMMENT_VALUE <> '') then
            begin
                KOMMENT_VALUE = '. Коммент: '||KOMMENT_VALUE;
            end
            res_error_text=left('Место работы кл. '||temp_str2||temp_str||CDV_VALUE||KOMMENT_VALUE, 255);
            suspend;
      end
    
      contr_check_result = coalesce(contr_check_result,'ControllerCheckStatus.NotChecked');
    
      if (contr_check_result = 'ControllerCheckStatus.NotChecked') then
      begin
        res_error_is_critical = 0;
        if ((KF_OFFICER = 1000000000004) and (kc_circle <=1)) then res_error_is_critical = 1;
        res_error_text='Визит - осмотр не проведен';
        suspend;
      end
    end
  end
end
