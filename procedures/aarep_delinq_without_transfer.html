CREATE PROCEDURE AAREP_DELINQ_WITHOUT_TRANSFER(
    REP_DATE D_DATE,
    REP_ORIG_DATE D_DATE,
    REP_NEW_DATE D_DATE)
RETURNS (
    K_ID D_ID,
    LOAN_BALANCE D_CURRENCY,
    DELINQ_BALANCE D_CURRENCY,
    DELINQ_DATE D_DATE,
    DELINQ_DAY D_INTEGER,
    K_KONTRACT_NO D_STR_20,
    REG_NAME D_STR_100,
    IS_TRANSFERED D_SMALLINT,
    TRANSFER_OS D_CURRENCY)
AS
declare variable OK_SH_ID D_INTEGER;
declare variable ZAD_OS D_CURRENCY;
declare variable OK_SH_OS D_CURRENCY;
begin
  for select oks_from_k_id, oks_balance, oks_delinq_date, k_kontract_no, reg_code || ' ' || reg_name
  from oksummary, kontract, region
  where :rep_date between oks_date_begin and oks_date_end and
        oks_balance > 0 and
        oks_from_k_id = k_id and
        oks_region_id = reg_unicode and
        oks_delinq_date >= (:rep_date - 360)
        and oks_close_state = 0 
  into k_id, loan_balance, delinq_date, k_kontract_no, reg_name
  do
  begin
    delinq_day = 0;
    delinq_balance = 0;
    is_transfered = 0;
    transfer_os = 0;
    if (delinq_date = :rep_new_date) then
    begin
      ok_sh_id = null;
      for select ok_sh_id, ok_sh_os
      from operkred, operlist
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date and
            ok_type = 2 and
            ok_sh_date = :rep_new_date and
            ok_sh_os > 0 and
            ok_from_ol_id = ol_id and
            ol_oper_type = 1 and
            ol_oper_subtype = 9
      into ok_sh_id, ok_sh_os
      do
      begin
        is_transfered = 1;
        zad_os = 0;
        transfer_os = transfer_os + ok_sh_os;

        select sum(ok_os + ok_sh_os)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date and
              ok_sh_id = :ok_sh_id and
              ok_type <= 6
        into zad_os;
        if (zad_os is null) then
          zad_os = 0;

        if (zad_os > 0) then
          delinq_date = rep_orig_date;
      end
    end
    if (delinq_date > rep_date) then
      delinq_date = null;

    if (delinq_date is not null) then
      delinq_day = datediff(day, delinq_date, rep_date);

    if (delinq_date is not null) then
      delinq_balance = loan_balance;

    suspend;
  end
end
