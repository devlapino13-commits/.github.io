CREATE PROCEDURE HP_EXTINFO_TO_SEND_KDN_SCORE(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable C_INN D_STR_20;
declare variable INCOME D_INTEGER;
declare variable MA_ID D_ID;
declare variable C_ID D_ID;
declare variable DEBT_AMOUNT numeric(18,4);
declare variable PAYMENT_AMOUNT numeric(18,4);
begin
    if (:object_kind <> 81) then
        exception e_default 'object_kind must be miniappclient';

    select coalesce(cl.c_inn, ''), mac.mac_amount,ma.ma_id,cl.c_id
    from miniappclient mac
    inner join clients cl on cl.c_id = mac.mac_from_c_id
    inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
    where mac.mac_id = :object_id
      and ma.ma_is_active = 1
      and ma.ma_date_end >= '31.12.9999'
      and mac.mac_amount > 0
    into :c_inn, :debt_amount, :ma_id, :c_id;

    if (coalesce(:c_inn, '') = '') then
    begin
        suspend;
        exit;
    end

    select ca.ca_avr_income
    from cfp_anketa ca
    where ca.ca_object_kind = :object_kind
      and ca.ca_object_id = :object_id
      and ca.ca_from_c_id = :c_id
    into :income;

  select sum(sh_all_with_tax) / max(sh_id)
  from fm_schedule_list(11, :ma_id)
  where sh_id <> 0
  into :payment_amount;

  payment_amount = coalesce(:payment_amount, 0);

    json_value = '{' || 
                   f_json_make_pair('consent', 1, 'num') || ', ' ||
                   f_json_make_pair('iin', :c_inn, 'str') || ', ' ||
                   f_json_make_pair('hide_debts', 1, 'num') || ', ' ||
                   f_json_make_pair('income', :income, 'num') || ', ' ||
                   f_json_make_pair('payment_amount', :payment_amount, 'num') || ', ' ||
                   f_json_make_pair('new_debt_amount', :debt_amount, 'num') ||
                 '}';

    suspend;
end
