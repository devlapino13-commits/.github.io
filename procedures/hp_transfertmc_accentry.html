CREATE PROCEDURE HP_TRANSFERTMC_ACCENTRY(
    TRT_ID D_INTEGER,
    DT_CURR D_DATE = null)
RETURNS (
    SAVE_OL_ID D_ID,
    SIGN_OL_ID D_ID)
AS
declare variable nyear d_integer;
declare variable nmonth d_integer;
declare variable curr_code d_str_lang;
declare variable jsdata d_blob_text;
declare variable dt_act d_date;
declare variable nm_act d_integer;
declare variable note d_str_long;
declare variable offices_id d_integer;
declare variable officer_id d_integer;
declare variable mols_id d_id;
declare variable molr_id d_id;
declare variable wps_id d_id;
declare variable wpr_id d_id;
declare variable un_id d_id;
declare variable numb d_integer;
declare variable ncount d_integer;
declare variable parent_ol_id d_id;
declare variable ol_id d_id;
declare variable ol_id1 d_id;
declare variable ol_id2 d_id;
declare variable ol_id3 d_id;
declare variable ob_detail d_str_long;
declare variable office_id d_integer;
declare variable bill_id d_id;
declare variable bill_line_no d_integer;
declare variable line_no d_integer;
declare variable ml_name d_str_255;
declare variable invent_str d_str_code;
declare variable state_id d_integer;
declare variable state_note d_str_100;
declare variable acc d_str_20;
declare variable acc1 d_str_20;
declare variable acc_balance d_str_20;
declare variable acc_amort d_str_20;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable qty d_currency;
declare variable summa d_currency;
declare variable summa_amort d_currency;
declare variable mc_id d_id;
declare variable pet_id d_integer;
declare variable ped_id d_integer;
begin
  dt_curr = coalesce(dt_curr, current_date);
  nyear = extract(year from dt_curr);
  nmonth = extract(month from dt_curr);
  curr_code = '417';

  jsdata = (
    select trt_json
    from transfertmc
    where trt_id = :trt_id);
  for
    select
      dt_act,
      nm_act,
      note,
      offices_id,
      officer_id,
      mols_id,
      molr_id,
      wpr_id
    from hp_transfertmc_jshead(:jsdata)
  into
    :dt_act,
    :nm_act,
    :note,
    :offices_id,
    :officer_id,
    :mols_id,
    :molr_id,
    :wpr_id
  do
  begin
    un_id = (
      select first 1 un_id
      from username
      where un_from_c_id = :molr_id
      order by un_id desc);

    insert into operlist (
      ol_from_reg_id,
      ol_oper_type,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id,
      ol_oper_subtype)
    values (
      :offices_id,
      3,
      :dt_curr,
      :nm_act,
      10,
      :note,
      :un_id,
      4)
    returning
      ol_id
    into
      :save_ol_id;

    -- выбытие -----------------------------------------------------------------
    ncount = 0;
    for
      select
        bill_id,
        bill_no,
        mc_id,
        wps_id,
        qty,
        summa,
        acc,
        invent_str,
        state_id,
        state_note
      from hp_transfertmc_jsdetail(:jsdata)
    into
      :bill_id,
      :bill_line_no,
      :mc_id,
      :wps_id,
      :qty,
      :summa,
      :acc,
      :invent_str,
      :state_id,
      :state_note
    do
    begin
      if (not exists(
        select
          om_bill_id
        from opermat
          inner join operlist
            on ol_id = om_from_ol_id
              and ol_date_oper_day <= :dt_curr
              and is_delete = 0
          inner join workplace
            on wp_id = om_workplace_id
              and wp_from_reg_id = :offices_id
        where om_matotvet_id = :mols_id
          and om_bill_id = :bill_id
          and om_bill_line_no = :bill_line_no
        group by
          om_bill_id,
          om_bill_line_no,
          om_workplace_id,
          om_matotvet_id
        having sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) >= :qty))
      then
        exception e_message f_format('ТМЦ с инв.номером %1 не может быть передана в данном АПП.', invent_str);

      ncount = ncount + 1;
      acc1 = iif(offices_id = officer_id, acc,
        overlay(acc placing substring(officer_id from 2 for 4) from 1 for 4));

      insert into opermat (
        om_from_ol_id,
        om_number,
        om_bill_id,
        om_bill_line_no,
        om_oper_type,
        om_workplace_id,
        om_matotvet_id,
        om_kolich_minus,
        om_summa_minus,
        om_summa_total,
        om_account,
        om_workplace_info,
        om_matotvet_info,
        om_account_info,
        om_main_line,
        om_state_rd_id,
        om_state_note)
      values (
        :save_ol_id,
        2 * :ncount - 1,
        :bill_id,
        :bill_line_no,
        3,
        :wps_id,
        :mols_id,
        :qty,
        :summa,
        :summa,
        :acc,
        :wpr_id,
        :molr_id,
        :acc,
        1,
        :state_id,
        :state_note);

      insert into opermat (
        om_from_ol_id,
        om_number,
        om_bill_id,
        om_bill_line_no,
        om_oper_type,
        om_workplace_id,
        om_matotvet_id,
        om_kolich_plus,
        om_summa_plus,
        om_summa_total,
        om_account,
        om_workplace_info,
        om_matotvet_info,
        om_account_info,
        om_main_line)
      values (
        :save_ol_id,
        2 * :ncount,
        :bill_id,
        :bill_line_no,
        3,
        :wpr_id,
        :molr_id,
        :qty,
        :summa,
        :summa,
        :acc1,
        :wps_id,
        :mols_id,
        :acc1,
        0);

      -- если это объект списания ГСМ
      if (mc_id in (31,36)) then
      begin
        for
          select
            max(pet_id),
            max(ped_id)
          from pettrans
            inner join petdata
              on ped_from_pet_id = pet_id
                and ped_is_active = 1
          where pet_bill_id = :bill_id
            and pet_bill_line_no = :bill_line_no
        into
          :pet_id,
          :ped_id
        do
        begin
          update petdata set
            ped_is_active = 0
          where ped_from_pet_id = :pet_id
            and ped_is_active = 1;

          insert into petdata (
            ped_from_pet_id,
            ped_id,
            ped_date,
            ped_region_place_id,
            ped_from_c_id,
            ped_expense_summer,
            ped_expense_winter,
            ped_acc_balance,
            ped_acc_number,
            ped_is_active,
            ped_expense_summer_mount,
            ped_expense_winter_mount)
          select
            :pet_id,
            :ped_id + 1,
            :dt_curr,
            :officer_id,
            :molr_id,
            ped_expense_summer,
            ped_expense_winter,
            ped_acc_balance,
            ped_acc_number,
            1,
            ped_expense_summer_mount,
            ped_expense_winter_mount
          from petdata
          where ped_from_pet_id = :pet_id
            and ped_id = :ped_id;
        end
      end
      --
    end

    -- перемещение -------------------------------------------------------------
    if (offices_id <> officer_id) then
    for
      select
        bill_id,
        bill_line_no,
        ml_name,
        invent_str,
        acc_balance,
        acc_amort,
        qty,
        summa
      from hp_transfertmc_viewd(:trt_id)
    into
      :bill_id,
      :bill_line_no,
      :ml_name,
      :invent_str,
      :acc_balance,
      :acc_amort,
      :qty,
      :summa
    do
    begin
      -- бух.проводки
      parent_ol_id = save_ol_id;
      ob_detail = f_format('%1 %2', invent_str, ml_name);

      acc_dt = (
        select al_id from acc_list
        where al_from_reg_id = :officer_id
          and al_currency = :curr_code
          and al_account = :acc_balance);
      acc_kt = (
        select al_id from acc_list
        where al_from_reg_id = :offices_id
          and al_currency = :curr_code
          and al_account = :acc_balance);

      summa_amort = coalesce((
        select
          gai_amort
        from get_amort_info(:bill_id, :bill_line_no, :dt_curr)), 0);

      for
        select
          res_reg_id,
          res_acc_dt,
          res_acc_kt,
          res_line_no
        from aarep_loronostro_link(:offices_id, :acc_dt, :acc_kt)
      into
        :office_id,
        :acc_dt,
        :acc_kt,
        :line_no
      do
      begin
        if (ol_id1 is null or ol_id2 is null or ol_id3 is null) then
        begin
          insert into operlist (
            ol_parent_id,
            ol_from_reg_id,
            ol_oper_type,
            ol_oper_subtype,
            ol_date,
            ol_doc_no,
            ol_doc_kind,
            ol_memo,
            ol_user_id)
          values (
            :parent_ol_id,
            :office_id,
            3,
            4,
            :dt_curr,
            :nm_act,
            10,
            'Перемещение ТМЦ',
            :un_id)
          returning ol_id
          into :ol_id;

          if (line_no = 1) then
          begin
            ol_id1 = ol_id;
            parent_ol_id = ol_id;
          end else
          if (line_no = 2) then ol_id2 = ol_id;
          else
          if (line_no = 3) then ol_id3 = ol_id;
        end

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val)
        values (
          case :line_no
            when 1 then :ol_id1
            when 2 then :ol_id2
            when 3 then :ol_id3
          END, 
          :curr_code,
          1,
          :acc_dt,
          :acc_kt,
          :ob_detail,
          :summa,
          :summa);
      end

      -- сумма амортизации
      if (summa_amort > 0 and qty > 0) then
      begin
        acc_dt = (
          select al_id from acc_list
          where al_from_reg_id = :offices_id
            and al_currency = :curr_code
            and al_account = :acc_amort);
        acc_kt = (
          select al_id from acc_list
          where al_from_reg_id = :officer_id
            and al_currency = :curr_code
            and al_account = :acc_amort);

        for
          select
            res_reg_id,
            res_acc_dt,
            res_acc_kt,
            res_line_no
          from aarep_loronostro_link(:offices_id, :acc_dt, :acc_kt)
        into
          :office_id,
          :acc_dt,
          :acc_kt,
          :line_no
        do
        begin
          insert into operbook (
            ob_from_ol_id,
            ob_currency,
            ob_curr_rate,
            ob_account_dt,
            ob_account_kt,
            ob_detail,
            ob_summa,
            ob_summa_val)
          values (
            case :line_no
              when 1 then :ol_id1
              when 2 then :ol_id2
              when 3 then :ol_id3
            end,
            :curr_code,
            1,
            :acc_dt,
            :acc_kt,
            :ob_detail,
            :summa_amort,
            :summa_amort);
        end
      end

      insert into operamorlist(
        oal_bill_id, oal_bill_line_no, oal_from_ol_id, oal_is_delete, oal_oper_type,
        oal_lifetime_month, oal_year, oal_month, oal_dont_calc, oal_summa_os, oal_summa_amort)
      values(
        :bill_id, :bill_line_no, :ol_id1, 0, 3,
        0, :nyear, :nmonth, 0,  -:summa, -:summa_amort);
      insert into operamorlist(
        oal_bill_id, oal_bill_line_no, oal_from_ol_id, oal_is_delete, oal_oper_type,
        oal_lifetime_month, oal_year, oal_month, oal_dont_calc, oal_summa_os, oal_summa_amort)
      values(
        :bill_id, :bill_line_no, coalesce(:ol_id3, :ol_id2) , 0, 0,
        0, :nyear, :nmonth, 0,  :summa, :summa_amort);
    end
  end
  sign_ol_id = ol_id1;

  suspend;
end
