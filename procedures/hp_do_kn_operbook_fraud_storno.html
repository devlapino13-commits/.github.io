CREATE PROCEDURE HP_DO_KN_OPERBOOK_FRAUD_STORNO(
    ONE_OFR_ID D_INTEGER NOT NULL,
    NEED_DELETE D_BOOL NOT NULL,
    OL_MEMO D_STR_255 NOT NULL,
    UN_ID D_ID NOT NULL)
AS
declare variable al_id d_str_code;
declare variable acc_id d_id;
declare variable orp_id d_integer;
declare variable orp_state d_integer;
declare variable ol_id d_id;
declare variable reg_id d_integer;
declare variable k_disb_date d_date;
declare variable kc_id d_id;
declare variable main_ol_id d_id;
declare variable tmp_ol_id d_id;
declare variable bal_perc d_currency;
declare variable max_number d_integer;
declare variable k_id d_id;
declare variable rep_date d_date; /* Дата сторнирования */
declare variable fraud_c_id d_id; /* C_ID сотрудника-мошеника */
declare variable ofr_id d_integer;
declare variable ofr_kk_date d_date;
declare variable ofr_kk_no d_str_code;
declare variable k_kontract_no d_str_code;
declare variable kcp_period d_integer;
declare variable need_generate_memo boolean;
begin
  -- Процедура по сторнированию выданного займа из-за мошеничества сотрудника
  -- Все действия, включая выдачи отменяются
  -- 'fraud_2020_12'
  need_generate_memo = (coalesce(ol_memo, '') = '');

  if (one_ofr_id is null) then
    one_ofr_id = 0;

  if (need_delete = 1 and one_ofr_id = 0) then
      exception e_message f_format('Нельзя запускать с параметром Need_delete = 1 и One_Ofr_ID = 0');

  for select ofr_id, ofr_from_k_id, ofr_storno_date, ofr_fraud_c_id, ofr_from_al_id,
             ofr_from_acc_id, ofr_storno_ol_id, ofr_kk_date, ofr_kk_no
  from operfraud
  where (:one_ofr_id = 0 or ofr_id = :one_ofr_id)
  into :ofr_id, :k_id, :rep_date, :fraud_c_id, :al_id, :acc_id, ol_id, ofr_kk_date, ofr_kk_no
  do
  begin
    if (rep_date is null) then
      exception e_default f_format('storno date is null, ofr_id: ', ofr_id);

    select kc_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into kc_id;

    if (ol_id is not null) then
    begin
      if (need_delete = 0) then
      begin
        if (exists(select * from operlist where ol_id = :ol_id and is_delete = 0)) then
          Continue;
      end else
      begin
        update operlist
        set is_delete = 1
        where ol_id = :ol_id;

        delete from kn_operbook
        where kob_from_kc_id = :kc_id and
              kob_ol_date >= :rep_date and
              kob_ol_is_active = 1;
      end

      ol_id = null;
    end

    if (al_id is null) then
      exception e_default f_format('76-02/5 al_id not found, reg_id: %1', reg_id);
  
    if (al_id is null) then
      exception e_default f_format('60323 acc_id not found, c_id: %1', fraud_c_id);

    -- Определяем офис где числится займ
    select kal_new_reg_id, k_disb_date, k_kontract_no
    from kontract, knaktlist kal
    where k_id = :k_id and
          k_from_kf_id = kal.kal_from_kf_id and
          kal.is_delete = 0 and
          kal.kal_date <= :rep_date and
          kal.kal_date_end >= :rep_date
    into reg_id, k_disb_date, k_kontract_no;

    if (need_generate_memo) then
    begin
      ol_memo = f_format('СТОРНО проводки по договору займа %1, протокол %2 от %3 г.',
                         k_kontract_no, ofr_kk_no, f_datetostr(ofr_kk_date));
    end
  
    -- Создаем новую операцию
    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind,
                          ol_memo, ol_parent_id, is_delete, ol_user_id)
    values (:reg_id, 1, 13, :rep_date, 9, :ol_memo, null, 0, :un_id)
    returning ol_id
    into :ol_id;

    update or insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id, opl_from_acc_id)
    values (:ol_id, 3, :al_id, :acc_id)
    matching (opl_from_ol_id);
  
    -- На дату сторнирования пересчитываем данные
    if (not exists(select * from kn_oksummary where oks_from_k_id = :k_id and oks_date_begin = :rep_date))
    then
      execute procedure hp_kn_oksummary_insert(:kc_id, :rep_date, 1);
    else
    begin
      update kn_oksummary
      set oks_is_fix_bal_perc = 0,
          oks_need_recalc = 1
      where oks_from_k_id = :k_id and
            oks_date_begin = :rep_date;
    end

    -- На дату сторнирования пересчитываем проводки
    execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :rep_date - 1);
  
    -- Все движения после даты сторнирования аннулируем, их нужно вручную перепринят на 60323
    for select distinct iif(ol_parent_id is not null, ol_parent_id, ol_id)
    from operkred, operlist
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date > :rep_date and
          ok_from_ol_id = ol_id
    into main_ol_id
    do
    begin
      update operlist
      set is_delete = 1
      where ol_id = :main_ol_id;
    end

    for select distinct iif(ol_parent_id is not null, ol_parent_id, ol_id)
    from operkroff, operlist
    where oko_from_k_id = :k_id and
          oko_ol_is_active = 1 and
          oko_ol_date > :rep_date and
          oko_from_ol_id = ol_id
    into main_ol_id
    do
    begin
      update operlist
      set is_delete = 1
      where ol_id = :main_ol_id;
    end
  
    -- Сторнируем движения по займу
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                          ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
    select :ol_id, -1, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, -ok_os, -ok_perc, -ok_fine, ok_pay_type,
           -ok_kommis, -ok_sh_os, -ok_sh_perc, -ok_kommis_adv, -ok_tax
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date and
          ok_from_ol_id <> :ol_id
    order by ok_ol_date, ok_from_ol_id, ok_number;

    -- Убираем пени
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                          ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
    select :ol_id, -1, :k_id, sh_id, sh_date, 5, 0, 0, -sh_calc_fine, 3, 0, 0, 0, 0, 0
    from aarep_make_zad_list_temp(:k_id, :rep_date, 0, 1)
    where sh_calc_fine <> 0;

    -- Сторнируем движения по займу
    insert into operkroff (oko_from_ol_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc, oko_other,
                           oko_sh_os, oko_sh_perc, oko_sh_other, oko_ol_is_active, oko_ol_date)
    select :ol_id, oko_from_k_id, oko_from_c_id, oko_type, -oko_os, -oko_perc, -oko_other, -oko_sh_os,
           -oko_sh_perc, -oko_sh_other, 1, :rep_date
    from operkroff
    where oko_from_k_id = :k_id and
          oko_ol_is_active = 1 and
          oko_ol_date <= :rep_date and
          oko_from_ol_id <> :ol_id
    order by oko_ol_date, oko_from_ol_id, oko_id;

    for select distinct iif(ol_parent_id is not null, ol_parent_id, ol_id)
    from operkred, operlist
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date and
          ok_from_ol_id = ol_id and
          ok_from_ol_id <> :ol_id
    union
    select iif(ol_parent_id is not null, ol_parent_id, ol_id)
    from operkroff, operlist
    where oko_from_k_id = :k_id and
          oko_ol_is_active = 1 and
          oko_ol_date <= :rep_date and
          oko_from_ol_id = ol_id and
          oko_from_ol_id <> :ol_id
    into main_ol_id
    do
    begin
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                            ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail)
      select :ol_id, -1, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, -ob_summa_val, -ob_summa,
             ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, 'Сторнирование: ' || coalesce(ob_detail, '')
      from operbook
      where ob_from_ol_id = :main_ol_id;
  
      for select ol_id
      from operlist
      where ol_parent_id = :main_ol_id and
            ol_id <> :main_ol_id
      into tmp_ol_id
      do
      begin
        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                              ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail)
        select :ol_id, -1, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, -ob_summa_val, -ob_summa,
               ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, 'Сторнирование: ' || coalesce(ob_detail, '')
        from operbook
        where ob_from_ol_id = :tmp_ol_id;
      end
    end

    select result_perc_end
    from aarep_calc_perc_one_cls(:k_id, :rep_date, :rep_date, 0, 1, 1)
    into bal_perc;
  
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine,
                          ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
    values (:ol_id, -1, :k_id, 0, :k_disb_date, 7, 0, -:bal_perc, 0,
            3, 0, 0, 0, 0, 0);
  
    if (not exists(select * from kontstate
    where ks_from_k_id = :k_id and
          ks_date_begin <= :rep_date and
          ks_state = 2 and
          ks_ol_is_delete = 0)) then
    begin
      insert into kontstate (ks_from_ol_id, ks_from_k_id, ks_date_begin, ks_ol_is_delete, ks_state)
      values (:ol_id, :k_id, :rep_date, 0, 2);
    end

    kcp_period = extract(year from rep_date) * 100 + extract(month from rep_date);
    if (exists(select * from kontcalcperc where kcp_from_k_id = :k_id and kcp_period = :kcp_period)) then
      execute procedure aarep_calc_perc_fill_one(:k_id, :kcp_period, :kcp_period);
  
    update operfraud
    set ofr_storno_ol_id = :ol_id
    where ofr_id = :ofr_id;

    update kn_oksummary
    set oks_bal_perc_fixed = 0,
        oks_is_fix_bal_perc = 1,
        oks_need_recalc = 1
    where oks_from_k_id = :k_id and
          oks_date_begin = :rep_date;
  
    insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                             kob_summa_tax_dt, kob_summa_tax_kt,
                             kob_detail_id, kob_client_dt, kob_client_kt, kob_from_ol_id, kob_from_kc_id, kob_from_rml_id)
    select kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, -kob_summa, -kob_summa_val, -kob_summa_tax_dt,
           -kob_summa_tax_kt, kob_detail_id, kob_client_dt, kob_client_kt,
           :ol_id, kob_from_kc_id, kob_from_rml_id
    from kn_operbook
    where kob_from_kc_id = :kc_id and
          kob_ol_is_active = 1 and
          kob_ol_date <= :rep_date and
          kob_from_ol_id <> :ol_id;
  
    max_number = 0;
    select max(ob_number)
    from operbook
    where ob_from_ol_id = :ol_id
    into max_number;
    if (max_number is null) then
      max_number = 0;
  
    -- Сажаем денежные операции на 76-02/5
    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail)
    select :ol_id, -1, ob_currency, ob_curr_rate, ob_client_dt, :al_id, -ob_summa_val, -ob_summa,
           ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_number <= :max_number and
          (ob_account_kt containing '50-01' or
           ob_account_kt containing '51-01' or
           ob_account_kt containing '76-10');
  
    insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                          ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail)
    select :ol_id, -1, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, -ob_summa_val, -ob_summa,
           ob_from_pc_id, ob_from_kp_id, ob_oper_mask, :al_id, ob_client_kt, ob_detail
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_number <= :max_number and
          (ob_account_dt containing '50-01' or
           ob_account_dt containing '51-01' or
           ob_account_dt containing '76-10');
  
    max_number = 0;
    select max(kob_id)
    from kn_operbook
    where kob_from_ol_id = :ol_id
    into max_number;
    if (max_number is null) then
      max_number = 0;
  
    insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                             kob_summa_tax_dt, kob_summa_tax_kt,
                             kob_detail_id, kob_client_dt, kob_client_kt, kob_from_ol_id, kob_from_kc_id, kob_from_rml_id)
    select kob_currency, kob_curr_rate, :acc_id, kob_acc_kt, -kob_summa, -kob_summa_val, -kob_summa_tax_dt,
           -kob_summa_tax_kt, kob_detail_id, kob_client_dt, kob_client_kt,
           :ol_id, kob_from_kc_id, kob_from_rml_id
    from kn_operbook, ref_account, sys_accbalance
    where kob_from_kc_id = :kc_id and
          kob_ol_is_active = 1 and
          kob_ol_date <= :rep_date and
          kob_from_ol_id = :ol_id and
          kob_id <= :max_number and
          kob_acc_kt = acc_id and
          acc_from_sab_id = sab_id and
          (sab_full_code starting 'Деньги.' or
           sab_full_code starting 'ПлатСистемы.');
  
    insert into kn_operbook (kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                             kob_summa_tax_dt, kob_summa_tax_kt,
                             kob_detail_id, kob_client_dt, kob_client_kt, kob_from_ol_id, kob_from_kc_id, kob_from_rml_id)
    select kob_currency, kob_curr_rate, kob_acc_dt, :acc_id, -kob_summa, -kob_summa_val, -kob_summa_tax_dt,
           -kob_summa_tax_kt, kob_detail_id, kob_client_dt, kob_client_kt,
           :ol_id, kob_from_kc_id, kob_from_rml_id
    from kn_operbook, ref_account, sys_accbalance
    where kob_from_kc_id = :kc_id and
          kob_ol_is_active = 1 and
          kob_ol_date <= :rep_date and
          kob_from_ol_id = :ol_id and
          kob_id <= :max_number and
          kob_acc_dt = acc_id and
          acc_from_sab_id = sab_id and
          (sab_full_code starting 'Деньги.' or
           sab_full_code starting 'ПлатСистемы.');

    delete from kn_operbook
    where kob_from_kc_id = :kc_id and
          kob_ol_date > :rep_date and
          kob_ol_is_active = 1;

    -- Включаем в список сторно-проводок
    insert into operrepeat (orp_from_k_id, orp_memo, orp_ol_date, orp_begin_date,
                            orp_book_reverse, orp_loan_reverse, orp_sql)
    values (:k_id, :ol_memo, :rep_date, :rep_date, 1, 1, 'EMPTY')
    returning orp_id, orp_state
    into :orp_id, orp_state;
  
    execute procedure hp_operrepeat_set_status(:orp_id, :orp_state, :un_id);
    execute procedure hp_operrepeat_set_status(:orp_id, 4, :un_id);
  end
end
