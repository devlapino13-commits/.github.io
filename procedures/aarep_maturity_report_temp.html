CREATE PROCEDURE AAREP_MATURITY_REPORT_TEMP(
    REG_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    YEAR_NO INTEGER,
    MONTH_NO INTEGER,
    DIFF_NO INTEGER,
    PLAN_OS DECIMAL(18,4),
    K_ID DECIMAL(18,0),
    PLAN_LLR_SUMMA DECIMAL(18,4),
    PLAN_PERC DECIMAL(18,4))
AS
declare variable plan_date date;
declare variable kont_id_prior decimal(18,0);
declare variable llr_perc decimal(18,7);
begin
  kont_id_prior = 0;
  rep_date = rep_date + 1;

  for select aarep_k_id, aarep_date_action, aarep_k_deliq_os, aarep_k_deliq_perc
      from aarep_kontract_id(:reg_id, :rep_date - 1, :rep_date - 1, 0, '000', 5)
      union
      select aarep_k_id, aarep_date_action, aarep_plan_os, aarep_plan_perc
      from aarep_kontract_id(:reg_id, :rep_date, '31.12.2999', 0, '000', 10)
      order by 1, 2
      into :k_id, :plan_date, :plan_os, plan_perc
  do
  begin
    if (kont_id_prior <> k_id) then
    begin
      select lk_percent
      from llc_main, llc_kont
      where lk_date <= :rep_date - 1 and
            lk_date_end >= :rep_date - 1 and
            lk_from_k_id = :k_id and
            lm_is_signed = 1 and
            lm_id = lk_from_lm_id
      order by lm_date desc
      rows 1
      into :llr_perc;
      if (llr_perc is null) then
        llr_perc = 0;
    end
    year_no = extract(year from plan_date);
    month_no = extract(month from plan_date);
    diff_no = (year_no * 12 + month_no) - (extract(year from rep_date) * 12 + extract(month from rep_date)) + 1;
    plan_llr_summa = cast((plan_os * llr_perc) / 100 as decimal(18,2));
    suspend;
    kont_id_prior = k_id;
  end
  suspend;

end
