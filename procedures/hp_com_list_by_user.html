CREATE PROCEDURE HP_COM_LIST_BY_USER(
    UN_ID D_ID,
    REG_ID D_INTEGER,
    FILTER_TEXT D_STR_255 = null)
RETURNS (
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    TXT_1 D_STR_255,
    TXT_2 D_STR_255)
AS
declare variable c_id d_id;
declare variable enum_cancel d_integer;
declare variable obj_kind_id d_integer;
declare variable k_id d_id;
declare variable txt_3 d_str_255;
declare variable cmc_name d_str_255;
declare variable need_suspend d_smallint;
begin
  obj_kind = 'com_list';

  if (filter_text is null) then
    filter_text = '';

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exit;

  enum_cancel = f_enum('com_action_status', 'cancelled', 'id');
  obj_kind_id = f_objkind('kontract', 'id');

  if (not exists(select ps_from_reg_id
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :c_id and
                       ps_date_begin <= current_date and
                       ps_date_end >= current_date and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code in ('JobSystem.Region_KM') and
                       ps_is_working = 1)) then
    reg_id = 0;

  for select res_c_id
      from hp_staff_changer_list(:un_id, 1)
      where :reg_id = 0
      union
      select res_c_id
      from hp_staff_changer_list(:un_id, -1)
      where :reg_id <> 0
      into :c_id
  do
  begin
    if (reg_id = 0) then
    begin
      for select cml_id,
                 iif(cml_object_kind = :obj_kind_id, cml_object_id, 0),
                 f_objname(cml_object_kind, cml_object_id),
                 cmc_name,
                 f_datetostr(cml_date) || '<br>' || f_enum('com_action_status', cml_last_status, 'name')
      from com_list, com_cfg
      where cml_init_c_id = :c_id and
            cml_from_cmc_id = cmc_id and
            cml_last_status <> :enum_cancel and
            not(cml_is_closed = 1 and datediff(day, cml_date, current_date) > 3) -- не показываем старые (более мес)
      into obj_id, k_id, txt_1, cmc_name, txt_2
      do
      begin
        if (k_id <> 0) then
        begin
          txt_3 = '';
          select kal.kal_new_kf_name
          from knaktlist kal, kontract
          where k_id = :k_id and
                kal.kal_from_kf_id = k_from_kf_id and
                current_date between kal.kal_date and kal.kal_date_end and
                kal.is_delete = 0
          into txt_3;
  
          if (trim(txt_3) <> '') then
            txt_1 = txt_3 || ', ' || txt_1;
        end
        txt_1 = txt_1 || '<br>' || cmc_name;

        need_suspend = 0;
        if (filter_text = '')
        then need_suspend = 1;
        else
        begin
          if (txt_1 containing filter_text or txt_2 containing filter_text)
          then need_suspend = 1;
          else
          begin
            if (exists(select * from knclient, clients
            where kc_from_k_id = :k_id and
                  kc_from_c_id = c_id and
                  c_name_fio containing :filter_text)) then need_suspend = 1;
          end
        end
        if (need_suspend = 1) then
          suspend;
      end
    end else
    begin
      for select cml_id,
                 iif(cml_object_kind = :obj_kind_id, cml_object_id, 0),
                 f_objname(cml_object_kind, cml_object_id),
                 cmc_name,
                 f_datetostr(cml_date) || '<br>' || f_enum('com_action_status', cml_last_status, 'name')
      from com_list, com_cfg, region_list(:reg_id, 1, 3, 1) reg
      where cml_from_reg_id = reg.id_list and
            cml_from_cmc_id = cmc_id and
            cml_last_status <> :enum_cancel and
            not(cml_is_closed = 1 and datediff(day, cml_date, current_date) > 1) -- не показываем старые (более мес)
      into obj_id, k_id, txt_1, cmc_name, txt_2
      do
      begin
        if (k_id <> 0) then
        begin
          txt_3 = '';
          select kal.kal_new_kf_name
          from knaktlist kal, kontract
          where k_id = :k_id and
                kal.kal_from_kf_id = k_from_kf_id and
                current_date between kal.kal_date and kal.kal_date_end and
                kal.is_delete = 0
          into txt_3;
  
          if (trim(txt_3) <> '') then
            txt_1 = txt_3 || ', ' || txt_1;
        end
        txt_1 = txt_1 || '<br>' || cmc_name;

        need_suspend = 0;
        if (filter_text = '')
        then need_suspend = 1;
        else
        begin
          if (txt_1 containing filter_text or txt_2 containing filter_text)
          then need_suspend = 1;
          else
          begin
            if (exists(select * from knclient, clients
            where kc_from_k_id = :k_id and
                  kc_from_c_id = c_id and
                  c_name_fio containing :filter_text)) then need_suspend = 1;
          end
        end
        if (need_suspend = 1) then
          suspend;
      end
    end
  end
end
