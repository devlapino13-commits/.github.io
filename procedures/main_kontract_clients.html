CREATE PROCEDURE MAIN_KONTRACT_CLIENTS(
    K_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_TYPE INTEGER,
    KC_STATUS INTEGER,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_NUMBER INTEGER,
    CA_ADDRESS VARCHAR(255),
    C_FULL_ADDRESS VARCHAR(255))
AS
declare variable dd date;
declare variable kk_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select k_disb_date, k_from_kf_id
  from kontract
  where k_id = :k_id
  into :dd, :kf_id;

  select k_id
  from kontract, product
  where k_disb_date <= :dd and
        k_from_pr_id = pr_id and
        pr_category = 0 and
        k_from_kf_id = :kf_id
  order by k_disb_date desc
  rows 1
  into :kk_id;

  if (kk_id is null) then
    kk_id = :k_id;

  for select c_name_fio, c_birth_date, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_type, kc_status,
             kc_amount_approved, kc_number, adl_address_str
      from clients, knclient, addrclient, addrlist
      where kc_from_k_id = :kk_id and
            kc_from_c_id = c_id and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      order by kc_number
      into :c_name_fio, :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_type, :kc_status,
           :kc_amount_approved, :kc_number, :c_full_address
  do
  begin
    ca_address = c_full_address;
    suspend;
  end

end
