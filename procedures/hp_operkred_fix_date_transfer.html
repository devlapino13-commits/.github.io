CREATE PROCEDURE HP_OPERKRED_FIX_DATE_TRANSFER(
    K_ID D_ID)
AS
declare variable ol_date d_date;
declare variable sh_date date;
declare variable ol_id bigint;
declare variable new_sh_date date;
begin
  -- Процедура исправляет минусовые графики, возникщие из-за переноса даты погашения из-за выходных задним числом

  -- получаем платежи с отрицательным остатком в платеже (sh_date)
  for select ok_sh_date
  from operkred
  where ok_from_k_id = :k_id and
        ok_ol_is_active = 1 and
        ok_type <= 7
  group by ok_sh_date
  having sum(ok_os + ok_sh_os) < 0 or sum(ok_perc + ok_sh_perc) < 0
  into sh_date
  do
  begin
    -- получаем список операций переноса даты погашения, где участвовал этот SH_DATE
    for select distinct ok_from_ol_id
    from operkred, operlist
    where ok_from_ol_id = ol_id and
          ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type = 2 and
          ok_sh_date = :sh_date and
          ol_memo starting 'Перевод даты погашения'
    into ol_id
    do
    begin
      -- вычисляем, на какую дату перенесен платеж
      new_sh_date = null;
      select max(ok_sh_date)
      from operkred
      where ok_from_ol_id = :ol_id and
            ok_from_k_id = :k_id and
            ok_type = 2 and
            ok_ol_is_active = 1
      having min(ok_sh_date) = :sh_date and -- старая дата
            sum(ok_sh_os) = 0 and -- сумма перенесена, поэтому общий итог должен быть 0.
            sum(ok_sh_perc) = 0   -- сумма перенесена, поэтому общий итог должен быть 0.
      into new_sh_date;

      -- если нашли новую дату, обновляем старую дату на новую с операциях,
      -- которые были проведены после даты переноса
      if (new_sh_date is not null) then
      begin
        select ol_date from operlist
        where ol_id = :ol_id
        into ol_date;

        update operkred
        set ok_sh_date = :new_sh_date 
        where ok_ol_date > :ol_date and
              ok_from_k_id = :k_id and
              ok_sh_date = :sh_date;
      end
    end
  end
end
