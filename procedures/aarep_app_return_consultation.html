CREATE PROCEDURE AAREP_APP_RETURN_CONSULTATION(
    REG_ID D_INTEGER)
RETURNS (
    C_FIO D_STR_100,
    K_NO D_STR_20,
    AMOUNT D_CURRENCY,
    STATUS D_SMALLINT,
    RETURN_REASON D_STR_255,
    RETURN_TYPE D_STR_20,
    SIGN_TYPE D_STR_20,
    ARC_ID D_INTEGER,
    C_ID D_ID,
    K_ID D_ID,
    VAR_STATUS D_STR_20)
AS
begin
  for select arc.arc_id, c.c_name_fio, k.k_kontract_no, arci.arci_amount, arc.arc_status,
             case arci.arci_return_reason
               when 0 then 'Ошибочно внесенная сумма'
               when 1 then 'Отказ от получения услуг'
             end,
             case arci.arci_return_type
               when 0 then 'Наличный рассчет'
               when 1 then 'На баланс телефона'
             end,
             case arci.arci_sign_type
               when 0 then 'Электронная подпись'
               when 1 then 'Физическая подпись'
             end,
             arc.arc_from_c_id, arc.arc_from_k_id,
             f_enum('app_return_consult', arc.arc_status, 'name')
      from app_return_consult arc
      inner join app_return_consult_info arci on arc.arc_id = arci.arci_id
      inner join clients c on arc.arc_from_c_id = c.c_id
      inner join kontract k on arc.arc_from_k_id = k.k_id
      inner join region_list(:reg_id, 1, -1, 1) on id_list = arci.arci_app_office_id
      where arc.arc_status in (f_enum('app_return_consult', 'process', 'id'), f_enum('app_return_consult', 'approved', 'id'))
      into :arc_id, :c_fio, :k_no, :amount, :status, :return_reason, :return_type, :sign_type, :c_id, :k_id, :var_status
  do
  begin
    suspend;
  end
end
