CREATE PROCEDURE AAREP_REPAYMENT_REPTMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_PROJECT VARCHAR(100))
RETURNS (
    AA_K_ID DECIMAL(18,0),
    AA_BRANCH_NAME VARCHAR(100),
    AA_OFFICE_NAME VARCHAR(100),
    AA_LO_NAME VARCHAR(100),
    AA_LOAN_KIND VARCHAR(100),
    AA_LOAN_NAME VARCHAR(100),
    AA_ACTIVE_BALANCE_B DECIMAL(18,4),
    AA_ACTIVE_KOLICH_CLIENT_B INTEGER,
    AA_ACTIVE_BALANCE_E DECIMAL(18,4),
    AA_ACTIVE_KOLICH_CLIENT_E INTEGER,
    AA_DISB_AMOUNT DECIMAL(18,4),
    AA_DISB_KOLICH_CLIENT INTEGER,
    AA_TRANS_OS DECIMAL(18,4),
    AA_TRANS_PERC DECIMAL(18,4),
    AA_TRANS_FINE DECIMAL(18,4),
    AA_TRANS_KOMMIS DECIMAL(18,4),
    AA_WRITEOFF_OS DECIMAL(18,4),
    AA_WRITEOFF_PERC DECIMAL(18,4),
    AA_WRITEOFF_FINE DECIMAL(18,4),
    AA_DELIQ_OS DECIMAL(18,4),
    AA_DELIQ_PERC DECIMAL(18,4),
    AA_DELIQ_BALANCE DECIMAL(18,4),
    AA_DELIQ_KOLICH_CLIENT INTEGER,
    AA_DELIQ_BALANCE7 DECIMAL(18,4),
    AA_DELIQ_KOLICH_CLIENT7 INTEGER,
    AA_DELIQ_BALANCE30 DECIMAL(18,4),
    AA_DELIQ_KOLICH_CLIENT30 INTEGER,
    AA_DELIQ_BALANCE60 DECIMAL(18,4),
    AA_DELIQ_KOLICH_CLIENT60 INTEGER,
    AA_DELIQ_BALANCE90 DECIMAL(18,4),
    AA_DELIQ_KOLICH_CLIENT90 INTEGER,
    AA_DELIQ_BALANCE180 DECIMAL(18,4),
    AA_DELIQ_KOLICH_CLIENT180 INTEGER,
    AA_DELIQ_BALANCE360 DECIMAL(18,4),
    AA_DELIQ_KOLICH_CLIENT360 INTEGER,
    AA_COLL_PLAN_OS DECIMAL(18,4),
    AA_COLL_FACT_OS DECIMAL(18,4),
    AA_LOAN_AMOUNT DECIMAL(18,4),
    AA_DISB_DATE DATE,
    AA_KONTRACT_NO VARCHAR(20),
    AA_CIRCLE INTEGER,
    AA_PERIOD INTEGER,
    AA_REPAY_TOTAL_OS DECIMAL(18,4),
    AA_KF_NAME VARCHAR(100),
    AA_K_FINANCED INTEGER)
AS
declare variable cl_bal decimal(18,4);
declare variable cl_kol integer;
declare variable cl_bal_all decimal(18,4);
declare variable vrem decimal(18,7);
declare variable k_from_kf_id curr_type;
begin
-- knclient_fix

  if (rep_project is null) then
    rep_project = '';
  for select k_financed, aa_k_id, k_amount_approved, k_disb_date, kontract.k_kontract_no,
             k_circle + k_circle_internal - 1, k_kolich_repay, aa_curr_branch_name, aa_curr_office_name, aa_curr_kc_fio,
             aa_loan_kind, aa_loan_common_name,
--         AA_KF_NAME,

             aa_amount_active_begin, aa_kolich_cl_active_begin, aa_amount_active_end, aa_kolich_cl_active_end,
             aa_amount_disb, aa_kolich_cl_disb, aa_repay_all_os, aa_repay_all_perc, aa_repay_all_fine,
             aa_repay_all_kommis, aa_repay_writeoff_os, aa_repay_writeoff_perc, aa_repay_writeoff_fine,

             aa_deliq_os_all, aa_deliq_perc_all,

             aa_deliq_balance, aa_deliq_cl_kolich,

             aa_deliq_balance7, aa_deliq_cl_kolich7,

             aa_deliq_balance30, aa_deliq_cl_kolich30,

             aa_deliq_balance60, aa_deliq_cl_kolich60,

             aa_deliq_balance90, aa_deliq_cl_kolich90,

             aa_deliq_balance180, aa_deliq_cl_kolich180,

             aa_deliq_balance360, aa_deliq_cl_kolich360,

             aa_amount_sh_os, /* coll_plan_os */
             aa_repay_normonth_os, /* coll_fact_os */
             k_from_kf_id

      from kontract, aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_project, 0, 0, 1, 1, 0, 0, 0)
      where aa_k_id = k_id
      into :aa_k_financed, :aa_k_id, :aa_loan_amount, :aa_disb_date, :aa_kontract_no, :aa_circle, :aa_period,
           :aa_branch_name, :aa_office_name, :aa_lo_name, :aa_loan_kind, :aa_loan_name,
--         :AA_KF_NAME,
           :aa_active_balance_b, :aa_active_kolich_client_b, :aa_active_balance_e, :aa_active_kolich_client_e,
           :aa_disb_amount, :aa_disb_kolich_client, :aa_trans_os, :aa_trans_perc, :aa_trans_fine, :aa_trans_kommis,
           :aa_writeoff_os, :aa_writeoff_perc, :aa_writeoff_fine,

           :aa_deliq_os, :aa_deliq_perc,

           :aa_deliq_balance, :aa_deliq_kolich_client,

           :aa_deliq_balance7, :aa_deliq_kolich_client7,

           :aa_deliq_balance30, :aa_deliq_kolich_client30,

           :aa_deliq_balance60, :aa_deliq_kolich_client60,

           :aa_deliq_balance90, :aa_deliq_kolich_client90,

           :aa_deliq_balance180, :aa_deliq_kolich_client180,

           :aa_deliq_balance360, :aa_deliq_kolich_client360,

           :aa_coll_plan_os, :aa_coll_fact_os, :k_from_kf_id
  do
  begin
    cl_kol = 0;
    cl_bal = 0;
    cl_bal_all = 0;
    for select kc_amount_approved
        from knclient
        where kc_from_k_id = :aa_k_id and
              kc_from_pt_id <> :rep_project and
              kc_amount_approved is not null and
              kc_amount_approved <> 0
        into :cl_bal
    do
    begin
      if (cl_bal is null) then
        cl_bal = 0;
      cl_bal_all = cl_bal_all + cl_bal;
      cl_kol = cl_kol + 1;
    end
    if (cl_bal_all <> 0) then
      vrem = 1 - cl_bal_all / aa_loan_amount;
    else
      vrem = 1;

    aa_loan_amount = aa_loan_amount - cl_bal_all;
    aa_active_balance_b = cast(aa_active_balance_b * vrem as decimal(15,2));
    aa_active_balance_e = cast(aa_active_balance_e * vrem as decimal(15,2));
    aa_disb_amount = cast(aa_disb_amount * vrem as decimal(15,2));
    aa_trans_os = cast(aa_trans_os * vrem as decimal(15,2));
    aa_trans_perc = cast(aa_trans_perc * vrem as decimal(15,2));
    aa_trans_fine = cast(aa_trans_fine * vrem as decimal(15,2));
    aa_trans_kommis = cast(aa_trans_kommis * vrem as decimal(15,2));
    aa_writeoff_os = cast(aa_writeoff_os * vrem as decimal(15,2));
    aa_writeoff_perc = cast(aa_writeoff_perc * vrem as decimal(15,2));
    aa_writeoff_fine = cast(aa_writeoff_fine * vrem as decimal(15,2));

    if (aa_active_kolich_client_b > 0) then
      aa_active_kolich_client_b = aa_active_kolich_client_b - cl_kol;
    if (aa_active_kolich_client_e > 0) then
      aa_active_kolich_client_e = aa_active_kolich_client_e - cl_kol;
    if (aa_disb_kolich_client > 0) then
      aa_disb_kolich_client = aa_disb_kolich_client - cl_kol;

    aa_deliq_os = cast(aa_deliq_os * vrem as decimal(15,2));
    aa_deliq_perc = cast(aa_deliq_perc * vrem as decimal(15,2));

    aa_deliq_balance = cast(aa_deliq_balance * vrem as decimal(15,2));
    if (aa_deliq_kolich_client > 0) then
      aa_deliq_kolich_client = aa_deliq_kolich_client - cl_kol;

    aa_deliq_balance7 = cast(aa_deliq_balance7 * vrem as decimal(15,2));
    if (aa_deliq_kolich_client7 > 0) then
      aa_deliq_kolich_client7 = aa_deliq_kolich_client7 - cl_kol;

    aa_deliq_balance30 = cast(aa_deliq_balance30 * vrem as decimal(15,2));
    if (aa_deliq_kolich_client30 > 0) then
      aa_deliq_kolich_client30 = aa_deliq_kolich_client30 - cl_kol;

    aa_deliq_balance60 = cast(aa_deliq_balance60 * vrem as decimal(15,2));
    if (aa_deliq_kolich_client60 > 0) then
      aa_deliq_kolich_client60 = aa_deliq_kolich_client60 - cl_kol;

    aa_deliq_balance90 = cast(aa_deliq_balance90 * vrem as decimal(15,2));
    if (aa_deliq_kolich_client90 > 0) then
      aa_deliq_kolich_client90 = aa_deliq_kolich_client90 - cl_kol;

    aa_deliq_balance180 = cast(aa_deliq_balance180 * vrem as decimal(15,2));
    if (aa_deliq_kolich_client180 > 0) then
      aa_deliq_kolich_client180 = aa_deliq_kolich_client180 - cl_kol;

    aa_deliq_balance360 = cast(aa_deliq_balance360 * vrem as decimal(15,2));
    if (aa_deliq_kolich_client360 > 0) then
      aa_deliq_kolich_client360 = aa_deliq_kolich_client360 - cl_kol;

    aa_coll_plan_os = cast(aa_coll_plan_os * vrem as decimal(15,2));
    aa_coll_fact_os = cast(aa_coll_fact_os * vrem as decimal(15,2));

    aa_repay_total_os = 0;
    aa_repay_total_os = :aa_loan_amount - :aa_active_balance_e;
    aa_kf_name = (select kal_new_kf_name
                  from knaktlist
                  where knaktlist.is_delete = 0 and
                        :rep_date_end between kal_date and kal_date_end and
                        kal_from_kf_id = :k_from_kf_id);

    suspend;
  end

end
