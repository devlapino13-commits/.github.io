CREATE PROCEDURE MC_LOGIN_ACTION_VISIT(
    ACTION_CODE D_STR_CODE,
    ACTION_DATA D_STR_LONG)
RETURNS (
    RES_CODE D_STR_CODE,
    ERR_TEXT D_STR_255)
AS
declare variable db_id d_integer;
declare variable mcl_login d_str_code;
declare variable mcl_id d_id;
declare variable pin_code d_integer;
declare variable mcc_pin_code d_integer;
declare variable send_code_cnt d_integer;
declare variable check_code_cnt d_integer;
declare variable login_try_cnt d_integer;
declare variable mcl_pass_hash d_str_code;
declare variable c_birth_date d_date;
declare variable birth_date_str d_str_code;
declare variable c_id d_id;
declare variable login_cnt d_integer;
declare variable login_correct d_smallint;
declare variable max_check_cnt d_smallint;
declare variable visit_access smallint;
declare variable mcl_id_old id_type;
begin
  --max_check_cnt = 1000; -- На время тестирования
  max_check_cnt = 3;
  err_text = '';
  res_code = 'undefined';
  db_id = f_get_db_id();

  if (action_code = 'send_code') then
  begin
    mcl_login = f_json_str_get_value(action_data, 'login', '');
    if (coalesce(mcl_login, '') = '') then
    begin
      res_code = 'send_error';
      err_text = 'Отсутствует номер телефона';
      suspend;
      exit;
    end
    if (mcl_login not similar to '(79|992|996|998|994)[[:DIGIT:]]{9}' or (mcl_login similar to '(9960|9961|9963|9964)[[:DIGIT:]]{8}')) then
    begin
      res_code = 'send_error';
      err_text = 'Неправильный номер телефона';
      suspend;
      exit;
    end

    send_code_cnt = 0;
    select mcc_send_code_cnt
    from mc_check
    where mcc_login = :mcl_login
    into send_code_cnt;

    if (send_code_cnt >= max_check_cnt) then
    begin
      res_code = 'send_block';
      err_text = 'Превышен лимит отправок кода.';
      suspend;
      exit;
    end

-- На вермя тестирования отправки кода не будет, код 7777 для всех
    mcc_pin_code = 7777;
--    mcc_pin_code = trunc(rand() * 9000 + 1000);
--    execute procedure hp_send_quick_message(
--      'sms', :mcl_login, '',
--      iif(db_id <> 94401,
--        'М Булак: Ваш код для подтверждения: ',
--        'Mol Bulak: Tesdiq uchun kod: ') || mcc_pin_code,
--      '', 365, 1000000000000);

    update or insert into mc_check (mcc_login, mcc_send_code_cnt, mcc_pin_code)
    values (:mcl_login, :send_code_cnt + 1, :mcc_pin_code)
    matching (mcc_login);

    res_code = 'send_ok';
    suspend;
  end

  if (action_code = 'check_code') then
  begin
    mcl_login = f_json_str_get_value(action_data, 'login', '');
    pin_code = f_json_str_get_value(action_data, 'pin_code', -1);
    if (coalesce(mcl_login, '') = '') then
    begin
      res_code = 'check_error';
      err_text = 'Отсутствует номер телефона';
      suspend;
      exit;
    end
    if (mcl_login not similar to '(79|992|996|998|994)[[:DIGIT:]]{9}' or (mcl_login similar to '(9960|9961|9963|9964)[[:DIGIT:]]{8}')) then
    begin
      res_code = 'check_error';
      err_text = 'Неправильный номер телефона';
      suspend;
      exit;
    end
    if (coalesce(pin_code, -1) = -1) then
    begin
      res_code = 'check_error';
      err_text = 'Код не указан';
      suspend;
      exit;
    end

    check_code_cnt = 0;
    mcc_pin_code = 0;
    select mcc_check_code_cnt, mcc_pin_code
    from mc_check
    where mcc_login = :mcl_login
    into check_code_cnt, mcc_pin_code;

    -- Если mcc_pin_code=0, значит не отправляли, отвечаем блокировкой
    if (check_code_cnt >= max_check_cnt or mcc_pin_code = 0) then
    begin
      res_code = 'check_block';
      err_text = 'Превышен лимит проверок кода.';
      suspend;
      exit;
    end

    if (mcc_pin_code <> pin_code) then
    begin
      update or insert into mc_check (mcc_login, mcc_check_code_cnt)
      values (:mcl_login, :check_code_cnt + 1)
      matching (mcc_login);
      res_code = 'check_error';
      err_text = 'Коды не совпадают.';
      suspend;
      exit;
    end

    res_code = 'check_ok';
    update mc_check
    set mcc_check_code_cnt = 0
    where mcc_login = :mcl_login;

    suspend;
  end

  if (action_code = 'update_pass') then
  begin
    mcl_login = f_json_str_get_value(action_data, 'login', '');
    pin_code = f_json_str_get_value(action_data, 'pin_code', -1);
    mcl_pass_hash = f_json_str_get_value(action_data, 'password', null);
    if (coalesce(mcl_login, '') = '') then
    begin
      res_code = 'pass_error';
      err_text = 'Отсутствует номер телефона';
      suspend;
      exit;
    end
    if (mcl_login not similar to '(79|992|996|998|994)[[:DIGIT:]]{9}' or (mcl_login similar to '(9960|9961|9963|9964)[[:DIGIT:]]{8}')) then
    begin
      res_code = 'pass_error';
      err_text = 'Неправильный номер телефона';
      suspend;
      exit;
    end
    if (coalesce(pin_code, -1) = -1) then
    begin
      res_code = 'pass_error';
      err_text = 'Код не указан';
      suspend;
      exit;
    end
    if (coalesce(mcl_pass_hash, '') = '') then
    begin
      res_code = 'pass_error';
      err_text = 'Пустой пароль';
      suspend;
      exit;
    end

    check_code_cnt = 0;
    mcc_pin_code = 0;
    select mcc_check_code_cnt, mcc_pin_code
    from mc_check
    where mcc_login = :mcl_login
    into check_code_cnt, mcc_pin_code;

    -- Если mcc_pin_code=0, значит не отправляли, отвечаем блокировкой
    if (check_code_cnt >= max_check_cnt or mcc_pin_code = 0) then
    begin
      res_code = 'pass_block';
      err_text = 'Превышен лимит попыток смены пароля';
      suspend;
      exit;
    end

    if (mcc_pin_code <> pin_code) then
    begin
      update or insert into mc_check (mcc_login, mcc_check_code_cnt)
      values (:mcl_login, :check_code_cnt + 1)
      matching (mcc_login);
      res_code = 'pass_error';
      err_text = 'Коды не совпадают.';
      suspend;
      exit;
    end

    mcl_id = null;
    c_id = null;
    select mcl_id, mcl_from_c_id
    from mc_login
    where mcl_login = :mcl_login and
          mcl_is_delete = 0
    into mcl_id, c_id;

    if (mcl_id is null or c_id is null) then
    begin
      -- Создаем/обновляем пользователя, привязанного клиента нет
      update or insert into mc_login (mcl_login, mcl_pass_hash)
      values (:mcl_login, :mcl_pass_hash)
      matching (mcl_login);

      -- Обнуляем счетчики проверок кода и логина
      update mc_check
      set mcc_check_code_cnt = 0,
          mcc_login_try_cnt = 0
      where mcc_login = :mcl_login;

      res_code = 'pass_ok_new';
    end
    else
    begin
      -- Пользователь уже есть, и к нему привязан клиент
      -- обновляем ему пароль, но выставляем требование проверить ДР
      -- Если это другой клиент, то проверка ДР создаст для него дубликат логина
      update mc_login
      set mcl_pass_hash = :mcl_pass_hash
      where mcl_id = :mcl_id;

      update mc_check
      set mcc_check_birth_date = 1,
          mcc_check_code_cnt = 0,
          mcc_login_try_cnt = 0
      where mcc_login = :mcl_login;

      res_code = 'pass_ok_exist';
    end

    suspend;
    exit;
  end
  if (action_code = 'check_birth_date') then
  begin
    mcl_login = f_json_str_get_value(action_data, 'login', '');
    birth_date_str = f_json_str_get_value(action_data, 'birth_date', '');

    if (coalesce(mcl_login, '') = '') then
    begin
      res_code = 'birth_error';
      err_text = 'Отсутствует номер телефона';
      suspend;
      exit;
    end

    if (mcl_login not similar to '(79|992|996|998|994)[[:DIGIT:]]{9}' or (mcl_login similar to '(9960|9961|9963|9964)[[:DIGIT:]]{8}')) then
    begin
      res_code = 'birth_error';
      err_text = 'Неправильный номер телефона';
      suspend;
      exit;
    end

    begin
      c_birth_date = cast(birth_date_str as date);
      when any do
        c_birth_date = null;
    end
    if (c_birth_date is null) then
    begin
      res_code = 'birth_error';
      err_text = 'Неправильный формат даты рождения';
      suspend;
      exit;
    end

    c_id = null;
    select mcl_from_c_id
    from mc_login, mc_check
    where mcl_login = :mcl_login and
          mcc_login = :mcl_login and
          mcc_check_birth_date = 1
    into c_id;

    if (c_id is null) then
    begin
      res_code = 'birth_error';
      err_text = 'Проверка ДР не требуется';
      suspend;
      exit;
    end

    if (exists(select 1
               from clients
               where c_id = :c_id and
                     c_birth_date = :c_birth_date)) then
      res_code = 'birth_ok_exists';
    else
    begin -- Дата рождения не совпала, требуется создание нового пользователя

      -- Расчитываем суффикс для логина login_nnn
      select count(*)
      from mc_login
      where mcl_login starting :mcl_login
      into login_cnt;

      -- Переименовываем существующего пользователя в вид login_nnn
      select mcl_id, mcl_from_c_id
      from mc_login
      where mcl_login = :mcl_login
      into :mcl_id_old, :c_id;

      update mc_login
      set mcl_login = mcl_login || '_' || lpad(:login_cnt, 3, '0')
      where mcl_id = :mcl_id_old  --mcl_login = :mcl_login
      returning mcl_pass_hash
      into mcl_pass_hash;

      -- Создаем нового пользователя, под старым именем, без клиента
      insert into mc_login (mcl_login, mcl_from_c_id, mcl_pass_hash)
      values (:mcl_login, :c_id, :mcl_pass_hash)
      returning mcl_id
      into :mcl_id;

      update mc_visit_login
      set mvl_mcl_id = :mcl_id
      where mvl_mcl_id = :mcl_id_old;

      res_code = 'birth_ok_new';
    end

    update mc_check
    set mcc_check_birth_date = 0
    where mcc_login = :mcl_login;

    suspend;
    exit;
  end

  if (action_code = 'login') then
  begin
    mcl_login = f_json_str_get_value(action_data, 'login',  '');
    mcl_pass_hash = f_json_str_get_value(action_data, 'password', '');

    if (coalesce(mcl_login, '') = '') then
    begin
      res_code = 'login_error';
      err_text = 'Отсутствует номер телефона';
      suspend;
      exit;
    end
    if (mcl_login not similar to '(79|992|996|998|994)[[:DIGIT:]]{9}' or (mcl_login similar to '(9960|9961|9963|9964)[[:DIGIT:]]{8}')) then
    begin
      res_code = 'login_error';
      err_text = 'Неправильный номер телефона';
      suspend;
      exit;
    end
    if (coalesce(mcl_pass_hash, '') = '') then
    begin
      res_code = 'login_error';
      err_text = 'Пустой пароль';
      suspend;
      exit;
    end

    login_try_cnt = 0;
    select mcc_login_try_cnt
    from mc_check
    where mcc_login = :mcl_login
    into login_try_cnt;

    if (login_try_cnt >= max_check_cnt) then
    begin
      res_code = 'login_block';
      err_text = 'Превышен лимит ошибочных попыток входа';
      suspend;
      exit;
    end

    login_correct = 0;
    visit_access = 0;

    select 1, iif(mvl_mcl_id is null, 0, 1)
    from mc_login
    left join mc_visit_login on mcl_id = mvl_mcl_id
        and mvl_is_active = 1
    where mcl_login = :mcl_login and
          mcl_pass_hash = :mcl_pass_hash and
          mcl_is_delete = 0
    into login_correct, visit_access;
    login_correct = coalesce(login_correct, 0);

    if (visit_access = 0) then
    begin
        login_correct = 0;
        res_code = 'login_error';
        err_text = 'Отказано в доступе';
    end

    if (login_correct = 1) then
    begin
      if (exists(select 1
                 from mc_check
                 where mcc_login = :mcl_login and
                       mcc_check_birth_date = 1)) then
      begin
        res_code = 'login_check_birth_date';
        err_text = 'Требуется проверка даты рождения';
      end
      else
      begin
        res_code = 'login_ok';
        -- Чистим счетчики
        update mc_check
        set mcc_check_code_cnt = 0,
            mcc_login_try_cnt = 0
        where mcc_login = :mcl_login;
      end
    end
    else
    begin
      res_code = 'login_error';
      err_text = 'Неправильный номер телефона или пароль';
      update or insert into mc_check (mcc_login, mcc_login_try_cnt)
      values (:mcl_login, :login_try_cnt + 1)
      matching (mcc_login);
    end
    suspend;
    exit;
  end
end
