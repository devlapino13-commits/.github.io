CREATE PROCEDURE PRINT_CLIENT_STATEMENT_SVOD(
    RI_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    C_NAME_FIO D_STR_100,
    QR_CODE D_STR_LONG,
    C_INN D_STR_20,
    C_ADDRESS_YUR D_STR_255,
    C_ADDRESS_FACT D_STR_255,
    LAST_CELL_MAIN STR_20,
    DATE_BEGIN DATE,
    DATE_END DATE,
    KML_BANK_NO VARCHAR(50),
    KML_BANK_BIK ID_TYPE,
    KML_BANK_NAME VARCHAR(255),
    OB_DT_ALL DECIMAL(18,4),
    OB_KT_ALL DECIMAL(18,4),
    SALDO_BEGIN_DT DECIMAL(18,4),
    SALDO_BEGIN_KT DECIMAL(18,4),
    SALDO_END_DT DECIMAL(18,4),
    SALDO_END_KT DECIMAL(18,4),
    SS D_STR_255,
    COMPANY_NAME D_STR_100)
AS
declare variable adl_address_str d_str_255;
declare variable adc_address_type d_smallint;
declare variable curr_str varchar(1024);
declare variable ri_from_c_id d_id;
declare variable ri_from_reg_id d_integer;
declare variable ri_balance_id d_str_20;
declare variable ri_date_begin date;
declare variable ri_date_end date;
declare variable ri_account_no d_integer;
declare variable ri_saldo_begin_dt decimal(18,2);
declare variable ri_saldo_begin_kt decimal(18,2);
declare variable ri_saldo_end_dt decimal(18,2);
declare variable ri_saldo_end_kt decimal(18,2);
declare variable edl_id d_id;
begin

  select ri_from_c_id, ri_from_reg_id, ri_date_begin, ri_date_end, ri_balance_id, ri_account_no, ri_saldo_begin_dt,
         ri_saldo_begin_kt, ri_saldo_end_dt, ri_saldo_end_kt
  from reconciliation_info ri
  where ri.ri_id = :ri_id
  into :ri_from_c_id, :ri_from_reg_id, :ri_date_begin, :ri_date_end, :ri_balance_id, :ri_account_no, :ri_saldo_begin_dt,
       :ri_saldo_begin_kt, :ri_saldo_end_dt, :ri_saldo_end_kt;

  saldo_begin_dt = coalesce(:ri_saldo_begin_dt, 0);
  saldo_begin_kt = coalesce(:ri_saldo_begin_kt, 0);
  saldo_end_dt = coalesce(:ri_saldo_end_dt, 0);
  saldo_end_kt = coalesce(:ri_saldo_end_kt, 0);

  select sum(ob_dt_all), sum(ob_kt_all)
  from aarep_ob_dt_ob_kt(:ri_id)
  into :ob_dt_all, ob_kt_all;


  /*select sum(sta_ob_dt), sum(sta_ob_kt)
  from aarep_client_statement(:ri_from_reg_id, :ri_date_begin, :ri_date_end, :ri_balance_id, :ri_account_no, :ri_from_c_id)
  into :ob_dt_all, ob_kt_all;*/

  c_name_fio = '';
  c_inn = '';
  select cl.c_name_fio, cl.c_inn
  from clients cl
  where cl.c_id = :ri_from_c_id
  into :c_name_fio, :c_inn;

  last_cell_main = f_cont_info(0, :ri_from_c_id, 'cell_main', 'ok');

  kml_bank_no = '';
  kml_bank_bik = 0;
  kml_bank_name = '';

  select bc.bc_account_no, bc.bc_from_b_id, b_name
  from bankclient bc, banks
  where bc.bc_from_c_id = :ri_from_c_id and
        bc.bc_is_active = 1 and
        bc.bc_from_b_id = b_id rows 1
  into kml_bank_no, kml_bank_bik, kml_bank_name;

  c_address_yur = '';
  c_address_fact = '';
  for select adl.adl_address_str, adc.adc_address_type
      from addrclient adc
      inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
      where adc_from_c_id = :ri_from_c_id and
            adc.adc_address_type in (1, 2)
      into :adl_address_str, adc_address_type
  do
  begin
    if (adc_address_type = 1) then
      c_address_yur = :adl_address_str;
    if (adc_address_type = 2) then
      c_address_fact = :adl_address_str;
  end

  select cm_name
  from company
  into :company_name;

  ss = '';
  if (ri_saldo_end_dt = ri_saldo_end_kt) then
    ss = 'На ' || f_datetostr(ri_date_end) || ' год между ' || company_name || ' и ' || c_name_fio || ' нет задолженности';
  else
  begin
    if (ri_saldo_end_dt > 0) then
    begin
      select curr_str
      from fhp_currencystr(:ri_saldo_end_dt, 2)
      into :curr_str;

      ss = 'На ' || f_datetostr(ri_date_end) || ' год между ' || company_name || ' и ' || c_name_fio || ' задолженность составляет ' || cast(ri_saldo_end_dt as d_str_20) || ' (' || curr_str || ') в пользу ' || company_name;
    end
    else
    begin
      select curr_str
      from fhp_currencystr(:ri_saldo_end_kt, 2)
      into :curr_str;

      ss = 'На ' || f_datetostr(ri_date_end) || ' год между ' || company_name || ' и ' || c_name_fio || ' задолженность составляет ' || cast(ri_saldo_end_kt as d_str_20) || ' (' || curr_str || ') в пользу ' || c_name_fio;
    end
  end

  date_begin = :ri_date_begin;
  date_end = :ri_date_end;

  -- QR-код электронной подписи клиента. Текст пустой если не подписан
  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(edl_id, :ri_from_c_id);

    edl_id = f_json_str_get_value(add_params, 'edl_id', null);

    if (edl_id > 0) then
    begin
      qr_code = f_edoc_qr_by_edl(edl_id, :ri_from_c_id);
    end
  end
  suspend;
end
