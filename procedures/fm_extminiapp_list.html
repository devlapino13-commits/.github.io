CREATE PROCEDURE FM_EXTMINIAPP_LIST(
    UN_ID ID_TYPE,
    REPORT_TYPE INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    REG_ID D_INTEGER)
RETURNS (
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    RES_TEXT_1 D_STR_255,
    RES_TEXT_2 D_STR_10,
    RES_POS D_INTEGER)
AS
declare variable C_ID ID_TYPE;
declare variable CM_DATABASE_ID integer;
declare variable LO_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable C_BIRTH_DATE D_DATE;
declare variable COUNTRY D_STR_100;
declare variable JK_SYSTEM_CODE D_STR_100;
declare variable LO_FIO D_STR_100;
declare variable IS_YOUNG_LO D_BOOL;
declare variable HAS_ACCESS D_BOOL;
begin
  cm_database_id = f_get_db_id();

  obj_kind = f_objkind('extminiapp', 'id');

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exit;      

  lo_id = c_id;

  if (coalesce(:reg_id, 0) = 0) then
    exit;

  is_young_lo = null;
  has_access = null;

  select iif (datediff(month, ps.ps_date_begin, current_date) < 18, 1, 0)
  from hr_pstatus ps
  inner join hr_orders ho on ps.ps_from_ho_id = ho.ho_id and
      ho.ho_order_type = 1 and
      ho.ho_status = 1
  where ps.ps_from_c_id = :c_id and
      ps.ps_date_begin <= current_date
  order by ps_date_begin
  rows 1
  into is_young_lo;

  has_access = 0;
  if (exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.OnlineAppNewClientAccess')
             where has_access = 1)) then
  has_access = 1;

  for select res_c_id, iif(res_c_id = :lo_id, 200, 100)
      from hp_staff_changer_list(:un_id, 1)
      union
      select null, 0
      from rdb$database
      order by 2 desc
      into :c_id, res_pos
  do
  begin
    select jk.jk_system_code
    from hr_pstatus h
    inner join joblist jl on h.ps_from_jl_id = jl.jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind
    where current_date between h.ps_date_begin and h.ps_date_end and
          h.ps_is_working = 1 and
          jk_system_code <> 'JobSystem.Specialist_KOBD' and
          h.ps_from_c_id = :c_id
    into :jk_system_code;

    for select ema_id, ema_date, ema_from_c_id, ema_amount, ema_period, ema_office_id, ema_lo_id, ema_status,
               ema_from_tkt_id, ema_block_datetime
        from extminiapp
        inner join ticket t on t.tkt_id = ema_from_tkt_id
        inner join region_list(:reg_id, 1, 3, 1) reg on ema_office_id = reg.id_list
        where ema_status = 0 and
              ema_date between :date_begin and :date_end and
              ema_from_c_id is not null and
              ((:report_type = 1 and t.tkt_source <> 2) or -- если report_type = 1 (источник не КМП), тогда берем все заявки отправленные не с КМП
               (:report_type = 2 and t.tkt_source = 2)) and -- если report_type = 2 (источник КМП), тогда берем все заявки отправленные только с КМП
              (ema_lo_id = :c_id or
              :c_id is null and (:jk_system_code = 'JobSystem.Region_KM' or ema_lo_id is null))
        as cursor ema
    do
    begin
      obj_id = ema.ema_id;

      res_pos = 0;
      res_text_1 = '';
      res_text_2 = '';
      country = '';

      if (ema.ema_from_c_id is not null) then
      begin
        select c_name_fio, c_birth_date, cit.rd_name || ', ' || nat.rd_name
        from clients, refdata cit, refdata nat
        where c_id = ema.ema_from_c_id and
              c_citizen = cit.rd_id and
              c_nation = nat.rd_id
        into c_name_fio, c_birth_date, country;
      end
      else
      begin
        select wmat_name, wmat_birthday, cit.rd_name || ', ' || nat.rd_name
               --               wma_citizenship, wma_language
        from cc_website_miniapp, cc_website_miniapp_tmp, refdata cit, refdata nat
        where wma_from_tkt_id = ema.ema_from_tkt_id and
              wma_last_wmat_id = wmat_id and
              wmat_citizenship = cit.rd_id and
              wmat_language = nat.rd_id
        order by wma_id desc
        into c_name_fio, c_birth_date, country;
      end

      select c_name_fio
      from clients
      where c_id = ema.ema_lo_id
      into lo_fio;

      if (cm_database_id = 81001) then
      begin
          if (ema.ema_block_datetime is not null) then
          begin
              if (current_timestamp - ema.ema_block_datetime <= 0.5) then
    
                  if (ema.ema_lo_id is not null and ema.ema_lo_id <> lo_id) then
                  begin
                      res_text_1 = lo_fio || '<br>'|| ema.ema_block_datetime ||'<br><b>' || c_name_fio || '</b>,<br>' || f_datetostr(c_birth_date) || iif(c_birth_date is not null, ', ', '') || country;
                      res_text_2 = '&#9899;';
                  end
                  else
                  begin
                      res_text_1 = '<b>' || c_name_fio || '</b>,<br>' || f_datetostr(c_birth_date) || iif(c_birth_date is not null, ', ', '') || country;
                      res_text_2 = '&#10004;';
                  end
              if (current_timestamp - ema.ema_block_datetime between 0.5 and 1) then
    
                  if (ema.ema_lo_id is not null and ema.ema_lo_id <> lo_id) then
                  begin
                       res_text_1 = lo_fio || '<br>'|| ema.ema_block_datetime ||'<font color="#FF9933"><br><b>' || c_name_fio || '</b>,<br></font>' || f_datetostr(c_birth_date) || iif(c_birth_date is not null, ', ', '') || country;
                       res_text_2 = '&#128993;';
                  end
                  else
                  begin
                       res_text_1 = '<font color="#FF9933"><b>' || c_name_fio || '</b>,<br></font>' || f_datetostr(c_birth_date) || iif(c_birth_date is not null, ', ', '') || country;
                       res_text_2 = '&#10071;';
                  end
              if (current_timestamp - ema.ema_block_datetime >= 1) then
    
                  if (ema.ema_lo_id is not null and ema.ema_lo_id <> lo_id) then
                  begin
                      res_text_1 = lo_fio || '<br>' || ema.ema_block_datetime ||'<font color="#FF0000"><br><b>' || c_name_fio || '</b>,<br></font>' || f_datetostr(c_birth_date) || iif(c_birth_date is not null, ', ', '') || country;
                      res_text_2 = '&#128308;';
                  end
                  else
                  begin
                       res_text_1 = '<font color="#FF0000"><b>' || c_name_fio || '</b>,<br></font>' || f_datetostr(c_birth_date) || iif(c_birth_date is not null, ', ', '') || country;
                       res_text_2 = '&#8252;';
                  end
          end
          else
          res_text_1 = '<b>' || c_name_fio || '</b>,<br>' || f_datetostr(c_birth_date) || iif(c_birth_date is not null, ', ', '') || country;
      end
      else
          begin
          res_text_1 = '<b>' || c_name_fio || '</b>,<br>' || f_datetostr(c_birth_date) || iif(c_birth_date is not null, ', ', '') || country;
          if (ema.ema_lo_id = lo_id) then
            res_text_2 = '&#10004;';
          else
          if (ema.ema_lo_id is not null and
              ema.ema_lo_id <> lo_id) then
            res_text_2 = '&#9989;';
      end

      if (cm_database_id <> 81001 ) then
          suspend;
      if (cm_database_id = 81001 ) then
          begin
              if (has_access = 1 or is_young_lo = 1 or jk_system_code = 'JobSystem.Region_KM') then
                 suspend;
              if (is_young_lo = 0 and has_access = 0 and ema.ema_lo_id = lo_id) then
                 suspend;
          end
    end
  end
end
