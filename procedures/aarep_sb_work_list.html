CREATE PROCEDURE AAREP_SB_WORK_LIST(
    REP_DATE DATE)
RETURNS (
    COUNTER INTEGER,
    REG_NAME STR_100,
    KF_NAME STR_100,
    PAYED_DAY CURR_TYPE,
    DELINQ_DEBT CURR_TYPE,
    TOTAL_DEBT CURR_TYPE,
    DELINQ_DAYS INTEGER,
    A_COEFF CURR_TYPE,
    B_COEFF INTEGER,
    C_COEFF PERC_TYPE,
    SUM_VZYSKAN CURR_TYPE)
AS
declare variable k_id id_type;
declare variable zad_os curr_type;
declare variable sh_calc_perc curr_type;
declare variable sh_calc_fine curr_type;
declare variable sh_date date;
declare variable min_date date;
declare variable ks_state integer;
begin
  a_coeff = 0;
  b_coeff = 0;
  c_coeff = 0;
  sum_vzyskan = 0;
  select a_coeff, -- Сумма погаш. процентов и пени с просрочкой < 60 дней
         b_coeff, -- Кол-во клиентов, вставших в график или внесших погашения от 4 тыс. руб
         c_coeff, -- Показатель эффективности СВ за отчетный месяц
         sum_vzyskan -- сумма взысканий с просрочками более 60 дней
  from cb_ru_sb_2015_08(:rep_date)
  into a_coeff, b_coeff, c_coeff, sum_vzyskan;

  counter = 0;

  for select sba_from_k_id, ks_state, kal_new_kf_name, reg_name
      from (select sba.sba_from_k_id, ks.ks_state
            from kontstate ks
            inner join sb_akt sba on sba.sba_from_k_id = ks.ks_from_k_id
            where :rep_date between sba.sba_date_begin and sba.sba_date_end and
                  sba.sba_date_begin <= sba.sba_date_end and
                  sba_is_delete = 0 and
                  sba.sba_from_sbl_id = 85 -- СБ у Руслана Илалова
                  and
                  :rep_date between ks.ks_date_begin and ks.ks_date_end and
                  ks.ks_ol_is_delete = 0 and
                  ks.ks_state = 1 -- беру только активные контракты
            union all
-- <<
            select sba.sba_from_k_id, ks.ks_state
            from kontstate ks
            inner join sb_akt sba on sba.sba_from_k_id = ks.ks_from_k_id
            where :rep_date between sba.sba_date_begin and sba.sba_date_end and
                  sba.sba_date_begin <= sba.sba_date_end and
                  sba_is_delete = 0 and
                  sba.sba_from_sbl_id = 85 -- СБ у Руслана Илалова
                  and
                  ks.ks_date_begin = :rep_date and
                  ks.ks_ol_is_delete = 0 and
                  ks.ks_state = 2 -- беру закрытые на дату, чтобы отразит погашения за день
      ) t
      inner join kontract kont on kont.k_id = t.sba_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      inner join region reg on reg.reg_unicode = kal.kal_new_reg_id and
            kal.is_delete = 0 and
            :rep_date between kal.kal_date and kal.kal_date_end
      order by reg.reg_name
      into :k_id, :ks_state, :kf_name, :reg_name
  do
  begin
    payed_day = null;

    select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine)
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_from_c_id = 0 and
          ok.ok_ol_date = :rep_date and
          ok.ok_type in (4, 9) and
          ok.ok_ol_is_active = 1
    into :payed_day; --  Погашения за день

    payed_day = coalesce(payed_day, 0);

    delinq_debt = 0; -- Текущая задолженность
    total_debt = 0; -- Общая задолженность
    delinq_days = 0; -- Кол-во просроченных дней

    min_date = null;

    if (ks_state = 1) then
    begin
      for select zad_os, sh_calc_perc, sh_calc_fine, sh_date
          from aarep_make_zad_list(:k_id, :rep_date, 1)
          where sh_date is not null
          order by sh_date
          into zad_os, sh_calc_perc, sh_calc_fine, sh_date
      do
      begin
        if (sh_date < :rep_date) then
        begin
          if (((zad_os <> 0) or (sh_calc_perc <> 0)) and
              min_date is null) then
            min_date = sh_date;
          delinq_debt = delinq_debt + zad_os + sh_calc_perc + sh_calc_fine;
        end
        total_debt = total_debt + zad_os + sh_calc_perc + sh_calc_fine;
      end

      if (min_date is not null) then
        delinq_days = (rep_date - min_date);
    end
    counter = counter + 1;
    suspend;
  end

end
