CREATE PROCEDURE HP_TRANSFER_ONLINE_LOAN_TO_LO(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    NEED_REFRESH D_SMALLINT)
AS
declare variable c_id d_id;
declare variable reg_id d_integer;
declare variable kf_id d_id;
declare variable kal_from_ka_id d_id;
declare variable kal_number d_integer;
declare variable kal_new_no d_integer;
declare variable kol d_integer;
declare variable ke_code d_str_100;
declare variable kal_date d_date;
declare variable reg_id_prior d_integer;
declare variable kf_new_name d_str_100;
begin
  need_refresh = 1;
  select un_from_c_id
  from username un
  where un.un_id = :un_id
  into c_id;

  select kt.k_from_kf_id
  from kontract kt
  where kt.k_id = :k_id
  into kf_id;

  select ps.ps_from_reg_id
  from hr_pstatus ps
  where ps.ps_from_c_id = :c_id and
        current_date between ps.ps_date_begin and ps.ps_date_end and
        ps.ps_status = 1 and
        ps.ps_is_working = 1
  into reg_id;

  select kal_from_ka_id, kal_number, kal_new_reg_id, kal_date
  from knaktlist
  where kal_from_kf_id = :kf_id and
        kal_date_end >= '31.12.2999' and
        knaktlist.is_delete = 0
  into kal_from_ka_id, kal_number, reg_id_prior, kal_date;

  if (reg_id is null) then
  begin
    res_text = 'Не определен офис кредитной папки';
    need_refresh = 0;
    suspend;
    exit;
  end

  if (reg_id <> reg_id_prior) then
  begin
    res_text = 'Передача папки возможна только внутри офиса';
    need_refresh = 0;
    suspend;
    exit;
  end 

  kol = 0;
  for select ke_code
      from krexpert
      where ke_from_p_id = :c_id and
            ke_from_reg_id = :reg_id and
            ke_status = 1 and
            ke_date_end >= '31.12.2999' and
            ke_date <= current_date
      into :ke_code
  do
  begin
    kol = kol + 1;
  end

  if (kol <> 1) then
  begin
    if (kol = 0) then
      res_text = 'У выбранного сотрудника отсутствует код СВЗ в этом офисе';
    else
      res_text = 'У выбранного сотрудника имеется ' || :kol || ' кода СВЗ в этом офисе. Сначала наведите порядок в справочнике СВЗ';
    suspend;
    exit;
  end

  select kal_new_no
  from knaktlist
  where kal_new_officer_code = :ke_code and
        kal_new_reg_id = :reg_id and
        kal_new_kf_officer = :c_id and
        knaktlist.is_delete = 0
  order by kal_new_officer_code desc, kal_new_no desc
  rows 1
  into :kal_new_no;
  if (kal_new_no is null) then
    kal_new_no = 0;
  kal_new_no = kal_new_no + 1;

  kf_new_name = ke_code || '-' || kal_new_no;

  update knaktlist
  set kal_new_kf_officer = :c_id,
      kal_new_kf_name = :kf_new_name,
      kal_new_no = :kal_new_no,
      kal_new_officer_code = :ke_code
  where (kal_from_ka_id = :kal_from_ka_id) and
        (kal_number = :kal_number) and
        (kal_from_kf_id = :kf_id);

  execute procedure knaktlist_update(:kf_id, kal_date);
  res_text = 'Операция прошла успешно';

  suspend;
end
