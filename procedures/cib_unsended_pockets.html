CREATE PROCEDURE CIB_UNSENDED_POCKETS(
    CIB_CODE D_STR_100 = 'cib.equifax')
RETURNS (
    POCKET_DATE D_DATE)
AS
declare variable CIB_ID D_INTEGER;
begin
  cib_id = f_get_rd_id(:cib_code);

  if (cib_code = 'cib.equifax') then
    pocket_date = cast('01.01.'||extract(year from current_date) as date);
  else
  if (cib_code = 'cib.pkb' or
      cib_code = 'cib.gkb') then
    pocket_date = '17.09.2025';
  else
  if (cib_code = 'cib.crif') then
    pocket_date = '24.12.2025';
  else
    pocket_date = '01.01.2019';

  while (pocket_date < current_date -1) do
  begin
   if (cib_code = 'cib.equifax') then
   begin
       if (not exists(
            select *
            from cib_pocket cp
            where cp.cp_pocket_type containing '.kont.'
            and cp.cp_cib_id = :cib_id
            and cp.cp_pocket_kib_status = 'Processed'
            and :pocket_date between cp.cp_date_begin and cp.cp_date_end)) then
       begin
            suspend;
       end
   end
   -- RFCKG-4913
   if (cib_code = 'cib.crif') then
   begin
       if (not exists(
            select *
            from cib_pocket cp
            where cp.cp_pocket_status > 0
            and cp.cp_cib_id = :cib_id
            and :pocket_date between cp.cp_date_begin and cp.cp_date_end)) then
       begin
            suspend;
       end
   end
   if (cib_code = 'cib.ishenim') then
   begin
       if (not exists(
            select *
            from cib_pocket cp
            where cp.cp_pocket_status = 2
            and cp.cp_cib_id = :cib_id
            and :pocket_date between cp.cp_date_begin and cp.cp_date_end)) then
       begin
            suspend;
       end
   end
   pocket_date = pocket_date + 1;
  end
  -- RFCKZ-96
  if (cib_code = 'cib.pkb' or
      cib_code = 'cib.gkb') then
  begin
    while (pocket_date < current_date) do
    begin
       if (not exists(
            select *
            from cib_pocket cp
            where cp.cp_cib_id = :cib_id
            and cp.cp_pocket_status = 2
            and :pocket_date between cp.cp_date_begin and cp.cp_date_end)) then
            suspend;
       pocket_date = pocket_date + 1;
    end
  end
end
