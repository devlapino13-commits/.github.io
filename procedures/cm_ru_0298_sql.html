CREATE PROCEDURE CM_RU_0298_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable is_documented d_bool;
declare variable res_code d_smallint;
declare variable res_text d_str_long;
declare variable mac_id d_id;
begin

  for select mcl.mac_id,
             maci.maci_is_documented
  from miniappclient mcl
  inner join miniappclient_income maci on maci.maci_from_mac_id = mcl.mac_id
  inner join kontract k on k.k_from_ma_app_id = mcl.mac_from_ma_id and
                           k.k_id = :k_id
  where mcl.mac_amount > 0
  into :mac_id,
       :is_documented
  do
  begin
    if (is_documented = 1) then
    begin
      select res_code, res_text
      from dpd_anketa_check('miniappclient_income', :mac_id, 0)
      into res_code, res_text;

      if (res_code = -1) then
      begin
        res_error_text = :res_text;
        res_error_is_critical = 2;
        suspend;
      end
    end
  end
end
