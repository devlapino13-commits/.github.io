CREATE PROCEDURE FIND_PETTRANS_TEMP
RETURNS (
    PET_ID INTEGER,
    PET_NAME VARCHAR(255),
    PET_LABEL VARCHAR(100),
    PET_IS_RENT INTEGER,
    PET_NORMA_YEDIZ INTEGER,
    PET_NORMA_KM INTEGER,
    PED_ID INTEGER,
    PED_DATE DATE,
    PED_REGION_PLACE_ID INTEGER,
    PED_FROM_C_ID DECIMAL(18,0),
    PED_IS_ACTIVE INTEGER,
    PED_OFFICE VARCHAR(100),
    PED_MATOTVET VARCHAR(100))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select pet_id, pet_name, pet_label, pet_is_rent, pet_norma_yediz, pet_norma_km
      from pettrans
      into :pet_id, :pet_name, :pet_label, :pet_is_rent, :pet_norma_yediz, :pet_norma_km
  do
  begin
    ped_id = null;
    ped_date = null;
    ped_region_place_id = null;
    ped_from_c_id = null;
    ped_is_active = null;
    ped_office = null;
    ped_matotvet = null;
    for select ped_id, ped_date, ped_region_place_id, ped_from_c_id, ped_is_active, reg_code || ' ' || reg_name,
               c_name_fio
        from petdata, region, clients
        where ped_from_pet_id = :pet_id and
              reg_unicode = ped_region_place_id and
              c_id = ped_from_c_id
        into :ped_id, :ped_date, :ped_region_place_id, :ped_from_c_id, :ped_is_active, :ped_office, :ped_matotvet
    do
    begin
      suspend;
    end
    if (ped_id is null) then
      suspend;
  end
end
