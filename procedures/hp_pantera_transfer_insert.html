CREATE PROCEDURE HP_PANTERA_TRANSFER_INSERT(
    TRANSFER_DATE DATE,
    PANT_ID_FROM ID_TYPE,
    PANT_ID_TO ID_TYPE,
    TRANSFER_REASON_CODE STR_100)
AS
declare variable delinq_days integer;
declare variable k_id id_type;
declare variable transfer_reason integer;
declare variable delinq_os curr_type;
declare variable delinq_perc curr_type;
declare variable delinq_fine curr_type;
declare variable zad_os_total curr_type;
declare variable zad_perc_total curr_type;
declare variable sh_date date;
declare variable zad_os curr_type;
declare variable sh_calc_perc curr_type;
declare variable sh_calc_fine curr_type;
declare variable pt_id d_id;
declare variable close_state d_smallint;
declare variable delinq_date d_date;
declare variable out_pcm_id d_integer;
declare variable input_pcm_id d_integer;
declare variable db_id d_integer;
begin
  /*  igsuev
    Процедура предназаначена для фиксирвоания передач, блокировок, разблокировок групп по проектам КЦ
    (Сирена, Пантера и аналогичным им проектам)

    таблица PANTERA_TRANSFER 
    */

  db_id = f_get_db_id();
  if (coalesce(transfer_reason_code, '') <> '') then
    transfer_reason = f_get_rd_id(:transfer_reason_code);

  if (pant_id_from is not null) then
  begin
    select pantk.pant_from_k_id
    from pantera_kontracts pantk
    where pantk.pant_id = :pant_id_from
    into :k_id;
  end
  else
  begin
    select pantk.pant_from_k_id
    from pantera_kontracts pantk
    where pantk.pant_id = :pant_id_to
    into :k_id;
  end

  delinq_days = null;

  select :transfer_date - oks.oks_delinq_date,
         oks.oks_close_state
  from oksummary oks
  where :transfer_date between oks.oks_date_begin and oks.oks_date_end and
        oks.oks_from_k_id = :k_id
  into :delinq_days,
       :close_state;

  if (close_state = 1) then
  begin
    delinq_date = null;

    select oks_delinq_date
    from oksummary
    where oks_from_k_id = :k_id and
          oks_delinq_date < '31.12.9999'
    order by oks_date_begin desc
    rows 1
    into delinq_date;

    if (delinq_date is not null) then
      delinq_days = :transfer_date - delinq_date;
  end

  delinq_days = coalesce(delinq_days, 0);
  if (delinq_days < 0) then
    delinq_days = 0;

  delinq_os = 0;
  delinq_perc = 0;
  delinq_fine = 0;

  zad_os_total = 0;
  zad_perc_total = 0;

  for select sh_date, zad_os, sh_calc_perc, sh_calc_fine
      from aarep_make_zad_list(:k_id, :transfer_date, 1)
      into :sh_date, :zad_os, :sh_calc_perc, :sh_calc_fine
  do
  begin
    if (coalesce(sh_date, '01.01.1900') < transfer_date) then -- просрочка
    begin
      delinq_os = delinq_os + zad_os;
      delinq_perc = delinq_perc + sh_calc_perc;
      delinq_fine = delinq_fine + sh_calc_fine;
    end

    -- общая задолжность по кредиту
    zad_os_total = zad_os_total + zad_os;
    zad_perc_total = zad_perc_total + sh_calc_perc;
  end

  if (db_id in (39801, 41701)) then
  begin
    if (f_tmp_const('kg_make_zad_list_okf_show') = 0) then
    begin
        for select coalesce(sum(okf.oko_os + okf.oko_sh_os), 0),
                   coalesce(sum(okf.oko_perc + okf.oko_sh_perc), 0),
                   coalesce(sum(okf.oko_other + okf.oko_sh_other), 0)
        from operkroff okf
        where okf.oko_from_k_id = :k_id
        and okf.oko_ol_is_active = 1
        and okf.oko_ol_date <= :transfer_date
        into :zad_os,
             :sh_calc_perc,
             :sh_calc_fine
        do
        begin
            delinq_os = delinq_os + zad_os;
            delinq_perc = delinq_perc + sh_calc_perc;
            delinq_fine = delinq_fine + sh_calc_fine;

            zad_os_total = zad_os_total + zad_os;
            zad_perc_total = zad_perc_total + sh_calc_perc;
        end
    end
  end

  -- фиксирую передачу
  insert into pantera_transfer (pt_out_pant_id, pt_input_pant_id, pt_delinq_os, pt_delinq_perc, pt_delinq_fine,
                                pt_zad_os, pt_zad_perc, pt_delinq_days, pt_reason_rd_id, pt_date)
  values (:pant_id_from, :pant_id_to, :delinq_os, :delinq_perc, :delinq_fine, :zad_os_total, :zad_perc_total,
          :delinq_days, :transfer_reason, :transfer_date)
  returning pt_id
  into :pt_id;

  out_pcm_id = (select pkm.pkc_from_pcm_id from pantera_kontracts_com pkm where pkm.pkc_from_pant_id = :pant_id_from);
  input_pcm_id = (select pkm.pkc_from_pcm_id from pantera_kontracts_com pkm where pkm.pkc_from_pant_id = :pant_id_to);

  if ((out_pcm_id is not null) or (input_pcm_id is not null)) then
  begin
     insert into pantera_transfer_com (ptc_from_pt_id, ptc_out_pcm_id, ptc_input_pcm_id)
     values (:pt_id, :out_pcm_id, :input_pcm_id);
  end
end
