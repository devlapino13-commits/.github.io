CREATE PROCEDURE FM_CLIENT_DOC_LIST(
    UN_ID D_ID,
    C_ID D_ID)
RETURNS (
    CDS_ID D_INTEGER,
    BLS1_ID D_ID,
    BLS1_STATUS D_SMALLINT,
    BLS1_EXT D_STR_LANG,
    BLS2_ID D_ID,
    BLS2_STATUS D_SMALLINT,
    BLS2_EXT D_STR_LANG,
    DOC_NAME D_STR_255,
    DOC_DETAIL1 D_STR_255,
    DOC_DETAIL2 D_STR_255,
    IS_OLD_DOC D_BOOL)
AS
declare variable lang d_str_lang;
declare variable need_photo1 d_bool;
declare variable need_photo2 d_bool;
declare variable pk_id d_integer;
declare variable pk_id_prior d_integer;
declare variable db_id d_integer;
declare variable doc_datetime d_datetime;
declare variable is_manager d_bool;
declare variable is_main_doc d_bool;
declare variable doc_no d_str_code;
declare variable doc_date_begin d_date;
declare variable pdt_need_date_begin d_integer;
declare variable doc_date_end d_date;
declare variable acl_status d_integer;
declare variable is_actual_photo d_bool;
declare variable photo_count d_integer;
declare variable pdt_code d_str_100;
declare variable new_bls_1 d_id;
declare variable new_bls_2 d_id;
declare variable need_show_temp d_bool;
declare variable is_photo_blocked d_bool;
begin
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  select un_id
  from hp_test_un_id(:un_id)
  into :un_id;

  need_show_temp = 1;

  new_bls_1 = 0;
  new_bls_2 = 0;
  select coalesce(pte_blob_pass1_id, 0), coalesce(pte_blob_pass2_id, 0)
  from perstemp
  where pte_from_c_id = :c_id
  into new_bls_1, new_bls_2;

  if (row_count = 0) then
    need_show_temp = 0;

  is_manager = 0;
  if (exists(
      select *
      from username, hr_pstatus, joblist, jobkind
      where un_id = :un_id
        and ps_from_c_id = un_from_c_id
        and ps_date_begin <= current_date
        and ps_date_end >= current_date
        and ps_is_working = 1
        and ps_status = 1
        and ps_from_jl_id = jl_id
        and jl_kind = jk_id
        and ((:db_id in (39801, 41701) and jk_system_code in ('JobSystem.Region_KM' , 'JobSystem.Region_Director')) or
            (:db_id in (81001, 94401) and jk_system_code = 'JobSystem.Region_KM')))
       ) then
    is_manager = 1;

  --RFCRU-5969 Если "Специалист" и "Ведущий специалист", блокируем доступ к просмотру фото клиента 
  is_photo_blocked = 0;
  if (:db_id = 81001) then
  begin
    if (exists(
        select *
        from username
        inner join hr_pstatus on ps_from_c_id = un_from_c_id
        inner join joblist on ps_from_jl_id = jl_id
      where un_id = :un_id
        and ps_date_begin <= current_date
        and ps_date_end >= current_date
        and ps_is_working = 1
        and ps_status = 1
        and jl_id in (31, 267, 279))  -- Специалист, Специалист, Ведущий специалист
      ) then
    is_photo_blocked = 1;
  end

  doc_name = f_lang(lang, 'Фото клиента');
  is_old_doc = 0;
  cds_id = 0;
  bls2_id = 0;
  bls2_status = 0;
  doc_detail2 = '';
  for select c_blob_photo_id, bls_datetime, coalesce(right(bls_filename, 3), '')
      from clients, blobstore
      where c_id = :c_id
        and c_blob_photo_id = bls_id
      into bls1_id, doc_datetime, bls1_ext
  do
  begin
    bls1_status = 1;
    if (bls1_id > 0) then
    begin
      doc_detail1 = f_format('%1: %2', f_lang(lang, 'Загружено'), f_datetimetostr(doc_datetime));
    end
    else
    begin
      bls1_id = 0;
      bls1_status = -1;
      doc_detail1 = f_lang(lang, 'Не загружено');
    end

    -- RFCRU-5969 блокировка просмотра фото клиента для Специалистов/Вед.специалистов
    if (is_photo_blocked = 1 and bls1_status = 1) then
    begin
      bls1_status = 2;
      bls1_id = 0;
    end
    suspend;
  end
  --Для ОРИВ показываем только фото клиента
  if (exists(select *
               from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessOriv')
               where has_access = 1)) then
                exit;

  -- Документы на оцифровке
  for select drl_id,
             pk_id,
             pk_name,
             coalesce(drl_from_bls_1, 0),
             coalesce(drl_from_bls_2, 0),
             drl_create_datetime,
             acl_status,
             coalesce(right(b1.bls_filename, 3), ''),
             coalesce(right(b2.bls_filename, 3), '')
  from docrecoglist
    inner join docrecogmain on drl_from_drm_id = drm_id
    inner join actionlist on drl_from_acl_id = acl_id
    inner join passkind on drl_from_pk_id = pk_id
    left join blobstore b1 on b1.bls_id = drl_from_bls_1
    left join blobstore b2 on b2.bls_id = drl_from_bls_2
  where drl_from_c_id = :c_id
    and drm_code not containing '.phonechange'
    and acl_status in (0, 1)
  into cds_id,
       pk_id,
       doc_name,
       bls1_id,
       bls2_id,
       doc_datetime,
       acl_status,
       bls1_ext,
       bls2_ext
  do
  begin
    if (bls1_id = new_bls_1 and bls2_id = new_bls_2) then
      need_show_temp = 0;

    doc_detail1 = f_format('%1: %2', f_lang(lang, 'Загружен'), f_datetimetostr(doc_datetime));
    doc_detail2 = f_lang(lang, 'На оцифровке') || ': ';
    if (acl_status = 0) then
      doc_detail2 = doc_detail2 || f_lang(lang, 'В ожидании');
    else
      doc_detail2 = doc_detail2 || f_lang(lang, 'Принят в обработку');

    select drm_has_page1, drm_has_page2
    from hp_get_drm_has_pages(:pk_id)
    into :need_photo1, :need_photo2;

    -- Первая страница
    if (need_photo1 = 1) then
      if (bls1_id > 0) then
        bls1_status = 1;
      else
      begin
        bls1_status = -1;
        bls1_id = 0;
      end
    else
      bls1_status = 0;

    -- Вторая страница
    if (need_photo2 = 1) then
      if (bls2_id > 0) then
        bls2_status = 1;
      else
      begin
        bls2_status = -1;
        bls2_id = 0;
      end
    else
      bls2_status = 0;

    suspend;
  end

  -- Остальные документы
  pk_id_prior = 0;
  is_actual_photo = 0;
  photo_count = 0;
  for select cds_id,
             pk_id, pk_name,
             coalesce(cds_doc_blob_1, 0),
             coalesce(cds_doc_blob_2, 0),
             cds_doc_no,
             cds_doc_date_begin,
             cds_doc_date_end,
             cds_is_main_doc,
             cds_create_datetime,
             pdt_need_date_begin,
             pdt_code,
             coalesce(right(b1.bls_filename, 3), ''),
             coalesce(right(b2.bls_filename, 3), '')
  from clientdocs
    inner join passkind on cds_from_pk_id = pk_id
    inner join passdoctype on pk_from_pdt_id = pdt_id
    left join blobstore b1 on cds_doc_blob_1 = b1.bls_id
    left join blobstore b2 on cds_doc_blob_2 = b2.bls_id
  where cds_from_c_id = :c_id
    and cds_is_delete = 0
    -- В РФ Андеррайтеры закрепляют, СВЗ в МП видеть не должны
    and (pdt_code not in ('credit_history', 'fssp', 'certificate_name_change') or :db_id <> 81001)
    and pdt_code not containing '_phone_change'
    and pk_id > 0
  order by pk_id, cds_create_datetime desc
  into cds_id,
       pk_id,
       doc_name,
       bls1_id,
       bls2_id,
       doc_no,
       doc_date_begin,
       doc_date_end,
       is_main_doc,
       doc_datetime,
       pdt_need_date_begin,
       pdt_code,
       bls1_ext,
       bls2_ext
  do
  begin
    if (bls1_id = new_bls_1 and bls2_id = new_bls_2) then
      need_show_temp = 0;

    -- Показываем только последний документ каждого вида, по pk_id
    if (pk_id <> pk_id_prior or
        (db_id in (39801, 41701) and :pdt_code = 'foto_on_date_disb' and photo_count < 4)) then --Выводит актульное фото на дату выдачи и 3 аннулированных фото на дату выдачи для сравнения
    begin
      select drm_has_page1, drm_has_page2
      from hp_get_drm_has_pages(:pk_id)
      into :need_photo1, :need_photo2;

      -- Первая страница
      if (need_photo1 = 1) then
        if (bls1_id > 0) then
          bls1_status = 1;
        else
        begin
          bls1_status = -1;
          bls1_id = 0;
        end
      else
        bls1_status = 0;

      -- Вторая страница
      if (need_photo2 = 1) then
        if (bls2_id > 0) then
          bls2_status = 1;
        else
        begin
          bls2_status = -1;
          bls2_id = 0;
        end
      else
        bls2_status = 0;

      -- Общее для документа
      is_old_doc = 0;
      if (doc_date_end < current_date) then
        is_old_doc = 1;

      -- BOARD-5097
      if (db_id in (39801, 41701) and
        :pdt_code = 'foto_on_date_disb') then
      begin
        photo_count = photo_count + 1;
        if (is_actual_photo = 0) then
        begin
          is_actual_photo = 1;
          is_old_doc = 0;
        end
        else
          is_old_doc = 1;  -- Все фото кроме актуального аннулируются (зачеркиваются)
      end
      -- Конец BOARD-5097

      doc_no = coalesce(doc_no, '');
      if (doc_no = '') then doc_no = f_lang(lang, 'б/н');

      if (pdt_need_date_begin = 1) then
        doc_detail1 = f_format(f_lang(lang,'№ %1, выдан: %2'), doc_no, f_datetostr(doc_date_begin));
      else
        doc_detail1 = f_format('№ %1', doc_no);

      doc_detail2 = f_format('%1: %2', f_lang(lang, 'Добавлен'), f_datetimetostr(doc_datetime));

      -- Если НЕ менеджер, блокируем доступ к паспорту (основному документу)
      if (is_main_doc = 1 and is_manager = 0) then
      begin
        if (bls1_status = 1) then
        begin
          bls1_status = 2;
          bls1_id = 0;
        end
        if (bls2_status = 1) then
        begin
          bls2_status = 2;
          bls2_id = 0;
        end
      end

      suspend;
    end
    pk_id_prior = pk_id;
  end

  if (need_show_temp = 1) then
  begin
    for select coalesce(pte_pass_drl_id, 0), pte_doc_type, pk_name,
               coalesce(pte_blob_pass1_id, 0), coalesce(pte_blob_pass2_id, 0),
               coalesce(b1.bls_datetime, pte_datetime_added),
               coalesce(right(b1.bls_filename, 3), ''),
               coalesce(right(b2.bls_filename, 3), '')
    from perstemp
         inner join passkind on pte_doc_type = pk_id
            left join blobstore b1 on b1.bls_id = pte_blob_pass1_id
            left join blobstore b2 on b2.bls_id = pte_blob_pass2_id
    where pte_from_c_id = :c_id
    into cds_id,
         pk_id,
         doc_name,
         bls1_id,
         bls2_id,
         doc_datetime,
         bls1_ext,
         bls2_ext
    do
    begin
      doc_detail1 = f_format('%1: %2', f_lang(lang, 'Загружен'), f_datetimetostr(doc_datetime));
      doc_detail2 = f_lang(lang, 'Добавлен') ;
    end
  end
end
