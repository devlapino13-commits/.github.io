CREATE PROCEDURE UXP$UXINDEX_INSERT_VIRTUAL(
    SOURCE_NAME D_STR_CODE,
    SOURCE_FIELDS D_STR_CODE,
    FK_SOURCE D_STR_CODE)
AS
declare variable tmp_source d_str_code;
declare variable fk_index d_str_100;
declare variable field_count d_integer;
begin
  fk_index = '';

  if (coalesce(source_name, '') = '') then
    exception e_default 'Source_name is empty';

  if (coalesce(source_fields, '') = '') then
    exception e_default 'Source_fields is empty';

  if (coalesce(fk_source, '') = '') then
    exception e_default 'FK_Source is empty';

  select count(*)
  from uxp$split_str(:source_fields, ';')
  into field_count;

  select uix_index
  from ux$index
  where uix_source = :fk_source and
        (uix_pk = 1 or uix_uniq = 1) and
        uix_count = :field_count
  order by uix_pk desc
  rows 1
  into fk_index;

  if (fk_index = '') then
  begin
    select lower(trim(i.rdb$index_name))
    from rdb$indices i
    where rdb$relation_name = upper(:fk_source) and
          i.rdb$unique_flag = 1 and
          rdb$segment_count = :field_count
    order by i.rdb$index_id
    rows 1
    into fk_index;

    if (fk_index <> '') then
    begin
      select source_name
      from uxp$index_load(:fk_index)
      into :tmp_source;

      if (tmp_source <> fk_source) then
        exception e_default f_format('Incorrect FK source name: "%1" and "%2"', upper(fk_source), upper(tmp_source));
    end else
      exception e_default f_format('PK or Unique field(s) not found. Source: %1', upper(fk_source));
  end

  insert into ux$index (uix_source, uix_fields, uix_fk, uix_virtual, uix_fk_source, uix_fk_index, uix_state)
  values (:source_name, :source_fields, 1, 1, :fk_source, :fk_index, 0);
end
