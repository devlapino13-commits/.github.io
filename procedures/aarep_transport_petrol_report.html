CREATE PROCEDURE AAREP_TRANSPORT_PETROL_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PED_ID INTEGER,
    PED_DATE DATE,
    PED_REGION_PLACE_ID INTEGER,
    PED_REGION_BALANCE_ID INTEGER,
    PED_FROM_C_ID DECIMAL(18,0),
    PED_EXPENSE_SUMMER DECIMAL(18,4),
    PED_EXPENSE_WINTER DECIMAL(18,4),
    PED_EXPENSE_SUMMER_MOUNT DECIMAL(18,4),
    PED_EXPENSE_WINTER_MOUNT DECIMAL(18,4),
    PED_ACC_BALANCE VARCHAR(20),
    PED_ACC_NUMBER INTEGER,
    PED_IS_ACTIVE INTEGER,
    PET_ID INTEGER,
    PET_NAME VARCHAR(255),
    PET_LABEL VARCHAR(50),
    PET_IS_RENT INTEGER,
    PET_NORMA_YEDIZ INTEGER,
    PET_NORMA_KM INTEGER,
    PED_CLIENT_FIO VARCHAR(100),
    PED_PLACE_NAME VARCHAR(100),
    PED_SALDO_LITER DECIMAL(18,4),
    PED_SALDO_SUMMA DECIMAL(18,4),
    PED_FILL_LITER DECIMAL(18,4),
    PED_FILL_LITER_TALON DECIMAL(18,4),
    PED_FILL_SUMMA DECIMAL(18,4),
    PET_NORMA_NAME VARCHAR(20),
    PED_WRITEOFF_LITER DECIMAL(18,4),
    PED_WRITEOFF_SUMMA DECIMAL(18,4),
    PED_WRITEOFF_KM DECIMAL(18,4),
    PED_SALDO_END_LITER DECIMAL(18,4),
    PED_SALDO_END_SUMMA DECIMAL(18,4),
    PET_BRANCH_PLACE_NAME VARCHAR(255))
AS
begin
  if (rep_region_id is null) then
    rep_region_id = 1000000;
  if (rep_region_id = 0) then
    rep_region_id = 1000000;
  for select ped_id, ped_date, ped_region_place_id, ped_region_balance_id, ped_from_c_id, ped_expense_summer,
             ped_expense_winter, ped_expense_summer_mount, ped_expense_winter_mount, ped_acc_balance, ped_acc_number,
             ped_is_active, pet_id, pet_name, pet_label, pet_is_rent, pet_norma_yediz, pet_norma_km, ped_client_fio,
             ped_place_name, sum(ped_saldo_liter), sum(ped_saldo_summa), sum(ped_fill_liter), sum(ped_fill_liter_talon),
             sum(ped_fill_summa), pet_norma_name, sum(ped_writeoff_liter), sum(ped_writeoff_summa),
             sum(ped_writeoff_km), r2.reg_code || ' ' || r2.reg_name
      from aarep_transport_petrol_temp(:rep_region_id, 0, :rep_date_begin, :rep_date_end), region r1, region r2
      where ped_region_place_id = r1.reg_unicode and
            r2.reg_unicode = r1.parent
      group by ped_id, ped_date, ped_region_place_id, ped_region_balance_id, ped_from_c_id, ped_expense_summer, ped_expense_winter, ped_expense_summer_mount, ped_expense_winter_mount, ped_acc_balance, ped_acc_number, ped_is_active, pet_id, pet_name, pet_label, pet_is_rent, pet_norma_yediz, pet_norma_km, ped_client_fio, ped_place_name, pet_norma_name, r2.reg_code, r2.reg_name

      having sum(ped_saldo_liter) <> 0 or sum(ped_saldo_summa) <> 0 or sum(ped_fill_liter) <> 0 or sum(ped_fill_liter_talon) <> 0 or sum(ped_fill_summa) <> 0
      into :ped_id, :ped_date, :ped_region_place_id, :ped_region_balance_id, :ped_from_c_id, :ped_expense_summer,
           :ped_expense_winter, :ped_expense_summer_mount, :ped_expense_winter_mount, :ped_acc_balance, :ped_acc_number,
           :ped_is_active, :pet_id, :pet_name, :pet_label, :pet_is_rent, :pet_norma_yediz, :pet_norma_km,
           :ped_client_fio, :ped_place_name, :ped_saldo_liter, :ped_saldo_summa, :ped_fill_liter, :ped_fill_liter_talon,
           :ped_fill_summa, :pet_norma_name, :ped_writeoff_liter, :ped_writeoff_summa, :ped_writeoff_km,
           :pet_branch_place_name
  do
  begin
    ped_saldo_end_liter = :ped_saldo_liter + :ped_fill_liter + :ped_fill_liter_talon - :ped_writeoff_liter;
    ped_saldo_end_summa = :ped_saldo_summa + :ped_fill_summa - :ped_writeoff_summa;
    suspend;
  end
end
