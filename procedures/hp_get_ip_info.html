CREATE PROCEDURE HP_GET_IP_INFO(
    IP_ADDRESS D_STR_CODE)
RETURNS (
    IPR_IP_ADDRESS D_STR_CODE,
    IPR_COUNTRY D_STR_CODE,
    IPR_REGION D_STR_CODE,
    IPR_PLACE D_STR_CODE,
    IPR_IS_LOCAL D_BOOL)
AS
declare variable ip_addr_int d_id;
declare variable t d_integer;
declare variable tmp_str d_str_code;
declare variable i d_integer;
begin
  -- Процедура возвращает информацию о стране/регионе/населенном пункте по ip
  -- из таблицы iprange
  -- IP_ADDRESS может быть как целочисленным, н-р: 1533531058,
  -- так и в виде классической строки, н-р: 91.103.211.178

  -- Если параметр строка вида ###.###.###.###, конвертируем в целое
  -- В результате ip_addr_int будет содержать адрес в числовом виде (int)
  ipr_ip_address = '';
  ip_addr_int = 0;
  t = position('.', ip_address);
  tmp_str = ip_address;
  while (t > 0) do
  begin
    ip_addr_int = ip_addr_int * 256 + cast(left(tmp_str, t - 1) as integer);
    tmp_str = substring(tmp_str from t + 1);
    t = position('.', tmp_str);
  end
  if (ip_addr_int > 0) then
  begin
    ip_addr_int = ip_addr_int * 256 + cast(tmp_str as integer);
    ipr_ip_address = ip_address; -- Адрес был текстовый, забираем в результат
  end
  else
    ip_addr_int = ip_address;

  -- Ищем нужный диапазон, для нахождения населенного пункта
  ipr_country = '<Страна неизвестна>';
  ipr_region = '<Регион неизвестен>';
  ipr_place = '<Населенный пункт неизвестен>';
  ipr_is_local = 0; -- Если не нашли, страна не наша

  select rd_name, ipr_region, ipr_place, ipr_is_local
  from iprange
  left join refdata on ipr_country = rd_id
  where :ip_addr_int between ipr_range_begin and ipr_range_end
  into ipr_country, ipr_region, ipr_place, ipr_is_local;

  if (ipr_ip_address = '') then
  begin
    -- Формируем ip-адрес в строковом виде
    i = 3;
    while (i >= 0) do
    begin
      t = ip_addr_int / cast(power(256, i) as integer);
      ipr_ip_address = ipr_ip_address || t || iif(i > 0, '.', '');
      ip_addr_int = mod(ip_addr_int, power(256, i));
      i = i - 1;
    end
  end

  suspend;
end
