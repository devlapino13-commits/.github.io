CREATE PROCEDURE HP_DO_KN_OPERBOOK_MEMORDER(
    EXIST_OL_ID D_ID,
    KC_ID D_ID,
    ACC_CODE_DT D_STR_CODE,
    ACC_CODE_KT D_STR_CODE,
    AMOUNT_SUM D_CURRENCY,
    AMOUNT_VAL D_CURRENCY,
    AMOUNT_TAX_DT D_CURRENCY,
    AMOUNT_TAX_KT D_CURRENCY,
    OB_DETAIL D_STR_255,
    ACC_DT D_ID,
    ACC_KT D_ID,
    OL_DATE D_DATE,
    OB_CURR D_STR_CODE)
RETURNS (
    OL_ID D_ID,
    KOB_ID D_ID)
AS
declare variable obd_id d_id;
declare variable ob_curr_rate d_currency = 1;
begin
  if (exist_ol_id is null) then
      exception e_default 'OL_ID is empty';

  ol_id = exist_ol_id;
  ob_curr_rate = 1;

  if (amount_val <> 0 and amount_sum <> amount_val) then
  ob_curr_rate = cast(amount_sum / amount_val as d_currency);

  if (ol_date is null) then
  begin
    select ol_date
    from operlist
    where ol_id = :ol_id
    into :ol_date;
  end

  if (ol_date is null) then
    exception e_default 'OL_ID not found';

  if (ol_date < '31.12.2017') then
    exception e_default 'Can not make record early 31.12.2017';

  acc_code_dt = replace(acc_code_dt, '.', '');
  acc_code_dt = replace(acc_code_dt, ' ', '');

  acc_code_kt = replace(acc_code_kt, '.', '');
  acc_code_kt = replace(acc_code_kt, ' ', '');

  if (acc_dt is null) then
  begin
    select acc_id
    from ref_account
    where acc_code = :acc_code_dt
    into acc_dt;
  end

  if (acc_kt is null) then
  begin
    select acc_id
    from ref_account
    where acc_code = :acc_code_kt
    into acc_kt;
  end

  if (acc_dt is null) then
    exception e_default 'Dt account not found: ' || coalesce(acc_code_dt, '<null>');

  if (acc_kt is null) then
    exception e_default 'Kt account not found: ' || coalesce(acc_code_kt, '<null>');

  select obd_id
  from kn_obdetail
  where obd_detail = :ob_detail
  into obd_id;

  if (obd_id is null) then
  begin
    insert into kn_obdetail (obd_detail)
    values (:ob_detail)
    returning obd_id
    into obd_id;
  end

  if (amount_tax_dt is null) then
    amount_tax_dt = amount_sum;
  if (amount_tax_kt is null) then
    amount_tax_kt = amount_sum;

  if (coalesce(ob_curr, '') = '') then
    ob_curr = f_get_const('cm_default_currency');

  insert into kn_operbook (kob_from_ol_id, kob_currency, kob_curr_rate, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                           kob_summa_tax_dt, kob_summa_tax_kt, kob_detail_id, kob_from_kc_id)
  values (:ol_id, :ob_curr, :ob_curr_rate, :acc_dt, :acc_kt, :amount_sum, :amount_val, :amount_tax_dt, :amount_tax_kt, :obd_id, :kc_id)
  returning (kob_id)
  into kob_id;

  suspend;
end
