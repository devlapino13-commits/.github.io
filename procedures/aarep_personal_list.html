AS
declare variable ps_is_on_staff smallint;
declare variable ps_is_working smallint;
declare variable jl_is_front integer;
declare variable c_sex smallint;
declare variable c_is_resident smallint;
declare variable st float;
declare variable hrc_is_forced integer;
declare variable js_code varchar(100);
declare variable adc_address_type integer;
declare variable adl_address_str d_str_255;
declare variable reg_hr_region_id d_integer;
declare variable cds_doc_blob_1 id_type;
declare variable cds_doc_blob_2 id_type;
declare variable asp_signed d_bool;
declare variable total_work_days d_integer;
declare variable work_years d_integer;
declare variable work_months d_integer;
declare variable work_days d_integer;
begin
  for select ps_from_c_id, ps_date_begin, ps_date_end, ps_period_month, ps_date_end_in_order, ps_is_on_staff,
             ps_is_working, ps_salary, ps_transfer_office, ps_tabel_no, ho_date, ho_from_reg_id, ho_text, ho_reason,
             ho_save_time, ho_order_no, jl_id, jl_name, jl_order_no, dep_name, dep_order_no, r1.reg_name,
             r1.reg_code || ' ' || r1.reg_name, r2.reg_name, jl_is_front, jl_category, c_id, c_name_fio, c_sex,
             c_birth_date, d1.rd_name, d2.rd_name, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire,
             (select cont_value
              from hp_contdata_get_email_main(0, c_id)), d3.rd_name, d4.rd_name, c_kol_family, c_kol_child, c_memo,
             c_reg_no, c_okpo, c_inn, c_account, c_org_from_b_id, /*c_hired_date,*/ c_is_resident, d5.rd_name,
             (select cont_value
              from hp_contdata_get_cell_main(0, c_id)), c_others,
             /*adl_address_str,*/ r3.reg_code || ' ' || r3.reg_name, jk_system_id, hrc_name, hrc_is_forced,
             ps_from_reg_id, br.reg_code || ' ' || br.reg_name, r1.reg_stat_rep_region, c_pincode, jl_name,
             case c_verify_status
               when -2 then 'Данные не верные'
               when -1 then 'не требуется верификация (юр.лица)'
               when -0 then 'Данные не полные'
               when 1 then 'В процессе верификации'
               when 2 then 'Данные верны'
             end,
             iif(c_blob_photo_id > 0, 'Есть', 'Нет'), iif(c_blob_pass1_id > 0, 'Есть', 'Нет'),
             iif(c_blob_pass2_id > 0, 'Есть', 'Нет'), jk_name, c_birth_city, c_pass_who_code, r1.reg_hr_region,
             iif(jl_id in (226, 227) and f_get_db_id() in (39801, 41701), 1, 0), d2.rd_str_id, jl_gni_code
      from hr_pstatus, hr_orders, region_list(:report_region_id, 1, -1, 1), joblist, depart, region r1, region r2, region r3, clients, region br, refdata d1, refdata d2, refdata d3, refdata d4, refdata d5, hr_class, jobkind
      where ho_id = ps_from_ho_id and
            (ps_from_reg_id = id_list) and
            ps_date_begin <= :report_date_end and
            ps_date_end >= :report_date_end and
            r3.reg_unicode = r1.reg_head_office and
            ps_from_reg_id = r1.reg_unicode and
            br.reg_unicode = r1.parent and
            ps_from_dep_id = dep_id and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            ho_from_reg_id = r2.reg_unicode and
            ps_class = hrc_id and
            (((ps_is_working = 0) and
            ((ps_transfer_office is null) or (ps_transfer_office = ps_from_reg_id)) and
            (:report_list_kind = 0)) or ((ps_is_working = 1) and
            (:report_list_kind = 1)) or (:report_list_kind = -1)) and
            ps_from_c_id = c_id and
            c_nation = d1.rd_id and
            c_citizen = d2.rd_id and
            c_family = d3.rd_id and
            c_education = d4.rd_id and
            c_army = d5.rd_id /*and
            adc_address_type = 1 and
            adc_from_c_id = c_id and
            adc_from_adl_id = adl_id */
      order by jl_order_no, c_name_fio

      into :ps_from_c_id, :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order, :ps_is_on_staff,
           :ps_is_working, :ps_salary, :ps_transfer_office, :ps_tabel_no, :ho_date, :ho_from_reg_id, :ho_text,
           :ho_reason, :ho_save_time, :ho_order_no, :jl_id, :jl_name, :jl_order_no, :dep_name, :dep_order_no, :reg_name,
           :reg_code_name, :ho_reg_name, :jl_is_front, :jl_category, :c_id, :c_name_fio, :c_sex, :c_birth_date,
           :c_nation, :c_citizen, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_email, :c_family,
           :c_education, :c_kol_family, :c_kol_child, :c_memo, :c_reg_no, :c_okpo, :c_inn, :c_account, :c_org_from_b_id,
           /*:c_hired_date,*/ :c_is_resident, :c_army, :c_plastcard_no, :c_others, /*:c_full_address,*/ :ps_branch,
           :jl_kind, :hrc_name, :hrc_is_forced, :ps_from_reg_id, :branch_name, :reg_stat_rep_region, :c_pincode,
           :jk_name, c_verify_status_str, c_blob_photo_check, c_blob_pass1_check, c_blob_pass2_check, jobkind_name,
           c_birth_city, c_pass_who_code, :reg_hr_region_id, :jl_acc, :c_citizen_code, jl_gni_code
  do
  begin

    select ps.ps_date_begin
    from hr_pstatus ps
    inner join hr_orders ho on ps.ps_from_ho_id = ho.ho_id and
          ho.ho_order_type = 1 and
          ho.ho_status = 1
    where ps.ps_from_c_id = :ps_from_c_id and
          ps.ps_date_begin <= :report_date_end
    order by 1 desc rows 1
    into :c_hired_date;

    reg_hr_region = '';

    if (reg_hr_region_id is not null) then
    begin
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :reg_hr_region_id
      into :reg_hr_region;
    end

    c_full_address = '';
    c_fact_address = '';

    for select adc_address_type, adl.adl_address_str
        from addrlist adl
        inner join addrclient adcl on adl.adl_id = adcl.adc_from_adl_id
        where adcl.adc_from_c_id = :ps_from_c_id and
              adc_address_type in (1, 2)
        into :adc_address_type, :adl_address_str
    do
    begin
      if (adc_address_type = 1) then
        c_full_address = adl_address_str;
      if (adc_address_type = 2) then
        c_fact_address = adl_address_str;
    end

    js_name = null;
    select rd_name, rd_code
    from refdata
    where rd_id = :jl_kind
    into js_name, js_code;
    if (js_name is null) then
      js_name = 'не определен';

    if (ps_is_on_staff = 0) then
      ps_is_on_staff_text = 'Вне штатный';
    else
      ps_is_on_staff_text = 'Штатный';

    if (ps_is_working = 1) then
      ps_is_working_text = 'Работает';
    else
      ps_is_working_text = 'Уволен';

    if (js_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
      jl_is_kred_person = 'Кредит.персонал';
    else
      jl_is_kred_person = 'Админ.персонал';

    if (jl_is_front = 1) then
      jl_is_front_text = 'Фронт офис';
    else
      jl_is_front_text = 'Бэк офис';

    if (c_sex = 1) then
      c_sex_text = 'Мужчина';
    else
      c_sex_text = 'Женщина';

    if (c_is_resident = 1) then
      c_is_resident_text = 'Резидент';
    else
      c_is_resident_text = 'Не резидент';

    if (c_hired_date is not null) then
      c_stage = datediff(month, c_hired_date, report_date_end);
    else
      c_stage = 0;
    st = c_stage / 12;
    c_stage_year = cast(st as integer);
    if (c_stage_year > (c_stage / 12)) then
      c_stage_year = c_stage_year - 1;

    c_stage_kc_date = null;
    if (js_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
    begin
      select ps_date_begin
      from hr_pstatus, joblist, jobkind
      where ps_from_c_id = :ps_from_c_id and
            ps_from_reg_id = :ps_from_reg_id and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            ps_status = 1
      order by ps_date_begin rows 1
      into :c_stage_kc_date;
    end
    jl_kred_kind = '';
    if (js_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')) then
      jl_kred_kind = 'Кредит.сотрудник';

    if (c_birth_date is not null) then
    begin
      c_age = cast(datediff(month, c_birth_date, report_date_end) / 12 as integer);
    end
    else
      c_age = 0;

    ps_curr_status = '';
    select sawt_name
    from hr_voctravel, hr_orders, sa_worktype
    where hrv_from_ho_id = ho_id and
          ho_order_type = sawt_id and
          hrv_from_c_id = :ps_from_c_id and
          hrv_status = 1 and
          hrv_date_begin <= :report_date_end and
          hrv_date_end_fact >= :report_date_end
    order by hrv_date_begin desc rows 1
    into :ps_curr_status;

    if (hrc_is_forced = 0) then
      hrc_is_force_txt = 'По желанию работника';
    else
      hrc_is_force_txt = 'По желанию работодателя';

    select aa_curr_level3_name
    from aarep_get_office_categ(:ps_from_reg_id, :report_date_end)
    into :reg_okrug;

    cs_inst_name = '';
    cs_spec_name = '';
    cs_year_end = null;
    cs_education_level = null;

    select cs_inst_name, cs_spec_name, cs_year_end,
           case cs_education_level
             when 18 then 'Неполное среднее'
             when 19 then 'Среднее'
             when 20 then 'Средне-специальное'
             when 21 then 'Неполное высшее'
             when 22 then 'Высшее'
             when 23 then 'Ученая степень'
             when 3554 then 'Неполное средне-специальное'
           end
    from cstudy
    where cs_from_c_id = :c_id
    order by cs_year_end desc rows 1
    into :cs_inst_name, :cs_spec_name, :cs_year_end, :cs_education_level;

    lml_fio = f_lo_mgr_name(f_lo_mgr_id(null, report_date_end, c_id, ps_from_reg_id));

    asp_cont = '';
    asp_signed = 0;

    asp_signed = F_ASP_SIGNED(:c_id, :report_date_end);

    if (asp_signed = 1) then
      asp_cont = 'Прикреплен';
    else
      asp_cont = 'Не прикреплен';

    doc_emp_history = '';
    select list(cds.cds_doc_no, ', ')
    from clientdocs cds
    inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
    inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
    where pdt.pdt_code = 'employment history' and  -- тип документа: трудовая книжка
          cds.cds_is_delete = 0 and -- документ сотрудника не удален
          cds.cds_from_c_id = :c_id -- фильтр по сотруднику
    into doc_emp_history; -- реквизиты трудовой книжки (через запятую вкладыша, если есть)

    -- RFCKG-2414 -- >
    total_work_days = 0;
    select coalesce(sum(datediff(day, cw.cw_date_from, cw.cw_date_till)), 0)
    from coldwork cw
    where cw.cw_from_c_id = :c_id
    group by cw.cw_from_c_id
    into :total_work_days;                  -- Суммарный стаж работы сотрудника в днях в предыдущих местах

    -- Далее идёт разбиение суммарного количества дней стажа работы на годы, месяцы и дни:
    work_years = total_work_days / 365;                                    -- Годы
    work_months = (total_work_days - work_years * 365) / 30;               -- Месяцы
    work_days = (total_work_days - work_years * 365) - (work_months * 30); -- Дни

    -- Проверка если последняя цифра года заканчивается на 1 - 4 и если год не в диапазоне между 11 и 14,
    -- то после года идёт слово - год (г.),
    -- если нет, то слово - лет (л.)
    if (right(cast(work_years as d_str_10), 1) in ('1', '2', '3', '4') and
        work_years not between 11 and 14) then
      work_experience = work_years || ' г. ' || work_months || ' м. ' || work_days || ' д.';
    else
      work_experience = work_years || ' л. ' || work_months || ' м. ' || work_days || ' д.';
    -- < --

    select rd.rd_str_id
    from client_gni_category cgc
    inner join refdata rd on rd.rd_id = cgc.cgc_from_category
    where cgc.cgc_from_c_id = :c_id and
          cgc.cgc_is_delete = 0 and
          cgc.cgc_date_end = '31.12.2999'
    into :cgc_from_category;

    suspend;
  end
