CREATE PROCEDURE OKT_IVR_CALL_LIST
RETURNS (
    PROJECT_CODE D_STR_100,
    PANT_ID D_ID,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID,
    PHONE D_STR_20,
    TIMEZONE D_INTEGER,
    PANT_DATE_BEGIN D_DATE,
    NEXT_CALL_DATETIME D_DATETIME)
AS
declare variable TIMEZONE_RU D_INTEGER;
declare variable K_ID D_ID;
declare variable HOURS_9 D_DATETIME;
declare variable SIRENA_1_ID D_INTEGER;
declare variable SIRENA_1_UN_ID D_ID;
declare variable KS_FINANACE_2_ID D_INTEGER;
declare variable KS_FINANACE_2_UN_ID D_ID;
declare variable SIRENA_4_UN_ID D_ID;
declare variable AUTO_INFORMER D_INTEGER;
begin
  --if (F_DATE_WORK_DAY(current_date) <> current_date) then
  --  exit;

  object_kind = 0;

  sirena_1_id = f_get_rd_id('oktell.projects.sirena1');
  sirena_1_un_id = (select un.un_id from username un where un.un_name = 'vosirena1');
  sirena_4_un_id = (select un.un_id from username un where un.un_name = 'voisirena4');

  ks_finanace_2_id = f_get_rd_id('oktell.projects.ks_finance_2');
  ks_finanace_2_un_id = (select un.un_id from username un where un.un_name = 'voksfinance');

  auto_informer = f_get_rd_id('oktell.projects.auto_informer');

  hours_9 = cast(cast(f_datetostr(current_date) as d_str_10) || ' 09:00' as timestamp);

  for select pant_date_begin,
             pant_id,
             pant_from_k_id,
             project_code,
             kncl.kc_from_c_id
  from
  (
      select pantk.pant_date_begin,
             pantk.pant_id,
             pantk.pant_from_k_id,
             'oktell.projects.auto_informer' as project_code
      from oksummary oks
      inner join pantera_kontracts pantk on pantk.pant_from_k_id = oks.oks_from_k_id
      where current_date between oks_date_begin and oks_date_end
      and current_date - oks_delinq_date between 7 and 20
      and current_date between pantk.pant_date_begin and pantk.pant_date_end
      and pantk.pant_rd_project_type = :auto_informer

      union all

      select pantk.pant_date_begin,
             pantk.pant_id,
             pantk.pant_from_k_id,
             'oktell.projects.sirena1' as project_code
      from oksummary oks
      inner join pantera_kontracts pantk on pantk.pant_from_k_id = oks.oks_from_k_id
      where current_date between oks_date_begin and oks_date_end
      and current_date - oks_delinq_date between iif(oks.oks_kc_id = 1000000000004, 1, 7) and 30
      and oks.oks_balance > 500
      and current_date between pantk.pant_date_begin and pantk.pant_date_end
      and pantk.pant_from_un_id in (:sirena_1_un_id, :sirena_4_un_id)
      and pantk.pant_rd_project_type = :sirena_1_id

      union all

      select pantk.pant_date_begin,
             pantk.pant_id,
             pantk.pant_from_k_id,
             'oktell.projects.ks_finance_2' as project_code
      from oksummary oks
      inner join pantera_kontracts pantk on pantk.pant_from_k_id = oks.oks_from_k_id
      where current_date between oks_date_begin and oks_date_end
      and current_date between pantk.pant_date_begin and pantk.pant_date_end
      and pantk.pant_from_un_id = :ks_finanace_2_un_id
      and pantk.pant_rd_project_type = :ks_finanace_2_id
      ) A
  inner join knclient kncl on kncl.kc_from_k_id =  pant_from_k_id
  where kncl.kc_amount_approved >0
  into :pant_date_begin,
       :pant_id,
       :k_id,
       :project_code,
       :object_id
  do
  begin
    select reg.reg_timezone
    from kontract kont
    inner join region reg on kont.k_last_reg_id = reg.reg_unicode
    where kont.k_id = :k_id
    into timezone_ru;

    timezone_ru = coalesce(timezone_ru, 3);

    timezone = (case when phone starting('996') then 6
                     when phone starting('998') then 5
                     when phone starting('992') then 5
                     else timezone_ru end);

    if (dateadd(hour, timezone -3, current_timestamp) < hours_9) then --  если сейчас локальное время клиента меньше 9 часов утра, ему не звоним
        continue;
    else if (dateadd(hour, timezone -3, current_timestamp) >= cast(cast(f_datetostr(current_date) as d_str_10) || ' 17:00' as timestamp)) then --  если сейчас локальное время клиента больше или равно 17:00, ему не звоним
        continue;

    next_call_datetime = null;

    select cv.ois_next_call_datetime
    from okt_ivr_calls_svod cv
    where cv.ois_from_pant_id = :pant_id
    and cv.ois_object_kind = 0
    and cv.ois_object_id = :object_id
    into :next_call_datetime;

    next_call_datetime = coalesce(next_call_datetime, hours_9);
    if (dateadd(hour, timezone -3, current_timestamp) >= next_call_datetime) then
    begin
        phone = left(f_cont_info('clients', :object_id, 'cell_main', 'ok'), 20);

        /*  для теста. Фейковые номера телефонов, один из которых телефон Ильи
        if (MOD(pant_id, 2) = 0) then
            phone = '996509757573';
        else
            phone = '79035544967';
        */

        if (phone <> '') then
            suspend;
    end
  end
end
