CREATE PROCEDURE AAREP_SITE_AGREEMENT(
    PAR D_STR_255)
RETURNS (
    C_FIO D_STR_255,
    C_INN D_STR_100,
    C_PASS_NO D_STR_100,
    REGISTRATION_PHONE D_STR_100,
    REGISTRATION_DATETIME D_DATETIME,
    PERSONAL_AGREEMENT STR_255)
AS
declare variable c_id d_id;
declare variable phone d_str_20;
declare variable pass_no d_str_100;
declare variable fio d_str_100;
declare variable inn d_str_20;
declare variable sql_text d_str_2048;
begin
  if (par <> '') then
  begin
    if (:par similar to '[0-9]+' and
        char_length(:par) = 13) then
    begin
      c_id = par;
      sql_text = 'select c_name_fio, c_pass_no, c_inn, c_id, coalesce(ok_cell_main, last_cell_main)
                  from clients
                  inner join continfo on cci_object_kind = 0 and cci_object_id = c_id
                  where c_id = ' || :c_id;
    end
    if (:par similar to '[0-9]+' and
        c_id is null) then
    begin
      inn = par;
      sql_text = 'select c_name_fio, c_pass_no, c_inn, c_id, coalesce(ok_cell_main, last_cell_main)
                  from clients
                  inner join continfo on cci_object_kind = 0 and cci_object_id = c_id
                  where c_inn = ''' || :inn || '''';
    end
    if (par similar to '%[а-яА-ЯA-Za-z]%' and
        not par similar to '%[0-9]%') then
    begin
      fio = par;
      sql_text = 'select c_name_fio, c_pass_no, c_inn, c_id, coalesce(ok_cell_main, last_cell_main)
                  from clients
                  inner join fiolist on fl_id = c_from_fl_id
                  inner join continfo on cci_object_kind = 0 and cci_object_id = c_id
                  where (c_name_fio = ''' || :fio || ''' or fl_name1 = ''' || :fio || ''' or fl_name2 = ''' || :fio || ''' or fl_name3 = ''' || :fio || ''')';
    end
    if (:par similar to '(79|992|996|998|994)[[:DIGIT:]]{9}' or (:par similar to '(9960|9961|9963|9964)[[:DIGIT:]]{8}')) then
    begin
      phone = par;
      sql_text = 'select c_name_fio, c_pass_no, c_inn, c_id, coalesce(ok_cell_main, last_cell_main)
                  from clients
                  inner join continfo on cci_object_kind = 0 and cci_object_id = c_id
                  where (ok_cell_main = ''' || :phone || '''
                  or last_cell_main = ''' || :phone || ''')';
    end
    if (par similar to '%[а-яА-ЯA-Za-z0-9]%' and
        fio is null and
        inn is null and
        c_id is null) then
    begin
      pass_no = par;
      sql_text = 'select c_name_fio, c_pass_no, c_inn, c_id, coalesce(ok_cell_main, last_cell_main)
                  from clients
                  inner join continfo on cci_object_kind = 0 and cci_object_id = c_id
                  where c_index_c_pass_no = ''' || :pass_no || '''';
    end

    for execute statement :sql_text
            into c_fio, c_pass_no, c_inn, c_id, registration_phone
    do
    begin
      for select wma_create_datetime, iif(wmat_anketa_memo containing 'Код из смс:', substring(wmat_anketa_memo from 12 for (position(ascii_char(13) in wmat_anketa_memo) - 12)), ''), wmat_name
          from cc_website_miniapp
          inner join cc_website_miniapp_tmp on wmat_id = wma_last_wmat_id
          where (wma_phone_no = :registration_phone or wma_from_c_id = :c_id)
          into registration_datetime, personal_agreement, c_fio
      do
      begin
        suspend;
      end

      registration_datetime = null;
      personal_agreement = null;

      for select emw_datetime, emw_sms_code, emw_cl_fio
          from extminiweb
          where emw_phone = :registration_phone
          into registration_datetime, personal_agreement, c_fio
      do
      begin
        suspend;
      end
    end
  end
end
