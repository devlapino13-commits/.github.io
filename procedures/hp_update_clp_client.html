CREATE PROCEDURE HP_UPDATE_CLP_CLIENT(
    K_ID D_ID,
    C_ID D_ID,
    UN_ID D_ID,
    STR_VALUE D_STR_255 = '',
    QUESTIONS_ID D_INTEGER = 1,
    CLARIFICATION_ID D_INTEGER = 1,
    COMAND_CODE D_STR_CODE = 'select')
RETURNS (
    CMC_QUESTIONS_ID D_INTEGER,
    CMC_CLARIFICATION_ID D_INTEGER)
AS
declare variable ma_id d_integer;
declare variable k_disb_date d_date;
declare variable ovr_value d_str_255;
declare variable opr_id d_id;
declare variable ovs_id d_integer;
begin
  select k.k_from_ma_app_id, k.k_disb_date
  from kontract k
  where k.k_id = :k_id
  into ma_id, k_disb_date;

  select coalesce(cmc.cmc_questions_id, 1),
         coalesce(cmc.cmc_clarification_id, 1)
  from clp_miniappclient cmc
  where cmc.cmc_from_ma_id = :ma_id
    and cmc.cmc_from_c_id = :c_id
  into cmc_questions_id,
       cmc_clarification_id;


  if (comand_code = 'update') then
  begin
    if (k_disb_date > '27.06.2024') then
    begin
      update clp_miniappclient cmc
      set cmc_questions_id = :questions_id,
          cmc_clarification_id = :clarification_id,
          cmc_from_un_id = :un_id
      where cmc.cmc_from_ma_id = :ma_id
        and cmc.cmc_from_c_id = :c_id;
    end
    else
    begin
      select coalesce(ovr.ovr_value, ''),
             opr.opr_id,
             ovs.ovs_id
      from objectprops opr
      inner join objectvalues ovs on ovs_from_opr_id = opr.opr_id
      inner join objectvalres ovr on ovr.ovr_from_opr_id = opr.opr_id and ovr.ovr_from_ovs_id = ovs.ovs_id
      inner join refdata rd on rd.rd_id = opr.opr_subject_id
      where rd.rd_code = 'AnketaQuestions.descr_of_loan'
      and opr.opr_object_kind = 0
      and opr.opr_object_id = :c_id
      and opr.opr_subject_kind = '-1'
      and ovs_date_begin <= :k_disb_date
      and ovs_date_end >= :k_disb_date
      rows 1
      into ovr_value, opr_id, ovs_id;

      if (coalesce(opr_id, -1) <> -1 and
          coalesce(ovs_id, -1) <> -1) then
      begin
        update objectvalres
        set ovr_value = :str_value
        where ovr_from_opr_id = :opr_id
          and ovr_from_ovs_id = :ovs_id
          and ovr_result_rd_id = 1435;
      end
    end
  end

   suspend;
end
