CREATE PROCEDURE HP_MIGRATION_CANCEL(
    HO_ID DECIMAL(18,0),
    PS_ID INTEGER,
    IS_ORDER INTEGER)
AS
declare variable dd date;
declare variable order_type integer;
declare variable c_id decimal(18,0);
begin
  if (is_order = 1) then
  begin
    select ho_order_type, ps_date_begin, ps_from_c_id
    from hr_pstatus, hr_orders
    where ps_from_ho_id = :ho_id and
          ps_id = :ps_id and
          ho_id = :ho_id
    into :order_type, :dd, :c_id;
    if (order_type = 2) then
    begin
      update empregist
      set er_cancelled = 1
      where er_from_c_id = :c_id and
            er_date_begin <= :dd and
            er_date_end >= :dd and
            er_cancelled <> 1;
    end
  end
  else
  begin
    select ho_order_type, hrv_date_begin, hrv_from_c_id
    from hr_voctravel, hr_orders
    where hrv_from_ho_id = :ho_id and
          hrv_number = :ps_id and
          ho_id = :ho_id
    into :order_type, :dd, :c_id;
    if (order_type in (7, 12)) then
    begin
      update empregist
      set er_cancelled = 1
      where er_from_c_id = :c_id and
            er_date_begin <= :dd and
            er_date_end >= :dd and
            er_cancelled <> 1;
    end
  end

end
