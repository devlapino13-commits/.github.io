CREATE PROCEDURE AAREP_PHONECALLS_LIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    OKS_DELINQ_DAY D_INTEGER,
    C_NAME_FIO_HR D_STR_100,
    PNC_SRC_PHONE_ID D_STR_20,
    PNC_START D_DATETIME,
    PNC_DURATION D_INTEGER,
    PNC_FILENAME D_STR_255,
    PNC_SOURCE D_STR_CODE)
AS
declare variable PNC_ID D_ID;
declare variable PNC_OBJECT_ID D_ID;
declare variable PNC_START_DATE D_DATE;
begin
  for select pc.pnc_id, cl.c_name_fio, pnc_object_id, cl_hr.c_name_fio, f_cont_code(pc.pnc_src_phone_id), pc.pnc_start,
             coalesce((datediff(second, pnc_start, pnc_finish)), 0), coalesce(pc.pnc_filename, ''), iif(pnc_source > 0, 'КЦ', 'МП')
      from phonecalls pc
      inner join clients cl on cl.c_id = pc.pnc_dst_c_id
      inner join clients cl_hr on cl_hr.c_id = pc.pnc_src_c_id
      inner join kontract k on k.k_id = pc.pnc_object_id
      inner join knaktlist kn on kn.kal_from_kf_id = k.k_from_kf_id
      inner join region_list(:rep_region_id, 1, -1, 1) on id_list = kn.kal_new_reg_id
      where pc.pnc_object_kind = 9 and
            cast(pc.pnc_start as d_date) between :rep_date_begin and :rep_date_end and
            cast(pc.pnc_start  as d_date) between kn.kal_date and kn.kal_date_end and
            pc.pnc_filename <> '' and
            kn.is_delete = 0
      into :pnc_id, c_name_fio, pnc_object_id, c_name_fio_hr, pnc_src_phone_id, pnc_start, pnc_duration, pnc_filename, :pnc_source
  do
  begin

    pnc_start_date = cast(pnc_start as d_date);

    k_kontract_no = '';
    select k.k_kontract_no
    from kontract k
    where k.k_id = :pnc_object_id
    into :k_kontract_no;

    if (pnc_start_date is not null) then
    begin
      oks_delinq_day = 0;
      select iif((cast(:pnc_start_date as date) - oks.oks_delinq_date < 0), 0, cast(:pnc_start_date as date) - oks.oks_delinq_date) -- число дней активной просрочки
      from oksummary oks
      where :pnc_start_date between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_from_k_id = :pnc_object_id and
            oks.oks_close_state = 0
      into :oks_delinq_day;
    end
    suspend;
  end
end
