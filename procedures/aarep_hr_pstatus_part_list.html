CREATE PROCEDURE AAREP_HR_PSTATUS_PART_LIST(
    REP_DATE D_DATE)
RETURNS (
    PS_ID D_INTEGER,
    HO_ID ID_TYPE,
    C_ID D_ID,
    C_INN D_STR_20,
    C_NAME_FIO D_STR_100,
    PART_HOURS D_SMALLINT,
    PS_IS_MAIN D_STR_20,
    OFFICE_ID D_INTEGER,
    OFFICE_NAME D_STR_100,
    DEP_NAME D_STR_100,
    JL_NAME D_STR_100,
    HO_TEXT D_STR_LONG)
AS
declare variable ps_from_dep_id d_integer;
declare variable ps_from_jl_id d_integer;
declare variable ps_is_not_main d_bool;
declare variable ho_order_type d_integer;
declare variable sawt_name d_str_100;
begin
  -- Возвращает cписок действующих совместителей (RFCAZ-372)

  for select ps_id,
             ps_from_ho_id,
             ps_from_c_id,
             ps_part_hours,
             ps_from_reg_id,
             ps_from_dep_id,
             ps_from_jl_id,
             ps_is_not_main,
             ho_order_type,
             left(ho_order_no || ' от ' || f_datetostr(ho_date) || '. ' || ho_text, 1024)
      from hr_pstatus
      inner join hr_orders on ho_id = ps_from_ho_id
      where ho_status = 1 and
            ps_status = 1 and
            ps_is_part = 1 and
            ps_is_working = 1 and
            :rep_date between ps_date_begin and ps_date_end
      into ps_id,
           ho_id,
           c_id,
           part_hours,
           office_id,
           ps_from_dep_id,
           ps_from_jl_id,
           ps_is_not_main,
           ho_order_type,
           ho_text
  do
  begin
    c_inn = null;
    c_name_fio = null;
    office_name = null;
    dep_name = null;
    jl_name = null;

    select c_inn,
           c_name_fio
    from clients
    where c_id = :c_id
    into c_inn,
         c_name_fio;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :office_id
    into office_name;

    select dep_name
    from depart
    where dep_id = :ps_from_dep_id
    into dep_name;

    select jl_name
    from joblist
    where jl_id = :ps_from_jl_id
    into jl_name;

    if (ps_is_not_main = 1) then
      ps_is_main = 'Не основное';
    else
      ps_is_main = 'Основное';

    select sawt_name
    from sa_worktype
    where sawt_id = :ho_order_type
    into sawt_name;

    ho_text = 'Тип - ' || sawt_name || '. № ' || ho_text;

    suspend;
  end
end
