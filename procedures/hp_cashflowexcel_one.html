CREATE PROCEDURE HP_CASHFLOWEXCEL_ONE(
    IN_CFX_ID D_ID)
RETURNS (
    CFX_ID D_ID,
    CFX_DATE D_DATE,
    CFX_DOC_KIND D_STR_20,
    CFX_BANK_BIK D_STR_20,
    CFX_DOC_NO D_STR_20,
    CFX_OPER_KIND D_STR_20,
    CFX_CODE D_STR_20,
    CFX_RESIDENCY D_STR_20,
    CFX_CLIENT_TYPE D_STR_20,
    CFX_CLIENT_NAME D_STR_255,
    CFX_CLIENT_COUNTRY D_STR_20,
    CFX_ACC_DT D_STR_20,
    CFX_ACC_KT D_STR_20,
    CFX_SUMMA D_CURRENCY,
    CFX_CURRENCY D_STR_10,
    CFX_CLIENT_BIRTH_DATE D_STR_20,
    CFX_LOAN_ID D_STR_20,
    CFX_KONTRACT_NO D_STR_20,
    CFX_CLIENT_ACC D_STR_20,
    CFX_OL_ID D_STR_20,
    CFX_OL_MEMO D_STR_LONG,
    CFX_NEED_RECALC D_BOOL,
    CFX_RES_SIGN D_SMALLINT,
    CFX_RES_NON_RESIDENT D_BOOL,
    CFX_RES_OL_ID D_ID,
    CFX_RES_OL_OB_NUMBER D_INTEGER,
    CFX_RES_KOB_ID D_ID,
    CFX_RES_CLIENT_ID D_ID,
    CFX_RES_ACCBANK_ID D_ID,
    CFX_RES_STATUS D_SMALLINT,
    CFX_RES_CHECK D_SMALLINT,
    ACB_BANK_NAME D_STR_255,
    ACB_REG_NO D_STR_20,
    ACB_BIK D_STR_20,
    ACB_BANK_ACCOUNT D_STR_20,
    ACB_ACC_KIND D_STR_20,
    KOB_ACC_DT_CODE D_STR_CODE,
    KOB_ACC_KT_CODE D_STR_CODE,
    OL_MEMO D_STR_LONG,
    C_ORG_FULL_NAME D_STR_LONG,
    C_ORG_SHORT_NAME D_STR_255,
    C_NAME_FULL_FIO D_STR_255,
    C_NAME_FIO D_STR_255,
    C_INN D_STR_20,
    C_PASS_NO D_STR_20,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE D_DATE,
    C_ADDRESS_LEGAL D_STR_255,
    C_ADDRESS_FACT D_STR_255,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    LOAN_ID D_ID,
    FACT_DATE D_DATE,
    CL_BANK_NAME D_STR_255,
    CL_BANK_BIK D_STR_20,
    CL_BANK_SWIFT D_STR_20,
    CL_BANK_COUNTRY D_STR_20)
AS
declare variable c_type d_smallint;
declare variable kc_disb_passport_id d_id;
declare variable cds_doc_no d_str_code;
declare variable cds_doc_issued_by d_str_255;
declare variable cds_doc_date_begin d_date;
declare variable ol_acc_no_client d_integer;
declare variable c_org_from_b_id d_str_20;
begin
  for select cfx_id,
             cfx_date,
             cfx_doc_kind,
             cfx_bank_bik,
             cfx_doc_no,
             cfx_oper_kind,
             cfx_code,
             cfx_residency,
             cfx_client_type,
             cfx_client_name,
             cfx_client_country,
             cfx_acc_dt,
             cfx_acc_kt,
             cfx_summa,
             cfx_currency,
             cfx_client_birth_date,
             cfx_loan_id,
             cfx_kontract_no,
             cfx_client_acc,
             cfx_ol_id,
             cfx_ol_memo,
             cfx_need_recalc,
             cfx_res_sign,
             cfx_res_non_resident,
             cfx_res_ol_id,
             cfx_res_ol_ob_number,
             cfx_res_kob_id,
             cfx_res_client_id,
             cfx_res_accbank_id,
             cfx_res_status,
             cfx_res_check,

             acb_bank_name,
             acb_reg_no,
             acb_bik,
             acb_bank_account,
             acb_acc_kind,

             a_dt.acc_add_code,
             a_kt.acc_add_code,

             ol_acc_no_client,
             ol_memo,

             c_type,
             c_name_full,
             c_name_fio, 

             c_inn,
             c_pass_no,
             c_pass_who,
             c_pass_date,
             c_org_from_b_id,

             k_kontract_no,
             k_disb_date,
             kc_id,
             kob_ol_date,

             kc_disb_passport_id,
             cds_doc_no,
             cds_doc_issued_by,
             cds_doc_date_begin

      from cashflowexcel
           left join ref_accbank on (cfx_res_accbank_id = acb_id)
           left join operlist on (cfx_res_ol_id = ol_id)
           left join clients on (cfx_res_client_id = c_id)
           left join kn_operbook on (cfx_res_kob_id = kob_id)
                left join ref_account a_dt on (a_dt.acc_id = kob_acc_dt)
                left join ref_account a_kt on (a_kt.acc_id = kob_acc_kt)
                left join knclient on (kob_from_kc_id = kc_id)
                     left join kontract on (kc_from_k_id = k_id)
                     left join clientdocs on (cds_id = kc_disb_passport_id)
      where cfx_id = :in_cfx_id
      into :cfx_id,
           :cfx_date,
           :cfx_doc_kind,
           :cfx_bank_bik,
           :cfx_doc_no,
           :cfx_oper_kind,
           :cfx_code,
           :cfx_residency,
           :cfx_client_type,
           :cfx_client_name,
           :cfx_client_country,
           :cfx_acc_dt,
           :cfx_acc_kt,
           :cfx_summa,
           :cfx_currency,
           :cfx_client_birth_date,
           :cfx_loan_id,
           :cfx_kontract_no,
           :cfx_client_acc,
           :cfx_ol_id,
           :cfx_ol_memo,
           :cfx_need_recalc,
           :cfx_res_sign,
           :cfx_res_non_resident,
           :cfx_res_ol_id,
           :cfx_res_ol_ob_number,
           :cfx_res_kob_id,
           :cfx_res_client_id,
           :cfx_res_accbank_id,
           :cfx_res_status,
           :cfx_res_check,

           :acb_bank_name,
           :acb_reg_no,
           :acb_bik,
           :acb_bank_account,
           :acb_acc_kind,

           :kob_acc_dt_code,
           :kob_acc_kt_code,

           :ol_acc_no_client,
           :ol_memo,

           :c_type,
           :c_org_full_name,
           :c_name_fio,

           :c_inn,
           :c_pass_no,
           :c_pass_who,
           :c_pass_date,
           :c_org_from_b_id,

           :k_kontract_no,
           :k_disb_date,
           :loan_id,
           :fact_date,

           :kc_disb_passport_id,
           :cds_doc_no,
           :cds_doc_issued_by,
           :cds_doc_date_begin
  do
  begin
    c_org_short_name = c_name_fio;
    c_name_full_fio = c_name_fio;

    c_address_legal = '';
    c_address_fact = '';

    if (cfx_res_client_id is not null) then
    begin
      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :cfx_res_client_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      into c_address_legal;

      if (c_type = 0) then
      begin
        select adl_address_str
        from addrclient, addrlist
        where adc_from_c_id = :cfx_res_client_id and
              adc_address_type = 2 and
              adc_from_adl_id = adl_id
        into c_address_fact;
      end
    end

    if (c_type is null) then
    begin
      c_org_full_name = '';
      c_org_short_name = '';
      c_name_full_fio = '';
      c_name_fio = '';
    end else
    if (c_type = 0) then
    begin
      c_org_full_name = '';
      c_org_short_name = '';
    end else
    begin
      c_name_full_fio = '';
      c_name_fio = '';
    end

    if (kc_disb_passport_id is not null) then
    begin
      c_pass_no = cds_doc_no;
      c_pass_who = cds_doc_issued_by;
      c_pass_date = cds_doc_date_begin;
    end

    cl_bank_name = '';
    cl_bank_bik = '';
    cl_bank_swift = '';
    cl_bank_country = '';

    if (cfx_res_accbank_id is not null and cfx_res_client_id is not null) then
    begin

      if (ol_acc_no_client is not null) then
      begin
        select b_id, b_name, b_swift
        from bankclient, banks
        where bc_from_c_id = :cfx_res_client_id and
              bc_number = :ol_acc_no_client and
              bc_from_b_id = b_id
        into cl_bank_bik, cl_bank_name, cl_bank_swift;
      end

      if (coalesce(cl_bank_name, '') = '' and coalesce(c_org_from_b_id, '') <> '') then
      begin
        select b_id, b_name, b_swift
        from banks
        where b_id = :c_org_from_b_id
        into cl_bank_bik, cl_bank_name, cl_bank_swift;
      end
    end

    suspend;
  end
end
