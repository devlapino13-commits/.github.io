CREATE PROCEDURE HP_AGREEMENTS(
    PAR1 D_INTEGER,
    PAR2 D_ID = 0)
RETURNS (
    AG_ID D_INTEGER,
    AG_PARENT_ID D_INTEGER,
    AG_FROM_RD_ID D_INTEGER,
    AG_OBJECT_KIND D_INTEGER,
    AG_OBJECT_ID D_ID,
    AG_FROM_REG_ID D_INTEGER,
    AG_FROM_INIT_REG_ID D_INTEGER,
    AG_CAPTION D_STR_255,
    AG_AGREEMENT_NO VARCHAR(30),
    AG_AGREEMENT_DATE D_DATE,
    AG_DATE_BEGIN D_DATE,
    AG_DATE_END D_DATE,
    AG_MEMO D_STR_255,
    AG_FROM_UN_ID D_ID,
    AG_DATETIME D_DATETIME,
    AG_STATUS D_INTEGER,
    AG_ELSOM_ACCOUNT D_STR_20,
    AG_ELCARD_ACCOUNT D_STR_20,
    C_NAME_FIO D_STR_100,
    AG_ELSOM_BANK_ID D_INTEGER,
    AG_FROM_AL_ID D_STR_20,
    AG_COMMIS_CARDS_MB D_PERCENT,
    AG_COMMIS_CARDS_OTHER D_PERCENT,
    AG_MBANK_ACCOUNT D_STR_20,
    AG_FROM_PAY_RD_ID D_INTEGER,
    AEA_SIGN D_BOOL,
    AEO_STATUS D_INTEGER,
    AG_TAX_PERCENT D_PERCENT,
    AEA_IS_BORDERO D_BOOL)
AS
begin
  select ag_id, ag_parent_id, ag_from_rd_id, ag_object_kind, ag_object_id, ag_from_reg_id, ag_from_init_reg_id,
         ag_caption, ag_agreement_no, ag_agreement_date, ag_date_begin, ag_date_end, ag_memo, ag_from_un_id,
         ag_datetime, ag_status, ag_elsom_account, ag_elcard_account, c_name_fio, ag_elsom_bank_id, ag_from_al_id,
         ag_commis_cards_mb, ag_commis_cards_other, ag_mbank_account, ag_from_pay_rd_id, iif(aea_sign = true,  1, 0), ag_tax_percent
  from agreements
  left join clients on (ag_object_id = c_id and
        ag_object_kind = 0)
  left join agreement_esign_access on (ag_id = aea_from_ag_id)
  where ag_id = :par1
  into :ag_id, :ag_parent_id, :ag_from_rd_id, :ag_object_kind, :ag_object_id, :ag_from_reg_id, :ag_from_init_reg_id,
       :ag_caption, :ag_agreement_no, :ag_agreement_date, :ag_date_begin, :ag_date_end, :ag_memo, :ag_from_un_id,
       :ag_datetime, :ag_status, :ag_elsom_account, :ag_elcard_account, :c_name_fio, :ag_elsom_bank_id, :ag_from_al_id,
       :ag_commis_cards_mb, :ag_commis_cards_other, :ag_mbank_account, :ag_from_pay_rd_id, :aea_sign, :ag_tax_percent;

  aeo_status = -1;

  if (:par1 <> 0) then
  begin
    select aeo.aeo_status
    from agreement_eds_original aeo
    where aeo.aeo_from_ag_id = :par1
    order by aeo.aeo_datetime desc
    rows 1
    into :aeo_status;
  end

  if (:par1 = 0 and
      :par2 <> 0) then
  begin
    select ag.ag_id, ag.ag_datetime
    from agreements ag
    where current_date between coalesce(ag.ag_date_begin, current_date) and coalesce(ag.ag_date_end, '31.12.2999') and
          ag.ag_status >= 0 and
          ag.ag_object_id = :par2
    order by 2 desc rows 1
    into :ag_id, :ag_datetime;

    if (:ag_id is not null) then
      select iif(aea.aea_sign = true, 1, 0)
      from agreement_esign_access aea
      where aea.aea_from_ag_id = :ag_id
      into :aea_sign;
    else
      select iif(coalesce(c.c_type, 0) = 0, 1, 0)
      from clients c
      where c.c_id = :par2
      into :aea_sign;
  end

  select aea.aea_is_bordero
  from agreement_esign_access aea
  where aea.aea_from_ag_id = :ag_id
  into aea_is_bordero;

  suspend;

end
