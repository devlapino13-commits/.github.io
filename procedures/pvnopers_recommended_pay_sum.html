CREATE PROCEDURE PVNOPERS_RECOMMENDED_PAY_SUM(
    K_ID ID_TYPE)
RETURNS (
    RECOM_PAY_SUM DECIMAL(18,2))
AS
declare variable rep_date_limit d_date;
declare variable min_sh_date d_date;
declare variable sh_date d_date;
declare variable rep_date d_date;
begin
  rep_date = current_date;

  select min(coalesce(sh_date, cast('01.01.1900' as date)))
  from aarep_make_zad_list(:k_id, :rep_date, 1)
  where sh_date is not null
  into :min_sh_date;
    
  rep_date_limit = rep_date + 20;
    
  if (min_sh_date < rep_date or min_sh_date is null ) then
  begin
      select round(sum(zad_os + iif(coalesce(sh_date, :rep_date) <= :rep_date, sh_calc_perc, zad_perc) + sh_calc_fine + zad_kommis + round(zad_perc * 2 / 100,0)),2)
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      where coalesce(sh_date, :rep_date_limit) <= :rep_date_limit
      into recom_pay_sum;
  end
  else
  begin
      select sh_date, round(zad_os + zad_perc + round(zad_perc * 2 / 100,0), 2)
      from aarep_make_zad_list(:k_id, :rep_date, 0)
      where sh_date >= :rep_date
      rows 1
      into sh_date, recom_pay_sum;
    
      if (:rep_date <= :sh_date - 20) then recom_pay_sum = 0;
  end
  recom_pay_sum = coalesce(recom_pay_sum, 0);

  suspend;
end
