CREATE PROCEDURE AAREP_DELETE_UPDATE_OPERLIST(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    STATUS D_SMALLINT = 3)
RETURNS (
    N D_ID,
    OLR_REG_STATE_STR D_STR_20,
    OL_DATE D_DATE,
    REG_NAME D_STR_100,
    OLR_REG_DATETIME TIMESTAMP,
    UN_FIO_REG D_STR_100,
    RD_NAME_REG D_STR_100,
    OLR_APPR_STATUS_STR D_STR_100,
    OLR_APPR_DATETIME TIMESTAMP,
    UN_FIO_APPR D_STR_100,
    RD_NAME_APPR D_STR_100)
AS
declare variable olr_reg_user d_id;
declare variable olr_appr_user d_id;
declare variable olr_reg_rd_id d_integer;
declare variable olr_appr_rd_id d_integer;
declare variable olr_from_ol_id d_id;
declare variable olr_appr_status d_smallint;
declare variable olr_reg_state d_smallint;
declare variable sql_text d_str_long;
declare variable sql_add d_str_100;
declare variable olr_from_reg_id d_integer;
begin
  n = 0;
  sql_add = (case status
               when 0 then '(0)'
               when 1 then '(1)'
               when 2 then '(2)'
               when 3 then '(0,1,2)'
               else '(0,1,2)'
             end);

  sql_text = 'select olr_from_ol_id, olr_reg_state, olr_reg_user, olr_reg_rd_id, olr_reg_datetime, olr_appr_status,
             olr_appr_user, olr_appr_datetime, olr_appr_rd_id, olr_from_reg_id, olr_appr_status
      from operlistappr opr, region_list(' || :rep_region_id || ', 1, 3, 1)
      where opr.olr_from_reg_id = id_list and
            cast(opr.olr_reg_datetime as date) between ''' || :rep_date_begin || ''' and  ''' || :rep_date_end || ''' and
            opr.olr_appr_status in ' || sql_add;

  for execute statement(sql_text)
          into :olr_from_ol_id, :olr_reg_state, :olr_reg_user, :olr_reg_rd_id, :olr_reg_datetime, :olr_appr_status,
          :olr_appr_user, :olr_appr_datetime, :olr_appr_rd_id, :olr_from_reg_id, olr_appr_status
  do
  begin
    n = n + 1;
    un_fio_reg = '';
    un_fio_appr = '';
    rd_name_reg = '';
    rd_name_appr = '';
    olr_reg_state_str = iif(olr_reg_state = 0, 'восстоновить', 'удалить');

    olr_appr_status_str = (case olr_appr_status
                             when 0 then 'в ожидании'
                             when 1 then 'подтвержден'
                             when 2 then 'отклонен'

                             else 'неизвестно'
                           end);

    select u.un_fio
    from username u
    where u.un_id = :olr_reg_user
    into :un_fio_reg;

    select r.rd_name
    from refdata r
    where r.rd_id = :olr_reg_rd_id
    into :rd_name_reg;

    if (olr_appr_status <> 0) then
    begin
      select u.un_fio
      from username u
      where u.un_id = :olr_appr_user
      into :un_fio_appr;
      if (olr_appr_status = 1) then
      begin
        select r.rd_name
        from refdata r
        where r.rd_id = :olr_appr_rd_id
        into :rd_name_appr;
      end
    end

    ol_date = null;
    reg_name = '';
    select o.ol_date, r.reg_name
    from operlist o
    inner join region r on r.reg_unicode = o.ol_from_reg_id
    where o.ol_id = :olr_from_ol_id
    into :ol_date, reg_name;
    suspend;
  end
end
