CREATE PROCEDURE AAREP_CLIENT_ACCOUNT_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE,
    REPORT_BALANCE_ID VARCHAR(20))
RETURNS (
    OB_ACCOUNT VARCHAR(20),
    OB_CLIENT DECIMAL(18,0),
    OB_SUMMA DECIMAL(18,4),
    OB_REGION_ID INTEGER)
AS
begin
  if (report_balance_id = '') then
  begin
    for select ob_account_dt, ob_client_dt, sum(ob_summa_val), ob_ol_reg_id
        from operbook, region_list(:report_region_id, 1, -1, 1), acc_list, balgroup
        where ob_ol_date <= :report_date_end and
              al_id = ob_account_dt and
              al_from_bg_id = bg_id and
              bg_report = 1 and
              ob_ol_is_active = 1 and
              ob_ol_reg_id = id_list
        group by ob_account_dt, ob_client_dt, ob_ol_reg_id
        into :ob_account, :ob_client, :ob_summa, :ob_region_id
    do
    begin
      suspend;
    end

    for select ob_account_kt, ob_client_kt, sum(-1 * ob_summa_val), ob_ol_reg_id
        from operbook, region_list(:report_region_id, 1, -1, 1), acc_list, balgroup
        where ob_ol_date <= :report_date_end and
              al_id = ob_account_kt and
              al_from_bg_id = bg_id and
              bg_report = 1 and
              ob_ol_is_active = 1 and
              ob_ol_reg_id = id_list
        group by ob_account_kt, ob_client_kt, ob_ol_reg_id
        into :ob_account, :ob_client, :ob_summa, :ob_region_id
    do
    begin
      suspend;
    end
  end
  else
  begin
    for select ob_account_dt, ob_client_dt, sum(ob_summa_val), ob_ol_reg_id
        from operbook, region_list(:report_region_id, 1, -1, 1), acc_list, balgroup
        where ob_ol_date <= :report_date_end and
              ob_ol_is_active = 1 and
              ob_ol_reg_id = id_list and
              ob_account_dt = al_id and
              al_from_bg_id = bg_id and
              bg_report = 1 and
              al_from_bg_id = :report_balance_id
        group by ob_account_dt, ob_client_dt, ob_ol_reg_id
        into :ob_account, :ob_client, :ob_summa, :ob_region_id
    do
    begin
      suspend;
    end

    for select ob_account_kt, ob_client_kt, sum(-1 * ob_summa_val), ob_ol_reg_id
        from operbook, region_list(:report_region_id, 1, -1, 1), acc_list, balgroup
        where ob_ol_date <= :report_date_end and
              ob_ol_is_active = 1 and
              ob_ol_reg_id = id_list and
              ob_account_kt = al_id and
              al_from_bg_id = :report_balance_id and
              al_from_bg_id = bg_id and
              bg_report = 1
        group by ob_account_kt, ob_client_kt, ob_ol_reg_id
        into :ob_account, :ob_client, :ob_summa, :ob_region_id
    do
    begin
      suspend;
    end
  end
end
