Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE AAREP_LOAN_REPORT(
    REP_DATE_END DATE)
RETURNS (
    DISB_SUM_ALL CURR_TYPE,
    PERCENT_EFFECT_ALL CURR_TYPE,
    DISB_SUM_KLAS CURR_TYPE,
    PERCENT_EFFECT_KLAS CURR_TYPE,
    DISB_SUM_IPF CURR_TYPE,
    PERCENT_EFFECT_IPF CURR_TYPE,
    DISB_KOL_ALL INTEGER,
    DISB_KOL_KLASS INTEGER,
    DISB_KOL_IPF INTEGER,
    BALANCE_ALL CURR_TYPE,
    BAL_EFFECT_ALL CURR_TYPE,
    BALANCE_KLASS CURR_TYPE,
    BAL_EFFECT_KLASS CURR_TYPE,
    BALANCE_IPF CURR_TYPE,
    BAL_EFFECT_IPF CURR_TYPE,
    DISB_SUM_ALL_2 CURR_TYPE,
    PERCENT_EFFECT_ALL_2 CURR_TYPE,
    DISB_SUM_KLAS_2 CURR_TYPE,
    PERCENT_EFFECT_KLAS_2 CURR_TYPE,
    DISB_SUM_IPF_2 CURR_TYPE,
    PERCENT_EFFECT_IPF_2 CURR_TYPE,
    DISB_KOL_ALL_2 INTEGER,
    DISB_KOL_KLASS_2 INTEGER,
    DISB_KOL_IPF_2 INTEGER,
    BALANCE_ALL_2 CURR_TYPE,
    BAL_EFFECT_ALL_2 CURR_TYPE,
    BALANCE_KLASS_2 CURR_TYPE,
    BAL_EFFECT_KLASS_2 CURR_TYPE,
    BALANCE_IPF_2 CURR_TYPE,
    BAL_EFFECT_IPF_2 CURR_TYPE,
    DISB_SUM_ALL_3 CURR_TYPE,
    PERCENT_EFFECT_ALL_3 CURR_TYPE,
    DISB_SUM_KLAS_3 CURR_TYPE,
    PERCENT_EFFECT_KLAS_3 CURR_TYPE,
    DISB_SUM_IPF_3 CURR_TYPE,
    PERCENT_EFFECT_IPF_3 CURR_TYPE,
    DISB_KOL_ALL_3 INTEGER,
    DISB_KOL_KLASS_3 INTEGER,
    DISB_KOL_IPF_3 INTEGER,
    BALANCE_ALL_3 CURR_TYPE,
    BAL_EFFECT_ALL_3 CURR_TYPE,
    BALANCE_KLASS_3 CURR_TYPE,
    BAL_EFFECT_KLASS_3 CURR_TYPE,
    BALANCE_IPF_3 CURR_TYPE,
    BAL_EFFECT_IPF_3 CURR_TYPE,
    DISB_SUM_ALL_4 CURR_TYPE,
    PERCENT_EFFECT_ALL_4 CURR_TYPE,
    DISB_SUM_KLAS_4 CURR_TYPE,
    PERCENT_EFFECT_KLAS_4 CURR_TYPE,
    DISB_SUM_IPF_4 CURR_TYPE,
    PERCENT_EFFECT_IPF_4 CURR_TYPE,
    DISB_KOL_ALL_4 INTEGER,
    DISB_KOL_KLASS_4 INTEGER,
    DISB_KOL_IPF_4 INTEGER,
    BALANCE_ALL_4 CURR_TYPE,
    BAL_EFFECT_ALL_4 CURR_TYPE,
    BALANCE_KLASS_4 CURR_TYPE,
    BAL_EFFECT_KLASS_4 CURR_TYPE,
    BALANCE_IPF_4 CURR_TYPE,
    BAL_EFFECT_IPF_4 CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable k smallint;
declare variable month_kv smallint;
begin
-- данные на текущую дату
  rep_date_begin = rep_date_end;
  select disb_sum_all, percent_effect_all, disb_sum_klas, percent_effect_klas, disb_sum_ipf, percent_effect_ipf,
         disb_kol_all, disb_kol_klass, disb_kol_ipf, balance_all, bal_effect_all, balance_klass, bal_effect_klass,
         balance_ipf, bal_effect_ipf
  from aarep_loan_report_temp(:rep_date_begin, :rep_date_end)
  into :disb_sum_all, :percent_effect_all, :disb_sum_klas, :percent_effect_klas, :disb_sum_ipf, :percent_effect_ipf,
       :disb_kol_all, :disb_kol_klass, :disb_kol_ipf, :balance_all, :bal_effect_all, :balance_klass, :bal_effect_klass,
       :balance_ipf, :bal_effect_ipf;
--данные с начала месяца
  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;
  select disb_sum_all, percent_effect_all, disb_sum_klas, percent_effect_klas, disb_sum_ipf, percent_effect_ipf,
         disb_kol_all, disb_kol_klass, disb_kol_ipf, balance_all, bal_effect_all, balance_klass, bal_effect_klass,
         balance_ipf, bal_effect_ipf
  from aarep_loan_report_temp(:rep_date_begin, :rep_date_end)
  into :disb_sum_all_2, :percent_effect_all_2, :disb_sum_klas_2, :percent_effect_klas_2, :disb_sum_ipf_2,
       :percent_effect_ipf_2, :disb_kol_all_2, :disb_kol_klass_2, :disb_kol_ipf_2, :balance_all_2, :bal_effect_all_2,
       :balance_klass_2, :bal_effect_klass_2, :balance_ipf_2, :bal_effect_ipf_2;
--данные с начала квартала
  k = extract(month from rep_date_end);
  month_kv = case
               when k in (1, 2, 3) then 1
               when k in (4, 5, 6) then 4
               when k in (7, 8, 9) then 7
               else 10
             end;
  rep_date_begin = 1 || '.' || month_kv || '.' || extract(year from rep_date_end);

  select disb_sum_all, percent_effect_all, disb_sum_klas, percent_effect_klas, disb_sum_ipf, percent_effect_ipf,
         disb_kol_all, disb_kol_klass, disb_kol_ipf, balance_all, bal_effect_all, balance_klass, bal_effect_klass,
         balance_ipf, bal_effect_ipf
  from aarep_loan_report_temp(:rep_date_begin, :rep_date_end)
  into :disb_sum_all_3, :percent_effect_all_3, :disb_sum_klas_3, :percent_effect_klas_3, :disb_sum_ipf_3,
       :percent_effect_ipf_3, :disb_kol_all_3, :disb_kol_klass_3, :disb_kol_ipf_3, :balance_all_3, :bal_effect_all_3,
       :balance_klass_3, :bal_effect_klass_3, :balance_ipf_3, :bal_effect_ipf_3;

-- данные с начало года
  rep_date_begin = 1 || '.' || 1 || '.' || extract(year from rep_date_end);
  select disb_sum_all, percent_effect_all, disb_sum_klas, percent_effect_klas, disb_sum_ipf, percent_effect_ipf,
         disb_kol_all, disb_kol_klass, disb_kol_ipf, balance_all, bal_effect_all, balance_klass, bal_effect_klass,
         balance_ipf, bal_effect_ipf
  from aarep_loan_report_temp(:rep_date_begin, :rep_date_end)
  into :disb_sum_all_4, :percent_effect_all_4, :disb_sum_klas_4, :percent_effect_klas_4, :disb_sum_ipf_4,
       :percent_effect_ipf_4, :disb_kol_all_4, :disb_kol_klass_4, :disb_kol_ipf_4, :balance_all_4, :bal_effect_all_4,
       :balance_klass_4, :bal_effect_klass_4, :balance_ipf_4, :bal_effect_ipf_4;
  suspend;
end
