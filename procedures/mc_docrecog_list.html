CREATE PROCEDURE MC_DOCRECOG_LIST(
    MCL_ID D_ID,
    LIST_KIND D_STR_100 = null)
RETURNS (
    DRM_NAME D_STR_255,
    DRM_CODE D_STR_255,
    DRM_API_CODE D_STR_255,
    DRM_IS_MANUAL D_BOOL,
    DRM_RECOG_2 D_BOOL,
    DRM_SHOW_FIELDS_CLIENT D_BOOL,
    DRM_HAS_PAGE1 D_BOOL,
    DRM_HAS_PAGE2 D_BOOL,
    DRM_USE_FRONT_CAMERA D_BOOL,
    DRM_CAN_LOAD_GALLERY D_BOOL)
AS
declare variable c_citizen d_integer;
declare variable pk_id d_integer;
declare variable c_id d_id;
declare variable pdt_code d_str_100;
declare variable db_id d_integer;
declare variable drm_id d_integer;
declare variable mcl_project d_integer;
declare variable oivo_id d_integer;
declare variable is_oivo d_bool;
declare variable tmp_prior d_smallint;
begin
  -- Процедура возвращает список документов, которые может добавить клиент
  -- Возвращаются только документы, которые есть в drm_main (т.е. распознаваемые автоматически либо вручную)
  db_id = f_get_db_id();
  list_kind = trim(coalesce(list_kind, ''));
  list_kind = iif(list_kind = '', 'basic', list_kind);
  drm_use_front_camera = 0;

  if (list_kind not in ('basic', 'full')) then
    exception e_default 'parametr "list_kind" has invalid value';

  select mcl_from_c_id, mcl_project
  from mc_login
  where mcl_id = :mcl_id
  into c_id, mcl_project;

  c_id = coalesce(c_id, 0);

  oivo_id = f_enum('mcl_project', 'oivo', 'id');
  is_oivo = iif(mcl_project = :oivo_id, 1, 0);      --RFCRU-5804

  if (c_id = 0) then
  begin
    -- Если c_id не передан, возвращаем только паспорта (pdt_is_main_doc=1)
    for
      select pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pk_can_load
      from docrecogmain drm, passkind, passdoctype
      where :is_oivo = 1
        and drm_code in ('rus.passport.national')
        and drm_inactive = 0
        and pk_from_drm_id = drm_id
        and pk_from_pdt_id = pdt_id
        and pdt_is_main_doc = 1
        and drm_show_to_client = 1
        and drm.drm_show_in_client_mobile = 1

    union all

      select pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pk_can_load
      from docrecogmain drm, passkind, passdoctype
      where :is_oivo = 0
        and drm_inactive = 0
        and pk_from_drm_id = drm_id
        and pk_from_pdt_id = pdt_id
        and pdt_is_main_doc = 1
        and drm_show_to_client = 1
        and drm.drm_show_in_client_mobile = 1

    order by 1

    into drm_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, drm_can_load_gallery
    do
    begin
      select res_value
      from mc_doc_name_change(:pk_id, :drm_name)
      into :drm_name;

      select drm_has_page1, drm_has_page2
      from hp_get_drm_has_pages(:pk_id)
      into :drm_has_page1, :drm_has_page2;

      suspend;
    end

  end
  else
  begin
    -- если есть запрос на прикрепление документа, показываем только их
    for select distinct drm_id
    from hp_docrecogrequest_docs(:c_id)
    into :drm_id
    do
    begin
      --RFCRU-5804
      for
        select pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pk_can_load
        from docrecogmain drm, passkind
        where :is_oivo = 1
          and drm_id = :drm_id
          and drm_code in ('rus.passport.national', 'd_162', 'd_208', 'd_139', 'd_138', 'd_145', 'd_132', 'd_82', 'd_124', 'd_137', 'd_163')
          and pk_from_drm_id = drm_id

      union all

        select pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pk_can_load
        from docrecogmain drm, passkind
        where :is_oivo = 0
          and drm_id = :drm_id
          and pk_from_drm_id = drm_id

      into drm_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, drm_can_load_gallery
      do
      begin
        select res_value
        from mc_doc_name_change(:pk_id, :drm_name)
        into :drm_name;

        select drm_has_page1, drm_has_page2
        from hp_get_drm_has_pages(:pk_id)
        into :drm_has_page1, :drm_has_page2;

        suspend;
      end
    end

    if (drm_id is not null) then
      exit;

    if (list_kind = 'basic') then
    begin
      -- Если c_id указан, возвращаем документы с учетом страны клиента
      -- Выясняем гражданство чтобы отфильтровать документы по нему
      select c_citizen
      from clients
      where c_id = :c_id
      into c_citizen;

      c_citizen = coalesce(c_citizen, 0);

      --RFCRU-5804
      for
        select pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pk_can_load
        from docrecogmain drm, passkind, hp_citizen_passkind(:c_citizen)
        where :is_oivo = 1
          and drm_code in ('rus.passport.national')
          and drm_inactive = 0
          and pk_from_drm_id = drm_id
          and pk_id > 0
          and pk_id = res_pk_id
          and drm_show_to_client = 1
          and drm.drm_show_in_client_mobile = 1

      union all

        select pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pk_can_load
        from docrecogmain drm, passkind, hp_citizen_passkind(:c_citizen)
        where :is_oivo = 0
          and drm_inactive = 0
          and res_pk_id = pk_id
          and pk_from_drm_id = drm_id
          and pk_id > 0
          and pk_id = res_pk_id
          and drm_show_to_client = 1
          and drm.drm_show_in_client_mobile = 1

      order by 1

      into drm_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, drm_can_load_gallery
      do
      begin
        select res_value
        from mc_doc_name_change(:pk_id, :drm_name)
        into :drm_name;

        select drm_has_page1, drm_has_page2
        from hp_get_drm_has_pages(:pk_id)
        into :drm_has_page1, :drm_has_page2;

        suspend;
      end
    end
    else
    begin
      --RFCRU-5804
      for
        select iif(pdt_code = 'selfie_photo', 0, 1), pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pdt_code, pk_can_load
        from docrecogmain
          inner join passkind on pk_from_drm_id = drm_id
          inner join passdoctype on pk_from_pdt_id = pdt_id
        where :is_oivo = 1
          and drm_code in ('d_162', 'd_208', 'd_139', 'd_138', 'd_145', 'd_132', 'd_82', 'd_124', 'd_137', 'd_163')
          and drm_inactive = 0
          and pk_id > 0
          and drm_show_in_client_mobile = 1

      union all

        select iif(pdt_code = 'selfie_photo', 0, 1), pk_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pdt_code, pk_can_load
        from docrecogmain
          inner join passkind on pk_from_drm_id = drm_id
          inner join passdoctype on pk_from_pdt_id = pdt_id
        where :is_oivo = 0
          and drm_inactive = 0
          and pk_id > 0
          and drm_show_in_client_mobile = 1
          and drm_code not in ('kgz.id.type2', 'uzb.passport.type3', 'rus.passport.national','kgz.passport.type1', 'tjk.passport.type1', 'kgz.id.type1', 'rus.pts.type1','uzb.passport.type1', 'aze.id.type1', 'aze.id.type1')

      order by 1, 2

      into tmp_prior, drm_name, drm_code, drm_api_code, drm_is_manual, drm_recog_2, drm_show_fields_client, pk_id, pdt_code, drm_can_load_gallery
      do
      begin
        drm_use_front_camera = 0;

        if (db_id = 81001 and pdt_code = 'selfie_photo') then
        begin
          drm_use_front_camera = 1;

          --RFCRU-5799
          -- Проверка на активную заявку — только для МБулак, поведение не меняется
          if (:is_oivo = 0
            and not exists
            (
              select *
              from miniapp, miniappclient
              where ma_is_active = 1
                and ma_date_end = '31.12.9999'
                and ma_id = mac_from_ma_id
                and mac_from_c_id = :c_id
                and mac_amount > 0
            )
          ) then
            continue; -- Если нет активной заявки, не даем Фото-идентификацию

        end

        select res_value
        from mc_doc_name_change(:pk_id, :drm_name)
        into :drm_name;

        select drm_has_page1, drm_has_page2
        from hp_get_drm_has_pages(:pk_id)
        into :drm_has_page1, :drm_has_page2;

        suspend;
      end

    end

  end

end
