CREATE PROCEDURE CB_RU_LO_201410_ONE_TMP(
    REP_REGION_ID INTEGER,
    REP_C_ID DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(20),
    AAREP_PLAN_ALL DECIMAL(18,4),
    AAREP_FACT_ALL DECIMAL(18,4),
    AAREP_FACT_PLANNED DECIMAL(18,4),
    AAREP_FACT_OTHER DECIMAL(18,4),
    AAREP_MONTH_PLAN_OS DECIMAL(18,4),
    AAREP_MONTH_PLAN_PERC DECIMAL(18,4),
    AAREP_DELINQ_OS DECIMAL(18,4),
    AAREP_DELINQ_PERC DECIMAL(18,4),
    AAREP_FACT_OS DECIMAL(18,4),
    AAREP_FACT_PERC DECIMAL(18,4),
    AAREP_FACT_FINE DECIMAL(18,4),
    AAREP_FACT_DELINQ_OS DECIMAL(18,4),
    AAREP_FACT_DELINQ_PERC DECIMAL(18,4),
    AAREP_FACT_ADV_OS DECIMAL(18,4),
    AAREP_FACT_ADV_PERC DECIMAL(18,4),
    AAREP_OKLAD DECIMAL(18,4),
    AAREP_PORT_BALANCE_OWN DECIMAL(18,4),
    AAREP_PORT_KOLICH_OWN INTEGER,
    AAREP_BONUS_OS DECIMAL(18,4),
    AAREP_BONUS_OS_ADV DECIMAL(18,4),
    PR_CODE STR_100,
    AAREP_BONUS_OS_ZAL_CAR CURR_TYPE)
AS
declare variable dd_disb_date date;
declare variable dd date;
declare variable k_stop_perc date;
declare variable sum1 decimal(18,4) = 0;
declare variable sum2 decimal(18,4) = 0;
declare variable ks_date_end date;
declare variable rep_month_end date;
declare variable sh_dd date;
declare variable sum3 decimal(18,4) = 0;
declare variable sum4 decimal(18,4) = 0;
declare variable is_sb integer;
declare variable kal_date date;
declare variable ks_kc_id decimal(18,0);
declare variable can_suspend integer;
declare variable kf_id decimal(18,0);
declare variable ok_with_agent integer;
declare variable last_sh_date date;
declare variable advanced_days integer;
declare variable new_policy_date date;
declare variable pr_id integer;
declare variable pr_id_prior integer;
declare variable aarep_bonus_os_not_zal_car curr_type;
declare variable ks_from_ol_id id_type;
declare variable loyal_calc_date date;
begin
  if (rep_date_begin is null) then
    rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;
  rep_month_end = rep_date_end - extract(day from :rep_date_end) + 1;
  rep_month_end = dateadd(month, 1, rep_month_end) - 1;

  aarep_port_balance_own = 0;
  aarep_port_kolich_own = 0;

  advanced_days = 0;
  if (rep_date_end >= '01.02.2017') then
    advanced_days = 15;
  if (rep_date_end >= '01.04.2017') then
    advanced_days = 10;

  new_policy_date = '31.01.2017'; -- Дата действия изменений

-- оклад сотрудника
  aarep_oklad = 0;
  select sum(sao_summa)
  from sa_oklad
  where sao_date <= :rep_date_end and
        sao_from_c_id = :rep_c_id
  into :aarep_oklad;
  if (aarep_oklad is null) then
    aarep_oklad = 0;
  pr_id_prior = 0;
  for select distinct ks_from_k_id, k_kontract_no, ks_date_begin, ks_date_end, k_stop_perc, k_kolich_client_approved,
                      ks_kc_id, kn1.kal_date, k_from_kf_id, k_from_pr_id, ks_from_ol_id
      from kontstate, kontract, knaktlist kn1
      where ks_ol_is_delete = 0 and
            ks_date_end >= (:rep_date_begin - 1) and
            ks_date_begin <= :rep_date_end and
            ks_state = 1 and
            kn1.kal_from_kf_id = k_from_kf_id and
            ks_from_k_id = k_id and
            kn1.kal_date <= :rep_date_end and
            kn1.kal_date_end >= :rep_date_end and
            kn1.is_delete = 0
--  and k_id=101010001252800201
--  and kal_new_reg_id=:rep_region_id
            and
            kn1.kal_new_kf_officer = :rep_c_id
--  and (kn1.kal_date<=maxvalue(:rep_date_begin, ks_date_begin) or ks_kc_id=kn1.kal_new_kf_officer)
      into :k_id, :k_kontract_no, :dd_disb_date, :ks_date_end, :k_stop_perc, :aarep_port_kolich_own, :ks_kc_id,
           :kal_date, :kf_id, pr_id, ks_from_ol_id
  do
  begin
    ok_with_agent = 0;
    can_suspend = 1;
    if (ks_kc_id <> :rep_c_id and
        not exists(select *
                   from krexpert
                   where ke_status = 1 and
                         ke_from_p_id = :rep_c_id and
                         ke_teacher_id = :ks_kc_id and
                         ke_teacher_date_end >= :kal_date -- до этого было :rep_date_end, при передаче наставник действующий должен быть
        )) then
    begin

      if (kal_date >= maxvalue(:rep_date_begin, :dd_disb_date)) then
      begin
        if (not(exists(select *
                       from knaktlist
                       where kal_from_kf_id = :kf_id and
                             is_delete = 0 and
                             kal_new_kf_officer = :rep_c_id and
                             kal_date <= (:kal_date - 1) and
                             kal_date_end >= (:kal_date - 1)))) then
          can_suspend = 0;
      end

    end
    if (can_suspend = 1) then
    begin
      if (pr_id_prior <> pr_id) then
      begin
        pr_id_prior = pr_id;
        select pr_code from product
        where pr_id = :pr_id
        into pr_code;
      end

      dd = maxvalue(:rep_date_begin, (dd_disb_date + 1));

      aarep_delinq_os = 0;
      aarep_delinq_perc = 0;
      aarep_month_plan_os = 0;
      aarep_month_plan_perc = 0;

      aarep_fact_all = 0;
      aarep_fact_other = 0;
      aarep_fact_planned = 0;
      aarep_plan_all = 0;

      aarep_fact_os = 0;
      aarep_fact_perc = 0;
      aarep_fact_fine = 0;
      aarep_fact_delinq_os = 0;
      aarep_fact_delinq_perc = 0;
      aarep_fact_adv_os = 0;
      aarep_fact_adv_perc = 0;

      aarep_bonus_os = 0;
      aarep_bonus_os_adv = 0;

      aarep_bonus_os_not_zal_car = 0;
      aarep_bonus_os_zal_car = 0;

      aarep_port_balance_own = 0;
      select oks_balance
      from oksummary
      where oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_from_k_id = :k_id and
            (:rep_date_end - oks_delinq_date <= 360) and
            oks_close_state = 0
      into :aarep_port_balance_own;
      if (aarep_port_balance_own is null) then
        aarep_port_balance_own = 0;
      if (aarep_port_balance_own = 0) then
      begin
        aarep_port_kolich_own = 0;
      end

      is_sb = 0;
      /*
        if (aarep_port_kolich_own=0) then
        begin
            if (exists(select * from sb_akt, sb_list
              where sba_from_k_id = :k_id
              and sba_status = 2
              and :rep_date_end between sba_date_begin and sba_date_end
              and sba_is_delete = 0
              and sba_from_sbl_id<>0
              and sba_from_sbl_id = sbl_id
              and sbl_type = 1)) then is_sb=1;
        end
        */
      if (is_sb = 0) then
      begin

        loyal_calc_date = null;
        select ogr_ol_date
        from operagr
        where ogr_from_ol_id = :ks_from_ol_id and
              ogr_sign = 1 and
              ogr_amount > 0
        into loyal_calc_date;

        if (loyal_calc_date is not null) then ok_with_agent = 2;
        else ok_with_agent = 0;
-- если выдан через агента и выдан с 01.05.2015, то за первое погашение не платим бонус
--          if ((dd_disb_date>='01.05.2015') and (REP_DATE_END>='29.06.2015') and (REP_DATE_END<'01.06.2016')) then
--          begin
--            for select (select cont_value from HP_CONTDATA_GET_cell_main(0, kc_from_c_id))
--            from knclient
--            where kc_from_k_id=:k_id
--            and kc_amount_approved>0
--            into :c_plastcard_no
--            do
--            begin
--              if (exists(select *
--              from smsdialogsend
--              where sds_action='Kupon_register'
--              and sds_cell_no=:c_plastcard_no
--              and (sds_text like 'Ваш%' or sds_text like 'Поздравляем! %')
--              and sds_from_c_id is not null
--              and sds_create_datetime between (:dd_disb_date-30) and (:dd_disb_date+1))) then ok_with_agent=1;
--            end
--          end

-- Просроченные платежи по кредиту за период
        for select ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc),
                   sum(iif(ok_ol_date < :dd, ok_os + ok_sh_os, 0)),
                   sum(iif(ok_ol_date < :dd, ok_perc + ok_sh_perc, 0))
            from operkred
            where ok_from_k_id = :k_id and
                  ok_ol_date <= :rep_date_end and
                  ok_sh_date between :rep_date_begin and :rep_month_end and
                  ok_ol_is_active = 1 and
                  ok_type < 7 and
                  ok_from_c_id = 0
            group by ok_sh_date
            into :sh_dd, :sum1, :sum2, sum3, sum4
        do
        begin
          if (sum1 is null) then
            sum1 = 0;
          if (sum2 is null) then
            sum2 = 0;
          if (sum3 is null) then
            sum3 = 0;
          if (sum4 is null) then
            sum4 = 0;
          aarep_month_plan_os = aarep_month_plan_os + sum3;
          aarep_month_plan_perc = aarep_month_plan_perc + sum4;
          if (sh_dd between :rep_date_begin and :rep_date_end) then
          begin
            aarep_delinq_os = aarep_delinq_os + sum1;
            aarep_delinq_perc = aarep_delinq_perc + sum2;

            if ((sh_dd = '25.11.2014') and
                (k_stop_perc <= '25.11.2014')) then -- не начислять штрафы если дата возникновения 25.11.2014
            begin
              aarep_delinq_os = aarep_delinq_os - sum1;
              aarep_delinq_perc = aarep_delinq_perc - sum2;
            end
          end
        end
        select -sum(iif(ok_sh_date between :rep_date_begin and :rep_month_end, ok_os, 0.0)), -- своевременные ОС
               -sum(iif(ok_sh_date between :rep_date_begin and :rep_month_end, ok_perc, 0.0)), -- своевременные %%
               -sum(ok_fine), -- пени
               -sum(iif(ok_sh_date < :rep_date_begin, ok_os, 0.0)), -- погашение просроченных ОС
               -sum(iif(ok_sh_date < :rep_date_begin, ok_perc, 0.0)), -- погашение просроченных %%
               -sum(iif(ok_sh_date > :rep_month_end, ok_os, 0.0)), -- досрочное погашение ОС
               -sum(iif(ok_sh_date > :rep_month_end, ok_perc, 0.0)), -- досрочное погашение %%
               -sum(iif(ok_sh_date >= :rep_date_begin and datediff(day, ok_ol_date, ok_sh_date) > :advanced_days, ok_os, 0.0)), -- досрочное погашение ОС (более чем 10 дней)
               -sum(iif(ok_sh_date >= :rep_date_begin and datediff(day, ok_ol_date, ok_sh_date) <= :advanced_days, ok_os, 0.0)), -- своевременно погашение ОС (не более чем 10 дней)
               -sum(iif(:pr_code not containing '.zalog_car.' and ok_sh_date >= :rep_date_begin and datediff(day, ok_ol_date, ok_sh_date) <= :advanced_days, ok_os, 0.0)), -- своевременно погашение ОС (не более чем 10 дней) кроме микрозайм залог
               -sum(iif(:pr_code containing '.zalog_car.' and ok_sh_date >= :rep_date_begin and datediff(day, ok_ol_date, ok_sh_date) <= 30, ok_os, 0.0)) -- своевременно погашение ОС (не более чем 30 дней) микрозайм залог
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date between :rep_date_begin and :rep_date_end and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_from_c_id = 0 and
              ok_sh_id > :ok_with_agent -- При расчете дохода не учитываются: погашения за первые два платежа по графику от клиентов, привлеченных агентами;
        into :aarep_fact_os, :aarep_fact_perc, :aarep_fact_fine, :aarep_fact_delinq_os, :aarep_fact_delinq_perc,
             :aarep_fact_adv_os, :aarep_fact_adv_perc, :aarep_bonus_os_adv, :aarep_bonus_os,
             :aarep_bonus_os_not_zal_car, aarep_bonus_os_zal_car;

        if (aarep_fact_os is null) then
          aarep_fact_os = 0;
        if (aarep_fact_perc is null) then
          aarep_fact_perc = 0;
        if (aarep_fact_fine is null) then
          aarep_fact_fine = 0;
        if (aarep_fact_delinq_os is null) then
          aarep_fact_delinq_os = 0;
        if (aarep_fact_delinq_perc is null) then
          aarep_fact_delinq_perc = 0;
        if (aarep_fact_adv_os is null) then
          aarep_fact_adv_os = 0;
        if (aarep_fact_adv_perc is null) then
          aarep_fact_adv_perc = 0;

        if (aarep_bonus_os_adv is null) then
          aarep_bonus_os_adv = 0;

        if (aarep_bonus_os is null) then
          aarep_bonus_os = 0;

        if (aarep_bonus_os_not_zal_car is null) then
          aarep_bonus_os_not_zal_car = 0;

        if (aarep_bonus_os_zal_car is null) then
          aarep_bonus_os_zal_car = 0;
-- если кредит закрыт за период, обнуляем сумму досрочных погашений
        if (rep_date_end <= :new_policy_date) then
        begin
          if (ks_date_end <= :new_policy_date and
              dateadd(day, 1, ks_date_end) between :rep_date_begin and :rep_date_end) then
          begin
            select max(ok_sh_date)
            from operkred, kontstate
            where ks_from_k_id = :k_id and
                  ks_state = 2 and
                  ks_ol_is_delete = 0 and
                  ks_from_ol_id = ok_from_ol_id
            into :last_sh_date;
            if (last_sh_date is null) then
              last_sh_date = rep_date_end;

            if (last_sh_date > (:ks_date_end + 10) and
                (extract(year from last_sh_date) * 100 + extract(month from last_sh_date) > extract(year from ks_date_end) * 100 + extract(month from ks_date_end))) then -- Если кредит закрыт досрочно более 10 дней
            begin
              aarep_fact_adv_os = 0.0;
              aarep_delinq_os = 0;
              aarep_delinq_perc = 0;
            end
          end
          aarep_fact_all = aarep_fact_os + aarep_fact_perc + aarep_fact_fine + aarep_fact_delinq_os + aarep_fact_delinq_perc + aarep_fact_adv_os + aarep_fact_adv_perc;
        end
        else
        begin
          if (rep_date_begin < '01.07.2019') then
          begin
            aarep_fact_all = aarep_bonus_os --  выплаченная сумма не более NN дней
            + aarep_fact_perc + aarep_fact_fine + aarep_fact_delinq_os + aarep_fact_delinq_perc + aarep_fact_adv_perc;
          end else
          begin
            aarep_bonus_os = aarep_bonus_os_not_zal_car;
            aarep_fact_all = aarep_bonus_os_not_zal_car + aarep_bonus_os_zal_car
            + aarep_fact_perc + aarep_fact_fine + aarep_fact_delinq_os + aarep_fact_delinq_perc + aarep_fact_adv_perc;
          end
        end

        aarep_fact_other = aarep_fact_fine + aarep_fact_delinq_os + aarep_fact_delinq_perc + aarep_fact_adv_os + aarep_fact_adv_perc;

        aarep_fact_planned = aarep_fact_os + aarep_fact_perc;
        aarep_plan_all = aarep_delinq_os + aarep_delinq_perc + aarep_fact_planned; -- все запланированные = выплаченные + новые просрочки
      end
      suspend;
    end
  end
end
