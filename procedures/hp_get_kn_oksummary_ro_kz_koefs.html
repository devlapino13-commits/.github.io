CREATE PROCEDURE HP_GET_KN_OKSUMMARY_RO_KZ_KOEFS(
    OKS_DATE_BEGIN D_DATE,
    OKS_LAST_SH_DATE D_DATE,
    OKS_DELINQ_DAYS D_INTEGER,
    OKS_IS_RESCHEDULE D_INTEGER,
    OKS_DISB_DATE D_DATE,
    OKS_DATE_STOP_PERC D_DATE,
    MAX_DELINQ_DAYS D_INTEGER,
    OKS_RESTRUCT_DATE D_DATE)
RETURNS (
    OKS_ECL_PERIOD_KOEF D_PERCENT,
    OKS_ECL_KIND D_INTEGER,
    OKS_ECL_ACC_KIND D_INTEGER,
    OKS_ECL_PERCENT D_PERCENT)
AS
declare variable ECL_PD_PERCENT PERC_TYPE;
declare variable ECL_LGD_PERCENT PERC_TYPE;
declare variable TMP_DELINQ_DAYS D_INTEGER;
declare variable ECL_PD_MACROECONOM PERC_TYPE;
declare variable ADD_KOEF D_PERCENT;
begin
  if (oks_disb_date is null) then
    oks_disb_date = current_date;

  -- После обновления ставок здесь, нужно запустить процедуру HP_DO_KN_OKSUMMARY_ALL_RECALC

  -- Вычисление коффициента по сроку до конца займа для Резерва по ЭСП
  if (oks_delinq_days > 0 or
      oks_is_reschedule = 1) then
    oks_ecl_period_koef = 1.0;
  else
  begin
    oks_ecl_period_koef = datediff(day, oks_date_begin, oks_last_sh_date);
    oks_ecl_period_koef = oks_ecl_period_koef / 365.25;
    if (oks_ecl_period_koef < 0 or
        oks_ecl_period_koef > 1) then
      oks_ecl_period_koef = 1;
  end

  tmp_delinq_days = oks_delinq_days;

  add_koef = 1.0000000;
  if (tmp_delinq_days = 0 and
      oks_is_reschedule = 1 and
      oks_restruct_date between '01.04.2020' and '31.05.2020')
  then
    add_koef = 1.5000000; -- письмо от Елены о 20.07.2020 16:17, если рестр. за апрель-май, оставить в перво корзине
  else
  begin
    -- начиная с 31.03.2019 учитывается реструктурированный займ или нет
    -- письмо от Елены от 26.03.2019 11:23
    if (oks_is_reschedule = 1 and
        tmp_delinq_days <= 30 and
        oks_date_begin >= '31.03.2019') then
      tmp_delinq_days = 31;
  end

  -- Определение ставки PD по ЭСП
    if (oks_date_begin >= '31.07.2026') then
       ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.73
                        when tmp_delinq_days <= 30 then 27.67
                        when tmp_delinq_days <= 60 then 61.62
                        when tmp_delinq_days <= 90 then 72.97
                        else 100.0
                      end;
  if (oks_date_begin >= '30.06.2026') then
       ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.78
                        when tmp_delinq_days <= 30 then 29.47
                        when tmp_delinq_days <= 60 then 61.62
                        when tmp_delinq_days <= 90 then 69.66
                        else 100.0
                      end;
  if (oks_date_begin >= '31.05.2026') then
       ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.78
                        when tmp_delinq_days <= 30 then 37.20
                        when tmp_delinq_days <= 60 then 64.35
                        when tmp_delinq_days <= 90 then 66.67
                        else 100.0
                      end;
  if (oks_date_begin >= '30.04.2026') then
       ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.83
                        when tmp_delinq_days <= 30 then 33.57
                        when tmp_delinq_days <= 60 then 57.41
                        when tmp_delinq_days <= 90 then 69.23
                        else 100.0
                      end;
  if (oks_date_begin >= '31.03.2026') then
       ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.83
                        when tmp_delinq_days <= 30 then 29.04
                        when tmp_delinq_days <= 60 then 58.33
                        when tmp_delinq_days <= 90 then 64
                        else 100.0
                      end;
   if (oks_date_begin >= '28.02.2026') then
       ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.85
                        when tmp_delinq_days <= 30 then 31.77
                        when tmp_delinq_days <= 60 then 60
                        when tmp_delinq_days <= 90 then 78.31
                        else 100.0
                      end;
    if (oks_date_begin >= '31.01.2026') then
       ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.85
                        when tmp_delinq_days <= 30 then 39.25
                        when tmp_delinq_days <= 60 then 69.64
                        when tmp_delinq_days <= 90 then 75
                        else 100.0
                      end;
    else
   if (oks_date_begin >= '31.12.2025') then
       ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.85
                        when tmp_delinq_days <= 30 then 32.29
                        when tmp_delinq_days <= 60 then 57.63
                        when tmp_delinq_days <= 90 then 68.75
                        else 100.0
                      end;
   else
   if (oks_date_begin >= '30.11.2025') then
    ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.96
                        when tmp_delinq_days <= 30 then 29.52
                        when tmp_delinq_days <= 60 then 54.41
                        when tmp_delinq_days <= 90 then 75.51
                        else 100.0
                      end;
  else
  if (oks_date_begin >= '31.10.2025') then
    ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.73
                        when tmp_delinq_days <= 30 then 22.31
                        when tmp_delinq_days <= 60 then 61.76
                        when tmp_delinq_days <= 90 then 65
                        else 100.0
                      end;
  else
  if (oks_date_begin >= '30.09.2025') then
    ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.62
                        when tmp_delinq_days <= 30 then 28.30
                        when tmp_delinq_days <= 60 then 49.18
                        when tmp_delinq_days <= 90 then 61.54
                        else 100.0
                      end;
  else
  if (oks_date_begin >= '01.01.2025') then
    ecl_pd_percent = case
                        when tmp_delinq_days = 0 then 0.53
                        when tmp_delinq_days <= 30 then 23.21
                        when tmp_delinq_days <= 60 then 48.00
                        when tmp_delinq_days <= 90 then 66.67
                        else 100.0
                      end;
  else
    ecl_pd_percent = 0;

  ecl_pd_macroeconom = 0.0;
  ecl_pd_macroeconom = case
                         when oks_date_begin >= '31.01.2021' then 0
                         when oks_date_begin >= '31.12.2020' then 3.36
                         when oks_date_begin >= '30.06.2020' then 6.24
                         else 0.0
                       end;

  if (add_koef <> 1.0) then
    ecl_pd_percent = ecl_pd_percent * add_koef;

  ecl_pd_percent = ecl_pd_percent + ecl_pd_macroeconom;
  if (ecl_pd_percent > 100.0) then
    ecl_pd_percent = 100.0;

  -- Определение ставки LGD по ЭСП
  if (oks_delinq_days > 360 or
      (oks_date_begin >= '01.12.2019' and
       max_delinq_days > 360)) then
    ecl_lgd_percent = 100.0;
  else
  begin
    ecl_lgd_percent = case
                        when oks_date_begin >= '31.07.2026' then 80.52
                        when oks_date_begin >= '30.06.2026' then 85.82
                        when oks_date_begin >= '31.05.2026' then 86.66
                        when oks_date_begin >= '30.04.2026' then 76.10
                        when oks_date_begin >= '31.03.2026' then 80.39
                        when oks_date_begin >= '28.02.2026' then 81.46
                        when oks_date_begin >= '31.01.2026' then 82.05
                        when oks_date_begin >= '31.12.2025' then 92.63
                        when oks_date_begin >= '30.11.2025' then 87.15
                        when oks_date_begin >= '31.10.2025' then 86.32
                        when oks_date_begin >= '30.09.2025' then 89.50
                        when oks_date_begin >= '01.01.2025' then 81.61
                        else 0
                      end;
  end

  -- Определение ставки резерва по ЭСП
  if (oks_date_begin < oks_last_sh_date) then
  begin
    oks_ecl_kind = 1;
    oks_ecl_percent = round(cast(ecl_pd_percent as double precision) / 100.0 * ecl_lgd_percent * oks_ecl_period_koef, 2);

    -- с 31.10.2019 для займов с остаточным сроком более года применяется другая формула
    -- письмо от Елены от 24.10.2019 г.
    if (datediff(day, oks_date_begin, oks_last_sh_date) > 365 and
        oks_date_begin >= '31.10.2019') then
    begin
      if ((oks_delinq_days between 31 and 90) or
          (oks_delinq_days <= 90 and
          oks_is_reschedule = 1)) then
        oks_ecl_percent = round((1 - power(1 - cast(ecl_pd_percent as double precision) / 100.0, 2)) * ecl_lgd_percent, 2);
    end
  end
  else
  begin
    oks_ecl_kind = 2;
    oks_ecl_percent = ecl_lgd_percent;
  end

  oks_ecl_acc_kind = 3;

  if (oks_date_begin >= '01.12.2019' and
      max_delinq_days > 360) then
  begin
    oks_ecl_kind = 3;
    oks_ecl_percent = ecl_lgd_percent;
  end

  if (oks_ecl_kind = 1) then
  begin
    if (oks_date_begin >= '31.10.2019') then
    begin
      if (oks_delinq_days <= 30 and
          oks_is_reschedule = 0) then
        oks_ecl_acc_kind = 1;
      else
      if (oks_delinq_days <= 90) then
        oks_ecl_acc_kind = 2;
    end
    else
    begin
      if (oks_delinq_days <= 30) then
        oks_ecl_acc_kind = 1;
      else
      if (oks_delinq_days <= 90) then
        oks_ecl_acc_kind = 2;
    end
  end

  -- Если договор расторжен (дефолт), призначем как 3 категорию
  if (oks_date_stop_perc is not null and
      oks_date_stop_perc <= oks_date_begin) then
  begin
    oks_ecl_acc_kind = 3;
    oks_ecl_kind = 2;
    oks_ecl_percent = ecl_lgd_percent;
  end

  if (oks_ecl_acc_kind = 3 and
      oks_date_begin < oks_restruct_date and
      oks_date_begin >= oks_last_sh_date) then
    oks_ecl_acc_kind = 2;

  suspend;
end
