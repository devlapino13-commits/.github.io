CREATE PROCEDURE HP_EXTINFO_TO_SEND_KONTUR(
    C_ID D_ID,
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
RETURNS (
    JSON_VALUE D_STR_LONG)
AS
declare variable name_fio d_str_100;
declare variable birth_date d_date;
declare variable drm_code d_str_code;
begin
  -- Если это DocRecog, может быть два документа
  if (f_objkind(obj_kind, 'name') = 'docrecoglist') then
  begin
    select drm_code
    from docrecoglist, docrecogmain
    where drl_id = :obj_id
      and drl_from_drm_id = drm_id
    into drm_code;
    if (drm_code = 'rus.cert.name_change') then
    begin
      -- Свидетельство о перемене ФИО
      name_fio = f_docrecog_value(obj_id, 'client_name_old', null);
      select c_birth_date
      from clients
      where c_id = :c_id
      into birth_date;
    end
    else if (drm_code = 'rus.other.kontur') then
    begin
      -- Данные для проверки Контур/Федресурс
      name_fio = f_docrecog_value(obj_id, 'name_full', null);
      birth_date = f_docrecog_value(obj_id, 'birth_date', null);
    end
  end
  else -- В остальных случаях берем через clients
  begin
    select c_name_fio, c_birth_date
    from clients
    where c_id = :c_id
    into name_fio, birth_date;
  end

   -- RFCRU-5215
   if (c_id = 1010100204764) then
     name_fio = 'Юлдашев Аслам Юрьевна';

  if (name_fio is not null and birth_date is not null) then
  begin
    json_value = f_format('{%1,%2}',
      f_json_make_pair('fio', name_fio, 'str'),
      f_json_make_pair('birthdate', f_datetostr(birth_date), 'str'));
    suspend;
  end
end
