CREATE PROCEDURE CRIF_GET_KONT_DATA(
    POCKET_ID D_INTEGER)
RETURNS (
    KREDIT_TYPE D_STR_10,
    CONTRACT_PROCESS_DATE D_DATE,
    C_ID_STR D_STR_100,
    C_ROLE D_STR_10,
    K_ID_STR D_STR_100,
    CONTRACT_TYPE D_STR_10,
    OWNERSHIP_TYPE D_STR_10,
    CONTRACT_PHASE D_STR_10,
    CONTRACT_STATUS D_STR_10,
    PERCENT D_CURRENCY,
    CURRENCY D_STR_10,
    S_CURRENCY D_STR_10,
    DISB_DATE D_DATE,
    MA_DATE D_DATE,
    PLAN_CLOSE_DATE D_DATE,
    FACT_CLOSE_DATE D_DATE,
    LAST_REPAYMENT_DATE D_DATE,
    REORGANIZED_CREDIT D_STR_10,
    CREDIT_LIMIT D_CURRENCY,
    BALANCE_USE D_CURRENCY,
    LAST_DEBITED_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_KOLICH_REPAY D_INTEGER,
    PAYMENT_PERIOD D_STR_10,
    PAYMENT_METHOD D_STR_10,
    M_PAYMENT_AMOUNT D_CURRENCY,
    KOL_OUTSTANDING_PAYMENTS D_STR_10,
    K_BALANCE D_CURRENCY,
    S_OS D_CURRENCY,
    S_PERC D_CURRENCY,
    COUNT_DELINQ_PAY D_STR_10,
    KOL_OS_DELINQ_PAY D_INTEGER,
    KOL_PERC_DELINQ_PAY D_INTEGER,
    DELINQ_OS D_CURRENCY,
    DELINQ_PERC D_CURRENCY,
    PORUCH_FIO D_STR_100,
    REGION_ID D_STR_100,
    INSTALLMENTTYPE VARCHAR(10),
    COUNT_REMAIN_PAYMENTS D_STR_10,
    DELINQ_SUM D_CURRENCY)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable office_id_old d_id;
declare variable rep_date_end d_date;
declare variable closed_ol_id d_id;
declare variable kf_id d_id;
declare variable office_id bigint;
declare variable reg_ishenim_region decimal(15,2);
declare variable k_kolich_client_approved integer;
declare variable pr_wm_kolich decimal(15,2);
declare variable pr_week_or_month decimal(15,2);
begin
  office_id_old = 0;

  select cp.cp_date_end
  from cib_pocket cp
  where cp.cp_id = :pocket_id
  into :rep_date_end;

  if (rep_date_end is null) then
    exit;

  for select k_id,
             phaseofcontract,
             days_past_due_domain
      from crif_get_kontract_data_tmp(:pocket_id)
      into :k_id,
           :contract_phase,
           :count_delinq_pay
  do
  begin
    select iif(kont.k_currency = 417, 'KGS', ''),
           ceil(kont.k_amount_approved),
           kont.k_kolich_repay,
           kont.k_fact_disb_date,
           kont.k_kredline_end,
           iif(:contract_phase = 'CL', kont.k_fact_closed_ol_id, null),
           kont.k_from_kf_id,
           iif(:contract_phase = 'CL', kont.k_fact_closed_date, null),
           kont.k_percent,
           pr.pr_week_or_month,
           pr.pr_wm_kolich,
           kont.k_kolich_client_approved
    from kontract kont
    inner join product pr on pr.pr_id = kont.k_from_pr_id
    where kont.k_id = :k_id
    order by kont.k_currency
    into :currency,
         :k_amount_approved,
         :k_kolich_repay,
         :disb_date,
         :plan_close_date,
         :closed_ol_id,
         :kf_id,
         :fact_close_date,
         :percent,
         :pr_week_or_month,
         :pr_wm_kolich,
         :k_kolich_client_approved;

    s_currency = currency;
    credit_limit = :k_amount_approved;
    kredit_type = 'CI';

    contract_process_date = :rep_date_end;

    select ma.ma_date_oper
    from miniapp ma
    inner join kontract k on k.k_from_ma_app_id = ma.ma_id and
        k.k_id = :k_id
    where ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.9999'
    into :ma_date;
    
    select kal.kal_new_reg_id
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
        :rep_date_end between kal.kal_date and kal.kal_date_end and
        kal.is_delete = 0
    into :office_id;
    
    if (closed_ol_id is not null) then
    begin
        select ol.ol_date
        from operlist ol
        where ol.ol_id = :closed_ol_id and
              ol.is_delete = 0
        into :fact_close_date;
    end
    
    if (office_id <> office_id_old) then
    begin
        office_id_old = office_id;
        
        select reg_ishenim_region
        from region
        where reg_unicode = :office_id
        into reg_ishenim_region;
        
        region_id = null;

        select rd_tag
        from refdata
        where rd_id = :reg_ishenim_region
        into region_id;

        region_id = coalesce(region_id, 'Other');
    end

    
    k_id_str = k_id;
    ownership_type = iif(:k_kolich_client_approved > 1, '2', '1');
    contract_type = '33'; --  потребительский кредит 88 - кредитная карта

    /*    if (k_disb_kind = 6) then contract_type = '88';  */
    
    /*
    ID Individual Физическое лицо
    BD Company Юридическое лицо
    
    SL Subject Link Связь с субъектом
    CI Installment В рассрочку
    CN Non Installment Не в рассрочку
    CC Credit Card Кредитная карта
    CS Services Усгуги
    NE Subject Negative Event Негативное событие по субъекту
    
    */
    
    if (kredit_type = 'CI') then
        installmenttype = 'F';
    else
        installmenttype = 'V';
    
    select (sh.sh_os + sh.sh_perc)
    from schedule sh
    where sh.sh_from_k_id = :k_id and
        sh.sh_kommis = 0 and
        ((sh.sh_os <> 0) or
        (sh.sh_perc <> 0))
    order by sh.sh_date
    rows 1
    into :m_payment_amount;
    
    select max(ok.ok_ol_date)
    from operkred ok
    where ok.ok_from_k_id = :k_id and
        ok.ok_type = 4 and
        ok.ok_ol_is_active = 1 and
        -(ok.ok_os + ok.ok_perc) <> 0 and
        ok.ok_ol_date <= :rep_date_end
    into last_repayment_date;
    
    last_debited_date = :last_repayment_date;

    if (:fact_close_date < last_repayment_date) then
        fact_close_date = last_repayment_date;
    
    delinq_os = 0;
    delinq_perc = 0;
    k_balance = 0;
    s_os = 0;
    s_perc = 0;
    balance_use = 0;
    kol_os_delinq_pay = 0;
    kol_outstanding_payments = 0;
    kol_perc_delinq_pay = 0;
    count_remain_payments = 0;
    delinq_sum = 0;

    if (contract_phase = 'AC') then
    begin
        select sum(zad_os), sum(zad_perc), sum(zad_os + sh_calc_perc + sh_calc_fine)
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date is not null
        into :s_os, :s_perc, :balance_use;
        
        s_os = ceil(s_os);
        s_perc = ceil(s_perc);
        balance_use = ceil(balance_use);
        k_balance = balance_use;

        select count(sh_id)
        from aarep_make_zad_list(:k_id, :rep_date_end, 0, 0)
        into count_remain_payments;
        
        if (count_remain_payments > k_kolich_repay) then
          k_kolich_repay = count_remain_payments;
        
        if (count_delinq_pay <= '1') then
        begin
          delinq_os = 0;
          delinq_perc = 0;
          delinq_sum = 0;
        end
        if (count_delinq_pay > '1') then
        begin
          select count(sh_id),
                 sum(zad_os),
                 sum(zad_perc)
          from aarep_make_zad_list(:k_id, :rep_date_end, 0, 0)
          where sh_date < :rep_date_end
          into kol_outstanding_payments,
               delinq_os,
               delinq_perc;
        
          count_remain_payments = cast(:count_remain_payments as integer) - cast(:kol_outstanding_payments as integer);
        
          delinq_os = ceil(delinq_os);
          delinq_perc = ceil(delinq_perc);
          delinq_sum = ceil(delinq_os + delinq_perc);
          kol_perc_delinq_pay = kol_outstanding_payments;
        
          select count(sh_id)
          from aarep_make_zad_list(:k_id, :rep_date_end, 0, 0)
          where sh_date < :rep_date_end and
                zad_os > 0
          into kol_os_delinq_pay;
        end
        k_balance = ceil(k_balance);
    end
    payment_method = 'OTH';
    
    ----------- частота выплат -------------------------------------------------
    payment_period = 'M';
    if (pr_wm_kolich = 1 and
      pr_week_or_month = 1) then -- еженедельно
    payment_period = 'W';
    else
    if (pr_wm_kolich = 2 and
      pr_week_or_month = 1) then -- раз в две недели
    payment_period = 'F';
    else
    if (pr_wm_kolich = 1 and
      pr_week_or_month = 2) then -- ежемесячно
    payment_period = 'M';
    else
    if (pr_wm_kolich = 2 and
      pr_week_or_month = 2) then -- раз в два месяца
    payment_period = 'B';
    else
    if (pr_wm_kolich = 3 and
      pr_week_or_month = 2) then -- поквартально
    payment_period = 'Q';
    else
    if (pr_wm_kolich = 4 and
      pr_week_or_month = 2) then -- три раза год
    payment_period = 'T';
    else
    if (pr_wm_kolich = 6 and -- раз в полгода
      pr_week_or_month = 2) then
    payment_period = 'S';
    else
    if (pr_wm_kolich = 12 and -- ежегодно
      pr_week_or_month = 2) then
    payment_period = 'Y';
    
    ----------------------------------------------------------------------------
    reorganized_credit = '0';
    if (contract_phase = 'CL' and
      fact_close_date < plan_close_date) then
    contract_phase = 'CA'; -- погашен досрочно

    if (contract_phase = 'AC') then
    begin
        select is_restruct
        from aarep_get_restructuring(:k_id, :rep_date_end, 41701)
        into :reorganized_credit;
    end
    poruch_fio = '';
    for select kc.kc_from_c_id,
        iif(kc.kc_amount_approved > 0, 'B', 'G')
    from knclient kc
    where kc.kc_from_k_id = :k_id
    into c_id, c_role
    do
    begin
        c_id_str = :c_id;
        if (:ownership_type = '2' and :c_role = 'B') then
        begin
          if (not exists(select *
                         from miniapp ma
                         inner join kontract k on k.k_from_ma_app_id = ma.ma_id
                         inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id and
                               mac.mac_is_cashier = 1 and
                               mac.mac_from_c_id = :c_id
                         where ma.ma_is_active = 1 and
                               ma.ma_date_end >= '31.12.9999')) then
            :c_role = 'C';
        end
        
        if (:c_role <> 'B') then
        begin
          select c.c_name_fio
          from clients c
          where c.c_id = :c_id
          into poruch_fio;
        end

        suspend;
      end
  end
end
