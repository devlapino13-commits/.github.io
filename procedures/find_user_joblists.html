CREATE PROCEDURE FIND_USER_JOBLISTS
RETURNS (
    JL_ID INTEGER,
    JL_NAME VARCHAR(100),
    JL_MEMO VARCHAR(255),
    JL_USER_KOLICH INTEGER,
    JL_ORDER_NO INTEGER,
    JK_ID INTEGER)
AS
declare variable jl_tmp varchar(100);
declare variable c_id decimal(18,0);
begin
  for select jl_order_no, jl_id, jl_name, jk_name, jk_id, sum((select un_is_active
                                                               from hp_un_is_active(un_id)))
      from hr_pstatus, username, joblist, jobkind
      where ps_from_c_id = un_from_c_id and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id
      group by jl_order_no, jl_name, jl_id, jk_name, jk_id
      into :jl_order_no, :jl_id, :jl_name, :jl_memo, :jk_id, :jl_user_kolich
  do
  begin
    if (jk_id = 0) then
    begin
      jl_memo = '';
      jk_id = 9999;
    end
    if ((jl_user_kolich = 1) and
        (jl_memo = '')) then
    begin
      select c_id, c_name_fio
      from hr_pstatus, clients
      where ps_from_c_id = c_id and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_is_working = 1 and
            ps_from_jl_id = :jl_id rows 1
      into :c_id, :jl_memo;

      if (exists(select *
                 from hp_client_un_is_active(:c_id)
                 where un_is_active = 0)) then
        jl_memo = '';
    end
    else
    begin
      if (jl_memo = '') then
      begin
        for select c_id, c_name_fio
            from hr_pstatus, clients
            where ps_from_c_id = c_id and
                  ps_date_begin <= current_date and
                  ps_date_end >= current_date and
                  ps_is_working = 1 and
                  ps_from_jl_id = :jl_id rows 2
            into :c_id, :jl_tmp
        do
        begin
          if (exists(select *
                     from hp_client_un_is_active(:c_id)
                     where un_is_active = 1)) then
          begin
            if (jl_memo <> '') then
              jl_memo = jl_memo || ', ' || jl_tmp;
            else
              jl_memo = jl_tmp;
          end
        end
        jl_memo = jl_memo || '...';
      end
    end
    suspend;
  end
end
