CREATE PROCEDURE FHP_ROUND_BANK(
    AMT CURR_TYPE,
    RND_TO_DECIMAL INTEGER)
RETURNS (
    RES CURR_TYPE)
AS
declare variable rounded_amt curr_type;
declare variable whole_amt integer;
declare variable i integer;
declare variable pow curr_type;
begin
  res = amt;
  if (amt - trunc(amt) <> 0.5) then
    res = round(amt, rnd_to_decimal);
  else
  begin
    if (rnd_to_decimal = 0) then
    begin
      res = trunc(amt);
      if (mod(res, 2) = 1) then
        res = res + 1;
    end
  end
  suspend;
  exit;

-- ниже старый механизм, который не работал
  whole_amt = trunc(amt, 0);
  rounded_amt = amt - whole_amt;
  i = 4; -- Если тип переменных вместо Curr_type будет другим, то указать здесь кол-во знаков после запятой
  while (i > rnd_to_decimal) do
  begin
    i = i - 1;
    pow = power(10, i);
    if ((trunc(rounded_amt * pow * 10) - (trunc(rounded_amt * pow) * 10)) = 5 and
        mod((trunc(rounded_amt * pow) - (trunc(rounded_amt * pow / 10) * 10)), 2) = 0) then
      rounded_amt = trunc(rounded_amt, i);
    else
      rounded_amt = round(rounded_amt, i);
  end
  res = rounded_amt + whole_amt;
  suspend;
end
