CREATE PROCEDURE HP_CLIENTCOMLOG_LIST(
    CCR_ID D_INTEGER)
RETURNS (
    CCL_ID D_INTEGER,
    CCL_FROM_CCR_ID D_INTEGER,
    CCL_DATETIME D_DATETIME,
    CCL_LIMIT_KIND D_SMALLINT,
    CCL_ACTION_KIND D_SMALLINT,
    CCL_REASON_KIND D_SMALLINT,
    CCL_FROM_UN_ID D_ID,
    CCL_IS_AUTO D_BOOL,
    CCL_LIMIT_KIND_STR D_STR_CODE,
    CCL_ACTION_KIND_STR D_STR_CODE,
    CCL_REASON_KIND_STR D_STR_CODE)
AS
begin
  for select ccl_id, ccl_from_ccr_id, ccl_datetime, ccl_limit_kind,
           ccl_action_kind, ccl_reason_kind, ccl_from_un_id,
           f_enum_name('clientcom_limit', ccl_limit_kind),
           f_enum_name('clientcom_action', ccl_action_kind),
           f_enum_name('clientcom_reason', ccl_reason_kind)
      from clientcomlog
      where ccl_from_ccr_id = :ccr_id
      order by ccl_datetime desc
      into ccl_id, ccl_from_ccr_id, ccl_datetime, ccl_limit_kind,
           ccl_action_kind, ccl_reason_kind, ccl_from_un_id, ccl_limit_kind_str,
           ccl_action_kind_str, ccl_reason_kind_str
  do
  begin
    ccl_is_auto = iif(ccl_from_un_id is null, 1, 0);
    suspend;
  end
end
