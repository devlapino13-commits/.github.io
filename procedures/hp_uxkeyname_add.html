CREATE PROCEDURE HP_UXKEYNAME_ADD(
    TABLE_NAME D_STR_CODE)
AS
declare variable field_name d_str_code;
declare variable key_field d_str_code;
declare variable name_field d_str_code;
declare variable is_key_field d_bool;
declare variable key_field_count d_integer;
begin
  key_field = '';
  name_field = '';
  key_field_count = 0;

  if (exists(select * from uxkeyname
  where ukn_table_name = :table_name)) then
    exit;

  for select res_field_name, res_is_key_field
  from hp_table_fields(:table_name)
  into :field_name, :is_key_field
  do
  begin
    if (is_key_field = 1) then
    begin
      key_field_count = key_field_count + 1;

      if (key_field = '') then
        key_field = field_name;
    end

    if (name_field = '' and (field_name containing 'name' or field_name containing 'caption')) then
      name_field = field_name;
  end

  if (key_field_count <> 1) then
    exception e_message 'В таблице должно быть одно ключевое поле';

  if (name_field = '') then
    name_field = key_field;

  insert into uxkeyname (ukn_table_name, ukn_name_field)
  values (:table_name, :name_field);
end
