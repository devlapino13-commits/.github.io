CREATE PROCEDURE AAREP_LIST_OF_OFFICES(
    REP_REG_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    OFFICER_NAME D_STR_100,
    ADL_ADDRESS_STR D_STR_255,
    RD_NAME VARCHAR(20),
    C_NAME_FIO D_STR_255,
    CELL_NO D_STR_64,
    CONT_VALUE D_STR_64,
    REG_DATE_OPEN D_STR_100,
    REG_STAT_REP_REGION D_STR_100,
    C_NAME_FIO_RM D_STR_255,
    REG_PHONE D_STR_255)
AS
declare variable jl_id_list d_str_255;
declare variable jl_id_list_rm d_str_255;
declare variable reg_id d_id;
declare variable c_id d_id;
declare variable reg_office_hours_id d_id;
begin

  select '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jk.jk_id = jl.jl_kind
  where jk.jk_system_code = 'JobSystem.Region_KM' and
        jl.jl_id = 225
  into :jl_id_list;

  select '.' || list(jb.jl_id, '.') || '.'
  from joblist jb
  inner join jobkind jd on jd.jk_id = jb.jl_kind
  where jd.jk_system_code = 'JobSystem.Region_Director'
  and jd.jk_name = 'Директор региона'
  into :jl_id_list_rm;

  for select reg2.reg_name,
             reg.reg_code || ' ' || reg.reg_name,
             adl.adl_address_str,
             reg.reg_stat_rep_region,
             reg.reg_date_open,
             reg.reg_unicode,
             reg.reg_office_hours_id,
             reg.reg_phone
      from region reg
      inner join region_list(:rep_reg_id, 1, 3, 1) rl on reg.reg_unicode = rl.id_list
      inner join region reg2 on reg2.reg_unicode = reg.parent
      inner join addrlist adl on adl.adl_id = reg.reg_address_fact
      where reg2.reg_unicode > 0 and
            :rep_date_end between reg.reg_date_open and coalesce(reg.reg_date_close, '31.12.9999')
      into :reg_name,
           :officer_name,
           :adl_address_str,
           :reg_stat_rep_region,
           :reg_date_open,
           :reg_id,
           :reg_office_hours_id,
           :reg_phone

  do
  begin
     rd_name = '';
  select rd.rd_name
  from refdata rd
  where rd.rd_id = :reg_office_hours_id
  into :rd_name;

    c_name_fio_rm = '';

    select list(cli.c_name_fio, ';')
    from hr_pstatus hr
    inner join clients cli on hr.ps_from_c_id = cli.c_id
    where :jl_id_list_rm containing '.' || hr.ps_from_jl_id || '.' and
          :rep_date_end between hr.ps_date_begin and hr.ps_date_end and
          hr.ps_from_reg_id = :reg_id and
          hr.ps_is_working = 1 and
          hr.ps_status = 1 and
          cli.c_id not in (1090100160185,1090100038080)
    into :c_name_fio_rm;

    cell_no = '';
    cont_value = '';
    c_name_fio = '';

    for select cl.c_name_fio,
               cl.c_id
        from hr_pstatus ps
        inner join clients cl on cl.c_id = ps.ps_from_c_id
        where :jl_id_list containing '.' || ps.ps_from_jl_id || '.' and
              :rep_date_end between ps.ps_date_begin and ps.ps_date_end and
              ps.ps_is_working = 1 and
              ps.ps_status = 1 and
              ps.ps_from_reg_id = :reg_id
        into :c_name_fio,
             :c_id

    do
    begin

      cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');
      cont_value = f_cont_info('clients', :c_id, 'email', 'ok');

    end
    suspend;
  end
end
