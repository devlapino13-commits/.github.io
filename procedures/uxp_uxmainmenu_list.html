CREATE PROCEDURE UXP$UXMAINMENU_LIST(
    UN_ID D_ID,
    APP_NAME D_STR_CODE,
    PARENT_ID D_INTEGER = null)
RETURNS (
    ML_ID D_INTEGER,
    ML_PARENT_ID D_INTEGER,
    ML_CODE D_STR_100,
    ML_PARENT_CODE D_STR_100,
    ML_CAPTION D_STR_255,
    ML_HINT D_STR_LONG,
    ML_UA_CODE D_STR_CODE,
    ML_UA_UUID D_UUID,
    ML_INDEX D_INTEGER,
    ML_KIND D_INTEGER,
    ML_INACTIVE D_BOOL,
    ML_IMAGE_CODE D_STR_CODE)
AS
declare variable ctx_app_name d_str_code;
begin
  app_name = trim(coalesce(app_name, ''));
  if (app_name = '') then
    app_name = uxf$appname();

  if (app_name = '' or app_name = 'ibexpert') then
    app_name = 'FsConfig';

  ctx_app_name = rdb$get_context('USER_SESSION', 'APP_NAME');
  if (coalesce(ctx_app_name, '') = '') then
    rdb$set_context('USER_SESSION', 'APP_NAME', app_name);

  if (un_id is null) then
    un_id = uxf$user();

  -- Новая версия процедуры
  for select ml_id, ml_parent_id, ml_code, ml_parent_code, ml_caption, ml_hint, ml_ua_code, ml_ua_uuid, ml_index, ml_kind,
             ml_inactive, ml_image_code
  from uxp$uxmainmenu_list_new(:un_id, :app_name)
  order by ml_level, ml_index
  into :ml_id, :ml_parent_id, :ml_code, :ml_parent_code, :ml_caption, :ml_hint, :ml_ua_code, :ml_ua_uuid, :ml_index,
       :ml_kind, :ml_inactive, :ml_image_code
  do
  begin
    ml_caption = uxf$lang(ml_caption);
    ml_hint = uxf$lang(ml_hint);

    suspend;
  end
end
