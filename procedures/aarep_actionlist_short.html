CREATE PROCEDURE AAREP_ACTIONLIST_SHORT(
    REP_K_ID D_ID)
RETURNS (
    RES_OBJECT_NAME STR_255,
    RES_RD_NAME STR_255,
    RES_OBJECT_DETAIL BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_OBJECT_MEMO BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable acl_id d_integer;
begin
  select acl_id
  from actionlist
  where acl_object_kind = 9
  and acl_object_id = :rep_k_id
  and acl_status <> -3
  order by acl_id desc 
  rows 1
  into acl_id;

  for select res_object_name, res_rd_name, res_object_detail, res_object_memo
  from aarep_actionlist_full(:acl_id, '', 0, '', 0, 'yes')
  where acl_from_rd_id in (5781, 8787, 5782, 7425)
  into res_object_name, res_rd_name, res_object_detail, res_object_memo
  do
  begin
    suspend;
  end 
end
