CREATE PROCEDURE FM_JSON_PARSE(
    JSON_OBJECT BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    JSON_LEVEL INTEGER = 0,
    JSON_OBJECT_NAME VARCHAR(255) = '')
RETURNS (
    JSON_NAME VARCHAR(255),
    JSON_KEY_PARENT VARCHAR(255),
    JSON_KEY_FULL VARCHAR(255),
    JSON_KEY VARCHAR(100),
    JSON_VALUE VARCHAR(255))
AS
declare variable kol integer;
declare variable i integer;
declare variable len integer;
declare variable sym varchar(1);
declare variable n integer;
declare variable json_key_parent_orig varchar(255);
declare variable json_key_parent_tmp varchar(255);
declare variable opened_count integer;
declare variable j integer;
declare variable json_key_tmp varchar(100);
declare variable has_key integer;
declare variable sym_open varchar(1);
declare variable sym_close varchar(1);
declare variable json_key_tmp2 varchar(100);
declare variable isobject integer;
declare variable json_tmp blob sub_type 1 segment size 80;
begin
-- Процедура возвращает значение параметра json-объекта
-- по переданному json-объекту и имени параметра
  if (json_level is null) then
    json_level = 0;
  if (json_object_name is null) then
    json_object_name = '';
  json_name = '';

  isobject = 0;
  if (left(json_object, 1) = '{') then
    isobject = 1;

  if (json_level = 0) then
  begin
    if (left(json_object, 1) = '{') then
      json_object = substring(json_object from 2 for char_length(json_object) - 2);

    select new_json
    from fm_json_clear_parse(:json_object)
    into :json_object;
  end

  len = char_length(json_object);

  if (json_object_name <> '') then -- Если задано имя объекта, находим нужный кусок и остальное убираем
  begin
    n = position('"' || json_object_name || '":', json_object);
    if (n = 0) then
      exit;
    i = n + char_length('"' || json_object_name || '":');
    sym = substring(json_object from i for 1);
    sym_open = '';
    sym_close = '';
    if (sym = '{') then
    begin
      sym_open = '{';
      sym_close = '}';
    end
    else
    if (sym = '[') then
    begin
      sym_open = '[';
      sym_close = ']';
    end
    if (sym_open = '') then
      exit;

    opened_count = 1;
    j = 0;
    kol = 0;
    while (opened_count > 0 and i < len and j = 0) do
    begin
      i = i + 1;
      sym = substring(json_object from i for 1);
      if (sym = '"') then
      begin
        kol = kol + 1;
        if (kol = 4) then
          kol = 0;
      end
      else
      begin
        if (kol in (0, 2)) then
        begin
          if (sym = sym_close) then
          begin
            opened_count = opened_count - 1;
            if (opened_count = 0) then
              j = i;
          end
          else
          if (sym = sym_open) then
            opened_count = opened_count + 1;
        end
      end
    end
    if (j = 0) then
      exit;
    json_object = substring(json_object from n for (j - n) + 1);
  end

  json_tmp = '';

  json_name = '';
  json_key_parent = '';
  json_key_parent_orig = '';

  i = 0;
  kol = 0;
  json_value = '';

  if (left(json_object, 1) = '[') then
  begin
    has_key = 0;
    while (i < len and has_key = 0) do
    begin
      i = i + 1;
      sym = substring(json_object from i for 1);
      if (sym = '"') then
        kol = kol + 1;
      if (kol = 2) then
        kol = 0;
      if (kol = 0 and
          sym = ':') then
        has_key = 1;
    end
    if (has_key = 0) then
    begin
      json_key_parent = '';
      json_name = '';
      json_key = '';
      json_key_full = '';
      i = 0;
      while (i < len and has_key = 0) do
      begin
        i = i + 1;
        sym = substring(json_object from i for 1);
        if (sym = '"') then
          kol = kol + 1;
        if (kol = 2) then
        begin
          kol = 0;
          json_value = replace(json_value, '•', '"');
          suspend;
          json_value = '';
        end
        if (kol = 1 and
            sym <> '"') then
          json_value = json_value || sym;
      end
      exit;
    end
  end

  i = 0;
  kol = 0;
  json_key_parent = '';
  json_name = '';
  json_key = '';
  json_value = '';
  json_key_parent_tmp = '';

  while (i < len) do
  begin
    i = i + 1;
    sym = substring(json_object from i for 1);
    if (sym in ('{', '[') and
        kol = 2) then
    begin
      if (sym = '{') then
      begin
        sym_open = '{';
        sym_close = '}';
      end
      if (sym = '[') then
      begin
        sym_open = '[';
        sym_close = ']';
      end

      opened_count = 1;
      j = i;
      n = 0;
      while (opened_count > 0 and j < len and n = 0) do
      begin
        j = j + 1;
        sym = substring(json_object from j for 1);
        if (sym = '"') then
        begin
          kol = kol + 1;
          if (kol = 4) then
            kol = 0;
        end
        else
        begin
          if (kol in (0, 2)) then
          begin
            if (substring(json_object from j for 1) = sym_open) then
              opened_count = opened_count + 1;
            if (substring(json_object from j for 1) = sym_close) then
            begin
              opened_count = opened_count - 1;
              if (opened_count = 0) then
                n = j;
            end
          end
        end
      end

      if (n > 0) then
      begin
        json_tmp = substring(json_object from i for (n - i + 1));
        i = n + 1;
        len = char_length(json_object);
        kol = 0;
        json_key_parent = json_key;
        json_key_parent_orig = json_key_parent;

        for select json_key_parent, json_key, json_value
            from fm_json_parse(:json_tmp, :json_level + 1, '')
            into :json_key_parent_tmp, :json_key_tmp, :json_value
        do
        begin
          if (json_key_tmp <> '') then
            json_key = lower(json_key_tmp);

          json_key_parent = json_key_parent_orig;
          if (json_key_parent_tmp = '' and
              json_key_tmp = '') then
          begin
            json_key_parent = '';
            json_name = '';
            if (json_key <> '') then
              json_name = json_key;
          end
          if (json_key_parent_tmp <> '') then
            json_key_parent = json_key_parent || '.' || json_key_parent_tmp;
          json_value = replace(json_value, '•', '"');
          if (json_level = 0 and
              json_name = '') then
            json_name = json_key_parent;
          json_key_full = trim(json_key_parent || iif(json_key_parent <> '', '.', '') || json_key);
          if (json_name <> '' and
              json_key_parent <> '') then
            json_key_full = substring(json_key_full from char_length(json_name) + 2);
          suspend;

          if (json_key_parent_tmp = '' and
              json_key_tmp = '' and
              json_level = 0 and
              isobject = 0) then
          begin
            json_key = 'endofobject';
            json_value = '';
            json_value = replace(json_value, '•', '"');
            if (json_level = 0 and
                json_name = '') then
              json_name = json_key_parent;
            json_key_full = trim(json_key_parent || iif(json_key_parent <> '', '.', '') || json_key);
            if (json_name <> '' and
                json_key_parent <> '') then
              json_key_full = substring(json_key_full from char_length(json_name) + 2);
            suspend;
            json_key = lower(json_key_parent_orig);
          end
        end
        json_key_parent = '';
        json_name = '';
        json_key_parent_tmp = '';
        json_key = '';
        json_value = '';
      end
    end

    if (sym = '"') then
    begin
      kol = kol + 1;
      if (kol = 4) then
      begin
        kol = 0;
        json_value = replace(json_value, '•', '"');
        if (json_level = 0 and
            json_name = '') then
          json_name = json_key_parent;
        json_key_full = trim(json_key_parent || iif(json_key_parent <> '', '.', '') || json_key);
        if (json_name <> '' and
            json_key_parent <> '') then
          json_key_full = substring(json_key_full from char_length(json_name) + 2);
        suspend;
        json_name = '';
        json_key_parent_tmp = '';
        json_key_parent = '';
        json_key = '';
        json_value = '';
      end
    end
    else
    begin
      if (kol = 1) then
        json_key = lower(json_key || sym);
      if (kol = 3) then
        json_value = json_value || sym;
    end
  end
end
