CREATE PROCEDURE CM_KG_0008_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable kh_ownamount curr_type;
declare variable kc_max_amount curr_type;
declare variable c_curr_bonus_status integer;
declare variable pr_code d_str_100;
begin
-- Разработка по новому паспорту от 06.03.2013г.     -------------------------------------------------------------------------
--            for execute statement :sql_stat
    for
    select kc_from_c_id,
           c_name_fio,
           C_pass_no,
           kc_amount_approved,
           k_disb_date,
           pr_code
    from knclient, clients, kontract, product
    where kc_from_k_id = :k_id
    and kc_amount_approved is not null
    and kc_amount_approved > 50000
    and kc_from_c_id = c_id
    and kc_from_k_id = k_id
    and k_from_pr_id = pr_id
    and pr_product_kind <> 3
    into :c_id,
         :res_info_text,
         :kf_client_pass_no,
         :kc_amount,
         :kf_disb_date,
         :pr_code
    do
    begin
      if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
        exit;

      res_error_is_critical = 0;
      res_error_text = '';
      if (kc_amount > 200000) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Сумма участника не должно превышать 200тыс.';
      end else
      begin
--              Проверка есть ли кредитная история в дргом ФКУ
        kh_ownamount = null;  -- сумма выдачи
        select max(kh_ownamount) -- максимальное сумма в другом ФКУ
        from kibminiapp, kibhistory kh
        where kbm_from_c_id = :c_id
        and kh_from_kbm_id = kbm_id
        and kh.kh_amount > 0
        and kh_is_our = 0 -- 0 -кредит в другом ФКУ; 1 -кредит в своем ФКУ
        into kh_ownamount;
        if (kh_ownamount is null) then kh_ownamount = -1;    -- если -1 то не имее кредитной истории в другом ФКУ
--              если у кл. нету кред. истории то проверяет по таблице CLIENTHISTORY, и берет поледний полученный кредит
        if (kh_ownamount = -1) then
        begin
          select ch_amount from clienthistory
          where ch_from_c_id = :c_id
          order by ch_disb_date desc
          rows 1
          into kh_ownamount;
          if (kh_ownamount is null) then kh_ownamount = -1;
        end 
--              Проверка клиент новый или нет
        KC_MAX_AMOUNT = 0;
        select max(kc_amount_approved)
        from kontstate, knclient
        where ks_state = 1
        and ks_ol_is_delete = 0
        and ks_date_begin < current_date
        and ks_from_k_id <> :k_id
        and ks_from_k_id = kc_from_k_id
        and kc_from_c_id = :c_id
        and kc_amount_approved > 0
        into :KC_MAX_AMOUNT;
        if (KC_MAX_AMOUNT is null) then KC_MAX_AMOUNT=0;
        if (kc_max_amount=0) then   -- Если новый клиент то проверяет следующее
        begin
          if (kc_amount > 30000 and kh_ownamount = -1) then
          begin
            res_error_is_critical = 0;
            res_error_text = 'Клиент без кред. истории. не разрешено > 30 тыс.';
            suspend;
          end else
          begin
            if (kc_amount > 100000 and kh_ownamount < 100000) then
            begin
              res_error_is_critical = 0;
              res_error_text = 'У клиента в другом ФКУ было до 100 тыс. Не разрешено > 100 тыс.';
              suspend;
            end
          end
        end
        /*
        else
        begin -- разработан по новому паспорту от 10.03.2015 действующий с 1го апреля 2015г.
--              вычисляет максимальную сумму клиента из предыдущих циклов
--        Проверка статуса контракта
              C_CURR_BONUS_STATUS = null;
              select C_CURR_BONUS_STATUS
              from HP_KONTSTATE_LOAN_STATUS(:k_id, 0, 0, :kf_disb_date)
              into :C_CURR_BONUS_STATUS;
              if (:C_CURR_BONUS_STATUS is null) then C_CURR_BONUS_STATUS = -1;
              if (C_CURR_BONUS_STATUS <= 1 and kc_amount > 100000) then
              begin
                res_error_is_critical = 0; -- сделано не критичным до выяснение с Алтыной, сделано от 05.08.2015
                res_error_text = 'Статус группы ниже серебяного и не разрешено брать более 100тыс.';
                suspend;
              end else
              if (C_CURR_BONUS_STATUS <= 2 and kc_amount > 150000) then
              begin
                res_error_is_critical = 0; -- сделано не критичным до выяснение с Алтыной, сделано от 05.08.2015
                res_error_text = 'Статус группы ниже золотого и не разрешено брать более 150тыс.';
                suspend;
              end
        end  */
      end
    end
end
