CREATE PROCEDURE CNP_PAYMENT_REQUEST_LIST
RETURNS (
    CPR_ID D_ID,
    C_ID D_ID,
    AMOUNT D_CURRENCY,
    TOKEN STR_100,
    PAY_ACTION D_STR_CODE,
    RES_TEXT D_STR_255)
AS
declare variable K_ID D_ID;
begin
  for select cpr.cpr_id, cpr.cpr_from_c_id, cpr.cpr_amount, cpr.cpr_from_k_id, cpr.cpr_action
      from cnp_payment_requests cpr
      where cpr.cpr_status not in (-2, 2)
      into cpr_id, c_id, amount, k_id, pay_action
  do
  begin
    select et.et_token
    from extsys_token et
    where et.et_from_c_id = :c_id and
          et.et_is_actual = 1
    into token;
     -- token = '151408-99371222' ;   для теста
    select f_format('Выдача контракта %1 на дату %2', k.k_kontract_no, k.k_disb_date)
    from kontract k
    where k.k_id = :k_id
    into res_text;
    -- res_text = 'test';
    suspend;
  end
end
