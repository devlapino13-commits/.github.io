CREATE PROCEDURE HP_HIGH_MOUNTAIN_EXP_CALC(
    C_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE D_DATE)
RETURNS (
    MONTH_OUT D_INTEGER,
    LAST_DATE D_DATE)
AS
declare variable DATE_END_TMP D_DATE;
declare variable DATE_BEGIN_TMP D_DATE;
declare variable COUNTER D_INTEGER;
declare variable DAYS_DIFF D_INTEGER;
declare variable MONTH_TMP D_INTEGER;
declare variable DATE_BEGIN D_DATE;
declare variable DATE_END D_DATE;
declare variable PS_REG_ID D_INTEGER;
declare variable RMK_START_DATE D_DATE;
begin
  counter = 0;
  month_tmp = 0;

  for select ps.ps_date_begin, ps.ps_date_end, ps.ps_from_reg_id
      from hr_pstatus ps
      inner join region reg on reg.reg_unicode = ps.ps_from_reg_id and
            f_reg_mount_koef(reg.reg_unicode, :rep_date) > 1 -- RFCKG-4782
      inner join hr_orders ho on ps.ps_from_ho_id = ho.ho_id and
            ho.ho_status = 1
      where ps.ps_from_c_id = :c_id and
            ps.ps_status = 1 and
            ps.ps_is_working = 1 and
            ps.ps_date_begin >= :rep_date_begin and
            ps.ps_date_begin <= :rep_date
      order by ps.ps_date_begin
      into :date_begin_tmp, :date_end_tmp, :ps_reg_id

  do
  begin
    rmk_start_date = null;
    select min(rmk.rmk_date_begin)
    from regmountkoefs rmk
    where rmk.rmk_from_reg_id = :ps_reg_id and
          rmk.rmk_mount_koef > 1
    into :rmk_start_date;

    if (:rmk_start_date is not null and :rmk_start_date > :date_begin_tmp) then
      :date_begin_tmp = :rmk_start_date;

    if (:date_begin_tmp > :date_end_tmp) then
      continue;

    if (:date_begin_tmp > :rep_date) then
      continue;

    counter = counter + 1;

    if (:rep_date between :date_begin_tmp and :date_end_tmp) then
      :date_end_tmp = :rep_date;

    if (counter = 1) then
    begin
      date_begin = :date_begin_tmp;
      date_end = :date_end_tmp;
      last_date = date_begin - 1;
    end
    else
    begin
      if (date_begin_tmp <> :date_end + 1) then
      begin
        month_out = (select datediff(month, :date_begin - 1, :date_end)
                     from rdb$database);
        days_diff = extract(day from :date_end) - extract(day from (:date_begin - 1));
        if (:days_diff < 0) then
          month_out = month_out - 1;
        if (:days_diff >= 14) then
          month_out = month_out + 1;

        month_tmp = month_tmp + month_out;

        date_begin = :date_begin_tmp;
        last_date = date_begin - 1;
      end

      date_end = :date_end_tmp;
    end
  end

    if (date_end > :rep_date) then
      date_end = :rep_date;

    if (date_end < date_begin) then
      month_out = 1;

    if (date_end = date_begin) then
      month_out = 0;

    else
    begin
      month_out = (select datediff(month, :date_begin - 1, :date_end)
                   from rdb$database);
      days_diff = extract(day from :date_end) - extract(day from (:date_begin - 1));
      if (:days_diff >= 14) then
        month_out = month_out + 1;
    end
    month_out = month_tmp + month_out;

  suspend;

end
