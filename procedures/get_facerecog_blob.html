CREATE PROCEDURE GET_FACERECOG_BLOB(
    FRL_ID D_ID)
RETURNS (
    BLOB_1 BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    BLOB_2 BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    FRL_KOEF D_PERCENT)
AS
declare variable bls_1_id d_id;
declare variable bls_2_id d_id;
begin
  for select frl.frl_photo_bls_id, frl.frl_doc_bls_id, frl.frl_koef
      from facerecoglist frl
      where frl.frl_id = :frl_id
      into bls_1_id, bls_2_id, frl_koef
  do
  begin
    select blx_blob
    from hp_get_blobex_blob(:bls_1_id)
    into blob_1;

    select blx_blob
    from hp_get_blobex_blob(:bls_2_id)
    into blob_2;
    suspend;
  end
end
