Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_PERSINFO_RESTORE_LIST(
    C_ID D_ID)
RETURNS (
    FIELD_NAME D_STR_CODE,
    PIF_KEY_ID D_ID,
    PIF_ACTIAL_DATA D_BOOL,
    PIF_UUID D_UUID,
    PIF_INDEXED D_BOOL,
    PIF_VALUE D_STR_255,
    PIF_USR_ID D_ID,
    PIF_DATETIME D_DATETIME,
    PIF_SIGNED D_BOOL)
AS
begin
  for select field_name, pif_key_id, pif_actial_data, p.pif_uuid, p.pif_indexed, pif_value, pif_signed, pif_usr_id, pif_datetime
  from hp_persinfo_restore_tmp(:c_id) p, ux$persfield t
  where p.pif_uuid = t.pif_uuid
  order by t.pif_source, t.pif_field_name, pif_actial_data, pif_datetime
  into :field_name, :pif_key_id, :pif_actial_data, :pif_uuid, :pif_indexed, :pif_value, :pif_signed, :pif_usr_id, :pif_datetime
  do
    suspend;
end
