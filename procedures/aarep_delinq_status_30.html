CREATE PROCEDURE AAREP_DELINQ_STATUS_30(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    IS_CATEGORY_3 INTEGER)
RETURNS (
    REG_NAME VARCHAR(60),
    KF_NAME VARCHAR(30),
    ZAD_OS CURR_TYPE,
    SH_CALC_PERC CURR_TYPE,
    SH_CALC_FINE CURR_TYPE,
    REPAY_SUM CURR_TYPE,
    REPAY_DATE DATE,
    SBA_DATE_BEGIN DATE,
    IN_OFFICE INTEGER,
    IN_COLLECTOR_RU INTEGER,
    IN_COLLECTOR_HOME INTEGER,
    SBL_NAME VARCHAR(100),
    OKS_FROM_K_ID DECIMAL(18,0))
AS
declare variable oks_region_id integer;
declare variable oks_kc_id id_type;
declare variable region_id_prior integer;
declare variable kc_id_prior id_type;
declare variable sbl_id integer;
declare variable zad_perc curr_type;
declare variable officer_name varchar(50);
begin
  if (is_category_3 is null or is_category_3 <> 1) then
    is_category_3 = 0;
  region_id_prior = 0;
  kc_id_prior = 0;
  for select oks_from_k_id, oks_region_id, oks_kc_id
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between oks_date_begin and oks_date_end and
            :rep_date_end - oks_delinq_date > 30 and
            ((:is_category_3 = 1 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) = 3) or
            (:is_category_3 = 0 and
            (select aa_category_id
             from aarep_get_category_id(oks_from_k_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2))) and
            oks_region_id = id_list and
            oks_close_state = 0
      order by oks_region_id, oks_kc_id
      into oks_from_k_id, oks_region_id, oks_kc_id
  do
  begin

    if (region_id_prior <> oks_region_id) then
    begin
      region_id_prior = oks_region_id;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :oks_region_id
      into reg_name;
    end
    if (kc_id_prior <> oks_kc_id) then
    begin
      kc_id_prior = oks_kc_id;
      select c_name_fio
      from clients
      where c_id = :oks_kc_id
      into officer_name;
    end
    select kal_new_kf_name
    from kontract, knaktlist
    where k_id = :oks_from_k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end
    into kf_name;

    select sum(zad_os), sum(zad_perc), sum(sh_calc_perc), sum(sh_calc_fine)
    from aarep_make_zad_list(:oks_from_k_id, :rep_date_end, 1)
    where sh_repay_type = 1 and
          sh_date is not null
    into zad_os, zad_perc, sh_calc_perc, sh_calc_fine;
    if (zad_perc > sh_calc_perc) then
      sh_calc_perc = zad_perc;

    repay_date = null;
    repay_sum = null;
    with tabl1
    as (select ok_ol_date, -sum(ok_os + ok_perc + ok_fine) repay_sum
        from operkred
        where ok_from_k_id = :oks_from_k_id and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_from_c_id = 0
        group by ok_ol_date)
    select ok_ol_date, repay_sum
    from tabl1
    where repay_sum <> 0
    order by ok_ol_date desc
    rows 1
    into repay_date, repay_sum;

    sbl_id = null;
    sbl_name = null;
    sba_date_begin = null;
    select sbl_id, sbl_name, sba_date_begin
    from sb_akt, sb_list
    where sba_from_k_id = :oks_from_k_id and
          sba_is_delete = 0 and
          sba_status = 2 and
          :rep_date_end between sba_date_begin and sba_date_end and
          sba_from_sbl_id = sbl_id
    into sbl_id, sbl_name, sba_date_begin;

    in_office = 0;
    in_collector_ru = 0;
    in_collector_home = 0;
    if (sbl_id is null or (sbl_id in (17, 18, 19))) then
    begin
      in_office = 1;
      sbl_name = officer_name;
    end
    else
    begin
      if (sbl_id in (1, 7, 8, 9, 10, 12, 13, 16, 17, 18,
                     19, 20)) then
        in_collector_ru = 1;
      else
        in_collector_home = 1;
    end

    suspend;
  end
end
