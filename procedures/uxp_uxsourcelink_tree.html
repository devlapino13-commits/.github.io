CREATE PROCEDURE UXP$UXSOURCELINK_TREE(
    US_UUID D_UUID,
    US_UUID_CODE D_STR_CODE = null)
RETURNS (
    RES_ID D_STR_100,
    RES_PARENT_ID D_STR_100,
    RES_SOURCE D_STR_CODE,
    RES_CAPTION D_STR_100,
    RES_CAN_EDIT D_BOOL,
    RES_FIELDS_LINK D_STR_255,
    RES_IMAGE D_SMALLINT,
    USL_US_UUID D_UUID,
    USL_FK_INDEX D_STR_100,
    USL_LINK_ID D_INTEGER,
    USL_PREFIX D_STR_10,
    USL_CHECKED D_BOOL,
    RES_SOURCE_FIELDS D_STR_LONG)
AS
declare variable tmp_source d_str_code;
declare variable tmp_fields d_str_100;
declare variable uix_fields d_str_100;
declare variable uix_fk_index d_str_code;
declare variable uix_state d_smallint;
declare variable tmp_uuid d_uuid;
begin
  if (coalesce(us_uuid, '') = '') then
    us_uuid = uxf$uniq(us_uuid_code);

  usl_us_uuid = us_uuid;

  for select us_code, 0, us_lookup_id, ''
  from ux$source
  where us_uuid = :us_uuid
  union
  select uix_fk_source, usl_link_id, usl_fk_index, usl_prefix
  from ux$sourcelink, ux$index
  where usl_us_uuid = :us_uuid and
        usl_fk_index = uix_index
  order by 2
  into res_source, usl_link_id, usl_fk_index, usl_prefix
  do
  begin
    res_can_edit = 0;
    res_parent_id = null;
    res_id = res_source || '$' || usl_fk_index;
    usl_checked = 1;
    res_fields_link = usl_fk_index;
    res_caption = res_source;
    uix_state = 0;
    res_image = 83; -- Таблица с выделенной колонкой

    if (usl_fk_index <> '') then
      res_caption = res_source || '  [' || usl_fk_index || ']';

    res_source_fields = '';
    if (usl_link_id = 0) then
    begin
      select left(list(uf_field_name, ', '), 1020)
      from ux$field
      where uf_owner = :us_uuid
      into res_source_fields;

      if (char_length(res_source_fields) = 1020) then
        res_source_fields = res_source_fields || '...';

      suspend;
    end

    res_parent_id = res_id;
    tmp_source = res_source;
    res_fields_link = '';

    for select uix_fk_source, uix_index, uix_fields, uix_fk_index, uix_state
    from ux$index
    where uix_source = :tmp_source and
          uix_fk_index <> ''
    into res_source, usl_fk_index, uix_fields, uix_fk_index, uix_state
    do
    begin
      res_id = res_source || '$' || usl_fk_index;
      res_fields_link = '';
      res_caption = res_source;
      usl_prefix = '';
      usl_link_id = null;

      tmp_fields = '';
      select uix_fields
      from ux$index
      where uix_index = :uix_fk_index
      into tmp_fields;

      if (tmp_fields <> '') then
      begin
        res_fields_link = uix_fields || ' -> ' || tmp_fields;
        res_caption = res_caption || '  [' || res_fields_link || ']';
      end

      usl_checked = 0;
      select usl_link_id, usl_prefix
      from ux$sourcelink
      where usl_us_uuid = :us_uuid and
            usl_fk_index = :usl_fk_index
      into usl_link_id, usl_prefix;

      if (row_count = 1) then
        usl_checked = 1;

      res_can_edit = 1;
      if (uix_state <> 0 and usl_checked = 0) then
        res_can_edit = 0;

      if (usl_checked = 1) then
        res_image = 83; -- Таблица с выделенной колонкой
      else
      begin
        res_image = case uix_state
        when 1 then 55 -- Таблица к добавлению
        when -1 then 56 -- Таблица к удалению
        else 85 -- Таблица
        end;
      end

      res_source_fields = '';
      tmp_uuid = uxf$uniq(res_source, 0);
      if (tmp_uuid is not null) then
      begin
        select left(list(uf_field_name, ', '), 1020)
        from ux$field
        where uf_owner = :tmp_uuid
        into res_source_fields;
  
        if (char_length(res_source_fields) = 1020) then
          res_source_fields = res_source_fields || '...';
      end

      suspend;
    end
  end
end
