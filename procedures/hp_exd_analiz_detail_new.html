CREATE PROCEDURE HP_EXD_ANALIZ_DETAIL_NEW(
    IFA_ID D_INTEGER,
    REC_NO D_INTEGER,
    TRANDT D_DATETIME,
    TRANPRDT D_DATE,
    BANK_CODE D_STR_LANG,
    FILIAL_CODE D_STR_LANG,
    ACCOUNT D_STR_20,
    CARD_NO D_STR_20,
    CURRENCY D_STR_LANG,
    SUMTRAN D_CURRENCY,
    OPERTYPE D_STR_SYM,
    TRANCODE D_STR_LANG,
    TERMTYPE D_STR_SYM,
    POINTCODE D_STR_20,
    POINTNAME D_STR_CODE,
    POINTMCC D_STR_10,
    CITY D_STR_20,
    EMITENTCOM D_CURRENCY,
    CMI D_STR_20,
    ACQUIRERCOM D_CURRENCY)
RETURNS (
    IED_ID D_INTEGER)
AS
declare variable dtoper d_date;
declare variable cca_id d_integer;
declare variable card_id d_integer;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable office_id d_integer;
declare variable itt_id d_integer;
declare variable ib_id d_integer;
declare variable pan d_str_20;
declare variable cur d_str_lang;
begin
  select
    cca_id,
    cci_from_c_id
  from cardclientacc
    inner join cardclientid
      on cci_id = cca_from_cci_id
  where cca_code = :account
  into
    :cca_id,
    :c_id;

  dtoper = trandt;
  select
    card_id,
    k_id
  from hp_client_bank_card_mb(:c_id, :dtoper)
  into
    :card_id,
    :k_id;

  select
    kal_new_reg_id
  from kontract
    inner join knaktlist
      on kal_from_kf_id = k_from_kf_id
        and is_delete = 0
        and :dtoper between kal_date and kal_date_end
  where k_id = :k_id
  into
    :office_id;

  select
    itt_id
  from ipc_transaction_types
  where itt_code = :trancode
  into
    :itt_id;

  select first 1
    ibc_from_ib_id
  from ipc_bank_cards
  where ibc_cmi = :cmi
  into
    :ib_id;

  pan = iif(char_length(:card_no) = 16,
    left(:card_no, 6) || '******' || right(:card_no, 4),
      :card_no);

  cur = iif(:currency = '', 'KGS', :currency);

  insert into ipc_exd_detail (
    ied_from_ifa_id,
    ied_rec_no,
    ied_trans_datetime,
    ied_ipc_trans_processed,
    ied_bank_code,
    ied_filial_code,
    ied_from_cca_id,
    ied_from_card_id,
    ied_from_c_id,
    ied_from_k_id,
    ied_from_office_id,
    ied_pan,
    ied_trans_currency,
    ied_trans_amount,
    ied_oper_type,
    ied_from_itt_id,
    ied_terminal_type,
    ied_trade_point_code,
    ied_trade_point_name,
    ied_trade_point_mcc,
    ied_city,
    ied_emitent_comission,
    ied_acquirer_cmi,
    ied_acquirer_bank_id,
    ied_acquirer_commission
  )
  values (
    :ifa_id,
    :rec_no,
    :trandt,
    :tranprdt,
    :bank_code,
    :filial_code,
    :cca_id,
    :card_id,
    :c_id,
    :k_id,
    :office_id,
    :pan,
    :cur,
    :sumtran,
    :opertype,
    :itt_id,
    :termtype,
    :pointcode,
    :pointname,
    :pointmcc,
    :city,
    :emitentcom,
    :cmi,
    :ib_id,
    :acquirercom
  )
  returning ied_id
  into :ied_id;

  suspend;
end
