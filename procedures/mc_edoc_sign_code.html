CREATE PROCEDURE MC_EDOC_SIGN_CODE(
    MCL_ID D_ID,
    EDS_ID D_ID,
    CODE D_STR_CODE)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable eds_status d_integer;
declare variable edm_status d_smallint;
declare variable edl_status d_integer;
declare variable hash_check d_smallint;
declare variable edl_id d_id;
declare variable msg_id d_integer;
declare variable eds_from_c_id d_id;
declare variable lang d_str_lang;
begin
  res_text = '';
  res_code = 0;
  lang = f_get_lang_code(null, mcl_id);

  if (code is null) then code = '';

  select edl_id, eds_status, iif(coalesce(edl_date_end, current_date) < current_date, -1, edl_status)
  from edoc_sign, edoc_list
  where eds_id = :eds_id and
        eds_from_edl_id = edl_id
  into edl_id, eds_status, edl_status;

  if (res_text = '' and edl_status = -1) then
    res_text = f_lang(lang, 'Данные устарели');
  if (res_text = '' and eds_status = 1) then
    res_text = f_lang(lang, 'Документы уже подписаны');

  if (res_text = '') then
  begin
    select edm_from_msg_id, edm_status, iif(:code = '', 0, iif(edm_hash_pin in (:code, f_md5(:code)), 1, 0)),
           eds_from_c_id
    from edoc_msg, edoc_sign
    where eds_id = :eds_id and
          edm_from_eds_id = eds_id
    order by iif(:code = '', 0, iif(edm_hash_pin in (:code, f_md5(:code)), 1, 0)) desc
    rows 1 -- берем наиболее успешный статус
    into msg_id, edm_status, hash_check,
         eds_from_c_id;

    if (res_text = '' and edm_status is null) then res_text = f_lang(lang, 'Неверный статус');
    if (res_text = '' and edm_status = 1) then res_text = f_lang(lang, 'Документы уже подписаны');

    if (res_text = '' and hash_check = 0) then
    begin
      res_text = f_lang(lang, 'Неверный код');
      res_code = 1;
    end
  end
  
  if (res_text = '') then
  begin
    update edoc_msg
    set edm_status = 1
    where edm_from_eds_id = :eds_id and
          edm_from_msg_id = :msg_id and
          edm_status = 0;

    if (eds_from_c_id is not null) then
    begin
      update edoc_sign
      set eds_enum_edoc_sign_kind =  f_enum('edoc_sign_kind', 'mobile', 'id'),
          eds_pin = :code
      where eds_id = :eds_id and
            (
              eds_enum_edoc_sign_kind is distinct from f_enum('edoc_sign_kind', 'mobile', 'id') or
              eds_pin is distinct from :code
            );
    end

    -- Если документы связаны с реструктуризацией
    select res_is_error, res_text
    from hp_edoc_com_action(:edl_id, 1000000000000) -- пользователь fsrobot
    into res_code, res_text;

    if (res_code = 0) then
      res_text = f_lang(lang, 'Документы успешно подписаны');
  end

  suspend;
end
