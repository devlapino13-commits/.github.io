CREATE PROCEDURE CM_KG_0149_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kc_id d_id;
begin
  if (k_id = 109010260262100101) then
    exit;
  res_error_is_critical = 2;
  for select k_disb_date,
             kc.kc_amount_approved,
             cl.c_name_fio,
             kc.kc_id
  from kontract kont
  inner join product pr on k_from_pr_id = pr_id and
        pr_code containing '.CONSULTATION.' and
        kont.k_refund = 0 -- не рефинансирование
  inner join knclient kc on k_id = kc_from_k_id and
        kc_amount_approved > 0
  inner join clients cl on cl.c_id = kc.kc_from_c_id
  where k_id = :k_id
  into :kf_disb_date, :kc_amount, :res_info_text, :kc_id
  do
  begin
        if(iif(f_edoc_status('consulting.agreement', :kc_id, :kf_disb_date) = 'signed', 1, 0) = 0) then  -- RFCKG-4363
        begin
            res_error_text = 'Договор на оказание консультационных услуг, на дату выдачи займа, не подписан';
            suspend;
        end
  end
end
