CREATE PROCEDURE HP_ISHENIM_TECH_PROBLEMS_SEND(
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_255,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable KBM_ID D_ID;
declare variable CNT D_INTEGER;
declare variable EIC_ID D_ID;
declare variable C_ID D_ID;
declare variable OBJ_ID D_ID;
begin
  -- переотправка в КИБ отчетов по которым была тех. ошибка (RFCKG-2181)

  res_info_text = '';
  res_error_text = '';
  cnt = 0;

  for select kbm_id
      from kibminiapp
      where cast(kbm_datetime as date) >= current_date and
            kbm_err_text like '%не удалось получить отчет по техническим причинам%'
      into kbm_id
  do
  begin
    update kibminiapp
    set kbm_ver_info = 0
    where kbm_id = :kbm_id;

    cnt = cnt + 1;
  end

  for select ma.ma_id, ma.ma_office_id, ma.ma_from_un_id
  from miniapp ma
  where ma.ma_date_end = '31.12.9999'
        and ma.ma_date_app >= current_date - 1
        and ma.ma_is_active = 1
  into :obj_id, :c_id, :un_id
  do
   begin
      if (not exists(
          select 1
          from extinfolist eil
          inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
          where eic.eic_code = 'ss_cib_report'
            and eil.eil_object_kind = f_objkind('miniapp', 'id')
            and eil.eil_object_id = :obj_id
            and eil.eil_status in (0, 2)
      )) then
      begin
        execute procedure hp_extinfo_insert('ss_cib_report', 'miniapp', :obj_id, 1, :un_id, null);
        cnt = cnt + 1;
      end
   end

  res_info_text = 'Количество переотправленных запросов в КИБ(SS) - ' || cnt;
  suspend;
  exit;

  when any do
  begin
    res_info_text = '';
    res_error_text = 'Ошибка: переотправить запросы в КИБ не удалось';
    suspend;
  end
end
