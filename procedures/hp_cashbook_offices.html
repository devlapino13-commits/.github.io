CREATE PROCEDURE HP_CASHBOOK_OFFICES(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_OFFICE_ID D_INTEGER,
    RES_OFFICE_NAME D_STR_100,
    RES_DATE_BEGIN D_DATE,
    RES_DATE_END D_DATE,
    RES_TOTAL_COUNT D_INTEGER,
    RES_SIGNED_COUNT D_INTEGER)
AS
begin
  res_date_begin = rep_date_begin;
  res_date_end = rep_date_end;

  for select id_list, reg_code || ' ' || reg_name
  from region_list(:rep_region_id, 1, 3, 1)
       inner join region on id_list = reg_unicode and coalesce(reg_date_close, :rep_date_end) >= :rep_date_begin
  into res_office_id, res_office_name
  do
  begin
    res_total_count = 0;
    res_signed_count = 0;

    select count(*), sum(iif(edl_status = 1, 1, 0))
    from cashbookmain
         left join edoc_list on cbm_from_edl_id = edl_id
    where cbm_office_id = :res_office_id and
          cbm_date between :rep_date_begin and :rep_date_end and
          cbm_invalid = 0
    into res_total_count, res_signed_count;

    if (res_total_count = 0) then
      res_total_count = null;

    if (res_signed_count = 0) then
      res_signed_count = null;

    suspend;
  end
end
