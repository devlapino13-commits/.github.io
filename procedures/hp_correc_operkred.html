CREATE PROCEDURE HP_CORREC_OPERKRED(
    K_ID ID_TYPE)
AS
declare variable orig_ol_id id_type;
declare variable orig_sh_id integer;
declare variable orig_sh_date date;
declare variable orig_ok_type integer;
declare variable ok_from_ol_id id_type;
declare variable ok_number integer;
declare variable ok_sh_id integer;
declare variable ok_sh_date date;
declare variable ok_type integer;
declare variable ok_os curr_type;
declare variable ok_perc curr_type;
declare variable ok_fine curr_type;
declare variable ok_pay_type integer;
declare variable ok_kommis curr_type;
declare variable ok_sh_os curr_type;
declare variable ok_sh_perc curr_type;
declare variable ok_kommis_adv curr_type;
declare variable ok_tax curr_type;
declare variable ok_ol_date date;
declare variable ok_ol_reg_id integer;
declare variable ok_ol_is_active smallint;
declare variable ok_from_c_id id_type;
declare variable orig_ok_number integer;
begin
  orig_ol_id = null;
  orig_sh_id = null;
  orig_sh_date = null;
  orig_ok_type = null;
  orig_ok_number = null;

  for select ok_from_ol_id, ok_number, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type, ok_kommis,
             ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_from_c_id
      from operkred
      where ok_from_k_id = :k_id
      order by ok_from_ol_id, ok_sh_id, ok_sh_date, ok_type, ok_number
--  rows 4
      into :ok_from_ol_id, :ok_number, :ok_sh_id, :ok_sh_date, :ok_type, :ok_os, :ok_perc, :ok_fine, :ok_pay_type,
           :ok_kommis, :ok_sh_os, :ok_sh_perc, :ok_kommis_adv, :ok_tax, :ok_ol_date, :ok_ol_reg_id, :ok_ol_is_active,
           :ok_from_c_id
  do
  begin

    if (orig_ol_id = ok_from_ol_id and
        orig_sh_id = ok_sh_id and
        orig_sh_date = ok_sh_date and
        orig_ok_type = ok_type) then
    begin
      update operkred
      set ok_os = ok_os + :ok_os,
          ok_perc = ok_perc + :ok_perc,
          ok_fine = ok_fine + :ok_fine,
          ok_kommis = ok_kommis + :ok_kommis,
          ok_sh_os = ok_sh_os + :ok_sh_os,
          ok_sh_perc = ok_sh_perc + :ok_sh_perc,
          ok_kommis_adv = ok_kommis_adv + :ok_kommis_adv,
          ok_tax = ok_tax + :ok_tax
      where (ok_from_ol_id = :orig_ol_id) and
            (ok_number = :orig_ok_number);

      delete from operkred
      where (ok_from_ol_id = :orig_ol_id) and
            (ok_number = :ok_number);
    end
    else
      orig_ok_number = ok_number;

    orig_ol_id = ok_from_ol_id;
    orig_sh_id = ok_sh_id;
    orig_sh_date = ok_sh_date;
    orig_ok_type = ok_type;

  end

--  delete from temp_oksum;
end
