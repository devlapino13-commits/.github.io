CREATE PROCEDURE HP_WAYBACKSHEET_VIEW(
    UN_ID D_ID)
RETURNS (
    WBS_ID D_INTEGER,
    WBS_COUNT_SIGNS D_INTEGER,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    HO_ID D_ID,
    HO_DATE D_DATE,
    HO_ORDER_NO D_STR_20,
    HO_FROM_REG_ID D_INTEGER,
    SIGN_NUMBER D_INTEGER,
    OFFICE_KIND D_INTEGER,
    OFFICE_NAME D_STR_CODE,
    JL_NAME D_STR_100,
    LAST_WORK_DATE D_DATE)
AS
declare variable un_from_c_id d_id;
declare variable reg_id d_integer;
declare variable lst_ho_id d_str_8100;
declare variable lst_reg_id d_str_8100;
declare variable sql_text d_str_8100;
begin
  lst_reg_id =
    (select list(r_list, ' ') from user_reg_list(:un_id));
  un_from_c_id =
    (select un_from_c_id from username where un_id = :un_id);
  reg_id = 1090100;
  lst_ho_id = '';
  sql_text =
    'select
       wbs_id,
       wbs_count_signs,
       ho_id,
       ho_date,
       ho_order_no,
       ho_from_reg_id
     from waybacksheet
       inner join hr_orders
         on ho_id = wbs_ho_id
           and ho_status = 1
     where wbs_signer%1 is null';
  for
    select
      rd_tag,
      wss_kind
    from waybacksheet_signer
      inner join refdata
        on rd_id = wss_rd_id
    where wss_c_id = :un_from_c_id
  into
    :sign_number,
    :office_kind
  do
  begin
    for
      execute statement(f_format(:sql_text, :sign_number))
    into
      :wbs_id,
      :wbs_count_signs,
      :ho_id,
      :ho_date,
      :ho_order_no,
      :ho_from_reg_id
    do
    begin
      if (lst_reg_id containing ho_from_reg_id) then
      begin
        if (sign_number = 7 and wbs_count_signs < 7) then
          continue;

        if (sign_number = 8 and :ho_date < cast('27.04.2026' as d_date)) then -- Подписанту номер 8 нужно подписывать приказы начиная с '27.04.2026'
          continue;

        if (lst_ho_id containing ho_id) then
          continue;
        else
          lst_ho_id = lst_ho_id || ho_id || ';';

        select first 1
          c_id,
          c_name_fio,
          jl_name,
          ps_date_begin - 1
        from hr_pstatus
          inner join joblist
            on jl_id = ps_from_jl_id
          inner join clients
            on c_id = ps_from_c_id
        where ps_from_ho_id = :ho_id
        order by ps_from_ho_id, ps_id desc
        into
          :c_id,
          :c_name_fio,
          :jl_name,
          :last_work_date;

        office_name =
          (select reg_name from region where reg_unicode = :ho_from_reg_id);

        suspend;
      end
    end
  end
end
