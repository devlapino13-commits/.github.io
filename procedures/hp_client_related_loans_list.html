CREATE PROCEDURE HP_CLIENT_RELATED_LOANS_LIST(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    UNIQ_ID D_STR_20,
    CLIENT_ID D_ID,
    CLIENT_FIO D_STR_100,
    LN_ID D_INTEGER,
    LN_DATE_BEGIN D_DATE,
    LN_DATE_END D_DATE,
    LN_COUNT D_INTEGER,
    CLIENT_POINT D_INTEGER,
    LN_ID_LIST D_STR_LONG,
    LN_AGREEMENT_LIST D_STR_LONG)
AS
begin
  for select distinct kc_from_c_id
  from knclient, kontstate
  where ks_date_begin <= (:rep_date_begin - 1) and
        ks_date_end >= (:rep_date_begin - 1) and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_from_k_id = kc_from_k_id and
        kc_amount_approved > 0
  into client_id
  do
  begin
    for select uniq_id, client_fio, ln_id, ln_date_begin, ln_date_end, ln_count, client_point, ln_id_list,
               ln_agreement_list
    from hp_client_related_loans(:client_id, :rep_date_begin, :rep_date_end)
    into :uniq_id, :client_fio, :ln_id, :ln_date_begin, :ln_date_end, :ln_count, :client_point, :ln_id_list, :ln_agreement_list
    do
      suspend;
  end
end
