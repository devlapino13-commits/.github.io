CREATE PROCEDURE HP_DO_REPAYMENT_RECALC_SH(
    OL_ID D_ID,
    K_ID D_ID)
AS
declare variable tmp_amount d_currency;
declare variable ol_date d_date;
begin
  -- Процедура для автореструктуризации графика
  -- Процедура приводит в актуальное состояние задолженности по процентам
  -- Является аналогом принятия нулевого погашения

  if (coalesce(k_id, 0) = 0) then
    exception e_default 'Input parameter K_ID is null';

  if (coalesce(ol_id, 0) = 0) then
    exception e_default 'Input parameter OL_ID is null';

  select ol_date from operlist
  where ol_id = :ol_id
  into ol_date;
  if (ol_date is null) then
    exception e_default 'Unknown date';

  -- Построение задолженности по графику на дату
  select res_amount
  from hp_repay_fill(:k_id, 0, :ol_date, 0, 1, 1, 1, 1, 1)
  into :tmp_amount;

  -- Автореструктуризация графика
  tmp_amount = 0;
  select has_advanced_pay
  from hp_repay_reschedule(:ol_date, 2, 1, :k_id)
  into :tmp_amount; -- Имеются ли досрочные погашения
  tmp_amount = 0;

  -- Сохранение на Operkred сумм автореструктуризации
  execute procedure hp_repay_save_operkred(:k_id, 0, :ol_id, :ol_date, 0, 3, 0);
end
