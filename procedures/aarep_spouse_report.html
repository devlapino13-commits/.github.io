CREATE PROCEDURE AAREP_SPOUSE_REPORT(
    REP_DATE D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    LKC_STATUS_STR VARCHAR(50),
    LKC_PERCENT PERC_TYPE,
    K_KONTRACT_NO D_STR_100,
    REG_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    BIRTH_DATE D_DATE,
    ADDRESS_FACT D_STR_255,
    C_NAME_FIO_SPOUSE D_STR_100,
    RELATION_TYPE D_BOOL,
    SPOUSE_FIO D_STR_100,
    SPOUSE_LKC_STATUS_STR VARCHAR(50),
    SPOUSE_LKC_PERCENT PERC_TYPE,
    SPOUSE_K_KONTRACT_NO D_STR_100,
    SPOUSE_REG_NAME D_STR_100,
    SPOUSE_OFFICE_NAME D_STR_100,
    SPOUSE_BIRTH_DATE D_DATE,
    SPOUSE_ADDRESS_FACT D_STR_255,
    SPOUSE_SPOUSE_FIO D_STR_100,
    SPOUSE_SPOUSE_BIRTH_DATE D_DATE,
    SPOUSE_FIO_REPEAT_COUNT D_INTEGER,
    IS_CONCURRENT D_BOOL)
AS
declare variable office_id d_id;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable current_office_id d_id;
declare variable c_id_old d_id;
declare variable spouse_c_id d_id;
declare variable spouse_c_id_tmp d_id;
declare variable spouse_office_id d_id;
declare variable spouse_k_id d_id;
declare variable spouse_spouse_c_id d_id;
declare variable spouse_current_office_id d_id;
declare variable lkc_status d_smallint;
declare variable k_from_kf_id d_id;
declare variable office_list d_str_long;
declare variable office_list1 d_str_long;
declare variable office_list2 d_str_long;
declare variable office_list3 d_str_long;
declare variable office_list4 d_str_long;
declare variable office_list5 d_str_long;
declare variable office_list6 d_str_long;
declare variable office_list_exception d_str_long;
begin
  -- Зоны поиска однофамильца
  --      НАЧАЛО
  -- Иссык-Кульский и Нарынский регион
  with reglist
  as (select id_list
      from region_list(1010000, 1, 3, 1)
      union
      select id_list
      from region_list(1090002, 1, 3, 1))
  select list(id_list)
  from reglist
  into office_list1;

  -- Бишкекский, Чуйский и Таласский регионы
  with reglist
  as (select id_list
      from region_list(1030000, 1, 3, 1)
      union
      select id_list
      from region_list(1060000, 1, 3, 1)
      union
      select id_list
      from region_list(1090004, 1, 3, 1))
  select list(id_list)
  from reglist
  into office_list2;

  --Ташкумырский и Жалалабадский
  with reglist
  as (select id_list
      from region_list(1050000, 1, 3, 1)
      union
      select id_list
      from region_list(1070000, 1, 3, 1))
  select list(id_list)
  from reglist
  into office_list3;

  -- Ошский, Узгенский, Кадамжайский, Сектор ИФУ «Амана Исламик»
  with reglist
  as (select id_list
      from region_list(1040000, 1, 3, 1)
      union
      select id_list
      from region_list(1130000, 1, 3, 1)
      union
      select id_list
      from region_list(1091000, 1, 3, 1)
      union
      select id_list
      from region_list(1270000, 1, 3, 1))
  select list(id_list)
  from reglist
  into office_list4;

  -- Баткенский, Кадамжайский, Кадамжайский, Сектор ИФУ «Амана Исламик»
  with reglist
  as (select id_list
      from region_list(1270000, 1, 3, 1)
      union
      select id_list
      from region_list(1130000, 1, 3, 1)
      union
      select id_list
      from region_list(1110000, 1, 3, 1))
  select list(id_list)
  from reglist
  into office_list5;

  -- Исключение т.к. Кадамжайский и Сектор ИФУ «Амана Исламик» регион входит в две зоны поиска: Ошский, Узгенский, Кадамжайский и Баткенский, Кадамжайский, Сектор ИФУ «Амана Исламик»
  with reglist
  as (select id_list
      from region_list(1040000, 1, 3, 1)
      union
      select id_list
      from region_list(1130000, 1, 3, 1)
      union
      select id_list
      from region_list(1091000, 1, 3, 1)
      union
      select id_list
      from region_list(1270000, 1, 3, 1)
      union
      select id_list
      from region_list(1110000, 1, 3, 1))
  select list(id_list)
  from reglist
  into office_list6;

  with reglist
  as (select id_list
      from region_list(1091000, 1, 3, 1)
      union
      select id_list
      from region_list(1130000, 1, 3, 1))
  select list(id_list)
  from reglist
  into office_list_exception;
  -- Зоны поиска однофамильца
  --       КОНЕЦ

  -- Данные по контракту (Активный индив)
  for select k.k_id, k.k_kontract_no, k.k_fact_disb_reg_id, c.c_id, c.c_name_fio, c.c_birth_date, k.k_from_kf_id
      from clients c
      inner join knclient kc on c.c_id = kc.kc_from_c_id
      inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
      inner join kontract k on k.k_id = kc.kc_from_k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id
      where kc.kc_amount_approved > 0 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end and
            pr.pr_code containing '.ind.'
            --and pr.pr_active = 1
      order by c.c_id
      into :k_id, :k_kontract_no, :office_id, :c_id, :c_name_fio, :birth_date, :k_from_kf_id
  do
  begin
    -- Пропускает повторение клиентов у которых больше 1 контракта
    if (c_id = c_id_old) then
      continue;
    c_id_old = c_id;

    select kal.kal_new_reg_id
    from knaktlist kal
    where kal.kal_from_kf_id = :k_from_kf_id and
          :rep_date between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into :current_office_id;

    -- Фактический адрес клиента
    select adl.adl_address_str
    from addrclient adc
    inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 2
    into :address_fact;

    -- Название офиса и региона выдачи кредита
    select reg.reg_name, reg.reg_stat_rep_region
    from region reg
    where reg.reg_unicode = :office_id
    into :office_name, :reg_name;

    -- Процент и классификация РППУ
    for select lkc.lkc_percent, lkc.lkc_status
        from llc_main lm
        inner join llc_kont_cli lkc on lkc.lkc_from_lm_id = lm.lm_id
        where lm_is_signed = 1 and
              lm_date_status between :rep_date - 4 and :rep_date + 4 and
              lm_from_reg_id = :current_office_id and
              lkc.lkc_from_k_id = :k_id and
              lkc.lkc_from_c_id = :c_id
        into :lkc_percent, :lkc_status
    do
    begin
      select cs.cs_name as lkc_status_str
      from classific cs
      where cs.cs_id = :lkc_status
      into :lkc_status_str;
    end

    -- Супруг/Однофамилец клиента НАЧАЛО
    -- ПРИМЕЧАНИЕ: Под однофамильцем имеется ввиду клиент у которого одинаковое ФИО, а не только фамилия
    spouse_c_id = null;
    spouse_fio = null;
    is_concurrent = null;
    spouse_c_id_tmp = null;
    spouse_fio_repeat_count = 0;

    -- Поиск супруга(и) по связям клиента
    select iif(crel.crel_from_c_id = :c_id, crel.crel_other_c_id, crel.crel_from_c_id)
    from clientrelation crel
    where (crel.crel_other_c_id = :c_id or
          crel.crel_from_c_id = :c_id) and
          crel.crel_is_active = 1 and
          crel.crel_from_rel_id in (1, 2) rows 1 --rows 1 т.к. есть клиенты с 2 и более супругами пример(1090100194146 1090100655075)
    into :spouse_c_id_tmp;

    -- Если у клиента нет супруга(и) пропускаем его
    if (spouse_c_id_tmp is null) then
      continue;

    -- ФИО, дата рождения супруга(и)
    select c.c_name_fio, c.c_birth_date
    from clients c
    where c.c_id = :spouse_c_id_tmp
    into :c_name_fio_spouse, :spouse_birth_date;

    spouse_fio = c_name_fio_spouse;

    office_list = case
                    when office_list1 containing :office_id then office_list1
                    when office_list2 containing :office_id then office_list2
                    when office_list3 containing :office_id then office_list3
                    when office_list_exception containing :office_id then office_list6
                    when office_list4 containing :office_id then office_list4
                    when office_list5 containing :office_id then office_list5
                  end;

    -- Подсчёт количества однофамильцев
    select count(c_id)
    from find_namesake(:c_name_fio_spouse, :spouse_c_id_tmp, :office_list, :rep_date)
    into spouse_fio_repeat_count;

    -- Если у клиента нет однофамильцев пропускаем его
    if (spouse_fio_repeat_count < 2) then
      continue;

    -- Проходим по супругу и всем однофамильцам и вытаскиваем их данные
    for select c_id
        from find_namesake(:c_name_fio_spouse, :spouse_c_id_tmp, :office_list, :rep_date)
        into spouse_c_id
    do
    begin
      spouse_spouse_fio = null;
      spouse_spouse_c_id = null;
      spouse_birth_date = null;
      spouse_address_fact = null;
      spouse_lkc_status_str = null;
      spouse_lkc_percent = null;
      spouse_k_kontract_no = null;
      spouse_reg_name = null;
      spouse_office_name = null;
      spouse_office_id = null;
      spouse_current_office_id = null;
      spouse_spouse_birth_date = null;
      k_from_kf_id = null;
      lkc_status = null;

      select c.c_birth_date
      from clients c
      where c.c_id = :spouse_c_id
      into :spouse_birth_date;

      -- Данные активного кредита супруга(и)
      select k.k_id, k.k_kontract_no, k.k_fact_disb_reg_id, k.k_from_kf_id
      from knclient kc
      inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
      inner join kontract k on k.k_id = kc.kc_from_k_id
      where kc.kc_from_c_id = :spouse_c_id and
            kc.kc_amount_approved > 0 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1 and
            :rep_date between ks.ks_date_begin and ks.ks_date_end rows 1
      into :spouse_k_id, :spouse_k_kontract_no, :spouse_office_id, :k_from_kf_id;

      -- Адрес супруга/однофамильца
      select adl.adl_address_str
      from addrclient adc
      inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
      where adc.adc_from_c_id = :spouse_c_id and
            adc.adc_address_type = 2
      into :spouse_address_fact;

      -- Название офиса и региона выдачи кредита супруга(и)/однофамильца
      select reg.reg_name, reg.reg_stat_rep_region
      from region reg
      where reg.reg_unicode = :spouse_office_id
      into :spouse_office_name, :spouse_reg_name;

      -- Для расчёта РППУ нужен текущий офис
      select kal.kal_new_reg_id
      from knaktlist kal
      where kal.kal_from_kf_id = :k_from_kf_id and
            :rep_date between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into spouse_current_office_id;

      if (spouse_k_id is not null) then
      begin
        -- Процент и классификация РППУ
        for select lkc.lkc_percent, lkc.lkc_status
            from llc_main lm
            inner join llc_kont_cli lkc on lkc.lkc_from_lm_id = lm.lm_id
            where lm_is_signed = 1 and
                  lm_date_status between :rep_date - 4 and :rep_date + 4 and
                  lm_from_reg_id = :spouse_current_office_id and
                  lkc.lkc_from_k_id = :spouse_k_id and
                  lkc.lkc_from_c_id = :spouse_c_id
            into :spouse_lkc_percent, :lkc_status
        do
        begin
          select cs.cs_name as lkc_status_str
          from classific cs
          where cs.cs_id = :lkc_status
          into :spouse_lkc_status_str;
        end
      end

      -- Поиск супруга(и) по связям супруга(и)/однофамильца
      select iif(crel.crel_from_c_id = :spouse_c_id, crel.crel_other_c_id, crel.crel_from_c_id)
      from clientrelation crel
      where (crel.crel_other_c_id = :spouse_c_id or
            crel.crel_from_c_id = :spouse_c_id) and
            crel.crel_is_active = 1 and
            crel.crel_from_rel_id in (1, 2) rows 1 --1090100194146 1090100655075
      into :spouse_spouse_c_id;

      -- ФИО супруга(и) супруга(и)/однофамильца
      select c.c_name_fio, c.c_birth_date
      from clients c
      where c.c_id = :spouse_spouse_c_id
      into :spouse_spouse_fio, :spouse_spouse_birth_date;

      -- Соответствуют ли данные по супругам (1 - да, 0 - нет)
      is_concurrent = iif(:c_name_fio = :spouse_spouse_fio, 1, 0);

      -- Признак связанности (1 - супруг, 0 однофамилец)
      relation_type = iif(c_id = spouse_spouse_c_id, 1, 0);

      suspend;
    end
  end
end
