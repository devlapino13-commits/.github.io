CREATE PROCEDURE FM_CLIENT_KONT_HISTORY(
    C_ID ID_TYPE,
    UN_ID ID_TYPE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable k_id id_type;
declare variable c_name_fio str_100;
declare variable k_kontract_no str_20;
declare variable k_kontract_date d_date;
declare variable k_amount_approved curr_type;
declare variable k_kolich_repay d_integer;
declare variable c_circle d_integer;
declare variable k_fine_max_days_str str_20;
declare variable k_close_date d_date;
declare variable sum_delay_days d_integer;
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';
  res_text = blob_append(res_text, f_format('<h1 class="green_text">%1</h1>', 'Внутренняя кредитная история клиента'));

  select c_name_fio
  from clients
  where c_id = :c_id
  into :c_name_fio;
  res_text = blob_append(res_text, f_format('<h2>%1</h2><br>', c_name_fio));

  res_text = blob_append(res_text, '<table width="100%">
       <col width="40%">
       <col width="60%">
       <tr>' || f_format('<th>%1</th><th>%2</th></tr>', 'Условия', 'Значения'));

  for select k_id, k_kontract_no, k_kontract_date, k_amount_approved, k_kolich_repay, c_circle, k_fine_max_days_str,
             k_close_date
      from kontract_client_one_full_info(:c_id, current_date)
      order by k_kontract_date desc, k_kontract_no desc
      into k_id, k_kontract_no, k_kontract_date, k_amount_approved, k_kolich_repay, c_circle, k_fine_max_days_str,
           k_close_date
  do
  begin

    select sum(ok.ok_ol_date - ok_sh_date)
    from operkred ok
    where ok.ok_ol_is_active = 1 and
          ok.ok_ol_date <= current_date and
          ok.ok_ol_date > ok_sh_date and
          ok_type < 7 and
          ok_from_k_id = :k_id and
          ok_os + ok_perc < 0 and
          ok_from_c_id = 0
    into sum_delay_days;

    res_text = blob_append(res_text, f_format('<tr><td class="filled" align="left"><b>%1</b></td><td class="filled" align="left"><b>%2</b></td></tr>', 'Номер договора', k_kontract_no) ||
                                     f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', 'Дата выдачи', f_datetostr(k_kontract_date)) ||
                                     f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', 'Сумма займа', f_numtostr(k_amount_approved, 0)) ||
                                     f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', 'Срок займа', k_kolich_repay) ||
                                     f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', 'Цикл', c_circle) ||
                                     f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', 'Количество просрочек в днях', coalesce(sum_delay_days, 0)) ||
                                     f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', 'Макс. просрочка в днях', k_fine_max_days_str) ||
                                     f_format('<tr><td align="left">%1</td><td align="left">%2</td></tr>', 'Дата закрытия', f_datetostr(k_close_date)));

  end
  res_text = blob_append(res_text, '</table>');
  res_text = blob_append(res_text, '</body>');
  suspend;
end
