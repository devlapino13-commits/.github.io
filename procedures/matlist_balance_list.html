CREATE PROCEDURE MATLIST_BALANCE_LIST(
    WORKPLACE_ID DECIMAL(18,0),
    MATLIST_ID DECIMAL(18,0),
    DD DATE,
    SHOW_ZERO SMALLINT,
    SHOW_BILL SMALLINT)
RETURNS (
    MBAL_BILL_ID DECIMAL(18,0),
    MBAL_BILL_LINE_NO INTEGER,
    MBAL_WP_ID DECIMAL(18,0),
    MBAL_MO_ID DECIMAL(18,0),
    MBAL_ML_ID DECIMAL(18,0),
    MBAL_KOLICH DECIMAL(18,4),
    MBAL_SUMMA DECIMAL(18,4),
    MBAL_BILL_COUNT INTEGER,
    MBAL_CPL_ID DECIMAL(18,0),
    MBAL_MC_ID DECIMAL(18,0),
    MBAL_INVENT_NO INTEGER,
    MBAL_INVENT_STR VARCHAR(50),
    MBAL_ACCOUNT VARCHAR(20),
    MBAL_IS_SKLAD SMALLINT,
    MBAL_FULL_DETAIL VARCHAR(2048),
    MBAL_WP_NAME VARCHAR(100),
    MBAL_REGION_ID INTEGER)
AS
declare variable reg integer;
declare variable wp decimal(18,0);
begin
  if (show_bill = 1) then
  begin
    for select mbal_bill_id, mbal_bill_line_no, mbal_wp_id, mbal_mo_id, mbal_kolich, mbal_summa, mbal_account,
               mbal_is_sklad, mbal_wp_name, mbal_region_id, ob_complect_id, ob_from_ml_id, ob_from_mc_id,
               ob_from_reg_id, ob_invent_no, ob_invent_str, ob_full_detail

        from matlist_balance_temp(:workplace_id, :matlist_id, :dd, :show_zero), operbill
        where ob_bill_id = mbal_bill_id and
              ob_bill_line_no = mbal_bill_line_no and
              ob_date_begin <= :dd and
              ob_date_end >= :dd and
              ob_is_delete = 0 and
              ((:matlist_id = 0) or
              (ob_from_ml_id = :matlist_id))
        into :mbal_bill_id, :mbal_bill_line_no, :mbal_wp_id, :mbal_mo_id, :mbal_kolich, :mbal_summa, :mbal_account,
             :mbal_is_sklad, :mbal_wp_name, :mbal_region_id, :mbal_cpl_id, :mbal_ml_id, :mbal_mc_id, :reg,
             :mbal_invent_no, :mbal_invent_str, :mbal_full_detail
    do
    begin
      if (mbal_account is null) then
        mbal_account = '';
      if (mbal_kolich is null) then
        mbal_kolich = 0;
      if (mbal_summa is null) then
        mbal_summa = 0;
      mbal_bill_count = 1;
      suspend;
    end
  end
  else
  begin

    if (show_zero = 0) then
    begin
      for select count(om_bill_id), om_workplace_id, om_matotvet_id, ob_from_ml_id, ob_from_mc_id, om_account,
                 fwp_status, fwp_name, fwp_region_id,
                 sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) as kol_balance,
                 sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) as sum_balance
          from opermat, operlist, full_wp_list(:workplace_id, 0), operbill
          where om_from_ol_id = ol_id and
                om_workplace_id = fwp_id and
                ob_bill_id = om_bill_id and
                ob_bill_line_no = om_bill_line_no and
                ob_date_begin <= :dd and
                ob_date_end >= :dd and
                ((:matlist_id = 0) or
                (ob_from_ml_id = :matlist_id)) and
                is_delete = 0 and
                ol_date <= :dd
          group by om_workplace_id, om_matotvet_id, ob_from_ml_id, ob_from_mc_id, om_account, fwp_status, fwp_name, fwp_region_id
          having(sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0) or (sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0)
          order by om_workplace_id, om_matotvet_id, ob_from_ml_id
          into :mbal_bill_count, :mbal_wp_id, :mbal_mo_id, :mbal_ml_id, :mbal_mc_id, :mbal_account, :wp, :mbal_wp_name,
               :mbal_region_id, :mbal_kolich, :mbal_summa
      do
      begin
        if (mbal_account is null) then
          mbal_account = '';
        if (mbal_kolich is null) then
          mbal_kolich = 0;
        if (mbal_summa is null) then
          mbal_summa = 0;
        mbal_bill_id = 0;
        mbal_bill_line_no = 0;
        mbal_cpl_id = null;
        mbal_invent_str = '';
        mbal_full_detail = '';
        if (wp = 6) then
          mbal_is_sklad = 1;
        else
          mbal_is_sklad = 0;
        suspend;
      end
    end
    else
    begin
      for select count(om_bill_id), om_workplace_id, om_matotvet_id, ob_from_ml_id, ob_from_mc_id, om_account,
                 fwp_status, fwp_name, fwp_region_id,
                 sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) as kol_balance,
                 sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) as sum_balance
          from opermat, operlist, full_wp_list(:workplace_id, 0), operbill
          where om_from_ol_id = ol_id and
                om_workplace_id = fwp_id and
                ob_bill_id = om_bill_id and
                ob_bill_line_no = om_bill_line_no and
                ob_date_begin <= :dd and
                ob_date_end >= :dd and
                ((:matlist_id = 0) or
                (ob_from_ml_id = :matlist_id)) and
                is_delete = 0 and
                ol_date <= :dd
          group by om_workplace_id, om_matotvet_id, ob_from_ml_id, ob_from_mc_id, om_account, fwp_status, fwp_name, fwp_region_id
          order by om_workplace_id, om_matotvet_id, ob_from_ml_id
          into :mbal_bill_count, :mbal_wp_id, :mbal_mo_id, :mbal_ml_id, :mbal_mc_id, :mbal_account, :wp, :mbal_wp_name,
               :mbal_region_id, :mbal_kolich, :mbal_summa
      do
      begin
        if (mbal_account is null) then
          mbal_account = '';
        if (mbal_kolich is null) then
          mbal_kolich = 0;
        if (mbal_summa is null) then
          mbal_summa = 0;
        mbal_bill_id = 0;
        mbal_bill_line_no = 0;
        mbal_cpl_id = null;
        mbal_invent_str = '';
        mbal_full_detail = '';
        if (wp = 6) then
          mbal_is_sklad = 1;
        else
          mbal_is_sklad = 0;
        suspend;
      end
    end
  end
end
