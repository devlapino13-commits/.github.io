CREATE PROCEDURE PRINT_HRORDER(
    ORDER_ID DECIMAL(18,0),
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    HRO_DATE D_DATE,
    HRO_TEXT D_STR_LONG,
    HRO_REASON D_STR_LONG,
    HRO_REASON_DATE D_DATE,
    HRO_SAVE_TIME TIMESTAMP,
    HRO_ORDER_NO D_STR_20,
    HRO_ORDER_KIND D_STR_100,
    HRO_OFFICE_NAME D_STR_100,
    HRO_SAVE_USER_POSITION VARCHAR(300),
    HRO_SAVE_USER_FIO D_STR_100,
    HRO_SAVE_POSITION D_STR_100,
    HRO_SAVE_USER_POSITION_KG D_STR_255,
    HRO_ORDER_SUBTYPE D_STR_255,
    HRO_ORDER_SUBTYPE_ID D_INTEGER,
    HRO_HELP_AMOUNT D_INTEGER,
    HRO_HELP_AMOUNT_KG D_STR_255,
    HRO_HELP_AMOUNT_RU D_STR_255,
    HRO_ACCOUNTANT_NAME D_STR_100,
    HRO_ACCOUNTANT_POSITION_RU D_STR_100,
    HRO_ACCOUNTANT_POSITION_KG D_STR_255,
    HRO_ACCOUNTANT_QR_CODE D_STR_LONG,
    QR_CODE D_STR_LONG,
    EDOC_SIGN_DATETIME D_DATETIME,
    EDOC_IS_ONLINE D_BOOL,
    HR_SIGNER D_STR_LONG,
    HO_SIGNER_NAME D_STR_255,
    HO_SIGNER_POSITION D_STR_255,
    HO_SIGNER_POSITION_KG D_STR_255,
    LO_SIGNER D_STR_LONG)
AS
declare variable ho_from_un_id d_id;
declare variable ho_reason d_str_255;
declare variable edl_id d_id;
declare variable c_id d_id;
declare variable hr_kind d_integer;
declare variable ho_signer_id d_id;
declare variable eds_id d_id;
declare variable ho_order_type d_integer;
declare variable ho_order_subtype d_integer;
declare variable curr_str d_str_long;
declare variable ho_accountant_id d_id;
declare variable ho_cert_marriage d_bool;
declare variable ho_cert_birth_child d_bool;
declare variable ho_cert_birth_emp d_bool;
declare variable ho_cert_death d_bool;
declare variable i d_smallint;
declare variable str d_str_255;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  hr_signer = '';

  for select ho_date,
             reg_name,
             ho_text,
             ho_reason,
             ho_reason_date,
             ho_save_time,
             ho_order_no,
             sawt_name,
             ho_from_un_id,
             ho_signer_id,
             ho_order_type,
             ho_order_subtype,
             ho_accountant_id,
             ho_help_amount,
             ho_cert_marriage,
             ho_cert_birth_child,
             ho_cert_birth_emp,
             ho_cert_death
      from hr_orders
      inner join region on reg_unicode = ho_from_reg_id
      inner join sa_worktype on sawt_id = ho_order_type
      where ho_id = :order_id
      into hro_date,
           hro_office_name,
           hro_text,
           ho_reason,
           hro_reason_date,
           hro_save_time,
           hro_order_no,
           hro_order_kind,
           ho_from_un_id,
           ho_signer_id,
           ho_order_type,
           ho_order_subtype,
           ho_accountant_id,
           hro_help_amount,
           ho_cert_marriage,
           ho_cert_birth_child,
           ho_cert_birth_emp,
           ho_cert_death
  do
  begin
    hro_save_position = '';
    hro_save_user_position = '';
    hro_save_user_fio = '';
    hro_reason = ho_reason;

    if (ho_from_un_id is not null) then
    begin
      select c_name_fio,
             jl_name || ', ' || reg_name,
             jl_name,
             coalesce(jl_name_origin, '')
      from hr_pstatus, joblist, region, clients, username
      where un_id = :ho_from_un_id and
            ps_from_c_id = un_from_c_id and
            un_from_c_id = c_id and
            ps_date_begin <= :hro_date and
            ps_date_end >= :hro_date and
            ps_from_reg_id = reg_unicode and
            ps_from_jl_id = jl_id
      order by ps_date_begin desc
      rows 1
      into :hro_save_user_fio,
           :hro_save_user_position,
           :hro_save_position,
           hro_save_user_position_kg;

      select cl.c_name_fio,
             jl.jl_name,
             coalesce(jl.jl_name_origin, '')
      from kontsignreg ksr
      inner join kontsignmen ksm on ksm.ksm_id = ksr.ksr_from_ksm_id and ksr.ksr_id = :ho_signer_id
      inner join clients cl on cl.c_id = ksm.ksm_from_c_id
      inner join hr_pstatus ps on ps.ps_from_c_id = cl.c_id and current_date between ps_date_begin and ps_date_end and ps_status = 1 and ps_is_working = 1 and ps.ps_is_on_staff = 1
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      into :ho_signer_name,
           :ho_signer_position,
           :ho_signer_position_kg;
    end

    -- QR-код электронной подписи казначей. Текст пустой если не подписан.
    qr_code = '';
    edoc_sign_datetime = null;
    edoc_is_online = 0;

    if (coalesce(add_params, '') <> '') then
      edl_id = f_json_str_get_value(add_params, 'edl_id', null);

    if (edl_id is null) then
    begin
      hr_kind = f_objkind('hr_orders', 'id');

      select edl_id
      from edoc_list
      where edl_object_kind = :hr_kind and
            edl_object_id = :order_id and
            edl_status >= 0
      order by edl_id desc
      rows 1
      into edl_id;
    end

    if (edl_id is not null) then
    begin
      edoc_is_online = 1;

      select eds_from_c_id,
             eds_id
      from edoc_sign
      where eds_from_edl_id = :edl_id rows 1
      into c_id,
           eds_id;

      qr_code = f_edoc_qr_by_edl(edl_id, c_id);
      lo_signer = f_edoc_cl_sign_web(eds_id);
      select max(eds_status_datetime)
      from edoc_sign
      where eds_from_edl_id = :edl_id and
            eds_status = 1
      into edoc_sign_datetime;

      hr_signer = f_edoc_hr_signer(edl_id);
    end

    if (db_id = 41701) then
    begin
      select saws_name
      from sa_worksubtype
      where saws_from_sawt_id = :ho_order_type and
            saws_id = :ho_order_subtype
      into hro_order_subtype;

      hro_order_subtype_id = ho_order_subtype;

      if (hro_help_amount > 0 and ho_order_type = 22) then -- материальная помощь
      begin
        select curr_str
        from fhp_currencystr_kg(:hro_help_amount)
        into curr_str;
        hro_help_amount_kg = lower(replace(curr_str, ' сом', ''));

        select curr_str
        from fhp_currencystr(:hro_help_amount, 2)
        into curr_str;
        hro_help_amount_ru = lower(replace(curr_str, ' сом. 00 тыйын.', ''));

        select rd_name
        from refdata
        where rd_code = 'ho_corporate_codex'
        into str;
        hro_reason = '1) ' || str || ascii_char(10);

        i = 2;
        if (ho_cert_marriage = 1) then
        begin
          select rd_name
          from refdata
          where rd_code = 'ho_cert_marriage'
          into str;
          hro_reason = hro_reason || i || ') ' || str || ascii_char(10);
          i = i + 1;
        end

        if (ho_cert_birth_child = 1) then
        begin
          select rd_name
          from refdata
          where rd_code = 'ho_cert_birth_child'
          into str;
          hro_reason = hro_reason || i || ') ' || str || ascii_char(10);
          i = i + 1;
        end

        if (ho_cert_death = 1) then
        begin
          select rd_name
          from refdata
          where rd_code = 'ho_cert_death'
          into str;
          hro_reason = hro_reason || i || ') ' || str || ascii_char(10);
          i = i + 1;
        end

        if (ho_cert_birth_emp = 1) then
        begin
          select rd_name
          from refdata
          where rd_code = 'ho_cert_birth_emp'
          into str;
          hro_reason = hro_reason || i || ') ' || str || ascii_char(10);
          i = i + 1;
        end

        if (ho_reason <> '') then
          hro_reason = hro_reason || i || ') Прочие основания: ' || ho_reason;

        select c_name_fio
        from clients
        where c_id = :ho_accountant_id
        into hro_accountant_name; -- бухгалтер-подписант

        select jl_name,
               jl_name_origin
        from hr_pstatus
        inner join joblist on jl_id = ps_from_jl_id
        where ps_from_c_id = :ho_accountant_id and
              ps_status = 1 and
              ps_is_working = 1 and
              ps_is_on_staff = 1 and
              current_date between ps_date_begin and ps_date_end
        into hro_accountant_position_ru,
             hro_accountant_position_kg;

        hro_accountant_qr_code = 'test';
      end
    end

    suspend;
  end
end
