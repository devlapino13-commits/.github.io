CREATE PROCEDURE HP_EFFECT_RATE_FILL_TMP(
    K_ID D_ID,
    OL_ID D_ID = null,
    APPLY_NEW_METHOD D_SMALLINT = 0)
AS
declare variable sh_date_prev d_date;
declare variable sh_id d_integer;
declare variable sh_date d_date;
declare variable sh_days d_integer;
declare variable sh_all d_currency = 0;
declare variable ipf_commission curr_type;
declare variable ol_date d_date;
declare variable sh_os d_currency;
declare variable total_os d_currency;
declare variable is_new_method d_smallint;
declare variable db_id d_integer;
declare variable is_mbf_card d_smallint;
begin
  -- Заполнение временной таблицы temp_schedule
  -- Вспомогательная процедура для расчета эффективной процентной ставки
  -- Все суммы помещаются в sh_os, т.к. по отдельности не нужны
  delete from temp_schedule
  where 1 = 1;

  if (apply_new_method is null) then
    apply_new_method = 0;

  if (ol_id is null) then -- Если OL_ID пусто, значит это первичный расчет ПСК
  begin
    ipf_commission = 0;
  
    select coalesce(pr.pr_ipf_commission, 0), iif(pr.pr_mbulak_cards_kind = 0, 0, 1)
    from product pr
    inner join kontract kont on kont.k_from_pr_id = pr.pr_id
    where kont.k_id = :k_id
    into :ipf_commission, :is_mbf_card;
  
    for select 0, min(sh_date), -sum(sh_os - iif(:is_mbf_card = 0, sh_kommis, 0)) + :ipf_commission
        from schedule
        where sh_from_k_id = :k_id
        union
        select sh_id, sh_date,
               (sh_os + sh_perc)
        from schedule
        where sh_from_k_id = :k_id and
              sh_id > 0
        into :sh_id, :sh_date, :sh_all
    do
    begin
      if (sh_date_prev is null) then
        sh_days = 0;
      else
        sh_days = sh_date - sh_date_prev;
  
      insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
      values (:sh_id, :sh_date, :sh_days, :sh_all);
      sh_date_prev = sh_date;
    end
  end else
  begin
    -- Если OL_ID передан, значит это пересчет после досрочного погашения или реструктуризации
    select ol_date
    from operlist
    where ol_id = :ol_id
    into ol_date;

    is_new_method = 0;
    if (apply_new_method = 1)
    then is_new_method = 1;
    else
    begin
      db_id = f_get_db_id();
      if (db_id = 81001 and ol_date >= cast('01.01.2021' as date)) then
        is_new_method = 1;
    end

    if (is_new_method = 0) then -- старый метод
    begin
      sh_id = 0;
      sh_date_prev = ol_date;
      total_os = 0;
      for select ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc)
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_type <= 7 and
            (ok_ol_date < :ol_date or (ok_ol_date = :ol_date and ok_from_ol_id <= :ol_id)) and
            ok_sh_date > :ol_date
      group by ok_sh_date
      having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 0
      order by ok_sh_date
      into sh_date, sh_os, sh_all
      do
      begin
        sh_id = sh_id + 1;
        sh_days = sh_date - sh_date_prev;
  
        total_os = total_os + sh_os;
  
        insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
        values (:sh_id, :sh_date, :sh_days, :sh_all);
        sh_date_prev = sh_date;
      end
  
      if (sh_id > 0 and total_os > 0) then
      begin
        sh_id = 0;
        sh_days = 0;
        total_os = -total_os;
        insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
        values (:sh_id, :ol_date, :sh_days, :total_os);
      end
    end else
    begin
      -- новый метод
      insert into temp_schedule (sh_id, sh_date, sh_days, sh_os)
      select sh_id, sh_date, sh_days, sh_all
      from hp_effect_rate_sh_list(:k_id, :ol_date, :ol_id);
    end
  end
end
