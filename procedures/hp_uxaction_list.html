CREATE PROCEDURE HP_UXACTION_LIST(
    UX_CODE D_STR_CODE)
RETURNS (
    UA_CODE D_STR_100,
    UA_UX_CODE D_STR_CODE,
    UA_UXA_CODE D_STR_100,
    UA_ADD_CODE D_STR_CODE,
    UA_KIND D_SMALLINT,
    UA_NAME D_STR_100,
    UA_HINT D_STR_255,
    UA_DISABLE_ON_EDIT D_BOOL,
    UA_READONLY D_BOOL,
    UA_SAFE_MODE D_BOOL,
    UA_INACTIVE D_BOOL,
    RES_UX_KIND D_INTEGER,
    RES_IS_OWN D_BOOL,
    RES_CAPTION D_STR_255,
    RES_MENU_COUNT D_INTEGER)
AS
declare variable tmp_name d_str_100;
begin
  res_is_own = 1;
  -- если передан ua_code, возващаем одну запись
  for select ua_code,
             ua_ux_code,
             ua_uxa_code,
             ua_usr_id_field,
             uxa_kind,
             ua_name,
             ua_hint,
             ua_disable_on_edit,
             ua_readonly,
             ua_safe_mode,
             ua_inactive,
             ux_kind,
             uxa_name
      from uxaction, uxalias, uxcode
      where ua_code = :ux_code and
            uxa_code = ua_uxa_code and
            uxa_ux_code = ux_code
      into :ua_code,
           :ua_ux_code,
           :ua_uxa_code,
           :ua_add_code,
           :ua_kind,
           :ua_name,
           :ua_hint,
           :ua_disable_on_edit,
           :ua_readonly,
           :ua_safe_mode,
           :ua_inactive,
           :res_ux_kind,
           :tmp_name
  do
  begin
    -- если Action добавлен в меню или кнопке, указываем этот объект
    res_menu_count = 0;

    select count(*) from uxaliasaction
    where uaa_ua_code = :ua_code
    into res_menu_count;

    res_caption = tmp_name;
    if (tmp_name not containing ua_uxa_code) then
      res_caption = res_caption || '   - ' || ua_uxa_code;

    suspend;

    -- Если вернул результат, значит это UxAction, завершаем работу
    exit;
  end

  -- Список действий объекта
  for select ua_code,
             ua_ux_code,
             ua_uxa_code,
             ua_usr_id_field,
             uxa_kind,
             ua_name,
             ua_hint,
             ua_disable_on_edit,
             ua_readonly,
             ua_safe_mode,
             ua_inactive,
             ux_kind,
             uxa_name
      from uxaction, uxalias, uxcode
      where ua_ux_code = :ux_code and
            uxa_code = ua_uxa_code and
            uxa_ux_code = ux_code and
            ux_code <> :ux_code
      into :ua_code,
           :ua_ux_code,
           :ua_uxa_code,
           :ua_add_code,
           :ua_kind,
           :ua_name,
           :ua_hint,
           :ua_disable_on_edit,
           :ua_readonly,
           :ua_safe_mode,
           :ua_inactive,
           :res_ux_kind,
           :tmp_name
  do
  begin
    -- если Action добавлен в меню или кнопке, указываем этот объект
    res_menu_count = 0;

    select count(*) from uxaliasaction
    where uaa_ua_code = :ua_code
    into res_menu_count;

    res_caption = tmp_name;
    if (tmp_name not containing ua_uxa_code) then
      res_caption = res_caption || '   - ' || ua_uxa_code;

    suspend;
  end

  -- Список действий где вызывается данный объект
  res_is_own = 0;
  for select ua_code,
             ua_ux_code,
             ua_uxa_code,
             ua_usr_id_field,
             uxa_kind,
             ua_name,
             ua_hint,
             ua_disable_on_edit,
             ua_inactive,
             ux_kind,
             ux_name
      from uxaction, uxalias, uxcode
      where uxa_ux_code = :ux_code and
            ua_uxa_code = uxa_code and
            ua_ux_code = ux_code and
            ua_ux_code <> :ux_code
      into :ua_code,
           :ua_ux_code,
           :ua_uxa_code,
           :ua_add_code,
           :ua_kind,
           :ua_name,
           :ua_hint,
           :ua_disable_on_edit,
           :ua_inactive,
           :res_ux_kind,
           :tmp_name
  do
  begin
    -- если Action добавлен в меню или кнопке, указываем этот объект
    res_menu_count = 0;

    if (ua_ux_code <> 'fm_MainMenu') then
    begin
      select count(*) from uxaliasaction
      where uaa_ua_code = :ua_code
      into res_menu_count;

      res_caption = tmp_name;
      if (tmp_name not containing ua_ux_code) then
        res_caption = res_caption || '   - ' || ua_ux_code;
    end else
    begin
      res_caption = 'Главное меню';

      if (exists(select * from menulist
      where ml_ua_code = :ua_code)) then
        res_menu_count = 1;
    end

    suspend;
  end
end
