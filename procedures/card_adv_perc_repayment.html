CREATE PROCEDURE CARD_ADV_PERC_REPAYMENT(
    OBJECT_KIND D_STR_100 NOT NULL,
    OBJECT_ID D_ID NOT NULL,
    PARENT_CS_ID D_INTEGER NOT NULL)
AS
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable ol_id d_id;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable office_id d_integer;
declare variable object_kind_str d_str_100;
declare variable adv_kommis d_currency;
declare variable un_id d_id;
declare variable card_id d_integer;
declare variable cca_id d_integer;
begin
    object_kind_str = f_objkind(object_kind, 'name');

    if (object_kind_str = 'kontract') then
        k_id = object_id;
    else
        exception e_message 'object_kind = '||object_kind_str||' не реализован';

    select cs.cs_from_un_id, cs.cs_card_id, cs.cs_from_cca_id
    from cardsync cs
    where cs.cs_id = :parent_cs_id
    into :un_id, :card_id, :cca_id;

    select kont.k_last_reg_id, k_kontract_no, k_disb_date
    from kontract kont
    where kont.k_id = :k_id
    into :office_id, :k_kontract_no, :k_disb_date;

    acc_dt = (
      select al_id
      from acc_list
      where al_from_reg_id = :office_id
        and al_currency = '417'
        and al_account = '20001/1');
    acc_kt = (
      select al_id
      from acc_list
      where al_from_reg_id = :office_id
        and al_currency = '417'
        and al_account = '21114/11');

    select sum(ok.ok_kommis)
    from operkred ok
    where ok.ok_from_k_id = :k_id
    and ok.ok_ol_is_active = 1
    and ok.ok_type in (1,4,5,9)
    into :adv_kommis;

    select kncl.kc_from_c_id, c_name_fio
    from knclient kncl
      inner join clients
        on c_id = kncl.kc_from_c_id
    where kncl.kc_from_k_id = :k_id
    and kncl.kc_amount_approved > 0
    into :c_id, :c_name_fio;
    
    if (adv_kommis < 0) then
        exception e_message 'По комисии не может быть переплаты (Коммиссия = '||adv_kommis||')';
    
    if (adv_kommis = 0) then -- если по продукту нет доп. %
        exit;

    select
      ol_id
    from  hp_do_repayment (
      :k_id,
      0,                    -- c_id
      current_date,         -- rep_date
      :adv_kommis,   -- repay_amount
      1,                    -- repay_to_os
      1,                    -- repay_to_perc
      1,                    -- repay_to_fine
      1,                    -- repay_to_kommis
      1,                    -- need_save
      0,                    -- is_full_repay
      0,                    -- is_writeoff
      :office_id,           -- repay_fact_office_id
      3,                    -- repay_dt_account_kind
      :acc_dt,              -- repay_dt_account_no
      0,                    -- repay_curr_rate
      :un_id,        -- repay_user_id
      :c_id,                -- repay_document_client
      null,                 -- repay_doc_client_name_reason
      0,                    -- repay_manager_c_id
      0)                    -- repay_from_rd_id
    into
      :ol_id;

    if (ol_id is not null) then
    begin
      update operlist set
        ol_memo = f_format('Кредитный дог.№%1 от %2%3%4',
          :k_kontract_no,
          f_datetostr(:k_disb_date),
          coalesce(f_format(' номер ID %1 (сч.%2) ', :card_id, :cca_id), ' '),
          :c_name_fio)
      where ol_id = :ol_id;
      update operbook set
        ob_account_kt = :acc_kt,
        ob_detail = 'Комиссия за выпуск и обслуживание карты Элкарт'
      where ob_from_ol_id = :ol_id;
    end else exception e_message 'Не удается сохранить платеж. ol_id = null';
end
