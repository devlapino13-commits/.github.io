CREATE PROCEDURE AAREP_IS_OFFICE_ONLINE(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RESULT_STR D_STR_20,
    OPC_DATETIME D_DATETIME,
    REG_NAME D_STR_CODE,
    NUMBER_OF_RECEIPTS_PAPER D_INTEGER,
    NUMBER_OF_CONSUMABLES_PAPER D_INTEGER,
    NUMBER_OF_RECEIPTS_ONLINE D_INTEGER,
    NUMBER_OF_CONSUMABLES_ONLINE D_INTEGER,
    RES_MANAGER_FIO D_STR_100)
AS
declare variable tmp_date d_date;
declare variable tmp_result_str d_str_20;
declare variable reg_unicode d_integer;
begin
  -- RFCKG-4607 Отчет Режим работы офисов онлайн/бумажный

  number_of_receipts_paper = 0;     -- Кол-во приходных бумажный
  number_of_consumables_paper = 0;  -- Кол-во расходных бумажный

  number_of_receipts_online = 0;    -- Кол-во приходных online
  number_of_consumables_online = 0; -- Кол-во расходных online

  for select
             opc.opc_datetime,
             iif(opc.opc_is_online = 1, 'онлайн', 'бумажный'),
             reg.reg_name,
             reg.reg_unicode
  from opercash opc
  inner join region_list(:rep_region_id, 1, -1, 1) on opc.opc_office_id = id_list
  inner join region reg on opc.opc_office_id = reg.reg_unicode
  where opc.opc_is_delete = 0 and
        opc.opc_status >= 0 and
        cast(opc.opc_datetime as d_date) between :rep_date_begin and :rep_date_end
  order by reg_unicode, opc.opc_datetime
  into opc_datetime,
       result_str,
       reg_name,
       reg_unicode
  do
  begin

    if ((tmp_date <> cast(opc_datetime as d_date)) or (tmp_result_str <> result_str)) then  -- мы берем только (первую запись за день) или (если меняется opc_is_online то есть мяняется с онлайн на бумажный и наоборот)
    begin
      select c.c_name_fio
      from cashbookmain cbm
      inner join clients c on cbm.cbm_cashier_id = c.c_id
      where cbm.cbm_office_id = :reg_unicode and
            cbm.cbm_date = cast(:opc_datetime as d_date) and
            cbm.cbm_invalid = 0 rows 1
      into :res_manager_fio;   -- ФИО Кассир(последний формировавший кассовый журнл)

      select 
         sum(case when opc.opc_sign = 1 and opc.opc_is_online = 0  then 1 else 0 end),
         sum(case when opc.opc_sign = -1 and opc.opc_is_online = 0 then 1 else 0 end),
         sum(case when opc.opc_sign = 1 and opc.opc_is_online = 1  then 1 else 0 end),
         sum(case when opc.opc_sign = -1 and opc.opc_is_online = 1 then 1 else 0 end)
      from opercash opc
      where opc.opc_office_id = :reg_unicode and
            opc.opc_is_delete = 0 and
            opc.opc_status >= 0 and
            cast(opc.opc_datetime as d_date) = cast(:opc_datetime as d_date)
      into :number_of_receipts_paper,
           :number_of_consumables_paper,
           :number_of_receipts_online,
           :number_of_consumables_online;

      suspend;
    end

    tmp_date = cast(opc_datetime as d_date);
    tmp_result_str = result_str;
  end

end
