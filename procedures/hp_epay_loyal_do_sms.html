CREATE PROCEDURE HP_EPAY_LOYAL_DO_SMS(
    AG_ID D_INTEGER,
    ACTION_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable edl_id d_id;
begin
  if (coalesce(action_kind, '') = '') then
    exception e_default 'Action_kind is empty';

  if (action_kind not in ('sms_doc_one', 'sms_card_one', 'sms_doc_all', 'sms_card_all')) then
    exception e_default 'Incorrect action_kind: ' || action_kind;

  res_text = '';
  res_is_error = 0;

  if (coalesce(ag_id, 0) = 0) then
  begin
    if (action_kind in ('sms_doc_one', 'sms_card_one')) then
    begin
      res_text = 'Соглашение не выбрано';
      res_is_error = 1;
    end
  end

  if (res_text = '' and action_kind in ('sms_doc_one', 'sms_doc_all')) then
  begin
    res_text = 'Данный тип рассылки еще не согласован';
    res_is_error = 1;
  end

  if (res_text = '' and action_kind in ('sms_card_one', 'sms_card_all')) then
  begin
    if (action_kind = 'sms_card_one') then
    begin
      select ag_object_id
      from agreements
      where ag_id = :ag_id
      into c_id;

      select edl_id, res_text
      from hp_edoc_ins_bankcard('bankcard.moneta', :c_id, :un_id)
      into edl_id, res_text;

      if (res_text <> '') then
        res_is_error = 1;
      else
      if (edl_id is null) then
      begin
        res_is_error = 1;
        res_text = 'Неизвестная ошибка';
      end
      else
        res_text = 'СМС отправлен';
    end else
    begin
      res_is_error = 0;
      res_text = 'Групповая рассылка на стадии тестирования';
    end
  end
  suspend;
end
