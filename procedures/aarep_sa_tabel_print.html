declare variable sat_number integer;
declare variable sat_prior decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable sat_year_month integer;
declare variable tnorm integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  select sum(scd_work_hours)
  from sa_calendays
  where scd_year_month = (:report_year * 100 + :report_month)
  into :tnorm;
  if (tnorm is null) then
    tnorm = 0;

  sat_no = '';
  sat_number = 0;
  sat_prior = 0;
  for select sat_from_c_id, sat_year_month, sat_from_sawt_id, sat_office_id, sat_day_1, sat_day_2, sat_day_3, sat_day_4,
             sat_day_5, sat_day_6, sat_day_7, sat_day_8, sat_day_9, sat_day_10, sat_day_11, sat_day_12, sat_day_13,
             sat_day_14, sat_day_15, sat_day_16, sat_day_17, sat_day_18, sat_day_19, sat_day_20, sat_day_21, sat_day_22,
             sat_day_23, sat_day_24, sat_day_25, sat_day_26, sat_day_27, sat_day_28, sat_day_29, sat_day_30, sat_day_31,
             sat_total_hours, sat_total_calc, sat_is_ready, sat_is_new_added, c_name_fio, sawt_name, sat_days_total,
             sat_days_calc
      from sa_tabel, clients, sa_worktype, region_list(:report_region, 1, -1, 1)
      where sat_from_c_id = c_id and
            sat_from_sawt_id = sawt_id and
            sat_year_month = (:report_year * 100 + :report_month) and
            (sat_office_id = id_list) and
            sat_is_ready = 1
      order by sat_office_id, c_name_fio, sat_from_sawt_id
      into :sat_from_c_id, :sat_year_month, :sat_from_sawt_id, :sat_office_id, :sat_day_1, :sat_day_2, :sat_day_3,
           :sat_day_4, :sat_day_5, :sat_day_6, :sat_day_7, :sat_day_8, :sat_day_9, :sat_day_10, :sat_day_11,
           :sat_day_12, :sat_day_13, :sat_day_14, :sat_day_15, :sat_day_16, :sat_day_17, :sat_day_18, :sat_day_19,
           :sat_day_20, :sat_day_21, :sat_day_22, :sat_day_23, :sat_day_24, :sat_day_25, :sat_day_26, :sat_day_27,
           :sat_day_28, :sat_day_29, :sat_day_30, :sat_day_31, :sat_total_hours, :sat_total_calc, :sat_is_ready,
           :sat_is_new_added, :sat_personal_fio, :sat_line_name, :sat_days_total, :sat_days_calc
  do
  begin
    if (sat_from_sawt_id = 0) then
      sat_total_norma = tnorm;
    else
      sat_total_norma = 0;

    sat_month = mod(sat_year_month, 100);
    sat_year = (sat_year_month - sat_month) / 100;

    if (sat_from_c_id <> sat_prior) then
    begin
      sat_number = sat_number + 1;
      sat_no = cast(sat_number as varchar(20));
    end
    else
      sat_no = '';

    sat_prior = sat_from_c_id;

    if (sat_day_1 = 0) then
      sat_day_1 = null;
    if (sat_day_2 = 0) then
      sat_day_2 = null;

      sat_day_3 = null;

    if (sat_day_20 = 0) then
