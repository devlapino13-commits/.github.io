CREATE PROCEDURE HP_KONTEFFRATE_INSERT(
    K_ID D_ID,
    CALC_DATE D_DATE,
    PERCENT_EFFECT_FIFD D_PERCENT = null)
AS
declare variable ker_id d_integer;
declare variable ker_state d_smallint;
declare variable new_ker_state d_smallint;
declare variable old_percent_effect_fifd d_percent;
begin
  if (not exists (select * from kontract where k_id = :k_id)) then
    exit;

  if (percent_effect_fifd <=0) then
    percent_effect_fifd = null;

  select ker_id, ker_state, ker_percent_effect_fifd
  from konteffrate
  where ker_from_k_id = :k_id
    and ker_date = :calc_date
  into ker_id, ker_state, old_percent_effect_fifd;

  if (ker_id is null) then
    insert into konteffrate (ker_date, ker_from_k_id, ker_percent_effect_fifd)
    values (:calc_date, :k_id, :percent_effect_fifd);
  else if (percent_effect_fifd is null) then
  begin
    new_ker_state = iif(ker_state <> 1, 1, null);
    percent_effect_fifd = 0;
  end
  else if (percent_effect_fifd <> old_percent_effect_fifd or ker_state <> 0) then
   new_ker_state = 0;

  if (new_ker_state is not null) then
    update konteffrate
    set ker_state = :new_ker_state,
        ker_percent_effect_fifd = :percent_effect_fifd
    where ker_id = :ker_id;

end
