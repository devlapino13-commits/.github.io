CREATE PROCEDURE HP_CARDPINCHANGE_EDOC_INSERT(
    CPH_ID D_INTEGER,
    UN_ID D_ID,
    REG_ID D_INTEGER)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable edl_id d_id;
begin

  select
    edl_id, res_text
  from hp_edoc_insert('change_pincode_application', f_objkind('cardpinchange', 'id'), :cph_id, :un_id)
  into :edl_id, :res_text;

  if (row_count = 0) then
    res_text = 'Эл. документ не создан';

  res_text = coalesce(res_text, '');
  if (res_text = '') then
    update cardpinchange
    set cph_from_edl_id = :edl_id,
        cph_status = 1,
        cph_from_un_id = :un_id,
        cph_region_id = :reg_id,
        cph_from_cs_id = null
    where cph_id = :cph_id;

  suspend;
end
