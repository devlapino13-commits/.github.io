CREATE PROCEDURE HP_BANKRUPTCY_INSERT(
    EIL_ID D_ID)
AS
declare variable K_ID D_ID;
declare variable C_ID D_ID;
declare variable BLS_ID D_ID;
declare variable BANK_STATE D_STR_255;
declare variable BANK_DATE D_DATE;
declare variable JSON_BLOB D_BLOB_TEXT;
declare variable IDX D_INTEGER;
declare variable PERSON_COUNT D_INTEGER;
declare variable BANK_TYPE D_STR_100;
declare variable IS_BANKRUPTCY d_integer;
begin
    -- Получаем данные из extinfolist
    select eil.eil_from_c_id,
           eil.eil_object_id,
           eil.eil_from_bls_id
    from extinfolist eil
    inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
    where eil.eil_id = :eil_id
      and eil.eil_status = 2
      and eic.eic_code = 'bankruptcy'
    rows 1
    into c_id, k_id, bls_id;

    select is_bankruptcy
    from get_bankruptcy_status(:c_id)
    into is_bankruptcy;

    if (:is_bankruptcy = 0) then
        exit;

    -- Получаем JSON
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into json_blob;

    if (row_count is null or coalesce(json_blob, '') = '') then
        exit;

    -- Определяем количество элементов массива PersonInfo
    person_count = f_json_blob_get_count(:json_blob,
        'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo');

    if (person_count > 0) then
    begin
        -- если массив, перебираем все элементы
        idx = 0;
        while (idx < person_count) do
        begin
            bank_state = f_json_blob_get_value(:json_blob,
                'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo[' || IDX || '].StatementStatus', null);

            if (bank_state = 'Zaversheno') then
            begin
                bank_date = cast(substring(f_json_blob_get_value(:json_blob,
                    'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo[' || IDX || '].BankruptStartDate', null)
                    from 1 for 10) as date);

                bank_type = f_json_blob_get_value(:json_blob,
                    'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo[' || IDX || '].BankruptcyType', null);

                insert into bankruptcy_for_exec (bfe_from_c_id, bfe_from_k_id, bfe_from_eil_id, 
                                                 bfe_bankruptcy_date, bfe_status, bfe_bank_type, bfe_bank_status)
                values (:c_id, :k_id, :eil_id, :bank_date, 0, :bank_type, :bank_state);

                exit;
            end

            IDX = IDX + 1;
        end
    end
    else
    begin
        -- Если массива нет, проверяем одиночный объект
        bank_state = f_json_blob_get_value(:json_blob,
            'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo.StatementStatus', null);

        if (bank_state = 'Zaversheno') then
        begin
            bank_date = cast(substring(f_json_blob_get_value(:json_blob,
                'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo.BankruptStartDate', null)
                from 1 for 10) as date);

            bank_type = f_json_blob_get_value(:json_blob,
                'response.responseData.data.ResponseByIin.PersonInfos.PersonInfo.BankruptcyType', null);

            insert into bankruptcy_for_exec (bfe_from_c_id, bfe_from_k_id, bfe_from_eil_id, 
                                             bfe_bankruptcy_date, bfe_status, bfe_bank_type, bfe_bank_status)
            values (:c_id, :k_id, :eil_id, :bank_date, 0, :bank_type, :bank_state);
        end
    end
end
