CREATE PROCEDURE CM_RU_0234_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id id_type;
declare variable pr_code d_str_100;
declare variable mac_loan_target d_integer;
declare variable c_inn str_20;
declare variable req_status d_smallint;
declare variable actuality_date d_date;
declare variable ip_status d_smallint;
begin
    for select pr.pr_code, mac.mac_from_c_id, mac.mac_loan_target
    from miniapp ma
    inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
        and mac.mac_amount > 0
    inner join product pr on ma.ma_from_pr_id = pr.pr_id
        and (pr.pr_code containing '.selfemployed.' or pr.pr_code containing '.business.')
    where ma.ma_id = :ma_id
        and ma.ma_is_active = 1
        and ma.ma_date_end = '31.12.9999'
    into :pr_code, :c_id, :mac_loan_target
    do
    begin
        c_inn = null;
        c_inn = (select c.c_inn from clients c where c.c_id = :c_id);

        if (c_inn is null) then
        begin
            res_error_is_critical = 2;
            res_error_text = 'ИНН не указан.';
            suspend;
        end 

        if (f_get_rd_id ('Loan_Target.72') <> mac_loan_target) then
        begin
            res_error_is_critical = 2;
            res_error_text = 'Некорректная цель займа.  Должна быть развитие бизнеса.';
            suspend;
        end


        -- RFCRU-5314
        -- Проверка анкеты
        if (check_kind in ('.miniapp.', '.create_loan.') and pr_code containing '.business.') then
        begin
            if (not exists
                (
                    select res_rd_code
                    from hp_objectprops_get(0, :c_id, -1, 'AnketaQuestions.LoanPurpose', current_date)
                    where res_rd_code = 'AnketaAnswers.9936'
                )
            ) then
            begin
                res_error_is_critical = 2;
                res_error_text = 'В анкете клиента должно быть выбрано значение "На развитие бизнеса"';
                suspend;
            end


            -- Проверка статуса
            res_error_is_critical = 0;
            res_error_text = '';

            select req_status, actuality_date, coalesce(ip_status, 0)
            from hp_kontur_ip_get_base_info('miniapp', :ma_id)
            into req_status, actuality_date, ip_status;

            -- Как видно из блока ниже, валиден только ip_status = 1 (Клиент является ИП)
            res_error_text =
                case
                    when req_status is null then 'Сначала проверьте статус ИП'
                    when current_date >= actuality_date then 'Сначала проверьте статус ИП'
                    when req_status < 0 then 'Ошибка'
                    when req_status = 0 then 'Сначала проверьте статус ИП'
                    when req_status = 1 then 'Сначала проверьте статус ИП'
                    when req_status = 2 and ip_status = 1 then ''
                    when req_status = 2 and ip_status = 0 then 'Нет действующего статуса ИП, измените продукт'
                    else 'Не предвиденная ошибка, просим обратиться в службу технической поддержки'
                end;

            if (res_error_text <> '') then
            begin
                res_error_is_critical = 2;
                suspend;
            end


        end

    end
end
