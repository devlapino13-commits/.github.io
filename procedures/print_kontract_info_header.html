CREATE PROCEDURE PRINT_KONTRACT_INFO_HEADER(
    K_ID D_ID)
RETURNS (
    K_NO D_STR_CODE,
    CLIENT_FIO D_STR_100,
    REG_NAME D_STR_CODE,
    ADDRESS_STR D_STR_255,
    MANAGER_FIO D_STR_100,
    MANAGER_CELL_NO D_STR_20,
    KS_NAME D_STR_100,
    KS_CELL_NO D_STR_100)
AS
declare variable reg_id d_integer;
declare variable jl_id_list d_str_255;
declare variable c_id d_id;
begin
  select k.k_kontract_no,
         reg.reg_name,
         adl.adl_address_str,
         reg.reg_unicode,
         c.c_name_fio,
         f_cont_info('clients', c.c_id, 'cell_main', 'last')
  from kontract k
  inner join region reg on reg.reg_unicode = k.k_last_reg_id
  inner join addrlist adl on adl.adl_id = reg.reg_address_fact
  inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id and
        current_date between kal.kal_date and coalesce(kal.kal_date_end, '31.12.2999')
  inner join clients c on kal.kal_new_kf_officer = c.c_id
  where k.k_id = :k_id
  into :k_no,
       :reg_name,
       :address_str,
       :reg_id,
       :ks_name,
       :ks_cell_no;

  -- Менеджер офиса

  select '.' || list(jl.jl_id, '.') || '.'
  from joblist jl
  inner join jobkind jk on jk.jk_id = jl.jl_kind
  where jk.jk_system_code = 'JobSystem.Region_KM'
  into :jl_id_list;

   for select cl.c_name_fio, cl.c_id
       from hr_pstatus ps
       inner join clients cl on cl.c_id = ps.ps_from_c_id
       where :jl_id_list containing '.' || ps.ps_from_jl_id || '.' and
             current_date between ps.ps_date_begin and ps.ps_date_end and
             ps.ps_is_working = 1 and
             ps.ps_status = 1 and
             ps.ps_from_reg_id = :reg_id
       into :manager_fio, :c_id
   do
   begin
     manager_cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');
   end

  for select c.c_name_fio
      from knclient kc
      inner join clients c on c.c_id = kc.kc_from_c_id
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      into :client_fio
  do
  begin
    suspend;
  end
end
