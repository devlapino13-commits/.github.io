CREATE PROCEDURE GET_STORED_PROC_OPTIONS
RETURNS (
    ST_PROC_NAME VARCHAR(255),
    ST_CLIENT_LEVEL INTEGER,
    ST_STAFF_LEVEL INTEGER,
    ST_USER_CLIENT_ACCESS INTEGER,
    ST_USER_STAFF_ACCESS INTEGER,
    RN_LEVEL_CLIENT_FIO INTEGER,
    RN_LEVEL_STAFF_FIO INTEGER,
    UN_LOGIN VARCHAR(255),
    JK_SYSTEM_ID INTEGER)
AS
declare variable un_name varchar(255);
declare variable role_id decimal(18,0);
declare variable un_id decimal(18,0);
declare variable tmp integer;
declare variable uniqtab_permactions integer;
begin
  st_user_client_access = 0;
  st_user_staff_access = 0;

  un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN');

  select un_id
  from username
  where un_name = :un_name
  into :un_id;
  if (un_id is null) then
    un_id = 0;

  tmp = 0;
  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Client_Data_Edit')
  into :tmp;
  if (tmp = 1) then
    st_user_client_access = 2;
  else
  begin
    select has_access
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Client_Data_Read')
    into :tmp;
    if (tmp = 1) then
      st_user_client_access = 1;
  end

  tmp = 0;
  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Staff_Data_Edit')
  into :tmp;
  if (tmp = 1) then
    st_user_staff_access = 2;
  else
  begin
    select has_access
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Staff_Data_Read')
    into :tmp;
    if (tmp = 1) then
      st_user_staff_access = 1;
  end

  select has_access
  from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
  into :tmp;
  if (tmp = 1) then
  begin
    st_user_client_access = 2;
    st_user_staff_access = 2;
  end
  suspend;
end
