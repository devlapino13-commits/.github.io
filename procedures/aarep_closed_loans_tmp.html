CREATE PROCEDURE AAREP_CLOSED_LOANS_TMP(
    REP_DATE_END DATE)
RETURNS (
    REP_K_ID ID_TYPE,
    DATE_DISB_YEAR SMALLINT,
    DATE_DISB_MONTH SMALLINT,
    DATE_DISB DATE,
    DATE_CLOSE DATE,
    DATE_CLOSE_PLAN DATE,
    KOL_CLI_DISB SMALLINT,
    DISB_AMOUNT CURR_TYPE,
    IS_DOSROCH SMALLINT,
    CLOSED_MONTH SMALLINT,
    CLOSE_MONTH_0_KOL SMALLINT,
    CLOSE_MONTH_1_KOL SMALLINT,
    CLOSE_MONTH_2_KOL SMALLINT,
    CLOSE_MONTH_3_KOL SMALLINT,
    CLOSE_MONTH_4_KOL SMALLINT,
    CLOSE_MONTH_5_KOL SMALLINT,
    CLOSE_MONTH_6_KOL SMALLINT,
    CLOSE_MONTH_7_KOL SMALLINT,
    CLOSE_MONTH_8_KOL SMALLINT,
    CLOSE_MONTH_9_KOL SMALLINT,
    CLOSE_MONTH_10_KOL SMALLINT,
    CLOSE_MONTH_11_KOL SMALLINT,
    CLOSE_MONTH_12_KOL SMALLINT,
    CLOSE_MONTH_13_KOL SMALLINT,
    CLOSE_MONTH_14_KOL SMALLINT,
    CLOSE_MONTH_15_KOL SMALLINT,
    CLOSE_MONTH_16_KOL SMALLINT,
    CLOSE_MONTH_17_KOL SMALLINT,
    CLOSE_MONTH_18_KOL SMALLINT)
AS
declare variable ks_date_end date;
begin
  for select k_id, ks_date_begin, ks_date_end, k_kolich_client_approved, k_amount_approved, k_kredline_end
      from kontstate, kontract
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin <= :rep_date_end and
            ks_from_k_id = k_id
      into rep_k_id, date_disb, ks_date_end, kol_cli_disb, disb_amount, date_close_plan
  do
  begin
    date_close = null;
    is_dosroch = 0;
    closed_month = null;
    if (ks_date_end <> '31.12.9999') then
    begin
      date_close = ks_date_end + 1;
      if (date_close < date_close_plan and
          extract(month from date_close) <> extract(month from date_close_plan)) then
      begin
        is_dosroch = 1;
        closed_month = datediff(month, :date_disb, :date_close);
      end
    end
    close_month_0_kol = 0;
    close_month_1_kol = 0;
    close_month_2_kol = 0;
    close_month_3_kol = 0;
    close_month_4_kol = 0;
    close_month_5_kol = 0;
    close_month_6_kol = 0;
    close_month_7_kol = 0;
    close_month_8_kol = 0;
    close_month_9_kol = 0;
    close_month_10_kol = 0;
    close_month_11_kol = 0;
    close_month_12_kol = 0;
    close_month_13_kol = 0;
    close_month_14_kol = 0;
    close_month_15_kol = 0;
    close_month_16_kol = 0;
    close_month_17_kol = 0;
    close_month_18_kol = 0;
    if (is_dosroch = 1) then
    begin
      if (closed_month = 0) then
        close_month_0_kol = kol_cli_disb;
      if (closed_month = 1) then
        close_month_1_kol = kol_cli_disb;
      if (closed_month = 2) then
        close_month_2_kol = kol_cli_disb;
      if (closed_month = 3) then
        close_month_3_kol = kol_cli_disb;
      if (closed_month = 4) then
        close_month_4_kol = kol_cli_disb;
      if (closed_month = 5) then
        close_month_5_kol = kol_cli_disb;
      if (closed_month = 6) then
        close_month_6_kol = kol_cli_disb;
      if (closed_month = 7) then
        close_month_7_kol = kol_cli_disb;
      if (closed_month = 8) then
        close_month_8_kol = kol_cli_disb;
      if (closed_month = 9) then
        close_month_9_kol = kol_cli_disb;
      if (closed_month = 10) then
        close_month_10_kol = kol_cli_disb;
      if (closed_month = 11) then
        close_month_11_kol = kol_cli_disb;
      if (closed_month = 12) then
        close_month_12_kol = kol_cli_disb;
      if (closed_month = 13) then
        close_month_13_kol = kol_cli_disb;
      if (closed_month = 14) then
        close_month_14_kol = kol_cli_disb;
      if (closed_month = 15) then
        close_month_15_kol = kol_cli_disb;
      if (closed_month = 16) then
        close_month_16_kol = kol_cli_disb;
      if (closed_month = 17) then
        close_month_17_kol = kol_cli_disb;
      if (closed_month = 18) then
        close_month_18_kol = kol_cli_disb;
    end
    date_disb_year = extract(year from :date_disb);
    date_disb_month = extract(month from :date_disb);

    suspend;
  end
end
