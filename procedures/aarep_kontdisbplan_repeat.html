CREATE PROCEDURE AAREP_KONTDISBPLAN_REPEAT(
    REP_DATE D_DATE)
RETURNS (
    OFFICE_NAME D_STR_100,
    CLIENT_NUM D_INTEGER,
    CLIENT_NAME D_STR_100)
AS
declare variable reg_id d_integer;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable prev_office_name d_str_100;
begin
  -- возвращает список повторных клиентов с договорами в процессе в разрезе офисов на дату (RFCAZ-244)

  for select kdp_office_id,
             k_id
      from kontract
      inner join kontdisbplan on kdp_office_id = k_last_reg_id and
            kdp_date_begin = k_disb_date
      where k_status = 0 and
            kdp_date_begin = :rep_date
      order by kdp_office_id
      into reg_id,
           k_id
  do
  begin
    c_id = null;
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 rows 1
    into c_id;

    if (exists(select *
               from kontstate
               inner join knclient on kc_from_k_id = ks_from_k_id
               where kc_from_c_id = :c_id and
                     kc_amount_approved > 0 and
                     ks_ol_is_delete = 0 and
                     ks_state in (1, 2) and
                     current_date between ks_date_begin and ks_date_end)) then
    begin
      office_name = null;
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into office_name;

      if (prev_office_name = office_name) then
        client_num = client_num + 1;
      else
        client_num = 1;

      client_name = null;
      select c_name_fio
      from clients
      where c_id = :c_id
      into client_name;

      suspend;
      prev_office_name = office_name;
    end
  end
end
