CREATE PROCEDURE FM_INSERT_CLDOC(
    C_ID DECIMAL(18,0),
    PK_ID INTEGER,
    UN_ID DECIMAL(18,0),
    IS_MAIN INTEGER,
    DOC_NO VARCHAR(30),
    DATE_BEGIN DATE,
    DATE_END DATE,
    DOC_ORG VARCHAR(255),
    DOC_ORG_CODE VARCHAR(20),
    FOR_COMPANY VARCHAR(255),
    TERRITORY VARCHAR(255),
    MEMO_1 VARCHAR(255),
    MEMO_2 VARCHAR(255),
    MEMO_3 VARCHAR(255),
    MEMO_4 VARCHAR(255),
    MEMO_5 VARCHAR(255),
    BLS_1 D_ID = null,
    BLS_2 D_ID = null)
RETURNS (
    CDS_ID INTEGER,
    ERR_TEXT VARCHAR(255))
AS
declare variable NEED_PHOTO2 integer;
declare variable CDS_IS_OLD_DOC integer;
declare variable CDS_VERIFY_STATUS integer;
declare variable DB_ID integer;
declare variable ACL_STATUS integer;
begin
  err_text = '';
  cds_id = 0;
  uxf$set_user(un_id);
  db_id = f_get_db_id();

  /*  RFCKZ-245
  if (db_id = 39801 and
     (exists(select *
            from passkind
            inner join passdoctype on pdt_id = pk_from_pdt_id
            where pk_id = :pk_id and
                 pdt_code = 'card_statement'))) then
  begin
    select cl.c_name_fio,
       iif(coalesce(nkh.nkh_is_gambler, 0) = 1
           or coalesce(nkh.nkh_is_overdue_90, 0) = 1
           or f_is_asp_client(cl.c_id) = 1,
                1, 0) as has_negativ_his
    from clients cl
    inner join negativ_kib_history nkh on nkh.nkh_from_c_id = cl.c_id
          and nkh.nkh_last = 1
    where cl.c_id = :c_id
    order by nkh.nkh_from_eil_id desc
    rows 1
    into :c_name_fio, :has_negativ_his;

    if (:has_negativ_his = 1) then
    begin
      err_text = 'Для клиента: ' || :c_name_fio || ' запрещено прикреплять выписку по карте';
      suspend;
      exit;
    end
  end
  */

  if (date_end is not null and date_end < current_date and
    not exists(
      select *
      from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_AttachExpiredDoc')
      where has_access = 1)) then
  begin
    err_text = 'Срок документа истек. Требуется действующий документ.';
    suspend;
    exit;
  end

  if (date_begin > current_date) then
  begin
    err_text = 'Неверная дата выдачи документа (еще не наступила).';
    suspend;
    exit;
  end

  if (extract(year from date_begin) < 1900) then
  begin
    err_text = 'Неверная дата выдачи документа (очень старая).';
    suspend;
    exit;
  end

  if (db_id = 81001) then
  begin
    if ((select inn_received
         from inn_query_status(:c_id)) = 1 and
        exists(select *
               from passkind
               inner join passdoctype on pdt_id = pk_from_pdt_id
               where pk_id = :pk_id and
                     pdt_code = 'info_inn')) then
    begin
      err_text = 'Сохранить не удалось, так как ИНН от ФНС ранее уже был получен';
      suspend;
      exit;
    end
  end

  if (db_id = 81001 and (select is_oivo from hp_check_oivo_assignment(:c_id, 1000000000000)) = 1 and
      exists(select *
             from passkind
             inner join passdoctype on pdt_id = pk_from_pdt_id
             where pk_id = :pk_id and
                   pdt_code = 'selfie_photo')) then
  begin
    select acl.acl_status
    from actionlist acl
    where acl.acl_object_kind = 0 and
          acl.acl_object_id = :c_id and
          acl.acl_from_rd_id = f_get_rd_id('ActionGroup.Passport_Verify_OiVO') and
          acl.acl_status in (1, 2)
    into :acl_status;

    if (acl_status is not null) then
    begin
      if (acl_status = 1) then
        err_text = 'Паспорт находится на рассмотрении. Добавление фотографии в слот «Фото идентификация» временно недоступно';
      else
        err_text = 'Проверка завершена успешно. Повторная загрузка фотографии не требуется';
      suspend;
      exit;
    end
  end

  if (doc_no is null) then
    doc_no = '';

  if (is_main = 1) then
  begin
    -- Паспорта


    if (exists(select *
               from actionorder
               where aco_object_kind = 0 and
                     aco_object_id = :c_id and
                     aco_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO')) and
                     aco_status = 1)) then
    begin
      err_text = 'Клиент находится на верификации. Изменение запрещено до принятия решения.';
      suspend;
      exit;
    end

    -- Корректировки по запросу Алмаза
    doc_no = upper(doc_no);
    doc_org = upper(doc_org);
    doc_org_code = upper(doc_org_code);
    if (db_id = 417001) then
    begin
      -- Гарантированный пробел после третьего символа
      doc_org = trim(f_replace_all(doc_org, '  ', ' '));
      if (substring(doc_org from 3 for 1) <> ' ') then
        doc_org = left(doc_org, 3) || ' ' || substring(doc_org from 3);
    end

    update clients
    set c_doc_type = :pk_id,
        c_pass_no = :doc_no,
        c_pass_date = :date_begin,
        c_pass_date_expire = :date_end,
        c_pass_who = :doc_org,
        c_pass_who_code = :doc_org_code,
        c_from_un_id = :un_id,
        c_blob_pass1_id = :bls_1,
        c_blob_pass2_id = :bls_2
    where c_id = :c_id;

    select cds_id
    from clientdocs
    where cds_from_c_id = :c_id and
          cds_is_delete = 0 and
          cds_is_main_doc = 1
    order by cds_id desc
    rows 1
    into :cds_id;

    select drm_has_page2
    from hp_get_drm_has_pages(:pk_id)
    into :need_photo2;

    -- Отправка на верификацию, если все необходимые данные заполнены
    if (exists(select *
               from clients
               where c_id = :c_id and
                     coalesce(c_name_fio, '') <> '' and
                     c_birth_date is not null and
                     coalesce(c_pass_no, '') <> '' and
                     coalesce(c_pass_date, '') <> '' and
                     coalesce(c_pass_who, '') <> '' and
                     coalesce(c_blob_photo_id, 0) > 0 and
                     coalesce(c_blob_pass1_id, 0) > 0 and
                     (coalesce(c_blob_pass2_id, 0) > 0 or
                     :need_photo2 = 0) and
                     c_verify_status <> 1)) then
    begin
      if (db_id in (39801, 41701)) then
      begin --Доп прооверки для КГ
        if (exists(select *
                   from hp_status_param_verif(:c_id)
                   where result = 0)) then
        begin
          update clients
          set c_verify_status = 1,
              c_from_un_id = :un_id
          where c_id = :c_id;
        end

      end
      else
      begin
        update clients
        set c_verify_status = 1,
            c_from_un_id = :un_id
        where c_id = :c_id;
      end
    end

  end
  else
  begin
    -- Все, кроме паспортов
    cds_id = null;
    if (doc_no <> '') then
    begin
      select cds_id, iif(cds_doc_date_end < current_date or cds_is_delete = 1, 1, 0) as cds_is_old_doc,
             cds_verify_status
      from clientdocs cds, passkind, passdoctype
      where cds_from_c_id = :c_id and
            cds_from_pk_id = :pk_id and
            pk_id = :pk_id and
            pk_from_pdt_id = pdt_id and
            cds_is_delete = 0 and
            replace(cds_doc_no, ' ', '') = replace(:doc_no, ' ', '') and
            cds.cds_datetime >= current_date - 7
      order by cds_datetime desc
      rows 1
      into :cds_id, :cds_is_old_doc, :cds_verify_status;
    end
    if ((cds_id is null) or (cds_is_old_doc = 1) or (cds_verify_status > 0)) then
    begin
      insert into clientdocs (cds_from_c_id, cds_from_pk_id, cds_from_un_id, cds_verify_status, cds_doc_no,
                              cds_doc_date_begin, cds_doc_date_end, cds_doc_issued_by, cds_doc_issued_code, cds_doc_for_company,
                              cds_doc_territory, cds_memo_1, cds_memo_2, cds_memo_3, cds_memo_4, cds_memo_5)
      values (:c_id, :pk_id, :un_id, -3, :doc_no, :date_begin, :date_end, :doc_org, :doc_org_code, :for_company, :territory, :memo_1,
              :memo_2, :memo_3, :memo_4, :memo_5)
      returning cds_id
      into :cds_id;
    end
    else
    begin
      update clientdocs
      set cds_doc_no = :doc_no,
          cds_doc_date_begin = :date_begin,
          cds_doc_date_end = :date_end,
          cds_doc_issued_by = :doc_org,
          cds_doc_issued_code = :doc_org_code,
          cds_doc_for_company = :for_company,
          cds_doc_territory = :territory,
          cds_memo_1 = :memo_1,
          cds_memo_2 = :memo_2,
          cds_memo_3 = :memo_3,
          cds_memo_4 = :memo_4,
          cds_memo_5 = :memo_5,
          cds_from_un_id = :un_id
      where (cds_id = :cds_id);
    end
  end
  suspend;
end
