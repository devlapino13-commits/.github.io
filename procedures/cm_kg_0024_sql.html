CREATE PROCEDURE CM_KG_0024_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable poruch_qty integer;
declare variable k_amount_approved curr_type;
declare variable k_disb_date date;
declare variable c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable kc_circle smallint;
declare variable ma_app_id integer;
declare variable kf_id id_type;
declare variable pr_name str_100;
declare variable pr_type str_100;
declare variable pr_code str_100;
declare variable min_perc decimal(15,2);
declare variable min_dop_perc decimal(15,2);
declare variable max_perc decimal(15,2);
declare variable max_dop_perc decimal(15,2);
declare variable k_percent perc_type;
declare variable k_kommis_perc perc_type;
declare variable unpaid_own_amount curr_type;
declare variable kh_repaystatus str_100;
declare variable kh_creditstatus str_100;
declare variable kh_deliqdayscount integer;
declare variable kh_is_our smallint;
declare variable kh_dateaward date;
declare variable own_amount curr_type;
declare variable group_unpaid_amount curr_type;
declare variable group_amount curr_type;
declare variable kh_count integer;
declare variable kh_clients_kol integer;
declare variable kbm_id id_type;
declare variable cu_id varchar(3);
declare variable cr_rate curr_type;
declare variable cli_percent curr_type;
declare variable cu_symbol varchar(3);
declare variable k_id_tmp id_type;
declare variable counter integer;
declare variable pr_type_old str_100;
declare variable k_percent_old perc_type;
declare variable k_kommis_perc_old perc_type;
declare variable k_id_old id_type;
declare variable max_delinq_days integer;
declare variable max_day_qty integer;
declare variable is_stop smallint;
declare variable k_kontract_no_old str_100;
declare variable k_amount_approved_max curr_type;
declare variable min_dop_perc_finished decimal(15,2);
declare variable min_perc_finished decimal(15,2);
begin
  poruch_qty = null;
  min_dop_perc_finished = 0;
  min_perc_finished = 0;
  for select kont.k_disb_date,
             kncl.kc_from_c_id,
             kncl.kc_amount_approved,
             kncl.kc_circle,
             cl.c_name_fio,
             kont.k_from_ma_app_id,
             kont.k_from_kf_id,
             pr.pr_code,
             kont.k_percent,
             kont.k_kommis_perc,
             pr.pr_name,
             iif (pr_code containing ('.EC.'), 'ECONOM', 'STANDART') -- если продукт не Эконом, тогда считаю что это Стандарт
  from kontract kont
  inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
  inner join product  pr   on pr.pr_id = kont.k_from_pr_id
  inner join clients  cl   on cl.c_id = kncl.kc_from_c_id
  where kont.k_id = :k_id
  and pr.pr_code containing('.BIRGE.')
  and pr_loan_type = 1 -- << Индив
  and kncl.kc_amount_approved > 0
  into :k_disb_date,
       :c_id,
       :kc_amount_approved,
       :kc_circle,
       :res_info_text,
       :ma_app_id,
       :kf_id,
       :pr_code,
       :k_percent,
       :k_kommis_perc,
       :pr_name,
       :pr_type
  do
  begin
    if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
      exit;
    poruch_qty = null;
    select datediff(year, c_birth_date, :k_disb_date)from zalog  zal, clients
    where zal.z_from_k_id = :k_id
    and zal.z_type = 0
    and zal.z_subtype = 1
    and zal.z_from_c_id = c_id
    rows 1
    into :poruch_qty;
    
    poruch_qty = coalesce(poruch_qty, 0);
    
    if (poruch_qty < 1) then
      begin
        res_error_text='Не указан поручитель!';
        res_error_is_critical=2;
        suspend;
      end
    
    if (poruch_qty not between 18 and 70) then
      begin
        res_error_text='Возраст поручителя должен быть от 18 до 70 лет!';
        res_error_is_critical=2;
        suspend;
      end
         --  Проверка на уровне КРЕДИТА ===================================== >>
         if (poruch_qty is null) then
         begin
             k_amount_approved_max = 55000;
             if (pr_type =  'STANDART') then
             begin
                min_perc = 37;
                max_perc = 43.5;
                min_dop_perc = 3.5;
                max_dop_perc = 3.9;
                min_perc_finished =  min_perc;
             end
             else if (pr_type = 'ECONOM') then
             begin
                min_perc = 34.9;
                max_perc = min_perc;
                min_perc_finished = min_perc;
                min_dop_perc = 4.5;
                max_dop_perc = 6.7;
             end
             if (pr_code containing ('.PAR.')) then
             begin     
                min_perc = 35; 
                min_perc_finished = min_perc;
                min_dop_perc = 3.5;
                max_dop_perc = min_dop_perc;
                pr_type = 'PARAL';
             end
             select count(*)
             from zalog zal
             where zal.z_from_k_id = :k_id
             and zal.z_type = 0
             and zal.z_subtype = 1
             into :poruch_qty;
    
             poruch_qty = coalesce(poruch_qty, 0);
    
             if (poruch_qty < 1) then
             begin
                res_error_text='Для продукта Бирге, минимум 1 поручитель';
                res_error_is_critical=1;
                suspend;
             end
             if ((k_amount_approved between 30001 and 55000) and (poruch_qty < 2))  then
             begin
                res_error_text='Для продукта Бирге, сумма кредита от 30001 до 55000, минимум 2 поручитель';
                res_error_is_critical=1;
                suspend;
             end
         end -- << =============================================================
         -- ПРОВЕРКА НА УРОВНЕ КЛИЕНТА  ===================================== >>
         -- ПРОВЕРКА КРЕД ИСТОРИИ КИБ --------------------------------------- >>
         select kbm_id
         from hp_kibminiapp_last(:c_id, :ma_app_id)
         into kbm_id;

         kh_count = 0;
         unpaid_own_amount = 0;
         for select trim(coalesce(kh.kh_repaystatus,  '-- не выбран --')),
                    trim(coalesce(kh.kh_creditstatus, '')),
                    coalesce(kh.kh_ownamount,0),
                    coalesce(kh.kh_deliqdayscount,0),
                    iif(coalesce(KH_IS_DEBTOR,0) = 1, coalesce(kh.kh_unpaidamount,0), 0),
                    coalesce(kh.kh_amount,0),
                    coalesce(kh.kh_is_our,0),
                    kh.kh_dateaward,
                    (case  kh_currency when 'Сом (Кыргызский)' then 'KGS'
                                       when 'Доллар США'  then 'USD'
                                       when 'Евро'  then 'EUR'
                                       when 'Российский рубл'  then 'RUB'
                                       else 'KGS'
                                       end ),
                   coalesce(kh_clients_kol,0)
         from kibhistory kh
         where kh.kh_from_kbm_id = :kbm_id
         and kh_creditstatus not containing('Отказано')  -- если отказано, это значит клиент кредит не получил
         and kh.kh_creditstatus not containing('Рассмотрение заявки') -- кредитный договор еще не оформлен
         and kh.kh_is_our = 0 -- только в другом ФКУ
         and kh.kh_is_debtor = 1
         order by kh.kh_dateaward desc
         rows 3
         into :kh_repaystatus,
              :kh_creditstatus,
              :own_amount,
              :kh_deliqdayscount,
              :group_unpaid_amount,
              :group_amount,
              :kh_is_our,
              :kh_dateaward,
              :CU_SYMBOL,
              :kh_clients_kol
         do
         begin
            cr_rate = null;
            if ((kh_clients_kol = 1) and  (own_amount = 0)) then own_amount = group_amount;
            select cl.cu_id from currlist cl
            where cl.cu_symbol = :CU_SYMBOL
            into :cu_id;
            select opc.cr_rate
            from opercurr opc
            where opc.cr_from_cu_id = :cu_id
            and :k_disb_date between opc.cr_date and opc.cr_date_end
            into :cr_rate;
            cr_rate = coalesce(cr_rate, 1);
            own_amount          = own_amount * cr_rate;
            group_unpaid_amount = group_unpaid_amount * cr_rate;
            group_amount        = group_amount * cr_rate;
            if (kh_repaystatus in (/*'Погашен гарантом',
                                   'Погашен за счет обеспечения в судебном порядке',
                                   'Погашен за счет обеспечения во внесудебном порядке',
                                   'Погашен за счет обеспечения на добровольной основе',
                                   'Погашен за счет обращ.взыск.на обесп.через суд.орг',
                                   'Погашен за счет обращ.вызск.на обесп. во внесуд.по',
                                   'Погашен за счет права уступки требования',
                                   'Погашен через судебные органы (кредиты без залога',
                                   'Списан через РППУ', */
                                   '120 дней просрочки и более',
                                   '60 дней до 89 дней просрочки',
                                   '60 дней просрочки и более',
                                   '90 дней  до 119 дней просрочки',
                                   '90 дней просрочки и более'
                                   ))  then
            begin
                res_error_text = 'кредит клиента из истории КИБ с датой выдачи '||cast(kh_dateaward as str_100)||' имеет негативный статус платежей('||kh_repaystatus||')';
                res_error_is_critical = 1;
                suspend;
            end else if ((kh_repaystatus = '30 дней просрочки и более') and (kh_deliqdayscount >= 60)) then
            begin
                res_error_text = 'кредит клиента из истории КИБ с датой выдачи '||cast(kh_dateaward as str_100)||' число дней просрочки более 60';
                res_error_is_critical = 1;
                suspend;                                
            end
            else if (kh_creditstatus = 'Судебное разбирательство')  then
            begin
                res_error_text = 'кредит клиента из истории КИБ '||cast(kh_dateaward as str_100)||' имеет негативный статус ('||kh_creditstatus||')';
                res_error_is_critical = 1;
                suspend;
            end
            -- рассматриваю все активные кредиты по киб
            if (KH_CREDITSTATUS in ('Выдан',
                                    'Просрочен',
                                    'Просрочен  - <не выбрано>',
                                    'Просрочен  -> Просрочки',
                                    'Просрочен -> Просрочки',
                                    'Реструктуризирован',
                                    'Своевременный платеж',
                                    'Начисление процентов')) then
            begin
                if (kh_is_our = 0) then  -- Интересуют только другие ФКУ
                begin
                    if ((own_amount > 0) and (group_unpaid_amount >0) and (group_amount >0))  then
                    begin
                        cli_percent = (cast(own_amount as double precision) * 100)/group_amount; -- доля клиента в группе
                        unpaid_own_amount = unpaid_own_amount + (group_unpaid_amount * cli_percent / 100);
                    end
                end
                if (kh_repaystatus in ('30 дней до 59 дней просрочки')) then
                begin
                    res_error_text = 'Активный кредит клиента из истории КИБ с датой выдачи '||cast(kh_dateaward as str_100)||', число дней просрочки более 30';
                    res_error_is_critical = 1;
                    suspend;          
                end
            end
         end
         -- Вычисляю баланс активных кредитов в МБФ для клиента
         for select kncl.kc_from_k_id,
                    kont.k_amount_approved,
                    kncl.kc_amount_approved
         from knclient kncl
         inner join kontstate ks on kncl.kc_from_k_id =  ks.ks_from_k_id
         inner join kontract kont on kont.k_id = kncl.kc_from_k_id
         where ks.ks_state = 1
         and ks.ks_ol_is_delete = 0
         and :k_disb_date between ks.ks_date_begin and ks.ks_date_end
         and kncl.kc_amount_approved >0
         and kncl.kc_from_c_id = :c_id
         and kncl.kc_from_k_id <> :k_id
         into :k_id_tmp,
              :group_amount,
              :own_amount
         do
         begin
            cli_percent = (cast(own_amount as double precision) * 100)/group_amount; -- доля клиента в группе
            
            group_unpaid_amount = null;
            select sum(oks.oks_balance)
            from oksummary oks
            where oks.oks_from_k_id = :k_id_tmp
            and :k_disb_date between oks.oks_date_begin and oks.oks_date_end
            and oks.oks_close_state = 0
            and oks.oks_balance > 0
            into :group_unpaid_amount;
            group_unpaid_amount = coalesce(group_unpaid_amount,0);
            unpaid_own_amount = unpaid_own_amount + (group_unpaid_amount * cli_percent / 100);
          end
          if (unpaid_own_amount > 300000) then
          begin
            res_error_text = 'Баланс активных кредитов в других ФКУ и МБФ более 300000';
            res_error_is_critical = 1;
            suspend;                
          end
         if (not (kc_amount_approved between 10000 and 55000)) then
         begin
            res_error_text = 'Сумма займа должна быть в диапазоне от 10000 до 55000 с.';
            res_error_is_critical = 1;
            suspend;
         end
         counter = 0;
         pr_type_old = null;
         is_stop = 0;
         if (pr_type <> 'PARAL') then
         begin
             for select kont.k_id,
                        kont.k_kontract_no,
                        kont.k_percent,
                        kont.k_kommis_perc,
                        iif (pr.pr_code containing ('.EC.'), 'ECONOM' , 'STANDART')
             from kontract kont
             inner join product pr on pr.pr_id = kont.k_from_pr_id
             inner join kontstate ks on ks.ks_from_k_id = kont.k_id
             inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
             where kncl.kc_from_c_id = :c_id
             and kncl.kc_amount_approved >0
             and ks.ks_state = 1
             and ks.ks_ol_is_delete = 0
             and ks.ks_date_begin < :k_disb_date
             and pr_code containing ('.BIRGE.')
             and pr_code not containing ('.PARAL.')  -- параллельный не рассмтриваю
             and pr_loan_type = 1 -- рассматриваю только индив займы
             order by ks.ks_date_begin desc
             into :k_id_old,
                  :k_kontract_no_old,
                  :k_percent_old,
                  :k_kommis_perc_old,
                  :pr_type_old
             do
             begin
                counter  = counter + 1;
                select cast(kd.kd_percent_per_day * 365  as decimal(15,2))
                from kontdata kd
                where kd.kd_ol_is_delete = 0
                and kd.kd_from_k_id =  :k_id_old
                and coalesce(kd_percent_per_day, 0) <> 0
                and kd.kd_date_end > kd.kd_ol_date
                order by kd.kd_ol_date desc
                rows 1
                into :k_percent_old;
                if ((pr_type_old = pr_type) and (is_stop = 0)) then
                begin
                    if (counter = 1) then  -- если в прошлый раз клиент кредитовался по тому же продуктк
                    begin
                         max_delinq_days = 0;
                         max_day_qty = 0;
                         -- рассчитываю макс. число дней просрочки
                         select max_day, count(max_day)
                         from
                         (
                             select (ok.ok_ol_date - ok_sh_date)  as max_day
                             from operkred ok
                             where ok.ok_ol_is_active = 1
                             and ok.ok_ol_date  <= :k_disb_date
                             and ok.ok_ol_date  > ok_sh_date
                             and ok_type < 7
                             and ok_from_k_id = :k_id_old
                             and ok_os + ok_perc < 0
                             and ok_from_c_id=0
                         ) A
                         group by max_day
                         order by  max_day desc
                         rows 1
                         into max_delinq_days, max_day_qty;
                         max_delinq_days = coalesce(max_delinq_days, 0);
                         max_day_qty = coalesce(max_day_qty, 0);
                         if (pr_type = 'STANDART') then
                         begin
                            if (max_delinq_days = 0) then
                            begin
                                if ((k_percent_old - 0.2) > min_perc) then min_perc = k_percent_old - 0.2;
                            end
                            if ((max_delinq_days between 1 and 2) and (max_day_qty = 1)) then min_perc = k_percent_old;
                            if ((max_delinq_days between 1 and 2) and (max_day_qty > 1))   then min_perc = k_percent_old + 0.5;
                            if (max_delinq_days between 3 and 7)   then min_perc = k_percent_old + 0.7;
                            if (max_delinq_days between 8 and 15)  then min_perc = k_percent_old + 1;
                            if (max_delinq_days between 16 and 30) then min_perc = max_perc;
                            if (max_delinq_days > 30) then
                            begin
                                min_perc = max_perc;
                                 res_error_text = 'У клиента было более 30 дней просрочки. Договор № '||k_kontract_no_old;
                                res_error_is_critical = 0;
                                suspend;
                            end
                            if (min_perc > max_perc) then  min_perc = max_perc;
                         end
                         if (pr_type = 'ECONOM') then
                         begin
                            if (max_delinq_days = 0) then
                            begin
                                if ((k_kommis_perc_old - 0.2) > min_dop_perc) then min_dop_perc = k_kommis_perc_old - 0.1;
                            end
                            if ((max_delinq_days between 1 and 2) and (max_day_qty = 1)) then min_dop_perc = k_kommis_perc_old;
                            if ((max_delinq_days between 1 and 2) and (max_day_qty > 1))   then min_dop_perc = k_kommis_perc_old + 0.2;
                            if (max_delinq_days between 3 and 7)   then min_dop_perc = k_kommis_perc_old + 0.3;
                            if (max_delinq_days between 8 and 15)  then min_dop_perc = k_kommis_perc_old + 0.5;
                            if (max_delinq_days between 16 and 30) then min_dop_perc = max_dop_perc;
                            if (max_delinq_days > 30) then
                            begin
                                min_perc = max_dop_perc;
                                res_error_text = 'У клиента было более 30 дней просрочки. Договор № '||k_kontract_no_old;
                                res_error_is_critical = 0;
                                suspend;
                            end
                            if (min_dop_perc > max_dop_perc) then  min_dop_perc = max_dop_perc;
                         end
                         is_stop = 1;
                    end
                    else  -- если происходит возвращение на ранее выбрнаныый продукт с другого
                    begin
                        res_error_text = 'Клиент вернулся на продукт '||pr_name||'. Годовая % ставка должна быть '||k_percent_old||' %, размер доп % должен быть = '||k_kommis_perc_old;
                        res_error_is_critical = 0;
                        suspend;     
                        is_stop = 1;
                    end
                end
             end
             if (min_perc < min_perc_finished) then  min_perc = min_perc_finished;
             if (min_dop_perc < min_dop_perc_finished) then  min_dop_perc = min_dop_perc_finished;
    
             if ((counter = 0) or (is_stop = 0))  then --  если клиент ранее не кредитовался по Бирге, или клиент еще не брал заем по текушему продукту
             begin
                 min_perc = max_perc;
                 if (min_perc_finished <> 0) then min_perc = min_perc_finished;
                 min_dop_perc = max_dop_perc;
                 if ( min_dop_perc_finished<>0) then min_dop_perc =  min_dop_perc_finished;
             end
             if (k_percent < min_perc) then
             begin
                res_error_text = 'Минимальная допустимая % ставка = '||min_perc;
                res_error_is_critical = 1;
                suspend;                   
             end
             if (k_kommis_perc < min_dop_perc) then
             begin
                res_error_text = 'Минимальный допустимы размер доп. % = '||min_dop_perc;
                res_error_is_critical = 1;
                suspend;                   
            end
        end
    end
end
