CREATE PROCEDURE CB_KG_KM_2014_02(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KC_BALANCE CURR_TYPE,
    DELINQ_BALANCE CURR_TYPE,
    PAR DECIMAL(5,5),
    REG_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    OKRUG_NAME VARCHAR(60),
    BONUS_MEMO VARCHAR(255),
    FACTOR_KOL_CLI DECIMAL(18,4),
    BONUS_END CURR_TYPE,
    KM_FIO VARCHAR(100),
    CURR_REG_ID INTEGER,
    AA_JUST_CL_END INTEGER,
    SKS_MEMO VARCHAR(100))
AS
declare variable rep_date_begin date;
declare variable cr_rate curr_type;
declare variable curr_kc_id decimal(18,0);
declare variable date_half_month date;
declare variable sks_date_begin date;
declare variable sks_reg_id_last varchar(50);
declare variable reg_name_last varchar(50);
declare variable kol_cli_field smallint;
declare variable ps_from_c_id id_type;
declare variable ps_date_begin date;
declare variable i smallint;
declare variable bonus_koef decimal(18,4);
declare variable bonus_sum decimal(18,4);
begin
  date_half_month = :rep_date_end - extract(day from :rep_date_end) + 15; -- 15е число
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end) + 1; -- Начало периода
  cr_rate = null;
  select cr_rate
  from opercurr
  where cr_from_cu_id = '840' and
        :rep_date_end between cr_date and cr_date_end
  into cr_rate;
  if (cr_rate is null) then
    cr_rate = 1;

-- Во временную таблицу загружает итоговые данные по КС
  insert into temp_kc_reg_info (kc_region_id, kc_kol_cli, kc_balance, kc_delinq_8_180)
  select aa_curr_region_id, sum(aa_just_cl_end), sum(aa_amount_active_end),
         sum(aa_deliq_balance7 - aa_deliq_balance180)
  from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 1, 1, 0, 0, 0, null)
  where not exists(select *
                   from sb_akt, sb_list -- Пункт 2.3
                   where sba_is_delete = 0 and
                         sba_from_sbl_id <> 0 and
                         sba_status = 2 and
                         :rep_date_end between sba_date_begin and sba_date_end and
                         sba_from_k_id = aa_k_id and
                         sba_from_sbl_id = sbl_id and
                         sbl_type = 1)
  group by aa_curr_region_id;

-- Вытаскивает всех КСников которые за отчетный период прозводили какое либо операцию или на конец есть активный портфель
  for select kc_region_id, kc_kol_cli, kc_balance, kc_delinq_8_180
      from temp_kc_reg_info
      order by kc_region_id
      into curr_reg_id, aa_just_cl_end, kc_balance, delinq_balance
  do
  begin
--  Расчет ПАР офиса
    par = 0;
    if (kc_balance <> 0) then
      par = round(delinq_balance / kc_balance * 100, 5);

-- Вычисления бонус фактора кол-во клиентов и пар, пункт 3.1
    kol_cli_field = case
                      when :aa_just_cl_end > 2000 then 1
                      when :aa_just_cl_end > 1500 then 2
                      when :aa_just_cl_end > 1000 then 3
                      when :aa_just_cl_end > 500 then 4
                      else 5
                    end;
    factor_kol_cli = case kol_cli_field
                       when 1 then case
                                     when par = 0 then 30000
                                     when par <= 1.0 then 24000
                                     when par <= 1.5 then 18000
                                     else 0
                                   end
                       when 2 then case
                                     when par = 0 then 22000
                                     when par <= 1.0 then 17600
                                     when par <= 1.5 then 13200
                                     else 0
                                   end
                       when 3 then case
                                     when par = 0 then 15000
                                     when par <= 1.0 then 12000
                                     when par <= 1.5 then 9000
                                     else 0
                                   end
                       when 4 then case
                                     when par = 0 then 12000
                                     when par <= 1.0 then 9600
                                     when par <= 1.5 then 7200
                                     else 0
                                   end
                       else 0
                     end;
    if (:rep_date_end >= '01.05.2014' and
        par > 1.0) then
      factor_kol_cli = 0;
--  Кол-во кредитных менеджеров
    i = 0;
    ps_from_c_id = 0;
    ps_date_begin = null;
    curr_kc_id = 0;
    select ps_from_c_id, ps_date_begin
    from hr_pstatus, joblist
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :curr_reg_id and
          ps_status = 1 and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = 6 -- Выборка по кредитным менеджерам
          and
          coalesce((select sawt_id
                    from hr_voctravel, hr_orders, sa_worktype
                    where hrv_from_ho_id = ho_id and
                          ho_order_type = sawt_id and
                          hrv_from_c_id = ps_from_c_id and
                          :rep_date_end between hrv_date_begin and hrv_date_end_fact
                    order by hrv_date_begin desc
                    rows 1), 0) not in (16, 17, 18, 8) -- Декретников не включает
          rows 1
    into curr_kc_id, sks_date_begin;
    if (curr_kc_id is null) then
      curr_kc_id = 0;

--  вытаскивает данные по выговорам и бонус коэффициент
    select bonus_koef, bonus_sum, bonus_memo
    from hp_get_vigovor_koef(:curr_kc_id, :rep_date_end)
    into bonus_koef, bonus_sum, bonus_memo;

    if (curr_kc_id <> 0) then
      bonus_end = factor_kol_cli * bonus_koef / 100 - bonus_sum;
    else
      bonus_end = 0;
    if (bonus_end < 0) then
      bonus_end = 0;

--  Если по системной должности СКС то дополнительный расчет бонуса для СКС
    sks_memo = null;
    if (sks_date_begin > date_half_month) then
    begin
      sks_reg_id_last = null;
      select ps_from_reg_id
      from hr_pstatus, joblist, jobkind
      where :date_half_month between ps_date_begin and ps_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.Region_KM' -- Выбор
            and
            ps_from_c_id = :curr_kc_id
      into sks_reg_id_last;
      if (sks_reg_id_last is not null) then
      begin
        reg_name_last = (select reg_code || ' ' || reg_name
                         from region
                         where reg_unicode = :sks_reg_id_last);
        sks_memo = 'КМ перевелся из офиса ' || reg_name_last || ', от ' || sks_date_begin;
      end
    end

    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
    from region r1, region r2, region r3
    where r1.reg_unicode = :curr_reg_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into reg_name, branch_name, okrug_name;

    km_fio = null;
    select c_name_fio
    from clients
    where c_id = :curr_kc_id
    into km_fio;

    suspend;
  end

end
