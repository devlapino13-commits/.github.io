CREATE PROCEDURE HP_VERIFY_AMOUNT_GOODS(
    K_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable k_no d_str_20;
declare variable k_amount_approved curr_type;
declare variable k_prepayment_summa curr_type;
declare variable kml_total curr_type;
begin
  res_text = '';
  select
    k_kontract_no,
    k_amount_approved,
    coalesce(k_prepayment_summa,0)
  from kontract
    inner join product
      on pr_id = k_from_pr_id
        and pr_code containing '.ipf.'
        and pr_code not containing '.KARDHASAN.' and
            pr_code not containing '.ISTISNAA.'
  where k_id = :k_id
  into
    :k_no,
    :k_amount_approved,
    :k_prepayment_summa;

  if (row_count = 1) then begin
    kml_total = coalesce((
      select
        sum(kml_total)
      from kontmat
      where kml_object_id = :k_id and kml_object_kind = 9),0);

    if ((k_amount_approved + k_prepayment_summa) <> kml_total)
      then res_text =
        'Итоговая сумма покупки товара ' || cast(kml_total as decimal(12,2)) ||
        ' - не соответствует сумме договора ' || cast(k_amount_approved as decimal(12,2)) ||
        iif(k_prepayment_summa > 0, ' и частичной оплаты стоимости товара ' ||
          cast(k_prepayment_summa as decimal(12,2)), '');
  end
  suspend;
end
