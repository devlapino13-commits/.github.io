CREATE PROCEDURE CARD_PAYMENT_CARD_LIMIT_UPDATE(
    OBJECT_KIND D_STR_100,
    OBJECT_ID D_ID,
    OL_ID D_ID)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable oks_balance d_currency;
declare variable new_limit d_currency;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable cca_code d_str_code;
declare variable cs_request_json d_str_8100;
declare variable ok_payed_os d_currency;
declare variable object_kind_str d_str_100;
declare variable cs_id d_integer;
declare variable ol_date d_date;
declare variable ol_parent_id d_id;
declare variable ol_from_reg_id d_integer;
declare variable un_id d_id;
declare variable is_writeoff d_bool;
begin
  /*
    Процедура отвечает за формирование операции для МПЦ
    "установка кредитного лимита по карт-счету",
    по факту погашения займа
  */
  if (f_get_const('cardprjclose') = 1) then
    exit;
  object_kind_str = f_objkind(object_kind, 'name');

  if (object_kind_str = 'kontract') then
    k_id = object_id;
  else
    exception e_message 'object_kind = '||object_kind_str||' не реализован';

  if (ol_id is null) then
    exception e_message 'ol_id не определен';

  select
    ol_parent_id,
    ol_from_reg_id,
    ol_user_id
  from operlist
  where ol_id = :ol_id
  into
    :ol_parent_id,
    :ol_from_reg_id,
    :un_id;

  if (ol_parent_id is not null) then
    select
      ol_from_reg_id
    from operlist
    where ol_id = :ol_parent_id
    into
      :ol_from_reg_id;

  -- списание(1) или погашение(0)
  is_writeoff = iif(exists(
    select *
    from operlist
    where ol_id = :ol_id
      and ol_oper_type = 1
      and ol_oper_subtype = 3), 1, 0);

  ok_payed_os = 0;
  select
    -sum(ok_os),
    ok_ol_date
  from operkred
  where ok_type = iif(:is_writeoff = 1, 5, 4)
    and ok_from_ol_id = :ol_id
    and ok_from_k_id = :k_id
    and ok_ol_is_active = 1
  group by ok_ol_date
  into
    :ok_payed_os,
    :ol_date;

  if (ok_payed_os < 0) then
    exception e_message 'Погашение увелиличило долг по ОС';

  if (ok_payed_os = 0) then
  begin
    -- если сумма платежа не затронула ОС, тогда запускаю операцию списания с карты
    execute procedure card_payment_writeoff(:object_kind, :object_id, null, :ol_id);
    exit;
  end

  if (ok_payed_os > 0) then
  begin
    -- если платеж затронул ОС, тогда формирется операция по изменению кредитного лимита
    select
      kncl.kc_from_c_id
    from knclient kncl
    where kncl.kc_from_k_id = :k_id
    into
      :c_id;

    select
      card_id,
      cca_id,
      cca_code
    from hp_client_bank_card_mb(:c_id, current_date, 0)
    into
      :card_id,
      :cca_id,
      :cca_code;

    oks_balance = (select amount from hp_balance_cardloans('kontract', :k_id, :ol_date));
    new_limit = (select amount from hp_balance_cardloans('cardacc', :cca_id));
    cs_request_json = f_format('{"bank_c":"21", "groupc":"01", "limit":%1, "card_acct":"%2"}',
      new_limit, cca_code);

    insert into cardsync(
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_amount,
      cs_request_json,
      cs_office_id,
      cs_from_un_id,
      cs_from_k_id,
      cs_from_c_id
    )
    values(
      f_get_cardsynccfg('payment_credit_limit_change', 'id'),
      :card_id,
      :cca_id,
      f_objkind('kontract', 'id'),
      :k_id,
      :new_limit,
      :cs_request_json,
      :ol_from_reg_id,
      :un_id,
      :k_id,
      :c_id
    )
    returning cs_id
    into :cs_id;

    insert into cardsyncopers (cso_from_cs_id, cso_from_ol_id)
    values (:cs_id, :ol_id);

    -- устанавливаю лимит по карт.счету = сумме баланса займа
    insert into cardclientacc_limit (
      ccl_from_cca_id, ccl_card_id, ccl_from_c_id, ccl_from_cs_id, ccl_from_ol_id,
      ccl_amount_limit_down, ccl_from_k_id, ccl_new_limit)
    values (
      :cca_id, :card_id, :c_id, :cs_id, :ol_id,
      :ok_payed_os, :k_id, :oks_balance);
  end
end
