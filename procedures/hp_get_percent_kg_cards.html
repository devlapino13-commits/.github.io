CREATE PROCEDURE HP_GET_PERCENT_KG_CARDS(
    PERC_IN PERC_TYPE,
    PERC_IN_MIN PERC_TYPE,
    PERC_IN_MAX PERC_TYPE,
    PR_CODE D_STR_100,
    KC_CIRCLE D_SMALLINT,
    LOANDISBDATE D_DATE,
    LOANPERIOD D_SMALLINT)
RETURNS (
    PERC_OUT PERC_TYPE,
    PERC_OUT_MIN PERC_TYPE,
    PERC_OUT_MAX PERC_TYPE)
AS
begin
  perc_out = perc_in;
  perc_out_min = perc_in_min;
  perc_out_max = perc_in_max;
  /*Процентные ставки для продуктов по выдачам на карту:
    КАРТА СТАНДАРТ
    КАРТА СТАНДАРТ-ОНЛАЙН
    КАРТА ИШЕНИМ
    КАРТА ИШЕНИМ-ОНЛАЙН    */

  if (pr_code containing '.consultation.') then
  begin
    --КАРТА ИШЕНИМ
    if (pr_code not containing '.online.') then
      perc_out = case
                   when kc_circle >= 6 then 24.0
                   when kc_circle between 4 and 5 then 27.0
                   when kc_circle between 2 and 3 then 30.0
                   else 33
                 end;
    --КАРТА ИШЕНИМ-ОНЛАЙН
    if (pr_code containing '.online.') then
      perc_out = case
                   when kc_circle >= 6 then 22.0
                   when kc_circle between 4 and 5 then 25.0
                   when kc_circle between 2 and 3 then 28.0
                   else 31
                 end;
  end
  else
  begin
    --КАРТА СТАНДАРТ
    if (pr_code not containing '.online.') then
      perc_out = case
                   when kc_circle >= 6 then 46.0
                   when kc_circle between 4 and 5 then 46.0
                   when kc_circle between 2 and 3 then 46.0
                   else 46.0
                 end;
    else --КАРТА СТАНДАРТ-ОНЛАЙН
      perc_out = case
                   when kc_circle >= 6 then 44.0
                   when kc_circle between 4 and 5 then 44.0
                   when kc_circle between 2 and 3 then 44.0
                   else 44.0
                 end;
  end

  if (perc_in <> perc_out) then
  begin
    perc_out = round(perc_out, 2);
    perc_out_min = perc_out;
    perc_out_max = perc_out;
  end

  suspend;
end
