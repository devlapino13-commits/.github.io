CREATE PROCEDURE HP_MENULIST_UX_INSERT(
    ML_PARENT_ID D_INTEGER,
    ML_INDEX D_INTEGER,
    ML_CAPTION D_STR_255,
    UA_CODE D_STR_100)
RETURNS (
    ML_ID D_INTEGER)
AS
declare variable ml_kind d_smallint;
begin
  ml_kind = 1; -- Ux папка

  if (trim(coalesce(ua_code, '')) <> '') then
  begin
    ml_kind = 2; -- UX элемент меню

    select ml_id
    from menulist
    where ml_ua_code = :ua_code
    rows 1
    into ml_id;

    if (ml_id is not null) then
    begin
      suspend;
      exit;
    end

    select ua_name
    from uxaction
    where ua_code = :ua_code
    into ml_caption;
  end else
    ua_code = Null;

  if (ua_code is not null and ml_index = -1) then
  begin
    -- Связываем с существующим пунктом
    ml_id = ml_parent_id;

    update menulist
    set ml_ua_code = :ua_code,
        ml_caption = :ml_caption
    where ml_id = :ml_parent_id;
  end else
  begin
    insert into menulist (ml_id, ml_parent_id, ml_caption, ml_index, ml_is_report, ml_kind, ml_ua_code)
    values (0, :ml_parent_id, :ml_caption, :ml_index, 0, :ml_kind, :ua_code)
    returning ml_id
    into ml_id;
  end

  suspend;
end
