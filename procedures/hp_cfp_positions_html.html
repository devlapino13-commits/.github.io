CREATE PROCEDURE HP_CFP_POSITIONS_HTML(
    CP_ID D_INTEGER)
RETURNS (
    HTML_TXT D_BLOB_TEXT)
AS
declare variable parent_id d_integer;
declare variable name_ru d_str_100;
declare variable name_kg d_str_100;
declare variable tp d_integer;
begin
  html_txt = '';
  for
    select
      cp_parenet_id,
      cp_name_ru,
      cp_name_kg,
      cp_type
    from cfp_positions
    where cp_id = :cp_id
  into
    :parent_id,
    :name_ru,
    :name_kg,
    :tp
  do
  begin
    html_txt = blob_append(html_txt, f_format('<b>%1</b><br>', f_enum_name('cfp_positions', tp)));
    html_txt = blob_append(html_txt, 'Наименование на русском:<br>');
    html_txt = blob_append(html_txt, f_format('<font color="blue">%1</font><br><br>', name_ru));
    html_txt = blob_append(html_txt, 'Наименование на кыргызском:<br>');
    html_txt = blob_append(html_txt, f_format('<font color="blue">%1</font>', name_kg));
  end
  suspend;
end
