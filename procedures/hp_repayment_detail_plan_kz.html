CREATE PROCEDURE HP_REPAYMENT_DETAIL_PLAN_KZ(
    K_ID D_ID,
    REP_DATE D_DATE)
RETURNS (
    SH_OS_BASE DECIMAL(18,2),
    KOL_DAY INTEGER,
    K_PERCENT_PER_DAY D_CURRENCY,
    CALC_PERC DECIMAL(18,0))
AS
declare variable k_amount_approved decimal(18,2);
declare variable k_disb_date date;
declare variable sh_date date;
declare variable sh_os_accum decimal(18,2);
declare variable prev_date date;
begin
  select k_amount_approved, k_percent_per_day, k_disb_date
  from kontract
  where k_id = :k_id
  into :k_amount_approved, :k_percent_per_day, :k_disb_date;

  for select sh_date, sh_calc_perc
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      where sh_date is not null and
            sh_calc_perc <> 0
      order by sh_date
      into :sh_date, :calc_perc
  do
  begin
    select max(sh.sh_date), sum(sh.sh_os)
    from schedule sh
    where sh.sh_from_k_id = :k_id and
          sh.sh_id > 0 and
          sh.sh_date < :sh_date
    into :prev_date, :sh_os_accum;

    if (prev_date is null) then
      prev_date = k_disb_date;

    sh_os_base = k_amount_approved - coalesce(:sh_os_accum, 0);
    kol_day = minvalue(:rep_date, :sh_date) - :prev_date;

    suspend;
  end
end
