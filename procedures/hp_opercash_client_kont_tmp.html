CREATE PROCEDURE HP_OPERCASH_CLIENT_KONT_TMP(
    C_ID D_ID,
    LKP_OFFICE_ID D_INTEGER,
    ADD_INFO D_STR_CODE)
RETURNS (
    RES_K_ID D_ID,
    RES_KONTRACT_NAME D_STR_255,
    RES_SORT_DATE D_DATE)
AS
declare variable ks_state d_smallint;
declare variable need_suspend d_bool;
declare variable tmp_amount d_currency;
declare variable kf_id d_id;
declare variable kf_name d_str_code;
declare variable k_status d_smallint;
declare variable edoc_kind d_integer;
declare variable k_disb_kind d_integer;
declare variable k_cons_kind d_smallint;
declare variable pr_code d_str_100;
declare variable res_text d_str_255;
begin
  add_info = coalesce(add_info, '');
  if (add_info = '') then
    Exit;

  if (lkp_office_id is null) then
    exit;

  add_info = f_enum_code('cashoperkind', add_info);

  for select kc_from_k_id, k_from_kf_id, k_status, k_disb_date,
             k_kontract_no || ' от ' || f_datetostr(k_disb_date) ||
             ', ' || pr_name, k_disb_kind, k_cons_kind, pr_code
  from knclient, kontract, product
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = k_id and
        k_from_pr_id = pr_id
  into res_k_id, kf_id, k_status, res_sort_date, res_kontract_name, k_disb_kind, k_cons_kind, pr_code
  do
  begin
    ks_state = null;
    select ks_state
    from kontstate
    where ks_from_k_id = :res_k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999'
    into ks_state;

    need_suspend = 1;
  
    if (add_info = 'minus.loan.disb' and pr_code containing '.insur_kontrakt.') then  -- RFCKG-4831
      need_suspend = 0;

    if (add_info = 'minus.loan.disb' and pr_code containing '.ipf.') then  -- RFCKG-5347
      need_suspend = 0;

    if (add_info in ('plus.loan.repay', 'plus.loan.cl_repay')) then
    begin
      if (ks_state <> 1) then
        need_suspend = 0;
      else
      begin
        select ok_sh_date
        from operkred
        where ok_from_k_id = :res_k_id and
              ok_ol_is_active = 1 and
              ok_type < 7
        group by 1
        having sum(ok_sh_os + ok_sh_perc + ok_os + ok_perc) > 0
        order by ok_sh_date -- RFCKG-5095
        rows 1
        into res_sort_date;
      end
    end else
    if (add_info = 'plus.loan.card_kommis') then
    begin
      /*
      if (not exists(select * from cardlistlnk
                     where cll_from_k_id = :res_k_id and
                           cll_date_end >= current_date)) then
        need_suspend = 0;
      */

      if (not exists(select *
                     from cardaktclient
                     where cac_from_k_id = :res_k_id and
                           cac_from_c_id = :c_id and
                           cac_status  = f_enum_id('cardaktclient', 'issued_delayed'))) then
        need_suspend = 0;
    end else
    if (add_info = 'minus.loan.overpay') then
    begin
      tmp_amount = 0;
      select sum(ok_os)
      from operkred
      where ok_from_k_id = :res_k_id and
            ok_ol_is_active = 1 and
            ok_type = 9
      into tmp_amount;

      if (tmp_amount is null) then
        tmp_amount = 0;

      if (tmp_amount = 0) then
        need_suspend = 0;

      if (need_suspend = 0) then
      begin
        select -sum(ok_perc + ok_tax)
        from operkred
        where ok_from_k_id = :res_k_id and
              ok_ol_is_active = 1 and
              ok_type = 9 and
              ok_ol_date = '31.12.2025'
        into tmp_amount; -- RFCKG-5143
  
        if (tmp_amount is null) then
          tmp_amount = 0;
  
        if (tmp_amount > 0) then
          need_suspend = 1;
      end
    end else
    if (add_info = 'plus.loan.offpay') then
    begin
      tmp_amount = 0;
      select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
      from operkroff
      where oko_from_k_id = :res_k_id and
            oko_from_c_id = :c_id and
            oko_ol_is_active = 1
      into tmp_amount;

      if (tmp_amount is null) then
        tmp_amount = 0;

      if (tmp_amount <= 0) then
        need_suspend = 0;
    end else
    if (add_info = 'plus.loan.sud') then
    begin
      tmp_amount = 0;

      select res_amount_saldo
      from aarep_operextra_saldo(9, :res_k_id, f_get_rd_id('OperExtraKind.LawClientExpense'), 'LoanRepay', current_date, 0)
      where res_amount_saldo > 0
      into tmp_amount;

      if (tmp_amount is null) then
        tmp_amount = 0;

      if (tmp_amount <= 0) then
        need_suspend = 0;
    end else
    if (add_info in ('minus.loan.konsult', 'minus.loan.disb', 'plus.loan.konsult', 'plus.loan.partial', 'plus.loan.insurance')) then
    begin
      -- Утвержденные договора в процессе, не выданные
      if (ks_state <> 0 or k_status <> 1) then
        need_suspend = 0;

      if (need_suspend = 1) then
      begin
        if (not exists(select * from loantodisb
                       where lnd_from_k_id = :res_k_id and
                             lnd_disb_office_id = :lkp_office_id and
                             lnd_is_ready = 1)) then
          need_suspend = 0;
      end

      if (need_suspend = 1) then
      begin
        if (add_info = 'minus.loan.disb') then
        begin
          if (k_disb_kind <> 1) then
            need_suspend = 0;
          else
          begin
            if (edoc_kind is null) then
              edoc_kind = f_edoc_kind('loan.agreement', 'id');
    
            -- Если нет подписанного эл.документа, не показываем займ
            if (not exists(select * from edoc_list
            where edl_object_kind = 9 and -- Kontract
                  edl_object_id = :res_k_id and
                  edl_from_edc_id = :edoc_kind and
                  edl_status = 1)) then
              need_suspend = 0;

            -- RFCKG-3947
            if (need_suspend = 1
                and exists
                (
                    select 1
                    from hp_opercash_loan_other_amount(:res_k_id, 'CashTransactionIncom.Consultation')
                    where res_amount > 0
                )
            ) then
                need_suspend = 0;

            -- RFCKG-3661
            if (need_suspend = 1
                and exists
                (
                    select 1
                    from hp_opercash_loan_other_amount(:res_k_id, 'CashTransactionIncom.Insurance')
                    where res_amount > 0
                )
            ) then
                need_suspend = 0;
          end
        end
      end
    end

    if (add_info = 'plus.loan.partial' and need_suspend = 1) then
    begin
      select res_amount
      from hp_opercash_loan_other_amount(:res_k_id, 'CashTransactionIncom.PartialPrice')
      into :tmp_amount;

      if (tmp_amount <= 0) then
        need_suspend = 0;
    end

    if (add_info = 'plus.loan.konsult' and k_cons_kind > 0) then
      need_suspend = iif(ks_state = 1, 1, 0); -- оплата конс. услуг в рассрочку принимается только если займ активный (RFCKG-4900)

    if (add_info = 'plus.loan.konsult' and need_suspend = 1) then
    begin
      select res_amount
      from hp_opercash_loan_other_amount(:res_k_id, 'CashTransactionIncom.Consultation')
      into :tmp_amount;

      if (tmp_amount <= 0) then
        need_suspend = 0;
    end

    if (add_info = 'plus.loan.insurance' and need_suspend = 0) then
    begin
      select res_text
      from hp_ipf_check_insurance_pay(:res_k_id)
      into res_text;

      if (res_text <> '') then
        need_suspend = 1; -- RFCKG-5390
    end

    if (add_info = 'plus.loan.insurance' and need_suspend = 1) then
    begin
      select res_amount
      from hp_opercash_loan_other_amount(:res_k_id, 'CashTransactionIncom.Insurance')
      into :tmp_amount;

      if (tmp_amount <= 0) then
        need_suspend = 0;
    end

    if (need_suspend = 1) then
    begin
      kf_name = '';
      select kal_new_kf_name
      from knaktlist
      where kal_from_kf_id = :kf_id and
            kal_date_end >=  '31.12.2999' and
            is_delete = 0
      into kf_name;
  
      if (kf_name <> '') then
        res_kontract_name = res_kontract_name || '  (' || kf_name || ')';
  
      suspend;
    end
  end
end
