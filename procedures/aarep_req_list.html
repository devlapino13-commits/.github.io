CREATE PROCEDURE AAREP_REQ_LIST(
    REP_REG_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REQ_NO D_INTEGER,
    APP_DATE D_DATE,
    OPER_DATE D_DATE,
    GEN_OFFICE D_STR_100,
    DEST_OFFICE D_STR_100,
    DEST_OFFICE_CODE D_STR_100,
    REQ_AUTHOR D_STR_100,
    ACC_NAME D_STR_255,
    REQ_STATUS VARCHAR(50),
    REQ_SUM D_INTEGER)
AS
declare variable RM_ID D_ID;
declare variable RM_REG_ID D_ID;
declare variable RL_REG_ID D_ID;
begin
  for select arl.rm_date, -- дата заявки
             arl.rm_region_name, -- основной офис
             arl.rm_c_name_fio, -- автор запроса
             arl.rm_no, -- номер запроса
             arl.rm_status_text, -- статус
             arl.rm_id,
             arl.rm_from_reg_id
      from aarep_request_list(:rep_reg_id, :rep_date_begin, :rep_date_end, 0, '', '', -2, -1) arl
      where arl.rm_from_rk_id = 0 -- тип запроса: подкрепление
      into :app_date,
           :gen_office,
           :req_author,
           :req_no,
           :req_status,
           :rm_id,
           :rm_reg_id
  do
  begin
    if (app_date <= rep_date_begin) then
      continue;

    for select reg.reg_name, -- офис назначения
               trim(reg.reg_code) || ' ' || trim(reg.reg_name),
               rl.rl_date, -- дата операции
               rl.rl_text, -- наименование счета
               rl.rl_summa_minus, -- сумма
               rl.rl_corr_office
        from reqlist rl
        inner join region reg on rl.rl_corr_office = reg.reg_unicode
        where rl.rl_from_rm_id = :rm_id
        into :dest_office,
             :dest_office_code,
             :oper_date,
             :acc_name,
             :req_sum,
             :rl_reg_id
    do
    begin
      if (rm_reg_id <> rl_reg_id) then
      suspend;
    end
  end
end
