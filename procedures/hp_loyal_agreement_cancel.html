CREATE PROCEDURE HP_LOYAL_AGREEMENT_CANCEL(
    AG_ID D_INTEGER,
    CANCEL_DATE D_DATE,
    UN_ID D_ID,
    AGT_FROM_RD_ID D_ID,
    AGT_MEMO D_STR_255)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_SMALLINT)
AS
declare variable reg_id d_integer;
declare variable db_id d_integer;
declare variable ol_id d_id;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable rd_id d_integer;
declare variable ogr_balance d_currency;
declare variable edc_id d_integer;
begin
  -- Процедура расторжения участия в программе лояльности
  -- Списывает накопленные баллы если имеются

  res_text = '';
  res_is_error = 0;

  rd_id = f_get_rd_id('AgreementKinds.loyal_agreement');
  if (rd_id is null) then
    exit;

  if (not exists(select *
                 from agreements
                 where ag_id = :ag_id and
                       ag_from_rd_id = :rd_id and
                       ag_status <> -1 and
                       :cancel_date between ag_date_begin and coalesce(ag_date_end, :cancel_date))) then
  begin
    res_is_error = 1;
    res_text = 'Нет действующей программы на указанную дату';
    suspend;
    exit;
  end

  db_id = f_get_db_id();

  reg_id = f_hq_id(db_id);

  ogr_balance = 0;

  select ag_object_kind, ag_object_id
  from agreements
  where ag_id = :ag_id
  into obj_kind, obj_id;

  select edc_id
  from edoc_cfg
  where edc_code = 'loyal.agreement'
  into edc_id;

  update edoc_list
  set edl_date_end = :cancel_date
  where edl_from_edc_id = :edc_id and
        edl_object_kind = :obj_kind and
        edl_object_id = :obj_id and
        edl_status <> -1 and
        edl_date_end is null;

  select sum(ogr_amount * ogr_sign + ogr_tax_fl * ogr_sign)
  from operagr
  where ogr_from_ag_id = :ag_id and
        ogr_ol_is_active = 1
  into ogr_balance;
  if (ogr_balance is null) then
    ogr_balance = 0;

 -- if (ogr_balance > 0) then
  begin

    update agreements
    set ag_date_end = :cancel_date
    where ag_id = :ag_id;

    insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_document_client, ol_memo, ol_user_id,
                          ol_oper_subtype)
    values (:reg_id, 2, :cancel_date, 9, :obj_id,
            f_format('Расторжение программы лояльности № %1', :ag_id), :un_id, 38)
    returning ol_id
    into :ol_id;

    insert into operagr (ogr_from_ol_id, ogr_from_ag_id, ogr_from_rd_id, ogr_sign, ogr_amount, ogr_tax_fl, ogr_ol_date)
    values (:ol_id, :ag_id, :rd_id, -1, :ogr_balance, 0, :cancel_date);

    insert into agreements_termination (agt_from_ag_id, agt_from_rd_id, agt_memo)
    values (:ag_id, :agt_from_rd_id, :agt_memo);
  end

 -- if (ogr_balance > 0) then
    res_text = f_format('Программа лояльности расторгнут, баллы списаны: %1', ogr_balance);
 -- else
  --  res_text = 'Программа лояльности расторгнут';

  suspend;
end
