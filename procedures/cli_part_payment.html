CREATE PROCEDURE CLI_PART_PAYMENT(
    OL_ID D_ID)
RETURNS (
    KC_ID D_ID,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    KC_AMOUNT_APPROVED D_CURRENCY,
    GROUP_LOAN_AMOUNT D_CURRENCY,
    KOEF D_PERCENT,
    GOS_PAYED_PERC_GROUP D_CURRENCY,
    GOS_PAYED_PERC D_CURRENCY)
AS
declare variable i d_integer;
declare variable n d_integer;
declare variable s d_currency;
begin
    gos_payed_perc_group = 0;

    select sum(ob_summa)
    from operbook
    where ob_from_ol_id = :ol_id
    and substring(ob_account_dt from 8 for 5) = '11531'
    into :gos_payed_perc_group;

    if (gos_payed_perc_group <> 0) then
    begin
        select count(*), sum(kncl.kc_amount_approved)
        from oper_compens oc
        inner join knclient kncl on kncl.kc_id = oc.oc_from_kc_id
        where oc.oc_from_ol_id = :ol_id
        into :n, :group_loan_amount;

        i = 0;
        s = 0;
        for select kc.kc_id, cl.c_id, cl.c_name_fio, kc.kc_amount_approved, /*oc.oc_group_loan_amount,*/ kc_amount_approved/:group_loan_amount
        from oper_compens oc
        inner join knclient kc on kc.kc_id = oc.oc_from_kc_id
        inner join operlist ol on ol.ol_id = oc.oc_from_ol_id
        inner join clients cl on cl.c_id = kc.kc_from_c_id
        where oc.oc_from_ol_id = :ol_id
        and ol.is_delete = 0
        into :kc_id, :c_id, :c_name_fio, :kc_amount_approved, /*:group_loan_amount,*/ :koef
        do
        begin
            i = i + 1;
            if (koef = 1) then
            begin
                gos_payed_perc = gos_payed_perc_group;
            end
            else
            begin
                if (i<n) then gos_payed_perc = round(gos_payed_perc_group*:koef, 0);
                else gos_payed_perc = gos_payed_perc_group - s;
                s = s + gos_payed_perc;
            end

            suspend;
        end
    end
end
