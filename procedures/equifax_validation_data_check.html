CREATE PROCEDURE EQUIFAX_VALIDATION_DATA_CHECK(
    K_ID ID_TYPE)
RETURNS (
    KF_CLIENT_FIO VARCHAR(100),
    KF_CLIENT_PASS_NO VARCHAR(30),
    KF_DISB_DATE DATE,
    KC_AMOUNT DECIMAL(18,4),
    KF_ERROR_TEXT VARCHAR(255),
    KF_ERROR_IS_CRITICAL INTEGER)
AS
declare variable C_ID D_ID;
declare variable ERR_TXT D_STR_255;
declare variable ADR_TYPE D_STR_100;
declare variable ADL_POST_INDEX D_STR_100;
declare variable C_PASS_WHO STR_255;
declare variable ERROR_TXT STR_255;
declare variable C_INN varchar(20);
declare variable RESULT integer;
declare variable C_PASS_WHO_CODE STR_255;
declare variable C_BIRTH_CITY STR_255;
declare variable K_DISB_DATE date;
declare variable DOC_CODE STR_100;
declare variable DOC_NO STR_100;
declare variable ADL_ADDRESS_STR D_STR_255;
declare variable ADC_ADDRESS_TYPE D_SMALLINT;
begin
  kf_error_is_critical = 2;

  for select kncl.kc_from_c_id, cl.c_name_fio, coalesce(cl.c_pass_who, ''), iif(cl.c_citizen = 13, cl.c_inn, ''),
             coalesce(cl.c_pass_who_code, ''), coalesce(cl.c_birth_city, ''), kont.k_disb_date
      from knclient kncl
      inner join clients cl on kncl.kc_from_c_id = cl.c_id
      inner join kontract kont on kont.k_id = kncl.kc_from_k_id
      where kncl.kc_from_k_id = :k_id and
            kncl.kc_amount_approved > 0
      into :c_id, :kf_client_fio, :c_pass_who, :c_inn, :c_pass_who_code, :c_birth_city, :k_disb_date
  do
  begin
    if (c_pass_who_code <> '') then
    begin
      if (c_pass_who_code not similar to('[-]?[a-zа-я0-9-.\ ]+')) then
      begin
        kf_error_is_critical = 2;
        kf_error_text = 'Код подразделения для документа удостоверяющего личность указан не корретно. Сейчас:' || c_pass_who_code;
        suspend;
      end
    end

    if (c_birth_city = '') then
    begin
      kf_error_is_critical = 2;
      kf_error_text = 'Место рождения клиента не указано';
      suspend;
    end
    ----------- >
    c_pass_who = trim(c_pass_who);
    c_pass_who = replace(c_pass_who, '`', '');
    c_pass_who = replace(c_pass_who, '*', '');
    c_pass_who = replace(c_pass_who, '''', '');
    c_pass_who = replace(c_pass_who, 'À', 'A');
    c_pass_who = replace(c_pass_who, 'İ', 'I');
    c_pass_who = replace(c_pass_who, 'ı', 'i');
    c_pass_who = replace(c_pass_who, '’', '');
    c_pass_who = replace(c_pass_who, 'ʻ', '');
    c_pass_who = replace(c_pass_who, '|', '');
    c_pass_who = replace(c_pass_who, ascii_char(13), '');
    c_pass_who = replace(c_pass_who, ascii_char(10), '');
    c_pass_who = replace(c_pass_who, '‘', '');
    c_pass_who = replace(c_pass_who, '‘', '');
    c_pass_who = replace(c_pass_who, 'Ĝ', '');
    c_pass_who = replace(c_pass_who, 'Õ', '');

    error_txt = null;
    select error_txt
    from hp_check_characters_str(:c_pass_who, '[ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyzа-я0-9@-@+@(@)@_@?#\/.,№!:;"'' ]+')
    into :error_txt;

    if (error_txt <> '') then
    begin
      kf_error_text = 'Место выдачи документа указано не корректно. ' || error_txt;
      suspend;
    end
    c_birth_city = trim(c_birth_city);
    c_birth_city = replace(c_birth_city, '`', '');
    c_birth_city = replace(c_birth_city, '*', '');
    c_birth_city = replace(c_birth_city, '''', '');
    c_birth_city = replace(c_birth_city, 'À', '');
    c_birth_city = replace(c_birth_city, 'İ', '');
    c_birth_city = replace(c_birth_city, 'ı', '');
    c_birth_city = replace(c_birth_city, '’', '');
    c_birth_city = replace(c_birth_city, 'ʻ', '');
    c_birth_city = replace(c_birth_city, ascii_char(13), '');
    c_birth_city = replace(c_birth_city, ascii_char(10), '');
    c_birth_city = replace(c_birth_city, '‘', '');
    c_birth_city = replace(c_birth_city, '‘', '');

    error_txt = null;
    select error_txt
    from hp_check_characters_str(:c_birth_city, '[ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyzа-я0-9@-@+@(@)@_@?#\/.,№!:;"'' ]+')
    into :error_txt;

    if (error_txt <> '') then
    begin
      kf_error_text = 'Место рождения указано не корректно. ' || error_txt;
      suspend;
    end

    for select adl.adl_address_str, adc.adc_address_type
        from addrclient adc
        inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
        where adc.adc_address_type in (1, 2) and
              adc.adc_from_c_id = :c_id
        into :adl_address_str, adc_address_type
    do
    begin
      error_txt = null;

      adl_address_str = replace(c_birth_city, '=', '');
      adl_address_str = replace(c_birth_city, '—', '-');



      select error_txt
      from hp_check_characters_str(:adl_address_str, '[ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyzа-я0-9@-@+@(@)@_@?#\/.,№!:;"'' ]+')
      into :error_txt;

      if (error_txt <> '') then
      begin
        if (adc_address_type = 1) then
          kf_error_text = 'Фактический адрес указан не корректно. ' || error_txt;
        else
          kf_error_text = 'Адрес регистрации указан не корректно. ' || error_txt;
        suspend;
      end
    end
    ------------ <

    if (c_inn <> '') then
    begin
      if (character_length(c_inn) <> 12) then
      begin
        kf_error_text = 'ИНН (' || c_inn || ') указан не корректно. Допустимая длина 12 символов, а сейчас ' || character_length(c_inn) || ' симв.';
        suspend;
      end
    end

    result = 1;
    if (substring(trim(kf_client_fio) from 1 for 1) similar to '[а-я]') then --Получаем 1й символ и проверяем на кириллицу
      :result = iif(kf_client_fio similar to '%[ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnoopqrstuvwxyz]+%', 1, 0); --кириллица
    else
      :result = iif(kf_client_fio similar to '%[а-я]+%', 1, 0); --латынь

    if (result = 1) then
    begin
      kf_error_text = 'ФИО указано не корреткно. В ФИО пристуствует как кириллица так и латиница';
      suspend;
    end

    for select trim(iif(adcl.adc_address_type = 1, 'Адрес регистрации', 'Адрес проживания')),
               coalesce(adl.adl_post_index, '')
        from addrclient adcl
        inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
        where adcl.adc_from_c_id = :c_id and
              adcl.adc_address_type in (1, 2)
        into :adr_type, :adl_post_index
    do
    begin
      err_txt = '';
      if (adl_post_index = '') then
        err_txt = 'почтовый индекс не указан';
      else
      if (character_length(adl_post_index) <> 6) then
        err_txt = 'длина почтового индекса должна быть 6 символов, сейчас: ' || character_length(adl_post_index) || ' символов. Почт. индекс: ' || adl_post_index;

      if (err_txt <> '') then
      begin
        kf_error_text = adr_type || ': ' || err_txt;
        suspend;
      end
    end

    for select pdt_code, cd.cds_doc_no
        from clientdocs cd
        inner join passkind on cds_from_pk_id = pk_id
        inner join passdoctype on pdt_id = pk_from_pdt_id
        where cds_from_c_id = :c_id and
              cds_doc_blob_1 > 0 and
              :k_disb_date between cds_doc_date_begin and coalesce(cds_doc_date_end, '31.12.9999') and
              cd.cds_verify_status in (0, 1) and
              cd.cds_is_delete = 0
        into :doc_code, doc_no
    do
    begin
      if (doc_code = 'snils') then
      begin
        if (char_length(doc_no) <> 11) then
        begin
          kf_error_is_critical = 2;
          kf_error_text = 'Ошибка в СНИЛС, длина мене 11 цифр. № документа: ' || doc_no;
          suspend;
        end
        else
        begin
          if (not(doc_no similar to '[-]?[0-9]+')) then
          begin
            kf_error_is_critical = 2;
            kf_error_text = 'Ошибка в СНИЛС, допустимы только цифры, без доп. символов: ' || doc_no;
            suspend;
          end
        end
      end
    end
  end
end
