CREATE PROCEDURE AAREP_CENTERBANK_REP_FILL(
    REP_DATE_BEGIN D_DATE,
    REP_QUERY_DATE D_DATE,
    RETURN_SCRIPT D_BOOL = 1,
    COMMIT_EACH_COUNT D_INTEGER = 1000)
RETURNS (
    RES_SQL D_STR_LONG)
AS
declare variable k_id d_id;
declare variable rec_no d_integer;
begin
  delete from rep_ru_cbank
  where 1 = 1;

  if (return_script is null) then
    return_script = 0;

  if (coalesce(commit_each_count, 0) = 0) then
    commit_each_count = 1000;

  if (return_script = 1) then
  begin
    res_sql = 'set statistics index pk_rep_ru_cbank;';
    suspend;
    res_sql = 'set statistics index rep_ru_cbank_k_id;';
    suspend;
  end

  rec_no = 0;
  for select distinct oks_from_k_id
      from kn_oksummary
      where oks_date_begin between :rep_date_begin and :rep_query_date
      into :k_id
  do
  begin
    rec_no = rec_no + 1;

    update or insert into rep_ru_cbank (line_no, k_id, rep_date_begin, rep_query_date, rep_need_recalc)
    values (:rec_no, :k_id, :rep_date_begin, :rep_query_date, 0)
    matching (k_id);

    if (return_script = 1) then
    begin
      res_sql = 'update rep_ru_cbank set rep_need_recalc = 1 where k_id = ' || k_id || ';';
  
      if (mod(rec_no, commit_each_count) = 0) then
      begin
        res_sql = res_sql || ' commit;';
        suspend;

        res_sql = 'set statistics index pk_rep_ru_cbank;';
        suspend;

        res_sql = 'set statistics index rep_ru_cbank_k_id;';
        suspend;
      end

      suspend;
    end
  end
  if (return_script = 1) then
    res_sql = res_sql || ' commit;';
  else
    res_sql = 'Заполнено строк: ' || rec_no;

  suspend;
end
