CREATE PROCEDURE HP_GENERATE_SMS_MESSAGE(
    C_ID D_ID)
RETURNS (
    RES_ERROR_TEXT D_STR_255,
    SMS_TEXT_STR D_STR_LONG)
AS
declare variable C_NAME_FIO D_STR_100;
declare variable C_SEX D_SMALLINT;
declare variable OK_CELL_MAIN D_STR_100;
begin
    -- Процедура обертка для модуля по отправке СМС.
    --
    -- RFCRU-2761

    res_error_text = '';

    if (coalesce(c_id, 0) = 0) then
    begin
        res_error_text = 'Клиент не выбран!';
        suspend;
        exit;
    end

    if (not exists(select * from clients c where c.c_id = :c_id)) then
    begin
        res_error_text = 'Клиент не найден';
        suspend;
        exit;
    end

    ok_cell_main = f_cont_info('clients', :c_id, 'cell_main', 'ok');

    if (coalesce(ok_cell_main, '') = '') then
    begin
        res_error_text = 'Ошибка: Номер телефона не указан в базе.';
        suspend;
        exit;
    end

    select c.c_name_fio, c.c_sex
    from clients c
    where c.c_id = :c_id
    into c_name_fio, c_sex;

    sms_text_str = iif(c_sex = 0, 'Уважаемая ', 'Уважаемый ') || c_name_fio ||', ';

    suspend;
end
