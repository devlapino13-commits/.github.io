CREATE PROCEDURE HP_OBJFUEL_VIEW(
    MC_ID D_ID)
RETURNS (
    ID D_STR_255,
    NAME D_STR_255,
    BILL_ID D_ID,
    BILL_LINE_NO D_INTEGER,
    INVENT_STR D_STR_CODE,
    ML_NAME D_STR_255,
    ML_MODEL D_STR_255,
    REG_ID D_INTEGER,
    REG_NAME D_STR_CODE,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    DT_OPER D_DATE)
AS
declare variable id_tmp d_str_255;
begin
  id_tmp = (select id from temp_objfuel_own where 1 = 0);
  for
    select
      ob_bill_id,
      ob_bill_line_no,
      ob_invent_str,
      ml_name,
      ml_model
    from operbill
      inner join matlist
        on ml_id = ob_from_ml_id
    where
      ob_from_mc_id = :mc_id
      and ob_is_delete = 0
      and not exists(
        select * from pettrans
        where pet_bill_id = ob_bill_id
          and pet_bill_line_no = ob_bill_line_no
          and pet_is_rent = 0)

  into
    :bill_id,
    :bill_line_no,
    :invent_str,
    :ml_name,
    :ml_model
  do
  begin
    select
      wp_from_reg_id,
      om_matotvet_id,
      max(ol_date)
    from opermat
      inner join operlist
        on ol_id = om_from_ol_id
          and is_delete = 0
      inner join workplace
        on wp_id = om_workplace_id
    where om_bill_id = :bill_id
      and om_bill_line_no = :bill_line_no
    group by
      wp_from_reg_id,
      om_matotvet_id
    having sum((1 - om_is_written_off) * om_kolich_plus - om_kolich_minus) <> 0
      or sum((1 - om_is_written_off) * om_summa_plus - om_summa_minus) <> 0
    into
      :reg_id,
      :c_id,
      :dt_oper;
    if (row_count = 1) then
    begin
      id = f_format('{"bill_id":%1,"bill_no":%2,"invent_str":"%3",', bill_id, bill_line_no, invent_str) ||
        f_format('"reg_id":%1,"c_id":%2,"dt_oper":"%3"}', reg_id, c_id, f_datetostr(dt_oper));
      name = invent_str || ' | ' || ml_name ||
        iif(ml_model in (ml_name, ''), '', ' ' || ml_model);
      reg_name = (select reg_name from region where reg_unicode = :reg_id);
      c_name_fio = (select c_name_fio from clients where c_id = :c_id);
      suspend;
    end
  end
end
