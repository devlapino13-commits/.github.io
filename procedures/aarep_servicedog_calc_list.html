CREATE PROCEDURE AAREP_SERVICEDOG_CALC_LIST(
    REP_REGION_ID INTEGER,
    REP_OFFICE_ID INTEGER,
    REP_DATE_END DATE,
    REP_SD_KIND INTEGER)
RETURNS (
    SD_ID INTEGER,
    SD_DOGOVOR_NO VARCHAR(20),
    SD_FROM_REG_ID INTEGER,
    SD_FROM_C_ID DECIMAL(18,0),
    SD_CATEGORY VARCHAR(255),
    SD_MEMO VARCHAR(1024),
    SD_DATE_BEGIN DATE,
    SD_DATE_END DATE,
    SD_DATE_CLOSED DATE,
    SD_ACC_MAIN VARCHAR(20),
    SD_ACC_ADVANCE VARCHAR(20),
    SD_ACC_EXPENSE VARCHAR(20),
    SD_USER_ID DECIMAL(18,0),
    SD_DATETIME TIMESTAMP,
    SD_REGION_NAME VARCHAR(100),
    SD_IS_INDIV_STR VARCHAR(50),
    SD_ONCE_ONLY_PAY_STR VARCHAR(50),
    SD_NEED_PATENT_STR VARCHAR(50),
    SD_NEED_SPOLIS_STR VARCHAR(50),
    SD_ONLY_WORKDAYS_STR VARCHAR(50),
    SD_CLIENT_NAME VARCHAR(100),
    SD_CLIENT_PASS_NO VARCHAR(30),
    SD_STATUS_STR VARCHAR(50),
    SD_STATUS INTEGER,
    SD_KIND_STR VARCHAR(100),
    SD_HAS_PATENT INTEGER,
    SD_HAS_SPOLIS INTEGER,
    SD_ALL_DAYS INTEGER,
    SD_BRANCH_ID INTEGER,
    SD_BRANCH_NAME VARCHAR(100),
    SD_DATE_LAST_CALC DATE,
    SD_DATE_LAST_DISB DATE,
    SD_SUM_PLAN DECIMAL(18,4),
    SD_SUM_SALDO DECIMAL(18,4),
    SD_PLUS_ALL DECIMAL(18,4),
    SD_MINUS_ALL DECIMAL(18,4),
    SD_DISB_ALL DECIMAL(18,4),
    SD_ONLY_WORKDAYS INTEGER,
    SK_SFOND_KOEF DECIMAL(18,7),
    SK_PDNAL_KOEF DECIMAL(18,7),
    SK_COMP_SFOND_KOEF DECIMAL(18,7),
    SK_SFOND_MIN_ZP DECIMAL(18,4),
    SD_EDOC_STATUS D_STR_20,
    SD_PAY_FREELANCE D_CURRENCY)
AS
declare variable sd_is_indiv integer;
declare variable sd_once_only_pay integer;
declare variable sd_need_patent integer;
declare variable sd_need_spolis integer;
declare variable sd_sum_calced_dog decimal(18,4);
declare variable can_suspend integer;
declare variable dd_temp date;
declare variable st_staff integer;
declare variable st_client integer;
declare variable db_id d_integer;
declare variable eds_status d_str_code;
begin
  db_id = f_get_db_id();
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select sd_id, sd_dogovor_no, sd_from_reg_id, sd_from_c_id, sd_category, sd_memo, sd_is_indiv, sd_date_begin,
             sd_date_end, sd_date_closed, sd_once_only_pay, sd_acc_main, sd_acc_advance, sd_acc_expense, sd_need_patent,
             sd_need_spolis, sd_only_workdays, sd_user_id, sd_datetime, r1.reg_name || ' ' || r1.reg_code, c_name_fio,
             c_pass_no, sd_status, sk_name, r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name, sk_sfond_koef,
             sk_pdnal_koef, sk_comp_sfond_koef, r1.reg_sfond_min_zp, sd_pay_freelance
      from servicedog, region r1, region_list(:rep_region_id, 1, -1, 1), clients, servicekind, region r2
      where r1.reg_unicode = sd_from_reg_id and
            r2.reg_unicode = r1.reg_head_office and
            ((:rep_office_id = 0) or
            (r1.reg_head_office = :rep_office_id)) and
            (sd_from_reg_id = id_list) and
            c_id = sd_from_c_id and
            sd_kind = sk_id and
            (:rep_sd_kind = -1 or
            sd_kind = :rep_sd_kind)
      into :sd_id, :sd_dogovor_no, :sd_from_reg_id, :sd_from_c_id, :sd_category, :sd_memo, :sd_is_indiv, :sd_date_begin,
           :sd_date_end, :sd_date_closed, :sd_once_only_pay, :sd_acc_main, :sd_acc_advance, :sd_acc_expense,
           :sd_need_patent, :sd_need_spolis, :sd_only_workdays, :sd_user_id, :sd_datetime, :sd_region_name,
           :sd_client_name, :sd_client_pass_no, :sd_status, :sd_kind_str, :sd_branch_id, :sd_branch_name,
           :sk_sfond_koef, :sk_pdnal_koef, :sk_comp_sfond_koef, :sk_sfond_min_zp, :sd_pay_freelance
  do
  begin
    if (sd_is_indiv = 0) then
      sd_is_indiv_str = 'Юр.лицо';
    else
      sd_is_indiv_str = 'Физ.лицо';
    if (sd_once_only_pay = 1) then
      sd_once_only_pay_str = 'Разовый';
    else
      sd_once_only_pay_str = 'Ежемесячный';
    if (sd_need_patent = 1) then
      sd_need_patent_str = 'Требуется';
    else
      sd_need_patent_str = 'Не требуется';
    if (sd_need_spolis = 1) then
      sd_need_spolis_str = 'Требуется';
    else
      sd_need_spolis_str = 'Не требуется';
    if (sd_only_workdays = 1) then
      sd_only_workdays_str = 'Почасовая';
    else
      sd_only_workdays_str = 'За месяц';
    if (sd_status = -1) then
      sd_status_str = 'Аннулирован';
    if (sd_status = 0) then
      sd_status_str = 'В процессе';
    if (sd_status = 1) then
      sd_status_str = 'Утвержден';
    if (sd_status = 2) then
      sd_status_str = 'Закрыт';
    if (sk_sfond_min_zp is null) then
      sk_sfond_min_zp = 0;

    sd_sum_plan = 0;
    sd_sum_saldo = 0;
    sd_date_last_calc = null;
    sd_sum_calced_dog = 0;
    sd_plus_all = 0;
    sd_minus_all = 0;
    sd_disb_all = 0;

    if (rep_sd_kind = 0 and
        db_id in (39801, 41701)) then
    begin
      eds_status = f_edoc_status('freelancer_kontract', :sd_id, current_date);
      if (eds_status = 'empty') then
        sd_edoc_status = 'Не отправлялся';
      if (eds_status = 'signed') then
        sd_edoc_status = 'Подписан';
      if (eds_status = 'unsigned') then
        sd_edoc_status = 'Не подписан';
      if (eds_status = 'cancelled') then
        sd_edoc_status = 'Аннулирован';
    end

    /* Всего к начислению по договору, по графику */
    if (sd_only_workdays = 0) then
    /* Ежемесячная оплата */
    begin
      select sum(ssh_amount_dogovor)
      from servicesh
      where ssh_date <= :rep_date_end and
            ssh_date_regist <= :rep_date_end and
            ssh_from_sd_id = :sd_id
      into :sd_sum_plan;
      if (sd_sum_plan is null) then
        sd_sum_plan = 0;
    end
    else
    begin /* Почасовая оплата */
      select ssh_date, sum(ssh_amount_dogovor)
      from servicesh
      where ssh_date <= :rep_date_end and
            ssh_date_regist <= :rep_date_end and
            ssh_from_sd_id = :sd_id
      group by ssh_date
      order by ssh_date desc
      rows 1
      into :dd_temp, :sd_sum_plan;
      if (sd_sum_plan is null) then
        sd_sum_plan = 0;
    end

    /* всего начислено уже по договору, также история начислений и удержаний */
    select sum(ssc_amount_dogovor), sum(ssc_amount_dogovor + ssc_other_plus),
           sum(ssc_tax_client_sfond + ssc_tax_client_pdnal + ssc_tax_client_other), sum(ssc_client_saldo),
           max(ssc_date)
    from servicecalc, operlist
    where ssc_from_sd_id = :sd_id and
          ol_id = ssc_from_ol_id and
          operlist.is_delete = 0 and
          ol_date <= :rep_date_end and
          ssc_kind = 0
    into :sd_sum_calced_dog, :sd_plus_all, :sd_minus_all, :sd_sum_saldo, :sd_date_last_calc;
    if (sd_sum_calced_dog is null) then
      sd_sum_calced_dog = 0;
    if (sd_sum_saldo is null) then
      sd_sum_saldo = 0;
    if (sd_plus_all is null) then
      sd_plus_all = 0;
    if (sd_minus_all is null) then
      sd_minus_all = 0;
    if (sd_only_workdays = 0) then
    begin
      sd_sum_plan = sd_sum_plan - sd_sum_calced_dog;
    end
    if (sd_sum_plan < 0) then
      sd_sum_plan = 0;
    if (sd_date_last_calc is null) then
      sd_date_last_calc = :sd_date_begin;
    else
      sd_date_last_calc = sd_date_last_calc + 1;
    if (sd_only_workdays = 1) then
    /* почасовая оплата */
    begin
      if (sd_date_last_calc > :rep_date_end) then
        sd_sum_plan = 0;
    end

    /* всего выдано до указанной даты */
    select sum(ssc_disb_amount), max(ol_date)
    from servicecalc, operlist
    where ssc_from_sd_id = :sd_id and
          ssc_disb_amount <> 0 and
          ol_id = ssc_from_ol_id and
          operlist.is_delete = 0 and
          ssc_kind = 1
    into :sd_disb_all, :sd_date_last_disb;
    if (sd_disb_all is null) then
      sd_disb_all = 0;
    sd_sum_saldo = sd_sum_saldo - sd_disb_all;

    sd_all_days = 0;
    sd_has_patent = 0;
    sd_has_spolis = 0;
    if (sd_date_last_calc < rep_date_end) then
    begin
      select res_days_all, res_days_patent, res_days_spolis
      from aarep_service_patent_kolich(:sd_id, :sd_date_last_calc, :rep_date_end)
      into :sd_all_days, :sd_has_patent, :sd_has_spolis;
    end
    /*
      SD_STATUS=-1 - Аннулирован';
      SD_STATUS=0 - В процессе';
      SD_STATUS=1 - Утвержден';
      SD_STATUS=2 - Закрыт';
      */

    can_suspend = 1;
    if (sd_status = -1) then
    begin
      /* Если договор аннулирован, то если нет задолженности то не отображаем в списке к начислению */
      if (sd_sum_saldo = 0) then
        can_suspend = 0;
    end
    if (sd_status = 2) then
    begin
      /* Если договор закрыт, то если нет задолженности то не отображаем в списке к начислению */
      if (sd_sum_saldo = 0) then
        can_suspend = 0;
    end
    if (sd_status = 0) then
    begin
      /* Если договор в процессе,
        и не были никакие расчеты, то не отображаем в списке к начислению */
      if ((sd_disb_all = 0) and
          (sd_plus_all = 0) and
          (sd_sum_plan = 0)) then
        can_suspend = 0;
    end
    if (can_suspend = 1) then
      suspend;
  end
end
