CREATE PROCEDURE HP_KS_EXP_CALC(
    REP_DATE_END D_DATE,
    REP_C_ID D_ID = 0)
RETURNS (
    REG_NAME_CHANGED D_STR_100,
    FIO_CHANGED D_STR_100,
    HIRED_DATE_CHANGED D_DATE,
    FIRST_ACTION_DATE_CHANGED D_DATE,
    LAST_EXP_CHANGED D_INTEGER,
    LAST_DEKR_CHANGED D_INTEGER,
    DEKR_CHANGED D_INTEGER,
    TOTAL_STAJE_CHANGED D_INTEGER,
    LO_ID D_ID)
AS
declare variable reg_id d_integer;
declare variable hrv_date_end_fact d_date;
declare variable hrv_date_begin d_date;
declare variable staje_tmp d_integer;
declare variable ks_kc_id id_type;
declare variable ps_date_begin d_date;
declare variable ps_from_jl_id d_id;
declare variable jk_system_code d_str_100;
declare variable ps_date_begin_temp d_date;
declare variable ps_date_end d_date;
declare variable ob_stag d_integer;
declare variable ho_order_type d_integer;
declare variable sql d_str_2048;
begin
  sql = f_format('select
        ps.ps_from_c_id,
        ps.ps_from_reg_id
      from hr_pstatus ps
        inner join joblist jl
          on jl.jl_id = ps.ps_from_jl_id
        inner join jobkind jk
          on jk.jk_id = jl.jl_kind
            and jk.jk_system_code in (''JobSystem.LO_Standart'', ''JobSystem.LO_Senior'')
      where ps.ps_status = 1
        and ps.ps_is_working = 1
        and ''%1'' between ps.ps_date_begin and ps.ps_date_end
        %2', rep_date_end, iif(rep_c_id > 0, 'and ps.ps_from_c_id = ' || :rep_c_id, ''));
  for execute statement sql
          into :lo_id, :reg_id
  do
  begin
    last_dekr_changed = 0;
    total_staje_changed = 0;
    dekr_changed = 0;
    staje_tmp = 0;
    hired_date_changed = null;
    last_exp_changed = 0;
    for select ps.ps_date_begin, ps.ps_from_jl_id, ho.ho_order_type
        from hr_pstatus ps
        inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id
        where ps.ps_from_c_id = :lo_id and
              ps_status = 1 and
              ps_date_begin <= ps_date_end and
              ps_date_begin <= :rep_date_end
        order by ps.ps_date_begin desc
        into ps_date_begin, ps_from_jl_id, ho_order_type
    do
    begin
      if (ho_order_type = 6) then
        break;
      select jk.jk_system_code
      from joblist jl
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where jl.jl_id = :ps_from_jl_id
      into jk_system_code;
      if (:jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior')) then
        ps_date_begin_temp = :ps_date_begin;
      else
        break;
    end
    for select ps.ps_date_begin, ps.ps_date_end
        from hr_pstatus ps
        inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id
        inner join joblist jl on ps.ps_from_jl_id = jl.jl_id
        inner join jobkind jk on jk.jk_id = jl.jl_kind and
              jk.jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior')
        where ps.ps_from_c_id = :lo_id and
              ps.ps_is_working = 1 and
              ps.ps_status = 1 and
              ps.ps_is_on_staff = 1 and
              ps_date_begin <= ps_date_end and
              ps_date_begin <= :rep_date_end
        order by ps.ps_date_begin desc
        into ps_date_begin, ps_date_end
    do
    begin
      if (ps_date_end > :rep_date_end) then
        ps_date_end = :rep_date_end;

      ob_stag = datediff(month, ps_date_begin, ps_date_end);
      if (extract(day from :ps_date_begin) = 1 and
          datediff(month, rep_date_end, rep_date_end + 1) <> 0) then
        total_staje_changed = total_staje_changed + 1;
      else
      if (extract(day from :ps_date_begin) > extract(day from :ps_date_end)) then
        total_staje_changed = total_staje_changed - 1;
      total_staje_changed = total_staje_changed + ob_stag;

      for select hrv_date_end_fact, hrv_date_begin
          from hr_voctravel, hr_orders, sa_worktype
          where hrv_from_ho_id = ho_id and
                ho_order_type = sawt_id and
                hrv_from_c_id = :lo_id and
                hrv_status = 1 and
                sawt_id in (17, 18, 24) and
                (hrv_date_begin between :ps_date_begin and :ps_date_end or
                (hrv_date_end_fact between :ps_date_begin and :ps_date_end))
          into hrv_date_end_fact, hrv_date_begin
      do
      begin
        staje_tmp = 0;
        if (hrv_date_end_fact > :ps_date_end) then
          hrv_date_end_fact = :ps_date_end;
        if (hrv_date_begin < :ps_date_begin) then
          hrv_date_begin = :ps_date_begin;
        staje_tmp = datediff(month, hrv_date_begin, hrv_date_end_fact);
        dekr_changed = dekr_changed + staje_tmp;
      end
    end
    hired_date_changed = :ps_date_begin_temp;
    select k_disb_date, ks_kc_id
    from kontract kn
    inner join kontstate ks on ks.ks_from_k_id = kn.k_id
    where kn.k_fact_disb_lo_id = :lo_id and
          k_disb_date > :hired_date_changed
    order by k_disb_date
    rows 1
    into first_action_date_changed, ks_kc_id;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name_changed;

    select c_name_fio
    from clients
    where c_id = :lo_id
    into fio_changed;
    last_exp_changed = datediff(month, :ps_date_begin_temp, :rep_date_end);
    if (extract(day from :ps_date_begin_temp) = 1 and
        datediff(month, :rep_date_end, :rep_date_end + 1) <> 0) then
      last_exp_changed = last_exp_changed + 1;
    if (last_exp_changed > total_staje_changed) then
      total_staje_changed = last_exp_changed;
    if (:lo_id = 1090100209233) then
    begin
      dekr_changed = dekr_changed - 1;
      last_dekr_changed = last_dekr_changed - 1;
      total_staje_changed = total_staje_changed + 1;
    end
    for select hrv_date_end_fact, hrv_date_begin
        from hr_voctravel, hr_orders, sa_worktype
        where hrv_from_ho_id = ho_id and
              ho_order_type = sawt_id and
              hrv_from_c_id = :lo_id and
              hrv_status = 1 and
              sawt_id in (17, 18, 24) and
              (hrv_date_begin between :ps_date_begin_temp and :rep_date_end or
              (hrv_date_end_fact between :ps_date_begin_temp and :rep_date_end))
        into hrv_date_end_fact, hrv_date_begin
    do
    begin
      staje_tmp = 0;
      if (hrv_date_begin < :ps_date_begin_temp) then
        hrv_date_begin = :ps_date_begin_temp;
      if (hrv_date_end_fact > :rep_date_end) then
        hrv_date_end_fact = :rep_date_end;
      staje_tmp = datediff(month, hrv_date_begin, hrv_date_end_fact);
      last_dekr_changed = last_dekr_changed + staje_tmp;
      if (extract(day from :ps_date_begin_temp) = 1 and
          datediff(month, :rep_date_end, :rep_date_end + 1) <> 0) then
        last_dekr_changed = last_dekr_changed + 1;
      if (last_dekr_changed > dekr_changed) then
        dekr_changed = last_dekr_changed;
    end
    total_staje_changed = total_staje_changed - dekr_changed;
    if (:lo_id = 1090100410319) then
      total_staje_changed = total_staje_changed - 1;
    suspend;
  end
end
