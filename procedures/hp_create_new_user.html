CREATE PROCEDURE HP_CREATE_NEW_USER(
    UN_NAME VARCHAR(30),
    UN_FROM_C_ID DECIMAL(18,0),
    UN_FIO VARCHAR(100),
    UN_CURR_REG_ID INTEGER,
    UN_LIKE_ID DECIMAL(18,0),
    RN_ID DECIMAL(18,0),
    REGISTERING_UN_ID DECIMAL(18,0))
RETURNS (
    RESULT_UN_ID DECIMAL(18,0),
    RESULT_MESSAGE VARCHAR(255))
AS
declare variable un_str varchar(100);
declare variable old_un_id decimal(18,0);
declare variable new_un_id decimal(18,0);
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
declare variable uniqtab_menulist integer;
declare variable uniqtab_permactions integer;
declare variable uniqtab_region integer;
declare variable db_id d_integer;
declare variable cell_no d_str_20;
begin
  db_id = f_get_db_id();

  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist;
  select ukt_id
  from hp_get_uniqtable_id('PermActions')
  into :uniqtab_permactions;
  select ukt_id
  from hp_get_uniqtable_id('Region')
  into :uniqtab_region;

  result_un_id = 0;
  result_message = '';
  if (un_from_c_id = 0) then
    un_from_c_id = null;

  un_str = '';
  old_un_id = null;
  select un_id, un_fio
  from username
  where un_name = :un_name
  into :old_un_id, :un_str;
  if (un_str is null) then
    un_str = '';
  if (old_un_id is not null) then
  begin
    result_message = 'Пользователь с таким логином существует в базе, ' || ascii_char(10) || ascii_char(13) || 'Имя: ' || :un_str;
    suspend;
    exit;
  end

  if (un_from_c_id is not null) then
  begin
    un_str = '';
    old_un_id = null;
    select un_id, un_name
    from username
    where un_from_c_id = :un_from_c_id rows 1
    into :old_un_id, :un_str;
    if (un_str is null) then
      un_str = '';
    if (old_un_id is not null) then
    begin
      result_message = 'У данного контрагента уже существует пользователь, ' || ascii_char(10) || ascii_char(13) || 'Логин: ' || :un_str;
      suspend;
      exit;
    end
  end

  if (un_like_id <> 0) then
  begin
-- Проверяем пользователя на наличие конфиденциальных данных
    if ((exists(select has_access
                from hp_permission_access_check(:un_like_id, 'PermActions', 0, 'FinStart.Superuser')
                where has_access = 1)) and
        (exists(select has_access
                from hp_permission_access_check(:registering_un_id, 'PermActions', 0, 'FinStart.Superuser')
                where has_access = 0))) then
    begin
      result_message = 'У вас нет прав на создание пользователя с правом Superuser.';
      suspend;
      exit;
    end

    if (exists(select has_access
               from hp_permission_access_check(:un_like_id, 'PermActions', 0, 'FinStart.Permission_Administrator')
               where has_access = 1) and
        (exists(select has_access
                from hp_permission_access_check(:registering_un_id, 'PermActions', 0, 'FinStart.Superuser')
                where has_access = 0))) then
    begin
      result_message = 'У вас нет прав на создание пользователя с правом Менеджер тех.поддержки.';
      suspend;
      exit;
    end

    if (exists(select has_access
               from hp_permission_access_check(:un_like_id, 'PermActions', 0, 'FinStart.Permission_Operator')
               where has_access = 1) and
        (exists(select has_access
                from hp_permission_access_check(:registering_un_id, 'PermActions', 0, 'FinStart.Superuser')
                where has_access = 0)) and
        (exists(select has_access
                from hp_permission_access_check(:registering_un_id, 'PermActions', 0, 'FinStart.Permission_Administrator')
                where has_access = 0))) then
    begin
      result_message = 'У вас нет прав на создание пользователя с правом Оператор тех.поддержки.';
      suspend;
      exit;
    end
  end
  else
  begin
    if (exists(select has_access
               from hp_permission_access_check(:registering_un_id, 'PermActions', 0, 'FinStart.Superuser')
               where has_access = 0)) then
    begin
      if (exists(select *
                 from permissions, menulist
                 where pm_object_kind = :uniqtab_rolename and
                       pm_object_kind = :rn_id and
                       pm_subject_kind = :uniqtab_menulist and
                       pm_subject_id = ml_id and
                       pm_is_delete = 0 and
                       coalesce(pm_date_begin, '01.01.2000') <= current_date and
                       coalesce(pm_date_end, '31.12.2999') >= current_date and
                       ml_secure_kind = -1)) then
      begin
        result_message = 'У вас нет прав на создание пользователя с этой ролю';
        suspend;
        exit;
      end

      if (exists(select *
                 from permissions, permactions
                 where pm_object_kind = :uniqtab_rolename and
                       pm_object_kind = :rn_id and
                       pm_subject_kind = :uniqtab_permactions and
                       pm_subject_id = pa_id and
                       pm_is_delete = 0 and
                       coalesce(pm_date_begin, '01.01.2000') <= current_date and
                       coalesce(pm_date_end, '31.12.2999') >= current_date and
                       pa_secure_kind = -1)) then
      begin
        result_message = 'У вас нет прав на создание пользователя с этой ролю';
        suspend;
        exit;
      end
    end
  end

  select result
  from get_kv_id(9)
  into :result_un_id;
  new_un_id = null;
  insert into username (un_id, un_name, un_curr_reg_id, un_registered_user, un_fio, un_from_c_id)
  values (:result_un_id, :un_name, :un_curr_reg_id, :registering_un_id, :un_fio, :un_from_c_id)
  returning un_id
  into :new_un_id;

  if (new_un_id is null) then
  begin
    result_message = 'Возникла ошибка при создании пользователя';
    suspend;
    exit;
  end
  result_un_id = new_un_id;

  -- S.K.
  if (db_id in (39801, 41701)) then
  begin
    cell_no = (
      select res_value
      from aarep_object_one_continfo('0', :un_from_c_id, 'ContactInfo.Cell_Main', null, null)
      where res_verify_status = 2);
    if (cell_no is null) then
    begin
      result_message = 'Верифицированный номер отсутствует, пользователь не может быть создан';
      suspend;
      exit;
    end else begin
      update username set
        un_password_new = ''
      where un_id = :new_un_id;
      execute procedure hp_send_quick_message(
        'sms',
        :cell_no,
        '',
        'Vash login ' || :un_name || '. Dlya sozdaniya parolya v FinStart najmite "Zabyli parol". ' ||
          'Pri vhode v FSMobile najmite "Sbrosit parol/Smenit telefon"',
        '',
        -1,
        1000000000000);
    end
  end

  if (un_like_id <> 0) then
  begin
-- Копирование ролей пользователя
    insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_date_begin,
                             pm_date_end, pm_access_kind, pm_is_delete, pm_from_un_id)
    select -1, 0, :new_un_id, pm_subject_kind, pm_subject_id, pm_date_begin, pm_date_end, pm_access_kind, pm_is_delete,
           :registering_un_id
    from permissions
    where pm_object_kind = :uniqtab_username and
          pm_object_id = :un_like_id and
          pm_is_delete = 0 and
          pm_subject_kind = :uniqtab_rolename and
          coalesce(pm_date_begin, '01.01.2000') <= current_date and
          coalesce(pm_date_end, '31.12.2999') >= current_date;

    insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_access_kind,
                             pm_is_delete, pm_from_un_id)
    values (-1, :uniqtab_username, :new_un_id, :uniqtab_region, :un_curr_reg_id, 1, 0, :registering_un_id);

    suspend;
    exit;
  end
  else
  begin
    if ((rn_id is not null) and
        (rn_id <> 0)) then
    begin
      insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_access_kind,
                               pm_is_delete, pm_from_un_id)
      values (-1, :uniqtab_region, :new_un_id, :uniqtab_rolename, :rn_id, 1, 0, :registering_un_id);
      suspend;

      insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_access_kind,
                               pm_is_delete, pm_from_un_id)
      values (-1, :uniqtab_username, :new_un_id, :uniqtab_region, :un_curr_reg_id, 1, 0, :registering_un_id);

      exit;
    end
  end
  result_message = 'Пользователь не создан из-за отсутствия роли.';
  suspend;
end
