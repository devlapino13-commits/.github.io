CREATE PROCEDURE HP_KONTUR_IP_GET_LIC_INFO(
    OBJECT_KIND D_STR_CODE,
    OBJECT_ID D_ID)
RETURNS (
    REQ_STATUS D_SMALLINT,
    REQ_DATE D_DATE,
    ACTUALITY_DATE D_DATE,
    OGRN D_STR_CODE,
    LICENSE_TXT D_STR_8100)
AS
declare variable jso d_integer;
declare variable licenses_jso d_integer;
declare variable services_jso d_integer;
declare variable official_num d_str_255;
declare variable date_start d_str_10;
declare variable issuer_name d_str_long;
declare variable date_end d_str_10;
declare variable srvices d_str_4096;
declare variable srvice d_str_long;
begin
  -- RFCRU-5304 получение ответа из Контур (Информация о лицензиях)
  -- входной object_kind может быть miniapp, kontract, knclient, clients.
/*
  req_status -- eil_status
  req_date -- Дата запроса
  actuality_date -- Дата актуальности (Дата запроса + eic_info_actual_days
*/

  select jso, req_status, req_date, actuality_date
  from hp_extinfo_get_jso('kontur_ip_lic', :object_kind, :object_id)
  into jso, req_status, req_date, actuality_date;

/*
  ogrn -- Регистрационный номер индивидуального предпринимателя (ОГРНИП)  Api3ReqResponse[0].ogrn
  license_txt --  Сведения о лицензии
        (номер, дата выдачи лицензии; кем выдана; срок действия; перечень видов лицензируемой деятельности)
         Api3LicencesResponse[0].licenses[n]:
        .officialNum
        .dateStart
        .issuerName
        .dateEnd
        .services (массив строк, склеить через запятую)
*/

  if (coalesce(req_status, 0) < 2) then   -- запрос не найден
  begin
    suspend;
    exit;
  end

  if (jso is null) then
    exception e_default 'json пустой. ' || iif(exists(select 1 from hp_db_is_actual), '', 'Необходимо синхронизировать blobfiles');

  select ogrn
  from hp_kontur_ip_get_base_info(:object_kind, :object_id)
  into ogrn;

  for select json.obj_get_val(jso, 'ogrn'),
             json.obj_get_arr(jso, 'licenses')
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'data'), 1)
  where (coalesce(:ogrn, '') = '' or json.obj_get_val(jso, 'ogrn') = :ogrn)
  rows 1
  into ogrn,
       licenses_jso
  do
  begin
    if (licenses_jso is not null) then
    begin
      license_txt = '';

      for select json.obj_get_val(jso, 'officialNum'),               -- номер
                 f_datetostr(json.obj_get_val(jso, 'dateStart')),    -- дата выдачи лицензии
                 json.obj_get_val(jso, 'issuerName'),                -- кем выдана
                 f_datetostr(json.obj_get_val(jso, 'dateEnd')),      -- срок действия
                 json.obj_get_arr(jso, 'services')
      from hp_bki_json_get_list(:licenses_jso, 1)
      into official_num,
           date_start,
           issuer_name,
           date_end,
           services_jso
      do
      begin
        srvices = '';
        if (services_jso is not null) then
        begin
          -- перечень видов лицензируемой деятельности
          for select val
          from hp_bki_json_get_list(:services_jso, 0)
          into srvice
          do
          begin
            srvices = srvices || iif(srvices = '', '', ', ') || srvice;
          end
        end
        license_txt = license_txt || f_format('%1, %2; %3; %4; %5 ', official_num, date_start, issuer_name, date_end, srvices);
      end
    end
  end

  suspend;
end
