CREATE PROCEDURE HP_EXTRACT_CIB(
    CONTRACT_NO D_STR_20)
RETURNS (
    CIB_ID D_INTEGER,
    CIB_NAME D_STR_255,
    JSON_TEXT D_BLOB_TEXT)
AS
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable bls_id d_integer;
declare variable eic_code d_str_code;
begin
  for select k.k_id, k.k_from_ma_app_id
      from kontract k
      where k.k_kontract_no = :contract_no
      into k_id, ma_id
  do
  begin
    for select eil.eil_from_bls_id, eic.eic_code, eic.eic_name
        from extinfolist eil
        inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
              eic.eic_id in (14, 16, 1, 2)
        where ((eil.eil_object_kind = 11 and
              eil.eil_object_id = :ma_id) or
              (eil.eil_object_kind = 9 and
              eil.eil_object_id = :k_id)) and
              eil.eil_status = 2 and
              eil.eil_is_json = 1
        into bls_id, eic_code, cib_name
    do
    begin
      if (eic_code containing 'nbki') then
        cib_id = 1;
      else
        cib_id = 2;
      select finish_json_text
      from hp_mask_client_data(:bls_id)
      into json_text;
      if (json_text is not null) then
        suspend;
    end
  end
end
