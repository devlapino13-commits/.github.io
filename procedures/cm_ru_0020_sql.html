CREATE PROCEDURE CM_RU_0020_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable KF_ID ID_TYPE;
declare variable CURR_K_PERIOD decimal(15,2);
declare variable CURR_K_DISB_DATE date;
declare variable IS_DELINQ smallint;
declare variable KC_FROM_C_ID CURR_TYPE;
declare variable K_KOLICH_CLIENT_APPROVED integer;
declare variable C_LAST_BONUS_STATUS integer;
declare variable C_ID ID_TYPE;
declare variable K_PERCENT PERC_TYPE;
declare variable K_KOLICH_REPAY integer;
declare variable OK_SH_ID integer;
declare variable OK_OS CURR_TYPE;
declare variable OK_PERC CURR_TYPE;
declare variable KC_AMOUNT_APPROVED CURR_TYPE;
declare variable K_KONTRACT_NO varchar(30);
declare variable CURR_K_ID id_type;
declare variable LAST_K_ID id_type;
begin
   res_error_text = '';
-- 5. Начиная с 1 ноября 2014г. статусные клиенты могут получить параллельный заем, при условии отсутсвтия просрочек гр. займа
-- по истечении 65 дней с даты выдачи и двух платежей в соответсвии с графиком платежей
   kf_id = null;
   select k_from_kf_id, k_disb_date, k_kolich_client_approved, k_percent, k_kolich_repay
   from kontract, product
   where k_id = :k_id
   and k_from_pr_id = pr_id
   and pr_product_kind = 2 -- проверка только по параллельным кредитам
   into kf_id, kf_disb_date, k_kolich_client_approved, k_percent, k_kolich_repay;
   if (kf_id is null) then
   begin
     Exit;
   end 
-- Определяется основной групповой кредит
   CURR_K_ID = null; -- ID основного контракта
   select k_id,
          k_period,
          ks_date_begin
   from kontstate, kontract, product
   where ks_ol_is_delete = 0
   and ks_state = 1
   and :KF_DISB_DATE between ks_date_begin and ks_date_end
   and ks_from_k_id = k_id
   and k_from_kf_id = :kf_id
   and k_from_pr_id = pr_id
   and pr_product_kind = 1   -- групповой кредит
   order by ks_date_begin desc
   rows 1
   into CURR_K_ID, curr_k_period, curr_k_disb_date;
   if (CURR_K_ID is null) then -- or curr_k_disb_date > cast('23.10.2014' as date)) then
   begin
     res_error_text = 'У данного кредита не имеется активный основной групповой заем на 23 октрября 2014г.';
     res_error_is_critical = 1;
     suspend;
   end
   if (curr_k_period < 5.90) then
   begin
     res_error_text = 'Срок основного группового кредита менее 6 месяцев!';
     res_error_is_critical = 1;
     suspend;
   end 
   if (kf_disb_date - curr_k_disb_date >= 65) then
   begin
     for
     select ok_sh_id, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
     from operkred
     where ok_ol_is_active = 1
     and ok_from_k_id = :curr_k_id
     and ok_ol_date <= :kf_disb_date
     and ok_from_c_id=0
--     and ok_sh_date <= :kf_disb_date
     and ok_type < 7
     group by ok_sh_id
     into ok_sh_id, ok_os, ok_perc
     do
     begin
       if (ok_sh_id in (1, 2) and ok_os + ok_perc > 0) then
       begin
         res_error_text = 'У основного кредита не произведены 2 успешные платежи!';
         res_error_is_critical = 1;
         suspend;
       end
     end 
   end else
   begin
     res_error_text = 'Со дня выдачи основного группового кредита еще не иcтекли 65 дней!';
     res_error_is_critical = 1;
     suspend;
   end 
-- Проверка просрочки группового займа
   if (CURR_K_ID is not null) then
   begin
     is_delinq = null;
     select 1 from oksummary
     where oks_from_k_id = :CURR_K_ID
     and :KF_DISB_DATE between oks_date_begin and oks_date_end
     and :KF_DISB_DATE > oks_delinq_date
     and oks_close_state = 0
     into is_delinq;
     if (is_delinq is null) then is_delinq = 0;
     if (is_delinq = 1) then
     begin
       res_error_text = 'Основной групповой займ Просрочен!';
       res_error_is_critical = 1;
       suspend;
     end else
     begin
--     имеется ли просроченные платежи
       is_delinq = null;
       select 1 from operkred
       where ok_from_k_id = :CURR_K_ID
       and ok_ol_is_active = 1
       and ok_ol_date <= :KF_DISB_DATE
       and ok_type in (4,5)
       and ok_ol_date > ok_sh_date
       and ok_os + ok_perc < 0
       and ok_from_c_id=0
       rows 1
       into is_delinq;
       if (is_delinq is null) then is_delinq = 0;
       if (is_delinq = 1) then
       begin
         res_error_text = 'В основном групповом займе были просроченные платежи!';
         res_error_is_critical = 1;
         suspend;
       end else
       begin
--      Проверяется предыдущий цикл группового займа
           LAST_K_ID = null; -- ID основного контракта
           select k_id from kontstate, kontract, product
           where ks_ol_is_delete = 0
           and ks_state = 1
           and ks_from_k_id = k_id
           and k_from_kf_id = :kf_id
           and k_from_pr_id = pr_id
           and pr_product_kind = 1
           and ks_date_begin <= :curr_k_disb_date
           and ks_from_k_id <> :curr_k_id
           order by ks_date_begin desc
           rows 1
           into LAST_K_ID;
--         если предыдущий контракт существует то проверяет на просроченные платежы
           if (LAST_K_ID is not null) then
           begin
               is_delinq = null;
               select 1 from operkred
               where ok_from_k_id = :LAST_K_ID
               and ok_ol_is_active = 1
               and ok_ol_date <= :KF_DISB_DATE
               and ok_type in (4,5)
               and ok_ol_date > ok_sh_date
               and ok_os + ok_perc < 0
               and ok_from_c_id=0
               rows 1
               into is_delinq;
               if (is_delinq is null) then is_delinq = 0;
        
               if (is_delinq = 1) then
               begin
                 res_error_text = 'В предыдущем групповом займе были просроченные платежи!';
                 res_error_is_critical = 1;
                 suspend;
               end
           end 
       end 
     end
   end
-- если не имеются предупреждении то проверяет следующее
--         Выстаскивает статус на начало основного кредита
--         Если в группе берут 2 и более участников параллельный кредит то проверяется статус на начало
--         А если один то статус данного клиента
  if (res_error_text = '') then
  begin
       if (k_kolich_client_approved > 1) then
       begin
         select C_LAST_BONUS_STATUS
         from HP_KONTSTATE_LOAN_STATUS(:k_id, 0, 0, :kf_disb_date, '')
         into :C_LAST_BONUS_STATUS;
       end else
       begin
         c_id = null;
         select kc_from_c_id from knclient
         where kc_from_k_id = :k_id
         and kc_amount_approved > 0
         into c_id;
         select C_LAST_BONUS_STATUS
         from HP_KONTSTATE_CLIENT_STATUS(:c_id, :kf_disb_date)
         into :C_LAST_BONUS_STATUS;
       end
       if (:C_LAST_BONUS_STATUS is null) then C_LAST_BONUS_STATUS = -1;
    --         Проверка процентной ставки и срок кредита
       if (C_LAST_BONUS_STATUS = 1) then
       begin
         if (k_percent <> 5.5) then
         begin
           res_error_text = 'Процентная ставка неправильно указан, у данного контракта статус Бронзовый';
           res_error_is_critical = 1;
           suspend;
         end 
         if (not(k_kolich_repay between 1 and 3)) then
         begin
           res_error_text = 'Срок кредита неправильно указан, у данного контракта статус Бронзовый';
           res_error_is_critical = 1;
           suspend;
         end 
       end else
       if (C_LAST_BONUS_STATUS = 2) then
       begin
         if (k_percent <> 5.0) then
         begin
           res_error_text = 'Процентная ставка неправильно указан, у данного контракта статус Серебряный';
           res_error_is_critical = 1;
           suspend;
         end 
         if (not(k_kolich_repay between 1 and 4)) then
         begin
           res_error_text = 'Срок кредита неправильно указан, у данного контракта статус Серебряный';
           res_error_is_critical = 1;
           suspend;
         end 
       end else
       if (C_LAST_BONUS_STATUS = 3) then
       begin
         if (k_percent <> 4.0) then
         begin
           res_error_text = 'Процентная ставка неправильно указан, у данного контракта статус Золотой';
           res_error_is_critical = 1;
           suspend;
         end 
         if (not(k_kolich_repay between 1 and 6)) then
         begin
           res_error_text = 'Срок кредита неправильно указан, у данного контракта статус Золотой';
           res_error_is_critical = 1;
           suspend;
         end 
       end else
       if (C_LAST_BONUS_STATUS = 4) then
       begin
         if (k_percent <> 3.5) then
         begin
           res_error_text = 'Процентная ставка неправильно указан, у данного контракта статус Бриллиантовый';
           res_error_is_critical = 1;
           suspend;
         end 
         if (not(k_kolich_repay between 1 and 9)) then
         begin
           res_error_text = 'Срок кредита неправильно указан, у данного контракта статус Бриллиантовый';
           res_error_is_critical = 1;
           suspend;
         end 
       end else
       if (C_LAST_BONUS_STATUS >= 5) then
       begin
         if (k_percent <> 2.5) then
         begin
           res_error_text = 'Процентная ставка неправильно указан, у данного контракта статус Бриллиантовый';
           res_error_is_critical = 1;
           suspend;
         end 
         if (not(k_kolich_repay between 1 and 9)) then
         begin
           res_error_text = 'Срок кредита неправильно указан, у данного контракта статус Бриллиантовый';
           res_error_is_critical = 1;
           suspend;
         end 
       end else
       begin
         res_error_text = 'Основной кредит без статуса.';
         res_error_is_critical = 1;
         suspend;
       end
  --         Проверка каждого клиента     --------------------------------------------------------------------------------------
           for
           select kc_amount_approved, c_name_fio, c_pass_no, kc_from_c_id
           from knclient, clients
           where kc_from_k_id = :k_id
           and kc_amount_approved > 0
           and kc_from_c_id = c_id
           into kc_amount_approved, res_info_text, kf_client_pass_no, c_id
           do
           begin
             if (C_LAST_BONUS_STATUS = 1) then
             begin
               if (kc_amount_approved <> 5000) then
               begin
                 res_error_text = 'Сумма клиента не правильно указан, статус контракта Бронзовый';
                 res_error_is_critical = 1;
                 suspend;
               end 
             end else
             if (C_LAST_BONUS_STATUS = 2) then
             begin
               if (not(kc_amount_approved between 5000 and 10000)) then
               begin
                 res_error_text = 'Сумма клиента не правильно указан, статус контракта Серебряный';
                 res_error_is_critical = 1;
                 suspend;
               end 
             end else
             if (C_LAST_BONUS_STATUS = 3) then
             begin
               if (not(kc_amount_approved between 5000 and 15000)) then
               begin
                 res_error_text = 'Сумма клиента не правильно указан, статус контракта Золотой';
                 res_error_is_critical = 1;
                 suspend;
               end 
             end else
             if (C_LAST_BONUS_STATUS in (4,5)) then
             begin
               if (not(kc_amount_approved between 5000 and 20000)) then
               begin
                 res_error_text = 'Сумма клиента не правильно указан, статус контракта Бриллиантовый';
                 res_error_is_critical = 1;
                 suspend;
               end 
             end
--  Проверка, клиент не находится ли у других в поручителях
             if (k_kolich_client_approved = 1) then
             begin
                k_kontract_no = null;
                select k_kontract_no
                from kontstate, kontract, product, zalog
                where ks_ol_is_delete = 0
                and ks_state = 1
                and :kf_disb_date between ks_date_begin and ks_date_end
                and ks_from_k_id = k_id
                and k_from_pr_id = pr_id
                and pr_loan_type = 1
                and k_id = z_from_k_id
                and z_type = 0
                and z_from_c_id = :kc_from_c_id
                rows 1
                into k_kontract_no;
                if (k_kontract_no is not null) then
                begin
                  res_error_is_critical = 1;
                  res_error_text = 'Клиент поручитель контракта №'||k_kontract_no||', выдача запрещено!';
                  suspend;
                end 
              end
           end
  end
end
