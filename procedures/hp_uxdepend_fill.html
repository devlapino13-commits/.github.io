CREATE PROCEDURE HP_UXDEPEND_FILL(
    UX_CODE D_STR_CODE,
    MAX_LEVEL D_INTEGER = 5,
    NEED_CLEAR D_BOOL = 0)
AS
declare variable tmp_ux_code d_str_code;
declare variable res_image_id d_integer;
begin
  if (need_clear is null) then
    need_clear = 0;

  if (need_clear = 1) then
  begin
    delete from uxdepend
    where udp_parent_code is null;
  end

  if (exists(select * from uxcode
  where ux_code = :ux_code)) then
    execute procedure hp_uxdepend_fill_tree(:ux_code, '', null, 0, :max_level);
  else
  begin
    -- таблица
    select res_image_id
    from hp_uxkind_image_id('table', '')
    into :res_image_id;

    insert into uxdepend (udp_owner_code, udp_parent_code, udp_ux_code, udp_ux_kind, udp_image_id)
    values (:ux_code, null, :ux_code, 'table', :res_image_id);

    for select ux_code
    from uxcode
    where ux_table_name = :ux_code
    order by iif(ux_kind in (0, 1), 1 - ux_kind, ux_kind)
    into tmp_ux_code
    do
      execute procedure hp_uxdepend_fill_tree(:tmp_ux_code, :ux_code, null, 0, :max_level);
  end
end
