CREATE PROCEDURE AAREP_CHECK_PRODUCT_ACC_TMP
RETURNS (
    PR_ID INTEGER,
    ACC_SET VARCHAR(1024))
AS
declare variable reg_id integer;
declare variable acc_type integer;
declare variable acc_no varchar(20);
begin
  for select distinct k_from_pr_id
      from kontstate, kontract
      where k_id = ks_from_k_id and
            ks_date_begin <= '31.12.2013' and
            ks_date_end >= '31.12.2013' and
            ks_state = 1 and
            ks_ol_is_delete = 0
      into :pr_id
  do
  begin
    for select reg_unicode
        from region
        where reg_status = 3
        into :reg_id
    do
    begin
      acc_set = '';
      for select sa_account_type, substring(coalesce(sa_account, '') from 5)
          from standacc
          where sa_account_type in (1, 2, 3, 5) and
                sa_from_pr_id = :pr_id and
                sa_from_reg_id = :reg_id
          order by sa_account_type
          into :acc_type, :acc_no
      do
      begin
        if (acc_no <> '') then
        begin
          if (acc_set <> '') then
            acc_set = acc_set || ', ' || :acc_type || '-' || :acc_no;
          else
            acc_set = :acc_type || '-' || :acc_no;
        end
      end
      if (acc_set <> '') then
        suspend;
    end
  end
end
