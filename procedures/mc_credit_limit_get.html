CREATE PROCEDURE MC_CREDIT_LIMIT_GET(
    MCL_ID D_ID)
RETURNS (
    LIMIT_AMOUNT D_STR_255,
    ERROR_TXT D_STR_255)
AS
declare variable C_ID D_ID;
declare variable KBM_ID D_ID;
declare variable DAYS_QTY D_INTEGER;
declare variable IS_NEW_CLIENT D_ID;
begin
  if (f_get_db_id() not in (39801, 41701)) then
    exit;

  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  select is_new_client
  from hp_is_new_client_for_cred_limit(:c_id)
  into :is_new_client;

  if (is_new_client = 0) then -- если клиент повторник, тогда нет смысла проверять отчет КИБ
  begin
    suspend;
    exit;
  end

  -- получаю id отчета киб, который по сроку валиден, для расчета кредитного лимита
  kbm_id = (select kbm_id from mc_online_loan_get_kib_id(:c_id));

  -- если нет отчета киб, валидного, для расчета кредитного лимита, тогда
  if (kbm_id is null) then
  begin
    days_qty = f_get_const('online_loan_cib_report_valid_days');

    -- проверяю, может есть запрос на получение отчета КИБ, который ожидает обработки
    if (exists(
       select *
       from kibminiapp kaa
       where kaa.kbm_from_c_id = :c_id and
       kaa.kbm_is_checked = 0 and
       cast(kbm_datetime as date) >= current_date - :days_qty and
       coalesce(kaa.kbm_err_text, '') = ''
    )) then
    begin
        -- если запрос есть, нет необходимости создавать задачу на получение отчета КИБ
        error_txt = 'Данные отправлены в КИБ';
        suspend;        
    end
    else
    begin
        --  если запроса на получение отета нет, тогда создаем задачу на получение свежего отчета КИБ
        execute procedure insert_kibminiapp_client(:c_id, 1000000000000);
        error_txt = 'Данные отправлены в КИБ';
        suspend;
    end
  end
end
