CREATE PROCEDURE AAREP_KREDIT_BALANCE_DIFFERENCE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_K_ID DECIMAL(18,0),
    RES_PR_ID INTEGER,
    RES_PR_NAME VARCHAR(50),
    RES_CURRENCY VARCHAR(3),
    RES_OS_BEGIN DECIMAL(18,4),
    RES_OS_END DECIMAL(18,4),
    RES_OS_DISB DECIMAL(18,4),
    RES_OS_REPAY DECIMAL(18,4),
    RES_BAL_OS_BEGIN DECIMAL(18,4),
    RES_BAL_OS_END DECIMAL(18,4),
    RES_BAL_OS_DISB DECIMAL(18,4),
    RES_BAL_OS_REPAY DECIMAL(18,4),
    RES_REG_NAME VARCHAR(100),
    RES_OL_ID DECIMAL(18,0),
    RES_BAL_OS_DIFF DECIMAL(18,4),
    RES_OL_MEMO VARCHAR(1024),
    RES_ACC_DT VARCHAR(20),
    RES_ACC_KT VARCHAR(20))
AS
declare variable res_ol_date date;
declare variable res_ok_type integer;
declare variable res_os decimal(18,4);
declare variable res_acc varchar(20);
declare variable res_region_id integer;
declare variable res_bal_os_dt decimal(18,4);
declare variable res_bal_os_kt decimal(18,4);
begin
  res_ol_memo = null;
  res_acc_dt = null;
  res_acc_kt = null;
  res_bal_os_diff = 0;
  for select k_id, pr_id, pr_name, k_currency
      from kontract, product, kontstate
      where k_from_pr_id = pr_id and
            ks_from_k_id = k_id and
            cast(ks_date_begin as date) <= :rep_date_end and
            ks_date_end >= :rep_date_begin and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            pr_is_employer = 0
      into :res_k_id, :res_pr_id, :res_pr_name, :res_currency
  do
  begin
    res_os_begin = 0;
    res_os_end = 0;
    res_os_disb = 0;
    res_os_repay = 0;
    res_bal_os_begin = 0;
    res_bal_os_end = 0;
    res_bal_os_disb = 0;
    res_bal_os_repay = 0;
    for select ok_ol_date, ok_ol_reg_id, ok_from_ol_id, ok_type, sum(ok_os), reg_code || ' ' || reg_name
        from operkred, region
        where ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date_end and
              ok_from_k_id = :res_k_id and
              ok_type <= 8 and
              ok_from_c_id = 0
        group by ok_ol_date, ok_ol_reg_id, ok_from_ol_id, ok_type, reg_code, reg_name
        having sum(ok_os) <> 0
        into :res_ol_date, :res_region_id, :res_ol_id, :res_ok_type, :res_os, :res_reg_name
    do
    begin
      if (res_os is null) then
        res_os = 0;
      res_acc = '';
      select res_account
      from hp_make_kredit_acclist(:res_pr_id, :res_region_id, 1) -- Кредитный портфель
      into :res_acc;

      res_bal_os_dt = 0;
      res_bal_os_kt = 0;
      if (res_acc <> '') then
      begin
        select sum(ob_summa_val)
        from operbook
        where ob_from_ol_id = :res_ol_id and
              ob_account_dt = :res_acc
        into :res_bal_os_dt;
        if (res_bal_os_dt is null) then
          res_bal_os_dt = 0;
        res_bal_os_dt = cast(res_bal_os_dt as decimal(15,2));

        select sum(ob_summa_val)
        from operbook
        where ob_from_ol_id = :res_ol_id and
              ob_account_kt = :res_acc
        into :res_bal_os_kt;
        if (res_bal_os_kt is null) then
          res_bal_os_kt = 0;
        res_bal_os_kt = cast(res_bal_os_kt as decimal(15,2));
      end
      if (res_ol_date < rep_date_begin) then
      begin
        res_os_begin = res_os_begin + res_os;
        res_bal_os_begin = res_bal_os_begin + :res_bal_os_dt - :res_bal_os_kt;
      end
      if (res_ol_date <= rep_date_end) then
      begin
        res_os_end = res_os_end + res_os;
        res_bal_os_end = res_bal_os_end + :res_bal_os_dt - :res_bal_os_kt;
      end
      if ((res_ol_date >= rep_date_begin) and
          (res_ol_date <= rep_date_end)) then
      begin
        if (res_ok_type = 8) then
          res_os_disb = res_os_disb + res_os;
        else
          res_os_repay = res_os_repay + res_os;
        res_bal_os_disb = res_bal_os_disb + :res_bal_os_dt;
        res_bal_os_repay = res_bal_os_repay + :res_bal_os_kt;
      end
    end
    if ((res_os_begin <> res_bal_os_begin) or (res_os_end <> res_bal_os_end) or (res_os_disb <> res_bal_os_disb) or (res_os_repay <> (0 - res_bal_os_repay))) then
      suspend;
  end
  res_os_begin = 0;
  res_os_end = 0;
  res_os_disb = 0;
  res_os_repay = 0;
  res_bal_os_begin = 0;
  res_bal_os_end = 0;
  res_bal_os_disb = 0;
  res_bal_os_repay = 0;
  res_k_id = null;
  res_pr_id = null;
  res_pr_name = null;

  for select distinct res_account, res_region_id
      from hp_make_all_kredit_acclist
      into :res_acc, :res_region_id
  do
  begin
    res_bal_os_diff = 0;
    for select ol_id, reg_code || ' ' || reg_name, ob_summa_val, ol_memo, ob_account_dt, ob_account_kt
        from operbook, region, operlist
        left join operkred on (ok_from_ol_id = ol_id and
              ok_from_c_id = 0)
        where ob_from_ol_id = ol_id and
              ob_ol_is_active = 1 and
              (ob_account_dt = :res_acc or
              ob_account_kt = :res_acc) and
              ob_ol_date between :rep_date_begin and :rep_date_end and
              ob_ol_reg_id = reg_unicode and
              ob_ol_reg_id = :res_region_id and
              ob_summa_val <> 0 and
              ok_from_ol_id is null
        into :res_ol_id, :res_reg_name, :res_bal_os_diff, :res_ol_memo, :res_acc_dt, :res_acc_kt
    do
    begin
      suspend;
    end
  end
end
