CREATE PROCEDURE HP_REP_REVISE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PORTF_CP DECIMAL(18,4),
    PORTF_K_QTY INTEGER,
    PORTF_C_QTY INTEGER,
    AK_CP DECIMAL(18,4),
    AK_K_QTY INTEGER,
    AK_C_QTY INTEGER,
    AC_CP DECIMAL(18,4),
    AC_C_QTY INTEGER,
    CALCP_PAYEDP DECIMAL(18,4),
    CALCP_WRITEOFFP DECIMAL(18,4),
    CALCP_CALCP DECIMAL(18,4),
    COPER_PAYEDP DECIMAL(18,4),
    COPER_WRITEOFFP DECIMAL(18,4),
    OBSV_CP DECIMAL(18,4),
    OBSV_PAYEDP DECIMAL(18,4),
    OBSV_CALCP DECIMAL(18,4))
AS
declare variable db_id integer;
declare variable acc_balance_no varchar(50);
declare variable end_dt decimal(18,4);
declare variable ob_kt decimal(18,4);
declare variable rep_currency varchar(3);
declare variable calcp_cp_temp decimal(18,4);
declare variable calcp_payedp_temp decimal(18,4);
declare variable calcp_writeoffp_temp decimal(18,4);
declare variable calcp_calcp_temp decimal(18,4);
declare variable ob_summa_from_val decimal(18,4);
declare variable reval_sum decimal(18,4);
declare variable year_buf varchar(4);
declare variable month_buf varchar(2);
declare variable total_date integer;
declare variable has_calc smallint;
begin
  portf_cp = 0;
  portf_k_qty = 0;
  portf_c_qty = 0;
  ak_cp = 0;
  ak_k_qty = 0;
  ak_c_qty = 0;
  ac_cp = 0;
  ac_c_qty = 0;
  calcp_payedp = 0;
  calcp_writeoffp = 0;
  calcp_calcp = 0;
  coper_payedp = 0;
  coper_writeoffp = 0;
  obsv_cp = 0;
  obsv_payedp = 0;
  obsv_calcp = 0;

  db_id = f_get_db_id();
  year_buf = extract(year from :rep_date_end);
  month_buf = extract(month from :rep_date_end);

  if (month_buf < 10) then
    month_buf = '0' || month_buf;

  total_date = cast(year_buf || month_buf as integer);

-- проверяю, сформированы ли начисленные % или нет за месяц, дыты конца отчета
  has_calc = 0;
  if (exists(select kcp_period
             from kontcalcperc kcp
             where kcp.kcp_period = :total_date)) then
    has_calc = 1;

-- Отчет - "Отчет о портфеле"
  select sum(aa_amount_active_end), sum(aa_kolich_active_end), sum(aa_kolich_cl_active_end)
  from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 0, 0, 0, 0, 0)
  into :portf_cp, :portf_k_qty, :portf_c_qty;

-- Отчет - "Активные кредиты"
  select sum(rep_active_balance), count(k_id), sum(k_kolich_client_approved)
  from aarep_kontract_list(:rep_region_id, :rep_date_begin, :rep_date_end, 0, '000', 3)
  into :ak_cp, :ak_k_qty, :ak_c_qty;

-- Отчет - "клиенты активных кредитов"
  select sum(knc_rep_active_balance), count(k_id)
  from aarep_kont_clients_list(:rep_region_id, :rep_date_begin, :rep_date_end, 0, '000', 3)
  into :ac_cp, :ac_c_qty;

-- т.к. кред. портфель из отчета "клиенты активных кредитов"
-- всегда отличается от реального кредитного портфеля на небольшое число,
-- установил логику описанную ниже. Согласована с Санжаром:
  if (abs(ac_cp - portf_cp) <= 100) then
    ac_cp = portf_cp;

  if (has_calc = 1) then
  begin
    calcp_writeoffp = 0;
-- начисленые %. Из данного отчета беру только суммы в национальной валюте.
--  Даже если отчет формировать сводный эквивалент валют, все равно отчет будет давать суммы не в эквиваленте, а в разбивке по валютам
    for select rep_currency, sum(aarep_balance_end), sum(aarep_perc_payed), sum(aarep_perc_writeoff),
               sum(aarep_perc_end)
        from aarep_calc_perc_list(:rep_region_id, :rep_date_begin, :rep_date_end, '000')
        group by rep_currency
        into :rep_currency, :calcp_cp_temp, :calcp_payedp_temp, :calcp_writeoffp_temp, :calcp_calcp_temp
    do
    begin
      if ((db_id in (39801, 41701)) and
          (rep_currency = '417')) then -- КГ
      begin
        calcp_payedp = calcp_payedp + calcp_payedp_temp;
        calcp_writeoffp = calcp_writeoffp + calcp_writeoffp_temp;
        calcp_calcp = calcp_calcp + calcp_calcp_temp;
      end
      if ((db_id = 94401) and
          (rep_currency = '944')) then -- АЗ
      begin
        calcp_payedp = calcp_payedp + calcp_payedp_temp;
        calcp_writeoffp = calcp_writeoffp + calcp_writeoffp_temp;
        calcp_calcp = calcp_calcp + calcp_calcp_temp;
      end
      if ((db_id = 81001) and
          (rep_currency = '810')) then -- РФ
      begin
        calcp_payedp = calcp_payedp + calcp_payedp_temp;
        calcp_writeoffp = calcp_writeoffp + calcp_writeoffp_temp;
        calcp_calcp = calcp_calcp + calcp_calcp_temp;
      end
    end

-- По КГ, после получения суммы выплаченных % , необходимо выявить сумму по кредитам в ин. валюте.
-- Для остальных БД активных индексированных кредитов нет.
    if (db_id in (39801, 41701)) then
    begin
      ob_summa_from_val = null;

      select sum(ob_summa)
      from operbook ob
      where ob_ol_is_active = 1 and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            ob.ob_short_acc_dt starting '11520' and
            ob.ob_short_acc_kt starting '21141'
      into :ob_summa_from_val;

      if (ob_summa_from_val is null) then
        ob_summa_from_val = 0;
      calcp_payedp = calcp_payedp + ob_summa_from_val;
    end
  end

-- Отчет - "операции по кредитам"
-- Для КГ имеет смысл формировать, только если есть начисленные %
  if (((db_id in (39801, 41701)) and
      (has_calc = 1)) or (db_id not in (39801, 41701))) then
  begin
    select -sum(payedp), -sum(writeoffp)
    from (select iif(trans_type = 4, trans_perc, 0) as payedp, iif(trans_type = 5, trans_perc, 0) as writeoffp
          from aarep_transkred_list(:rep_region_id, :rep_date_begin, :rep_date_end, '000')
          where trans_type in (4, 5)) a
    into :coper_payedp, :coper_writeoffp;
  end

  /* ОБСВ*/
  if (db_id in (39801, 41701)) then -- КГ
  begin
    /*
        Т.к. в КГ имеются индексированные кредиты,
        нет смысла сверять ОБСВ с остальными отчетами до момента, пока не произвели начисление %.
        Т.к. в ОБС по кредитному портфелю бухгалтерия начисляет процентыный доход, и это делатеся
        раз в месяц, когда начисляются %.
    */

    if (has_calc = 1) then
    begin
      for select acc_balance_no, sum(end_dt) end_dt, sum(ob_kt) ob_kt
          from aarep_account_oborsaldo(:rep_region_id, :rep_date_begin, :rep_date_end, 0, '000')
          where acc_balance_no in ('10931', '10941', -- портфель
                               '11437') -- выплаченные, начисленные %
          group by acc_balance_no
          into :acc_balance_no, :end_dt, :ob_kt
      do
      begin
        if (acc_balance_no in ('10931', '10941')) then
          obsv_cp = obsv_cp + end_dt;
        if (acc_balance_no = '11437') then
        begin
          obsv_payedp = obsv_payedp + ob_kt;

--if (has_calc = 1) then
          obsv_calcp = obsv_calcp + end_dt;
        end
      end

-- т.к. в obsv_payedp ( выплаченные %)сейчас сидит сумма списанных % и пероценка, удаляю эти суммы
-- удаляю списанные %
      if (calcp_writeoffp is null) then
        calcp_writeoffp = 0;
      obsv_payedp = obsv_payedp - calcp_writeoffp;

-- вычисляю сумму переоценки
      reval_sum = 0;
      select sum(ob_summa)
      from operbook ob
      where ob_ol_is_active = 1 and
            ob_ol_date between :rep_date_begin and :rep_date_end and
            ob.ob_short_acc_dt = '11520/1' and
            ob.ob_short_acc_kt starting '11437'
      into :reval_sum;
-- вычитаю сумму переоценки
      if (reval_sum is null) then
        reval_sum = 0;
      obsv_payedp = obsv_payedp - reval_sum;
    end
  end
------------------------------------------------------------------------------

  if (db_id = 94401) then -- АЗ
  begin
    for select acc_balance_no, sum(end_dt) end_dt, sum(ob_kt) ob_kt
        from aarep_account_oborsaldo(:rep_region_id, :rep_date_begin, :rep_date_end, 0, '000')
        where acc_balance_no in ('21110', '21113', '21210', '21213', -- портфель
                             '21112', '21212') -- выплаченные, начисленные %
        group by acc_balance_no
        into :acc_balance_no, :end_dt, :ob_kt
    do
    begin
      if (acc_balance_no in ('21110', '21113', '21210', '21213')) then
        obsv_cp = obsv_cp + end_dt;
      if (acc_balance_no in ('21112', '21212')) then
      begin
        obsv_payedp = obsv_payedp + ob_kt;
        if (has_calc = 1) then
          obsv_calcp = obsv_calcp + end_dt;
      end
    end
  end
------------------------------------------------------------------------------
  if (db_id = 81001) then -- РФ
  begin
    for select acc_balance_no, sum(end_dt) end_dt, sum(ob_kt) ob_kt
        from aarep_account_oborsaldo(:rep_region_id, :rep_date_begin, :rep_date_end, 0, '000')
        where acc_balance_no in ('58-03', -- портфель
                             '76-09') -- выплаченные, начисленные %
        group by acc_balance_no
        into :acc_balance_no, :end_dt, :ob_kt
    do
    begin
      if (acc_balance_no = '58-03') then
        obsv_cp = obsv_cp + end_dt;
      if (acc_balance_no = '76-09') then
      begin
        obsv_payedp = obsv_payedp + ob_kt;
        if (has_calc = 1) then
          obsv_calcp = obsv_calcp + end_dt;
      end
    end
  end

-- индексированные кредиты в КГ могут дать небольшую разницу между отчетами по показателю, "начисленные %"
-- допустимая разница = 0.5 сома. Даный момент согласован с Керимбековым А.
  if (db_id in (39801, 41701)) then
  begin
    if ((abs(obsv_calcp - calcp_payedp)) <= 0.5) then
      obsv_calcp = calcp_payedp;
  end

  suspend;

end
