CREATE PROCEDURE AAREP_OFFICER_CLI_IPF(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(60),
    BRANCH_NAME VARCHAR(60),
    OKRUG_NAME VARCHAR(60),
    KF_OFFICER_NAME VARCHAR(100),
    KOLICH_CLI_BEGIN INTEGER,
    KOLICH_CLI_BEGIN_OWN INTEGER,
    KOLICH_CLI_END INTEGER,
    KOLICH_CLI_END_OWN INTEGER,
    PRIROST_CLI INTEGER,
    PRIROST_CLI_OWN INTEGER,
    DISB_CLI INTEGER,
    DISB_CLI_NEW INTEGER,
    DISB_CLI_NOT_HISTORY INTEGER,
    DISB_CLI_NOT_HISTORY_IPF INTEGER,
    JS_NAME VARCHAR(100))
AS
declare variable kal_new_reg_id integer;
declare variable kal_new_kf_officer id_type;
begin
  for with tabl1
      as (select kal_new_reg_id, kal_new_kf_officer, count(distinct kc_from_c_id) as kolich_cli_begin, count(distinct
                 case
                   when ks_kc_id = kal_new_kf_officer then kc_from_c_id
                 end) as kolich_cli_begin_own,
                 0 as kolich_cli_end, 0 as kolich_cli_end_own
          from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1), knclient
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :rep_date_begin - 1 between ks_date_begin and ks_date_end and
                ks_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                kal_new_reg_id = id_list and
                :rep_date_begin - 1 between kal_date and kal_date_end and
                kc_from_k_id = k_id and
                kc_amount_approved > 0 and
                exists(select *
                       from hp_k_pr_type(ks_from_k_id)
                       where k_ipf = 1)
          group by kal_new_reg_id, kal_new_kf_officer
          union
          select kal_new_reg_id, kal_new_kf_officer, 0 as kolich_cli_begin, 0 as kolich_cli_begin_own,
                 count(distinct kc_from_c_id) as kolich_cli_end, count(distinct
                 case
                   when ks_kc_id = kal_new_kf_officer then kc_from_c_id
                 end) as kolich_cli_end_own
          from kontstate, kontract, knaktlist, region_list(:rep_region_id, 1, 3, 1), knclient
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                :rep_date_end between ks_date_begin and ks_date_end and
                ks_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                knaktlist.is_delete = 0 and
                kal_new_reg_id = id_list and
                :rep_date_end between kal_date and kal_date_end and
                kc_from_k_id = k_id and
                kc_amount_approved > 0 and
                exists(select *
                       from hp_k_pr_type(ks_from_k_id)
                       where k_ipf = 1)
          group by kal_new_reg_id, kal_new_kf_officer)
      select kal_new_reg_id, kal_new_kf_officer, sum(kolich_cli_begin), sum(kolich_cli_begin_own), sum(kolich_cli_end),
             sum(kolich_cli_end_own)
      from tabl1
      group by kal_new_reg_id, kal_new_kf_officer
      into kal_new_reg_id, kal_new_kf_officer, kolich_cli_begin, kolich_cli_begin_own, kolich_cli_end,
           kolich_cli_end_own
  do
  begin
    office_name = null;
    branch_name = null;
    okrug_name = null;
    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
    from region r1, region r2, region r3
    where r1.reg_unicode = :kal_new_reg_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into office_name, branch_name, okrug_name;
    kf_officer_name = null;
    select c_name_fio
    from clients
    where c_id = :kal_new_kf_officer
    into :kf_officer_name;

    prirost_cli = kolich_cli_end - kolich_cli_begin;
    prirost_cli_own = kolich_cli_end_own - kolich_cli_begin_own;
    disb_cli = null;
    disb_cli_new = null;
    disb_cli_not_history = null;
    select count(distinct kc_from_c_id), count(distinct
           case
             when kc_circle <= 1 then kc_from_c_id
           end),
           count(distinct
           case
             when kc_circle <= 1 then case
                                        when not exists(select 1
                                                        from kibminiapp, kibhistory
                                                        where kbm_from_c_id = kc_from_c_id and
                                                              kh_from_kbm_id = kbm_id and
                                                              kh_is_our = 0) then kc_from_c_id
                                      end
           end)
    from kontstate, knclient
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :kal_new_kf_officer and
          ks_region_id = :kal_new_reg_id and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          exists(select *
                 from hp_k_pr_type(ks_from_k_id)
                 where k_ipf = 1)
    into disb_cli, disb_cli_new, disb_cli_not_history;
    if (disb_cli is null) then
      disb_cli = 0;
    if (disb_cli_new is null) then
      disb_cli_new = 0;
    if (disb_cli_not_history is null) then
      disb_cli_not_history = 0;
--  Вычисляет системный должность Сотрудника
    js_name = null;
    select rd_name
    from hr_pstatus, joblist, jobkind, refdata
    where ps_from_c_id = :kal_new_kf_officer and
          ps_is_working = 1 and
          ps_status = 1 and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_id = rd_id
    into js_name;
    if (js_name is null) then
      js_name = 'не определен';

    suspend;
  end
end
