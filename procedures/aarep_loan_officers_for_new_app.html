CREATE PROCEDURE AAREP_LOAN_OFFICERS_FOR_NEW_APP(
    REG_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    KE_FROM_P_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    KE_CODE VARCHAR(20),
    KE_FROM_REG_ID INTEGER,
    KE_OPER_DAYS INTEGER)
AS
declare variable st_staff integer;
declare variable st_client integer;
declare variable reg_code varchar(20);
declare variable is_hq_user integer;
declare variable un_name varchar(30);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  ke_from_reg_id = :reg_id;
  for select distinct ke_from_p_id, c_name_fio, ke_code, ke_oper_days
      from krexpert, clients
      where ke_from_p_id = c_id and
            ke_status = 1 and
            ke_date <= :rep_date and
            ke_date_end >= :rep_date and
            ke_is_krs <= 2 and
            ke_from_reg_id = :reg_id
      order by c_name_fio
      into :ke_from_p_id, :c_name_fio, :ke_code, :ke_oper_days
  do
  begin
    suspend;
  end

  un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN');

  is_hq_user = 0;
  if (exists(select *
             from username
             where un_curr_reg_id = f_hq_id() and
                   un_name = :un_name)) then
    is_hq_user = 1;

-- Виртуальный сотрудник
  if (f_get_db_id() = 81001) then
  begin
    select reg_code
    from region
    where reg_unicode = :reg_id
    into :reg_code;
    reg_code = cast(left(reg_code, 2) as integer) || '-' || cast(right(reg_code, 2) as integer);

    ke_from_p_id = null;
    for select c_id, c_name_fio
        from clients
        where c_id in (1000000000004)
        into :ke_from_p_id, :c_name_fio
    do
    begin
      ke_code = reg_code || '-0';
      ke_oper_days = 127;
      suspend;
    end
    if (is_hq_user = 1) then
    begin
      ke_from_p_id = null;
      for select c_id, c_name_fio
          from clients
          where c_id in (1000000000002)
          into :ke_from_p_id, :c_name_fio
      do
      begin
        ke_code = reg_code || '-БС';
        ke_oper_days = 127;
        suspend;
      end
    end
  end
end
