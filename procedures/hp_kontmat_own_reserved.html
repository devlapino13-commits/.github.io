CREATE PROCEDURE HP_KONTMAT_OWN_RESERVED(
    BILL_ID D_ID,
    BILL_LINE_NO D_INTEGER,
    WP_ID D_ID,
    MOTV_ID D_ID,
    DT D_DATE = current_date,
    C_ID D_ID = null,
    ML_ID D_ID = null)
RETURNS (
    QUANTITY D_CURRENCY,
    TOTAL D_CURRENCY)
AS
begin
  select
    sum(kmo_quantity),
    sum(kmo_total)
  from kontmat_own
  where kmo_bill_id = :bill_id
    and kmo_bill_line_no = :bill_line_no
    and kmo_wp_id = :wp_id
    and kmo_motv_id = :motv_id
    and kmo_date <= :dt
    and (kmo_ol_id is null or exists(select * from operlist where ol_id = kmo_ol_id and is_delete = 1))
    and kmo_c_id = coalesce(:c_id, kmo_c_id)
    and kmo_ml_id = coalesce(:ml_id, kmo_ml_id)
  group by
    kmo_c_id,
    kmo_wp_id,
    kmo_motv_id,
    kmo_bill_id,
    kmo_bill_line_no,
    kmo_ml_id,
    kmo_ol_id
  into
    :quantity,
    :total;
  quantity = coalesce(quantity, 0);
  total = coalesce(total, 0);
  suspend;
end
