CREATE PROCEDURE FSROBOT_SMS_DISMISSED_LO(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable OFFICE_PHONE varchar(100);
declare variable KA_DATE D_DATE;
declare variable DB_ID D_INTEGER;
declare variable CL_NAME STR_100;
declare variable CL_NAME_RUS STR_100;
declare variable KAL_NEW_KF_OFFICER ID_TYPE;
declare variable SVZ_NAME STR_100;
declare variable SVZ_NAME_RUS STR_100;
declare variable SVZ_PHONE STR_100;
declare variable PUSH_BODY blob sub_type 1 segment size 80;
declare variable MSG_KIND D_STR_10;
declare variable MCL_PROJECT integer;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды, ID-Акт прием-передачи
-- Rep_UN_ID - пользователь

  res_maillist = :rep_email;

  db_id = f_get_db_id();

  select reg_phone, ka_date
  from region, knakt
  where reg_unicode = ka_from_reg_id and
        ka_id = :rep_params
  into :office_phone, ka_date;

  res_subject = '';

  for select distinct kc_from_c_id,
                      (select cont_value
                       from hp_contdata_get_cell_valid(0, kc_from_c_id)), kal_new_kf_officer
      from knaktlist, kontract, oksummary, knclient
      where kal_from_ka_id = :rep_params and
            kal_from_ka_id > 0 and
            kal_from_kf_id = k_from_kf_id and
            oks_from_k_id = k_id and
            oks_date_begin <= current_date and
            oks_date_end >= current_date and
            oks_balance > 0 and
            oks_close_state = 0 and
            kc_from_k_id = k_id and
            kc_amount_approved > 0
      into :res_c_id, :res_maillist, kal_new_kf_officer
  do
  begin
    if (res_maillist <> '' and db_id <> 81001) then
      suspend;
    if (res_maillist <> '' and db_id = 81001) then
    begin

    -- Определяем проект (М Булак / OiVO) по клиенту
        mcl_project = f_get_mcl_project(f_objkind('clients','id'), :res_c_id);

        if (mcl_project = 1) then     -- М Булак  RFCRU-4257
      begin
    res_body = 'Ваш специалист больше не работает в '||iif(f_get_db_id() = 81001, 'М Булак', 'М Булак')||'. ';
    res_body = blob_append(res_body, 'Чтобы получить повторный заём позвоните по ');
    res_body = blob_append(res_body, 'тел.' || :office_phone || ' или приходите в офис. ');
    res_body = blob_append(res_body, 'Спасибо, что выбрали '||iif(f_get_db_id() = 81001, 'М Булак', 'М Булак')||'!!');
      end
      else
      if (mcl_project = 2) then     -- OiVO  RFCRU-4257
      begin
     res_body = 'Ваш специалист больше не работает в '||iif(f_get_db_id() = 81001, 'OiVO Кредит', 'OiVO Кредит')||'. ';
    res_body = blob_append(res_body, 'Чтобы получить повторный заём позвоните по ');
    res_body = blob_append(res_body, 'тел.' || :office_phone || ' или приходите в офис. ');
    res_body = blob_append(res_body, 'Спасибо, что выбрали '||iif(f_get_db_id() = 81001, 'OiVO Кредит', 'OiVO Кредит')||'!!');
      end
        select (select txt2
                from hp_fio_clear(c_alternat_fio)), (select txt2
                                                     from hp_fio_clear(c_name_fio))
        from clients
        where c_id = :res_c_id
        into cl_name, cl_name_rus;

        select (select txt2
                from hp_fio_clear(c_alternat_fio)), (select txt1 || ' ' || txt2
                                                     from hp_fio_clear(c_name_fio)), (select cont_value
                                                                                      from hp_contdata_get_cell_valid(0, :kal_new_kf_officer))
        from clients
        where c_id = :kal_new_kf_officer
        into svz_name, svz_name_rus, svz_phone;

        msg_kind = f_get_msg_kind(res_c_id, 'fsrobot_sms_dismissed_lo', '', db_id);
        if (f_clientcom_allowed(res_c_id, 1, 1, 1) = 1) then
        begin
          if ((msg_kind = 'sms') and (mcl_project = 1)) then     -- М Булак  RFCRU-4257
          begin
            res_body = cl_name || ', s ' || f_datetostr(ka_date) || ' s vami budet rabotat specialist ' || svz_name || '! Po vsem voprosam zvonite po nomeru +' || svz_phone || '! S zabotoi o vas, M Bulak';
            suspend;
          end
          else
          if ((msg_kind = 'sms')  and (mcl_project = 2)) then     -- OiVO  RFCRU-4257
          begin
            res_body = cl_name || ', s ' || f_datetostr(ka_date) || ' s vami budet rabotat specialist ' || svz_name || '! Po vsem voprosam zvonite po nomeru +' || svz_phone || '! S zabotoi o vas, OiVO Kredit';
            suspend;
          end
          if ((msg_kind = 'push') and (mcl_project = 1)) then     -- М Булак  RFCRU-4257
          begin
            push_body = 'Здравствуйте, ' || cl_name_rus || '. С ' || f_datetostr(ka_date) || ' с вами будет работать специалист ' || svz_name_rus || '! По всем вопросам звоните по номеру +' || coalesce(svz_phone, '') || '! С заботой о вас, М Булак';
            execute procedure mc_send_push('clients', :res_c_id, 'Новый Кредитный Специалист', :push_body, 35);
          end
          else
          if ((msg_kind = 'push') and (mcl_project = 2)) then     --  OiVO  RFCRU-4257
          begin
            push_body = 'Здравствуйте, ' || cl_name_rus || '. С ' || f_datetostr(ka_date) || ' с вами будет работать специалист ' || svz_name_rus || '! По всем вопросам звоните по номеру +' || coalesce(svz_phone, '') || '! С заботой о вас, OiVO Кредит';
            execute procedure mc_send_push('clients', :res_c_id, 'Новый Кредитный Специалист', :push_body, 35);
          end
        end
    end
  end
end
