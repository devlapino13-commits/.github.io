CREATE PROCEDURE FT_FIO_CHANGE_HIST(
    C_ID D_ID,
    SHOW_ALL D_SMALLINT = 0)
RETURNS (
    RES_NAME_FIO D_STR_255,
    RES_DATETIME D_DATETIME)
AS
declare variable c_name_fio d_str_255;
declare variable c_datetime d_datetime;
begin
  -- Первое значение res_name_fio не возвращаем, пустая строка служит маркером
  res_name_fio = '';

  for with
    tab as (
      select c_name_fio, c_datetime
      from clients
      where c_id = :c_id
      union
      select cll_name_fio, cll_datetime
      from clientlog
      where cll_from_c_id = :c_id
        and cll_name_fio not similar to '%[[:DIGIT:]]%' -- исключаем из лога временные правки, типа 'Осмонова Калипа1'
      union
      select 'конец выборки', current_timestamp  -- это маркер конца цикла, он также не возвращается
      from rdb$database
      )
    select c_name_fio, c_datetime
    from tab
    order by c_datetime
  into c_name_fio, c_datetime
  do
  begin
    if (res_name_fio <> c_name_fio) then
    begin
      if (res_name_fio <> '') then
        suspend;
      res_name_fio = c_name_fio;
      res_datetime = c_datetime;
    end
  end
end
