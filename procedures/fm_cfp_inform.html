CREATE PROCEDURE FM_CFP_INFORM(
    MAC_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_BLOB_TEXT)
AS
declare variable c_id d_id;
declare variable reg_id d_id;
declare variable mac_amount d_currency;
declare variable kf_name d_str_100;
declare variable lo_id d_id;
declare variable lo_fio d_str_255;
declare variable c_name_fio d_str_255;
declare variable reg_name d_str_255;
declare variable acl_id d_id;
begin
  if(coalesce(mac_id, 0) < 0) then    -- RFCKG-5565
  begin
    acl_id = mac_id * -1;

    select acl.acl_object_id
    from actionlist acl
    where acl.acl_id = :acl_id
    into mac_id;
  end

  for select mac.mac_from_c_id, ma.ma_office_id, mac.mac_amount, kal.kal_new_kf_name, kal.kal_new_kf_officer
      from miniappclient mac
      inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id and
            ma.ma_is_active = 1 and
            current_date between ma.ma_date_app and ma.ma_date_end
      inner join knaktlist kal on kal.kal_from_kf_id = ma.ma_from_kf_id and
            current_date between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      where mac.mac_id = :mac_id
      into c_id, reg_id, mac_amount, kf_name, lo_id
  do
  begin
    select c.c_name_fio
    from clients c
    where c.c_id = :lo_id
    into lo_fio;

    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into c_name_fio;

    select reg.reg_name
    from region reg
    where reg.reg_unicode = :reg_id
    into reg_name;

    res_html = (select res_style
                from fm_html_styles) || f_format('<body><a name="top"></a><h2>%1</h2><br>', :c_name_fio);
    res_html = blob_append(res_html, f_format('Офис: %1<br>', :reg_name));
    res_html = blob_append(res_html, f_format('Кредитная папка: %1<br>', :kf_name));
    res_html = blob_append(res_html, f_format('Сумма займа: %1<br>', :mac_amount));
    res_html = blob_append(res_html, f_format('Специалист: %1<br>', :lo_fio));
    suspend;
  end
end
