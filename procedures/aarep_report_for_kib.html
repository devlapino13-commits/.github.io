CREATE PROCEDURE AAREP_REPORT_FOR_KIB(
    REP_DATE_END DATE)
RETURNS (
    C_ID ID_TYPE,
    CREL_OTHER_C_ID ID_TYPE,
    C_NAME_FIO VARCHAR(100),
    C_FROM_FL_ID ID_TYPE,
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_INN VARCHAR(20),
    FL_NAME1 VARCHAR(100),
    FL_NAME2 VARCHAR(100),
    FL_NAME3 VARCHAR(100),
    C_SEX_STR VARCHAR(10),
    CREL_OTHER_C_ID_STR VARCHAR(20),
    ROW_NUM INTEGER)
AS
begin
  row_num = 0;
  if (extract(day from :rep_date_end) = 1) then
    rep_date_end = rep_date_end - 1; -- отчет будет отправляться по первым числам, дата отчета последний день месяца
  for select distinct kc_from_c_id as c_id
      from kontstate, knclient
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0
      into c_id
  do
  begin

    for select crel_other_c_id
        from hp_client_relation_list(:c_id, 0)
        where crel_from_rel_id in ('1', '2') and
              crel_is_active = 1
        into crel_other_c_id
    do
    begin
      crel_other_c_id_str = crel_other_c_id;

      select c_name_fio, c_from_fl_id, c_birth_date, trim(c_pass_no), c_inn, iif(c_sex = 0, 'Ж', 'М')
      from clients
      where c_id = :crel_other_c_id
      into c_name_fio, c_from_fl_id, c_birth_date, c_pass_no, c_inn, c_sex_str;

      select fl_name1, fl_name2, fl_name3
      from fiolist
      where fl_id = :c_from_fl_id
      into fl_name1, fl_name2, fl_name3;

      if (c_name_fio containing 'уулу ') then
        fl_name1 = fl_name1 || ' УУЛУ';
      if (c_name_fio containing 'кызы ') then
        fl_name1 = fl_name1 || ' КЫЗЫ';

      row_num = row_num + 1;
      suspend;
    end
  end
end
