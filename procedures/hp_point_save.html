CREATE PROCEDURE HP_POINT_SAVE(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
AS
declare variable ma_id d_id;
declare variable mac_id d_id;
declare variable c_id d_id;
begin
  -- RFCRU-5380
  -- процедура фиксирует баллы клиента на определенный момент
  if (object_kind = 11) then
    ma_id = object_id;
  else if (object_kind = 9) then
    select k_from_ma_app_id
    from kontract 
    where k_id = :object_id
    into ma_id;
  else 
    exception e_default 'Unknown object_kind = ' || coalesce(object_kind, '<NULL>');

  for select mac_id, mac_from_c_id
  from miniappclient
  where mac_from_ma_id = :ma_id
  into mac_id, c_id
  do
    update or insert into miniappclient_data (mcd_from_mac_id, mcd_point)
    values (:mac_id, f_point(:c_id, current_date))
    matching (mcd_from_mac_id);

end
