CREATE PROCEDURE AAREP_SMS_DELINQ(
    REP_DATE DATE,
    REP_DAYS INTEGER)
RETURNS (
    PHONE VARCHAR(20),
    SMS_MESSAGE VARCHAR(255),
    CLIENT_ID DECIMAL(18,0),
    CL_NAME STR_100)
AS
declare variable DB_ID integer;
declare variable K_ID decimal(18,0);
declare variable REP_AMOUNT varchar(20);
declare variable CURR varchar(10);
declare variable NSD_PERCENT decimal(15,4);
declare variable MBF_PHONE varchar(20);
declare variable DATE_FORMAT_STR varchar(20);
declare variable C_ALTERNAT_FIO STR_100;
declare variable CELL_MAIN_RD_ID integer;
declare variable IS_MAIN D_SMALLINT;
begin

  db_id = f_get_db_id();

  if (db_id = 81001) then
    curr = ' руб.';
  if (db_id in (39801, 41701)) then
    curr = ' сом';
  if (db_id = 94401) then
    curr = ' AZN';

  nsd_percent = 0;
  select coalesce(nsd_percent / 100, 0)
  from nspconfig_data
  where nsd_from_nsm_id = -1 and
        (:rep_date + :rep_days) between nsd_date_begin and nsd_date_end
  into nsd_percent;
  if (nsd_percent is null) then
    nsd_percent = 0;

  date_format_str = right('0' || extract(day from rep_date - rep_days), 2) || '.' || right('0' || extract(month from rep_date - rep_days), 2) || '.' || extract(year from rep_date - rep_days);

  for select oks.oks_from_k_id
  from oksummary oks
  where :rep_date - oks.oks_delinq_date >= :rep_days
  and oks_close_state = 0
  and :rep_date between oks.oks_date_begin and oks.oks_date_end
  into :k_id
  do
  begin
      if (exists (
          select ok_from_k_id
          from operkred, kontstate
          where ks_from_k_id = ok_from_k_id and
                ks_ol_is_delete = 0 and
                ks_date_begin <= :rep_date and
                ks_date_end >= :rep_date and
                ks_state = 1 and
                ok_ol_is_active = 1 and
                ok_sh_date = :rep_date - cast(:rep_days as integer) and
                ok_type <> 8 and
                ok_type <> 7 and
                ok_from_c_id = 0 and
                ok_from_k_id = :k_id
          group by ok_from_k_id
          having sum(ok_sh_os + ok_os + ok_sh_perc + ok_perc) > 0
          )) then
      begin
        select sum(round(zad_os + sh_calc_perc + sh_calc_perc * :nsd_percent + sh_calc_fine)) delinq_sum
        from aarep_make_zad_list_temp(:k_id, :rep_date, 1)
        where sh_date = :rep_date - :rep_days
        into rep_amount;
    
        rep_amount = cast(rep_amount as decimal(15,0));
        for select f_cont_info('clients', kc_from_c_id, 'cell_main', 'ok'), kc_from_c_id
            from knclient
            where kc_from_k_id = :k_id and
                  kc_amount_approved > 0
            into phone, client_id
        do
        begin
          if ((db_id = 81001) and
              (rep_days = 1) and
              (phone = '')) then -- RF и первый день просрочки
          begin
            select rd.rd_id, iif(rd.rd_code = 'ContactInfo.Cell_Main', 1, 0)
            from refdata rd
            where rd.rd_code = 'ContactInfo.Cell_Main'
            into :cell_main_rd_id, is_main;
    
            if (f_get_const('new_contdata') = 0) then
            begin
              select cff.cv_contact_value
              from contverify cff
              where cff.cv_object_kind = 0 and
                    cff.cv_object_id = :client_id and
                    cff.cv_from_rd_id = :cell_main_rd_id and
                    cff.cv_status > 0
              order by cv_id desc
              rows 1
              into phone;
            end else
            begin
              select f_cont_code(cda_from_clt_id)
              from contdata, contveri
              where cda_from_c_id = :client_id and
                    cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and
                    cda_is_main = :is_main and
                    cda_id = cvr_from_cda_id and
                    cvr_status = 1
              order by cvr_id desc
              rows 1
              into phone;
            end
    
            phone = coalesce(phone, '');
          end
    
          if (phone <> '') then
          begin
            if (db_id in (39801, 41701)) then -- KG
            begin
              sms_message = 'Ваш долг от ' || date_format_str || 'г.до сих пор не погашен. Сумма:' || rep_amount || 'сом';
    
              suspend;
            end
            else
            if (db_id = 81001) then -- RF
            begin
              cl_name = '';
    
              select cl.c_alternat_fio
              from clients cl
              where cl.c_id = :client_id
              into :c_alternat_fio;
    
              select txt2
              from hp_fio_clear(:c_alternat_fio)
              into :cl_name;
    
              cl_name = coalesce(cl_name, '');
              if (cl_name = '') then
                cl_name = 'klient';

              mbf_phone = '88005557171';
    
              sms_message = '';
    
              if (rep_days = 1) then
                sms_message = 'Vnimanie! Uvajaemyi ' || cl_name || ', srok oplaty prosrochen. ' || 'Сhtoby izbejat lishnih rashodov prosim oplatit ' || :rep_amount || ' rub. S zabotoi o Vas, M Bulak';
              else
              if (rep_days = 2) then
                sms_message = 'Uvajaemyi ' || cl_name || ', srok oplaty prosrochen. Сhtoby izbejat lishnih rashodov prosim vnesti plateg. S zabotoi o Vas, M Bulak!';
              else
              if (rep_days between 4 and 6) then
                sms_message = 'Vnimanie! Uvajaemyi ' || cl_name || '! Vash dolg uvelichivaetsa, svyajites s nami. T.' || mbf_phone || ' My gotovy Vam pomoch. S zabotoi o Vas, M Bulak!';
              else
              if (rep_days = 7) then
                sms_message = 'Uvajaemyi ' || cl_name || '! Vashe delo peredano v Slujbu Vzyskaniy. My pytaemsya Vam POMOCH. Pojaluista, svyajites s nami T.' || mbf_phone || ' Vash M Bulak';
              else
              if (rep_days = 10) then
                sms_message = 'Uvajaemyi ' || cl_name || '! My gotovy Vam POMOCH. Ne narushayte usloviya dogovora. Pojaluista, svyajites s nami T.' || mbf_phone || '. Vash, M Bulak!';
              else
              if (rep_days = 27) then
                sms_message = 'Uvajaemyi ' || cl_name || '! V sluchae neoplaty dolga, summa zajma mojet byt potrebovana dosrochno v ramkah prinuditelnogo vzyskaniya. Svyajites s nami, my gotovy Vam POMOCH T.' || mbf_phone || ' S zabotoi o Vas, M Bulak!';
              suspend;
            end
            else
            if (db_id = 94401) then -- AZ
            begin
              sms_message = 'MBulaka odenishiniz ' || :rep_days || ' gun gecikir. Bugune olan gecikme borcunuz ' || :rep_amount || :curr || '. Gecikmede olan her gune cerime hesablanir';
              suspend;
            end
          end
        end
     end
  end
end
