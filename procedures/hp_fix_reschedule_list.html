CREATE PROCEDURE HP_FIX_RESCHEDULE_LIST(
    REP_DATE D_DATE,
    REP_K_ID D_ID = null)
RETURNS (
    K_ID D_ID,
    KONTRACT_NO STR_100,
    OL_DATE D_DATE,
    SH_ID D_INTEGER,
    SH_DATE D_DATE,
    NEED_FIX D_BOOL,
    SQL_SCRIPT D_STR_255)
AS
declare variable is_found d_bool;
declare variable ok_sh_id d_integer;
declare variable ok_sh_date d_date;
declare variable zad_os d_currency;
declare variable zad_perc d_currency;
declare variable zad_all_last d_currency;
declare variable zad_all_prior d_currency;
declare variable zad_no_grace d_currency;
declare variable sh_count d_integer;
declare variable grace_found d_bool;
declare variable balance d_currency;
begin
  if (rep_date is null) then
    rep_date = '01.04.2026';

  for select distinct ok_from_k_id
      from operkred
      where ok_type = 3 and
            ok_ol_date >= :rep_date and
            ok_sh_date >= :rep_date and
            ok_ol_is_active = 1 and
            ok_sh_os <> 0 and
            coalesce(:rep_k_id, ok_from_k_id) = ok_from_k_id
      into :k_id
  do
  begin
    if (exists(select * from kontstate
               where ks_from_k_id = :k_id and
                     ks_ol_is_delete = 0 and
                     ks_state = 2)) then
      Continue;

    ol_date = null;
    sh_id = null;
    sh_date = null;
    sql_script = '';

    is_found = 0;
    for select sh_id, sh_date
    from schedule
    where sh_from_k_id = :k_id and
          sh_date >= :rep_date and
          sh_grace = 1 and
          extract(month from sh_date) in (1, 2)
    into sh_id, sh_date
    do
    begin
      is_found = 0;
      select ok_ol_date
      from operkred
      where ok_from_k_id = :k_id and
            ok_sh_id = :sh_id and
            ok_sh_date = :sh_date and
            ok_ol_is_active = 1 and
            ok_type = 3 and
            ok_sh_os <> 0
      order by ok_ol_date
      rows 1
      into ol_date;

      if (ol_date is not null) then
      begin
        is_found = 1;
        break;
      end
    end

    if (ol_date is null or is_found = 0) then
      Continue;

    if (ol_date < rep_date) then
      Continue;

    kontract_no = '';
    select k_kontract_no
    from kontract
    where k_id = :k_id
    into kontract_no;

    zad_all_last = 0;
    zad_all_prior = 0;
    sh_count = 0;
    grace_found = 0;
    balance = 0;
    zad_no_grace = 0;
    sql_script = '';
    for select ok_sh_date, ok_sh_id, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_date <= current_date and
          ok_ol_is_active = 1 and
          ok_type <= 7
    group by ok_sh_date, ok_sh_id
    having sum(ok_os + ok_sh_os) <> 0 or sum(ok_perc + ok_sh_perc) <> 0
    order by ok_sh_date, ok_sh_id
    into ok_sh_date, ok_sh_id, zad_os, zad_perc
    do
    begin
      zad_all_prior = zad_no_grace;
      if (zad_os <> 0) then
        zad_no_grace = zad_os + zad_perc;
      zad_all_last = zad_os + zad_perc;
      sh_count = sh_count + 1;

      if (zad_os = 0 and zad_perc > 0 and extract(month from ok_sh_date) in (1, 2)) then
        grace_found = 1;

      balance = balance + zad_os;
    end

    need_fix = 0;
    if (balance <= 500) then
      sql_script = ' -- игнор: баланс <= 500 руб.';
    else
    if (sh_count = 1) then
      sql_script = ' -- игнор: остался один платеж';
    else
    if (grace_found = 1 and (zad_all_last - zad_all_prior) <= 200) then
      sql_script = ' -- игнор: график уже исправлен, отличие послед.платежа не более 200 руб.';
    else
    begin
      need_fix = 1;
      sql_script = 'execute procedure hp_do_fix_reschedule(' || k_id || ', current_date, ''' ||  f_datetostr(ol_date) || ''', 1); commit;';
    end

    suspend;
  end
end
