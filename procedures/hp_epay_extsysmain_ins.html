CREATE PROCEDURE HP_EPAY_EXTSYSMAIN_INS(
    KPT_ID INTEGER,
    PAY_ACTION STR_100,
    TRANS_ID VARCHAR(20),
    ERROR_TXT STR_255,
    REQUEST_BLOB BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    ANSWER_BLOB BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    ESM_ID ID_TYPE)
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable account varchar(20);
declare variable amount curr_type;
declare variable project str_100;
declare variable status integer;
declare variable un_id id_type;
declare variable rep_date date;
declare variable status_tmp integer;
begin
  error_txt = coalesce(error_txt, '');
  esm_id = -1;
  rep_date = current_date;

  /*
  1: ОПЛАТА
     а) Перед фактической оплатой чз ЭЛСОМ происходит check-запрос к ЭЛ-СОМ
     и если ошибка не возникал, тогда закрываю задачу на оплату на нашей стороне, считаю что платеж прошел успешно (т. KONTPAYTASKS поле KPT_STATUS = 1)
     (для того чтобы при ошибке в БД не получилась ситуация когда мы платеж отправили, но на своей стороне не зафиксировали и из за этого отправили его еще раз)
     при этом формируется запись в EXTSYSMAIN и тип операции check

     б) Затем уже после попытки перечисления денежных средств чз ЭЛ-СОМ
      вне зависимости от того возникла ошибка или нет формирую запись в EXTSYSMAIN с типом операции PAY и
     если ошибка не возникла, тогда ESM_STATUS = 0 иначе ESM_STATUS = -1

  2: ФОРМИРОВАНИЕ ПРОВОДОК
     Есть робот который отрабатывает каждую минуту
     вытягивает все EXTSYSMAIN по проекту elsom со статусом ESM_STATUS = 0
     и строит бух проводки для этих платежей.
  */

  for select kpt.kpt_id, kpt.kpr_from_k_id, kpt.kpt_seller_id, kpt.kpt_account, kpt.kpt_amount, kpt.kpt_project,
             kpt.kpt_from_un_id
      from kontpaytasks kpt
      where kpt.kpt_id = :kpt_id
      into :kpt_id, :k_id, :c_id, :account, :amount, :project, :un_id
  do
  begin
    status = iif(error_txt = '', 1, -1);

    status_tmp = status;
    if ((pay_action = 'pay') and
        (status = 1)) then
      status_tmp = 0;

    insert into extsysmain (esm_trans_id, esm_project, esm_action, esm_date, esm_account, esm_object_kind,
                            esm_object_id, esm_datetime, esm_result_id, esm_status, esm_full_source, esm_amount,
                            esm_from_k_id)
    values (:trans_id, :project, :pay_action, :rep_date, :account, 0, :c_id, current_timestamp, :status_tmp,
            :status_tmp, :request_blob, :amount, :k_id)
    returning esm_id
    into :esm_id;

    insert into extsysusers (esu_from_esm_id, esu_user_id)
    values (:esm_id, :un_id);

    insert into extsysanswer (esa_from_esm_id, esa_message, esa_result_id, esa_status)
    values (:esm_id, :answer_blob, :status, :status);

    /* ¤ намеренно на оперции check закрываю задачу, если она проходит успешно,
       т.к. слудующа¤ операци¤ pay вернет задачу в статус - ошибка, если будет возвращена ошибка от сервиса "ЁЋ—ќћ" */
    update kontpaytasks
    set kpt_from_esm_id = :esm_id,
        kpt_error_text = :error_txt,
        kpt_status = :status
    where (kpt_id = :kpt_id);

    suspend;
  end
end
