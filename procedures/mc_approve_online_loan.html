CREATE PROCEDURE MC_APPROVE_ONLINE_LOAN(
    MCL_ID D_ID,
    K_ID D_ID)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable ool_id d_id;
declare variable ma_id d_id;
declare variable edl_id d_id;
begin

    res_code = 'success';
    res_text = '';

    c_id = (select mcl_from_c_id from mc_login where mcl_id = :mcl_id);
    if (coalesce(c_id, 0) = 0) then
    begin
        res_code = 'error';
        res_text = 'Не удалось загрузить данные профиля. Попробуйте войти в приложение заново';
        suspend;
        exit;
    end

    if (coalesce(k_id, 0) = 0) then
    begin
        res_code = 'error';
        res_text = 'Не удалось получить информацию о кредитном договоре. Попробуйте снова или обратитесь к технической поддержке';
        suspend;
        exit;
    end

    ma_id = (select k.k_from_ma_app_id from kontract k where k.k_id = :k_id);

    -- Не даем утверждать займы, которые не относятся к онлайн-кредитованию ОЙВО
    select ool_id
    from mc_oivo_online_loan ool
    where ool.ool_from_mcl_id = :mcl_id
        and ool.ool_from_ma_id = :ma_id
        and ool.ool_from_k_id = :k_id
    into :ool_id;

    if (ool_id is null) then
    begin
        res_code = 'error';
        res_text = 'Произошла ошибка. Пожалуйста, обратитесь к технической поддержке';
        suspend;
        exit;
    end

    -- Утверждаем договор. Процедура hp_approve_loan здесь не используется, из-за проверок, которые могут
    -- помешать утверждению. Все необходимые проверки в рамках онлайн-кредитования OiVO выполняются отдельно
    update kontract
    set k_status = 1,
        k_user_id = 1000000000000
    where k_id = :k_id;

    -- Отправляем клиенту кредитный договор на подпись. Процедура hp_edoc_ins_loan_agreement здесь
    -- не используется, чтобы избежать проверок, которые в ней проводятся
    select edl_id, res_text
    from hp_edoc_insert('loan.agreement', f_objkind('kontract', 'id'), :k_id, 1000000000000)
    into :edl_id, :res_text;

    if (coalesce(res_text, '') <> '') then
    begin
        -- Заменяем системный текст из процедуры, его нельзя показывать клиенту
        res_code = 'error';
        res_text = 'Не удалось отправить электронный документ. Пожалуйста, попробуйте снова или обратитесь к техническое поддержке';
        suspend;
        exit;
    end

    update mc_oivo_online_loan_edoc oole
    set oole.oole_packet_kontract = :edl_id
    where oole.oole_from_ool_id = :ool_id;

    update mc_oivo_online_loan ool
    set ool.ool_from_k_id = :k_id
        --ool.ool_stage = f_enum_id()
    where ool.ool_id = :ool_id;

    suspend;

    when any do
    begin
        res_code = 'error';
        res_text = 'Что-то пошло не так. Повторите попытку позже или обратитесь к технической поддержке';
        suspend;
    end
end
