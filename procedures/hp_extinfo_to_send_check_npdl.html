CREATE PROCEDURE HP_EXTINFO_TO_SEND_CHECK_NPDL(
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    JSON_VALUE D_BLOB_TEXT)
AS
declare variable LAST_NAME D_STR_CODE;
declare variable FIRST_NAME D_STR_CODE;
declare variable PATRONYMIC D_STR_CODE;
declare variable C_NAME_FIO D_STR_100;
declare variable FL_ID D_ID;
begin
  if (f_objkind('miniapp', 'id') <> :object_kind) then
        exception e_default 'object_kind is not miniapp';

  for select cl.c_name_fio,
             cl.c_from_fl_id
  from clients cl
  inner join miniappclient mac on mac.mac_from_c_id = cl.c_id
  where mac.mac_from_ma_id = :object_id and
        mac.mac_amount > 0
  into c_name_fio,
       fl_id
  do
  begin
    select upper(substring(fl.fl_name1 from 1 for 1)) || lower(substring(fl.fl_name1 from 2)),
           upper(substring(fl.fl_name2 from 1 for 1)) || lower(substring(fl.fl_name2 from 2)),
           upper(substring(fl.fl_name3 from 1 for 1)) || lower(substring(fl.fl_name3 from 2))
    from fiolist fl
    where fl.fl_id = :fl_id
    into :last_name, :first_name, :patronymic;

    if (c_name_fio like '%УУЛУ%') then
       last_name = last_name || ' уулу';
    if (c_name_fio like '%КЫЗЫ%') then
       last_name = last_name || ' кызы';
    if (c_name_fio like '%UULU%') then
       last_name = last_name || ' uulu';
    if (c_name_fio like '%KYZY%') then
       last_name = last_name || ' kyzy';

    json_value = '{' || f_json_make_pair('last_name', :last_name, 'str') || ', '
                     || f_json_make_pair('first_name', :first_name, 'str') || ', '
                     || f_json_make_pair('patronymic', :patronymic, 'str') || '}';
    suspend;
  end
end
