Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE LLC_KONT_LIST(
    REP_LM_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    LK_FROM_LM_ID DECIMAL(18,0),
    LK_FROM_K_ID DECIMAL(18,0),
    LK_DELIQ_DAYS INTEGER,
    LK_FROM_REG_ID INTEGER,
    LK_BALANCE DECIMAL(18,4),
    LK_STATUS SMALLINT,
    LK_PERCENT DECIMAL(18,7),
    LK_DATE DATE,
    LK_DATE_END DATE,
    LK_MEMO VARCHAR(1024),
    LK_IS_INDIV SMALLINT,
    LK_IS_SIGNED SMALLINT,
    LK_IS_CHANGED_MANNUAL INTEGER,
    LK_PERCENT_ORIG DECIMAL(18,7),
    KF_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    CS_NAME VARCHAR(50),
    CS_LLR_KIND_TEXT VARCHAR(50),
    CS_CLASS_TEXT VARCHAR(50),
    REG_NAME VARCHAR(100),
    LK_RPPU DECIMAL(18,4))
AS
declare variable k_clients_cashier_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select lk_from_lm_id, lk_from_k_id, lk_deliq_days, lk_from_reg_id, lk_balance, lk_status, lk_percent, lk_date,
             lk_date_end, lk_memo, lk_is_indiv, lk_is_signed, lk_is_changed_mannual, lk_percent_orig, kal_new_kf_name,
             k_clients_cashier_id, cs_name, cs_llr_kind_text, cs_class_text, reg_name
      from llc_kont, knaktlist, kontract, classific, region
      where lk_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            kal_date <= :rep_date and
            kal_date_end >= :rep_date and
            knaktlist.is_delete = 0 and
            lk_from_lm_id = :rep_lm_id and
            lk_status = cs_id and
            reg_unicode = kal_new_reg_id
      order by reg_unicode, cs_id, kal_new_kf_name, k_id
      into :lk_from_lm_id, :lk_from_k_id, :lk_deliq_days, :lk_from_reg_id, :lk_balance, :lk_status, :lk_percent,
           :lk_date, :lk_date_end, :lk_memo, :lk_is_indiv, :lk_is_signed, :lk_is_changed_mannual, :lk_percent_orig,
           :kf_name, :k_clients_cashier_id, :cs_name, :cs_llr_kind_text, :cs_class_text, :reg_name
  do
  begin
    c_name_fio = '';
    if (k_clients_cashier_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :k_clients_cashier_id
      into c_name_fio;
    end
    lk_rppu = cast((lk_balance * lk_percent / 100) as decimal(18,2));
    suspend;
  end
end
