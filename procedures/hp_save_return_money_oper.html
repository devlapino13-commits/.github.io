CREATE PROCEDURE HP_SAVE_RETURN_MONEY_OPER(
    TYPE_OPER D_STR_20,
    UN_ID D_ID,
    REG_ID D_INTEGER,
    REG_ID_TMP D_INTEGER,
    K_DISB_DATE DATE,
    KONTRACT_NO D_STR_20,
    OPER_DATE DATE,
    C_ID D_ID,
    SUM_OVP D_CURRENCY,
    K_ID D_ID,
    LAST_OVP_DATE DATE = null,
    AL_ID D_STR_20 = null)
RETURNS (
    ACCOUNT_OVERPAYMENTS D_STR_20,
    ERR_MSG D_STR_100,
    NEW_OL_ID D_ID)
AS
declare variable un_from_c_id d_id;
declare variable uc_num_cash d_integer;
declare variable account_cash d_str_20;
declare variable bg_pereplat d_str_10;
declare variable db_id d_integer;
declare variable ol_memo d_str_255;
declare variable doc_no d_integer;
declare variable acc_from_1 d_str_20;
declare variable acc_from_2 d_str_20;
declare variable acc_from_3 d_str_20;
declare variable acc_from_4 d_str_20;
declare variable parent_ol_id d_id;
declare variable def_curr varchar(3);
declare variable bg_kassa d_str_10;
declare variable is_ipf_prod d_integer;
declare variable ob_detail varchar(1024);
declare variable account_doverit d_str_20;
declare variable memo_date date;
declare variable c_fio d_str_100;
declare variable arci_from_ol_id d_id;
declare variable from_to_bishkek d_smallint;
declare variable k_disb_kind d_integer;
begin
  err_msg = '';
  new_ol_id = 0;
  account_overpayments = '';
  from_to_bishkek = 0;
  db_id = f_get_db_id();
  def_curr = f_get_const('cm_default_currency');

  select iif(pr.pr_code containing '.ipf.', 101, 1), k.k_disb_kind
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where k_id = :k_id
  into :is_ipf_prod, :k_disb_kind;

  if (db_id = 81001) then
  begin
    if (al_id is null) then
    begin
      bg_pereplat = '76-06';
      bg_kassa = '50-01';
    end
    else
    begin
      select al_from_bg_id
      from acc_list al
      where al_id in ('010181076-06000002')
      into :bg_pereplat;

      select acl.al_from_bg_id
      from acc_list acl
      where acl.al_id = :al_id
      into :bg_kassa;

    end
  end
  else
  if (db_id in (39801, 41701)) then
  begin
    if (type_oper = 'overpayment') then
      bg_pereplat = '21199';
    else
    if (type_oper = 'consultation') then
    begin
      bg_pereplat = '21114';
      is_ipf_prod = 10;

      select al.al_id
      from acc_list al
      where al.al_from_reg_id = :reg_id_tmp and
            al.al_from_bg_id = '21199' and
            al.al_number = 110
      into account_doverit;

    end
    bg_kassa = '10001';
  end

  if (al_id is null) then
    select al.al_id
    from acc_list al
    where al.al_from_reg_id = :reg_id_tmp and
          al.al_from_bg_id = :bg_pereplat and
          al.al_number = :is_ipf_prod
    into account_overpayments;
  else
    account_overpayments = '010181076-06000002';
  if (:k_disb_kind = 6 and db_id in (39801, 41701)) then
  begin
    select al.al_id
    from acc_list al
    where al.al_from_reg_id = :reg_id_tmp and
          al.al_from_bg_id = :bg_pereplat and
          al.al_number = 1101
    into account_overpayments;
  end
  -- находим номер кассы заданного пользователя
  if (db_id in (39801, 41701)) then
  begin
    select un_from_c_id
    from username
    where un_id = :un_id
    into un_from_c_id;

    select uc_num_cash
    from hp_user_cash(null, 0, :un_from_c_id)
    where uc_from_reg_id = :reg_id
    into uc_num_cash;
    uc_num_cash = coalesce(uc_num_cash, 0);

    if (uc_num_cash = 0) then
    begin
      err_msg = 'У вас не имеется доступ к кассе!';
      suspend;
      exit;
    end

    if (is_ipf_prod = 101) then
      uc_num_cash = uc_num_cash + 100;

    select al_id
    from acc_list
    where al_from_bg_id = :bg_kassa and
          al_from_reg_id = :reg_id and
          acc_list.is_delete = 0 and
          al_number = :uc_num_cash
    into :account_cash;

    if (account_cash is null or account_overpayments is null) then
    begin
      if (account_cash is null) then
        err_msg = 'Cash Account is null';
      else
        err_msg = 'Overpayment Account is null';
      suspend;
      exit;
    end

  end
  else
  begin
    -- вычисляем счет кассы, откуда будет возврат переплаты
    if (al_id is null) then
    begin
      select al_id
      from acc_list
      where al_from_bg_id = :bg_kassa and
            al_from_reg_id = :reg_id and
            acc_list.is_delete = 0
      into account_cash;
    end
    else
      account_cash = al_id;
  end

  select result
  from get_kv_id(5)
  into new_ol_id;

  if (type_oper = 'consultation') then
  begin
    select arc.arc_date, c.c_name_fio, arci.arci_from_ol_id
    from app_return_consult arc
    inner join app_return_consult_info arci on arci.arci_id = arc.arc_id
    inner join clients c on c.c_id = arc.arc_from_c_id
    where arc.arc_from_c_id = :c_id and
          arc.arc_from_k_id = :k_id
    into :memo_date, :c_fio, :arci_from_ol_id;

    if (arci_from_ol_id is null) then
      err_msg = 'Невозможно продолжить выдачу из-за отсутствия родительской проводки.';
  end
  if (type_oper = 'overpayment') then
    memo_date = k_disb_date;

  ol_memo = right((100 + extract(day from memo_date)), 2) || '.' || right((100 + extract(month from memo_date)), 2) || '.' || extract(year from memo_date) || ' г.';
  doc_no = 0;
  select num
  from last_doc_number(:oper_date, 0, :reg_id)
  into :doc_no;

  if (type_oper = 'overpayment') then
    :parent_ol_id = :new_ol_id;
  if (type_oper = 'consultation') then
    :parent_ol_id = :arci_from_ol_id;

  --------------------------------------------------
  if ((:reg_id <> :reg_id_tmp) and
      (db_id in (39801, 41701))) then
  begin
    if (type_oper = 'consultation') then
    begin
      ol_memo = :c_fio || ' согл.заявления от ' || :ol_memo;
      ob_detail = 'Возврат оплаты за консультационные услуги через другой офис ';
    end
    if (type_oper = 'overpayment') then
    begin
      ol_memo = kontract_no || ' от ' || :ol_memo;
      ol_memo = 'Возврат переплаты по кредитному договору ' || :ol_memo;
      ob_detail = 'Возврат переплаты через другой офис';
    end
    if (:reg_id_tmp = 1060100 or :reg_id = 1060100) then
    begin
      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = :reg_id_tmp and
            lr.ln_region_nostro = :reg_id
      into :acc_from_1;

      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = :reg_id and
            lr.ln_region_nostro = :reg_id_tmp
      into :acc_from_4;

      from_to_bishkek = 1;
    end
    begin

      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = :reg_id_tmp and
            lr.ln_region_nostro = 1060100
      into :acc_from_1;

      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = 1060100 and
            lr.ln_region_nostro = :reg_id_tmp
      into :acc_from_2;

      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = 1060100 and
            lr.ln_region_nostro = :reg_id
      into :acc_from_3;

      select lr.ln_account
      from loronostro lr
      where lr.ln_region_loro = :reg_id and
            lr.ln_region_nostro = 1060100
      into :acc_from_4;

    end

    if (type_oper = 'overpayment') then
      execute procedure hp_overpayment_return_save_tmp(:new_ol_id, :reg_id_tmp, :oper_date, :doc_no, :ol_memo, :un_id,
          :c_id, :def_curr, :account_overpayments, :sum_ovp, :acc_from_1, k_id, last_ovp_date, 0, :parent_ol_id,
          :ob_detail);

    if (type_oper = 'consultation') then
      execute procedure hp_overpayment_return_save_tmp(:new_ol_id, :reg_id_tmp, :oper_date, :doc_no, :ol_memo, :un_id,
          :c_id, :def_curr, :account_doverit, :sum_ovp, :acc_from_1, k_id, last_ovp_date, 0, :parent_ol_id, :ob_detail);

    if (from_to_bishkek = 0) then
    begin
      :new_ol_id = :new_ol_id + 1;

      execute procedure hp_overpayment_return_save_tmp(:new_ol_id, 1060100, :oper_date, :doc_no, :ol_memo, :un_id,
          :c_id, :def_curr, :acc_from_2, :sum_ovp, :acc_from_3, k_id, last_ovp_date, 0, :parent_ol_id, :ob_detail);

    end
    :new_ol_id = :new_ol_id + 1;

    execute procedure hp_overpayment_return_save_tmp(:new_ol_id, :reg_id, :oper_date, :doc_no, :ol_memo, :un_id, :c_id,
        :def_curr, :acc_from_4, :sum_ovp, :account_cash, k_id, last_ovp_date, 1, :parent_ol_id, :ob_detail);

    update operloan
    set opl_from_al_id = :account_cash,
        opl_acc_kind = 1
    where opl_from_ol_id = :parent_ol_id;

  end
  else
  --------------------------------------------------
  begin

    if (type_oper = 'consultation') then
    begin
      ol_memo = :c_fio || ' согл.заявления от ' || :ol_memo;
      ob_detail = 'Возврат оплаты за консультационные услуги ';
    end
    if (type_oper = 'overpayment') then
    begin
      ol_memo = kontract_no || ' от ' || :ol_memo;
      ol_memo = 'Возврат переплаты по кредитному договору ' || :ol_memo;
      if (al_id is not null) then
        ol_memo = :ol_memo || ' безналичным расчетом через банк';
      ob_detail = 'Возврат переплаты';
    end

    if (type_oper = 'overpayment') then
      execute procedure hp_overpayment_return_save_tmp(:new_ol_id, :reg_id, :oper_date, :doc_no, :ol_memo, :un_id,
          :c_id, :def_curr, :account_overpayments, :sum_ovp, :account_cash, k_id, last_ovp_date, 1, :parent_ol_id,
          :ob_detail);

    if (type_oper = 'consultation') then
      execute procedure hp_overpayment_return_save_tmp(:new_ol_id, :reg_id_tmp, :oper_date, :doc_no, :ol_memo, :un_id,
          :c_id, :def_curr, :account_doverit, :sum_ovp, :account_cash, k_id, last_ovp_date, 0, :parent_ol_id,
          :ob_detail);

    update operloan
    set opl_from_al_id = :account_cash,
        opl_acc_kind = 1
    where opl_from_ol_id = :new_ol_id;
  end

  suspend;
end
