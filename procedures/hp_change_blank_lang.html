CREATE PROCEDURE HP_CHANGE_BLANK_LANG(
    C_ID D_ID,
    NEW_BLANK_LANG D_STR_LANG,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable ma_id d_integer;
declare variable ma_sign_status d_str_code;
declare variable kon_sign_status d_str_code;
declare variable can_update_blank_lang d_smallint;
declare variable lang d_str_lang;
begin
    -- RFCKG-4156: Запрет на изменение языка формирования бланков, если заявка на займ уже подписана, но договор еще нет

    lang = f_get_lang_code(un_id, null);
    can_update_blank_lang = 0;
    
    -- Ищем последнюю заявку
    select ma.ma_id
    from miniappclient mac
    inner join miniapp ma on mac.mac_from_ma_id = ma.ma_id
    where
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999' and
        ma_date_oper >= (current_date - 30) and
        mac.mac_from_c_id = :c_id and
        mac_amount > 0
    order by ma.ma_date_app desc
    rows 1
    into
      :ma_id;
    
    ma_sign_status = f_edoc_status('loan.application', :ma_id, current_date);
    
    if (ma_sign_status = 'signed') then -- Если заявка подписана
    begin
        -- Проверям, подписан ли договор, связанный с этой заявкой
        kon_sign_status = f_edoc_status('loan.agreement', (select k.k_id from kontract k where k.k_from_ma_app_id = :ma_id), current_date);
        
        -- Если договор еще не подписан
        if (kon_sign_status <> 'signed') then
            res_text = f_lang(:lang, 'Изменение языка формирования бланков запрещается до подписания договора, т.к. заявка уже подписана. Чтобы изменить язык, аннулируйте текущую заявку');
        else
            can_update_blank_lang = 1; -- Если и заявка, и договор уже подписаны, изменение языка формирования бланков разрешается
    end
    else
        can_update_blank_lang = 1; -- Если заявка еще не подписана, изменение языка формирования бланков разрешается
    
    if (can_update_blank_lang = 1) then
    begin
        update clients
        set c_blank_lang = :new_blank_lang
        where c_id = :c_id;

        res_text = f_lang(lang, 'Язык для подготовки бланков изменен');
    end

    suspend;
end
