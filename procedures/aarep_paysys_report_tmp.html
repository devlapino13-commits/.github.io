CREATE PROCEDURE AAREP_PAYSYS_REPORT_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_ID INTEGER,
    REP_REGION_NAME VARCHAR(100),
    REP_CLIENT_FIO VARCHAR(100),
    REP_POLIS_SALDO_BEGIN INTEGER,
    REP_POLIS_PLUS INTEGER,
    REP_POLIS_MINUS INTEGER,
    REP_POLIS_SALDO_END INTEGER,
    REP_POLIS_DISB INTEGER,
    REP_POLIS_CANCEL INTEGER,
    REP_PROJECT_NAME VARCHAR(255))
AS
declare variable c_id decimal(18,0);
begin
  rep_polis_saldo_begin = 0;
  rep_polis_plus = 0;
  rep_polis_minus = 0;
  rep_polis_saldo_end = 0;
  rep_polis_disb = 0;
  rep_polis_cancel = 0;
  rep_client_fio = 'Офис';

  for select reg_unicode, reg_code || ' ' || reg_name, psp_name
      from region, region_list(:rep_region_id, 1, 3, 1), paysys_project
      where reg_unicode = id_list and
            psp_is_active = 1
      into :reg_id, :rep_region_name, :rep_project_name
  do
  begin
    suspend;
  end

  for select reg_unicode, reg_code || ' ' || reg_name, psp_name, sum(psl_sign)
      from paysys_list, paysys_main, region, region_list(:rep_region_id, 1, 3, 1), paysys_item, paysys_project
      where psl_object_kind = 1 and
            psl_from_psm_id = psm_id and
            psl_is_delete = 0 and
            psm_is_delete = 0 and
            psl_object_id = reg_unicode and
            psm_date < :rep_date_begin and
            reg_unicode = id_list and
            psp_id = psi_from_psp_id and
            psl_from_psi_id = psi_id
      group by reg_unicode, reg_code, reg_name, psp_name
      having sum(psl_sign) <> 0
      into :reg_id, :rep_region_name, :rep_project_name, :rep_polis_saldo_begin
  do
  begin
    suspend;
  end

  rep_polis_saldo_begin = 0;
  rep_polis_plus = 0;
  rep_polis_minus = 0;
  rep_polis_saldo_end = 0;

  for select reg_unicode, reg_code || ' ' || reg_name, psp_name, sum(iif(psl_sign > 0, psl_sign, 0)),
             sum(iif(psl_sign < 0, -psl_sign, 0))
      from paysys_list, paysys_main, region, region_list(:rep_region_id, 1, 3, 1), paysys_item, paysys_project
      where psl_object_kind = 1 and
            psl_from_psm_id = psm_id and
            psl_is_delete = 0 and
            psm_is_delete = 0 and
            psl_object_id = reg_unicode and
            psm_date between :rep_date_begin and :rep_date_end and
            reg_unicode = id_list and
            psp_id = psi_from_psp_id and
            psl_from_psi_id = psi_id
      group by reg_unicode, reg_code, reg_name, psp_name
      into :reg_id, :rep_region_name, :rep_project_name, :rep_polis_plus, :rep_polis_minus
  do
  begin
    suspend;
  end

  rep_polis_saldo_begin = 0;
  rep_polis_plus = 0;
  rep_polis_minus = 0;
  rep_polis_saldo_end = 0;

  for select reg_unicode, reg_code || ' ' || reg_name, psp_name, sum(psl_sign)
      from paysys_list, paysys_main, region, region_list(:rep_region_id, 1, 3, 1), paysys_item, paysys_project
      where psl_object_kind = 1 and
            psl_from_psm_id = psm_id and
            psl_is_delete = 0 and
            psm_is_delete = 0 and
            psl_object_id = reg_unicode and
            psm_date <= :rep_date_end and
            reg_unicode = id_list and
            psp_id = psi_from_psp_id and
            psl_from_psi_id = psi_id
      group by reg_unicode, reg_code, reg_name, psp_name
      into :reg_id, :rep_region_name, :rep_project_name, :rep_polis_saldo_end
  do
  begin
    suspend;
  end

  rep_polis_saldo_begin = 0;
  rep_polis_plus = 0;
  rep_polis_minus = 0;
  rep_polis_saldo_end = 0;
  rep_polis_disb = 0;
  rep_polis_cancel = 0;

  for select reg_unicode, reg_code || ' ' || reg_name, psp_name, sum(iif(psm_oper_kind = 0, psl_sign, 0)),
             sum(iif(psm_oper_kind = 1, -psl_sign, 0))
      from paysys_list, paysys_main, region, region_list(:rep_region_id, 1, 3, 1), paysys_item, paysys_project
      where psl_object_kind > 2 -- Получатель это контракт или операция
            and
            psl_from_psm_id = psm_id and
            psl_is_delete = 0 and
            psm_is_delete = 0 and
            psm_from_reg_id = reg_unicode and
            psm_date between :rep_date_begin and :rep_date_end and
            reg_unicode = id_list and
            psp_id = psi_from_psp_id and
            psl_from_psi_id = psi_id
      group by reg_unicode, reg_code, reg_name, psp_name
      into :reg_id, :rep_region_name, :rep_project_name, :rep_polis_disb, :rep_polis_cancel
  do
  begin
    suspend;
  end

  rep_polis_saldo_begin = 0;
  rep_polis_plus = 0;
  rep_polis_minus = 0;
  rep_polis_saldo_end = 0;
  rep_polis_disb = 0;
  rep_polis_cancel = 0;

  for select c_id, c_name_fio, psp_name, sum(psl_sign)
      from paysys_list, paysys_main, clients, paysys_item, paysys_project
      where psl_object_kind = 2 and
            psl_from_psm_id = psm_id and
            psl_is_delete = 0 and
            psm_is_delete = 0 and
            psl_object_id = c_id and
            psm_date <= :rep_date_end and
            psp_id = psi_from_psp_id and
            psl_from_psi_id = psi_id
      group by c_id, c_name_fio, psp_name
      into :c_id, :rep_client_fio, :rep_project_name, :rep_polis_saldo_end
  do
  begin
    rep_region_name = '';
    select ps_from_reg_id, reg_code || ' ' || reg_name
    from hr_pstatus, region_list(:rep_region_id, 1, 3, 1), region
    where ps_from_c_id = :c_id and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_from_reg_id = id_list and
          ps_status = 1 and
          ps_from_reg_id = reg_unicode
    into :reg_id, :rep_region_name;
    if (rep_region_name is null) then
      rep_region_name = '';
    if (rep_region_name <> '') then
      suspend;
  end

  rep_polis_saldo_begin = 0;
  rep_polis_plus = 0;
  rep_polis_minus = 0;
  rep_polis_saldo_end = 0;
  rep_polis_disb = 0;
  rep_polis_cancel = 0;

  for select c_id, c_name_fio, psp_name, sum(psl_sign)
      from paysys_list, paysys_main, clients, paysys_item, paysys_project
      where psl_object_kind = 2 and
            psl_from_psm_id = psm_id and
            psl_is_delete = 0 and
            psm_is_delete = 0 and
            psl_object_id = c_id and
            psm_date < :rep_date_begin and
            psp_id = psi_from_psp_id and
            psl_from_psi_id = psi_id
      group by c_id, c_name_fio, psp_name
      into :c_id, :rep_client_fio, :rep_project_name, :rep_polis_saldo_begin
  do
  begin
    rep_region_name = '';
    select ps_from_reg_id, reg_code || ' ' || reg_name
    from hr_pstatus, region_list(:rep_region_id, 1, 3, 1), region
    where ps_from_c_id = :c_id and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_from_reg_id = id_list and
          ps_status = 1 and
          ps_from_reg_id = reg_unicode
    into :reg_id, :rep_region_name;
    if (rep_region_name is null) then
      rep_region_name = '';
    if (rep_region_name <> '') then
      suspend;
  end

  rep_polis_saldo_begin = 0;
  rep_polis_plus = 0;
  rep_polis_minus = 0;
  rep_polis_saldo_end = 0;
  rep_polis_disb = 0;
  rep_polis_cancel = 0;

  for select c_id, c_name_fio, psp_name, sum(iif(psl_sign > 0, psl_sign, 0)), sum(iif(psl_sign < 0, -psl_sign, 0))
      from paysys_list, paysys_main, clients, paysys_item, paysys_project
      where psl_object_kind = 2 and
            psl_from_psm_id = psm_id and
            psl_is_delete = 0 and
            psm_is_delete = 0 and
            psl_object_id = c_id and
            psm_date between :rep_date_begin and :rep_date_end and
            psp_id = psi_from_psp_id and
            psl_from_psi_id = psi_id
      group by c_id, c_name_fio, psp_name
      into :c_id, :rep_client_fio, :rep_project_name, :rep_polis_plus, :rep_polis_minus
  do
  begin
    rep_region_name = '';
    select ps_from_reg_id, reg_code || ' ' || reg_name
    from hr_pstatus, region_list(:rep_region_id, 1, 3, 1), region
    where ps_from_c_id = :c_id and
          ps_date_begin <= :rep_date_end and
          ps_date_end >= :rep_date_end and
          ps_from_reg_id = id_list and
          ps_status = 1 and
          ps_from_reg_id = reg_unicode
    into :reg_id, :rep_region_name;
    if (rep_region_name is null) then
      rep_region_name = '';
    if (rep_region_name <> '') then
      suspend;
  end
end
