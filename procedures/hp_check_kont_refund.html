CREATE PROCEDURE HP_CHECK_KONT_REFUND(
    K_ID ID_TYPE,
    REP_DATE D_DATE)
RETURNS (
    RES_TEXT STR_255)
AS
declare variable disb_date d_date;
declare variable pr_code str_100;
declare variable k_id_tmp id_type;
declare variable disb_date_tmp d_date;
declare variable pr_code_tmp str_100;
declare variable k_kontract_no str_20;
declare variable c_id id_type;
declare variable pr_id d_integer;
declare variable pr_id_tmp d_integer;
begin
  res_text = '';

  for select k_fact_disb_date, pr_code, kc_from_c_id, pr_id
      from kontract
      inner join product on pr_id = k_from_pr_id
      inner join knclient on kc_from_k_id = k_id
      where k_id = :k_id and
            kc_amount_approved > 0
      into disb_date, pr_code, c_id, pr_id
  do
  begin
    k_kontract_no = null;

    select k_kontract_no, k_fact_disb_date
    from kontract, product, kontstate, knclient
    where k_from_pr_id = pr_id and
          ks_from_k_id = k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date between ks_date_begin and ks_date_end and
          pr_id = :pr_id and
          kc_from_k_id = k_id and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          k_fact_disb_date < :disb_date
    order by k_fact_disb_date rows 1
    into :k_kontract_no, :disb_date_tmp;

    if (k_kontract_no is not null) then
    begin
      res_text = 'Заем по договору ' || k_kontract_no || ' от ' || f_datetostr(disb_date_tmp) || ' был рефинансирован, сперва требуется погасить данный заем. ';
      suspend;
    end

    k_kontract_no = null;
    k_id_tmp = null;

    select knc.kc_from_c_id, pr_code, min(k_id), pr_id
    from kontstate ks
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join knclient knc on k.k_id = knc.kc_from_k_id
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k_from_pr_id = pr_id and
          ks_from_k_id = k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date between ks_date_begin and ks_date_end and
          pr_id <> :pr_id and
          kc_from_k_id = k_id and
          kc_from_c_id = :c_id and
          kc_amount_approved > 0
    group by 1, 2, 4
    having count(kc_from_c_id) > 1 
    order by 1,2,3 rows 1
    into c_id, pr_code_tmp, k_id_tmp, pr_id_tmp;

    if (k_id_tmp is not null) then
    begin
      select k_kontract_no, k_fact_disb_date
      from kontract
      where k_id = :k_id_tmp
      into :k_kontract_no, :disb_date_tmp;
      res_text = 'Заем по договору ' || k_kontract_no || ' от ' || f_datetostr(disb_date_tmp) || ' был рефинансирован, сперва требуется погасить данный заем. ';
      suspend;
    end
  end
end
