CREATE PROCEDURE CC_DELINQ_LIST_TMP(
    REP_DATE DATE,
    PROJECT STR_100,
    CALL_TYPE SMALLINT,
    CALLED_OPER STR_100,
    K_FOLDER_NAME STR_100,
    KONTRACT_NO STR_100,
    DELINQ_DAYS_MIN INTEGER,
    DELINQ_DAYS_MAX INTEGER)
RETURNS (
    K_ID ID_TYPE,
    KF_ID ID_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    DELINQ_DAYS INTEGER,
    OKS_KC_ID ID_TYPE)
AS
declare variable K_ID_TMP ID_TYPE;
declare variable DELINQ_DAYS_MIN_TMP integer;
declare variable DB_ID integer;
declare variable DELINQ_DAYS_MAX_TMP integer;
declare variable PROJECT_RD_ID integer;
begin
-- CALL_TYPE 1 - КЦ Сирена,
--           2 - звонки прочрочникам

  :db_id = f_get_db_id();

  delinq_days_max_tmp = 9999;
  delinq_days_min_tmp = 7;

  if (db_id = 94401) then -- АЗ
  begin
    delinq_days_min_tmp = 21;
    delinq_days_max_tmp = 9999; --  до 25.01.2019 , было  9999
  end

  if (db_id = 81001) then
  begin
    if (project = 'OKTELL.Projects.Sirena') then
    begin
      if (call_type = 1) then
        delinq_days_min_tmp = 1;
      else
        delinq_days_min_tmp = 7;
      delinq_days_max_tmp = 60;
    end
    else
    if (project = 'OKTELL.Projects.Sirena1') then
    begin
      if (call_type = 1) then
        delinq_days_min_tmp = 1;
      else
        delinq_days_min_tmp = 7;
      delinq_days_max_tmp = 30;
    end
    else
    if (project = 'OKTELL.Projects.Sirena2') then
    begin
      delinq_days_min_tmp = 31;
      delinq_days_max_tmp = 60;
    end
  end

  select rd.rd_id
  from refdata rd
  where rd.rd_code = :project
  into :project_rd_id;

  called_oper = trim(coalesce(called_oper, ''));
  if (k_folder_name = '') then
    k_folder_name = null;

  kontract_no = trim(coalesce(kontract_no, ''));
  k_id_tmp = 0;
  delinq_days_min = coalesce(delinq_days_min, 1);
  if (delinq_days_min <= 0) then
    delinq_days_min = 1;

  delinq_days_max = coalesce(delinq_days_max, 29);
  if (delinq_days_max <= 0) then
    delinq_days_max = 29;

  if (delinq_days_min < delinq_days_min_tmp) then
    delinq_days_min = delinq_days_min_tmp;
  if (delinq_days_max > delinq_days_max_tmp) then
    delinq_days_max = delinq_days_max_tmp;

  if (kontract_no <> '') then
  begin
    select kont.k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id_tmp;

    if (k_id_tmp is null) then
      k_id_tmp = 0;
  end

  if (call_type = 1) then
  begin
    for select oks.oks_from_k_id as k_id, k_kontract_no, kont.k_from_kf_id as kf_id,
               cast(:rep_date as date) - oks_delinq_date as delinq_days, oks.oks_kc_id
        from oksummary oks
        inner join kontract kont on kont.k_id = oks.oks_from_k_id
        inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
        where :rep_date between oks_date_begin and oks_date_end and
              oks_delinq_date < :rep_date and
              cast(:rep_date as date) - oks_delinq_date between :delinq_days_min and :delinq_days_max and
              oks_close_state = 0
            --and ((kont.k_id = :k_id_tmp) or (:k_id_tmp = 0))
              and
              oks.oks_kc_id = 1000000000004 and
              :rep_date between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0
              --and ((kal.kal_new_kf_name = :K_FOLDER_NAME)  or (:K_FOLDER_NAME is null))
        into :k_id, :k_kontract_no, :kf_id, :delinq_days, :oks_kc_id
    do
    begin
      if (not exists(select *
                     from pantera_ignore pig
                     where pig.pi_object_kind = 9 and
                           pig.pi_object_id = :k_id and
                           pig.pi_is_active = 1 and
                           :rep_date between pig.pi_date_begin and pig.pi_date_end and
                           ((pig.pi_rd_project_id is null) or
                           (pig.pi_rd_project_id = :project_rd_id)))) then
      begin
        suspend;
      end
    end
  end
  if (call_type = 2) then
  begin
    for select oks.oks_from_k_id as k_id, k_kontract_no, kont.k_from_kf_id as kf_id,
               cast(:rep_date as date) - oks_delinq_date as delinq_days, oks.oks_kc_id
        from oksummary oks
        inner join kontract kont on kont.k_id = oks.oks_from_k_id
        inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
        where :rep_date between oks_date_begin and oks_date_end and
              oks_delinq_date < :rep_date and
              cast(:rep_date as date) - oks_delinq_date between :delinq_days_min and :delinq_days_max and
              oks_close_state = 0
              --and ((kont.k_id = :k_id_tmp) or (:k_id_tmp = 0))
              and
              oks.oks_kc_id <> 1000000000004 and
              :rep_date between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0
              --and ((kal.kal_new_kf_name = :K_FOLDER_NAME)  or (:K_FOLDER_NAME is null))
        into :k_id, :k_kontract_no, :kf_id, :delinq_days, :oks_kc_id
    do
    begin
      if (not exists(select *
                     from pantera_ignore pig
                     where pig.pi_object_kind = 9 and
                           pig.pi_object_id = :k_id and
                           pig.pi_is_active = 1 and
                           :rep_date between pig.pi_date_begin and pig.pi_date_end and
                           ((pig.pi_rd_project_id is null) or
                           (pig.pi_rd_project_id = :project_rd_id)))) then
      begin
        suspend;
      end
    end
  end
end
