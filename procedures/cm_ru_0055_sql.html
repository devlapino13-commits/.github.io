CREATE PROCEDURE CM_RU_0055_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kf_officer id_type;
declare variable officer_name varchar(100);
declare variable kf_id id_type;
begin
  for
  select kal_new_kf_officer, ma_from_kf_id
  from miniapp, knaktlist
  where
  ma_id = :ma_id and
  ma_from_kf_id = kal_from_kf_id
  and knaktlist.is_delete = 0
  and current_date between kal_date and kal_date_end
  and ma_is_active = 1
  and kal_new_kf_officer in (
        1010100000724,
        1010100301143,
        1010100018673,
        1010100020082,
        1010100200408,
        1010100065165,
        1170100000166,
        1010800000351,
        1010100190437,
        1010100222748,
        1010300007049,
        1010100253352,
        1010100275448,
        1010100294749
        )
    order by ma_number desc
    rows 1
    into kf_officer, kf_id
    do
    begin
--    проверка контракта новый или нет, если новый только тогда выдавать следующее сообщение
      if (not exists(select * from kontract, kontstate
                     where k_from_kf_id = :kf_id
                     and k_id = ks_from_k_id
                     and ks_ol_is_delete = 0
                     and ks_state = 1)) then
      begin
          officer_name = '';
          select c_name_fio from clients
          where c_id = :kf_officer
          into officer_name;
    
          res_error_text = 'СВЗ: '||officer_name||' не имеет права выдавать новым группам из-за увеличение PAR';
          res_error_is_critical = 2;
          suspend;
      end
    end
end
