CREATE PROCEDURE UXP$UXFRAME_SQL_TREE_MOVE(
    UFR_UUID D_UUID,
    REPLACE_FRAME_FIELDS D_BOOLEAN = false)
RETURNS (
    RES_TREE_MOVE_SQL D_STR_8100)
AS
declare variable ufr_tree_position_field d_str_code;
declare variable us_code d_str_code;
declare variable us_main_table d_str_code;
declare variable us_keyfields d_str_code;
declare variable us_tree_parent_field d_str_code;
declare variable us_tree_position_field d_str_code;
declare variable tree_source d_str_code;
declare variable ufr_tree_parent_field d_str_code;
declare variable orig_keyfields d_str_code;
declare variable ufr_add_table_fields d_str_code;
begin
  select ufr_tree_position_field, ufr_tree_parent_field, ufr_tree_position_field,
         us_code, us_main_table, ufr_add_table_fields,
         us_keyfields, us_tree_parent_field, us_tree_position_field
  from ux$frame, ux$source
  where ufr_uuid = :ufr_uuid and
        ufr_us_uuid = us_uuid
  into ufr_tree_position_field, ufr_tree_parent_field, ufr_tree_position_field,
       us_code, us_main_table, ufr_add_table_fields,
       us_keyfields, us_tree_parent_field, us_tree_position_field;

  res_tree_move_sql = '';
  orig_keyfields = us_keyfields;
  if (ufr_tree_position_field <> '') then
  begin
    tree_source = us_code;

    if (us_keyfields = '' or us_tree_parent_field = '' or us_tree_position_field = '') then
    begin
      if (coalesce(ufr_add_table_fields, '') <> '') then
      begin
        select us_keyfields, us_tree_parent_field, us_tree_position_field
        from ux$source
        where us_code = :ufr_add_table_fields
        into us_keyfields, us_tree_parent_field, :us_tree_position_field;

        tree_source = ufr_add_table_fields;
      end
    end

    if (us_keyfields = '' or us_tree_parent_field = '' or us_tree_position_field = '') then
    begin
      if (us_main_table <> '') then
      begin
        select us_keyfields, us_tree_parent_field, us_tree_position_field
        from ux$source
        where us_code = :us_main_table
        into us_keyfields, us_tree_parent_field, :us_tree_position_field;

        tree_source = us_main_table;
      end
    end

    if (us_keyfields <> '' and us_tree_parent_field <> '' and us_tree_position_field <> '') then
    begin
      select tree_move_sql
      from uxp$tree_move_sql(:tree_source, :us_keyfields, :us_tree_parent_field, :us_tree_position_field)
      into :res_tree_move_sql;

      if (res_tree_move_sql <> '' and replace_frame_fields) then
      begin
        res_tree_move_sql = replace(res_tree_move_sql, ':tree$id', ':' || orig_keyfields);
        res_tree_move_sql = replace(res_tree_move_sql, ':tree$parent', ':' || ufr_tree_parent_field);
        res_tree_move_sql = replace(res_tree_move_sql, ':tree$position', ':' || ufr_tree_position_field);
      end
    end
  end

  suspend;
end
