CREATE PROCEDURE HP_DB_STAT(
    PAGE_LEVEL D_INTEGER)
RETURNS (
    ST_STAT_ID D_INTEGER,
    ST_STAT_GROUP D_SMALLINT,
    ST_PAGE_READS D_ID,
    ST_PAGE_WRITES D_ID,
    ST_PAGE_FETCHES D_ID,
    ST_PAGE_MARKS D_ID,
    CS_CALL_ID D_ID,
    CS_STATEMENT_ID D_ID,
    CS_CALLER_ID D_ID,
    CS_OBJECT_NAME CHAR(31),
    CS_OBJECT_TYPE D_SMALLINT,
    CS_TIMESTAMP D_DATETIME,
    CS_SOURCE_LINE D_INTEGER,
    CS_SOURCE_COLUMN D_INTEGER,
    S_ATTACHMENT_ID D_ID,
    S_TRANSACTION_ID D_ID,
    S_STATE D_SMALLINT,
    S_TIMESTAMP D_DATETIME,
    S_SQL_TEXT D_STR_8100,
    S_OBJECT_LIST D_STR_8100)
AS
begin
  for select st.mon$stat_id,
             st.mon$stat_group,
             st.mon$page_reads,
             st.mon$page_writes,
             st.mon$page_fetches,
             st.mon$page_marks
  from mon$io_stats st
  where st.mon$page_fetches > :page_level
  into st_stat_id,
       st_stat_group,
       st_page_reads,
       st_page_writes,
       st_page_fetches,
       st_page_marks
  do
  begin
    for select
         cs.mon$call_id,
         cs.mon$statement_id,
         cs.mon$caller_id,
         cs.mon$object_name,
         cs.mon$object_type,
         cs.mon$timestamp,
         cs.mon$source_line,
         cs.mon$source_column,
         s.mon$attachment_id,
         s.mon$transaction_id,
         s.mon$state,
         s.mon$timestamp,
         cast(left(s.mon$sql_text, 8100) as varchar(8100)) as mon$sql_text
    from mon$call_stack cs,
          mon$statements s
    where cs.mon$stat_id = :st_stat_id and
          cs.mon$statement_id = s.mon$statement_id
    into cs_call_id,
         cs_statement_id,
         cs_caller_id,
         cs_object_name,
         cs_object_type,
         cs_timestamp,
         cs_source_line,
         cs_source_column,
         s_attachment_id,
         s_transaction_id,
         s_state,
         s_timestamp,
         s_sql_text
    do
    begin
      s_object_list = '';
      suspend;

      with recursive
        tab as (
        select
        cs.mon$statement_id, cs.mon$call_id, cs.mon$object_name
        from mon$call_stack cs
        where cs.mon$caller_id is null
        union all
        select
        cs.mon$statement_id, cs.mon$call_id, cs.mon$object_name
        from tab, mon$call_stack cs
        where cs.mon$caller_id = tab.mon$call_id
        order by cs.mon$call_id
        )
      select list(trim(mon$object_name), ' <- ')
      from tab, mon$statements s
      where s.mon$statement_id = :cs_statement_id and
            s.mon$statement_id = tab.mon$statement_id
      into s_object_list;

      suspend;
    end
  end
end
