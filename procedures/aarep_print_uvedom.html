CREATE PROCEDURE AAREP_PRINT_UVEDOM(
    REP_K_ID DECIMAL(18,0),
    REP_DATE_END DATE,
    REP_CURR_RATE DECIMAL(18,4) = 0)
RETURNS (
    REP_DATE DATE,
    K_ORIG_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_KOLICH_REPAY INTEGER,
    K_PERCENT DECIMAL(18,7),
    K_DATE_STOP_PERC DATE,
    KDI_PAYED_TOTAL DECIMAL(18,4),
    KDI_ZAD_OS_NOW DECIMAL(18,4),
    KDI_CALC_PERC_NOW DECIMAL(18,4),
    KDI_CALC_FINE_NOW DECIMAL(18,4),
    KDI_SUM_TOTAL_NOW DECIMAL(18,4),
    KDI_DEL_DAYS INTEGER,
    K_CLIENTS_FIO VARCHAR(2048),
    K_CLIENTS_FIO_ADDRESS VARCHAR(2048),
    KDI_DOCS VARCHAR(2048),
    KDI_SUM_TOTAL_NOW_KG DECIMAL(18,4),
    K_CLIENTS_FIO_ADDRESS_BIRTH VARCHAR(2048),
    K_CLIENTS_FIO_ADDRESS2 VARCHAR(2048))
AS
declare variable k_id_tmp decimal(18,0);
declare variable kdi_zad_perc_now decimal(18,4) = 0;
declare variable sh_date date;
declare variable c_name_fio varchar(100);
declare variable c_address varchar(255);
declare variable k_id decimal(18,0);
declare variable ol_id decimal(18,0);
declare variable ol_parent_id decimal(18,0);
declare variable ol_doc_no integer;
declare variable ol_date date;
declare variable doc_type varchar(20);
declare variable ok_sum decimal(18,4) = 0;
declare variable ol_date_str varchar(20);
declare variable kol integer;
declare variable c_id decimal(18,0);
declare variable c_birth_date date;
declare variable c_birth_date_str varchar(20);
begin
  if (rep_date_end is null) then
    rep_date_end = current_date;
  rep_date = :rep_date_end;

  k_id = :rep_k_id;
  k_id_tmp = null;
  select kd_k_orig
  from kontdevides
  where kd_k_new = :k_id
  into :k_id_tmp;

  if (k_id_tmp is null) then
    k_id_tmp = k_id;

-- номер оригинального договора
  select k_kontract_no, k_disb_date
  from kontract
  where k_id = :k_id_tmp
  into :k_orig_kontract_no, :k_disb_date;

-- условия договора
  select k_amount_approved, k_kolich_repay, k_percent, k_stop_perc
  from kontract
  where k_id = :k_id
  into :k_amount_approved, k_kolich_repay, k_percent, k_date_stop_perc;

  select sum(-ok_os - ok_perc - ok_fine)
  from operkred
  where ok_type = 4 and
        ok_ol_is_active = 1 and
        ok_from_k_id = :k_id and
        ok_ol_date <= :rep_date and
        ok_from_c_id = 0
  into :kdi_payed_total;
  if (kdi_payed_total is null) then
    kdi_payed_total = 0;

  kdi_del_days = 0;
  select min(iif(sh_calc_fine + sh_calc_perc + zad_os <> 0, sh_date, :rep_date)), sum(zad_os), sum(zad_perc),
         sum(sh_calc_perc), sum(sh_calc_fine)
  from aarep_make_zad_list(:k_id, :rep_date, 1)
  where sh_date is not null
  into :sh_date, :kdi_zad_os_now, :kdi_zad_perc_now, :kdi_calc_perc_now, :kdi_calc_fine_now;
  kdi_del_days = :rep_date - sh_date;
  if (kdi_del_days is null) then
    kdi_del_days = 0;

  if ((abs(kdi_calc_perc_now - kdi_zad_perc_now)) <= 1) then
    kdi_calc_perc_now = kdi_zad_perc_now;

  kdi_sum_total_now = kdi_zad_os_now + kdi_calc_perc_now + kdi_calc_fine_now;
  kdi_sum_total_now_kg = round(cast(kdi_sum_total_now * rep_curr_rate as decimal(18,2)), 0);

  k_clients_fio = '';
  k_clients_fio_address = '';
  k_clients_fio_address2 = '';
  k_clients_fio_address_birth = '';

  kol = 0;
  for select c_id, c_name_fio, c_birth_date
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_from_c_id = c_id and
            kc_amount_approved > 0
      into :c_id, :c_name_fio, :c_birth_date
  do
  begin
    c_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :c_id and
          adc_address_type = 4 and
          adc_from_adl_id = adl_id
    into :c_address;

    if (c_address is null) then
      c_address = '';

    if (c_address = '') then
    begin
      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      into :c_address;
    end
    if (c_address is null) then
      c_address = '';

    c_birth_date_str = '';
    if (extract(day from c_birth_date) >= 10) then
    begin
      if (extract(month from c_birth_date) >= 10) then
        c_birth_date_str = extract(day from c_birth_date) || '.' || extract(month from c_birth_date) || '.' || extract(year from c_birth_date);
      else
        c_birth_date_str = extract(day from c_birth_date) || '.0' || extract(month from c_birth_date) || '.' || extract(year from c_birth_date);
    end
    else
    begin
      if (extract(month from c_birth_date) >= 10) then
        c_birth_date_str = '0' || extract(day from c_birth_date) || '.' || extract(month from c_birth_date) || '.' || extract(year from c_birth_date);
      else
        c_birth_date_str = '0' || extract(day from c_birth_date) || '.0' || extract(month from c_birth_date) || '.' || extract(year from c_birth_date);
    end

    kol = kol + 1;
    if (k_clients_fio = '') then
      k_clients_fio = c_name_fio;
    else
      k_clients_fio = k_clients_fio || ', ' || c_name_fio;

    if (k_clients_fio_address = '') then
      k_clients_fio_address = 'Должник ' || kol || ': ' || c_name_fio || ', ' || c_birth_date_str || ' г.р.' || ascii_char(10) || 'Адрес: ' || c_address;
    else
      k_clients_fio_address = k_clients_fio_address || ascii_char(10) || ascii_char(10) || 'Должник ' || kol || ': ' || c_name_fio || ', ' || c_birth_date_str || ' г.р.' || ascii_char(10) || 'Адрес: ' || c_address;

    if (k_clients_fio_address2 = '') then
      k_clients_fio_address2 = c_name_fio || ', проживающий по адресу: ' || c_address;
    else
      k_clients_fio_address2 = k_clients_fio_address2 || ', ' || c_name_fio || ', проживающий по адресу: ' || c_address;

    if (k_clients_fio_address_birth = '') then
      k_clients_fio_address_birth = c_name_fio || ', ' || c_birth_date_str || ' г.р., уроженец Кыргызстана, проживающий по адресу: ' || c_address;
    else
      k_clients_fio_address_birth = k_clients_fio_address_birth || ', ' || c_name_fio || ', ' || c_birth_date_str || ' г.р., уроженец Кыргызстана, проживающий по адресу: ' || c_address;

  end

  kdi_docs = '';
  for select ol.ol_id, ol.ol_parent_id, ol.ol_doc_no, ol.ol_date,
             (
             case ok_type
               when 8 then 'Расходный'
               when 4 then 'Приходный'
             end) as doc_type,
             sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_kommis)
      from operkred ok, operlist ol
      where ol.ol_id = ok.ok_from_ol_id and
            ok.ok_ol_is_active = 1 and
            ok.ok_type in (8, 4) and
            ok.ok_from_k_id = :k_id and
            ok_ol_date <= :rep_date and
            ok_from_c_id = 0
      group by ol.ol_id, ol.ol_parent_id, ol_doc_no, ol.ol_date, ok_type
      into :ol_id, :ol_parent_id, :ol_doc_no, :ol_date, :doc_type, :ok_sum
  do
  begin
    ol_date_str = '';
    if (extract(day from ol_date) >= 10) then
    begin
      if (extract(month from ol_date) >= 10) then
        ol_date_str = extract(day from ol_date) || '.' || extract(month from ol_date) || '.' || extract(year from ol_date);
      else
        ol_date_str = extract(day from ol_date) || '.0' || extract(month from ol_date) || '.' || extract(year from ol_date);
    end
    else
    begin
      if (extract(month from ol_date) >= 10) then
        ol_date_str = '0' || extract(day from ol_date) || '.' || extract(month from ol_date) || '.' || extract(year from ol_date);
      else
        ol_date_str = '0' || extract(day from ol_date) || '.0' || extract(month from ol_date) || '.' || extract(year from ol_date);
    end

    if (ol_parent_id is not null) then
    begin
      select ol_doc_no
      from operlist ol
      where ol.ol_id = :ol_parent_id
      into :ol_doc_no;
    end
    if (kdi_docs = '') then
      kdi_docs = '№ ' || ol_doc_no || ' от ' || ol_date_str || ' года (' || doc_type || ')';
    else
      kdi_docs = kdi_docs || ascii_char(10) || '№ ' || ol_doc_no || ' от ' || ol_date_str || ' года (' || doc_type || ')';
  end

  suspend;

end
