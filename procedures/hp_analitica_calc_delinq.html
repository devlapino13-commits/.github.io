CREATE PROCEDURE HP_ANALITICA_CALC_DELINQ(
    K_ID DECIMAL(18,0),
    K_DISB_DATE DATE,
    K_DISB_AMOUNT DECIMAL(18,4),
    REP_DATE_END DATE,
    REP_DELINQ_DAYS INTEGER)
RETURNS (
    DELINQ_00 DECIMAL(18,4),
    DELINQ_01 DECIMAL(18,4),
    DELINQ_02 DECIMAL(18,4),
    DELINQ_03 DECIMAL(18,4),
    DELINQ_04 DECIMAL(18,4),
    DELINQ_05 DECIMAL(18,4),
    DELINQ_06 DECIMAL(18,4),
    DELINQ_07 DECIMAL(18,4),
    DELINQ_08 DECIMAL(18,4),
    DELINQ_09 DECIMAL(18,4),
    DELINQ_10 DECIMAL(18,4),
    DELINQ_11 DECIMAL(18,4),
    DELINQ_12 DECIMAL(18,4),
    DELINQ_13 DECIMAL(18,4),
    AMOUNT_00 DECIMAL(18,4),
    AMOUNT_01 DECIMAL(18,4),
    AMOUNT_02 DECIMAL(18,4),
    AMOUNT_03 DECIMAL(18,4),
    AMOUNT_04 DECIMAL(18,4),
    AMOUNT_05 DECIMAL(18,4),
    AMOUNT_06 DECIMAL(18,4),
    AMOUNT_07 DECIMAL(18,4),
    AMOUNT_08 DECIMAL(18,4),
    AMOUNT_09 DECIMAL(18,4),
    AMOUNT_10 DECIMAL(18,4),
    AMOUNT_11 DECIMAL(18,4),
    AMOUNT_12 DECIMAL(18,4),
    AMOUNT_13 DECIMAL(18,4))
AS
declare variable yy integer;
declare variable mm integer;
declare variable i integer;
declare variable is_delinq integer;
declare variable oks_date_begin date;
declare variable oks_date_end date;
declare variable oks_balance decimal(18,4) = 0;
declare variable oks_delinq_date date;
declare variable dd date;
begin
  delinq_00 = 0;
  amount_00 = 0;
  delinq_01 = 0;
  amount_01 = 0;
  delinq_02 = 0;
  amount_02 = 0;
  delinq_03 = 0;
  amount_03 = 0;
  delinq_04 = 0;
  amount_04 = 0;
  delinq_05 = 0;
  amount_05 = 0;
  delinq_06 = 0;
  amount_06 = 0;
  delinq_07 = 0;
  amount_07 = 0;
  delinq_08 = 0;
  amount_08 = 0;
  delinq_09 = 0;
  amount_09 = 0;
  delinq_10 = 0;
  amount_10 = 0;
  delinq_11 = 0;
  amount_11 = 0;
  delinq_12 = 0;
  amount_12 = 0;
  delinq_13 = 0;
  amount_13 = 0;

  for select oks_date_begin, oks_date_end, oks_balance, oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            oks_close_state = 0 and
            oks_balance > 0 and
            oks_date_begin <= :rep_date_end
      into oks_date_begin, oks_date_end, oks_balance, oks_delinq_date
  do
  begin
    i = -1;
    dd = k_disb_date - extract(day from k_disb_date);
    yy = extract(year from dd);
    mm = extract(month from dd);
    while (i < 13 and dd < rep_date_end) do
    begin
      i = i + 1;
      if (mm < 12) then
        mm = mm + 1;
      else
      begin
        yy = yy + 1;
        mm = 1;
      end
      dd = cast('01.' || right(100 + mm, 2) || '.' || yy as date);
      dd = dateadd(month, 1, :dd) - 1;
      is_delinq = 0;

      if (dd between oks_date_begin and oks_date_end and
          datediff(day, oks_delinq_date, dd) > :rep_delinq_days and
          dd <= rep_date_end) then
        is_delinq = 1;

      if (i = 0) then
      begin
        if (is_delinq = 1) then
          delinq_00 = oks_balance;
        amount_00 = k_disb_amount;
      end
      if (i = 1) then
      begin
        if (is_delinq = 1) then
          delinq_01 = oks_balance;
        amount_01 = k_disb_amount;
      end
      if (i = 2) then
      begin
        if (is_delinq = 1) then
          delinq_02 = oks_balance;
        amount_02 = k_disb_amount;
      end
      if (i = 3) then
      begin
        if (is_delinq = 1) then
          delinq_03 = oks_balance;
        amount_03 = k_disb_amount;
      end
      if (i = 4) then
      begin
        if (is_delinq = 1) then
          delinq_04 = oks_balance;
        amount_04 = k_disb_amount;
      end
      if (i = 5) then
      begin
        if (is_delinq = 1) then
          delinq_05 = oks_balance;
        amount_05 = k_disb_amount;
      end
      if (i = 6) then
      begin
        if (is_delinq = 1) then
          delinq_06 = oks_balance;
        amount_06 = k_disb_amount;
      end
      if (i = 7) then
      begin
        if (is_delinq = 1) then
          delinq_07 = oks_balance;
        amount_07 = k_disb_amount;
      end
      if (i = 8) then
      begin
        if (is_delinq = 1) then
          delinq_08 = oks_balance;
        amount_08 = k_disb_amount;
      end
      if (i = 9) then
      begin
        if (is_delinq = 1) then
          delinq_09 = oks_balance;
        amount_09 = k_disb_amount;
      end
      if (i = 10) then
      begin
        if (is_delinq = 1) then
          delinq_10 = oks_balance;
        amount_10 = k_disb_amount;
      end
      if (i = 11) then
      begin
        if (is_delinq = 1) then
          delinq_11 = oks_balance;
        amount_11 = k_disb_amount;
      end
      if (i = 12) then
      begin
        if (is_delinq = 1) then
          delinq_12 = oks_balance;
        amount_12 = k_disb_amount;
      end
      if (i = 13) then
      begin
        if (is_delinq = 1) then
          delinq_13 = oks_balance;
        amount_13 = k_disb_amount;
      end
    end
  end
  suspend;
end
