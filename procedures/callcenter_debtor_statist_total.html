CREATE PROCEDURE CALLCENTER_DEBTOR_STATIST_TOTAL(
    DATE_BEGIN TIMESTAMP,
    DATE_END TIMESTAMP)
RETURNS (
    NEGATIVE INTEGER,
    POSITIVE INTEGER,
    NOTCALLED INTEGER,
    TOTAL INTEGER,
    KONTRACTS_COUNT INTEGER,
    CONTACTS_COUNT INTEGER,
    NEGATIVE_P DECIMAL(18,2),
    POSITIVE_P DECIMAL(18,2),
    NOTCALLED_P DECIMAL(18,2))
AS
declare variable d1 date;
declare variable d2 date;
declare variable ccdr_result smallint;
declare variable k_id decimal(18,0);
declare variable contacts_count_k integer;
declare variable kontracts_count_k integer;
begin
  /*
    CREATED BY: IGUSEV 29.06.2012

    PURPOSE: Итоговая статистика по обзвону клиентов - должников

    24.07.2012 CHANGED BY IGUSEV
    CAHNGES:   Оптимизирована выборка данных, реализована привязка к kontstate 

    7.08.2012 CHANGED BY IGUSEV
    CAHNGES: Добавлено условие    and ((cdbl.ccdl_exclude_date <=  :d2 + 1) or (cdbl.ccdl_exclude_date is null))  - дата когда оператор исключил контакт из списка текущих

  */
  d1 = cast(date_begin as date);
  d2 = cast(date_end as date);

  negative = 0;
  positive = 0;
  notcalled = 0;
  total = 0;

  for select cdr.ccdr_result
      from callcenter_debtors_result cdr
      where cdr.ccdr_call_datetime >= :d1 and
            cdr.ccdr_call_datetime < :d2 + 1 and
            cdr.ccdr_is_auto = 0
      order by cdr.ccdr_result
      into :ccdr_result
  do
  begin
    if (ccdr_result = 0) then
      negative = negative + 1;
    if (ccdr_result = 1) then
      positive = positive + 1;
    if (ccdr_result = -1) then
      notcalled = notcalled + 1;
    total = total + 1;
  end

  if (total > 0) then
  begin
    negative_p = negative * 100 / total;
    positive_p = positive * 100 / total;
    notcalled_p = notcalled * 100 / total;
  end
  else
  begin
    negative_p = 0;
    positive_p = 0;
    notcalled_p = 0;
  end

  if (ccdr_result is null) then
    ccdr_result = 0;

  kontracts_count = 0;
  contacts_count = 0;

  for select distinct cdbl.ccdl_from_k_id as k_id
      from callcenter_debtors_list cdbl
      inner join kontstate ks on ccdl_from_k_id = ks_from_k_id
      where ((cdbl.ccdl_deliq_closed_date is null) or
            (cdbl.ccdl_deliq_closed_date < :d2)) /* не погашенные*/
            and
            ((cdbl.ccdl_timeout_date is null) or
            (cdbl.ccdl_timeout_date < :d2)) /* только должники/конт. лица которые не исчесчезли по timeout-у*/
            and
            ks_ol_is_delete = 0 and
            ((cdbl.ccdl_exclude_date <= :d2 + 1) or
            (cdbl.ccdl_exclude_date is null)) /* дата когда оператор исключил контакт из списка текущих */
            and
            ks_state = 1 /*только активные*/
            and
            :date_end between ks_date_begin and ks_date_end
      into :k_id
  do
  begin
    kontracts_count = kontracts_count + 1;
  end

  select count(cdbl.ccdl_from_c_id)
  from callcenter_debtors_list cdbl
  inner join kontstate ks on ccdl_from_k_id = ks_from_k_id
  where ((cdbl.ccdl_deliq_closed_date is null) or
        (cdbl.ccdl_deliq_closed_date < :d2)) /* не погашенные*/
        and
        ((cdbl.ccdl_timeout_date is null) or
        (cdbl.ccdl_timeout_date < :d2)) /* только должники/конт. лица которые не исчесчезли по timeout-у т.к. им звонили на протяжении 3 дней*/
        and
        ks_ol_is_delete = 0 and
        ((cdbl.ccdl_exclude_date <= :d2 + 1) or
        (cdbl.ccdl_exclude_date is null)) /* дата когда оператор исключил контакт из списка текущих */
        and
        ks_state = 1 /*только активные*/
        and
        :date_end between ks_date_begin and ks_date_end
  into :contacts_count;

  suspend;
end
