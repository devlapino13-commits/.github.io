CREATE PROCEDURE HP_JSON_ACTION(
    ID D_STR_CODE,
    IN_CLASS_NAME D_STR_CODE,
    IN_COMMAND D_STR_CODE,
    IN_ITEM_NAME D_STR_255,
    IN_ITEM_JSON D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_ID D_STR_CODE,
    RES_ERROR_TEXT D_STR_255)
AS
begin
  res_id = 0;
  res_error_text = '';

  if (coalesce(in_class_name, '') = '') then
    exception e_default 'in_class_name param is empty';

  if (id is null or id = '') then id = '0';

  if (in_command not in -- допустимык действия для справочников
   ('doLoad',
    'doUpload',
    'doSave')) then
    exception e_default 'Incorrect param for: '||in_class_name||', command: '||coalesce(in_command, '<NULL>');


  -- doLoad используется только для предзаполнения полей для нового создаваемого элемента
  -- Поэтому при doLoad запрещено передавать ID <> 0
  if (in_command = 'doLoad' and id <> '0') then
    exception e_default 'Loading data by JSON: expecting ID 0, actual ID: '||Id;

  if (in_command = 'doUpload') then
  begin
    if (coalesce(in_item_name, '') = '') then
      exception e_default 'in_item_name param is empty';

    if (coalesce(in_item_json, '') = '') then
      exception e_default 'in_item_json param is empty';

    insert into temp_json_action (ja_name, ja_json)
    values (:in_item_name, :in_item_json);

    suspend;
    Exit;
  end

  if (in_command = 'doSave') then
  begin
    if (not exists(select * from temp_json_action rows 1)) then
      exception e_default 'Items is empty';
  end

  if (in_class_name = 'TAdminOperNewForm') then
  begin
    select res_id, res_error_text
    from hp_json_action_memorder(:id, :in_class_name, :in_command, :in_item_json, :un_id)
    into res_id, res_error_text;
  end
  if (in_class_name = 'TRefAccEditForm') then
  begin
    select res_id, res_error_text
    from hp_json_action_refaccount(:id, :in_class_name, :in_command, :in_item_json, :un_id)
    into res_id, res_error_text;
  end

  Suspend;
end
