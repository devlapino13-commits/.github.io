CREATE PROCEDURE FIND_US_REF_MONEYOPER(
    FIND_ID D_STR_CODE,
    FIND_TEXT D_STR_255,
    ADD_PARAMS D_STR_8100,
    UN_ID D_ID)
RETURNS (
    RES_ID D_ID,
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_LONG)
AS
declare variable num_1 d_str_255;
declare variable txt d_str_255;
declare variable max_row d_integer;
begin
  if (un_id is null) then
    un_id = 0;

  if (find_text is null) then
    find_text = '';

  if (find_id is null) then
    find_id = '';

  if (find_id <> '' and
      find_id <> '0' and
      find_text = '') then
  begin
    select rmo_id, rmo_code, f_objname('ref_moneyoper', rmo_id)
    from ref_moneyoper
    where rmo_id = :find_id
    into res_id, res_code, res_text;

    suspend;

    exit;
  end

  max_row = 500;

  find_text = replace(find_text, '.', '');

  select num_1, txt
  from find_editor_parcer(:find_text)
  into :num_1, :txt;

  if (num_1 <> '') then
  begin
    for select rmo_id, rmo_code, f_objname('ref_moneyoper', rmo_id)
    from ref_moneyoper
    where rmo_code starting :num_1 and
          rmo_name containing :txt
    rows :max_row
    into res_id, res_code, res_text
    do
      Suspend;

    Exit;
  end

  if (txt <> '') then
  begin
    for select rmo_id, rmo_code, f_objname('ref_moneyoper', rmo_id)
    from ref_moneyoper
    where rmo_name containing :txt
    rows :max_row
    into res_id, res_code, res_text
    do
      Suspend;
  end
end
