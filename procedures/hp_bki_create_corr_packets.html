CREATE PROCEDURE HP_BKI_CREATE_CORR_PACKETS(
    BKK_ID D_ID NOT NULL)
AS
declare variable object_id d_id;
declare variable object_kind d_integer;
declare variable bkb_date d_date;
declare variable object_kind_kontract d_integer;
declare variable object_kind_miniapp d_integer;
begin
  object_kind_kontract = f_objkind('kontract', 'id');
  object_kind_miniapp  = f_objkind('miniapp',  'id');

  for select bkb_object_id, bkb_object_kind, bkb_date
  from bki_buffer
  where bkb_tag = 25 --perstemp
    and bkb_from_bkk_id = :bkk_id
    and bkb_object_kind in (:object_kind_kontract, :object_kind_miniapp)
  into :object_id, object_kind, bkb_date
  do
  begin
    if (:bkk_id = 1 and
       (exists
         (select distinct bkp.bkp_from_bkk_id
          from bki_loans bkl
          inner join bki_packet bkp on bkp.bkp_id = bkl.bkl_from_bkp_id
          where bkl.bkl_object_id = :object_id
            and bkl.bkl_object_kind = :object_kind
            and bkp.bkp_date < :bkb_date
            and bkp_from_bkk_id = 2))) then
      execute procedure hp_bki_add_to_buffer('.hp_bki_get_packet.nbki.', :object_kind, :object_id, :bkb_date);

    execute procedure hp_bki_corr_event_3_1(:object_id, :object_kind, :bkk_id, :bkb_date);
  end
end
