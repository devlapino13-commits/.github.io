CREATE PROCEDURE HP_DO_KN_OPERBOOK_FIX_TAX(
    REP_DATE D_DATE)
RETURNS (
    TOTAL_LINE D_INTEGER,
    CHANGED_LINE D_INTEGER)
AS
declare variable kob_id d_id;
declare variable kob_summa d_currency;
declare variable kob_summa_tax_dt d_currency;
declare variable kob_summa_tax_kt d_currency;
declare variable tax_kind_dt d_smallint;
declare variable tax_kind_kt d_smallint;
declare variable orig_summa_tax_dt d_currency;
declare variable orig_summa_tax_kt d_currency;
declare variable b_acc_dt d_str_code;
declare variable b_acc_kt d_str_code;
begin
  total_line = 0;
  changed_line = 0;
  for select kob_id, kob_summa, kob_summa_tax_dt, kob_summa_tax_kt,
             b_dt.ab_tax_kind,
             b_kt.ab_tax_kind,
             b_dt.ab_code,
             b_kt.ab_code
      from kn_operbook, ref_account a_dt, ref_account a_kt, ref_accbalance b_dt, ref_accbalance b_kt
      where kob_ol_date >= :rep_date and
            kob_acc_dt = a_dt.acc_id and
            kob_acc_kt = a_kt.acc_id and
            a_dt.acc_from_add_ab_id = b_dt.ab_id and
            a_kt.acc_from_add_ab_id = b_kt.ab_id and
            kob_acc_dt <> 0 and
            kob_acc_kt <> 0
      into kob_id, kob_summa, orig_summa_tax_dt, orig_summa_tax_kt, tax_kind_dt, tax_kind_kt, b_acc_dt, b_acc_kt
  do
  begin
    if (b_acc_dt = '60324 (23)' and b_acc_kt = '60323 (20)') then
    begin
      update kn_operbook
      set kob_summa_tax_dt = 0,
          kob_summa_tax_kt = 0
      where kob_id = :kob_id;

      changed_line = changed_line + 1;
      Continue;
    end

    if (b_acc_dt = '71802 (556061)' and b_acc_kt = '60323 (20)') then
      Continue;

    total_line = total_line + 1;
    kob_summa_tax_dt = orig_summa_tax_dt;
    kob_summa_tax_kt = orig_summa_tax_kt;

    if (tax_kind_dt = 0) then
      kob_summa_tax_dt = kob_summa;
    if (tax_kind_dt = 1) then
      kob_summa_tax_dt = 0;

    if (tax_kind_kt = 0) then
      kob_summa_tax_kt = kob_summa;
    if (tax_kind_kt = 1) then
      kob_summa_tax_kt = 0;

    if (tax_kind_dt = 2 and
        tax_kind_kt in (0, 1)) then
      kob_summa_tax_dt = kob_summa_tax_kt;
    if (tax_kind_kt = 2 and
        tax_kind_dt in (0, 1)) then
      kob_summa_tax_kt = kob_summa_tax_dt;

    if (tax_kind_dt = 2 and
        tax_kind_kt = 2 and
        kob_summa <> 0) then
    begin
      if (kob_summa_tax_dt <> 0 and
          kob_summa_tax_dt <> kob_summa) then
        kob_summa_tax_dt = kob_summa;
      if (kob_summa_tax_kt <> 0 and
          kob_summa_tax_kt <> kob_summa) then
        kob_summa_tax_kt = kob_summa;
    end

    if (orig_summa_tax_dt <> kob_summa_tax_dt or orig_summa_tax_kt <> kob_summa_tax_kt) then
    begin
      update kn_operbook
      set kob_summa_tax_dt = :kob_summa_tax_dt,
          kob_summa_tax_kt = :kob_summa_tax_kt
      where kob_id = :kob_id;

      changed_line = changed_line + 1;
    end
  end
  suspend;
end
