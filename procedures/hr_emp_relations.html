CREATE PROCEDURE HR_EMP_RELATIONS(
    ORDER_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE DATE,
    C_PLASTCARD_NO VARCHAR(30),
    REL_NAME D_STR_255,
    REL_ADDRESS D_STR_2048)
AS
begin
  for
    select
        rel.c_name_fio,
        rel.c_birth_date,
        rel.c_plastcard_no,
        rel.rel_name,
        adl.adl_address_str
    from hr_pstatus ps
    inner join hp_client_relation_list(ps.ps_from_c_id, 0) rel on 1 = 1
    left join addrclient adcl on adcl.adc_from_c_id = rel.crel_other_c_id and adcl.adc_address_type = 1
    left join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
    where ps.ps_from_ho_id = :order_id
    and rel.crel_is_active = 1
    and rel.cnt_country_name <> 'xxx'
    into
        :c_name_fio,
        :c_birth_date,
        :c_plastcard_no,
        :rel_name,
        :rel_address
  do
    suspend;
end
