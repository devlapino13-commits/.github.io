CREATE PROCEDURE AAREP_TAX_NSP_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    TAX_KIND VARCHAR(50),
    TAX_BASE DECIMAL(18,4),
    TAX_KOEF DECIMAL(15,7),
    TAX_REGION_ID INTEGER)
AS
begin
  for select id_list, txn_percent
      from region_orig_list(:report_region_id, 1, 3, 1), taxnsp
      where txn_percent <> 0
      into :tax_region_id, :tax_koef
  do
  begin
    tax_base = 0;
    tax_kind = 'Ставка ' || cast(:tax_koef as decimal(15,1)) || ' %:';
    suspend;
  end

  for select ol_from_reg_id, txn_percent, sum(ob_summa)
      from taxnsp, operlist, operbook, acc_list, region_list(:report_region_id, 1, -1, 1)
      where ol_id = ob_from_ol_id and
            al_id = ob_account_kt and
            (ol_from_reg_id = id_list) and
            al_from_bg_id = txn_from_bg_id and
            al_number >= txn_acc_from and
            al_number <= txn_acc_till and
            operlist.is_delete = 0 and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            txn_percent <> 0
      group by ol_from_reg_id, txn_percent
      order by txn_percent, ol_from_reg_id
      into :tax_region_id, :tax_koef, :tax_base
  do
  begin
    tax_kind = 'Ставка ' || cast(:tax_koef as decimal(15,1)) || ' %:';
    suspend;
  end
  for select ol_from_reg_id, txn_percent, sum(ob_summa)
      from taxnsp, operlist, operbook, acc_list, region_list(:report_region_id, 1, -1, 1)
      where ol_id = ob_from_ol_id and
            al_id = ob_account_dt and
            txn_from_bg_id <> '61011' and
            (ol_from_reg_id = id_list) and
            al_from_bg_id = txn_from_bg_id and
            al_number >= txn_acc_from and
            al_number <= txn_acc_till and
            operlist.is_delete = 0 and
            ol_date_oper_day between :report_date_begin and :report_date_end and
            txn_percent <> 0
      group by ol_from_reg_id, txn_percent
      order by txn_percent, ol_from_reg_id
      into :tax_region_id, :tax_koef, :tax_base
  do
  begin
    tax_base = 0 - :tax_base;
    tax_kind = 'Ставка ' || cast(:tax_koef as decimal(15,1)) || ' %:';
    suspend;
  end
end
