        values (:ad_id, :cl_delinq_00, :cl_delinq_01, :cl_delinq_02, :cl_delinq_03, :cl_delinq_04, :cl_delinq_05,
                :cl_delinq_06, :cl_delinq_07, :cl_delinq_08, :cl_delinq_09, :cl_delinq_10, :cl_delinq_11, :cl_delinq_12,

      end
    end

  end

  begin -- данные на уровне группы
    for select ks_from_k_id, ks_region_id, ks_kc_id, 0,
               iif(k_index_curr_code = :def_curr, k_amount_approved, round(k_amount_approved * k_index_curr_rate, 2)),
               0, k_disb_date
        from kontstate, temp_analitica_regions, kontract
        where ks_date_begin between :rep_include_closed and (:rep_date_begin - 1) and
              ks_state = 2 and
              ks_ol_is_delete = 0 and
              ks_region_id = reg_id and
              ks_from_k_id = k_id and
              k_disb_date >= :rep_include_closed
        order by ks_kc_id
        into :res_k_id, :reg_id, :lo_id, :c_id, :cl_loan_amount, cl_circle, k_disb_date
    do
    begin
      ad_id = ad_id + 1;
      if (cl_circle <= 0) then
        cl_circle = 1;

      if (lo_id <> lo_id_prior) then
      begin
        lo_id_prior = lo_id;
        update or insert into temp_analitica_lo (tal_from_ap_id, tal_lo_id)
        values (:rep_period_id, :lo_id)
        matching (tal_from_ap_id, tal_lo_id);
      end

      insert into temp_analitica_data (ad_id, ad_date_begin, ad_date_end, ad_from_ap_id, ad_region_id, ad_lo_id,

      values (:ad_id, :rep_date_begin, :rep_date_end, :rep_period_id, :reg_id, :lo_id, :res_k_id, :c_id,

      if (rep_kind = 'delinq') then
      begin

        insert into temp_analitica_delinq (aq_from_ad_id, aq_delinq_00, aq_delinq_01, aq_delinq_02, aq_delinq_03,

                                           aq_amount_10, aq_amount_11, aq_amount_12, aq_amount_13)

               amount_03, amount_04, amount_05, amount_06, amount_07, amount_08, amount_09, amount_10, amount_11,
               amount_12, amount_13
        from hp_analitica_calc_delinq(:res_k_id, :k_disb_date, :cl_loan_amount, :rep_date_end, :rep_delinq_days);
      end
    end
    exit;
  end
