CREATE PROCEDURE AAREP_KONTCALCPERC_CHECK
RETURNS (
    REP_DD DATE,
    REP_REGION_ID INTEGER,
    REP_REGION_NAME VARCHAR(100),
    REP_ACCOUNT VARCHAR(20),
    REP_KONT_SALDO DECIMAL(18,4),
    REP_BALANCE_SALDO DECIMAL(18,4),
    REP_LOAN_COUNT INTEGER,
    REP_DIFF DECIMAL(18,4),
    REP_DIFF_AVG DECIMAL(18,4))
AS
begin
  for select rep_dd, rep_region_id, rep_region_name, rep_account, sum(rep_kont_saldo), sum(rep_balance_saldo),
             sum(rep_loan_count)
      from aarep_kontcalcperc_check_tmp
      group by rep_dd, rep_region_id, rep_region_name, rep_account
      into :rep_dd, :rep_region_id, :rep_region_name, :rep_account, :rep_kont_saldo, :rep_balance_saldo,
           :rep_loan_count
  do
  begin
    if ((rep_kont_saldo <> 0) or (rep_balance_saldo <> 0)) then
    begin
      rep_diff = rep_balance_saldo - rep_kont_saldo;
      if (rep_loan_count <> 0) then
        rep_diff_avg = cast(rep_diff / :rep_loan_count as decimal(15,2));
      else
        rep_diff_avg = 0;
      suspend;
    end
  end
end
