CREATE PROCEDURE CB_AZ_LO_201602_ONE_LO_2_OLIMP(
    REP_REGION_ID INTEGER,
    REP_C_ID DECIMAL(18,0),
    REP_DATE_END DATE,
    NEED_ONE_DAY INTEGER)
RETURNS (
    AAREP_DATE DATE,
    AAREP_RETURNMENT_KOEF DECIMAL(18,7),
    AAREP_BONUS_PLUS DECIMAL(18,4),
    AAREP_BONUS_MINUS DECIMAL(18,4),
    AAREP_BONUS_TOTAL DECIMAL(18,4),
    AAREP_DAY_RETURNMENT_KOEF DECIMAL(18,7),
    AAREP_DAY_BONUS_PLUS DECIMAL(18,4),
    AAREP_DAY_BONUS_MINUS DECIMAL(18,4),
    AAREP_DAY_BONUS_TOTAL DECIMAL(18,4),
    AAREP_BONUS_IDEAL DECIMAL(18,4),
    AAREP_DELINQ_OS DECIMAL(18,4),
    AAREP_DELINQ_PERC DECIMAL(18,4),
    AAREP_FACT_OS DECIMAL(18,4),
    AAREP_FACT_PERC DECIMAL(18,4),
    AAREP_FACT_FINE DECIMAL(18,4),
    AAREP_FACT_DELINQ_OS DECIMAL(18,4),
    AAREP_FACT_DELINQ_PERC DECIMAL(18,4),
    AAREP_FACT_ADV_OS DECIMAL(18,4),
    AAREP_FACT_ADV_PERC DECIMAL(18,4),
    AAREP_MEMO VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    REG_NAME VARCHAR(100),
    AAREP_OKLAD DECIMAL(18,4),
    AAREP_BONUS_TOTAL_TO_PAY DECIMAL(18,4),
    AAREP_BONUS_TOTAL_FACT DECIMAL(18,4),
    AAREP_PORT_BALANCE_ALL DECIMAL(18,4),
    AAREP_PORT_KOLICH_ALL INTEGER,
    AAREP_PAR_BALANCE1 DECIMAL(18,4),
    AAREP_PAR_BALANCE7 DECIMAL(18,4),
    AAREP_PORT_BALANCE_OWN DECIMAL(18,4),
    AAREP_PORT_KOLICH_OWN INTEGER,
    AAREP_PORT_KOLICH_EX_TRANFER INTEGER,
    AAREP_CLIENTS_QTY INTEGER,
    AAREP_LO_STATUS STR_100,
    AAREP_FACT_KOMMIS DECIMAL(18,4),
    JK_NAME VARCHAR(100),
    C_ID DECIMAL(18,0),
    REG_ID INTEGER,
    LO_STAGE_MONTH DECIMAL(4,2))
AS
declare variable AAREP_PORT_KOLICH_ALL_TMP integer;
declare variable AAREP_PAR_BALANCE1_TMP decimal(18,4);
declare variable AAREP_PAR_BALANCE7_TMP decimal(18,4);
declare variable AAREP_PORT_BALANCE_ALL_TMP decimal(18,4);
declare variable AAREP_MONTH_PLAN_OS decimal(18,4) = 0;
declare variable AAREP_MONTH_PLAN_PERC decimal(18,4) = 0;
declare variable AAREP_PLAN_ALL decimal(18,4);
declare variable AAREP_FACT_ALL decimal(18,4);
declare variable AAREP_FACT_PLANNED decimal(18,4);
declare variable AAREP_FACT_OTHER decimal(18,4);
declare variable KOEF_BONUS decimal(18,4) = 0;
declare variable KOEF_PENALTY decimal(18,4) = 0;
declare variable AAREP1_PLAN_ALL decimal(18,4) = 0;
declare variable AAREP1_FACT_ALL decimal(18,4) = 0;
declare variable AAREP1_FACT_PLANNED decimal(18,4) = 0;
declare variable AAREP1_FACT_OTHER decimal(18,4) = 0;
declare variable C_EMAIL varchar(100);
declare variable K_ID_TMP ID_TYPE;
declare variable K_DISB_DATE_TMP date;
declare variable CLIENTS_QTY_TMP integer;
begin
  koef_bonus = 90.0;
  koef_penalty = iif(:rep_date_end = '31.03.2022', 400.0, 300.0);
  aarep_port_kolich_ex_tranfer = 0;

  for select ps_from_c_id, c_email, ps_from_reg_id, jk_name
      from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1)
      where :rep_c_id = 0 and
            ps_from_c_id = c_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id > 1000000000003 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner', 'JobSystem.Region_KM') and
            ps_from_reg_id = id_list
      union
      select c_id, c_email, :rep_region_id, ''
      from clients
      where :rep_c_id <> 0 and
            c_id = :rep_c_id
      order by 3
      into :c_id, :c_email, :reg_id, jk_name
  do
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    aarep_lo_status = '';

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into :reg_name;

    aarep_memo = '';
    if (c_email is null) then
      c_email = '';
    if (c_email like '%@mbulak.az') then
      aarep_memo = 'Ok';
    else
      aarep_memo = 'Не будет отправлено, нет EMAIL';

    aarep_port_balance_all = 0;
    aarep_port_kolich_all = 0;

    aarep_par_balance1 = 0;
    aarep_par_balance7 = 0;

    aarep_port_balance_all_tmp = 0;
    aarep_port_kolich_all_tmp = 0;
    aarep_par_balance1_tmp = 0;
    aarep_par_balance7_tmp = 0;

    aarep_clients_qty = 0;

    for select kont.k_id, kont.k_disb_date, sum(oks_balance), sum(k_kolich_client_approved),
               sum(iif(:rep_date_end - oks_delinq_date > 1, oks_balance, 0.0)),
               sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0.0))
        from oksummary, kontract kont
        where oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_kc_id = :c_id and
              oks_balance <> 0 and
              oks_from_k_id = k_id
              --      and (:rep_date_end-oks_delinq_date<=180)
              and
              oks_close_state = 0
        group by k_id, k_disb_date
        into :k_id_tmp, :k_disb_date_tmp, :aarep_port_balance_all_tmp, :aarep_port_kolich_all_tmp,
             :aarep_par_balance1_tmp, :aarep_par_balance7_tmp
    do
    begin
      aarep_port_balance_all = aarep_port_balance_all + aarep_port_balance_all_tmp;
      aarep_port_kolich_all = aarep_port_kolich_all + aarep_port_kolich_all_tmp;
      aarep_par_balance1 = aarep_par_balance1 + aarep_par_balance1_tmp;
      aarep_par_balance7 = aarep_par_balance7 + aarep_par_balance7_tmp;

      select res_just_client_count
      from hp_get_just_client_in_loan(:rep_date_end, :k_id_tmp, :k_disb_date_tmp)
      into clients_qty_tmp;

      if (clients_qty_tmp is null) then
        clients_qty_tmp = 0;

      aarep_clients_qty = aarep_clients_qty + clients_qty_tmp;
    end

    if (aarep_port_balance_all is null) then
      aarep_port_balance_all = 0;
    if (aarep_port_kolich_all is null) then
      aarep_port_kolich_all = 0;
    if (aarep_par_balance1 is null) then
      aarep_par_balance1 = 0;
    if (aarep_par_balance7 is null) then
      aarep_par_balance7 = 0;

    ----------------------------------------------------------------------- <<
    if (aarep_port_balance_all = 0) then
      aarep_memo = 'Не будет отправлено, нет КП.';

    aarep_date = :rep_date_end;
    aarep_plan_all = 0;
    aarep_fact_all = 0;
    aarep_fact_planned = 0;
    aarep_fact_other = 0;
    aarep_delinq_os = 0;
    aarep_delinq_perc = 0;
    aarep_fact_os = 0;
    aarep_fact_perc = 0;
    aarep_fact_fine = 0;
    aarep_fact_delinq_os = 0;
    aarep_fact_delinq_perc = 0;
    aarep_fact_adv_os = 0;
    aarep_fact_adv_perc = 0;
    aarep_port_balance_own = 0;
    aarep_port_kolich_own = 0;

    select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other),
           sum(aarep_month_plan_os), sum(aarep_month_plan_perc), sum(aarep_delinq_os), sum(aarep_delinq_perc),
           sum(aarep_fact_os), sum(aarep_fact_perc), sum(aarep_fact_fine), sum(aarep_fact_delinq_os),
           sum(aarep_fact_delinq_perc), sum(aarep_fact_adv_os), sum(aarep_fact_adv_perc), max(aarep_oklad),
           sum(aarep_port_balance_own), sum(aarep_port_kolich_own), sum(aarep_fact_kommis)
    from cb_az_lo_201602_one_tmp_olimp(:reg_id, :c_id, :rep_date_end)
    into :aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os,
         aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine,
         aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc, aarep_oklad,
         aarep_port_balance_own, aarep_port_kolich_own, aarep_fact_kommis;

    if (aarep_plan_all is null) then
      aarep_plan_all = 0; -- Сумма новых просрочек + выплаченные по графику за период
    if (aarep_fact_all is null) then
      aarep_fact_all = 0; -- Сумма новых просрочек + выплаченные по графику за период
    if (aarep_fact_planned is null) then
      aarep_fact_planned = 0; -- выплаченные по графику за период
    if (aarep_fact_other is null) then
      aarep_fact_other = 0; -- выплаченные по графику за период
    if (aarep_month_plan_os is null) then
      aarep_month_plan_os = 0;
    if (aarep_month_plan_perc is null) then
      aarep_month_plan_perc = 0;
    if (aarep_delinq_os is null) then
      aarep_delinq_os = 0;
    if (aarep_delinq_perc is null) then
      aarep_delinq_perc = 0;
    if (aarep_fact_os is null) then
      aarep_fact_os = 0;
    if (aarep_fact_perc is null) then
      aarep_fact_perc = 0;
    if (aarep_fact_fine is null) then
      aarep_fact_fine = 0;
    if (aarep_fact_delinq_os is null) then
      aarep_fact_delinq_os = 0;
    if (aarep_fact_delinq_perc is null) then
      aarep_fact_delinq_perc = 0;
    if (aarep_fact_adv_os is null) then
      aarep_fact_adv_os = 0;
    if (aarep_fact_adv_perc is null) then
      aarep_fact_adv_perc = 0;
    if (aarep_oklad is null) then
      aarep_oklad = 0;

    aarep_returnment_koef = 1.0;

    if (aarep_plan_all <> 0) then
      aarep_returnment_koef = round(aarep_fact_planned / aarep_plan_all, 4);
    else
      aarep_returnment_koef = 0.0;
    if (aarep_returnment_koef > 1.0) then
      aarep_returnment_koef = 1.0;

    aarep_bonus_plus = round((aarep_fact_all / 1000.0) * koef_bonus, 0);

    -- формула: aarep_plan_all=aarep_delinq_os+aarep_fact_os
    if (aarep_plan_all > (aarep_fact_planned + (aarep_fact_delinq_os / 2))) then
      aarep_bonus_minus = round(((aarep_plan_all - aarep_fact_planned - (aarep_fact_delinq_os / 2)) / 1000.0) * koef_penalty, 0);
    else
      aarep_bonus_minus = 0;

    aarep_bonus_total_fact = 0;
    aarep_bonus_total_to_pay = 0;
    aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus;
    aarep_bonus_total_fact = aarep_bonus_total - aarep_oklad;
    aarep_bonus_total_to_pay = aarep_bonus_total_fact;
    if (aarep_bonus_total_to_pay < 0) then
      aarep_bonus_total_to_pay = 0;
    aarep_bonus_ideal = round(((aarep_fact_perc + aarep_delinq_perc + aarep_month_plan_perc + aarep_fact_adv_perc + aarep_fact_fine) / 1000.0) * koef_bonus, 0);

    aarep_day_bonus_plus = aarep_bonus_plus;
    aarep_day_bonus_minus = aarep_bonus_minus;
    aarep_day_bonus_total = aarep_bonus_total;
    aarep_day_returnment_koef = aarep_returnment_koef;

    aarep1_plan_all = 0;
    aarep1_fact_all = 0;
    aarep1_fact_planned = 0;
    aarep1_fact_other = 0;

    if ((extract(day from :rep_date_end) > 1) and
        (need_one_day = 1)) then
    begin
      select sum(aarep_plan_all), sum(aarep_fact_all), sum(aarep_fact_planned), sum(aarep_fact_other)
      from cb_az_lo_201602_one_tmp_olimp(:reg_id, :c_id, :rep_date_end - 1)
      into :aarep1_plan_all, :aarep1_fact_all, :aarep1_fact_planned, :aarep1_fact_other;

      if (aarep1_plan_all is null) then
        aarep1_plan_all = 0;
      if (aarep1_fact_all is null) then
        aarep1_fact_all = 0;
      if (aarep1_fact_planned is null) then
        aarep1_fact_planned = 0;
      if (aarep1_fact_other is null) then
        aarep1_fact_other = 0;

      aarep_day_returnment_koef = 1.0;
      if (aarep_plan_all - aarep1_plan_all <> 0) then
        aarep_day_returnment_koef = round((aarep_fact_planned - aarep1_fact_planned) / (aarep_plan_all - aarep1_plan_all), 4);
      else
        aarep_day_returnment_koef = 0.0;
      if (aarep_day_returnment_koef > 1.0) then
        aarep_day_returnment_koef = 1.0;
      if (aarep_day_returnment_koef < 0) then
        aarep_day_returnment_koef = 1.0;

      aarep_day_bonus_plus = round((aarep1_fact_all / 1000.0) * koef_bonus, 0);
      if (aarep1_plan_all > aarep1_fact_planned) then
        aarep_day_bonus_minus = round(((aarep1_plan_all - aarep1_fact_planned) / 1000.0) * koef_penalty, 0);
      else
        aarep_day_bonus_minus = 0;
      aarep_day_bonus_total = aarep_day_bonus_plus - aarep_day_bonus_minus;

      aarep_day_bonus_plus = aarep_bonus_plus - aarep_day_bonus_plus;
      aarep_day_bonus_minus = aarep_bonus_minus - aarep_day_bonus_minus;
      aarep_day_bonus_total = aarep_bonus_total - aarep_day_bonus_total;
    end

    lo_stage_month = 0;
    select lo_stage_month_h
    from hp_get_lo_stage(:c_id, :rep_date_end)
    into lo_stage_month;

    if (lo_stage_month < 4.0) then
      aarep_bonus_total_to_pay = 0;

    if (rep_c_id <> 0) then
      suspend;
    else
    begin
      if ((aarep_port_balance_all <> 0) or (aarep_bonus_total <> 0)) then
        suspend;
    end
  end
end
