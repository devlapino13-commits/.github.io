CREATE PROCEDURE HP_GENERATE_C_UNIQ_ID(
    C_ID DECIMAL(18,0),
    C_BIRTH_DATE DATE,
    C_ACTION INTEGER,
    UN_ID DECIMAL(18,0))
AS
declare variable cunq_ext_id_new integer;
declare variable cunq_ext_id_curr integer;
declare variable cul_ext_id integer;
begin
-- c_action отрицательный - убрать лицевой счет,
-- c_action положительный - вычислить лицевой счет
  cunq_ext_id_new = 0;
  if (c_action = 0) then
    exit;

-- Получаем текущий лицевой счет
  select coalesce(cunq_ext_id, 0)
  from clientuniq
  where cunq_from_c_id = :c_id
  into :cunq_ext_id_curr;

-- Если у него имеется положительный лицевой счет, то не меняем
  if ((c_action > 0) and
      (cunq_ext_id_curr > 0)) then
    exit;

  if (c_action > 0) then
  begin
    cul_ext_id = null;

-- Сначала ищем среди старых лицевых счетов этого контрагента
    select cul_ext_id
    from clientulog
    where cul_from_c_id = :c_id and
          cul_birth_date = :c_birth_date and
          cul_ext_id > 0
    order by cul_id desc
    rows 1
    into :cul_ext_id;
    if (cul_ext_id is null) then
      cul_ext_id = 0;

-- если имеется старый лицевой счет, уточняем, не занято ли он
    if (cul_ext_id > 0) then
    begin
      if (exists(select *
                 from clientuniq
                 where cunq_birth_date = :c_birth_date and
                       cunq_ext_id = :cul_ext_id)) then
        cul_ext_id = 0;
    end

    if (cul_ext_id > 0) then
      cunq_ext_id_new = cul_ext_id; -- если не занят, присваиваем старое
    else
    begin
-- если занят, генерируем новый лицевой счет
      select cunq_ext_id
      from clientuniq
      where cunq_birth_date = :c_birth_date and
            cunq_ext_id > 0
      order by cunq_birth_date desc, cunq_ext_id desc
      rows 1
      into :cunq_ext_id_new;
      cunq_ext_id_new = coalesce(cunq_ext_id_new, 0) + 1;
    end
  end
  else
  begin
    if (c_birth_date is not null) then
    begin
      select cunq_ext_id
      from clientuniq
      where cunq_birth_date = :c_birth_date and
            cunq_ext_id < 0
      order by cunq_birth_date asc, cunq_ext_id asc
      rows 1
      into :cunq_ext_id_new;
      cunq_ext_id_new = coalesce(cunq_ext_id_new, 0) - 1;
    end
    else
    begin
      select cunq_ext_id
      from clientuniq
      where cunq_birth_date is null and
            cunq_ext_id < 0
      order by cunq_birth_date asc, cunq_ext_id asc
      rows 1
      into :cunq_ext_id_new;
      cunq_ext_id_new = coalesce(cunq_ext_id_new, 0) - 1;
    end
  end

  update or insert into clientuniq (cunq_from_c_id, cunq_birth_date, cunq_ext_id, cunq_from_un_id)
  values (:c_id, :c_birth_date, :cunq_ext_id_new, :un_id)
  matching (cunq_from_c_id);
end
