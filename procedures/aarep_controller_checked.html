CREATE PROCEDURE AAREP_CONTROLLER_CHECKED(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(50),
    KF_NAME VARCHAR(100),
    REG_NAME VARCHAR(100),
    CLIENT_FIO VARCHAR(100),
    CONTROLLER_FIO VARCHAR(100),
    CHECK_RESULT VARCHAR(20),
    CHECKED_DATE DATE,
    KONT_STATE VARCHAR(20))
AS
begin
  for select kont.k_kontract_no, kal.kal_new_kf_name, reg.reg_name,

             cl.c_name_fio as client_fio, cl_contr.c_name_fio as controller_fio, rd_res.rd_name,
             cont.contr_checked_date,
             (
             case ks.ks_state
               when 0 then 'В процессе'
               when 1 then 'Выдан'
               when 2 then 'Закрыт'
               when -1 then 'Аннулир.'
             end) as kont_state
      from controller cont
      inner join kontract kont on kont.k_id = cont.contr_from_k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
      inner join kontstate ks on ks.ks_from_k_id = kont.k_id
      inner join clients cl on cl.c_id = cont.contr_from_c_id
      inner join region_list(:rep_region_id, 1, -1, 1) rl on rl.id_list = cast(kal_new_reg_id as id_type)
      inner join region reg on reg.reg_unicode = rl.id_list
      inner join username un on un.un_id = cont.contr_from_un_id
      inner join clients cl_contr on cl_contr.c_id = un.un_from_c_id
      inner join refdata rd_res on rd_res.rd_id = cont.contr_check_result
      where cont.contr_checked_date between :rep_date_begin and :rep_date_end and
            cont.contr_is_delete = 0 and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0 and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end
      order by reg_name, kal_new_kf_name, k_kontract_no
      into :k_kontract_no, :kf_name, :reg_name, :client_fio, :controller_fio, :check_result, :checked_date,
           :kont_state
  do
  begin
    suspend;
  end
end
