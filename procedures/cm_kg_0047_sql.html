CREATE PROCEDURE CM_KG_0047_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable KC_AMOUNT CURR_TYPE;
declare variable C_ID ID_TYPE;
declare variable C_NAME_FIO STR_100;
declare variable C_COMMENT STR_255;
declare variable CL_FIO STR_100;
declare variable MA_ID integer;
declare variable ADD_INFO STR_100;
declare variable NDPL_ID integer;
declare variable IS_NDPL_ID D_BOOL;
declare variable PR_CODE D_STR_100;
begin
  if (k_id = 109010230638400401) then
    exit;

  ma_id = (select k_from_ma_app_id
           from kontract
           where k_id = :k_id);
  -- проверка под ФТ на уровне контракта

  select pr.pr_code
  from kontract kt
  inner join product pr on pr.pr_id = kt.k_from_pr_id
  where kt.k_id = :k_id
  into pr_code;

  select rd_id
  from refdata
  where rd_code = 'FionpolisDirectory.NDPLLst'
  into :ndpl_id;

  for select kc_from_c_id, 'клиент'
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      union
      select rel_other_c_id, 'родственник клиента'
      from knclient
      inner join hp_client_relation_tmp(kc_from_c_id, 4) on 1 = 1
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0
      union
      select distinct ag_object_id, 'продавец'
      from kontmat, agreements
      where kml_object_id = :k_id and
            kml_object_kind = 9 and
            kml_from_ag_id = ag_id and
            ag_object_kind = 0
      union
      select z_from_c_id, 'поручитель'
      from zalog zal
      where z_from_k_id = :k_id and
            zal.z_from_c_id is not null
      union
      select rel_other_c_id, 'родственник поручителя'
      from zalog zal
      inner join hp_client_relation_tmp(z_from_c_id, 4) on 1 = 1
      where z_from_k_id = :k_id and
            zal.z_from_c_id is not null
      into :c_id, :add_info
  do
  begin
    -- комплаенс даёт "добро"
    if (exists(select *
               from fp_verify
               where fpv_ma_id = :ma_id and
                     fpv_c_id = :c_id and
                     fpv_is_confirm = 1)) then
      continue;
    if (c_id <> 1090101074749) then
    begin
      for select c_name_fio, c_comment, iif(:add_info = 'клиент' and :ndpl_id = terr_categ_id, 1, 0)
          from hp_check_client_fio_ft_fuzzy(:c_id, '', '', '', null, '', 1, iif(:add_info = 'клиент', 1, 0))
          where c_comment_id = 1
          into :c_name_fio, :c_comment, :is_ndpl_id
      do
      begin
        select cl.c_name_fio
        from clients cl
        where cl.c_id = :c_id
        into :res_info_text;

        if (c_comment = 'РФМ (СФР): ФИО') then
          res_error_is_critical = 1;
        else
          res_error_is_critical = 2;

        if (:is_ndpl_id = 1) then
          res_error_text = 'Совпадение с лицом со списка НПДЛ (национальное публичное должностное лицо) ' || c_name_fio || ', в случае если лицо является НПДЛ, сформировать и подписать Анкету НПДЛ.';
        else
          res_error_text = 'Совпадение с лицом по ' || c_comment || ' ' || c_name_fio || ' | ' || add_info;

        if (k_id in (109010061182700201, 109010039739800501, 109010094626200101)) then
          res_error_is_critical = 0;
        suspend;
      end
    end
  end
end
