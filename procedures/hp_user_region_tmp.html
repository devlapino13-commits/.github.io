CREATE PROCEDURE HP_USER_REGION_TMP(
    UN_ID D_ID)
RETURNS (
    C_ID D_ID)
AS
begin
  select un.un_from_c_id
  from username un
  where un.un_id = :un_id
  into :c_id;
  suspend;
  for select hrv.hrv_from_c_id
      from hr_voctravel hrv
      inner join hr_orders ho on hrv.hrv_from_ho_id = ho.ho_id and
            ho.ho_order_type <> 20
      where hrv.hrv_changer_id = :c_id and
            current_date between hrv.hrv_date_begin and hrv.hrv_date_end and
            hrv_status = 1
      into :c_id
  do
    suspend;
end
