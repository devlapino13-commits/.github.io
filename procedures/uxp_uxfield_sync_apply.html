CREATE PROCEDURE UXP$UXFIELD_SYNC_APPLY(
    UF_UUID D_UUID,
    UF_SOURCE D_STR_CODE = null,
    UF_FIELD D_STR_CODE = null)
AS
declare variable uf_state d_smallint;
declare variable us_uuid d_uuid;
begin
  -- Применяет синхронизацию
  -- если -1, то удаляет
  -- если 1, то оставляет как поле

  if (coalesce(uf_uuid, '') = '') then
    uf_uuid = uxf$uuid(uxf$uniq(uf_source), 'uxfield', uf_field);

  uf_state = 0;

  select uf_field_name, uf_state, uf_owner
  from ux$field
  where uf_uuid = :uf_uuid
  into uf_field, uf_state, us_uuid;

  if (uf_state = 0) then
    exit;

  if (uf_state = -1) then
  begin
    update ux$field
    set uf_type = -1,
        uf_state = 0
    where uf_uuid = :uf_uuid;
  end

  if (uf_state = 1) then
  begin
    update ux$field
    set uf_state = 0
    where uf_uuid = :uf_uuid;
  end

  execute procedure uxp$uxsource_diff_sync(:us_uuid);
end
