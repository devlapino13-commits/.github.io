CREATE PROCEDURE AAREP_TRANSFER_BEELINE_DOCS(
    C_ID D_ID,
    REP_DATE D_DATE,
    HBI D_INTEGER,
    IS_FIRST D_BOOL)
RETURNS (
    CLIENT_ID D_ID,
    ID D_INTEGER,
    NUMBER D_STR_CODE,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    ISSUED_BY D_STR_255,
    AUTHORITY_CODE D_STR_255,
    DATA_ALL D_STR_8100)
AS
declare variable DOC_TYPE_NAME D_STR_100;
begin
  -- Документы удостоверяющие личность    01( 0000 0001)
  -- Вид на жительство                    02( 0000 0010)
  -- Разрешение на временное проживание   04( 0000 0100)
  -- Патент                               08( 0000 1000)
  -- Миграционная карта                   16( 0001 0000)
  -- ДДЛ + РВП                            05( 0000 0101)
  -- ДДЛ + РВП + ВНЖ                      07( 0000 0111)
  -- Патент + РВП + ВНЖ                   14( 0000 1110)
  data_all = '';
  for
    select
          q.cds_from_c_id
        , q.id
        , q.name
        , q.cds_doc_no
        , q.cds_doc_date_begin
        , q.cds_doc_date_end
        , q.cds_doc_issued_by
        , q.authority_code
      from (
        with document_types as (
                    select 01 as id, 'Паспорт РФ'                      as name, ',passport_local,'                  as code, 01 as hbi from rdb$database
--          union all select 02 as id, 'Военный билет'                   as name, ',military_id,'                     as code, 01 as hbi from rdb$database
--          union all select 03 as id, 'Удостоверение личности'          as name, ''                                  as code, 01 as hbi from rdb$database
          union all select 04 as id, 'Паспорт иностранного гражданина' as name, ',ID_karta_uzb,passport_card,'      as code, 01 as hbi from rdb$database
--          union all select 05 as id, 'Паспорт СССР'                    as name, ''                                  as code, 01 as hbi from rdb$database
          union all select 04 as id, 'Загранпаспорт'                   as name, ',passport_trans,passport_trans_2,' as code, 01 as hbi from rdb$database
          union all select 07 as id, 'ВНЖ'                             as name, ',registration_vnj,'                as code, 02 as hbi from rdb$database
          union all select 08 as id, 'РВП'                             as name, ',registration_rvp,'                as code, 04 as hbi from rdb$database
          union all select 09 as id, 'Патент'                          as name, ',work_patent,'                     as code, 08 as hbi from rdb$database
          union all select 10 as id, 'Миграционная карта'              as name, ',magration_card,'                  as code, 16 as hbi from rdb$database
        )
        select
              cd.cds_from_c_id
            , dt.id
            , dt.name
            , cd.cds_doc_no
            , cd.cds_doc_date_begin
            , cd.cds_doc_date_end
            , cd.cds_doc_issued_by
            , iif(cd.cds_from_drl_id is null, '', f_docrecog_value(cd.cds_from_drl_id, 'authority_code', '')) as authority_code
          from
            clientdocs                          as cd
            inner join passkind                 as pk
              on (cd.cds_from_pk_id = pk.pk_id)
            inner join passdoctype              as pdt
              on (pk.pk_from_pdt_id = pdt.pdt_id)
            inner join document_types           as dt
              on
                    (dt.code > '')
                and ((bin_and(dt.hbi, :hbi) = dt.hbi)
                and (dt.code containing ',' || pdt.pdt_code || ',' ))
          where
                (cds_from_c_id = :c_id)
            and (cds_is_delete = 0)
            and (:rep_date between cds_doc_date_begin and coalesce(cds_doc_date_end, :rep_date))
          order by
              cd.cds_from_pk_id     asc
            , cd.cds_doc_date_begin desc) as q
      order by
        q.id asc
      into
        :client_id
        , :id
        , :doc_type_name
        , :number
        , :date_begin
        , :date_end
        , :issued_by
        , :authority_code
  do
  begin
    doc_type_name = replace(:number, ' ', '') || ' ' || :date_begin || coalesce(' ' || :date_end, '');
    if (:data_all > '') then
      :data_all = :data_all || ',';
    :data_all = :data_all || ' ' || :doc_type_name;
    if (:is_first = 1) then
      break;
  end

  if (:data_all > '') then
  begin
    if (:is_first = 0) then
    begin
      id = null;
      number = null;
      date_begin = null;
      date_end = null;
      issued_by = null;
      authority_code = null;
    end
    :data_all = trim(:data_all);
    suspend;
  end
end
