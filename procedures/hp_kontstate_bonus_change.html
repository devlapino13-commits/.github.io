CREATE PROCEDURE HP_KONTSTATE_BONUS_CHANGE(
    KS_FROM_K_ID DECIMAL(18,0),
    REP_DATE DATE,
    KONT_IS_ACTIVE INTEGER)
RETURNS (
    MAX_DELINQ_DAY INTEGER,
    BONUS_CHANGE INTEGER)
AS
declare variable ok_sh_date date;
declare variable ok_delinq_day integer;
declare variable balance decimal(18,4);
declare variable db_id integer;
declare variable k_fact_closed_date_last date;
declare variable k_disb_date date;
declare variable k_fact_disb_date_last date;
declare variable k_from_kf_id id_type;
declare variable ks_bonus_change integer;
declare variable max_delinq_new integer;
declare variable max_delinq_old integer;
declare variable k_id_last id_type;
declare variable k_kredline_end date;
declare variable k_kolich_repay integer;
begin

  db_id = f_get_db_id();
  bonus_change = 0;
  max_delinq_day = 0;

  if (db_id in (39801, 41701)) then
  begin
    for select ok_sh_date, max(ok_ol_date - ok_sh_date)
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :ks_from_k_id and
              ok_type = 4 and
              (ok_os + ok_perc) < -100 -- не включать до 100 руб. прсрочки
              and
              ok_sh_date < ok_ol_date and
              ok_ol_date <= :rep_date and
              ok_from_c_id = 0
        group by ok_sh_date
        into :ok_sh_date, :ok_delinq_day
    do
    begin
      if (max_delinq_day < ok_delinq_day) then
        max_delinq_day = :ok_delinq_day;
    end

    if (kont_is_active = 1) then
    begin
      for select ok_sh_date, sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc)
          from operkred
          where ok_from_k_id = :ks_from_k_id and
                ok_ol_is_active = 1 and
                ok_sh_date <= :rep_date and
                ok_ol_date <= :rep_date and
                ok_type <= 6 and
                ok_sh_id > 0 and
                ok_from_c_id = 0
          group by ok_sh_date
          having(sum(ok_os + ok_sh_os) > 100 -- Учитывается просрочка более 100 руб.
          and (:rep_date - ok_sh_date) > :max_delinq_day)
          into :ok_sh_date, :balance
      do
      begin
        if (max_delinq_day < (:rep_date - ok_sh_date)) then
          max_delinq_day = :rep_date - ok_sh_date;
      end
    end
    bonus_change = 0;

    if (rep_date >= cast('01.08.2015' as date)) then -- Добавлено по приказу от 01.08.2015 Улан
    begin
      if (max_delinq_day = 0) then
        bonus_change = 1;
      else
      if (max_delinq_day <= 15) then
        bonus_change = 0;
      else
      if (max_delinq_day <= 30) then
        bonus_change = -1;
      else
        bonus_change = -999;
    end
    else
    if (rep_date >= cast('24.10.2012' as date)) then
    begin
      if (max_delinq_day = 0) then
        bonus_change = 1;
      else
      if (max_delinq_day <= 30) then
        bonus_change = 0;
      else
        bonus_change = -999;
    end
    else
    begin
      if (max_delinq_day = 0) then
        bonus_change = 1;
      else
        bonus_change = -999;
    end
--    Добавлено от 31.01.2014
    if (bonus_change = 1 and
        rep_date >= cast('30.12.2013' as date) and
        rep_date < cast('01.08.2015' as date)) then
    begin
--      Клиенты (группа) сохраняет свой статус в течение трех месяцев со дня последнего погашения займа
      select k_from_kf_id, k_disb_date
      from kontract
      where k_id = :ks_from_k_id
      into k_from_kf_id, k_disb_date;

      if (kont_is_active = 1) then
      begin
        k_fact_closed_date_last = null;
        k_fact_disb_date_last = null;
        select k_fact_disb_date, k_fact_closed_date
        from kontract
        where k_from_kf_id = :k_from_kf_id and
              k_disb_date < :k_disb_date and
              k_fact_disb_date is not null and
              k_fact_closed_date is not null
        order by k_fact_closed_date desc
        rows 1
        into k_fact_disb_date_last, k_fact_closed_date_last;

        if (k_fact_closed_date_last is not null) then
        begin
--        если после закрытия в течении 12 месяцев не кредитовался то их статус не меняется
--        По паспорту пунтк 23
          if (k_fact_closed_date_last < dateadd(month, -12, k_disb_date)) then
            bonus_change = 0;

--        Группы, погасившим заем досрочно в течение первых 3х месяцев не присваивается новый статус
          else
          if (k_fact_disb_date_last > dateadd(month, -3, k_fact_closed_date_last)) then
            bonus_change = 0;
        end
      end
      else
      begin
--      Если контракт закрывается то по текущему контракту будем проверять
--      не закрылся ли досрочно, если закрылся в течении 3 месяцев после выдачи то новый статус не присваиваем
        if (k_disb_date > dateadd(month, -3, rep_date)) then
          bonus_change = 0;
      end
    end
    else
    if (rep_date >= cast('01.08.2015' as date)) then
    begin
--      Клиенты (группа) сохраняет свой статус в течение трех месяцев со дня последнего погашения займа
      select k_from_kf_id, k_disb_date
      from kontract
      where k_id = :ks_from_k_id
      into k_from_kf_id, k_disb_date;

      if (kont_is_active = 1) then
      begin
--          вычисляется предыдущий цикл данного контракта
        k_fact_closed_date_last = null;
        k_id_last = null;
        select ks_date_end, k_id
        from kontstate, kontract
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_from_k_id = k_id and
              k_from_kf_id = :k_from_kf_id and
              ks_from_k_id <> :ks_from_k_id and
              ks_date_begin < :k_disb_date
        order by ks_date_end desc
        rows 1
        into k_fact_closed_date_last, k_id_last;

        if (k_id_last is not null) then
        begin
          if (k_fact_closed_date_last < dateadd(month, -36, k_disb_date)) then
          begin
--            если не кредитовался более 36 месяцев то теряет все статусы
            bonus_change = -999;
          end
          else
          if (k_fact_closed_date_last >= dateadd(month, -36, k_disb_date) and
              k_fact_closed_date_last <= dateadd(month, -3, k_disb_date)) then
          begin
--            если со дня закрытия кредита прошли более 3х месяев и не более 36 то действует следующие условии
--            вычисляет максимальное кол-во просроченных дней
            max_delinq_day = 0;
            for select ok_sh_date, max(ok_ol_date - ok_sh_date)
                from operkred
                where ok_ol_is_active = 1 and
                      ok_from_k_id = :k_id_last and
                      ok_type = 4 and
                      (ok_os + ok_perc) < -100 -- не включать до 100 руб. прсрочки
                      and
                      ok_sh_date < ok_ol_date and
                      ok_ol_date <= :rep_date and
                      ok_from_c_id = 0
                group by ok_sh_date
                into :ok_sh_date, :ok_delinq_day
            do
            begin
              if (max_delinq_day < ok_delinq_day) then
                max_delinq_day = :ok_delinq_day;
            end

            if (max_delinq_day < 1) then
              bonus_change = 0;
            else
            if (max_delinq_day <= 15) then
              bonus_change = -1;
            else
              bonus_change = -999;
          end
        end
      end
      else
      begin
--      Если контракт закрывается то по текущему контракту будем проверять
--      не закрылся ли досрочно, если закрылся в течении 3 месяцев после выдачи то новый статус не присваиваем
        if (k_disb_date > dateadd(month, -3, rep_date)) then
          bonus_change = 0;
      end
    end

  end
  else
  if (db_id = 81001) then
  begin
    max_delinq_new = 0;
    max_delinq_old = 0;
    for select ok_sh_date, max(ok_ol_date - ok_sh_date)
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :ks_from_k_id and
              ok_type = 4 and
              (ok_os + ok_perc) < -100 -- не включать до 100 руб. прсрочки
              and
              ok_sh_date < ok_ol_date and
              ok_ol_date <= :rep_date and
              ok_from_c_id = 0
        group by ok_sh_date
        into :ok_sh_date, :ok_delinq_day
    do
    begin
      if (max_delinq_day < ok_delinq_day) then
        max_delinq_day = :ok_delinq_day;
      if (ok_sh_date >= '01.11.2014' and
          max_delinq_new < ok_delinq_day) then
        max_delinq_new = :ok_delinq_day;
      if (ok_sh_date < '01.11.2014' and
          max_delinq_old < ok_delinq_day) then
        max_delinq_old = :ok_delinq_day;
    end
--    Расчте максимальное кол-во просроченных дней до 01.11.2014г.

    if (kont_is_active = 1) then
    begin
      for select ok_sh_date, sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc)
          from operkred
          where ok_from_k_id = :ks_from_k_id and
                ok_ol_is_active = 1 and
                ok_sh_date <= :rep_date and
                ok_ol_date <= :rep_date and
                ok_type <= 6 and
                ok_sh_id > 0 and
                ok_from_c_id = 0
          group by ok_sh_date
          having(sum(ok_os + ok_sh_os) > 100 -- Учитывается просрочка более 100 руб.
--                and (:rep_date - ok_sh_date) > :MAX_DELINQ_DAY
          )
          into :ok_sh_date, :balance
      do
      begin
        if (max_delinq_day < (:rep_date - ok_sh_date)) then
          max_delinq_day = :rep_date - ok_sh_date;
        if (ok_sh_date >= '01.11.2014' and
            max_delinq_new < (:rep_date - ok_sh_date)) then
          max_delinq_new = (:rep_date - ok_sh_date);
        if (ok_sh_date < '01.11.2014' and
            max_delinq_old < (:rep_date - ok_sh_date)) then
          max_delinq_old = (:rep_date - ok_sh_date);
      end
    end

    bonus_change = 0;

    if (rep_date < cast('01.11.2014' as date)) then
    begin
      if (max_delinq_day <= 7) then
        bonus_change = 1;
      else
      if (max_delinq_day < 30) then
        bonus_change = 0;
      else
        bonus_change = -999;
    end
    else
    if (rep_date < cast('05.10.2015' as date)) then
    begin -- программа лояльности от 23.10.2014
      if (max_delinq_new = 0 and
          max_delinq_old <= 7) then
      begin
        bonus_change = 1;
      end
      else
      begin
        if (max_delinq_old > 7) then
        begin
          if (max_delinq_old < 30) then
            bonus_change = 0;
          else
            bonus_change = -999;
        end
        else
        if (max_delinq_new >= 1) then
        begin
------------------------------------------------------------------------------------------------------------------------------------------
-- нужно вычислить текущий статус и если более бронзового статуса то спустить до бронзового а если бронзовый то сделать без статуса
-- нужно
          select ks_bonus_change
          from kontstate
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_from_k_id = :ks_from_k_id rows 1
          into ks_bonus_change;

          if (ks_bonus_change > 1) then
            bonus_change = -998;
          else
            bonus_change = -999;
        end
      end
    end
    else
    begin
--      добавлено по приказу №23-09-15 от 30.09.2015г.
      if (max_delinq_day <= 3) then
        bonus_change = 1;
      else
      if (max_delinq_day <= 7) then
        bonus_change = 0;
      else
      if (max_delinq_day <= 20) then
        bonus_change = -1; -- данную строку нужно уточнить
      else
        bonus_change = -999;
    end

    if (bonus_change = 1) then
    begin
--      Клиенты (группа) сохраняет свой статус в течение трех месяцев со дня последнего погашения займа
      if (rep_date < cast('05.10.2015' as date)) then
      begin
        select k_from_kf_id, k_disb_date
        from kontract
        where k_id = :ks_from_k_id
        into k_from_kf_id, k_disb_date;

        if (kont_is_active = 1) then
        begin
          k_fact_closed_date_last = null;
          k_fact_disb_date_last = null;

          select k_fact_disb_date, k_fact_closed_date
          from kontract
          where k_from_kf_id = :k_from_kf_id and
                k_disb_date < :k_disb_date and
                k_fact_disb_date is not null and
                k_fact_closed_date is not null
          order by k_fact_closed_date desc
          rows 1
          into k_fact_disb_date_last, k_fact_closed_date_last;

          if (k_fact_closed_date_last is not null) then
          begin
--        если после закрытия в течении 3х месяцев не кредитовался то их статус не меняется
            if (k_fact_closed_date_last < dateadd(month, -3, k_disb_date)) then
              bonus_change = 0;

--        Группы, погасившим заем досрочно в течение первых 3х месяцев не присваивается новый статус
            else
            if (k_fact_disb_date_last > dateadd(month, -3, k_fact_closed_date_last)) then
              bonus_change = 0;
          end
        end
        else
        begin
--      Если контракт закрывается то по текущему контракту будем проверять
          if (k_disb_date > dateadd(month, -3, rep_date)) then
            bonus_change = 0;
        end
      end
      else
      begin
--        добавлено по приказу №23-09-15 от 30.09.2015г.
        if (kont_is_active <> 1) then
        begin
--        для повышения статуса клиент должен воспользоваться микрозаймом полный срок, А для займов, выданных на 9 и более месяцев
--        - не менее 2/3 от общего срока
--        если микрозаем был полностью выплачен за 10 дней и менее до кончания срока микрозайма по графику,
--        в данном случае признается , что клиент воспользовался микрозаймом полный срок.

--          вытаскиваем кол-во погашений, дату выдачи и план дату закрытия
          select k_disb_date, k_kredline_end, k_kolich_repay
          from kontract
          where k_id = :ks_from_k_id
          into k_disb_date, k_kredline_end, k_kolich_repay;

          if (k_kredline_end - rep_date > 10) then -- rep_date в этом случае дата закрытия если KONT_IS_ACTIVE <> 1
          begin
            if (k_kolich_repay < 9) then
              bonus_change = 0;
            else
            begin
              if (cast((rep_date - k_disb_date) as double precision) < cast((k_kredline_end - k_disb_date) as double precision) * 2 / 3) then
                bonus_change = 0;
            end
          end
        end
      end
    end
  end
  if (db_id = 94401) then
  begin
    for select ok_sh_date, max(ok_ol_date - ok_sh_date)
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :ks_from_k_id and
              ok_type = 4 and
              (ok_os + ok_perc) < -100 -- не включать до 100 руб. прсрочки
              and
              ok_sh_date < ok_ol_date and
              ok_ol_date <= :rep_date and
              ok_from_c_id = 0
        group by ok_sh_date
        into :ok_sh_date, :ok_delinq_day
    do
    begin
      if (max_delinq_day < ok_delinq_day) then
        max_delinq_day = :ok_delinq_day;
    end

    if (kont_is_active = 1) then
    begin
      for select ok_sh_date, sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc)
          from operkred
          where ok_from_k_id = :ks_from_k_id and
                ok_ol_is_active = 1 and
                ok_sh_date <= :rep_date and
                ok_ol_date <= :rep_date and
                ok_type <= 6 and
                ok_sh_id > 0 and
                ok_from_c_id = 0
          group by ok_sh_date
          having(sum(ok_os + ok_sh_os) > 100 -- Учитывается просрочка более 100 руб.
          and (:rep_date - ok_sh_date) > :max_delinq_day)
          into :ok_sh_date, :balance
      do
      begin
        if (max_delinq_day < (:rep_date - ok_sh_date)) then
          max_delinq_day = :rep_date - ok_sh_date;
      end
    end
    bonus_change = 0;
    if (rep_date >= cast('24.10.2012' as date)) then
    begin
      if (max_delinq_day = 0) then
        bonus_change = 1;
      else
      if (max_delinq_day <= 30) then
        bonus_change = 0;
      else
        bonus_change = -999;
    end
    else
    begin
      if (max_delinq_day = 0) then
        bonus_change = 1;
      else
        bonus_change = -999;
    end
--    Добавлено от 31.01.2014
    if (bonus_change = 1 and
        rep_date >= cast('30.12.2013' as date) and
        db_id in (39801, 41701)) then
    begin
--      Клиенты (группа) сохраняет свой статус в течение трех месяцев со дня последнего погашения займа
      select k_from_kf_id, k_disb_date
      from kontract
      where k_id = :ks_from_k_id
      into k_from_kf_id, k_disb_date;

      if (kont_is_active = 1) then
      begin
        k_fact_closed_date_last = null;
        k_fact_disb_date_last = null;
        select k_fact_disb_date, k_fact_closed_date
        from kontract
        where k_from_kf_id = :k_from_kf_id and
              k_disb_date < :k_disb_date and
              k_fact_disb_date is not null and
              k_fact_closed_date is not null
        order by k_fact_closed_date desc
        rows 1
        into k_fact_disb_date_last, k_fact_closed_date_last;

        if (k_fact_closed_date_last is not null) then
        begin
--        если после закрытия в течении 12 месяцев не кредитовался то их статус не меняется
--        По паспорту пунтк 23
          if (k_fact_closed_date_last < dateadd(month, -12, k_disb_date)) then
            bonus_change = 0;

--        Группы, погасившим заем досрочно в течение первых 3х месяцев не присваивается новый статус
          else
          if (k_fact_disb_date_last > dateadd(month, -3, k_fact_closed_date_last)) then
            bonus_change = 0;
        end
      end
      else
      begin
--      Если контракт закрывается то по текущему контракту будем проверять
        if (k_disb_date > dateadd(month, -3, rep_date)) then
          bonus_change = 0;
      end

    end
  end

  suspend;
end
