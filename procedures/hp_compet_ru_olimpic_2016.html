CREATE PROCEDURE HP_COMPET_RU_OLIMPIC_2016(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer;
declare variable rd_id integer;
declare variable res_period_id integer;
declare variable res_team_id integer;
declare variable res_lo_id decimal(18,0);
declare variable ad_active_client integer;
declare variable ad_active_balance decimal(18,4) = 0;
declare variable ad_diff_client integer;
declare variable ad_delinq_balance_7 decimal(18,4) = 0;
declare variable ad_delinq_balance_diff_30 decimal(18,4) = 0;
declare variable points_1 decimal(18,4) = 0;
declare variable points_2 decimal(18,4) = 0;
declare variable points_all decimal(18,4) = 0;
declare variable res_team_id_prior integer;
declare variable cpr_id integer;
declare variable cpr_group_name varchar(255);
declare variable cpd_comment d_str_4096;
declare variable lo_panelty_count integer;
declare variable avg_points decimal(18,4) = 0;
declare variable total_points decimal(18,4) = 0;
declare variable add_points decimal(18,4) = 0;
declare variable max_points decimal(18,4) = 0;
declare variable group_name varchar(255);
declare variable cpp_orderno integer;
declare variable kind integer;
declare variable cpr_final_points decimal(18,4) = 0;
declare variable ad_disb_loan decimal(18,4) = 0;
declare variable lo_count decimal(18,4) = 0;
declare variable total_disb decimal(18,4) = 0;
declare variable avg_disb decimal(18,4) = 0;
declare variable cpr_group_name_prior varchar(100);
declare variable place_no integer;
declare variable points_3 decimal(18,4) = 0;
declare variable points_4 decimal(18,4) = 0;
declare variable is_otpusk varchar(100);
declare variable res_closed integer;
declare variable res_repeat integer;
declare variable res_koef decimal(18,7);
declare variable avg_disb_prior decimal(18,4) = 0;
declare variable kol integer;
begin
-- Расчет итогов олимпиады РФ 2016
-- Stage:
-- 0 - период (неделя)
-- 1 - Промежуточные итоги
-- 2 - Итоги
-- 3 - Финальные результаты

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

-- rd_id для теста: 1381

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  if (stage = 0) then -- Если уровень недели
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

-- Формируем в памяти данные по РФ
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

-- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1, res_field_2, sum(ad_active_client), sum(ad_active_balance), sum(ad_diff_client),
               sum(ad_delinq_balance_end2), sum(ad_delinq_balance_diff3), sum(ad_disb_loan)
        from hp_analitica_report('kont_lo_teamname_id, lo_id', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        group by 1, 2
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :res_team_id, :res_lo_id, :ad_active_client, -- Клиенты на конец периода
             :ad_active_balance, -- Баланс на конец периода
             :ad_diff_client, -- Прирост клиентов
             :ad_delinq_balance_7, -- Баланс просрочек 7 дней
             :ad_delinq_balance_diff_30, -- Изменение баланс 30 дней
             :ad_disb_loan -- Выданные займы
    do
    begin
      if (ad_disb_loan is null) then
        ad_disb_loan = 0;

-- Исключаем если нет действующего
      if (not(exists(select *
                     from hr_pstatus
                     where ps_from_c_id = :res_lo_id and
                           ps_date_end >= :res_date_end and
                           ps_date_begin <= :res_date_end and
                           ps_status = 1 and
                           ps_is_working = 1))) then
        res_team_id = 0;

      if (res_team_id <> 0) then
      begin
-- Исключаем если не СВЗ/КС
        if (exists(select *
                   from hr_pstatus, joblist, jobkind
                   where ps_from_c_id = :res_lo_id and
                         ps_date_end >= :res_date_end and
                         ps_date_begin <= :res_date_end and
                         ps_status = 1 and
                         ps_from_jl_id = jl_id and
                         jl_kind = jk_id and
                         not(jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Partner')) and
                         ps_is_working = 1)) then
          res_team_id = 0;
      end

      is_otpusk = '';
      if (res_team_id <> 0 and
          res_date_end >= '26.09.2016') then
      begin
-- Исключаем если в отпуске
        if (exists(select *
                   from hr_voctravel, hr_orders, sa_worktype
                   where hrv_from_ho_id = ho_id and
                         ho_order_type = sawt_id and
                         hrv_from_c_id = :res_lo_id and
                         sawt_base_type in (4, 5) -- отпуск, декретный отпуск
                         and
                         sawt_id <> 20 and
                         sawt_hours_real >= 0 and
                         hrv_date_begin <= :res_date_begin and
                         hrv_date_end_fact >= (:res_date_begin + 4) and
                         hrv_status = 1)) then
        begin
          if (not(exists(select *
                         from hr_voctravel, hr_orders
                         where hrv_from_ho_id = ho_id and
                               ho_order_type = 20 and
                               hrv_from_c_id = :res_lo_id and
                               hrv_date_begin <= :res_date_end and
                               hrv_date_end_fact >= :res_date_begin and
                               hrv_status = 1))) then
          begin
--            res_team_id=0;
            is_otpusk = 'Отпуск.';
          end
        end
      end

      points_1 = 0.0;
      points_2 = 0.0;
      points_3 = 0.0;

      cpd_comment = is_otpusk || '<br>';
      cpd_comment = cpd_comment || 'Актив.кл: <b>' || round(ad_active_client) || '</b><br>';
      cpd_comment = cpd_comment || 'Прирост кл: <b>' || round(ad_diff_client) || '</b><br>';
      cpd_comment = cpd_comment || 'PAR7: <b>' || round(ad_delinq_balance_7 * 100.00 / ad_active_balance, 2) || '%</b><br>';
      cpd_comment = cpd_comment || 'Изменение PAR30: <b>' || round(ad_delinq_balance_diff_30, 2) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Выдачи за период СВЗ: <b>' || round(ad_disb_loan, 0) || '</b><br>';

-- Вычисляем баллы по приросту
-- п.1.1 - Прирост клиентов >=2 то 10 баллов
      if (ad_diff_client >= 2) then
        points_1 = 10.0;

-- п.1.2 - иначе, количество кл >=500 то 5 баллов
      if (points_1 = 0.0 and
          ad_active_client >= 500) then
        points_1 = 5.0;

-- п.1.4 - иначе, прирост кл отрицательный и клиентов меньше 500, то -5 баллов
      if (points_1 = 0.0 and
          ad_diff_client < 0 and
          ad_active_client < 500) then
        points_1 = -5.0;

-- Вычисляем баллы по снижению баланса просрочек 30 дн.
-- п.2.1 - Если снижен более 5 тыс., то 20 баллов
      if (ad_delinq_balance_diff_30 <= -5000.0) then
        points_2 = 20.0;

-- п.2.2 - Если снижен более 2 тыс., то 10 баллов
      if (points_2 = 0.0 and
          ad_delinq_balance_diff_30 <= -2000.0) then
        points_2 = 10.0;

-- п.2.3 - Если par7<=7%, то 5 баллов
      if (points_2 = 0.0 and
          (round(ad_delinq_balance_7 * 100.00 / ad_active_balance, 2)) < 7) then
        points_2 = 5.0;

-- п.2.4 - Если ниодин пункт не выполнен, то -5 баллов
      if (points_2 = 0.0 and
          ad_delinq_balance_diff_30 > 0 and
          ad_delinq_balance_diff_30 < 5000) then
        points_2 = -5.0;

-- п.2.5 - Если ниодин пункт не выполнен, то -5 баллов
      if (points_2 = 0.0 and
          ad_delinq_balance_diff_30 >= 5000.0) then
        points_2 = -10.0;

      if (res_date_end >= '09.10.2016') then -- Расчет коэффициента удержания клиентов
      begin
        if (datediff(day, cast('25.09.2016' as date), res_date_end) = (datediff(day, cast('25.09.2016' as date), res_date_end) / 14) * 14) then
        begin
          res_closed = 0;
          res_repeat = 0;
          res_koef = 100.0;
          select res_closed, res_repeat, res_koef
          from aarep_lo_retention_koef(:res_lo_id, dateadd(day, -13, :res_date_end), :res_date_end)
          into :res_closed, res_repeat, res_koef;

          cpd_comment = cpd_comment || 'Удержание клиентов за две недели:<br>';
          cpd_comment = cpd_comment || 'Закрытые кредиты: <b>' || round(res_closed, 0) || '</b><br>';
          cpd_comment = cpd_comment || 'Повторные выдачи по закрывшим за период: <b>' || round(res_repeat, 0) || '</b><br>';

-- Вычисляем баллы по удержанию клиентов

          if (res_closed <> 0) then
          begin
            if (res_koef >= 40.0) then
              points_3 = 25.0; -- п.3.1 - коэф.удержания >=40 то 25 баллов
            else
            if (res_koef >= 35.0) then
              points_3 = 20.0; -- п.3.2 - коэф.удержания >=35 то 20 баллов
            else
            if (res_koef >= 30.0) then
              points_3 = 10.0; -- п.3.3 - коэф.удержания >=30 то 10 баллов
            else
            if (res_koef >= 25.0) then
              points_3 = 0.0; -- п.3.4 - коэф.удержания >=25 то 0 баллов
            else
              points_3 = -10.0; -- п.3.5 - иначе -10 балла
            cpd_comment = cpd_comment || 'Коэффициент удержания: <b>' || round(res_koef, 1) || '</b><br>';
          end
          else
          begin
            points_3 = 0.0;
            cpd_comment = cpd_comment || 'Коэффициент удержания: <b>---</b><br>';
          end
        end
      end

      points_all = points_1 + points_2 + points_3;

      if (res_team_id <> res_team_id_prior) then -- Если команда отличается
      begin
-- Находим нужную строку команды за этот период
        cpr_id = null;
        select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id and
              cpr_object_kind = -1 and
              cpr_object_id = :res_team_id
        into :cpr_id;

        if (cpr_id is null) then -- если такой строки нет, создаем новую
        begin
          cpr_group_name = '';

-- Получаем название Группы где последний раз была команда
          if (cpp_id <> 17) then -- Если это не первая неделя второго Этапа
          begin
            select cpr_group_name
            from competresult, competperiod
            where cpr_from_cpp_id = cpp_id and
                  cpp_from_rd_id = :rd_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id and
                  cpr_group_name <> '' and
                  cpp_orderno < :cpp_orderno
            order by cpp_orderno desc
            rows 1
            into :cpr_group_name;
          end
          else
          begin
-- Если это первая неделя второго этапа, название лиги берем из поля комментария предыдущего расчета
            select coalesce(cpr_comment, '')
            from competresult, competperiod
            where cpr_from_cpp_id = 16 and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id rows 1
            into :cpr_group_name;
          end
          if (cpr_group_name is null) then
            cpr_group_name = '';

          insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                    cpr_from_un_id, cpr_comment)
          values (:cpp_id, :cpr_group_name, -1, :res_team_id, 0, :un_id, '')
          returning cpr_id
          into :cpr_id;
        end
      end

-- Вставляем данные по сотруднику
      insert into competdetails (cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, cpd_points_1,
                                 cpd_points_2, cpd_points_3, cpd_comment, cpd_add_data_1)
      values (:cpr_id, 0, :res_lo_id, :points_all, :points_1, :points_2, :points_3, left(:cpd_comment, 4032), :ad_disb_loan);
    end

-- Расчет средней выдачи команды на одного СВЗ
-- Вступает в силу с 26.09.2016
    if (res_date_begin >= '26.09.2016') then
    begin
-- Первые 9 мест с начала
      avg_disb_prior = -777;
      kol = 0;
      cpr_group_name_prior = '';
      place_no = 0;
      for select cpr_id, cpr_group_name, cpr_object_id, round(avg(cast(cpd_add_data_1 as double precision)), 2)
          from competresult, competdetails
          where cpr_from_cpp_id = :cpp_id and
                cpr_group_name <> '' and
                cpr_object_kind = -1 -- Refdata, названия групп
                and
                cpd_from_cpr_id = cpr_id and
                not(cpd_comment starting 'Отпуск.')
          group by 1, 2, 3
          order by 2, 4 desc
          into :cpr_id, :cpr_group_name, res_team_id, avg_disb
      do
      begin
        points_3 = 0;
        points_4 = 0;
        if (cpr_group_name <> cpr_group_name_prior) then
        begin
          place_no = 0;
          kol = 0;
          avg_disb_prior = -777;
          cpr_group_name_prior = cpr_group_name;
        end

        if (avg_disb_prior <> avg_disb) then
        begin
          place_no = place_no + 1 + kol;
          avg_disb_prior = avg_disb;
          kol = 0;
        end
        else
          kol = kol + 1;

        if (place_no <= 9) then
        begin
          points_4 = (case
                        when place_no = 1 then 25
                        when place_no = 2 then 22
                        when place_no = 3 then 20
                        when place_no = 4 then 17
                        when place_no = 5 then 15
                        when place_no = 6 then 12
                        when place_no = 7 then 10
                        when place_no = 8 then 7
                        when place_no = 9 then 5
                      end);

          update competresult
          set cpr_comment = cpr_comment || iif(cpr_comment = '', '', ', ') || 'Ср.выдача на СВЗ: ' || round(:avg_disb, 1) || ', ' || round(:points_4, 0) || ' балл за ' || round(:place_no, 0) || ' место среди лучших'
          where cpr_id = :cpr_id;

          update competresult
          set cpr_additional_points = cpr_additional_points + :points_4
          where (cpr_id = :cpr_id);

        end
      end

-- Последние 6 мест с конца
      cpr_group_name_prior = '';
      place_no = 0;
      avg_disb_prior = -777;
      for select cpr_id, cpr_group_name, cpr_object_id, round(avg(cast(cpd_add_data_1 as double precision)), 2)
          from competresult, competdetails
          where cpr_from_cpp_id = :cpp_id and
                cpr_group_name <> '' and
                cpr_object_kind = -1 -- Refdata, названия групп
                and
                cpd_from_cpr_id = cpr_id and
                not(cpd_comment starting 'Отпуск.')
          group by 1, 2, 3
          order by 2, 4
          into :cpr_id, :cpr_group_name, res_team_id, avg_disb
      do
      begin
        points_3 = 0;
        points_4 = 0;

        if (cpr_group_name <> cpr_group_name_prior) then
        begin
          place_no = 0;
          kol = 0;
          avg_disb_prior = -777;
          cpr_group_name_prior = cpr_group_name;
        end

        if (avg_disb_prior <> avg_disb) then
        begin
          place_no = place_no + 1 + kol;
          avg_disb_prior = avg_disb;
          kol = 0;
        end
        else
          kol = kol + 1;

        if (place_no <= 6) then
        begin
          points_4 = (case
                        when place_no = 1 then -15
                        when place_no = 2 then -13
                        when place_no = 3 then -11
                        when place_no = 4 then -9
                        when place_no = 5 then -7
                        when place_no = 6 then -5
                      end);

          update competresult
          set cpr_comment = cpr_comment || iif(cpr_comment = '', '', ', ') || 'Ср.выдача на СВЗ: ' || round(:avg_disb, 1) || ', ' || round(:points_4, 0) || ' балл за ' || round(:place_no, 0) || ' место среди худших'
          where cpr_id = :cpr_id;

          update competresult
          set cpr_additional_points = cpr_additional_points + :points_4
          where (cpr_id = :cpr_id);
        end
      end

    end -- конец расчета средней выдачи на СВЗ в команде

-- Пересчитываем итоги по командам
    for select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id
        into :cpr_id
    do
    begin
      avg_points = 0.0;
      lo_panelty_count = 0.0;
      select round(avg(cpd_points_all), 2), -- средний балл
             sum(iif(cpd_points_1 < 0 -- кол-во свз со штрафами
             or cpd_points_2 < 0 or cpd_points_3 < 0 or cpd_points_4 < 0 or cpd_points_5 < 0 or cpd_points_6 < 0 or cpd_points_7 < 0 or cpd_points_8 < 0 or cpd_points_9 < 0, 1, 0))
      from competdetails
      where cpd_from_cpr_id = :cpr_id and
            not(cpd_comment starting 'Отпуск.')
      into :avg_points, :lo_panelty_count;

      if (avg_points is null) then
        avg_points = 0;
      if (lo_panelty_count is null) then
        lo_panelty_count = 0;

      total_points = avg_points; -- Средний балл комманды
      add_points = 0.0;

      if (lo_panelty_count = 0) -- Если никто не штрафован, плюс 2 балла
      then
        add_points = 2.0;

      if (lo_panelty_count >= 2) -- Если оштрафованы 2 и более СВЗ, минус 2 балла
      then
        add_points = -2.0;

      update competresult
      set cpr_calced_points = :total_points,
          cpr_additional_points = cpr_additional_points + :add_points,
          cpr_comment = cpr_comment || iif(cpr_comment = '', '', ', ') || 'Балл за минусовые СВЗ: ' || round(:add_points, 0)
      where (cpr_id = :cpr_id);
    end

-- Присвоение одного балла у кого самый больший балл в группе
    add_points = 1;
    for select cpr_group_name, max(cpr_final_points)
        from competresult
        where cpr_from_cpp_id = :cpp_id
        group by cpr_group_name
        into :group_name, max_points
    do
    begin
      update competresult
      set cpr_additional_points = cpr_additional_points + :add_points,
          cpr_comment = cpr_comment || iif(cpr_comment = '', '', ', ') || 'Балл за 1 место: ' || round(:add_points, 0)
      where cpr_from_cpp_id = :cpp_id and
            cpr_group_name = :group_name and
            cpr_final_points = :max_points;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  if (stage = 1) then -- если промежуточные итоги тура
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, sum(cpr_final_points)
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      cpr_group_name = '';

-- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  if (stage = 2) then -- если промежуточные итоги этапа
  begin
-- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, avg(cpr_final_points)
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      cpr_group_name = '';

-- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end

    execute procedure hp_competition_recalc_order(:cpp_id);
  end

  if (stage = 3) then -- если финальные итоги
  begin
    res_text = 'Автоматизация итогов еще не готов';
  end
  suspend;
end
