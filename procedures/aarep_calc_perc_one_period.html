CREATE PROCEDURE AAREP_CALC_PERC_ONE_PERIOD(
    K_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RESULT_PERC_BEGIN D_CURRENCY,
    RESULT_CALCED D_CURRENCY,
    RESULT_PAYED D_CURRENCY,
    RESULT_WRITEOFF D_CURRENCY,
    RESULT_PERC_END D_CURRENCY,
    RESULT_OUT_PERC_END D_CURRENCY,
    RESULT_OUT_PERC_BEGIN D_CURRENCY,
    RESULT_OUT_CALCED D_CURRENCY)
AS
declare variable REP_PERIOD_BEGIN D_INTEGER;
declare variable REP_PERIOD_END D_INTEGER;
declare variable K_ORIG ID_TYPE;
declare variable REP_DATE_BEGIN_PRV D_DATE;
declare variable REP_PERIOD_BEGIN_TMP D_INTEGER;
begin
  rep_period_begin = 0;
  rep_period_end = 0;

  result_perc_begin = 0;
  result_calced = 0;
  result_payed = 0;
  result_writeoff = 0;
  result_perc_end = 0;
  result_out_perc_begin = 0;
  result_out_calced = 0;
  result_out_perc_end = 0;

  rep_date_begin_prv = rep_date_begin - 1;

  if (extract(day from :rep_date_begin) = 1) then
    rep_period_begin = extract(year from :rep_date_begin) * 100 + extract(month from :rep_date_begin);

  if (rep_period_begin > 0) then
  begin
    if (not(exists(select *
                   from kontcalcperc
                   where kcp_from_k_id = :k_id and
                         kcp_period >= :rep_period_begin))) then
    begin
      k_orig = null;
      select kd_k_orig
      from kontdevides
      where kd_k_new = :k_id
      into :k_orig;
      if (k_orig is not null) then
      begin
        if (not(exists(select *
                       from kontcalcperc
                       where kcp_from_k_id = :k_orig and
                             kcp_period >= :rep_period_begin))) then
          rep_period_begin = 0;
      end
      else
        rep_period_begin = 0;
    end
  end

  if (rep_period_begin > 0) then -- начальный период совпадает
  begin
    select kcp_perc_begin_saldo, kcp_out_perc_begin_saldo
    from kontcalcperc
    where kcp_period = :rep_period_begin and
          kcp_from_k_id = :k_id
    into :result_perc_begin, :result_out_perc_begin;
  end else
  begin
    select result_perc_end, result_out_perc_end
    from aarep_calc_perc_one_saldo(:k_id, :rep_date_begin_prv, :rep_date_begin_prv)
    into :result_perc_begin, :result_out_perc_begin;
  end

  if (extract(day from :rep_date_end) = 1) then
    rep_period_end = extract(year from :rep_date_end) * 100 + extract(month from :rep_date_end);

  if (rep_period_end > 0) then
  begin
    if (not(exists(select *
                   from kontcalcperc
                   where kcp_from_k_id = :k_id and
                         kcp_period >= :rep_period_end))) then
    begin
      k_orig = null;
      select kd_k_orig
      from kontdevides
      where kd_k_new = :k_id
      into :k_orig;
      if (k_orig is not null) then
      begin
        if (not(exists(select *
                       from kontcalcperc
                       where kcp_from_k_id = :k_orig and
                             kcp_period >= :rep_period_end))) then
          rep_period_end = 0;
      end
      else
        rep_period_end = 0;
    end
  end

  if (rep_period_end > 0) then -- конечный период совпадает
  begin
    rep_period_begin_tmp = rep_period_begin;
    if (rep_period_begin_tmp = 0) then
      rep_period_begin_tmp = rep_period_end;

    select sum(kcp_perc_payed),
           sum(kcp_perc_writeoff),
           sum(iif(kcp_period = :rep_period_end, kcp_perc_saldo, 0)),
           sum(iif(kcp_period = :rep_period_end, kcp_out_perc_saldo, 0))
    from kontcalcperc
    where kcp_from_k_id = :k_id and
          kcp_period between :rep_period_begin_tmp and :rep_period_end
    into result_payed, result_writeoff, result_perc_end, result_out_perc_end;

    if (rep_period_begin = 0) then
    begin
      result_payed = 0;
      result_writeoff = 0;

      select sum(iif(ok_type = 4, -ok_perc, 0)), sum(iif(ok_type = 5, -ok_perc, 0))
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_from_k_id = :k_id
      into :result_payed, result_writeoff;

      if (result_payed is null) then
        result_payed = 0;

      if (result_writeoff is null) then
        result_writeoff = 0;
    end
  end else
  begin
    select result_payed, result_writeoff, result_perc_end, result_out_perc_end
    from aarep_calc_perc_one_saldo(:k_id, :rep_date_begin, :rep_date_end)
    into :result_payed, :result_writeoff, :result_perc_end, :result_out_perc_end;
  end

  result_calced = result_perc_end - result_perc_begin + (result_payed + result_writeoff);
  result_out_calced = (result_perc_end + result_out_perc_end) - (result_perc_begin + result_out_perc_begin) + (result_payed + result_writeoff) - result_calced;

  suspend;
end
