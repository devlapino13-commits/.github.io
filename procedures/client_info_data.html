CREATE PROCEDURE CLIENT_INFO_DATA(
    C_ID ID_TYPE)
RETURNS (
    CLIENT_DATA BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable cdc_code str_255;
declare variable cdv_value_string str_255;
declare variable rep_date date;
declare variable check_result str_100;
declare variable checked_date date;
declare variable is_svo smallint;
declare variable annul_status str_100;
declare variable cdv_value_number decimal(15,2);
declare variable c_name_fio str_100;
declare variable c_memo varchar(1024);
declare variable work_place_name str_255;
declare variable work_job_name str_255;
declare variable work_address_name str_255;
declare variable work_address_orientir str_255;
declare variable work_address_metro str_255;
declare variable work_status str_255;
declare variable svo_annul_status str_100;
declare variable svo_check_result str_100;
declare variable svo_checked_date date;
declare variable svo_work_reputation str_100;
declare variable svo_work_experience str_100;
declare variable svo_work_yield str_100;
declare variable svo_work_yield_sum curr_type;
declare variable svo_work_komment str_255;
declare variable cont_check_result str_100;
declare variable cont_checked_date date;
begin
  rep_date = current_date;

  for select cl.c_id, cl.c_name_fio, cl.c_memo
      from clients cl
      where c_id = :c_id
      into :c_id, :c_name_fio, :c_memo
  do
  begin
    work_place_name = '';
    work_job_name = '';
    work_address_name = '';
    work_address_orientir = '';
    work_address_metro = '';
    work_status = '';

-- Информация о месте работы из Анкеты-Личный лист
--    наименование организацц
--    должность
--    адрес работы
--    ориентир
--    ближайшее метро
--    работает/безработный
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_place_name', :rep_date)
    into work_place_name;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_job_name', :rep_date)
    into work_job_name;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_address_orientir', :rep_date)
    into work_address_orientir;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_address_metro', :rep_date)
    into work_address_metro;
    select ovs_value
    from hp_objectprops_get(0, :c_id, -1, 'work_status', :rep_date)
    into work_status;

    select adl_address_str
    from addrclient, addrlist
    where adc_from_c_id = :c_id and
          adc_address_type = 3 and
          adc_from_adl_id = adl_id
    into :work_address_name;

    if (coalesce(work_address_name, 'Не указан') = 'Не указан') then
      work_address_name = '';
    if (work_address_name = '') then
    begin
      select ovs_value
      from hp_objectprops_get(0, :c_id, -1, 'work_address_name', :rep_date)
      into work_address_name;
    end

--      Старый вариант, отключено 29.03.2016, Sanjari
-- Инфрмация указанная в карточке контрагента, владка - "Доп. ифно."
--      for select CDC_CODE, CDV_VALUE_STRING
--      from HP_CUSTOMDATACFG_ONE(0, 0, :C_ID, :REP_DATE)
--      into :cdc_code, :cdv_value_string
--      do
--      begin
--        if (cdc_code = 'work_place_name')       then work_place_name       = cdv_value_string; -- наименование организацц
--        if (cdc_code = 'work_job_name')         then work_job_name         = cdv_value_string; -- должность
--        if (cdc_code = 'work_address_name')     then work_address_name     = cdv_value_string; -- адрес работы
--        if (cdc_code = 'work_address_orientir') then work_address_orientir = cdv_value_string; -- ориентир
--        if (cdc_code = 'work_address_metro')    then work_address_metro    = cdv_value_string; -- ближайшее метро
--        if (cdc_code = 'work_status')           then work_status           = cdv_value_string; -- статус (работает/безрабтный)
--      end

    c_memo = coalesce(c_memo, '');
    c_memo = replace(c_memo, ascii_char(13), ' ');
    c_memo = replace(c_memo, ascii_char(10), ' ');

    work_address_name = iif(work_address_name <> '', work_address_name, 'Адрес не указан');

-- Информация указанная СВО, Контролерами
    for select iif(cont.contr_is_delete = 2, 'Аннулирован (Изменили данные по месту работы)', ''),
               cont_res.rd_name, cont.contr_checked_date,
               iif(cont_type.rd_code = 'ControllerType.Specialist_Visitor', 1, 0)
        from controller cont
        inner join refdata cont_type on cont_type.rd_id = cont.contr_rd_type
        inner join refdata cont_res on cont_res.rd_id = cont.contr_check_result
        where cont.contr_from_c_id = :c_id and
              cont_type.rd_code in ('ControllerType.Specialist_Visitor', 'ControllerType.Controller') and
              cont.contr_is_delete in (0, 2) and
              cont.contr_checked_date is not null
        order by cont.contr_checked_date desc
        into :annul_status, :check_result, :checked_date, :is_svo
    do
    begin
      if (is_svo = 1) then -- если это проверка СВО
      begin
        if (svo_checked_date is null) then -- беру только последний результат проверки СВО
        begin
          svo_annul_status = annul_status;
          svo_check_result = check_result;
          svo_checked_date = checked_date;

          for select cdc_code, cdv_value_string, cdv_value_number
              from hp_customdatacfg_one(0, 1, :c_id, :rep_date)
              into :cdc_code, :cdv_value_string, :cdv_value_number
          do
          begin
            if (cdc_code = 'work_reputation') then
              svo_work_reputation = cdv_value_string; -- репутация
            if (cdc_code = 'work_experience') then
              svo_work_experience = cdv_value_string; -- стаж работы
            if (cdc_code = 'work_yield') then
              svo_work_yield = cdv_value_string; -- доход
            if (cdc_code = 'work_yield_sum') then
              svo_work_yield_sum = cdv_value_number; -- сумма дохода
            if (cdc_code = 'work_komment') then
              svo_work_komment = cdv_value_string; -- комментрий СВО
          end
        end
      end
      else -- если это контролер
      begin
        if (cont_checked_date is null) then -- беру только последний результат проверки Контролером
        begin
          cont_check_result = check_result;
          cont_checked_date = checked_date;
        end
      end
    end

    client_data = 'Место работы: <b>' || coalesce(work_place_name, 'Не указан') || '</b><br>' || 'Должность:    <b>' || coalesce(work_job_name, 'Не указан') || '</b><br>' || 'Адрес работы: <b>' || coalesce(work_address_name, 'Не указан') || '</b><br>' || 'Ориентир:     <b>' || coalesce(work_address_orientir, 'Не указан') || '</b><br>' || 'Станция метро:<b>' || coalesce(work_address_metro, 'Не указан') || '</b><br>' || 'Статус:       <b>' || coalesce(work_status, 'Не указан') || '</b><br>';

    if (svo_checked_date is not null) then
    begin
      client_data = BLOB_APPEND(client_data, 'Репутатия:    <b>' || coalesce(svo_work_reputation, 'Не указан') || '</b><br>' || 'Стаж работы:  <b>' || coalesce(svo_work_experience, 'Не указан') || '</b><br>' || 'Доход:        <b>' || coalesce(svo_work_yield, 'Не указан') || '</b><br>' || 'Сумма дохода: <b>' || coalesce(svo_work_yield_sum, 'Не указан') || '</b><br>' || 'Комментарий:  <b>' || coalesce(svo_work_komment, 'Не указан') || '</b><br>');
    end
    client_data = BLOB_APPEND(client_data, 'Примечание:   <b>' || coalesce(c_memo, 'Не указан') || '</b><br>');
    suspend;
  end
end
