CREATE PROCEDURE HP_TXT_CLEAR(
    FIO_TXT VARCHAR(1024))
RETURNS (
    RES_TEXT VARCHAR(1024))
AS
declare variable sym_old varchar(1);
declare variable sym_new varchar(1);
declare variable len integer;
declare variable i integer;
declare variable fio_txt_orig varchar(1024);
begin
  fio_txt = trim(:fio_txt);
  if ((fio_txt is null) or (fio_txt = '')) then
  begin
    res_text = '';
    suspend;
    exit;
  end
  fio_txt = ' ' || fio_txt || ' ';
  fio_txt = replace(fio_txt, 'ң', 'н');
  fio_txt = replace(fio_txt, 'Ң', 'Н');
  fio_txt = replace(fio_txt, 'ү', 'у');
  fio_txt = replace(fio_txt, 'Ү', 'У');
  fio_txt = replace(fio_txt, '�©', 'о');
  fio_txt = replace(fio_txt, 'Ө', 'О');

  fio_txt = upper(fio_txt);

  fio_txt = replace(fio_txt, ' УУЛУ ', ' ');
  fio_txt = replace(fio_txt, ' КЫЗЫ ', ' ');
  fio_txt = replace(fio_txt, '-УУЛУ ', ' ');
  fio_txt = replace(fio_txt, '-КЫЗЫ ', ' ');

  fio_txt = replace(fio_txt, ' УГЛИ ', ' ');
  fio_txt = replace(fio_txt, ' ЎГЛИ ', ' ');
  fio_txt = replace(fio_txt, ' ЎГЛЫ ', ' ');
  fio_txt = replace(fio_txt, ' КИЗИ ', ' ');

  fio_txt = replace(fio_txt, ' O''GLI ', ' ');
  fio_txt = replace(fio_txt, ' OG''LI ', ' ');
  fio_txt = replace(fio_txt, ' O''G''LI ', ' ');
  fio_txt = replace(fio_txt, ' O''GLU ', ' ');
  fio_txt = replace(fio_txt, ' OG''LU ', ' ');
  fio_txt = replace(fio_txt, ' O''G''LU ', ' ');
  fio_txt = replace(fio_txt, ' QIZI ', ' ');

  fio_txt = replace(fio_txt, ' qizi ', ' ');
  fio_txt = replace(fio_txt, ' qızı ', ' ');
  fio_txt = replace(fio_txt, ' qizı ', ' ');
  fio_txt = replace(fio_txt, ' qızi ', ' ');
  fio_txt = replace(fio_txt, ' QIZI ', ' ');
  fio_txt = replace(fio_txt, ' QİZİ ', ' ');
  fio_txt = replace(fio_txt, ' QIZİ ', ' ');
  fio_txt = replace(fio_txt, ' QİZI ', ' ');

  fio_txt = replace(fio_txt, ' UULU ', ' ');
  fio_txt = replace(fio_txt, ' KYZY ', ' ');
  fio_txt = replace(fio_txt, '-UULU ', ' ');
  fio_txt = replace(fio_txt, '-KYZY ', ' ');
  fio_txt = replace(fio_txt, ' OGLU ', ' ');
  fio_txt = replace(fio_txt, ' OĞLU ', ' ');
  fio_txt = replace(fio_txt, ' OĞLİ ', ' ');
  fio_txt = replace(fio_txt, ' OĞLI ', ' ');

  fio_txt = replace(fio_txt, ' ОГЛЫ ', ' ');
  fio_txt = replace(fio_txt, ' ОГЛУ ', ' ');
  fio_txt = replace(fio_txt, ' ОГЛУУ ', ' ');
  fio_txt = replace(fio_txt, ' КЫЗИ ', ' ');
  fio_txt = replace(fio_txt, ' КИЗЫ ', ' ');

  len = char_length(fio_txt);
  fio_txt_orig = fio_txt;
  fio_txt = '';
  i = 1;
  while (i <= len) do
  begin
    sym_old = substring(fio_txt_orig from i for 1);
    sym_new = (case sym_old
                 when ':' then ' '
                 when ';' then ' '
                 when '«' then ' '
                 when '«' then ' '
                 when '»' then ' '
                 when '%' then ' '
                 when '_' then ' '
                 when '.' then ' '
                 when ',' then ','
                 when '`' then ''
                 when '|' then ' '
                 when '0' then ' '
                 when '1' then ' '
                 when '2' then ' '
                 when '3' then ' '
                 when '4' then ' '
                 when '5' then ' '
                 when '6' then ' '
                 when '7' then ' '
                 when '8' then ' '
                 when '9' then ' '
                 when '"' then ' '
                 when '(' then ' '
                 when ')' then ' '
                 when '[' then ' '
                 when ']' then ' '
                 when '{' then ' '
                 when '}' then ' '
                 when '/' then ' '
                 when '\' then ' '
                 when '&' then ' '
                 when '*' then ' '
                 when '$' then ' '
                 when '@' then ' '
                 when '!' then ' '
                 when '<' then ' '
                 when '>' then ' '
                 when '~' then ' '
                 when '?' then ' '
                 when '''' then ''
                 else :sym_old
               end);

    fio_txt = fio_txt || sym_new;
    i = i + 1;
  end
  while (position('  ', fio_txt) > 0) do
    fio_txt = replace(fio_txt, '  ', ' ');
  res_text = trim(:fio_txt);

  suspend;
end
