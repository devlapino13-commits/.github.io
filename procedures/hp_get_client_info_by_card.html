CREATE PROCEDURE HP_GET_CLIENT_INFO_BY_CARD(
    CARD_ID D_INTEGER)
RETURNS (
    C_ID D_ID,
    CARD_REGION_ID D_INTEGER,
    RES_TEXT D_STR_255)
AS
declare variable crlf varchar(2);
declare variable reg_name d_str_code;
declare variable is_error d_bool;
begin
  crlf = ascii_char(13)||ascii_char(10);
  is_error = 0;

  select
    c_id,
    'Клиент: ' || c_name_fio || iif(c_pass_no is not null, :crlf ||
      'Паспорт № ' || c_pass_no, '') || iif(c_inn is not null, :crlf ||
        'ИНН: ' || c_inn, '')
  from find_client_by_mb_card(:card_id, 'all')
  into
    :c_id,
    :res_text;

  if (c_id is null) then
  begin
    is_error = 1;
    res_text = 'Данная карта не выдана клиенту';
  end else begin
    select
      cpl.cpl_from_reg_id,
      r.reg_name
    from cardlist cal
      inner join cardplace cpl
        on cal.cal_last_cpl_id = cpl.cpl_id
      inner join region r
        on cpl.cpl_from_reg_id = r.reg_unicode
    where cal.cal_card_id = :card_id
    into
      :card_region_id,
      :reg_name;

    if (card_region_id is null) then
    begin
      is_error = 1;
      res_text = 'Офис выдачи карты не определен';
    end else
      res_text = res_text || :crlf || 'Офис выдачи карты: ' || reg_name;
  end

  if (is_error = 1) then
  begin
    c_id = -1;
    card_region_id = -1;
  end

  suspend;
end
