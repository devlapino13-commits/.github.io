CREATE PROCEDURE AAREP_LO_PERFORMANCE(
    REG_ID D_ID,
    REP_DATE D_DATE,
    C_ID D_ID = null)
RETURNS (
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REG_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    LO_STAGE_MONTH_FULL D_CURRENCY,
    LO_WEEK_DISB D_INTEGER,
    LO_PERFORMANCE D_CURRENCY,
    OTPUSK_DAYS INTEGER,
    LO_WORK_DAYS INTEGER,
    MGR_NAME D_STR_100,
    LO_WEEK_NEW_CLIENT D_INTEGER)
AS
declare variable KS_REGION_ID D_INTEGER;
declare variable KS_KC_ID D_ID;
declare variable N integer;
declare variable DD_TMP date;
begin
  /* loan officer performance - эффективность кредитного специалиста
  По заказу Бабура Тольбаева */
  if (c_id is not null) then
    reg_id = 1000000;

  rep_date_begin = rep_date - 7;
  rep_date_end = rep_date - 1;

  for with t_lo
      as (select ks_region_id, ks_kc_id, count(ks_from_k_id) count_lo_disb, sum(case when kc.kc_circle = 1 then 1 else 0 end) count_lo_new_client
          from region_list(:reg_id, 1, -1, 1)
          inner join kontstate ks on id_list = ks.ks_region_id and
                ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin between :rep_date_begin and :rep_date_end and
                ks.ks_kc_id = coalesce(:c_id, ks_kc_id)
          inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id and kc.kc_amount_approved > 0
          group by ks_region_id, ks_kc_id)
      select ks_region_id, r.reg_name, ks_kc_id, c_name_fio, count_lo_disb, count_lo_new_client
      from t_lo
      inner join region r on ks_region_id = reg_unicode
      inner join clients c on ks_kc_id = c_id
      order by count_lo_disb
      into ks_region_id, reg_name, ks_kc_id, c_name_fio, lo_week_disb, lo_week_new_client
  do
  begin

    -- Проверяем статус отпуска на каждый день недели:
    -- если все 5 дней был в отпуске, исключаем из списка
    otpusk_days = 0;
    n = 0;
    while (n <= 4) do
    begin
      dd_tmp = dateadd(day, n, rep_date_begin);
      if (exists(select *
                 from hr_voctravel, hr_orders, sa_worktype
                 where hrv_from_ho_id = ho_id and
                       ho_order_type = sawt_id and
                       hrv_from_c_id = :ks_kc_id and
                       sawt_base_type in (4, 5) and -- отпуск, декретный отпуск
                       sawt_id <> 20 and
                       sawt_hours_real >= 0 and
                       hrv_date_begin <= :dd_tmp and
                       hrv_date_end_fact >= :dd_tmp and
                       hrv_status = 1)) then
        otpusk_days = otpusk_days + 1;
      n = n + 1;
    end
    lo_work_days = 5 - otpusk_days;

    if (lo_work_days > 0) then
      lo_performance = lo_week_disb / cast(lo_work_days as decimal(18,2));
    else
      lo_performance = 0.0;

    lo_stage_month_full = null;
    select lo_stage_month_h
    from hp_get_lo_stage(:ks_kc_id, :rep_date_end)
    into lo_stage_month_full; --Стаж

    mgr_name = null;
    if (exists(select * from region
        where reg_unicode = :ks_region_id
        and reg_logic_label containing '.mega_office.')) then
    begin
      mgr_name = f_lo_mgr_name(f_lo_mgr_id(null, :rep_date_end, :ks_kc_id, :ks_region_id));
    end 

    suspend;
  end
end
