CREATE PROCEDURE AAREP_OSV_EPS_OSV_STATEMENT(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_ACC_ID ID_TYPE)
RETURNS (
    ST_LINE_NO INTEGER,
    ST_KOB_ID ID_TYPE,
    ST_DATE DATE,
    ST_ACC_ID ID_TYPE,
    ST_ACC_CODE D_STR_CODE,
    ST_ACC_NAME D_STR_255,
    ST_DT_SUM CURR_TYPE,
    ST_KT_SUM CURR_TYPE,
    ST_DT_VAL CURR_TYPE,
    ST_KT_VAL CURR_TYPE,
    ST_DT_TAX CURR_TYPE,
    ST_KT_TAX CURR_TYPE,
    ST_OB_DETAIL D_STR_LONG,
    ST_OPER_MEMO D_STR_LONG,
    ST_DATA_KIND INTEGER)
AS
declare variable total_dt_sum curr_type;
declare variable total_kt_sum curr_type;
declare variable total_dt_val curr_type;
declare variable total_kt_val curr_type;
declare variable total_dt_tax curr_type;
declare variable total_kt_tax curr_type;
begin
  st_data_kind = -1;
  st_line_no = 0;
  st_kob_id = null;
  st_date = rep_date_begin - 1;
  st_acc_id = null;
  st_acc_code = 'begin_saldo';
  st_acc_name = '';
  st_ob_detail = '';
  st_oper_memo = '';

  st_dt_sum = 0;
  st_kt_sum = 0;
  st_dt_val = 0;
  st_kt_val = 0;
  st_dt_tax = 0;
  st_kt_tax = 0;

  select kobs_bal_dt_sum, kobs_bal_kt_sum, kobs_bal_dt_val, kobs_bal_kt_val, kobs_bal_dt_tax, kobs_bal_kt_tax
  from kn_obsummary
  where kobs_acc_id = :rep_acc_id and
        kobs_date <= :st_date and
        kobs_date_end >= :st_date
  into :st_dt_sum, :st_kt_sum, :st_dt_val, :st_kt_val, :st_dt_tax, :st_kt_tax;

  suspend;

  st_data_kind = 0;
  total_dt_sum = 0;
  total_kt_sum = 0;
  total_dt_val = 0;
  total_kt_val = 0;
  total_dt_tax = 0;
  total_kt_tax = 0;

  for select kob_id, kob_ol_date, iif(kob_acc_dt = :rep_acc_id, kob_acc_kt, kob_acc_dt), obd_detail, ol_memo,
             iif(kob_acc_dt = :rep_acc_id, kob_summa, 0), iif(kob_acc_kt = :rep_acc_id, kob_summa, 0),
             iif(kob_acc_dt = :rep_acc_id, kob_summa_val, 0), iif(kob_acc_kt = :rep_acc_id, kob_summa_val, 0),
             iif(kob_acc_dt = :rep_acc_id, kob_summa_tax_dt, 0), iif(kob_acc_kt = :rep_acc_id, kob_summa_tax_kt, 0)
      from kn_operbook, operlist, kn_obdetail
      where kob_ol_is_active = 1 and
            kob_ol_date between :rep_date_begin and :rep_date_end and
            (kob_acc_dt = :rep_acc_id or
            kob_acc_kt = :rep_acc_id) and
            kob_from_ol_id = ol_id and
            kob_detail_id = obd_id and
            (kob_summa <> 0 or
            kob_summa_val <> 0 or
            kob_summa_tax_dt <> 0 or
            kob_summa_tax_kt <> 0)
      order by kob_ol_date, kob_from_ol_id, kob_id
      into :st_kob_id, :st_date, :st_acc_id, :st_ob_detail, :st_oper_memo, :st_dt_sum, :st_kt_sum, :st_dt_val,
           :st_kt_val, :st_dt_tax, :st_kt_tax
  do
  begin
    st_line_no = st_line_no + 1;
    st_acc_code = f_fmt_acc(st_acc_id);

    st_acc_name = '';
    select acc_name
    from ref_account
    where acc_id = :st_acc_id
    into st_acc_name;

    total_dt_sum = total_dt_sum + st_dt_sum;
    total_kt_sum = total_kt_sum + st_kt_sum;

    total_dt_val = total_dt_val + st_dt_val;
    total_kt_val = total_kt_val + st_kt_val;

    total_dt_tax = total_dt_tax + st_dt_tax;
    total_kt_tax = total_kt_tax + st_kt_tax;

    suspend;
  end

  st_data_kind = 1;
  st_line_no = st_line_no + 1;
  st_kob_id = null;
  st_date = rep_date_end + 1;
  st_acc_id = null;
  st_acc_code = 'end_saldo';
  st_acc_name = '';
  st_ob_detail = '';
  st_oper_memo = '';

  st_dt_sum = 0;
  st_kt_sum = 0;
  st_dt_val = 0;
  st_kt_val = 0;
  st_dt_tax = 0;
  st_kt_tax = 0;

  select kobs_bal_dt_sum, kobs_bal_kt_sum, kobs_bal_dt_val, kobs_bal_kt_val, kobs_bal_dt_tax, kobs_bal_kt_tax
  from kn_obsummary
  where kobs_acc_id = :rep_acc_id and
        kobs_date <= :rep_date_end and
        kobs_date_end >= :rep_date_end
  into :st_dt_sum, :st_kt_sum, :st_dt_val, :st_kt_val, :st_dt_tax, :st_kt_tax;
  suspend;
end
