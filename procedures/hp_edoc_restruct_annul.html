CREATE PROCEDURE HP_EDOC_RESTRUCT_ANNUL
AS
declare variable edc_id d_id;
begin
    -- RFCKZ-587

    -- Аннулируем "Заявление на реструктуризацию"
    edc_id = (select edc.edc_id from edoc_cfg edc where edc.edc_code = 'restruct_application');
    update edoc_list edl
    set edl.edl_status = -1
    where edl.edl_from_edc_id = :edc_id
        and edl.edl_blank_date between current_date - 7 and current_date
        and edl.edl_status = 0;
    
    -- Аннулируем "Доп соглашение на реструктуризацию"
    edc_id = (select edc.edc_id from edoc_cfg edc where edc.edc_code = 'add_loan_consent');
    update edoc_list edl
    set edl.edl_status = -1
    where edl.edl_from_edc_id = :edc_id
        and edl.edl_blank_date between current_date - 7 and current_date
        and edl.edl_status = 0;
end
