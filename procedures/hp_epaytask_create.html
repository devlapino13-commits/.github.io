CREATE PROCEDURE HP_EPAYTASK_CREATE(
    K_ID D_ID)
AS
declare variable pr_ipf boolean;
declare variable pr_goods boolean;
declare variable pr_sugar boolean;
declare variable acc_absent boolean;
declare variable seller_id d_id;
declare variable elsom_account d_str_20;
declare variable elcard_account d_str_20;
declare variable mbank_account d_str_20;
declare variable pay_amount d_currency;
declare variable un_id d_id;
declare variable master_acl_id d_id;
declare variable child_acl_id d_id;
declare variable reg_id d_integer;
declare variable ag_id d_integer;
declare variable rd_code d_str_255;
declare variable qr_account d_str_8100;
declare variable acl_id d_id;
begin
  select
    pr_code containing '.ipf.',
    pr_code containing '.installment_goods.',
    pr_code containing '.sugarmill.',
    k_last_reg_id,
    un_id
  from kontract
    inner join product
      on pr_id = k_from_pr_id
    inner join username
      on un_from_c_id = k_approved_manager
  where k_id = :k_id
    and k_status = 1
  into
    :pr_ipf,
    :pr_goods,
    :pr_sugar,
    :reg_id,
    :un_id;
  if (row_count = 0 or not (pr_ipf or pr_goods or pr_sugar)) then
    exit;

  if (pr_goods or pr_sugar) then
  begin
    -- товары в рассрочку
    if (pr_goods) then
      select
        cs_from_c_id,
        mp_amount * (1 - cs_discount / 100)
      from kontract
        inner join mc_purchase
          on mp_from_ma_id = k_from_ma_app_id
        inner join clientshops
          on cs_id = mp_from_cs_id
            and cs_from_rd_id = f_get_rd_id('ClientShops.outlets')
            and cs_is_delete = 0
      where k_id = :k_id
      into
        :seller_id,
        :pay_amount;
    else
    -- сах.завод
    if (pr_sugar) then
      select
        spt_seller_id,
        spt_amount
      from sugarpaymenttask
      where spt_k_id = :k_id
      into
        :seller_id,
        :pay_amount;

    select first 1
      ag_id,
      f_get_rd_code(ag_from_pay_rd_id)
    from agreements
    where ag_object_kind = 0
      and ag_object_id = :seller_id
      and current_date between
        coalesce(ag_date_begin, '01.01.1900') and
        coalesce(ag_date_end, '31.12.9999')
      and ag_from_rd_id = f_get_rd_id('AgreementKinds.installment_goods')
      and ag_status = 1
    order by ag_id desc
    into
      :ag_id,
      :rd_code;

    if (ag_id is null) then
      exit;
  end

  -- если есть задачи по контракту, перешедшие на уровень оплаты
  if (exists(
    select *
    from kontpaytasks
    where kpr_from_k_id = :k_id
      and kpt_status >= 0))
  then exit;

  -- 'откатываем'(если таковые есть) все задачи по оплате в режиме 'ожидания'
  -- для данного контракта
  select acl.acl_id
  from actionlist acl
  where acl_object_id = :k_id
    and acl_object_kind = 9
    and acl_from_rd_id = f_get_rd_id('EPayProjects')
    and acl_status = 0
  into :acl_id;

  if (acl_id is not null) then
  begin
      update actionlist set
        acl_result_rd_id = f_get_rd_id('EPayTaskResults.TaskCancel'),
        acl_status = -2
      where acl_id = :acl_id;

      update actionlist acl set
        acl_result_rd_id = f_get_rd_id('EPayTaskResults.TaskCancel'),
        acl_status = -2
      where acl.acl_parent_id = :acl_id
      and acl.acl_status = 0;
  end

  -- список продавцов в рамках контракта с их счетами и суммами
  for
    select
      :ag_id,
      :seller_id,
      :rd_code,
      :pay_amount
    from rdb$database
    where :pr_goods or :pr_sugar
    union
    select
      ag_id,
      ag_object_id,
      f_get_rd_code(ag_from_pay_rd_id),
      coalesce(sum(kml_total), 0)
    from kontmat
      inner join agreements
        on ag_id = kml_from_ag_id
          and ag_from_rd_id in (
            f_get_rd_id('AgreementKinds.IPF_Sellers'),
            f_get_rd_id('AgreementKinds.TSP_and_IPF'),
            f_get_rd_id('AgreementKinds.IPF_WithCommission'),
            f_get_rd_id('AgreementKinds.IPF_Idjara'))
          and ag_status = 1
    where :pr_ipf
      and kml_object_kind = 9
      and kml_object_id = :k_id
      and kml_status = 1
    group by
      ag_id,
      ag_object_id,
      ag_from_pay_rd_id
  into
    :ag_id,
    :seller_id,
    :rd_code,
    :pay_amount
  do
  begin
    select
      trim(coalesce(ag_elsom_account, '')),
      trim(coalesce(ag_elcard_account, '')),
      trim(coalesce(ag_mbank_account, ''))
    from agreements ag
    where ag_id = :ag_id
    into
      :elsom_account,
      :elcard_account,
      :mbank_account;

    if (rd_code = 'EPayProjects.QR') then
        qr_account = coalesce((select url_data from agreement_payment_qr_json(:ag_id)), '');

    acc_absent = (
      (rd_code = 'EPayProjects.ElSom' and elsom_account = '') or
      (rd_code = 'EPayProjects.ElCard' and elcard_account = '') or
      (rd_code = 'EPayProjects.MBank' and mbank_account = '') or
      (rd_code = 'EPayProjects.QR' and qr_account = ''));

    if ((pr_sugar or pr_goods) and acc_absent) then
    begin
      execute procedure hp_sugarpaymenttask_noacc(:k_id, :pr_sugar);
      exit;
    end

    -- родительская запись в Actionlist на уровне договора
    if (master_acl_id is null) then
      execute procedure hp_actionlist_insert(
        null,
        :reg_id,
        'Kontract',
        :k_id,
        'EPayProjects',
        0,
        :un_id)
      returning_values :master_acl_id;

    -- создаю задачу на оплату
    child_acl_id = null;
    execute procedure hp_actionlist_insert(
      :master_acl_id,
      :reg_id,
      'Clients',
      :seller_id,
      :rd_code,
      0,
      :un_id)
    returning_values :child_acl_id;

    if (pay_amount = 0)
    then
      update actionlist set
        acl_result_rd_id = f_get_rd_id('EPayTaskResults.ZeroSumm'),
        acl_status = -2
      where acl_id = :child_acl_id;
    else
      if (acc_absent) then
        update actionlist set
          acl_result_rd_id = f_get_rd_id('EPayTaskResults.AccountAbsent'),
          acl_status = -2
        where acl_id = :child_acl_id;
  end
end
