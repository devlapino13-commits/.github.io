CREATE PROCEDURE MC_PURCHASE_SAVE(
    MCL_ID D_ID,
    MP_RECEIPT_NUMBER D_STR_20,
    FEG_ID D_ID,
    MP_AMOUNT D_CURRENCY)
RETURNS (
    ERROR_TXT D_STR_255,
    UID D_STR_CODE,
    MP_ID D_ID)
AS
declare variable cs_id d_id;
declare variable temp_uid d_uuid;
begin
  if (f_get_db_id() not in (39801, 41701)) then
  begin
    error_txt = '"Товары в рассрочку" доступны в Кыргызстане.';
    suspend;
    exit;
  end

  if (coalesce(:mp_receipt_number, '') = '') then
  begin
    error_txt = 'Номер чека введен некорректно';
    suspend;
    exit;
  end

  if (coalesce(:feg_id, -777) = -777) then
  begin
    error_txt = 'Категория не выбрана.';
    suspend;
    exit;
  end

  if (coalesce(:mp_amount, 0) <= 0) then
  begin
    error_txt = 'Сумма к оплате должна быть больше нуля.';
    suspend;
    exit;
  end

  select cs.cs_id
  from clientshops cs
  inner join mc_login mcl on cs.cs_from_c_id = mcl.mcl_from_c_id
  where mcl.mcl_id = :mcl_id and
        cs.cs_is_delete = 0 and
        trim(cs.cs_phone) = mcl.mcl_login
  into :cs_id;

  if (coalesce(:cs_id, -777) <> -777) then
  begin
    insert into mc_purchase (mp_from_mcl_id, mp_from_cs_id, mp_date, mp_datetime, mp_amount, mp_receipt_number,
                             mp_status, mp_from_feg_id, mp_is_finished)
    values (null, :cs_id, current_date, current_timestamp, :mp_amount, :mp_receipt_number, 1, :feg_id, 0)
    returning mp_id
    into mp_id;

    insert into uuidlist (udl_object_kind, udl_object_id)
    values (f_objkind('mc_purchase', 'id'), :mp_id)
    returning udl_uuid
    into :temp_uid;

    :uid = uuid_to_char(:temp_uid);
    error_txt = '';

    suspend;
  end
  else
  begin
    error_txt = 'Агент не найден.';
    suspend;
    exit;
  end
end
