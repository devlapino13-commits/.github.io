CREATE PROCEDURE HP_KNCLIENT_PDN_UPDATE(
    KC_ID D_ID,
    USE_EXIST_EIL_ID D_SMALLINT = 0,
    CML_ID D_SMALLINT = null,
    BASE_CALC_DATE D_DATE = null,
    REFINANCE_KONT_LIST D_STR_LONG = null,
    KCP_KIND D_SMALLINT = null,
    MODE D_SMALLINT = null,
    TAG D_STR_CODE = null)
AS
declare variable k_disb_date d_date;
declare variable un_id d_id;
declare variable history_count d_integer;
declare variable calc_date d_date;
declare variable eil_id d_id;
declare variable ssp_eil_id d_id;
declare variable calc_eil_date d_date;
declare variable kcp_id d_id;
declare variable k_id d_id;
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable k_refund d_smallint;
declare variable tag_id d_integer;
declare variable tracelist d_str_long;
declare variable tracelist_id d_id;
declare variable ks_state d_integer;
declare variable pr_code d_str_100;
declare variable ok_cell_main d_str_100;
declare variable bsc_scoring_level d_integer;
declare variable ma_summa_minus d_currency;
declare variable ma_no d_integer;
declare variable client_point d_integer;
declare variable ki_jso d_integer;
declare variable ssp_jso d_integer;
declare variable ki_result_id d_integer;
declare variable ki_result d_str_code;
declare variable ssp_result_id d_integer;
declare variable ssp_result d_str_code;
begin
  -- если передан use_exist_eil_id = 1, то не будем искать новую истори БКИ,
  -- берем ранее использованную из knclient_pdn.kc_from_eil_id
  -- cml_id - при реструктуризации ID задачи КК

  un_id = coalesce(uxf$usr_un(null, 0),1000000000000);

  if (kcp_kind is null and cml_id is not null) then
    kcp_kind = 1;

  kcp_kind = coalesce(kcp_kind, 0);  -- Тип расчета: 0 - при выдаче, 1 - при реструктуризации, 2 - 60 дней после выдачи (рефинанс)
  mode = coalesce(mode, 0);  -- режим запуска: 0 - автоматический, 1 - вручную
  use_exist_eil_id = coalesce(use_exist_eil_id, 0);

  select k_disb_date, k_id, k_refund, pr_code, kc_from_c_id, k_from_ma_app_id
  from knclient, kontract, product
  where kc_id = :kc_id
    and kc_from_k_id = k_id
    and pr_id = k_from_pr_id
  into k_disb_date, k_id, k_refund, pr_code, c_id, ma_id;

  if (kcp_kind = 0 and mode = 0 and f_edoc_status('loan.agreement', k_id, k_disb_date) in ('signed', 'unsigned'))
  then
    if (tag = 'check_preview') then -- исключение на просмотр условий
      exit;
    else
      exception e_message 'ПДН не может быть пересчитан т.к. личный лист подписан или ожидает подписания';

  if (kcp_kind = 0 and k_refund = 0) then
    delete from kontrefinance
    where kr_from_ma_id = :ma_id;

  if (kcp_kind in (0,2)) then    -- выдача, 60 дней рефинанс
  begin
    select kcp_id, kc_date
    from knclient_pdn
    where kc_id = :kc_id
      and kcp_kind = :kcp_kind
    into kcp_id, calc_date;

    if (calc_date is null) then
      calc_date = current_date;

    if (kcp_kind = 0) then
      calc_date = minvalue(k_disb_date, current_date);
  end
  else if (kcp_kind = 1) then    -- реструктуризация
  begin
    select kcp_id, cml_date
    from knclient_pdn, com_list, com_cfg
    where cml_id = :cml_id
      and cml_from_kcp_id = kcp_id
      and cml_from_cmc_id = cmc_id
    into kcp_id, calc_date;

    if (calc_date is null) then
      select cml_date
      from com_list
      where cml_id = :cml_id
      into calc_date;
  end
  else
    exception e_default 'Unknown kcp_kind = ' || coalesce(kcp_kind, '<NULL>');

  if (kcp_id is null) then
  begin
    insert into knclient_pdn (kc_id, kcp_kind, kc_from_c_id, kc_date, kc_from_un_id)
    values (:kc_id, :kcp_kind, :c_id, :calc_date, :un_id)
    returning kcp_id
    into kcp_id;

    if (kcp_kind = 1) then
      update com_list
      set cml_from_kcp_id = :kcp_id
      where cml_id = :cml_id;
  end
  else
  begin
    delete from knclient_pdnlist
    where kpl_from_kcp_id = :kcp_id;

    update knclient_pdn
    set kc_id = :kc_id,
        kc_from_c_id = :c_id,
        kc_from_un_id = :un_id
    where kcp_id = :kcp_id;

    -- RFCRU-2652 Пока кредит не выдан, то поле с адресом надо обновлять
    if (not exists(select 1 from kontstate, knclient
                where ks_from_k_id = kc_from_k_id
                  and kc_id = :kc_id
                  and ks_ol_is_delete = 0
                  and ks_state > 0
                  and current_date between ks_date_begin and ks_date_end))
    then
      update knclient_pdn
      set kc_from_adl_id = null
      where kcp_id = :kcp_id;
  end

  -- RFCRU-3169. Выявляем откуда вызван расчет ПДН. Пишем лог.
  tag_id = f_enum('pdn_calc_source', tag, 'id', 0);

  if (tag_id is null) then
  begin
    with recursive head as (
      select 1 level_,
        cl.mon$statement_id, cl.mon$call_id,
        cl.mon$object_name, cl.mon$object_type
      from mon$call_stack cl
      where cl.mon$caller_id is null
      union all
      select level_ + 1,
        cl.mon$statement_id, cl.mon$call_id,
        cl.mon$object_name, cl.mon$object_type
      from mon$call_stack cl
        join head on cl.mon$caller_id = head.mon$call_id
    ),
    t1 as (
      select level_, mon$object_name
      from head
        join mon$statements stmt on stmt.mon$statement_id = head.mon$statement_id
      where stmt.mon$attachment_id = current_connection
      order by 1
    )
    select list('.'||trim(mon$object_name)||'.', '')
    from t1
    into tracelist;

    tracelist_id = f_memo_id(coalesce(tag, '<NULL>') || ';' || tracelist);
  end 

  history_count = 0;

  if (not exists(select kc_from_k_id from knclient
                 where kc_id = :kc_id
                   and kc_from_c_id is not null))
  then
    exception e_default 'Нет займа KC_ID: ' || coalesce(kc_id, '<NULL>');

  if (use_exist_eil_id = 1) then
  begin
    select kc_from_eil_id, kcp_kbki_eil_id
    from knclient_pdn
    where kcp_id = :kcp_id
    into eil_id, ssp_eil_id;

    select json.parse(blx_blob)
    from extinfolist, hp_get_blobex_blob(eil_from_bls_id)
    where eil_id = :eil_id
    into ki_jso;

    select json.parse(blx_blob)
    from extinfolist, hp_get_blobex_blob(eil_from_bls_id)
    where eil_id = :ssp_eil_id
    into ssp_jso;
  end
  else
  begin
    ki_jso = rdb$get_context('USER_TRANSACTION', 'bki_ki_jso_for_c_id_' || c_id);
    ssp_jso = rdb$get_context('USER_TRANSACTION', 'bki_ssp_jso_for_c_id_' || c_id);
    eil_id = rdb$get_context('USER_TRANSACTION', 'bki_ki_eil_for_c_id_' || c_id);
    ssp_eil_id = rdb$get_context('USER_TRANSACTION', 'bki_ssp_eil_for_c_id_' || c_id);
  
    if (ki_jso + ssp_jso is null) then
    begin
      select ki_result_id, ki_result, ki_jso, ssp_result_id, ssp_result, ssp_jso, ki_eil_id, ssp_eil_id
      from hp_pdn_get_request_status(:c_id, null, :calc_date, :ma_id)
      into ki_result_id, ki_result, ki_jso, ssp_result_id, ssp_result, ssp_jso, eil_id, ssp_eil_id;
    
      if (coalesce(ki_result_id, 0) not in (5,6,9)) then
        exception e_message 'Ошибка получения КИ: ' || ki_result;
  
      if (coalesce(ssp_result_id, 0) not in (5,6,9)) then
        exception e_message 'Ошибка получения ССП: ' || ssp_result;
  
      rdb$set_context('USER_TRANSACTION', 'bki_ssp_jso_for_c_id_' || c_id, ssp_jso);
      rdb$set_context('USER_TRANSACTION', 'bki_ki_jso_for_c_id_' || c_id, ki_jso);
      rdb$set_context('USER_TRANSACTION', 'bki_ki_eil_for_c_id_' || c_id, eil_id);
      rdb$set_context('USER_TRANSACTION', 'bki_ssp_eil_for_c_id_' || c_id, ssp_eil_id);
    end
  end

  --- RFCRU-3235 Заполнение информации по самозапрету
  if (calc_date = current_date) then -- не выполняем если расчет задним числом
    execute procedure hp_selfblock_save(:ssp_eil_id);

  execute procedure hp_extinfolist_pdn_amp_update(:kcp_id, :kc_id, maxvalue(:k_disb_date, :calc_date), :ssp_jso);

  select eil_date
  from extinfolist
  where eil_id = :eil_id
  into calc_eil_date;

  for select
    rec_no, creduid, tag, credcurrency, creddate, credenddate, credenddatefact, credupdate,
    credsumdebt, credsumoverdue, creddayoverdue, credfullcost, credsum, credpartnertype_id, credtype_id,
    credactive_str, credcollateral, credcollateralitemtype_id, credcollateralfactenddate, credguarantee,
    credguaranteefactenddate, credcategory_id, credconsumer, credratio_id, credpurpose_id, calc_koef_period,
    calc_period_month, calc_month_repay, calc_is_credit_card, kc_id, kc_stat_avg_income,
    kc_anketa_income, kc_final_income, final_pdn, res_last_income_period, adl_id, str_id,
    res_calc_date, calc_month_repay_final, use_amp, credit_line_type_id, cred_owner_type_id,
    debtcurr_opsum, debtcurr_percentsum, debtcurr_othersum, debtoverdue_opsum, debtoverdue_percentsum,
    debtoverdue_othersum, fullcost_sum, fullcost_date, average_payment_date, kc_anketa_income_calc, kc_documented_income,
    gracedate, graceenddate, dealsigncreditcard, minsumpaycc, income_discount_perc, exclusion_reason, jso_log, debtoverdue_date
  from hp_extinfolist_pdn_one_loan(:kcp_id, :kc_id, :calc_eil_date, :calc_date, :ki_jso, 'equifax_40', :kcp_kind, :cml_id)
  as cursor a
  do
  begin
    if (a.rec_no <> 0 and a.rec_no <> -1) then
      history_count = history_count + 1;

    if (a.rec_no <> -1) then
    begin
      insert into knclient_pdnlist (
        kpl_from_kcp_id, kpl_rec_no, kpl_credid, kpl_credcurrency, kpl_creddate,
        kpl_credenddate, kpl_credenddatefact, kpl_credupdate, kpl_credsumdebt, kpl_credsumoverdue,
        kpl_creddayoverdue, kpl_credfullcost, kpl_credsum, kpl_credpartnertype, kpl_credtype,
        kpl_credactive_id, kpl_status_id, kpl_koef_period, kpl_period_month, kpl_is_credit_card,
        kpl_month_repay, kpl_history_kc_id, kpl_zalog, kpl_poruch,
        kpl_tag, kpl_credcategory, kpl_month_repay_final,
        kpl_credguaranteefactenddate, kpl_credcollateralitemtype, kpl_credcollateralfactenddate,
        kpl_credconsumer, kpl_credratio, kpl_credpurpose, kpl_use_amp, kpl_credit_line_type, kpl_cred_owner_type,
        kpl_debtcurr_opsum, kpl_debtcurr_percentsum, kpl_debtcurr_othersum, kpl_debtoverdue_opsum,
        kpl_debtoverdue_percentsum, kpl_debtoverdue_othersum, kpl_fullcost_sum, kpl_fullcost_date,
        kpl_average_payment_date, kpl_gracedate, kpl_graceenddate, kpl_dealsigncreditcard,
        kpl_minsumpaycc, kpl_exclusion_reason, kpl_debtoverdue_date)
      values (
      :kcp_id, a.rec_no, a.creduid, a.credcurrency, a.creddate,
      a.credenddate, a.credenddatefact, a.credupdate, a.credsumdebt, a.credsumoverdue,
      a.creddayoverdue, a.credfullcost, a.credsum, a.credpartnertype_id, a.credtype_id,
      f_memo_id(a.credactive_str), f_memo_id('История получена'), a.calc_koef_period, a.calc_period_month, a.calc_is_credit_card,
      a.calc_month_repay, a.kc_id, a.credCollateral, a.credGuarantee,
      a.tag, a.credcategory_id, a.calc_month_repay_final,
      a.credguaranteefactenddate, a.credcollateralitemtype_id, a.credcollateralfactenddate,
      a.credconsumer, a.credratio_id, a.credpurpose_id, a.use_amp, a.credit_line_type_id, a.cred_owner_type_id,
      a.debtcurr_opsum, a.debtcurr_percentsum, a.debtcurr_othersum, a.debtoverdue_opsum, a.debtoverdue_percentsum,
      a.debtoverdue_othersum, a.fullcost_sum, a.fullcost_date, a.average_payment_date,
      a.gracedate, a.graceenddate, a.dealsigncreditcard, a.minsumpaycc, a.exclusion_reason, a.debtoverdue_date);
    end else
    begin
      update knclient_pdn
      set kc_status = iif(:eil_id is null, -1, 0),
        kc_date = a.res_calc_date,
        kc_from_adl_id = a.adl_id,
        kc_from_str_id = a.str_id,
        kc_from_eil_id = :eil_id,
        kcp_kbki_eil_id = :ssp_eil_id,
        kc_stat_last_period = a.res_last_income_period,
        kc_stat_avg_income = a.kc_stat_avg_income,
        kc_anketa_income = a.kc_anketa_income,
        kc_documented_income = a.kc_documented_income,
        kc_final_income = a.kc_final_income,
        kc_loan_expense = a.calc_month_repay_final,
        kc_pdn_koef = a.final_pdn,
        kc_from_mml_id = 0,
        kc_anketa_income_calc = a.kc_anketa_income_calc,
        kcp_income_discount_perc = a.income_discount_perc
      where kcp_id = :kcp_id;

      insert into knclient_pdn_log (kplg_from_kcp_id, kplg_from_un_id, kplg_tag_id, kplg_tracelist_id, kplg_pdn_koef, kplg_sp_new, kplg_sp_other,
                                    kplg_calc_period_month, kplg_percent_sum, kplg_income, kplg_creddate, kplg_credenddate, kplg_calc_date)
      values (:kcp_id, :un_id, :tag_id, :tracelist_id, a.final_pdn,
            json.obj_get_val(a.jso_log, 'sp_new'),
            a.calc_month_repay_final - cast(json.obj_get_val(a.jso_log, 'sp_new', 0) as d_currency),
            json.obj_get_val(a.jso_log, 'calc_period_month'),
            json.obj_get_val(a.jso_log, 'percent_sum'),
            a.kc_final_income,
            json.obj_get_val(a.jso_log, 'creddate'),
            json.obj_get_val(a.jso_log, 'credenddate'),
            json.obj_get_val(a.jso_log, 'calc_date'));
    end
  end

  if (history_count = 0) then
    update knclient_pdnlist
    set kpl_status_id = f_memo_id('Клиент не имеет кредитной истории')
    where kpl_from_kcp_id = :kcp_id and
          kpl_status_id = f_memo_id('История получена');

  --- RFCRU-3640 Скоринг для продуктов "Ойва" (Oiva) и "Универсальный"_от Скоринг Бюро
  ---8.3.  Если результат скоринга низкий и у клиента менее 13 баллов по программе лояльности, в займе должно быть отказано автоматически,
  -- причина отказа «Отрицательная кредитная история клиента»
  if (kcp_kind = 0  -- при выдаче
      and mode = 0  -- автоматический запуск
      and (pr_code containing ('.universal.') or pr_code containing ('.oivo_pack.')))
  then
  begin
    -- проверим что займ в процессе
    ks_state = 1;

    select ks_state
    from kontstate
    where ks_from_k_id = :k_id
      and :calc_date between ks_date_begin and ks_date_end
      and ks_ol_is_delete = 0
    into ks_state;

    client_point = f_point_get(9, k_id, c_id, calc_date);

    if (ks_state = 0 and client_point < 13) then
    begin
      select bsc_scoring_level
      from bki_scoring
      where bsc_from_kc_id = :kc_id
      into bsc_scoring_level;

      if (bsc_scoring_level = 0) then
      begin
        insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                                 krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
        values (3, 1000000000000, :k_id, 0, null, current_timestamp, current_date, current_date, 0);
    
        update kontract
        set k_status = -1,
            k_user_id = 1000000000000
        where k_id = :k_id;
  
        select ok_cell_main
        from continfo
        where cci_object_kind = 0
        and cci_object_id = :c_id
        rows 1
        into ok_cell_main;
  
        select ma_summa_minus, ma_no
        from miniapp ma
        where ma_id = :ma_id
        order by ma_number desc
        rows 1
        into ma_summa_minus, ma_no;
  
        if (coalesce(ok_cell_main, '') <> '') then
          execute procedure hp_send_quick_message('sms', :ok_cell_main, '',
              f_format('По вашей заявке на заем №%1 на сумму %2руб. отказано в связи с отрицательной кредитной историей. ООО «МКК М Булак».', ma_no, ma_summa_minus),
               '', -1, 1000000000000);
  
        execute procedure fsrobot_insert_event('DoMailing', 'Push_scoring_miniapp_cancel', f_format('%1;%2;', :ma_id, :c_id), :un_id); -- RFCRU-3652
      end
    end
  end
end
