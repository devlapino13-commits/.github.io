CREATE PROCEDURE AAREP_LLC_REPORT(
    REG_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    LLC_NAME VARCHAR(50),
    LLC_RESERV_KIND VARCHAR(50),
    LLC_OFFICE_NAME VARCHAR(100),
    LLC_BRANCH_NAME VARCHAR(100),
    BAL_GR DECIMAL(18,4),
    RPPU_GR DECIMAL(18,4),
    BAL_IND DECIMAL(18,4),
    RPPU_IND DECIMAL(18,4),
    PERC_GR DECIMAL(18,7),
    PERC_IND DECIMAL(18,7),
    BAL DECIMAL(18,4),
    PERC DECIMAL(15,7),
    RPPU DECIMAL(18,4))
AS
begin
  llc_reserv_kind = '';
  for select llc_name, llc_office_name, llc_branch_name, sum(llc_gr_balance), sum(llc_gr_rppu), sum(llc_ind_balance),
             sum(llc_ind_rppu), sum(llc_k_balance), sum(llc_rppu) as rppu
      from aarep_llc_report_temp(:reg_id, :rep_date)
      group by llc_name, llc_office_name, llc_branch_name
      order by llc_office_name
      into :llc_name, :llc_office_name, :llc_branch_name, :bal_gr, :rppu_gr, :bal_ind, :rppu_ind, :bal, :rppu
  do
  begin
    perc_gr = 0;
    perc_ind = 0;
    perc = 0;
    if (bal_gr <> 0) then
      perc_gr = cast((rppu_gr / bal_gr * 100) as decimal(15,2));
    if (bal_ind <> 0) then
      perc_ind = cast((rppu_ind / bal_ind * 100) as decimal(15,2));
    if (bal <> 0) then
      perc = cast((rppu / bal * 100) as decimal(15,2));
    suspend;
  end
end
