CREATE PROCEDURE HP_OPERKRED_DD_ALL_CHANGE(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_DATE_NEW DATE,
    REPORT_USER_ID DECIMAL(18,0))
AS
declare variable reg_id integer;
begin
  for select reg_unicode
      from region, region_orig_list(:report_region_id, 1, -1, 1)
      where reg_unicode = id_list and
            reg_status = 3
      order by reg_unicode
      into :reg_id
  do
  begin
    execute procedure hp_operkred_dd_change(:reg_id, :report_date_begin, :report_date_end, :report_date_new,
        :report_user_id);
  end
end
