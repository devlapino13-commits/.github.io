CREATE PROCEDURE HP_CHECK_CLIENT_FIO_FT(
    REP_C_ID D_ID,
    IS_CRITICAL_FIO D_SMALLINT)
RETURNS (
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    C_COMMENT D_STR_100)
AS
declare variable found_client_fio d_smallint;
declare variable found_birth_date d_smallint;
declare variable found_pass_no d_smallint;
declare variable found_inn_no d_smallint;
begin
  for select c_id, fp_name, max(found_client_fio), max(found_birth_date),
    max(found_pass_no), max(found_inn_no)
  from aarep_check_finpolis_tmp(:rep_c_id)
  group by c_id, fp_name
  into c_id, c_name_fio, found_client_fio, found_birth_date, found_pass_no, found_inn_no
  do
  begin
    c_comment = '';
    if ((found_birth_date + found_pass_no + found_inn_no > 0) or
        (is_critical_fio = 1 and found_client_fio = 1)) then
    begin
      if (found_client_fio = 1) then
        c_comment = 'РФМ (СФР): ФИО';
      if (found_pass_no = 1) then
        c_comment = iif(c_comment <> '', c_comment || ', ', '') || 'Паспорт';
      if (found_inn_no = 1) then
        c_comment = iif(c_comment <> '', c_comment || ', ', '') || 'ИНН';
      if (found_birth_date = 1) then
        c_comment = iif(c_comment <> '', c_comment || ', ', '') || 'Д.Р.';
      suspend;
    end
  end
end
