CREATE PROCEDURE FM_SET_CETRIFICATE_TIME(
    UN_ID DECIMAL(18,0),
    MA_ID INTEGER,
    SET_TIME TIMESTAMP)
RETURNS (
    RES_ID INTEGER,
    RES_TXT VARCHAR(255))
AS
declare variable reg_id integer;
declare variable manager_id decimal(18,0);
declare variable ma_plan_disb_time timestamp;
declare variable delta integer;
declare variable db_id integer;
begin
  delta = 15;

  :db_id = f_get_db_id();
  if (db_id = 81001) then
    delta = 10;
  res_id = 1;
  if (set_time is not null) then
  begin
    select kal_new_reg_id, ma_manager_id, ma_plan_disb_time
    from miniapp, knaktlist
    where ma_id = :ma_id and
          ma_date_oper = cast(:set_time as date) and
          ma_date_end >= '31.12.2999' and
          ma_is_active = 1 and
          ma_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_date <= current_date and
          kal_date_end >= current_date
    into :reg_id, :manager_id, :ma_plan_disb_time;

    if (ma_plan_disb_time = set_time) then
    begin
      res_txt = 'Время сертификации установлено успешно.';
      suspend;
      exit;
    end

    if (exists(select *
               from miniapp, knaktlist
               where ma_id <> :ma_id and
                     ma_date_oper = cast(:set_time as date) and
                     ma_date_end >= '31.12.2999' and
                     ma_is_active = 1 and
                     ma_from_kf_id = kal_from_kf_id and
                     knaktlist.is_delete = 0 and
                     kal_date <= current_date and
                     kal_date_end >= current_date and
                     kal_new_reg_id = :reg_id and
                     ma_plan_disb_time >= :set_time and
                     ma_plan_disb_time < (dateadd(minute, :delta, :set_time)))) then
    begin
      res_id = -1;
      res_txt = 'На выбранное время уже назначена сертификация.';
      suspend;
      exit;
    end
    execute procedure hp_update_miniapp_cert_time(:ma_id, :manager_id, :set_time, :un_id, null);
    res_txt = 'Время сертификации установлено успешно.';
  end
  else
  begin
    execute procedure hp_update_miniapp_cert_time(:ma_id, :manager_id, :set_time, :un_id, null);
    res_txt = 'Назначенное время сертификации успешно освобождено.';
  end
  suspend;
end
