CREATE PROCEDURE CM_KG_0118_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable c_id id_type;
declare variable pr_id d_integer;
declare variable pr_code d_str_100;
declare variable date_oper d_date;
declare variable same_product d_integer;
declare variable pr_id_tmp d_integer;
declare variable max_amount d_integer;
declare variable min_amount d_integer;
declare variable refund d_smallint;
begin
  res_error_is_critical = 1;
  for select mac_from_c_id,
             p.pr_code,
             m.ma_refund,
             mcl.mac_amount,
             cl.c_name_fio
      from miniappclient mcl, miniapp m, product p, clients cl
      where mac_from_ma_id = :ma_id and
            mac_amount > 0 and
            ma_id = mcl.mac_from_ma_id and
            ma_is_active = 1 and
            ma_date_end = '31.12.9999' and
            p.pr_id = m.ma_from_pr_id and
            cl.c_id = mcl.mac_from_c_id
      into :c_id,
           :pr_code,
           :refund,
           :kc_amount,
           :res_info_text
  do
  begin
    max_amount = 0;
    min_amount = 0;
    if ((pr_code containing '.GR.'))  then -- если это Групповой
        min_amount = iif(refund = 0, 10000, 2000);
    /*  igusev 16.03.2021 убрал блок т.к. реализация проверки макс по бирге Бирге на уровне блока 139 + вместо текущего блока добален блок ниже
    if ((pr_code containing '.BIRGE.') and (pr_code not containing '.EXPRESS.')) then -- если это Бирге Стандарт
    begin
        min_amount = iif(refund = 0, 15000, 2000);
        max_amount = iif(refund = 0, 150000, 160000);
    end
    */
    -- igusev 16.03.2021
    if ((pr_code containing '.BIRGE.') and (pr_code not containing '.EXPRESS.') and (refund = 1)) then -- если это Бирге Стандарт
    begin
        min_amount = 2000;
        max_amount = 160000;
    end
    if ((pr_code containing '.BIRGE.') and (pr_code containing '.EXPRESS.')) then -- если это Бирге Экспресс
        min_amount = iif(refund = 0, 5000, 2000);
    /* igusev 16.03.2021 убрал блок т.к. реализация проверки Бирге на уровне блока 139
    if ((pr_code containing '.BIRGE.') and (pr_code containing '.MUGALIM.')) then -- если это Бирге mugalim
        max_amount = 160000;
    */
    if ((pr_code containing '.FARMER.')) then -- если это Бизнесмен-Фермер
        min_amount = iif(refund = 0, 50000, 2000);
     if (kc_amount < min_amount) then
     begin
        res_error_text = 'Сумма клиента указана не верно. Допустимая минимальная сумма займа = '||min_amount;
        suspend;
     end
     if (max_amount <> 0) then
     begin
        if (kc_amount > max_amount) then
        begin
            res_error_text = 'Сумма клиента указана не верно. Допустимая максимальная сумма займа = '||max_amount;
            suspend;
        end
     end
  end
end
