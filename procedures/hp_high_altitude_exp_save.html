CREATE PROCEDURE HP_HIGH_ALTITUDE_EXP_SAVE(
    C_ID D_ID,
    EXP_YEARS D_INTEGER,
    EXP_MONTH D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable DATE_FOR_AI D_DATE;
declare variable COUNTER D_INTEGER;
declare variable DATE_BEGIN_TMP D_DATE;
declare variable DATE_END_TMP D_DATE;
declare variable IS_WORKING D_SMALLINT;
declare variable DATE_END D_DATE;
declare variable HA_KOEF D_PERCENT;
declare variable DELIMITER D_STR_10;
declare variable RES_MONTH D_INTEGER;
declare variable REG_ID D_INTEGER;
begin
  delimiter = ascii_char(13) || ascii_char(10);
  res_text = 'Выбранный сотрудник не подключен к системе начисления высокогорного стажа.' || delimiter || 'Причина: ';
  counter = 0;
  for select ps.ps_date_begin, ps.ps_date_end, ps.ps_is_working, ps.ps_from_reg_id
      from hr_pstatus ps
      where ps.ps_from_c_id = :c_id and
            ps.ps_status = 1
      into :date_begin_tmp, :date_end_tmp, :is_working, :reg_id

  do
  begin
    -- RFCKG-4782
    ha_koef = f_reg_mount_koef(:reg_id);
    if (:ha_koef > 1) then
    begin
      counter = counter + 1;

      if (counter = 1) then
        date_for_ai = :date_begin_tmp;
      else
      if (:date_begin_tmp - 1 <> :date_end) then
        date_for_ai = :date_begin_tmp;
      date_end = date_end_tmp;
    end
  end

  if (:is_working = 0) then
    res_text = res_text || 'не является действующим сотрудником на данный момент,' || delimiter;

  if (:date_for_ai is null) then
  begin
    res_text = res_text || 'не был закреплен за высокогорным офисом компании,' || delimiter;
    is_working = 0;
  end

  if (:ha_koef = 1) then
  begin
    res_text = res_text || 'не является сотрудником высокогорного офиса на данный момент,' || delimiter;
    is_working = 0;
  end

  res_text = trim(res_text);
  res_text = overlay(res_text placing '.' from (char_length(res_text) - 2));

  date_for_ai = date_for_ai - 1;
  res_month = exp_years * 12 + exp_month;

  if (exists(select *
             from high_altitude_experience hae
             where hae.hae_from_c_id = :c_id)) then
  begin
    update high_altitude_experience
    set hae_month = :res_month,
        hae_ai_is_active = :is_working,
        hae_datetime = current_timestamp,
        hae_from_un_id = :un_id
    where hae_from_c_id = :c_id;
    if (:is_working = 1) then
      res_text = 'Успешно сохранено.';
  end
  else
  begin
    insert into high_altitude_experience (hae_from_c_id, hae_month, hae_day_for_ai, hae_ai_is_active, hae_from_un_id)
    values (:c_id, :res_month, extract(day from :date_for_ai), :is_working, :un_id);

    if (:is_working = 1) then
      res_text = 'Успешно добавлено.' || delimiter || 'День начисления стажа: ' || extract(day from :date_for_ai);
  end
  suspend;
end
