CREATE PROCEDURE UXP$UX_LIST_MENU(
    ONLY_OWN D_SMALLINT = 0)
RETURNS (
    RES_ID D_STR_255,
    RES_PARENT_ID D_STR_255,
    RES_KIND D_STR_CODE,
    RES_IMAGE D_SMALLINT,
    RES_NAME D_STR_100,
    RES_FCODE D_STR_100,
    RES_MEMO D_STR_LONG,
    RES_KIND_CAPTION D_STR_CODE,
    RES_DATETIME D_DATETIME,
    RES_SOURCE_UUID D_UUID,
    RES_WARNING_STATE D_SMALLINT,
    RES_UUID D_UUID,
    RES_UUID_STR D_STR_CODE,
    RES_SCODE D_STR_CODE,
    RES_CAPTION D_STR_100,
    RES_SOURCE_CODE D_STR_CODE,
    RES_READONLY D_BOOL)
AS
declare variable ux_root d_str_code;
declare variable ua_uuid d_uuid;
declare variable mnu_is_folder d_boolean;
declare variable ux_kind d_str_code;
declare variable us_is_table d_bool;
declare variable us_user d_str_code;
declare variable ux_draft_user d_str_code;
declare variable user_login d_str_code;
declare variable ux_draft_uuid d_uuid;
begin
  if (only_own is null) then
    only_own = 0;

  res_warning_state = 0;
  res_readonly = 0;

  res_source_uuid = null;
  res_source_code = '';

  res_kind_caption = '';
  res_datetime = null;
  res_uuid = null;
  res_uuid_str = '';

  res_image = 79; -- зеленая кника
  res_kind = 'object.main';
  res_kind_caption = 'Гл.меню';
  res_parent_id = null;
  res_id = 'object.main';
  res_fcode = '';
  res_scode = '';
  res_name = 'Главное меню';
  res_memo = 'Навигация через главное меню приложения';
  res_caption = 'Главное меню';
  suspend;

  ux_root = res_id;
  user_login = uxf$user_login(uxf$user());

  for select mnu_id, mnu_parent_id, mnu_code, mnu_name, mnu_memo, mnu_ua_uuid, mnu_is_folder
  from uxp$mainmenu_tree(1, 0)
  into :res_id, :res_parent_id, :res_scode, :res_caption, :res_memo, :ua_uuid, mnu_is_folder
  do
  begin
    res_uuid = null;
    res_uuid_str = '';

    res_kind = 'menu';
    res_kind_caption = 'Меню';
    res_name = res_caption;
    res_fcode = res_scode;

    res_source_uuid = null;
    res_source_code = '';
    res_uuid_str = '';

    if (res_memo is null) then
      res_memo = '';

    if (mnu_is_folder) then
    begin
      res_kind = 'folder';
      res_kind_caption = 'Папка';
      res_image = 119; -- Желтая папка
    end else
    begin
      if (ua_uuid is null) then
        Continue;

      ux_kind = null;
      select ux_us_uuid, us_code, uxf$kind(uu_kind, 'code'), us_is_table,
             us_user, ux_draft_user, ux_draft_uuid
      from ux$action, ux$alias, ux$code, ux$uniq, ux$source
      where ua_uuid = :ua_uuid and
            ua_uxa_uuid = uxa_uuid and
            uxa_owner = ux_uuid and
            ux_uuid = uu_id and
            ux_us_uuid = us_uuid
      into res_source_uuid, res_source_code, ux_kind, us_is_table,
           us_user, ux_draft_user, ux_draft_uuid;

      if (user_login <> '' and only_own = 1 and us_user <> user_login and ux_draft_user <> user_login) then
        continue;

      if (ux_draft_user <> current_user) then
      begin
        if (res_memo <> '') then
          res_memo = res_memo || ascii_char(13);

        res_memo = res_memo || 'User: ' || ux_draft_user;
      end

      if (us_is_table = 1) then
        res_kind = 'table';
      else
        res_kind = 'procedure';

      res_uuid = res_source_uuid;

      res_uuid_str = '';
      if (ux_draft_uuid is not null) then
        res_uuid_str = uuid_to_char(ux_draft_uuid); -- в строковом формате передаем UUID формы

      res_image = case ux_kind
        when 'uxcommand' then 35
        when 'uxreport' then 26
        else 19
      end;
    end

    if (res_parent_id is null) then
      res_parent_id = ux_root;

    suspend;
  end
end
