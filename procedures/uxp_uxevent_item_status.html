CREATE PROCEDURE UXP$UXEVENT_ITEM_STATUS(
    ITEM_UUID D_UUID,
    EVENT_CONTROL D_STR_CODE,
    LIST_KIND D_STR_CODE)
RETURNS (
    EVN_STATUS D_SMALLINT)
AS
declare variable rec_count d_integer;
begin
  evn_status = -1; -- События не доступны
  
  if (list_kind = 'sender') then
  begin
    if (exists(select * from uxp$uxevent_links
               where evl_control = :event_control and
                     evl_event_code starting 'on')) then
                 evn_status = 0; -- События доступны
  
    select count(*)
    from ux$event
    where evt_sender_uuid = :item_uuid
    into rec_count;

    if (rec_count = 1) then
      evn_status = 1; -- Событие прикреплено
    else
      if (rec_count > 1) then
        evn_status = 2; -- События прикреплены

  end else
  begin
    if (exists(select * from uxp$uxevent_links
               where evl_control = :event_control and
                     evl_event_code not starting 'on')) then
                 evn_status = 0; -- Обработчики доступны
  
    select count(*)
    from ux$event
    where evt_target_uuid = :item_uuid
    into rec_count;

    if (rec_count = 1) then
      evn_status = 1; -- Обработчик прикреплен
    else
      if (rec_count > 1) then
        evn_status = 2; -- Обработчики прикреплены
  end

  suspend;
end
