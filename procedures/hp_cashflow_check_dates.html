CREATE PROCEDURE HP_CASHFLOW_CHECK_DATES(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    NEED_UPDATE D_BOOL = 0)
RETURNS (
    KOB_ID D_ID,
    KOB_OL_DATE D_DATE,
    KOB_FROM_OL_ID D_ID,
    KOB_CASHFLOW_DATE D_DATE,
    CASH_DATE D_STR_LONG,
    RES_STATUS D_STR_255)
AS
declare variable cash_acc_id d_id;
begin
  if (need_update is null) then
    need_update = 0;

  for select acc_id
  from ref_account, sys_accbalance
  where sab_full_code = 'ПлатСистемы.Погашено' and
        acc_from_sab_id = sab_id
  into cash_acc_id
  do
  begin
    for select kob_id, kob_from_ol_id, kob_ol_date, kob_cashflow_date
    from kn_operbook
    where kob_ol_date between :rep_date_begin and :rep_date_end and
          kob_ol_is_active = 1 and
          kob_acc_dt = :cash_acc_id
    plan (kn_operbook index (kn_operbook_acc_dt_kt, kn_operbook_date))
    into kob_id, kob_from_ol_id, kob_ol_date, kob_cashflow_date
    do
    begin
      cash_date = '';
      res_status = '';
      select list(distinct f_datetostr(kob_ol_date))
      from operperiod, kn_operbook, ref_account, sys_accbalance
      where :kob_ol_date between olp_date_begin and olp_date_end and
            kob_ol_is_active = 1 and
            olp_from_ol_id = kob_from_ol_id and
            kob_acc_kt + 0 = :cash_acc_id and
            kob_acc_dt = acc_id and
            acc_from_sab_id = sab_id and
            sab_full_code starting 'Деньги.Безнал.Банк.'
      into cash_date;
  
      if (cash_date not containing f_datetostr(kob_cashflow_date)) then
      begin
        if (need_update = 1) then
        begin
          update kn_operbook
          set kob_from_rml_id = -2
          where kob_id = :kob_id;

          res_status = 'Дата ДДС исправлена';
        end

        suspend;
      end
    end
  end
end
