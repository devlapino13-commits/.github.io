CREATE PROCEDURE AAREP_MAKE_DELINQ_OBOOK(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    POB_AMOUNT DECIMAL(18,4),
    POB_REGION_NAME VARCHAR(100),
    POB_REGION_ID INTEGER,
    POB_BRANCH_ID INTEGER,
    POB_AMOUNT_BALANCE DECIMAL(18,4),
    POB_AMOUNT_DIFF DECIMAL(18,4),
    POB_ACC_DT VARCHAR(20),
    POB_ACC_KT VARCHAR(20),
    POB_BOOK_MEMO VARCHAR(1024))
AS
declare variable acc_main varchar(20);
declare variable acc_delinq varchar(20);
declare variable k_id decimal(18,0);
declare variable tmp_amount decimal(18,4) = 0;
declare variable bg_id d_str_20;
declare variable db_id d_integer;
declare variable al_id d_str_20;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable oks_balance d_currency;
declare variable pr_id d_integer;
declare variable pob_acc_temp varchar(20);
begin
  -- SQL_TEXT должен выдавать поля в порядке:
  -- 1. Сумма в аналитической части
  -- 2. Название офиса
  -- 3. ID офиса, за которую формируются данная проводка
  -- 4. ID офиса где будет формироваться проводки (обычно это тот же самый офис)
  -- 5. Сумму в балансе (бух.учете)
  -- 6. Сумму разницы (аналитика - баланс). Это и есть сумма проводки, если сумма минусовая, система поменяет Дт и Кт счета местами
  -- 7. Дт счет
  -- 8. Кт счет
  -- 9. Примечание в проводке, краткое описание

  -- получаем группу счетов, где счет учета нормальных кредитов отличается от счета просроченных
  db_id = f_get_db_id();
  if (db_id = 94401) then
  begin

    for select b.bg_id
        from balgroup b
        where b.bg_id in ('21113', '21213', '21143', '21243')
        into :bg_id
    do
    begin
      for select distinct reg_unicode, reg_code || ' ' || reg_name
          from region, region_list(:report_region_id, 1, 3, 1)
          where reg_unicode = id_list and
                reg_status = 3

          into :pob_region_id, :pob_region_name
      do
      begin
        pob_amount = 0;
        pob_amount_balance = 0;
        pob_amount_diff = 0;
        pob_book_memo = 'Новые просрочки';

        -- находим просроченные кредиты, возникщие сегодня
        pob_acc_dt = acc_delinq;
        pob_acc_kt = acc_main;
        pob_branch_id = pob_region_id;

        acc_dt = bg_id;
        if (acc_dt = '21113') then
          acc_kt = '21110';
        else
        if (acc_dt = '21143') then
          acc_kt = '21140';
        else
        if (acc_dt = '21213') then
          acc_kt = '21210';
        else
        if (acc_dt = '21243') then
          acc_kt = '21240';

        pob_acc_dt = '';
        pob_acc_kt = '';

        select acl.al_id
        from acc_list acl
        where acl.al_from_reg_id = :pob_region_id and
              acl.al_from_bg_id = :acc_dt
        into :pob_acc_dt;

        select acl.al_id
        from acc_list acl
        where acl.al_from_reg_id = :pob_region_id and
              acl.al_from_bg_id = :acc_kt
        into :pob_acc_kt;

        pob_amount_balance = 0;
        select coalesce(sum(summa_val), 0)
        from aarep_account_saldo_temp(:pob_region_id, :report_date_end), acc_list, balgroup b1, balgroup b2
        where account = al_id and
              al_from_bg_id = b1.bg_id and
              b1.parent = b2.bg_id and
              ((currency = '000') or
              ('000' = '---') or
              ('000' = '000')) and
              account = :pob_acc_dt
        group by account, al_name, al_number, al_from_bg_id, b1.bg_name, b2.bg_name, b2.parent, currency, b2.bg_id, b2.bg_kind, al_type
        having sum(summa_val) <> 0 or sum(summa_ekv) <> 0
        into :pob_amount_balance;

        pob_amount = 0;
        for select coalesce(oks.oks_balance, 0), k.k_from_pr_id
            from oksummary oks
            inner join kontract k on k.k_id = oks.oks_from_k_id
            where :report_date_end between oks.oks_date_begin and oks.oks_date_end and
                  oks.oks_close_state = 0 and
                  oks.oks_region_id = :pob_region_id and
                  oks_delinq_date < :report_date_end
            into :oks_balance, :pr_id
        do
        begin
          if (exists(select distinct coalesce(pal_main_acc, 'не указан')
                     from product, prodacctype, prodacclist
                     where pr_from_pat_id = pat_id and
                           pal_from_pat_id = pat_id and
                           pal_from_at_id = 1 and
                           pr_id = :pr_id and
                           pal_main_acc = :acc_kt)) then
          begin
            pob_amount = pob_amount + oks_balance;
          end

        end

        pob_amount_diff = pob_amount - pob_amount_balance;

        if (pob_amount_diff < 0) then
        begin
          pob_amount_diff = -pob_amount_diff;
          pob_acc_temp = pob_acc_dt;
          pob_acc_dt = pob_acc_kt;
          pob_acc_kt = pob_acc_temp;
        end

        if (pob_amount_diff <> 0) then
        begin
          suspend;
        end
      end
    end
  end
  else
  begin
    for select distinct reg_unicode, reg_code || ' ' || reg_name,
                        reg_code || pal_currency || pal_main_acc || right('000000' || pal_main_acc_no, 6),
                        reg_code || pal_currency || pal_delinq_acc || right('000000' || pal_delinq_acc_no, 6)
        from prodacclist, region, region_list(:report_region_id, 1, 3, 1)
        where coalesce(pal_main_acc, '') <> '' and
              coalesce(pal_main_acc_no, -1) <> -1 and
              coalesce(pal_delinq_acc, '') <> '' and
              coalesce(pal_delinq_acc_no, -1) <> -1 and
              reg_unicode = id_list and
              reg_status = 3 and
              reg_code || pal_currency || pal_main_acc || right('000000' || pal_main_acc_no, 6) <> reg_code || pal_currency || pal_delinq_acc || right('000000' || pal_delinq_acc_no, 6)
        into :pob_region_id, :pob_region_name, :acc_main, :acc_delinq
    do
    begin
      pob_amount = 0;
      pob_amount_balance = 0;
      pob_amount_diff = 0;
      pob_book_memo = 'Новые просрочки';

      -- находим просроченные кредиты, возникщие сегодня
      pob_acc_dt = acc_delinq;
      pob_acc_kt = acc_main;
      pob_branch_id = pob_region_id;
      for select oks_from_k_id
          from oksummary, kontract, product, prodacclist
          where oks_from_k_id = k_id and
                oks_close_state = 0 and
                k_from_pr_id = pr_id and
                oks_date_begin <= :report_date_end and
                oks_date_end >= :report_date_end and
                oks_delinq_date = :report_date_end and
                oks_balance > 0 and
                oks_region_id = :pob_region_id and
                pr_from_pat_id = pal_from_pat_id and
                pal_from_at_id = 1 and
                coalesce(pal_main_acc, '') <> '' and
                coalesce(pal_main_acc_no, -1) <> -1 and
                coalesce(pal_delinq_acc, '') <> '' and
                coalesce(pal_delinq_acc_no, -1) <> -1 and
                pal_currency || pal_main_acc || right('000000' || pal_main_acc_no, 6) <> pal_currency || pal_delinq_acc || right('000000' || pal_delinq_acc_no, 6) and
                pal_currency || pal_main_acc || right('000000' || pal_main_acc_no, 6) = right(:acc_main, 14) and
                pal_currency || pal_delinq_acc || right('000000' || pal_delinq_acc_no, 6) = right(:acc_delinq, 14)
          into :k_id
      do
      begin
        -- по этим кредитам вычисляем просроченную основную сумму, возникщая в день отчета
        select sum(ok_os + ok_sh_os)
        from operkred
        where ok_ol_is_active = 1 and
              ok_ol_date <= :report_date_end and
              ok_type <= 6 and
              ok_from_k_id = :k_id and
              ok_sh_date = :report_date_end and
              ok_from_c_id = 0 having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 0
        into :tmp_amount;
        if (tmp_amount is null) then
          tmp_amount = 0;
        pob_amount = pob_amount + :tmp_amount;
      end
      tmp_amount = 0;
      select sum(ob_summa_val)
      from operbook
      where ob_ol_date = :report_date_end and
            ob_ol_reg_id = :pob_region_id and
            ob_account_dt = :pob_acc_dt and
            ob_account_kt = :pob_acc_kt and
            ob_ol_is_active = 1
      into :tmp_amount;
      if (tmp_amount is null) then
        tmp_amount = 0;
      pob_amount_balance = pob_amount_balance + tmp_amount;

      select -sum(ob_summa_val)
      from operbook
      where ob_ol_date = :report_date_end and
            ob_ol_reg_id = :pob_region_id and
            ob_account_kt = :pob_acc_dt and
            ob_account_dt = :pob_acc_kt and
            ob_ol_is_active = 1
      into :tmp_amount;
      if (tmp_amount is null) then
        tmp_amount = 0;
      pob_amount_balance = pob_amount_balance + tmp_amount;

      pob_amount_diff = pob_amount - pob_amount_balance;
      if (pob_amount_diff <> 0) then
      begin
        suspend;
      end
    end
  end
end
