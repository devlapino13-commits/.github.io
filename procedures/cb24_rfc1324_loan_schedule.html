CREATE PROCEDURE CB24_RFC1324_LOAN_SCHEDULE(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    LOAN_ID D_ID,
    LOAN_UID D_STR_CODE,
    PAYMENT_PLAN_SUM D_CURRENCY,
    PAYMENT_PLAN_SUM_PRINCIPAL D_CURRENCY,
    PAYMENT_PLAN_SUM_INTEREST D_CURRENCY,
    PAYMENT_PLAN_DATE D_DATE,
    SUBJECT_TYPE D_STR_20,
    SUBJECT_ID D_ID,
    SUBJECT_PRIVATE_FULLNAME D_STR_100,
    SUBJECT_PRIVATE_BIRTH_DATE D_DATE,
    SUBJECT_PRIVATE_COUNTRY_CODE D_STR_20,
    SUBJECT_PRIVATE_DOC_SERIAL D_STR_CODE,
    SUBJECT_PRIVATE_DOC_NUMBER D_STR_CODE,
    SUBJECT_LEGAL_FULLNAME D_STR_255,
    SUBJECT_LEGAL_INN D_STR_20,
    SUBJECT_LEGAL_OGRN D_STR_20,
    RES_KONTRACT_NO D_STR_20,
    RES_DISB_DATE D_DATE,
    RES_SH_ID D_INTEGER,
    RES_DOC_ID D_ID,
    RES_PK_NAME D_STR_100)
AS
declare variable kc_disb_passport_id d_integer;
declare variable pk_id d_integer;
declare variable cds_doc_no d_str_code;
declare variable k_id d_id;
begin
  subject_type = 'ФЛ';
  subject_legal_fullname = '';
  subject_legal_inn = '';
  subject_legal_ogrn = '';

  for select ks_from_k_id, ks_date_begin, k_kontract_no
  from kontstate, kontract, region_list(:rep_region_id, 1, 3, 1)
  where ks_state = 1 and
        ks_ol_is_delete = 0 and
        ks_date_begin between :rep_date_begin and :rep_date_end and
        ks_from_k_id = k_id and
        ks_region_id = id_list
  into k_id, res_disb_date, res_kontract_no
  do
  begin
    for select kc_id, kc_disb_passport_id, kc_from_c_id, c_name_fio, c_birth_date
    from knclient, clients
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    into loan_id, kc_disb_passport_id, subject_id, subject_private_fullname, subject_private_birth_date
    do
    begin
      loan_uid = f_uuid_cbr(f_uuid('knclient', loan_id));

      res_doc_id = null;
      subject_private_doc_serial = '';
      subject_private_doc_number = '';
      subject_private_country_code = '';
      res_pk_name = '';

      if (kc_disb_passport_id is not null) then
      begin
        res_doc_id = kc_disb_passport_id;

        select cds_doc_no, pk_id, pk_name, left(pk_country, 2)
        from clientdocs, passkind
        where cds_id = :res_doc_id and
              cds_from_pk_id = pk_id
        into cds_doc_no, pk_id, res_pk_name, subject_private_country_code;

        select res_doc_seria, res_doc_number
        from hp_passport_serial_num(:cds_doc_no, :pk_id)
        into subject_private_doc_serial, subject_private_doc_number;
      end

      -- Базовый график погашений
      for select sh_date, sh_id, sh_os, sh_perc
      from hp_get_base_schedule(:k_id)
      where sh_id > 0
      order by sh_date
      into payment_plan_date, res_sh_id, payment_plan_sum_principal, payment_plan_sum_interest
      do
      begin
        payment_plan_sum = payment_plan_sum_principal + payment_plan_sum_interest;
        suspend;
      end

      -- Актуальный после выдачи
      /*
      for select sh_date, sh_id, sh_os, sh_perc
      from schedule
      where sh_from_k_id = :k_id and
            sh_id > 0
      order by sh_date
      into payment_plan_date, res_sh_id, payment_plan_sum_principal, payment_plan_sum_interest
      do
      begin
        payment_plan_sum = payment_plan_sum_principal + payment_plan_sum_interest;
        suspend;
      end
      */
    end
  end
end
