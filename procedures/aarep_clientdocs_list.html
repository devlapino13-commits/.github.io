CREATE PROCEDURE AAREP_CLIENTDOCS_LIST(
    C_ID DECIMAL(18,0))
RETURNS (
    CDS_FROM_C_ID D_ID,
    CDS_ID D_INTEGER,
    CDS_FROM_PK_ID D_INTEGER,
    CDS_DOC_NO VARCHAR(30),
    CDS_DOC_DATE_BEGIN D_DATE,
    CDS_DOC_DATE_END D_DATE,
    CDS_DOC_ISSUED_BY D_STR_255,
    CDS_DOC_ISSUED_CODE D_STR_CODE,
    CDS_DOC_FOR_COMPANY D_STR_255,
    CDS_DOC_TERRITORY D_STR_255,
    CDS_DOC_BLOB_1 D_ID,
    CDS_DOC_BLOB_2 D_ID,
    CDS_FROM_UN_ID D_ID,
    CDS_DATETIME D_DATETIME,
    CDS_IS_DELETE D_INTEGER,
    CDS_VERIFY_STATUS D_INTEGER,
    CDS_VERIFY_DATETIME D_DATETIME,
    CDS_VERIFY_UN_ID D_ID,
    CDS_VERIFY_CC_DATETIME D_DATETIME,
    CDS_VERIFY_CC_UN_ID D_ID,
    CDS_MEMO_1 D_STR_255,
    CDS_MEMO_2 D_STR_255,
    CDS_MEMO_3 D_STR_255,
    CDS_MEMO_4 D_STR_255,
    CDS_MEMO_5 D_STR_255,
    PDT_IS_MAIN_DOC D_INTEGER,
    CDS_IS_OLD_DOC D_INTEGER,
    PK_NAME D_STR_255,
    CDS_IS_MAIN_DOC D_INTEGER,
    CDS_FROM_DRL_ID D_ID,
    EDIT_KIND D_INTEGER,
    PDT_NEED_DATE_BEGIN D_INTEGER,
    PDT_NEED_DATE_END D_INTEGER,
    CDS_CREATE_DATETIME D_DATETIME,
    BLOB_ADD_KIND D_STR_100,
    SHOW_IN_KK D_SMALLINT,
    VERIFY_CC_STATUS D_BOOL,
    UN_NAME D_STR_CODE,
    C_NAME_FIO D_STR_100)
AS
declare variable pdt_code d_str_100;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  for select cds_from_c_id, cds_id, cds_from_pk_id, cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_doc_issued_by, cds_doc_issued_code,
             cds_doc_for_company, cds_doc_territory, cds_doc_blob_1, cds_doc_blob_2, cds_from_un_id, cds_datetime,
             cds_is_delete, cds_verify_status, cds_verify_datetime, cds_verify_un_id,
             cds_memo_1, cds_memo_2, cds_memo_3, cds_memo_4, cds_memo_5, pk_name, pdt_is_main_doc, cds_is_main_doc,
             cds_from_drl_id, pdt_code, pdt_need_date_begin, pdt_need_date_end,
             cdk_datetime, cdk_un_id, cds_create_datetime, iif(coalesce(cdk_un_id, 0) > 0, 1, 0), un_name
      from clientdocs
        inner join passkind
          on pk_id = cds_from_pk_id
        inner join passdoctype
          on pdt_id = pk_from_pdt_id
        left join clientdocs_kk_verify
          on cdk_cds_id = cds_id
        left join username
          on un_id = cds_from_un_id
      where cds_from_c_id = :c_id
      into cds_from_c_id, cds_id, cds_from_pk_id, cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_doc_issued_by, cds_doc_issued_code,
           cds_doc_for_company, cds_doc_territory, cds_doc_blob_1, cds_doc_blob_2, cds_from_un_id, cds_datetime,
           cds_is_delete, cds_verify_status, cds_verify_datetime, cds_verify_un_id,
           cds_memo_1, cds_memo_2, cds_memo_3, cds_memo_4, cds_memo_5, pk_name, pdt_is_main_doc, cds_is_main_doc,
           cds_from_drl_id, pdt_code, pdt_need_date_begin, pdt_need_date_end,
           cds_verify_cc_datetime, cds_verify_cc_un_id, cds_create_datetime, verify_cc_status, :un_name
  do
  begin
    show_in_kk = 1;
    -- Исключаем кадровые документы (Страховой полис, Уведомление о приеме ИГ, Уведомление об увольнении ИГ)
    if (db_id = 81001 and pdt_code in ('Insurance_ID', 'notification', 'notification_2')) then
      show_in_kk = 0;

    edit_kind = 0;
    cds_is_old_doc = 0;
    if (cds_doc_date_end < current_date or cds_is_delete = 1) then
      cds_is_old_doc = 1;
    if (f_get_const('new_clientdocs') = 1 and cds_from_drl_id is not null) then
      edit_kind = 1;
    else if (pdt_code = 'transport' and
        db_id = 81001 and
        cds_from_drl_id is not null) then
      edit_kind = 1;

    if (cds_doc_blob_1 is not null) then
    begin
      select f_blob_kind(bls_source_system, bls_kind)
      from blobstore
      where bls_id = :cds_doc_blob_1
      into blob_add_kind;
    end else
    if (cds_doc_blob_2 is not null) then
    begin
      select f_blob_kind(bls_source_system, bls_kind)
      from blobstore
      where bls_id = :cds_doc_blob_2
      into blob_add_kind;
    end

    if (pdt_code = 'certificate_ip' and db_id in (39801, 41701))
      then cds_memo_1 = f_docrecog_value(:cds_from_drl_id, 'memo1');
    c_name_fio = (select c_name_fio from clients where c_id = :c_id);
    suspend;
  end
end
