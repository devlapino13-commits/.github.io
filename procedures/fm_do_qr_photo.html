CREATE PROCEDURE FM_DO_QR_PHOTO(
    C_ID D_ID,
    GPS_LAT D_DOUBLE,
    GPS_LONG D_DOUBLE,
    REASON D_INTEGER,
    UN_ID D_ID)
RETURNS (
    QRL_ID D_ID,
    ERR_TEXT D_STR_255)
AS
declare variable UN_ANDROID_ID varchar(50); /* Уникальный код аппарата пользователя */
declare variable DB_ID D_INTEGER;
begin
  -- Процедура генерирует запрос для подтвреждения фото как альтернатива QR-подтверждения
  db_id = f_get_db_id();
  err_text = '';

  if (coalesce(gps_lat, 0) = 0 or
      coalesce(gps_long, 0) = 0) then
  begin
    err_text = 'Не удалось сгенерировать qps-данные';
    suspend;
    exit;
  end

  if (c_id is null) then
    exception e_default 'Неверный вызов'; -- данная процедура должна вызываться только из МП клиента

  if (db_id = 81001) then
  begin
    if (exists(select *
               from hp_check_oivo_assignment(:c_id, :un_id)
               where is_oivo = 1 and
                     res_id <> 1)) then
    begin
      err_text = 'Верификация не прошла, так как клиент закреплен не за Вами!';
      suspend;
      exit;
    end
  end

  select un_android_id
  from username
  where un_id = :un_id
  into un_android_id;

  -- Аннулируем другие если еще не подтверждены
  update qr_list
  set qrl_is_delete = 1
  where qrl_from_c_id = :c_id and
        qrl_state = 0 and
        qrl_is_delete = 0;

  insert into qr_list (qrl_from_c_id, qrl_gps_lat, qrl_gps_long, qrl_phone_id)
  values (:c_id, :gps_lat, :gps_long, :un_android_id)
  returning qrl_id
  into :qrl_id;

  insert into qr_photo (qrf_from_qrl_id, qrf_reason, qrf_from_un_id)
  values (:qrl_id, :reason, :un_id);

  suspend;
end
