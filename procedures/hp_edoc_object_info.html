CREATE PROCEDURE HP_EDOC_OBJECT_INFO(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID,
    REP_DATE D_DATE = null)
RETURNS (
    BLANK_DATE D_DATE,
    PR_ID D_INTEGER,
    DOC_NAME D_STR_255,
    REG_ID D_INTEGER)
AS
declare variable kf_id d_id;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable obj_kind_tmp d_integer;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  -- Процедура возвращает дату электронного документа, продукт, регион и номер документа
  obj_kind = f_objkind(obj_kind, 'name');
  doc_name = '';

  if (obj_kind = 'Kontract') then
  begin
    select k_kontract_date, k_from_kf_id, k_from_pr_id, k_kontract_no || ' от ' || f_datetostr(k_kontract_date)
    from kontract
    where k_id = :obj_id
    into blank_date, kf_id, pr_id, doc_name;
  end

  if (obj_kind = 'knclient') then
  begin
    /*
    select k_kontract_date
    from kontract kont
    inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
    where kc_id = :obj_id
    into :blank_date;
    */

    blank_date = current_date;
  end

  if (obj_kind = 'Miniapp') then
  begin
    select ma_date_app, ma_from_kf_id, ma_from_pr_id, ma_no || ' от ' || f_datetostr(ma_date_app)
    from miniapp
    where ma_id = :obj_id
    order by ma_id desc, ma_number desc
    rows 1
    into blank_date, kf_id, pr_id, doc_name;
  end

  if (obj_kind = 'Agreements') then
  begin
    if (:db_id in (39801, 41701)) then
    begin
      blank_date = current_date;
    end
    else
    begin
      select ag_agreement_date, coalesce(ag_from_init_reg_id, ag_from_reg_id),
             ag_agreement_no || ' от ' || f_datetostr(ag_agreement_date)
      from agreements
      where ag_id = :obj_id
      into blank_date, reg_id, doc_name;
    end
  end
  if (obj_kind = 'kontmatag') then
  begin
    blank_date = current_date;
  end
  if (obj_kind = 'kontract_istinaa_prepay') then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'RECONCILIATION_INFO') then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'kontrefund_disb') then
  begin
    select ol_date, ol_from_reg_id, 'РКО № ' || coalesce(ol_doc_no_str, '') || ' от ' || f_datetostr(ol_date)
    from kontrefund_disb, operlist
    where krd_from_ol_id = ol_id and
          krd_id = :obj_id
    into blank_date, reg_id, doc_name;
  end

  if (obj_kind = 'com_list') then
  begin
    select cml_date, cml_from_reg_id, cml_id || ' от ' || f_datetostr(cml_date), cml_object_kind, cml_object_id
    from com_list
    where cml_id = :obj_id
    into blank_date, reg_id, doc_name, obj_kind_tmp, k_id;

    if (f_objkind(obj_kind_tmp, 'name') = 'kontract') then
    begin
      select k_from_pr_id
      from kontract
      where k_id = :k_id
      into pr_id;
    end
  end

  if (obj_kind = 'kontpsk') then
  begin
    blank_date = current_date;
    reg_id = 1000000;
  end

  if (obj_kind = 'ft_anketa_list') then
  begin
    select fal_date
    from ft_anketa_list
    where fal_id = :obj_id
    into blank_date;
  end

  if (obj_kind = 'cardaktclient') then
  begin
    select cac_date, cac_from_reg_id, cac_id || ' от ' || f_datetostr(cac_date)
    from cardaktclient
    where cac_id = :obj_id
    into :blank_date, :reg_id, :doc_name;
  end

  if (obj_kind = 'overpayment') then
  begin
    select ovp_date, ovp_id || ' от ' || f_datetostr(ovp_date)
    from overpayment
    where ovp_id = :obj_id
    into :blank_date, :doc_name;
  end

  if (obj_kind = 'cardblockopers' and
      :db_id in (39801, 41701)) then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'Clients') then
  begin
    blank_date = current_date;

    select c_name_fio
    from clients
    where c_id = :obj_id
    into doc_name;
  end

  if (obj_kind = 'expressapp') then
  begin
    blank_date = current_date;

    select c_name_fio, exa_from_pr_id
    from expressapp, clients
    where exa_id = :obj_id and
          exa_from_c_id = c_id
    into doc_name, pr_id;

    doc_name = 'Отправлено эл.предложение:' || coalesce(doc_name, '');
  end

  if (obj_kind = 'finerefund') then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'app_return_consult') then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'cardpinchange') then
  begin
    blank_date = current_date;
  end

  if (obj_kind in ('freelancer_kontract', 'probationer_kontract', 'kochegar_kontract', 'add_for_freelancers', 'termination_freelancer')) then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'freelancer_akt') then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'mc_purchase') then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'hr_orders') then
  begin
    select ho_date, ho_from_reg_id,
           trim(iif(coalesce(ho_order_no, '') = '', 'б/н', ho_order_no)) || ' от ' || f_datetostr(ho_date)
    from hr_orders
    where ho_id = :obj_id
    into blank_date, reg_id, doc_name;
  end

  if (obj_kind = 'servicedog') then
  begin
    select sd_date_begin, sd_from_reg_id,
           trim(iif(coalesce(sd_dogovor_no, '') = '', 'б/н', sd_dogovor_no)) || ' от ' || f_datetostr(sd_date_begin)
    from servicedog
    where sd_id = :obj_id
    into blank_date, reg_id, doc_name;
  end

  if (obj_kind = 'opercash') then
  begin
    select opc_date, opc_office_id, opc_doc_no || ' от ' || f_datetostr(opc_date)
    from opercash
    where opc_id = :obj_id
    into :blank_date, reg_id, :doc_name;
  end

  if (obj_kind = 'transfertmc') then
  begin
    select trt_date, trt_number || ' от ' || f_datetostr(trt_date)
    from transfertmc
    where trt_id = abs(:obj_id)
    into :blank_date, :doc_name;
  end

  if (obj_kind = 'fuelgenreg') then
  begin
    select fg_dtwriteoff, fg_id || ' от ' || f_datetostr(fg_dtwriteoff)
    from fuelgenreg
    where fg_id = abs(:obj_id)
    into :blank_date, :doc_name;
  end

  if (obj_kind = 'accesscashroom') then
  begin
    select acr_date, acr_id || ' от ' || f_datetostr(acr_date)
    from accesscashroom
    where acr_id = :obj_id
    into :blank_date, :doc_name;
  end

  if (obj_kind = 'cashcheckmain') then
  begin
    select ccm_date, ccm_id || ' от ' || f_datetostr(ccm_date)
    from cashcheckmain
    where ccm_id = :obj_id
    into :blank_date, :doc_name;
  end

  if (obj_kind = 'signdoc') then
  begin
    select sgn_datetime, sgn_id || ' от ' || f_datetostr(sgn_datetime)
    from signdoc
    where sgn_id = :obj_id
    into :blank_date, :doc_name;
  end

  if (obj_kind = 'info_loan_client') then
  begin
    blank_date = current_date;
    reg_id = 1000000;
  end

  if (obj_kind = 'invmember') then
  begin
    select im.im_date
    from invmember imem
    inner join invmain im on im.im_id = imem.imem_from_im_id
    where imem.imem_id = :obj_id
    into blank_date;
  end 

  if (obj_kind = 'cashbookmain') then
  begin
    select cbm_date, cbm_office_id
    from cashbookmain
    where cbm_id = :obj_id
    into blank_date, reg_id;
  end 

  if (obj_kind = 'sud_mainreq') then
  begin
    select k_from_pr_id, current_date
    from sud_mainreq, kontract
    where k_id = smr_from_k_id
      and smr_id = :obj_id
    into pr_id, blank_date;
  end

  if (obj_kind = 'overpay_main') then
  begin
    blank_date = current_date;
  end

  if (obj_kind = 'smevpass_msg') then
  begin
    blank_date = current_date;
  end

  if (rep_date is null) then
    rep_date = blank_date;

  if (kf_id is not null) then
  begin
    -- если это кредитный документ, берем регион из кредитной папки
    select k.kal_new_reg_id
    from knaktlist k
    where k.kal_from_kf_id = :kf_id and
          k.is_delete = 0 and
          :rep_date between k.kal_date and k.kal_date_end rows 1
    into reg_id;

    if (reg_id is null) then
    begin
      select k.kal_new_reg_id
      from knaktlist k
      where k.kal_from_kf_id = :kf_id and
            k.is_delete = 0 and
            k.kal_date_end >= '31.12.2999' rows 1
      into reg_id;
    end
  end

  if (reg_id is null) then
  begin
    -- если регион не найден, пробуем получить из пользователя
    select un_from_c_id
    from username
    where un_id = :un_id
    into c_id;

    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :c_id and
          ps_date_begin <= :rep_date and
          ps_date_end >= :rep_date and
          ps_is_working = 1 and
          ps_status = 1
    into reg_id;

    if (reg_id is null) then
      reg_id = f_hq_id();
  end

  -- Если по указанным выше кодам дата не определена, завершаем работу с ошибкой
  -- Данная реализация помогает когда появляется новый тип объекта для электронного документа,
  -- а программист не учел его выше
  if (blank_date is null) then
    exception e_default 'Blank date is undefined for ' || obj_kind;

  suspend;
end
