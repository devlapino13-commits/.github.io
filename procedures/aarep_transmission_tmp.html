CREATE PROCEDURE AAREP_TRANSMISSION_TMP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_PRODUCT_TYPE VARCHAR(255),
    REP_PRODUCT_KIND VARCHAR(255),
    REP_PRODUCT_NAME VARCHAR(255),
    REP_REGION_NAME VARCHAR(255),
    PAR_STATUS_AT_BEGIN VARCHAR(100),
    PAR_STATUS_AT_END VARCHAR(100),
    REP_AMOUNT CURR_TYPE,
    REP_DATE DATE)
AS
declare variable rep_date_dd date;
declare variable k_id id_type;
begin
  rep_amount = 0;
  rep_date_dd = :rep_date_begin;
  while (rep_date_dd < :rep_date_end) do
  begin
    rep_date_begin = rep_date_dd;
    rep_date_dd = dateadd(month, 1, rep_date_dd);
    rep_date = rep_date_dd - 1;

    rep_amount = 0;
    for select oks_from_k_id, reg_code || ' ' || reg_name as rep_region_name, pr_common_name as rep_product_kind,
               pr_name as rep_product_name, iif(pr_loan_type = 0, 'Group loans', 'Indiv loans') as rep_product_type,
               oks_balance,
               case
                 when (:rep_date_begin - oks_delinq_date) <= 0 then 'A. Non_PAR'
                 when (:rep_date_begin - oks_delinq_date) between 1 and 30 then 'B. PAR 1-30'
                 when (:rep_date_begin - oks_delinq_date) between 31 and 60 then 'C. PAR 31-60'
                 when (:rep_date_begin - oks_delinq_date) between 61 and 90 then 'D. PAR 61-90'
                 when (:rep_date_begin - oks_delinq_date) between 91 and 120 then 'E. PAR 91-120'
                 when (:rep_date_begin - oks_delinq_date) between 121 and 150 then 'F. PAR 121-150'
                 when (:rep_date_begin - oks_delinq_date) between 151 and 180 then 'G. PAR 151-180'
                 when (:rep_date_begin - oks_delinq_date) > 180 then 'H.PAR 180+'
               end
        from oksummary, kontract, product, region
        where oks_from_k_id = k_id and
              k_from_pr_id = pr_id and
              oks_region_id = reg_unicode and
              oks_date_begin <= :rep_date_begin and
              oks_date_end >= :rep_date_begin and
              oks_balance > 0 and
              oks_close_state = 0 and
              k_disb_date >= '01.01.2012'
        into :k_id, :rep_region_name, rep_product_kind, rep_product_name, rep_product_type, rep_amount,
             par_status_at_begin
    do
    begin
      par_status_at_end = '0. Outstanding at begin';
      suspend;
      rep_amount = 0;
      par_status_at_end = '';
      select oks_balance,
             (
             case
               when (:rep_date - oks_delinq_date) <= 0 then '1. Non_PAR'
               when (:rep_date - oks_delinq_date) between 1 and 30 then '2. PAR 1-30'
               when (:rep_date - oks_delinq_date) between 31 and 60 then '3. PAR 31-60'
               when (:rep_date - oks_delinq_date) between 61 and 90 then '4. PAR 61-90'
               when (:rep_date - oks_delinq_date) between 91 and 120 then '5. PAR 91-120'
               when (:rep_date - oks_delinq_date) between 121 and 150 then '6. PAR 121-150'
               when (:rep_date - oks_delinq_date) between 151 and 180 then '7. PAR 151-180'
               when (:rep_date - oks_delinq_date) > 180 then '8.PAR 180+'
             end)
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :rep_date and
            oks_date_end >= :rep_date and
            oks_balance > 0 and
            oks_close_state = 0
      into :rep_amount, par_status_at_end;
      if (par_status_at_end = '') then
        par_status_at_end = '1. Non_PAR';
      suspend;

      rep_amount = 0;
      select -sum(ok_os)
      from operkred
      where ok_from_k_id = :k_id and
            ok_type = 5 and
            ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date
      into :rep_amount;
      par_status_at_end = '9. Write-off';
      if (rep_amount <> 0) then
        suspend;

      rep_amount = 0;
      select -sum(ok_os)
      from operkred
      where ok_from_k_id = :k_id and
            ok_type = 4 and
            ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date
      into :rep_amount;
      par_status_at_end = '10. Repayment';
      if (rep_amount <> 0) then
        suspend;

    end
  end
end
