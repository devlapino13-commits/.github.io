CREATE PROCEDURE HP_DO_KN_OPERLOAN_FILL(
    OL_DATE D_DATE,
    ONE_OL_ID D_ID)
AS
declare variable al_id d_str_code;
declare variable ol_id d_id;
declare variable acc_kind d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_parent_id d_id;
begin
  if (one_ol_id is not null) then
  begin
    select ol_date
    from operlist
    where ol_id = :one_ol_id
    into ol_date;
  end

  for select ol_id, ol_parent_id, ol_oper_subtype
      from operlist
      left join operloan on (ol_id = opl_from_ol_id)
      where ol_oper_type = 1 and
            ol_date = :ol_date and
            opl_from_acc_id is null and
            (:one_ol_id is null or ol_id = :one_ol_id)
      into ol_id, ol_parent_id, ol_oper_subtype
  do
  begin
    al_id = null;
    acc_kind = null;

    if (ol_parent_id is not null and
        ol_oper_subtype = 1) then
    begin
      select
             case
               when ob_account_kt containing '50-01' then 1
               when ob_account_dt containing '50-01' then 1
               when ob_account_kt containing '51-01' then 2
               when ob_account_dt containing '51-01' then 2
               when ob_account_kt containing '76-10' then 3
               when ob_account_dt containing '76-10' then 3
               when ob_account_dt containing '70-01' then 3
               else 4
             end,
             case
               when ob_account_kt containing '50-01' then ob_account_kt
               when ob_account_dt containing '50-01' then ob_account_dt
               when ob_account_kt containing '51-01' then ob_account_kt
               when ob_account_dt containing '51-01' then ob_account_dt
               when ob_account_kt containing '76-10' then ob_account_kt
               when ob_account_dt containing '76-10' then ob_account_dt
               when ob_account_dt containing '70-01' then ob_account_dt
               else iif(:ol_oper_subtype in (1, 10), ob_account_kt, ob_account_dt)
             end
      from operbook
      where ob_from_ol_id = :ol_id
      order by 1
      rows 1
      into :acc_kind, :al_id;

      if (acc_kind = 4) then
        acc_kind = 0;
      update or insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id)
      values (:ol_id, :acc_kind, :al_id)
      matching (opl_from_ol_id);
    end

    al_id = null;
    acc_kind = null;

    if (ol_parent_id is not null and
        ol_oper_subtype <> 1) then
      ol_id = ol_parent_id;

    select
           case
             when ob_account_kt containing '50-01' then 1
             when ob_account_dt containing '50-01' then 1
             when ob_account_kt containing '51-01' then 2
             when ob_account_dt containing '51-01' then 2
             when ob_account_kt containing '76-10' then 3
             when ob_account_dt containing '76-10' then 3
             when ob_account_dt containing '70-01' then 3
             else 4
           end,
           case
             when ob_account_kt containing '50-01' then ob_account_kt
             when ob_account_dt containing '50-01' then ob_account_dt
             when ob_account_kt containing '51-01' then ob_account_kt
             when ob_account_dt containing '51-01' then ob_account_dt
             when ob_account_kt containing '76-10' then ob_account_kt
             when ob_account_dt containing '76-10' then ob_account_dt
             when ob_account_dt containing '70-01' then ob_account_dt
             else iif(:ol_oper_subtype in (1, 10), ob_account_kt, ob_account_dt)
           end
    from operbook
    where ob_from_ol_id = :ol_id
    order by 1
    rows 1
    into :acc_kind, :al_id;

    if (acc_kind = 4) then
      acc_kind = 0;
    update or insert into operloan (opl_from_ol_id, opl_acc_kind, opl_from_al_id)
    values (:ol_id, :acc_kind, :al_id)
    matching (opl_from_ol_id);

  end
end
