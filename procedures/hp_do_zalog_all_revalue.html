CREATE PROCEDURE HP_DO_ZALOG_ALL_REVALUE(
    REP_DATE D_DATE)
RETURNS (
    OL_ID D_ID,
    TOTAL_DT D_CURRENCY,
    TOTAL_KT D_CURRENCY,
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_BOOL)
AS
declare variable kc_id d_id;
declare variable amount_sum d_currency;
declare variable sab_id d_id;
declare variable acc_dt d_id;
declare variable acc_kt d_id;
declare variable acc_id d_id;
declare variable kob_id d_id;
declare variable acc_99998 d_id;
declare variable kob_summa d_currency;
declare variable new_zalog_amount d_currency;
declare variable k_id d_id;
declare variable non_resident d_smallint;
declare variable client_type d_smallint;
declare variable zap_id d_integer;
declare variable za_id d_integer;
declare variable zap_from_ol_id d_id;
declare variable ol_is_delete d_smallint;
declare variable res_count d_integer;
begin
-- процедура переоценивает по займу залоговую стоимость по всем активным займам

  total_dt = 0;
  total_kt = 0;
  res_count = 0;
  res_is_error = 0;
  res_text = '';

  select acc_id
  from ref_account, sys_accbalance
  where acc_from_sab_id = sab_id and
        sab_full_code = 'Внебаланс.99998'
  into Acc_99998;

  select sab_id
  from sys_accbalance
  where sab_full_code = 'Микрозаймы.Внебаланс.Залог'
  into :sab_id;

  for select ks_from_k_id, za_id
  from kontstate, zalogauto
  where ks_date_begin <= :rep_date and
        ks_date_end >= :rep_date and
        ks_state = 1 and
        ks_ol_is_delete = 0 and
        za_from_k_id = ks_from_k_id
  into k_id, za_id
  do
  begin
    zap_id = null;
    new_zalog_amount = 0;
    zap_from_ol_id = null;
    ol_is_delete = 0;

    select zap_id, zap_zalog_cost, zap_from_ol_id, is_delete
    from zalogautoprof
         left join operlist on (zap_from_ol_id = ol_id)
    where zap_date <= :rep_date and
          zap_status = 2 and
          zap_from_za_id = :za_id
    order by zap_date desc
    rows 1
    into zap_id, new_zalog_amount, zap_from_ol_id, ol_is_delete;

    if (row_count = 0) then
      Continue;

    kc_id = null;
    select kc_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into kc_id;

    if (kc_id is not null) then
    begin
      if (zap_from_ol_id is not null and ol_is_delete = 0) then
      begin
        if (exists(select * from kn_operbook
        where kob_from_ol_id = :zap_from_ol_id and
              kob_from_kc_id = :kc_id)) then
          Continue;
      end

      acc_id = null;
      amount_sum = 0;
  
      select acc_id
      from ref_account
      where acc_from_sab_id = :sab_id and
            acc_object_kind = 8 and
            acc_object_id = :kc_id
      into acc_id;
  
      if (acc_id is null) then
      begin
        select non_resident, client_type
        from hp_get_kn_client_resident(:kc_id, :rep_date)
        into non_resident, client_type;
  
        select acc_id
        from get_ref_account_knclient('Микрозаймы', 'Залог.ВнеБаланс', :kc_id, :non_resident, :client_type, :rep_date)
        into acc_id;
      end
  
      if (acc_id is null) then
        exception e_default 'Acc_ID not found for KC_ID: ' || kc_id;
  
      select -kobs_bal_sum
      from kn_obsummary
      where kobs_acc_id = :acc_id and
            kobs_date <= :rep_date
      order by kobs_acc_id desc, kobs_date desc
      rows 1
      into amount_sum;
  
      kob_summa = abs(amount_sum - new_zalog_amount);
  
      if (kob_summa = 0) then
        Continue;
  
      if (new_zalog_amount > amount_sum) then
      begin
        acc_dt = acc_99998;
        acc_kt = acc_id;
        total_kt = total_kt + kob_summa;
      end
      else
      begin
        acc_dt = acc_id;
        acc_kt = acc_99998;
        total_dt = total_dt + kob_summa;
      end
  
      if (ol_id is null) then
      begin
        insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no,
                              ol_doc_no_str, ol_memo, ol_parent_id, is_delete, ol_user_id)
        values (1010100, 2, 11, :rep_date, 9, 0, '', 'Переоценка залоговой стоимости',
                null, 0, 1000000000001)
        returning ol_id
        into :ol_id;
      end
  
      kob_id = null;
      select kob_id
      from hp_do_kn_operbook_memorder(:ol_id, :kc_id, null, null, :kob_summa, 0, 0, 0, 'Переоценка залоговой стоимостипо по займу за балансом', :acc_dt, :acc_kt, :rep_date, '810')
      into :kob_id;
      if (kob_id is null) then
        exception e_default 'kob_id is null';
  
      update zalogautoprof
      set zap_from_ol_id = :ol_id
      where zap_id = :zap_id;

      res_count = res_count + 1;
  
      if (exists(select * from kontstate
      where ks_from_k_id = :k_id and
            ks_state = 2 and
            ks_ol_is_delete = 0 and
            ks_date_begin >= :rep_date)) then
        execute procedure hp_do_kn_operbook_one_online(:k_id, current_date, null, :rep_date);
    end
  end

  if (res_count > 0) then
    res_text = 'Переоценка залога успешно проведена, количество: ' || res_count;
  else
  begin
    res_is_error = 1;
    res_text = 'Отсутствуют залоги для переоценки';
  end

  suspend;
end
