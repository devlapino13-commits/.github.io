CREATE PROCEDURE HP_CORRECT_RESH_TEMP(
    KK_ID DECIMAL(18,0),
    OPER_DD DATE)
RETURNS (
    MIN_SH_ID INTEGER,
    MIN_SH_DD DATE,
    OK_TYPE INTEGER,
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    OK_OS DECIMAL(18,4),
    OK_BALANCE DECIMAL(18,4),
    SH_DIFF_DATE INTEGER)
AS
begin
  /* Вытаскиваем график погашения на указанную дату */
  select min(ok_sh_id), min(ok_sh_date)
  from operkred
  where ok_ol_is_active = 1 and
        ok_from_k_id = :kk_id and
        ok_sh_date <= :oper_dd and
        ok_ol_date = :oper_dd and
        ok_os <> 0 and
        ok_from_c_id = 0
  into :min_sh_id, :min_sh_dd;

  if (min_sh_dd >= :oper_dd) then
  begin
    select max(ok_sh_id), max(ok_sh_date)
    from operkred
    where ok_ol_is_active = 1 and
          ok_from_k_id = :kk_id and
          ok_sh_date < :oper_dd and
          ok_os <> 0 and
          ok_from_c_id = 0
    into :min_sh_id, :min_sh_dd;
  end

  if (min_sh_id > 0) then
  begin
    select sum(ok_os)
    from operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :min_sh_dd and
          ok_from_k_id = :kk_id and
          ok_from_c_id = 0
    into :ok_balance;
  end
  else
  begin
    ok_balance = 0;
    select sum(ok_os)
    from operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :oper_dd and
          ok_from_k_id = :kk_id and
          ok_from_c_id = 0
    into :ok_balance;
  end

  sh_diff_date = 0;

  ok_type = 1;/* График */
  ok_os = 0;
  for select ok_sh_id, ok_sh_date, sum(ok_sh_os), sum(ok_sh_perc), sh_diff_date
      from operkred, schedule
      where ok_ol_is_active = 1 and
            ok_ol_date <= :oper_dd and
            ok_sh_date > :min_sh_dd and
            ok_from_k_id = :kk_id and
            sh_from_k_id = :kk_id and
            sh_id = ok_sh_id and
            ok_from_c_id = 0
      group by ok_sh_id, ok_sh_date, sh_diff_date
      order by ok_sh_date
      into :sh_id, :sh_date, :sh_os, :sh_perc, :sh_diff_date
  do
  begin
    suspend;
  end

  ok_type = 0;/* погашения */
  sh_id = 0;
  sh_os = 0;
  sh_perc = 0;
  sh_diff_date = 0;

  ok_balance = 0;
  select sum(ok_os)
  from operkred
  where ok_ol_is_active = 1 and
        ok_ol_date < :min_sh_dd and
        ok_from_k_id = :kk_id and
        ok_from_c_id = 0
  into :ok_balance;

  if (ok_balance is null) then
    ok_balance = 0;

  for select ok_ol_date, sum(ok_os)
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date between :min_sh_dd and :oper_dd and
            ok_from_k_id = :kk_id and
            ok_from_c_id = 0
      group by ok_ol_date
      order by ok_ol_date
      into :sh_date, :ok_os
  do
  begin
    select sh_diff_date
    from schedule
    where sh_date = :sh_date and
          sh_from_k_id = :kk_id
    into :sh_diff_date;

    if (sh_diff_date is null) then
      sh_diff_date = 0;
    ok_balance = ok_balance + :ok_os;
    suspend;
  end

end
