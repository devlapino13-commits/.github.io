CREATE PROCEDURE AAREP_KO_EQUIFAX_PASS_LIST(
    BLS_ID D_ID)
RETURNS (
    DOC_ACTIVITY D_STR_CODE,
    DOC_COUNTRY D_STR_255,
    DOC_TYPE_STR D_STR_255,
    DOC_SERIAL D_STR_CODE,
    DOC_NUMBER D_STR_CODE,
    DOC_DATE D_DATE,
    DOC_END_DATE D_DATE)
AS
declare variable jso d_integer;
declare variable jso_tmp d_integer;

declare function getcatval(kind d_str_code, code d_str_code)
  returns d_str_255
  as
  begin
    return coalesce((select bct_name from bki_catalog where bct_kind = :kind and bct_code = :code), 'Неопределено');
  end

begin
  jso = f_bki_bls2json(bls_id);

  -- ДУЛ текущий
  doc_activity = 'Текущий';
  jso_tmp = json.obj_get_obj(jso, 'response.titlePart._private.doc');

  if (jso_tmp is not null) then
  begin
    doc_country = iif(json.obj_get_val(jso_tmp, 'country') = '999',
              json.obj_get_val(jso_tmp, 'countryText'),
              getcatval('CountryOKSM', json.obj_get_val(jso_tmp, 'country')));

    doc_type_str = getcatval('bki_doc_type', json.obj_get_val(jso_tmp, 'typeValue'));

    doc_serial = json.obj_get_val(jso_tmp, 'serial');
    doc_number = json.obj_get_val(jso_tmp, 'number');
    doc_date = f_bki_json_get_date(jso_tmp, 'date');
    doc_end_date = f_bki_json_get_date(jso_tmp, 'endDate');

    suspend;
  end

  doc_activity = 'Предыдущий';

  for select jso
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'response.titlePart._private.history.doc'), 1)
  into jso_tmp
  do
  begin
    doc_country = iif(json.obj_get_val(jso_tmp, 'country') = '999',
              json.obj_get_val(jso_tmp, 'countryText'),
              getcatval('CountryOKSM', json.obj_get_val(jso_tmp, 'country')));

    doc_type_str = getcatval('bki_doc_type', json.obj_get_val(jso_tmp, 'typeValue'));

    doc_serial = json.obj_get_val(jso_tmp, 'serial');
    doc_number = json.obj_get_val(jso_tmp, 'number');
    doc_date = f_bki_json_get_date(jso_tmp, 'date');
    doc_end_date = f_bki_json_get_date(jso_tmp, 'endDate');

    if (doc_number is not null or doc_date is not null) then
      suspend;
  end
end
