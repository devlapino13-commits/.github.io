CREATE PROCEDURE HP_DO_KN_OPERBOOK_KONT_RSV(
    KC_ID ID_TYPE,
    OL_DATE DATE,
    OL_REG_ID INTEGER,
    OL_IS_ACTIVE INTEGER,
    UN_ID ID_TYPE)
AS
declare variable oks_llr_os curr_type;
declare variable oks_llr_perc curr_type;
declare variable oks_diff_ecl_llr curr_type;
declare variable reserv_kind integer;
declare variable ol_doc_no integer;
declare variable ol_doc_no_str d_str_20;
declare variable kontract_no d_str_255;
declare variable kontract_date date;
declare variable ob_detail d_str_long;
declare variable close_state d_integer;
declare variable oks_off_esp_bal curr_type;
declare variable non_resident d_integer;
declare variable client_type d_integer;
declare variable reserv_kind_prv d_integer;
declare variable loan_acc_group d_str_255;
declare variable reserv_os d_currency;
declare variable reserv_perc d_currency;
declare variable reserv_diff_ro d_currency;
declare variable reserv_law d_currency;
declare variable dt_minus_acc_id d_id;
declare variable dt_acc_id d_id;
declare variable kt_acc_id d_id;
declare variable kt_sys_acc d_str_code;
declare variable rsv_kind d_smallint;
declare variable rsv_code d_str_code;
declare variable detail_1 d_str_255;
declare variable detail_2 d_str_255;
declare variable detail_3 d_str_255;
declare variable sab_id d_id;
declare variable ol_id d_id;
begin
  loan_acc_group = 'Микрозаймы'; -- в будущем нужно прикрепить название раздела системного счета в таблицу продуктов.

  -- начисления резерва
  if (ol_date is null) then
    exception e_default 'Date is empty';

  if (ol_date >= cast('01.03.2019' as date)) then
  begin
    select ol_id
    from hp_get_kn_operbook_ol(:ol_date, :ol_reg_id, 'calc_reserv')
    into :ol_id;
  end
  else
  begin
    if (ol_id is null) then
    begin
      select k_kontract_no, k_kontract_date
      from kontract, knclient
      where kc_id = :kc_id and
            kc_from_k_id = k_id
      into :kontract_no, kontract_date;
      kontract_no = 'договору займа ' || kontract_no || ' от ' ||f_datetostr(:kontract_date) || ' г.';

      ob_detail = 'Начисления по ' || kontract_no;
      insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_doc_no,
                            ol_doc_no_str, ol_memo, ol_parent_id, is_delete, ol_user_id)
      values (:ol_reg_id, 2, 30, :ol_date, 9, :ol_doc_no, :ol_doc_no_str, :ob_detail, null, 0, :un_id)
      returning ol_id
      into :ol_id;
    end
  end

  for select oks_from_kc_id, oks_ecl_acc_kind, oks_llr_os, oks_llr_perc, oks_diff_ecl_llr, oks_close_state,
             oks_off_esp_bal, oks_reserv_law
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            oks_date_begin = :ol_date and
            (oks_close_state <> 2 or
            (oks_close_state = 2 and
            (oks_acr_law <> 0 or
            oks_pmt_law <> 0 or
            oks_off_law <> 0)))
      into kc_id, reserv_kind, oks_llr_os, oks_llr_perc, oks_diff_ecl_llr, close_state, oks_off_esp_bal, reserv_law
  do
  begin
    if (reserv_os is null) then
      reserv_os = 0;
    if (oks_llr_perc is null) then
      oks_llr_perc = 0;
    if (oks_diff_ecl_llr is null) then
      oks_diff_ecl_llr = 0;

    reserv_os = 0;
    reserv_perc = 0;
    reserv_diff_ro = 0;

    if (close_state = 0) then
    begin
      reserv_os = oks_llr_os;
      reserv_perc = oks_llr_perc;
      reserv_diff_ro = oks_diff_ecl_llr;
    end
    else
    if (close_state = 1) then -- если займ закрыт, доводим РО до 100% остатка ВБС
      reserv_diff_ro = oks_diff_ecl_llr;
--      reserv_diff_ro = oks_off_esp_bal;

    select non_resident, client_type
    from hp_get_kn_client_resident(:kc_id, :ol_date)
    into non_resident, client_type;

    -- Перенос остатка на новую корзину
    rsv_kind = 1;
    while (rsv_kind <= 4) do
    begin
      kt_sys_acc = '';
      if (close_state <> 2) then
      begin
        if (rsv_kind = 1) then -- Резерв на основную сумму
        begin
          kt_sys_acc = 'ОС' || reserv_kind || '.Резерв';
          rsv_code = '.Резерв.ОС';
          detail_1 = 'Перенос остатка резерва на ОД с "Кор-ка до РО ';
          detail_2 = 'ой корзины" на "Кор-ка до РО ';
          detail_3 = 'ей корзины"';
        end
        if (rsv_kind = 2) then -- Резерв на проценты
        begin
          kt_sys_acc = 'Процент' || reserv_kind || '.Резерв';
          rsv_code = '.Резерв.Процент';
          detail_1 = 'Перенос остатка резерва на %% с "Кор-ка до РО ';
          detail_2 = 'ой корзины" на "Кор-ка до РО ';
          detail_3 = 'ей корзины"';
        end
        if (rsv_kind = 3) then -- Резерв на РО
        begin
          kt_sys_acc = 'Корректировка' || reserv_kind || '.Резерв';
          rsv_code = '.Резерв.Корректировка';
          detail_1 = 'Перенос остатка резерва РО с "Кор-ка до РО ';
          detail_2 = 'ой корзины" на "Кор-ка до РО ';
          detail_3 = 'ей корзины"';
        end
      end
      if (rsv_kind = 4) then -- Резерв на госпошлину
      begin
        kt_sys_acc = 'Госпошлина' || reserv_kind || '.Резерв';
        rsv_code = '.Резерв.Госпошлина';
        detail_1 = 'Перенос остатка резерва госпошлины с корзины № ';
        detail_2 = ' на корзину № ';
        detail_3 = '';
      end

      if (kt_sys_acc <> '') then
      begin
        select acc_id
        from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
        into :kt_acc_id;

        reserv_kind_prv = 1;
        while (reserv_kind_prv <= 3) do
        begin
          ob_detail = detail_1 || reserv_kind_prv || detail_2 || reserv_kind || detail_3;

          for select sab_id
              from sys_accbalance
              where sab_full_code = :loan_acc_group || :rsv_code || :reserv_kind_prv
              into sab_id
          do
          begin
            for select acc_id
                from ref_account, kn_obsummary
                where acc_from_sab_id = :sab_id and
                      acc_object_kind = 8 and
                      acc_object_id = :kc_id and
                      kobs_acc_id = acc_id and
                      kobs_date <= :ol_date and
                      kobs_date_end >= :ol_date and
                      acc_id <> :kt_acc_id and
                      (kobs_bal_sum <> 0 or
                      kobs_bal_tax <> 0)
                into dt_acc_id
            do
            begin
              execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, null, null, 0, 0, 0, :ol_date,
                  :ol_reg_id, :ol_is_active, :ob_detail, 1, 1);
            end
          end
          reserv_kind_prv = reserv_kind_prv + 1;
        end
      end
      rsv_kind = rsv_kind + 1;
    end
    -- Конец переноса остатка на новую корзину

    if (close_state <> 2) then
    begin
      dt_minus_acc_id = (select acc_id
                         from get_ref_account_office(:loan_acc_group, 'ИзмРезерва.Доход', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date));

      ob_detail = 'РВПЗ на основную сумму';

      select acc_id
      from get_ref_account_office(:loan_acc_group, 'ИзмРезерва.Расход', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
      into :dt_acc_id;

      kt_sys_acc = 'ОС' || reserv_kind || '.Резерв';
      select acc_id
      from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
      into :kt_acc_id;

      execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, :dt_minus_acc_id, null, :reserv_os, 0,
          :reserv_os, :ol_date, :ol_reg_id, :ol_is_active, :ob_detail, -1, 0);

      -- Резерв на %%
      ob_detail = 'РВПЗ на сумму процентов';

      select acc_id
      from get_ref_account_office(:loan_acc_group, 'ИзмРезерва.Расход', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
      into :dt_acc_id;

      kt_sys_acc = 'Процент' || reserv_kind || '.Резерв';

      select acc_id
      from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
      into :kt_acc_id;

      execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, :dt_minus_acc_id, null, :reserv_perc,
          0, :reserv_perc, :ol_date, :ol_reg_id, :ol_is_active, :ob_detail, -1, 0);

      -- Резерв РО
      select acc_id
      from get_ref_account_office(:loan_acc_group, 'ИзмРезерваРО.Доход', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
      into :dt_minus_acc_id;

      ob_detail = 'Корректировка до РО';

      select acc_id
      from get_ref_account_office(:loan_acc_group, 'ИзмРезерваРО.Расход', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
      into :dt_acc_id;

      kt_sys_acc = 'Корректировка' || reserv_kind || '.Резерв';
      select acc_id
      from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
      into :kt_acc_id;

      execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, :dt_minus_acc_id, null,
          :reserv_diff_ro, 0, 0, :ol_date, :ol_reg_id, :ol_is_active, :ob_detail, -1, 0);
    end

    -- резерв по Госпошлине
    ob_detail = 'Резерв по госпошлине';

    select acc_id
    from get_ref_account_office(:loan_acc_group, 'ИзмРезерваГоспошлина.Доход', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
    into :dt_minus_acc_id;

    select acc_id
    from get_ref_account_office(:loan_acc_group, 'ИзмРезерваГоспошлина.Расход', :ol_reg_id, '', -1, :non_resident, :client_type, :ol_date)
    into :dt_acc_id;

    kt_sys_acc = 'Госпошлина' || reserv_kind || '.Резерв';
    select acc_id
    from get_ref_account_knclient(:loan_acc_group, :kt_sys_acc, :kc_id, :non_resident, :client_type, :ol_date)
    into :kt_acc_id;

    execute procedure hp_do_kn_operbook_ins(:ol_id, :kc_id, :dt_acc_id, :kt_acc_id, :dt_minus_acc_id, null, :reserv_law, 0,
        0, :ol_date, :ol_reg_id, :ol_is_active, :ob_detail, -1, 0);
  end
end
