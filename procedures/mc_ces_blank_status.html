CREATE PROCEDURE MC_CES_BLANK_STATUS(
    MCL_ID D_ID,
    EDS_ID D_ID)
RETURNS (
    BLANK_STATUS D_SMALLINT,
    ERROR_TEXT D_STR_255)
AS
declare variable c_id d_id;
begin
    error_text = '';

    select coalesce(mcl_from_c_id, 0)
    from mc_login
    where mcl_id = :mcl_id
        and mcl_is_delete = 0
    into :c_id;

    if (c_id = 0) then
    begin
        error_text = 'id пользователя не подвязан к КМП';
        suspend;
        exit;
    end

    select edb_status
    from edoc_sign
    inner join edoc_blank on eds_from_edl_id = edb_from_edl_id
    where eds_id = :eds_id
        and eds_from_c_id = :c_id
    into blank_status;

    suspend;
end
