CREATE PROCEDURE HP_FPS_APPROVE_CLIENT_NAME(
    K_ID D_ID,
    UN_ID D_ID)
AS
declare variable fcn_id d_integer;
begin
  -- Монета СБП, подтверждаем ФИО клиента,
  -- если выдача по СБП, и требуется подтверждение имени клиента
  for select fcn_id
  from kontract, knclient, fps_loan_banks, fps_client_banks, fps_client_names
  where k_id = :k_id and
        k_disb_kind = 7 and
        kc_from_k_id = k_id and
        flb_from_kc_id = kc_id and
        fbl_from_fcb_id  = fcb_id and
        fcn_from_fcb_id = fcb_id and
        fcb_status = 1 and       -- На проверке Андеррайтеров
        fcn_status = 0           -- Ожидает подтверждения
  into fcn_id
  do
  begin
    -- fps_client_bank.fcb_status обновится через триггер
    update fps_client_names
    set fcn_status = 2,
        fcn_approve_un_id = :un_id
    where fcn_id = :fcn_id;
  end
end
