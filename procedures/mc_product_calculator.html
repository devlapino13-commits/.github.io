CREATE PROCEDURE MC_PRODUCT_CALCULATOR(
    MCL_ID D_ID,
    PR_ID D_INTEGER,
    AMOUNT D_CURRENCY,
    PERIOD D_INTEGER,
    IS_TEST_DATA D_SMALLINT = 0,
    TEST_C_ID D_ID = null)
RETURNS (
    MONTHLY_PAYMENT D_STR_100)
AS
declare variable SH_OS D_CURRENCY;
declare variable SH_PERC D_CURRENCY;
declare variable SH_KOMMIS D_CURRENCY;
declare variable SH_PERC_TAX D_CURRENCY;
declare variable NSP_PERCENT D_CURRENCY;
declare variable C_CURR_STATUS smallint;
declare variable LOAN_PERCENT PERC_TYPE;
declare variable SUM_PAUMENT integer;
declare variable C_ID D_ID;
declare variable DB_ID D_INTEGER;
begin

  db_id = f_get_db_id();
  if (pr_id = 134 and db_id = 81001) then
  begin
    monthly_payment = '';
    suspend;
    exit;
  end

  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  if (coalesce(pr_id, 0) <= 0) then
    exception e_default 'parametr "pr_id" has invalid value';

  if (coalesce(amount, 0) <= 0) then
    exception e_default 'parametr "amount" has invalid value';

  if (coalesce(period, 0) <= 0) then
    exception e_default 'parametr "period" has invalid value';

  if (is_test_data = 1) then
    c_id = f_get_const('test_c_id');
  else
  if (test_c_id is not null) then
    c_id = test_c_id;
  else
  begin
    select mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into :c_id;
  end

  c_curr_status = null;
  select c_curr_status
  from hp_get_client_status(:c_id, current_date)
  into c_curr_status;
  if (c_curr_status is null) then
    c_curr_status = 0;

  select perc_default
  from hp_get_percent(:pr_id, current_date, 1, :amount, :period, :amount, :amount, :c_id)
  into loan_percent;

  --  Метод 1, которы работает в самом финстарте
  execute procedure hp_make_schedule_date(:amount, :period, 0, current_date, :pr_id, 0, 0, '', 1, 0);

  execute procedure hp_make_schedule_perc(:amount, :period, :pr_id, :loan_percent, 0, 0, 0);

  -- RFCKZ-349
  if (db_id in (81001, 39801)) then --RF, KZ
  begin
    select sh_os + sh_perc
    from temp_schedule
    where sh_id = 1
    into sum_paument;
    if (db_id = 81001) then
        monthly_payment = sum_paument || ' руб.';
    else
        monthly_payment = sum_paument || ' тенге';
  end

  if (db_id = 41701) then --KG
  begin
    select sh_os, sh_perc, sh_kommis
    from temp_schedule
    where sh_id = 2
    into sh_os, sh_perc, sh_kommis;

    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= current_date and
          nsd_date_end >= current_date
    into nsp_percent;

    select res
    from fhp_round_bank(((:sh_perc + :sh_kommis) * :nsp_percent / 100.0), 0)
    into :sh_perc_tax;

    sum_paument = sh_os + sh_perc + sh_kommis + sh_perc_tax;

    monthly_payment = sum_paument || ' сом';
  end

  if (db_id = 94401) then --AZ
  begin
    select sh_os + sh_perc
    from temp_schedule
    where sh_id = 1
    into sum_paument;
    monthly_payment = sum_paument || ' manat.';
  end
  delete from temp_schedule;
  suspend;
end
