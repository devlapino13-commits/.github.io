CREATE PROCEDURE HP_KONTRACT_LOAD_FOR_DISB(
    UN_ID DECIMAL(18,0),
    K_ID DECIMAL(18,0))
RETURNS (
    RES_ID INTEGER,
    RES_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable kf_id decimal(18,0);
declare variable k_disb_date date;
declare variable pr_id integer;
declare variable k_kontract_date date;
declare variable k_disb_amount decimal(18,4) = 0;
declare variable k_kommis_perc decimal(18,7);
declare variable k_kontract_no varchar(20);
declare variable k_kontract_date_str varchar(20);
declare variable k_disb_date_str varchar(20);
declare variable k_kommis_amount decimal(18,4) = 0;
declare variable tmp_txt d_blob_text;
declare variable region_id integer;
declare variable folder_name varchar(100);
declare variable lo_name varchar(100);
declare variable tmp_amount decimal(18,4) = 0;
declare variable manager_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable c_name_fio varchar(100);
declare variable pass_id decimal(18,0);
declare variable pr_name varchar(255);
declare variable k_nsp_amount decimal(18,4) = 0;
declare variable nsp_perc decimal(18,7);
declare variable txt varchar(2000);
declare variable db_id integer;
declare variable pr_round_admin integer;
declare variable is_ipf integer;
declare variable pr_code varchar(100);
declare variable psi_id id_type;
declare variable psi_code str_20;
declare variable psi_pin_code str_20;
declare variable crd_txt d_blob_text;
declare variable psp_id integer;
declare variable cardnumber str_100;
declare variable default_disb_kind integer;
declare variable region_id_tmp d_id;
declare variable k_kolich_client_approved d_integer;
declare variable card_txt d_str_100;
declare variable is_power_user d_smallint;
declare variable can_disb_on_card d_bool;
declare variable v_delimiter varchar(3);
begin
  -- Загрузка данных для отображения в окне выдачи
  res_id = 1;
  res_txt = '';

  is_power_user = 0;

  if (is_power_user = 0 and exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Superuser')
             where has_access = 1)) then
    is_power_user = 1;

  if (is_power_user = 0 and exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Programmer')
             where has_access = 1)) then
    is_power_user = 1;

  if (is_power_user = 0 and exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Operator')
             where has_access = 1)) then
    is_power_user = 1;

  if (is_power_user = 0 and exists(select *
             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Permission_Administrator')
             where has_access = 1)) then
    is_power_user = 1;

  db_id = f_get_db_id();
  select k_from_kf_id, k_disb_date, k_from_pr_id, k_kontract_date, k_amount_approved, k_kommis_perc, k_kontract_no,
         k_approved_manager,
         pr_name || trim(iif(coalesce(pr_memo, '') <> '' and coalesce(pr_memo, '') <> pr_name, ' (' || pr_memo || ')', '')),
         pr_round_admin, pr.pr_code, kont.k_kolich_client_approved
  from kontract kont, product pr
  where k_id = :k_id and
        k_from_pr_id = pr_id
  into kf_id, k_disb_date, pr_id, k_kontract_date, k_disb_amount, k_kommis_perc, k_kontract_no, manager_id, pr_name,
       pr_round_admin, pr_code, k_kolich_client_approved;

  if (k_kommis_perc is null) then
    k_kommis_perc = 0;
  if (manager_id is null) then
    manager_id = 0;

  k_kontract_date_str = right('00' || extract(day from k_kontract_date), 2) || '.' || right('00' || extract(month from k_kontract_date), 2) || '.' || extract(year from k_kontract_date);

  k_disb_date_str = right('00' || extract(day from k_disb_date), 2) || '.' || right('00' || extract(month from k_disb_date), 2) || '.' || extract(year from k_disb_date);

  select kal_new_reg_id, kal_new_kf_name, c_name_fio
  from knaktlist, clients
  where knaktlist.is_delete = 0 and
        kal_date <= :k_disb_date and
        kal_date_end >= :k_disb_date and
        kal_from_kf_id = :kf_id and
        kal_new_kf_officer = c_id
  into :region_id, :folder_name, :lo_name;

  -- Сумма комиссии
  k_kommis_amount = round(k_disb_amount * k_kommis_perc / 100.0, -pr_round_admin);
  select sum(kc_kommis_lgota)
  from knclient
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into :tmp_amount;
  if (tmp_amount is null) then
    tmp_amount = 0;
  k_kommis_amount = k_kommis_amount - tmp_amount;
  if (k_kommis_amount < 0) then
    k_kommis_amount = 0;

  nsp_perc = 0;
  select nsp_percent_for_extra_int
  from hp_get_kontract_nsp_acc(:region_id, :k_disb_date)
  into :nsp_perc;
  if (nsp_perc is null) then
    nsp_perc = 0;

  k_nsp_amount = round(k_kommis_amount * nsp_perc / 100.0, 0);
  if (k_nsp_amount > 0 and
      mod(round(k_kommis_amount * nsp_perc, 0), 100) = 50) then
    k_nsp_amount = k_nsp_amount - 1;

  -- Получаем казначея
  c_id = 0;
  select kc_from_c_id
  from knclient
  where kc_from_k_id = :k_id and
        kc_status = 2 and
        kc_amount_approved > 0 rows 1
  into c_id;
  if (c_id is null) then
    c_id = 0;
  if (c_id = 0) then
  begin
    select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 rows 1
    into c_id;
  end
  is_ipf = 0;
  if (db_id in (39801, 41701)) then
  begin
    if (exists(select *
               from kontmat
               where kml_object_kind = 9 and
                     kml_object_id = :k_id and
                     kml_total > 0)) then
      is_ipf = 1;
  end

  if (k_kommis_amount is null) then
    k_kommis_amount = 0;
  if (k_nsp_amount is null) then
    k_nsp_amount = 0;

  can_disb_on_card = f_can_disb_on_card('kontract', k_id, db_id);

  if ((db_id = 94401)) then
  begin

    if (pr_code containing '.CardDisb.') then
      default_disb_kind = 4;
    else
    if (pr_code containing '.president_decree.') then
      default_disb_kind = 6;
    else
    if (pr_code containing '.president_decree_operkroff.') then
      default_disb_kind = 7;
    else
    if (pr_code containing '.BizCardDisb.') then
      default_disb_kind = 8;
    else
      default_disb_kind = 2;
  end
  else
  begin
    select k_disb_kind
    from kontract
    where k_id = :k_id
    into default_disb_kind;

    -- После запуска БС СОФР в РФ, по-умолчанию выдача через СБП
    if (default_disb_kind is null) then
      default_disb_kind = iif(can_disb_on_card = 0, 7, 1);
  end

  region_id_tmp = region_id;
  if (default_disb_kind = 5) then
  begin
     if(db_id = 81001) then
        region_id_tmp = 1010100;
     else if(db_id = 41701) then
        region_id_tmp = 1090100;
     else if(db_id = 39801) then
        region_id_tmp = 1010100;
  end

  tmp_txt = '';
  tmp_txt = BLOB_APPEND(tmp_txt, '"disb_date":"' || k_disb_date_str || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"kontract_no":"' || k_kontract_no || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"kontract_date":"' || k_kontract_date_str || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"folder_name":"' || folder_name || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"region_id":"' || region_id || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"disb_region_id":"' || region_id_tmp || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"repay_region_id":"' || region_id || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"lo_name":"' || lo_name || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"product_id":"' || pr_id || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"product_code":"' || pr_code || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"product_name":"' || pr_name || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"disb_amount":"' || k_disb_amount || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"kommis_amount":"' || k_kommis_amount || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"nsp_amount":"' || k_nsp_amount || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"total_repay":"' ||(k_kommis_amount + k_nsp_amount) || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"cashier_id":"' || c_id || '"');
  v_delimiter = iif(tmp_txt = '','',',');
  tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter || '"manager_id":"' || manager_id || '"');
  v_delimiter = iif(tmp_txt = '','',',');

  if ((db_id = 94401)) then
  begin
    tmp_txt = BLOB_APPEND(tmp_txt, ',"default_disb_kind":"' || default_disb_kind || '"');
  end
  else
  begin
    tmp_txt = BLOB_APPEND(tmp_txt, ',"default_disb_kind":"' || coalesce(default_disb_kind, iif(is_ipf = 0, 1, 3)) || '"');
  end

  tmp_txt = BLOB_APPEND(tmp_txt, f_format(',"default_repay_kind":"%1"', iif(db_id = 94401, 3, 1)));

  tmp_txt = BLOB_APPEND(tmp_txt, ',"can_change_accounts":"'|| is_power_user ||'"');

  res_txt = BLOB_APPEND(res_txt, '"loan":{', tmp_txt, '}');

  tmp_txt = '';
  if (is_ipf = 0 and db_id <> 94401) then
  begin
    -- Если кредит числится в ГО, даем другие способы выдачи
    tmp_txt = '{"id":"1", "name":"Наличными"}';
    if ((is_power_user = 1 and db_id in (39801, 41701, 94401)) or
        (db_id = 41701 and region_id = 1090100) or -- ГО
        (db_id = 39801 and region_id = 1010100) or -- ГО
        (db_id = 94401)) then -- AZ
      tmp_txt = BLOB_APPEND(tmp_txt, ', {"id":"2", "name":"Безналичными"}, {"id":"3", "name":"Другой счет"}');
  end
  else
    tmp_txt = '{"id":"3", "name":"Другой счет"}';
  res_txt = BLOB_APPEND(res_txt, ',"repay_kinds":[', tmp_txt, ']');

  if ((db_id = 81001) or
      (db_id in (39801, 41701)) or
      ((db_id = 94401) and
      (pr_code like '%.CardDisb.%'))) then
  begin
    if (db_id = 81001) then -- если РФ
    begin
      if (default_disb_kind = 5) then
        select first 1 et_cardnumber
        from extsys_token
        where et_from_c_id = :c_id
          and et_is_actual = 1
        into cardnumber;

      tmp_txt = BLOB_APPEND(tmp_txt, trim(iif(can_disb_on_card = 1,
        ', {"id":"-1", "name":"Банковская карта онлайн [карта: ' || coalesce(cardnumber, 'Отсутствует') || ']"}' ||
        ', {"id":"5", "name":"Банковская карта офлайн [карта: ' || coalesce(cardnumber, 'Отсутствует') || ']"}', '')) ||
        iif(f_tmp_const('moneta_fps_active') = 1, ', {"id":"7", "name":"СБП по номеру телефона "}', ''));
    end
    else if (db_id = 94401) then -- если АЗ
    begin
      tmp_txt = '{"id":"4", "name":"На банковскую карту MilliKart"}';
      psp_id = 1;
    end
    else if (db_id in (39801, 41701)) then -- если КГ
    begin
      card_txt = '';
      if (default_disb_kind = 5) then
      begin
          for select kncl.kc_from_c_id
          from knclient kncl
          where kncl.kc_from_k_id = :k_id
          and kncl.kc_amount_approved > 0
          into :c_id
          do
          begin
            if (card_txt = '') then
            begin
                cardnumber = (select first 1 et_cardnumber
                             from extsys_token
                             where et_from_c_id = :c_id and
                                   et_is_actual = 1);
    
                if (cardnumber is null) then
                    card_txt = '[карта: Отсутствует]';
            end
          end
      end

      if (card_txt = '') then
      begin
          if (k_kolich_client_approved = 1) then
            card_txt = '[карта: '||coalesce(:cardnumber, '')||']';
          else
            if (k_kolich_client_approved = 1) then
            card_txt = '[групповой займ]';
      end
      tmp_txt = BLOB_APPEND(tmp_txt, ', {"id":"5", "name":"на карту Элькарт', card_txt, '"}');
    end
    if (default_disb_kind = 4) then
    begin
      crd_txt = '';
      v_delimiter = '';
      for select psi_id, psi_code, coalesce(psi_pin_code, '')
          from hp_paysys_get_cl_info(:c_id, :psp_id)
          into :psi_id, :psi_code, :psi_pin_code
      do
      begin
        crd_txt = BLOB_APPEND(crd_txt, v_delimiter, '{"id":"' || psi_id || '", "code":"' || psi_code || '", "pin":"' || psi_pin_code || '"}');
        v_delimiter = ',';
      end
      if (crd_txt <> '') then
        res_txt = BLOB_APPEND(res_txt, ',"bank_cards":[', crd_txt, ']');
    end
  end
  if (db_id = 94401) then
  begin
    if (pr_code containing '.president_decree.') then
    begin
      tmp_txt = '{"id":"6", "name":"Прочие дебиторы и другие активы"}';
    end else
    if (pr_code containing '.president_decree_operkroff.') then
    begin
      tmp_txt = '{"id":"7", "name":"Прочие дебиторы и другие активы"}';
    end else
    if (pr_code containing '.BizCardDisb.') then
    begin
        tmp_txt = '{"id":"8", "name":"На банк. карту MilliKart Бизнес"}';

        crd_txt = '';
          v_delimiter = '';
        for select psi_id, psi_code, coalesce(psi_pin_code, '')
          from hp_paysys_get_cl_info(:c_id, 2)
          into :psi_id, :psi_code, :psi_pin_code
        do
        begin
         crd_txt = BLOB_APPEND(crd_txt, v_delimiter, '{"id":"' || psi_id || '", "code":"' || psi_code || '", "pin":"' || psi_pin_code || '"}');
         v_delimiter = ',';
        end
        if (crd_txt <> '') then
            res_txt = BLOB_APPEND(res_txt, ',"bank_cards":[', crd_txt, ']');

    end
  end
  res_txt = BLOB_APPEND(res_txt, ',"disb_kinds":[', tmp_txt, ']');

  -- Счет склада по ИПФ
  tmp_txt = '';
  if (is_ipf = 1) then
  begin
    if (pr_code containing '.ISTISNAA_PROCESS.') then
    begin
      select list('{"id":"' || al_id || '", "name":"' || al_name || '", "code":"' || al_account || '", "a_currency":"' || al_currency || '"}', ',')
      from find_acc_list('11208/1') -- было find_acc_list('11571/3') изменен от 28.05.2018 Улан
      where al_from_reg_id = :region_id
      into tmp_txt;
      if (tmp_txt is null) then
        tmp_txt = '';
    end
    else
    if (pr_code containing '.IDJARA.') then
    begin
      select list('{"id":"' || al_id || '", "name":"' || al_name || '", "code":"' || al_account || '", "a_currency":"' || al_currency || '"}', ',')
      from find_acc_list('21199/99')
      where al_from_reg_id = :region_id
      into tmp_txt;
      if (tmp_txt is null) then
        tmp_txt = '';
    end
    else
    begin
      select list('{"id":"' || al_id || '", "name":"' || al_name || '", "code":"' || al_account || '", "a_currency":"' || al_currency || '"}', ',')
      from workplace, acc_list
      where wp_from_reg_id = :region_id and
            wp_ml_account = al_id and
            (wp_name like '%Собственность%')
            --             or wp_name containing 'Истиснаа')
      into tmp_txt;
      if (tmp_txt is null) then
        tmp_txt = '';
    end
  end
  res_txt = BLOB_APPEND(res_txt, ',"kontmatacc":[', tmp_txt, ']');

  tmp_txt = '';
  v_delimiter = '';
           
  for select kc_from_c_id, c_name_fio
      from clients, knclient
      where kc_from_c_id = c_id and
            kc_from_k_id = :k_id and
            kc_amount_approved > 0
      into c_id, c_name_fio
  do
  begin
    c_name_fio = replace(c_name_fio, '"', '\"');
    pass_id = 0;
    select c_blob_pass1_id
    from clients
    where c_id = :c_id
    into pass_id;
    if (pass_id is null) then
      pass_id = 0;
    tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter, '{"id":"' || c_id || '", "name":"' || c_name_fio || '", "pass_id":"' || :pass_id || '"}');
    v_delimiter = ',';
  end
  res_txt = BLOB_APPEND(res_txt, ',"clients":[', tmp_txt, ']');

  -- Возвращаем список сертифицирующих (руководителей) офиса
  tmp_txt = '';
  v_delimiter = '';
  for select '{"id": "' || manager_c_id || '","name":"' || manager_fio || '"}'
      from aarep_manager_list(:region_id, :k_disb_date, 1)
      into txt
  do
  begin
    tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter, txt);
    v_delimiter = ',';
  end
  res_txt = BLOB_APPEND(res_txt, ',"managers":[', tmp_txt, ']');

  -- Список офисов где можно выдавать кредит
  tmp_txt = '';
  v_delimiter = '';
  for select '{"id": "' || reg_unicode || '","name":"' || reg_name || '", "code":"' || reg_code || '"}'
      from region
      where reg_unicode = :region_id
      into txt
  do
  begin
    tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter,  txt);
    v_delimiter = ',';
  end

  if (region_id <> iif(db_id = 41701, 1090100, 1010100) and (is_power_user = 1 or default_disb_kind = 5)) then
  begin
    for select '{"id": "' || reg_unicode || '","name":"' || reg_name || '", "code":"' || reg_code || '"}'
        from region
        where reg_unicode = (case :db_id when 81001 then 1010100
                                        when 41701 then 1090100
                                        when 39801 then 1010100
                                        else null end)
        into txt
    do
    begin
      tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter, txt);
      v_delimiter = ',';
    end
  end

  res_txt = BLOB_APPEND(res_txt, ',"regions":[', tmp_txt, ']');

  -- Список офисов где можно выдавать кредит
  tmp_txt = '';
  v_delimiter = '';
  for select '{"sh_id":"' || sh_id || '",' || '"sh_date":"' || sh_date || '",' || '"sh_os":"' || round(sh_os, 2) || '",' || '"sh_perc":"' || round(sh_perc + sh_kommis, 2) || '",' || '"sh_all":"' || round(sh_all, 2) || '",' || '"sh_perc_tax":"' || round(sh_perc_tax, 2) || '",' || '"sh_all_with_tax":"' || round(sh_all_with_tax, 2) || '"}'
      from fm_schedule_list(9, :k_id)
      where sh_id > 0
      order by sh_id
      into txt
  do
  begin
    tmp_txt = BLOB_APPEND(tmp_txt, v_delimiter, txt);
    v_delimiter = ',';
  end
  res_txt = BLOB_APPEND(res_txt, ',"schedule":[',  tmp_txt, ']');

  res_txt = BLOB_APPEND('{', res_txt, '}');

  if (coalesce(res_txt, '') = '') then
  begin
    res_txt = 'Произошла ошибка при загрузке данных. Обратитесь к технической поддержке и сообщите номер договора.';
    res_id = -1;
  end
  suspend;
end
