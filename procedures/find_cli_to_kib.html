CREATE PROCEDURE FIND_CLI_TO_KIB(
    C_FIO VARCHAR(100),
    C_PASS_NO VARCHAR(30),
    C_BIRTH_DATE DATE,
    C_KIB_ID ID_TYPE)
RETURNS (
    C_ID DECIMAL(18,0),
    CC_NAME_FIO VARCHAR(100),
    CC_PASS_NO VARCHAR(30),
    CC_BIRTH_DATE DATE,
    V_FOUND_FIO INTEGER,
    V_FOUND_PASS_NO INTEGER,
    V_FOUND_BIRTH_DATE INTEGER,
    V_FOUND_KIB_ID INTEGER)
AS
begin
  if (c_kib_id is null) then
    c_kib_id = 0;
  v_found_fio = 0;
  v_found_pass_no = 0;
  v_found_birth_date = 0;
  v_found_kib_id = 0;

-- сперва ищеть из KIBCLIID (справочник ID  из киба)
  c_id = 0;
  select kci_from_c_id
  from kibcliid
  where kci_kib_id = :c_kib_id
  into c_id;
-- если находит, то из справочника заполняет данные
  if (c_id > 0) then
  begin
    v_found_kib_id = 1;
    select c_id, c_name_fio, c_pass_no, c_birth_date
    from clients
    where c_id = :c_id
    into c_id, cc_name_fio, cc_pass_no, cc_birth_date;
    suspend;
  end
  else
  begin
    for select c_id, cc_name_fio, cc_pass_no, cc_birth_date, v_found_fio, v_found_pass_no, v_found_birth_date
        from find_one_client_id(:c_fio, :c_pass_no, :c_birth_date)
        into :c_id, :cc_name_fio, :cc_pass_no, :cc_birth_date, :v_found_fio, :v_found_pass_no, :v_found_birth_date
    do
    begin
--    если тут находит то нужно добавить в KIBCLIID
      if (c_id > 0 and
          c_kib_id > 0) then
      begin
        update or insert into kibcliid (kci_from_c_id, kci_kib_id)
        values (:c_id, :c_kib_id)
        matching (kci_from_c_id);
      end
      suspend;
    end
  end
end
