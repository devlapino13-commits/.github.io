CREATE PROCEDURE AAREP_DISB_DAILY(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REP_DATE DATE,
    WEEK_DAY_STR VARCHAR(2),
    KOL_DISB INTEGER,
    SUM_DISB CURR_TYPE,
    KOL_CLI_DISB INTEGER,
    SUM_CLI_DISB CURR_TYPE,
    KOL_DISB_EXPRESS INTEGER,
    SUM_DISB_EXPRESS CURR_TYPE,
    KOL_DISB_SHANS INTEGER,
    SUM_DISB_SHANS CURR_TYPE,
    KOL_DISB_BLAGO INTEGER,
    SUM_DISB_BLAGO CURR_TYPE,
    KOL_DISB_USPEH INTEGER,
    SUM_DISB_USPEH CURR_TYPE,
    KOL_DISB_PATENT INTEGER,
    SUM_DISB_PATENT CURR_TYPE,
    KOL_DISB_PATENT_PLUS INTEGER,
    SUM_DISB_PATENT_PLUS CURR_TYPE,
    KOL_DISB_STATUS INTEGER,
    SUM_DISB_STATUS CURR_TYPE,
    KOL_DISB_LOYAL INTEGER,
    SUM_DISB_LOYAL CURR_TYPE,
    KOL_CLI_DISB_NEW INTEGER,
    SUM_DISB_NEW CURR_TYPE,
    KOL_DISB_ZALOG_CAR INTEGER,
    SUM_DISB_ZALOG_CAR CURR_TYPE,
    KOL_DISB_EDUCATION INTEGER,
    SUM_DISB_EDUCATION CURR_TYPE,
    KOL_DISB_FOOD INTEGER,
    SUM_DISB_FOOD CURR_TYPE,
    KOL_DISB_FLOAT_PERCENT INTEGER,
    SUM_DISB_FLOAT_PERCENT CURR_TYPE,
    KOL_DISB_REL_CL INTEGER,
    SUM_DISB_REL_CL CURR_TYPE,
    KOL_DISB_REL_PL INTEGER,
    SUM_DISB_REL_PL CURR_TYPE)
AS
declare variable rep_date_begin date;
declare variable week_day smallint;
begin
  rep_date_begin = :rep_date_end - extract(day from :rep_date_end);
  rep_date = :rep_date_begin;
  while (:rep_date < :rep_date_end) do
  begin
    rep_date = rep_date + 1;
    week_day = extract(weekday from :rep_date);
    week_day_str = case week_day
                     when 1 then 'пн'
                     when 2 then 'вт'
                     when 3 then 'ср'
                     when 4 then 'чт'
                     when 5 then 'пт'
                     when 6 then 'сб'
                     else 'вс'
                   end;

    select count(kc1.kc_from_c_id) as kol_disb, sum(kc1.kc_amount_approved) as sum_disb,
           sum(iif(kc1.kc_circle < 2, 1, 0)) as kol_cli_disb_new,
           sum(iif(kc1.kc_circle < 2, kc1.kc_amount_approved, 0)) as sum_disb_new,

           sum(iif(pr_code containing '.express.', 1, 0)) as kol_disb_express,
           sum(iif(pr_code containing '.express.', kc1.kc_amount_approved, 0)) as sum_disb_express,

           sum(iif(pr_code containing '.shans.', 1, 0)) as kol_disb_shans,
           sum(iif(pr_code containing '.shans.', kc1.kc_amount_approved, 0)) as sum_disb_shans,

           sum(iif(pr_code containing '.blago.', 1, 0)) as kol_disb_blago,
           sum(iif(pr_code containing '.blago.', kc1.kc_amount_approved, 0)) as sum_disb_blago,

           sum(iif(pr_code containing '.uspeh.', 1, 0)) as kol_disb_uspeh,
           sum(iif(pr_code containing '.uspeh.', kc1.kc_amount_approved, 0)) as sum_disb_uspeh,

           sum(iif(pr_code containing '.patent plus.' and pr_code not containing '.par.', 1, 0)) as kol_disb_patent,
           sum(iif(pr_code containing '.patent plus.' and pr_code not containing '.par.', kc1.kc_amount_approved, 0)) as sum_disb_patent,

           sum(iif(pr_code containing '.patent plus.' and pr_code containing '.par.', 1, 0)) as kol_disb_patent_plus,
           sum(iif(pr_code containing '.patent plus.' and pr_code containing '.par.', kc1.kc_amount_approved, 0)) as sum_disb_patent_plus,

           sum(iif(pr_code containing '.status.', 1, 0)) as kol_disb_status,
           sum(iif(pr_code containing '.status.', kc1.kc_amount_approved, 0)) as sum_disb_status,

           sum(iif(pr_code containing '.loyal.', 1, 0)) as kol_disb_loyal,
           sum(iif(pr_code containing '.loyal.', kc1.kc_amount_approved, 0)) as sum_disb_loyal,

           sum(iif(pr_code containing '.zalog_car.', 1, 0)) as kol_disb_status,
           sum(iif(pr_code containing '.zalog_car.', kc1.kc_amount_approved, 0)) as sum_disb_status,

           sum(iif(pr_code containing '.education.', 1, 0)) as kol_disb_status,
           sum(iif(pr_code containing '.education.', kc1.kc_amount_approved, 0)) as sum_disb_status,

           sum(iif(pr_code containing '.food.', 1, 0)) as kol_disb_status,
           sum(iif(pr_code containing '.food.', kc1.kc_amount_approved, 0)) as sum_disb_food,

           sum(iif(pr_code containing '.FLOAT_PERCENT.', 1, 0)) as kol_disb_status, --
           sum(iif(pr_code containing '.FLOAT_PERCENT.', kc1.kc_amount_approved, 0)) as sum_disb_food,

           sum(iif(pr_code containing '.reliable_client.', 1, 0)) as kol_disb_rel_cl, --
           sum(iif(pr_code containing '.reliable_client.', kc1.kc_amount_approved, 0)) as sum_disb_rel_cl,

           sum(iif(pr_code containing '.reliable_client_plus.', 1, 0)) as kol_disb_rel_pl, --
           sum(iif(pr_code containing '.reliable_client_plus.', kc1.kc_amount_approved, 0)) as sum_disb_rel_pl   --' or pr_code containing '.reliable_client_plus.'

    from kontstate ks1, knclient kc1, kontract k, product, region_list(:rep_region_id, 1, 3, 1)
    where ks1.ks_ol_is_delete = 0 and
          ks1.ks_state = 1 and
          cast(ks1.ks_date_begin as date) = :rep_date and
          ks1.ks_from_k_id = kc1.kc_from_k_id and
          kc1.kc_amount_approved > 0 and
          ks1.ks_from_k_id = k_id and
          k_from_pr_id = pr_id and
          ks_region_id = id_list

    into kol_disb, sum_disb, kol_cli_disb_new, sum_disb_new, kol_disb_express, sum_disb_express, kol_disb_shans,
         sum_disb_shans, kol_disb_blago, sum_disb_blago, kol_disb_uspeh, sum_disb_uspeh, kol_disb_patent,
         sum_disb_patent, kol_disb_patent_plus, sum_disb_patent_plus, kol_disb_status, sum_disb_status, kol_disb_loyal,
         sum_disb_loyal, kol_disb_zalog_car, sum_disb_zalog_car, kol_disb_education, sum_disb_education, kol_disb_food,
         sum_disb_food, kol_disb_float_percent, sum_disb_float_percent, kol_disb_rel_cl, sum_disb_rel_cl, kol_disb_rel_pl, sum_disb_rel_pl;

    if (sum_disb is null) then
      sum_disb = 0;

    kol_disb_express = coalesce(kol_disb_express, 0);
    sum_disb_express = coalesce(sum_disb_express, 0);
    if (kol_cli_disb_new is null) then
      kol_cli_disb_new = 0;
    if (sum_disb_new is null) then
      sum_disb_new = 0;
    kol_disb_shans = coalesce(kol_disb_shans, 0);
    sum_disb_shans = coalesce(sum_disb_shans, 0);
    kol_disb_blago = coalesce(kol_disb_blago, 0);
    sum_disb_blago = coalesce(sum_disb_blago, 0);
    kol_disb_uspeh = coalesce(kol_disb_uspeh, 0);
    sum_disb_uspeh = coalesce(sum_disb_uspeh, 0);
    kol_disb_patent = coalesce(kol_disb_patent, 0);
    sum_disb_patent = coalesce(sum_disb_patent, 0);
    kol_disb_patent_plus = coalesce(kol_disb_patent_plus, 0);
    sum_disb_patent_plus = coalesce(sum_disb_patent_plus, 0);
    kol_disb_status = coalesce(kol_disb_status, 0);
    sum_disb_status = coalesce(sum_disb_status, 0);
    kol_disb_loyal = coalesce(kol_disb_loyal, 0);
    sum_disb_loyal = coalesce(sum_disb_loyal, 0);
    kol_disb_zalog_car = coalesce(kol_disb_zalog_car, 0);
    sum_disb_zalog_car = coalesce(sum_disb_zalog_car, 0);
    kol_disb_education = coalesce(kol_disb_education, 0);
    sum_disb_education = coalesce(sum_disb_education, 0);
    kol_disb_food = coalesce(kol_disb_food, 0);
    sum_disb_food = coalesce(sum_disb_food, 0);
    kol_disb_float_percent = coalesce(kol_disb_float_percent, 0);
    sum_disb_float_percent = coalesce(sum_disb_float_percent, 0);
    kol_disb_rel_cl = coalesce(kol_disb_rel_cl, 0);
    sum_disb_rel_cl = coalesce(sum_disb_rel_cl, 0);
    kol_disb_rel_pl = coalesce(kol_disb_rel_pl, 0);
    sum_disb_rel_pl = coalesce(sum_disb_rel_pl, 0);

    kol_cli_disb = kol_disb - kol_disb_loyal - kol_disb_patent_plus;
    sum_cli_disb = sum_disb - sum_disb_loyal - sum_disb_patent_plus;

    suspend;
  end
end
