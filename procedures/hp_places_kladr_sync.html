CREATE PROCEDURE HP_PLACES_KLADR_SYNC(
    PARENT_PKI_ID D_ID = null,
    PARENT_PL_ID D_ID = null,
    PARENT_KLADR_OKATO D_STR_20 = null,
    PLT_JSO D_INTEGER = null)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable pl_type d_str_20;
declare variable kladr_type d_str_20;
declare variable pki_id d_id;
declare variable pl_id d_id;
declare variable plt_id d_id;
declare variable kst_id d_id;
declare variable err d_boolean;
declare variable need_edit d_boolean;
declare variable obj_name d_str_long;
declare variable kladr_code d_str_code;
declare variable pl_code d_str_code;
declare variable kladr_okato d_str_20;
declare variable pl_okato d_str_20;
declare variable comm d_str_long;
declare variable status d_str_code;
declare variable ksm_status d_smallint;
begin
  -- процедура регулярной сверки адресных справочников КЛАДР и БД
  -- обрабатываем только АКТУАЛЬНЫЕ записи
  error_text = '';

  if (parent_pki_id is null) then -- запуск основной процедуры
  begin
    select ksm_status
    from kladr_sync_main
    where ksm_status < 4
    into ksm_status;
  
    if (ksm_status is null) then
    begin
      error_text = 'Справочник не загружен';
      suspend;
      exit;
    end 
  
    if (ksm_status < 2) then
    begin
      error_text = 'Иерархия еще не построена';
      suspend;
      exit;
    end 

    if (ksm_status > 2) then
    begin
      error_text = 'Сверка уже произведена';
      suspend;
      exit;
    end 

    parent_pki_id = (select pki_id from places_kladr_import where pki_code = rpad('', 13, '0'));

    if (parent_pki_id is null) then
    begin
      error_text = 'Постройте иерархию загруженного справочника КЛАДР';
      suspend;
      exit;
    end

    parent_pl_id = 810000000001;
    plt_jso = json.obj_create();

    for select plt_short_name, max(plt_id)
    from placetype
    where plt_can_select < 2
      and plt_short_name <> ''
      and plt_id starting with '810'
    group by 1
    into pl_type, plt_id
    do
    begin
      json.obj_set_s(plt_jso, replace(pl_type,'.','#'), plt_id);
      json.obj_set_s(plt_jso, replace(pl_type,'.',''), plt_id);
    end
  end

  update places_kladr_import
  set pki_okatd = :parent_kladr_okato
  where pki_parent = :parent_pki_id
    and pki_okatd is null;

  -- шаг 1. поиск
    -- бывают нас. пункты актуальные и с одинаковым названием, различаются только коды. Чтобы избежать дублей ищем два раза, с кодом и без
  -- шаг 1.1. найдем полное соответствие по названию и типу и коду АКТУАЛЬНЫЕ
  for select pki_id, pki_name, replace(pki_type, '.', ''), left(pki_code, 11)
  from places_kladr_import
  where pki_parent = :parent_pki_id
    and right(pki_code, 2) = '00'
  into pki_id, obj_name, kladr_type, kladr_code
  do
  begin
    update kladr_sync_tmp
    set kst_from_pki_id = :pki_id
    where kst_pl_name = :obj_name
      and kst_pl_type_index = :kladr_type
      and kst_pl_parent = :parent_pl_id
      and kst_pl_code = '810' || :kladr_code
      and coalesce(kst_from_pki_id, 0) = 0;
  end

  -- шаг 1.2. найдем полное соответствие по названию и типу АКТУАЛЬНЫЕ
  for select pki_id, pki_name, replace(pki_type, '.', '')
  from places_kladr_import
  where pki_parent = :parent_pki_id
    and right(pki_code, 2) = '00'
  into pki_id, obj_name, kladr_type
  do
  begin
    update kladr_sync_tmp
    set kst_from_pki_id = :pki_id
    where kst_pl_name = :obj_name
      and kst_pl_type_index = :kladr_type
      and kst_pl_parent = :parent_pl_id
      and coalesce(kst_from_pki_id, 0) = 0;
  end

  -- шаг 2. добавим остальное из КЛАДР
  for select pki_id
  from places_kladr_import
  where pki_parent = :parent_pki_id
    and right(pki_code, 2) = '00' -- только актуальные
    and not exists(select 1 from kladr_sync_tmp where kst_from_pki_id = pki_id)
  into pki_id
  do
  begin
    insert into kladr_sync_tmp (kst_from_pki_id)
    values(:pki_id);
  end

  -- шаг 3. обработаем
  for select kst_id, pki_id, pl_id, pki_type, left(pki_code, 11), substring(kst_pl_code from 4), pki_okatd, pl_add_code
  from places_kladr_import
  inner join kladr_sync_tmp on kst_from_pki_id = pki_id + 0
  left join places on pl_id = kst_from_pl_id
  where pki_parent = :parent_pki_id
  into kst_id, pki_id, pl_id, kladr_type, kladr_code, pl_code, kladr_okato, pl_okato
  do
  begin
    status = '';
    comm = '';
    need_edit = false;
    err = false;

    if (pl_id is null) then  -- Новый
    begin
      status = 'Новый.';
      comm = '';
  
      plt_id = json.obj_get_val(plt_jso, replace(kladr_type,'.','#'), json.obj_get_val(plt_jso, replace(kladr_type,'.','')));

      if (plt_id is null) then
      begin
        status = 'Ошибка создания.';
        comm = 'Не найден тип: ' || coalesce(kladr_type, '<NULL>');
      end
    end
    else
    begin
      if (pl_code is distinct from kladr_code) then
      begin
        need_edit = true;
        status = status || ' ~Код';
        comm = comm || f_format(' Код: %1 -> %2  ', pl_code, kladr_code);
      end 
  
      if (pl_okato is distinct from kladr_okato) then
      begin
        need_edit = true;
        status = status || ' ~ОКАТО';
        comm = comm || f_format(' ОКАТО: %1 -> %2  ', pl_okato, kladr_okato);
      end 
  
      if (err) then
        status = 'Ошибка изменеия:' || status;
      else if (need_edit) then
        status = 'Изменение:' || status;
      else
        status = 'ОК' || status;
    end

    update kladr_sync_tmp
    set kst_status = :status,
        kst_comm = :comm,
        kst_need_update = :need_edit
    where kst_id = :kst_id;

    select error_text from hp_places_kladr_sync(:pki_id, :pl_id, :kladr_okato, :plt_jso)
    into error_text;

    if (error_text <> '') then
      break;
  end


  update kladr_sync_main
  set ksm_status = 3
  where ksm_status = 2;

  suspend;
end
