CREATE PROCEDURE PAYSYS_PRINT_AKT_COMMON(
    PSM_ID ID_TYPE)
RETURNS (
    PSM_AKT_NO DECIMAL(18,0),
    REG_NAME VARCHAR(50),
    C_NAME_FIO_SRC VARCHAR(100),
    C_NAME_FIO_DST VARCHAR(100),
    JL_NAME_SRC VARCHAR(100),
    JL_NAME_DST VARCHAR(100),
    PSM_DATE DATE,
    PSL_COUNT INTEGER,
    OBJ_NAME_SRC VARCHAR(100),
    OBJ_KIND_SRC INTEGER,
    OBJ_KIND_TXT_SRC VARCHAR(100),
    OBJ_NAME_DST VARCHAR(100),
    OBJ_KIND_TXT_DST VARCHAR(100),
    OBJ_KIND_DST INTEGER)
AS
declare variable c_id_src decimal(18,0);
declare variable c_id_dst decimal(18,0);
declare variable obj_kind integer;
declare variable obj_id decimal(18,0);
begin
  select reg_name, psm_date, psm_from_c_source, psm_from_c_destin
  from paysys_main, region
  where psm_id = :psm_id and
        psm_from_reg_id = reg_unicode
  into reg_name, psm_date, c_id_src, c_id_dst;
  psm_akt_no = psm_id;
-- ФИО выдающего и получающего (через которых производится передача)
  c_name_fio_src = '';
  if (c_id_src is not null) then
    select c_name_fio
    from clients
    where c_id = :c_id_src
    into c_name_fio_src;

  c_name_fio_dst = '';
  if (c_id_dst is not null) then
    select c_name_fio
    from clients
    where c_id = :c_id_dst
    into c_name_fio_dst;

-- Должность выдающего/получающего сотрудника
  jl_name_src = '';
  select jl_name
  from hr_pstatus, joblist
  where ps_from_c_id = :c_id_src and
        ps_status = 1 and
        :psm_date between ps_date_begin and ps_date_end and
        ps_from_jl_id = jl_id
  into jl_name_src;

  select jl_name
  from hr_pstatus, joblist
  where ps_from_c_id = :c_id_dst and
        ps_status = 1 and
        :psm_date between ps_date_begin and ps_date_end and
        ps_from_jl_id = jl_id
  into jl_name_dst;

-- Получение данных отправителя отправителя
  select psl_object_id, psl_object_kind
  from paysys_list, paysys_item
  where psl_from_psm_id = :psm_id and
        psl_from_psi_id = psi_id and
        psl_sign = -1 rows 1
  into obj_id, obj_kind;
  obj_kind_src = obj_kind;
  if (obj_kind = 0) then
  begin
    obj_kind_txt_src = 'Проект';
    select psp_name
    from paysys_project
    where psp_id = :obj_id
    into obj_name_src;
  end
  else
  if (obj_kind = 1) then
  begin
    obj_kind_txt_src = 'Офис';
    select reg_name
    from region
    where reg_unicode = :obj_id
    into obj_name_src;
  end
  else
  if (obj_kind = 2) then
  begin
    obj_kind_txt_src = jl_name_src;
    obj_name_src = c_name_fio_src;
  end

-- Получение данных отправителя получателя
  select psl_object_id, psl_object_kind
  from paysys_list, paysys_item
  where psl_from_psm_id = :psm_id and
        psl_from_psi_id = psi_id and
        psl_sign = 1 rows 1
  into obj_id, obj_kind;
  obj_kind_dst = obj_kind;
  if (obj_kind = 0) then
  begin
    obj_kind_txt_dst = 'Проект';
    select psp_name
    from paysys_project
    where psp_id = :obj_id
    into obj_name_dst;
  end
  else
  if (obj_kind = 1) then
  begin
    obj_kind_txt_dst = 'Офис';
    select reg_name
    from region
    where reg_unicode = :obj_id
    into obj_name_dst;
  end
  else
  if (obj_kind = 2) then
  begin
    obj_kind_txt_dst = jl_name_dst;
    obj_name_dst = c_name_fio_dst;
  end

-- Количество передаваемых карт
  select count(*)
  from paysys_list
  where psl_from_psm_id = :psm_id and
        psl_sign = -1
  into psl_count;

  suspend;
end
