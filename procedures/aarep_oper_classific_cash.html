CREATE PROCEDURE AAREP_OPER_CLASSIFIC_CASH(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_GROUP_REP_2_1 VARCHAR(20),
    RES_OPERCODE_STR VARCHAR(20),
    REP_AMOUNT_PLUS CURR_TYPE,
    REP_AMOUNT_MINUS CURR_TYPE,
    RES_SALDO_BEGIN DECIMAL(18,4),
    RES_SALDO_END DECIMAL(18,4),
    ERR_TEXT VARCHAR(1024))
AS
declare variable tmp_summa decimal(18,4) = 0;
declare variable atl_acc_mask varchar(20);
declare variable al_id varchar(20);
begin
  err_text = '';
  if (not(exists(select *
                 from temp_moneyclassific
                 rows 1))) then
  begin
-- Отчет запускается первый раз, заполняем временную таблицу.
    select err_text
    from aarep_oper_classific_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
    into :err_text;
    if (err_text <> '') then
    begin
      suspend;
      exit;
    end
  end
-- Ниже данные будут браться из временной таблице

  res_saldo_begin = 0;
  res_saldo_end = 0;
  for select atl_acc_mask
      from acctypelist
      where atl_from_at_id = 11
      into :atl_acc_mask
  do
  begin
    for select al_id
        from acc_list, region_list(:rep_region_id, 1, -1, 1)
        where al_from_bg_id starting with :atl_acc_mask and
              al_from_reg_id = id_list
        into :al_id
    do
    begin
      tmp_summa = f_al_bal(al_id, rep_date_begin - 1);
      res_saldo_begin = res_saldo_begin + tmp_summa;

      tmp_summa = f_al_bal(al_id, rep_date_end);
      res_saldo_end = res_saldo_end + tmp_summa;
    end
  end

  for select res_group_rep_2_1, res_opercode_str, sum(iif(res_oper_sign = 1, res_amount, 0.0)), -- Зачисление Руб
             sum(iif(res_oper_sign = -1, res_amount, 0.0)) -- Списание Руб
      from temp_moneyclassific
      where res_is_bank = 0
      group by res_group_rep_2_1, res_opercode_str
      into :res_group_rep_2_1, res_opercode_str, :rep_amount_plus, :rep_amount_minus
  do
  begin
    suspend;
  end
end
