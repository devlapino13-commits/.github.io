CREATE PROCEDURE HP_MAKE_SCHEDULE(
    UN_ID DECIMAL(18,0),
    OBJ_KIND INTEGER,
    OBJ_ID DECIMAL(18,0),
    ID INTEGER,
    NEW_DATE DATE,
    NEW_GRACE INTEGER)
RETURNS (
    RES_ID INTEGER,
    RES_TXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable LOAN_DISB_DATE date;
declare variable LOAN_REPAY_DAY integer;
declare variable LOAN_REPAY_KOLICH integer;
declare variable LOAN_PRODUCT_ID integer;
declare variable LOAN_AMOUNT decimal(18,4) = 0;
declare variable LOAN_PERCENT decimal(18,7);
declare variable LOAN_KOMMIS_PERC decimal(18,7);
declare variable PR_PERCENT_FOR_DAYS integer;
declare variable PR_IS_365 integer;
declare variable PR_IS_WEEKEND_WORK integer;
declare variable CALC_KOMMIS_LGOTA decimal(18,4) = 0;
declare variable SH_DATE_PRIOR date;
declare variable PR_ROUND integer;
declare variable PR_ROUND_PERC integer;
declare variable PR_ROUND_KOMMIS integer;
declare variable PR_CODE varchar(100);
declare variable TOTAL_GRACE integer;
declare variable SH_ID integer;
declare variable SH_DATE date;
declare variable SH_DAYS integer;
declare variable SH_GRACE integer;
declare variable DD_1 date;
declare variable DD_2 date;
declare variable IS_CHECKING integer;
declare variable NEW_OBJ_ID decimal(18,0);
declare variable MAX_DD date;
declare variable SUM_PERC decimal(18,4) = 0;
declare variable PERC_BEGIN_ANNU decimal(18,7);
declare variable EFFECT_RATE decimal(18,7);
declare variable EFFECT_RATE_COMMON decimal(18,7);
declare variable DB_ID integer;
declare variable MIN_DAYS integer;
declare variable MAX_DAYS integer;
declare variable SH_GRACES varchar(255);
declare variable ADD_MONTH D_INTEGER;
declare variable CLI_ID D_ID;
declare variable REG_ID D_INTEGER;
declare variable ANOTHER_SCHEDULE D_SMALLINT;
declare variable FIRST_WEEKDAY_WEEKEND D_SMALLINT;
declare variable SECOND_WEEKDAY_WEEKEND D_SMALLINT;
declare variable KF_ID D_ID;
declare variable MA_INSUR_KIND D_SMALLINT;
declare variable CLIENT_SCORE D_INTEGER;
begin
  db_id = f_get_db_id();
  min_days = 15;
  max_days = 45;

  res_id = 1;
  res_txt = '';
  is_checking = 0;
  total_grace = 0;
  another_schedule = 0;

  if (obj_kind = 9) then
    new_obj_id = obj_id;
  else
  if (obj_kind = 11) then
  begin
    select k_id
    from kontract
    where k_from_ma_app_id = :obj_id
    order by k_disb_date desc
    rows 1
    into new_obj_id;
    if (new_obj_id is null) then
      new_obj_id = -:obj_id;
  end
  else
    exit;

  if (id = 0 and
      new_grace = -1 and
      obj_kind = 9) then
  begin
    is_checking = 1;
    id = null;
    select sh_id, sh_date, coalesce(sh_grace, 0)
    from schedule
    where sh_from_k_id = :new_obj_id and
          sh_id = 1
    into :id, new_date, new_grace;
    if (id is null) then
      exit;
  end
  if (new_grace = -2) then
    is_checking = -1; -- График трогать не будем, контрольно вычислим ставки и обновим кредит
-- Будет использовано при создании кредита из заявки
  reg_id = null;

  if (obj_kind = 9) then
  begin
    select k_disb_date, k_repay_day, k_kolich_repay, k_from_pr_id, k_amount_approved, k_percent, k_kommis_perc, k_add_month, k_from_kf_id
    from kontract
    where k_id = :obj_id
    into loan_disb_date, loan_repay_day, loan_repay_kolich, loan_product_id, loan_amount, loan_percent,
         loan_kommis_perc, add_month, kf_id;

    select coalesce(kal.kal_new_reg_id, kal.kal_prior_reg_id)
    from knaktlist kal
    where kal.kal_from_kf_id = :kf_id and
          :loan_disb_date between kal.kal_date and kal.kal_date_end and
          kal.is_delete = 0
    into reg_id;
  end
  if (obj_kind = 11) then
  begin
    select ma_date_oper, ma_repay_day, ma_kolich_repay, ma_from_pr_id, ma_summa_minus, ma_percent, ma_kommis_perc, ma_add_month, ma_office_id
    from miniapp
    where ma_id = :obj_id and
          ma_date_end >= '31.12.2999' and
          ma_is_active = 1
    into loan_disb_date, loan_repay_day, loan_repay_kolich, loan_product_id, loan_amount, loan_percent,
         loan_kommis_perc, add_month, reg_id;
  end

  if (coalesce(obj_id, 0) = 0) then
    exit;
  if (loan_product_id is null) then
    exit;

  if (is_checking >= 0) then
  begin
-- Получаем необходимые настройки кредитного продукта
    select pr_percent_for_days, pr_is_360, pr_round, pr_round_perc, pr_round_admin, pr_is_weekend_work, pr_code
    from product
    where pr_id = :loan_product_id
    into :pr_percent_for_days, :pr_is_365, :pr_round, :pr_round_perc, :pr_round_kommis, :pr_is_weekend_work, pr_code;

    pr_round = -pr_round;
    pr_round_perc = -pr_round_perc;
    pr_round_kommis = -pr_round_kommis;

    if (coalesce(pr_percent_for_days, 0) = 0) then
      exit;

    delete from temp_schedule
    where 1 = 1;

    --Для офиса Амана-Исламик Карасуу другой рабочий график
    --В будущем если запросов на изменение графика рабочих дней будет больше, то нужно создать таблицу с соответствующими полями
    if (db_id in (39801, 41701) and reg_id = 1280400) then
    begin
      another_schedule = 1;
      first_weekday_weekend = 0; --Выходной день воскресенье
    end 

-- Вычисляем сумму доп.процента с учетом предоставленных скидок
    calc_kommis_lgota = 0;
    if (obj_kind = 9) then
    begin
      select sum(kc_kommis_lgota)
      from knclient
      where kc_from_k_id = :obj_id and
            kc_amount_approved > 0
      into :calc_kommis_lgota;
    end
    if (obj_kind = 11) then
    begin
      select sum(mac_kommis_lgota)
      from miniappclient
      where mac_from_ma_id = :obj_id and
            mac_amount > 0
      into :calc_kommis_lgota;
    end
    if (calc_kommis_lgota is null) then
      calc_kommis_lgota = 0;

    if (id = 0) then -- Даты не передаются, вычисляем самостоятельно
    begin
      sh_graces = '';
      select list('[' || sh_id || ']')
      from schedule
      where sh_from_k_id = :new_obj_id and
            sh_grace = 1
      into :sh_graces;
      if (sh_graces is null) then
        sh_graces = '';

      if (db_id = 81001 and
          sh_graces = '') then
      begin
        if (not(exists(select *
                       from schedule
                       where sh_from_k_id = :new_obj_id))) then
          sh_graces = 'NEW';
      end

      execute procedure hp_make_schedule_date(:loan_amount, :loan_repay_kolich, :loan_repay_day, :loan_disb_date,
          :loan_product_id, :loan_kommis_perc, :calc_kommis_lgota, :sh_graces, 0, :add_month, :reg_id);
    end
    else -- передается дата для редактирования
    begin
      if (is_checking = 0) then
      begin
        select sh_date
        from schedule
        where sh_id = :id and
              sh_from_k_id = :new_obj_id
        into dd_2;

        if (db_id = 81001) then
        begin
          if (dd_2 <> new_date) then
          begin
-- В РФ запрещено изменение дат погашений
            res_id = -1;
            res_txt = 'Изменение дат погашений запрещено.';
            suspend;
            exit;
          end
        end else
        begin
          if (id = 1 and add_month <> 0 and dd_2 <> new_date) then
          begin
            -- Если первое погашение через 2-3 месяца
            res_id = -1;
            res_txt = 'Изменение первой даты погашения запрещено.';
            suspend;
            exit;
          end
        end

        if (db_id = 94401 and
            new_grace = 1) then
        begin
          -- RFCAZ-609: для продуктов 130-133, 142-145, 150-165 при балле >= 30 и сумме >= 4000
          -- разрешаем льготу до 6 платежей; иначе действует старое ограничение (только 1-2)
          if (loan_product_id between 130 and 133 or
              loan_product_id between 142 and 145 or
              loan_product_id between 150 and 165) then
          begin
            -- клиент для расчета балла
            cli_id = null;
            if (obj_kind = 9) then
            begin
              select kc_from_c_id
              from knclient
              where kc_from_k_id = :obj_id and
                    kc_amount_approved > 0 rows 1
              into :cli_id;
            end
            else if (obj_kind = 11) then
            begin
              select mac_from_c_id
              from miniappclient
              where mac_from_ma_id = :obj_id and
                    mac_amount > 0 rows 1
              into :cli_id;
            end
        
            client_score = f_point(:cli_id, :loan_disb_date);
        
            -- условия не выполнены -> старое ограничение (грейс только на 1-2)
            if ((client_score < 30 or
                loan_amount < 4000) and
                id > 2) then
            begin
              res_id = -1;
              res_txt = 'Льготное погашение не применимо'; -- RFCAZ-393
              suspend;
              exit;
            end
        
            -- условия выполнены, но грейс дальше 6-го платежа не даем
            if (id >= 6) then
            begin
              res_id = -1;
              res_txt = 'Льготный период не может превышать 6 месяцев'; -- RFCAZ-609
              suspend;
              exit;
            end
          end
          else
          begin
            -- прочие продукты: без изменений, грейс только на 1-2
            if (id > 2) then
            begin
              res_id = -1;
              res_txt = 'Льготное погашение не применимо'; -- RFCAZ-393
              suspend;
              exit;
            end
          end
        end

        if (db_id = 81001) then
        begin
          if (pr_code containing '.EMP.' and
              pr_code containing '.reliable_emp.' and
              pr_code containing '.IND.'and
              new_grace = 1) then
          begin
            res_id = -1;
            res_txt = 'Данный продукт не может иметь льготный период';
            suspend;
            exit;
          end
        end 

        if (id = loan_repay_kolich and
            new_grace = 1) then
        begin
          res_id = -1;
          res_txt = 'Последний платеж не может быть льготным';
          suspend;
          exit;
        end

        if (extract(weekday from :new_date) in (0, 6) and
            (pr_is_weekend_work = 0) and another_schedule = 0) then
        begin
          res_id = -1;
          res_txt = 'Погашения не должны попадать на выходные сб-вс.';
          suspend;
          exit;
        end

        if (another_schedule = 1) then
        begin
          if (extract(weekday from :new_date) = first_weekday_weekend) then
          begin
            res_id = -1;
            res_txt = 'Погашения не должны попадать на выходные вс.';
            suspend;
            exit;
          end 
        end 

        select sh_date
        from schedule
        where sh_id = (:id - 1) and
              sh_from_k_id = :new_obj_id
        into dd_1;

        dd_2 = new_date;
        if (dd_1 is not null) then
        begin
          if (dd_1 >= dd_2) then
          begin
            res_id = -1;
            res_txt = 'Неправильная последовательность дат.';
            suspend;
            exit;
          end
          if (datediff(day, dd_1, dd_2) < min_days) then
          begin
            res_id = -1;
            res_txt = 'Дни между платежами не может быть меньше ' || :min_days || ' дней';
            suspend;
            exit;
          end
          if (datediff(day, dd_1, dd_2) > max_days) then
          begin
            if (db_id = 81001) then
            begin
              -- Если платежи попадают между 9-12 января или 1-11 мая, НЕ выводим сообщение о несоответствии по максимальным дням между платежами
              if (not((extract(month from dd_2) = 1 and extract(day from dd_2) between 9 and 12) or
                  (extract(month from dd_2) = 5 and extract(day from dd_2) between 1 and 11)))  then
              begin
                res_id = -1;
                res_txt = 'Дни между платежами не может быть больше ' || max_days || ' дней';
                suspend;
                exit;
              end
            end
            else
            begin
              res_id = -1;
              res_txt = 'Дни между платежами не может быть больше ' || max_days || ' дней';
              suspend;
              exit;
            end
          end
        end
        if (id < loan_repay_kolich) then
        begin
          select sh_date
          from schedule
          where sh_id = (:id + 1) and
                sh_from_k_id = :new_obj_id
          into dd_2;
          dd_1 = new_date;
          if (dd_2 is not null) then
          begin
            if (dd_1 >= dd_2) then
            begin
              res_id = -1;
              res_txt = 'Неправильная последовательность дат.';
              suspend;
              exit;
            end
            if (datediff(day, dd_1, dd_2) < min_days) then
            begin
              res_id = -1;
              res_txt = 'Дни между платежами не может быть меньше ' || min_days || ' дней';
              suspend;
              exit;
            end
            if (datediff(day, dd_1, dd_2) > max_days) then
            begin
              if (db_id = 81001) then
              begin
                if (not((extract(month from dd_2) = 1 and extract(day from dd_2) between 9 and 12) or
                    (extract(month from dd_2) = 5 and extract(day from dd_2) between 1 and 11)))  then
                begin
                  res_id = -1;
                  res_txt = 'Дни между платежами не может быть больше ' || max_days || ' дней';
                  suspend;
                  exit;
                end
              end
              else
              begin
                res_id = -1;
                res_txt = 'Дни между платежами не может быть больше ' || max_days || ' дней';
                suspend;
                exit;
              end
            end
          end
        end
        else
        begin
          select sh_date
          from schedule
          where sh_id = :id and
                sh_from_k_id = :new_obj_id
          into dd_2;
          if (dd_2 < new_date) then
          begin
            res_id = -1;
            res_txt = 'Нельзя увеличивать срок последнего платежа';
            suspend;
            exit;
          end
        end
      end
      insert into temp_schedule (sh_id, sh_date, sh_os, sh_perc, sh_kommis, sh_grace)
      select sh_id, sh_date, sh_os, sh_perc, sh_kommis, sh_grace
      from schedule
      where sh_from_k_id = :new_obj_id
      order by sh_date, sh_id;

      update temp_schedule
      set sh_date = :new_date,
          sh_grace = :new_grace
      where sh_id = :id;

      sh_date_prior = loan_disb_date;
      total_grace = 0;
      for select sh_id, sh_date, sh_grace
          from temp_schedule
          where sh_id > 0
          order by sh_id
          into :sh_id, sh_date, sh_grace
      do
      begin
        sh_days = (select res_days
                   from fhp_agein360(:sh_date_prior, :sh_date, :pr_is_365));
        update temp_schedule
        set sh_days = :sh_days
        where sh_id = :sh_id;

        total_grace = total_grace + sh_grace;
        sh_date_prior = sh_date;
      end
    end

    cli_id = null;
    if (obj_kind = 9) then
    begin
      select kc_from_c_id
      from knclient
      where kc_from_k_id = :obj_id and
            kc_amount_approved > 0
      rows 1
      into cli_id;
    end else
    if (obj_kind = 11) then
    begin
      select mac_from_c_id
      from miniappclient
      where mac_from_ma_id = :obj_id and
            mac_amount > 0
      rows 1
      into cli_id;
    end

    if (db_id = 41701) then
    begin
      if (obj_kind = 9) then
      begin
        select first 1 ma.ma_insur_kind
        from kontract k
        inner join miniapp ma on k.k_from_ma_app_id = ma.ma_id
        where k_id = :obj_id and
              ma.ma_date_end >= '31.12.2999' and
              ma.ma_is_active = 1
        order by ma.ma_date_app desc
        into :ma_insur_kind;
      end
      else if (obj_kind = 11) then
      begin
        select first 1 ma.ma_insur_kind
        from miniapp ma
        where ma.ma_id = :obj_id and
              ma.ma_date_end >= '31.12.2999' and
              ma.ma_is_active = 1
        order by ma.ma_date_app desc
        into :ma_insur_kind;
      end
    end

    execute procedure hp_make_schedule_perc(:loan_amount, :loan_repay_kolich, :loan_product_id, :loan_percent,
        :loan_kommis_perc, :calc_kommis_lgota, :add_month, :cli_id, 0, coalesce(:ma_insur_kind, 0));

    if (is_checking = 0) then
    begin
-- показываем график, далее уберем это.
      delete from schedule
      where sh_from_k_id = :new_obj_id;

      insert into schedule (sh_from_k_id, sh_id, sh_date, sh_os, sh_perc, sh_kommis, sh_disb_amount, sh_grace)
      select :new_obj_id, sh_id, sh_date, 0, 0, sh_kommis, :loan_amount, 0
      from temp_schedule
      where sh_id = 0;

      insert into schedule (sh_from_k_id, sh_id, sh_date, sh_os, sh_perc, sh_kommis, sh_grace, sh_kommis_monthly)
      select :new_obj_id, sh_id, sh_date, sh_os, sh_perc, sh_kommis, sh_grace, sh_kommis_monthly
      from temp_schedule
      where sh_id > 0;

      if (new_obj_id > 0) then
        execute procedure hp_kontmat_sell_update(:new_obj_id);
    end
  end

  if (is_checking in (0, -1)) then
  begin
    select max(sh_date), sum(sh_perc), sum(iif(sh_os = 0, 1, 0))
    from schedule
    where sh_from_k_id = :new_obj_id and
          sh_id > 0
    into :max_dd, sum_perc, total_grace;

    effect_rate = 0.0;
    effect_rate_common = 0.0;
    perc_begin_annu = 0.0;
    if (sum_perc is null) then
      sum_perc = 0;

    select effect_rate, effect_rate_common
    from hp_effect_rate_common(:new_obj_id, null)
    into :effect_rate, :effect_rate_common;

    if (loan_disb_date < max_dd and
        loan_amount <> 0) then
      perc_begin_annu = round(((sum_perc / cast(datediff(day, :loan_disb_date, :max_dd) as double precision)) * 365.0000) / :loan_amount, 4);

    update kontract
    set k_kontract_no = '', -- переформирование номера договора так как папка может быть передан другому офису
        k_kredline_end = :max_dd,
        k_kont_last_date = :max_dd,
        k_kolich_lgota = coalesce(:total_grace, 0),
        k_percent_begin_annual = :perc_begin_annu,
        k_percent_effect = :effect_rate_common,
        k_percent_effect_fifd = :effect_rate,
        k_fix_percent_amount = :sum_perc
    where k_id = :new_obj_id;
  end
  else
  begin
-- значит (is_checking=1)
-- Сверка двух графиков
    res_id = 1;
    res_txt = '';
    if (exists(select *
               from schedule sh
               left join temp_schedule tsh on (sh.sh_id = tsh.sh_id)
               where sh.sh_from_k_id = :new_obj_id and
                     ((abs(coalesce(sh.sh_os, -1) - coalesce(tsh.sh_os, -1)) > 2) or
                     (abs(coalesce(sh.sh_perc, -1) - coalesce(tsh.sh_perc, -1)) > 2)))) then
    begin
      res_id = -1;
      res_txt = new_obj_id;
    end
  end
  suspend;

  delete from temp_schedule
  where 1 = 1;
end
