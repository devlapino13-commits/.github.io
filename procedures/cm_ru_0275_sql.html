CREATE PROCEDURE CM_RU_0275_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable ma_id d_id;
declare variable c_citizen d_integer;
declare variable c_citizen_str d_str_255;
declare variable status_code d_integer;
declare variable error_text d_str_255;
begin
  ma_id = (select k_from_ma_app_id from kontract where k_id = :k_id);

  for select kc_from_c_id, c_citizen
  from knclient, clients
  where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    and kc_from_c_id = c_id
  into c_id, c_citizen
  do
  begin
    c_citizen_str = '';
    status_code = 0;
    error_text = '';

    select coalesce(rd_code, '')
    from refdata
    where rd_id = :c_citizen
    into c_citizen_str;

    if (c_citizen_str <> 'Citizen.RU') then
      continue;

    select coalesce(status_code, 0), error_text
    from hp_smevpass_get_result(:c_id, :ma_id, current_date)
    into status_code, error_text;

    if (error_text <> '') then
    begin
      res_error_is_critical = 2;
      res_error_text = 'СМЭВ. Проверка паспорта: ' || error_text;
      suspend;
      exit;
    end

    if (status_code <> 300) then
    begin
      res_error_text = 'Действительность паспорта не подтверждена. Откажите в заявке';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
