CREATE PROCEDURE AAREP_RETURN_KOEF_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    PR_COMMON_NAME VARCHAR(255),
    AAREP_REPAY_DATE DATE,
    AAREP_PLAN_OS DECIMAL(18,4),
    AAREP_PLAN_PERC DECIMAL(18,4),
    AAREP_REG_ID INTEGER,
    AAREP_LO_ID DECIMAL(18,0),
    AAREP_REPAY_YEAR_MONTH INTEGER,
    AAREP_FACT_OS DECIMAL(18,4),
    AAREP_FACT_PERC DECIMAL(18,4),
    AAREP_REPAY_WEEK INTEGER,
    AAREP_BALANCE_END DECIMAL(18,4),
    AAREP_LOAN_AMOUNT DECIMAL(18,4),
    AAREP_BALANCE_BEGIN DECIMAL(18,4),
    AAREP_DELINQ180_BEGIN DECIMAL(18,4),
    AAREP_DELINQ180_END DECIMAL(18,4),
    AAREP_DELINQ60_BEGIN DECIMAL(18,4),
    AAREP_DELINQ60_END DECIMAL(18,4))
AS
declare variable dd_disb_date date;
declare variable dd date;
declare variable aarep_loan_amount_tmp integer;
declare variable dd_tmp date;
begin
  for select distinct ks_from_k_id, kal_new_reg_id, kal_new_kf_officer, ks_date_begin, pr_common_name,
                      k_amount_approved
      from kontstate, kontract, knaktlist, product
      where ks_ol_is_delete = 0 and
            ks_date_end >= (:rep_date_begin - 1) and
            ks_date_begin <= :rep_date_end and
            ks_state = 1 and
            kal_from_kf_id = k_from_kf_id and
            ks_from_k_id = k_id and
            kal_date <= maxvalue(:rep_date_begin, ks_date_begin) and
            kal_date_end >= maxvalue(:rep_date_begin, ks_date_begin) and
            knaktlist.is_delete = 0 and
            k_from_pr_id = pr_id
      into :k_id, :aarep_reg_id, :aarep_lo_id, :dd_disb_date, :pr_common_name, :aarep_loan_amount_tmp
  do
  begin
    dd = maxvalue(:rep_date_begin, (dd_disb_date + 1));
    aarep_fact_os = 0;
    aarep_fact_perc = 0;
    aarep_balance_end = 0;
    aarep_balance_begin = 0;
    aarep_loan_amount = 0;

    aarep_delinq180_begin = 0;
    aarep_delinq180_end = 0;
    aarep_delinq60_begin = 0;
    aarep_delinq60_end = 0;

    for select ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date < :dd and
              ok_sh_date between :rep_date_begin and :rep_date_end and
              ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_from_c_id = 0
        group by ok_sh_id, ok_sh_date
        having(sum(ok_os + ok_sh_os) <> 0) or (sum(ok_perc + ok_sh_perc) <> 0)
        into :aarep_repay_date, :aarep_plan_os, :aarep_plan_perc
    do
    begin
      aarep_repay_year_month = extract(year from :aarep_repay_date) * 100 + extract(month from :aarep_repay_date);
      aarep_repay_week = extract(year from :aarep_repay_date) * 100 + extract(week from :aarep_repay_date);
      suspend;
    end

    aarep_plan_os = 0;
    aarep_plan_perc = 0;
    aarep_balance_end = 0;
    aarep_balance_begin = 0;
    aarep_loan_amount = 0;

    for select ok_ol_date, -sum(ok_os), -sum(ok_perc)
        from operkred
        where ok_from_k_id = :k_id and
              ok_ol_date between :rep_date_begin and :rep_date_end and
              ok_sh_date between :rep_date_begin and :rep_date_end and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_from_c_id = 0
        group by ok_ol_date
        having(sum(ok_os) <> 0) or (sum(ok_perc) <> 0)
        into :aarep_repay_date, :aarep_fact_os, :aarep_fact_perc
    do
    begin
      aarep_repay_year_month = extract(year from :aarep_repay_date) * 100 + extract(month from :aarep_repay_date);
      aarep_repay_week = extract(year from :aarep_repay_date) * 100 + extract(week from :aarep_repay_date);
      suspend;
    end

    aarep_plan_os = 0;
    aarep_plan_perc = 0;
    aarep_balance_end = 0;
    aarep_balance_begin = 0;
    aarep_loan_amount = 0;
    aarep_fact_os = 0;
    aarep_fact_perc = 0;
    aarep_delinq180_begin = 0;
    aarep_delinq180_end = 0;
    aarep_delinq60_begin = 0;
    aarep_delinq60_end = 0;

    aarep_loan_amount = aarep_loan_amount_tmp;
    dd_tmp = null;
    select oks_balance, oks_delinq_date
    from oksummary
    where oks_date_begin <= (:rep_date_begin - 1) and
          oks_date_end >= (:rep_date_begin - 1) and
          oks_from_k_id = :k_id and
          oks_balance > 0 and
          oks_close_state = 0
    into :aarep_balance_begin, dd_tmp;
    if (((:rep_date_begin - 1) - dd_tmp) > 60) then
      aarep_delinq60_begin = aarep_balance_begin;
    if (((:rep_date_begin - 1) - dd_tmp) > 180) then
      aarep_delinq180_begin = aarep_balance_begin;

    select oks_balance, oks_delinq_date
    from oksummary
    where oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_from_k_id = :k_id and
          oks_balance > 0 and
          oks_close_state = 0
    into :aarep_balance_end, dd_tmp;
    if ((:rep_date_end - dd_tmp) > 60) then
      aarep_delinq60_end = aarep_balance_end;
    if ((:rep_date_end - dd_tmp) > 180) then
      aarep_delinq180_end = aarep_balance_end;

    suspend;
    aarep_balance_end = 0;
    aarep_balance_begin = 0;
  end
end
