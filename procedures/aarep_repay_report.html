CREATE PROCEDURE AAREP_REPAY_REPORT(
    REP_REGION_ID DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OL_ID DECIMAL(18,0),
    PLAN_OS DECIMAL(18,4),
    PLAN_PERC DECIMAL(18,4),
    REPAY_FIO VARCHAR(255),
    KF_NAME VARCHAR(100),
    FACT_DATE DATE,
    FACT_OS DECIMAL(18,4),
    FACT_PERC DECIMAL(18,4),
    FACT_FINE DECIMAL(18,4),
    K_KONTRACT_NO VARCHAR(20),
    CASHIER_FIO VARCHAR(100),
    REPAY_MEMO VARCHAR(100),
    PLAN_DATE_STR VARCHAR(100),
    REG_NAME VARCHAR(100),
    OL_DOC_NO INTEGER)
AS
declare variable plan_max_date date;
declare variable plan_date date;
declare variable k_id decimal(18,0);
begin
  for select max(fact_date), reg_name, max(ol_doc_no), kf_name, k_id, max(ol_id), k_kontract_no, min(plan_date),
             max(plan_date), sum(plan_os), sum(plan_perc), max(repay_fio), sum(fact_os), sum(fact_perc),
             sum(fact_fine)
      from aarep_repay_report_tmp(:rep_region_id, :rep_date_begin, :rep_date_end)
      group by kf_name, k_id, k_kontract_no, reg_name, k_id
      order by 1, 2, 3, kf_name
      into :fact_date, :reg_name, ol_doc_no, :kf_name, :k_id, :ol_id, :k_kontract_no, :plan_date, :plan_max_date,
           :plan_os, :plan_perc, :repay_fio, :fact_os, :fact_perc, :fact_fine

  do
  begin
    cashier_fio = '';
    select c_name_fio
    from clients, knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          kc_status = 2 and
          kc_from_c_id = c_id
    into :cashier_fio;
    if (cashier_fio is null) then
      cashier_fio = '';
    if (cashier_fio = '') then
    begin
      select c_name_fio
      from clients, knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id rows 1
      into :cashier_fio;
    end
    if (plan_date <> plan_max_date) then
    begin
      plan_date_str = right('00' || extract(day from plan_date), 2) || '.' || right('00' || extract(month from plan_date), 2) || '.' || extract(year from plan_date);
      plan_date_str = plan_date_str || '-' || right('00' || extract(day from plan_max_date), 2) || '.' || right('00' || extract(month from plan_max_date), 2) || '.' || extract(year from plan_max_date);
    end
    else
    begin
      plan_date_str = right('00' || extract(day from plan_date), 2) || '.' || right('00' || extract(month from plan_date), 2) || '.' || extract(year from plan_date);
    end
    repay_memo = '';
    if (ol_id = 0) then
      repay_memo = 'Платеж не произведен';
    else
    begin
      if (fact_date = plan_date) then
      begin
        if (fact_date = plan_max_date) then
          repay_memo = 'По графику';
        if (fact_date < plan_max_date) then
          repay_memo = 'По графику с предоплатой';
      end
      if (fact_date > plan_date) then
      begin
        if (fact_date = plan_max_date) then
          repay_memo = 'Оплата просрочки и текущего';
        if (fact_date < plan_max_date) then
          repay_memo = 'Оплата просрочки и предоплата';
        if (fact_date > plan_max_date) then
          repay_memo = 'Оплата просрочки';
      end
      if (fact_date < plan_date) then
        repay_memo = 'Предоплата';
    end
    if (exists(select *
               from kontstate
               where ks_from_k_id = :k_id and
                     ks_state = 2 and
                     ks_ol_is_delete = 0 and
                     ks_date_begin >= :rep_date_end)) then
      repay_memo = 'Кредит закрыт. ' || repay_memo;

    if (ol_doc_no = 0) then
      ol_doc_no = null;
    suspend;
  end
end
