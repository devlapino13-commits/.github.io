CREATE PROCEDURE HP_PAYSYS_GET_CL_CARDINFO(
    C_ID DECIMAL(18,0))
RETURNS (
    PSM_ID DECIMAL(18,0),
    PSL_ID DECIMAL(18,0),
    PSI_ID DECIMAL(18,0),
    PSI_CODE VARCHAR(20),
    PSM_DATE DATE,
    PSP_NAME VARCHAR(255))
AS
declare variable psp_id integer;
begin
-- Ищем все закрепления банковских карт за клиентом
  psm_id = null;
  psl_id = null;
  psi_id = null;
  psi_code = null;
  for select psi_id, psi_code, psp_name
      from paysys_main, paysys_list, paysys_item, paysys_project
      where psl_from_psm_id = psm_id and
            psl_from_psi_id = psi_id and
            psl_object_kind = 5 -- Клиент/Сотрудник (Clients)
            and
            psp_is_card = 1 -- банковская карта, привязка к клиенту
            and
            psm_is_delete = 0 and
            psi_from_psp_id = psp_id and
            psl_object_id = :c_id
      group by psi_id, psi_code, psp_name
      having sum(psl_sign) = 1
      into psi_id, psi_code, psp_name
  do
  begin
    select psl_id, psl_from_psm_id, psl_date
    from paysys_list
    where psl_from_psi_id = :psi_id and
          psl_sign = 1
    order by psl_from_psm_id desc
    rows 1
    into psl_id, psm_id, psm_date;

    suspend;
  end
end
