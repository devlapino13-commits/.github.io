CREATE PROCEDURE HP_DO_CLIENT_ACCESS_BLOCK(
    C_ID DECIMAL(18,0),
    BLOCK_DATE DATE,
    BLOCK_UN_ID DECIMAL(18,0))
AS
declare variable ACCESS_KIND integer;
declare variable PA_ID integer;
declare variable UNIQTAB_USERNAME integer;
declare variable UNIQTAB_PERMACTIONS integer;
declare variable MAX_DATE date;
declare variable UN_ID decimal(18,0);
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('PermActions')
  into :uniqtab_permactions;

  if (block_date is null) then
    block_date = current_date;

  for select un_id
      from username
      where un_from_c_id = :c_id
      into :un_id
  do
  begin
    update username
    set un_google_id = ''
    where (un_id = :un_id);

    access_kind = 0;
    for select pa_id, pm_access_kind
        from permactions, permissions
        where pa_parent_id is null and
              pm_object_kind = :uniqtab_username and
              pm_object_id = :un_id and
              pm_subject_kind = :uniqtab_permactions and
              pm_subject_id = pa_id and
              pm_is_delete = 0 and
              (coalesce(pm_date_begin, date '01.01.2000') <= :block_date) and
              (coalesce(pm_date_end, date '31.12.2999') >= :block_date)
        into :pa_id, :access_kind
    do
    begin
      if (access_kind = 1) then
      begin
        max_date = null;
        select max(pm_date_begin)
        from permactions, permissions
        where pa_parent_id is null and
              pm_object_kind = :uniqtab_username and
              pm_object_id = :un_id and
              pm_subject_kind = :uniqtab_permactions and
              pm_subject_id = pa_id and
              pm_is_delete = 0 and
              (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
              (coalesce(pm_date_end, '31.12.2999') >= current_date)
        into :max_date;

        if (max_date is not null and
            max_date > block_date) then
          block_date = current_date;

        insert into permissions (pm_id, pm_object_kind, pm_object_id, pm_subject_kind, pm_subject_id, pm_date_begin,
                                 pm_access_kind, pm_is_delete, pm_from_un_id)
        values (-1, :uniqtab_username, :un_id, :uniqtab_permactions, :pa_id, :block_date, -1, 0, :block_un_id);
      end
    end
  end
end
