CREATE PROCEDURE HP_EXPRESSAPP_STATUS_INFO(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_100,
    RES_HTML D_STR_LONG)
AS
declare variable OBJ_KIND_STR D_STR_CODE;
declare variable EXA_WAIT_STATUS D_INTEGER;
declare variable WAIT_TEXT D_STR_LONG;
declare variable EXC_NAME D_STR_100;
declare variable EXA_DATETIME_STR D_STR_CODE;
declare variable LANG D_STR_10;
declare variable EXA_FINAL_STATUS D_INTEGER;
declare variable EXC_ACTION_CODE D_STR_CODE;
declare variable EXA_ID D_INTEGER;
begin
  obj_kind_str = f_objkind(obj_kind, 'code');
  obj_kind = f_objkind(obj_kind, 'id');

  res_text= '';
  res_html = '';

  exa_wait_status = null;
  wait_text = '';

  if (obj_kind_str = 'kontract') then
  begin
    select exa_wait_status, exa_final_status, exc_name, f_datetimetostr(exa_datetime), exc_action_code, exa_id
    from expressapp, expresscfg
    where exa_from_k_id = :obj_id and
          exa_from_exc_id = exc_id
    order by exa_id desc
    rows 1
    into exa_wait_status, exa_final_status, exc_name, exa_datetime_str, exc_action_code, exa_id;
  end
  if (obj_kind_str = 'miniapp') then
  begin
    select exa_wait_status, exa_final_status, exc_name, f_datetimetostr(exa_datetime), exc_action_code, exa_id
    from expressapp, expresscfg
    where exa_from_ma_id = :obj_id and
          exa_from_exc_id = exc_id
    order by exa_id desc
    rows 1
    into exa_wait_status, exa_final_status, exc_name, exa_datetime_str, exc_action_code, exa_id;
  end
  if (obj_kind_str = 'expressapp') then
  begin
    select exa_wait_status, exa_final_status, exc_name, f_datetimetostr(exa_datetime), exc_action_code, exa_id
    from expressapp, expresscfg
    where exa_id = :obj_id and
          exa_from_exc_id = exc_id
    into exa_wait_status, exa_final_status, exc_name, exa_datetime_str, exc_action_code, exa_id;
  end

--   Финальный статус экспресс заявки exa_final_status:
--   -2 Отказ от клиента
--   -1 - Отменен в компании
--    0 - Новая запись
--    1 - Клиент дал добро и процесс начат
--    2 - Выдан

  if (exa_wait_status is not null) then
  begin
    lang = f_get_lang_code(un_id, null);

    res_text = case exa_wait_status
      when -1 then 'Робот &#128736;' -- черный   &#128736; Молоток и гаечный ключ, настройка
      when  0 then 'Робот &#9989;'   -- зеленый  &#9989; Зеленая галочка
      when  1 then 'Робот &#128293;' -- Красный &#128293; Огонек
      when  2 then 'Робот &#9203;'   -- Желтый  &#9203; песочные часы
    end;
    res_text = trim(res_text);

    if (exa_final_status  in (0, 1)) then
    begin
      wait_text = case exa_wait_status
        when -1 then '&#128736; Ошибка, отправлено разработчикам' -- черный   &#128736; Молоток и гаечный ключ, настройка
        when  0 then '&#9989; В обработке'   -- зеленый  &#9989; Зеленая галочка
        when  1 then '&#128293; Требуется ваше участие2' -- Красный &#128293; Огонек
        when  2 then '&#9203; В ожидании'   -- Желтый  &#9203; песочные часы
      end;
      wait_text = trim(wait_text);

      res_html = res_html || '<font color="#02b316"><b>' || f_lang(lang, 'Заявку ведет робот.') || '</b></font><br>';
    end else
    begin
      wait_text = case exa_final_status
        when -2 then '&#10006; Клиент отказал'
        when -1 then '&#10060; Отказ от компании'
        when  2 then '&#9989; Займ выдан'
      end;
      wait_text = trim(wait_text);

      res_html = res_html || '<font color="#02b316"><b>' || f_lang(lang, 'Заявка обработана роботом.') || '</b></font><br>';
    end

    if (exc_action_code = 'loan.next' ) then
    begin
      if (exa_wait_status = 1) then
      begin
        wait_text = '&#128293; Перед тем как подтвердить проверьте все условии: ';
        wait_text = blob_append(wait_text, f_uri_sql('Подвтердить', 'approved_exa_process', '60', exa_id, 'none',
                              'Подтвердить?', '', 0, 255, '', '', 'dialog', ''));
      end else
      if (exa_wait_status = 0) then
      begin
        wait_text = '&#9989; Подтврежден, в течении 5 минут процесс продолжиться.';
      end
    end

    res_html = res_html || exa_datetime_str || ', статус:' || '<br>';
    res_html = res_html || '<b>' || exc_name || '</b><br>';
    res_html = res_html || trim(wait_text) || '<br><br>';
  end

  suspend;
end
