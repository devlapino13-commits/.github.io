CREATE PROCEDURE AAREP_BLANK_LIST(
    BLANK_ID INTEGER)
RETURNS (
    BLN_ID INTEGER,
    BLN_NAME VARCHAR(255),
    BLN_MEMO VARCHAR(255),
    BLN_IS_ACTIVE SMALLINT,
    BLN_KIND INTEGER,
    BLN_SHOW_MODE INTEGER,
    BLN_IS_READ_ONLY INTEGER,
    BLB_NUMBER INTEGER,
    BLB_DATE_BEGIN DATE,
    BLB_DATE_END DATE,
    BLB_VERSION INTEGER,
    BLB_PARAMS VARCHAR(255),
    BLB_FRX_STATUS INTEGER,
    BLB_FONT_STYLE VARCHAR(100),
    BLB_FONT_SIZE INTEGER,
    BLN_CAN_EXPORT INTEGER,
    BLN_PRINT_STATUS VARCHAR(20))
AS
begin
  for select bln_id, bln_name, bln_memo, iif(blb_blob is null and blb_frx_blob is null, 0, 1), bln_kind, bln_show_mode,
             bln_is_read_only, bln_can_export, bln_print_status, blb_number, blb_date_begin, blb_date_end, blb_version,
             blb_params, blb_frx_status, blb_font_style, blb_font_size
      from blankblob, blankname
      where bln_id = blb_from_bln_id and
            (:blank_id = 0 or
            (bln_id = :blank_id and
            :blank_id <> 0))
      order by bln_id, blb_number
      into :bln_id, :bln_name, :bln_memo, :bln_is_active, :bln_kind, :bln_show_mode, :bln_is_read_only, :bln_can_export,
           :bln_print_status, :blb_number, :blb_date_begin, :blb_date_end, :blb_version, :blb_params, :blb_frx_status,
           :blb_font_style, :blb_font_size
  do
  begin
    if (blb_font_style is null) then
      blb_font_style = '';
    suspend;
  end
end
