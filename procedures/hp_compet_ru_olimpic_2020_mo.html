CREATE PROCEDURE HP_COMPET_RU_OLIMPIC_2020_MO(
    CPP_ID INTEGER,
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT VARCHAR(255))
AS
declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer = 0;
declare variable rd_id integer;
declare variable cpp_orderno integer;
declare variable res_period_id integer;
declare variable res_team_id_prior decimal(18,0);
declare variable reg_id integer;
declare variable ad_active_balance decimal(18,4) = 0;
declare variable ad_begin_active_balance decimal(18,4) = 0;
declare variable points_1 decimal(18,4) = 0;
declare variable points_2 decimal(18,4) = 0;
declare variable points_3 curr_type = 0;
declare variable points_4 curr_type = 0;
declare variable points_5 curr_type = 0;
declare variable points_6 curr_type = 0;
declare variable points_all decimal(18,4) = 0;
declare variable cpr_id integer = 0;
declare variable cpr_group_name varchar(255) = '';
declare variable cpd_comment d_str_4096;
declare variable total_points decimal(18,4) = 0;
declare variable add_points decimal(18,4) = 0;
declare variable group_name varchar(255) = '';
declare variable kol_svz integer = 0;
declare variable reg_name varchar(100) = '';
declare variable res_team_id decimal(18,0) = 0;
declare variable res_total1 integer = 0;
declare variable ad_delinq_balance_7 curr_type;
declare variable ad_delinq_balance_7_beg curr_type;
declare variable par7 d_percent;
declare variable par7_begin d_percent;
declare variable par7_diff d_percent;
declare variable date_end_prv date;
declare variable date_begin_prv date;
declare variable zarabotok_end curr_type;
declare variable zarabotok_begin curr_type;
declare variable zarabotok_diff curr_type;
declare variable rank_point_4 integer;
declare variable max_count integer;
declare variable ad_disb_loan curr_type;
declare variable avg_plan_disb_svz d_percent;
declare variable res_qual_koef d_currency;
declare variable count_reg integer;
declare variable avg_res_qual_koef d_currency;
declare variable rank_point_5 integer;
declare variable rep_indicator integer;
declare variable rep_period integer;
declare variable sum_plan d_currency;
declare variable sum_fact d_currency;
declare variable sum_ost d_currency;
declare variable par30 d_percent;
declare variable notdel_portfel_begin d_currency;
declare variable notdel_portfel_end d_currency;
declare variable growth_notdelinq d_currency;
declare variable super_pokazatel d_currency;
declare variable kind integer;
declare variable cpr_final_points decimal(18,4) = 0;
declare variable reg_disb_1_15 d_currency;
declare variable save_point smallint;
declare variable avg_disb_amount d_percent;
declare variable res_lo_id d_id;
declare variable plan_sum d_currency;
declare variable disb_sum d_currency;
declare variable dosroch_sum d_currency;
declare variable avg_plan_disb_lo d_percent;
declare variable lo_qty_from_plan d_integer;
declare variable par30_balance_begin d_currency;
declare variable par30_balance_end d_currency;
declare variable rank_point_9 integer;
declare variable count_lo_1_15 integer;
begin
  -- Расчет итогов олимпиады РФ 2020
  -- Stage:
  -- 0 - период (месяц)
  -- 1 - Промежуточные итоги
  -- 2 - Итоги
  -- 3 - Финальные результаты

  cpd_comment = '';
  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  date_end_prv = res_date_begin - 1;
  date_begin_prv = date_end_prv - extract(day from date_end_prv) + 1;

  if (stage = 0) then -- Если уровень МЕСЯЦ
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    -- Формируем в памяти данные по РФ
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'standart')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID Региона
    res_team_id_prior = -1;
    for select res_field_1, res_field_2, --
               sum(ad_delinq_par_begin2), -- par7_begin
               sum(ad_delinq_par_end2), -- par7
               sum(ad_disb_loan) ad_disb_loan, --
               sum(ad_delinq_balance_diff3) par30, --
               sum(ad_delinq_balance_begin3) par30, --
               sum(ad_delinq_balance_end3) par30 --

        from hp_analitica_report('reg_id, reg_name', 'tree_group', 8, 360, 0, 360, 8, 360, 31, 360, 181, 360, 360,
        --                             -1
        9, 'standart')
        group by 1, 2
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        into :reg_id, -- ID офиса
             :reg_name, -- наименование офиса
             :par7_begin, --
             :par7, --
             :ad_disb_loan, --
             :par30, --
             :par30_balance_begin, --
             :par30_balance_end

    do
    begin
      par7_diff = -(par7 - par7_begin);

      --    кол-во СВЗ
      kol_svz = null;
      select count(*)
      from hr_pstatus, joblist, jobkind, clients
      where ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.LO_Standart' and -- , 'JobSystem.LO_Partner')
            :res_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_id and
            ps_from_c_id = c_id and
            coalesce((select sawt_id
                      from hr_voctravel, hr_orders, sa_worktype -- убираем декретников и неоплачиваемых отпускников
                      where hrv_from_ho_id = ho_id and
                            ho_order_type = sawt_id and
                            hrv_from_c_id = ps_from_c_id and
                            :res_date_end between hrv_date_begin and hrv_date_end_fact
                      order by hrv_date_begin desc
                      rows 1), 0) not in (16, 17, 18, 21) and -- Убрал 8 - неоплачиваемый отпуск
            exists(select *
                   from oksummary -- берем только тех у кого на конец имеется активный портфель
                   where oks_kc_id = ps_from_c_id and
                         oks_region_id = ps_from_reg_id and
                         :res_date_end between oks_date_begin and oks_date_end and
                         oks_balance > 0)
      into kol_svz;
      kol_svz = coalesce(kol_svz, 0);

      if (kol_svz = 0) then
        kol_svz = 1; -- если в офисе не имеется СВЗ то будем считать 1 СВЗ
      --    Разработка по рейтингу
      --    Вычисляем балы
      points_1 = 0.0;
      points_2 = 0.0;
      points_3 = 0.0;
      points_4 = 0.0;
      points_5 = 0.0;
      points_6 = 0.0;

      cpr_group_name = reg_name;

      /* --    2. Выполнение плана по штату ВИПС
      select rd_id
      from refdata
      where rd_code in ('PlanIndicators.LO_Count')
      into rep_indicator;
      sum_plan = coalesce((select sum(pd_data_1)
                           from hp_tmp_plandata_show(:reg_id, null, :rep_indicator, :rep_period, 0, 0)
                           where pd_is_office = 1), 0);  */
      sum_plan = 0;
      select po_amount
      from hp_get_planoper_info(:reg_id, 0, 'planindicators.LO_Count', :res_date_end, :rep_period)
      into sum_plan;

      sum_fact = 0;
      sum_ost = 0;
      /*--План по выдачам
      rep_indicator = null;
      select rd_id
      from refdata
      where rd_code in ('planindicators.disb_amount')
      into rep_indicator;     */

      avg_plan_disb_lo = 0;
      avg_plan_disb_svz = 0;
      lo_qty_from_plan = 0;
      for select ps_from_c_id, count(h.ps_from_c_id) over() --общее количество СВЗ
          from hr_pstatus h
          inner join region r on r.reg_unicode = h.ps_from_reg_id
          inner join joblist jl on jl.jl_id = h.ps_from_jl_id
          inner join jobkind jk on jk.jk_id = jl.jl_kind
          where :res_date_end between h.ps_date_begin and h.ps_date_end and
                h.ps_is_working = 1 and
                jk.jk_system_code = 'JobSystem.LO_Standart' and
                r.reg_unicode = :reg_id and
                ps_from_c_id not in (select hrv.hrv_from_c_id -- ИСКЛЮЧАЕМ
                                     from hr_voctravel hrv, sa_worktype swt, hr_orders hr --  •    Отпуск без содержания по уходу за ребенком
                                     where hrv.hrv_date_begin <= hrv.hrv_date_end_fact and --  •    Приказ_отпуск_по_уходу_за_ребенком
                                           hrv.hrv_status = 1 and --  •    Декретный отпуск.
                                           ho_order_type = sawt_id and
                                           hrv_from_ho_id = ho_id and --     sawt_base_type =5 чтобы исключить 21-ю позицию в кг и  аз
                                           :res_date_end between hrv.hrv_date_begin and hrv.hrv_date_end_fact and
                                           swt.sawt_id in (16, 17, 18, 21) and
                                           swt.sawt_base_type = 5)
          into :res_lo_id, :sum_fact
      do
      begin /*
        execute procedure hp_plandata_fill(:reg_id, :res_date_end, :rep_indicator, :rep_period, 1, null);

        --      реализация план выдач
        plan_sum = 0;
        select sum(pd_data_1)
        from temp_plandata
        where pd_from_c_id = :res_lo_id and
              pd_is_office = 2 and
              pd_data_1 > 0
        into plan_sum;
        if (plan_sum is null) then
          plan_sum = 0;   */

        if (lo_qty_from_plan = 0) then
          lo_qty_from_plan = sum_fact;
        plan_sum = 0;
        select po_amount
        from hp_get_planoper_info(:reg_id, :res_lo_id, 'planindicators.disb_amount', :res_date_end, :rep_period)
        into plan_sum;

        if (coalesce(plan_sum, 0) = 0) then
        begin
          plan_sum = 0;
          lo_qty_from_plan = lo_qty_from_plan - 1;
        end
        else
        begin
          --      обьем выдач
          disb_sum = 0;
          select sum(k_amount_approved)
          from kontstate, kontract
          where ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin between :res_date_begin and :res_date_end and
                ks_kc_id = :res_lo_id and
                ks_from_k_id = k_id
          into disb_sum;
          if (disb_sum is null) then
            disb_sum = 0;

          --      Обьем досроных погашений
          select -sum(ok_os)
          from operkred, kontract, knaktlist
          where ok_ol_is_active = 1 and
                ok_type = 4 and
                ok_os < 0 and
                ok_ol_date between :res_date_begin and :res_date_end and
                ok_from_k_id = k_id and
                k_from_kf_id = kal_from_kf_id and
                ok_ol_date between kal_date and kal_date_end and
                kal_new_kf_officer = :res_lo_id and
                ok_ol_date < ok_sh_date and
                extract(month from ok_ol_date) <> extract(month from ok_sh_date)
          into dosroch_sum;
          if (dosroch_sum is null) then
            dosroch_sum = 0;

          --    3.   Среднее значение выполнения плана по выдачам Команды за месяц
          avg_plan_disb_lo = (disb_sum - dosroch_sum) / plan_sum * 100.0;

          avg_plan_disb_svz = avg_plan_disb_svz + avg_plan_disb_lo;
        end
      end

      sum_ost = sum_plan - sum_fact;
      sum_ost = iif(sum_ost > 0, sum_ost, 0);

      sum_plan = round(sum_plan, 2);
      sum_fact = round(sum_fact, 2);
      sum_ost = round(sum_ost, 2);
      --2. Выполнение плана по штату ВИПС (конец)

      --    3. Среднее количество выдач на 1 СВЗ офиса
      if (lo_qty_from_plan = 0) then
        avg_plan_disb_svz = 100.0;
      else
        avg_plan_disb_svz = coalesce(cast(avg_plan_disb_svz as float) / cast(lo_qty_from_plan as float), 0.0);

      --4.    Прирост заработка офиса  >>>>
      zarabotok_diff = 0;
      zarabotok_begin = 0;
      zarabotok_end = 0;
      if (rep_period < 202009) then
      begin
        select aarep_bonus_total
        from cb_ru_office_staff_201908(:reg_id, :date_end_prv)
        where jl_name is null
        into zarabotok_begin;

        select aarep_bonus_total
        from cb_ru_office_staff_201908(:reg_id, :res_date_end)
        where jl_name is null
        into zarabotok_end;
      end
      else
      begin
        select aarep_bonus_total
        from cb_ru_office_staff_202009(:reg_id, :date_end_prv)
        where jl_name is null
        into zarabotok_begin;

        select aarep_bonus_total
        from cb_ru_office_staff_202009(:reg_id, :res_date_end)
        where jl_name is null
        into zarabotok_end;
      end

      zarabotok_diff = zarabotok_end - zarabotok_begin;
      --4.    Прирост заработка офиса <<<<

      cpd_comment = '';
      cpd_comment = coalesce(cpd_comment, '') || 'Количество СВЗ: <b>' || round(:kol_svz, 0) || ' </b><br><br>';
      cpd_comment = coalesce(cpd_comment, '') || 'Кол.выдач: <b>' || round(:ad_disb_loan, 0) || ' </b><br><br>';
      cpd_comment = coalesce(cpd_comment, '') || 'PAR7 офиса на начало: <b>' || round(par7_begin, 2) || ' %.</b><br>';
      cpd_comment = coalesce(cpd_comment, '') || 'PAR7 офиса: <b>' || round(par7, 2) || ' %.</b><br>';
      cpd_comment = coalesce(cpd_comment, '') || 'Изменение PAR7 офиса: <b>' || round(par7_diff, 2) || ' %.</b><br><br>';

      cpd_comment = coalesce(cpd_comment, '') || 'План по штату ВИПС: <b>' || round(sum_plan, 2) || '.</b><br>';

      points_1 = case
                   when par7 < 4.0 then 15
                   when par7_diff >= 1.0 then 10
                   when par7_diff >= 0.50 then 7
                   when par7_diff >= 0.20 then 5
                   when par7_diff >= 0.0 then 2
                   when par7_diff >= -0.20 then -3
                   else -8
                 end;

      points_2 = 0;
      if (sum_plan > 0) then
      begin
        cpd_comment = coalesce(cpd_comment, '') || 'Выполнение плана по штату ВИПС: <b>' || round(sum_fact, 2) || '.</b><br>';
        cpd_comment = coalesce(cpd_comment, '') || 'Остаток по штату ВИПС: <b>' || round(sum_ost, 2) || '.</b><br><br>';

        points_2 = case
                     when sum_ost = 0 then 2
                     when sum_ost = 1 then -8
                     else -15
                   end;
      end
      --
      cpd_comment = coalesce(cpd_comment, '') || '3. Выполнение плана по выдачам в ср. на 1 ВИПС офиса: <b>' || round(avg_plan_disb_svz, 2) || '% </b><br><br>';
      points_3 = case
                   when avg_plan_disb_svz > 100.0 then 8
                   when avg_plan_disb_svz > 90.0 then 6
                   when avg_plan_disb_svz > 80.0 then 4
                   when avg_plan_disb_svz > 70.0 then 0
                   when avg_plan_disb_svz > 60.0 then -3
                   else -5
                 end;

      cpd_comment = coalesce(cpd_comment, '') || '4. Прирост заработка офиса: <b>' || round(zarabotok_diff, 2) || ' р.</b><br><br>';

      --      вычисляем коэффициент Качество принемаемых решений МО  >>>>>>
      res_qual_koef = null;

      select avg(--
             (iif(res_total1 > 0, res_qual_koef1, 0) + iif(res_total2 > 0, res_qual_koef2, 0) + iif(res_total3 > 0, res_qual_koef3, 0)) --
             / --
             (iif(res_total1 > 0, 1, 0) + iif(res_total2 > 0, 1, 0) + iif(res_total3 > 0, 1, 0)) --
             ) res_qual_qoef
      from aarep_disb_manager_quality(:reg_id, :res_date_begin, :res_date_end, 0) dmq
      where exists(select *
                   from aarep_personal_list(:reg_id, :res_date_end, 1) pl
                   where pl.c_name_fio = dmq.c_name_fio and
                         (jl_name containing 'менеджер' or
                         jl_name containing 'ведущий'))
      into res_qual_koef;
      res_qual_koef = round(coalesce(res_qual_koef, 0), 2);
      --1. Качество принемаемых решений МО по первому погашению <<<<

      super_pokazatel = 0;
      if (rep_period = 202003) then
      begin /*    Супер показатель март 2020
        На март супер-показатель для всех соревнований – это прирост портфеля без просрочек.
        В расчет будет включаться портфель, у которого нет ни одного дня просрочки, т.е.
        прирост за месяц считаем как: КП без просрочек на 31.03.20 – КП без просрочек на 29.02.20           */

        select sum(oks_balance) as portfel_begin
        from oksummary oks
        where oks.oks_region_id = :reg_id and
              :date_end_prv between oks_date_begin and oks_date_end and
              cast(:date_end_prv as date) - oks_delinq_date < 1 and
              oks_close_state = 0
        into notdel_portfel_begin;

        select sum(oks_balance) as svz_portfel_end
        from oksummary oks
        where oks.oks_region_id = :reg_id and
              :res_date_end between oks_date_begin and oks_date_end and
              cast(:res_date_end as date) - oks_delinq_date < 1 and
              oks_close_state = 0
        into notdel_portfel_end;

        growth_notdelinq = notdel_portfel_end - notdel_portfel_begin;

        cpd_comment = coalesce(cpd_comment, '') || 'КП без просрочек на 29.02.20: <b>' || round(notdel_portfel_begin, 2) || '.</b><br>';
        cpd_comment = coalesce(cpd_comment, '') || 'КП без просрочек на 31.03.20: <b>' || round(notdel_portfel_end, 2) || '.</b><br>';
        cpd_comment = coalesce(cpd_comment, '') || 'Прирост портфеля без просроче: <b>' || round(growth_notdelinq, 2) || '.</b><br><br>';
      end --Superpokazatel konec
      else
      if (rep_period = 202006) then
      begin /*    Супер показатель июнь 2020 выдачи онлайн    */
        select count(ks_from_k_id)
        from kontstate ks
        inner join kontract k on ks_from_k_id = k_id and
              k_is_online = 1
        where ks.ks_region_id = :reg_id and
              ks.ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks.ks_date_begin between :res_date_begin and :res_date_end
        into super_pokazatel; --конец супер-показателя за июнь
      end
      else
      if (rep_period = 202007) then
      begin /*    Супер показатель июнь 2020 выдачи    */
        select count(ks_from_k_id)
        from kontstate ks
        where ks.ks_region_id = :reg_id and
              ks.ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks.ks_date_begin between :res_date_begin and :res_date_end
        into super_pokazatel; --конец супер-показателя за июль
      end
      else
      if (rep_period = 202009) then
      begin
        /*    Сентябрь – снижение баланса просрочки PAR30 в процентах от
          баланса на начало месяца. То есть на начало Сентября баланс 30 млн, а на
          конец стал 15 млн, это значит, что снижение 50%. Результаты ранжируются и
          баллы начисляются точно так-же как и в предыдущих месяцах. Для ОИ – снижение
          рассчитывается от суммарного баланса команды, то есть все балансы участников
          на начало складываются, все балансы участников на конец складываются и, затем,
          вычисляется снижение баланса команды. Для ОИ МО – это снижение баланса офиса.     */
        super_pokazatel = ((par30_balance_end / par30_balance_begin) - 1) * 100; --конец супер-показателя за 202009
      end
      else
      if (rep_period = 202010) then
      begin /*    Супер показатель октябрь 2020 выдачи новым клиентам   */
        super_pokazatel = 0;
        select count(*)
        from kontstate ks
        inner join knclient kc on ks_from_k_id = kc_from_k_id and
              kc_amount_approved > 0 and
              kc_circle = 1

      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end
    execute procedure hp_competition_recalc_order(:cpp_id);
    --    res_text = 'Автоматизация итогов еще не готов';
  end

  execute procedure hp_competition_recalc_order(:cpp_id);

  update competresult
  set cpr_final_orderno = 0,
      cpr_calced_points = 0,
      cpr_additional_points = 0,
      cpr_order_in_group = 0,
      cpr_final_points = 0
  where cpr_from_cpp_id = :cpp_id and
        cpr_object_id = 0;

  cpr_final_points = 0;
  for select cpr_id, dense_rank() over(order by cpr_final_orderno)
      from competresult cpr
      where cpr.cpr_from_cpp_id = :cpp_id and
            cpr.cpr_object_id <> 0 and
            coalesce(cpr_group_name, 'Не заполнен') <> 'Не заполнен'
      into :cpr_id, :cpr_final_points
  do
  begin
    update competresult
    set cpr_final_orderno = :cpr_final_points
    where cpr_id = :cpr_id;
  end
  suspend;
