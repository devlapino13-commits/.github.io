CREATE PROCEDURE UXP$TRANSMAIN_SYNC_SCRIPT(
    ONLY_TRANSLATED D_BOOL)
RETURNS (
    SQL_TEXT D_STR_8100)
AS
declare variable sql_mask d_str_long;
declare variable tmp_text d_str_long;
begin
  sql_mask = 'insert into ux$transimport (trm_orig_text, trm_is_delphi, trl_lang_code, trl_translate_text, trl_actual_text, trl_checked) values (';

  for select trm_orig_text, trm_is_delphi, trl_lang_code, trl_translate_text, trl_actual_text, trl_checked
  from uxp$transmain_sync_list(:only_translated)
  where coalesce(trm_orig_text, '') <> ''
  as cursor cur
  do
  begin
    sql_text = sql_mask || '''' || replace(cur.trm_orig_text, '''', '''''') || '''';
    sql_text = sql_text || ', ' || coalesce(cur.trm_is_delphi, 0);

    tmp_text = coalesce(cur.trl_lang_code, '');
    sql_text = sql_text || ', ' ||  '''' || replace(tmp_text, '''', '''''') || '''';

    tmp_text = coalesce(cur.trl_translate_text, '');
    sql_text = sql_text || ', ' ||  '''' || replace(tmp_text, '''', '''''') || '''';

    tmp_text = coalesce(cur.trl_actual_text, '');
    sql_text = sql_text || ', ' ||  '''' || replace(tmp_text, '''', '''''') || '''';

    if (cur.trl_checked is null) then
      tmp_text = 'null';
    else
      tmp_text = cur.trl_checked;

    sql_text = sql_text || ', ' ||  tmp_text || ');';
    suspend;
  end
end
