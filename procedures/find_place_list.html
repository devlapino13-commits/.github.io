CREATE PROCEDURE FIND_PLACE_LIST(
    PARENT_ID DECIMAL(18,0),
    PLACE_STR VARCHAR(255),
    USER_ID DECIMAL(18,0))
RETURNS (
    PL_ID DECIMAL(18,0),
    PARENT DECIMAL(18,0),
    PL_PATH VARCHAR(255),
    PL_NAME VARCHAR(100),
    PL_STATUS INTEGER,
    PL_CAN_SELECT INTEGER,
    PLT_NAME VARCHAR(255),
    PLACE_FULL_REGION VARCHAR(255),
    PLACE_FULL VARCHAR(255),
    PL_NAME_KOMITET VARCHAR(100))
AS
declare variable txt varchar(255);
declare variable p_id decimal(18,0);
declare variable finish integer;
declare variable kol integer;
declare variable can_suspend integer;
declare variable parent_tmp decimal(18,0);
declare variable plt_type_name varchar(255);
declare variable plt_short_name_side integer;
begin
  /* Procedure Text */
  if (place_str is null) then
    place_str = '';
  place_str = (select res_txt
               from hp_clear_symbols(:place_str));
  if ((parent_id is null) or (parent_id <= 0)) then
    parent_id = 0;
  if (char_length(place_str) <= 1) then
    place_str = '';
  if ((place_str <> '') and
      (parent_id <> 0)) then
  begin
    if (char_length(place_str) >= 3) then
    begin
      for select pl_id, parent, pl_name, pl_name_komitet, pl_status, plt_name, plt_can_select
          from places, placetype, find_in_workdoc_words('places', :place_str, 1)
          where pl_status = plt_id and
                pl_id = wdlink_object_id and
                plt_can_select = 1
          into :pl_id, :parent, :pl_name, :pl_name_komitet, :pl_status, :plt_name, :pl_can_select
      do
      begin
        parent_tmp = :pl_id;
        if (pl_id <> parent_id) then
          can_suspend = 0;
        else
          can_suspend = 1;
        while ((parent_tmp is not null) and (can_suspend = 0)) do
        begin
          select parent
          from places
          where pl_id = :parent_tmp
          into :parent_tmp;
          if (parent_tmp = :parent_id) then
            can_suspend = 1;
        end
        if (can_suspend = 1) then
        begin
          pl_path = '';
          finish = 0;
          for select pl_id, pl_name, plt_name, plt_short_name_side
              from places_list(:pl_id, 0, -1, 0), places, placetype
              where id_list = pl_id and
                    pl_status = plt_id
              into :p_id, :txt, :plt_type_name, :plt_short_name_side
          do
          begin
            if (plt_short_name_side = -1) then
            begin
              if (right(plt_type_name, 1) <> '.') then
                plt_type_name = plt_type_name || ' ';
              txt = plt_type_name || txt;
            end
            else
            if (plt_short_name_side = 1) then
              txt = txt || ' ' || plt_type_name;
            if (finish = 0) then
              pl_path = :txt || ' ' || :pl_path;
            if (:parent_id = :p_id) then
              finish = 1;
          end
          place_full_region = '';
          place_full = '';
          select place_full, place_full_region
          from aarep_places_link(:pl_id)
          into :place_full, :place_full_region;
          suspend;
        end
      end
    end
    else
    begin
      for select pl_id, parent, pl_name, pl_name_komitet, pl_status, plt_name, plt_can_select
          from places, placetype
          where pl_status = plt_id and
                (pl_name = :place_str or
                pl_name_komitet = :place_str) and
                plt_can_select = 1
          into :pl_id, :parent, :pl_name, :pl_name_komitet, :pl_status, :plt_name, :pl_can_select
      do
      begin
        parent_tmp = :pl_id;
        if (pl_id <> parent_id) then
          can_suspend = 0;
        else
          can_suspend = 1;
        while ((parent_tmp is not null) and (can_suspend = 0)) do
        begin
          select parent
          from places
          where pl_id = :parent_tmp
          into :parent_tmp;
          if (parent_tmp = :parent_id) then
            can_suspend = 1;
        end
        if (can_suspend = 1) then
        begin
          pl_path = '';
          finish = 0;
          for select pl_id, pl_name, plt_name, plt_short_name_side
              from places_list(:pl_id, 0, -1, 0), places, placetype
              where id_list = pl_id and
                    pl_status = plt_id
              into :p_id, :txt, :plt_type_name, :plt_short_name_side
          do
          begin
            if (plt_short_name_side = -1) then
            begin
              if (right(plt_type_name, 1) <> '.') then
                plt_type_name = plt_type_name || ' ';
              txt = plt_type_name || txt;
            end
            else
            if (plt_short_name_side = 1) then
              txt = txt || ' ' || plt_type_name;
            if (finish = 0) then
              pl_path = :txt || ' ' || :pl_path;
            if (:parent_id = :p_id) then
              finish = 1;
          end
          place_full_region = '';
          place_full = '';
          select place_full, place_full_region
          from aarep_places_link(:pl_id)
          into :place_full, :place_full_region;
          suspend;
        end
      end
    end
  end
  else
  begin
    for select ups_from_pl_id, pl_name, pl_name_komitet, ups_access_kolich, parent, pl_status, plt_name,
               plt_can_select
        from userplacestat, places, placetype
        where ups_from_un_id = :user_id and
              pl_id = ups_from_pl_id and
              pl_status = plt_id
        order by ups_access_kolich desc
        into :pl_id, :pl_name, :pl_name_komitet, :kol, :parent, :pl_status, :plt_name, :pl_can_select
    do
    begin
      pl_path = '';
      finish = 0;
      for select pl_id, pl_name, plt_name, plt_short_name_side
          from places_list(:pl_id, 0, -1, 0), places, placetype
          where id_list = pl_id and
                pl_status = plt_id
          into :p_id, :txt, :plt_type_name, :plt_short_name_side
      do
      begin
        if (plt_short_name_side = -1) then
        begin
          if (right(plt_type_name, 1) <> '.') then
            plt_type_name = plt_type_name || ' ';
          txt = plt_type_name || txt;
        end
        else
        if (plt_short_name_side = 1) then
          txt = txt || ' ' || plt_type_name;
        if (finish = 0) then
          pl_path = :txt || ' ' || :pl_path;
        if (:parent_id = :p_id) then
          finish = 1;
      end
      place_full_region = '';
      place_full = '';
      select place_full, place_full_region
      from aarep_places_link(:pl_id)
      into :place_full, :place_full_region;
      suspend;
    end

  end
end
