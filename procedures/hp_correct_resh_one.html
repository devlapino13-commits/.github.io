CREATE PROCEDURE HP_CORRECT_RESH_ONE(
    KK_ID DECIMAL(18,0))
RETURNS (
    K_ID DECIMAL(18,0),
    MIN_SH_DD DATE,
    OK_TYPE INTEGER,
    SH_ID INTEGER,
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    OK_OS DECIMAL(18,4),
    SH_DIFF_DATE INTEGER,
    K_PERCENT_PER_DAY DECIMAL(18,7),
    PR_IS_360 INTEGER,
    RES_KOL_DAYS INTEGER,
    BALANCE_NEW DECIMAL(18,4),
    OK_PERC DECIMAL(18,4),
    RES_DIFF_PERC DECIMAL(18,4),
    OPER_DD DATE,
    OPER_ID DECIMAL(18,0),
    PERC_TEMP DECIMAL(18,4),
    SH_PERC_NEXT DECIMAL(18,4),
    FIRST_LINE INTEGER)
AS
declare variable dd_prior date;
declare variable dd date;
declare variable ok_balance decimal(18,4);
declare variable min_sh_id integer;
declare variable pp2 decimal(18,4);
begin
  k_id = :kk_id;

  delete from operkred
  where ok_from_k_id = :kk_id
--  and ok_from_c_id=0
        and
        ok_type = 3;

  for select ok_ol_date, min(ok_from_ol_id)
      from operkred
      where ok_ol_is_active = 1 and
            ok_os <> 0 and
            ok_from_k_id = :kk_id and
            ok_from_c_id = 0
      group by ok_ol_date
      order by ok_ol_date
      into :oper_dd, :oper_id
  do
  begin
    dd_prior = null;
    ok_perc = 0;
    perc_temp = 0;
    sh_perc_next = 0;
    balance_new = 0;
    first_line = 0;

    for select min_sh_id, min_sh_dd, ok_type, sh_id, sh_date, sh_os, sh_perc, ok_os, ok_balance, sh_diff_date,
               k_percent_per_day, pr_is_360
        from hp_correct_resh_temp(:kk_id, :oper_dd), kontract, product
        where k_id = :kk_id and
              k_from_pr_id = pr_id
        order by sh_date, ok_type
        into :min_sh_id, :min_sh_dd, :ok_type, :sh_id, :sh_date, :sh_os, :sh_perc, :ok_os, :ok_balance, :sh_diff_date,
             :k_percent_per_day, :pr_is_360
    do
    begin
      res_diff_perc = 0;
      if (dd_prior is null) then
      begin
        first_line = 1;
        dd_prior = :min_sh_dd;
        balance_new = :ok_balance;
      end
      else
        first_line = 0;
      if (sh_date > :dd_prior) then
      begin
        if (sh_diff_date = 0) then
        begin
          execute procedure fhp_agein360(:dd_prior, :sh_date, :pr_is_360)
              returning_values :res_kol_days;
          pp2 = balance_new * res_kol_days / 100;
          pp2 = :pp2 * k_percent_per_day;
          perc_temp = :perc_temp + :pp2;
          perc_temp = :perc_temp + :sh_perc_next;
          sh_perc_next = 0;
        end
        else
        begin
          dd = :sh_date - :sh_diff_date;
          if (not((sh_diff_date > 0) and
              (dd = dd_prior))) then
          begin
            perc_temp = :perc_temp + :sh_perc_next;
            sh_perc_next = 0;
          end
          if (sh_diff_date > 0) then
          begin
            execute procedure fhp_agein360(:dd, :sh_date, :pr_is_360)
                returning_values :res_kol_days;
            sh_perc_next = :sh_perc_next + cast(:balance_new * :res_kol_days * :k_percent_per_day / 100 as decimal(15,0));

            execute procedure fhp_agein360(:dd_prior, :dd, :pr_is_360)
                returning_values :res_kol_days;
            if (res_kol_days > 0) then
            begin
              pp2 = balance_new * res_kol_days / 100;
              pp2 = :pp2 * k_percent_per_day;
              perc_temp = :perc_temp + :pp2;
            end
          end
          else
          begin
            execute procedure fhp_agein360(:sh_date, :dd, :pr_is_360)
                returning_values :res_kol_days;
            if (res_kol_days > 0) then
              perc_temp = :perc_temp + (:balance_new - :sh_os + :ok_os) * res_kol_days * k_percent_per_day / 100;

            execute procedure fhp_agein360(:dd_prior, :sh_date, :pr_is_360)
                returning_values :res_kol_days;
            if (res_kol_days > 0) then
            begin
              pp2 = balance_new * res_kol_days / 100;
              pp2 = :pp2 * k_percent_per_day;
              perc_temp = :perc_temp + :pp2;
            end
          end
        end
      end
      if ((ok_type = 1) and
          (sh_perc <> 0)) then
      begin
        ok_perc = cast(:perc_temp as decimal(15,0));
        perc_temp = 0;
        res_diff_perc = :ok_perc - :sh_perc;
        if ((res_diff_perc >= -1) and
            (res_diff_perc <= 1)) then
        begin
          res_diff_perc = 0;
          ok_perc = :sh_perc;
        end
      end
      else
        res_diff_perc = 0;
      if (res_diff_perc <> 0) then
      begin
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:oper_id, -1, :kk_id, :sh_id, :sh_date, 3, 0, 0, 0, 2, 0, 0, :res_diff_perc, 0, 0);
      end
      suspend;
      if (first_line = 0) then
      begin
        if (sh_date <= :oper_dd) then
        begin
          balance_new = :balance_new + :ok_os;
        end
        else
        begin
          balance_new = :balance_new - :sh_os;
        end
      end
      if (sh_diff_date >= 0) then
        dd_prior = :sh_date;
      else
        dd_prior = :sh_date - :sh_diff_date;
    end
  end
end
