CREATE PROCEDURE HP_PRE_SUD_CLI_DOCS(
    REP_C_ID ID_TYPE)
RETURNS (
    PK_NAME STR_255,
    CDS_DOC_DATE_BEGIN D_DATE,
    CDS_DOC_DATE_END D_DATE,
    CDS_DOC_BLOB_1 ID_TYPE,
    CDS_DOC_BLOB_2 ID_TYPE,
    CDS_DATETIME D_DATETIME,
    CDS_IS_DELETE D_INTEGER)
AS
begin
  for select pk_name, cds_doc_date_begin, cds_doc_date_end, cds_doc_blob_1,
             cds_doc_blob_2, cds_datetime, cds_is_delete
      from aarep_clientdocs_list(:rep_c_id)
      into pk_name, cds_doc_date_begin, cds_doc_date_end, cds_doc_blob_1,
           cds_doc_blob_2, cds_datetime, cds_is_delete
  do
  begin
    suspend;
  end
end
