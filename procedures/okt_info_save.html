CREATE PROCEDURE OKT_INFO_SAVE(
    ACL_ID ID_TYPE,
    UN_ID ID_TYPE,
    PARAMS BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
RETURNS (
    RES SMALLINT)
AS
declare variable al_project str_255;
declare variable t integer;
declare variable nextpos integer;
declare variable field_value str_255;
declare variable res_timer integer;
declare variable default_un_id id_type;
begin
  /*
  OKTELL

  На данный момент, процедура предназначена для сохранения длительности диалога между оператором и абонентом (Duration=...)
  В будущем, может использоваться для сохранения каких-либо дополнительных значений.

  */

  select rd.rd_code
  from actionlist al
  inner join refdata rd on al.acl_from_rd_id = rd.rd_id
  where al.acl_id = :acl_id
  into :al_project;

  field_value = null;

  if (al_project containing ('OKTELL.')) then
  begin
    t = position('Duration=', params); -- продолжительность диалога в секундах
    if (t <> 0) then
    begin
      nextpos = position('&', params, t + 1);
      if (nextpos = 0) then
        field_value = substring(params from t + 9 for char_length(params) - t - 8);
      else
        field_value = substring(params from t + 9 for nextpos - t - 9);
    end
    field_value = coalesce(field_value, '');

    if (field_value <> '') then
    begin
      select res_timer
      from hp_actionlist_timer(:acl_id, cast(:field_value as integer), 'set')
      into :res_timer;
    end

    if (al_project containing ('OKTELL.Projects.WebSite')) then -- звонки по заявкам с сайта для мигрантов
    begin
      select un.un_id
      from username un
      where un.un_name = 'IGUSEV'
      into :default_un_id;

      execute procedure fsrobot_insert_event('DoMailing', 'OKTELL_MAILING', :acl_id, :default_un_id);
    end
  end

  suspend;
end
