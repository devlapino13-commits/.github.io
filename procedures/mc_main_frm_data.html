CREATE PROCEDURE MC_MAIN_FRM_DATA(
    MCL_ID D_ID,
    C_ID D_ID)
RETURNS (
    CLI_ACCOUNT D_STR_100,
    CLI_LOAN_INFO D_STR_100,
    CLI_FIO D_STR_100)
AS
begin
-- "МП для клиентов" Главное окно
-- CLI_ACCOUNT - лицевой счет
  cli_account = coalesce((select uniq_acc_str
                          from hp_show_clientuniq_acc_no(:c_id)), '');
  cli_account = iif(cli_account = '', 'Отсутствует', cli_account);
  cli_account = iif(cli_account = 'XXX', 'Отсутствует', cli_account);

  cli_loan_info = 'У Вас нет активных займов';
  if (exists(select *
             from knclient kncl
             inner join kontstate ks on ks.ks_from_k_id = kncl.kc_from_k_id and
                   ks.ks_state = 1 and
                   ks.ks_ol_is_delete = 0 and
                   current_date between ks.ks_date_begin and ks.ks_date_end
             where kncl.kc_from_c_id = :c_id and
                   kncl.kc_amount_approved > 0)) then
    cli_loan_info = 'Информация по активным займам';

  cli_fio = (select cl.c_name_fio
             from clients cl
             where cl.c_id = :c_id);

  suspend;
end
