CREATE PROCEDURE PRINT_RESULT_ANTIFROD(
    OBJECT_KIND INTEGER,
    OBJECT_ID ID_TYPE)
RETURNS (
    REP_FIO STR_100,
    REP_BIRTHDATE D_DATE,
    REP_KONTRACT_NO D_STR_20,
    REP_APPLICATE_NO D_INTEGER,
    REP_APPLICATE_DATE D_DATE,
    REP_DISB_DATE D_DATE,
    REP_KND_DATETIME D_DATETIME,
    REP_CATEGORY D_STR_LONG,
    REP_CRITERIA D_STR_8100,
    REP_RES_TEXT D_STR_255,
    RES_CATEGORIA D_PERCENT,
    RISK_LEVEL D_SMALLINT,
    RISK_LEVEL_STR D_STR_CODE,
    AFC_ID D_ID,
    TOTAL_PERCENT D_CURRENCY)
AS
declare variable k_id id_type;
declare variable acr_id d_id;
declare variable jso d_integer;
declare variable jso_cat d_integer;
declare variable jso_cri d_integer;
declare variable acr_percent d_percent;
declare variable acr_name d_str_long;
declare variable criteria_code d_str_code;
begin
  if (object_kind = 9) then
    k_id = object_id;

  if (k_id is null) then
    exit;

  if (f_loan_is_online(k_id) = 0) then
    exit;  

  select c_name_fio, c_birth_date, k_kontract_no, k_applicate_date, k_applicate_no, k_disb_date
  from knclient, kontract, clients
  where k_id = kc_from_k_id
    and c_id = kc_from_c_id
    and kc_from_k_id = :k_id
    and kc_amount_approved > 0
  into :rep_fio, :rep_birthdate, :rep_kontract_no, rep_applicate_date, rep_applicate_no, rep_disb_date;

  select rep_knd_datetime, jso
  from hp_antifrod_get_res(:k_id)
  into rep_knd_datetime, jso;

  risk_level = json.obj_get_val(:jso, 'risk_level', 0);

  risk_level_str = case risk_level
                     when 0 then 'Низкий'
                     when 1 then 'Значительный'
                     else 'Высокий'
                   end;

  total_percent = 0;

  for select json.obj_get_obj(:jso, val), val
  from hp_bki_json_get_list(json.obj_get_arr(:jso, 'cat_arr'), 0)
  into jso_cat, afc_id
  do
  begin
    rep_category = (select afc_name from antifrod_categories where afc_id = :afc_id);
    res_categoria = json.obj_get_val(:jso_cat, 'category_res', 0);
    rep_criteria = '';
    total_percent = total_percent + res_categoria;

    for select json.obj_get_obj(:jso_cat, val), val
    from hp_bki_json_get_list(json.obj_get_arr(:jso_cat, 'cri_arr'), 0)
    into jso_cri, acr_id
    do
    begin
      acr_name = null;

      select acr_name, acr_percent
      from antifrod_criteria
      where acr_id = :acr_id
      into acr_name, acr_percent;

      if (afc_id < 0) then
      begin
        rep_criteria = acr_name;
        criteria_code = json.obj_get_val(jso_cri, 'criteria_code');
        rep_res_text = json.obj_get_val(jso, f_format('criteria_data.%1.res_text', criteria_code));
        suspend;
      end
      else
      begin
        rep_criteria = rep_criteria || f_format('%1 <b>(%2%)</b>%3', acr_name, f_numtostr(json.obj_get_val(jso_cri, 'criteria_res', acr_percent), 0), ascii_char(10));
      end
    end

    if (total_percent < 0) then
      total_percent = 0;

    if (afc_id > 0) then
      suspend;
  end
end
