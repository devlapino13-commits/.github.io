declare variable k_id_prior id_type;
declare variable kf_id id_type;
declare variable reg_id integer;
declare variable reg_id_prior integer;
declare variable k_amount curr_type;
declare variable k_balance curr_type;
declare variable k_from_pr_id integer;
declare variable kf_officer_id id_type;
declare variable kf_officer_id_prior id_type;
declare variable busuness_subtype integer;
declare variable cl_dolya double precision;
declare variable k_llc curr_type;
declare variable ks_state smallint;
declare variable kc_loan_target integer;
declare variable kc_loan_target_prior integer;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable c_army integer;
declare variable c_education integer;
declare variable c_family integer;
declare variable c_citizen integer;
declare variable c_nation integer;
declare variable c_doc_type integer;
declare variable c_doc_type_prior integer;
declare variable c_is_resident smallint;
declare variable ks_date_begin date;
declare variable ks_from_ol_id id_type;
declare variable k_percent perc_type;
declare variable rep_deliq_perc_kont curr_type;
declare variable pr_is_secured integer;
declare variable pr_int_calc_type integer;
declare variable c_type integer;
declare variable kc_is_new_business integer;
declare variable kc_is_registered integer;
declare variable kc_status integer;
declare variable lk_status integer;
declare variable k_delinq_balance curr_type;
declare variable k_delinq_os curr_type;
declare variable k_kolich_client_approved smallint;
begin
  k_id_prior = 0;
  reg_id_prior = 0;
  k_balance = 0;
  kf_officer_id_prior = 0;
  c_doc_type_prior = 0;
  for select c_id, ks_from_k_id, kc_amount_approved, kc_circle, kc.kc_business_name, kc.kc_busuness_subtype, ks_state,
             kc_loan_target, kc.kc_amount_applicate, ks_date_begin, ks_from_ol_id, kc_is_new_business, kc_is_registered,
             kc_monthly_oborot, kc_kapital, kc_month_profit, kc_work_kolich, kc_work_kolich_family, kc_status,

             '' as c_name_fio, c_army, c_education, c_family, c_citizen, c_nation, c_doc_type, '' as c_plastcard_no,
             c_is_resident, c_inn, c_okpo, c_kol_child, c_kol_family,
             (select cont_value
              from hp_contdata_get_email_valid(0, c_id)), c_pass_date_expire, c_pass_who, c_pass_date, '' as c_pass_no,
             c_birth_date, c_sex, c_type, c_memo, c_account, c_org_from_b_id, c_hired_date

      from kontstate, knclient kc, clients
      where ks_ol_is_delete = 0 and
            ks_state in (1, 2) and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into c_id, k_id, cl_amount, cl_circle, business_name, busuness_subtype, ks_state, kc_loan_target,
           cl_amount_applicate, ks_date_begin, ks_from_ol_id, kc_is_new_business, kc_is_registered, knc_monthly_oborot,
           knc_kapital, knc_month_profit, knc_work_kolich, knc_work_kolich_family, kc_status,

           cl_fio, c_army, c_education, c_family, c_citizen, c_nation, c_doc_type, c_plastcard_no, c_is_resident, c_inn,
           c_okpo, c_kol_child, c_kol_family, c_email, c_pass_date_expire, c_pass_who, c_pass_date, c_pass_no,
           c_birth_date, c_sex, c_type, knc_memo, knc_account, knc_org_from_b_id, knc_hired_date

  do
  begin
--  филиал папки и офис папки
--  и все данные которые берутся из таблицы knclient
    if (k_id_prior <> k_id) then
    begin
      k_id_prior = k_id;
      select k_from_kf_id, k_amount_approved, k_from_pr_id, k_disb_date, k_kontract_no, k.k_kredline_end,
             k_kolich_repay, k_kolich_lgota, k_period, k.k_amount_applicate, k.k_kontract_date, k_applicate_no,
             k_applicate_date, k.k_index_curr_code, k_percent_lgota,
--             K_PERCENT_REAL,
             k_from_f_id, k_from_pt_id, k_percent_for_days, k_kommis_perc, k_percent, k_kolich_client_approved

      from kontract k
      where k_id = :k_id
      into kf_id, k_amount, k_from_pr_id, k_disb_date, k_kontract_no, k_kredline_end, k_kolich_repay, k_kolich_lgota,
           k_period, k_amount_applicate, k_kontract_date, k_applicate_no, k_applicate_date, k_index_curr_code,
           k_percent_lgota,
--           K_PERCENT_REAL,
           k_from_f_id, k_from_pt_id, k_percent_for_days, k_kommis_perc, k_percent, k_kolich_client_approved

      ;

      k_percent_real = :k_percent - :k_percent_lgota;
--    Тип кредита
--    Кр. продукт
      select iif(pr_loan_type = 0, 'групп.', 'индив.'), pr_name, pr_is_secured, pr_interest_method
      from product
      where pr_id = :k_from_pr_id
      into k_type, pr_name, pr_is_secured, pr_int_calc_type;

      pr_is_secured_str = '';
      if (pr_is_secured = 0) then
        pr_is_secured_str = 'Без залога';
      if (pr_is_secured = 1) then
        pr_is_secured_str = 'Залог';
      if (pr_is_secured = 2) then
        pr_is_secured_str = 'Поручители';
      if (pr_is_secured = 3) then
        pr_is_secured_str = 'Залог,поруч.';

      pr_int_calc_type_str = '';
      if (pr_int_calc_type = 0) then
        pr_int_calc_type_str = 'На сумму кредита';
      if (pr_int_calc_type = 1) then
        pr_int_calc_type_str = 'На остаток';
      if (pr_int_calc_type = 2) then
        pr_int_calc_type_str = 'Аннуитет';
--  /*
      select kal_new_reg_id, kal_new_kf_officer, kal_new_kf_name,
             iif(kal_from_ka_id <> 0 and kal_date > :k_disb_date, kal_date, null)
      from knaktlist
      where knaktlist.is_delete = 0 and
            kal_from_kf_id = :kf_id and
            :rep_date_end between kal_date and kal_date_end
      into reg_id, kf_officer_id, kf_name, kf_date_in_to_krs;
--           */

      if (kf_officer_id_prior <> kf_officer_id) then
      begin
        kf_officer_id_prior = kf_officer_id;
        select c_name_fio
        from clients
        where c_id = :kf_officer_id
        into kf_officer;
      end

--    Филиал папки;    Офис папки
      if (reg_id_prior <> reg_id) then
      begin
        reg_id_prior = reg_id;
        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r1.reg_stat_rep_region
        from region r1, region r2
        where r1.reg_unicode = :reg_id and
              r1.parent = r2.reg_unicode
        into reg_name, branch_name, reg_stat_rep_region;
      end

      k_balance = 0;
      k_delinq_date = null;
      select oks_balance, oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into k_balance, k_delinq_date;

      k_rest = null;
      if (exists(select *
                 from operrest, operlist
                 where or_from_ol_id = ol_id and
                       operlist.is_delete = 0 and
                       or_from_k_id = :k_id and
                       :rep_date_end between ol_date and coalesce(or_date_end, cast('31.12.9999' as date)))) then
        k_rest = 'Рекструкт.';

      k_llc = 0;
      llc_percent = 0;
      zad_os = 0;
      zad_perc = 0;
      k_delinq_os = 0;

      k_fact_close_date = null;
      if (ks_state = 1) then
      begin
--      РППУ
        select cast(lk_balance as double precision) * lk_percent / 100, lk_percent, lk_status
        from llc_kont, llc_main
        where lk_from_k_id = :k_id and
              :rep_date_end between lk_date and lk_date_end and
              lk_from_lm_id = lm_id and
              lm_is_signed = 1
        into k_llc, llc_percent, lk_status;

        select cs_class_text
        from classific
        where cs_id = :lk_status
        into llc_current_class;

--    Задолженность клиента
        select sum(zad_os), sum(sh_calc_perc), sum(iif(sh_repay_type = 1, sh_calc_perc, 0)), --Задолж. по %
               sum(iif(sh_repay_type = 1, zad_os, 0))
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        where sh_date <= :rep_date_end
        into zad_os, zad_perc, rep_deliq_perc_kont, k_delinq_os;
      end
      else
      begin
        k_fact_close_date = ks_date_begin;
      end

--    Дата нач. проср.    Кол проср. дней
      k_delinq_date = coalesce(k_delinq_date, '31.12.9999');
      k_delinq_day = 0;
      if (k_delinq_date <= :rep_date_end) then
      begin
        k_delinq_day = rep_date_end - k_delinq_date;
      end
      else
      begin
        k_delinq_date = null;
      end

      if (k_delinq_day > 0) then
      begin
        k_delinq_balance = k_balance;
      end
      else
        k_delinq_balance = 0;

--    Кол. проср. выплат
      k_kol_delinq_repay = 0;
      select count(distinct ok_sh_date)
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_ol_date > ok_sh_date and
            ok_type in (4, 9) and
            ok_os + ok_perc < 0 and
            ok_from_c_id = 0
      into k_kol_delinq_repay;
--    Процент в день
      percent_daye = 0;
      select kd_percent_per_day
      from kontdata
      where kd_from_k_id = :k_id and
            :rep_date_end between kd_ol_date and kd_date_end and
            kd_ol_is_delete = 0
      into percent_daye;
--    Месяц выдачи, год выдачи
      disb_month = extract(month from :k_disb_date);
      disb_year = extract(year from :k_disb_date);

      is_par7_month6 = 0;
      is_par7_year1 = 0;
      is_par30_month6 = 0;
      is_par30_year1 = 0;

      select
--      popal li klient v PaR7 v pervie 6 mesyaces posle vidaci (1 esli da, 0 esli net)
             max(iif(ok_ol_date between :k_disb_date and dateadd(month, 6, :k_disb_date), 1, 0)),
--      popal li klient v PaR30 v pervie 6 mesyaces posle vidaci (1 esli da, 0 esli net)
             max(iif(ok_ol_date between :k_disb_date and dateadd(year, 1, :k_disb_date), 1, 0)),

--      popal li klient v PaR30 v pervie 6 mesyaces posle vidaci (1 esli da, 0 esli net)
             max(iif(ok_ol_date - ok_sh_date > 30 and ok_ol_date between :k_disb_date and dateadd(month, 6, :k_disb_date), 1, 0)),
--      popal li klient v PaR30 v perviy qod posle vidaci (1 esli da, 0 esli net)
             max(iif(ok_ol_date - ok_sh_date > 30 and ok_ol_date between :k_disb_date and dateadd(year, 1, :k_disb_date), 1, 0))
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1
--      and ok_ol_date between :k_disb_date and dateadd(month, 6, :k_disb_date)
            and
            ok_type in (4, 5, 9) and
            ok_os + ok_perc < 0 and
            ok_ol_date - ok_sh_date > 7 and
            ok_from_c_id = 0
      into is_par7_month6, is_par7_year1, is_par30_month6, is_par30_year1;

      if (is_par7_month6 is null) then
        is_par7_month6 = 0;
      if (is_par7_year1 is null) then
        is_par7_year1 = 0;
      if (is_par30_month6 is null) then
        is_par30_month6 = 0;
      if (is_par30_year1 is null) then
        is_par30_year1 = 0;

      if (ks_state = 1) then
      begin
        if (is_par7_month6 = 0 and
            :k_disb_date >= dateadd(month, -6, :rep_date_end) and
            k_delinq_day > 7) then
          is_par7_month6 = 1;
        if (is_par7_year1 = 0 and
            :k_disb_date >= dateadd(year, -1, :rep_date_end) and
            k_delinq_day > 7) then
          is_par7_year1 = 1;
        if (is_par30_month6 = 0 and
            :k_disb_date >= dateadd(month, -6, :rep_date_end) and
            k_delinq_day > 30) then
          is_par7_month6 = 1;
        if (is_par30_year1 = 0 and
            :k_disb_date >= dateadd(year, -1, :rep_date_end) and
            k_delinq_day > 30) then
          is_par7_year1 = 1;
      end
--      status (aktivniy kredit, zakritiy kredit, spisanniy kredit)

      if (exists(select *
                 from operkred
                 where ok_from_ol_id = :ks_from_ol_id and
                       ok_from_k_id = :k_id and
                       ok_ol_is_active = 1 and
                       ok_type in (5, 6) and
                       ok_os + ok_perc < 0 and
                       ok_from_c_id = 0)) then
      begin
        k_status = 'Списан.';
      end
      else
      begin
        if (ks_state = 1) then
          k_status = 'Активный';
        else
          k_status = 'Закрыт.';
      end

    end
    c_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :c_id and
          adc_from_adl_id = adl_id
    into :c_full_address;

    if (c_type = 0) then
      knc_type_str = 'Физ.лицо';
    else
      knc_type_str = 'Юр.лицо';

    if (c_pass_date_expire is not null) then
    begin
      if (c_pass_date_expire < :rep_date_end) then
        knc_pass_date_expired_str = 'Просрочен';
      else
      begin
        if (datediff(day, :rep_date_end, c_pass_date_expire) <= 30) then
          knc_pass_date_expired_str = 'Истекает';
        else
          knc_pass_date_expired_str = 'Действителен';
      end
    end
    else
      knc_pass_date_expired_str = 'Не указан';

    cl_dolya = cast(cl_amount as double precision) / k_amount;
--  возрост клиента, месяц рождения клиента
    c_vozrast = datediff(year, :c_birth_date, :rep_date_end);
    c_birth_month = extract(month from :c_birth_date);
    c_birth_year = extract(year from :c_birth_date);
    if (c_sex is not null) then
    begin
      if (c_sex = trim('0')) then
        c_sex = 'жен.';
      else
        c_sex = 'муж.';
    end

    if (kc_is_new_business = 1) then
      knc_is_new_buss_str = 'Да';
    else
      knc_is_new_buss_str = 'Нет';

    if (kc_is_registered = 1) then
      knc_is_regist_str = 'Да';
    else
      knc_is_regist_str = 'Нет';

    if (kc_status = 0) then
      knc_status_str = 'Участник';
    if (kc_status = 1) then
      knc_status_str = 'Лидер';
    if (kc_status = 2) then
      knc_status_str = 'Казначей';

--  Баланс кл.
    cl_balance = cl_dolya * k_balance;
    cl_llc = cl_dolya * k_llc;

    cl_zad_os = cl_dolya * zad_os;
    cl_zad_perc = cl_dolya * zad_perc;

    rep_deliq_perc = cl_dolya * rep_deliq_perc_kont;

    cl_delinq_balance = cl_dolya * k_delinq_balance;
    cl_delinq_os = cl_dolya * k_delinq_os;
--  Цель кредита
    if (kc_loan_target_prior <> kc_loan_target) then
    begin
      kc_loan_target_prior = kc_loan_target;
      select rd_name
      from refdata
      where rd_id = :kc_loan_target
      into :kc_loan_target_str;
    end

    if (cl_circle <= 1) then
      cl_new = 'новый';
    else
      cl_new = 'повторный';

--    Субтип бизнеса
--    Тип бизнеса кл.
    select bs_name, bt_name
    from bussubtype, bustype
    where bs_id = :busuness_subtype and
          bs_from_bt_id = bt_id
    into bs_name, bt_name;
--  Национальность, гражданство, семей. положение, образование и армия
    nation = null;
    citizen = null;
    family = null;
    education = null;
    army = null;
    with tabl1
    as (select iif(rd_main_code = 'Nation', rd_name, '') nation, iif(rd_main_code = 'Citizen', rd_name, '') citizen,
               iif(rd_main_code = 'Family', rd_name, '') family, iif(rd_main_code = 'Education', rd_name, '') education,
               iif(rd_main_code = 'Army', rd_name, '') army
        from refdata
        where rd_id in (:c_army, :c_education, :c_family, :c_citizen, :c_nation))
    select max(nation), max(citizen), max(family), max(education), max(army)
    from tabl1
    into nation, citizen, family, education, army;
--  Тип документа
    if (c_doc_type_prior <> c_doc_type) then
    begin
      c_doc_type_prior = c_doc_type;
      select pk_name
      from passkind
      where pk_id = :c_doc_type
      into c_doc_type_str;
    end
    c_is_resident_str = '';
    if (c_is_resident = 1) then
      c_is_resident_str = 'Резидент';

--    if (k_fact_close_date < :rep_date_end) then k_fact_close_date = k_fact_close_date + 1;
--    else k_fact_close_date = null;

    def30 = iif(k_delinq_day >= 30, 1, 0);
    idcontract = k_kolich_client_approved;
    ncontract = k_kolich_client_approved;
    loanterm = k_kredline_end - k_disb_date;

    c_inn = coalesce(c_inn, '');
    inn = iif(c_inn <> '', 1, 0);

    c_okpo = coalesce(c_okpo, '');
    okpo = iif(c_okpo <> '', 1, 0);

    c_email = coalesce(c_email, '');
    email = iif(c_email <> '', 1, 0);

    daystillpassportexpiry = c_pass_date_expire - k_disb_date;
    dayssincepassportissue = k_disb_date - c_pass_date;

    knc_account = coalesce(knc_account, '');
    clientaccount = iif(knc_account <> '', 1, 2);
    frombirthdatetoissuedate = k_disb_date - c_birth_date;

    suspend;
  end
