CREATE PROCEDURE GET_KONTPAYTASK_LIST(
    PROJECT D_STR_100)
RETURNS (
    KPT_ID D_INTEGER,
    ACCOUNT D_STR_20,
    AMOUNT D_CURRENCY,
    URL D_STR_100)
AS
declare variable BANK_ID D_INTEGER;
begin
  if (project = 'elsom') then
    exit;

  for
    select
      kpt.kpt_id,
      kpt.kpt_account,
      kpt.kpt_amount,
      kpt_bank_id
    from kontpaytasks kpt
    where kpt.kpt_project = :project
      and kpt.kpt_status = 0
      and kpt.kpt_error_text = ''
    into
      :kpt_id,
      :account,
      :amount,
      :bank_id
  do
  begin
    if (project = 'elsom') then
    begin
      url = coalesce((select rd_str_id from refdata where rd_id = :bank_id), '');
      if (url = '') then
        continue;
    end
    suspend;
  end
end
