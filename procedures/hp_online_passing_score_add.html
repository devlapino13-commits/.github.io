CREATE PROCEDURE HP_ONLINE_PASSING_SCORE_ADD(
    OPS_MIN_POINT D_INTEGER,
    OPS_DATE_BEGIN D_DATE,
    OPS_FROM_UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_255,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable ops_id d_id;
declare variable current_score d_integer;
declare variable str d_str_100;
begin
  -- процедура добавляет запись в online_passing_score - новый минимальный балл клиента для онлайн займа (RFCKG-2253)

  res_info_text = '';
  res_error_text = '';

  select ops_id,
         ops_min_point
  from online_passing_score
  where current_date between ops_date_begin and ops_date_end
  into ops_id,
       current_score;

  if (ops_min_point = current_score) then
    res_error_text = 'Ошибка: новый балл не отличается от текущего';

  if (coalesce(ops_min_point, 0) not between 3 and 100) then
    exception e_message 'Ошибка: минимальный балл должен быть в диапазоне 3-100';

  if (ops_date_begin is null) then
    res_error_text = 'Ошибка: не указана дата начала действия';

  if (ops_date_begin < current_date) then
    res_error_text = 'Ошибка: дата начала действия не может быть прошедшей датой';

  if (not exists(select *
                 from username
                 where un_id = :ops_from_un_id)) then
    res_error_text = 'Ошибка: пользователь не найден';

  if (res_error_text = '') then
  begin
    -- удаляем балл с будущей датой (если есть)
    select 'Удален балл на будущую дату - ' || ops_min_point || ', начало действия ' || f_datetostr(ops_date_begin)
    from online_passing_score
    where ops_date_begin > current_date
    into str;

    delete from online_passing_score
    where ops_date_begin > current_date;

    if (row_count = 0) then
      str = '';

    -- добавляем новый балл
    insert into online_passing_score (ops_date_begin, ops_min_point, ops_from_un_id)
    values (:ops_date_begin, :ops_min_point, :ops_from_un_id);

    if (row_count = 1) then
    begin
      if (str <> '') then
        res_info_text = str || ascii_char(13) || ascii_char(10) || ascii_char(13) || ascii_char(10);
      res_info_text = res_info_text || 'Добавлен новый минимальный балл - ' || ops_min_point || ', действует с ' || f_datetostr(ops_date_begin);
    end
    else
      res_error_text = 'Ошибка: новый минимальный балл добавить не удалось';
  end

  suspend;
end
