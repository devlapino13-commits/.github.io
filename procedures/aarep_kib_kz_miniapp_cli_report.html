CREATE PROCEDURE AAREP_KIB_KZ_MINIAPP_CLI_REPORT(
    MA_ID INTEGER,
    C_ID ID_TYPE)
RETURNS (
    CHECKED_DATE_TIME D_DATETIME,
    UN_FIO D_STR_255,
    DOB D_DATE,
    ALL_CREDIT_HISTORY_COUNT D_INTEGER,
    DOC_NUMBER D_STR_100,
    PRINCIPAL_AMOUNT D_STR_100,
    TYPE_CONTRACT D_STR_CODE,
    IS_GAMBLING D_SMALLINT,
    IS_STOP_CREDIT D_SMALLINT,
    IS_BANKRUPTCY D_SMALLINT,
    IS_TAX_DEBT D_SMALLINT,
    IS_FAKE_BUSINESS D_SMALLINT,
    IS_BAD_SUPPLIER D_SMALLINT,
    IS_BLACK_LIST D_SMALLINT,
    IS_WANTED_PERSON D_SMALLINT,
    IS_WANTED_ALIMENT D_SMALLINT,
    DATE_OF_CREDIT_START D_DATE,
    DATE_OF_CREDIT_END D_DATE,
    LOAN_TERM D_INTEGER,
    TOTAL_AMOUNT DECIMAL(18,2),
    OUTSTANDING_AMOUNT DECIMAL(18,2),
    CONTRACT_STATUS D_SMALLINT,
    EX_OVERDUE_MAX D_INTEGER,
    CLIENT_FIO D_STR_255,
    MA_KOLICH_REPAY D_INTEGER,
    MA_AMOUNT DECIMAL(18,2),
    MONTHLY_REPAY DECIMAL(18,2),
    COMMON_POINTS D_INTEGER,
    IS_OUR D_STR_20,
    SUBJECT D_STR_64,
    PORUCH_NAME D_STR_255,
    PAYMENT_STATUS D_STR_10,
    IS_ACTIVE D_STR_64,
    IS_DISPLAYED D_INTEGER,
    K_NO D_INTEGER,
    CURRENCY_CODE D_STR_10,
    NUMBER_OF_CONTRACTS D_INTEGER)
AS
declare variable BLS_ID D_ID;
declare variable FINANCIAL_INSTITUTION D_STR_255;
declare variable HAS_PRINT D_SMALLINT;
begin
  has_print = 0;
  --exception e_message(f_format('MA_FROM_REG_ID: %1  MA_ID: %2  C_ID: %3', MA_FROM_REG_ID, MA_ID, C_ID));

  select cl.c_name_fio,
         ma.ma_kolich_repay,
         mac.mac_amount
  from miniappclient mac
  inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
        and ma.ma_is_active = 1
        and ma.ma_date_end >= '31.12.2999'
  inner join clients cl on cl.c_id = mac.mac_from_c_id
  where ma.ma_id = :ma_id and
        mac.mac_from_c_id = :c_id
  order by ma.ma_number desc rows 1
  into: client_fio,
        ma_kolich_repay,
        ma_amount;

  monthly_repay = ma_amount / ma_kolich_repay;
  common_points = f_point(:c_id, current_date, 0);

  select eil.eil_from_bls_id, eil.eil_status_datetime
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eic.eic_code = 'pkb_cib_report'
      and eil.eil_object_kind = f_objkind('miniapp', 'id')
      and eil.eil_from_c_id = :c_id
      and eil.eil_status = 2
  order by eil.eil_status_datetime desc
  rows 1
  into :bls_id, checked_date_time;
  if (bls_id is null) then
    exit;

  select cl.c_name_fio
  from miniapp maa
  inner join username un on un.un_id = maa.ma_from_un_id
  inner join clients cl on cl.c_id = un.un_from_c_id
  where maa.ma_id = :ma_id
    and maa.ma_is_active = 1
    and maa.ma_date_end >= '31.12.2999' rows 1
  into :un_fio;


  select c_name_fio, dob, all_credit_history_count, doc_number, principal_amount,
            is_gambling, is_stop_credit, is_bankruptcy, is_tax_debt, is_fake_business,
            is_bad_supplier, is_black_list, is_wanted_person, is_wanted_aliment
  from hp_parse_pkb_report(:bls_id)
  rows 1
  into: client_fio, dob, all_credit_history_count, doc_number, principal_amount,
        is_gambling, is_stop_credit, is_bankruptcy, is_tax_debt, is_fake_business,
        is_bad_supplier, is_black_list, is_wanted_person, is_wanted_aliment;

  for select type_contract, date_of_credit_start, date_of_credit_end, loan_term,
            total_amount, outstanding_amount, contract_status, ex_overdue_max,
            financial_institution, poruch_name, payment_status, active_kont, k_no,
            currency_code, number_of_contracts
  from hp_parse_pkb_report(:bls_id)
  into: type_contract, date_of_credit_start, date_of_credit_end, loan_term,
        total_amount, outstanding_amount, contract_status, ex_overdue_max,
        financial_institution, poruch_name, payment_status, is_displayed, k_no,
        currency_code, number_of_contracts
  do
  begin
    has_print = 1;
    is_our = iif(:financial_institution not containing 'МБулак', 'В другом ФКУ', 'В МБулак');
    subject = iif(poruch_name = '', 'Заемщик', poruch_name);
    is_active = iif(contract_status = 1, 'Активный', 'Закрытый');
    suspend;
  end
  if (has_print = 0) then
    suspend;
end


/*

un_fio -- фио випса string(255)
