CREATE PROCEDURE HP_FT_INACTIVE_ASP_CANCEL
AS
declare variable c_id d_id;
declare variable cancel_date d_date;
declare variable asp_kind d_integer;
declare variable cell_app_kind d_integer;
declare variable push_body d_str_long;
declare variable is_overpay d_bool;
declare variable last_loan_blank_date d_date;
begin
  -- RFCRU-4869
  -- процедура ищет клиентов у кого все договоры закрыты более двух лет, нет активных заявок и прочих долгов/переплат а АСП активен
  -- и расторгает АСП с отправкой ПУШ, если
  asp_kind = f_edoc_kind('asp.agreement', 'id');
  cell_app_kind = f_edoc_kind('cell.application', 'id');

  for select kc_from_c_id
  from oksummary
  inner join knclient on kc_from_k_id = oks_from_k_id
  inner join edoc_list on edl_object_kind = 0
                      and edl_object_id = kc_from_c_id
                      and edl_from_edc_id = :asp_kind
                      and coalesce(edl_date_end, current_date) >= current_date
                      and edl_status >= 0
  where (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc) > 0 -- нет долгов по ОД + %%
  group by 1
  having max(oks_date_end) < dateadd(year, -2, current_date)
  into c_id
  do
  begin
    -- удостоверимся что нет открытых заявок
    if (exists(select 1 from hp_clientstate_get(:c_id) where res_client_state in (0,1))) then
      continue;

    -- нет прочих долгов
    if (exists(select 1
              from knclient, oksummary
              where kc_from_k_id = oks_from_k_id
                and current_date between oks_date_begin and oks_date_end
                and kc_from_c_id = :c_id
                and (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc + oks_out_other + oks_extra) > 0))
    then
      continue;

    -- нет переплат
    select iif(sum(ok_os) <> 0, 1, 0)
    from knclient, operkred
    where kc_from_k_id = ok_from_k_id
      and ok_ol_is_active = 1
      and ok_type = 9
      and kc_from_c_id = :c_id
    into is_overpay;

    if (is_overpay > 0) then
      continue;

    -- Переплату могли перенести на счет заморозки
    is_overpay = 0;

    select iif(coalesce(f_acc_bal(acc_id, current_date, 'sum'), 0) <> 0, 1, 0)
    from ref_account, ref_accbalance
    where ab_code = '60322'
    and acc_from_ab_id = ab_id
    and acc_add_code = '60322 (83)'
    and acc_object_kind = 0
    and acc_object_id = :c_id
    into is_overpay;

    if (is_overpay > 0) then
      continue;

    -- расторгаем АСП
    cancel_date = f_date_work_day(current_date + 1); -- дата расторжения - следующий рабочий день

    update edoc_list
    set edl_date_end = :cancel_date - 1
    where edl_object_kind = 0
      and edl_object_id = :c_id
      and edl_from_edc_id in (:asp_kind, :cell_app_kind)
      and coalesce(edl_date_end, :cancel_date) >= :cancel_date
      and edl_status >= 0;

    -- последняя дата договора. нужны подписанные договоры. ориентируемся на дату бланка,
    -- т.к. с 21.01.2026 в бланк договора добавили пункт о расторжении АСП через 2 года и оповещать не нужно
    select coalesce(max(edl_blank_date), '01.01.1900')
    from knclient, edoc_list, edoc_sign
    where edl_object_kind = 9
      and edl_object_id = kc_from_k_id
      and edl_from_edc_id = 3 --  f_edoc_kind('loan.agreement','id')
      and eds_from_edl_id = edl_id
      and eds_status = 1
      and kc_from_c_id = :c_id
    into last_loan_blank_date;
 
    if (last_loan_blank_date < '21.01.2026') then
    begin
      -- отправка ПУШ клиенту
      push_body = f_format('Уважаемый(ая) %1, уведомляем Вас о досрочном расторжении Договора на оказание информационных услуг заключенного с ООО «МКК М Булак» с %2.',
                           (select c_name_fio from clients where c_id = :c_id),
                           f_datetostr(cancel_date));
  
      f_send_push('clients', c_id, 'Расторжение договора АСП', push_body, 'hp_writeoff_asp_cancel', -1);
    end
  end
end
