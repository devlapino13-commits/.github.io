CREATE PROCEDURE CM_RU_0203_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable PR_CODE D_STR_100;
declare variable C_ID ID_TYPE;
declare variable DISB_DATE D_DATE;
declare variable CLI_BALANCE CURR_TYPE;
declare variable CLI_AMOUNT CURR_TYPE;
declare variable RE_DISB_DATE D_DATE;
declare variable LAST_ACTIVE D_SMALLINT;
declare variable CURR_ACTIVE D_SMALLINT;
declare variable PR_CODE_ACTIVE D_STR_100;
declare variable BAL_REFUND CURR_TYPE;
declare variable KOL_REFUND D_SMALLINT;
declare variable LAST_BALANCE CURR_TYPE;
declare variable LIMIT CURR_TYPE;
declare variable K_PERIOD CURR_TYPE;
declare variable DISB_DATE_REFUND D_DATE;
begin
    if (ma_id = 2460871) then
    exit;

  for
  select pr_code, mac_from_c_id, ma_date_oper, mac_amount from miniapp, product, miniappclient
  where ma_id = :ma_id
  and ma_is_active = 1
  and ma_date_end = '31.12.9999'
  and ((ma_refund = 1 and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%')
       or (pr_code containing '.support.'))
  and ma_from_pr_id = pr_id
  and ma_id = mac_from_ma_id
  and mac_amount > 0
  into pr_code, c_id, disb_date, cli_amount
  do
  begin
    -- у клиента не должно быть активной просрочки
    if (exists(
    select * from knclient,oksummary
    where kc_from_c_id = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id = oks_from_k_id
    and :disb_date between oks_date_begin and oks_date_end
    and oks_close_state = 0
    and :disb_date - oks_delinq_date > 0)) then
    begin
      res_error_text = 'У клиента имеется активная просрочка.';
      res_error_is_critical = 2;
      suspend;
    end

    if (pr_code containing '.support.') then
    begin
      res_error_text = 'По акции Поддержка выдачи запрещены.';
      res_error_is_critical = 2;
      suspend;
      exit;

      -- 1) если у клиента два займа, выданных до 1 марта по участвующим в акции продуктам, чтобы подучить новый заём «Поддержка», он должен рефинансировать оба займа;
      -- т.е. если имеется надежный сотрудник и параллельный то обе займа должны рефинанироваться чтобы получить поддержку
      last_active = null;
      select max(iif(pr_code containing '.reliable_client', 1, 0)) + max(iif(pr_code containing '.loyal.', 1, 0))
      from kontstate, kontract, product, knclient
      where ks_ol_is_delete = 0
      and ks_state = 1
      and ks_from_k_id = k_id
      and k_disb_date < '01.03.2022'  -- выданные до 1 марта 2022г.
      and :disb_date - 1 between ks_date_begin and ks_date_end -- проверяем на предыдущую дату
      and k_from_pr_id = pr_id
      and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%'
      and k_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = :c_id
      into last_active;
      last_active = coalesce(last_active, 0);

      -- проверка кол-во рефинансированных кредитов
      curr_active = null;
      select max(iif(pr_code containing '.reliable_client', 1, 0)) + max(iif(pr_code containing '.loyal.', 1, 0))
      from kontract, product, kontstate, knclient
      where k_refund = 1
      and ks_date_begin >= '01.03.2022'
      and k_from_pr_id = pr_id
      and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%'
      and k_id = ks_from_k_id
      and ks_state = 1
      and :disb_date between ks_date_begin and ks_date_end
      and k_id = kc_from_k_id
      and kc_amount_approved > 0
      and kc_from_c_id = :c_id
      into curr_active;
      curr_active = coalesce(curr_active, 0);

      if (curr_active = 0) then
      begin
        res_error_text = 'по продукту "Поддержка" могут получить рефинансированные займы';
        res_error_is_critical = 2;
        suspend;
      end else
      if (last_active > curr_active) then
      begin
        res_error_text = 'чтобы получить "Поддержка" нужно обе займа рефинансировать';
        res_error_is_critical = 2;
        suspend;
      end 

      k_period = null;
      select cast((max(sh_date) - min(sh_date)) as curr_type) / 30
      from schedule
      where sh_from_k_id = -:ma_id
      into k_period;
      k_period = round(k_period,1);
  
      if (k_period not between 5.5 and 6.5) then
      begin
        res_error_text = 'Срок займа не соответсвует параметрам продукта';
        res_error_is_critical = 2;
        suspend;
      end
    end else
    begin
      -- вычисляем баланс рефинансированных займов полученные от 1 марта
      kol_refund = 0;
      limit = 20000;
      for
      select iif(pr_code containing '.reliable_client', '.reliable_client', '.loyal.'), k_disb_date, sum(oks_balance)
      from knclient, kontract, product, oksummary
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and kc_from_k_id = oks_from_k_id
      and :disb_date between oks_date_begin and oks_date_end
      and oks_close_state = 0
      and oks_from_k_id = k_id
      and k_refund = 1
      and k_disb_date >= '01.03.2022'
      and k_from_pr_id = pr_id
      and pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.)%'
      group by 1, 2
      into pr_code_active, disb_date_refund, bal_refund
      do
      begin
        kol_refund = kol_refund + 1;
        if (kol_refund >= 2 and 1<>1) then   -- убрал по письму от Лауры от 17.04.2024
        begin
          res_error_text = 'Клиент 2 раза рефинансировался.';
          res_error_is_critical = 2;
          suspend;
          exit;
        end

        -- вычисляем предыдущий баланс рефинансированного займа
        select oks_balance from knclient, kontract, product, oksummary
        where kc_from_c_id = :c_id
        and kc_amount_approved > 0
        and kc_from_k_id = oks_from_k_id
        and :disb_date_refund - 1 between oks_date_begin and oks_date_end
        and oks_close_state = 0
        and oks_from_k_id = k_id
        and k_from_pr_id = pr_id
        and pr_code containing trim(:pr_code_active)
        and k_disb_date < '01.03.2022'
        order by k_disb_date desc
        rows 1
        into last_balance;

        limit = limit - (bal_refund - last_balance);
        if (limit > 20000) then limit = 20000;

      end 


--      if (pr_code containing '.reliable_client') then pr_code = '.reliable_client';
--      else pr_code = '.loyal.';

      -- находим находим остаток
--      cli_balance = null;
--      select oks_balance, k_disb_date from knclient, kontract, product, oksummary

--      and :disb_date between oks_date_begin and oks_date_end
--      and oks_close_state = 0
--      and oks_from_k_id = k_id
--      and k_from_pr_id = pr_id
--      and pr_code containing :pr_code
--      order by k_disb_date desc
--      rows 1
--      into cli_balance, re_disb_date;

--      cli_balance = coalesce(cli_balance,0);

--      begin

--          res_error_is_critical = 2;
--          suspend;
--        end
--      end else
--      begin
--        res_error_text = 'Рефинансирование запрещено';  -- RFCRU-3486
--        res_error_is_critical = 2;
--        suspend;
--      end

--      if (re_disb_date >= '01.03.2022') then
--      begin
--        res_error_text = 'Рефинансируются выданные до 1 марта 2022.';
--        res_error_is_critical = 2;
--        suspend;
--      end
    end
  end
end
