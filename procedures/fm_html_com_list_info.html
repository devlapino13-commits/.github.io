CREATE PROCEDURE FM_HTML_COM_LIST_INFO(
    CML_ID D_ID,
    UN_ID D_ID,
    SOURCE_SYSTEM D_SMALLINT)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable TMP_STR_LONG D_STR_8100;
declare variable CAPTION D_STR_CODE;
declare variable URI D_STR_4096;
declare variable DESCRIPTION D_STR_255;
declare variable EDIT_KIND D_STR_CODE;
declare variable MAX_LENGTH D_SMALLINT;
declare variable PARAM_NAME D_STR_CODE;
declare variable PARAM_VALUE D_STR_CODE;
declare variable PARAM_STR_VALUE D_STR_255;
declare variable CML_STATUS D_SMALLINT;
declare variable LANG D_STR_LANG;
declare variable CML_IS_CLOSED D_SMALLINT;
declare variable CMC_COM_KIND D_STR_CODE;
declare variable CML_ADD_PARAMS D_STR_LONG;
declare variable RES_USER D_STR_100;
declare variable RES_DATETIME D_DATETIME;
declare variable RES_IS_ACTION D_SMALLINT;
declare variable RES_MEMO D_STR_LONG;
declare variable CML_DATE D_DATE;
declare variable CMC_ID D_INTEGER;
declare variable CMCP_DEFAULT_VALUE D_STR_CODE;
declare variable CMCP_DEFAULT_TEXT D_STR_CODE;
declare variable CMCP_READ_ONLY D_SMALLINT;
declare variable OBJ_KIND D_STR_CODE;
declare variable OBJ_ID D_ID;
declare variable KF_NAME D_STR_100;
declare variable C_ID D_ID;
declare variable C_NAME_FIO D_STR_100;
declare variable PR_NAME D_STR_255;
declare variable DB_ID D_INTEGER;
declare variable IS_MANAGER D_SMALLINT;
declare variable CMC_CODE D_STR_CODE;
begin
  -- Процедура готовится как для МП, так и для Finstart
  -- Для Finstart ссылки нужно прятать

  lang = f_get_lang_code(un_id, null);
  db_id = f_get_db_id();

  is_manager = 0;
  if (db_id = 81001) then
  begin
    -- Проверка на менеджера нужна только в РФ, в остальных не используем
    if (exists(select *
       from hr_pstatus, joblist, jobkind, username
       where un_id = :un_id and
             ps_date_begin <= current_date and
             ps_date_end >= current_date and
             ps_is_working = 1 and
             ps_status = 1 and
             ps_from_jl_id = jl_id and
             jl_kind = jk_id and
             un_from_c_id = ps_from_c_id and
             jk_system_code = 'JobSystem.Region_KM')) then
      is_manager = 1;
  end

  res_text = (select res_style from fm_html_styles) || '<body>' ||
    f_format('<h2>%1</h2>', f_lang(lang, 'Общая информация'));


  select f_objkind(cml_object_kind, 'name'), cml_object_id
  from com_list
  where cml_id = :cml_id
  into obj_kind, obj_id;

  if (obj_kind = 'kontract') then
  begin
    select kal_new_kf_name, pr_name
      from knaktlist, kontract, product
      where k_id = :obj_id
      and k_from_kf_id = kal_from_kf_id
      and knaktlist.is_delete = 0
      and current_date between kal_date and kal_date_end
      and k_from_pr_id = pr_id
      into kf_name, pr_name;
  end
  kf_name = coalesce(kf_name , '');
  pr_name = coalesce(pr_name, '');

  select
    f_format('%1: <b>%2</b><br>', f_lang(:lang, 'Тип запроса'), cmc_name) ||
    f_format('%1: <b>%2</b><br>', f_lang(:lang, 'Объект'), f_objname(cml_object_kind, cml_object_id)) ||
    iif(:kf_name <> '', f_format('%1: <b>%2</b><br>', f_lang(:lang, 'Кредитная папка'), :kf_name), '') ||
    f_format('%1: <b>%2</b>', f_lang(:lang, 'Дата'), f_datetostr(cml_date)) || ' $$$<br>' ||
    f_format('%1: <b>%2</b><br>', f_lang(:lang, 'Продукт'), :pr_name) ||
    f_format('%1: <b>%2</b><br>', f_lang(:lang, 'Офис'), reg_name) ||
    f_format('%1: <b>%2</b><br>', f_lang(:lang, 'Заявитель'), c_name_fio) ||
    f_format('%1: <b>%2</b><br>', f_lang(:lang, 'Статус'), f_lang(:lang, f_enum('com_action_status', cml_last_status, 'name'))),
    cmc_id, f_enum('com_kind', cmc_com_kind, 'code'), cml_last_status, cml_is_closed, cml_add_params, cml_date, cmc_code
  from com_list, com_cfg, region, clients
  where cml_id = :cml_id and
        cml_from_cmc_id = cmc_id and
        cml_from_reg_id = reg_unicode and
        cml_init_c_id = c_id
  into tmp_str_long, cmc_id, cmc_com_kind, cml_status, cml_is_closed,
    cml_add_params, cml_date, cmc_code;

  if (cml_date <> current_date and
      f_enum('com_action_status', cml_status, 'code') in ('new', 'rework')
      and source_system = 1
    ) then
  begin
    uri = f_uri_sql(f_lang(lang, 'Обновить'), 'com_date_update',
      f_objkind('com_list', 'id'), cml_id, 'none',
      f_lang(lang,  'Дата запроса будет изменена на текущую, продолжить?'),
      null, null, null, null, null, 'dialog', null);
    tmp_str_long = replace(tmp_str_long, '$$$', uri);
  end
  else
    tmp_str_long = replace(tmp_str_long, '$$$', '');

  res_text = BLOB_APPEND(res_text, tmp_str_long);

  -- Для реструктуризируемых контрактов выводим клиентов
  if (obj_kind = 'kontract') then
  begin
    tmp_str_long = f_format(
      '<br><table width="100%" class="msg"><col width="100%"><tr><th>%1</th></tr>',
      f_lang(lang, 'Клиенты договора'));

    for select c_id, c_name_fio
    from knclient, clients
    where kc_from_k_id = :obj_id
      and kc_from_c_id = c_id
      and kc_amount_approved > 0
    into c_id, c_name_fio do
    begin
      if (source_system = 1) then
        uri = f_uri_open(c_name_fio, 'client', 'Clients', c_id, c_name_fio, '', un_id);
      else
        uri = c_name_fio;
      tmp_str_long = tmp_str_long ||'<tr><td>' || uri || '</td></tr>';
    end
    tmp_str_long = tmp_str_long || '</table>';
  end
  res_text = BLOB_APPEND(res_text, tmp_str_long);

  -- Отображаем таблицу параметров параметров.
  -- Если еще не отправлен на КК, то даем редактировать.
  res_text = BLOB_APPEND(res_text, f_format(
    '<br><table width="100%" class="msg"><col width="40%"><col width="40%">' ||
    '<col width="20%"><tr><th colspan="3">%1</th></tr>',
    f_lang(lang, 'Параметры реструктуризации')));


  for select cmcp_param_name, cmcp_caption, cmcp_description, cmcp_edit_kind,
             cmcp_max_length, cmcp_default_value, cmcp_default_text, cmcp_read_only
  from com_cfg_params
  where cmcp_from_cmc_id = :cmc_id and
        cmcp_is_hidden = 0
  order by cmcp_order
  into :param_name, :caption, :description, :edit_kind,
       :max_length, :cmcp_default_value, :cmcp_default_text, :cmcp_read_only
  do
  begin
    param_value = f_json_str_get_value(cml_add_params, param_name, cmcp_default_value);

    if (edit_kind in ('text', 'date')) then
    begin
      if (param_value = cmcp_default_value)
      then param_str_value = cmcp_default_text;
      else param_str_value = param_value;
    end

    if (edit_kind = 'refdata') then
      edit_kind = 'combo';

    -- Получаем текстовое представление значения для выпадающих списков
    if (edit_kind = 'combo') then
    begin
      param_str_value = f_format(f_lang(lang, 'Значение: %1'), param_value);
      select m_name
      from fm_report_cmd_mru('cml_set_rest_' || :param_name, f_objkind('com_list', 'id'), :cml_id, :un_id)
      where m_id = :param_value
      into :param_str_value;
    end

    caption = trim(caption);
    param_value = trim(param_value);

    if (f_enum('com_action_status', cml_status, 'code') in ('new', 'rework') and source_system = 1 and cmcp_read_only = 0) then
    begin
      uri = f_uri_sql(f_lang(lang, 'Изменить'), 'cml_set_rest_' || param_name,
        f_objkind('com_list', 'id'), cml_id, edit_kind, description, '', 1,
        max_length, param_value, '', 'dialog', param_name);
      res_text = BLOB_APPEND(res_text,
        '<tr><td>' || caption ||
        '</td><td align="center">' || param_str_value ||
        '</td><td align="center">' || uri || '</td></tr>');
    end
    else
    res_text = BLOB_APPEND(res_text,
      f_format(
        '<tr><td>%1</td><td align="center" colspan="2">%2</td></tr>',
        caption, param_str_value));
  end
  res_text = BLOB_APPEND(res_text, '</table>');

  if (cml_is_closed = 0 and source_system = 1) then
  begin
    res_text = BLOB_APPEND(res_text, f_format('<h2>%1</h2>', f_lang(lang, 'Действия')));
  
    -- Графики новый и старый
    if (cmc_com_kind in ('reschedule','termination')) then
    begin
      res_text = BLOB_APPEND(res_text, f_uri_html(f_lang(lang, 'Показать график'),
        'com_list_reschedule', f_lang(lang, 'Графики погашений'),
        'com_list', cml_id, null, null, '') || '<br><br>');
    end

    -- Отправка на КК
    if (f_enum('com_action_status', cml_status, 'code') in ('new', 'rework')) then
    begin

      if (db_id <> 81001 or is_manager = 1) then
      begin
        res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(lang, 'Отправить на КК'),
          'com_list_send_kk', f_objkind('com_list', 'id'), cml_id, 'none',
          f_lang(lang, 'Подтвердите отправку запроса на КК'), '', 0, 0, '', '',
          'dialog', '') || '<br><br>');
      end

      res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(:lang, 'Отменить'),
        'com_list_cancel_kk', f_objkind('com_list', 'id'), cml_id, 'none',
        f_lang(lang, 'Подтвердите отмену. Запрос будет исключен из списка КК'),
        '', 0, 0, '', '', 'dialog', '') || '<br><br>');
    end
  
    -- Добавление комментария
    res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(lang, 'Добавить комментарий'),
      'com_memo_add', f_objkind('com_list', 'id'), cml_id, 'text',
      f_lang(lang, 'Введите комментарий'), '', 5, 1024, '',
      f_lang(lang, 'Комментарий не введен'), 'dialog', '') || '<br><br>');

    -- Добавление документа
    res_text = BLOB_APPEND(res_text, f_uri_open(f_lang(lang, 'Добавить документ'),
      'com_blob', f_objkind('com_list', 'id'), cml_id,
      f_lang(lang, 'Добавить документ'), '', un_id) || '<br><br>');
  end

  if (cmc_com_kind in ('reschedule', 'termination') and source_system = 0) then
  begin
    res_text = BLOB_APPEND(res_text, (select res_text
      from fm_html_com_reschedule(:cml_id, :un_id, 1)) || '<br>');
  end

  if (source_system = 1) then
  begin
    res_text = BLOB_APPEND(res_text, f_format(
      '<table width="100%" class="msg"><col width="70%"><col width="30%">' ||
      '<tr><th colspan="2">%1</th></tr>', f_lang(lang, 'Действия и комментарии')));
  
    for select res_user, res_datetime, res_memo, res_is_action
    from hp_com_list_memo(:cml_id, 0, :un_id)
    into res_user, res_datetime, res_memo, res_is_action
    do
    begin
      res_text = BLOB_APPEND(res_text,
        f_format(
          '<tr><td class="msg_header">%1</td><td class="msg_header" align="right">%2</td></tr>',
          res_user, f_datetimetostr(res_datetime)) ||
        f_format(
          '<tr><td class="msg_text" colspan="2"><font color="%1">%2</font></td></tr>',
          trim(iif(res_is_action=0, 'blue', 'black')), res_memo));
    end
    res_text = BLOB_APPEND(res_text, '</table>');
  end

  res_text = BLOB_APPEND(res_text, '</body>');
  suspend;
end
