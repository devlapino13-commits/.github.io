CREATE PROCEDURE AAREP_MONEY_SALD_NEW(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    RES_SALDO_CASH DECIMAL(18,4),
    RES_SALDO_BANK DECIMAL(18,4),
    RES_SALDO_PORT DECIMAL(18,4))
AS
declare variable s1 decimal(18,4);
declare variable al_id varchar(20);
declare variable res_acc_kind integer;
declare variable atl_acc_mask varchar(20);
begin
-- Sanjari, 19.07.2012

  res_saldo_cash = 0;
  res_saldo_bank = 0;
  res_saldo_port = 0;

  for select atl_from_at_id, atl_acc_mask
      from acctypelist
      where atl_from_at_id in (1, 11, 12)
      into :res_acc_kind, atl_acc_mask
  do
  begin
    for select al_id
        from acc_list, region_list(:report_region_id, 1, -1, 1)
        where al_from_bg_id starting with :atl_acc_mask and
              al_from_reg_id = id_list
        into :al_id
    do
    begin
      s1 = f_al_bal(al_id, report_date_end);

      if (res_acc_kind = 11) then
        res_saldo_cash = res_saldo_cash + s1;
      if (res_acc_kind = 12) then
        res_saldo_bank = res_saldo_bank + s1;
      if (res_acc_kind = 1) then
        res_saldo_port = res_saldo_port + s1;
    end
  end
  res_saldo_cash = cast(res_saldo_cash as decimal(15,2));
  res_saldo_bank = cast(res_saldo_bank as decimal(15,2));
  res_saldo_port = cast(res_saldo_port as decimal(15,2));
  suspend;
end
