CREATE PROCEDURE CM_AZ_0027_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable c_id id_type;
declare variable ovs_value str_255;
begin
    for select kncl.kc_from_c_id,
               cl.c_name_fio
    from knclient kncl
    inner join clients cl on kncl.kc_from_c_id = cl.c_id
    where kncl.kc_amount_approved >0
    and kncl.kc_from_k_id = :k_id
    into :c_id, :res_info_text
    do
    begin
        select ovs_value from  HP_OBJECTPROPS_GET(0, :c_id, -1, 'AnketaQuestions.communal_expenses', current_date)
        into :ovs_value;
        if (ovs_value is not null) then
        begin
           ovs_value = trim(replace(ovs_value, 'təsviri: ', ''));
           if (ovs_value <> '') then
           begin
              if (not (ovs_value similar to '[-]?[[:DIGIT:]]+')) then
              begin
                  res_error_is_critical = 2;
                  res_error_text = 'В Анкете клиента, вопрос: "Расход на жилье и ком.платежи (в месяц)" ошибка в ответе. Укажите сумму в AZN в месяц. Сейчас значение:'||ovs_value;
                  suspend;
              end
           end
        end
        ------------------
        select ovs_value from  HP_OBJECTPROPS_GET(0, :c_id, -1, 'AnketaQuestions.living_expenses', current_date)
        into :ovs_value;
        if (ovs_value is not null) then
        begin
           ovs_value = trim(replace(ovs_value, 'təsviri: ', ''));
           if (ovs_value <> '') then
           begin
              if (not (ovs_value similar to '[-]?[[:DIGIT:]]+')) then
              begin
                  res_error_is_critical = 2;
                  res_error_text = 'В Анкете клиента, вопрос: "Расход на проживание (в месяц)" ошибка в ответе. Укажите сумму в AZN в месяц. Сейчас значение:'||ovs_value;
                  suspend;
              end
              /*
              else if (ovs_value = '0') then
              begin
                res_error_is_critical = 2;
                res_error_text = 'В Анкете клиента, вопрос: "Расход на проживание (в месяц)" значение не может быть = 0';
                suspend;
              end
              */
           end
           /*
           else
           begin
              res_error_is_critical = 2;
              res_error_text = 'В Анкете клиента, вопрос: "Расход на проживание (в месяц)" ответ не заполнен';
              suspend;            
           end  */
        end
        else
        begin
          res_error_is_critical = 2;
          res_error_text = 'В Анкете клиента, вопрос: "Расход на проживание (в месяц)" ответ не заполнен';
          suspend;            
        end
        ------------
        select ovs_value from  HP_OBJECTPROPS_GET(0, :c_id, -1, 'work_monthly_income', current_date)
        into :ovs_value;
        if (ovs_value is not null) then
        begin
           ovs_value = trim(replace(ovs_value, 'təsviri: ', ''));
           if (ovs_value <> '') then
           begin
              if (not (ovs_value similar to '[-]?[[:DIGIT:]]+')) then
              begin
                  res_error_is_critical = 2;
                  res_error_text = 'В Анкете клиента, вопрос: "Ежемесячный доход" ошибка в ответе. Укажите сумму в AZN в месяц. Сейчас значение:'||ovs_value;
                  suspend;
              end
           end
        end
        ----
        select ovs_value from  HP_OBJECTPROPS_GET(0, :c_id, -1, 'work_extra_amount', current_date)
        into :ovs_value;
        if (ovs_value is not null) then
        begin
           ovs_value = trim(replace(ovs_value, 'təsviri: ', ''));
           if (ovs_value <> '') then
           begin
              if (not (ovs_value similar to '[-]?[[:DIGIT:]]+')) then
              begin
                  res_error_is_critical = 2;
                  res_error_text = 'В Анкете клиента, вопрос: "Сумма доп. дохода (в месяц)" ошибка в ответе. Укажите сумму в сомах в месяц. Сейчас значение:'||ovs_value;
                  suspend;
              end
           end
        end
    end
end
