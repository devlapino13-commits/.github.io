CREATE PROCEDURE STAROFFICES_UPDATE(
    REG_ID DECIMAL(18,0))
AS
declare variable d date;
declare variable d_prior date;
declare variable chet integer;
declare variable k_no decimal(18,0);
begin
  /* Procedure Text */
  chet = 0;
  for select so_date_begin, so_number
      from staroffices
      where so_from_reg_id = :reg_id
      order by so_date_begin desc, so_number desc
      into :d, :k_no
  do
  begin
    chet = :chet + 1;
    if (chet > 1) then
    begin
      update staroffices
      set so_date_end = :d_prior - 1
      where so_from_reg_id = :reg_id and
            so_number = :k_no and
            so_date_end <> :d_prior - 1;
    end
    else
    begin
      update staroffices
      set so_date_end = '31.12.9999'
      where so_from_reg_id = :reg_id and
            so_number = :k_no and
            so_date_end <> '31.12.9999';
    end
    d_prior = d;
  end
end
