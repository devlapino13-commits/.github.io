CREATE PROCEDURE PRINT_CLIENT_BEN(
    REP_C_ID ID_TYPE)
RETURNS (
    BEN_C_ID ID_TYPE,
    BEN_C_TYPE INTEGER,
    BEN_C_NAME_FIO STR_100,
    BEN_C_NAME_FULL STR_255,
    BEN_C_BIRTH_DATE DATE,
    BEN_C_PASS_NO STR_100,
    BEN_C_PASS_DATE DATE,
    BEN_C_PASS_WHO STR_255,
    BEN_C_PASS_DATE_EXPIRE DATE,
    BEN_C_SEX INTEGER,
    BEN_C_IS_RESIDENT INTEGER,
    BEN_C_BIRTH_CITY STR_100,
    BEN_C_INN VARCHAR(20),
    BEN_C_NATION_STR STR_100,
    BEN_C_CITIZEN_STR STR_255,
    BEN_C_FAMILY_STR STR_255,
    BEN_C_CELL_NO STR_100,
    BEN_C_EMAIL STR_100,
    BEN_C_ADDRESS_YUR STR_255,
    BEN_C_ADDRESS_FACT STR_255,
    BEN_CB_PERCENT DECIMAL(5,2),
    BEN_C_PFT_RISK_KOEF INTEGER,
    BEN_C_PFT_RISK_MEMO VARCHAR(1024),
    BEN_C_IS_PUBLIC INTEGER,
    BEN_CLI_IS_VERIF INTEGER,
    BEN_CLI_VERIF_DATE DATE,
    BEN_C_ALTERNAT_FIO D_STR_100,
    BEN_C_OKPO D_STR_100,
    BEN_C_FORM_STATUS_LONG D_STR_255,
    BEN_C_FORM_STATUS_SHORT D_STR_20,
    BEN_C_ORG_TYPE_LONG D_STR_255,
    BEN_C_ORG_TYPE_SHORT D_STR_20)
AS
begin

  for
    select
      c_id,
      c_type,
      cb_ben_name,
      c_name_full,
      c_birth_date,
      c_pass_no,
      c_pass_date,
      c_pass_who,
      c_pass_date_expire,
      c_sex,
      c_is_resident,
      c_birth_city,
      c_inn,
      c_pft_risk_koef,
      c_pft_risk_memo,
      c_is_public,
      rd1.rd_name,
      rd2.rd_name,
      rd3.rd_name,
      cdt1.cont_value,
      cdt2.cont_value,
      addr1.adc_address_str,
      addr2.adc_address_str,
      cb_percent,
      c_alternat_fio,
      c_okpo,
      c_form_status_long,
      c_form_status_short,
      c_org_type_long,
      c_org_type_short
    from clientbenefic
      left join clients
        on c_id = cb_ben_from_c_id
      left join refdata as rd1
        on rd1.rd_id = c_nation
      left join refdata as rd2
        on rd2.rd_id = c_citizen
      left join refdata as rd3
        on rd3.rd_id = c_family
      left join hp_contdata_get_cell_main(0, c_id) as cdt1
        on 1 = 1
      left join hp_contdata_get_email_main(0, c_id) as cdt2
        on 1 = 1
      left join hp_addrclient(c_id) addr1
        on addr1.adc_address_type = 1
      left join hp_addrclient(c_id) addr2
        on addr2.adc_address_type = 2
    where cb_from_c_id = :rep_c_id and cb_is_owner = 1
  into
    :ben_c_id,
    :ben_c_type,
    :ben_c_name_fio,
    :ben_c_name_full,
    :ben_c_birth_date,
    :ben_c_pass_no,
    :ben_c_pass_date,
    :ben_c_pass_who,
    :ben_c_pass_date_expire,
    :ben_c_sex,
    :ben_c_is_resident,
    :ben_c_birth_city,
    :ben_c_inn,
    :ben_c_pft_risk_koef,
    :ben_c_pft_risk_memo,
    :ben_c_is_public,
    :ben_c_nation_str,
    :ben_c_citizen_str,
    :ben_c_family_str,
    :ben_c_cell_no,
    :ben_c_email,
    :ben_c_address_yur,
    :ben_c_address_fact,
    :ben_cb_percent,
    :ben_c_alternat_fio,
    :ben_c_okpo,
    :ben_c_form_status_long,
    :ben_c_form_status_short,
    :ben_c_org_type_long,
    :ben_c_org_type_short



  do
  begin
    ben_cli_is_verif = null;
    ben_cli_verif_date = null;

    if (ben_c_id > 0 ) then
    select iif(al.acl_status = 2, 1, 0),
           iif(al.acl_status = 2, al.acl_date, null)
    from actionlist al
    where al.acl_from_rd_id in (f_get_rd_id('ActionGroup.Passport_Verify'), f_get_rd_id('ActionGroup.Passport_Verify_OiVO'))
    and al.acl_object_kind = 0
    and al.acl_object_id = :ben_c_id
    --and al.acl_date <= :rep_date
    rows 1
    into :ben_cli_is_verif, ben_cli_verif_date;
    suspend;
  end
end
