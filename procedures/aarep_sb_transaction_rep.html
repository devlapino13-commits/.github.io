CREATE PROCEDURE AAREP_SB_TRANSACTION_REP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SB_KIND INTEGER)
RETURNS (
    SBA_FROM_K_ID DECIMAL(18,0),
    SBA_FROM_SBL_ID INTEGER,
    SBA_DATE_BEGIN DATE,
    SBA_DATE_END DATE,
    SBA_REP_BEGIN DATE,
    SBA_REP_END DATE,
    SBL_NAME VARCHAR(255),
    SBL_TYPE_STR VARCHAR(20),
    REP_REPAY_LAST_DATE DATE,
    REP_REPAY_OS DECIMAL(18,4),
    REP_REPAY_PERC DECIMAL(18,4),
    REP_REPAY_FINE DECIMAL(18,4),
    REP_BEGIN_BALANCE DECIMAL(18,4),
    REP_BEGIN_OS DECIMAL(18,4),
    REP_BEGIN_PERC DECIMAL(18,4),
    REP_END_BALANCE DECIMAL(18,4),
    REP_END_OS DECIMAL(18,4),
    REP_END_PERC DECIMAL(18,4),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    REP_STATUS_BEGIN VARCHAR(50),
    REP_STATUS_END VARCHAR(50),
    REP_KOL_BEGIN INTEGER,
    REP_KOL_REPAY INTEGER,
    REP_KOL_END INTEGER,
    REP_KOL_TO_SB INTEGER,
    REP_KOL_FROM_SB INTEGER,
    REP_KOL_NORMAL INTEGER,
    REP_KOL_DELINQ INTEGER,
    REP_KOL_CLOSED INTEGER,
    REP_KF_NAME VARCHAR(100),
    REP_LO_NAME VARCHAR(100))
AS
declare variable sba_number integer;
declare variable sbl_type integer;
declare variable is_active_begin integer;
declare variable is_active_end integer;
declare variable d1 date;
declare variable d2 date;
declare variable delinq_date date;
declare variable k_from_kf_id decimal(18,0);
begin
  for select sba_from_k_id, sba_number, sba_date_begin, sba_date_end, sba_from_sbl_id, sbl_name, sbl_type,
             k_kontract_no, k_disb_date, k_from_kf_id
      from sb_akt, sb_list, kontract, region_list(:rep_region_id, 1, -1, 1)
      where sba_is_delete = 0 and
            sba_status = 2 and
            sba_date_begin <= :rep_date_end and
            sba_date_end >= :rep_date_begin and
            sba_date_begin <= sba_date_end and
            sba_from_sbl_id = sbl_id and
            sba_from_k_id = k_id and
            coalesce(sbl_from_reg_id, 1000000) = id_list and
            sbl_id <> 0 and
            ((:rep_sb_kind = -1) or
            (:rep_sb_kind >= 0 and
            sbl_type = :rep_sb_kind))
      into :sba_from_k_id, :sba_number, :sba_date_begin, :sba_date_end, :sba_from_sbl_id, :sbl_name, :sbl_type,
           :k_kontract_no, :k_disb_date, :k_from_kf_id
  do
  begin
    rep_kol_begin = 0;
    rep_kol_repay = 0;
    rep_kol_to_sb = 0;
    rep_kol_from_sb = 0;
    rep_kol_end = 0;
    rep_kol_normal = 0;
    rep_kol_delinq = 0;
    rep_kol_closed = 0;

    rep_end_balance = 0;
    rep_end_os = 0;
    rep_end_perc = 0;

    rep_begin_os = 0;
    rep_begin_perc = 0;
    rep_begin_balance = 0;

    is_active_begin = 0;
    is_active_end = 0;

    rep_lo_name = '';
    rep_kf_name = '';
    select kal_new_kf_name, c_name_fio
    from knaktlist, clients
    where kal_from_kf_id = :k_from_kf_id and
          knaktlist.is_delete = 0 and
          kal_date <= :rep_date_end and
          kal_date_end >= :rep_date_end and
          kal_new_kf_officer = c_id
    into :rep_kf_name, :rep_lo_name;

    sba_rep_begin = rep_date_begin;
    sba_rep_end = rep_date_end;
    if (sba_date_begin > rep_date_begin) then
      sba_rep_begin = sba_date_begin;
    if (sba_date_end < rep_date_end) then
      sba_rep_end = sba_date_end;

    if (sbl_type = 0) then
      sbl_type_str = 'сб';
    else
      sbl_type_str = 'коллекторы';

    rep_repay_last_date = null;
    select ok_ol_date
    from operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :sba_rep_end and
          ok_from_k_id = :sba_from_k_id and
          ok_type = 4 and
          ok_from_c_id = 0
    order by ok_ol_date desc
    rows 1
    into :rep_repay_last_date;

    d1 = null;
    d2 = null;
    select ks_date_begin, ks_date_end
    from kontstate
    where ks_from_k_id = :sba_from_k_id and
          ks_ol_is_delete = 0 and
          ks_state = 1
    into :d1, d2;
    if (d1 is not null) then
    begin
      if ((d1 <= (sba_rep_begin - 1)) and
          (d2 >= (sba_rep_begin - 1))) then
        is_active_begin = 1;
    end
    if (d2 is not null) then
    begin
      if ((d1 <= sba_rep_end) and
          (d2 >= sba_rep_end)) then
        is_active_end = 1;
    end

    if (sba_date_begin >= :rep_date_begin) then
      rep_kol_to_sb = 1;
    if (sba_date_end < :rep_date_end) then
      rep_kol_from_sb = 1;
    if (d2 between :sba_rep_begin and sba_rep_end) then
      rep_kol_closed = 1;
    if (sba_date_begin <= (:rep_date_begin - 1) and
        sba_date_end >= (:rep_date_begin - 1) and
        is_active_begin = 1) then
      rep_kol_begin = 1;
    if (sba_date_begin <= :rep_date_end and
        sba_date_end >= :rep_date_end and
        is_active_end = 1) then
      rep_kol_end = 1;

    if (is_active_begin = 1) then
    begin
      delinq_date = null;
      rep_begin_balance = 0;
      select oks_balance, oks_delinq_date
      from oksummary
      where oks_from_k_id = :sba_from_k_id and
            oks_date_begin <= (:sba_rep_begin - 1) and
            oks_date_end >= (:sba_rep_begin - 1) and
            oks_close_state = 0
      into :rep_begin_balance, :delinq_date;
      if (rep_begin_balance is null) then
        rep_begin_balance = 0;
      if (rep_begin_balance = 0) then
      begin
        rep_kol_begin = 0;
        rep_kol_from_sb = 0;
        rep_kol_to_sb = 0;
      end

      if ((delinq_date is not null) and
          (delinq_date <= (sba_rep_begin - 1)) and
          (rep_begin_balance > 0)) then
      begin
        select sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_ol_date <= (:sba_rep_begin - 1) and
              ok_sh_date <= (:sba_rep_begin - 1) and
              ok_ol_is_active = 1 and
              ok_from_k_id = :sba_from_k_id and
              ok_type < 7 and
              ok_from_c_id = 0
        into :rep_begin_os, :rep_begin_perc;
        if (rep_begin_os is null) then
          rep_begin_os = 0;
        if (rep_begin_perc is null) then
          rep_begin_perc = 0;
      end
    end

    if (rep_kol_end = 1) then
    begin
      delinq_date = null;
      rep_end_balance = 0;
      select oks_balance, oks_delinq_date
      from oksummary
      where oks_from_k_id = :sba_from_k_id and
            oks_date_begin <= :sba_rep_end and
            oks_date_end >= :sba_rep_end and
            oks_close_state = 0
      into :rep_end_balance, delinq_date;
      if (rep_end_balance is null) then
        rep_end_balance = 0;
      if (rep_end_balance = 0) then
        rep_kol_end = 0;
      if ((rep_kol_begin = 1) and
          (rep_end_balance = 0)) then
        rep_kol_closed = 1;

      if ((delinq_date is not null) and
          (delinq_date <= sba_rep_end) and
          (rep_end_balance > 0)) then
      begin
        if (rep_kol_end = 1) then
          rep_kol_delinq = 1;

        select sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
        from operkred
        where ok_ol_date <= :sba_rep_end and
              ok_sh_date <= :sba_rep_end and
              ok_ol_is_active = 1 and
              ok_from_k_id = :sba_from_k_id and
              ok_type < 7 and
              ok_from_c_id = 0
        into :rep_end_os, :rep_end_perc;
        if (rep_end_os is null) then
          rep_end_os = 0;
        if (rep_end_perc is null) then
          rep_end_perc = 0;
      end
      else
      begin
        if ((rep_end_balance > 0) and
            (rep_kol_end = 1)) then
          rep_kol_normal = 1;
      end
    end
    if (rep_kol_begin = 0) then
      rep_kol_closed = 0;
    if (rep_kol_from_sb = 1) then
      rep_kol_closed = 0;
    if (rep_kol_to_sb = 1 and
        rep_kol_from_sb = 0 and
        rep_kol_end = 0) then
      rep_kol_closed = 1;

    rep_repay_os = 0;
    rep_repay_perc = 0;
    rep_repay_fine = 0;
    if (is_active_begin = 1) then
    begin
      select sum(-ok_os), sum(-ok_perc), sum(-ok_fine)
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date between :sba_rep_begin and :sba_rep_end and
            ok_from_k_id = :sba_from_k_id and
            ok_type = 4 and
            ok_from_c_id = 0
      into :rep_repay_os, :rep_repay_perc, :rep_repay_fine;
      if (rep_repay_os is null) then
        rep_repay_os = 0;
      if (rep_repay_perc is null) then
        rep_repay_perc = 0;
      if (rep_repay_fine is null) then
        rep_repay_fine = 0;
      if ((rep_repay_os + rep_repay_perc + rep_repay_fine) <> 0) then
        rep_kol_repay = 1;
    end

    if (is_active_begin = 1) then
    begin
      if ((rep_begin_os + rep_begin_perc) <> 0) then
        rep_status_begin = 'просроченный';
      else
        rep_status_begin = 'нормальный';
    end
    else
      rep_status_begin = 'закрытый';

    if (is_active_end = 1) then
    begin
      if ((rep_end_os + rep_end_perc) <> 0) then
        rep_status_end = 'просроченный';
      else
        rep_status_end = 'нормальный';
    end
    else
      rep_status_end = 'закрытый';

    if ((is_active_begin + is_active_end) > 0) then
      suspend;
  end
end
