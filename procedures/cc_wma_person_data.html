CREATE PROCEDURE CC_WMA_PERSON_DATA(
    FIO STR_255,
    BIRTH_DATE DATE,
    PHONE_NO STR_100)
RETURNS (
    TXT D_BLOB_TEXT)
AS
declare variable c_id id_type;
declare variable res_comment str_255;
declare variable c_name_fio str_100;
declare variable c_birth_date str_10;
declare variable k_disb_date str_10;
declare variable kf_name str_100;
declare variable reg_name str_100;
begin
  --  #FF9882
  txt = '';
  for select c_id, res_comment, c_name_fio, f_datetostr(c_birth_date) c_birth_date
      from hp_find_clients(null, :fio, :birth_date, null, null, null, :phone_no, null, null)
      where is_critical = 1
      into :c_id, :res_comment, :c_name_fio, :c_birth_date
  do
  begin
    txt = BLOB_APPEND(txt,'<div style="border: thin solid; padding: 5px;">');
    txt = BLOB_APPEND(txt,'<div style="background-color: #FFDDCC">');
    txt = BLOB_APPEND(txt,'Совпадение с лицом: <b>'||res_comment||'</b><br>');
    txt = BLOB_APPEND(txt,'ФИО: <b>'||c_name_fio||'</b><br>');
    txt = BLOB_APPEND(txt,'Дата рождения: <b>'||c_birth_date||'</b><br>');

    for select f_datetostr(kont.k_disb_date),
               kal.kal_new_kf_name,
               reg.reg_name
    from kontstate ks
    inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
    inner join kontract kont on kont.k_id = kncl.kc_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
    inner join region reg on  reg.reg_unicode = kal.kal_new_reg_id
    where ks.ks_state = 1
    and ks.ks_ol_is_delete = 0
    and current_date between ks.ks_date_begin and ks.ks_date_end
    and kncl.kc_amount_approved > 0
    and kncl.kc_from_c_id = :c_id
    into :k_disb_date, :kf_name, :reg_name
    do
    begin
        txt = BLOB_APPEND(txt,'<br>Активный займ<br>');
        txt = BLOB_APPEND(txt,'Офис: <b>'||reg_name||'</b><br>');
        txt = BLOB_APPEND(txt,'Группа: <b>'||kf_name||'</b><br>');
        txt = BLOB_APPEND(txt,'Дата выдачи: <b>'||k_disb_date||'</b><br>');
    end

    txt = BLOB_APPEND(txt,'</div>');
    txt = BLOB_APPEND(txt,'</div>');

    suspend;
  end
end
