              ok_sh_os <> 0

        having max(ok_sh_date) >= :dd and min(ok_sh_date) <= :dd
        order by ok_sh_id

      end

          res_max_fine_amount = 0;
      end

      if (total_calc_perc is null) then
        total_calc_perc = 0;
      if (total_calc_perc < 0) then
        total_calc_perc = 0;
      if (pr_perc_round = -2) then
        total_calc_perc = cast(total_calc_perc as decimal(18,2));
      if (pr_perc_round = -1) then
        total_calc_perc = cast(total_calc_perc as decimal(18,1));
      if (pr_perc_round = 0) then
        total_calc_perc = cast(total_calc_perc as decimal(18,0));
      if (pr_perc_round = 1) then
        total_calc_perc = cast(total_calc_perc / 10 as decimal(18,0)) * 10;
      if (pr_perc_round = 2) then
        total_calc_perc = cast(total_calc_perc / 100 as decimal(18,0)) * 100;
      if (pr_perc_round = 3) then
        total_calc_perc = cast(total_calc_perc / 1000 as decimal(18,0)) * 1000;

      if (result_vac_perc_saldo is null) then
        result_vac_perc_saldo = 0;
    end
    select kal_new_reg_id, reg_head_office
    from knaktlist, region
    where kal_from_kf_id = :kf_id and
          kal_date <= :dd and
          kal_date_end >= :dd and
          knaktlist.is_delete = 0 and
          reg_unicode = kal_new_reg_id rows 1
    into :reg_unicode, :reg_head_office;

    disb_os = 0;

    -- Вычисление задолженности по доп.проценту
    if ((db_id = 94401) or (f_tmp_const('add_perc_later_repay') = 1)) then  -- добавил условие db_id = 94401, т.к.АЗ сперва переходил на отложенный доп. %, затем вернулся на старый механизм, без отложенного доп. %
    begin
      if (:pr_code containing '.ipf.' and
          :disb_dd > cast('07.12.2021' as date) and
          :db_id in (39801, 41701)) then
      begin
        select pr.pr_ipf_commission
        from product pr
        where pr.pr_id = :pr_id
        into pr_ipf_commission;

        pr_ipf_commission = coalesce(pr_ipf_commission, 0);
        rd_id = f_get_rd_id('CashTransactionIncom.IPFCommission');

        select sum(ok.ok_amount)
        from operkontract ok
        inner join operlist ol on ol.ol_id = ok.ok_from_ol_id and
              ol.is_delete = 0
        where ok_from_k_id = :kont_id and
              ok.ok_operation_type = :rd_id
        into ok_amount_ipf;

        ok_amount_ipf = coalesce(ok_amount_ipf, 0);
        sh_add_perc = pr_ipf_commission - ok_amount_ipf;
        sh_add_perc = coalesce(sh_add_perc, 0);

        if (sh_add_perc > 0) then
        begin
          is_extra = 1;
          sh_repay_type = 1;
          sh_date = disb_dd;
         sh_date_orig = disb_dd;
          sh_date_sh_orig = disb_dd;
          sh_id = 0;
          s_all = sh_add_perc;
          suspend;
        end
        sh_add_perc = 0;
        is_extra = 0;
      end
      else
      begin
        select sum(ok_kommis)
        from operkred
        where ok_from_k_id = :kont_id and
              ok_ol_is_active = 1 and
              ok_ol_date <= :dd and
              ok_sh_id = 0 and
              ok_from_c_id = 0
        into sh_add_perc;
        if (sh_add_perc is null) then
          sh_add_perc = 0;

        if (sh_add_perc > 0) then
        begin
          is_extra = 1;
          sh_repay_type = 1;
          sh_date = disb_dd;
          sh_date_orig = disb_dd;
         sh_date_sh_orig = disb_dd;
          sh_id = 0;
          s_all = sh_add_perc;
          suspend;
        end
        sh_add_perc = 0;
        is_extra = 0;
      end
    end

    total_fine_amount = 0;
    for select distinct ok_sh_id, ok_sh_date
        from operkred
        where ok_from_k_id = :kont_id and
              ok_ol_is_active = 1 and
              ok_ol_date <= :dd and
              ok_sh_id > 0 and
              ok_from_c_id = 0
        order by ok_sh_date, ok_sh_id -- Sanjari, 22.04.2020
        into :sh_id, :sh_date
    do
    begin
      real_dd = sh_date;
      pr_fine_percent = k_fine_percent;
      sh_date_orig = :sh_date;
      add_fine = 0;
      if (sh_date = kont_last_date) then
      begin
        if ((dd > :sh_date) and
            (pr_calc_perc_for_deliq > 0)) then
        begin
          sh_date = :dd;
        end
      end
      sh_date_sh_orig = :sh_date;
      if (pr_fine_method in (4, 5)) then
      begin
        if ((last_fine_dd < sh_date_orig) or (last_fine_dd is null)) then
          last_fine_dd = sh_date_orig;
      end
      dd2 = dd;
      v_pay_os = 0;
      v_pay_perc = 0;
      s_os = 0;
      sss_os = 0; -- Накопитель планового тела перед вложенным циклом

      s_perc = 0;
      sh_calc_fine = 0;
      sh_calc_perc = 0;
      zad_o = disb_os;
      zad_p = 0;
      last_dd = null;
      max_dd = cast((sh_date_orig + pr_fine_max_day) as date);
      zad_perc = 0;
      zad_os = 0;
      sh_fine_kol_day = 0;
      add_fine = 0;
      last_type = 0; -- и далее новая логика с last_type для корректного определения льготного периода, Sanjari, RFCKG-5395
      is_grace = 0;

      for with tab
          as (select ok_ol_date as oper_dd, ok_type as v_type, ok_sh_id as sh_id, ok_os as v_os, ok_perc as v_perc,
                     ok_fine as v_fine, ok_sh_os as sh_os, ok_sh_perc as sh_perc, -1.0 as fine_perc_tmp
              from operkred
              where ok_ol_is_active = 1 and
                    ok_ol_date <= :dd2 and
                    ok_sh_id = :sh_id and
                    ok_sh_date = :sh_date_orig and
                    ok_from_k_id = :kont_id and
                    ok_from_c_id = 0 and
                    ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/

              union all

              select kd_ol_date as oper_dd, 4 as v_type, :sh_id as sh_id, 0.0 as v_os, 0.0 as v_perc, 0.0 as v_fine,
                     0.0 as sh_os, 0.0 as sh_perc, kd_fine_percent as fine_perc_tmp
              from kontdata
              where kd_from_k_id = :kont_id and
                    kd_ol_date < :dd2 and
                    kd_date_end >= kd_ol_date and
                    kd_ol_is_delete = 0)

          select oper_dd, v_type, sum(v_os), sum(v_perc), sum(v_fine), sum(sh_os), sum(sh_perc), max(fine_perc_tmp)
          from tab
          group by oper_dd, v_type
          order by oper_dd

          into :oper_dd, :v_type, :v_os, :v_perc, :v_fine, :sh_os, :sh_perc, :fine_perc_tmp
      do
      begin
        is_hidden_line = 0;

        if (real_dd < oper_dd) then
          real_dd = oper_dd;

        if (pr_fine_method = 5) then
        begin
          select oks_balance
          from oksummary
          where oks_from_k_id = :kont_id and
                oks_date_begin <= :real_dd and
                oks_date_end >= :real_dd and
                oks_close_state = 0
          into :base_amount_for_fine;
        end

        if (v_type <= 3) then
        /* */
        begin
          s_os = s_os + sh_os;
          s_perc = s_perc + sh_perc;
        end
           --RFCRU-5730
        if (db_id in (81001,41701)) then
        begin
          if (v_type <= 2 or sh_date = k_stop_perc) then
          begin
            sss_os = sss_os + sh_os;
          end
        end

        if ((v_type < 8) and
            (v_type >= 1)) then
        begin
          v_pay_os = v_pay_os - v_os;
          v_pay_perc = v_pay_perc - v_perc;
        end

        zad_perc = zad_perc + sh_perc + v_perc;
        zad_os = zad_os + sh_os + v_os;

        if (v_type <= 6 or sh_date = k_stop_perc) then
        begin
          if (sh_os <> 0) then
          begin
            last_type = v_type;
            if (sh_date = k_stop_perc and zad_os = 0 and sh_os <> 0) then
              is_grace = 1;
          end
        end

        is_calc_fine = 1;

        if (reg_head_office = 1040100) then
        begin
          if ((sh_date_orig >= '11.06.2010') and
              (sh_date_orig <= '11.07.2010')) then
          begin
            is_calc_fine = 0;
          end
        end
        if (reg_unicode = 1040100) then
        begin
          if ((sh_date_orig >= '11.06.2010') and
              (sh_date_orig <= '30.09.2010')) then
          begin
            is_calc_fine = 0;
          end
        end
        if ((reg_unicode = 1040100) and
            (dd >= '15.10.2010') and
            (disb_dd <= '11.06.2010') and
            (sh_date_orig >= '15.10.2010')) then
        begin
          is_calc_fine = 0;
        end
        if ((reg_unicode = 1040100) and
            (dd >= '15.10.2010') and
            (disb_dd <= '11.06.2010') and
            (pr_loan_type = 0)) then
        begin
          is_calc_fine = 0;
        end

        if (zad_o + zad_p > 0) then
        begin

          if (oper_dd > max_dd) then
            oper_dd = max_dd;
          if (pr_fine_method in (4, 5)) then
          begin
            if (last_dd is not null) then
            begin
              if (last_dd < last_fine_dd) then
                last_dd = last_fine_dd;
            end
          end

          if ((db_id = 94401) and
            (pr_id in (4, 7)) and
            (last_dd <= '30.11.2012')) then
            last_dd = '30.11.2012';

          d_begin = last_dd;
          d_end = minvalue(:oper_dd, coalesce(:k_stop_fine, :oper_dd));

          kol_day_tmp = 0;
          if (d_begin < k_covid_fine_stop_date and
              d_end > k_covid_fine_start_date) then
          begin
            select wd_count
            from hp_get_work_days(:k_covid_fine_stop_date, :k_covid_fine_start_date, :pr_is_fine_weekend)
            into :kol_day_tmp;
          end
          else
          begin
            if (d_begin between k_covid_fine_stop_date and (k_covid_fine_start_date - 1)) then
              d_begin = k_covid_fine_start_date - 1;

            if (d_end between k_covid_fine_stop_date and (k_covid_fine_start_date - 1)) then
              d_end = k_covid_fine_stop_date;

        if (kol_day > 0) then

          if (not(pr_fine_method in (4, 5))) then
          begin
            sh_calc_fine = sh_calc_fine + kol_day * zad_o * pr_fine_percent / 100 * is_calc_fine + kol_day * zad_p * pr_fine_for_percent / 100 * is_calc_fine;

            if (add_fine = 0) then
            begin
              add_fine = 1;
              sh_calc_fine = sh_calc_fine + zad_o * add_fine_koef / 100 * is_calc_fine + zad_p * add_fine_koef / 100 * is_calc_fine;
            end
          end
          else
          begin
            sh_calc_fine = sh_calc_fine + kol_day * base_amount_for_fine * pr_fine_percent / 100 * is_calc_fine;
            sh_calc_fine = sh_calc_fine + kol_day * base_amount_for_fine * k_percent_per_day * pr_fine_like_perc / 10000 * is_calc_fine;
            last_fine_dd = oper_dd;
          end
        end
        sh_fine_kol_day = :kol_day;
      end
      sh_calc_fine = sh_calc_fine + v_fine;
      if (pr_fine_method = 0) then
          zad_o = zad_o + sh_os + v_os;
      if (pr_fine_method = 1) then
          zad_o = zad_o + sh_os;
      if (pr_fine_method = 2) then
      begin
          zad_o = zad_o + sh_os + v_os;
          zad_p = zad_p + sh_perc + v_perc;
      end
      if (pr_fine_method = 3) then
      begin
          zad_o = zad_o + sh_os;
          zad_p = zad_p + sh_perc;
      end
      if (pr_fine_method in (4, 5)) then
      begin
          zad_o = zad_o + sh_os + v_os;
          zad_p = zad_p + sh_perc + v_perc;

        else
          last_dd = oper_dd;

        if (coalesce(fine_perc_tmp, -1) <> -1) then
          pr_fine_percent = fine_perc_tmp;

        if (pr_fine_method = 5) then
        begin
          select oks_balance
          from oksummary
          where oks_from_k_id = :kont_id and
                oks_date_begin <= :real_dd and
                oks_date_end >= :real_dd and
                oks_close_state = 0
          into :base_amount_for_fine;
        end
      end
      if (sh_calc_fine = 0) then
        add_fine = 0;
      if ((dd2 > last_dd) and
          (zad_o + zad_p > 0)) then
      begin
        if (dd2 > max_dd) then
          dd2 = max_dd;
        if (pr_fine_method in (4, 5)) then
        begin
          if (last_dd is not null) then
          begin
            if (last_dd < last_fine_dd) then
              last_dd = last_fine_dd;
          end
        end

        if ((db_id = 94401) and
            (pr_id in (4, 7)) and
            (last_dd <= '30.11.2012')) then
          last_dd = '30.11.2012';

        d_begin = last_dd;
        d_end = minvalue(:dd2, coalesce(:k_stop_fine, :dd2));

        kol_day_tmp = 0;
        if (d_begin < k_covid_fine_stop_date and
            d_end > k_covid_fine_start_date) then
        begin
          select wd_count
          from hp_get_work_days(:k_covid_fine_stop_date, :k_covid_fine_start_date, :pr_is_fine_weekend)
          into :kol_day_tmp;
        end
        else
        begin
          if (d_begin between k_covid_fine_stop_date and (k_covid_fine_start_date - 1)) then
            d_begin = k_covid_fine_start_date - 1;

          if (d_end between k_covid_fine_stop_date and (k_covid_fine_start_date - 1)) then
            d_end = k_covid_fine_stop_date;
        end

        select wd_count
        from hp_get_work_days(:d_begin, :d_end, :pr_is_fine_weekend)
        into :kol_day;

        kol_day = kol_day - kol_day_tmp;

        if (kol_day < 0) then
          kol_day = 0;
        if (kol_day > 0) then
        begin
          if (not(pr_fine_method in (4, 5))) then
          begin
            sh_calc_fine = sh_calc_fine + kol_day * zad_o * pr_fine_percent / 100 * is_calc_fine + kol_day * zad_p * pr_fine_for_percent / 100 * is_calc_fine;
            sh_calc_fine = sh_calc_fine + kol_day * zad_o * k_percent_per_day / 100 * pr_fine_like_perc / 100 * is_calc_fine + kol_day * zad_p * k_percent_per_day / 100 * pr_fine_like_perc / 100 * is_calc_fine;
            if (add_fine = 0) then
            begin
              add_fine = 1;
              sh_calc_fine = sh_calc_fine + zad_o * add_fine_koef / 100 * is_calc_fine + zad_p * add_fine_koef / 100 * is_calc_fine;
            end
          end
          else
          begin
            sh_calc_fine = sh_calc_fine + kol_day * base_amount_for_fine * pr_fine_percent / 100 * is_calc_fine;
            sh_calc_fine = sh_calc_fine + kol_day * base_amount_for_fine * k_percent_per_day / 100 * pr_fine_like_perc / 100 * is_calc_fine;
            last_fine_dd = dd2;
          end
        end
        sh_fine_kol_day = :kol_day;
      end
      if ((sh_calc_fine > 0) and
          (sh_calc_fine < pr_min_fine)) then
        sh_calc_fine = pr_min_fine;
      if (pr_fine_round = -2) then
        sh_calc_fine = cast(sh_calc_fine as decimal(18,2));
      if (pr_fine_round = -1) then
        sh_calc_fine = cast(sh_calc_fine as decimal(18,1));
      if (pr_fine_round = 0) then
        sh_calc_fine = cast(sh_calc_fine as decimal(18,0));
      if (pr_fine_round = 1) then
        sh_calc_fine = cast(sh_calc_fine / 10 as decimal(18,0)) * 10;
      if (pr_fine_round = 2) then
        sh_calc_fine = cast(sh_calc_fine / 100 as decimal(18,0)) * 100;
      if (pr_fine_round = 3) then
        sh_calc_fine = cast(sh_calc_fine / 1000 as decimal(18,0)) * 1000;
      if ((new_ipf_repay_mode = 1 and show_zero_line = 1) or -- если погашение по ИФП новый метод
          (cast(sh_calc_fine as decimal(18,2)) <> 0) or
          (cast(zad_perc as decimal(18,2)) <> 0) or (cast(zad_os as decimal(18,2)) <> 0)) then
      begin
        if (first_date is null) then
          first_date = :sh_date_orig;
        if (sh_date_orig < dd) then
          sh_repay_type = 1;
        else
        begin
          if (first_current = 0) then
          begin
            first_sh_id = sh_id;
            sh_repay_type = 2;
            first_current = 1;
          end
          else
          begin
            sh_repay_type = 3;
          end
        end

        if ((cast(zad_perc as decimal(18,2)) <> 0) or (cast(zad_os as decimal(18,2)) <> 0)) then
        begin
          if (sh_id <= coalesce(vac_os_sh_id, 999999)) then
          begin
            if (total_calc_perc - zad_perc > 1) then
            begin
              total_calc_perc = total_calc_perc - zad_perc;
              sh_calc_perc = zad_perc;
            end
            else
            begin
              sh_calc_perc = total_calc_perc;
              total_calc_perc = 0;
  
              if (abs(sh_calc_perc - zad_perc) <= 1) then
                sh_calc_perc = zad_perc;
            end
          end

          -- Распределение процентов по кредитным каникулам
          if (result_vac_perc_saldo > 0 and sh_id > coalesce(vac_os_sh_id, 999999)) then
          begin
            if (result_vac_perc_saldo - zad_perc > 1) then
            begin
              result_vac_perc_saldo = result_vac_perc_saldo - zad_perc;
              sh_calc_perc = zad_perc;
            end
            else
            begin
              sh_calc_perc = result_vac_perc_saldo;
              result_vac_perc_saldo = 0;

              if (abs(sh_calc_perc - zad_perc) <= 1) then
                sh_calc_perc = zad_perc;
            end
          end
          result_balance_end = result_balance_end - zad_os;
        end
      end
      --      if (((sh_repay_type=2) or (sh_date_orig=kont_last_date)) and (total_calc_perc>0)) then
      if (((sh_repay_type = 2 and
          dd_change_end is null) or (sh_date_orig = k_kredline_end)) and
          (total_calc_perc > 0)) then
      begin
        if (db_id <> 39801 or k_stop_perc is null) then
        begin
          sh_calc_perc = sh_calc_perc + total_calc_perc;
          total_calc_perc = 0;
        end
      end

      -- получение ежедневной процентной ставки ДО текущей даты платежа
      if (perc_is_new_mode = 1) then
      begin
        sh_prior_percent_per_day = null;
        select kd_percent_per_day
        from kontdata
        where kd_from_k_id = :kont_id and
              kd_ol_is_delete = 0 and
              kd_ol_date < :sh_date
        order by kd_from_k_id desc, kd_ol_date desc
        rows 1
        into sh_prior_percent_per_day;
        if (sh_prior_percent_per_day is null) then
          sh_prior_percent_per_day = k_percent_per_day;
      end

      if (sh_calc_fine < 0) then
        sh_calc_fine = 0;

      need_suspend = 0;
      if ((cast(sh_calc_fine as decimal(18,2)) <> 0) or (cast(sh_calc_perc as decimal(18,2)) <> 0) or (cast(zad_kommis as decimal(18,2)) <> 0) or (cast(zad_perc as decimal(18,2)) <> 0) or (cast(zad_os as decimal(18,2)) <> 0)) then
      begin
        -- если имеется задолженность, возвращаем строки
        need_suspend = 1;

        -- если будущим периодом встретили строку с задоженностью, отключаем последующий показ нулевых строк
        if (sh_date >= dd) then
          show_zero_line = 0;
      end

      -- если нет задолженности в будущих платежах, но входящий параметр требует нулевых, отображаем
      if (need_suspend = 0 and
          sh_date >= dd and
          show_zero_line = 1) then
      begin
        need_suspend = 1;
        is_hidden_line = 1;
      end
       --RFCRU-5730
      if (need_suspend = 1) then
      begin
        if (db_id in (81001, 41701)) then
        begin
          if ((sss_os = 0) or (zad_os = 0 and last_type in (1, 2))) then 
            is_grace = 1;
          else
            is_grace = 0;
        end
        else
        begin
          if (zad_os = 0 and last_type in (1, 2)) then 
            is_grace = 1;
        end

        s_all = s_os + s_perc;
        pac_percent = 0;
        if (pr_pay_pred <> 3) then
        begin
          if (sh_id <= pac_calc_repay) then
            pac_percent = pac_percent_before_term;
          else
            pac_percent = pac_percent_after_term;
        end
        else
        begin
          if (sh_id >= (k_kolich_repay - (pac_calc_repay - first_sh_id) + 1)) then
            pac_percent = pac_percent_before_term;
          else
          begin
            if (sh_repay_type = 2) then
            begin
              if (sh_id <= pac_calc_repay) then
                pac_percent = pac_percent_before_term;
              else
                pac_percent = pac_percent_after_term;
            end
            if (sh_repay_type = 3) then
              pac_percent = pac_percent_after_term;
          end
        end
        if (pac_calc_near_pay = 0) then
        begin
          if (sh_repay_type = 2) then
            pac_percent = 0;
        end
        if (sh_date < :dd) then
        begin
          sh_fine_base = (zad_o + zad_p);
          sh_fine_base_os = zad_o;
          sh_fine_base_perc = zad_p;
        end
        else
        begin
          sh_fine_base = 0;
          sh_fine_base_os = 0;
          sh_fine_base_perc = 0;
        end
        sh_date = :sh_date_orig;

        if (sh_calc_fine > 0) then
        begin
          if (total_fine_amount + sh_calc_fine - v_fine > res_max_fine_amount) then
            sh_calc_fine = res_max_fine_amount - total_fine_amount + v_fine;

          if (sh_calc_fine < 0) then
            sh_calc_fine = 0;

          total_fine_amount = total_fine_amount + sh_calc_fine - v_fine;
        end

        suspend;
      end
    end
  end

  if (is_extra_at_end = 1) then
  begin
    is_hidden_line = 0;
    zad_kommis = 0;
    sss_os = 0;
    pr_fine_method = -1;
    total_calc_perc = 0;
    result_vac_perc_saldo = 0;
    zad_os = 0;
    zad_perc = 0;
    zad_o = 0;
    zad_p = 0;
    sh_repay_type = 0;
    first_current = 0;
    v_pay_os = 0;
    v_pay_perc = 0;
    s_os = 0;
    s_perc = 0;
    first_sh_id = 0;
    k_currency_rate = 1;
    sh_fine_base = 0;
    sh_fine_base_os = 0;
    sh_fine_base_perc = 0;

    sh_calc_perc = 0;
    sh_calc_fine = 0;
    s_all = 0;
    is_grace = 0;
    pac_percent = 0;
    sh_fine_kol_day = 0;
    dd_change_days = 0;

    for select res_rd_id, res_amount_saldo
        from hp_repay_extra_saldo(:kont_id, 0, :dd)
        into :res_rd_id, :zad_kommis
    do
    begin
      if (zad_kommis > 0) then
      begin
        is_extra = 1;
        if (sh_date < current_date) then
          sh_date = current_date;

        sh_calc_fine = 0;
        sh_repay_type = 3;
        sh_date_orig = sh_date;
        sh_id = res_rd_id;
        s_all = zad_kommis;
        suspend;
      end
    end
  end
