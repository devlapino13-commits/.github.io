CREATE PROCEDURE UXP$FIELD_CLONE_VIRTUAL_FK(
    UF_UUID D_UUID)
AS
declare variable field_name d_str_code;
declare variable us_code d_str_code;
declare variable fk_source d_str_code;
declare variable fk_index d_str_100;
begin
  select us_code, uf_field_name
  from ux$field, ux$source
  where uf_uuid = :uf_uuid and
        uf_owner = us_uuid
  into us_code, field_name;

  for select distinct i.uix_fk_source, i.uix_fk_index
  from ux$index i
  where i.uix_fields = :field_name and
        i.uix_virtual = 1 and
        i.uix_fk = 1 and
        i.uix_state = 0 and
        i.uix_count = 1
  into fk_source, fk_index
  do
  begin
    if (not exists(select * from ux$index
                   where uix_fields = :field_name and
                         uix_source = :us_code and
                         uix_fk_source = :fk_source and
                         uix_fk_index = :fk_index)) then
    begin
      insert into ux$index (uix_source, uix_fields, uix_fk, uix_virtual, uix_count,
                            uix_fk_source, uix_fk_index, uix_state)
      values (:us_code, :field_name, 1, 1, 1,
              :fk_source, :fk_index, 0);
    end
  end
end
