CREATE PROCEDURE HP_CASHBOOK_INVALIDS(
    IN_OFFICE_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    RES_DATE D_DATE,
    RES_STATUS D_INTEGER,
    RES_SIGNED D_BOOL,
    RES_BLANK_ID D_INTEGER,
    CBM_ID D_INTEGER,
    CBM_CASHIER_ID D_ID,
    CBM_MANAGER_ID D_ID,
    CBM_AMOUNT_PLUS D_CURRENCY,
    CBM_AMOUNT_MINUS D_CURRENCY,
    CBM_AMOUNT_SALDO D_CURRENCY,
    CBM_COUNT_PLUS D_INTEGER,
    CBM_COUNT_MINUS D_INTEGER,
    CBM_AMOUNT_IPF_PLUS D_CURRENCY,
    CBM_AMOUNT_IPF_MINUS D_CURRENCY,
    CBM_AMOUNT_IPF_SALDO D_CURRENCY,
    CBM_COUNT_IPF_PLUS D_INTEGER,
    CBM_COUNT_IPF_MINUS D_INTEGER,
    CBM_AMOUNT_CLS_PLUS D_CURRENCY,
    CBM_AMOUNT_CLS_MINUS D_CURRENCY,
    CBM_AMOUNT_CLS_SALDO D_CURRENCY,
    CBM_COUNT_CLS_PLUS D_INTEGER,
    CBM_COUNT_CLS_MINUS D_INTEGER,
    CBM_FROM_UN_ID D_ID,
    CBM_DATETIME D_DATETIME,
    CBM_FROM_EDL_ID D_ID,
    QR_CODE D_STR_LONG,
    EDOC_SIGN_DATETIME D_DATETIME,
    RES_SALDO_BEGIN D_CURRENCY,
    RES_SALDO_IPF_BEGIN D_CURRENCY,
    RES_SALDO_CLS_BEGIN D_CURRENCY,
    RES_CASHIER_FIO D_STR_100,
    RES_MANAGER_FIO D_STR_100,
    RES_WEEKDAY D_STR_10,
    RES_IS_WEEKEND D_BOOL)
AS
declare variable WEEKDAY_ID D_INTEGER;
begin
  if (rep_date_begin > rep_date_end) then
    exit;

  for select cbm_id, cbm_date, cbm_cashier_id, cbm_manager_id, cbm_amount_plus,
         cbm_amount_minus, cbm_amount_saldo, cbm_count_plus, cbm_count_minus, cbm_amount_ipf_plus, cbm_amount_ipf_minus,
         cbm_amount_ipf_saldo, cbm_count_ipf_plus, cbm_count_ipf_minus, cbm_from_un_id, cbm_datetime, cbm_from_edl_id
  from cashbookmain
  where cbm_office_id = :in_office_id and
        cbm_date between :rep_date_begin and :rep_date_end and
        cbm_invalid = 1
  order by cbm_date, cbm_id
  into :cbm_id, :res_date, :cbm_cashier_id, :cbm_manager_id,
       :cbm_amount_plus, :cbm_amount_minus, :cbm_amount_saldo, :cbm_count_plus, :cbm_count_minus, :cbm_amount_ipf_plus,
       :cbm_amount_ipf_minus, :cbm_amount_ipf_saldo, :cbm_count_ipf_plus, :cbm_count_ipf_minus, :cbm_from_un_id,
       :cbm_datetime, :cbm_from_edl_id
  do
  begin
    weekday_id = extract(weekday from res_date);
    res_is_weekend = 0;
    if (weekday_id in (0, 6)) then
      res_is_weekend = 1;

    res_weekday = case weekday_id
      when 0 then 'Вс'
      when 1 then 'Пн'
      when 2 then 'Вт'
      when 3 then 'Ср'
      when 4 then 'Чт'
      when 5 then 'Пт'
      when 6 then 'Сб'
    end;

    res_blank_id = 78; -- Кассовый журнал

    cbm_amount_cls_plus = 0;
    cbm_amount_cls_minus = 0;
    cbm_amount_cls_saldo = 0;
    cbm_count_cls_plus = 0;
    cbm_count_cls_minus = 0;

    cbm_amount_cls_plus = cbm_amount_plus - cbm_amount_ipf_plus;
    cbm_amount_cls_minus = cbm_amount_minus - cbm_amount_ipf_minus;
    cbm_amount_cls_saldo = cbm_amount_saldo - cbm_amount_ipf_saldo;
    cbm_count_cls_plus = cbm_count_plus - cbm_count_ipf_plus;
    cbm_count_cls_minus = cbm_count_minus - cbm_count_ipf_minus;

    res_saldo_begin = cbm_amount_saldo - (cbm_amount_plus - cbm_amount_minus);
    res_saldo_cls_begin = cbm_amount_cls_saldo - (cbm_amount_cls_plus - cbm_amount_cls_minus);
    res_saldo_ipf_begin = cbm_amount_ipf_saldo - (cbm_amount_ipf_plus - cbm_amount_ipf_minus);

    qr_code = '';
    edoc_sign_datetime = null;
    res_signed = null;

    if (cbm_from_edl_id is not null) then
    begin
      qr_code = f_edoc_qr_by_edl(cbm_from_edl_id, cbm_manager_id);

      select max(eds_status_datetime)
      from edoc_sign
      where eds_from_edl_id = :cbm_from_edl_id and
            eds_status = 1
      into edoc_sign_datetime;

      res_signed = 0;
      if (exists(select * from edoc_list
                 where edl_id = :cbm_from_edl_id and
                       edl_status = 1)) then
        res_signed = 1;
    end

    res_status = -2; -- Аннулирован

    res_cashier_fio = '';
    select c_name_fio
    from clients
    where c_id = :cbm_cashier_id
    into res_cashier_fio;

    res_manager_fio = '';
    select c_name_fio
    from clients
    where c_id = :cbm_manager_id
    into res_manager_fio;

    suspend;
  end
end
