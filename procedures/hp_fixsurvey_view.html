CREATE PROCEDURE HP_FIXSURVEY_VIEW(
    KONT_NO D_STR_20,
    RD_ID_INP D_INTEGER = 0)
RETURNS (
    RD_ID D_INTEGER,
    RD_CODE D_STR_255,
    RD_NAME D_STR_255,
    ACL_ID D_ID,
    ACL_DATE D_DATE,
    ACL_LAST_DATETIME D_DATETIME,
    ACL_STATUS D_INTEGER,
    ACL_STATE D_STR_20,
    ACL_FROM_UN_ID D_ID,
    ACL_RESPONS_C_ID D_ID,
    EMP_NAME D_STR_100,
    NEW_EMP_ID D_ID,
    NEW_EMP_NAME D_STR_100,
    UN_FIO D_STR_100,
    K_ID D_ID,
    K_DISB_DATE D_DATE,
    K_FACT_CLOSE_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    IS_IPF D_INTEGER,
    KC_ID D_ID,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    MA_ID D_INTEGER,
    MAC_ID D_INTEGER)
AS
declare variable obj_kind d_str_20;
declare variable obj_id d_id;
declare variable on_date d_date;
declare variable acl_id1 d_id;
declare variable k_reg_id d_integer;
declare variable kf_id d_id;
begin
  if (kont_no similar to '[-]?[[:DIGIT:]]+') then
    k_id = (select k_id from kontract where k_id = :kont_no);
  else
    k_id = (select k_id from kontract where k_kontract_no = :kont_no);
  if (k_id is null) then
    exception e_message 'Договор не определен.';

  select
    k_disb_date,
    k_kontract_no,
    iif(pr_code containing '.ipf.', 1 ,0),
    ma_id,
    c_id,
    c_name_fio,
    coalesce(ol_date, current_date),
    kal_new_reg_id,
    kal_from_kf_id
  from kontract
    inner join product
      on pr_id = k_from_pr_id
    inner join miniapp
      on ma_id = k_from_ma_app_id
        and ma_date_end = '31.12.9999'
    inner join knaktlist
      on kal_from_kf_id = k_from_kf_id
        and current_date between kal_date and kal_date_end
        and knaktlist.is_delete = 0
    inner join clients
      on c_id = kal_new_kf_officer
    left join operlist
      on ol_id = k_fact_closed_ol_id
        and operlist.is_delete = 0
  where k_id = :k_id
  into
    :k_disb_date,
    :k_kontract_no,
    :is_ipf,
    :ma_id,
    :new_emp_id,
    :new_emp_name,
    :k_fact_close_date,
    :k_reg_id,
    :kf_id;
  -- Если папка у Виртуального сотрудника передаём задачу МО офиса RFCKG-329
  if (:new_emp_id = 1000000000100 or
      :new_emp_id = 1000000000005) then
  begin
    new_emp_id = null;
    -- Получаем c_id менеджера офиса
    select ps.ps_from_c_id
    from hr_pstatus ps
    inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
    inner join jobkind jk on jk.jk_id = jl.jl_kind and
                             jk.jk_system_code = 'JobSystem.Region_KM'
    where ps.ps_from_reg_id = :k_reg_id and
          current_date between ps.ps_date_begin and ps.ps_date_end and
          ps.ps_is_working = 1
    rows 1
    into :new_emp_id;
    -- Если у офиса нет МО передаём задачу КС у которого была папка до Виртуального сотрудника
    if (:new_emp_id is null) then
    begin
      select kal.kal_new_kf_officer
      from knaktlist kal
      where kal.kal_from_kf_id = :kf_id and
            kal.kal_new_kf_officer not in (1000000000100, 1000000000005) and
            kal.is_delete = 0
      rows 1
      into :new_emp_id;
    end 
  end 

  for
    select
      rd_id,
      rd_code,
      rd_name
    from refdata
    where rd_main_code = 'ActionGroup'
      and rd_parent_id > 0
      and rd_str_id = 'Anketa'
      and rd_is_active = 1 and
      iif(:rd_id_inp > 0, rd_id = :rd_id_inp, rd_id > 0)
    order by rd_id
  into
    :rd_id,
    :rd_code,
    :rd_name
  do
  begin
    for
      select
        kc_id,
        c_id,
        c_name_fio,
        mac_id
      from knclient
        inner join clients
          on c_id = kc_from_c_id
        inner join miniappclient
          on mac_from_ma_id = :ma_id
            and mac_from_c_id = c_id
      where kc_from_k_id = :k_id
        and kc_amount_approved > 0
    into
      :kc_id,
      :c_id,
      :c_name_fio,
      :mac_id
    do
    begin
      acl_id = null;

      if (rd_code = 'Anketa.Main') then
        select
          acl_id,
          acl_date,
          acl_last_datetime,
          acl_status,
          acl_respons_c_id,
          acl_from_un_id
        from actionlist
        where acl_object_kind = f_objkind('Clients', 'id')
          and acl_object_id = :c_id
          and acl_from_rd_id = :rd_id
          and acl_date <= :k_disb_date
        order by acl_id desc
        rows 1
        into
          :acl_id,
          :acl_date,
          :acl_last_datetime,
          :acl_status,
          :acl_respons_c_id,
          :acl_from_un_id;

      if (rd_code = 'Anketa.TargetUseLoan') then
      begin
        if (k_disb_date >= '03.03.2022') then
        begin
          obj_kind = 'KnClient';
          obj_id = kc_id;
          select
            acl_id,
            acl_date,
            acl_last_datetime,
            acl_status,
            acl_respons_c_id,
            acl_from_un_id
          from actionlist
          where acl_object_kind = f_objkind(:obj_kind, 'id')
            and acl_object_id = :obj_id
            and acl_from_rd_id = :rd_id
          order by acl_id desc
          rows 1
          into
            :acl_id,
            :acl_date,
            :acl_last_datetime,
            :acl_status,
            :acl_respons_c_id,
            :acl_from_un_id;
        end else begin
          obj_kind = 'Clients';
          obj_id = c_id;
    
          if (k_disb_date < '14.02.2022')
          then
            on_date = k_disb_date +
              iif(k_disb_date >= '22.12.2021' and is_ipf = 1, 1, 8);
          else
            on_date = k_disb_date + 1;

          select
            acl_id,
            acl_date,
            acl_last_datetime,
            acl_status,
            acl_respons_c_id,
            acl_from_un_id
          from actionlist
          where acl_object_kind = f_objkind(:obj_kind, 'id')
            and acl_object_id = :obj_id
            and acl_from_rd_id = :rd_id
            and acl_date = :on_date
          order by acl_id desc
          rows 1
          into
            :acl_id,
            :acl_date,
            :acl_last_datetime,
            :acl_status,
            :acl_respons_c_id,
            :acl_from_un_id;
        end
        acl_id1 = acl_id;
      end

      if (rd_code = 'Anketa.Solvency') then
      begin
        obj_kind = 'MiniAppClient';
        obj_id = mac_id;
        select
          acl_id,
          acl_date,
          acl_last_datetime,
          acl_status,
          acl_respons_c_id,
          acl_from_un_id
        from actionlist
        where acl_object_kind = f_objkind(:obj_kind, 'id')
          and acl_object_id = :obj_id
          and acl_from_rd_id = :rd_id
        order by acl_id desc
        rows 1
        into
          :acl_id,
          :acl_date,
          :acl_last_datetime,
          :acl_status,
          :acl_respons_c_id,
          :acl_from_un_id;
      end

      if (rd_code = 'Anketa.Delinq.1') then
      begin
        obj_kind = 'KnClient';
        obj_id = kc_id;
        if (not exists(
          select *
          from actionlist
          where acl_object_kind = f_objkind(:obj_kind, 'id')
            and acl_object_id = :obj_id
            and acl_from_rd_id = :rd_id)) then
        begin
          obj_kind = 'Clients';
          obj_id = c_id;
        end
        select
          acl_id,
          acl_date,
          acl_last_datetime,
          acl_status,
          acl_respons_c_id,
          acl_from_un_id
        from actionlist
        where acl_object_kind = f_objkind(:obj_kind, 'id')
          and acl_object_id = :obj_id
          and acl_from_rd_id = :rd_id
          and acl_date between :k_disb_date and :k_fact_close_date
        order by acl_id desc
        rows 1
        into
          :acl_id,
          :acl_date,
          :acl_last_datetime,
          :acl_status,
          :acl_respons_c_id,
          :acl_from_un_id;
      end

      if (rd_code = 'Anketa.ClientAnaliz' and acl_id1 > 0) then
        select
          acl_id,
          acl_date,
          acl_last_datetime,
          acl_status,
          acl_respons_c_id,
          acl_from_un_id
        from actionlist
        where acl_object_kind = f_objkind('Clients', 'id')
          and acl_object_id = :c_id
          and acl_from_rd_id = :rd_id
          and acl_parent_id = :acl_id1
        order by acl_id desc
        rows 1
        into
          :acl_id,
          :acl_date,
          :acl_last_datetime,
          :acl_status,
          :acl_respons_c_id,
          :acl_from_un_id;

      if (acl_id > 0) then
      begin
        emp_name = (select c_name_fio from clients where c_id = :acl_respons_c_id);
        un_fio = (select un_fio from username where un_id = :acl_from_un_id);
        acl_state =
          case acl_status
            when -3 then 'отмена'
            when -2 then 'отказ'
            when -1 then 'доработка'
            when  0 then 'ожидание'
            when  1 then 'обработка'
            when  2 then 'подтверждение'
          end;
        suspend;
      end
    end
  end
end
