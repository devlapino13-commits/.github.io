CREATE PROCEDURE OKT_PANTERA_CALL_LIST(
    REP_DATE D_DATE,
    CALLED_OPER STR_100,
    KFOLDER_NAME STR_100,
    KONTRACT_NO D_STR_20,
    NATION D_INTEGER,
    CITIZENSHIP D_INTEGER,
    SBL_ID D_INTEGER,
    ROWS_MIN D_INTEGER = 1,
    ROWS_MAX D_INTEGER = 1000000,
    DELINQ_DAYS_MIN D_INTEGER = 0,
    DELINQ_DAYS_MAX D_INTEGER = 9999)
RETURNS (
    COUNTER D_INTEGER,
    SBA_K_ID D_ID,
    PANT_ID D_ID,
    SBA_NUMBER D_INTEGER,
    PANT_OPER_CALLED D_STR_100,
    K_KONTRACT_NO D_STR_CODE,
    KF_NAME D_STR_100,
    REG_NAME D_STR_CODE,
    NATION_STR D_STR_CODE,
    CITIZEN_STR D_STR_CODE,
    C_ID D_ID,
    C_PLASTCARD_NO D_STR_20,
    CL_CALL_QTY D_INTEGER,
    CL_TRY_QTY D_INTEGER,
    CL_LAST_CALL_DATE D_DATE,
    CL_LAST_TRY_DATE D_DATE,
    CON_CALL_QTY D_INTEGER,
    CON_TRY_QTY D_INTEGER,
    C_NAME_FIO D_STR_100,
    RESULT_STR D_STR_LONG,
    NEXT_CALL D_DATETIME,
    CAN_CALL D_BOOL,
    PROJECT D_STR_100,
    CLIENT_CELL_NO D_STR_100,
    OKS_DELINQ_DAY D_INTEGER)
AS
declare variable C_NATION D_INTEGER;
declare variable C_CITIZEN D_INTEGER;
declare variable K_FROM_KF_ID_OLD D_ID;
declare variable C_NATION_OLD D_INTEGER;
declare variable K_FROM_KF_ID D_ID;
declare variable C_CITIZEN_OLD D_INTEGER;
declare variable IS_OK D_BOOL;
declare variable PROJEC_TYPE D_SMALLINT;
declare variable PANT_DATE_BEGIN D_DATE;
declare variable K_ID D_ID;
declare variable CL_CALL_RESULT STR_100;
declare variable CL_MEMO D_STR_LONG;
declare variable PCL_RES D_INTEGER;
declare variable PCL_RES_OLD D_INTEGER;
declare variable LIMIT_ROWS D_INTEGER;
declare variable BALANCE_OS CURR_TYPE;
declare variable DELINQ_OS CURR_TYPE;
declare variable DB_ID D_INTEGER;
declare variable SQL_TEXT_1 D_STR_8100;
declare variable DELINQ_DAYS_MIN_TMP D_INTEGER;
declare variable DELINQ_DAYS_MAX_TMP D_INTEGER;
declare variable ACTIVE_DELINQ_DAYS D_INTEGER;
declare variable OKS_CLOSE_STATE D_SMALLINT;
declare variable DELINQ_DATE D_DATE;
begin
  kontract_no = coalesce(kontract_no, '');
  called_oper = coalesce(called_oper, '');
  kfolder_name = coalesce(kfolder_name, '');
  nation = coalesce(nation, 0);
  citizenship = coalesce(citizenship, 0);
  limit_rows = 0;
  counter = 0;

  delinq_days_max_tmp = 9999;
  delinq_days_min_tmp = 0;

   if (delinq_days_min < delinq_days_min_tmp) then
    delinq_days_min = delinq_days_min_tmp;
  if (delinq_days_max > delinq_days_max_tmp) then
    delinq_days_max = delinq_days_max_tmp;

  k_from_kf_id_old = 0;
  c_nation_old = 0;
  c_citizen_old = 0;
  pcl_res_old = -999;

  db_id = f_get_db_id();

  select rd.rd_code,
         cast(rd.rd_tag as d_integer)
  from refdata rd
  where rd.rd_main_code = 'OKTELL.Projects'
  and rd.rd_str_id = cast(:sbl_id as d_str_10)
  into :project,
       :projec_type;

  if (sbl_id is null) then
    exception e_default 'sbl_id is null ';

  sql_text_1 = 'select sba.sba_from_k_id, cast('''||f_datetostr(:rep_date)||''' as date) - oks.oks_delinq_date, oks_close_state, ';
  sql_text_1 = sql_text_1||'sba.sba_number, oks.oks_balance + oks.oks_out_os, oks.oks_delinq_os + oks.oks_out_os ';
  sql_text_1 = sql_text_1||'from oksummary oks ';
  sql_text_1 = sql_text_1||'inner join sb_akt sba on sba.sba_from_k_id = oks.oks_from_k_id ';
  sql_text_1 = sql_text_1||'where sba.sba_date_end >= '''||f_datetostr(:rep_date)||''' and ';
  sql_text_1 = sql_text_1||'sba.sba_date_begin <= sba.sba_date_end and ';
  sql_text_1 = sql_text_1||'sba_is_delete = 0 and ';
  sql_text_1 = sql_text_1||'sba.sba_status = 2 and ';
  sql_text_1 = sql_text_1||'sba.sba_from_sbl_id = '||:sbl_id||' and ';
  sql_text_1 = sql_text_1||''''||f_datetostr(:rep_date)||''' between oks.oks_date_begin and oks.oks_date_end ';

  if (kontract_no <> '') then
  begin
    select kont.k_id
    from kontract kont
    where kont.k_kontract_no = :kontract_no
    into :k_id;
  end

  if (k_id is not null) then
    sql_text_1 = sql_text_1||'and sba_from_k_id = '||:k_id||' ';
  sql_text_1 = sql_text_1||' and oks_balance + oks_out_os + oks_sh_perc + oks_out_perc > 0 ';
  sql_text_1 = sql_text_1||' order by sba.sba_from_k_id ';
  for execute statement(sql_text_1)
  into :sba_k_id, :oks_delinq_day, :oks_close_state, :sba_number, :balance_os, :delinq_os
  do
  begin
    balance_os = coalesce(balance_os, 0);
    delinq_os = coalesce(delinq_os, 0);

    if (db_id = 94401) then
    begin
        if (oks_close_state = 1) then
        begin
           delinq_date = null;
    
           select oks_delinq_date
           from oksummary
           where oks_from_k_id = :sba_k_id
           and oks_delinq_date < '31.12.9999'
           order by oks_date_begin desc
           rows 1
           into delinq_date;
            
           if (delinq_date is not null) then
                oks_delinq_day = :rep_date - delinq_date;
        end
        if (oks_delinq_day < 0) then oks_delinq_day = 0;

        if (not(oks_delinq_day between delinq_days_min and delinq_days_max)) then
            continue;
    end

    pant_id = null;

    if (db_id = 81001 and (balance_os < 500 or delinq_os < 500)) then
        continue;

    is_ok = 1;
    if ((kfolder_name <> '') and (kontract_no = '')) then
    begin
        select kal_new_kf_name
        from knaktlist kal
        inner join kontract kont on kont.k_from_kf_id = kal.kal_from_kf_id
        where kal.is_delete = 0 and
              :rep_date between kal_date and kal_date_end and
              kont.k_id = :sba_k_id rows 1
        into kf_name;
    
        if (kf_name <> kfolder_name) then
          is_ok = 0;
    end
    
    if (db_id = 81001 and projec_type = 3) then -- ДРВП
    begin
        if (not exists(select *
                       from drvp_actual_list t
                       where t.k_id = :sba_k_id)) then
          is_ok = 0;
    end
    
    if (is_ok = 1) then
    begin
        pant_oper_called = null;
        pant_date_begin = null;
    
        select pant.pant_id,
               pant.pant_oper_called,
               pant.pant_date_begin
        from pantera_kontracts pant
        where pant.pant_from_k_id = :sba_k_id and
              :rep_date between pant.pant_date_begin and pant.pant_date_end and
              pant.pant_project_type = :projec_type
              rows 1
        into :pant_id,
             :pant_oper_called,
             :pant_date_begin;
    
        if (pant_oper_called is null) then
            pant_oper_called = '';

        is_ok = 1;
        if ((pant_oper_called <> called_oper) and
            (called_oper <> '')) then
          is_ok = 0;
    
        if (pant_date_begin is null) then
          pant_date_begin = rep_date;
    
        if ((rep_date > pant_date_begin) and
            (called_oper = '')) then
          is_ok = 0;
    
        if (is_ok = 1) then
        begin
           for select kont.k_kontract_no,
                     k_from_kf_id,
                     reg.reg_name,
                     cl.c_nation,
                     cl.c_citizen,
                     cl.c_id,
                     cl.c_name_fio,
                     f_cont_info('clients', kncl.kc_from_c_id, 'cell_main', 'last')
              from kontract kont
              inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
              inner join clients cl on cl.c_id = kncl.kc_from_c_id
              inner join region reg on reg.reg_unicode = kont.k_last_reg_id
              where kont.k_id = :sba_k_id and
                    kncl.kc_amount_approved > 0 and
                    kont.k_id = kncl.kc_from_k_id and
                    cl.c_id = kncl.kc_from_c_id and
                    reg.reg_unicode = kont.k_last_reg_id
              into :k_kontract_no,
                   :k_from_kf_id,
                   :reg_name,
                   :c_nation,
                   :c_citizen,
                   :c_id,
                   :c_name_fio,
                   :c_plastcard_no
          do
          begin
            client_cell_no = f_cont_info('clients', c_id, 'cell_main', 'last');

            cl_try_qty = 0;
            cl_call_qty = 0;
            cl_last_call_date = null;
            cl_last_try_date = null;
            con_try_qty = 0;
            con_call_qty = 0;
            cl_call_result = '';
            cl_memo = '';
            result_str = '';
            next_call = null;
            can_call = 0;
    
            is_ok = 1;
            if ((citizenship <> 0) and
                (c_citizen <> citizenship)) then
              is_ok = 0;
            if ((nation <> 0) and
                (c_nation <> nation)) then
              is_ok = 0;
    
            if (is_ok = 1) then
            begin
              counter = counter + 1;
              if (not (counter between rows_min and rows_max)) then
                continue;
              if (counter > rows_max) then
                exit;

              if (called_oper <> '') then -- только для режима звонков показываю эти цифры
              begin
                select pds.pds_try_qty,
                       pds.pds_call_qty,
                       pds.pds_last_call_date,
                       pds.pds_last_try_date
                from pantera_data_svod pds
                where pds.pds_from_pan_id = :pant_id and
                      pds.pds_from_c_id = :c_id
                      rows 1
                into :cl_try_qty,
                     :cl_call_qty,
                     :cl_last_call_date,
                     :cl_last_try_date;
    
                select sum(pds.pds_try_qty),
                       sum(pds.pds_call_qty)
                from pantera_data_svod pds
                inner join contacts cont on cont.cc_id = pds.pds_from_cc_id
                where pds.pds_from_pan_id = :pant_id and
                      cont.cc_from_c_id = :c_id
                into :con_try_qty,
                     :con_call_qty;
    
                if (cl_try_qty is null) then cl_try_qty = 0;
                if (cl_call_qty is null) then cl_call_qty = 0;
                if (con_try_qty is null) then con_try_qty = 0;
                if (con_call_qty is null) then con_call_qty = 0;
    
                if (cl_try_qty > 0) then
                begin
                  select pc.pcl_res,
                         pc.pcl_memo
                  from pantera_clients pc
                  where pc.pcl_from_pan_id = :pant_id and
                        pc.pcl_from_c_id = :c_id
                  order by pc.pcl_called_datetime desc
                  rows 1
                  into :pcl_res,
                       :cl_memo;
    
                  if (pcl_res <> pcl_res_old) then
                  begin
                    cl_call_result = null;
    
                    select rd.rd_name
                    from refdata rd
                    where rd.rd_main_code = 'OKTELL.SirenaResults' and
                          rd.rd_tag = :pcl_res and
                          rd.rd_parent_id is not null
                    into :cl_call_result;
    
                    pcl_res_old = pcl_res;
                  end
    
                  cl_call_result = coalesce(cl_call_result, '');
                  cl_memo = coalesce(cl_memo, '');
    
                  result_str = 'Результат: ' || cl_call_result || '. Примечание: ' || cl_memo;
                end
    
                -- получаю дату следующего звонка
                select next_call_date,
                       can_call
                from hp_can_call_check(:sba_k_id, 'Clients', :c_id, null)
                into :next_call,
                     :can_call;
              end
    
              if (k_from_kf_id <> k_from_kf_id_old) then
              begin
                select kal_new_kf_name
                from knaktlist
                where is_delete = 0 and
                      :rep_date between kal_date and kal_date_end and
                      kal_from_kf_id = :k_from_kf_id
                      rows 1
                into kf_name;
    
                k_from_kf_id_old = k_from_kf_id;
              end
    
              if (c_nation <> c_nation_old) then
              begin
                nation_str = null;

                select ddr.rd_name
                from refdata ddr
                where ddr.rd_id = :c_nation
                into :nation_str;

                if (nation_str is null) then nation_str = '';
                c_nation_old = c_nation;
              end
    
              if (c_citizen <> c_citizen_old) then
              begin
                citizen_str = null;

                select ddr.rd_name
                from refdata ddr
                where ddr.rd_id = :c_citizen
                into :citizen_str;

                if (citizen_str is null) then citizen_str = '';
                c_citizen_old = c_citizen;
              end
    
              /*
              if (called_oper = '') then
              begin
                if (limit_rows <= 1000) then
                begin
                  limit_rows = limit_rows + 1;
                  suspend;
                end
                else
                  exit;
              end
              else
              begin
                suspend;
              end
              */
              suspend;
            end
          end
        end
    end
  end
end
