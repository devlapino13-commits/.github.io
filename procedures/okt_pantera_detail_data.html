CREATE PROCEDURE OKT_PANTERA_DETAIL_DATA(
    K_ID ID_TYPE,
    NUM_ID INTEGER,
    C_ID ID_TYPE,
    CC_ID ID_TYPE,
    PORUCH_ID ID_TYPE,
    REP_DATE DATE,
    REP_TYPE SMALLINT,
    PROJECT_TYPE SMALLINT)
RETURNS (
    PANT_ID ID_TYPE,
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED CURR_TYPE,
    K_PERIOD INTEGER,
    PDS_MEMO_QTY INTEGER,
    ZAD_SUMM CURR_TYPE,
    DELINQ_DAYS INTEGER,
    PDS_LAST_MEMO VARCHAR(1000),
    K_KREDLINE_END DATE,
    EUROPLAT_ACC_NO VARCHAR(10),
    NEXT_PAY_DATE DATE,
    LAST_PAY_DATE DATE,
    KRED_FULL_BALANCE CURR_TYPE,
    QTY_LEFT INTEGER,
    DELINQ_OS CURR_TYPE,
    DELINQ_PERC CURR_TYPE,
    DELINQ_FINE CURR_TYPE,
    BALANCE_OS CURR_TYPE,
    BALANCE_PERC CURR_TYPE,
    BALANCE_FINE CURR_TYPE,
    C_BIRTH_DATE DATE)
AS
declare variable SH_MIN_DATE date;
declare variable SH_DATE date;
declare variable ZAD_SUMM_TEMP CURR_TYPE;
declare variable ZAD_OS_TEMP CURR_TYPE;
declare variable ZAD_PERC_TEMP CURR_TYPE;
declare variable ZAD_FINE_TEMP CURR_TYPE;
begin
  /*   PROJECT_TYPE
      1 - Пантера
      4 - Сирена
  */
  c_id = coalesce(c_id, 0);
  cc_id = coalesce(cc_id, 0);
  poruch_id = coalesce(poruch_id, 0);

  select pk.pant_id
  from pantera_kontracts pk
  where pk.pant_from_k_id = :k_id and
        :rep_date between pk.pant_date_begin and pk.pant_date_end and
        pk.pant_project_type = :project_type
  into :pant_id;

  if (rep_type = 1) then -- для первого открытия формы по группе нужен полный набор данных
  begin
    select kont.k_disb_date, kont.k_amount_approved, kont.k_period, kont.k_kredline_end -- палновая дата закрытия займа
    from kontract kont
    where kont.k_id = :k_id
    into :k_disb_date, :k_amount_approved, :k_period, :k_kredline_end;

    sh_min_date = null;
    zad_summ = 0;
    next_pay_date = null;
    last_pay_date = null;
    kred_full_balance = 0;
    qty_left = 0;

    delinq_os = 0;
    delinq_perc = 0;
    delinq_fine = 0;

    balance_os = 0;
    balance_perc = 0;
    balance_fine = 0;

    for select sh_date , sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine), sum(zad_os + sh_calc_perc + sh_calc_fine)
        from aarep_make_zad_list(:k_id, :rep_date, 1)
        where sh_date is not null
        group by sh_date
        order by sh_date
        into :sh_date, :zad_os_temp, :zad_perc_temp, :zad_fine_temp, :zad_summ_temp
    do
    begin
      if (sh_date < :rep_date) then
      begin
        if (sh_min_date is null) then
          sh_min_date = sh_date;
        zad_summ = zad_summ + zad_summ_temp;

        delinq_os = delinq_os + zad_os_temp;
        delinq_perc = delinq_perc + zad_perc_temp;
        delinq_fine = delinq_fine + zad_fine_temp;
      end
      if (sh_date >= rep_date) then
      begin
        if (next_pay_date is null) then
          next_pay_date = sh_date;
      end
      last_pay_date = sh_date;

      kred_full_balance = kred_full_balance + zad_summ_temp;

      balance_os = balance_os + zad_os_temp;
      balance_perc = balance_perc + zad_perc_temp;
      balance_fine = balance_fine + zad_fine_temp;

      qty_left = qty_left + 1;
    end

    if (sh_min_date is not null) then
      delinq_days = datediff(day, sh_min_date, :rep_date);
  end
  if (delinq_days is null) then
    delinq_days = 0;

  pds_last_memo = '';
  if (rep_type in (1, 2)) then -- если это клиент
  begin
    select pds.pds_memo_qty, pds.pds_last_memo
    from pantera_data_svod pds
    where pds.pds_from_pan_id = :pant_id and
          pds.pds_from_c_id = :c_id
    into :pds_memo_qty, pds_last_memo;
  end
  if (rep_type = 3) then -- если это контактное лицо
  begin
    if (poruch_id = 0) then
    begin
      select pds.pds_memo_qty, pds_last_memo
      from pantera_data_svod pds
      where pds.pds_from_pan_id = :pant_id and
            pds.pds_from_cc_id = :cc_id
      into :pds_memo_qty, :pds_last_memo;
    end
    else
    begin
      select pds.pds_memo_qty, pds_last_memo
      from pantera_data_svod pds
      where pds.pds_from_pan_id = :pant_id and
            pds.pds_from_poruch_id = :poruch_id
      into :pds_memo_qty, :pds_last_memo;
    end
  end

  if (pds_memo_qty is null) then
    pds_memo_qty = 0;

  europlat_acc_no = '';

  select uniq_acc_str
  from hp_show_clientuniq_acc_no(:c_id)
  into :europlat_acc_no; -- № Лицевого счета

  select cl.c_birth_date
  from clients cl
  where cl.c_id = :c_id
  into :c_birth_date;

  suspend;

end
