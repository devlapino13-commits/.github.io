CREATE PROCEDURE HP_SHOW_CONDITION_IGNORE(
    SHOW_CODE D_STR_CODE,
    C_ID D_ID = 0)
RETURNS (
    CDM_ID D_INTEGER,
    CDM_MEMO D_STR_255,
    CDM_OBJECT_KIND D_STR_CODE,
    MA_NO D_INTEGER,
    MA_SUMMA_MINUS D_CURRENCY,
    MA_DATE_APP D_DATE,
    MA_ID D_INTEGER,
    MA_NUMBER D_INTEGER)
AS
begin
  -- RFCKG-3846 Модуль исключения блоков для конкретных заявок/контрактов
  -- Процедура показывает информацию для модуля
  -- show_code = 'condition' процедура отображает болки в окне "Выбор блока"
  -- show_code = 'miniappByClient' процедура отображает заявки клиента окне "Выбор заявки"
  cdm_id = 0;
  cdm_memo = '';
  cdm_object_kind = '';
  ma_no = 0;
  ma_summa_minus = 0;
  ma_date_app = '31.12.9999';
  ma_id = 0;
  ma_number = 0;

  if (show_code = 'condition') then
  begin
    for select cdm.cdm_id,
               cdm.cdm_memo,
               cdm.cdm_object_kind
        from conditionmain cdm
        order by cdm.cdm_id
        into cdm_id,
             cdm_memo,
             cdm_object_kind
    do
    begin
      suspend;
    end
  end

  if (show_code = 'miniappByClient' and c_id <> 0) then
  begin
    for select ma.ma_no,
               ma.ma_summa_minus,
               ma.ma_date_app,
               ma.ma_id,
               ma.ma_number
        from miniapp ma
        inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
        where mac.mac_from_c_id = :c_id and
              mac.mac_amount > 0 and
              ma.ma_is_active = 1 and
              ma.ma_date_end = '31.12.9999'
        order by ma.ma_date_app
        into ma_no,
             ma_summa_minus,
             ma_date_app,
             ma_id,
             ma_number
    do
    begin
      suspend;
    end
  end
end
