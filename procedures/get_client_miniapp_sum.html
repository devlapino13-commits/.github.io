CREATE PROCEDURE GET_CLIENT_MINIAPP_SUM(
    C_ID D_ID)
RETURNS (
    MA_SUM D_INTEGER)
AS
declare variable ma_id d_id;
declare variable mac_amount d_integer;
declare variable tkt_id d_id;
declare variable ma_id_list d_str_long;
declare variable mc_purchase_sum d_currency;
begin
  ma_id_list = '.'; --Проверенные заявки
  ma_sum = 0;

  -- Проходим по всем кредитным заявкам клиента
  for select ma.ma_id, mac.mac_amount
      from miniappclient mac
      inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id and
            ma.ma_date_end = '31.12.9999' and
            ma.ma_is_active = 1
      where mac.mac_from_c_id = :c_id and
            mac.mac_amount > 0
      into :ma_id, :mac_amount
  do
  begin
    -- Если заявка уже имеет выданный кредит то пропускаем
    if (exists(select *
               from kontract k
               inner join kontstate ks on ks.ks_from_k_id = k.k_id and
                     current_date between ks.ks_date_begin and ks.ks_date_end and
                     ks.ks_ol_is_delete = 0 and
                     ks.ks_state <> 0
               where k.k_from_ma_app_id = :ma_id)) then
      continue;
    --Если контракта нет или контракт не активный/закрытый, то прибавляем к сумме и добавляем ID
    ma_id_list = ma_id_list || ma_id || '.';
    ma_sum = ma_sum + mac_amount;
  end

  -- Проходим по внешним заявкам
  for select ema.ema_from_tkt_id, ema.ema_amount
      from extminiapp ema
      inner join product pr on pr.pr_id = ema.ema_from_pr_id and
            pr.pr_code containing '.sugarmill.'
      where ema.ema_from_c_id = :c_id and
            ema.ema_amount > 0 and
            ema.ema_status = 0
      into tkt_id, mac_amount
  do
  begin
    -- Если заявку уже проверили в блоке выше, то пропускаем
    if (exists(select *
               from miniapp ma
               where ma.ma_from_tkt_id = :tkt_id and
                     ma.ma_date_end = '31.12.9999' and
                     ma.ma_is_active = 1 and
                     :ma_id_list containing ma.ma_id)) then
      continue;
    ma_sum = ma_sum + mac_amount;
  end

  -- Сумма в еще необработанных заявках(не отсканированы) с КМП
  select coalesce(sum(mp.mp_amount),0)
  from mc_purchase mp
  inner join mc_login mcl on mcl.mcl_id = mp.mp_from_mcl_id and
                             mcl.mcl_from_c_id = :c_id
  where mp.mp_from_ma_id is null and
        mp.mp_from_exa_id is null and
        mp.mp_status = 2 and
        mp.mp_is_finished = 1
  into :mc_purchase_sum;

  ma_sum = ma_sum + :mc_purchase_sum;
  suspend;
end
