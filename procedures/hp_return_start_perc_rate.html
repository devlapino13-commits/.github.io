CREATE PROCEDURE HP_RETURN_START_PERC_RATE(
    K_ID D_ID,
    OL_DATE D_DATE,
    TASK_MEMO STR_100)
AS
declare variable kf_id d_id;
declare variable un_id d_id;
declare variable res_ol_id d_id;
declare variable new_kd_percent d_percent;
declare variable kd_percent_per_day d_percent;
declare variable kd_percent_effect d_percent;
declare variable kd_percent_effect_fifd d_percent;
declare variable kd_fine_percent d_percent;
declare variable reg_id d_integer;
declare variable tmp_amount d_currency;
declare variable k_kontract_no d_str_100;
declare variable k_kontract_date d_date;
declare variable ol_memo d_str_long;
declare variable new_kd_id d_integer;
declare variable kd_float_perc_level d_integer;
begin
    -- процедура возращает начальную % ставку по договору
    un_id = 1000000000000;

    select kont.k_from_kf_id,
           kont.k_kontract_no,
           k_kontract_date
    from kontract kont
    where kont.k_id = :k_id
    into :kf_id,
         :k_kontract_no,
         :k_kontract_date;

    select kal_new_reg_id
    from knaktlist k
    where k.kal_from_kf_id = :kf_id and
        k.is_delete = 0 and
        k.kal_date <= :ol_date and
        k.kal_date_end >= :ol_date
    into reg_id;
    
    if (reg_id is null) then
    exception e_default 'Office not found';
    
    ol_memo = f_format('Изменение плавающей процентной ставки по договору %1 от %2', k_kontract_no, f_datetostr(k_kontract_date));
    
    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_memo, is_delete,
                        ol_user_id)
    values (:reg_id, 1, 12, :ol_date, 9, :ol_memo, 0, :un_id)
    returning ol_id
    into :res_ol_id;

    select kd.kd_percent,
           kd.kd_percent_per_day,
           kd.kd_percent_effect,
           kd.kd_percent_effect_fifd,
           kd.kd_fine_percent
    from kontdata kd
    where kd.kd_is_primary = 1
    and kd.kd_ol_is_delete = 0
    and kd.kd_from_k_id = :k_id
    into new_kd_percent,
         kd_percent_per_day,
         kd_percent_effect,
         kd_percent_effect_fifd,
         kd_fine_percent;

    select kd.kd_float_perc_level
    from kontdata kd
    where kd.kd_ol_is_delete = 0
    and kd.kd_from_k_id = :k_id
    and current_date between kd.kd_ol_date and kd.kd_date_end
    into :kd_float_perc_level;

    insert into kontdata (kd_from_k_id, kd_from_ol_id, kd_ol_date, kd_ol_is_delete, kd_percent, kd_percent_per_day,
                        kd_percent_effect, kd_percent_effect_fifd, kd_memo, kd_fine_percent, kd_float_perc_level)
    values (:k_id, :res_ol_id, :ol_date, 0, :new_kd_percent, :kd_percent_per_day, :kd_percent_effect,
          :kd_percent_effect_fifd, 'Изменение процентной ставки согласно: ' || :task_memo,
          :kd_fine_percent, :kd_float_perc_level)
    returning kd_id
    into new_kd_id;
    
    -- Построение задолженности по графику на дату
    select res_amount
    from hp_repay_fill(:k_id, 0, :ol_date, 0, 1, 1, 1, 1, 1)
    into :tmp_amount;
    
    -- Автореструктуризация графика
    tmp_amount = 0;
    select has_advanced_pay
    from hp_repay_reschedule(:ol_date, 3, 1, :k_id)
    into :tmp_amount; -- Имеются ли досрочные погашения
    tmp_amount = 0;
    
    -- Сохранение на Operkred сумм автореструктуризации
    execute procedure hp_repay_save_operkred(:k_id, 0, :res_ol_id, :ol_date, 0, 3, 0);
    
    select kps_psk
    from hp_edoc_ins_kontpsk(:k_id, :res_ol_id, :un_id)
    into :kd_percent_effect;
    
    if (kd_percent_effect is not null) then
    begin
    update kontdata
    set kd_percent_effect_fifd = :kd_percent_effect
    where kd_id = :new_kd_id;
    end
end
