CREATE PROCEDURE SPOUSES_CONSENT_CHECK(
    MA_ID D_ID NOT NULL)
RETURNS (
    RES_ERROR_TEXT D_STR_255)
AS
declare variable C_ID D_ID;
declare variable MA_DATE_OPER D_DATE;
declare variable MA_DATE_APP D_DATE;
declare variable SUM_MA_KC_AMOUNT D_CURRENCY;
declare variable RES_STR_VALUES D_STR_100;
declare variable CREL_OTHER_C_ID D_ID;
declare variable PARENT_REG_ID D_INTEGER;
declare variable PR_ID D_INTEGER;
begin
  select reg.parent,
         ma.ma_from_pr_id,
         ma.ma_date_oper,
         ma.ma_date_app
  from miniapp ma
  inner join region reg on reg.reg_unicode = ma.ma_office_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end = '31.12.9999'
  into parent_reg_id,
       pr_id,
       ma_date_oper,
       ma_date_app;

  if (exists(select *
             from product pr
             where pr.pr_id = :pr_id and
                   (pr.pr_code containing '.sugarmill.' or pr.pr_code containing '.installment_goods.' or pr.pr_code containing '.insur_kontrakt.'))) then
    exit; -- для сах. завода и товаров в расрочку согласие супруга не требуется
          -- И для «Страхование в кредит» RFCKG-4831

  for select mc.mac_from_c_id
      from miniappclient mc
      where mc.mac_from_ma_id = :ma_id and
            mc.mac_amount > 0
      into :c_id
  do
  begin
    sum_ma_kc_amount = 0;
    select ma_kc_amount
    from hp_get_ma_kc_amount(:ma_id, :c_id)
    into sum_ma_kc_amount;

    if (sum_ma_kc_amount < 100000) then
      continue; -- если общая сумма менее 100 тыс. сомов, то согласие супруга не требуется (RFCKG-3181)

    res_str_values = null;
    select rd.rd_code
    from clients cs
    inner join refdata rd on rd.rd_id = cs.c_family
    where cs.c_id = :c_id
    into res_str_values;

    crel_other_c_id = null;
    select crel_other_c_id
    from clientrelation crel
    where crel.crel_from_c_id = :c_id and
          crel.crel_is_active = 1 and
          crel.crel_from_rel_id in ('1', '2')
    into crel_other_c_id;

    if (res_str_values = 'Family.married' or crel_other_c_id is not null) then
    begin
      -- если нет подписанного электронного документа согласие супруга
      if (not exists(select *
                     from edoc_list el
                     inner join edoc_cfg edc on edc.edc_id = el.edl_from_edc_id and edc.edc_code = 'loan.spouses_consent'
                     where el.edl_object_id = :ma_id and
                           el.edl_object_kind = 11 and
                           el.edl_status = 1)) then
      begin
        if (not exists(select *
                       from clientdocs cds
                       inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                       inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
                       where cds.cds_from_c_id = :c_id and
                             cds.cds_is_delete = 0 and
                             pk.pk_is_delete = 0 and
                             ((pdt.pdt_code in ('certificate_annulment', -- свид. о расторжении брака
                                            'certificate_death', -- свид. о смерти
                                            'judicial_act', -- судебный акт
                                            'power_of_attorney') and -- дов-ть супруга
                             :ma_date_oper between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.9999')) or (pdt.pdt_code = 'consent_to_a_loan' and -- бумажное согласие супруга
                             cds.cds_doc_date_begin between :ma_date_app and :ma_date_oper)))) then -- согласие супруга проверяем в промежутке между датой заявки и датой выдачи, согласно задаче RFCKG-1857
        begin
          res_error_text = 'Прикрепите согласие супруга, дата закрепления должна быть в промежутке между датой создания заявки и датой выдачи займа.';
          suspend;
        end
      end
    end
  end
end
