CREATE PROCEDURE CM_AZ_0050_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_id id_type;
declare variable reg_id integer;
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_percent_per_day perc_type;
declare variable pr_product_kind integer;
declare variable k_circle integer;
declare variable k_kommis_perc perc_type;
declare variable k_amount_approved curr_type;
declare variable pr_name varchar(50);
declare variable pr_code varchar(100);
declare variable c_id id_type;
begin
  for
  select k_disb_date, k_from_kf_id, k_last_reg_id, k_kolich_repay, k_kolich_client_approved, k_percent_per_day,
           pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code
  from kontract, product
  where k_id=:k_id
  and k_from_pr_id = pr_id
  and pr_code containing 'akcia'
  into :kf_disb_date, :kf_id, :reg_id, :k_kolich_repay, :k_kolich_client_approved, k_percent_per_day,
                 :pr_product_kind, k_circle, k_kommis_perc, k_amount_approved, pr_name, pr_code
  do
  begin
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id
        and kc_amount_approved > 0
    into c_id
    do
    begin
      for select c_name_fio
          from knclient, kontstate, clients, kontract
          where kc_from_c_id = :c_id
          and kc_amount_approved > 0
          and kc_from_k_id = ks_from_k_id
          and kc_from_k_id = k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and :kf_disb_date between ks_date_begin and ks_date_end
          and kc_from_c_id = c_id
          into res_info_text
      do
      begin
       -- У клиента есть активный кредит!
        res_error_is_critical = 2;
        res_error_text='müştərinin aktiv krediti var!';
        suspend;    
      end
    end
  end
end
