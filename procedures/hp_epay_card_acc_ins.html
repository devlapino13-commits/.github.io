CREATE PROCEDURE HP_EPAY_CARD_ACC_INS(
    C_ID D_ID,
    CARD_ID D_STR_CODE,
    CARD_NO D_STR_CODE,
    UPD_DATETIME D_DATETIME = null)
AS
declare variable epc_id d_integer;
declare variable epa_id d_integer;
begin
  -- Привязка банковской карты к клиенту по новой структуре (epay_acc)

  if (upd_datetime is null) then
    upd_datetime = current_timestamp;

  select epc_id
  from epay_card
  where epc_card_code = :card_no and
        epc_card_id = :card_id
  into epc_id;

  if (epc_id is null) then
  begin
    insert into epay_card (epc_card_id, epc_card_code)
    values (:card_id, :card_no)
    returning epc_id
    into :epc_id;
  end

  select epa_id
  from epay_acc
  where epa_object_kind = f_objkind('epay_card', 'id') and
        epa_object_id = :epc_id and
        epa_from_c_id = :c_id
  order by epa_id desc
  rows 1
  into epa_id;

  if (epa_id is not null) then
  begin
    update epay_acc
    set epa_status = 1,
        epa_datetime = :upd_datetime
    where epa_id = :epa_id;
  end else
  begin
    insert into epay_acc (epa_from_c_id, epa_object_kind, epa_object_id, epa_status, epa_from_un_id, epa_datetime)
    values (:c_id, f_objkind('epay_card', 'id'), :epc_id, 1, 1000000000000, :upd_datetime);
  end
end
