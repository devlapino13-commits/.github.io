CREATE PROCEDURE HP_KONTMAT_ACTION(
    UN_ID DECIMAL(18,0),
    ACTION_KIND VARCHAR(255),
    KONTMAT_DATA BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    IS_MP SMALLINT = 0,
    K_ID BIGINT = 0)
RETURNS (
    RES_ID INTEGER,
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable region_id integer;
declare variable seller_id decimal(18,0);
declare variable matcateg_id decimal(18,0);
declare variable workplace_id decimal(18,0);
declare variable account_dt varchar(20);
declare variable json_key varchar(255);
declare variable json_value varchar(255);
declare variable oper_date date;
declare variable workplace_name varchar(255);
declare variable seller_name varchar(100);
declare variable matcateg_name varchar(100);
declare variable tmp_str varchar(5000);
declare variable account_dt_name varchar(255);
declare variable account_kt varchar(20);
declare variable k_disb_date date;
declare variable matotvet_id decimal(18,0);
declare variable matotvet_name varchar(255);
declare variable txt varchar(1000);
declare variable ol_id decimal(18,0);
declare variable curr_region_id integer;
declare variable ol_doc_kind integer;
declare variable ol_memo varchar(1024);
declare variable line_no integer;
declare variable ml_id decimal(18,0);
declare variable kol decimal(18,4) = 0;
declare variable kml_count decimal(18,4) = 0;
declare variable kml_total decimal(18,4) = 0;
declare variable kml_commis d_currency;
declare variable ml_name varchar(512);
declare variable ob_currency varchar(3);
declare variable kml_id integer;
declare variable ml_model varchar(255);
declare variable kml_cost decimal(18,4) = 0;
declare variable kml_bill_id decimal(18,0);
declare variable kml_bill_line_no integer;
declare variable is_buyed integer;
declare variable kml_guarantee_month decimal(18,4) = 0;
declare variable kf_name varchar(100);
declare variable client_id decimal(18,0);
declare variable fio d_str_100;
declare variable k_status integer;
declare variable is_selected integer;
declare variable selected_kml_id integer;
declare variable error_in_date integer;
declare variable ag_agreement_no varchar(100);
declare variable ag_agreement_date date;
declare variable k_kontract_no varchar(100);
declare variable kml_id_lst d_str_8100;
declare variable rd_id d_integer;
declare variable rd_id_1 d_integer;
declare variable rd_id_2 d_integer;
declare variable pr_code d_str_100;
declare variable obj_id d_id;
declare variable obj_kind d_integer;
declare variable obj_code d_str_code;
declare variable exec_sql d_str_long;
declare variable method d_smallint;
declare variable tmp_str1 varchar(5000);
declare variable ob_detail d_str_long;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
begin
  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.IPF_Sellers'
  into :rd_id;

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.tsp_and_ipf'
  into :rd_id_1;

  select rd_id
  from refdata
  where rd_code = 'AgreementKinds.IPF_WithCommission'
  into :rd_id_2;

  res_id = 1;
  res_text = '';
  method = 1;

  select cm_default_currency
  from company
  into :ob_currency;

  -- KontMat_Blob формат:
  -- matcateg_id
  -- workplace_id
  -- matotvet_id
  -- account_dt
  -- account_kt
  -- seller_id
  -- oper_date

  for select json_key, json_value
      from fm_json_parse(:kontmat_data, 0, 'main')
      into :json_key, :json_value
  do
  begin
    if (json_key = 'matcateg_id' or json_key = 'workplace_id' or json_key = 'matotvet_id' or json_key = 'seller_id' -- AG_ID из Agreements
        or json_key = 'region_id') then
    begin
      if (coalesce(json_value, '') = '') then
        json_value = '0';
    end

    if (json_key = 'matcateg_id') then
      matcateg_id = json_value;
    if (json_key = 'workplace_id') then
      workplace_id = json_value;
    if (json_key = 'matotvet_id') then
      matotvet_id = json_value;
    if (json_key = 'seller_id') then
      seller_id = json_value;
    if (json_key = 'obj_id') then
      obj_id = json_value;
    if (json_key = 'obj_kind') then
      obj_kind = json_value;
    if (json_key = 'region_id') then
      region_id = json_value;
    if (json_key = 'method') then
      method = json_value;
    if (json_key = 'oper_date') then
      oper_date = json_value;
    if (json_key = 'account_dt') then
      account_dt = json_value;
    if (json_key = 'account_kt') then
      account_kt = json_value;
  end -- Конец персинга

  if (obj_kind is null) then
    select obk.obk_id, obk.obk_code
    from objkind obk
    where obk.obk_code = 'agreements'
    into :obj_kind, :obj_code;
  else
    select obk.obk_id, obk.obk_code
    from objkind obk
    where obk.obk_id = :obj_kind
    into :obj_kind, :obj_code;

  if (obj_code = 'agreements' and
      :seller_id is null) then
    seller_id = obj_id;

  if (obj_code = 'kontract') then
    select kal.kal_new_reg_id
    from kontract k
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id and
          kal.is_delete = 0 and
          kal.kal_date_end >= current_date
    where k.k_id = :obj_id
    into :region_id;

  if (:action_kind <> 'doSave' and
      :method > 0) then
  begin
    if (obj_code = 'agreements' and
        coalesce(seller_id, 0) = 0) then
    begin
      res_id = -1;
      res_text = 'Поставщик не указан';
      suspend;
      exit;
    end
    if (obj_code = 'kontract' and
        coalesce(obj_id, 0) = 0) then
    begin
      res_id = -1;
      res_text = 'Необходимо выбрать договор';
      suspend;
      exit;
    end
  end

  select ag_object_id, ag_agreement_no, ag_agreement_date, c_name_fio
  from agreements ag
    inner join clients
      on c_id = ag_object_id
  where ag_id = :seller_id and
        ag.ag_from_rd_id in (:rd_id, :rd_id_1, :rd_id_2)
  into :client_id, ag_agreement_no, ag_agreement_date, fio;

  if (ag_agreement_no is null) then
    ag_agreement_no = '';

  if (action_kind = 'doLoad' and
      is_mp = 0) then -- загрузки первичных данных
  begin

    -- Категория ТМЦ
    --    select mc_id
    --    from matcateg
    --    where mc_name='ТМЦ для купли-продажи'
    --    rows 1
    --    into :matcateg_id;
    --    if (matcateg_id is null) then matcateg_id=97;
    --
    matcateg_id = 133; -- в КР все ТМЦ по Мурабаха сажаем в одну категорию

    select mc_name
    from matcateg
    where mc_id = :matcateg_id
    into matcateg_name;
    matcateg_name = replace(matcateg_name, '"', '\"');
    res_text = '"main":{';
    res_text = BLOB_APPEND(res_text, '"matcateg_id":"' || :matcateg_id || '"');
    if (matotvet_id is null) then
      matotvet_id = 0;
    res_text = BLOB_APPEND(res_text, ',"region_id":"' || :region_id || '"');
    res_text = BLOB_APPEND(res_text, ',"matotvet_id":"' || :matotvet_id || '"');

    select wp_id, wp_name, wp_ml_account
    from workplace
    where wp_from_reg_id = :region_id and
          wp_status = 6 and
          wp_is_active = 1 and
          (wp_name like '%Собственность%' or
          wp_name containing 'Истиснаа') rows 1
    into :workplace_id, workplace_name, account_dt;
    if (workplace_id is null) then
      workplace_id = 0;
    if (account_dt is null) then
      account_dt = '';

    res_text = BLOB_APPEND(res_text, ',"workplace_id":"' || :workplace_id || '"');
    res_text = BLOB_APPEND(res_text, ',"account_dt":"' || :account_dt || '"');

    account_kt = '';
    res_text = BLOB_APPEND(res_text, ',"account_kt":"' || :account_kt || '"');
    if (obj_code = 'agreements') then
      res_text = BLOB_APPEND(res_text, ',"seller_id":"' || :seller_id || '"');

    res_text = BLOB_APPEND(res_text, '}');
    tmp_str = '{"id":"' || :matcateg_id || '", "name":"' || matcateg_name || '"}';
    res_text = BLOB_APPEND(res_text, ',"matcategs":[' || :tmp_str || ']');

    tmp_str = '';

    res_text = BLOB_APPEND(res_text, ',"accounts_kt":[' ||(select list('{"id":"' || al_id || '", "name":"' || replace(al_name, '"', '\"') || '", "code":"' || al_account || '"}', ',')
                                                 from acc_list
                                                 where al_from_reg_id = :region_id and
                                                       al_from_bg_id in ('10001', '10150', '21199', '11571')) || ']');

    for select wp_id, replace(wp_name, '"', '\"')
        from workplace
        where wp_from_reg_id = :region_id and
              wp_status = 6 and
              wp_is_active = 1 and
              (wp_name like '%Собственность%' or
              wp_name containing 'Истиснаа')
        into :workplace_id, workplace_name
    do
    begin
      tmp_str = tmp_str || iif(tmp_str = '', '', ',') || '{"id":"' || :workplace_id || '", "name":"' || workplace_name || '"}';
    end
    res_text = BLOB_APPEND(res_text, ',"workplaces":[' || :tmp_str || ']');

    res_text = BLOB_APPEND(res_text, ',"accounts_dt":[' ||(select list('{"id":"' || wp_ml_account || '", "name":"' || replace(al_name, '"', '\"') || '", "code":"' || al_account || '"}', ',')
                                                 from workplace, acc_list
                                                 where wp_from_reg_id = :region_id and
                                                       wp_status = 6 and
                                                       wp_is_active = 1 and
                                                       (wp_name like '%Собственность%' or
                                                       wp_name containing 'Истиснаа') and
                                                       wp_ml_account = al_id) || ']');

    tmp_str = '';

    if (matotvet_id <> 0) then
    begin
      select replace(c_name_fio, '"', '\"')
      from clients
      where c_id = :matotvet_id
      into :matotvet_name;

      tmp_str = '{"id":"' || :matotvet_id || '", "name":"' || matotvet_name || '"}';
    end

    txt = '';
    for select '{"id": "' || manager_c_id || '","name":"' || replace(manager_fio, '"', '\"') || '"}'
        from aarep_manager_list(:region_id, current_date, 0)
        where manager_c_id <> :matotvet_id
        into txt
    do
    begin
      tmp_str = tmp_str || iif(tmp_str = '', '', ',') || txt;
    end
    res_text = BLOB_APPEND(res_text, ',"matotvets":[' || :tmp_str || ']');

    select replace(al_name, '"', '\"')
    from acc_list
    where al_id = :account_dt
    into :account_dt_name;
    if (account_dt_name is null) then
      account_dt_name = 'Не указан';
    if (obj_code = 'agreements') then
      exec_sql = f_format('select %1 from rdb$database', :seller_id);
    else
      exec_sql = f_format('select distinct(kml.kml_from_ag_id)
                            from kontmat kml
                            where kml.kml_object_kind = 9 and
                                  kml.kml_status = 1 and
                                  kml.kml_object_id = %1 and
                                  not exists(select *
                                             from kontmat_own
                                             where kmo_kml_id = kml.kml_id)', :obj_id);

    for execute statement(:exec_sql)
            into :seller_id
    do
    begin
      select replace(ag_caption, '"', '\"')
      from agreements ag
      where ag_id = :seller_id and
            ag.ag_from_rd_id in (:rd_id, :rd_id_1, :rd_id_2)
      into :seller_name;

      tmp_str = '{"id":"' || :seller_id || '", "name":"' || seller_name || '"}';
      res_text = (select result
                  from hp_append_arr_from_json('sellers', '{' || :res_text || '}', :tmp_str, 1));

      kol = 0;
      tmp_str = '';

      for select kml_id, ml_name, ml_model, kml_count, kml_cost, kml_total, kml_commis, kml_bill_id, kml_bill_line_no,
                 kml_guarantee_month, k_kontract_no, kal_new_kf_name, k_status
          from kontmat kml
          inner join matlist ml on ml.ml_id = kml.kml_from_ml_id
          inner join kontract k on kml.kml_object_id = k.k_id and
                k.k_status >= 0
          inner join kontstate ks on ks.ks_from_k_id = kml.kml_object_id and
                ks.ks_ol_is_delete = 0 and
                current_date between ks.ks_date_begin and ks.ks_date_end and
                ks.ks_state = 0
          inner join product pr on k.k_from_pr_id = pr.pr_id and
                pr.pr_code not like '%.IDJARA.%'
          inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id and
                k.k_disb_date between kal.kal_date and kal.kal_date_end and
                kal.kal_new_reg_id = :region_id and
                kal.is_delete = 0
          where kml.kml_object_kind = 9 and
                kml.kml_status = 1 and
                kml.kml_from_ag_id = :seller_id and
                iif(:obj_code = 'kontract', kml.kml_object_id = :obj_id, kml.kml_object_id > 0) and
                not exists(select *
                           from kontmat_own
                           where kmo_kml_id = kml.kml_id)
          into :kml_id, ml_name, ml_model, kml_count, kml_cost, kml_total, kml_commis, kml_bill_id, kml_bill_line_no,
               kml_guarantee_month, :k_kontract_no, kf_name, k_status
      do
      begin
        if (coalesce(k_kontract_no, '') = '') then
          k_kontract_no = '-';
        if (coalesce(kf_name, '') = '') then
          kf_name = '-';
        tmp_str1 = '';
        is_selected = 0;

        if (kml_guarantee_month is null) then
          kml_guarantee_month = 0;
        is_buyed = 0;
        if (kml_bill_id is not null) then
        begin
          if (exists(select *
                     from opermat, operlist
                     where om_from_ol_id = :kml_bill_id and
                           om_number = :kml_bill_line_no and
                           om_from_ol_id = ol_id and
                           operlist.is_delete = 0)) then
            is_buyed = 1;
        end
        if (is_buyed = 0) then
        begin
          kol = kol + 1;
          if (k_status >= 0 and
              kml_cost <> 0 and
              kml_count <> 0) then
            --S.K.
            --is_selected = 1;
            tmp_str1 = '{' || f_json_make_pair('kml_id', kml_id, 'str') || ',' || f_json_make_pair('ml_name', ml_name, 'str') || ',' || f_json_make_pair('ml_model', ml_model, 'str') || ',' || f_json_make_pair('kml_count', kml_count, 'str') || ',' || f_json_make_pair('kml_cost', kml_cost, 'str') || ',' || f_json_make_pair('kml_total', kml_total, 'str') || ',' || f_json_make_pair('k_kontract_no', k_kontract_no, 'str') || ',' || f_json_make_pair('kf_name', kf_name, 'str') || ',' || f_json_make_pair('k_status', k_status, 'str') || ',' || f_json_make_pair('is_selected', iif(:is_mp = 1, 1, is_selected), 'str') || '}';
          if (tmp_str1 <> '') then
            tmp_str = tmp_str || tmp_str1 || ', ';
        end
      end
      if (tmp_str <> '') then
      begin
        if (right(tmp_str, 2) = ', ') then
          tmp_str = substring(:tmp_str from 1 for (char_length(:tmp_str) - 2));
        res_text = (select result
                    from hp_append_arr_from_json('kontmat', '{' || :res_text || '}', :tmp_str, 1));
      end
    end

    res_text = BLOB_APPEND('{', res_text, '}');
    suspend;
    exit;
  end

  if (action_kind = 'doLoad' and
      is_mp = 1) then -- загрузки первичных данных МП
  begin

    res_text = '';
    matcateg_id = 133; -- в КР все ТМЦ по Мурабаха сажаем в одну категорию

    select mc_name
    from matcateg
    where mc_id = :matcateg_id
    into matcateg_name;
    matcateg_name = replace(matcateg_name, '"', '\"');

    select k.k_approved_manager, pr.pr_code
    from kontract k
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where k.k_id = :k_id
    into :matotvet_id, :pr_code;
    res_text = BLOB_APPEND(res_text,'"matcateg_id":"' || :matcateg_id || '"');
    if (matotvet_id is null) then
      matotvet_id = 0;
    res_text = BLOB_APPEND(res_text, ',"region_id":"' || :region_id || '"');
    res_text = BLOB_APPEND(res_text, ',"matotvet_id":"' || :matotvet_id || '"');

    select wp_id, wp_name
    from workplace
    where wp_from_reg_id = :region_id and
          wp_status = 6 and
          wp_is_active = 1 and
          (wp_name like '%Собственность%' or
          wp_name containing 'Истиснаа') rows 1
    into :workplace_id, workplace_name;
    if (workplace_id is null) then
      workplace_id = 0;

    res_text = BLOB_APPEND(res_text, ',"workplace_id":"' || :workplace_id || '"');

    select al.al_id
    from acc_list al
    where al.al_from_reg_id = :region_id and
          al.al_is_active = 1 and
          al.is_delete = 0 and
          ((al.al_account = '11208/1' and
          :pr_code containing '.istisnaa.') or
          (al.al_account = '11209/1' and
          :pr_code containing '.murabaha.'))
    into :account_dt;
    if (account_dt is null) then
      account_dt = '';

    res_text = BLOB_APPEND(res_text, ',"account_dt":"' || :account_dt || '"');

    select al.al_id
    from acc_list al
    where al.al_from_reg_id = :region_id and
          al.al_is_active = 1 and
          al.is_delete = 0 and
          ((al.al_account = '11571/3' and
          :pr_code containing '.istisnaa.') or
          (al.al_account = '21199/99' and
          :pr_code containing '.murabaha.'))
    into :account_kt;

    res_text = BLOB_APPEND(res_text, ',"account_kt":"' || :account_kt || '"');

    select replace(al_name, '"', '\"')
    from acc_list
    where al_id = :account_dt
    into :account_dt_name;
    if (account_dt_name is null) then
      account_dt_name = 'Не указан';

    select replace(ag_caption, '"', '\"')
    from agreements ag
    where ag_id = :seller_id and
          ag.ag_from_rd_id in (:rd_id, :rd_id_1, :rd_id_2)
    into :seller_name;

    res_text = BLOB_APPEND(res_text, ',"seller_id":"' || :seller_id || '"');

    res_text = BLOB_APPEND('"main":{', res_text, '}');

    tmp_str = '{"id":"' || :matcateg_id || '", "name":"' || matcateg_name || '"}';
    res_text = BLOB_APPEND(res_text, ',"matcategs":[' || :tmp_str || ']');

    res_text = BLOB_APPEND(res_text, ', "kontmat":[');
    kol = 0;
    for select kml_id, ml_name, ml_model, kml_count, kml_cost, kml_total, kml_commis, kml_bill_id, kml_bill_line_no,
               kml_guarantee_month, k_kontract_no, kal_new_kf_name, k_status
        from kontmat kml
        inner join matlist ml on ml.ml_id = kml.kml_from_ml_id
        inner join kontract k on kml.kml_object_id = k.k_id and
              k.k_status >= 0
        inner join kontstate ks on ks.ks_from_k_id = kml.kml_object_id and
              ks.ks_ol_is_delete = 0 and
              current_date between ks.ks_date_begin and ks.ks_date_end and
              ks.ks_state = 0
        inner join product pr on k.k_from_pr_id = pr.pr_id and
              pr.pr_code not like '%.IDJARA.%'
        inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id and
              k.k_disb_date between kal.kal_date and kal.kal_date_end and
              kal.is_delete = 0
        where kml.kml_object_kind = 9 and
              kml.kml_status = 1 and
              kml.kml_from_ag_id = :seller_id and
              kml.kml_object_id = :k_id
        into :kml_id, ml_name, ml_model, kml_count, kml_cost, kml_total, kml_commis, kml_bill_id, kml_bill_line_no,
             kml_guarantee_month, :k_kontract_no, kf_name, k_status
    do
    begin
      if (coalesce(k_kontract_no, '') = '') then
        k_kontract_no = '-';
      if (coalesce(kf_name, '') = '') then
        kf_name = '-';

      is_selected = 0;

      if (kml_guarantee_month is null) then
        kml_guarantee_month = 0;
      is_buyed = 0;
      if (kml_bill_id is not null) then
      begin
        if (exists(select *
                   from opermat, operlist
                   where om_from_ol_id = :kml_bill_id and
                         om_number = :kml_bill_line_no and
                         om_from_ol_id = ol_id and
                         operlist.is_delete = 0)) then
          is_buyed = 1;
      end
      if (is_buyed = 0) then
      begin
        kol = kol + 1;
        if (k_status >= 0 and
            kml_cost <> 0 and
            kml_count <> 0) then
          --S.K.
          --is_selected = 1;
          tmp_str = '{' || f_json_make_pair('selected_kml_id', kml_id, 'str') || '}';
        if (kol > 1) then
          res_text = BLOB_APPEND(res_text, ', ');
        res_text = BLOB_APPEND(res_text, tmp_str);
      end
    end
    res_text = BLOB_APPEND(res_text, ']');
    res_text = BLOB_APPEND('{', res_text, '}');
    suspend;
    exit;
  end

  if (action_kind = 'doSave') then -- Сохранение данных
  begin
    res_id = 1;
    res_text = '';

    is_buyed = 0;
    line_no = 0;

    if (res_text = '' and
        coalesce(matcateg_id, 0) = 0) then
      res_text = 'Категория ТМЦ не указана';

    if (res_text = '' and
        coalesce(workplace_id, 0) = 0) then
      res_text = 'Место для прихода не указано';

    if (res_text = '' and
        coalesce(matotvet_id, 0) = 0) then
      res_text = 'Материально-ответственный сотрудник не указан';

    if (res_text = '' and
        coalesce(seller_id, 0) = 0 and
        method > 0) then
      res_text = 'Продавец товара(ов) не указан. Нужно указать в окне Договора, во вкладке "Поручители".';

    if (res_text = '' and
        coalesce(region_id, 0) = 0) then
      res_text = 'Офис не указан';

    if (res_text = '' and
        coalesce(account_dt, '') = '') then
      res_text = 'Дебет счет не указан';

    if (res_text = '' and
        coalesce(account_kt, '') = '') then
      res_text = 'Кредит счет не указан';

    if (is_mp = 1 and
        oper_date is null) then
      oper_date = current_date;

    if (res_text = '' and
        oper_date is null) then
      res_text = 'Дата покупки не указана';

    if (res_text <> '') then
    begin
      res_id = -1;
      suspend;
      exit;
    end

    line_no = 0;
    is_buyed = 0;
    error_in_date = 0;

    for select json_key, json_value
        from fm_json_parse(:kontmat_data, 0, 'kontmat')
        into :json_key, :json_value
    do
    begin
      if (json_key = 'selected_kml_id') then
        selected_kml_id = json_value;
      if (json_key = 'endofobject') then
      begin
        for select kml_bill_id, kml_bill_line_no, k_disb_date
            from kontmat, kontract
            where kml_id = :selected_kml_id and
                  kml_object_id = k_id
            into :kml_bill_id, :kml_bill_line_no, k_disb_date
        do
        begin
          line_no = line_no + 1;
          if (kml_bill_id is not null) then
          begin
            if (exists(select *
                       from opermat, operlist
                       where om_from_ol_id = :kml_bill_id and
                             om_number = :kml_bill_line_no and
                             om_from_ol_id = ol_id and
                             operlist.is_delete = 0)) then
              is_buyed = is_buyed + 1;
          end
          if (k_disb_date < oper_date) then
            error_in_date = 1;
        end
        if (is_buyed > 0) then
        begin
          res_text = 'В списке ' || is_buyed || ' из ' || line_no || ' товаров ранее оприходованы на склад.';
          res_id = -1;
          suspend;
          exit;
        end

        if (error_in_date = 1) then
        begin
          res_id = -1;
          res_text = 'Покупка должна быть до даты выдачи. Исправьте дату.';
          suspend;
          exit;
        end

        -- Все условия проверены, можно сохранять
        curr_region_id = f_hq_id();

        select kv
        from get_kv(:curr_region_id, 5)
        into :ol_id;

        ol_doc_kind = 9; -- Мем.ордер
        if (ag_agreement_no <> '' and
            ag_agreement_date is not null) then
          ol_memo = 'Покупка товаров по ИПФ согласно договора № ' || ag_agreement_no || ' от ' || coalesce(ag_agreement_date, current_date);
        else
          ol_memo = 'Покупка товаров по ИПФ';

        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_user_id,
                              ol_document_client, ol_oper_subtype, ol_is_client)
        values (:ol_id, :region_id, 3, :oper_date, :ol_doc_kind, :ol_memo, :un_id, :client_id, 9, 1)
        returning ol_id
        into :ol_id;
      end
    end

    kml_id_lst = '';
    for select json_key, json_value
        from fm_json_parse(:kontmat_data, 0, 'kontmat')
        into :json_key, :json_value
    do
    begin
      if (json_key = 'selected_kml_id') then
        selected_kml_id = json_value;
      if (json_key = 'endofobject') then
      begin
        if (:method = 0 and selected_kml_id is not null) then
        begin
          select ag.ag_object_id
          from kontmat kml
          inner join agreements ag on kml.kml_from_ag_id = ag.ag_id and
                ag.ag_object_kind = 0
          where kml.kml_id = :selected_kml_id
          into :client_id;
        end
        if (kml_id_lst containing selected_kml_id || ';') then
          continue;
        else
          kml_id_lst = kml_id_lst || selected_kml_id || ';';
        for select kml_id, kml_from_ml_id, kml_cost, kml_count, kml_total, kml_commis,
                   ml_name || iif(coalesce(ml_model, '') <> '', ' (' || ml_model || ')', ''), kml_bill_id,
                   kml_bill_line_no, k_kontract_no, k_disb_date
            from kontmat, matlist, kontract
            where kml_id = :selected_kml_id and
                  kml_from_ml_id = ml_id and
                  kml_object_id = k_id
            into :kml_id, :ml_id, :kml_cost, :kml_count, :kml_total, :kml_commis, ml_name, :kml_bill_id, :kml_bill_line_no,
                 k_kontract_no, k_disb_date
        do
        begin
          insert into opermat (om_from_ol_id, om_oper_type, om_workplace_id, om_matotvet_id, om_summa_plus,
                               om_summa_minus, om_kolich_plus, om_kolich_minus, om_nds_percent, om_account,
                               om_is_written_off, om_account_info)
          values (:ol_id, 1, :workplace_id, :matotvet_id, :kml_total, 0, :kml_count, 0, 0, :account_dt, 0, :account_kt)
          returning (om_number)
          into :line_no;

          update kontmat
          set kml_bill_id = :ol_id,
              kml_bill_line_no = :line_no
          where kml_id = :kml_id;

          insert into operbill (ob_bill_id, ob_bill_line_no, ob_from_ml_id, ob_from_reg_id, ob_from_mc_id,
                                ob_date_begin, ob_from_ol_id, ob_invent_no, ob_invent_str, ob_full_detail)
          values (:ol_id, :line_no, :ml_id, :region_id, :matcateg_id, :oper_date, :ol_id, 0, '', '');

          insert into operbook (ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                                ob_summa, ob_account_kt, ob_client_kt, ob_detail)
          values (:ol_id, :ob_currency, 1, null, :account_dt, :kml_total, :kml_total, :account_kt, :client_id,
                  'Покупка по ИПФ: ' || :ml_name || ', дог.финансирования ' || :k_kontract_no || ' от ' || :k_disb_date);

          -- RFCKG-5488
          if (kml_commis > 0) then
          begin
            acc_dt = (select al_id from find_acc_list('21199/99') where al_from_reg_id = :region_id);
            acc_kt = (select al_id from find_acc_list('21199/299') where al_from_reg_id = :region_id);
            ob_detail = trim(f_format('Обязательства к оплате контрагенту/поставщику %1 по договору № %2', fio, k_kontract_no));
            insert into operbook (
              ob_from_ol_id, ob_currency, ob_curr_rate, ob_account_dt,
              ob_summa_val, ob_summa, ob_account_kt, ob_client_kt, ob_detail)
            values (
              :ol_id, '417', 1, :acc_dt,
              :kml_total, :kml_total, :acc_kt, :client_id, :ob_detail);

            acc_dt = (select al_id from find_acc_list('21199/299') where al_from_reg_id = :region_id);
            acc_kt = (select al_id from find_acc_list('21114/299') where al_from_reg_id = :region_id);
            ob_detail = trim(f_format('Услуги по ГП договорам ИПФ %1 по договору № %2', fio, k_kontract_no));
            insert into operbook (
              ob_from_ol_id, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
              ob_summa_val, ob_summa, ob_account_kt, ob_detail)
            values (
              :ol_id, '417', 1, :client_id, :acc_dt,
              :kml_commis, :kml_commis, :acc_kt, :ob_detail);
          end
        end
      end
    end

    res_text = '{"results":{"ol_id":"' || :ol_id || '"}}';
    suspend;
    exit;
  end
end
