CREATE PROCEDURE CM_KG_0006_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable k_kontract_no varchar(20);
declare variable kf_id id_type;
declare variable pr_code d_str_100;
begin
--          Проверка, не допускает к полученю второго параллельного кредита
    for
    select k_from_kf_id, k_disb_date, pr_code
    from kontract kont, product
    where k_id = :k_id
    and k_from_pr_id = pr_id
    and pr_product_kind = 2
    and pr_code not containing '.IPF.'
    and pr_code containing '.PAR.'
    and coalesce(k_refund,0) = 0
    into kf_id, kf_disb_date, pr_code
    do
    begin
      if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
        exit;
      for
      select kc_from_c_id from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into :c_id
      do
      begin
--              Запрос который вытаскивает активный параллельный кредит
        k_kontract_no = null;
        select k_kontract_no from kontstate, knclient, kontract, product
        where ks_ol_is_delete = 0
        and ks_state = 1
        and :kf_disb_date between ks_date_begin and ks_date_end
        and ks_from_k_id = kc_from_k_id
        and kc_from_k_id <> :k_id
        and kc_amount_approved > 0
        and kc_from_k_id = k_id
        and k_from_pr_id = pr_id
        and pr_product_kind = 2
        and k_from_kf_id = :kf_id
        and pr_code not containing '.IPF.'
        and pr_code containing '.PAR.'
        rows 1
        into :k_kontract_no;
        if (k_kontract_no is not null) then
        begin
          select c_name_fio, c_pass_no from clients
          where c_id = :c_id
          into res_info_text, kf_client_pass_no;
          res_error_is_critical = 1;
          res_error_text = 'У клиента имеется активный параллельный кредит в контракте №'||:k_kontract_no;
          suspend;
        end
      end
    end
end
