CREATE PROCEDURE HP_DO_KN_OKSUMMARY_DATE_RECALC(
    REP_DATE D_DATE)
RETURNS (
    KC_ID D_ID,
    ERR_TEXT D_STR_255)
AS
declare variable min_date d_date;
begin
  -- Процедура производит пересчет по всем займам после указанной даты.
  -- Применяется после обновления ставок резервов или после изменения алгоритма в триггерах

  -- Формула для Excel:
  -- ="execute procedure hp_do_kn_oksummary_all_recalc('"&ТЕКСТ(A1;"ДД.ММ.ГГГГ")&"', '"&ТЕКСТ(A1;"ДД.ММ.ГГГГ")&"'); commit;"
  select count(distinct oks_from_kc_id)
  from kn_oksummary
  where oks_date_begin >= :rep_date
  into kc_id;

  kc_id = -kc_id;
  suspend;

  for select oks_from_kc_id, min(oks_date_begin)
  from kn_oksummary
  where oks_date_begin >= :rep_date
  group by oks_from_kc_id
  into kc_id, min_date
  do
  begin
    begin
      err_text = '';
      in autonomous transaction do
        update kn_oksummary set oks_need_recalc = 1 where oks_from_kc_id = :kc_id and oks_date_begin = :min_date;

      when any do
      begin
        err_text = f_format('update kn_oksummary set oks_need_recalc = 1 where oks_from_kc_id = %1 and oks_date_begin = ''%2''; commit;', kc_id, min_date);
        suspend;
      end
    end
  end

  err_text = 'Finish';
  suspend;
end
