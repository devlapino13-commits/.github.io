CREATE PROCEDURE AAREP_MONEY_SALDO_LIST_DTMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    RES_REGION_NAME VARCHAR(100),
    RES_ACCOUNT VARCHAR(20),
    RES_ACCOUNT_NAME VARCHAR(200),
    RES_CURRENCY VARCHAR(3),
    RES_SALDO DECIMAL(18,4),
    RES_SALDO_EKV DECIMAL(18,4),
    RES_ACC_KIND INTEGER,
    RES_REGION_ID INTEGER,
    AL_ID VARCHAR(20))
AS
declare variable atl_acc_mask varchar(20);
begin
-- Sanjari, 19.07.2012
  for select atl_from_at_id, atl_acc_mask
      from acctypelist
      where (atl_from_at_id = 11) or
            (atl_from_at_id = 12)
      into :res_acc_kind, :atl_acc_mask
  do
  begin
    for select distinct al_id, reg_code || ' ' || reg_name, al_account, al_name, al_currency, reg_unicode
        from acc_list, region_list(:report_region_id, 1, -1, 1), region
        where al_from_bg_id = :atl_acc_mask and
              (al_from_reg_id = id_list) and
              reg_unicode = al_from_reg_id
        into :al_id, :res_region_name, :res_account, :res_account_name, :res_currency, :res_region_id
    do
    begin
      res_saldo_ekv = f_al_bal(al_id, report_date_end);
      res_saldo = f_al_bal(al_id, report_date_end, 'val');

      if ((res_saldo <> 0) or (res_saldo_ekv <> 0)) then
        suspend;
    end
  end
end
