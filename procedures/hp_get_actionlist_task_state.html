CREATE PROCEDURE HP_GET_ACTIONLIST_TASK_STATE(
    RD_CODE D_STR_255,
    OBJECT_KIND D_INTEGER,
    OBJECT_ID D_ID)
RETURNS (
    IS_QUERY_ALREADY_SENDED_TODAY D_BOOL,
    ACL_STATUS D_INTEGER,
    OBM_MEMO D_BLOB_TEXT)
AS
declare variable RD_ID D_INTEGER;
begin
  is_query_already_sended_today = 0;
  rd_id = f_get_rd_id(:rd_code);

  -- если запрос уже был сегодня отправлен
  if (exists (select *
              from actionlist acl
              where acl.acl_from_rd_id = :rd_id
                and acl.acl_date = current_date
                and acl.acl_object_kind = :object_kind
                and acl.acl_object_id = :object_id)) then
     is_query_already_sended_today = 1;

  -- статус последнего запроса
  select acl.acl_status
  from actionlist acl
  where acl.acl_object_kind = :object_kind
    and acl.acl_object_id = :object_id
    and acl.acl_from_rd_id = :rd_id
  order by acl.acl_last_datetime desc
  rows 1
  into acl_status;

  -- коментарий последнего запроса со статусом "На доработку"
  select obm.obm_memo
  from actionlist acl
  inner join objectprops opr on opr.opr_object_id = acl.acl_object_id
  inner join objectmemo obm on obm.obm_from_opr_id = opr.opr_id
  where opr.opr_subject_id = acl.acl_from_rd_id
    and acl.acl_from_rd_id = :rd_id
    and opr.opr_object_kind = acl.acl_object_kind
    and opr.opr_object_id = :object_id
    and acl.acl_status = -1
  order by obm.obm_datetime desc
  rows 1
  into obm_memo;

  obm_memo = coalesce(obm_memo, '');
  suspend;
end
