CREATE PROCEDURE HP_CHECK_KONTRACT_AZ(
    REP_K_ID DECIMAL(18,0),
    REP_DATE_END DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(20),
    CHECK_MEMO VARCHAR(255),
    OK_BALANCE CURR_TYPE,
    OB_BALANCE CURR_TYPE,
    OK_OL_DATE DATE,
    OK_FROM_OL_ID ID_TYPE,
    LINK VARCHAR(2048))
AS
declare variable OK_TYPE integer;
declare variable OK_OS CURR_TYPE;
declare variable OK_PERC CURR_TYPE;
declare variable OK_FINE CURR_TYPE;
declare variable OK_KOMMIS CURR_TYPE;
declare variable OK_SH_OS CURR_TYPE;
declare variable ACC_BALANCE varchar(20);
declare variable ACC_PERC_DOHOD varchar(20);
declare variable ACC_PERCENT varchar(20);
declare variable ACC_FINE varchar(20);
declare variable ACC_KOMMIS varchar(20);
declare variable ACC_PEREPLATA varchar(20);
declare variable ACC_LLC_OS varchar(20);
declare variable ACC_LLC_PERCENT varchar(20);
declare variable ACC_KOMMIS_DOSROCH varchar(20);
declare variable OB_SUMMA_VAL CURR_TYPE;
declare variable AT_ID integer;
declare variable PAL_MAIN_ACC varchar(20);
declare variable PAL_DELINQ_ACC varchar(20);
declare variable AT_NAME varchar(50);
declare variable OK_TYPE_2 decimal(18,4);
declare variable OK_TYPE_3 decimal(18,4);
declare variable SH_KOMMIS CURR_TYPE;
declare variable OK_KOMMIS_TOTAL CURR_TYPE;
declare variable K_DISB_DATE date;
declare variable OK_KOMMIS_ADV CURR_TYPE;
declare variable K_KOLICH_CLIENT integer;
declare variable KC_KOLICH_CLIENT integer;
declare variable KD_K_NEW decimal(18,0);
declare variable K_KONTRACT_NO_BUF varchar(50);
declare variable OK_BALANCE_BUF decimal(18,4);
declare variable KS_STATE_BUF integer;
declare variable OK_SUM_BUF decimal(18,4);
declare variable OK_TYPE_BUF integer;
declare variable SH_OS decimal(18,4);
declare variable K_AMOUNT_APPROVED decimal(18,4);
declare variable K_PERCENT_FOR_DAYS integer;
declare variable PR_PERCENT_FOR_DAYS integer;
declare variable P_SH_ID integer;
declare variable P_SH_DATE date;
declare variable P_SH_OS CURR_TYPE;
declare variable P_SH_PERC CURR_TYPE;
declare variable OK_SH_PERC CURR_TYPE;
declare variable ACC_KOMMIS_OTLOJ varchar(20);
begin

  -- ориентироватся по таблице ACCTYPE, PRODACCTYPE
  -- Счета по новому стандарту
  for select at_id, pal_main_acc || '/' || pal_main_acc_no, at_name, k_kontract_no, k_disb_date,
             k_kolich_client_approved, pal_delinq_acc || '/' || pal_delinq_acc_no
      from kontract, product, prodacclist, acctype
      where k_id = :rep_k_id and
            k_from_pr_id = pr_id and
            pr_from_pat_id = pal_from_pat_id and
            pal_from_at_id = at_id and
            at_kind = 1 and -- выборка только по кредитным счетам
            pal_main_acc is not null
      into at_id, pal_main_acc, at_name, k_kontract_no, k_disb_date, k_kolich_client, pal_delinq_acc
  do
  begin
    if (at_id = 1) then
      acc_balance = :pal_main_acc; -- Портфель
    if (at_id = 2) then
      acc_perc_dohod = :pal_main_acc; -- Начисленные проценты
    if (at_id = 3) then
      acc_percent = :pal_main_acc; -- Полученные проценты
    if (at_id = 5) then
      acc_fine = :pal_main_acc; -- Полученные штрафы
    if (at_id = 7) then
      acc_kommis = :pal_main_acc; -- Полученные комиссионные (ДБП)
    if (at_id = 8) then
      acc_pereplata = :pal_main_acc; -- Счета переплаты по ОС
    if (at_id = 10) then
      acc_llc_os = :pal_main_acc; -- Счета РППУ для списание ОС
    if (at_id = 15) then
      acc_llc_percent = :pal_main_acc; -- Счета РППУ для списание Процентов
    if (at_id = 26) then
      acc_kommis_dosroch = :pal_main_acc; -- Комиссия за досрочное погашениеС
    if (at_id = 43) then
      acc_kommis_otloj = :pal_main_acc; --  Отложенный доп.процент
  end

  ok_type_2 = 0;
  ok_type_3 = 0;
  ok_balance = 0;
  ok_kommis_total = 0;
  for with tabl1
      as (select kd_k_new
          from kontdevides
          where kd_k_orig = :rep_k_id
          union
          select :rep_k_id
          from rdb$database)
      select ok_from_ol_id, ok_type, ok_ol_date, sum(ok_os), sum(ok_perc), sum(ok_fine), sum(ok_kommis), sum(ok_sh_os),
             sum(ok_sh_perc), sum(ok_kommis_adv)
      from operkred ok, tabl1, operlist ol
      where ok_ol_is_active = 1 and
            ok_from_c_id = 0 and
            ok_type in (2, 3, 4, 5, 8, 9) and
            ok_from_ol_id not in (1010100002949117,1010100002924236,1010100002921198,1010100002919320,1010100002913368,1010100002890376,1010100002888025,1010100002864863,1010100002845496,
            1010100002838380,1010100002794884,1010100002704568,1010100002699288,1010100002688792,1010100002675423,1010100002670638,1010100002670658,1010100002668264,1010100002659927,
            1010100002645059,1010100002609474,1010100002607108,1010100002603099,1010100002600281,1010100002600706,1010100002597151,1010100002596882,1010100002581775,1010100002570420,
            1010100002560995,1010100002548243,1010100002547069,1010100002521780,1010100002491133,1010100002453579,1010100002451970,1010100002434055,1010100002431825,1010100002426063,
            1010100002427930,1010100002418535,1010100002416737,1010100002412773,1010100002408182,1010100002390706,1010100002377918,1010100002376729,1010100002375887,1010100002374763,
            1010100002366564,1010100002366538,1010100002358960,1010100002352370,1010100002331507,1010100002327790,1010100002323883,1010100002319497,1010100002309902,1010100002304302,
            1010100002301196,1010100002300674,1010100002298952,1010100002297255,1010100002296654,1010100002289650,1010100002286651,1010100002274025,1010100002251776,1010100002244199,
            1010100002240023,1010100002233593,1010100002215290,1010100002207782,1010100002184859,1010100002165563,1010100002154251,1010100002131225,1010100002131129,1010100002115674,
            1010100002110691,1010100002085536,1010100002079495,1010100002073466,1010100002008222,1010100002039341 ,1010100003520321,1010100003627483) and
            ok_from_k_id = kd_k_new and
            ol.ol_id = ok.ok_from_ol_id
      group by ok_from_ol_id, ok_type, ok_ol_date
      order by ok_ol_date
      into ok_from_ol_id, ok_type, ok_ol_date, ok_os, ok_perc, ok_fine, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv
  do
  begin
    ok_fine = ok_fine + ok_kommis_adv;
    if (ok_type = 2) then
      ok_type_2 = ok_type_2 + ok_sh_os;
    if (ok_type = 3) then
      ok_type_3 = ok_type_3 + ok_sh_os;
    if (ok_type in (4, 5, 8)) then
      ok_balance = ok_balance + ok_os;
    ok_kommis_total = ok_kommis_total + ok_kommis;

    if (ok_type = 4 and
        ok_os > 0) then
    begin
      check_memo = 'Сумма погашения по ОС больше 0. operkred.ok_os = ' ||(ok_os);
      link = 'http://mbrain.molbulak.kg/ArticleFrm.aspx?ART_ID=725';
      suspend;
    end
    if (ok_type = 4 and
        ok_perc > 0) then
    begin
      check_memo = 'Сумма погашения по %% больше 0. operkred.ok_perc = ' ||(ok_perc);
      link = 'http://mbrain.molbulak.kg/ArticleFrm.aspx?ART_ID=727';
      suspend;
    end
    if (ok_type = 4 and
        ok_sh_os > 0) then
    begin
      check_memo = 'Сумма погашения по ОС больше 0. operkred.ok_sh_os = ' ||(ok_os);
      suspend;
    end
    if (ok_type = 4 and
        ok_sh_perc > 0) then
    begin
      check_memo = 'Сумма погашения по %% больше 0. operkred.ok_sh_perc = ' ||(ok_perc);
      suspend;
    end
    --   Проверка выдачи
    if (ok_type = 8) then
    begin
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
      ob_short_acc_dt in ('21110/1', '21210/1', :acc_balance)  -- '185-1/1', '180-1/1' на 18 начинаются они уже не используются
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;

      ob_balance = ob_summa_val;
      if (ok_os <> ob_summa_val) then
      begin
        check_memo = 'Выданная сумма не совпадает на: ' ||(ok_os - ob_summa_val);
        suspend;
      end
    end

    if (ok_type = 4) then
    begin
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            ob_short_acc_kt in (:acc_kommis, :acc_kommis_otloj)
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_kommis <> ob_summa_val) then
      begin
        check_memo = 'Разница по полученному доп проценту: ' ||(ok_kommis + ob_summa_val);
        suspend;
      end

      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook ob
      where (ob.ob_short_acc_dt in ('21110/1', '21210/1', '21140/1', '21240/1', '21113/1', '21213/1', '21143/1', '21243/1',
                                    '23220/1') or
            ob.ob_short_acc_kt in ('21110/1', '21210/1', '21140/1', '21240/1', '21113/1', '21213/1', '21143/1', '21243/1',
                                   '23220/1')) and
                                  -- '185-1/1', '180-1/1', --Вот те которые на 18 начинаются они уже не используются
            ob_from_ol_id = :ok_from_ol_id into ob_summa_val;

      if (ob_summa_val is null) then
        ob_summa_val = 0;
      ob_balance = ob_balance - ob_summa_val;
      if (-ok_os <> ob_summa_val) then
      begin

        check_memo = 'Разница по полученному ОС: ' ||(ok_os + ob_summa_val);
        suspend;
      end

      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            ob_short_acc_kt = :acc_percent
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_perc <> ob_summa_val) then
      begin
        check_memo = 'Разница по выплаченному проценту: ' ||(ok_perc + ob_summa_val);
        suspend;
      end

      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            (ob_short_acc_dt = :acc_fine or
            ob_short_acc_kt = :acc_fine)
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_fine <> ob_summa_val) then
      begin
        check_memo = 'Разница по выплаченной пени: ' ||(ok_fine + ob_summa_val);
        suspend;
      end
    end
    --   Переплата кредита
    if (ok_type = 9) then
    begin
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            (ob_short_acc_dt = :acc_pereplata or
            ob_short_acc_kt = :acc_pereplata)

      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_os <> ob_summa_val) then
      begin
        check_memo = 'Разница по переплате ОС: ' ||(ok_os + ob_summa_val);
        suspend;
      end
    end
    --   Списание
    if (ok_type = 5) then
    begin
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            (ob_short_acc_kt = :acc_balance or --'185-1/1', '180-1/1',
            ob_short_acc_kt in ( '21110/1', '21210/1', '21140/1', '21240/1', '21113/1', '21213/1', '21143/1', '21243/1',
                                '23220/1')) and
            ob_short_acc_dt = :acc_llc_os

      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      ob_balance = ob_balance - ob_summa_val;
      if (-ok_os <> ob_summa_val) then
      begin
        check_memo = 'Разница по списанному ОС: ' ||(ok_os + ob_summa_val);
        suspend;
      end
      ob_summa_val = null;
      select sum(ob_summa_val)
      from operbook
      where ob_from_ol_id = :ok_from_ol_id and
            ob_short_acc_kt = :acc_perc_dohod and
            ob_short_acc_dt in (:acc_llc_percent, '11002/2')
      into ob_summa_val;
      if (ob_summa_val is null) then
        ob_summa_val = 0;
      if (-ok_perc <> ob_summa_val) then
      begin
        check_memo = 'Разница по списанному Проценту: ' ||(ok_perc + ob_summa_val);
        suspend;
      end
    end
  end

  if (ok_balance < 0) then
  begin
    check_memo = 'Баланс контракта < 0: ' ||(ok_balance);
    suspend;
  end

  /* Добавил igusev 23.09.2014, вместо кода который идет выше

    Если выявлено, что суммарный баланс включая амнистированные контракты <> 0,
      прохожусь по контрактам связанным с rep_k_id, проверяю их баланс и статус
   */
  if (ok_balance <> 0) then
  begin
    for select kd_k_new, kont.k_kontract_no, ks.ks_state
        from (select kd_k_new
              from kontdevides kd
              where kd_k_orig = :rep_k_id
              union
              select :rep_k_id
              from rdb$database) a
        inner join kontract kont on kont.k_id = a.kd_k_new
        inner join kontstate ks on ks.ks_from_k_id = kont.k_id
        where :rep_date_end between ks.ks_date_begin and ks.ks_date_end
        into :kd_k_new, :k_kontract_no_buf, :ks_state_buf
    do
    begin
      ok_balance_buf = 0;

      for select sum(ok.ok_sh_os), ok_type
          from operkred ok
          where ok.ok_from_k_id = :kd_k_new and
                ok.ok_type in (2, 3) and
                ok.ok_ol_is_active = 1 and
                ok_from_c_id = 0
          group by ok_type
          into :ok_sum_buf, :ok_type_buf
      do
      begin
        if (coalesce(ok_sum_buf, 0) <> 0) then
        begin
          check_memo = k_kontract_no_buf || '; ' || 'У контракта ok_type = ' || ok_type_buf || ' <> 0: ' ||(ok_sum_buf);
          suspend;
        end
      end

      select sum(ok.ok_os)
      from operkred ok
      where ok.ok_from_k_id = :kd_k_new and
            ok.ok_type in (4, 5, 8) and
            ok.ok_ol_is_active = 1 and
            ok_from_c_id = 0
      into :ok_balance_buf;

      if ((ks_state_buf = 2) and
          (ok_balance_buf <> 0)) then
      begin
        check_memo = k_kontract_no_buf || '; ' || 'У закрытого контракта Баланс <> 0: ' ||(ok_balance_buf);
        suspend;
      end

      if (ks_state_buf = 1) then
      begin
        for select ok_sh_id, ok.ok_sh_date, sum(ok.ok_os + ok_sh_os) ok_os, sum(ok_perc + ok_sh_perc) sh_perc
            from operkred ok
            where ok.ok_from_k_id = :kd_k_new and
                  ok.ok_type in (1, 2, 3, 4, 5, 6, 9) and
                  ok.ok_ol_is_active = 1 and
                  ok_from_c_id = 0 and
                  ok.ok_sh_id <> 0
            group by ok.ok_sh_id, ok.ok_sh_date
            having sum(ok.ok_os + ok_sh_os) < 0 or (sum(ok_perc + ok_sh_perc) < 0)
            into p_sh_id, p_sh_date, p_sh_os, p_sh_perc
        do
        begin
          check_memo = k_kontract_no_buf || '; ' || 'Планируемое сумма погашении меньше 0(нуля) ' || p_sh_id || '; ' || p_sh_date || '; ' || p_sh_os || '; ' || p_sh_perc;
          link = 'http://mbrain.molbulak.kg/ArticleFrm.aspx?ART_ID=734';
          suspend;
        end
      end
    end
  end

  if (ok_balance <> ob_balance) then
  begin
    check_memo = 'Баланс контракта не идет с обороткой, Разница: ' ||(ob_balance - ok_balance);
    suspend;
  end
  if (ok_type_2 <> 0) then
  begin
    check_memo = 'Сумма реструктуризации по ОС <> 0, ok_type = 2: ' || ok_type_2;
    suspend;
  end
  if (ok_type_3 <> 0) then
  begin
    check_memo = 'Сумма автореструктуризации по ОС <> 0, ok_type = 3: ' || ok_type_3;
    suspend;
  end
  -- Проверка комиссии с графиком
  -- ok_from_c_id=0 добавлен для пары

  sh_kommis = null;
  with tabl1
  as (select kd_k_new
      from kontdevides
      where kd_k_orig = :rep_k_id
      union
      select :rep_k_id
      from rdb$database)
  select sum(sh_kommis)
  from schedule, tabl1
  where sh_from_k_id = kd_k_new
  into sh_kommis;
  if (sh_kommis is null) then
    sh_kommis = 0;
  if (sh_kommis <> -ok_kommis_total and
      coalesce(acc_kommis_otloj, '') = '') then
  begin
    check_memo = 'Расхождения в сумме комиссии, SCHEDULE <> OPERKRED Разница: ' ||(sh_kommis + ok_kommis_total);
    suspend;
  end

  kc_kolich_client = null;
  select count(*)
  from knclient
  where kc_from_k_id = :rep_k_id and
        kc_amount_approved > 0
  into kc_kolich_client;
  if (kc_kolich_client is null) then
    kc_kolich_client = 0;

  if (k_kolich_client <> kc_kolich_client) then
  begin
    check_memo = 'Разница в кол-ве клиентов, KONTRACT <> KNCLIENT';
    suspend;
  end

  --Проверка на соответствие суммы выдачи в k_amount_approved и sh_os
  select sum(sh_os) sh_os
  from schedule sch
  where sch.sh_from_k_id = :rep_k_id
  into sh_os;

  if (sh_os is null) then
    sh_os = 0;

  select k_amount_approved, coalesce(kont.k_percent_for_days, -777), coalesce(pr.pr_percent_for_days, -777)
  from kontract kont
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where k_id = :rep_k_id
  into k_amount_approved, k_percent_for_days, pr_percent_for_days;

  if (sh_os <> k_amount_approved) then
  begin
    check_memo = 'Разница в утвержденной сумме, sh_os <> k_amount_approved';
    suspend;
  end

  if (k_percent_for_days <> pr_percent_for_days) then
  begin
    check_memo = 'Неправильное значение ежедневной ставки. k_percent_for_days = ' || k_percent_for_days || ', pr_percent_for_days=' || pr_percent_for_days;
    suspend;
  end
end
