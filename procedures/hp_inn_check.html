CREATE PROCEDURE HP_INN_CHECK(
    INN STR_100,
    BIRTH_DATE DATE,
    C_SEX SMALLINT,
    C_ID ID_TYPE = null)
RETURNS (
    ERROR_TEXT STR_255,
    FIO STR_100)
AS
declare variable gender smallint;
declare variable birth_date_str str_100;
begin
  inn = trim(coalesce(inn, ''));
  error_text = '';

  if (c_id is not null) then
  begin
    select trim(coalesce(cl.c_inn, '')), c_sex, cl.c_birth_date, cl.c_name_fio
    from clients cl
    where cl.c_id = :c_id
    into :inn, :c_sex, :birth_date, :fio;
  end

  if ((c_id is null) and
      ((inn = '') and
      (birth_date is null) and
      (c_sex is null))) then
  begin
    error_text = 'Проблема с набором входящих параметров';
    suspend;
    exit;
  end

  if (inn = '') then
  begin
    error_text = 'ИНН не указан; ';
    suspend;
    exit;
  end
  else
  if (birth_date is null) then
  begin
    error_text = 'Дата рождения не указана; ';
    suspend;
    exit;
  end
  else
  if (c_sex is null) then
  begin
    error_text = 'Пол не указан; ';
    suspend;
    exit;
  end
  begin
    birth_date_str = f_datetostr(birth_date,'ddmmyyyy');
    gender = c_sex + 1;

    if (char_length(inn) <> 14) then
      error_text = error_text || 'Длина ИНН должна быть = 14 символов; ';
    else
    begin
      if (substring(inn from 2 for 8) <> birth_date_str) then
        error_text = error_text || 'Дата рождения не соотвествует ИНН; ';
    end

    if (left(inn, 1) <> gender) then
      error_text = error_text || 'Пол не соответствует ИНН; ';
  end

  suspend;
end
