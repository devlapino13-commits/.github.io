CREATE PROCEDURE HP_SEND_ASP_FOR_EMPLOYEE(
    C_ID D_ID,
    UN_ID D_ID,
    SIGNER_C_ID D_ID,
    LANG D_STR_10 = null)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable edoc_kind d_str_code;
declare variable obj_kind d_str_code;
declare variable edc_id d_integer;
declare variable object_kind d_integer;
declare variable signed_date d_date;
declare variable has_access d_bool;
declare variable employee_region d_integer;
begin
  edl_id = 0;
  res_text = '';

  if (lang = 'defbln') then
    lang = (select c_blank_lang from clients where c_id = :c_id);

  if (signer_c_id is null or signer_c_id = 0) then
  begin
    res_text = 'Необходимо выбрать подписанта.';
    suspend;
    exit;
  end

  if (res_text = '' and
      exists(select *
             from hr_pstatus ps
             where ps.ps_from_c_id = :c_id and
                   ps.ps_is_working = 1 and
                   ps.ps_status = 1 and
                   current_date between ps.ps_date_begin and ps.ps_date_end)) then
  begin
    select ps.ps_from_reg_id
    from hr_pstatus ps
    where ps.ps_from_c_id = :c_id and
          ps.ps_is_working = 1 and
          ps.ps_status = 1 and
          current_date between ps.ps_date_begin and ps.ps_date_end
    into :employee_region;

    select iif(id_list is not null, 1, 0)
    from permissions pm
    inner join region_list(pm.pm_subject_id, 1, 3, 1) on id_list = :employee_region
    where current_date between pm.pm_date_begin and pm.pm_date_end and
          pm.pm_object_id = :un_id and
          pm.pm_is_delete = 0 and
          pm.pm_subject_kind = 4
    into :has_access;

    if (has_access is null or has_access = 0) then
    begin
      res_text = 'Отправить код можно только сотрудникам ввереного региона';
      suspend;
      exit;
    end
  end

  obj_kind = f_objkind('clients', 'id');

  edoc_kind = 'asp.agreement';
  edc_id = f_edoc_kind('asp.agreement', 'id');

  select edc_object_kind
  from edoc_cfg
  where edc_id = :edc_id
  into :object_kind;

  select cast(eds.eds_status_datetime as date), edl.edl_id
  from edoc_list edl
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id and
        eds.eds_status = 1
  where edl.edl_object_kind = :object_kind and
        edl.edl_object_id = :c_id and
        edl.edl_from_edc_id = :edc_id and
        current_date between edl.edl_blank_date and coalesce(edl.edl_date_end, current_date) and
        edl.edl_status = 1
  order by edl.edl_status desc, edl.edl_id desc
  rows 1
  into :signed_date, :edl_id;

  /*
    11.01.2022 был создан новый договор ЭЦП, единый для клиентов, внештатников и продавцов.
    Если договор был подписан до этой даты, то аннулирую старый и отправляю на подписание новый,
    иначе переподписывать ЭЦП не нужно.
    */

  /*
    С 20.05.2022 ЭЦП общий для всех.*/
  if (:signed_date < '20.05.2022') then
    update edoc_list
    set edl_date_end = (current_date - 1)
    where edl_id = :edl_id;

  select edl_id, res_text
  from hp_edoc_insert(:edoc_kind, :obj_kind, :c_id, :un_id, :lang)
  into :edl_id, :res_text;

  if (:edl_id is null) then
    edl_id = 0;

  if (res_text = '' and
      edl_id > 0) then
    update edoc_list
    set edl_manager_c_id = :signer_c_id
    where edl_id = :edl_id;

  suspend;

end
