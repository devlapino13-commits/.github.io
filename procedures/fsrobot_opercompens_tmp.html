CREATE PROCEDURE FSROBOT_OPERCOMPENS_TMP(
    REP_DATE D_DATE,
    STEP D_SMALLINT = 2,
    REP_K_ID D_ID = null)
RETURNS (
    KCC_ID D_INTEGER,
    K_ID D_ID,
    C_ID D_ID,
    KAL_NEW_REG_ID D_INTEGER,
    PR_CODE D_STR_100,
    CLI_PERC_DOLIA D_CURRENCY,
    SH_CALC_PERC D_CURRENCY,
    CALC_PERC_OSTAT CURR_TYPE,
    CLI_PERC_DOLIA_CORRECTED CURR_TYPE,
    KC_ID D_ID)
AS
declare variable k_id_old d_id;
declare variable kc_closed_ol_id d_id;
declare variable ks_state d_integer;
declare variable k_amount_approved_tmp d_currency;
declare variable k_amount_approved d_currency;
declare variable k_kolich_client_approved d_integer;
declare variable kc_amount_approved d_currency;
declare variable kc_amount_approved_minus d_currency;
declare variable dolia d_percent;
declare variable has_sh_payment d_smallint;
declare variable is_ipf d_smallint;
declare variable sh_calc_perc_tmp d_currency;
declare variable has_next_oper d_smallint;
declare variable sql_text d_blob_text;
begin
    -- дата запуска не может выходить за диапазон
    if (not(coalesce(rep_date, cast('01.01.1900' as date)) between '01.12.2020' and '28.02.2021')) then
        exit;

    if (rep_date > current_date) then --дата запуска не может быть больше текущей
        exit;

    if (step = 1) then
    begin
        k_id_old = 0;
    
        sql_text = 'select kcc.kcc_id, kncl.kc_from_k_id, kncl.kc_from_c_id, kncl.kc_amount_approved, kncl.kc_closed_ol_id, kcc.kcc_from_kc_id ';
        sql_text = blob_append(sql_text,'from knclient_compens kcc ');
        sql_text = blob_append(sql_text,'inner join knclient kncl on kncl.kc_id = kcc.kcc_from_kc_id ');
        sql_text = blob_append(sql_text,'where '''||f_datetostr(:REP_DATE)||''' between kcc.kcc_date_begin and kcc.kcc_date_end '); -- беру активных участников на дату

        if (rep_k_id is not null) then
            sql_text = blob_append(sql_text,'and kcc.kcc_from_k_id = '||:rep_k_id||' ');
        sql_text = blob_append(sql_text,'order by kncl.kc_from_k_id ');
        for execute statement(sql_text)
        into :kcc_id, :k_id, :c_id, :kc_amount_approved, :kc_closed_ol_id, :kc_id
        do
        begin
            -- если клиент из программы компенсации закрыл свою долю по нему не отрабатываем
            if (kc_closed_ol_id is not null) then
            begin
                if (exists (select *
                            from operlist ol
                            where ol.ol_id = :kc_closed_ol_id
                            and ol.is_delete = 0
                            and ol.ol_date < :rep_date)) then
                    continue;
            end
    
            if (k_id <> k_id_old) then
            begin
                k_id_old = k_id;

                has_sh_payment = 0;

                -- есть платеж по графику на дату ДА/НЕТ
                if (exists (
                    with t_zad
                    as (select ok_sh_id, ok_sh_date, sum(ok.ok_os + ok_sh_os) sh_os, sum(ok.ok_perc + ok_sh_perc) sh_perc
                        from operkred ok
                        where ok.ok_from_k_id = :k_id and
                              ok.ok_type < 7 and
                              ok.ok_ol_is_active = 1 and
                              ok_from_c_id = 0 and
                              ok.ok_ol_date <= :rep_date
                        group by 1, 2)
                    select ok_sh_date
                    from t_zad
                    where (sh_os <> 0 or
                          sh_perc <> 0) and
                          ok_sh_date = :rep_date)) then
                    has_sh_payment = 1;
    
                select kont.k_kolich_client_approved,
                       kont.k_amount_approved,
                       kal.kal_new_reg_id,
                       pr_code,
                       ks.ks_state,
                       iif(pr_code like '%.ipf.%', 1, 0)
                from kontract kont
                inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
                inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                inner join product pr on pr.pr_id = kont.k_from_pr_id
                where kont.k_id = :k_id
                and :rep_date between kal.kal_date and kal.kal_date_end
                and kal.is_delete = 0
                and :rep_date between ks.ks_date_begin and ks.ks_date_end
                and ks.ks_ol_is_delete = 0
                into :k_kolich_client_approved,
                     :k_amount_approved,
                     :kal_new_reg_id,
                     :pr_code,
                     :ks_state,
                     :is_ipf;

                has_next_oper = 0;

                if (exists(
                      select *
                      from operkred ok
                      inner join operlist ol on ok.ok_from_ol_id = ol.ol_id
                      where ok.ok_from_k_id = :k_id
                      and ok.ok_ol_date >= :rep_date
                      and ok.ok_ol_is_active = 1
                )) then
                    has_next_oper = 1;
            end

            if (has_next_oper = 1) then continue; -- если есть платеж датой запуска робота или более поздней датой, исключаю
            if ((has_sh_payment = 0) and (is_ipf = 0)) then continue; -- если нет платежа по графику + классика, исключаю
            if ((rep_date <> '01.12.2020') and (is_ipf = 1)) then continue; -- если это ИПФ и дата операции не 01.12.2020, тогда исключаю
            if (ks_state <> 1) then continue; -- если заем не активный по нему робот отрабатывать не будет

            sh_calc_perc = 0;
    
            if(is_ipf = 0) then
            begin
                select round(sum(sh_calc_perc) * 1.02, 0)
                from aarep_make_zad_list(:k_id, :rep_date, 1)
                where sh_date = :rep_date
                into :sh_calc_perc;
            end
            else
            begin
                -- Если ИПФ важно получить % + НСП по каждому платежу в отдельности
                for select round(zad_perc * 1.02, 0)
                from aarep_make_zad_list(:k_id, :rep_date, 0)
                where sh_date between cast('01.12.2020' as date) and cast('28.02.2021' as date)
                into :sh_calc_perc_tmp
                do
                begin
                    sh_calc_perc = sh_calc_perc + sh_calc_perc_tmp;
                end
            end
    
            sh_calc_perc = coalesce(sh_calc_perc, 0);
    
            if (sh_calc_perc <> 0) then -- если не найден платеж на дату или сумма накопленных % = 0, тогда платеж % от гос-ва сформирован не будет
            begin
                if (k_kolich_client_approved = 1) then -- если в договоре всего 1 участник, тогда его доля = 100%, поэтому сумма накопленных % по нему = сумме накоопленных % по всему займу
                begin
                    cli_perc_dolia = sh_calc_perc;
                    suspend;
                end
                else
                begin
                    k_amount_approved_tmp = k_amount_approved;
    
                    kc_amount_approved_minus = 0;
    
                    select sum(kncl.kc_amount_approved)
                    from kontract kont
                    inner join knclient kncl on kont.k_id = kncl.kc_from_k_id
                    inner join operlist ol on ol.ol_id = kncl.kc_closed_ol_id
                    where ol.is_delete = 0
                    and ol.ol_date < :rep_date
                    and kncl.kc_amount_approved > 0
                    and kncl.kc_from_k_id = :k_id
                    into :kc_amount_approved_minus;
    
                    kc_amount_approved_minus = coalesce(kc_amount_approved_minus, 0);
    
                    k_amount_approved_tmp = k_amount_approved_tmp - kc_amount_approved_minus;
    
                    dolia = (kc_amount_approved * 100 / k_amount_approved_tmp);
                    cli_perc_dolia = round(sh_calc_perc * dolia/100);
    
                    suspend;
                end
            end
        end
    end
    else if (step = 2) then
    begin
        /*
        данный шаг нужен для устранения неточностей с округлением. Из-за окрегления может получиться ситуация
        что робот либо недопосадил в качестве платежа 1 -2 сома либо посадил на 1 - 2 сома больше.
        Актуально для групповых займов
        */

        k_id_old = 0;

        for select kcc_id,
                   k_id,
                   c_id,
                   kal_new_reg_id,
                   pr_code,
                   cli_perc_dolia,
                   sh_calc_perc,
                   kc_id
        from fsrobot_opercompens_tmp (:rep_date, 1, :rep_k_id)
        order by k_id
        into kcc_id,
             k_id,
             c_id,
             kal_new_reg_id,
             pr_code,
             cli_perc_dolia,
             sh_calc_perc,
             kc_id
         do
         begin
            if (k_id_old <> k_id) then
            begin
                k_id_old  = k_id;
                calc_perc_ostat = sh_calc_perc;
            end
            calc_perc_ostat = calc_perc_ostat - cli_perc_dolia;

            cli_perc_dolia_corrected = cli_perc_dolia;
            if (calc_perc_ostat between -2 and 2) then
                cli_perc_dolia_corrected =  cli_perc_dolia_corrected + calc_perc_ostat;

            suspend;
         end
    end
end
