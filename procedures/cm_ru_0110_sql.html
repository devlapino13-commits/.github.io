CREATE PROCEDURE CM_RU_0110_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable C_ID ID_TYPE;
declare variable C_BIRTH_DATE date;
declare variable KOL_CL integer;
declare variable PR_CODE varchar(100);
begin
  select PR_CODE, K_DISB_DATE
  from KONTRACT, PRODUCT
  where K_ID = :K_ID and
        K_FROM_PR_ID = PR_ID
  into PR_CODE, KF_DISB_DATE;
  if (PR_CODE containing '.PATENT') then
    exit;
  else
    for select KC_FROM_C_ID, KC_AMOUNT_APPROVED, C_NAME_FIO, C_PASS_NO, C_BIRTH_DATE
        from KNCLIENT, CLIENTS
        where KC_FROM_K_ID = :K_ID and
              KC_AMOUNT_APPROVED > 0 and
              KC_FROM_C_ID = C_ID
        into C_ID, KC_AMOUNT, res_info_text, KF_CLIENT_PASS_NO, C_BIRTH_DATE
    do
    begin
      --        Изменен по письму от 14.06.2013 19:19 Если у клиента возраст не достик 21годам на дату получения кредита то проверяет следующее
      if (C_BIRTH_DATE > dateadd(year, -21, KF_DISB_DATE)) then
      begin
        if (:C_BIRTH_DATE <= dateadd(year, -18, :KF_DISB_DATE) and
            PR_CODE containing '.FAMILY') then
        begin
          KOL_CL = coalesce(KOL_CL, 0) + 1;
          if (KOL_CL > 1) then
          begin
            res_error_is_critical = 0;
            res_error_text = 'Количество клиентов младше 21-лет, не может быть больше одного!';
            suspend;
          end
          if ( --в рамках продукта «Семья» и «Семья Плюс», сейчас те, кто моложе 21 года, но достигли 18 лет могут получить с родителями. Мы хотим, чтобы они могли получать еще с супругами или родными братьями/сестрами. Зара 25.12.17
              exists(select *
                     from HP_CLIENT_RELATION_LIST(:C_ID, 0), KNCLIENT
                     where CREL_FROM_REL_ID between 1 and 8 and
                           CREL_OTHER_C_ID = KC_FROM_C_ID and
                           KC_AMOUNT_APPROVED > 0 and
                           KC_FROM_K_ID = :K_ID)) then --Добавлено 01.01.18 Письмо от Зары от
          begin
            if (exists(select *
                       from HP_CLIENT_RELATION_LIST(:C_ID, 0), KNCLIENT
                       where CREL_FROM_REL_ID in (3, 4) and
                             CREL_OTHER_C_ID = KC_FROM_C_ID and
                             KC_AMOUNT_APPROVED > 0 and
                             KC_FROM_K_ID = :K_ID)) then
            begin
              res_error_is_critical = 0;
              res_error_text = 'Клиент младше 21 лет, получает с родителями, не забудьте запросить свидетельство о рождении';
              suspend;
            end    /*
            else
            begin
              --          Проверка клиента новый или нет, если новый то не будет пропускать
              if (not exists(select *
                             from KONTSTATE, KNCLIENT
                             where KS_OL_IS_DELETE = 0 and
                                   KS_STATE = 1 and
                                   KS_DATE_BEGIN <= :KF_DISB_DATE and
                                   KS_FROM_K_ID = KC_FROM_K_ID and
                                   KC_AMOUNT_APPROVED > 0 and
                                   KC_FROM_C_ID = :C_ID)) then
              begin
                res_error_is_critical = 2;
                res_error_text = 'Кредит по возрасту клиента не допускается.';
                suspend;
              end
            end  */
          end
        end
        else
        begin
          res_error_is_critical = 2;
          res_error_text = 'Кредит по возрасту клиента не допускается.';
          suspend;
        end
      end
    end
end
