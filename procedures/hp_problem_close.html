CREATE PROCEDURE HP_PROBLEM_CLOSE
RETURNS (
    K_ID D_ID,
    KONTRACT_NO D_STR_100,
    CLOSE_DATE D_DATE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_FINE CURR_TYPE,
    ZAD_OTHER CURR_TYPE)
AS
begin
  for select k_id, -- ID контракта
             k_kontract_no, -- № контракта
             oks_date_begin -- Дата закрытия
      from oksummary oks
      inner join kontract k on oks.oks_from_k_id = k.k_id
      where current_date between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_close_state = 0 and
            oks.oks_balance = 0
      into :k_id, :kontract_no, :close_date
  do
  begin
    select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine)
    from aarep_make_zad_list(:k_id, current_date, 1)
    into :zad_os, :zad_perc, :zad_fine;

    zad_os = coalesce(zad_os, 0); -- Задолж. по ОС
    zad_perc = coalesce(zad_perc, 0); -- Накопленные  %
    zad_fine = coalesce(zad_fine, 0); -- Долг по штрафам

    select sum(ox_amount_plus + ox_amount_minus)
    from operextra o
    inner join operlist ol on ox_from_ol_id = ol_id and
          ol.is_delete = 0
    where o.ox_object_id = :k_id
    into :zad_other;

    zad_other = coalesce(zad_other, 0); -- Прочие долги

    if (zad_os = 0 and
        zad_perc = 0 and
        zad_fine = 0 and
        zad_other = 0) then
      suspend;
  end
end
