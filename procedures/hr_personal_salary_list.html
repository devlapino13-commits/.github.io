CREATE PROCEDURE HR_PERSONAL_SALARY_LIST(
    REG_ID INTEGER,
    DEP_ID INTEGER,
    JL_ID INTEGER,
    DD1 DATE,
    DD2 DATE)
RETURNS (
    PS_ID INTEGER,
    PS_FROM_HO_ID DECIMAL(18,0),
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_FROM_REG_ID INTEGER,
    PS_FROM_DEP_ID INTEGER,
    PS_FROM_JL_ID INTEGER,
    PS_FROM_C_ID DECIMAL(18,0),
    PS_IS_ON_STAFF_TEXT VARCHAR(20),
    PS_IS_WORKING_TEXT VARCHAR(20),
    PS_SALARY DECIMAL(18,4),
    PS_REG_NAME VARCHAR(100),
    PS_DEP_NAME VARCHAR(100),
    PS_JL_NAME VARCHAR(100),
    PS_C_NAME_FIO VARCHAR(100),
    PS_ORDER_TYPE INTEGER,
    PS_ORDER_BASE_TYPE INTEGER,
    PS_TABEL_NO VARCHAR(20),
    PS_DATE_HIRE DATE,
    PS_DATE_HIRE_OFFICE DATE,
    PS_DATE_GONE_OFFICE DATE,
    SAT_TOTAL_NORMA INTEGER,
    SAT_TOTAL_WORKED INTEGER,
    SAT_TOTAL_OTPUSK INTEGER,
    SAT_TOTAL_BOLNICH INTEGER,
    SAC_ID INTEGER,
    SAC_PRIORITY_ORDER INTEGER,
    SAC_VALUE DECIMAL(18,4),
    SAT_IS_NEW INTEGER,
    SAT_PDNAL_LGOTA DECIMAL(18,4),
    PS_C_SEX_TEXT VARCHAR(20),
    PS_IS_ON_STAFF SMALLINT,
    PS_IS_WORKING SMALLINT,
    SAT_HAS_TABEL INTEGER)
AS
declare variable MONTH_NO integer;
declare variable YEAR_NO integer;
declare variable KOL integer;
declare variable SCN_VALUE decimal(18,4);
declare variable C_ID_PRIOR decimal(18,0);
declare variable PD_LGOTA integer;
declare variable PD_NAL_LG decimal(18,4);
declare variable SCN_VALUE1 decimal(18,4);
declare variable SAT_TOTAL_CALC decimal(18,4);
declare variable SAT_BASE_TYPE integer;
declare variable REG_ID_PRIOR integer;
declare variable SAO_MOUNT_KOEF_ID integer;
declare variable HAS_MOUNT_KOEF integer;
declare variable SAO_MOUNT_ID integer;
declare variable TNORM integer;
declare variable CM_DATABASE_ID integer;
declare variable USE_TABEL_HOURS D_BOOL;
declare variable TABEL_NORMA D_INTEGER;
declare variable SHORT_DAYS D_INTEGER;
declare variable TOTAL_DAYS D_INTEGER;
declare variable DAY_NORMA_HOURS D_INTEGER;
begin
  year_no = extract(year from dd1);
  month_no = extract(month from dd1);
  reg_id_prior = 0;
  c_id_prior = 0;

  :cm_database_id = f_get_db_id();

  select sum(scd_work_hours), sum(iif(scd_work_hours = 7, 1, 0)), sum(iif(scd_work_hours <> 0, 1, 0))
  from sa_calendays
  where scd_year_month = (:year_no * 100 + :month_no)
  into :tnorm, short_days, total_days;
  if (tnorm is null) then
    tnorm = 0;

  if (short_days is null) then
    short_days = 0; -- кол-во коротких дней (7 часовых)

  if (total_days is null) then
    total_days = 0; -- кол-во рабочих дней

  pd_lgota = 1;
  select scn_value
  from sa_get_current_consvalue(:dd1)
  where scn_id = 1
  into :scn_value1;

  scn_value = 0;
  select scn_value
  from sa_get_current_consvalue(:dd1)
  where scn_id = 2
  into :scn_value;

  pd_nal_lg = scn_value * scn_value1;

  sao_mount_id = 0;
  select sac_id, sac_priority_order
  from sa_config
  where sac_is_credit = 7
  into :sao_mount_id, :sao_mount_koef_id;
  if (sao_mount_id is null) then
    sao_mount_id = 0;
  if (sao_mount_koef_id is null) then
    sao_mount_koef_id = 0;

  use_tabel_hours = 0;
  if (cm_database_id <> 94401 or (year_no * 100 + :month_no) >= 202511) then
    use_tabel_hours = 1;

-- новый вариант, который отображает при расчете ЗП все не зависимо от табели
  for select ps_id, ps_from_ho_id, ps_date_begin, ps_date_end, ps_period_month, ps_date_end_in_order, ps_from_reg_id,
             ps_from_dep_id, ps_from_jl_id, ps_from_c_id, ps_is_on_staff_text, ps_is_working_text, ps_salary,
             ps_reg_name, ps_dep_name, ps_jl_name, ps_c_name_fio, ps_order_type, ps_order_base_type, ps_tabel_no,
             ps_date_hire, ps_date_hire_office, ps_date_gone_office, ps_c_sex_text, ps_is_on_staff, ps_is_working, ps_norma_hours
      from hr_all_personal_new_pstatus(:reg_id, :dd1, :dd2)
      into :ps_id, :ps_from_ho_id, :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order,
           :ps_from_reg_id, :ps_from_dep_id, :ps_from_jl_id, :ps_from_c_id, :ps_is_on_staff_text, :ps_is_working_text,
           :ps_salary, :ps_reg_name, :ps_dep_name, :ps_jl_name, :ps_c_name_fio, :ps_order_type, :ps_order_base_type,
           :ps_tabel_no, :ps_date_hire, :ps_date_hire_office, :ps_date_gone_office, :ps_c_sex_text, :ps_is_on_staff,
           :ps_is_working, :day_norma_hours
  do
  begin
    sat_has_tabel = 0;
    sat_total_norma = 0;
    sat_total_worked = 0;
    sat_total_otpusk = 0;
    sat_total_bolnich = 0;

    if (cm_database_id = 94401) then
      sat_has_tabel = 1;

    if (coalesce(day_norma_hours, 0) = 0) then
      day_norma_hours = 8;

    sat_total_norma = tnorm;

    for select sum(iif(:use_tabel_hours = 1, sat_total_calc, round(sat_days_calc, 0) * 8.0)), sawt_base_type, sum(sat_total_hours)
        from sa_tabel, sa_worktype
        where sawt_id = sat_from_sawt_id and
              sat_is_ready = 1 and
              sat_year_month = (:year_no * 100 + :month_no) and
              sat_office_id = :ps_from_reg_id and
              sat_from_c_id = :ps_from_c_id
        group by sawt_base_type
        order by sawt_base_type
        into :sat_total_calc, :sat_base_type, :tabel_norma
    do
    begin
      sat_has_tabel = 1;
      if (sat_total_calc is null) then
        sat_total_calc = 0;
      if (sat_base_type = 0) then
      begin
        sat_total_norma =  tnorm;
        if(use_tabel_hours = 1 and cm_database_id = 94401) then
          sat_total_norma = total_days * day_norma_hours - short_days;
        sat_total_worked = sat_total_worked + sat_total_calc;
      end
      if (sat_base_type = 4) then
        sat_total_otpusk = sat_total_otpusk + sat_total_calc;
      if (sat_base_type = 5) then
        sat_total_bolnich = sat_total_bolnich + sat_total_calc;
    end

    sat_pdnal_lgota = 0;
    if ((c_id_prior <> :ps_from_c_id) or (ps_from_reg_id <> :reg_id_prior)) then
    begin
      select res_value
      from hp_get_opersal_sacvalue(:ps_from_c_id, (:year_no * 100 + :month_no), (:year_no * 100 + :month_no), null, null, ' where SAC_Kind=6 ')
      into :sat_pdnal_lgota;

      if (sat_pdnal_lgota = 0) then
        sat_pdnal_lgota = pd_nal_lg;
    end
    kol = 0;
    has_mount_koef = 0;
    for select osd_from_sac_id, sac_priority_order, sum(osd_value)
        from opersalmain, opersaldetail, sa_config
        where osm_id = osd_from_osm_id and
              osm_ol_is_active = 1 and
              osm_year_month = (:year_no * 100 + :month_no) and
              osm_from_c_id = :ps_from_c_id and
              osd_from_sac_id = sac_id
        group by osd_from_sac_id, sac_priority_order
        order by sac_priority_order, osd_from_sac_id
        into :sac_id, :sac_priority_order, :sac_value
    do
    begin
      if (sac_id is not null) then
      begin
        sat_is_new = 0;
        kol = kol + 1;
--        if (sac_id=SAO_MOUNT_KOEF_ID) then HAS_MOUNT_KOEF=1; Старый вариант
        if (sac_id = sao_mount_id) then
          has_mount_koef = 1;
        suspend;
      end
    end

    if (kol = 0) then
    begin
      sat_is_new = 1;
      sac_id = 0;
      sac_priority_order = 0;
      sac_value = 0;
      suspend;
    end
    c_id_prior = :ps_from_c_id;
    reg_id_prior = :ps_from_reg_id;
  end
end
