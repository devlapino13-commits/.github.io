CREATE PROCEDURE HP_CELL_MAIN_STATUS(
    C_ID D_ID)
RETURNS (
    OK_CELL_MAIN D_STR_100,
    CDA_STATUS D_SMALLINT,
    CDA_STATUS_NAME D_STR_20)
AS
declare variable clt_id d_integer;
begin
  -- процедура возвращает верифицированный основной номер телефона клиента и его статус в contdata (RFCKG-2578)

  select f_cont_info('clients', :c_id, 'cell_main', 'ok')
  from rdb$database
  into ok_cell_main;

  if (coalesce(ok_cell_main, '') = '') then
    ok_cell_main = null;
  else
  begin
    select clt_id
    from contlist
    where clt_code = :ok_cell_main
    into clt_id;

    select cda_status,
           iif(cda_status = 1, 'Подтвержден', 'Не подтвержден')
    from contdata
    where cda_from_c_id = :c_id and
          cda_from_clt_id = :clt_id and
          cda_is_delete = 0 and
          cda_is_main = 1
    order by cda_date desc
    rows 1
    into cda_status,
         cda_status_name;
  end

  suspend;
end
