CREATE PROCEDURE AAREP_EPAY_BANKCARD_DISB(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    NUM SMALLINT,
    MIGRATION_SER STR_255,
    MIGRATION_NO STR_255,
    NAME STR_255,
    SURNAME STR_255,
    MIDLENAME STR_255,
    BIRTH_DATE DATE,
    CITIZEN STR_100,
    PASS_TYPE STR_100,
    PASS_SER STR_10,
    PASS_NO STR_100,
    DURATION_FROM DATE,
    DURATION_TO DATE,
    DOCUMENT STR_100)
AS
declare variable c_id id_type;
begin
  --для ВАЛЮТНОГО КОНТРОЛЯ
  num = 0;

  for select ag_object_id
  from operagr, agreements
  where ogr_ol_date between :rep_date_begin and :rep_date_end and
        ogr_ol_is_active = 1 and
        ogr_from_ag_id = ag_id and
        ag_object_kind = f_objkind('clients', 'id')
  group by ag_object_id
  into :c_id
  do
  begin
    select name, surname, midlename, citizen, birthday,
           pass_type, pass_ser, pass_no, duration_from, duration_to, document, migration_ser, migration_no
    from aarep_epay_client_info(:c_id, :rep_date_end)
    into :name, :surname, :midlename, :citizen, birth_date,
         :pass_type, :pass_ser, :pass_no, :duration_from, :duration_to, :document, :migration_ser, :migration_no;

    num = num + 1;
    suspend;
  end
end
