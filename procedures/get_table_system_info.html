CREATE PROCEDURE GET_TABLE_SYSTEM_INFO(
    TABLE_NAME VARCHAR(50))
RETURNS (
    RES_TABLE_NAME VARCHAR(50),
    RES_FIELD_NAME VARCHAR(50),
    RES_FIELD_SIZE INTEGER,
    RES_FIELD_TYPE VARCHAR(50),
    RES_FIELD_DESCRIPTION VARCHAR(4096),
    RES_IS_KEY_FIELD INTEGER,
    RES_FK_TABLE VARCHAR(50),
    RES_FK_FIELD VARCHAR(50),
    RES_INDEX INTEGER)
AS
declare variable res_field_type_id integer;
declare variable res_field_sub_type integer;
declare variable res_field_scale integer;
begin
  table_name = upper(table_name);

  for select trim(lower(r.rdb$relation_name)), trim(lower(r.rdb$field_name)), trim(left(r.rdb$description, 4096)),
             f.rdb$field_length, f.rdb$field_type, f.rdb$field_scale, coalesce(f.rdb$field_sub_type, 0),
             r.rdb$field_position
      from rdb$fields f, rdb$relation_fields r
      where f.rdb$field_name = r.rdb$field_source and
            r.rdb$system_flag = 0 and
            (r.rdb$relation_name = :table_name or
            (:table_name = '' and
            not(r.rdb$relation_name starting 'VREF')))
      order by r.rdb$relation_name, r.rdb$field_position
      into :res_table_name, :res_field_name, res_field_description, :res_field_size, :res_field_type_id,
           :res_field_scale, :res_field_sub_type, res_index
  do
  begin
    res_field_type = '';

--  7   SMALLINT
--  8   INTEGER
--  10  FLOAT
--  12  DATE
--  16  BIGINT
--  27  DOUBLE PRECISION
--  35  TIMESTAMP
--  37  VARCHAR
--  261 BLOB

--smallint
--integer
--float
--date
--bigint
--percent
--currency
--double
--datetime
--string
--memo
--blob
    if (res_field_type_id = 7) then
      res_field_type = 'smallint';
    if (res_field_type_id = 8) then
      res_field_type = 'integer';
    if (res_field_type_id = 10) then
      res_field_type = 'float';
    if (res_field_type_id = 12) then
      res_field_type = 'date';
    if (res_field_type_id = 16) then
    begin
      if (res_field_scale = 0) then
        res_field_type = 'bigint';
      else
      if (res_field_scale <= -6) then
        res_field_type = 'percent';
      else
        res_field_type = 'currency';
    end
    if (res_field_type_id = 27) then
      res_field_type = 'double';
    if (res_field_type_id = 35) then
      res_field_type = 'datetime';
    if (res_field_type_id = 37) then
    begin
      res_field_type = 'string';
      res_field_size = res_field_size / 4;
    end
    if (res_field_type_id = 261) then
    begin
      if (res_field_sub_type = 1) then
        res_field_type = 'memo';
      else
        res_field_type = 'blob';
    end

    res_is_key_field = 0;
    if (exists(select *
               from rdb$relation_constraints r
               inner join rdb$index_segments i on i.rdb$index_name = r.rdb$index_name
               inner join rdb$relation_fields rf on rf.rdb$field_name = i.rdb$field_name and
                     rf.rdb$relation_name = r.rdb$relation_name
               inner join rdb$fields f on f.rdb$field_name = rf.rdb$field_source
               where r.rdb$constraint_type = 'PRIMARY KEY' and
                     r.rdb$relation_name = upper(:res_table_name) and
                     i.rdb$field_name = upper(:res_field_name))) then
      res_is_key_field = 1;

    res_fk_table = '';
    res_fk_field = '';
    select trim(lower(c.rdb$relation_name)), trim(lower(d.rdb$field_name))
    from rdb$ref_constraints b, rdb$relation_constraints a, rdb$relation_constraints c, rdb$index_segments d, rdb$index_segments e, rdb$indices i
    where (a.rdb$constraint_type = 'FOREIGN KEY') and
          (a.rdb$constraint_name = b.rdb$constraint_name) and
          (b.rdb$const_name_uq = c.rdb$constraint_name) and
          (c.rdb$index_name = d.rdb$index_name) and
          (a.rdb$index_name = e.rdb$index_name) and
          (a.rdb$index_name = i.rdb$index_name) and
          (a.rdb$relation_name) = upper(:res_table_name) and
          (e.rdb$field_name) = upper(:res_field_name)
    rows 1
    into :res_fk_table, :res_fk_field;
    if (res_fk_table is null) then
      res_fk_table = '';
    if (res_fk_field is null) then
      res_fk_field = '';

    suspend;
  end
end
