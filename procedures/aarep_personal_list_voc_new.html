CREATE PROCEDURE AAREP_PERSONAL_LIST_VOC_NEW(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    JL_NAME VARCHAR(100),
    C_HIRED_DATE DATE,
    BALANCE_END_YEAR_31_12_2008 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2009 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2010 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2011 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2012 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2013 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2014 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2015 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2016 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2017 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2018 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2019 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2020 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2021 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2022 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2023 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2024 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2025 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2026 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2027 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2028 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2029 DECIMAL(18,4),
    BALANCE_END_YEAR_31_12_2030 DECIMAL(18,4),
    TOTAL_BALANCE DECIMAL(18,4))
AS
declare variable c_id id_type;
declare variable current_year d_integer;
declare variable sat_voc_payed decimal(18,4);
declare variable sat_voc_end decimal(18,4);
declare variable sat_voc_calc decimal(18,4);
declare variable sat_voc_end_year decimal(18,4);
declare variable report_date_end date;
declare variable year_end date;
declare variable payed_tmp decimal(18,4);
declare variable calc_tmp decimal(18,4);
declare variable dd1 date;
declare variable dd2 date;
begin
  balance_end_year_31_12_2008 = 0;
  balance_end_year_31_12_2009 = 0;
  balance_end_year_31_12_2010 = 0;
  balance_end_year_31_12_2011 = 0;
  balance_end_year_31_12_2012 = 0;
  balance_end_year_31_12_2013 = 0;
  balance_end_year_31_12_2014 = 0;
  balance_end_year_31_12_2015 = 0;
  balance_end_year_31_12_2016 = 0;
  balance_end_year_31_12_2017 = 0;
  balance_end_year_31_12_2018 = 0;
  balance_end_year_31_12_2019 = 0;
  balance_end_year_31_12_2020 = 0;
  balance_end_year_31_12_2021 = 0;
  balance_end_year_31_12_2022 = 0;
  balance_end_year_31_12_2023 = 0;
  balance_end_year_31_12_2024 = 0;
  balance_end_year_31_12_2025 = 0;
  balance_end_year_31_12_2026 = 0;
  balance_end_year_31_12_2027 = 0;
  balance_end_year_31_12_2028 = 0;
  balance_end_year_31_12_2029 = 0;
  balance_end_year_31_12_2030 = 0;

  total_balance = 0;
  -- использую процедуру получаю список дуйствуюших сотрудников и вывожу их данные
  for select reg_name, c_name_fio, jl_name, c_hired_date, ps_from_c_id, sat_voc_end_year
      from aarep_personal_list_voc(:rep_region_id, :rep_date_end, 1)
      into :reg_name, :c_name_fio, :jl_name, :c_hired_date, :c_id, :balance_end_year_31_12_2022

  do
  begin
    current_year = 2008; -- присваиваю переменную для цикла
    while (current_year < 2030) do
    -- отработать цикл до 2030 года
    begin

      select sat_vocation_end -- остаток на дату отчета
      from aarep_sa_tabel_one_voc(:rep_region_id, :c_id, :current_year, 12)
      into :sat_voc_end;
      sat_voc_payed = 0;
      sat_voc_calc = 0;
      sat_voc_end_year = sat_voc_end;
      report_date_end = cast('31.12.' || :current_year as date);
      year_end = report_date_end;

      for select hrv_date_begin, hrv_date_end --
          from hr_voctravel, sa_worktype swt, hr_orders
          where ((hrv_date_begin > :report_date_end and
                hrv_date_begin <= :year_end) or (hrv_date_end > :report_date_end and
                hrv_date_end <= :year_end)) and
                hrv_from_c_id = :c_id and
                ho_order_type = sawt_id and
                hrv_from_ho_id = ho_id and
                hrv_status = 1 and
                swt.sawt_base_type = 4 /* трудовой отпуск */
                and
                sawt_is_paid = 1 /* опалчиваемы отпуск */
          into :dd1, :dd2
      do
      begin
        if (dd1 <= :report_date_end) then
          dd1 = :report_date_end + 1;
        if (dd2 > :year_end) then
          dd2 = :year_end;
        payed_tmp = cast((dd2 - dd1 + 1) as integer);
        sat_voc_payed = sat_voc_payed + payed_tmp;
      end
      -- получаем остаток на конец года
      sat_voc_end_year = coalesce(sat_voc_end, 0) + coalesce(sat_voc_calc, 0) - coalesce(sat_voc_payed, 0);

      if (extract(year from :rep_date_end) < current_year) then
        sat_voc_end_year = 0;

      if (current_year = 2008) then
        balance_end_year_31_12_2008 = sat_voc_end_year;

      if (current_year = 2009) then
        balance_end_year_31_12_2009 = sat_voc_end_year;

      if (current_year = 2010) then
        balance_end_year_31_12_2010 = sat_voc_end_year;

      if (current_year = 2011) then
        balance_end_year_31_12_2011 = sat_voc_end_year;

      if (current_year = 2012) then
        balance_end_year_31_12_2012 = sat_voc_end_year;

      if (current_year = 2013) then
        balance_end_year_31_12_2013 = sat_voc_end_year;

      if (current_year = 2014) then
        balance_end_year_31_12_2014 = sat_voc_end_year;

      if (current_year = 2015) then
        balance_end_year_31_12_2015 = sat_voc_end_year;

      if (current_year = 2016) then
        balance_end_year_31_12_2016 = sat_voc_end_year;

      if (current_year = 2017) then
        balance_end_year_31_12_2017 = sat_voc_end_year;

      if (current_year = 2018) then
        balance_end_year_31_12_2018 = sat_voc_end_year;

      if (current_year = 2019) then
        balance_end_year_31_12_2019 = sat_voc_end_year;

      if (current_year = 2020) then
        balance_end_year_31_12_2020 = sat_voc_end_year;

      if (current_year = 2021) then
        balance_end_year_31_12_2021 = sat_voc_end_year;

      if (current_year = 2022) then
        balance_end_year_31_12_2022 = sat_voc_end_year;
        
      if (current_year = 2023) then
        balance_end_year_31_12_2023 = sat_voc_end_year;
        
      if (current_year = 2024) then
        balance_end_year_31_12_2024 = sat_voc_end_year;
        
      if (current_year = 2025) then
        balance_end_year_31_12_2025 = sat_voc_end_year;
        
      if (current_year = 2026) then
        balance_end_year_31_12_2026 = sat_voc_end_year;
        
      if (current_year = 2027) then
        balance_end_year_31_12_2027 = sat_voc_end_year;
        
      if (current_year = 2028) then
        balance_end_year_31_12_2028 = sat_voc_end_year;
        
      if (current_year = 2029) then
        balance_end_year_31_12_2029 = sat_voc_end_year;

      if (current_year = 2030) then
        balance_end_year_31_12_2030 = sat_voc_end_year;

      total_balance = balance_end_year_31_12_2008 -- получаем общий баланс
      + balance_end_year_31_12_2009
      + balance_end_year_31_12_2010
      + balance_end_year_31_12_2011
      + balance_end_year_31_12_2012
      + balance_end_year_31_12_2013
      + balance_end_year_31_12_2014
      + balance_end_year_31_12_2015
      + balance_end_year_31_12_2016
      + balance_end_year_31_12_2017
      + balance_end_year_31_12_2018
      + balance_end_year_31_12_2019
      + balance_end_year_31_12_2020
      + balance_end_year_31_12_2021
      + balance_end_year_31_12_2022
      + balance_end_year_31_12_2023
      + balance_end_year_31_12_2024
      + balance_end_year_31_12_2025
      + balance_end_year_31_12_2026
      + balance_end_year_31_12_2027
      + balance_end_year_31_12_2028
      + balance_end_year_31_12_2029
      + balance_end_year_31_12_2030;

      current_year = current_year + 1;
    end
    suspend;
  end
end
