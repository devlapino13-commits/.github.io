CREATE PROCEDURE HP_IS_RESTRUCTURED_KZ(
    C_ID D_ID,
    MONTHS D_INTEGER)
RETURNS (
    IS_RESTRUCTURED D_SMALLINT,
    RESTRUCT_DATE D_DATE,
    RESTRUCT_K_ID D_ID)
AS
declare variable K_ID              D_ID;
declare variable IS_RESTRUCT       D_SMALLINT;
declare variable RESTRUCT_DATE_FST D_DATE;
declare variable RESTRUCT_DATE_LST D_DATE;
begin
  is_restructured = 0;
  restruct_date   = null;
  restruct_k_id   = null;

  for select distinct kc.kc_from_k_id
      from knclient kc
      where kc.kc_from_c_id = :c_id
        and kc.kc_amount_applicate > 0
      into :k_id
  do
  begin
    select is_restruct,
           restruct_date,
           restruct_date_last
    from AAREP_GET_RESTRUCTURING(:k_id, current_date, 39801)
    into :is_restruct,
         :restruct_date_fst,
         :restruct_date_lst;

    -- проверяем попадает ли последняя реструктуризация в нужный период
    if (is_restruct = 1 and
        coalesce(restruct_date_lst, restruct_date_fst) >= dateadd(-:months month to current_date)) then
    begin
      is_restructured = 1;
      restruct_date   = coalesce(restruct_date_lst, restruct_date_fst);
      restruct_k_id   = k_id;
      suspend;
      exit;
    end
  end

  suspend;
end
