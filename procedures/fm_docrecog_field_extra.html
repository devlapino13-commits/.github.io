CREATE PROCEDURE FM_DOCRECOG_FIELD_EXTRA(
    DRF_CODE D_STR_CODE,
    PK_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    RES_JSON_EXTRA D_STR_8100)
AS
declare variable db_id d_integer;
declare variable lang d_str_lang;
begin
  res_json_extra = '';
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);

  -- Страна рождения
  if (drf_code = 'birth_country') then
  begin
    select list(f_format('{"id":"%1", "name":"%1"}', rd_name), ',')
    from fm_get_ref('citizen')
    into res_json_extra;
  
    res_json_extra = f_format('[%1]', res_json_extra);
  end

  -- Пол
  if (drf_code = 'calc_gender') then
  begin
    res_json_extra = f_format(
      '[{"id":"0", "name":"%1"},{"id":"1", "name":"%2"}]',
      f_lang(lang, 'Женский'), f_lang(lang, 'Мужской'));
  end

   -- МФО
  if (drf_code = 'compet_mfo') then
  begin
    -- Конкуренты в РФ, не переводим
    res_json_extra = f_format(
      '[{"id":"0","name":"%1"},{"id":"1","name":"%2"},{"id":"2","name":"%3"}]',
      'Другое', 'Байбол', 'МЦК');
  end

  -- Поле Код УГНС
  if (drf_code = 'memo_3' and pk_id in (44,68,71) and db_id in (39801, 41701)) then
  begin
    select list(f_format('{"id":"%1", "name":"%1"}', rd_code), ',')
    from refdata
    where rd_main_code = 'TerritorialTaxAdmin'
      and rd_parent_id > 0
    into res_json_extra;

    res_json_extra = f_format('[%1]', res_json_extra);
  end

   -- МФО
  if (drf_code = 'compet_mfo') then
  begin
    -- Конкуренты в РФ, не переводим
    res_json_extra = f_format(
      '[{"id":"0","name":"%1"},{"id":"1","name":"%2"},{"id":"2","name":"%3"}]',
      'Другое', 'Байбол', 'МЦК');
  end

  if (drf_code = 'calc_family_status') then
  begin
    -- Семейное положение (КГ)
    res_json_extra = f_format(
      '[{"id":"0","name":"%1"},{"id":"1","name":"%2"},{"id":"2","name":"%3"},{"id":"3","name":"%4"},{"id":"4","name":"%5"}]',
      f_lang(lang, 'Не указано'),
      f_lang(lang, 'Состоит в браке'),
      f_lang(lang, 'В браке не состоял(а)'),
      f_lang(lang, 'Разведен(а)'),
      f_lang(lang, 'Вдовец/вдова'));
  end

  suspend;
end
