CREATE PROCEDURE AAREP_ANALITICA_LO_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_PROJECT VARCHAR(100),
    REP_PERIOD_KIND INTEGER)
RETURNS (
    AA_K_ID DECIMAL(18,0),
    AA_DELIQ_DAYS INTEGER,
    AA_LO_ID DECIMAL(18,0),
    AA_CURR_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_CURR_KOLICH_CL_ACTIVE_END INTEGER,
    AA_CURR_DELIQ_CL_KOLICH INTEGER,
    AA_CURR_DELIQ_BALANCE DECIMAL(18,4),
    AA_OWN_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_OWN_KOLICH_CL_ACTIVE_END INTEGER,
    AA_OWN_DELIQ_CL_KOLICH INTEGER,
    AA_OWN_DELIQ_BALANCE DECIMAL(18,4),
    AA_IN_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_IN_KOLICH_CL_ACTIVE_END INTEGER,
    AA_IN_DELIQ_CL_KOLICH INTEGER,
    AA_IN_DELIQ_BALANCE DECIMAL(18,4),
    AA_OUT_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_OUT_KOLICH_CL_ACTIVE_END INTEGER,
    AA_OUT_DELIQ_CL_KOLICH INTEGER,
    AA_OUT_DELIQ_BALANCE DECIMAL(18,4),
    AA_DISB_LO_FIO VARCHAR(100),
    AA_LO_OFFICE_NAME VARCHAR(100),
    AA_LO_JOBNAME VARCHAR(100),
    PS_IS_ON_STAFF_STR VARCHAR(100),
    PS_IS_WORKING_STR VARCHAR(100),
    AA_LO_BRANCH_NAME VARCHAR(100),
    AA_LO_LEVEL_NAME VARCHAR(100),
    AA_LO_LEVEL2_NAME VARCHAR(100),
    AA_LO_LEVEL3_NAME VARCHAR(100))
AS
declare variable aa_disb_kc_id decimal(18,0);
declare variable aa_curr_kc_id decimal(18,0);
declare variable aa_is_own integer;
declare variable aa_lo_region_id integer;
declare variable ps_is_on_staff integer;
declare variable ps_is_working integer;
declare variable lev_id integer;
declare variable s_level varchar(100);
declare variable jl_id_prior integer;
declare variable ps_from_jl_id integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  jl_id_prior = -1;
  for select aa_k_id, aa_amount_active_end, aa_kolich_cl_active_end, aa_deliq_cl_kolich, aa_deliq_balance,
             aa_disb_kc_id, aa_curr_kc_id, aa_is_own, aa_deliq_days
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_project, :rep_period_kind, 0, 0, 0, 0, 0, 0)
      where aa_amount_active_end <> 0
      into :aa_k_id, :aa_curr_amount_active_end, :aa_curr_kolich_cl_active_end, :aa_curr_deliq_cl_kolich,
           :aa_curr_deliq_balance, :aa_disb_kc_id, :aa_curr_kc_id, :aa_is_own, :aa_deliq_days
  do
  begin
    if (aa_deliq_days <= 7) then
    begin
      aa_curr_deliq_cl_kolich = 0;
      aa_curr_deliq_balance = 0;
    end
    aa_own_amount_active_end = 0;
    aa_own_kolich_cl_active_end = 0;
    aa_own_deliq_cl_kolich = 0;
    aa_own_deliq_balance = 0;

    aa_in_amount_active_end = 0;
    aa_in_kolich_cl_active_end = 0;
    aa_in_deliq_cl_kolich = 0;
    aa_in_deliq_balance = 0;

    aa_out_amount_active_end = 0;
    aa_out_kolich_cl_active_end = 0;
    aa_out_deliq_cl_kolich = 0;
    aa_out_deliq_balance = 0;

    aa_lo_id = aa_curr_kc_id;/* текущий владелец кредита */
    if (aa_is_own = 0) then
    begin
      aa_in_amount_active_end = aa_curr_amount_active_end;
      aa_in_kolich_cl_active_end = aa_curr_kolich_cl_active_end;
      aa_in_deliq_cl_kolich = aa_curr_deliq_cl_kolich;
      aa_in_deliq_balance = aa_curr_deliq_balance;
    end
    else
    begin
      aa_own_amount_active_end = aa_curr_amount_active_end;
      aa_own_kolich_cl_active_end = aa_curr_kolich_cl_active_end;
      aa_own_deliq_cl_kolich = aa_curr_deliq_cl_kolich;
      aa_own_deliq_balance = aa_curr_deliq_balance;
    end

    select c_name_fio
    from clients
    where c_id = :aa_lo_id
    into :aa_disb_lo_fio;

    aa_lo_region_id = null;
    aa_lo_office_name = 'Нет приказа';
    ps_from_jl_id = null;
    ps_is_on_staff = 0;
    select reg_unicode, reg_code || ' ' || reg_name, ps_from_jl_id, ps_is_on_staff, ps_is_working
    from region, hr_pstatus
    where reg_unicode = ps_from_reg_id and
          ps_date_end >= :rep_date_end and
          ps_date_begin <= :rep_date_end and
          ps_from_c_id = :aa_lo_id
    order by ps_is_working desc
    rows 1
    into :aa_lo_region_id, :aa_lo_office_name, :ps_from_jl_id, :ps_is_on_staff, :ps_is_working;

    if (ps_from_jl_id is null) then
    begin
      ps_from_jl_id = 0;
      jl_id_prior = -1;
      aa_lo_jobname = 'Нет приказа';
    end
    else
    if (ps_from_jl_id <> jl_id_prior) then
    begin
      select jl_name
      from joblist
      where jl_id = :ps_from_jl_id
      into :aa_lo_jobname;
      jl_id_prior = :ps_from_jl_id;
    end

    if (ps_is_on_staff is null) then
      ps_is_on_staff_str = 'Нет приказа';
    if (ps_is_on_staff = 0) then
      ps_is_on_staff_str = 'Внештатный';
    if (ps_is_on_staff = 1) then
      ps_is_on_staff_str = 'Штатный';

    if (ps_is_working = 0) then
      ps_is_working_str = 'Уволен';
    if (ps_is_working = 1) then
      ps_is_working_str = 'Работает';

    if (aa_lo_region_id is not null) then
    begin
      lev_id = 0;
      select branch.reg_code || ' ' || branch.reg_name, r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
      from region office, region branch, region r2
      where office.reg_unicode = :aa_lo_region_id and
            branch.reg_unicode = office.parent and
            r2.reg_unicode = branch.parent
      into :aa_lo_branch_name, :lev_id, :s_level;

      if ((lev_id is not null) and
          (lev_id <> 0) and
          (lev_id <> 1000000)) then
      begin
        aa_lo_level_name = s_level;
      end
      else
      begin
        aa_lo_level_name = aa_lo_branch_name;
        lev_id = 0;
      end

      select r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :lev_id and
            r2.reg_unicode = r1.parent
      into :lev_id, :s_level;
      if ((lev_id is not null) and
          (lev_id <> 0) and
          (lev_id <> 1000000)) then
      begin
        aa_lo_level2_name = s_level;
      end
      else
      begin
        aa_lo_level2_name = aa_lo_level_name;
        lev_id = 0;
      end

      select r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
      from region r1, region r2
      where r1.reg_unicode = :lev_id and
            r2.reg_unicode = r1.parent
      into :lev_id, :s_level;
      if ((lev_id is not null) and
          (lev_id <> 0) and
          (lev_id <> 1000000)) then
      begin
        aa_lo_level3_name = s_level;
      end
      else
      begin
        aa_lo_level3_name = aa_lo_level2_name;
        lev_id = 0;
      end
    end
    else
    begin
      aa_lo_branch_name = 'Не определен';
      aa_lo_level_name = 'Не определен';
      aa_lo_level2_name = 'Не определен';
      aa_lo_level3_name = 'Не определен';
    end

    suspend;
    if (aa_is_own = 0) then
    begin
      aa_own_amount_active_end = 0;
      aa_own_kolich_cl_active_end = 0;
      aa_own_deliq_cl_kolich = 0;
      aa_own_deliq_balance = 0;

      aa_in_amount_active_end = 0;
      aa_in_kolich_cl_active_end = 0;
      aa_in_deliq_cl_kolich = 0;
      aa_in_deliq_balance = 0;

      aa_out_amount_active_end = 0;
      aa_out_kolich_cl_active_end = 0;
      aa_out_deliq_cl_kolich = 0;
      aa_out_deliq_balance = 0;

      aa_lo_id = aa_disb_kc_id;/* КС который выдао кредит */
      aa_out_amount_active_end = aa_curr_amount_active_end;
      aa_out_kolich_cl_active_end = aa_curr_kolich_cl_active_end;
      aa_out_deliq_cl_kolich = aa_curr_deliq_cl_kolich;
      aa_out_deliq_balance = aa_curr_deliq_balance;

      aa_own_amount_active_end = aa_curr_amount_active_end;
      aa_own_kolich_cl_active_end = aa_curr_kolich_cl_active_end;
      aa_own_deliq_cl_kolich = aa_curr_deliq_cl_kolich;
      aa_own_deliq_balance = aa_curr_deliq_balance;

      aa_curr_amount_active_end = 0;
      aa_curr_kolich_cl_active_end = 0;
      aa_curr_deliq_cl_kolich = 0;
      aa_curr_deliq_balance = 0;

      select c_name_fio
      from clients
      where c_id = :aa_lo_id
      into :aa_disb_lo_fio;

      aa_lo_region_id = null;
      aa_lo_office_name = 'Нет приказа';
      ps_from_jl_id = null;
      ps_is_on_staff = 0;
      select reg_unicode, reg_code || ' ' || reg_name, ps_from_jl_id, ps_is_on_staff, ps_is_working
      from region, hr_pstatus
      where reg_unicode = ps_from_reg_id and
            ps_date_end >= :rep_date_end and
            ps_date_begin <= :rep_date_end and
            ps_from_c_id = :aa_lo_id
      order by ps_is_working desc
      rows 1
      into :aa_lo_region_id, :aa_lo_office_name, :ps_from_jl_id, :ps_is_on_staff, :ps_is_working;
      if (ps_from_jl_id is null) then
      begin
        ps_from_jl_id = 0;
        jl_id_prior = -1;
        aa_lo_jobname = 'Нет приказа';
      end
      else
      if (ps_from_jl_id <> jl_id_prior) then
      begin
        select jl_name
        from joblist
        where jl_id = :ps_from_jl_id
        into :aa_lo_jobname;
        jl_id_prior = :ps_from_jl_id;
      end

      if (ps_is_on_staff is null) then
        ps_is_on_staff_str = 'Нет приказа';
      if (ps_is_on_staff = 0) then
        ps_is_on_staff_str = 'Внештатный';
      if (ps_is_on_staff = 1) then
        ps_is_on_staff_str = 'Штатный';

      if (ps_is_working = 0) then
        ps_is_working_str = 'Уволен';
      if (ps_is_working = 1) then
        ps_is_working_str = 'Работает';

      if (aa_lo_region_id is not null) then
      begin
        lev_id = 0;
        select branch.reg_code || ' ' || branch.reg_name, r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
        from region office, region branch, region r2
        where office.reg_unicode = :aa_lo_region_id and
              branch.reg_unicode = office.parent and
              r2.reg_unicode = branch.parent
        into :aa_lo_branch_name, :lev_id, :s_level;

        if ((lev_id is not null) and
            (lev_id <> 0) and
            (lev_id <> 1000000)) then
        begin
          aa_lo_level_name = s_level;
        end
        else
        begin
          aa_lo_level_name = aa_lo_branch_name;
          lev_id = 0;
        end

        select r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
        from region r1, region r2
        where r1.reg_unicode = :lev_id and
              r2.reg_unicode = r1.parent
        into :lev_id, :s_level;
        if ((lev_id is not null) and
            (lev_id <> 0) and
            (lev_id <> 1000000)) then
        begin
          aa_lo_level2_name = s_level;
        end
        else
        begin
          aa_lo_level2_name = aa_lo_level_name;
          lev_id = 0;
        end

        select r2.reg_unicode, r2.reg_code || ' ' || r2.reg_name
        from region r1, region r2
        where r1.reg_unicode = :lev_id and
              r2.reg_unicode = r1.parent
        into :lev_id, :s_level;
        if ((lev_id is not null) and
            (lev_id <> 0) and
            (lev_id <> 1000000)) then
        begin
          aa_lo_level3_name = s_level;
        end
        else
        begin
          aa_lo_level3_name = aa_lo_level2_name;
          lev_id = 0;
        end
      end
      else
      begin
        aa_lo_branch_name = 'Не определен';
        aa_lo_level_name = 'Не определен';
        aa_lo_level2_name = 'Не определен';
        aa_lo_level3_name = 'Не определен';
      end

      suspend;
    end
  end

end
