CREATE PROCEDURE MC_CNP_TOKEN_CHECK(
    MCL_ID D_ID,
    ACTION_CODE D_STR_10)
RETURNS (
    RES_CODE D_STR_CODE,
    ERROR_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable k_id d_id;
declare variable mcl_login d_str_20;
begin
    res_code = 'success';
    error_text = '';
    
    select
        mcl.mcl_from_c_id,
        mcl.mcl_login
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
        and mcl.mcl_is_delete = 0
    into
        c_id,
        mcl_login;

    select ks.ks_from_k_id
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
        and kc.kc_from_c_id = :c_id
        and kc.kc_amount_approved > 0
    inner join kontract k on ks.ks_from_k_id = k.k_id
        and k.k_status = 0
    where current_date between ks.ks_date_begin and ks.ks_date_end
        and ks.ks_ol_is_delete = 0
        and ks.ks_state = 0
    into :k_id;


    -- Проверка 1: Договор в процессе
    if (error_text = ''
        and action_code in ('update', 'delete')
        and coalesce(:k_id, 0) > 0
    ) then
    begin
        res_code = 'error';
        error_text = f_format('Имеется кредитный договор в процессе. %1 карты недоступно', iif(action_code = 'update', 'Изменение', 'Удаление'));

        -- Если с выдачей на карту произошли какие-то проблемы, тогда разрешаем изменение карты
        if (exists
            (
                select 1
                from cnp_payment_requests cpr
                where cpr.cpr_from_c_id = :c_id
                    and cpr.cpr_from_k_id = :k_id
                    and cpr.cpr_action = 'pay'
                    and cpr.cpr_status = -2
            )
        ) then
        begin
            res_code = 'success';
            error_text = '';
        end
    end

    -- Проверка 2: Биометрия (только при update)
    if (error_text = ''
        and action_code = 'update'
        and exists
        (
            select 1
            from try_counter_verify tcv
            where tcv.tcv_from_c_id = :c_id
                and tcv.tcv_phone = :mcl_login
                and tcv.tcv_count >= 3
        )
    ) then
    begin
        res_code = 'error';
        error_text = 'Добавление карты временно недоступно. Пожалуйста, повторите попытку через 15 минут.';
    end

    suspend;
end
