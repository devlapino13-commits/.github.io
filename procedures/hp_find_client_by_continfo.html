CREATE PROCEDURE HP_FIND_CLIENT_BY_CONTINFO(
    CURR_C_ID D_ID,
    FIND_CONT_VALUE D_STR_CODE,
    IS_PHYS_FACE D_SMALLINT = -1)
RETURNS (
    C_ID D_ID,
    C_BIRTH_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    C_ALTERNAT_FIO D_STR_100,
    C_PASS_NO D_STR_CODE,
    C_PASS_DATE D_DATE,
    C_INN D_STR_CODE,
    C_EXT_INN D_STR_CODE,
    C_PINCODE D_STR_CODE)
AS
declare variable clt_id d_integer;
declare variable hash_code d_str_100;
declare variable is_new_contdata d_smallint;
begin
  if (curr_c_id is null) then
    curr_c_id = 0;

  is_new_contdata = f_get_const('new_contdata');
  is_new_contdata = 1; -- процедуру переводим на новый механизм, Sanjari, 21.12.2019

  if (is_new_contdata = 0) then
  begin
    hash_code = hash(lower(:find_cont_value));

    for select c_id, c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, c_inn, c_ext_inn, c_pincode
    from objectprops, objectvalues, refdata, clients
    where cast(opr_object_kind as integer) = 0 and
          cast(opr_object_id as decimal(18,0)) <> :curr_c_id and
          cast(opr_object_id as decimal(18,0)) <> 0 and
          opr_object_id = c_id and
          opr_subject_kind = -1 and
          opr_subject_id = rd_id and
          opr_id = ovs_from_opr_id and
          hash(lower(ovs_value)) = :hash_code and
          ovs_value = :find_cont_value and
          rd_main_code = 'ContactInfo' and
          opr_actual_ovs_id = ovs_id and
          (:is_phys_face = -1 or c_type = :is_phys_face)
     into c_id, c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, c_inn, c_ext_inn, c_pincode
     do
       suspend;
  end else
  begin
    clt_id = f_cont_id(find_cont_value, 'null');
    if (clt_id is not null) then
    begin
      for select distinct cda_from_c_id
      from contdata
      where cda_from_clt_id = :clt_id and
            cda_is_delete = 0 and
            cda_from_c_id <> 0 and
            cda_from_c_id <> :curr_c_id
      into c_id


      do
      begin

      if(f_cont_info('clients', :c_id, 'cell_main', 'ok') = '') then continue;

        for select c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, c_inn, c_ext_inn, c_pincode
        from clients
        where c_id = :c_id and
              (:is_phys_face = -1 or c_type = :is_phys_face)
        into c_birth_date, c_name_fio, c_alternat_fio, c_pass_no, c_pass_date, c_inn, c_ext_inn, c_pincode
        do
          Suspend;
      end
    end
  end
end
