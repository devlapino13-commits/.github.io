CREATE PROCEDURE HP_BKI_JSON_GET_LIST(
    JSA D_INTEGER,
    IS_OBJ D_SMALLINT)
RETURNS (
    JSO D_INTEGER,
    VAL D_STR_8100,
    I D_INTEGER)
AS
declare variable cnt d_integer;
begin
  -- is_obj = 0 - значение, 1 - объеке, 2 - массив
  if (jsa is null) then
    exit;

  i = 0;
  cnt = json.count_get(jsa);

  while (i < cnt)
  do
  begin
    if (is_obj = 1) then
      jso = json.arr_get_obj(jsa, i);
    else if (is_obj = 2) then
      jso = json.arr_get_arr(jsa, i);
    else
      val = json.arr_get_val(jsa, i);

    i = i + 1;
    suspend;
  end
end
