CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_FINISH(
    EXA_ID D_INTEGER)
RETURNS (
    RES_STATUS D_SMALLINT)
AS
declare variable PR_ID D_INTEGER;
declare variable UN_ID D_ID;
begin
  res_status = 0;

  select exa_from_pr_id, exa_from_un_id
  from expressapp
  where exa_id = :exa_id
  into pr_id, un_id;

  if (not exists(select * from product
    where pr_id = :pr_id
    and pr_code containing '.tms.')) then
  begin
    suspend;
    exit;
  end 

  execute procedure hp_extinfo_insert('tms', 'expressapp', :exa_id, 0, :un_id);

  suspend;
end
