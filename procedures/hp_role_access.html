CREATE PROCEDURE HP_ROLE_ACCESS(
    ID_1 VARCHAR(100),
    ID_2 VARCHAR(100))
RETURNS (
    RES_TEXT VARCHAR(1024),
    RES_KIND VARCHAR(20),
    RES_CHECKED_1 INTEGER,
    RES_CHECKED_2 INTEGER)
AS
declare variable UNIQTAB_ROLENAME integer;
declare variable UNIQTAB_MENULIST integer;
declare variable UNIQTAB_PERMACTION integer;
declare variable ML_ID integer;
declare variable ML_PARENT_ID integer;
declare variable ML_CAPTION varchar(255);
declare variable RES_ID integer;
declare variable RN_ID_1 decimal(18,0);
declare variable RN_ID_2 decimal(18,0);
begin
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename; -- 1
  select ukt_id
  from hp_get_uniqtable_id('Menulist')
  into :uniqtab_menulist; -- 2
  select ukt_id
  from hp_get_uniqtable_id('Permaction')
  into :uniqtab_permaction; -- 3

  if (id_1 similar to '[-]?[[:DIGIT:]]+') then
    rn_id_1 = id_1;
  else
    select rn_id
    from rolename
    where rn_name = :id_1
    into :rn_id_1;

  if (id_2 similar to '[-]?[[:DIGIT:]]+') then
    rn_id_2 = id_2;
  else
    select rn_id
    from rolename
    where rn_name = :id_2
    into :rn_id_2;

  res_kind = 'menu';
  for select ml_id, ml_parent_id, ml_caption
      from menulist
      into :ml_id, ml_parent_id, ml_caption
  do
  begin
    res_text = ml_caption;
    res_id = ml_id;
    while (ml_parent_id is not null) do
    begin
      select ml_id, ml_parent_id, ml_caption
      from menulist
      where ml_id = :ml_parent_id
      into :ml_id, ml_parent_id, ml_caption;
      res_text = ml_caption || ' -> ' || res_text;
    end

    res_checked_1 = 0;
    res_checked_2 = 0;

    if (exists(select *
               from permissions
               where pm_object_kind = 1 and
                     pm_object_id = :rn_id_1 and
                     pm_subject_kind = 2 and
                     pm_subject_id = :res_id and
                     pm_is_delete = 0 and
                     pm_access_kind >= 0 and
                     (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                     (coalesce(pm_date_end, '31.12.2999') >= current_date))) then
      res_checked_1 = 1;

    if (exists(select *
               from permissions
               where pm_object_kind = 1 and
                     pm_object_id = :rn_id_2 and
                     pm_subject_kind = 2 and
                     pm_subject_id = :res_id and
                     pm_is_delete = 0 and
                     pm_access_kind >= 0 and
                     (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                     (coalesce(pm_date_end, '31.12.2999') >= current_date))) then
      res_checked_2 = 1;

    suspend;
  end

  res_kind = 'actions';
  for select pa_id, pa_parent_id, pa_name
      from permactions
      where pa_id <> pa_id -- Временно отключаем проверку
      into :ml_id, ml_parent_id, ml_caption
  do
  begin
    res_text = ml_caption;
    res_id = ml_id;
    while (ml_parent_id is not null) do
    begin
      select pa_id, pa_parent_id, pa_name
      from permactions
      where pa_id = :ml_parent_id
      into :ml_id, ml_parent_id, ml_caption;
      res_text = ml_caption || ' -> ' || res_text;
    end

    res_checked_1 = 0;
    res_checked_2 = 0;

    if (exists(select *
               from permissions
               where pm_object_kind = 1 and
                     pm_object_id = :rn_id_1 and
                     pm_subject_kind = 3 and
                     pm_subject_id = :res_id and
                     pm_is_delete = 0 and
                     pm_access_kind >= 0 and
                     (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                     (coalesce(pm_date_end, '31.12.2999') >= current_date))) then
      res_checked_1 = 1;

    if (exists(select *
               from permissions
               where pm_object_kind = 1 and
                     pm_object_id = :rn_id_2 and
                     pm_subject_kind = 3 and
                     pm_subject_id = :res_id and
                     pm_is_delete = 0 and
                     pm_access_kind >= 0 and
                     (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
                     (coalesce(pm_date_end, '31.12.2999') >= current_date))) then
      res_checked_2 = 1;

    suspend;
  end

end
