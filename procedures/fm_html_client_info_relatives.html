CREATE PROCEDURE FM_HTML_CLIENT_INFO_RELATIVES(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable rel_fio d_str_100;
declare variable rel_name d_str_100;
declare variable rel_phone d_str_100;
declare variable cont_fio d_str_100;
declare variable cont_phone d_str_255;
begin
  rel_fio = '';
  rel_name = '';
  rel_phone = '';
  res_text = '';
  res_text = BLOB_APPEND(res_text, f_format('<h2 class="green_text">%1</h2>', 'Родственники'));

  res_text = BLOB_APPEND(res_text, '<table width="100%">
         <col width="50%">
         <col width="25%">
         <col width="25%">
         <tr>' || f_format('<th>%1</th>', 'ФИО') || f_format('<th>%1</th>', 'Связь') || f_format('<th>%1</th>', 'Контакт') || '</tr>');
  for select c_name_fio, rel_name, c_plastcard_no
      from hp_client_relation_list(:c_id, 0)
      into :rel_fio, rel_name, rel_phone
  do
  begin
    res_text = BLOB_APPEND(res_text, f_format('<tr><td>%1</td><td>%2</td><td>%3</td></tr>', coalesce(rel_fio, ''), coalesce(rel_name, ''), f_uri_phone(rel_phone, 0, c_id, 1)));
  end
  res_text = BLOB_APPEND(res_text, '</table>');

  res_text = BLOB_APPEND(res_text, f_format('<h2 class="green_text">%1</h2>', 'Дополнительные контакты'));
  cont_fio = '';
  cont_phone = '';
  res_text = BLOB_APPEND(res_text, '<table width="100%">
         <col width="50%">
         <col width="50%">
         <tr>' || f_format('<th>%1</th>', 'ФИО') || f_format('<th>%1</th>', 'Связь') || '</tr>');
  for select cc_name, cc_cell_no
      from hp_contact_data(:c_id, 0)
      into :cont_fio, cont_phone
  do
  begin
    res_text = BLOB_APPEND(res_text, f_format('<tr><td>%1</td><td>%2</a></td></tr>', coalesce(:cont_fio, ''), coalesce(:cont_phone, '')));
  end
  res_text = BLOB_APPEND(res_text, '</table>');
  suspend;
  res_text = res_text||'';
end
