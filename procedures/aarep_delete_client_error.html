CREATE PROCEDURE AAREP_DELETE_CLIENT_ERROR
RETURNS (
    CDB_FROM_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_CODE,
    C_INN D_STR_20,
    C_EXT_INN D_STR_20,
    CDB_ERR_MESSAGE D_STR_255)
AS
begin
    -- RFCRU-5616
    for select
        cdb.cdb_from_c_id, 
        cdb.c_name_fio, 
        cdb.c_birth_date, 
        cdb.c_pass_no, 
        cdb.c_inn, 
        cdb.c_ext_inn, 
        cdb.cdb_err_message
    from client_delete_buffer cdb
    where cdb.cdb_is_delete_error = 1
    into 
        :cdb_from_c_id, 
        :c_name_fio, 
        :c_birth_date, 
        :c_pass_no, 
        :c_inn, 
        :c_ext_inn,
        :cdb_err_message
    do
        suspend;

end
