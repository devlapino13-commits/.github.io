CREATE PROCEDURE AAREP_STAT_ACTIVE_TEMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    ST_PERIOD_KIND INTEGER,
    ST_AMOUNT_KIND INTEGER,
    ST_CLIENT_KOLICH INTEGER,
    ST_DISB DECIMAL(18,4),
    ST_PERC DECIMAL(18,4))
AS
declare variable mon float;
declare variable disb decimal(18,4);
declare variable k_disb decimal(18,4);
declare variable kc_disb decimal(18,4);
begin
-- knclient_fix

  st_client_kolich = 0;
  st_disb = 0;
  st_perc = 0;
  st_amount_kind = 1;
  st_period_kind = 1;
  suspend;
  st_period_kind = 2;
  suspend;
  st_period_kind = 3;
  suspend;
  st_period_kind = 4;
  suspend;
  st_period_kind = 5;
  suspend;

  st_amount_kind = 2;
  st_period_kind = 1;
  suspend;
  st_period_kind = 2;
  suspend;
  st_period_kind = 3;
  suspend;
  st_period_kind = 4;
  suspend;
  st_period_kind = 5;
  suspend;

  st_amount_kind = 3;
  st_period_kind = 1;
  suspend;
  st_period_kind = 2;
  suspend;
  st_period_kind = 3;
  suspend;
  st_period_kind = 4;
  suspend;
  st_period_kind = 5;
  suspend;

  st_amount_kind = 4;
  st_period_kind = 1;
  suspend;
  st_period_kind = 2;
  suspend;
  st_period_kind = 3;
  suspend;
  st_period_kind = 4;
  suspend;
  st_period_kind = 5;
  suspend;

  for select k_period, kc_amount_approved * k_index_curr_rate, k_amount_approved * k_index_curr_rate, aarep_k_balance
      from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, 0, '000', 3)
      left join kontract on (aarep_k_id = k_id)
      left join(knclient left join clients on (kc_from_c_id = c_id)) on (kc_from_k_id = aarep_k_id)
      where kc_amount_approved <> 0

      into :mon, :kc_disb, :k_disb, :st_disb
  do
  begin
    st_client_kolich = 1;
    if (kc_disb = k_disb) then
      disb = :st_disb;
    else
    begin
      if (k_disb <> 0) then
        disb = st_disb * (kc_disb / k_disb);
      else
      begin
        disb = 0;
        st_client_kolich = 0;
      end
    end

    if (mon <= 3) then
      st_period_kind = 1;
    else
    if (mon <= 6) then
      st_period_kind = 2;
    else
    if (mon <= 12) then
      st_period_kind = 3;
    else
    if (mon <= 36) then
      st_period_kind = 4;
    else
      st_period_kind = 5;

    if (kc_disb <= 10000) then
      st_amount_kind = 1;
    else
    if (kc_disb <= 25000) then
      st_amount_kind = 2;
    else
    if (kc_disb <= 50000) then
      st_amount_kind = 3;
    else
      st_amount_kind = 4;

    st_disb = disb;
    suspend;
  end

end
