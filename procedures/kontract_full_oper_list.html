CREATE PROCEDURE KONTRACT_FULL_OPER_LIST(
    KONT_ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    OL_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_MEMO VARCHAR(1024),
    OL_DATETIME TIMESTAMP,
    UN_NAME VARCHAR(30),
    SUM_OF_OK_OS DECIMAL(18,4),
    SUM_OF_OK_PERC DECIMAL(18,4),
    SUM_OF_OK_FINE DECIMAL(18,4),
    OK_PAY_TYPE INTEGER,
    SUM_OF_OK_KOMMIS DECIMAL(18,4),
    SUM_OF_OK_TAX DECIMAL(18,4),
    SUM_OF_DEPO DECIMAL(18,4),
    OK_TYPE INTEGER,
    OK_PAY_TYPE_TEXT VARCHAR(20),
    OK_TYPE_TEXT VARCHAR(100),
    SUM_OF_ALL DECIMAL(18,4),
    OL_ID DECIMAL(18,0),
    OL_DATE_OPER_DAY DATE,
    OL_PARENT_ID DECIMAL(18,0))
AS
declare variable lawclientexpense_id integer;
begin
  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'OperExtraKind.LawClientExpense'
  into :lawclientexpense_id;

  for select ol_id, ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, sum(ok_sh_os + ok_os), sum(ok_sh_perc + ok_perc),
             sum(ok_fine), sum(ok_tax), ok_pay_type, sum(ok_kommis + ok_kommis_adv), cast(0 as decimal(18,4)), ok_type,
             sum(ok_sh_os + ok_os + ok_sh_perc + ok_perc + ok_fine + ok_tax + ok_kommis + ok_kommis_adv),
             ol_date_oper_day, ot_name, ol_parent_id
      from operkred, kroper, operlist, username
      where (ok_from_ol_id = operlist.ol_id) and
            (ol_user_id = un_id) and
            (is_delete = 0) and
            ok_from_k_id = :kont_id and
            (ok_type <> 3 and
            ok_type <> 1) and
            ol_date <= :dd and
            ok_type = ot_id and
            ok_from_c_id = 0 and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
      group by ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, ok_pay_type, ok_type, ol_id, ol_date_oper_day, ot_name, ol_parent_id
      union
      select ol_id, ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, cast(0 as decimal(18,4)),
             cast(0 as decimal(18,4)), cast(0 as decimal(18,4)), cast(0 as decimal(18,4)), cast(3 as integer),
             cast(0 as decimal(18,4)), sum(okd_summa_minus - okd_summa_plus), cast(4 as integer),
             sum(okd_summa_minus - okd_summa_plus), ol_date_oper_day, cast('' as varchar(50)), ol_parent_id
      from operkreddepo, operlist, username
      where (okd_from_ol_id = operlist.ol_id) and
            (ol_user_id = un_id) and
            (is_delete = 0) and
            okd_from_k_id = :kont_id and
            ol_date <= :dd
      group by ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, ol_id, ol_date_oper_day, ol_parent_id
      union
      select ol_id, ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, sum(oko_os), sum(oko_perc), sum(oko_other),
             max(0), max(1), sum(0), cast(0 as decimal(18,4)), oko_type, sum(oko_os + oko_perc + oko_other),
             ol_date_oper_day, ot_name, ol_parent_id
      from operkroff, kroper, operlist, username
      where (oko_from_ol_id = operlist.ol_id) and
            (ol_user_id = un_id) and
            (is_delete = 0) and
            oko_from_k_id = :kont_id and
            oko_type in (4, 9) and
            ol_date <= :dd and
            oko_type = ot_id
      group by ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, oko_type, ol_id, ol_date_oper_day, ot_name, ol_parent_id
      union
      select ol_id, ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, 0, 0, 0, 0, 0,
             sum(oe.ox_amount_plus - oe.ox_amount_minus), 0, 4, sum(oe.ox_amount_plus - oe.ox_amount_minus),
             ol_date_oper_day,
             iif(sum(oe.ox_amount_plus - oe.ox_amount_minus) >= 0, 'Судеб. взыскание', 'Судеб. издержки') as ok_type_text,
             ol.ol_parent_id
      from operextra oe, operlist ol, username un
      where oe.ox_object_kind = 9 and
            oe.ox_object_id = :kont_id and
            oe.ox_from_rd_id = :lawclientexpense_id and
            ol.ol_id = oe.ox_from_ol_id and
            ol.is_delete = 0 and
            ol_user_id = un_id and
            ol.ol_date <= :dd
      group by ol_date, ol_doc_no, ol_memo, ol_datetime, un_name, ol_id, ol_date_oper_day, ol_parent_id
      order by 2, 1, 3
      into :ol_id, :ol_date, :ol_doc_no, :ol_memo, :ol_datetime, :un_name,

           :sum_of_ok_os, :sum_of_ok_perc, :sum_of_ok_fine, :sum_of_ok_tax,

           :ok_pay_type, :sum_of_ok_kommis,

           :sum_of_depo, :ok_type,

           :sum_of_all, :ol_date_oper_day, :ok_type_text, :ol_parent_id
  do
  begin
    if (ok_pay_type = 1) then
      ok_pay_type_text = 'Налич.';
    if (ok_pay_type = 2) then
      ok_pay_type_text = 'Безнал';
    if (ok_pay_type = 3) then
      ok_pay_type_text = 'Внутр.';

    suspend;
  end
end
