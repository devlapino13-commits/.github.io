CREATE PROCEDURE HP_PRODUCTBLANK_LIST(
    PR_ID D_INTEGER,
    BLN_ID D_INTEGER)
RETURNS (
    PRB_ID D_INTEGER,
    PRB_FROM_PR_ID D_INTEGER,
    PRB_FROM_BLN_ID D_INTEGER,
    PRB_DATE_BEGIN D_DATE,
    PRB_DATE_END D_DATE,
    PRB_ORDER_NO D_INTEGER,
    PRB_FROM_UN_ID D_ID,
    PRB_DATETIME D_DATETIME,
    PR_NAME D_STR_255,
    BLN_NAME D_STR_255,
    UN_NAME D_STR_100)
AS
declare variable pr_memo d_str_255;
declare variable bln_memo d_str_255;
begin
  select pr_name, pr_memo
  from product
  where pr_id = :pr_id
  into pr_name, pr_memo;

  if (pr_memo is null) then
    pr_memo = '';

  pr_name = f_str_join(pr_name, pr_memo, '/');

  select f_str_join(bln_name, bln_memo, '/'), bln_memo
  from blankname
  where bln_id = :bln_id
  into bln_name, bln_memo;

  bln_memo = trim(coalesce(bln_memo, ''));

  for select prb_id, prb_from_pr_id, prb_from_bln_id,
             prb_date_begin,  prb_date_end,
             prb_order_no, prb_from_un_id, prb_datetime
  from productblank
  where prb_from_pr_id = :pr_id and
        prb_from_bln_id = :bln_id
  order by prb_date_begin desc, prb_date_end desc
  into prb_id, prb_from_pr_id, prb_from_bln_id,
       prb_date_begin,  prb_date_end,
       prb_order_no, prb_from_un_id, prb_datetime
  do
  begin
    un_name = '';
    select un_name
    from username
    where un_id = :prb_from_un_id
    into un_name;

    suspend;
  end

end
