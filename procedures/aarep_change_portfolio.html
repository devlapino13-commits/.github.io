CREATE PROCEDURE AAREP_CHANGE_PORTFOLIO(
    REG_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    ID_PARENT ID_TYPE,
    KC_REGION_ID ID_TYPE,
    KC_ID ID_TYPE,
    PARENT_NAME VARCHAR(50),
    KC_REG_NAME STR_100,
    KC_FIO STR_100,
    KC_BALANCE CURR_TYPE,
    KC_PAR7 CURR_TYPE,
    KC_PAR30 CURR_TYPE,
    REP_DATE_MONTH DATE,
    KC_BALANCE_MONTH CURR_TYPE,
    KC_PAR7_MONTH CURR_TYPE,
    KC_PAR30_MONTH CURR_TYPE,
    KC_EMP INTEGER,
    KC_CL INTEGER,
    KC_CL_MONTH INTEGER,
    KC_DISB_SUM_MONTH CURR_TYPE,
    KC_PAR7_PERC PERC_TYPE,
    KC_PAR30_PERC PERC_TYPE)
AS
begin
  rep_date_month = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date) - 1;

  for select id_parent, parent_name, oks_region_id, oks_kc_id, sum(k_kolich_client_approved),
             sum(oks_balance) as kc_balance,
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 7, oks_balance, 0)) as kc_par7,
             sum(iif(cast(:rep_date_end as date) - oks_delinq_date > 30, oks_balance, 0)) as kc_par30
      from oksummary oks, kontract k, region_list(:reg_id, 1, -1, 1)
      where :rep_date_end between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_close_state = 0 and
            oks_from_k_id = k_id and
            oks.oks_region_id = id_list
      group by id_parent, parent_name, oks_region_id, oks_kc_id
      into id_parent, parent_name, kc_region_id, kc_id, kc_cl, kc_balance, kc_par7, kc_par30
  do
  begin
    select sum(oks_balance) as kc_balance,
           sum(iif(cast(:rep_date_month as date) - oks_delinq_date > 7, oks_balance, 0)) as kc_par7,
           sum(iif(cast(:rep_date_month as date) - oks_delinq_date > 30, oks_balance, 0)) as kc_par30
    from oksummary oks
    where :rep_date_month between oks.oks_date_begin and oks.oks_date_end and
          oks.oks_close_state = 0 and
          oks_kc_id = :kc_id and
          oks.oks_region_id = :kc_region_id
    into kc_balance_month, kc_par7_month, kc_par30_month;

    if (kc_balance_month is null) then
      kc_balance_month = 0;
    if (kc_par7_month is null) then
      kc_par7_month = 0;
    if (kc_par30_month is null) then
      kc_par30_month = 0;
    if (kc_balance is null) then
      kc_balance = 0;
    if (kc_par7 is null) then
      kc_par7 = 0;
    if (kc_par30 is null) then
      kc_par30 = 0;

    kc_par7_perc = 0;
    if (kc_par7 <> 0 and
        kc_balance <> 0) then
      kc_par7_perc = kc_par7 / kc_balance;

    kc_par30_perc = 0;
    if (kc_par30 <> 0 and
        kc_balance <> 0) then
      kc_par30_perc = kc_par30 / kc_balance;

    kc_cl_month = null;
    kc_disb_sum_month = null;
    select sum(k_kolich_client_approved), sum(k_amount_approved)
    from kontstate ks, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_month + 1 and :rep_date_end and
          ks_kc_id = :kc_id and
          ks_from_k_id = k_id
    into kc_cl_month, kc_disb_sum_month;
    if (kc_cl_month is null) then
      kc_cl_month = 0;
    if (kc_disb_sum_month is null) then
      kc_disb_sum_month = 0;

    kc_emp = null;
    select c_name_fio, datediff(month, c_hired_date, :rep_date_end)
    from clients cl
    where c_id = :kc_id
    into kc_fio, kc_emp;
    if (kc_emp is null) then
      kc_emp = 0;

    select reg_name
    from region
    where reg_unicode = :kc_region_id
    into kc_reg_name;

    suspend;
  end
end
