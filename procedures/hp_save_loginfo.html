CREATE PROCEDURE HP_SAVE_LOGINFO(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    PRV_ID D_ID,
    LOG_ACTION D_STR_CODE,
    STATUS D_SMALLINT,
    IS_DELETE D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    LGI_ID D_ID)
AS
declare variable action_id d_smallint;
declare variable obj_kind_id d_integer;
declare variable un_name d_str_code;
declare variable stack_list d_str_code;
begin
  -- Процедура логирует данные в общей таблице лога: LogInfo

  if (log_action is null) then
    exception e_default 'log_action is null';

  if (log_action not in ('insert', 'update', 'delete')) then
    exception e_default f_format('Unknown log_action: %1', log_action);

  obj_kind_id = f_objkind(obj_kind, 'id');

  -- Если пользователь не указан, пытаемся определить
  if (un_id is null) then
  begin
    un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN');
    if (coalesce(un_name, '') = '' or un_name = 'OTHER_USERS') then
      un_name = current_user;

    select un_id
    from username
    where un_name = :un_name
    into un_id;

    if (un_id is null) then
    begin
      select trim(call_s.mon$object_name)
      from mon$call_stack call_s, mon$statements stmt
      where call_s.mon$caller_id is null and
            call_s.mon$statement_id = stmt.mon$statement_id and
            stmt.mon$attachment_id = current_connection
      rows 1
      into stack_list;

      if (stack_list is not null) then
      begin
        update logerror
        set lge_datetime_last = current_timestamp,
            lge_count = coalesce(lge_count, 0) + 1
        where lge_object_kind = :obj_kind_id and
              lge_proc_name = :stack_list;
  
        if (row_count = 0) then
        begin
          insert into logerror (lge_object_kind, lge_proc_name)
          values (:obj_kind_id, :stack_list);
        end
      end
    end
  end

  action_id = case log_action
    when 'insert' then 1
    when 'update' then 0
    when 'delete' then -1
  end;

  insert into loginfo (lgi_object_kind, lgi_object_id, lgi_prv_id, lgi_action, lgi_status, lgi_is_delete, lgi_from_un_id)
  values (:obj_kind_id, :obj_id, :prv_id, :action_id, :status, :is_delete, :un_id)
  returning lgi_id
  into :lgi_id;

  suspend;
end
