CREATE PROCEDURE CM_KG_0146_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable rd_id integer;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable cons_price d_currency;
declare variable payed_summa d_currency;
declare variable k_cons_kind d_smallint;
begin
  rd_id = f_get_rd_id('CashTransactionIncom.Consultation');
  res_error_is_critical = 2;
  for select kncl.kc_id, kncl.kc_from_c_id, cl.c_name_fio, kont.k_cons_kind
  from kontract kont
  inner join product pr on pr_id = k_from_pr_id
  inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  where k_id = :k_id
  and pr.pr_code containing '.consultation.'
  and kncl.kc_amount_approved > 0
  into :kc_id, :c_id, :res_info_text, :k_cons_kind
  do
  begin
    -- RFCKG-4902
    if (coalesce(:k_cons_kind, 0) > 0) then
    exit;

    cons_price = (select price from clt_get_price_for_consultation(:kc_id));
    payed_summa = 0;
    select
      sum(ok.ok_amount)
    from operkontract ok
      inner join operlist ol
        on ol.ol_id = ok.ok_from_ol_id
          and ol.is_delete = 0
          and ok.ok_from_k_id = :k_id
          and ok.ok_operation_type = :rd_id
          and ol_document_client = :c_id
    into
      :payed_summa;
    payed_summa = coalesce(payed_summa, 0);
    if (payed_summa = 0) then
    begin
      res_error_text = 'Оплата за консультацию не произведена';
      suspend;
    end
    else if (payed_summa < cons_price) then
    begin
      res_error_text = 'Оплата за консультацию произведена не в полном объеме. Цена за консультацию = '||round(cons_price)||' сом . Оплачено: '||round(payed_summa)||' сом';
      suspend;
    end
    else if (payed_summa > cons_price) then
    begin
      res_error_text = 'Оплата за консультацию произведена в большем объеме. Цена за консультацию = '||round(cons_price)||' сом . Оплачено: '||round(payed_summa)||' сом';
      suspend;
    end
  end
end
