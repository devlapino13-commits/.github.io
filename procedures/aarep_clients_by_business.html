CREATE PROCEDURE AAREP_CLIENTS_BY_BUSINESS(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_AMOUNT_ALL_DISB DECIMAL(18,4),
    AAREP_AMOUNT_ALL_BALANCE DECIMAL(18,4),
    AAREP_KOLICH_ALL_DISB INTEGER,
    AAREP_KOLICH_ALL_BALANCE INTEGER,
    AAREP_AMOUNT_FEMAIL_DISB DECIMAL(18,4),
    AAREP_AMOUNT_FEMAIL_BALANCE DECIMAL(18,4),
    AAREP_KOLICH_FEMAIL_DISB INTEGER,
    AAREP_KOLICH_FEMAIL_BALANCE INTEGER,
    AAREP_BST_ID INTEGER,
    AAREP_BST_CODE VARCHAR(10),
    AAREP_BST_NAME VARCHAR(255))
AS
begin
  /* Procedure Text */
  for select sum(aarep_amount_all_disb), sum(aarep_amount_all_balance), sum(aarep_kolich_all_disb),
             sum(aarep_kolich_all_balance), sum(aarep_amount_femail_disb), sum(aarep_amount_femail_balance),
             sum(aarep_kolich_femail_disb), sum(aarep_kolich_femail_balance), bst_id, bst_code, bst_name
      from bussubtypetarget, aarep_clients_by_business_temp(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency)
      where bst_id = aarep_bst_id
      group by bst_id, bst_code, bst_name
      order by bst_code, bst_name
      into :aarep_amount_all_disb, :aarep_amount_all_balance, :aarep_kolich_all_disb, :aarep_kolich_all_balance,
           :aarep_amount_femail_disb, :aarep_amount_femail_balance, :aarep_kolich_femail_disb,
           :aarep_kolich_femail_balance, :aarep_bst_id, :aarep_bst_code, :aarep_bst_name
  do
  begin
    if (aarep_amount_all_disb is null) then
      aarep_amount_all_disb = 0;
    if (aarep_amount_all_balance is null) then
      aarep_amount_all_balance = 0;
    if (aarep_kolich_all_disb is null) then
      aarep_kolich_all_disb = 0;
    if (aarep_kolich_all_balance is null) then
      aarep_kolich_all_balance = 0;
    if (aarep_amount_femail_disb is null) then
      aarep_amount_femail_disb = 0;
    if (aarep_amount_femail_balance is null) then
      aarep_amount_femail_balance = 0;
    if (aarep_kolich_femail_disb is null) then
      aarep_kolich_femail_disb = 0;
    if (aarep_kolich_femail_balance is null) then
      aarep_kolich_femail_balance = 0;
    suspend;
  end

end
