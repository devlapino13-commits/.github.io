CREATE PROCEDURE HP_EXTINFO_INSERT(
    EIC_CODE D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    FORCE_INSERT D_SMALLINT,
    UN_ID D_ID = null,
    POCKET_ID D_INTEGER = null)
AS
declare variable EIC_ID D_ID;
declare variable OBJ_KIND_STR D_STR_CODE;
declare variable C_ID D_ID;
declare variable EIC_HAS_API D_SMALLINT;
declare variable EIL_DATE D_DATE;
declare variable EIL_ID D_ID;
declare variable EIL_OBJECT_KIND D_INTEGER;
declare variable EIC_INFO_ACTUAL_DAYS D_INTEGER;
begin
  select eic_id, eic_has_api, eic_info_actual_days
  from extinfocfg
  where eic_code = :eic_code
  into eic_id, eic_has_api, eic_info_actual_days;

  if (eic_id is null) then
    exception e_default 'ExtinfoCfg not found: ' || coalesce(eic_code, '<NULL>');

  obj_kind_str = f_objkind(obj_kind, 'name');

  for select kc_from_c_id
      from knclient
      where :obj_kind_str = 'Kontract' and
            kc_from_k_id = :obj_id and
            kc_amount_approved > 0
      union
      select mac_from_c_id
      from miniappclient
      where :obj_kind_str = 'Miniapp' and
            mac_from_ma_id = :obj_id and
            mac_amount > 0
      union
      select c_id
      from clients
      where :obj_kind_str = 'Clients' and
            c_id = :obj_id
      union
      select drl_from_c_id
      from docrecoglist
      where :obj_kind_str = 'DocrecogList' and
            drl_id = :obj_id
      union
      select exa_from_c_id
      from expressapp
      where :obj_kind_str = 'expressapp' and
            exa_id = :obj_id
      into c_id
  do
  begin
    eil_date = null;
    eil_id = null;

    select eil_id, eil_date, eil_object_kind
    from extinfolist
    where eil_from_c_id = :c_id and
          eil_from_eic_id = :eic_id and
          eil_status in (-1, 0, 1, 2) and
          datediff(day, eil_date, current_date) <= :eic_info_actual_days
    order by eil_date desc, eil_create_datetime desc
    rows 1
    into eil_id, eil_date, eil_object_kind;

    if (eil_date is null or
       (eil_date < current_date and force_insert = 1) or
       (eic_code = 'kontur' and obj_kind_str = 'docrecoglist') or
       eic_code in ('tunduk', 'fin', 'inn_gnivc', 'inn_smev', 'salary_check', 'pension_check', 'checkme_info', 'stop_credit')) then
    begin
      insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_date,
                               eil_object_kind, eil_object_id, eil_is_json,
                               eil_from_eip_id, eil_from_un_id)
      values (:c_id, :eic_id, current_date, f_objkind(:obj_kind, 'id'), :obj_id,
              :eic_has_api, :pocket_id, :un_id);
    end
    else
    begin
      if (eil_id is not null and
          f_objkind(eil_object_kind, 'name') is distinct from f_objkind(obj_kind, 'name')) then
      begin
        update extinfolist
        set eil_object_kind = f_objkind(:obj_kind, 'id'),
            eil_object_id = :obj_id
        where eil_id = :eil_id;
      end
    end
  end
end
