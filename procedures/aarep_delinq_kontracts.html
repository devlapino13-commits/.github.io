CREATE PROCEDURE AAREP_DELINQ_KONTRACTS(
    REP_REGION_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(50),
    K_KF_NAME VARCHAR(100),
    PR_NAME VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    C_ADDRESS VARCHAR(255),
    C_PHONE VARCHAR(20),
    BALANCE_KONTRACT DECIMAL(18,4),
    DELINQ_DATE DATE,
    DELINQ_DAYS INTEGER,
    C_AMOUNT_APPROVED DECIMAL(18,4),
    C_OS_BALANCE DECIMAL(18,4),
    C_PERCENT_BALANCE DECIMAL(18,4),
    C_FINE_BALANCE DECIMAL(18,4),
    TOTAL_BALANCE DECIMAL(18,4),
    PAR30_SUM DECIMAL(18,4))
AS
declare variable k_id id_type;
declare variable c_id id_type;
declare variable reg_id id_type;
declare variable os_balance decimal(18,4);
declare variable fine_balance decimal(18,4);
declare variable percent_balance decimal(18,4);
declare variable c_approved_all decimal(18,4);
declare variable c_k_perc curr_type;
begin
-- для всех активных контрактов на дату, вычисляю задолженность
  for select oks_from_k_id, oks_delinq_date, oks_region_id, oks_balance,
             cast(:rep_date as date) - oks_delinq_date as delinq_days
      from oksummary oks, region_list(:rep_region_id, 1, -1, 1)
      where oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end and
            :rep_date > oks_delinq_date and
            (oks_region_id = id_list)
--  and oks_from_k_id = 101010001393900101
      into :k_id, :delinq_date, :reg_id, balance_kontract, delinq_days
  do
  begin
    for select kc_from_c_id, kc_amount_approved
        from knclient kc
        where kc.kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc.kc_closed_ol_id is null
        union
        select kc_from_c_id, kc_amount_approved
        from knclient kc, operlist ol
        where kc.kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              kc.kc_closed_ol_id is not null and
              kc.kc_closed_ol_id = ol_id and
              ol.is_delete = 1
        into c_id, c_amount_approved
    do
    begin

      select k_kontract_no, kal.kal_new_kf_name, pr.pr_name
      from kontract, knaktlist kal, product pr
      where k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            kal.is_delete = 0 and
            :rep_date between kal_date and kal_date_end and
            k_from_pr_id = pr.pr_id
      into k_kontract_no, k_kf_name, pr_name;

      select sum(kc_amount_approved) c_kolich
      from knclient kc
      where kc.kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            kc.kc_closed_ol_id is null
      into c_approved_all;

      select sum(zad_os), sum(sh_calc_perc), sum(sh_calc_fine) sh_fine
      from aarep_make_zad_list_temp(:k_id, :rep_date, 1)
      where sh_date < :rep_date
      into :os_balance, percent_balance, fine_balance;

      c_k_perc = c_amount_approved / c_approved_all;

      c_os_balance = c_k_perc * os_balance;
      c_percent_balance = c_k_perc * percent_balance;
      c_fine_balance = c_k_perc * fine_balance;

      total_balance = c_os_balance + c_percent_balance + c_fine_balance;

      par30_sum = null;
      if (:delinq_days > 29) then
        par30_sum = c_k_perc * (select sum(zad_os + sh_calc_perc + sh_calc_fine) k_balance
                                from aarep_make_zad_list_temp(:k_id, :rep_date, 1)
                                where :rep_date > sh_date + 29);

      if (par30_sum is null) then
        par30_sum = 0;

      select reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;

      select c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_valid(0, :c_id))
      from clients c
      where c_id = :c_id
      into c_name_fio, c_phone;

      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :c_id and
            adc_address_type = 2 and
            adc_from_adl_id = adl_id
      into :c_address;

      suspend;
    end
  end
end
