CREATE PROCEDURE FSROBOT_ON_MAILING_TMP(
    KIND VARCHAR(10))
RETURNS (
    MSG_TYPE INTEGER,
    RES_MAILLIST VARCHAR(255),
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    MSG_LIFETIME_DAYS INTEGER,
    REP_UN_ID DECIMAL(18,0),
    REP_COMMAND_KIND VARCHAR(255),
    RES_C_ID DECIMAL(18,0),
    MSG_SENDER SMALLINT)
AS
declare variable proc_name varchar(255);
declare variable sql_text varchar(2048);
declare variable user_fio varchar(100);
declare variable un_name varchar(30);
declare variable rep_params varchar(255);
declare variable fel_id integer;
declare variable rd_is_active integer;
declare variable obj_kind integer;
declare variable obj_id d_id;
begin
  -- Процедура вызывается из через сервис по графику (ежеминутно)
  -- Параметры берет из таблицы FsRobot_EventList
  -- Процедура готовит список уведомлений для отправки роботом по почте или sms (Event 'DoMailing')
  -- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
  -- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
  -- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS
  
  -- RefData.RD_Memo    - Список получателей
  -- RefData.RD_Tag     - Срок жизни сообщения
  -- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения
  
  -- Rep_Params - входящий параметры в зависимости от команды
  -- Rep_UN_ID - пользователь

  if (kind = 'MAIL') then
    msg_type = 0;
  if (kind = 'SMS') then
    msg_type = 1;
  if (kind = 'PUSH') then
    msg_type = 2;

  for select fel_id, fel_command_kind, fel_params, fel_from_un_id
      from fsrobot_eventlist
      where fel_event_name = 'DoMailing' and
            fel_status = 0
      into :fel_id, :rep_command_kind, :rep_params, :rep_un_id
  do
  begin
    user_fio = '';
    if (rep_command_kind = 'Push_Loan_Status_Change') then
      rep_un_id = 1000000000000;

    if (f_get_db_id() = 81001) then
    begin
        msg_sender = 0;

        select obj_kind, obj_id
        from hp_fsrobot_mailing_get_obj(:rep_command_kind, :rep_params)
        into :obj_kind, :obj_id;
    
        msg_sender = iif(f_get_mcl_project(:obj_kind, :obj_id) = 2, 1, 0);
    end

    if (coalesce(rep_un_id, 0) = 0) then
    begin
      un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN');
      un_name = coalesce(un_name, 'OTHERS');
      if (un_name <> 'OTHERS') then
      begin
        select un_id, c_name_fio
        from clients, username
        where un_name = :un_name and
              un_from_c_id = c_id
        into :rep_un_id, :user_fio;
      end
      else
        exit;
    end
    else
    begin
      select c_name_fio
      from clients, username
      where un_id = :rep_un_id and
            un_from_c_id = c_id
      into :user_fio;
    end

    proc_name = '';
    rep_params = (trim(rep_params));

    for select coalesce(rd_memo, ''), coalesce(rd_tag, 0), coalesce(rd_str_id, ''), rd_is_active
        from refdata
        where rd_code = 'Notify' || :kind || '.' || :rep_command_kind and
              coalesce(rd_str_id, '') <> ''
        into :res_maillist, msg_lifetime_days, :proc_name, rd_is_active
    do
    begin
      if (rd_is_active = 1) then
      begin
        res_maillist = (trim(res_maillist));

        sql_text = 'select RES_SUBJECT, RES_BODY, RES_MAILLIST, RES_C_ID';
        sql_text = sql_text || ' from ' || :proc_name || '(:REP_PARAMS, :REP_EMAIL, :USER_FIO)';
        for execute statement(:sql_text)(rep_params := :rep_params, rep_email := :res_maillist, user_fio := :user_fio)
                into :res_subject, :res_body, :res_maillist, :res_c_id
        do
        begin
          if (coalesce(res_body, '') <> '' and
              coalesce(res_maillist, '') not in ('', '-')) then
          begin
            if (kind = 'push') then
              select res_text
              from fm_json_data(:res_body, :res_subject, 'mailmessages_push', 'fsrobot_msgbox', null, null, null, null)
              into res_body;
            suspend;
          end
        end
      end
      -- Удаляем задачу из списка, она уже исполнена
      delete from fsrobot_eventlist
      where fel_id = :fel_id;
    end
  end

end
