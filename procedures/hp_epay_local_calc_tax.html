CREATE PROCEDURE HP_EPAY_LOCAL_CALC_TAX(
    AG_ID D_INTEGER,
    IS_RESIDENT D_BOOL,
    REP_DATE D_DATE,
    ADD_TEST_AMOUNT D_CURRENCY = null)
RETURNS (
    EPD_AMOUNT D_CURRENCY,
    EPAY_PAYED_YEAR D_CURRENCY,
    EPAY_AMOUNT D_CURRENCY,
    EPAY_TAX_FL D_CURRENCY,
    OGR_TAX_PAYED D_CURRENCY,
    EPAY_NEW_AMOUNT D_CURRENCY,
    EPAY_NEW_TAX D_CURRENCY,
    EPAY_NEW_LGOTA D_CURRENCY,
    EPAY_NEW_BASE D_CURRENCY,
    TAX_LGOTA D_CURRENCY,
    TAX_BASE D_CURRENCY,
    TAX_PERCENT D_CURRENCY,
    TAX_TAX_DIFF D_CURRENCY)
AS
declare variable rep_year_begin d_date;
begin
  if (rep_date is null) then
    rep_date = current_date;

  if (add_test_amount is null) then
    add_test_amount = 0;

  rep_year_begin = cast('01.01.' || extract(year from current_date) as date);

  epd_amount = 0;
  epay_payed_year = 0;
  select sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, current_date) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, current_date)),
         sum(iif(ogr_sign = -1 and ogr_object_kind is null and ogr_ol_date >= :rep_year_begin, ogr_amount + ogr_tax_fl, 0)),
         sum(iif(ogr_sign = -1 and ogr_object_kind is null and ogr_ol_date >= :rep_year_begin, ogr_tax_fl, 0))
  from operagr
  where ogr_from_ag_id = :ag_id and
        ogr_ol_is_active = 1
  into epd_amount, epay_payed_year, ogr_tax_payed;

  if (epd_amount is null) then
    epd_amount = 0;

  epd_amount = epd_amount + add_test_amount;

  if (epay_payed_year is null) then
    epay_payed_year = 0;

  if (ogr_tax_payed is null) then
    ogr_tax_payed = 0;

  tax_tax_diff = 0;
  tax_base = 0;

  epay_amount = 0;
  epay_tax_fl = 0;
  epay_new_amount = 0;
  epay_new_tax = 0;
  tax_lgota = 4000;

  tax_lgota = tax_lgota - epay_payed_year;
  if (tax_lgota < 0) then
    tax_lgota = 0;

  tax_base = epd_amount - tax_lgota;
  if (tax_base < 0) then
    tax_base = 0;

  if (is_resident = 1)
  then tax_percent = 0.13;
  else tax_percent = 0.30;

  if (tax_base < 0) then
    tax_base = 0;

  epay_tax_fl = round(tax_base * tax_percent);
  epay_amount = epd_amount - epay_tax_fl;

  epay_new_lgota = 4000;
  epay_new_base = epd_amount + epay_payed_year - epay_new_lgota;
  if (epay_new_base < 0) then
    epay_new_base = 0;

  epay_new_tax = round(epay_new_base * tax_percent) - ogr_tax_payed;

  epay_new_amount = epd_amount - epay_new_tax;
  if (epay_new_amount < 0) then
    epay_new_amount = 0;

  tax_tax_diff = 0;
  if (epay_tax_fl <> epay_new_tax) then
    tax_tax_diff = 1;

  suspend;
end
