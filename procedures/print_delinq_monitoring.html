CREATE PROCEDURE PRINT_DELINQ_MONITORING(
    KC_ID D_ID)
RETURNS (
    CLIENT_FIO D_STR_100,
    ANALIZ_DATE D_DATE,
    ANALIZ_LO_FIO D_STR_100,
    DELINQ_REASON D_STR_255,
    REPAY_WHEN D_STR_255,
    DELINQ_MEMO D_STR_255,
    REPAY_EXPECTATION D_STR_255,
    KONTRACT_NO D_STR_100,
    LOAN_AMOUNT D_STR_100,
    PR_NAME D_STR_100,
    FOLDER_NAME D_STR_100,
    HAS_ANKETA D_SMALLINT,
    LOAN_TARGET D_STR_255,
    REPAY_REMINDER_DELINQ D_STR_255)
AS
declare variable k_id d_id;
declare variable c_id d_id;
declare variable k_disb_date d_date;
declare variable rd_id d_integer;
declare variable k_fact_close_date d_date;
declare variable delinq_days d_integer;
declare variable acl_id d_id;
declare variable acl_date d_date;
declare variable kf_id d_id;
declare variable obj_kind d_str_100;
declare variable obj_id d_id;
declare variable un_id d_id;
begin
  rd_id = f_get_rd_id('Anketa.Delinq.1');
  un_id = 1000000000001;

  select
    kc_from_c_id,
    k_id,
    k_disb_date,
    c_name_fio,
    coalesce(ol_date, current_date),
    k_kontract_no,
    kc_amount_approved,
    pr_name,
    k_from_kf_id,
    rd_name
  from knclient
    inner join kontract
      on k_id = kc_from_k_id
    inner join clients
      on c_id = kc_from_c_id
    inner join product
      on pr_id = k_from_pr_id
    left join operlist
      on ol_id = k_fact_closed_ol_id
        and operlist.is_delete = 0
    left join refdata
      on rd_id = kc_loan_target
  where kc_id = :kc_id
    and kc_amount_approved > 0
  into
    :c_id,
    :k_id,
    :k_disb_date,
    :client_fio,
    :k_fact_close_date,
    :kontract_no,
    :loan_amount,
    :pr_name,
    :kf_id,
    :loan_target;

  obj_kind = 'KnClient';
  obj_id = kc_id;
  if (not exists(
    select *
    from actionlist
    where acl_object_kind = f_objkind(:obj_kind, 'id')
      and acl_object_id = :obj_id
      and acl_from_rd_id = :rd_id)) then
  begin
    obj_kind = 'Clients';
    obj_id = c_id;
  end

  select
    acl_id,
    acl_date,
    acl_last_datetime,
    c_name_fio
  from actionlist
    inner join clients
      on c_id = acl_respons_c_id
  where acl_object_kind = f_objkind(:obj_kind, 'id')
    and acl_object_id = :obj_id
    and acl_date between :k_disb_date and :k_fact_close_date
    and acl_from_rd_id = :rd_id
    and acl_status = 2
    and acl_from_un_id <> :un_id
  order by acl_id desc
  rows 1
  into
    :acl_id,
    :acl_date,
    :analiz_date,
    :analiz_lo_fio;

  has_anketa = 0;
  if (acl_id > 0) then
  begin
    select
      coalesce(:acl_date - oks_delinq_date, 0)
    from oksummary
    where oks_from_k_id = :k_id
      and :acl_date between oks_date_begin and oks_date_end
      and oks_close_state = 0
    into
      :delinq_days;

    -- Если на дату создания задачи по работе над просрочкой по займу
    -- было 2 дня просрочки, тогда это нужный заем
    -- 21.10.22 Многие контракты с заполненными анкетами по работе с просрочками не отображаются так как разница составляет больше двух дней
    if (delinq_days >= 2) then
    begin
      has_anketa = 1;

      select
        coalesce(kal_new_kf_name, '')
      from knaktlist
      where kal_from_kf_id = :kf_id
        and :analiz_date between kal_date and kal_date_end
        and is_delete = 0
      into
        :folder_name;

      select
        coalesce(res_str_values, '')
      from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'DelinqReport.Reason', :analiz_date, null)
      into
        :delinq_reason;

      select
        coalesce(res_str_values, '')
      from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'DelinqReport.RepayWhen', :analiz_date, null)
      into
        :repay_when;

      select
        coalesce(res_str_values, '')
      from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'DelinqReport.Memo', :analiz_date, null)
      into
        :delinq_memo;
      delinq_memo = replace(delinq_memo, 'Значение: ', '');
      delinq_memo = replace(delinq_memo, 'Значение', '');

      select
        coalesce(res_str_values, '')
      from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'DelinqReport.RepayExpectation', :analiz_date, null)
      into
        :repay_expectation;

      select
        coalesce(res_str_values, '')
      from hp_objectprops_anketa_get(:obj_kind, :obj_id, -1, 'AnketaQuestions.Reminder.Delinq', :analiz_date, null)
      into
        :repay_reminder_delinq;

      suspend;
    end
  end
end
