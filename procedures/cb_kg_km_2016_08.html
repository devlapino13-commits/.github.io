CREATE PROCEDURE CB_KG_KM_2016_08(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    DELINQ_BALANCE_7 CURR_TYPE,
    BALANCE_BEGIN CURR_TYPE,
    ZNACH_PAR7 CURR_TYPE,
    DELINQ_BALANCE_7_BEGIN CURR_TYPE,
    WOFF_OS CURR_TYPE,
    DINAMIKA_PAR7 CURR_TYPE,
    KOL_KC INTEGER,
    TOTAL_BONUS CURR_TYPE,
    AVG_BONUS_KC CURR_TYPE,
    BONUS_FACTOR INTEGER,
    KOL_SSVZ INTEGER,
    FIO_SSVZ VARCHAR(1024),
    REG_NAME VARCHAR(100),
    BONUS_KM INTEGER,
    BONUS_FACTOR1 INTEGER,
    BONUS_FACTOR2 INTEGER,
    AVG_ZARABOTOK_KC CURR_TYPE,
    TOTAL_BONUS_OKLAD DECIMAL(18,4))
AS
declare variable rep_date_begin date;
declare variable reg_id integer;
begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end);

  for select oks_region_id, sum(iif(:rep_date_end - oks_delinq_date > 7, oks_balance, 0))
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where oks_region_id = id_list and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            oks_close_state = 0 
      group by oks_region_id
      into reg_id, delinq_balance_7
  do
  begin
--  Балан КП и баланс PAR7 на начало
    balance_begin = null;
    delinq_balance_7_begin = null;
    select sum(oks_balance), sum(iif(:rep_date_begin - oks_delinq_date > 7, oks_balance, 0))
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_begin between oks_date_begin and oks_date_end and
          oks_close_state = 0 
    into balance_begin, delinq_balance_7_begin;
    balance_begin = coalesce(balance_begin, 0);
    delinq_balance_7_begin = coalesce(delinq_balance_7_begin, 0);
--  Значение PAR7* в % от КП
    if (balance_begin = 0 and
        delinq_balance_7 > 0) then
      znach_par7 = 100;
    else
    if (delinq_balance_7 = 0) then
      znach_par7 = 0;
    else
      znach_par7 = cast(delinq_balance_7 as double precision) / balance_begin * 100;
--  Основная сумма списанных кредитов
    woff_os = null;
    select -sum(ok_os)
    from operkred
    where ok_ol_reg_id = :reg_id and
          ok_ol_is_active = 1 and
          ok_ol_date > :rep_date_begin and
          ok_ol_date <= :rep_date_end and
          ok_type = 5
    into woff_os;
    woff_os = coalesce(woff_os, 0);
--  динамики баланса PAR7
    if (delinq_balance_7_begin > 0) then
    begin
--      dinamika_par7 = (delinq_balance_7_begin - delinq_balance_7 - woff_os) / delinq_balance_7_begin * 100;

      dinamika_par7 = (delinq_balance_7 - delinq_balance_7_begin + woff_os) / delinq_balance_7_begin * 100;
    end
    else
    if (delinq_balance_7 + woff_os > 0) then
      dinamika_par7 = -100;
    else
      dinamika_par7 = 0;
--  Средняя сумма бонусов КС
    kol_kc = 0;
    total_bonus = 0;
    total_bonus_oklad = 0;
    select count(*), sum(aarep_bonus_total_to_pay), sum(aarep_bonus_total)
    from cb_kg_lo_from_km(:reg_id, :rep_date_end)
    into kol_kc, total_bonus, -- бонус КС, после вычета оклада
         total_bonus_oklad -- бонус КС до вычета оклада
    ;

    total_bonus = coalesce(total_bonus, 0);
    total_bonus_oklad = coalesce(total_bonus_oklad, 0);
    if (kol_kc > 0) then
    begin
      avg_bonus_kc = cast(total_bonus as double precision) / kol_kc;
      avg_zarabotok_kc = cast(total_bonus_oklad as double precision) / kol_kc;
    end
    else
    begin
      avg_bonus_kc = 0;
      avg_zarabotok_kc = 0;
    end
--  Бонус фактор
    /*
    bonus_factor = case
                     when znach_par7 <= 1 then 150
                     when znach_par7 <= 3 and dinamika_par7 > 10 then 130
                     when znach_par7 <= 3 and dinamika_par7 > 5 and dinamika_par7 <= 10 then 110
                     when znach_par7 <= 3 and dinamika_par7 > 0 and dinamika_par7 <= 5 then 90
                     when znach_par7 <= 3 and dinamika_par7 = 0 then 70
                     when znach_par7 <= 3 and dinamika_par7 < 0 and dinamika_par7 >= -3 then 50
                     else 0
                   end;
    */
    bonus_factor1 = case
                      when znach_par7 = 0 then 120
                      when znach_par7 <= 0.5 then 100
                      when znach_par7 <= 1 then 90
                      when znach_par7 <= 1.5 then 50
                      else 0
                    end;

    bonus_factor2 = 0;
    if (rep_date_end <= cast('31.12.2016' as date)
--        and znach_par7 > 1.5 -- убран по письму Отабека от 06.12.2016
        ) then
    begin
      dinamika_par7 = -dinamika_par7;
      bonus_factor2 = case
                        when dinamika_par7 > 5 then 80
                        when dinamika_par7 >= 1 and dinamika_par7 <= 5 then 50
                        when dinamika_par7 > 0 and dinamika_par7 < 1 then 30
                        else 0
                      end;
      dinamika_par7 = -dinamika_par7;
    end

    if (bonus_factor1 > bonus_factor2) then
      bonus_factor = bonus_factor1;
    else
      bonus_factor = bonus_factor2;

    kol_ssvz = 0;
    fio_ssvz = '';
    select count(*), list(c_name_fio, '; ')
    from hr_pstatus, joblist, jobkind, clients
    where ps_status = 1 and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code = 'JobSystem.Region_KM' and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_reg_id = :reg_id and
          ps_from_c_id = c_id and
          (ps_date_begin <= dateadd(month, -3, :rep_date_end) or
          (select min(hr1.ps_date_begin)
           from hr_pstatus hr1
           where hr1.ps_is_working = 1 and
                 hr1.ps_status = 1 and
                 hr1.ps_from_c_id = c_id) <= dateadd(month, -3, :rep_date_end)) and
          coalesce((select sawt_id
                    from hr_voctravel, hr_orders, sa_worktype
                    where hrv_from_ho_id = ho_id and
                          ho_order_type = sawt_id and
                          hrv_from_c_id = ps_from_c_id and
                          :rep_date_end between hrv_date_begin and hrv_date_end_fact
                    order by hrv_date_begin desc
                    rows 1), 0) not in (16, 17, 18, 8) -- Декретников не включает
    into kol_ssvz, fio_ssvz;
    if (kol_ssvz is null) then
      kol_ssvz = 0;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :reg_id
    into reg_name;

    if (rep_date_end < cast('01.09.2016' as date)) then
    begin
      bonus_km = avg_bonus_kc * bonus_factor / 100;
    end
    else
    begin
      bonus_km = (cast(total_bonus_oklad as double precision) * 15 * bonus_factor) / 10000;
    end

    if (bonus_km < 0) then
      bonus_km = 0;

    suspend;
  end
end
