CREATE PROCEDURE HP_PARSE_JSON_GRS(
    JSON_TEXT D_BLOB_TEXT)
RETURNS (
    VOIDSTATUS D_INTEGER,
    SURNAME D_STR_100,
    NAME D_STR_100,
    PATRONYMIC D_STR_100,
    FAMILY_STATUS D_INTEGER,
    DATE_OF_BIRTH D_DATE,
    PASSPORT_SERIES D_STR_10,
    PASSPORT_NUMBER D_STR_20,
    PIN D_STR_20,
    ISSUED_DATE D_DATE,
    PASSPORT_AUTHORITY D_STR_64,
    PASSPORT_AUTHORITY_CODE D_STR_64,
    EXPIRED_DATE D_DATE,
    GENDER D_STR_SYM,
    COUNTRY_TXT D_BLOB_TEXT,
    POST D_INTEGER,
    PARENT_ID D_ID,
    PL_STREET_ID D_ID,
    ADDRESS_STR D_BLOB_TEXT,
    ADDRESS_BUILDING D_BLOB_TEXT,
    ADDRESS_APARTMENT D_BLOB_TEXT,
    ADDRESS_ARRAY D_BLOB_TEXT,
    RD_MARITAL_STATUS D_INTEGER,
    DOC_TYPE D_INTEGER)
AS
declare variable PL_REGION_ID D_ID;
declare variable PL_DISTRICT_ID D_ID;
declare variable PL_AYMAK_ID D_ID;
declare variable PL_VILLAGE_ID D_ID;
declare variable PL_HOUSE_ID D_ID;
declare variable PL_FLAT_ID D_ID;
declare variable COUNTRY_ID D_STR_20;
declare variable REGION_ID D_STR_20;
declare variable DISTRICT_ID D_STR_20;
declare variable AYMAK_ID D_STR_20;
declare variable VILLAGE_ID D_STR_20;
declare variable STREET_ID D_STR_20;
declare variable HOUSE_ID D_STR_20;
declare variable FLAT_ID D_STR_20;
declare variable HOUSE_TXT D_STR_CODE;
declare variable FLAT_TXT D_STR_CODE;
declare variable POST_STR D_STR_CODE;
declare variable REGION_TXT D_BLOB_TEXT;
declare variable DISTRICT_TXT D_BLOB_TEXT;
declare variable AYMAK_TXT D_BLOB_TEXT;
declare variable VILLAGE_TXT D_BLOB_TEXT;
declare variable STREET_TXT D_BLOB_TEXT;
declare variable FAMILY_STATUS_TMP D_INTEGER;
declare variable MARITAL_STATUS D_STR_100;
declare variable DATE_OF_BIRTH_STR D_STR_20;
declare variable ADDRESSREGION D_BLOB_TEXT;
declare variable PL_TYPE D_INTEGER;
declare variable UN_VIRT D_ID;
begin
  un_virt = 1000000000000;
  rd_marital_status = 0;
  voidstatus = f_json_str_get_value(:json_text, 'voidStatus', 1); -- Актуальность паспорта (не просрочен ли)
  pin = f_json_str_get_value(:json_text, 'pin', ''); -- Персональный идентификационный номер
  surname = cast(f_json_blob_get_value(:json_text, 'surname', '') as d_blob_text); -- Фамилия
  name = cast(f_json_blob_get_value(:json_text, 'name', '') as d_blob_text); -- Имя
  patronymic = cast(f_json_blob_get_value(:json_text, 'patronymic', '') as d_blob_text); -- Отчество
  family_status_tmp = f_json_str_get_value(:json_text, 'familyStatus', 4); -- Семейное положение из паспорта - Возможные значения (0 - Состоит в браке, 1 - Не состоит в браке, 2 - Разведен(а) )
  marital_status = f_json_str_get_value(:json_text, 'maritalStatus', ''); -- Семейное положение из ЗАГС - Возм. строк. знач., одно из (Состоит в браке, В браке не состоял(а), Разведен(а), Вдовец/вдова, null)
  gender = f_json_str_get_value(:json_text, 'gender', ''); -- Пол. Возможные значения "M" - муж и "F" - жен
  date_of_birth_str = f_json_str_get_value(:json_text, 'dateOfBirth', ''); -- Дата рождения
  passport_series = f_json_str_get_value(:json_text, 'passportSeries', ''); -- Серия паспорта, пример "ID"
  passport_number = f_json_str_get_value(:json_text, 'passportNumber', ''); --  Номер паспорта, пример "1501045"
  issued_date = cast(left(f_json_str_get_value(:json_text, 'issuedDate', ''), 10) as date); -- Дата выдачи паспорта
  passport_authority = cast(f_json_blob_get_value(:json_text, 'passportAuthority', '') as d_blob_text); -- Орган выдавший паспорт
  passport_authority_code = f_json_str_get_value(:json_text, 'passportAuthorityCode', ''); -- Код органа выдачи паспорта (описение и поле отсутствует в документации)
  expired_date = cast(left(f_json_str_get_value(:json_text, 'expiredDate', ''), 10) as date); -- Дата окончания срока действия паспорта
  date_of_birth = left(date_of_birth_str, 10);
  doc_type = case
                when passport_series containing 'AN' then 8
                when passport_series containing 'ID' then 296
                when passport_series containing 'IK' then 372
                else 0
             end;
  -----------------------------------------------------------------------<
  -- address ----------------------------------------------------------- >
  address_str = cast(f_json_blob_get_value(:json_text, 'message', '') as d_blob_text); -- Полный адрес из АСБ(Адресно-Справочное Бюро) - при отсутствии в АБС выдается сообщение (Ошибка: адрес регистрации не найден)
  addressregion = cast(f_json_blob_get_value(:json_text, 'addressRegion', '') as d_blob_text);
  address_building = cast(f_json_blob_get_value(:json_text, 'addressBuilding', '') as d_blob_text);
  address_apartment = cast(f_json_blob_get_value(:json_text, 'addressApartment', '') as d_blob_text);
  -- RFCKG-5004
  if (coalesce(:marital_status, '') <> '') then
  begin
    family_status = case :marital_status
                      when 'Состоит в браке' then f_get_rd_id('family.married')
                      when 'В браке не состоял(а)' then f_get_rd_id('family.unmarried')
                      when 'Разведен(а)' then f_get_rd_id('family.divorced')
                      when 'Вдовец/вдова' then f_get_rd_id('family.widower')
                    end;
    rd_marital_status = :family_status;
  end
  else
  if (coalesce(:family_status_tmp, -1) <> -1) then
  begin
    family_status = case :family_status_tmp
                      when 0 then f_get_rd_id('family.married') -- замужем
                      when 1 then f_get_rd_id('family.unmarried') -- холост
                      when 2 then f_get_rd_id('family.divorced') -- разведен
                      when 3 then f_get_rd_id('family.widower') -- вдовец
                      when 4 then 0 -- не заполнен
                    end;
    rd_marital_status = :family_status;
  end

  if ((addressregion = 'null') or (addressregion = '') or (addressregion = '-')) then
    addressregion = null;

  if ((address_building = 'null') or (address_building = '') or (address_building = '-')) then
    address_building = null;

  if ((address_apartment = 'null') or (address_apartment = '') or (address_apartment = '-') or (address_apartment not similar to '[-]?[[:DIGIT:]]+')) then
    address_apartment = null;

  address_array = cast(f_json_blob_get_value(:json_text, 'addressArray', '') as d_blob_text);

  if (address_array <> '' and
      address_str not containing 'ликвидирован') then
  begin

    country_id = f_json_str_get_value(:address_array, 'countryId', '');
    region_id = f_json_str_get_value(:address_array, 'regionId', '');
    district_id = f_json_str_get_value(:address_array, 'districtId', '');
    aymak_id = f_json_str_get_value(:address_array, 'aymakId', '');
    village_id = f_json_str_get_value(:address_array, 'villageId', '');
    street_id = f_json_str_get_value(:address_array, 'streetId', '');
    house_id = f_json_str_get_value(:address_array, 'houseId', '');
    flat_id = f_json_str_get_value(:address_array, 'flatId', '');

    country_txt = cast(f_json_blob_get_value(:address_array, 'countryTxt', '') as d_blob_text);
    region_txt = cast(f_json_blob_get_value(:address_array, 'regionTxt', '') as d_blob_text);
    district_txt = cast(f_json_blob_get_value(:address_array, 'districtTxt', '') as d_blob_text);
    aymak_txt = cast(f_json_blob_get_value(:address_array, 'aymakTxt', '') as d_blob_text);
    village_txt = cast(f_json_blob_get_value(:address_array, 'villageTxt', '') as d_blob_text);
    street_txt = cast(f_json_blob_get_value(:address_array, 'streetTxt', '') as d_blob_text);
    house_txt = cast(f_json_blob_get_value(:address_array, 'houseTxt', '') as d_blob_text);
    flat_txt = cast(f_json_blob_get_value(:address_array, 'flatTxt', '') as d_blob_text);

    post_str = cast(f_json_str_get_value(:address_array, 'post', '') as d_blob_text);

    -- проверяем наличие дочернего элемента и ес его нет добавляем

    if (country_id <> '4948') then
      exception e_default 'Страна не Кыргызстан';

    parent_id = 417000000001; -- Кыргызстан

    if (region_id <> '0') then
    begin
      pl_type = 0;
      pl_region_id = (select pl.pl_id
                      from places pl
                      where pl.pl_code = :region_id rows 1);

      if (left(:region_txt, 2) = 'г.') then
        pl_type = 417005;
      else
        pl_type = 417002;
      if (pl_region_id is null) then
        pl_region_id = (select pl_id
                        from hp_insert_place_node(:parent_id, :region_txt, :pl_type, :un_virt, :region_id, 1));

      parent_id = :pl_region_id;
    end

    if (district_id <> '0') then
    begin
      pl_district_id = (select pl.pl_id
                        from places pl
                        where pl.pl_code = :district_id rows 1);
      pl_type = 0;
      if (left(:district_txt, 2) = 'г.') then
        pl_type = 417005;
      else
      if (right(:district_txt, 3) = 'р-н') then
        pl_type = 417003;

      if (pl_district_id is null) then
        pl_district_id = (select pl_id
                          from hp_insert_place_node(:parent_id, :district_txt, :pl_type, :un_virt, :district_id, 1));

      parent_id = :pl_district_id;
    end

    if (aymak_id <> '0') then
    begin
      pl_aymak_id = (select pl.pl_id
                     from places pl
                     where pl.pl_code = :aymak_id rows 1);
      pl_type = 0;
      if (right(:aymak_txt, 3) = 'а/а') then
        pl_type = 417004;
      else
      if (left(:street_txt, 8) = 'переулок') then
        pl_type = 417103;
      else
      if (left(:street_txt, 5) = 'тупик') then
        pl_type = 417106;
      else
      if (left(:street_txt, 3) = 'ж/м') then
        pl_type = 417010;
      else
      if (left(:street_txt, 5) = 'шоссе') then
        pl_type = 417108;
      else
      if ((left(:street_txt, 7) = 'участок') or (left(:street_txt, 3) = 'уч.')) then
        pl_type = 417111;
      else
      if (left(:street_txt, 8) = 'проспект') then
        pl_type = 417104;
      else
      if (left(:street_txt, 7) = 'квартал') then
        pl_type = 417101;
      else
      if (left(:street_txt, 2) = 'г.') then
        pl_type = 417005;
      else
      if (left(:street_txt, 6) = 'массив') then
        pl_type = 417116;
      else
      if (left(:street_txt, 7) = 'бульвар') then
        pl_type = 417115;
      else
      if (left(:street_txt, 5) = 'аллея') then
        pl_type = 417114;
      else
      if (left(:street_txt, 6) = 'контур') then
        pl_type = 417113;
      else
      if (left(:street_txt, 2) = 'ст') then
        pl_type = 417112;
      else
      if (left(:street_txt, 7) = 'поселок') then
        pl_type = 417007;
      else
      if (left(:street_txt, 10) = 'микрорайон') then
        pl_type = 417011;
      else
      if (left(:street_txt, 6) = 'проезд') then
        pl_type = 417105;
      else
      if (left(:street_txt, 4) = 'пгт.') then
        pl_type = 417006;
      else
        pl_type = 417000;
      if (pl_aymak_id is null) then
        pl_aymak_id = (select pl_id
                       from hp_insert_place_node(:parent_id, :aymak_txt, :pl_type, :un_virt, :aymak_id, 1));

      parent_id = :pl_aymak_id;
    end

    if (village_id <> '0') then
    begin
      pl_village_id = (select pl.pl_id
                       from places pl
                       where pl.pl_code = :village_id rows 1);
      pl_type = 0;
      if (left(:village_txt, 2) = 'с.') then
        pl_type = 417008;
      if (pl_village_id is null) then
        pl_village_id = (select pl_id
                         from hp_insert_place_node(:parent_id, :village_txt, :pl_type, :un_virt, :village_id, 1));
      if (pl_village_id is not null) then
        parent_id = :pl_village_id;
    end

    if (street_id <> '0') then
    begin
      pl_street_id = (select pl.pl_id
                      from places pl
                      where pl.pl_code = :street_id rows 1);
      pl_type = 0;
      if (left(:street_txt, 5) = 'улица') then
        pl_type = 417107;
      else
      if (left(:street_txt, 8) = 'переулок') then
        pl_type = 417103;
      else
      if (left(:street_txt, 5) = 'тупик') then
        pl_type = 417106;
      else
      if (left(:street_txt, 3) = 'ж/м') then
        pl_type = 417010;
      else
      if (left(:street_txt, 5) = 'шоссе') then
        pl_type = 417108;
      else
      if ((left(:street_txt, 7) = 'участок') or (left(:street_txt, 3) = 'уч.')) then
        pl_type = 417111;
      else
      if (left(:street_txt, 8) = 'проспект') then
        pl_type = 417104;
      else
      if (left(:street_txt, 7) = 'квартал') then
        pl_type = 417101;
      else
      if (left(:street_txt, 2) = 'г.') then
        pl_type = 417005;
      else
      if (left(:street_txt, 6) = 'массив') then
        pl_type = 417116;
      else
      if (left(:street_txt, 7) = 'бульвар') then
        pl_type = 417115;
      else
      if (left(:street_txt, 5) = 'аллея') then
        pl_type = 417114;
      else
      if (left(:street_txt, 6) = 'контур') then
        pl_type = 417113;
      else
      if (left(:street_txt, 2) = 'ст') then
        pl_type = 417112;
      else
      if (left(:street_txt, 7) = 'поселок') then
        pl_type = 417007;
      else
      if (left(:street_txt, 10) = 'микрорайон') then
        pl_type = 417102;
      else
      if (left(:street_txt, 6) = 'проезд') then
        pl_type = 417105;
      else
      if (left(:street_txt, 4) = 'пгт.') then
        pl_type = 417006;
      else
        pl_type = 417000;
      if (pl_street_id is null) then
        pl_street_id = (select pl_id
                        from hp_insert_place_node(:parent_id, :street_txt, :pl_type, :un_virt, :street_id, 1));

      parent_id = :pl_street_id;
    end

    if ((post_str = 'null') or (post_str = '')) then
      post = 0;
    else
      post = cast(post_str as d_integer);
  end
  suspend;
end
