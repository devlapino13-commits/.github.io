CREATE PROCEDURE HP_OPERKROFF_CLOSE(
    REP_K_ID ID_TYPE,
    REP_DATE DATE,
    OL_MEMO VARCHAR(255) = null,
    OL_ID ID_TYPE = null)
AS
declare variable oko_os curr_type;
declare variable oko_perc curr_type;
declare variable oko_other curr_type;
declare variable new_ol_id id_type;
declare variable reg_id integer;
declare variable doc_no smallint;
declare variable un_id id_type;
declare variable oko_from_c_id id_type;
begin
  /* Procedure Text */
  reg_id = null;
  new_ol_id = null;
  if (ol_memo is null) then ol_memo = 'Очистка внесистемного учета. Письмо от 28.05.2019 Азамат Махмутов.';
  un_id = 1000000000000;

  for
  select oko_from_c_id, sum(oko_os + oko_sh_os), sum(oko_perc + oko_sh_perc), sum(oko_other + oko_sh_other)
  from operkroff
  where oko_from_k_id = :rep_k_id
  and oko_ol_is_active = 1
  and (:ol_id is null or oko_from_ol_id = :ol_id)
  group by oko_from_c_id
  having sum(oko_os + oko_sh_os) > 0 or sum(oko_perc + oko_sh_perc) > 0 or sum(oko_other + oko_sh_other) > 0
  into oko_from_c_id, oko_os, oko_perc, oko_other
  do
  begin
    if (reg_id is null) then
    begin
      select k_last_reg_id from kontract
      where k_id = :rep_k_id
      into reg_id;
    end

    --создаем новый OL_ID
    if (new_ol_id is null) then
    begin
      select kv
      from get_kv(:reg_id, 5)
      into :new_ol_id;

      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_no, ol_doc_kind, ol_memo,
                          ol_status, is_delete, ol_user_id, ol_document_client)
      values (:new_ol_id, :reg_id, 1, 10, :rep_date, :doc_no, 0, :ol_memo, 0, 0, :un_id, 0);
    end

    doc_no = 0;
    /*
    select num
    from last_doc_number(:oper_date, 0, :reg_id)
    into :doc_no;
    */

    insert into operkroff (oko_from_ol_id, oko_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc, oko_other,
                           oko_sh_os, oko_sh_perc, oko_sh_other, oko_ol_is_active, oko_ol_date)
    values (:new_ol_id, null, :rep_k_id, :oko_from_c_id, 5, -:oko_os, -:oko_perc, -:oko_other, 0,
            0, 0, 1, :rep_date);
  end

end
