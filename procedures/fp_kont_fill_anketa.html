CREATE PROCEDURE FP_KONT_FILL_ANKETA(
    REP_K_ID DECIMAL(18,0),
    REP_DATE DATE,
    REP_UN_ID DECIMAL(18,0))
AS
declare variable c_id decimal(18,0);
declare variable ank_date date;
declare variable database_id integer;
begin

  :database_id = f_get_db_id();

  if (database_id = 81001) then
  begin
    for select distinct c_id from
    (
    select kncl.kc_from_c_id  as c_id
    from knclient kncl
    where kncl.kc_from_k_id = :rep_k_id and
          kncl.kc_amount_approved > 0
    union all
    select zal.z_from_c_id
    from zalog zal
    where zal.z_from_k_id = :rep_k_id and
          zal.z_from_c_id is not null and
          zal.z_is_ready = 1 and
          :rep_date between coalesce(zal.z_date_begin, '01.01.1900') and coalesce(zal.z_date_end, '31.12.9999')
    ) A
    into :c_id
    do
    begin
      ank_date = null;
      select fp.fpa_date
      from fp_anketa_cl fp
      where fp.fpa_c_id = :c_id
      order by fp.fpa_date desc
      rows 1
      into :ank_date;

      execute procedure fp_fill_anketa(:c_id, :rep_date, :rep_un_id);
      /*
        if (ANK_DATE is null) then
            execute procedure FP_FILL_ANKETA(:C_ID, :REP_DATE, :REP_UN_ID);
        else
        begin
             if (abs(datediff(DAY, cast(:ANK_DATE as date), cast(:REP_DATE as date))) >= 30) then 
                execute procedure FP_FILL_ANKETA(:C_ID, :REP_DATE, :REP_UN_ID);
        end
        */
    end
  end
end
