CREATE PROCEDURE HP_CHECK_BIOMETRY_CONDITIONS(
    UN_ID D_ID,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    EDC_ID D_INTEGER = null)
RETURNS (
    HAS_ACCESS D_SMALLINT)
AS
declare variable c_id d_id;
declare variable edl_status d_smallint;
declare variable eds_id d_id;
begin
  has_access = 0;

  --Пока что только для КР
  if (f_get_db_id() not in (39801, 41701)) then
  begin
    suspend;
    exit;
  end 

  --Проверяем есть ли у данного сотрудника доступ к подписанию эл. док-ов через биометрию
  if (not exists(select *
                 from temp_objlist tol
                 where tol.tol_kind = 'selfie_sign' and
                       tol.tol_object_kind = f_objkind('username', 'id') and
                       tol.tol_object_id = :un_id)) then
  begin
    suspend;
    exit;
  end

  --Временный костыль пока не переведём edoc_cfg.edc_code = 'consulting.agreement' с clients на knclient
  --Знаю что с edc_code = 'consulting.agreement' прилетает knclient.kc_id
  if (f_edoc_kind(:edc_id, 'code') = 'consulting.agreement') then
  begin                                                                                         -- RFCKG-4363
    select kc.kc_id, 'knclient'
    from knclient kc
    where kc.kc_id = :obj_id
    into obj_id, obj_kind;
  end 

  --Проверяем подписан ли документ или отправлено СМС, если подписан или отправлено СМС то has_access = 0
  for select edl.edl_status, eds.eds_id
      from edoc_list edl
      inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
      where edl.edl_object_kind = f_objkind(:obj_kind, 'id') and
            edl.edl_object_id = :obj_id and
            edl.edl_status >= 0 and
            current_date between edl.edl_blank_date and coalesce(edl.edl_date_end, current_date) and
            iif(:edc_id is not null, edl.edl_from_edc_id = :edc_id, true)
      into edl_status, eds_id
  do
  begin
    if (edl_status = 1) then
    begin
      suspend;
      exit;
    end 

    if (exists(select *
               from edoc_msg edm
               where edm.edm_from_eds_id = :eds_id)) then
    begin
      suspend;
      exit;
    end 
  end


  --Проверяем наличие фото с ГРС
  if (obj_kind in ('miniapp', 'kontract', 'kontmatag')) then
  begin
    --Перезаписываю переменную obj_id и привожу её в конечном итоге к ma_id
    if (obj_kind in ('kontract', 'kontmatag')) then
    begin
      if (obj_kind = 'kontmatag') then
      begin
        select kma.kma_from_k_id
        from kontmatag kma
        where kma.kma_id = :obj_id rows 1
        into obj_id;
      end

      select kt.k_from_ma_app_id
      from kontract kt
      where kt.k_id = :obj_id
      into obj_id;
    end
    ------------------------------------------------------------------------

    for select mac.mac_from_c_id
        from miniapp ma
        inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
        where ma.ma_id = :obj_id and
              ma.ma_is_active = 1 and
              ma.ma_date_end = '31.12.9999' and
              mac.mac_amount > 0
        into c_id
    do
    begin
      if (not exists(select *
                     from extinfolist eil
                     inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                     inner join extinfolistbls eib on eib.eib_from_eil_id = eil.eil_id
                     where eil.eil_from_c_id = :c_id and
                           eic.eic_code = 'Tunduk' and
                           eil.eil_status = 2)) then
      begin
        suspend;
        exit;
      end
    end
  end
  else
  if (obj_kind = 'clients') then
  begin
    if (not exists(select *
                   from extinfolist eil
                   inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                   inner join extinfolistbls eib on eib.eib_from_eil_id = eil.eil_id
                   where eil.eil_from_c_id = :obj_id and
                         eic.eic_code = 'Tunduk' and
                         eil.eil_status = 2)) then
    begin
      suspend;
      exit;
    end
  end

  has_access = 1;
  suspend;
end
