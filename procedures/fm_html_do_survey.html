CREATE PROCEDURE FM_HTML_DO_SURVEY(
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable svq_id integer;
declare variable svq_caption varchar(255);
declare variable svq_memo varchar(255);
declare variable c_id decimal(18,0);
declare variable selected_sva_id integer;
declare variable sva_codes varchar(255);
declare variable sva_caption varchar(255);
declare variable sva_id integer;
declare variable res_link varchar(4096);
declare variable kol integer;
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  res_html = (select res_style from fm_html_styles) ||
    '<body><h1 class="green_text">Онлайн голосование.</h1>';

  kol = 0;
  for select svq_id, svq_caption, svq_memo
      from survey_ques
      where svq_time_start <= current_timestamp and
            (svq_time_finish >= current_timestamp or
            svq_time_finish is null)
      into :svq_id, :svq_caption, :svq_memo
  do
  begin
    kol = kol + 1;
    res_html = BLOB_APPEND(res_html,
      '<hr width="100%"><h2 class="green_text">' || :svq_caption || '</h2>');
    if (coalesce(svq_memo, '') <> '') then
      res_html = BLOB_APPEND(res_html, svq_memo || '<br><br>');

    selected_sva_id = null;
    select svr_from_sva_id
    from survey_result
    where svr_source_kind = 0 and
          svr_source_id = :c_id and
          svr_from_svq_id = :svq_id rows 1
    into :selected_sva_id;
    if (selected_sva_id is null) then
      selected_sva_id = 0;

    if (selected_sva_id = 0) then
      res_html = BLOB_APPEND(res_html, 'Выберите один из ответов:' || '<br>');
    else
      res_html = BLOB_APPEND(res_html, 'Ваш ответ зафиксирован.' || '<br>');

    for select sva_id, sva_codes, sva_caption
        from survey_answer
        where sva_from_svq_id = :svq_id
        order by sva_codes
        into :sva_id, :sva_codes, sva_caption
    do
    begin
      sva_codes = replace(sva_codes, '~', '');
      res_link = sva_codes || '. ' || sva_caption;

      if (selected_sva_id = 0) then
      begin
        res_link = f_uri_sql(res_link, 'set_survey_answer', '0', :c_id,
          'none', :sva_caption || '<br><br>Хотите зафиксировать этот ответ?',
          '', 4, 355, '', '', 'dialog', :sva_id);
      end

      if (sva_id = selected_sva_id) then
        res_html = BLOB_APPEND(res_html, '<br><b>' || res_link || '</b><br>');
      else
        res_html = BLOB_APPEND(res_html, '<br>' || res_link || '<br>');
    end
  end
  if (kol = 0) then
    res_html = BLOB_APPEND(res_html,
      '<hr width="100%">В настоящий момент нет открытого голосования.');
  res_html = BLOB_APPEND(res_html, '</body>');
  suspend;

end
