CREATE PROCEDURE AAREP_BATCH_WRITEOFF(
    REP_DATE DATE)
RETURNS (
    K_KONTRACT_NO STR_100,
    OK_OS CURR_TYPE,
    OK_PERC CURR_TYPE,
    OK_FINE CURR_TYPE,
    OPEREXTRA_AMOUNT_WRITEOFF CURR_TYPE,
    ERR_TEXT STR_100,
    IS_OK INTEGER,
    WRITEOFF_REASON STR_100)
AS
declare variable K_ID ID_TYPE;
declare variable OL_ID ID_TYPE;
begin
  for select kfw.k_id, k_fact_closed_ol_id, kont.k_kontract_no, is_ok, rd.rd_name
      from kont_for_writeoff kfw
      inner join kontract kont on kont.k_id = kfw.k_id
      left join refdata rd on rd.rd_id = kfw.writeoff_reason_id
      where kfw.writeoff_date = :rep_date and
            kfw.is_ok > -5
      into :k_id, :ol_id, :k_kontract_no, :is_ok, :writeoff_reason
  do
  begin
    ok_os = 0;
    ok_perc = 0;
    ok_fine = 0;
    err_text = '';
    operextra_amount_writeoff = 0;

    if (is_ok = 1) then
    begin
      select -sum(ok.ok_os), -sum(ok.ok_perc), -sum(ok.ok_fine)
      from operkred ok
      where ok.ok_from_ol_id = :ol_id and
            ok.ok_from_k_id = :k_id and
            ok.ok_type = 5  and
            ok.ok_ol_is_active = 1
      into :ok_os, :ok_perc, :ok_fine;

      select oe.ox_amount_writeoff
      from operextra oe
      where oe.ox_ol_is_active = 1
      and oe.ox_object_kind = 9
      and oe.ox_object_id = :k_id
      and oe.ox_from_ol_id = :ol_id
      into :operextra_amount_writeoff;
    end
    else
    begin
      err_text = (case is_ok
                    when -1 then 'Кредит не активный'
                    when -2 then 'Кредит не активный'
                    when -3 then 'Просрочка меньше 361 день.'
                    when -4 then 'Есть операция более поздней датой'
                    when 0 then 'Не обработан'
                  end);

    end
    suspend;
  end
end
