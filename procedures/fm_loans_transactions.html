CREATE PROCEDURE FM_LOANS_TRANSACTIONS(
    UN_ID DECIMAL(18,0),
    REP_DATE DATE,
    REG_ID INTEGER)
RETURNS (
    OBJ_ID DECIMAL(18,0),
    KF_NAME VARCHAR(100),
    RES_AMOUNT VARCHAR(100),
    RES_TIME VARCHAR(100),
    OK_TYPE VARCHAR(20),
    PR_ID ID_TYPE)
AS
declare variable c_id decimal(18,0);
declare variable k_id decimal(18,0);
declare variable ol_datetime timestamp;
declare variable time_delta integer;
declare variable ok_amount decimal(18,4) = 0;
declare variable c_name_fio varchar(100);
declare variable cl_id decimal(18,0);
begin
  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exit;

  time_delta = 0;
  if (f_get_db_id() = 81001) then
    time_delta = 3;
  
  if (not exists(select ps_from_reg_id
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :c_id and
                       ps_date_begin <= current_date and
                       ps_date_end >= current_date and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       iif(f_get_db_id() in (39801, 41701), jk_system_code = 'JobSystem.Region_KM', jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior')) and
                       ps_is_working = 1)) then
    reg_id = 0;

  for select res_c_id
      from hp_staff_changer_list(:un_id, 1)
      where :reg_id = 0
      union
      select res_c_id
      from hp_staff_changer_list(:un_id, -1)
      where :reg_id <> 0
      into :c_id
  do
  begin
    for select oks_from_k_id
        from oksummary
        where oks_date_begin = :rep_date and
              oks_date_end >= :rep_date and
              oks_kc_id = :c_id and
              :reg_id = 0 and
              oks_close_state = 0
        union
        select oks_from_k_id
        from oksummary
        inner join region_list(:reg_id, 1, 3, 1) reg on oks_region_id = reg.id_list
        where oks_date_begin = :rep_date and
              oks_date_end >= :rep_date and
              :reg_id <> 0 and
              oks_close_state = 0
        into :k_id
    do
    begin
      for select ok_from_ol_id, iif(ok_type in (4, 9), 'Погаш.', iif(ok_type = 4, 'Списан', 'Выдача')),
                 sum(-ok_os - ok_perc - ok_fine - ok_kommis - ok_kommis_adv - ok_tax)
          from operkred
          where ok_from_k_id = :k_id and
                ok_type in (4, 5, 8, 9) and
                ok_ol_is_active = 1 and
                ok_ol_date = :rep_date
          group by 1, 2
          into :obj_id, :ok_type, ok_amount
      do
      begin
        kf_name = '';
        select kal_new_kf_name, k_from_pr_id
        from knaktlist, kontract
        where knaktlist.is_delete = 0 and
              kal_from_kf_id = k_from_kf_id and
              k_id = :k_id and
              kal_date <= :rep_date and
              kal_date_end >= :rep_date
        into :kf_name, :pr_id;

        res_amount = round(ok_amount, 2);
        res_time = '';
        select ol_datetime + coalesce(reg_timezone, 0) - :time_delta, ol_document_client
        from operlist, region
        where ol_id = :obj_id and
              ol_from_reg_id = reg_unicode
        into :ol_datetime, cl_id;
        res_time = right('0' || extract(hour from ol_datetime), 2) || ':' || right('0' || extract(minute from ol_datetime), 2);

        if (cl_id is not null) then
        begin
          select c_name_fio
          from clients
          where c_id = :cl_id
          into :c_name_fio;

          select short_fio
          from hp_fio_clear(:c_name_fio)
          into :c_name_fio;
          kf_name = kf_name || ', ' || c_name_fio;
        end
        suspend;
      end
    end
  end
end
