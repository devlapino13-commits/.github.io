CREATE PROCEDURE CB_KG_LO_CHECK(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    JL_NAME STR_100,
    K_KONTRACT_NO VARCHAR(20),
    AAREP_PLAN_ALL DECIMAL(18,4),
    AAREP_FACT_ALL DECIMAL(18,4),
    AAREP_FACT_PLANNED DECIMAL(18,4),
    AAREP_FACT_OTHER DECIMAL(18,4),
    AAREP_MONTH_PLAN_OS DECIMAL(18,4),
    AAREP_MONTH_PLAN_PERC DECIMAL(18,4),
    AAREP_DELINQ_OS DECIMAL(18,4),
    AAREP_DELINQ_PERC DECIMAL(18,4),
    AAREP_FACT_OS DECIMAL(18,4),
    AAREP_FACT_PERC DECIMAL(18,4),
    AAREP_FACT_FINE DECIMAL(18,4),
    AAREP_FACT_DELINQ_OS DECIMAL(18,4),
    AAREP_FACT_DELINQ_PERC DECIMAL(18,4),
    AAREP_FACT_ADV_OS DECIMAL(18,4),
    AAREP_FACT_ADV_PERC DECIMAL(18,4),
    AAREP_PORT_BALANCE_OWN DECIMAL(18,4),
    AAREP_PORT_KOLICH_OWN INTEGER,
    AAREP_FACT_KOMMIS DECIMAL(18,4),
    AAREP_FACT_KOMMIS_IPF CURR_TYPE,
    OFFICER_NAME STR_100)
AS
declare variable c_id id_type;
declare variable reg_id integer;
declare variable k_id id_type;
declare variable aarep_oklad decimal(18,4);
begin
  for select ps_from_c_id, ps_from_reg_id, jl_name, c_name_fio
  from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1)
  where ps_from_c_id = c_id and
        ps_date_begin <= :rep_date_end and
        ps_date_end >= :rep_date_end and
        ps_status = 1 and
        ps_is_working = 1 and
        ps_from_c_id > 1000000000003 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id and
        jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
        ps_from_reg_id = id_list
  into :c_id,  :reg_id, jl_name, officer_name
  do
  begin
    for
    select k_id, k_kontract_no, aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other,
           aarep_month_plan_os, aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc,
           aarep_fact_os, aarep_fact_perc, aarep_fact_fine, aarep_fact_delinq_os,
           aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc, aarep_oklad,
           aarep_port_balance_own, aarep_port_kolich_own, aarep_fact_kommis, aarep_fact_kommis_ipf
    from cb_kg_lo_201602_one_tmp(:reg_id, :c_id, :rep_date_end)
    into k_id, k_kontract_no, aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os,
         aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine,
         aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc, aarep_oklad,
         aarep_port_balance_own, aarep_port_kolich_own, aarep_fact_kommis, aarep_fact_kommis_ipf
    do
    begin


      suspend;
    end 
  end
end
