CREATE PROCEDURE HP_INACTIVE_CLIENTS_GET(
    REP_DATE D_DATE)
RETURNS (
    C_ID D_ID,
    K_ID D_ID,
    CLOSE_DATE D_DATE)
AS
declare variable last_k_id d_id;
declare variable has_active_loan d_bool;
begin
  if (rep_date is null) then
    rep_date = current_date;

  for
    select
      kc.kc_from_c_id,
      ks.ks_from_k_id,
      ks.ks_date_begin
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
    where ks.ks_state = 2
      and ks.ks_ol_is_delete = 0
      and ks.ks_date_end = date '31.12.9999'
      -- С момента закрытия прошло не менее 60 календарных дней
      and ks.ks_date_begin <= :rep_date - 60
      --and ks.ks_date_begin >= date '01.01.2026' -- test
      and kc.kc_amount_approved > 0
    into
      :c_id,
      :k_id,
      :close_date
  do
  begin
    -- Проверяем, что выбранный договор является последним займом клиента
    last_k_id = null;

    select first 1 kc.kc_from_k_id
    from knclient kc
    where kc.kc_from_c_id = :c_id
      and kc.kc_amount_approved > 0
    order by kc.kc_date_disb desc, kc.kc_from_k_id desc
    into :last_k_id;

    if (last_k_id is distinct from k_id) then
      continue;

    -- Проверяем, что у клиента нет другого действующего займа
    has_active_loan = 0;

    select first 1
      1
    from kontstate ks
    inner join knclient kc on ks.ks_from_k_id = kc.kc_from_k_id
    where kc.kc_from_c_id = :c_id
      and kc.kc_amount_approved > 0
      and ks.ks_state = 1
      and ks.ks_ol_is_delete = 0
      and :rep_date between ks.ks_date_begin and ks.ks_date_end
    into :has_active_loan;

    if (has_active_loan = 1) then
      continue;

    -- Здесь потом можно добавить проверку заявок

    suspend;
  end
end
