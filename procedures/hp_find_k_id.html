CREATE PROCEDURE HP_FIND_K_ID(
    KONTRACT_NO D_STR_100)
RETURNS (
    K_ID DECIMAL(18,0),
    KONTRACT_NO_LOC D_STR_20)
AS
declare variable COUNT_SYMBOL D_SMALLINT;
declare variable POINT_1 D_SMALLINT;
declare variable POINT_2 D_SMALLINT;
declare variable PART_1 D_STR_20;
declare variable PART_2 D_STR_20;
declare variable PART_3 D_STR_20;
begin
  -- Входящий параметр: набор символов
  -- Если все нормально, K_ID
  -- Если нет такого догоовра, результат -1
  k_id = -1;

  --вывод ошибки
  if ((kontract_no is null) or (kontract_no = '')) then
  begin
    suspend;
    exit;
  end

  count_symbol = char_length(kontract_no);

  --вывод ошибки
  if (count_symbol < 12) then
  begin
    suspend;
    exit;
  end

  --вывод ошибки
  if (count_symbol > 17) then    -- RFCKG-5278
  begin
    suspend;
    exit;
  end

  if (kontract_no containing '-') then
    kontract_no_loc = kontract_no;
  else
  begin
    --устанавливаем позицию "-" в наборе символов
    point_1 = 4;
    point_2 = count_symbol - 2;

    --делим на три части
    part_1 = substring(kontract_no from 1 for point_1);
    part_2 = substring(kontract_no from point_1 + 1 for count_symbol - 6);
    part_3 = substring(kontract_no from count_symbol - 1 for 2);

    --собираем номер договора
    kontract_no_loc = part_1 || '-' || part_2 || '-' || part_3;
  end

  select coalesce(k_id, -1)
  from kontract k
  where k.k_kontract_no = trim(:kontract_no_loc)
  into :k_id;
  suspend;
end
