CREATE PROCEDURE PRINT_CLI_ADDRESS_FACT(
    C_ID D_ID,
    ADD_PARAMS D_ID)
RETURNS (
    C_NAME_FIO D_STR_255,
    C_INN D_STR_20,
    EIL_DATE D_DATE,
    STATE D_STR_255,
    CITY D_STR_255,
    REGION_NAME D_STR_255,
    DISTRICT D_STR_255,
    STREET D_STR_255,
    HOUSE D_STR_255,
    FLAT D_STR_255)
AS
declare variable bls_id d_id;
declare variable json d_blob_text;
declare variable k_disb_date d_date;
begin
  select k.k_disb_date
  from kontract k
  where k.k_id = :add_params
  into k_disb_date;

  k_disb_date = coalesce(k_disb_date, current_date);

  select eil_date, eil_from_bls_id
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id and
        eil_from_eic_id = eic_id and
        eic_code = 'address_fact' and
        eil_is_json = 1 and
        eil_status = 2 and
        eil_date <= :k_disb_date --RFCKG-3426
  order by eil_date desc, eil_id
  rows 1
  into eil_date, bls_id;

  if (bls_id is null) then -- Если не удалось найти бланк берем последний
  begin
    select eil_date, eil_from_bls_id
    from extinfolist, extinfocfg
    where eil_from_c_id = :c_id and
        eil_from_eic_id = eic_id and
        eic_code = 'address_fact' and
        eil_is_json = 1 and
        eil_status = 2
    order by eil_date desc, eil_id
    rows 1
    into eil_date, bls_id;

    eil_date = k_disb_date;
  end

  select c_name_fio, c_inn
  from clients
  where c_id = :c_id
  into c_name_fio, c_inn;

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json;

  if (coalesce(:json, '') <> '') then
  begin
    state = f_json_str_get_value(:json, 'state', '');
    city = f_json_str_get_value(:json, 'city', '');
    if (:state containing 'г.') then
    begin
      city = :state;
      state = '';
    end
    region_name = f_json_str_get_value(:json, 'region', '');
    if (:region_name containing 'г.') then
    begin
      city = :region_name;
      region_name = '';
    end
    district = f_json_str_get_value(:json, 'district', '');
    if (:district containing 'г.') then
    begin
      city = :district;
      district = '';
    end
    street = f_json_str_get_value(:json, 'street', '');
    house = f_json_str_get_value(:json, 'house', '');
    flat = f_json_str_get_value(:json, 'flat', '');
  end
  suspend;
end
