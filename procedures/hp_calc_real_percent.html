CREATE PROCEDURE HP_CALC_REAL_PERCENT(
    AMOUNT DECIMAL(18,4),
    KOMMIS DECIMAL(18,4),
    PERIOD INTEGER,
    CALC_PERC_FOR_DAY DECIMAL(18,7))
RETURNS (
    RES_PERCENT DECIMAL(18,7),
    K INTEGER)
AS
declare variable annu_perc decimal(18,4);
declare variable annu decimal(18,4);
declare variable calc_perc_for_month decimal(18,7);
declare variable annu_need decimal(18,4);
declare variable ext decimal(18,7);
declare variable a2 decimal(18,4);
declare variable min2 decimal(18,4);
declare variable min1 decimal(18,7);
declare variable res_max decimal(18,7);
declare variable res_min decimal(18,7);
begin
  if (kommis = 0) then
  begin
    res_percent = round(calc_perc_for_day * 30.0, 9);
    suspend;
    exit;
  end
  if (period <= 1) then
  begin
    res_percent = 0;
    suspend;
    exit;
  end

  calc_perc_for_month = round(cast(calc_perc_for_day as double precision) * 30 / 100.0, 7);
  annu_perc = (calc_perc_for_month * power((1 + calc_perc_for_month), period)) / (power((1 + calc_perc_for_month), period) - 1);
  annu_need = amount * annu_perc;
  annu_need = round(annu_need, 0);

  amount = amount - kommis;
  ext = 0.000001;
  res_percent = 10000.0;
  res_max = res_percent;
  res_min = 0.0;

  calc_perc_for_month = round(res_percent / 100.0, 7);
  annu_perc = (calc_perc_for_month * power((1 + calc_perc_for_month), period)) / (power((1 + calc_perc_for_month), period) - 1);
  annu = amount * annu_perc;
  annu = round(annu, 0);

  a2 = annu;
  k = 0;
  min2 = a2;
--  while (abs(annu_need-a2)>ext and k<1000000) do
  while ((res_max - res_min) > ext and k < 1000000) do
  begin
    k = k + 1;
    if (a2 >= annu_need) then
    begin
      res_max = res_percent;
      res_percent = res_percent - (res_max - res_min) / 2.0000000;
    end
    if (a2 < annu_need) then
    begin
      res_min = res_percent;
      res_percent = res_percent + (res_max - res_min) / 2.0000000;
    end
    res_percent = round(res_percent, 7);

    if (res_percent <> 0) then
    begin
      calc_perc_for_month = res_percent / 100.0;
      annu_perc = (calc_perc_for_month * power((1 + calc_perc_for_month), period)) / (power((1 + calc_perc_for_month), period) - 1);
      annu = amount * annu_perc;
      annu = round(annu, 0);
      a2 = annu;
    end
    else
      a2 = 0;

    if (min2 > abs(annu_need - a2)) then
    begin
      min2 = abs(annu_need - a2);
      min1 = res_percent;
    end
  end
  if (k = 1000000) then
    res_percent = min1;
  suspend;
end
