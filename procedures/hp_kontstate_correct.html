CREATE PROCEDURE HP_KONTSTATE_CORRECT
RETURNS (
    K_ID DECIMAL(18,0),
    KS_NUMBER INTEGER,
    KS_DATE_BEGIN DATE,
    KS_REGION_ID INTEGER,
    KS_KC_ID DECIMAL(18,0),
    KS_KFOLDER_NO VARCHAR(100))
AS
begin
  /* поправляет код офиса и КС */
  for select ks_from_k_id, ks_number, ks_date_begin
      from kontstate
      into :k_id, :ks_number, :ks_date_begin
  do
  begin
    select kal_new_reg_id, kal_new_kf_officer, kal_new_kf_name
    from knaktlist, kontract
    where is_delete = 0 and
          kal_from_kf_id = k_from_kf_id and
          k_id = :k_id and
          kal_date <= :ks_date_begin and
          kal_date_end >= :ks_date_begin rows 1
    into :ks_region_id, :ks_kc_id, :ks_kfolder_no;

    update kontstate
    set ks_region_id = :ks_region_id,
        ks_kc_id = :ks_kc_id,
        ks_kfolder_no = :ks_kfolder_no
    where ks_from_k_id = :k_id and
          ks_number = :ks_number;
  end
end
