CREATE PROCEDURE UXP$UXCOMMAND_INFO(
    UX_UUID D_UUID,
    UX_UUID_CODE D_STR_CODE = null)
RETURNS (
    CMD_UUID D_UUID,
    CMD_OPEN_KIND D_SMALLINT,
    CMD_SOURCE_ROWS D_SMALLINT,
    CMD_SOURCE_REFRESH D_SMALLINT,
    CMD_COMMIT_EXPR D_STR_LONG,
    CMD_MESSAGE_FIELD D_STR_LONG,
    CMD_EXPRESSION D_STR_LONG,
    CMD_COMMIT_DIALOG D_BOOL,
    UX_NAME D_STR_100,
    UX_CODE D_STR_CODE,
    RES_CMD_SOURCE D_STR_CODE,
    RES_HAS_OUTPUT D_BOOL)
AS
declare variable fr_uuid d_uuid;
begin
  if (coalesce(ux_uuid, '') = '') then
    ux_uuid = uxf$uniq(ux_uuid_code);

  for select cmd_uuid, cmd_open_kind, cmd_source_rows, cmd_source_refresh, cmd_message_field, cmd_commit_dialog,
             ux_name, ux_fcode
  from ux$command, ux$form, ux$code
  where cmd_uuid = :ux_uuid and
        cmd_uuid = ufm_uuid and
        ufm_ux_uuid = ux_uuid
  into :cmd_uuid, :cmd_open_kind, :cmd_source_rows, :cmd_source_refresh, :cmd_message_field, :cmd_commit_dialog,
       :ux_name, ux_code
  do
  begin
    res_cmd_source = null;
    res_has_output = 0;
    cmd_commit_expr = '';
    cmd_expression = '';

    select exp_calc_expr, exp_expression
    from ux$expression
    where exp_ux_uuid = :cmd_uuid and
          exp_kind = 0
    into :cmd_commit_expr, cmd_expression;

    select ufr_uuid, us_code
    from ux$frame, ux$source
    where ufr_owner = :ux_uuid and
          ufr_us_uuid = us_uuid
    into fr_uuid, res_cmd_source;

    if (exists(select *
    from ux$frparam, ux$field
    where ufp_ufr_uuid = :fr_uuid and
          ufp_uf_uuid = uf_uuid and
          uf_is_param = 0)) then
      res_has_output = 1;

    suspend;
  end
end
