CREATE PROCEDURE UXP$UXEXPRESSION_FRAME_LOOKUP(
    UX_UUID D_UUID)
RETURNS (
    UFR_UUID D_UUID,
    UFR_UUID_CODE D_STR_CODE,
    UFR_NAME D_STR_255)
AS
declare variable ufm_uuid d_uuid;
declare variable tmp_uuid d_uuid;
declare variable ufi_owner d_uuid;
begin
  select ufi_owner
  from ux$fritem
  where ufi_uuid = :ux_uuid
  into ufi_owner;

  if (exists(select * from ux$form
             where ufm_uuid = :ufi_owner)) then
  begin
    for select ufr_uuid
    from ux$frame
    where ufr_owner = :ufi_owner
    into ufr_uuid
    do
    begin
      ufr_uuid_code = uuid_to_char(ufr_uuid);
      ufr_name = uxf$code(ufr_uuid, 'caption');
      suspend;
    end
  end else
  begin
    ufr_uuid = null;
    ufr_uuid_code = '';
    ufr_name = 'Текущий фрейм';
    suspend;
  
    select ufr_uuid
    from uxp$uxframe_source(:ux_uuid, null)
    into :tmp_uuid;
  
    if (tmp_uuid is not null) then
    begin
      select ufr_owner
      from ux$frame
      where ufr_uuid = :tmp_uuid
      into ufm_uuid;
  
      for select ufr_uuid
      from ux$frame
      where ufr_owner = :ufm_uuid and
            ufr_uuid <> :tmp_uuid
      into ufr_uuid
      do
      begin
        ufr_uuid_code = uuid_to_char(ufr_uuid);
        ufr_name = uxf$code(ufr_uuid, 'caption');
        suspend;
      end
    end
  end
end
