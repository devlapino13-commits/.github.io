CREATE PROCEDURE HP_GET_CUSTOMDATAVALUE(
    OBJECT_KIND INTEGER,
    OBJECT_ID DECIMAL(18,0),
    FIELD_CODE VARCHAR(255),
    REP_DATE DATE)
RETURNS (
    CDC_ID INTEGER,
    CDC_CODE VARCHAR(255),
    CDV_VALUE VARCHAR(255),
    CDV_VALUE_NUMBER DECIMAL(15,2),
    CDV_VALUE_DATE DATE,
    CDV_VALUE_STRING VARCHAR(255))
AS
declare variable cdc_data_type varchar(100);
begin
  cdc_id = null;
  cdv_value = '';
  cdv_value_number = 0;
  cdv_value_date = null;
  cdv_value_string = '';

  select cdc_id, cdc_data_type
  from customdatacfg
  where cdc_code = :field_code
  into :cdc_id, :cdc_data_type;
  if (cdc_id is null) then
  begin
    cdc_id = 0;
    cdc_code = '';
    suspend;
    exit;
  end

  cdc_code = field_code;
  select cdv_value_number, cdv_value_date, cdv_value_string
  from customdatavalue
  where cdv_object_kind = :object_kind and
        cdv_object_id = :object_id and
        cdv_date_begin <= :rep_date and
        cdv_date_end >= :rep_date and
        cdv_from_cdc_id = :cdc_id rows 1
  into :cdv_value_number, :cdv_value_date, :cdv_value_string;

  if (cdc_data_type = 'number') then
    cdv_value = round(cdv_value_number);
  if (cdc_data_type = 'float') then
    cdv_value = cdv_value_number;
  if (cdc_data_type = 'date') then
    cdv_value = right('0' || extract(day from cdv_value_date), 2) || '.' || right('0' || extract(month from cdv_value_date), 2) || '.' || extract(year from cdv_value_date);
  if (cdc_data_type = 'string') then
    cdv_value = cdv_value_string;

  suspend;
end
