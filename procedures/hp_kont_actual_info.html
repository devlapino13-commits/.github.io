CREATE PROCEDURE HP_KONT_ACTUAL_INFO(
    KONTRACT_NO D_STR_20,
    REP_DATE D_DATE)
RETURNS (
    CONTRACT_STATUS D_STR_20,
    CLOSE_DATE D_DATE,
    ZAD_OS D_CURRENCY,
    ZAD_PERC D_CURRENCY,
    ZAD_FINE D_CURRENCY,
    OKO_SH_OS D_CURRENCY,
    OKO_SH_PERC D_CURRENCY,
    DELINQ_DAYS D_INTEGER,
    IS_RETRUCT D_SMALLINT,
    PR_ID D_ID,
    PR_NAME D_STR_255,
    TOTAL_MAX_DELINQ_DAYS D_INTEGER)
AS
declare variable K_ID D_ID;
declare variable KS_STATE D_SMALLINT;
declare variable KS_DATE_BEGIN D_DATE;
declare variable MAX_INACTIVE_DELINQ_DAYS D_INTEGER;
begin
  for select k.k_id, k.k_from_pr_id, pr.pr_name
      from kontract k
      inner join product pr on pr.pr_id = k.k_from_pr_id
      where k.k_kontract_no = :kontract_no
      into k_id, pr_id, pr_name
  do
  begin
    close_date = null;
    zad_os = 0;
    zad_perc = 0;
    zad_fine = 0;
    oko_sh_os = 0;
    oko_sh_perc = 0;
    select ks.ks_state, ks.ks_date_begin
    from kontstate ks
    where ks.ks_from_k_id = :k_id and
          cast(:rep_date as date) between ks.ks_date_begin and ks.ks_date_end and
          ks.ks_ol_is_delete = 0
    into ks_state, ks_date_begin;
    if (ks_state = 1) then
    begin
      contract_status = 'активный';

      select sum(zad_os), sum(zad_perc), sum(sh_calc_fine)
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      into zad_os, zad_perc, zad_fine;

      select iif((cast(:rep_date as date) - oks.oks_delinq_date < 0), 0, cast(:rep_date as date) - oks.oks_delinq_date) -- числоднейактивнойпросрочки
      from oksummary oks
      where :rep_date between oks.oks_date_begin and oks.oks_date_end and
            oks.oks_from_k_id = :k_id and
            oks_close_state = 0
      into delinq_days;
    end
    if (ks_state = 2) then
    begin
      contract_status = 'закрытый';
      if (exists(select 1
                 from oksummary oks
                 where oks_from_k_id = :k_id and
                       current_date between oks.oks_date_begin and oks.oks_date_end and
                       (oks_curr_sh_os + oks_curr_sh_perc + oks_delinq_os + oks_delinq_perc + oks_out_os + oks_out_perc + oks_out_other + oks_extra) > 0)) then
      begin
        contract_status = 'спец учет';

        select sum(oko.oko_sh_os + oko.oko_os), sum(oko.oko_sh_perc + oko.oko_perc)
        from operkroff oko
        where oko.oko_from_k_id = :k_id and
              oko.oko_ol_is_active = 1
        into oko_sh_os, oko_sh_perc;
      end
    end
    if (contract_status <> 'активный') then
      close_date = ks_date_begin;

    select is_restruct -- 1 – реструктуризирован, 0 - нет
    from aarep_get_restructuring(:k_id, :rep_date, null)
    into is_retruct;


    oko_sh_os = coalesce(oko_sh_os, 0);
    oko_sh_perc = coalesce(oko_sh_perc, 0);

    max_inactive_delinq_days = 0;
    total_max_delinq_days = 0;
    delinq_days = coalesce(delinq_days, 0);

    select coalesce(max(ok.ok_ol_date - ok_sh_date), 0)
    from operkred ok
    where ok.ok_ol_is_active = 1
    and ok.ok_ol_date <= :rep_date
    and ok.ok_ol_date > ok_sh_date
    and ok_type < 7
    and ok_from_k_id = :k_id
    and ok_os + ok_perc < 0
    and ok_from_c_id = 0
    into :max_inactive_delinq_days;

    total_max_delinq_days = maxvalue(max_inactive_delinq_days, delinq_days);

    suspend;
  end
end
