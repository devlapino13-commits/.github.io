CREATE PROCEDURE AAREP_KONTRACT_DELIQ_TEMP(
    REG_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    DK_ID DECIMAL(18,0),
    DK_DATE DATE,
    DK_OS DECIMAL(18,4),
    DK_PERC DECIMAL(18,4),
    DK_DAYS INTEGER,
    DK_REGION_ID INTEGER,
    CURR_CODE VARCHAR(3),
    DK_KF_ID DECIMAL(18,0),
    DK_BALANCE_END DECIMAL(18,4),
    DK_REPAYS_STR VARCHAR(10),
    DK_DISB_DATE DATE,
    DK_BALANCE_BEGIN DECIMAL(18,4),
    AAREP_REPAY_OS DECIMAL(18,4),
    AAREP_REPAY_PERC DECIMAL(18,4),
    AAREP_REPAY_FINE DECIMAL(18,4),
    AAREP_WRITEOFF_OS DECIMAL(18,4),
    AAREP_WRITEOFF_PERC DECIMAL(18,4),
    AAREP_WRITEOFF_FINE DECIMAL(18,4),
    AAREP_LAST_OL_DATE DATE,
    DK_CLIENT_KOL INTEGER,
    DK_LOAN_TYPE INTEGER,
    DK_DISB_KC_ID DECIMAL(18,0),
    DK_PR_CODE VARCHAR(100))
AS
declare variable pr_set_rate_in_kont integer;
declare variable ol_date date;
declare variable ok_type integer;
declare variable ok_sh_date date;
declare variable ok_sh_id integer;
declare variable sh_os decimal(18,4);
declare variable sh_perc decimal(18,4);
declare variable ok_os decimal(18,4);
declare variable ok_perc decimal(18,4);
declare variable is_first_line integer;
declare variable dd_prior date;
declare variable deliq_os decimal(18,4);
declare variable deliq_perc decimal(18,4);
declare variable aa_deliq_repay_kolich integer;
declare variable kolich integer;
declare variable curr_rate decimal(18,4);
declare variable kont_curr_rate decimal(18,4);
declare variable def_curr varchar(3);
declare variable ok_fine decimal(18,4);
declare variable pr_id integer;
declare variable pr_id_prior integer;
declare variable need_curr_rate integer;
begin
  /* Procedure Text */
  select cm_default_currency
  from company
  into :def_curr;
  if (rep_date_begin is null) then
    rep_date_begin = :rep_date_end;
  dk_repays_str = '';

  if (report_currency = '000') then
    need_curr_rate = 1;
  else
    need_curr_rate = 0;
  pr_id_prior = 0;

  for select kk_from_kf_id, kk_currency, kk_id, kk_region_id, kk_disb_date, kk_curr_rate, kk_kolich_client,
             kk_from_pr_id, kk_balance, kk_disb_kc_id
      from aarep_active_kont_id_in_period(:reg_id, null, :rep_date_end, 0, 1, :need_curr_rate, 0, 0)
      into :dk_kf_id, :curr_code, :dk_id, :dk_region_id, :dk_disb_date, :curr_rate, :dk_client_kol, pr_id,
           :dk_balance_end, :dk_disb_kc_id
  do
  begin
    if (pr_id <> pr_id_prior) then
    begin
      select pr_loan_type, pr.pr_code
      from product pr
      where pr_id = :pr_id
      into :dk_loan_type, dk_pr_code;
      pr_id_prior = :pr_id;
    end

    dd_prior = null;
    is_first_line = 1;
    aa_deliq_repay_kolich = 0;
    dk_date = :rep_date_end;
    kolich = 0;
    dk_balance_begin = 0;

    deliq_os = 0;
    deliq_perc = 0;

    dk_os = 0;
    dk_perc = 0;

    aarep_repay_os = 0;
    aarep_repay_perc = 0;
    aarep_repay_fine = 0;

    aarep_writeoff_os = 0;
    aarep_writeoff_perc = 0;
    aarep_writeoff_fine = 0;
    aarep_last_ol_date = null;

    for select ok_ol_date, ok_type, ok_sh_date, ok_sh_id, ok_sh_os, ok_sh_perc, ok_os, ok_perc, ok_fine
        from operkred
        where ok_ol_is_active = 1 and
              ok_from_k_id = :dk_id and
              ok_ol_date between '01.01.1900' and :rep_date_end and
              ok_type <> 7 and
              ok_from_c_id = 0
        order by ok_sh_date, ok_sh_id, ok_ol_date
        into :ol_date, :ok_type, :ok_sh_date, :ok_sh_id, :sh_os, :sh_perc, :ok_os, :ok_perc, :ok_fine
    do
    begin
      if ((ol_date <= :rep_date_end) and
          (ok_sh_date <= :rep_date_end)) then
      begin
        if (((ol_date > aarep_last_ol_date) or (aarep_last_ol_date is null)) and
            ok_type = 4) then
          aarep_last_ol_date = :ol_date;
        if (ok_sh_id <> 0) then
        begin
          if (ok_type < 8) then
          begin
            if ((is_first_line = 1) or (:ok_sh_date = :dd_prior)) then
            begin
              deliq_os = :deliq_os + :sh_os + :ok_os;
              deliq_perc = :deliq_perc + :sh_perc + :ok_perc;
              if (is_first_line = 1) then
                dd_prior = :ok_sh_date;
            end
            else
            begin
              if ((deliq_os > 0) or (deliq_perc > 0)) then
              begin
                aa_deliq_repay_kolich = :aa_deliq_repay_kolich + 1;
                dk_os = :dk_os + :deliq_os;
                dk_perc = :dk_perc + :deliq_perc;
                if (dk_date = :rep_date_end) then
                  dk_date = :dd_prior;
              end
              dd_prior = :ok_sh_date;
              deliq_os = :sh_os + :ok_os;
              deliq_perc = :sh_perc + :ok_perc;
            end
            is_first_line = 0;
          end
        end
      end
      if (:ol_date < :rep_date_begin) then
      begin
        if (ok_type <= 8) then
          dk_balance_begin = :dk_balance_begin + :ok_os;
      end
      if ((ol_date >= :rep_date_begin) and
          (ol_date <= :rep_date_end)) then
      begin
        if (ok_type = 4) then
        begin
          aarep_repay_os = aarep_repay_os - ok_os;
          aarep_repay_perc = aarep_repay_perc - ok_perc;
          aarep_repay_fine = aarep_repay_fine - ok_fine;
        end
        if (ok_type in (5, 6)) then
        begin
          aarep_writeoff_os = aarep_writeoff_os - ok_os;
          aarep_writeoff_perc = aarep_writeoff_perc - ok_perc;
          aarep_writeoff_fine = aarep_writeoff_fine - ok_fine;
        end
      end

      kolich = 1;
    end /* Расчет сумм окончен */
    if (kolich > 0) then
    begin
      if ((:deliq_os > 0) or (:deliq_perc > 0)) then
      begin
        aa_deliq_repay_kolich = :aa_deliq_repay_kolich + 1;
        dk_os = :dk_os + :deliq_os;
        dk_perc = :dk_perc + :deliq_perc;
        if (dk_date = :rep_date_end) then
          dk_date = :dd_prior;
      end
    end
    if ((dk_os + dk_perc) >= 0.01) then
    begin
      if ((curr_code <> def_curr) and
          (report_currency = '000')) then
      begin
        dk_os = cast(dk_os * curr_rate as decimal(15,2));
        dk_perc = cast(dk_perc * curr_rate as decimal(15,2));
        dk_balance_begin = cast(dk_balance_begin * curr_rate as decimal(15,2));
        dk_balance_end = cast(dk_balance_end * curr_rate as decimal(15,2));
      end
      dk_days = cast((:rep_date_end - :dk_date) as integer);
      suspend;
    end
  end

end
