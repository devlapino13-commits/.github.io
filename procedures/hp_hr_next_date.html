CREATE PROCEDURE HP_HR_NEXT_DATE(
    DATE_BEGIN D_DATE,
    PLUS_DAYS D_INTEGER,
    WORK_TYPE D_INTEGER = 0,
    REG_ID D_INTEGER = null)
RETURNS (
    DATE_END D_DATE,
    NEXT_WORK_DATE D_DATE)
AS
declare variable week_workdays d_smallint;
declare variable sawt_include_holiday d_smallint;
declare variable rd_code d_str_code;
declare variable calendar_days d_bool;
begin
  if (work_type = -1) then
  begin
    sawt_include_holiday = 0;
    if (plus_days = 1) then
      plus_days = 2;
  end else
  begin
    select sawt_include_holiday, iif(sawt_base_type = 5 and sawt_is_paid = 0, 1, 0)
    from sa_worktype
    where sawt_id = :work_type
    into sawt_include_holiday, calendar_days;
  end

  week_workdays = 5;
  if (sawt_include_holiday = 1) then
  begin
    if (calendar_days = 0) then
      week_workdays = 7; -- включая сб и вс кроме праздников
    else
      week_workdays = -1; -- календарные дни
  end
  else
  begin
    if (reg_id is not null) then
    begin
      select rd_code
      from region, refdata
      where reg_unicode = :reg_id and
            reg_office_hours_id = rd_id
      into rd_code;

      if (rd_code starting 'office_hours_6') then
        week_workdays = 6;
      else
      if (rd_code starting 'office_hours_7') then
        week_workdays = 7;
    end
  end

  select date_end, next_work_date
  from hp_next_work_date(:date_begin, :plus_days, :week_workdays)
  into :date_end, :next_work_date;

  suspend;
end
