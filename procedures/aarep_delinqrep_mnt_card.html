CREATE PROCEDURE AAREP_DELINQREP_MNT_CARD(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID ID_TYPE,
    K_KONTRACT_NO STR_20,
    K_DISB_DATE DATE,
    C_NAME_FIO STR_100,
    METHOD_ISSUE STR_100,
    CARD_MASK STR_100)
AS
declare variable OEK_RD_ID integer;
begin
    oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

    for with tabl1
      as (select oks_from_k_id as k_id, oks_balance balance, :rep_date_end - oks_delinq_date as delinq_day,
                 oks_delinq_date delinq_date, 0 as zad_oko_os, 0 as zad_oko_perc, 0 as zad_oko_other,
                 null as oko_ol_date, 0 as oko_os, 0 as oko_perc, 0 as oko_other, null as date_woff_temp,0 as ox_amount_rez
          from oksummary
          where :rep_date_end between oks_date_begin and oks_date_end and
                :rep_date_end - oks_delinq_date > 0
                and oks_close_state = 0 
          union all
          select oko_from_k_id, 0 oks_balance, 0 delinq_day, null, oko_os + oko_sh_os, oko_perc + oko_sh_perc,
                 oko_other + oko_sh_other, iif(oko_type = 4, oko_ol_date, null), iif(oko_type = 4, oko_os, 0),
                 iif(oko_type = 4, oko_perc, 0), iif(oko_type = 4, oko_other, 0), oko_ol_date as date_woff_temp, 0 as ox_amount_rez
          from operkroff
          where oko_ol_is_active = 1 and
                oko_ol_date <= :rep_date_end
          union all
          select ox_object_id as k_id, 0 oks_balance, 0 delinq_day, null,0 as zad_oko_os, 0 as zad_oko_perc, 0 as zad_oko_other,
                 null as oko_ol_date, 0 as oko_os, 0 as oko_perc, 0 as oko_other, null as date_woff_temp , ox_amount_plus - ox_amount_minus - ox_amount_writeoff as ox_amount_rez
          from operextra
          where ox_ol_is_active = 1 and
              ox_from_rd_id = :oek_rd_id and
              ox_object_kind = f_objkind('kontract', 'id') and
              ox_ol_date<=:rep_date_end)

      select k_id
      from tabl1
      group by k_id
      having sum(balance) <> 0 or sum(zad_oko_os) <> 0 or sum(zad_oko_perc) <> 0 or sum(zad_oko_other) <> 0 or max(delinq_day) > 0 or sum(ox_amount_rez)<>0
      into k_id

      do
      begin

          for select k_kontract_no, k_disb_date, c_name_fio,
                 case k_disb_kind
                     when 5 then 'Банк.карта ' || iif(k_is_online=1, 'онлайн', 'офлайн')
                     when 7 then 'СБП'
                     when 8 then 'Расчетный счет'
                     else k_disb_kind
                   end
          from kontract
          inner join knclient on kc_from_k_id = k_id 
          inner join clients on c_id = kc_from_c_id
          where k_id = :k_id
          and k_disb_kind <> 1
          into k_kontract_no, k_disb_date, c_name_fio, method_issue

          do
          begin

              card_mask = null;
              select et_cardnumber
              from extsys_token, extsysmain
              where esm_from_k_id = :k_id
              and esm_account = et_token
              and esm_project = 'moneta_disb'
              and esm_action = 'carddisb'
              and esm_status = 1
              and esm_date <= :k_disb_date
              order by esm_date desc
              rows 1
              into card_mask;
    
              suspend;
          end
      end

end
