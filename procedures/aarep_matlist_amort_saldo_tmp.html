CREATE PROCEDURE AAREP_MATLIST_AMORT_SALDO_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_ID INTEGER,
    OM_BILL_ID DECIMAL(18,0),
    OM_BILL_LINE_NO INTEGER,
    OM_WORKPLACE_ID DECIMAL(18,0),
    OM_MATOTVET_ID DECIMAL(18,0),
    OM_ACCOUNT VARCHAR(20),
    SALDO_KOL_BEGIN DECIMAL(18,4),
    SALDO_SUM_BEGIN DECIMAL(18,4),
    SALDO_KOL_END DECIMAL(18,4),
    SALDO_SUM_END DECIMAL(18,4),
    OBOR_KOL_PLUS DECIMAL(18,4),
    OBOR_SUM_PLUS DECIMAL(18,4),
    OBOR_KOL_MINUS DECIMAL(18,4),
    OBOR_SUM_MINUS DECIMAL(18,4),
    REG_NAME VARCHAR(100),
    WORKPLACE_NAME VARCHAR(100),
    MATOTVET_FIO VARCHAR(100),
    OM_MATLIST_ID DECIMAL(18,0),
    OM_MATLIST_NAME VARCHAR(255),
    OM_MATCATEG_NAME VARCHAR(255),
    SALDO_AMORT_BEGIN DECIMAL(18,4),
    SALDO_AMORT_END DECIMAL(18,4),
    SALDO_AMORT_OBOR DECIMAL(18,4),
    WP_STATUS_STR VARCHAR(100),
    OM_IS_NEW VARCHAR(100),
    OM_ACC_AMORT VARCHAR(20),
    OB_INVENT_STR VARCHAR(20),
    OB_DATE_BEGIN DATE,
    LIFETIME_MONTH INTEGER,
    LIFETIME_MONTH_SALDO INTEGER)
AS
declare variable reg_id_prior integer;
declare variable om_workplace_id_prior decimal(18,0);
declare variable om_matotvet_id_prior decimal(18,0);
declare variable wp_status integer;
declare variable db_id d_id;
declare variable ob_date_begin_tmp d_date;
begin
  db_id = f_get_db_id();
  reg_id_prior = 0;
  om_workplace_id_prior = 0;
  om_matotvet_id_prior = 0;
  for select om_bill_id, om_bill_line_no, om_workplace_id, om_matotvet_id, om_account, reg_unicode, wp_status, sum(iif(ol_date < :rep_date_begin,
             (1 - om_is_written_off) * om_kolich_plus - om_kolich_minus, 0)), sum(iif(ol_date < :rep_date_begin,
             (1 - om_is_written_off) * om_summa_plus - om_summa_minus, 0)), sum(iif(ol_date <= :rep_date_end,
             (1 - om_is_written_off) * om_kolich_plus - om_kolich_minus, 0)), sum(iif(ol_date <= :rep_date_end,
             (1 - om_is_written_off) * om_summa_plus - om_summa_minus, 0)),
             sum(iif(ol_date between :rep_date_begin and :rep_date_end, om_kolich_plus, 0)),
             sum(iif(ol_date between :rep_date_begin and :rep_date_end, om_summa_plus, 0)),
             sum(iif(ol_date between :rep_date_begin and :rep_date_end, om_kolich_minus, 0)),
             sum(iif(ol_date between :rep_date_begin and :rep_date_end, om_summa_minus, 0))
      from operlist, opermat, region_list(:rep_region_id, 1, -1, 1), workplace, region
      where ol_id = om_from_ol_id and
            operlist.is_delete = 0 and
            ol_date <= :rep_date_end and
            om_workplace_id = wp_id and
            wp_from_reg_id = reg_unicode and
            reg_unicode = id_list and
            om_bill_id is not null
      group by om_bill_id, om_bill_line_no, om_workplace_id, om_matotvet_id, om_account, reg_unicode, wp_status
      into :om_bill_id, :om_bill_line_no, :om_workplace_id, :om_matotvet_id, :om_account, :reg_id, wp_status,
           :saldo_kol_begin, :saldo_sum_begin, :saldo_kol_end, :saldo_sum_end, :obor_kol_plus, :obor_sum_plus,
           :obor_kol_minus, :obor_sum_minus
  do
  begin
    if ((saldo_kol_begin <> 0) or (saldo_sum_begin <> 0) or (saldo_kol_end <> 0) or (saldo_sum_end <> 0) or (obor_kol_plus <> 0) or (obor_sum_plus <> 0) or (obor_kol_minus <> 0) or (obor_sum_minus <> 0)) then
    begin
      om_matlist_name = null;
      om_matcateg_name = null;
      select ob_from_ml_id, ml_name, mc_name, mc_acc_amort, ob_invent_str, ob_date_begin,
             iif(((ob_bill_id = ob_from_ol_id) and (ob_date_begin between :rep_date_begin and :rep_date_end)), 'Новый', 'Существующий')
      from operbill, matlist, matcateg
      where ob_bill_id = :om_bill_id and
            ob_bill_line_no = :om_bill_line_no and
            ob_date_begin <= :rep_date_end and
            ob_date_end >= :rep_date_end and
            operbill.ob_is_delete = 0 and
            ob_from_ml_id = ml_id and
            ob_from_mc_id = mc_id rows 1
      into :om_matlist_id, :om_matlist_name, :om_matcateg_name, :om_acc_amort, :ob_invent_str, :ob_date_begin,
           :om_is_new;

      if (om_matcateg_name is not null) then
      begin
        if (reg_id_prior <> reg_id) then
        begin
          select reg_code || ' ' || reg_name
          from region
          where reg_unicode = :reg_id
          into :reg_name;
          reg_id_prior = :reg_id;
        end
        if (om_workplace_id_prior <> om_workplace_id) then
        begin
          select wp_name
          from workplace
          where wp_id = :om_workplace_id
          into :workplace_name;
          om_workplace_id_prior = :om_workplace_id;
        end
        if (om_matotvet_id_prior <> om_matotvet_id) then
        begin
          select c_name_fio
          from clients
          where c_id = :om_matotvet_id
          into :matotvet_fio;
          om_matotvet_id_prior = :om_matotvet_id;
        end

        saldo_amort_begin = 0;
        saldo_amort_end = 0;
        saldo_amort_obor = 0;
        lifetime_month = null;
        select sum(iif(ol_date < :rep_date_begin, oal_summa_amort, 0)),
               sum(iif(ol_date <= :rep_date_end, oal_summa_amort, 0)),
               sum(iif(ol_date between :rep_date_begin and :rep_date_end, oal_summa_amort, 0)),
               max(oal_lifetime_month)
        from operlist, operamorlist
        where ol_id = oal_from_ol_id and
              operlist.is_delete = 0 and
              ol_date <= :rep_date_end and
              oal_bill_id = :om_bill_id and
              oal_bill_line_no = :om_bill_line_no
        into :saldo_amort_begin, :saldo_amort_end, :saldo_amort_obor, :lifetime_month;
        if (saldo_amort_begin is null) then
          saldo_amort_begin = 0;
        if (saldo_amort_end is null) then
          saldo_amort_end = 0;
        if (saldo_amort_obor is null) then
          saldo_amort_obor = 0;


        ob_date_begin_tmp = null;
        if (db_id = 94401) then  --Дата начало амортизации только для АЗ
        begin
          ob_date_begin_tmp = ob_date_begin;
          ob_date_begin = null;

          select ol.ol_date
          from opermat om
          inner join operlist ol on ol.ol_id = om.om_from_ol_id
          where om.om_bill_id = :om_bill_id and
                ol.ol_date <= :rep_date_end and
                om.om_bill_line_no = :om_bill_line_no and
                om.om_oper_type = 3
          order by ol.ol_date
          rows 1
          into :ob_date_begin;

          if (lifetime_month is not null) then
          begin
            lifetime_month_saldo = lifetime_month - (datediff(month, :ob_date_begin_tmp, :rep_date_end) + 1);
            lifetime_month_saldo = iif(coalesce(lifetime_month_saldo, 0) <= 0, 0, lifetime_month_saldo);
          end
          else
            lifetime_month_saldo = null;
        end
        else
        begin
        if (lifetime_month is not null) then
          begin
            lifetime_month_saldo = lifetime_month - (datediff(month, :ob_date_begin, :rep_date_end) + 1);
            lifetime_month_saldo = iif(coalesce(lifetime_month_saldo, 0) <= 0, 0, lifetime_month_saldo);
          end
          else
            lifetime_month_saldo = null;
        end

        if (saldo_sum_begin = 0) then
        begin
          saldo_amort_begin = 0;
        end
        if (saldo_sum_end = 0) then
        begin
          saldo_amort_end = 0;
          saldo_amort_obor = 0;
        end

        if (wp_status = 5) then
          wp_status_str = 'Рабочие места';
        if (wp_status = 6) then
          wp_status_str = 'Склады';
        if (wp_status = 7) then
          wp_status_str = 'Отделы';
        if (wp_status = 8) then
          wp_status_str = 'Трасферы';

        suspend;
      end
    end
  end
end
