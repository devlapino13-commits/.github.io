CREATE PROCEDURE EQUIFAX_COLLATERAL_DATA(
    K_ID ID_TYPE)
RETURNS (
    SUBJECT INTEGER,
    COLLATERAL_ID STR_255,
    COLLATERAL_NO STR_20,
    COLLATERAL_DATE DATE,
    COLLATERAL_ENDDATE DATE,
    ASSESSMENT DECIMAL(15,2),
    ASSESSMENT_DATE DATE)
AS
declare variable drl_id id_type;
declare variable zs_id integer;
begin
  for
    select
      zs_id,
      k_kontract_no,
      k_kontract_date,
      k_kredline_end,
      z_cost_likvid,
      k_disb_date,
      cds_from_drl_id
    from zalog
      inner join zalogsubtype
        on zs_id = z_subtype
      inner join kontract
        on k_id = z_from_k_id
      inner join knclient
        on kc_from_k_id = k_id
          and kc_amount_approved > 0
      left join clientdocs
        on cds_from_c_id = kc_from_c_id
          and cast(cds_create_datetime as date) <= k_disb_date -- прикреплен не позднее дня выдачи кредита
          and cds_from_drl_id > 0
          and cds_from_pk_id = 315 -- Паспорт трансп. средства (ПТС)
    where z_from_k_id = :k_id
      and z_type = 1
    order by cds_is_delete, cds_create_datetime desc   -- последний прикрепленный не аннулированный, если все аннулированы, то последний из аннулированных
    rows 1
  into
    :zs_id,
    :collateral_no,
    :collateral_date,
    :collateral_enddate,
    :assessment,
    :assessment_date,
    :drl_id
  do
  begin
    if (drl_id is not null) then begin
      collateral_id = f_docrecog_value(:drl_id, 'vin');

      if (zs_id = 1) then
      begin
        subject = 1;
      end

      subject = (case zs_id when 2 then 1 -- FS: Недвижимость =  Equifax: Недвижимость
                            when 3 then 3 -- FS: Оборудования =  Equifax: Машина, оборудование
                            when 4 then 2 -- FS: Транспортные средства = Equifax: Транспортное средство
                            when 5 then 8 -- FS: Товары = Equifax: Товар
                            when 6 then 6 -- FS: Патенты, лицензии = Equifax: Ценные бумаги
                 end);
      suspend;
    end
  end
end
