CREATE PROCEDURE FM_KONTMAT_OWN_UPDATE(
    KML_ID D_ID,
    K_ID D_ID,
    C_ID D_ID,
    REG_ID D_INTEGER,
    WP_ID D_ID,
    MOTV_ID D_ID,
    BILL_ID D_ID,
    BILL_LINE_NO D_INTEGER,
    ML_ID D_ID,
    QUANTITY D_CURRENCY,
    COST D_CURRENCY)
RETURNS (
    ERR_TEXT VARCHAR(255))
AS
declare variable kol d_integer;
declare variable kml_tmp_id d_id;
declare variable total d_currency;
declare variable ag_id d_id;
begin
  err_text = '';
  if (coalesce(c_id, 0) = 0) then
    err_text = 'Не указан продавец';
  if (coalesce(ml_id, 0) = 0) then
    err_text = 'Не указано название товара';
  ag_id = (select first 1 ag_id
           from hp_get_agreement(:k_id)
           where ag_object_id = :c_id
           order by ag_id desc);
  if (ag_id is null) then
    err_text = 'Договор с контрагентом не определен.';
  select hg.quantity
  from hp_get_own_good(:c_id) hg
  where hg.ml_id = :ml_id and
        hg.bill_id = :bill_id and
        hg.bill_line_no = :bill_line_no and
        hg.wp_id = :wp_id and
        hg.motv_id = :motv_id
  into kol;
  if (kol < :quantity) then
    err_text = 'Для данной номенклатуры, количество превышает максимально допустимое';

  if (err_text <> '') then
  begin
    suspend;
    exit;
  end
  if (:kml_id is null) then
  begin
    insert into kontmat (kml_object_kind, kml_object_id, kml_from_ml_id, kml_count, kml_cost, kml_bill_id,
                         kml_bill_line_no, kml_from_ag_id)
    values (9, :k_id, :ml_id, :quantity, :cost, :bill_id, :bill_line_no, :ag_id)
    returning kml_id, kml_total
    into :kml_tmp_id, :total;
    insert into kontmat_own (kmo_kml_id, kmo_k_id, kmo_c_id, kmo_reg_id, kmo_wp_id, kmo_motv_id, kmo_bill_id,
                             kmo_bill_line_no, kmo_ml_id, kmo_quantity, kmo_total)
    values (:kml_tmp_id, :k_id, :c_id, :reg_id, :wp_id, :motv_id, :bill_id, :bill_line_no, :ml_id, :quantity, :total);
  end
  else
  begin
    update or insert into kontmat (kml_id, kml_object_kind, kml_object_id, kml_from_ml_id, kml_count, kml_cost,
                                   kml_bill_id, kml_bill_line_no, kml_from_ag_id)
    values (:kml_id, 9, :k_id, :ml_id, :quantity, :cost, :bill_id, :bill_line_no, :ag_id)
    matching (kml_id);
    update or insert into kontmat_own (kmo_kml_id, kmo_k_id, kmo_c_id, kmo_reg_id, kmo_wp_id, kmo_motv_id, kmo_bill_id,
                                       kmo_bill_line_no, kmo_ml_id, kmo_quantity, kmo_total)
    values (:kml_id, :k_id, :c_id, :reg_id, :wp_id, :motv_id, :bill_id, :bill_line_no, :ml_id, :quantity,
            (:quantity * :cost))
    matching (kmo_kml_id);
  end
  suspend;
end
