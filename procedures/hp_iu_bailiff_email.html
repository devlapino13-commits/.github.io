CREATE PROCEDURE HP_IU_BAILIFF_EMAIL(
    RD_ID D_INTEGER,
    EMAIL D_STR_255,
    MEMO D_STR_255,
    IS_ACTIVE D_SMALLINT)
AS
declare variable rd_parent_id d_integer;
declare variable orderno d_integer;
declare variable rd_code d_str_255;
declare variable rd_id_old d_integer;
declare variable is_active_old d_smallint;
begin
  if ((select result
       from hp_check_email_domens(:email)) = 0) then
    exception e_message 'Не допустимое значение почтового домена.';

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'Bailiff.Email' and
        rd.rd_parent_id is null
  into :rd_parent_id;

  if (:rd_id = 0) then
  begin
    select count(*)
    from refdata rd
    where rd.rd_main_code = 'Bailiff.Email'
    into :orderno;

    select rd.rd_id, rd.rd_is_active, rd.rd_code
    from refdata rd
    where rd.rd_main_code = 'Bailiff.Email' and
          rd.rd_name = :email
    into :rd_id_old, :is_active_old, :rd_code;

    if (:rd_id_old is not null) then
    begin
      if (is_active_old = 0) then
      begin
        rd_id = :rd_id_old;
        is_active = 1;
      end
      else
        exception e_message 'Указанный почтовый адрес существует.';
    end
  end
  else
    select rd.rd_code, rd.rd_orderno
    from refdata rd
    where rd.rd_id = :rd_id
    into :rd_code, :orderno;

  update or insert into refdata (rd_id, rd_parent_id, rd_code, rd_main_code, rd_name, rd_memo, rd_orderno, rd_is_active,
                                 rd_is_locked, rd_str_id, rd_tag)
  values (iif(:rd_id = 0, null, :rd_id), :rd_parent_id,
          iif((:rd_id = 0 and :rd_code is null), 'Bailiff.Email.' || :orderno, :rd_code), 'Bailiff.Email', :email, :memo,
          :orderno, :is_active, -1, '', 0)
  matching (rd_id);
end
