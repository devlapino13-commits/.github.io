CREATE PROCEDURE HP_MOVE_ACTIONGROUP_ORDER(
    ACG_ID INTEGER,
    MOVE_DIRECTION INTEGER)
AS
declare variable acg_parent_id integer;
declare variable tmp_step integer;
declare variable acg_orderno integer;
declare variable next_acg_id integer;
declare variable acg_group_rd_id integer;
begin
  select acg_parent_id, acg_orderno, acg_group_rd_id
  from actiongroup
  where acg_id = :acg_id
  into :acg_parent_id, acg_orderno, acg_group_rd_id;

  tmp_step = 1;
  if (acg_parent_id is null) then
    tmp_step = 100;
  tmp_step = tmp_step * move_direction;

  select acg_id
  from actiongroup
  where acg_orderno = (:acg_orderno + :tmp_step) and
        coalesce(acg_parent_id, 0) = coalesce(:acg_parent_id, 0) and
        acg_group_rd_id = :acg_group_rd_id
  into :next_acg_id;

  if (next_acg_id is not null) then
  begin
    update actiongroup
    set acg_orderno = acg_orderno + :tmp_step
    where acg_id = :acg_id;

    update actiongroup
    set acg_orderno = acg_orderno - :tmp_step
    where acg_id = :next_acg_id;
  end
end
