CREATE PROCEDURE HP_OPERVAC_REPAYMENT(
    REP_K_ID D_ID,
    REP_OL_ID D_ID)
RETURNS (
    RES_NUMBER D_INTEGER,
    RES_TYPE D_STR_20,
    RES_SH_ID D_ID,
    RES_SH_DATE D_DATE,
    RES_PAY_OS D_CURRENCY,
    RES_PAY_PERC D_CURRENCY,
    RES_PAY_FINE D_CURRENCY,
    RES_ALL D_CURRENCY)
AS
declare variable ok_type d_integer;
begin
  for select ok_number, ok_type, ok_sh_id, ok_sh_date, ok_os, ok_perc, ok_fine
  from operkred
  where ok_from_k_id = :rep_k_id and
        ok_from_ol_id = :rep_ol_id and
        ok_type in (4, 5, 8) and
        (ok_os <> 0 or ok_perc <> 0 or ok_fine <> 0)
  order by ok_sh_date, ok_sh_id
  into res_number, ok_type, res_sh_id, res_sh_date, res_pay_os, res_pay_perc, res_pay_fine
  do
  begin
    res_type = '';
    res_type = trim(case ok_type
      when 4 then 'Погашен'
      when 5 then 'Списан'
      when 8 then 'Ввыдача'
    end);

    res_all = res_pay_os + res_pay_perc + res_pay_fine;
    suspend;
  end
end
