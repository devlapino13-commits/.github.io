CREATE PROCEDURE AAREP_BUDOPER_REGION(
    REP_REGION_ID INTEGER,
    REP_YEAR INTEGER,
    REP_WITH_RESERV INTEGER,
    REP_DATE DATE,
    REP_WITH_FACT INTEGER)
RETURNS (
    BIT_ID INTEGER,
    BIT_PARENT INTEGER,
    BIT_CODE VARCHAR(20),
    BIT_NAME VARCHAR(255),
    BIT_MEMO VARCHAR(1024),
    BIT_HAS_CHILD INTEGER,
    BOI_REAL_01 DECIMAL(18,4),
    BOI_REAL_02 DECIMAL(18,4),
    BOI_REAL_03 DECIMAL(18,4),
    BOI_REAL_04 DECIMAL(18,4),
    BOI_REAL_05 DECIMAL(18,4),
    BOI_REAL_06 DECIMAL(18,4),
    BOI_REAL_07 DECIMAL(18,4),
    BOI_REAL_08 DECIMAL(18,4),
    BOI_REAL_09 DECIMAL(18,4),
    BOI_REAL_10 DECIMAL(18,4),
    BOI_REAL_11 DECIMAL(18,4),
    BOI_REAL_12 DECIMAL(18,4),
    BOI_OPTIM_01 DECIMAL(18,4),
    BOI_OPTIM_02 DECIMAL(18,4),
    BOI_OPTIM_03 DECIMAL(18,4),
    BOI_OPTIM_04 DECIMAL(18,4),
    BOI_OPTIM_05 DECIMAL(18,4),

    BOI_OPTIM_10 DECIMAL(18,4),

    BOI_OPTIM_12 DECIMAL(18,4),
    BOI_PESSIM_01 DECIMAL(18,4),
    BOI_PESSIM_02 DECIMAL(18,4),
    BOI_PESSIM_03 DECIMAL(18,4),
    BOI_PESSIM_04 DECIMAL(18,4),
    BOI_PESSIM_05 DECIMAL(18,4),
    BOI_PESSIM_06 DECIMAL(18,4),
    BOI_PESSIM_07 DECIMAL(18,4),
    BOI_PESSIM_08 DECIMAL(18,4),
    BOI_PESSIM_09 DECIMAL(18,4),
    BOI_PESSIM_10 DECIMAL(18,4),
    BOI_PESSIM_11 DECIMAL(18,4),
    BOI_PESSIM_12 DECIMAL(18,4),
    BOI_FACT_01 DECIMAL(18,4),
    BOI_FACT_02 DECIMAL(18,4),
    BOI_FACT_03 DECIMAL(18,4),
    BOI_FACT_04 DECIMAL(18,4),
    BOI_FACT_05 DECIMAL(18,4),
    BOI_FACT_06 DECIMAL(18,4),
    BOI_FACT_07 DECIMAL(18,4),
    BOI_FACT_08 DECIMAL(18,4),
    BOI_FACT_09 DECIMAL(18,4),
    BOI_FACT_10 DECIMAL(18,4),
    BOI_FACT_11 DECIMAL(18,4),
    BOI_FACT_12 DECIMAL(18,4))
AS
begin
  for select boi_from_bit_id, sum(boi_real_01), sum(boi_real_02), sum(boi_real_03), sum(boi_real_04), sum(boi_real_05),
             sum(boi_real_06), sum(boi_real_07), sum(boi_real_08), sum(boi_real_09), sum(boi_real_10), sum(boi_real_11),
             sum(boi_real_12), sum(boi_optim_01), sum(boi_optim_02), sum(boi_optim_03), sum(boi_optim_04),
             sum(boi_optim_05), sum(boi_optim_06), sum(boi_optim_07), sum(boi_optim_08), sum(boi_optim_09),
             sum(boi_optim_10), sum(boi_optim_11), sum(boi_optim_12), sum(boi_pessim_01), sum(boi_pessim_02),
             sum(boi_pessim_03), sum(boi_pessim_04), sum(boi_pessim_05), sum(boi_pessim_06), sum(boi_pessim_07),
             sum(boi_pessim_08), sum(boi_pessim_09), sum(boi_pessim_10), sum(boi_pessim_11), sum(boi_pessim_12),
             sum(boi_fact_01), sum(boi_fact_02), sum(boi_fact_03), sum(boi_fact_04), sum(boi_fact_05), sum(boi_fact_06),
             sum(boi_fact_07), sum(boi_fact_08), sum(boi_fact_09), sum(boi_fact_10), sum(boi_fact_11), sum(boi_fact_12)

      from aarep_budoper_region_tmp(:rep_region_id, :rep_year, :rep_with_reserv, :rep_date, :rep_with_fact)
      group by boi_from_bit_id
      into :bit_id, :boi_real_01, :boi_real_02, :boi_real_03, :boi_real_04, :boi_real_05, :boi_real_06, :boi_real_07,
           :boi_real_08, :boi_real_09, :boi_real_10, :boi_real_11, :boi_real_12, :boi_optim_01, :boi_optim_02,
           :boi_optim_03, :boi_optim_04, :boi_optim_05, :boi_optim_06, :boi_optim_07, :boi_optim_08, :boi_optim_09,
           :boi_optim_10, :boi_optim_11, :boi_optim_12, :boi_pessim_01, :boi_pessim_02, :boi_pessim_03, :boi_pessim_04,
           :boi_pessim_05, :boi_pessim_06, :boi_pessim_07, :boi_pessim_08, :boi_pessim_09, :boi_pessim_10,
           :boi_pessim_11, :boi_pessim_12, :boi_fact_01, :boi_fact_02, :boi_fact_03, :boi_fact_04, :boi_fact_05,
           :boi_fact_06, :boi_fact_07, :boi_fact_08, :boi_fact_09, :boi_fact_10, :boi_fact_11, :boi_fact_12
  do
  begin
    select bit_parent, bit_code, bit_name, bit_memo, bit_has_child
    from buditem
    where bit_id = :bit_id
    into :bit_parent, :bit_code, :bit_name, :bit_memo, :bit_has_child;

    suspend;
  end
end
