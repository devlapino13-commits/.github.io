CREATE PROCEDURE PRINT_PAYMENT_MONITORING_CLIENT(
    OBJ_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_100,
    C_AMOUNT DECIMAL(18,2),
    C_CIRCLE D_INTEGER,
    C_STATUS_BEGIN D_STR_100,
    C_STATUS_AFTER D_STR_100,
    C_LOANS_ACTIVE D_INTEGER,
    C_LOANS_PROCESS D_INTEGER,
    C_LOANS_CLOSED D_INTEGER)
AS
declare variable rep_date d_date;
begin
  rep_date = current_date;

  for select t.c_name_fio, t.c_amount, t.c_circle, t.c_status_begin, t.c_status_after, t.c_loans_active,
             t.c_loans_process, t.c_loans_closed
      from kontract_client_status(:obj_id, :rep_date) t
      into :c_name_fio, :c_amount, :c_circle, :c_status_begin, :c_status_after, :c_loans_active, :c_loans_process,
           :c_loans_closed
  do
  begin
    suspend;
  end
end
