CREATE PROCEDURE HP_EPAY_LIST_MONETA_DISB(
    EPT_ID ID_TYPE)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable token str_100;
declare variable c_id d_id;
declare variable ept_amount d_currency;
declare variable esm_id d_id;
declare variable esm_status d_smallint;
declare variable c_name_fio d_str_255;
declare variable c_comment d_str_100;
begin
  res_text = '';

  select ept_from_esm_id, ept_from_c_id, ept_amount
  from epay_trans
  where ept_id = :ept_id
  into esm_id, c_id, ept_amount;

  -- Проверка по ПОД/ФТ
  for select c_name_fio, c_comment
      from hp_check_client_fio_ft(:c_id, 0)
      into c_name_fio, c_comment
  do
  begin
    res_text = 'Совпадение с лицом по ' || c_comment || ' ' || c_name_fio;
    break;
  end

  if (exists(select *
             from extsysmain em
             where em.esm_project = 'moneta_disb' and
                   em.esm_action = 'CardDisb' and
                   em.esm_id = :esm_id and
                   em.esm_status in (0, 1))) then
    res_text = 'Оплата уже в процессе';

  if (res_text = '') then
  begin
    esm_id = null;

    token = (select token
             from hp_client_bank_card(:c_id));
  
    if (token is null) then
    begin
      res_text = 'Банковская карта не привязана';
    end else
    begin
      insert into extsysmain (esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
                              esm_full_source, esm_amount, esm_from_k_id, esm_account)
      values ('moneta_disb', 'CardDisb', current_date, 0, :c_id, 0, '', :ept_amount, null, :token)
      returning esm_id, esm_status
      into esm_id, esm_status;

      update epay_trans
      set ept_from_esm_id = :esm_id,
          ept_status = :esm_status
      where ept_id = :ept_id;
    end
  end

  suspend;
end
