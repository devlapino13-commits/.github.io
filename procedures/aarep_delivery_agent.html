CREATE PROCEDURE AAREP_DELIVERY_AGENT(
    DATE_BEGIN DATE,
    DATE_END DATE)
RETURNS (
    KONTRACT_NOMER D_STR_20,
    DATA_VYDACHY DATE,
    SUMMA_ZAIMA CURR_TYPE,
    FIO_KS STR_100,
    REGON_R STR_100,
    FIO_AGENT STR_100,
    ID_AGENT ID_TYPE,
    FIO_CLIENT STR_100,
    DATE_CLOSE_SCHEDULE DATE,
    DATE_CLOSE_FACT DATE)
AS
declare variable k_id id_type;
begin
  for select K.K_KONTRACT_NO, K.K_FACT_DISB_DATE, KNC.KC_AMOUNT_APPROVED, CLL.C_NAME_FIO, REG_NAME,cl3.c_name_fio,ag.ag_object_id,cl.c_name_fio,k.k_kredline_end,
  k.k_fact_closed_date
      from KONTSTATE KS
      inner join OPERAGR O on OGR_FROM_OL_ID = KS_FROM_OL_ID
      inner join KONTRACT K on K.K_ID = KS.KS_FROM_K_ID
      inner join KNCLIENT KNC on KNC.KC_FROM_K_ID = K.K_ID
      inner join CLIENTS CL on CL.C_ID = KNC.KC_FROM_C_ID
      inner join CLIENTS CLL on CLL.C_ID = K.K_FACT_DISB_LO_ID
      inner join KNAKTLIST KAL on KAL.KAL_FROM_KF_ID = K.K_FROM_KF_ID
      inner join REGION REG on REG.REG_UNICODE = KAL.KAL_NEW_REG_ID
      inner join agreements ag on ag.ag_id = o.ogr_from_ag_id
      inner join refdata r on r.rd_id = ag.ag_from_rd_id
      inner join clients cl3 on cl3.c_id = ag.ag_object_id
      where KS_OL_IS_DELETE = 0 and
            KS_STATE = 1 and
            OGR_SIGN = 1 and
            KS_DATE_BEGIN between :DATE_BEGIN and :DATE_END and
            OGR_AMOUNT > 0 and
            KNC.KC_AMOUNT_APPROVED > 0 and
            :DATE_END between KAL.KAL_DATE and KAL.KAL_DATE_END and
            KAL.IS_DELETE = 0 and -- передача папок не должна быть удаленной
            r.rd_code = 'AgreementKinds.loyal_agreement'
      into :KONTRACT_NOMER, :DATA_VYDACHY, :SUMMA_ZAIMA, :FIO_KS, :REGON_R,FIO_agent,id_agent,FIO_CLIENT,DATE_CLOSE_SCHEDULE,DATE_CLOSE_FACT

  do
  begin
    suspend;
  end
end
