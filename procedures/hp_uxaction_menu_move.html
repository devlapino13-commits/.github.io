Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_UXACTION_MENU_MOVE(
    UX_CODE D_STR_CODE,
    UA_CODE D_STR_100,
    MOVE_SIGN D_INTEGER)
AS
declare variable len d_integer;
declare variable menu_index d_str_code;
begin
  select ua_menu_index
  from uxaction
  where ua_code = :ua_code
  into menu_index;

  if (menu_index is null) then
    exit;

  len = char_length(menu_index);

  if (move_sign = 1) then
  begin
    update uxaction
    set ua_menu_index = left(ua_menu_index, :len - 2) ||
                       right('0' || cast((cast(substring(ua_menu_index from :len - 1 for 2) as integer) + :move_sign) as d_str_code), 2) ||
                       substring(ua_menu_index from :len + 1)
    where ua_ux_code = :ux_code and
          ua_menu_index starting left(:menu_index, :len - 2) and
          ua_menu_index >= :menu_index and
          ua_code <> :ua_code
    order by ua_menu_index desc;
  end else
  begin
    update uxaction
    set ua_menu_index = left(ua_menu_index, :len - 2) ||
                       right('0' || cast((cast(substring(ua_menu_index from :len - 1 for 2) as integer) + :move_sign) as d_str_code), 2) ||
                       substring(ua_menu_index from :len + 1)
    where ua_ux_code = :ux_code and
          ua_menu_index starting left(:menu_index, :len - 2) and
          ua_menu_index >= :menu_index and
          ua_code <> :ua_code
    order by ua_menu_index;
  end
end
