CREATE PROCEDURE AAREP_SSVZ_DELINQ_TEMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REP_SVZ_NAME VARCHAR(100),
    REP_KF_NAME VARCHAR(50),
    REP_CATEGORY SMALLINT,
    REP_DELINQ_DAY INTEGER,
    REP_DELINQ_BALANCE CURR_TYPE,
    REP_SVZ_JOB_SYSTEM VARCHAR(100),
    REP_OFFICE_ID INTEGER,
    REP_BRANCH_ID INTEGER,
    REP_OKRUG_ID INTEGER,
    REP_DELINQ_OS CURR_TYPE,
    REP_DELINQ_PERC CURR_TYPE,
    REP_DELINQ_SUM CURR_TYPE,
    REP_MEMO VARCHAR(50),
    REP_DATE_MATURITY DATE,
    REP_SUM_MATURITY CURR_TYPE)
AS
declare variable rep_svz_id decimal(18,0);
declare variable rep_k_id decimal(18,0);
declare variable rep_svz_id_prior id_type;
declare variable rep_office_id_prior integer;
begin
  rep_svz_id_prior = 0;
  rep_office_id_prior = 0;
-- Вытаскиваем все просроченныхе кредтиы в данном регионе
-- чтобы не включать просрчоенные по категории 3, заранее будем брать просроченные до 180 дней включительно
  for select oks_kc_id, oks_from_k_id, cast(:rep_date_end as date) - oks_delinq_date, oks_balance, oks_region_id
      from oksummary, region_list(:rep_region_id, 1, 3, 1)
      where oks_balance > 0 and
            :rep_date_end between oks_date_begin and oks_date_end and
            :rep_date_end >= oks_delinq_date and
            :rep_date_end - 180 <= oks_delinq_date -- просроченные свыше 180 не брать
            and
            oks_region_id = id_list and
            oks_close_state = 0
      order by oks_region_id, oks_kc_id
      into rep_svz_id, rep_k_id, rep_delinq_day, rep_delinq_balance, rep_office_id
  do
  begin
--   Категория
    rep_category = (select aa_category_id
                    from aarep_get_category_id(:rep_k_id, :rep_date_end, null, null, null, null, null, null));
    if (rep_category < 3) then
    begin
      if (rep_svz_id_prior <> rep_svz_id) then
      begin
        rep_svz_id_prior = :rep_svz_id;
        rep_svz_name = (select c_name_fio
                        from clients
                        where c_id = :rep_svz_id);
--     Системная должность
        rep_svz_job_system = null;
        select rd_name
        from hr_pstatus, joblist, jobkind, refdata
        where ps_from_c_id = :rep_svz_id and
              ps_is_working = 1 and
              ps_status = 1 and
              :rep_date_end between ps_date_begin and ps_date_end and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_id = rd_id
        into rep_svz_job_system;
        if (rep_svz_job_system is null) then
          rep_svz_job_system = 'Не определен';
      end
--       Название папки
      rep_kf_name = null;
      select kal_new_kf_name
      from kontract, knaktlist
      where k_id = :rep_k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end
      into rep_kf_name;
--       Просроченная сумма
      with tabl1
      as (select ok_sh_date, sum(ok_os + ok_sh_os) zad_os, sum(ok_perc + ok_sh_perc) zad_perc
          from operkred
          where ok_from_k_id = :rep_k_id and
                ok_ol_is_active = 1 and
                ok_ol_date <= :rep_date_end and
                ok_type in (1, 2, 3, 4, 5, 6, 9) and
                ok_from_c_id = 0
          group by ok_sh_date
          having sum(ok_os + ok_sh_os) + sum(ok_perc + ok_sh_perc) <> 0)
      select sum(zad_os), sum(zad_perc)
      from tabl1
      where ok_sh_date <= :rep_date_end
      into rep_delinq_os, rep_delinq_perc;
      if (rep_delinq_os is null) then
        rep_delinq_os = 0;
      if (rep_delinq_perc is null) then
        rep_delinq_perc = 0;
      rep_delinq_sum = rep_delinq_os + rep_delinq_perc;
--       Проверка на реструктуризацию
      rep_memo = null;
      if (exists(select *
                 from operrest, operlist
                 where or_from_ol_id = ol_id and
                       operlist.is_delete = 0 and
                       ol_date_oper_day <= :rep_date_end and
                       or_from_k_id = :rep_k_id)) then
        rep_memo = 'Реструтуризирован';
--    Дата и сумма последнего погашения
      rep_date_maturity = null;
      rep_sum_maturity = null;
      with tabl1
      as (select ok_ol_date as ol_date
          from operkred
          where ok_ol_is_active = 1 and
                ok_type = 4 and
                ok_os + ok_perc < 0 and
                ok_from_k_id = :rep_k_id and
                ok_ol_date <= :rep_date_end and
                ok_from_c_id = 0
          order by ok_ol_date desc
          rows 1)
      select ol_date, -sum(ok_os + ok_perc)
      from tabl1, operkred
      where ok_ol_is_active = 1 and
            ok_type = 4 and
            ok_ol_date = ol_date and
            ok_from_k_id = :rep_k_id and
            ok_from_c_id = 0
      group by ol_date
      into rep_date_maturity, rep_sum_maturity;
      if (rep_sum_maturity is null) then
        rep_sum_maturity = 0;
--    Округ и регион для сортировки
      if (rep_office_id_prior <> rep_office_id) then
      begin
        rep_office_id_prior = rep_office_id;
        rep_branch_id = (select parent
                         from region
                         where reg_unicode = :rep_office_id);
        rep_okrug_id = (select parent
                        from region
                        where reg_unicode = :rep_branch_id);
      end
      suspend;
    end
  end
end
