CREATE PROCEDURE SAVE_CFP_KPD(
    MAC_ID D_ID,
    ACL_ID D_ID = 0)
RETURNS (
    KPD_ERR D_STR_255)
AS
declare variable MA_ID D_ID;
declare variable C_ID D_ID;
declare variable MAC_AMOUNT D_CURRENCY;
declare variable CA_ID D_ID;
declare variable K_ID D_ID;
declare variable KBM_ID D_ID;
declare variable KPD D_CURRENCY;
declare variable CA_AVR_INCOME decimal(15,2);
declare variable EXTRA_AMOUNT decimal(15,2);
declare variable COMMUNAL_EXPENSES decimal(15,2);
declare variable LIVING_EXPENSES decimal(15,2);
declare variable LEGAL_DOCS_EXPENSES decimal(15,2);
declare variable OTHER_BUSINESS_EXPENSES decimal(15,2);
declare variable REPAYMENT_OBLIGATIONS decimal(15,2);
declare variable NET_PROFIT decimal(15,2);
declare variable AMOUNT_MONTHLY_PAYMENTS decimal(15,2);
declare variable KBM_REL_ID D_ID;
declare variable croc_id d_id;
begin
  for select mac.mac_from_ma_id, mac.mac_from_c_id, mac.mac_amount
      from miniappclient mac
      inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id and
            ma.ma_is_active = 1 and
            coalesce(ma.ma_date_end, '31.12.9999') >= '31.12.2999'
      where mac.mac_id = :mac_id and
            mac.mac_amount > 0
      into ma_id, c_id, mac_amount
  do
  begin
    select ca.ca_id, ca.ca_avr_income
    from cfp_anketa ca
    where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
          ca.ca_object_id = :mac_id and
          ca.ca_from_c_id = :c_id and
          (
            coalesce(:acl_id, 0) = 0
            or ca.ca_from_acl_id = :acl_id
          ) and
          (
            coalesce(:acl_id, 0) <> 0
            or ca.ca_from_acl_id is null
          )
    into ca_id, ca_avr_income;

    select k.k_id
    from kontract k
    where k.k_from_ma_app_id = :ma_id
    into k_id;

    k_id = coalesce(k_id, 0);
    extra_amount = (select sum(caa.caa_value)
                    from cfp_anketa_answers caa
                    inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                          cq.cq_code_name in ('cl_add_income_spouse_salary', 'cl_add_income_household', 'cl_add_income_service')
                    where caa.caa_from_ca_id = :ca_id);
    communal_expenses = (select caa.caa_value
                         from cfp_anketa_answers caa
                         inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                               cq.cq_code_name = 'monthly_housing_expenses'
                         where caa.caa_from_ca_id = :ca_id);
    living_expenses = (select caa.caa_value
                       from cfp_anketa_answers caa
                       inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                             cq.cq_code_name = 'monthly_family_expenses'
                       where caa.caa_from_ca_id = :ca_id);
    legal_docs_expenses = (select caa.caa_value
                           from cfp_anketa_answers caa
                           inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                                 cq.cq_code_name = 'monthly_tax_expenses'
                           where caa.caa_from_ca_id = :ca_id);
    other_business_expenses = (select caa.caa_value
                               from cfp_anketa_answers caa
                               inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                                     cq.cq_code_name = 'monthly_other_expenses'
                               where caa.caa_from_ca_id = :ca_id);

    select repayment_obligations,
           kbm_id,                                       -- (Погашение других долговых обязательств)
           kbm_rel_id,
           croc_id,
           text_error
    from hp_calc_repayment_obligations(:ma_id, :acl_id)
    into repayment_obligations,
         kbm_id,
         kbm_rel_id,
         croc_id,
         kpd_err;

   if (coalesce(kpd_err, '') <> '') then
   begin
     suspend;
     exit;
   end

    net_profit = ca_avr_income + extra_amount - communal_expenses - living_expenses - legal_docs_expenses - other_business_expenses - repayment_obligations;
    amount_monthly_payments = (select avg(sh.sh_os + sh.sh_perc + (sh.sh_perc * 0.02) + sh.sh_kommis + sh.sh_perc_tax)
                               from schedule sh
                               where sh.sh_from_k_id = iif(:k_id > 0, :k_id, -:ma_id) and
                                     sh.sh_id > 0);
    kpd = (net_profit / amount_monthly_payments) * 100;
    kpd = coalesce(kpd, 0);

    if (:kpd < 150 and coalesce(:acl_id, 0) = 0) then    -- RFCKG-5146
    begin
      kpd_err = 'КПД должно быть > 150% '; -- Для всех продуктов
      suspend;
      exit;
    end
--    if ((mac_amount > 200000) and   -- RFCKG-5146
--        (:kpd < 140)) then
--    begin
--      kpd_err = 'КПД должно быть не менее 140% , т.к. сумма займа более 200 000'; -- для всех продуктов
--      suspend;
--      exit;
--    end

    update cfp_repayment_oblig_calc croc
       set croc.croc_is_actual = 1
    where croc.croc_id = :croc_id;

    update or insert into cfp_anketa_kpd (cak_from_ca_id, cak_kpd, cak_net_profit, cak_other_dept, cak_monthly_payment,
                                          cak_kbm_id, cak_rel_kbm_id, cak_change_datetime)
    values (:ca_id, :kpd, :net_profit, :repayment_obligations, :amount_monthly_payments,
            iif(:kbm_id = 0, null, :kbm_id), iif(:kbm_rel_id = 0, null, :kbm_rel_id), current_timestamp)
    matching (cak_from_ca_id);
    suspend;
  end
end
