CREATE PROCEDURE HP_HR_PSTATUS_RELATIVES(
    ORDER_ID ID_TYPE)
RETURNS (
    REL_NAME STR_100,
    C_NAME_FIO STR_255,
    C_BIRTH_DATE DATE)
AS
declare variable c_id id_type;
declare variable has_rec smallint;
begin
  for select ps_from_c_id
      from hr_pstatus ps
      where ps_from_ho_id = :order_id
      into :c_id
  do
  begin
    has_rec = 0;
    for select rel_name, c_name_fio, c_birth_date
        from hp_client_relation_list(:c_id, 0)
        where crel_from_rel_id in ('4', '3', '8', '7', '5', '6') rows 6 -- мать, отца, сестру, брата, детей
        into :rel_name, :c_name_fio, :c_birth_date
    do
    begin
      has_rec = 1;
      suspend;
    end

    if (has_rec = 0) then
    begin
        rel_name = '';
        c_name_fio = '';
        c_birth_date = null;
        suspend;
    end
  end
end
