CREATE PROCEDURE AAREP_OKA_DELINQ_PAYMENT_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    APL_DATE DATE,
    APL_REGION VARCHAR(100),
    APL_CLIENT_NAME VARCHAR(100),
    APL_MEMO VARCHAR(1024),
    APL_DATETIME TIMESTAMP,
    APL_USER_NAME VARCHAR(100),
    APL_OS DECIMAL(18,4),
    APL_PERC DECIMAL(18,4),
    APL_FINE DECIMAL(18,4),
    APL_KF_NAME VARCHAR(100),
    APL_KOL_DAY INTEGER,
    APL_SH_DATE DATE,
    K_KONTRACT_CATEGORY VARCHAR(50),
    APL_TOTAL_SUM DECIMAL(18,4),
    APL_LO_FIO VARCHAR(100),
    APL_FACT_DISB_DATE DATE)
AS
declare variable apl_ol_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
declare variable k_id decimal(18,0);
declare variable k_is_problem decimal(18,0);
declare variable def_curr varchar(3);
declare variable k_from_pr_id integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select cm_default_currency
  from company
  into :def_curr;
  for select ol_id, ol_date, reg_code || ' ' || reg_name, left(ol_doc_client_name, 100), ol_memo, ol_datetime,
             un.c_name_fio, lo.c_name_fio, kal_new_kf_name, ok_sh_date, k_id, k_is_problem, k_from_pr_id,
             k_fact_disb_date, sum(-1 * ok_os), sum(-1 * ok_perc - ok_kommis_adv), sum(-1 * ok_fine),
             -sum(ok_os + ok_perc - ok_kommis_adv + ok_fine)
      from operkred, operlist, username, region, clients un, region_list(:rep_region_id, 1, 3, 1), knaktlist, kontract kont, clients lo
      where ok_from_ol_id = ol_id and
            (region.reg_unicode = region_list.id_list) and
            ol_from_reg_id = reg_unicode and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_type = 4 and
            ok_sh_date < ol_date and
            ol_user_id = un_id and
            un_from_c_id = un.c_id and
            operlist.is_delete = 0 and
            ok_ol_is_active = 1 and
            ok_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            knaktlist.is_delete = 0 and
            kal_new_kf_officer = lo.c_id and
            ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
            and
            ok_from_c_id = 0
      group by ol_id, ol_date, reg_code, reg_name, ol_doc_client_name, ol_memo, ol_datetime, un.c_name_fio, kal_new_kf_name, ok_sh_date, k_id, k_is_problem, k_from_pr_id, lo.c_name_fio, k_fact_disb_date
      having sum(-1 * ok_os) <> 0 or sum(-1 * ok_perc - ok_kommis_adv) <> 0 or sum(-1 * ok_fine) <> 0
      into :apl_ol_id, :apl_date, :apl_region, :apl_client_name, :apl_memo, :apl_datetime, :apl_user_name, :apl_lo_fio,
           :apl_kf_name, :apl_sh_date, :k_id, :k_is_problem, :k_from_pr_id, :apl_fact_disb_date, :apl_os, :apl_perc,
           :apl_fine, :apl_total_sum
  do
  begin
    apl_kol_day = :apl_date - :apl_sh_date;

    k_kontract_category = 'Категория A';
    if (k_is_problem = 1) then
    begin
      if (exists(select *
                 from sb_akt, sb_list
                 where sba_from_k_id = :k_id and
                       sba_date_begin <= :rep_date_end and
                       sba_date_end >= :rep_date_end and
                       sba_status = 2 and
                       sba_from_sbl_id = sbl_id and
                       sbl_type = 1 and
                       sba_is_delete = 0 and
                       sba_from_sbl_id <> 0)) then
        k_kontract_category = 'Категория C';
    end
    if ((k_kontract_category = 'Категория A') and
        (def_curr = '810') and
        (k_from_pr_id >= 18) and
        (k_from_pr_id <= 30)) then
      k_kontract_category = 'Категория B';

    suspend;
  end
end
