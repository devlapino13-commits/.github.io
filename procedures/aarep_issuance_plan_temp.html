CREATE PROCEDURE AAREP_ISSUANCE_PLAN_TEMP(
    REP_DATE DATE)
RETURNS (
    C_ID ID_TYPE,
    SUM_PLAN CURR_TYPE,
    SUM_AHEAD CURR_TYPE,
    SUM_FACT CURR_TYPE,
    IS_IPF SMALLINT,
    REG_ID INTEGER)
AS
declare variable rep_date_begin date;
declare variable rep_date_end date;
declare variable rep_period integer;
declare variable rd_id integer;
begin
  rep_date_begin = rep_date - extract(day from rep_date) + 1;
  rep_date_end = dateadd(month, 1, rep_date_begin) - 1;
  rep_period = 100 * extract(year from rep_date) + extract(month from rep_date);

  c_id = 0;
  sum_plan = 0;
  sum_ahead = 0;
  sum_fact = 0;
  is_ipf = 0;

  for select rd_id, iif(rd_code = 'planindicators.disb_amount_ipf', 1, 0)
      from refdata
      where rd_code in ('planindicators.disb_amount', 'planindicators.disb_amount_ipf')
      into rd_id, is_ipf
  do
  begin
    c_id = 0;
    sum_plan = 0;
    sum_ahead = 0;
    sum_fact = 0;
    for select pd_from_c_id, pd_data_1, pd_from_reg_id
        from hp_tmp_plandata_show(1000000, :rep_date, :rd_id, :rep_period, 1, null)
        where pd_is_office = 2 and
              pd_data_1 > 0
        into c_id, sum_plan, reg_id
    do
    begin
      suspend;
    end
  end

  c_id = 0;
  sum_plan = 0;
  sum_ahead = 0;
  sum_fact = 0;
  is_ipf = 0;

  for select kal_new_kf_officer, -ok_os, iif(pr_code containing '.ipf.', 1, 0), k_last_reg_id
      from operkred, kontract, knaktlist, product
      where ok_ol_is_active = 1 and
            ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_os < 0 and
            ok_type in (4, 9) and
            ok_sh_date > :rep_date_end and
            ok_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            ok_ol_date between kal_date and kal_date_end and
            k_from_pr_id = pr_id
      into c_id, sum_ahead, is_ipf, reg_id
  do
  begin

    suspend;
  end

  c_id = 0;
  sum_plan = 0;
  sum_ahead = 0;
  sum_fact = 0;
  is_ipf = 0;

  for select ks_kc_id, k_amount_approved, iif(pr_code containing '.ipf.', 1, 0), k_last_reg_id
      from kontstate, kontract, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id
      into c_id, sum_fact, is_ipf, reg_id
  do
  begin

    suspend;
  end

end
