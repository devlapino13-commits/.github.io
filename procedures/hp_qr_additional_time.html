CREATE PROCEDURE HP_QR_ADDITIONAL_TIME(
    C_ID D_ID)
RETURNS (
    ADD_TIME D_INTEGER)
AS
begin
  -- возвращает дополнительное время жизни QR в добавок ко времени по умолчанию (для случаев, когда надо увеличить время жизни QR)

  add_time = 0; -- дополнительное время жизни QR в минутах

  -- Если QR требуется для подписания договора АСП, при этом клиент повторный и у него слетела верификация, то срок действия QR равно 120 минут (RFCRU-2203)
  if (f_get_db_id() = 81001
      and f_edoc_status('asp.agreement', c_id, current_date) = 'unsigned' -- договор АСП не подписан, значит QR нужен для его подписания
      and exists(select *
                 from kontstate
                 inner join knclient on kc_from_k_id = ks_from_k_id
                 where kc_from_c_id = :c_id and
                       kc_amount_approved > 0 and
                       ks_ol_is_delete = 0 and
                       ks_state in (1, 2) and
                       current_date between ks_date_begin and ks_date_end) -- клиент повторный
      and not exists(select *
                     from clients
                     where c_id = :c_id and
                           c_verify_status = 2)) then -- клиент не верифицирован
    add_time = 60; -- т.к. время жизни QR по умолчанию 60 минут, то для достижения 120 минут доп. время равно 60

  suspend;
end
