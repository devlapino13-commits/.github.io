CREATE PROCEDURE HP_GET_NEXT_DOV_NO(
    DL_ID INTEGER,
    DC_ID INTEGER,
    DC_DATE DATE)
RETURNS (
    DOV_NO_1 INTEGER,
    DOV_NO_2 INTEGER)
AS
declare variable str_start varchar(50);
declare variable str_end varchar(50);
declare variable dd_year integer;
declare variable dd_begin date;
declare variable dd_begin_next date;
begin
  select dc_mask_no_start, dc_mask_no_end
  from dovcateg
  where dc_id = :dc_id
  into :str_start, :str_end;
  if (str_start is null) then
    str_start = '';
  if (str_end is null) then
    str_end = '';

  dd_year = extract(year from :dc_date);
  dd_begin = cast('01.01.' || :dd_year as date);
  dd_begin_next = cast('01.01.' ||(:dd_year + 1) as date) - 1;

  select max(dl_dov_no_1)
  from dovlist, dovcateg
  where dl_date_begin <= :dc_date and
        dl_date_begin >= :dd_begin and
        dl_from_dc_id = dc_id and
        dc_mask_no_start = :str_start and
        dc_mask_no_end = :str_end and
        dl_id <> :dl_id and
        dl_status > -1
  into :dov_no_1;
  if (dov_no_1 is null) then
    dov_no_1 = 0;
  dov_no_2 = 0;

  if (exists(select *
             from dovlist, dovcateg
             where dl_date_begin > :dc_date and
                   dl_date_begin <= :dd_begin_next and
                   dl_from_dc_id = dc_id and
                   dc_mask_no_start = :str_start and
                   dl_id <> :dl_id and
                   dl_status > -1 and
                   dc_mask_no_end = :str_end)) then
  begin
    select max(dl_dov_no_2)
    from dovlist, dovcateg
    where dl_date_begin <= :dc_date and
          dl_date_begin >= :dd_begin and
          dl_from_dc_id = dc_id and
          dl_dov_no_1 = :dov_no_1 and
          dc_mask_no_start = :str_start and
          dc_mask_no_end = :str_end
    into :dov_no_2;
    if (dov_no_2 is null) then
      dov_no_2 = 0;
    dov_no_2 = dov_no_2 + 1;
  end
  else
    dov_no_1 = dov_no_1 + 1;
  suspend;
end
