CREATE PROCEDURE HP_BKI_GET_UID_INFO(
    CBR_UID D_STR_CODE)
RETURNS (
    OBJECT_TYPE D_STR_100,
    KONTRACT_NO D_STR_20,
    OBJECT_DATE D_DATE,
    CLIENT_FIO D_STR_255,
    PORUCH_FIO D_STR_255,
    OBJECT_ID D_ID,
    C_ID D_ID,
    PORUCH_C_ID D_ID,
    UID_OBJECT_ID D_ID)
AS
declare variable uid_object_kind d_integer;
begin
  if (cbr_uid is null) then
    exit;

  select obj_kind, obj_id
  from hp_find_cbr_uuid(:cbr_uid)
  into uid_object_kind, uid_object_id;

  if (uid_object_kind is null) then
    exit;

  if (uid_object_kind = f_objkind('miniappclient', 'id')) then
  begin
    object_type = 'Заявка';

    select ma_id, ma_date_app, c_id, c_name_fio
    from miniappclient, miniapp, clients
    where mac_id = :uid_object_id
      and ma_id = mac_from_ma_id
      and c_id = mac_from_c_id
    order by ma_number
    rows 1
    into object_id, object_date, c_id, client_fio;
  end
  else if (uid_object_kind in (f_objkind('knclient', 'id'), f_objkind('equifax_poruch', 'id'))) then
  begin
    object_type = 'Договор';

    if (uid_object_kind = f_objkind('equifax_poruch', 'id')) then
    begin
      object_type = 'Договор поручительства';
      select epr_client_kc_id, c_id, c_name_fio
      from equifax_poruch, knclient, clients
      where kc_id = epr_poruch_kc_id
        and c_id = kc_from_c_id
        and epr_id = :uid_object_id
      into uid_object_id, poruch_c_id, poruch_fio;
    end

    select k_id, k_disb_date, k_kontract_no, kc_from_c_id, c_name_fio
    from knclient, kontract, clients
    where k_id = kc_from_k_id
      and c_id = kc_from_c_id
      and kc_id = :uid_object_id
    into object_id, object_date, kontract_no, c_id, client_fio;
  end
  else
    exception e_message 'not found object_kind:' || coalesce(uid_object_kind, '<NULL>');

  suspend;
end
