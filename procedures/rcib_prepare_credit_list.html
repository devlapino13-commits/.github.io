CREATE PROCEDURE RCIB_PREPARE_CREDIT_LIST(
    REP_DATE DATE,
    BASE_RCP_ID ID_TYPE,
    REP_K_ID ID_TYPE,
    REP_C_ID ID_TYPE,
    STEP SMALLINT)
RETURNS (
    CRED_ID VARCHAR(50),
    CRED_NO VARCHAR(50),
    CRED_DATE DATE,
    CRED_SUM DECIMAL(15,2),
    CRED_PERCENT DECIMAL(3,2),
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    CRED_ACCOUNT_NO VARCHAR(50),
    CRED_ACTIVE SMALLINT,
    DELIQ_DAYS INTEGER,
    CRED_ENDDATE DATE,
    TA_CRED_DATE_OVERDUE DATE,
    CDER_ACTIVE_DATE DATE,
    CRED_DATE_MISSEDPAYOUT DATE,
    CRED_AMOUNT_FREQUENCY VARCHAR(2),
    OP_CRED_SUM_PAYOUT DECIMAL(15,2),
    OP_CRED_DATE_PAYOUT DATE,
    OP_CRED_DATE_OVERDUE DATE,
    TA_CRED_SUM_PAYOUT DECIMAL(15,2),
    TA_CRED_DATE_PAYOUT DATE,
    OP_CRED_SUM_DEBT DECIMAL(15,2),
    OP_CRED_SUM_OVERDUE DECIMAL(15,2),
    DELIQ_PERCENTAMOUNT DECIMAL(15,2),
    DELIQ_UNPAIDFINE DECIMAL(15,2),
    OP_CRED_DATE_NEXTPAYOUT DATE,
    TA_CRED_SUM_NEXTPAYOUT DECIMAL(15,2),
    TA_CRED_DATE_NEXTPAYOUT DATE,
    TA_CRED_SUM_DEBT DECIMAL(15,2),
    TA_CRED_SUM_OVERDUE DECIMAL(15,2),
    SH_CALC_FINE DECIMAL(15,2),
    SH_DATE DATE,
    OP_CRED_SUM_PAID DECIMAL(15,2),
    TA_CRED_SUM_PAID DECIMAL(15,2),
    CRED_STATUS SMALLINT,
    CRED_JOIN SMALLINT,
    CRED_CURRENCY VARCHAR(3),
    OP_CRED_SUM_NEXTPAYOUT DECIMAL(15,2),
    HAS_ORIG_ID SMALLINT,
    REP_DATE_BEGIN DATE,
    PERS_TYPE INTEGER,
    IS_PORUCH SMALLINT,
    KC_NUMBER INTEGER,
    IS_OBESP SMALLINT)
AS
declare variable k_kontract_no varchar(50);
declare variable k_amount_approved decimal(15,2);
declare variable unpaidpercentamount_debt decimal(15,2);
declare variable fact_close_date date;
declare variable om_perc decimal(18,7);
declare variable op_cred_sum_paid_tmp decimal(15,2);
declare variable ta_cred_sum_paid_tmp decimal(15,2);
declare variable ta_cred_date_payout_tmp date;
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable max_delinq_d integer;
declare variable k_devide_date date;
declare variable k_orig_id id_type;
declare variable cred_no_prev str_100;
declare variable sh_calc_perc curr_type;
declare variable rcp_date_old date;
declare variable has_changes smallint;
declare variable null_date date;
begin
  rep_k_id = coalesce(rep_k_id, 0);
  rep_c_id = coalesce(rep_c_id, 0);
  cred_amount_frequency = '03'; -- частота выплат - ежемесячно
  null_date = '01.02.1900';
  if (base_rcp_id = 0) then
    base_rcp_id = null;

  if (step = 1) then
  begin
    for select k_kontract_no, k_id, k_amount_approved, c_id, cred_sum, kc_number, cred_percent, cred_join, cred_status,
               fact_close_date, cred_enddate, cred_date, cred_currency, k_devide_date, k_orig_id, cred_no_prev,
               rep_date_begin, type_, is_poruch, is_obesp
        from rcib_prepare_credit_list_temp(:rep_date, :base_rcp_id, :rep_k_id, :rep_c_id, 3)
        order by k_id, kc_number, is_poruch -- Именно такая последоательность важна !!!
        into :k_kontract_no, :k_id, :k_amount_approved, :c_id, :cred_sum, :kc_number, :cred_percent, :cred_join,
             :cred_status, :fact_close_date, :cred_enddate, :cred_date, :cred_currency, :k_devide_date, :k_orig_id,
             :cred_no_prev, :rep_date_begin, :pers_type, :is_poruch, :is_obesp
    do
    begin
      if (is_poruch = 0) then -- рессчитываю только для заемщиков
      begin
        if ((cred_no_prev = '') or (cred_no_prev = '-')) then
          cred_no = k_kontract_no || '(' || cast(kc_number as varchar(10)) || ')';
        else
          cred_no = cred_no_prev;

        if (cred_no_prev = '') then
          has_orig_id = 0;
        else
          has_orig_id = 1;

        cred_account_no = cred_no;
        cred_id = cred_no;

        op_cred_sum_paid = 0;
        ta_cred_sum_paid = 0;

        ta_cred_sum_payout = 0;
        ta_cred_date_payout = null;

        op_cred_sum_payout = 0;
        op_cred_date_payout = null;

        ta_cred_date_overdue = null;

        for select -sum(ok.ok_os), -sum(ok.ok_os + ok.ok_perc + ok.ok_fine + ok.ok_tax), ok.ok_ol_date
            from operkred ok
            where ok.ok_from_k_id = :k_id and
                  ok.ok_type in (4, 9) and
                  ok.ok_ol_is_active = 1 and
                  ok.ok_ol_date <= :rep_date and
                  ok.ok_kommis = 0 -- не комиссия
                  and
                  ok_from_c_id = 0
            group by ok.ok_ol_date
            order by ok.ok_ol_date
            into :op_cred_sum_paid_tmp, :ta_cred_sum_paid_tmp, :ta_cred_date_payout_tmp
        do
        begin
          if (op_cred_sum_paid_tmp <> 0) then
          begin
            op_cred_date_payout = ta_cred_date_payout_tmp; --   дата последнего погашения по кредиту по ОС
            op_cred_sum_payout = op_cred_sum_paid_tmp; --   Сумма последнего платежа по ОС
          end
          op_cred_sum_paid = op_cred_sum_paid + op_cred_sum_paid_tmp; -- Общая сумма, заплаченная заемщиком по кредиту по ОС
          ta_cred_sum_paid = ta_cred_sum_paid + :ta_cred_sum_paid_tmp; -- Общая сумма, заплаченная заемщиком

          ta_cred_date_payout = ta_cred_date_payout_tmp; --   дата последнего погашения по кредиту
          ta_cred_sum_payout = ta_cred_sum_paid_tmp; --   Сумма последнего платежа
        end

        op_cred_sum_payout = coalesce(op_cred_sum_payout, 0);
        op_cred_sum_paid = coalesce(op_cred_sum_paid, 0);
        ta_cred_sum_paid = coalesce(ta_cred_sum_paid, 0);
        ta_cred_sum_payout = coalesce(ta_cred_sum_payout, 0);

        if (op_cred_sum_payout = 0) then
        begin
          op_cred_date_payout = cast('01.02.1900' as date);
        end

--**********************************************************************

        if (cred_status = 1) then --  ЕСЛИ ЭТО АКТИВНЫЙ КРЕДИТ
        begin
          op_cred_date_overdue = null;
          cred_date_missedpayout = null;
          op_cred_sum_debt = 0;
          unpaidpercentamount_debt = 0;
          op_cred_sum_overdue = 0;
          deliq_percentamount = 0;
          deliq_unpaidfine = 0;
          op_cred_date_nextpayout = null;
          ta_cred_sum_nextpayout = 0;
          ta_cred_date_nextpayout = null;
          ta_cred_sum_debt = 0;
          ta_cred_sum_overdue = 0;
          ta_cred_date_overdue = null;
          op_cred_sum_nextpayout = 0;

          for select zad_os, sh_calc_perc, zad_perc, sh_calc_fine, sh_date
              from aarep_make_zad_list(:k_id, :rep_date, 1)
              where sh_date is not null
              into :zad_os, sh_calc_perc, :zad_perc, :sh_calc_fine, :sh_date
          do
          begin
            if (sh_date < rep_date) then
            begin
              if ((cred_date_missedpayout is null) and
                  (zad_os > 0)) then
                cred_date_missedpayout = sh_date; -- Дата возникновения текущей просрочки

              op_cred_sum_overdue = op_cred_sum_overdue + zad_os; -- Сумма текущей просроченной задолженности по ОС
              deliq_percentamount = deliq_percentamount + sh_calc_perc;
              deliq_unpaidfine = deliq_unpaidfine + sh_calc_fine;
            end
            if (sh_date >= rep_date) then
            begin
              unpaidpercentamount_debt = unpaidpercentamount_debt + zad_perc;
              op_cred_sum_debt = op_cred_sum_debt + zad_os; -- Сумма текущей задолженности по ОС
              if ((op_cred_date_nextpayout is null) and
                  (zad_os > 0) and
                  (sh_date > rep_date)) then
              begin
                op_cred_date_nextpayout = sh_date; -- Дата след платежа ОС
                op_cred_sum_nextpayout = zad_os;
              end
              if ((ta_cred_date_nextpayout is null) and
                  (sh_date > rep_date)) then
              begin
                ta_cred_date_nextpayout = sh_date; -- Дата следующего платежа
                ta_cred_sum_nextpayout = zad_os + :zad_perc + sh_calc_fine; -- Сумма следующего платежа
              end
            end
          end
          if (op_cred_sum_overdue > 0) then
            op_cred_date_overdue = cred_date_missedpayout;
          else
            op_cred_date_overdue = cast('01.02.1900' as date);

          ta_cred_sum_debt = op_cred_sum_debt + unpaidpercentamount_debt; -- Сумма текущей задолженности
          ta_cred_sum_overdue = op_cred_sum_overdue + deliq_percentamount + deliq_unpaidfine; --  текущая просрочка (ОС+%+пеня)
          ta_cred_date_overdue = cred_date_missedpayout; -- Дата возникновения текущей просрочки

------------------------------------------------------------------------

          cred_active = 1; -- кредит активный нет просроч. платежей -- <<< СЮДА!
          cder_active_date = :cred_date; -- изначально беру дату выдачи

          if (cred_date_missedpayout is not null) then
            deliq_days = datediff(day, :cred_date_missedpayout, :rep_date); -- -1;
          else
            deliq_days = 0;

          if (deliq_days > 0) then
          begin
            cder_active_date = cred_date_missedpayout; -- дата измнения статуса это дата начала просрочки
            if (deliq_days between 1 and 5) then
              cred_active = 10;
            else
            if (deliq_days between 6 and 29) then
              cred_active = 11;
            else
            if (deliq_days between 30 and 59) then
              cred_active = 12;
            else
            if (deliq_days between 60 and 89) then
              cred_active = 13;
            else
            if (deliq_days between 90 and 119) then
              cred_active = 14;
            else
            if (deliq_days between 120 and 149) then
              cred_active = 15;
            else
            if (deliq_days between 150 and 179) then
              cred_active = 16;
            else
            if (deliq_days between 180 and 209) then
              cred_active = 17;
            else
            if (deliq_days between 210 and 239) then
              cred_active = 18;
            else
            if (deliq_days >= 240) then
              cred_active = 19;
          end
          else -- если нет просрочек на данный момент, но тем не менее статус кредита поменялся,
-- считаю, что дата измнения статуса - это дата последнего платежа
          begin
            cder_active_date = :cred_date;
            max_delinq_d = 0;

            select max(ok_ol_date - ok_sh_date)
            from operkred
            where ok_ol_is_active = 1 and
                  ok_from_k_id = :k_id and
                  ok_ol_date > ok_sh_date and
                  ok_ol_date <= :rep_date and
                  ok_from_c_id = 0 and
                  ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
                  and
                  ok_os + ok_perc + ok_fine <> 0
            into :max_delinq_d;

            max_delinq_d = coalesce(max_delinq_d, 0);
            if (max_delinq_d > 0) then
            begin
              select ok_ol_date
              from operkred
              where ok_ol_is_active = 1 and
                    ok_from_k_id = :k_id and
                    ok_ol_date > ok_sh_date and
                    ok_ol_date <= :rep_date and
                    ok_type <> 7 and
                    ok_os + ok_perc + ok_fine <> 0 and
                    ok_from_c_id = 0
              order by ok_from_k_id desc, ok_ol_date desc
              rows 1
              into :cder_active_date;
            end
          end

          if (cred_active > 10) then
          begin
            if (cred_active = 11) then
              cder_active_date = cder_active_date + 6;
            else
            if (cred_active = 12) then
              cder_active_date = cder_active_date + 30;
            else
            if (cred_active = 13) then
              cder_active_date = cder_active_date + 60;
            else
            if (cred_active = 14) then
              cder_active_date = cder_active_date + 90;
            else
            if (cred_active = 15) then
              cder_active_date = cder_active_date + 120;
            else
            if (cred_active = 16) then
              cder_active_date = cder_active_date + 150;
            else
            if (cred_active = 17) then
              cder_active_date = cder_active_date + 180;
            else
            if (cred_active = 18) then
              cder_active_date = cder_active_date + 210;
            else
            if (cred_active = 19) then
              cder_active_date = cder_active_date + 240;
          end
        end
        else -- ЕСЛИ ЭТО ЗАКРЫТЫЙ КРЕДИТ
        begin
          ta_cred_sum_debt = 0;
          ta_cred_sum_overdue = 0;

          op_cred_sum_debt = 0; -- Сумма текущей задолженности по ОС
          op_cred_date_overdue = null;
          cred_date_missedpayout = null;
          op_cred_sum_overdue = 0; -- Сумма текущей просроченной задолженности по ОС
          deliq_percentamount = 0;
          deliq_unpaidfine = 0;

          op_cred_date_nextpayout = null;
          op_cred_sum_nextpayout = 0;
          ta_cred_sum_nextpayout = 0; -- Сумма следующего платежа
          ta_cred_date_nextpayout = null; -- Дата следующего платежа

          cred_active = 0; -- кредит закрыт!
          cder_active_date = :fact_close_date; -- дата закрытия
        end
------------------------------------------------------------------------
        om_perc = cred_sum * 100 / k_amount_approved; -- доля уч-ка в процентном соотношении к доли группы

        ta_cred_sum_payout = ta_cred_sum_payout * om_perc / 100;
        op_cred_sum_payout = op_cred_sum_payout * om_perc / 100;

        ta_cred_sum_debt = ta_cred_sum_debt * om_perc / 100;
        op_cred_sum_debt = op_cred_sum_debt * om_perc / 100;

        ta_cred_sum_paid = ta_cred_sum_paid * om_perc / 100;
        op_cred_sum_paid = op_cred_sum_paid * om_perc / 100;

        ta_cred_sum_overdue = ta_cred_sum_overdue * om_perc / 100;
        op_cred_sum_overdue = op_cred_sum_overdue * om_perc / 100;

        ta_cred_sum_nextpayout = ta_cred_sum_nextpayout * om_perc / 100;
        op_cred_sum_nextpayout = op_cred_sum_nextpayout * om_perc / 100;

        deliq_percentamount = deliq_percentamount * om_perc / 100;
        deliq_unpaidfine = deliq_unpaidfine * om_perc / 100;

        if (cred_status = 2) then
        begin
          op_cred_sum_paid = round(op_cred_sum_paid);
        end

        if (cred_active in (0, 1)) then -- даже если  нет текущей просрочки, нужно определить дату последней просрочки, требование КИБ
        begin
          select max(ok_sh_date)
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :k_id and
                ok_type = 4 and
                ok_ol_date > ok_sh_date and
                ok_ol_date <= :rep_date and
                ok_from_c_id = 0 and
                ok_type <> 7 and
                ok_os + ok_perc <> 0 --  !!! Иначе ты можешь поймать погашения просроченной пени (отложенные пени, или по новому руководству, пеня берется после погашения ОС и процентов).
          into :cred_date_missedpayout;
        end

        if (cred_date_missedpayout is null) then
        begin
          select max(rck_cred_date_missedpayout)
          from rcib_kontract rck
          inner join rcib_pocket rcp on rcp.rcp_id = rck.rck_from_cp_id and
                rck.rck_from_c_id = :c_id and
                rck.rck_from_k_id = :k_id and
                rcp.rcp_is_kredit = 1
          into :cred_date_missedpayout;
        end
        op_cred_date_overdue = coalesce(op_cred_date_overdue, cast('01.02.1900' as date));
        ta_cred_date_overdue = coalesce(ta_cred_date_overdue, cast('01.02.1900' as date));

        cred_date_missedpayout = coalesce(cred_date_missedpayout, cast('01.02.1900' as date));

        op_cred_date_nextpayout = coalesce(op_cred_date_nextpayout, cast('01.02.1900' as date));
        ta_cred_date_nextpayout = coalesce(ta_cred_date_nextpayout, cast('01.02.1900' as date));

        if (ta_cred_sum_nextpayout = 0) then
          ta_cred_date_nextpayout = cast('01.02.1900' as date);
        if (ta_cred_date_nextpayout > cred_enddate) then
          ta_cred_date_nextpayout = cred_enddate;

        ta_cred_sum_payout = coalesce(ta_cred_sum_payout, 0);

        if (ta_cred_sum_payout = 0) then
          ta_cred_sum_paid = 0;
        if (cred_active = 0) then
          ta_cred_sum_debt = 0;

        if ((op_cred_date_overdue = ta_cred_date_overdue) and
            (ta_cred_sum_overdue < op_cred_sum_overdue)) then
          ta_cred_sum_overdue = op_cred_sum_overdue;

        if ((op_cred_date_overdue <> cast('01.02.1900' as date)) and
            op_cred_sum_overdue = 0) then
          op_cred_sum_overdue = ta_cred_sum_overdue;

        if (ta_cred_date_overdue = '01.02.1900') then
          ta_cred_sum_overdue = 0;
      end
      else
      begin
        cred_date_missedpayout = '01.02.1900';
        op_cred_date_payout = '01.02.1900';
        ta_cred_date_payout = '01.02.1900';
        op_cred_sum_payout = 0;
        ta_cred_sum_payout = 0;
        op_cred_sum_paid = 0;
        ta_cred_sum_paid = 0;
      end

      suspend;
    end
  end
  else
  if (step = 2) then
  begin
    has_changes = 1;

    for select cred_id, cred_no, cred_date, cred_sum, cred_percent, k_id, c_id, cred_account_no, cred_active,
               deliq_days, cred_enddate, ta_cred_date_overdue, cder_active_date, cred_date_missedpayout,
               cred_amount_frequency, op_cred_sum_payout, op_cred_date_payout, op_cred_date_overdue, ta_cred_sum_payout,
               ta_cred_date_payout, op_cred_sum_debt, op_cred_sum_overdue, deliq_percentamount, deliq_unpaidfine,
               op_cred_date_nextpayout, ta_cred_sum_nextpayout, ta_cred_date_nextpayout, ta_cred_sum_debt,
               ta_cred_sum_overdue, sh_calc_fine, sh_date, op_cred_sum_paid, ta_cred_sum_paid, cred_status, cred_join,
               cred_currency, op_cred_sum_nextpayout, has_orig_id, rep_date_begin, kc_number, is_poruch, pers_type,
               is_obesp
        from rcib_prepare_credit_list(:rep_date, :base_rcp_id, :rep_k_id, :rep_c_id, 1)
        order by k_id, kc_number, is_poruch
        into :cred_id, :cred_no, :cred_date, :cred_sum, :cred_percent, :k_id, :c_id, :cred_account_no, :cred_active,
             :deliq_days, :cred_enddate, :ta_cred_date_overdue, :cder_active_date, :cred_date_missedpayout,
             :cred_amount_frequency, :op_cred_sum_payout, :op_cred_date_payout, :op_cred_date_overdue,

        has_changes = 0;

-- Проверяю, есть ли измнения по кредиту. Если измнения есть создаю новую запись

        if ((base_rcp_id is null) and
            (rep_k_id = 0)) then
        begin
          select rck.rck_cred_update,
                 iif(rck_cred_active = :cred_active, 0, 1) + iif(coalesce(rck_cred_sum, -777) = coalesce(:cred_sum, -777), 0, 1) + iif(round(coalesce(rck_cred_percent, -777), 2) = round(coalesce(:cred_percent, -777), 2), 0, 1) + iif(coalesce(rck_cred_date, :null_date) = coalesce(:cred_date, :null_date), 0, 1) + iif(coalesce(rck_cred_enddate, :null_date) = coalesce(:cred_enddate, :null_date), 0, 1) + iif(coalesce(rck_cred_date_missedpayout, :null_date) = coalesce(:cred_date_missedpayout, :null_date), 0, 1) + iif(coalesce(rck_op_cred_sum_payout, -777) = coalesce(:op_cred_sum_payout, -777), 0, 1) + iif(coalesce(rck_op_cred_date_payout, :null_date) = coalesce(:op_cred_date_payout, :null_date), 0, 1) + iif(coalesce(rck_op_cred_sum_debt, -777) = coalesce(:op_cred_sum_debt, -777), 0, 1) + iif(coalesce(rck_op_cred_sum_paid, -777) = coalesce(:op_cred_sum_paid, -777), 0, 1) + iif(coalesce(rck_op_cred_sum_overdue, -777) =           from rcib_kontract rck
          where rck.rck_cred_no = :cred_no --'0103-053496-01(3)'
          order by rck.rck_from_cp_id desc
          rows 1
          into rcp_date_old, has_changes;

          rcp_date_old = coalesce(rcp_date_old, cast('01.01.2014' as date));
          if (datediff(day, rcp_date_old, rep_date) >= 10) then
            has_changes = 1; -- даже если изменений нет, прошло 10 дней и более, все равно создаю запись
        end
        else
          has_changes = 1;
      end

      if (has_changes >= 1) then
      begin
        if (ta_cred_date_payout is null) then
          ta_cred_date_payout = null_date;
        if (op_cred_date_payout is null) then
          op_cred_date_payout = null_date;

        suspend;
      end
    end
