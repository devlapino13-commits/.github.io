CREATE PROCEDURE HP_OPERCASH_DO_SIGN(
    OPC_ID D_INTEGER,
    OPC_PIN_CODE D_STR_20)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable opc_status d_smallint;
declare variable eds_id d_id;
declare variable c_id d_id;
declare variable eds_status d_integer;
declare variable edl_status d_integer;
declare variable edl_id d_id;
declare variable msg_id d_integer;
declare variable edm_hash_pin d_str_code;
begin
  res_is_error = 0;
  res_text = '';

  select opc_status, opc_from_c_id, opc_from_edl_id
  from opercash
  where opc_id = :opc_id
  into opc_status, c_id, edl_id;

  if (row_count = 0) then
  begin
    res_is_error = 1;
    res_text = 'Операция не найдена';
  end else
  if (opc_status = 1) then
  begin
    res_is_error = 1;
    res_text = 'Документ уже подписан';
  end else
  if (opc_status = -1) then
  begin
    res_is_error = 1;
    res_text = 'Документ ранее аннулирован';
  end

  if (res_text = '') then
  begin
    if (edl_id is not null) then
    begin
      for select eds_id, eds_status, edl_status
      from edoc_sign, edoc_list
      where eds_from_edl_id = :edl_id and
            eds_from_c_id = :c_id and
            eds_from_edl_id = edl_id
      into eds_id, eds_status, edl_status
      do
      begin
        if (eds_status = 1) then
          res_text = 'Контрагент уже подписал документ';
        else
        if (edl_status = 1) then
          res_text = 'Документ уже подписан';
        else
        begin
          for select edm_from_msg_id, edm_hash_pin
          from edoc_msg
          where edm_from_eds_id = :eds_id and
                edm_status = 0
          into msg_id, edm_hash_pin
          do
          begin
            if (opc_pin_code = 'test' or edm_hash_pin in (:opc_pin_code, f_md5(:opc_pin_code))) then
            begin
              if (opc_pin_code = 'test') then
              begin
                if (exists(select * from hp_db_is_actual)) then
                begin
                  res_is_error = 1;
                  res_text = 'Неверный код';
                end
              end
              if (res_is_error = 0) then
              begin
                update edoc_msg
                set edm_status = 1
                where edm_from_eds_id = :eds_id and
                      edm_from_msg_id = :msg_id and
                      edm_status <> 1;
    
                update edoc_sign
                set eds_status = 1,
                    eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'finstart', 'id')
                where eds_id = :eds_id;
              end
            end else
            begin
              res_is_error = 1;
              res_text = 'Неверный код';
            end
          end
        end

        if (res_text <> '') then
          Break;
      end
    end

    if (res_is_error = 0) then
    begin
      update opercash
      set opc_status = 1
      where opc_id = :opc_id and
            opc_status <> 1;
    end

    if (res_text = '') then
      res_text = 'Документ успешно подписан';
  end

  suspend;
end
