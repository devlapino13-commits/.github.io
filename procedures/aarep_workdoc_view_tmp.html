CREATE PROCEDURE AAREP_WORKDOC_VIEW_TMP(
    WDL_ID INTEGER)
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_OPEN_KOLICH INTEGER,
    C_LAST_OPENED TIMESTAMP,
    C_JL_NAME VARCHAR(100),
    C_REG_NAME VARCHAR(100),
    C_FIRST_OPENED TIMESTAMP)
AS
declare variable wdl_is_open_access integer;
begin
  c_open_kolich = 0;
  c_last_opened = null;
  c_first_opened = null;
  wdl_is_open_access = 0;
  select wdl_is_open_access
  from workdoclist
  where wdl_id = :wdl_id
  into :wdl_is_open_access;
  if (wdl_is_open_access is null) then
    wdl_is_open_access = 0;
  if (wdl_is_open_access = 0) then
  begin
    for select distinct wdc_from_c_id, c_name_fio
        from workdoccl, clients
        where wdc_from_wdl_id = :wdl_id and
              wdc_from_c_id = c_id
        union
        select wdvip_from_c_id, c_name_fio
        from workdocvipusers, clients
        where wdvip_from_c_id = c_id
        into :c_id, :c_name_fio
    do
    begin
      c_jl_name = 'Нет действ.приказа';
      c_reg_name = 'Нет действ.приказа';

      select jl_name, reg_code || ' ' || reg_name
      from hr_pstatus, joblist, region
      where ps_from_c_id = :c_id and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            ps_from_reg_id = reg_unicode rows 1
      into :c_jl_name, :c_reg_name;

      suspend;
    end
    for select distinct c_id, c_name_fio, jl_name, reg_code || ' ' || reg_name
        from workdocjl, clients, hr_pstatus, joblist, region
        where wdu_from_wdl_id = :wdl_id and
              wdu_from_jl_id = ps_from_jl_id and
              ps_from_c_id = c_id and
              ps_date_begin <= current_date and
              ps_date_end >= current_date and
              ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              ps_from_reg_id = reg_unicode
        into :c_id, :c_name_fio, :c_jl_name, :c_reg_name
    do
    begin
      suspend;
    end
  end
  else
  begin
    for select un_from_c_id, c_name_fio
        from username, clients
        where un_from_c_id = c_id
        into :c_id, :c_name_fio
    do
    begin
      if (exists(select *
                 from hp_client_un_is_active(:c_id)
                 where un_is_active = 1)) then
      begin
        c_jl_name = 'Нет действ.приказа';
        c_reg_name = 'Нет действ.приказа';

        select jl_name, reg_code || ' ' || reg_name
        from hr_pstatus, joblist, region
        where ps_from_c_id = :c_id and
              ps_date_begin <= current_date and
              ps_date_end >= current_date and
              ps_is_working = 1 and
              ps_from_jl_id = jl_id and
              ps_from_reg_id = reg_unicode rows 1
        into :c_jl_name, :c_reg_name;

        suspend;
      end
    end
  end

  for select c_id, c_name_fio, wds_kolich, wds_datetime, wds_first_datetime
      from workdocstat, clients, username
      where wds_from_wdl_id = :wdl_id and
            wds_from_un_id = un_id and
            un_from_c_id = c_id
      into :c_id, :c_name_fio, :c_open_kolich, c_last_opened, :c_first_opened
  do
  begin
    c_jl_name = 'Нет действ.приказа';
    c_reg_name = 'Нет действ.приказа';

    select jl_name, reg_code || ' ' || reg_name
    from hr_pstatus, joblist, region
    where ps_from_c_id = :c_id and
          ps_date_begin <= current_date and
          ps_date_end >= current_date and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id and
          ps_from_reg_id = reg_unicode rows 1
    into :c_jl_name, :c_reg_name;

    suspend;
  end

end
