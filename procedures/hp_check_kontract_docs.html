CREATE PROCEDURE HP_CHECK_KONTRACT_DOCS(
    K_ID ID_TYPE)
RETURNS (
    RES_ID D_INTEGER,
    RES_SET_ID D_INTEGER,
    RES_DOC_SET_NAME D_STR_100,
    RES_DOCPROP_NAME D_STR_255,
    RES_VALID D_SMALLINT,
    CLIENT_KIND SMALLINT,
    RES_OBJECT_NAME D_STR_255)
AS
declare variable pr_id d_integer;
declare variable rep_date d_date;
declare variable z_type d_smallint;
declare variable z_from_c_id id_type;
declare variable c_id id_type;
declare variable k_kontract_no d_str_code;
begin
  --  client_kind: 0 - Клиент; 1 - Клиент и он сам является залогодателем; 2 - Залогодатель, он отличается от клиента; 3 - Поручитель
  res_id = 0;
  for
  select k_from_pr_id,k_disb_date, k_kontract_no from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  into pr_id, rep_date, k_kontract_no
  do
  begin
    -- проверка клиентов
    for
    select kc_from_c_id from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    into c_id
    do
    begin
      client_kind = 0;
      if (exists(select * from zalog
      where z_from_c_id = :c_id
      and z_from_k_id = :k_id
      and :rep_date between coalesce(z_date_begin,'01.01.1990')
      and coalesce(z_date_end,'31.12.9999'))) then client_kind = 1;

      if ((f_docprloan_valid(:c_id,:pr_id,:client_kind,:rep_date) = -1)) then
      begin
        for select res_set_id, res_doc_set_name, res_docprop_name, res_valid
        from hp_docprloan_valid(:c_id, :pr_id, :client_kind, :rep_date, 1)
--        where res_valid = 0 --возвращает только по недостающимся документам
        into :res_set_id, :res_doc_set_name, :res_docprop_name, :res_valid
        do
        begin
          res_object_name = '';
          select :k_kontract_no || ', '||trim(iif(:client_kind = 0, 'клиент:','клиент/залогодатель:'))||c_name_fio
          from clients
          where c_id = :c_id
          into res_object_name;

          res_id = res_id + 1;
          suspend;
        end
      end
    end 
    -- проверка поручителей и залогодателей
    for
    select distinct z_type, z_from_c_id from zalog
    where z_from_k_id = :k_id
    and :rep_date between coalesce(z_date_begin,'01.01.1990') and coalesce(z_date_end,'31.12.9999')
    and z_from_c_id is not null
    into z_type, z_from_c_id
    do
    begin
      if (exists(select * from knclient
      where kc_from_k_id = :k_id
      and kc_from_c_id = :z_from_c_id
      and kc_amount_approved > 0
      )) then continue;

      if (z_type = 1)
      then client_kind = 2;
      else client_kind = 3;

      if (f_docprloan_valid(:z_from_c_id,:pr_id,:client_kind,:rep_date) = -1) then
      begin
        for select res_set_id, res_doc_set_name, res_docprop_name, res_valid
        from hp_docprloan_valid(:z_from_c_id, :pr_id, :client_kind, :rep_date, 1)
--        where res_valid = 0 -- возвращает только по недостающимся документам
        into :res_set_id, :res_doc_set_name, :res_docprop_name, :res_valid
        do
        begin
          res_object_name = '';
          select :k_kontract_no || ', '||trim(iif(:client_kind = 2, 'залогодатель:','поручитель:'))||c_name_fio
          from clients
          where c_id = :c_id
          into res_object_name;

          res_id = res_id + 1;
          suspend;
        end
      end
    end 
  end
end
