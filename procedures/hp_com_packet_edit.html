CREATE PROCEDURE HP_COM_PACKET_EDIT(
    NEW_CMP_ID D_INTEGER,
    NEW_CMP_STR_NO D_STR_CODE,
    NEW_CMP_DATE D_DATE,
    NEW_CMP_FROM_REG_ID D_INTEGER,
    NEW_CMP_FROM_CMK_ID D_INTEGER,
    NEW_CMP_FROM_CMD_ID D_INTEGER,
    NEW_CMP_STATUS D_SMALLINT,
    NEW_CMP_IS_IPF D_SMALLINT,
    NEW_CMP_MEMO D_STR_LONG)
RETURNS (
    CMP_ID D_INTEGER,
    CMP_STR_NO D_STR_CODE,
    CMP_DATE D_DATE,
    CMP_FROM_REG_ID D_INTEGER,
    CMP_FROM_CMK_ID D_INTEGER,
    CMP_FROM_CMD_ID D_INTEGER,
    CMP_STATUS D_SMALLINT,
    CMP_IS_IPF D_SMALLINT,
    CMP_MEMO D_STR_LONG)
AS
declare variable last_cmp_id d_integer;
begin
  if (coalesce(new_cmp_from_cmk_id, 0) = 0) then
    exception e_message 'Тип кредитного комитета не выбран';

  if (new_cmp_id is null) then
    new_cmp_id = 0;

  if (exists(select *
             from com_packet
             where cmp_str_no = :cmp_str_no and
                   cmp_id <> :new_cmp_id)) then
    exception e_message 'Номер дублируется, введите другой номер';

  if (coalesce(new_cmp_from_cmd_id, 0) <> 0) then
  begin
    if (not exists(select *
                   from com_kind_dec
                   where cmkd_from_cmk_id = :new_cmp_from_cmk_id and
                         cmkd_from_cmd_id = :new_cmp_from_cmd_id)) then
      exception e_message 'Выбранный вид комитета не может принимать такое решение';
  end

  select cmk_is_ipf
  from com_kind
  where cmk_id = :new_cmp_from_cmk_id
  into new_cmp_is_ipf;

  if (new_cmp_id = 0) then
  begin
    select cmp_id
    from com_packet
    where cmp_from_cmk_id = :new_cmp_from_cmk_id
    order by cmp_id desc
    rows 1
    into last_cmp_id;

    insert into com_packet (cmp_str_no, cmp_date, cmp_from_reg_id, cmp_from_cmk_id, cmp_from_cmd_id, cmp_status,
                            cmp_is_ipf, cmp_memo)
    values (:new_cmp_str_no, :new_cmp_date, :new_cmp_from_reg_id, :new_cmp_from_cmk_id, :new_cmp_from_cmd_id,
            :new_cmp_status, :new_cmp_is_ipf, :new_cmp_memo)
    returning cmp_id, cmp_str_no, cmp_date, cmp_from_reg_id, cmp_from_cmk_id, cmp_from_cmd_id, cmp_status, cmp_is_ipf, cmp_memo
    into cmp_id, cmp_str_no, cmp_date, cmp_from_reg_id, cmp_from_cmk_id, cmp_from_cmd_id, cmp_status, cmp_is_ipf,
         cmp_memo;

    if (last_cmp_id is not null) then
    begin
      insert into com_staff (cms_from_cmp_id, cms_from_c_id, cms_status)
      select :cmp_id, cms_from_c_id, cms_status
      from com_staff
      where cms_from_cmp_id = :last_cmp_id;
    end
  end
  else
  begin
    update com_packet
    set cmp_str_no = :new_cmp_str_no,
        cmp_date = :new_cmp_date,
        cmp_from_reg_id = :new_cmp_from_reg_id,
        cmp_from_cmk_id = :new_cmp_from_cmk_id,
        cmp_from_cmd_id = :new_cmp_from_cmd_id,
        cmp_status = :new_cmp_status,
        cmp_is_ipf = :new_cmp_is_ipf,
        cmp_memo = :new_cmp_memo
    where (cmp_id = :new_cmp_id)
    returning cmp_id, cmp_str_no, cmp_date, cmp_from_reg_id, cmp_from_cmk_id, cmp_from_cmd_id, cmp_status, cmp_is_ipf, cmp_memo
    into cmp_id, cmp_str_no, cmp_date, cmp_from_reg_id, cmp_from_cmk_id, cmp_from_cmd_id, cmp_status, cmp_is_ipf,
         cmp_memo;
  end
  suspend;
end
