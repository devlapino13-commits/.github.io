CREATE PROCEDURE HP_BKI_CORR_COPY_EVENT(
    BEV_ID D_ID)
AS
declare variable bfv_id d_id;
  declare variable bkk_id d_id;
  declare variable event_id d_integer;
  declare variable object_id d_id;
  declare variable c_id d_id;
  declare variable new_bev_id d_id;
  declare variable var_value_old d_str_8100;
  declare variable var_value_new d_str_8100;
begin
  select bkp_from_bkk_id, bkl_object_id, bkc_from_c_id, bev_event_id
  from bki_events, bki_clients, bki_loans, bki_packet
  where bev_id = :bev_id
    and bkc_id = bev_from_bkc_id
    and bkl_id = bkc_from_bkl_id
    and bkp_id = bkl_from_bkp_id
    and bkl_object_kind = 9
  into bkk_id, object_id, c_id, event_id;

  if (object_id is null) then
    exit;

  select new_bev_id
  from hp_bki_corr_add_loan (:object_id, :c_id, :event_id, current_date, :bkk_id)
  into new_bev_id;

  for select bfv_id, var_value_old, var_value_new
  from hp_bki_parameters_list(:bev_id, 1)
  into :bfv_id, :var_value_old, :var_value_new
  do
  begin
    execute procedure hp_bki_corr_set_parametr(:new_bev_id, :bfv_id, 1);
    execute procedure hp_bki_corr_set_value(:bfv_id, :new_bev_id, :var_value_old, :var_value_new);
  end
end
