CREATE PROCEDURE OKT_WEBSITE_LIST(
    INPUT_ACL_ID ID_TYPE)
RETURNS (
    ACL_ID ID_TYPE,
    ACL_CELL_NO VARCHAR(20),
    ACL_PROJECT STR_100,
    ACL_TIMEZONE INTEGER,
    ACL_LANGUAGE STR_100,
    ERR_TEXT STR_255,
    ACL_URL BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable rd_id integer;
declare variable t_id integer;
declare variable object_id id_type;
declare variable res_has_code integer;
declare variable str_values str_255;
declare variable next_call_value timestamp;
begin
  /*
  OKTELL
  Список задач для обработки OKTELL по проекту "Обработка заявок с сайта для Мигрантов"
  */
  input_acl_id = coalesce(input_acl_id, 0);

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'OKTELL.Projects.Website'
  into :rd_id;

  t_id = f_objkind('ExtSysMain', 'id');

  acl_project = 'OKTELL.Projects.WebSite';

  acl_timezone = 3;

  for select ao.aco_id, ao.aco_object_id
      from actionorder ao
      where ao.aco_from_rd_id = :rd_id and
            ((ao.aco_id = :input_acl_id) or
            (:input_acl_id = 0))
      into :acl_id, :object_id
  do
  begin
    err_text = '';

    select res_str_values
    from hp_objectprops_anketa_get(:t_id, :object_id, -1, 'AnketaQuestions.OKTELL.WebSite.Phone_No', current_date, 'Results.StrValue')
    into :acl_cell_no;

    acl_cell_no = left(coalesce(acl_cell_no, ''), 20);

    acl_language = 'Language.RU';

    select res_has_code
    from hp_objectprops_anketa_get(:t_id, :object_id, -1, 'AnketaQuestions.OKTELL.WebSite.Cl_Language', current_date, 'Language.KG')
    into :res_has_code;

    res_has_code = coalesce(res_has_code, 0);
    if (res_has_code = 0) then
    begin
      select res_has_code
      from hp_objectprops_anketa_get(:t_id, :object_id, -1, 'AnketaQuestions.OKTELL.WebSite.Cl_Language', current_date, 'Language.TJ')
      into :res_has_code;

      res_has_code = coalesce(res_has_code, 0);

      if (res_has_code = 0) then
      begin
        select res_has_code
        from hp_objectprops_anketa_get(:t_id, :object_id, -1, 'AnketaQuestions.OKTELL.WebSite.Cl_Language', current_date, 'Language.UZ')
        into :res_has_code;

        res_has_code = coalesce(res_has_code, 0);

        if (res_has_code = 1) then
        begin
          acl_language = 'Language.UZ';
        end
      end
      else
      begin
        acl_language = 'Language.TJ';
      end
    end
    else
    begin
      acl_language = 'Language.KG';
    end

    str_values = null;
    next_call_value = null;

    select res_str_values
    from hp_objectprops_anketa_get(:t_id, :object_id, -1, 'AnketaQuestions.OKTELL.WebSite.Cl_Next_Call_Value', current_date, 'Results.StrValue')
    into :str_values;

    str_values = trim(coalesce(str_values, ''));
    if (str_values <> '') then
    begin
      next_call_value = cast(str_values as timestamp);

      when sqlcode -413 do
-- convert_error
      begin
        err_text = 'Неправильный формат даты/времени след. звонка';
      end
    end

    next_call_value = coalesce(next_call_value, '01.01.1900');

    if (acl_cell_no <> '') then
    begin
      if (current_timestamp >= next_call_value) then
      begin
        acl_url = 'http://oktell.molbulak.ru/CustomAnketaSelection.aspx?anketa_category=OKTELL.Projects.Website&abonent_number=' || acl_cell_no || '&acl_id=' || acl_id || '&un_id=';
        suspend;
      end
    end
  end
end
