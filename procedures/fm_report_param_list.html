CREATE PROCEDURE FM_REPORT_PARAM_LIST(
    UN_ID D_ID,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    REP_ID D_STR_255,
    REP_OBJ_KIND D_STR_CODE = '',
    REP_OBJ_ID D_ID = 0,
    FIND_TEXT D_STR_100 = '',
    REG_ID D_INTEGER = null)
RETURNS (
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    TEXT_1 D_STR_255,
    TEXT_2 D_STR_255,
    TEXT_3 D_STR_255,
    TEXT_4 D_STR_255)
AS
declare variable dd_tmp d_date;
declare variable k_id d_id;
declare variable ma_id d_integer;
declare variable delinq_days d_integer;
declare variable limit d_currency = 0;
declare variable res_sender_fio d_str_100;
declare variable res_subject d_str_255;
declare variable res_datetime d_datetime;
declare variable res_kind d_str_code;
declare variable time_str d_str_code;
declare variable day_between d_integer;
declare variable ma_plan_disb_time d_datetime;
declare variable is_ipf d_integer;
declare variable db_id d_integer;
declare variable pr_id_old d_integer;
declare variable pr_id d_integer;
declare variable lo_id d_id;
declare variable mo_reg_id d_integer;
declare variable lang d_str_lang;
declare variable rn_name varchar(30);
declare variable wait_text d_str_code;
declare variable lo_reg_id d_integer;
declare variable counter d_integer;
declare variable c_name_fio varchar(100);
declare variable temp_text_1 d_str_long;
declare variable temp_text_length d_integer;
begin
  -- Процедура возвращает список (до 4 колонок в МП) для выбора пользователем
  -- rep_obj_kind, rep_object_id приходят во входящих параметрах только когда
  -- в fm_report_param_view указано next_action = 'list'.
  -- Т.е. окно параметров вызывает следующее окно параметров, а не html.
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);
  date_begin = coalesce(date_begin, current_date);
  date_end = coalesce(date_end, current_date);
  if (coalesce(rep_id, '') = '') then
    exception e_default 'Input param rep_id is not defined';
  find_text = coalesce(find_text, '');
  -- Подмена пользователя
  if (not(rep_id starting 'mailmessages_')) then
  begin
    select un_id
    from hp_test_un_id(:un_id)
    into un_id;
  end
  select un_from_c_id, un_curr_reg_id
  from username
  where un_id = :un_id
  into lo_id, lo_reg_id;

  -- Если параметр reg_id не был передан, пытаемся взять его из кадровых приказов сотрудника
  if (reg_id is null) then
  begin
      select ps_from_reg_id
      from hr_pstatus
      where ps_from_c_id = :lo_id and
            ps_date_begin <= current_date and
            ps_date_end >= current_date and
            ps_status = 1 and
            ps_is_working = 1
      into :reg_id;
  end

  mo_reg_id = 0; -- Офис МО, используется в одном отчете, там же и вычисляется
  if (rep_id = 'loan_in_progress_status') then
  begin
    -- Мои заявки в процессе
    -- title_1 = 'Группа'
    -- title_2 = 'Статус'
    -- title_3 = 'Дата'
    -- title_4 = 'Время'
    text_4 = '';
    pr_id_old = 0;
    for select k_id, ma_id, kf_name, loan_status, disb_date, ma_plan_disb_time, pr_id
        from fm_loans_in_process(:un_id, :reg_id)
        order by kf_name
        into k_id, ma_id, text_1, text_2, dd_tmp, ma_plan_disb_time, pr_id
    do
    begin
      if (k_id <> 0) then
      begin
        obj_kind = 'Kontract';
        obj_id = k_id;
      end
      else
      begin
        obj_kind = 'Miniapp';
        obj_id = ma_id;
      end
      wait_text = '';
      select res_text
      from hp_expressapp_status_info(:obj_kind, :obj_id, :un_id)
      into wait_text;
      if (wait_text <> '') then
        text_1 = text_1 || '<br>' || wait_text;
      if (pr_id <> pr_id_old and
          db_id in (39801, 41701)) then
      begin
        pr_id_old = pr_id;
        select iif(pr_code containing '.IPF.', 1, 0)
        from product pr
        where pr.pr_id = :pr_id
        into is_ipf;
        if (is_ipf = 1) then
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'ИПФ'));
        else
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'Займ'));
      end
      text_3 = iif(dd_tmp is not null, f_datetostr(dd_tmp), f_lang(lang, 'Не указана'));
      text_4 = iif(ma_plan_disb_time is not null, right('0' || extract(hour from ma_plan_disb_time), 2) || ':' || right('0' || extract(minute from ma_plan_disb_time), 2), '-');
      if (find_text = '' or f_format('%1~%2~%3~%4', text_1, text_2, text_3, text_4) containing find_text) then
        suspend;
    end
    exit;
  end
  if (rep_id = 'bank_card_disb_status') then
  begin
    obj_kind = 'new';
    text_1 = 'К выдаче';
    obj_id = 1;
    suspend;
    exit;
  end
  if (rep_id = 'bank_card_disb_list') then
  begin
    mo_reg_id = (select ps.ps_from_reg_id
                 from hr_pstatus ps
                 where ps.ps_from_c_id = :lo_id and
                       ps.ps_is_working = 1 and
                       current_date between ps.ps_date_begin and ps.ps_date_end and
                       ps.ps_status = 1);
    obj_kind = :rep_obj_kind;
    for select hk.k_id, hk.k_kontract_no, hk.c_name_fio
        from hp_kontracts_for_card_disb(:mo_reg_id, :date_end, :rep_obj_kind, :un_id) hk
        into obj_id, text_2, text_1
    do
    begin
      if (find_text = '' or f_format('%1~%2', text_1, text_2) containing find_text) then
        suspend;
    end
    exit;
  end

  if (rep_id = 'extminiapp_list') then --КЦ
  begin
    -- Онлайн заявки в процессе
    -- title_1 = 'ФИО'
    -- title_2 = 'Мои'
    for select obj_kind, obj_id, res_text_1, res_text_2
        from fm_extminiapp_list(:un_id, 1, :date_begin, :date_end, :reg_id)
        order by res_pos desc, obj_id
        into obj_kind, obj_id, text_1, text_2
    do
    begin
      if (find_text = '' or f_format('%1~%2', text_1, text_2) containing find_text) then
        suspend;
    end
    exit;
  end
  if (rep_id = 'extminiapp_list_kmp') then --КМП
  begin
    -- Онлайн заявки в процессе
    -- title_1 = 'ФИО'
    -- title_2 = 'Мои'
    for select obj_kind, obj_id, res_text_1, res_text_2
        from fm_extminiapp_list(:un_id, 2, :date_begin, :date_end, :reg_id)
        order by res_pos desc, obj_id
        into obj_kind, obj_id, text_1, text_2
    do
    begin
      if (find_text = '' or f_format('%1~%2', text_1, text_2) containing find_text) then
        suspend;
    end
    exit;
  end
  if (rep_id = 'clientlink_list') then
  begin
    -- Список друзей клиента
    -- title_1 = 'ФИО'
    -- title_2 = 'Смайлик'
    for select obj_kind, obj_id, res_text_1, res_text_2
        from fm_clientlink_list(:un_id, :find_text)
        into obj_kind, obj_id, text_1, text_2
    do
    begin
      suspend;
    end
    exit;
  end
    if (rep_id = 'clientlink_list_no_lo') then
  begin
    -- Список друзей клиента
    -- title_1 = 'ФИО'
    -- title_2 = 'Смайлик'
    for select obj_kind, obj_id, res_text_1, res_text_2
        from fm_clientlink_list_no_lo(:un_id, :find_text)
        into obj_kind, obj_id, text_1, text_2
    do
    begin
      suspend;
    end
    exit;
  end

  if (rep_id = 'docrecog_request_list') then
  begin
    -- Запросы на фотографирование
    -- title_1 = 'ФИО'
    -- title_2 = 'Документ'
    -- title_3 = 'Дата'
    -- title_4 = 'Статус'

    obj_kind = 'docrecogrequest';
    for select drr_id, c_name_fio, obj_name, drr_create_date, drr_status_str
    from fm_docrecogrequest_list(:un_id)
    into obj_id, text_1, text_2, text_3, text_4
    do
    begin
      if (find_text = '' or f_format('%1~%2~%3~%4', text_1, text_2, text_3, text_4) containing find_text) then
        suspend;
    end
    exit;
  end

  if (rep_id = 'loan_daily_repay') then
  begin
    -- Планируемые погашениия
    -- title_1 = '№ группы'
    -- title_2 = 'Дата погашения'
    -- title_3 = 'Текущий долг'
    -- title_4 = 'Оплачено'
    pr_id_old = 0;
    for select k_id, gr_name, plan_date, round(sum_zadol, 1), round(sum_repay, 1)
        from fm_loan_daily_repay(:un_id, :date_begin, :date_end, :reg_id)
        order by plan_date, gr_name
        into k_id, text_1, dd_tmp, text_3, text_4
    do
    begin
      if (pr_id <> pr_id_old and
          db_id in (39801, 41701)) then
      begin
        pr_id_old = pr_id;
        select iif(pr_code containing '.IPF.', 1, 0)
        from product pr
        where pr.pr_id = :pr_id
        into is_ipf;
        if (is_ipf = 1) then
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'ИПФ'));
        else
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'Займ'));
      end
      if (dd_tmp is null) then
        text_2 = 'Кол: ' || k_id;
      else
        text_2 = f_datetostr(dd_tmp);
      if (k_id < 1000000) then
        k_id = 0; -- В МП дальше не переходит, при нажатии на obj_id = 0
      obj_kind = 'Kontract';
      obj_id = k_id;
      if (find_text = '' or f_format('%1~%2~%3~%4', text_1, text_2, text_3, text_4) containing find_text) then
        suspend;
    end
    exit;
  end
  if (rep_id = 'loan_planning_repay') then
  begin
    -- Планируемые погашениия
    -- title_1 = '№ группы'
    -- title_2 = 'Дата погашения'
    -- title_3 = 'Сумма'
    -- title_4 = 'История'
    for select k_id, plan_date, round(sum_zadol, 1), repay_no_str
        from fm_loan_planning_repay(:un_id, :date_begin, :date_end, 0, :reg_id)
        order by plan_date, k_id
        into :k_id, :dd_tmp, :text_3, :text_4
    do
    begin
      obj_kind = 'Kontract';
      obj_id = k_id;
      text_1 = '';
      select iif(:db_id = 39801, k_kontract_no, kal_new_kf_name), iif(pr_code containing '.IPF.', 1, 0), c.c_name_fio
      from kontract, knaktlist, product, clients c
      where k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            k_clients_cashier_id = c.c_id and
            knaktlist.is_delete = 0 and
            k_from_pr_id = pr_id and
            current_date between kal_date and kal_date_end
      into :text_1, is_ipf, c_name_fio;
      if (db_id in (39801, 41701)) then
      begin
        if (is_ipf = 1) then
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'ИПФ'));
        else
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'Займ'));
        if (text_1 is not null) then
        begin
          select short_fio
          from hp_fio_clear(:c_name_fio)
          into :c_name_fio;
          text_1 = text_1 || ', ' || c_name_fio;
        end
      end
      text_2 = iif(dd_tmp is not null, f_datetostr(dd_tmp), f_lang(lang, 'Не указана'));
      if (find_text = '' or f_format('%1~%2~%3~%4', text_1, text_2, text_3, text_4) containing find_text) then
        suspend;
    end
    exit;
  end
  if (rep_id = 'loan_planning_last_repay') then
  begin
    -- Погашения в ближайшие дни
    -- title_1 = 'Группа'
    -- title_2 = 'Дата'
    -- title_3 = 'Сумма'
    -- title_4 = 'История просрочек'
    limit = -500.0;
    if (db_id = 94401) then
      limit = -10;
    for select k_id, plan_date, round(sum_zadol, 1), repay_no_str
        from fm_loan_planning_repay(:un_id, :date_begin, :date_end, 1, :reg_id)
        order by plan_date, k_id
        into :k_id, :dd_tmp, :text_3, :text_4
    do
    begin
      obj_kind = 'Kontract';
      obj_id = k_id;
      text_4 = '';
      delinq_days = 0;
      select max(datediff(day, ok_sh_date, ok_ol_date))
      from operkred
      where ok_from_k_id = :k_id and
            ok_type in (4, 5) and
            ok_sh_date < ok_ol_date and
            (ok_os + ok_perc) < :limit
      into :delinq_days;
      if (delinq_days is null) then
        delinq_days = 0;
      if (delinq_days = 0) then
        text_4 = 'Нет';
      else
      begin
        text_4 = (case delinq_days
                    when 1 then delinq_days || ' день'
                    when 2 then delinq_days || ' дня'
                    when 3 then delinq_days || ' дня'
                    when 4 then delinq_days || ' дня'
                    else delinq_days || ' дней'
                  end);
      end
      text_1 = '';
      select kal_new_kf_name, iif(pr_code containing '.IPF.', 1, 0), c.c_name_fio
      from kontract, knaktlist, product, clients c
      where k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            k_clients_cashier_id = c.c_id and
            knaktlist.is_delete = 0 and
            k_from_pr_id = pr_id and
            current_date between kal_date and kal_date_end
      into :text_1, is_ipf, c_name_fio;
      if (db_id in (39801, 41701)) then
      begin
        if (is_ipf = 1) then
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'ИПФ'));
        else
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'Займ'));
        if (text_1 is not null) then
        begin
          select short_fio
          from hp_fio_clear(:c_name_fio)
          into :c_name_fio;
          text_1 = text_1 || ', ' || c_name_fio;
        end
      end
      text_2 = iif(dd_tmp is not null, f_datetostr(dd_tmp), f_lang(lang, 'Не указана'));
      if (find_text = '' or f_format('%1~%2~%3~%4', text_1, text_2, text_3, text_4) containing find_text) then
        suspend;
    end
    exit;
  end
  if (rep_id = 'loan_transactions') then
  begin
    -- Фактические операции
    -- title_1 = '№ группы'
    -- title_2 = 'Сумма'
    -- title_3 = 'Время'
    -- title_4 = 'Тип погашения'
    obj_kind = 'operlist';
    pr_id_old = 0;
    for select obj_id, kf_name, res_amount, res_time, ok_type, pr_id
        from fm_loans_transactions(:un_id, :date_end, :reg_id)
        order by res_time desc, obj_id desc
        into :obj_id, :text_1, :text_2, :text_3, :text_4, :pr_id
    do
    begin
      if ((pr_id <> pr_id_old) and
          (db_id in (39801, 41701))) then
      begin
        pr_id_old = pr_id;
        select iif(pr_code containing '.IPF.', 1, 0)
        from product pr
        where pr.pr_id = :pr_id
        into is_ipf;
        if (is_ipf = 1) then
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'ИПФ'));
        else
          text_1 = text_1 || f_format(' (%1)', f_lang(lang, 'Займ'));
      end
      if (find_text = '' or f_format('%1~%2~%3~%4', text_1, text_2, text_3, text_4) containing find_text) then
        suspend;
    end
    exit;
  end
  if (rep_id = 'region_data_list') then
  begin
    -- Справочник офисов
    -- title_1 = 'Офис'
    -- title_2 = 'Регион'
    obj_kind = 'region';
    text_3 = '';
    text_4 = '';

    for select rg.reg_unicode, rg.reg_code || ' ' || rg.reg_name, iif(:db_id in (39801, 41701), rg2.reg_name,rg.reg_stat_rep_region)
        from region rg
        inner join addrlist adl on adl.adl_id = rg.reg_address_fact
        inner join region rg2 on rg2.reg_unicode = rg.parent
        where rg.reg_status = 3 and
              coalesce(rg.reg_date_close, '31.12.2999') >= current_date and
              (rg.reg_code like '%' || :find_text || '%' or
              rg.reg_name like '%' || :find_text || '%' or
              iif(:db_id in (39801, 41701), trim(rg2.reg_name), trim(rg.reg_stat_rep_region)) like '%' || :find_text || '%' or
              iif(:db_id in (39801, 41701), true, trim(adl_address_str)) like '%' || :find_text || '%')
        order by 2, 1
        into :obj_id, :text_1, :text_2
    do
    begin
      suspend;
    end
    exit;
  end

  if (rep_id = 'app_return_consult' and
      db_id in (39801, 41701)) then
  begin
    select ps_from_reg_id
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :lo_id and
          ps_date_begin <= current_date and
          ps_date_end >= current_date and
          ps_status = 1 and
          ps_from_jl_id = jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
          ps_is_working = 1
    into mo_reg_id;
    mo_reg_id = coalesce(mo_reg_id, 0);
    obj_kind = 'app_return_consult';
    text_3 = '';
    text_4 = '';
    for select arc.arc_id, c.c_name_fio, k.k_kontract_no
        from app_return_consult arc
        inner join clients c on arc.arc_from_c_id = c.c_id
        inner join kontract k on k.k_id = arc.arc_from_k_id
        inner join app_return_consult_info arci on arci.arci_id = arc.arc_id and
              arci.arci_app_office_id = :mo_reg_id
        where arc.arc_status = f_enum('app_return_consult', 'process', 'id')
        into :obj_id, :text_1, :text_2
    do
    begin
      suspend;
    end
    exit;
  end
  if (rep_id starting 'mailmessages_') then
  begin
    -- Группа почтовых отчетов
    -- title_1 = 'Заголовок'
    -- title_2 = 'Время'
    text_3 = '';
    text_4 = '';
    for select obj_kind, obj_id, res_sender_fio, res_subject, res_datetime, res_kind
        from fm_mailmessage_list(:un_id, :rep_id)
        into :obj_kind, :obj_id, :res_sender_fio, :res_subject, :res_datetime, :res_kind
    do
    begin
      time_str = '';
      day_between = datediff(day, cast(res_datetime as date), current_date);
      time_str = trim(case day_between
                        when 0 then f_lang(lang, 'сегодня')
                        when 1 then f_lang(lang, 'вчера')
                        else f_datetostr(res_datetime)
                      end);
      time_str = time_str || '<br>' || right('0' || extract(hour from res_datetime), 2) || ':' || right('0' || extract(minute from res_datetime), 2);

      temp_text_1 = res_sender_fio || '<br><small>' || res_kind || ': ' || res_subject || '</small>';
      temp_text_length = char_length(temp_text_1);
      if (char_length(temp_text_1) > 255) then
      begin
        res_subject = substring(res_subject from 1 for (char_length(res_subject) - (temp_text_length - 255) - 3));
        text_1 = res_sender_fio || '<br><small>' || res_kind || ': ' || res_subject || '...' || '</small>';
      end
      else
