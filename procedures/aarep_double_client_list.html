CREATE PROCEDURE AAREP_DOUBLE_CLIENT_LIST(
    C_ID_1 D_ID,
    C_ID_2 D_ID,
    LIST_KIND D_STR_CODE = null,
    SHOW_COUNT D_INTEGER = 500)
RETURNS (
    DC_FROM_C_ID_1 DECIMAL(18,0),
    DC_FROM_C_ID_2 DECIMAL(18,0),
    DC_IDENTITY_KOEF INTEGER,
    DC_IS_CRITICAL INTEGER,
    DC_SAME_FIO INTEGER,
    DC_SAME_BIRTH_DATE INTEGER,
    DC_SAME_PASS_NO INTEGER,
    DC_SAME_PASS_DATE INTEGER,
    DC_SAME_CONTACT INTEGER,
    DC_SAME_CODE INTEGER,
    DC_FROM_UN_ID DECIMAL(18,0),
    DC_DATETIME TIMESTAMP,
    DC_CLIENT_FIO_1 VARCHAR(100),
    DC_CLIENT_FIO_2 VARCHAR(100),
    DC_BIRTH_DATE_1 DATE,
    DC_BIRTH_DATE_2 DATE,
    DC_PASS_NO_1 VARCHAR(30),
    DC_PASS_NO_2 VARCHAR(30),
    DC_PASS_DATE_1 DATE,
    DC_PASS_DATE_2 DATE,
    DC_PASS_DATE_END_1 DATE,
    DC_PASS_DATE_END_2 DATE,
    DC_CODE_1 VARCHAR(50),
    DC_CODE_2 VARCHAR(50),
    DC_ADDRESS_1 VARCHAR(255),
    DC_ADDRESS_2 VARCHAR(255),
    DC_VERIFY_STATUS_1 INTEGER,
    DC_VERIFY_STATUS_2 INTEGER,
    DC_PHOTO_1 INTEGER,
    DC_PASS1_SCAN_1 INTEGER,
    DC_PASS2_SCAN_1 INTEGER,
    DC_PHOTO_2 INTEGER,
    DC_PASS1_SCAN_2 INTEGER,
    DC_PASS2_SCAN_2 INTEGER,
    DC_STATUS_1 VARCHAR(255),
    DC_STATUS_2 VARCHAR(255),
    DC_LINE_1 VARCHAR(20),
    DC_LINE_2 VARCHAR(20),
    CLIENT_IMAGES D_STR_8100,
    DC_STATE_UN_NAME D_STR_100,
    DC_STATE_DATETIME D_DATETIME)
AS
declare variable c_pincode varchar(20);
declare variable state_id d_smallint;
declare variable client_blob d_str_8100;
declare variable dc_state_un_id d_id;
declare variable dc_state d_integer;
begin
  dc_line_1 = 'Первый';
  dc_line_2 = 'Второй';
  if (list_kind is null) then
    list_kind = '';

  if (coalesce(show_count, 0) = 0) then
    show_count = 500;

  state_id = -2;
  if (list_kind = 'list') then
    state_id = 0;

  if (list_kind = 'full') then
  begin
    state_id = 0;
    show_count = 100000;
  end

  if (list_kind = 'process') then
  begin
    state_id = 1;
    show_count = 100000;
  end

  if (list_kind = 'ignored') then
  begin
    state_id = -1;
    show_count = 100000;
  end

  for select dc_from_c_id_1, dc_from_c_id_2, dc_identity_koef, dc_is_critical, dc_same_fio, dc_same_birth_date,
             dc_same_pass_no, dc_same_pass_date, dc_same_contact, dc_same_code, dc_from_un_id, dc_datetime,
             dc_state, dc_state_un_id, dc_state_datetime
      from doubleclients
      where (:c_id_1 = 0 or
            dc_from_c_id_1 = :c_id_1) and
            (:c_id_2 = 0 or
            dc_from_c_id_2 = :c_id_2) and
            (:state_id = -2 or
            (:state_id = -1 and
            dc_state = -1) or
            (:state_id = 0 and
            dc_state >= 0) or
            (:state_id = 1 and
            dc_state > 0))
      order by dc_state desc, dc_same_fio desc, dc_identity_koef desc, coalesce(dc_state_datetime, dc_datetime) desc
      rows :show_count
      into :dc_from_c_id_1, :dc_from_c_id_2, :dc_identity_koef, :dc_is_critical, :dc_same_fio, :dc_same_birth_date,
           :dc_same_pass_no, :dc_same_pass_date, :dc_same_contact, :dc_same_code, :dc_from_un_id, :dc_datetime,
           dc_state, dc_state_un_id, dc_state_datetime
  do
  begin
    dc_address_1 = '';
    dc_address_2 = '';
    dc_verify_status_1 = 0;
    dc_verify_status_2 = 0;
    dc_state_datetime = dc_datetime;
    dc_state_un_name = '';

    if (dc_state = -1 and dc_state_un_id is not null) then
    begin
      select un_name
      from username
      where un_id = :dc_state_un_id
      into dc_state_un_name;

      dc_state_datetime = dc_state_datetime;
    end

    -- Данные по первому клиенту
    c_pincode = '';
    select c_name_fio, c_birth_date, c_pass_no, c_pass_date, c_pass_date_expire, c_inn, c_pincode, c_verify_status,
           iif(coalesce(c_blob_photo_id, 0) <> 0, 1, 0), iif(coalesce(c_blob_pass1_id, 0) <> 0, 1, 0),
           iif(coalesce(c_blob_pass2_id, 0) <> 0, 1, 0)
    from clients
    where c_id = :dc_from_c_id_1
    into :dc_client_fio_1, dc_birth_date_1, dc_pass_no_1, dc_pass_date_1, dc_pass_date_end_1, dc_code_1, c_pincode,
         dc_verify_status_1, dc_photo_1, dc_pass1_scan_1, dc_pass2_scan_1;

    if (dc_code_1 is null) then
      dc_code_1 = '';
    if (c_pincode is null) then
      c_pincode = '';
    if (c_pincode <> '') then
    begin
      if (dc_code_1 <> '') then
        dc_code_1 = dc_code_1 || '/' || c_pincode;
      else
        dc_code_1 = c_pincode;
    end
    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :dc_from_c_id_1 and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into :dc_address_1;

    dc_status_1 = '';
    select c_status
    from hp_find_client_status(:dc_from_c_id_1)
    into :dc_status_1;

    -- Данные по второму клиенту
    c_pincode = '';
    select c_name_fio, c_birth_date, c_pass_no, c_pass_date, c_pass_date_expire, c_inn, c_pincode, c_verify_status,
           iif(coalesce(c_blob_photo_id, 0) <> 0, 1, 0), iif(coalesce(c_blob_pass1_id, 0) <> 0, 1, 0),
           iif(coalesce(c_blob_pass2_id, 0) <> 0, 1, 0)
    from clients
    where c_id = :dc_from_c_id_2
    into :dc_client_fio_2, dc_birth_date_2, dc_pass_no_2, dc_pass_date_2, dc_pass_date_end_2, dc_code_2, c_pincode,
         dc_verify_status_2, dc_photo_2, dc_pass1_scan_2, dc_pass2_scan_2;

    if (dc_code_2 is null) then
      dc_code_2 = '';
    if (c_pincode is null) then
      c_pincode = '';
    if (c_pincode <> '') then
    begin
      if (dc_code_2 <> '') then
        dc_code_2 = dc_code_2 || '/' || c_pincode;
      else
        dc_code_2 = c_pincode;
    end
    select adl_address_str
    from addrlist, addrclient
    where adc_from_c_id = :dc_from_c_id_2 and
          adc_address_type = 1 and
          adc_from_adl_id = adl_id
    into :dc_address_2;

    dc_status_2 = '';
    select c_status
    from hp_find_client_status(:dc_from_c_id_2)
    into :dc_status_2;

    client_images = '';
    client_blob = '';

    select res_image_list
    from hp_get_client_doc_images(:dc_from_c_id_1, 1, 1, '', null)
    into :client_images;

    select res_image_list
    from hp_get_client_doc_images(:dc_from_c_id_2, 1, 1, '', null)
    into :client_blob;

    if (client_blob <> '') then
    begin
      if (client_images = '')
      then client_images = client_blob;
      else
      begin
        client_images = client_images || ',' || client_blob;
      end
    end

    client_images = '[' || client_images || ']';

    suspend;
  end
end
