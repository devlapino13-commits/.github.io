CREATE PROCEDURE HP_OFFICERPAR7_UPDATE_AZ(
    REP_DATE_END DATE)
AS
declare variable par1_table curr_type;
declare variable oks_kc_id decimal(18,0);
declare variable oks_balance decimal(18,4);
declare variable oks_delinq_balance decimal(18,4);
declare variable par1 decimal(18,4);
begin

  for select oks_kc_id, sum(oks_balance), sum(iif(:rep_date_end - oks_delinq_date >= 2, oks_balance, 0))
      from oksummary
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0 and
            oks_balance > 0
      group by oks_kc_id
      into oks_kc_id, oks_balance, oks_delinq_balance
  do
  begin
    par1 = oks_delinq_balance / oks_balance * 100;
    par1_table = null;
    select op_par8_180
    from officerpar7
    where op_kc_id = :oks_kc_id and
          :rep_date_end between op_date_begin and op_date_end
    into par1_table;
    if (par1_table is null) then
      par1_table = 0;

    if (par1 > 10) then
    begin
      if (par1_table <= 10.00) then
      begin
        insert into officerpar7 (op_kc_id, op_date_begin, op_balance, op_delinq_bal8_180, op_par8_180, op_memo)
        values (:oks_kc_id, :rep_date_end + 1, :oks_balance, :oks_delinq_balance, :par1,
                'Доступ закрыт к выдачам');
      end
    end
    else
    begin
      if (par1_table > 10.00) then
      begin
        insert into officerpar7 (op_kc_id, op_date_begin, op_balance, op_delinq_bal8_180, op_par8_180, op_memo)
        values (:oks_kc_id, :rep_date_end + 1, :oks_balance, :oks_delinq_balance, :par1,
                'Доступ открыт к выдачам');
      end
    end

--    suspend;
  end
end
