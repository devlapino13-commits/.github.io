CREATE PROCEDURE FM_PHONECALL_SET_STATUS(
    PNC_ID D_ID,
    PNC_CALL_STATUS D_STR_CODE,
    PNC_PRV_PHONE D_STR_20,
    PNC_START D_DATETIME,
    PNC_FINISH D_DATETIME,
    PNC_DURATION D_INTEGER,
    PNC_FILENAME D_STR_255)
AS
declare variable pnc_call_status_id d_integer;
declare variable pnc_prv_phone_id d_integer;
begin
  -- Процедура вызывается из Asterisk, после завершения звонка, или в случае
  -- неудачной попытки, для сохранения результатов

  -- Получаем clt_id для переданного номера провайдера
  pnc_prv_phone_id = f_cont_id(pnc_prv_phone, 'add');

  -- Конвертируем статус
  -- Есть еще искусственный статус 0 - not dialed, когда еще не было звонка,
  -- но этот статус не приходит с Asterisk'а
  pnc_call_status_id =
    case pnc_call_status
      when 'answered' then 1
      when 'no answer' then -1
      when 'failed' then -2
      when 'busy' then -3
      else -4 -- 'unknown'
    end;

  update phonecalls
  set pnc_prv_phone_id = coalesce(:pnc_prv_phone_id, pnc_prv_phone_id),
      pnc_call_status_id = coalesce(:pnc_call_status_id, pnc_call_status_id),
      pnc_start = coalesce(:pnc_start, pnc_start),
      pnc_finish = coalesce(:pnc_finish, pnc_finish),
      pnc_duration = coalesce(:pnc_duration, pnc_duration),
      pnc_filename = coalesce(:pnc_filename, pnc_filename)
  where pnc_id = :pnc_id;

end
