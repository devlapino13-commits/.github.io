CREATE PROCEDURE FM_LOAN_TOTAL_PLAN_TMP(
    UN_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    PLAN_DATE DATE,
    SUM_PLAN DECIMAL(18,4),
    SUM_LAST DECIMAL(18,4),
    SUM_ADVANCE DECIMAL(18,4),
    SUM_DISB DECIMAL(18,4),
    SUM_WRITEOFF DECIMAL(18,4),
    COUNT_DISB DECIMAL(18,4),
    COUNT_CLOSE DECIMAL(18,4),
    COUNT_CLOSE_ADV DECIMAL(18,4))
AS
declare variable c_id id_type;
declare variable reg_id integer;
declare variable last_date date;
declare variable k_id decimal(18,0);
declare variable sum_tmp decimal(18,4) = 0;
declare variable aa_id integer;
declare variable aa_client_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable count_tmp decimal(18,4);
declare variable del_amount decimal(18,4) = 0;
begin
-- Планируемое изменение КП за период

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exit;

  select ps_from_reg_id
  from hr_pstatus, joblist, jobkind
  where ps_from_c_id = :c_id and
        ps_date_begin <= current_date and
        ps_date_end >= current_date and
        ps_status = 1 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id and
        jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
        ps_is_working = 1
  into :reg_id;
  if (reg_id is null) then
    reg_id = 0;

  sum_plan = 0;
  sum_last = 0;
  sum_advance = 0;
  sum_disb = 0;
  sum_writeoff = 0;
  count_close = 0;
  count_disb = 0;
  count_close_adv = 0;

  plan_date = rep_date_begin;
  while (plan_date <= rep_date_end) do
  begin
    suspend;
    plan_date = plan_date + 1;
  end

  if (reg_id = 0) then
  begin
    plan_date = rep_date_begin - 1;
    sum_plan = 0;
    sum_last = 0;
    sum_advance = 0;
    sum_disb = 0;
    sum_writeoff = 0;
    count_close = 0;
    count_disb = 0;
    count_close_adv = 0;

-- Фактиские операции за сегодня
    for select distinct oks_from_k_id
        from oksummary
        where oks_kc_id = :c_id and
              oks_close_state = 0 and
              ((oks_date_begin = :rep_date_begin and
              oks_date_end >= :rep_date_begin) or
              (oks_date_end = :rep_date_begin - 1))
        into k_id
    do
    begin
      sum_plan = 0;
      sum_last = 0;
      sum_advance = 0;
      sum_disb = 0;
      sum_writeoff = 0;
      count_close = 0;
      count_disb = 0;
      count_close_adv = 0;

      select sum(iif(ok_type = 4 and ok_sh_date = ok_ol_date, ok_os, 0.0)),
             sum(iif(ok_type = 4 and ok_sh_date < ok_ol_date, ok_os, 0.0)),
             sum(iif(ok_type = 4 and ok_sh_date > ok_ol_date, ok_os, 0.0)), sum(iif(ok_type = 5, ok_os, 0.0)),
             sum(iif(ok_type = 8, ok_os, 0.0))
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date = :rep_date_begin and
            ok_type in (4, 5, 8) and
            ok_from_k_id = :k_id
      into sum_plan, sum_last, sum_advance, sum_writeoff, sum_disb;
      if (sum_plan is null) then
        sum_plan = 0;
      if (sum_last is null) then
        sum_last = 0;
      if (sum_advance is null) then
        sum_advance = 0;
      if (sum_writeoff is null) then
        sum_writeoff = 0;
      if (sum_disb is null) then
        sum_disb = 0;
      if (sum_plan + sum_last + sum_advance + sum_writeoff <> 0.0 or sum_disb <> 0.0) then
      begin
        select k_kolich_client_approved
        from kontract
        where k_id = :k_id
        into :count_tmp;

        if (sum_disb <> 0.0) then
          count_disb = count_tmp;
        else
        begin
          if (exists(select *
                     from kontstate
                     where ks_ol_is_delete = 0 and
                           ks_date_begin = :rep_date_begin and
                           ks_state = 2 and
                           ks_from_k_id = :k_id)) then
          begin
            if (sum_advance = 0) then
              count_close = -count_tmp;
            else
              count_close_adv = -count_tmp;
          end
        end
      end
      suspend;
    end

    sum_plan = 0;
    sum_last = 0;
    sum_advance = 0;
    sum_disb = 0;
    sum_writeoff = 0;
    count_close = 0;
    count_disb = 0;
    count_close_adv = 0;

-- Планируемые погашения
    for select oks_from_k_id, oks_delinq_date, k_kredline_end, k_kolich_client_approved
        from oksummary, kontract
--      where oks_delinq_date between :rep_date_begin and :rep_date_end
        where oks_delinq_date <= :rep_date_end and
              oks_kc_id = :c_id and
              oks_close_state = 0 and
              oks_date_begin <= :rep_date_begin and
              oks_date_end >= :rep_date_begin and
              k_id = oks_from_k_id
--      and oks_delinq_date>=:rep_date_begin
        into k_id, plan_date, last_date, count_tmp
    do
    begin
      if (plan_date < :rep_date_begin) then
      begin
        plan_date = null;
        select min(ok_sh_date)
        from operkred
        where ok_from_k_id = :k_id and
              ok_sh_date between :rep_date_begin and :rep_date_end and
              ok_ol_date <= :rep_date_end and
              ok_type < 7 and
              ok_ol_is_active = 1 having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 0
        into plan_date;
      end
      if (plan_date is not null) then
      begin
        sum_plan = 0;
        sum_last = 0;
        sum_advance = 0;
        sum_disb = 0;
        sum_writeoff = 0;
        count_close = 0;
        count_disb = 0;
        count_close_adv = 0;

        select sum(zad_os)
        from aarep_make_zad_list(:k_id, :plan_date, 1)
        where sh_date = :plan_date
        into sum_plan;
        if (sum_plan is null) then
          sum_plan = 0;
        sum_plan = -sum_plan;

        if (plan_date = last_date) then
        begin
          sum_last = sum_plan;
          sum_plan = 0;
          count_close = -count_tmp;
        end
        suspend;
      end
    end

-- Досрочное закрытия
    for select aa_id, aa_date_repay, aa_from_k_id, aa_from_c_id, k_kolich_client_approved
        from apadvanced, kontract, knaktlist
        where aa_from_k_id = k_id and
              aa_is_deleted = 0 and
              aa_date_app <= :rep_date_begin and
              aa_date_repay between :rep_date_begin and :rep_date_end and
              knaktlist.is_delete = 0 and
              kal_date <= :rep_date_begin and
              kal_date_end >= :rep_date_begin and
              kal_from_kf_id = k_from_kf_id and
              kal_new_kf_officer = :c_id
        into :aa_id, plan_date, :k_id, :aa_client_id, :count_tmp
    do
    begin
      sum_plan = 0;
      sum_last = 0;
      sum_advance = 0;
      sum_disb = 0;
      sum_writeoff = 0;
      count_close = 0;
      count_disb = 0;
      count_close_adv = 0;

      select sum(rep_amount)
      from aarep_apadvanced_print(:aa_id)
      into :sum_advance;
      if (sum_advance is null) then
        sum_advance = 0;

      if (exists(select *
                 from oksummary
                 where oks_delinq_date between :rep_date_begin and :rep_date_end and
                       oks_close_state = 0 and
                       oks_date_begin <= :rep_date_begin and
                       oks_date_end >= :rep_date_begin and
                       oks_from_k_id = :k_id)) then
      begin
        select sum(zad_os)
        from aarep_make_zad_list(:k_id, :plan_date, 1)
        where sh_date = :plan_date
        into sum_tmp;
        if (sum_tmp is null) then
          sum_tmp = 0;
        if (sum_tmp > 0 and
            aa_client_id is not null) then
        begin
          select (cast(:sum_tmp as decimal(18,2)) / k_amount_approved) * kc_amount_approved
          from kontract, knclient
          where k_id = :k_id and
                kc_from_k_id = :k_id and
                kc_amount_approved > 0 and
                k_amount_approved > 0 and
                kc_from_c_id = :aa_client_id
          into :sum_tmp;
        end
        sum_advance = sum_advance - sum_tmp;
        if (sum_advance <> 0) then
        begin
          if (aa_client_id is not null) then
            count_close_adv = -1;
          else
            count_close_adv = -count_tmp;
        end
      end
      suspend;
    end

-- Планируемые выдачи по заявкам
    for select ma_from_kf_id, ma_date_oper, ma_summa_minus, ma_kolich_client
        from miniapp, knaktlist
        where ma_is_active = 1 and
              ma_date_app <= :rep_date_begin and
              ma_date_end >= :rep_date_begin and
              ma_date_oper between :rep_date_begin and :rep_date_end and
              ma_is_active = 1 and
              knaktlist.is_delete = 0 and
              kal_date <= :rep_date_begin and
              kal_date_end >= :rep_date_begin and
              kal_from_kf_id = ma_from_kf_id and
              kal_new_kf_officer = :c_id
        into :kf_id, plan_date, sum_tmp, count_tmp
    do
    begin
      sum_plan = 0;
      sum_last = 0;
      sum_advance = 0;
      sum_disb = 0;
      sum_writeoff = 0;
      count_close = 0;
      count_disb = 0;
      count_close_adv = 0;

      count_disb = count_tmp;

      if (plan_date = rep_date_begin) then
      begin
        if (exists(select *
                   from kontstate, kontract
                   where ks_date_begin = :rep_date_begin and
                         ks_ol_is_delete = 0 and
                         ks_state = 1 and
                         ks_from_k_id = k_id and
                         k_from_kf_id = :kf_id)) then
        begin
          sum_tmp = 0;
          count_disb = 0;
        end
      end
      sum_disb = sum_tmp;
      suspend;
    end

  end
end
