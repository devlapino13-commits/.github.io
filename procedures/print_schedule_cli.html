CREATE PROCEDURE PRINT_SCHEDULE_CLI(
    KONT_ID DECIMAL(18,0))
RETURNS (
    SH_ID VARCHAR(10),
    SH_DATE DATE,
    SH_OS DECIMAL(18,4),
    SH_PERC DECIMAL(18,4),
    SH_TOTAL DECIMAL(18,4),
    SH_KOMMIS DECIMAL(18,4),
    SH_DISB_AMOUNT DECIMAL(18,4),
    SH_NAME_FIO VARCHAR(100),
    SH_PERC_TAX DECIMAL(18,4),
    SH_IS_HEADER SMALLINT,
    SH_SUM_DISB CURR_TYPE,
    SH_TOTAL_EX_TAX DECIMAL(18,4),
    KC_BONUS_TEXT VARCHAR(20),
    SH_CL_FULL_TOTAL DECIMAL(18,4))
AS
declare variable sh_os1 decimal(18,4);
declare variable sh_perc1 decimal(18,4);
declare variable sh_kommis1 decimal(18,4);
declare variable disb_all decimal(18,4);
declare variable sh_perc_tax1 decimal(18,4);
declare variable k_kolich_repay smallint;
declare variable last_sum_os decimal(18,4);
declare variable sh_os_summ curr_type;
declare variable sh_perc_summ curr_type;
declare variable sh_total_summ curr_type;
declare variable sh_kommis_summ curr_type;
declare variable sum_perc_kommis curr_type;
declare variable k_amount_approved decimal(18,4);
declare variable k_disb_date date;
declare variable k_nsp_percent decimal(18,7);
declare variable sh_total_ex_tax_summ curr_type;
declare variable sh_perc_tax_summ curr_type;
declare variable c_id decimal(10,0);
declare variable k_bonus integer;
declare variable kc_kommis_lgota decimal(18,4);
declare variable sum_perc decimal(18,4);
declare variable k_kommis_perc decimal(18,7);
declare variable sh_os2 decimal(18,4);
declare variable sh_perc2 decimal(18,4);
declare variable sh_kommis2 decimal(18,4);
declare variable sh_perc_tax2 decimal(18,4);
begin
  k_nsp_percent = 0;
  select k_amount_approved, k_disb_date, kont.k_kommis_perc
  from kontract kont
  where k_id = :kont_id
  into :k_amount_approved, :k_disb_date, :k_kommis_perc;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -1 -- нсп от процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end

  for select k_amount_approved, c_name_fio, kc_amount_approved, k_kolich_repay, c_id, kc.kc_kommis_lgota
      from knclient kc, clients, kontract
      where kc_from_c_id = c_id and
            kc_from_k_id = :kont_id and
            kc_amount_approved <> 0 and
            k_id = :kont_id
      order by kc_number
      into :disb_all, :sh_name_fio, :sh_disb_amount, :k_kolich_repay, :c_id, :kc_kommis_lgota
  do
  begin
    select c_last_bonus_status, c_last_bonus_status_str
    from hp_kontstate_client_status(:c_id, :k_disb_date)
    into :k_bonus, kc_bonus_text;

    sh_os_summ = 0;
    sh_perc_summ = 0;
    sh_perc_tax_summ = 0;
    sh_total_summ = 0;
    sh_total_ex_tax_summ = 0;
    sh_kommis_summ = 0;
    last_sum_os = sh_disb_amount;

    for select sh_id, sh_date, sh_os, sh_perc, sh_kommis
    from hp_get_base_schedule(:kont_id)
     order by sh_date
    into :sh_id, :sh_date, :sh_os1, :sh_perc1, :sh_kommis1
    do
    begin
      sh_perc_tax1 = round((sh_perc1 + sh_kommis1) * k_nsp_percent / 100);

      sum_perc_kommis = 0;

      select sum(sh_perc)
      from schedule
      where sh_from_k_id = :kont_id
      into :sum_perc;
      if (sum_perc is null) then
        sum_perc = 0;

      sum_perc = cast(sum_perc * (sh_disb_amount / disb_all) as decimal(18,2));

      sh_os = cast(sh_os1 * (sh_disb_amount / disb_all) as decimal(18,2));
      sh_perc = cast(sh_perc1 * (sh_disb_amount / disb_all) as decimal(18,2));

      if (sh_kommis1 > 0) then
        sh_kommis = sh_disb_amount * k_kommis_perc / 100 - kc_kommis_lgota; -- было до 28.11.13 Cast(sh_kommis1*(sh_disb_amount/disb_all) as decimal(18,2));
      else
        sh_kommis = 0;

      sum_perc_kommis = sum_perc + sh_kommis;

      sh_perc_tax = cast(sh_perc_tax1 * (sh_disb_amount / disb_all) as decimal(18,2));
      sh_total = sh_os + sh_perc + sh_perc_tax + sh_kommis;
      sh_total_ex_tax = sh_os + sh_perc + sh_kommis;
      sh_sum_disb = null;
      if (sh_id = 0) then
      begin
        sh_is_header = 1;
        sh_sum_disb = sum_perc_kommis;

        sh_cl_full_total = 0;
        for select sum(sh_os), sum(sh_perc), sum(sh_kommis)
            from schedule
            where sh_from_k_id = :kont_id
            into :sh_os1, :sh_perc1, :sh_kommis1
        do
        begin
          sh_os2 = cast(sh_os1 * (sh_disb_amount / disb_all) as decimal(18,2));
          sh_perc2 = cast(sh_perc1 * (sh_disb_amount / disb_all) as decimal(18,2));

          if (sh_kommis1 > 0) then
            sh_kommis2 = sh_disb_amount * k_kommis_perc / 100 - kc_kommis_lgota; -- было до 28.11.13 Cast(sh_kommis1*(sh_disb_amount/disb_all) as decimal(18,2));
          else
            sh_kommis2 = 0;

          sh_perc_tax2 = round((sh_perc2 + sh_kommis2) * k_nsp_percent / 100);

          sh_perc_tax2 = cast(sh_perc_tax2 * (sh_disb_amount / disb_all) as decimal(18,2));
          sh_cl_full_total = sh_cl_full_total + sh_os2 + sh_perc2 + sh_perc_tax2 + sh_kommis2;
        end
      end
      else
      begin
        sh_is_header = 0;
        sh_sum_disb = null;
      end
      if (sh_id = k_kolich_repay) then
      begin
        sh_os = last_sum_os;
        sh_total = sh_os + sh_perc + sh_perc_tax;
        sh_total_ex_tax = sh_os + sh_perc;
      end

--      sh_id1 = sh_id;
--      sh_id = null;
      suspend;
--      sh_id = sh_id1;

      sh_os_summ = sh_os_summ + sh_os;
      sh_perc_summ = sh_perc_summ + sh_perc;
      sh_perc_tax_summ = sh_perc_tax_summ + sh_perc_tax;
      sh_total_summ = sh_total_summ + sh_total; -- + sh_kommis;
      sh_total_ex_tax_summ = sh_total_ex_tax_summ + sh_total_ex_tax; -- + sh_kommis;
      sh_kommis_summ = sh_kommis_summ + sh_kommis;
      if (sh_id = k_kolich_repay) then
      begin
        sh_os = sh_os_summ;
        sh_perc = sh_perc_summ;
        sh_perc_tax = sh_perc_tax_summ;
        sh_total = sh_total_summ;
        sh_total_ex_tax = sh_total_ex_tax_summ;
        sh_kommis = sh_kommis_summ;
        sh_id = null;
        sh_date = null;
--        sh_disb_amount = null;
--        sh_name_fio = null;
        sh_sum_disb = :sh_disb_amount;
        sh_id = 'Всего';
        suspend;
      end

      last_sum_os = last_sum_os - coalesce(sh_os, 0);
    end
  end
end
