CREATE PROCEDURE LLC_LIST_PRIOR(
    ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    LK_KONTRACT_ID DECIMAL(18,0),
    LK_DELIQ_DAYS_PRIOR INTEGER,
    LK_STATUS_PRIOR SMALLINT,
    LK_PERCENT_PRIOR DECIMAL(18,7),
    LK_DISB_DATE DATE,
    LK_KONT_NO VARCHAR(20))
AS
declare variable perc_gr decimal(18,7);
declare variable perc_ind decimal(18,7);
declare variable loan_t integer;
declare variable prior_status integer;
begin
  /* Procedure Text */
  for select lk_from_k_id, k_disb_date, k_kontract_no, k_from_cs_id, cs_is_for_group, cs_is_for_indiv, pr_loan_type
      from llc_kont, kontract, classific, product
      where lk_from_lm_id = :id and
            k_id = lk_from_k_id and
            k_from_cs_id = cs_id and
            k_from_pr_id = pr_id
      into :lk_kontract_id, :lk_disb_date, :lk_kont_no, :prior_status, :perc_gr, :perc_ind, :loan_t
  do
  begin
    lk_deliq_days_prior = 0;
    lk_status_prior = 0;
    lk_percent_prior = 0;
    select lk_status, lk_percent, lk_deliq_days
    from llc_kont
    where lk_date < :dd and
          lk_date_end >= (:dd - 1) and
          lk_from_k_id = :lk_kontract_id
    order by lk_date desc
    rows 1
    into :lk_status_prior, :lk_percent_prior, :lk_deliq_days_prior;
    if (lk_status_prior = 0) then
    begin
      if (loan_t = 0) then
        lk_percent_prior = :perc_gr;
      else
        lk_percent_prior = :perc_ind;
      lk_status_prior = :prior_status;
    end
    suspend;
  end
end
