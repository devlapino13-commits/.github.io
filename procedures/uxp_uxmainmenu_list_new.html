CREATE PROCEDURE UXP$UXMAINMENU_LIST_NEW(
    UN_ID D_ID,
    APP_NAME D_STR_CODE)
RETURNS (
    ML_ID D_INTEGER,
    ML_PARENT_ID D_INTEGER,
    ML_CODE D_STR_100,
    ML_PARENT_CODE D_STR_100,
    ML_CAPTION D_STR_255,
    ML_HINT D_STR_LONG,
    ML_UA_CODE D_STR_CODE,
    ML_UA_UUID D_UUID,
    ML_INDEX D_INTEGER,
    ML_KIND D_INTEGER,
    ML_INACTIVE D_BOOL,
    ML_IMAGE_CODE D_STR_CODE,
    ML_LEVEL D_INTEGER)
AS
declare variable ua_image_linked d_bool;
begin
  app_name = trim(coalesce(app_name, ''));
  if (app_name = '') then
    app_name = uxf$appname();

  if (app_name = '' or app_name = 'ibexpert') then
    app_name = 'FsConfig';

  if (un_id is null) then
    un_id = uxf$user();

  ml_inactive = 0;
  for select uml_ml_id, min(ml_level), max(uml_ua_code), max(ua_image_linked)
  from menulist, uxp$uxmainmenu_list_tmp(:un_id, :app_name)
  where uml_ml_id = ml_id
  group by 1
  order by min(ml_level), max(ml_position), 1
  into :ml_id, :ml_level, :ml_ua_code, :ua_image_linked
  do
  begin
    ml_ua_uuid = null;
    if (ml_ua_code <> '') then
      ml_ua_uuid = char_to_uuid(ml_ua_code);

    for select ml_parent_id, ml_code, ml_parent_code, ml_caption,
               ml_description, ml_position, ml_kind
    from menulist
    where ml_id = :ml_id
    into ml_parent_id, ml_code, ml_parent_code, ml_caption,
         ml_hint, ml_index, ml_kind
    do
    begin
      ml_image_code = '';
      if (ua_image_linked = 1) then
      begin
        select uuid_to_char(ua_img_uuid)
        from ux$action
        where ua_uuid = :ml_ua_uuid
        into ml_image_code;
      end

      suspend;
    end
  end
end
