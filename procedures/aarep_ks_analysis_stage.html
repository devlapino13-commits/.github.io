CREATE PROCEDURE AAREP_KS_ANALYSIS_STAGE(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    KF_OFFICER_NAME VARCHAR(100),
    K_KONTRACT_BALANCE DECIMAL(18,4),
    K_KOLICH_CLIENT_ACTIVE INTEGER,
    K_DELIQ_BALANCE DECIMAL(18,4),
    K_DELIQ_BALANCE7 DECIMAL(18,4),
    K_DELIQ_BALANCE30 DECIMAL(18,4),
    PS_IS_WORKING_TEXT VARCHAR(50),
    PS_IS_ON_STAFF_STR VARCHAR(20),
    JL_NAME VARCHAR(100),
    KC_STAGE_YEARS INTEGER,
    KC_STAGE_MONTH INTEGER,
    AA_KC_STAGE VARCHAR(50))
AS
declare variable kal_new_kf_officer decimal(18,0);
declare variable ps_is_working smallint;
declare variable ps_is_on_staff integer;
declare variable ps_date date;
begin
  for select reg_name, parent_name, kal_new_kf_officer, kf_officer_name, sum(k_kontract_balance),
             sum(k_kolich_client_active), sum(k_deliq_balance), sum(k_deliq_balance7), sum(k_deliq_balance30)
      from aarep_ks_analysis_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_currency)
      group by reg_name, parent_name, kal_new_kf_officer, kf_officer_name

      into :reg_name, :parent_name, :kal_new_kf_officer, :kf_officer_name, :k_kontract_balance, :k_kolich_client_active,
           :k_deliq_balance, :k_deliq_balance7, :k_deliq_balance30
  do
  begin
    ps_is_working = null;
    ps_is_working_text = null;

    select ps_is_working
    from hr_pstatus, hr_orders, sa_worktype
    where :rep_date_end between ps_date_begin and ps_date_end and
          ps_from_c_id = :kal_new_kf_officer and
          ps_from_ho_id = ho_id and
          ho_order_type = sawt_id
    into :ps_is_working;

    if (ps_is_working = 1) then
      ps_is_working_text = 'Работает';
    else
    begin
      if (ps_is_working = 0) then
        ps_is_working_text = 'Уволен';
      else
        ps_is_working_text = 'Не оперделен';
    end

    ps_is_on_staff = null;
    jl_name = null;
    select ps_is_on_staff, jl_name
    from hr_pstatus, joblist
    where ps_from_c_id = :kal_new_kf_officer and
          :rep_date_end between ps_date_begin and ps_date_end and
          ps_is_working = 1 and
          ps_from_jl_id = jl_id
    into :ps_is_on_staff, :jl_name;

    if (ps_is_on_staff is not null) then
      if (ps_is_on_staff = 1) then
        ps_is_on_staff_str = 'Штатный';
      else
        ps_is_on_staff_str = 'Внештатный';
    else
      ps_is_on_staff_str = 'Приказ не оформлен';

    ps_date = '01.01.1900';
    kc_stage_month = null;
    kc_stage_years = null;
    aa_kc_stage = null;

    select ps_date_begin
    from hr_pstatus, hr_orders, joblist, jobkind /*Вычисляет стаж КС*/
    where ps_from_c_id = :kal_new_kf_officer and
          ps_from_ho_id = ho_id and
          (ps_date_begin is not null) and
          jl_id = ps_from_jl_id and
          jl_kind = jk_id and
          jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner')
    order by ps_date_begin
    rows 1
    into :ps_date;
    if ((ps_date is null) or (ps_date = '01.01.1900')) then
      aa_kc_stage = 'XX Стаж не определен';
    else
    begin
      kc_stage_month = datediff(month, :ps_date, :rep_date_end);/*между двумя датами определяет сколько месяцев*/
      kc_stage_years = kc_stage_month / 12;
      if (kc_stage_years < 1) then
        aa_kc_stage = '0. Стаж КС менее года';
      else
      if (kc_stage_years = 1) then
        aa_kc_stage = '1. Стаж КС год';
      else
      if (kc_stage_years > 1) then
        aa_kc_stage = '2. Стаж КС более года';
    end
    suspend;
  end

end
