Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXUNITITEM_EVENTS(
    UNIT_UUID D_UUID)
RETURNS (
    UNP_UNI_UUID D_UUID,
    UNP_UNC_UUID D_UUID,
    UNI_CODE D_STR_CODE,
    UNI_CAPTION D_STR_100,
    UNC_CLASS_NAME D_STR_CODE,
    UNC_PROP_NAME D_STR_100,
    UNI_JSON_PATH D_STR_LONG,
    UNP_STATE D_SMALLINT,
    UNP_VALUE D_STR_255,
    UNP_MEMO D_STR_255)
AS
begin
  for select uni_uuid, unp_unc_uuid, uni_json_path, uni_code, uni_caption,
             unc_class_name, unc_prop_name,
             unp_state, unp_value, unp_memo
  from ux$unititem, ux$unitprops p, ux$unitclass
  where uni_unt_uuid = :unit_uuid and
        unp_uni_uuid = uni_uuid and
        unp_unc_uuid = unc_uuid and
        unc_is_event = 1 and
        unp_value <> ''
  into unp_uni_uuid, unp_unc_uuid, uni_json_path, uni_code, uni_caption,
       unc_class_name, unc_prop_name,
       unp_state, unp_value, unp_memo
  do
    suspend;
end
