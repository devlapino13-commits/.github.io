CREATE PROCEDURE AAREP_STAFFTEAM_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    RES_LO_ID DECIMAL(18,0),
    RES_REGION_ID INTEGER,
    RES_IS_WORKING INTEGER,
    RES_JOBNAME VARCHAR(100),
    RES_JOBKIND VARCHAR(100),
    RES_JOBMEMO VARCHAR(100),
    RES_LO_FIO VARCHAR(100),
    RES_REGION_NAME VARCHAR(100),
    RES_TEAM_ID INTEGER,
    RES_TEAM_NAME VARCHAR(100),
    RES_TEAM_LEADER INTEGER,
    RES_TEAM_DATE DATE)
AS
begin
  for select res_lo_id, res_region_id, res_is_working, res_jobname, res_jobkind, res_jobmemo, res_lo_fio,
             res_region_name, res_team_id, res_team_name, res_team_leader, res_team_date
      from aarep_staffteam_list_tmp(:rep_region_id, :rep_date_end)
      where res_lo_id <> 0
      into :res_lo_id, :res_region_id, :res_is_working, :res_jobname, :res_jobkind, :res_jobmemo, :res_lo_fio,
           :res_region_name, :res_team_id, :res_team_name, :res_team_leader, :res_team_date
  do
  begin
    suspend;
  end
end
