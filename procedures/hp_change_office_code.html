CREATE PROCEDURE HP_CHANGE_OFFICE_CODE(
    REGION_ORIG_ID INTEGER,
    REGION_ORIG_SHORT_CODE VARCHAR(10),
    REGION_NEW_ID INTEGER,
    REGION_NEW_CODE VARCHAR(10),
    REGION_NEW_SHORT_CODE VARCHAR(10),
    REGION_NEW_BRANCH INTEGER)
AS
declare variable region_orig_code varchar(10);
declare variable region_acc_format varchar(20);
declare variable short_code_len integer;
declare variable parent_id integer;
begin
  region_orig_code = null;
  select cm_account_format
  from company
  into :region_acc_format;
  short_code_len = char_length(:region_orig_short_code);

  select parent
  from region
  where reg_unicode = :region_new_branch
  into :parent_id;

  select reg_code
  from region
  where reg_unicode = :region_orig_id
  into :region_orig_code;
  if (region_orig_code is not null) then
  begin
    update region
    set reg_unicode = :region_new_id,
        reg_code = :region_new_code,
        reg_head_office = :region_new_branch,
        parent = :parent_id
    where reg_unicode = :region_orig_id;
    if (region_acc_format = 'OOOOCCCBBBBBNNNNNN') then
    begin
      update acc_list
      set al_id = :region_new_code || right(al_id, 14)
      where al_id like :region_orig_code || '%';

      /*
     update standacc
     set sa_account=:REGION_NEW_CODE||Right(sa_account,14)
     where sa_account like :REGION_ORIG_CODE||'%';
     */

      update bankregion
      set br_name = :region_new_code || right(br_name, 14)
      where br_name like :region_orig_code || '%';
    end
    update krexpert
    set ke_code = :region_new_short_code || substring(ke_code from :short_code_len + 1 for 10)
    where ke_code like :region_orig_short_code || '%';
    update knaktlist
    set kal_new_officer_code = :region_new_short_code || substring(kal_new_officer_code from :short_code_len + 1 for 10),
        kal_new_kf_name = :region_new_short_code || substring(kal_new_kf_name from :short_code_len + 1 for 10)
    where kal_new_officer_code like :region_orig_short_code || '%';
  end
end
