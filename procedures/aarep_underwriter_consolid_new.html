CREATE PROCEDURE AAREP_UNDERWRITER_CONSOLID_NEW(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ACL_ID ID_TYPE,
    KF_NAME STR_100,
    K_KONTRACT_NO STR_20,
    K_ID ID_TYPE,
    CL_FIO D_STR_LONG,
    K_AMOUNT DECIMAL(18,2),
    PR_NAME VARCHAR(50),
    K_CIRCLE INTEGER,
    CL_BALL D_INTEGER,
    REG_NAME VARCHAR(50),
    REG_NAMEP STR_100,
    FIRST_ACL_DATE DATE,
    FIRST_LINE_DATETIME TIMESTAMP,
    FIRST_REVIEW_DATETIME TIMESTAMP,
    FIRST_PROCESSING STR_20,
    FIRST_STATUS_DATETIME TIMESTAMP,
    FIRST_PROCESSING2 STR_20,
    FIRST_STATUS STR_100,
    FIRST_STATUS_REASON STR_255,
    FIRST_STATUS_MEMO VARCHAR(2048),
    FIRST_USER_FIO STR_100,
    LAST_ACL_DATE DATE,
    LAST_LINE_DATETIME TIMESTAMP,
    LAST_REVIEW_DATETIME TIMESTAMP,
    LAST_PROCESSING STR_20,
    LAST_STATUS_DATETIME TIMESTAMP,
    LAST_PROCESSING2 STR_20,
    LAST_STATUS STR_100,
    LAST_STATUS_REASON STR_255,
    LAST_STATUS_MEMO VARCHAR(2048),
    LAST_USER_FIO STR_100,
    SUM_TIME_PROCESSING STR_20,
    INFO_FOR_MNM_FROM_KK VARCHAR(2048),
    INFO_STOP_FACTOR VARCHAR(2048),
    PLAN_DISB_DATE DATE,
    FACT_DISB_DATE DATE,
    KF_SVZ STR_100,
    OFFICER_NAME STR_100,
    USER_SEND_TO_KK STR_100,
    K_IS_ONLINE VARCHAR(50),
    DISB_TYPE VARCHAR(50),
    FIRST_REFUSED INTEGER,
    LAST_REFUSED D_INTEGER,
    FIRST_COMMENT_CODE D_STR_10,
    LAST_COMMENT_CODE D_STR_10,
    OTHER_COMMENT_CODE STR_100)
AS
declare variable opr_id id_type;
declare variable sertifikator str_255;
declare variable db_id integer;
declare variable rd_id d_integer;
declare variable robot_name d_str_100;
begin
    db_id = f_get_db_id();
    robot_name = (select un_fio from username where un_id = 1000000000001);
    
    for select res_acl_id, sertifikator, res_acl_date, res_status, k_kontract_no, kf_name, res_reg_name, res_reg_namep,
               k_circle, cl_ball, product_name, k_amount, cl_fio_all, k_disb_date, k_fact_disb_date, officer_name,
               k_is_online, k_is_online1, res_memo, stop_fact
    from aarep_underwriter_consolidated(:rep_region_id, :rep_date_begin, :rep_date_end)
    into acl_id, sertifikator, first_acl_date, last_status, k_kontract_no, kf_name, reg_name, reg_namep,
         k_circle, cl_ball, pr_name, k_amount, cl_fio, plan_disb_date, fact_disb_date, officer_name,
         k_is_online, disb_type, info_for_mnm_from_kk, info_stop_factor
    do
    begin
      if (:db_id in (39801, 41701) and
           sertifikator = 'Виртуальный сотрудник') then
          continue;
      
        select acl_object_id
        from actionlist
        where acl_id = :acl_id                      
        into k_id;

        last_refused = 0;
        first_refused = 0;
        opr_id = null;
        last_acl_date = null;
        last_status_datetime = null;
        last_user_fio = '';

        select rd_id
        from hp_get_check_loan_data(:k_id)
        into:rd_id;


        select opr_id, min(ovs_datetime), un_fio /*ovs_date_begin, ovs_datetime, un_fio, case ovs_value when -1 then 1
                                                                            when -2 then 2
                                                                            else 0
                                                             end    -- последний статус в КК  */
        from objectprops
        inner join objectvalues on ovs_from_opr_id = opr_id
        inner join username on un_id = ovs_from_un_id
        where opr_object_kind = 9
        and opr_object_id = :k_id
        and ovs_is_active = 1
        and ovs_value = 0 -- в очереди
        and opr_subject_kind = -1
        and opr_subject_id = :rd_id
        and cast(ovs_datetime as date) between :rep_date_begin and :rep_date_end
        group by 1,3
        order by 2
        rows 1
        into opr_id, first_line_datetime, user_send_to_kk;


        select ovs_date_begin, ovs_datetime, un_fio, case ovs_value when -1 then 1
                                                                            when -2 then 2
                                                                            else 0
                                                     end    -- последний статус в КК--min(ovs_datetime), un_fio
        from objectvalues
        inner join username on un_id = ovs_from_un_id
        where ovs_from_opr_id = :opr_id
        and ovs_is_active = 1
        and cast(ovs_datetime as date) between :rep_date_begin and :rep_date_end
        and exists(select has_access
                   from hp_permission_access_check(ovs_from_un_id, 'PermActions', 0, 'FinStart.RepUnderConsVisible')
                   where has_access = 1)
        order by 2 desc
        rows 1
        into last_acl_date, last_status_datetime, last_user_fio, last_refused;

        last_line_datetime = null;

        select max(ovs_datetime)
        from objectvalues
        where ovs_from_opr_id =  :opr_id
        and ovs_value = 0 -- в очереди
        and ovs_is_active = 1
        and cast(ovs_datetime as date) between :rep_date_begin and :rep_date_end
        order by 1 desc
        rows 1
        into last_line_datetime;

        first_review_datetime = null;

        select min(ovs_datetime)
        from objectvalues
        where ovs_from_opr_id =  :opr_id
        and ovs_value = 1 -- в обработке
        and ovs_is_active = 1
        and cast(ovs_datetime as date) between :rep_date_begin and :rep_date_end
        order by 1
        rows 1
        into first_review_datetime;

        last_review_datetime = null;

        select max(ovs_datetime)
        from objectvalues
        where ovs_from_opr_id =  :opr_id
        and ovs_value = 1 -- в обработке
        and ovs_is_active = 1
        and cast(ovs_datetime as date) between :rep_date_begin and :rep_date_end
        order by 1 desc
        rows 1
        into last_review_datetime;

        first_status_datetime = null;
        first_user_fio = '';
        first_status = '';

        select min(ovs_datetime), un_fio, case ovs_value when -1 then 'Доработка'
                                                         when -2 then 'Отказан'
                                                         when 2 then 'Одобрен'
                                          end,
                                          case ovs_value when -1 then 1
                                                         when -2 then 2
                                                         else 0
                                          end
        from objectvalues
        inner join username on un_id = ovs_from_un_id
        where ovs_from_opr_id =  :opr_id
        and ovs_is_active = 1
        and ovs_value in (-1, -2, 2) -- первичное присвоение статуса
        and cast(ovs_datetime as date) between :rep_date_begin and :rep_date_end
        and exists(select has_access
                   from hp_permission_access_check(ovs_from_un_id, 'PermActions', 0, 'FinStart.RepUnderConsVisible')
                   where has_access = 1)
        group by 2,3,4
        order by 1
        rows 1
        into first_status_datetime, first_user_fio, first_status, first_refused;

        if (first_status = '') then
            last_status = '';

        first_status_memo = '';
        first_status_reason = '';
        last_status_reason = '';
        last_status_memo = '';
        last_comment_code = '';
        first_comment_code = '';
        other_comment_code = '';

        if (first_refused in (1, 2)) then
        begin
            select f_get_verify_memo(obm_memo, 'reason'), trim(leading ascii_char(13)|| ascii_char(10) from f_get_verify_memo(obm_memo, 'memo')), f_get_verify_memo(obm_memo, 'code')
            from objectmemo
            where obm_from_opr_id = :opr_id
            and obm_memo not containing ('Для сведения:')
            and cast(obm_datetime as date) between :rep_date_begin and :rep_date_end
            order by obm_id
            rows 1
            into first_status_reason, first_status_memo, first_comment_code;
        end

        if (last_refused in (1,2)) then
        begin
            select f_get_verify_memo(obm_memo, 'reason'), trim(leading ascii_char(13)|| ascii_char(10) from f_get_verify_memo(obm_memo, 'memo')), f_get_verify_memo(obm_memo, 'code')
          from objectmemo
          where obm_from_opr_id = :opr_id
          and obm_memo not containing ('Для сведения:')
          and cast(obm_datetime as date) between :rep_date_begin and :rep_date_end
          order by obm_id desc
          rows 1
            into last_status_reason, last_status_memo, last_comment_code;
        end

        select coalesce(list(distinct(f_get_verify_memo(obm_memo, 'code'))), '')
        from objectmemo
        where obm_from_opr_id = :opr_id
        and iif(f_get_verify_memo(obm_memo, 'code') <> '', left(f_get_verify_memo(obm_memo, 'code'), position('.' in f_get_verify_memo(obm_memo, 'code'))) in('7.', '8.', '10.'), null)
        and cast(obm_datetime as date) between :rep_date_begin and :rep_date_end
        into other_comment_code;

        select c_name_fio
        from knaktlist
        inner join kontract on k_from_kf_id = kal_from_kf_id
        inner join clients on c_id = kal_new_kf_officer
        where k_id = :k_id
        and kal_date_end = '31.12.9999'
        and is_delete = 0
        into kf_svz;

        first_processing = f_get_time(first_review_datetime, first_status_datetime);
        last_processing = f_get_time(last_review_datetime, last_status_datetime);
        sum_time_processing = f_get_time(first_line_datetime, last_status_datetime);
        first_processing2 = f_get_time(first_line_datetime, first_status_datetime);
        last_processing2 = f_get_time(last_line_datetime, last_status_datetime);

        if (db_id = 81001 and rd_id = 9978 and last_user_fio = '') then
        begin
          first_user_fio = robot_name;
          last_user_fio = robot_name;
        end

        suspend;
    end
end
