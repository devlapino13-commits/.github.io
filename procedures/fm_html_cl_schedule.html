CREATE PROCEDURE FM_HTML_CL_SCHEDULE(
    C_ID DECIMAL(18,0),
    DATE_END DATE)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable kont_no varchar(255);
declare variable kont_no_prior varchar(255);
declare variable str varchar(1024);
declare variable db_id integer;
declare variable k_id d_id;
declare variable round_value d_smallint;
begin
  :db_id = f_get_db_id();
  round_value = iif(db_id = 94401, 2, 0);

  if (date_end is null) then
    date_end = current_date;

  select res_text
  from fm_html_client_zadol(:c_id, :date_end, 1)
  into res_text;

  res_text = BLOB_APPEND(res_text, f_format('<h1 class="green_text">%1</h1>', iif(db_id <> 94401, 'Актуальный график погашений', 'Cari ödəniş cədvəli')));

  kont_no_prior = '';

  for select k_id,
             k_kontract_no,
             '<tr>' || '<td align="center">' || res_id || '<br>' || right(100 + extract(day from res_date), 2) || '.' || right(100 + extract(month from res_date), 2) || '.' || extract(year from res_date) || '</td>' || '<td align="right">' || round(res_gr_os, :round_value) || '<br>' || round(res_cl_os, :round_value) || '</td>' || '<td align="right">' || round(res_gr_perc, :round_value) || '<br>' || round(res_cl_perc, :round_value) || '</td>' || '<td align="right">' || round(res_gr_fine, :round_value) || '<br>' || round(res_cl_fine, :round_value) || '</td>' || '<td align="right">' || round(res_gr_plan_amount, :round_value) || '<br>' || round(res_cl_plan_amount, :round_value) || '</td>' || '</tr>'
      from aarep_get_client_schedule(:c_id, :date_end)
      order by k_id, k_kontract_date, res_date, res_id
      into k_id, kont_no, :str
  do
  begin
    if (kont_no <> kont_no_prior) then
    begin
      if (kont_no_prior <> '') then
      begin
        res_text = BLOB_APPEND(res_text, '</table>');
        res_text = BLOB_APPEND(res_text, '<br>');
      end
      res_text = BLOB_APPEND(res_text, f_format('%1№ %2<br><br>', iif(db_id <> 94401, 'Дог.', 'Müq.'), kont_no));

      res_text = BLOB_APPEND(res_text, '<table width="100%">
           <col width="20%">
           <col width="20%">
           <col width="20%">
           <col width="20%">
           <col width="20%">
           <tr>' || f_format('<th>№<br>%1</th>', iif(db_id <> 94401, 'Дата', 'Tarix')) ||
                    f_format('<th>%1<br>%2</th>', iif(db_id <> 94401, 'ОС гр.', 'Qrupun əsas məbl.'),
                                                  iif(db_id <> 94401, 'ОС кл.', 'Müştərinin əsas məbl')) ||
                    f_format('<th>%% %1<br>%% %2</th>', iif(db_id <> 94401, 'гр.', 'Qrupun'),
                                                        iif(db_id <> 94401, 'кл.', 'Müşt.')) ||
                    f_format('<th>%1<br>%2</th>', iif(db_id <> 94401, 'Пеня гр.', 'Qrupun cəriməsi'),
                                                  iif(db_id <> 94401, 'Пеня кл.', 'Müşt. cəriməsi')) ||
                    f_format('<th>%1<br>%2</th>', iif(db_id <> 94401, 'Всего гр.', 'Ümumi qrup'),
                                                  iif(db_id <> 94401, 'Всего кл.', 'Ümumi müşt.')) || '</tr>');
      kont_no_prior = kont_no;
      --BOARD-5114 распечатка графика погашения
      if (:db_id in (39801, 41701)) then
        res_text = BLOB_APPEND(res_text, f_uri_blank('Распечатать график погашения', 'print', f_objkind('kontract', 'id'), :k_id, 1711, 'Печать графика погашения', 0, 0, '') || '<br><br>');
      --BOARD-5114 END
    end
    res_text = BLOB_APPEND(res_text, str);
  end
  res_text = BLOB_APPEND(res_text, '</table>');
  res_text = BLOB_APPEND(res_text, '</body>');
  suspend;
end
