CREATE PROCEDURE UXP$FINSTART_MENULIST
RETURNS (
    ML_ID D_INTEGER,
    ML_PARENT_ID D_INTEGER,
    ML_CODE D_STR_100,
    ML_PARENT_CODE D_STR_100,
    ML_CAPTION D_STR_255,
    ML_DESCRIPTION D_STR_LONG,
    ML_INDEX D_INTEGER,
    ML_IS_REPORT D_INTEGER,
    ML_SECURE_KIND D_INTEGER,
    ML_KIND D_INTEGER)
AS
begin
  for select ml_id, ml_parent_id, ml_code, ml_parent_code, ml_caption,
             ml_description, ml_index, ml_is_report, ml_secure_kind, ml_kind
  from
      menulist
  where ml_id < 0 and
        ml_kind = 0
  into ml_id, ml_parent_id, ml_code, ml_parent_code, ml_caption,
       ml_description, ml_index, ml_is_report, ml_secure_kind, ml_kind
  do
    suspend;
end
