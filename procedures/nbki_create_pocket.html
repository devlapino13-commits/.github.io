CREATE PROCEDURE NBKI_CREATE_POCKET
RETURNS (
    POCKET_ID INTEGER,
    POCKET_NAME STR_100)
AS
declare variable cib_id integer;
declare variable y integer;
declare variable m_int integer;
declare variable m_str varchar(10);
declare variable d_int integer;
declare variable d_str varchar(10);
declare variable h_int integer;
declare variable h_str varchar(10);
declare variable s_int integer;
declare variable s_str varchar(10);
declare variable min_int integer;
declare variable min_str varchar(10);
begin
  -- Пакет в НБКИ формируется на текущую дату
  cib_id = f_get_rd_id('CIB.NBKI');

  --[14:54, 24.4.2019] Антон Ицкович: P701NN000001_20190420_150653
  --[14:55, 24.4.2019] Антон Ицкович: P701NN000001_YYYYMMDD_HHMMSS

  y = extract(year from current_timestamp);

  m_int = extract(month from current_timestamp);
  m_str = iif(m_int<10, '0'||m_int, m_int);

  d_int = extract(day from current_timestamp);
  d_str = iif(d_int<10, '0'||d_int, d_int);

  h_int = extract(hour from current_timestamp);
  h_str = iif(h_int<10, '0'||h_int, h_int);

  min_int = extract(minute from current_timestamp);
  min_str = iif(min_int<10, '0'||min_int, min_int);

  s_int = extract(second from current_timestamp);
  if (s_int = 60) then s_int = s_int - 1;
  s_str = iif(s_int<10, '0'||s_int, s_int);

  pocket_name = 'P701NN000001_'||y||m_str||d_str||'_'||h_str||min_str||s_str;

  insert into cib_pocket (cp_counter, cp_date_begin, cp_date_end, cp_pocket_name, cp_pocket_type, cp_pocket_status,
                          cp_pocket_kib_status, cp_cib_id)
  values (0, null, current_date, :pocket_name, '.d.kont.', 0, null, :cib_id)
  returning cp_id
  into :pocket_id;

  suspend;
end
