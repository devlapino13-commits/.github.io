CREATE PROCEDURE AAREP_CHANGED_PERS_DATA(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    OFFICE STR_100,
    DOC_DATETIME D_DATETIME,
    EDIT_DATETIME D_DATETIME,
    CL_FIO STR_100,
    CATEGORY STR_20,
    CTZN_CHNG STR_20,
    ACTUAL_DATA STR_255,
    OLD_DATA STR_255,
    LO_FIO STR_100,
    DOC_FIO STR_100,
    DOC_JOB_NAME STR_255,
    EDITOR_FIO STR_100,
    EDITOR_JOB_NAME STR_255,
    CLIENT_STATUS D_STR_20)
AS
declare variable C_ID ID_TYPE;
declare variable DOC_CR_UN_ID ID_TYPE;
declare variable EDITOR_UN_ID ID_TYPE;
declare variable PK_COUNTRY_ID D_INTEGER;
declare variable CDS_DOC_NO STR_100;
declare variable CDS_DOC_DATE_BEGIN D_DATE;
begin

  select reg_name
  from region
  where reg_unicode = :rep_region_id
  into office;
  -- проверка замены номеров телефона
  category = 'номер';
  for select edl_create_datetime, edl_object_id, c_name_fio, edl_create_un_id
      from edoc_list
      inner join miniapp on ma_id = (select max(mac_from_ma_id)
                                     from miniappclient
                                     where mac_from_c_id = edl_object_id)
      inner join knaktlist on kal_from_kf_id = ma_from_kf_id
      inner join clients on c_id = kal_new_kf_officer
      where edl_from_edc_id = 9 and
            edl_status = 1 and
            cast(edl_create_datetime as date) between :rep_date_begin and :rep_date_end and
            kal_new_reg_id = :rep_region_id and
            ma_date_end = '31.12.9999' and
            is_delete = 0 and
            kal_date_end = '31.12.9999'
      into doc_datetime, c_id, lo_fio, doc_cr_un_id
  do
  begin
    actual_data = '';
    old_data = '';
    cl_fio = '';
    edit_datetime = null;
    editor_fio = '';
    editor_job_name = '';
    doc_fio = '';
    doc_job_name = '';
    editor_un_id = null;

    select c_name_fio
    from clients
    where c_id = :c_id
    into cl_fio;

    select clt_code, clg_from_un_id, clg_datetime
    from contdata
    inner join contlog on clg_from_cda_id = cda_id
    inner join contlist on clt_id = cda_from_clt_id
    where cda_from_c_id = :c_id and
          clg_is_main = 1 and
          clg_is_delete = 0 and
          clg_datetime >= :doc_datetime
    order by clg_id rows 1
    into actual_data, editor_un_id, edit_datetime;

    select coalesce(clt_code, '')
    from contdata
    inner join contlist on clt_id = cda_from_clt_id
    inner join contlog on clg_from_cda_id = cda_id
    where cda_from_c_id = :c_id and
          clg_is_main = 1 and
          clg_is_delete = 1 and
          clt_id <> 0 and
          clg_datetime >= :doc_datetime and
          coalesce(clt_code, '') <> :actual_data
    order by clg_id rows 1
    into old_data;

    if (old_data = '') then
    begin
      select coalesce(clt_code, '')
      from contdata
      inner join contlist on clt_id = cda_from_clt_id
      inner join contlog on clg_from_cda_id = cda_id
      where cda_from_c_id = :c_id and
            clg_is_main = 1 and
            clg_is_delete = 1 and
            clt_id <> 0 and
            clg_datetime <= :doc_datetime and
            coalesce(clt_code, '') <> :actual_data
      order by clg_id desc rows 1
      into old_data;
    end

    select un_fio, list(rn_name || ' ' || coalesce('(' || rn_memo || ')', ''), ', ')
    from username
    inner join permissions on pm_object_kind = 0 and
          pm_object_id = un_id
    inner join rolename on pm_subject_kind = 1 and
          pm_subject_id = rn_id
    where un_id = :doc_cr_un_id and
          pm_is_delete = 0 and
          iif(pm_date_end is not null, pm_date_end >= '31.12.2999', pm_date_end is null)
    group by 1
    into doc_fio, doc_job_name;

    select un_fio, list(rn_name || ' ' || coalesce('(' || rn_memo || ')', ''), ', ')
    from username
    inner join permissions on pm_object_kind = 0 and
          pm_object_id = un_id
    inner join rolename on pm_subject_kind = 1 and
          pm_subject_id = rn_id
    where un_id = :editor_un_id and
          pm_is_delete = 0 and
          iif(pm_date_end is not null, pm_date_end >= '31.12.2999', pm_date_end is null)
    group by 1
    into editor_fio, editor_job_name;

    -- RFCRU-4497
    if (exists(select *
               from oksummary oks
               inner join knclient kc on kc.kc_from_k_id = oks.oks_from_k_id
               where kc.kc_from_c_id = :c_id and
                     kc.kc_amount_approved > 0 and
                     current_date between oks.oks_date_begin and oks.oks_date_end and
                     oks.oks_close_state = 0 and
                     current_date - oks.oks_delinq_date > 0)) then
        client_status = 'Просрочен';
    else
        client_status = 'Не просрочен';

    suspend;
  end
  -- проверка замены паспортных данных
  category = 'паспортные данные';
  for select cds_from_c_id, cds_create_datetime, cds_create_datetime, cds_doc_no, cds_doc_date_begin, c_name_fio,
             cds_create_un_id, pk_country_id
      from clientdocs
      inner join passkind on pk_id = cds_from_pk_id
      inner join miniapp on ma_id = (select max(mac_from_ma_id)
                                     from miniappclient
                                     where mac_from_c_id = cds_from_c_id)
      inner join knaktlist on kal_from_kf_id = ma_from_kf_id
      inner join clients on c_id = kal_new_kf_officer
      where cast(cds_create_datetime as date) between :rep_date_begin and :rep_date_end and
            cds_is_delete = 0 and
            pk_id <> 0 and
            cds_is_main_doc = 1 and
            kal_new_reg_id = :rep_region_id and
            ma_date_end = '31.12.9999' and
            cds_verify_status = 2 and
            is_delete = 0 and
            kal_date_end = '31.12.9999'
      into c_id, doc_datetime, edit_datetime, cds_doc_no, cds_doc_date_begin, lo_fio, doc_cr_un_id, pk_country_id
  do
  begin
    old_data = '';
    cl_fio = '';
    editor_fio = '';
    editor_job_name = '';
    doc_fio = '';
    doc_job_name = '';
    editor_un_id = null;

    actual_data = cds_doc_no || ' выдан: ' || f_datetostr(cds_doc_date_begin);

    select c_name_fio
    from clients
    where c_id = :c_id
    into cl_fio;

    select coalesce(cds_doc_no || ' выдан: ' || f_datetostr(cds_doc_date_begin), ''),
           iif(pk_country_id <> :pk_country_id, 'смена гражданства', '')
    from clientdocs
    inner join passkind on pk_id = cds_from_pk_id
    where cds_from_c_id = :c_id and
          cds_is_delete = 1 and
          cds_is_main_doc = 1 and
          --cds_verify_status = 2 and
          cds_create_datetime <= :doc_datetime - 2 and
          coalesce(cds_doc_no, '') <> '' and
          cds_from_pk_id <> 0 and
          replace(cds_doc_no, ' ', '') <> replace(:cds_doc_no, ' ', '')
    order by cds_create_datetime desc rows 1
    into old_data, ctzn_chng;

    select un_fio, list(rn_name || ' ' || coalesce('(' || rn_memo || ')', ''), ', ')
    from username
    inner join permissions on pm_object_kind = 0 and
          pm_object_id = un_id
    inner join rolename on pm_subject_kind = 1 and
          pm_subject_id = rn_id
    where un_id = :doc_cr_un_id and
          pm_is_delete = 0 and
          iif(pm_date_end is not null, pm_date_end >= '31.12.2999', pm_date_end is null)
    group by 1
    into doc_fio, doc_job_name;

    editor_fio = doc_fio;
    editor_job_name = doc_job_name;

    -- RFCRU-4497
    if (exists(select *
               from oksummary oks
               inner join knclient kc on kc.kc_from_k_id = oks.oks_from_k_id
               where kc.kc_from_c_id = :c_id and
                     kc.kc_amount_approved > 0 and
                     current_date between oks.oks_date_begin and oks.oks_date_end and
                     oks.oks_close_state = 0 and
                     current_date - oks.oks_delinq_date > 0)) then
        client_status = 'Просрочен';
    else
        client_status = 'Не просрочен';

    if (old_data <> '') then
      suspend;
  end
  -- проверка замены фио
  ctzn_chng = '';
  category = 'фио';
  for select cds_from_c_id, cds_create_datetime, c_name_fio, cds_create_un_id
      from clientdocs
      inner join miniapp on ma_id = (select max(mac_from_ma_id)
                                     from miniappclient
                                     where mac_from_c_id = cds_from_c_id)
      inner join knaktlist on kal_from_kf_id = ma_from_kf_id
      inner join clients on c_id = kal_new_kf_officer
      where cast(cds_create_datetime as date) between :rep_date_begin and :rep_date_end and
            cds_is_delete = 0 and
            cds_from_pk_id in (56, 57, 66) and
            kal_new_reg_id = :rep_region_id and
            ma_date_end = '31.12.9999' and
            is_delete = 0 and
            kal_date_end = '31.12.9999'
      into :c_id, doc_datetime, lo_fio, doc_cr_un_id
  do
  begin
    actual_data = '';
    old_data = '';
    cl_fio = '';
    edit_datetime = null;
    editor_fio = '';
    editor_job_name = '';
    doc_fio = '';
    doc_job_name = '';
    editor_un_id = null;

    select c_name_fio, c_name_fio
    from clients
    where c_id = :c_id
    into cl_fio, actual_data;

    for select cll_name_fio, cll_curr_datetime, cll_curr_user_id
        from clientlog
        where cll_from_c_id = :c_id and
              cll_name_fio <> :actual_data and
              cll_curr_datetime >= :doc_datetime - 7 and
              cll_fraud_fio = 0
        order by cll_curr_datetime desc rows 1
        into old_data, edit_datetime, editor_un_id

    do
    begin
      select un_fio, list(rn_name || ' ' || coalesce('(' || rn_memo || ')', ''), ', ')
      from username
      inner join permissions on pm_object_kind = 0 and
            pm_object_id = un_id
      inner join rolename on pm_subject_kind = 1 and
            pm_subject_id = rn_id
      where un_id = :doc_cr_un_id and
            pm_is_delete = 0 and
            iif(pm_date_end is not null, pm_date_end >= '31.12.2999', pm_date_end is null)
      group by 1
      into doc_fio, doc_job_name;

      select un_fio, list(rn_name || ' ' || coalesce('(' || rn_memo || ')', ''), ', ')
      from username
      inner join permissions on pm_object_kind = 0 and
            pm_object_id = un_id
      inner join rolename on pm_subject_kind = 1 and
            pm_subject_id = rn_id
      where un_id = :editor_un_id and
            pm_is_delete = 0 and
            iif(pm_date_end is not null, pm_date_end >= '31.12.2999', pm_date_end is null)
      group by 1
      into editor_fio, editor_job_name;

      -- RFCRU-4497
      if (exists(select *
                 from oksummary oks
                 inner join knclient kc on kc.kc_from_k_id = oks.oks_from_k_id
                 where kc.kc_from_c_id = :c_id and
                       kc.kc_amount_approved > 0 and
                       current_date between oks.oks_date_begin and oks.oks_date_end and
                       oks.oks_close_state = 0 and
                       current_date - oks.oks_delinq_date > 0)) then
          client_status = 'Просрочен';
      else
          client_status = 'Не просрочен';

      suspend;
    end
  end
end
