CREATE PROCEDURE HP_SURVEY_DTCHANGES_LST(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    RD_ID D_INTEGER)
RETURNS (
    DT_LST D_STR_8100,
    DT_MAX D_DATE,
    DT_MIN D_DATE)
AS
declare variable rd_code d_str_255;
declare variable delim d_str_lang;
begin
  rd_code = f_get_rd_code(:rd_id);
  delim = ascii_char(13) || ascii_char(10);

  if (rd_code = 'Anketa.Main') then
    select
      coalesce(list(f_datetostr(ovs_date_begin), :delim), ''),
      max(ovs_date_begin),
      min(ovs_date_begin)
    from (
      select
        ovs_date_begin
      from actiongroup
        inner join objectprops
          on opr_object_kind = :obj_kind
            and opr_object_id = :obj_id
            and opr_subject_kind = -1
            and opr_subject_id = acg_from_rd_id
        inner join objectvalues
          on ovs_from_opr_id = opr_id
            and ovs_date_end >= ovs_date_begin
            and ovs_is_active = 1
      where acg_group_rd_id = :rd_id
        and acg_parent_id is null
        and acg_is_delete = 0
      group by ovs_date_begin
      order by ovs_date_begin desc)
    into
      :dt_lst,
      :dt_max,
      :dt_min;
  else
  if (rd_code = 'Anketa.TargetUseLoan') then
    select
      coalesce(list(f_datetostr(dt), :delim), ''),
      max(dt),
      min(dt)
    from (
      select
        iif(k_disb_date < '14.02.2022',
          k_disb_date + iif(k_disb_date >= '22.12.2021' and is_ipf = 1, 1, 8), k_disb_date + 1) as dt
      from hp_survey_view_kont(:obj_id)
      group by 1
      order by 1 desc)
    into
      :dt_lst,
      :dt_max,
      :dt_min;
  else
  if (rd_code = 'Anketa.Delinq.1') then
    select
      coalesce(list(f_datetostr(dt), :delim), ''),
      max(dt),
      min(dt)
    from (
      select
        cast(acl_last_datetime as d_date) as dt
      from actionlist
      where acl_object_kind = :obj_kind
        and acl_object_id = :obj_id
        and acl_from_rd_id = :rd_id
        and acl_status = 2
      group by 1
      order by 1 desc)
    into
      :dt_lst,
      :dt_max,
      :dt_min;

  suspend;
end
