CREATE PROCEDURE HP_CHECK_INFO_HTML_GET(
    C_ID ID_TYPE)
RETURNS (
    COMMENT_CHECK_INFO STR_255,
    CR_LEVEL INTEGER,
    ACL_STATUS_CHECK_INFO INTEGER)
AS
declare variable acl_id_check_info id_type;
begin
  acl_id_check_info = 0;
  acl_status_check_info = 0;

  select al.acl_id, al.acl_status
  from actionlist al
  where al.acl_object_kind = 0 and
        al.acl_object_id = :c_id and
        al.acl_from_rd_id in (f_get_rd_id('ActionGroup.Check_Info'),
                              f_get_rd_id('ActionGroup.Check_Info_Error'))
  order by al.acl_id desc
  rows 1
  into :acl_id_check_info, :acl_status_check_info;

  if (acl_id_check_info > 0) then
  begin
    for select acl_comment, cr_level
        from hp_actionlist_memo_info(:acl_id_check_info)
        into :comment_check_info, :cr_level
    do
    begin
      suspend;
    end
  end
end
