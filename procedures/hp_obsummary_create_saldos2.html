CREATE PROCEDURE HP_OBSUMMARY_CREATE_SALDOS2
RETURNS (
    CHANGED_RECORDS INTEGER)
AS
declare variable al_id_prior varchar(20);
declare variable saldo_ekv decimal(18,4);
declare variable saldo_val decimal(18,4);
declare variable obs_from_al_id varchar(20);
declare variable obs_date date;
declare variable obs_dt_ekv decimal(18,4);
declare variable obs_kt_ekv decimal(18,4);
declare variable obs_dt_val decimal(18,4);
declare variable obs_kt_val decimal(18,4);
declare variable obs_saldo_ekv decimal(18,4);
declare variable obs_saldo_val decimal(18,4);
begin
-- если остатки лицевых счетов не совпадает с реальным остатков,
-- то исправляет остатки по лицевым счетам на каждый опер.день
  saldo_ekv = 0;
  saldo_val = 0;
  al_id_prior = '';
  changed_records = 0;

  for select obs_from_al_id, obs_date, obs_dt_ekv, obs_kt_ekv, obs_dt_val, obs_kt_val, obs_saldo_ekv, obs_saldo_val
      from obsummary
      order by obs_from_al_id, obs_date
      into :obs_from_al_id, :obs_date, :obs_dt_ekv, :obs_kt_ekv, :obs_dt_val, :obs_kt_val, :obs_saldo_ekv, :obs_saldo_val
--  as cursor cur
  do
  begin
    if (obs_saldo_ekv is null) then
      obs_saldo_ekv = 0;
    if (obs_saldo_val is null) then
      obs_saldo_val = 0;

    if (obs_from_al_id = al_id_prior) then
    begin
      saldo_ekv = saldo_ekv + obs_dt_ekv - obs_kt_ekv;
      saldo_val = saldo_val + obs_dt_val - obs_kt_val;
    end
    else
    begin
      saldo_ekv = obs_dt_ekv - obs_kt_ekv;
      saldo_val = obs_dt_val - obs_kt_val;
    end
    al_id_prior = obs_from_al_id;
    if ((obs_saldo_ekv <> saldo_ekv) or (obs_saldo_val <> saldo_val)) then
    begin
      changed_records = changed_records + 1;
      update obsummary
      set obs_saldo_ekv = :saldo_ekv,
          obs_saldo_val = :saldo_val
      where obs_from_al_id = :obs_from_al_id and
            obs_date = :obs_date;
--        where current of cur;
    end
  end
  suspend;
end
