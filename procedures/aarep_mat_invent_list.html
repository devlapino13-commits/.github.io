CREATE PROCEDURE AAREP_MAT_INVENT_LIST(
    WORKPLACE_ID DECIMAL(18,0),
    REPORT_DATE DATE)
RETURNS (
    MBAL_WP_NAME VARCHAR(100),
    MBAL_MO_NAME VARCHAR(100),
    MBAL_ML_NAME VARCHAR(255),
    MBAL_KOLICH DECIMAL(18,4),
    MBAL_SUMMA DECIMAL(18,4),
    MBAL_MC_NAME VARCHAR(100),
    MBAL_INVENT_STR VARCHAR(50),
    MBAL_FULL_DETAIL VARCHAR(2048),
    MBAL_ACCOUNT VARCHAR(20),
    MBAL_IS_SKLAD_STR VARCHAR(20),
    MBAL_REGION_NAME VARCHAR(100),
    MBAL_ML_MODEL VARCHAR(255),
    MBAL_MC_STATUS_STR VARCHAR(20),
    MBAL_WP_ID DECIMAL(18,0),
    MBAL_MO_ID DECIMAL(18,0),
    MBAL_MC_STATUS INTEGER,
    MBAL_MC_ID INTEGER,
    MBAL_BILL_ID DECIMAL(18,0),
    MBAL_BILL_LINE_NO INTEGER,
    MBAL_DATE DATE,
    MBAL_AMORT_BEGIN DATE,
    OL_CLIENT VARCHAR(100),
    MBAL_ML_ID D_ID)
AS
declare variable mbal_is_sklad smallint;
declare variable st_staff integer;
declare variable st_client integer;
declare variable ol_document_client decimal(18,0);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ol_date, mbal_bill_id, mbal_bill_line_no, wp_name, c_name_fio, ml_name, ml_model, mbal_kolich, mbal_summa,
             mc_name, mc_status, mbal_invent_str, mbal_full_detail, mbal_account, mbal_is_sklad, reg_name, wp_id, c_id,
             mc_id, mbal_ml_id
      from operlist, matlist_balance_list(:workplace_id, 0, :report_date, 0, 1)
      left join(workplace left join region on (wp_from_reg_id = reg_unicode)) on (mbal_wp_id = wp_id)
      left join matcateg on (mbal_mc_id = mc_id)
      left join matlist on (mbal_ml_id = ml_id)
      left join clients on (mbal_mo_id = c_id)
      where mc_status <= 2 /* Только ОС, МБП и Нематериальные активы */
            and
            ol_id = mbal_bill_id
      into :mbal_date, :mbal_bill_id, :mbal_bill_line_no, :mbal_wp_name, :mbal_mo_name, :mbal_ml_name, :mbal_ml_model,
           :mbal_kolich, :mbal_summa, :mbal_mc_name, :mbal_mc_status, :mbal_invent_str, :mbal_full_detail,
           :mbal_account, :mbal_is_sklad, :mbal_region_name, mbal_wp_id, mbal_mo_id, :mbal_mc_id, :mbal_ml_id
  do
  begin
    if (mbal_is_sklad = 1) then
      mbal_is_sklad_str = 'На складе';
    else
      mbal_is_sklad_str = 'На пользовании';
    if (mbal_mc_status = 0) then
      mbal_mc_status_str = '1 ОС';
    if (mbal_mc_status = 1) then
      mbal_mc_status_str = '2 МБП';
    if (mbal_mc_status = 2) then
      mbal_mc_status_str = '3 Немат.актив';
    if (mbal_mc_status = 3) then
      mbal_mc_status_str = '4 Расход.матер.';
    if (mbal_region_name is not null) then
    begin
      mbal_amort_begin = null;
      select ol_date
      from operamormain, operlist
      where oam_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            oam_bill_id = :mbal_bill_id and
            oam_bill_line_no = :mbal_bill_line_no
      order by ol_date
      rows 1
      into :mbal_amort_begin;
      ol_client = '';
      select c_name_fio
      from operlist, clients
      where ol_id = :mbal_bill_id and
            ol_document_client = c_id
      into :ol_client;
      if (ol_client is null) then
        ol_client = '';
      suspend;
    end
  end
end
