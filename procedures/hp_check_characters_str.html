CREATE PROCEDURE HP_CHECK_CHARACTERS_STR(
    IN_STRING D_STR_8100,
    CHECK_MASK D_STR_8100 = '')
RETURNS (
    ERROR_TXT D_STR_8100,
    OUT_STRING D_STR_8100)
AS
declare variable counter integer;
declare variable len integer;
declare variable str varchar(1);
begin
  error_txt = '';
  out_string = '';

  if (check_mask <> '') then
  begin
      if (in_string not similar to check_mask escape '@') then
      begin
        len = character_length(in_string);
        counter = 1;
        while (counter <= len) do
        begin
            str = substring(in_string from counter for 1);
            if (str similar to check_mask escape '@')
              then out_string = out_string || str;
                else error_txt = error_txt || '(' || str || '); ';
            counter = counter + 1;
        end
      end
  end

  if (error_txt <> '') then
  begin
    error_txt = 'В строке ('||in_string||') недопустимые символы: '||error_txt;
    suspend;
  end
end
