CREATE PROCEDURE CM_RU_0297_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable k_disb_date d_date;
declare variable kc_amount_approved d_currency;
declare variable k_kolich_lgota d_integer;
declare variable pr_code d_str_100;
declare variable min_amount d_currency;
declare variable pr_name d_str_code;
begin
  for select kc.kc_from_c_id, kc_amount_approved, k.k_kolich_lgota, pr.pr_code, pr.pr_name
      from kontract k
      inner join knclient kc on kc.kc_from_k_id = k.k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id
      where k.k_id = :k_id
        and kc.kc_amount_approved > 0
        and pr.pr_code containing '.easy_start'
      into :c_id, kc_amount_approved, k_kolich_lgota, pr_code, pr_name
  do
  begin
    min_amount = iif(pr_code containing '.easy_start.', 30000, 15000);
    if (:kc_amount_approved < min_amount or :kc_amount_approved > 50000) then
    begin
      res_error_text = f_format('Минимальная сумма по продукту «%1» - %2, максимальная - 50000 рублей', :pr_name, :min_amount);
      res_error_is_critical = 2;
      suspend;
    end

    if (:k_kolich_lgota > 0) then
    begin
      res_error_text = f_format('Льготный период не предоставляется по продукту «%1»', :pr_name);
      res_error_is_critical = 2;
      suspend;
    end

    --заем может быть выдан только новым клиентам и только один раз
    if (exists
       (select 1
       from knclient, kontstate
       where kc_from_c_id = :c_id
         and kc_amount_approved > 0
         and kc_from_k_id = ks_from_k_id
         and ks_ol_is_delete = 0
         and ks_state = 1)) then
    begin
      res_error_text = f_format('Продукт «%1» выдается только один раз и только новому клиенту, клиент ранее получал займ в Компании', :pr_name);
      res_error_is_critical = 2;
      suspend; 
      exit;
    end
  end

  for
  select kc.kc_from_c_id, k.k_disb_date
  from kontract k
  inner join knclient kc on kc.kc_from_k_id = k.k_id
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where k.k_id = :k_id
    and kc.kc_amount_approved > 0
    and pr.pr_code not containing '.easy_start'
  into :c_id, k_disb_date
  do
  begin
    --клиент не должен получить новый заем по любому другому продукту, пока действует льготный период по "ЛС"
    if (exists (select 1
        from knclient kc
        inner join kontstate ks on ks.ks_from_k_id = kc.kc_from_k_id
                              and current_date between ks_date_begin and ks_date_end
        inner join kontract k on k.k_id = ks.ks_from_k_id
        inner join product pr on pr.pr_id = k.k_from_pr_id
        where kc.kc_from_c_id = :c_id
          and kc.kc_amount_approved > 0
          and ks.ks_ol_is_delete = 0
          and ks.ks_state = 1
          and pr.pr_code containing '.easy_start'
          and :k_disb_date - ks.ks_date_begin <= 15)) then
    begin
      res_error_text = f_format('Клиент не может получить новый заем, пока действует льготный период по продукту "Легкий старт"');
      res_error_is_critical = 2;
      suspend;
      exit;
    end
  end
end
