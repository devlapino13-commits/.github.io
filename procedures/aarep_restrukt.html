CREATE PROCEDURE AAREP_RESTRUKT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_TYPE VARCHAR(100))
RETURNS (
    K_ID DECIMAL(18,0),
    OL_DATE DATE,
    OL_MEMO VARCHAR(1024),
    KF_NAME VARCHAR(100),
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    K_KREDLINE_END DATE,
    K_DATE_END DATE,
    KOL_PROLOG INTEGER,
    REP_ACTIVE_BALANCE DECIMAL(18,2),
    RT_KIND VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    PR_NAME VARCHAR(50),
    PR_COMMON_NAME VARCHAR(100),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    IS_PROLONGED SMALLINT,
    BALANCE_REST CURR_TYPE,
    K_AMOUNT_APPROVED CURR_TYPE,
    REG_STAT_REP_REGION VARCHAR(255),
    K_DISB_KINDD D_STR_100,
    K_BY_RATING D_STR_64,
    UUID D_STR_CODE)
AS
declare variable CURR varchar(3);
declare variable DEF_CURR varchar(3);
declare variable CURR_RATE decimal(18,4);
declare variable K_INDEX_CURR_RATE decimal(18,4);
declare variable PR_SET_RATE_IN_KONT integer;
declare variable MAX_DATE_GRAPH date;
declare variable MAX_DATE_REST date;
declare variable DB_ID integer;
declare variable REG_ID_PRIOR integer;
declare variable OR_KIND smallint;
declare variable OR_KIND_PRIOR smallint;
declare variable REG_ID integer;
declare variable DO_CONTINUE smallint;
declare variable K_DISB_KIND D_INTEGER;
begin
   k_disb_kindd = '';
  select cm_default_currency
  from company
  into def_curr;

  :db_id = f_get_db_id();
  reg_id_prior = 0;
  or_kind_prior = -1;

  for select or_from_k_id, ol_date, id_list, or_kind, ol_memo
      from operrest opr, operlist, region_list(:rep_region_id, 1, 3, 1)
      where opr.or_ol_is_active = 1 and
            opr.or_ol_date between :rep_date_begin and :rep_date_end and
            or_from_ol_id = ol_id and
            opr.or_from_reg_id = id_list
      into k_id, ol_date, reg_id, or_kind, ol_memo
  do
  begin
    do_continue = 1;

    if (rep_type = 'Restructuring') then
    begin
      if ((select is_restruct
           from aarep_get_res_for_oper(:k_id, :ol_date, :db_id)) = 0) then
        do_continue = 0;
    end

    if (do_continue = 1) then
    begin
      if (or_kind_prior <> or_kind) then
      begin
        or_kind_prior = or_kind;
        select rt_name
        from restkind
        where rt_id = :or_kind
        into rt_kind;
      end

      select kal_new_kf_name, k_kredline_end, k_kontract_no, k_kolich_client_approved, k_currency, k_index_curr_rate,
             pr_name, pr_common_name, pr_set_rate_in_kont, k_amount_approved, k_disb_kind, iif(k_by_rating = 1, 'По кредитному рейтингу', 'Стандартная выдача')
      from kontract, knaktlist, product
      where k_id = :k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :ol_date between kal_date and kal_date_end and
            k_from_pr_id = pr_id
      into kf_name, k_kredline_end, k_kontract_no, k_kolich_client_approved, curr, k_index_curr_rate, pr_name,
           pr_common_name, pr_set_rate_in_kont, k_amount_approved, k_disb_kind, k_by_rating;

      if (:db_id in (39801, 41701)) then
      begin
        k_disb_kindd = case :k_disb_kind
                         when 6 then 'На карту Мбулак'
                         when 5 then 'На карту клиента'
                         else 'Касса'
                       end;
      end

      if (reg_id_prior <> reg_id) then
      begin
        reg_id_prior = reg_id;

        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r1.reg_stat_rep_region
        from region r1, region r2
        where r1.reg_unicode = :reg_id and
              r1.parent = r2.reg_unicode
        into reg_name, parent_name, reg_stat_rep_region;
      end

      select max(iif(ok_type = 1, ok_sh_date, cast('01.01.1900' as date))),
             max(iif(ok_type = 2, ok_sh_date, cast('01.01.1900' as date)))
      from operkred
      where ok_ol_is_active = 1 and
            ok_ol_date <= :rep_date_end and
            ok_from_k_id = :k_id and
            ok_type in (1, 2) and
            ok_from_c_id = 0
      into max_date_graph, max_date_rest;

      --      k_kredline_end = max_date_graph;

      if (max_date_rest > max_date_graph) then
      begin
        is_prolonged = 1;
        k_date_end = max_date_rest;
      end
      else
      begin
        is_prolonged = 0;
        k_date_end = max_date_graph;
      end

      balance_rest = null;
      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            :ol_date between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into balance_rest;
      if (balance_rest is null) then
        balance_rest = 0;

      rep_active_balance = 0;
      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into rep_active_balance;
      if (rep_active_balance is null) then
        rep_active_balance = 0;

      kol_prolog = datediff(month, max_date_graph, k_date_end);

      curr_rate = 1;
      if (curr <> def_curr) then
      begin
        select cr_rate
        from opercurr
        where cr_from_cu_id = :curr and
              cr_date <= :rep_date_end and
              cr_date_end >= :rep_date_end rows 1
        into :curr_rate;
        if ((curr_rate is null) or (curr_rate = 0)) then
          curr_rate = 1;

        if (pr_set_rate_in_kont = 1) then
        begin
          if (curr_rate < k_index_curr_rate) then
            curr_rate = :k_index_curr_rate;
        end

        rep_active_balance = rep_active_balance * :curr_rate;
        rep_active_balance = cast(rep_active_balance as decimal(15,2));

        balance_rest = balance_rest * :curr_rate;
        balance_rest = cast(balance_rest as decimal(15,2));
      end

      if (db_id <> 81001) then
        suspend;

      uuid = null;
      if (db_id = 81001) then
      begin
        for select f_uuid_cbr(f_uuid('knclient', kc_id))
            from knclient
            where kc_from_k_id = :k_id
            and kc_amount_approved > 0
            into uuid
        do
        begin
          suspend;
        end
      end
    end
  end
end
