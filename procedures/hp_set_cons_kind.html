CREATE PROCEDURE HP_SET_CONS_KIND(
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    CONS_KIND_ID D_SMALLINT)
RETURNS (
    RES_TEXT D_STR_255,
    NEED_REFRESH D_SMALLINT)
AS
declare variable linked_id d_id;
declare variable ks_state d_smallint;
declare variable edoc_status varchar(20);
begin
  res_text = '';
  need_refresh = 0;

  if (not exists(
    select 1
    from enum_list
    where el_kind = 'consultation_kind'
        and el_id = :cons_kind_id))
    then
        res_text = 'Некорректный тип оплаты за конс. услуги';

  if (res_text = '') then
  begin
    if (obj_kind = 'miniapp') then
    begin
      select coalesce(k.k_id, 0)
      from miniapp ma
      left join kontract k on k.k_from_ma_app_id = ma.ma_id
      where ma.ma_id = :obj_id
        and ma.ma_is_active = 1
        and ma.ma_date_end = '31.12.9999'
      into linked_id;
      if (row_count = 0) then res_text = 'Заявка не найдена';
      else if (linked_id > 0) then
      begin
        select ks_state
        from kontstate
        where ks_from_k_id = :linked_id
            and ks_ol_is_delete = 0
            and ks_date_end = '31.12.9999'
        into ks_state;
        if (ks_state in (1, 2)) then res_text = 'Контракт уже выдан, изменить тип оплаты нельзя';
      end
    end
    else
    begin
      select k.k_from_ma_app_id, ks_state
      from kontract k
      left join kontstate ks on ks_from_k_id = k.k_id and ks_ol_is_delete = 0 and ks_date_end >= '31.12.2999'
      where k.k_id = :obj_id
      into linked_id, ks_state;
      if (row_count = 0) then res_text = 'Контракт не найден';
      else if (ks_state in (1, 2)) then res_text = 'Контракт уже выдан, изменить тип оплаты нельзя';
      else if (ks_state <> 0) then res_text = 'Контракт должен быть в процессе';
    end

--    if (res_text = '' and coalesce(linked_id, :obj_id) > 0) then
--    begin
--      edoc_status = f_edoc_status('consulting.agreement', coalesce(linked_id, :obj_id), current_date);
--      if (edoc_status = 'signed') then res_text = 'Документ консультации уже подписан, изменить нельзя';
--    end

    if (res_text = '') then
    begin
      if (obj_kind = 'miniapp') then
      begin
        update miniapp
        set ma_cons_kind = :cons_kind_id
        where ma_id = :obj_id
            and ma_is_active = 1
            and ma_date_end = '31.12.9999';
        if (linked_id > 0) then
        begin
          update kontract
          set k_cons_kind = :cons_kind_id
          where k_id = :linked_id;
        end
      end
      else
      begin
        update kontract
        set k_cons_kind = :cons_kind_id
        where k_id = :obj_id;
        if (linked_id > 0) then
        begin
          update miniapp
          set ma_cons_kind = :cons_kind_id
          where ma_id = :linked_id
              and ma_is_active = 1
              and ma_date_end = '31.12.9999';
        end
      end
      res_text = 'Тип оплаты консультационных услуг успешно изменен';
      need_refresh = 1;
    end
  end
  suspend;
end
