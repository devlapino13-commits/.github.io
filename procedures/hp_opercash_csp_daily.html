CREATE PROCEDURE HP_OPERCASH_CSP_DAILY(
    REP_DATE D_DATE,
    OFFICE_ID D_INTEGER,
    CASH_ID D_INTEGER,
    DOC_STATUS D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    OFFICE_NAME D_STR_100,
    CASH_NAME D_STR_100,
    CASHIER_NAME D_STR_100,
    CLS_RES_BEGIN D_STR_100,
    CLS_RES_END D_STR_100,
    IPF_RES_BEGIN D_STR_100,
    IPF_RES_END D_STR_100,
    OPC_ID D_INTEGER,
    OPC_PARENT_ID D_INTEGER,
    OPC_TIME_STR D_STR_10,
    OPC_DOC_NO D_INTEGER,
    OPC_SIGN_STR D_STR_10,
    OPC_IS_IPF D_BOOL,
    OPC_CORR_AL_ID D_STR_20,
    OPC_IS_ONLINE D_BOOL,
    OPC_MEMO D_STR_LONG,
    AMOUNT_PLUS D_CURRENCY,
    AMOUNT_MINUS D_CURRENCY,
    RES_CORR_OFFICE D_STR_100,
    RES_CLIENT_FIO D_STR_255,
    RES_STATUS_STR D_STR_20,
    RES_OPER_KIND D_STR_100,
    RES_OPER_CODE D_STR_CODE,
    RES_MANAGER_FIO D_STR_100,
    RES_USERNAME D_STR_CODE)
AS
declare variable cls_al_account d_str_code;
declare variable ipf_al_account d_str_code;
declare variable res_line_cation d_str_100;
declare variable res_begin d_currency;
declare variable res_end d_currency;
declare variable csp_cashier_id d_id;
begin

  select reg_code || ' ' || reg_name,
         cls.al_account,
         ipf.al_account,
         csp_cashier_id
  from cashplace
  inner join region on reg_unicode = csp_office_id
  left join acc_list cls on csp_cls_al_id = cls.al_id
  left join acc_list ipf on csp_ipf_al_id = ipf.al_id
  where csp_office_id = :office_id and
        csp_id = :cash_id
  into office_name,
       cls_al_account,
       ipf_al_account,
       csp_cashier_id;

  for select res_line_cation,
             res_begin,
             res_end
      from hp_opercash_daily_total(:cash_id, :rep_date)
      where res_category starting with '# Касса'
      into res_line_cation,
           res_begin,
           res_end
  do
  begin
    if (res_line_cation = cls_al_account) then
    begin
      cls_res_begin = cls_al_account || ' - ' || f_numtostr(res_begin, 2);
      cls_res_end = cls_al_account || ' - ' || f_numtostr(res_end, 2);
    end
    else if (res_line_cation = ipf_al_account) then
    begin
      ipf_res_begin = ipf_al_account || ' - ' || f_numtostr(res_begin, 2);
      ipf_res_end = ipf_al_account || ' - ' || f_numtostr(res_end, 2);
    end
  end

  cash_name = cls_al_account || ', ' || ipf_al_account;

  select c_name_fio
  from clients
  where c_id = :csp_cashier_id
  into cashier_name;

  for select opc_id,
             opc_parent_id,
             left(cast(opc_datetime as time), 5),
             opc_doc_no,
             decode(opc_sign, 1, 'П', -1, 'Р'),
             opc_is_ipf,
             opc_corr_al_id,
             opc_is_online,
             opc_memo,
             res_amount_plus,
             res_amount_minus,
             res_corr_office,
             res_client_fio,
             res_status_str,
             res_oper_kind,
             res_oper_code,
             res_manager_fio,
             lower(res_username)
      from hp_opercash_office(:office_id, :cash_id, :rep_date, :doc_status, :un_id)
      order by opc_datetime, opc_doc_no
      into opc_id,
           opc_parent_id,
           opc_time_str,
           opc_doc_no,
           opc_sign_str,
           opc_is_ipf,
           opc_corr_al_id,
           opc_is_online,
           opc_memo,
           amount_plus,
           amount_minus,
           res_corr_office,
           res_client_fio,
           res_status_str,
           res_oper_kind,
           res_oper_code,
           res_manager_fio,
           res_username
  do
    suspend;

end
