CREATE PROCEDURE OKT_WEBSITE_MINIAPP_MAKE_CALL(
    CK_ID INTEGER,
    CK_MEMO VARCHAR(1000),
    CK_STATUS INTEGER,
    CK_FIO STR_100,
    CK_CITIZENSHIP INTEGER,
    CK_BIRTH_DATE DATE,
    CK_CITY ID_TYPE,
    CK_CRED_AMOUNT INTEGER,
    CK_PERIOD STR_100,
    CK_LEGAL_DOCS STR_255,
    CK_JOB_STATUS SMALLINT,
    CK_NEXT_CALL_DATETIME TIMESTAMP,
    CK_LO_ID ID_TYPE,
    CK_FROM_REG_ID INTEGER,
    CK_CELL_NO VARCHAR(20),
    CK_PASS_DATA VARCHAR(30),
    CK_SVZ_ID ID_TYPE,
    CK_SMS_TXT STR_255)
AS
declare variable sec_data varchar(255);
begin
  sec_data = '';
  ck_sms_txt = coalesce(ck_sms_txt, '');

  ck_cell_no = trim(coalesce(ck_cell_no, ''));
  if (ck_cell_no = '79') then
    ck_cell_no = '';

  ck_pass_data = trim(coalesce(ck_pass_data, ''));
  if (ck_pass_data <> '') then
    sec_data = '<pass>' || ck_pass_data || '</>';
  if (ck_cell_no <> '') then
    sec_data = sec_data || '<cell>' || ck_cell_no || '</>';

  if (sec_data = '') then
    sec_data = null;

  update cc_check_knclient chk
  set ck_type = 6,
      ck_is_blocked = 0,
      ck_checked_oper = chk.ck_blocked_oper,
      ck_memo = :ck_memo,
      ck_status = :ck_status,
      ck_call_qty = 1,
      ck_fio = :ck_fio,
      ck_citizenship = :ck_citizenship,
      ck_birth_date = :ck_birth_date,
      ck_city = :ck_city,
      ck_cred_amount = :ck_cred_amount,
      ck_period = :ck_period,
      ck_legal_docs = :ck_legal_docs,
      ck_job_status = :ck_job_status,
      ck_next_call_datetime = :ck_next_call_datetime,
      ck_lo_id = :ck_lo_id,
      ck_from_reg_id = :ck_from_reg_id,
      ck_secondary_data = :sec_data
  where (ck_id = :ck_id);

  if (ck_status = 2) then
    execute procedure hp_cc_lo_sms_send(:ck_id, :ck_svz_id, :ck_sms_txt);
end
