CREATE PROCEDURE HP_EXTINFO_MILITARY_UPDATE(
    EIL_ID D_ID)
AS
declare variable EIL_STATUS D_SMALLINT;
declare variable EIL_STATUS_DATETIME D_DATETIME;
declare variable JSON_STR D_STR_255;
declare variable ANSWER_DATE D_DATE;
declare variable ANSWER_STATUS D_BOOLEAN;
begin
  -- RFCKG-4079 получения информации о статусе прохождения воинской службы в КЗ
  select cast(blx_blob as d_blob_text),
         eil.eil_status,
         eil.eil_status_datetime
  from extinfolist eil, extinfocfg, hp_get_blobex_blob(eil_from_bls_id)
  where eic_code = 'military_status' and
        eil_from_eic_id = eic_id and
        eil_id = :eil_id
  into json_str,
       eil_status,
       eil_status_datetime;

  if (eil_status = 2) then
  begin

    -- Если клиент не найден
    if (coalesce(json_str, '') = '') then
    begin
      update military_check_query
      set mcq_status = -1, -- ответ получен клиент НЕ найден
          mcq_received = 1,
          mcq_status_datetime = :eil_status_datetime
      where mcq_from_eil_id = :eil_id;
      exit;
    end

    if (f_json_str_get_value(json_str, 'code', 0) = 500 or
        f_json_str_get_value(json_str, 'code', 0) = 410) then -- RFCKG-4518
    begin
      update military_check_query
        set mcq_status = 3,
            mcq_received = 1,
            mcq_status_datetime = :eil_status_datetime
       where mcq_from_eil_id = :eil_id;
    end

    -- Получаем ответ
    answer_date = cast(f_json_str_get_value(json_str, 'date', null) as d_date);
    answer_status = cast(f_json_str_get_value(json_str, 'status', null) as d_boolean);

    if (coalesce(answer_date, '') <> '' and coalesce(answer_status, '') <> '') then
    begin
      update military_check_query
      set mcq_status = iif(:answer_status, 1, 0),
          mcq_answer_date = :answer_date, -- ответ получен клиент найден
          mcq_received = 1,
          mcq_status_datetime = :eil_status_datetime
      where mcq_from_eil_id = :eil_id;
    end
  end

  if (eil_status in (-3, -2)) then --Обработка ошибок
  begin
    update military_check_query
    set mcq_status = -2,
        mcq_received = 1,
        mcq_status_datetime = :eil_status_datetime
    where mcq_from_eil_id = :eil_id;
  end
end
