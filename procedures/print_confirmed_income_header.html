CREATE PROCEDURE PRINT_CONFIRMED_INCOME_HEADER(
    K_ID D_ID)
RETURNS (
    KONTRACT_NO D_STR_20,
    PROF_NO D_STR_20,
    FROM_DATE D_DATE,
    FROM_MONTH D_STR_20)
AS
begin
  -- RFCRU-4782
  -- ПДН; Профсуждение;
  for
    select
      k_kontract_no,
      k_disb_date
    from kontract
    where k_id = :k_id
  into
    :kontract_no,
    :from_date
  do
  begin
    prof_no = kontract_no || ' пдн.';
    from_month =
      case extract(month from from_date)
        when 1  then 'Январь'
        when 2  then 'Февраль'
        when 3  then 'Март'
        when 4  then 'Апрель'
        when 5  then 'Май'
        when 6  then 'Июнь'
        when 7  then 'Июль'
        when 8  then 'Август'
        when 9  then 'Сентябрь'
        when 10 then 'Октябрь'
        when 11 then 'Ноябрь'
        when 12 then 'Декабрь'
      end || ' ' || extract(year from from_date);
    suspend;
  end
end
