CREATE PROCEDURE CB_AZ_LO2013_03(
    REP_REGION_ID INTEGER,
    REP_BONUS_ID INTEGER,
    REP_USER_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    AA_CURR_OFFICE_NAME VARCHAR(60),
    AA_CURR_KC_FIO VARCHAR(100),
    AA_AMOUNT_ACTIVE_END CURR_TYPE,
    AA_DELIQ_BALANCE7 CURR_TYPE,
    PAR7 PERC_TYPE,
    CB_DISB_CLI_NEW INTEGER,
    CB_DISB_CLI_OLD INTEGER,
    BONUS_TOTAL CURR_TYPE)
AS
declare variable aa_curr_kc_id id_type;
declare variable rep_date_begin date;
declare variable aa_curr_region_id integer;
declare variable bonus_disb curr_type;
declare variable bonus_portfel curr_type;
declare variable bonus_perc decimal(2,2);
begin
  rep_date_begin = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date);
  for select aa_curr_kc_id, aa_curr_region_id, aa_curr_office_name, aa_curr_kc_fio, sum(aa_amount_active_end),
             sum(aa_deliq_balance7)
      from aarep_analitica_port_temp(:rep_region_id, :rep_date_begin, :rep_date_end, '', 0, 0, 0, 0, 0, 0, 0)
      where aa_amount_active_end <> 0
      group by aa_curr_kc_id, aa_curr_region_id, aa_curr_office_name, aa_curr_kc_fio
      order by aa_curr_office_name, aa_curr_kc_fio
      into aa_curr_kc_id, aa_curr_region_id, aa_curr_office_name, aa_curr_kc_fio, aa_amount_active_end,
           aa_deliq_balance7
  do
  begin
    par7 = 0;
    if (aa_amount_active_end > 0) then
      par7 = aa_deliq_balance7 / aa_amount_active_end * 100;
--  Вычисляет кол-во выданных кредитов, новым и повторным
    cb_disb_cli_new = null;
    cb_disb_cli_old = null;
    select sum(
           case
             when kc_circle <= 1 then 1
           end),
           sum(
           case
             when kc_circle > 1 then 1
           end)
    from kontstate, knclient
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end and
          ks_kc_id = :aa_curr_kc_id and
          ks_region_id = :aa_curr_region_id and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0
    into cb_disb_cli_new, cb_disb_cli_old;
    if (cb_disb_cli_new is null) then
      cb_disb_cli_new = 0;
    if (cb_disb_cli_old is null) then
      cb_disb_cli_old = 0;

    bonus_disb = cb_disb_cli_new * 7 + cb_disb_cli_old * 5;
    bonus_portfel = case
                      when aa_amount_active_end >= 50000 then aa_amount_active_end * 0.001
                      else 0
                    end;
    bonus_perc = case
                   when par7 < 1 then 1
                   when par7 >= 1 and par7 < 2 then 0.75
                   when par7 >= 2 and par7 < 3 then 0.5
                   else 0
                 end;
    bonus_total = (bonus_disb + bonus_portfel) * bonus_perc;
    bonus_total = round(bonus_total, 0);
    suspend;
  end

end
