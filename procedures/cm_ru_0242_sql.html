CREATE PROCEDURE CM_RU_0242_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable pr_code str_100;
declare variable c_id id_type;
declare variable final_pdn d_currency;
declare variable pdn_limit d_currency;
declare variable kc_id d_id;
declare variable ma_id d_integer;
begin
  res_error_is_critical = 2;

  for select pr_code, kc_from_c_id, kc_id, k_from_ma_app_id
      from kontract, product, knclient
      where k_id = :k_id and
            k_from_pr_id = pr_id and
            kc_from_k_id = k_id and
            kc_amount_approved > 0
      into pr_code, c_id, kc_id, ma_id
  do
  begin
    if (exists(select *
               from miniapp ma
               inner join miniappclient mac on mac_from_ma_id = ma_id
               inner join miniappclient_income maci on maci.maci_from_mac_id = mac.mac_id and
                     maci.maci_is_documented = 1
               where ma.ma_id = :ma_id and
                     ma.ma_is_active = 1 and
                     ma.ma_date_end >= '31.12.9999' and
                     mac_from_c_id = :c_id) and
        check_kind containing '.send_kk.') then
      continue;

    final_pdn = coalesce(final_pdn, 0);

    if (pr_code not containing '.selfemployed.' and
        pr_code not containing '.business.') then
    begin
      final_pdn = 0;
      pdn_limit = 0;

      select coalesce(kc_pdn_koef, 0)
      from knclient_pdn
      where kc_id = :kc_id and
            kcp_kind = 0
      into final_pdn;

      select pdn_limit
      from hp_pdn_get_limit(:kc_id)
      into pdn_limit;

      if (final_pdn > pdn_limit) then
      begin
        if (pdn_limit = 0) then
          continue;

        res_error_text = 'Кредитование запрещено: ПДН более ' || round(pdn_limit) || '%';
        suspend;
      end
    end
  end
end
