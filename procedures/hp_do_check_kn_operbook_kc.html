CREATE PROCEDURE HP_DO_CHECK_KN_OPERBOOK_KC(
    REP_SAB_FULL_CODE D_STR_CODE NOT NULL,
    REP_KC_ID D_ID NOT NULL,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE NOT NULL)
RETURNS (
    KC_ID D_ID,
    K_ID D_ID,
    C_ID D_ID,
    K_KONTRACT_NO D_STR_CODE,
    SAB_ID D_INTEGER,
    SAB_FULL_CODE D_STR_CODE,
    OKS_CLOSE_STATE D_INTEGER,
    OKS_AMOUNT D_CURRENCY,
    EPS_AMOUNT D_CURRENCY,
    ACC_ID_LIST D_STR_255)
AS
declare variable acc_id d_id;
declare variable tmp_amount d_currency;
declare variable oks_balance d_currency;
declare variable oks_bal_perc d_currency;
declare variable oks_diff_ecl_llr d_currency;
declare variable oks_ecl_acc_kind d_integer;
declare variable oks_llr_os d_currency;
declare variable oks_llr_perc d_currency;
declare variable oks_reserv_law d_currency;
declare variable oks_diff_esp_bal d_currency;
declare variable operkroff_os d_currency;
declare variable operkroff_perc d_currency;
declare variable ok_pereplata d_currency;
declare variable oks_bal_law d_currency;
declare variable oks_bal_esp_end d_currency;
declare variable oks_ecl_vbs d_currency;
declare variable oks_off_esp_bal d_currency;
begin
  if (rep_sab_full_code = '') then
    rep_sab_full_code = 'full';

  if (rep_date_begin is null) then
    rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  kc_id = rep_kc_id;
  k_id = null;
  oks_close_state = -1;
  oks_balance = 0;
  oks_bal_perc = 0;
  oks_diff_ecl_llr = 0;
  oks_ecl_acc_kind = -1;
  oks_llr_os = 0;
  oks_llr_perc = 0;
  oks_reserv_law = 0;
  oks_diff_esp_bal = 0;
  operkroff_os = 0;
  operkroff_perc = 0;
  ok_pereplata = 0;
  oks_bal_law = 0;
  oks_bal_esp_end = 0;
  oks_ecl_vbs = 0;
  oks_off_esp_bal = 0;

  select kc_from_k_id, kc_from_c_id, k_kontract_no
  from knclient, kontract
  where kc_id = :kc_id and
        kc_from_k_id = k_id
  into k_id, c_id, k_kontract_no;

  if (c_id is null) then
    exit;

  select oks_close_state,
         oks_balance,
         oks_bal_perc,
         oks_ecl_acc_kind,
         iif(oks_close_state = 0, oks_diff_ecl_llr, 0),
         oks_llr_os,
         oks_llr_perc,
         oks_reserv_law,
         iif(oks_close_state = 0, oks_diff_esp_bal, 0),
         oks_bal_law,
         oks_bal_esp_end - oks_off_esp_bal, -- vbs
         oks_ecl_vbs -- все резервы кроме госпошлины
  from kn_oksummary
  where oks_from_kc_id = :kc_id and
        oks_date_begin <= :rep_date_end
  order by oks_from_kc_id desc, oks_date_begin desc
  rows 1
  into oks_close_state,
       oks_balance,
       oks_bal_perc,
       oks_ecl_acc_kind,
       oks_diff_ecl_llr,
       oks_llr_os,
       oks_llr_perc,
       oks_reserv_law,
       oks_diff_esp_bal,
       oks_bal_law,
       oks_bal_esp_end,
       oks_ecl_vbs;

  select sum(oko_os + oko_sh_os),  sum(oko_perc + oko_sh_perc)
  from operkroff
  where oko_ol_date <= :rep_date_end and
        oko_ol_is_active = 1 and
        oko_from_k_id = :k_id and
        oko_from_c_id = :c_id
  into operkroff_os, operkroff_perc;

  if (operkroff_os is null) then
    operkroff_os = 0;

  if (operkroff_perc is null) then
    operkroff_perc = 0;

  select sum(ok_os)
  from kn_operkred
  where ok_ol_date <= :rep_date_end and
        ok_ol_is_active = 1 and
        ok_from_kc_id = :kc_id and
        ok_type = 9
  into ok_pereplata;

  if (ok_pereplata is null) then
    ok_pereplata = 0;

  for select sab_id, sab_full_code
  from sys_accbalance
  where sab_object_kind = 8 and
        (:rep_sab_full_code = 'full' or sab_full_code = :rep_sab_full_code)
  union
  select -1, 'vbs' from rdb$database
  where :rep_sab_full_code in ('full', 'vbs')
  union
  select -2, 'reserv all' from rdb$database
  where :rep_sab_full_code in ('full', 'reserv all')
  into sab_id, sab_full_code
  do
  begin
    oks_amount = 0;

    if (sab_full_code = 'Микрозаймы.Баланс.ОС') then
      oks_amount = oks_balance;
    else
    if (sab_full_code = 'Микрозаймы.Начислено.Процент') then
      oks_amount = oks_bal_perc;
    else
    if (sab_full_code starting 'Микрозаймы.Резерв.Корректировка') then
    begin
      if (oks_ecl_acc_kind = right(sab_full_code, 1)) then
        oks_amount = oks_diff_ecl_llr;
    end
    else
    if (sab_full_code starting 'Микрозаймы.Резерв.ОС') then
    begin
      if (oks_ecl_acc_kind = right(sab_full_code, 1)) then
        oks_amount = oks_llr_os;
    end
    else
    if (sab_full_code starting 'Микрозаймы.Резерв.Процент') then
    begin
      if (oks_ecl_acc_kind = right(sab_full_code, 1)) then
        oks_amount = oks_llr_perc;
    end
    else
    if (sab_full_code starting 'Микрозаймы.Резерв.Госпошлина') then
    begin
      if (oks_ecl_acc_kind = right(sab_full_code, 1)) then
        oks_amount = oks_reserv_law;
    end
    else
    if (sab_full_code = 'Микрозаймы.Баланс.ИзмВБС') then
      oks_amount = oks_diff_esp_bal;
    else
    if (sab_full_code = 'Микрозаймы.Внебаланс.ОС') then
      oks_amount = operkroff_os;
    else
    if (sab_full_code = 'Микрозаймы.Внебаланс.Процент') then
      oks_amount = operkroff_perc;
    else
    if (sab_full_code = 'Микрозаймы.Погашено.Переплата') then
      oks_amount = ok_pereplata;
    else
    if (sab_full_code = 'Микрозаймы.Погашено.Госпошлина') then
      oks_amount = oks_bal_law;
    else
    if (sab_full_code = 'vbs') then
    begin
      if (oks_close_state <> 2) then
        oks_amount = oks_bal_esp_end;
    end
    else
    if (sab_full_code = 'reserv all') then
    begin
      if (oks_close_state = 0) then
        oks_amount = oks_ecl_vbs;
    end else
      Continue;

    acc_id = -1;
    acc_id_list = '';
    eps_amount = 0;

    if (sab_full_code = 'vbs') then
    begin
      for select acc_id
      from ref_account
      where acc_from_sab_id in (12, 13, 15, 16, 20, 21, 54, 55) and
            acc_object_kind = 8 and
            acc_object_id = :kc_id
      into acc_id
      do
      begin
         tmp_amount = f_acc_bal(acc_id, rep_date_end, 'sum');
         eps_amount = eps_amount + tmp_amount;
  
         if (tmp_amount <> 0) then
         begin
           if (acc_id_list <> '') then
             acc_id_list = acc_id_list || ',';
  
           acc_id_list = acc_id_list || acc_id;
         end
      end
    end else
    if (sab_full_code = 'reserv all') then
    begin
      for select acc_id
      from ref_account
      where acc_from_sab_id between 30 and 38 and -- все счета резерва кроме госпошлины
            acc_object_kind = 8 and
            acc_object_id = :kc_id
      into acc_id
      do
      begin
         tmp_amount = f_acc_bal(acc_id, rep_date_end, 'sum');
         eps_amount = eps_amount + tmp_amount;
  
         if (tmp_amount <> 0) then
         begin
           if (acc_id_list <> '') then
             acc_id_list = acc_id_list || ',';
  
           acc_id_list = acc_id_list || acc_id;
         end
      end
    end else
    begin
      for select acc_id
      from ref_account
      where acc_from_sab_id = :sab_id and
            acc_object_kind = 8 and
            acc_object_id = :kc_id
      into acc_id
      do
      begin
         tmp_amount = f_acc_bal(acc_id, rep_date_end, 'sum');
         eps_amount = eps_amount + tmp_amount;
  
         if (tmp_amount <> 0) then
         begin
           if (acc_id_list <> '') then
             acc_id_list = acc_id_list || ',';
  
           acc_id_list = acc_id_list || acc_id;
         end
      end
    end

    if (acc_id_list = '') then
      acc_id_list = acc_id;

    if (sab_full_code starting 'Микрозаймы.Резерв.Корректировка' or
        sab_full_code starting 'Микрозаймы.Резерв.ОС' or
        sab_full_code starting 'Микрозаймы.Резерв.Процент' or
        sab_full_code starting 'Микрозаймы.Резерв.Госпошлина' or
        sab_full_code = 'reserv all') then
      eps_amount = -eps_amount;


    if (eps_amount <> oks_amount) then
      suspend;
  end

  eps_amount = 0;
  oks_amount = 0;

  if (rep_sab_full_code not in ('full', 'vbs_out')) then
    Exit;

  sab_full_code = 'vbs_out';
  select oks_off_esp_bal
  from kn_oksummary
  where oks_date_begin between :rep_date_begin and :rep_date_end and
        oks_from_kc_id = :kc_id and
        oks_close_state = 1
  into oks_off_esp_bal;

  oks_amount = oks_off_esp_bal;

  select sum(iif(a_dt.acc_from_sab_id in (57, 89), kob_summa, -kob_summa))
  from ref_account a_dt, ref_account a_kt, kn_operbook
  where kob_from_kc_id = :kc_id and
        kob_ol_date between :rep_date_begin and :rep_date_end and
        kob_ol_is_active = 1 and
        a_dt.acc_id = kob_acc_dt and
        a_kt.acc_id = kob_acc_kt and
        ((a_dt.acc_from_sab_id = 13 and a_kt.acc_from_sab_id in (57, 89)) or
         (a_dt.acc_from_sab_id in (57, 89) and a_kt.acc_from_sab_id = 13))
  into eps_amount;

  if (eps_amount is null) then
    eps_amount = 0;

  if (eps_amount <> oks_amount) then
    suspend;
end
