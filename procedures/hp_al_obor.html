Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_AL_OBOR(
    AL_ID D_STR_20,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    DT_EKV D_CURRENCY,
    KT_EKV D_CURRENCY,
    DT_VAL D_CURRENCY,
    KT_VAL D_CURRENCY,
    MAX_DATE D_DATE)
AS
declare variable recalc_date d_date;
declare variable tmp_date_begin d_date;
declare variable tmp_date_end d_date;
declare variable tmp_sum d_currency;
declare variable tmp_val d_currency;
begin
  select osc_date
  from obsumrecalc
  where osc_from_al_id = :al_id and
        osc_date <= :rep_date_end
  order by osc_date
  rows 1
  into recalc_date;

  tmp_date_begin = rep_date_begin;
  tmp_date_end = rep_date_end;

  if (recalc_date is not null) then
  begin
    if (recalc_date <= rep_date_begin) then
    begin
      tmp_date_begin = null;
      tmp_date_end = null;
    end else
    begin
      if (recalc_date <= rep_date_end) then
        tmp_date_end = recalc_date - 1;
    end
  end

  if (tmp_date_begin is not null) then
  begin
    select sum(obs_dt_ekv), sum(obs_kt_ekv), sum(obs_dt_val), sum(obs_kt_val), max(obs_date)
    from obsummary
    where obs_from_al_id = :al_id and              
          obs_date between :tmp_date_begin and :tmp_date_end
    into :dt_ekv, :kt_ekv, :dt_val, :kt_val, max_date;
  end

  if (dt_ekv is null) then
    dt_ekv = 0;

  if (kt_ekv is null) then
    kt_ekv = 0;

  if (dt_val is null) then
    dt_val = 0;

  if (kt_val is null) then
    kt_val = 0;

  if (recalc_date is not null) then
  begin
    tmp_date_begin = maxvalue(recalc_date, rep_date_begin);
    tmp_date_end = rep_date_end;

    tmp_sum = 0;
    tmp_val = 0;

    select sum(ob_summa), sum(ob_summa_val)
    from operbook
    where ob_account_dt = :al_id and
          ob_ol_date between :tmp_date_begin and :tmp_date_end and
          ob_ol_is_active = 1
    into tmp_sum, tmp_val;

    if (tmp_sum is null) then
      tmp_sum = 0;

    if (tmp_val is null) then
      tmp_val = 0;

    dt_ekv = dt_ekv + tmp_sum;
    dt_val = dt_val + tmp_val;

    tmp_sum = 0;
    tmp_val = 0;

    select sum(ob_summa), sum(ob_summa_val)
    from operbook
    where ob_account_kt = :al_id and
          ob_ol_date between :tmp_date_begin and :tmp_date_end and
          ob_ol_is_active = 1
    into tmp_sum, tmp_val;

    if (tmp_sum is null) then
      tmp_sum = 0;

    if (tmp_val is null) then
      tmp_val = 0;

    kt_ekv = kt_ekv + tmp_sum;
    kt_val = kt_val + tmp_val;
  end

  suspend;
end
