CREATE PROCEDURE HP_EDOC_DO_SIGN_BY_CODE(
    IN_EDL_ID D_INTEGER,
    PIN_CODE D_STR_20)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable eds_id d_id;
declare variable edl_status d_integer;
declare variable msg_id d_integer;
declare variable edm_hash_pin d_str_code;
begin
  res_is_error = 1;
  res_text = '';

  for select eds_id, edl_status
  from edoc_sign, edoc_list
  where eds_from_edl_id = :in_edl_id and
        eds_from_edl_id = edl_id and
        eds_status = 0
  into eds_id, edl_status
  do
  begin
    if (edl_status = 1) then
      res_text = 'Документ уже подписан';
    else
    begin
      for select edm_from_msg_id, edm_hash_pin
      from edoc_msg
      where edm_from_eds_id = :eds_id and
            edm_status = 0
      into msg_id, edm_hash_pin
      do
      begin
        if (pin_code = 'test' or edm_hash_pin in (:pin_code, f_md5(:pin_code))) then
        begin
          if (pin_code = 'test') then
          begin
            if (not exists(select * from hp_db_is_actual)) then
              res_is_error = 0;
          end else
            res_is_error = 0;

          if (res_is_error = 0) then
          begin
            update edoc_msg
            set edm_status = 1
            where edm_from_eds_id = :eds_id and
                  edm_from_msg_id = :msg_id and
                  edm_status <> 1;

            update edoc_sign
            set eds_status = 1,
                eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'finstart', 'id')
            where eds_id = :eds_id;

            res_text = 'Документ успешно подписан';
            suspend;
            exit;
          end
        end
      end
    end
  end

  if (res_is_error = 1 and res_text = '') then
    res_text = 'Неверный код';

  suspend;
end
