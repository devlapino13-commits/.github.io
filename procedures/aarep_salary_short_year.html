CREATE PROCEDURE AAREP_SALARY_SHORT_YEAR(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    ZP_PERSONAL_ID DECIMAL(18,0),
    ZP_PERSONAL_FIO VARCHAR(100),
    ZP_INN VARCHAR(20),
    ZP_REGION VARCHAR(100),
    ZP_SUMMA_1 DECIMAL(18,4),
    ZP_SUMMA_2 DECIMAL(18,4),
    ZP_SUMMA_3 DECIMAL(18,4),
    ZP_SUMMA_4 DECIMAL(18,4),
    ZP_SUMMA_5 DECIMAL(18,4),
    ZP_SUMMA_6 DECIMAL(18,4),
    ZP_SUMMA_7 DECIMAL(18,4),
    ZP_SUMMA_8 DECIMAL(18,4),
    ZP_SUMMA_9 DECIMAL(18,4),
    ZP_SUMMA_10 DECIMAL(18,4),
    ZP_SUMMA_11 DECIMAL(18,4),
    ZP_SUMMA_12 DECIMAL(18,4),
    SD_BIRTH_DATE DATE,
    SD_SEX_STR VARCHAR(20),
    SD_COMMENT_SF VARCHAR(100),
    SD_SF_LGOTA DECIMAL(18,7))
AS
declare variable c_sex integer;
declare variable c_id decimal(18,0);
declare variable dd date;
begin
  for select zp_personal_id, zp_personal_fio, c_inn, reg_code || ' ' || reg_name, c_sex, c_birth_date, c_id,
             sum(zp_summa_1), sum(zp_summa_2), sum(zp_summa_3), sum(zp_summa_4), sum(zp_summa_5), sum(zp_summa_6),
             sum(zp_summa_7), sum(zp_summa_8), sum(zp_summa_9), sum(zp_summa_10), sum(zp_summa_11), sum(zp_summa_12)
      from aarep_salary_short2(:report_region_id, :report_date_begin, :report_date_end), clients, region
      where zp_personal_id = c_id and
            zp_region = reg_unicode
      group by zp_personal_id, zp_personal_fio, c_inn, reg_code, reg_name, c_sex, c_birth_date, c_id
      order by reg_code, zp_personal_fio
      into :zp_personal_id, :zp_personal_fio, :zp_inn, :zp_region, :c_sex, :sd_birth_date, :c_id, :zp_summa_1,
           :zp_summa_2, :zp_summa_3, :zp_summa_4, :zp_summa_5, :zp_summa_6, :zp_summa_7, :zp_summa_8, :zp_summa_9,
           :zp_summa_10, :zp_summa_11, :zp_summa_12
  do
  begin
    select sum(sao_sf_lgota)
    from sa_oklad
    where sao_from_c_id = :c_id
    into :sd_sf_lgota;
    if (sd_sf_lgota is null) then
      sd_sf_lgota = 0;
    if (c_sex = 0) then
      sd_sex_str = 'Женщина';
    else
      sd_sex_str = 'Мужчина';
    sd_comment_sf = '001 Стандарт';
    if (c_sex = 0) then
      dd = cast('01.01.1969' as date);
    else
      dd = cast('01.01.1964' as date);
    if (sd_birth_date < dd) then
      sd_comment_sf = '007 Муж.ранее 01.01.1964 и жен.ранее 01.01.1969';
    if (sd_sf_lgota > 0) then
      sd_comment_sf = '003 Работники пенсионеры';
    suspend;
  end
end
