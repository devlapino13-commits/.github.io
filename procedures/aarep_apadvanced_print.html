CREATE PROCEDURE AAREP_APADVANCED_PRINT(
    AA_ID INTEGER)
RETURNS (
    RES_AA_ID INTEGER,
    AA_FROM_K_ID DECIMAL(18,0),
    AA_FROM_C_ID DECIMAL(18,0),
    AA_DATE_APP DATE,
    AA_DATE_REPAY DATE,
    AA_FROM_REG_ID INTEGER,
    AA_REASON_ID INTEGER,
    COMMENTS VARCHAR(512),
    AA_IS_DELETED DECIMAL(18,0),
    AA_FROM_UN_ID DECIMAL(18,0),
    AA_DATETIME TYPE OF COLUMN APADVANCED.AA_DATETIME,
    AA_IS_ONE_CLIENT TYPE OF COLUMN APADVANCED.AA_IS_ONE_CLIENT,
    REP_KONTRACT_NO VARCHAR(100),
    REP_KF_NAME VARCHAR(100),
    REP_CLIENT_FIO VARCHAR(100),
    REP_REGION_NAME VARCHAR(100),
    REP_DISB_DATE DATE,
    REP_AMOUNT DECIMAL(18,4),
    RES_MEMBER VARCHAR(100),
    REP_COMPANY_NAME VARCHAR(255),
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_FINE CURR_TYPE,
    ZAD_NCP CURR_TYPE,
    ZAD_ALL CURR_TYPE)
AS
declare variable c_id decimal(18,0);
declare variable s_rec_no integer;
declare variable zad_kommis decimal(18,4) = 0;
begin
  rep_amount = 0;
  res_aa_id = :aa_id;
  for select aa_from_k_id, aa_from_c_id, aa_date_app, aa_date_repay, aa_from_reg_id, aa_reason_id, comments,
             aa_is_deleted, aa_from_un_id, aa_datetime, coalesce(aa_is_one_client, 0)
      from apadvanced
      where aa_id = :aa_id
      into :aa_from_k_id, :aa_from_c_id, :aa_date_app, :aa_date_repay, :aa_from_reg_id, :aa_reason_id, :comments,
           :aa_is_deleted, :aa_from_un_id, :aa_datetime, :aa_is_one_client
  do
  begin
    rep_kontract_no = '';
    rep_kf_name = '';
    rep_client_fio = '';
    rep_region_name = '';

    if (aa_from_c_id is null) then
    begin
      select kc_from_c_id
      from knclient
      where kc_from_k_id = :aa_from_k_id and
            kc_amount_approved > 0
      order by kc_status desc
      rows 1
      into :aa_from_c_id;
    end
    if (aa_from_c_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :aa_from_c_id
      into :rep_client_fio;
    end

    select k_kontract_no, k_disb_date, kal_new_kf_name
    from kontract, knaktlist
    where k_id = :aa_from_k_id and
          k_from_kf_id = kal_from_kf_id and
          kal_date <= :aa_date_app and
          kal_date_end >= :aa_date_app and
          knaktlist.is_delete = 0
    into :rep_kontract_no, :rep_disb_date, :rep_kf_name;

    select reg_code || ' ' || reg_name, reg_company_name
    from region
    where reg_unicode = :aa_from_reg_id
    into :rep_region_name, :rep_company_name;

    c_id = 0;
    res_member = '';
    if (aa_is_one_client = 1) then
    begin
      if (coalesce(aa_from_c_id, 0) <> 0) then
        c_id = :aa_from_c_id;
      else
        c_id = 0;
    end
    if (aa_is_one_client = 1) then
      res_member = ' за участника: ' || rep_client_fio;
    else
      res_member = ' за всех членов группы ' || rep_kf_name;

-- вычисляем сумму для досрочного погашения
    select ol_repay_amount
    from hp_do_repayment(:aa_from_k_id, :c_id, :aa_date_repay, 0, 1, 1, 1, 1, 0, -- :NEED_SAVE,
    1, 0, 0, 3, '010141710001000001', 1, 1000000000001, null, '', null, 0, 0)
    into :rep_amount;

    select max(s_rec_no)
    from temp_vrepay
    into :s_rec_no;

    zad_kommis = 0;
    select zad_kommis
    from temp_vrepay
    where s_rec_no = :s_rec_no
    into :zad_kommis;
    if (zad_kommis is null) then
      zad_kommis = 0;

    rep_amount = rep_amount - zad_kommis;

    select sum(pay_os) zad_os, sum(pay_perc) zad_perc, sum(pay_fine) zad_fine, sum(pay_tax) zad_ncp
    from temp_vrepay
    into zad_os, zad_perc, zad_fine, zad_ncp;

    zad_all = zad_os + zad_perc + zad_ncp + zad_fine;

    suspend;
  end
end
