CREATE PROCEDURE DATETOSTR(
    DTVAL DATE)
RETURNS (
    SVAL STR_100)
AS
begin
  sval = right('0' || extract(day from dtval), 2) || '.' || right('0' || extract(month from dtval), 2) || '.' || extract(year from dtval);
  suspend;
end
