Определение

Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_UXPARAMDEF_UPDATE(
    UFP_CODE D_STR_100,
    UXA_CODE D_STR_CODE,
    DEF_VALUE D_STR_100,
    IS_LOCKED D_BOOL,
    IS_INSERTING D_BOOL)
AS
begin
  if (uxa_code = '') then
    uxa_code = null;

  if (coalesce(def_value, '') = '' and is_inserting = 0) then
  begin
    delete from uxparamdef
    where upd_ufp_code = :ufp_code and
          upd_uxa_code is not distinct from :uxa_code;
  end else
  begin
    update or insert into uxparamdef (upd_ufp_code, upd_uxa_code, upd_def_value, upd_locked)
    values (:ufp_code, :uxa_code, :def_value, :is_locked)
    matching (upd_ufp_code, upd_uxa_code);

    -- если вставлен и значение пусто, то удаляем
    delete from uxparamdef
    where upd_ufp_code = :ufp_code and
          upd_uxa_code is not distinct from :uxa_code and
          upd_def_value = '';
  end
end
