CREATE PROCEDURE AAREP_KANS_SEND_SALDO_TMP(
    REP_DATE DATE,
    REP_REGION_FROM INTEGER,
    REP_REGION_TO INTEGER)
RETURNS (
    KOL_FROM_KSL_ID INTEGER,
    KOL_KOL_SALDO DECIMAL(18,4),
    KOL_KOL_ORDER DECIMAL(18,4))
AS
begin
  /* купленные минус отправленные: Остаток чего надо отправить */
  kol_kol_order = 0;
  kol_kol_saldo = 0;
  for select ksl_id
      from kanslist
      into kol_from_ksl_id
  do
  begin
    suspend;
  end

  for select kopl_from_ksl_id, sum(kopl_plus - kopl_minus)
      from kansoperlist, kansoper, region_list(:rep_region_from, 1, -1, 1)
      where kopl_from_kop_id = kop_id and
            kop_date <= :rep_date and
            (kop_from_reg_id = id_list) and
            kop_is_delete = 0
      group by kopl_from_ksl_id
      having sum(kopl_plus - kopl_minus) <> 0
      into :kol_from_ksl_id, :kol_kol_saldo
  do
  begin
    suspend;
  end
  kol_kol_saldo = 0;
  for select kol_from_ksl_id, sum(kol_kolich)
      from kansorlist, kansorder
      where kol_from_kso_id = kso_id and
            kso_date <= :rep_date and
            kso_from_reg_id = :rep_region_to and
            kso_state > 0 and
            kso_is_delete = 0
      group by kol_from_ksl_id
      having sum(kol_kolich) <> 0
      into :kol_from_ksl_id, :kol_kol_order
  do
  begin
    suspend;
  end

  kol_kol_order = 0;
  for select kopl_from_ksl_id, sum(kopl_minus)
      from kansoperlist, kansoper
      where kopl_from_kop_id = kop_id and
            kop_date <= :rep_date and
            kop_to_office = :rep_region_to and
            kop_is_delete = 0
      group by kopl_from_ksl_id
      having sum(kopl_minus) <> 0
      into :kol_from_ksl_id, :kol_kol_order
  do
  begin
    kol_kol_order = 0 - kol_kol_order;
    suspend;
  end
end
