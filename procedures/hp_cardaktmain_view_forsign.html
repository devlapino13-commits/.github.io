CREATE PROCEDURE HP_CARDAKTMAIN_VIEW_FORSIGN(
    C_ID D_ID)
RETURNS (
    CAM_ID D_INTEGER,
    CAM_DATE D_DATE,
    CAM_CARD_LIST D_STR_8100,
    CAM_CARD_COUNT D_INTEGER,
    CAM_SRC_CPL_ID D_INTEGER,
    REGS_ID D_INTEGER,
    REGS_NAME D_STR_CODE,
    CS_ID D_ID,
    CS_NAME_FIO D_STR_100,
    CPLS_IS_TRANSFER D_SMALLINT,
    CAM_DST_CPL_ID D_INTEGER,
    REGD_ID D_INTEGER,
    REGD_NAME D_STR_CODE,
    CD_ID D_ID,
    CD_NAME_FIO D_STR_100,
    CAM_DST_PIN_CPL_ID D_INTEGER,
    CD_PIN_ID D_ID,
    CD_PIN_NAME_FIO D_STR_100,
    CPLD_IS_TRANSFER D_SMALLINT,
    KIND_AKT D_STR_100,
    CAM_STATUS D_SMALLINT,
    UN_ID D_ID,
    UN_FIO D_STR_100,
    COMPANY_NAME D_STR_255,
    COMPANYF_NAME D_STR_255,
    CAM_DATETIME D_DATETIME,
    CAM_FROM_OL_ID D_ID,
    OL_IS_DELETE D_BOOL,
    CAM_OL_DATETIME D_DATETIME)
AS
begin
  for
    select
      c.cam_id,
      c.cam_date,
      c.cam_card_list,
      c.cam_card_count,
      c.cam_src_cpl_id,
      rs.reg_unicode, 
      rs.reg_name,
      cs.c_id, 
      cs.c_name_fio,
      cps.cpl_is_transfer,
      c.cam_dst_cpl_id,
      rd.reg_unicode, 
      rd.reg_name,
      cd.c_id, 
      cd.c_name_fio,
      c.cam_dst_pin_cpl_id,
      cdp.c_id,
      cdp.c_name_fio,
      cpd.cpl_is_transfer,
      c.cam_status,
      u.un_id,
      u.un_fio,
      c.cam_datetime,
      c.cam_from_ol_id,
      c.cam_ol_datetime
    from cardaktmain as c
      inner join cardplace as cps
        on cps.cpl_id = c.cam_src_cpl_id
      inner join region as rs
        on rs.reg_unicode = cps.cpl_from_reg_id
      inner join clients as cs
        on cs.c_id = cps.cpl_from_c_id
      inner join cardplace as cpd
        on cpd.cpl_id = c.cam_dst_cpl_id
          and cpd.cpl_from_c_id = :c_id
      inner join region as rd
        on rd.reg_unicode = cpd.cpl_from_reg_id
      inner join clients as cd
        on cd.c_id = cpd.cpl_from_c_id
      inner join username as u
        on u.un_id = c.cam_from_un_id
      left join cardplace as cpdp
        on cpdp.cpl_id = c.cam_dst_pin_cpl_id
      left join clients as cdp
        on cdp.c_id = cpdp.cpl_from_c_id
    where c.cam_status = 0
  into
    :cam_id,
    :cam_date,
    :cam_card_list,
    :cam_card_count,
    :cam_src_cpl_id,
    :regs_id,
    :regs_name,
    :cs_id,
    :cs_name_fio,
    :cpls_is_transfer,
    :cam_dst_cpl_id,
    :regd_id,
    :regd_name,
    :cd_id,
    :cd_name_fio,
    :cam_dst_pin_cpl_id,
    :cd_pin_id,
    :cd_pin_name_fio,
    :cpld_is_transfer,
    :cam_status,
    :un_id,
    :un_fio,
    :cam_datetime,
    :cam_from_ol_id,
    :cam_ol_datetime
  do
  begin
    if (cam_from_ol_id > 0) then
      ol_is_delete = (select is_delete from operlist where ol_id = :cam_from_ol_id);
    else ol_is_delete = 0;

    kind_akt =
      case
        when cam_src_cpl_id = 0 then 'приход карт из МПЦ'
        when cam_dst_cpl_id = 0 then 'передача заготовок в МПЦ'
        when cpls_is_transfer = 0 and cpld_is_transfer = 1  then 'передача карт из офиса'
        when cpls_is_transfer = 1 and cpld_is_transfer = 0  then 'приход карт в офис'
        when cpls_is_transfer = 0 and cpld_is_transfer = 0  then 'смена мат.ответственного лица'
      else '' end;

    suspend;
  end
end
