CREATE PROCEDURE HP_CARDOPER_ANNUL_VIEW(
    DTBEG D_DATE = null,
    DTEND D_DATE = null)
RETURNS (
    CON_ID D_INTEGER,
    CON_OL_ID D_ID,
    CON_REG_ID D_INTEGER,
    CON_K_ID D_ID,
    CON_C_ID D_ID,
    CON_CCA_ID D_INTEGER,
    CON_SUMMA D_CURRENCY,
    CON_SUMMAPERC D_CURRENCY,
    CON_CS_ID1 D_INTEGER,
    CON_CS_ID2 D_INTEGER,
    CON_CS_ID3 D_INTEGER,
    CON_STATUS D_SMALLINT,
    CON_UN_ID D_ID,
    CON_DATECREATE D_DATE,
    CON_DATEUPDATE D_DATETIME,
    OL_DATE D_DATE,
    OL_MEMO D_STR_LONG,
    REG_NAME D_STR_CODE,
    K_KONTRACT_NO D_STR_20,
    C_NAME_FIO D_STR_100,
    UN_FIO D_STR_100,
    STATUS D_STR_CODE)
AS
begin
  dtbeg = coalesce(dtbeg, current_date - 1);
  dtend = coalesce(dtend, current_date);

  for
    select
      con_id,
      con_ol_id,
      con_reg_id,
      con_k_id,
      con_c_id,
      con_cca_id,
      con_summa,
      con_summaperc,
      con_cs_id1,
      con_cs_id2,
      con_cs_id3,
      con_status,
      con_un_id,
      con_datecreate,
      con_dateupdate,
      ol_date,
      ol_memo,
      reg_name,
      k_kontract_no,
      c_name_fio,
      un_fio
    from cardoper_annul
      inner join operlist
        on ol_id = con_ol_id
      inner join region
        on reg_unicode = con_reg_id
      inner join kontract
        on k_id = con_k_id
      inner join clients
        on c_id = con_c_id
      left join cardsync as cs1
        on cs1.cs_id = con_cs_id1
      left join cardsync as cs2
        on cs2.cs_id = con_cs_id2
      left join cardsync as cs3
        on cs3.cs_id = con_cs_id3
      inner join username
        on un_id = con_un_id
    where con_datecreate between :dtbeg and :dtend
  into
    :con_id,
    :con_ol_id,
    :con_reg_id,
    :con_k_id,
    :con_c_id,
    :con_cca_id,
    :con_summa,
    :con_summaperc,
    :con_cs_id1,
    :con_cs_id2,
    :con_cs_id3,
    :con_status,
    :con_un_id,
    :con_datecreate,
    :con_dateupdate,
    :ol_date,
    :ol_memo,
    :reg_name,
    :k_kontract_no,
    :c_name_fio,
    :un_fio
  do
  begin
    status =
      case con_status
        when 0
          then 'создана'
        when 1
          then 'отправлена в МПЦ'
        when 2
          then 'сумма аннулирована в МПЦ'
        when 3
          then 'сумма % и НСП аннулирована в МПЦ'
        when 4
          then 'лимит изменен в МПЦ'
        when 5
          then 'аннулирована'
        when -1
          then 'ошибка аннулирования в FS'
      end;
    suspend;
  end
end
