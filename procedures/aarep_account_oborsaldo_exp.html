CREATE PROCEDURE AAREP_ACCOUNT_OBORSALDO_EXP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    ACCOUNT VARCHAR(20),
    ACCOUNT_NAME VARCHAR(255),
    ACC_BALANCE_NO VARCHAR(20),
    ACC_BALANCE VARCHAR(255),
    ACC_GROUP VARCHAR(255),
    ACC_CLASS VARCHAR(255),
    CURR_CODE VARCHAR(3),
    BEG_DT DECIMAL(18,4),
    BEG_KT DECIMAL(18,4),
    OB_DT DECIMAL(18,4),
    OB_KT DECIMAL(18,4),
    END_DT DECIMAL(18,4),
    END_KT DECIMAL(18,4),
    LAST_OPER_DATE DATE,
    ACC_GROUP_NO VARCHAR(20),
    ACC_KIND INTEGER,
    ACC_KIND_NAME VARCHAR(255))
AS
declare variable summa decimal(18,4);
begin
  for select account, account_name, acc_balance_no, acc_balance, acc_group, acc_class, curr_code, acc_group_no,
             acc_kind, acc_kind_name, sum(ob_dt), sum(ob_kt), sum(end_dt), sum(end_kt), max(last_oper_date)
      from aarep_account_oborsaldo_temp(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency), balgroup
      where bg_id = acc_balance_no and
            ((bg_kind = 4) or
            (bg_kind = 5))
      group by account, account_name, acc_balance_no, acc_balance, acc_group, acc_class, curr_code, acc_group_no, acc_kind, acc_kind_name
      order by curr_code, acc_balance_no, account
      into :account, :account_name, :acc_balance_no, :acc_balance, :acc_group, :acc_class, :curr_code, :acc_group_no,
           :acc_kind, :acc_kind_name, :ob_dt, :ob_kt, :end_dt, :end_kt, :last_oper_date
  do
  begin
    beg_dt = 0;
    beg_kt = 0;
    summa = (-ob_dt + ob_kt + end_dt - end_kt);
    if (summa >= 0) then
      beg_dt = summa;
    else
      beg_kt = -1 * summa;
    suspend;
  end
end
