CREATE PROCEDURE UXP$UXGRID_ITEMS(
    UX_UUID D_UUID,
    URT_UUID D_UUID,
    IS_READONLY D_BOOL)
RETURNS (
    UFI_UUID D_UUID,
    UFI_PARENT_UUID D_UUID,
    UFI_UX_CODE D_STR_100,
    UFI_UFR_UUID D_UUID,
    UFI_UFR_CODE D_STR_100,
    UFI_UF_UUID D_UUID,
    UFI_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_NAME D_STR_100,
    UFI_CAPTION_ALIGN D_INTEGER,
    UFI_HINT D_STR_LONG,
    UFI_HIDDEN D_SMALLINT,
    UFI_READONLY D_BOOL,
    UFI_GROUP_DIRECTION D_INTEGER,
    UFI_H_ALIGN D_INTEGER,
    UFI_V_ALIGN D_INTEGER,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_SIZABLE D_BOOL,
    UFI_ORDER_INDEX D_SMALLINT,
    UFI_GROUP_INDEX D_SMALLINT,
    UFI_FIXED_KIND D_SMALLINT,
    UFI_SUMMARY_KIND D_SMALLINT,
    IMAGE_INDEX D_INTEGER,
    UF_IS_PARAM D_SMALLINT,
    UF_FIELD_NAME D_STR_CODE,
    UF_SOURCE D_STR_CODE,
    UF_NOT_NULL D_BOOL,
    UF_SECURED D_BOOL,
    UFI_PARENT_UUID_STR D_STR_CODE,
    UFI_BAND_INDEX D_INTEGER,
    UFI_HEADER_ALIGN D_SMALLINT,
    UFI_VALUE_ALIGN D_SMALLINT)
AS
declare variable res_ufi_uuid d_uuid;
declare variable res_ufi_readonly d_bool;
declare variable fr_uuid d_uuid;
declare variable ux_kind d_str_code;
declare variable ufi_band_id d_integer;
begin
  if (coalesce(ux_uuid, '') = '') then
    exit;

  if (is_readonly is null) then
    is_readonly = 0;

  ufi_ux_code = uxf$code(ux_uuid, 'fcode');
  ux_kind = uxf$code(ux_uuid, 'kind');

  ufi_header_align = 0;
  ufi_value_align = 0;

  if (ux_kind = 'uxunit') then
  begin
    ufi_ufr_uuid = null;
    uf_is_param = 0;
    uf_not_null = 0;
    uf_secured = 0;
    ufi_band_id = -1;

    for select uni_uuid, uni_parent_uuid, uni_parent_uuid_code, uni_readonly, uni_fieldname, uni_datasource,
               uni_kind, uni_control, uni_position, uni_caption, uni_caption_align,
               uni_hint, uni_hidden, uni_group_direction,
               uni_h_align, uni_v_align, uni_width, uni_height,
               uni_sizable, uni_order_index, uni_group_index, uni_fixed_kind, uni_summary_kind
    from uxp$unititem_tree(:ux_uuid, :urt_uuid, :urt_uuid, 1)
    into :ufi_uuid, :ufi_parent_uuid, :ufi_parent_uuid_str, :res_ufi_readonly, :uf_field_name, :uf_source,
         :ufi_kind, :ufi_control, :ufi_position, ufi_name, :ufi_caption_align,
         :ufi_hint, :ufi_hidden, :ufi_group_direction,
         :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height,
         :ufi_sizable, :ufi_order_index, :ufi_group_index, :ufi_fixed_kind, :ufi_summary_kind
    do
    begin
      ufi_readonly = res_ufi_readonly;

      ufi_uf_uuid = null;
      if (coalesce(uf_field_name, '') <> '' and coalesce(uf_source, '') <> '') then
        ufi_uf_uuid = f_md5_uuid(uf_source || '.' || uf_field_name);

      suspend;
    end
  end else
  if (ux_kind <> 'uxgrid') then -- Это не отчет, вызывается для построения контейнера из UxFrItem
  begin
    ufi_band_id = -1;

    for select res_ufi_uuid, res_parent_ufi, res_ufi_readonly
    from uxp$uxfritem_tree(:ux_uuid, null, 1, :is_readonly)
    into :res_ufi_uuid, :ufi_parent_uuid, :res_ufi_readonly
    do
    begin
      ufi_readonly = res_ufi_readonly;
      for select ufi_uuid, ufi_ufr_uuid, ufi_uf_uuid, ufi_kind, ufi_control,
                 ufi_position, ufi_name, ufi_caption_align, ufi_hint, ufi_hidden,
                 ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
                 coalesce(uf_is_param, 0), coalesce(uf_field_name, ''), coalesce(us_code, ''),
                 coalesce(uf_not_null, 0), coalesce(uf_secured, 0), ufi_sizable,
                 ufi_order_index, ufi_group_index, ufi_fixed_kind, ufi_summary_kind
          from ux$fritem
               left join ux$field on (ufi_uf_uuid = uf_uuid)
               left join ux$source on (uf_owner = us_uuid)
          where ufi_uuid = :res_ufi_uuid
          into :ufi_uuid, :ufi_ufr_uuid, :ufi_uf_uuid, :ufi_kind, :ufi_control,
               :ufi_position, :ufi_name, :ufi_caption_align, :ufi_hint, :ufi_hidden,
               :ufi_group_direction, :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height,
               :uf_is_param, :uf_field_name, :uf_source,
               :uf_not_null, :uf_secured, :ufi_sizable,
               :ufi_order_index, :ufi_group_index, :ufi_fixed_kind, :ufi_summary_kind
      do
      begin
        ufi_parent_uuid_str = uuid_to_char(ufi_parent_uuid);

        suspend;
      end
    end
  end else
  begin
    ufi_parent_uuid = ux_uuid;
    ufi_parent_uuid_str = uuid_to_char(ufi_parent_uuid);

    uf_not_null = 0;
    uf_secured = 0;
    ufi_position = 0;

    uf_is_param = 0;
    uf_field_name = '';
    uf_source = '';
    uf_not_null = 0;
    uf_secured = 0;
    ufi_uf_uuid = null;
    ufi_kind = -1;
    ufi_control = 'grid';
    ufi_caption_align = -1; -- Скрытый

    ufi_group_direction = -1; -- Не группа
    ufi_h_align = 3; -- На всю ширину
    ufi_v_align = 3; -- На всю высоту
    uf_secured = 0;
    ufi_readonly = is_readonly;
    image_index = -1;

    select ufr_uuid, ufr_fcode, us_code
    from ux$grid, ux$frame, ux$source
    where ugr_uuid = :ux_uuid and
          ugr_ufr_uuid = ufr_uuid and
          ufr_us_uuid = us_uuid
    into ufi_ufr_uuid, ufi_ufr_code, uf_source;

    fr_uuid = ufi_ufr_uuid;

    for select ugr_uuid, ugr_name, ugr_memo, ugr_inactive
    from ux$grid, ux$frame, ux$form, ux$code
    where ugr_uuid = :ux_uuid and
          ugr_inactive = 0 and
          ugr_ufr_uuid = ufr_uuid and
          ufr_owner = ufm_uuid and
          ufm_ux_uuid = ux_uuid
    into :ufi_uuid, :ufi_name, ufi_hint, ufi_hidden
    do
      suspend;

    if (row_count = 0) then
      exception e_message f_format('Не найден шаблон отчета: %1', uuid_to_char(ux_uuid));

    ufi_parent_uuid = ufi_uuid;

    ufi_kind = 0;
    ufi_h_align = 4; -- По умолчанию
    ufi_v_align = 4; -- По умолчанию
    ufi_height = 0;
    uf_not_null = 0;
    uf_secured = 0;

    for select ugi_uuid, ugi_uf_uuid, ugi_position, ugi_width, coalesce(ugi_bnd_id, -1),
               ugi_sizable, ugi_order_index, ugi_group_index, ugi_fixed_kind, ugi_summary_kind,
               ugi_caption, ugi_hint, ugi_hidden, iif(:is_readonly = 1, 1, ugi_readonly),
               ugi_header_align, ugi_value_align
    from ux$gritem
    where ugi_ugr_uuid = :ux_uuid
    order by ugi_position
    into :ufi_uuid, :ufi_uf_uuid, :ufi_position, :ufi_width, :ufi_band_id,
         :ufi_sizable, :ufi_order_index, :ufi_group_index, :ufi_fixed_kind, :ufi_summary_kind,
         :ufi_name, :ufi_hint, :ufi_hidden, :ufi_readonly,
         :ufi_header_align, :ufi_value_align
    do
    begin
      ufi_band_index = -1;
      if (ufi_band_id >= 0) then
      begin
        select bnd_index
        from ux$band
        where bnd_grid_uuid = :ux_uuid and
              bnd_id = :ufi_band_id
        into ufi_band_index;
      end

      for select ufp_editor, uf_field_name, iif(uf_state = -1, 1, :ufi_hidden)
      from ux$frparam, ux$field
      where ufp_ufr_uuid = :fr_uuid and
            ufp_uf_uuid = :ufi_uf_uuid and
            ufp_uf_uuid = uf_uuid
      into :ufi_control, uf_field_name, :ufi_hidden
      do
        suspend;
    end
  end
end
