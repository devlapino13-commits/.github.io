CREATE PROCEDURE AAREP_SB_REPAYS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SB_TYPE INTEGER)
RETURNS (
    OL_DATE DATE,
    REG_NAME VARCHAR(100),
    SB_NAME VARCHAR(255),
    SB_TEAM_NAME VARCHAR(255),
    DELINQ_REPAY_DAYS INTEGER,
    REPAY_OS DECIMAL(18,4),
    REPAY_PERC DECIMAL(18,4),
    REPAY_FINE DECIMAL(18,4),
    REPAY_ALL DECIMAL(18,4),
    REPAY_TAX_KC CURR_TYPE,
    REPAY_ALL_IN_TAX CURR_TYPE,
    K_KONTRACT_NO VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    K_STATE_STR VARCHAR(100),
    AAREP_DELINQ_DATE DATE,
    LAST_REPAY_DATE DATE,
    OL_DOC_NO INTEGER,
    OL_DOC_CLIENT_NAME VARCHAR(1024),
    SBA_DATE_BEGIN DATE,
    DELINQ_DAY_OF_TRNSFER INTEGER,
    KC_ID DECIMAL(18,0))
AS
declare variable k_id decimal(18,0);
declare variable koef double precision;
declare variable k_amount decimal(18,4);
declare variable kc_amount decimal(18,4);
declare variable repay_os1 decimal(18,4);
declare variable repay_perc1 decimal(18,4);
declare variable repay_fine1 decimal(18,4);
declare variable dd_begin date;
declare variable ol_id decimal(18,0);
declare variable cl_id decimal(18,0);
declare variable cc_name varchar(100);
declare variable cm_is_client_mode integer;
declare variable c_id id_type;
declare variable repay_tax curr_type;
declare variable sum_zadol curr_type;
declare variable oek_rd_id integer;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');

  -- operkred_fix
  select cm_is_client_mode
  from company
  into cm_is_client_mode;

  if (cm_is_client_mode = 0) then
  begin
    for with tabl1
        as (select ok_ol_date, ok_from_ol_id, ok_from_k_id, reg_code || ' ' || reg_name as reg_name, sbl_name,
                   sbl_team_name, sba_date_begin, ok_ol_date - ok_sh_date as delinq_day, ok_os, ok_perc, ok_fine,
                   0 as ox_amount_minus
            from operkred, region, sb_akt, sb_list, region_list(:rep_region_id, 1, -1, 1)
            where ok_ol_date between :rep_date_begin and :rep_date_end and
                  ok_ol_is_active = 1 and
                  ok_ol_reg_id = reg_unicode and
                  sba_from_k_id = ok_from_k_id and
                  sba_is_delete = 0 and
                  sba_from_sbl_id <> 0 and
                  sba_date_begin <= ok_ol_date and
                  sba_date_end >= ok_ol_date and
                  sba_status = 2 and
                  sba_from_sbl_id = sbl_id and
                  sbl_type = :rep_sb_type and
                  ok_type = 4 and
                  ok_ol_reg_id = id_list and
                  sba_date_begin <= sba_date_end -- отсеиваю лишние записи
                  and
                  ok_from_c_id = 0
            union all
            select ox_ol_date, ox_from_ol_id, ox_object_id, reg_code || ' ' || reg_name, sbl_name, sbl_team_name,
                   sba_date_begin, 0, 0, 0, 0, ox_amount_minus
            from operextra, region, sb_akt, sb_list, region_list(:rep_region_id, 1, -1, 1)
            where ox_from_rd_id = :oek_rd_id and
                  ox_object_kind = 9 and
                  ox_ol_is_active = 1 and
                  ox_ol_date between :rep_date_begin and :rep_date_end and
                  ox_amount_minus > 0 and
                  ox_from_reg_id = reg_unicode and
                  reg_unicode = id_list and
                  ox_object_id = sba_from_k_id and
                  sba_is_delete = 0 and
                  sba_from_sbl_id <> 0 and
                  sba_date_begin <= ox_ol_date and
                  sba_date_end >= ox_ol_date and
                  sba_status = 2 and
                  sba_from_sbl_id = sbl_id and
                  sbl_type = :rep_sb_type and
                  sba_date_begin <= sba_date_end
            union all

            select oko_ol_date, oko_from_ol_id, oko_from_k_id, reg_code || ' ' || reg_name as reg_name, sbl_name,
                   sbl_team_name, sba_date_begin, 0 as delinq_day, oko_os, oko_perc, oko_other, 0 as ox_amount_minus
            from operkroff, region, sb_akt, sb_list, region_list(:rep_region_id, 1, -1, 1), operlist
            where oko_ol_date between :rep_date_begin and :rep_date_end and
                  oko_ol_is_active = 1 and
                  oko_from_ol_id = ol_id and
                  operlist.is_delete = 0 and
                  ol_from_reg_id = reg_unicode and
                  sba_from_k_id = oko_from_k_id and
                  sba_is_delete = 0 and
                  sba_from_sbl_id <> 0 and
                  sba_date_begin <= oko_ol_date and
                  sba_date_end >= oko_ol_date and
                  sba_status = 2 and
                  sba_from_sbl_id = sbl_id and
                  sbl_type = :rep_sb_type and
                  oko_type = 4 and
                  reg_unicode = id_list and
                  sba_date_begin <= sba_date_end -- отсеиваю лишние записи
                  --        and ok_from_c_id=0
        )
        select ok_ol_date, ok_from_ol_id, ok_from_k_id, reg_name, sbl_name, sbl_team_name, sba_date_begin,
               max(delinq_day), sum(-ok_os), sum(-ok_perc), sum(-ok_fine), sum(ox_amount_minus)
        from tabl1
        group by ok_ol_date, ok_from_ol_id, ok_from_k_id, reg_name, sbl_name, sbl_team_name, sba_date_begin
        having sum(-ok_os - ok_perc - ok_fine) <> 0 or sum(ox_amount_minus) <> 0
        into :ol_date, :ol_id, :k_id, :reg_name, :sb_name, sb_team_name, sba_date_begin, delinq_repay_days, repay_os1,
             repay_perc1, repay_fine1, repay_tax
    do
    begin
      select ol_doc_no, ol_document_client, ol_doc_client_name
      from operlist
      where ol_id = :ol_id
      into :ol_doc_no, :cl_id, :ol_doc_client_name;
      if (ol_doc_client_name is null) then
        ol_doc_client_name = '';

      cc_name = '';
      if (cl_id is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :cl_id
        into :cc_name;
      end
      if (ol_doc_client_name <> '') then
      begin
        if (cc_name <> '') then
          ol_doc_client_name = cc_name || ' через ' || ol_doc_client_name;
      end
      else
      if (cc_name <> '') then
        ol_doc_client_name = cc_name;

      k_state_str = 'Закрытый';
      aarep_delinq_date = null;
      dd_begin = null;
      select oks_date_begin, oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_close_state = 0
      into :dd_begin, :aarep_delinq_date;
      if (dd_begin is not null) then
      begin
        k_state_str = 'Активный (без просрочки)';
        if (aarep_delinq_date > :rep_date_end) then
          aarep_delinq_date = null;
        else
          k_state_str = 'Активный (просроченный)';
      end
      else
      begin
        sum_zadol = null;
        select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
        from operkroff
        where oko_from_k_id = :k_id and
              oko_ol_is_active = 1 and
              oko_ol_date <= :rep_date_end
        into sum_zadol;
        sum_zadol = coalesce(sum_zadol, 0);
        if (sum_zadol > 0) then
        begin
          k_state_str = 'Активный (на внесистемном учете)';
        end

      end
      begin
        sum_zadol = null;
        select sum(ox.ox_amount_plus - ox.ox_amount_minus - ox.ox_amount_writeoff) as debtamountforstateduty
        from operextra ox
        where ox_ol_is_active = 1
           and ox.ox_object_kind = 9
           and ox_object_id = :k_id
        into sum_zadol;
        sum_zadol = coalesce(sum_zadol, 0);
        if (sum_zadol > 0) then
        begin
          k_state_str = 'Активный (спец учет)';
        end
      end

      --  Кол-во просроченных дней на дату передачи коллекторам
      delinq_day_of_trnsfer = null;
      select :sba_date_begin - oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            :sba_date_begin between oks_date_begin and oks_date_end and
            :sba_date_begin > oks_delinq_date and
            oks_close_state = 0
      into delinq_day_of_trnsfer;

      select ok_ol_date
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_type = 4 and
            ok_from_c_id = 0
      order by ok_from_k_id desc, ok_ol_date desc
      rows 1
      into :last_repay_date;

      for select k_kontract_no, c_name_fio, k_amount_approved, kc_amount_approved, kc_id
          from knclient, kontract, clients
          where kc_from_k_id = k_id and
                k_id = :k_id and
                kc_from_c_id = c_id and
                kc_amount_approved > 0
          into :k_kontract_no, :c_name_fio, :k_amount, :kc_amount, :kc_id
      do
      begin
        koef = cast(kc_amount as double precision) / k_amount;
        repay_os = cast(repay_os1 * koef as decimal(15,2));
        repay_perc = cast(repay_perc1 * koef as decimal(15,2));
        repay_fine = cast(repay_fine1 * koef as decimal(15,2));
        repay_all = repay_os + repay_perc + repay_fine;
        repay_tax_kc = round(koef * repay_tax, 2);
        repay_all_in_tax = repay_all + repay_tax_kc;

        suspend;
      end

    end
    exit;
  end

  for select ok_ol_date, ok_from_ol_id, ok_from_k_id, reg_code || ' ' || reg_name, sbl_name, sbl_team_name,
             sba_date_begin, max(ok_ol_date - ok_sh_date), sum(-ok_os), sum(-ok_perc), sum(-ok_fine), ok_from_c_id,
             k_kontract_no
      from operkred, region, sb_akt, sb_list, region_list(:rep_region_id, 1, -1, 1), kontract
      where ok_ol_date between :rep_date_begin and :rep_date_end and
            ok_ol_is_active = 1 and
            ok_ol_reg_id = reg_unicode and
            sba_from_k_id = ok_from_k_id and
            sba_is_delete = 0 and
            sba_from_sbl_id <> 0 and
            sba_date_begin <= ok_ol_date and
            sba_date_end >= ok_ol_date and
            sba_status = 2 and
            sba_from_sbl_id = sbl_id and
            sbl_type = :rep_sb_type and
            ok_type = 4 and
            ok_ol_reg_id = id_list and
            sba_date_begin <= sba_date_end -- отсеиваю лишние записи
            and
            ok_from_c_id <> 0 and
            ok_from_k_id = k_id
      group by ok_ol_date, ok_from_ol_id, ok_from_k_id, reg_code || ' ' || reg_name, sbl_name, sbl_team_name, sba_date_begin, ok_from_c_id, k_kontract_no
      having sum(-ok_os - ok_perc - ok_fine) <> 0
      into :ol_date, :ol_id, :k_id, :reg_name, :sb_name, sb_team_name, sba_date_begin, delinq_repay_days, repay_os1,
           repay_perc1, repay_fine1, c_id, k_kontract_no
  do
  begin
    select ol_doc_no, ol_document_client, ol_doc_client_name
    from operlist
    where ol_id = :ol_id
    into :ol_doc_no, :cl_id, :ol_doc_client_name;
    if (ol_doc_client_name is null) then
      ol_doc_client_name = '';

    cc_name = '';
    if (cl_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :cl_id
      into :cc_name;
    end
    if (ol_doc_client_name <> '') then
    begin
      if (cc_name <> '') then
        ol_doc_client_name = cc_name || ' через ' || ol_doc_client_name;
    end
    else
    if (cc_name <> '') then
      ol_doc_client_name = cc_name;

    k_state_str = 'Закрыт';
    aarep_delinq_date = null;
    dd_begin = null;
    select oks_date_begin, oks_delinq_date
    from oksummary
    where oks_from_k_id = :k_id and
          oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_close_state = 0
    into :dd_begin, :aarep_delinq_date;
    if (dd_begin is not null) then
    begin
      k_state_str = 'Активный (без просрочки)';
      if (aarep_delinq_date > :rep_date_end) then
        aarep_delinq_date = null;
      else
        k_state_str = 'Активный (просроченный)';
    end
    else
    begin
      sum_zadol = null;
      select sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc + oko_other + oko_sh_other)
      from operkroff
      where oko_from_k_id = :k_id and
            oko_ol_is_active = 1 and
            oko_ol_date <= :rep_date_end
      into sum_zadol;
      sum_zadol = coalesce(sum_zadol, 0);
      if (sum_zadol > 0) then
      begin
        k_state_str = 'Активный (на внесистемном учете)';
      end
    end

    sum_zadol = null;
    select sum(ox.ox_amount_plus - ox.ox_amount_minus - ox.ox_amount_writeoff) as debtamountforstateduty
    from operextra ox
        where ox_ol_is_active = 1
          and ox.ox_object_kind = 9
          and ox_object_id = :k_id
    into sum_zadol;
    sum_zadol = coalesce(sum_zadol, 0);
    if (sum_zadol > 0) then
    begin
      k_state_str = 'Активный (спец учет)';
    end

    --  Кол-во просроченных дней на дату передачи коллекторам
    delinq_day_of_trnsfer = null;
    select :sba_date_begin - oks_delinq_date
    from oksummary
    where oks_from_k_id = :k_id and
          :sba_date_begin between oks_date_begin and oks_date_end and
          :sba_date_begin > oks_delinq_date and
          oks_close_state = 0
    into delinq_day_of_trnsfer;

    select ok_ol_date
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_from_c_id = 0
    order by ok_from_k_id desc, ok_ol_date desc
    rows 1
    into :last_repay_date;

    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    repay_os = repay_os1;
    repay_perc = repay_perc1;
    repay_fine = repay_fine1;
    repay_all = repay_os + repay_perc + repay_fine;
    /*
        for
        select k_kontract_no,
               c_name_fio,
               k_amount_approved,
               kc_amount_approved
        from knclient, kontract, clients
        where kc_from_k_id=k_id
        and k_id=:k_id
        and kc_from_c_id=c_id
        and kc_amount_approved>0
        into :k_kontract_no,
             :c_name_fio,
             :k_amount,
             :kc_amount
        do
        begin
          koef=kc_amount/k_amount;
          repay_os=cast(repay_os1*koef as decimal(15,2));
          repay_perc=cast(repay_perc1*koef as decimal(15,2));
          repay_fine=cast(repay_fine1*koef as decimal(15,2));
          REPAY_ALL=repay_os+repay_perc+repay_fine;
          Suspend;
        end
        */

    suspend;
  end
end
