CREATE PROCEDURE AAREP_ZALOG_INFO(
    REP_REGION_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    K_KONTRACT_NO VARCHAR(100),
    K_AMOUNT_APPROVED CURR_TYPE,
    KC_AMOUNT_APPROVED CURR_TYPE,
    K_BALANCE CURR_TYPE,
    K_FACT_DISB_DATE DATE,
    K_FACT_CLOSED_DATE DATE,
    REG_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    P_NAME_FIO VARCHAR(1024),
    KC_LOAN_TARGET_STR VARCHAR(255),
    Z_TYPE_TEXT VARCHAR(50),
    Z_SUBTYPE_TEXT VARCHAR(50),
    Z_NAME VARCHAR(100),
    Z_MEMO2 VARCHAR(2048),
    Z_COST_MARKET CURR_TYPE,
    Z_REGISTRY_DATE DATE,
    Z_REGISTRY_DATE_END DATE,
    Z_NAME_FIO VARCHAR(100),
    Z_ADDRESS VARCHAR(255),
    KS_STATE_TEXT VARCHAR(20),
    SH_CALC_PERC CURR_TYPE,
    PR_NAME VARCHAR(100),
    Z_DATE_BEGIN DATE,
    Z_DATE_END DATE)
AS
declare variable c_id id_type;
declare variable k_id id_type;
declare variable k_id_temp id_type;
declare variable k_reg_id id_type;
declare variable kc_loan_target integer;
declare variable z_type integer;
declare variable z_subtype integer;
declare variable ks_state integer;
declare variable z_from_adl_id id_type;
declare variable p_name_fio_tmp varchar(100);
declare variable pr_id id_type;
begin
--Получаем все активные и закрыте кредиты на отчетную дату
  for select ks_from_k_id, k_last_reg_id, ks_state, iif(ks_state = 2, ks_date_begin, null), k_kontract_no,
             k_amount_approved, k.k_fact_disb_date, k_from_pr_id
      from kontstate ks, kontract k, region_list(:rep_region_id, 1, -1, 1)
      where ks_ol_is_delete = 0 and
            ks_state in (1, 2) and
            :rep_date between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k.k_last_reg_id = id_list
      into k_id, k_reg_id, ks_state, k_fact_closed_date, k_kontract_no, k_amount_approved, k_fact_disb_date, pr_id
  do
  begin
--Получаем только те контракты где есть залог
    z_type = null;
    z_subtype = null;
    z_name = null;
    z_memo2 = null;
    z_cost_market = null;
    z_name_fio = null;
    z_from_adl_id = null;
    z_date_begin = null;
    z_date_end = null;

    for select z.z_type, z.z_subtype, z.z_name, z.z_memo2, z.z_cost_market, z.z_name_fio, z.z_from_adl_id,
               z.z_date_begin, z.z_date_end
        from zalog z
        where z.z_from_k_id = :k_id and
              z_type in (1, 2)
        into z_type, z_subtype, z_name, z_memo2, z_cost_market, z_name_fio, z_from_adl_id, z_date_begin, z_date_end
    do
    begin
--Адрес залога
      select al.adl_address_str
      from addrlist al
      where al.adl_id = :z_from_adl_id
      into :z_address;
      z_address = coalesce(z_address, 'Не указан');

--Тип залога
      if (z_type = 1) then
        z_type_text = 'Движимое имущество';
      else
        z_type_text = 'Недвижимое имущество';

--Суб.тип залога
      z_subtype_text = 'Не указан';
      select zst.zs_name
      from zalogsubtype zst
      where zst.zs_from_zt_id = :z_type and
            zst.zs_id = :z_subtype
      into z_subtype_text;

--Дата регистрации и снятия залога(всегда пусто, так как еще не автоматизирован в Финстарте)
      z_registry_date = null;
      z_registry_date_end = null;

--Информация по кредиту
      if (coalesce(k_id_temp, 0) <> k_id) then
      begin
        k_id_temp = k_id;

--Статус кредита
        ks_state_text = case ks_state
                          when 1 then 'Активный кредит'
                          when 2 then 'Закрытий кредит'
                        end;

--Баланс кредита, начисленные проценты
        k_balance = 0;
        sh_calc_perc = 0;
        select sum(zad_os), sum(sh_calc_perc)
        from aarep_make_zad_list(:k_id, :rep_date, 1)
        where sh_date is not null
        into k_balance, sh_calc_perc;

--product
        pr_name = null;
        select pr_name
        from product
        where pr_id = :pr_id
        into pr_name;

--Название офиса
        reg_name = null;
        select reg_name
        from region reg
        where reg_unicode = :k_reg_id
        into reg_name;

--Поручители
        p_name_fio = null;
        for select z2.z_name_fio
            from zalog z2
            where z2.z_from_k_id = :k_id and
                  z2.z_type = 0
            into p_name_fio_tmp
        do
        begin
          p_name_fio = coalesce(p_name_fio || '; ', '') || p_name_fio_tmp;
          p_name_fio_tmp = null;
        end

        p_name_fio = coalesce(p_name_fio, 'Поручитель не указан');
      end

--ИНформация по клиентам
      c_id = null;
      kc_amount_approved = null;
      kc_loan_target = null;
      for select kc_from_c_id, kc_amount_approved, kc.kc_loan_target
          from knclient kc
          where kc_amount_approved > 0 and
                kc_from_k_id = :k_id
          into c_id, kc_amount_approved, kc_loan_target
      do
      begin
        c_name_fio = '';
        select c_name_fio
        from clients
        where c_id = :c_id
        into c_name_fio;

--Цель кредита
        kc_loan_target_str = '';
        select rd_name
        from refdata
        where rd_id = :kc_loan_target
        into :kc_loan_target_str;

        suspend;
      end
    end
  end
end
