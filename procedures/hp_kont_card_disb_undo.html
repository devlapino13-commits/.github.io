CREATE PROCEDURE HP_KONT_CARD_DISB_UNDO(
    K_ID D_ID,
    UN_ID D_ID,
    WITH_KONTRACT D_BOOL = 1)
RETURNS (
    ERR_TEXT D_STR_255)
AS
declare variable k_disb_kind d_integer;
declare variable esm_status d_integer;
declare variable disb_ol_id d_id;
begin
  -- Отмена выдачи (если была ошибка)
  -- Вызывается из Finstart, окно TKontCardDisbForm

  err_text = '';
  if (f_get_db_id() <> 81001) then
  begin
    err_text = 'Разрешено только для РФ';
    suspend;
    exit;
  end

  select k_disb_kind from kontract
  where k_id = :k_id
  into k_disb_kind;

  if (k_disb_kind is null) then
  begin
    err_text = f_format('Договор не найден id:%1', coalesce(k_id, 'NULL'));
    suspend;
    exit;
  end

  if (k_disb_kind not in (5, 7, 8)) then
  begin
    err_text = 'Разрешено только для выдач карту/СБП/расчетный счет';
    suspend;
    exit;
  end

  -- Проверяем статус контракта. Должен быть активный
  if (not exists(select *
                 from kontstate
                 where ks_from_k_id = :k_id and
                       ks_state = 1 and
                       ks_date_end >= '31.12.2999' and
                       ks_ol_is_delete = 0)) then
  begin
    err_text = 'Договор должен быть активным';
    suspend;
    exit;
  end

  -- Проверяем статус выплаты
  select esm_status
  from extsysmain em
  where esm_from_k_id = :k_id and
        esm_project in ('moneta_disb', 'wirebank_disb') and
        esm_action in ('CardDisb', 'PhoneDisb', 'AccDisb')
  order by esm_id desc
  rows 1
  into esm_status;

  if (esm_status is null) then
  begin
    err_text = 'Не найдена операция выплаты на карту/СБП/расчетный счет';
    suspend;
    exit;
  end

  if (esm_status <> -1) then
  begin
    err_text = 'Операция выплаты на карту/СБП/расчетный счет, должна быть в статусе Ошибка';
    suspend;
    exit;
  end

  -- Проверки пройдены, откатываем
  select coalesce(ol_parent_id, ol_id)
  from kontstate, operlist
  where ks_from_k_id = :k_id
  and ks_state = 1
  and ks_ol_is_delete = 0
  and ks_from_ol_id = ol_id
  into disb_ol_id;

  if (disb_ol_id is not null) then
  begin
    update operlist
    set
      is_delete = 1,
      ol_user_id = :un_id
    where ol_id = :disb_ol_id;
  end

  if (with_kontract = 1) then
  begin
    update kontract
    set k_status = 0,
        k_user_id = :un_id
    where k_id = :k_id and
          k_status <> 0;
  end
  suspend;
end
