CREATE PROCEDURE AAREP_UNDERWRITER_REPORT(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    RES_ACL_ID DECIMAL(18,0),
    RES_ACL_DATE DATE,
    RES_STATUS VARCHAR(100),
    RES_APPROVED INTEGER,
    RES_REJECTED INTEGER,
    RES_REFUSED INTEGER,
    RES_REFUSED_NEW INTEGER,
    RES_LOAN_STATUS VARCHAR(100),
    K_KONTRACT_NO VARCHAR(100),
    RES_MIN_DATETIME TIMESTAMP,
    RES_MAX_DATETIME TIMESTAMP,
    RES_REG_NAMEP VARCHAR(100),
    RES_REG_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    K_CLIENT_COUNT INTEGER,
    RES_DISB_DATE DATE,
    RES_LIFETIME_IN_MINUTE FLOAT,
    RES_MEMO VARCHAR(2048),
    KFOLDER_ID ID_TYPE,
    K_ID ID_TYPE,
    PRODUCT_ID INTEGER,
    K_AMOUNT CURR_TYPE,
    K_CIRCLE INTEGER,
    K_REFUND_TXT VARCHAR(20),
    STOP_FACT VARCHAR(2048),
    RES_FIRST_SOLUTION_DATETIME TIMESTAMP,
    RES_FIRST_SOLUTION_IN_MINUTE STR_20,
    RES_ALL_DATETIME STR_20,
    K_FACT_DISB_DATE D_DATE,
    K_DISB_DATE D_DATE,
    RES_SENDER D_STR_100,
    RES_LIFETIME D_STR_20)
AS
declare variable acl_status d_integer;
declare variable rd_id d_integer;
declare variable c_id d_id;
declare variable reg_id d_integer;
declare variable k_status d_integer;
declare variable ovs_add_value d_integer;
declare variable ovs_datetime d_datetime;
declare variable opr_id d_id;
declare variable opr_subject_id d_integer;
declare variable ovs_un_id d_id;
declare variable ovs_c_id d_id;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();

  for select acl_id, acl_date, acl_object_id, acl_status, acl_from_rd_id, acl_respons_c_id, acl_from_reg_id,
             round(cast(acl_lifetime_in_second as float) / 60.0, 1), un_fio,
             f_get_time(time'00:00', dateadd(second, acl_lifetime_in_second, time'00:00:00'))
      from actionlist, refdata, region_list(:rep_region_id, 1, -1, 1), username
      where acl_from_rd_id = rd_id and
            acl_date between :rep_date_begin and :rep_date_end and
            rd_main_code = 'actiongroup' and
            rd_str_id starting 'underwriter' and
            acl_dont_show_in_order = 0 and
            acl_object_kind = 9 and
            acl_from_reg_id = id_list and
            acl_from_un_id = un_id
      into res_acl_id, res_acl_date, k_id, acl_status, rd_id, c_id, reg_id, res_lifetime_in_minute, res_sender,
           res_lifetime
  do
  begin
    res_approved = 0;
    res_rejected = 0;
    res_refused = 0;
    res_status = '';
    res_loan_status = '';
    res_min_datetime = null;
    res_max_datetime = null;
    k_client_count = 0;
    res_disb_date = null;
    res_memo = null;
    res_first_solution_datetime = null;
    res_first_solution_in_minute = null;

    if (db_id <> 94401) then -- RFCAZ-433
      c_id = null;

    for select ovs_add_value, ovs_datetime, opr_id, opr_subject_id, un_id, un_from_c_id
        from objectprops, objectvalues, username
        where ovs_from_opr_id = opr_id and
              opr_object_kind = 9 and
              opr_object_id = :k_id and
              opr_subject_kind = -1 and
              ovs_is_active = 1 and
              ovs_from_un_id = un_id
              /*and OPR_SUBJECT_ID = :RD_ID --Muzaffar чтобы добавить другие комментарии КК*/
        order by ovs_from_opr_id, ovs_id
        into :ovs_add_value, ovs_datetime, opr_id, opr_subject_id, ovs_un_id, ovs_c_id
    do
    begin
      if (opr_subject_id = rd_id) then
      begin
        if (ovs_add_value in (-2,-1,2) and cast(ovs_datetime as date) between rep_date_begin and rep_date_end) then
        begin
          -- Берем последнего пользователя принимавшего решение за период
          if (exists(select has_access
           from hp_permission_access_check(:ovs_un_id, 'PermActions', 0, 'FinStart.RepUnderConsVisible')
           where has_access = 1)) then
            c_id = ovs_c_id;
        end
        if (res_min_datetime is null) then
          res_min_datetime = ovs_datetime;
        if (res_max_datetime is null) then
          res_max_datetime = ovs_datetime;

        if (res_min_datetime is not null and
            res_first_solution_datetime is null and
            ovs_add_value not in (0, 1)) then
        begin
          res_first_solution_datetime = ovs_datetime;
          res_first_solution_in_minute = f_get_time(res_min_datetime, ovs_datetime);
        end

        if (ovs_add_value = 2) then
          res_approved = 1;
        if (ovs_add_value = -1) then
          res_refused = 1;
        if (ovs_add_value = -2) then
          res_rejected = 1;

        if (res_refused = 1 or res_rejected = 1) then
        begin
          select f_get_verify_memo(obm_memo, 'reason_memo')
          from objectmemo
          where obm_from_opr_id = :opr_id
          order by obm_id desc
          rows 1
          into res_memo;
        end

        --S.K.
        if (exists(select *
                   from objectmemo
                   where obm_from_opr_id = :opr_id and
                         obm_memo starting '[Доработка: ' and
                         obm_memo containing ascii_char(2))) then
          res_refused_new = 1;
        else
          res_refused_new = 0;

        if (res_min_datetime > ovs_datetime) then
          res_min_datetime = ovs_datetime;
        if (res_max_datetime < ovs_datetime) then
          res_max_datetime = ovs_datetime;
      end
      else
      begin
        stop_fact = '';
        select f_get_verify_memo(left(list(obm_memo, '. Сл.текст: '), 2048), 'reason_memo')
        from objectmemo
        where obm_from_opr_id = :opr_id
        into stop_fact;
      end
    end
    if (res_approved = 1 or res_rejected = 1 or res_refused = 1) then
    begin
      select k_kontract_no, k_status, k_kolich_client_approved, k_from_kf_id, k_from_pr_id, k_amount_approved, k_circle,
             iif(k_refund = 1, 'Перекредитование', ''), k_fact_disb_date, k_disb_date
      from kontract k
      where k_id = :k_id
      into :k_kontract_no, :k_status, :k_client_count, :kfolder_id, :product_id, :k_amount, :k_circle, k_refund_txt,
           :k_fact_disb_date, k_disb_date;

      if (acl_status = -2) then
        res_status = 'Отказан';
      if (acl_status = -1) then
        res_status = 'Доработка';
      if (acl_status = 0) then
        res_status = 'В очереди';
      if (acl_status = 1) then
        res_status = 'В процессе';
      if (acl_status = 2) then
        res_status = 'Одобрен';

      select ks_date_begin
      from kontstate
      where ks_from_k_id = :k_id and
            ks_state = 1 and
            ks_ol_is_delete = 0
      into :res_disb_date;

      if (res_disb_date is not null) then
        res_loan_status = 'Выдан';
      else
      begin
        if (k_status = -1) then
          res_loan_status = 'Отказан';
        if (k_status = -2) then
          res_loan_status = 'Заявка';
        if (k_status = 0) then
          res_loan_status = 'В процессе';
        if (k_status = 1) then
          res_loan_status = 'Утвержден';
      end

      select rc.reg_code || ' ' || rc.reg_name, rp.reg_name
      from region as rc
      left join region as rp on rp.reg_unicode = rc.parent
      where rc.reg_unicode = :reg_id
      into :res_reg_name, :res_reg_namep;

      c_name_fio = '';

      select c_name_fio
      from clients
      where c_id = :c_id
      into :c_name_fio;

      res_all_datetime = null;
      if (res_min_datetime is not null and
          res_max_datetime is not null) then
      begin
        res_all_datetime = f_get_time(res_min_datetime, res_max_datetime);
      end

      suspend;
    end
  end
end
