CREATE PROCEDURE HP_CASHOPER_CASHINVLIST(
    REP_OFFICE_ID D_INTEGER,
    REP_DATE D_DATE,
    DOC_STATUS D_SMALLINT)
RETURNS (
    CIL_ID D_INTEGER,
    CIL_FROM_CSP_ID D_INTEGER,
    CIL_DATE D_DATE,
    CIL_CASHIER_ID D_ID,
    CIL_INSPECTOR_ID D_ID,
    CIL_NEED_BONDS D_BOOL,
    CIL_INV_AMOUNT D_CURRENCY,
    CIL_DATETIME D_DATETIME,
    CIL_FROM_UN_ID D_ID,
    CIL_FROM_EDL_ID D_ID,
    CIL_INV_BLS_ID D_ID,
    CIL_FACT_BLS_ID D_ID,
    CIL_STATE D_SMALLINT,
    CIL_IS_CLOSED D_BOOL,
    CIL_IS_DELETE D_BOOL,
    CASHIER_FIO D_STR_100,
    INSPECTOR_FIO D_STR_100,
    CIL_STATE_STR D_STR_CODE,
    USER_NAME D_STR_CODE,
    CIL_KIND_STR D_STR_CODE,
    CTM_DOC_NO D_INTEGER,
    CTM_TERMINAL_ID D_STR_100,
    CTM_PERIOD_BEGIN D_DATETIME,
    CTM_PERIOD_END D_DATETIME,
    CTM_AMOUNT_CLS D_CURRENCY,
    CTM_AMOUNT_IPF D_CURRENCY,
    CTM_AMOUNT_OTHER D_CURRENCY,
    CALC_AMOUNT_TOTAL D_CURRENCY,
    INV_CASH_NAME D_STR_CODE,
    FACT_CASH_NAME D_STR_CODE,
    FACT_DATE D_DATE,
    FACT_AMOUNT D_CURRENCY,
    FACT_CIL_ID D_INTEGER,
    INV_CIL_ID D_INTEGER,
    RES_IS_TERMINAL D_BOOL,
    RES_IS_FACT D_BOOL)
AS
declare variable edl_blank_status d_smallint;
declare variable cil_parent_id d_integer;
declare variable cil_office_id d_integer;
declare variable cil_oper_kind d_integer;
declare variable oper_code d_str_code;
begin
  if (rep_date is null) then
    rep_date = current_date;

  if (doc_status is null) then
    doc_status = 0;

  for select cil_id,
             cil_date,
             cil_parent_id,
             cil_from_csp_id,
             cil_oper_kind,
             cil_office_id,
             cil_cashier_id,
             cil_inspector_id,
             cil_need_bonds,
             cil_inv_amount,
             cil_datetime,
             cil_from_un_id,
             cil_from_edl_id,
             cil_from_bls_id,
             cil_state,
             cil_is_closed,
             cil_is_delete,
             csh.c_name_fio,
             ins.c_name_fio,
             u.un_name,
             ed.edl_blank_status,
             trim(iif(csp_kind = 0, 'Касса', 'Расчет')) || ' ' || csp_cash_no
      from cashinvlist
           inner join cashplace on (cil_from_csp_id = csp_id and csp_office_id  = :rep_office_id)
           left join clients csh on (csh.c_id = cil_cashier_id)
           left join clients ins on (ins.c_id = cil_inspector_id)
           left join edoc_list ed on (ed.edl_id = cil_from_edl_id)
           left join username u on (u.un_id = cil_from_un_id)
      where cil_date between dateadd(day, -7, :rep_date) and :rep_date
      union
      select cil_id,
             cil_date,
             cil_parent_id,
             cil_from_csp_id,
             cil_oper_kind,
             cil_office_id,
             cil_cashier_id,
             cil_inspector_id,
             cil_need_bonds,
             cil_inv_amount,
             cil_datetime,
             cil_from_un_id,
             cil_from_edl_id,
             cil_from_bls_id,
             cil_state,
             cil_is_closed,
             cil_is_delete,
             csh.c_name_fio,
             ins.c_name_fio,
             u.un_name,
             ed.edl_blank_status,
             trim(iif(csp_kind = 0, 'Касса', 'Расчет')) || ' ' || csp_cash_no
      from cashinvlist
           inner join cashplace on (cil_from_csp_id = csp_id and csp_office_id  <> :rep_office_id)
           left join clients csh on (csh.c_id = cil_cashier_id)
           left join clients ins on (ins.c_id = cil_inspector_id)
           left join edoc_list ed on (ed.edl_id = cil_from_edl_id)
           left join username u on (u.un_id = cil_from_un_id)
      where cil_date between dateadd(day, -7, :rep_date) and :rep_date and
            cil_office_id = :rep_office_id
      order by 2 desc, 1 desc
      into :cil_id,
           :cil_date,
           :cil_parent_id,
           :cil_from_csp_id,
           :cil_oper_kind,
           :cil_office_id,
           :cil_cashier_id,
           :cil_inspector_id,
           :cil_need_bonds,
           :cil_inv_amount,
           :cil_datetime,
           :cil_from_un_id,
           :cil_from_edl_id,
           :cil_inv_bls_id,
           :cil_state,
           :cil_is_closed,
           :cil_is_delete,
           :cashier_fio,
           :inspector_fio,
           :user_name,
           :edl_blank_status,
           :inv_cash_name
  do
  begin
    cil_state_str = '-';
    res_is_fact = 0;
    if (cil_parent_id is not null) then
      res_is_fact = 1;

    if (doc_status = 1) then
    begin
      if (cil_is_delete = 1) then
        Continue;
    end else
    if (doc_status = -1) then
    begin
      if (not(cil_is_delete = 1)) then
        Continue;
    end

    if (cil_from_edl_id is not null) then
    begin
      cil_state_str = trim(case cil_state
        when -1 then 'Аннулирован'
        when 0 then 'Формируется'
        when 1 then 'Подписан'
      end);

      if (cil_state = 0 and edl_blank_status = 1) then
        cil_state_str = 'На подписании';
    end

    cil_kind_str = f_enum_name('cashoperkind', cil_oper_kind);
    res_is_terminal = 0;

    ctm_amount_cls = 0;
    ctm_amount_ipf = 0;
    ctm_amount_other = 0;

    ctm_doc_no = null;
    ctm_terminal_id = null;
    ctm_period_begin = null;
    ctm_period_end = null;
    cil_fact_bls_id = null;

    select ctm_doc_no, t.tm_terminal_id, ctm_period_begin, ctm_period_end,
           ctm_amount_cls, ctm_amount_ipf, ctm_amount_other
    from cashterminal
         inner join terminals t on (t.tm_id = ctm_from_tm_id)
    where ctm_from_cil_id = :cil_id
    into ctm_doc_no, ctm_terminal_id, ctm_period_begin, ctm_period_end,
         ctm_amount_cls, ctm_amount_ipf, ctm_amount_other;

    if (row_count > 0) then
    begin
      res_is_terminal = 1;
      cil_kind_str = 'Чек терминала';
    end

    calc_amount_total = ctm_amount_cls + ctm_amount_ipf + ctm_amount_other;

    fact_cash_name = '';
    fact_date = null;
    fact_amount = null;

    inv_cil_id = cil_id;
    fact_cil_id = null;

    if (cil_parent_id is not null) then
    begin
      if (exists(select * from cashinvlist
                 where cil_id = :cil_parent_id and
                       cil_office_id = :cil_office_id)) then
        Continue;

      cil_fact_bls_id = cil_inv_bls_id;
      select reg_code || ', ' || trim(iif(csp_kind = 0, 'Касса', 'Расчет')) || ' ' || csp_cash_no,
             cil_date, cil_inv_amount, cil_from_bls_id
      from cashinvlist
           inner join cashplace on (cil_from_csp_id = csp_id)
           inner join region on (csp_office_id = reg_unicode)
      where cil_id = :cil_parent_id
      into :fact_cash_name,
           :fact_date, :fact_amount, cil_inv_bls_id;

      inv_cil_id = cil_parent_id;
      fact_cil_id = cil_id;
    end else
    begin
      select cil_id, reg_code || ', ' || trim(iif(csp_kind = 0, 'Касса', 'Расчет')) || ' ' || csp_cash_no,
             cil_date, cil_inv_amount, cil_from_bls_id
      from cashinvlist
           inner join cashplace on (cil_from_csp_id = csp_id)
           inner join region on (csp_office_id = reg_unicode)
      where cil_parent_id = :cil_id and
            cil_is_delete = 0
      into :fact_cil_id, :fact_cash_name,
           :fact_date, :fact_amount, cil_fact_bls_id;
    end

    if (cil_date < rep_date) then
    begin
      if (fact_date < rep_date) then
        Continue;
      else
      begin
        oper_code = f_enum_code('cashoperkind', cil_oper_kind);

        if (oper_code not in ('minus.office', 'minus.to_kassa')) then
          Continue;
      end
    end

    suspend;
  end
end
