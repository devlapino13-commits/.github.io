CREATE PROCEDURE AAREP_QUANT_KD_PERCENT_IZM(
    K_ID DECIMAL(18,0),
    REP_DATE_END D_DATE)
RETURNS (
    QUANT_KD_PERCENT_IZM D_INTEGER)
AS
declare variable kd_ol_date_per_day d_date;
declare variable kd_percent_per_day perc_type;
declare variable kd_percent_per_day_old perc_type;
begin
    kd_ol_date_per_day=null;
    kd_percent_per_day=0;
    quant_kd_percent_izm=0;
    kd_percent_per_day_old=0;
  --Количество изменений ставки с даты выдачи до отчетной даты
       for select distinct kd_ol_date, kd_percent_per_day
        from kontdata
        where kd_ol_date <= :rep_date_end and
              kd_ol_is_delete = 0 and
              kd_from_k_id = :k_id and
              kd_from_ol_id is not null
       -- group by kd_ol_date
        into :kd_ol_date_per_day,:kd_percent_per_day
        do
        begin
         if (kd_percent_per_day<>kd_percent_per_day_old) then
         :quant_kd_percent_izm =quant_kd_percent_izm+1;
         :kd_percent_per_day_old=:kd_percent_per_day;
        end
  suspend;
end
