CREATE PROCEDURE HP_CLIENTSTATE_ORDER_HANDLE
AS
declare variable sql_text d_str_long;
declare variable c_id d_id;
declare variable field d_str_100;
declare variable state d_smallint;
begin

    for select
        cso_c_id,
        cso_upd_field,
        cso_upd_state
    from clientstate_order cso
    where cso.cso_handled = 0
    into
        :c_id,
        :field,
        :state
    do
    begin
        /*
            ПРОШУ ОЗНАКОМИТЬСЯ ДЛЯ ЛУЧШЕГО ПОНИМАНИЯ !!!

            Возможно скрипт ниже выгляжит трудно читаемым, но это единственный наиболее оптимальный, а также наикратчайший способ реализации,
            который не разведёт "лапша-кода" на пятьсот с лишним строк

            Пример полноценного запроса с уже подставленными параметрами
            update clientstate_ru
            set csr_edoc_asp_signed = 1
            where csr_from_c_id = 1010101173339
                 and csr_edoc_asp_signed <> 1;
    
            %1 - id клиента (c_id)
            %2 - названием одного из полей таблицы clientstate_ru (на вход поля проверяются в процедуре hp_clientstate_order_create)
            %3 - само состояние
        */


        sql_text = f_format(
        '
            update clientstate_ru
            set %2 = %3
            where csr_from_c_id = %1
                and %2 <> %3;
        ',
        :c_id, :field, :state);

        execute statement (:sql_text);


        -- Удаляем успешно обработанную задачу + все записи с идентичными значениями (дубликаты)
        delete from clientstate_order cso
        where cso.cso_c_id = :c_id
            and cso.cso_upd_field = :field
            and cso.cso_upd_state = :state;

        when gdscode deadlock do
        begin
            -- Прячем исключение при параллельном выполнении
            -- В таком случае задача будет обработана следующим кругом

        end
    
        when any do
        begin
            -- Помечаем запись как -1
            -- Когда таких записей накопиться 20 шт (на момент реализации), то придёт письмо программистам на почту, о неисправном механизме
            update clientstate_order cso
            set cso.cso_handled = -1
            where cso.cso_c_id = :c_id
                and cso.cso_upd_field = :field;

        end
    end
end
