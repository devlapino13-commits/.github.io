CREATE PROCEDURE AAREP_RATING_DELIQ_TEMP(
    REG_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    DK_REGION_ID INTEGER,
    DK_DISB_DATE DATE,
    DK_CURR_LO_ID DECIMAL(18,0),
    DK_DISB_LO_ID DECIMAL(18,0),
    DK_DELINQ_BALANCE DECIMAL(18,4),
    DK_DELINQ_CL_KOLICH INTEGER,
    DK_BALANCE_END DECIMAL(18,4),
    DK_CLIENT_KOL INTEGER,
    DK_DATE DATE)
AS
declare variable kont_curr_rate decimal(18,4);
declare variable dk_id decimal(18,0);
begin
  /* Procedure Text */
  for select kk_id, kk_region_id, kk_kc_id, kk_disb_kc_id, kk_disb_date, kk_curr_rate, kk_kolich_client,
             kk_delinq_balance, kk_delinq_date
      from aarep_active_kont_id_in_period(:reg_id, null, :rep_date_end, 0, 1, 1, 1, 0)
      into :dk_id, :dk_region_id, :dk_curr_lo_id, :dk_disb_lo_id, :dk_disb_date, :kont_curr_rate, :dk_delinq_cl_kolich,
           dk_delinq_balance, dk_date
  do
  begin
    dk_client_kol = dk_delinq_cl_kolich;
    dk_delinq_balance = dk_delinq_balance * kont_curr_rate;
    dk_balance_end = dk_delinq_balance;
    suspend;
  end

end
