CREATE PROCEDURE AAREP_CALC_AMORT_KOMMIS_OSALDO(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    AAREP_OFFICE_ID INTEGER,
    OB_OBOROT_DT DECIMAL(18,4),
    OB_OBOROT_KT DECIMAL(18,4),
    OB_SALDO_BEGIN DECIMAL(18,4),
    OB_SALDO_END DECIMAL(18,4),
    AAREP_ACCOUNT VARCHAR(20),
    AAREP_ACCOUNT_PROFIT VARCHAR(20),
    AAREP_ACCOUNT_PAYED VARCHAR(20),
    AAREP_ACCOUNT_TYPE INTEGER)
AS
declare variable ob_end_dt decimal(18,4);
declare variable ob_end_kt decimal(18,4);
begin
  for select distinct aarep_office_id, aarep_account_profit, aarep_account_payed
      from aarep_calc_amort_kommis_obook(:report_region_id, :report_date_begin, :report_date_end, 0)
      into :aarep_office_id, :aarep_account_profit, :aarep_account_payed
  do
  begin
    if (aarep_account_profit is not null) then
    begin
      aarep_account_type = 2;
      aarep_account = aarep_account_profit;
      ob_end_dt = 0;
      ob_end_kt = 0;
      ob_oborot_dt = 0;
      ob_oborot_kt = 0;
      ob_saldo_begin = 0;
      ob_saldo_end = 0;

      select sum(obs_dt_val), sum(obs_kt_val)
      from obsummary
      where obs_from_al_id = :aarep_account and
            obs_date between :report_date_begin and :report_date_end
      into :ob_oborot_dt, :ob_oborot_kt;
      if (ob_oborot_dt is null) then
        ob_oborot_dt = 0;
      if (ob_oborot_kt is null) then
        ob_oborot_kt = 0;

      ob_saldo_end = f_al_bal(aarep_account, report_date_end, 'val');

      ob_saldo_begin = ob_saldo_end - (ob_oborot_dt - ob_oborot_kt);
      suspend;
    end
    if (aarep_account_payed is not null) then
    begin
      aarep_account_type = 1;
      aarep_account = aarep_account_payed;
      ob_end_dt = 0;
      ob_end_kt = 0;
      ob_oborot_dt = 0;
      ob_oborot_kt = 0;
      ob_saldo_begin = 0;
      ob_saldo_end = 0;

      select sum(obs_dt_val), sum(obs_kt_val)
      from obsummary
      where obs_from_al_id = :aarep_account and
            obs_date between :report_date_begin and :report_date_end
      into :ob_oborot_dt, :ob_oborot_kt;
      if (ob_oborot_dt is null) then
        ob_oborot_dt = 0;
      if (ob_oborot_kt is null) then
        ob_oborot_kt = 0;

      ob_saldo_end = f_al_bal(aarep_account, report_date_end, 'val');
      ob_saldo_begin = ob_saldo_end - (ob_oborot_dt - ob_oborot_kt);
      suspend;
    end
  end
end
