CREATE PROCEDURE UXP$UXGRID_COPY_LIST(
    GRID_UUID D_UUID,
    SAME_FIELD_COUNT D_INTEGER = 3)
RETURNS (
    SRC_GRID_UUID D_UUID,
    RES_FRAME D_STR_255,
    RES_FORM D_STR_255,
    RES_KIND D_STR_CODE,
    RES_COUNT D_INTEGER,
    RES_FIELDS D_STR_LONG)
AS
declare variable us_uuid d_uuid;
declare variable ufr_add_table_fields d_str_code;
declare variable main_us_uuid d_uuid;
begin
  if (coalesce(same_field_count, 0) = 0) then
    same_field_count = 3;

  select ufr_us_uuid, ufr_add_table_fields
  from ux$grid, ux$frame
  where ugr_uuid = :grid_uuid and
        ugr_ufr_uuid = ufr_uuid
  into us_uuid, ufr_add_table_fields;

  if (ufr_add_table_fields <> '') then
  begin
    select us_uuid
    from ux$source
    where us_code = :ufr_add_table_fields
    into main_us_uuid;
  end

  if (main_us_uuid is null) then
    main_us_uuid = us_uuid;

  for select src.ugi_ugr_uuid, count(*)
  from ux$field fld, ux$gritem src, ux$field sf
  where fld.uf_owner in (:us_uuid, :main_us_uuid) and
        fld.uf_is_param = 0 and
        fld.uf_field_name = sf.uf_field_name and
        src.ugi_uf_uuid = sf.uf_uuid and
        src.ugi_ugr_uuid <> :grid_uuid
  group by 1
  into src_grid_uuid, res_count
  do
  begin
    if (res_count < same_field_count) then
      Continue;

    res_fields = '';
    for select ufr_name,
               iif(ufr_name = ux_name, ufr_fcode, left(ufr_fcode || '  - ' || ux_name, 255)),
               uxf$kind(uu_kind, 'caption')
    from ux$grid, ux$frame, ux$form, ux$code, ux$uniq
    where ugr_uuid = :src_grid_uuid and
          ugr_ufr_uuid = ufr_uuid and
          ufr_owner = ufm_uuid and
          ufm_ux_uuid = ux_uuid and
          (ux_draft_uuid = ufm_uuid or ux_release_uuid = ufm_uuid) and
          ux_uuid = uu_id
    into :res_frame, :res_form, :res_kind
    do
    begin
      select left(list(uf_field_name), 1020)
      from ux$gritem, ux$field
      where ugi_ugr_uuid = :src_grid_uuid and
            ugi_uf_uuid = uf_uuid
      into res_fields;

      if (char_length(res_fields) = 1020) then
        res_fields = res_fields || '...';

      suspend;
    end
  end

  res_kind = 'FinStart форма';
  for select par.uni_uuid, count(*)
  from ux$field fld, ux$unititem sf, ux$unititem par
  where fld.uf_owner = :us_uuid and
        fld.uf_is_param = 0 and
        fld.uf_field_name = sf.uni_fieldname and
        par.uni_uuid = sf.uni_parent_uuid and
        par.uni_name in ('TcxGridDBTableView', 'TcxGridDbBandedTableView', 'TcxDBTreeList')
  group by 1
  into src_grid_uuid, res_count
  do
  begin
    if (res_count < same_field_count) then
      Continue;

    if (src_grid_uuid = grid_uuid) then
      Continue;

    res_fields = '';
    for select uni.uni_datasource || '  - ' || uni.uni_code,
               itm.uni_code || '   - ' || itm.uni_caption
    from ux$unititem uni,ux$unititem itm
    where uni.uni_uuid = :src_grid_uuid and
          uni.uni_unt_uuid = itm.uni_uuid
    into :res_frame, :res_form
    do
    begin
      select left(list(uni_fieldname), 1020)
      from ux$unititem uni
      where uni.uni_parent_uuid = :src_grid_uuid
      into res_fields;

      if (char_length(res_fields) = 1020) then
        res_fields = res_fields || '...';

      suspend;
    end
  end
end
