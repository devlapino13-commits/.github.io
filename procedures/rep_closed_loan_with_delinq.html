CREATE PROCEDURE REP_CLOSED_LOAN_WITH_DELINQ(
    OFFICE_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    FIO_SVZ STR_100,
    KONTRACT_NO STR_100,
    DATE_END D_DATE,
    DEBT_OS CURR_TYPE,
    DEBT_PERC CURR_TYPE,
    DEBT_FINE CURR_TYPE,
    DEBT_OTHER CURR_TYPE)
AS
begin
    for select c.c_name_fio,
           k.k_kontract_no,
           k.k_fact_closed_date,
           sum(oks.oks_out_os),
           sum(oks.oks_out_perc),
           sum(oks.oks_out_other),
           sum(oks.oks_extra)
    from kontract k
    inner join clients c on c.c_id = k.k_fact_closed_lo_id 
    inner join oksummary oks on oks.oks_from_k_id = k.k_id
    inner join region_list(:office_id, 1, 3, 1) on id_list = oks.oks_region_id
    where oks.oks_close_state = 1
    and k.k_fact_closed_date between :rep_date_begin and :rep_date_end
    group by 1,2,3
    into fio_svz, kontract_no, date_end, debt_os, debt_perc, debt_fine, debt_other
    do
    begin
        suspend;
    end
end
