CREATE PROCEDURE SA_TABEL_SELECT(
    SA_REGION_ID INTEGER,
    SA_YEAR INTEGER,
    SA_MONTH INTEGER)
RETURNS (
    SAT_FROM_C_ID DECIMAL(18,0),
    SAT_OFFICE_ID INTEGER,
    SAT_TOTAL_NORMA INTEGER,
    SAT_TOTAL_WORKED INTEGER,
    SAT_TOTAL_OTPUSK INTEGER,
    SAT_TOTAL_BOLNICH INTEGER)
AS
begin
  for select sat_from_c_id, sat_office_id, sum(sat_total_norma), sum(sat_total_worked), sum(sat_total_otpusk),
             sum(sat_total_bolnich)
      from sa_tabel_select_temp(:sa_region_id, :sa_year, :sa_month)
      group by sat_from_c_id, sat_office_id
      into :sat_from_c_id, :sat_office_id, :sat_total_norma, :sat_total_worked, :sat_total_otpusk, :sat_total_bolnich
  do
  begin
    suspend;
  end
end
