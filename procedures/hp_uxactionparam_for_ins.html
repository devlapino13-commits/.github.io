CREATE PROCEDURE HP_UXACTIONPARAM_FOR_INS(
    UA_CODE D_STR_CODE)
RETURNS (
    ID D_STR_100,
    ID_VISIBLE D_STR_255,
    TEXT D_STR_255)
AS
declare variable ux_code d_str_code;
declare variable fr_code d_str_code;
declare variable fr_name d_str_100;
declare variable ufd_source d_str_100;
declare variable ufp_code d_str_100;
declare variable ufp_origin_name d_str_100;
declare variable ufi_caption d_str_100;
declare variable ux_kind d_str_code;
declare variable uxa_kind d_smallint;
begin
  select ux_code, ux_kind, uxa_kind
  from hp_get_uxalias(:ua_code)
  into :ux_code, :ux_kind, uxa_kind;

  if (uxf$kind(ux_kind, 'code') <> 'uxform' or uxa_kind = 4) then
    exit;

  for select ufi_ufr_code, ux_name
  from uxfritem, uxcode
  where ufi_ux_code = :ux_code and
        ufi_kind = 1 and -- frame
        ufi_ux_code = ux_code
  into fr_code, fr_name
  do
  begin
    for select ufd_source
    from uxfrdata
    where ufd_ufr_code = :fr_code and
          ufd_kind = 0
    into ufd_source
    do
    begin
      for select ufp_code, ufp_origin_name
      from uxfrparam
      where ufp_ufr_code = :fr_code and
            ufp_is_delete = 0 and
            ufp_in_use = 1 and
            ufp_is_param = 0
      into :ufp_code, ufp_origin_name
      do
      begin
        ufi_caption = '';
        select ufi_caption
        from uxfritem
        where ufi_ufp_code = :ufp_code
        into ufi_caption;

        -- Если поле отсутствует в интерфейсе, не отображаем в списке
        if (row_count = 0) then
          Continue;

        if (ufi_caption = '') then
          id_visible = '%1.%2';
        else
          id_visible = ufi_caption || ' (%1.%2)';

        id = ufp_code;
        id_visible = f_format(id_visible, ufd_source, ufp_origin_name);

        if (fr_code <> fr_name and not(fr_name containing fr_code)) then
          text = fr_code || '   - ' || fr_name;
        else
          text = fr_name;

        suspend;
      end
    end
  end
end
