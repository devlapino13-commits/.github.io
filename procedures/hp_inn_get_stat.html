CREATE PROCEDURE HP_INN_GET_STAT(
    DATE_BEGIN D_DATE)
RETURNS (
    CURR_REQ_TYPE D_SMALLINT,
    EIC_NAME D_STR_CODE,
    STATUS D_STR_CODE,
    CNT D_INTEGER,
    FIRST_CREATE_DATETIME D_DATETIME,
    LAST_CREATE_DATETIME D_DATETIME)
AS
declare variable eil_status d_integer;
begin
  curr_req_type = f_get_const('inn_ru_source');

  for select eic_name, eil_status, count(*), min(eil_create_datetime), max(eil_create_datetime)
  from extinfolist, extinfocfg
  where eic_id = eil_from_eic_id
    and eic_code in ('inn_gnivc','inn_smev')
    and eil_date >= :date_begin
  group by 1,2
  into eic_name, eil_status, cnt, first_create_datetime, last_create_datetime
  do
  begin
    status = case eil_status
                when -3 then 'Отменена из-за простоя'
                when -2 then 'Ошибка'
                when -1 then 'Временная ошибка'
                when 0  then 'В очереди для отправки'
                when 1  then 'Отправлен, в ожидании ответа'
                when 2  then 'Ответ получен'
                else 'Неизвестно'
             end;

    suspend;
  end 
end
