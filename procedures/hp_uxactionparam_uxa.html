CREATE PROCEDURE HP_UXACTIONPARAM_UXA(
    UA_CODE D_STR_100,
    UXA_CODE D_STR_CODE,
    PARAM_KIND D_SMALLINT)
RETURNS (
    UAP_UA_CODE D_STR_100,
    UAP_LINK_UFP_CODE D_STR_100,
    UAP_FROM_UFP_CODE D_STR_100,
    UAP_KIND D_SMALLINT,
    UAP_DEFAULT D_STR_255,
    UAP_USE_DEFAULT D_SMALLINT,
    UAP_LINK_UFP_CAPTION D_STR_255,
    UAP_LOCKED D_BOOL,
    RES_EDITOR D_STR_CODE,
    RES_STATE D_INTEGER,
    RES_READONLY D_BOOL)
AS
declare variable ufd_link_source d_str_code;
declare variable ufp_name d_str_code;
declare variable fr_code d_str_code;
declare variable ufd_is_table d_bool;
declare variable ufp_pk_field d_smallint;
declare variable max_pk d_integer;
declare variable ufp_is_output d_bool;
declare variable ua_kind d_integer;
declare variable ucf_is_mainframe d_bool;
declare variable ufp_origin_name d_str_code;
begin
  uap_ua_code = ua_code;
  res_state = 0;
  ua_kind = 0;

  if (ua_code <> '') then
  begin
    select uxa_code
    from hp_get_uxalias(:ua_code)
    into uxa_code;
  end

  for select ucf_ufr_code, ucf_is_mainframe, ufd_source, ufd_is_table
  from uxcodeframe, uxfrdata, uxalias
  where uxa_code = :uxa_code and
        ucf_ux_code = uxa_ux_code and
        ucf_ufr_code = ufd_ufr_code and
        ufd_kind = 0
  order by ucf_is_mainframe desc, ucf_ufr_code
  into fr_code, ucf_is_mainframe, :ufd_link_source, ufd_is_table
  do
  begin
    max_pk = 0;
    if (ufd_is_table = 1) then
    begin
      select max(ufp_pk_field)
      from uxfrparam
      where ufp_ufr_code = :fr_code
      into max_pk;

      if (max_pk is null) then
        max_pk = 0;
    end

    for select ufp_code, ufp_name, ufp_origin_name, ufp_editor, ufp_pk_field,
               iif(ufp_is_param = 1 or (ufp_pk_field <> 0 and :ufd_is_table = 1), 0, 1)
    from uxfrparam
    where ufp_ufr_code = :fr_code and
          ufp_in_use = 1
    into uap_link_ufp_code, ufp_name, ufp_origin_name, res_editor, ufp_pk_field,
         ufp_is_output
    do
    begin
      -- Если это действие на создание записи, не показываем последнее ключевое поле
      if (ua_kind = 2 and max_pk > 0 and ufp_pk_field = max_pk) then
        Continue;

      if (param_kind = 0 and ufp_is_output <> 0) then
        Continue;

      if (param_kind = 1 and ufp_is_output <> 1) then
        Continue;

      uap_kind = ufp_is_output;

      -- если в форме один фрейм, не показываем источник чтобы текст был покороче
      if (ucf_is_mainframe = 1) then
        uap_link_ufp_caption = ufp_origin_name;
      else
        uap_link_ufp_caption = ufd_link_source || '.' || ufp_origin_name;

      if (ufp_pk_field <> 0) then
        uap_link_ufp_caption = uap_link_ufp_caption || ' (pk' || ufp_pk_field || ')';

      if (uap_link_ufp_caption not containing ufp_name) then
        uap_link_ufp_caption = uap_link_ufp_caption || '   - ' || ufp_name;
  
      select res_def_value, res_source_ufp_code, res_use_default, res_state, res_locked, res_readonly
      from hp_uxparam_final_value(:uap_link_ufp_code, :uxa_code, :ua_code)
      into :uap_default, :uap_from_ufp_code, :uap_use_default, :res_state, :uap_locked, :res_readonly;

      suspend;
    end
  end
end
