CREATE PROCEDURE AAREP_CONSULTATION_PAY(
    REG_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME D_STR_100,
    PARENT_NAME VARCHAR(50),
    KS_DATE_BEGIN D_DATE,
    KAL_NEW_KF_NAME D_STR_100,
    K_KONTRACT_NO D_STR_20,
    PR_NAME D_STR_100,
    K_PERCENT PERC_TYPE,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    PAYED_SUMMA D_CURRENCY,
    OFFICER_NAME_FIO D_STR_100,
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    CONSULTATION_COUNT D_INTEGER,
    NON_CONSULTATION_COUNT D_INTEGER,
    ALL_KONTRACT_COUNT D_INTEGER,
    LO_INN D_STR_20,
    NEW_OR_REPEAT_CLIENT STR_100)
AS
declare variable rd_id id_type;
declare variable c_id d_id;
declare variable k_id id_type;
declare variable k_from_kf_id id_type;
declare variable ks_kc_id d_id;
declare variable k_fact_disb_date d_date;
begin
  rd_id = f_get_rd_id('CashTransactionIncom.Consultation');

  for select ks.ks_date_begin, ks.ks_from_k_id, reg_name, parent_name, ks.ks_kc_id,
             k.k_kontract_no, k.k_from_kf_id, pr.pr_name, k.k_percent, k.k_fact_disb_date
      from kontstate ks
      inner join region_list(:reg_id, 1, -1, 1) on ks.ks_region_id = id_list
      inner join region reg on id_list = reg_unicode
      inner join kontract k on k.k_id = ks.ks_from_k_id
      inner join product pr on k.k_from_pr_id = pr.pr_id
      where pr.pr_code not containing '.IPF.' and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end
  order by ks.ks_kc_id
  into :ks_date_begin, :k_id, :office_name, :parent_name, :ks_kc_id,
       :k_kontract_no, :k_from_kf_id, :pr_name, :k_percent, :k_fact_disb_date
  do
  begin
    consultation_count = 0;
    non_consultation_count = 0;
    all_kontract_count = 0;

    if (k_kontract_no is not null) then
    begin
      kal_new_kf_name = null;
      lo_inn = null;

      select kal.kal_new_kf_name, cl.c_name_fio, coalesce(cl.c_inn, '') as lo_inn
      from knaktlist kal
      inner join clients cl on cl.c_id = kal.kal_new_kf_officer
      where kal.kal_from_kf_id = :k_from_kf_id and
            kal.is_delete = 0 and
            :rep_date_end between kal.kal_date and kal.kal_date_end and
            kal.is_delete = 0
      into :kal_new_kf_name, :officer_name_fio, :lo_inn;

      select coalesce(sum(case when pr.pr_code containing '.consultation.' then 1 else 0 end), 0),
             coalesce(sum(case when not (pr.pr_code containing '.consultation.') then 1 else 0 end), 0)
      from kontstate ks
      left join kontract k on k.k_id = ks.ks_from_k_id
      left join product pr on pr_id = k.k_from_pr_id
      where ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0 and
            ks.ks_date_begin between :rep_date_begin and :rep_date_end and
            ks.ks_kc_id = :ks_kc_id
      into consultation_count, non_consultation_count;

      all_kontract_count = consultation_count + non_consultation_count;

      for select kc_from_c_id, kc_amount_approved, c_name_fio, c_inn
          from knclient
          inner join clients on kc_from_c_id = c_id
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0
          into :c_id, :kc_amount_approved, :c_name_fio, :c_inn
      do
      begin
        payed_summa = 0;

        select sum(ok.ok_amount)
        from operkontract ok
        inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
        where ol.is_delete = 0 and
              ok.ok_from_k_id = :k_id and
              ok.ok_operation_type = :rd_id and
              ol_document_client = :c_id
        into :payed_summa;

        payed_summa = coalesce(payed_summa, 0);

        if (exists(select 1
                   from kontstate ks
                   inner join knclient kncl on ks.ks_from_k_id = kncl.kc_from_k_id
                   where ks.ks_ol_is_delete = 0 and
                         ks.ks_state = 1 and
                         kncl.kc_amount_approved > 0 and
                         kncl.kc_from_c_id = :c_id and
                         ks.ks_from_k_id <> :k_id and
                         ks.ks_date_begin < :k_fact_disb_date))-- дата выдачи кредита меньше даты выдачи текущего кредита
          then
            new_or_repeat_client = 'Повторный клиент';
          else
            new_or_repeat_client = 'Новый клиент';

        suspend;
      end
    end
    k_kontract_no = null;
    k_from_kf_id = null;
    pr_name = null;
    k_percent = 0;
  end
end
