CREATE PROCEDURE FIND_PETTRANS_LIST(
    FIND_TXT VARCHAR(100))
RETURNS (
    PET_ID INTEGER,
    PET_NAME VARCHAR(255),
    PET_LABEL VARCHAR(50),
    PET_IS_RENT INTEGER,
    PET_NORMA_YEDIZ INTEGER,
    PET_NORMA_KM INTEGER,
    PED_ID INTEGER,
    PED_DATE DATE,
    PED_REGION_PLACE_ID INTEGER,
    PED_FROM_C_ID DECIMAL(18,0),
    PED_IS_ACTIVE INTEGER,
    PED_OFFICE VARCHAR(100),
    PED_MATOTVET VARCHAR(100))
AS
begin
  find_txt = upper(:find_txt);
  for select pet_id, pet_name, pet_label, pet_is_rent, pet_norma_yediz, pet_norma_km, ped_id, ped_date,
             ped_region_place_id, ped_from_c_id, ped_is_active, ped_office, ped_matotvet
      from find_pettrans_temp
      where upper(pet_name) like '%' || :find_txt || '%' or
            upper(pet_label) like '%' || :find_txt || '%' or
            upper(ped_office) like '%' || :find_txt || '%' or
            upper(ped_matotvet) like '%' || :find_txt || '%'
      into :pet_id, :pet_name, :pet_label, :pet_is_rent, :pet_norma_yediz, :pet_norma_km, :ped_id, :ped_date,
           :ped_region_place_id, :ped_from_c_id, :ped_is_active, :ped_office, :ped_matotvet
  do
  begin
    suspend;
  end
end
