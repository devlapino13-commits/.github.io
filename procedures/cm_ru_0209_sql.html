CREATE PROCEDURE CM_RU_0209_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable pr_code d_str_100;
declare variable c_id d_id;
declare variable disb_date d_date;
declare variable active_loan_cnt d_integer;
declare variable tmp_k_id d_id;
declare variable delinq_days d_integer;
declare variable k_kontract_no d_str_code;
declare variable loan_is_closed d_bool;
declare variable active_loan_balance d_currency;
declare variable is_refund d_bool;
declare variable client_points d_integer;
declare variable client_allowed_amount d_currency;
declare variable pr_name d_str_100;
declare variable kc_amount_approved d_currency;
declare variable delinq_day d_integer;
begin
  -- Проверка договоров по продукту Лето Плюс

  select pr_code, k_disb_date, k_refund, pr_name
  from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  into pr_code, disb_date, is_refund, pr_name;

  if (is_refund = 1 and
      (disb_date not between '01.08.2022' and '23.08.2022' and
       disb_date not between '18.10.2022' and '31.10.2022' and
       disb_date not between '02.12.2022' and '30.12.2022' and
       disb_date not between '15.05.2023' and '31.05.2023'
       )) then
  begin
--    res_error_text = 'Рефинансирование запрещается!';
--    res_error_is_critical = 2;
--    suspend;
  end 

  -- Проверяем только Лето Плюс
  if (pr_code not containing '.summer_plus.' and pr_code not containing '.comfortable.') then
    exit;

  -- Проверка даты выдачи
  if (pr_code containing '.summer_plus.' and disb_date not between '01.08.2022' and '23.08.2022') then
  begin
    res_error_text = 'Продукт Лето Плюс действует только с 01.08.2022 по 23.08.2022';
    res_error_is_critical = 2;
    suspend;
  end
--  if (pr_code containing '.comfortable.' and
--      disb_date not between '19.10.2022' and '31.10.2022' and
--      disb_date not between '02.12.2022' and '30.12.2022' and
--      disb_date not between '15.05.2023' and '31.05.2023'
--      ) then
--  begin
--    res_error_text = 'Продукт Удобный действует только с 19.10.2022 по 31.10.2022';
--    res_error_is_critical = 2;
--    suspend;
--  end

  -- Проверки уровня клиента
  for select kc_from_c_id, c_name_fio, kc_amount_approved
  from knclient, clients
  where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    and kc_from_c_id = c_id
  into c_id, res_info_text, kc_amount_approved
  do
  begin

    -- Клиент может получить заем только один раз по продукту Лето Плюс
    if (exists(
        select * from kontract, knclient, product
        where kc_from_c_id = :c_id
          and kc_from_k_id = k_id
          and kc_amount_approved > 0
          and k_from_pr_id = pr_id
          and pr_code containing '.summer_plus.'
          and k_status > 0
          and k_id <> :k_id
          ) and pr_code containing '.summer_plus.'
          ) then
    begin
      res_error_text = 'По продукту Лето Плюс разрешена выдача только одного займа';
      res_error_is_critical = 2;
      suspend;
    end
    /*
    if (exists(
        select * from kontract, knclient, product
        where kc_from_c_id = :c_id
          and kc_from_k_id = k_id
          and kc_amount_approved > 0
          and k_from_pr_id = pr_id
          and pr_code containing '.comfortable.'
          and k_status > 0
          and k_id <> :k_id
          and k_disb_date >= '02.12.2022'
          )and pr_code containing '.comfortable.') then
    begin
      res_error_text = 'По продукту "Удобный" разрешена выдача только одного займа';
      res_error_is_critical = 2;
      suspend;
    end
    */
    /*if (exists(
        select * from kontract, knclient, product, kontstate
        where kc_from_c_id = :c_id
          and kc_from_k_id = k_id
          and kc_amount_approved > 0
          and k_from_pr_id = pr_id
          and pr_code containing '.comfortable.'
          and k_status > 0
          and k_id <> :k_id
          and k_id = ks_from_k_id
          and ks_ol_is_delete = 0
          and ks_state = 1
          and :disb_date between ks_date_begin and ks_date_end
          ) and
          pr_code containing '.comfortable.' and
          is_refund = 0
          ) then
    begin
      res_error_text = 'Повторный "Удоабный" необходимо рефинансировать.';
      res_error_is_critical = 2;
      suspend;
    end*/

    active_loan_cnt = 0;
    -- Проверка просрочки разрешено не более 30 дней по активным займам
    -- Если нет активных, проверяем последний закрытый
    for with
      -- Берем все активные займы клиента
      active_loans as (
        select kc_from_k_id as tmp_k_id, 0 as is_closed, (:disb_date - oks_delinq_date) as delinq_day
        from knclient, oksummary
        where kc_from_c_id = :c_id
        and kc_amount_approved > 0
        and oks_from_k_id = kc_from_k_id
        and :disb_date between oks_date_begin and oks_date_end
        and oks_close_state = 0
      ),
      -- Берем последний закрытый займ клиента
      closed_loans as (
        select ks_from_k_id as tmp_k_id, 1 as is_closed, 0 delinq_day
        from kontstate,  knclient
        where kc_from_c_id = :c_id
          and kc_from_k_id = ks_from_k_id
          and kc_amount_approved > 0
          and ks_ol_is_delete = 0
          and ks_state = 2
          and ks_date_begin <= :disb_date
        order by ks_date_begin desc
        rows 1
      ),
      all_loans as (
        select tmp_k_id, is_closed, delinq_day from active_loans
        union
        select tmp_k_id, is_closed, delinq_day from closed_loans
      )
    select tmp_k_id, is_closed, k_kontract_no, delinq_day
    from all_loans, kontract
    where tmp_k_id = k_id
    order by is_closed -- Сначала активные, потом закрытый
    into tmp_k_id, loan_is_closed, k_kontract_no, delinq_day do
    begin
      if (delinq_day < 0) then delinq_day = 0;
      if (loan_is_closed = 0) then
        active_loan_cnt = active_loan_cnt + 1;

      -- Закрытый займ проверяем только если нет активных
      if (loan_is_closed = 1 and active_loan_cnt > 0) then
        break;

      select max(ok_ol_date - ok_sh_date)
      from operkred
      where ok_ol_is_active = 1
      and ok_from_k_id = :tmp_k_id
      and ok_os + ok_perc < 0
      and ok_type in (4,5)
      and ok_ol_date <= :disb_date
      into delinq_days;

      if (delinq_days > 30 and pr_code not containing '.comfortable.') then
      begin
        res_error_text = f_format('Есть просрочка по договору %1 более 30 дней (%2).', k_kontract_no, delinq_days);
        res_error_is_critical = 2;
        suspend;
      end

      if (delinq_day > 0) then
      begin
        res_error_text = f_format('Есть активная просрочка по договору %1, (%2).', k_kontract_no, delinq_day);
        res_error_is_critical = 2;
        suspend;
      end
    end

    if (active_loan_cnt > 2 and pr_code not containing '.comfortable.') then
    begin
      res_error_text = f_format('У клиента больше двух активных займов (%1), нельзя выдавать продукт %2.', active_loan_cnt, pr_name);
      res_error_is_critical = 2;
      suspend;
    end

    -- Проверка баллов на момент получения займа
    -- Если рефинансирование, мы не проверяем. Т.к. будет неясно
    if (is_refund = 0 and active_loan_cnt > 0) then
    begin
      active_loan_balance = 0;
      select sum(oks_balance)
      from oksummary, knclient
      where kc_from_c_id = :c_id
      and kc_amount_approved > 0
      and oks_from_k_id = kc_from_k_id
      and :disb_date between oks_date_begin and oks_date_end
      and oks_from_k_id <> :k_id
      and oks_close_state = 0
      into active_loan_balance;

      client_points = f_point_get(9, k_id, c_id, disb_date);
      if (pr_code containing '.summer_plus.') then
      begin
        client_allowed_amount =
          case
            when client_points <= 6 then 50000
            when client_points <= 11 then 100000
            else 150000
          end;
      end else
      if (pr_code containing '.comfortable.') then
      begin
        client_allowed_amount =
        case
          when client_points <= 6 then 100000
          when client_points <= 11 then 160000
          else 200000
        end;
      end 

      if (client_allowed_amount < active_loan_balance + kc_amount_approved) then
      begin
        res_error_text = f_format('Превышена макс. допустимая сумма продукта %3, тек. баланс: %1, балл: %2.', f_numtostr(active_loan_balance, 0), client_points, pr_name);
        res_error_is_critical = 2;
        suspend;
      end
    end
  end
end
