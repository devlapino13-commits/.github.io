CREATE PROCEDURE AAREP_GET_KC_CLIENT(
    REP_KC DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    REP_CLIENT_KOLICH INTEGER,
    REP_CLIENT_KOLICH_IND INTEGER)
AS
declare variable pr_tmp integer;
declare variable cl_tmp integer;
begin
  /* групповое */
  rep_client_kolich = 0;
  rep_client_kolich_ind = 0;
  for select pr_loan_type, sum(k_kolich_client_approved)
      from kontract, oksummary, product
      where k_id = oks_from_k_id and
            oks_date_begin <= :rep_date and
            oks_date_end >= :rep_date and
            k_from_pr_id = pr_id and
            oks_kc_id = :rep_kc and
            oks_close_state = 0
      group by pr_loan_type
      into :pr_tmp, :cl_tmp
  do
  begin
    if (cl_tmp is null) then
      cl_tmp = 0;
    if (pr_tmp = 0) then
      rep_client_kolich = rep_client_kolich + cl_tmp;
    if (pr_tmp = 1) then
      rep_client_kolich_ind = rep_client_kolich_ind + cl_tmp;
  end
  suspend;
end
