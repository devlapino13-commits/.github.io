CREATE PROCEDURE HP_SUD_STAGES
RETURNS (
    CODE D_STR_10,
    NAME D_STR_100)
AS
begin
  -- все
  code = 'A0'; name = 'все активные'; suspend;
  code = 'A1'; name = 'Все за период'; suspend;

  -- этап заявка
  code = 'R0'; name = 'Этап заявки'; suspend;
  code = 'R1'; name = 'Заявка от офиса'; suspend;
  code = 'R2'; name = 'Заявки на утверждении'; suspend;

  -- этап документы
  code = 'D0'; name = 'Сбор документов'; suspend;
  code = 'D1'; name = 'Сбор документов ДРПВ'; suspend;
  code = 'D2'; name = 'Сбор документов офис'; suspend;
  code = 'D3'; name = 'Сбор документов архив'; suspend;
  code = 'D5'; name = 'Документы собраны'; suspend;

  -- этап оплаты ГП
  code = 'P0'; name = 'Этап оплаты ГП'; suspend;
  code = 'P1'; name = 'Выбор адреса для подсудности'; suspend;
  code = 'P2'; name = 'Подтверждение подсудности'; suspend;
  code = 'P3'; name = 'Оплата ГП'; suspend;

  -- этап суда
  code = 'J0'; name = 'Этап суда'; suspend;
  code = 'J1'; name = 'Отправка в суд'; suspend;
  code = 'J2'; name = 'В суде'; suspend;
  code = 'J3'; name = 'Оцифровке решения'; suspend;
  code = 'J4'; name = 'Оцифровк ИД'; suspend;
  code = 'J5'; name = 'Положительное решение суда'; suspend;
  code = 'J6'; name = 'Отрицательное решение суда'; suspend;

  -- этап ФССП
  code = 'С0'; name = 'Этап ФССП'; suspend;
  code = 'C1'; name = 'Выбор ФССП'; suspend;
  code = 'C2'; name = 'Отправка в ФССП'; suspend;
  code = 'C3'; name = 'Отправлены в ФССП'; suspend;
  code = 'C4'; name = 'ИП возбуждено'; suspend;
  code = 'C5'; name = 'ИП окончено'; suspend;
  code = 'C6'; name = 'У контрагента'; suspend;

end
