CREATE PROCEDURE HP_CC_KNCL_CHECK(
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    KONT_TYPE SMALLINT,
    ACT_TYPE SMALLINT,
    OPER STR_100,
    MEMO VARCHAR(1000),
    STATUS SMALLINT,
    CALL_QTY SMALLINT)
AS
begin
  if (act_type = 0) then -- разблокир
  begin
    update or insert into cc_check_knclient (ck_type, ck_from_k_id, ck_from_c_id, ck_is_blocked, ck_blocked_oper,
                                             ck_status)
    values (:kont_type, :k_id, :c_id, 0, null, :status)
    matching (ck_type, ck_from_k_id, ck_from_c_id);
  end

  if (act_type = 1) then -- блок
  begin
    update or insert into cc_check_knclient (ck_type, ck_from_k_id, ck_from_c_id, ck_is_blocked, ck_blocked_oper,
                                             ck_status)
    values (:kont_type, :k_id, :c_id, 1, :oper, :status)
    matching (ck_type, ck_from_k_id, ck_from_c_id);
  end

  if (act_type = 2) then -- позвонил
  begin
    update or insert into cc_check_knclient (ck_type, ck_from_k_id, ck_from_c_id, ck_checked_oper, ck_memo, ck_status,
                                             ck_call_qty)
    values (:kont_type, :k_id, :c_id, :oper, :memo, :status, :call_qty)
    matching (ck_type, ck_from_k_id, ck_from_c_id);
  end

  if (act_type = 3) then -- не дозвонился
  begin
    update or insert into cc_check_knclient (ck_type, ck_from_k_id, ck_from_c_id, ck_checked_oper, ck_memo, ck_status,
                                             ck_call_qty)
    values (:kont_type, :k_id, :c_id, :oper, :memo, :status, :call_qty)
    matching (ck_type, ck_from_k_id, ck_from_c_id);
  end
end
