CREATE PROCEDURE HP_REF_MONEYMASK(
    OLD_RMM_ID D_ID)
RETURNS (
    RMM_ID D_ID,
    RMM_FROM_AB_DT D_ID,
    RMM_FROM_AB_KT D_ID,
    RMM_MASK_DT D_STR_CODE,
    RMM_MASK_KT D_STR_CODE,
    RMM_ADD_INFO D_STR_CODE,
    RMM_FROM_RMO_RESIDENT D_ID,
    RMM_FROM_RMO_NORESIDENT D_ID,
    RMM_LAW_RMO_RESIDENT D_ID,
    RMM_LAW_RMO_NORESIDENT D_ID,
    RMM_NAME D_STR_255,
    RMM_MEMO D_STR_255,
    RMM_ORDER_NO D_INTEGER,
    RMM_FROM_AB_DT_TEXT D_STR_LONG,
    RMM_FROM_AB_KT_TEXT D_STR_LONG,
    RMM_FROM_RMO_RESIDENT_TEXT D_STR_LONG,
    RMM_FROM_RMO_NORESIDENT_TEXT D_STR_LONG,
    RMM_LAW_RMO_RESIDENT_TEXT D_STR_LONG,
    RMM_LAW_RMO_NORESIDENT_TEXT D_STR_LONG)
AS
begin
  for select rmm_id,
             coalesce(rmm_from_ab_dt, 0),
             coalesce(rmm_from_ab_kt, 0),
             rmm_mask_dt,
             rmm_mask_kt,
             rmm_add_info,
             coalesce(rmm_from_rmo_resident, 0),
             coalesce(rmm_from_rmo_noresident, 0),
             coalesce(rmm_law_rmo_resident, 0),
             coalesce(rmm_law_rmo_noresident, 0),
             rmm_name,
             rmm_memo,
             rmm_order_no
      from ref_moneymask
      where (:old_rmm_id is null or rmm_id = :old_rmm_id)
      into :rmm_id,
           :rmm_from_ab_dt,
           :rmm_from_ab_kt,
           :rmm_mask_dt,
           :rmm_mask_kt,
           :rmm_add_info,
           :rmm_from_rmo_resident,
           :rmm_from_rmo_noresident,
           :rmm_law_rmo_resident,
           :rmm_law_rmo_noresident,
           :rmm_name,
           :rmm_memo,
           :rmm_order_no
  do
  begin
    rmm_from_ab_dt_text = f_objname('ref_accbalance', rmm_from_ab_dt);
    rmm_from_ab_kt_text = f_objname('ref_accbalance', rmm_from_ab_kt);
    rmm_from_rmo_resident_text = f_objname('ref_moneyoper', rmm_from_rmo_resident);
    rmm_from_rmo_noresident_text = f_objname('ref_moneyoper', rmm_from_rmo_noresident);
    rmm_law_rmo_resident_text = f_objname('ref_moneyoper', rmm_law_rmo_resident);
    rmm_law_rmo_noresident_text = f_objname('ref_moneyoper', rmm_law_rmo_noresident);

    rmm_from_ab_dt_text = replace(rmm_from_ab_dt_text, '      ', '             ');
    rmm_from_ab_kt_text = replace(rmm_from_ab_kt_text, '      ', '             ');
    rmm_from_rmo_resident_text = replace(rmm_from_rmo_resident_text, '      ', '               ');
    rmm_from_rmo_noresident_text = replace(rmm_from_rmo_noresident_text, '      ', '               ');
    rmm_law_rmo_resident_text = replace(rmm_law_rmo_resident_text, '      ', '               ');
    rmm_law_rmo_noresident_text = replace(rmm_law_rmo_noresident_text, '      ', '               ');

    suspend;
  end
end
