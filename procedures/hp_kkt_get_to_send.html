CREATE PROCEDURE HP_KKT_GET_TO_SEND(
    KKT_KIND_STR D_STR_CODE NOT NULL,
    OL_ID D_ID,
    KKD_CODE D_STR_CODE = null)
RETURNS (
    KKM_ID D_ID,
    JSON_STR D_STR_2048,
    IS_CORRECTION D_BOOL,
    CASH_GROUP D_SMALLINT,
    DONT_PRINT D_BOOL)
AS
declare variable kkt_kind d_smallint;
declare variable our_inn d_str_20;
declare variable corr_datetime_str d_str_code;
declare variable cred_no d_str_20;
declare variable cred_date d_date;
declare variable is_return d_bool;
declare variable corr_datetime d_datetime;
declare variable corr_ol_id d_id;
declare variable correction_json d_str_long;
declare variable add_part_json d_str_long;
declare variable items_json d_str_long;
declare variable items_json_line d_str_long;
declare variable items_json_template d_str_long;
declare variable total_sum decimal(15,2);
declare variable kki_name d_str_255;
declare variable payment_subject d_str_255;
declare variable operation_type d_str_255;
declare variable kki_type d_smallint;
declare variable kki_amount decimal(15,2);
declare variable custom_property_name d_str_255;
declare variable custom_property_value d_str_255;
declare variable rep_datetime d_datetime;
declare variable cashier_name d_str_255;
declare variable cashier_inn d_str_code;
declare variable corr_fpd d_str_code;
declare variable un_id d_id;
declare variable electronically boolean;
declare variable error_text d_str_255;
declare variable msg_subject d_str_255;
declare variable msg_text d_str_long;
declare variable reg_id d_integer;
declare variable kkm_duplicated_kkm_id d_id;
declare variable kkm_add_number d_smallint;
declare variable kkm_operation d_smallint;
declare variable kkm_uid d_str_code;
declare variable db_id d_integer;
declare variable err_mail d_str_code;
declare variable err_prefix d_str_10;
declare variable return_basis_details d_str_long;
declare variable send_check_to d_str_255;
declare variable c_id d_id;
declare variable k_id d_id;

declare function my_f_get_template(template_name d_str_code, kkt_kind d_smallint, rep_datetime d_datetime)
returns d_str_2048
as
begin
return(
  select ktm_value
  from kkt_template
  where ktm_kkt_kind = :kkt_kind
    and ktm_name = :template_name
    and ktm_datetime <= :rep_datetime
  order by ktm_datetime desc
  rows 1);
end

begin
  db_id = f_get_db_id();
  cash_group = 213; -- группа касс, указывается в ЛК ОФД. тестовая 7
  our_inn = '7728744752'; -- наш инн. боевой '7728744752' тестовый '4205389890'.
  cashier_name = '';
  cashier_inn = '';
  corr_datetime_str = '';
  corr_fpd = '';
  dont_print = 0;
  kkt_kind = f_enum('kkt_kind', kkt_kind_str, 'id');

  if (db_id = 81001) then
  begin
    electronically = f_get_const('kkt_print_mode') = 0;
  
    if (kkt_kind_str = '1ofd_cloud') then
      select kkm_id, kkm_add_number, kkm_is_return, kkm_corr_ol_id, kkm_from_ol_id, kkm_duplicated_kkm_id
      from kkt_main
      where kkm_kkt_kind = :kkt_kind
        and kkm_status = 0
      order by kkm_id
      rows 1
      into kkm_id, kkm_add_number, is_return, corr_ol_id, ol_id, kkm_duplicated_kkm_id;
    else if (kkt_kind_str = 'atoll') then
    begin
      if (ol_id is not null) then
        select kkm_id, kkm_add_number, kkm_is_return, kkm_corr_ol_id, kkm_duplicated_kkm_id
        from kkt_main
        where kkm_kkt_kind = :kkt_kind
          and kkm_status = 0
          and kkm_from_ol_id = :ol_id
        order by kkm_id
        rows 1
        into kkm_id, kkm_add_number, is_return, corr_ol_id, kkm_duplicated_kkm_id;
      else
      begin
        -- корректировочные чеки для касс в офисах (фоновая печать)
        reg_id = (select kkd_from_reg_id from kkt_devices where kkd_code = :kkd_code);
  
        select kkm_id, kkm_add_number, kkm_is_return, kkm_corr_ol_id, kkm_from_ol_id, kkm_duplicated_kkm_id
        from kkt_main
        where kkm_kkt_kind = :kkt_kind
          and kkm_status = 0
          and kkm_corr_reg_id = :reg_id
        order by kkm_id
        rows 1
        into kkm_id, kkm_add_number, is_return, corr_ol_id, ol_id, kkm_duplicated_kkm_id;
      end
    end
  end
  else if (db_id = 39801) then
  begin
    if (kkt_kind_str = 'WebKassa') then
      select kkm_id, kkm_add_number, kkm_is_return, kkm_corr_ol_id, kkm_from_ol_id, kkm_duplicated_kkm_id, kkm_operation, uuid_to_char(kkm_uuid)
      from kkt_main
      where kkm_kkt_kind = :kkt_kind
        and kkm_status = 0
      order by kkm_id
      rows 1
      into kkm_id, kkm_add_number, is_return, corr_ol_id, ol_id, kkm_duplicated_kkm_id, kkm_operation, kkm_uid;
  end

  if (kkm_id is null) then
  begin
    if (kkt_kind_str = 'atoll' and ol_id is not null) then
    begin
      dont_print = 1;
      suspend;
    end

    exit;
  end

  corr_datetime = null;

  if (db_id = 81001 and corr_ol_id is not null) then
  begin
    if (kkm_duplicated_kkm_id is not null) then
      select kkm_fisc_datetime, kkm_fpd
      from kkt_main
      where kkm_id = :kkm_duplicated_kkm_id
      into corr_datetime, corr_fpd;
    else
    begin
      select kkm_fisc_datetime, kkm_fpd
      from kkt_main
      where kkm_from_ol_id = :corr_ol_id
        and kkm_ol_is_delete = 0
      rows 1
      into corr_datetime, corr_fpd;
  
      if (corr_datetime is null) then
        select ol_date
        from operlist
        where ol_id = :corr_ol_id
        into corr_datetime;
    end
  end

  select k_kontract_no, k_disb_date, ol_user_id, kc_from_c_id, k_id
  from operlist, operkred, kontract, knclient
  where :ol_id in (ol_parent_id, ol_id)
    and ok_from_ol_id = ol_id
    and k_id = ok_from_k_id
    and kc_from_k_id = k_id
  rows 1
  into cred_no, cred_date, un_id, c_id, k_id;

  if (cred_no is null) then
    select k_kontract_no, k_disb_date, ol_user_id, kc_from_c_id, k_id
    from operlist, operkroff, kontract, knclient
    where :ol_id in (ol_parent_id, ol_id)
      and oko_from_ol_id = ol_id
      and k_id = oko_from_k_id
      and kc_from_k_id = k_id
    rows 1
    into cred_no, cred_date, un_id, c_id, k_id;

  if (kkt_kind_str = '1ofd_cloud') then
  begin
    if (corr_datetime is not null) then
      corr_datetime_str =
        lpad(extract(year   from corr_datetime), 4, '0') || '-' ||
        lpad(extract(month  from corr_datetime), 2, '0') || '-' ||
        lpad(extract(day    from corr_datetime), 2, '0') || 'T' ||
        lpad(extract(hour   from corr_datetime), 2, '0') || ':' ||
        lpad(extract(minute from corr_datetime), 2, '0') || ':' ||
        lpad(cast(extract(second from corr_datetime) as integer), 2, '0') || '+03:00';

    operation_type  = trim(iif(is_return = 0, 'INCOME', 'INCOME_RETURN'));
  end 
  else if (kkt_kind_str = 'atoll') then
  begin
    if (corr_datetime is not null) then
      corr_datetime_str =
        lpad(extract(year   from corr_datetime), 4, '0') || '.' ||
        lpad(extract(month  from corr_datetime), 2, '0') || '.' ||
        lpad(extract(day    from corr_datetime), 2, '0');

    operation_type  = trim(iif(is_return = 0, 'sell', 'sellReturn'));

    select fio, inn
    from hp_kkt_get_userdata(:un_id)
    into cashier_name, cashier_inn;

    cashier_inn = trim(iif(cashier_inn <> '', ',"vatin":"' || cashier_inn || '"', ''));
  end
  else if (kkt_kind_str = 'WebKassa') then
  begin
    operation_type = case
                       when kkm_operation = 2 and is_return = 0 then 0 --  покупка
                       when kkm_operation = 2 and is_return = 1 then 1 --  возврат покупки
                       when kkm_operation = 1 and is_return = 0 then 2 --  продажа
                       when kkm_operation = 1 and is_return = 1 then 3 --  возврат продажи
                     end;

    select kkd_code
    from kkt_devices
    where current_date between kkd_date_begin and kkd_date_end
    rows 1
    into kkd_code;

    if (not exists(select 1 from hp_db_is_actual)) then
      kkd_code = 'SWK00033830';

    if (kkd_code is null) then
    begin
      error_text = 'Not found kkd_code';
      exception e_default error_text;
    end
  end
  else
  begin
    error_text = 'Unknown kkt_kind: ' || coalesce(kkt_kind_str, '<NULL>');
    exception e_default error_text;
  end

  rep_datetime = coalesce(corr_datetime, current_timestamp);
  items_json_template = my_f_get_template('item_json', kkt_kind, rep_datetime);

  total_sum = 0;
  items_json = '';

  for select kki_type, kki_amount
  from kkt_items
  where kki_from_kkm_id = :kkm_id
  order by kki_position
  into kki_type, kki_amount
  do
  begin
    if (db_id = 81001) then
    begin
      if (kkt_kind_str = '1ofd_cloud') then
        payment_subject = trim(iif(kki_type = 2, 'NONE_OPERATIONAL_INCOME', 'SERVICE'));
      else if (kkt_kind_str = 'atoll') then
        payment_subject = trim(iif(kki_type = 2, 'nonOperatingIncome', 'service'));
    end

    total_sum = total_sum + kki_amount;
    kki_name = my_f_get_template('item_name' || kki_type, 0, rep_datetime);
    kki_name = replace(kki_name, '<#cred_no#>', cred_no);
    kki_name = replace(kki_name, '<#cred_date#>', f_datetostr(cred_date));

    items_json_line = items_json_template;
    items_json_line = replace(items_json_line, '<#kki_name#>', kki_name);
    items_json_line = replace(items_json_line, '<#kki_amount#>', kki_amount);
    items_json_line = replace(items_json_line, '<#payment_subject#>', coalesce(payment_subject, ''));

    items_json = items_json || trim(iif(items_json <> '', ',', '')) || items_json_line;
  end
  custom_property_name = my_f_get_template('custom_property_name', 0, rep_datetime);
  custom_property_value = my_f_get_template('custom_property_value', 0, rep_datetime);
  custom_property_value = replace(custom_property_value, '<#cred_no#>', cred_no);

  correction_json = '';
  add_part_json = '';
  is_correction = 0;

  if (corr_datetime is not null) then
  begin
    is_correction = 1;
    if (kkt_kind_str = 'atoll') then
      operation_type = operation_type || 'Correction';

    correction_json = my_f_get_template('correction_json', kkt_kind, rep_datetime);
    correction_json = replace(correction_json, '<#corr_datetime_str#>', corr_datetime_str);

    if (coalesce(corr_fpd, '') <> '') then
    begin
      add_part_json = my_f_get_template('add_part_json', kkt_kind, rep_datetime);
      add_part_json = replace(add_part_json, '<#corr_fpd#>', corr_fpd);
    end
  end

  json_str = my_f_get_template('cheque_json', kkt_kind, rep_datetime);
  json_str = replace(json_str, '<#operation_type#>', operation_type);
  json_str = replace(json_str, '<#items_json#>', items_json);
  json_str = replace(json_str, '<#total_sum#>', total_sum);

  if (db_id = 81001) then
  begin
    json_str = replace(json_str, '<#cash_group#>', cash_group);
    json_str = replace(json_str, '<#kkm_id#>', kkm_id || coalesce('_' || kkm_add_number, ''));
    json_str = replace(json_str, '<#electronically#>', lower(electronically));
    json_str = replace(json_str, '<#correction_json#>', correction_json);
    json_str = replace(json_str, '<#add_part#>', add_part_json);
    json_str = replace(json_str, '<#cashier_name#>', cashier_name);
    json_str = replace(json_str, '<#cashier_inn#>', cashier_inn);
    json_str = replace(json_str, '<#custom_property_name#>', custom_property_name);
    json_str = replace(json_str, '<#custom_property_value#>', custom_property_value);
    json_str = replace(json_str, '<#our_inn#>', our_inn);

    -- RFCRU-4775
    send_check_to = '';

    -- если продан по цессии, то send_to не формируем
    if (not exists(select 1 from sud_cession_packet, sud_cession_loans
              where scl_from_scp_id = scp_id
                and scl_from_k_id = :k_id
                and scp_status > 0))
    then
    begin
      select coalesce(cv_contact_value, '')
      from contverify
      where cv_object_kind = 0
        and cv_object_id = :c_id
        and cv_from_rd_id = 5847
        and cv_status > 0
        and cv_verify_datetime > '18.03.2026'
      order by cv_verify_datetime desc
      rows 1
      into send_check_to;
  
      if (send_check_to = '') then
      begin
        send_check_to = coalesce(f_cont_info(0, c_id, 'cell', 'ok'), '');
  
        if (send_check_to = '') then
          send_check_to = coalesce(f_cont_info(0, c_id, 'cell', 'last'), '');
    
        if (send_check_to <> '') then
          send_check_to = '+' || send_check_to;
      end

      if (send_check_to <> '') then
      begin
        update kkt_main
        set kkm_send_to = :send_check_to
        where kkm_id = :kkm_id;
  
        if (kkt_kind_str = '1ofd_cloud') then
          send_check_to = '"sendCheckTo": "' || send_check_to || '",';
        else
          send_check_to = '"clientInfo": {"emailOrPhone": "' || send_check_to || '"},';
      end
    end

    json_str = replace(json_str, '<#send_check_to#>', send_check_to);
  end

  if (db_id = 39801) then
  begin
    json_str = replace(json_str, '<#kkd_code#>', kkd_code);
    json_str = replace(json_str, '<#kkm_uid#>', kkm_uid);

    -- RFCKZ-173 формат 2.0.3
    return_basis_details = '';

    if (is_return = 1 and exists(select cv_value from const_var  where cv_code = 'kkt_kz_format_203' and cv_value = 1)) then
    begin
      -- Найдем данные для возврата
      with t1 as(
        select kkm_fpd, replace(kkm_fisc_datetime, ' ', 'T') || 'Z' dt, kkf_code, cast(sum(kki_amount) as decimal(15,2)) summ
        from kkt_main, kkt_fn, kkt_devices, kkt_items
        where kkf_id = kkm_from_kkf_id
          and kkd_id = kkf_from_kkd_id
          and kki_from_kkm_id = kkm_id
          and kkm_from_ol_id = :ol_id
          and kkm_is_return = 0
        group by 1,2,3
      )
      select f_format('"ReturnBasisDetails":{"CheckNumber": "%1","DateTime": "%2","RegistrationNumber": "%3","Total": %4,"IsOffline": false},', kkm_fpd, dt, kkf_code, summ)
      from t1
      rows 1
      into return_basis_details;
    end

    json_str = replace(json_str, '<#return_basis_details#>', return_basis_details);
  end

  suspend;

  when any do
  begin
    err_mail = trim(iif(db_id = 81001, 'ru.finstart@molbulak.com', 'finstart@mbfplanet.com'));
    err_prefix = case db_id
                   when 81001 then 'RU: '
                   when 39801 then 'KZ: '
                   when 41701 then 'KG: '
                   when 94401 then 'AZ: '
                   else ''
                 end;

    msg_subject = err_prefix || 'ККТ: Ошибка в hp_kkt_send_message.';
    msg_text  = 'Ошибка в hp_kkt_get_to_send: ' || coalesce(error_text, 'gdscode: ' || gdscode) || ascii_char(10) || ascii_char(13) ||
                'Параметры:' || ascii_char(10) || ascii_char(13) ||
                ' kkt_kind_str = ' || coalesce(kkt_kind_str, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' ol_id = ' || coalesce(ol_id, '<NULL>') || ascii_char(10) || ascii_char(13) ||
                ' kkd_code = ' || coalesce(kkd_code, '<NULL>');
    in autonomous transaction do
      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind)
      values (0, :err_mail, :msg_subject, :msg_text, 7, 1000000000001, '');

    exception;
  end
end
