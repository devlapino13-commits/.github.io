CREATE PROCEDURE HP_CONTDATA_ONE_CHECK(
    REP_C_ID D_ID,
    EDIT_CHECK_STATUS D_SMALLINT = 1)
RETURNS (
    C_ID D_ID,
    RD_CODE D_STR_100,
    STATUS D_STR_100,
    VALUE_OLD D_STR_100,
    VALUE_NEW D_STR_100)
AS
declare variable cont_kind d_str_code;
declare variable cont_kind_id d_integer;
declare variable cont_type d_str_code;
declare variable check_state d_smallint;
begin
  if (rep_c_id is null) then rep_c_id = 0;
  if (rep_c_id = 0) then exit;

  c_id = rep_c_id;
  if (edit_check_status is null) then
    edit_check_status = 0;

  check_state = 1;
  for select rd_code
  from refdata
  where rd_main_code = 'ContactInfo' and
        rd_main_code <> rd_code and
        rd_code <> 'ContactInfo.Insta'
  into rd_code
  do
  begin
    cont_kind = replace(rd_code, 'ContactInfo.', '');
    if (cont_kind in ('cell_main', 'cell_add')) then
      cont_kind = 'cell';

    cont_kind_id = f_enum('cont_kind', cont_kind, 'id');

    cont_type = 'add';
    if (rd_code = 'ContactInfo.cell_main') then cont_type = 'main';

    status = 'last';
    value_old = f_cont_info(0, c_id, rd_code, status);
    value_new = f_cont(c_id, cont_kind_id, status, cont_type);

    if (value_old <> value_new) then
    begin
      check_state = -1;
      Suspend;
    end

    status = 'ok';
    value_old = f_cont_info(0, c_id, rd_code, status);
    value_new = f_cont(c_id, cont_kind_id, status, cont_type);

    if (value_old <> value_new) then
    begin
      check_state = -1;
      Suspend;
    end
  end

  if (edit_check_status = 1) then
  begin
    update continfo
    set cci_check_status = :check_state
    where cci_object_kind = 0 and
          cci_object_id = :c_id and
          coalesce(cci_check_status, -777) <> :check_state;
  end
end
