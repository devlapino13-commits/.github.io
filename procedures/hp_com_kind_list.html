CREATE PROCEDURE HP_COM_KIND_LIST
RETURNS (
    CMK_ID D_INTEGER,
    CMK_NAME D_STR_LONG)
AS
declare variable cmk_reason d_str_long;
begin
  for select cmk_id, cmk_name, cmk_reason
  from com_kind
  where cmk_inactive = 0
  into :cmk_id, :cmk_name, cmk_reason
  do
  begin
    if (cmk_name is null) then
      cmk_name = '';

    if (cmk_reason is null) then
      cmk_reason = '';

    if (cmk_id <> 0) then
    begin
      if (cmk_reason <> '') then
      begin
        if (cmk_name <> '')
        then cmk_name = cmk_name || ' (' || cmk_reason || ')';
        else cmk_name = cmk_reason;
      end
    end

    if (cmk_name = '') then
      cmk_name = 'Тип КК ' || cmk_id;

    suspend;
  end
end
