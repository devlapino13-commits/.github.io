CREATE PROCEDURE HP_EDOC_CLEANUP
AS
declare variable edl_id d_id;
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable c_name_fio d_str_100;
declare variable res_text str_255;
declare variable un_id d_id;
declare variable k_disb_date d_date;
begin
  --В 16:00 электронный документ (договор займа) должен аннулироваться, если он не подписан, а выдача стоит текущей датой, fsrobot 215
  for select edl.edl_id, k.k_id, k.k_kontract_no, k.k_disb_date
    from edoc_list edl
    inner join edoc_cfg edc on edc.edc_id = edl.edl_from_edc_id
    inner join kontract k on k.k_id = edl.edl_object_id
    inner join product pr on pr.pr_id = k.k_from_pr_id
    where edl.edl_object_kind = 9
      and edc.edc_code = 'loan.agreement'
      and edl.edl_status = 0
      and k.k_disb_date = current_date
      and pr.pr_code containing '.business.'
    into edl_id, k_id, k_kontract_no, k_disb_date
  do
  begin
    select c.c_name_fio
    from clients c
    inner join knclient kc on kc.kc_from_c_id = c.c_id
    where kc.kc_from_k_id = :k_id
      and kc.kc_amount_approved > 0
    into :c_name_fio;

    update edoc_list
    set edl_status = -1
    where edl_id = :edl_id;

    select un.un_id
    from kontract k
    inner join knaktlist kal on kal.kal_from_kf_id = k.k_from_kf_id
    inner join username un on kal.kal_new_kf_officer = un.un_from_c_id
    where k.k_id = :k_id
      and kal.is_delete = 0
      and current_date between kal_date and kal_date_end
    into :un_id;

    res_text = 'Договор займа № ' || k_kontract_no || ' от ' || cast(k_disb_date as date) || ' ' ||
                     coalesce(c_name_fio,'') || ' - аннулирован';
    if (un_id is not null) then
          f_send_push('username', :un_id, 'Аннулирование договора', :res_text, 'hp_edoc_cleanup_new', 7, 1000000000000);
  end
end
