CREATE PROCEDURE AAREP_HUMAN_HOURS(
    REP_REGION_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    TERRITORY D_STR_255,
    OFFICE D_STR_255,
    GEOGRAPH_AREA D_STR_255,
    PARENT D_STR_255,
    COUNT_EMP_JAN D_INTEGER,
    COUNT_EMP_WOMAN_JAN D_INTEGER,
    SAT_HOURS_JAN D_CURRENCY,
    COUNT_EMP_FEB D_INTEGER,
    COUNT_EMP_WOMAN_FEB D_INTEGER,
    SAT_HOURS_FEB D_CURRENCY,
    COUNT_EMP_MAR D_INTEGER,
    COUNT_EMP_WOMAN_MAR D_INTEGER,
    SAT_HOURS_MAR D_CURRENCY,
    COUNT_EMP_APR D_INTEGER,
    COUNT_EMP_WOMAN_APR D_INTEGER,
    SAT_HOURS_APR D_CURRENCY,
    COUNT_EMP_MAY D_INTEGER,
    COUNT_EMP_WOMAN_MAY D_INTEGER,
    SAT_HOURS_MAY D_CURRENCY,
    COUNT_EMP_JUN D_INTEGER,
    COUNT_EMP_WOMAN_JUN D_INTEGER,
    SAT_HOURS_JUN D_CURRENCY,
    COUNT_EMP_JUL D_INTEGER,
    COUNT_EMP_WOMAN_JUL D_INTEGER,
    SAT_HOURS_JUL D_CURRENCY,
    COUNT_EMP_AUG D_INTEGER,
    COUNT_EMP_WOMAN_AUG D_INTEGER,
    SAT_HOURS_AUG D_CURRENCY,
    COUNT_EMP_SEP D_INTEGER,
    COUNT_EMP_WOMAN_SEP D_INTEGER,
    SAT_HOURS_SEP D_CURRENCY,
    COUNT_EMP_OKT D_INTEGER,
    COUNT_EMP_WOMAN_OKT D_INTEGER,
    SAT_HOURS_OKT D_CURRENCY,
    COUNT_EMP_NOV D_INTEGER,
    COUNT_EMP_WOMAN_NOV D_INTEGER,
    SAT_HOURS_NOV D_CURRENCY,
    COUNT_EMP_DEC D_INTEGER,
    COUNT_EMP_WOMAN_DEC D_INTEGER,
    SAT_HOURS_DEC D_CURRENCY)
AS
declare variable year_month d_integer;
declare variable rep_year d_integer;
declare variable sat_year d_integer;
declare variable sat_month d_integer;
declare variable first_day d_date;
declare variable last_day d_date;
declare variable reg_hr_region_id d_integer;
declare variable reg_id d_integer;
declare variable emp_c_id d_id;
declare variable cl_count d_integer;
declare variable cl_w_count d_integer;
declare variable count_emp d_integer;
declare variable count_emp_woman d_integer;
declare variable sat_hours d_currency;
declare variable current_month d_integer;
begin
  rep_year = extract(year from :rep_date);

  --- получаем поля по региону
  for select reg.reg_stat_rep_region, reg.reg_code || ' ' || reg.reg_name, reg.reg_hr_region, regp.reg_name,
             reg.reg_unicode
      from region reg
      inner join region_list(:rep_region_id, 1, -1, 1) rl on reg.reg_unicode = rl.id_list
      inner join region regp on reg.parent = regp.reg_unicode
      where reg.reg_hr_region is not null and
            reg.reg_date_close is null and
            regp.reg_date_close is null
      into :territory, :office, :reg_hr_region_id, :parent, :reg_id
  do
  begin
    if (reg_hr_region_id is not null) then
    begin
      geograph_area = '';
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :reg_hr_region_id
      into :geograph_area;
    end

    -- получаем циклом каждый месяц в году на год даты отчета
    for select distinct sat.sat_year_month
        from sa_tabel sat
        inner join sa_worktype sawt on sat.sat_from_sawt_id = sawt.sawt_id
        where substring(cast(sat.sat_year_month as d_str_10) from 1 for 4) = cast(:rep_year as d_str_10) and
              sat.sat_is_ready = 1 and
              sat.sat_is_new_added = 1 and
              sat.sat_office_id = :reg_id
        into :year_month
    do
    begin
      sat_year = substring(cast(year_month as d_str_10) from 1 for 4);
      sat_month = right(cast(year_month as d_str_10), 2);
      current_month = extract(month from :rep_date);
      first_day = cast('01.' || iif(char_length(sat_month) > 1, sat_month, '0' || sat_month) || '.' || sat_year as d_date);
      last_day = dateadd(1 month to first_day) - 1;

      sat_hours = null;
      --- получаем общее кол-во отработанных часов на офис
      select sum(sat.sat_total_hours)
      from sa_tabel sat
      inner join sa_worktype sawt on sat.sat_from_sawt_id = sawt.sawt_id
      where sat.sat_office_id = :reg_id and
            sat.sat_is_ready = 1 and
            sat.sat_year_month = (:sat_year * 100 + :sat_month) and
            sawt.sawt_id in (0, 11, 12)
      group by sat.sat_year_month
      into :sat_hours;

      count_emp = 0;
      count_emp_woman = 0;

      --- сотрудники на офис
      for select ps.ps_from_c_id
          from hr_pstatus ps
          inner join hr_orders ho on ps.ps_from_ho_id = ho.ho_id
          where ps.ps_is_working = 1 and
                ps.ps_date_begin <= :last_day and
                ps.ps_date_end >= :last_day and
                ps.ps_from_reg_id = :reg_id --and
--                ((ps.ps_from_reg_id = ps.ps_transfer_office) or (ps.ps_transfer_office is null))
          into :emp_c_id
      do
      begin
        --- общее количество сотрудников офиса
        cl_count = null;
        select count(cl.c_id)
        from clients cl
        where cl.c_id = :emp_c_id
        into :cl_count;

        count_emp = count_emp + cl_count;

        --- количество женщин из общего количества сотрудников офиса
        cl_w_count = null;
        select count(cl.c_id)
        from clients cl
        where cl.c_id = :emp_c_id and
              cl.c_sex = 0
        into :cl_w_count;

        count_emp_woman = count_emp_woman + cl_w_count;
      end

      sat_hours = coalesce(sat_hours, 0);
      count_emp = coalesce(count_emp, 0);
      count_emp_woman = coalesce(count_emp_woman, 0);

      --- обнуляем количество сотрудников и часы, если месяц на текущей итерации больше, чем месяц даты отчета (если табель еще не сформирован)
      if (sat_month > current_month) then
      begin
        sat_hours = 0;
        count_emp = 0;
        count_emp_woman = 0;
      end

      --- Январь
      if (sat_month = 1) then
      begin
        count_emp_jan = count_emp;
        count_emp_woman_jan = count_emp_woman;
        sat_hours_jan = sat_hours;
      end
      --- Февраль
      if (sat_month = 2) then
      begin
        count_emp_feb = count_emp;
        count_emp_woman_feb = count_emp_woman;
        sat_hours_feb = sat_hours;
      end
      --- Март
      if (sat_month = 3) then
      begin
        count_emp_mar = count_emp;
        count_emp_woman_mar = count_emp_woman;
        sat_hours_mar = sat_hours;
      end
      --- Апрель
      if (sat_month = 4) then
      begin
        count_emp_apr = count_emp;
        count_emp_woman_apr = count_emp_woman;
        sat_hours_apr = sat_hours;
      end
      --- Май
      if (sat_month = 5) then
      begin
        count_emp_may = count_emp;
        count_emp_woman_may = count_emp_woman;
        sat_hours_may = sat_hours;
      end
      --- Июнь
      if (sat_month = 6) then
      begin
        count_emp_jun = count_emp;
        count_emp_woman_jun = count_emp_woman;
        sat_hours_jun = sat_hours;
      end
      --- Июль
      if (sat_month = 7) then
      begin
        count_emp_jul = count_emp;
        count_emp_woman_jul = count_emp_woman;
        sat_hours_jul = sat_hours;
      end
      --- Август
      if (sat_month = 8) then
      begin
        count_emp_aug = count_emp;
        count_emp_woman_aug = count_emp_woman;
        sat_hours_aug = sat_hours;
      end
      --- Сентябрь
      if (sat_month = 9) then
      begin
        count_emp_sep = count_emp;
        count_emp_woman_sep = count_emp_woman;
        sat_hours_sep = sat_hours;
      end
      --- Октябрь
      if (sat_month = 10) then
      begin
        count_emp_okt = count_emp;
        count_emp_woman_okt = count_emp_woman;
        sat_hours_okt = sat_hours;
      end
      --- Ноябрь
      if (sat_month = 11) then
      begin
        count_emp_nov = count_emp;
        count_emp_woman_nov = count_emp_woman;
        sat_hours_nov = sat_hours;
      end
      --- Декабрь
      if (sat_month = 12) then
      begin
        count_emp_dec = count_emp;
        count_emp_woman_dec = count_emp_woman;
        sat_hours_dec = sat_hours;
      end
    end
    suspend;
  end
end
