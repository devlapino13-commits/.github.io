CREATE PROCEDURE HP_OVERPAYMENT_RETURN_SAVE_TMP(
    NEW_OL_ID ID_TYPE,
    REG_ID INTEGER,
    OPER_DATE DATE,
    DOC_NO INTEGER,
    OL_MEMO VARCHAR(255),
    UN_ID ID_TYPE,
    C_ID ID_TYPE,
    DEF_CURR VARCHAR(3),
    ACC_1 VARCHAR(20),
    SUM_OVP CURR_TYPE,
    ACC_2 VARCHAR(20),
    K_ID ID_TYPE,
    LAST_OVP_DATE DATE,
    INSERT_TO_OK SMALLINT,
    OL_PARENT_ID ID_TYPE,
    OB_DETAIL VARCHAR(1024))
AS
begin
  insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_no, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_user_id, ol_document_client, ol_parent_id)
  values (:new_ol_id, :reg_id, iif(:insert_to_ok = 1, 1, 2), iif(:insert_to_ok = 1, 0, 20), :oper_date, :doc_no,
  iif(:insert_to_ok = 1, 0, 9), :ol_memo, 0, 0, :un_id, :c_id, iif(:ol_parent_id <> :new_ol_id, :ol_parent_id, null)); --проверить ol_memo, ol_oper_subtype

  if (:insert_to_ok = 1) then
  begin
    insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                          ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                          ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
    select :new_ol_id, -1, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, -ok_os, 0, 0, ok_pay_type, 0, 0, 0, 0, 0, null,
           null, null, 0
    from operkred
    where ok_from_k_id = :k_id and
          ok_type = 9 and
          ok_ol_is_active = 1 and
          ok_os < 0 and
          ok_ol_date > :last_ovp_date;

    insert_to_ok = 0;
  end

  insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt, ob_summa_val,
                        ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt, ob_client_kt, ob_detail,
                        ob_from_sac_id, ob_ol_date, ob_ol_reg_id)
  values (:new_ol_id, null, :def_curr, 1, :c_id, :acc_1, :sum_ovp, :sum_ovp, null, null, null, :acc_2, null,
          :ob_detail, null, :oper_date, :reg_id);
end
