CREATE PROCEDURE HP_SUD_LOANDOCS_ADD(
    SMD_ID D_ID NOT NULL,
    SLC_ID D_ID,
    DOC_KIND D_INTEGER NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_SLD_ID D_ID)
AS
declare variable sld_status d_integer;
begin
  if (slc_id = 0) then
    slc_id = null;

  -- Проверяем наличие уже созданного sud_loandocs
  select sld_id, sld_status
  from sud_loandocs
  where sld_from_smd_id = :smd_id and
        coalesce(sld_from_slc_id, 0) = coalesce(:slc_id, 0) and
        sld_doc_kind = :doc_kind and
        sld_inactive = 0
  into res_sld_id, sld_status;

  if (res_sld_id is null) then
  begin
    -- если нет, создаем новую запись
    insert into sud_loandocs (sld_from_smd_id, sld_from_slc_id, sld_doc_kind, sld_from_un_id)
    values (:smd_id, :slc_id, :doc_kind, :un_id)
    returning sld_id
    into res_sld_id;
  end else
  begin
    -- Если есть и (на доработке или отказ), то восстанавливаем статус
    if (sld_status < 0) then
    begin
      update sud_loandocs
      set sld_status = 0,
          sld_from_un_id = :un_id
      where sld_id = :res_sld_id;
    end
  end

  suspend;
end
