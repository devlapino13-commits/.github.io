CREATE PROCEDURE AAREP_OSV_EPS_OSV_ACC(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AB_ID ID_TYPE,
    REP_FOLD_ACCOUNT INTEGER,
    REP_AB_CODE D_STR_CODE = null)
RETURNS (
    ACC_ID ID_TYPE,
    ACC_CODE D_STR_CODE,
    DT_SUM CURR_TYPE,
    DT_VAL CURR_TYPE,
    DT_TAX CURR_TYPE,
    KT_SUM CURR_TYPE,
    KT_VAL CURR_TYPE,
    KT_TAX CURR_TYPE,
    B_DT_SUM CURR_TYPE,
    B_DT_VAL CURR_TYPE,
    B_DT_TAX CURR_TYPE,
    B_KT_SUM CURR_TYPE,
    B_KT_VAL CURR_TYPE,
    B_KT_TAX CURR_TYPE,
    E_DT_SUM CURR_TYPE,
    E_DT_VAL CURR_TYPE,
    E_DT_TAX CURR_TYPE,
    E_KT_SUM CURR_TYPE,
    E_KT_VAL CURR_TYPE,
    E_KT_TAX CURR_TYPE,
    ACC_NAME D_STR_255)
AS
declare variable n_start integer;
declare variable acc_mask d_str_code;
begin
  if (rep_ab_code is null) then
    rep_ab_code = '';
  acc_mask = '';
  n_start = position('(', rep_ab_code);
  if (n_start > 0) then
  begin
    acc_mask = left(rep_ab_code, 5);
    rep_ab_code = substring(rep_ab_code from n_start + 1);
    rep_ab_code = left(rep_ab_code, char_length(rep_ab_code) - 1);
    n_start = char_length(rep_ab_code);
    acc_mask = acc_mask || '_____' || rep_ab_code || left('__________', 10 - n_start);
  end

  if (rep_ab_id is null) then
  begin
    for select osv_acc_id, dt_sum, dt_val, dt_tax, kt_sum, kt_val, kt_tax, b_dt_sum, b_dt_val, b_dt_tax, b_kt_sum,
               b_kt_val, b_kt_tax, e_dt_sum, e_dt_val, e_dt_tax, e_kt_sum, e_kt_val, e_kt_tax, acc_name
        from aarep_osv_eps_osv_tmp(:rep_date_begin, :rep_date_end, null, :rep_fold_account), ref_account
        where osv_acc_id = acc_id
        into :acc_id, :dt_sum, :dt_val, :dt_tax, :kt_sum, :kt_val, :kt_tax, :b_dt_sum, :b_dt_val, :b_dt_tax, :b_kt_sum,
             :b_kt_val, :b_kt_tax, :e_dt_sum, :e_dt_val, :e_dt_tax, :e_kt_sum, :e_kt_val, :e_kt_tax, acc_name
    do
    begin
      acc_code = f_fmt_acc(acc_id);
      suspend;
    end
  end
  else
  begin
    for select osv_acc_id, dt_sum, dt_val, dt_tax, kt_sum, kt_val, kt_tax, b_dt_sum, b_dt_val, b_dt_tax, b_kt_sum,
               b_kt_val, b_kt_tax, e_dt_sum, e_dt_val, e_dt_tax, e_kt_sum, e_kt_val, e_kt_tax, acc_name
        from aarep_osv_eps_osv_tmp(:rep_date_begin, :rep_date_end, :rep_ab_id, :rep_fold_account), ref_account
        where osv_acc_id = acc_id and
              acc_from_ab_id = :rep_ab_id and
              (:acc_mask = '' or
              acc_code like :acc_mask)
        into :acc_id, :dt_sum, :dt_val, :dt_tax, :kt_sum, :kt_val, :kt_tax, :b_dt_sum, :b_dt_val, :b_dt_tax, :b_kt_sum,
             :b_kt_val, :b_kt_tax, :e_dt_sum, :e_dt_val, :e_dt_tax, :e_kt_sum, :e_kt_val, :e_kt_tax, acc_name
    do
    begin
      acc_code = f_fmt_acc(acc_id);
      suspend;
    end
  end
end
