CREATE PROCEDURE HP_EMPLOYER_IN_DEKRET(
    REP_C_ID ID_TYPE,
    REP_DATE DATE)
RETURNS (
    IS_DECRET SMALLINT)
AS
declare variable sawt_id integer;
begin
  sawt_id = null;
  select sawt_id
  from hr_voctravel, hr_orders, sa_worktype
  where hrv_from_ho_id = ho_id and
        ho_order_type = sawt_id and
        hrv_from_c_id = :rep_c_id and
        hrv_status = 1 and
        :rep_date between hrv_date_begin and hrv_date_end_fact
  order by hrv_date_begin desc
  rows 1
  into sawt_id;

  sawt_id = coalesce(sawt_id, 0);

--  if (sawt_id in (16,17,18,8, 21)) then is_decret = 1; -- У,рал 8, Sanjari, 03/05/2017
  if (sawt_id in (16, 17, 18, 21)) then
    is_decret = 1;
  else
    is_decret = 0;

  suspend;
end
