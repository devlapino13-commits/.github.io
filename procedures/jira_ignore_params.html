CREATE PROCEDURE JIRA_IGNORE_PARAMS(
    IGNORE_BA D_BOOL,
    IGNORE_CLARIFICATION D_BOOL,
    JIRA_CODE D_STR_CODE,
    TASK_LAST_STATE D_STR_CODE,
    PROGRAMMER_FIO D_STR_100)
RETURNS (
    IS_IGRORE D_BOOL)
AS
begin
    is_igrore = 0;

    if ((ignore_ba = 1) and (task_last_state in ('БА', 'Бизнес анализ'))) then
        is_igrore = 1;
    
    if ((ignore_clarification = 1) and (task_last_state in ('На согласовании', 'Очередь на анализ', 'Уточнение требований'))) then
        is_igrore = 1;
    
    if ((jira_code starting 'fin2') or (jira_code starting 'rfcxx')) then
        is_igrore = 1;
    
    if (programmer_fio in ('Светлана Симоненко', 'Комилжон Каримов loc')) then
        is_igrore = 1;
    
    if (exists (select * from jira_system_issues jsi where jsi.jst_issue_code = :jira_code)) then
        is_igrore = 1;
    suspend;
end
