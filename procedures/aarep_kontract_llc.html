CREATE PROCEDURE AAREP_KONTRACT_LLC(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_REVERSE SMALLINT,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_LLC_K_ID VARCHAR(20),
    AAREP_LLC_K_DELIQ_DAYS INTEGER,
    AAREP_LLC_K_BALANCE DECIMAL(18,4),
    AAREP_LLC_LOAN_TYPE INTEGER,
    AAREP_LLC_STATUS INTEGER,
    AAREP_LLC_REGION_ID INTEGER,
    AAREP_NEED_ONE_UP INTEGER,
    AAREP_MAX_DAY_CIRCLE INTEGER,
    AAREP_MEMO VARCHAR(50),
    AAREP_PERCENT DECIMAL(18,7))
AS
declare variable kind integer;
declare variable prior decimal(18,0);
declare variable m_day integer;
declare variable kol integer;
declare variable kol_pros_day integer;
declare variable min_day_2 integer;
declare variable min_day_1 integer;
declare variable kk_id decimal(18,0);
declare variable kf_id decimal(18,0);
declare variable main_circle integer;
declare variable aarep_llc_k_deliq_days1 integer;
declare variable aarep_llc_status_temp integer;
declare variable cm_default_currency varchar(3);
declare variable kol_rest integer; /* кол-во реструктуризации */
declare variable kk integer;
begin
  aarep_need_one_up = 0;
  aarep_max_day_circle = 0;/*Параметр для проверки, если в последних в 2х циклах были просрочки от 4-30 дней
                            или "первый кредит" то классфификацию на один увеличивает */
  /* Procedure Text */
  select cm_default_currency
  from company
  into :cm_default_currency;
  if (cm_default_currency <> '810') then
  begin
    kol = 0;
    min_day_1 = 0;
    min_day_2 = 0;
    aarep_llc_k_deliq_days1 = -1;
    /*Вытаскивает первые 2 цифры из поле cs_day_max, из таблицы classific*/
    for select cs_day_max
        from classific
        where cs_day_max > -1
        order by cs_day_max
        rows 2
        into :m_day
    do
    begin
      kol = kol + 1;
      if (kol = 1) then
      begin
        min_day_1 = :m_day;
      end
      else
      begin
        min_day_2 = :m_day;
      end
    end
    prior = 0;
    /*Вытаскивает просроченных и активных кредитов и их соединяет*/
    /*
      for select aarep_k_id, Cast(0 as Integer), aarep_k_deliq_days, aarep_k_balance, pr_loan_type,
      kal_new_reg_id, k_from_kf_id
      from aarep_kontract_id(:report_region_id,:report_date_begin,:report_date_end,:report_reverse,:report_currency,5),
      kontract, knaktlist, product
      where aarep_k_id=k_id
      and k_from_pr_id=pr_id
      and kal_from_kf_id=k_from_kf_id
      and :report_date_end between kal_date and kal_date_end
      union
      select ac_k_id, Cast(1 as Integer), Cast(0 as Integer), ac_k_balance, ac_loan_type,
      ac_region_id, ac_kf_id
      from AAREP_KONTRACT_ACTIVE_TEMP(:report_region_id, :report_date_end, :report_currency, :report_reverse)
      order by 1, 2  */
    for with tab1
        as (select aarep_k_id as aarep_llc_k_id, cast(0 as integer) as kind,
                   aarep_k_deliq_days as aarep_llc_k_deliq_days, aarep_k_balance as aarep_llc_k_balance,
                   case
                     when (k_kolich_client_approved > 2 or pr_product_kind = 2) then 0
                     else 1
                   end as aarep_llc_loan_type,
                   kal_new_reg_id as aarep_llc_region_id, k_from_kf_id as kf_id
            from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 5), kontract, knaktlist, product
            where aarep_k_id = k_id and
                  k_from_pr_id = pr_id and
                  kal_from_kf_id = k_from_kf_id and
                  :report_date_end between kal_date and kal_date_end and
                  knaktlist.is_delete = 0
            union
            select ac_k_id as aarep_llc_k_id, cast(1 as integer) as kind, cast(0 as integer) as aarep_llc_k_deliq_days,
                   ac_k_balance as aarep_llc_k_balance,
                   case
                     when (ac_client_kolich > 2 or pr_product_kind = 2) then 0
                     else 1
                   end as aarep_llc_loan_type,
                   ac_region_id as aarep_llc_region_id, ac_kf_id as kf_id
            from aarep_kontract_active_temp(:report_region_id, :report_date_end, :report_currency, :report_reverse), kontract, product
            where ac_k_id = k_id and
                  k_from_pr_id = pr_id)

        select aarep_llc_k_id, kind, aarep_llc_k_deliq_days, aarep_llc_k_balance, aarep_llc_loan_type,
               aarep_llc_region_id, kf_id
        from tab1
        order by aarep_llc_k_id, kind

        into :aarep_llc_k_id, :kind, :aarep_llc_k_deliq_days, :aarep_llc_k_balance, :aarep_llc_loan_type,
             :aarep_llc_region_id, :kf_id
    do
    begin
      aarep_memo = '';
      if (aarep_llc_k_deliq_days is null) then
        aarep_llc_k_deliq_days = 0;
      /*Если поле aarep_llc_k_id не равен к предыдущему то выполняет т.е. повторно не будет считать для этого контракта*/
      if (aarep_llc_k_id <> prior) then
      begin
        aarep_need_one_up = 0;
        aarep_llc_status_temp = 3;
        if (cm_default_currency = '417') then
          min_day_1 = -1; -- если БД КГ то не будет смотреть предыдущий цикл
        /*Если кол-во просроченных дней <= первой ограничений из таблицы clasific поле min_day_1, делает следующее*/
        if (aarep_llc_k_deliq_days <= min_day_1) then
        begin
          main_circle = 0;
          /*Вычисляет цикл контракта*/
          select distinct count(k_id)
          from kontract, product
          where k_from_kf_id = :kf_id and
                k_id <= :aarep_llc_k_id and
                pr_category = 0 -- Параллельные кредиты не включает
                and
                k_status >= 0 and
                k_from_pr_id = pr_id and
                k_disb_date <= :report_date_end
          into :main_circle;
          if (main_circle is null) then
            main_circle = 0;
          /*Если цикл контракта>1 и тип контракта Групповой то делает следующее*/
          if ((main_circle > 1) or (aarep_llc_loan_type = 0)) then
          begin
            kol_pros_day = 0;
            aarep_max_day_circle = 0;
            /*И берет последние 2 цикла этой папки*/
            for select distinct k_id
                from kontract, product
                where ((k_from_kf_id = :kf_id and
                      k_id < :aarep_llc_k_id and
                      pr_category = 0) or
                      (k_id = :aarep_llc_k_id)) and
                      k_status >= 0 and
                      k_from_pr_id = pr_id
                order by k_disb_date desc
                rows 2
                into :kk_id
            do
            begin
              kol_pros_day = 0;
              /*Вычисляет максимальное кол-во просроченных дней в последних двух циклах*/
              select cast(max(ok_ol_date - ok_sh_date) as integer)
              from operkred
              where ok_ol_is_active = 1 and
                    ok_from_k_id = :kk_id and
                    ok_ol_date <= :report_date_end and
                    ok_sh_date <= :report_date_end and
                    (ok_os + ok_perc + ok_fine) <> 0 and
                    ok_from_c_id = 0 and
                    ok_type <> 7 /* 12/12/10 чтоб доначисления не влияли на график*/
              into :kol_pros_day;
              if (kol_pros_day is null) then
                kol_pros_day = 0;
              if (kol_pros_day > :aarep_max_day_circle) then
                aarep_max_day_circle = :kol_pros_day;
            end
            if (aarep_max_day_circle > min_day_1) then
            begin
              aarep_need_one_up = 1;
              aarep_memo = cast(aarep_max_day_circle as varchar(5)) || ' дней просрочки в послед.2 циклах';
            end
          end
          else
          begin
            aarep_need_one_up = 1;
            aarep_memo = 'Первый кредит';
          end
        end
--        Вычисляет кол-во реструктуризации
        kol_rest = 0;
        select count(ol_id)
        from operlist, operrest
        where or_from_ol_id = ol_id and
              operlist.is_delete = 0 and
              or_from_k_id = :aarep_llc_k_id and
              ol_date <= :report_date_end
        into :kol_rest;
        if (kol_rest is null) then
          kol_rest = 0;

        /*Если кол-во просроченных дней не совпадает с предыдущим контрактом то вытаскивает классификацию*/
        if (aarep_llc_k_deliq_days1 <> aarep_llc_k_deliq_days) then
        begin
          /*Статусы для групповых кредитов*/
          /*
            select cs_id from classific
            where cs_day_max>-1
            and :aarep_llc_k_deliq_days between cs_day_min and cs_day_max
            into :aarep_llc_status_temp; */

          /*Присваивает статусы классификации для групповых кредитов*/
          if (cm_default_currency <> '944') then
          begin
            if (:aarep_llc_loan_type = 0) then
            begin
              aarep_llc_status_temp = case
                                        when (:aarep_llc_k_deliq_days = 0 and (kol_rest = 0)) then 3
                                        when ((:aarep_llc_k_deliq_days between 1 and 29) or (kol_rest = 1 and :aarep_llc_k_deliq_days = 0)) then 4
                                        when ((:aarep_llc_k_deliq_days between 30 and 59) and (kol_rest < 2)) then 6
                                        when ((:aarep_llc_k_deliq_days >= 60) or (kol_rest >= 2)) then 7
                                      end;
            end
            /*Присваивает статусы классификации для индивидуальных кредитов*/
            if (:aarep_llc_loan_type = 1) then
            begin
--            Определяет максимальное кол-во просроченных дней для классификации РППУ по индив c другой ФКУ
              kk = null;
              with tabl1
              as (select
                         case kt_status_pogash
                           when 'Платежи по графику' then 2
                           when 'до 30 дней просрочки' then 3
                           when '30 дней до 59 дней просрочки' then 4
                           when '60 дней до 89 дней просрочки' then 5
                           when '90 дней  до 119 дней просрочки' then 6
                           when '120 дней просрочки и более' then 7
                           else 1
                         end as kk
                  from knclient, kibtaken
                  where kc_amount_approved > 0 and
                        kc_from_k_id = :aarep_llc_k_id and
                        kc_from_c_id = kt_from_c_id and
                        kt_year = extract(year from :report_date_end) and
                        kt_month = extract(month from :report_date_end))
              select max(kk)
              from tabl1
              into kk;

              aarep_llc_status_temp = case
                                        when (:aarep_llc_k_deliq_days <= 30 and (kk is null) and (kol_rest = 0)) then 1
                                        when (:aarep_llc_k_deliq_days <= 30 and (kk <= 3) and (kol_rest = 0)) then 4
                                        when (:aarep_llc_k_deliq_days <= 30 and (kk > 3)) then 5
                                        when ((:aarep_llc_k_deliq_days between 31 and 90) or (kol_rest > 0 and :aarep_llc_k_deliq_days <= 30)) then 5
                                        when :aarep_llc_k_deliq_days between 91 and 180 then 6
                                        when :aarep_llc_k_deliq_days > 180 then 7
                                      end;
            end
          end
          else
          begin
            /*Классификация для AZ БД*/

            /*select cs_id from classific
              where cs_day_max<>-1
              and cs_day_min<=:aarep_llc_k_deliq_days
              and cs_day_max>=:aarep_llc_k_deliq_days
              into :aarep_llc_status_temp; */
            /*Присваивает статусы классификации для групповых кредитов*/
            if (:aarep_llc_loan_type = 0) then
            begin
              aarep_llc_status_temp = case
                                        when :aarep_llc_k_deliq_days <= 30 then 3
                                        when :aarep_llc_k_deliq_days between 31 and 180 then 6
                                        when :aarep_llc_k_deliq_days > 180 then 7
                                      end;
            end
            /*Присваивает статусы классификации для индивидуальных кредитов*/
            if (:aarep_llc_loan_type = 1) then
            begin
              aarep_llc_status_temp = case
                                        when :aarep_llc_k_deliq_days <= 30 then 3
                                        when :aarep_llc_k_deliq_days between 31 and 180 then 5
                                        when :aarep_llc_k_deliq_days between 181 and 365 then 6
                                        when :aarep_llc_k_deliq_days > 365 then 7
                                      end;
            end
          end
        end

        aarep_llc_status = aarep_llc_status_temp + aarep_need_one_up;
        /*      if ((AAREP_LLC_STATUS<5) and (aarep_llc_loan_type=1)) then */
--          if (AAREP_LLC_STATUS<5) then
--          begin
--                /* Проверка на реструктуризацию */
--            if (exists(select ol_id from operlist, operrest
--                       where or_from_ol_id=ol_id
--                       and operlist.is_delete=0
--                       and or_from_k_id=:aarep_llc_k_id
--                       and ol_date<=:REPORT_DATE_END)) then
--            begin
--              AAREP_LLC_STATUS=5;
--              aarep_memo='Реструктурированный кредит';
--            end
--          end

        suspend;
        /*   aarep_llc_k_deliq_days1=aarep_llc_k_deliq_days;*/
      end
      prior = aarep_llc_k_id;
    end
  end
  else
  begin
    prior = 0;
    for select aarep_k_id, cast(0 as integer), aarep_k_deliq_days, aarep_k_balance, pr_loan_type, kal_new_reg_id,
               k_from_kf_id
        from aarep_kontract_id(:report_region_id, :report_date_begin, :report_date_end, :report_reverse, :report_currency, 5), kontract, knaktlist, product
        where aarep_k_id = k_id and
              k_from_pr_id = pr_id and
              kal_from_kf_id = k_from_kf_id and
              kal_date <= :report_date_end and
              kal_date_end >= :report_date_end
        union
        select ac_k_id, cast(1 as integer), cast(0 as integer), ac_k_balance, ac_loan_type, ac_region_id, ac_kf_id
        from aarep_kontract_active_temp(:report_region_id, :report_date_end, :report_currency, :report_reverse)
        order by 1, 2
        into :aarep_llc_k_id, :kind, :aarep_llc_k_deliq_days, :aarep_llc_k_balance, :aarep_llc_loan_type,
             :aarep_llc_region_id, :kf_id
    do
    begin
      if (prior <> aarep_llc_k_id) then
      begin
        aarep_memo = '';
        if (aarep_llc_k_deliq_days is null) then
          aarep_llc_k_deliq_days = 0;
        select cs_id, cs_is_for_group
        from classific
        where cs_day_max <> -1 and
              cs_day_min <= :aarep_llc_k_deliq_days and
              cs_day_max >= :aarep_llc_k_deliq_days
        into :aarep_llc_status_temp, :aarep_percent;
        aarep_llc_status = aarep_llc_status_temp;
        suspend;
      end
      prior = aarep_llc_k_id;
    end
  end

end
