CREATE PROCEDURE AAREP_EFFECTIVENESS_WEEK(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_NAME VARCHAR(50),
    WEEK_1_DISB_SVZ_ONE CURR_TYPE,
    WEEK_2_DISB_SVZ_ONE CURR_TYPE,
    WEEK_3_DISB_SVZ_ONE CURR_TYPE,
    WEEK_4_DISB_SVZ_ONE CURR_TYPE,
    WEEK_5_DISB_SVZ_ONE CURR_TYPE,
    WEEK_6_DISB_SVZ_ONE CURR_TYPE,
    WEEK_1_SVZ_CLI_AVG CURR_TYPE,
    WEEK_2_SVZ_CLI_AVG CURR_TYPE,
    WEEK_3_SVZ_CLI_AVG CURR_TYPE,
    WEEK_4_SVZ_CLI_AVG CURR_TYPE,
    WEEK_5_SVZ_CLI_AVG CURR_TYPE,
    WEEK_6_SVZ_CLI_AVG CURR_TYPE,
    WEEK_1_SVZ_KOL INTEGER,
    WEEK_2_SVZ_KOL INTEGER,
    WEEK_3_SVZ_KOL INTEGER,
    WEEK_4_SVZ_KOL INTEGER,
    WEEK_5_SVZ_KOL INTEGER,
    WEEK_6_SVZ_KOL INTEGER)
AS
begin
  for select office_name, sum(week_1_disb_svz_one), sum(week_2_disb_svz_one), sum(week_3_disb_svz_one),
             sum(week_4_disb_svz_one), sum(week_5_disb_svz_one), sum(week_6_disb_svz_one), sum(week_1_svz_cli_avg),
             sum(week_2_svz_cli_avg), sum(week_3_svz_cli_avg), sum(week_4_svz_cli_avg), sum(week_5_svz_cli_avg),
             sum(week_6_svz_cli_avg), sum(week_1_svz_kol), sum(week_2_svz_kol), sum(week_3_svz_kol),
             sum(week_4_svz_kol), sum(week_5_svz_kol), sum(week_6_svz_kol)
      from aarep_effectiveness_week_tmp(:rep_region_id, :rep_date_end)
      group by office_name
      into :office_name, :week_1_disb_svz_one, :week_2_disb_svz_one, :week_3_disb_svz_one, :week_4_disb_svz_one,
           :week_5_disb_svz_one, :week_6_disb_svz_one, :week_1_svz_cli_avg, :week_2_svz_cli_avg, :week_3_svz_cli_avg,
           :week_4_svz_cli_avg, :week_5_svz_cli_avg, :week_6_svz_cli_avg, :week_1_svz_kol, :week_2_svz_kol,
           :week_3_svz_kol, :week_4_svz_kol, :week_5_svz_kol, :week_6_svz_kol
  do
  begin
    if (week_1_svz_kol <> 0) then
    begin
      week_1_disb_svz_one = cast(week_1_disb_svz_one as double precision) / week_1_svz_kol;
      week_1_svz_cli_avg = cast(week_1_svz_cli_avg as double precision) / week_1_svz_kol;
    end
    else
    begin
      week_1_disb_svz_one = 0;
      week_1_svz_cli_avg = 0;
    end
    if (week_2_svz_kol <> 0) then
    begin
      week_2_disb_svz_one = cast(week_2_disb_svz_one as double precision) / week_2_svz_kol;
      week_2_svz_cli_avg = cast(week_2_svz_cli_avg as double precision) / week_2_svz_kol;
    end
    else
    begin
      week_2_disb_svz_one = 0;
      week_2_svz_cli_avg = 0;
    end
    if (week_3_svz_kol <> 0) then
    begin
      week_3_disb_svz_one = cast(week_3_disb_svz_one as double precision) / week_3_svz_kol;
      week_3_svz_cli_avg = cast(week_3_svz_cli_avg as double precision) / week_3_svz_kol;
    end
    else
    begin
      week_3_disb_svz_one = 0;
      week_3_svz_cli_avg = 0;
    end
    if (week_4_svz_kol <> 0) then
    begin
      week_4_disb_svz_one = cast(week_4_disb_svz_one as double precision) / week_4_svz_kol;
      week_4_svz_cli_avg = cast(week_4_svz_cli_avg as double precision) / week_4_svz_kol;
    end
    else
    begin
      week_4_disb_svz_one = 0;
      week_4_svz_cli_avg = 0;
    end
    if (week_5_svz_kol <> 0) then
    begin
      week_5_disb_svz_one = cast(week_5_disb_svz_one as double precision) / week_5_svz_kol;
      week_5_svz_cli_avg = cast(week_5_svz_cli_avg as double precision) / week_5_svz_kol;
    end
    else
    begin
      week_5_disb_svz_one = 0;
      week_5_svz_cli_avg = 0;
    end
    if (week_6_svz_kol <> 0) then
    begin
      week_6_disb_svz_one = cast(week_6_disb_svz_one as double precision) / week_6_svz_kol;
      week_6_svz_cli_avg = cast(week_6_svz_cli_avg as double precision) / week_6_svz_kol;
    end
    else
    begin
      week_6_disb_svz_one = 0;
      week_6_svz_cli_avg = 0;
    end
    suspend;
  end
end
