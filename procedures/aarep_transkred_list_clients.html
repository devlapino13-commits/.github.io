    K_ID VARCHAR(18),
    K_FROM_KF_ID DECIMAL(18,0),
    K_CURRENCY VARCHAR(3),
    K_APPLICATE_DATE DATE,
    K_APPLICATE_NO INTEGER,
    K_KONTRACT_NO VARCHAR(20),
    K_KONTRACT_DATE DATE,
    K_FROM_PR_ID INTEGER,
    K_DISB_DATE DATE,
    K_DISB_YEAR INTEGER,
    K_DISB_MONTH INTEGER,
    K_AMOUNT_APPLICATE DECIMAL(18,4),
    K_STAGE INTEGER,
    K_PERIOD DECIMAL(15,2),
    K_KOLICH_LGOTA INTEGER,
    K_PERCENT DECIMAL(18,7),
    K_PERCENT_LGOTA DECIMAL(18,7),
    K_PERCENT_REAL DECIMAL(18,7),
    K_KOLICH_REPAY INTEGER,

    KF_OFFICER_CODE VARCHAR(20),

    KF_DATE_IN_TO_KRS DATE,
    KF_DATE_OUT_FROM_KRS DATE,
    KF_DATE_CREATE DATE,
    KF_MEMO VARCHAR(2048),
    TRANS_DATE DATE,
    TRANS_YEAR INTEGER,
    TRANS_MONTH INTEGER,
    TRANS_DOC_NO INTEGER,
    TRANS_REGION_ID INTEGER,
    TRANS_REGION_NAME VARCHAR(100),
    TRANS_TYPE INTEGER,
    TRANS_TYPE_TEXT VARCHAR(50),
    TRANS_DISB_OS DECIMAL(18,4),
    TRANS_OS DECIMAL(18,4),
    TRANS_PERC DECIMAL(18,4),
    TRANS_FINE DECIMAL(18,4),
    TRANS_KOMMIS DECIMAL(18,4),
    TRANS_KOMMIS_ADV DECIMAL(18,4),
    TRANS_PAY_TYPE INTEGER,
    TRANS_PAY_TYPE_TEXT VARCHAR(20),
    TRANS_OPERLIST_ID DECIMAL(18,0),
    TRANS_PAY_ALL DECIMAL(18,4),

    CL_LEADER_NAME VARCHAR(100),

    CL_LEADER_PASS_DATE DATE,

    TRANS_DATETIME TIMESTAMP,
    TRANS_OTHER_DAY VARCHAR(20),
    TRANS_WEEKEND VARCHAR(20),
    CL_ID DECIMAL(18,0),
    CL_NAME VARCHAR(100),
    CL_PASS_NO VARCHAR(30),
    CL_PASS_DATE DATE,
    CL_PASS_WHO VARCHAR(50),
    CL_FULL_ADDRESS VARCHAR(255),
    BT_NAME VARCHAR(50),
    BS_NAME VARCHAR(50),
    REG_STAT_REP_REGION VARCHAR(255))
AS
declare variable pr_is_secured integer;
declare variable pr_int_calc_type integer;
declare variable k_is_kredline integer;
begin
-- knclient_fix

--  if(f_get_db_id() in (39801, 41701)) then           --Блок отключен из-за переноса БД на сервер отчетности
--  begin
--    if (extract(hour from current_timestamp) between 8 and 16) then
--        exception e_message 'Отчет не оптимизирован. Допустимо запускать либо до 8:00 либо после 17:00';
--  end

  for select tr.trans_kontract_id, k_from_kf_id, k_currency, k_applicate_date, k_applicate_no, k_kontract_no,
             k_kontract_date, k_from_pr_id, k_disb_date, k_amount_applicate, k_stage, k_period, k_kolich_lgota,
             k_percent, k_percent_lgota, k_kolich_repay, k_from_f_id, k_from_pt_id, k_circle, k_circle_internal,
             k_percent_per_day, k_memo, k_amount_approved, k_percent_for_days, k_kolich_client_applicate,
             k_kolich_client_approved, k_repay_kolich_closed, k_class, k_kommis_perc, k_from_cs_id, k_credit_target,
             k_is_kredline, k_kredline_end, kakt1.kal_new_kf_name, kakt1.kal_new_no, r1.reg_unicode, r1.reg_name,
             r2.reg_unicode, r2.reg_name, kakt1.kal_new_kf_officer, kakt1.kal_new_officer_code, c1.c_name_fio,
             kakt1.kal_date, kakt1.kal_date_end, kakt2.kal_date, kf_memo, tr.trans_date, tr.trans_doc_no,
             tr.trans_region_id, r3.reg_name, tr.trans_type, tr.trans_type_text,
             cast(tr.trans_disb_os as double precision) / k_amount_approved * kc_amount_approved,
             cast(tr.trans_os as double precision) / k_amount_approved * kc_amount_approved,
             cast(tr.trans_perc as double precision) / k_amount_approved * kc_amount_approved,
             cast(tr.trans_fine as double precision) / k_amount_approved * kc_amount_approved,
             cast(tr.trans_kommis as double precision) / k_amount_approved * kc_amount_approved,
             cast(tr.trans_kommis_adv as double precision) / k_amount_approved * kc_amount_approved, tr.trans_pay_type,
             tr.trans_pay_type_text, tr.trans_operlist_id,
             cast(tr.trans_pay_all as double precision) / k_amount_approved * kc_amount_approved, k_clients_leader_id,
             c2.c_name_fio, c2.c_pass_no, c2.c_pass_date, substring(c2.c_pass_who from 1 for 50), k_clients_cashier_id,
             c3.c_name_fio, c3.c_pass_no, c3.c_pass_date, substring(c3.c_pass_who from 1 for 50), pr_code, pr_name,
             pr_loan_type, pr_is_secured, pr_interest_method, pr_percent_for_days, tr.trans_other_client,
             tr.trans_datetime, tr.trans_other_day, c5.c_id, c5.c_name_fio, c5.c_pass_no, c5.c_pass_date,
             substring(c5.c_pass_who from 1 for 50), bt_name, bs_name, r1.reg_stat_rep_region

      from aarep_transkred_id(:report_region_id, :report_date_begin, :report_date_end, :report_currency) tr
      left join region r3 on (tr.trans_region_id = r3.reg_unicode)

      left join(knaktlist kakt1 left join clients c1 on (kakt1.kal_new_kf_officer = c1.c_id) left join(region r1 left join region r2 on (r1.parent = r2.reg_unicode)) on (kakt1.kal_new_reg_id = r1.reg_unicode)) on (tr.trans_kf_id = kakt1.kal_from_kf_id and
            kakt1.kal_date <= tr.trans_date and
            kakt1.kal_date_end >= tr.trans_date)

      left join(kontract left join clients c2 on (kontract.k_clients_leader_id = c2.c_id) left join clients c3 on (kontract.k_clients_cashier_id = c3.c_id) left join product on (kontract.k_from_pr_id = pr_id) left join kfolder on (kontract.k_from_kf_id = kf_id) left join knaktlist kakt2 on kontract.k_from_kf_id = kakt2.kal_from_kf_id and
            kakt2.kal_from_ka_id = 0 and
            kakt2.kal_number = 1 left join(knclient left join(bussubtype left join bustype on (bs_from_bt_id = bt_id)) on (bs_id = kc_busuness_subtype) left join clients c5 on (kc_from_c_id = c5.c_id)) on (kontract.k_id = kc_from_k_id and
            kc_amount_approved <> 0)) on (k_id = tr.trans_kontract_id)

      into :k_id, :k_from_kf_id, :k_currency, :k_applicate_date, :k_applicate_no, :k_kontract_no, :k_kontract_date,
           :k_from_pr_id, :k_disb_date, :k_amount_applicate, :k_stage, :k_period, :k_kolich_lgota, :k_percent,
           :k_percent_lgota, :k_kolich_repay, :k_from_f_id, :k_from_pt_id, :k_circle, :k_circle_internal,
           :k_percent_per_day, :k_memo, :k_amount_approved, :k_percent_for_days, :k_kolich_client_applicate,
           :k_kolich_client_approved, :k_repay_kolich_closed, :k_risk_category, :k_kommis_perc, :k_from_cs_id,
           :k_credit_target, :k_is_kredline, :k_kredline_end, :kf_name, :kf_number, :kf_region_id, :kf_region_name,
           :kf_branch_id, :kf_branch_name, :kf_officer_id, :kf_officer_code, :kf_officer_name, :kf_date_in_to_krs,
           :kf_date_out_from_krs, :kf_date_create, :kf_memo, :trans_date, :trans_doc_no, :trans_region_id,
           :trans_region_name, :trans_type, :trans_type_text, :trans_disb_os, :trans_os, :trans_perc, :trans_fine,
           :trans_kommis, :trans_kommis_adv, :trans_pay_type, :trans_pay_type_text, :trans_operlist_id, :trans_pay_all,
           :cl_leader_id, :cl_leader_name, :cl_leader_pass_no, :cl_leader_pass_date, :cl_leader_pass_who,
           :cl_cashier_id, :cl_cashier_name, :cl_cashier_pass_no, :cl_cashier_pass_date, :cl_cashier_pass_who, :pr_code,
           :pr_name, :pr_loan_type, :pr_is_secured, :pr_int_calc_type, :pr_int_for_days, :trans_other_client,
           :trans_datetime, :trans_other_day, :cl_id, :cl_name, :cl_pass_no, :cl_pass_date, :cl_pass_who, :bt_name,
           :bs_name, :reg_stat_rep_region

  do
  begin
    select kd_percent, kd_percent_per_day
    from kontdata
    where kd_ol_is_delete = 0 and
          :trans_date between kd_ol_date and kd_date_end and
          kd_from_k_id = :k_id
    into k_percent, k_percent_per_day;

    cl_leader_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_leader_id and
          adc_from_adl_id = adl_id
    into :cl_leader_full_address;

    cl_cashier_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_cashier_id and
          adc_from_adl_id = adl_id
    into :cl_cashier_full_address;

    cl_full_address = '';
    select adl_address_str
    from addrclient, addrlist
    where adc_address_type = 1 and
          adc_from_c_id = :cl_id and
          adc_from_adl_id = adl_id
    into :cl_full_address;

    k_disb_year = extract(year from k_disb_date);
    k_disb_month = extract(month from k_disb_date);
    trans_year = extract(year from trans_date);
    trans_month = extract(month from trans_date);

    if (k_is_kredline = 0) then
      k_is_kredline_str = 'Нет';
    if (k_is_kredline = 1) then
      k_is_kredline_str = 'Да';

    pr_loan_type_str = '';
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Группа';
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    pr_is_secured_str = '';
    if (pr_is_secured = 0) then
      pr_is_secured_str = 'Без залога';
    if (pr_is_secured = 1) then
      pr_is_secured_str = 'Залог';
    if (pr_is_secured = 2) then
      pr_is_secured_str = 'Поручители';
    if (pr_is_secured = 3) then
      pr_is_secured_str = 'Залог,поручит.';

    pr_int_calc_type_str = '';
    if (pr_int_calc_type = 0) then
      pr_int_calc_type_str = 'На сумму кредита';
    if (pr_int_calc_type = 1) then
      pr_int_calc_type_str = 'На остаток';
    if (pr_int_calc_type = 2) then
      pr_int_calc_type_str = 'Аннуитет';

    k_circle_real = :k_circle + :k_circle_internal - 1;
    k_percent_real = :k_percent - :k_percent_lgota;

    if ((extract(weekday from :trans_date) = 0) or (extract(weekday from :trans_date) = 6)) then
      trans_weekend = 'Выходной день';
    else
      trans_weekend = 'Рабочий день';
    suspend;
  end
end
