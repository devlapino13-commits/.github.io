CREATE PROCEDURE OKT_WEBSITE_VERIF_UPDATE(
    ACL_ID ID_TYPE)
AS
declare variable acl_object_id id_type;
declare variable acl_object_kind integer;
declare variable acl_status integer;
declare variable acl_from_un_id id_type;
declare variable acl_parent_id id_type;
declare variable res_has_code integer;
begin

  res_has_code = 0;

  for select al.acl_object_id, al.acl_object_kind, al.acl_status, al.acl_from_un_id, al.acl_parent_id
      from actionlist al
      where al.acl_id = :acl_id and
            al.acl_parent_id is not null
      into :acl_object_id, :acl_object_kind, :acl_status, :acl_from_un_id, :acl_parent_id
  do
  begin
    if (acl_status = 2) then
    begin
      select res_has_code
      from hp_objectprops_anketa_get(:acl_object_kind, :acl_object_id, -1, 'AnketaQuestions.OKTELL.WebSite.Result', current_date, 'CallCentreCheckStatus.Interest')
      into :res_has_code;

      res_has_code = coalesce(res_has_code, 0);

      if (res_has_code <> 0) then
      begin
        update actionlist al
        set acl_status = 2,
            acl_result_rd_id = (select rd.rd_id
                                from refdata rd
                                where rd.rd_code = 'CallCentreCheckStatus.Interest'),
            al.acl_from_un_id = :acl_from_un_id
        where al.acl_id = :acl_parent_id and
              al.acl_parent_id is null;
      end
      else
      begin
        select res_has_code
        from hp_objectprops_anketa_get(:acl_object_kind, :acl_object_id, -1, 'AnketaQuestions.OKTELL.WebSite.Result', current_date, 'CallCentreCheckStatus.NotInterest')
        into :res_has_code;

        res_has_code = coalesce(res_has_code, 0);

        if (res_has_code <> 0) then
        begin
          update actionlist al
          set acl_status = 2,
              acl_result_rd_id = (select rd.rd_id
                                  from refdata rd
                                  where rd.rd_code = 'CallCentreCheckStatus.NotInterest'),
              al.acl_from_un_id = :acl_from_un_id
          where al.acl_id = :acl_parent_id and
                al.acl_parent_id is null;
        end
      end
    end
  end
end
