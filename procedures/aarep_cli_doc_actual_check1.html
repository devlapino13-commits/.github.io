CREATE PROCEDURE AAREP_CLI_DOC_ACTUAL_CHECK1(
    REGION_ID D_INTEGER = 1000000,
    REP_DATE D_DATE = current_date)
RETURNS (
    NUMB D_INTEGER,
    C_ID D_ID,
    C_NAME_FIO STR_100,
    C_BIRTH_DATE D_DATE,
    C_IS_RESIDENT D_INTEGER,
    RESIDENT D_STR_10,
    KS_STATE D_INTEGER,
    STATE D_STR_10,
    DOC_TYPE D_STR_20,
    DOC_DATE D_DATE,
    DOC_DATE_EXPIRE D_DATE,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    K_KFOLDER_NAME D_STR_100,
    OFFICE_NAME D_STR_CODE,
    CPDS_TIMESTAMP D_DATETIME)
AS
declare variable reg_id d_integer;
declare variable kf_id d_id;
declare variable fil_id d_integer;
declare variable fic_id d_integer;
declare variable fic_date d_date;
declare variable fiv_value d_str_long;
declare variable dt_finish d_date;
begin
  numb = 0;
  for
    select
      k_last_reg_id,
      k_from_kf_id,
      kal_new_kf_name,
      k_id,
      k_kontract_no,
      ks_state,
      k_disb_date,
      kc_from_c_id
  from kontstate
    inner join kontract
      on k_id = ks_from_k_id
    inner join knclient
      on kc_from_k_id = k_id
        and kc_amount_approved > 0
        and not exists(select * from temp_rep where id = kc_from_c_id)
    inner join knaktlist
      on kal_from_kf_id = k_from_kf_id
        and :rep_date between kal_date and kal_date_end
        and knaktlist.is_delete = 0
    inner join region_list(:region_id, 1, -1, 1)
      on id_list = kal_new_reg_id
  where :rep_date between ks_date_begin and ks_date_end
    and ks_state in (1, 2)
    and ks_ol_is_delete = 0
  order by k_disb_date
  into
    :reg_id,
    :kf_id,
    :k_kfolder_name,
    :k_id,
    :k_kontract_no,
    :ks_state,
    :k_disb_date,
    :c_id
  do
  begin
    if (ks_state = 2 and not exists(
      select oko_from_k_id
      from operkroff
      where oko_from_k_id = :k_id
        and oko_ol_is_active = 1
      group by oko_from_k_id
      having sum(oko_os + oko_sh_os + oko_perc + oko_sh_perc) > 0))
    then continue;

    for
      select
        c_id,
        c_name_fio,
        c_birth_date,
        c_is_resident,
        c_pass_date,
        c_pass_date_expire
      from clients
      where c_id = :c_id
    into
      :c_id,
      :c_name_fio,
      :c_birth_date,
      :c_is_resident,
      :doc_date,
      :doc_date_expire
    do
    begin
      select first 1
        fil_id
      from ft_infolist
      where fil_from_c_id = :c_id
      order by fil_id desc
      into
        :fil_id;

      select first 1
        fic_id,
        fic_date
      from ft_infocheck
      where fic_from_fil_id = :fil_id
      order by fic_id desc
      into
        :fic_id,
        :fic_date;

      select
        fiv_value
      from hp_ft_infofield_value(:fic_id, :fil_id, :fic_date, 'relation_finished')
      into
        :fiv_value;

      dt_finish = iif(fiv_value = '', '31.12.9999', fiv_value);
      if (dt_finish <= rep_date) then
        continue;

      doc_type = 'Паспорт';
      if (c_is_resident = 0) then
      begin
        select first 1
          iif(cds_from_pk_id = 39, 'ВНЖ', 'РВП'),
          cds_doc_date_begin,
          cds_doc_date_end
        from clientdocs
        where cds_from_c_id = :c_id
          and cds_from_pk_id in (39, 46)
          and cds_is_delete = 0
          and cds_verify_status = 2
        order by cds_datetime desc
        into
          :doc_type,
          :doc_date,
          :doc_date_expire;
      end

      if (rep_date between doc_date and coalesce(doc_date_expire, '31.12.9999'))
        then continue;

      resident = iif(c_is_resident = 1, '+', '');
      state = iif(ks_state = 2, '+', '');
      office_name = (
        select
          reg_name
        from region
        where reg_unicode = :reg_id);
      cpds_timestamp = (
        select first 1
          cpds_timestamp
        from clipassdelaysms
        where cpds_c_id = :c_id
        order by cpds_timestamp desc);
      numb = numb + 1;

      insert into temp_rep(id) values(:c_id);
      suspend;
    end
  end
end
