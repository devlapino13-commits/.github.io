CREATE PROCEDURE AAREP_MINIAPP_DISB_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_AUTO_PERIOD INTEGER,
    REP_AUTO_KOEF DECIMAL(18,7),
    REP_INCLUDE_FIRST_DAY INTEGER)
RETURNS (
    MA_DATE_OPER DATE,
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_SUMMA_MINUS_REPEAT DECIMAL(18,4),
    MA_SUMMA_MINUS_NEW DECIMAL(18,4),
    MA_SUMMA_MINIAPPNEW DECIMAL(18,4),
    MA_REGION_ID INTEGER,
    MA_KF_ID DECIMAL(18,0),
    MA_DISB_KOLICH INTEGER,
    MA_FACT_DATE DATE,
    MA_FACT_DISB_AMOUNT DECIMAL(18,4),
    MA_PLAN_DISB_ALL DECIMAL(18,4),
    MA_PRODUCT VARCHAR(100),
    MA_CIRCLE INTEGER,
    MA_UNIQ_ID INTEGER)
AS
begin
  ma_product = '';
  ma_circle = 0;
  for select ma_date_oper, ma_summa_minus, ma_summa_minus_repeat, ma_summa_minus_new, ma_summa_miniappnew, ma_region_id,
             ma_kf_id, ma_disb_kolich, ma_uniq_id
      from aarep_miniapp_temp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_auto_period, :rep_auto_koef, :rep_include_first_day, 0)
      where ma_summa_minus <> 0 or
            ma_summa_minus_repeat <> 0 or
            ma_summa_minus_new <> 0 or
            ma_summa_miniappnew <> 0
      into :ma_date_oper, :ma_summa_minus, :ma_summa_minus_repeat, :ma_summa_minus_new, :ma_summa_miniappnew,
           :ma_region_id, :ma_kf_id, :ma_disb_kolich, :ma_uniq_id
  do
  begin
    ma_plan_disb_all = ma_summa_minus + ma_summa_minus_repeat + ma_summa_minus_new + ma_summa_miniappnew;

    ma_fact_date = null;
    ma_fact_disb_amount = 0;
    select ks_date_begin, k_amount_approved, pr_common_name, k_circle
    from kontract, kontstate, product
    where ((ks_date_begin between(:rep_date_begin - 30) and (:rep_date_begin - 1)) or
          (ks_date_begin between(:rep_date_end + 1) and (:rep_date_end + 30))) and
          ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_from_k_id = k_id and
          k_from_kf_id = :ma_kf_id and
          k_from_pr_id = pr_id rows 1
    into :ma_fact_date, :ma_fact_disb_amount, :ma_product, :ma_circle;

    suspend;
  end
  ma_plan_disb_all = 0;
  ma_date_oper = null;
  ma_summa_minus = 0;
  ma_summa_minus_repeat = 0;
  ma_summa_minus_new = 0;
  ma_summa_miniappnew = 0;
  ma_disb_kolich = 0;
  ma_uniq_id = 0;
  for select k_from_kf_id, ks_date_begin, k_amount_approved, kal_new_reg_id, pr_common_name, k_circle
      from kontract, kontstate, knaktlist, region_list(:rep_region_id, 1, -1, 1), product
      where ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_state = 1 and
            ks_from_k_id = k_id and
            ks_ol_is_delete = 0 and
            kal_from_kf_id = k_from_kf_id and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            knaktlist.is_delete = 0 and
            (kal_new_reg_id = id_list) and
            k_from_pr_id = pr_id
      into :ma_kf_id, :ma_fact_date, :ma_fact_disb_amount, :ma_region_id, :ma_product, :ma_circle
  do
  begin
    suspend;
  end
end
