CREATE PROCEDURE AAREP_KS_VS_HR(
    REP_REGION INTEGER,
    REP_DATE_END DATE,
    REP_CURRENCY VARCHAR(3))
RETURNS (
    REG_UNICODE INTEGER,
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    KF_OFFICER DECIMAL(18,0),
    KF_OFFICER_NAME VARCHAR(100),
    MEMO VARCHAR(100),
    PS_CURR_STATUS VARCHAR(100),
    REP_ACTIVE_BALANCE DECIMAL(18,4))
AS
declare variable is_porfel integer;
declare variable rep_active_balance_1 decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select distinct reg_unicode, /*Вытаскивает КСников у которых есть активный порфель но не существует в кадровом отчете*/
                      reg_code || ' ' || reg_name, parent_name, kal_new_kf_officer, c_name_fio,
                      cast('Имеется активный портфель но в Кадровом отчете отсутствует' as varchar(100)),
                      cast(1 as integer)
      from kontstate, kontract, knaktlist, region, region_list(:rep_region, 1, 3, 1), clients
      where ks_ol_is_delete = 0 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_state = 1 and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            (reg_unicode = id_list) and
            kal_new_reg_id = reg_unicode and
            reg_status = 3 and
            c_id = kal_new_kf_officer and
            kal_new_kf_officer not in (select ps_from_c_id
                                       from hr_pstatus, joblist, jobkind
                                       where ps_from_jl_id = jl_id and
                                             jl_kind = jk_id and
                                             jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
                                             :rep_date_end between ps_date_begin and ps_date_end and
                                             (ps_from_reg_id = region.reg_unicode) and
                                             ps_is_working = 1)

      union

      select reg_unicode, /*Вытаскивает КСников у которых нет активного займа но в кадровом отчете существует*/
             reg_code || ' ' || reg_name, parent_name, ps_from_c_id, c_name_fio,
             cast('В кадровом отчете отражается но не имеется активный портфель' as varchar(100)),
             cast(0 as integer)
      from hr_pstatus, joblist, region, region_list(:rep_region, 1, 3, 1), clients, jobkind
      where ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            :rep_date_end between ps_date_begin and ps_date_end and
            (reg_unicode = id_list) and
            ps_from_reg_id = reg_unicode and
            reg_status = 3 and
            ps_is_working = 1 and
            c_id = ps_from_c_id and
            ps_from_c_id not in (select distinct kal_new_kf_officer
                                 from kontstate, kontract, knaktlist
                                 where ks_ol_is_delete = 0 and
                                       :rep_date_end between ks_date_begin and ks_date_end and
                                       ks_state = 1 and
                                       ks_from_k_id = k_id and
                                       k_from_kf_id = kal_from_kf_id and
                                       knaktlist.is_delete = 0 and
                                       :rep_date_end between kal_date and kal_date_end and
                                       kal_new_reg_id = region.reg_unicode)

      into :reg_unicode, :reg_name, :parent_name, :kf_officer, :kf_officer_name, :memo, :is_porfel

  do
  begin
    rep_active_balance = 0;
    rep_active_balance_1 = 0;
    ps_curr_status = '';
    if (is_porfel = 1) then
    begin /*Считает баланс кредита у которых есть активный займ*/
      for select sum(
                 case
                   when :rep_currency = '000' then (case
                                                      when cr_rate < k_index_curr_rate then oks_balance * k_index_curr_rate /*Вычисляет активный портфель текущего сотрудника*/
                                                      else oks_balance * cr_rate
                                                    end)
                   when :rep_currency = '---' then oks_balance
                   when :rep_currency = k_index_curr_code then oks_balance
                 end)
          from oksummary, kontract, opercurr
          where oks_from_k_id = k_id and
                k_index_curr_code = cr_from_cu_id and
                oks_date_begin <= :rep_date_end and
                oks_date_end >= :rep_date_end and
                :rep_date_end between cr_date and cr_date_end and
                ((:rep_currency = '---') or
                (:rep_currency = '000') or
                (:rep_currency = k_index_curr_code)) and
                oks_region_id = :reg_unicode and
                oks_kc_id = :kf_officer and
                oks_balance > 0 and
                oks_close_state = 0
          into :rep_active_balance_1
      do
      begin
        rep_active_balance = rep_active_balance_1;
      end
    end
    else
    begin /*Вытаскивает текущий статус по кадровым приказам КСника*/
      select sawt_name
      from hr_voctravel, hr_orders, sa_worktype
      where hrv_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            hrv_from_c_id = :kf_officer and
            hrv_status = 1 and
            :rep_date_end between hrv_date_begin and hrv_date_end_fact
      order by hrv_date_begin desc
      rows 1
      into :ps_curr_status;
    end
    suspend;
  end
end
