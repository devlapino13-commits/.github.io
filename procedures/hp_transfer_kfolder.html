CREATE PROCEDURE HP_TRANSFER_KFOLDER(
    REG_ID D_INTEGER,
    REP_DATE D_DATE,
    KA_MEMO D_STR_2048,
    KAL_FROM_KF_ID ID_TYPE,
    KAL_NEW_KF_OFFICER ID_TYPE,
    KAL_NEW_KF_NAME STR_100,
    KAL_NEW_NO D_INTEGER,
    KAL_NEW_OFFICER_CODE STR_20,
    C_ID ID_TYPE,
    ONLY_MAIL D_SMALLINT)
AS
declare variable ka_id id_type;
declare variable ka_akt_no str_20;
declare variable c_name_fio str_100;
declare variable lml_email str_255;
declare variable email_body d_str_long;
begin
  if (only_mail = 0) then
  begin
    ka_id = (select kv
             from get_kv(1010100, 7));
    ka_akt_no = coalesce((select max(cast(ka_akt_no as integer))
                          from knakt
                          where ka_from_reg_id = :reg_id), 0) + 1;
    insert into knakt (ka_id, ka_date, ka_from_reg_id, ka_akt_no, ka_memo, is_delete, ka_from_ol_id)
    values (:ka_id, :rep_date, :reg_id, :ka_akt_no, :ka_memo, 0, null);

    insert into knaktlist (kal_from_ka_id, kal_number, kal_from_kf_id, kal_date, kal_date_end, kal_new_kf_officer,
                           kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no, kal_new_officer_code)
    values (:ka_id, 1, :kal_from_kf_id, :rep_date, '31.12.9999', :kal_new_kf_officer,
            :kal_new_kf_name || :kal_new_officer_code, 0, :reg_id, :kal_new_no, :kal_new_officer_code);
  end

  if (only_mail = 1) then
  begin
    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    select coalesce(list(distinct lml_email, '; ') || '; ', '') || 'y.pankova@molbulak.com; r.ilalov@molbulak.com'
    from loanmgrlist
    where lml_from_reg_id = :reg_id and
          lml_email is not null and
          lml_date_end >= current_date
    into lml_email;

    email_body = '<b>Внимание!</b><br><br><br>Клиент офиса: <b>' || c_name_fio || ', кр.папка ' || kal_new_kf_name || ' – банкрот.<br><br>С этого момента с клиентом - банкротом запрещено взаимодействовать по взысканию долга: звонить, отправлять любые сообщения, письма, уведомления, осуществлять выезды.</b>';
    insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                                msg_command_kind, msg_object_kind, msg_object_id)
    values (0, :lml_email, 'Клиент - банкрот', :email_body, 7, 1000000000000, '', 0, 0);
  end
end
