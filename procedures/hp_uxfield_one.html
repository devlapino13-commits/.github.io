CREATE PROCEDURE HP_UXFIELD_ONE(
    TABLE_NAME D_STR_CODE,
    FIELD_NAME D_STR_CODE)
RETURNS (
    RES_POSITION D_INTEGER,
    RES_TABLE_NAME D_STR_CODE,
    RES_FIELD_NAME D_STR_CODE,
    RES_DOMAIN D_STR_CODE,
    RES_FIELD_SIZE D_INTEGER,
    RES_FIELD_TYPE D_STR_CODE,
    RES_COMMENT D_STR_4096,
    RES_IS_KEY_FIELD D_BOOL,
    RES_IS_FK_FIELD D_BOOL,
    RES_NOT_NULL D_BOOL,
    RES_DEFAULT D_STR_255,
    RES_FK_TABLE D_STR_CODE,
    RES_FK_FIELD D_STR_CODE,
    RES_JSON_PAIR_TYPE D_STR_CODE)
AS
declare variable res_field_type_id integer;
declare variable res_field_sub_type integer;
declare variable res_field_scale integer;
begin
  table_name = upper(trim(table_name));
  field_name = upper(trim(field_name));
  res_table_name = lower(table_name);
  res_is_fk_field = 0;
  for select rdb$field_position, trim(r.rdb$field_name), trim(left(r.rdb$description, 4096)),
             f.rdb$field_length, f.rdb$field_type, f.rdb$field_scale, coalesce(f.rdb$field_sub_type, 0),
             coalesce(r.rdb$null_flag, 0), left(r.rdb$default_source, 100),
             lower(f.rdb$field_name)
      from rdb$fields f, rdb$relation_fields r
      where f.rdb$field_name = r.rdb$field_source and
            r.rdb$system_flag = 0 and
            r.rdb$relation_name = :table_name and
            r.rdb$field_name = :field_name
      order by r.rdb$field_position
      into res_position, :res_field_name, res_comment, :res_field_size, :res_field_type_id,
           :res_field_scale, :res_field_sub_type, :res_not_null, :res_default, :res_domain
  do
  begin
    res_field_type = '';
    res_default = replace(res_default, 'default ', '');
    res_default = replace(res_default, '''', '');

    if (res_domain starting 'RDB$') then
      res_domain = '';

    select res_field_type, res_field_size
    from hp_uxfield_type(:res_field_type_id, :res_field_size, :res_field_scale, :res_field_sub_type)
    into :res_field_type, :res_field_size;

    res_is_key_field = 0;

    if (exists(select * from hp_uxfield_get_pk(:table_name)
               where pk_field = :res_field_name)) then
      res_is_key_field = 1;

    res_fk_table = '';
    res_fk_field = '';

    select fk_table, fk_field
    from hp_uxfield_get_fk(:table_name, :res_field_name)
    into :res_fk_table, :res_fk_field;

    if (res_fk_table is null) then
      res_fk_table = '';
    if (res_fk_field is null) then
      res_fk_field = '';

    if (res_fk_field <> '') then
    begin
      res_is_fk_field = 1;
      res_field_type = 'id';
    end

    res_json_pair_type = 'num';

    if (res_field_type in ('string', 'memo', 'blob')) then
      res_json_pair_type = 'str';

    if (res_field_type = 'date') then
      res_json_pair_type = 'date';

    if (res_field_type = 'datetime') then
      res_json_pair_type = 'datetime';

    suspend;
  end
end
