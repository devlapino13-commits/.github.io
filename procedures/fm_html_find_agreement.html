CREATE PROCEDURE FM_HTML_FIND_AGREEMENT(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable number d_integer;
declare variable edc_code d_str_code;
declare variable title d_str_100;
declare variable kc_id d_integer;
declare variable reg_id d_integer;
begin
  res_text = '';

  for select edf.edc_code, t.nnl_from_kc_id,k.k_last_reg_id
      from nbkr_notif_list t
      inner join edoc_cfg edf on edf.edc_id = t.nnl_from_edc_id
      inner join kontract k on k.k_id = t.nnl_k_id
      where t.nnl_c_id = :c_id
      into :edc_code, kc_id,reg_id
  do
  begin
    number = 0;
  title = '';
    if (edc_code = 'nbkr.cli_close_we_owe') then
    begin
      title = 'Согласие';
      number = 1688; -- categ = 'Закрытый; Нам должны';
    end
    else
    if (edc_code = 'nbkr.cli_active_we_owe') then
    begin
      title = 'Доп.соглашение';
      number = 1690; -- categ = 'активный; Нам должны';
    end
    else
    if (edc_code = 'nbkr.cli_close_we_must') then
    begin
      title = 'Согласие';
      number = 1687; -- categ = 'Закрытый; Мы должны';
    end
    else
    if (edc_code = 'nbkr.cli_active_we_must') then
    begin
      title = 'Доп.соглашение';
      number = 1689; --categ = 'Активный; Мы должны';
    end

    if (coalesce(number, 0) > 0) then
    begin
      res_text = res_text||'<h2>' || title || '</h2>';
      res_text = res_text || f_uri_blank('Распечатать ' || title, 'print', 1, :kc_id, number, 'Распечатать', 1, 0, f_format('{"reg_id":%1}', reg_id)) || '<br>';
    end
  end
  suspend;
end
