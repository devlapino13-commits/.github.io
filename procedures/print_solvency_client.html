CREATE PROCEDURE PRINT_SOLVENCY_CLIENT(
    OBJECT_ID ID_TYPE,
    OBJECT_KIND INTEGER)
RETURNS (
    REP_ID ID_TYPE,
    REP_FIO STR_100,
    REP_DTBIRTH DATE,
    REP_PASSDATA STR_255,
    REP_WORK_PLACE STR_255,
    REP_WORK_PLACE_ADD STR_255,
    REP_STATUS D_STR_255,
    REP_OFFICE VARCHAR(50),
    REP_RESSOLV STR_20,
    REP_DTSOLV DATE,
    REP_BASIC_INCOME CURR_TYPE,
    REP_ADD_INCOME CURR_TYPE,
    REP_AVG_INCOME_ROSSTAT CURR_TYPE,
    REP_AVG_INCOME CURR_TYPE,
    REP_RES_INCOME CURR_TYPE,
    REP_EXPENSES CURR_TYPE,
    REP_COMMITMENTS CURR_TYPE,
    REP_ALIVEMIN CURR_TYPE,
    REP_PAY_EXTERNAL CURR_TYPE,
    REP_RES_EXPENSES CURR_TYPE,
    REP_RES_COMMITMENT CURR_TYPE,
    REP_FREE_BALANCE CURR_TYPE,
    REP_AVG_REPAYMENT CURR_TYPE,
    REP_SOLVENCY_RATIO CURR_TYPE)
AS
declare variable K_ID ID_TYPE;
declare variable KC_ID D_ID;
declare variable KC_ANKETA_INCOME_CALC D_CURRENCY;
begin
  if (object_kind = 9) then
    k_id = object_id;
  else
  begin
    select kont.k_id
    from kontract kont
    where kont.k_from_ma_app_id = :object_id rows 1
    into :k_id;
  end

  if (k_id is null) then
    exit;

  for select c_id, c_name_fio, c_birth_date,
             'Номер: ' || c_pass_no || ';' || ' действ.: ' || f_datetostr(c_pass_date) || '-' || f_datetostr(c_pass_date_expire) || ';' || ' выдан: ' || coalesce(c_pass_who_code, '') || ' ' || coalesce(c_pass_who, ''),
             kca_work_place_name, kca_work_place_add_name,
             iif(kca_c_status = 0, 'без статуса', rpad('', kca_c_status, '*')), reg_code || ' ' || reg_name,
             kca_date_verify, kca_work_monthly_income, kca_work_extra_amount, kca_result_income,
             kca_communal_expenses + kca_living_expenses + kca_transfer_to_family + kca_legal_docs_expenses,
             kca_commitment_expenses, kca_alive_minimum, kca_monthly_payments_loans, kca_result_expenses,
             kca_result_commitment, kca_result_income - kca_result_expenses - kca_result_commitment,
             kca_avg_montly_payment, kca_avg_income_rosstat, kca_avg_income, kca_result_solvency,
             iif(kca_result_solvency < 1, 'неплатежеспособен', 'платежеспособен'),
             kc_id
      from knclient
      inner join clients on c_id = kc_from_c_id
      left join knclient_anketa on kca_from_kc_id = kc_id
      left join region on reg_unicode = kca_from_reg_id
      where kc_from_k_id = :k_id
      into :rep_id, :rep_fio, :rep_dtbirth, :rep_passdata, :rep_work_place, :rep_work_place_add, :rep_status,
           :rep_office, :rep_dtsolv, :rep_basic_income, :rep_add_income, :rep_res_income, :rep_expenses,
           :rep_commitments, :rep_alivemin, :rep_pay_external, :rep_res_expenses, :rep_res_commitment,
           :rep_free_balance, :rep_avg_repayment, :rep_avg_income_rosstat, :rep_avg_income, :rep_solvency_ratio,
           :rep_ressolv, :kc_id
  do
  begin
    if (f_get_db_id() = 81001) then
    begin
      -- RFCRU-1970 ПДН: меры в целях понижения ПДН клиента до <=80%
      kc_anketa_income_calc = null;

      select kc_anketa_income_calc
      from knclient_pdn
      where kc_id = :kc_id
        and kcp_kind = 0
      into kc_anketa_income_calc;

      if (kc_anketa_income_calc is not null) then
        rep_basic_income = kc_anketa_income_calc - rep_add_income;
    end

    suspend;
  end
end
