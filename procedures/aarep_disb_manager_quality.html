CREATE PROCEDURE AAREP_DISB_MANAGER_QUALITY(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_MANAGER_C_ID DECIMAL(18,0),
    FOR_IS_BONUS SMALLINT = 0)
RETURNS (
    RES_DISB_PERIOD INTEGER,
    RES_DELINQ1 INTEGER,
    RES_DELINQ2 INTEGER,
    RES_DELINQ3 INTEGER,
    RES_PERIOD_DATE DATE,
    RES_QUAL_KOEF1 DECIMAL(18,7),
    RES_QUAL_KOEF2 DECIMAL(18,7),
    RES_QUAL_KOEF3 DECIMAL(18,7),
    RES_TOTAL1 INTEGER,
    RES_TOTAL2 INTEGER,
    RES_TOTAL3 INTEGER,
    C_NAME_FIO VARCHAR(100),
    RES_DELINQ1_STAGE_12 INTEGER,
    RES_DELINQ2_STAGE_12 INTEGER,
    RES_DELINQ3_STAGE_12 INTEGER,
    RES_QUAL_KOEF1_STAGE_12 DECIMAL(18,7),
    RES_QUAL_KOEF2_STAGE_12 DECIMAL(18,7),
    RES_QUAL_KOEF3_STAGE_12 DECIMAL(18,7),
    RES_TOTAL1_STAGE_12 INTEGER,
    RES_TOTAL2_STAGE_12 INTEGER,
    RES_TOTAL3_STAGE_12 INTEGER,
    RES_DELINQ1_STAGE_13 INTEGER,
    RES_DELINQ2_STAGE_13 INTEGER,
    RES_DELINQ3_STAGE_13 INTEGER,
    RES_TOTAL1_STAGE_13 INTEGER,
    RES_TOTAL2_STAGE_13 INTEGER,
    RES_TOTAL3_STAGE_13 INTEGER,
    REG_ID INTEGER,
    LML_ID D_INTEGER,
    JK_SYSTEM_CODE D_STR_100,
    MGR_C_ID ID_TYPE,
    RES_TOTAL1_NEW_CL D_INTEGER,
    RES_TOTAL2_NEW_CL D_INTEGER,
    RES_TOTAL3_NEW_CL D_INTEGER,
    RES_DELINQ1_NEW_CL D_INTEGER,
    RES_DELINQ2_NEW_CL D_INTEGER,
    RES_DELINQ3_NEW_CL D_INTEGER,
    RES_TOTAL1_OLD_CL D_INTEGER,
    RES_TOTAL2_OLD_CL D_INTEGER,
    RES_TOTAL3_OLD_CL D_INTEGER,
    RES_DELINQ1_OLD_CL D_INTEGER,
    RES_DELINQ2_OLD_CL D_INTEGER,
    RES_DELINQ3_OLD_CL D_INTEGER,
    JL_NAME STR_100,
    REG_NAME STR_100,
    LOAN_REG STR_100)
AS
declare variable c_id decimal(18,0);
declare variable res_delinq1_tmp integer;
declare variable res_delinq2_tmp integer;
declare variable res_delinq3_tmp integer;
declare variable res_total1_tmp integer;
declare variable res_total2_tmp integer;
declare variable res_total3_tmp integer;
declare variable res_qual_koef1_tmp perc_type;
declare variable res_qual_koef2_tmp perc_type;
declare variable res_qual_koef3_tmp perc_type;
declare variable res_delinq1_stage_12_tmp integer;
declare variable res_delinq2_stage_12_tmp integer;
declare variable res_delinq3_stage_12_tmp integer;
declare variable res_total1_stage_12_tmp integer;
declare variable res_total2_stage_12_tmp integer;
declare variable res_total3_stage_12_tmp integer;
declare variable res_qual_koef1_stage_12_tmp perc_type;
declare variable res_qual_koef2_stage_12_tmp perc_type;
declare variable res_qual_koef3_stage_12_tmp perc_type;
declare variable loan_reg_id d_integer;
declare variable db_id integer;
begin
  db_id = f_get_db_id();
  if (rep_region_id is null) then
    rep_region_id = 0;
  for select c_id, res_disb_period, iif(:rep_manager_c_id = 0 and :for_is_bonus = 0, iif(:db_id = 81001, kal_new_reg_id, :rep_region_id), :rep_region_id) loan_reg_id,
--           по всем СВЗ
             sum(res_delinq1), sum(res_delinq2), sum(res_delinq3), sum(res_total1), sum(res_total2), sum(res_total3),
--           со стажем менее года
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_delinq1, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_delinq2, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_delinq3, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_total1, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_total2, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_total3, 0)),
--           со стажем более года
             sum(iif(lo_experience >= 12, res_delinq1, 0)), sum(iif(lo_experience >= 12, res_delinq2, 0)),
             sum(iif(lo_experience >= 12, res_delinq3, 0)), sum(iif(lo_experience >= 12, res_total1, 0)),
             sum(iif(lo_experience >= 12, res_total2, 0)), sum(iif(lo_experience >= 12, res_total3, 0)),
--           разбиение по первичным и повторным
             sum(iif(is_new_cl = 0  and res_delinq1 = 1, 1, 0)),
             sum(iif(is_new_cl = 1 and res_delinq1 = 1, 1, 0)),
             sum(iif(is_new_cl = 0 and res_delinq2 = 1, 1, 0)),
             sum(iif(is_new_cl = 1 and res_delinq2 = 1, 1, 0)),
             sum(iif(is_new_cl = 0 and res_delinq3 = 1, 1, 0)),
             sum(iif(is_new_cl = 1 and res_delinq3 = 1, 1, 0)),
             sum(iif(is_new_cl = 0 and res_total1 = 1, 1, 0)),
             sum(iif(is_new_cl = 1 and res_total1 = 1, 1, 0)),
             sum(iif(is_new_cl = 0 and res_total2 = 1, 1, 0)),
             sum(iif(is_new_cl = 1 and res_total2 = 1, 1, 0)),
             sum(iif(is_new_cl = 0 and res_total3 = 1, 1, 0)),
             sum(iif(is_new_cl = 1 and res_total3 = 1, 1, 0))

      from aarep_disb_manager_quality_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, :rep_manager_c_id, :for_is_bonus)
      group by c_id, res_disb_period, loan_reg_id
      order by c_id, res_disb_period, loan_reg_id
      into c_id, res_disb_period, loan_reg_id, res_delinq1, res_delinq2, res_delinq3, res_total1, res_total2, res_total3,
           res_delinq1_stage_12, res_delinq2_stage_12, res_delinq3_stage_12, res_total1_stage_12, res_total2_stage_12,
           res_total3_stage_12, res_delinq1_stage_13, res_delinq2_stage_13, res_delinq3_stage_13, res_total1_stage_13,
           res_total2_stage_13, res_total3_stage_13,
           res_delinq1_old_cl, res_delinq1_new_cl, res_delinq2_old_cl, res_delinq2_new_cl, res_delinq3_old_cl, res_delinq3_new_cl,
           res_total1_old_cl, res_total1_new_cl, res_total2_old_cl, res_total2_new_cl, res_total3_old_cl, res_total3_new_cl

  do
  begin
    reg_id = null;
    select ps_from_reg_id, jk_system_code, jl_name
    from hr_pstatus, joblist, jobkind
    where ps_from_c_id = :c_id
    and :rep_date_end between ps_date_begin and ps_date_end
    and ps_status = 1
    and ps_from_jl_id = jl_id
    and jl_kind = jk_id
    into reg_id, jk_system_code, jl_name;

    select reg_name || '(' || reg_code || ')'
    from region
    where reg_unicode = :reg_id
    into reg_name;

    select reg_name || '(' || reg_code || ')'
    from region
    where reg_unicode = :loan_reg_id
    into loan_reg;

    res_period_date = cast('01.' || right(res_disb_period, 2) || '.' || left(res_disb_period, 4) as date);
    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    lml_id = null;
    mgr_c_id = null;
    select lml_id, lml_from_c_id from loanmgrlist
    where lml_from_c_id = :c_id
    and :rep_date_end between lml_date_begin and lml_date_end
    into lml_id, mgr_c_id;

    if (lml_id is null) then
    begin
      lml_id = f_lo_mgr_id(null, :rep_date_end, :c_id, :reg_id);
      mgr_c_id = f_lo_mgr_c_id(:lml_id);
    end

    res_qual_koef1 = 100.0;
    res_qual_koef2 = 100.0;
    res_qual_koef3 = 100.0;

    res_qual_koef1_tmp = 100.0;
    res_qual_koef2_tmp = 100.0;
    res_qual_koef3_tmp = 100.0;

    res_qual_koef1_stage_12_tmp = 100.0;
    res_qual_koef2_stage_12_tmp = 100.0;
    res_qual_koef3_stage_12_tmp = 100.0;

    if (for_is_bonus = 1 and
        (res_total1 = 0 or res_total2 = 0 or res_total3 = 0 or res_total1_stage_12 = 0 or res_total2_stage_12 = 0 or res_total3_stage_12 = 0)) then
    begin
--  П.2.3.2.3. Если у МНМ/Зам.МНМ нет показателей  КПР по какому-либо из первых трех погашений, в расчет принимается средний показатель офиса по данному погашению
--  средний считается по всем погашениям офиса, уточнял по телефону с Иродой
      res_delinq1_tmp = 0;
      res_delinq2_tmp = 0;
      res_delinq3_tmp = 0;
      res_total1_tmp = 0;
      res_total2_tmp = 0;
      res_total3_tmp = 0;

      res_delinq1_stage_12_tmp = 0;
      res_delinq2_stage_12_tmp = 0;
      res_delinq3_stage_12_tmp = 0;
      res_total1_stage_12_tmp = 0;
      res_total2_stage_12_tmp = 0;
      res_total3_stage_12_tmp = 0;

      select sum(res_delinq1), sum(res_delinq2), sum(res_delinq3), sum(res_total1), sum(res_total2), sum(res_total3),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_delinq1, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_delinq2, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_delinq3, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_total1, 0)),
             sum(iif(lo_experience >= 0 and lo_experience < 12, res_total2, 0)), sum(iif(lo_experience >= 0 and lo_experience < 12, res_total3, 0))

      from aarep_disb_manager_quality_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, 0, :for_is_bonus)
      where res_disb_period = :res_disb_period
      into res_delinq1_tmp, res_delinq2_tmp, res_delinq3_tmp, res_total1_tmp, res_total2_tmp, res_total3_tmp,
           res_delinq1_stage_12_tmp, res_delinq2_stage_12_tmp, res_delinq3_stage_12_tmp, res_total1_stage_12_tmp,
           res_total2_stage_12_tmp, res_total3_stage_12_tmp;
      res_delinq1_tmp = coalesce(res_delinq1_tmp, 0);
      res_delinq2_tmp = coalesce(res_delinq2_tmp, 0);
      res_delinq3_tmp = coalesce(res_delinq3_tmp, 0);
      res_total1_tmp = coalesce(res_total1_tmp, 0);
      res_total2_tmp = coalesce(res_total2_tmp, 0);
      res_total3_tmp = coalesce(res_total3_tmp, 0);

      res_delinq1_stage_12_tmp = coalesce(res_delinq1_stage_12_tmp, 0);
      res_delinq2_stage_12_tmp = coalesce(res_delinq2_stage_12_tmp, 0);
      res_delinq3_stage_12_tmp = coalesce(res_delinq3_stage_12_tmp, 0);
      res_total1_stage_12_tmp = coalesce(res_total1_stage_12_tmp, 0);
      res_total2_stage_12_tmp = coalesce(res_total2_stage_12_tmp, 0);
      res_total3_stage_12_tmp = coalesce(res_total3_stage_12_tmp, 0);

      if (res_total1_tmp <> 0) then
        res_qual_koef1_tmp = (1.0 - cast(res_delinq1_tmp as double precision) / cast(res_total1_tmp as double precision)) * 100.0;
      if (res_total2_tmp <> 0) then
        res_qual_koef2_tmp = (1.0 - cast(res_delinq2_tmp as double precision) / cast(res_total2_tmp as double precision)) * 100.0;
      if (res_total3_tmp <> 0) then
        res_qual_koef3_tmp = (1.0 - cast(res_delinq3_tmp as double precision) / cast(res_total3_tmp as double precision)) * 100.0;

      if (res_total1_stage_12_tmp <> 0) then
        res_qual_koef1_stage_12_tmp = (1.0 - cast(res_delinq1_stage_12_tmp as double precision) / cast(res_total1_stage_12_tmp as double precision)) * 100.0;
      if (res_total2_stage_12_tmp <> 0) then
        res_qual_koef2_stage_12_tmp = (1.0 - cast(res_delinq2_stage_12_tmp as double precision) / cast(res_total2_stage_12_tmp as double precision)) * 100.0;
      if (res_total3_stage_12_tmp <> 0) then
        res_qual_koef3_stage_12_tmp = (1.0 - cast(res_delinq3_stage_12_tmp as double precision) / cast(res_total3_stage_12_tmp as double precision)) * 100.0;

    end

    if (res_total1 <> 0) then
      res_qual_koef1 = (1.0 - cast(res_delinq1 as double precision) / cast(res_total1 as double precision)) * 100.0;
    else
      res_qual_koef1 = res_qual_koef1_tmp;

    if (res_total2 <> 0) then
      res_qual_koef2 = (1.0 - cast(res_delinq2 as double precision) / cast(res_total2 as double precision)) * 100.0;
    else
      res_qual_koef2 = res_qual_koef2_tmp;

    if (res_total3 <> 0) then
      res_qual_koef3 = (1.0 - cast(res_delinq3 as double precision) / cast(res_total3 as double precision)) * 100.0;
    else
      res_qual_koef3 = res_qual_koef3_tmp;

--    res_qual_koef1_stage_12=100.0;
--    res_qual_koef2_stage_12=100.0;
--    res_qual_koef3_stage_12=100.0;

    res_qual_koef1_stage_12 = 0.0;
    res_qual_koef2_stage_12 = 0.0;
    res_qual_koef3_stage_12 = 0.0;

--    П.2.3.1.1. Если нет показателя по какому-либо из первых трех погашений ВИПС со стажем работы менее 12 месяцев на момент выдачи займа,
--    в расчет принимается средний показатель по данному погашению по всем ВИПС офиса (данное условие применяется ко всему пункту 2.3 настоящей Бонусной системы)
    if (res_total1_stage_12 <> 0) then
      res_qual_koef1_stage_12 = (1.0 - cast(res_delinq1_stage_12 as double precision) / cast(res_total1_stage_12 as double precision)) * 100.0;
    else
    if (res_total1_stage_12_tmp <> 0) then
      res_qual_koef1_stage_12 = res_qual_koef1_stage_12_tmp;
    else
      res_qual_koef1_stage_12 = res_qual_koef1;

    if (res_total2_stage_12 <> 0) then
      res_qual_koef2_stage_12 = (1.0 - cast(res_delinq2_stage_12 as double precision) / cast(res_total2_stage_12 as double precision)) * 100.0;
    else
    if (res_total2_stage_12_tmp <> 0) then
      res_qual_koef2_stage_12 = res_qual_koef2_stage_12_tmp;
    else
      res_qual_koef2_stage_12 = res_qual_koef2;

    if (res_total3_stage_12 <> 0) then
      res_qual_koef3_stage_12 = (1.0 - cast(res_delinq3_stage_12 as double precision) / cast(res_total3_stage_12 as double precision)) * 100.0;
    else
    if (res_total3_stage_12_tmp <> 0) then
      res_qual_koef3_stage_12 = res_qual_koef3_stage_12_tmp;
    else
      res_qual_koef3_stage_12 = res_qual_koef3;

    suspend;
  end
end
