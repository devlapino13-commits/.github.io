CREATE PROCEDURE HP_ACTIONS_PRINT_WORK_DETAIL(
    C_ID DECIMAL(18,0))
RETURNS (
    RES_ACL_ID DECIMAL(18,0),
    RES_QUESTION VARCHAR(255),
    RES_ANSWER VARCHAR(1024))
AS
declare variable anketa_name varchar(255);
declare variable anketa_id integer;
begin
  res_answer = '';

  anketa_name = 'Anketa.Main';
  select rd_id
  from refdata
  where rd_code = :anketa_name
  into :anketa_id;
  if (anketa_id is null) then
    exit;

  select acl_id
  from actionlist
  where acl_object_kind = 0 and
        acl_object_id = :c_id and
        acl_from_rd_id = :anketa_id
  order by acl_id desc
  rows 1
  into :res_acl_id;
  if (res_acl_id is null) then
    res_acl_id = 0;

  for select ques_text, res_text
      from hp_anketa_print(0, :c_id, :anketa_name, null)
      where ques_code starting 'work_'
      into :res_question, res_answer
  do
  begin
    suspend;
  end

  res_question = 'Примечание';
  res_answer = '';
  select c_memo
  from clients
  where c_id = :c_id
  into :res_answer;
  if (res_answer is null) then
    res_answer = '';
  if (res_answer <> '') then
    suspend;
end
