CREATE PROCEDURE HP_UPLOAD_CLIENT_PHOTO_CHECK(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable GRS_ACTIVE D_SMALLINT;
declare variable C_CITIZEN D_INTEGER;
declare variable BLOB_1 D_INTEGER;
declare variable BLOB_2 D_INTEGER;
begin
  res_text = '';

  select cv.cv_value
  from const_var cv
  where cv.cv_code = 'grs_is_active'
  into :grs_active;

  select c.c_citizen
  from clients c
  where c.c_id = :c_id
  into :c_citizen;

  if (:c_citizen = f_get_rd_id('Citizen.KG') and
      :grs_active = 1 and
      not exists(select *
                 from grs_clients gc
                 where gc.gc_from_c_id = :c_id and
                       gc.gc_is_last = 1 and
                       gc.gc_status in (2, 3, -3))) then
  begin
    res_text = 'Сначала надо отправить запрос в ГРС, а после сфотографировать клиента';
    suspend;
    exit;
  end

  select first 1 cds.cds_doc_blob_1, cds.cds_doc_blob_2
  from clientdocs cds
  where cds.cds_from_c_id = :c_id and
        cds.cds_is_delete = 0 and
        cds.cds_is_main_doc = 1
  order by cds.cds_create_datetime asc
  into blob_1, blob_2;

  if ((coalesce(:blob_1, 0) = 0) or (coalesce(:blob_2, 0) = 0)) then
  begin
    res_text = 'Скан (Фото) паспорта не прикреплен, прошу проверить';
    suspend;
    exit;
  end
end
