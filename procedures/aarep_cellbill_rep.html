CREATE PROCEDURE AAREP_CELLBILL_REP(
    REP_DATE_BEGIN DATE)
RETURNS (
    OFFICE VARCHAR(50),
    CAL_NUMBER STR_100,
    CBL_CELL_NO STR_20,
    JOB_NAME STR_100,
    FIO STR_100,
    AMOUNT CURR_TYPE,
    NDS CURR_TYPE,
    SUM_BILL CURR_TYPE,
    LIMIT_AMOUNT CURR_TYPE,
    COMPANY_PAY CURR_TYPE,
    PAY_BY_DISMISSED CURR_TYPE,
    FIN_EMP_PAY CURR_TYPE)
AS
declare variable date_year integer;
declare variable date_month integer;
declare variable c_id id_type;
declare variable cal_date_begin date;
declare variable ps_date_begin date;
declare variable cal_has_limit integer;
begin

/*  поля на выход
  - офис
  - номер дог/лиц счета
  - номер телефона
  - должность
  - ФИО сотрудника
  - расход по связи
  - НДС (?)
  - Итог по счету (расход по связи + ндс (?))
  - утвержденный лимит
  - расход компании
  - расход в лк для уволнных  заполняется руками в выгруженном xlsx файле по факту
  - всего к удержанию
*/
    date_month = extract(month from rep_date_begin);
    date_year = extract(year from rep_date_begin);

    for select cbl_cell_no,
               cbl_amount
    from cellbilloper
    inner join cellbill on cb_bill_no = cbl_bill_no
    where cb_year = :date_year
    and cb_month = :date_month
    into cbl_cell_no, amount
    do
    begin
        select cal_from_c_id,
               iif(clk_limit_amount = 1000000.0000, 0.0000, clk_limit_amount),
               cal_number_with_operator,
               cal_date_begin,
               cal_has_limit
        from cellaktlist
        inner join cellimitkind on clk_id = cal_has_limit
        where cal_cell_no = :cbl_cell_no
        order by 4 desc
        rows 1
        into c_id, limit_amount, cal_number, cal_date_begin, cal_has_limit;

        nds = amount * 0.2;
        sum_bill = amount + nds;

        select reg_name,
               c_name_fio,
               jl_name,
               ps_date_begin
        from hr_pstatus
        inner join joblist on ps_from_jl_id = jl_id
        inner join jobkind on jl_kind = jk_id
        inner join clients on c_id = ps_from_c_id
        inner join region on reg_unicode = ps_from_reg_id
        where c_id = :c_id
        order by 4 desc
        rows 1
        into office, fio, job_name, ps_date_begin;

        if (cal_has_limit in (4,6,8,10,13,14)) then
            company_pay = sum_bill;
        else company_pay = limit_amount;

        fin_emp_pay = sum_bill - company_pay;

        if (fin_emp_pay <= 0.0000) then
            fin_emp_pay = 0.0000 ;

        suspend;
    end

end
