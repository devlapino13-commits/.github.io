CREATE PROCEDURE SVC_JIRA_CHECK_PASSWORD(
    LOGIN D_STR_CODE,
    PASS D_STR_CODE)
RETURNS (
    DEV_ID D_ID,
    ERROR_TEXT D_STR_255,
    SVC_PERMISSIONS D_STR_8100)
AS
declare variable PASS_HASH D_STR_CODE;
begin
  svc_permissions = '{}';
  error_text = '';
  login = coalesce(login, '');
  pass = coalesce(pass, '');

  if (login || pass = '') then
  begin
    error_text = 'Не указан логин или пароль';
    suspend;
    exit;
  end

  pass_hash = hex_encode(crypt_hash(:pass using md5));

  select jd.jrd_id
  from jira_dev jd
  where jd.jrd_db_login = :login
  and jd.jrd_password = :pass_hash
  and jd.jrd_password is not null
  into :dev_id;

  if (dev_id is null) then
  begin
    error_text = 'Некорректный логин или пароль';
    suspend;
    exit;
  end

  svc_permissions = (select result from svc_jira_permission(:login));

  suspend;
end
