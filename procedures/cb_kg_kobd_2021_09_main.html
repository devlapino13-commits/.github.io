CREATE PROCEDURE CB_KG_KOBD_2021_09_MAIN(
    REP_REGION_ID D_INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    REPAY_SUM_NOT_OS CURR_TYPE,
    REPAY_KOMMIS_IPF CURR_TYPE,
    DELINQ_OS_PERIOD CURR_TYPE,
    AAREP_CONSULT_SUMMA CURR_TYPE,
    SUM_VOZNOG_OFFICE CURR_TYPE,
    KOL_KOBD D_SMALLINT,
    FIO_KOBD D_STR_100,
    BONUS_KOBD CURR_TYPE,
    MEMO D_STR_2048)
AS
declare variable fio d_str_2048;
begin
  for select reg_name,
             repay_sum_not_os,
             repay_kommis_ipf,
             delinq_os_period,
             aarep_consult_summa,
             sum_voznog_office,
             kol_kobd,
             fio_kobd,
             bonus_kobd,
             memo
  from cb_kg_kobd_2021_09(:rep_region_id, :rep_date_end)
  into :reg_name,
       :repay_sum_not_os,
       :repay_kommis_ipf,
       :delinq_os_period,
       :aarep_consult_summa,
       :sum_voznog_office,
       :kol_kobd,
       :fio,
       :bonus_kobd,
       :memo
  do
  begin
    if (kol_kobd > 1) then
    begin
      for select trim(value_out)
          from str_with_separator_parser(:fio, ';')
          into :fio_kobd
      do
      begin
        suspend;
      end
    end
    if (kol_kobd <= 1) then
    begin
      fio_kobd = fio;
      suspend;
    end
  end
end
