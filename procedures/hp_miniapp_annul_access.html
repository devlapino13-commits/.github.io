CREATE PROCEDURE HP_MINIAPP_ANNUL_ACCESS(
    MA_ID D_ID,
    MA_DATE_OPER D_DATE,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_100)
AS
declare variable k_id d_id;
declare variable ks_date_begin d_date;
begin
  --RFCKG 2286

  if (f_get_db_id() not in (39801, 41701)) then
    exit;

  if (current_user = 'dkydyraliev') then
    exit;

  select kt.k_id
  from kontract kt
  where kt.k_from_ma_app_id = :ma_id
  into k_id;

  if (k_id is not null) then
  begin
    select ks.ks_date_begin
    from kontstate ks
    where ks.ks_from_k_id = :k_id and
          ks.ks_ol_is_delete = 1 and
          ks.ks_state = 1
    order by ks.ks_date_begin desc
    rows 1
    into ks_date_begin;

    ks_date_begin = coalesce(ks_date_begin, current_date);

    if (ks_date_begin < current_date and
        ma_date_oper < current_date) then
    begin
      if (not exists(select *
                     from hp_permission_access_check(:un_id, 'permactions', 0, 'FinStart.Miniapp_annul_after_disb')
                     where has_access = 1)) then
        res_text = 'Ошибка при аннулировании. Обратитесь к тех. поддержке.';
    end 
  end 
  suspend;
end
