CREATE PROCEDURE HP_EXPRESSAPP_BY_EXTMINIAPP(
    EMA_ID D_ID)
AS
declare variable c_id d_id;
declare variable amount d_currency;
declare variable ema_date d_date;
declare variable period d_integer;
declare variable reg_id d_id;
declare variable tkt_id d_id;
declare variable esw_id d_id;
declare variable pr_id d_integer;
declare variable exc_id_begin d_integer;
declare variable percent d_percent;
declare variable repay_day d_integer;
declare variable exa_id d_id;
begin
  select ema.ema_from_c_id, ema.ema_amount, ema.ema_date, ema.ema_period, ema.ema_office_id, ema.ema_from_tkt_id,
         ema.ema_from_esw_id, ema.ema_from_pr_id
  from extminiapp ema
  where ema.ema_id = :ema_id
  into c_id, amount, ema_date, period, reg_id, tkt_id, esw_id, pr_id;

  select exc.exc_id
  from expresscfg exc
  where exc.exc_action_code = 'miniapp.create'
  into exc_id_begin; --Создание кредитной заявки

  percent = 0;
  repay_day = 0;

  insert into expressapp (exa_from_exp_id, exa_date, exa_group_id, exa_done, exa_from_c_id, exa_base_k_id,
                          exa_from_exc_id, exa_final_status, exa_wait_status, exa_from_ma_id, exa_from_k_id,
                          exa_from_edl_id, exa_disb_date, exa_from_pr_id, exa_amount, exa_period, exa_percent,
                          exa_repay_day, exa_lo_id, exa_from_reg_id, exa_try_count, exa_from_un_id, exa_datetime,
                          exa_reshipping_count)
  values (-2, current_date, 0, 0, :c_id, null, :exc_id_begin, 0, 0, null, null, null, current_date, :pr_id, :amount,
          :period, :percent, :repay_day, 1000000000101, :reg_id, 0, 1000000000000, current_timestamp, 0)
  returning exa_id
  into :exa_id;

  execute procedure hp_expressapp_disb_loan(:exa_id, :tkt_id);

end
