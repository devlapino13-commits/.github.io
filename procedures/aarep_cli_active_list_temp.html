CREATE PROCEDURE AAREP_CLI_ACTIVE_LIST_TEMP(
    REP_DATE_END DATE)
RETURNS (
    AC_ID ID_TYPE,
    AC_DISB_DATE DATE,
    AC_KONTRACT_ID ID_TYPE,
    AC_IS_CLIENT SMALLINT,
    AC_REGION_ID INTEGER,
    AC_SHARE_CLIENT PERC_TYPE,
    AC_CLI_BALANCE CURR_TYPE,
    AC_CURR_CODE VARCHAR(3),
    AC_CURR_RATE CURR_TYPE,
    AC_PRODUCT_ID INTEGER,
    AC_KC_ID ID_TYPE,
    AC_AMOUNT_APPROVED CURR_TYPE,
    AC_PERIOD DECIMAL(15,2),
    AC_KONTRACT_NO VARCHAR(30),
    AC_KF_ID ID_TYPE,
    AC_DELINQ_DAY INTEGER)
AS
declare variable c_id_prior id_type;
declare variable cr_rate curr_type;
declare variable oks_delinq_date date;
begin
  select cr_rate
  from opercurr
  where cr_from_cu_id = '840' and
        :rep_date_end between cr_date and cr_date_end
  into cr_rate;
  if (cr_rate is null) then
    cr_rate = 0;
  c_id_prior = 0;
  for select kc_from_c_id, k_disb_date, k_id, oks_region_id,
             cast(kc_amount_approved as double precision) / k_amount_approved,
             cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance,
             k_index_curr_code, k_index_curr_rate, k_from_pr_id, oks_kc_id, kc_amount_approved, k_period,
             k_kontract_no, k_from_kf_id, oks_delinq_date
      from kontract, knclient, oksummary
      where k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            k_id = oks_from_k_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0
      order by kc_from_c_id, k_disb_date, k_id
      into ac_id, ac_disb_date, ac_kontract_id, ac_region_id, ac_share_client, ac_cli_balance, ac_curr_code,
           ac_curr_rate, ac_product_id, ac_kc_id, ac_amount_approved, ac_period, ac_kontract_no, ac_kf_id,
           oks_delinq_date
  do
  begin
    if (ac_curr_code = '840') then
    begin
      if (cr_rate > ac_curr_rate) then
        ac_cli_balance = ac_cli_balance * cr_rate;
      else
        ac_cli_balance = ac_cli_balance * ac_curr_rate;
    end
    if (c_id_prior <> ac_id) then
    begin
      c_id_prior = ac_id;
      ac_is_client = 1;
    end
    else
      ac_is_client = 0;
    if (:rep_date_end > :oks_delinq_date) then
      ac_delinq_day = :rep_date_end - :oks_delinq_date;
    else
      ac_delinq_day = 0;
    suspend;
  end
end
