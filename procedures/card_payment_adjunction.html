CREATE PROCEDURE CARD_PAYMENT_ADJUNCTION(
    OBJECT_KIND D_STR_100 NOT NULL,
    OBJECT_ID D_ID NOT NULL,
    PARENT_CS_ID D_INTEGER = null,
    OL_ID D_ID = null)
AS
declare variable ok_overpayment d_currency;
declare variable ok_perc d_currency;
declare variable ok_fine d_currency;
declare variable ok_tax d_currency;
declare variable c_id d_id;
declare variable card_id d_integer;
declare variable cca_id d_integer;
declare variable cca_code d_str_code;
declare variable cs_request_json d_str_8100;
declare variable comment_txt varchar(140);
declare variable cs_id d_integer;
declare variable cst_packet_no varchar(8);
declare variable cst_ticket_no varchar(7);
declare variable trans_datetime d_datetime;
declare variable object_kind_str d_str_100;
declare variable k_id d_id;
declare variable ok_os d_currency;
declare variable full_sum d_currency;
declare variable ol_parent_id d_id;
declare variable ol_from_reg_id d_integer;
declare variable un_id d_id;
declare variable is_writeoff d_bool;
begin
    /*
         Процедура отвечает за формирование операции для МПЦ
         "списание с карт счета на сумму = сумма платежа минус сумма платежа по ОС",
         по факту погашения займа
    */
    if (f_get_const('cardprjclose') = 1) then
      exit;
    object_kind_str = f_objkind(object_kind, 'name');

    if (object_kind_str = 'kontract') then
        k_id = object_id;
    else
        exception e_message 'object_kind = '||object_kind_str||' не реализован';

    if ((ol_id is null) and
        (parent_cs_id is not null)) then
    begin
        select cso.cso_from_ol_id
        from cardsyncopers cso
        where cso.cso_from_cs_id = :parent_cs_id
        into :ol_id;
    end

    if (ol_id is null) then exception e_message 'ol_id не определен';

    select ol.ol_parent_id,
           ol.ol_from_reg_id,
           ol.ol_user_id
    from operlist ol
    where ol.ol_id = :ol_id
    into :ol_parent_id,
         :ol_from_reg_id,
         :un_id;

    if (ol_parent_id is not null) then
    begin
      select ol.ol_from_reg_id
      from operlist ol
      where ol.ol_id = :ol_parent_id
      into :ol_from_reg_id;
    end

    -- списание(1) или погашение(0)
    is_writeoff = iif(exists(
      select * from operlist
      where ol_id = :ol_id
        and ol_oper_type = 1
        and ol_oper_subtype = 3), 1, 0);

    ok_os = 0;
    ok_overpayment = 0;
    ok_perc = 0;
    ok_fine = 0;
    ok_tax = 0;
    select -sum(iif(ok_type = iif(:is_writeoff = 1, 5, 4), ok.ok_os, 0)),
           -sum(iif(ok_type = 9, ok.ok_os, 0)),
           -sum(ok.ok_perc),
           -sum(ok.ok_fine),
           -sum(ok.ok_tax)
    from operkred ok
    where iif(:is_writeoff = 1, ok.ok_type = 5, ok.ok_type in (4, 9))
    and ok.ok_from_ol_id = :ol_id
    and ok.ok_ol_is_active = 1
    and ok.ok_from_k_id = :k_id
    into
       :ok_os,
       :ok_overpayment,
       :ok_perc,
       :ok_fine,
       :ok_tax;
    
    full_sum = ok_os + ok_overpayment + ok_perc + ok_fine + ok_tax;
    
    if (full_sum < 0) then
        exception e_message 'Погашение увелиличило долг клиента';
    
    if (full_sum = 0) then -- если это нулевой платеж
        exit;
    
    comment_txt = '';

    if (ok_os <> 0) then
        comment_txt = comment_txt || ' основная сумма: ' || round(ok_os, 2)|| '; ';

    if (ok_overpayment <> 0) then
        comment_txt = comment_txt || ' переплата: ' || round(ok_overpayment, 2)|| '; ';

    if (ok_perc  <> 0) then
        comment_txt = comment_txt || ' %: ' || round(ok_perc, 2)|| '; ';

    if (ok_fine  <> 0) then
        comment_txt = comment_txt || ' пени: ' || round(ok_fine, 2)|| '; ';

    if (ok_tax  <> 0) then
        comment_txt = comment_txt || ' НСП: ' || round(ok_tax, 2)|| '; ';

    select kncl.kc_from_c_id
    from knclient kncl
    where kncl.kc_from_k_id = :k_id
      and kncl.kc_amount_approved > 0
    into :c_id;

    select packet_no,
           ticket_no
    from cardsynctrn_create
    into :cst_packet_no,
         :cst_ticket_no;

    if (cst_packet_no is null) then
        exception e_message '№ пакета не опредлен';
    
    if (cst_ticket_no is null) then
        exception e_message '№ тикета не опредлен';

    select ol.ol_datetime
    from operlist ol
    where ol.ol_id = :ol_id
    into :trans_datetime;
    
    select
        card_id,
        cca_id,
        cca_code
    from hp_client_bank_card_mb(:c_id, current_date, 0)
    into
        :card_id,
        :cca_id,
        :cca_code;

    if (full_sum is null) then
      exit;

    cs_request_json = f_format('{"card_acct": "%1","card_acct_ccy": "KGS","tran_type": "11%2",', cca_code, iif(is_writeoff = 1, 'L', 'C'));
    cs_request_json = cs_request_json ||
      f_format('"tran_ccy": "KGS","tran_amount": %1,"tran_pocket": "%2","tran_ticket": "%3","tran_comment": "%4","bank_c": "21","groupc": "01","tran_date_time": "%5"}',
        round(:full_sum, 2), cst_packet_no, cst_ticket_no, comment_txt, f_datetimetostr(trans_datetime));

    insert into cardsync(
      cs_parent_id,
      cs_from_csc_id,
      cs_card_id,
      cs_from_cca_id,
      cs_object_kind,
      cs_object_id,
      cs_amount,
      cs_request_json,
      cs_office_id,
      cs_from_un_id,
      cs_from_k_id,
      cs_from_c_id
    )
    values(
      :parent_cs_id,
      f_get_cardsynccfg('payment_adjunction', 'id'),
      :card_id,
      :cca_id,
      f_objkind('kontract', 'id'),
      :k_id,
      :full_sum,
      :cs_request_json,
      :ol_from_reg_id,
      :un_id,
      :k_id, 
      :c_id
    )
    returning cs_id
    into :cs_id;

    update cardsynctrn cst
    set cst.cst_from_cs_id = :cs_id
    where cst.cst_packet_no = :cst_packet_no and
          cst.cst_ticket_no = :cst_ticket_no;

    insert into cardsyncopers (cso_from_cs_id, cso_from_ol_id)
    values (:cs_id, :ol_id);
end
