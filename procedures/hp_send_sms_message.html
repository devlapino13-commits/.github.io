CREATE PROCEDURE HP_SEND_SMS_MESSAGE(
    C_ID D_ID,
    SMS_TEXT_STR D_STR_LONG,
    UN_ID D_ID)
RETURNS (
    RES_ERROR_TEXT D_STR_255,
    RES_INFO_TEXT D_STR_255)
AS
declare variable OK_CELL_MAIN D_STR_100;
declare variable PHONE_ID D_INTEGER;
begin
    -- Процедура выполняет отправку СМС для модуля по отправке СМС.
    --  sms_text_str текст для смс сообщения приходит из модуля см. hp_generate_sms_message
    -- RFCRU-2761

    res_error_text = '';
    res_info_text = '';

    if (coalesce(c_id, 0) = 0) then
        res_error_text = 'Клиент не выбран!';

    if ((res_error_text = '') and (not exists(select * from clients c where c.c_id = :c_id))) then
        res_error_text = 'Клиент не найден.';

    if (res_error_text = '') then
    begin
        ok_cell_main = f_cont_info('clients', :c_id, 'cell_main', 'ok');
        if (coalesce(ok_cell_main, '') = '') then
        begin
            res_error_text = 'Ошибка: Номер телефона не указан в базе.';
            phone_id = f_cont_id(:ok_cell_main, 'add');
        end
    end

    if (res_error_text <> '') then
    begin
        suspend;
        exit;
    end

    phone_id = f_cont_id(:ok_cell_main, null);

    --отправляем СМС
    execute procedure hp_send_quick_message('sms', :ok_cell_main, '', :sms_text_str, '', -1, :un_id);
    res_info_text = 'Сообщение отправлено.';

    insert into user_sms_send_log (ussl_from_c_id, ussl_un_id, ussl_msg_txt, ussl_phone_id)
    values (:c_id, :un_id, :sms_text_str, :phone_id);

    suspend;
end
