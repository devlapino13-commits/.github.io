CREATE PROCEDURE HP_EXTINFOLIST_PDN_GET_EIL(
    KC_ID D_ID,
    CALC_DATE D_DATE,
    USE_EXIST_EIL_ID INTEGER,
    EIC_CODE D_STR_CODE NOT NULL = 'equifax',
    KCP_KIND D_SMALLINT NOT NULL = 0)
RETURNS (
    EIL_ID D_ID,
    DATE_TILL_DISB D_DATE,
    DATE_SUCCESS_TILL_DISB D_DATE,
    CHECK_COUNT D_INTEGER,
    CALC_STATUS D_STR_100,
    CALC_EIL_DATE D_DATE,
    BLS_ID D_ID,
    HAS_BLOB D_SMALLINT,
    K_DISB_DATE D_DATE,
    C_ID D_ID,
    HISTORY_NOT_FOUND D_SMALLINT)
AS
declare variable eil_status d_smallint;
declare variable eil_is_json d_smallint;
declare variable eil_id_tmp d_id;
declare variable eil_id_till_disb d_id;
declare variable dd_tmp d_date;
declare variable ma_id d_integer;
declare variable k_id d_id;
declare variable history_search_date d_date;
begin
  if (not exists(select eic_id from extinfocfg where eic_code = :eic_code)) then
    exception e_default 'Не найден проект: ' || eic_code;

  history_not_found = 1;
  if (calc_date is null) then calc_date = current_date;

  select kc_from_k_id, kc_from_c_id, k_from_ma_app_id, k_disb_date
  from knclient, kontract
  where kc_id = :kc_id and
        kc_from_k_id = k_id
  into :k_id, :c_id, ma_id, k_disb_date;

  if (k_id is null or c_id is null) then
    exception e_default 'Нет займа KC_ID: ' || coalesce(kc_id, '<NULL>');

  date_till_disb = null; -- Дата последнего запроса в БКИ до выдачи займа включительно
  date_success_till_disb = null; -- Дата последнего УСПЕШНОГО запроса в БКИ до выдачи
  check_count = 0; -- Кол-во запросов в БКИ по займу

  eil_id = null;

  -- если передан use_exist_eil_id = 1, то не будем искать новую истори БКИ,
  -- берем ранее использованную из knclient_pdn.kc_from_eil_id

  if (use_exist_eil_id is null) then
    use_exist_eil_id = 0;

  if (use_exist_eil_id = 1) then
  begin
    select kc_from_eil_id
    from knclient_pdn
    where kc_id = :kc_id
      and kcp_kind = 0
    into eil_id;

    if (eil_id is not null) then
    begin
      select eil_date
      from extinfolist
      where eil_id = :eil_id
      into calc_eil_date;
    end else
      use_exist_eil_id = 0;
  end

  history_search_date = iif(kcp_kind = 0, k_disb_date, calc_date);

  if (eil_id is null) then
  begin
    select eil_id, eil_date
    from extinfolist, extinfocfg
    where eil_from_c_id = :c_id and
          eil_from_eic_id = eic_id and
          eic_code = :eic_code and
          eil_is_json = 1 and
          eil_status = 2 and
          eil_date between (:history_search_date - 5) and :history_search_date
    order by eil_date desc, eil_id desc
    rows 1
    into eil_id, calc_eil_date;
  end

  if (eil_id is null) then
  begin
    select eil_id, eil_date
    from extinfolist, extinfocfg
    where eil_from_c_id = :c_id and
          eil_from_eic_id = eic_id and
          eic_code = :eic_code and
          eil_is_json = 1 and
          eil_status = 2 and
          ((eil_object_kind = f_objkind('kontract', 'id') and eil_object_id = :k_id) or
           (eil_object_kind = f_objkind('miniapp', 'id') and eil_object_id = :ma_id)) and
          eil_date >= (minvalue(:calc_date, :history_search_date) - 5)
    order by eil_date desc, eil_id desc
    rows 1
    into eil_id, calc_eil_date;
  end

  eil_id_till_disb = null;
  for select eil_id, eil_date, eil_is_json, eil_status
  from extinfolist, extinfocfg
  where eil_from_c_id = :c_id and
        eil_from_eic_id = eic_id and
        eic_code = :eic_code and
        ((eil_object_kind = f_objkind('kontract', 'id') and eil_object_id = :k_id) or
         (eil_object_kind = f_objkind('miniapp', 'id') and eil_object_id = :ma_id))
  order by eil_date
  into eil_id_tmp, dd_tmp, eil_is_json, eil_status
  do
  begin
    check_count = check_count + 1;
    if (eil_id is null and eil_is_json = 1 and eil_status = 2) then eil_id = eil_id_tmp;
    if (dd_tmp <= history_search_date) then date_till_disb = dd_tmp;
    if (eil_is_json = 1 and eil_status = 2 and dd_tmp <= history_search_date) then
    begin
      date_success_till_disb = dd_tmp;
      eil_id_till_disb = eil_id_tmp;
    end
  end
  if (use_exist_eil_id = 0 and eil_id_till_disb is not null and calc_eil_date is null) then
    eil_id = eil_id_till_disb;

  if (eil_id is null) then eil_id = eil_id_tmp;

  if (eil_id is not null) then
  begin
    select eil_id, eil_date, eil_from_bls_id, eil_status, eil_is_json
    from extinfolist
    where eil_id = :eil_id
    into eil_id, calc_eil_date, bls_id, eil_status, eil_is_json;
  end

  calc_status = 'Нет запроса в БКИ';

  has_blob = 0;

  if (calc_eil_date is not null) then
  begin
    history_not_found = 0;
    if (coalesce(bls_id, 0) <= 0) then
    begin
      if (eil_status = 2)
      then calc_status = 'Клиент не имеет кредитной истории';
      else
      begin
        calc_status = 'История не получена';
        history_not_found = 1;
      end
    end else
    begin
      if (eil_is_json = 1) then
      begin
        calc_status = 'История получена';
        if (datediff(day, calc_eil_date, calc_date) > 5) then
          calc_status = 'Полученная история БКИ устарела';

        has_blob = 1;
      end else
      begin
        calc_status = 'История в формате PDF';
        history_not_found = 1;
      end
    end
  end

  suspend;
end
