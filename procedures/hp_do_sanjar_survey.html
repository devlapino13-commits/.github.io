CREATE PROCEDURE HP_DO_SANJAR_SURVEY(
    CELL_FROM VARCHAR(20),
    SMS_TEXT VARCHAR(255))
AS
declare variable svq_id integer;
declare variable sva_id integer;
declare variable svq_time_start timestamp;
declare variable svq_time_finish timestamp;
declare variable sva_caption varchar(255);
declare variable svq_caption varchar(255);
declare variable svq_memo varchar(255);
begin
  if (not exists(select *
                 from survey_allowedcells
                 where sac_is_test >= 0 and
                       sac_cell_no = :cell_from)) then
  begin
    execute procedure hp_send_quick_message('sms', :cell_from, '',
        'Ваш номер не участвует в голосовании. Обратитесь к организаторам.',
        '', null, 1000000000001);
    exit;
  end
  sms_text = replace(sms_text, ' ', '');
  sms_text = left(sms_text, 20);

  select svq_id, sva_id, svq_time_start, svq_time_finish, sva_caption, svq_caption, svq_memo
  from survey_answer, survey_ques
  where svq_id = sva_from_svq_id and
        svq_id between 42 and 53 -- Тут прописать номера вопросов
        and
        sva_codes = '~' || :sms_text || '~' rows 1
  into svq_id, sva_id, svq_time_start, svq_time_finish, sva_caption, svq_caption, svq_memo;

  if (svq_id is null) then -- Не найден совпадающий ответ
  begin
    execute procedure hp_send_quick_message('sms', :cell_from, '',
        'SMS должен быть в формате <№ номинации>#<№ участника>. Например: 1#77',
        '', null, 1000000000001);
    exit;
  end

  if (svq_time_start is null or svq_time_start > current_timestamp) then
  begin
    execute procedure hp_send_quick_message('sms', :cell_from, '',
        'Голосование еще не началось.', '', null, 1000000000001);
    exit;
  end

  if (svq_time_finish < current_timestamp) then
  begin
    execute procedure hp_send_quick_message('sms', :cell_from, '', 'Голосование завершено.', '',
        null, 1000000000001);
    exit;
  end

-- Проверяет наличие дубликата
  if (exists(select *
             from survey_result
             where svr_source_kind = 1 -- СМС
                   and
                   svr_source_id = :cell_from and
                   svr_from_svq_id = :svq_id)) then
  begin
    execute procedure hp_send_quick_message('sms', :cell_from, '',
        'В каждой номинации засчитывается один голос за юношу и один за девушку',
        '', null, 1000000000001);
    exit;
  end

  insert into survey_result (svr_source_kind, svr_source_id, svr_text, svr_server_sim, svr_from_svq_id, svr_from_sva_id)
  values (1, :cell_from, :sms_text, '', :svq_id, :sva_id);

  execute procedure hp_send_quick_message('sms', :cell_from, '',
      'Ваш голос в номинации ' || :svq_caption || ' (' || lower(:svq_memo) || ') принят за участника ' || :sva_caption,
      '. Спасибо!', null, 1000000000001);
end
