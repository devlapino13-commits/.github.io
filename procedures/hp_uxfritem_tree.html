CREATE PROCEDURE HP_UXFRITEM_TREE(
    PARENT_CODE D_STR_100,
    REPLACE_READONLY D_BOOL)
RETURNS (
    UFI_CODE D_STR_100,
    UFI_PARENT_CODE D_STR_100,
    UFI_UFR_CODE D_STR_CODE,
    UFI_UX_CODE D_STR_CODE,
    UFI_UFP_CODE D_STR_100,
    UFI_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_CAPTION D_STR_CODE,
    UFI_CAPTION_ALIGN D_INTEGER,
    UFI_HINT D_STR_LONG,
    UFI_HIDDEN D_BOOL,
    UFI_READONLY D_BOOL,
    UFI_GROUP_DIRECTION D_INTEGER,
    UFI_H_ALIGN D_INTEGER,
    UFI_V_ALIGN D_INTEGER,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_ADD_PARAMS D_STR_CODE,
    ALLOW_CHILD D_BOOL,
    UFP_IS_PARAM D_SMALLINT,
    UFP_ORIGIN_NAME D_STR_CODE,
    UFD_SOURCE D_STR_CODE,
    UFP_REQUIRED D_BOOL,
    UFP_ACCESS_KIND D_SMALLINT)
AS
declare variable tmp_group d_str_code;
declare variable parent_is_readonly d_bool;
begin
  if (replace_readonly is null) then
    replace_readonly = 0;

  parent_is_readonly = 0;
  if (replace_readonly = 1) then
  begin
    select ufi_readonly
    from uxfritem
    where ufi_code = :parent_code
    into parent_is_readonly;
  end

  for select ufi_code, ufi_parent_code, ufi_ufr_code, ufi_ux_code, ufi_ufp_code, ufi_kind, ufi_control, ufi_position,
             ufi_caption, ufi_caption_align, ufi_hint, ufi_hidden, iif(:parent_is_readonly = 1, 1, ufi_readonly),
             ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
             ufi_add_params, ufp_origin_name, ufp_is_param, ufd_source, ufp_required, ufp_access_kind
  from uxfritem
    left join uxfrparam on (ufi_ufp_code = ufp_code)
    left join uxfrdata on (ufd_ufr_code = ufp_ufr_code and ufd_kind = 0)
  where ufi_parent_code = :parent_code
  order by ufi_position
  into :ufi_code, :ufi_parent_code, :ufi_ufr_code, :ufi_ux_code, :ufi_ufp_code, :ufi_kind, :ufi_control, :ufi_position,
       :ufi_caption, :ufi_caption_align, :ufi_hint, :ufi_hidden, :ufi_readonly,
       :ufi_group_direction, :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height,
       :ufi_add_params, :ufp_origin_name, :ufp_is_param, :ufd_source, ufp_required, ufp_access_kind
  do
  begin
    if (ufp_access_kind is null) then
      ufp_access_kind = 0;

    allow_child = iif(ufi_kind in (-1, -2, -3), 1, 0);
    suspend;

    if (ufi_kind in (-1, -3)) then
    begin
      tmp_group = ufi_code;

      for select ufi_code, ufi_parent_code, ufi_ufr_code, ufi_ux_code, ufi_ufp_code, ufi_kind, ufi_control,
                 ufi_position, ufi_caption, ufi_caption_align, ufi_hint, ufi_hidden,
                 iif(:parent_is_readonly = 1, 1, ufi_readonly),
                 ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
                 ufi_add_params, ufp_origin_name, ufp_is_param, ufd_source, allow_child, ufp_required, ufp_access_kind
          from hp_uxfritem_tree(:tmp_group, :replace_readonly)
          into :ufi_code, :ufi_parent_code, :ufi_ufr_code, :ufi_ux_code, :ufi_ufp_code, :ufi_kind, :ufi_control,
               :ufi_position, :ufi_caption, :ufi_caption_align, :ufi_hint, :ufi_hidden,
               :ufi_readonly,
               :ufi_group_direction, :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height,
               :ufi_add_params, :ufp_origin_name, :ufp_is_param, :ufd_source, :allow_child, :ufp_required, :ufp_access_kind
      do
        suspend;
    end
  end
end
