CREATE PROCEDURE HP_EDOC_WEB_BLANK_ONLINE(
    UUID D_STR_CODE,
    ID D_ID)
RETURNS (
    EDL_ID D_ID,
    EDB_ID D_ID,
    REG_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    UN_ID D_ID,
    BLN_ID D_INTEGER,
    BLB_BLOB D_BLOB_BINARY,
    MANAGER_KSR_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG,
    HASH_KIND D_INTEGER)
AS
declare variable manager_c_id d_id;
declare variable eds_id d_id;
declare variable eds_status d_bool;
declare variable edb_from_c_id d_id;
begin
  -- Используется для подготовки онлайн бланков
  -- Возвращает информацию, для подготовки бланка в FsService
  hash_kind = 0; -- Для онлайн бланков hash пока не требуется

  select obj_id
  from hp_uuid(:uuid)
  into eds_id;
  if (eds_id is null) then
    exit;

  select eds_from_edl_id, eds_status
  from edoc_sign
  where eds_id = :eds_id
  into edl_id, eds_status;
  if (edl_id is null) then
    exit;

  for select edb_id, edb_from_bln_id, edl_object_kind, edl_object_id,
    coalesce(edl_manager_c_id, edc_default_manager_id), edl_create_un_id,
    edl_blank_date, edb_from_c_id, edl_from_reg_id
  from edoc_blank, edoc_list, edoc_cfg
  where edb_from_edl_id = edl_id
    and edl_from_edc_id = edc_id
    and edb_id = :id
    and edb_from_edl_id = :edl_id
  into :edb_id, :bln_id, obj_kind, obj_id, manager_c_id, un_id, date_begin, edb_from_c_id, reg_id
  do
  begin
    add_params = '';

    -- Поля для вычисления
    if (edb_from_c_id is not null)
    then add_params = f_format('{"edl_id":%1,"c_id":%2}', edl_id, edb_from_c_id);
    else add_params = f_format('{"edl_id":%1}', edl_id);

    if (date_begin is null) then
      date_begin = current_date;
    date_end = date_begin;

    if (reg_id is null) then
    begin
      select reg_id
      from hp_edoc_object_info(:obj_kind, :obj_id, :un_id, :date_begin)
      into :reg_id;
    end

    -- Получаем подписанта, приоритет у активных, потом у последних добавленных
    -- Т.е. допускаем что подписант может стать неактивным, когда клиент подписал
    select ksr_id
    from kontsignmen, kontsignreg, region_list(:reg_id, 1, -1, 0)
    where ksm_id = ksr_from_ksm_id and
          ksm_from_c_id = :manager_c_id and
          ksr_from_reg_id = id_list and
          coalesce(ksr_date_begin, '01.01.1990') <= :date_begin and
          coalesce(ksr_date_end, '31.12.9999') >= :date_begin
    order by (ksr_is_active + ksm_is_active) desc, ksr_id desc
    rows 1
    into manager_ksr_id;

    select iif(blb_frx_blob is null, blb_blob, blb_frx_blob) as blb_blob
    from blankblob
    where blb_date_begin <= :date_begin
      and blb_date_end >= :date_begin
      and blb_from_bln_id = :bln_id
    into blb_blob;

    if (blb_blob is not null) then
      Suspend;

    if (eds_status = 0) then
    begin
      update or insert into edoc_open (edo_from_eds_id, edo_from_edb_id, edo_before_sign_datetime)
      values (:eds_id, :edb_id, current_timestamp)
      matching (edo_from_eds_id, edo_from_edb_id);
    end else
    begin
      update or insert into edoc_open (edo_from_eds_id, edo_from_edb_id, edo_after_sign_datetime)
      values (:eds_id, :edb_id, current_timestamp)
      matching (edo_from_eds_id, edo_from_edb_id);
    end
  end
end
