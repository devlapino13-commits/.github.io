Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_BUSSTANDARTLINK(
    BSM_ID D_INTEGER,
    SHOW_ZERO D_BOOL = 0)
RETURNS (
    BSLK_FROM_BS_ID D_INTEGER,
    BS_NAME D_STR_CODE,
    BS_IS_DELETE D_INTEGER,
    BSLK_FROM_BSL_ID D_INTEGER,
    BSLK_FROM_BSM_ID D_INTEGER)
AS
begin
  if (show_zero is null) then
    show_zero = 0;

  for select bslk.bslk_from_bs_id,
             bs.bs_name,
             bs.bs_is_delete,
             bslk.bslk_from_bsl_id,
             bslk.bslk_from_bsm_id
      from busstandartlink bslk
      inner join bussubtype bs on bs.bs_id = bslk.bslk_from_bs_id
      where bslk.bslk_from_bsm_id = :bsm_id and
            (:show_zero = 0 or coalesce(bslk_from_bsl_id, 0) = 0)
      order by bslk.bslk_from_bs_id
      into bslk_from_bs_id,
           bs_name,
           bs_is_delete,
           bslk_from_bsl_id,
           bslk_from_bsm_id
  do
  begin
    suspend;
  end
end
