CREATE PROCEDURE CM_AZ_0008_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable pr_product_kind integer;
declare variable k_kolich_repay integer;
declare variable k_kolich_client_approved integer;
declare variable k_amount_approved curr_type;
declare variable kf_id id_type;
declare variable pr_loan_type integer;
declare variable max_delinq_day integer;
declare variable last_k_id id_type;
declare variable delinq_day integer;
declare variable k_amount_last curr_type;
declare variable pr_product_kind_last integer;
declare variable k_id_last id_type;
declare variable k_disb_date_last date;
declare variable C_LAST_BONUS_STATUS integer;
declare variable k_last_balance curr_type;
declare variable max_balance curr_type;
declare variable kc_amount_last curr_type;
declare variable kc_last_balance curr_type;
declare variable kc_amount_approved curr_type;
declare variable kc_last_balance1 curr_type;
declare variable kc_amount_approved1 curr_type;
begin
  for
  select pr_product_kind,
         ma_kolich_repay,
         ma_kolich_client,
         ma_summa_minus,
         ma_from_kf_id,
         ma_date_oper,
         pr_loan_type
  from miniapp, product
  where ma_id=:ma_id
  and ma_from_pr_id = pr_id
  and ma_date_end='31.12.9999'
  into pr_product_kind,
       k_kolich_repay,
       k_kolich_client_approved,
       k_amount_approved,
       kf_id,
       kf_disb_date,
       pr_loan_type
  do
  begin
--     Проверка на параллельный, если pr_product_kind = 2 то параллельный кредит
    if (pr_product_kind = 2) then
    begin
--       Вытаскиает данные Основного контракта для дольнейшей проверки
      k_amount_last = null;
      pr_product_kind_last = null;
      k_id_last = null;
      select k_amount_approved, pr_product_kind, k_id, k_disb_date
      from kontstate, kontract, product
      where ks_ol_is_delete = 0
      and ks_state = 1
      and :kf_disb_date between ks_date_begin and ks_date_end
      and ks_from_k_id = k_id
--      and ks_from_k_id <> :k_id
      and k_from_kf_id = :kf_id
      and k_from_pr_id = pr_id
       and pr_product_kind = 1 -- выбор группового кредита
      order by k_disb_date desc
      rows 1
      into k_amount_last, pr_product_kind_last, k_id_last, k_disb_date_last;
      if (k_amount_last is null) then
      begin
        res_error_is_critical = 1;
        res_error_text = 'Əsas aktiv kredit mövcüd deyil/Отсутствует активный основной кредит';
        suspend;
      end else
      if (pr_product_kind_last = 2) then
       begin
        res_error_is_critical = 1;
        res_error_text = 'Müqavilən aktiv parallel krediti mövcüddur/У данного контракта уже имеется активный параллельный кредит.';
        suspend;
      end else
      if (pr_product_kind_last = 1) then  -- если Основной кредит групповой то проверяет следующее
      begin
        if (k_kolich_repay not between 3 and 18) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Parallel kredit 3 aydan 18 ayadək verilir/Параллельный кредит выдается сроком от 3 до 18 месяцев';
          suspend;
         end else
  /*      if ((select count(kc2.kc_from_c_id)  -- проверка на то что из основного кредита не менее 2х активных участников берут параллельный
             from knclient kc1, knclient kc2
             where kc1.kc_from_k_id = :k_id_last
              and kc1.kc_amount_approved > 0
             and kc2.kc_from_k_id = :k_id
             and kc2.kc_amount_approved > 0
              and kc1.kc_from_c_id = kc2.kc_from_c_id) < 2 and :k_id <> 0) then
        begin
          res_error_is_critical = 1;
          res_error_text = 'Параллельный кредит могут получить не менее 2 человека из активных клиентов';
        end else  */
         if ((kf_disb_date - k_disb_date_last) / 30 < 2) then
         begin
           res_error_is_critical = 1;
           res_error_text = 'Əsas kreditin buraxılış tarixindən 2 ay keçməyib/После выдачи основного кредита еще 2 месяца не прошло.';
           suspend;
         end else
         if (coalesce((select rep_delinq_days from
            aarep_kont_delinq_status(:k_id_last,current_date)),0) > 0) then  -- если заявка то проверяет на просрочку на текущую дату
         begin
           res_error_is_critical = 1;
           res_error_text = 'Əsas qrup krediti gecikdirilib/Основной групповой кредит просрочен.';
           suspend;
         end else
         begin
--           Вычиление статуса группы
           select C_LAST_BONUS_STATUS
           from HP_KONTSTATE_LOAN_STATUS(:k_id_last, 0, 0, :kf_disb_date, '')
           into :C_LAST_BONUS_STATUS;
           if (:C_LAST_BONUS_STATUS is null) then C_LAST_BONUS_STATUS = -1;
           if (:C_LAST_BONUS_STATUS < 1) then
           begin
             res_error_is_critical = 1;
             res_error_text = 'Əsas qrup kreditin statusu ilə parallel kreditin buraxılışı mümkün deyil/Статус основной группы не позволяет получить паралельный кредит.';
             suspend;
           end else
           begin
--              Вычисляет баланс основного группового кредита
             select coalesce(sum(ok_os),0) from operkred
             where ok_ol_is_active=1
             and ok_ol_date <= :kf_disb_date
             and ok_from_k_id = :k_id_last
             and ok_type in (4,5,6,8)
             and ok_from_c_id=0
             into k_last_balance;
--             Вычисляет баланс основного контракта и суммы выдачи параллельного кредита на уровне клиента
             if (k_last_balance > 0) then
             begin
                max_balance = 0;
                kc_amount_last = 0;
                kc_last_balance = 0;
                kc_amount_approved = 0;
                    for
                    select  kc1.kc_amount_approved,
                            (kc1.kc_amount_approved / :k_amount_last * :k_last_balance),  -- баланс клиента предыдущего Основного контракта
                            mac_amount -- Сумма в заявке
                    from knclient kc1, miniappclient
                    where kc1.kc_from_k_id = :k_id_last -- Основной контракт
                    and kc1.kc_amount_approved > 0
                    and mac_from_ma_id = :ma_id     -- Заявка
                    and mac_amount > 0
                    and kc1.kc_from_c_id = mac_from_c_id
                    into kc_amount_last, kc_last_balance1, kc_amount_approved1
                    do
                    begin
                      if (max_balance < kc_last_balance1 + kc_amount_approved1) then
                      begin
                        max_balance = kc_last_balance1 + kc_amount_approved1;
                        kc_last_balance = kc_last_balance1;
                        kc_amount_approved = kc_amount_approved1;
                      end
                      if (:C_LAST_BONUS_STATUS = 1 and kc_amount_approved1 > kc_amount_last / 2) then
                      begin
                         res_error_is_critical = 1;
                         res_error_text = 'Bürünc status ilə əsas kredit buraxılış məbləğindən 50 % çox götürmək mümkün deyil/C бронзовым статусом не разрешено брать более 50% от выданной суммы основного кредита';
                         suspend;
                      end
                    end
            end else
            begin
              res_error_is_critical = 1;
              res_error_text = 'Əsas müqavilənin aktiv balansı mövcüd deyil/У основного контракта нету активного баланса';
              suspend;
            end
            if (res_error_text = '') then
            begin
                if (kc_last_balance + kc_amount_approved > 3000.00) then
                begin
                  res_error_is_critical = 1;
                  res_error_text = 'Əsas və parallel kreditin balansı 3000 AZN-dən artıqdır/Общий баланс основного и параллельного кредита превышает 3000 манатов';
                  suspend;
                end else
                if (kc_amount_approved > kc_amount_last) then
                begin
                  res_error_is_critical = 1;
                  res_error_text = 'Parallel kreditin məbləği əsas kreditin məbləğindən artıqdır/Сумма параллельного кредита превышает сумму основного кредита';
                  suspend;
                end
            end
          end
        end
      end else
      begin
        res_error_is_critical = 1;
        res_error_text = 'Parallel kreditlər ancaq qrup kreditləri üçün nəzərdə tutulub/Параллельные создаются только для групповых кредитов.';
        suspend;
      end
    end
  end
end
