CREATE PROCEDURE FIND_PAYSYS_PACKETS(
    PROJECT_ID INTEGER,
    KIND INTEGER,
    ID DECIMAL(18,0))
RETURNS (
    FIRST_PSI_ID DECIMAL(18,0),
    FIRST_PSI_CODE VARCHAR(20),
    PSI_COUNT INTEGER,
    LAST_PSI_CODE VARCHAR(20),
    LAST_PSI_ID DECIMAL(18,0))
AS
declare variable psi_id decimal(18,0);
declare variable psi_code varchar(20);
begin
-- Выдает список пачек бланков по проекту, по типу и ID источника
-- kind = 1 поиск в офисе
-- kind = 2 поиск у сотрудника
  last_psi_id = null;
  for select psi_id, psi_code
      from paysys_list, paysys_item
      where psl_from_psi_id = psi_id and
            psl_object_kind = :kind and
            psl_object_id = :id and
            psl_is_delete = 0 and
            psi_from_psp_id = :project_id
      group by psi_id, psi_code
      having sum(psl_sign) = 1
      into :psi_id, :psi_code
  do
  begin
    if (last_psi_id is null) then
    begin
      last_psi_id = psi_id;
      last_psi_code = psi_code;
      first_psi_id = psi_id;
      first_psi_code = psi_code;
    end
    else
    if (not(psi_id = last_psi_id + 1 and
        cast(psi_code as decimal(18,0)) = cast(last_psi_code as decimal(18,0)) + 1)) then
    begin
-- Встретили разрыв, пачка кончилась
      psi_count = last_psi_id - first_psi_id + 1;
      suspend;
-- Начинаем новую
      first_psi_id = psi_id;
      first_psi_code = psi_code;
    end
    last_psi_id = psi_id;
    last_psi_code = psi_code;
  end
  if (last_psi_id is not null) then
  begin
    psi_count = last_psi_id - first_psi_id + 1;
    suspend;
  end
end
