CREATE PROCEDURE HP_SOLVE_PROBLEM_VEREFY_NUMBER(
    C_ID D_ID)
RETURNS (
    RES_ERROR_TEXT D_STR_255,
    RES_INFO_TEXT D_STR_255)
AS
declare variable OVS_ID D_INTEGER;
declare variable OVS_FROM_OPR_ID D_ID;
declare variable OVS_VALUE D_STR_255;
declare variable OVS_ADD_VALUE D_CURRENCY;
declare variable OVS_DATETIME D_DATETIME;
declare variable OVS_DATE_BEGIN D_DATE;
declare variable PREV_DATE_BEGIN D_DATETIME;
declare variable PREV_VALUE D_STR_255;
declare variable PREV_ADD_VALUE D_CURRENCY;
declare variable OK_CELL_MAIN D_STR_100;
begin
  --RFCRU-2760 Процедура команда выполнить для модуля "Верифицированный номер не воспринимается подтверждённым"

      res_error_text = '';
      res_info_text = '';

      select ok_cell_main
      from continfo con
      where con.cci_object_kind = f_objkind('clients', 'id')
        and con.cci_object_id = :c_id
      into ok_cell_main;

      select
              ovs.ovs_id,
              ovs.ovs_from_opr_id,
              ovs.ovs_value,
              ovs.ovs_add_value,
              ovs.ovs_datetime,
              ovs.ovs_date_begin
      from objectvalues ovs
      inner join objectprops opr on ovs.ovs_from_opr_id = opr.opr_id
      inner join refdata rd on rd.rd_id = opr.opr_subject_id
      where opr.opr_object_kind = f_objkind('clients', 'id')
        and opr.opr_object_id = :c_id
        and opr.opr_subject_kind = f_objkind('refdata', 'id')
        and ovs.ovs_is_active = 1
        and rd.rd_code = 'ContactInfo.Cell_Main'
      order by ovs.ovs_id desc    --Берем последную запись
      rows 1
      into ovs_id,
           ovs_from_opr_id,
           ovs_value,
           ovs_add_value,
           ovs_datetime,
           ovs_date_begin;

      if(ovs_id is null or ovs_from_opr_id is null) then
      begin
        res_error_text = 'Номер не найден!';
        suspend;
        exit;
      end

      if(ovs_id > 1) then
      begin
        select
            ovs.ovs_value,
            ovs.ovs_add_value,
            ovs.ovs_date_begin
        from objectvalues ovs                -- получаем предыдущий номер
        where ovs.ovs_id = :ovs_id - 1
          and ovs.ovs_from_opr_id = :ovs_from_opr_id
        into  prev_value,
              prev_add_value,
              prev_date_begin;
      end

      if(prev_value is null and prev_add_value is null and prev_date_begin is null) then
      begin
        res_error_text = 'Предыдущий номер не найден!';
        suspend;
        exit;
      end 

      if(ovs_add_value = 2 and coalesce(ovs_value, '') = '') then -- если верефицированный номер не заполнен
      begin

        if(coalesce(prev_value, '') <> '') then
        begin
            --обновляем верефицированный номер
            update objectvalues ovs
            set ovs.ovs_value = :prev_value
            where ovs.ovs_id = :ovs_id
              and ovs.ovs_from_opr_id = :ovs_from_opr_id;

           ovs_value = prev_value;
        end
      end

      if(ovs_add_value = 0 and
         prev_add_value = 2 and
         prev_value = ovs_value and
         prev_date_begin = ovs_date_begin) then  -- если номер не верефицирован и дата начала и номер совпадает с предыдущим верефицированнмым
      begin
         -- удаляем последную запись
         delete from objectvalues ovs
         where ovs.ovs_id = :ovs_id
           and ovs.ovs_from_opr_id = :ovs_from_opr_id;
      end

      if (coalesce(ok_cell_main, '') = '' and not coalesce(ovs_value, '') = '') then
      begin
            --Если номера нет в continfo добавляем ее
            update continfo con
            set con.OK_CELL_MAIN = :ovs_value
            where con.cci_object_id = :c_id
              and con.cci_object_kind = f_objkind('clients', 'id');
      end

      res_info_text = 'Проблема решена';
      suspend;
end
