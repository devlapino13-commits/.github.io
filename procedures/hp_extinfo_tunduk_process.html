CREATE PROCEDURE HP_EXTINFO_TUNDUK_PROCESS(
    EIL_ID D_ID)
AS
declare variable eil_status d_smallint;
declare variable eil_from_c_id d_id;
declare variable eil_from_mcl_id d_id;
begin
    -- RFCRU-4656
    select
        eil.eil_status,
        eil.eil_from_c_id,
        eil.eil_from_mcl_id
    from extinfolist eil
    where eil.eil_id = :eil_id
    into
        :eil_status,
        :eil_from_c_id,
        :eil_from_mcl_id;

    if (eil_status <> 1) then
    begin
        update grs_clients gc
        set gc.gc_is_last = 0
        where (gc.gc_from_c_id = :eil_from_c_id or gc.gc_from_mcl_id = :eil_from_mcl_id) and
              gc.gc_is_last = 1;

        insert into grs_clients (gc_from_c_id, gc_from_eil_id, gc_status, gc_is_last, gc_from_mcl_id)
        values (:eil_from_c_id, :eil_id, :eil_status, 1, :eil_from_mcl_id);
    end

    if (eil_status = 2 and coalesce(eil_from_mcl_id, 0) > 0) then
         execute procedure mc_create_action_facerecog(:eil_from_mcl_id);


    if (eil_status in (-2, -3) and eil_from_mcl_id is not null) then
        execute procedure mc_send_push('mc_login', :eil_from_mcl_id,
            'Авторизация не прошла',
            'Повторите попытку позже!' || ascii_char(10) || 'Для этого нажмите на кнопку "Авторизоваться"');
end
