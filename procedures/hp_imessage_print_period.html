CREATE PROCEDURE HP_IMESSAGE_PRINT_PERIOD(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_STATUS D_SMALLINT = null,
    REP_TYPE D_INTEGER = 0,
    IM_TYPE D_INTEGER = 0)
RETURNS (
    IM_ID D_ID,
    CM_NAME D_STR_255,
    KONTRACT_NO D_STR_20,
    CALC_KONTRACT_NO D_STR_CODE,
    DISB_DATE D_DATE,
    REG_NAME D_STR_100,
    PERS_FIO D_STR_255,
    RES_SUMM ID_TYPE,
    PART_STR D_STR_2048,
    OL_DATE D_DATE,
    OL_DATETIME D_DATETIME,
    CASHIER_FIO D_STR_255,
    CASHIER_POS D_STR_255,
    CALC_CASHIER VARCHAR(512),
    CREATE_DATE D_DATE,
    CREATE_DATETIME D_DATETIME,
    CASHIER_SIGN_DATETIME D_DATETIME,
    CONTROLLER_1_FIO D_STR_100,
    CONTROLLER_1_POS D_STR_100,
    CALC_CONTROLLER_1 VARCHAR(512),
    CONTROLLER_1_SIGN_DATETIME D_DATETIME,
    CONTROLLER_2_FIO D_STR_100,
    CONTROLLER_2_POS D_STR_100,
    CALC_CONTROLLER_2 VARCHAR(512),
    CONTROLLER_2_SIGN_DATETIME D_DATETIME,
    STATUS D_INTEGER)
AS
begin
  if (rep_type <> 0) then
    exit;
  rep_date_begin = coalesce(rep_date_begin, '05.11.2020');
  rep_date_end = coalesce(rep_date_end, current_date);
  for
    select
      im_id,
      reg_code || ' ' || reg_name
    from internal_message
      left join operlist
        on ol_id = im_from_ol_id
          and is_delete = 0
      inner join region_list(:rep_region_id, 1, -1, 1)
        on id_list = im_from_reg_id
      inner join region
        on reg_unicode = im_from_reg_id
    where im_create_datetime >= :rep_date_begin
      and im_create_datetime < (:rep_date_end + 1)
      and im_status = coalesce(:rep_status, im_status)
      and im_type = :im_type
    order by reg_code, im_controller_2_sign_datetime
  into
      :im_id,
      :reg_name
  do
  begin
    select
      cm_name,
      kontract_no,
      disb_date,
      pers_fio,
      res_summ,
      part_str,
      ol_date,
      ol_datetime,
      cashier_fio,
      cashier_pos,
      create_datetime,
      cashier_sign_datetime,
      controller_1_fio,
      controller_1_pos,
      controller_1_sign_datetime,
      controller_2_fio,
      controller_2_pos,
      controller_2_sign_datetime,
      status
    from hp_imessage_print(:im_id)
    into
      :cm_name,
      :kontract_no,
      :disb_date,
      :pers_fio,
      :res_summ,
      :part_str,
      :ol_date,
      :ol_datetime,
      :cashier_fio,
      :cashier_pos,
      :create_datetime,
      :cashier_sign_datetime,
      :controller_1_fio,
      :controller_1_pos,
      :controller_1_sign_datetime,
      :controller_2_fio,
      :controller_2_pos,
      :controller_2_sign_datetime,
      :status;
    create_date = cast(create_datetime as date);
    calc_kontract_no = kontract_no || ' ' || f_datetostr(disb_date);
    calc_cashier = cashier_fio || ' ' || cashier_pos;
    calc_controller_1 = controller_1_fio || ' ' || controller_1_pos;
    calc_controller_2 = controller_2_fio || ' ' || controller_2_pos;
    suspend;
  end
end
