CREATE PROCEDURE HP_SET_INV_INFO(
    OL_ID D_ID,
    OLA_OPER_NO D_STR_CODE,
    OLA_DTSUPPLY D_DATE,
    OB_DETAIL D_STR_LONG = null)
AS
declare variable oper_no d_str_code;
declare variable dtsupply d_date;
declare variable ol_id1 d_id;
declare variable ob_number1 d_integer;
declare variable ob_number d_integer;
declare variable nump d_integer;
declare variable num d_integer;
declare variable npos d_integer;
declare variable inv_ch boolean;
begin
  if (not exists(select * from operlist where ol_id = :ol_id and is_delete = 0)) then
    exit;

  ola_oper_no = trim(coalesce(ola_oper_no, ''));
  if (ola_oper_no = '') then
  begin
    for
      select
        ol_id
      from operlist
      where ol_parent_id = :ol_id
    into
      :ol_id1
    do
      delete from operlist_add where ola_from_ol_id = :ol_id1;
    delete from operlist_add where ola_from_ol_id = :ol_id;
    exit;
  end

  select
    ola_oper_no,
    ola_dtsupply
  from hp_get_inv_info (:ol_id)
  into
    :oper_no,
    :dtsupply;

  inv_ch = (coalesce(ola_oper_no, '') <> coalesce(oper_no, '') or
      coalesce(ola_dtsupply, '01.01.1900') <> coalesce(dtsupply, '01.01.1900'));

  if (inv_ch) then
  begin
    update or insert into operlist_add (ola_from_ol_id, ola_oper_no, ola_dtsupply)
      values (:ol_id, :ola_oper_no, :ola_dtsupply)
      matching (ola_from_ol_id);

    for
      select
        ol_id
      from operlist
      where ol_parent_id = :ol_id
    into
      :ol_id1
    do
    begin
      update or insert into operlist_add (ola_from_ol_id, ola_oper_no, ola_dtsupply)
        values (:ol_id1, :ola_oper_no, :ola_dtsupply)
        matching (ola_from_ol_id);
    end
  end

  nump = coalesce((
    select count(*)
    from operbook
    where ob_from_ol_id = :ol_id), 0);
  num = coalesce((
    select count(*)
    from operlist
      inner join operbook
        on ob_from_ol_id = ol_id
    where ol_parent_id = :ol_id), 0);

  for
    select
      ob_number,
      ob_detail
    from operbook
    where ob_from_ol_id = :ol_id
    order by ob_number
  into
    :ob_number,
    :ob_detail
  do
  begin
    npos = position('сч-ф.№', ob_detail);
    if (npos > 0) then
      ob_detail = trim(left(ob_detail, npos - 1));
    if (coalesce(ola_oper_no, '') <> '') then
      ob_detail = ob_detail || ' сч-ф.№ ' || ola_oper_no ||
        iif(ola_dtsupply is null, '', ' от ' || f_datetostr(ola_dtsupply) || 'г.');

    update operbook set
      ob_detail = :ob_detail
    where ob_from_ol_id = :ol_id
      and ob_number = :ob_number
      and ob_detail <> :ob_detail;

    if (nump = num) then
    begin
      select
        ob_from_ol_id,
        ob_number
      from operbook
        inner join operlist
          on ol_parent_id = :ol_id
            and ol_id = ob_from_ol_id
      order by ol_id, ob_number
      rows :ob_number to :ob_number
      into
        :ol_id1,
        :ob_number1;
      update operbook set
        ob_detail = :ob_detail
      where ob_from_ol_id = :ol_id1
        and ob_number = :ob_number1
        and ob_detail <> :ob_detail;
    end else
    if (nump = 1 and num > 1) then
    begin
      for
        select
          ob_from_ol_id,
          ob_number
        from operbook
          inner join operlist
            on ol_parent_id = :ol_id
              and ol_id = ob_from_ol_id
        order by ol_id, ob_number
      into
        :ol_id1,
        :ob_number1
      do
        update operbook set
          ob_detail = :ob_detail
        where ob_from_ol_id = :ol_id1
          and ob_number = :ob_number1
          and ob_detail <> :ob_detail;
    end
  end
end
