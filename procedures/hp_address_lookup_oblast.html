CREATE PROCEDURE HP_ADDRESS_LOOKUP_OBLAST(
    COUNTRY_PL_ID D_ID)
RETURNS (
    PL_ID D_ID,
    PL_NAME D_STR_255)
AS
declare variable pl_name_komitet d_str_255;
declare variable plt_short_name d_str_100;
declare variable plt_short_name_side d_smallint;
declare variable plt_alt_short_name d_str_100;
declare variable plt_alt_short_name_side d_smallint;
begin
  if (country_pl_id is null) then
    exit;

  for select pl_id, pl_name, pl_name_komitet,
             plt_short_name, plt_short_name_side,
             plt_alt_short_name, plt_alt_short_name_side
  from places, placetype
  where parent = :country_pl_id and
        pl_status = plt_id
  into pl_id, pl_name, pl_name_komitet,
       plt_short_name, plt_short_name_side,
       plt_alt_short_name, plt_alt_short_name_side
  do
  begin
    if (plt_short_name is null) then
      plt_short_name = '';

    if (plt_alt_short_name <> '') then
      plt_alt_short_name = '';

    if (coalesce(pl_name_komitet, '') <> '') then
    begin
      if (pl_name not containing pl_name_komitet and pl_name_komitet not containing pl_name) then
      begin
        if (plt_short_name <> '') then
        begin
          if (plt_short_name_side = -1) then
            pl_name = plt_short_name || ' ' || pl_name;
          else
          if (plt_short_name_side = 1) then
            pl_name = pl_name || ' ' || plt_short_name;
        end

        if (plt_alt_short_name <> '') then
        begin
          if (plt_alt_short_name_side = -1) then
            pl_name_komitet = plt_alt_short_name || ' ' || pl_name_komitet;
          else
          if (plt_alt_short_name_side = 1) then
            pl_name_komitet = pl_name_komitet || ' ' || plt_alt_short_name;
        end

        pl_name = pl_name || '   / ' || pl_name_komitet;
      end
      else
      begin
        if (plt_short_name <> '') then
        begin
          if (plt_short_name_side = -1) then
            pl_name = plt_short_name || ' ' || pl_name;
          else
          if (plt_short_name_side = 1) then
            pl_name = pl_name || ' ' || plt_short_name;
        end
      end
    end

    pl_name = replace(pl_name, '. ', '.');

    suspend;
  end
end
