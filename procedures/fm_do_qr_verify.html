CREATE PROCEDURE FM_DO_QR_VERIFY(
    QR_CODE D_STR_100,
    GPS_LAT D_DOUBLE,
    GPS_LONG D_DOUBLE,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable C_ID D_ID;
declare variable QRL_ID D_ID;
declare variable OBJ_KIND D_INTEGER;
declare variable QRL_FROM_C_ID D_ID;
declare variable MAX_DISTANCE D_INTEGER;
declare variable MAX_TIME_SECOND D_INTEGER;
declare variable GPS_DISTANCE D_DOUBLE;
declare variable DIFF_SECOND D_INTEGER;
declare variable C_NAME_FIO D_STR_100;
declare variable MCL_ID D_ID;
declare variable CLT_ID D_INTEGER;
declare variable FOUND_COUNT D_INTEGER;
declare variable MCL_LOGIN D_STR_CODE;
declare variable DB_ID D_INTEGER;
begin
  -- Процедура проводит QR-верификацию ранее сгенерированного в МП клиенте QR-кода
  -- Вызывается из МП сотрудников

  db_id = f_get_db_id();

  res_is_error = 1;
  res_text = '';

  max_distance = 500; -- максимально допустимая дистанция в метрах
  max_time_second = 10; -- максимально допустимая разница по времени между генерацией и верификацией QR кода

  select un_from_c_id
  from username
  where un_id = :un_id
  into c_id;

  if (res_text = '' and
      c_id is null) then
    res_text = 'ID сотрудника не определен';

  if (res_text = '' and
      coalesce(trim(qr_code), '') = '') then
    res_text = 'QR код пустой';

  if (res_text = '' and
      (coalesce(gps_lat, 0) = 0 or
      coalesce(gps_long, 0) = 0)) then
    res_text = 'Местонахождение не определено';

  if (res_text = '') then
  begin
    select obj_kind, obj_id
    from hp_uuid(:qr_code)
    into :obj_kind, :qrl_id;

    if (res_text = '' and
        qrl_id is null) then
      res_text = 'QR код не найден в БД';

    if (res_text = '' and f_objkind(obj_kind, 'name') <> 'qr_list') then
      res_text = 'Несоответствующий QR код';
  end

  if (res_text = '' and
      exists(select *
             from qr_verify
             where qrv_from_qrl_id = :qrl_id)) then
    res_text = 'Этот QR-код ранее уже был отсканирован. Для верификации клиент должен повторно сгенерировать QR-код.';

  if (res_text = '') then
  begin
    insert into qr_verify (qrv_from_qrl_id, qrv_from_c_id, qrv_gps_lat, qrv_gps_long)
    values (:qrl_id, :c_id, :gps_lat, :gps_long);

    select qrl_from_mcl_id, qrl_from_c_id, f_gps_distance(qrl_gps_lat, qrl_gps_long, qrv_gps_lat, qrv_gps_long),
           datediff(second, qrl_datetime, qrv_datetime)
    from qr_list, qr_verify
    where qrl_id = :qrl_id and
          qrl_id = qrv_from_qrl_id
    into mcl_id, qrl_from_c_id, gps_distance, diff_second;

    if (res_text = '' and
        coalesce(gps_distance, max_distance + 1) > max_distance) then
      res_text = 'QR код не верифицирован: дистанция';

    if (res_text = '' and
        coalesce(diff_second, max_time_second + 1) > max_time_second) then
      res_text = 'QR код не верифицирован: устарел';

    -- если клиент не идентифицирован в МП, пробуем найти по номеру телефону
    if (res_text = '' and
        qrl_from_c_id is null) then
    begin
      select mcl_login, mcl_from_c_id
      from mc_login
      where mcl_id = :mcl_id
      into mcl_login, qrl_from_c_id; -- mcl_login - номер телефона

      if (qrl_from_c_id is null) then
      begin
        clt_id = f_cont_id(mcl_login, 'null');

        if (clt_id is not null) then
        begin
          -- Получаем количество клиентов с таким основным номером телефона
          select max(cda_from_c_id), count(*)
          from contdata
          where cda_cont_kind = f_enum('cont_kind', 'cell', 'id') and 
                cda_from_clt_id = :clt_id and
                cda_is_delete = 0 and
                cda_is_main = 1
          into qrl_from_c_id, found_count;

          -- Если найден один клиент, прикрепляем C_ID к пользователю
          if (found_count = 1) then
          begin
            update mc_login
            set mcl_from_c_id = :qrl_from_c_id
            where mcl_id = :mcl_id;
          end

          if (found_count > 1) then
          begin
            qrl_from_c_id = null;
            res_text = 'Найдено несколько клиентов с основным номером %1. Требуется прикрепление паспорта через МП клиента.';
            res_text = f_format(res_text, mcl_login);
          end
        end
      end

      if (res_text = '') then
      begin
        if (qrl_from_c_id is not null) then
        begin
          update qr_list
          set qrl_from_c_id = :qrl_from_c_id
          where qrl_id = :qrl_id;
        end
        else
          res_text = 'ID клиента не определен, возможно клиент еще не создан в БД';
      end
    end
  end

  if (db_id = 81001 and qrl_from_c_id is not null) then
  begin
    if (exists(select *
               from hp_check_oivo_assignment(:qrl_from_c_id, :un_id)
               where is_oivo = 1 and
                     res_id <> 1)) then
    begin
      res_text = 'Верификация не прошла, так как клиент закреплен не за Вами!';
      suspend;
      exit;
    end
  end

  if (res_text = '') then
  begin
    update qr_list
    set qrl_state = 2
    where qrl_id = :qrl_id;

    select c_name_fio
    from clients
    where c_id = :qrl_from_c_id
    into c_name_fio;

    res_is_error = 0;
    res_text = f_format('%1,<br>QR верификация прошла успешно!', c_name_fio);
  end

  suspend;
end
