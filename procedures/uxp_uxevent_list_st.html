Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$UXEVENT_LIST_ST(
    UFI_UUID D_UUID,
    UFI_EVENT_KIND D_INTEGER,
    IS_TARGET D_BOOL)
RETURNS (
    EVT_UUID D_UUID,
    EVT_SENDER_UUID D_UUID,
    EVT_EVENT_KIND D_INTEGER,
    EVT_TARGET_UUID D_UUID,
    EVT_HANDLER_KIND D_INTEGER,
    EVT_POSITION D_INTEGER,
    EVT_UFR_UUID D_UUID,
    EVT_EXPRESSION D_STR_8100,
    EVT_LINK_KIND D_SMALLINT)
AS
declare variable ux_kind d_str_code;
begin
  if (ufi_uuid is null) then
    exit;

  if (is_target is null) then
    is_target = 0;

  if (coalesce(ufi_event_kind, 0) = 0) then
    ufi_event_kind = null;

  if (is_target = 0) then
  begin
    for select evt_uuid, evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position
    from ux$event
    where evt_sender_uuid = :ufi_uuid and
          evt_event_kind = coalesce(:ufi_event_kind, evt_event_kind)
    order by evt_sender_uuid, evt_event_kind, evt_position
    into evt_uuid, evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position
    do
    begin
      evt_ufr_uuid = null;
      evt_expression = '';
      select exp_expression, exp_calc_ufr_uuid
      from ux$expression
      where exp_ux_uuid = :evt_uuid
      into :evt_expression, :evt_ufr_uuid;

      ux_kind = uxf$code(evt_target_uuid, 'kind');
  
      evt_link_kind = 0;
  
      if (ux_kind = 'uxevent') then
        evt_link_kind = 1; -- Связанный обработчик

      suspend;
    end
  end else
  begin
    for select evt_uuid, evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position
    from ux$event
    where evt_target_uuid = :ufi_uuid and
          evt_handler_kind = coalesce(:ufi_event_kind, evt_event_kind)
    order by evt_sender_uuid, evt_event_kind, evt_position
    into :evt_uuid, :evt_sender_uuid, evt_event_kind, evt_target_uuid, evt_handler_kind, evt_position
    do
    begin
      evt_ufr_uuid = null;
      evt_expression = '';
      select exp_expression, exp_calc_ufr_uuid
      from ux$expression
      where exp_ux_uuid = :evt_uuid
      into :evt_expression, :evt_ufr_uuid;

      ux_kind = uxf$code(evt_target_uuid, 'kind');
  
      evt_link_kind = 0;
  
      if (ux_kind = 'uxevent') then
        evt_link_kind = 1; -- Связанный обработчик

      suspend;
    end
  end
end
