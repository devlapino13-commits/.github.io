CREATE PROCEDURE HP_KONT_RESTRUCT_DEL(
    REP_KONTRACT_NO D_STR_20,
    REP_DATE D_DATE,
    REP_UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable k_id d_id;
declare variable ok_type d_integer;
declare variable ok_ol_date d_date;
declare variable ol_datetime d_datetime;
declare variable ol_datetime_restruct d_datetime;
declare variable i d_integer;
begin
  res_info_text = '';
  res_error_text = '';
  i = 0;

  select k_id
  from kontract
  where k_kontract_no = :rep_kontract_no
  into k_id;

  if (k_id is null) then
    res_error_text = 'Ошибка: контракт не найден';

  -- операции по контаркту на заданную дату или позже
  for select ol_datetime,
             ok_type,
             ok_ol_date
      from operkred
      inner join operlist on ol_id = ok_from_ol_id
      where ok_ol_is_active = 1 and
            ok_from_k_id = :k_id and
            ok_ol_date >= :rep_date
      order by ok_ol_date, ok_type
      into ol_datetime,
           ok_type,
           ok_ol_date
  do
    if (ok_type = 2 and ok_ol_date = rep_date) then
      begin
        i = i + 1; -- счетчик записей (проводок) реструктуризации на заданную дату
        ol_datetime_restruct = ol_datetime; -- запоминаем время реструктуризации
      end
    else if (ok_ol_date = rep_date and ol_datetime_restruct > ol_datetime) then
      continue; -- в день реструктуризации была другая операция, даем удалить если время реструктуризации позже
    else
      res_error_text = 'Ошибка: есть другая операция на заданную дату или позже';

  if (exists(select *
             from kontdata
             where kd_ol_is_delete = 0 and
                   kd_from_k_id = :k_id and
                   kd_ol_date >= :rep_date)) then
    res_error_text = 'Ошибка: есть изменения ставки на заданную дату или позже';

  if (res_error_text = '') then
  begin
     -- ошибок нет
    insert into operkred_restruct_del (ord_ok_from_ol_id, ord_ok_number, ord_ok_from_k_id, ord_ok_sh_id, ord_ok_sh_date,
      ord_ok_type, ord_ok_os, ord_ok_perc, ord_ok_fine, ord_ok_pay_type, ord_ok_kommis, ord_ok_sh_os, ord_ok_sh_perc,
      ord_ok_kommis_adv, ord_ok_tax, ord_ok_ol_date, ord_ok_ol_reg_id, ord_ok_ol_is_active, ord_ok_from_c_id, ord_ok_from_un_id)
    select ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc, ok_fine, ok_pay_type,
      ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date, ok_ol_reg_id, ok_ol_is_active, ok_from_c_id, :rep_un_id
    from operkred
    where ok_ol_is_active = 1 and
          ok_type = 2 and
          ok_from_k_id = :k_id and
          ok_ol_date = :rep_date;

    delete from operkred
    where ok_ol_is_active = 1 and
          ok_type = 2 and
          ok_from_k_id = :k_id and
          ok_ol_date = :rep_date;

    res_info_text = 'Реструктуризация успешно удалена,' || ascii_char(13) || ascii_char(10) || 'количество удаленных записей - ' || i;
  end

  suspend;
end
