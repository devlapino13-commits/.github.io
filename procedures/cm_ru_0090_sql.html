CREATE PROCEDURE CM_RU_0090_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable k_kolich_repay integer;
declare variable c_id id_type;
declare variable k_from_kf_id id_type;
declare variable re_k_id id_type;
declare variable re_kolich_repay integer;
declare variable delinq_day integer;
declare variable max_delinq_day integer;
declare variable kol_delinq integer;
declare variable k_from_pr_id integer;
declare variable re_kol_fact_repay integer;
declare variable no_error_repay integer;
declare variable kc_circle integer;
declare variable pr_code varchar(100);
begin
  for
  select k_from_kf_id,
         k_disb_date,
         k_from_pr_id,
         pr_code
  from kontract, product
  where k_id = :k_id
  and coalesce(k_refund, 0) = 1
  and k_from_pr_id = pr_id
  and pr_code not containing '.PILOT.'
  into k_from_kf_id,
       kf_disb_date,
       k_from_pr_id,
       pr_code
  do
  begin
   /*
    if (not exists(select * from product
        where pr_id = :k_from_pr_id
        and pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.)%')) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Выберите правильный продукт для перекредитование(«Семья», «Семья Плюс», «Баракат», «Баракат Плюс»).';
      suspend;
      exit;
    end
    */
    if (not pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.)%') then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Выберите правильный продукт для перекредитование(«Семья», «Семья Плюс», «Баракат», «Баракат Плюс»).';
      suspend;
      exit;
    end
--    6.1.1.    клиент имеет активный микрозаем в Компании, который был выдан на срок от 6 месяцев включительно
--    по продуктам «Семья», «Семья Плюс», «Баракат», «Баракат Плюс».
--  при пере кредитовании новых клиентов не должны быть
    for
    select kc_from_c_id,
           kc_amount_approved
    from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    and kc_circle <= 1
    into c_id, kc_amount
    do
    begin
      select c_name_fio, c_pass_no
      from clients
      where c_id = :c_id
      into res_info_text, kf_client_pass_no;
      res_error_is_critical = 2;
      res_error_text = 'В перекредитовании новых клиентов нельзя добавлять!';
      suspend;
    end 
--  проверка активного кредита
--  по какому продукту берут то по такому долны перекредитоваться или могут другие продукты выбирать?
--  из группового не могут ведь перекредитоваться в индив?
    re_k_id = null;
    re_kolich_repay = null;
    select k_id, k_kolich_repay
    from kontract, product, kontstate
    where k_from_kf_id = :k_from_kf_id
    and k_from_pr_id = pr_id
    and pr_product_kind <> 2
    and pr_code similar to '%(.FAMILY.|.FAMILY PLUS.|.BARAKAT.|.BARAKAT PLUS.)%'
    and k_id = ks_from_k_id
    and ks_ol_is_delete = 0
    and ks_state = 1
    and :kf_disb_date between ks_date_begin and ks_date_end
    and ks_date_begin < :kf_disb_date
    order by ks_date_begin desc
    rows 1
    into re_k_id, re_kolich_repay;
    if (re_k_id is null) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Не имеется активный заем для перекредитование!';
      suspend;
      exit;
    end 
    if (coalesce(re_kolich_repay, 0) < 6) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Срок активного заема менее 6ти месяцев!';
      suspend;
      exit;
    end 
--  проверка на просрочку
    delinq_day = null;
    select :kf_disb_date - oks_delinq_date
    from oksummary
    where :kf_disb_date between oks_date_begin and oks_date_end
    and oks_from_k_id = :re_k_id
    and oks_close_state = 0
    into delinq_day;
    delinq_day = coalesce(delinq_day,0);
    if (delinq_day > 0) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'Активный заем просрочен!';
      suspend;
      exit;
    end
--  расчет максмальное просроченные дни
    max_delinq_day = null;
    kol_delinq = null;
--    select max(ok_ol_date - ok_sh_date),
--           count(*)
--    from operkred
--    where ok_ol_is_active = 1
--    and ok_from_k_id = :re_k_id
--    and ok_type in (4,5)
--    and ok_ol_date > ok_sh_date
--    and ok_os + ok_perc < 0
    with tabl1 as (
      select distinct ok_ol_date, ok_sh_date
      from operkred
      where ok_ol_is_active = 1
      and ok_from_k_id = :re_k_id
      and ok_type in (4,5)
      and ok_ol_date > ok_sh_date
      and ok_os + ok_perc < 0)
    select max(ok_ol_date - ok_sh_date),
          count(*)
    from tabl1
    into max_delinq_day, kol_delinq;
    max_delinq_day = coalesce(max_delinq_day, 0);
    kol_delinq = coalesce(kol_delinq, 0);
    if (max_delinq_day > 7) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'В активном кредите допускали просрочки более 7 дней!';
      suspend;
      exit;
    end 
    if (kol_delinq > 1) then
    begin
      res_error_is_critical = 2;
      res_error_text = 'В активном кредите допускали непрерывные просрочки более одного раза!';
      suspend;
      exit;
    end
--   6.3.    Активный микрозаем может быть перекредитован на следующих условиях
--   в зависимости от срока активного микрозайма и цикла клиента:
--  вычисляем кол-во платежей и кол-во правильных платежей т.е. погашали своевременно т.е. не досрочно за 10 дней
--  и не произвели просрочку более 7ми дней
    re_kol_fact_repay = null;
    no_error_repay = null;
    with tabl1 as (
        select ok_sh_date,
               max(ok_ol_date) as ol_date,
               sum(ok_os + ok_sh_os) as os,
               sum(ok_perc + ok_sh_perc) as perc
        from operkred
        where ok_ol_is_active = 1
        and ok_type < 7
        and ok_ol_date <= :kf_disb_date
        and ok_from_k_id = :re_k_id
        and ok_sh_id > 0 -- дату выдачу не включать
        and (ok_os <> 0 or ok_sh_os <> 0
             or ok_perc <> 0 or ok_sh_perc <> 0)
        group by ok_sh_date
        having sum(ok_os + ok_sh_os) <= 0 and sum(ok_perc + ok_sh_perc) <= 0),
    tabl2 as (
        select ok_sh_date,
               ol_date,
               iif(ol_date between ok_sh_date - 10 and ok_sh_date + 7, 1,0) as no_error
        from tabl1)
    select count(*) as kol_repay,
           sum(no_error) as no_error_repay
    from tabl2
    into re_kol_fact_repay, no_error_repay;
    re_kol_fact_repay = coalesce(re_kol_fact_repay, 0);
    no_error_repay = coalesce(no_error_repay, 0);
--  проверяем каждого клиента активного кредита
    for
    select kc_from_c_id,
           kc_circle,
           kc_amount_approved
    from knclient kc1
    where kc_from_k_id = :re_k_id
    and kc_amount_approved > 0
    and exists(select * from knclient kc2
               where kc2.kc_from_k_id = :k_id
               and kc2.kc_amount_approved > 0
               and kc2.kc_from_c_id = kc1.kc_from_c_id)
    into c_id, kc_circle, kc_amount
    do
    begin
      select c_name_fio, c_pass_no from clients
      where c_id = :c_id
      into res_info_text, kf_client_pass_no;
      if (re_kolich_repay between 6 and 9) then
      begin
        if (kc_circle <= 1) then
        begin
--      если до конца срока микрозайма осталось не более 1 платежа, при условии полной оплаты*
--      предыдущих платежей по графику погашения.
          if (re_kolich_repay - re_kol_fact_repay > 1) then
          begin
            res_error_is_critical = 2;
            res_error_text = 'У активного контракта(цикл клиента=1) до конца срока осталось более одного платежа!';
            suspend;
          end
        end else
        begin
--      если до конца срока микрозайма осталось 2 платежа и менее,
--      при условии полной оплаты* предыдущих платежей по графику погашения.
          if (re_kolich_repay - re_kol_fact_repay > 2) then
          begin
            res_error_is_critical = 2;
            res_error_text = 'У активного контракта(цикл клиента 2 и более) до конца срока осталось более двух платежей!';
            suspend;
          end
        end
--      полная оплата платежей т.е. без просрочки и не досрочно
        if (no_error_repay <> re_kol_fact_repay) then
        begin
          res_error_is_critical = iif(k_id = 101010023680800201, 0, 1);
          res_error_text = 'Платежи не считаются полностью оплаченными!'; --смотрите пукнт 6.3
          suspend;
        end
      end 
      if (re_kolich_repay between 10 and 12) then
      begin
        if (kc_circle <= 1) then
        begin
          if (no_error_repay < 9) then
          begin
            res_error_is_critical = 1;
            res_error_text = 'в активном кредите не было полностью оплачено 9 платежей! (цикл клиента=1)'; --смотрите пукнт 6.3
            suspend;
          end
        end else
        begin
          if (no_error_repay < 8) then
          begin
            res_error_is_critical = 1;
            res_error_text = 'в активном кредите не было полностью оплачено 8 платежей!(цикл клиента 2 и более)'; --смотрите пукнт 6.3
            suspend;
          end
        end
      end
    end
  end
end
