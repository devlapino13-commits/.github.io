CREATE PROCEDURE ELCARD_CARD_CHECK(
    ESM_ID ID_TYPE,
    FULL_SOURCE D_BLOB_TEXT,
    ANSWER D_BLOB_TEXT,
    ERROR_CODE D_BLOB_TEXT)
AS
declare variable esm_date d_date;
declare variable code d_str_20;
begin
  error_code = trim(coalesce(error_code, ''));
  answer = trim(coalesce(answer, ''));
  full_source = trim(coalesce(full_source, ''));

  if (esm_id is null) then
    exception e_default 'ESM_ID is null';

  select esm.esm_date
  from extsysmain esm
  where esm.esm_id = :esm_id
  and esm.esm_project = 'elcard_disb'
  and esm.esm_action = 'CardCheck'
  into :esm_date;

  if (esm_date is null) then -- исхожу из того, что не может быть такого, чтобы  у extsysmain не был заполнен esm_date, поэтому если esm_date не заполняется считаю что записи вообще нет и поднимаю exception
    exception e_default 'ESM_ID is not assigned to "elcard_disb"';

  delete from extsysanswer ea
  where ea.esa_from_esm_id = :esm_id;

  if (error_code <> '') then
  begin
    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'error: ' || :error_code);
  end
  else
  begin
    code = f_json_str_get_value(answer, 'status', '');
    if (code <> '600') then
    begin
        error_code = code;

        insert into extsysanswer (esa_from_esm_id, esa_message)
        values (:esm_id, 'error: ' || :error_code);
    end
  end

  if (answer <> '') then
  begin
    insert into extsysanswer (esa_from_esm_id, esa_message)
    values (:esm_id, 'answer: ' || :answer);
  end

  update extsysmain em
  set em.esm_status = iif(:error_code = '', 1, -1),
      em.esm_full_source = :full_source
  where em.esm_id = :esm_id;
end
