CREATE PROCEDURE HP_CLIENTSTATE_GET(
    C_ID DECIMAL(18,0))
RETURNS (
    RES_DATE D_DATE,
    RES_CLIENT_STATE D_INTEGER,
    RES_REGION_ID D_INTEGER,
    RES_LO_ID D_ID,
    RES_KF_NAME D_STR_100,
    RES_CIRCLE D_INTEGER,
    RES_STAFF_STATE D_INTEGER)
AS
declare variable k_id d_id;
declare variable kf_id d_id;
declare variable ks_date_end d_date;
declare variable staff_office d_integer;
declare variable tmp_date d_date;
begin
-- Процедура возвращает актуальный статус контрагента.
  res_client_state = -1;
  res_staff_state = -1;
  res_circle = 0;

-- Ищем среди активных или закрытых кредитов
  for select ks_from_k_id, ks_date_begin, ks_date_end, k_from_kf_id
      from kontstate, knclient, kontract
      where kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            kc_from_k_id = ks_from_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 1 and
            k_id = ks_from_k_id
      order by ks_date_end, ks_date_begin desc
      into :k_id, res_date, ks_date_end, :kf_id
  do
  begin
    res_circle = res_circle + 1;
  end
  if (k_id is not null) then
  begin
    if (ks_date_end = '31.12.9999') then
      res_client_state = 1;
    else
    begin
      res_client_state = 2;
      res_date = ks_date_end + 1;
    end
  end

  if (res_client_state <> 1) then
  begin
-- Ищем среди заявок за последний 45 дней
    for select ma_from_kf_id, ma_date_app
        from miniapp, miniappclient
        where mac_from_c_id = :c_id and
              mac_amount > 0 and
              mac_from_ma_id = ma_id and
              ma_is_active = 1 and
              ma_date_end = '31.12.9999' and
              ma_date_app >= (current_date - 45)
        order by ma_date_app desc
        rows 1
        into :kf_id, :res_date
    do
    begin
      res_client_state = 0; -- Заявка
    end
  end
  if (kf_id is not null) then
  begin
    select kal_new_reg_id, kal_date, kal_new_kf_officer, kal_new_kf_name
    from knaktlist
    where knaktlist.is_delete = 0 and
          kal_date_end = '31.12.9999' and
          kal_from_kf_id = :kf_id
    into :res_region_id, :tmp_date, :res_lo_id, res_kf_name;
  end

  res_date = maxvalue(res_date, tmp_date);

  select ps_from_reg_id, ps_date_begin, ps_is_working
  from hr_pstatus
  where ps_from_c_id = :c_id and
        ps_date_end = '31.12.9999' and
        ps_status = 1
  order by ps_date_begin desc
  rows 1
  into :staff_office, tmp_date, :res_staff_state;

  res_date = maxvalue(res_date, tmp_date);

  if (res_region_id is null) then
    res_region_id = staff_office;

  if (res_staff_state is null) then
    res_staff_state = -1;

  suspend;
end
