CREATE PROCEDURE EQUIFAX_KONT_LIST(
    POCKET_ID D_INTEGER)
RETURNS (
    K_ID D_ID)
AS
declare variable ACTIVE_DATE D_DATE;
declare variable EQUIFAX_CIB_ID D_INTEGER;
declare variable CIB_CODE D_STR_100;
declare variable POCKET_TYPE D_STR_100;
declare variable REP_DATE_BEGIN D_DATE;
declare variable REP_DATE_END D_DATE;
declare variable POCKET_CREATE_DATE D_DATE;
declare variable CP_ID D_INTEGER;
begin
    select cp.cp_date_begin,
           cp.cp_date_end,
           cp.cp_pocket_type,
           rd.rd_code,
           cast(cp.cp_datetime as date),
           cp.cp_id
    from cib_pocket cp
    inner join refdata rd on cp.cp_cib_id = rd.rd_id
    where cp.cp_id = :pocket_id
    into :rep_date_begin,
         :rep_date_end,
         :pocket_type,
         :cib_code,
         :pocket_create_date,
         :cp_id;

  equifax_cib_id = f_get_rd_id('cib.equifax');

  if(cib_code = 'cib.equifax') then
  begin
      active_date  = '01.07.2014'; -- дата запуска проекта;

      if (exists (select * from cib_knclient_in_pocket ckp where ckp.ckp_from_cp_id = :cp_id)) then -- если по пакету уже имеется сформированный набор кредитов, тогда тянем эти кредиты
      begin
          for select distinct kncl.kc_from_k_id
          from cib_knclient_in_pocket ckp
          inner join knclient kncl on kncl.kc_id = ckp.ckp_from_kc_id
          where ckp.ckp_from_cp_id = :cp_id
          into :k_id
          do
          begin
            suspend;
          end
          exit;
      end

      if (pocket_type containing '.error_check.') then -- если нужно подтянуть набор кредитов, для правки в эквифакс
      begin
        for select cke.k_id
        from cib_actuality cke
        where cke.k_id <> 101010000000100101
        into :k_id
        do
        begin
            suspend;
        end
      end

      if (pocket_type containing '.errors_corrected.') then -- если это пакет по исправленным ошибкам
      begin
          for select distinct cke.cke_from_k_id
          from cib_kontract_errors cke
          inner join cib_pocket cp on cp.cp_id = cke.cke_from_cp_id
          -- отталкиваюсь именно от даты создания пакета, т.к если пакет был создан сегодня должен содержать ошибки исправленные вчера -->
          where cke.cke_checked_datetime >= :pocket_create_date - 1
          and cke.cke_checked_datetime < :pocket_create_date
          ---------------------------------------------------------------------<
          and cp.cp_date_end = :rep_date_end
          and cp.cp_cib_id = :equifax_cib_id
          into :k_id
          do
          begin
            -- интересуют только кредиты, с даты автоматизации в Эквифакс
            if (exists (select *
                        from kontract kont
                        where kont.k_id = :k_id
                        and kont.k_disb_date >= :active_date
                        )) then
            suspend;
          end
      end

      if (pocket_type containing '.rest_prev_d.') then -- все реструктуризации на дату, которые на эту дату в пакет не попали
      begin
            for select distinct ok.ok_from_k_id
            from operkred ok
            where ok.ok_type = 2
            and ok.ok_ol_is_active = 1
            and ok.ok_ol_date = :rep_date_end
            and ((ok.ok_os <> 0) or
                 (ok.ok_sh_os <> 0) or
                 (ok.ok_perc <> 0) or
                 (ok.ok_sh_perc <> 0) or
                 (ok.ok_fine <> 0))
            into :k_id
            do
            begin
                -- интересуют только кредиты, с даты автоматизации в Эквифакс
                if (exists (select *
                            from kontract kont
                            where kont.k_id = :k_id
                            and kont.k_disb_date >= :active_date
                            )) then
                begin
                    -- если кредит не попадал в пакет на эту дату
                    if (not exists(
                                select *
                                from cib_knclient_in_pocket ckp
                                inner join knclient kncl on kncl.kc_id = ckp.ckp_from_kc_id
                                inner join cib_pocket cp on cp.cp_id = ckp.ckp_from_cp_id
                                where kncl.kc_from_k_id = :k_id
                                and kncl.kc_amount_approved > 0
                                and cp.cp_pocket_type containing '.d.'
                                and cp.cp_pocket_type containing '.kont.'
                                and cp.cp_cib_id = :equifax_cib_id -- нас интересует только Эквифакс
                                and cp.cp_pocket_status > 0 -- негативный статус не учитываю
                                and :rep_date_end between cp.cp_date_begin and cp.cp_date_end
            
                    )) then
                        suspend;
                end
            end
      end

      if (pocket_type containing '.okf_pay_prev_d.') then  -- все погашения на внесистемном учете, где платеж был посажен задней датой и на эту дату кредит в пакет не попал
      begin
            for select distinct okf.oko_from_k_id
            from operkroff  okf
            inner join operlist ol on ol.ol_id = okf.oko_from_ol_id
            where okf.oko_type = 4
            and okf.oko_ol_is_active = 1
            and okf.oko_ol_date = :rep_date_end
            and okf.oko_os + okf.oko_perc <> 0
            and cast(ol.ol_datetime as date) > okf.oko_ol_date -- платеж посажен задним числом
            into :k_id
            do
            begin
                -- интересуют только кредиты, с даты автоматизации в Эквифакс
                if (exists (select *
                            from kontract kont
                            where kont.k_id = :k_id
                            and kont.k_disb_date >= :active_date
                            )) then
                begin
                    -- если кредит не попадал в пакет на эту дату
                    if (not exists(
                                select *
                                from cib_knclient_in_pocket ckp
                                inner join knclient kncl on kncl.kc_id = ckp.ckp_from_kc_id
                                inner join cib_pocket cp on cp.cp_id = ckp.ckp_from_cp_id
                                where kncl.kc_from_k_id = :k_id
                                and kncl.kc_amount_approved > 0
                                and cp.cp_pocket_type containing '.d.'
                                and cp.cp_pocket_type containing '.kont.'
                                and cp.cp_cib_id = :equifax_cib_id -- нас интересует только Эквифакс
                                and cp.cp_pocket_status > 0 -- негативный статус не учитываю
                                and :rep_date_end between cp.cp_date_begin and cp.cp_date_end
            
                    )) then
                        suspend;  
                end
            end
      end

      if (pocket_type containing '.ok_pay_prev_d.') then -- все погашения на стандатном учете, где платеж был посажен задней датой и на эту дату кредит в пакет не попал
      begin
            for select distinct ok.ok_from_k_id
            from operkred ok
            inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
            where ok.ok_ol_date = :rep_date_end
            and ok.ok_type = 4
            and ok.ok_ol_is_active = 1
            and ok.ok_os + ok.ok_perc + ok.ok_fine <> 0
            and cast(ol.ol_datetime as date) > ok.ok_ol_date -- платеж посажен задним числом  
            into :k_id
            do
            begin
                -- интересуют только кредиты, с даты автоматизации в Эквифакс
                if (exists (select *
                            from kontract kont
                            where kont.k_id = :k_id
                            and kont.k_disb_date >= :active_date
                            )) then
                begin
                    -- если кредит не попадал в пакет на эту дату
                    if (not exists(
                                select *
                                from cib_knclient_in_pocket ckp
                                inner join knclient kncl on kncl.kc_id = ckp.ckp_from_kc_id
                                inner join cib_pocket cp on cp.cp_id = ckp.ckp_from_cp_id
                                where kncl.kc_from_k_id = :k_id
                                and kncl.kc_amount_approved > 0
                                and cp.cp_pocket_type containing '.d.'
                                and cp.cp_pocket_type containing '.kont.'
                                and cp.cp_cib_id = :equifax_cib_id -- нас интересует только Эквифакс
                                and cp.cp_pocket_status > 0 -- негативный статус не учитываю
                                and :rep_date_end between cp.cp_date_begin and cp.cp_date_end
            
                    )) then
                        suspend;
                end
            end
      end

      if (pocket_type containing '.perc_ch.') then -- кредиты по которым было изменение % ставки из-за уровня просрочки
      begin
        -- кредиты по которым было измнение % ставки из-за уровня просрочки
        for select distinct kd.kd_from_k_id
        from kontdata kd
        inner join operlist ol on ol.ol_id = kd.kd_from_ol_id
        where kd.kd_ol_date = :rep_date_end
        and kd.kd_float_perc_level in (1, 2)
        and kd.kd_ol_is_delete = 0
        and ol.ol_oper_type = 1  -- на счет ol_oper_type = 1 и ol_oper_subtype = 12 подсказал Санжар
        and ol.ol_oper_subtype = 12
        and cast(ol.ol_datetime as date) > ol.ol_date -- платеж посажен задним числом
        into :k_id
        do
        begin
             -- интересуют только кредиты, с даты автоматизации в Эквифакс
            if (exists (select *
                        from kontract kont
                        where kont.k_id = :k_id
                        and kont.k_disb_date >= :active_date
                        )) then
            begin
             if (not exists(
                            select *
                            from cib_knclient_in_pocket ckp
                            inner join knclient kncl on kncl.kc_id = ckp.ckp_from_kc_id
                            inner join cib_pocket cp on cp.cp_id = ckp.ckp_from_cp_id
                            where kncl.kc_from_k_id = :k_id
                            and kncl.kc_amount_approved > 0
                            and cp.cp_pocket_type containing '.d.'
                            and cp.cp_pocket_type containing '.kont.'
                            and cp.cp_cib_id = :equifax_cib_id -- нас интересует только Эквифакс
                            and cp.cp_pocket_status > 0 -- негативный статус не учитываю
                            and :rep_date_end between cp.cp_date_begin and cp.cp_date_end
        
                )) then
                    suspend;    
            end
        end
      end
    
      if (pocket_type containing '.standard_daily.')  then  -- ЕСЛИ ЭТО СТАНДАРТНЫЙ ЕЖЕДЕНВНЫЙ ПАКЕТ
      begin
        -- активные кредиты на дату (попадает в стандартный дневной пакет)
        for select ks.ks_from_k_id
        from kontstate ks
        where ks.ks_state = 1
        and ks.ks_ol_is_delete = 0
        and :rep_date_end between ks.ks_date_begin and ks.ks_date_end
        and ks_date_begin >= :active_date  -- интересуют только кредиты, с даты автоматизации в Эквифакс
        into :k_id
        do
        begin
            suspend;
        end

        -- закрытые кредиты за период (попадает в стандартный дневной пакет)
        for select ks.ks_from_k_id
        from kontstate ks
        where ks.ks_state = 2
        and ks.ks_ol_is_delete = 0
        and ks.ks_date_begin between :rep_date_begin and :rep_date_end
        and ks_date_begin >= :active_date  -- интересуют только кредиты, с даты автоматизации в Эквифакс
        into :k_id
        do
        begin
            -- интересуют только кредиты, с даты автоматизации в Эквифакс
            if (exists (select *
                        from kontract kont
                        where kont.k_id = :k_id
                        and kont.k_disb_date >= :active_date
                        )) then
            suspend;
        end

        -- закрытые кредиты с долгом на внесистемке (попадает в стандартный дневной пакет)
        for select oko_from_k_id
        from
        (
            select okf.oko_from_k_id
            from operkroff okf
            inner join kontstate ks on ks.ks_from_k_id = okf.oko_from_k_id
            where okf.oko_ol_is_active = 1
            and okf.oko_ol_date <= :rep_date_end
            and ks.ks_state = 1
            and ks.ks_date_begin >= :active_date
            and ks.ks_ol_is_delete = 0
            group by okf.oko_from_k_id
            having sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc) > 0
        ) b
        inner join kontstate ks_close on ks_close.ks_from_k_id = oko_from_k_id
        where ks_close.ks_ol_is_delete = 0
        and ks_date_begin < :rep_date_end
        and ks_close.ks_state = 2
        into :k_id
        do
        begin
            -- интересуют только кредиты, с даты автоматизации в Эквифакс
            if (exists (select *
                        from kontract kont
                        where kont.k_id = :k_id
                        and kont.k_disb_date >= :active_date
                        )) then
            suspend;    
        end

        -- кредиты, по которым долг по внесистемке был закрыт в рамках периода пакета (попадает в стандартный дневной пакет)
        for select distinct okf.oko_from_k_id -- получаю список погашений и списаний по внесистмке за период
        from operkroff okf
        where okf.oko_ol_is_active = 1
        and okf.oko_type in (4,5)
        and okf.oko_ol_date between :rep_date_begin and :rep_date_end
        into :k_id
        do
        begin
            -- интересуют только кредиты, с даты автоматизации в Эквифакс
            if (exists (select *
                        from kontract kont
                        where kont.k_id = :k_id
                        and kont.k_disb_date >= :active_date
                        )) then
            begin
              -- если после погашения долга по внесистемке больше нет
              if ((select sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc)
                   from operkroff okf
                   where okf.oko_type in (1, 4, 5)
                   and okf.oko_ol_is_active =1
                   and okf.oko_from_k_id = :k_id
                   and okf.oko_ol_date <= :rep_date_end) <= 0) then
              begin
                suspend;
              end
            end
        end

        -- кредиты по которым было измнение % ставки из-за уровня просрочки
        for select distinct kd.kd_from_k_id
        from kontdata kd
        inner join operlist ol on ol.ol_id = kd.kd_from_ol_id
        where kd.kd_ol_date between :rep_date_begin and :rep_date_end
        and kd.kd_float_perc_level in (1, 2)
        and kd.kd_ol_is_delete = 0
        and ol.ol_oper_type = 1  -- на счет ol_oper_type = 1 и ol_oper_subtype = 12 подсказал Санжар
        and ol.ol_oper_subtype = 12
        into :k_id
        do
        begin
             -- интересуют только кредиты, с даты автоматизации в Эквифакс
            if (exists (select *
                        from kontract kont
                        where kont.k_id = :k_id
                        and kont.k_disb_date >= :active_date
                        )) then
            begin
                suspend;    
            end
        end
    end
  end -- < --- if(CIB_CODE = 'CIB.EQUIFAX') then
  if (cib_code = 'cib.nbki') then  -- ЕСЛИ ЭТО ПАКЕТ ПО НБКИ
  begin
    for select nkl.nkl_from_k_id
    from nbki_kontract_list nkl
    where :rep_date_end between nkl.nkl_date_begin and nkl.nkl_date_end
    into :k_id
    do
    begin
        suspend;
    end
  end
end
