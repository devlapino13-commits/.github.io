CREATE PROCEDURE UXP$UXFRAME_SQL_MODIFY(
    FR_UUID D_UUID,
    DATA_KIND D_INTEGER,
    FR_UUID_CODE D_STR_CODE = null)
RETURNS (
    SQL_TEXT D_STR_8100,
    EXEC_PROC D_BOOL)
AS
declare variable ufs_field_name d_str_code;
declare variable ufp_field_name d_str_code;
declare variable field_list d_str_8100;
declare variable enter_sym d_str_10;
declare variable sql_where d_str_long;
declare variable data_source d_str_code;
declare variable is_table d_bool;
declare variable uf_pk_field d_smallint;
declare variable value_list d_str_8100;
declare variable f_temp d_str_255;
declare variable frame_us_uuid d_uuid;
declare variable data_us_uuid d_uuid;
declare variable fr_code_msg d_str_100;
declare variable ufs_uf_uuid d_uuid;
declare variable is_blob d_bool;
declare variable need_sql d_bool;
declare variable sql_kind d_str_code;
declare variable ufs_default d_str_100;
declare variable us_autoincfield d_str_code;
declare variable ufp_uuid d_uuid;
declare variable ufs_is_param d_bool;
declare variable ufp_is_param d_bool;
declare variable ufr_add_table_fields d_str_code;
declare variable main_us_uuid d_uuid;
begin
  -- Приготовление Update sql для фрейма

  if (coalesce(fr_uuid, '') = '') then
    fr_uuid = uxf$uniq(fr_uuid_code);

  fr_code_msg = uxf$code(fr_uuid, 'fcode');

  if (data_kind is null) then
    exception e_default f_format('Empty data_kind, frame: %1', fr_code_msg);

  if (data_kind not in (1, 2, 3)) then
    exception e_default f_format('Undefined data_kind, frame: %1, data_kind: %2', fr_code_msg, data_kind);

  exec_proc = 0;
  sql_text = '';
  field_list = '';
  value_list = '';
  sql_where = '';
  enter_sym = ascii_char(13) || ascii_char(10);

  f_temp = 'coalesce(uxf$set_blob(:new_%1$blob, :new_%1$file), :new_%1)';

  select ufr_us_uuid, ufr_add_table_fields,
         iif(:data_kind = 1, ufr_can_update, iif(:data_kind = 2, ufr_can_insert, ufr_can_delete))
  from ux$frame
  where ufr_uuid = :fr_uuid
  into frame_us_uuid, ufr_add_table_fields, need_sql;

  if (need_sql = 0) then
  begin
    suspend;
    exit;
  end

  if (ufr_add_table_fields <> '') then
  begin
    select us_uuid
    from ux$source
    where us_code = :ufr_add_table_fields
    into main_us_uuid;
  end

  select ufd_us_uuid, us_code, us_is_table, us_autoincfield
  from ux$frdata, ux$source
  where ufd_ufr_uuid = :fr_uuid and
        ufd_kind = :data_kind and
        ufd_us_uuid = us_uuid
  into data_us_uuid, data_source, is_table, us_autoincfield;

  if (row_count = 0) then
    exception e_default f_format('Data source not found, frame: %1', fr_code_msg);

  if (is_table = 1) then
  begin
    for select uf_uuid, uf_field_name, uf_pk_field
    from ux$field
    where uf_owner = :data_us_uuid and
          uf_type = 0 and
          not(uf_readonly)
    order by uf_position
    into ufs_uf_uuid, ufs_field_name, uf_pk_field
    do
    begin
      ufp_uuid = null;
      ufp_field_name = '';
      ufs_default = '';
      ufs_is_param = 0;

      ufp_is_param = ufs_is_param;

      select ufs_ufp_uuid, ufs_default
      from ux$framesql
      where ufs_ufr_uuid = :fr_uuid and
            ufs_sql_kind = :data_kind and
            ufs_uf_uuid = :ufs_uf_uuid
      into ufp_uuid, ufs_default;

      if (row_count = 0) then
      begin
        if (data_us_uuid = frame_us_uuid or data_us_uuid = main_us_uuid) then
        begin
          ufp_field_name = ufs_field_name;

          select ufp_uuid
          from ux$frparam
          where ufp_ufr_uuid = :fr_uuid and
                ufp_uf_uuid = :ufs_uf_uuid
          into ufp_uuid;
        end

        if (ufp_uuid is null) then
          Continue;
      end else
      if (ufp_uuid is not null) then
      begin
        select uf_field_name, uf_is_param
        from ux$frparam, ux$field
        where ufp_uuid = :ufp_uuid and
              ufp_uf_uuid = uf_uuid
        into ufp_field_name, ufp_is_param;

        ufs_default = '';
      end else
        if (data_kind in (1, 3)) then
          Continue;

      is_blob = 0;
      if (ufp_uuid is not null) then
      begin
        select iif(ufp_editor = 'blob', 1, 0)
        from ux$frparam
        where ufp_uuid = :ufp_uuid
        into is_blob;

        if (row_count = 0) then
          Continue;
      end

      if (not(data_kind = 2 and ufs_field_name = us_autoincfield)) then
      begin
        if (field_list <> '') then
          field_list = field_list || ', ' || enter_sym;
  
        if (data_kind = 2 and value_list <> '') then
          value_list = value_list || ', ' || enter_sym;
      end
  
      if (data_kind = 1) then
      begin
        if (is_blob = 0) then
        begin
          if (ufp_is_param = 0) then
            field_list = field_list || ufs_field_name || ' = :new_' || ufp_field_name;
          else
            field_list = field_list || ufs_field_name || ' = :' || ufp_field_name;
        end
        else
          field_list = field_list || ufs_field_name || ' = ' || f_format(f_temp, ufp_field_name);
      end

      if (data_kind = 2 and ufs_field_name <> us_autoincfield) then
      begin
        field_list = field_list || ufs_field_name;

        if (ufs_default = '') then
        begin
          if (is_blob = 0) then
          begin
            if (ufp_is_param = 0) then
              value_list = value_list || ':new_' || ufp_field_name;
            else
              value_list = value_list || ':' || ufp_field_name;
          end
          else
            value_list = value_list || f_format(f_temp, ufp_field_name);
        end else
          value_list = value_list || ufs_default;
      end
  
      if (uf_pk_field <> 0) then
      begin
        if (data_kind in (1, 3)) then
        begin
          if (sql_where <> '') then
            sql_where = sql_where || ' and ' || enter_sym;
    
          sql_where = sql_where || ufs_field_name || ' = :old_' || ufp_field_name;
        end
      end
    end

    if (data_kind = 1) then
    begin
      if (field_list <> '') then
      begin
        sql_text = 'update ' || data_source || ' set ';
        sql_text = sql_text || enter_sym || field_list;
  
        if (sql_where <> '') then
        begin
          sql_text = sql_text || enter_sym || 'where (' || sql_where || ')';
          if (us_autoincfield <> '') then
            sql_text = sql_text || enter_sym || 'returning (' || us_autoincfield || ')';
        end
        else
          exception e_message ('Update SQL: Нет условия по ключевым полям (WHERE)');
      end
    end else
    if (data_kind = 2) then
    begin
      if (field_list <> '') then
      begin
        sql_text = 'insert into ' || data_source;
        sql_text = sql_text || enter_sym || ' (' || field_list || ')';
        sql_text = sql_text || enter_sym || 'values (' || value_list || ')';

        if (us_autoincfield <> '') then
          sql_text = sql_text || enter_sym || 'returning (' || us_autoincfield || ')';
      end
    end else
    if (data_kind = 3) then
    begin
      if (sql_where <> '') then
      begin
        sql_text = 'delete from ' || data_source;
        sql_text = sql_text || enter_sym || 'where (' || sql_where || ')';
      end else
        exception e_message ('Delete SQL: Нет условия по ключевым полям (WHERE)');
    end
  end else
  begin
    for select uf_uuid, uf_field_name, uf_pk_field
    from ux$field
    where uf_owner = :data_us_uuid and
          uf_is_param = 1 and
          uf_type = 0
    order by uf_position
    into ufs_uf_uuid, ufs_field_name, uf_pk_field
    do
    begin
      ufp_uuid = null;
      ufp_field_name = '';
      ufs_default = '';

      select ufs_ufp_uuid, ufs_default
      from ux$framesql
      where ufs_ufr_uuid = :fr_uuid and
            ufs_sql_kind = :data_kind and
            ufs_uf_uuid = :ufs_uf_uuid
      into ufp_uuid, ufs_default;

      if (row_count = 0) then
      begin
        if (data_us_uuid = frame_us_uuid or data_us_uuid = main_us_uuid) then
        begin
          ufp_field_name = ufs_field_name;

          select ufp_uuid
          from ux$frparam
          where ufp_ufr_uuid = :fr_uuid and
                ufp_uf_uuid = :ufs_uf_uuid
          into ufp_uuid;
        end
      end else
      if (ufp_uuid is not null) then
      begin
        select uf_field_name
        from ux$frparam, ux$field
        where ufp_uuid = :ufp_uuid and
              ufp_uf_uuid = uf_uuid
        into ufp_field_name;

        ufs_default = '';
      end

      is_blob = 0;
      if (ufp_uuid is not null) then
      begin
        select iif(ufp_editor = 'blob', 1, 0)
        from ux$frparam
        where ufp_uuid = :ufp_uuid
        into is_blob;
      end

      if (field_list <> '') then
        field_list = field_list || ', ' || enter_sym;

      if (ufp_field_name <> '') then
      begin
        if (is_blob = 0) then
          field_list = field_list || ':' || ufp_field_name;
        else
          field_list = field_list || f_format(f_temp, ufp_field_name);
      end
      else
      begin
        if (ufs_default <> '') then
          field_list = field_list || ufs_default;
        else
          field_list = field_list || 'null';
      end
    end

    exec_proc = 1;
    if (exists(select * from ux$field
    where uf_owner = :data_us_uuid and
          uf_is_param = 0)) then
      exec_proc = 0;

    if (field_list <> '') then
    begin
      if (exec_proc = 0) then
        sql_text = 'select * from ' || data_source || '(' || field_list || ')';
      else
        sql_text = 'execute procedure ' || data_source || '(' || field_list || ')';
    end
    else
    begin
      sql_kind = case data_kind
        when 1 then 'Update'
        when 2 then 'Insert'
        when 3 then 'Delete'
      end;

      exception e_default f_format('Не настроены параметры процедуры %1, фрейм: %2, SQL: %3',
                                   upper(data_source), fr_code_msg, sql_kind);
    end
  end

  suspend;
end
