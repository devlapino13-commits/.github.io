CREATE PROCEDURE HP_PVNOPERS_REPAYMENT(
    PO_ID D_INTEGER NOT NULL,
    CS_ID D_INTEGER = null)
AS
declare variable acc_dt d_str_20;
declare variable is_ipf d_bool;
declare variable ol_id d_id;
declare variable c_id d_id;
declare variable card_id d_integer;
declare variable k_id d_id;
declare variable amount d_currency;
declare variable ticket_no d_str_code;
declare variable office_id d_integer;
declare variable un_id d_id;
declare variable is_mb_card d_bool;
declare variable card_acc_no d_str_code;
begin
  select po.po_from_c_id,
         po.po_card_id,
         po.po_from_k_id,
         po.po_amount,
         po.po_ticket_no,
         po.po_office_id,
         po.po_un_id,
         po.po_is_mb_card
  from pvnopers po
  where po.po_id = :po_id
  into :c_id,
       :card_id,
       :k_id,
       :amount,
       :ticket_no,
       :office_id,
       :un_id,
       :is_mb_card;

  select iif(pr_code like '%.ipf.%', 1, 0)
  from kontract kont
  inner join product pr on pr.pr_id = kont.k_from_pr_id
  where kont.k_id = :k_id
  into :is_ipf;

  if (is_mb_card = 1) then
  begin
    acc_dt = (
      select al_id
      from acc_list
      where al_from_reg_id = :office_id
        and al_account = '20001/1');
  end
  else
  begin
      acc_dt = (
          select al_id
          from acc_list
          where al_from_reg_id = :office_id
            and al_account = iif(:is_ipf = 1, '10024/102', '10024/2'));
  end

  select
      ol_id
  from  hp_do_repayment (
      :k_id,
      0,                    -- c_id
      current_date,           -- rep_date
      :amount,              -- repay_amount
      1,                    -- repay_to_os
      1,                    -- repay_to_perc
      1,                    -- repay_to_fine
      1,                    -- repay_to_kommis
      1,                    -- need_save
      0,                    -- is_full_repay
      0,                    -- is_writeoff
      :office_id,           -- repay_fact_office_id
      18,                    -- repay_dt_account_kind
      :acc_dt,              -- repay_dt_account_no
      0,                    -- repay_curr_rate
      :un_id,               -- repay_user_id
      :c_id,                -- repay_document_client
      null,   -- repay_doc_client_name_reason
      0,                    -- repay_manager_c_id
      0)                    -- repay_from_rd_id
  into
    :ol_id;

  if (:is_mb_card = 1) then
    select cca.cca_code
    from cardlist cal
       inner join cardclientacc cca on cal.cal_from_cca_id = cca.cca_id
    where cal.cal_card_id = :card_id
    into :card_acc_no;

  update operlist
  set ol_memo = iif(:is_mb_card = 0, f_format('Погашение кредита по ПВН согласно чека № %1 от %2 ', :ticket_no, f_datetostr(current_date)), f_format('Погашение кредита по ПВН: ID %1 (сч.%2) ', :card_id, :card_acc_no)) || ol_memo
  where ol_id = :ol_id;

  if (ol_id is null) then
      exception e_message 'Не удается сохранить платеж. ol_id = null';
    
  update pvnopers po
  set po.po_status = f_enum('pvnopers_status', 'operation_ok', 'id'),
      po.po_from_ol_id = :ol_id
  where po.po_id = :po_id;

  if (:cs_id is not null) then
    insert into cardsyncopers (cso_from_cs_id, cso_from_ol_id, cso_after_exec)
    values (:cs_id, :ol_id,  1);

end
