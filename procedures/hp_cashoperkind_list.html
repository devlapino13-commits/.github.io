CREATE PROCEDURE HP_CASHOPERKIND_LIST
RETURNS (
    ID D_INTEGER,
    CODE D_STR_CODE,
    NAME D_STR_CODE,
    ID_SIGN D_STR_20,
    ACC_LIST_CLS D_STR_LONG,
    ACC_LIST_IPF D_STR_LONG)
AS
declare variable bg_id d_str_20;
declare variable min_acc_no d_integer;
declare variable acc_no_list d_str_long;
declare variable acc_count d_integer;
begin
  for select id, code, name
  from hp_enum_list('cashoperkind')
  where id <> 0
  into :id, :code, :name
  do
  begin
    if (id < 0) then
      id_sign = 'Расход';
    else
      id_sign = 'Приход';

    acc_list_cls = '';
    acc_list_ipf = '';

    for select cak_from_bg_id, min(cak_acc_no), count(*), list(cak_acc_no)
    from cashacckind
    where cak_oper_kind = :id and
          cak_is_ipf = 0
    group by 1
    into bg_id, min_acc_no, acc_count, acc_no_list
    do
    begin
      if (acc_list_cls <> '') then
        acc_list_cls = acc_list_cls || ',  ';

      if (min_acc_no = 0) then
        acc_list_cls = acc_list_cls || bg_id;
      else
      if (acc_count = 1) then
        acc_list_cls = acc_list_cls || bg_id || '/' || min_acc_no;
      else
        acc_list_cls = acc_list_cls || bg_id || '/(' || acc_no_list || ')';

      if (char_length(acc_list_cls) >= 900) then
      begin
        acc_list_cls = acc_list_cls || '...';
        break;
      end
    end

    for select cak_from_bg_id, min(cak_acc_no), count(*), list(cak_acc_no)
    from cashacckind
    where cak_oper_kind = :id and
          cak_is_ipf = 1
    group by 1
    into bg_id, min_acc_no, acc_count, acc_no_list
    do
    begin
      if (acc_list_ipf <> '') then
        acc_list_ipf = acc_list_ipf || ',  ';

      if (min_acc_no = 0) then
        acc_list_ipf = acc_list_ipf || bg_id;
      else
      if (acc_count = 1) then
        acc_list_ipf = acc_list_ipf || bg_id || '/' || min_acc_no;
      else
        acc_list_ipf = acc_list_ipf || bg_id || '/(' || acc_no_list || ')';

      if (char_length(acc_list_ipf) >= 900) then
      begin
        acc_list_ipf = acc_list_ipf || '...';
        break;
      end
    end

    suspend;
  end
end
