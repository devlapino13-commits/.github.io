CREATE PROCEDURE MC_EDOC_STATUS(
    EDS_ID D_ID)
RETURNS (
    RES_CODE D_STR_CODE,
    RES_TEXT D_STR_255,
    EDOC_STATUS D_INTEGER)
AS
declare variable edb_status d_integer;
declare variable edl_status d_integer;
begin
    res_code = 'success';
    edoc_status = -1;
    res_text = '';

    select
        edb_status,
        edl_status
    from edoc_sign
    inner join edoc_list on eds_from_edl_id = edl_id
    inner join edoc_blank on edb_from_edl_id = edl_id
    where
        eds_id = :eds_id
    into
        :edb_status,
        :edl_status;
    
    if (edb_status is null) then
        res_text = 'Документ не найден';
    else if (edl_status < 0) then
        res_text = 'Документ недействителен';
    else if (edb_status < 0) then
        res_text = 'Ошибка приготовления документа';
    else if (edb_status = 0) then
        edoc_status = 0;
    else if (edb_status > 0) then
        edoc_status = 1;

    if (res_text <> '') then
        res_code = 'error';
    
    suspend;
end
