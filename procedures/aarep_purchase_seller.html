CREATE PROCEDURE AAREP_PURCHASE_SELLER(
    OL_ID ID_TYPE)
RETURNS (
    SELLER_NAME D_STR_100,
    SELLER_INN D_STR_20,
    SELLER_PASS_NO VARCHAR(30),
    SELLER_ADDRESS D_STR_255,
    SELLER_REGDOC_NAME D_STR_100,
    SELLER_REGDOC_NO VARCHAR(30),
    SELLER_REGDOC_DATE D_DATE,
    SELLER_TAX_CODE D_STR_255,
    SELLER_TAX_NAME D_STR_255,
    SELLER_ELSOM D_STR_20,
    SELLER_DATE D_DATE,
    SELLER_DOCNO D_INTEGER)
AS
begin
  for
    select
      c_name_fio,
      c_inn,
      c_pass_no,
      adl_address_str,
      pk_name,
      cds_doc_no,
      cds_doc_date_begin,
      rd_code,
      rd_memo,
      ag_elsom_account,
      ol_date,
      ol_doc_no
    from operlist
      inner join clients
        on c_id = ol_document_client
      left join addrclient
        on adc_from_c_id = c_id
          and adc_address_type = 1
      left join addrlist
        on adl_id = adc_from_adl_id
      left join clientdocs
        on cds_id = (
          select first 1 cds_id
          from clientdocs
          where cds_from_c_id = c_id
            and cds_from_pk_id in (44, 68, 71)
            and cds_is_delete = 0
          order by cds_id desc)
      left join passkind
        on pk_id = cds_from_pk_id
      left join refdata
        on rd_parent_id = 3606
          and rd_code = cds_memo_3
      left join kontmat
        on kml_id = (
          select first 1 kml_id
          from kontmat
          where kml_bill_id = :ol_id)
      left join agreements
        on ag_id = kml_from_ag_id
    where ol_id = :ol_id
  into
      :seller_name,
      :seller_inn,
      :seller_pass_no,
      :seller_address,
      :seller_regdoc_name,
      :seller_regdoc_no,
      :seller_regdoc_date,
      :seller_tax_code,
      :seller_tax_name,
      :seller_elsom,
      :seller_date,
      :seller_docno
  do suspend;
end
