CREATE PROCEDURE FM_HTML_CLIENT_FT_ANKETA_SELLER(
    C_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_8100)
AS
declare variable code d_str_code;
declare variable signed d_integer;
declare variable lang d_str_lang;
begin
  res_text = '';
  code = 'ft_anketa_seller';

  if (exists(
    select *
    from hp_vislink_send_seller_anketa(:c_id, :un_id)
    where is_visible = 1)) then
  begin
    if (exists(
      select edoc_status from hp_edoc_status(:code, 0, :c_id)
      where client_sign_status = 1))
    then signed = 1; else signed = 0;

    lang = f_get_lang_code(:un_id, null);
    res_text = '<h2>' || f_lang(:lang, 'Анкета ПФТД и Гендоговор') || '</h2>';
    res_text = res_text || f_uri_sql(
      f_lang(:lang, 'Отправить ссылку на подпись'),
      :code,
      'client',
      :c_id,
      'none',
      f_lang(lang, iif(:signed = 1, 'Анкета ранее уже была подписана. ', '') ||
        'Отправить ссылку на подписание через СМС?'),
      '',
      0,
      0,
      '',
      '',
      'dialog',
      '');
  end
  if (res_text <> '') then
    res_text = res_text || '<br><br>';
  suspend;
end
