CREATE PROCEDURE HP_E_HR_ORDERS_SIGNER_LOOKUP(
    IN_OFFICE_ID D_INTEGER,
    IN_DATE D_DATE)
RETURNS (
    ID D_INTEGER,
    TEXT D_STR_255)
AS
declare variable is_hq d_integer;
declare variable db_id d_integer;
declare variable ksm_id d_integer;
declare variable c_id d_id;
begin
  id = null;
  text = '-';
  suspend;

  db_id = f_get_db_id();

  is_hq = 0;
  if ((db_id = 41701 and in_office_id = 1090100) or
      (db_id = 39801 and in_office_id = 1010100) or
      (db_id not in (39801, 41701) and in_office_id = 1010100)) then
    is_hq = 1;

  if (is_hq = 1) then
  begin
    for select ksm_id, ksm_from_c_id
    from kontsignmen
    where ksm_hr_hq > 0
    order by ksm_hr_hq
    into ksm_id, c_id
    do
    begin
      id = null;
      select ksr_id
      from kontsignreg, region_list(:in_office_id, 1, -1, 0)
      where ksr_from_ksm_id = :ksm_id and
            ksr_from_reg_id = id_list and
            coalesce(ksr_date_begin, :in_date) <= :in_date and
            coalesce(ksr_date_end, :in_date) >= :in_date
      order by ksr_is_active desc, ksr_id desc
      rows 1
      into id;

      if (id is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :c_id
        into text;

        if (text = 'Исламов Санжар Якубжанович') then
          Continue;

        suspend;
      end
    end
  end else
  begin
    for select ksm_id, ksm_from_c_id
    from kontsignmen
    where ksm_hr_branch > 0
    order by ksm_hr_branch
    into ksm_id, c_id
    do
    begin
      id = null;
      select ksr_id
      from kontsignreg, region_list(:in_office_id, 1, -1, 0)
      where ksr_from_ksm_id = :ksm_id and
            ksr_from_reg_id = id_list and
            coalesce(ksr_date_begin, :in_date) <= :in_date and
            coalesce(ksr_date_end, :in_date) >= :in_date
      order by ksr_is_active desc, ksr_id desc
      rows 1
      into id;

      if (id is not null) then
      begin
        select c_name_fio
        from clients
        where c_id = :c_id
        into text;

        if (text = 'Исламов Санжар Якубжанович') then
          Continue;

        suspend;
      end
    end
  end

end
