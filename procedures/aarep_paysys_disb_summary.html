CREATE PROCEDURE AAREP_PAYSYS_DISB_SUMMARY(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REP_REGION_NAME VARCHAR(100),
    REP_AMOUNT_APPROVED DECIMAL(18,4),
    REP_POLIS_AMOUNT DECIMAL(18,4),
    REP_LO_FIO VARCHAR(100),
    REP_OKRUG_NAME VARCHAR(100),
    REP_POLIS_KOLICH INTEGER,
    REP_DISB_CLIENT_KOLICH INTEGER,
    REP_POLIS_PERCENT DECIMAL(18,4))
AS
begin
  for select rep_okrug_name, rep_region_name, rep_lo_fio, sum(rep_amount_approved), sum(rep_polis_amount),
             sum(rep_polis_kolich), sum(rep_disb_client_kolich)
      from aarep_paysys_disb_report(:rep_region_id, :rep_date_begin, :rep_date_end, 0)
      group by rep_okrug_name, rep_region_name, rep_lo_fio
      into :rep_okrug_name, :rep_region_name, rep_lo_fio, rep_amount_approved, :rep_polis_amount, :rep_polis_kolich,
           :rep_disb_client_kolich
  do
  begin
    if (rep_disb_client_kolich <> 0) then
      rep_polis_percent = cast(rep_polis_kolich as float) / rep_disb_client_kolich * 100;
    else
      rep_polis_percent = 0;
    suspend;
  end
end
