CREATE PROCEDURE FM_USER_REGIONS(
    UN_ID D_ID)
RETURNS (
    REG_ID D_INTEGER,
    REG_NAME D_STR_100)
AS
declare variable is_office d_bool;
begin
  -- Процедура возвращает список офисов, доступных пользователю
  -- Используется для фильтра по региону в отчетах МП
  -- RFCRU-4679

  if (un_id is null) then
    exit;

  select un_id
  from hp_test_un_id(:un_id)
  into :un_id;

  for select distinct reg_list, reg_code || '  ' || reg_name,
                      iif(reg_status = 3, 1, 0)
      from aarep_region_user_tmp(:un_id), region
      where reg_unicode = reg_list and
            reg_code <> reg_name and -- Офисы без названия, н-р 0127 не кредитуют
            coalesce(reg_date_close, current_date) >= current_date -- Закрытые не возвращаем
      order by reg_code, reg_unicode
      into reg_id, :reg_name, is_office
  do
  begin
    if (is_office = 0) then
    begin
      -- Не возвращаем родителей (узлы), если все подчиненные офисы закрыты
      if (not exists(select *
                     from region
                     where parent = :reg_id and
                           reg_status = 3 and
                           coalesce(reg_date_close, current_date) >= current_date)) then
      begin
        -- Проверка второго уровня вложенности
        if (not exists(select *
                       from region r1, region r2
                       where r1.parent = :reg_id and
                             r1.reg_unicode = r2.parent and
                             r2.reg_status = 3 and
                             coalesce(r2.reg_date_close, current_date) >= current_date)) then
          continue;
      end
    end

    suspend;
  end
end
