CREATE PROCEDURE FM_KONTRACT_TO_MINIAPP(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_LONG,
    RES_NEED_ROLLBACK D_SMALLINT)
AS
begin
  -- в МП возвращает договор в статус заявки

  select res_text, res_need_rollback
  from hp_kontract_set_status(:k_id, -2, 0, :un_id)
  into :res_text, :res_need_rollback;

  suspend;
end
