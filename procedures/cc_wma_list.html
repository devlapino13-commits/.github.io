CREATE PROCEDURE CC_WMA_LIST(
    UN_ID ID_TYPE,
    REP_KIND STR_100,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE = null,
    REP_PHONE_NO STR_100 = null,
    REP_FIO STR_100 = null)
RETURNS (
    WMA_ID INTEGER,
    WMA_DATE DATE,
    FIO STR_100,
    CITIZENSHIP STR_100,
    LANGUAGE STR_100,
    WMA_CANT_CALL_QTY INTEGER,
    LAST_CANT_CALL_DATETIME TIMESTAMP,
    BLOCKED_USER STR_100,
    PHONE_NO STR_100,
    CALLBACK_DATETIME TIMESTAMP,
    LAST_USER STR_100,
    MEMO D_STR_8100,
    CREATE_DATETIME STR_100,
    CITY STR_100,
    APP_STATUS STR_100,
    DETAILS_READ_ONLY SMALLINT,
    DELINQ_CALL SMALLINT,
    WMA_SEND_TO_OFFICE_DATETIME STR_100,
    WMAT_OFFICE STR_100,
    EMA_LO_FIO STR_100,
    KFOLDER_NAME STR_100,
    EMA_CURRENT_STATUS STR_255,
    REFUSE_REASON STR_255)
AS
declare variable wma_status smallint;
declare variable blocked_un_id id_type;
declare variable tkt_id id_type;
declare variable ema_status smallint;
declare variable svz_refuse str_255;
declare variable ema_id id_type;
declare variable ma_id integer;
declare variable ma_refuse_reason str_255;
declare variable ma_date_oper date;
declare variable ma_krl_id integer;
declare variable k_id id_type;
declare variable k_refuse_reason str_255;
declare variable k_krl_id integer;
declare variable ks_state integer;
declare variable sql_text d_blob_text;
declare variable param str_100;
begin
  if (rep_phone_no = '') then
    rep_phone_no = null;

  if (rep_fio = '') then
    rep_fio = null;

  if ((rep_kind = 'SearchByPhone') and
      (rep_phone_no is null)) then
    exit;

  if (rep_kind = 'new') then
  begin
    details_read_only = 0;
    for select wma_id,
               wma_date,
               iif(wmat_name = '', '[Анкета в процессе заполнения]', wmat_name),
               rd_c.rd_name as citizenship,
               rd_l.rd_name as language,
               wma_cant_call_qty,
               wma_phone_no,
               wma_last_cant_call_datetime,
               --f_datetimetostr(wma_last_cant_call_datetime),
               coalesce(un.un_name, ''),
               f_datetimetostr(wma_create_datetime),
               wmat_city
        from cc_website_miniapp wma
        inner join cc_website_miniapp_tmp wmt on wmt.wmat_id = wma.wma_last_wmat_id
        left join refdata rd_c on rd_c.rd_id = wmt.wmat_citizenship
        left join refdata rd_l on rd_l.rd_id = wmat_language
        left join username un on wma_blocked_un_id = un.un_id
        where wma_date between :rep_date_begin and :rep_date_end and
              wma_status = 0
        order by wma_id
        into :wma_id,
             :wma_date,
             :fio,
             :citizenship,
             :language,
             :wma_cant_call_qty,
             :phone_no,
             :last_cant_call_datetime,
             :blocked_user,
             :create_datetime,
             :city
    do
    begin
      suspend;
    end
  end
  if (rep_kind = 'callBack') then
  begin
    details_read_only = 0;
    for select wma_id,
               --f_datetimetostr(wmt.wmat_call_back_date),
               wmt.wmat_call_back_date,
               wma_date,
               wmat_name,
               rd_c.rd_name as citizenship,
               rd_l.rd_name as language,
               wma_phone_no,
               coalesce(un_block.un_name, ''),
               coalesce(un_last.un_name, ''),
               wma_cant_call_qty,
               wma_last_cant_call_datetime,
               --f_datetimetostr(wma_last_cant_call_datetime),
               coalesce(wmt.wmat_anketa_memo, ''),
               f_datetimetostr(wma_create_datetime),
               wmat_city,
               iif(wmat_call_back_date < current_timestamp, 1, 0)
        from cc_website_miniapp wma
        inner join cc_website_miniapp_tmp wmt on wmt.wmat_id = wma_last_wmat_id
        left join refdata rd_c on rd_c.rd_id = wmt.wmat_citizenship
        left join refdata rd_l on rd_l.rd_id = wmat_language
        left join username un_block on un_block.un_id = wma_blocked_un_id
        left join username un_last on un_last.un_id = wma_last_un_id
        where wma_status = 1
        order by wma_id
        into :wma_id,
             :callback_datetime,
             :wma_date,
             :fio,
             :citizenship,
             :language,
             :phone_no,
             :blocked_user,
             :last_user,
             :wma_cant_call_qty,
             :last_cant_call_datetime,
             :memo,
             :create_datetime,
             :city,
             :delinq_call
    do
    begin
      suspend;
    end
  end
  if (rep_kind = 'SearchByPhone') then
  begin
    -- f_datetimetostr(wmt.wmat_call_back_date),
    -- f_datetimetostr(wma_last_cant_call_datetime),

    sql_text = 'select wma_id,
               wmt.wmat_call_back_date,
               wma_date,
               wmat_name,
               rd_c.rd_name as citizenship,
               rd_l.rd_name as language,
               wma_phone_no,
               coalesce(un_block.un_name, ''''),
               coalesce(un_last.un_name, ''''),
               wma_cant_call_qty,
               wma_last_cant_call_datetime,
               coalesce(wmt.wmat_anketa_memo, ''''),
               f_datetimetostr(wma_create_datetime),
               wmat_city,
               (case wma_status
                    when -1 then ''Закрыта. Негатив. статус''
                    when 0  then ''Новая''
                    when 1  then ''Перезвонить''
                    when 2  then ''Передана в офис''
               end),
               wma_status,
               wma_blocked_un_id
        from cc_website_miniapp wma
        inner join cc_website_miniapp_tmp wmt on wmt.wmat_id = wma_last_wmat_id
        left join refdata rd_c on rd_c.rd_id = wmt.wmat_citizenship
        left join refdata rd_l on rd_l.rd_id = wmat_language
        left join username un_block on un_block.un_id = wma_blocked_un_id
        left join username un_last on un_last.un_id = wma_last_un_id';
        if (rep_phone_no is not null) then
        begin
            param = :rep_phone_no;
            sql_text = blob_append(sql_text, ' where wma.wma_phone_no = :param ');
        end
        else if (rep_fio is not null) then
        begin
            param = :rep_fio;
            sql_text = blob_append(sql_text, ' where wmat_name = :param ');
        end
        sql_text = blob_append(sql_text, ' order by wma_id');

        for execute statement(:sql_text)(param := :param)
        into :wma_id,
             :callback_datetime,
             :wma_date,
             :fio,
             :citizenship,
             :language,
             :phone_no,
             :blocked_user,
             :last_user,
             :wma_cant_call_qty,
             :last_cant_call_datetime,
             :memo,
             :create_datetime,
             :city,
             :app_status,
             :wma_status,
             :blocked_un_id
    do
    begin
      details_read_only = (case wma_status
                                when -1 then 1
                                when 0  then 0
                                when 1  then 0
                                when 2  then 1
                           end);

      if (blocked_un_id <> un_id) then -- если заявка заблокирована за другим обператором, тогда всегда открытие только для чтения.
        details_read_only = 1;
      suspend;
    end
  end
  if (rep_kind = 'SendedToOffice') then
  begin
    for select wma_id,
               wma_date,
               wmat_name,
               rd_c.rd_name as citizenship,
               rd_l.rd_name as language,
               wma_phone_no,
               f_datetimetostr(wma.wma_send_to_office_datetime),
               reg.reg_name,
               wma.wma_from_tkt_id,
               f_datetimetostr(wma_create_datetime)
        from cc_website_miniapp wma
        inner join cc_website_miniapp_tmp wmt on wmt.wmat_id = wma_last_wmat_id
        inner join region reg on reg.reg_unicode = wmt.wmat_office
        left join refdata rd_c on rd_c.rd_id = wmt.wmat_citizenship
        left join refdata rd_l on rd_l.rd_id = wmat_language
        where wma.wma_status = 2
        and wma.wma_date between :rep_date_begin and :rep_date_end
        order by wma_id
        into :wma_id,
             :wma_date,
             :fio,
             :citizenship,
             :language,
             :phone_no,
             :wma_send_to_office_datetime,
             :wmat_office,
             :tkt_id,
             :create_datetime
    do
    begin
        ema_lo_fio = null;
        ema_status = null;
        svz_refuse = null;
        ema_id = null;

        select cl.c_name_fio,
               ema.ema_status,
               rd_refuse.rd_name,
               ema.ema_id
        from extminiapp ema
        left join refdata rd_refuse on rd_refuse.rd_id = ema.ema_result_rd_id
        left join clients cl on cl.c_id = ema.ema_lo_id
        where ema.ema_from_tkt_id = :tkt_id
        into :ema_lo_fio,
             :ema_status,
             :svz_refuse,
             :ema_id;

        ma_id = null;
        kfolder_name= null;
        ma_refuse_reason = null;
        ma_date_oper= null;
        ma_krl_id = null;

        for select maa.ma_id
        from miniapp maa
        where maa.ma_from_tkt_id = :tkt_id
        rows 1
        into :ma_id
        do
        begin
            select map.ma_id,
                   kal.kal_new_kf_name,
                   krs.kr_text,
                   map.ma_date_oper,
                   krl.krl_id
            from miniapp map
            inner join knaktlist  kal on kal.kal_from_kf_id = map.ma_from_kf_id
            left join kontreslist krl on krl.krl_ma_id = map.ma_id
            left join kontres     krs on krs.kr_id = krl.krl_from_kr_id
            where map.ma_id = :ma_id
            and map.ma_date_end = '31.12.9999'
            and current_date between kal.kal_date and kal.kal_date_end
            and kal.is_delete = 0
            rows 1
            into
                :ma_id,
                :kfolder_name,
                :ma_refuse_reason,
                :ma_date_oper,
                :ma_krl_id;
        end

        k_id = null;
        k_refuse_reason = null;
        k_krl_id = null;
        ks_state = null;

        select kont.k_id,
               krs.kr_text,
               krl.krl_id,
               ks.ks_state
        from kontract kont
        left join kontreslist krl on krl.krl_from_k_id = kont.k_id
        left join kontres     krs on krs.kr_id = krl.krl_from_kr_id
        left join kontstate   ks  on ks.ks_from_k_id = kont.k_id
        left join operlist    ol  on ol.ol_id = kont.k_fact_disb_ol_id
            and ol.is_delete = 0
        where kont.k_from_ma_app_id = :ma_id
        and current_date between ks.ks_date_begin and ks.ks_date_end
        and ks.ks_ol_is_delete = 0
        rows 1
        into :k_id,
             :k_refuse_reason,
             :k_krl_id,
             :ks_state;

        ema_current_status = '';
        refuse_reason = '';

        if (k_id is not null) then
        begin
            if (ks_state in (1,2)) then
                ema_current_status = 'Заем выдан';
            else if (k_krl_id is not null) then
            begin
                ema_current_status = 'Отказ на уровне кредитного договора';
                refuse_reason = k_refuse_reason;
            end
            else
                ema_current_status = 'Кредитный договор в процессе';
        end
        else if (ma_id is not null) then
        begin
            if (ma_krl_id is not null) then
            begin
                ema_current_status = 'Отказ на уровне кредитной заявки';
                refuse_reason = ma_refuse_reason;
            end
            else
                ema_current_status = 'Кредитная заявка';
        end
        else if (ema_id is not null) then
        begin
           if (ema_status = 0) then
           begin
                if (ema_lo_fio is null) then
                    ema_current_status = 'Заявка на уровне FsMobile. Не заблокирована ВИПС';
                else
                    ema_current_status = 'Заявка на уровне FsMobile. Заблокирвоана ВИПС';
           end
           else if (ema_status = 1) then
           begin
                ema_current_status = 'Заявка на уровне FsMobile. Обработана ВИПС';
           end
            else if (ema_status = -1) then
            begin
                refuse_reason = svz_refuse;
                ema_current_status = 'Отказ на уровне FsMobile со стороны ВИПС';
            end
        end
        else
        begin
            ema_current_status = 'Заявка с КЦ передана в офис';
        end
        suspend;
    end    
  end
end
