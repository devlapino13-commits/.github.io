CREATE PROCEDURE HP_BLANKBLOB_JSON_UPDATE(
    BLN_ID D_INTEGER,
    BLB_NUMBER D_INTEGER,
    JSON_BLOB D_BLOB_TEXT,
    FIELD_LIST D_BLOB_TEXT,
    UN_ID D_ID)
AS
declare variable table_count d_integer;
declare variable table_id d_integer;
declare variable table_item d_blob_text;
declare variable table_name d_str_255;
declare variable field_count d_integer;
declare variable field_id d_integer;
declare variable field_name d_str_255;
declare variable bjf_id d_integer;
begin
  update blankblob
  set blb_json_blob = :json_blob,
      blb_from_un_id = :un_id
  where blb_from_bln_id = :bln_id and
        blb_number = :blb_number;


  delete from blankjsonlink
  where bjl_from_bln_id = :bln_id and
        bjl_from_blb_number = :blb_number;

  table_count = 0;
  table_count = f_json_blob_get_count(field_list, 'tables');
  if (table_count > 0) then
  begin
    table_id = 0;
    while (table_id < table_count) do
    begin
      table_item = f_json_blob_get_value(field_list, 'tables['||table_id||']', '');
      if (table_item <> '') then
      begin
        table_name = f_json_blob_get_value(table_item, 'name', '');
        field_count = f_json_blob_get_count(table_item, 'fields');

        if (field_count > 0) then
        begin
          field_id = 0;
          while (field_id < field_count) do
          begin
            field_name = f_json_blob_get_value(table_item, 'fields['||field_id||']', '');

            if (field_name <> '') then
            begin
              update or insert into blankjsonfields (bjf_table_name, bjf_field_name)
              values (:table_name, :field_name)
              matching (bjf_table_name, bjf_field_name)
              returning bjf_id
              into bjf_id;

              insert into blankjsonlink (bjl_from_bln_id, bjl_from_blb_number, bjl_from_bjf_id)
              values (:bln_id, :blb_number, :bjf_id);
            end

            field_id = field_id + 1;
          end
        end
      end
      table_id = table_id + 1;
    end
  end
end
