CREATE PROCEDURE INVMAIN_PRINT(
    PAR1 DECIMAL(18,0))
RETURNS (
    IM_ID DECIMAL(18,0),
    IM_DATE DATE,
    IM_ORDER_DATE DATE,
    IM_ORDER_NO VARCHAR(20),
    IM_DATE_INV DATE,
    IM_WORKPLACE_ID DECIMAL(18,0),
    IM_STATUS INTEGER,
    IM_DATE_APPROVED DATE,
    IM_PRESIDENT DECIMAL(18,0),
    IM_NO INTEGER,
    IM_CATEG_ID INTEGER,
    IM_PRESIDENT_FIO VARCHAR(100),
    IM_WORKPLACE_NAME VARCHAR(100),
    UN_NAME VARCHAR(30),
    IM_CATEG_NAME VARCHAR(100),
    PRESIDENT_LOGIN D_STR_64)
AS
declare variable wp_name2 varchar(100);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select im_id, im_date, im_order_date, im_order_no, im_date_inv, im_workplace_id, im_status, im_date_approved,
             im_president, c_name_fio, un_name, wp_name, im_no, im_categ_id, reg_name
      from invmain
      left join clients on (im_president = c_id)
      left join username on (im_user_id = un_id)
      left join workplace on (im_workplace_id = wp_id)
      left join region on (im_workplace_id = reg_unicode)
      where im_id = :par1
      into :im_id, :im_date, :im_order_date, :im_order_no, :im_date_inv, :im_workplace_id, :im_status,
           :im_date_approved, :im_president, :im_president_fio, :un_name, :im_workplace_name, :im_no, :im_categ_id,
           :wp_name2
  do
  begin
    if (im_workplace_name is null) then
    begin
      im_workplace_name = wp_name2;
    end
    if (im_categ_id = -1) then
      im_categ_name = 'Все ТМЦ';
    if (im_categ_id = 0) then
      im_categ_name = 'Основные средства';
    if (im_categ_id = 1) then
      im_categ_name = 'МБП';
    if (im_categ_id = 2) then
      im_categ_name = 'Немат.активы';
    if (im_categ_id = 3) then
      im_categ_name = 'Расход.материалы';

    president_login = '';
    if (im_status = 1) then
      president_login = (select un_name from username where un_from_c_id = :im_president);
    suspend;
  end
end
