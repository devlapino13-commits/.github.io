CREATE PROCEDURE HP_EDOC_SIGN_LOCK(
    EDS_ID D_ID,
    UN_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_255,
    C_PINCODE D_STR_CODE,
    ESL_CREATE_DATETIME D_DATETIME,
    ESL_EXPIRE_DATETIME D_DATETIME)
AS
declare variable eds_status d_smallint;
declare variable edl_id d_id;
declare variable c_name_fio d_str_100;
declare variable c_id d_id;
begin
  -- Блокировка документа для подписания клиентом/пользователем в рамках интеграции с sima.az
  -- Проверки перед блокировкой
  error_text = '';

  -- Ищем подписанта
  if (error_text = '') then
  begin
    select eds_from_edl_id, eds_status, c_id, c_pincode
    from edoc_sign
      left join clients on eds_from_c_id = c_id
    where eds_id = :eds_id
    into edl_id, eds_status, c_id, c_pincode;

    if (edl_id is null) then
      error_text = f_format('Подписант не найден, id: %1', coalesce(eds_id, 'NULL'));

    if (error_text = '' and eds_status = 1) then
      error_text = 'Документ был ранее подписан клиентом.';

    -- У контролера eds_from_c_id is null, PIN проверяем для клиентов
    if (error_text = '' and c_id is not null and coalesce(c_pincode, '') = '') then
    error_text = 'У клиента не заполнен pin';
  end

  -- Контрольно снимаем устаревшие блокировки
  execute procedure hp_edoc_sign_unlock(null);

  -- Если у текущего подписанта есть действительная блокировка, возвращаем ее
  select esl_create_datetime, esl_expire_datetime
  from edoc_sign_lock
  where esl_from_eds_id = :eds_id
  into esl_create_datetime, esl_expire_datetime;
  if (esl_create_datetime is not null) then
  begin
    suspend;
    exit;
  end


  if (error_text = '') then
  begin
      -- проверить на eds_from_edc_id is null
    c_name_fio = null;
    select coalesce(c_name_fio, 'Контролер')
    from edoc_sign_lock
      left join edoc_sign on esl_from_eds_id = eds_id
      left join clients on eds_from_c_id = c_id
    where esl_from_edl_id = :edl_id
    into c_name_fio;
    if (c_name_fio is not null) then
      error_text = f_format('Договор сейчас подписывается: %1', c_name_fio);
  end

  if (error_text = '') then
  begin
    insert into edoc_sign_lock (
      esl_from_edl_id, esl_from_eds_id, esl_from_un_id)
    values (:edl_id, :eds_id, :un_id)
    returning esl_create_datetime, esl_expire_datetime
    into esl_create_datetime, esl_expire_datetime;
    when gdscode unique_key_violation do
      -- Не смогли вставить из-за того что запись уже есть
      error_text = 'Ошибка получения блокировки. Попробуйте еще раз через 10-15 секунд';
  end

  if (error_text = '') then
  begin
    -- Сохраняем список бланков, которые будет подписывать клиент (подписант)
    insert into edoc_blank_taken (
      ebt_from_edb_id, ebt_from_eds_id, ebt_from_edl_id)
    select edb_id, eds_id, eds_from_edl_id
    from edoc_sign, edoc_blank
    where eds_id = :eds_id and
          edb_from_edl_id = eds_from_edl_id and
          (edb_from_c_id is null or edb_from_c_id = eds_from_c_id); -- выдаем общие бланки или конкретного участника
  end
  suspend;
end
