CREATE PROCEDURE HP_PERSTEMP_TEST_CHECK(
    C_ID D_ID)
RETURNS (
    RES_CHECK D_BOOL)
AS
begin
  res_check = 1;
  if (res_check = 1 and exists(select * from perslist
             where pls_from_c_id = :c_id)) then
    res_check = 0;

  if (res_check = 1 and exists(select * from perstemp
             where pte_from_c_id = :c_id)) then
    res_check = 0;

  if (res_check = 1 and exists(select * from actionlist
             where acl_object_kind = 0 and
                   acl_object_id = :c_id and
                   acl_parent_id is null and
                   acl_date = current_date)) then
    res_check = 0;

  suspend;
end
