CREATE PROCEDURE UXP$PERSFIELD_PREPARE_VIEW(
    TABLE_NAME D_STR_CODE)
RETURNS (
    RES_IS_ERROR D_BOOL,
    RES_SCRIPT D_STR_8100)
AS
declare variable field_name d_str_code;
declare variable is_pers_field d_bool;
declare variable return_fields d_str_8100;
declare variable select_fields d_str_8100;
declare variable key_field d_str_code;
declare variable key_count d_integer;
declare variable pif_mask_kind d_str_code;
declare variable pif_mask_add_field d_str_code;
declare variable pif_link_source d_str_code;
begin
  res_is_error = 0;
  res_script = '';
  return_fields = '';
  select_fields = '';

  /*
create or alter view v$contlist(
    clt_id,
    clt_code)
as
select
    clt_id,
    clt_code
from contlist
;
  */

  table_name = trim(lower(table_name));
  key_field = '';
  key_count = 0;

  if (not exists(select * from ux$persfield where pif_source = :table_name)) then
    res_script = 'Нет полей в таблице зарегистрированные как персональные данные ' || upper(table_name);
  else
  begin
    select key_fields, key_count
    from uxp$uxfield_db_keyfields(:table_name, 0)
    into key_field, key_count;
  
    if (key_field = '') then
      res_script = 'Не найдено ключевое поле таблицы: ' || upper(table_name);
    else
    if (key_count > 1) then
      res_script = 'Таблицы с составными первичными ключами не поддерживаются: ' || upper(table_name);
  end

  if (res_script <> '') then
  begin
    res_is_error = 1;
    suspend;
    exit;
  end

  for select lower(trim(r.rdb$field_name))
  from rdb$relation_fields r
  where r.rdb$relation_name = upper(:table_name)
  order by r.rdb$field_position
  into field_name
  do
  begin
    is_pers_field = 0;
    for select pif_mask_kind, pif_mask_add_field, pif_link_source
    from ux$persfield
    where pif_source = :table_name and
          pif_field_name = :field_name
    into pif_mask_kind, pif_mask_add_field, pif_link_source
    do
    begin
      is_pers_field = 1;

      if (return_fields <> '') then
        return_fields = return_fields || ',' || ascii_char(13) || '    ';
      else
        return_fields = '    ';

      return_fields = return_fields || field_name;
      return_fields = return_fields || ',' || ascii_char(13) || '    ';
      return_fields = return_fields || 'orig$' || field_name;
      return_fields = return_fields || ',' || ascii_char(13) || '    ';
      return_fields = return_fields || 'mask$' || field_name;

      if (select_fields <> '') then
        select_fields = select_fields || ',' || ascii_char(13) || '    ';
      else
        select_fields = '    ';

      select_fields = select_fields || field_name;
      select_fields = select_fields || ',' || ascii_char(13) || '    ';
      select_fields = select_fields || 'cast(pif$value(''' || table_name || ''', ''' || field_name || ''', ' || key_field || ', ' || field_name || ')';
      select_fields = select_fields || ' as type of column ' || table_name || '.' || field_name || ')';
      select_fields = select_fields || ',' || ascii_char(13) || '    ';

      if (pif_mask_kind = '' and pif_link_source = '') then
        pif_mask_kind = 'text';

      if (pif_mask_kind <> '') then
      begin
        if (pif_mask_add_field <> '') then
          select_fields = select_fields || 'pif$mask(' || field_name || ', ''' || pif_mask_kind || ''', ' || pif_mask_add_field || ')';
        else
          select_fields = select_fields || 'pif$mask(' || field_name || ', ''' || pif_mask_kind || ''')';
      end else
      begin
        if (pif_link_source <> '') then
          select_fields = select_fields || field_name;
        else
        begin
          if (pif_mask_add_field <> '') then
            select_fields = select_fields || '''Id_''' || pif_mask_add_field;
          else
            select_fields = select_fields || '''Id_''' || field_name;
        end
      end
    end

    if (is_pers_field = 0) then
    begin
      if (return_fields <> '') then
        return_fields = return_fields || ',' || ascii_char(13) || '    ';
      else
        return_fields = '    ';

      return_fields = return_fields || field_name;

      if (select_fields <> '') then
        select_fields = select_fields || ',' || ascii_char(13) || '    ';
      else
        select_fields = '    ';

      select_fields = select_fields || field_name;
    end
  end

  res_script = 'create or alter view v$' || lower(table_name) || '(' || ascii_char(13) || return_fields || ')' || ascii_char(13) ||
               'as' || ascii_char(13) ||
               'select' || ascii_char(13) ||
               select_fields || ascii_char(13) ||
               'from ' || table_name || ascii_char(13) ||
               ';' || ascii_char(13) || ascii_char(13);

  suspend;
end
