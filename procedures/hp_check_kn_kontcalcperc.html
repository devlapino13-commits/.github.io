CREATE PROCEDURE HP_CHECK_KN_KONTCALCPERC
RETURNS (
    K_BASE_K_ID ID_TYPE,
    RES_K_ID ID_TYPE,
    RES_KC_ID ID_TYPE,
    RES_FACT_CLOSE_PERIOD INTEGER,
    RES_LAST_PERIOD INTEGER,
    KCP_PERC_SALDO CURR_TYPE,
    KCP_HAS_SALDO_AFTER_CLOSE INTEGER,
    KCP_HAS_OBOROT_AFTER_CLOSE INTEGER)
AS
begin
  for select distinct kd_k_orig
      from kontdevides
      into k_base_k_id
  do
  begin
    for select res_k_id
        from hp_get_linked_kontract_list(:k_base_k_id, 1)
        into :res_k_id
    do
    begin
      res_kc_id = 0;
      res_fact_close_period = null;

      select extract(year from ks_date_begin) * 100 + extract(month from ks_date_begin)
      from kontstate
      where ks_from_k_id = :res_k_id and
            ks_ol_is_delete = 0 and
            ks_state = 2 and
            ks_date_begin <= '31.07.2018'
      into :res_fact_close_period;

      res_last_period = null;
      kcp_has_saldo_after_close = 0;
      kcp_has_oborot_after_close = 0;

      select kcp_period, kcp_perc_saldo
      from kontcalcperc
      where kcp_from_k_id = :res_k_id
      order by kcp_from_k_id desc, kcp_period desc
      rows 1
      into res_last_period, kcp_perc_saldo;
      if (kcp_perc_saldo is null) then
        kcp_perc_saldo = 0;

      if (res_fact_close_period is not null) then
      begin
        if (exists(select *
                   from kontcalcperc
                   where kcp_from_k_id = :res_k_id and
                         kcp_period > :res_fact_close_period and
                         kcp_perc_saldo <> 0)) then
          kcp_has_saldo_after_close = 1;

        if (exists(select *
                   from kontcalcperc
                   where kcp_from_k_id = :res_k_id and
                         kcp_period > :res_fact_close_period and
                         (kcp_perc_calced <> 0 or
                         kcp_perc_payed <> 0 or
                         kcp_perc_writeoff <> 0))) then
          kcp_has_oborot_after_close = 1;
      end

      if (coalesce(res_fact_close_period, 201807) <> res_last_period or (res_fact_close_period is not null and
          kcp_perc_saldo <> 0)) then
        suspend;

      for select kc_id
          from knclient
          where kc_from_k_id = :res_k_id and
                kc_amount_approved > 0
          into res_kc_id
      do
      begin
        res_last_period = null;
        kcp_has_saldo_after_close = 0;
        kcp_has_oborot_after_close = 0;

        select kcp_period, kcp_perc_saldo
        from kn_kontcalcperc
        where kcp_from_kc_id = :res_kc_id
        order by kcp_from_kc_id desc, kcp_period desc
        rows 1
        into res_last_period, kcp_perc_saldo;
        if (kcp_perc_saldo is null) then
          kcp_perc_saldo = 0;

        if (res_fact_close_period is not null) then
        begin
          if (exists(select *
                     from kn_kontcalcperc
                     where kcp_from_k_id = :res_kc_id and
                           kcp_period > :res_fact_close_period and
                           kcp_perc_saldo <> 0)) then
            kcp_has_saldo_after_close = 1;

          if (exists(select *
                     from kn_kontcalcperc
                     where kcp_from_k_id = :res_kc_id and
                           kcp_period > :res_fact_close_period and
                           (kcp_perc_calced <> 0 or
                           kcp_perc_payed <> 0 or
                           kcp_perc_writeoff <> 0))) then
            kcp_has_oborot_after_close = 1;
        end

        if (coalesce(res_fact_close_period, 201807) <> res_last_period or (res_fact_close_period is not null and
            kcp_perc_saldo <> 0)) then
          suspend;
      end
    end
  end
end
