CREATE PROCEDURE AAREP_ACCOUNT_BANK_SALDO_TMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    LINE_NAME D_STR_255,
    SALDO_HQ_417 D_CURRENCY,
    SALDO_HQ_840 D_CURRENCY,
    SALDO_HQ_978 D_CURRENCY,
    SALDO_HQ_OTHER D_CURRENCY,
    SALDO_BRU D_CURRENCY,
    CURRENCY VARCHAR(3),
    SALDO_ISK D_CURRENCY,
    SALDO_OSH D_CURRENCY,
    SALDO_JAL D_CURRENCY,
    SALDO_NAR D_CURRENCY,
    SALDO_TAL D_CURRENCY,
    SALDO_KAD D_CURRENCY,
    SALDO_BAT D_CURRENCY,
    SALDO_UZG D_CURRENCY,
    SALDO_TASH D_CURRENCY,
    SALDO_CHUY D_CURRENCY,
    SALDO_BISH D_CURRENCY,
    SALDO_OTHER D_CURRENCY,
    SALDO_REGION D_CURRENCY,
    CR_RATE D_CURRENCY,
    BG_ID D_STR_20)
AS
declare variable account d_str_20;
declare variable summa_ekv d_currency;
declare variable obs_saldo_val d_currency;
declare variable al_memo d_str_255;
declare variable reg_id d_integer;
declare variable region_id id_type;
begin
  -- вычисляет остаток по денежных средств в разбивке по округам.
  -- работает только для КР
  for select al_id, al_from_bg_id, al_currency, al_from_reg_id, coalesce(al_memo, ''), op.cr_rate
      from region_list(:report_region_id, 1, 3, 1)
      inner join acc_list al on al_from_reg_id = id_list and
            al_from_bg_id in ('10150', '10001', '10101') and
            al.al_id <> 090141710150000101 --  BOARD-5892 задача от финансистов
      left join opercurr op on op.cr_from_cu_id = al_currency and
            :report_date_end between op.cr_date and op.cr_date_end
      into :account, :bg_id, :currency, :reg_id, :al_memo, :cr_rate
  do
  begin
    saldo_hq_417 = 0;
    saldo_hq_840 = 0;
    saldo_hq_978 = 0;
    saldo_hq_other = 0;
    saldo_bru = 0;
    saldo_isk = 0;
    saldo_osh = 0;
    saldo_jal = 0;
    saldo_tash = 0;
    saldo_nar = 0;
    saldo_tal = 0;
    saldo_kad = 0;
    saldo_bat = 0;
    saldo_uzg = 0;
    saldo_chuy = 0;
    saldo_bish = 0;
    saldo_other = 0;
    saldo_region = 0;

    line_name = '';
    summa_ekv = f_al_bal(account, report_date_end);
    obs_saldo_val = f_al_bal(account, report_date_end, 'val');

    if (currency <> '417') then
      summa_ekv = obs_saldo_val * cr_rate;

    summa_ekv = coalesce(summa_ekv, 0);

    if (summa_ekv <> 0.00) then
    begin
      if (bg_id = '10001') then
        line_name = 'Наличные в кассах';
      else
        line_name = al_memo;
      if (line_name = '') then
        line_name = 'X Банк не указан';

      if (reg_id = 1090100) then
      begin
        if (currency = '417') then
          saldo_hq_417 = summa_ekv;
        else
        if (currency = '840') then
          saldo_hq_840 = summa_ekv;
        else
        if (currency = '978') then
          saldo_hq_978 = summa_ekv;
        else
          saldo_hq_other = summa_ekv;
      end
      else
      begin
        if (reg_id = 1060100) then --1060100  Бишкекский офис
        begin
          saldo_bru = summa_ekv;
        end
        else
        begin
          saldo_region = summa_ekv;

          select parent
          from region reg
          where reg.reg_unicode = :reg_id
          into region_id;
          /*
          MRAKHMANOV
          |REG_UNICODE  REG_NAME
          1010000  Иссык-Кульский р.
          1030000  Чуйский р.
          1040000  Ошский р.
          1050000  Жалал-Абадский р.
          1060000  Бишкекский р.
          1060100  Бишкекский офис
          1070000  Ташкомурский р.
          1090002  Нарынский р.
          1090004  Таласский р.
          1090306  Закрытые офисы и регионы
          1091000  Кадамжайский р.
          1110000  Баткенский р.
          1270000  Узгенский р. */

          if (region_id = 1010000) then
            saldo_isk = summa_ekv; --1010000    Иссык-Кульский р.
          else
          if (region_id = 1030000) then
            saldo_chuy = summa_ekv; --1030000    Чуйский р. (был Кантский)
          else
          if (region_id = 1040000) then
            saldo_osh = summa_ekv; --1040000    Ошский р.
          else
          if (region_id = 1050000) then
            saldo_jal = summa_ekv; --1050000    Жалал-Абадский р.
          else
          if (region_id = 1060000) then
            saldo_bish = summa_ekv; --1060000  Бишкекский р.
          else
          if (region_id = 1070000) then
            saldo_tash = summa_ekv; --1070000    Ташкомурский р.
          else
          if (region_id = 1090002) then
            saldo_nar = summa_ekv; --1090002    Нарынский р.
          else
          if (region_id = 1090004) then
            saldo_tal = summa_ekv; --1090004    Таласский р.
          else
          if (region_id = 1091000) then
            saldo_kad = summa_ekv; --1091000    Кадамжайский р.
          else
          if (region_id = 1110000) then
            saldo_bat = summa_ekv; --1110000    Баткенский р.
          else
          if (region_id = 1270000) then
            saldo_uzg = summa_ekv; --1270000    Узгенский р.
          else
            saldo_other = summa_ekv; --Прочие
        end
      end
      suspend;
    end
  end
end
