CREATE PROCEDURE HP_PERSTEMP_DIFF(
    C_ID ID_TYPE)
RETURNS (
    LINE_NO D_INTEGER,
    INFO_NAME D_STR_100,
    INFO_OLD D_STR_255,
    INFO_NEW D_STR_255)
AS
declare variable c_adl_id_1 d_id;
declare variable c_adl_id_2 d_id;
begin
  if (coalesce(c_id, 0) = 0) then
    exit;

  line_no = 0;

  for select pte_name_fio, pte_alternat_fio, pte_name_eng, pte_sex, pte_birth_date, pte_citizen,
             pte_inn, pte_ext_inn, pte_pincode, pte_snils,
             pte_doc_type, pte_pass_no, pte_pass_date, pte_pass_date_expire, pte_pass_who,
             pte_pass_who_code, pte_blob_pass1_id, pte_blob_pass2_id, pte_blob_photo_id,
             pte_birth_country, pte_birth_city, pte_adl_id_1, pte_adl_id_2
  from perstemp
  where pte_from_c_id = :c_id
  as cursor p
  do
  begin
    select adc_from_adl_id
    from addrclient
    where adc_from_c_id = :c_id and
          adc_address_type = 1
    into c_adl_id_1;

    select adc_from_adl_id
    from addrclient
    where adc_from_c_id = :c_id and
          adc_address_type = 2
    into c_adl_id_2;

    for select c_name_fio, c_alternat_fio, c_name_eng, c_sex, c_birth_date, c_citizen,
               c_inn, c_ext_inn, c_pincode, c_reg_no,
               c_doc_type, c_pass_no, c_pass_date, c_pass_date_expire, c_pass_who,
               c_pass_who_code, c_blob_pass1_id, c_blob_pass2_id, c_blob_photo_id,
               c_birth_country, c_birth_city
    from clients
    where c_id = :c_id
    as cursor c
    do
    begin
      if (coalesce(p.pte_name_fio, '') is distinct from coalesce(c.c_name_fio, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'ФИО';
        info_old = c.c_name_fio;
        info_new = p.pte_name_fio;
        suspend;
      end

      if (coalesce(p.pte_alternat_fio, '') is distinct from coalesce(c.c_alternat_fio, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'ФИО на латинском';
        info_old = c.c_alternat_fio;
        info_new = p.pte_alternat_fio;
        suspend;
      end

      if (coalesce(p.pte_name_eng, '') is distinct from coalesce(c.c_name_eng, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'ФИО на английском';
        info_old = c.c_name_eng;
        info_new = p.pte_name_eng;
        suspend;
      end

      if (p.pte_sex is distinct from c.c_sex) then
      begin
        line_no = line_no + 1;
        info_name = 'Пол';
        info_old = iif(c.c_sex = 0, 'Женский', 'Мужской');
        info_new = iif(p.pte_sex = 0, 'Женский', 'Мужской');
        suspend;
      end

      if (p.pte_birth_date is distinct from c.c_birth_date) then
      begin
        line_no = line_no + 1;
        info_name = 'Дата рождения';
        info_old = f_datetostr(c.c_birth_date);
        info_new = f_datetostr(p.pte_birth_date);
        suspend;
      end

      if (p.pte_citizen is distinct from c.c_citizen) then
      begin
        line_no = line_no + 1;
        info_name = 'Гражданство';

        info_old = '';
        info_new = '';

        select rd_name
        from refdata
        where rd_id = c.c_citizen
        into info_old;

        select rd_name
        from refdata
        where rd_id = p.pte_citizen
        into info_new;

        suspend;
      end

      if (coalesce(p.pte_inn, '') is distinct from coalesce(c.c_inn, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'ИНН';
        info_old = c.c_inn;
        info_new = p.pte_inn;
        suspend;
      end

      if (coalesce(p.pte_ext_inn, '') is distinct from coalesce(c.c_ext_inn, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'ИНН на родине';
        info_old = c.c_ext_inn;
        info_new = p.pte_ext_inn;
        suspend;
      end

      if (coalesce(p.pte_pincode, '') is distinct from coalesce(c.c_pincode, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'ПИН';
        info_old = c.c_pincode;
        info_new = p.pte_pincode;
        suspend;
      end

      if (coalesce(p.pte_snils, '') is distinct from coalesce(c.c_reg_no, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'СНИЛС';
        info_old = c.c_reg_no;
        info_new = p.pte_snils;
        suspend;
      end

      if (p.pte_doc_type is distinct from c.c_doc_type) then
      begin
        info_name = 'Тип документа';

        info_old = '';
        info_new = '';
        select pk_name
        from passkind
        where pk_id = c.c_doc_type
        into info_old;

        select pk_name
        from passkind
        where pk_id = p.pte_doc_type
        into info_new;

        if (info_old is distinct from info_new) then
        begin
          line_no = line_no + 1;
          suspend;
        end
      end

      if (coalesce(p.pte_pass_no, '') is distinct from coalesce(c.c_pass_no, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'Номер документа';
        info_old = c.c_pass_no;
        info_new = p.pte_pass_no;
        suspend;
      end

      if (p.pte_pass_date is distinct from c.c_pass_date) then
      begin
        line_no = line_no + 1;
        info_name = 'Дата выдачи';
        info_old = f_datetostr(c.c_pass_date);
        info_new = f_datetostr(p.pte_pass_date);
        suspend;
      end

      if (p.pte_pass_date_expire is distinct from c.c_pass_date_expire) then
      begin
        line_no = line_no + 1;
        info_name = 'Срок действия';

        info_old = trim(iif(c.c_pass_date_expire >= '31.12.2999', 'Бессрочный',  f_datetostr(c.c_pass_date_expire)));
        info_new = trim(iif(p.pte_pass_date_expire >= '31.12.2999', 'Бессрочный',  f_datetostr(p.pte_pass_date_expire)));
        suspend;
      end

      if (coalesce(p.pte_pass_who, '') is distinct from coalesce(c.c_pass_who, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'Орган выдачи';
        info_old = c.c_pass_who;
        info_new = p.pte_pass_who;
        suspend;
      end

      if (coalesce(p.pte_pass_who_code, '') is distinct from coalesce(c.c_pass_who_code, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'Код органа';
        info_old = c.c_pass_who_code;
        info_new = p.pte_pass_who_code;
        suspend;
      end

      if (coalesce(p.pte_birth_country, '') is distinct from coalesce(c.c_birth_country, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'Страна рождения';
        info_old = c.c_birth_country;
        info_new = p.pte_birth_country;
        suspend;
      end

      if (coalesce(p.pte_birth_city, '') is distinct from coalesce(c.c_birth_city, '')) then
      begin
        line_no = line_no + 1;
        info_name = 'Место рождения';
        info_old = c.c_birth_city;
        info_new = p.pte_birth_city;
        suspend;
      end

      if (p.pte_adl_id_1 is distinct from c_adl_id_1) then
      begin
        info_name = 'Адрес регистрации';

        info_old = '';
        info_new = '';
        select adl_address_str
        from addrlist
        where adl_id = :c_adl_id_1
        into info_old;

        select adl_address_str
        from addrlist
        where adl_id = p.pte_adl_id_1
        into info_new;

        if (info_old is distinct from info_new) then
        begin
          line_no = line_no + 1;
          suspend;
        end
      end

      if (p.pte_adl_id_2 is distinct from c_adl_id_2) then
      begin
        info_name = 'Адрес проживания';

        info_old = '';
        info_new = '';
        select adl_address_str
        from addrlist
        where adl_id = :c_adl_id_2
        into info_old;

        select adl_address_str
        from addrlist
        where adl_id = p.pte_adl_id_2
        into info_new;

        if (info_old is distinct from info_new) then
        begin
          line_no = line_no + 1;
          suspend;
        end
      end

      if (p.pte_blob_pass1_id is distinct from c.c_blob_pass1_id) then
      begin
        line_no = line_no + 1;
        info_name = 'Скан документа 1';

        info_old = '';
        info_new = '';

        select f_datetimetostr(bls_datetime)
        from blobstore
        where bls_id = c.c_blob_pass1_id
        into info_old;

        select f_datetimetostr(bls_datetime)
        from blobstore
        where bls_id = p.pte_blob_pass1_id
        into info_new;

        suspend;
      end

      if (p.pte_blob_pass2_id is distinct from c.c_blob_pass2_id) then
      begin
        line_no = line_no + 1;
        info_name = 'Скан документа 2';
        info_old = '';
        info_new = '';

        select f_datetimetostr(bls_datetime)
        from blobstore
        where bls_id = c.c_blob_pass2_id
        into info_old;

        select f_datetimetostr(bls_datetime)
        from blobstore
        where bls_id = p.pte_blob_pass2_id
        into info_new;

        suspend;
      end

      if (p.pte_blob_photo_id is distinct from c.c_blob_photo_id) then
      begin
        line_no = line_no + 1;
        info_name = 'Фото';

        info_old = '';
        info_new = '';

        select f_datetimetostr(bls_datetime)
        from blobstore
        where bls_id = c.c_blob_photo_id
        into info_old;

        select f_datetimetostr(bls_datetime)
        from blobstore
        where bls_id = p.pte_blob_photo_id
        into info_new;

        suspend;
      end
    end
  end
end
