CREATE PROCEDURE HP_INTERN_RESULT(
    I_ID D_INTEGER,
    UN_ID D_ID,
    RESULT D_SMALLINT,
    COMM D_STR_255,
    ACCESS D_STR_100)
RETURNS (
    RES_TEXT D_STR_100)
AS
declare variable sd_id d_integer;
declare variable date_begin d_date;
declare variable work_days d_integer;
begin
  res_text = 'Произошла ошибка.';
  if (access = 'Committee.personnel_committee') then
  begin
    if (i_id > 0 and
        exists(select *
               from intern_committee_result
               where icr_from_i_id = :i_id)) then
      update intern_committee_result
      set icr_personnel_com = :un_id,
          icr_personnel_com_res = :result,
          icr_personnel_com_comment = :comm,
          icr_personnel_com_timestamp = current_timestamp
      where (icr_from_i_id = :i_id);
    else
    begin
      insert into intern_committee_result (icr_from_i_id, icr_personnel_com, icr_personnel_com_res,
                                           icr_personnel_com_comment, icr_personnel_com_timestamp)
      values (:i_id, :un_id, :result, :comm, current_timestamp);

      execute procedure hp_intern_send_email(:i_id, 'personnel_committee');
    end
    res_text = 'Стажер утвержден.';
  end
  if (access = 'Committee.educational_committee') then
  begin
    update intern_committee_result
    set icr_educational_com = :un_id,
        icr_educational_com_res = :result,
        icr_educational_com_comment = :comm,
        icr_educational_com_timestamp = current_timestamp
    where (icr_from_i_id = :i_id);

    update interns
    set i_date_end_fact = current_date
    where (i_id = :i_id);

    select icr.icr_from_sd_id
    from intern_committee_result icr
    where icr.icr_from_i_id = :i_id
    into :sd_id;

    if (:result = 0) then
    begin
      update servicedog
      set sd_date_closed = current_date,
          sd_datetime = current_timestamp,
          sd_status = -1
      where (sd_id = :sd_id);
      res_text = 'Стажер отклонен.';
    end
    else
    begin
      update servicedog
      set sd_date_closed = current_date,
          sd_datetime = current_timestamp
      where (sd_id = :sd_id);

      select sd.sd_date_begin
      from servicedog sd
      where sd.sd_id = :sd_id
      into :date_begin;

      select counter
      from hp_get_work_date(10, :date_begin, current_date)
      into :work_days;

      update servicesh
      set ssh_date = current_date,
          ssh_amount_dogovor = 125 * :work_days
      where (ssh_from_sd_id = :sd_id) and
            (ssh_id = 1);

      execute procedure hp_intern_send_email(:i_id, 'educational_committee');
      res_text = 'Стажер утвержден.';
    end
  end
  suspend;
end
