CREATE PROCEDURE AAREP_WORKDOC_VIEW_LIST(
    WDL_ID INTEGER)
RETURNS (
    LINE_NO INTEGER,
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    C_OPEN_KOLICH INTEGER,
    C_LAST_OPENED TIMESTAMP,
    C_JL_NAME VARCHAR(100),
    C_REG_NAME VARCHAR(100),
    C_IS_OPEN INTEGER,
    C_FIRST_OPENED TIMESTAMP)
AS
begin
  line_no = 0;
  for select c_id, c_name_fio, sum(c_open_kolich), max(c_last_opened), min(c_first_opened), c_jl_name, c_reg_name
      from aarep_workdoc_view_tmp(:wdl_id)
      group by c_name_fio, c_id, c_jl_name, c_reg_name
      into :c_id, :c_name_fio, :c_open_kolich, :c_last_opened, :c_first_opened, :c_jl_name, :c_reg_name
  do
  begin
    line_no = line_no + 1;
    if (c_open_kolich > 0) then
      c_is_open = 1;
    else
      c_is_open = 0;
    suspend;
  end
end
