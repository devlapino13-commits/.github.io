CREATE PROCEDURE AAREP_KR_SB_CLI(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    AA_K_ID ID_TYPE,
    AA_CURR_LEVEL_NAME STR_100,
    AA_CURR_BRANCH_NAME STR_100,
    AA_CURR_OFFICE_NAME STR_100,
    AA_KF_NAME STR_100,
    AA_KONTRACT_NO STR_100,
    AA_SB_TEAM_NAME STR_100,
    AA_SB_NAME STR_100,
    SBA_DATE_BEGIN DATE,
    AA_KOLICH_CL_ACTIVE_BEGIN INTEGER,
    AA_KOLICH_CL_ACTIVE_END INTEGER,
    AA_KOLICH_CLOSED INTEGER,
    AA_CURR_KC_FIO STR_100,
    K_CASHIER_FIO STR_100,
    AAREP_LEADER_FIO STR_100,
    AA_LOAN_AMOUNT CURR_TYPE,
    SBA_DISB_DATE DATE,
    AA_PRODUCT_NAME STR_100,
    AA_LOAN_CURRENCY VARCHAR(30),
    AA_LOAN_KIND VARCHAR(30),
    AA_DELIQ_REPAY_KOLICH INTEGER,
    AA_DELIQ_DAYS INTEGER,
    AA_DELIQ_BALANCE CURR_TYPE,
    AA_DELIQ_OS_ALL CURR_TYPE,
    AA_DELIQ_PERC_ALL CURR_TYPE,
    OKO_BALANCE_OTHER CURR_TYPE,
    OKO_BALANCE_TOTAL CURR_TYPE,
    AA_REPAY_WRITEOFF_FINE CURR_TYPE,
    AA_REPAY_WRITEOFF_PERC CURR_TYPE,
    AA_REPAY_WRITEOFF_OS CURR_TYPE,
    AA_REPAY_ALL_OS CURR_TYPE,
    AA_REPAY_ALL_PERC CURR_TYPE,
    AA_REPAY_ALL_KOMMIS CURR_TYPE,
    AA_REPAY_ALL_KOMMIS_ADV CURR_TYPE,
    AA_REPAY_ALL_FINE CURR_TYPE,
    OKO_PAYED_OTHER CURR_TYPE,
    AA_REPAY_ALL CURR_TYPE,
    AA_STATUS_TEXT STR_100,
    AA_AMOUNT_ACTIVE_END CURR_TYPE,
    AA_AMOUNT_ACTIVE_BEGIN CURR_TYPE,
    AA_CURR_KC_SEX STR_20,
    AA_DELIQ_BEG_CL_KOLICH CURR_TYPE,
    AA_DELIQ_BEG_BALANCE CURR_TYPE,
    SBL_TYPE_STR STR_100,
    OKO_PAYED_OS_PER CURR_TYPE,
    OKO_PAYED_PERC_PER CURR_TYPE,
    OKO_PAYED_OTHER_PER CURR_TYPE,
    OKO_PAYED_TOTAL_PER CURR_TYPE,
    TYPE_STR VARCHAR(20),
    C_ADDRESS_FACT STR_255,
    C_ADDRESS_YUR STR_255,
    SBA_CLIENT_NAME STR_100,
    SBA_CL_TEL_NO STR_20,
    SBA_CL_BIRTH_DATE DATE,
    KC_AMOUNT_APPROVED CURR_TYPE)
AS
declare variable dolya curr_type;
declare variable kc_from_c_id id_type;
declare variable oko_payed_total_per1 curr_type;
declare variable aa_amount_active_begin1 curr_type;
declare variable aa_amount_active_end1 curr_type;
declare variable aa_deliq_balance1 curr_type;
declare variable aa_deliq_os_all1 curr_type;
declare variable aa_deliq_perc_all1 curr_type;
declare variable aa_loan_amount1 curr_type;
declare variable aa_repay_all_fine1 curr_type;
declare variable aa_repay_all_kommis1 curr_type;
declare variable aa_repay_all_kommis_adv1 curr_type;
declare variable aa_repay_all_os1 curr_type;
declare variable aa_repay_all_perc1 curr_type;
declare variable aa_repay_writeoff_fine1 curr_type;
declare variable aa_repay_writeoff_os1 curr_type;
declare variable aa_repay_writeoff_perc1 curr_type;
declare variable oko_balance_other1 curr_type;
declare variable oko_payed_os_per1 curr_type;
declare variable oko_payed_other_per1 curr_type;
declare variable oko_payed_perc_per1 curr_type;
begin
  for select aa_k_id, aa_curr_level_name, aa_curr_branch_name, aa_curr_office_name, aa_kf_name, aa_kontract_no,
             aa_sb_team_name, aa_sb_name, sba_date_begin, aa_kolich_cl_active_begin, aa_kolich_cl_active_end,
             aa_kolich_closed, aa_curr_kc_fio, k_cashier_fio, aarep_leader_fio, aa_loan_amount, sba_disb_date,
             aa_product_name, aa_loan_currency, aa_loan_kind, aa_deliq_repay_kolich, aa_deliq_days, aa_deliq_balance,
             aa_deliq_os_all, aa_deliq_perc_all, oko_balance_other, oko_balance_total, aa_repay_writeoff_fine,
             aa_repay_writeoff_perc, aa_repay_writeoff_os, aa_repay_all_os, aa_repay_all_perc, aa_repay_all_kommis,
             aa_repay_all_kommis_adv, aa_repay_all_fine, oko_payed_other, aa_repay_all, aa_status_text,
             aa_amount_active_end, aa_amount_active_begin, aa_curr_kc_sex, aa_deliq_beg_cl_kolich, aa_deliq_beg_balance,
             sbl_type_str, oko_payed_os_per, oko_payed_perc_per, oko_payed_other_per, oko_payed_total_per, type_str
      from aarep_kr_sb(:rep_region_id, :rep_date_begin, :rep_date_end)
      into :aa_k_id, :aa_curr_level_name, :aa_curr_branch_name, :aa_curr_office_name, :aa_kf_name, :aa_kontract_no,
           :aa_sb_team_name, :aa_sb_name, :sba_date_begin, :aa_kolich_cl_active_begin, :aa_kolich_cl_active_end,
           :aa_kolich_closed, :aa_curr_kc_fio, :k_cashier_fio, :aarep_leader_fio, :aa_loan_amount, :sba_disb_date,
           :aa_product_name, :aa_loan_currency, :aa_loan_kind, :aa_deliq_repay_kolich, :aa_deliq_days,
           :aa_deliq_balance, :aa_deliq_os_all, :aa_deliq_perc_all, :oko_balance_other, :oko_balance_total,
           :aa_repay_writeoff_fine, :aa_repay_writeoff_perc, :aa_repay_writeoff_os, :aa_repay_all_os,
           :aa_repay_all_perc, :aa_repay_all_kommis, :aa_repay_all_kommis_adv, :aa_repay_all_fine, :oko_payed_other,
           :aa_repay_all, :aa_status_text, :aa_amount_active_end, :aa_amount_active_begin, :aa_curr_kc_sex,
           :aa_deliq_beg_cl_kolich, :aa_deliq_beg_balance, :sbl_type_str, :oko_payed_os_per, :oko_payed_perc_per,
           :oko_payed_other_per, :oko_payed_total_per, :type_str
  do
  begin
    if (aa_kolich_cl_active_end > 0) then
      aa_kolich_cl_active_end = 1;
    if (aa_kolich_cl_active_begin > 0) then
      aa_kolich_cl_active_begin = 1;
    if (type_str = 'Внесистемный учет') then
      aa_kolich_closed = 1;

    aa_deliq_balance1 = aa_deliq_balance;
    aa_deliq_perc_all1 = aa_deliq_perc_all;
    aa_deliq_os_all1 = aa_deliq_os_all;
    aa_repay_writeoff_fine1 = aa_repay_writeoff_fine;
    aa_repay_writeoff_perc1 = aa_repay_writeoff_perc;
    aa_repay_writeoff_os1 = aa_repay_writeoff_os;
    aa_repay_all_kommis_adv1 = aa_repay_all_kommis_adv;
    aa_repay_all_kommis1 = aa_repay_all_kommis;
    aa_repay_all_fine1 = aa_repay_all_fine;
    aa_repay_all_perc1 = aa_repay_all_perc;
    aa_repay_all_os1 = aa_repay_all_os;
    aa_loan_amount1 = aa_loan_amount;
    aa_amount_active_end1 = aa_amount_active_end;
    aa_amount_active_begin1 = aa_amount_active_begin;
    oko_balance_other1 = oko_balance_other;
    oko_payed_os_per1 = oko_payed_os_per;
    oko_payed_perc_per1 = oko_payed_perc_per;
    oko_payed_other_per1 = oko_payed_other_per;
    oko_payed_total_per1 = oko_payed_total_per;

    for select kc_from_c_id, kc_amount_approved
        from knclient
        where kc_from_k_id = :aa_k_id and
              kc_amount_approved > 0
        into kc_from_c_id, kc_amount_approved
    do
    begin
      dolya = kc_amount_approved / aa_loan_amount1;
      aa_deliq_balance = aa_deliq_balance1;
      aa_deliq_perc_all = dolya * aa_deliq_perc_all1;
      aa_deliq_os_all = dolya * aa_deliq_os_all1;
      aa_repay_writeoff_fine = dolya * aa_repay_writeoff_fine1;
      aa_repay_writeoff_perc = dolya * aa_repay_writeoff_perc1;
      aa_repay_writeoff_os = dolya * aa_repay_writeoff_os1;
      aa_repay_all_kommis_adv = dolya * aa_repay_all_kommis_adv1;
      aa_repay_all_kommis = dolya * aa_repay_all_kommis1;
      aa_repay_all_fine = dolya * aa_repay_all_fine1;
      aa_repay_all_perc = dolya * aa_repay_all_perc1;
      aa_repay_all_os = dolya * aa_repay_all_os1;
      aa_loan_amount = kc_amount_approved;
      aa_amount_active_end = dolya * aa_amount_active_end1;
      aa_amount_active_begin = dolya * aa_amount_active_begin1;
      oko_balance_other = dolya * oko_balance_other1;
      oko_payed_os_per = dolya * oko_payed_os_per1;
      oko_payed_perc_per = dolya * oko_payed_perc_per1;
      oko_payed_other_per = dolya * oko_payed_other_per1;
      oko_payed_total_per = dolya * oko_payed_total_per1;

      select c_name_fio, c_birth_date,
             (select cont_value
              from hp_contdata_get_cell_main(0, :kc_from_c_id))
      from clients
      where c_id = :kc_from_c_id
      into sba_client_name, sba_cl_birth_date, sba_cl_tel_no;

      c_address_yur = null;
      c_address_fact = null;

      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :kc_from_c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      into :c_address_yur;
      if (c_address_yur is null) then
        c_address_yur = 'Не указан';

      select adl_address_str
      from addrclient, addrlist
      where adc_from_c_id = :kc_from_c_id and
            adc_address_type = 2 and
            adc_from_adl_id = adl_id
      into :c_address_fact;
      if (c_address_fact is null) then
        c_address_fact = 'Не указан';

      suspend;
    end
  end
end
