CREATE PROCEDURE HP_INSURANCE_GET_RANDOM_COMPANY(
    IA_ID D_INTEGER)
RETURNS (
    AG_ID D_INTEGER,
    EDC_CODE D_STR_CODE)
AS
declare variable rnd_val d_percent;
declare variable current_ag_id d_integer;
declare variable current_edc_code d_str_code;
declare variable percent d_smallint;
declare variable accumulated_percent d_smallint;
begin
    ag_id = null;
    accumulated_percent = 0;

    -- 1. генерируем случайное число от 0.0 до 100.0
    rnd_val = rand() * 100.0;

    -- 2. бежим по компаниям, привязанным к этому договору
    for select ic.ic_from_ag_id, ic.ic_edc_code, ic.ic_percent
    from insurance_company ic
    where ic.ic_from_ia_id = :ia_id
    order by ic.ic_percent desc
    into :current_ag_id, :current_edc_code, :percent
    do
    begin
        -- накапливаем проценты, сдвигая границу отрезка
        accumulated_percent = accumulated_percent + percent;

        -- если случайное число попало в текущий отрезок
        if (rnd_val < accumulated_percent) then
        begin
            ag_id = :current_ag_id;
            edc_code = :current_edc_code;
            break;
        end
    end

    -- 3. страховка на случай, если сумма процентов в базе была меньше 100%,
    -- а рандом выдал число больше, чем удалось накопить.
    if (ag_id is null and current_ag_id is not null) then
        ag_id = current_ag_id;

    if (edc_code is null and current_edc_code is not null) then
        edc_code = current_edc_code;

    suspend;
end
