CREATE PROCEDURE PREPARE_UNIQKEY_METADATA(
    ACTION_KIND VARCHAR(10),
    ACTION_TABLE VARCHAR(100))
RETURNS (
    SCRIPT_SQL BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable table_name varchar(100);
declare variable field_name varchar(100);
begin
-- ACTION_KIND  - 'CREATE' or 'DROP'
-- ACTION_TABLE - Название таблицы для создания или удаления триггеров и данных

  script_sql = 'SET TERM ^ ;';
  suspend;
  if (action_table is null) then
    action_table = '';

  if (action_table = '') then Exit;

  if (action_kind = 'CREATE') then
  begin
    table_name = f_objkind(action_table, 'name');
    field_name = f_objkind(action_table, 'field');

    table_name = trim(table_name);
    field_name = trim(field_name);

    if (not(exists(select *
                   from rdb$triggers t
                   where t.rdb$trigger_name = 'UNQ_TRIGGER_' || :table_name || '_AU777'))) then
    begin
-- Триггер на After Update
      script_sql = 'CREATE OR ALTER TRIGGER UNQ_TRIGGER_' || :table_name || '_AU777 FOR ' || :table_name || ascii_char(13) || ascii_char(10);
      script_sql = BLOB_APPEND(script_sql, 'ACTIVE AFTER UPDATE POSITION 777' || ascii_char(13) || ascii_char(10));
      script_sql = BLOB_APPEND(script_sql, 'AS ' || ascii_char(13) || ascii_char(10));
      script_sql = BLOB_APPEND(script_sql, 'begin' || ascii_char(13) || ascii_char(10));
      script_sql = BLOB_APPEND(script_sql, '  execute procedure HP_UNIQKEYS_UPDATE(2, ''' || :table_name || ''', old.' || :field_name || ', new.' || :field_name || ');' || ascii_char(13) || ascii_char(10));
      script_sql = BLOB_APPEND(script_sql, 'end ^' || ascii_char(13) || ascii_char(10));
      suspend;
    end

    if (not(exists(select *
                   from rdb$triggers t
                   where t.rdb$trigger_name = 'UNQ_TRIGGER_' || :table_name || '_AD777'))) then
    begin
-- Триггер на After Delete
      script_sql = 'CREATE OR ALTER TRIGGER UNQ_TRIGGER_' || :table_name || '_AD777 FOR ' || :table_name || ascii_char(13) || ascii_char(10);
      script_sql = BLOB_APPEND(script_sql, 'ACTIVE AFTER DELETE POSITION 777' || ascii_char(13) || ascii_char(10));
      script_sql = BLOB_APPEND(script_sql, 'AS ' || ascii_char(13) || ascii_char(10));
      script_sql = BLOB_APPEND(script_sql, 'begin' || ascii_char(13) || ascii_char(10));
      script_sql = BLOB_APPEND(script_sql, '  execute procedure HP_UNIQKEYS_UPDATE(3, ''' || :table_name || ''', old.' || :field_name || ', null);' || ascii_char(13) || ascii_char(10));
      script_sql = BLOB_APPEND(script_sql, 'end ^' || ascii_char(13) || ascii_char(10));
      suspend;
    end
  end

  if (action_kind = 'DROP') then
  begin
    table_name = f_objkind(action_table, 'name');
    field_name = f_objkind(action_table, 'field');
    table_name = trim(table_name);
    field_name = trim(field_name);

    if (not(exists(select *
                   from rdb$triggers t
                   where t.rdb$trigger_name = 'UNQ_TRIGGER_' || :table_name || '_AI777'))) then
    begin
-- Триггер на After Insert
      script_sql = 'DROP TRIGGER UNQ_TRIGGER_' || :table_name || '_AI777 ^';
      suspend;
    end

    if (not(exists(select *
                   from rdb$triggers t
                   where t.rdb$trigger_name = 'UNQ_TRIGGER_' || :table_name || '_AU777'))) then
    begin
-- Триггер на After Update
      script_sql = 'DROP TRIGGER UNQ_TRIGGER_' || :table_name || '_AU777 ^';
      suspend;
    end

    if (not(exists(select *
                   from rdb$triggers t
                   where t.rdb$trigger_name = 'UNQ_TRIGGER_' || :table_name || '_AD777'))) then
    begin
-- Триггер на After Delete
      script_sql = 'DROP TRIGGER UNQ_TRIGGER_' || :table_name || '_AD777 ^';
      suspend;
    end
  end

  script_sql = 'SET TERM ; ^';
  suspend;
end
