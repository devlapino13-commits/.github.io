CREATE PROCEDURE HP_ONLINE_LOAN_RU_KIB_CHECK(
    EIL_ID D_ID)
AS
declare variable eil_object_kind d_integer;
declare variable eil_object_id d_id;
declare variable eic_code d_str_code;
declare variable pr_code d_str_100;
declare variable eil_status d_smallint;
declare variable eil_kk_result d_smallint;
begin
  select eil_object_kind, eil_object_id, eic_code, eil_status, eil_kk_result
  from extinfolist
  inner join extinfocfg on eic_id = eil_from_eic_id
  where eil_id = :eil_id
  into eil_object_kind, eil_object_id, eic_code, eil_status, eil_kk_result;

  if (eic_code = 'check_in_ru_db' and
      f_objkind('miniapp', 'id') = eil_object_kind) then
  begin
    select pr_code
    from miniapp
    inner join product pr on pr_id = ma_from_pr_id
    where ma_id = :eil_object_id and
          ma_is_active = 1 and
          ma_date_end >= '31.12.9999'
    into pr_code;

    if (pr_code containing '.online_loan.' and
       (eil_status < 0 or eil_kk_result = 1)) then
      update expressapp
      set exa_final_status = -1
      where exa_from_ma_id = :eil_object_id;
  end 
end
