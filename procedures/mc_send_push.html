CREATE PROCEDURE MC_SEND_PUSH(
    OBJECT_KIND D_STR_100,
    OBJECT_ID D_ID,
    SUBJECT D_STR_255,
    MSG_BODY D_STR_LONG,
    LIFETIME_DAYS D_INTEGER = -1)
AS
declare variable res_body d_str_2048;
declare variable google_id d_str_255;
declare variable mcl_id d_id;
declare variable msg_id d_integer;
declare variable rep_id d_str_100;
declare variable db_id d_integer;
begin

  db_id = f_get_db_id();
  if (object_kind is null) then
    exception e_default 'parametr "object_kind" is null';

  if (object_id is null) then
    exception e_default 'parametr "object_id" is null';

  if (subject is null) then
    exception e_default 'parametr "subject" is null';

  if (msg_body is null) then
    exception e_default 'parametr "msg_body" is null';

  if (object_kind = 'mc_login') then
  begin
    mcl_id = :object_id;

    select mcl_google_id
    from mc_login mcl
    where mcl_id = :object_id and
          mcl.mcl_is_delete = 0
    into :google_id;
  end
  else
  if (object_kind = 'clients') then
  begin
    -- выявляем пользоватля МП по c_id
    select mcl_google_id, mcl.mcl_id
    from mc_login mcl
    where mcl.mcl_from_c_id = :object_id and
          mcl.mcl_login not containing ('_') and
          mcl.mcl_is_delete = 0
    order by mcl.mcl_id desc
    rows 1
    into :google_id, :mcl_id;

    if (mcl_id is null) then -- mcl_id is null в случае если клиент не активный пользователь МП.
      exit;
  end
  else
    exception e_default 'parametr "object_kind" has invalid value. ';

  google_id = coalesce(google_id, ''); -- если у нас нет google_id (а это вполне допустимо, если на телефоне не устанволены сервисы google, тогда пуш-фактически не уйдет, а мы сразу отразим его как отправленнынй status = 1)

  rep_id = 'mailmessages_push';

  if (db_id = 81001) then
  begin
    if (msg_body containing 'Станьте Партнером М Булак и начните зарабатывать прямо сейчас.') then
      rep_id = 'partner_program'; --Описание партнерской программы
    else
    if (msg_body containing 'Уважаемый Партнер, благодарим вас за сотрудничество.') then
      rep_id = 'partner_program_invite'; --Пригласить друга/партнера
  end

  if (db_id = 39801) then
  begin
    -- RFCKG-4284
    if (subject containing 'Ознакомление с согласиями') then
        rep_id = 'cons_fam';

    -- RFCKG-4371
    if (subject containing 'Добавьте банковскую карту') then
        rep_id = 'bank_card_add';
  end

  select res_text
  from mc_fcm_data(:msg_body, :subject, :rep_id, 'fsrobot_msgbox', null, null, null, null, :mcl_id)
  into res_body;

  insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id,
                              msg_command_kind, msg_object_kind, msg_object_id, msg_status)
  values (2, :google_id, :subject, :res_body, :lifetime_days, 1000000000000, 'msg_to_clients',
          f_objkind('mc_login', 'id'), :mcl_id, 0)
  returning msg_id
  into :msg_id;

  if (google_id = '') then
  begin
    update fsrobot_msgbox fmb
    set fmb.msg_status = 1
    where fmb.msg_id = :msg_id;
  end
end
