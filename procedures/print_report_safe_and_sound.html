CREATE PROCEDURE PRINT_REPORT_SAFE_AND_SOUND(
    MA_ID D_INTEGER,
    C_ID D_ID)
RETURNS (
    EIL_DATETIME D_DATETIME,
    UN_FIO D_STR_100,
    IPF_POINTS D_INTEGER,
    COMMON_POINTS D_INTEGER,
    C_NAME_FIO D_STR_255,
    MAC_AMOUNT D_CURRENCY,
    MA_KOLICH_REPAY D_INTEGER,
    BIRTH_DATE D_DATE,
    PASS_NO D_STR_CODE,
    INN D_STR_20,
    TOTAL_OUTSTANDING D_CURRENCY,
    AMOUNT D_INTEGER,
    TOTAL_HISTORY D_INTEGER,
    ROWS_NO D_INTEGER,
    SCORE D_INTEGER,
    RISK_STR D_STR_20,
    RISK_LEVEL D_STR_10,
    PROVIDER_INST D_STR_255,
    OWN_TYPE_INST D_STR_20,
    START_DATE_INST D_DATE,
    END_DATE_INST D_DATE,
    BORROWERS_INST D_STR_2048,
    GUARANTORS_INST D_STR_2048,
    LOAN_AMOUNT_INST D_CURRENCY,
    CURRENCY_INST D_STR_20,
    ROLE_STR_INST D_STR_20,
    PAYMENT_AMOUNT_INST D_CURRENCY,
    DURATION_INST D_INTEGER,
    OUTSTANDING_INST D_CURRENCY,
    INTEREST_OUTSTANDING_INST D_CURRENCY,
    LOAN_STATUS_INST D_STR_100,
    PAYMENT_STATUS_INST D_STR_100,
    MAX_PAST_DUE_INST D_STR_100,
    IS_OUR_INST D_SMALLINT,
    INTEREST_RATE D_STR_10)
AS
declare variable json d_blob_text;
declare variable ma_date_app d_date;
declare variable bls_id d_id;
declare variable base_path d_str_code;
declare variable un_id d_id;
declare variable h_qty d_integer;
declare variable closed_qty d_integer;
declare variable current_closed d_integer;
declare variable pr_code d_str_100;
declare variable base_k_id d_id;
declare variable base_ma_id d_integer;
declare variable insur_ma_id d_integer;
declare variable kib_change_actualiti_date d_date;
begin
  kib_change_actualiti_date = f_get_const('kib_change_actualiti_date');

  select pr.pr_code
  from miniapp ma
  inner join product pr on pr.pr_id = ma.ma_from_pr_id
  where ma.ma_id = :ma_id and
        ma.ma_is_active = 1 and
        ma.ma_date_end >= '31.12.2999'
  into pr_code;

  if (pr_code containing '.insur_kontrakt.') then  -- RFCKG-5051
  begin
    select lir.lir_kontract_base
    from kontract k
    inner join loan_insurance_relation lir on k.k_id = lir.lir_kontract_insurance
    where k.k_from_ma_app_id = :ma_id
    into base_k_id;

    select k.k_from_ma_app_id
    from kontract k
    where k.k_id = :base_k_id
    into base_ma_id;

    if (coalesce(base_ma_id, 0) <> 0) then
    begin
      insur_ma_id = ma_id;
      ma_id = base_ma_id;
    end
  end

  -- Получаем данные клиента
  select ma.ma_date_app, mac_amount, ma.ma_kolich_repay, c.c_name_fio, c.c_birth_date,  c.c_pass_no, c.c_inn
  from miniapp ma
  inner join miniappclient mac on ma.ma_id = mac.mac_from_ma_id
  inner join clients c on c.c_id = mac.mac_from_c_id
  where ma.ma_id = :ma_id
  and mac.mac_from_c_id = :c_id
  order by ma.ma_date_app desc
  rows 1
  into ma_date_app, mac_amount, ma_kolich_repay, c_name_fio, birth_date, pass_no, inn;

  if (coalesce(insur_ma_id, 0) <> 0) then
  begin
    select mac_amount, ma.ma_kolich_repay, c.c_name_fio, c.c_birth_date,  c.c_pass_no, c.c_inn
    from miniapp ma
    inner join miniappclient mac on ma.ma_id = mac.mac_from_ma_id
    inner join clients c on c.c_id = mac.mac_from_c_id
    where ma.ma_id = :insur_ma_id
    and mac.mac_from_c_id = :c_id
    order by ma.ma_date_app desc
    rows 1
    into mac_amount, ma_kolich_repay, c_name_fio, birth_date, pass_no, inn;
  end

  -- Получаем последний отчет SS
  select eil.eil_from_bls_id, eil.eil_status_datetime, eil.eil_from_un_id
  from extinfolist eil
  inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id and
        eic.eic_code = 'ss_cib_report'
  where eil.eil_from_c_id = :c_id and
        eil.eil_is_json = 1 and
        eil.eil_status = 2 and
        iif(:ma_date_app >= :kib_change_actualiti_date,
            eil.eil_date between :ma_date_app - 1 and :ma_date_app,                       -- RFCKG-5294
            eil.eil_object_kind = f_objkind('miniapp', 'id') and eil.eil_object_id = :ma_id)
  order by eil.eil_id desc
  rows 1
  into bls_id, eil_datetime, un_id;

  if (row_count = 0) then  -- RFCKG-5396
  begin
    select eil.eil_from_bls_id, eil.eil_status_datetime, eil.eil_from_un_id
    from extinfolist eil
    inner join extinfocfg eic on eil.eil_from_eic_id = eic.eic_id and
          eic.eic_code = 'ss_cib_report'
    where eil.eil_from_c_id = :c_id and
          eil.eil_object_kind = f_objkind('miniapp', 'id') and
          eil.eil_object_id = :ma_id and
          eil.eil_is_json = 1 and
          eil.eil_status = 2
    order by eil.eil_id desc
    rows 1
    into bls_id, eil_datetime, un_id;
  end

  if (bls_id is null) then
  begin
      c_name_fio = 'Клиент не найден';
      eil_datetime = null;
      un_fio = 'Не определен';
      suspend;
      exit;
  end

  select c_name_fio
  from clients, username
  where un_id = :un_id and
        un_from_c_id = c_id
  into :un_fio;
  if (un_fio is null) then
        un_fio = 'Не определен';

  ipf_points = f_point(:c_id, cast(eil_datetime as d_date), 2);
  common_points = f_point(:c_id, cast(eil_datetime as d_date), 0);

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json;

  if (row_count is null or coalesce(:json, '') = '') then
  begin
      c_name_fio = 'Клиент не найден';
      eil_datetime = null;
      un_fio = 'Не определен';
      suspend;
      exit;
  end

  if (f_json_blob_get_value(:json, 'result', '') = 'NO SUBJECT FOUND') then
  begin
    c_name_fio = 'Клиент не найден';
    eil_datetime = null;
    un_fio = 'Не определен';
    suspend;
    exit;
  end

  base_path = 'other_element.monitoring_enquiry';

  total_outstanding = cast(f_json_blob_get_value(:json, base_path || '.credit_report.contracts_history.aggregated_data.bctotal_amounts.total_outstanding', 0) as d_currency);
  amount = cast(f_json_blob_get_value(:json, base_path || '.credit_report.contracts_history.installments.amounts_summary.contracts_counter', 0) as d_integer) +
           cast(f_json_blob_get_value(:json, base_path || 'credit_report.contracts_history.non_installments.amounts_summary.contracts_counter', 0) as d_integer) +
           cast(f_json_blob_get_value(:json, base_path || 'credit_report.contracts_history.credit_cards.amounts_summary.contracts_counter', 0) as d_integer) +
           cast(f_json_blob_get_value(:json, base_path || 'credit_report.contracts_history.services.amounts_summary.contracts_counter', 0) as d_integer);
  total_history = cast(f_json_blob_get_value(:json, base_path || '.credit_report.contracts_history.aggregated_data.contracts_number', 0) as d_integer);

  ---------------cbscore
  score = cast(f_json_blob_get_value(:json, base_path || '.cbscore.cbsglocal.score_data.score_raw', 0) as d_integer);
  risk_level = f_json_blob_get_value(:json, base_path || '.cbscore.cbsglocal.score_data.score_range', '');
  risk_str = case
               when risk_level = 'A' then 'Очень низкий риск'
               when risk_level = 'B' then 'Очень низкий риск'
               when risk_level = 'C' then 'Низкий риск'
               when risk_level = 'D' then 'Низкий риск'
               when risk_level = 'E' then 'Средний риск'
               when risk_level = 'F' then 'Средний риск'
               when risk_level = 'G' then 'Высокий риск'
               when risk_level = 'H' then 'Высокий риск'
               when risk_level = 'I' then 'Очень высокий риск'
               when risk_level = 'J' then 'Очень высокий риск'
               else '--'
             end;

  rows_no = 0;
  h_qty = 0;
  closed_qty = 0;

  -- цикл по контрактам
  for
  select *
  from (
    select  ------------В рассрочку
        provider,
        ownership_type,
        start_date,
        end_date,
        borrowers,
        guarantors,
        loan_amount,
        currency,
        role_str,
        payments,
        duration,
        outstanding,
        interest_outstanding,
        loan_status,
        payment_status,
        max_past_due,
        is_our,
        interest_rate
    from get_contracts_ss('installments', :base_path, :bls_id, :c_name_fio)
    union all
    select  ----------------Не в рассрочку
        provider,
        ownership_type,
        start_date,
        end_date,
        borrowers,
        guarantors,
        loan_amount,
        currency,
        role_str,
        null,
        duration,
        null,
        null,
        loan_status,
        null,
        null,
        is_our,
        interest_rate
    from get_contracts_ss('non_installments', :base_path, :bls_id, :c_name_fio)
    union all
    select  -----------------На карту
        provider,
        ownership_type,
        start_date,
        end_date,
        borrowers,
        guarantors,
        loan_amount,
        currency,
        role_str,
        payments,
        duration,
        outstanding,
        interest_outstanding,
        loan_status,
        payment_status,
        max_past_due,
        is_our,
        interest_rate
    from get_contracts_ss('credit_cards', :base_path, :bls_id, :c_name_fio)
    union all
    select  -----------------На услуги
        provider,
        null,
        start_date,
        end_date,
        borrowers,
        guarantors,
        loan_amount,
        currency,
        role_str,
        null,
        duration,
        null,
        null,
        loan_status,
        payment_status,
        max_past_due,
        is_our,
        interest_rate
    from get_contracts_ss('services', :base_path, :bls_id, :c_name_fio)
    ) t
    order by
    case
      when loan_status containing 'Закры' then 1
      else 0
    end
  into :provider_inst,
       :own_type_inst,
       :start_date_inst,
       :end_date_inst,
       :borrowers_inst,
       :guarantors_inst,
       :loan_amount_inst,
       :currency_inst,
       :role_str_inst,
       :payment_amount_inst,
       :duration_inst,
       :outstanding_inst,
       :interest_outstanding_inst,
       :loan_status_inst,
       :payment_status_inst,
       :max_past_due_inst,
       :is_our_inst,
       :interest_rate
  do
  begin
    rows_no = rows_no + 1;

    if (loan_status_inst containing 'Закры') then
    begin
      current_closed = 1;
      closed_qty = closed_qty + 1;
    end
    else
     begin
      current_closed = 0;
     end
    
    if (current_closed = 0) then
      suspend;
    else
    begin
      if ((closed_qty <= 5) and (h_qty <= 7)) then
        suspend;
    end
    h_qty = h_qty + 1;
  end
  if (rows_no = 0) then
   suspend;
end
