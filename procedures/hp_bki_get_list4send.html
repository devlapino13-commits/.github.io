CREATE PROCEDURE HP_BKI_GET_LIST4SEND(
    PACKET_ID D_ID)
RETURNS (
    RES_STR D_STR_8100)
AS
declare variable bkc_id d_id;
declare variable bev_id d_id;
declare variable bkk_code d_str_code;
declare variable packet_name d_str_code;
declare variable packet_datetime d_datetime;
declare variable subjects_count d_integer;
declare variable records_count d_integer;
declare variable bkp_number d_integer;
declare variable bkp_status d_integer;
declare variable rec_no d_integer;
declare variable reset_rec_no d_integer;
declare variable event_id d_integer;
declare variable bkt_action_code d_str_code;
declare variable xml_version_id d_integer;
declare variable schema_version d_str_10;
declare variable event_code d_str_code;
declare variable bkc_add d_str_long;
declare variable uid d_str_code;
declare variable object_kind d_integer;
declare variable object_id d_id;
declare variable k_disb_date d_date;
declare variable jso d_integer;
declare variable bkt_format d_smallint;
begin
  xml_version_id = null;

  select bxv_id, bxv_schema_version
  from bki_xml_version
  where bxv_date_begin <= current_date
  order by bxv_date_begin desc
  rows 1
  into xml_version_id, schema_version;

  select bkk_code, bkp_code, bkp_send_datetime, bkp_number, bkp_status, f_enum('bki_packet_action', bkt_action, 'code'), bkt_format
  from bki_packet, bki_kind, bki_packet_type
  where bkp_id = :packet_id
    and bkk_id = bkp_from_bkk_id
    and bkt_id = bkp_from_bkt_id
  into bkk_code, packet_name, packet_datetime, bkp_number, bkp_status, bkt_action_code, bkt_format;

  -- шапка
  subjects_count = 0;
  records_count = 0;

  select count(distinct bkc_from_c_id), count(bev_id)
  from bki_loans, bki_clients, bki_events
  where bkl_from_bkp_id = :packet_id
    and bkc_from_bkl_id = bkl_id
    and bev_from_bkc_id = bkc_id
    and bev_status = 1
    and bev_event_id is not null
  into subjects_count, records_count;

  -- при переотправке (3.2), для какждого субъекта по одному события 3.2
  if (bkt_action_code in ('reset', 'recreate')) then
    records_count = subjects_count;

  res_str = '<?xml version="1.0" encoding="utf-8"?>' ||
            '<Document ' ||
              f_format('schemaVersion="%1" ', schema_version) ||
              'inn="7728744752" ogrn="1107746627779" ' ||
              f_format('sourceID="%1" ', trim(iif(bkk_code = 'ru.equifax', '47L', 'P701NN000001')))||
              f_format('regNumberDoc="%1" ', trim(iif(bkk_code = 'ru.equifax', lpad(bkp_number, 4, '0'), packet_name))) ||
              f_format('dateTimeDoc="%1" ', f_bki_datetime_to_str(packet_datetime, 3)) ||
              f_format('subjectsCount="%1" groupBlocksCount="%2">', subjects_count, records_count) ||
            '<Source><FL_46_UL_36_OrgSource><sourceCode>2</sourceCode><sourceRegistrationFact_1></sourceRegistrationFact_1>' ||
            '<fullName>ОБ�©ЕСТВО С ОГРАНИЧЕННОЙ ОТВЕТСТВЕННОСТЬЮ «МИКРОКРЕДИТНАЯ КОМПАНИЯ М БУЛАК»</fullName>' ||
            '<shortName>ООО «МКК М БУЛАК»</shortName><sourceDateStart>2010-08-09</sourceDateStart><regNum>1107746627779</regNum>' ||
            '<TaxNum_group_FL_46_UL_36_OrgSource><taxCode>1</taxCode><taxNum>7728744752</taxNum></TaxNum_group_FL_46_UL_36_OrgSource>' ||
            f_format('<sourceCreditInfoDate>%1</sourceCreditInfoDate>', cast(packet_datetime as date)) ||
            '</FL_46_UL_36_OrgSource></Source>' || trim(iif(bkt_format = 2, '<Data_AF>', '<Data>'));
  suspend;

  -- тело
  rec_no = 0;
  reset_rec_no = 0;

  for select distinct bkc_id, bkc_add, bkl_object_kind, bkl_object_id, f_bki_data_text(bkc_uid_bkd_id)
  from bki_loans, bki_clients, bki_events
  where bkl_from_bkp_id = :packet_id
    and bkc_from_bkl_id = bkl_id
    and bev_from_bkc_id = bkc_id
    and bev_status = 1
    and bev_event_id is not null
  order by bkl_id, bkc_id
  into bkc_id, bkc_add, object_kind, object_id, uid
  do
  begin
    -- Титульная часть
    reset_rec_no = reset_rec_no + 1;
    res_str = '<Subject_FL>' || (select res_xml from hp_bki_get_xml4send_equifax5(:xml_version_id, 1, :bkc_id, 'Title', 0));
    res_str = res_str || '<Events>';

    if (bkt_action_code in ('reset', 'recreate')) then
    begin
      -- запишем номер корректировочной записи
      jso = iif(coalesce(bkc_add, '') = '', json.obj_create(), json.parse(bkc_add));
      json.obj_set_s(jso, 'orderNum', reset_rec_no);
      bkc_add = json.obj_to_json(jso);
  
      update bki_clients
      set bkc_add = :bkc_add
      where bkc_id = :bkc_id;

      if (object_kind = f_objkind('miniapp', 'id')) then
        select k_disb_date
        from kontract
        where k_from_ma_app_id = :object_id
        order by k_status desc
        rows 1
        into k_disb_date;
      else
        select k_disb_date
        from kontract
        where k_id = :object_id
        into k_disb_date;

      res_str = res_str || f_format('<FL_Event_3_2 operationCode="C.1" orderNum="%1" eventDateTime="%2"', reset_rec_no, f_bki_datetime_to_str(packet_datetime, 3));
      if (bkt_action_code = 'reset') then
         res_str = res_str || f_format(' startDateTime="%1T00:00:00+03:00"', json.obj_get_val(jso, 'start_date', ''));

      res_str = res_str || f_format(' changeReason="%1"', json.obj_get_val(jso, 'change_reason', '1'));
      res_str = res_str || '>';

      res_str = res_str || f_format('<FL_17_DealUid><uid>%1</uid><openDate>%2</openDate></FL_17_DealUid>', uid, k_disb_date);
    end

    suspend;

    insert into bki_send_data (bsd_id, bsd_from_bkd_id, bsd_from_bfv_id, bsd_obj_kind, bsd_obj_id, bsd_number)
    select bsd_id, bsd_from_bkd_id, bsd_from_bfv_id, bsd_obj_kind, bsd_obj_id, bsd_number
    from bki_send_data_oper
    where bsd_obj_kind = 1
      and bsd_obj_id = :bkc_id;

    -- События
    for select bev_event_id, bev_id
    from bki_events
    where bev_from_bkc_id = :bkc_id
      and bev_status = 1
      and bev_event_id is not null
    order by bev_event_date, bev_id
    into event_id, bev_id
    do
    begin
      rec_no = rec_no + 1;

      update bki_events
      set bev_rec_no = :rec_no
      where bev_id = :bev_id;

      event_code = f_enum('bki_event', event_id, 'code');
      event_code = left(event_code, position('_', event_code) - 1);

      if (event_code in ('3.1')) then
        res_str = (select res_xml from hp_bki_corr_get_xml_equifax(:xml_version_id, :bev_id, 'Event_' || :event_code, :rec_no));
      else
        res_str = (select res_xml from hp_bki_get_xml4send_equifax5(:xml_version_id, 2, :bev_id, 'Event_' || :event_code, :rec_no));

      insert into bki_send_data (bsd_id, bsd_from_bkd_id, bsd_from_bfv_id, bsd_obj_kind, bsd_obj_id, bsd_number)
      select bsd_id, bsd_from_bkd_id, bsd_from_bfv_id, bsd_obj_kind, bsd_obj_id, bsd_number
      from bki_send_data_oper
      where bsd_obj_kind = 2
        and bsd_obj_id = :bev_id;
  
      if (bkp_status = 1) then
        update bki_events
        set bev_status = 2
        where bev_id = :bev_id;

      suspend;
    end

    res_str = '';
    if (bkt_action_code in ('reset', 'recreate')) then
      res_str = res_str || '</FL_Event_3_2>';

    res_str = res_str || '</Events></Subject_FL>';
    suspend;
  end
  res_str =  trim(iif(bkt_format = 2, '</Data_AF>', '</Data>')) || '</Document>';
  suspend;

  -- пакет забран
  if (bkp_status = 1) then
    update bki_packet bkp
    set bkp.bkp_status = 2,
        bkp.bkp_result_status = 'Taken'
    where bkp.bkp_id = :packet_id;
end
