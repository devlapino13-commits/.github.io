CREATE PROCEDURE FIND_BANKS(
    FIND_TXT VARCHAR(100))
RETURNS (
    B_ID VARCHAR(10),
    B_NAME VARCHAR(255),
    B_SWIFT VARCHAR(20),
    B_ADDRESS VARCHAR(255),
    B_MFO_OLD VARCHAR(10),
    IS_DELETE SMALLINT,
    B_CORR_ACC VARCHAR(255),
    B_PHONE VARCHAR(255),
    B_DATE_CHANGE DATE)
AS
begin
  for select b_id, b_name, b_swift, b_address, b_mfo_old, is_delete, b_corr_acc, b_phone, b_date_change
      from banks, find_in_workdoc_words('banks', :find_txt, 1)
      where b_id = wdlink_object_id
      into :b_id, :b_name, :b_swift, :b_address, :b_mfo_old, :is_delete, :b_corr_acc, :b_phone, :b_date_change
  do
  begin
    suspend;
  end
end
