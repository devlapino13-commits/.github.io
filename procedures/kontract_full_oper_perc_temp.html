CREATE PROCEDURE KONTRACT_FULL_OPER_PERC_TEMP(
    KONT_ID D_ID,
    DD D_DATE)
RETURNS (
    OL_DATE D_DATE,
    SUM_OF_OK_OS D_CURRENCY,
    SUM_OF_OK_PERC D_CURRENCY,
    SUM_OF_OK_FINE D_CURRENCY,
    SUM_OF_OK_KOMMIS D_CURRENCY,
    SUM_OF_OK_KOMMIS_ADV D_CURRENCY,
    SUM_OF_ALL D_CURRENCY,
    OL_IS_END_MONTH D_INTEGER,
    OPER_TYPE D_INTEGER,
    OPER_KIND D_STR_CODE)
AS
declare variable min_dd d_date;
declare variable ye d_integer;
declare variable mo d_integer;
declare variable dd_month_end d_date;
begin
  ol_is_end_month = 0;
  for
    select
      ok_ol_date,
      ot_name,
      ok_type,
      sum(ok_os),
      sum(ok_perc),
      sum(ok_fine),
      sum(ok_kommis),
      sum(ok_kommis_adv),
      sum(ok_sh_os + ok_os + ok_sh_perc + ok_perc + ok_fine + ok_kommis_adv)
    from operkred
      inner join kroper
        on ot_id = ok_type
    where ok_from_k_id = :kont_id
      and ok_ol_date <= :dd
      and ok_ol_is_active = 1
      and ok_type >= 4
      and ok_type <> 7
      and ok_from_c_id = 0
    group by ok_ol_date, ot_name, ok_type
    order by ok_ol_date, ot_name
  into
    :ol_date,
    :oper_kind,
    :oper_type,
    :sum_of_ok_os,
    :sum_of_ok_perc,
    :sum_of_ok_fine,
    :sum_of_ok_kommis,
    :sum_of_ok_kommis_adv,
    :sum_of_all
  do
  begin
    if (min_dd is null) then
      min_dd = ol_date;
    suspend;
  end

  ol_is_end_month = 1;
  sum_of_ok_os = 0;
  sum_of_ok_perc = 0;
  sum_of_ok_fine = 0;
  sum_of_ok_kommis = 0;
  sum_of_ok_kommis_adv = 0;
  sum_of_all = 0;
  if (min_dd is not null) then
  begin
    ye = extract(year from min_dd);
    mo = extract(month from min_dd);
    dd_month_end = min_dd;
    while (dd_month_end <= :dd) do
    begin
      if (mo < 12) then
      begin
        dd_month_end = cast(('01.' || cast((mo + 1) as varchar(2)) || '.' || cast(ye as varchar(4))) as date) - 1;
      end
      else
      begin
        dd_month_end = cast('31.12.' || cast(ye as varchar(4)) as date);
      end
      if (mo < 12) then
      begin
        mo = mo + 1;
      end
      else
      begin
        mo = 1;
        ye = ye + 1;
      end
      ol_date = dd_month_end;
      if (ol_date > :dd) then
        ol_date = dd;
      suspend;
    end
  end
end
