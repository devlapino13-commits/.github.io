CREATE PROCEDURE HP_FLOATING_CHARTS(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_ERROR_TEXT D_STR_255,
    RES_INFO_TEXT D_STR_255)
AS
declare variable k_status d_integer;
declare variable ok_from_ol_id d_id;
declare variable sh_date d_date;
declare variable first_update_percent d_date;
declare variable second_update_percent d_date;
begin
  -- RFCKG-3422 Команда выполнить для модуля по "Тестирование плавающих графиков"

  res_error_text = '';
  res_info_text = '';

  if (exists(select * from hp_db_is_actual)) then
  begin
    res_error_text = 'БД дожна быть тестовой';
    exit;
  end

  select k.k_status
  from kontract k
  where k.k_id = :k_id
  into k_status;

  select ok.ok_from_ol_id
  from operkred ok
  where ok.ok_from_k_id = :k_id
    and ok.ok_type = 8 --Выдача
    and ok_ol_is_active = 1
  into ok_from_ol_id;

  if (k_status = 1 and ok_from_ol_id is not null ) then   -- если контракт активный и была выдача
  begin
       select sh.sh_date
       from schedule sh
       where sh.sh_from_k_id = :k_id    --Дата ближайшего погашения
         and sh.sh_date >= current_date
       order by sh.sh_date
       rows 1
       into sh_date;

     if (sh_date is not null) then
     begin
       first_update_percent = f_date_work_nday(sh_date, 3);
       second_update_percent = f_date_work_nday(sh_date, 8);

       execute procedure hp_do_reschedule_float(k_id, first_update_percent, un_id);
       execute procedure hp_do_reschedule_float(k_id, second_update_percent, un_id);
       res_info_text = f_format('Выполнено'|| ascii_char(13) || ascii_char(10) ||'Дата ближайшего погашения: %1'|| ascii_char(13) || ascii_char(10) ||'Первое повышение плавающей процентной ставки: %2'|| ascii_char(13) || ascii_char(10) ||'Второе повышение  плавающей процентной ставки: %3',
       f_datetostr(sh_date), f_datetostr(first_update_percent), f_datetostr(second_update_percent));
     end
  end
  suspend;
end
