CREATE PROCEDURE HP_BIOGUARD_UPDATE_STATUS(
    BR_ID D_ID NOT NULL,
    EIL_ID D_ID NOT NULL,
    RD_CODE D_STR_255 NOT NULL,
    C_ID D_ID = null)
AS
declare variable EIL_STATUS D_SMALLINT;
begin
    -- Ранее по процессу ответ уже был сохраняен в процедуре hp_bioguard_save_response
    -- и eil_status на данном моменте равен 2
    update extinfolist
    set eil_from_c_id = :c_id,
        eil_object_kind = f_objkind('clients', 'id'),
        eil_object_id = :c_id
    where eil_id = :eil_id;

    -- Сохраняем уже признак успешного и не успешного ответа
    -- Все статусы, которые сюда могут попасть находятся в таблице refdata -> rd_main_code = 'BioguardStatus'
    update bioguard_results
    set br_from_c_id = :c_id, 
        br_status = f_get_rd_id(:rd_code),
        br_last_datetime = current_timestamp
    where br_id = :br_id;
end
