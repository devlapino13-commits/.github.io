CREATE PROCEDURE UXP$UXSOURCE_SYNC(
    US_UUID D_UUID,
    AUTO_APPLY D_BOOL = 0,
    US_UUID_CODE D_STR_CODE = null)
AS
declare variable us_source d_str_code;
declare variable us_kind d_smallint;
declare variable res_memo d_str_8100;
declare variable us_is_view d_bool;
declare variable field_name d_str_code;
declare variable uf_uuid d_uuid;
declare variable table_name d_str_code;
declare variable view_field d_str_code;
declare variable rel_type d_bool;
declare variable us_is_table d_smallint;
declare variable view_name d_str_code;
declare variable view_uuid d_uuid;
begin
  -- Сихронизирует индексы БД с таблицей uxIndex

  if (coalesce(us_uuid, '') = '') then
    us_uuid = uxf$uniq(us_uuid_code, 0);

  if (auto_apply is null) then
    auto_apply = 0;

  select us_code, us_kind, us_is_table, us_is_view
  from ux$source
  where us_uuid = :us_uuid
  into us_source, us_kind, us_is_table, us_is_view;

  if (us_kind <> 0) then
    exit;

  select trim(rdb$description), r.rdb$relation_type
  from rdb$relations r
  where r.rdb$system_flag = 0 and
        r.rdb$relation_type in (0, 1) and
        r.rdb$relation_name = upper(:us_source)
  into res_memo, rel_type;

  if (res_memo is null) then
    res_memo = '';

  if (rel_type is null) then
    rel_type = 0;

  update ux$source
  set us_memo = :res_memo,
      us_is_view = :rel_type
  where us_uuid = :us_uuid
  returning us_is_view
  into us_is_view;

  execute procedure uxp$uxsource_sync_params(:us_uuid, :auto_apply);

  if (us_is_view = 0) then
    execute procedure uxp$uxsource_sync_index(:us_uuid);
  else
  begin
    table_name = '';
    select trim(rdb$relation_name)
    from rdb$view_relations
    where rdb$view_name = upper(:us_source) and
          rdb$context_type = 0
    order by rdb$view_context
    rows 1
    into :table_name;

    if (table_name <> '') then
    begin
      for select lower(trim(i.rdb$field_name))
      from rdb$relation_constraints r
      inner join rdb$index_segments i on i.rdb$index_name = r.rdb$index_name
      inner join rdb$relation_fields rf on rf.rdb$field_name = i.rdb$field_name and
            rf.rdb$relation_name = r.rdb$relation_name
      inner join rdb$fields f on f.rdb$field_name = rf.rdb$field_source
      where r.rdb$constraint_type = 'PRIMARY KEY' and
           r.rdb$relation_name = upper(:table_name)
      into field_name
      do
      begin
        for select lower(trim(rdb$field_name))
        from rdb$relation_fields
        where rdb$relation_name = upper(:us_source) and
              rdb$base_field = upper(:field_name)
        into view_field
        do
        begin
          for select uf_uuid
          from ux$field
          where uf_owner = :us_uuid and
                uf_field_name = :view_field
          into uf_uuid
          do
            execute procedure uxp$uxfield_set_keyfield(:uf_uuid);
        end
      end
    end

    update ux$field
    set uf_origin_uf_uuid = uf_origin_uf_uuid
    where uf_owner = :us_uuid and
          uf_origin_uf_uuid is null;
  end

  execute procedure uxp$uxsource_diff_sync(:us_uuid);

  if (us_is_table = 1 and us_is_view = 0) then
  begin
    for select lower(trim(rdb$view_name))
    from rdb$view_relations
    where rdb$relation_name = upper(:us_source)
    into view_name
    do
    begin
      for select us_uuid
      from ux$source
      where us_code = :view_name
      into view_uuid
      do
        execute procedure uxp$uxsource_sync(:view_uuid, 1);
    end
  end
end
