CREATE PROCEDURE MC_CLI_EXTMINIAPP_LIST(
    MCL_ID D_ID)
RETURNS (
    OBJECT_KIND STR_100,
    OBJECT_ID D_ID,
    AMOUNT D_STR_100,
    PERIOD D_STR_100,
    POINT_1_TEXT D_STR_100,
    POINT_1_DATE D_STR_100,
    POINT_1_STATE D_SMALLINT,
    POINT_2_TEXT D_STR_100,
    POINT_2_DATE D_STR_100,
    POINT_2_STATE D_SMALLINT,
    POINT_3_TEXT D_STR_100,
    POINT_3_DATE D_STR_100,
    POINT_3_STATE D_SMALLINT,
    SHOW_REJECTION_BUTTON D_BOOL)
AS
declare variable c_id d_id;
declare variable send_date d_date;
declare variable ema_status d_smallint;
declare variable ema_id d_id;
declare variable ma_is_active d_smallint;
declare variable ma_id d_integer;
declare variable ema_lo_id d_id;
declare variable ema_from_c_id d_id;
declare variable tkt_id d_id;
declare variable ema_lo_block_date d_date;
declare variable ema_status_date d_date;
declare variable ma_date_save d_date;
declare variable days_for_show_negative_status d_smallint;
declare variable create_date_limit d_date;
declare variable ema_result_rd_id d_integer;
declare variable db_id d_integer;
declare variable lang_code d_str_lang;
declare variable c_name_fio_officer varchar(100);
declare variable cell_main_officer varchar(50);
declare variable pr_id d_integer;
declare variable pr_code d_str_100;
declare variable mcl_project d_str_20;
begin
  db_id = f_get_db_id();
  lang_code = f_get_lang_code(null, :mcl_id);

  days_for_show_negative_status = 7; -- сколько дней мы продолжаем показвать инфу по заявке, если по ней негативный статус.
  create_date_limit = current_date - 30; -- если заявка создана до этой даты, тогда ее не рассматриваем. Данный ограничитель для ограничения выборки и анализа данных

  if (coalesce(mcl_id, 0) <= 0) then
    exception e_default 'parametr "mcl_id" has invalid value';

  select
    mcl.mcl_from_c_id,
    f_enum('mcl_project', mcl.mcl_project, 'code', 0)
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into
    :c_id,
    :mcl_project;

  -- за основу берем таблицу онлайн-заявок extminiapp------ >
  for select
    round(iif(maa.ma_id is null, ema.ema_amount, maa.ma_summa_minus), 2),
    iif(maa.ma_id is null, ema.ema_period, maa.ma_kolich_repay),
    ema.ema_date,
    ema.ema_status,
    ema.ema_id,
    maa.ma_is_active,
    maa.ma_id,
    ema.ema_lo_id,
    ema.ema_block_datetime,
    cast(ema.ema_status_datetime as date),
    cast(maa.ma_date_save as date),
    ema.ema_result_rd_id,
    ema.ema_from_c_id,
    coalesce(ema.ema_from_pr_id, maa.ma_from_pr_id)
  from extminiapp ema
  left join miniapp maa on ema.ema_from_tkt_id = maa.ma_from_tkt_id
    and maa.ma_date_end = '31.12.9999'
  left join miniappclient macl on maa.ma_id = macl.mac_from_ma_id
    and macl.mac_amount > 0
  where ema.ema_from_mcl_id = :mcl_id
    and ema.ema_date >= :create_date_limit
  into
    :amount,
    :period,
    :send_date,
    :ema_status,
    :ema_id,
    :ma_is_active,
    :ma_id,
    :ema_lo_id,
    :ema_lo_block_date,
    :ema_status_date,
    :ma_date_save,
    :ema_result_rd_id,
    :ema_from_c_id,
    :pr_id
  do
  begin
    select pr.pr_code
    from product pr
    where pr.pr_id = :pr_id
    into :pr_code;

    -- RFCRU-3766
    if (mcl_project = 'kmp' and pr_code <> '' and pr_code containing '.oivo_pack.') then
      continue;

    if (mcl_project = 'oivo' and pr_code <> '' and pr_code not containing '.oivo_pack.') then
      continue;


    select cl.c_name_fio, f_cont_info('clients', cl.c_id, 'cell_main', 'ok')
    from clients cl
    where cl.c_id = :ema_lo_id
    into :c_name_fio_officer, :cell_main_officer;

    -- даннное условие нужно, т.к. если заявка превратилась в договор, который был выдан, эта заявка уже не должна светиться среди заявок в процессе
    if (exists(select *
               from kontract kont
               inner join kontstate ks on ks.ks_from_k_id = kont.k_id
               where kont.k_from_ma_app_id = :ma_id and
                     ks.ks_state = 1 and
                     ks.ks_ol_is_delete = 0)) then
      continue;

    point_1_text = f_lang(lang_code, 'Отправлена');
    point_1_date = f_datetostr(send_date);
    point_1_state = 1;

    -- это значения по умолчанию --------
    point_2_text = f_lang(lang_code, 'На обработке');
    point_2_date = '';

    point_3_text = f_lang(lang_code, 'Решение');
    point_3_date = '';
    ------------------------------------

    if (ma_id is null) then -- если кредитная заявка не была найдена
    begin
      object_kind = 'extminiapp';
      object_id = ema_id;

      ------------------------- когда заявка ушла в обработку -->
      point_2_date = '';
      point_2_state = 0;
      if ((ema_lo_id is not null) or (ema_status <> 0)) then
      begin
        point_2_date = f_datetostr(ema_lo_block_date);
        point_2_state = 1;
      end
      -------------------------- <
      -------------------------- когда заявка ушла в отказ -->
      point_3_date = '';
      point_3_state = 0;
      if (ema_status = -1) then
      begin
        point_3_date = f_datetostr(ema_status_date);
        point_3_state = -1;

        if (datediff(day, ema_status_date, current_date) > days_for_show_negative_status) then -- если по заявке был отказ более X дней назад, то прекращаем ее показ
          continue;

        if (ema_result_rd_id = f_get_rd_id('website_miniapp.Refuse.MiniappCreateCancel')) then -- если онланй-заявка была аннулироавана ввиду подачи кредитной заявки, аннулированную онлайн-заявку я не показваю, чтобы она не влияла на отказ.
          continue;
      end
      -------------------------- <
    end
    else -- если кредитная заявка найдена, значит заявка уже в обработке
    begin
      object_kind = 'miniapp';
      object_id = ma_id;

      point_2_state = 1;
      point_2_date = null;

      -- что считаем обработкой? Приоритет отдает блокировке заявки на уровне extminiapp, если там блокировки не было но есть кредитная заявка, за начало обработки берем дату формирвоания этой заявкию
      if ((ema_lo_id is not null) or (ema_status <> 0)) then
        point_2_date = f_datetostr(ema_lo_block_date);

      if (point_2_date is null) then
        point_2_date = f_datetostr(ma_date_save);
      ------------------------- >
      point_3_state = 0;

      if (ma_is_active < 1) then -- если на этапе договора или заявки был отказ, тогда негативный статус
      begin
        point_3_date = f_datetostr(ma_date_save);
        point_3_state = -1;

        if (datediff(day, ma_date_save, current_date) > days_for_show_negative_status) then -- если по заявке был отказ более X дней назад, то прекращаем ее показ
          continue;
      end
    end

    amount = replace(amount, '.00', '');
    amount = amount || iif(db_id = 41701, ' сом.', iif(db_id = 39801, ' тенге', ' руб.'));
    period = period || ' ' || f_lang(lang_code, 'мес') || '.';

    if (ema_lo_id is not null and
        (point_2_state = 1) and (ema_from_c_id is not null)) then
    begin
      if (not exists(select *
                     from fsrobot_msgbox f
                     where f.msg_body containing f_lang(:lang_code, 'Вашу заявку принял в работу специалист') and
                           f.msg_object_id = :mcl_id)) then
      begin
        execute procedure mc_send_push('clients', :ema_from_c_id,
                                       f_lang(:lang_code, 'Заявка на рассмотрении'),
                                       f_format(f_lang(:lang_code, 'Вашу заявку принял в работу специалист %1. Свяжется с вами в течение ближайшего рабочего дня. По вопросам можете связаться по номеру: %2'), :c_name_fio_officer, :cell_main_officer)
                                      );
      end
    end

    show_rejection_button = f_freeze_get_reject_status(ma_id);

    suspend;
  end
  ---------------------------------------------------------------------------- <
  --- Поднимаем кредитные заявки пользователя МП, которые не связаны с онлайн-заявками -->
  for select
    round(maa.ma_summa_minus, 2),
    maa.ma_kolich_repay,
    maa.ma_is_active,
    maa.ma_id,
    maa.ma_from_tkt_id,
    cast(maa.ma_date_save as date),
    maa.ma_from_pr_id
  from miniapp maa
  inner join miniappclient macl on maa.ma_id = macl.mac_from_ma_id
  where macl.mac_from_c_id = :c_id
    and macl.mac_amount > 0
    and maa.ma_date_end = '31.12.9999'
    and maa.ma_date_app >= :create_date_limit
  into
    :amount,
    :period,
    :ma_is_active,
    :ma_id,
    :tkt_id,
    :ma_date_save,
    :pr_id
  do
  begin
    select pr.pr_code
    from product pr
    where pr.pr_id = :pr_id
    into :pr_code;

    -- RFCRU-3766
    if (mcl_project = 'kmp' and pr_code <> '' and pr_code containing '.oivo_pack.') then
      continue;

    if (mcl_project = 'oivo' and pr_code <> '' and pr_code not containing '.oivo_pack.') then
      continue;

    -- если есть онлайн-заявка, дельше не идем т.к. данная заявка уже обрабатывалась в скрипте выше
    if (exists(select *
               from extminiapp ema
               where ema.ema_from_tkt_id = :tkt_id)) then
      continue;

    -- если есть активный договор, дельше не идем т.к. если заявка превратилась в договор,
    -- который был выдан, эта заявка уже не должна светиться среди заявок в процессе
    if (exists(select *
               from kontract kont
               inner join kontstate ks on ks.ks_from_k_id = kont.k_id
               where kont.k_from_ma_app_id = :ma_id and
                     ks.ks_state = 1 and
                     ks.ks_ol_is_delete = 0)) then
      continue;

    object_kind = 'miniapp';
    object_id = ma_id;

    -- получаю дату когда заявка была создана
    point_1_text = f_lang(lang_code, 'Отправлена');
    point_1_state = 1;
    select f_datetostr(maa.ma_date_save)
    from miniapp maa
    where maa.ma_id = :ma_id
    order by maa.ma_number
    rows 1
    into :point_1_date;
    ------------------------- >
    -- т.к. заявка была создана не онлайн значит она сразу перещла в обработку
    point_2_text = f_lang(lang_code, 'На обработке');
    point_2_state = 1;
    point_2_date = point_1_date;
    ------------------------- <
    ------------------------- >
    point_3_text = f_lang(lang_code, 'Решение');
    point_3_state = 0;
    point_3_date = '';

    if (ma_is_active <= 0) then
    begin
      point_3_date = f_datetostr(ma_date_save);
      point_3_state = -1;

      if (datediff(day, ma_date_save, current_date) > days_for_show_negative_status) then -- если по заявке был отказ более X дней назад, то прекращаем ее показ
        continue;
    end
    ------------------------- <
    amount = replace(amount, '.00', '');
    amount = amount || iif(db_id = 41701, ' сом.', iif(db_id = 39801, ' тенге', ' руб.'));
    period = period || ' ' || f_lang(lang_code, 'мес') || '.';

    show_rejection_button = f_freeze_get_reject_status(ma_id);

    suspend;
  end
end
