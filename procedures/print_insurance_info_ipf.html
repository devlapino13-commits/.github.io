CREATE PROCEDURE PRINT_INSURANCE_INFO_IPF(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    CLIENT_FULL_NAME D_STR_255,
    BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_64,
    PASSPORT_ISSUED_BY D_STR_255,
    C_PASS_DATE D_DATE,
    LEGAL_ADDRESS D_STR_255,
    PHONE_NUMBER D_STR_100,
    WORK_PLACE D_STR_255,
    INSURANCE_AMOUNT D_CURRENCY,
    CONTRACT_START_DATE D_DATE,
    CONTRACT_END_DATE D_DATE,
    APPLICATION_DATE D_DATETIME,
    QR_CODE D_STR_LONG,
    K_DISB_DATE D_DATE,
    K_KONTRACT_NO D_STR_20,
    IK_COMMISSION D_CURRENCY,
    SH_DATE D_DATE)
AS
declare variable c_id d_id;
declare variable edl_id d_id;
declare variable tmp_id d_id;
declare variable ma_id d_id;
declare variable mac_id d_id;
begin

    select cl.c_id,
           cl.c_name_fio,
           cl.c_birth_date,
           cl.c_pass_no,
           cl.c_pass_who,
           cl.c_pass_date,
           f_cont_info('clients', c_id, 'cell_main', 'ok'),
           kc.kc_amount_approved,
           kn.k_kontract_date,
           kn.k_kredline_end,
           kn.k_from_ma_app_id,
           kn.k_kontract_no,
           kn.k_disb_date
    from kontract kn
    inner join knclient kc on kc.kc_from_k_id = kn.k_id
    inner join clients cl on cl.c_id = kc.kc_from_c_id
    inner join product pr on pr.pr_id = kn.k_from_pr_id
    where kn.k_id = :k_id
        and kc.kc_amount_approved > 0
    into :c_id,
         :client_full_name,
         :birth_date,
         :c_pass_no,
         :passport_issued_by,
         :c_pass_date,
         :phone_number,
         :insurance_amount,
         :contract_start_date,
         :contract_end_date,
         :ma_id,
         :k_kontract_no,
         :k_disb_date;

    if ((contract_end_date - contract_start_date) < 365) then
     contract_end_date = dateadd(year, 1, contract_start_date);

    select adl.adl_address_str
    from addrclient adc
    join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where adc.adc_from_c_id = :c_id and
          adc.adc_address_type = 1
    order by adc.adc_datetime desc
    rows 1
    into :legal_address;

    if (not exists(select * from clients
    where c_id = :c_id)) then
    begin
        tmp_id = f_json_str_get_value(add_params, 'c_id', null);
        if (tmp_id is not null) then
        c_id = tmp_id;
    end

    if (coalesce(add_params, '') <> '') then
    begin
        edl_id = f_json_str_get_value(add_params, 'edl_id', null);
        qr_code = f_edoc_qr_by_edl(edl_id, c_id);
        
        select edl.edl_create_datetime
        from edoc_list edl
        where edl.edl_id = :edl_id
        into :application_date;
    end

    select mac_id
    from miniappclient mac
    where mac.mac_from_c_id = :c_id
        and mac.mac_from_ma_id = :ma_id
    into :mac_id;

    select cp.cp_name_ru
    from cfp_positions cp
    join cfp_anketa ca on ca.ca_from_cp_id = cp.cp_id
    where ca.ca_from_c_id = :c_id
        and ca.ca_object_kind = f_objkind('miniappclient', 'id')
        and ca.ca_object_id = :mac_id
    into :work_place;

    ik_commission = 0;
    select coalesce(ik.ik_commission, 0)
    from insurance_kontract ik
    where ik.ik_from_k_id = :k_id
    into :ik_commission;

    select sh.sh_date
    from schedule sh
    where sh.sh_from_k_id = :k_id
    and sh.sh_id = 1
    into :sh_date;

    suspend;
end
