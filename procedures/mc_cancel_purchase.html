CREATE PROCEDURE MC_CANCEL_PURCHASE(
    MCL_ID D_ID,
    MP_ID D_ID,
    USER_CODE D_INTEGER)
RETURNS (
    ERROR_TXT D_STR_100,
    IS_CANCELLED D_INTEGER)
AS
declare variable other_c_id d_id;
declare variable c_id d_id;
declare variable subject d_str_255;
declare variable msg_body varchar(500);
begin
  error_txt = '';
  select mcl.mcl_from_c_id
  from mc_login mcl
  where mcl.mcl_id = :mcl_id
  into :c_id;

  if (exists(select *
             from mc_purchase mp
             where mp.mp_id = :mp_id and
                   mp.mp_status = 1)) then
  begin
    update mc_purchase mp
    set mp.mp_status = f_enum_id('mc_purchase_status', 'canceled', null)
    where mp.mp_id = :mp_id;

    is_cancelled = 1;
  end
  else
  begin
    error_txt = 'Покупка не может быть отменена';
    is_cancelled = 0;
    suspend;
    exit;
  end

  if (user_code = 0) then -- Если клиент
  begin
    subject = 'Отмена рассрочки';
    msg_body = 'Вы отменили оформление рассрочки.' || ascii_char(10) || ascii_char(10) || ascii_char(10) || 'При необходимости вы можете еще оформить' || ascii_char(10) || 'рассрочку' || ascii_char(10);
    execute procedure mc_send_push('clients', :c_id, :subject, :msg_body);

    select cs.cs_from_c_id
    from mc_purchase mp
    inner join clientshops cs on cs.cs_id = mp.mp_from_cs_id
    where mp.mp_id = :mp_id
    into :other_c_id;

    subject = 'Отмена покупки';
    msg_body = 'Клиент отменил покупку товара в рассрочку';
    execute procedure mc_send_push('clients', :other_c_id, :subject, :msg_body);

  end
  else --Если агент
  begin
    subject = 'Отмена продажи в рассрочку';
    msg_body = 'Подтверждая отмену продажи товара' || ascii_char(10) || 'в рассрочку вы подтверждаете, что не продали' || ascii_char(10) || 'товар и поэтому не получите возмещение' || ascii_char(10) || 'денежных средств от Компании М Булак';
    execute procedure mc_send_push('clients', :c_id, :subject, :msg_body);
  end
  suspend;
end
