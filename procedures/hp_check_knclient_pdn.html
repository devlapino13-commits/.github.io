CREATE PROCEDURE HP_CHECK_KNCLIENT_PDN(
    K_ID D_ID)
RETURNS (
    KF_CLIENT_FIO D_STR_100,
    KF_ERROR_TEXT D_STR_255,
    KF_ERROR_IS_CRITICAL D_SMALLINT)
AS
declare variable kc_id d_id;
declare variable kc_status d_integer;
begin
  kf_error_is_critical = 1;

  for select kc_id, c_name_fio
  from knclient, clients
  where kc_from_k_id = :k_id and
        kc_from_c_id = c_id and
        kc_amount_approved > 0
  into kc_id, kf_client_fio
  do
  begin
    kf_error_text = '';
    kc_status = null;

    select kc_status
    from knclient_pdn
    where kc_id = :kc_id
      and kcp_kind = 0
    into kc_status;

    if (coalesce(kc_status, -1) = -1) then
    begin
      kf_error_text = 'По клиенту отсутствует расчет ПДН';
      suspend;
    end
  end
end
