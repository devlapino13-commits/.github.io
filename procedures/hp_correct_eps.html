CREATE PROCEDURE HP_CORRECT_EPS(
    REP_DATE D_DATE)
AS
declare variable kd_from_ol_id d_id;
declare variable kd_id d_id;
declare variable k_id d_id;
declare variable effect_rate_common decimal(18,4);
begin
  for select k.k_id
      from kontract k
      inner join kontstate ks on ks.ks_from_k_id = k.k_id
      inner join product pr on pr.pr_id = k.k_from_pr_id and
            pr.pr_code not containing '.ipf.' and
            pr.pr_mbulak_cards_kind = 1
      where
            ks.ks_date_begin = :rep_date and
            ks.ks_ol_is_delete = 0 and
            ks.ks_state = 1
      into k_id
  do
  begin
    effect_rate_common = 0;
    execute procedure hp_effect_rate_fill_tmp(:k_id, null, 0);

    select calc_eff
    from hp_effect_rate_kg(:k_id)
    into effect_rate_common;

    update kontract k
    set k.k_percent_effect = :effect_rate_common,
        k.k_percent_effect_fifd = :effect_rate_common,
        k.k_percent_market = :effect_rate_common
    where k.k_id = :k_id;

    update kontdata kd
    set kd.kd_percent_effect = :effect_rate_common,
        kd.kd_percent_effect_fifd = :effect_rate_common
    where kd.kd_from_k_id = :k_id and
          kd.kd_ol_is_delete = 0 and
          kd.kd_float_perc_level in (0, 1, 2);

    if ((select count(*)
         from kontdata kd
         where kd.kd_from_k_id = :k_id) >= 2) then
    begin
      for select kd.kd_id, kd.kd_from_ol_id
          from kontdata kd
          where kd.kd_from_k_id = :k_id and
                kd.kd_ol_is_delete = 0 and
                kd.kd_float_perc_level > 0
          order by kd_float_perc_level
          into kd_id, kd_from_ol_id
      do
      begin
        effect_rate_common = 0;
        execute procedure hp_effect_rate_fill_tmp(:k_id, :kd_from_ol_id, 0);

        select calc_eff
        from hp_effect_rate_kg(:k_id)
        into effect_rate_common;

        update kontdata kd
        set kd.kd_percent_effect_fifd = :effect_rate_common
        where kd.kd_from_k_id = :k_id and
              kd.kd_id = :kd_id;
      end
    end
  end
end
