CREATE PROCEDURE AAREP_DTI(
    REP_REGION_ID ID_TYPE,
    K_ID_IN ID_TYPE,
    REP_DATE DATE,
    REP_TYPE D_STR_100 = 'active',
    REP_DATE_BEGIN DATE = null)
RETURNS (
    CLIENT_FIO VARCHAR(100),
    K_KONTRACT_NO VARCHAR(50),
    PR_NAME VARCHAR(100),
    OFFICE_NAME VARCHAR(100),
    KC_DISB_OFFICER_NAME VARCHAR(100),
    KC_OFFICER_NAME VARCHAR(100),
    KF_DISB_DATE DATE,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    KC_AMOUNT DECIMAL(18,4),
    KF_NO VARCHAR(100),
    KC_CIRCLE INTEGER,
    K_BALANCE_OS CURR_TYPE,
    INCOME_TYPE VARCHAR(255),
    JOB_NAME VARCHAR(255),
    PMT CURR_TYPE,
    DTI_D CURR_TYPE,
    DTI_I CURR_TYPE,
    DTI_INCOME CURR_TYPE,
    DTI PERC_TYPE,
    DTI_TXT STR_255)
AS
declare variable KS_STATE integer;
declare variable PR_ID integer;
declare variable RD_ID integer;
declare variable K_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable OKS_KC_ID ID_TYPE;
declare variable K_FROM_PR_ID integer;
declare variable OFFICE_ID ID_TYPE;
declare variable REG_ID ID_TYPE;
declare variable K_FACT_DISB_LO_ID ID_TYPE;
declare variable RD_NAME STR_255;
declare variable RD_CODE STR_255;
declare variable OVS_VALUE STR_255;
declare variable OVS_VALUE_NO STR_255;
declare variable CL_PROPORTION PERC_TYPE;
declare variable SQL_TEXT D_STR_2048;
begin
  /*3.    Есть необходимость в создании отдельного отчёта, с нижеуказанными полями:

•    ФИО клиента -
•    № контракта -
•    Кредитный продукт -
•    Офис выдачи -
•    Кредитный специалист, выдавший кредит
•    Дата выдачи
•    Сумма выдачи
•    № кредитной папки
•    Цикл Клиента
•    Остаток задолженности по ОС
•    Gəlirin növü (Вид дохода – это инф. из анкетных данных клиента)
•    Vəzifəsi (Должность – это инф. из анкетных данных клиента)
•    DTI_D (Müştərinin borc yükü) --финансовая загруженность клиента 
•    DTI_I  (Müştərinin xalis qəliri (Vergi tutulmaları xaric)) - чистый доход клиента (после вычета налогов)
•    Доход (Müştərinin aylıq gəliri (Vergi tutulmaları daxil))
•    PMT (месячное погашение по кредиту)
•    DTI_TXT
*/
  if (coalesce(k_id_in, 0) < 101010000000100101) then k_id_in = null;
  if (rep_region_id is null) then rep_region_id = 0;
  if (rep_date is null) then rep_date = '01.01.1900';

  if (rep_type = 'active') then
  begin
      if (k_id_in is null) then
      begin
        sql_text = '
        select k.k_last_reg_id,
               k_disb_date,
               k_id,
               k_kontract_no,
               k_from_pr_id,
               k.k_fact_disb_lo_id,
               k.k_fact_disb_kf_no,
               k_amount_approved
        from kontract k, kontstate ks, region_list(:rep_region_id, 1, -1, 1)
        where k_id = ks_from_k_id
        and ks_state = 1
        and ks_ol_is_delete = 0
        and :rep_date between ks_date_begin and ks_date_end
        and cast(k_last_reg_id as integer) = id_list
        and k_from_pr_id in (54, 55, 56, 57, 64, 65, 66, 67)
        and :k_id_in is null
        and :rep_date_begin is null
        order by k_last_reg_id, k_from_pr_id, k_disb_date';
      end else
      begin
        sql_text = '
        select k.k_last_reg_id,
               k_disb_date,
               k_id,
               k_kontract_no,
               k_from_pr_id,
               k.k_fact_disb_lo_id,
               k.k_fact_disb_kf_no,
               k_amount_approved
        from kontract k
        where k_id = :k_id_in
        and cast(:rep_region_id as integer) = :rep_region_id
        and cast(:rep_date as date) = :rep_date
        and :rep_date_begin is null
        and k_from_pr_id in (54, 55, 56, 57, 64, 65, 66, 67)';
      end 
  end
  else if (rep_type = 'disb') then
  begin
    sql_text = '
        select k.k_last_reg_id,
               k_disb_date,
               k_id,
               k_kontract_no,
               k_from_pr_id,
               k.k_fact_disb_lo_id,
               k.k_fact_disb_kf_no,
               k_amount_approved
        from kontract k, kontstate ks, region_list(:rep_region_id, 1, -1, 1)
        where k_id = ks_from_k_id
        and ks_state = 1
        and ks_ol_is_delete = 0
        and ks_date_begin between :rep_date_begin and :rep_date
        and cast(k_last_reg_id as integer) = id_list
        and :k_id_in is null
        and k_from_pr_id in (54, 55, 56, 57, 64, 65, 66, 67)
        order by k_last_reg_id, k_from_pr_id, k_disb_date';
  end

  for execute statement(sql_text)(rep_region_id := rep_region_id, rep_date := rep_date, k_id_in := k_id_in, rep_date_begin := rep_date_begin)
      into office_id, kf_disb_date, k_id, k_kontract_no, k_from_pr_id, k_fact_disb_lo_id, kf_no, k_amount_approved
  do
  begin
    if (coalesce(pr_id, 0) <> k_from_pr_id) then
    begin
      select pr_name
      from product pr
      where pr_id = :k_from_pr_id
      into pr_name;

      pr_id = k_from_pr_id;
    end

    if (coalesce(reg_id, 0) <> office_id) then
    begin
      reg_id = office_id;

      select reg_name
      from region r
      where r.reg_unicode = :office_id
      into :office_name;
    end

    k_balance_os = 0;
    oks_kc_id = null;
    select oks_balance, oks_kc_id
    from oksummary oks
    where :rep_date between oks_date_begin and oks_date_end and
          oks_from_k_id = :k_id
          and oks_close_state = 0 
    into k_balance_os, oks_kc_id;

    kc_officer_name = '';
    select c_name_fio
    from clients
    where c_id = :oks_kc_id
    into :kc_officer_name;

    kc_disb_officer_name = '';
    select c_name_fio
    from clients
    where c_id = :k_fact_disb_lo_id
    into :kc_disb_officer_name;

    pmt = 0;
    select sh_os + sh_perc
    from schedule sh
    where sh.sh_from_k_id = :k_id and
          sh.sh_id = 1
    into pmt;
    pmt = coalesce(pmt, 0);

    for select kc_from_c_id, kc_amount_approved, kc_circle
        from knclient kc
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, kc_amount, kc_circle
    do
    begin
      if (k_amount_approved > kc_amount) then
      begin
        cl_proportion = kc_amount / k_amount_approved;
        k_balance_os = k_balance_os * cl_proportion;
        pmt = pmt * cl_proportion;
      end

      client_fio = '';
      select c_name_fio
      from clients
      where c_id = :c_id
      into client_fio;

      dti_txt = '';
      dti = 0;
      dti_d = 0;
      dti_i = 0;
      dti_income = 0;
      income_type = '';
      job_name = '';
      for select rd.rd_id, rd_name, rd_code
          from refdata rd
          where rd.rd_main_code = 'AnketaQuestions' and
                (rd.rd_code containing 'AnketaQuestions.DTI' or
                rd_name in ('Gəlirin növü', 'Vəzifəsi'))
          into rd_id, rd_name, rd_code
      do
      begin
        ovs_value = '';
        ovs_value_no = 0;
        select ovs_value
        from hp_objectprops_get(0, :c_id, -1, :rd_id, /*:rep_date*/ :kf_disb_date)
        into ovs_value;
        ovs_value = coalesce(ovs_value, '');

        if (rd_name = 'Gəlirin növü') then --Gəlirin növü (Вид дохода – это инф. из анкетных данных клиента)
          income_type = ovs_value;
        else
        if (rd_name = 'Vəzifəsi') then --Vəzifəsi (Должность – это инф. из анкетных данных клиента)
          job_name = ovs_value;
        else
        if (ovs_value <> '') then
        begin
          select string_txt_out
          from string_char_to_decimal(:ovs_value)
          into ovs_value_no;

          ovs_value_no = coalesce(ovs_value_no, 0);

          if (rd_code containing 'DTI_D') then
            dti_d = ovs_value_no; --финансовая загруженность клиента

          if (rd_code containing 'DTI_I' and
              rd_code not containing 'DTI_Income') then
            dti_i = ovs_value_no; --чистый доход клиента (после вычета налогов)

          if (rd_code containing 'DTI_Income') then
            dti_income = ovs_value_no; --Доход
        end
      end

      if (coalesce(dti_i, 0) = 0 and
          dti_income > 0) then
      begin
        if (dti_income <= 200) then
        begin
          /*a.    Если сумма дохода в справке  200,00 или меньше 200,00  (1-ый метод)
            то значение I вычисляется как указано ниже:
             
                I = Доход – (Доход *0.035)
                   
               Например доход = 200,00 то:
                I = 200 – (200*0.035)
               I = 193
        */
          dti_i = dti_income - (dti_income * 0.035);
        end

        if (dti_income between 200.01 and 8000) then
        begin
          /*b.    Если сумма дохода в справке находится в промежутке 201 – 8000,00 (восемь тысяч включительно) (2-ой метод)
        то значение I вычисляется более сложным способом (200 и 6 это фиксированные цифры):
         
        I = Доход - (((Доход – 200.00) * 0.10 ) + 6 ) – (Доход*0.005)
        
        Например, доход  равен 5000,00  то:
        I = 5000 –  (((5000 – 200) * 0.10 ) + 6 ) – (5000*0.005)
        I = 5000 – 486 –  25 
        I = 4489
                */
          dti_i = dti_income - (((dti_income - 200) * 0.1) + 6) - (dti_income * 0.005);
        end

        if (dti_income > 8000) then
        begin
          /*c.    Если сумма дохода в справке больше 8000,00 (3-ий метод)
        то значение о значение I вычисляется нижеуказанным способом (8000, 200 и 6 это фиксированные цифры):
         
        I = Доход - (((Доход – 200.00) * 0.10 ) + 6 ) – (Доход*0.005) - ((Доход – 8000) * 0.14 )
        
        Например, доход  равен 8500  то:
        I = 8500 – (((8500 – 200.00) * 0.10 ) + 6 ) – (8500*0.005) –  ((8500 – 8000) * 0.14 ) 
        I = 8500  – 836 – 42,50 – 70 
        I = 7551,50
                        */
          dti_i = dti_income - (((dti_income - 200) * 0.1) + 6) - (dti_income * 0.005) - ((dti_income - 8000) * 0.14);
        end
      end

      dti = 0;
      if (dti_i > 0) then
        dti = ((dti_d + pmt) / dti_i) * 100;
      else
--        DTI_TXT = 'Расчет DTI не возможен! Необходимо заполнить данные!';
        dti_txt = 'DTI hesablanması mümkün deyil. Anket məlumatlarında müştərinin gəliri və borc yükü barədə məlumatları qeyd edin.';

      /*•         Если значение I равно, или меньше 519.00 то для выдачи кредита значение DTI не должно превышать 40 %
        •         Если значение I  больше 519.00 то для выдачи кредита значение DTI не должно превышать 50 %
        */
      if (dti > 40) then
      begin
        if (kf_disb_date < '04.02.19') then
        begin
          if (dti_i <= 519) then
            dti_txt = 'Müştərinin xalis qəliri 519.0 AZN və daha az olarkən DTI 40 %-dən çox ola bilməz';
          else
          if (dti > 50) then
            dti_txt = 'Müştərinin xalis qəliri 519.0 AZN çox olarkən DTI 50 %-dən çox ola bilməz';
        end
        else     
        if (kf_disb_date < '04.11.19') then
        begin
          /*
            SYSDBA
        •    Если значение I равно, или меньше 300 то для выдачи кредита значение DTİ не должно превышать 70 %
        •    Если значение I в промежутке 301- 573 то для выдачи кредита значение DTİ не должно превышать 40 %
        •    Если значение I больше 573 то для выдачи кредита значение DTİ не должно превышать 50 %
                                              */

          if (dti_i <= 400 and
              dti > 70) then
            dti_txt = 'Müştərinin xalis qəliri 400 AZN və daha az olarkən DTI 70 %-dən çox ola bilməz';
          if (dti_i between 401 and 573 and
              dti > 40) then
            dti_txt = 'Müştərinin xalis qəliri 401 AZN çox, 573 və daha az olarkən DTI 40 %-dən çox ola bilməz';
          else
          if (dti_i > 573 and
              dti > 50) then
            dti_txt = 'Müştərinin xalis qəliri 573.0 AZN çox olarkən DTI 50 %-dən çox ola bilməz';

        end else
        begin  
          if (dti > 50) then
            dti_txt = 'DTI 50 %-dən çox ola bilməz.';
        end
      end

      suspend;
    end
  end
end
