CREATE PROCEDURE AAREP_MAT_POKUPKA_FULL(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    REP_ML_NAME VARCHAR(255),
    REP_INVENT_STR VARCHAR(50),
    REP_OPER_DATE DATE,
    REP_REGION_NAME VARCHAR(100),
    REP_ACCOUNT VARCHAR(20),
    REP_KOLICH_PLUS DECIMAL(18,4),
    REP_KOLICH_MINUS DECIMAL(18,4),
    REP_SUMMA_PLUS DECIMAL(18,4),
    REP_SUMMA_MINUS DECIMAL(18,4),
    REP_MATOTVET VARCHAR(100),
    REP_WORKPLACE VARCHAR(100),
    REP_CATEGORY VARCHAR(100),
    REP_CATEGORY_PARENT VARCHAR(100),
    REP_CATEGORY_TYPE VARCHAR(100),
    REP_ACCOUNT_NAME VARCHAR(100),
    REP_ACC_BALANCE VARCHAR(20),
    REP_MATOTVET_INFO VARCHAR(100),
    REP_WORKPLACE_INFO VARCHAR(100),
    REP_ACCOUNT_INFO VARCHAR(20),
    REP_BALANCE_NAME VARCHAR(100),
    REP_OPER_KIND VARCHAR(50),
    REP_MODERN DECIMAL(18,4),
    REP_SUMMA_ALL DECIMAL(18,4))
AS
declare variable om_bill_id decimal(18,0);
declare variable om_bill_line_no integer;
declare variable rep_is_modern integer;
begin
  for select rep_ml_name, rep_invent_str, rep_oper_date, rep_region_name, rep_account, rep_kolich_plus,
             rep_kolich_minus, rep_summa_plus, rep_summa_minus, rep_matotvet, rep_workplace, rep_category,
             rep_category_parent, rep_category_type, rep_account_name, rep_acc_balance, rep_matotvet_info,
             rep_workplace_info, rep_account_info, rep_balance_name, rep_oper_kind, rep_modern, om_bill_id,
             om_bill_line_no, rep_is_modern
      from aarep_mat_pokupka(:report_region_id, :report_date_begin, :report_date_end)
      into :rep_ml_name, :rep_invent_str, :rep_oper_date, :rep_region_name, :rep_account, :rep_kolich_plus,
           :rep_kolich_minus, :rep_summa_plus, :rep_summa_minus, :rep_matotvet, :rep_workplace, :rep_category,
           :rep_category_parent, :rep_category_type, :rep_account_name, :rep_acc_balance, :rep_matotvet_info,
           :rep_workplace_info, :rep_account_info, :rep_balance_name, :rep_oper_kind, :rep_modern, :om_bill_id,
           :om_bill_line_no, :rep_is_modern
  do
  begin
    if (rep_is_modern = 1) then
    begin
      rep_modern = rep_summa_plus - rep_summa_minus;
      rep_summa_plus = 0;
      rep_summa_minus = 0;
    end
    else
      rep_modern = 0;
    rep_summa_all = rep_summa_plus - rep_summa_minus + rep_modern;
    suspend;
  end
end
