CREATE PROCEDURE HP_MOR_SYNCHRON
AS
declare variable SQL_TEXT varchar(2048);
declare variable SQL_TEXT1 varchar(2048);
declare variable MSS_ID integer;
declare variable MSS_DB_CONNECT varchar(255);
declare variable MSS_USER_CONNECT varchar(50);
declare variable MSS_PASS varchar(50);
declare variable MSS_USER_SAVE decimal(18,0);
declare variable MSS_OFFICE_SAVE integer;
declare variable CM_DEFAULT_CURRENCY varchar(3);
declare variable LMS_NUMBER integer;
declare variable M_CLIENT_FIO2 varchar(100);
declare variable M_CLIENT_BIRTH_DATE2 date;
declare variable TIME_BEGIN timestamp;
declare variable M_ID decimal(18,0);
declare variable M_NUMBER integer;
declare variable M_NEW_ID decimal(18,0);
declare variable M_ID1 decimal(18,0);
declare variable M_NUMBER1 integer;
declare variable M_IS_BLACK integer;
declare variable M_DATE_BEGIN date;
declare variable M_DATE_END date;
declare variable M_MONTH integer;
declare variable M_IS_ACTIVE smallint;
declare variable M_COMMENT_GROUP varchar(2048);
declare variable M_COMMENT_OFFICER varchar(2048);
declare variable M_COMMENT_MANAGER varchar(2048);
declare variable M_CLIENT_FIO varchar(100);
declare variable M_CLIENT_PASS_NO varchar(30);
declare variable M_CLIENT_ADDRESS varchar(255);
declare variable M_CLIENT_BIRTH_DATE date;
declare variable M_OUR_CLIENTS integer;
declare variable M_FROM_REG_ID integer;
declare variable M_NO integer;
declare variable M_DATETIME timestamp;
declare variable M_DATETIME_EDIT timestamp;
declare variable M_USER_CREATE decimal(18,0);
declare variable M_USER_EDIT decimal(18,0);
declare variable M_DB_CODE varchar(3);
declare variable M_ORIGINAL_ID decimal(18,0);
declare variable K integer;
declare variable DB_ID integer;
begin
  db_id = f_get_db_id();

  for select mss_id, mss_db_connect, mss_user_connect, mss_pass, mss_user_save, mss_office_save
      from morsynsettings
      where (:db_id = 81001 and mss_db_id in (94401, 41701))
      or (:db_id in (39801, 41701) and mss_db_id in (81001))
      into :mss_id, :mss_db_connect, :mss_user_connect, :mss_pass, :mss_user_save, :mss_office_save
  do
  begin
-- Генерирует номер записи для логирования
    select coalesce(max(lms_number), 0) + 1
    from logmorsyn
    where lms_from_mss_id = :mss_id and
          lms_date = current_date
    into :lms_number;

--  До синхранизации создает новую запись в таблице логов
--    time_begin = current_timestamp;
    time_begin = cast('now' as timestamp);
    insert into logmorsyn (lms_from_mss_id, lms_date, lms_number, lms_rec_count, lms_datetime_begin, lms_syn)
    values (:mss_id, current_date, :lms_number, 0, :time_begin, 0);

-- счетчик для определения кол-во копируемой записей
    k = 0;

--  Из другой БД вытаскивает валюту для кода ДБ
    execute statement 'select cm_default_currency from company'
        on external :mss_db_connect
        as user :mss_user_connect password :mss_pass
        into :cm_default_currency;

--  Из другой БД вытягивает ID для сравнения с текущей
    sql_text = 'select m_id, m_number, m_client_fio, m_client_birth_date ' || 'from moratory ' || 'where m_is_black > 0 ' || 'and m_is_active = 1 ' || 'and m_original_id is null';
    for execute statement :sql_text
            on external :mss_db_connect
            as user :mss_user_connect password :mss_pass
            into :m_id, :m_number, m_client_fio2, m_client_birth_date2
    do
    begin
      m_new_id = null;
      select m_id
      from moratory
      where (m_original_id = :m_id and
            m_number = :m_number) -- Проверка, что раньше не было синхронизировано
            or
            (m_client_fio = :m_client_fio2 and
            m_client_birth_date = :m_client_birth_date2) -- Проверка по ФИО и по дате рождения
            rows 1
      into :m_new_id;
-- Если такой записи нету в текущей БД то выполняет следующее
      if (:m_new_id is null) then
      begin
        sql_text1 = 'select
             -255 as m_id1, m_number, m_is_black, m_date_begin, m_date_end, m_month,
             m_is_active,(
               select ''Импортирован из офиса:''||r1.reg_code||'' ''||r1.reg_name||
               '', филиал:''||r2.reg_code||'' ''||r2.reg_name||'', округ:''||
               r3.reg_code||'' ''||r3.reg_name
               from region r1, region r2, region r3
               where r1.parent = r2.reg_unicode
                 and r2.parent = r3.reg_unicode
                 and r1.reg_unicode = m_from_reg_id
               ) as m_comment_group,
             m_comment_officer, m_comment_manager, m_client_fio, m_client_pass_no,
             m_client_address, m_client_birth_date, m_our_clients,' || :mss_office_save || ' as m_from_reg_id, m_no, m_datetime, m_datetime_edit,' || :mss_user_save || ' as m_user_create,' || :mss_user_save || ' as m_user_edit,' || :cm_default_currency || ' as m_db_code,
             m_id as m_original_id
          from moratory
          where m_id = ' || :m_id || ' and m_number =' || :m_number;
        for execute statement :sql_text1
                on external :mss_db_connect
                as user :mss_user_connect password :mss_pass
                into m_id1, m_number1, m_is_black, m_date_begin, m_date_end, m_month, m_is_active, m_comment_group,
                m_comment_officer, m_comment_manager, m_client_fio, m_client_pass_no, m_client_address,
                m_client_birth_date, m_our_clients, m_from_reg_id, m_no, m_datetime, m_datetime_edit, m_user_create,
                m_user_edit, m_db_code, m_original_id
        do
        begin
          if (m_from_reg_id is null) then m_from_reg_id = mss_office_save;
          insert into moratory (m_id, m_number, m_is_black, m_date_begin, m_date_end, m_month, m_is_active,
                                m_comment_group, m_comment_officer, m_comment_manager, m_client_fio, m_client_pass_no,
                                m_client_address, m_client_birth_date, m_our_clients, m_from_reg_id, m_no, m_datetime,
                                m_datetime_edit, m_user_create, m_user_edit, m_db_code, m_original_id)
          values (:m_id1, :m_number1, :m_is_black, :m_date_begin, :m_date_end, :m_month, :m_is_active, :m_comment_group,
                  :m_comment_officer, :m_comment_manager, :m_client_fio, :m_client_pass_no, :m_client_address,
                  :m_client_birth_date, :m_our_clients, :m_from_reg_id, :m_no, :m_datetime, :m_datetime_edit,
                  :m_user_create, :m_user_edit, :m_db_code, :m_original_id);
          k = :k + 1;
        end
      end
    end
--  Если все прошло успешно то обновляет запись лога на успешное
    update logmorsyn
    set lms_rec_count = :k,
--        lms_datetime_end = current_timestamp,
        lms_datetime_end = cast('now' as timestamp),
        lms_syn = 1
    where (lms_from_mss_id = :mss_id) and
          (lms_date = current_date) and
          (lms_number = :lms_number);
  end
end
