CREATE PROCEDURE HP_MONETA_DIFF_KONT_PAY_SBP_TMP(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    STEP SMALLINT)
RETURNS (
    ESM_ID ID_TYPE,
    K_ID ID_TYPE,
    SBPPHONE STR_100,
    AMOUNT CURR_TYPE,
    ESM_DATE DATE,
    REG_ID INTEGER,
    KF_OFFICER ID_TYPE,
    IS_FOUND SMALLINT)
AS
declare variable esm_full_source blob sub_type 1 segment size 80;
declare variable sbpphone_old str_100;
declare variable k_id_old id_type;
declare variable is_found_tmp smallint;
begin
  -- цель процедуры: получить все платежи по Монете (Тинькову), где за период с одной и той же карте были платежи по разным договорам

  if (step = 1) then
  begin
    -- а)  делаю выборку из  extsysmain по проекту moneta(tinkoff)
    -- б)  получаю номер карточки клиента

    for select em.esm_id, em.esm_from_k_id, em.esm_full_source, em.esm_amount,
               em.esm_date, kal.kal_new_reg_id, kal.kal_new_kf_officer
        from extsysmain em
        inner join kontract kont on em.esm_from_k_id = kont.k_id
        inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
        inner join region_list(:rep_region_id, 1, -1, 1) on id_list = kal.kal_new_reg_id
        where em.esm_action = 'pay' and
              em.esm_project in ('moneta') and
              em.esm_date between :rep_date_begin and :rep_date_end and
              em.esm_status = 1 and
              kal.is_delete = 0 and
              :rep_date_end between kal.kal_date and kal.kal_date_end
        into :esm_id, :k_id, :esm_full_source, :amount, :esm_date, :reg_id,
             :kf_officer
    do
    begin
      sbpphone = '';
      sbpphone = (select field_value
                     from hp_extsys_parcer(:esm_full_source, 'sbpphone'));

      sbpphone = coalesce(sbpphone, '');
      if (sbpphone <> '') then
        suspend;
    end
  end

  if (step = 2) then
  begin
    -- если по одной и той же карточке есть палтежи по разным займам маркирую один из займов как  is_found = 1

    sbpphone_old = '';
    k_id_old = 0;

    for select esm_id, k_id, amount, esm_date, reg_id, kf_officer, sbpphone
        from hp_moneta_diff_kont_pay_sbp_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, 1)
        order by sbpphone, k_id
        into :esm_id, :k_id, :amount, :esm_date, :reg_id, :kf_officer,
             :sbpphone
    do
    begin
      is_found = 0;

      if (sbpphone = sbpphone_old) then
      begin
        if (k_id <> k_id_old) then
        begin
          k_id_old = k_id;

          is_found = 1;
        end
      end
      else
      begin
        sbpphone_old = sbpphone;
        k_id_old = k_id;
      end

      suspend;
    end
  end
  if (step = 3) then
  begin
    -- если по карте есть хоть одна маркровка is_found = 1, маркирую остальные платежи этой же маркировкой is_found = 1,
    -- данная маркровка говрит о том что с карты был платеж по разным договорам

    sbpphone_old = '';

    for select esm_id, k_id, amount, esm_date, reg_id, kf_officer, sbpphone,
               is_found
        from hp_moneta_diff_kont_pay_sbp_tmp(:rep_region_id, :rep_date_begin, :rep_date_end, 2)
        order by sbpphone, is_found desc
        into :esm_id, :k_id, :amount, :esm_date, :reg_id, :kf_officer,
             :sbpphone, :is_found_tmp
    do
    begin
      if (sbpphone <> sbpphone_old) then
      begin
        is_found = is_found_tmp;
        sbpphone_old = sbpphone;
      end
      if (is_found = 1) then
        suspend;
    end
  end
end
