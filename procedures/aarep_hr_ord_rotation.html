CREATE PROCEDURE AAREP_HR_ORD_ROTATION(
    ORDER_ID D_ID)
RETURNS (
    C_NAME_FIO D_STR_100,
    PS_DATE_BEGIN D_DATE)
AS
declare variable c_id d_id;
declare variable order_date d_date;
begin

  select ho_date
  from hr_orders
  where ho_id = :order_id
  into order_date;

  for select c_name_fio,
             c_id
      from hr_pstatus
      inner join clients on c_id = ps_from_c_id
      where ps_from_ho_id = :order_id
      order by c_name_fio
      into c_name_fio,
           c_id
  do
  begin
    for select ps_date_begin
        from hr_pstatus
        inner join hr_orders on ho_id = ps_from_ho_id
        where ho_status = 1 and
              ho_order_type in (1, 2, 3) and
              ho_date <= :order_date and
              ps_from_c_id = :c_id
        group by ps_date_begin
        order by ps_date_begin
        into ps_date_begin
    do
      suspend;

  end
end
