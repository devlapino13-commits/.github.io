CREATE PROCEDURE HP_CLIENTSTATE_ORDER_ERR_NOTIFICATION
AS
declare variable orders d_integer;
declare variable mail_to d_str_255;
declare variable mail_title d_str_255;
declare variable mail_body d_str_long;
begin

    orders = (select count(cso_c_id) from clientstate_order);
    if (orders < 1200) then
        exit;

    mail_to = 'finstart@mbfplanet.com';
    mail_title = 'Скопилось много необработанных записей (возможно, сломался обработчик)';
    mail_body =
        'В таблице <b>clientstate_order</b> скопилось большое количество записей: ' || orders || '.<br><br>' ||
        '<b>Возможные причины:</b><br>' ||
        ' 1. Обработчик исправен, но записи просто не успели обработаться (fsm_id = 211).<br>' ||
        ' 2. С роботом либо с процедурой hp_clientstate_order_handle действительно какие-то проблемы.<br><br>' ||
        '<b>Что делать:</b><br>' ||
        'Если запустить вручную <code>execute procedure hp_clientstate_order_handle;</code> и все записи будут обработаны, ' ||
        'то процедура исправна. В таком случае робот просто не успел их обработать (стоит подождать) ' ||
        'либо требуется перезагрузка самого робота.';

    execute procedure hp_send_quick_message('email', mail_to, mail_title, mail_body, 'clientstate_orders', 30, 1000000000000);
end
