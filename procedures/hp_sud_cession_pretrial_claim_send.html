CREATE PROCEDURE HP_SUD_CESSION_PRETRIAL_CLAIM_SEND
AS
declare variable cnt d_integer;
declare variable max_cnt d_integer;
declare variable sud_mainreq_obj_kind d_integer;
declare variable smr_id d_id;
declare variable sld_id d_id;
declare variable c_id d_id;
declare variable error_text d_str_255;
declare variable tol_kind d_str_code;
declare variable tol_date d_date;
declare variable cell_no d_str_100;
declare variable msk_call_time_start d_time;
declare variable msk_call_time_end d_time;
begin
  cnt = 0;
  max_cnt = 500;
  tol_kind = 'hp_sud_load_loans';
  sud_mainreq_obj_kind = f_objkind('sud_mainreq', 'id');

  -- удалим старые ошибки
  delete from temp_objlist
  where tol_kind = :tol_kind
    and tol_object_kind = :sud_mainreq_obj_kind
    and tol_status < 0
    and tol_date + 60 < current_date;

  for select tol_object_id, tol_date
  from temp_objlist
  where tol_kind = :tol_kind
    and tol_object_kind = :sud_mainreq_obj_kind
    and tol_status = 0
  into smr_id, tol_date
  do
  begin
    sld_id = null;
    error_text = '';

    select sld_id
    from sud_maindocs, sud_loandocs, sud_cfgdockind
    where sld_from_smd_id = smd_id
      and scd_id = sld_doc_kind
      and smd_id = :smr_id
      and scd_code = 'pretrial_claim'
    into sld_id;

    if (sld_id is null) then
      error_text = 'Не найден документ "Досудебная претензия" в модуле ДРПВ';

    if (error_text = ''
      and (select count(distinct slc_from_kc_id) from sud_loanclient where slc_from_smr_id = :smr_id) > 1)
    then
      error_text = 'Групповой займ. Нельзя определить время отправки СМС';

    if (error_text = '') then
    begin
      select distinct kc_from_c_id
      from sud_loanclient, knclient
      where kc_id = slc_from_kc_id
        and slc_from_smr_id = :smr_id
      into c_id;

      cell_no = coalesce(f_cont_info(0, c_id, 'cell', 'ok'), '');

      if (cell_no = '') then
        cell_no = coalesce(f_cont_info(0, c_id, 'cell', 'last'), '');

      if (cell_no = '') then
        error_text = 'Не удалось найти номер телефона для отправки СМС';
    end

    if (error_text = '') then
    begin
      select msk_call_time_start, msk_call_time_end
      from hp_places_get_timezone_c(:c_id, :cell_no)
      into msk_call_time_start, msk_call_time_end;
  
      if (cast(current_time as d_time) not between msk_call_time_start and msk_call_time_end) then
        continue;
    end 

    if (error_text = '') then
    begin
      select res_text
      from hp_pretrial_claim_send(:sld_id, 1000000000000)
      into error_text;

      if (error_text = '') then
      begin
        delete from temp_objlist
        where tol_kind = :tol_kind
          and tol_object_kind = :sud_mainreq_obj_kind
          and tol_object_id = :smr_id;

        cnt = cnt + 1;
    
        if (cnt > max_cnt) then
          exit;
      end
    end 

    if (coalesce(error_text, '') <> '') then
    begin
      update temp_objlist
      set tol_add_info_1 = coalesce(:error_text, '<NULL>'),
          tol_status = -1
      where tol_kind = :tol_kind
        and tol_object_kind = :sud_mainreq_obj_kind
        and tol_object_id = :smr_id;
    end

    when any do
    begin
      error_text = 'Ошибка при создании электронного документа';

      update or insert into temp_objlist (tol_kind, tol_object_kind, tol_object_id, tol_status, tol_date)
      values (:tol_kind, :sud_mainreq_obj_kind, :smr_id, -1, :tol_date);
    end
  end 
end
