CREATE PROCEDURE MC_BUTTON_LIST(
    MCL_ID D_ID,
    REP_ID D_STR_CODE,
    OBJ_KIND D_STR_CODE,
    OBJ_ID D_ID,
    PARAMS D_STR_LONG)
RETURNS (
    RES_REP_ID D_STR_CODE,
    RES_KIND D_STR_CODE,
    RES_CAPTION D_STR_CODE,
    RES_IS_BUTTON D_SMALLINT,
    RES_PARAMS D_STR_LONG)
AS
declare variable db_id integer;
declare variable c_id decimal(18,0);
begin
-- Процедура возвращает список меню для окна в мобильном приложении.
-- Если вызывается из основного окна клиента, Rep_ID = Client

-- Возвращаемые результаты:
-- RES_KIND - варианты: Open, HTML, Report, SQL
--   Для RES_KIND=Open,
--     в RES_REP_ID передаем кодовое название окна для перехода, например Client
--     в RES_PARAMS - Json, состав параметров зависит от окна. Н-р для access_kind требуется для окна клиента
--   Для RES_KIND=HTML,
--     в RES_REP_ID передаем Rep_ID (Кодовое название окна HTML)
--     в RES_PARAMS передаем
--       {"title":"Заголовок окна отчета", "obj_kind":"obj_kind", "obj_id":"obj_id",
--        "date_begin":"Дата нач.", "date_end":"Дата окон."}
--   Для RES_KIND=Report,
--     в RES_REP_ID передаем ID отчета FastReport
--     в RES_PARAMS передаем требуемые параметры FastReport, пример:
--       {"title":"Заголовок окна отчета", "obj_kind":"obj_kind", "obj_id":"obj_id",
--        "date_begin":"Дата нач.", "date_end":"Дата окон."}
--   Для RES_KIND=SQL,
--     в RES_REP_ID передаем название команды, эта команда будет обработана в процедуре fm_report_cmd
--     RES_PARAMS={"obj_kind":"obj_kind", "obj_id":"obj_id", }
-- RES_IS_BUTTON - это пункт меню или кнопка в нижней части окна
-- RES_CAPTION - текст пункта меню
--   Для RES_KIND=SQL,
--     в RES_REP_ID передаем название команды, эта команда будет обработана в процедуре fm_report_cmd
--     RES_PARAMS={
--       "obj_kind":"obj_kind",
--       "obj_id":"obj_id",
--       "description":"description", -- Основной текст окна диалогового окна выше редактора
--       "edit_kind": "none", -- none, text, date, combo, mru, по умолчанию - none
--       "comment":"comment", -- Примечание, в нижней части окна ниже редактора
--       "max_length":"max_length", -- Макс длина строки для edit_kind = (text, mru), по-умолчанию не ограничена
--       "max_lines":"1", -- Макс. количество строк для edit_kind = (text, mru), по-умолчанию: 1
--       "empty_err":"empty_err", -- Текст ошибки, когда пользователь ничего не указал в редакторе
--          если не указано, то допускается пустое значение в редакторе, для edit_kind = (text, mru, date)
--       "def_text":"def_text", -- Текст в редакторе по-умолчанию, для edit_kind = (text, mru, date)
--       "show_res":"dialog" -- Отображение результата в отдельном окне "window", или диалоге "dialog" (по-умолчанию)
--     }
--     Для параметров description, comment, empty_err - допустимы базовые теги HTML
--     Пример, для текстового параметра:
--     RES_PARAMS =
--       '{' ||
--       '"description":"Вы действительно <b>этого</b> хотите?",' ||
--       '"obj_kind":"' || obj_kind || '",' ||
--       '"obj_id":"' || obj_id || '",' ||
--       '"comment":"Если это сделать будет <b>вообще</b> замечательно, ты даже не представляешь, как!",' ||
--       '"edit_kind":"text",' ||
--       '"max_lines":4' ||
--       '"max_length":255,' ||
--       '"empty_err":"Вводи давай!",'  ||
--       '"def_text":"Hello World!"' ||
--       '}';

-- Инициализация

  :db_id = f_get_db_id();

-- Подмена отчета, при необходимости
  select res_rep_id
  from mc_replace_rep_id(:mcl_id, :rep_id)
  into rep_id;

  select mcl_from_c_id
  from mc_login
  where mcl_id = :mcl_id
  into :c_id;

  if (rep_id = 'my_profile') then
  begin
    res_caption = 'Указать ФИО и ДР';
    res_kind = 'open';
    res_rep_id = 'edit_client';
    res_is_button = 1;
    res_params = '';
    suspend;
  end

  if (rep_id = 'my_profile') then
  begin
    res_caption = 'Сфотографировать паспорт';
    res_kind = 'open';
    res_rep_id = 'docrecog_main';
    res_is_button = 1;
    res_params = '';
    suspend;
  end
-- Из окна "О Компании"
  if (rep_id = 'about_company') then
  begin
    res_caption = 'Инфо о клиенте';
    res_kind = 'Report';
    res_rep_id = 'ClientInfo';
    res_is_button = 1;
    res_params = '{' || '"title":"' || res_caption || '",' || '"obj_id":"' || 1000000000001 || '"' || '}';
    suspend;
  end

  if (rep_id = 'html_test') then
  begin
    res_caption = 'Тестовое окно 2';
    res_kind = 'html';
    res_rep_id = 'html_test2';
    res_is_button = 0;

    res_params = '{' || '"title":"' || res_caption || '",' || '"obj_id":"' || 1000000000001 || '"' || '}';
    suspend;

    res_caption = 'Тестовое окно 3';
    res_kind = 'html';
    res_rep_id = 'html_test2';
    res_is_button = 0;

    res_params = '{' || '"title":"' || res_caption || '",' || '"obj_id":"' || 1000000000001 || '"' || '}';
    suspend;
  end
end
