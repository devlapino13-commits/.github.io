CREATE PROCEDURE AHPLIST_CADDRESS(
    C_ID DECIMAL(18,0))
RETURNS (
    CA_NUMBER INTEGER,
    CA_TYPE INTEGER,
    CA_FROM_C_ID DECIMAL(18,0),
    CA_FROM_PL_ID DECIMAL(18,0),
    CA_ADDRESS VARCHAR(255),
    CA_PHONE VARCHAR(100),
    CA_DATE DATE,
    CA_IS_ACTIVE SMALLINT,
    CA_IS_MAIN SMALLINT,
    CA_MEMO VARCHAR(255),
    CA_TYPE_TEXT VARCHAR(30),
    CA_IS_MAIN_TEXT VARCHAR(20))
AS
begin
  for select adc_address_type, adc_address_type, adc_from_c_id, adl_from_pl_id, left(adl_address_str, 255),
             left(adc_phone, 100), adc_date, adc_memo
      from addrclient, addrlist
      where adc_from_c_id = :c_id and
            adc_from_adl_id = adl_id
      into :ca_number, :ca_type, :ca_from_c_id, :ca_from_pl_id, :ca_address, :ca_phone, :ca_date, :ca_memo
  do
  begin
    if (ca_type = 1) then
      ca_type_text = 'Юридический';
    if (ca_type = 2) then
      ca_type_text = 'Проживание';
    if (ca_type = 3) then
      ca_type_text = 'Место бизнеса';
    if (ca_type = 4) then
      ca_type_text = 'Юрид.на родине';
    if (ca_type = 5) then
      ca_type_text = 'Факт.на родине';

    if (ca_type = 1) then
      ca_is_main_text = 'Основной';
    else
      ca_is_main_text = 'Дополнительный';
    ca_is_active = 1;
    if (ca_type = 1) then
      ca_is_main = 1;
    else
      ca_is_main = 0;

    suspend;
  end
end
