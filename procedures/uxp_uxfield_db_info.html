CREATE PROCEDURE UXP$UXFIELD_DB_INFO(
    OBJ_NAME D_STR_CODE,
    FIELD_NAME D_STR_CODE,
    IS_TABLE D_BOOL)
RETURNS (
    FIELD_TYPE D_STR_CODE,
    FIELD_SIZE D_INTEGER,
    FIELD_SCALE D_INTEGER,
    FIELD_DEFAULT D_STR_100,
    FIELD_REQUIRED D_BOOL,
    FIELD_DOMAIN D_STR_CODE,
    FIELD_POSITION D_INTEGER,
    FIELD_READONLY D_BOOLEAN,
    FIELD_ORIG_DEFAULT D_STR_100)
AS
declare variable field_subtype d_integer;
declare variable field_type_id d_integer;
declare variable pkg_name d_str_code;
declare variable field_length d_integer;
declare variable field_db_domain d_str_code;
begin
  obj_name = upper(trim(obj_name));
  field_name = upper(trim(field_name));

-- boolean
-- smallint
-- integer
-- float
-- date
-- time
-- bigint
-- percent
-- currency
-- double
-- datetime
-- string
-- memo  -- это текстовый блоб
-- blob  -- это бинарный блоб


  if (obj_name <> '') then
  begin
    if (coalesce(is_table, 0) = 0) then
    begin
      select f.rdb$field_type, f.rdb$field_length, f.rdb$field_scale, coalesce(f.rdb$field_sub_type, 0),
             p.rdb$field_source, (f.rdb$computed_source is not null),
             left(trim(p.rdb$default_source), 100),
             coalesce(p.rdb$null_flag, 0),
             rdb$parameter_number
      from rdb$procedure_parameters p, rdb$fields as f, rdb$types t
      where p.rdb$procedure_name = :obj_name and
            p.rdb$parameter_name = :field_name and
            p.rdb$package_name is null and
            f.rdb$field_name = p.rdb$field_source and
            f.rdb$field_type = t.rdb$type and
            t.rdb$field_name = 'RDB$FIELD_TYPE'
      rows 1
      into field_type_id, field_length, field_scale, field_subtype,
           field_db_domain, field_readonly, field_default, field_required, field_position;

      if (row_count = 0) then
      begin
        for select rdb$package_name
        from rdb$packages
        into pkg_name
        do
        begin
          select f.rdb$field_type, f.rdb$field_length, f.rdb$field_scale, coalesce(f.rdb$field_sub_type, 0),
                 p.rdb$field_source, (f.rdb$computed_source is not null),
                 left(trim(p.rdb$default_source), 100),
                 coalesce(p.rdb$null_flag, 0),
                 p.rdb$parameter_number
          from rdb$procedure_parameters p, rdb$fields as f, rdb$types t
          where p.rdb$procedure_name = :obj_name and
                p.rdb$parameter_name = :field_name and
                p.rdb$package_name = :pkg_name and
                f.rdb$field_name = p.rdb$field_source and
                f.rdb$field_type = t.rdb$type and
                t.rdb$field_name = 'RDB$FIELD_TYPE'
          rows 1
          into field_type_id, field_length, field_scale, field_subtype,
               field_db_domain, field_readonly, field_default, field_required, field_position;
  
          if (row_count = 1) then
            break;
        end
      end
    end else
    begin
      select f.rdb$field_type, f.rdb$field_length, f.rdb$field_scale, coalesce(f.rdb$field_sub_type, 0),
             f.rdb$field_name, (f.rdb$computed_source is not null or f.rdb$computed_blr is not null), left(trim(r.rdb$default_source), 100), coalesce(r.rdb$null_flag, 0), r.rdb$field_position
      from rdb$fields f, rdb$relation_fields r
      where f.rdb$field_name = r.rdb$field_source and
            r.rdb$system_flag = 0 and
            r.rdb$relation_name = :obj_name and
            r.rdb$field_name = :field_name
      order by r.rdb$field_position
      rows 1
      into :field_type_id, :field_length, :field_scale, :field_subtype,
           field_db_domain, field_readonly, field_default, field_required, field_position;
    end
  end else
  begin
    -- Значит передан название домена
    field_default = '';
    field_required = 0;
    field_position = 0;
    field_db_domain = lower(field_name);
    field_readonly = false;

    select f.rdb$field_type, f.rdb$field_length, f.rdb$field_scale, coalesce(f.rdb$field_sub_type, 0)
    from rdb$fields f
    where f.rdb$field_name = :field_name
    into :field_type_id, :field_length, :field_scale, :field_subtype;
  end

  if (field_type_id is null) then
    exit;

  if (field_db_domain is null) then
    field_db_domain = '';

  if (field_required is null) then
    field_required = 0;

  field_db_domain = lower(trim(field_db_domain));
  field_domain = field_db_domain;

  if (field_default is not null and left(field_default, 8) = 'DEFAULT ') then
  begin
    field_default = substring(field_default from 9);
    field_default = trim(field_default);
  end

  if (field_default = '= null') then
    field_default = null;

  if (field_default is not null and left(field_default, 1) = '=') then
  begin
    field_default = substring(field_default from 2);
    field_default = trim(field_default);
  end

  field_orig_default = field_default;

  if (field_default is not null) then
  begin
    if (left(field_default, 1) = '''') then
      field_default = substring(field_default from 2);
    if (right(field_default, 1) = '''') then
      field_default = left(field_default, char_length(field_default) - 1);
  end

  if (field_default = '= null') then
    field_default = null;

  if (field_default is not null and left(field_default, 1) = '=') then
  begin
    field_default = substring(field_default from 2);
    field_default = trim(field_default);
    if (left(field_default, 1) = '''') then
      field_default = substring(field_default from 2);
    if (right(field_default, 1) = '''') then
      field_default = left(field_default, char_length(field_default) - 1);
  end


  if (field_scale is null) then
    field_scale = 0;

  field_type = trim(case field_type_id
    when 7 then 'smallint'
    when 8 then 'integer'
    when 10 then 'float'
    when 12 then 'date'
    when 13 then 'time'
    when 23 then 'boolean'
    when 27 then 'double'
    when 35 then 'datetime'
    when 37 then 'string'
    when 16 then
      case
        when field_scale = 0 then 'bigint'
        when field_scale <= -6 then 'percent'
        else 'currency'
      end
    when 261 then
      case
        when field_subtype = 1 then 'memo'
        else 'blob'
      end
    else null
  end);

  if (field_type is null and field_db_domain = 'd_uuid') then
    field_type = 'uuid';

  if (field_type is null) then
    exception e_default 'Unknown field type: ' || field_type_id || ' for ' || coalesce(obj_name, '<Null>') || '.' || coalesce(field_name, '<Null>');

  field_size = coalesce(field_length, 0);
  if (field_type = 'string') then
    field_size = field_length / 4;

  if (field_domain starting 'rdb$') then
  begin
    if (field_type = 'string') then
    begin
      if (field_size in (10, 20, 64, 100, 255, 2048, 4096, 8100)) then
        field_domain = 'd_str_' || field_size;
      else
      field_domain = trim(case field_size
        when 1 then 'd_str_sym'
        when 3 then 'd_str_lang'
        when 50 then 'd_str_code'
        when 1024 then 'd_str_long'
        else 'varchar(' || field_size || ')'
      end);
    end
    else if (field_type = 'memo') then
      field_domain = 'd_blob_text';
    else if (field_type = 'blob') then
      field_domain = 'd_blob_binary';
    else if (field_type = 'bigint') then
      field_domain = 'd_id';
    else
      field_domain = 'd_' || field_type;
  end

  suspend;
end
