CREATE PROCEDURE HP_BANKIMPORT_CREATE_CLIENT(
    OL_C_ID DECIMAL(18,0),
    OL_CLIENT_NAME VARCHAR(255),
    OL_CLIENT_ACCOUNT VARCHAR(20),
    OL_CLIENT_BANK VARCHAR(20),
    OL_CLIENT_INN VARCHAR(20),
    UN_ID DECIMAL(18,0))
RETURNS (
    C_ID DECIMAL(18,0))
AS
declare variable short_status varchar(20);
declare variable c_type integer;
declare variable c_from_reg_id integer;
begin
  if (coalesce(ol_c_id, 0) <> 0) then
  begin
    c_id = ol_c_id;
    if (coalesce(ol_client_account, '') <> '') then
    begin
      if (not(exists(select *
                     from bankclient
                     where bc_from_c_id = :c_id and
                           bc_account_no = :ol_client_account))) then
      begin
        if (not(exists(select *
                       from banks
                       where b_id = :ol_client_bank))) then
        begin
          insert into banks (b_id, b_name)
          values (:ol_client_bank, 'без имени');
        end
        insert into bankclient (bc_from_c_id, bc_account_no, bc_name, bc_is_main, bc_is_active, bc_from_b_id)
        values (:c_id, :ol_client_account, 'Текущий счет', 1, 1, :ol_client_bank);
      end
    end
    suspend;
    exit;
  end
  c_id = 0;
  if (char_length(ol_client_name) > 100) then
  begin
    ol_client_name = left(ol_client_name, 100);
  end

  select bc_from_c_id
  from bankclient
  where bc_account_no = :ol_client_account
  order by bc_from_c_id desc
  rows 1
  into c_id;
  if (coalesce(c_id, 0) > 0) then
  begin
    suspend;
    exit;
  end

  if (ol_client_name <> '' and
      ol_client_account <> '' and
      ol_client_inn <> '') then
  begin
    select c_id
    from clients
    where c_name_fio = :ol_client_name and
          c_account = :ol_client_account and
          c_inn = :ol_client_inn
    into :c_id;
    if (coalesce(c_id, 0) > 0) then
    begin
      suspend;
      exit;
    end
  end
  if (ol_client_name <> '' and
      ol_client_account <> '' and
      ol_client_inn = '') then
  begin
    select c_id
    from clients
    where c_name_fio = :ol_client_name and
          c_account = :ol_client_account
    into :c_id;
    if (coalesce(c_id, 0) > 0) then
    begin
      suspend;
      exit;
    end
  end

  c_from_reg_id = f_hq_id();

  short_status = '';
  c_type = 0;
  if (ol_client_inn <> '') then
  begin
    c_type = 1;
    if (ol_client_name like '% ИП' or ol_client_name like 'ИП %') then
    begin
      short_status = 'ИП';
    end
  end
  else
  begin
    if (ol_client_name like '%альфа%') then
      c_type = 1;
  end

  select result
  from get_kv_id(2)
  into :c_id;

  insert into clients (c_id, c_type, c_name_fio, c_sex, c_org_type_short, c_nation, c_citizen, c_from_reg_id, c_family,
                       c_education, c_kol_family, c_inn, c_is_resident, c_army, c_doc_type, c_from_un_id, c_kol_child)
  values (:c_id, :c_type, :ol_client_name, 2, :short_status, 0, 0, :c_from_reg_id, 0, 0, 0, :ol_client_inn, 1, 0, 0,
          :un_id, 0);

  if (ol_client_bank <> '' and
      ol_client_account <> '') then
  begin
    if (not(exists(select *
                   from banks
                   where b_id = :ol_client_bank))) then
    begin
      insert into banks (b_id, b_name)
      values (:ol_client_bank, 'без имени');
    end

    insert into bankclient (bc_from_c_id, bc_account_no, bc_name, bc_is_main, bc_is_active, bc_from_b_id)
    values (:c_id, :ol_client_account, 'Текущий счет', 1, 1, :ol_client_bank);
  end
  suspend;
end
