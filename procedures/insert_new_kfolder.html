CREATE PROCEDURE INSERT_NEW_KFOLDER(
    REG_ID INTEGER,
    KC_ID DECIMAL(18,0),
    KF_DATE DATE,
    IS_YUR INTEGER,
    KF_NAME VARCHAR(100),
    KF_NO INTEGER,
    KF_OF_CODE VARCHAR(20))
RETURNS (
    KF_ID DECIMAL(18,0))
AS
begin
  select result
  from get_kv_id(1)
  into :kf_id;

  insert into kfolder (kf_id, kf_type, kf_memo, is_delete, kf_is_yur, kf_is_new)
  values (:kf_id, 0, 'FsMobile', 0, :is_yur, 0);

  if (kf_name = '') then
  begin
    kf_of_code = '';

    if (kc_id > 1000000000004) then
    begin
      select ke_code
      from krexpert
      where ke_from_p_id = :kc_id and
            ke_status = 1 and
            ke_date <= :kf_date and
            ke_date_end >= :kf_date and
            ke_is_krs <= 2 and
            ke_from_reg_id = :reg_id
      into kf_of_code;
    end
    else
    begin
      if (kc_id in (1000000000004, 1000000000002)) then
      begin
        select cast(left(reg_code, 2) as integer) || '-' || cast(right(reg_code, 2) as integer) || trim(iif(:kc_id = 1000000000004, '-0', '-БС'))
        from region
        where reg_unicode = :reg_id
        into kf_of_code;
      end
    end
    if (kf_of_code is null) then
      kf_of_code = '';
    kf_no = 0;

    if (kf_of_code <> '') then
    begin
      select max_no
      from get_max_officer_code_no(:kf_of_code)
      into :kf_no;
      kf_no = kf_no + 1;
      kf_name = trim(kf_of_code) || '-' || kf_no;
    end
  end

  insert into knaktlist (kal_from_ka_id, kal_number, kal_from_kf_id, kal_date, kal_new_kf_officer, kal_new_kf_name,
                         is_delete, kal_new_reg_id, kal_new_no, kal_new_officer_code)
  values (0, -1, :kf_id, :kf_date, :kc_id, :kf_name, 0, :reg_id, :kf_no, :kf_of_code);
  suspend;
end
