CREATE PROCEDURE HP_CANCEL_LOAN_STATUS(
    K_ID D_ID,
    UN_ID D_ID,
    IS_APP D_SMALLINT = 0)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable KL_DATE D_DATE;
declare variable KL_DATE_TO D_DATE;
begin
  res_text = '';
  if (exists(select kl_datetime from kontlog
      where kl_from_k_id = :k_id
      and kl_state = 1)
--        exists(select * from kontract where k_id = :k_id and (k_status = 1
--        or :is_app = 1))
--      and un_id <> 1010100000050 -- доступ у Гитинова Зара Тургуновна
  ) then
  begin
    kl_date = null;
    select min(cast(msg_datetime as date))
    from edoc_sign, edoc_list, edoc_msg, fsrobot_msgbox
    where eds_from_edl_id = edl_id
    and edm_from_eds_id = eds_id
    and msg_id = edm_from_msg_id
    and edl_from_edc_id = f_edoc_kind('loan.agreement','id')
    and edl_object_kind = 9
    and edl_object_id = :k_id
    into kl_date;

    -- дата утверждение
    if (kl_date is null) then
    begin
      select kl_datetime from kontlog
      where kl_from_k_id = :k_id
      and kl_state = 1
      order by kl_datetime desc
      rows 1
      into kl_date;
    end

    kl_date_to = f_date_work_nday((:kl_date + 1), 5);

    if (current_date <= kl_date_to) then
      res_text = 'Нельзя отказать в займе раньше '||f_datetostr(kl_date_to)||' либо пока клиент не подпишет Заявление';
  end
  suspend;
end
