CREATE PROCEDURE UXP$UXGRID_EXCEL_COLUMNS(
    GRID_UUID D_UUID,
    GRID_UUID_CODE D_STR_CODE = null)
RETURNS (
    UF_UUID D_UUID,
    UF_IS_PARAM D_BOOL,
    UF_FIELD_NAME D_STR_CODE,
    UF_DATA_TYPE D_STR_CODE,
    UGI_CAPTION D_STR_100,
    UGI_POSITION D_INTEGER,
    UGI_WIDTH D_INTEGER,
    UGI_GROUP_INDEX D_INTEGER,
    UGI_SUMMARY_KIND D_INTEGER,
    UGI_HINT D_STR_LONG)
AS
declare variable us_uuid d_uuid;
declare variable edi_name d_str_code;
declare variable ufr_uuid d_uuid;
declare variable ufp_uuid d_uuid;
declare variable ufp_editor d_str_code;
begin
  if (coalesce(grid_uuid, '') = '') then
    grid_uuid = uxf$uniq(grid_uuid_code);

  select ufr_uuid, ufr_us_uuid
  from ux$grid, ux$frame
  where ugr_uuid = :grid_uuid and
        ugr_ufr_uuid = ufr_uuid
  into ufr_uuid, us_uuid;

  if (us_uuid is null) then
    exception e_default 'Grid_uuid is not Grid items';

  uf_uuid = null;
  uf_field_name = '';
  ugi_caption = '';
  ugi_position = 0;
  ugi_width = 0;
  ugi_group_index = null;
  ugi_summary_kind = 0;
  ugi_hint = '';

  uf_is_param = 1;
  for select uf_uuid, uf_field_name, uf_name, uf_position, uf_data_type,
             iif(uf_edi_uuid is not null, uxf$code(uf_edi_uuid, 'code'), null)
  from ux$field
  where uf_owner = :us_uuid and
        uf_state = 0 and
        uf_is_param = 1
  order by uf_position
  into uf_uuid, uf_field_name, ugi_caption, ugi_position, uf_data_type, edi_name
  do
  begin
    if (edi_name in ('current$reverse', 'current$user')) then
      Continue;

    ufp_uuid = null;
    select ufp_uuid
    from ux$frparam
    where ufp_ufr_uuid = :ufr_uuid and
          ufp_uf_uuid = :uf_uuid and
          ufp_parent is null
    into ufp_uuid;

    if (ufp_uuid is null) then
      Continue;

    if (exists(select * from ux$paramdef
    where upd_ufp_uuid = :ufp_uuid and
          upd_uxa_uuid is null and
          upd_locked = 1)) then
      Continue;

    suspend;
  end

  uf_is_param = 0;
  for select uf_uuid, uf_field_name, ugi_caption, ugi_position, uf_data_type,
             ugi_width, ugi_group_index, ugi_summary_kind, ugi_hint
  from ux$field, ux$gritem
  where ugi_ugr_uuid = :grid_uuid and
        ugi_uf_uuid = uf_uuid and
        uf_state = 0
  order by ugi_position
  into uf_uuid, uf_field_name, ugi_caption, ugi_position, uf_data_type,
       ugi_width, ugi_group_index, ugi_summary_kind, ugi_hint
  do
  begin
    ufp_editor = '';
    select ufp_editor
    from ux$frparam
    where ufp_ufr_uuid = :ufr_uuid and
          ufp_uf_uuid = :uf_uuid and
          ufp_parent is null
    into ufp_editor;

    if (ufp_editor = '') then
      Continue;

    if (ufp_editor = 'json' or ufp_editor starting 'blob') then
      Continue;

    if (ufp_editor in ('buttonedit', 'dialogedit')) then
      uf_field_name = uf_field_name || '$text';

    if (ugi_width = 0) then
    begin
      select default_width
      from uxp$control_kind
      where editor_id = :ufp_editor
      into ugi_width;
    end

    suspend;
  end
end
