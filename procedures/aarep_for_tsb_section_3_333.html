CREATE PROCEDURE AAREP_FOR_TSB_SECTION_3_333(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KOL_DAY INTEGER,
    PLAN_OS CURR_TYPE,
    PLAN_PERC CURR_TYPE)
AS
declare variable K_ID ID_TYPE;
declare variable ZAD_OS CURR_TYPE;
declare variable ZAD_PERC CURR_TYPE;
begin
--[15:10:25] Ulan Bakachiev: 1. за выбранный период выданных не будем брать
--2. на начало периода не должно быть просрочки т.е. если отчет будем готовить за май т.е. 1 по 31 мая то на 30 апреля не должно быть просрочки
--3. считается все план погашении за выбранный период
--[15:11:32] Елена Грабовская: 1. Да, берем план по займам, выданным по состоянию на начало периода отчета
--[15:12:16] Елена Грабовская: 2. Да, берем план по займам, которые по состоянию на начало периода отчета имеют 0 дней просрочки по ОС или процентам
--[15:13:01] Елена Грабовская: 3. В отчет попадают плановые платежи по графикам займов, актуальным по состоянию на начало периода отчета
  kol_day = :rep_date_end - :rep_date_begin + 1;
  plan_os = 0;
  plan_perc = 0;

  for select oks_from_k_id
      from oksummary
      where :rep_date_begin - 1 between oks_date_begin and oks_date_end and
            :rep_date_begin - 1 <= oks_delinq_date
            and oks_close_state = 0 
      into k_id
  do
  begin

    zad_os = null;
    zad_perc = null;
    select sum(ok_os + ok_sh_os) ok_os, sum(ok_perc + ok_sh_perc) ok_perc
    from operkred
    where ok_from_k_id = :k_id and
          ok_ol_is_active = 1 and
          ok_type < 7 and
          ok_ol_date < :rep_date_begin and
          ok_sh_date between :rep_date_begin and :rep_date_end
    into zad_os, zad_perc;
    zad_os = coalesce(zad_os, 0);
    zad_perc = coalesce(zad_perc, 0);

    plan_os = plan_os + zad_os;
    plan_perc = plan_perc + zad_perc;

  end

  suspend;
end
