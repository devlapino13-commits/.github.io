CREATE PROCEDURE AAREP_SMS_REPAY_3DAY(
    REP_DATE DATE,
    REP_DAYS INTEGER,
    MAX_DELINQ INTEGER = 0)
RETURNS (
    PHONE VARCHAR(20),
    SMS_MESSAGE VARCHAR(255),
    CLIENT_ID DECIMAL(18,0),
    KONTRACT_ID D_ID,
    PUSH_MESSAGE D_STR_LONG)
AS
declare variable DB_ID integer;
declare variable K_ID decimal(18,0);
declare variable REP_AMOUNT varchar(20);
declare variable DD_STR varchar(10);
declare variable COMP_NAME varchar(20);
declare variable IS_DELINQ smallint;
declare variable IS_OK smallint;
declare variable CL_NAME STR_100;
declare variable MCL_PROJECT integer;
begin

  db_id = f_get_db_id();

  comp_name = decode(db_id, 81001, 'МБулак', 41701, 'МБулак', 94401, 'MBulak', 39801, 'MBulak', '');
  if (comp_name = '') then
    exit;

  dd_str = f_datetostr(rep_date + rep_days);

  for select ok_from_k_id, sum(ok_sh_os + ok_os + ok_sh_perc + ok_perc)
      from operkred, kontstate
      where ks_from_k_id = ok_from_k_id and
            ks_ol_is_delete = 0 and
            ks_date_begin <= :rep_date and
            ks_date_end >= :rep_date and
            ks_state = 1 and
            ok_ol_is_active = 1 and
            ok_sh_date = :rep_date + :rep_days and
            ok_type <> 8 and
            ok_type <> 7 and
            ok_from_c_id = 0
      group by ok_from_k_id
      having sum(ok_sh_os + ok_os + ok_sh_perc + ok_perc) > 0
      into k_id, rep_amount
  do
  begin
    is_delinq = null;

    -- RFCRU-4537: Новый выходной параметр добавлен, чтобы можно было корректно определить отправителя СМС (MBulak/OiVO)
    kontract_id = :k_id;

    select 1
    from oksummary
    where :rep_date between oks_date_begin and oks_date_end and
          :rep_date > oks_delinq_date and
          oks_from_k_id = :k_id and
          oks_balance > 0 and
          oks_close_state = 0
    into is_delinq;

    if (is_delinq is null) then
      is_delinq = 0;

    is_ok = 1;
    if (max_delinq > 0) then
    begin
      if (exists(select max(iif(oks_date_end > :rep_date, :rep_date, oks_date_end + 1) - oks_delinq_date)
                 from oksummary oks
                 where oks_from_k_id = :k_id and
                       oks_close_state = 0 having max(iif(oks_date_end > :rep_date, :rep_date, oks_date_end + 1) - oks_delinq_date) >= :max_delinq)) then
        is_ok = 1;
      else
        is_ok = 0;
    end

    rep_amount = cast(rep_amount as decimal(15,0));
    for select (select cont_value
                from hp_contdata_get_cell_main(0, kc_from_c_id)), kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0 and
              :is_delinq = 0 and
              :is_ok = 1
        into phone, client_id
    do
    begin
      if (phone <> '') then
      begin
        if (db_id in (39801, 41701)) then
          sms_message = comp_name || ': У вашей группы погашение ' || :dd_str || ' г.';
        if (db_id = 81001) then
        begin
          cl_name = '';

          select txt2
          from clients, hp_fio_clear(c_alternat_fio)
          where c_id = :client_id
          into cl_name;

          cl_name = coalesce(cl_name, '');
          if (cl_name = '') then
            cl_name = 'klient';

     -----RFCRU-4257 Разделение SMS на Мбулак\OIVO
         mcl_project = f_get_mcl_project(f_objkind('clients','id'), :client_id);

          if (rep_days <> 0) then
          begin
    if (mcl_project = 2) then
          begin
      -- OiVO
      sms_message = cl_name || ', skoro srok oplaty po zaimu. ' || dd_str || ' k oplate ' || :rep_amount || '.Oplatit mojno na saite https://oivokredit.ru/repayment ili v mob. prilojenii OiVO Kredit. S zabotoy o Vas OiVO Kredit!';
      push_message = 'Здравствуйте, ' || cl_name || '. Скоро срок оплаты по займу. ' || dd_str || ' к оплате ' || :rep_amount || ' рублей. ' || 'Вы можете легко и быстро оплатить на сайте компании по ссылке: https://oivokredit.ru/repayment ' || 'или в мобильном приложении OiVO Кредит. ' || 'Если у Вас есть вопросы, Ваш специалист всегда на связи и ответит на Ваши вопросы. ' || 'С заботой о Вас OiVO Кредит!';
          end
          else
          begin
      -- М Булак (по умолчанию)
      sms_message = cl_name || ', skoro srok oplaty po zaimu. ' || dd_str || ' k oplate ' || :rep_amount || '.Oplatit mojno legko na www.mbulak.ru/p ili v mob. prilojenii M Bulak. S zabotoy o Vas M BULAK!';
      push_message = 'Здравствуйте, ' || cl_name || '. Скоро срок оплаты по займу. ' || dd_str || ' к оплате ' || :rep_amount || ' рублей. ' || 'Вы можете легко и быстро оплатить на сайте компании по ссылке: www.molbulak.ru/p ' || 'или в мобильном приложении М Булак. ' || 'Если у Вас есть вопросы, Ваш специалист всегда на связи и ответит на Ваши вопросы. ' || 'С заботой о Вас М Булак!';
          end
          end
          else
          begin
            sms_message = 'Zdravstvuite!' || cl_name || ', SEGODNYA den oplaty po zaimu. Ne zabudte oplatit ' || :rep_amount || 'rub. Mojno oplatit onlain po ssylke www.molbulak.ru/p ili v ofise:)';
          end
        end
        if (db_id = 94401) then
          sms_message = comp_name || ': Hormetli mushteri, ' || :dd_str || ' tarixinde MolBulaka kredit borcunuzun novbeti odenish vaxtidir. ' || 'Odenishi ofisde ve ya E-manat sistemi ile odeye bilersiz';
        if (db_id <> 81001 or f_clientcom_allowed(client_id, 1, 1, 1) = 1) then
          suspend;
      end
    end
  end
end
