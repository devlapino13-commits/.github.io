CREATE PROCEDURE CM_KG_0009_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare kf_client_pass_no varchar(30);
declare variable c_id id_type;
declare variable kh_ownamount curr_type;
declare variable kc_max_amount curr_type;
declare variable c_curr_bonus_status integer;
declare variable pr_id d_integer;
declare variable pr_code d_str_100;
declare variable ma_refund d_smallint;
begin
  -- Разработка по новому паспорту от 06.03.2013г.     -------------------------------------------------------------------------
  --            for execute statement :sql_stat
  for select mac_from_c_id, c_name_fio, c_pass_no, mac_amount, ma_date_oper, pr_id, p.pr_code, m.ma_refund
      from miniappclient, clients, miniapp m, product p
      where mac_from_reg_id = 0 -- регион в заявках уже не фиксируются
            and
            mac_from_ma_id = :ma_id and
            mac_amount is not null and
            mac_amount > 50000 and
            mac_from_c_id = c_id and
            mac_from_ma_id = ma_id and
            ma_from_pr_id = pr_id and
            pr_product_kind <> 3 and
            ma_date_end = '31.12.9999'
      into :c_id, :res_info_text, :kf_client_pass_no, :kc_amount, :kf_disb_date, :pr_id, :pr_code, :ma_refund
  do
  begin
    res_error_is_critical = 0;
    res_error_text = '';
    if (kf_disb_date > current_date + 30) then
    begin
      res_error_text = 'Дата выдачи кредита указана не корректно (выставлена слишком поздняя дата по отношению к текущей)';
      res_error_is_critical = 1;
      suspend;
    end
    if (kc_amount > 200000 and
        pr_id not in (240, 266)) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Сумма участника не должно превышать 200тыс.';
      if (ma_refund = 1) then --board 1986
        if ((pr_code containing '.gr.' and
            pr_code containing '.mugalim.') or (pr_code containing '.gr.' and
            pr_code not containing '.par.' and
            pr_code not containing '.mugalim.') or (pr_code containing '.farmer.' and
            pr_code not containing '.zal.')) then
          if (kc_amount between 200000 and 225000) then
          begin
            res_error_is_critical = 0;
          end
      suspend;
    end
    else
    begin
      --              Проверка есть ли кредитная история в дргом ФКУ
      kh_ownamount = null; -- сумма выдачи
      select max(kh_ownamount) -- максимальное сумма в другом ФКУ
      from kibminiapp, kibhistory
      where kbm_from_c_id = :c_id and
            kh_from_kbm_id = kbm_id and
            kh_ownamount > 0 and
            kh_is_our = 0 -- 0 -кредит в другом ФКУ; 1 -кредит в своем ФКУ
      into kh_ownamount;
      if (kh_ownamount is null) then
        kh_ownamount = -1; -- если -1 то не имее кредитной истории в другом ФКУ
      --              если у кл. нету кред. истории то проверяет по таблице CLIENTHISTORY, и берет поледний полученный кредит
      if (kh_ownamount = -1) then
      begin
        select ch_amount
        from clienthistory
        where ch_from_c_id = :c_id
        order by ch_disb_date desc
        rows 1
        into kh_ownamount;
        if (kh_ownamount is null) then
          kh_ownamount = -1;
      end
      --              Проверка клиент новый или нет
      kc_max_amount = 0;
      select max(kc_amount_approved)
      from kontstate, knclient
      where ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_date_begin < :kf_disb_date
            --        and ks_from_k_id <> :k_id
            and
            ks_from_k_id = kc_from_k_id and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0
      into :kc_max_amount;
      if (kc_max_amount is null) then
        kc_max_amount = 0;
      if (kc_max_amount = 0) then -- Если новый клиент то проверяет следующее
      begin
        if (kf_disb_date < '15.01.2021' and
            kc_amount > 60000 and
            kh_ownamount = -1) then
        begin
          res_error_is_critical = 0;
          res_error_text = 'Клиент без кред. истории. не разрешено > 60 тыс.';
          suspend;
        end
        if (kf_disb_date >= '15.01.2021' and
            kc_amount > 80000 and
            kh_ownamount = -1) then
        begin
          res_error_is_critical = 0;
          res_error_text = 'Клиент без кред. истории. не разрешено > 80 тыс.';
          suspend;
        end
        else
        begin
          if (kc_amount > 100000 and
              kh_ownamount < 100000) then
          begin
            res_error_is_critical = 0;
            res_error_text = 'У клиента в другом ФКУ было до 100 тыс. Не разрешено > 100 тыс.';
            suspend;
          end
        end
      end
    end
  end
end
