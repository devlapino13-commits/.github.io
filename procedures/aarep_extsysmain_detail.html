CREATE PROCEDURE AAREP_EXTSYSMAIN_DETAIL(
    ESM_ID DECIMAL(18,0))
RETURNS (
    FIELD_NAME VARCHAR(255),
    FIELD_VALUE VARCHAR(1000))
AS
declare variable esm_full_source blob sub_type 1 segment size 80;
declare variable esm_project varchar(20);
declare variable field_code varchar(1000);
begin
  select esm_project, esm_full_source
  from extsysmain
  where esm_id = :esm_id
  into :esm_project, :esm_full_source;

  for select field_code, field_value
      from hp_extsys_parcer(:esm_full_source, '')
      into :field_code, :field_value
  do
  begin
    field_name = '';
    select esf_field_name
    from extsysfields
    where esf_project = :esm_project and
          esf_field_code = :field_code
    into :field_name;
    if (coalesce(field_name, '') = '') then
      field_name = field_code;

    suspend;
  end
end
