CREATE PROCEDURE INN_KK_FILL(
    C_ID D_ID)
RETURNS (
    FIRST_NAME D_STR_CODE,
    LAST_NAME D_STR_CODE,
    MIDDLE_NAME D_STR_CODE,
    BIRTH_DATE D_DATE,
    DOC_NO D_STR_CODE,
    PASS_SERIA D_STR_10,
    PASS_NUMBER D_STR_20,
    DOC_DATE D_DATE,
    DOC_TYPE D_INTEGER,
    BLS1_ID D_ID,
    BLS2_ID D_ID,
    BLSI1_ID D_ID,
    BLSI2_ID D_ID,
    BLSD1_ID D_ID,
    BLSD2_ID D_ID,
    NEED_RETRY D_BOOL,
    NEED_SERIAL D_BOOL,
    HTML_TXT D_BLOB_TEXT)
AS
declare variable c_name_fio d_str_100;
declare variable info_souce d_str_code;
declare variable pk_id d_integer;
declare variable name_tail d_str_code;
begin
    if (exists (select * from perstemp t where t.pte_from_c_id = :c_id)) then
    begin
      info_souce = 'perstemp';
      select pte.pte_name_fio,
             pte.pte_birth_date,
             pte.pte_doc_type,
             coalesce(pte_pass_no, ''),
             pte.pte_pass_date
      from perstemp pte
      where pte.pte_from_c_id = :c_id
      into c_name_fio,
           birth_date,
           pk_id,
           doc_no,
           doc_date;
    end
    else
    begin
      info_souce = 'clients';
      select c_name_fio,
             c_birth_date
      from clients
      where c_id = :c_id
      into c_name_fio,
           birth_date;
    end

    if ((info_souce = 'clients') or (info_souce = 'perstemp' and doc_no = ''))  then
    begin
        select distinct
                pk_id,
                cds_doc_no,
                cds_doc_date_begin
        from clientdocs cds
        inner join passkind on cds_from_pk_id = pk_id
        inner join passdoctype on pk_from_pdt_id = pdt_id
        where cds_from_c_id = :c_id
        and pdt_is_main_doc = 1
        and cds_verify_status = 2
        and cds.cds_is_delete = 0
        rows 1
        into pk_id, doc_no, doc_date;
    end

    select txt1, txt2, txt3
    from hp_fio_clear(:c_name_fio)
    into last_name, first_name, middle_name;
    name_tail = '';
    
    if (c_name_fio like '%УУЛУ%') then
      name_tail = 'УУЛУ';
    if (c_name_fio like '%КЫЗЫ%') then
      name_tail = 'КЫЗЫ';
    if (c_name_fio like '%УГЛИ%') then
       name_tail = 'УГЛИ';
    if (c_name_fio like '%КИЗИ%') then
      name_tail = 'КЫЗЫ';
    
    if (name_tail <> '') then
      if (middle_name <> '') then
        middle_name = middle_name || ' ' || name_tail;
      else
        last_name = last_name || ' ' || name_tail;

     doc_no = replace(doc_no, ' ', '');

    -- Паспорт/Загранпаспорт РФ
    if (pk_id = 12) then
    begin
        doc_type = 21;
        --Серия и номер паспорта гражданина РФ имеют формат 99 99 9999990, где:
        --9 – десятичная цифра (обязательная к вводу);
        --0 – десятичная цифра (необязательная, может отсутствовать).
        doc_no = left(doc_no, 2) || ' ' || substring(doc_no from 3 for 2) || ' ' || substring(doc_no from 5);
    end
    else
        doc_type = 10;

  select
    c_blob_pass1_id,
    c_blob_pass2_id
  from clients
  where c_id = :c_id
  into
    :bls1_id,
    :bls2_id;

  select first 1
    cds_doc_blob_1,
    cds_doc_blob_2
  from clientdocs cds
    inner join passkind
      on pk_id = cds_from_pk_id
    inner join passdoctype
      on pdt_code = 'info_inn'
        and pdt_id = pk_from_pdt_id
  where cds_from_c_id = :c_id
    and cds.cds_is_delete = 0
    and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
    order by cds_id desc
  into
    :blsi1_id,
    :blsi2_id;

  select first 1
    cds_doc_blob_1,
    cds_doc_blob_2
  from clientdocs cds
    inner join passkind
      on pk_id = cds_from_pk_id
    inner join passdoctype
      on pdt_code = 'dul_inn'
        and pdt_id = pk_from_pdt_id
  where cds_from_c_id = :c_id
    and cds.cds_is_delete = 0
    and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
    order by cds_id desc
  into
    :blsd1_id,
    :blsd2_id;

  select
    pass_seria,
    pass_number
  from split_passport_number(:pk_id, :doc_no)
  into
    :pass_seria,
    :pass_number;
  need_retry = iif(f_get_const('inn_ru_source') > 0, 1, 0);
  need_serial = iif(f_get_const('inn_pass_split') = 1, 1, 0);
  html_txt = (select html_txt from inn_kk_info(:c_id));

  suspend;
end
