CREATE PROCEDURE UXP$UXSOURCE_LIST(
    IS_TABLE D_BOOL,
    LIST_KIND D_STR_CODE)
RETURNS (
    US_CODE D_STR_CODE,
    US_NAME D_STR_100,
    US_MEMO D_STR_8100)
AS
declare variable is_pro d_boolean;
begin
  is_pro = uxf$pro();

  if (coalesce(list_kind, '') = '') then
    list_kind = 'source';

  if (list_kind not in ('new', 'source', 'all')) then
    exception e_default f_format('Undefined list kind: %1', list_kind);

  if (is_table = 1) then
  begin
    if (list_kind in ('new', 'all')) then
    begin
      for select lower(trim(r.rdb$relation_name)), trim(rdb$description)
      from rdb$relations r
      where r.rdb$system_flag = 0 and
            r.rdb$relation_type in (0, 1, 4)
      order by 1
      into us_code, us_memo
      do
      begin
        us_name = us_code;

        if (is_pro and us_code containing '$' and not(us_code starting 'ust$') and not(us_code starting 'imp$') and us_code <> 'ux$currency') then
          Continue;

        select us_name
        from ux$source
        where us_code = :us_code
        into us_name;

        if (list_kind = 'new' and row_count = 1) then
          Continue;

        suspend;
      end
    end else
    begin
      for select us_code, us_name, us_memo
      from ux$source
      where us_is_table = 1
      order by us_code
      into us_code, us_name, us_memo
      do
        suspend;
    end
  end else
  begin
    if (list_kind in ('new', 'all')) then
    begin
      for select lower(trim(p.rdb$procedure_name)), trim(rdb$description)
      from rdb$procedures p
      order by 1
      into us_code, us_memo
      do
      begin
        us_name = us_code;

        select us_name
        from ux$source
        where us_code = :us_code
        into us_name;

        if (list_kind = 'new' and row_count = 1) then
          Continue;

        suspend;
      end
    end else
    begin
      for select us_code, us_name, us_memo
      from ux$source
      where us_is_table = 0 and    
            us_kind <> -1
      order by us_code
      into us_code, us_name, us_memo
      do
        suspend;
    end
  end
end
