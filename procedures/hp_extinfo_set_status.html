CREATE PROCEDURE HP_EXTINFO_SET_STATUS(
    ID D_ID,
    STATUS D_STR_CODE,
    EXT_ID D_STR_CODE,
    REQUEST_TEXT D_STR_8100 = null,
    ANSWER_TEXT D_STR_8100 = null)
AS
declare variable status_id d_id;
declare variable k_kontract_no d_str_100;
declare variable email_msg d_blob_text;
declare variable email_subject d_str_255;
declare variable eil_object_kind integer;
declare variable eil_object_id d_id;
declare variable c_name_fio d_str_100;
declare variable kf_name d_str_100;
declare variable c_id d_id;
declare variable reg_id integer;
declare variable eic_name d_str_255;
declare variable eic_memo d_str_255;
begin
  -- процедура устанавливает статус запроса

  status_id = case status
  when 'send_ok' then 1
  when 'send_temp_error' then -1
  when 'send_error' then -2
  end;

  if (status_id is null) then
    exception e_default 'Incorrect status: '||coalesce(status, '<NULL>');

  if (id is null) then
    exception e_default 'ID is null';

  if (ext_id is null) then ext_id = '';

  update extinfolist
  set eil_status = :status_id,
      eil_ext_id = iif(coalesce(eil_ext_id, '') <> '', eil_ext_id, :ext_id)
  where eil_id = :id;

  if (status_id = -2) then
  begin
      request_text = coalesce(request_text, '');
      answer_text = coalesce(answer_text, '');

      insert into extinfoerr (eir_from_eil_id, eir_request, eir_answer)
      values (:id, :request_text, :answer_text);

      select eil_object_kind,
             eil_object_id,
             c_name_fio,
             c_id,
             eic_name,
             eic_memo
      from extinfolist
      inner join clients on c_id = eil_from_c_id
      inner join extinfocfg on eil_from_eic_id = eic_id
      where eil_id = :id
      into eil_object_kind, eil_object_id, c_name_fio, c_id, eic_name, eic_memo;

      email_msg = '';

      if (eil_object_kind = 9) then
      begin
          select kont.k_kontract_no,
                 kont.k_last_reg_id
          from kontract kont
          where kont.k_id = :eil_object_id
          into :k_kontract_no, :reg_id;

          email_msg =
            '<b>Запрос на уровне кредитного договора</b><br>' ||
            'Номер договора: <b>'||k_kontract_no||'</b><br>';
      end
      else if (eil_object_kind = 11) then
      begin
        select kal.kal_new_kf_name,
               kal.kal_new_reg_id
        from miniapp maa
        inner join knaktlist kal on kal.kal_from_kf_id = maa.ma_from_kf_id
        where maa.ma_date_end = '31.12.9999'
        and maa.ma_id = :eil_object_id
        and current_date between kal.kal_date and kal.kal_date_end
        and kal.is_delete = 0
        into :kf_name, :reg_id;

        email_msg =
          '<b>Запрос на уровне кредитной заявки</b><br>' ||
          'Номер группы: <b>'||kf_name||'</b><br>';
      end

      if (f_get_db_id() = 81001) then
      begin
        email_msg = blob_append(email_msg,
          f_format('Провайдер: <b>%1(%2)</b><br>', eic_memo, eic_name) ||
          f_format('ФИО клиента: <b>%1</b><br>', c_name_fio) ||
          f_format('Id запроса: <b>%1</b><br><br>', id) ||
          '<b>Запрос:</b> <br>' || request_text ||
          '<br><br>' ||
          '<b>Ответ:</b> <br>' || answer_text);
  
        email_subject = f_format('Ошибка при запросе в %1', eic_name);
  
        execute procedure hp_send_quick_message('email', 'ru.bki_check@molbulak.com', email_subject, email_msg, '', null, 1000000000000);
      end

      execute procedure hp_actionlist_create_check_info(:c_id, :reg_id, 1, 1000000000000);
  end
end
