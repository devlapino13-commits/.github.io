CREATE PROCEDURE HP_DO_EXPRESSAPP_ACTION_KKSEND(
    EXA_ID D_INTEGER)
RETURNS (
    RES_STATUS D_SMALLINT)
AS
declare variable exa_from_un_id id_type;
declare variable exa_from_k_id id_type;
declare variable acl_object_kind d_integer;
declare variable acl_from_rd_id d_integer;
declare variable exa_date d_date;
begin
  res_status = 0;

  select exa_from_un_id, exa_from_k_id, exa_date
  from expressapp
  where exa_id = :exa_id
  into exa_from_un_id, exa_from_k_id, exa_date;

  if (exa_from_k_id is null) then
    res_status = -1;

  if (res_status = 0) then
  begin
    acl_object_kind = f_objkind('Kontract', 'id');
    --acl_from_rd_id = f_get_rd_id('ActionGroup.Check_Loan_Data');

    if (not exists(select *
                   from actionlist
                   where acl_object_kind = :acl_object_kind and
                         acl_object_id = :exa_from_k_id and
                         acl_from_rd_id = (select rd_id
                                           from hp_get_check_loan_data(:exa_from_k_id)) and
                         acl_date >= :exa_date)) then
    begin
      -- если еще не отправлен в КК то отправляем
      select wait_status
      from hp_send_kontract_to_kk(:exa_from_k_id, :exa_from_un_id)
      into res_status;
    end
  end
  suspend;
end
