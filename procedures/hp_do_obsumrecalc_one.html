CREATE PROCEDURE HP_DO_OBSUMRECALC_ONE(
    RECALC_DATE D_DATE,
    RECALC_AL_ID D_STR_20)
AS
declare variable dt_ekv d_currency;
declare variable kt_ekv d_currency;
declare variable dt_val d_currency;
declare variable kt_val d_currency;
declare variable obs_date d_date;
declare variable obs_dt_ekv d_currency;
declare variable obs_kt_ekv d_currency;
declare variable obs_dt_val d_currency;
declare variable obs_kt_val d_currency;
begin
  dt_ekv = 0;
  kt_ekv = 0;
  dt_val = 0;
  kt_val = 0;

  in autonomous transaction do
  begin
    update obsumrecalc
    set osc_in_process = 1
    where osc_date = :recalc_date and
          osc_from_al_id = :recalc_al_id and
          osc_in_process = 0;

    when gdscode unique_key_violation do
    begin
      -- Не смогли обновить из-за того что запись уже есть
      -- игнорируем только данную ошибку: Violation of PRIMARY or UNIQUE KEY constraint "@1" on table "@2"
    end
  end

  in autonomous transaction do
  begin
    select sum(ob_summa), sum(ob_summa_val)
    from operbook
    where ob_ol_date = :recalc_date and
          ob_account_dt = :recalc_al_id and
          ob_ol_is_active = 1
    into dt_ekv, dt_val;
  
    if (dt_ekv is null) then
      dt_ekv = 0;
  
    if (dt_val is null) then
      dt_val = 0;
  
    select sum(ob_summa), sum(ob_summa_val)
    from operbook
    where ob_ol_date = :recalc_date and
          ob_account_kt = :recalc_al_id and
          ob_ol_is_active = 1
    into kt_ekv, kt_val;
  
    if (kt_ekv is null) then
      kt_ekv = 0;
  
    if (kt_val is null) then
      kt_val = 0;
  
    select obs_date, obs_dt_ekv, obs_kt_ekv, obs_dt_val, obs_kt_val
    from obsummary
    where obs_from_al_id = :recalc_al_id and
          obs_date <= :recalc_date
    order by obs_from_al_id desc, obs_date desc
    rows 1
    into obs_date, obs_dt_ekv, obs_kt_ekv, obs_dt_val, obs_kt_val;

    if (obs_date is null or obs_date < recalc_date) then
    begin
      insert into obsummary (obs_from_al_id, obs_date, obs_dt_ekv, obs_kt_ekv, obs_dt_val, obs_kt_val)
      values (:recalc_al_id, :recalc_date, :dt_ekv, :kt_ekv, :dt_val, :kt_val);
    end else
    begin
      if (obs_dt_ekv <> :dt_ekv or obs_kt_ekv <> :kt_ekv or obs_dt_val <> :dt_val or obs_kt_val <> :kt_val) then
      begin
        update obsummary
        set obs_dt_ekv = :dt_ekv,
            obs_kt_ekv = :kt_ekv,
            obs_dt_val = :dt_val,
            obs_kt_val = :kt_val
        where obs_from_al_id = :recalc_al_id and
              obs_date = :recalc_date;
      end
    end

    delete from obsumrecalc
    where osc_date = :recalc_date and
          osc_from_al_id = :recalc_al_id and
          osc_in_process = 1;
  end
end
