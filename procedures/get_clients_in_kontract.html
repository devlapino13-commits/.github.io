CREATE PROCEDURE GET_CLIENTS_IN_KONTRACT(
    K_ID DECIMAL(18,0),
    KONT_NO VARCHAR(20))
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    RES_K_ID DECIMAL(18,0),
    K_STOP_PERC DATE,
    PR_NAME VARCHAR(50))
AS
begin
  if (k_id > 0) then
  begin
    res_k_id = :k_id;
    for select kc_from_c_id, c_name_fio, k_kontract_no, k_stop_perc, pr_name
        from clients, kontract, knclient, product
        where k_id = kc_from_k_id and
              k_id = :k_id and
              pr_id = k_from_pr_id and
              kc_from_c_id = c_id and
              kc_amount_approved > 0
        into :c_id, :c_name_fio, :k_kontract_no, :k_stop_perc, :pr_name
    do
    begin
      suspend;
    end
  end
  if ((kont_no <> '') and
      (kont_no is not null)) then
  begin
    k_kontract_no = :kont_no;
    for select kc_from_c_id, c_name_fio, k_id, k_stop_perc, pr_name
        from clients, kontract, knclient, product
        where k_id = kc_from_k_id and
              k_kontract_no = :kont_no and
              pr_id = k_from_pr_id and
              kc_from_c_id = c_id and
              kc_amount_approved > 0
        into :c_id, :c_name_fio, :res_k_id, :k_stop_perc, :pr_name
    do
    begin
      suspend;
    end
  end
end
