CREATE PROCEDURE HP_KLADR_SYNC_SAVE(
    PARENT_PKI_ID D_ID = null,
    PARENT_PL_ID D_ID = null,
    PLT_JSO D_INTEGER = null,
    UN_ID D_ID = null,
    KSM_ID D_ID = null)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable pl_type d_str_20;
declare variable kladr_type d_str_20;
declare variable pl_id d_id;
declare variable plt_id d_id;
declare variable pki_id d_id;
declare variable obj_name d_str_255;
declare variable kladr_code d_str_code;
declare variable kladr_okato d_str_20;
declare variable comm d_str_long;
declare variable ksm_status d_smallint;
begin
  un_id = coalesce(un_id, uxf$usr_un(null, 0),1000000000000);
  error_text = '';

  -- процедура сохранения результатов сверки адресных справочников КЛАДР и БД
  if (parent_pki_id is null) then -- запуск основной процедуры
  begin
    select ksm_status, ksm_id
    from kladr_sync_main
    where ksm_status < 4
    into ksm_status, ksm_id;
  
    if (ksm_status is null) then
    begin
      error_text = 'Последняя сверка уже сохранена';
      suspend;
      exit;
    end 
  
    if (ksm_status < 3) then
    begin
      error_text = 'Сверка еще не произведена';
      suspend;
      exit;
    end 

    parent_pl_id = 810000000001;
    parent_pki_id = (select pki_id from places_kladr_import where pki_code = rpad('', 13, '0'));
    plt_jso = json.obj_create();

    for select plt_short_name, max(plt_id)
    from placetype
    where plt_can_select < 2
      and plt_short_name <> ''
    group by 1
    into pl_type, plt_id
    do
    begin
      json.obj_set_s(plt_jso, replace(pl_type,'.','#'), plt_id);
      json.obj_set_s(plt_jso, replace(pl_type,'.',''), plt_id);
    end

    -- обработаем изменения
    for select kst_from_pl_id, left(pki_code, 11), pki_okatd, kst_comm
    from kladr_sync_tmp, places_kladr_import
    where pki_id = kst_from_pki_id
      and kst_need_update = true
    into pl_id, kladr_code, kladr_okato, comm
    do
    begin
      update places
      set pl_code = '810' || :kladr_code,
          pl_add_code = :kladr_okato
      where pl_id = :pl_id;
  
      insert into kladr_sync_log (ksl_from_pl_id, ksl_comm, ksl_from_ksm_id)
      values (:pl_id, :comm, :ksm_id);
    end
  end

  -- вставим новые
  for select pki_id, pki_name, pki_type, left(pki_code, 11), pki_okatd, kst_from_pl_id
  from places_kladr_import
  left join kladr_sync_tmp on kst_from_pki_id = pki_id + 0
  where pki_parent = :parent_pki_id
    and right(pki_code, 2) = '00'
    and kst_from_pki_id is not null
  into pki_id, obj_name, kladr_type, kladr_code, kladr_okato, pl_id
  do
  begin
    if (pl_id is null) then
    begin
      plt_id = json.obj_get_val(plt_jso, replace(kladr_type,'.','#'), json.obj_get_val(plt_jso, replace(kladr_type,'.','')));

      if (plt_id is null) then
      begin
        error_text = 'Не найден тип: ' || coalesce(kladr_type, '<NULL>');
        suspend;
        exit;
      end

      insert into places (parent, pl_name, pl_name_komitet, pl_status, is_delete, pl_detail_kolich, pl_from_un_id, pl_code, pl_add_code)
      values (:parent_pl_id, :obj_name, '', :plt_id, 0, 0, :un_id, '810' || :kladr_code, :kladr_okato)
      returning pl_id
      into pl_id;

      insert into kladr_sync_log (ksl_from_pl_id, ksl_comm, ksl_from_ksm_id)
      values (:pl_id, 'Новый', :ksm_id);
    end 

    select error_text
    from hp_kladr_sync_save(:pki_id, :pl_id, :plt_jso, :un_id, :ksm_id)
    into error_text;

    if (error_text <> '') then
    begin
      suspend;
      exit;
    end
  end

  if (parent_pki_id is null) then -- запуск основной процедуры
    delete from kladr_sync_tmp;

  update kladr_sync_main
  set ksm_status = 4
  where ksm_id = :ksm_id;

  suspend;
end
