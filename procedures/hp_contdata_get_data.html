CREATE PROCEDURE HP_CONTDATA_GET_DATA(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    DATA_FORMAT VARCHAR(100),
    DATA_TYPE VARCHAR(100),
    DATA_STATE VARCHAR(100),
    DATA_ROW VARCHAR(100))
RETURNS (
    CONT_NAME D_STR_100,
    CONT_VALUE D_STR_255,
    CONT_STATUS D_INTEGER,
    CONT_STATUS_TXT D_STR_100)
AS
declare variable cd_type varchar(100);
declare variable kol integer;
declare variable db_id d_integer;
begin
  -- Если параметр не указан, не возращаем знаение
  db_id = f_get_db_id();
  if (coalesce(obj_id, '') = '' or obj_id = 0) then
    exit;

  if (coalesce(data_format, '') = '') then
    data_format = 'value';
  if (coalesce(data_type, '') = '') then
    data_type = 'all';
  if (coalesce(data_state, '') = '') then
    data_state = 'verified';

  if (coalesce(data_row, '') = '') then
    exception e_default 'Incorrect DATA_ROW params: ALL or ONE must be';

  if (data_row = 'one' and
      data_type = 'all') then
    exception e_default 'Incorrect params: ALL contact types with ONE rows';

  obj_kind = f_objkind(obj_kind, 'id');

  cont_name = '';
  cont_status_txt = '';
  cont_value = '';
  kol = 0;

  if (data_state = 'valid') then
  begin
    for select cont_value, cd_type, cont_status, trim(
               case
                 when cont_status = 0 then 'Добавлен'
                 when cont_status = 1 then 'Подтвержден'
                 when cont_status = 2 then 'Основной'
                 else 'Неопределен'
               end) as cont_status_txt
        from hp_contdata_get_data_tmp(:obj_kind, :obj_id, :data_type)
        order by cd_orderno, cont_status -- Берем первую по возрастанию - новые, не верифицированные в приоритете
        into :cont_value, :cd_type, :cont_status, :cont_status_txt
    do
    begin
      kol = kol + 1;
      cont_name = '';

      select rd_name
      from refdata
      where rd_code = :cd_type
      into cont_name;

      if (data_format = 'href' and cd_type in ('ContactInfo.cell_main', 'ContactInfo.cell_add')) then
        cont_value = f_uri_phone(cont_value, obj_kind, obj_id);
      else if (data_format = 'secured') then
        cont_value = f_cont_format(cont_value, cd_type, data_format);

      suspend;
      if (data_row = 'one') then
        exit;
    end
  end
  else
  begin
    for select cont_value, cd_type, cont_status, trim(
               case
                 when cont_status = 0 then 'Добавлен'
                 when cont_status = 1 then 'Подтвержден'
                 when cont_status = 2 then 'Основной'
                 else 'Неопределен'
               end) as cont_status_txt
        from hp_contdata_get_data_tmp(:obj_kind, :obj_id, :data_type)
        where iif(:data_state = 'all' or
              (:data_state = 'main' and
              cont_status = 2) or
              (:data_state = 'verified' and
              cont_status >= 1) or
              (:data_state = 'valid' and
              cont_status >= 0), 1, 0) = 1
        order by cd_orderno, cont_status desc -- Берем с наивысшим статусом, главные и верифицированные в приоритете
        into :cont_value, :cd_type, :cont_status, :cont_status_txt
    do
    begin
      kol = kol + 1;

      cont_name = '';
      select rd_name
      from refdata
      where rd_code = :cd_type
      into cont_name;

      if (data_format = 'href' and cd_type in ('ContactInfo.cell_main', 'ContactInfo.cell_add')) then
        cont_value = f_uri_phone(cont_value, obj_kind, obj_id);
      else if (data_format = 'secured') then
        cont_value = f_cont_format(cont_value, cd_type, data_format);

      if (db_id in (39801, 41701) and cd_type = 'ContactInfo.cell_main') then
      begin
        if (cont_status < 2) then
          cont_name = 'Новый: ' || :cont_name;
        else
        begin
          suspend;
          continue;
        end
      end

      suspend;
      if (data_row = 'one') then
        exit;
    end
  end

  if (kol = 0) then
  begin
    cont_name = '';
    cont_status = -3;
    cont_status_txt = 'Нет данных';
    cont_value = '';

    suspend;
  end
end
