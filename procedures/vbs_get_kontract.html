CREATE PROCEDURE VBS_GET_KONTRACT(
    KONT_NO D_STR_CODE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(50),
    K_PERCENT_EFFECT DECIMAL(18,7),
    K_PERCENT_MARKET DECIMAL(18,7),
    K_DISB_DATE DATE,
    K_KOLICH_REPAY INTEGER,
    K_AMOUNT_APPROVED CURR_TYPE,
    K_KOMMIS_PERC PERC_TYPE,
    K_PERCENT PERC_TYPE,
    K_PERCENT_ANNUAL PERC_TYPE)
AS
begin
  if (kont_no is null) then
    kont_no = '';

  for select k_id, k_kontract_no, k_percent_effect, k_percent_market, k_disb_date, k_kolich_repay, k_amount_approved,
             k_kommis_perc, k_percent, round(k_percent_per_day * 365, 7)
      from kontract
      where k_kontract_no starting :kont_no rows 100
      into :k_id, :k_kontract_no, :k_percent_effect, :k_percent_market, :k_disb_date, :k_kolich_repay,
           :k_amount_approved, :k_kommis_perc, :k_percent, :k_percent_annual
  do
  begin
    suspend;
  end
end
