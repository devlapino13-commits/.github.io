CREATE PROCEDURE HP_CHECK_PASPORT_ACTUAL_GRS(
    C_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    STATE D_SMALLINT)
AS
declare variable GC_STATUS D_SMALLINT;
declare variable GC_STATUS_DATE D_DATE;
declare variable ACTUAL_DAYS D_INTEGER;
declare variable KIB_NAME D_STR_10;
declare variable DB_ID D_INTEGER;
declare variable DAYS_TO_EXPIRE D_INTEGER;
declare variable IS_BIOGARD_ACTUAL D_BOOL;
begin
  db_id = f_get_db_id();
  kib_name = iif(db_id <> 39801, 'ГРС', 'ГБД ФЛ'); -- RFCKG-4140
  days_to_expire = iif(db_id <> 39801, 180, 1);
  res_text = '';
  actual_days = 0;
  state = 0;

  if (exists(select *
             from extinfolist eil
             inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
             where eic.eic_code = 'bioguard_info' and
                   eil.eil_from_c_id = :c_id and
                   eil.eil_date = current_date)) then
     is_biogard_actual = 1;

  is_biogard_actual = coalesce(is_biogard_actual, 0);

  if (exists(select *
             from grs_clients gc
             where gc.gc_from_c_id = :c_id and
                   gc.gc_is_last = 1)) then
  begin
    select gc.gc_status, cast(gc.gc_status_datetime as date)
    from grs_clients gc
    where gc.gc_from_c_id = :c_id and
          gc.gc_is_last = 1
          rows 1
    into gc_status, gc_status_date;
    :actual_days = current_date - gc_status_date;

    if (gc_status in (2, 3)) then
    begin
      if (:actual_days >= :days_to_expire) then
        res_text = 'Информация с ' || :kib_name || ' просроченная';
      else
      begin
        res_text = 'Информация актуальная';
        state = 1;
      end
    end
    else
    begin
      res_text = f_enum_name('grs_status', :gc_status);
      if (:actual_days < 30 and db_id <> 39801) then
        state = 1;
    end
  end
  else
  begin
    if (db_id <> 39801) then
      res_text = 'Запрос в ' || :kib_name || ' не отправлен';
    else if (db_id = 39801 and is_biogard_actual = 1) then
    begin
      gc_status = 2;
      res_text = f_enum_name('grs_status', :gc_status);
    end
    else
      res_text = 'Информация с ' || :kib_name || ' просроченная';
  end
  suspend;
end
