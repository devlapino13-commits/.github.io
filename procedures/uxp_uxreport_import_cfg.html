CREATE PROCEDURE UXP$UXREPORT_IMPORT_CFG(
    IMPORT_KIND D_SMALLINT,
    IMPORT_ID D_INTEGER,
    JSON_DATA D_BLOB_TEXT = null)
RETURNS (
    RES_US_UUID_STR D_STR_CODE,
    RES_UUID_STR D_STR_CODE,
    RES_ERROR_TEXT D_STR_LONG)
AS
declare variable rep_ux_uuid d_uuid;
declare variable uxa_uuid d_uuid;
declare variable rl_name d_str_100;
declare variable rl_hint d_str_long;
declare variable mm_uuid d_uuid;
declare variable ua_uuid d_uuid;
declare variable rl_report_kind d_smallint;
declare variable ufm_uuid d_uuid;
declare variable us_uuid d_uuid;
declare variable rl_datetime d_datetime;
declare variable uxa_fcode d_str_100;
begin
  if (coalesce(import_kind, 0) = 0) then
    exception e_message 'Не указан вариант импорта';

  mm_uuid = uxf$mainmenu();

  res_us_uuid_str = '';
  res_uuid_str = '';
  res_error_text = '';

  delete from ux$repimport
  where rim_kind = :import_kind and
        rim_id = :import_id;

  if (import_kind = 1) then -- Импорт отчета
  begin
    select rl_name_rus, rl_hint_rus, rl_report_kind, rl_datetime
    from reportlist
    where rl_id = :import_id
    into rl_name, rl_hint, rl_report_kind, rl_datetime;
  
    if (row_count = 0) then
      res_error_text = f_format('Не найден отчет с RL_ID: %1', coalesce(import_id, '<Null>'));
    else
    begin
      if (rl_report_kind in (1, 2)) then
      begin
        select rep_ux_uuid, res_uuid_str, res_error_text
        from uxp$uxreport_import_grid(:import_id, :json_data)
        into :rep_ux_uuid, :res_uuid_str, res_error_text;
      end else
      if (rl_report_kind in (3, 4)) then -- FastReport, Бух.расчеты
      begin
        select rep_ux_uuid, res_uuid_str
        from uxp$uxreport_import_frx(1, :import_id, :json_data)
        into :rep_ux_uuid, :res_uuid_str;
      end else
        res_error_text = f_format('Unknown report kind: %1', rl_report_kind);
     end
  end else
  if (import_kind = 2) then -- Импорт бланка
  begin
    rl_hint = '';
    select bln_name
    from blankname
    where bln_id = :import_id
    into rl_name;

    if (row_count = 0) then
      res_error_text = f_format('Не найден бланк № %1', coalesce(import_id, '<Null>'));
    else
    begin
      select rep_ux_uuid, res_uuid_str, res_datetime
      from uxp$uxreport_import_frx(2, :import_id, :json_data)
      into :rep_ux_uuid, :res_uuid_str, rl_datetime;
    end
  end

  if (res_error_text = '' and rep_ux_uuid is null) then
    res_error_text = 'Не удалось импортировать';

  if (res_error_text <> '') then
  begin
    suspend;
    exit;
  end

  select uuid_to_char(ux_us_uuid), ufm_uuid
  from ux$code, ux$form
  where ux_uuid = :rep_ux_uuid and
        ufm_ux_uuid = ux_uuid
  into res_us_uuid_str, ufm_uuid;

  if (ufm_uuid is not null) then
  begin
    us_uuid = null;

    select ufr_us_uuid
    from ux$frame, ux$source
    where ufr_owner = :ufm_uuid and
          ufr_us_uuid = us_uuid and
          coalesce(us_sql, '') = ''
    order by us_is_table desc, ufr_is_mainframe desc
    rows 1
    into us_uuid;

    if (us_uuid is not null) then
    begin
      update ux$code
      set ux_us_uuid = :us_uuid
      where ux_uuid = :rep_ux_uuid;
    end
  end

  select uxa_uuid, uxa_fcode
  from ux$alias
  where uxa_owner = :rep_ux_uuid
  into uxa_uuid, uxa_fcode;

  update or insert into ux$action (ua_owner, ua_uxa_uuid, ua_uxa_fcode, ua_name, ua_hint, ua_readonly)
  values (:mm_uuid, :uxa_uuid, :uxa_fcode, :rl_name, :rl_hint, 1)
  matching (ua_owner, ua_uxa_fcode)
  returning ua_uuid
  into ua_uuid;

  update ux$report
  set urt_import_kind = :import_kind,
      urt_import_id = :import_id,
      urt_import_datetime = :rl_datetime
  where urt_uuid = :ufm_uuid;

  -- если импортируется отчет, добавляем в Главное меню
  if (import_kind = 1) then
  begin
    update or insert into ux$menulist (uml_ml_id, uml_ua_uuid)
    values (:import_id, :ua_uuid)
    matching (uml_ml_id);
  end

  suspend;
end
