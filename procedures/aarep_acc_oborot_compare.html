CREATE PROCEDURE AAREP_ACC_OBOROT_COMPARE(
    REPORT_REGION_ID INTEGER,
    REPORT_YEAR INTEGER,
    REPORT_MONTH INTEGER,
    REPORT_PERIOD_KIND INTEGER,
    REPORT_PERIOD_KOLICH INTEGER,
    REPORT_SHOW_ACC_NO INTEGER,
    REPORT_SHOW_OFFICE INTEGER)
RETURNS (
    REP_ACCOUNT VARCHAR(20),
    REP_ACCOUNT_FULL VARCHAR(20),
    REP_BALANCE_NAME VARCHAR(255),
    REP_ACCOUNT_NAME VARCHAR(255),
    REP_BG_KIND_TEXT VARCHAR(30),
    REP_GROUP_NO VARCHAR(255),
    REP_SUMMA1 DECIMAL(18,4),
    REP_SUMMA2 DECIMAL(18,4),
    REP_SUMMA3 DECIMAL(18,4),
    REP_SUMMA4 DECIMAL(18,4),
    REP_SUMMA5 DECIMAL(18,4),
    REP_SUMMA6 DECIMAL(18,4),
    REP_SUMMA7 DECIMAL(18,4),
    REP_SUMMA8 DECIMAL(18,4),
    REP_SUMMA9 DECIMAL(18,4),
    REP_SUMMA10 DECIMAL(18,4),
    REP_SUMMA11 DECIMAL(18,4),
    REP_SUMMA12 DECIMAL(18,4),
    REGION_NAME VARCHAR(100),
    OFFICE_NAME VARCHAR(100))
AS
declare variable rep_account_no integer;
begin
  if (report_show_acc_no = 1) then
  begin
    for select region_name, office_name, rep_account, rep_account_no, rep_balance_name, max(rep_account_name),
               bg_kind_text, rep_group_no, sum(rep_summa1), sum(rep_summa2), sum(rep_summa3), sum(rep_summa4),
               sum(rep_summa5), sum(rep_summa6), sum(rep_summa7), sum(rep_summa8), sum(rep_summa9), sum(rep_summa10),
               sum(rep_summa11), sum(rep_summa12)
        from aarep_acc_oborot_compare2(:report_region_id, :report_year, :report_month, :report_period_kind, :report_period_kolich, :report_show_acc_no, :report_show_office)
        group by region_name, office_name, rep_account, rep_account_no, rep_balance_name, bg_kind_text, rep_group_no
        into :region_name, :office_name, :rep_account, :rep_account_no, :rep_balance_name, :rep_account_name,
             :rep_bg_kind_text, :rep_group_no, :rep_summa1, :rep_summa2, :rep_summa3, :rep_summa4, :rep_summa5,
             :rep_summa6, :rep_summa7, :rep_summa8, :rep_summa9, :rep_summa10, :rep_summa11, :rep_summa12
    do
    begin
      rep_account_full = :rep_account || '/' || cast(:rep_account_no as varchar(10));
      suspend;
    end
  end
  else
  begin
    rep_account_no = 0;
    for select region_name, office_name, rep_account, rep_balance_name, max(rep_account_name), bg_kind_text,
               rep_group_no, sum(rep_summa1), sum(rep_summa2), sum(rep_summa3), sum(rep_summa4), sum(rep_summa5),
               sum(rep_summa6), sum(rep_summa7), sum(rep_summa8), sum(rep_summa9), sum(rep_summa10), sum(rep_summa11),
               sum(rep_summa12)
        from aarep_acc_oborot_compare2(:report_region_id, :report_year, :report_month, :report_period_kind, :report_period_kolich, :report_show_acc_no, :report_show_office)
        group by region_name, office_name, rep_account, rep_balance_name, bg_kind_text, rep_group_no
        into :region_name, :office_name, :rep_account, :rep_balance_name, :rep_account_name, :rep_bg_kind_text,
             :rep_group_no, :rep_summa1, :rep_summa2, :rep_summa3, :rep_summa4, :rep_summa5, :rep_summa6, :rep_summa7,
             :rep_summa8, :rep_summa9, :rep_summa10, :rep_summa11, :rep_summa12
    do
    begin
      rep_account_full = :rep_account;
      suspend;
    end
  end
end
