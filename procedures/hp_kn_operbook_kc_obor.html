CREATE PROCEDURE HP_KN_OPERBOOK_KC_OBOR(
    KC_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    SAB_FULL_CODE_DT D_STR_255,
    SAB_FULL_CODE_KT D_STR_255)
RETURNS (
    RES_AMOUNT D_CURRENCY)
AS
declare variable sab_dt d_id;
declare variable sab_kt d_id;
declare variable acc_dt d_id;
declare variable acc_kt d_id;
declare variable tmp_amount d_currency;
declare variable sab_kind_dt d_integer;
declare variable sab_kind_kt d_integer;
begin
  select sab_id, sab_object_kind
  from sys_accbalance
  where sab_full_code = :sab_full_code_dt
  into sab_dt, sab_kind_dt;

  select sab_id, sab_object_kind
  from sys_accbalance
  where sab_full_code = :sab_full_code_kt
  into sab_kt, sab_kind_kt;

  res_amount = 0;

  if (coalesce(sab_kind_dt, 0) = 8 and
      coalesce(sab_kind_kt, 0) = 8) then
  begin
    for select acc_id from ref_account
    where acc_from_sab_id = :sab_dt and
          acc_object_kind = 8 and
          acc_object_id = :kc_id
    into acc_dt
    do
    begin
      for select acc_id from ref_account
      where acc_from_sab_id = :sab_kt and
            acc_object_kind = 8 and
            acc_object_id = :kc_id
      into acc_kt
      do
      begin
        tmp_amount = 0;
        select sum(kob_summa)
        from kn_operbook
        where kob_ol_is_active = 1 and
              kob_ol_date between :rep_date_begin and :rep_date_end and
              kob_acc_dt = :acc_dt and
              kob_acc_kt = :acc_kt
        into tmp_amount;
        if (tmp_amount is null) then tmp_amount = 0;
  
        res_amount = res_amount + tmp_amount;
      end
    end
  end

  if (coalesce(sab_kind_dt, 0) = 8 and
      coalesce(sab_kind_kt, 0) <> 8) then
  begin
    for select acc_id from ref_account
    where acc_from_sab_id = :sab_dt and
          acc_object_kind = 8 and
          acc_object_id = :kc_id
    into acc_dt
    do
    begin
      for select acc_id from ref_account
      where acc_from_sab_id = :sab_kt
      into acc_kt
      do
      begin
        tmp_amount = 0;
        select sum(kob_summa)
        from kn_operbook
        where kob_ol_is_active = 1 and
              kob_ol_date between :rep_date_begin and :rep_date_end and
              kob_acc_dt = :acc_dt and
              kob_acc_kt = :acc_kt
        into tmp_amount;
        if (tmp_amount is null) then tmp_amount = 0;
  
        res_amount = res_amount + tmp_amount;
      end
    end
  end

  if (coalesce(sab_kind_dt, 0) <> 8 and
      coalesce(sab_kind_kt, 0) = 8) then
  begin
    for select acc_id from ref_account
    where acc_from_sab_id = :sab_dt
    into acc_dt
    do
    begin
      for select acc_id from ref_account
      where acc_from_sab_id = :sab_kt and
            acc_object_kind = :sab_kind_kt and
            acc_object_id = :kc_id
      into acc_kt
      do
      begin
        tmp_amount = 0;
        select sum(kob_summa)
        from kn_operbook
        where kob_ol_is_active = 1 and
              kob_ol_date between :rep_date_begin and :rep_date_end and
              kob_acc_dt = :acc_dt and
              kob_acc_kt = :acc_kt
        into tmp_amount;
        if (tmp_amount is null) then tmp_amount = 0;
  
        res_amount = res_amount + tmp_amount;
      end
    end
  end

  if (coalesce(sab_kind_dt, 0) <> 8 and
      coalesce(sab_kind_kt, 0) <> 8) then
  begin
    for select acc_id from ref_account
    where acc_from_sab_id = :sab_dt
    into acc_dt
    do
    begin
      for select acc_id from ref_account
      where acc_from_sab_id = :sab_kt
      into acc_kt
      do
      begin
        tmp_amount = 0;
        select sum(kob_summa)
        from kn_operbook
        where kob_ol_is_active = 1 and
              kob_ol_date between :rep_date_begin and :rep_date_end and
              kob_acc_dt = :acc_dt and
              kob_acc_kt = :acc_kt
        into tmp_amount;
        if (tmp_amount is null) then tmp_amount = 0;
  
        res_amount = res_amount + tmp_amount;
      end
    end
  end


  Suspend;
end
