  for with tabl1
      as (select oks_from_k_id as k_id, oks_balance balance, :rep_date_end - oks_delinq_date as delinq_day,
                  oks_delinq_date delinq_date, 0 as zad_oko_os, 0 as zad_oko_perc, 0 as zad_oko_other,
                  null as oko_ol_date, 0 as oko_os, 0 as oko_perc, 0 as oko_other, null as date_woff_temp,0 as ox_amount_rez
          from oksummary
          where :rep_date_end between oks_date_begin and oks_date_end and
                :rep_date_end - oks_delinq_date > 0
                and oks_close_state = 0 
          union all
          select oko_from_k_id, 0 oks_balance, 0 delinq_day, null, oko_os + oko_sh_os, oko_perc + oko_sh_perc,
                 oko_other + oko_sh_other, iif(oko_type = 4, oko_ol_date, null), iif(oko_type = 4, oko_os, 0),
                 iif(oko_type = 4, oko_perc, 0), iif(oko_type = 4, oko_other, 0), oko_ol_date as date_woff_temp, 0 as ox_amount_rez
          from operkroff
          where oko_ol_is_active = 1 and
                oko_ol_date <= :rep_date_end
          union all
          select ox_object_id as k_id, 0 oks_balance, 0 delinq_day, null,0 as zad_oko_os, 0 as zad_oko_perc, 0 as zad_oko_other,
                 null as oko_ol_date, 0 as oko_os, 0 as oko_perc, 0 as oko_other, null as date_woff_temp , ox_amount_plus - ox_amount_minus - ox_amount_writeoff as ox_amount_rez
          from operextra
          where ox_ol_is_active = 1 and
              ox_from_rd_id = :oek_rd_id and
              ox_object_kind = f_objkind('kontract', 'id') and
              ox_ol_date<=:rep_date_end)

      select k_id, sum(balance), max(delinq_day), max(delinq_date), sum(zad_oko_os), sum(zad_oko_perc),
             sum(zad_oko_other), max(oko_ol_date), -sum(oko_os), -sum(oko_perc), -sum(oko_other), min(date_woff_temp),sum(ox_amount_rez)
      from tabl1
--      where k_id in (

      group by k_id
      having sum(balance) <> 0 or sum(zad_oko_os) <> 0 or sum(zad_oko_perc) <> 0 or sum(zad_oko_other) <> 0 or max(delinq_day) > 0 or sum(ox_amount_rez)<>0


      into k_id, balance, delinq_day, delinq_date, zad_oko_os, zad_oko_perc, zad_oko_other, oko_ol_date, repay_oko_os,

                     else ''
                   end;

    if (disb_reg_id_prior <> disb_reg_id) then
    begin
      disb_reg_id_prior = disb_reg_id;
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r1.reg_stat_rep_region
      from region r1, region r2
      where r1.reg_unicode = :disb_reg_id and
            r1.parent = r2.reg_unicode
      into disb_reg_name, disb_branch_name, reg_stat_rep_region;
    end

    into pr_code, pr_name;

      ox_amount_plus = 0;

      ox_amount_zadol = 0;

    poruch = 'Не определен';
    if (pr_code containing '.GR.' and
        pr_code containing '.FAMILY') then
    begin
      if (k_disb_date between '05.10.15' and '21.10.15') then
      begin
        poruch = 'Групповой без поручительства';

      begin

      end

      begin
        poruch = 'Групповой без поручительства';

      if (:k_disb_date between '15.01.15' and '30.06.15') then

        poruch = 'Групповой с поручительством';

        begin
          poruch = 'Индивидуальный с поручительством';

        begin
          poruch = 'Индивидуальный без поручительства';

      end

      if (pr_name containing 'Будущий сотрудник' or (pr_code similar to '%(.EMP.|.PP.)%')) then

        poruch = 'Индивидуальный с поручительством';
      end
      if (pr_code similar to '%(.PATENT.|.BARAKAT|.LOYAL.|.WEEK.)%') then
      begin
        poruch = 'Индивидуальный без поручительства';
      end
    end
    ------------------
    delinq_os = 0;
    delinq_perc = 0;
    delinq_fine = 0;

    repay_os = 0;
    repay_perc = 0;

    begin

        delinq_perc = 0;

    end
      select max(iif(ok_type <> 5, ok_ol_date, null)),
             -sum(iif(ok_type <> 5 /*and ok_ol_date >= :rep_date_begin*/ , ok_os, 0)),
             -sum(iif(ok_type <> 5 /*and ok_ol_date >= :rep_date_begin*/ , ok_perc, 0)),
             -sum(iif(ok_type <> 5 /*and ok_ol_date >= :rep_date_begin*/ , ok_fine, 0)),
             min(iif(ok_type = 5, ok_ol_date, null))
      from operkred
      where ok_from_k_id = :k_id and
            ok_ol_is_active = 1 and
            ok_type in (4, 9, 5) and
            ok_os + ok_perc + ok_fine < 0 and
            ok_ol_date <= :rep_date_end
      into last_date_repay, repay_os, repay_perc, repay_fine, date_woff;
      if (last_date_repay <= :k_disb_date) then
        last_date_repay = null;
      if (repay_os is null) then
        repay_os = 0;
      if (repay_perc is null) then
        repay_perc = 0;
      if (repay_fine is null) then
        repay_fine = 0;



    zadol_os_perc = balance + zad_oko_os + zad_oko_perc + sh_calc_perc;
    if (zadol_os_perc < 0) then
    continue;
    zadol_os = balance + zad_oko_os;
    zadol_perc = zad_oko_perc + sh_calc_perc;

    if (date_woff is not null and
        date_woff > date_woff_temp) then
      date_woff = date_woff_temp;
    else
      date_woff = date_woff_temp;

    --  название службы
    sbl_name = '';
    sba_date_begin = null;
    if (k_is_problem = 1) then
    begin
      select sbl_name, sba_date_begin
      from sb_akt, sb_list
      where sba_from_k_id = :k_id and
            :rep_date_end between sba_date_begin and sba_date_end and
            sba_is_delete = 0 and
            sba_status = 2 and
            sba_from_sbl_id = sbl_id
      into sbl_name, sba_date_begin;
    end

    kd_date = null;
    select kd_date
    from kontdevides
    where kd_k_new = :k_id and
          kd_date <= :rep_date_end
    into kd_date;

    avg_annuity_payment = 0;
    sh_total_pay = 0;
    sh_cnt = 0;

    begin
      select sum(bs.sh_os + bs.sh_perc), count(*)
      from hp_get_base_schedule(:k_id) bs
      where bs.sh_id >= 1 and
            bs.sh_os + bs.sh_perc > 0
      into sh_total_pay, sh_cnt;
    
      if (coalesce(sh_cnt, 0) > 0) then
        avg_annuity_payment = round(sh_total_pay / sh_cnt, 2);
      when any do
        avg_annuity_payment = null;
    end

    k = 0;
    balance_delta = balance;
    zad_oko_os_delta = zad_oko_os;
    zad_oko_perc_delta = zad_oko_perc;
    zad_oko_other_delta = zad_oko_other;

    repay_oko_os_delta = repay_oko_os;
    repay_oko_perc_delta = repay_oko_perc;
    repay_oko_other_delta = repay_oko_other;

    delinq_os_delta = delinq_os;
    delinq_perc_delta = delinq_perc;
    delinq_fine_delta = delinq_fine;

    repay_os_delta = repay_os;
    repay_perc_delta = repay_perc;
    repay_fine_delta = repay_fine;
    ox_amount_plus_delta = ox_amount_plus;
    ox_amount_zadol_delta = ox_amount_zadol;

    for select kc_from_c_id, kc_amount_approved, kc_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id, kc_amount, kc_id
    do
    begin
      cl_status = '';
      if (c_id = k_clients_leader_id) then
        cl_status = 'Лидер';
      else
      if (c_id = k_clients_cashier_id) then
        cl_status = 'Казначей';

      --      suspend;
      select c_name_fio, c_citizen, c_birth_country, c_birth_date, c_pass_no, c_pass_date, c_pass_who, coalesce(nullif(trim(c_inn), ''), c_ext_inn)
      from clients
      where c_id = :c_id
      into cli_fio, c_citizen, c_birth_country_str, c_birth_date, c_pass_no, c_pass_date, c_pass_who, c_inn;

      c_email = f_cont_info('clients', c_id, 'email', 'ok');

      if (c_citizen_prior <> c_citizen) then
      begin
        c_citizen_prior = c_citizen;
        c_citizen_str = null;
        select rd_name
        from refdata
        where rd_id = :c_citizen
        into c_citizen_str;
      end
      /*
      if (c_birth_country_prior <> c_birth_country) then
      begin
        c_birth_country_prior = c_birth_country;
        c_birth_country_str = null;
        select rd_name from refdata
        where rd_id = :c_birth_country
        into c_birth_country_str;
      end
      */

      rep_c_address_legal = '';
      rep_c_address_fact = '';
      rep_c_rodina_legal = '';
      rep_c_rodina_fact = '';
      rep_cell_no_additional = '';

      select adl_address_str, adc_phone
      from addrlist, addrclient
      where adc_from_c_id = :c_id and
            adl_id = adc_from_adl_id and
            adc_address_type = 1
      into rep_c_address_legal, ss_phone;
      if ((coalesce(ss_phone, '') <> '') and
          (position(ss_phone in rep_cell_no_additional) = 0)) then
      begin
        if (rep_cell_no_additional = '') then
          rep_cell_no_additional = ss_phone;
        else
          rep_cell_no_additional = rep_cell_no_additional || ', ' || :ss_phone;
      end

      select adl_address_str, adc_phone
      from addrlist, addrclient
      where adc_from_c_id = :c_id and
            adl_id = adc_from_adl_id and
            adc_address_type = 2
      into rep_c_address_fact, ss_phone;
      if ((coalesce(ss_phone, '') <> '') and
          (position(ss_phone in rep_cell_no_additional) = 0)) then
      begin
        if (rep_cell_no_additional = '') then
          rep_cell_no_additional = ss_phone;
        else
          rep_cell_no_additional = rep_cell_no_additional || ', ' || :ss_phone;
      end

      select adl_address_str, adc_phone
      from addrlist, addrclient
      where adc_from_c_id = :c_id and
            adl_id = adc_from_adl_id and
            adc_address_type = 4
      into rep_c_rodina_legal, ss_phone;
      if ((coalesce(ss_phone, '') <> '') and
          (position(ss_phone in rep_cell_no_additional) = 0)) then
      begin
        if (rep_cell_no_additional = '') then
          rep_cell_no_additional = ss_phone;
        else
          rep_cell_no_additional = rep_cell_no_additional || ', ' || :ss_phone;
      end

      select adl_address_str, adc_phone
      from addrlist, addrclient
      where adc_from_c_id = :c_id and
            adl_id = adc_from_adl_id and
            adc_address_type = 5
      into rep_c_rodina_fact, ss_phone;
      if ((coalesce(ss_phone, '') <> '') and
          (position(ss_phone in rep_cell_no_additional) = 0)) then
      begin
        if (rep_cell_no_additional = '') then
          rep_cell_no_additional = ss_phone;
        else
          rep_cell_no_additional = rep_cell_no_additional || ', ' || :ss_phone;
      end

      personal_sheet_phones = '';
      contact_no_ru = '';
      contact_no_rodina = '---';
       if (contact_no_rodina = '---') then
        begin
          last_cc_date_rf = null;
          contact_no_ru = '';

          select max(cc.cc_date)
          from contacts cc
          where cc_from_c_id = :c_id and
                -- cc_is_old_contact = 0 and
                coalesce(cc_cell_no, '') <> '' and
                cc_type = 0
          into :last_cc_date_rf;

          select list(cc_cell_no, ', ')
          from contacts cc
          where cc_from_c_id = :c_id and
                cc.cc_date = :last_cc_date_rf and
                coalesce(cc_cell_no, '') <> '' and
                cc_type = 0 -- Контакты в РФ
          into  contact_no_ru;
          if (contact_no_ru is null) then
            contact_no_ru = '';
          if (contact_no_ru <> '') then
            personal_sheet_phones = contact_no_ru;
        end

        if (contact_no_rodina = '---') then
        begin
          last_cc_date_rodina = null;
          contact_no_rodina = '';

          select max(cc.cc_date)
          from contacts cc
          where cc_from_c_id = :c_id and
                coalesce(cc_cell_no, '') <> '' and
                cc_type = 1
          into :last_cc_date_rodina;

          select left(list(cc_cell_no, ', '), 255)
          from contacts
          where cc_from_c_id = :c_id and
                cc_date = :last_cc_date_rodina and
                coalesce(cc_cell_no, '') <> '' and
                cc_type = 1 -- Контакты в Родина
          into  contact_no_rodina;
          if (contact_no_rodina is null) then
            contact_no_rodina = '';
          if (contact_no_rodina = '') then
            contact_no_rodina = '---';
        end
      if (contact_no_rodina is not null and contact_no_rodina <> '') then
      begin
        if (personal_sheet_phones = '') then
          personal_sheet_phones = contact_no_rodina;
        else
          personal_sheet_phones = left(personal_sheet_phones || ', ' || contact_no_rodina, 500);
      end

      --ДОбавлен по запросу Ильи Онищенко от 05.04.18
      rep_cell_no_additional = coalesce(f_cont_info('clients', c_id, 'ContactInfo.Cell_Add', 'last'), '') || rep_cell_no_additional;

      cc_kontact_cells = '';
      cc_kontact_cells_rf = '';
      cc_cell_no_old = '';

      for select coalesce(cc_name, ''),
                 trim(coalesce(cc_cell_no, '')),
                 cc_type
      from contacts cc
      where cc_from_c_id = :c_id and
            coalesce(cc_name, '') <> '' and
            trim(coalesce(cc_cell_no, '')) <> ''
      order by  cc.cc_cell_no,
                cc.cc_create_date desc
      into :cc_name1, :cc_cell_no1, :cc_type
      do
      begin
        if (cc_cell_no1 <> cc_cell_no_old) then -- 22.07.2019
        begin
            cc_cell_no_old = cc_cell_no1;

            if (cc_type = 1) then -- родина
            begin
                if (cc_kontact_cells = '') then
                  cc_kontact_cells = left(cc_name1 || ':' || cc_cell_no1, 255);
                else
                  cc_kontact_cells = left(cc_kontact_cells || ', ' || cc_name1 || ':' || cc_cell_no1, 255);
            end
            else -- в РФ
            begin
                if (cc_kontact_cells_rf = '') then
                  cc_kontact_cells_rf = left(cc_name1 || ':' || cc_cell_no1, 255);
                else
                  cc_kontact_cells_rf = left(cc_kontact_cells_rf || ', ' || cc_name1 || ':' || cc_cell_no1, 255);
            end
        end
      end


      -- Данные о поручителе
      cc_poruch_cell = '';

      for select coalesce(z_name_fio, ''), coalesce(z_phone, ''), zal.z_from_c_id
          from zalog zal
          inner join addrlist adl on adl.adl_id = zal.z_from_adl_id
          where zal.z_from_k_id = :k_id and
                zal.z_type = 0
          into poruch_fio, :poruch_phone, :poruch_c_id
      do
      begin
        if ((poruch_fio = '') or (poruch_phone = '')) then
        begin
          select coalesce(cl.c_name_fio, ''),
                 (select cont_value
                  from hp_contdata_get_cell_valid(0, :poruch_c_id))
          from clients cl
          where cl.c_id = :poruch_c_id
          into :poruch_fio, :poruch_phone;
        end
        if ((poruch_fio <> '') or (poruch_phone <> '')) then
        begin
          poruch_phone = ':' || poruch_phone;

          if (cc_poruch_cell = '') then
            cc_poruch_cell = left(poruch_fio || poruch_phone, 255);
          else
            cc_poruch_cell = left(cc_poruch_cell || ', ' || poruch_fio || poruch_phone, 255);
        end
      end

      rep_cell_no_relations = '';
      for select (select cont_value
                  from hp_contdata_get_cell_valid(0, c_id)), crel_from_rel_id
          from clients, clientrelation
          where crel_from_c_id = :c_id and
                crel_other_c_id = c_id and
                (select cont_value
                 from hp_contdata_get_cell_valid(0, c_id)) <> '' rows 10
          into :ss_phone, :crel_from_rel_id
      do
      begin
        if (position(ss_phone in rep_cell_no_relations) = 0) then
        begin
          rel_name = '';
          select rel_name
          from relationtype
          where rel_id = :crel_from_rel_id
          into :rel_name;
          if (rel_name is null) then
            rel_name = '';

          if (rel_name = '') then
          begin
            if (rep_cell_no_relations = '') then
              rep_cell_no_relations = ss_phone;
            else
              rep_cell_no_relations = rep_cell_no_relations || ', ' || ss_phone;
          end
          else
          begin
            if (rep_cell_no_relations = '') then
              rep_cell_no_relations = rel_name || ': ' || ss_phone;
            else
              rep_cell_no_relations = rep_cell_no_relations || ', ' || rel_name || ': ' || ss_phone;
          end
        end
      end

      select uniq_acc_str
      from hp_show_clientuniq_acc_no(:c_id)
      into :acc_no; -- № лицевого счета

      if (kol_cli > 1) then
        cl_dolya = cast(kc_amount as double precision) / k_amount;
      else
        cl_dolya = 1;
      k = k + 1;
      if (k < kol_cli) then
      begin
        cli_balance = round(cl_dolya * balance, 0);
        cli_zad_oko_os = round(cl_dolya * zad_oko_os, 0);
        cli_zad_oko_perc = round(cl_dolya * zad_oko_perc, 0);
        cli_zad_oko_other = round(cl_dolya * zad_oko_other, 0);

        cli_repay_oko_os = round(cl_dolya * repay_oko_os, 0);
        cli_repay_oko_perc = round(cl_dolya * repay_oko_perc, 0);
        cli_repay_oko_other = round(cl_dolya * repay_oko_other, 0);

        cli_delinq_os = round(cl_dolya * delinq_os, 0);
        cli_delinq_perc = round(cl_dolya * delinq_perc, 0);
        cli_delinq_fine = round(cl_dolya * delinq_fine, 0);

        cli_repay_os = round(cl_dolya * repay_os, 0);
        cli_repay_perc = round(cl_dolya * repay_perc, 0);
        cli_repay_fine = round(cl_dolya * repay_fine, 0);

        cli_ox_amount_plus = round(cl_dolya * ox_amount_plus, 0);
        cli_ox_amount_zadol = round(cl_dolya * ox_amount_zadol, 0);

        balance_delta = balance_delta - cli_balance;
        zad_oko_os_delta = zad_oko_os_delta - cli_zad_oko_os;
        zad_oko_perc_delta = zad_oko_perc_delta - cli_zad_oko_perc;
        zad_oko_other_delta = zad_oko_other_delta - cli_zad_oko_other;

        repay_oko_os_delta = repay_oko_os_delta - cli_repay_oko_os;
        repay_oko_perc_delta = repay_oko_perc_delta - cli_repay_oko_perc;
        repay_oko_other_delta = repay_oko_other_delta - cli_repay_oko_other;

        delinq_os_delta = delinq_os_delta - cli_delinq_os;
        delinq_perc_delta = delinq_perc_delta - cli_delinq_perc;
        delinq_fine_delta = delinq_fine_delta - cli_delinq_fine;

        repay_os_delta = repay_os_delta - cli_repay_os;
        repay_perc_delta = repay_perc_delta - cli_repay_perc;
        repay_fine_delta = repay_fine_delta - cli_repay_fine;

        ox_amount_plus_delta = ox_amount_plus_delta - cli_ox_amount_plus;
        ox_amount_zadol_delta = ox_amount_zadol_delta - cli_ox_amount_zadol;

      end else
      begin
        cli_balance = balance_delta;
        cli_zad_oko_os = zad_oko_os_delta;
        cli_zad_oko_perc = zad_oko_perc_delta;
        cli_zad_oko_other = zad_oko_other_delta;

        cli_repay_oko_os = repay_oko_os_delta;
        cli_repay_oko_perc = repay_oko_perc_delta;
        cli_repay_oko_other = repay_oko_other_delta;

        cli_delinq_os = delinq_os_delta;
        cli_delinq_perc = delinq_perc_delta;
        cli_delinq_fine = delinq_fine_delta;

        cli_repay_os = repay_os_delta;
        cli_repay_perc = repay_perc_delta;
        cli_repay_fine = repay_fine_delta;

        cli_ox_amount_plus = ox_amount_plus_delta;
        cli_ox_amount_zadol = ox_amount_zadol_delta;
      end

      --      cli_repay_total = cli_repay_oko_os + cli_zad_oko_perc + cli_repay_os  + cli_repay_perc;
      cli_repay_total = cli_repay_oko_os + cli_repay_oko_perc + cli_repay_oko_other + cli_repay_os + cli_repay_perc + cli_repay_fine;
      cli_zadol = cli_balance + cli_zad_oko_os + cli_zad_oko_perc + cli_zad_oko_other + cli_delinq_perc + cli_delinq_fine + cli_ox_amount_zadol;
      cli_repay_os_perc = cli_repay_oko_os + cli_repay_oko_perc + cli_repay_os + cli_repay_perc;

      rep_cell_no = (select cont_value
                     from hp_contdata_get_cell_valid(0, :c_id));
      begin
      cli_del_os_all = cli_delinq_os + cli_zad_oko_os;
      cli_del_perc_all = cli_delinq_perc + cli_zad_oko_perc;
      cli_del_fine_all = cli_delinq_fine + cli_zad_oko_other;
      end
      suspend;
    end
  end
