CREATE PROCEDURE AAREP_ACTIVE_FOR_FT(
    REP_DATE DATE)
RETURNS (
    "ФИО клиента" STR_100,
    "Статус" STR_100)
AS
declare variable C_ID ID_TYPE;
declare variable C_ID_OLD D_ID;
declare variable COUNTER D_INTEGER;
begin
      c_id_old = 0;
      counter = 0;

      for select c_id, status
      from
      (
          select distinct knc.kc_from_c_id as c_id,
                          'Активный займ' as status,
                          1 order_no
          from kontstate ks
          inner join knclient knc on knc.kc_from_k_id = ks.ks_from_k_id
          where ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1 and
                knc.kc_amount_approved > 0 and
                :rep_date between ks.ks_date_begin and ks.ks_date_end -- беру всех клиентлв активных кредитов
          ---------------------------------------------------------------------
          union all
          --беру клиентов у которых есть долг по внесистемке
          select kncl.kc_from_c_id,
                 'Долг на внесистемном учете',
                 2
          from
          (
              select okf.oko_from_k_id
              from operkroff okf
              where okf.oko_ol_is_active = 1 and
                    okf.oko_ol_date <= :rep_date
              group by okf.oko_from_k_id
              having sum(okf.oko_os + okf.oko_sh_os + okf.oko_perc + okf.oko_sh_perc + oko_other + oko_sh_other) > 0
          )B
          inner join knclient kncl on kncl.kc_from_k_id = B.oko_from_k_id
          and kncl.kc_amount_approved > 0
          union all
          ----------------------------------------------------------------------------
          -- активным договором АСП счиатем, если на дату отчета он активный + был подписан на момент активного займа у клиента или после подписания клиент получал заем.
          select distinct eds.eds_from_c_id,
                          'Активный договор АСП',
                          3
          from edoc_list edl
          inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
          inner join knclient kncl on kc_from_c_id = eds.eds_from_c_id
          inner join kontstate ks on kc_from_k_id = ks_from_k_id
          inner join kontract kont on kc_from_k_id = k_id
          where edl.edl_status = 1 and
                coalesce(edl.edl_date_end, '31.12.9999') >= cast(:rep_date as date) and --все клиенты с которыми был подписан договор АСП
                edl.edl_from_edc_id = f_edoc_kind('asp.agreement', 'id') and
                eds.eds_status = 1 and

                kc_amount_approved > 0 and
                ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_end >= edl_blank_date and
                ks_date_begin <= :rep_date
      ) A
      order by c_id, order_no
      into :c_id, "Статус"
  do
  begin
    if (c_id <> c_id_old) then
    begin
        counter = counter + 1;

        c_id_old = c_id;

        select cl.c_name_fio
        from clients cl
        where c_id = :c_id
        into "ФИО клиента";

        suspend;
    end
  end

  "ФИО клиента" = 'ИТОГО';
  "Статус" = counter;

  suspend;
end
