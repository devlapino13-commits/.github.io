CREATE PROCEDURE CM_RU_0251_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable obj_kind d_integer;
declare variable rd_1 d_id;
declare variable rd_2 d_id;
declare variable k_disb_date d_date;
declare variable acl_date d_date;
begin
  /*if (1 = 1) then
  begin
    res_error_text = 'Выдачи временно приостановлены';
    res_error_is_critical = 2;
    suspend;
  end*/

  obj_kind = f_objkind('kontract', 'id');
  rd_1 = f_get_rd_id('ActionGroup.Check_Loan_Data');
  rd_2 = f_get_rd_id('ActionGroup.Check_Loan_Data_Special');

  for select kc.kc_from_c_id, k.k_disb_date
      from knclient kc
      inner join kontract k on k.k_id = kc.kc_from_k_id
      inner join clients c on c.c_id = kc.kc_from_c_id
      where k.k_id = :k_id and
            kc.kc_amount_approved > 0 and
            c.c_citizen <> f_get_rd_id('Citizen.RU')
      into :c_id, :k_disb_date
  do
  begin
    acl_date = null;

    select acl.acl_date
    from actionlist acl
    where acl.acl_object_kind = :obj_kind and
          acl.acl_object_id = :k_id and
          acl.acl_status = 2 and
          (acl.acl_from_rd_id = :rd_1 or
          acl.acl_from_rd_id = :rd_2)
    order by acl.acl_date desc
    rows 1
    into :acl_date;

    if (k_disb_date > acl_date) then
    begin
      exit;
    end

    if (not exists(select *
                   from clientdocs cds
                   inner join passkind pk on pk.pk_id = cds.cds_from_pk_id
                   inner join passdoctype pdt on pdt.pdt_id = pk.pk_from_pdt_id
                   where cds.cds_from_c_id = :c_id and
                         cds.cds_is_delete = 0 and
                         cast(cds.cds_create_datetime as date) = current_date and
                         pdt.pdt_name = 'Результаты проверки по РКЛ' and
                         (coalesce(cds.cds_doc_blob_1, 0) > 0 or
                         coalesce(cds.cds_doc_blob_2, 0) > 0))) then
    begin
      res_error_text = 'Нет документа в слоте "Результаты проверки по РКЛ".';
      res_error_is_critical = 1;
      suspend;
    end
  end
end
