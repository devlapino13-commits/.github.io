CREATE PROCEDURE AAREP_STAT_DISB_TEMP2(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    K_PERIOD DECIMAL(15,2),
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_FROM_K_ID DECIMAL(18,0),
    DISB_OS DECIMAL(18,4),
    PERC_OS DECIMAL(18,4),
    KC_FROM_C_ID DECIMAL(18,0))
AS
declare variable cm_is_client_mode integer;
begin
-- operkred_fix
  select cm_is_client_mode
  from company
  into cm_is_client_mode;

  if (cm_is_client_mode = 0) then
  begin
    for select k_period,
               cast(cast((ok_os / k_amount_approved) * kc_amount_approved as decimal(18,4)) * cr_rate as decimal(18,2)),
               kc_from_k_id, kc_from_c_id,
               cast(cast((ok_os / k_amount_approved) * kc_amount_approved as decimal(18,4)) * cr_rate as decimal(18,2)),
               cast(cast(cast((ok_os / k_amount_approved) * kc_amount_approved as decimal(18,4)) * cr_rate as decimal(18,2)) * k_percent_per_day / 100 * 360 as decimal(18,2))
        from kontract, knclient, operkred, region_list(:report_region_id, 1, -1, 1), opercurr
        where kc_amount_approved > 0 and
              kc_from_k_id = k_id and
              ok_ol_date between :report_date_begin and :report_date_end and
              ok_ol_reg_id = id_list and
              k_amount_approved <> 0 and
              k_id = ok_from_k_id and
              ok_ol_is_active = 1 and
              ok_type = 8 and
              k_currency = cr_from_cu_id and
              ok_ol_date between cr_date and cr_date_end and
              ok_from_c_id = 0
        into :k_period, :kc_amount_approved, :kc_from_k_id, :kc_from_c_id, :disb_os, :perc_os
    do
    begin
      suspend;
    end
    exit;
  end

  for select k_period, kc_amount_approved * k_index_curr_rate, k_id, kc_from_c_id,
             kc_amount_approved * k_index_curr_rate,
             kc_amount_approved * k_index_curr_rate * k_percent_per_day / 100 * 360
      from kontstate, kontract, knclient, region_list(:report_region_id, 1, -1, 1)
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :report_date_begin and :report_date_end and
            ks_region_id = id_list and
            ks_from_k_id = k_id and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0
      into :k_period, :kc_amount_approved, :kc_from_k_id, :kc_from_c_id, :disb_os, :perc_os
  do
  begin
    suspend;
  end
end
