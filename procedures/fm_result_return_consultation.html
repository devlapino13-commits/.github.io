CREATE PROCEDURE FM_RESULT_RETURN_CONSULTATION(
    M_OBJ_ID D_ID,
    M_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable sign_type d_smallint;
declare variable edl_id d_id;
begin
  select arci.arci_sign_type
  from app_return_consult_info arci
  where arci.arci_id = :m_obj_id
  into :sign_type;

  if (m_kind = 'confirm_return_consultation') then
  begin
    if (:sign_type = 0) then
    begin
      select res_text
      from hp_return_consult_send_sms(:m_obj_id, :un_id)
      into :res_text;
      if (res_text = '') then
        res_text = 'СМС было отправлено на подписание.';
    end

    if (:sign_type = 1) then
    begin
      update app_return_consult
      set arc_status = 2,
          arc_status_un_id = :un_id,
          arc_status_datetime = current_timestamp
      where (arc_id = :m_obj_id);
      res_text = 'Заявление утверждено.';
    end
  end

  if (m_kind = 'decline_return_consultation') then
  begin
    select arci.arci_from_edl_id
    from app_return_consult_info arci
    where arci.arci_id = :m_obj_id
    into :edl_id;

    if (:edl_id is not null) then
      res_text = 'СМС было отправлено клиенту. Заявление не может быть отклонено.';

    else
    begin
      update app_return_consult
      set arc_status = -1,
          arc_status_un_id = :un_id,
          arc_status_datetime = current_timestamp
      where (arc_id = :m_obj_id);

      res_text = 'Заявление отклонено.';
    end
  end
  suspend;
end
