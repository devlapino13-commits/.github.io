CREATE PROCEDURE HP_OPERCASH_HISTORY_REP(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REG_ID D_ID,
    AL_ID D_STR_20,
    OL_ID D_ID,
    OL_DATE D_DATE,
    OL_IS_DELETE D_BOOL,
    OB_SUMMA_PLUS D_CURRENCY,
    OB_SUMMA_MINUS D_CURRENCY,
    CSP_ID D_INTEGER,
    CASH_NO D_INTEGER,
    K_ID_LIST D_STR_255,
    LOAN_COUNT D_INTEGER,
    OL_DOC_SIGN D_SMALLINT,
    OL_DOC_NO D_INTEGER,
    OL_OPER_TYPE D_INTEGER,
    OL_OPER_SUBTYPE D_INTEGER,
    OL_IS_IPF D_BOOL,
    OL_DOCUMENT_CLIENT_ID D_ID,
    OL_MAIN_K_ID D_ID,
    OL_USER_ID D_ID,
    OL_DATETIME D_DATETIME,
    OL_MEMO D_STR_LONG)
AS
declare variable bal_code d_str_20;
declare variable db_id d_integer;
declare variable tmp_ol_id d_id;
declare variable k_id d_id;
begin
  db_id = f_get_db_id();

  bal_code = case db_id
    when 41701 then '10001'
    when 81001 then '50-01'
    when 94401 then '10010'
  end;

  for select id_list
  from region_list(:rep_region_id, 1, 3, 1)
  into reg_id
  do
  begin
    for select al_id
    from acc_list
    where al_from_reg_id = :reg_id and
          al_from_bg_id = :bal_code
    into al_id
    do
    begin
      csp_id = null;
      cash_no = null;
      ol_is_ipf = null;

      select csp_id, csp_cash_no, iif(csp_cls_al_id = :al_id, 0, 1)
      from cashplace
      where csp_office_id = :reg_id and
            (csp_cls_al_id = :al_id or csp_ipf_al_id = :al_id)
      into csp_id, cash_no, ol_is_ipf;

      ol_date = rep_date_begin;
      while (ol_date <= rep_date_end) do
      begin
        ob_summa_plus = 0;
        ob_summa_minus = 0;
        ol_doc_sign = 1;

        for select ob_from_ol_id, 1 - ob_ol_is_active,  sum(ob_summa)
        from operbook
        where ob_ol_reg_id = :reg_id and
              ob_ol_date = :ol_date and
              ob_account_dt = :al_id
        group by 1, 2
        into ol_id, ol_is_delete, ob_summa_plus
        do
        begin
          select ol_doc_no, ol_oper_type, ol_oper_subtype, ol_document_client,
                 ol_user_id, ol_datetime, ol_memo
          from operlist
          where ol_id = :ol_id
          into ol_doc_no, ol_oper_type, ol_oper_subtype, ol_document_client_id,
               ol_user_id, ol_datetime, ol_memo;

          k_id_list = '';
          loan_count = 0;
          ol_main_k_id = null;

          for select ol_id
          from operlist
          where (ol_id = :ol_id or ol_parent_id = :ol_id)
          into tmp_ol_id
          do
          begin
            for select distinct ok_from_k_id
            from operkred
            where ok_from_ol_id = :tmp_ol_id
            into k_id
            do
            begin
              if (ol_main_k_id is null) then
                ol_main_k_id = k_id;

              if (position(k_id, k_id_list) = 0) then
              begin
                if (k_id_list <> '') then
                  k_id_list = k_id_list || ',';

                k_id_list = k_id_list || k_id;
                loan_count = loan_count + 1;
              end
            end

            for select distinct oko_from_k_id
            from operkroff
            where oko_from_ol_id = :tmp_ol_id
            into k_id
            do
            begin
              if (position(k_id, k_id_list) = 0) then
              begin
                if (k_id_list <> '') then
                  k_id_list = k_id_list || ',';

                k_id_list = k_id_list || k_id;
                loan_count = loan_count + 1;
              end
            end
          end

          suspend;
        end
  
        ob_summa_plus = 0;
        ob_summa_minus = 0;
        ol_doc_sign = -1;
  
        for select ob_from_ol_id, 1 - ob_ol_is_active,  sum(ob_summa)
        from operbook
        where ob_ol_reg_id = :reg_id and
              ob_ol_date = :ol_date and
              ob_account_kt = :al_id
        group by 1, 2
        into ol_id, ol_is_delete, ob_summa_minus
        do
        begin
          select ol_doc_no, ol_oper_type, ol_oper_subtype, ol_document_client,
                 ol_user_id, ol_datetime, ol_memo
          from operlist
          where ol_id = :ol_id
          into ol_doc_no, ol_oper_type, ol_oper_subtype, ol_document_client_id,
               ol_user_id, ol_datetime, ol_memo;

          k_id_list = '';
          loan_count = 0;
          ol_main_k_id = null;

          for select ol_id
          from operlist
          where (ol_id = :ol_id or ol_parent_id = :ol_id)
          into tmp_ol_id
          do
          begin
            for select distinct ok_from_k_id
            from operkred
            where ok_from_ol_id = :tmp_ol_id
            into k_id
            do
            begin
              if (ol_main_k_id is null) then
                ol_main_k_id = k_id;

              if (position(k_id, k_id_list) = 0) then
              begin
                if (k_id_list <> '') then
                  k_id_list = k_id_list || ',';

                k_id_list = k_id_list || k_id;
                loan_count = loan_count + 1;
              end
            end

            for select distinct oko_from_k_id
            from operkroff
            where oko_from_ol_id = :tmp_ol_id
            into k_id
            do
            begin
              if (position(k_id, k_id_list) = 0) then
              begin
                if (k_id_list <> '') then
                  k_id_list = k_id_list || ',';

                k_id_list = k_id_list || k_id;
                loan_count = loan_count + 1;
              end
            end
          end

          suspend;
        end

        ol_date = dateadd(day, 1, ol_date);
      end
    end
  end
end
