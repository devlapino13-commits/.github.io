CREATE PROCEDURE HP_SA_CONFIG_INFO
RETURNS (
    REC_NO D_INTEGER,
    SAC_ID D_INTEGER,
    SAC_CODE D_STR_20,
    SAC_SHORT_NAME D_STR_100,
    SAC_SIGN_INFO D_STR_CODE,
    SAC_IS_VISIBLE D_STR_CODE,
    SAC_FORMULA_INFO D_STR_8100,
    SAC_LINK_INFO D_STR_8100)
AS
declare variable tmp_code d_str_20;
declare variable tmp_short_name d_str_100;
declare variable sac_sign d_integer;
declare variable sac_formula d_str_4096;
begin
  for select sac_id, sac_code, sac_short_name, sac_sign, sac_formula, sac_priority_order,
             iif(sac_is_visible = 0, 'скрытое', '')
  from sa_config
  order by sac_priority_order
  into :sac_id, :sac_code, :sac_short_name, :sac_sign, :sac_formula, :rec_no, :sac_is_visible
  do
  begin
    sac_formula_info = '';
    sac_link_info = '';

    select list(sac_short_name)
    from sa_config
    where sac_formula containing :sac_code
    into sac_link_info;

    if (sac_link_info is null) then
      sac_link_info = '';

    sac_link_info = replace(sac_link_info, ',', ', ');

    sac_sign_info = '';
    if (sac_sign = 1) then
      sac_sign_info = 'плюс';
    if (sac_sign = -1) then
      sac_sign_info = '(минус)';

    sac_formula_info = sac_formula;
    for select sac_code, sac_short_name
    from sa_config
    into tmp_code, tmp_short_name
    do
      sac_formula_info = replace(sac_formula_info, tmp_code, '  ' || tmp_short_name || '  ');


    sac_formula_info = trim(sac_formula_info);
    sac_formula_info = replace(sac_formula_info, '(  ', '(');
    sac_formula_info = replace(sac_formula_info, '  )', ')');

    sac_formula_info = replace(sac_formula_info, '*', ' * ');
    sac_formula_info = replace(sac_formula_info, '/', ' / ');
    sac_formula_info = replace(sac_formula_info, '+', ' + ');
    sac_formula_info = replace(sac_formula_info, '-', ' - ');

    while (position('   ', sac_formula_info) > 0) do
      sac_formula_info = replace(sac_formula_info, '   ', '  ');

    suspend;
  end
end
