CREATE PROCEDURE FIND_HRORDERS_FIO(
    HO_ID DECIMAL(18,0))
RETURNS (
    HR_PERONAL_FIO VARCHAR(100),
    HR_DATE_BEGIN DATE,
    HR_DATE_END DATE,
    HR_REGION_NAME VARCHAR(100),
    HR_JOBNAME VARCHAR(100))
AS
begin
  for select c_name_fio, ps_date_begin, ps_date_end_in_order, reg_code || ' ' || reg_name, jl_name
      from hr_pstatus, clients, region, joblist
      where ps_from_ho_id = :ho_id and
            ps_from_jl_id = jl_id and
            ps_from_reg_id = reg_unicode and
            ps_from_c_id = c_id
      into :hr_peronal_fio, :hr_date_begin, :hr_date_end, :hr_region_name, :hr_jobname
  do
  begin
    suspend;
  end

  hr_jobname = '';
  for select c_name_fio, hrv_date_begin, hrv_date_end, hrv_direction
      from hr_voctravel, clients
      where hrv_from_ho_id = :ho_id and
            hrv_from_c_id = c_id
      into :hr_peronal_fio, :hr_date_begin, :hr_date_end, :hr_region_name
  do
  begin
    suspend;
  end

end
