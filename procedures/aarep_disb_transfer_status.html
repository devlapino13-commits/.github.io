CREATE PROCEDURE AAREP_DISB_TRANSFER_STATUS(
    PROJECT_CODE STR_100,
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ESM_ID ID_TYPE,
    REG_NAME STR_100,
    KF_NAME STR_100,
    K_KONTRACT_NO STR_100,
    K_AMOUNT_APPROVED INTEGER,
    C_NAME_FIO STR_100,
    TRANSFER_STATUS STR_255,
    ESM_AMOUNT CURR_TYPE,
    ESM_DATE DATE,
    K_DISB_DATE DATE,
    ERROR_TXT STR_255,
    CARD_NUMBER STR_100,
    CREDITID STR_100,
    ESM_DATETIME TIMESTAMP,
    MNT_OPERATION_ID STR_100,

    C_INN D_STR_20,
    C_PINCODE D_STR_20,

    CTID D_STR_100)

declare variable esm_status integer;
declare variable ma_id d_integer;
declare variable ma_date_app d_date;
declare variable k_id id_type;
declare variable c_id id_type;
declare variable pk_id d_integer;
declare variable db_id d_integer;
declare variable k_disb_kind d_integer;
begin
  db_id = f_get_db_id();
  k_is_online = '';
  ctid = '';
  if (project_code in ('moneta_disb')) then
  begin
    select pk_id
    from passkind
    inner join passdoctype on pdt_id = pk_from_pdt_id and

        inner join kontract kont on k_id = em.esm_from_k_id
        inner join kontstate ks on ks.ks_from_k_id = kont.k_id and
              ks.ks_ol_is_delete = 0 and
              ks.ks_state = 1 and
              ks.ks_date_begin between ks.ks_date_begin and ks.ks_date_end

              kal.is_delete = 0

                         end);
