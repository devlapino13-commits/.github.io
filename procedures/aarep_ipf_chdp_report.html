CREATE PROCEDURE AAREP_IPF_CHDP_REPORT(
    REP_REGION_ID D_ID)
RETURNS (
    REG_NAME STR_100,
    CLIENT_NAME STR_255,
    CONTRACT_NO STR_100,
    CLIENT_PHONE STR_100,
    SIGN_STATUS STR_20,
    SENT_STATUS STR_20,
    SPECIALIST_NAME STR_255,
    ISSUE_DATE D_DATE)
AS
declare variable K_ID D_ID;
declare variable EDOC_RES D_STR_CODE;
declare variable C_ID D_ID;
declare variable SPEC_ID D_ID;
declare variable REG_ID D_ID;
begin
  for
    select
        ks.ks_from_k_id,
        kc.kc_from_c_id,
        ks.ks_kc_id,
        ks.ks_region_id,
        k.k_disb_date,
        k.k_kontract_no
    from kontstate ks
    inner join knclient kc on kc.kc_from_k_id = ks.ks_from_k_id
    inner join kontract k on k.k_id = ks.ks_from_k_id
    inner join product pr on pr.pr_id = k.k_from_pr_id
    inner join region_list(:rep_region_id, 1, 3, 1) rl on rl.id_list = ks.ks_region_id
    where kc.kc_amount_approved > 0
      and ks.ks_date_begin between '10.11.2025' and '28.02.2026'
      and ks.ks_ol_is_delete = 0
      and ks.ks_state = 1
      and current_date between ks.ks_date_begin and ks.ks_date_end
      and pr.pr_code containing '.ipf.'
    into
        :k_id,
        :c_id,
        :spec_id,
        :reg_id,
        :issue_date,
        :contract_no
  do
  begin

    select c_name_fio
    from clients
     where c_id = :c_id
     into :client_name;

    select c_name_fio
    from clients
    where c_id = :spec_id
    into :specialist_name;

    select reg_name
    from region
    where reg_unicode = :reg_id
    into :reg_name;

    client_phone = f_cont_info('clients', :c_id, 'cell_main', 'ok');

    edoc_res = f_edoc_status('ipf.chdp.agreement', :k_id, current_date);

    if (edoc_res = 'signed') then
        sign_status = 'Подписал';

    else if (edoc_res = 'cancelled') then
       sign_status = 'Аннулирован';
    else
        sign_status = 'Не подписал';

    if (edoc_res in ('signed','unsigned')) then
        sent_status = 'Да';
    else
        sent_status = 'Нет';

    suspend;
  end
end
