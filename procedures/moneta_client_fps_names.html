CREATE PROCEDURE MONETA_CLIENT_FPS_NAMES(
    C_ID D_ID,
    KIND D_STR_CODE = 'json')
RETURNS (
    CLIENT_NAMES D_STR_255)
AS
declare variable tmp_txt d_str_100;
declare variable surname d_str_100;
declare variable name d_str_100;
declare variable patronymic1 d_str_100;
declare variable patronymic2 d_str_100;
declare variable word_num d_integer;
declare variable clients_list d_str_8100;
begin
  -- Процедура возвращает допустимые ФИО клиента для Системы Быстрых Платежей (Moneta)
  -- Возвращать нужно в виде Имя Отчество и Первая буква фамилии
  -- Пример: Анатолий Викторович �©

  if (kind not in ('json', 'list')) then
    exception e_default f_format('Invalid kind:%1', coalesce(kind, 'null'));

  select txt1, txt2, txt3
  from clients, hp_fio_clear(c_name_fio)
  where c_id = :c_id
  into surname, name, patronymic1;
  client_names = '';
  if (name <> '') then
   client_names = name;
  if (patronymic1 <> '') then
    client_names = iif(client_names <> '', client_names || ' ', '') || patronymic1;
  if (surname <> '') then
    client_names = iif(client_names <> '', client_names || ' ', '') || left(surname, 1);
  clients_list = '"' || client_names || '"';
  if (kind = 'list') then
    suspend;
  if (client_names <> replace(client_names, 'ё', 'е')) then
  begin
    client_names = replace(client_names, 'ё', 'е');
    clients_list = clients_list || ',"' || client_names || '"';
    if (kind = 'list') then
      suspend;
  end
  word_num = 1;
  surname = ''; -- Фамилия
  name = ''; -- Имя
  patronymic1 = ''; -- Отчество (последнее слово)
  patronymic2 = ''; -- Отчество (все слова после второго)
  for select res_str
      from clients, hp_split_str(c_name_fio, ' ')
      where c_id = :c_id
      into tmp_txt
  do
  begin
    if (word_num = 1) then
      surname = left(tmp_txt, 1);
    else if (word_num = 2) then
      name = tmp_txt;
    else
    begin
      patronymic1 = tmp_txt;
      if (patronymic2 <> '') then
        patronymic2 = patronymic2 || ' ';
      patronymic2 = patronymic2 || tmp_txt;

    end
    word_num = word_num + 1;
  end

  client_names = name;
  if (patronymic1 <> '') then
    client_names = client_names || ' ' || patronymic1;
  if (surname <> '') then
    client_names = client_names || ' ' || surname;

  if (position(client_names, clients_list) = 0) then
  begin
    clients_list = clients_list || ',"' || client_names || '"';
    if (kind = 'list') then
      suspend;
    if (client_names <> replace(client_names, 'ё', 'е')) then
    begin
      client_names = replace(client_names, 'ё', 'е');
      clients_list = clients_list || ',"' || client_names || '"';
      if (kind = 'list') then
        suspend;
    end
  end

  if (patronymic2 <> '') then
    client_names = client_names || ' ' || patronymic2;
  if (surname <> '') then
    client_names = client_names || ' ' || surname;

  if (position(client_names, clients_list) = 0) then
  begin
    clients_list = clients_list || ',"' || client_names || '"';
    if (kind = 'list') then
      suspend;
    if (client_names <> replace(client_names, 'ё', 'е')) then
    begin
      client_names = replace(client_names, 'ё', 'е');
      clients_list = clients_list || ',"' || client_names || '"';
      if (kind = 'list') then
        suspend;
    end
  end

  -- Добавляем подтвержденные андеррайтерами имена, ФИО как предоставили банки
  for select fcn_name
  from fps_client_names, fps_client_banks
  where fcn_from_fcb_id = fcb_id
    and fcb_from_c_id = :c_id
    and fcn_status = 2
    and fcb_status = 2
  into client_names do
  begin
    if (position(client_names, clients_list) = 0) then
    begin
      clients_list = clients_list || ',"' || client_names || '"';
      if (kind = 'list') then
        suspend;
      if (client_names <> replace(client_names, 'ё', 'е')) then
      begin
        client_names = replace(client_names, 'ё', 'е');
        clients_list = clients_list || ',"' || client_names || '"';
        if (kind = 'list') then
          suspend;
      end
    end
  end

  -- Формируем массив JSON
  if (kind = 'json') then
  begin
    client_names = '[' || clients_list || ']';
    suspend;
  end
end
