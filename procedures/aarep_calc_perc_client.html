Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE AAREP_CALC_PERC_CLIENT(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE,
    REPORT_CURRENCY VARCHAR(3))
RETURNS (
    AAREP_K_ID DECIMAL(18,0),
    AAREP_BALANCE_BEGIN DECIMAL(18,7),
    AAREP_PERC_BEGIN DECIMAL(18,7),
    AAREP_PERC_CALCED DECIMAL(18,7),
    AAREP_PERC_PAYED DECIMAL(18,7),
    AAREP_PERC_WRITEOFF DECIMAL(18,7),
    AAREP_PERC_END DECIMAL(18,7),
    AAREP_BALANCE_END DECIMAL(18,7),
    AAREP_KONTRACT_NO VARCHAR(20),
    AAREP_CLIENT_NAME VARCHAR(100),
    AAREP_IS_INDIV VARCHAR(10),
    AAREP_PRODUCT_NAME VARCHAR(50),
    AAREP_DISB_DATE DATE,
    AAREP_DISB_AMOUNT DECIMAL(18,4),
    AAREP_CALCED_ALL DECIMAL(18,4),
    K_KREDLINE_END DATE,
    AAREP_REST_KOL INTEGER,
    K_KREDLINE_END_REST DATE,
    AAREP_REST_KIND VARCHAR(100),
    K_KOMMIS_AMOUNT DECIMAL(18,4),
    K_PERCENT DECIMAL(18,7),
    K_PERCENT_LAST DECIMAL(18,7),
    K_PERC_START_DATE DATE,
    K_DISB_DATE DATE,
    AAREP_DELINQ_OS DECIMAL(18,7),
    AAREP_DELINQ_PERC DECIMAL(18,7),
    AAREP_LLC_PERC DECIMAL(18,7),
    AAREP_NONDELINQ_BALANCE DECIMAL(18,4),
    AAREP_DELINQ_DAY_OS INTEGER,
    AAREP_DELINQ_DAY_PERC INTEGER,
    AAREP_PERC_END_DELINQ DECIMAL(18,4),
    AAREP_LLC_OS_AMOUNT DECIMAL(18,4),
    AAREP_LLC_PERC_AMOUNT DECIMAL(18,4),
    AAREP_PERC_END_ALL DECIMAL(18,4),
    AAREP_REGION_NAME VARCHAR(100),
    AAREP_ACTIVE_AT_END SMALLINT,
    MAX_DELINQ_DAYS INTEGER,
    KC_FROM_C_ID ID_TYPE)
AS
declare variable aarep_loan_t integer;
declare variable kc_amount decimal(18,4);
declare variable aarep_balance_begin_t decimal(18,4);
declare variable aarep_perc_begin_t decimal(18,4);
declare variable aarep_perc_calced_t decimal(18,4);
declare variable aarep_perc_payed_t decimal(18,4);
declare variable aarep_perc_writeoff_t decimal(18,4);
declare variable aarep_perc_end_t decimal(18,4);
declare variable aarep_balance_end_t decimal(18,4);
declare variable aarep_disb_amount_t decimal(18,4);
declare variable curr_code varchar(3);
declare variable curr_prior varchar(3);
declare variable curr_rate_def decimal(18,4);
declare variable k_curr_rate decimal(18,4);
declare variable curr_rate decimal(18,4);
declare variable temp_os decimal(18,4);
declare variable temp_perc decimal(18,4);
declare variable delinq_os decimal(18,4);
declare variable delinq_perc decimal(18,4);
declare variable sh_delinq_begin date;
declare variable sh_dd_min date;
declare variable k_kommis_tmp decimal(18,4);
declare variable dolya double precision;
declare variable k_from_kf_id id_type;
begin
  aarep_llc_perc_amount = 0;
  curr_prior = '000';

  for select aarep_k_id, sum(aarep_balance_begin), sum(aarep_perc_begin), sum(aarep_perc_calced),
             sum(aarep_perc_payed), sum(aarep_perc_writeoff), sum(aarep_perc_end), sum(aarep_balance_end),
             aarep_disb_amount, aarep_disb_date
      from aarep_calc_perc_list(:report_region_id, :report_date_begin, :report_date_end, :report_currency)
      group by aarep_k_id, aarep_disb_amount, aarep_disb_date
      into :aarep_k_id, :aarep_balance_begin_t, :aarep_perc_begin_t, :aarep_perc_calced_t, :aarep_perc_payed_t,
           :aarep_perc_writeoff_t, :aarep_perc_end_t, :aarep_balance_end_t, :aarep_disb_amount_t, :aarep_disb_date
  do
  begin

    select k_kontract_no, k_index_curr_code, k_index_curr_rate, k_kredline_end, k_percent, k_disb_date, -- такое поле есть
           pr_loan_type, pr_name, k_from_kf_id
    from kontract, product
    where k_id = :aarep_k_id and
          k_from_pr_id = pr_id
    into :aarep_kontract_no, :curr_code, :k_curr_rate, :k_kredline_end, :k_percent, :k_disb_date, :aarep_loan_t,
         :aarep_product_name, :k_from_kf_id;

    select reg_code || ' ' || reg_name
    from knaktlist, region
    where knaktlist.is_delete = 0 and
          kal_from_kf_id = :k_from_kf_id and
          :report_date_end between kal_date and kal_date_end and
          kal_new_reg_id = reg_unicode
    into :aarep_region_name;

    aarep_rest_kol = 0;
    aarep_rest_kind = '';
    k_kredline_end_rest = :k_kredline_end;
    select count(*)
    from operrest, operlist
    where or_from_ol_id = ol_id and
          operlist.is_delete = 0 and
          ol_date <= :report_date_end and
          or_from_k_id = :aarep_k_id and
          or_date_end >= :report_date_end
    into :aarep_rest_kol;

    if (aarep_rest_kol is null) then
      aarep_rest_kol = 0;
    if (aarep_rest_kol > 0) then
    begin
      aarep_rest_kind = 'срок';
      select ok_sh_date --, sum(ok_os+ok_sh_os)
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :aarep_k_id and
            ok_type <= 2 and
            ok_from_c_id = 0
      group by ok_sh_date
      having sum(ok_os + ok_sh_os) > 0
      order by ok_sh_date desc
      rows 1
      into :k_kredline_end_rest; --, :TEMP;
    end

    k_perc_start_date = :k_disb_date;
    k_percent_last = :k_percent;

    aarep_llc_perc = 0;
    select lk_percent
    from llc_kont
    where lk_from_k_id = :aarep_k_id and
          lk_date <= :report_date_end and
          lk_date_end >= :report_date_end rows 1
    into :aarep_llc_perc;
    if (aarep_llc_perc is null) then
      aarep_llc_perc = 0;

    temp_os = 0;
    temp_perc = 0;
    sh_delinq_begin = null;
    delinq_os = 0;
    delinq_perc = 0;
    k_kommis_tmp = 0;
    k_kommis_amount = 0;
    aarep_delinq_os = 0;
    aarep_delinq_perc = 0;

    if (exists(select *
               from oksummary
               where :report_date_end between oks_date_begin and oks_date_end and
                     :report_date_end >= oks_delinq_date and
                     oks_balance > 0 and
                     oks_from_k_id = :aarep_k_id and
                     oks_close_state = 0)) then
      for select ok_sh_date, sum(ok_os + ok_sh_os), sum(ok_perc + ok_sh_perc)
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :aarep_k_id and
                ok_type < 7 and
                ok_ol_date <= :report_date_end and
                ok_sh_date <= :report_date_end and
                ok_from_c_id = 0
          group by ok_sh_date
          order by ok_sh_date desc
          into :sh_dd_min, :temp_os, :temp_perc
      do
      begin
        if ((temp_os >= 0.01) or (temp_perc >= 0.01)) then
        begin
          sh_delinq_begin = sh_dd_min;
          delinq_os = delinq_os + temp_os;
          delinq_perc = delinq_perc + temp_perc;
        end
      end

    select sum(ok_kommis)
    from operkred
    where ok_ol_is_active = 1 and
          ok_type = 1 and
          ok_from_k_id = :aarep_k_id and
          ok_from_c_id = 0
    into k_kommis_tmp;
    if (k_kommis_tmp is null) then
      k_kommis_tmp = 0;

    aarep_calced_all = 0;
    if (curr_prior <> :curr_code) then
    begin
      curr_rate = 1;
      select cr_rate
      from opercurr
      where cr_from_cu_id = :curr_code and
            cr_date <= :report_date_end and
            cr_date_end >= :report_date_end
      into :curr_rate;
      if (curr_rate is null) then
        curr_rate = 1;
      curr_prior = :curr_code;
    end
    curr_rate_def = curr_rate;
    if (curr_rate < k_curr_rate) then
      curr_rate_def = k_curr_rate;

    if (aarep_loan_t = 1) then
      aarep_is_indiv = 'Индив.';
    else
      aarep_is_indiv = 'Группа';

    max_delinq_days = null;
    select max(iif(oks_date_end <= :report_date_end, oks_date_end + 1, :report_date_end) - oks_delinq_date) as max_del_days
    from oksummary
    where oks_from_k_id = :aarep_k_id and
          oks_close_state = 0 and
          oks_delinq_date <= :report_date_end and
          oks_date_begin <= :report_date_end
    into max_delinq_days;
    if (max_delinq_days is null or max_delinq_days < 0) then
      max_delinq_days = 0;

    aarep_active_at_end = 0;
    if (aarep_balance_end_t <> 0) then
      aarep_active_at_end = 1;

    for select c_name_fio, kc_amount_approved, kc_from_c_id
        from knclient, clients
        where kc_amount_approved is not null and
              kc_amount_approved > 0 and
              kc_from_k_id = :aarep_k_id and
              kc_from_c_id = c_id
        into :aarep_client_name, :kc_amount, kc_from_c_id
    do
    begin
      dolya = cast(kc_amount as double precision) / aarep_disb_amount_t;
      aarep_disb_amount = :kc_amount;
      aarep_delinq_os = cast(dolya * :delinq_os as curr_type);
      aarep_delinq_perc = cast(dolya * :delinq_perc as curr_type);
      k_kommis_amount = cast(dolya * :k_kommis_tmp as curr_type);
      aarep_balance_begin = cast(dolya * :aarep_balance_begin_t as curr_type);
      aarep_perc_begin = cast(dolya * :aarep_perc_begin_t as curr_type);
      aarep_perc_calced = cast(dolya * :aarep_perc_calced_t as curr_type);
      aarep_perc_payed = cast(dolya * :aarep_perc_payed_t as curr_type);
      aarep_perc_writeoff = cast(dolya * :aarep_perc_writeoff_t as curr_type);
      aarep_perc_end = cast(dolya * :aarep_perc_end_t as curr_type);
      aarep_balance_end = cast(dolya * :aarep_balance_end_t as curr_type);

      if (curr_rate_def <> 1) then
      begin
        aarep_balance_begin = cast(:aarep_balance_begin * :curr_rate_def as decimal(15,2));
        aarep_perc_begin = cast(:aarep_perc_begin * :curr_rate_def as decimal(15,2));
        aarep_perc_calced = cast(:aarep_perc_calced * :curr_rate_def as decimal(15,2));
        aarep_perc_payed = cast(:aarep_perc_payed * :curr_rate_def as decimal(15,2));
        aarep_perc_writeoff = cast(:aarep_perc_writeoff * :curr_rate_def as decimal(15,2));
        aarep_perc_end = cast(:aarep_perc_end * :curr_rate_def as decimal(15,2));
        aarep_balance_end = cast(:aarep_balance_end * :curr_rate_def as decimal(15,2));
        aarep_delinq_os = cast(:aarep_delinq_os * :curr_rate_def as decimal(15,2));
        aarep_delinq_perc = cast(:aarep_delinq_perc * :curr_rate_def as decimal(15,2));
      end

      aarep_perc_end_all = aarep_perc_end;
      if (aarep_delinq_os > aarep_balance_end) then
        aarep_delinq_os = aarep_balance_end; -- добавлено от 07.11.2014 по письму Санжара
      aarep_nondelinq_balance = aarep_balance_end - aarep_delinq_os;
      aarep_perc_end_delinq = 0;
      aarep_delinq_day_os = 0;
      if (sh_delinq_begin is not null) then
      begin
        if (aarep_delinq_os > 0.01) then
          aarep_delinq_day_os = :report_date_end - sh_delinq_begin;
        if (aarep_delinq_perc > 0.01) then
        begin
          aarep_delinq_day_perc = :report_date_end - sh_delinq_begin;
          if (aarep_perc_end > aarep_delinq_perc) then
          begin
            aarep_perc_end_delinq = aarep_delinq_perc;
            aarep_perc_end = aarep_perc_end - aarep_delinq_perc;
          end
          else
          begin
            aarep_perc_end_delinq = aarep_perc_end;
            aarep_perc_end = 0;
          end
        end
      end
      else
      begin
        aarep_delinq_day_os = 0;
        aarep_delinq_day_perc = 0;
      end

      aarep_llc_os_amount = cast(aarep_balance_end / 100 as decimal(18,4));
      aarep_llc_os_amount = cast(aarep_llc_os_amount * aarep_llc_perc as decimal(18,2));

      aarep_llc_perc_amount = 0;
      if (k_disb_date >= '28.02.2012') then
      begin
        aarep_llc_perc_amount = cast(aarep_perc_end_all / 100 as decimal(18,4));
        aarep_llc_perc_amount = cast(aarep_llc_perc_amount * aarep_llc_perc as decimal(18,2));
      end

      suspend;
    end
  end
end
