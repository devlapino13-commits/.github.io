CREATE PROCEDURE AAREP_LAST_MINIAPPNEW_PERIOD(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    REGION_NAME VARCHAR(100),
    REPORT_MAX_DATE DATE,
    MA_SUMMA_MINUS DECIMAL(18,4),
    MA_DATE_OPER DATE,
    MA_PLAN DECIMAL(18,4),
    MA_PLAN_SALDO DECIMAL(18,4))
AS
declare variable reg_id integer;
declare variable ma_from_reg_id integer;
declare variable ma_id integer;
declare variable ma_number integer;
declare variable ma_is_active integer;
declare variable ma_kf_id decimal(18,0);
declare variable kol integer;
declare variable reg_id_prior integer;
declare variable ma_id_prior integer;
declare variable ma_is_new integer;
begin
  for select region_name, report_max_date, sum(ma_summa_minus), ma_date_oper, sum(ma_plan)
      from aarep_last_man_period_temp(:report_region_id, :report_date_end)
      group by region_name, report_max_date, ma_date_oper
      order by region_name, ma_date_oper
      into :region_name, :report_max_date, :ma_summa_minus, :ma_date_oper, :ma_plan
  do
  begin
    ma_plan_saldo = ma_plan - ma_summa_minus;
    suspend;
  end
end
