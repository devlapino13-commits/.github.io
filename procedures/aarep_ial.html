CREATE PROCEDURE AAREP_IAL(
    REP_REGION_ID ID_TYPE,
    REP_DATE_END DATE)
RETURNS (
    K_ID ID_TYPE,
    DELINQ_DAYS INTEGER,
    K_BALANCE CURR_TYPE,
    K_KONTRACT_NO VARCHAR(50),
    KF_NAME VARCHAR(100),
    REG_NAME VARCHAR(50),
    K_AMOUNT_APPROVED CURR_TYPE,
    PR_NAME VARCHAR(50),
    PR_LOAN_TYPE SMALLINT,
    PR_LOAN_TYPE_STR VARCHAR(10),
    KCP_PERIOD INTEGER,
    KCP_PERC_SALDO CURR_TYPE,
    KCP_KOMMIS_SALDO CURR_TYPE)
AS
begin
--IAL - Information on active loans

  kcp_period = extract(year from :rep_date_end) || extract(month from :rep_date_end);

  /*iif(extract(month from :rep_date_end) = 1, -->>next
                     extract(year from :rep_date_end) -1,
                     extract(year from :rep_date_end)) ||
                     iif(extract(month from :rep_date_end)-1 = 0, 12,
                         extract(month from :rep_date_end)-1); --<<end */
  for select k_id, kal.kal_new_kf_name, k_kontract_no, pr.pr_name, pr.pr_loan_type, reg.reg_name, k_amount_approved
      from kontstate ks, kontract k, knaktlist kal, region reg, product pr, region_list(:rep_region_id, 1, -1, 1)
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            kal.is_delete = 0 and
            kal_new_reg_id = reg_unicode and
            k.k_from_pr_id = pr_id and
            reg_unicode = id_list
      into k_id, kf_name, k_kontract_no, pr_name, pr_loan_type, reg_name, k_amount_approved
  do
  begin
    pr_loan_type_str = '';
    if (pr_loan_type = 0) then
      pr_loan_type_str = 'Группа';
    if (pr_loan_type = 1) then
      pr_loan_type_str = 'Индив.';

    select oks_balance, iif(oks_delinq_date < :rep_date_end, :rep_date_end - oks_delinq_date, 0)
    from oksummary oks
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_from_k_id = :k_id
          and oks_close_state = 0 
    into k_balance, delinq_days;

    kcp_perc_saldo = 0;
    kcp_kommis_saldo = 0;

    select kcp.kcp_perc_saldo, kcp.kcp_kommis_saldo
    from kontcalcperc kcp
    where kcp.kcp_period = :kcp_period and
          kcp.kcp_from_k_id = :k_id
    into kcp_perc_saldo, kcp_kommis_saldo;

    suspend;
  end
end
