CREATE PROCEDURE AAREP_CHECK_FINPOLIS_TMP_639(
    REP_C_ID D_ID)
RETURNS (
    FPS_ID D_INTEGER,
    C_ID D_ID,
    C_NAME_FIO D_STR_100,
    FPS_FIO D_STR_255,
    C_BIRTH_DATE D_DATE,
    C_PASS_NO D_STR_CODE,
    C_INN D_STR_20,
    C_ADDRESS D_STR_255,
    FOUND_PASS_NO D_SMALLINT,
    FOUND_INN_NO D_SMALLINT,
    FOUND_CLIENT_FIO D_SMALLINT,
    FOUND_BIRTH_DATE D_SMALLINT)
AS
declare variable c_index_c_pass_no d_str_code;
declare variable db_id d_integer;
begin
  db_id = f_get_db_id();
  if (coalesce(:rep_c_id, 0) = 0) then
  begin
    for with cte_finpolis
        as (select fp.fps_id, fp.birthdate,
                   replace((coalesce(fp.docseria, '') || coalesce(fp.docnumber, '')), ' ', '') as fp_pass_no,
                   trim(coalesce(fp.inn_flip, '')) as fp_inn,
                   fp.fps_from_fl_id
            from finpolis_639 fp
            where fps_fio > ''),
        cte_no_address
        as (select fp.fps_id as fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn,
                   iif(fp.birthdate = cl.c_birth_date, 1, 0) as found_birth_date, 1 as kind
            from clients cl
            inner join cte_finpolis fp on cl.c_index_c_pass_no = fp.fp_pass_no and
                  fp.fp_pass_no > ''

            union all

            select fp.fps_id as fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn,
                   iif(fp.birthdate = cl.c_birth_date, 1, 0) as found_birth_date, 2 as kind
            from clients cl
            inner join cte_finpolis fp on cl.c_inn = fp.fp_inn and
                  fp.fp_inn > ''

            union all

            select fp.fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn,
                   iif(fp.birthdate = cl.c_birth_date, 1, 0) as found_birth_date, 3 as kind
            from clients cl, cte_finpolis fp, fiolist fl1, fiolist fl2
            where fl1.fl_id = fp.fps_from_fl_id
              and fl2.fl_id in (cl.c_from_fl_id, cl.c_name_eng_fl_id, cl.c_alternat_fl_id)
              and fl1.fl_name1 = fl2.fl_name1
              and fl1.fl_name2 = fl2.fl_name2
              and (fl1.fl_name3 = fl2.fl_name3 or (:db_id = 81001 and fp.birthdate = cl.c_birth_date)) -- В РФ кроме ФИ должно совпасть отчество или ДР

            union all

            select fp.fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn,
                   iif(fp.birthdate = cll.cll_birth_date, 1, 0) as found_birth_date, 3 as kind
            from clients cl, clientlog cll, cte_finpolis fp, fiolist fl1, fiolist fl2
            where cll.cll_from_c_id = c_id
              and fl1.fl_id = fp.fps_from_fl_id
              and fl2.fl_id = cll.cll_from_fl_id
              and fl1.fl_name1 = fl2.fl_name1
              and fl1.fl_name2 = fl2.fl_name2
              and (fl1.fl_name3 = fl2.fl_name3 or (:db_id = 81001 and fp.birthdate = cl.c_birth_date)) -- В РФ кроме ФИ должно совпасть отчество или ДР

            union all

            -- Проверка clients через f_levenstain2, дата рождения должна совпадать
            select fp.fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn,
                   1 as found_birth_date, 3 as kind
            from clients cl, cte_finpolis fp, fiolist fl1, fiolist fl2
            where fl1.fl_id = fp.fps_from_fl_id
              and fp.birthdate = cl.c_birth_date
              and fl2.fl_id in (cl.c_from_fl_id, cl.c_name_eng_fl_id, cl.c_alternat_fl_id)
              and f_levenstain2(fl1.fl_name1, fl2.fl_name1, 2) < 3
              and f_levenstain2(fl1.fl_name2, fl2.fl_name2, 2) < 3
              and (:db_id = 81001 or f_levenstain2(fl1.fl_name3, fl2.fl_name3, 2) < 3)
            )

        select fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, adl_address_str,
               max(iif(kind = 1, 1, 0)) as found_pass_no, max(iif(kind = 2, 1, 0)) as found_inn_no,
               max(iif(kind = 3, 1, 0)) as found_client_fio, found_birth_date
        from (select fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, adl.adl_address_str, found_birth_date,
                     kind
              from cte_no_address cl
              inner join addrclient adc on cl.c_id = adc.adc_from_c_id and
                    adc_address_type = 1
              inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id)
        group by fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, adl_address_str, found_birth_date
        into fps_id, c_id, c_name_fio, c_birth_date, c_pass_no, c_inn, c_address, found_pass_no, found_inn_no,
             found_client_fio, found_birth_date
    do
    begin
      select fps_fio
      from finpolis_639
      where fps_id = :fps_id
      into fps_fio;

      suspend;
    end
  end
  else
  begin
    select c_id, coalesce(c_name_fio, '') as c_name_fio, c_birth_date, coalesce(c_pass_no, '') as c_pass_no,
           coalesce(c_index_c_pass_no, '') as c_index_c_pass_no, coalesce(c_inn, '') as c_inn
    from clients
    where c_id = :rep_c_id
    into c_id, c_name_fio, c_birth_date, c_pass_no, c_index_c_pass_no, c_inn;

    for with cte_finpolis
        as (select fps_id, birthdate,
                   replace((coalesce(docseria, '') || coalesce(docnumber, '')), ' ', '') as fp_pass_no,
                   trim(coalesce(inn_flip, '')) as fp_inn,
                   fps_from_fl_id
            from finpolis_639
            where fps_fio > '')

        select fps_id, max(iif(kind = 1, 1, 0)) as found_pass_no, max(iif(kind = 2, 1, 0)) as found_inn_no,
               max(iif(kind = 3, 1, 0)) as found_client_fio, max(found_birth_date)
        from (select fps_id, iif(fp.birthdate = :c_birth_date, 1, 0) as found_birth_date, 1 as kind
              from cte_finpolis fp
              where fp.fp_pass_no = :c_index_c_pass_no and
                    fp.fp_pass_no > ''

              union all

              select fps_id, iif(fp.birthdate = :c_birth_date, 1, 0) as found_birth_date, 2 as kind
              from cte_finpolis fp
              where fp.fp_inn = :c_inn and
                    fp.fp_inn > ''

              union all

              select fp.fps_id, iif(fp.birthdate = cl.c_birth_date, 1, 0) as found_birth_date, 3 as kind
              from clients cl, cte_finpolis fp, fiolist fl1, fiolist fl2
              where cl.c_id = :rep_c_id
                and fl1.fl_id = fp.fps_from_fl_id
                and fl2.fl_id in (cl.c_from_fl_id, cl.c_name_eng_fl_id, cl.c_alternat_fl_id)
                and fl1.fl_name1 = fl2.fl_name1
                and fl1.fl_name2 = fl2.fl_name2
                and (fl1.fl_name3 = fl2.fl_name3 or (:db_id = 81001 and fp.birthdate = cl.c_birth_date)) -- В РФ кроме ФИ должно совпасть отчество или ДР

              union all
  
              select fp.fps_id, iif(fp.birthdate = cll.cll_birth_date, 1, 0) as found_birth_date, 3 as kind
              from clientlog cll, cte_finpolis fp, fiolist fl1, fiolist fl2
              where cll.cll_from_c_id = :rep_c_id
                and fl1.fl_id = fp.fps_from_fl_id
                and fl2.fl_id = cll.cll_from_fl_id
                and fl1.fl_name1 = fl2.fl_name1
                and fl1.fl_name2 = fl2.fl_name2
                and (fl1.fl_name3 = fl2.fl_name3 or (:db_id = 81001 and fp.birthdate = cll.cll_birth_date)) -- В РФ кроме ФИ должно совпасть отчество или ДР

              union all

              -- Проверка clients через f_levenstain2, дата рождения должна совпадать
              select fp.fps_id, iif(fp.birthdate = cl.c_birth_date, 1, 0) as found_birth_date, 3 as kind
              from clients cl, cte_finpolis fp, fiolist fl1, fiolist fl2
              where cl.c_id = :rep_c_id
                and fl1.fl_id = fp.fps_from_fl_id
                and fl2.fl_id in (cl.c_from_fl_id, cl.c_name_eng_fl_id, cl.c_alternat_fl_id)
                and f_levenstain2(fl1.fl_name1, fl2.fl_name1, 2) < 3
                and f_levenstain2(fl1.fl_name2, fl2.fl_name2, 2) < 3
                and (:db_id = 81001 or f_levenstain2(fl1.fl_name3, fl2.fl_name3, 2) < 3)

              )
        group by fps_id
        into fps_id, found_pass_no, found_inn_no, found_client_fio, found_birth_date
    do
    begin
      select fps_fio
      from finpolis_639
      where fps_id = :fps_id
      into fps_fio;

      suspend;
    end
  end
end
