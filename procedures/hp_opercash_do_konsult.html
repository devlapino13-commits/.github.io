CREATE PROCEDURE HP_OPERCASH_DO_KONSULT(
    OPC_ID D_ID,
    OL_ID D_ID,
    CODE D_STR_CODE)
AS
declare variable office_id d_integer;
declare variable office_name d_str_100;
declare variable k_id d_id;
declare variable k_fact_disb_date d_date;
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable opc_date d_date;
declare variable opc_amount d_currency;
declare variable new_ol_id d_id;
declare variable ol_date d_date;
declare variable ol_memo d_str_long;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_doc_kind d_integer;
declare variable ol_is_client d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable ob_detail d_str_255;
declare variable sh_date d_date;
declare variable sh_id d_integer;
declare variable fix_sum d_currency;
declare variable sh_sum d_currency;
declare variable paid_sum d_currency;
declare variable delta d_currency;
declare variable kc_id d_id;
declare variable common_err_msg d_str_100;
begin
  -- Построение проводок забалансового учета при оплате конс. услуг в рассрочку (RFCKG-4900)

  common_err_msg = 'Ошибка при обработке платежа конс. услуг в рассрочку, opc_id = ' || opc_id;

  if (code = 'accept') then -- Первоначальное признание стоимости услуги к получению
  begin
    select ok_from_k_id,
           ok_ol_reg_id,
           kc_id,
           kc_from_c_id
    from knclient
    inner join operkred on ok_from_k_id = kc_from_k_id
    where kc_amount_approved > 0 and
          ok_ol_is_active = 1 and
          ok_type = 8 and -- выдача займа
          ok_from_ol_id = :ol_id -- id операции выдачи
          rows 1
    into k_id,
         office_id,
         kc_id,
         c_id;

    select reg_code || ' ' || reg_name
    from region
    where reg_unicode = :office_id
    into office_name;

    select c_name_fio
    from clients
    where c_id = :c_id
    into c_name_fio;

    select price
    from clt_get_price_for_consultation(:kc_id)
    into :opc_amount; -- стоимость конс. услуги

    -- создаем записи в operlist/operbook
    ol_date = current_date;
    ol_oper_type = 2; -- Чисто бухгалтерская
    ol_oper_subtype = 11; -- Мемориальный ордер
    ol_doc_kind = 9; -- Мемориальный ордер
    ol_is_client = 1;

    ol_memo = 'Сумма к получению по договору Консультационных услуг / ' || f_datetostr(ol_date) || ' / ' || c_name_fio;
    ob_detail = 'Первоначальное признание стоимости консультационных услуг в рассрочку.';

    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = '99911/1'
    into acc_dt;

    if (coalesce(acc_dt, '') = '') then
      exception e_message 'Дт. счет к первоначальному признанию стоимости конс. услуг не определен, офис ' || office_name;

    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = '99912/1'
    into acc_kt;

    if (coalesce(acc_kt, '') = '') then
      exception e_message 'Кт. счет к первоначальному признанию стоимости конс. услуг не определен, офис ' || office_name;

    -- забалансовый учет
    insert into operlist (ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no_str, ol_document_client, ol_is_client, ol_memo, ol_user_id, ol_date)
    values (:office_id, :office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :k_id, :c_id, :ol_is_client, :ol_memo, 1000000000000, :ol_date)
    returning (ol_id)
    into new_ol_id;

    insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
    values (:new_ol_id, :acc_dt, :acc_kt, :opc_amount, :opc_amount, :opc_date, :office_id, :ob_detail);

    exit;
  end

  select opc_office_id,
         opc_from_k_id,
         opc_from_c_id,
         opc_date,
         opc_amount
  from opercash
  where opc_id = :opc_id and
        opc_is_delete = 0 and
        opc_oper_kind = f_enum_id('cashoperkind', 'plus.loan.konsult')
  into office_id,
       k_id,
       c_id,
       opc_date,
       opc_amount;

  if (office_id is null or
      k_id is null or
      c_id is null or
      ol_id is null or
      opc_date is null or
      opc_amount is null or
      opc_amount = 0) then
    exception e_message common_err_msg;

  select reg_code || ' ' || reg_name
  from region
  where reg_unicode = :office_id
  into office_name;

  select c_name_fio
  from clients
  where c_id = :c_id
  into c_name_fio;

  select k_fact_disb_date
  from kontract
  where k_id = :k_id
  into k_fact_disb_date;

  if (code = 'repayment') then -- поступление погашения по конс. услугам в рассрочку
  begin
    -- 1) фиксация платежа в operbook

    ob_detail = 'Оплата Консультационных услуг / ' || f_datetostr(k_fact_disb_date) || ' / ' || c_name_fio;

    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = '99912/1'
    into acc_dt;

    if (coalesce(acc_dt, '') = '') then
      exception e_message 'Дт. счет к получению оплаты конс. услуг не определен (забалансовый учет), офис ' || office_name;

    select al_id
    from acc_list
    where al_from_reg_id = :office_id and
          al_currency = '417' and
          al_account = '99911/1'
    into acc_kt;

    if (coalesce(acc_kt, '') = '') then
      exception e_message 'Кт. счет к получению оплаты конс. услуг не определен (забалансовый учет), офис ' || office_name;

    insert into operbook (ob_from_ol_id, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_client_kt, ob_detail)
    values (:ol_id, :acc_dt, :acc_kt, :opc_amount, :opc_amount, :opc_date, :office_id, :c_id, :ob_detail);

    -- 2) фиксация платежа в opercons
    for select oc_sh_id,
               oc_schedule_date,
               oc_schedule_principal
        from opercons
        where oc_from_k_id = :k_id and
              oc_ol_is_active = 1 and
              oc_operation_type_id = 1 -- график оплаты конс. услуг в рассрочку
        order by oc_schedule_date
        into sh_id,
             sh_date,
             sh_sum
    do
    begin
      fix_sum = 0; -- сумма платежа, фиксируемая в новой записи opercons
      paid_sum = 0; -- ранее оплаченная сумма

      select sum(oc_paid_principal)
      from opercons
      where oc_from_k_id = :k_id and
            oc_ol_is_active = 1 and
            oc_operation_type_id = 4 and
            oc_schedule_date = :sh_date
      into paid_sum;

      delta = coalesce(sh_sum, 0) + coalesce(paid_sum, 0); -- разница между суммой по графику и ранее оплаченной суммой (paid_sum равен нулю или отриц. числу)

      if (delta < 0) then
        exception e_message common_err_msg;

      if (delta = 0) then -- платеж по графику закрыт
        continue;

      if (delta > 0) then -- платеж по графику не закрыт
      begin
        if (opc_amount < delta) then
        begin
          -- внесенная сумма меньше суммы платежа по графику
          fix_sum = opc_amount;
          opc_amount = 0;
        end
        else
        begin
          -- внесенная сумма больше или равно сумме платежа по графику
          fix_sum = delta;
          opc_amount = opc_amount - delta;
        end

        insert into opercons (oc_from_ol_id, oc_number, oc_from_k_id, oc_from_c_id, oc_sh_id, oc_schedule_date, oc_schedule_principal, oc_paid_principal, oc_operation_type_id, oc_ol_date, oc_ol_reg_id, oc_ol_is_active, ok_pay_source)
        values (:ol_id, -1, :k_id, :c_id, :sh_id, :sh_date, 0, -:fix_sum, 4, :opc_date, :office_id, 1, 3);
      end

      if (opc_amount = 0) then -- внесенной суммы не осталось для распределения
        leave;
    end

    if (coalesce(sh_id, 0) = 0) then
      exception e_message 'График оплаты конс. услуг в рассрочку не найден';

    if (opc_amount > 0) then -- есть переплата, т.е. есть остаток внесенной суммы после ее распределения по графику оплаты
      insert into opercons (oc_from_ol_id, oc_number, oc_from_k_id, oc_from_c_id, oc_sh_id, oc_schedule_date, oc_schedule_principal, oc_paid_principal, oc_operation_type_id, oc_ol_date, oc_ol_reg_id, oc_ol_is_active, ok_pay_source)
      values (:ol_id, -1, :k_id, :c_id, :sh_id, :sh_date, 0, -:opc_amount, 9, :opc_date, :office_id, 1, 3);
  end
end
