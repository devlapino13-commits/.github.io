CREATE PROCEDURE AAREP_BILL_PERC_PRINT(
    REP_OL_ID DECIMAL(18,0))
RETURNS (
    OL_ID DECIMAL(18,0),
    IS_DELETE INTEGER,
    OK_PERC DECIMAL(18,4),
    BILL_NO INTEGER,
    HEAD_ID INTEGER)
AS
declare variable reg_id integer;
declare variable oper_dd date;
begin
  bill_no = 0;
  select ol_from_reg_id, reg_head_office, ol_date
  from operlist, region
  where ol_id = :rep_ol_id and
        reg_unicode = ol_from_reg_id
  into :reg_id, :head_id, :oper_dd;
  if (reg_id is not null) then
  begin
    for select ok_from_ol_id,
               (1 - ok_ol_is_active), sum(ok_perc)
        from operkred, region
        where ok_ol_reg_id = reg_unicode and
              reg_head_office = :head_id and
              ok_type = 4 and
              ok_ol_date = :oper_dd and
              ok_from_c_id = 0
        group by ok_from_ol_id, ok_ol_is_active
        order by ok_from_ol_id
        into :ol_id, :is_delete, :ok_perc
    do
    begin
      bill_no = bill_no + 1;
      if (ol_id = :rep_ol_id) then
      begin
        ok_perc = 0 - ok_perc;
        suspend;
      end
    end
  end
end
