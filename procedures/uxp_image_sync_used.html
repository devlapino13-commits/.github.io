Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE UXP$IMAGE_SYNC_USED
AS
declare variable img_uuid d_uuid;
begin
  update ux$image
  set img_used = 0
  where 1 = 1;

  for select distinct ua_img_uuid
  from ux$action
  where ua_img_uuid is not null
  union
  select ucl_img_uuid
  from ux$combolist
  where ucl_img_uuid is not null
  union
  select edi_img_uuid
  from ux$editor
  where edi_img_uuid is not null
  union
  select ufb_img_uuid
  from ux$frbutton
  where ufb_img_uuid is not null
  into img_uuid
  do
  begin
    update ux$image
    set img_used = 1
    where img_uuid = :img_uuid;
  end
end
