CREATE PROCEDURE HP_KN_OKSUMMARY_FIX_ONE(
    REP_K_ID DECIMAL(18,0),
    REP_OL_ID DECIMAL(18,0),
    NEED_CLOSE_LOAN INTEGER)
AS
declare variable ok_number integer;
declare variable ok_sh_date date;
declare variable kc_id id_type;
declare variable gr_zad_os curr_type;
declare variable gr_zad_perc curr_type;
declare variable arr_zad_os varchar(400);
declare variable arr_zad_perc varchar(400);
declare variable zad_os curr_type;
declare variable zad_perc curr_type;
declare variable curr_line integer;
declare variable max_number integer;
declare variable ok_ol_date date;
begin
-- корректировка фактической ОС
  if (need_close_loan = 1) then
  begin
    ok_number = 0;
    max_number = 0;
    for select ok_number, ok_ol_date
        from operkred
        where ok_from_k_id = :rep_k_id and
              ok_from_ol_id = :rep_ol_id and
              ok_ol_is_active = 1 and
              ok_type < 7 and
              ok_os <> 0
        order by ok_from_ol_id desc, ok_number desc
        rows 1
        into :ok_number, :ok_ol_date
    do
    begin
      gr_zad_os = 0;
      arr_zad_os = '';

      for select kc_id
          from knclient
          where kc_from_k_id = :rep_k_id and
                kc_amount_approved > 0
          into :kc_id
      do
      begin
        zad_os = 0;
        select oks_balance
        from kn_oksummary
        where oks_from_kc_id = :kc_id and
              oks_date_begin = :ok_ol_date
        into :zad_os;
        if (zad_os is null) then
          zad_os = 0;

        if (zad_os <> 0) then
        begin
          update kn_operkred
          set ok_os = ok_os - :zad_os
          where ok_from_kc_id = :kc_id and
                ok_from_ol_id = :rep_ol_id and
                ok_number = :ok_number;
        end
      end
    end
  end -- конец факт ОС

-- корректировка по ОС
  ok_number = 0;
  max_number = 0;
  for select ok_number, ok_sh_date
      from operkred
      where ok_from_k_id = :rep_k_id and
            ok_from_ol_id = :rep_ol_id and
            ok_ol_is_active = 1 and
            ok_type in (2, 3) and
            ok_sh_os <> 0
      order by ok_from_ol_id, ok_number
      into :ok_number, :ok_sh_date
  do
  begin
    gr_zad_os = 0;
    arr_zad_os = '';

    for select kc_id
        from knclient
        where kc_from_k_id = :rep_k_id and
              kc_amount_approved > 0
        into :kc_id
    do
    begin
      zad_os = 0;
      select sum(ok_os + ok_sh_os)
      from kn_operkred
      where ok_from_kc_id = :kc_id and
            ok_sh_date = :ok_sh_date and
            ok_ol_is_active = 1 and
            ok_type < 7
      into :zad_os;
      if (zad_os is null) then
        zad_os = 0;

      gr_zad_os = gr_zad_os + zad_os;
      arr_zad_os = arr_zad_os || right('                    ' || zad_os, 20);
    end

    if (gr_zad_os = 0) then
    begin
      curr_line = 0;
      for select kc_id
          from knclient
          where kc_from_k_id = :rep_k_id and
                kc_amount_approved > 0
          into :kc_id
      do
      begin
        curr_line = curr_line + 1;
        zad_os = substring(arr_zad_os from ((curr_line - 1) * 20 + 1) for 20);

        if (zad_os <> 0) then
        begin
          update kn_operkred
          set ok_sh_os = ok_sh_os - :zad_os
          where ok_from_kc_id = :kc_id and
                ok_from_ol_id = :rep_ol_id and
                ok_ol_is_active = 1 and
                ok_number = :ok_number;
        end
      end
    end
    else
    begin
      if (max_number < ok_number) then
        max_number = ok_number;
    end
  end

  if (max_number <> 0) then
  begin
    for select kc_id
        from knclient
        where kc_from_k_id = :rep_k_id and
              kc_amount_approved > 0
        into :kc_id
    do
    begin
      zad_os = 0;
      select sum(ok_sh_os)
      from kn_operkred
      where ok_from_kc_id = :kc_id and
            ok_from_ol_id = :rep_ol_id and
            ok_number <> :max_number and
            ok_ol_is_active = 1 and
            ok_type in (2, 3)
      into :zad_os;

      if (zad_os is null) then
        zad_os = 0;

      update kn_operkred
      set ok_sh_os = -:zad_os
      where ok_from_kc_id = :kc_id and
            ok_from_ol_id = :rep_ol_id and
            ok_number = :max_number and
            ok_ol_is_active = 1 and
            ok_sh_os <> -:zad_os;
    end
  end

-- корректировка по %%
  ok_number = 0;
  for select ok_number, ok_sh_date
      from operkred
      where ok_from_k_id = :rep_k_id and
            ok_from_ol_id = :rep_ol_id and
            ok_ol_is_active = 1 and
            ok_type in (2, 3) and
            ok_sh_perc <> 0
      order by ok_from_ol_id, ok_number
      into :ok_number, :ok_sh_date
  do
  begin
    gr_zad_perc = 0;
    arr_zad_perc = '';

    for select kc_id
        from knclient
        where kc_from_k_id = :rep_k_id and
              kc_amount_approved > 0
        into :kc_id
    do
    begin
      zad_perc = 0;
      select sum(ok_sh_perc + ok_perc)
      from kn_operkred
      where ok_from_kc_id = :kc_id and
            ok_sh_date = :ok_sh_date and
            ok_ol_is_active = 1 and
            ok_type < 7
      into :zad_perc;
      if (zad_perc is null) then
        zad_perc = 0;

      gr_zad_perc = gr_zad_perc + zad_perc;
      arr_zad_perc = arr_zad_perc || right('                    ' || zad_perc, 20);
    end

    if (gr_zad_perc = 0) then
    begin
      curr_line = 0;
      for select kc_id
          from knclient
          where kc_from_k_id = :rep_k_id and
                kc_amount_approved > 0
          into :kc_id
      do
      begin
        curr_line = curr_line + 1;
        zad_perc = substring(arr_zad_perc from ((curr_line - 1) * 20 + 1) for 20);

        if (zad_perc <> 0) then
        begin
          update kn_operkred
          set ok_sh_perc = ok_sh_perc - :zad_perc
          where ok_from_kc_id = :kc_id and
                ok_from_ol_id = :rep_ol_id and
                ok_ol_is_active = 1 and
                ok_number = :ok_number;
        end
      end
    end
  end
end
