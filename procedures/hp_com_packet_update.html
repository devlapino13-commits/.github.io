CREATE PROCEDURE HP_COM_PACKET_UPDATE(
    CMP_ID D_INTEGER,
    CMP_STR_NO D_STR_CODE,
    CMP_DATE D_DATE,
    CMP_IS_IPF D_SMALLINT,
    CMP_MEMO D_STR_LONG)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
begin
  res_is_error = 0;
  res_text = '';

  if (exists(select *
             from com_packet
             where cmp_str_no = :cmp_str_no)) then
  begin
    res_is_error = 1;
    res_text = 'Номер дублируется, введите другой номер';
  end else
  begin
    update com_packet
    set cmp_str_no = :cmp_str_no,
        cmp_date = :cmp_date,
        cmp_is_ipf = :cmp_is_ipf,
        cmp_memo = :cmp_memo
    where cmp_id = :cmp_id;

    res_text = 'Данные успешно сохранены';
  end
  suspend;
end
