CREATE PROCEDURE HP_FREELANCER_AKT(
    PAR1 D_INTEGER)
RETURNS (
    SSH_DATE D_DATE,
    AMOUNT D_CURRENCY,
    OL_LAST_DATE D_DATE,
    SSHE_IS_COMPLITED D_SMALLINT,
    SSHE_MEMO D_STR_255,
    EDL_STATUS D_STR_CODE,
    SSH_ID D_INTEGER,
    SSH_FROM_SD_ID D_INTEGER,
    SSHE_ID D_INTEGER)
AS
begin
  ssh_from_sd_id = par1;

  for select ssh.ssh_date, sum(ssh.ssh_amount_dogovor) as amount, max(ssh.ssh_date_regist) as ol_last_date, max(ssh.ssh_id)
      from servicesh ssh
      where ssh.ssh_from_sd_id = :par1
      group by ssh.ssh_date
      order by ssh.ssh_date
      into :ssh_date, :amount, :ol_last_date, :ssh_id
  do
  begin
    sshe_id = -1;
    sshe_memo = null;
    sshe_is_complited = null;
    select sshe_is_complited, sshe_memo, sshe_id
    from servicesh_edoc
    where sshe_from_sd_id = :par1 and
        sshe_from_ssh_id = :ssh_id
    into :sshe_is_complited, :sshe_memo, :sshe_id;

    edl_status = f_edoc_status('freelancer_akt', :sshe_id, current_date);

    if (edl_status = 'empty') then
      edl_status = 'Не отправлялся';
    if (edl_status = 'signed') then
      edl_status = 'Подписан';
    if (edl_status = 'unsigned') then
      edl_status = 'Не подписан';
    if (edl_status = 'cancelled') then
      edl_status = 'Аннулирован';
    suspend;
  end
end
