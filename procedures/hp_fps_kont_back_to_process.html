CREATE PROCEDURE HP_FPS_KONT_BACK_TO_PROCESS(
    K_ID D_ID NOT NULL,
    NEW_STATUS D_SMALLINT NOT NULL,
    MEMO D_STR_255 NOT NULL,
    UN_ID D_ID NOT NULL)
AS
declare variable acl_id d_id;
declare variable rd_id integer;
begin
  -- Процедура вызывается из окна выдач по СБП, в Finstart'е, для отказа/доработки

  if (new_status not in (-2, -1)) then
    exception e_default 'Bad new_status, should be: -2, -1';

  --rd_id = f_get_rd_id('ActionGroup.Check_Loan_Data');

  select rd_id
  from hp_get_check_loan_data(:k_id)
  into :rd_id;

  select acl_id
  from actionlist
  where acl_object_kind = 9 and
        acl_object_id = :k_id and
        acl_from_rd_id = :rd_id
  order by acl_id desc
  rows 1
  into acl_id;

  -- Вносим изменения в решение КК
  if (acl_id is not null) then
  begin
    update actionlist
    set acl_status = :new_status,
        acl_from_un_id = :un_id
    where acl_id = :acl_id;
  end

  -- Добавляем комментарий при отказе/доработке
  execute procedure hp_objectprops_memo_set('kontract', :k_id, -1, :rd_id, :memo, :un_id);

  -- Возвращаем договор в процесс
  update kontract
  set k_status = 0,
      k_user_id = :un_id
  where k_id = :k_id and
        k_status <> 0;
end
