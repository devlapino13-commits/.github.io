CREATE PROCEDURE HP_DO_REPAYMENT_BI(
    K_ID ID_TYPE,
    REP_DATE DATE,
    REPAY_AMOUNT CURR_TYPE,
    REPAY_FACT_OFFICE_ID INTEGER,
    REPAY_DT_ACCOUNT_NO STR_20,
    REPAY_USER_ID ID_TYPE,
    REPAY_DOCUMENT_CLIENT ID_TYPE,
    REPAY_DOC_CLIENT_NAME_REASON STR_255)
RETURNS (
    OL_ID ID_TYPE,
    ERROR_TEXT STR_255)
AS
declare variable debt_check integer;
declare variable k_kontract_no str_20;
declare variable operdel_date d_date;
begin
  ol_id = 0;
  error_text = '';

  debt_check = (select result
                from hp_debt_check(:k_id, null, :rep_date));
  if (debt_check = -1) then
    error_text = 'Договор не определён';

  if (f_get_db_id() = 81001 and error_text = '') then
  begin
    operdel_date = null;

    -- Есть ли подтвержленная заявка на удаление
    select min(odm_oper_date)
    from operdelkont, operdelmain
    where odk_from_k_id = :k_id and
          odk_from_odm_id = odm_id and
          odm_state = 1 and
          odm_oper_date >= (current_date - 7)
    into operdel_date;

    if (operdel_date is null) then
      operdel_date = cast('31.12.2999' as date);

    if (exists(select * from operkred
    where ok_ol_is_active = 1
    and ok_from_k_id = :k_id 
    and ok_ol_date between :rep_date and :operdel_date
    and ok_type in (4,5,9)
    and (ok_os <> 0 or ok_perc <> 0))) then
    begin
      select k_kontract_no from kontract
      where k_id = :k_id
      into k_kontract_no;

      debt_check = -1;
      error_text = 'ВНИМАНИЕ! По займу '||k_kontract_no||
      ' ранее принято погашение с более поздней датой. Распроведите более раннее погашение, примите в ручном режиме погашение в рамках загружаемой выписки, затем примите снова распроведенное погашение.';
    end 
  end 

  if (debt_check = 1) then
-- погашение
    select ol_id, error_text
    from hp_do_repayment(:k_id, 0, :rep_date, :repay_amount, 1, 1, 1, 1, 1, 0, 0, :repay_fact_office_id, 2, :repay_dt_account_no, 0, :repay_user_id, :repay_document_client, :repay_doc_client_name_reason, null, 0, 0)
    into :ol_id, :error_text;

  if (debt_check = 0) then
-- переплата
    select ol_id, error_text
    from hp_do_repayment(:k_id, 0, :rep_date, :repay_amount, 1, 0, 0, 0, 1, 0, 0, :repay_fact_office_id, 3, :repay_dt_account_no, 1, :repay_user_id, :repay_document_client, :repay_doc_client_name_reason, null, 0, 0)
    into :ol_id, :error_text;
  suspend;
end
