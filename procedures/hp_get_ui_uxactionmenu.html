CREATE PROCEDURE HP_GET_UI_UXACTIONMENU(
    UXA_CODE D_STR_100,
    UN_ID D_ID)
RETURNS (
    RES_CODE D_STR_100,
    RES_PARENT_CODE D_STR_100,
    RES_CAPTION D_STR_100,
    RES_HINT D_STR_255,
    RES_UA_CODE D_STR_100,
    RES_TAG D_SMALLINT,
    RES_JSONDOC_FIELD D_STR_CODE)
AS
declare variable ux_code d_str_100;
declare variable res_disable_on_edit d_bool;
declare variable res_inactive d_bool;
declare variable ux_kind d_str_code;
declare variable fr_code d_str_code;
begin
  select uxa_ux_code, uxf$kind(ux_kind, 'code')
  from uxalias, uxcode
  where uxa_code = :uxa_code and
        uxa_ux_code = ux_code
  into ux_code, ux_kind;

  fr_code = '';
  if (ux_kind = 'uxform') then
  begin
    select ucf_ufr_code
    from uxcodeframe
    where ucf_ux_code = :ux_code and
          ucf_is_mainframe = 1
    into fr_code;
  end else
    fr_code = uxa_code;

  res_hint = '';
  res_ua_code = '';
  res_tag = 0;
  res_parent_code = '';
  res_jsondoc_field = '';

  if (fr_code <> '') then
  begin
    for select ufp_origin_name, ufp_name
    from uxfrparam
    where ufp_ufr_code = :fr_code and
          ufp_editor = 'blob_json' and
          ufp_in_use = 1 and
          ufp_is_delete = 0
    into res_jsondoc_field, res_caption
    do
    begin
      res_code = 'b_jsondoc_'  || res_jsondoc_field;
      res_jsondoc_field = res_jsondoc_field || '$json';
      suspend;
    end
  end

  if (res_jsondoc_field <> '') then
  begin
    res_code = res_code || '_';
    res_caption = '-';
    suspend;
  end

  res_jsondoc_field = '';
  res_hint = '';
  res_ua_code = '';
  res_tag = 0;
  res_parent_code = '';

  for select res_code, res_parent_code, res_caption, res_ua_code, res_inactive, res_disable_on_edit, res_hint
  from hp_uxsubmenu_items(:ux_code, :uxa_code, 1, 0)
  into :res_code, :res_parent_code, :res_caption, :res_ua_code, :res_inactive, res_disable_on_edit, res_hint
  do
  begin
    res_tag = 0;
    if (res_inactive = 1) then
      res_tag = -1;
    else
      if (res_disable_on_edit = 1) then
        res_tag = 1;

    if (uxf$access(un_id, res_ua_code) <> -1) then
      suspend;
  end
end
