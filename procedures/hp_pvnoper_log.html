CREATE PROCEDURE HP_PVNOPER_LOG(
    REGION_ID D_INTEGER,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    PO_DATE D_DATE,
    OL_ID D_ID,
    OPER_TYPE D_SMALLINT,
    CARD_ID D_INTEGER,
    NAME_FIO VARCHAR(100),
    KONTRACT_NO VARCHAR(20),
    PO_AMOUNT D_CURRENCY,
    TICKET_NO D_STR_CODE,
    USER_FIO VARCHAR(100),
    REG_NAME VARCHAR(50),
    STATUS D_STR_CODE,
    CARD_BALANCE D_CURRENCY,
    IS_MB_CARD D_BOOL,
    CARD_TYPE_NAME VARCHAR(20),
    OPER_TYPE_NAME VARCHAR(20),
    PO_ID D_INTEGER,
    STATUS_ID D_INTEGER)
AS
begin

  for select o.po_date, po_from_ol_id, po_oper_type, po_card_id, cl.c_name_fio, k.k_kontract_no, po_amount,
    po_ticket_no, un.un_fio, r.reg_name, el.el_name, po_card_balance_amount, po_is_mb_card,
    iif(po_is_mb_card = 1, 'Карта МБулак', 'Чужая карта'), iif(po_oper_type = 0, 'Обналичивание', 'Платеж по кредиту'), o.po_id, o.po_status
  from pvnopers o
    inner join clients cl on o.po_from_c_id = cl.c_id
    inner join username un on o.po_un_id = un.un_id
    inner join region_list(:region_id, 1, 3, 1) rl on o.po_office_id = rl.id_list
    inner join region r on o.po_office_id = r.reg_unicode
    inner join enum_list el on o.po_status = el.el_id and el.el_kind = 'pvnopers_status'
    left join kontract k on o.po_from_k_id = k.k_id
  where o.po_is_delete = 0
    and o.po_date between :date_begin and :date_end
  order by o.po_id
  into :po_date, :ol_id, :oper_type, :card_id, :name_fio, :kontract_no, :po_amount,
    :ticket_no, :user_fio, :reg_name, :status, :card_balance, :is_mb_card, :card_type_name, :oper_type_name, :po_id, :status_id
  do
    suspend;
end
