CREATE PROCEDURE HP_BKI_CONTRACTCHANGESLIST_EDIT(
    BOD_ID D_ID NOT NULL,
    CHANGE_DATE D_DATE NOT NULL,
    CHANGE_TYPE D_STR_CODE NOT NULL,
    SPECIAL_TYPE D_STR_CODE NOT NULL,
    TYPE_TEXT D_STR_LONG,
    APPLY_DATE D_DATE NOT NULL,
    END_DATE D_DATE NOT NULL)
AS
declare variable jso d_integer;
declare variable status d_integer;
begin
  select json.parse(bod_add)
  from bki_once_data
  where bod_id = :bod_id
  into jso;

  status = json.obj_get_val(jso, 'status');
  if (status is null or status = 2) then  --В очереди на отправку
    exception e_message 'Запись не может быть редактирована';

  json.obj_set_s(jso, 'date', f_datetostr(change_date));
  json.obj_set_s(jso, 'type', change_type);
  json.obj_set_s(jso, 'specialType', special_type);
  json.obj_set_s(jso, 'applyDate', f_datetostr(apply_date));
  json.obj_set_s(jso, 'endDate', f_datetostr(end_date));

  if (coalesce(type_text, '') <> '') then
    json.obj_set_s(jso, 'typeText', type_text);

  update bki_once_data
    set bod_add = json.obj_to_json(:jso)
  where bod_id = :bod_id;
end
