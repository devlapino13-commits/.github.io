CREATE PROCEDURE CM_RU_0148_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable c_id id_type;
declare variable sts_car_owner str_100;
declare variable cds_doc_no str_100;
begin
  if (k_id in (101010043997800201, 101010085788300301)) then
    exit;
  for select f_docrecog_value(cd.cds_from_drl_id, 'calc_name_full'), knc.kc_from_c_id, cd.cds_doc_no
      from knclient knc
      inner join kontract kk on kk.k_id = knc.kc_from_k_id
      inner join product pr on pr.pr_id = kk.k_from_pr_id and
            pr.pr_code like '%.zalog_car.%'
      left join clientdocs cd on cd.cds_from_c_id = knc.kc_from_c_id and
            cd.cds_is_delete = 0 and
            k_disb_date between cd.cds_doc_date_begin and coalesce(cd.cds_doc_date_end, k_disb_date) and
            cd.cds_from_pk_id = 310 -- ctc
      where knc.kc_from_k_id = :k_id and
            knc.kc_amount_approved > 0
      into :sts_car_owner, :c_id, :cds_doc_no
  do
  begin
    sts_car_owner = coalesce(sts_car_owner, '');
    if (sts_car_owner = '') then
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :c_id
      into :res_info_text;
      res_error_is_critical = 1;
      res_error_text = 'Не заполнено поле "собственник по СТС" (№ ' || coalesce(cds_doc_no, '') || ')';
      suspend;
    end
  end
end
