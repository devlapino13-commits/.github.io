CREATE PROCEDURE FM_LOAN_PLANNING_REPAY(
    UN_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    JUST_LAST_REPAY INTEGER,
    REG_ID INTEGER)
RETURNS (
    K_ID ID_TYPE,
    PLAN_DATE DATE,
    SUM_ZADOL CURR_TYPE,
    REPAY_NO_STR VARCHAR(20))
AS
declare variable C_ID ID_TYPE;
declare variable SUM_REPAY decimal(18,4);
declare variable REPAY_NO integer;
declare variable MAX_REPAY integer;
declare variable LAST_DATE date;
declare variable CM_DATABASE_ID integer;
declare variable C_NAME_FIO varchar(100);
begin
  /*  select cm_database_id from company
  into :cm_database_id;  */
  :cm_database_id = f_get_db_id();

  select un_from_c_id
  from username
  where un_id = :un_id
  into :c_id;

  if (c_id is null) then
    exit;

  if (not exists(select ps_from_reg_id
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :c_id and
                       ps_date_begin <= current_date and
                       ps_date_end >= current_date and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
                       ps_is_working = 1)) then
    reg_id = 0;

  for select res_c_id
      from hp_staff_changer_list(:un_id, 1)
      where :reg_id = 0
      union
      select res_c_id
      from hp_staff_changer_list(:un_id, -1)
      where :reg_id <> 0
      into :c_id
  do
  begin
    if (datediff(day, rep_date_begin, rep_date_end) > 30) then
      rep_date_end = dateadd(month, 1, rep_date_begin) - 1;

    if (reg_id = 0) then
    begin
      for select oks_from_k_id, oks_delinq_date, k_kredline_end, c.c_name_fio
          from oksummary, kontract k, clients c
          where oks_delinq_date <= :rep_date_end and
                oks_kc_id = :c_id and
                k.k_clients_cashier_id = c.c_id and
                oks_close_state = 0 and
                oks_date_begin+0 <= :rep_date_begin and
                oks_date_end >= :rep_date_begin and
                k_id = oks_from_k_id
          into k_id, plan_date, last_date, c_name_fio
      do
      begin
        if (plan_date < :rep_date_begin) then
        begin
          plan_date = null;
          select min(ok_sh_date)
          from operkred
          where ok_from_k_id = :k_id and
                ok_sh_date between :rep_date_begin and :rep_date_end and
                ok_ol_date <= :rep_date_end and
                ok_type < 7 and
                ok_ol_is_active = 1 having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 0
          into plan_date;
        end
        if (plan_date is not null) then
        begin

          if (just_last_repay = 0 or plan_date = last_date) then
          begin
            sum_repay = 0;
            sum_zadol = 0;
            repay_no_str = '';
            repay_no = 0;
            max_repay = 0;
            select sum(iif(sh_date <= :plan_date, zad_os + sh_calc_perc + sh_calc_fine, 0.0)),
                   sum(iif(sh_date > :plan_date, zad_os + sh_calc_perc + sh_calc_fine, 0.0)),
                   max(iif(sh_date <= :plan_date, sh_id, 0)), max(sh_id)
            from aarep_make_zad_list(:k_id, iif(:plan_date > current_date, :plan_date, current_date), 1) -- RFCKZ-394
            where sh_date is not null
            into sum_zadol, sum_repay, repay_no, max_repay;

            if (sum_zadol is null) then
              sum_zadol = 0;
            if (sum_repay is null) then
              sum_repay = 0;
            if (repay_no is null) then
              repay_no = 0;
            if (max_repay is null) then
              max_repay = 0;
            if (sum_repay > 0) then
              repay_no_str = repay_no || f_format(' %1 ', iif(cm_database_id <> 94401, 'из', 'from')) || max_repay;
            else
              repay_no_str = repay_no || f_format(', %1.', iif(cm_database_id <> 94401, 'посл.', 'sonuncu'));
            if (sum_zadol <> 0 and
                (just_last_repay = 0 or (just_last_repay = 1 and
                sum_repay <= 0))) then
              suspend;
          end
        end
      end
    end
    else
    begin
      for select oks_from_k_id, oks_delinq_date, k_kredline_end
          from oksummary, kontract, region_list(:reg_id, 1, 3, 1) reg
          where oks_delinq_date <= :rep_date_end and
                oks_region_id = reg.id_list and
                oks_close_state = 0 and
                oks_date_begin+0 <= :rep_date_begin and
                oks_date_end >= :rep_date_begin and
                k_id = oks_from_k_id
          into k_id, plan_date, last_date
      do
      begin
        if (plan_date < :rep_date_begin) then
        begin
          plan_date = null;
          select min(ok_sh_date)
          from operkred
          where ok_from_k_id = :k_id and
                ok_sh_date between :rep_date_begin and :rep_date_end and
                ok_ol_date <= :rep_date_end and
                ok_type < 7 and
                ok_ol_is_active = 1 having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 0
          into plan_date;
        end
        if (plan_date is not null) then
        begin
          if (just_last_repay = 0 or plan_date = last_date) then
          begin
            sum_repay = 0;
            sum_zadol = 0;
            repay_no_str = '';
            repay_no = 0;
            max_repay = 0;
            select sum(iif(sh_date <= :plan_date, zad_os + sh_calc_perc + sh_calc_fine, 0.0)),
                   sum(iif(sh_date > :plan_date, zad_os + sh_calc_perc + sh_calc_fine, 0.0)),
                   max(iif(sh_date <= :plan_date, sh_id, 0)), max(sh_id)
            from aarep_make_zad_list(:k_id, iif(:plan_date > current_date, :plan_date, current_date), 1) -- RFCKZ-394
            where sh_date is not null
            into sum_zadol, sum_repay, repay_no, max_repay;

            if (sum_zadol is null) then
              sum_zadol = 0;
            if (sum_repay is null) then
              sum_repay = 0;
            if (repay_no is null) then
              repay_no = 0;
            if (max_repay is null) then
              max_repay = 0;
            if (sum_repay > 0) then
              repay_no_str = repay_no || f_format(' %1 ', iif(cm_database_id <> 94401, 'из', 'from')) || max_repay;
            else
              repay_no_str = repay_no || f_format(', %1', iif(cm_database_id <> 94401, 'посл.', 'sonuncu'));
            if (sum_zadol <> 0 and
                (just_last_repay = 0 or (just_last_repay = 1 and
                sum_repay <= 0))) then
              suspend;
          end
        end
      end
    end
  end

end
