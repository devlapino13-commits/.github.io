CREATE PROCEDURE AAREP_SURVEY_STATUS_LIST(
    SVQ_ID INTEGER)
RETURNS (
    SVQ_CAPTION VARCHAR(255),
    RESULT_STATUS VARCHAR(255))
AS
declare variable total_count integer;
declare variable svr_from_sva_id integer;
declare variable svq_time_finish timestamp;
declare variable kol integer;
declare variable voice_count integer;
begin
  total_count = 0;
  select count(*)
  from survey_allowedcells
  where sac_is_test >= 0
  into :total_count;

  for select svq_caption || ' (' || svq_memo || ')', svq_time_finish
      from survey_ques
      where svq_id = :svq_id
      into svq_caption, svq_time_finish
  do
  begin
    kol = 0;
    voice_count = 0;
    select count(*)
    from survey_result
    where svr_from_svq_id = :svq_id
    into voice_count;

    result_status = 'Проголосовало ' || voice_count || ' из ' || total_count;

    if (current_timestamp > svq_time_finish) then
    begin
      result_status = 'Нет СМС';

      select svr_from_sva_id, count(*)
      from survey_result
      where svr_from_svq_id = :svq_id
      group by svr_from_sva_id
      order by 2 desc
      rows 1
      into :svr_from_sva_id, :kol;

      if (kol is null) then
        kol = 0;
      if (kol > 0) then
      begin
        for select svr_from_sva_id
            from survey_result
            where svr_from_svq_id = :svq_id
            group by svr_from_sva_id
            having count(*) = :kol
            into :svr_from_sva_id
        do
        begin
          result_status = '';
          select sva_caption
          from survey_answer
          where sva_id = :svr_from_sva_id
          into :result_status;
          suspend;
        end
      end
      else
        suspend;
    end
    else
      suspend;
  end
end
