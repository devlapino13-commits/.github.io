CREATE PROCEDURE HP_HR_VOCTRAVEL_LIST(
    ORDER_ID D_ID)
RETURNS (
    HRV_FROM_HO_ID D_ID,
    HRV_NUMBER D_INTEGER,
    HRV_FROM_C_ID D_ID,
    HRV_DATE_BEGIN D_DATE,
    HRV_DATE_END D_DATE,
    HRV_DAYS D_INTEGER,
    HRV_DAY_BACK D_DATE,
    HRV_DIRECTION D_STR_100,
    HRV_CLASS D_STR_20,
    HRV_STATUS D_INTEGER,
    HRV_DATE_END_FACT D_DATE,
    HRV_CHANGER_ID D_ID,
    HRV_NOT_CLOSE_ACCESS D_BOOL,
    HRV_FROM_REG_ID D_INTEGER,
    C_NAME_FIO D_STR_100,
    RES_REGION_NAME D_STR_100,
    HRV_CHANGER_FIO D_STR_100)
AS
begin
  for select hrv_from_ho_id,
             hrv_number,
             hrv_from_c_id,
             hrv_date_begin,
             hrv_date_end,
             hrv_days,
             hrv_day_back,
             hrv_direction,
             hrv_class,
             hrv_status,
             hrv_date_end_fact,
             hrv_changer_id,
             hrv_not_close_access,
             hrv_from_reg_id,
             c_name_fio,
             reg_code || ' ' || reg_name
      from hr_voctravel
      inner join clients on hrv_from_c_id = c_id
      left join region on hrv_from_reg_id = reg_unicode
      where hrv_from_ho_id = :order_id
      into :hrv_from_ho_id,
           :hrv_number,
           :hrv_from_c_id,
           :hrv_date_begin,
           :hrv_date_end,
           :hrv_days,
           :hrv_day_back,
           :hrv_direction,
           :hrv_class,
           :hrv_status,
           :hrv_date_end_fact,
           :hrv_changer_id,
           :hrv_not_close_access,
           :hrv_from_reg_id,
           :c_name_fio,
           :res_region_name
  do
  begin
    hrv_changer_fio = '';

    if (hrv_changer_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :hrv_changer_id
      into hrv_changer_fio;
    end

    if (hrv_date_end_fact = hrv_date_end) then
      hrv_date_end_fact = null;

    suspend;
  end
end
