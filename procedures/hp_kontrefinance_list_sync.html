CREATE PROCEDURE HP_KONTREFINANCE_LIST_SYNC(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    CALC_DATE D_DATE,
    REFINANCE_KONT_LIST D_STR_LONG)
AS
declare variable res_k_id d_id;
declare variable loan_balance d_currency;
declare variable loan_perc d_currency;
declare variable disb_saldo d_currency;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable ma_id d_integer;
declare variable disb_amount d_currency;
declare variable kontract_no d_str_code;
declare variable kc_id d_id;
begin
  -- Сначала отмечаем все займы для рефинансирования по заявке -1
  if (calc_date is null) then
    exception e_message 'calc_date is empty';

  k_id = 0;
  if (obj_kind = 9) then
  begin
    k_id = obj_id;
    select k_from_ma_app_id, k_amount_approved
    from kontract
    where k_id = :k_id
    into ma_id, disb_amount;

    select kc_from_c_id
    from knclient
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0
    into c_id;
  end else
  if (obj_kind = 11) then
  begin
    ma_id = obj_id;

    if (coalesce(disb_amount, 0) = 0) then
      exception e_message 'Disb_amount is 0';

    select k_id
    from kontract
    where k_from_ma_app_id = :ma_id and
          k_status = 1 and
          k_disb_date <= :calc_date
    order by k_disb_date desc
    rows 1
    into k_id;

    select mac_from_c_id, mac_amount
    from miniappclient
    where mac_from_ma_id = :ma_id and
          mac_amount > 0
    into c_id, disb_amount;
  end
  
  update kontrefinance
  set kr_sync_state = -1
  where kr_from_ma_id = :ma_id;

  if (coalesce(refinance_kont_list, '') <> '') then
  begin
    for select trim(res_str)
    from hp_split_str(:refinance_kont_list, ',')
    into :kontract_no
    do
    begin
      res_k_id = null;

      select k_id
      from kontract
      where k_kontract_no = :kontract_no
      into res_k_id;

      if (res_k_id is not null) then
      begin
        update or insert into kontrefinance (kr_from_ma_id, kr_from_k_id, kr_date, kr_balance,
                                             kr_perc_saldo, kr_sync_state)
        values (:ma_id, :res_k_id, :calc_date, 0, 0, 0)
        matching (kr_from_ma_id, kr_from_k_id);
      end else
      if (kontract_no like '________-____-____-____-____________-_') then -- проверка на формат UUID
      begin
        update or insert into kontrefinance (kr_from_ma_id, kr_loan_uuid, kr_date, kr_balance,
                                             kr_perc_saldo, kr_sync_state)
        values (:ma_id, :kontract_no, :calc_date, 0, 0, 0)
        matching (kr_from_ma_id, kr_loan_uuid);
      end else
        exception e_message 'Loan not found: ' || kontract_no;
    end
  end else
  begin
    -- Заполняем список займов к рефинансированию
    disb_saldo = disb_amount;
    for select res_k_id, loan_balance, loan_perc
    from hp_kontrefinance_list_tmp(:c_id, :calc_date)
    where (loan_balance + loan_perc) < :disb_amount and
          res_k_id <> :k_id
    order by res_avg_repay desc
    into :res_k_id, loan_balance, loan_perc
    do
    begin
      if (disb_saldo < (loan_balance + loan_perc)) then
        Continue;
  
      update or insert into kontrefinance (kr_from_ma_id, kr_from_k_id, kr_date, kr_balance,
                                           kr_perc_saldo, kr_sync_state)
      values (:ma_id, :res_k_id, :calc_date, :loan_balance, :loan_perc, 0)
      matching (kr_from_ma_id, kr_from_k_id);
  
      disb_saldo = disb_saldo - (loan_balance + loan_perc);
    end
  end

  -- Удаляем займы если они не вошли в список
  delete from kontrefinance
  where kr_from_ma_id = :ma_id and
        kr_sync_state = -1;

  if (not exists(select * from kontrefinance
                 where kr_from_ma_id = :ma_id)) then
    exception e_message 'У клиента нет займа для рефинансирования';
end
