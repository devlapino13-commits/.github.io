CREATE PROCEDURE AAREP_GRANTED_LOANS_IPF_TMP(
    REP_REGION_ID ID_TYPE,
    K_ID_IN ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    OFFICE_ID ID_TYPE,
    K_ID ID_TYPE,
    K_KONTRACT_NO STR_100,
    K_DISB_DATE DATE,
    K_AMOUNT_APPROVED CURR_TYPE,
    PR_NAME STR_100,
    KS_FIO STR_100,
    KF_ID ID_TYPE)
AS
begin
  /*Процедура для aarep_granted_loans_ipf разбивает один запрос на два в зависимости от входного параметра.
  Для лушего выбора индексов*/
  k_id_in = coalesce(k_id_in, 0);
  if (k_id_in = 0) then
    for select distinct k_last_reg_id, k_id, k_kontract_no, k_disb_date, k_amount_approved, pr_name, c.c_name_fio, -- кс, который выдал заем
                        k.k_from_kf_id
        from kontstate ks, kontract k, product pr, region_list(:rep_region_id, 1, -1, 1), clients c
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_from_k_id = k_id and
              cast(k_last_reg_id as integer) = id_list and
              k_from_pr_id = pr_id and
              pr_code containing '.ipf.' and
              k_fact_disb_lo_id = c.c_id --added by aleksei on 31-01-2019
        into :office_id, :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :pr_name, :ks_fio, :kf_id
    do
    begin

      suspend;
    end
  else
    for select distinct k_last_reg_id, k_id, k_kontract_no, k_disb_date, k_amount_approved, pr_name, c.c_name_fio, -- кс, который выдал заем
                        k.k_from_kf_id
        from kontstate ks, kontract k, product pr, region_list(:rep_region_id, 1, -1, 1), clients c
        where ks_ol_is_delete = 0 and
              ks_from_k_id = :k_id_in and
              ks_date_begin between :rep_date_begin and :rep_date_end and
              ks_from_k_id = k_id and
              cast(k_last_reg_id as integer) = id_list and
              k_from_pr_id = pr_id and
              pr_code containing '.ipf.' and
              k_fact_disb_lo_id = c.c_id --added by aleksei on 31-01-2019
        into :office_id, :k_id, :k_kontract_no, :k_disb_date, :k_amount_approved, :pr_name, :ks_fio, :kf_id
    do
    begin

      suspend;
    end

end
