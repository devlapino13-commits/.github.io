CREATE PROCEDURE AAREP_OFFICERPAR7_LIST(
    REP_DATE DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    OFFICER_NAME VARCHAR(100),
    BALANCE CURR_TYPE,
    DELINQ_BALANCE7 CURR_TYPE,
    PAR7 CURR_TYPE,
    OP_DATE_BEGIN DATE,
    MEMO VARCHAR(50))
AS
declare variable c_id id_type;
declare variable reg_id integer;
declare variable reg_id_prior integer;
begin
  reg_id_prior = -1;
  for select c_name_fio, op_balance, op_delinq_bal8_180, op_par8_180, c_id, op_date_begin, op_reg_id
      from officerpar7, clients
      where :rep_date between op_date_begin and op_date_end and
            op_par8_180 >= 7.0000 and
            op_kc_id = c_id
      order by op_date_begin desc, op_reg_id
      into officer_name, balance, delinq_balance7, par7, c_id, op_date_begin, reg_id
  do
  begin
    memo = '';
    if (par7 > 100.0000) then
    begin
      balance = null;
      delinq_balance7 = null;
      select sum(oks_balance), sum(iif(:op_date_begin - oks_delinq_date > 7, oks_balance, 0))
      from oksummary
      where :op_date_begin between oks_date_begin and oks_date_end and
            oks_kc_id = :c_id
            and oks_close_state = 0
      into balance, delinq_balance7;
      balance = coalesce(balance, 0);
      delinq_balance7 = coalesce(delinq_balance7, 0);

      if (balance > 0) then
        par7 = cast(delinq_balance7 as double precision) / balance * 100;
      else
        par7 = 0;

      memo = 'Полный запрет к выдачам';
    end

    if (reg_id_prior <> reg_id) then
    begin
      reg_id_prior = reg_id;

      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into reg_name;
    end
    suspend;
  end
end
