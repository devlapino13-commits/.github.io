CREATE PROCEDURE HP_SEND_MAIL_HIGH_ALT_EXP
AS
declare variable curr_day_7 d_integer;
declare variable c_id d_id;
declare variable hired_date d_date;
declare variable hm_mount d_integer;
declare variable exp d_integer;
declare variable date_start d_date;
declare variable mail_to d_str_255;
declare variable mail_subject d_str_255;
declare variable mail_body varchar(1024);
declare variable reg_id d_integer;
declare variable reg_id_tmp d_integer;
declare variable jl_name d_str_100;
declare variable reg_concat d_str_100;
declare variable fio d_str_100;
declare variable birth_date d_date;
declare variable curr_date d_date;
declare variable curr_month d_integer;
begin
  curr_date = current_date;
  curr_day_7 = extract(day from (:curr_date + 7));
  mail_to = 'hr_branch@mbulak.kg, ch.abdykerimova@mbulak.kg, b.abdraeva@mbulak.kg';
  reg_id = 0;

  for select ps.ps_from_c_id, jl.jl_name, rl.id_list
      from hr_pstatus ps
      inner join region_list(1000000, 1, 3, 1) rl on rl.id_list = cast(ps.ps_from_reg_id as integer)
      inner join region reg on reg.reg_unicode = rl.id_list and
            f_reg_mount_koef(reg.reg_unicode) > 1 -- RFCKG-4782
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      where :curr_date between ps.ps_date_begin and ps.ps_date_end and
            ps.ps_is_working = 1 and
            ps.ps_status = 1 and
            (((ps_is_working = 0) and
            ((ps_transfer_office is null) or
            (ps_transfer_office = ps_from_reg_id))) or
            (ps_is_working = 1))
      order by rl.id_list
      into :c_id, :jl_name, :reg_id_tmp

  do
  begin
    mail_body = '';
    hm_mount = 0;

    select ps.ps_date_begin, hm.hm_mount
    from hr_pstatus ps
    inner join hr_mount hm on ps.ps_from_ho_id = hm.hm_ps_from_ho_id and
          ps.ps_id = hm.hm_from_ps_id
    where ps.ps_from_c_id = :c_id
    order by ps.ps_date_begin desc
    rows 1
    into :hired_date, :hm_mount;

    select month_out, last_date
    from hp_high_mountain_exp_calc(:c_id, :hired_date, :curr_date)
    into :exp, :date_start;

    exp = :exp + :hm_mount;

    select c.c_name_fio, c.c_birth_date
    from clients c
    where c.c_id = :c_id
    into :fio, :birth_date;

    if (:reg_id <> :reg_id_tmp) then
    begin
      reg_id = :reg_id_tmp;

      select reg.reg_code || ' ' || reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into :reg_concat;
    end

    if (f_is_valid_date(extract(day from :date_start) || '.' || extract(month from :curr_date) || '.' || extract(year from :curr_date)) = 0) then
      :date_start = extract(day from f_date_month_end(:curr_date)) || '.' || extract(month from :date_start) || '.' || extract(year from :date_start);

    if (:curr_day_7 = extract(day from :date_start) and
        mod(:exp, 60) = 0) then
    begin
      mail_subject = 'Уведомление о высокогорном стаже';
      mail_body = '<b>Внимание !</b><br><br>' || '<b>Высокорный стаж составил <font color="red">' ||((:exp) / 60 * 5) || '</font> лет:</b><br><br>' || 'ОП: <b>' || :reg_concat || '</b><br>' || 'ФИО: <b>' || :fio || '</b><br>' || 'Должность: <b>' || :jl_name || '</b><br><br>' || 'Дата наступления стажа: <b>' || f_datetostr(:curr_date + 7) || '</b>';
      execute procedure hp_send_quick_message('email', :mail_to, :mail_subject, :mail_body, '', 1, 1000000000000);
    end

    if (f_age(:birth_date, :curr_date) = 27 and
        extract(day from :birth_date) = :curr_day_7 and
        extract(month from :birth_date) = extract(month from :curr_date + 7)) then
    begin
      mail_subject = 'Уведомление о высокогорном стаже, 28 лет';
      mail_body = '<b>Внимание !</b><br><br>' || '<b>Исполнилось <font color="red">' || f_age(:birth_date, :curr_date + 7) || '</font> лет:</b><br><br>' || 'ОП: <b>' || :reg_concat || '</b><br>' || 'ФИО: <b>' || :fio || '</b><br>' || 'Должность: <b>' || :jl_name || '</b><br><br>' || 'Дата исполнения: <b>' || f_datetostr(:curr_date + 7) || '</b>';
      execute procedure hp_send_quick_message('email', :mail_to, :mail_subject, :mail_body, '', 1, 1000000000000);
    end
  end
end
