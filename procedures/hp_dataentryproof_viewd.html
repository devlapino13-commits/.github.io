CREATE PROCEDURE HP_DATAENTRYPROOF_VIEWD(
    KI_ID D_INTEGER)
RETURNS (
    BCT_ID D_ID,
    BCT_CODE D_STR_CODE,
    BCT_NAME D_STR_255,
    MARK D_BOOL)
AS
begin
  for
    select
      bct_id,
      bct_code,
      bct_name
    from bki_catalog
    where bct_kind = 'incomeInfoSource'
  into
    :bct_id,
    :bct_code,
    :bct_name
  do
  begin
    mark = iif(exists(
      select * from kk_income_cib
      where kic_from_ki_id = :ki_id
        and kic_from_bct_id = :bct_id), 1, 0);
    suspend;
  end
end
