CREATE PROCEDURE AAREP_DELINQ_LOANS_BY_STATUS(
    REP_REGION_ID D_INTEGER,
    REP_DATE D_DATE)
RETURNS (
    C_NAME_FIO D_STR_100,
    K_KONTRACT_NO D_STR_20,
    KAL_NEW_KF_NAME D_STR_100,
    KAL_DATE D_DATE,
    SBL_NAME D_STR_255,
    ACC_TYPE D_STR_20,
    K_KREDLINE_END D_DATE,
    DELINQ_DAYS_AMOUNT D_INTEGER,
    K_AMOUNT_APPROVED CURR_TYPE,
    SUM_OS CURR_TYPE,
    SUM_PERC CURR_TYPE,
    SUM_TOTAL CURR_TYPE,
    LAST_PAY_DATE D_DATE,
    DAYS_FROM_TRANSFER D_INTEGER)
AS
declare variable REG_ID D_INTEGER;
declare variable K_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable PR_CODE D_STR_100;
declare variable PREV_K_ID ID_TYPE;
declare variable I D_INTEGER;
declare variable C_QTY D_INTEGER;
declare variable C_SHARE_TMP decimal(18,2);
declare variable C_SHARE decimal(18,2);
declare variable DISB_OL_ID ID_TYPE;
begin
  for select id_list
      from region_list(:rep_region_id, 1, 3, 1)
  into :reg_id
  do
  begin
    k_id = 0;
    delinq_days_amount = null;
    acc_type = '';

    for select oks.oks_from_k_id,
               iif(:rep_date - oks.oks_delinq_date < 0, 0, :rep_date - oks.oks_delinq_date),
               'Активный'
        from oksummary oks
        where oks.oks_region_id = :reg_id and
              oks.oks_close_state = 0 and
              :rep_date between oks.oks_date_begin and oks.oks_date_end and
              :rep_date - oks.oks_delinq_date > 0
        union
        select oks.oks_from_k_id, 
               iif(:rep_date - oks.oks_delinq_date < 0, 0, :rep_date - oks.oks_delinq_date),
               'Спец. учет'
        from oksummary oks
        where oks.oks_region_id = :reg_id and
              oks.oks_close_state = 1 and
              :rep_date between oks.oks_date_begin and oks.oks_date_end
    into :k_id,
         :delinq_days_amount,
         :acc_type
    do
    begin
      prev_k_id = 0;
      i = 0;
      c_share_tmp = 0;

      for select kn.k_kontract_no,
                  kn.k_kredline_end,
                  kal.kal_new_kf_name,
                  kal.kal_date,
                  kn.k_amount_approved,
                  pr.pr_code,
                  kc.kc_from_c_id,
                  (kc.kc_amount_approved / kn.k_amount_approved),
                  kn.k_kolich_client_approved,
                  kn.k_fact_disb_ol_id
      from kontract kn
      inner join knclient kc on kc.kc_from_k_id = kn.k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kn.k_from_kf_id
      inner join product pr on pr.pr_id = kn.k_from_pr_id
      where kn.k_id = :k_id and
            kc.kc_amount_approved > 0 and
            kal.is_delete = 0 and
            :rep_date between kal.kal_date and kal.kal_date_end
      into :k_kontract_no,
            :k_kredline_end,
            :kal_new_kf_name,
            :kal_date,
            :k_amount_approved,
            :pr_code,
            :c_id,
            :c_share,
            :c_qty,
            :disb_ol_id
      do
      begin
        days_from_transfer = cast(:rep_date as date) - cast(:kal_date as date);
        last_pay_date = null;
        select max(pay_date)
        from (
            select max(ok_ol_date) as pay_date from operkred
            where ok_from_k_id = :k_id and ok_ol_is_active = 1 and
                  ok_type = 4 and
                  (ok_os + ok_perc < 0)
            union all
            select max(oko_ol_date) as pay_date from operkroff
            where oko_from_k_id = :k_id and
                  oko_ol_is_active = 1 and
                  oko_type = 4
        ) into :last_pay_date;

        sbl_name = 'ВИПС офиса';
        select sbl.sbl_name
        from sb_akt sba
        inner join sb_list sbl on sbl.sbl_id = sba.sba_from_sbl_id
        where sba.sba_from_k_id = :k_id and
              sba.sba_is_delete = 0 and
              :rep_date between sba.sba_date_begin and sba.sba_date_end and
              sba.sba_status = 2
        into :sbl_name;

        select cl.c_name_fio
        from clients cl
        where cl.c_id = :c_id
        into :c_name_fio;

        if (k_id <> prev_k_id) then
        begin
            c_share_tmp = 0; --Временная переменная для подсчёта доли клиента в процентах
            i = 1;                --Итератор для последнего клиента
        end


        if (c_qty > 1) then
        begin
          prev_k_id = k_id;

          if (i <> c_qty) then
          begin
            c_share_tmp = c_share_tmp + c_share;
          end

          if (i = c_qty) then
          begin
            c_share = 1 - c_share_tmp;
          end
          i = i + 1;
       end

        if (acc_type = 'Активный' and pr_code not containing 'ipf') then
        begin
          select sum(zad_os),
                 sum(sh_calc_perc),
                 sum(zad_os) + sum(sh_calc_perc) + sum(round(sh_calc_perc * 0.02)) + sum(sh_calc_fine)
          from aarep_make_zad_list(:k_id, :rep_date, 1)
          group by :k_id
          into :sum_os,
               :sum_perc,
               :sum_total;

          sum_os = sum_os * c_share;
          sum_perc = sum_perc * c_share;
          sum_total = sum_total * c_share;
        end

        if (acc_type = 'Активный' and pr_code containing '.ipf.') then
        begin
          select sum(zad_os),
                 sum(zad_perc),
                 sum(zad_os) + sum(zad_perc) + sum(round(zad_perc * 0.02)) + sum(sh_calc_fine)
          from aarep_make_zad_list(:k_id, :rep_date, 1)
          group by :k_id
          into :sum_os,
               :sum_perc,
               :sum_total;

          sum_os = sum_os * c_share;
          sum_perc = sum_perc * c_share;
          sum_total = sum_total * c_share;
        end

        if (acc_type = 'Спец. учет') then
        begin
          select sum(oko.oko_sh_os),
                 sum(oko.oko_sh_perc),
                 sum(oko.oko_sh_os) + sum(oko.oko_sh_perc)
          from operkroff oko
          where oko.oko_from_c_id = :c_id and
                oko.oko_from_k_id = :k_id and
                oko.oko_ol_is_active = 1 and
                oko.oko_type = 1
          group by oko.oko_from_c_id, oko.oko_from_k_id
          --order by oko.oko_ol_date desc
          --rows 1
          into :sum_os,
               :sum_perc,
               :sum_total;
        end
        suspend;
      end
    end
  end
end
