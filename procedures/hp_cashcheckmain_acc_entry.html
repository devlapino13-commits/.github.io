CREATE PROCEDURE HP_CASHCHECKMAIN_ACC_ENTRY(
    CCM_ID D_INTEGER)
RETURNS (
    OL_ID D_ID,
    ERR_TEXT D_STR_255)
AS
declare variable dtoper d_date;
declare variable curr_code d_str_lang;
declare variable reg_id d_integer;
declare variable cashier_id d_id;
declare variable cashier_name d_str_100;
declare variable ccm_num_cash d_integer;
declare variable un_id d_id;
declare variable is_deficit boolean;
declare variable acc d_str_20;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable amount d_currency;
declare variable cash_id d_integer;
declare variable is_ipf d_bool;
declare variable ol_memo d_str_long;
begin
  err_text = '';
  curr_code = '417';

  for
    select
      ccm_date,
      ccm_from_reg_id,
      c_id,
      c_name_fio,
      iif(ccm_num_cash = 0 or ccm_num_cash is null, 1, ccm_num_cash),
      ccm_from_un_id,
      coalesce(ccm_cash_saldo_mannual,0) > coalesce(ccm_cash_saldo_fact,0),
      abs(coalesce(ccm_cash_saldo_mannual,0) - coalesce(ccm_cash_saldo_fact,0))
    from cashcheckmain
      inner join clients
        on c_id = ccm_cashier_curr
    where ccm_id = :ccm_id
      and ccm_is_delete = 0
      and coalesce(ccm_cash_saldo_mannual,0) <> coalesce(ccm_cash_saldo_fact,0)
      and ccm_ol_id is null
  into
    :dtoper,
    :reg_id,
    :cashier_id,
    :cashier_name,
    :ccm_num_cash,
    :un_id,
    :is_deficit,
    :amount
  do
  begin
    ol_memo = f_format('%1 при инвентаризации кассы; %2',
      trim(iif(:is_deficit, 'Недостача', 'Излишек')), :cashier_name);

    insert into operlist (
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id,
      ol_document_client
    )
    values (
      :reg_id,
      iif(:is_deficit, 2, 1),
      2,
      :dtoper,
      :ccm_id,
      iif(:is_deficit, 0, 1),
      :ol_memo,
      :un_id,
      :cashier_id
    )
    returning ol_id
    into :ol_id;

    acc = '10001/' || ccm_num_cash;
    acc_dt = (
      select
        al_id
      from acc_list
      where al_from_reg_id = :reg_id
        and al_currency = :curr_code
        and al_account = iif(:is_deficit, '11504/1', :acc));
    acc_kt = (
      select
        al_id
      from acc_list
      where al_from_reg_id = :reg_id
        and al_currency = :curr_code
        and al_account = iif(:is_deficit, :acc, '80509/1'));

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_summa,
      ob_summa_val,
      ob_client_dt,
      ob_client_kt
    )
    values (
      :ol_id,
      :curr_code,
      1,
      :acc_dt,
      :acc_kt,
      :amount,
      :amount,
      iif(:is_deficit, :cashier_id, null),
      iif(:is_deficit, null, :cashier_id)
    );

    if (ol_id > 0) then
    begin
      acc = iif(:is_deficit, :acc_kt, :acc_dt);

      select
        csp_id,
        iif(csp_cls_al_id = :acc, 0, 1)
      from cashplace
      where csp_office_id = :reg_id
        and (csp_cls_al_id = :acc or csp_ipf_al_id = :acc)
      into
        :cash_id,
        :is_ipf;

      insert into opercash (
        opc_date,
        opc_doc_no,
        opc_office_id,
        opc_from_csp_id,
        OPC_OPER_KIND,
        opc_sign,
        opc_amount,
        opc_cashier_id,
        opc_from_c_id,
        opc_is_ipf,
        opc_corr_al_id,
        opc_corr_reg_id,
        opc_memo,
        opc_from_ol_id,
        opc_from_un_id)
      values (
        :dtoper,
        :ccm_id,
        :reg_id,
        :cash_id,
        f_enum_id('cashoperkind', iif(:is_deficit, 'minus.saldo', 'plus.saldo')),
        iif(:is_deficit, -1, 1),
        :amount,
        :cashier_id,
        :cashier_id,
        :is_ipf,
        iif(:is_deficit, :acc_dt, :acc_kt),
        :reg_id,
        :ol_memo,
        :ol_id,
        :un_id);
    end

    suspend;
  end
end
