CREATE PROCEDURE HP_VERSIONLOG_INFO(
    ITEM_NAME D_STR_100 NOT NULL,
    VER_DATETIME D_DATETIME)
RETURNS (
    LAST_DATE D_DATETIME,
    LAST_USER D_STR_CODE,
    LAST_BODY D_STR_CODE,
    MAX_DATE D_DATETIME)
AS
declare variable last_body_blob d_blob_text;
declare variable tmp_blob_w blob sub_type 1 segment size 80 character set win1251;
declare variable curr_body_blob d_blob_text;
declare variable pos d_integer;
declare variable is_exists d_bool;
declare variable pkg_name d_str_code;
declare variable obj_type d_integer;
declare variable tmp_blob d_blob_binary;
declare function prepare_blob(bl_in d_blob_text)
returns blob
as
declare variable bl_out d_blob_text;
declare variable ppos d_integer;
begin
  bl_out = bl_in;
  bl_out = lower(bl_out);

  ppos = position(ascii_char(13)||ascii_char(10) || 'as' || ascii_char(13)||ascii_char(10), bl_out);

  if (ppos > 0) then
    bl_out = substring(bl_out from ppos + 1);
  bl_out = replace(bl_out, ' ', '');
  bl_out = replace(bl_out, ascii_char(13), '');
  bl_out = replace(bl_out, ascii_char(10), '');

  --Убрать завершающую семиколон ";"
  ppos = char_length(bl_out);
  if (substring(bl_out from ppos for 1) = ';') then
    bl_out =  substring(bl_out from 1 for ppos - 1);
  return bl_out;
end

begin
  last_body = '';

  if (item_name is null) then
    item_name = '';

  if (not exists(select * from ibe$version_history v
  where v.ibe$vh_object_name = :item_name)) then
  begin
    for select rdb$package_name
    from rdb$packages
    union
    select null
    from rdb$database
    into pkg_name
    do
    begin
      select p.rdb$procedure_source
      from rdb$procedures p
      where p.rdb$package_name is not distinct from :pkg_name and
            p.rdb$procedure_name = :item_name
      into tmp_blob;

      if (tmp_blob is null) then
      begin
        select f.rdb$function_source
        from rdb$functions f
        where f.rdb$package_name is not distinct from :pkg_name and
              f.rdb$function_name = :item_name
        into tmp_blob;
      end

      if (tmp_blob is not null) then
        break;
    end

    if (tmp_blob is null) then
    begin
      select t.rdb$trigger_source
      from rdb$triggers t
      where t.rdb$trigger_name = :item_name
      into tmp_blob;
    end

    if (tmp_blob is not null) then
    begin
      begin
        curr_body_blob = tmp_blob;
        when any do
        begin
          tmp_blob_w = tmp_blob;
          curr_body_blob = tmp_blob_w;
        end
      end

      curr_body_blob = prepare_blob(curr_body_blob);

      last_body = f_md5_blob(curr_body_blob);
    end else
      last_body = 'deleted';

    last_date = current_timestamp;
    last_user = 'SISLAMOV';
    max_date = current_timestamp;

    suspend;
    exit;
  end

  for select ibe$vh_modify_date as change_date,
             ibe$vh_user_name as change_user,
             ibe$vh_object_name as obj_name,
             ibe$vh_body as obj_body,
             ibe$vh_object_type as obj_type
  from ibe$version_history
  where ibe$vh_object_name = :item_name and
        ibe$vh_modify_date >= coalesce(:ver_datetime, ibe$vh_modify_date)
  order by ibe$vh_modify_date desc
  as cursor cur
  do
  begin
    if (max_date is null) then
      max_date = cur.change_date;

    obj_type = cur.obj_type;

    is_exists = 0;
    if (obj_type in (1, 4)) then
    begin
      for select rdb$package_name
      from rdb$packages
      union
      select null
      from rdb$database
      into pkg_name
      do
      begin
        if (obj_type = 1) then
        begin
          if (exists(
          select * from rdb$procedures p
          where p.rdb$package_name is not distinct from :pkg_name and
                p.rdb$procedure_name = cur.obj_name)) then
            is_exists = 1;
        end else
        begin
          if (exists(
          select * from rdb$functions f
          where f.rdb$package_name is not distinct from :pkg_name and
                f.rdb$function_name = cur.obj_name)) then
            is_exists = 1;
        end
      end
    end else
    if (obj_type = 2) then
    begin
    for  select t.rdb$trigger_source from rdb$triggers t
      where t.rdb$trigger_name = cur.obj_name
      into :curr_body_blob do
        is_exists = 1;
    end

    if (is_exists = 0) then
    begin
      last_date = cur.change_date;
      last_user = cur.change_user;

      last_body = 'deleted';

      suspend;
      exit;
    end

    begin
      if (obj_type <> 2) then
        curr_body_blob = cur.obj_body;
      when any do
      begin
        tmp_blob_w = cur.obj_body;
        curr_body_blob = tmp_blob_w;
      end
    end

    curr_body_blob = prepare_blob(curr_body_blob);

    if (last_body_blob is not null and last_body_blob is distinct from curr_body_blob) then
    begin
      last_body = f_md5_blob(last_body_blob);
      suspend;

      exit;
    end

    last_body_blob = curr_body_blob;
    last_date = cur.change_date;
    last_user = cur.change_user;
  end

  if (max_date is not null) then
  begin
    last_body = f_md5_blob(last_body_blob);
    suspend;
  end
end
