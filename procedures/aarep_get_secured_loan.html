CREATE PROCEDURE AAREP_GET_SECURED_LOAN(
    REP_PR_CODE VARCHAR(100),
    REP_PR_LOAN_TYPE INTEGER,
    REP_PR_PRODUCT_KIND INTEGER,
    REP_PR_ID INTEGER,
    REP_K_DISB_DATE DATE,
    REP_KOL_CLI INTEGER,
    REP_K_ID ID_TYPE)
RETURNS (
    IS_SECURED SMALLINT,
    MEMO VARCHAR(100))
AS
declare variable Z_NAME_FIO varchar(255);
begin
  -- Данная процедура определяет кредит с обеспечением или без обеспечения
  -- если результат вернет = 1 то кредит с обеспечением
  if (rep_pr_loan_type is null) then
    rep_pr_loan_type = -1;
  if (rep_pr_product_kind is null) then
    rep_pr_product_kind = -1;
  if (rep_pr_id is null) then
    rep_pr_id = 0;
  if (rep_kol_cli is null) then
    rep_kol_cli = 0;

  memo = '';
  if (rep_k_id is null) then
  begin
    is_secured = -1;
    memo = 'ID контракта обязательное поле, необходимо задать!';
    suspend;
    exit;
  end

  if ((rep_pr_code is null or rep_pr_loan_type not in (0, 1) or rep_pr_product_kind not in (1, 2, 3, 4)) and
      rep_pr_id > 0) then
  begin
    --  если не задан один из вышестоящих параметров то вытягивает из продукта
    rep_pr_code = null;
    rep_pr_loan_type = -1;
    rep_pr_product_kind = -1;

    select pr_code, pr_loan_type, pr_product_kind
    from product
    where pr_id = :rep_pr_id
    into rep_pr_code, rep_pr_loan_type, rep_pr_product_kind;
  end

  if ((rep_pr_code is null or rep_pr_loan_type not in (0, 1) or rep_pr_product_kind not in (1, 2, 3, 4) or rep_k_disb_date is null or rep_kol_cli = 0) and
      rep_k_id > 0) then
  begin
    --  если не задан один из вышестоящих параметров то вытягивает из продукта
    rep_pr_code = null;
    rep_pr_loan_type = -1;
    rep_pr_product_kind = -1;

    select pr_code, pr_loan_type, pr_product_kind, k_disb_date, k_kolich_client_approved
    from product, kontract
    where pr_id = k_from_pr_id and
          k_id = :rep_k_id
    into rep_pr_code, rep_pr_loan_type, rep_pr_product_kind, rep_k_disb_date, rep_kol_cli;
  end

  if (rep_pr_code is null) then
  begin
    is_secured = -1;
    memo = 'Не правильно заданы входящие параметры';
    suspend;
    exit;
  end

  if (rep_pr_code containing '.winner.' and
      rep_pr_code containing '.emp.' and
      rep_pr_code containing '.ind.') then
  begin
    is_secured = 0;
    suspend;
    exit;
  end

  --      section = 1 - обеспеченный
  --      section = 2 - не обеспеченный
  --      is_secured = 0 - без обеспечение
  --      is_secured = 1 - с обеспечение

  if (rep_pr_loan_type = 1) then
  begin
    if (rep_pr_code containing '.IND.' and
        rep_pr_code containing '.BZ.') then
    begin
      z_name_fio = null;
      select z_name_fio
      from zalog
      where z_from_k_id = :rep_k_id
      order by z_name_fio desc
      rows 1
      into z_name_fio;
      z_name_fio = trim(z_name_fio);
      if (z_name_fio is null) then
        z_name_fio = '';

      if (z_name_fio = '') then
        is_secured = 0;
      else
        is_secured = 1;
    end
    else
    begin
      if (rep_pr_code containing '.IND.' and
          rep_pr_code containing '.OB.') then
        is_secured = 1;
      else
        is_secured = 0;
    end
    --        добавлено по письму от 23.04.2015 Ёльнура
    if (rep_k_disb_date >= '01.01.2015' and
        rep_pr_code containing '.PP.' and
        rep_pr_code similar to '%(.ZNAKOM.|.UVAJENIE.|.DOVERIE.)%') then
    begin
      is_secured = 1;
    end
    if (rep_k_disb_date between '01.01.2015' and '14.01.2015') then
    begin
      if (rep_pr_code containing '.ONE.') then
      begin
        is_secured = 0;
      end
    end
    else
    if (rep_k_disb_date >= '15.01.2015') then
    begin
      if (rep_pr_code containing '.ONE.') then
      begin
        is_secured = 1;
      end
    end

  end
  else
  begin -- все групповые обеспечены с поручительствовами
    --        если параллельный займ и кол-во участников = 1 тогда будет не обеспеченным кредитом
    if (rep_kol_cli = 1 and
        rep_pr_product_kind = 2) then
      is_secured = 0;
    else
      is_secured = 1;
    if (rep_k_disb_date between '15.12.2014' and '14.01.2015' and
        is_secured = 1) then -- добавлено от 21.01.2014 по письму Эльнуры, тема письма: ТЗ -  Отчет по ПСК (для ЦБ) - СРОЧНО!
    begin
      is_secured = 0;
    end
    --        добавлено по письму от 23.04.2015 Эльнура
    if (rep_k_disb_date between '01.01.2015' and '14.01.2015') then
    begin
      if (rep_pr_code similar to '%(.ASALAM.|.KEL.|.PARTNER.|.ONE.)%' or (rep_pr_product_kind = 2 and
          rep_kol_cli = 2)) then
      begin
        is_secured = 0;
      end
    end
    else
    if (rep_k_disb_date >= '15.01.2015') then
    begin
      if (rep_pr_code similar to '%(.ASALAM.|.KEL.|.PARTNER.|.ONE.)%' or (rep_pr_product_kind = 2 and
          rep_kol_cli = 2)) then
      begin
        is_secured = 1;
      end
    end
    if (rep_kol_cli = 1 and
        rep_pr_product_kind = 2 and
        rep_k_disb_date >= '01.01.2015') then
    begin
      is_secured = 0;
    end
  end

  /*
        Парный 1+1:   15.12.2014 - 14.01.2015 - необеспечен
                      15.01.2015 - 30.06.2015 - обеспечен
                      — 01.07.2015 - необеспечен
        */
  if ((rep_pr_code containing ('.ONE.')) /*and (k_kolich_client_approved = 2)*/ ) then
  begin
    if (rep_k_disb_date between cast('15.12.2014' as date) and cast('14.01.2015' as date)) then
      is_secured = 0;
    if (rep_k_disb_date between cast('15.01.2015' as date) and cast('30.06.2015' as date)) then
      is_secured = 1;
    if (rep_k_disb_date >= cast('01.07.2015' as date)) then
      is_secured = 0;
  end

  -- Микрозаем для сотрудников Ц обеспечен
  if (rep_pr_code containing ('.EMP.')) then
    is_secured = 1;
  -- << ----- igusev 21.07.2015 ----

  --      Добавлено по письму Эльнуры от 17.08.2015 тема письма:
  --      Приказ: №26-06-15 од от 25.06.15г. об отмене Индив. микрозайма и изменени...
  --      групповые и парные займы выданные с 1 июл¤ 2015 считаютс¤ необеспеченными
  if (rep_k_disb_date >= cast('01.07.2015' as date) and
      rep_pr_product_kind in (1, 2)) then
  begin
    is_secured = 0;
  end

  --    Разработка по приказу от 26.10.2015
  if (rep_pr_code similar to '%(.BARAKAT.|.BARAKAT PLUS.)%') then
    is_secured = 0;

  if (rep_pr_code containing '.FAMILY') then
  begin
    if (rep_k_disb_date >= cast('22.10.2015' as date)) then
      is_secured = 1;
    else
      is_secured = 0;
  end

  -------------------------------------------------------------------------->>
  /* Добавлено, согласно письма от Эльнуры, от 19.06.2016

    Отчет по средневзвешенному ПСК (Отчеты/Кредитные отчеты) неверно относит займы по продукту «Групповой микрозаем на патент» в категорию Необеспеченные.

    Продукт является обеспеченным поручительством и в отчете ПСК относится к категории «Потребительские займы с иным обеспечением» .
    Пожалуйста, исправьте отчет.
    */
  if (rep_pr_code containing '.PATENT.' and
      rep_pr_code containing '.GR.') then
  begin
    is_secured = 1;
  end
  --------------------------------------------------------------------------<<
  --  добавлено по письму от Эльнуры от 17.10.2016
  if (is_secured = 1 and
      rep_pr_code containing '.EMP.' and
      rep_pr_code containing '.FOREIGN.') then
  begin
    is_secured = 0;
  end

  -- все микрозаймы по продуктам Экспресс, Шанс, Благо, Успех, Параллельный, Патент, Патент плюс и Статус в группу "иные микрозаймы" от 27.09.2018
  -- у Зары уточнил их нужно отнести к категонии "Необеспеченные займы"
  if (is_secured = 1 and
      rep_k_disb_date >= '01.08.2018' and
      rep_pr_code similar to '%(.express.|.shans.|.blago.|.uspeh.|.loyal.|.patent plus.|.status.|.patent.)%') then
    is_secured = 0;
  if (rep_pr_code containing '.zalog_car.') then
    is_secured = 1;
  if (rep_pr_code containing '.zalog.') then
    is_secured = 1;
  if (rep_pr_code containing '.education.') then
    is_secured = 0;
  -- добавлено по письмо Елены от 12.04.2021
  if (is_secured = 1 and rep_pr_code containing '.reliable_emp.') then
    is_secured = 0;
  -- Письмо Зары, по продукту Лето Плюс от 21.07.2022
  if (rep_pr_code containing '.summer_plus.') then
    is_secured = 0;
  if (rep_pr_code containing '.comfortable.') then
    is_secured = 0;


  suspend;
end
