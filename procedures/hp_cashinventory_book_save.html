CREATE PROCEDURE HP_CASHINVENTORY_BOOK_SAVE(
    IN_CIL_ID D_INTEGER,
    IN_ROUND_AMOUNT D_CURRENCY,
    UN_ID D_ID)
AS
declare variable akt_oper_kind_code d_str_code;
declare variable opc_id d_integer;
declare variable cfl_from_cil_id d_integer;
declare variable cfl_from_csp_id d_integer;
declare variable cfl_date d_date;
declare variable cfl_cashier_id d_id;
declare variable cfl_inspector_id d_id;
declare variable cfl_fact_amount d_currency;
declare variable main_opc_id d_integer;
declare variable oper_memo d_str_long;
declare variable csp_office_id d_integer;
declare variable amount d_currency;
declare variable al_id d_str_20;
declare variable is_ipf d_bool;
declare variable oper_kind_other d_integer;
declare variable diff_amount d_currency;
declare variable oper_sign d_integer;
declare variable cil_inv_amount d_currency;
declare variable cil_parent_id d_integer;
declare variable cfl_id d_integer;
declare variable cil_oper_kind d_integer;
declare variable oper_name d_str_255;
declare variable akt_person_id d_id;
declare variable saldo d_currency;
declare variable in_round_ipf d_currency;
begin
  select cil_parent_id,
         f_enum_name('cashoperkind', cil_oper_kind),
         f_enum_code('cashoperkind', cil_oper_kind)
  from cashinvlist
  where cil_id = :in_cil_id
  into cil_parent_id, oper_name, akt_oper_kind_code;

  if (row_count = 0) then                     --
    exception e_message 'Акт не найден, id: ' || coalesce(cfl_from_cil_id, '< Null>');

  if (cil_parent_id is not null) then
  begin
    cfl_id = in_cil_id;
    in_cil_id = cil_parent_id;
  end else
  begin
    select cil_id
    from cashinvlist
    where cil_parent_id = :in_cil_id and
          cil_is_delete = 0
    into cfl_id;
  end

  if (exists(select * from opercash
             where opc_from_cil_id = :in_cil_id and
                   opc_is_delete = 0)) then
    exception e_message 'По акту уже созданы приходные проводки';

  select cim_from_c_id
  from cashinvmembers
  where cim_from_cil_id = :in_cil_id and
        cim_role_kind = -1
  into akt_person_id;

  for select cil_id, cil_oper_kind, cil_from_csp_id, cil_date, cil_inv_amount,
             cil_cashier_id, cil_inspector_id, csp_office_id
  from cashinvlist, cashplace
  where cil_id = :in_cil_id and
        cil_from_csp_id = csp_id
  into cfl_from_cil_id, cil_oper_kind, cfl_from_csp_id, cfl_date, cil_inv_amount,
       cfl_cashier_id, cfl_inspector_id, csp_office_id
  do
  begin
    oper_memo = oper_name;
    if (akt_person_id is null) then
      akt_person_id = cfl_cashier_id;

    saldo = f_cash_saldo(cfl_from_csp_id, cfl_date, 1);

    in_round_ipf = round(saldo) - saldo;
    if (in_round_ipf <> 0) then
    begin
      is_ipf = 1;

      if (in_round_ipf > 0) then
      begin
        amount = in_round_ipf;
        oper_sign = 1;
        oper_kind_other = f_enum_id('cashoperkind', 'plus.round');
        oper_memo = oper_memo || ', округление остатка';
      end else
      begin
        amount = -in_round_ipf;
        oper_sign = -1;
        oper_kind_other = f_enum_id('cashoperkind', 'minus.round');
        oper_memo = oper_memo || ', округление остатка';
      end

      -- Излишка или недосдача с терминалом всегда берем из настройки прихода по излишке
      al_id = f_cashacc(csp_office_id, oper_kind_other, is_ipf);

      insert into opercash (opc_date, opc_parent_id, opc_from_csp_id, opc_oper_kind, opc_sign,
                            opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                            opc_manager_id, opc_from_cil_id, opc_memo, opc_from_un_id)
      values (:cfl_date, :main_opc_id, :cfl_from_csp_id, :oper_kind_other, :oper_sign,
              :amount, null, :is_ipf, :al_id,
              :cfl_inspector_id, :cfl_from_cil_id, :oper_memo, :un_id)
      returning opc_id
      into opc_id;

      if (main_opc_id is null) then
        main_opc_id = opc_id;

      in_round_amount = in_round_amount - in_round_ipf;
    end

    if (in_round_amount <> 0) then
    begin
      is_ipf = 0;
      if (in_round_amount > 0) then
      begin
        amount = in_round_amount;
        oper_sign = 1;
        oper_kind_other = f_enum_id('cashoperkind', 'plus.round');
        oper_memo = oper_memo || ', округление остатка';
      end else
      begin
        amount = -in_round_amount;
        oper_sign = -1;
        oper_kind_other = f_enum_id('cashoperkind', 'minus.round');
        oper_memo = oper_memo || ', округление остатка';
      end

      -- Излишка или недосдача с терминалом всегда берем из настройки прихода по излишке
      al_id = f_cashacc(csp_office_id, oper_kind_other, is_ipf);

      insert into opercash (opc_date, opc_parent_id, opc_from_csp_id, opc_oper_kind, opc_sign,
                            opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                            opc_manager_id, opc_from_cil_id, opc_memo, opc_from_un_id)
      values (:cfl_date, :main_opc_id, :cfl_from_csp_id, :oper_kind_other, :oper_sign,
              :amount, null, :is_ipf, :al_id,
              :cfl_inspector_id, :cfl_from_cil_id, :oper_memo, :un_id)
      returning opc_id
      into opc_id;

      if (main_opc_id is null) then
        main_opc_id = opc_id;
    end

    diff_amount = cfl_fact_amount - cil_inv_amount;

    cfl_fact_amount = 0;
    select cil_inv_amount
    from cashinvlist
    where cil_id = :cfl_id
    into cfl_fact_amount;

    diff_amount = cfl_fact_amount - cil_inv_amount;
    oper_memo = oper_name;
    if (diff_amount <> 0) then
    begin
      is_ipf = 0;

      if (diff_amount > 0) then
      begin
        amount = diff_amount;
        oper_sign = 1;
        oper_kind_other = f_enum_id('cashoperkind', 'plus.saldo');
        oper_memo = oper_memo || ', излишка';
      end else
      begin
        amount = -diff_amount;
        oper_sign = -1;
        oper_kind_other = f_enum_id('cashoperkind', 'minus.saldo');
        oper_memo = oper_memo || ', недостача';
      end

      -- Излишка или недосдача с терминалом всегда берем из настройки прихода по излишке
      al_id = f_cashacc(csp_office_id, 'plus.saldo', is_ipf);

      insert into opercash (opc_date, opc_parent_id, opc_from_csp_id, opc_oper_kind, opc_sign,
                            opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                            opc_manager_id, opc_from_cil_id, opc_memo, opc_from_un_id)
      values (:cfl_date, :main_opc_id, :cfl_from_csp_id, :oper_kind_other, :oper_sign,
              :amount, :akt_person_id, :is_ipf, :al_id,
              :cfl_inspector_id, :cfl_from_cil_id, :oper_memo, :un_id)
      returning opc_id
      into opc_id;

      if (main_opc_id is null) then
        main_opc_id = opc_id;
    end
  end

  if (akt_oper_kind_code in ('check.close', 'check.audit')) then
  begin
    oper_sign = 0;
    amount = 0;
    al_id = null;
    is_ipf = 0;

    if (cfl_inspector_id is not null) then
    begin
      update opercash
      set opc_manager_id = :cfl_inspector_id
      where opc_from_csp_id = :cfl_from_csp_id and
            opc_date = :cfl_date and
            opc_is_delete = 0 and
            opc_manager_id is null;
    end

    oper_memo = oper_name;
    insert into opercash (opc_date, opc_parent_id, opc_from_csp_id, opc_oper_kind, opc_sign,
                          opc_amount, opc_from_c_id, opc_is_ipf, opc_corr_al_id,
                          opc_manager_id, opc_from_cil_id, opc_memo, opc_from_un_id)
    values (:cfl_date, :main_opc_id, :cfl_from_csp_id, :cil_oper_kind, :oper_sign,
            :amount, null, :is_ipf, :al_id,
            :cfl_inspector_id, :cfl_from_cil_id, :oper_memo, :un_id);
  end
end
