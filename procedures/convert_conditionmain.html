CREATE PROCEDURE CONVERT_CONDITIONMAIN
AS
declare variable tmp$id type of column conditionmain.cdm_id;
declare variable tmp$sql type of column conditionmain.cdm_sql;
declare variable tmp$i d_id;
declare variable tmp$j d_integer;
declare variable tmp$str d_str_255;
declare variable tmp$res_info_text d_str_100;
declare variable tmp$res_error_text d_str_255;
declare variable tmp$res_error_is_critical d_integer;
declare variable tmp$enter char(1);
declare variable tmp$var d_str_8100;
declare variable tmp$header d_str_8100;
declare variable tmp$body type of column conditionmain.cdm_sql;
declare variable tmp$kind_str type of column conditionmain.cdm_object_kind;
declare variable tmp$kind_int d_integer;
declare variable tmp$db_id d_integer;
begin
  -- Неразрывный пробел
  TMP$ENTER = ascii_char(64);
  TMP$DB_ID = F_GET_DB_ID();
  for
    select
        cm.CDM_ID
        , cm.CDM_SQL
      from
        ConditionMain as cm
      into
          :TMP$ID
        , :TMP$SQL
    do
    begin

      TMP$HEADER = '';
      TMP$VAR    = '';

      -- Заменяем все анл буквы A..Z, на a..z
--      TMP$I = 65;
--      while (:TMP$I < 91) do
--      begin
--        TMP$SQL = cast(replace(:TMP$SQL, ascii_char(:TMP$I), ascii_char(:TMP$I + 32)) as type of column conditionmain.cdm_sql);
--        TMP$I = :TMP$I + 1;
--      end

      TMP$SQL = replace(:TMP$SQL, ascii_char(13) || ascii_char(10), :TMP$ENTER);
      TMP$SQL = replace(:TMP$SQL, ascii_char(10), :TMP$ENTER);
      TMP$SQL = replace(:TMP$SQL, ascii_char(13), :TMP$ENTER);
      TMP$SQL = replace(:TMP$SQL, :TMP$ENTER || :TMP$ENTER, :TMP$ENTER);
      TMP$SQL = replace(:TMP$SQL, :TMP$ENTER || 'As' || :TMP$ENTER, ' as ');
      TMP$SQL = replace(:TMP$SQL, '@begin--', '@begin --');

--      TMP$SQL = replace(:TMP$SQL, 'user_session', 'USER_SESSION');
--      TMP$SQL = replace(:TMP$SQL, 'FINSTART_USER_LOGIN', 'FINSTART_USER_LOGIN');

      -- Удаляем если есть послений символ ";"
      TMP$STR = right(:TMP$SQL, 1);
      if (TMP$STR = ';') then
        TMP$SQL = left(:TMP$SQL, char_length(:TMP$SQL) - 1);

      TMP$I = position('execute', :TMP$SQL);
      TMP$J = position('returns', :TMP$SQL);

      -- Не execute block
      if ((:TMP$I = 0) and (:TMP$J = 0)) then
      begin
        TMP$BODY   = TMP$SQL;

        TMP$BODY = left(:TMP$SQL, position('from', :TMP$SQL) - 1);
        TMP$BODY = right(:TMP$BODY, char_length(:TMP$BODY) - (position('select', :TMP$SQL) + 6));
        TMP$BODY = 'for ' || :TMP$ENTER || :TMP$SQL || :TMP$ENTER || ' into ' || :TMP$BODY || :TMP$ENTER || ' do suspend; end';

        TMP$J = position(':k_id', :TMP$SQL);
        if (:TMP$J > 0) then
          TMP$KIND_INT = 1;
        else
          TMP$KIND_INT = 0;
      end
      else
      begin
        -- execute block

        TMP$HEADER = left(:TMP$SQL, :TMP$J - 1);
        TMP$BODY   = right(:TMP$SQL, char_length(:TMP$SQL) - (char_length(:TMP$HEADER) + 7));

        TMP$J = position('@begin@', :TMP$BODY);
        TMP$I = position('@begin ', :TMP$BODY);
        if ((:TMP$J = 0) or ((:TMP$I > 0) and (:TMP$I < :TMP$J))) then
          TMP$J = :TMP$I;
        TMP$I = position(' begin@', :TMP$BODY);
        if ((:TMP$J = 0) or ((:TMP$I > 0) and (:TMP$I < :TMP$J))) then
          TMP$J = :TMP$I;
        TMP$I = position(' begin ', :TMP$BODY);
        if ((:TMP$J = 0) or ((:TMP$I > 0) and (:TMP$I < :TMP$J))) then
          TMP$J = :TMP$I;
        TMP$I = position('@begin --', :TMP$BODY);
        if ((:TMP$J = 0) or ((:TMP$I > 0) and (:TMP$I < :TMP$J))) then
          TMP$J = :TMP$I;

        TMP$I = position('declare', :TMP$BODY);

        if ((:TMP$I > 0) and (:TMP$I < :TMP$J)) then
        begin
          TMP$VAR   = left(:TMP$BODY, :TMP$J - 1);
          TMP$I     = position('declare', :TMP$VAR) - 1;
          TMP$VAR   = right(:TMP$VAR, char_length(:TMP$VAR) - :TMP$I);
        end

        TMP$BODY  = right(:TMP$BODY, char_length(:TMP$BODY) - (:TMP$J + 6));

        -- проверяем какой тип
        TMP$I = position(':k_id', :TMP$HEADER);
        TMP$J = position(':ma_id', :TMP$HEADER);
        if ((TMP$I > 0) and ((TMP$J = 0) or (TMP$J > TMP$I))) then
          TMP$KIND_INT = 1;
        else
          TMP$KIND_INT = 0;
      end


      TMP$BODY = replace(:TMP$BODY, 'kf_client_fio',        'res_info_text');
      TMP$BODY = replace(:TMP$BODY, 'kf_error_text',        'res_error_text');
      TMP$BODY = replace(:TMP$BODY, 'kf_error_is_critical', 'res_error_is_critical');

      TMP$I = 0;

      if (:TMP$DB_ID = 81001) then -- База РФ, ИД-исключения
      begin
        if (:TMP$ID = 19) then
          TMP$I = position('hp_check_loan_kond_ru_indiv', :TMP$BODY);
        else if (:TMP$ID = 129) then
          TMP$I = position('hp_check_loan_docx', :TMP$BODY);
        else if (:TMP$ID = 154) then
          TMP$I = position('hp_check_knclient_pdn', :TMP$BODY);
        else if (:TMP$ID = 155) then
          TMP$I = position('hp_edoc_sign_check_kont', :TMP$BODY);
        else if (:TMP$ID = 156) then
          TMP$I = position('hp_edoc_sign_check_miniapp', :TMP$BODY);
        else if (:TMP$ID = 165) then
          TMP$I = position('hp_anketa_check_miniapp', :TMP$BODY);
        else if (:TMP$ID = 166) then
          TMP$I = position('hp_anketa_check_kont', :TMP$BODY);
        else if (:TMP$ID = 185) then
          TMP$I = position('hp_bankrupt_check', :TMP$BODY);
        else if (:TMP$ID = 186) then
          TMP$I = position('hp_bankrupt_check', :TMP$BODY);
        else if (:TMP$ID = 149) then
          TMP$BODY = replace(:TMP$BODY, '@molbulak.com',        '_molbulak.com');
        else if (:TMP$ID = 28) then
          TMP$BODY = replace(:TMP$BODY, 'if (ma_id = 994559,1100997) then exit;', 'if (ma_id in (994559,1100997)) then exit;');
        else if (:TMP$ID = 48) then
          TMP$VAR =  'declare res_error_text_bl d_str_255;'  || :TMP$ENTER || :TMP$VAR;
      end
      else if (:TMP$db_id in (39801, 41701)) then -- База КГ, ИД-исключения
      begin
        if (:TMP$ID in (112,119)) then
          TMP$I = position('hp_edoc_sign_check_kont', :TMP$BODY);
        else if (:TMP$ID = 134) then
          TMP$I = position('check_match_card_name', :TMP$BODY);
        else if (:TMP$ID = 135) then
          TMP$I = position('cdm_kg_135', :TMP$BODY);
        else if (:TMP$ID = 136) then
        begin
          TMP$BODY = replace(:TMP$BODY, 'from cdm_kg_136(:k_id);', 'from cdm_kg_136(:k_id)');
          TMP$I = position('cdm_kg_136', :TMP$BODY);
        end
        else if (:TMP$ID = 47) then
          TMP$BODY = replace(
              :TMP$BODY
            , 'if(k_id = 109010061182700201,109010039739800501,109010094626200101) then'
            , 'if(k_id in (109010061182700201,109010039739800501,109010094626200101)) then');
      end
      else if (:TMP$DB_ID = 94401) then -- База АЗ, ИД-исключения
      begin
        TMP$I = :TMP$I;
      end

      -- Обратная замена именований если ИД подали в исключения и найден предмет обработки
      if (:TMP$I > 0) then
      begin
        TMP$SQL  = left(:TMP$BODY, :TMP$I);
        TMP$BODY = right(:TMP$BODY, char_length(:TMP$BODY) - char_length(:TMP$SQL));
        TMP$SQL  = replace(:TMP$SQL, 'res_info_text',         'kf_client_fio');
        TMP$SQL  = replace(:TMP$SQL, 'res_error_text',        'kf_error_text');
        TMP$SQL  = replace(:TMP$SQL, 'res_error_is_critical', 'kf_error_is_critical');
        TMP$BODY = :TMP$SQL || :TMP$BODY;
      end

      TMP$I  = position('kf_client_pass_no', :TMP$BODY);
      if (:TMP$I > 0) then
        TMP$VAR =  'declare kf_client_pass_no varchar(30);' || :TMP$ENTER || :TMP$VAR;
    
      TMP$I = position('kf_disb_date', :TMP$BODY);
      if (:TMP$I > 0) then
        TMP$VAR =  'declare kf_disb_date d_date;' || :TMP$ENTER || :TMP$VAR;
    
      TMP$I = position('kc_amount', :TMP$BODY);
      if (:TMP$I > 0) then
        TMP$VAR =  'declare kc_amount curr_type;' || :TMP$ENTER || :TMP$VAR;

      if (TMP$KIND_INT = 1) then
        TMP$KIND_STR = 'kontract';
      else
        TMP$KIND_STR = 'miniapp';

      if (:TMP$KIND_INT  = 1) then  --k_id, kontract
        TMP$HEADER = 'execute block ('                       || ascii_char(13) || ascii_char(10) ||
                  '    k_id d_id = :k_id,'                   || ascii_char(13) || ascii_char(10) ||
                  '    check_kind d_str_code = :check_kind)' || ascii_char(13) || ascii_char(10) ||
                  '  returns ('                              || ascii_char(13) || ascii_char(10) ||
                  '    res_info_text d_str_100,'             || ascii_char(13) || ascii_char(10) ||
                  '    res_error_text d_str_255,'            || ascii_char(13) || ascii_char(10) ||
                  '    res_error_is_critical d_integer)'     || ascii_char(13) || ascii_char(10) ||
                  'as'                                       || ascii_char(13) || ascii_char(10);

      else                          --ma_id, miniapp
        TMP$HEADER = 'execute block ('                       || ascii_char(13) || ascii_char(10) ||
                  '    ma_id d_id = :ma_id,'                 || ascii_char(13) || ascii_char(10) ||
                  '    check_kind d_str_code = :check_kind)' || ascii_char(13) || ascii_char(10) ||
                  '  returns ('                              || ascii_char(13) || ascii_char(10) ||
                  '    res_info_text d_str_100,'             || ascii_char(13) || ascii_char(10) ||
                  '    res_error_text d_str_255,'            || ascii_char(13) || ascii_char(10) ||
                  '    res_error_is_critical d_integer)'     || ascii_char(13) || ascii_char(10) ||
                  'as'                                       || ascii_char(13) || ascii_char(10);

      TMP$SQL = :TMP$HEADER || :TMP$VAR || :TMP$ENTER || 'begin' || :TMP$ENTER || :TMP$BODY;
      TMP$SQL = replace(:TMP$SQL, :TMP$ENTER, ascii_char(13) || ascii_char(10));

      begin
        TMP$J = 1;
        if (:TMP$KIND_INT = 1) then --k_id, kontract
        begin
          for
            select
              q.id
            from
              (
                       select cast(IIF(:TMP$DB_ID = 81001, 101010028537300701, IIF(:TMP$db_id in (39801, 41701), 109010040197400701, 101010008744900301)) as d_id) as id from RDB$DATABASE
                union  select cast(IIF(:TMP$DB_ID = 81001, 101010071495500101, IIF(:TMP$db_id in (39801, 41701), 109010138854100101, 101010009548600301)) as d_id) as id from RDB$DATABASE
                union  select cast(IIF(:TMP$DB_ID = 81001, 101010071499500101, IIF(:TMP$db_id in (39801, 41701), 109010138856500101, 101010013240100101)) as d_id) as id from RDB$DATABASE
              ) q
            into
              :TMP$I
          do
            for
              execute statement
                (:TMP$SQL)
                (k_id := :TMP$I, check_kind := :TMP$KIND_STR)
              into
                  :tmp$res_info_text
                , :tmp$res_error_text
                , :tmp$res_error_is_critical
              do
                TMP$J = 1;
        end
        else
        begin                       --ma_id, miniapp
          for
            select
              q.id
            from
              (
                       select cast(IIF(:TMP$DB_ID = 81001, 1182652, IIF(:TMP$db_id in (39801, 41701), 2309835, 169372)) as d_id) as id from RDB$DATABASE
                union  select cast(IIF(:TMP$DB_ID = 81001, 1183061, IIF(:TMP$db_id in (39801, 41701), 2309825, 169375)) as d_id) as id from RDB$DATABASE
                union  select cast(IIF(:TMP$DB_ID = 81001, 1183147, IIF(:TMP$db_id in (39801, 41701), 2309856, 169131)) as d_id) as id from RDB$DATABASE
              ) q
            into
              :TMP$I
          do
            for
              execute statement
                (:TMP$SQL)
                (ma_id := :TMP$I, check_kind := :TMP$KIND_STR)
              into
                  :tmp$res_info_text
                , :tmp$res_error_text
                , :tmp$res_error_is_critical
              do
                TMP$J = 1;
        end
      when any do
        TMP$J = -1;
      end

      update
          ConditionMain as cm
        SET
            cm.CDM_OBJECT_KIND   = :TMP$KIND_STR
          , cm.CDM_SQL_TMP       = :TMP$SQL
          , cm.CDM_SQL_TMP_CHECK = :TMP$J
          , cm.CDM_USE_STORED_PROC = IIF(:TMP$J = 1, 1, 0)
        WHERE
          (cm.cdm_id = :tmp$id);
  end
end
