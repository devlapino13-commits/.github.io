CREATE PROCEDURE AAREP_FOR_TSB_SECTION_1_2(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    VID_OP VARCHAR(100),
    REG_ADDRESS_FACT_STR VARCHAR(255),
    REG_PHONE VARCHAR(50),
    REG_EMAIL VARCHAR(50),
    UKAZ_MFC VARCHAR(255),
    DISB_SUM CURR_TYPE,
    REG_DATE_OPEN DATE,
    REG_DATE_CLOSE DATE)
AS
declare variable reg_id integer;
declare variable reg_address_fact id_type;
begin
  for select reg_code || ' ' || reg_name, -- Наименование обособленного подразделения
             'Иное обособленное подразделение' vid_op, -- Вид обособленного подразделения
             reg_address_fact, -- Адрес фактического нахождения
             reg_phone, -- Номер телефона
             'office@molbulak.ru' reg_email, -- Адрес электронной почты
             'Осуществляет' ukaz_mfk, -- Указание на осуществление микрофинансовой деятельности
             reg_unicode, reg_date_open, reg_date_close
      from region
      where coalesce(reg_date_open, '01.01.1990') <= :rep_date_end and
            coalesce(reg_date_close, '31.12.9999') >= :rep_date_begin and
            is_delete = 0
      into reg_name, vid_op, reg_address_fact, reg_phone, reg_email, ukaz_mfc, reg_id, reg_date_open, reg_date_close
  do
  begin
    disb_sum = null;
    select sum(k_amount_approved)
    from kontstate, kontract
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_region_id = :reg_id and
          ks_from_k_id = k_id
    into disb_sum;

    disb_sum = disb_sum / 1000;

    select adl_address_str
    from addrlist
    where adl_id = :reg_address_fact
    into reg_address_fact_str;

    suspend;
  end
end
