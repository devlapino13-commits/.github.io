CREATE PROCEDURE FSROBOT_MAILING_KOMPLAINS(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable c_name_fio varchar(100);
declare variable reg_name varchar(100);
declare variable jl_name varchar(100);
declare variable ps_date date;
declare variable c_login varchar(100);
declare variable c_email varchar(100);
declare variable c_cell_no varchar(100);
declare variable c_id decimal(18,0);
declare variable kol integer;
declare variable hr_order_no varchar(20);
declare variable date_str varchar(255);
declare variable date_format_str varchar(20);
declare variable ol_id id_type;
declare variable ob_number integer;
declare variable ob_summa curr_type;
declare variable ob_summa_val curr_type;
declare variable ob_currency integer;
declare variable ob_ol_date date;
declare variable ob_short_acc_dt varchar(20);
declare variable ob_short_acc_kt varchar(20);
declare variable ob_ol_reg_id integer;
declare variable ob_detail varchar(1024);
declare variable cu_symbol varchar(10);
declare variable ob_account_kt varchar(20);
declare variable ob_account_dt varchar(20);
declare variable ol_document_client id_type;
declare variable ol_memo varchar(1024);
declare variable al_name_kt varchar(200);
declare variable al_name_dt varchar(200);
declare variable cl_name str_100;
declare variable attantion str_100;
begin
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь
  res_c_id = 0;
  ol_id = substring(rep_params from 1 for position(';' in rep_params) - 1);
  rep_params = substring(rep_params from position(';' in rep_params) + 1);
  ob_number = substring(rep_params from 1 for position(';' in rep_params) - 1);
  attantion = substring(rep_params from position(';' in rep_params) + 1);
  attantion = trim(attantion);

  select coalesce(ob.ob_summa, 0), coalesce(ob.ob_summa_val, 0),
         (select cl.cu_symbol
          from currlist cl
          where cl.cu_id = ob.ob_currency),

         ob.ob_ol_date, coalesce(ob.ob_short_acc_dt, ''), coalesce(ob.ob_short_acc_kt, ''),

         (select reg.reg_name
          from region reg
          where reg.reg_unicode = ob.ob_ol_reg_id), coalesce(ob.ob_detail, ''), ob_account_kt,

         ob_account_dt, coalesce(ol_memo, ''), ol_document_client
  from operbook ob
  inner join operlist ol on ob.ob_from_ol_id = ol.ol_id
  where ob.ob_ol_is_active = 1 and
        ob.ob_from_ol_id = :ol_id and
        ob.ob_number = :ob_number
  into :ob_summa, :ob_summa_val, :cu_symbol,

       :ob_ol_date, :ob_short_acc_dt, :ob_short_acc_kt,

       :reg_name, :ob_detail, :ob_account_kt,

       :ob_account_dt, :ol_memo, :ol_document_client;

  date_format_str = right('0' || extract(day from ob_ol_date), 2) || '.' || right('0' || extract(month from ob_ol_date), 2) || '.' || extract(year from ob_ol_date);

--  наименование кредит
  select al_name
  from acc_list
  where al_id = :ob_account_kt
  into al_name_kt;

--  наименование дебит
  select al_name
  from acc_list
  where al_id = :ob_account_dt
  into al_name_dt;

  cl_name = null;
  select c_name_fio
  from clients
  where c_id = :ol_document_client
  into cl_name;
  cl_name = coalesce(cl_name, '');

  res_maillist = :rep_email;

  res_subject = attantion || ' Сформирована проводка на сумму 1 млн. сом и выше';

  res_body = '<b>Сформирована проводка на сумму 1 млн. сом и выше</b><br>';
  res_body = BLOB_APPEND(res_body, '<br>');
  res_body = BLOB_APPEND(res_body, 'Дата операции.........' || '<span style = color:red> <b>' || date_format_str || '</span> <br> </b>');
  res_body = BLOB_APPEND(res_body, 'Сумма (в нац. валюте).' || cast(ob_summa as varchar(50)) || '<br>');
  res_body = BLOB_APPEND(res_body, 'Сумма (в валюте)......' || cast(ob_summa_val as varchar(50)) || '<br>');
  res_body = BLOB_APPEND(res_body, 'Валюта................' || cu_symbol || '<br>');
  res_body = BLOB_APPEND(res_body, 'Наименование Кт.......' || al_name_kt || '<br>');
  res_body = BLOB_APPEND(res_body, 'Наименование Дт.......' || al_name_dt || '<br>');
  res_body = BLOB_APPEND(res_body, 'Счет Дт...............' || ob_short_acc_dt || '<br>');
  res_body = BLOB_APPEND(res_body, 'Счет Кт...............' || ob_short_acc_kt || '<br>');
  res_body = BLOB_APPEND(res_body, 'Офис..................' || reg_name || '<br>');
  res_body = BLOB_APPEND(res_body, '<span style = color:red> <b>' || 'Клиент................' || cl_name || '</span> </b> <br>');
  res_body = BLOB_APPEND(res_body, '<br>');

  if (ob_detail <> '') then
  begin
    res_body = BLOB_APPEND(res_body, 'Детали проводки:<br>');
    res_body = BLOB_APPEND(res_body, ob_detail || '<br>');
  end

  if (ol_memo <> '') then
  begin
    res_body = BLOB_APPEND(res_body, 'Примечание:<br>');
    res_body = BLOB_APPEND(res_body, '<span style = color:red> <b>' || ol_memo || '</span> </b> <br>');
  end
  suspend;
end
