CREATE PROCEDURE CRM_SVZ_LIST
RETURNS (
    OFFICE_ID D_INTEGER,
    C_ID D_ID,
    FIO D_STR_100,
    CELL_NO D_STR_100)
AS
begin
  for select ps_from_reg_id,
             ps_from_c_id,
             cl.c_name_fio
          from hr_pstatus hps
          inner join joblist jl on jl.jl_id = ps_from_jl_id
          inner join jobkind jk on jk.jk_id = jl.jl_kind
          inner join clients cl on cl.c_id = ps_from_c_id
          inner join region_list(1000000, 1, -1, 1) on id_list = ps_from_reg_id
          where current_date between ps_date_begin and ps_date_end and
                ps_is_working = 1 and
                jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.Region_KM')
          order by 1
          into :office_id,
               :c_id,
               :fio
  do
  begin
    cell_no = '';
    select cont_value
    from hp_contdata_get_cell_main(0, :c_id)
    where cont_value is not null
    into cell_no;

    suspend;
  end
end
