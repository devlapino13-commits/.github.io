CREATE PROCEDURE HP_HTML_GET_LOAN_STATUS_KONTMAT(
    K_ID D_ID)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable ml_name d_str_255;
declare variable ml_model d_str_255;
declare variable kml_guarantee_month curr_type;
declare variable kml_count curr_type;
declare variable kml_cost curr_type;
declare variable kml_total curr_type;
declare variable ag_caption d_str_255;
begin
  res_text = '';

  for select ml_name, ml_model, kml_guarantee_month, kml_count, kml_cost, kml_total, ag_caption
      from hp_kontmat_view(:k_id)
      into :ml_name, :ml_model, :kml_guarantee_month, :kml_count, :kml_cost, :kml_total, :ag_caption
  do
  begin
    res_text = f_format('<h3>%1</h3><table width="100%"><col width="40%"><col width="60%">', '');
    res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Наименование товара</b>', ml_name, '', '', -1, -1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Модель товара</b>', ml_model, '', '', -1, -1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Гарантия в мес.</b>', round(kml_guarantee_month, 0), '', '', -1, -1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Кол-во</b>', round(kml_count, 2), '', '', -1, -1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Стоимость покупки</b>', round(kml_cost, 2), '', '', -1, -1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Сумма покупки</b>', round(kml_total, 2), '', '', -1, -1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row('<b>Продавец</b>', ag_caption, '', '', -1, -1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, '</table>');
    suspend;
  end

  if (coalesce(ml_name, '') = '') then
  begin
    res_text = '';
    suspend;
  end
end
