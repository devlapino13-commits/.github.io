CREATE PROCEDURE HP_SEND_PUSH_TO_CLIENT(
    C_ID D_ID,
    MSG_SUBJECT D_STR_255,
    MSG_BODY D_BLOB_TEXT,
    LIFE_TIME D_INTEGER,
    PROJECT D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable mcl_id d_id;
declare variable push_end_date d_date;
begin
    :res_text = '';
    :push_end_date = null;

    if (coalesce(:c_id, 0) = 0) then
        :res_text = 'Не указан клиент для отправки push - уведомления.';

    if (:res_text = '' and coalesce(:msg_subject, '') = '') then    
        :res_text = 'Не заполнен заголовок уведомления.';

    if (:res_text = '' and coalesce(:msg_body, '') = '') then
        :res_text = 'Не заполнен текст уведомления.';

    if (:res_text = '' and coalesce(:life_time, 0) <= 0) then
        :res_text = 'Число дней хранения сообщения должно быть больше нуля.';

    if (:res_text = '' and coalesce(:project, 0) = 0) then
        :res_text = 'Не выбрано клиентское мобильное приложение.';

    if (:res_text = '') then
    begin
        for select ml.mcl_id
        from mc_login ml
        where ml.mcl_is_delete = 0
          and ml.mcl_from_c_id = :c_id
          and ml.mcl_project = :project
          and ml.mcl_datetime = (select max(ml2.mcl_datetime)
                                from mc_login ml2
                                where ml2.mcl_from_c_id = ml.mcl_from_c_id
                                  and ml2.mcl_is_delete = 0
                                  and ml2.mcl_project = :project)
        into :mcl_id
        do
        begin
            execute procedure mc_send_push('mc_login', :mcl_id, :msg_subject, :msg_body, :life_time);

            push_end_date = current_date + :life_time;

            insert into push_log (pl_create_datetime, pl_title, pl_content, pl_date_begin, pl_date_end, pl_project,
                                  pl_from_un_id, pl_from_c_id)
            values (current_timestamp, :msg_subject, :msg_body, current_date, :push_end_date, :project, :un_id,
                    :c_id);
        end

        if (coalesce(:mcl_id, 0) = 0) then
            :res_text = 'У клиента не установлен КМП.';
    end 

    suspend;
end
