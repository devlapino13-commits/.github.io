CREATE PROCEDURE HP_EXD_ANALIZ_DETAIL_VIEW(
    IFA_ID D_INTEGER)
RETURNS (
    IED_ID D_INTEGER,
    IED_REC_NO D_INTEGER,
    IED_TRANS_DATETIME D_DATETIME,
    IED_IPC_TRANS_PROCESSED D_DATE,
    IED_BANK_CODE D_STR_LANG,
    IED_FILIAL_CODE D_STR_LANG,
    IED_FROM_CARD_ID D_INTEGER,
    IED_FROM_CCA_ID D_INTEGER,
    IED_PAN D_STR_20,
    IED_TRANS_CURRENCY D_STR_LANG,
    IED_TRANS_AMOUNT D_CURRENCY,
    IED_OPER_TYPE D_STR_SYM,
    IED_FROM_ITT_ID D_INTEGER,
    ITT_NAME D_STR_255,
    IED_TERMINAL_TYPE D_STR_SYM,
    IED_TRADE_POINT_CODE D_STR_20,
    IED_TRADE_POINT_NAME D_STR_CODE,
    IED_TRADE_POINT_MCC D_STR_10,
    IED_CITY D_STR_20,
    IED_EMITENT_COMISSION D_CURRENCY,
    IED_ACQUIRER_CMI D_STR_20,
    IED_ACQUIRER_BANK_ID D_INTEGER,
    BANK_NAME D_STR_255,
    IED_ACQUIRER_COMMISSION D_CURRENCY,
    IS_ERROR D_BOOL,
    IED_ERROR_ID D_INTEGER,
    ERROR_NAME D_STR_255,
    IED_FROM_OL_ID D_ID)
AS
begin
  for
    select
      ied_id,
      ied_rec_no,
      ied_trans_datetime,
      ied_ipc_trans_processed,
      ied_bank_code,
      ied_filial_code,
      ied_from_card_id,
      ied_from_cca_id,
      ied_pan,
      ied_trans_currency,
      ied_trans_amount,
      ied_oper_type,
      ied_from_itt_id,
      itt_name_en,
      ied_terminal_type,
      ied_trade_point_code,
      ied_trade_point_name,
      ied_trade_point_mcc,
      ied_city,
      ied_emitent_comission,
      ied_acquirer_cmi,
      ied_acquirer_bank_id,
      ib_ipc_bank_name,
      ied_acquirer_commission,
      iif(ied_error_id > 0, 1, 0),
      ied_error_id,
      rd_name,
      ied_from_ol_id
    from ipc_exd_detail
      left join ipc_banks
        on ib_id = ied_acquirer_bank_id
      left join ipc_transaction_types
        on itt_id = ied_from_itt_id
      left join refdata
        on rd_id = ied_error_id
    where ied_from_ifa_id = :ifa_id
  into
    :ied_id,
    :ied_rec_no,
    :ied_trans_datetime,
    :ied_ipc_trans_processed,
    :ied_bank_code,
    :ied_filial_code,
    :ied_from_card_id,
    :ied_from_cca_id,
    :ied_pan,
    :ied_trans_currency,
    :ied_trans_amount,
    :ied_oper_type,
    :ied_from_itt_id,
    :itt_name,
    :ied_terminal_type,
    :ied_trade_point_code,
    :ied_trade_point_name,
    :ied_trade_point_mcc,
    :ied_city,
    :ied_emitent_comission,
    :ied_acquirer_cmi,
    :ied_acquirer_bank_id,
    :bank_name,
    :ied_acquirer_commission,
    :is_error,
    :ied_error_id,
    :error_name,
    :ied_from_ol_id
  do
  begin
    suspend;
  end
end
