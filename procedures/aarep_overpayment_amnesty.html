CREATE PROCEDURE AAREP_OVERPAYMENT_AMNESTY(
    REP_DATE DATE)
RETURNS (
    K_KONTRACT_NO STR_20,
    CURR_DEBT CURR_TYPE,
    LIST_KONTRACT_NO STR_255,
    SUM_OVERPAY CURR_TYPE)
AS
declare variable debt_operkroff curr_type;
declare variable kd_k_orig id_type;
declare variable kd_k_new id_type;
declare variable can_suspend smallint;
declare variable k_id decimal(18,0);
declare variable sum_overpay_temp curr_type;
begin
  for
  select kd_k_orig from kontdevides
  where kd_date <= :rep_date
  union
  select kd_k_new from kontdevides
  where kd_date <= :rep_date
  into k_id
  do
  begin
--  проверка договор активный или имеется задолженность на спец учете.
    debt_operkroff = null;
    select sum(oko_os + oko_perc + oko_sh_os + oko_sh_perc)
    from operkroff
    where oko_from_k_id = :k_id
    and oko_ol_date <= :rep_date
    and oko_ol_is_active = 1
    into debt_operkroff;
    debt_operkroff = coalesce(debt_operkroff,0);

    can_suspend = 0;
    sum_overpay = 0;
    if (exists(select * from kontstate
    where ks_ol_is_delete = 0
    and ks_state = 1
    and ks_from_k_id = :k_id
    and :rep_date between ks_date_begin and ks_date_end)
    or
    debt_operkroff > 0) then
    begin
--    Ищем связанные контракты по которым имеется переплата
--    Находим основной контракт
      kd_k_orig = null;
      select kd_k_orig
      from kontdevides
      where kd_k_new = :k_id
      into kd_k_orig;
      kd_k_orig = coalesce(kd_k_orig, k_id);

--    далее каждый контракт проверяем на переплату
      list_kontract_no = '';
      for
      select kd_k_orig from kontdevides
      where kd_k_orig = :kd_k_orig
      union
      select kd_k_new from kontdevides
      where kd_k_orig = :kd_k_orig
      into kd_k_new
      do
      begin
--      если закрытый то только тогда проверяем на переплату
        if (exists(select * from kontstate
        where ks_ol_is_delete = 0
        and ks_from_k_id = :kd_k_new
        and ks_state = 2
        and :rep_date between ks_date_begin and ks_date_end)) then
        begin
          sum_overpay_temp = null;
          select -sum(ok_os) from operkred
          where ok_ol_is_active = 1
          and ok_from_k_id = :kd_k_new
          and ok_type = 9
          into sum_overpay_temp;
          sum_overpay_temp = coalesce(sum_overpay_temp, 0);

          if (sum_overpay_temp > 0) then
          begin
            sum_overpay = sum_overpay + sum_overpay_temp;
            select k_kontract_no from kontract
            where k_id = :kd_k_new
            into k_kontract_no;

            if (list_kontract_no = '') then list_kontract_no = k_kontract_no;
            else list_kontract_no = list_kontract_no||'; '||k_kontract_no;
          end
        end
      end

    end

    if (sum_overpay > 0) then
    begin
      select k_kontract_no from kontract
      where k_id = :k_id
      into k_kontract_no;

      curr_debt = null;
      select sum(zad_os + sh_calc_perc)
      from aarep_make_zad_list(:k_id, :rep_date, 1)
      into curr_debt;
      curr_debt = coalesce(curr_debt,0);

      suspend;
    end

  end
end
