CREATE PROCEDURE AAREP_REPORT_KP_PAR(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    PARENT_NAME STR_255,
    DISB_SUM DECIMAL(18,4),
    KOL_DISB INTEGER,
    SVZ_KOL_ACT INTEGER,
    WORK_DAYS_QTY INTEGER,
    DISB_PLAN DECIMAL(18,4),
    DIFFERENCE DECIMAL(18,4),
    COUNT_DISB_AVG DECIMAL(4,2),
    WORK_DAYS_MONTH INTEGER)
AS
declare variable date_end d_date;
declare variable first_date_month d_date;
begin
  for select apr.parent_name, apr.reg_name, sum(apr.disb_sum), sum(apr.kol_disb), sum(apr.svz_kol_act),
             apr.work_days_qty
      from aarep_report_apr(:rep_region_id, :rep_date_end) apr
      where apr.reg_name <> '0101 Bakı'
      group by parent_name, reg_name, work_days_qty
      into parent_name, reg_name, disb_sum, kol_disb, svz_kol_act, work_days_qty
  do
  begin
    first_date_month = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date);
    date_end = f_date_month_end(:rep_date_end); -- Конец месяца

    --считаем количество рабочих дней в месяце
    work_days_month = null;
    select count(scd_work_hours) work_days_qty
    from sa_calendays sc
    where sc.scd_date between :first_date_month and :date_end and
          scd_work_hours > 0
    into :work_days_month;

    disb_plan = (1.5 * work_days_month * svz_kol_act);
    difference = (kol_disb - disb_plan);
    -- Средняя выдача на 1 КС считается как = количество выдач/количество КС/количество рабочих дней, рабочих дней 
    if (svz_kol_act > 0 and
        work_days_qty <> 0) then
      count_disb_avg = round(cast(kol_disb as d_currency) / cast(svz_kol_act as d_currency) / cast(work_days_qty as d_currency), 2);
    else
      count_disb_avg = 0.0;
    suspend;
  end
end
