CREATE PROCEDURE LAW_EXECUTORY_PROCESS(
    LAM_ID INTEGER,
    REP_DATE DATE,
    SHOW_FULL_DATA SMALLINT)
RETURNS (
    LAK_FROM_LAM_ID INTEGER,
    LAK_FROM_K_ID ID_TYPE,
    LAK_FROM_C_ID ID_TYPE,
    LAK_COST_ALL CURR_TYPE,
    LAK_COST_OS CURR_TYPE,
    LAK_COST_PERC CURR_TYPE,
    LAK_COST_FINE CURR_TYPE,
    LAK_COST_OTHER CURR_TYPE,
    LAK_FROM_UN_ID ID_TYPE,
    LAK_DATETIME TIMESTAMP,
    LAK_EL_REC_DATE DATE,
    LAK_EB STR_255,
    LAK_EL_FILING_DATE DATE,
    LAK_ENF_PROC_DATE DATE,
    LAK_ENF_NUMBER STR_255,
    LAK_MOVEMENT_DATE DATE,
    LAK_PROC_STOP_DATE DATE,
    LAK_BAILIFFS_CONTACTS STR_255,
    LAK_MEMO STR_255,
    LAK_EL_RETURN_DATE DATE,
    LAK_ZADOL_ALL CURR_TYPE,
    LAK_DELINQ_DAY INTEGER,
    KONTRACT_NO VARCHAR(20),
    CLIENT_FIO STR_100,
    C_CITIZEN STR_100,
    KF_NAME STR_100,
    CL_ZAD_TOTAL CURR_TYPE,
    C_ADDRESS_YUR STR_255,
    C_ADDRESS_FACT STR_255,
    C_ADDRESS_RODINA_YUR STR_255,
    C_ADDRESS_RODINA_FACT STR_255,
    C_PASS_NO VARCHAR(30),
    C_PASS_WHO STR_255,
    C_PASS_DATE DATE,
    IN_BLACK_LIST SMALLINT,
    C_BIRTH_DATE DATE,
    OWN_PAYED_AMOUNT CURR_TYPE,
    MAX_PAYED_DATE DATE,
    LAK_IL_ZADOL CURR_TYPE,
    LAK_NATIONAL_DUTY CURR_TYPE,
    LAK_FILING_TYPE STR_100,
    LAK_REST_MOV_FROM DATE,
    LAK_REST_MOV_IN DATE,
    LAK_FMC STR_255,
    LAK_SEARCH STR_255,
    LAK_HL_PENALTY STR_255,
    LAK_STOP_REASON STR_255,
    LAK_FROM_PL_ID ID_TYPE,
    LAK_FROM_PL_NAME STR_255,
    LAK_RESOLUTION_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable k_id id_type;
declare variable k_amount_approved curr_type;
declare variable kc_amount_approved curr_type;
declare variable zad_total curr_type;
declare variable c_id id_type;
declare variable adc_address_type integer;
declare variable adc_address_str str_255;
declare variable lam_date_begin date;
declare variable kc_closed_ol_id id_type;
declare variable goup_payed_amount curr_type;
declare variable own_payed_amount_tmp curr_type;
declare variable max_payed_date_tmp date;
declare variable cl_share decimal(18,7);
declare variable par_c_inn varchar(30);
declare variable par_c_cell_no varchar(100);
begin
  show_full_data = coalesce(show_full_data, 0);

  select lm.lam_date_begin
  from lawmain lm
  where lm.lam_id = :lam_id
  into :lam_date_begin;

  for select lak_from_lam_id, lak_from_k_id, lak_from_c_id,

             lak_cost_all, lak_cost_os, lak_cost_perc,

             lak_cost_fine, lak_cost_other, lak_from_un_id,

             lak_datetime, lak_el_rec_date, lak_eb,

             lak_el_filing_date, lak_enf_proc_date, -- Дата Постановления о возбуждении ИП
             lak_enf_number, -- Номер ИП

             lak_movement_date, lak_proc_stop_date, -- Окончание ИП
             lak_bailiffs_contacts, -- Пристав
-- Примечение
             lak_memo, lak_el_return_date, lak_zadol_all,

             lak_delinq_day, lak_il_zadol, -- Сумма задолженности по ИЛ
             lak_national_duty, -- Госпошлина

             lak_filing_type, -- Способ предъявления
             lak_rest_mov_from, -- Дата ограничения выезда из РФ
             lak_rest_mov_in, -- Дата ограничения въезда в РФ

             lak_fmc, -- Управление ФМС
             lak_search, -- Розыск
             lak_hl_penalty, -- Взыскание на Родине

             lak_stop_reason, -- Основание окончания,
             lak_from_pl_id, -- Регион (Регион места нахождения отдела судебных приставов)

             lk.lak_resolution_text,

             kont.k_kontract_no, cl.c_name_fio,

             rd_name, kal.kal_new_kf_name, kont.k_id,

             k_amount_approved, kncl.kc_amount_approved, cl.c_id,

             coalesce(cl.c_pass_no, ''), cl.c_pass_who, cl.c_pass_date,

             cl.c_birth_date, iif(exists(select *
                                         from operlist ol
                                         where ol.ol_id = kncl.kc_closed_ol_id and
                                               ol.is_delete = 0), kc_closed_ol_id, null), coalesce(cl.c_inn, ''),
             (select cont_value
              from hp_contdata_get_cell_main(0, kncl.kc_from_c_id))
      from refdata, lawkredit lk, kontract kont, clients cl, knaktlist kal, knclient kncl
      where lk.lak_from_lam_id = :lam_id and
            cl.c_citizen = rd_id and
            lk.lak_from_k_id = kont.k_id and
            cl.c_id = lk.lak_from_c_id and
            k_from_kf_id = kal_from_kf_id and
            kal.is_delete = 0 and
            kncl.kc_from_c_id = lk.lak_from_c_id and
            kncl.kc_from_k_id = lk.lak_from_k_id and
            current_date between kal_date and kal_date_end
      into :lak_from_lam_id, :lak_from_k_id, :lak_from_c_id, :lak_cost_all, :lak_cost_os, :lak_cost_perc,
           :lak_cost_fine, :lak_cost_other, :lak_from_un_id, :lak_datetime, :lak_el_rec_date, :lak_eb,
           :lak_el_filing_date, :lak_enf_proc_date, :lak_enf_number, :lak_movement_date, :lak_proc_stop_date,
           :lak_bailiffs_contacts, :lak_memo, :lak_el_return_date, :lak_zadol_all, :lak_delinq_day, :lak_il_zadol,
           :lak_national_duty, :lak_filing_type, :lak_rest_mov_from, :lak_rest_mov_in, :lak_fmc, :lak_search,
           :lak_hl_penalty, :lak_stop_reason, :lak_from_pl_id, :lak_resolution_text,

           :kontract_no, :client_fio, :c_citizen, :kf_name, :k_id, :k_amount_approved, :kc_amount_approved, :c_id,
           :c_pass_no, :c_pass_who, :c_pass_date, :c_birth_date, :kc_closed_ol_id, :par_c_inn, :par_c_cell_no

  do
  begin
    lak_from_pl_name = null;

    if (lak_from_pl_id is not null) then
    begin
      select pl.pl_name
      from places pl
      where pl.pl_id = :lak_from_pl_id and
            pl.is_delete = 0
      into :lak_from_pl_name;
    end

    lak_from_pl_name = coalesce(lak_from_pl_name, '');
    if (lak_from_pl_name = '') then
      lak_from_pl_name = 'Не указано';

    if (show_full_data = 1) then
    begin
      cl_share = kc_amount_approved * 100 / k_amount_approved;

      cl_zad_total = 0;
      if (kc_closed_ol_id is null) then -- если клиент не оплатил совю долю, тогда
      begin

        zad_total = null;

        select sum(zad_os + sh_calc_perc + sh_calc_fine)
        from aarep_make_zad_list(:k_id, :rep_date, 1)
        where sh_date is not null
        into :zad_total;

        zad_total = coalesce(zad_total, 0);
        cl_zad_total = cast((zad_total * cl_share / 100) as decimal(18,2)); -- Текущая задолженность клиента (ОС+%%+пеня)
      end

      c_address_yur = '';
      c_address_fact = '';
      c_address_rodina_yur = '';
      c_address_rodina_fact = '';

      for select adc_address_type, adc_address_str
          from hp_addrclient(:c_id)
          where adc_address_type in (1, 2, 4, 5)
          into adc_address_type, adc_address_str
      do
      begin
        if (adc_address_type = 1) then
          c_address_yur = adc_address_str;
        if (adc_address_type = 2) then
          c_address_fact = adc_address_str;
        if (adc_address_type = 4) then
          c_address_rodina_yur = adc_address_str;
        if (adc_address_type = 5) then
          c_address_rodina_fact = adc_address_str;
      end

      in_black_list = 0;

      if (exists(select c_comment || ', ' || bl_comment
                 from hp_blacklist_search(:c_id, :client_fio, :c_birth_date, :c_pass_no, :par_c_inn, :par_c_cell_no, 1, 0, 0)
                 where c_comment_id = 1 rows 1)) then
        in_black_list = 1;

--  Сумма выплат за период с даты передачи в суд ------------------------ >>
      own_payed_amount = 0;
      goup_payed_amount = null;
      own_payed_amount_tmp = null;
      max_payed_date = null;
      max_payed_date_tmp = null;

      select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine), max(ok.ok_ol_date)
      from operkred ok
      where ok.ok_type = 4 and
            ok.ok_ol_is_active = 1 and
            ok.ok_from_k_id = :k_id and
            ok.ok_ol_date >= :lam_date_begin and
            ok.ok_from_ol_id <> coalesce(:kc_closed_ol_id, 0)
-- and ((ok.ok_os) <> 0  or (ok.ok_perc <>0) or (ok.ok_fine<>0))
            and
            ok.ok_from_c_id = 0
      into :goup_payed_amount, max_payed_date;

      goup_payed_amount = coalesce(goup_payed_amount, 0);

      if (kc_closed_ol_id is not null) then
      begin
        select -sum(ok.ok_os + ok.ok_perc + ok.ok_fine), max(ok.ok_ol_date)
        from operkred ok
        where ok.ok_type = 4 and
              ok.ok_ol_is_active = 1 and
              ok.ok_from_k_id = :k_id and
              ok.ok_ol_date >= :lam_date_begin and
              ok.ok_from_ol_id = :kc_closed_ol_id and
              ok.ok_from_c_id = 0
        into :own_payed_amount_tmp, max_payed_date_tmp;
      end

      own_payed_amount_tmp = coalesce(own_payed_amount_tmp, 0);

      own_payed_amount = cast(((goup_payed_amount * cl_share / 100) + own_payed_amount_tmp) as decimal(18,2));
------------------------------------------------------------------------- <<

      if ((max_payed_date is null) and
          (max_payed_date_tmp is not null)) then
        max_payed_date = max_payed_date_tmp;
      if ((max_payed_date is not null) and
          (max_payed_date_tmp is not null)) then
      begin
        if (max_payed_date < max_payed_date_tmp) then
          max_payed_date = max_payed_date_tmp;
      end
    end
    suspend;
  end
end
