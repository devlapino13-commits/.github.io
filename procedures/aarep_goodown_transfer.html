CREATE PROCEDURE AAREP_GOODOWN_TRANSFER(
    REG_ID D_INTEGER,
    DT_BEGIN D_DATE,
    DT_END D_DATE)
RETURNS (
    OPER_DATE D_DATE,
    SELLER_ID D_ID,
    INVENT_STR D_STR_CODE,
    OPER_KIND D_STR_CODE,
    KONTRACT_NO D_STR_20,
    WP_ID D_ID,
    MO_ID D_ID,
    ML_ID D_ID,
    MC_ID D_ID,
    REGION_ID D_INTEGER,
    ACCOUNT D_STR_20,
    QTY D_CURRENCY,
    PRICE D_CURRENCY,
    SUMMA D_CURRENCY,
    SELLER_NAME D_STR_100,
    WP_NAME D_STR_100,
    MO_NAME D_STR_100,
    MC_NAME D_STR_100,
    REG_NAME D_STR_CODE,
    ML_NAME D_STR_255,
    YEDIZ D_STR_20)
AS
declare variable ol_id d_id;
declare variable ol_parent_id d_id;
declare variable oper_type d_integer;
begin
  for
    select
      ol_id,
      ol_parent_id,
      ol_date_oper_day,
      ob_from_own_c_id,
      ob_invent_str,
      wp_from_reg_id,
      om_oper_type,
      om_workplace_id,
      om_matotvet_id,
      ob_from_ml_id,
      ob_from_mc_id,
      om_account,
      iif(om_kolich_plus = 0, -om_kolich_minus, om_kolich_plus),
      iif(om_summa_plus = 0, -om_summa_minus, om_summa_plus)
    from operbill
      inner join opermat
        on om_bill_id = ob_bill_id
          and om_bill_line_no = ob_bill_line_no
      inner join operlist
        on ol_id = om_from_ol_id
          and ol_date_oper_day between :dt_begin and :dt_end
          and is_delete = 0
      inner join workplace
        on wp_id = om_workplace_id
          and exists(select * from region_list(:reg_id, 1, -1, 1) where id_list = wp_from_reg_id)
    where ob_from_own_c_id > 0
      and ob_is_delete = 0
  into
    :ol_id,
    :ol_parent_id,
    :oper_date,
    :seller_id,
    :invent_str,
    :region_id,
    :oper_type,
    :wp_id,
    :mo_id,
    :ml_id,
    :mc_id,
    :account,
    :qty,
    :summa
  do
  begin
    seller_name = (select c_name_fio from clients where c_id = :seller_id);
    wp_name = (select wp_name from workplace where wp_id = :wp_id);
    mo_name = (select c_name_fio from clients where c_id = :mo_id);
    mc_name = (select mc_name from matcateg where mc_id = :mc_id);
    reg_name = (select reg_name from region where reg_unicode = :region_id);
    price = iif(qty = 0, 0, summa / qty);

    select
      ml_name,
      yl_code
    from matlist
      inner join yedizlist
        on yl_id = ml_from_yl_id
    where ml_id = :ml_id
    into
      :ml_name,
      :yediz;

    oper_kind =
      case oper_type
        when  1 then 'Покупка, приобретение'
        when  2 then 'Приход от производства'
        when  3 then 'Перемещение'
        when  4 then 'Внутренние списания из баланса'
        when  5 then 'Продажа, Передача филиалам (Выбытие)'
        when  6 then 'Модернизация, реконструкция'
        when  7 then 'Разделение цельного / Объединение'
        when  8 then 'Комплектация / Декомплектация'
        when  9 then 'Переоценка (доход, убыток)'
        when 10 then 'Переоценка за счет амортизации'
      else '???' end;

     kontract_no = null;
     if (oper_type = 5) then
     begin
       kontract_no = (
         select first 1 k_kontract_no
         from kontmat_own
           inner join kontract
             on k_id = kmo_k_id
         where kmo_ol_id = :ol_parent_id);
       if (kontract_no is not null) then
         oper_kind = 'Финансирование';
     end

    suspend;
  end
end
