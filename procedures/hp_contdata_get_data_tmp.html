CREATE PROCEDURE HP_CONTDATA_GET_DATA_TMP(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    DATA_TYPE VARCHAR(100))
RETURNS (
    CONT_VALUE VARCHAR(255),
    CONT_STATUS INTEGER,
    CD_TYPE VARCHAR(100),
    CD_ORDERNO INTEGER)
AS
declare variable ok_data_main varchar(100);
declare variable ok_data_add varchar(100);
declare variable last_data_main varchar(100);
declare variable last_data_add varchar(100);
declare variable main_founded integer;
declare variable cda_is_main d_smallint;
declare variable cda_status d_smallint;
begin
  -- Data_type:         - тип контактных данных
  -- cell, email и т.д. - из справочника
  -- all                - все виды (только заполненные)
  -- Можно задавать через запятую

  -- Если параметр не указан, не возращаем знаение
  if (coalesce(obj_id, '') = '' or
      obj_id = 0) then
    exit;

  obj_kind = f_objkind(:obj_kind, 'id');

  if (f_get_const('new_contdata') = 0) then
  begin
    if (position('cell', data_type) > 0 or
        data_type = 'all') then -- сотовый или все существующие контактные данные
    begin
      cd_orderno = 0;
      main_founded = 0;
      for select coalesce(ok_cell_main, ''), coalesce(ok_cell_add, ''), coalesce(last_cell_main, ''),
                 coalesce(last_cell_add, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, ok_data_add, last_data_main, last_data_add
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный номер
          cd_type = 'ContactInfo.cell_main';
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
          main_founded = 1;
        end

        if (ok_data_add <> '' and
            ok_data_add <> ok_data_main) then
        begin
          -- Дополнительный верифицированный номер
          if (main_founded = 1) then
          begin
            cd_type = 'ContactInfo.cell_add';
            cont_value = ok_data_add;
            cont_status = 1;
          end
          else
          begin
            cd_type = 'ContactInfo.cell_main';
            cont_value = ok_data_main;
            cont_status = 2;
          end
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main and
            last_data_main <> ok_data_add) then
        begin
          -- Новый основной номер, не подтвержден
          cd_type = 'ContactInfo.cell_main';
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end

        if (last_data_add <> '' and
            last_data_add <> ok_data_add and
            last_data_add <> ok_data_main and
            last_data_add <> last_data_main) then
        begin
          -- Новый доп.номер, не подтвержден
          cd_type = 'ContactInfo.cell_add';
          cont_value = last_data_add;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('email', data_type) > 0 or
        data_type = 'all') then -- email или все существующие контактные данные
    begin
      cd_orderno = 1;
      cd_type = 'ContactInfo.email';
      for select coalesce(ok_email, ''), coalesce(last_email, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный email
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый email, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('phone', data_type) > 0 or
        data_type = 'all') then -- стационарные или все существующие контактные данные
    begin
      cd_orderno = 2;
      for select coalesce(ok_phone_home, ''), coalesce(ok_phone_work, ''), coalesce(last_phone_home, ''),
                 coalesce(last_phone_work, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, ok_data_add, last_data_main, last_data_add
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной домашний номер
          cd_type = 'ContactInfo.phone_home';
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (ok_data_add <> '' and
            ok_data_add <> ok_data_main) then
        begin
          -- Основном рабочий номер
          cd_type = 'ContactInfo.phone_work';
          cont_value = ok_data_add;
          cont_status = 1;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main and
            last_data_main <> ok_data_add) then
        begin
          -- Новый основной номер, не подтвержден
          cd_type = 'ContactInfo.phone_home';
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end

        if (last_data_add <> '' and
            last_data_add <> ok_data_add and
            ok_data_add <> ok_data_main and
            ok_data_add <> last_data_main) then
        begin
          -- Новый доп.номер, не подтвержден
          cd_type = 'ContactInfo.phone_work';
          cont_value = last_data_add;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('skype', data_type) > 0 or
        data_type = 'all') then -- skype или все существующие контактные данные
    begin
      cd_orderno = 3;
      cd_type = 'ContactInfo.skype';
      for select coalesce(ok_skype, ''), coalesce(last_skype, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный skype
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый skype, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('whatsapp', data_type) > 0 or
        data_type = 'all') then -- whatsapp или все существующие контактные данные
    begin
      cd_orderno = 4;
      cd_type = 'ContactInfo.whatsapp';
      for select coalesce(ok_whatsapp, ''), coalesce(last_whatsapp, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный whatsapp
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый whatsapp, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('viber', data_type) > 0 or
        data_type = 'all') then -- viber или все существующие контактные данные
    begin
      cd_orderno = 4;
      cd_type = 'ContactInfo.viber';
      for select coalesce(ok_viber, ''), coalesce(last_viber, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный viber
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый viber, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('telegram', data_type) > 0 or
        data_type = 'all') then -- telegram или все существующие контактные данные
    begin
      cd_orderno = 4;
      cd_type = 'ContactInfo.telegram';
      for select coalesce(ok_telegram, ''), coalesce(last_telegram, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный telegram
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый telegram, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('imo', data_type) > 0 or
        data_type = 'all') then -- imo или все существующие контактные данные
    begin
      cd_orderno = 4;
      cd_type = 'ContactInfo.imo';
      for select coalesce(ok_imo, ''), coalesce(last_imo, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный imo
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый imo, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('OK', data_type) > 0 or
        data_type = 'all') then -- OK или все существующие контактные данные
    begin
      cd_orderno = 4;
      cd_type = 'ContactInfo.OK';
      for select coalesce(ok_ok, ''), coalesce(last_ok, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный OK
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый OK, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('facebook', data_type) > 0 or
        data_type = 'all') then -- facebook или все существующие контактные данные
    begin
      cd_orderno = 4;
      cd_type = 'ContactInfo.facebook';
      for select coalesce(ok_facebook, ''), coalesce(last_facebook, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный facebook
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый facebook, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end

    if (position('VK', data_type) > 0 or
        data_type = 'all') then -- VK или все существующие контактные данные
    begin
      cd_type = 'ContactInfo.VK';
      cd_orderno = 4;
      for select coalesce(ok_vk, ''), coalesce(last_vk, '')
          from continfo
          where cci_object_kind = :obj_kind and
                cci_object_id = :obj_id
          into :ok_data_main, last_data_main
      do
      begin
        if (ok_data_main <> '') then
        begin
          -- Основной верифицированный VK
          cont_value = ok_data_main;
          cont_status = 2;
          suspend;
        end

        if (last_data_main <> '' and
            last_data_main <> ok_data_main) then
        begin
          -- Новый VK, не подтвержден
          cont_value = last_data_main;
          cont_status = 0;
          suspend;
        end
      end
    end
  end
  else
  begin
    -- новая структура хранения данных
    cd_orderno = 0;
    for select f_cont_code(cda_from_clt_id), f_enum('cont_kind', cda_cont_kind, 'code'),
               cda_is_main, cda_status, cda_cont_kind
        from contdata
        where cda_from_c_id = :obj_id and
              cda_is_delete = 0 and
              (:data_type = 'all' or
              position(f_enum('cont_kind', cda_cont_kind, 'code'), :data_type) > 0)
        order by cda_cont_kind, cda_is_main desc, cda_status desc, cda_id desc
        into cont_value, cd_type, cda_is_main, cda_status, cd_orderno
    do
    begin
      cont_status = cda_status;

      if (cda_status = 1 and
          cda_is_main = 1) then
        cont_status = 2;

      if (cd_type = 'cell') then
      begin
        if (cda_is_main = 1) then
          cd_type = 'cell_main';
        else
          cd_type = 'cell_add';
      end
      cd_type = 'ContactInfo.' || cd_type;

      suspend;
    end
  end
end
