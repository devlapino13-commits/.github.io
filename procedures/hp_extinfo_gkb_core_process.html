CREATE PROCEDURE HP_EXTINFO_GKB_CORE_PROCESS(
    EIL_ID D_ID)
AS
declare variable BLS_ID          D_ID;
declare variable EIL_STATUS      D_SMALLINT;
declare variable C_ID            D_ID;
declare variable OBJ_KIND        D_INTEGER;
declare variable OBJ_ID          D_ID;
declare variable JSON_INFO       D_BLOB_TEXT;
declare variable RATING          D_STR_CODE;
declare variable SCORE_VAL       D_INTEGER;
declare variable SESSION_ID      D_STR_64;
declare variable REP_START       D_STR_64;
declare variable DECISION        D_SMALLINT;
declare variable IS_ERROR        D_SMALLINT;
declare variable MAIL_TO_FRONT   D_STR_255;
declare variable MAIL_TO_FIN     D_STR_255;
declare variable SERVICE_DOWN    D_STR_CODE;
declare variable MSG_BODY        D_STR_2048;
begin
  mail_to_front = 'b.djumabekova@mbulak.kz;zh.batyrova@mbulak.kz';
  mail_to_fin   = 'n.ismankulova@mbulak.kz;d.kassymbek@mbulak.kz';

  select eil.eil_from_bls_id, eil.eil_status, eil.eil_from_c_id,
         eil.eil_object_kind, eil.eil_object_id
  from extinfolist eil
  inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
  where eil.eil_id = :eil_id
    and eic.eic_code = 'gkb_pay_checks_core'
  into bls_id, eil_status, c_id, obj_kind, obj_id;

  if (obj_id is null) then
    exit;  -- не наша запись

  rating = null; score_val = null; session_id = null; rep_start = null;
  decision = null; is_error = 0;

  if (eil_status = 2 and bls_id is not null) then
  begin
    json_info = coalesce((select blx_blob from hp_get_blobex_blob(:bls_id)), '{}');

    rating     = lower(coalesce(f_json_blob_get_value(json_info, 'score.rating', ''), ''));
    score_val  = cast(coalesce(f_json_blob_get_value(json_info, 'score.score', '0'), '0') as d_integer);
    session_id = left(coalesce(f_json_blob_get_value(json_info, 'info.session_id', ''), ''), 64);
    rep_start  = left(coalesce(f_json_blob_get_value(json_info, 'info.report_start_date', ''), ''), 64);

    if (rating = '') then
    begin
      is_error = 1;
      rating = 'E';
    end
    else
    begin
      -- Правило «рейтинг -> решение». Держится ТОЛЬКО здесь: результат
      -- кладётся в GPC_DECISION, и все проверки сравнивают уже число.
      -- Дублировать сравнение строк по блокам нельзя — разъедется.
      --
      -- Шкала ГКБ (балл тем выше, чем ниже риск):
      --   A1  91-100 | A2  81-90            наименее рискованные
      --   B1  71-80  | B2  61-70            менее рискованные
      --   C1  51-60  | C2  41-50            рискованные
      --   D1  31-40  | D2  21-30 | D3 0-20  наиболее рискованные
      --
      --   0 - выдача без ограничений                      (A1-B2)
      --   1 - выдача с уведомлением о повышенном риске,
      --       возможно снижение суммы кредита             (C1-D1)
      --   2 - стоп-фактор, отказ                          (D2, D3)
      --
      -- Любой код вне перечисленных списков попадает в else и трактуется
      -- как отказ — это защита на случай новых категорий от ГКБ.
      decision =
        case
          when rating in ('a1', 'a2', 'b1', 'b2') then 0
          when rating in ('c1', 'c2', 'd1')       then 1
          else 2
        end;
    end
  end
  else
  begin
    is_error = 1;  -- статус < 0 или нет блоба → недоступность
    rating = 'E';
  end

  -- Пишем/обновляем результат по объекту (одна актуальная запись на объект)
  update or insert into gkb_pay_check
    (gpc_object_kind, gpc_object_id, gpc_from_c_id, gpc_from_eil_id,
     gpc_rating, gpc_score, gpc_session_id, gpc_req_date, gpc_req_time,
     gpc_decision, gpc_is_error)
  values
    (:obj_kind, :obj_id, :c_id, :eil_id,
     :rating, :score_val, :session_id, current_date, :rep_start,
     :decision, :is_error)
  matching (gpc_object_kind, gpc_object_id);

  -- Уведомления о недоступности / восстановлении сервиса.
  --
  -- Состояние сервиса хранится в константе gkb_fraud_service_down:
  --   '1' — сервис признан недоступным, '0' — работает.
  -- Письмо «недоступен»   шлётся на КАЖДУЮ заявку с категорией E.
  -- Письмо «восстановлен» шлётся РОВНО ОДИН РАЗ, при переходе 1 -> 0.
  --
  -- Раньше факт простоя определялся по последней записи GKB_PAY_CHECK
  -- другого объекта — это давало ложные срабатывания и пропуски.
  service_down = coalesce((select cv.cv_value
                           from const_var cv
                           where cv.cv_code = 'gkb_fraud_service_down'), '0');

  if (is_error = 1) then
  begin
    msg_body = 'Сервис Fraud Score ГКБ временно недоступен. ' ||
               'Выдача кредитов осуществляется без проверки Fraud Score.';
    execute procedure hp_send_quick_message('email', :mail_to_front,
      'Fraud Score ГКБ недоступен', :msg_body, 'gkb_fraud_down', 7, 1000000000000);
    execute procedure hp_send_quick_message('email', :mail_to_fin,
      'Fraud Score ГКБ недоступен', :msg_body, 'gkb_fraud_down', 7, 1000000000000);

    if (service_down <> '1') then
      update or insert into const_var (cv_code, cv_value, cv_comment)
      values ('gkb_fraud_service_down', '1',
              'RFCKZ-877 Признак недоступности сервиса Fraud Score ГКБ (1-лежит, 0-работает)')
      matching (cv_code);
  end
  else
  begin
    -- Успешный ответ. Если сервис числился недоступным — фиксируем восстановление.
    if (service_down = '1') then
    begin
      msg_body = 'Работа сервиса Fraud Score ГКБ восстановлена. ' ||
                 'Проверка клиентов вновь выполняется в штатном режиме.';
      execute procedure hp_send_quick_message('email', :mail_to_front,
        'Fraud Score ГКБ восстановлен', :msg_body, 'gkb_fraud_up', 7, 1000000000000);
      execute procedure hp_send_quick_message('email', :mail_to_fin,
        'Fraud Score ГКБ восстановлен', :msg_body, 'gkb_fraud_up', 7, 1000000000000);

      update or insert into const_var (cv_code, cv_value, cv_comment)
      values ('gkb_fraud_service_down', '0',
              'RFCKZ-877 Признак недоступности сервиса Fraud Score ГКБ (1-лежит, 0-работает)')
      matching (cv_code);
    end
  end
end
