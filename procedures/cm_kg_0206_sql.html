CREATE PROCEDURE CM_KG_0206_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_circle d_integer;
declare variable k_disb_date d_date;
declare variable c_birth_date date;
declare variable ma_id d_id;
declare variable mac_is_poruch d_smallint;
declare variable c_id d_id;
declare variable poruch_id d_id;
declare variable poruch_birth_date d_date;
begin
  if (k_id = 109010225929100101) then
    exit;
  if (k_id = 109010228089800101) then
    exit; -- jira 282
  if (k_id = 109010229094800101) then
    exit; --  Ticket#2023040888000502 08.04.2023

  res_error_is_critical = 0;

  for select kc_circle, k_disb_date, c_birth_date, k_from_ma_app_id, c_id
      from knclient kc
      inner join kontract kn on kn.k_id = kc.kc_from_k_id
      inner join clients cl on cl.c_id = kc.kc_from_c_id
      where kc.kc_from_k_id = :k_id and
            kc.kc_amount_approved > 0
      into kc_circle, k_disb_date, c_birth_date, ma_id, c_id
  do
  begin
    if ((dateadd(year, 70, c_birth_date) < k_disb_date) and
        (kc_circle < 2)) then
    begin
      res_error_text = 'У нового клиента возраст не должен быть выше 70 лет.';
      res_error_is_critical = 2;
      suspend;
    end
  end

  if (exists(select *
             from zalog zl
             where zl.z_from_k_id = :k_id and
                   zl.z_type = 0)) then
  begin
    for select zl.z_from_c_id, c.c_birth_date
        from zalog zl
        inner join clients c on c.c_id = zl.z_from_c_id
        where zl.z_from_k_id = :k_id and
              zl.z_type = 0
        into poruch_id, poruch_birth_date
    do
    begin
      if (not exists(select *
                     from zalog zl
                     where zl.z_from_k_id <> :k_id and
                           zl.z_from_c_id = :poruch_id and
                           zl.z_type = 0) and
          dateadd(year, 65, :poruch_birth_date) < k_disb_date) then
      begin
        res_error_text = 'У поручителя возраст не должен быть выше 65 лет.';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end
end
