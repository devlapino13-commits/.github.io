CREATE PROCEDURE AAREP_SURVEY_TEST_TMP(
    QUES_LIST VARCHAR(255))
RETURNS (
    OBJ_NAME VARCHAR(255),
    IS_CORRECT INTEGER)
AS
declare variable svr_source_kind integer;
declare variable svr_source_id decimal(18,0);
declare variable res_str varchar(255);
declare variable svq_correct_sva_id integer;
begin
  for select res_str
      from hp_split_str(:ques_list, ' ')
      into :res_str
  do
  begin
    svq_correct_sva_id = null;
    select svq_correct_sva_id
    from survey_ques
    where svq_id = :res_str
    into svq_correct_sva_id;
    if (svq_correct_sva_id is null) then
      svq_correct_sva_id = -1;

    for select svr_source_kind, svr_source_id, iif(svr_from_sva_id = :svq_correct_sva_id, 1, 0)
        from survey_result
        where svr_from_svq_id = :res_str
        into :svr_source_kind, :svr_source_id, is_correct
    do
    begin
      obj_name = '';
      if (svr_source_kind = 0) then
      begin
        obj_name = (select c_name_fio
                    from clients
                    where c_id = :svr_source_id);
      end
      if (svr_source_kind = 1) then
      begin
        obj_name = svr_source_id;
      end
      suspend;
    end
  end
end
