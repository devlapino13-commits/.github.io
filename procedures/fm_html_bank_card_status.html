CREATE PROCEDURE FM_HTML_BANK_CARD_STATUS(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable lang d_str_code;
declare variable k_kontract_no d_str_100;
declare variable k_amount_approved d_currency;
declare variable k_disb_date d_date;
declare variable k_disb_kind d_integer;
declare variable pr_name d_str_255;
declare variable reg_id d_id;
declare variable pay_status d_integer;
declare variable pay_status_text d_str_255;
declare variable reg_name d_str_255;
declare variable client_total d_integer;
declare variable client_cards d_integer;
declare variable client_done d_integer;
declare variable client_error d_integer;
declare variable pay_error_text d_str_255;
declare variable k_id d_id;
declare variable kc_id d_id;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable fps_name_fio d_str_255;
declare variable res_check_state d_integer;
declare variable check_kind d_integer;
declare variable rec_id d_integer;
declare variable k_id_str d_integer;
declare variable is_critical d_str_100;
declare variable ccr_from_cdm_id d_id;
declare variable cdm_memo d_str_255;
declare variable ccr_info d_str_255;
declare variable ccr_message d_str_255;
declare variable ccr_is_critical d_smallint;
declare variable ccr_datetime d_datetime;
declare variable counter d_integer;
declare variable res_check_state_str d_str_100;
begin
  lang = f_get_lang_code(:un_id, null);
  check_kind = f_enum_id('condition_check_kind', 'loan_disb');
  res_text = (select res_style
              from fm_html_styles) || '<body>';
  if (obj_kind = 'new' and
      exists(select *
             from kontstate ks
             where ks.ks_from_k_id = :obj_id and
                   ks.ks_date_begin = current_date and
                   ks.ks_ol_is_delete = 0 and
                   ks.ks_state = 1)) then
  begin
    res_text = BLOB_APPEND(res_text, '<h2>' || f_lang(lang, 'Контракт выдан') || '</h2>');
    suspend;
    exit;
  end
  res_text = BLOB_APPEND(res_text, '<h2>' || f_lang(lang, 'Информация о договоре') || '</h2>');

  if (obj_kind = 'new') then -- готовые к выдаче
  begin
    pay_status = 0;
    pay_status_text = '';

    for select kont.k_kontract_no, kont.k_amount_approved, kont.k_disb_date, kont.k_disb_kind, p.pr_name,
               kd.kkd_from_reg_id
        from kontcarddisb kd, kontract kont, product p
        where kd.kkd_from_k_id = kont.k_id and
              kont.k_from_pr_id = p.pr_id and
              kont.k_id = :obj_id
        into :k_kontract_no, :k_amount_approved, :k_disb_date, :k_disb_kind, :pr_name, :reg_id
    do
    begin
      select reg.reg_name
      from region reg
      where reg.reg_unicode = :reg_id
      into reg_name;
      pay_status = 2; -- Еще не выданные
      pay_status_text = '';
      pay_error_text = '';
      client_total = 0;
      client_cards = 0; -- проверить наличие банк карт
      client_done = 0;
      client_error = 0;

      for select kc_id, kc_from_c_id
          from knclient
          where kc_from_k_id = :obj_id and
                kc_amount_approved > 0
          into kc_id, c_id
      do
      begin
        c_name_fio = '';
        fps_name_fio = '';

        select c_name_fio, ''
        from clients
        where c_id = :c_id
        into c_name_fio, fps_name_fio;

        client_total = client_total + 1;
        if (k_disb_kind = 6) then
        begin
          if (exists(select *
                     from hp_client_bank_card_mb(:c_id))) then
            client_cards = client_cards + 1;
        end
        else
        begin
          if ((select coalesce(token, '')
               from hp_client_bank_card(:c_id)) <> '') then
            client_cards = client_cards + 1;
        end
      end

      res_check_state = 0; -- не проверен на соответствие
      select iif(ccs_errors > 0, -2, 1)
      from condcheckstate
      where ccs_object_kind = f_objkind('kontract', 'id') and
            ccs_object_id = :obj_id and
            ccs_check_kind = :check_kind
      into res_check_state;
      if (res_check_state is null) then
        res_check_state = 0;
      rec_id = rec_id + 1;
      res_check_state_str = case :res_check_state
                              when 0 then 'Не проверен'
                              when 1 then 'Проверен успешно'
                              else 'Ошибка'
                            end;
    end
    res_text = BLOB_APPEND(res_text, '<table width="100%"><col width="40%"><col width="60%">' || f_html_table_row(f_lang(lang, 'Условия'), :res_check_state_str, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Номер договора'), :k_kontract_no, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Дата выдачи'), :k_disb_date, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Сумма'), :k_amount_approved, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Офис'), :reg_name, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Кредитный продукт'), :pr_name, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Клиенты'), :client_total, '', '', -1, 1, 1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'С картой'), :client_cards, '', '', -1, 1, 1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'С ошибкой'), :client_error, '', '', -1, 1, 1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Статус оплаты'), :pay_status_text, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Текст ошибки'), :pay_error_text, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Провайдер'), 'elcard', '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Тип карты'), 'карта М булак', '', '', -1, 1, -1, -1, 2, '') || '</table>');
  end
  else
  begin
    -- выданные по базе
    for select kont.k_kontract_no, kont.k_amount_approved, kont.k_disb_date, kont.k_disb_kind, p.pr_name
        from kontstate ks, kontract kont, product p
        where ks.ks_state = 1 -- выданные по Kontstate
              and
              ks.ks_ol_is_delete = 0 and
              ks.ks_from_k_id = kont.k_id and
              kont.k_is_online = 1 and -- только онлайн выдачи
              kont.k_disb_kind in (5, 6, 7) and -- только перевод на карту + СБП
              kont.k_from_pr_id = pr_id and
              kont.k_id = :obj_id
        into :k_kontract_no, :k_amount_approved, :k_disb_date, :k_disb_kind, :pr_name
    do
    begin
      c_name_fio = '';
      fps_name_fio = '';
      select status, description, client_total, client_done, client_error, client_cards
      from hp_kont_card_disb_status(:obj_id)
      into :pay_status, :pay_error_text, :client_total, :client_done, :client_error, client_cards;

      if (pay_status = 2) then -- если займ выдан, НО нет записи в ExtSysMain, считаем ошибкой
      begin
        if (pay_error_text = '') then
          pay_error_text = 'Займ выдан, но нет записи об оплате на карту';

        pay_status = -1;
      end

      pay_status_text = trim(case pay_status
                               when -1 then 'Ошибка'
                               when 0 then 'На обработке'
                               when 1 then 'Выдан'
                             end);

      if ((obj_kind = 'process' and
          pay_status = 0) or (obj_kind = 'done' and
          pay_status = 1) or (obj_kind = 'error' and
          pay_status = -1)) then
      begin
        rec_id = rec_id + 1;

        res_check_state = 0; -- не проверен на соответствие
        select iif(ccs_errors > 0, -2, 1)
        from condcheckstate
        where ccs_object_kind = :obj_kind and
              ccs_object_id = :obj_id and
              ccs_check_kind = :check_kind
        into res_check_state;
        if (res_check_state is null) then
          res_check_state = 0;

        select c_name_fio
        from knclient
        left join clients on kc_from_c_id = c_id
        where kc_from_k_id = :obj_id and
              kc_amount_approved > 0 rows 1
        into c_name_fio;
        res_check_state_str = case :res_check_state
                                when 0 then 'Не проверен'
                                when 1 then 'Проверен успешно'
                                else 'Ошибка'
                              end;
      end
      res_text = BLOB_APPEND(res_text, '<table width="100%"><col width="40%"><col width="60%">' || f_html_table_row(f_lang(lang, 'Условия'), :res_check_state_str, '', '', -1, 1, -1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Номер договора'), :k_kontract_no, '', '', -1, 1, -1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Дата выдачи'), :k_disb_date, '', '', -1, 1, -1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Сумма'), :k_amount_approved, '', '', -1, 1, -1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Офис'), :reg_name, '', '', -1, 1, -1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Кредитный продукт'), :pr_name, '', '', -1, 1, -1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Клиенты'), :client_total, '', '', -1, 1, 1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'С картой'), :client_cards, '', '', -1, 1, 1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Выдан'), :pay_status_text, '', '', -1, 1, 1, -1, 2, '')); -- втапдл
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'С ошибкой'), :client_error, '', '', -1, 1, 1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Статус оплаты'), :pay_status_text, '', '', -1, 1, -1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Текст ошибки'), :pay_error_text, '', '', -1, 1, -1, -1, 2, ''));
      res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Провайдер'), 'elcard', '', '', -1, 1, -1, -1, 2, '') || '</table>');
    end
  end
  res_text = BLOB_APPEND(res_text, '<br>');
  res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(lang, 'Проверить условия'), 'check_condition_bankcard', 'Kontract', :obj_id, 'none', f_lang(lang, 'Подтверждаете отправку контракта на проверку условий?'), '', 0, 0, '', '', 'window', '') || '<br><br>');
  counter = 1;
  for select ccr_from_cdm_id, cdm_memo, ccr_info, ccr_message, ccr_is_critical, ccr_datetime
      from hp_check_loan_konditions_show('kontract', :obj_id, 'loan_disb', :un_id)
      into ccr_from_cdm_id, cdm_memo, ccr_info, ccr_message, ccr_is_critical, ccr_datetime
  do
  begin
    is_critical = case :ccr_is_critical
                    when 2 then 'очень критично'
                    when 1 then 'критично'
                    else 'информация'
                  end;
    res_text = BLOB_APPEND(res_text, '<h3>' || f_lang(lang, 'Проверка условий (Блок № ' || :counter || ' )') || '</h3>');
    res_text = BLOB_APPEND(res_text, '<table width="100%"><col width="40%"><col width="60%">' || f_html_table_row(f_lang(lang, 'ID'), :ccr_from_cdm_id, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Проверка'), :ccr_info, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Текст предупреждения'), :cdm_memo, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Описание'), :ccr_message, '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Время проверки'), f_datetimetostr(:ccr_datetime), '', '', -1, 1, -1, -1, 2, ''));
    res_text = BLOB_APPEND(res_text, f_html_table_row(f_lang(lang, 'Критичность'), :is_critical, '', '', -1, 1, -1, -1, 2, '') || '</table><br>');
    counter = counter + 1;
  end
  if (:res_check_state = 1) then
    res_text = BLOB_APPEND(res_text, f_uri_sql(f_lang(lang, 'Выдать на карту'), 'bankcard_disb', 'Kontract', :obj_id, 'none', f_lang(lang, 'Подтверждаете выдачу на карту?'), '', 0, 0, '', '', 'dialog', '') || '<br>');
  res_text = BLOB_APPEND(res_text, '</body>');
  suspend;
end
