CREATE PROCEDURE HP_PASSPORT_NUMBER_SPLIT(
    PASS_NO D_STR_CODE)
RETURNS (
    PASS_SERIA D_STR_CODE,
    PASS_NUMBER D_STR_CODE)
AS
declare variable counter d_integer;
declare variable str_leng d_integer;
declare variable ch d_str_sym;
begin
  -- Процедура разделяет номера паспорта на серию и номер
  if (pass_no is null) then
    pass_no = '';

  pass_seria = '';
  pass_number = '';

  pass_no = replace(pass_no, ' ', '');
  if(pass_no similar to '[-]?[[:digit:]]+') then -- если состоит полностью из цифр - РФ формат, первые 4 символа это серия
  begin
    pass_seria = substring(pass_no from 1 for 4);
    pass_number = substring(pass_no from 5);
  end
  else
  begin
    -- Если имеются символы, выделяем символы как серию, остальное как номер
    counter = 1;
    str_leng = character_length(pass_no);
    while (counter < str_leng-1) do
    begin
      ch = substring(pass_no from counter for 1);
      if(ch not similar to '[-]?[[:digit:]]+') then
        pass_seria = pass_seria||ch;
      else
      begin
        pass_number = substring(pass_no from counter);
        counter = str_leng-1;
      end
      counter = counter + 1;
    end
  end

  suspend;
end
