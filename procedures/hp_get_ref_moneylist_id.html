CREATE PROCEDURE HP_GET_REF_MONEYLIST_ID(
    OL_ID D_ID,
    OL_DATE D_DATE,
    ACC_DT D_ID,
    ACC_KT D_ID,
    KC_ID D_ID,
    OB_C_ID D_ID)
RETURNS (
    RES_RML_ID D_INTEGER,
    RES_CASHFLOW_DATE D_DATE)
AS
declare variable ab_id_dt d_id;
declare variable ab_id_kt d_id;
declare variable acc_code_dt d_str_20;
declare variable acc_code_kt d_str_20;
declare variable sab_full_code_dt d_str_code;
declare variable sab_full_code_kt d_str_code;
declare variable kc_id_dt d_id;
declare variable kc_id_kt d_id;
declare variable is_resident d_bool;
declare variable is_law d_bool;
declare variable rmm_id d_id;
declare variable non_resident_dt d_bool;
declare variable non_resident_kt d_bool;
declare variable need_resident d_bool;
declare variable need_law d_bool;
declare variable rmm_from_rml_resident d_integer;
declare variable rmm_from_rml_noresident d_integer;
declare variable rmm_law_rml_resident d_integer;
declare variable rmm_law_rml_noresident d_integer;
declare variable c_id d_id;
declare variable law_k_id d_id;
declare variable plat_acc_id d_id;
begin
  res_rml_id = -1;

  res_cashflow_date = null;

  select acc_from_ab_id, sab_full_code, acc_code,
         iif(acc_code starting '603',
             iif(substring(acc_code from 13 for 1) = 2, 1, 0), acc_non_resident),
         iif(acc_object_kind = 8, acc_object_id, null)
  from ref_account, sys_accbalance
  where acc_id = :acc_dt and
        acc_from_sab_id = sab_id
  into ab_id_dt, sab_full_code_dt, acc_code_dt, non_resident_dt, kc_id_dt;

  if (kc_id is null and kc_id_dt is not null) then
    kc_id = kc_id_dt;

  select acc_from_ab_id, sab_full_code, acc_code,
         iif(acc_code starting '603',
             iif(substring(acc_code from 13 for 1) = 2, 1, 0), acc_non_resident),
         iif(acc_object_kind = 8, acc_object_id, null)
  from ref_account, sys_accbalance
  where acc_id = :acc_kt and
        acc_from_sab_id = sab_id
  into ab_id_kt, sab_full_code_kt, acc_code_kt, non_resident_kt, kc_id_kt;

  if (kc_id is null and kc_id_kt is not null) then
    kc_id = kc_id_kt;

  if (sab_full_code_dt not in ('Деньги.Безнал.Банк.РасчСчет', 'Деньги.Безнал.Банк.СпецСчет',
                               'Деньги.Нал.Касса',
                               'ПлатСистемы.Выдано',
                               'ПлатСистемы.Погашено') and
      sab_full_code_kt not in ('Деньги.Безнал.Банк.РасчСчет', 'Деньги.Безнал.Банк.СпецСчет',
                               'Деньги.Нал.Касса',
                               'ПлатСистемы.Выдано',
                               'ПлатСистемы.Погашено')) then
  begin
    -- Не денежная операция, завершаем работу возвращая -1.
    suspend;
    exit;
  end

  if (sab_full_code_dt = 'ПлатСистемы.Погашено') then
  begin
    plat_acc_id = null;
    select kob_acc_dt
    from kn_operbook, ref_account, sys_accbalance
    where kob_from_ol_id = :ol_id and
          kob_acc_dt = acc_id and
          acc_from_sab_id = sab_id and
          sab_full_code = :sab_full_code_dt
    rows 1
    into plat_acc_id;

    select kob_ol_date
    from operperiod, kn_operbook, ref_account, sys_accbalance
    where :ol_date between olp_date_begin and olp_date_end and
          olp_from_ol_id = kob_from_ol_id and
          kob_acc_kt + 0 = :plat_acc_id and
          kob_acc_dt = acc_id and
          acc_from_sab_id = sab_id and
          sab_full_code starting 'Деньги.Безнал.Банк.'
    rows 1
    into res_cashflow_date;

    if (res_cashflow_date = ol_date) then
      res_cashflow_date = null;
  end

  res_rml_id = 0;
  is_resident = 1;
  is_law = 0;

  select rmm_id,
         rmm_from_rml_resident, rmm_from_rml_noresident,
         rmm_law_rml_resident, rmm_law_rml_noresident
  from ref_moneymask
  where rmm_id > 0 and
        rmm_from_ab_dt = :ab_id_dt and
        rmm_from_ab_kt = :ab_id_kt and
        :acc_code_dt like rmm_mask_dt and
        :acc_code_kt like rmm_mask_kt
  order by rmm_order_no
  rows 1
  into rmm_id,
       rmm_from_rml_resident, rmm_from_rml_noresident,
       rmm_law_rml_resident, rmm_law_rml_noresident;

  if (rmm_id is not null) then
  begin
    need_resident = 0;
    need_law = 0;

    if (rmm_from_rml_resident > 0 and
        rmm_from_rml_noresident > 0 and
        rmm_from_rml_resident <> rmm_from_rml_noresident) then
      need_resident = 1;

    if (rmm_law_rml_resident > 0 and
        rmm_law_rml_noresident > 0 and
        rmm_law_rml_resident <> rmm_law_rml_noresident) then
      need_resident = 1;

    if (kc_id is not null and
       (rmm_law_rml_resident > 0 or rmm_law_rml_noresident > 0)) then
      need_law = 1;

    if (kc_id is null and ob_c_id is null) then
    begin
      if (need_resident = 1) then
      begin
        if (non_resident_dt = 1 or non_resident_kt = 1) then
          is_resident = 0;
      end
    end else
    begin
      if (need_resident = 1) then
      begin
        if (kc_id is not null) then
        begin
          select kc_from_c_id
          from knclient
          where kc_id = :kc_id
          into c_id;
        end else
          c_id = ob_c_id;

        select 1 - res_non_resident
        from hp_get_client_type(:c_id, null, null, :ol_date)
        into is_resident;
      end

      if (need_law = 1) then
      begin
        if (exists(select * from kn_oksummary
        where oks_from_kc_id = :kc_id and
              oks_date_begin <= :ol_date and
              (oks_acr_law <> 0 or oks_pmt_law <> 0)))
        then is_law = 1;
        else
        begin
          law_k_id = null;
          select coalesce(k_base_k_id, k_id)
          from knclient, kontract
          where kc_id = :kc_id and
                kc_from_k_id = k_id
          into law_k_id;

          if (exists(select * from opersud
          where osd_from_k_id = :law_k_id and
                osd_ol_date <= :ol_date and
                osd_ol_is_delete = 0))
          then is_law = 1;

        end
      end
    end

    if (need_resident = 1) then
    begin
      if (need_law = 1) then
      begin
        if (is_resident = 1) then
        begin
          if (is_law = 0)
          then res_rml_id = rmm_from_rml_resident;
          else res_rml_id = rmm_law_rml_resident;
        end else
        begin
          if (is_law = 0)
          then res_rml_id = rmm_from_rml_noresident;
          else res_rml_id = rmm_law_rml_noresident;
        end
      end else
      begin
        if (is_resident = 1)
        then res_rml_id = rmm_from_rml_resident;
        else res_rml_id = rmm_from_rml_noresident;
      end
    end else
    begin
      if (need_law = 1) then
      begin
        if (is_law = 0)
        then res_rml_id = maxvalue(rmm_from_rml_resident, rmm_from_rml_noresident);
        else res_rml_id = maxvalue(rmm_law_rml_resident, rmm_law_rml_noresident);
      end else
      begin
        res_rml_id = maxvalue(rmm_from_rml_resident, rmm_from_rml_noresident);
      end
    end
  end

  suspend;
end
