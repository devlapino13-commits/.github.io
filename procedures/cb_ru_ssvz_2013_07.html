CREATE PROCEDURE CB_RU_SSVZ_2013_07(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    ID_LIST INTEGER,
    OKRUG_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    REG_NAME VARCHAR(50),
    REG_BALANCE_DELINQ CURR_TYPE,
    REG_BALANCE_LAST CURR_TYPE,
    PEREDAN_BALANCE CURR_TYPE,
    PAR7 DECIMAL(15,1),
    CB_BONUS_SVZ CURR_TYPE,
    CB_BONUS_SSVZ_TOTAL CURR_TYPE,
    PAR7_LAST DECIMAL(15,1),
    KOL_SVZ INTEGER)
AS
declare variable rep_date_begin date;
declare variable rep_date_end_2 date;
declare variable rep_date_begin_2 date;
declare variable rep_date_end_3 date;
declare variable reg_balance_last_2 curr_type;
declare variable reg_balance_delinq_2 curr_type;
declare variable bonus_perc curr_type;
declare variable par_razn decimal(15,1);
declare variable peredan_balance_2 curr_type;
declare variable cb_personal_id id_type;
declare variable cb_region_id integer;
declare variable cb_bonus_total curr_type;
begin
  rep_date_begin = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date); -- Начало периода
  rep_date_end_2 = rep_date_begin - 1; -- Конец предыдущего месяца

  rep_date_begin_2 = cast('01.' || extract(month from :rep_date_end_2) || '.' || extract(year from :rep_date_end_2) as date);
  rep_date_end_3 = rep_date_begin_2 - 1;

  for select cb_personal_id, cb_region_id, cb_bonus_total
      from cb_ru_lo_2013_11(:rep_region_id, :rep_date_end)
      into :cb_personal_id, :cb_region_id, :cb_bonus_total
  do
  begin
    insert into temp_bonusru_lo (bl_personal_id, bl_region_id, bl_bonus_total)
    values (:cb_personal_id, :cb_region_id, :cb_bonus_total);
  end

  for select id_list, r3.reg_code || ' ' || r3.reg_name, r2.reg_code || ' ' || r2.reg_name,
             r1.reg_code || ' ' || r1.reg_name
      from region_list(:rep_region_id, 1, 3, 1), region r1, region r2, region r3
      where r1.reg_unicode = id_list and
            r2.reg_unicode = r1.parent and
            r3.reg_unicode = r2.parent
      order by r3.reg_code, r2.reg_code, r1.reg_code
      into id_list, okrug_name, branch_name, reg_name
  do
  begin
--  Портфель всех займов, выданных СВЗ данного офиса с 1 июня 2012
--  Портфель, выданных СВЗ до 1 июня 2012, просрочка на 01 декабря отсутствует
--  Портфель, выданных СВЗ до 1 июня 2012, просрочка на 01 декабря от 1 до 180 дней.

-- B - Портфель займов всего офиса без просрочек и с просрочками 1-180 дней на предыдущую дату
    reg_balance_last = null;
    with tabl1
    as (select
               case
                 when coalesce((select rep_delinq_days
                                from aarep_kont_delinq_status_new(k_id, :rep_date_end_2)), 0) <= 180 then k_id
                 else 0
               end as k_id
        from kontstate, kontract, knaktlist
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end_2 between ks_date_begin and ks_date_end and
              ks_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              kal_new_reg_id = :id_list and
              :rep_date_end_2 between kal_date and kal_date_end)
    select sum(ok_os)
    from tabl1, operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end_2 and
          ok_from_c_id = 0 and
          ok_type in (4, 5, 6, 8) and
          ok_from_k_id = k_id
    into reg_balance_last;
    if (reg_balance_last is null) then
      reg_balance_last = 0;

--  С - Портфель займов всего офиса (без просрочек и с просрочками 1-180 дней на предыдущую отчетную дату),
--  переданный в течение отчетоного периода из другово офиса

--  все активные папки находящийся на конец отчетной даты в данном офисе и не находящийся на начало в данном офисе
    peredan_balance = null;
    with tabl1
    as (select iif((select rep_delinq_days
                    from aarep_kont_delinq_status_new(k_id, :rep_date_end_2)) <= 180, k_id, 0) as k_id,
               kal_end.kal_date as kal_date
        from knaktlist kal_end, knaktlist kal_begin, kontract, kontstate
        where kal_end.is_delete = 0 and
              :rep_date_end between kal_end.kal_date and kal_end.kal_date_end and
              kal_end.kal_new_reg_id = :id_list and
              kal_end.kal_from_kf_id = kal_begin.kal_from_kf_id and
              kal_begin.is_delete = 0 and
              :rep_date_end_2 between kal_begin.kal_date and kal_begin.kal_date_end and
              kal_begin.kal_new_reg_id <> :id_list and
              kal_end.kal_from_kf_id = k_from_kf_id and
              k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end)
    select sum(oks_balance)
    from tabl1, oksummary
    where k_id = oks_from_k_id and
          kal_date between oks_date_begin and oks_date_end and
          oks_close_state = 0
    into peredan_balance;
    if (peredan_balance is null) then
      peredan_balance = 0;

--  Расчет просроченного баланса текущего месяца; A - Портфель займов всего офиса с просрочками 8-180 дней на отчетную дату
    reg_balance_delinq = null;
    with tabl1
    as (select
               case
                 when coalesce((select rep_delinq_days
                                from aarep_kont_delinq_status_new(k_id, :rep_date_end)), 0) between 8 and 180 then k_id
                 else 0
               end as k_id
        from kontstate, kontract, knaktlist
        where ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end and
              ks_from_k_id = k_id and
              k_from_kf_id = kal_from_kf_id and
              knaktlist.is_delete = 0 and
              kal_new_reg_id = :id_list and
              :rep_date_end between kal_date and kal_date_end)
    select sum(ok_os)
    from tabl1, operkred
    where ok_ol_is_active = 1 and
          ok_ol_date <= :rep_date_end and
          ok_from_c_id = 0 and
          ok_type in (4, 5, 6, 8) and
          ok_from_k_id = k_id
    into reg_balance_delinq;
    if (reg_balance_delinq is null) then
      reg_balance_delinq = 0;
--  Расчет PAR,
    if (reg_balance_last + peredan_balance > 0) then
      par7 = reg_balance_delinq / (reg_balance_last + peredan_balance) * 100;
    else
      par7 = 100;
--  Расчет суммы бонусов
    cb_bonus_svz = 0;

--    select sum(cb_bonus_total) from CB_RU_LO_2013_07(1000000,:rep_date_end)
--    where cb_bonus_total > 0
--    and cb_is_get = 1
--    and cb_region_id = :id_list
--    into cb_bonus_svz;

    select sum(bl_bonus_total)
    from temp_bonusru_lo
    where bl_region_id = :id_list
    into cb_bonus_svz;

    par7_last = 0;
    cb_bonus_ssvz_total = 0;
    if (cb_bonus_svz is null) then
      cb_bonus_svz = 0;
    if (cb_bonus_svz > 0) then
      if (par7 <= 7) then
      begin
        cb_bonus_ssvz_total = cb_bonus_svz * 30 / 100;
      end
      else
      begin
--    Если пар7>7% то проверяет на снижения пара
        reg_balance_last_2 = null;
        with tabl1
        as (select
                   case
                     when coalesce((select rep_delinq_days
                                    from aarep_kont_delinq_status_new(k_id, :rep_date_end_3)), 0) <= 180 then k_id
                     else 0
                   end as k_id
            from kontstate, kontract, knaktlist
            where ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  :rep_date_end_3 between ks_date_begin and ks_date_end and
                  ks_from_k_id = k_id and
                  k_from_kf_id = kal_from_kf_id and
                  knaktlist.is_delete = 0 and
                  kal_new_reg_id = :id_list and
                  :rep_date_end_3 between kal_date and kal_date_end)
        select sum(ok_os)
        from tabl1, operkred
        where ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date_end_3 and
              ok_from_c_id = 0 and
              ok_type in (4, 5, 6, 8) and
              ok_from_k_id = k_id
        into reg_balance_last_2;
        if (reg_balance_last_2 is null) then
          reg_balance_last_2 = 0;

        peredan_balance_2 = null;
        with tabl1
        as (select iif((select rep_delinq_days
                        from aarep_kont_delinq_status_new(k_id, :rep_date_end_3)) <= 180, k_id, 0) as k_id,
                   kal_end.kal_date as kal_date
            from knaktlist kal_end, knaktlist kal_begin, kontract, kontstate
            where kal_end.is_delete = 0 and
                  :rep_date_end_2 between kal_end.kal_date and kal_end.kal_date_end and
                  kal_end.kal_new_reg_id = :id_list and
                  kal_end.kal_from_kf_id = kal_begin.kal_from_kf_id and
                  kal_begin.is_delete = 0 and
                  :rep_date_begin_2 between kal_begin.kal_date and kal_begin.kal_date_end and
                  kal_begin.kal_new_reg_id <> :id_list and
                  kal_end.kal_from_kf_id = k_from_kf_id and
                  k_id = ks_from_k_id and
                  ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  :rep_date_end_2 between ks_date_begin and ks_date_end)
        select sum(oks_balance)
        from tabl1, oksummary
        where k_id = oks_from_k_id and
              kal_date between oks_date_begin and oks_date_end and
              oks_close_state = 0
        into peredan_balance_2;
        if (peredan_balance_2 is null) then
          peredan_balance_2 = 0;

--  Расчет просроченного баланса текущего месяца; A - Портфель займов всего офиса с просрочками 8-180 дней на отчетную дату
        reg_balance_delinq_2 = null;
        with tabl1
        as (select
                   case
                     when coalesce((select rep_delinq_days
                                    from aarep_kont_delinq_status_new(k_id, :rep_date_end_2)), 0) between 8 and 180 then k_id
                     else 0
                   end as k_id
            from kontstate, kontract, knaktlist
            where ks_ol_is_delete = 0 and
                  ks_state = 1 and
                  :rep_date_end_2 between ks_date_begin and ks_date_end and
                  ks_from_k_id = k_id and
                  k_from_kf_id = kal_from_kf_id and
                  knaktlist.is_delete = 0 and
                  kal_new_reg_id = :id_list and
                  :rep_date_end_2 between kal_date and kal_date_end)
        select sum(ok_os)
        from tabl1, operkred
        where ok_ol_is_active = 1 and
              ok_ol_date <= :rep_date_end_2 and
              ok_from_c_id = 0 and
              ok_type in (4, 5, 6, 8) and
              ok_from_k_id = k_id
        into reg_balance_delinq_2;
        if (reg_balance_delinq_2 is null) then
          reg_balance_delinq_2 = 0;
--  Расчет PAR,
        if (reg_balance_last_2 + peredan_balance_2 > 0) then
          par7_last = reg_balance_delinq_2 / (reg_balance_last_2 + peredan_balance_2) * 100;
        else
          par7_last = 100;
        par_razn = par7_last - par7;

        bonus_perc = case
                       when par_razn <= 0.9 then 0
                       when par_razn between 1.0 and 2.0 then 5
                       when par_razn between 2.1 and 4.0 then 10
                       when par_razn >= 4.1 then 15
                     end;
        cb_bonus_ssvz_total = cb_bonus_svz * bonus_perc / 100;
      end
--  Вычисляет кол-во СВЗ у которых стаж не менее одного месяца
    kol_svz = null;
    with tabl1
    as (select ps1.ps_from_c_id,
               (select ps2.ps_date_begin
                from hr_pstatus ps2, joblist jl2
                where ps2.ps_from_c_id = ps1.ps_from_c_id and
                      ps2.ps_date_begin is not null and
                      ps2.ps_status = 1 and
                      ps2.ps_from_jl_id = jl2.jl_id and
                      jl2.jl_kind = 5
                order by ps2.ps_date_begin
                rows 1) as ps_stage
        from hr_pstatus ps1, joblist jl1
        where :rep_date_end between ps1.ps_date_begin and ps1.ps_date_end and
              ps1.ps_status = 1 and
              ps1.ps_is_working = 1 and
              ps1.ps_from_reg_id = :id_list and
              ps1.ps_from_jl_id = jl1.jl_id and
              jl1.jl_kind = 5)
    select count(*)
    from tabl1
    where cast(:rep_date_end as date) - ps_stage >= 30
    into kol_svz;
    if (kol_svz is null) then
      kol_svz = 0;
--    if (kol_svz < 6) then cb_bonus_ssvz_total = 0;

    suspend;
  end
end
