CREATE PROCEDURE HP_CONTDATA_GET_CELL_VALID(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0))
RETURNS (
    CONT_VALUE VARCHAR(255))
AS
begin
  -- упращенная процедура, возвращает только один актуальный номер
  -- Приоритетно берется последний веденный номер. Если такого нет, берет приортитетно главную.

  cont_value = f_cont_info(obj_kind, obj_id, 'cell', 'last');
  suspend;
end
