CREATE PROCEDURE AAREP_PLANNING_REPAY_KIVA_TMP(
    REPORT_REGION_ID INTEGER,
    REPORT_DATE_END DATE)
RETURNS (
    REP_YEAR INTEGER,
    REP_MONTH INTEGER,
    REP_KONTRACT_NO VARCHAR(20),
    REP_DISB_AMOUNT DECIMAL(18,4),
    REP_MONTH_AMOUNT DECIMAL(18,4))
AS
declare variable k_id decimal(18,0);
declare variable rep_date date;
declare variable k_amount decimal(18,4);
declare variable rep_amount decimal(18,4);
declare variable cm_is_client_mode integer;
begin
-- operkred_fix
  select cm_is_client_mode
  from company
  into cm_is_client_mode;

  if (cm_is_client_mode = 0) then
  begin
    for select distinct kc_from_k_id, k_amount_approved, k_kontract_no
        from knclient, kontstate, kontract
        where ks_from_k_id = k_id and
              kc_from_k_id = k_id and
              kc_amount_approved <> 0 and
              kc_amount_approved is not null and
              kc_from_pt_id = 'KIVA' and
              cast(ks_date_begin as date) <= :report_date_end and
              ks_date_end >= :report_date_end and
              ks_state = 1 and
              k_financed = 1
        into :k_id, :k_amount, :rep_kontract_no
    do
    begin
      for select ok_sh_date, sum(ok_os + ok_sh_os)
          from operkred
          where ok_ol_is_active = 1 and
                ok_from_k_id = :k_id and
                ok_ol_date <= :report_date_end and
                ok_type <= 7 and
                ok_type <> 3 and
                ok_from_c_id = 0
          group by ok_sh_date
          having sum(ok_os + ok_sh_os) > 0
          into :rep_date, :rep_amount
      do
      begin
        rep_year = extract(year from :rep_date);
        rep_month = extract(month from :rep_date);
        for select kc_amount_approved
            from knclient
            where kc_amount_approved <> 0 and
                  kc_from_k_id = :k_id and
                  kc_amount_approved is not null and
                  kc_from_pt_id = 'KIVA'
            into :rep_disb_amount
        do
        begin
          rep_month_amount = cast(:rep_amount / :k_amount as decimal(15,7));
          rep_month_amount = cast(:rep_month_amount * :rep_disb_amount as decimal(15,2));
          suspend;
        end
      end
    end
    exit;
  end

  for select k_kontract_no, ok_sh_date, kc_amount_approved, sum(ok_os + ok_sh_os)
      from kontstate, knclient, operkred, kontract
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :report_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = kc_from_k_id and
            kc_amount_approved > 0 and
            kc_from_pt_id = 'KIVA' and
            kc_from_c_id = ok_from_c_id and
            kc_from_k_id = ok_from_k_id and
            ok_ol_is_active = 1 and
            ok_ol_date <= :report_date_end and
            ok_type <= 7 and
            ok_type <> 3
--and ok_from_c_id = 0
            and
            kc_from_k_id = k_id
      group by k_kontract_no, ok_sh_date, kc_amount_approved
      having sum(ok_os + ok_sh_os) > 0
      into rep_kontract_no, rep_date, rep_disb_amount, rep_month_amount
  do
  begin
    rep_year = extract(year from :rep_date);
    rep_month = extract(month from :rep_date);
    suspend;
  end
end
