CREATE PROCEDURE AAREP_VERIFY(
    REGION_ID ID_TYPE,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    ACL_DATE DATE,
    ACL_STATUS VARCHAR(50),
    ACL_DATETIME TIMESTAMP,
    ACL_LAST_DATETIME TIMESTAMP,
    ACL_LIFETIME_IN_SECOND INTEGER,
    ACL_DONT_SHOW_IN_ORDER INTEGER,
    ACL_SOL_TIME INTEGER,
    ACL_UN_REG_FIO VARCHAR(100),
    ACL_FROM_UN_FIO VARCHAR(100),
    ACL_UN_REG_FIO_LAST VARCHAR(100),
    ACL_C_FIO VARCHAR(100),
    PLS_EDIT_KIND D_STR_100,
    REG_NAME VARCHAR(100),
    ACL_MEMO VARCHAR(255),
    C_PASS_WHO VARCHAR(255),
    C_PASS_WHO_CODE VARCHAR(20),
    PK_NAME VARCHAR(100),
    BLOB_ADD_KIND D_STR_100)
AS
declare variable ACL_ID decimal(18,0);
declare variable ACL_PARENT_ID decimal(18,0);
declare variable ACL_FROM_REG_ID integer;
declare variable ACL_OBJECT_KIND integer;
declare variable ACL_OBJECT_ID decimal(18,0);
declare variable ACL_FROM_RD_ID integer;
declare variable ACL_RESPONS_C_ID decimal(18,0);
declare variable ACL_FROM_UN_ID decimal(18,0);
declare variable ACL_RESULT_RD_ID integer;
declare variable ACL_QUOTA_USED integer;
declare variable ACM_MEMO varchar(4000);
declare variable JSON_KEY varchar(255);
declare variable JSON_VALUE varchar(255);
begin
  for select acl_id, acl_parent_id, acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id,
             acl_respons_c_id, acl_datetime, acl_from_un_id, acl_lifetime_in_second, acl_result_rd_id,
             acl_last_datetime, acl_dont_show_in_order, acl_quota_used,
             case acl_status
               when -3 then 'Отмена заполнения'
               when -2 then 'Отказан'
               when -1 then 'На доработку'
               when 0 then 'В ожидании'
               when 1 then 'Принят на обработку'
               when 2 then 'Подтвежден'
             end
      from refdata, actionlist acl, region_list(:region_id, 1, -1, 1)
      where rd_code starting 'ActionGroup.Passport_Verify' and
            rd_id = acl_from_rd_id and
            acl_object_kind = 0 and
            acl_date between :rep_date_begin and :rep_date_end and
            acl.acl_from_reg_id = id_list
      into acl_id, acl_parent_id, acl_date, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id,
           acl_respons_c_id, acl_datetime, acl_from_un_id, acl_lifetime_in_second, acl_result_rd_id, acl_last_datetime,
           acl_dont_show_in_order, acl_quota_used, acl_status
  do
  begin
    acl_sol_time = datediff(minute, acl_datetime, acl_last_datetime);

    acl_from_un_fio = '';

    select c.c_name_fio
    from clients c        --RFCRU-4515
    where c.c_id = :acl_respons_c_id
    into :acl_from_un_fio;

    if (acl_from_un_fio = '') then
    begin
        select un.un_fio
        from username un
        where un.un_id = :acl_from_un_id
        into acl_from_un_fio;
    end

    acl_c_fio = '';
    c_pass_who = null;
    c_pass_who_code = null;
    pk_name = null;
    select c_name_fio, c.c_pass_who, c.c_pass_who_code,
           (select pk_name
            from passkind
            where pk_id = c.c_doc_type rows 1), --
           un_fio
    from clients c, username
    where c_id = :acl_object_id and
          c.c_from_un_id = un_id
    into acl_c_fio, c_pass_who, c_pass_who_code, pk_name, acl_un_reg_fio;

    select
      un_fio,
      f_blob_kind(bls_source_system, bls_kind)
    from clients cl, blobstore bs, username un
    where c_id = :acl_object_id and
          bls_id in (cl.c_blob_pass1_id, c_blob_pass2_id, c_blob_photo_id) and
          bls_datetime <= :acl_last_datetime and
          bs.bls_from_un_id = un_id
    order by bls_datetime desc
    rows 1
    into
      acl_un_reg_fio_last,
      blob_add_kind;


    -- RFCRU-3398 -->
    pls_edit_kind = '';
    
    select
      case aek_editkind
            when -1 then 'Подтверждение без изменений'
            when 0 then 'Исправление ошибки'
            when 1 then  'Смена персональных данных'
        end
    from actioneditkind aek
    where aek.aek_from_acl_id = :acl_id
    into
        :pls_edit_kind;
    -- <

    reg_name = '';
    select reg_name
    from region
    where reg_unicode = :acl_from_reg_id
    into reg_name;

    acm_memo = '';
    acl_memo = '';
    json_key = null;
    json_value = null;
    select acm_memo
    from actionmemo
    where acm_from_acl_id = :acl_id
    into :acm_memo;
    if (coalesce(acm_memo, '') <> '') then
    begin
      for select json_key, json_value
          from fm_json_parse(:acm_memo, 0, 'verify')
          into :json_key, :json_value
      do
      begin
        if (json_key = 'comment') then
          acl_memo = json_value;
      end
    end

    suspend;
  end
end
