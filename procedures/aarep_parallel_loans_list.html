CREATE PROCEDURE AAREP_PARALLEL_LOANS_LIST(
    REP_DATE_END DATE)
RETURNS (
    REP_LINE_NO INTEGER,
    REP_C_ID DECIMAL(18,0),
    REP_LOANS_KOLICH INTEGER,
    REP_CLIENT_FIO VARCHAR(100),
    REP_DISB_DATE DATE,
    REP_KONTRACT_NO VARCHAR(20),
    REP_INDEX_CURR VARCHAR(3),
    REP_DISB_AMOUNT DECIMAL(18,4),
    REP_PERIOD DECIMAL(18,4),
    REP_PERCENT DECIMAL(18,7),
    REP_PRODUCT VARCHAR(100),
    REP_BALANCE DECIMAL(18,4),
    REP_OFFICE VARCHAR(100),
    PR_COMMON_NAME VARCHAR(100),
    REP_BALANCE_EKV DECIMAL(18,4),
    KF_NAME VARCHAR(100),
    REP_DISB_AMOUNT_EKV DECIMAL(18,4),
    REP_REGION VARCHAR(255),
    BT_NAME VARCHAR(50),
    BS_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(60),
    OKRUG_NAME VARCHAR(60),
    KNC_SEX_STR VARCHAR(10),
    KF_OFFICER_NAME VARCHAR(100))
AS
declare variable k_id decimal(18,0);
declare variable def_curr varchar(3);
declare variable curr_rate decimal(18,4);
declare variable kf_id decimal(18,0);
declare variable gr_amount decimal(18,4);
declare variable tmp decimal(18,7);
declare variable kc_busuness_subtype integer;
declare variable kal_new_kf_officer curr_type;
begin
  select cm_default_currency
  from company
  into def_curr;

  curr_rate = 1;
  for select kc_from_c_id, count(*)
      from oksummary, knclient
      where oks_from_k_id = kc_from_k_id and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_balance > 0 and
            oks_close_state = 0
      group by kc_from_c_id
      having count(*) > 1
      into :rep_c_id, :rep_loans_kolich
  do
  begin
    select c_name_fio,
           case c_sex
             when 0 then 'Женщина'
             when 1 then 'Мужчина'
             when 2 then 'Юр.лицо'
           end
    from clients
    where c_id = :rep_c_id
    into :rep_client_fio, knc_sex_str;

    rep_balance = 0;
    rep_line_no = 0;
    for select k_id, k_from_kf_id, ks_date_begin, k_kontract_no, k_index_curr_code, kc_amount_approved, k_period,
               k_percent, pr_name, pr_common_name, k_amount_approved, kc_busuness_subtype
        from kontract, knclient, kontstate, product, oksummary
        where k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_date_begin <= :rep_date_end and
              ks_date_end >= :rep_date_end and
              ks_state = 1 and
              kc_from_c_id = :rep_c_id and
              kc_from_k_id = k_id and
              kc_amount_approved <> 0 and
              kc_amount_approved is not null and
              ks_ol_is_delete = 0 and
              k_from_pr_id = pr_id and
              oks_from_k_id = k_id and
              oks_date_begin <= :rep_date_end and
              oks_date_end >= :rep_date_end and
              oks_balance > 0 and
              oks_close_state = 0
        order by ks_date_begin
        into :k_id, :kf_id, :rep_disb_date, :rep_kontract_no, :rep_index_curr, :rep_disb_amount, :rep_period,
             :rep_percent, :rep_product, pr_common_name, :gr_amount, :kc_busuness_subtype
    do
    begin
      rep_line_no = rep_line_no + 1;
      select r1.reg_code || ' ' || r1.reg_name, kal_new_kf_name, r1.reg_stat_rep_region,
             r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name, kal_new_kf_officer
      from knaktlist, region r1, region r2, region r3
      where kal_new_reg_id = r1.reg_unicode and
            kal_from_kf_id = :kf_id and
            kal_date <= :rep_date_end and
            kal_date_end >= :rep_date_end and
            knaktlist.is_delete = 0 and
            r1.parent = r2.reg_unicode and
            r2.parent = r3.reg_unicode
      into :rep_office, :kf_name, :rep_region, :branch_name, :okrug_name, kal_new_kf_officer;

      kf_officer_name = (select c_name_fio
                         from clients
                         where c_id = :kal_new_kf_officer);

      rep_balance = 0;
      select oks_balance
      from oksummary
      where oks_from_k_id = :k_id and
            oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_close_state = 0
      into :rep_balance;
      if (rep_balance is null) then
        rep_balance = 0;

      tmp = rep_balance / gr_amount;
      rep_balance = cast(tmp * rep_disb_amount as decimal(15,2));

      if (rep_index_curr <> def_curr) then
      begin
        select cr_rate
        from opercurr
        where cr_date <= :rep_date_end and
              cr_date_end >= :rep_date_end and
              cr_from_cu_id = :rep_index_curr
        into :curr_rate;
        if (curr_rate is null) then
          curr_rate = 1;
        rep_balance_ekv = cast(rep_balance * curr_rate as decimal(15,2));

        select cr_rate
        from opercurr
        where cr_date <= :rep_disb_date and
              cr_date_end >= :rep_disb_date and
              cr_from_cu_id = :rep_index_curr
        into :curr_rate;
        if (curr_rate is null) then
          curr_rate = 1;
        rep_disb_amount_ekv = cast(rep_disb_amount_ekv * curr_rate as decimal(15,2));
      end
      else
      begin
        rep_balance_ekv = rep_balance;
        rep_disb_amount_ekv = rep_disb_amount;
      end

      -- Вытаскивает тип бизнеса клиента
      bt_name = null;
      bs_name = null;
      select bt_name, bs_name
      from bussubtype, bustype
      where bs_from_bt_id = bt_id and
            bs_id = :kc_busuness_subtype
      into :bt_name, :bs_name;

      suspend;
    end
  end
end
