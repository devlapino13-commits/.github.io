CREATE PROCEDURE PRINT_EPS_ADD_AGREEMENT(
    KC_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    K_KONTRACT_NO D_STR_100,
    KS_FIO D_STR_100,
    CLIENT_FIO D_STR_100,
    PASS_NO D_STR_255,
    PASS_DATE D_DATE,
    PASS_WHO D_STR_255,
    EDOC_DATE D_DATE,
    QR_CODE D_STR_LONG,
    K_DISB_DATE D_DATE,
    CL_ADDRESS D_STR_255,
    CL_EMAIL D_STR_100,
    CL_PHONE D_STR_100,
    OFFICE_NAME D_STR_100,
    NEW_EFF_PERC PERC_TYPE,
    NEW_EFF_PERC1 PERC_TYPE,
    NEW_EFF_PERC2 PERC_TYPE)
AS
declare variable EDL_ID D_ID;
declare variable C_ID D_ID;
declare variable REGION_ID D_INTEGER;
declare variable K_ID D_ID;
declare variable KF_ID D_ID;
begin
  select kont.k_kontract_no, cl.c_name_fio, kncl.kc_from_c_id, kont.k_disb_date, cl.c_pass_no, cl.c_pass_date,
         cl.c_pass_who, kont.k_id, kont.k_from_kf_id, round(kont.k_percent_effect, 2)
  from knclient kncl
  inner join kontract kont on kont.k_id = kncl.kc_from_k_id
  inner join geps_agreement_card gs on gs.gs_from_kc_id = kncl.kc_id
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  where kncl.kc_id = :kc_id
  into :k_kontract_no, :client_fio, :c_id, :k_disb_date, :pass_no, :pass_date, :pass_who, :k_id, :kf_id, :new_eff_perc;

  select round(calc_eff, 2)
  from hp_calc_eps_float_schedule(:k_id, 1) --   эффективная процентная ставка, при просрочке больше 2 дней
  into new_eff_perc1;

  select round(calc_eff, 2)
  from hp_calc_eps_float_schedule(:k_id, 2) --   эффективная процентная ставка, при просрочке больше 7 дней
  into new_eff_perc2;

  cl_email = f_cont_info('clients', :c_id, 'email', 'ok');
  cl_phone = f_cont_info('clients', :c_id, 'cell_main', 'ok');

  select adl.adl_address_str
  from addrclient adcl
  inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
  where adcl.adc_from_c_id = :c_id and
        adcl.adc_address_type = 1
  into :cl_address;

  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(add_params, 'edl_id', null);

    select cast(edl.edl_create_datetime as date)
    from edoc_list edl
    where edl.edl_id = :edl_id
    into :edoc_date;

    select reg.reg_name, reg.reg_unicode
    from knaktlist kal
    inner join region reg on reg.reg_unicode = kal.kal_new_reg_id
    where kal.is_delete = 0 and
          :edoc_date between kal.kal_date and kal.kal_date_end and
          kal.kal_from_kf_id = :kf_id
    into :office_name, :region_id;

    select cl.c_name_fio
    from hr_pstatus hps
    inner join joblist jl on ps_from_jl_id = jl_id
    inner join jobkind jk on jl.jl_kind = jk.jk_id
    inner join clients cl on cl.c_id = hps.ps_from_c_id
    where :edoc_date between ps_date_begin and ps_date_end and
          ps_status = 1 and -- утвержденная операция
          ps_is_working = 1 and --активные сотрудники
          jk.jk_system_code = 'JobSystem.Region_KM' and
          hps.ps_from_reg_id = :region_id and
          not exists(select *
                     from hr_voctravel, hr_orders, sa_worktype
                     where hrv_from_ho_id = ho_id and
                           ho_order_type = sawt_id and
                           hrv_from_c_id = cl.c_id and
                           hrv_date_begin <= :edoc_date and
                           hrv_date_end_fact >= :edoc_date and
                           hrv_status = 1 and
                           sawt_base_type = 5 and
                           sawt_include_holiday = 1 and
                           sawt_is_paid = 0 /* декретный отпуск */
                           rows 1)
    order by c_id
    rows 1
    into :ks_fio; -- пишем ФИО КМ

    if (ks_fio is null) then
      ks_fio = 'Джумабекова Бактыгуль Ниязахуновна';

    qr_code = f_edoc_qr_by_edl(edl_id, c_id);
  end
  suspend;
end
