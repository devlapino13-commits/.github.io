CREATE PROCEDURE CC_LAWYER_QESTION_LIST
RETURNS (
    CP_ID INTEGER,
    CPC_NO INTEGER,
    COUNTER INTEGER,
    CPC_QESTION_DATE DATE,
    CPC_QESTION_OPER STR_100,
    CPC_BLOCKED_OPER_ID ID_TYPE,
    CPC_BLOCKED_OPER STR_100,
    CPC_QESTION_TEXT STR_255,
    CPC_ANSWER_TEXT STR_255,
    IS_CLIENT SMALLINT)
AS
begin
  counter = 0;

  for select cpc.cpc_from_cp_id, cpc.cpc_no, cpc.cpc_qestion_date,

             un.un_name as cpc_qestion_oper, cpc.cpc_blocked_oper_id, left(cpc.cpc_qestion_text, 150),

             left(cpc.cpc_answer_text, 150),
             (
             case
               when cp.cp_from_c_id is null then 0
               else 1
             end)
      from cc_person_calls cpc
      inner join cc_persons cp on cp.cp_id = cpc.cpc_from_cp_id
      inner join username un on un.un_id = cpc.cpc_qestion_oper_id
      where cpc.cpc_qestion_open = 1 and
            cpc.cpc_to_lawyer = 1 and
            cpc.cpc_no > 0
      order by cpc.cpc_qestion_date, cpc.cpc_from_cp_id, cpc.cpc_no
      into :cp_id, :cpc_no, :cpc_qestion_date, :cpc_qestion_oper, :cpc_blocked_oper_id, :cpc_qestion_text,
           :cpc_answer_text, :is_client
  do
  begin
    cpc_blocked_oper = '';
    if (cpc_blocked_oper_id is not null) then
    begin
      select un.un_name
      from username un
      where un.un_id = :cpc_blocked_oper_id
      into :cpc_blocked_oper;
    end

    counter = counter + 1;

    cpc_answer_text = replace(cpc_answer_text, ascii_char(13), ' ');
    cpc_answer_text = replace(cpc_answer_text, ascii_char(10), ' ');
    cpc_answer_text = trim(cpc_answer_text);

    cpc_answer_text = replace(cpc_answer_text, ascii_char(13), ' ');
    cpc_answer_text = replace(cpc_answer_text, ascii_char(10), ' ');
    cpc_answer_text = trim(cpc_answer_text);

    if (cpc_qestion_text <> '') then
    begin
      if (char_length(cpc_qestion_text) > 100) then
        cpc_qestion_text = left(cpc_qestion_text, 100) || '...';
    end

    suspend;
  end
end
