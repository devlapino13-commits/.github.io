CREATE PROCEDURE HP_EPAY_DOC_DISB(
    EPD_ID D_INTEGER NOT NULL,
    UN_ID D_ID)
RETURNS (
    RES_IS_ERROR D_SMALLINT,
    RES_TEXT D_STR_255)
AS
declare variable epr_code d_str_code;
declare variable epr_fix_amount d_smallint;
declare variable epl_status d_smallint;
declare variable epr_id d_integer;
declare variable main_ol_id d_id;
declare variable epl_id d_id;
declare variable trans_ol_id d_id;
declare variable epay_amount d_currency;
declare variable resident d_integer;
declare variable c_id d_id;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable epd_date_end d_date;
declare variable epd_amount d_currency;
declare variable epd_status d_smallint;
declare variable epd_from_epl_id d_integer;
declare variable epay_acc_id d_integer;
declare variable epay_tax_fl d_currency;
declare variable epay_payed_year d_currency;
declare variable rep_year_begin d_date;
declare variable ol_memo d_str_long;
declare variable ept_id d_id;
begin
  res_is_error = 1;
  res_text = '';

  if (un_id is null) then un_id = 1000000000000;

  rep_year_begin = cast('01.01.' || extract(year from current_date) as date);

  select epr_id, epr_code, epr_fix_amount, epd_from_c_id, epd_object_kind, epd_object_id,
         epd_date_end, epd_amount, epd_status, epd_from_epl_id
  from epay_project, epay_doc
  where epd_id = :epd_id and
        epr_id = epd_from_epr_id
  into epr_id, epr_code, epr_fix_amount, c_id, obj_kind, obj_id,
       epd_date_end, epd_amount, epd_status, epd_from_epl_id;

  if (epr_code is null) then
    res_text = 'Запись об оплате не найдена, EPD_ID: ' || epd_id;

  if (res_text = '' and epd_date_end < current_date) then
    res_text = 'Срок актуальности проверки документов истёк';

  if (res_text = '' and epd_status <> 1) then
    res_text = 'Оплата не одобрена';

  if (res_text = '' and epr_fix_amount = 1 and epd_amount = 0) then
    res_text = 'Нулевая сумма оплаты';

  if (res_text = '' and epr_fix_amount = 0) then
  begin
    epd_amount = 0;
    epay_payed_year = 0;
    select sum(ogr_amount * ogr_sign * f_loyal(ogr_from_ol_id, current_date) + ogr_tax_fl * ogr_sign * f_loyal(ogr_from_ol_id, current_date)),
           sum(iif(ogr_sign = -1 and ogr_object_kind is null and ogr_ol_date >= :rep_year_begin, ogr_amount + ogr_tax_fl, 0))
    from operagr
    where ogr_from_ag_id = :obj_id and
          ogr_ol_is_active = 1
    into epd_amount, epay_payed_year;
    if (epd_amount is null) then
      epd_amount = 0;

    if (epay_payed_year is null) then
      epay_payed_year = 0;

    if (epd_amount = 0) then
      res_text = 'Нулевая сумма оплаты';
  end

  if (res_text = '' and epd_from_epl_id is not null) then
  begin
    select epl_status
    from epay_list
    where epl_id = :epd_from_epl_id
    into epl_status;

    res_text = trim('Запись об оплате ранее создана, статус: ' ||
    case epl_status
    when 1 then 'Оплачено успешно'
    when 0 then 'В процессе'
    when -1 then 'Ошибка'
    end);
  end

  if (res_text = '') then
  begin
    main_ol_id = null;
    trans_ol_id = null;
    epay_acc_id = null;

    select epa_id
    from epay_acc
    where epa_from_c_id = :c_id and
          epa_status = 1
    order by epa_id desc
    rows 1
    into epay_acc_id;

    if (epay_acc_id is null) then
      res_text = 'Банковская карта не привязана';
  end

  if (res_text = '') then
  begin
    select resident
    from aarep_epay_client_info(:c_id, current_date)
    into :resident;

    epay_amount = 0;
    epay_tax_fl = 0;

    select epay_new_amount, epay_new_tax
    from hp_epay_local_calc_tax(:obj_id, :resident, current_date, 0)
    into :epay_amount, :epay_tax_fl;

    ol_memo = 'Выплата вознаграждения по программе лояльности';

    insert into operlist (ol_from_reg_id, ol_oper_type, ol_oper_subtype, ol_date, ol_doc_kind, ol_memo, ol_parent_id,
                          is_delete, ol_user_id)
    values (1010100, 2, 37, current_date, 9, :ol_memo, null, 0, :un_id)
    returning ol_id
    into :main_ol_id;
    trans_ol_id = main_ol_id;

    insert into epay_list (epl_from_epr_id, epl_object_kind, epl_object_id, epl_date, epl_status, epl_from_ol_id)
    values (:epr_id, :obj_kind, :obj_id, current_date, 0, :main_ol_id)
    returning epl_id
    into :epl_id;

    insert into epay_trans (ept_from_epl_id, ept_from_c_id, ept_from_epa_id, ept_amount, ept_status, ept_from_ol_id)
    values (:epl_id, :c_id, :epay_acc_id, :epay_amount, 0, :trans_ol_id)
    returning ept_id
    into :ept_id;

    select res_text
    from hp_epay_list_moneta_disb(:ept_id)
    into res_text;
  end

  if (res_text = '') then
  begin
    update epay_doc
    set epd_from_epl_id = :epl_id
    where epd_id = :epd_id;

    insert into operagr (ogr_from_ol_id, ogr_from_ag_id, ogr_sign, ogr_amount, ogr_tax_fl, ogr_ol_date, ogr_ol_is_active)
    values (:trans_ol_id, :obj_id, -1, :epay_amount, :epay_tax_fl, current_date, 1);

    -- Строим бух.проводки по Минфин и ЕПС
    execute procedure hp_epay_doc_disb_operbook(:trans_ol_id);

    res_text = 'Выплата на карту успешно создана';
    res_is_error = 0;
  end

  suspend;
end
