CREATE PROCEDURE AAREP_SITE_MINIAPP(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    CK_CHECKED_DATETIME TIMESTAMP,
    CK_CHECKED_DATE DATE,
    CK_CHECKED_OPER STR_100,
    OPER_MEMO VARCHAR(1000),
    OPER_CL_FIO STR_255,
    OPER_CL_BIRTH_DATE DATE,
    CK_CRED_AMOUNT CURR_TYPE,
    CK_LEGAL_DOCS STR_255,
    CK_NEXT_CALL_DATETIME TIMESTAMP,
    CK_PERIOD STR_100,
    CALL_RES STR_100,
    OPER_CITIZENSHIP STR_100,
    OPER_CL_CITY STR_100,
    JOB_STATUS STR_100,
    OPER_REG_NAME STR_100,
    OPER_LO_NAME STR_100,
    WMA_CL_NAME STR_100,
    WMA_CL_BIRTHDAY DATE,
    WMA_CL_CITIZENSHIP STR_100,
    WMA_CL_GENDER STR_100,
    WMA_CL_LOAN_TERM STR_100,
    WMA_PHONE_NO STR_100,
    WMA_CL_LOAN_AMOUNT STR_100,
    WMA_CITY STR_100,
    WMA_DATE DATE,
    WMA_DATETIME TIMESTAMP,
    ESM_ID VARCHAR(18))
AS
declare variable ck_citizenship integer;
declare variable ck_status integer;
declare variable ck_from_esm_id id_type;
declare variable ck_city id_type;
declare variable ck_job_status integer;
declare variable ck_lo_id id_type;
declare variable ck_from_reg_id integer;
declare variable esv_value_txt str_255;
declare variable esf_field_code str_100;
declare variable ck_parent_id id_type;
begin
  for select chk.ck_checked_datetime, chk.ck_checked_date, chk.ck_checked_oper,

             chk.ck_memo, ck_status, ck_from_esm_id,

             ck_fio, chk.ck_citizenship, chk.ck_birth_date,

             chk.ck_city, ck_cred_amount, ck_legal_docs,

             ck_job_status, ck_next_call_datetime, ck_lo_id,

             ck_from_reg_id, chk.ck_period, chk.ck_parent_id
      from cc_check_knclient chk
      where chk.ck_type = 6 and
            chk.ck_checked_date between :rep_date_begin and :rep_date_end
      order by chk.ck_checked_datetime
      into :ck_checked_datetime, :ck_checked_date, :ck_checked_oper,

           :oper_memo, :ck_status, :ck_from_esm_id,

           :oper_cl_fio, :ck_citizenship, :oper_cl_birth_date,

           :ck_city, :ck_cred_amount, :ck_legal_docs,

           :ck_job_status, :ck_next_call_datetime, :ck_lo_id,

           :ck_from_reg_id, :ck_period, :ck_parent_id
  do
  begin
    oper_memo = replace(oper_memo, ascii_char(13), '');
    oper_memo = replace(oper_memo, ascii_char(10), '');

    if (ck_parent_id is not null) then
    begin
      select ck_from_esm_id, ck_fio, chk.ck_citizenship,

             chk.ck_birth_date, chk.ck_city, ck_cred_amount,

             ck_legal_docs, ck_job_status, chk.ck_period
      from cc_check_knclient chk
      where chk.ck_id = :ck_parent_id rows 1
      into :ck_from_esm_id, :oper_cl_fio, :ck_citizenship,

           :oper_cl_birth_date, :ck_city, :ck_cred_amount,

           :ck_legal_docs, :ck_job_status, :ck_period;
    end

    esm_id = cast(esm_id as varchar(18));

    if (ck_status = -1) then
      call_res = 'не дозвон';
    else
    if (ck_status = -2) then
      call_res = 'не интресуется';
    else
    if (ck_status = 1) then
      call_res = 'заинтересован (перезвонить)';
    else
    if (ck_status = 2) then
      call_res = 'заинтересован (конечный статус)';

    oper_citizenship = '';
    if (ck_citizenship is not null) then
    begin
      select rd.rd_name
      from refdata rd
      where rd.rd_id = :ck_citizenship
      into :oper_citizenship;
    end

    oper_cl_city = '';
    if (ck_city is not null) then
    begin
      select pl.pl_name
      from places pl
      where pl.pl_id = :ck_city
      into :oper_cl_city;
    end

    if (ck_job_status = 1) then
      job_status = 'работает';
    else
      job_status = 'безработный';

    oper_lo_name = '';
    if (ck_lo_id is not null) then
    begin
      select cl.c_name_fio
      from clients cl
      where cl.c_id = :ck_lo_id
      into :oper_lo_name;
    end

    oper_reg_name = '';
    if (ck_from_reg_id is not null) then
    begin
      select reg.reg_name
      from region reg
      where reg.reg_unicode = :ck_from_reg_id
      into :oper_reg_name;
    end

-----
    wma_cl_name = '';
    wma_cl_birthday = null;
    wma_cl_citizenship = null;
    wma_cl_gender = '';
    wma_cl_loan_term = '';
    wma_phone_no = '';
    wma_cl_loan_amount = 0;
    wma_city = '';

    for select ef.esf_field_code, ev.esv_value_txt, em.esm_date, em.esm_datetime
        from extsysmain em, extsysfields ef, extsysvalues ev
        where em.esm_project = ef.esf_project and
              em.esm_project = 'website' and
              ev.esv_from_esm_id = em.esm_id and
              ev.esv_from_esf_id = ef.esf_id and
              em.esm_status = 0 -- если = 0 значит успешная команда
              and
              em.esm_id = :ck_from_esm_id and
              esm_action = 'save_cl_info'
        into :esf_field_code, :esv_value_txt, :wma_date, :wma_datetime
    do
    begin
      if (esf_field_code = 'cl_name') then
        wma_cl_name = esv_value_txt;
      if (esf_field_code = 'cl_birthday') then
        wma_cl_birthday = cast(esv_value_txt as date);
      if (esf_field_code = 'cl_citizenship') then
        wma_cl_citizenship = esv_value_txt;
      if (esf_field_code = 'cl_gender') then
        wma_cl_gender = esv_value_txt;
      if (esf_field_code = 'cl_loan_term') then
        wma_cl_loan_term = esv_value_txt;
      if (esf_field_code = 'cl_loan_amount') then
        wma_cl_loan_amount = esv_value_txt;
      if (esf_field_code = 'phone_no') then
        wma_phone_no = esv_value_txt;
      if (esf_field_code = 'cl_city') then
        wma_city = esv_value_txt;
    end

    suspend;
  end
end
