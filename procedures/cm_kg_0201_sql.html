CREATE PROCEDURE CM_KG_0201_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable sh_grace integer;
declare variable sh_grace_temp integer;
declare variable sh_date d_date;
declare variable point d_integer;
declare variable k_id d_id;
begin
  if (exists(select *
             from miniappclient mcl
             inner join miniapp mp on mp.ma_id = mcl.mac_from_ma_id
             inner join product pr on pr.pr_id = mp.ma_from_pr_id
             where ma_id = :ma_id and
                   ma_is_active = 1 and
                   ma_date_end = '31.12.9999' and
                   mac_amount > 0 and
                   pr_code containing '.jonokoi.')) then
  begin
    sh_grace = 0;
    select k.k_id
    from kontract k
    where k.k_from_ma_app_id = :ma_id
    into k_id;

    for select sc.sh_grace, sc.sh_date
        from schedule sc
        where sc.sh_from_k_id = iif(coalesce(:k_id, 0) > 0, :k_id, -:ma_id)
        into sh_grace_temp, sh_date
    do
    begin
      sh_grace = :sh_grace + :sh_grace_temp;
      if (extract(month from :sh_date) not in (1) and
          :sh_grace_temp = 1) then
      begin
        res_error_text = 'По данному продукту льготный период можно получить только на январь.';
        res_error_is_critical = 1;
        suspend;
        exit;
      end
    end
    if (:sh_grace = 0) then
    begin
      res_error_text = 'Льготный период не указан';
      res_error_is_critical = 1;
      suspend;
      exit;
    end
    if ((select f_point(mac.mac_from_c_id, ma.ma_date_oper)
         from miniappclient mac
         inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
         where mac_from_ma_id = :ma_id and
               mac_amount > 0 and
               ma.ma_is_active = 1 and
               ma.ma_date_end >= '31.12.9999') < 3) then
    begin
      res_error_text = 'По данному продукту у клиентка балл должен быть выше 3';
      res_error_is_critical = 1;
      suspend;
      exit;
    end
  end
end
