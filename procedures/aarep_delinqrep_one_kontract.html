CREATE PROCEDURE AAREP_DELINQREP_ONE_KONTRACT(
    K_ID DECIMAL(18,0))
RETURNS (
    DQ_ID INTEGER,
    DQ_DATE DATE,
    DQ_FROM_K_ID DECIMAL(18,0),
    DQ_FROM_C_ID DECIMAL(18,0),
    DQ_CATEGORY VARCHAR(255),
    DQ_MEMO VARCHAR(1024),
    DQ_BLOB BLOB SUB_TYPE BINARY SEGMENT SIZE 80,
    DQ_FROM_UN_ID DECIMAL(18,0),
    DQ_DATETIME TIMESTAMP,
    DQ_FROM_DR_ID INTEGER,
    DQ_REPAYMENT_RATE INTEGER,
    DQ_CURRENT_ADDRESS VARCHAR(255),
    DQ_CURRENT_PHONE VARCHAR(255),
    DQ_DR_NAME VARCHAR(255),
    DQ_DR_CATEGORY VARCHAR(255),
    DQ_USER_FIO VARCHAR(100),
    DQ_CLIENT_FIO VARCHAR(100),
    DQ_OKRUG VARCHAR(100),
    DQ_COUNTRY VARCHAR(100))
AS
declare variable c_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select kc_from_c_id, c_name_fio
      from knclient, clients
      where kc_from_k_id = :k_id and
            kc_amount_approved is not null and
            kc_amount_approved > 0 and
            kc_from_c_id = c_id
      into :dq_from_c_id, :dq_client_fio
  do
  begin
    dq_id = null;
    dq_date = null;
    dq_category = null;
    dq_memo = null;
    dq_user_fio = null;
    dq_datetime = null;
    dq_dr_name = null;
    dq_dr_category = null;
    dq_repayment_rate = null;
    dq_current_address = null;
    dq_current_phone = null;

    select dq_id, dq_date, dq_category, dq_memo, c_name_fio, dq_datetime, dr_name, dr_category, dq_repayment_rate,
           dq_current_address, dq_current_phone, okl_name, rd_name
    from delinqrep, delinqrating, clients, username, okruglist, refdata
    where dq_from_c_id = :dq_from_c_id and
          dq_from_dr_id = dr_id and
          dq_from_un_id = un_id and
          un_from_c_id = c_id and
          dq_okrug = okl_id and
          dq_country = rd_id
    order by dq_id desc
    rows 1
    into :dq_id, :dq_date, :dq_category, :dq_memo, :dq_user_fio, :dq_datetime, :dq_dr_name, :dq_dr_category,
         :dq_repayment_rate, :dq_current_address, :dq_current_phone, :dq_okrug, :dq_country;

    suspend;
  end
end
