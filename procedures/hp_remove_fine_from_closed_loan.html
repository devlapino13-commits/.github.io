CREATE PROCEDURE HP_REMOVE_FINE_FROM_CLOSED_LOAN(
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    K_ID ID_TYPE,
    K_KONTRACT_NO STR_100)
AS
declare variable ol_id id_type;
declare variable sh_id integer;
declare variable sh_date date;
declare variable calc_fine curr_type;
declare variable ok_number integer;
begin
-- получаю спсиок закрытых займов за период
  for select ks.ks_from_k_id, ks.ks_from_ol_id, kont.k_kontract_no
      from kontstate ks
      inner join kontract kont on kont.k_id = ks.ks_from_k_id
      where ks_state = 2 and
            ks.ks_date_begin between :date_begin and :date_end and
            ks.ks_ol_is_delete = 0
      into :k_id, :ol_id, :k_kontract_no
  do
  begin
-- если ест активная задолженность по пеням по закрытому займу, убираю пени чз ok_type = 3
    if (exists(select *
               from aarep_make_zad_list(:k_id, current_date, 1)
               where sh_calc_fine > 0)) then
    begin
      for select sh_id, sh_date, sh_calc_fine
          from aarep_make_zad_list(:k_id, current_date, 1)
          where sh_calc_fine > 0
          into :sh_id, :sh_date, :calc_fine
      do
      begin
        select max(ok.ok_number)
        from operkred ok
        where ok.ok_from_ol_id = :ol_id
        into :ok_number;

        ok_number = ok_number + 1;

        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax)
        values (:ol_id, :ok_number, :k_id, :sh_id, :sh_date, 3, 0, 0, :calc_fine * -1, 3, 0, 0, 0, 0, 0);
      end
      suspend;
    end
  end
end
