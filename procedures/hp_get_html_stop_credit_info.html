CREATE PROCEDURE HP_GET_HTML_STOP_CREDIT_INFO(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID)
RETURNS (
    RES_HTML D_STR_2048)
AS
declare variable STATUS_STOP_CREDIT D_SMALLINT;
declare variable STATUS_TEXT_STOP_CREDIT D_STR_255;
declare variable QUERY_DATE_STOP_CREDIT D_DATE;
declare variable QUERY_DATETIME_STOP_CREDIT D_DATETIME;
declare variable ANSWER_DATE_STOP_CREDIT D_DATE;
declare variable LINK_STOP_CREDIT D_STR_LONG;
declare variable LINK_STOP_CREDIT_REPEAT D_STR_LONG;
declare variable STOP_CREDIT_MAX_QUERIES D_INTEGER;
declare variable STOP_CREDIT_QUERY_COUNT D_INTEGER;
declare variable STOP_CREDIT_STATUS_IS_ACTUAL D_BOOL;
declare variable SERVICE_MAX_DELAY_SECONDS D_INTEGER;
declare variable IS_STOP_CREDIT_DOC_LOADED D_INTEGER;
declare variable TEXT_TMP D_STR_255;
begin
  -- RFCKG-4392, RFCKZ-23
  if (f_get_const('stop_credit_is_active') = 0) then
    exit;

  text_tmp = '';
  res_html = '';

  select status, query_date, query_datetime, answer_date
  from hp_stop_credit_query_status(:obj_kind, :obj_id)
  into :status_stop_credit, :query_date_stop_credit, :query_datetime_stop_credit, :answer_date_stop_credit;

  status_text_stop_credit = (case status_stop_credit
                   when 0 then 'У клиента не установлен стоп-кредит.'
                   when 1 then 'У клиента действует стоп-кредит, выдача невозможна'
                   when -1 then 'Запрос не отправлен'
                   when -2 then 'Запрос отправлен'
                   when -3 then 'Внутренняя ошибка сервиса'
                   else ''
                 end);

  stop_credit_status_is_actual = f_is_stopcred_actual(:answer_date_stop_credit);
  service_max_delay_seconds = cast(f_get_const('stop_credit_service_max_delay_seconds') as d_integer);
  is_stop_credit_doc_loaded = cast(f_is_stopcred_doc_loaded(:obj_kind, :obj_id) as d_integer);
  link_stop_credit = f_uri_sql('Проверить', 'stop_credit_status', :obj_kind, :obj_id, 'none', 'Подтверждаете отправку запроса для проверки статуса стоп-кредита?', '', 0, 0, '', '', 'dialog', '');
  link_stop_credit_repeat = f_uri_sql('Проверить повторно', 'stop_credit_status', :obj_kind, :obj_id, 'none', 'Подтверждаете отправку запроса для проверки статуса стоп-кредита?', '', 0, 0, '', '', 'dialog', '');

  if (status_stop_credit = -1) then
    begin
        -- На уровне контракта нет автоматического триггера отправки запроса, он должен отправляться вручную
        if (obj_kind = f_objkind('kontract', 'id')) then
          begin
              -- Если документ был прикреплен на этапе заявки (и он все еще актуален), то отправлять запрос на уровне контракта нет смысла
              if (is_stop_credit_doc_loaded = 1) then
                begin
                    res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
                    res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                      'Прикреплен документ',
                                                                      '', '', -1, -1, -1, -1, 2, ''));
                end
              else
                begin
                    res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="40%"><col width="20%">');
                    res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                      f_format('<span style="color: red;">%1</span>', status_text_stop_credit),
                                                                      link_stop_credit,
                                                                      -1, -1, -1, -1, -1, 3, ''));
                end
          end
        else
          begin
              res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
              res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                :status_text_stop_credit,
                                                                '', '', -1, -1, -1, -1, 2, ''));
          end
    end
  else if (status_stop_credit = -2) then
    begin
          if (query_datetime_stop_credit is not null and
              datediff(second, query_datetime_stop_credit, current_timestamp) >= service_max_delay_seconds) then
            begin
                res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
                res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                  '<span style="color: red;">Сервис не отвечает. Приложите документ</span>',
                                                                  '', '', -1, -1, -1, -1, 2, ''));
            end
          else
            begin
                res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
                res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                  :status_text_stop_credit,
                                                                  '', '', -1, -1, -1, -1, 2, ''));
            end
    end
  else if (status_stop_credit = 1) then
    begin
        -- Если дата ответа актуальна, то статус актуальный и кнопку "Проверить повторно" выводить не нужно
        if (stop_credit_status_is_actual = 1) then
          begin
              res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
              res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                f_format('<span style="color: red;">%1</span>', status_text_stop_credit),
                                                                '', '', -1, -1, -1, -1, 2, ''));
          end
        else -- Если дата ответа НЕ актуальна, то нужно сделать запрос заново, т.к. результат уже не актуален
          begin
              res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="40%"><col width="20%">');
              res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                f_format('<span style="color: red;">%1</span>', status_text_stop_credit),
                                                                f_format('Дата проверки: %1<br>%2', f_datetostr(answer_date_stop_credit), link_stop_credit_repeat),
                                                                -1, -1, -1, -1, -1, 3, ''));
          end
    end
  else if (status_stop_credit = 0) then
    begin
        if (stop_credit_status_is_actual = 1) then
          begin
              res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
              res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                status_text_stop_credit,
                                                                '', '', -1, -1, -1, -1, 2, ''));
          end
        else
          begin
              res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="40%"><col width="20%">');
              res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                status_text_stop_credit,
                                                                f_format('Дата проверки: %1<br>%2', f_datetostr(answer_date_stop_credit), link_stop_credit_repeat),
                                                                -1, -1, -1, -1, -1, 3, ''));
          end
    end
  else if (status_stop_credit = -3) then
    begin
        -- Получаем количество ошибочных запросов к сервису стоп-кредит для пары object_kind/object_id
        select count(eil.eil_id)
        from extinfolist eil
        where eil.eil_object_kind = :obj_kind
            and eil.eil_object_id = :obj_id
            and eil.eil_from_eic_id = 4 -- интересует только стоп-кредит
            and eil.eil_status in (-2, -3) -- Ошибка
            and eil.eil_date = :query_date_stop_credit
        into :stop_credit_query_count;

        stop_credit_max_queries = cast(f_get_const('stop_credit_max_queries') as integer);

        if (stop_credit_query_count <= stop_credit_max_queries) then
          begin
              res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="40%"><col width="20%">');
              res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                f_format('<span style="color: red;">%1</span>', status_text_stop_credit),
                                                                link_stop_credit,
                                                                -1, -1, -1, -1, -1, 3, ''));
          end
        else
          begin
              if (is_stop_credit_doc_loaded = 1) then
                text_tmp = 'Ошибка при получении статуса клиента. Документ прикреплен';
              else
                text_tmp = f_format('<span style="color: red;">%1</span>', 'Ошибка при получении статуса клиента. Прикрепите документ');

              if (stop_credit_status_is_actual = 1) then
                begin
                    res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
                    res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                       text_tmp,
                                                                        '', '', -1, -1, -1, -1, 2, ''));
                end
              else
                begin
                    res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="40%"><col width="20%">');
                    res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                                        text_tmp,
                                                                        f_format('Дата проверки: %1<br>%2', f_datetostr(answer_date_stop_credit), link_stop_credit_repeat),
                                                                        -1, -1, -1, -1, -1, 3, ''));
                end
          end
    end
  else
    begin
        res_html = blob_append(res_html, '<table width="100%" style="margin: 0; padding 0;"><col width="40%"><col width="60%">');
        res_html = blob_append(res_html, f_html_table_row('Стоп Кредит',
                                                          f_format('<span style="color: red;">%1</span>', 'Произошла неизвестная ошибка'),
                                                          '', '', -1, -1, -1, -1, 2, ''));
    end


  res_html = blob_append(res_html, '</table>');
    
  suspend;
end
