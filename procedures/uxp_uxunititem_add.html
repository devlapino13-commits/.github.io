CREATE PROCEDURE UXP$UXUNITITEM_ADD(
    UNIT_CODE D_STR_CODE NOT NULL,
    PARENT_UUID D_UUID,
    JSON_PATH D_STR_LONG,
    CLASS_TYPE D_STR_CODE NOT NULL,
    CLASS_NAME D_STR_CODE,
    CLASS_PATH D_STR_4096,
    PROP_NAMES D_STR_4096 NOT NULL,
    PROP_VALUES D_STR_4096,
    UNI_POSITION D_INTEGER)
RETURNS (
    UNI_UUID D_UUID)
AS
declare variable unt_uuid d_uuid;
declare variable prop_name d_str_100;
declare variable cls_name d_str_code;
declare variable cls_parent d_str_code;
declare variable last_class d_str_code;
declare variable cls_visual d_smallint;
declare variable prop_value d_str_255;
declare variable prop_no d_integer;
declare variable prop_val_no d_integer;
begin
  select unt_uuid
  from ux$unit
  where unt_code = :unit_code
  into unt_uuid;

  if (unt_uuid is null) then
    exception e_message f_format('Unit not found: %1', unit_code);

  last_class = '';
  for select res_str
  from uxp$split_str(:class_path, ';')
  into :cls_name
  do
    last_class = cls_name;

  cls_parent = null;
  for select res_str
  from uxp$split_str(:class_path, ';')
  into :cls_name
  do
  begin
    cls_visual = -2;

    select cls_visual
    from ux$class
    where cls_name = :cls_name
    into cls_visual;

    if (row_count = 0) then
    begin
      if (cls_name = last_class) then
        cls_visual = -1;

      insert into ux$class (cls_name, cls_parent, cls_visual)
      values (:cls_name, :cls_parent, :cls_visual);
    end else
    begin
      if (cls_name = last_class and cls_visual = -2) then
      begin
        update ux$class
        set cls_visual = -1
        where cls_name = :cls_name;
      end
    end

    cls_parent = cls_name;
  end

  insert into ux$unititem (uni_unt_uuid, uni_parent_uuid, uni_json_path, uni_code, uni_name, uni_position)
  values (:unt_uuid, :parent_uuid, :json_path, :class_name, :class_type, :uni_position)
  returning uni_uuid
  into :uni_uuid;

  if (coalesce(prop_names, '') <> '') then
  begin
    prop_no = 0;
    for select res_str
    from uxp$split_str(:prop_names, ';')
    into :prop_name
    do
    begin
      prop_no = prop_no + 1;

      prop_value = '';
      prop_val_no = 0;
      for select res_str
      from uxp$split_str(:prop_values, ';')
      into :prop_value
      do
      begin
        prop_val_no = prop_val_no + 1;
        if (prop_val_no = prop_no) then
          break;
      end

      execute procedure uxp$uxunitprops_add(:unit_code, :uni_uuid, :class_type, :prop_name, :prop_value);
    end
  end

  suspend;
end
