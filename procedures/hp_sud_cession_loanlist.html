CREATE PROCEDURE HP_SUD_CESSION_LOANLIST(
    IN_SCP_ID D_ID)
RETURNS (
    SCL_ID D_ID,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_CODE,
    BKP_CODE D_STR_CODE,
    BKP_STATUS D_INTEGER,
    BEV_REC_NO D_INTEGER,
    BEV_STATUS D_INTEGER,
    SCL_FROM_OL_ID D_ID,
    SCL_OL_IS_ACTIVE D_STR_CODE,
    SCL_OL_DATE D_DATE)
AS
declare variable bev_id d_id;
declare variable event_date d_date;
begin
  select scp_date
  from sud_cession_packet
  where scp_id = :in_scp_id
    and scp_status = 1
  into event_date;

  for select scl_id, k_id, k_kontract_no, scl_from_bev_id, scl_from_ol_id, iif(scl_ol_is_active = 1, 'Активна', iif(scl_ol_is_active = 0, 'Аннулирована', 'Списания не было')), scl_ol_date
  from sud_cession_loans, kontract
  where k_id = scl_from_k_id
    and scl_from_scp_id = :in_scp_id
  into scl_id, k_id, k_kontract_no, bev_id, scl_from_ol_id, scl_ol_is_active, scl_ol_date
  do
  begin
    bkp_code = null;
    bkp_status = null;
    bev_status = null;

    if (bev_id is not null) then
    begin
      select bkp_code, bkp_status, bev_status, bev_rec_no
      from bki_events, bki_clients, bki_loans, bki_packet
      where bkc_id = bev_from_bkc_id
        and bkl_id = bkc_from_bkl_id
        and bkp_id = bkl_from_bkp_id
        and bev_id = :bev_id
      into bkp_code, bkp_status, bev_status, bev_rec_no;
    end
    else if (event_date is not null) then
    begin
      select 'В очереди на отправку'
      from bki_buffer
      where bkb_from_bkk_id = 1
        and bkb_object_kind = 9
        and bkb_object_id = :k_id
        and bkb_date = :event_date
        and bkb_type = 0
      rows 1
      into bkp_code;
    end 

    suspend;
  end 
end
