CREATE PROCEDURE MC_EDOC_SIGN_STATUS(
    EDL_ID D_ID,
    C_ID D_ID)
RETURNS (
    SIGN_STATUS D_STR_CODE,
    IS_SHOW D_BOOL)
AS
declare variable EDC_CODE D_STR_100;
declare variable EDC_PROCESS_KIND D_SMALLINT;
declare variable EDL_BLANK_DATE D_DATE;
declare variable EDL_OBJECT_ID D_ID;
declare variable EDS_STATUS D_INTEGER;
declare variable PASSED_DAYS D_INTEGER;
declare variable MA_IS_ACTIVE D_SMALLINT;
declare variable KS_STATE D_SMALLINT;
declare variable RES_ARCHIVE D_STR_CODE;
declare variable RES_SIGNED D_STR_CODE;
declare variable RES_IN_PROCESS D_STR_CODE;
begin
  -- процедура возвращает статус подписи edoc клиентом (RFCKG-1258), а также
  -- статус показа в КМП уведомления, связанного с переданным edoc, is_show: 1 - показать, 0 - скрыть (RFCKG-1261)

  -- коды статусов:
  res_in_process = 'in_process'; -- не подписан
  res_signed = 'signed'; -- подписан
  res_archive = 'archive'; -- архив

  sign_status = ''; -- возвращаемый статус подписи пока пустой
  is_show = 0; -- по умолчанию уведомления скрывать

  select edc.edc_code,
         edc.edc_process_kind,
         edl.edl_blank_date,
         edl.edl_object_id
  from edoc_list edl
  inner join edoc_cfg edc on edl.edl_from_edc_id = edc.edc_id
  where edl.edl_id = :edl_id and
        edl.edl_status >= 0 and
        edl.edl_blank_status = 1
  into edc_code, -- код конфигурации edoc
       edc_process_kind, -- вариант подписания документа
       edl_blank_date, -- дата создания edoc
       edl_object_id; -- id объекта (заявки или контракта)

  -- RFCKG-4732 перенос Информационного листа в раздел подписанные документы
  if (edc_code = 'loan.application_new_conditions') then
  begin
    sign_status = res_signed;
    suspend;
    exit;
  end

  -- кол-во дней, прошедших с даты создания документа
  passed_days = datediff(day, edl_blank_date, current_date);

  -- документы, не требующие подписания (в т.ч. 'loan.debt')
  if (edc_process_kind = 0) then
  begin
    if (passed_days > 7) then
      sign_status = res_archive;
    else
    begin
      sign_status = res_signed;
      is_show = 1;
    end

    suspend;
    exit;
  end

  -- далее идут документы, требующие подписания

  eds_status = 0;
  select eds_status
  from edoc_sign
  where eds_from_edl_id = :edl_id and
        eds_from_c_id = :c_id
  order by eds_status_datetime desc
  rows 1
  into eds_status; -- последний статус подписи клиентом

  -- документ еще не подписан
  if (eds_status = 0) then
  begin
    sign_status = res_in_process;
    is_show = 1;
    suspend;
    exit;
  end

  -- документ - кредитная заявка
  if (edc_code = 'loan.application') then
  begin
    select ma_is_active
    from miniapp
    where ma_id = :edl_object_id and
          ma_date_end = '31.12.9999'
    into ma_is_active; -- статус заявки (0 - аннулирована, 1 - активна)

    if (ma_is_active = 0) then
      sign_status = res_archive;
    else
    begin
      ks_state = 0;

      select ks.ks_state
      from kontstate ks
      inner join kontract k on k.k_id = ks.ks_from_k_id
      where k.k_from_ma_app_id = :edl_object_id and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end rows 1
      into :ks_state; -- статус контракта

      -- контракт в процессе или активный, статус заявки = подписан
      if (ks_state in (0, 1)) then
      begin
        sign_status = res_signed;
      end
      else
        sign_status = res_archive;
    end

    suspend;
    exit;
  end

  -- документ - кредитный договор
  if (edc_code = 'loan.agreement') then
  begin
    select ks_state
    from kontstate
    where ks_ol_is_delete = 0 and
          current_date between ks_date_begin and ks_date_end and
          ks_from_k_id = :edl_object_id
    into ks_state;

    -- контракт в процессе или активный
    if (ks_state in (0, 1)) then
    begin
      sign_status = res_signed;
    end
    else
      sign_status = res_archive;

    suspend;
    exit;
  end

  -- документ подписан
  if (eds_status = 1) then
    if (passed_days > 7) then
      sign_status = res_archive;
    else
      sign_status = res_signed;

  suspend;
end
