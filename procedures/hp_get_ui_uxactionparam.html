CREATE PROCEDURE HP_GET_UI_UXACTIONPARAM(
    UA_CODE D_STR_100,
    PARENT_UA_CODE D_STR_100)
RETURNS (
    UAP_CODE D_STR_CODE,
    UAP_FIELD D_STR_CODE,
    UAP_IS_PARAM D_BOOL,
    UAP_SOURCE_CODE D_STR_CODE,
    UAP_SOURCE_FIELD D_STR_CODE,
    UAP_SOURCE_IS_PARAM D_BOOL,
    UAP_VALUE D_STR_255,
    UAP_IS_PARENT_UA D_BOOL,
    UAP_LOOKUP_CODE D_STR_100)
AS
declare variable uap_default d_str_255;
declare variable uap_use_default d_bool;
declare variable uap_source_ufp_code d_str_100;
declare variable ux_code d_str_code;
declare variable ufp_code d_str_100;
declare variable uap_kind d_smallint;
declare variable param_kind d_smallint;
declare variable uap_readonly d_bool;
declare variable uxa_code d_str_code;
begin
  if (parent_ua_code is null) then
    parent_ua_code = '';

  select uxa_code, ux_code
  from hp_get_uxalias(:ua_code)
  into uxa_code, :ux_code;

  for select ucf_ufr_code
  from uxcodeframe
  where ucf_ux_code = :ux_code
  into uap_code
  do
  begin
    for select ufp_code, iif(ufp_pk_field <> 0 or ufp_is_param = 1, 0, 1)
    from uxfrparam
    where ufp_ufr_code = :uap_code and
          ufp_is_delete = 0
    into ufp_code, param_kind
    do
    begin
      uap_is_parent_ua = 0;
      uap_kind = param_kind;

      for select res_fr_code, res_field_name, res_is_param, res_def_value,
                 res_source_code, res_source_ufp_code, res_source_field,
                 res_source_is_param, res_readonly, res_lookup_code
      from hp_uxparam_final_value(:ufp_code, :uxa_code, :ua_code)
      into :uap_code, :uap_field, :uap_is_param, :uap_value,
           :uap_source_code, uap_source_ufp_code, :uap_source_field,
           :uap_source_is_param, :uap_readonly, :uap_lookup_code
      do
      begin
        if (uap_readonly = 0 and parent_ua_code <> '') then
        begin
          select uap_from_ufp_code, uap_use_default,
                 iif(uap_kind = 0, 1, 0), trim(coalesce(uap_default, '')), 1
          from uxactionparam
          where uap_ua_code = :parent_ua_code and
                uap_link_ufp_code = :ufp_code and
                uap_kind = :uap_kind
          into uap_source_ufp_code, uap_use_default,
               uap_is_param, uap_default, uap_is_parent_ua;
  
          if (uap_source_ufp_code is null) then
            uap_source_ufp_code = '';
  
          if (uap_use_default = 1 and uap_default = '') then
            uap_use_default = 0;
        end
  
        if (uap_source_ufp_code <> '' or uap_value <> '') then
          suspend;
      end
    end
  end
end
