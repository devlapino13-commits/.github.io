CREATE PROCEDURE FM_USER_SETTINGS(
    UN_ID D_ID)
RETURNS (
    RES_DATA D_STR_8100)
AS
declare variable NEW_CLIENTDOCS D_SMALLINT;
declare variable QR_PHOTO_ENABLED D_SMALLINT;
declare variable QR_VERIFY_ENABLED D_SMALLINT;
declare variable MENU_CLIENTS D_BOOL;
declare variable MENU_LOANS D_BOOL;
declare variable MENU_REPORTS D_BOOL;
declare variable MENU_TASKS D_BOOL;
declare variable MENU_NOTIFY D_BOOL;
declare variable MENU_UPLOADS D_BOOL;
declare variable MENU_CALC D_BOOL;
declare variable MENU_CALC_RU D_BOOL;
declare variable MENU_QR_INFO D_BOOL;
declare variable MENU_QR_VERIFY D_BOOL;
declare variable MENU_SETTINGS D_BOOL;
declare variable MENU_SURVEYS D_BOOL;
declare variable MENU_USER_CERT D_BOOL;
declare variable MENU_WHATS_NEW D_BOOL;
declare variable DB_ID D_ID;
declare variable ADDR_LANG_ENABLED D_BOOL;
declare variable REP_REFRESH_DELAY D_INTEGER;
declare variable TUNDUK_EPASS_ENABLED D_BOOL;
declare variable ANTIFROD_PHOTO_ACTIVE D_BOOL;
declare variable ANTIFROD_PHONE_ACTIVE D_BOOL;
declare variable CONTACT_NAME_VISIBLE D_BOOL;
declare variable SCREENSHOT_ENABLED D_BOOL;
declare variable BIOMETRY_SIGN_TRY_QTY D_INTEGER;
declare variable QR_SIMA_DESCRIPTION D_STR_100;
declare variable LANG D_STR_LANG;
declare variable UN_NAME D_STR_255;
declare variable PHONECALL_ENABLED D_BOOL;
declare variable TLS_GOST D_BOOL;
begin
  -- Процедура, возвращающая настройки для пользователя мобильного приложения
  -- Вызывается при логине пользователя
  db_id = f_get_db_id();
  lang = f_get_lang_code(un_id, null);
  new_clientdocs = f_get_const('new_clientdocs');
  antifrod_photo_active = f_get_const('antifrod_photo_active');
  antifrod_phone_active = f_get_const('antifrod_phone_active');
  qr_verify_enabled = f_get_const('qr_verify_enabled');
  biometry_sign_try_qty = f_get_const('biometry_sign_try_qty');
  qr_photo_enabled = 0;
  contact_name_visible = 1; -- Отображение поля ввода ФИО контактного лица
  qr_sima_description = ''; -- Текст подсказки пользователю, при открытии окна qr-подписания через sima.az
  tls_gost = 0; -- Шифрование по ГОСТ (РФ)

  if (db_id = 94401) then
    qr_sima_description = f_lang(lang,  'Для подписания документа клиенту необходимо отсканировать QR код приложением "Sima"');
  phonecall_enabled = f_get_const('fsm_phonecall_enabled'); -- Звонки через ip-телефонию

  addr_lang_enabled = 0;  -- В редактировании адреса отображать галочку "Формировать на кыргызском языке"
  if (db_id in (39801, 41701)) then
    addr_lang_enabled = 1;

  un_id = (select un_id
           from hp_test_un_id(:un_id));

  if (qr_verify_enabled = 2) then -- Временно, до боевого запуска
    qr_verify_enabled = 1;

  if (qr_verify_enabled = 1) then
  begin
    qr_photo_enabled = (select has_access
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.QrPhotoVerify'));
  end

  -- Получение скриншотов в МП
  screenshot_enabled = 1;
  if (db_id in (41701, 39801)) then
  begin
    screenshot_enabled = (select has_access
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ScreenshotAccess'));
  end

  if (db_id = 81001) then
  begin
    tls_gost = (select has_access
    from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.TlsGost'));
  end

  -- Кнопка создания клиента/получения паспорта через Тундук
  tunduk_epass_enabled = 0;
  if (db_id in (39801, 41701)) then
    tunduk_epass_enabled = 1;

  -- Время в секундах, через которое html-отчет будет перезапрашиваться на сервере
  -- при свайпе вниз (обновление страницы). Если прошло меньше, приложение не
  -- будет обращаться к серверу.
  -- Чем значение меньше, тем больше нагрузка на сервер.
  -- Изменение не заметно для клиента, для него выполняется имитация обновления,
  -- когда прошло меньше времени.
  -- Если будет передано отрицательное (неверное) число секунд, МП будет использовать 15-секундную задержку
  rep_refresh_delay = 15;

  -- Меню общее
  menu_clients = 1; -- Меню Клиенты
  menu_loans = 1; -- Меню Кредиты и справочники
  menu_reports = 1; -- Меню Мои отчеты
  menu_tasks = 1; -- Меню Мои задачи
  menu_notify = 1; -- Меню Новости и уведомления
  menu_uploads = 1; -- Меню История закачек
  menu_calc = 1; -- Меню Калькулятор графика погашений
  menu_calc_ru = 0; -- Меню Калькулятор доходов/расходов (только РФ)
  menu_qr_info = iif(db_id = 39801, 0, 1); -- Меню Информация по QR-коду             -- RFCKG-4140 
  menu_qr_verify = iif(db_id = 39801, 0, 1); -- Меню Верификация по QR-коду
  menu_settings = 1; -- Меню Настройки
  menu_surveys = 0; -- Меню Опросы (отключили, пока не используются)
  menu_user_cert = 1; -- Меню Удостоверение сотрудника
  menu_whats_new = 1; -- Меню Что нового?

  if (db_id = 81001) then
    menu_calc_ru = 1; -- Меню Калькулятор доходов/расходов (только РФ)
  if (db_id = 94401) then
    menu_qr_info = 0; -- Меню Информация по QR-коду (убрать после запуска эл. документов в АЗ)

  if (exists(select * from hp_permission_access_check(:un_id, 'PermActions', 0, 'FsMobile.ViewAccessOriv')
      where has_access = 1)) then
  begin
    phonecall_enabled = 0; -- Запрет на звонки
    -- Меню для ОРИВ
    menu_loans = 0; -- Меню Кредиты и справочники
    menu_tasks = 0; -- Меню Мои задачи
    menu_notify = 0; -- Меню Новости и уведомления
    menu_uploads = 0; -- Меню История закачек
    menu_calc = 0; -- Меню Калькулятор графика погашений
    menu_calc_ru = 0; -- Меню Калькулятор доходов/расходов (только РФ)
    menu_qr_info = 0; -- Меню Информация по QR-коду
    menu_qr_verify = 0; -- Меню Верификация по QR-коду
    tunduk_epass_enabled = 0; -- Кнопка создания клиента/получения паспорта через Тундук
  end

  select un.un_name from username un
  where un.un_id = :un_id
  into un_name;

  if (un_name = 'fmtest') then
  begin
   menu_clients = 0; -- Меню Клиенты
   menu_loans = 0; -- Меню Кредиты и справочники
   menu_reports = 1; -- Меню Мои отчеты
   menu_tasks = 0; -- Меню Мои задачи
   menu_notify = 1; -- Меню Новости и уведомления
   menu_uploads = 0; -- Меню История закачек
   menu_calc = 0; -- Меню Калькулятор графика погашений
   menu_calc_ru = 1; -- Меню Калькулятор доходов/расходов (только РФ)
   menu_qr_info = 1; -- Меню Информация по QR-коду
   menu_qr_verify = 0; -- Меню Верификация по QR-коду
   menu_settings = 0; -- Меню Настройки
   menu_surveys = 0; -- Меню Опросы (отключили, пока не используются)
   menu_user_cert = 1; -- Меню Удостоверение сотрудника
   menu_whats_new = 0; -- Меню Что нового?
  end
  -- Полная сборка
  res_data =
    '{' ||
      f_json_make_pair('qr_photo_enabled', qr_photo_enabled, 'str') || ',' ||
      f_json_make_pair('add_client_enabled', iif(db_id = 39801, 0, 1 - new_clientdocs), 'str') || ',' ||  -- RFCKG-4140
      f_json_make_pair('add_client_recog_enabled', iif(db_id = 39801, 0, 1), 'str') || ',' ||
      f_json_make_pair('new_clientdocs', new_clientdocs, 'str') || ',' ||
      f_json_make_pair('addr_lang_enabled', addr_lang_enabled, 'str') || ',' ||
      f_json_make_pair('rep_refresh_delay', rep_refresh_delay, 'str') || ',' ||
      f_json_make_pair('tunduk_epass_enabled', tunduk_epass_enabled, 'str') || ',' ||
      f_json_make_pair('antifrod_photo_active', antifrod_photo_active, 'str') || ',' ||
      f_json_make_pair('antifrod_phone_active', antifrod_phone_active, 'str') || ',' ||
      f_json_make_pair('biometry_sign_try_qty', biometry_sign_try_qty, 'str') || ',' ||
      f_json_make_pair('contact_name_visible', contact_name_visible, 'str') || ',' ||
      f_json_make_pair('screenshot_enabled', screenshot_enabled, 'str') || ',' ||
      f_json_make_pair('qr_sima_description', qr_sima_description, 'str') || ',' ||
      f_json_make_pair('phonecall_enabled', phonecall_enabled, 'str') || ',' ||
      f_json_make_pair('tls_gost', tls_gost, 'str') || ',' ||
      '"menu":{'||
        f_json_make_pair('clients', menu_clients, 'str') || ',' ||
        f_json_make_pair('loans', menu_loans, 'str') || ',' ||
        f_json_make_pair('reports', menu_reports, 'str') || ',' ||
        f_json_make_pair('tasks', menu_tasks, 'str') || ',' ||
        f_json_make_pair('notifications', menu_notify, 'str') || ',' ||
        f_json_make_pair('uploads', menu_uploads, 'str') || ',' ||
        f_json_make_pair('calc', menu_calc, 'str') || ',' ||
        f_json_make_pair('calc_ru', menu_calc_ru, 'str') || ',' ||
        f_json_make_pair('qr_info', menu_qr_info, 'str') || ',' ||
        f_json_make_pair('qr_verify', menu_qr_verify, 'str') || ',' ||
        f_json_make_pair('settings', menu_settings, 'str') || ',' ||
        f_json_make_pair('surveys', menu_surveys, 'str') || ',' ||
        f_json_make_pair('user_certificate', menu_user_cert, 'str') || ',' ||
        f_json_make_pair('whats_new', menu_whats_new, 'str')||
      '}' ||
    '}';

  suspend;
end
