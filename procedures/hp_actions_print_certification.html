CREATE PROCEDURE HP_ACTIONS_PRINT_CERTIFICATION(
    C_ID D_ID)
RETURNS (
    RES_ACL_ID D_ID,
    RES_QUESTION D_STR_255,
    RES_ANSWER D_STR_LONG)
AS
declare variable ANKETA_NAME D_STR_255;
declare variable ANKETA_ID D_INTEGER;
declare variable MA_ID D_INTEGER;
begin
  res_answer = '';

  anketa_name = 'Anketa.Main';
  select rd_id
  from refdata
  where rd_code = :anketa_name
  into :anketa_id;
  if (anketa_id is null) then
    exit;

  select acl_id
  from actionlist
  where acl_object_kind = 0 and
        acl_object_id = :c_id and
        acl_from_rd_id = :anketa_id
  order by acl_id desc
  rows 1
  into :res_acl_id;
  if (res_acl_id is null) then
    res_acl_id = 0;

  for select ques_text, res_text
      from hp_anketa_print(0, :c_id, :anketa_name, null)
      where ques_code in (
        'AnketaQuestions.1333',
        'AnketaQuestions.BankCardSystem',
        'AnketaQuestions.1337',
        'work_status',
        'Organization_type',
        'AnketaQuestions.1340',
        'work_job_name',
        'AnketaQuestions.Other_business_expenses',
        'AnketaQuestions.communal_expenses',
        'living_expenses',
        'monthly_income',
        'Ind_Entrepreneur',
        'AnketaQuestions.InfoSource',
        'AnketaQuestions.1332',
        'AnketaQuestions.Nation',
        'AnketaQuestions.Education',
        'work_place_name',
        'work_address_orientir',
        'AnketaQuestions.1336',
        'AnketaQuestions.repayment_obligations',
        --'AnketaQuestions.descr_of_loan',  -- RFCKG-2951
        'AnketaQuestions.1339',
        'AnketaQuestions.legal_docs_expenses',
        'AnketaQuestions.1335',
        'AnketaQuestions.Family',
        'AnketaQuestions.Family_fact', -- BOARD-4872
        'AnketaQuestions.1338',
        'AnketaQuestions.work_staj',
        'amount_monthly_payments',
        'extra_amount',
        'work_business_kind',
        'AnketaQuestions.TypeOfCard',
        'AnketaQuestions.VisitHomeNeeded',
        'AnketaQuestions.VisitBusinessNeeded',
        'AnketaQuestions.GetLoanCard',
        'AnketaQuestions.RepayLoanCard',
        'net_profit',
        'AnketaQuestions.1348'
      )
      into :res_question, res_answer
  do
  begin
  if (res_question = (select rd.rd_name
                      from refdata rd
                      where rd.rd_code ='AnketaQuestions.Family') and
      res_answer = (select ovs_value
                    from hp_objectprops_get(0, :c_id, -1, f_get_rd_id('AnketaQuestions.Family'), current_date)
                    where ovs_datetime >= (select gc.gc_status_datetime
                                           from grs_clients gc
                                           where gc.gc_from_c_id = :c_id and
                                                 gc.gc_status = 2 and
                                                 gc.gc_is_last = 1) and
                                                 ovs_from_un_id = 1000000000000)) then
        res_answer = res_answer || '(Инфо с ГРС)';
    suspend;
  end
    -- RFCKG-2951
    select first 1 k_from_ma_app_id
    from knclient
    inner join kontstate on kc_from_k_id = ks_from_k_id
        and current_date between ks_date_begin and ks_date_end
        and ks_state in (0,1)
    inner join kontract on kc_from_k_id = k_id
    where kc_from_c_id = :c_id
         and kc_amount_approved > 0
    order by ks_date_begin desc
    into :ma_id;

    res_question = 'AnketaQuestions.descr_of_loan';
    res_answer = (select loan_description from hp_get_loan_description(:c_id, :ma_id));

    suspend;
end
