CREATE PROCEDURE EQUIFAX_MISSEDPAY_120_CHECK(
    K_ID ID_TYPE,
    K_DISB_DATE DATE,
    CHECK_DATE DATE)
RETURNS (
    MAX_DELINQ_DAYS INTEGER)
AS
declare variable date_end date;
begin
-- контракты у которых в течении 120 дней с момента выдачи была непрерывная просрочка более двух платежей

  /*
    График платежей по договору:
    01.01.2015     01.02.2015   01.03.2015   01.04.2015   01.05.2015   01.06.2015   01.07.2015   01.08.2015   01.09.2015   ***
    Дата договора  1-ый платеж  2-ой платеж  3-ий платеж  4-ый платеж  5-ый платеж  6-ой платеж  7-ой платеж  8-ой платеж  *** 
                                                                                    ~ 120 дней с 
                                                                                    первого платежа
     
    1.  1) Если субъект КИ  не погасил подряд  1-ый и 2-ой  платеж –  пропущено 2 подряд платежа  в течение 120 календарных дней с момента 
    первого платежа, – то необходимо передать cred_missedpay_120 = 1. 

    2.  Если субъект КИ погасил 1-ый и 2-ой платеж, но не погасил подряд 3-ий и 4-ий платеж (или 4-ый и 5-ый) – пропущено 2 подряд платежа в 
    течение 120 календарных дней с момента первого платежа, – то необходимо передать cred_missedpay_120 = 1. 

    3.  Если субъект КИ  не погасил 5-ый и 6-ой платеж, то необходимо передать cred_missedpay_120  =  0,  т.к. в течение 120 календарных дней с 
    момента первого платежа был только один пропущенный платеж (5-ый платеж). 
  */

  if (k_disb_date is null) then
  begin
    select ks_date_begin
    from kontstate ks
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks.ks_from_k_id = :k_id
    into :k_disb_date;
  end

  date_end = dateadd(120 day to k_disb_date);

  max_delinq_days = null;
  if (date_end > check_date) then
    date_end = check_date;

-- проверяю max число дней просрочки на дату = дата выдачи  + 120 дней
--(считаю что 1 платеж = 30 дням), т.е. непрерывная просрочка в 2-а платежа = 60 дней.
  select max(iif(oks_date_end + 1 > cast(:date_end as date), cast(:date_end as date), oks_date_end + 1) - oks_delinq_date) max_delinq_days
  from oksummary oks
  where oks_delinq_date < :date_end and
        oks_close_state = 0 and
        (iif(oks_date_end + 1 > :date_end, :date_end, oks_date_end + 1) - oks_delinq_date) > 60 and
        oks_from_k_id = :k_id
  into max_delinq_days;

  if (max_delinq_days is not null) then
  begin
    suspend;
  end
end
