CREATE PROCEDURE UXP$TRANSMAIN_SYNC_LIST(
    ONLY_TRANSLATED D_BOOL)
RETURNS (
    TRM_ORIG_TEXT D_STR_255,
    TRM_IS_DELPHI D_BOOL,
    TRL_LANG_CODE D_STR_10,
    TRL_TRANSLATE_TEXT D_STR_255,
    TRL_ACTUAL_TEXT D_STR_255,
    TRL_CHECKED D_BOOL)
AS
declare variable trm_id d_id;
begin
  if (only_translated is null) then
    only_translated = 0;

  for select trm_id, trm_orig_text, trm_is_delphi
  from ux$transmain
  into trm_id, trm_orig_text, trm_is_delphi
  do
  begin
    trl_lang_code = '';
    trl_translate_text = '';
    trl_actual_text = '';
    trl_checked = null;

    for select trl_lang_code, trl_translate_text, trl_actual_text, trl_checked
    from ux$translate
    where trl_from_trm_id = :trm_id
    into :trl_lang_code, :trl_translate_text, :trl_actual_text, :trl_checked
    do
      suspend;

    if (row_count = 0 and only_translated = 0) then
      suspend;
  end
end
