CREATE PROCEDURE HP_AFM_GET_CLIENT_CATEGORIES(
    C_ID D_ID NOT NULL,
    DELIMITER D_STR_10 NOT NULL,
    VALUES_LIST D_STR_20 NOT NULL,
    CATEGORIES_IDS D_STR_CODE NOT NULL)
RETURNS (
    CATEGORIES D_STR_2048)
AS
begin
    -- Убираем пробелы, иначе будет работать некорректно
    values_list = replace(:values_list, ' ', '');
    categories_ids = replace(:categories_ids, ' ', '');

    with last_check as (
        select acc.*,
               row_number() over (partition by acc.ac_from_c_id order by acc.ac_date desc) rn
        from afm_check acc
        where acc.ac_from_c_id = :c_id
          and acc.ac_is_actual = 1
    )
    select list(ac.ac_name, :delimiter)
    from (
        select 1 as category_id, lc.ac_category_1 as val from last_check lc where lc.rn = 1
        union all select 2, lc.ac_category_2 from last_check lc where lc.rn = 1
        union all select 3, lc.ac_category_3 from last_check lc where lc.rn = 1
        union all select 4, lc.ac_category_4 from last_check lc where lc.rn = 1
        union all select 5, lc.ac_category_5 from last_check lc where lc.rn = 1
        union all select 6, lc.ac_category_6 from last_check lc where lc.rn = 1
        union all select 7, lc.ac_category_7 from last_check lc where lc.rn = 1
        union all select 8, lc.ac_category_8 from last_check lc where lc.rn = 1
        union all select 9, lc.ac_category_9 from last_check lc where lc.rn = 1
        union all select 10, lc.ac_category_10 from last_check lc where lc.rn = 1
        union all select 11, lc.ac_category_11 from last_check lc where lc.rn = 1
        union all select 12, lc.ac_category_12 from last_check lc where lc.rn = 1
        union all select 13, lc.ac_category_13 from last_check lc where lc.rn = 1
        union all select 14, lc.ac_category_14 from last_check lc where lc.rn = 1
        union all select 15, lc.ac_category_15 from last_check lc where lc.rn = 1
    ) x
    inner join afm_categories ac on ac.ac_id = x.category_id
    where
        position(',' || x.val || ',' in ',' || :values_list || ',') > 0
        and position(',' || x.category_id || ',' in ',' || :categories_ids || ',') > 0
    into :categories;

    suspend;
end
