CREATE PROCEDURE HP_CASHOPER_CLIENT_KONTRACTS(
    C_ID D_ID,
    KONT_STATE D_SMALLINT,
    ADD_INFO D_STR_CODE)
RETURNS (
    RES_K_ID D_ID,
    RES_KONTRACT_NAME D_STR_255)
AS
declare variable ks_state d_smallint;
declare variable need_suspend d_bool;
declare variable tmp_amount d_currency;
declare variable kf_id d_id;
declare variable kf_name d_str_code;
begin
  add_info = coalesce(add_info, '');
  if (kont_state is null) then
    kont_state = -1;

  for select kc_from_k_id, k_from_kf_id, k_kontract_no || ' от ' || f_datetostr(k_disb_date)
  from knclient, kontract
  where kc_from_c_id = :c_id and
        kc_amount_approved > 0 and
        kc_from_k_id = k_id and
        k_status = 1
  into res_k_id, kf_id, res_kontract_name
  do
  begin
    select ks_state
    from kontstate
    where ks_from_k_id = :res_k_id and
          ks_ol_is_delete = 0 and
          ks_date_end >= '31.12.2999'
    into ks_state;

    kf_name = '';
    select kal_new_kf_name
    from knaktlist
    where kal_from_kf_id = :kf_id and
          kal_date_end >=  '31.12.2999' and
          is_delete = 0
    into kf_name;

    if (kf_name <> '') then
      res_kontract_name = res_kontract_name || '  (' || kf_name || ')';

    need_suspend = 0;
    if (kont_state = -1 or ks_state = kont_state) then
      need_suspend = 1;

    if (add_info = 'minus.loan.overpay') then
    begin
      tmp_amount = 0;
      select sum(ok_os)
      from operkred
      where ok_from_k_id = :res_k_id and
            ok_ol_is_active = 1 and
            ok_type = 9
      into tmp_amount;

      if (tmp_amount is null) then
        tmp_amount = 0;

      if (tmp_amount = 0) then
        need_suspend = 0;
    end

    if (need_suspend = 1) then
      suspend;
  end
end
