CREATE PROCEDURE UXP$UXFIELD_LINKED_SOURCES(
    UF_UUID D_UUID,
    SHOW_ALL D_BOOL = 0)
RETURNS (
    ID D_STR_CODE,
    TEXT D_STR_255)
AS
declare variable uf_source d_str_code;
declare variable field_name d_str_code;
declare variable source_name d_str_code;
declare variable is_found d_integer;
declare variable uf_origin_uf_uuid d_uuid;
begin
  uf_source = uxf$owner(uf_uuid, 'code');
  field_name = uxf$code(uf_uuid, 'code');

  if (show_all is null) then
    show_all = 0;

  is_found = 0;
  for select us_code, iif(us_code = us_name, us_code, us_code || '    - ' || us_name)
  from ux$index, ux$source
  where uix_source = :uf_source and
        uix_fields = :field_name and
        uix_fk_source = us_code and
        coalesce(us_lookup_text, '') <> ''
  into id, text
  do
  begin
    is_found = 1;
    suspend;

    source_name = id;

    for select us_code, iif(us_code = us_name, us_code, us_code || '    - ' || us_name)
    from ux$source
    where us_main_table = :source_name and
          coalesce(us_lookup_text, '') <> '' and
          us_kind <> -1 -- кроме SQL
    into id, text
    do
      suspend;
  end

  select uf_origin_uf_uuid
  from ux$field
  where uf_uuid = :uf_uuid and
        uf_origin_uf_uuid is not null
  into uf_origin_uf_uuid;

  if (uf_origin_uf_uuid is not null) then
  begin
    select us_code, uf_field_name
    from ux$field, ux$source
    where uf_uuid = :uf_origin_uf_uuid and
          uf_owner = us_uuid
    into uf_source, field_name;

    if (row_count = 1) then
    begin
      for select us_code, iif(us_code = us_name, us_code, us_code || '    - ' || us_name)
      from ux$index, ux$source
      where uix_source = :uf_source and
            uix_fields = :field_name and
            uix_fk_source = us_code and
            coalesce(us_lookup_text, '') <> ''
      into id, text
      do
      begin
        is_found = 1;
        suspend;
      end
    end
  end

  if (is_found = 0 and show_all = 1) then
  begin
    for select us_code, us_name
    from uxp$uxsource_list(0, 'source')
    into :id, :text
    do
      suspend;
  end
end
