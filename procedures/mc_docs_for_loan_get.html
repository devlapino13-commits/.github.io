CREATE PROCEDURE MC_DOCS_FOR_LOAN_GET(
    MCL_ID D_ID,
    C_ID D_ID,
    PASSKIND_ID D_INTEGER)
RETURNS (
    CDS_ID D_INTEGER,
    DOC_EXISTS D_SMALLINT,
    API_CODE D_STR_255,
    BLS_1 D_ID,
    BLS_2 D_ID)
AS
begin
  doc_exists = 0;

  if (c_id is null) then
  begin
    select mcl.mcl_from_c_id
    from mc_login mcl
    where mcl.mcl_id = :mcl_id
    into :c_id;
  end

  for select cds_id, cds_doc_blob_1, cds_doc_blob_2
      from (
      -- получаю из clientdocs
      select cds.cds_id, cds.cds_doc_blob_1, cds.cds_doc_blob_2, cds.cds_create_datetime
      from clientdocs cds
      where cds.cds_from_c_id = :c_id and
            cds.cds_from_pk_id = :passkind_id and
            cds.cds_is_delete = 0 and
            coalesce(cds.cds_doc_blob_1, 0) + coalesce(cds.cds_doc_blob_2, 0) > 0 and
            current_date between cds.cds_doc_date_begin and coalesce(cds.cds_doc_date_end, '31.12.9999')
      union
      -- получаю из docrecoglist
      select null as cds_id, drl.drl_from_bls_1, drl.drl_from_bls_2, drl.drl_create_datetime
      from docrecoglist drl
      inner join actionlist al on al.acl_id = drl.drl_from_acl_id
      where drl.drl_from_mcl_id = :mcl_id and
            drl.drl_from_pk_id = :passkind_id and
            ((coalesce(drl.drl_from_bls_1, 0) > 0) or
            (coalesce(drl.drl_from_bls_2, 0) > 0)) and
            al.acl_status >= 0 -- учитываю только документы по которым статус проверки <> Отказ
      ) a
      order by cds_create_datetime desc
      rows 1
      into :cds_id, :bls_1, :bls_2
  do
  begin
    doc_exists = 1;
  end

  select drm.drm_api_code
  from passkind pask
  left join docrecogmain drm on drm.drm_id = pask.pk_from_drm_id
  where pask.pk_id = :passkind_id
  into api_code;

  suspend;
end
