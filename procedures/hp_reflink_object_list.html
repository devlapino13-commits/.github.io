CREATE PROCEDURE HP_REFLINK_OBJECT_LIST(
    REP_CATEGORY VARCHAR(255),
    FILTER_RD_ID INTEGER)
RETURNS (
    ID DECIMAL(18,0),
    NAME VARCHAR(255),
    MEMO VARCHAR(255),
    IS_ACTIVE INTEGER,
    LABEL_VALUE VARCHAR(255),
    CATEGORY VARCHAR(255))
AS
declare variable select_sql varchar(5000);
declare variable obj_id decimal(18,0);
begin
-- Процедура возвращает список объектов
-- либо список объектов имеющее свойство Filter_RD_ID

  if (filter_rd_id = 0) then -- Полный список объектов
  begin
    select ref_select_sql
    from hp_reflink_category(0)
    where ref_name = :rep_category
    into select_sql;
    if (select_sql is null) then
      exit;

    category = rep_category;
    for execute statement :select_sql
            into :id, :name, :memo, :is_active, :label_value
    do
      suspend;

  end
  else
  begin
    for select rlk_object_id, rd_memo
        from reflink, refdata
        where rlk_from_rd_id = :filter_rd_id and
              rd_id = :filter_rd_id
        into :obj_id, category
    do
    begin
      select ref_select_sql
      from hp_reflink_category(:obj_id)
      where ref_name = :rep_category
      into select_sql;

      for execute statement :select_sql
              into :id, :name, :memo, :is_active, :label_value
      do
        suspend;
    end
  end
end
