CREATE PROCEDURE MC_ONLINE_LOAN_GET_KIB_ID(
    C_ID D_ID)
RETURNS (
    KBM_ID D_ID,
    KBM_SCORE_RATE D_INTEGER)
AS
declare variable DAYS_QTY D_INTEGER;
begin
  days_qty = f_get_const('online_loan_cib_report_valid_days');

  select kaa.kbm_id,
         kaa.kbm_score_rate
  from kibminiapp kaa
  where kaa.kbm_from_c_id = :c_id and
  kaa.kbm_is_checked = 1 and
  cast(kbm_checked_datetime as date) >= current_date - :days_qty
  order by kaa.kbm_id desc
  rows 1
  into :kbm_id,
       :kbm_score_rate;

  suspend;
end
