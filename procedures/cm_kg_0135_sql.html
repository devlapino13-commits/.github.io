CREATE PROCEDURE CM_KG_0135_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable c_id d_id;
declare variable pr_code d_str_100;
begin
  if (k_id = 109010264931500101) then
    exit; ---RFCKG-613
  if (k_id = 109010248810200101) then
    exit; --RFCKG-847
  if (k_id = 109010317188000101) then
    exit; --RFCKG-2905

  res_error_is_critical = 1;
  res_error_text = 'Загрузите фото клиента на дату выдачи.';
  for select k_disb_date, pr.pr_code
      from kontract k
      inner join product pr on pr.pr_id = k.k_from_pr_id
      where k_id = :k_id and
            pr.pr_code not containing '.istisnaa.'
      into kf_disb_date, pr_code
  do
  begin
    if (pr_code containing '.express_online.') then
      exit;
    if (pr_code containing '.jonokoi_cards.') then
      exit;
    if (pr_code containing '.online_loan.') then
      exit;
    if (pr_code containing '.TMS.') then
      exit;

    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into :c_id
    do
    begin
      if (not exists(select *
                     from clientdocs cdc
                     inner join passkind pk on pk.pk_id = cds_from_pk_id
                     inner join passdoctype pdt on pk.pk_from_pdt_id = pdt.pdt_id and
                           pdt.pdt_code = 'foto_on_date_disb' -- фото на дату выдачи
                     where cds_from_c_id = :c_id and
                           cds_is_delete = 0 and
                           --cds_verify_status = 2 and
                           cds_doc_date_begin between :kf_disb_date - 3 and :kf_disb_date -- JIRA Board-4073
                           and
                           (cdc.cds_doc_blob_1 is not null or
                           cdc.cds_doc_blob_2 is not null) --явная проверка на наличие блоба
          )) then
      begin
        if (exists(select *
                   from edoc_sign eds
                   inner join edoc_list edl on edl.edl_id = eds.eds_from_edl_id and
                         edl.edl_object_kind = f_objkind('kontract', 'id') and
                         edl.edl_object_id = :k_id and
                         edl.edl_status = 1 and
                         edl.edl_blank_status = 1
                   where eds_enum_edoc_sign_kind = f_enum('edoc_sign_kind', 'face', 'id') and
                         eds_status = 1 and
                         eds.eds_from_c_id = :c_id)) then
          continue;

        suspend;
      end
    end
  end
end
