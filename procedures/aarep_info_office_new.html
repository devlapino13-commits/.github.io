CREATE PROCEDURE AAREP_INFO_OFFICE_NEW(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    RES_REGION_ID INTEGER,
    SVZ_KOL_ACT INTEGER,
    SVZ_DISB1 INTEGER,
    DISB_KOL_KRED1 INTEGER,
    DISB_KOL_PARALEL1 INTEGER,
    DISB_KOL_VSEGO1 INTEGER,
    SVZ_ONE_DISB1 CURR_TYPE,
    SVZ_DISB7 INTEGER,
    DISB_KOL_KRED7 INTEGER,
    DISB_KOL_PARALEL7 INTEGER,
    DISB_KOL_VSEGO7 INTEGER,
    SVZ_ONE_DISB7 CURR_TYPE,
    SVZ_DISB30 INTEGER,
    DISB_KOL_KRED30 INTEGER,
    DISB_KOL_PARALEL30 INTEGER,
    DISB_KOL_VSEGO30 INTEGER,
    SVZ_ONE_DISB30 CURR_TYPE,
    REP_BALANCE CURR_TYPE,
    DELINQ_BALANCE_1 CURR_TYPE,
    DELINQ_BALANCE_7 CURR_TYPE,
    DELINQ_BALANCE_30 CURR_TYPE,
    PAR1 CURR_TYPE,
    PAR7 CURR_TYPE,
    PAR30 CURR_TYPE,
    KOL_CLOSE_KRED1 INTEGER,
    KOL_DISB_KRED1 INTEGER,
    PERC_UDERJ1 CURR_TYPE,
    REP_BALANCE_LAST1 CURR_TYPE,
    KOL_CLOSE_KRED7 INTEGER,
    KOL_DISB_KRED7 INTEGER,
    PERC_UDERJ7 CURR_TYPE,
    REP_BALANCE_LAST7 CURR_TYPE,
    KOL_CLOSE_KRED30 INTEGER,
    KOL_DISB_KRED30 INTEGER,
    PERC_UDERJ30 CURR_TYPE,
    REP_BALANCE_LAST30 CURR_TYPE,
    PRIROST_PORTFEL1 CURR_TYPE,
    PRIROST_PORTFEL7 CURR_TYPE,
    PRIROST_PORTFEL30 CURR_TYPE,
    KOL_CLI INTEGER,
    KOL_CLI_LAST1 INTEGER,
    PRIROST_CLI1 INTEGER,
    KOL_CLI_LAST7 INTEGER,
    PRIROST_CLI7 INTEGER,
    KOL_CLI_LAST30 INTEGER,
    PRIROST_CLI30 INTEGER,
    OFFICE_NAME VARCHAR(50),
    REP_BALANCE_WEEK1 DECIMAL(18,4),
    REP_BALANCE_WEEK2 DECIMAL(18,4),
    REP_BALANCE_BEGIN_MONTH DECIMAL(18,4),
    DELINQ_BALANCE_WEEK1_7 DECIMAL(18,4),
    DELINQ_BALANCE_WEEK2_7 DECIMAL(18,4),
    DELINQ_BALANCE_BEGIN_MONTH DECIMAL(18,4))
AS
begin
  for select res_region_id, sum(svz_kol_act), sum(svz_disb1), sum(disb_kol_kred1), sum(disb_kol_paralel1),
             sum(disb_kol_vsego1), sum(svz_one_disb1), sum(svz_disb7), sum(disb_kol_kred7), sum(disb_kol_paralel7),
             sum(disb_kol_vsego7), sum(svz_one_disb7), sum(svz_disb30), sum(disb_kol_kred30), sum(disb_kol_paralel30),
             sum(disb_kol_vsego30), sum(svz_one_disb30), sum(rep_balance), sum(delinq_balance_1), sum(delinq_balance_7),
             sum(delinq_balance_30), sum(par1), sum(par7), sum(par30), sum(kol_close_kred1), sum(kol_disb_kred1),
             sum(kol_close_kred7), sum(kol_disb_kred7), sum(kol_close_kred30), sum(kol_disb_kred30), sum(perc_uderj1),
             sum(perc_uderj7), sum(perc_uderj30), sum(rep_balance_last1), sum(rep_balance_last7),
             sum(rep_balance_last30), sum(kol_cli), sum(kol_cli_last1), sum(kol_cli_last7), sum(kol_cli_last30),
             sum(rep_balance_week1), sum(rep_balance_week2), sum(rep_balance_begin_month), sum(delinq_balance_week1_7),
             sum(delinq_balance_week2_7), sum(delinq_balance_begin_month)
      from aarep_info_office_new_tmp(:rep_region_id, :rep_date_end)
      group by res_region_id
      into :res_region_id, :svz_kol_act, :svz_disb1, :disb_kol_kred1, :disb_kol_paralel1, :disb_kol_vsego1,
           :svz_one_disb1, :svz_disb7, :disb_kol_kred7, :disb_kol_paralel7, :disb_kol_vsego7, :svz_one_disb7,
           :svz_disb30, :disb_kol_kred30, :disb_kol_paralel30, :disb_kol_vsego30, :svz_one_disb30, :rep_balance,
           :delinq_balance_1, :delinq_balance_7, :delinq_balance_30, :par1, :par7, :par30, :kol_close_kred1,
           :kol_disb_kred1, :kol_close_kred7, :kol_disb_kred7, :kol_close_kred30, :kol_disb_kred30, :perc_uderj1,
           :perc_uderj7, :perc_uderj30, :rep_balance_last1, :rep_balance_last7, :rep_balance_last30, :kol_cli,
           :kol_cli_last1, :kol_cli_last7, :kol_cli_last30, :rep_balance_week1, rep_balance_week2,
           rep_balance_begin_month, :delinq_balance_week1_7, :delinq_balance_week2_7, :delinq_balance_begin_month

  do
  begin
    prirost_portfel1 = rep_balance - rep_balance_last1;
    prirost_portfel7 = rep_balance - rep_balance_last7;
    prirost_portfel30 = rep_balance - rep_balance_last30;

    prirost_cli1 = kol_cli - kol_cli_last1;
    prirost_cli7 = kol_cli - kol_cli_last7;
    prirost_cli30 = kol_cli - kol_cli_last30;
    select reg_name
    from region
    where reg_unicode = :res_region_id
    into office_name;

    suspend;
  end
end
