CREATE PROCEDURE HP_ADDITIONAL_KOMMIS_PERC_SHOW(
    PR_ID D_INTEGER,
    PRL_ID D_INTEGER,
    PAR_DISB_KIND D_INTEGER)
RETURNS (
    PAR_ID D_ID,
    PAR_KOMMIS_PERC_DEFAULT D_PERCENT)
AS
begin
  -- RFCKG-4129
  -- Показывает текущее значение  Доп. процента для определенного продукта и "Способа выдачи"
  select par.par_id,
         par.par_kommis_perc_default
  from product_additional_rate par
  where par.par_from_pr_id = :pr_id and
        par.par_from_prl_id = :prl_id and
        par.par_is_delete = 0 and
        par.par_disb_kind = :par_disb_kind
  order by par.par_create_datetime desc
  rows 1
  into par_id, par_kommis_perc_default;

  par_id = coalesce(par_id, 0);
  par_kommis_perc_default = coalesce(par_kommis_perc_default, 0);
  suspend;
end
