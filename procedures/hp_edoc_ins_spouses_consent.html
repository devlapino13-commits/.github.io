CREATE PROCEDURE HP_EDOC_INS_SPOUSES_CONSENT(
    MA_ID D_INTEGER,
    UN_ID ID_TYPE)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable c_id id_type;
declare variable c_name_fio d_str_100;
declare variable is_verified d_bool;
declare variable msg_err d_str_255;
begin
  -- spouses consent = согласие супруга(и) на займ
  -- процедура проверяет в заявке каждое лицо, дающее согласие,
  -- если все прошли проверки, то вызывает hp_edoc_insert, передав в нее ma_id и un_id

  res_text = '';
  msg_err = 'Согласие супруга(и) на займ не отправлено. Ошибка: ';

  for select c.c_id,
             c_name_fio
      from miniappclient mac
      inner join clients c on c.c_id = mac.mac_from_c_id
      where mac.mac_from_ma_id = :ma_id and
            mac.mac_is_poruch = -1 and
            mac_amount = 0
      into c_id,
           c_name_fio
  do
  begin
    msg_err = msg_err || c_name_fio;

    is_verified = 0;
    select is_verified
    from hp_verify_consent(:ma_id, :c_id)
    into is_verified;

    if (is_verified = 0) then
      res_text = msg_err || ' не является супругом(ой) клиенту в заявке';
  end

  if (res_text = '') then
    select edl_id,
           res_text
    from hp_edoc_insert('loan.spouses_consent', f_objkind('miniapp', 'id'), :ma_id, :un_id)
    into edl_id,
         res_text;
  else
    exception e_message res_text;

  suspend;
end
