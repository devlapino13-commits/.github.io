CREATE PROCEDURE CEO_DISB_LOAN_GET(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    K_ID D_ID,
    K_KONTRACT_NO D_STR_CODE,
    K_FACT_DISB_DATE D_DATE,
    K_AMOUNT_APPROVED D_CURRENCY,
    K_PERIOD DECIMAL(15,2),
    K_PERCENT PERC_TYPE,
    K_FACT_DISB_LO_ID D_ID,
    K_FACT_DISB_REG_ID D_INTEGER,
    K_PERCENT_EFFECT PERC_TYPE,
    K_FROM_PR_ID D_INTEGER,
    C_LO_BIRTH_DATE D_DATE,
    C_LO_SEX D_SMALLINT,
    C_LO_CITIZEN D_INTEGER,
    C_LO_NATION D_INTEGER,
    KC_ID ID_TYPE,
    KC_FROM_C_ID ID_TYPE,
    KC_AMOUNT_APPROVED D_CURRENCY,
    C_CLI_BIRTH_DATE D_DATE,
    C_CLI_SEX D_SMALLINT,
    C_CLI_CITIZEN D_INTEGER,
    C_CLI_NATION D_INTEGER,
    KC_AGE D_INTEGER,
    KC_POINT D_INTEGER,
    KC_PDN D_CURRENCY)
AS
declare variable DB_ID D_INTEGER;
begin
    -- Формирую список выданных кредитов в связки с клиентами

    db_id = f_get_db_id();
    for
        select
            k.k_id,
            k.k_kontract_no,
            k.k_fact_disb_date,
            k.k_amount_approved,
            k.k_period,
            k.k_percent,
            k.k_fact_disb_lo_id,
            k.k_fact_disb_reg_id,
            k.k_percent_effect,
            k.k_from_pr_id,
            c.c_birth_date,
            c.c_sex,
            c.c_citizen,
            c.c_nation
        from kontstate ks
        inner join kontract k
            on k.k_id = ks.ks_from_k_id
        inner join clients c
            on c.c_id = k.k_fact_disb_lo_id
        where ks.ks_state = 1
          and ks.ks_ol_is_delete = 0
          and ks.ks_date_begin >= :rep_date_begin
          and ks.ks_date_begin <= :rep_date_end
        into
            :k_id,
            :k_kontract_no,
            :k_fact_disb_date,
            :k_amount_approved,
            :k_period,
            :k_percent,
            :k_fact_disb_lo_id,
            :k_fact_disb_reg_id,
            :k_percent_effect,
            :k_from_pr_id,
            :c_lo_birth_date,
            :c_lo_sex,
            :c_lo_citizen,
            :c_lo_nation
    do
    begin
        for
            select
                kncl.kc_id,
                kncl.kc_from_c_id,
                kncl.kc_amount_approved,
                c.c_birth_date,
                c.c_sex,
                c.c_citizen,
                c.c_nation,
                f_age(c.c_birth_date, :k_fact_disb_date)
            from knclient kncl
            inner join clients c
                on c.c_id = kncl.kc_from_c_id
            where kncl.kc_amount_approved > 0
              and kncl.kc_from_k_id = :k_id
            into
                :kc_id,
                :kc_from_c_id,
                :kc_amount_approved,
                :c_cli_birth_date,
                :c_cli_sex,
                :c_cli_citizen,
                :c_cli_nation,
                :kc_age
        do
        begin
            if (db_id = 81001) then
                kc_point =  f_point_get(9, :k_id, :kc_from_c_id, :k_fact_disb_date);
            else
                kc_point =  f_point(:kc_from_c_id, :k_fact_disb_date, 0);

            kc_pdn = f_get_kdn(:kc_id);

            suspend;
        end
    end
end
