CREATE PROCEDURE AAREP_LOAN_OFFICERS_INFO(
    REG_ID INTEGER,
    REP_DATE DATE,
    LIST_KIND VARCHAR(10))
RETURNS (
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100),
    KE_CODE VARCHAR(20),
    CLIENT_COUNT INTEGER,
    KF_COUNT INTEGER,
    TOTAL_BAL CURR_TYPE,
    DELINQ_BAL CURR_TYPE)
AS
begin
  if (list_kind = 'DstList') then
  begin
-- Возвращаем список получателей, заполняются все поля
    for select distinct c_id, c_name_fio, ke_code
        from krexpert, clients
        where ke_from_p_id = c_id and
              ke_status = 1 and
              ke_date <= :rep_date and
              ke_date_end >= :rep_date and
              ke_is_krs <= 2 and
              ke_from_reg_id = :reg_id
        order by c_name_fio
        into c_id, c_name_fio, ke_code
    do
    begin
-- Для каждого сотрудника получаем:
      select count(distinct(kc_from_c_id)), -- Количество клиентов
             count(distinct(k_from_kf_id)), -- Количество папок
             sum(cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance), -- Баланс кредитов
             sum(iif(:rep_date > oks_delinq_date, cast(kc_amount_approved as double precision) / k_amount_approved * oks_balance, 0)) -- Баланс просрочек
      from kontract, oksummary, knaktlist, knclient
      where k_from_kf_id = kal_from_kf_id and
            k_id = oks_from_k_id and
            :rep_date between oks_date_begin and oks_date_end and
            is_delete = 0 and
            :rep_date between kal_date and kal_date_end and
            kal_new_kf_officer = :c_id and
            kc_from_k_id = k_id and
            kc_amount_approved > 0 and
            oks_close_state = 0
      into client_count, kf_count, total_bal, delinq_bal;
      total_bal = coalesce(total_bal, 0);
      delinq_bal = coalesce(delinq_bal, 0);

      suspend;
    end
  end -- if (List_Kind = 'DstList')

  if (list_kind = 'SrcCombo') then
  begin
-- Возвращаем список передающих КС/СВЗ + Виртуальный СВЗ (Для РФ)
-- Поля: c_id, c_name_fio, ke_code
    for select c_id, c_name_fio, ke_code
        from (
-- КС/СВЗ
        select distinct c_id, c_name_fio, ke_code
        from krexpert, clients
        where ke_from_p_id = c_id and
              ke_status = 1 and
              ke_date <= :rep_date and
              ke_date_end >= :rep_date and
              ke_is_krs <= 2 and
              ke_from_reg_id = :reg_id
        union
-- Виртуальный сотрудник
        select c_id, c_name_fio, '.ВС'
        from clients
        where c_id = 1000000000000)
        order by c_name_fio
        into c_id, c_name_fio, ke_code
    do
      suspend;
  end -- if (List_Kind = 'SrcCombo')
end
