CREATE PROCEDURE HP_GET_UI_UXFRAMES(
    UA_CODE D_STR_100,
    UN_ID D_STR_CODE)
RETURNS (
    FR_CODE D_STR_CODE,
    UFR_CAN_SELECT D_BOOL,
    UFR_CAN_UPDATE D_BOOL,
    UFR_CAN_INSERT D_BOOL,
    UFR_CAN_DELETE D_BOOL,
    UFR_TABLE_NAME D_STR_CODE,
    UFR_KEY_FIELDS D_STR_100,
    UFR_PARAM_SQL D_STR_8100,
    UFR_SELECT_SQL D_STR_8100,
    UFR_UPDATE_SQL D_STR_8100,
    UFR_INSERT_SQL D_STR_8100,
    UFR_DELETE_SQL D_STR_8100,
    UFR_MASTER_FRAME D_STR_CODE,
    UFR_MASTER_FIELD D_STR_100,
    UFR_DETAIL_FIELD D_STR_100,
    UFR_IS_MAIN D_BOOL,
    UFR_HAS_EXEC_PROC D_BOOL,
    UFR_SHOW_KIND D_SMALLINT,
    UFR_SELECT_IS_EXEC D_BOOL)
AS
declare variable ux_code d_str_code;
declare variable main_fr_code d_str_code;
declare variable need_secure_mask d_bool;
declare variable ua_usr_id_field d_str_code;
begin
  select ux_code, ua_usr_id_field
  from hp_get_uxalias(:ua_code)
  into ux_code, ua_usr_id_field;

  need_secure_mask = 0;
  if (ua_usr_id_field <> '' and uxf$access(un_id, ua_code) = -1) then
    need_secure_mask = 1;

  if (ua_usr_id_field <> '') then
    need_secure_mask = 1;

  select ucf_ufr_code
  from uxcodeframe
  where ucf_ux_code = :ux_code and
        ucf_is_mainframe = 1
  into main_fr_code;

  for select fr_code, master_frame, master_field, detail_field, show_kind
  from hp_uxframe_md_tree(:ux_code, null)
  into :fr_code, :ufr_master_frame, :ufr_master_field, :ufr_detail_field, :ufr_show_kind
  do
  begin
    ufr_is_main = 0;
    if (fr_code = main_fr_code) then
      ufr_is_main = 1;

    ufr_has_exec_proc = 0;
    for select ufr_can_select, ufr_can_update, ufr_can_insert, ufr_can_delete,
               ufr_table_name, ufr_key_fields, ufr_param_sql,
               ufr_select_sql, ufr_update_sql, ufr_insert_sql, ufr_delete_sql,
               ufr_has_exec_proc, ufr_select_is_exec
    from hp_uxframe_sql(:ux_code, :fr_code, :need_secure_mask)
    into :ufr_can_select, :ufr_can_update, :ufr_can_insert, :ufr_can_delete,
         :ufr_table_name, :ufr_key_fields, :ufr_param_sql,
         :ufr_select_sql, :ufr_update_sql, :ufr_insert_sql, :ufr_delete_sql,
         :ufr_has_exec_proc, ufr_select_is_exec
    do
      suspend;
  end
end
