CREATE PROCEDURE AAREP_MANAGER_LIST(
    REP_REGION_ID INTEGER,
    REP_DATE DATE,
    REP_IS_SIGN INTEGER)
RETURNS (
    MANAGER_C_ID DECIMAL(18,0),
    MANAGER_FIO VARCHAR(100))
AS
declare variable db_id integer;
declare variable ksr_can_sign_loans integer;
declare variable reg_id integer;
declare variable reg_head_office integer;
declare variable non_standard integer;
begin
  db_id = f_get_db_id();
  if (rep_is_sign = 1) then -- Список подписантов
  begin
    for select c_id, c_name_fio, 0
        from clients, hr_pstatus, joblist, jobkind
        where c_id = ps_from_c_id and
              ps_date_begin <= :rep_date and
              ps_date_end >= :rep_date and
              ps_status = 1 and
              ps_from_jl_id = jl_id and
              jl_kind = jk_id and
              jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
              ps_from_reg_id = :rep_region_id and
              ps_is_working = 1
        union
        select c_id, c_name_fio, 1
        from hr_pstatus
          inner join joblist
            on jl_id = ps_from_jl_id
          inner join jobkind
            on jk_id = jl_kind
              and jk_system_code = 'JobSystem.LO_Standart'
          inner join clients
            on c_id = ps_from_c_id
        where :db_id = 81001 and :rep_date between ps_date_begin and ps_date_end and
          ps_from_reg_id = :rep_region_id and ps_is_working = 1 and ps_status = 1
        into :manager_c_id, :manager_fio, :non_standard
    do
    begin
      ksr_can_sign_loans = 0;
      if (rep_date is not null) then
      begin
        if (non_standard=0) then begin
          for select distinct dvc_from_reg_id
              from dovlist, dovcateg, dovclients
              where dl_from_dc_id = dc_id and
                    dl_id = dvc_from_dl_id and
                    dvc_from_c_id = :manager_c_id and
                    dc_sign_loans = 1 and
                    dl_status = 1 and
                    dl_date_begin <= :rep_date and
                    dl_date_end >= :rep_date and
                    ((dl_date_end_fact is null) or
                    (dl_date_end_fact >= :rep_date))
              into :reg_id
          do
          begin
            if (reg_id is not null) then
            begin
              if (exists(select *
                         from region_list(:rep_region_id, 1, -1, 0)
                         where id_list = :reg_id)) then
              begin
                ksr_can_sign_loans = 1;
              end
            end
          end
        end else begin
          if (exists(
            select *
            from kontsignmen
              inner join kontsignreg
                on ksr_from_ksm_id = ksm_id
                  and :rep_date between coalesce(ksr_date_begin, '01.01.1990') and coalesce(ksr_date_end, '31.12.9999')
                  and ksr_is_active = 1
              inner join region_list(:rep_region_id, 1, -1, 0)
                on id_list = ksr_from_reg_id
            where ksm_from_c_id = :manager_c_id and ksm_is_active = 1))
          then ksr_can_sign_loans = 1;
        end
      end

      if (ksr_can_sign_loans = 1) then
        suspend;
    end
    for select c_id, c_name_fio
        from kontsignreg, clients, kontsignmen
        where ksr_from_reg_id = :rep_region_id and
              ksm_id = ksr_from_ksm_id and
              ksm_from_c_id = c_id and
              ksr_is_active = 1 and
              ksm_is_active = 1 and
              coalesce(ksr_date_begin, '01.01.1990') <= current_date and
              coalesce(ksr_date_end, '31.12.9999') >= current_date
        into :manager_c_id, :manager_fio
    do
    begin
      if (not(exists(select *
                     from hr_pstatus, joblist, jobkind
                     where ps_from_c_id = :manager_c_id and
                           ps_date_begin <= :rep_date and
                           ps_date_end >= :rep_date and
                           ps_status = 1 and
                           ps_from_jl_id = jl_id and
                           jl_kind = jk_id and
                           jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Senior') and
                           ps_from_reg_id = :rep_region_id and
                           ps_is_working = 1))) then
      begin
        ksr_can_sign_loans = 0;
        if (rep_date is not null) then
        begin
          for select distinct dvc_from_reg_id
              from dovlist, dovcateg, dovclients
              where dl_from_dc_id = dc_id and
                    dl_id = dvc_from_dl_id and
                    dvc_from_c_id = :manager_c_id and
                    dc_sign_loans = 1 and
                    dl_status = 1 and
                    dl_date_begin <= :rep_date and
                    dl_date_end >= :rep_date and
                    ((dl_date_end_fact is null) or
                    (dl_date_end_fact >= :rep_date))
              into :reg_id
          do
          begin
            if (reg_id is not null) then
            begin
              if (exists(select *
                         from region_list(:rep_region_id, 1, -1, 0)
                         where id_list = :reg_id)) then
              begin
                ksr_can_sign_loans = 1;
              end
            end
          end
        end
        if (ksr_can_sign_loans = 1) then
          suspend;
      end
    end
  end
  if (rep_is_sign = 0) then -- Список сертификанта
  begin
    reg_head_office = f_hq_id(db_id);

    for select distinct c_id, c_name_fio
        from kontsignreg, clients, kontsignmen
        where ksr_from_reg_id = :rep_region_id and
              ksm_id = ksr_from_ksm_id and
              ksm_from_c_id = c_id and
              ksr_is_active = 1 and
              ksm_is_active = 1 and
              coalesce(ksr_date_begin, '01.01.1990') <= current_date and
              coalesce(ksr_date_end, '31.12.9999') >= current_date
        into :manager_c_id, :manager_fio
    do
    begin
      suspend;
    end

    for select distinct c_id, c_name_fio
        from kontsignreg, clients, kontsignmen, region_list(:rep_region_id, 0, -1, 0)
        where ksr_from_reg_id = id_list and
              ksm_id = ksr_from_ksm_id and
              ksm_from_c_id = c_id and
              ksr_is_active = 1 and
              ksm_is_active = 1 and
              coalesce(ksr_date_begin, '01.01.1990') <= current_date and
              coalesce(ksr_date_end, '31.12.9999') >= current_date
        into :manager_c_id, :manager_fio
    do
    begin
      if (exists(select *
                 from hr_pstatus, joblist, jobkind
                 where ps_from_c_id = :manager_c_id and
                       ps_date_begin <= :rep_date and
                       ps_date_end >= :rep_date and
                       ps_status = 1 and
                       ps_from_jl_id = jl_id and
                       jl_kind = jk_id and
                       jk_system_code in (
                         'JobSystem.LO_Standart',
                         'JobSystem.Region_KM',
                         'JobSystem.Region_Director',
                         'JobSystem.LO_Senior',
                         'JobSystem.LO_Partner',
                         'JobSystem.Director_Islamic_Window') and
                       ps_is_working = 1) or :manager_c_id = 1090100001937) then
      begin
        suspend;
      end
    end
  end

  if (rep_is_sign = 2) then -- Список экспертов-аналитиков
  begin
    for select ps_from_c_id, c_name_fio
        from hr_pstatus, joblist, jobkind, clients
        where ps_from_reg_id = :rep_region_id and
              ps_date_begin <= :rep_date and
              ps_date_end >= :rep_date and
              ps_from_jl_id = jl_id and
              ps_from_c_id = c_id and
              ps_status = 1 and
              jl_kind = jk_id and
              jk_system_code = 'JobSystem.Specialist_Expert' -- Эксперт-аналитик
        into :manager_c_id, :manager_fio
    do
    begin
      suspend;
    end
    manager_c_id = 1000000000004;
    manager_fio = 'КЦ Сирена (для льготных займов)';
    suspend;
  end
end
