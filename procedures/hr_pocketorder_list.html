CREATE PROCEDURE HR_POCKETORDER_LIST(
    HO_NEED_ID DECIMAL(18,0))
RETURNS (
    HO_ID DECIMAL(18,0),
    HO_DATE DATE,
    HO_FROM_REG_ID INTEGER,
    HO_TEXT VARCHAR(1024),
    HO_REASON VARCHAR(255),
    HO_ORDER_TYPE INTEGER,
    HO_ORDER_NO VARCHAR(20),
    HRV_FROM_HO_ID DECIMAL(18,0),
    HRV_NUMBER INTEGER,
    HRV_FROM_C_ID DECIMAL(18,0),
    HRV_DATE_BEGIN DATE,
    HRV_DATE_END DATE,
    HRV_DAYS INTEGER,
    HRV_DAY_BACK DATE,
    HRV_DIRECTION VARCHAR(100),
    HRV_CLASS VARCHAR(20),
    C_NAME_FIO VARCHAR(100),
    C_OFFICE_NAME VARCHAR(100),
    C_JOBLIST VARCHAR(100),
    C_IS_WORK VARCHAR(50),
    HRV_CHANGER_ID DECIMAL(18,0),
    HRV_CHANGER_FIO VARCHAR(100),
    HRV_NOT_CLOSE_ACCESS D_BOOL)
AS
declare variable ps_is_working integer;
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select hrv_from_ho_id, hrv_number, hrv_from_c_id, hrv_date_begin, hrv_date_end, hrv_days, hrv_day_back,
             hrv_direction, hrv_class, ho_id, ho_date, ho_from_reg_id, ho_text, ho_reason, ho_order_type, ho_order_no,
             c_name_fio, hrv_changer_id, hrv_not_close_access
      from hr_voctravel, hr_orders, clients
      where ho_id = hrv_from_ho_id and
            ho_id = :ho_need_id and
            hrv_from_c_id = c_id
      into :hrv_from_ho_id, :hrv_number, :hrv_from_c_id, :hrv_date_begin, :hrv_date_end, :hrv_days, :hrv_day_back,
           :hrv_direction, :hrv_class, :ho_id, :ho_date, :ho_from_reg_id, :ho_text, :ho_reason, :ho_order_type,
           :ho_order_no, :c_name_fio, :hrv_changer_id, :hrv_not_close_access

  do
  begin
    c_office_name = '';
    c_joblist = '';
    ps_is_working = 1;
    select reg_code || ' ' || reg_name, jl_name, ps_is_working
    from hr_pstatus, region, joblist
    where ps_from_reg_id = reg_unicode and
          ps_from_jl_id = jl_id and
          ps_from_c_id = :hrv_from_c_id
    order by ps_date_begin desc, ps_date_end desc
    rows 1
    into :c_office_name, :c_joblist, :ps_is_working;
    if (ps_is_working = 0) then
      c_is_work = 'Уволен';
    else
      c_is_work = 'Работает';

    hrv_changer_fio = '';
    if (hrv_changer_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :hrv_changer_id
      into hrv_changer_fio;
    end
    suspend;
  end
end
