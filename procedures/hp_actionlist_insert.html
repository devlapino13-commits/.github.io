CREATE PROCEDURE HP_ACTIONLIST_INSERT(
    ACL_PARENT_ID ID_TYPE,
    ACL_FROM_REG_ID D_INTEGER,
    ACL_OBJECT_KIND_STR STR_100,
    ACL_OBJECT_ID ID_TYPE,
    ACL_ACTION_CODE STR_100,
    ACL_STATUS D_INTEGER,
    ACL_FROM_UN_ID ID_TYPE)
RETURNS (
    RES_ACL_ID ID_TYPE)
AS
declare variable acl_object_kind    d_integer;
declare variable acl_from_rd_id     d_integer;
declare variable dont_show_in_order d_integer;
declare variable db_id              d_integer;
declare variable lo_id              d_id;
begin
  db_id = f_get_db_id();
  acl_from_rd_id = f_get_rd_id(acl_action_code);
  acl_object_kind = f_objkind(acl_object_kind_str, 'id');

  select acl_id
  from actionlist
  where acl_object_kind = :acl_object_kind and
        acl_object_id = :acl_object_id and
        acl_from_rd_id = :acl_from_rd_id and
        acl_parent_id is null
  order by acl_id desc
  rows 1
  into :res_acl_id;

  if (acl_action_code starting 'OKTELL.Projects.') then
  begin
    if (acl_action_code = 'OKTELL.Projects.Website' and
        acl_parent_id is not null) then
      dont_show_in_order = 1;
    else
      dont_show_in_order = 0;

    insert into actionlist (acl_parent_id, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                            acl_from_un_id, acl_dont_show_in_order)
    values (:acl_parent_id, :acl_from_reg_id, :acl_object_kind, :acl_object_id, :acl_from_rd_id, :acl_status,
            :acl_from_un_id, :dont_show_in_order)


    returning acl_id
    into :res_acl_id;
  end
  else
  if (acl_action_code in ('EPayProjects', 'EPayProjects.ElSom', 'EPayProjects.ElCard', 'EPayProjects.MBank')) then
  begin
    insert into actionlist (acl_parent_id, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                            acl_from_un_id, acl_dont_show_in_order)
    values (:acl_parent_id, :acl_from_reg_id, :acl_object_kind, :acl_object_id, :acl_from_rd_id, :acl_status,
            :acl_from_un_id, 0)


    returning acl_id
    into :res_acl_id;
  end

  else
  if (acl_action_code = 'ActionGroup.Check_loan_fix') then
  begin
    select kal.kal_new_kf_officer
    from kontract kt
    inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id
    where kt.k_id = :acl_object_id and
          kal.is_delete = 0 and
          current_date between kal.kal_date and kal.kal_date_end
    into lo_id;

    insert into actionlist (acl_parent_id, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id, acl_status,
                            acl_from_un_id, acl_dont_show_in_order, acl_respons_c_id)
    values (:acl_parent_id, :acl_from_reg_id, :acl_object_kind, :acl_object_id, :acl_from_rd_id, 0,
            :acl_from_un_id, 0, :lo_id)


    returning acl_id
    into :res_acl_id;
  end
  else
  begin
    if (res_acl_id is null) then
    begin
      insert into actionlist (acl_parent_id, acl_from_reg_id, acl_object_kind, acl_object_id, acl_from_rd_id,
                              acl_status, acl_from_un_id)
      values (:acl_parent_id, :acl_from_reg_id, :acl_object_kind, :acl_object_id, :acl_from_rd_id, :acl_status,
              :acl_from_un_id)
      returning acl_id
      into :res_acl_id;
    end
    else
    begin
      update actionlist
      set acl_status = :acl_status,
          acl_from_un_id = :acl_from_un_id
      where acl_id = :res_acl_id and
            coalesce(acl_status, -777) <> coalesce(:acl_status, -777);
    end
  end
  suspend;
end
