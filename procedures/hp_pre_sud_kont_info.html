CREATE PROCEDURE HP_PRE_SUD_KONT_INFO(
    REP_K_ID ID_TYPE)
RETURNS (
    K_KONTRACT_NO STR_20,
    DISB_DATE D_DATE,
    C_ID ID_TYPE,
    C_NAME_FIO STR_100,
    ZAD_ALL CURR_TYPE,
    ZAD_OS CURR_TYPE,
    ZAD_PERC CURR_TYPE,
    ZAD_FINE CURR_TYPE,
    ZAD_EXTRA CURR_TYPE,
    LAST_PAY_DATE D_DATETIME,
    LAST_PAY_SUM CURR_TYPE,
    LAST_COMMUNICATION D_DATE,
    SBL_NAME STR_255,
    SBA_DATE_BEGIN D_DATE)
AS
declare variable part double precision;
declare variable kc_amount_approved curr_type;
declare variable oek_rd_id d_integer;
declare variable oko_sum curr_type;
declare variable oko_date d_datetime;
declare variable ox_sum curr_type;
declare variable ox_date d_datetime;
begin
  oek_rd_id = f_get_rd_id('OperExtraKind.LawClientExpense');


  for select c_id, c_name_fio, k_kontract_no, k_disb_date, kc_amount_approved,
             kc_amount_approved / k_amount_approved
      from knclient
      inner join kontract on k_id = kc_from_k_id
      inner join clients on c_id = kc_from_c_id
      where k_id = :rep_k_id and
            kc_amount_approved > 0
      into c_id, c_name_fio, k_kontract_no, disb_date, kc_amount_approved, part
  do
  begin
    zad_os = 0;
    zad_perc = 0;
    zad_fine = 0;
    zad_extra = 0;

    select round(sum(zad_os) * :part), round(sum(sh_calc_perc) * :part), round(sum(sh_calc_fine) * :part)
    from aarep_make_zad_list(:rep_k_id, current_date, 1)
    into zad_os, zad_perc, zad_fine;

    select round(coalesce(sum(ox_amount_plus - ox_amount_minus - ox_amount_writeoff), 0) * :part)
    from operextra
    where ox_ol_is_active = 1 and
          ox_from_rd_id = :oek_rd_id and
          ox_object_kind = f_objkind('kontract', 'id') and
          ox_object_id = :rep_k_id and
          ox_ol_date <= current_date
    into zad_extra;

    zad_all = zad_os + zad_perc + zad_fine + zad_extra;

    last_pay_sum = 0;
    last_pay_date = null;
    oko_sum = 0;
    oko_date = null;
    ox_sum = 0;
    ox_date = null;

    select sum(ok_os + ok_perc + ok_fine) * -1, ol_datetime
    from operkred
    inner join operlist on ol_id = ok_from_ol_id
    where ok_from_k_id = :rep_k_id and
          is_delete = 0 and
          ok_type = 4
    group by 2
    order by 2 desc rows 1
    into last_pay_sum, last_pay_date;

    select (oko_os + oko_perc + oko_other) * -1, ol_datetime
    from operkroff
    inner join operlist on ol_id = oko_from_ol_id
    where oko_from_k_id = :rep_k_id and
          is_delete = 0 and
          oko_type = 4
    order by 2 desc rows 1
    into oko_sum, oko_date;

    if (oko_date is not null) then
    begin
      last_pay_sum = oko_sum;
      last_pay_date = oko_date;
    end

    select ox_amount_plus, ol_datetime
    from operextra
    inner join operlist on ol_id = ox_from_ol_id
    where ox_ol_is_active = 1 and
          ox_from_rd_id = :oek_rd_id and
          ox_object_kind = f_objkind('kontract', 'id') and
          ox_object_id = :rep_k_id
    order by 2 desc rows 1
    into ox_sum, ox_date;

    if (ox_date is not null and
        ox_date > last_pay_date) then
    begin
      last_pay_sum = ox_sum;
      last_pay_date = ox_date;
    end

    --  название службы
    sbl_name = '';
    sba_date_begin = null;

    select sbl_name, sba_date_begin
    from sb_akt, sb_list
    where sba_from_k_id = :rep_k_id and
          current_date between sba_date_begin and sba_date_end and
          sba_is_delete = 0 and
          sba_status = 2 and
          sba_from_sbl_id = sbl_id
    into sbl_name, sba_date_begin;

    -- дата послeдней коммуникации
    last_communication = null;

    select max(com_date)
    from hp_pre_sud_communicate(:rep_k_id)
    where rep_type = 0
    into last_communication;
    suspend;
  end
end
