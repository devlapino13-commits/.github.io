CREATE PROCEDURE AAREP_KN_OKSUMMARY_TOTAL(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    REP_PERIOD D_INTEGER,
    PR_NAME D_STR_100,
    DISB_COUNT D_INTEGER,
    DISB_AMOUNT D_CURRENCY,
    ACTIVE_COUNT D_INTEGER,
    ACTIVE_BALANCE D_CURRENCY,
    ACR_PERC D_CURRENCY,
    EXPENSE_RO D_CURRENCY)
AS
declare variable dd_begin d_date;
declare variable dd_end d_date;
begin
  dd_begin = rep_date_begin;
  while (dd_begin < rep_date_end) do
  begin
    dd_end = f_date_month_end(dd_begin);

    rep_period = extract(year from dd_end) * 100 +extract(month from dd_end);
  
    for select pr_name,
         sum(iif(oks_disb_amount <> 0, 1, 0)) as disb_count,
         sum(-oks_disb_amount) as disb_amount,
         sum(iif(oks_balance > 0, 1, 0)) as active_count,
         sum(oks_balance) as active_balance,
         sum(oks_acr_esp_perc) as acr_perc,
         sum(r_ro) as expense_ro
    from aarep_kn_oksummary_rep(:dd_begin, :dd_end)
    group by 1
    into pr_name, disb_count, disb_amount, active_count, active_balance, acr_perc, expense_ro
    do
    begin
      if (active_balance = 0) then
        active_count = 0;
  
      if (disb_count <> 0 or
          disb_amount <> 0 or
          active_count <> 0 or
          active_balance <> 0 or
          acr_perc <> 0 or
          expense_ro <> 0) then
      suspend;
    end

    dd_begin = dateadd(month, 1, dd_begin);
  end
end
