CREATE PROCEDURE PRINT_CLIENT_LGOTA(
    AAREP_K_ID DECIMAL(18,0))
RETURNS (
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_PASS_DATE_EXPIRE DATE,
    C_TYPE INTEGER,
    KC_STATUS INTEGER,
    KC_AMOUNT_APPROVED DECIMAL(18,4),
    KC_AMOUNT_APPROVED_EKV DECIMAL(18,4),
    KC_NUMBER INTEGER,
    KC_PATENT_NO VARCHAR(40),
    KC_PATENT_DATE DATE,
    CA_ADDRESS VARCHAR(255),
    C_FULL_ADDRESS VARCHAR(255),
    KC_CIRCLE INTEGER,
    C_BANK_ACCOUNT_NO VARCHAR(50),
    C_BANK_CODE VARCHAR(10),
    C_BANK_NAME VARCHAR(255),
    C_BANK_CORR_ACC VARCHAR(255),
    K_KOMMIS_AMOUNT DECIMAL(18,4),
    KC_KOMMIS INTEGER,
    KC_BONUS_TEXT VARCHAR(30),
    KC_KOMMIS_LGOTA DECIMAL(18,4),
    KC_NSP_AMOUNT DECIMAL(18,4),
    KC_KOMMIS_AND_NSP DECIMAL(18,4))
AS
declare variable kc_from_c_id decimal(18,0);
declare variable k_kommis_perc decimal(18,7);
declare variable k_amount_approved decimal(18,4);
declare variable st_staff integer;
declare variable st_client integer;
declare variable k_disb_date date;
declare variable k_bonus integer;
declare variable k_nsp_percent decimal(18,7);
declare variable k_kolich_client_approved integer;
declare variable k_nsp_amount decimal(18,4);
declare variable kc_nsp_amount_all decimal(18,4);
declare variable kol integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;

  k_nsp_percent = 0;
  select k_kommis_perc, k_amount_approved, k_disb_date, k_kolich_client_approved
  from kontract
  where k_id = :aarep_k_id
  into :k_kommis_perc, :k_amount_approved, :k_disb_date, :k_kolich_client_approved;

  if (exists(select *
             from nspconfig_data
             where nsd_from_nsm_id = 0 and
                   nsd_date_begin <= :k_disb_date and
                   nsd_date_end >= :k_disb_date)) then
  begin
    select nsd_percent
    from nspconfig_data
    where nsd_from_nsm_id = -2 -- нсп от доп.процента
          and
          nsd_date_begin <= :k_disb_date and
          nsd_date_end >= :k_disb_date
    into :k_nsp_percent;
    if (k_nsp_percent is null) then
      k_nsp_percent = 0;
  end
  k_nsp_amount = 0;

  kc_nsp_amount_all = 0;
  kol = 0;

  for select c_name_fio, c_birth_date, c_pass_no, c_pass_date, c_pass_who, c_pass_date_expire, c_type, kc_status,
             kc_amount_approved, kc_amount_approved * k_index_curr_rate, kc_number, kc_patent_no, kc_patent_date,
             adl_address_str, kc_circle, kc_from_c_id, kc_kommis_lgota
      from clients, knclient, kontract, addrclient, addrlist
      where kc_from_k_id = :aarep_k_id and
            k_id = :aarep_k_id and
            kc_from_c_id = c_id and
            kc_amount_approved <> 0 and
            kc_amount_approved is not null and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id and
            kc_kommis_lgota > 0

      order by kc_number
      into :c_name_fio, :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who, :c_pass_date_expire, :c_type, :kc_status,
           :kc_amount_approved, :kc_amount_approved_ekv, :kc_number, :kc_patent_no, :kc_patent_date, :c_full_address,
           :kc_circle, :kc_from_c_id, :kc_kommis_lgota
  do
  begin
    select c_last_bonus_status, c_last_bonus_status_str
    from hp_kontstate_client_status(:kc_from_c_id, :k_disb_date)
    into :k_bonus, kc_bonus_text;

    ca_address = c_full_address;

    k_kommis_amount = cast(k_amount_approved * k_kommis_perc as decimal(15,0));
    k_kommis_amount = cast(k_kommis_amount / 100 as decimal(15,2));
    c_bank_account_no = '';
    c_bank_code = '';
    c_bank_name = '';
    kc_kommis = cast(kc_amount_approved * k_kommis_perc as decimal(15,0));
    kc_kommis = cast(kc_kommis / 100 as decimal(15,2));

    k_kommis_amount = cast(k_amount_approved * k_kommis_perc as decimal(15,0));
    k_kommis_amount = cast(k_kommis_amount / 100 as decimal(15,2));
    k_nsp_amount = round(k_kommis_amount * k_nsp_percent / 100);

    kc_nsp_amount = round(kc_kommis * k_nsp_percent / 100);
    if (kol < k_kolich_client_approved) then
      kc_nsp_amount_all = kc_nsp_amount_all + kc_nsp_amount;
    else
      kc_nsp_amount = k_nsp_amount - kc_nsp_amount_all;

    kc_kommis_and_nsp = kc_kommis + kc_nsp_amount;

    if (kc_from_c_id is not null) then
    begin
      select bc_account_no, bc_from_b_id, b_name, b_corr_acc
      from bankclient, banks
      where bc_from_c_id = :kc_from_c_id and
            bc_from_b_id = b_id and
            bc_is_main = 1 and
            bc_is_active = 1 rows 1
      into :c_bank_account_no, :c_bank_code, :c_bank_name, :c_bank_corr_acc;
    end
    suspend;
  end
end
