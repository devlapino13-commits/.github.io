CREATE PROCEDURE HP_PREPARE_FIO(
    SURNAME D_STR_100,
    NAME D_STR_100,
    PATRONYMIC D_STR_100,
    SURNAME_ENG D_STR_100,
    NAME_ENG D_STR_100,
    PATRONYMIC_ENG D_STR_100,
    NOT_CREATE D_BOOL = 0)
RETURNS (
    FL_ID D_ID,
    FL_ID_ENG D_ID,
    C_NAME_FIO D_STR_100,
    C_NAME_FIO_ENG D_STR_100)
AS
declare variable v_fio d_str_255;
declare variable v_from_lang d_str_10;
declare variable v_country d_str_10;
declare variable db_id d_id;
begin
  db_id = f_get_db_id();

  v_fio = coalesce(surname, '') || ' ' || coalesce(name, '') || ' ' || coalesce(patronymic, '');

  if (not_create = 0) then
    select fl_id
    from hp_fiolist_get_or_create(:surname, :name, :patronymic)
    into :fl_id;

  if (db_id = 41701) then
  begin
    v_from_lang = 'cyr';
    v_country = 'kg';

    c_name_fio = f_translit_fio_grs(:v_from_lang, :v_from_lang, :v_fio, :v_country);
  end
  else if (db_id = 39801) then
  begin
    v_from_lang = 'orig';
    v_country = 'kz';

    c_name_fio = f_translit_fio_grs(:v_from_lang, :v_from_lang, :v_fio, :v_country);
    v_from_lang = 'cyr';
  end
  else if (db_id = 94401) then
  begin
    v_from_lang = 'lat';
    v_country = 'az';

    surname = f_correct_fio(replace(surname, 'I', 'ı'));
    name = f_correct_fio(replace(name, 'I', 'ı'));
    patronymic = f_correct_fio(replace(patronymic, 'I', 'ı'));

    c_name_fio =
        coalesce(surname, '') || ' ' ||
        coalesce(name, '') || ' ' ||
        coalesce(patronymic, '');
  end
  else
  begin
    v_from_lang = 'cyr';
    v_country = 'ru';

    c_name_fio = v_fio;
  end

  -- eng fio перводим только если не передали их
  if (surname_eng is null) then
    surname_eng = f_translit_fio_grs(v_from_lang, 'eng', surname, v_country);

  if (name_eng is null) then
    name_eng = f_translit_fio_grs(v_from_lang, 'eng', name, v_country);

  if (patronymic_eng is null) then
    patronymic_eng = f_translit_fio_grs(v_from_lang, 'eng', patronymic, v_country);

  c_name_fio_eng =
      coalesce(surname_eng, '') || ' ' ||
      coalesce(name_eng, '') || ' ' ||
      coalesce(patronymic_eng, '');

  if (not_create = 0) then
    select fl_id
    from hp_fiolist_get_or_create(:surname_eng, :name_eng, :patronymic_eng)
    into :fl_id_eng;

  suspend;
end
