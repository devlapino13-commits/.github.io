Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_FUELGENREG_CALC(
    DT_BEG D_DATE,
    DT_END D_DATE,
    REG_ID D_INTEGER,
    PET_ID D_INTEGER,
    WHOURS D_CURRENCY,
    WRATE D_CURRENCY)
RETURNS (
    FG_START D_CURRENCY,
    FG_START_SUM D_CURRENCY,
    FG_FILLUP D_CURRENCY,
    FG_FILLUP_SUM D_CURRENCY,
    FG_USEDUP D_CURRENCY,
    FG_USEDUP_SUM D_CURRENCY,
    FG_END D_CURRENCY,
    FG_END_SUM D_CURRENCY,
    ERR_TXT D_STR_255)
AS
declare variable usedup d_currency;
declare variable fillup d_currency;
declare variable fillup_cost d_currency;
declare variable fillup_sum d_currency;
declare variable calc_usedup d_currency;
begin
  err_txt = '';
  if (pet_id = 0) then
    pet_id = -1;

  select
    round(sum(ped_saldo_liter), 2),
    round(sum(ped_saldo_summa), 2),
    round(sum(ped_fill_liter + ped_fill_liter_talon), 2),
    round(sum(ped_fill_summa), 2)
  from aarep_transport_petrol(:reg_id, :pet_id, :dt_beg, :dt_end)
  into
    :fg_start,
    :fg_start_sum,
    :fg_fillup,
    :fg_fillup_sum;

  fg_usedup = round(whours * wrate, 2);

  if (fg_usedup > fg_start)
  then calc_usedup = fg_usedup - fg_start;
    else calc_usedup = fg_usedup;
  fg_usedup_sum = 0;
  for
    select
      peo_liter,
      peo_cost
    from aarep_trans_talon_saldo(:pet_id, :dt_end)
    where /*peo_from_reg_id = :reg_id and*/ peo_liter > 0
    order by peo_bill_id desc
  into
    :fillup,
    :fillup_cost
  do
  begin
    fg_usedup_sum = fg_usedup_sum + round(iif(calc_usedup < fillup, calc_usedup, fillup) * fillup_cost, 2);
    calc_usedup = iif(calc_usedup < fillup, 0, calc_usedup - fillup);
    if (calc_usedup <= 0) then
      break;
  end
  if (fg_usedup > fg_start) then
    fg_usedup_sum = fg_usedup_sum + fg_start_sum;
  if (fg_usedup = 0) then
    fg_usedup_sum = 0;
  fg_end = fg_start + fg_fillup - fg_usedup;
  fg_end_sum = fg_start_sum + fg_fillup_sum - fg_usedup_sum;

  fg_start = coalesce(fg_start, 0);
  fg_start_sum = coalesce(fg_start_sum, 0);
  fg_fillup = coalesce(fg_fillup, 0);
  fg_fillup_sum = coalesce(fg_fillup_sum, 0);
  fg_usedup = coalesce(fg_usedup, 0);
  fg_usedup_sum = coalesce(fg_usedup_sum, 0);
  fg_end = coalesce(fg_end, 0);
  fg_end_sum = coalesce(fg_end_sum, 0);

  suspend;
end
