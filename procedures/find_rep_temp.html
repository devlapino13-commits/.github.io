CREATE PROCEDURE FIND_REP_TEMP(
    REP_STR VARCHAR(255))
RETURNS (
    REP_ID INTEGER)
AS
declare variable temp_id integer;
begin
  /* Procedure Text */
  if (rep_str <> '') then
  begin
    rep_str = upper(:rep_str);
    for select distinct rl_id
        from reportlist
        where (upper(rl_name_rus) like '%' || :rep_str || '%') or
              (upper(rl_hint_rus) like '%' || :rep_str || '%') or
              (upper(rl_name_eng) like '%' || :rep_str || '%') or
              (upper(rl_hint_eng) like '%' || :rep_str || '%')
        into :temp_id
    do
    begin
      for select id_list
          from report_list(:temp_id, 1, -1, 0)
          into :rep_id
      do
        suspend;
    end
    for select distinct rt_from_rl_id
        from reptemplates
        where (upper(rt_name) like '%' || :rep_str || '%') or
              (upper(rt_name_eng) like '%' || :rep_str || '%')
        into :temp_id
    do
    begin
      for select id_list
          from report_list(:temp_id, 1, -1, 0)
          into :rep_id
      do
        suspend;
    end
  end
  else
  begin
    for select distinct rl_id
        from reportlist
        into :rep_id
    do
      suspend;
  end
end
