CREATE PROCEDURE HP_ONLINE_LOAN_LIST(
    GROUP_ID D_INTEGER)
AS
declare variable exa_id d_id;
declare variable oa_id d_integer;
declare variable sql_request d_str_255;
declare variable obj_kind d_integer;
declare variable eic_id d_id;
begin
  obj_kind = f_objkind('miniapp', 'id');
  eic_id = (select eic_id from extinfocfg where eic_code = 'check_in_ru_db');

  for select distinct oa.oa_from_exa_id, oa_id
      from online_app oa
      inner join expressapp exa on exa.exa_id = oa.oa_from_exa_id and
            exa.exa_group_id = :group_id
      inner join kibminiapp kbm on kbm.kbm_id = oa.oa_from_kbm_id and
            kbm.kbm_is_checked = 1
      inner join kibminiapp_link kbl on kbl.kbl_from_kbm_id = kbm.kbm_id and
            kbl.kbl_object_id = exa.exa_from_ma_id and
            kbl.kbl_object_kind = :obj_kind
      inner join extinfolist eil on eil.eil_from_c_id = oa.oa_from_c_id and
            eil.eil_object_kind = :obj_kind and
            eil.eil_object_id = exa.exa_from_ma_id and
            eil.eil_from_eic_id = :eic_id and
            eil.eil_status = 2 and
            eil.eil_kk_result = 0  --просрочек нет  (определяется в процедуре HP_EXTINFO_SET_RESULT)
      where oa.oa_status = 1
      into exa_id, oa_id
  do
  begin
    sql_request = 'execute procedure hp_online_loan_do_process(' || :oa_id || ', ' || :exa_id || ')';

    execute statement sql_request
        with autonomous transaction;
  end
end
