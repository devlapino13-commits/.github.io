CREATE PROCEDURE AAREP_GET_ACTIONLIST_QUOTA(
    K_ID DECIMAL(18,0))
RETURNS (
    RES_ACL_ID DECIMAL(18,0),
    RES_ID INTEGER,
    RES_TXT VARCHAR(255),
    RES_TOTAL_QUOTA INTEGER,
    RES_QUOTA_SALDO INTEGER,
    CURR_IN_ORDER INTEGER,
    ESTIMATE_IN_ORDER INTEGER,
    RES_TOTAL_QUOTED INTEGER)
AS
declare variable total_quota_used integer;
declare variable acl_from_reg_id integer;
declare variable acl_quota_used integer;
declare variable rd_id integer;
declare variable total_in_order integer;
declare variable acl_id decimal(18,0);
declare variable aco_orderno decimal(18,0);
begin
  select first 1 acl_id, acl_from_reg_id, coalesce(acl_quota_used, 0), acl_from_rd_id
  from actionlist
  where acl_object_kind = 9 and
        acl_object_id = :k_id
  order by acl_datetime desc
  into :res_acl_id, :acl_from_reg_id, :acl_quota_used, :rd_id;

  res_id = 1;
  res_txt = '';
  res_total_quota = 0;
  res_quota_saldo = 0;
  total_quota_used = 0;

  if (acl_quota_used <> 0) then
  begin
    res_id = -1;
    res_txt = 'Договор уже находится в приортитетном обслуживании.';
    suspend;
    exit;
  end

  -- Лимит трех офисов увеличен согласно письма Ю.Каштановой от 27.04.2021
  if (acl_from_reg_id in (
        1150100, -- Южно-Сахалинск
        1180100, -- Хабаровск
        1190100) -- Владивосток
      ) then
    res_total_quota = 5;
  else
  if (exists(select *
             from region
             where reg_unicode = :acl_from_reg_id and
                   reg_logic_label like '%.mega_office.%')) then
    res_total_quota = 5;
  else
    res_total_quota = 3;

  select sum(iif(coalesce(acl_quota_used, 0) > 0, 1, 0))
  from actionlist
  where acl_date = current_date and
        acl_parent_id is null and
        acl_object_kind = 9 and
        acl_from_rd_id = :rd_id and
        acl_from_reg_id = :acl_from_reg_id
  into :total_quota_used;
  if (total_quota_used is null) then
    total_quota_used = 0;

  res_quota_saldo = res_total_quota - total_quota_used;
  if (res_quota_saldo < 0) then
    res_quota_saldo = 0;

  total_in_order = 0;
  estimate_in_order = 0;
  res_total_quoted = 0;

  for select aco_orderno, aco_id
      from actionorder
      where aco_from_rd_id = :rd_id
      order by aco_orderno, aco_id
      into :aco_orderno, :acl_id
  do
  begin
    total_in_order = total_in_order + 1;
    if (acl_id = res_acl_id) then
      curr_in_order = total_in_order; -- Текущий номер в очереди
    if (aco_orderno = 0 and
        acl_id < res_acl_id) then
      estimate_in_order = estimate_in_order + 1; -- ожидаемый номер в очереди если перевести на приоритетный
    if (aco_orderno = 0) then
      res_total_quoted = res_total_quoted + 1; -- Всего приоритетных
  end
  estimate_in_order = estimate_in_order + 1;

  if (res_quota_saldo = 0) then
  begin
    res_id = -1;
    res_txt = 'Ежедневная квота на офис: ' || res_total_quota || '. На сегодня лимит исчерпан. Ваш номер в очереди: ' || curr_in_order;
    suspend;
    exit;
  end

  if (curr_in_order <= 2) then
  begin
    res_id = -1;
    res_txt = 'Вы уже ' || iif(curr_in_order = 2, 'второй', 'первый') || ' в очереди. Нельзя использовать квоту.';
    suspend;
    exit;
  end

  res_id = 1;
  res_txt = '';
  suspend;
end
