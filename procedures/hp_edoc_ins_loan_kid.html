CREATE PROCEDURE HP_EDOC_INS_LOAN_KID(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    RES_TEXT D_STR_255)
AS
declare variable DB_ID D_INTEGER;
declare variable K_STATUS D_SMALLINT;
begin
    -- RFCKZ-805
    -- Создает эл. документ КИД + проект договора после утверждения договора менеджером.
    -- Вызывается из hp_approve_loan.
    res_text = '';
    db_id = f_get_db_id();

    if (db_id <> 39801 or
        f_get_const('kid_active') <> 1) then
        exit;

    select k_status
    from kontract
    where k_id = :k_id
    into k_status;

    if (coalesce(k_status, 0) <> 1) then
    begin
        res_text = 'Сначала утвердите договор займа';
        suspend;
        exit;
    end

    -- При переутверждении договора предыдущий КИД аннулируется явно.
    -- В hp_edoc_insert при edc_object_uniq = 1 проверка подписанного документа
    -- стоит раньше вызова hp_edoc_cancel и не дала бы создать новый.
    execute procedure hp_edoc_cancel('loan.kid', f_objkind('kontract', 'name'), :k_id);

    select edl_id, res_text
    from hp_edoc_insert('loan.kid', f_objkind('kontract', 'id'), :k_id, :un_id)
    into edl_id, res_text;

    suspend;
end
