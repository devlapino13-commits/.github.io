CREATE PROCEDURE HP_CHECK_LOAN_PERIOD(
    MA_ID D_INTEGER)
RETURNS (
    ERR_TEXT STR_255)
AS
declare variable ma_kolich_repay d_integer;
declare variable ma_summa_minus curr_type;
declare variable pr_code str_100;
declare variable ma_date_oper d_date;
declare variable c_id d_id;
declare variable c_curr_ball d_integer;
begin
  err_text = '';

  select ma_kolich_repay, ma_summa_minus, pr_code, ma_date_oper, mac.mac_from_c_id
  from miniapp ma
  inner join miniappclient mac on mac.mac_from_ma_id = ma.ma_id
  inner join product on pr_id = ma_from_pr_id
  where ma_id = :ma_id and
        ma_is_active = 1 and
        ma_date_end = '31.12.9999' and
        mac.mac_amount > 0
  into ma_kolich_repay, ma_summa_minus, pr_code, ma_date_oper, c_id;

  c_curr_ball = f_point_get(11, ma_id, c_id, ma_date_oper);

  if (pr_code containing '.reliable_client.' or pr_code containing '.universal.' or pr_code containing '.oivo.' or pr_code containing '.par.') then
  begin
      if (ma_summa_minus between 0 and 40000) then
      begin
         -- RFCRU-5704
         if (ma_date_oper < cast('01.07.2026' as date)) then
         begin
            if (c_curr_ball <= 45 and
                ma_kolich_repay not between 3 and 12) then
              err_text = 'У клиента меньше или равно 45 баллов, сумма займа меньше или равно 40 000 рублей, срок должен быть 3-12 мес.';
            if (c_curr_ball between 46 and 60 and
                ma_kolich_repay not between 3 and 24) then
              err_text = 'У клиента от 46 до 60 баллов, сумма займа меньше или равно 40 000 рублей, срок должен быть 3-24 мес.';
            if (c_curr_ball > 60 and
                ma_kolich_repay not between 3 and 24) then
              err_text = 'У клиента больше 60 баллов, сумма займа меньше или равно 40 000 рублей, срок должен быть 3-24 мес.';
         end else
         begin
            if (c_curr_ball <= 12 and
                ma_kolich_repay not between 3 and 12) then
              err_text = 'У клиента меньше или равно 12 баллов, сумма займа меньше или равно 40 000 рублей, срок должен быть 3-12 мес.';
            if (c_curr_ball >= 13 and
                ma_kolich_repay not between 3 and 24) then
              err_text = 'У клиента больше или равно 13 баллов, сумма займа меньше или равно 40 000 рублей, срок должен быть 3-24 мес.';
         end
      end
      else
      if (ma_summa_minus between 40001 and 80000) then
      begin
        if (ma_date_oper < cast('01.07.2026' as date)) then
        begin
          if (c_curr_ball <= 40 and
              ma_kolich_repay not between 6 and 24) then
            err_text = 'У клиента меньше или равно 40 баллов, сумма займа от 40001 до 80000 рублей, срок должен быть 6-24 мес.';
          if (c_curr_ball between 41 and 45 and
              ma_kolich_repay not between 6 and 24) then
            err_text = 'У клиента от 41 до 45 баллов, сумма займа от 40001 до 80000 рублей, срок должен быть 6-24 мес.';
          if (c_curr_ball > 45 and
              ma_kolich_repay not between 6 and 24) then
            err_text = 'У клиента больше 45 баллов, сумма займа от 40001 до 80000 рублей, срок должен быть 6-24 мес.';
        end else
        begin
            if (c_curr_ball <= 12 and
                ma_kolich_repay not between 6 and 12) then
              err_text = 'У клиента меньше или равно 12 баллов, сумма займа от 40001 до 80000 рублей, срок должен быть 6-12 мес.';
            if (c_curr_ball >= 13 and
                ma_kolich_repay not between 6 and 24) then
              err_text = 'У клиента больше или равно 13 баллов, сумма займа от 40001 до 80000 рублей, срок должен быть 6-24 мес.';
        end

      end
      else
      if (ma_summa_minus between 80001 and 100000) then
      begin
        if (ma_date_oper < cast('01.07.2026' as date)) then
        begin
          if (c_curr_ball <= 40 and
              ma_kolich_repay not between 9 and 24) then
            err_text = 'У клиента меньше или равно 40 баллов, сумма займа от 80001 до 100000 рублей, срок должен быть 9-24 мес.';
          if (c_curr_ball between 41 and 45 and
              ma_kolich_repay not between 9 and 24) then
            err_text = 'У клиента от 41 до 45 баллов, сумма займа от 80001 до 100000 рублей, срок должен быть 9-24 мес.';
          if (c_curr_ball > 45 and
              ma_kolich_repay not between 9 and 24) then
            err_text = 'У клиента больше 45 баллов, сумма займа от 80001 до 100000 рублей, срок должен быть 9-24 мес.';
        end else
        begin
            if (c_curr_ball <= 12 and
                ma_kolich_repay not between 9 and 12) then
              err_text = 'У клиента меньше или равно 12 баллов, сумма займа от 80001 до 100000 рублей, срок должен быть 9-12 мес.';
            if (c_curr_ball >= 13 and
                ma_kolich_repay not between 9 and 24) then
              err_text = 'У клиента больше или равно 13 баллов, сумма займа от 80001 до 100000 рублей, срок должен быть 9-24 мес.';
        end
      end
      else
      if (ma_summa_minus between 100001 and 150000) then
      begin
        if (ma_date_oper < cast('01.07.2026' as date)) then
        begin
            if (c_curr_ball between 46 and 60 and
                ma_kolich_repay not between 13 and 24) then
              err_text = 'У клиента от 46 до 60 баллов, сумма займа от 100001 до 150000 рублей, срок должен быть 13-24 мес.';
            if (c_curr_ball > 60 and
                ma_kolich_repay not between 13 and 24) then
              err_text = 'У клиента больше 60 баллов, сумма займа от 100001 до 150000 рублей, срок должен быть 13-24 мес.';
        end else
        begin
            if (c_curr_ball >= 51 and
                ma_kolich_repay not between 13 and 24) then
              err_text = 'У клиента больше или равно 51 баллов, сумма займа от 100001 до 150000 рублей, срок должен быть 13-24 мес.';
        end
      end
      else
      if (ma_summa_minus > 150000) then
      begin

        /*if (c_curr_ball between 46 and 60 and
            ma_kolich_repay not between 18 and 24) then
          err_text = 'У клиента от 46 до 60 баллов, сумма займа от 100001 до 150000 рублей, срок должен быть 18-24 мес.';*/
        if (ma_date_oper < cast('01.07.2026' as date)) then
        begin
            if (c_curr_ball > 60 and
                ma_kolich_repay not between 18 and 24) then
              err_text = 'У клиента больше 60 баллов, сумма займа больше 150000 рублей, срок должен быть 18-24 мес.';
        end else
        begin
            if (c_curr_ball >= 51 and
                ma_kolich_repay not between 18 and 24) then
              err_text = 'У клиента больше или равно 51 баллов, сумма займа больше 150000 рублей, срок должен быть 18-24 мес.';
        end
      end
  end
  if (pr_code containing '.zalog.' or pr_code containing '.selfemployed.' or
      pr_code containing '.business.') then
  begin
      if (ma_kolich_repay not between 6 and 24) then
        err_text = 'Срок должен быть 6-24 мес.';

      if (ma_summa_minus <= 40000 and
          ma_kolich_repay < 3) then
        err_text = 'Сумма займа меньше 40000, срок должен быть от 3 мес.';

      if (ma_summa_minus between 40001 and 80000 and
          ma_kolich_repay < 6) then
        err_text = 'Сумма займа от 40001 до 80000, срок должен быть от 6 мес.';

      if (ma_summa_minus between 80001 and 100000 and
          ma_kolich_repay < 9) then
        err_text = 'Сумма займа от 80001 до 100000, срок должен быть от 9 мес.';

      if (ma_summa_minus between 100001 and 150000 and
          ma_kolich_repay < 13) then
        err_text = 'Сумма займа от 100001 до 150000, срок должен быть от 13 мес.';

      if (ma_summa_minus > 150000 and
          ma_kolich_repay < 18) then
        err_text = 'Сумма займа больше 150 000 срок должен быть от 18 мес.';
  end

  suspend;
end
