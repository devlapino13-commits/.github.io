CREATE PROCEDURE HP_CLOSEMONTHLIST_FILL(
    CMM_ID D_INTEGER NOT NULL)
AS
declare variable cmm_sql d_str_8100;
declare variable rep_date_begin d_date;
declare variable rep_date_end d_date;
declare variable sql_text d_str_8100;
declare variable cmm_kind d_smallint;
begin
  delete from closemonthlist
  where cml_from_cmm_id = :cmm_id;

  select cmm_date_begin, cmm_date_end, cmm_sql, cmm_kind
  from closemonthmain
  where cmm_id = :cmm_id
  into rep_date_begin, rep_date_end, cmm_sql, cmm_kind;

  cmm_sql = replace(cmm_sql, ':rep_date_begin', '''' || f_datetostr(:rep_date_begin) || '''');
  cmm_sql = replace(cmm_sql, ':rep_date_end', '''' || f_datetostr(:rep_date_end) || '''');
  cmm_sql = replace(cmm_sql, ':cmm_id', cmm_id);

  if (cmm_kind = 1) then
  begin
    execute statement (cmm_sql);
  end else
  if (cmm_kind = 2) then
  begin
    for execute statement (cmm_sql)
    into sql_text
    do
    begin
      insert into closemonthlist (cml_from_cmm_id, cml_sql)
      values (:cmm_id, :sql_text);
    end
  end else
    exception e_message f_format('Неверный cmm_kind %1 для вызова HP_CLOSEMONTHLIST_FILL', cmm_kind);
end
