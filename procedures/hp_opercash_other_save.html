CREATE PROCEDURE HP_OPERCASH_OTHER_SAVE(
    OPER_DATE D_DATE,
    REG_ID D_INTEGER,
    OPER_SIGN D_INTEGER,
    DOC_NO D_INTEGER,
    MEMO D_STR_2048,
    CLIENT_ID D_ID,
    CORR_OFFICE_ID D_INTEGER,
    DT_ACC D_STR_CODE,
    KT_ACC D_STR_CODE,
    AMOUNT D_CURRENCY,
    OB_DETAIL D_STR_LONG,
    UN_ID D_ID,
    MAIN_OL_ID D_ID)
RETURNS (
    OL_ID ID_TYPE,
    RES_TEXT D_STR_LONG,
    RES_MAIN_OL_ID D_ID)
AS
declare variable ob_client_dt id_type;
declare variable ob_client_kt id_type;
declare variable oper_subtype d_integer;
declare variable doc_kind d_integer;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_doc_kind d_integer;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable total_line d_integer;
declare variable res_line_no d_integer;
begin
-- Процедура сохранения денежной операции
  ol_id = 0;
  res_text = '';

  if (client_id = 0) then
    client_id = null;
  if (corr_office_id = 0) then
    corr_office_id = null;

  if (res_text = '' and
      oper_date is null) then
    res_text = 'Дата не указана';
  if (res_text = '' and
      coalesce(reg_id, 0) = 0) then
    res_text = 'Офис не указан';

  if (oper_sign <> 0) then
  begin
    if (res_text = '' and
        coalesce(dt_acc, '') = '') then
      res_text = 'Дебет счет не указан';
    if (res_text = '' and
        coalesce(kt_acc, '') = '') then
      res_text = 'Кредит счет не указан';
    if (res_text = '' and
        coalesce(amount, 0) = 0) then
      res_text = 'Сумма не указана';
    if (res_text = '' and
        coalesce(memo, '') = '') then
      res_text = 'Назначение платежа не указано';
  end

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end

  ob_client_dt = null;
  ob_client_kt = null;

  if (oper_sign = 0) then
  begin
    oper_subtype = 1;
    doc_kind = 9; -- Приходный ордер
  end else
  if (oper_sign = 1) then
  begin
    oper_subtype = 1;
    ob_client_kt = client_id;
    doc_kind = 1; -- Приходный ордер
  end else
  begin
    oper_subtype = 2;
    ob_client_dt = client_id;
    doc_kind = 0; -- Расходный ордер
  end

  insert into operlist (ol_parent_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no, ol_doc_no_str, ol_doc_kind, ol_memo,
                        ol_status, is_delete, ol_user_id, ol_document_client,
                        ol_document_office, ol_oper_subtype)
  values (:main_ol_id, :reg_id, 2, -- денежная операция
          :oper_date, :doc_no, :doc_no, :doc_kind, :memo, 0, -- ol_status
          0, -- is_delete
          :un_id, :client_id,
          :reg_id, :oper_subtype)
  returning (ol_id)
  into ol_id;

  res_main_ol_id = ol_id;

  if (amount <> 0) then
  begin
    if (corr_office_id is null or corr_office_id = reg_id) then
    begin
      insert into operbook (ob_from_ol_id, ob_summa_val, ob_summa,
                            ob_account_dt, ob_client_dt,
                            ob_account_kt, ob_client_kt, ob_detail,
                            ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
      values (:ol_id,  :amount, :amount,
              :dt_acc, :ob_client_dt, :kt_acc, :ob_client_kt,
              :ob_detail, :oper_date, 1, :reg_id);
    end else
    begin
      -- Проводки с учетом лоро-ностро если выдается из другого офиса
      for select res_reg_id, res_acc_dt, res_acc_kt, total_line, res_line_no
          from aarep_loronostro_link(:reg_id, :dt_acc, :kt_acc)
          into :res_reg_id, :res_acc_dt, :res_acc_kt, :total_line, :res_line_no
      do
      begin
        if (res_line_no = 1 and total_line > 1) then
        begin
          insert into operbook (ob_from_ol_id, ob_summa_val, ob_summa,
                                ob_account_dt, ob_client_dt,
                                ob_account_kt, ob_client_kt, ob_detail,
                                ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
          values (:ol_id,  :amount, :amount,
                  :res_acc_dt, :ob_client_dt, :res_acc_kt, :ob_client_kt,
                  :ob_detail, :oper_date, 1, :reg_id);
        end
        else
        begin
          if (main_ol_id is null) then
            main_ol_id = ol_id;

          ol_doc_kind = 9;

          -- перевод между офисами
          ol_oper_type = 2;
          ol_oper_subtype = 20;

          -- Вставка новой операции в Operlist
          insert into operlist (ol_parent_id, ol_from_reg_id, ol_date, ol_doc_no, ol_doc_no_str, ol_oper_type, ol_oper_subtype,
                                ol_doc_kind, ol_memo, ol_user_id, ol_document_client, ol_doc_client_name, ol_is_client)
          values (:main_ol_id, :res_reg_id, :oper_date, :doc_no, :doc_no, :ol_oper_type, :ol_oper_subtype,
                  :ol_doc_kind, :memo, :un_id, :client_id, '', iif(:client_id is not null, 1, 0))
          returning (ol_id)
          into ol_id;

          -- Вставка новой проводки в Operbook
          insert into operbook (ob_from_ol_id, ob_curr_rate, ob_account_dt, ob_account_kt, ob_summa_val,
                                ob_summa, ob_detail, ob_ol_date, ob_ol_is_active, ob_ol_reg_id)
          values (:ol_id, 1, :res_acc_dt, :res_acc_kt, :amount, :amount, :ob_detail, :oper_date,
                  1, :res_reg_id);
        end
      end
    end
  end

  suspend;
end
