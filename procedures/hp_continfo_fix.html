CREATE PROCEDURE HP_CONTINFO_FIX(
    NEW_C_ID D_ID,
    NEED_SUSPEND D_SMALLINT = 0)
RETURNS (
    C_ID D_ID,
    RD_CODE D_STR_CODE,
    CURR_VALUE D_STR_CODE,
    NEW_VALUE D_STR_CODE,
    F_NAME VARCHAR(50))
AS
declare variable rd_id d_integer;
declare variable opr_id d_id;
declare variable orig_opr_id d_id;
declare variable last_cell d_str_code;
declare variable ok_cell d_str_code;
declare variable cv_verify_datetime d_datetime;
declare variable ovs_datetime d_datetime;
begin
  for select cci_object_id
  from continfo
  where cci_object_kind = 0 and
        cci_object_id = :new_c_id
  into c_id
  do
  begin
    for select rd_id, rd_code
    from refdata
    where rd_main_code = 'contactinfo' and
          rd_main_code <> rd_code
    into rd_id, rd_code
    do
    begin
      f_name = replace(rd_code, 'contactinfo.', '');
      f_name = 'last_' || f_name;
      if (f_name = 'last_insta') then
        continue;

      orig_opr_id = null;
      select opr_id
      from objectprops
      where opr_object_kind = 0 and
            opr_object_id = :c_id and
            opr_subject_kind = -1 and
            opr_subject_id = :rd_id
      into orig_opr_id;

      if (orig_opr_id is not null) then
        execute procedure hp_objectvalues_dates(orig_opr_id);

      curr_value = '';
      execute statement ('select ' || f_name || ' from continfo where cci_object_kind = 0 and cci_object_id = :c_id')
                        (c_id := c_id)
      into curr_value;

      opr_id = null;
      new_value = '';
      select opr_id, coalesce(ovs_value, '')
      from objectprops, objectvalues
      where opr_id = :orig_opr_id and
            ovs_from_opr_id = opr_id and
            opr_actual_ovs_id = ovs_id and  -- actual
            coalesce(ovs_value, '') <> :curr_value
      into opr_id, new_value;
  
      if (opr_id is not null) then
      begin
        if (need_suspend = 1) then
          suspend;

        execute statement ('update continfo set ' || f_name || ' = :new_value where cci_object_kind = 0 and cci_object_id = :c_id;')
                          (c_id := c_id, new_value := new_value);
      end

      f_name = replace(rd_code, 'contactinfo.', '');
      f_name = 'ok_' || f_name;

      curr_value = '';
      execute statement ('select ' || f_name || ' from continfo where cci_object_kind = 0 and cci_object_id = :c_id')
                        (c_id := c_id)
      into curr_value;

      if (rd_code in ('contactinfo.cell_main', 'contactinfo.cell_add', 'contactinfo.email')) then
      begin
        opr_id = null;
        new_value = '';
        select opr_id, coalesce(ovs_value, '')
        from objectprops, objectvalues
        where opr_id = :orig_opr_id and
              ovs_from_opr_id = opr_id and
              opr_fact_ovs_id = ovs_id and -- fact
              coalesce(ovs_value, '') <> :curr_value
        into opr_id, new_value;
    
        if (opr_id is not null) then
        begin
          if (need_suspend = 1) then
            suspend;
  
          execute statement ('update continfo set ' || f_name || ' = :new_value where cci_object_kind = 0 and cci_object_id = :c_id;')
                            (c_id := c_id, new_value := new_value);
        end
      end

      if (rd_code in ('contactinfo.cell_main', 'contactinfo.cell_add')) then
      begin
        last_cell = '';
        ok_cell = '';
        if (rd_code = 'contactinfo.cell_main') then
        begin
          select last_cell_main, ok_cell_main
          from continfo
          where cci_object_kind = 0 and
                cci_object_id = :c_id and
                last_cell_main <> ok_cell_main
          into last_cell, ok_cell;
        end else
        begin
          select last_cell_add, ok_cell_add
          from continfo
          where cci_object_kind = 0 and
                cci_object_id = :c_id and
                last_cell_add <> ok_cell_add
          into last_cell, ok_cell;
        end
        if (last_cell is null) then last_cell = '';
        if (ok_cell is null) then ok_cell = '';

        if (ok_cell <> last_cell) then
        begin
          cv_verify_datetime = null;
          select cv_verify_datetime
          from contverify
          where cv_object_kind = 0 and
                cv_object_id = :c_id and
                cv_from_rd_id = :rd_id and
                cv_contact_value = :ok_cell and
                cv_status > 0 and
                cv_verify_datetime is not null
          order by cv_verify_datetime desc
          rows 1
          into cv_verify_datetime;
  
          if (cv_verify_datetime is not null) then
          begin
            ovs_datetime = null;
            select ovs_datetime
            from objectprops, objectvalues
            where opr_id = :orig_opr_id and
                  ovs_from_opr_id = opr_id and
                  coalesce(ovs_value, '') = :last_cell
            order by ovs_datetime desc
            rows 1
            into ovs_datetime;

            if (ovs_datetime < cv_verify_datetime) then
            begin
              if (rd_code = 'contactinfo.cell_main') then
              begin
                update continfo
                set last_cell_main = ok_cell_main
                where cci_object_kind = 0 and
                      cci_object_id = :c_id;
              end else
              begin
                update continfo
                set last_cell_add = ok_cell_add
                where cci_object_kind = 0 and
                      cci_object_id = :c_id;
              end

              if (need_suspend = 1) then
                suspend;
            end
          end
        end
      end
    end
  end
end
