CREATE PROCEDURE HP_OBJECTPROPS_MEMO_GET(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID VARCHAR(255),
    SUBJECT_KIND VARCHAR(100),
    SUBJECT_ID VARCHAR(255),
    RESULT_ROW_COUNT INTEGER)
RETURNS (
    OBM_FROM_OPR_ID DECIMAL(18,0),
    OBM_ID INTEGER,
    OBM_MEMO BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    OBM_DATETIME TIMESTAMP)
AS
begin
-- процедура получает значение для пары object/subject
  select ovs_from_opr_id
  from hp_objectprops_get_opr_id(:object_kind, :object_id, :subject_kind, :subject_id)
  into :obm_from_opr_id;

  if (obm_from_opr_id is null) then
    exit;

  if (result_row_count > 0) then
  begin
    for select obm_id, obm_memo, obm_datetime
        from objectmemo
        where obm_from_opr_id = :obm_from_opr_id
        order by obm_from_opr_id desc, obm_id desc
        rows :result_row_count
        into :obm_id, :obm_memo, :obm_datetime
    do
    begin
      suspend;
    end
  end
  else
  begin
    for select obm_id, obm_memo, obm_datetime
        from objectmemo
        where obm_from_opr_id = :obm_from_opr_id
        order by obm_from_opr_id desc, obm_id desc
        into :obm_id, :obm_memo, :obm_datetime
    do
    begin
      suspend;
    end
  end

end
