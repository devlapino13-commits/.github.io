CREATE PROCEDURE PRINT_NEW_SCHEDULE_KZ(
    K_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    SH_DATE D_DATE,
    SH_OS CURR_TYPE,
    SH_PERC CURR_TYPE,
    SH_TOTAL_OS_PERC CURR_TYPE,
    SH_REMAIN_OS CURR_TYPE)
AS
declare variable counter d_integer;
declare variable bls_id d_id;
declare variable json_text d_blob_text;
declare variable json_schedule d_blob_text;
declare variable schedule_counter d_integer;
declare variable sh_balance_before curr_type;
begin
  for select eil.eil_from_bls_id
      from extinfolist eil
      inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id and
            eic.eic_code = 'restruct_schedule'
      where eil.eil_status = 2 and
            eil.eil_object_kind = f_objkind('kontract', 'id') and
            eil.eil_object_id = :k_id and
            eil.eil_is_json = 1 and
            eil.eil_from_bls_id > 0
      order by eil.eil_id desc rows 1
      into bls_id
  do
  begin

    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into json_text;

    sh_balance_before = f_json_str_get_value(:json_text, 'sh_balance_before', '');
    json_schedule = cast(f_json_blob_get_value(:json_text, 'schedule', '') as d_blob_text);
    schedule_counter = f_json_str_get_count(:json_text, 'schedule');
    counter = 0;
    sh_date = null;
    sh_os = 0;
    sh_perc = 0;
    sh_total_os_perc = 0;
    sh_remain_os = sh_balance_before;
    while (json_schedule <> '' and schedule_counter > counter) do
    begin
      sh_date = f_json_str_get_value(:json_text, 'schedule[' || cast(counter as varchar(25)) || '].sh_date', '');
      sh_os = f_json_str_get_value(:json_text, 'schedule[' || cast(counter as varchar(25)) || '].zad_os', '');
      sh_perc = f_json_str_get_value(:json_text, 'schedule[' || cast(counter as varchar(25)) || '].zad_perc', '');
      sh_total_os_perc = sh_os + sh_perc;
      sh_remain_os = sh_remain_os - sh_os;
      counter = counter + 1;
      suspend;
    end
  end
end
