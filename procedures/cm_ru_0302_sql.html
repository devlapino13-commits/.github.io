CREATE PROCEDURE CM_RU_0302_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable is_refund d_integer;
declare variable amount curr_type;
declare variable disb_date d_date;
declare variable point d_integer;
declare variable cli_balance curr_type;
declare variable amount_max curr_type;
declare variable pr_id d_integer;
declare variable delinq_days d_integer;
declare variable ref_count d_integer;
declare variable ref_balance curr_type;
declare variable office_id d_integer;
declare variable disb_date_last d_date;
declare variable ref_disb_date d_date;
begin
  res_error_is_critical = 2;

  for select mac_from_c_id, ma_refund, mac_amount, ma_date_oper, pr_id, ma_office_id
      from miniapp
      inner join miniappclient on mac_from_ma_id = ma_id
      inner join product on pr_id = ma_from_pr_id
      where ma_id = :ma_id and
            ma_is_active = 1 and
            ma_date_end = '31.12.9999' and
            pr_code containing '.refinance_plus.'
      into c_id, is_refund, amount, disb_date, pr_id, office_id
  do
  begin
    point = f_point_get(11, ma_id, c_id, disb_date);

    if (office_id in (1012800, 1490600, 1050600)) then
    begin
      res_error_text = 'В текущем офисе выдача запрещена';
      suspend;
    end

    if (point < 11) then
    begin
      res_error_text = 'У клиента менее 11 баллов, выдача запрещена';
      suspend;
    end

    if (is_refund <> 1) then
    begin
      res_error_text = 'Продукт предназначен для рефинансирования';
      suspend;
    end

    cli_balance = 0;
    delinq_days = 0;
    select sum(oks_balance), sum(iif((:disb_date - oks_delinq_date) < 0, 0, :disb_date - oks_delinq_date))
    from knclient, oksummary
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = oks_from_k_id and
          oks_balance > 0 and
          oks_close_state = 0 and
          :disb_date between oks_date_begin and oks_date_end
    into cli_balance, delinq_days;
    cli_balance = coalesce(cli_balance, 0);
    delinq_days = coalesce(delinq_days, 0);

    if (cli_balance + amount > 500000) then
    begin
      res_error_text = 'Общий баланс всех займов превысит 500 000 р.';
      suspend;
    end

    if (delinq_days > 0) then
    begin
      res_error_text = 'У клиента имеется активная просрочка';
      suspend;
    end

    select amount_max
    from perc_amount_interval_by_point(:pr_id, :point)
    into amount_max;

    if (amount > amount_max) then
    begin
      res_error_text = f_format('Бал клиента: %1, Разрешенная максимальная сумма для выдачи: %2', point, round(amount_max));
      suspend;
    end

    if (amount not in (100000, 200000)) then
    begin
      res_error_text = 'Выбрана непрвильная сумма для выдачи';
      suspend;
    end

    ref_count = 0;
    ref_balance = 0;
    select count(kr_from_k_id), sum(kr_balance), max(k_disb_date)
    from kontrefinance
    inner join kontract on k_id = kr_from_k_id
    where kr_from_ma_id = :ma_id and
          kr_from_k_id is not null
    into ref_count, ref_balance, ref_disb_date;
    ref_count = coalesce(ref_count, 0);
    ref_balance = coalesce(ref_balance, 0);

    if (ref_disb_date >= '13.04.2026') then
    begin
      res_error_text = 'Заем не подходит под условия Акции';
      suspend;
    end 

    if (ref_count = 0) then
    begin
      res_error_text = 'Не выбраны займы для рефинансирования';
      suspend;
    end

    if (amount < ref_balance) then
    begin
      res_error_text = 'Суммы нового займа недостаточно для рефинансирования всех выбранных займов, измените выбор';
      suspend;
    end

    if (amount = 200000 and
        ref_count < 2) then
    begin
      res_error_text = 'Сумма займа 200 000 руб.: к рефинансированию должно быть выбрано не менее 2 займов';
      suspend;
    end

    select max(k_disb_date)
    from knclient, kontstate, kontract, product
    where kc_from_c_id = :c_id and
          kc_amount_approved > 0 and
          kc_from_k_id = ks_from_k_id and
          kc_from_k_id = k_id and
          k_from_pr_id = pr_id and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          :disb_date between ks_date_begin and ks_date_end and
          pr_code containing '.refinance_plus.'
    into disb_date_last;

    if (disb_date_last + 3 > disb_date) then
    begin
      res_error_text = 'Новый заем Рефинанс Плюс может быть выдан не ранее ' || f_datetostr((disb_date_last + 3));
      suspend;
    end

  end
end
