Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE HP_GET_UI_UXFRITEMS(
    UA_CODE D_STR_100,
    UN_ID D_STR_CODE)
RETURNS (
    UFI_CODE D_STR_100,
    UFI_PARENT_CODE D_STR_100,
    UFI_KIND D_INTEGER,
    UFI_CONTROL D_STR_CODE,
    UFI_POSITION D_INTEGER,
    UFI_CAPTION D_STR_CODE,
    UFI_CAPTION_ALIGN D_INTEGER,
    UFI_HINT D_STR_LONG,
    UFI_HIDDEN D_BOOL,
    UFP_REQUIRED D_BOOL,
    UFP_ACCESS_KIND D_SMALLINT,
    UFI_READONLY D_BOOL,
    UFI_GROUP_DIRECTION D_INTEGER,
    UFI_H_ALIGN D_INTEGER,
    UFI_V_ALIGN D_INTEGER,
    UFI_WIDTH D_INTEGER,
    UFI_HEIGHT D_INTEGER,
    UFI_DATASET_FIELD D_STR_100,
    UFI_LOOKUP_SQL D_STR_LONG,
    UFI_UFR_CODE D_STR_CODE,
    UFI_IS_PARAM D_SMALLINT,
    UFI_UCM_CODE D_STR_CODE,
    UFI_MAIN_FIELD D_STR_100,
    FR_CODE D_STR_CODE,
    UFI_UA_CODE D_STR_100,
    USR_ID_FIELD D_STR_CODE,
    USR_TEXT_FIELD D_STR_CODE,
    UA_USR_ID_FIELD D_STR_CODE,
    UA_USR_TEXT_FIELD D_STR_CODE,
    USR_EDITOR_PARAM_FIELD D_STR_CODE,
    USR_EDITOR_READONLY D_BOOL,
    TREE_PARENT D_STR_CODE,
    UFP_EDIT_MASK D_STR_LONG,
    UFP_MASK_MESSAGE D_STR_255)
AS
declare variable is_btn_edit d_bool;
declare variable fr_parent d_str_100;
declare variable ux_code d_str_code;
declare variable parent_is_readonly d_bool;
declare variable ua_uxa_code d_str_code;
declare variable ufp_code d_str_100;
declare variable is_image d_bool;
declare variable blob_edit d_bool;
begin
  if (exists(select * from uxcode
             where ux_code = :ua_code)) then
    ux_code = ua_code;
  else
  begin
    -- Значит передается Действие
    select ux_code
    from hp_get_uxalias(:ua_code)
    into :ux_code;
  end

  select ufi_readonly
  from uxfritem
  where ufi_ux_code = :ux_code and
        ufi_parent_code is null
  into parent_is_readonly;

  if (parent_is_readonly is null) then
    parent_is_readonly = 0;

  ufi_position = -1;
  for select ufi_code, ufi_parent_code, ufi_kind, ufi_control, ufi_ufr_code,
             ufi_caption, ufi_caption_align, ufi_hint, ufi_hidden,
             iif(:parent_is_readonly = 1, 1, ufi_readonly),
             ufi_group_direction, ufi_h_align, ufi_v_align, ufi_width, ufi_height,
             ufi_ufp_code, ufi_origin_name, ufi_is_param, ufp_required, ufp_access_kind
  from hp_uxfritem_list(:ux_code, 1)
  into :ufi_code, :ufi_parent_code, :ufi_kind, :ufi_control, ufi_ufr_code,
       :ufi_caption, :ufi_caption_align, :ufi_hint, :ufi_hidden,
       :ufi_readonly,
       :ufi_group_direction, :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height,
       :ufp_code, :ufi_dataset_field, :ufi_is_param, ufp_required, ufp_access_kind
  do
  begin
    ufi_lookup_sql = '';
    ufi_ua_code = '';
    usr_id_field = '';
    usr_text_field = '';
    ua_usr_id_field = '';
    ua_usr_text_field = '';
    usr_editor_readonly = 1;
    usr_editor_param_field = '';
    ufp_edit_mask = '';
    ufp_mask_message = '';

    is_btn_edit = 0;
    ufi_ucm_code = '';
    tree_parent = '';
    ufi_lookup_sql = '';

    if (ufi_parent_code is null and ufi_control = 'form') then
    begin
      -- Если это форма, растягиваем на всю свободную область
      ufi_h_align = 3; -- ahClient
      ufi_v_align = 3; -- avClient
      ufi_caption_align = -1;
      ufi_caption = 'Форма';
    end

    fr_code = '';
    if (ufi_control = 'frame') then
    begin
      fr_code = ufi_ufr_code;
      fr_parent = ufi_parent_code;

      if (ufi_readonly = 1) then
        parent_is_readonly = 1;

      for select ufi_code, ufi_position, ufi_caption, ufi_caption_align,
             iif(ufi_parent_code is null, :fr_parent, ufi_parent_code),
             iif(ufi_parent_code is null, :ufi_kind, ufi_kind),
             iif(ufi_parent_code is null, :ufi_control, ufi_control),
             iif(ufi_parent_code is null, :ufi_h_align, ufi_h_align),
             iif(ufi_parent_code is null, :ufi_v_align, ufi_v_align),
             iif(ufi_parent_code is null, :ufi_width, ufi_width),
             iif(ufi_parent_code is null, :ufi_height, ufi_height),
             iif(:parent_is_readonly = 1, 1, ufi_readonly), -- Если родитель только для чтения, то все дети тоже
             ufi_hint, ufi_hidden, ufi_group_direction,
             ufi_dataset_field, ufi_lookup_sql, ufi_ufr_code, ufi_is_param,
             ufi_ucm_code, ufi_main_field, ufi_ua_code, ufp_required,
             usr_id_field, usr_text_field, ua_usr_id_field, ua_usr_text_field,
             usr_editor_readonly, usr_editor_param_field, ufp_access_kind,
             ufp_edit_mask, ufp_mask_message
      from hp_get_ui_uxfritems(:fr_code, :un_id)
      into :ufi_code, :ufi_position, :ufi_caption, :ufi_caption_align,
           :ufi_parent_code, :ufi_kind, :ufi_control,
           :ufi_h_align, :ufi_v_align, :ufi_width, :ufi_height, :ufi_readonly, 
           :ufi_hint, :ufi_hidden, :ufi_group_direction,
           :ufi_dataset_field, :ufi_lookup_sql, :ufi_ufr_code, :ufi_is_param,
           :ufi_ucm_code, :ufi_main_field, ufi_ua_code, ufp_required,
           :usr_id_field, :usr_text_field, :ua_usr_id_field, :ua_usr_text_field,
           :usr_editor_readonly, :usr_editor_param_field, ufp_access_kind,
           :ufp_edit_mask, :ufp_mask_message
      do
      begin
        select iif(:ufi_width > 0, :ufi_width, default_width), iif(:ufi_height > 0, :ufi_height, default_height)
        from hp_uxfritem_control_kind
        where editor_id = :ufi_control
        into ufi_width, ufi_height;

        suspend;
      end

      Continue;
    end

    select iif(:ufi_width > 0, :ufi_width, default_width), iif(:ufi_height > 0, :ufi_height, default_height)
    from hp_uxfritem_control_kind
    where editor_id = :ufi_control
    into ufi_width, ufi_height;

    if (ufi_dataset_field is null) then
      ufi_dataset_field = '';

    ufi_lookup_sql = '';
    ufi_ua_code = '';
    usr_id_field = '';
    usr_text_field = '';
    ua_usr_id_field = '';
    ua_usr_text_field = '';
    is_btn_edit = 0;
    ufi_ucm_code = '';
    tree_parent = '';
    usr_editor_readonly = 1;
    usr_editor_param_field = '';

    if (ufi_control = 'combobox') then
    begin
      select ufc_ucm_code
      from uxfrcombobox
      where ufc_ufp_code = :ufp_code
      into ufi_ucm_code;
    end

    if (ufi_control = 'tree') then
    begin
      select ufr_tree_parent_field
      from uxframe
      where ufr_code = :fr_code
      into tree_parent;
    end

    if (ufi_control = 'lookup') then
    begin
      select sql_text
      from hp_uxframe_sql_lookup(:ufp_code)
      into :ufi_lookup_sql;
    end

    if (ufi_control = 'mask') then
    begin
      select uex_expression, uex_message
      from uxexpression
      where uex_ufp_code = :ufp_code
      into ufp_edit_mask, ufp_mask_message;
    end

    if (ufi_control = 'cell') then
    begin
      ufp_edit_mask = '(79|992|996|998|994)[0-9]{9}';
      ufp_mask_message = 'Неверный формат номера телефона';
      ufi_control = 'mask';
    end

    if (ufi_control = 'email') then
    begin
      ufp_edit_mask = '[\w\-.]+@[\w\-]+(.[\w\-]+)+';
      ufp_mask_message = 'Неверный формат электронный почты';
      ufi_control = 'mask';
    end

    if (ufi_control = 'mask' and ufp_edit_mask = '') then
    begin
      ufi_control = 'text';
      ufp_mask_message = '';
    end

    -- DialogEdit - это тоже самое ButtonEdit,
    -- отличие в том что процедура HP_GET_UI_UXACTION будет обрабатывать его по особому
    -- Поисковик привязанный как DialogEdit будет открываться до появления формы для выбора значения

    if (ufi_control = 'dialogedit') then
      ufi_control = 'buttonedit';

    if (ufi_control = 'buttonedit') then
    begin
      ufi_control = 'text';

      if (ufi_hidden = 0) then
      begin
        ufi_hidden = 1;
        is_btn_edit = 1;
      end
    end

    is_image = 0;
    blob_edit = 0;
    if (ufi_control in ('blob_json', 'blob_bls', 'blob')) then
    begin
      if (ufi_hidden = 0) then
      begin
        ufi_hidden = 1;
        is_image = 1;
        if (ufi_control = 'blob') then
          blob_edit = 1;
      end
      ufi_control = 'text';
    end

    ufi_position = ufi_position + 1;
    suspend;

    if (is_btn_edit = 1) then
    begin
      select ua_code, ua_uxa_code, ua_usr_id_field, ua_usr_text_field, usr_id_field, usr_text_field,
             usr_editor_readonly, usr_editor_param_field
      from uxfrparam, uxaction, uxsearch
      where ufp_code = :ufp_code and
            ufp_ufr_code = ua_ux_code and
            ufp_origin_name = ua_usr_id_field and
            ua_uxa_code = usr_code
      into ufi_ua_code, ua_uxa_code, ua_usr_id_field, ua_usr_text_field, usr_id_field, usr_text_field,
           usr_editor_readonly, usr_editor_param_field;

      if (coalesce(ufi_ua_code, '') = '') then
        exception e_default f_format('ButtonEdit action not found, ufi_code: %1', ufi_code);

      if (ua_usr_text_field = '') then
      begin
        -- если поле нужно формировать динамически, получаем текст SQL
        select ukn_sql
        from uxalias, uxcode, uxkeyname
        where uxa_code = :ua_uxa_code and
              uxa_ux_code = ux_code and
              ux_table_name = ukn_table_name
        into ufi_lookup_sql;

        if (ufi_lookup_sql is null) then
          ufi_lookup_sql = '';

        if (ufi_lookup_sql = '') then
          exception e_default f_format('Keyname lookup sql not found, ufi_code: %1', ufi_code);
      end

      ua_usr_text_field = ua_usr_id_field || '$text';
      ufi_code = ufi_code || '$text';
      ufi_position = ufi_position + 1;
      ufi_dataset_field = ua_usr_text_field;
      ufi_hidden = 0;
      ufi_control = 'buttonedit';

      suspend;

      ufi_ua_code = '';
    end

    if (is_image = 1) then
    begin
      ufi_dataset_field = ufi_dataset_field || '$text';
      ufi_code = ufi_code || '$text';
      ufi_hidden = 0;
      ufi_position = ufi_position + 1;

      if (blob_edit = 0) then
        ufi_control = 'blob_json';
      else
        ufi_control = 'blob';

      suspend;
    end
  end
end
