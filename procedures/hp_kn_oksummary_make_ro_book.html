CREATE PROCEDURE HP_KN_OKSUMMARY_MAKE_RO_BOOK(
    REP_DATE_END D_DATE)
RETURNS (
    REC_ID D_INTEGER,
    REG_ID D_INTEGER,
    REG_NAME D_STR_100,
    PAT_ID D_INTEGER,
    PAT_NAME D_STR_100,
    OKS_BALANCE D_CURRENCY,
    OKS_PERCENT D_CURRENCY,
    OKS_BAL_ESP_END D_CURRENCY,
    OKS_DIFF_ESP_BAL D_CURRENCY,
    OKS_ECL_VBS D_CURRENCY,
    OKS_DELINQ_OS D_CURRENCY,
    OKS_DELINQ_PERC D_CURRENCY,
    BAL_ECL_VBS D_CURRENCY,
    DT_ECL_VBS D_STR_20,
    KT_ECL_VBS D_STR_20,
    SUM_ECL_VBS D_CURRENCY,
    BAL_DELINQ_OS D_CURRENCY,
    DT_DELINQ_OS D_STR_20,
    KT_DELINQ_OS D_STR_20,
    SUM_DELINQ_OS D_CURRENCY,
    BAL_DELINQ_PERC D_CURRENCY,
    DT_DELINQ_PERC D_STR_20,
    KT_DELINQ_PERC D_STR_20,
    SUM_DELINQ_PERC D_CURRENCY,
    DT_CALC_PERC D_STR_20,
    KT_CALC_PERC D_STR_20,
    SUM_CALC_PERC D_CURRENCY)
AS
declare variable rep_date_begin d_date;
declare variable k_id d_id;
declare variable tmp_balance d_currency;
declare variable tmp_percent d_currency;
declare variable tmp_bal_esp_end d_currency;
declare variable tmp_diff_esp_bal d_currency;
declare variable tmp_ecl_vbs d_currency;
declare variable tmp_delinq_os d_currency;
declare variable tmp_delinq_perc d_currency;
declare variable pr_id d_integer;
declare variable acc_ro_norm d_str_20;
declare variable acc_ro_expense d_str_20;
declare variable acc_ro_income d_str_20;
declare variable acc_os_norm d_str_20;
declare variable acc_os_delinq d_str_20;
declare variable acc_perc_norm d_str_20;
declare variable acc_perc_delinq d_str_20;
declare variable acc_perc_income d_str_20;
declare variable bal_perc_saldo d_currency;
declare variable diff_perc_saldo d_currency;
begin
  rec_id = 0;

  rep_date_begin = rep_date_end - extract(day from rep_date_end) + 1;

  for select reg_unicode, reg_code || ' ' || reg_name
  from region
  where reg_status = 3 and
        coalesce(reg_date_close, :rep_date_end) >= :rep_date_begin
  into reg_id, reg_name
  do
  begin
    for select pat_id, pat_name
    from prodacctype, prodacclist
    where pat_id = pal_from_pat_id and
          pal_from_at_id = 2 and
          coalesce(pal_main_acc, '') <> ''
    into :pat_id, :pat_name
    do
    begin
      select pr_id
      from product
      where pr_from_pat_id = :pat_id
      rows 1
      into pr_id;

      oks_balance = 0;
      oks_percent = 0;
      oks_bal_esp_end = 0;
      oks_diff_esp_bal = 0;
      oks_ecl_vbs = 0;
      oks_delinq_os = 0;
      oks_delinq_perc = 0;

      for select oks_from_k_id
      from oksummary, kontract, product
      where oks_date_begin <= :rep_date_end and
            oks_date_end >= :rep_date_end and
            oks_region_id = :reg_id and
            oks_balance > 0 and
            oks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_from_pat_id = :pat_id
      order by k_from_pr_id, k_id
      into k_id
      do
      begin
        -- Обороты:
        -- oks_diff_esp_perc
        -- oks_diff_esp_norm

        tmp_balance = 0;
        tmp_percent = 0;
        tmp_bal_esp_end = 0;
        tmp_diff_esp_bal = 0;
        tmp_ecl_vbs = 0;
        tmp_delinq_os = 0;
        tmp_delinq_perc = 0;

        select sum(oks_balance), sum(oks_bal_perc),
               sum(oks_bal_esp_end), sum(oks_diff_esp_bal), sum(oks_ecl_vbs),
               sum(iif(oks_delinq_date < :rep_date_end, oks_bal_zad_os, 0)),
               sum(iif(oks_delinq_date < :rep_date_end, oks_bal_zad_perc, 0))
        from kn_oksummary
        where oks_from_k_id = :k_id and
              oks_date_begin = :rep_date_end
        into tmp_balance, tmp_percent,
             tmp_bal_esp_end, tmp_diff_esp_bal, tmp_ecl_vbs,
             tmp_delinq_os, tmp_delinq_perc;

        if (tmp_balance is null) then
          tmp_balance = 0;

        if (tmp_percent is null) then
          tmp_percent = 0;

        if (tmp_bal_esp_end is null) then
          tmp_bal_esp_end = 0;

        if (tmp_diff_esp_bal is null) then
          tmp_diff_esp_bal = 0;

        if (tmp_ecl_vbs is null) then
          tmp_ecl_vbs = 0;

        if (tmp_delinq_os is null) then
          tmp_delinq_os = 0;

        if (tmp_delinq_perc is null) then
          tmp_delinq_perc = 0;

        oks_balance = oks_balance + tmp_balance;
        oks_percent = oks_percent + tmp_percent;

        oks_bal_esp_end = oks_bal_esp_end + tmp_bal_esp_end;
        oks_diff_esp_bal = oks_diff_esp_bal + tmp_diff_esp_bal;
        oks_ecl_vbs = oks_ecl_vbs + tmp_ecl_vbs;

        oks_delinq_os = oks_delinq_os + tmp_delinq_os;
        oks_delinq_perc = oks_delinq_perc + tmp_delinq_perc;
      end

      -- Счет РППУ
      select res_account
      from hp_make_kredit_acclist(:pr_id, :reg_id, 9)
      into :acc_ro_norm;

      -- Расход по резервам
      select res_account
      from hp_make_kredit_acclist(:pr_id, :reg_id, 49)
      into :acc_ro_expense;

      -- Доход (восстановление) по резервам
      select res_account
      from hp_make_kredit_acclist(:pr_id, :reg_id, 51)
      into :acc_ro_income;

      -- Счет Портфеля
      select res_account, res_acc_delinq
      from hp_make_kredit_acclist(:pr_id, :reg_id, 1)
      into :acc_os_norm, :acc_os_delinq;

      -- Счет Начисл.%%
      select res_account, res_acc_delinq
      from hp_make_kredit_acclist(:pr_id, :reg_id, 2)
      into :acc_perc_norm, :acc_perc_delinq;

      -- Счет %% дохода
      select res_account
      from hp_make_kredit_acclist(:pr_id, :reg_id, 36)
      into :acc_perc_income;

      bal_ecl_vbs = -f_al_bal(acc_ro_norm, :rep_date_end);

      sum_ecl_vbs = 0;
      dt_ecl_vbs = '';
      kt_ecl_vbs = '';

      if (oks_ecl_vbs > bal_ecl_vbs) then
      begin
        sum_ecl_vbs = oks_ecl_vbs - bal_ecl_vbs;
        dt_ecl_vbs = acc_ro_expense;
        kt_ecl_vbs = acc_ro_norm;
      end else
      if (oks_ecl_vbs < bal_ecl_vbs) then
      begin
        sum_ecl_vbs = bal_ecl_vbs - oks_ecl_vbs;
        dt_ecl_vbs = acc_ro_norm;
        kt_ecl_vbs = acc_ro_income;
      end

      sum_calc_perc = 0;
      dt_calc_perc = '';
      kt_calc_perc = '';

      sum_delinq_os = 0;
      dt_delinq_os = '';
      kt_delinq_os = '';

      if (acc_os_delinq <> '') then
      begin
        bal_delinq_os = f_al_bal(acc_os_delinq, :rep_date_end);

        if (oks_delinq_os > bal_delinq_os) then
        begin
          sum_delinq_os = oks_delinq_os - bal_delinq_os;
          dt_delinq_os = acc_os_delinq;
          kt_delinq_os = acc_os_norm;
        end else
        if (oks_delinq_os < bal_delinq_os) then
        begin
          sum_delinq_os = bal_delinq_os - oks_delinq_os;
          dt_delinq_os = acc_os_norm;
          kt_delinq_os = acc_os_delinq;
        end
      end

      sum_delinq_perc = 0;
      dt_delinq_perc = '';
      kt_delinq_perc = '';

      bal_perc_saldo = f_al_bal(acc_perc_norm, rep_date_end) + f_al_bal(acc_perc_delinq, rep_date_end);
      diff_perc_saldo = oks_percent - bal_perc_saldo;

      if (acc_perc_norm <> '' and acc_perc_income <> '' and diff_perc_saldo <> 0) then
      begin
        if (diff_perc_saldo > 0) then
        begin
          sum_calc_perc = diff_perc_saldo;
          dt_calc_perc = acc_perc_norm;
          kt_calc_perc = acc_perc_income;
        end else
        begin
          sum_calc_perc = -diff_perc_saldo;
          dt_calc_perc = acc_perc_income;
          kt_calc_perc = acc_perc_norm;
        end
      end

      if (acc_perc_delinq <> '') then
      begin
        bal_delinq_perc = f_al_bal(acc_perc_delinq, :rep_date_end);

        if (oks_delinq_perc > bal_delinq_perc) then
        begin
          sum_delinq_perc = oks_delinq_perc - bal_delinq_perc;
          dt_delinq_perc = acc_perc_delinq;
          kt_delinq_perc = acc_perc_norm;
        end else
        if (oks_delinq_perc < bal_delinq_perc) then
        begin
          sum_delinq_perc = bal_delinq_perc - oks_delinq_perc;
          dt_delinq_perc = acc_perc_norm;
          kt_delinq_perc = acc_perc_delinq;
        end
      end

      rec_id = rec_id + 1;
      suspend;
    end
  end
end
