CREATE PROCEDURE GET_CFP_KPD(
    MAC_ID D_ID,
    ACL_ID D_ID = 0)
RETURNS (
    KPD D_CURRENCY,
    KPD_ERR D_STR_255,
    CA_AVR_INCOME DECIMAL(15,2),
    EXTRA_AMOUNT DECIMAL(15,2),
    COMMUNAL_EXPENSES DECIMAL(15,2),
    LIVING_EXPENSES DECIMAL(15,2),
    LEGAL_DOCS_EXPENSES DECIMAL(15,2),
    OTHER_BUSINESS_EXPENSES DECIMAL(15,2),
    REPAYMENT_OBLIGATIONS DECIMAL(15,2),
    NET_PROFIT DECIMAL(15,2),
    AMOUNT_MONTHLY_PAYMENTS DECIMAL(15,2),
    DATA_SOURCE D_STR_255)
AS
declare variable ma_id d_id;
declare variable c_id d_id;
declare variable mac_amount d_currency;
declare variable ca_id d_id;
declare variable c_name_fio d_str_255;
begin
  for select mac.mac_from_ma_id, mac.mac_from_c_id, mac.mac_amount
      from miniappclient mac
      inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id and
            ma.ma_is_active = 1 and
            coalesce(ma.ma_date_end, '31.12.9999') >= '31.12.2999'
      where mac.mac_id = :mac_id and
            mac.mac_amount > 0
      into ma_id, c_id, mac_amount
  do
  begin
    select ca.ca_id, ca.ca_avr_income
    from cfp_anketa ca
    where ca.ca_object_kind = f_objkind('miniappclient', 'id') and
          ca.ca_object_id = :mac_id and
          ca.ca_from_c_id = :c_id and
          (
            coalesce(:acl_id, 0) = 0                    -- RFCKG-5566
            or ca.ca_from_acl_id = :acl_id
          ) and
          (
            coalesce(:acl_id, 0) <> 0
            or ca.ca_from_acl_id is null
          )
    into ca_id, ca_avr_income;

    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into c_name_fio;

    extra_amount = (select sum(caa.caa_value)
                    from cfp_anketa_answers caa
                    inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                          cq.cq_code_name in ('cl_add_income_spouse_salary', 'cl_add_income_household', 'cl_add_income_service')
                    where caa.caa_from_ca_id = :ca_id);
    communal_expenses = (select caa.caa_value
                         from cfp_anketa_answers caa
                         inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                               cq.cq_code_name = 'monthly_housing_expenses'
                         where caa.caa_from_ca_id = :ca_id);
    living_expenses = (select caa.caa_value
                       from cfp_anketa_answers caa
                       inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                             cq.cq_code_name = 'monthly_family_expenses'
                       where caa.caa_from_ca_id = :ca_id);
    legal_docs_expenses = (select caa.caa_value
                           from cfp_anketa_answers caa
                           inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                                 cq.cq_code_name = 'monthly_tax_expenses'
                           where caa.caa_from_ca_id = :ca_id);
    other_business_expenses = (select caa.caa_value
                               from cfp_anketa_answers caa
                               inner join cfp_questions cq on cq.cq_id = caa.caa_from_cq_id and
                                     cq.cq_code_name = 'monthly_other_expenses'
                               where caa.caa_from_ca_id = :ca_id);
    repayment_obligations = 0;
    amount_monthly_payments = 0;
    net_profit = 0;
    select cak.cak_kpd, cak.cak_net_profit, cak.cak_other_dept, cak.cak_monthly_payment
    from cfp_anketa_kpd cak
    where cak.cak_from_ca_id = :ca_id
    into kpd, net_profit, repayment_obligations, amount_monthly_payments;

    select first 1 rd.rd_name
    from cfp_repayment_oblig_calc croc
    inner join refdata rd on rd.rd_id = croc.croc_source
    where croc.croc_from_ca_id = :ca_id and
          croc.croc_is_actual = 1
    order by croc.croc_create_datetime desc
    into data_source;

    if (:kpd < 150 and coalesce(:acl_id, 0) = 0) then     -- RFCKG-5146
    begin
      kpd_err = 'КПД должно быть > 150% ' || c_name_fio; -- Для всех продуктов
      suspend;
      exit;
    end

--    if ((mac_amount > 200000) and      -- RFCKG-5146
--        (:kpd < 140)) then
--    begin
--      kpd_err = 'КПД должно быть не менее 150% , т.к. сумма займа более 200 000' || c_name_fio; -- для всех продуктов
--      suspend;
--      exit;
--    end

    suspend;
  end
end
