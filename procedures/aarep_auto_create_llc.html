CREATE PROCEDURE AAREP_AUTO_CREATE_LLC(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_ID INTEGER,
    REG_CODE VARCHAR(20),
    REG_NAME VARCHAR(100),
    LM_ID DECIMAL(18,0),
    AA_K_ID DECIMAL(18,0),
    AA_AMOUNT_ACTIVE_END DECIMAL(18,4),
    AA_DELIQ_DAYS INTEGER,
    AA_PR_IS_INDIV INTEGER,
    SH_CALC_PERC CURR_TYPE,
    REP_PERIOD INTEGER)
AS
declare variable aa_llc_percent decimal(18,7);
declare variable aa_llc_id integer;
declare variable lm_no integer;
declare variable db_id integer;
declare variable k_index_curr_code varchar(3);
declare variable k_index_curr_rate curr_type;
declare variable cr_rate curr_type;
declare variable k_disb_date d_date;
declare variable k_percent d_percent;
declare variable lir_rate d_percent;
declare variable k_by_rating d_smallint;
declare variable not_secured boolean;
begin
  db_id = f_get_db_id();
  cr_rate = null;

  select cr_rate
  from opercurr
  where cr_from_cu_id = '840' and
        :rep_date_end between cr_date and cr_date_end
  into cr_rate;

  if (cr_rate is null) then
    cr_rate = 1;

  delete from llc_main
  where lm_date = :rep_date_end;

  for select reg_unicode, reg_code, reg_name
      from region
      where reg_status = 3
      order by reg_code
      into :reg_id, :reg_code, :reg_name
  do
  begin
    select result
    from get_kv_id(10)
    into :lm_id;

    lm_no = right(:lm_id, 4);
    insert into llc_main (lm_id, lm_date, lm_date_status, lm_from_reg_id, lm_no, lm_memo, lm_no_str, lm_is_signed)
    values (:lm_id, :rep_date_end, :rep_date_end, :reg_id, :lm_no, '', :lm_no, 1);

    if (db_id = 81001) then
    begin
      insert into llc_sign (ls_from_lm_id, ls_number, ls_sign_type, ls_fio)
      values (:lm_id, 1, '1', 'Белоусова Лариса');
    end
    else
    begin
      insert into llc_sign (ls_from_lm_id, ls_number, ls_sign_type, ls_fio)
      values (:lm_id, 1, '1', '');
    end

--  Переделанный переделанный вариант
    for select ks_from_k_id as aa_k_id, 0 as aa_amount_active_end, 0 as aa_deliq_days
        from kontstate
        where ks_ol_is_delete = 0 and
              ks_state = 2 and
              ks_region_id = :reg_id and
              cast(ks_date_begin as date) between :rep_date_begin and :rep_date_end
        union
        select oks_from_k_id, oks_balance, cast(:rep_date_end as date) - oks_delinq_date
        from oksummary
        where oks_region_id = :reg_id and
              :rep_date_end between oks_date_begin and oks_date_end and
              oks_close_state = 0
        into :aa_k_id, :aa_amount_active_end, :aa_deliq_days
    do
    begin
      select
        pr_loan_type,
        k_index_curr_code,
        k_index_curr_rate,
        k_disb_date,
        k_percent,
        coalesce(k_by_rating, 0),
        (:db_id = 94401) and (pr_name like '%t_minatsız%')
      from kontract, product
      where k_id = :aa_k_id
        and k_from_pr_id = pr_id
      into
        :aa_pr_is_indiv,
        :k_index_curr_code,
        :k_index_curr_rate,
        :k_disb_date,
        :k_percent,
        :k_by_rating,
        :not_secured;

      if (aa_deliq_days < 0) then
        aa_deliq_days = 0;
      if (aa_amount_active_end <> 0 and
          k_index_curr_code = '840') then
      begin
        if (k_index_curr_rate > cr_rate) then
          aa_amount_active_end = cast(aa_amount_active_end * k_index_curr_rate as decimal(18,2));
        else
          aa_amount_active_end = cast(aa_amount_active_end * cr_rate as decimal(18,2));
      end
--    классификация группового кредита в АЗ отличается от индив, по названиям классификации и по процентам
      if (db_id = 94401 and
          aa_pr_is_indiv = 0) then
      begin
        if (aa_deliq_days < 31) then
        begin
          aa_llc_percent = 0.0;
          aa_llc_id = 3;
        end
        else
        if (aa_deliq_days between 31 and 180) then
        begin
          aa_llc_percent = 50.0;
          aa_llc_id = 6;
        end
        else
        if (aa_deliq_days > 180) then
        begin
          aa_llc_percent = 100.0;
          aa_llc_id = 7;
        end
      end
      else
      begin
        select
          cs_id,
          iif(:aa_pr_is_indiv = 1,
            iif(:k_disb_date >= '12.01.2024' and ((:k_by_rating = 1) or
              (:k_percent > (select first 1 lir_rate from llc_intrate where lir_date <= :k_disb_date order by lir_date desc))),
                cs_is_for_group, cs_is_for_indiv),
            cs_is_for_group)
        from classific
        where cs_day_min <= :aa_deliq_days and
              cs_day_max >= :aa_deliq_days
        into :aa_llc_id, :aa_llc_percent;
        if (aa_llc_percent is null) then
        begin
          aa_llc_percent = 0;
          aa_llc_id = 1;
        end

        -- s.k.
        if (not_secured and aa_deliq_days >= 31) then
        begin
          aa_llc_percent = iif(aa_deliq_days between 31 and 180, 50, 100);
          aa_llc_id = iif(aa_deliq_days between 31 and 180, 6, 7);
        end
      end

      insert into llc_kont (lk_from_lm_id, lk_from_k_id, lk_deliq_days, lk_from_reg_id, lk_balance, lk_status,
                            lk_percent, lk_date, lk_date_end, lk_memo, lk_is_indiv, lk_is_signed, lk_is_changed_mannual,
                            lk_percent_orig)
      values (:lm_id, :aa_k_id, :aa_deliq_days, :reg_id, :aa_amount_active_end, :aa_llc_id, :aa_llc_percent,
              :rep_date_end, null, '', :aa_pr_is_indiv, 1, 0, :aa_llc_percent);
    end
    suspend;
  end

  if (db_id = 94401) then
  begin
    reg_id = null;
    reg_code = null;
    reg_name = null;
    lm_id = null;
    aa_k_id = null;
    aa_amount_active_end = null;
    aa_deliq_days = null;
    aa_pr_is_indiv = null;
    sh_calc_perc = null;
    rep_period = null;
    for select lk_from_lm_id, lk_from_k_id, lk_percent, sh_calc_perc, rep_period
        from hp_llc_update_calcperc_az(:rep_date_end)
        into :lm_id, :aa_k_id, :aa_llc_percent, :sh_calc_perc, :rep_period
    do
    begin
      suspend;
    end
  end
end
