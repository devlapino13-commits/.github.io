CREATE PROCEDURE HP_ANNUAL_INSURANCE_LOAN(
    K_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255)
AS
declare variable pr_code d_str_100;
declare variable insur_k_id d_id;
declare variable ol_id d_id;
declare variable kr_id d_integer;
declare variable krl_from_un_id d_id;
declare variable k_status d_integer;
begin
  -- RFCKG-5090
  res_text = '';

  if (f_get_db_id() <> 41701) then
  begin
    suspend;
    exit;
  end

  select pr.pr_code,
         k.k_status
  from kontract k
  inner join product pr on pr.pr_id = k.k_from_pr_id
  where k.k_id = :k_id
  into pr_code,
       k_status;

  if (:pr_code containing '.insur_base.') then  -- Если анулировали основной займ
  begin
    select lir.lir_kontract_insurance
    from loan_insurance_relation lir
    where lir.lir_kontract_base = :k_id
    into insur_k_id;

    if (coalesce(:insur_k_id, 0) <> 0) then
    begin

      select ok.ok_from_ol_id
      from operkontract ok
      inner join operlist ol on ol.ol_id = ok.ok_from_ol_id
      where ok.ok_from_k_id = :k_id and
            ok.ok_operation_type = f_get_rd_id('CashTransactionIncom.Insurance') and
            ol.is_delete = 0
      into ol_id;

      if (coalesce(:ol_id, 0) <> 0) then -- Если успели выдать анулируем
      begin
        update operlist ol         -- удаляем оплату за страхование
           set ol.is_delete = 1
        where ol.ol_id = :ol_id;

        ol_id = 0;

        select ks.ks_from_ol_id
        from kontstate ks
        where ks.ks_from_k_id = :insur_k_id and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0
        into ol_id;

        if (coalesce(:ol_id, 0) <> 0) then
        begin
          update operlist ol   -- удаляем выдачу
             set ol.is_delete = 1
          where ol.ol_id = :ol_id;
        end
      end

      update kontract
      set k_status = :k_status
      where k_id = :insur_k_id;

      select first 1 krl.krl_from_kr_id,
                     krl.krl_from_un_id
      from kontreslist krl
      where krl.krl_from_k_id = :k_id
      order by krl.krl_datetime desc
      into kr_id, krl_from_un_id;

      if (coalesce(kr_id, 0) <> 0) then
      begin
        -- Причина отказа из основного займа
        insert into kontreslist ( krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                                 krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
        values (:kr_id, :krl_from_un_id, :insur_k_id, 0, null, current_timestamp, current_date, current_date, 0);
      end
    end
  end
  suspend;
end
