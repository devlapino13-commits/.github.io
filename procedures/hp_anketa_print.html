CREATE PROCEDURE HP_ANKETA_PRINT(
    OBJECT_KIND INTEGER,
    OBJECT_ID DECIMAL(18,0),
    ANKETA_NAME VARCHAR(100),
    REP_DATE DATE)
RETURNS (
    QUES_ID INTEGER,
    QUES_CODE VARCHAR(255),
    QUES_TEXT VARCHAR(255),
    RES_TEXT VARCHAR(255),
    RES_DATE DATE,
    IS_HISTORY INTEGER)
AS
declare variable anketa_id integer;
begin
  select rd_id
  from refdata
  where rd_code = :anketa_name
  into :anketa_id;
  if (anketa_id is null) then
    exit;

  for select acg_from_rd_id, rd_code, rd_name
      from actiongroup, refdata
      where acg_group_rd_id = :anketa_id and
            acg_parent_id is null and
            acg_is_delete = 0 and
            acg_from_rd_id = rd_id
      order by acg_orderno
      into :ques_id, ques_code, ques_text
  do
  begin
    res_text = '';
    select ovs_value, ovs_date, is_history
    from hp_objectprops_get(:object_kind, :object_id, -1, :ques_id, :rep_date)
    into :res_text, :res_date, :is_history;
    if (coalesce(res_text, '') = '') then
      res_text = '-';
    suspend;
  end
end
