CREATE PROCEDURE CM_KG_0075_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable ma_id integer;
declare variable object_qty integer;
declare variable description varchar(2000);
declare variable finish_date date;
declare variable pr_code d_str_100;
begin
  for select kont.k_from_ma_app_id, pr_code
  from kontract kont, product pr
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr.pr_code containing '.ISTISNAA.'
  into :ma_id, :pr_code
  do
  begin
    select mif.mai_from_ma_id,
           mif.mai_object_qty,
           mif.mai_project_description,
           mif.mai_project_finish_date
    from miniapp_ipf mif
    where mif.mai_from_ma_id = :ma_id
    rows 1
    into :ma_id, :object_qty, :description, :finish_date;
    object_qty = coalesce(object_qty, 0);
    description = coalesce(description, '');
    if (ma_id is null) then
    begin
      res_error_text = 'Отсутсвует запись в таблице miniapp_ipf';
      res_error_is_critical = 2;
      suspend;
      exit;
    end
    if (object_qty = 0) then
    begin
      res_error_text = 'Число объектов не указано';
      res_error_is_critical = 2;
      suspend;
    end
    if (description = '') then
    begin
      res_error_text = 'Описание проекта не указано';
      res_error_is_critical = 2;
      suspend;
    end
    if (finish_date is null) then
    begin
      res_error_text = 'Дата сдачи объекта не указана';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
