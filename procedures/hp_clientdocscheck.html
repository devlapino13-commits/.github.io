CREATE PROCEDURE HP_CLIENTDOCSCHECK(
    C_ID ID_TYPE)
RETURNS (
    CDS_ID D_INTEGER,
    PK_NAME D_STR_100,
    PK_NAME_ISCORRECT D_INTEGER,
    CDS_DOC_NO VARCHAR(30),
    CDS_DOC_DATE_BEGIN DATE,
    CDS_DOC_DATE_END DATE,
    CDS_DOC_ISSUED_BY D_STR_255,
    CDS_CREATE_DATETIME D_DATETIME,
    CDS_DOC_BLOB_1 ID_TYPE,
    CDS_DOC_BLOB_2 ID_TYPE,
    CDS_ISCHANGED D_INTEGER,
    CDS_FROM_PK_ID TYPE OF COLUMN PASSKIND.PK_ID,
    CDS_PREVIOUSLY_EDITED D_BOOL)
AS
begin
  for
    select
      cds_id,
      pk_name,
      coalesce(dcl_pk_name_iscorrect,1),
      cds_doc_no,
      cds_doc_date_begin,
      cds_doc_date_end,
      cds_doc_issued_by,
      cds_create_datetime,
      cds_doc_blob_1,
      cds_doc_blob_2,
      iif(dcl_id is null, 0, 1),
      cds_from_pk_id,
      iif(dcl_id is null, 0, 1)
    from clientdocs
      inner join passkind
        on pk_id = cds_from_pk_id
      inner join passdoctype
        on pdt_id = pk_from_pdt_id
      left join clientdocscheck_log
        on dcl_cds_id = cds_id
    where cds_from_c_id = :c_id
      and (cds_doc_blob_1 > 0 or cds_doc_blob_2 > 0)
      -- исключаем фото клиента, и кадровые документы (Страховой полис,Уведомление о приеме ИГ,Уведомление об увольнении ИГ)
      and pdt_code not in ('Business_place_photo', 'Insurance_ID', 'notification', 'notification_2')
      and not (cds_from_drl_id > 0 and pdt_code = 'transport')
      and cds_delete_datetime is null
  into
      :cds_id,
      :pk_name,
      :pk_name_iscorrect,
      :cds_doc_no,
      :cds_doc_date_begin,
      :cds_doc_date_end,
      :cds_doc_issued_by,
      :cds_create_datetime,
      :cds_doc_blob_1,
      :cds_doc_blob_2,
      :cds_ischanged,
      :cds_from_pk_id,
      :cds_previously_edited
   do
   begin
     suspend;
   end
end
