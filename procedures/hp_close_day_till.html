CREATE PROCEDURE HP_CLOSE_DAY_TILL(
    REG_ID INTEGER,
    REG_DATE DATE,
    REG_USER_ID DECIMAL(18,0),
    REG_IS_CASH INTEGER,
    REG_MEMO VARCHAR(255))
AS
declare variable min_dd date;
declare variable last_is_closed integer;
declare variable last_dd date;
declare variable i integer;
begin
  /* Procedure Text */
  select reg_date_open
  from region
  where reg_unicode = :reg_id
  into :min_dd;
  if (min_dd is null) then
  begin
    select min(ol_date_oper_day)
    from operlist
    where ol_from_reg_id = :reg_id and
          operlist.is_delete = 0
    into :min_dd;
  end
  if (min_dd is null) then
    min_dd = current_date;

  last_dd = :reg_date;
  last_is_closed = 0;
  while ((last_is_closed <> 1) and (last_dd > min_dd)) do
  begin
    select day_closed
    from hp_is_day_closed(:reg_id, :last_dd, :reg_is_cash)
    into :last_is_closed;
    if (last_is_closed = 0) then
    begin
      if (reg_is_cash = 1) then
      begin
        insert into closeday (cd_from_reg_id, cd_date, cd_number, cd_is_closed, cd_open_memo, cd_datetime, cd_user_id,
                              cd_is_cash_closed)
        values (:reg_id, :last_dd, -1, -1, :reg_memo, null, :reg_user_id, 1);
      end
      else
      begin
        insert into closeday (cd_from_reg_id, cd_date, cd_number, cd_is_closed, cd_open_memo, cd_datetime, cd_user_id,
                              cd_is_cash_closed)
        values (:reg_id, :last_dd, -1, 1, :reg_memo, null, :reg_user_id, -1);
      end
    end
    last_dd = dateadd(day, -1, last_dd);
    reg_memo = '';
  end
end
