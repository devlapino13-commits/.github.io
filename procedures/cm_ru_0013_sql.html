CREATE PROCEDURE CM_RU_0013_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable reg_id integer;
declare variable DISB_LIMIT_AMOUNT curr_type;
begin
    select k_last_reg_id, k_amount_approved, k_disb_date from kontract
    where k_id=:k_id
    into :reg_id, :KC_AMOUNT, :KF_DISB_DATE;
    if (KC_AMOUNT is null) then KC_AMOUNT=0;
--  Отменено 31.07.2014 согласно приказа об отмене Приказа №103-05-14 от 26.05.2014 г.
    if (reg_id is not null) then
    begin
      if ((KF_DISB_DATE>=cast('01.06.2014' as date) and KF_DISB_DATE<= cast('31.07.2014' as date))
           or (KF_DISB_DATE>=cast('17.09.2014' as date))) then
      begin
        select END_DISB_SALDO
        from AAREP_CASH_DISB_CHECK(:reg_id, :KF_DISB_DATE)
        into :DISB_LIMIT_AMOUNT;
  
        if (DISB_LIMIT_AMOUNT is null) then DISB_LIMIT_AMOUNT=0;
        if (Disb_limit_amount<KC_AMOUNT) then
        begin
          res_error_is_critical = 0;
          res_error_text = 'Недостаточно средств для кредитования! Вы можете утвердить, но при выдаче будет заблокировано если непомолнить кассу';
          if (reg_id = 1010300) then res_error_is_critical = 0;  -- Ордынка
          if (reg_id = 1010100) then res_error_is_critical = 0;  -- Ордынка
          suspend;
        end
      end
    end
end
