CREATE PROCEDURE CM_KG_0002_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kc_amount curr_type;
declare variable kf_disb_date d_date;
declare variable kol_new_cli smallint;
declare variable kol_old_cli smallint;
declare variable kal_new_kf_officer id_type;
declare variable op_par8_180 decimal(5,5);
declare variable op_date_begin date;
declare variable rep_date_str varchar(20);
declare variable officer_name varchar(100);
declare variable pr_code d_str_100;
begin
  select pr.pr_code
  from kontract kt
  inner join product pr on pr.pr_id = kt.k_from_pr_id
  where kt.k_id = :k_id
  into pr_code;

  if (pr_code containing '.auto_kk.' and check_kind = '.send_kk.') then
   exit;

--    Проверка гурппа новая или повторная
      kol_new_cli = null;
      kol_old_cli = null;
      select sum(iif(kc_circle <= 1, 1, 0)),
             sum(iif(kc_circle > 1, 1, 0))
      from knclient
      where kc_from_k_id = :k_id
      and kc_amount_approved > 0
      into kol_new_cli, kol_old_cli;
      if (kol_new_cli is null) then kol_new_cli = 0;
      if (kol_old_cli is null) then kol_old_cli = 0;

      if (current_date > '09.11.2014') then -- Разработка по приказу от 10.11.2014г.
      begin
          if (kol_new_cli > kol_old_cli) then
          begin
            select kal_new_kf_officer, k_disb_date from knaktlist, kontract
            where knaktlist.is_delete = 0
            and k_disb_date between kal_date and kal_date_end
            and kal_from_kf_id = k_from_kf_id
            and k_id = :k_id
            into kal_new_kf_officer, KF_DISB_DATE;
            op_par8_180 = null;
            select op_par8_180, op_date_begin from officerpar7
            where op_kc_id = :kal_new_kf_officer
            and :KF_DISB_DATE between op_date_begin and op_date_end
            into op_par8_180, op_date_begin;
            if (op_par8_180 is null) then op_par8_180 = 0;
            if (op_par8_180 > 1.00000) then
            begin
--              op_date_begin = op_date_begin - 1;
              rep_date_str = right((extract(day from :op_date_begin) + 100),2);
              rep_date_str = rep_date_str||'.'||extract(month from :op_date_begin)||'.'||extract(year from :op_date_begin);
    
              officer_name = (select c_name_fio from clients where c_id = :kal_new_kf_officer);
              res_error_text = 'У КС - '||:officer_name||' на дату '||:rep_date_str||' PAR = '||op_par8_180||'; Запрет выдачи новым группам.';
              res_error_is_critical = 1;
              suspend;
            end
          end
      end 
end
