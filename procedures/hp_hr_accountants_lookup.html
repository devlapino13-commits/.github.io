CREATE PROCEDURE HP_HR_ACCOUNTANTS_LOOKUP(
    IN_OFFICE_ID D_INTEGER)
RETURNS (
    ID D_ID,
    TEXT D_STR_255)
AS
declare variable c_name_fio d_str_100;
declare variable jl_name d_str_100;
declare variable db_id d_integer;
begin
  -- список бухгалтеров-подписантов в кадровых приказах (RFCKG: 4340, 4382)

  db_id = f_get_db_id();

  if (db_id <> 41701) then
    exit;

  if (in_office_id = 1090100) then
  begin
    for select c_id,
               c_name_fio,
               jl_name
        from hr_pstatus
        inner join clients on c_id = ps_from_c_id
        inner join joblist on jl_id = ps_from_jl_id
        where ps_status = 1 and
              ps_is_working = 1 and
              ps_is_on_staff = 1 and
              current_date between ps_date_begin and ps_date_end and
              ps_from_c_id in(1090100000003, 1090100615665) -- бухгалтеры-подписанты по офису ГО (RFCKG-4382)
        order by c_name_fio
        into id,
             c_name_fio,
             jl_name
    do
    begin
      text = c_name_fio || ', ' || jl_name;
      suspend;
    end
    exit;
  end

  for select c_id,
             c_name_fio,
             jl_name
      from hr_pstatus
      inner join clients on c_id = ps_from_c_id
      inner join joblist on jl_id = ps_from_jl_id
      inner join committees on co_from_c_id = ps_from_c_id
      where ps_status = 1 and
            ps_is_working = 1 and
            ps_is_on_staff = 1 and
            current_date between ps_date_begin and ps_date_end and
            ps_from_reg_id = 1090100 and -- офис ГО
            ps_from_dep_id = 13 and -- бухгалтерия
            co_from_rd_id = f_get_rd_id('Committee.payroll_committee') and -- справочник Комитеты -> Бухгалтеры-подписанты
            co_is_delete = 0
      order by c_name_fio
      into id,
           c_name_fio,
           jl_name
  do
  begin
    text = c_name_fio || ', ' || jl_name;
    suspend;
  end
end
