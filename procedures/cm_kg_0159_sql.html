CREATE PROCEDURE CM_KG_0159_SQL(
    K_ID ID_TYPE NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable kf_id id_type;
declare variable k_kredline_end date;
declare variable rep_date date;
declare variable k_kontract_no str_100;
declare variable ok_from_k_id id_type;
declare variable is_found integer;
declare variable c_id_tmp id_type;
declare variable ma_id d_id;
declare variable k_disb_date date;
declare variable k_id_tmp d_id;
declare variable k_id_tmp1 d_id;
declare variable crel_other_c_id id_type;
declare variable c_name_fio varchar(255);
begin
  for select kont.k_id, kont.k_disb_date
      from kontract kont
      inner join product pr on pr.pr_id = kont.k_from_pr_id
      where kont.k_id = :k_id and
            pr.pr_code containing '.Actiya_Ishenim_Refin.'
      into :k_id, :k_disb_date
  do
  begin
    is_found = 0;

    for select kncl.kc_from_c_id
        from knclient kncl
        where kncl.kc_from_k_id = :k_id and
              kncl.kc_amount_approved > 0
        into :c_id_tmp
    do
    begin
      if (is_found = 0) then
      begin
        for select kont.k_id
            from kontract kont
            inner join product pr on pr.pr_id = kont.k_from_pr_id
            inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
            inner join kontstate ks on ks.ks_from_k_id = kont.k_id
            where kncl.kc_from_c_id = :c_id_tmp and
                  kncl.kc_amount_approved > 0 and
                  ks.ks_state = 1 and
                  ks.ks_ol_is_delete = 0 and
                  pr.pr_code not containing '.Actiya_Ishenim_Refin.' and
                  kont.k_disb_date < cast('01.04.2021' as date) and
                  kont.k_kredline_end < cast('01.10.2021' as date) and
                  :k_disb_date between ks.ks_date_begin and ks.ks_date_end
            into :k_id_tmp
        do
        begin
          if (exists(select *
                     from operkred ok
                     where ok.ok_from_k_id = :k_id_tmp and
                           ok.ok_type = 4 and
                           ok.ok_ol_is_active = 1 and
                           ok.ok_ol_date between cast('01.05.2021' as date) and cast('15.06.2021' as date))) then
          begin
            is_found = 1;
            break;
          end
        end
      end
    end

    if (is_found = 0) then
    begin
        for select clt.crel_other_c_id, clt.c_name_fio
            from hp_client_relation_list(:c_id_tmp, 0) clt
            where crel_from_rel_id in ('1', '2', '3', '4', '5', '6')
            into :crel_other_c_id, res_info_text
        do
        begin
          if (is_found = 0) then
          begin
            for select kont.k_id
                from kontract kont
                inner join product pr on pr.pr_id = kont.k_from_pr_id
                inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
                inner join kontstate ks on ks.ks_from_k_id = kont.k_id
                where kncl.kc_from_c_id = :crel_other_c_id and
                      kncl.kc_amount_approved > 0 and
                      ks.ks_state = 1 and
                      ks.ks_ol_is_delete = 0 and
                      pr.pr_code not containing '.Actiya_Ishenim_Refin.' and
                      kont.k_disb_date < cast('01.04.2021' as date) and
                      kont.k_kredline_end < cast('01.10.2021' as date) and
                      :k_disb_date between ks.ks_date_begin and ks.ks_date_end
                into :k_id_tmp1
            do
            begin
              if (exists(select *
                         from operkred ok
                         where ok.ok_from_k_id = :k_id_tmp1 and
                               ok.ok_type = 4 and
                               ok.ok_ol_is_active = 1 and
                               ok.ok_ol_date between cast('01.05.2021' as date) and cast('15.06.2021' as date))) then
              begin
                is_found = 1;
                break;
              end
            end
          end
        end
    end

    if (is_found = 0) then
    begin
      res_error_is_critical = 1;
      res_error_text = 'Кредит не соответсвует условиям Акции по рефинансирвоанию ч/з продукт Ишеним';
      suspend;
    end
  end
end
