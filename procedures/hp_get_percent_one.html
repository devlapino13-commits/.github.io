CREATE PROCEDURE HP_GET_PERCENT_ONE(
    K_ID D_ID,
    K_LEVEL D_SMALLINT,
    K_DISB_DATE D_DATE = null,
    K_PERCENT_PER_DAY D_PERCENT = null,
    K_FROM_PR_ID D_INTEGER = null)
RETURNS (
    NEW_KD_PERCENT DECIMAL(18,4))
AS
declare variable NEW_LEVEL D_SMALLINT;
declare variable K_PERCENT_ON_DISB D_PERCENT;
declare variable DB_ID D_INTEGER;
begin
  db_id = f_get_db_id();

  if (k_disb_date is null or k_percent_per_day is null or k_from_pr_id is null) then
  begin
    select k_disb_date,
           round(cast(k_percent_per_day as double precision) * iif(:db_id = 81001, 30.0, pr_percent_for_days), 2) -- для РФ это РФ, вычисляется месячная % ставка, если КГ тогда годовая
    from kontract, product pr
    where k_id = :k_id and
          k_from_pr_id = pr_id and
          pr_percent_kind = 1 -- плавающая процентная ставка
    into k_disb_date,
         k_percent_on_disb;
  end else
  begin
    select round(cast(:k_percent_per_day as double precision) * iif(:db_id = 81001, 30.0, pr_percent_for_days), 2) from product
    where pr_id = :k_from_pr_id
    and  pr_percent_kind = 1
    into k_percent_on_disb;
  end 

  if (k_percent_on_disb is null) then exit; -- не имеется плавающей ставки

  new_level = k_level;

  if (db_id = 81001) then
  begin
    if (k_disb_date < '01.11.2020') then
    begin
      if (new_level = 1) then
        new_kd_percent = 8.2;
      if (new_level = 2) then
        new_kd_percent = 9.0;
    end
    else
    if (k_disb_date < '08.09.2021') then
    begin
      if (new_level = 1) then
        new_kd_percent = k_percent_on_disb + 1.0;
      if (new_level = 2) then
        new_kd_percent = k_percent_on_disb + 2.0;
      if (new_level = 3) then
        new_kd_percent = k_percent_on_disb + 2.5;
    end
    else
    if (new_level = 1) then
      new_kd_percent = k_percent_on_disb;

    if (k_id in (101010081081400101, -- СРОЧНО! ВАЖНО! Не выдавать Займы сотрудникам свыше 100 рублей! письмо от Зары от Пт 21.01.2022 18:39
             101010081152300101, 101010081446600101, 101010081922200101, 101010082131400101, 101010083321700101, 101010083676500101, 101010083873900101, 101010084172300101, 101010084863500101)) then
      new_kd_percent = 4;

  end
  suspend;
end
