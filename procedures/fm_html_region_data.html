CREATE PROCEDURE FM_HTML_REGION_DATA(
    REG_ID INTEGER)
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable DB_ID integer;
declare variable FIO varchar(100);
declare variable CELL varchar(100);
declare variable C_ID ID_TYPE;
declare variable CHANGER_FIO D_STR_100;
declare variable CHANGER_CELL D_STR_100;
declare variable CHANGER_ID D_ID;
begin
  res_text = (select res_style
              from fm_html_styles) || '<body>';

  db_id = f_get_db_id();

  res_text = blob_append(res_text,(select '<h1 class="green_text">' || reg_code || ' ' || reg_name || '</h1>' || 'Телефон: <b>' || coalesce(reg_phone, 'Не указан') || '</b><br>' || 'Email: <b>' || coalesce(reg_email_manager, 'Не указан') || '</b><br>' || 'Регион: <b>' || reg_stat_rep_region || '</b><br>' || 'Адрес: <b>' || adl_address_str || '</b><br>' || 'Ориентир: <b>' || coalesce(adl_comment, 'Не указан') || '</b><br>'
                         from region, addrlist
                         where reg_unicode = :reg_id and
                               reg_address_fact = adl_id));

  res_text = blob_append(res_text, '<br>Менеджер:<br>');

  for select c_name_fio,
             (select cont_value
              from hp_contdata_get_cell_valid(0, c_id)), c_id
      from hr_pstatus, joblist, jobkind, clients
      where ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.Region_KM' and
            current_date between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_id and
            ps_from_c_id = c_id
      into :fio, :cell, :c_id
  do
  begin
    select hrv_changer_id
    from hr_voctravel, hr_orders
    where hrv_from_ho_id = ho_id and
    hrv_from_c_id = :c_id and
    ho_order_type in (7, 8, 9, 10, 13, 16, 17, 18) and
    current_date between hrv_date_begin and hrv_date_end_fact
    order by hrv_date_begin desc
    rows 1
    into :changer_id;

    if (changer_id is not null and changer_id > 0) then
    begin
      select c_name_fio, (select cont_value
              from hp_contdata_get_cell_valid(0, :changer_id))
      from clients
      where c_id = :changer_id
      into :changer_fio, changer_cell;

      res_text = blob_append(res_text, '/заменяющее лицо: ' || changer_fio || ', ' || '<td align="center">'|| f_uri_phone(changer_cell) ||'<br>');
    end
    else
    begin
      res_text = blob_append(res_text, fio || ', ' || '<td align="center">'|| f_uri_phone(cell) ||'<br>');
    end
  end
  res_text = blob_append(res_text, '</body>');
  suspend;
end
