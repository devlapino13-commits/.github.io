CREATE PROCEDURE CM_RU_0069_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare variable c_id id_type;
declare variable err integer;
declare variable rd_id integer;
declare variable adl_id id_type;
begin
  select rd.rd_id
  from refdata rd where rd.rd_code = 'Citizen.RU'
  into :rd_id;

  if (not exists (select *
                    from kontract, product
                    where k_id = :k_id
                    and k_from_pr_id = pr_id
                    and pr_code containing '.EMP.')) then
  
  for select cl.c_id, cl.c_name_fio
  from knclient kncl
  inner join clients cl on cl.c_id = kncl.kc_from_c_id
  and kncl.kc_amount_approved >0
  and kncl.kc_from_k_id = :k_id
  and cl.c_citizen <> :rd_id -- не граждане РФ
  into :c_id, :res_info_text
  do
  begin
    /*
    if (adc_address_type=1) 'Юридически';
    if (adc_address_type=2) 'Фактически';
    if (adc_address_type=3) 'Место работы/бизнеса';
    if (adc_address_type=4) 'Юр.родина';
    if (adc_address_type=5) 'Факт.родина';
    */
    select adcl.adc_from_adl_id
    from addrclient adcl
    where adcl.adc_from_c_id = :c_id
    and adcl.adc_address_type = 5
    into :adl_id;
    err = 0;
    res_error_text = null;
    res_error_is_critical = null;
    if (adl_id is null) then err = 1;
    else if(right(adl_id,9) = '000000000') then err = 1;
    if(err = 1) then
    begin
        res_error_text = 'Укажите фактический адрес клиента на Родине';
        res_error_is_critical = 1;
        suspend;
    end
  end
end
