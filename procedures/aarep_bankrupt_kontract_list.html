CREATE PROCEDURE AAREP_BANKRUPT_KONTRACT_LIST(
    REP_DATE_END D_DATE)
RETURNS (
    REC_NO D_INTEGER,
    KC_ID D_ID,
    C_ID D_ID,
    EIL_ID D_ID,
    K_KONTRACT_NO D_STR_20,
    STATUS_DATE D_DATE,
    STATUS D_STR_CODE,
    IS_NEW D_BOOL,
    K_DISB_DATE D_DATE,
    PR_NAME VARCHAR(50),
    OKS_DELINQ_DAYS D_INTEGER,
    OKS_ECL_ACC_KIND D_SMALLINT,
    OKS_LAST_SH_DATE D_DATE,
    C_NAME_FIO D_STR_100,
    DATE_SUD_RD D_DATE,
    DATE_SUD_RI D_DATE,
    DATE_SUD_RIZ D_DATE,
    DATE_SUD_KPZ D_DATE,
    DATE_NSUD_VPVB D_DATE,
    DATE_NSUD_ZPVB D_DATE,
    STATUS_OTHER D_STR_100,
    IS_FINISHED D_BOOL,
    K_STOP_PERC D_DATE,
    DATE_OFF D_DATE,
    OKS_BALANCE D_CURRENCY,
    OKS_BAL_PERC D_CURRENCY,
    OX_AMOUNT_ZADOL D_CURRENCY,
    OKS_DATE_ECL_ACC_KIND_3 D_DATE,
    STATUS_OD D_SMALLINT,
    OKS_DELINQ_DATE DATE)
AS
declare variable rep_date_begin d_date;
declare variable brt_date d_date;
declare variable k_id d_id;
declare variable bal_os_off d_currency;
declare variable bal_perc_off d_currency;
begin
  rec_no = 0;
  rep_date_begin = dateadd(day, -extract(day from rep_date_end) + 1, rep_date_end);

  -- Получаем список всех банкротов за отчетный месяц.

  for select distinct brt_from_c_id, max(brt_from_eil_id), min(brt_date_begin)
      from bankrupt
      where (:rep_date_begin between brt_date_begin and brt_date_end or
            :rep_date_end between brt_date_begin and brt_date_end)
      group by brt_from_c_id
      into c_id, eil_id, brt_date
  do
  begin
    select status_date, status, is_finished
    from hp_get_kontur_info(:eil_id)
    into status_date, status, is_finished;

    is_new = 0;
    if (brt_date between rep_date_begin and rep_date_end) then
      is_new = 1;

    for select kc_id, k_kontract_no, k_disb_date, pr_name, iif(oks_close_state = 0, :rep_date_end - oks_delinq_date, -1),
               oks_last_sh_date, c_name_fio, k_stop_perc, k_id
        from knclient, kontract, oksummary, product, clients
        where kc_from_c_id = c_id and
              kc_from_k_id = k_id and
              k_from_pr_id = pr_id and
              oks_from_k_id = k_id and
              kc_amount_approved > 0 and
              :rep_date_end between oks_date_begin and oks_date_end and
              oks_close_state in (0, 1) and
              c_id = :c_id
        into kc_id, k_kontract_no, k_disb_date, pr_name, oks_delinq_days, oks_last_sh_date, c_name_fio, k_stop_perc,
             k_id
    do
    begin

      oks_balance = 0;
      oks_bal_perc = 0;
      ox_amount_zadol = 0;
      select oks_ecl_acc_kind, oks_balance, oks_bal_perc, iif(oks_is_month = 1, oks_bal_law, 0) + oks_off_law
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            :rep_date_end between oks_date_begin and oks_date_end
      into oks_ecl_acc_kind, oks_balance, oks_bal_perc, ox_amount_zadol;

      bal_os_off = 0;
      bal_perc_off = 0;

      select sum(oko_os + oko_sh_os) os, sum(oko_perc + oko_sh_perc) perc
      from operkroff
      where oko_from_k_id = :k_id and
            oko_from_c_id = :c_id and
            oko_ol_is_active = 1 and
            oko_ol_date <= :rep_date_end
      into bal_os_off, bal_perc_off;

      if (bal_os_off is null) then
        bal_os_off = 0;
      if (bal_perc_off is null) then
        bal_perc_off = 0;

      oks_balance = oks_balance + bal_os_off;
      oks_bal_perc = oks_bal_perc + bal_perc_off;

      select date_sud_rd, date_sud_ri, date_sud_riz, date_sud_kpz, date_nsud_vpvb, date_nsud_zpvb, status_other
      from hp_get_kontur_ext_info(:c_id, :k_disb_date, :rep_date_end)
      into date_sud_rd, date_sud_ri, date_sud_riz, date_sud_kpz, date_nsud_vpvb, date_nsud_zpvb, status_other;

      date_off = null;
      select ok_ol_date
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :k_id and
            ok_type = 5 and
            ok_os + ok_perc < 0
      order by ok_ol_date
      rows 1
      into date_off;

      if (oks_delinq_days = -1) then

      begin
        select oks_delinq_date
        from kn_oksummary
        where oks_from_kc_id = :kc_id and
              oks_date_begin < :rep_date_end and
              oks_delinq_date is not null
        order by oks_date_begin desc
        rows 1
        into :oks_delinq_date;

        oks_delinq_days= datediff(day, oks_delinq_date, date_off);
      end

      oks_date_ecl_acc_kind_3 = null;
      select oks_date_begin
      from kn_oksummary
      where oks_date_begin <= :rep_date_end and
            oks_from_k_id = :k_id and
            oks_ecl_acc_kind = 3
      order by oks_date_begin asc
      rows 1
      into :oks_date_ecl_acc_kind_3;

      status_od = 1;

      if ((date_off is not null) and
          (date_off <= :rep_date_end)) then
        status_od = 2;
      if ((date_off is not null) and
          (date_off > :rep_date_end)) then
        status_od = 1;

      rec_no = rec_no + 1;
      suspend;
    end
  end
end
