CREATE PROCEDURE CM_KZ_0040_SQL(
    MA_ID D_INTEGER NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable C_ID D_ID;
declare variable RES_STATUS D_SMALLINT;
begin
  -- RFCKZ-996
  res_info_text = 'Проверка статуса СУСН';
  res_error_text = '';
  res_error_is_critical = 0;

  for select mac.mac_from_c_id
      from miniappclient mac
      where mac.mac_from_ma_id = :ma_id and
            mac.mac_amount > 0
      into :c_id
  do
  begin
    res_status = null;

    select res_status
    from hp_susn_get_status(:c_id, :ma_id)
    into :res_status;

    if (coalesce(res_status, 0) <> 2) then
    begin
      res_error_text = 'Статус СУСН еще не получен. Для утверждения займа без данного статуса необходимо выполнить утверждение через ФС.';
      res_error_is_critical = 0;
      suspend;
      exit;
    end
  end
end
