CREATE PROCEDURE AAREP_KOMMIS_DIFF_ALL_CL(
    NEW_VALUE D_PERCENT,
    SHOW_ROW_COUNT D_INTEGER,
    FIX_3 D_INTEGER,
    CUSTOM_KOEF D_PERCENT)
RETURNS (
    REC_NO D_INTEGER,
    K_ID D_ID,
    K_KONTRACT_NO D_STR_CODE,
    K_DISB_DATE D_DATE,
    K_PERIOD D_CURRENCY,
    K_PERCENT_YEAR D_PERCENT,
    C_NAME_FIO D_STR_100,
    C_BIRTH_DATE D_DATE,
    ORIG_AMOUNT D_CURRENCY,
    ORIG_PERC D_CURRENCY,
    ORIG_KOMMIS D_CURRENCY,
    NEW_AMOUNT D_CURRENCY,
    NEW_PERC D_CURRENCY,
    DIFF_PLUS_PERC D_CURRENCY,
    LOAN_STATUS D_STR_CODE,
    LAST_DATE D_DATE,
    PERCENT_WITH_KOEF D_PERCENT,
    FACT_CLOSE_DATE D_DATE)
AS
declare variable tmp_orig_amount d_currency;
declare variable tmp_orig_perc d_currency;
declare variable tmp_orig_kommis d_currency;
declare variable tmp_new_amount d_currency;
declare variable tmp_new_perc d_currency;
declare variable tmp_diff_plus_perc d_currency;
declare variable koef d_double;
declare variable client_count d_integer;
begin
  -- Если new_perc <= 2, то это коэффициент
  -- Иначе это абсолютное значение к проценту

  rec_no = 0;
  for select a.k_id, k.k_kontract_no, k.k_disb_date, client_count, orig_amount, orig_perc, orig_kommis,
         new_amount, new_perc, diff_plus_perc, k.k_period, round(k.k_percent_per_day * 365.0, 2),
         last_date, percent_with_koef
  from aarep_kommis_diff_list_short2(:new_value, :show_row_count, :fix_3, :custom_koef) a, kontract k
  where diff_client_count > 0 and
        a.k_id = k.k_id
  into :k_id, k_kontract_no, k_disb_date, :client_count, :tmp_orig_amount, :tmp_orig_perc, :tmp_orig_kommis,
       :tmp_new_amount, :tmp_new_perc, :tmp_diff_plus_perc, k_period, k_percent_year,
       :last_date, :percent_with_koef
  do
  begin
    loan_status = 'Займ активный';
    /*

    if (exists(select * from kontstate
    where ks_from_k_id = :k_id and
          ks_state = 1 and
          ks_ol_is_delete = 0 and
          ks_date_end <= cast('31.12.2999' as date))) then
      loan_status = 'Закрыт';  */

    fact_close_date = null;

    select ks.ks_date_begin
    from kontstate ks
    where ks.ks_from_k_id = :k_id
    and ks.ks_ol_is_delete = 0
    and current_date between ks.ks_date_begin and ks.ks_date_end
    and ks.ks_state = 2
    into :fact_close_date;

    if (fact_close_date is not null) then
        loan_status = 'Закрыт';    

    for select kc_amount_approved, c_name_fio, c_birth_date
    from knclient, clients
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    into orig_amount, c_name_fio, c_birth_date
    do
    begin
      orig_perc = tmp_orig_perc;
      orig_kommis = tmp_orig_kommis;
      new_amount = tmp_new_amount;
      new_perc = tmp_new_perc;
      diff_plus_perc = tmp_diff_plus_perc;

      if (client_count > 1) then
      begin
        koef = cast(orig_amount as d_double) / tmp_orig_amount;
        orig_perc = round(orig_perc * koef, 0);
        orig_kommis = round(orig_kommis * koef, 0);
        new_amount = round(new_amount * koef, 0);
        new_perc = round(new_perc * koef, 0);
        diff_plus_perc = round(diff_plus_perc * koef, 0);
      end

      rec_no = rec_no + 1;
      suspend;
    end
  end
end
