CREATE PROCEDURE CM_RU_0306_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable fname d_str_100;
declare variable mname d_str_100;
declare variable sname d_str_100;
declare variable fname_eng d_str_100;
declare variable mname_eng d_str_100;
declare variable sname_eng d_str_100;
declare variable c_alternat_fio d_str_100;
begin
  -- RFCRU-5644
  for
    select
      c_id,
      c_name_fio,
      fl1.fl_name2,
      fl1.fl_name3,
      fl1.fl_name1,
      fl2.fl_name2,
      fl2.fl_name3,
      fl2.fl_name1,
      c_alternat_fio
    from kontract
      inner join knclient
        on kc_from_k_id = k_id
          and kc_amount_approved > 0
      inner join clients
        on c_id = kc_from_c_id
      left join fiolist as fl1
        on fl1.fl_id = c_from_fl_id
      left join fiolist as fl2
        on fl2.fl_id = c_name_eng_fl_id
    where k_id = :k_id
  into
    :c_id,
    :c_name_fio,
    :fname,
    :mname,
    :sname,
    :fname_eng,
    :mname_eng,
    :sname_eng,
    :c_alternat_fio
  do
  begin
    res_error_text = (
      select
        err_txt
      from hp_separate_fio_checkvalid(:c_id, :fname, :mname, :sname, :fname_eng, :mname_eng, :sname_eng, :c_alternat_fio));
    if (res_error_text <> '') then
    begin
      res_error_is_critical = 2;
      suspend;
    end
  end
end
