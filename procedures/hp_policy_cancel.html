CREATE PROCEDURE HP_POLICY_CANCEL
AS
declare variable k_id d_id;
begin
    for select ik.ik_from_k_id
    from insurance_kontract ik
    inner join enum_list el on ik.ik_enum_status = el.el_id
        and el.el_kind = 'insurance_kontract'
        and el.el_code = 'edoc_was_cancelled'
    where ik.ik_policy_no <> ''
        and cast(ik.ik_last_datetime as date) = current_date
    into :k_id
    do
        execute procedure hp_extinfo_insert('policy_cancel', f_objkind('kontract', 'id'), :k_id, 1);
end
