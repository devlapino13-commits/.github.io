CREATE PROCEDURE FSROBOT_SMS_SEND_DAILY
AS
declare variable DB_ID integer;
declare variable USER_ID ID_TYPE;
declare variable CLIENT_ID ID_TYPE;
declare variable MSG_BODY varchar(255);
declare variable MSG_MAIL_TO varchar(255);
declare variable MSG_LIFETIME_DELINQ integer;
declare variable MSG_LIFETIME_BIRTHDAY integer;
declare variable MSG_LIFETIME_REPAY integer;
declare variable MSG_COMMAND_KIND varchar(30);
declare variable PUSH_BODY D_STR_LONG;
declare variable CL_NAME STR_100;
declare variable C_ALTERNAT_FIO STR_100;
declare variable MSG_KIND D_STR_CODE;
declare variable CFG_NAME D_STR_CODE;
declare variable MSG_LIFETIME_EXP_PASS D_INTEGER;
declare variable KC_ID D_ID;
declare variable mcl_project d_integer;
declare variable msg_sender_code d_str_code;
declare variable k_id d_id;
begin
 if (exists(select * from smssendok -- если в данной таблице имеется запись на данную дату то СМС уже было отправлено
 where sso_date = current_date)) then
 begin
   exit;
 end

  cl_name = '';
  -- Отправка ежедневных СМС (бывш. SmsSvc). Вызывается в FSROBOT_SH_MAIN

  -- Отправка от имени пользователя FINSTART_ROBOT (id: 1000000000000)
  user_id = 1000000000000;

  -- Сроки хранения отправленных СМС:
  msg_lifetime_delinq = 365 * 3; -- Уведомления о просрочках, 3 года
  msg_lifetime_birthday = 1; -- Поздравления с днем рождения, 1 день
  msg_lifetime_repay = 365; -- Напоминания о погашении, 1 год
  msg_lifetime_exp_pass = 30; -- Уведомление об истечении паспорта 30 дней

  -- Имя конфигурации отправки (эта процедура)
  msg_command_kind = 'fsrobot_sms_send_daily';

  db_id = f_get_db_id();

  -- СМС для клиентов РФ об истечении паспорта гражданина РФ
  if (db_id = 81001) then
  begin
    msg_body = 'Уважаемый, клиент! В связи с истечением срока действия паспорта гражданина РФ, просим Вас обратиться в Компанию, предоставив документ, удостоверяющий личность.';
    push_body = msg_body;
    for select distinct c_id
        from aarep_cli_passport_actual_check(1000000, current_date)
        where c_pass_date_expire = current_date - 1
        into client_id
    do
    begin
      msg_kind = f_get_msg_kind(client_id, 'aarep_cli_passport_actual_check', '', db_id);
      if (msg_kind = 'sms' and
          msg_body <> '' and
          msg_mail_to starting '7') then
      begin
        select cont_value
        from hp_contdata_get_cell_main(0, :client_id)
        into msg_mail_to;

        mcl_project = f_get_mcl_project(f_objkind('clients', 'id'), client_id);
        msg_sender_code = f_enum_code('sms_sender', iif(:mcl_project = 2, 1, 0));

        f_send_sms(msg_mail_to, msg_body, 'clients', client_id,
                   'aarep_cli_passport_actual_check', msg_lifetime_exp_pass, user_id, 0, msg_sender_code);

      end
      else
      if (msg_kind = 'push' and
          push_body <> '') then
        execute procedure mc_send_push('clients', client_id,
            'Истек срок действия паспорта', push_body,
            msg_lifetime_exp_pass);

      if (msg_kind in ('push', 'sms')) then
        insert into clipassdelaysms (cpds_date, cpds_timestamp, cpds_c_id,
                                     cpds_msg_id)
        values (current_date, current_timestamp, :client_id, 0);
    end
  end

  -- Поздравление клиентов с днем рождения
  for select distinct phone, client_id, c_alternat_fio
      from aarep_sms_birthday(current_date)
      into msg_mail_to, client_id, c_alternat_fio

  do
  begin
    select txt2
    from hp_fio_clear(:c_alternat_fio)
    into :cl_name;

    cl_name = coalesce(cl_name, '');
    if (cl_name = '') then
      cl_name = 'klient';

    push_body = '';

    if (db_id = 94401) then
      msg_body = 'Hormetli mushteri, Sizin ad gunuzu tebrik edirik! ' || 'Ishlerinizde ugurlar ve can sagligi arzu edirik. Hormetle M Bulak.';

    if (db_id = 81001) then
    begin
      msg_body = :cl_name || ', pozdravlyaem s dnem rojdeniya! Jelaem zdorovya, dostatka i blagopoluchiya v semye. Budte schastlivy i uspeshny!';
      push_body = 'Уважаемый ' || :cl_name || '. Спешим поздравить Вас с днём рождения. В этот день мы хотим пожелать Вам благополучия, крепкого здоровья и успехов в любых начинаниях! Надеемся быть полезными Вам и в будущем';
    end

    --RFCKG-2987
    if (db_id in (39801, 41701)) then
      push_body = 'Уважаемый клиент М Булак, С днем рождения! Желаем счастья и удачи';

    msg_kind = f_get_msg_kind(client_id, 'aarep_sms_birthday', '', db_id);

    --RFCKG-2987
    if (db_id in (39801, 41701)) then
      msg_kind = 'push';

    if (db_id <> 81001 or f_clientcom_allowed(client_id, 1, 1, 1) = 1) then
    begin
      if (msg_kind = 'sms' and
          msg_body <> '') then
      begin
        msg_sender_code = null;

        if (db_id = 81001) then
        begin
            mcl_project = f_get_mcl_project(f_objkind('clients', 'id'), client_id);
            msg_sender_code = f_enum_code('sms_sender', iif(:mcl_project = 2, 1, 0));
        end

        f_send_sms(msg_mail_to, msg_body, 'clients', client_id, 'aarep_sms_birthday', msg_lifetime_birthday, user_id, 0, msg_sender_code);
      end
      else
      if (msg_kind = 'push' and
          push_body <> '') then
        execute procedure mc_send_push('clients', client_id,
            'Поздравление с днем рождения!', push_body,
            msg_lifetime_birthday);
    end
  end

  if (db_id = 81001) then
  begin
    -- Напоминание на 1, 3, 5, 10 день просрочки
    for select distinct phone, sms_message, push_message, client_id, '1', kc_id
        from aarep_sms_kont_delinq(current_date, 1)
        union
        select distinct phone, sms_message, push_message, client_id, '3', kc_id
        from aarep_sms_kont_delinq(current_date, 3)
        union
        select distinct phone, sms_message, push_message, client_id, '5', kc_id
        from aarep_sms_kont_delinq(current_date, 5)
        union
        select distinct phone, sms_message, push_message, client_id, '10', kc_id
        from aarep_sms_kont_delinq(current_date, 10)
        into msg_mail_to, msg_body, push_body, client_id, cfg_name, kc_id
    do
    begin
      msg_kind = f_get_msg_kind(client_id, 'aarep_sms_kont_delinq', cfg_name, db_id);
      if (msg_kind = 'sms' and
          msg_body <> '') then
      begin
        select kc.kc_from_k_id
        from knclient kc
        where kc.kc_id = :kc_id
        into :k_id;

        mcl_project = f_get_mcl_project(f_objkind('kontract', 'id'), :k_id);
        msg_sender_code = f_enum_code('sms_sender', iif(:mcl_project = 2, 1, 0));

        f_send_sms(msg_mail_to, msg_body, 'knclient', kc_id, 'aarep_sms_kont_delinq.' || cfg_name, msg_lifetime_delinq, user_id, 0, :msg_sender_code);
      end
      else
      if (msg_kind = 'push' and
          push_body <> '') then
        f_send_push('knclient', kc_id, 'Напоминание о просрочке', push_body, 'aarep_sms_kont_delinq.' || cfg_name, msg_lifetime_delinq, user_id);   -- RFCRU-4115
    end
  end

  if (db_id = 94401) then
  begin
    -- Напоминание на 1, 5 день просрочки
    for select distinct phone, sms_message, client_id
        from aarep_sms_delinq(current_date, 1)
        into msg_mail_to, msg_body, client_id
    do
    begin
      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body,
                                  msg_lifetime_days, msg_from_un_id,
                                  msg_command_kind, msg_object_kind,
                                  msg_object_id)
      values (1, :msg_mail_to, '', :msg_body, :msg_lifetime_delinq, :user_id,
              :msg_command_kind, 0, :client_id);
    end
  end

  if (db_id = 81001) then
  begin
    -- Напоминание о погашении за 3 дня
    for select distinct phone, sms_message, client_id, kontract_id, push_message
        from aarep_sms_repay_3day(current_date, 3, 0)
        into msg_mail_to, msg_body, client_id, :k_id, push_body
    do
    begin
      msg_kind = f_get_msg_kind(client_id, 'aarep_sms_repay_3day', '3', db_id);
      if (msg_kind = 'sms' and
          msg_body <> '') then
      begin
        mcl_project = f_get_mcl_project(f_objkind('kontract', 'id'), :k_id);
        msg_sender_code = f_enum_code('sms_sender', iif(:mcl_project = 2, 1, 0));

        f_send_sms(msg_mail_to, msg_body, 'clients', client_id, 'aarep_sms_repay_3day.3', msg_lifetime_repay, user_id, 0, msg_sender_code);
      end
      else
      if (msg_kind = 'push' and
          push_body <> '') then
        execute procedure mc_send_push('clients', client_id,
            'Напоминание о погашении', push_body,
            msg_lifetime_repay);
    end
  end

  if (db_id = 94401) then
  begin
    -- Напоминание о погашении, за 1 день
    for select distinct phone, sms_message, client_id
        from aarep_sms_repay_3day(current_date, 1, 0)
        into msg_mail_to, msg_body, client_id
    do
    begin
      insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body,
                                  msg_lifetime_days, msg_from_un_id,
                                  msg_command_kind, msg_object_kind,
                                  msg_object_id)
      values (1, :msg_mail_to, '', :msg_body, :msg_lifetime_repay, :user_id,
              :msg_command_kind, 0, :client_id);
    end
  end

  if (db_id = 94401) then
  begin
    -- Напоминание об отпуске (Эльнара из отдела УЧР)
    for select c_id,
               'Hörmətli ' || c_name_fio || ', ' || right('0' || extract(day from vr_date_begin), 2) || '.' || right('0' || extract(month from vr_date_begin), 2) || '.' || extract(year from vr_date_begin) || '-ci il tarixindən ' || right('0' || extract(day from vr_date_end), 2) || '.' || right('0' || extract(month from vr_date_end), 2) || '.' || extract(year from vr_date_end) || '-ci il tarixinə kimi məzuniyyətdə hesablanacaqsınız.'
        from vacationrequest vr, hr_pstatus, clients
        where current_date = vr_date_begin - 5 and
              vr.vr_from_c_id = c_id and
              ps_from_c_id = c_id and
              current_date between ps_date_begin and ps_date_end and
              ps_status = 1 and
              ps_is_working = 1
        into client_id, msg_body
    do
    begin
      select cont_value
      from hp_contdata_get_cell_main(0, :client_id)
      into msg_mail_to;

      if (msg_mail_to starting '994') then
      begin
        insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject,
                                    msg_body, msg_lifetime_days, msg_from_un_id,
                                    msg_command_kind, msg_object_kind,
                                    msg_object_id)
        values (1, :msg_mail_to, '', :msg_body, 30, :user_id, :msg_command_kind,
                0, :client_id);
      end
    end
  end

  -- Отчет об отправке
  client_id = 1000000000000;
  msg_mail_to = 'finstart@mbfplanet.com';
  msg_body = 'СМС поставлены в очередь на отправку ' || '(Дни рождения, напоминания о погашении, уведомления о просрочках).';

  insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body,
                              msg_lifetime_days, msg_from_un_id,
                              msg_command_kind, msg_object_kind, msg_object_id)
  values (0, :msg_mail_to, 'СМС рассылка произведена',
          :msg_body, 1, :user_id, :msg_command_kind, 0, :client_id);

  insert into smssendok (sso_date)
  values (current_date);
end
