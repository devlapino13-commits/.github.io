CREATE PROCEDURE HP_DO_KONTMATOWN_SELL(
    K_ID D_ID,
    OL_ID D_ID)
AS
declare variable kontract_no d_str_20;
declare variable disb_date d_date;
declare variable kml_id d_integer;
declare variable bill_id d_id;
declare variable bill_no d_integer;
declare variable wps_id d_id;
declare variable wpr_id d_id;
declare variable mols_id d_id;
declare variable molr_id d_id;
declare variable seller_id d_id;
declare variable offices_id d_integer;
declare variable officer_id d_integer;
declare variable ml_id d_id;
declare variable ml_name d_str_255;
declare variable qty d_currency;
declare variable summa d_currency;
declare variable summaw d_currency;
declare variable un_id d_id;
declare variable ol_id1 d_id;
declare variable ol_id2 d_id;
declare variable ol_id3 d_id;
declare variable parent_ol_id d_id;
declare variable ob_detail d_str_long;
declare variable acc_bal d_str_20;
declare variable acc_stock d_str_20;
declare variable acc1 d_str_20;
declare variable acc2 d_str_20;
declare variable office_id d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable line_no d_integer;
declare variable num d_integer;
declare variable kc_id d_id;
declare variable kwp_id d_id;
begin
  un_id = (
    select ol_user_id
    from operlist
    where ol_id = :ol_id);

  -- проводки при выдаче займа для ТМЦ по ИПФ (собств.)
  select
    k_kontract_no,
    k_disb_date,
    k_last_reg_id,
    ks_kc_id
  from kontract
    inner join kontstate
      on ks_from_k_id = k_id
        and k_disb_date between ks_date_begin and ks_date_end
        and ks_state = 1
        and ks_ol_is_delete = 0
  where k_id = :k_id
  into
    :kontract_no,
    :disb_date,
    :officer_id,
    :kc_id;
  kwp_id = (
    select
      wp_id
    from workplace
    where wp_from_reg_id = :officer_id
      and wp_status = 6
      and wp_name starting 'Собственность по ИПФ');

  summaw = 0;
  for
    select
      kmo_kml_id,
      kmo_bill_id,
      kmo_bill_line_no,
      kmo_wp_id,
      kmo_motv_id,
      kmo_c_id,
      kmo_reg_id,
      kmo_ml_id,
      ml_name,
      kmo_quantity,
      kmo_total
    from kontmat_own
      inner join matlist
        on ml_id = kmo_ml_id
    where kmo_k_id = :k_id
  into
    :kml_id,
    :bill_id,
    :bill_no,
    :wps_id,
    :mols_id,
    :seller_id,
    :offices_id,
    :ml_id,
    :ml_name,
    :qty,
    :summa
  do
  begin
    molr_id = mols_id;
    wpr_id = wps_id;
    summaw = summaw + summa;

    select
      mc_acc_balance || '/' || mc_acc_balance_no,
      mc_acc_stock || '/' || mc_acc_stock_no
    from operbill
      inner join matcateg
        on mc_id = ob_from_mc_id
    where ob_bill_id = :bill_id
      and ob_bill_line_no = :bill_no
      and :disb_date between ob_date_begin and ob_date_end
      and ob_is_delete = 0
    into
      :acc_bal,
      :acc_stock;

    acc1 = (
      select
        al_id
      from acc_list
      where al_from_reg_id = :offices_id
        and al_currency = '417'
        and al_account = :acc_stock);
    acc2 = acc1;
    ob_detail = f_format('%1, договор %2', trim(ml_name), kontract_no);

    -- когда офис хранения ТМЦ не соответствует офису выдачи займа
    if (offices_id <> officer_id) then
    begin
      molr_id = kc_id;
      wpr_id = kwp_id;
      -- бух.проводки по перемещению ТМЦ между складами
      parent_ol_id = ol_id;
      acc_dt = (
        select al_id from acc_list
        where al_from_reg_id = :officer_id
          and al_currency = '417'
          and al_account = '11209/10');
      acc_kt = (
        select al_id from acc_list
        where al_from_reg_id = :offices_id
          and al_currency = '417'
          and al_account = '11209/10');
      num = (select num from last_doc_number(:disb_date, 10, :offices_id));
      for
        select
          res_reg_id,
          res_acc_dt,
          res_acc_kt,
          res_line_no
        from aarep_loronostro_link(:offices_id, :acc_dt, :acc_kt)
      into
        :office_id,
        :acc_dt,
        :acc_kt,
        :line_no
      do
      begin
        insert into operlist (
          ol_parent_id,
          ol_from_reg_id,
          ol_oper_type,
          ol_oper_subtype,
          ol_date,
          ol_doc_no,
          ol_doc_kind,
          ol_memo,
          ol_user_id)
        values (
          :parent_ol_id,
          :office_id,
          3,
          4,
          :disb_date,
          :num,
          10,
          'Перемещение ТМЦ',
          :un_id)
        returning ol_id
        into :ol_id1;

        if (line_no = 1) then
        begin
          parent_ol_id = ol_id1;
          -- АПП ТМЦ
          acc2 = (
            select
              al_id
            from acc_list
            where al_from_reg_id = :officer_id
              and al_currency = '417'
              and al_account = :acc_stock);

          insert into opermat (
            om_from_ol_id,
            om_number,
            om_bill_id,
            om_bill_line_no,
            om_oper_type,
            om_workplace_id,
            om_matotvet_id,
            om_kolich_minus,
            om_summa_minus,
            om_summa_total,
            om_account,
            om_workplace_info,
            om_matotvet_info,
            om_account_info,
            om_main_line)
          values (
            :ol_id1,
            1,
            :bill_id,
            :bill_no,
            3,
            :wps_id,
            :mols_id,
            :qty,
            :summa,
            :summa,
            :acc1,
            :wpr_id,
            :molr_id,
            :acc1,
            1);
          insert into opermat (
            om_from_ol_id,
            om_number,
            om_bill_id,
            om_bill_line_no,
            om_oper_type,
            om_workplace_id,
            om_matotvet_id,
            om_kolich_plus,
            om_summa_plus,
            om_summa_total,
            om_account,
            om_workplace_info,
            om_matotvet_info,
            om_account_info,
            om_main_line)
          values (
            :ol_id1,
            2,
            :bill_id,
            :bill_no,
            3,
            :wpr_id,
            :molr_id,
            :qty,
            :summa,
            :summa,
            :acc2,
            :wps_id,
            :mols_id,
            :acc2,
            0);
        end

        insert into operbook (
          ob_from_ol_id,
          ob_currency,
          ob_curr_rate,
          ob_account_dt,
          ob_account_kt,
          ob_detail,
          ob_summa,
          ob_summa_val)
        values (
          :ol_id1,
          '417',
          1,
          :acc_dt,
          :acc_kt,
          :ob_detail,
          :summa,
          :summa);
      end
    end

    -- Выбытие ТМЦ
    num = (select num from last_doc_number(:disb_date, 12, :officer_id));
    insert into operlist (
      ol_parent_id,
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id)
    values (
      :ol_id,
      :officer_id,
      3,
      5,
      :disb_date,
      :num,
      12,
      'Выбытие ТМЦ',
      :un_id)
    returning
      ol_id
    into
      :ol_id2;

    insert into opermat (
      om_from_ol_id,
      om_number,
      om_bill_id,
      om_bill_line_no,
      om_oper_type,
      om_workplace_id,
      om_matotvet_id,
      om_kolich_minus,
      om_summa_minus,
      om_summa_total,
      om_account,
      om_is_written_off,
      om_main_line)
    values (
      :ol_id2,
      1,
      :bill_id,
      :bill_no,
      5,
      :wpr_id,
      :molr_id,
      :qty,
      :summa,
      :summa,
      :acc2,
      1,
      1);

    update kontmat_own set
      kmo_ol_id = :ol_id
    where kmo_kml_id = :kml_id;
  end

  if (summaw > 0) then
  begin
    num = (select num from last_doc_number(:disb_date, 12, :officer_id));
    acc_dt = (
      select al_id from acc_list
      where al_from_reg_id = :officer_id
        and al_currency = '417'
        and al_account = '11209/1');
    acc_kt = (
      select al_id from acc_list
      where al_from_reg_id = :officer_id
        and al_currency = '417'
        and al_account = '11209/10');

    insert into operlist (
      ol_parent_id,
      ol_from_reg_id,
      ol_oper_type,
      ol_oper_subtype,
      ol_date,
      ol_doc_no,
      ol_doc_kind,
      ol_memo,
      ol_user_id)
    values (
      :ol_id,
      :officer_id,
      3,
      5,
      :disb_date,
      :num,
      12,
      'ИПФ-cобственные товары фин-ние согл.дог. ' || :kontract_no,
      :un_id)
    returning
      ol_id
    into
      :ol_id2;

    insert into operbook (
      ob_from_ol_id,
      ob_currency,
      ob_curr_rate,
      ob_account_dt,
      ob_account_kt,
      ob_summa,
      ob_summa_val)
    values (
      :ol_id2,
      '417',
      1,
      :acc_dt,
      :acc_kt,
      :summaw,
      :summaw);
  end
end
