CREATE PROCEDURE HP_SELECT_FP_EXCLUDES
RETURNS (
    FPE_ID D_ID,
    C_ID D_ID,
    C_NAME_FIO D_STR_255,
    FPE_NAME D_STR_255,
    FPE_GR D_STR_255,
    REASON D_STR_255,
    CREATE_DATETIME D_DATETIME,
    CHANGE_DATETIME D_DATETIME,
    CREATE_UN_FIO D_STR_255,
    CHANGE_UN_FIO D_STR_255,
    IS_DELETE D_STR_255)
AS
begin
  for select fpe.fpe_id,
             c.c_id,
             c.c_name_fio,
             fpe.fpe_nameu,
             fpe.fpe_gr,
             fpe.fpe_reason,
             fpe.fpe_create_datetime,
             fpe.fpe_change_datetime,
             u1.c_name_fio,
             u2.c_name_fio,
             iif(fpe.fpe_is_delete = 0, 'Актуальна', 'Удалена')
  from finpolis_excludes fpe
  join username un1 on un1.un_id = fpe.fpe_create_un_id
  join username un2 on un2.un_id = fpe.fpe_change_un_id
  join clients c on c.c_id = fpe.fpe_from_c_id
  join clients u1 on u1.c_id = un1.un_from_c_id
  join clients u2 on u2.c_id = un2.un_from_c_id
  into fpe_id,
       c_id,
       c_name_fio,
       fpe_name,
       fpe_gr,
       reason,
       create_datetime,
       change_datetime,
       create_un_fio,
       change_un_fio,
       is_delete
  do
  suspend;
end
