CREATE PROCEDURE HP_PAYSYS_ONE_POLICY_OPER(
    PSI_ID DECIMAL(18,0),
    K_ID DECIMAL(18,0),
    UN_ID DECIMAL(18,0),
    OPER_MEMO VARCHAR(255),
    PSIP_AMOUNT DECIMAL(18,4),
    PSIP_TAX_AMOUNT DECIMAL(18,4),
    PSIP_PERC DECIMAL(18,7),
    C_ID DECIMAL(18,0))
RETURNS (
    PSM_ID DECIMAL(18,0),
    RES_TEXT VARCHAR(150))
AS
declare variable psi_date_begin date;
declare variable psi_date_end date;
declare variable kol integer;
declare variable source_kind integer;
declare variable source_id decimal(18,0);
declare variable str varchar(100);
declare variable disb_date date;
begin
  /* Надстройка над hp_paysys_one_card_oper, с доп. действиями над полисами*/

  if (k_id is not null) then
  begin
-- Проверка возраста клиента
    select k_disb_date
    from kontract
    where k_id = :k_id
    into disb_date;
    if (disb_date is null) then
    begin
      res_text = 'Дата рождения клиента не указана';
      suspend;
      exit;
    end
    if (exists(select 1
               from clients
               where c_id = :c_id and
                     c_birth_date <= dateadd(year, -70, :disb_date))) then
    begin
      res_text = 'Возраст клиента должен быть менее 70 лет';
      suspend;
      exit;
    end

-- Проверка на параллельные кредиты
    if (exists(select 1
               from kontract, product
               where k_id = :k_id and
                     k_from_pr_id = pr_id and
                     pr_product_kind = 2)) then
    begin
      res_text = 'По льготным продуктам страхование не допускается';
      suspend;
      exit;
    end
  end

  kol = 0;
  for select psl_object_kind, psl_object_id
      from paysys_list
      where psl_from_psi_id = :psi_id and
            psl_is_delete = 0
      group by psl_object_kind, psl_object_id
      having sum(psl_sign) > 0
      into :source_kind, source_id
  do
    kol = kol + 1;
  if (kol = 1) then -- Проверяем только когда 1, остальные проверки в hp_paysys_one_card_oper
  begin
    if (source_kind = 3 and
        k_id is not null) then
    begin
      str = '';
      select k_kontract_no
      from kontract
      where k_id = :source_id
      into :str;
      res_text = 'Бланк уже закреплен за дог. № ' || str;
      str = '';
      select c_name_fio
      from paysys_item_props, clients
      where psip_from_c_id = c_id and
            psip_from_psi_id = :psi_id
      into :str;
      res_text = res_text || ', кл. ' || str;
      suspend;
      exit;
    end
    /*Бланк не должен висеть на контракте, передача должна быть только на контракт*/
  end

  select psm_id, res_text
  from hp_paysys_one_card_oper(:psi_id, 3, :k_id, :un_id, :oper_memo)
  into :psm_id, :res_text;
  /*
    При аннулировании обновляется дата окончания в paysys_item (берем текущую)
    При выдаче обновляются даты начала и окончания (из контракта)
    добавляются клиент, сумма, налог, процент в paysys_item_props
  */
  if (res_text = '') then
  begin
    if (k_id is null) then -- Аннулирование
    begin
-- Обновляем дату окончания полиса
      update paysys_item
      set psi_date_end = current_date
      where psi_id = :psi_id;
    end
    else -- Выдача полиса
    begin
-- Обновляем дату начала, окончания полиса, берем с контракта
      select k_disb_date, k_kredline_end
      from kontract
      where k_id = :k_id
      into :psi_date_begin, :psi_date_end;

      update paysys_item
      set psi_date_end = :psi_date_end,
          psi_date_begin = :psi_date_begin
      where psi_id = :psi_id;

-- Добавляем запись с доп характеристиками полиса в paysys_item_props
      update or insert into paysys_item_props (psip_from_psi_id, psip_amount, psip_tax_amount, psip_perc,
                                               psip_from_c_id)
      values (:psi_id, :psip_amount, :psip_tax_amount, :psip_perc, :c_id);
    end
  end
  suspend;
end
