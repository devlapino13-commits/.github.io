CREATE PROCEDURE HP_CARDOPER_ANNUL_RUN(
    CON_ID D_INTEGER)
RETURNS (
    CS_ID D_INTEGER)
AS
declare variable ol_id d_id;
declare variable ol_parent_id d_id;
declare variable reg_id d_integer;
declare variable k_id d_id;
declare variable c_id d_id;
declare variable cca_id d_integer;
declare variable card_id d_integer;
declare variable summa d_currency;
declare variable summaperc d_currency;
declare variable cs_id1 d_integer;
declare variable cs_id2 d_integer;
declare variable cs_id3 d_integer;
declare variable status d_integer;
declare variable un_id d_id;
declare variable evnt_str d_str_100;
declare variable json_str d_str_8100;
declare variable packet_no d_str_10;
declare variable ticket_no d_integer;
declare variable new_limit d_currency;
begin
  for
    select
      con_ol_id,
      con_reg_id,
      con_k_id,
      con_c_id,
      con_cca_id,
      con_summa,
      con_summaperc,
      con_cs_id1,
      con_cs_id2,
      con_cs_id3,
      con_un_id,
      con_status
    from cardoper_annul
    where con_id = :con_id
      and con_status < 4
  into
    :ol_id,
    :reg_id,
    :k_id,
    :c_id,
    :cca_id,
    :summa,
    :summaperc,
    :cs_id1,
    :cs_id2,
    :cs_id3,
    :un_id,
    :status
  do
  begin
    if (status in (0, 1, 2)) then
    begin
      if (status = 1 and summaperc = 0) then
        status = 2;

      card_id = (
        select card_id
        from hp_client_bank_card_mb(:c_id, current_date, 0));

      select
        packet_no,
        ticket_no
      from cardsynctrn_create
      into
        :packet_no,
        :ticket_no;
      if (status = 0) then
      begin
        evnt_str = 'annul_repayment';
        json_str = f_format(
          '{"card_acct": "%1", "card_acct_ccy": "KGS", "tran_type": "120", ' ||
          '"tran_ccy": "KGS", "tran_amount": %2, "tran_pocket": "%3", "tran_ticket": "%4", ' ||
          '"bank_c": "21", "groupc": "01", "tran_date_time": "%5"}',
          cca_id, summa, packet_no, ticket_no, f_datetimetostr(current_timestamp));
      end else
      if (status = 1) then
      begin
        evnt_str = 'annul_repayment_perc';
        json_str = f_format(
          '{"card_acct": "%1", "card_acct_ccy": "KGS", "tran_type": "110", ' ||
          '"tran_ccy": "KGS", "tran_amount": %2, "tran_pocket": "%3", "tran_ticket": "%4", ' ||
          '"bank_c": "21", "groupc": "01", "tran_date_time": "%5"}',
          cca_id, summaperc, packet_no, ticket_no, f_datetimetostr(current_timestamp));
      end else
      if (status = 2) then
      begin
        evnt_str = 'annul_repayment_credit_limit_change';
        new_limit = summa - summaperc +
          coalesce((select amount from hp_balance_cardloans('cardacc', :cca_id)), 0);
        json_str = f_format(
          '{"bank_c": "21", "groupc": "01", "limit": %1, "card_acct": "%2"}',
          new_limit, cca_id);
      end

      insert into cardsync(
        cs_parent_id,
        cs_from_csc_id,
        cs_card_id,
        cs_from_cca_id,
        cs_object_kind,
        cs_object_id,
        cs_amount,
        cs_request_json,
        cs_office_id,
        cs_from_un_id,
        cs_from_k_id,
        cs_from_c_id
      )
      values(
        iif(:status = 0, null,
          iif(:status = 1 or :summaperc = 0, :cs_id1, :cs_id2)),
        f_get_cardsynccfg(:evnt_str, 'id'),
        :card_id,
        :cca_id,
        f_objkind('cardoper_annul', 'id'),
        :con_id,
        iif(:status = 0, :summa,
          iif(:status = 1, :summaperc, :new_limit)),
        :json_str,
        :reg_id,
        :un_id,
        :k_id,
        :c_id
      )
      returning
        cs_id
      into
        :cs_id;

      if (status = 0) then
        update cardoper_annul set
          con_status = 1,
          con_cs_id1 = :cs_id
        where con_id = :con_id;
      else
      if (status = 1) then
        update cardoper_annul set
          con_status = 2,
          con_cs_id2 = :cs_id
        where con_id = :con_id;
      else
      if (status = 2) then
        update cardoper_annul set
          con_status = 3,
          con_cs_id3 = :cs_id
        where con_id = :con_id;
    end else begin
      ol_parent_id = (
        select ol_parent_id
        from operlist where ol_id = :ol_id);
      update operlist set
        is_delete = 1
      where ol_id = coalesce(:ol_parent_id, :ol_id)
        and is_delete = 0;
      update opercash set
        opc_is_delete = 1
      where opc_from_ol_id = coalesce(:ol_parent_id, :ol_id)
        and opc_is_delete = 0;
      update cardoper_annul set
        con_status = 5
      where con_id = :con_id;

      when any do
      begin
        update cardoper_annul set
          con_status = - 1
        where con_id = :con_id;
      end
    end

    suspend;
  end
end
