CREATE PROCEDURE HP_REFLINK_CATEGORY(
    KEY_ID DECIMAL(18,0))
RETURNS (
    REF_CAPTION VARCHAR(255),
    REF_NAME VARCHAR(255),
    REF_TABLE_NAME VARCHAR(100),
    REF_LABEL_FIELD VARCHAR(255),
    REF_SELECT_SQL VARCHAR(5000))
AS
begin
-- Список доступных категорий для привязки объекта к свойству из RefData
-- Данные (связки) будут храниться в таблице RefLink
-- Возможные значения нижеуказанных справочников будут храниться в RefData (справочник RefLink)
-- Поле RD_Memo указывает к какому справочнику относится, RD_Str_ID - кодовое значение
-- Пример: RD_Memo='Region', RD_Str_ID='.mega.'

  ref_name = 'ConditionMain';
  ref_caption = 'Проверка кредитов';
  ref_table_name = 'ConditionMain';
  ref_label_field = 'cdm_check_kind';
  ref_select_sql = 'select cdm_id as id, cdm_memo as name, cdm_category as memo, cdm_is_active as is_active, coalesce(cdm_check_kind, '''') as label_value from conditionmain';
  if (coalesce(key_id, 0) <> 0) then
    ref_select_sql = ref_select_sql || ' where cdm_id=' || key_id;
  suspend;

  ref_name = 'Product';
  ref_caption = 'Продукты';
  ref_table_name = 'Product';
  ref_label_field = 'pr_code';
  ref_select_sql = 'select pr_id as id, pr_name as name, pr_memo as memo, pr_active as is_active, coalesce(pr_code, '''') as label_value from product';
  if (coalesce(key_id, 0) <> 0) then
    ref_select_sql = ref_select_sql || ' where pr_id=' || key_id;
  suspend;

  ref_name = 'Region';
  ref_caption = 'Офисы, регионы';
  ref_table_name = 'Region';
  ref_label_field = 'reg_logic_label';
  ref_select_sql = 'select r1.reg_unicode as id, r1.reg_code||'' ''||r1.reg_name as name, iif(r1.reg_status=3, ''Офис'', ''Регион'')||'' ''||r2.reg_name as memo, iif(r1.reg_date_close is null, 1, 0) as is_active, coalesce(r1.reg_logic_label, '''') as label_value from region r1, region r2 where r1.parent=r2.reg_unicode and r1.parent is not null';
  if (coalesce(key_id, 0) <> 0) then
    ref_select_sql = ref_select_sql || ' and r1.reg_unicode=' || key_id;
  suspend;

  ref_name = 'BlankName';
  ref_caption = 'Бланки';
  ref_table_name = 'BlankName';
  ref_label_field = '';
  ref_select_sql = 'select bln_id as id, bln_name as name, bln_memo as memo, bln_is_active as is_active, ''Поле-метка не используется'' as label_value from BlankName where bln_kind=11';
  if (coalesce(key_id, 0) <> 0) then
    ref_select_sql = ref_select_sql || ' and bln_id=' || key_id;
  suspend;
end
