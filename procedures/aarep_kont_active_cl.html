CREATE PROCEDURE AAREP_KONT_ACTIVE_CL(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_REGION_ID D_INTEGER)
RETURNS (
    K_KONTRACT_NO STR_20,
    K_DISB_DATE D_DATE,
    KS_DATE_BEGIN D_DATE,
    K_FROM_PT_ID STR_20,
    KAL_NEW_KF_NAME STR_100,
    KC_CIRCLE D_INTEGER,
    K_STAGE D_INTEGER,
    K_PERIOD DECIMAL(18,2),
    K_KOLICH_CLIENT_APPROVED D_INTEGER,
    K_PERCENT_REAL PERC_TYPE,
    K_AMOUNT_APPROVED CURR_TYPE,
    LO_NAME_FIO STR_100,
    PR_NAME STR_100,
    REG_NAME STR_100,
    CL_NAME_FIO STR_100,
    CL_IS_ACTIVE STR_20,
    POINT D_INTEGER)
AS
declare variable c_id id_type;
begin
  for select k_disb_date, k_kontract_no, kf_officer_name, kf_name, kf_region_name, k_stage, k_period,
             k_percent_real, k_kolich_client_approved, k_amount_approved, k_from_pt_id,
             cl_cashier_id, cl_cashier_name, k_circle, rep_date_closed, pr_name
      from aarep_kontract_list(:rep_region_id, :rep_date_begin, :rep_date_end, null, 810, 4)
      into k_disb_date, k_kontract_no, lo_name_fio, kal_new_kf_name, reg_name, k_stage, k_period,
           k_percent_real, k_kolich_client_approved, k_amount_approved, k_from_pt_id,
           c_id, cl_name_fio, kc_circle, ks_date_begin, pr_name
  do
  begin
    point = f_point(:c_id, :rep_date_end);
    cl_is_active = '';
    if (exists(select kc_from_k_id
               from knclient
               inner join kontstate on kc_from_k_id = ks_from_k_id
               where kc_from_c_id = :c_id and
                     ks_ol_is_delete = 0 and
                     ks_state = 1 and
                     :rep_date_end between ks_date_begin and ks_date_end)) then
      cl_is_active = 'Активный';
    else
      cl_is_active = 'Закрытый';
    suspend;
  end
end
