CREATE PROCEDURE FM_CONSULT_PORTAL_KG(
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
begin
  -- процедура отображает ссылку на сайт консультационного портала для ВИПСов и МО (RFCKG-3358)

  res_text = (select res_style
              from fm_html_styles) || '<body>';

  res_text = blob_append(res_text, '<a href="https://consult.mbulak.kg/?uid=' || :un_id || '">Консультационная платформа для клиентов - Демо</a><br></br>');
  res_text = blob_append(res_text, 'Консультационная платформа М Булак – обучающая платформа для сотрудников М Булак.
                                    Мы хотим сделать консультации по микрокредитованию и финансовой грамотности более доступными.
                                    Чтобы наши клиенты развивались и становились успешнее и счастливее!');

  res_text = blob_append(res_text, '</body>');

  suspend;
end
