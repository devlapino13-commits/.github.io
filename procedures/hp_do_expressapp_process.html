CREATE PROCEDURE HP_DO_EXPRESSAPP_PROCESS(
    EXA_ID D_INTEGER NOT NULL,
    DEBUG_MODE D_BOOL = 0)
AS
declare variable UN_ID D_ID;
declare variable CURR_EXC_ID D_INTEGER;
declare variable WAIT_STATUS D_SMALLINT;
declare variable EXC_ID D_INTEGER;
declare variable EXC_OBJ_KIND D_INTEGER;
declare variable EXC_ACTION_KIND D_STR_CODE;
declare variable EXC_ACTION_CODE D_STR_CODE;
declare variable MA_ID D_ID;
declare variable K_ID D_ID;
declare variable OBJ_ID D_ID;
declare variable EDL_STATUS D_SMALLINT;
declare variable EDL_ID D_ID;
declare variable CURR_WAIT_STATUS D_SMALLINT;
declare variable OBJ_CODE D_STR_CODE;
declare variable FINAL_STATUS D_SMALLINT;
declare variable EXA_LO_ID D_ID;
declare variable EXA_FROM_C_ID D_ID;
declare variable K_DISB_DATE D_DATE;
declare variable K_STATUS D_INTEGER;
declare variable RES_TXT D_STR_255;
declare variable EDS_STATUS D_SMALLINT;
declare variable DISB_DATE D_DATE;
declare variable DB_ID D_ID;
declare variable EXA_FROM_REG_ID D_INTEGER;
declare variable EXA_FROM_PR_ID D_INTEGER;
begin
  -- Процедура выполняет задачи на каждом этапе и переходит на следующую задачу.
  -- Работает пока не дошел до конца или не перешел в режим ожидания решения специалиста или клиента
  db_id = f_get_db_id();
  un_id = 1000000000000;

  wait_status = 0;
  -- Возможные значения wait_status:
  --  0 - В текущем этапе все нормально, можно переходить на следующий этап
  --  1 - В ожидании решения от специалистов
  --  2 - В ожидании ответа от клиента, например подписание электронного документа.
  -- -1 - Появилась ошибка в процессе

  while (wait_status = 0) do
  -- Пока этап завершается нормально, крутим цикл и переходим на следующие этапы
  begin
    select exa_from_exc_id, exa_final_status, exa_wait_status, exa_from_ma_id, exa_from_k_id, exa_from_edl_id,
           exc_obj_kind, exc_action_kind, exc_action_code, exa_lo_id, exa_from_c_id, exa_from_reg_id, exa_from_pr_id
    from expressapp, expresscfg
    where exa_id = :exa_id and
          exa_from_exc_id = exc_id and
          exa_final_status in (0, 1)
    into curr_exc_id, final_status, curr_wait_status, ma_id, k_id, edl_id, exc_obj_kind, exc_action_kind,
         exc_action_code, exa_lo_id, exa_from_c_id, exa_from_reg_id, exa_from_pr_id;

    if (row_count = 0 or -- Если exa_final_status not in (0, 1) завершаем работу
        curr_exc_id = 0) then -- Если экспресс-заявке не дан старт, завершаем работу
      exit;

    wait_status = 0; -- Предположим что все нрмально

    if (exists(select *
               from kontstate
               where ks_from_k_id = :k_id and
                     ks_ol_is_delete = 0 and
                     ks_state = 1)) then
    begin
      final_status = 2; -- если контракты был выдан то завершаем процесс

      update expressapp
      set exa_final_status = :final_status,
          exa_wait_status = :wait_status,
          exa_try_count = exa_try_count + 1
      where exa_id = :exa_id;

      wait_status = 0;
      if (db_id in (39801, 41701)) then
      begin
        select res_status from hp_do_expressapp_action_finish(:exa_id)
        into wait_status;
      end 

      exit;
    end

    obj_code = f_objkind(exc_obj_kind, 'code');

    obj_id = null;

    if (obj_code = 'expressapp') then
      obj_id = exa_id;

    if (obj_code = 'miniapp') then
      obj_id = ma_id;

    if (obj_code = 'kontract') then
      obj_id = k_id;

    if (obj_code = 'clients') then
      obj_id = exa_from_c_id;

    if (obj_id is null and
        exc_action_kind in ('edoc', 'check')) then
      exception e_default 'Unknown obj_id';

    if (obj_code = 'kontract') then
    begin
      select k_disb_date, k_status
      from kontract
      where k_id = :obj_id
      into k_disb_date, k_status;

      if (k_disb_date < current_date) then
      begin
        if (k_status = 1) then
        begin
          update kontract
          set k_status = 0,
              k_user_id = :un_id
          where k_id = :k_id;
        end

        res_txt = '';

        select res_txt
        from fm_change_disb_date(:un_id, :obj_code, :obj_id, current_date)
        into res_txt;

        if (res_txt = '') then
        begin
          update expressapp
          set exa_from_exc_id = 100,
              exa_disb_date = current_date
          where exa_id = :exa_id;

          continue;
        end
      end
    end

    if (exc_action_kind = 'action') then
    begin
      -- Нужно выполнить некие действия, например создать Miniapp и т.д.
      -- Всевозможные действия должны быть автоматизированы внутри нижеуказанной процедуры
      select res_status
      from hp_do_expressapp_action(:exa_id, :exc_action_code)
      into :wait_status;

      if (exists(select * from product
      where pr_id = :exa_from_pr_id
      and pr_code containing '.TMS.')
      and :exc_action_code in ('consult_send', 'loan_send')) then
        wait_status = 0;

      if (wait_status = -1) then
        final_status = -1;

      when any do
      begin
        wait_status = -1;
        if (debug_mode = 1) then
          exception;
      end
    end

    if (exc_action_kind = 'check') then
    begin
      -- Запуск CheckLoanCondition
      -- Если неходим ошибку, отправляем Push специалисту и выходим из цикла (wait_status <> 0)
      select res_is_error
      from hp_check_loan_konditions_update(:exc_obj_kind, :obj_id, :exc_action_code, 0, :un_id)
      into :wait_status;

      if (exists(select * from product  -- если контракт утвержден то переходим на следюущий этап
      where pr_id = :exa_from_pr_id and
      pr_code containing '.tms.') and
      :exc_action_code in ('loan_approve') and
      exists(select * from kontract
      where k_id = :k_id and
      k_status = 1)) then
        wait_status = 0;

      if (wait_status = 1 and
          curr_wait_status <> 1) then
      begin
        -- Отправляем Push специалисту
        execute procedure hp_do_expressapp_push_loan(:exa_lo_id, :un_id, :exc_obj_kind, :obj_id, :obj_code,
            :exa_from_c_id);
      end

      when any do
      begin
        wait_status = -1;
        if (debug_mode = 1) then
          exception;
      end
    end

    if (exc_action_kind = 'edoc') then
    begin
      -- Проверка электронного документа
      -- Если не создан, то создаем
      -- Если ранее создан, проверяем подписан ли документ
      -- Все виды электронных документов создаются внутри нижеуказанной процедуры

      if (edl_id is null) then
      begin
        -- Создаем новый электронный документ
        select wait_status
        from hp_do_expressapp_edoc(:exa_id, :exc_action_code, :exc_obj_kind, :obj_id, :un_id)
        into :wait_status;
      end
      else
      begin
        edl_status = 0;

        -- В тестовой базе всегда подписываем eDoc автоматически
        execute procedure hp_edoc_sign_draft(:edl_id);

        select edl_status
        from edoc_list
        where edl_id = :edl_id
        into edl_status;
        
        if (:edl_status = 1 and :exc_action_code = 'loan.express' and :db_id in (39801, 41701)) then
        begin
          disb_date = null;
          select eds_status_datetime
          from edoc_sign eds
          where eds_from_edl_id = :edl_id
          and eds_status = 1
          order by eds_status desc
          rows 1
          into disb_date;
          disb_date = coalesce(disb_date,current_date);
          disb_date = disb_date + 2;

          if (extract(weekday from cast(:disb_date as date)) = 6) then disb_date = disb_date + 2;
          if (extract(weekday from cast(:disb_date as date)) = 0) then disb_date = disb_date + 1;

          update expressapp exa
          set exa.exa_disb_date = :disb_date
          where exa.exa_id = :exa_id;

        end

        if (edl_status = 0) then -- Документ не подписан, ждем
          wait_status = 2;

        if (edl_status = -1) then -- Документ аннулирован, что делать?
        begin
          wait_status = -1;

          -- Если документ аннулирован и это "Предложение по акции", то закрываем экспресс-заявку
          if (exc_action_code = 'loan.express') then
            final_status = -2;
        end
      end
    end

    if (wait_status = 0) then
    begin
      -- Если дошли до этого момента, значит этап завершился нормально
      -- Устанавливаем следующий этап и оставляем wait_status = 0 чтобы не выйти из цикла
      exc_id = null;
      select exc_id
      from expresscfg
      where exc_id > :curr_exc_id and
            iif(:db_id in (39801, 41701), exc_action_code <> 'mc_purchase.agreement_autosign', true)  --Чтобы не сломать логику с object_kind нужно пропустить данный этап для обычных экспрессов
      order by exc_id
      rows 1
      into :exc_id;

      if (row_count = 0) then -- Если запрос выше ничего не вернул, значит дошли до конца
      begin
        -- Устанавливаем финальный статус как выданный займ
        update expressapp
        set exa_final_status = 2
        where exa_id = :exa_id and
              exa_final_status <> 2;
      end
      else
      begin
        -- устанавливаем следующий этап, цикл отработает повторно
        update expressapp
        set exa_from_exc_id = :exc_id
        where exa_id = :exa_id;
      end
    end
    else
    begin
      -- если wait_status <> 0, фиксируем результат, завершается работа процедуры
      -- Обработка продолжится когда придет очередь к этой заявке следующий раз
      update expressapp
      set exa_final_status = :final_status,
          exa_wait_status = :wait_status,
          exa_try_count = exa_try_count + 1
      where exa_id = :exa_id;
    end
  end -- конец цикла, крутимся пока final_status = 0 или не долшли до конца
end
