CREATE PROCEDURE HP_CHECK_EXISTS_PERSTEMP(
    C_ID D_ID,
    FIELD_GROUP D_STR_CODE)
RETURNS (
    RES_ERROR_TEXT D_STR_255,
    RES_IS_EXISTS D_BOOL)
AS
begin
  -- Процедура проверяет были ли изменения перс. данных по клиенту, используется при const_tmp.use_perstemp = 1
  -- входные параметры:  c_id - id клиента, field_group - группа перс. данных, по которым проверяются изменения
  -- выходные параметры: res_is_exists = 1, если в буфере изменений перс. данных есть поля из заданной группы (field_group), иначе res_is_exists = 0
  --                     res_error_text - инфо о возникшей ошибке при проверке изменений

  res_error_text = '';
  res_is_exists = 0;

  if (f_tmp_const('use_perstemp') <> '1') then
    res_error_text = 'Ошибка: модуль буфера изменений перс. данных отключен, проверка изменений невозможна';

  if (coalesce(field_group, '') = '') then
    res_error_text = 'Ошибка: не указана группа перс. данных';

  if (coalesce(c_id, 0) = 0) then
    res_error_text = 'Ошибка: не указан клиент';

  if (res_error_text = '' and field_group = 'name_date_place_birth') then -- RFCRU-2205
    if (exists(select *
               from hp_perstemp_diff(:c_id)
               where info_name in ('ФИО', 'ФИО на латинском', 'ФИО на английском', 'Дата рождения', 'Страна рождения', 'Место рождения'))) then
      res_is_exists = 1;

  suspend;
end
