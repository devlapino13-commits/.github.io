CREATE PROCEDURE HP_PRINT_AFM_RESULT(
    C_ID D_ID,
    MA_ID D_ID)
RETURNS (
    RES_TEXT D_BLOB_TEXT)
AS
declare variable AC_ID D_SMALLINT;
declare variable AC_STATUS D_SMALLINT;
declare variable AC_NAME D_STR_255;
declare variable SQL_TEXT D_STR_LONG;
begin
    res_text = '';
    ac_id = 0;
    ac_status = 0;
    ac_name = '';

    while (ac_id < 15) do
    begin
        ac_id  = ac_id + 1;
        if (ac_id in (2, 3)) then
            continue;

        sql_text = f_format(
            'select first 1 ac_category_%1
             from afm_check
             where ac_from_c_id = %2
               and ac_from_ma_id = %3
               and ac_is_actual = 1',
            :ac_id, :c_id, :ma_id
        );
        execute statement :sql_text
        into :ac_status;

        if (ac_status not in (1, 2, 3, 4, 5)) then
            continue;

        select first 1 ac_name
        from afm_categories
        where ac_id = :ac_id
        into :ac_name;

        if (ac_name is null) then
            continue;

        if (res_text <> '') then
            res_text = res_text || ascii_char(13) || ascii_char(10);
        if (ac_status = 1) then
            res_text = res_text || 'Не состоит в списке ' || :ac_name;
        else
            res_text = res_text || 'Состоит в списке ' || :ac_name;
    end
    suspend;
end
