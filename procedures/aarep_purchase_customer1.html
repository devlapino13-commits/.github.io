CREATE PROCEDURE AAREP_PURCHASE_CUSTOMER1(
    OL_ID ID_TYPE)
RETURNS (
    CUSTOMER_NAME D_STR_255,
    CUSTOMER_INN D_STR_20,
    CUSTOMER_ADR1 D_STR_2048,
    CUSTOMER_ADR2 D_STR_255,
    CUSTOMER_FIO D_STR_100,
    CUSTOMER_JOBNAME D_STR_100)
AS
declare variable pl_id id_type;
declare variable pl_name d_str_100;
declare variable adl_id id_type;
begin
  select 
    reg_kont_company,
    reg_inn,
    reg_from_pl_id,
    reg_address_legal                          
  from region 
  where reg_code = '001'
  into
    customer_name,
    customer_inn,
    pl_id,
    adl_id;
    
  select first 1
    c_name_fio,
    jl_name        
  from opermat
    inner join clients
      on c_id = om_matotvet_id
    left join hr_pstatus
      on ps_from_c_id = c_id
        and ps_status = 1 and ps_is_working = 1
        and current_date between ps_date_begin and ps_date_end
    left join joblist
      on jl_id = ps_from_jl_id
  where om_from_ol_id = :ol_id
  into        
    customer_fio,
    customer_jobname;
             
  customer_adr1 = '';
  while (pl_id <> 417000000001) do
  begin
    select
      pl_name,
      parent
    from places
    where pl_id = :pl_id
    into
      pl_name,
      pl_id;
    if (row_count = 1) then customer_adr1 = pl_name || iif(customer_adr1='','',', ') || customer_adr1;
    else leave;
  end
     
  customer_adr2 = '';
  select
    iif(pl_id is null, '', 'ул.' || pl_name) ||
    iif(adl_depart_no in ('0', ''), '', ' д.' || adl_depart_no ||
      iif(adl_depart_pref='', '', '/' || adl_depart_pref)) ||
    iif(adl_flat_no in ('0', ''), '', ' кв.' || adl_flat_no)
  from addrlist
    left join places
      on adl_street_id = pl_id
  where adl_id = :adl_id
  into
    customer_adr2;
    
  suspend;
end
