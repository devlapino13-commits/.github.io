CREATE PROCEDURE AAREP_PLAN_REPAY_SSVZ_NEW(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE,
    REP_C_ID ID_TYPE = 0)
RETURNS (
    KF_OFFICER_NAME VARCHAR(100),
    KF_NAME VARCHAR(20),
    AA_CATEGORY_ID SMALLINT,
    REP_DATE DATE,
    WEEK_NAME VARCHAR(3),
    IS_CLOSED VARCHAR(1),
    KOLICH_CLI_CLOSE SMALLINT,
    OK_SH_OS CURR_TYPE,
    OK_SH_PERC CURR_TYPE,
    OK_SH_TOTAL CURR_TYPE,
    PLAN_BALANCE CURR_TYPE,
    DELINQ_ZAD_OS CURR_TYPE,
    DELINQ_ZAD_PERC CURR_TYPE,
    DELINQ_ZAD_FINE CURR_TYPE,
    DELINQ_ZAD_TOTAL CURR_TYPE,
    DELINQ_BALANCE CURR_TYPE,
    DELINQ_DAY INTEGER,
    C_PLASTCARD_NO VARCHAR(30),
    CASHIER_NAME VARCHAR(100),
    PR_COMMON_NAME VARCHAR(100))
AS
declare variable AA_KC_STAGE varchar(30);
declare variable OKS_KC_ID ID_TYPE;
declare variable OKS_FROM_K_ID ID_TYPE;
declare variable IS_DELINQ smallint;
declare variable OKS_KC_ID_PRIOR ID_TYPE;
declare variable CASHIER_ID ID_TYPE;
declare variable K_KREDLINE_END date;
declare variable K_KOLICH_CLIENT_APPROVED smallint;
declare variable OKS_BALANCE CURR_TYPE;
begin
  if (rep_c_id is null or rep_c_id = -1) then rep_c_id = 0;

  oks_kc_id_prior = 0;
  for select rep_date, oks_kc_id, oks_from_k_id, ok_sh_os, ok_sh_perc, delinq_zad_os, delinq_zad_perc, delinq_zad_fine,
             delinq_zad_total, delinq_day, is_delinq, aa_category_id
      from aarep_plan_repay_ssvz_temp(:rep_region_id, :rep_date_end, :rep_c_id)
      order by oks_kc_id, is_delinq desc, delinq_day, rep_date
      into :rep_date, :oks_kc_id, :oks_from_k_id, :ok_sh_os, :ok_sh_perc, :delinq_zad_os, :delinq_zad_perc,
           :delinq_zad_fine, :delinq_zad_total, :delinq_day, :is_delinq, :aa_category_id
  do
  begin
    ok_sh_total = :ok_sh_os + :ok_sh_perc;

    if (oks_kc_id_prior <> oks_kc_id) then
    begin
      oks_kc_id_prior = oks_kc_id;
      kf_officer_name = (select c_name_fio
                         from clients
                         where c_id = :oks_kc_id);
--     Системная должность
      aa_kc_stage = null;
      select rd_name
      from hr_pstatus, joblist, jobkind, refdata
      where ps_from_c_id = :oks_kc_id and
            ps_is_working = 1 and
            ps_status = 1 and
            :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_id = rd_id
      into aa_kc_stage;
      if (aa_kc_stage is null) then
        aa_kc_stage = 'Не определен';
      kf_officer_name = kf_officer_name || ' - ' || aa_kc_stage;
    end

    cashier_id = null;
    kf_name = null;
    k_kolich_client_approved = null;
    pr_common_name = null;
    select kal_new_kf_name, k_clients_cashier_id, k_kredline_end, k_kolich_client_approved, pr_common_name
    from knaktlist, kontract, product
    where knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_from_kf_id = k_from_kf_id and
          k_id = :oks_from_k_id and
          k_from_pr_id = pr_id
    into kf_name, cashier_id, k_kredline_end, k_kolich_client_approved, pr_common_name;
    if (cashier_id is null) then
      cashier_id = 0;
    if (kf_name is null) then
      kf_name = '';
    if (k_kolich_client_approved is null) then
      k_kolich_client_approved = 0;
    if (pr_common_name is null) then
      pr_common_name = '';

    is_closed = '';
    kolich_cli_close = null;
    if (rep_date = k_kredline_end) then
    begin
      is_closed = 'a';
      kolich_cli_close = k_kolich_client_approved;
    end

    cashier_name = null;
    c_plastcard_no = null;
    select c_name_fio, '+' ||(select cont_value
                              from hp_contdata_get_cell_valid(0, :cashier_id))
    from clients
    where c_id = :cashier_id
    into cashier_name, c_plastcard_no;

    oks_balance = null;
    select sum(oks_balance)
    from oksummary
    where :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          oks_from_k_id = :oks_from_k_id and
          oks_close_state = 0
    into oks_balance;
    if (oks_balance is null) then
      oks_balance = 0;

    delinq_balance = 0;
    plan_balance = 0;
    if (is_delinq = 1) then
      delinq_balance = oks_balance;
    else
      plan_balance = oks_balance;

    week_name = case extract(weekday from :rep_date)
                  when 0 then 'Вс.'
                  when 1 then 'Пн.'
                  when 2 then 'Вт.'
                  when 3 then 'Ср.'
                  when 4 then 'Чт.'
                  when 5 then 'Пт.'
                  when 6 then 'Сб.'
                end;

    suspend;
  end

end
