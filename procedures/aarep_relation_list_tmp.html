        if (position(rel_c_id, rel_list) <= 0) then
        begin
-- rel_list = rel_list || ',' || :rel_c_id;
          if (char_length(rel_list || ',' || :rel_c_id) <= 2048) --30/05/2019
          then
            rel_list = rel_list || ',' || :rel_c_id;
          rel_fio = '';
          rel_cell_no = '***';
          rel_address = '***';

          select c_name_fio
          from clients
          where c_id = :rel_c_id
          into :rel_fio;

          found_count = found_count + 1;
          suspend;
        end
      end

      rel_name = 'Живет в одном доме с активным клиентом';
      for select kc_from_c_id
          from knclient, kontstate, addrclient, addrlist
          where kc_amount_approved > 0 and
                kc_from_c_id <> :c_id and
                ks_from_k_id = kc_from_k_id and
                ks_state = 1 and
                ks_ol_is_delete = 0 and
                ks_date_end >= '31.12.2999' and
                adc_from_c_id = kc_from_c_id and
                adc_address_type = 2 and
                adc_from_adl_id = adl_id and
                adl_street_id = :adl_street_id and
                adl_depart_no = :adl_dep_no
          into :rel_c_id
      do
      begin
        if (position(rel_c_id, rel_list) <= 0) then
        begin
          if (char_length(rel_list || ',' || :rel_c_id) <= 2048) --30/05/2019
          then
            rel_list = rel_list || ',' || :rel_c_id;
          rel_fio = '';
          rel_cell_no = '';

          select c_name_fio,
                 (select cont_value
                  from hp_contdata_get_cell_valid(0, :rel_c_id)), adl_address_str
          from clients, addrclient, addrlist
          where c_id = :rel_c_id and
                adc_from_adl_id = adl_id and
                adc_from_c_id = :rel_c_id and
                adc_address_type = 2
          into :rel_fio, :rel_cell_no, rel_address;

          if (rel_cell_no is null) then
            rel_cell_no = '';
          if (rel_address is null) then
            rel_address = '';
          found_count = found_count + 1;
          suspend;
        end
      end
    end
  end
