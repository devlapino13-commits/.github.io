CREATE PROCEDURE FSROBOT_PUSH_AGR_EXCHANGE
AS
declare variable c_id id_type;
declare variable mcl_id id_type;
declare variable combo_active d_bool;
declare variable score d_smallint;
declare variable cl_name str_100;
declare variable push_text d_str_long;
declare variable rd_id d_integer;
begin
  rd_id = f_get_rd_id('AgreementKinds.loyal_agreement');

  for select ag_object_id
      from agreements ag
      where ag.ag_object_kind = 0 and
            ag.ag_from_rd_id = :rd_id and
            coalesce(ag_date_end, '31.12.9999') >= current_date and
            ag_status = 1
      into c_id
  do
  begin
    mcl_id = null;
    combo_active = null;
    cl_name = '';

    select fl_name2
    from clients
    inner join fiolist on fl_id = c_from_fl_id
    where c_id = :c_id
    into cl_name;

    select mcl_id
    from mc_login
    where mcl_from_c_id = :c_id and
          mcl_is_delete = 0 and
          mcl_google_id <> ''
    order by mcl_last_login_datetime desc
    rows 1
    into mcl_id;

    if (mcl_id is not null) then
    begin
      combo_active = 0;
      score = 0;
      push_text = '';

      select combo_active, score
      from mc_get_activity(:mcl_id)
      into combo_active, score;
      if (combo_active = 1) then
      begin
        push_text = 'Уважаемый, ' || cl_name || ascii_char(10) || 'Поздравляем! Вы накопили ' || :score || ' бонусов.' || ascii_char(10) || 'Вы можете обменять их на деньги или продолжить копить дальше в разделе Партнерская программа в мобильном приложении.';
        execute procedure mc_send_push('mc_login', :mcl_id, 'У Вас накопились бонусы!', :push_text,
            7);
      end
    end
  end
end
