CREATE PROCEDURE FM_GET_PREPAYMENT_INFO(
    K_ID ID_TYPE)
RETURNS (
    CL_ACC D_INTEGER,
    CL_DATE D_DATE,
    AMOUNT D_CURRENCY,
    QR_COMMENT D_STR_255,
    CL_DATE_STR D_STR_20)
AS
declare variable C_ID ID_TYPE;
declare variable K_KONTRACT_NO D_STR_20;
declare variable K_DISB_DATE D_DATE;
begin

  for select ma.ma_prepayment_summa,
             kc.kc_from_c_id,
             kn.k_kontract_no,
             kn.k_disb_date
      from miniapp ma
      inner join kontract kn on kn.k_from_ma_app_id = ma.ma_id
      inner join knclient kc on kc.kc_from_k_id = kn.k_id
      where kn.k_id = :k_id and
            kc.kc_amount_approved > 0 and
            ma.ma_is_active = 1 and
            ma.ma_date_end >= '31.12.9999'
  into :amount,
       :c_id,
       :k_kontract_no,
       :k_disb_date
  do
  begin
    select uniq_acc,
           uniq_birth_date
    from hp_show_clientuniq_acc_no(:c_id)
    into :cl_acc,
         :cl_date;

    cl_date_str = extract(year from cl_date) || right(0 || extract(month from cl_date), 2) || right(0 || extract(day from cl_date), 2);

    qr_comment = 'Договор № ' || :k_kontract_no || ' от ' || f_datetostr(:k_disb_date) ||
                 '. Сумма частичной оплаты ' || f_numtostr(:amount, 2) || ' сом.';
    suspend;
  end
end
