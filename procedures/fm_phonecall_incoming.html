CREATE PROCEDURE FM_PHONECALL_INCOMING(
    SRC_PHONE_NO D_STR_CODE,
    PRV_PHONE_NO D_STR_CODE)
RETURNS (
    PNC_ID D_ID,
    DST_PHONE_NO D_STR_CODE,
    PNC_DIRECTION D_SMALLINT)
AS
declare variable pnc_src_phone_id d_integer;
declare variable pnc_prv_phone_id d_integer;
begin
  -- Процедура вызывается из Asterisk, в случае входящего звонка
  -- Если за последний час был созвон с номером src_phone_no, создаем запись
  -- в phonecalls, и возвращаем номер, с которым был созвон
  -- Используется также для исходящих звонков из МП
  -- В этом случае должна вернуться ранее созданная запись, а не новая

  -- pnc_id = -1 - Когда звонит сотрудник, но звонок не зарегистрирован (включаем инструкцию)
  -- pnc_id = -2 - Когда не смогли определить, с кем соединять

  pnc_src_phone_id = f_cont_id(src_phone_no, 'null');
  pnc_prv_phone_id = f_cont_id(prv_phone_no, 'null');
  dst_phone_no = '0';
  pnc_direction = 0;
  if (pnc_src_phone_id is null or pnc_prv_phone_id is null) then
  begin
    pnc_id = -2;
    suspend;
    exit;
  end

  -- Тестирование
  --  79032063710(Комилжон)
  --  74952258089(Провайде)
  --  79036824207(Виктор)
  if (src_phone_no = '79032063710' and
      prv_phone_no = '74952258057') then
  begin
    pnc_id = -999;
    dst_phone_no =  '79036824207';
    pnc_direction = 1;
    suspend;
    exit;
  end

  -- Ищем номер во входящих и исходящих звонках
  for select pnc_id, pnc_src_phone_id, pnc_dst_phone_id, pnc_prv_phone_id,
    pnc_src_c_id, pnc_dst_c_id, pnc_direction, pnc_object_kind, pnc_object_id,
    pnc_is_delinq, pnc_call_status_id, pnc_call_result_id
  from phonecalls
  where datediff(minute, current_timestamp, pnc_start) < 1440 -- 24 часа
     and :pnc_src_phone_id in (pnc_src_phone_id, pnc_dst_phone_id)
  order by pnc_start desc -- Приоритет у исходящих звонков
  rows 1
  as cursor cur
  do
  begin
    pnc_direction = cur.pnc_direction;

    -- Если найден исходящий звонок и провайдер совпал, и он в начальном статусе
    -- значит нужно вернуть его, не вставляя новую запись
    if (cur.pnc_prv_phone_id = pnc_prv_phone_id and
        cur.pnc_call_status_id = 0 and
        cur.pnc_call_result_id = 0 and
        pnc_direction = 1 and
        cur.pnc_src_phone_id = pnc_src_phone_id) then
    begin
      pnc_id = cur.pnc_id;
      dst_phone_no = f_cont_code(cur.pnc_dst_phone_id);
      suspend;
      exit;
    end

    if (pnc_src_phone_id = cur.pnc_src_phone_id and f_is_employee(cur.pnc_src_c_id, current_date) = 1) then
    begin
      -- Повторный звонок от сотрудника, без инициализации в МП (FM_PHONECALL_INIT)
      -- Т.е. просто повторно вызывает номер провайдера
      pnc_id = -1; -- Такой звонок запрещен
    end
    else if (cur.pnc_src_phone_id = pnc_src_phone_id and
             cur.pnc_prv_phone_id = pnc_prv_phone_id) then
    begin
      -- Повторный звонок от клиента
      insert into phonecalls (
        pnc_src_phone_id, pnc_dst_phone_id, pnc_src_c_id, pnc_dst_c_id,
        pnc_direction, pnc_object_kind, pnc_object_id, pnc_is_delinq)
      values (cur.pnc_src_phone_id, cur.pnc_dst_phone_id, cur.pnc_src_c_id,
        cur.pnc_dst_c_id, cur.pnc_direction, cur.pnc_object_kind, cur.pnc_object_id,
        cur.pnc_is_delinq)
      returning pnc_id
      into pnc_id;
      dst_phone_no =  f_cont_code(cur.pnc_dst_phone_id);
    end
    else if (cur.pnc_prv_phone_id = pnc_prv_phone_id) then
    begin
      -- Обратный звонок от клиента
      pnc_direction = -1 * cur.pnc_direction;
      insert into phonecalls (
        pnc_src_phone_id, pnc_dst_phone_id, pnc_src_c_id, pnc_dst_c_id,
        pnc_direction, pnc_object_kind, pnc_object_id, pnc_is_delinq)
      values (cur.pnc_dst_phone_id, cur.pnc_src_phone_id, cur.pnc_dst_c_id,
        cur.pnc_src_c_id, :pnc_direction, cur.pnc_object_kind, cur.pnc_object_id,
        cur.pnc_is_delinq)
      returning pnc_id
      into pnc_id;
      dst_phone_no =  f_cont_code(cur.pnc_src_phone_id);
    end
  end

  if (pnc_id is null) then
  begin
    -- Перекидываем на КЦ, т.к. не нашли исходящий звонок
    -- Нужен признак OiVO
    pnc_id = -2;
    pnc_direction = 0;
  end
  suspend;
end
