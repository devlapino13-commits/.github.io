CREATE PROCEDURE HP_GET_MC_APP_VERSION(
    MCL_LOGIN D_STR_CODE,
    MCL_PROJECT D_STR_CODE,
    APPLICATION_VERSION D_STR_10,
    OPERATING_SYSTEM D_STR_10)
RETURNS (
    APP_VERSION_IN_DECIMAL DECIMAL(18,7),
    RES_CODE D_STR_CODE,
    ERR_TEXT D_STR_255)
AS
declare variable APP_VERSION D_STR_10;
declare variable TMP_APP_VER_IN_DECIMAL decimal(18,7);
declare variable IS_MANDATORY_UPDATE D_SMALLINT;
begin
    --RFCKG 2344
    res_code = '';
    err_text = '';

    app_version = f_get_const(:mcl_project||'_client_'||:operating_system ||'_ver');
    app_version_in_decimal = left(application_version, position('.', application_version) - 1) || '.' || replace(substring(application_version from position('.', application_version)), '.', '');
    tmp_app_ver_in_decimal = left(app_version, position('.', app_version) - 1) || '.' || replace(substring(app_version from position('.', app_version)), '.', '');

    -- RFCKG-2670
    if(tmp_app_ver_in_decimal > app_version_in_decimal) then
    begin
        is_mandatory_update = cast(f_get_const(:mcl_project||'_is_mandatory_update') as d_smallint);
        if(is_mandatory_update = 1) then
        begin
            -- RFCRU-2599
            res_code = 'version_error';
            if (f_get_db_id() = 81001 and operating_system = 'android') then
                err_text = 'Обновите приложение через наш сайт';
            else
                err_text = 'Приложение устарело. Необходимо установить актуальную версию приложения';
        end
        else if(exists
        (
            select 1
            from mc_login mcl
            where mcl.mcl_login = :mcl_login
                and mcl.mcl_is_delete = 0
                and datediff
                (
                    day,
                    cast(mcl.mcl_last_login_datetime as date),
                    coalesce(cast(mcl.mcl_last_update_notify as date), current_date - 1)
                ) < 0
        )) then
        begin
            -- RFCRU-2599
            res_code = 'version_warning';
            if (f_get_db_id() = 81001 and operating_system = 'android') then
                err_text = 'Обновите приложение через наш сайт';
            else
                err_text = 'Приложение устарело. Необходимо установить актуальную версию приложения';

            update mc_login mcl
            set mcl.mcl_last_update_notify = current_timestamp,
                mcl.mcl_last_login_datetime = current_timestamp
            where mcl.mcl_login = :mcl_login
                and mcl.mcl_is_delete = 0;

        end
    end

    suspend;
end
