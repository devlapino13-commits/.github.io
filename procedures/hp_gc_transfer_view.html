CREATE PROCEDURE HP_GC_TRANSFER_VIEW(
    IGA_ID D_INTEGER)
RETURNS (
    IGT_ID D_INTEGER,
    IGT_FROM_C_ID D_ID,
    C_NAME_FIO D_STR_100,
    IGT_FROM_AG_ID D_INTEGER,
    IGT_ACCOUNT D_STR_20,
    ACCOUNT D_STR_20,
    IGT_ACCOUNT_BANK_ID D_INTEGER,
    BANK_NAME D_STR_CODE,
    IGT_AMOUNT D_CURRENCY,
    IGT_STATUS D_SMALLINT,
    STATUS D_STR_CODE,
    IGT_DB_ERROR_ID D_INTEGER,
    IGT_ERROR_TEXT D_STR_255,
    IGT_DO_TRANSFER D_BOOL,
    INFO D_SMALLINT)
AS
begin
  for
    select
      igt_id,
      igt_from_c_id,
      c_name_fio,
      igt_from_ag_id,
      igt_account,
      left(igt_account, 4) || lpad(right(igt_account,2),6,'*'),
      igt_account_bank_id,
      case f_get_rd_code(igt_account_bank_id)
        when 'ElSomBank.KICB' then 'KICB'
        when 'ElSomBank.Finca' then 'Finca'
      else '' end,
      igt_amount,
      igt_status,
      case igt_status
        when -2 then 'ошибка данных БД'
        when -1 then 'ошибка перевода Elsom'
        when  0 then 'на перевод не отправлялась'
        when  1 then 'в очереди на перевод'
        when  2 then 'перевод осуществлен'
        when  3 then 'бух.проводка построена'
      end,
      igt_db_error_id,
      igt_error_text,
      igt_do_transfer,
      case
        when igt_status < 0 then -1
        when igt_status in (0,1,2) then 0
        when igt_status = 3 then 1
      end
    from ipc_gc_transfer
      inner join clients
        on c_id = igt_from_c_id
      inner join agreements
        on ag_id = igt_from_ag_id
    where igt_from_iga_id = :iga_id
  into
    :igt_id,
    :igt_from_c_id,
    :c_name_fio,
    :igt_from_ag_id,
    :igt_account,
    :account,
    :igt_account_bank_id,
    :bank_name,
    :igt_amount,
    :igt_status,
    :status,
    :igt_db_error_id,
    :igt_error_text,
    :igt_do_transfer,
    :info
  do
  begin
    if (igt_status = -2 and igt_db_error_id > 0) then
      igt_error_text = (select rd_name from refdata where rd_id = :igt_db_error_id);
    suspend;
  end
end
