CREATE PROCEDURE AAREP_KONTRACT_ACTIVE_STATUS(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REGION_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    OKRUG_NAME VARCHAR(50),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    K_CIRCLE INTEGER,
    K_CIRCLE_INTERNAL INTEGER,
    K_PERIOD DECIMAL(15,2),
    K_PERCENT DECIMAL(15,2),
    K_PERCENT_CATEGORY DECIMAL(15,1),
    KF_OFFICER VARCHAR(100),
    KAL_NEW_KF_NAME VARCHAR(20),
    K_BONUS_TEXT VARCHAR(20),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_BONUS_TEXT_ON_DISB VARCHAR(20))
AS
declare variable k_id decimal(18,0);
declare variable c_curr_bonus_status integer;
declare variable c_last_bonus_status integer;
begin
-- Вытаскивает всех активных групповых кредитов
  for select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name,
             k_kontract_no, k_disb_date, k_circle, k_circle_internal, k_period, k_percent, k_percent_category,
             c_name_fio, kal_new_kf_name, k_id, k_kolich_client_approved, k_amount_approved
      from kontstate, kontract, knaktlist, clients, region r1, region_list(:rep_region_id, 1, 3, 1), region r2, region r3, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            :rep_date_end between ks_date_begin and ks_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            :rep_date_end between kal_date and kal_date_end and
            kal_new_reg_id = id_list and
            kal_new_kf_officer = c_id and
            id_list = r1.reg_unicode and
            r1.parent = r2.reg_unicode and
            r2.parent = r3.reg_unicode and
            k_from_pr_id = pr_id and
            pr_loan_type = 0
      into :region_name, :branch_name, :okrug_name, :k_kontract_no, :k_disb_date, :k_circle, :k_circle_internal,
           :k_period, :k_percent, :k_percent_category, :kf_officer, :kal_new_kf_name, :k_id, :k_kolich_client_approved, :k_amount_approved

  do
  begin
--  Проверка статуска контракта
    select c_last_bonus_status, c_curr_bonus_status, c_last_bonus_status_str, c_curr_bonus_status_str
    from hp_kontstate_loan_status(:k_id, 0, 0, :rep_date_end, '')
    into :c_last_bonus_status, :c_curr_bonus_status, :k_bonus_text_on_disb, k_bonus_text;

    if (:c_curr_bonus_status > 0) then
    begin
      suspend;
    end
  end
end
