CREATE PROCEDURE HP_KONTSTATE_CLIENT_STATUS(
    C_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    C_LAST_BONUS_STATUS INTEGER,
    C_CURR_BONUS_STATUS INTEGER,
    C_LAST_BONUS_STATUS_STR VARCHAR(100),
    C_CURR_BONUS_STATUS_STR VARCHAR(100))
AS
declare variable k_id decimal(18,0);
declare variable bonus_change integer;
declare variable bonus_change_min integer;
declare variable db_id integer;
declare variable latter_close_date date;
begin
  if (c_id = 0) then
  begin
    c_last_bonus_status = 0;
    c_curr_bonus_status = 0;
    suspend;
    exit;
  end

  db_id = f_get_db_id();
  latter_close_date = '31.12.9999';
  c_last_bonus_status = 0;
  c_curr_bonus_status = 0;
  for select ks_from_k_id, ks_bonus_change, ks_date_begin
      from kontstate, knclient
      where ks_from_k_id = kc_from_k_id and
            ks_ol_is_delete = 0 and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            ks_state = 2 and
            ks_date_begin <= :rep_date
      order by ks_date_begin
      into :k_id, :bonus_change, :latter_close_date
  do
  begin
--  Проверка контракта на параллельность
    if (exists(select 1
               from kontract, product
               where k_id = :k_id and
                     k_from_pr_id = pr_id and
                     pr_product_kind = 2)) then
    begin
--    Если параллельный то их на повышение статуса не включает
      if (bonus_change > 0) then
        bonus_change = 0;
    end
    /*  убран, данная функция работает в процедуре  HP_KONTSTATE_BONUS_CHANGE
--Так же проверим на досрочное закрытие кредита в течении 3х месяцев
    if (db_id in (39801, 41701) and bonus_change > 0
         and (exists(select 1 from kontstate
              where ks_ol_is_delete = 0
              and ks_state = 1
              and ks_from_k_id = :k_id
--              and datediff(month, ks_date_begin, :latter_close_date) < 4
              and dateadd(month, 3, ks_date_begin) > :latter_close_date
              ))) then
    begin
      bonus_change = 0;
    end  */

    c_last_bonus_status = c_last_bonus_status + :bonus_change;
    if (c_last_bonus_status < 0) then
      c_last_bonus_status = 0;
--  для РФ базы, если, отпустит статус клиента до брозны
    if (bonus_change = -998) then
      c_last_bonus_status = 1;

--  добавлено по приказу №23-09-15 от 30.09.2015г.
    if (rep_date >= cast('05.10.2015' as date) and
        db_id = 81001) then
    begin
      if (c_last_bonus_status > 5 and
          latter_close_date < cast('05.10.2015' as date)) then
      begin
        c_last_bonus_status = 5;
      end
    end
  end

  if (db_id in (39801, 41701)) then
  begin
    if (:rep_date < cast('01.08.2015' as date)) then
    begin
      if (latter_close_date < dateadd(month, -12, :rep_date) and
          not exists(select *
                     from kontstate, knclient
                     where ks_ol_is_delete = 0 and
                           ks_state = 1 and
                           :rep_date between ks_date_begin and ks_date_end and
                           ks_from_k_id = kc_from_k_id and
                           kc_amount_approved > 0 and
                           kc_from_c_id = :c_id)) then
      begin
        c_last_bonus_status = 0;
      end
    end
    else
    begin -- действует с 1го августа, добавил Улан
      if (latter_close_date < dateadd(month, -36, :rep_date)) then
      begin
        if (not exists(select *
                       from kontstate, knclient
                       where ks_ol_is_delete = 0 and
                             ks_state = 1 and
                             :rep_date between ks_date_begin and ks_date_end and
                             ks_from_k_id = kc_from_k_id and
                             kc_amount_approved > 0 and
                             kc_from_c_id = :c_id)) then
        begin
          c_last_bonus_status = 0;
        end
      end
    end
  end

  c_curr_bonus_status = c_last_bonus_status;
  bonus_change_min = 0;
  for select ks_from_k_id
      from kontstate, knclient, product, kontract
      where ks_from_k_id = kc_from_k_id and
            ks_ol_is_delete = 0 and
            kc_from_c_id = :c_id and
            kc_amount_approved > 0 and
            ks_state = 1 and
            ks_date_begin <= :rep_date and
            ks_date_end >= :rep_date and
            ks_from_k_id = k_id and
            k_from_pr_id = pr_id and
            pr_product_kind = 1 -- категории считает только по групповым гредитам
      order by ks_date_begin
      into :k_id
  do
  begin
    bonus_change_min = 1000;
    select bonus_change
    from hp_kontstate_bonus_change(:k_id, :rep_date, 1)
    into :bonus_change;
    if (bonus_change_min > :bonus_change) then
      bonus_change_min = :bonus_change;
  end
  if (bonus_change_min = 1000) then
    bonus_change_min = 0;
  c_curr_bonus_status = c_curr_bonus_status + :bonus_change_min;
  if (c_curr_bonus_status < 0) then
    c_curr_bonus_status = 0;
  if (bonus_change = -998) then
    c_last_bonus_status = 1;

  if (db_id = 81001) then
  begin
    if (rep_date < cast('05.10.2015' as date)) then
    begin
      if (c_last_bonus_status > 5) then
        c_last_bonus_status = 5;
      if (c_curr_bonus_status > 5) then
        c_curr_bonus_status = 5;
    end
    else
    begin
--      добавлено по приказу №23-09-15 от 30.09.2015г.
      /*
      с 5го октября 2015 по РФ статусы будут продолжаться до 10ти
      1 - Бронза
      2 - Серебро
      3 - Золото
      4 - Бриллиант
      5 - Бриллиант*
      6 - Бриллиант**
      7 - Бриллиант***
      8 - Бриллиант****
      9 - Бриллиант*****
      10 - Платина
      */
      if (c_last_bonus_status > 10) then
        c_last_bonus_status = 10;
      if (c_curr_bonus_status > 10) then
        c_curr_bonus_status = 10;
    end
  end
  else
  begin
    if (c_last_bonus_status > 4) then
      c_last_bonus_status = 4;
    if (c_curr_bonus_status > 4) then
      c_curr_bonus_status = 4;
  end

  c_last_bonus_status_str = '';
  c_curr_bonus_status_str = '';

  if (c_last_bonus_status = -1) then
    c_last_bonus_status_str = 'Не определен';
  if (c_last_bonus_status = 0) then
    c_last_bonus_status_str = 'Без статуса';
  if (c_last_bonus_status = 1) then
    c_last_bonus_status_str = 'Бронзовый';
  if (c_last_bonus_status = 2) then
    c_last_bonus_status_str = 'Серебрянный';
  if (c_last_bonus_status = 3) then
    c_last_bonus_status_str = 'Золотой';
  if (c_last_bonus_status = 4) then
    c_last_bonus_status_str = 'Бриллиантовый';
  if (c_last_bonus_status = 5) then
    c_last_bonus_status_str = 'Бриллиант-миллионер';

  if (c_curr_bonus_status = -1) then
    c_curr_bonus_status_str = 'Не определен';
  if (c_curr_bonus_status = 0) then
    c_curr_bonus_status_str = 'Без статуса';
  if (c_curr_bonus_status = 1) then
    c_curr_bonus_status_str = 'Бронзовый';
  if (c_curr_bonus_status = 2) then
    c_curr_bonus_status_str = 'Серебрянный';
  if (c_curr_bonus_status = 3) then
    c_curr_bonus_status_str = 'Золотой';
  if (c_curr_bonus_status = 4) then
    c_curr_bonus_status_str = 'Бриллиантовый';
  if (c_curr_bonus_status = 5) then
    c_curr_bonus_status_str = 'Бриллиант-миллионер';

  if ((db_id = 81001) and
      (rep_date >= cast('05.10.2015' as date))) then
  begin
    if (c_last_bonus_status = 0) then
      c_last_bonus_status_str = 'Без статуса';
    if (c_last_bonus_status = 1) then
      c_last_bonus_status_str = 'Бронза';
    if (c_last_bonus_status = 2) then
      c_last_bonus_status_str = 'Серебро';
    if (c_last_bonus_status = 3) then
      c_last_bonus_status_str = 'Золото';
    if (c_last_bonus_status = 4) then
      c_last_bonus_status_str = 'Бриллиант';
    if (c_last_bonus_status = 5) then
      c_last_bonus_status_str = 'Бриллиант*';
    if (c_last_bonus_status = 6) then
      c_last_bonus_status_str = 'Бриллиант**';
    if (c_last_bonus_status = 7) then
      c_last_bonus_status_str = 'Бриллиант***';
    if (c_last_bonus_status = 8) then
      c_last_bonus_status_str = 'Бриллиант****';
    if (c_last_bonus_status = 9) then
      c_last_bonus_status_str = 'Бриллиант*****';
    if (c_last_bonus_status = 10) then
      c_last_bonus_status_str = 'Платинум';

    if (c_curr_bonus_status = 0) then
      c_curr_bonus_status_str = 'Без статуса';
    if (c_curr_bonus_status = 1) then
      c_curr_bonus_status_str = 'Бронза';
    if (c_curr_bonus_status = 2) then
      c_curr_bonus_status_str = 'Серебро';
    if (c_curr_bonus_status = 3) then
      c_curr_bonus_status_str = 'Золото';
    if (c_curr_bonus_status = 4) then
      c_curr_bonus_status_str = 'Бриллиант';
    if (c_curr_bonus_status = 5) then
      c_curr_bonus_status_str = 'Бриллиант*';
    if (c_curr_bonus_status = 6) then
      c_curr_bonus_status_str = 'Бриллиант**';
    if (c_curr_bonus_status = 7) then
      c_curr_bonus_status_str = 'Бриллиант***';
    if (c_curr_bonus_status = 8) then
      c_curr_bonus_status_str = 'Бриллиант****';
    if (c_curr_bonus_status = 9) then
      c_curr_bonus_status_str = 'Бриллиант*****';
    if (c_curr_bonus_status = 10) then
      c_curr_bonus_status_str = 'Платинум';
  end

  suspend;
end
