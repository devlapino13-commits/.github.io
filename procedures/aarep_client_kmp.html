CREATE PROCEDURE AAREP_CLIENT_KMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME VARCHAR(50),
    PARENT_NAME VARCHAR(50),
    C_NAME_FIO VARCHAR(100),
    MCL_APP_STATUS_STR VARCHAR(10),
    MCL_APP_VERSION_STR VARCHAR(10),
    MCL_LAST_LOGIN_DATETIME TIMESTAMP,
    C_NAME_FIO_OFFICER VARCHAR(100))
AS
declare variable C_ID decimal(18,0);
declare variable MCL_APP_VERSION integer;
declare variable MCL_ID decimal(18,0);
declare variable KF_OFFICER decimal(18,0);
begin

  for select distinct cl.c_id, reg.reg_name, parent_name, cl.c_name_fio, knl.kal_new_kf_officer
      from knclient kncl
      inner join kontract kon on kncl.kc_from_k_id = kon.k_id
      inner join region reg on reg.reg_unicode = kon.k_fact_disb_reg_id
      inner join region_list(:REP_REGION_ID, 1, 3, 1) on id_list = reg.reg_unicode
      inner join kontstate ks on ks.ks_from_k_id = kon.k_id
      inner join clients cl on cl.c_id = kncl.kc_from_c_id
      inner join knaktlist knl on knl.kal_from_kf_id = kon.k_from_kf_id and
                 :Rep_Date_End between knl.kal_date and knl.kal_date_end
      where kncl.kc_amount_applicate > 0 and
            :Rep_Date_End between ks_date_begin and ks_date_end and
            ks.ks_state = 1 and
            ks.ks_ol_is_delete = 0
      into :c_id, :reg_name, :parent_name, :c_name_fio, :kf_officer

  do
  begin

    mcl_id = null;
    mcl_app_version = null;
    mcl_last_login_datetime = null;
    --mcl_app_version_str = null;
    --mcl_app_status_str = null;

    select f_fsm_client_status(c_id)
    from clients cl
    where cl.c_id = :c_id
    into :mcl_app_version;

    select c_name_fio
    from clients cl
    where cl.c_id = :kf_officer
    into :c_name_fio_officer;

    select ml.mcl_id, mcl_last_login_datetime
    from mc_login ml
    where ml.mcl_from_c_id = :c_id and
          ml.mcl_is_delete = 0
    order by mcl_last_login_datetime desc rows 1
    into :mcl_id, :mcl_last_login_datetime;

    if (mcl_id is null) then
      mcl_app_status_str = 'нет';
    else
      mcl_app_status_str = 'да';

    if (mcl_app_version > 1) then
      mcl_app_version_str = 'да';
    else
      mcl_app_version_str = 'нет';

    suspend;
  end
end
