CREATE PROCEDURE MC_EDOC_LIST(
    MCL_ID D_ID)
RETURNS (
    EDS_ID D_ID,
    EDC_NAME D_STR_255,
    EDL_BLANK_DATE D_DATE,
    DOC_NUMBER D_STR_CODE,
    EDS_STATUS D_INTEGER,
    EDS_STATUS_STR D_STR_CODE,
    SIGN_TYPE D_STR_20)
AS
begin
   -- измененеи алгоритма процедуры MC_EDOC_LIST связано с PM-133
   if(iif(mcl_id = 1111700, false, f_get_const('mc_edoc_list_new_alg') = 0)) then
   begin
       sign_type = 'standard';
       -- до запуска проекта pm-133 (весь серипт mc_edoc_list был перенесен в mc_edoc_list_old)
       for select eds_id, edc_name, edl_blank_date, doc_number, eds_status, eds_status_str
       from mc_edoc_list_old(:mcl_id)
       into :eds_id, :edc_name, :edl_blank_date, :doc_number, :eds_status, :eds_status_str
       do
        suspend;
   end
   else
   begin
       for select eds_id, edc_name, edl_blank_date, doc_number, eds_status, eds_status_str, sign_type
       from mc_edoc_list_new(:mcl_id)
       order by edl_blank_date desc
       into :eds_id, :edc_name, :edl_blank_date, :doc_number, :eds_status, :eds_status_str, sign_type
       do
        suspend;
   end
end
