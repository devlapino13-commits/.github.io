CREATE PROCEDURE AAREP_SB_CHECK_LIST_TMP
RETURNS (
    SBL_ID INTEGER,
    SBL_FROM_C_ID DECIMAL(18,0),
    SBL_C_NAME_FIO VARCHAR(100),
    SBL_DATE_BEGIN DATE,
    SBL_DATE_END DATE,
    SBL_FROM_REG_ID INTEGER,
    SBL_HR_REG_ID INTEGER,
    SBL_HR_DATE_BEGIN DATE,
    SBL_HR_DATE_END DATE,
    SBL_HR_IS_WORKING INTEGER,
    SBL_RESULT VARCHAR(100),
    SBL_RESULT_ID INTEGER)
AS
begin
  sbl_result_id = 0;
  sbl_result = '';
  for select sbl_id, sbl_from_c_id, sbl_date_begin, sbl_date_end, sbl_from_reg_id, c_name_fio
      from sb_list, clients
      where sbl_type = 0 and
            sbl_from_c_id = c_id
      into :sbl_id, :sbl_from_c_id, sbl_date_begin, sbl_date_end, sbl_from_reg_id, sbl_c_name_fio
  do
  begin
    sbl_hr_date_begin = null;
    sbl_hr_date_end = null;
    sbl_hr_reg_id = null;
    sbl_hr_is_working = -1;

    select ps_date_begin, ps_date_end, ps_from_reg_id, ps_is_working
    from joblist, jobkind, hr_pstatus
    where jl_kind = jk_id and
          jk_system_code = 'JobSystem.Specialist_SB' and
          ps_from_jl_id = jl_id and
          ps_status = 1 and
          ps_from_c_id = :sbl_from_c_id
    order by ps_date_begin desc
    rows 1
    into :sbl_hr_date_begin, :sbl_hr_date_end, :sbl_hr_reg_id, :sbl_hr_is_working;
    if (sbl_hr_is_working <> 1) then
    begin
      sbl_result = 'Необходимо исключить из списка СБ';
      sbl_result_id = -1;
    end
    else
    begin
      if (sbl_from_reg_id <> sbl_hr_reg_id) then
      begin
        sbl_result = 'Необходима поправка';
        sbl_result_id = -2;
      end
    end
    suspend;
  end

  sbl_id = 0;
  sbl_date_begin = null;
  sbl_date_end = null;
  sbl_from_reg_id = null;
  sbl_hr_is_working = 1;
  for select ps_from_c_id, ps_date_begin, ps_date_end, ps_from_reg_id, ps_is_working, c_name_fio
      from joblist, jobkind, hr_pstatus, clients
      where jl_kind = jk_id and
            jk_system_code = 'JobSystem.Specialist_SB' and
            ps_from_jl_id = jl_id and
            ps_date_end >= '31.12.2999' and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id = c_id
      into :sbl_from_c_id, :sbl_hr_date_begin, :sbl_hr_date_end, :sbl_hr_reg_id, :sbl_hr_is_working, :sbl_c_name_fio
  do
  begin
    if (not(exists(select *
                   from sb_list
                   where sbl_from_c_id = :sbl_from_c_id and
                         sbl_type = 0))) then
    begin
      sbl_id = sbl_id - 1;
      sbl_result = 'Необходимо добавить в список СБ';
      sbl_result_id = 1;
      suspend;
    end
  end
end
