CREATE PROCEDURE HP_GET_KONTRACT_PARAMS(
    KK_ID DECIMAL(18,0),
    DD DATE)
RETURNS (
    K_ID DECIMAL(18,0),
    K_DISB_DATE DATE,
    K_PERCENT_PER_DAY DECIMAL(18,7),
    K_PERCENT DECIMAL(18,7),
    K_KOLICH_CLIENT_APPROVED INTEGER,
    K_AMOUNT_APPROVED DECIMAL(18,4),
    K_FROM_KF_ID DECIMAL(18,0),
    K_KONTRACT_NO VARCHAR(80),
    K_KOLICH_REPAY INTEGER,
    K_CURRENCY VARCHAR(12),
    PR_ID INTEGER,
    PR_ROUND_FINE SMALLINT,
    PR_INTEREST_METHOD SMALLINT,
    PR_FINE_METHOD SMALLINT,
    PR_FINE_PERCENT DECIMAL(18,7),
    PR_FINE_FOR_PERCENT DECIMAL(18,7),
    PR_FINE_MAX_DAY INTEGER,
    PR_IS_FINE_WEEKEND SMALLINT,
    PR_PAY_PROS_MORE INTEGER,
    PR_IS_WEEKEND_WORK INTEGER,
    PR_PAY_PROS SMALLINT,
    PR_PAY_CURR SMALLINT,
    PR_PAY_PRED SMALLINT,
    PR_CURRENCY VARCHAR(12),
    PR_INDEX_CURR_CODE VARCHAR(12),
    K_INDEX_CURR_RATE DECIMAL(18,4),
    K_INDEX_CURR_CODE VARCHAR(12),
    PR_SET_RATE_IN_KONT INTEGER,
    PR_CALC_PERC_TAX INTEGER,
    PR_TWO_REPAY_DATE INTEGER,
    KAL_NEW_REG_ID INTEGER,
    PR_NAME VARCHAR(100),
    C_NAME_FIO VARCHAR(100),
    PR_ADV_PAY_PERCENT DECIMAL(18,7),
    PR_ADV_CLOSE_PERCENT DECIMAL(18,7),
    PR_IS_360 INTEGER,
    PR_IS_LOCAL_CURRENCY INTEGER,
    PR_IS_WEEKEND_WORK1 INTEGER,
    PR_CATEGORY_PERC INTEGER,
    PR_CAN_DEPOSIT SMALLINT,
    PR_KOMMIS_IS_PERCENT INTEGER,
    PR_NEED_RESCHEDULE INTEGER,
    PR_ROUND_PERC SMALLINT,
    PR_NEED_KOMMIS_ADV_PERCENT SMALLINT,
    KAL_NEW_KF_NAME VARCHAR(100),
    PR_CALC_PERC_FOR_DELIQ INTEGER,
    NSP_PERCENT_FOR_INT DECIMAL(18,7),
    NSP_ACC_INT_MAIN VARCHAR(20),
    NSP_ACC_INT_EXPENSE VARCHAR(20),
    NSP_ACC_INT_REPAY VARCHAR(20),
    NSP_PERCENT_FOR_EXTRA_INT DECIMAL(18,7),
    NSP_ACC_EXTRA_INT_MAIN VARCHAR(20),
    NSP_ACC_EXTRA_INT_EXPENSE VARCHAR(20),
    NSP_ACC_EXTRA_INT_REPAY VARCHAR(20))
AS
declare variable k_clients_cashier_id decimal(18,0);
begin
  for select k_id, k_disb_date, kd_percent_per_day, kd_percent, k_kolich_client_approved, k_amount_approved,
             k_from_kf_id, k_kontract_no, k_kolich_repay, k_currency, pr_id, pr_round_fine, pr_interest_method,
             pr_fine_method, pr_fine_percent, pr_fine_for_percent, pr_fine_max_day, pr_is_fine_weekend,
             pr_pay_pros_more, pr_is_weekend_work, pr_pay_pros, pr_pay_curr, pr_pay_pred, pr_currency,
             pr_index_curr_code, k_index_curr_rate, k_index_curr_code, pr_set_rate_in_kont, pr_calc_perc_tax,
             pr_two_repay_date, kal_new_reg_id, pr_name, pr_adv_pay_percent, pr_adv_close_percent, pr_is_360,
             pr_is_local_currency, pr_is_weekend_work, pr_category_perc, pr_can_deposit, pr_kommis_is_percent,
             pr_need_reschedule, pr_round_perc, pr_need_kommis_adv_percent, kal_new_kf_name, pr_calc_perc_for_deliq,
             k_clients_cashier_id
      from kontract, product, knaktlist, kontdata
      where k_id = :kk_id and
            k_from_pr_id = pr_id and
            k_from_kf_id = kal_from_kf_id and
            kal_date <= :dd and
            kal_date_end >= :dd and
            knaktlist.is_delete = 0 and
            kd_ol_is_delete = 0 and
            kd_ol_date <= :dd and
            kd_date_end >= :dd and
            kd_from_k_id = :kk_id
      into :k_id, :k_disb_date, :k_percent_per_day, :k_percent, :k_kolich_client_approved, :k_amount_approved,
           :k_from_kf_id, :k_kontract_no, :k_kolich_repay, :k_currency, :pr_id, :pr_round_fine, :pr_interest_method,
           :pr_fine_method, :pr_fine_percent, :pr_fine_for_percent, :pr_fine_max_day, :pr_is_fine_weekend,
           :pr_pay_pros_more, :pr_is_weekend_work, :pr_pay_pros, :pr_pay_curr, :pr_pay_pred, :pr_currency,
           :pr_index_curr_code, :k_index_curr_rate, :k_index_curr_code, :pr_set_rate_in_kont, :pr_calc_perc_tax,
           :pr_two_repay_date, :kal_new_reg_id, :pr_name, :pr_adv_pay_percent, :pr_adv_close_percent, :pr_is_360,
           :pr_is_local_currency, :pr_is_weekend_work1, :pr_category_perc, :pr_can_deposit, :pr_kommis_is_percent,
           :pr_need_reschedule, :pr_round_perc, :pr_need_kommis_adv_percent, :kal_new_kf_name, :pr_calc_perc_for_deliq,
           :k_clients_cashier_id
  do
  begin
    c_name_fio = null;
    if (k_clients_cashier_id is not null) then
    begin
      select c_name_fio
      from clients
      where c_id = :k_clients_cashier_id
      into :c_name_fio;
    end

    select nsp_percent_for_int, nsp_percent_for_extra_int, nsp_acc_int_main, nsp_acc_int_expense, nsp_acc_int_repay,
           nsp_acc_extra_int_main, nsp_acc_extra_int_expense, nsp_acc_extra_int_repay
    from hp_get_kontract_nsp_acc(:kal_new_reg_id, :k_disb_date)
    into :nsp_percent_for_int, :nsp_percent_for_extra_int, :nsp_acc_int_main, :nsp_acc_int_expense, :nsp_acc_int_repay,
         :nsp_acc_extra_int_main, :nsp_acc_extra_int_expense, :nsp_acc_extra_int_repay;

    suspend;
  end
end
