CREATE PROCEDURE HP_OPERCASH_AKT_PLUS(
    IN_OFFICE_ID D_INTEGER,
    IN_CASH_ID D_INTEGER,
    IN_DATE D_DATE,
    IN_CIL_ID D_INTEGER,
    IN_MANAGER_ID D_ID,
    IN_UN_ID D_ID)
RETURNS (
    CFL_FROM_CIL_ID D_INTEGER,
    CFL_FROM_CSP_ID D_INTEGER,
    CFL_DATE D_DATE,
    CFL_CASHIER_ID D_ID,
    CFL_INSPECTOR_ID D_ID,
    CFL_FACT_AMOUNT D_CURRENCY,
    CFL_FACT_AMOUNT$BOND D_STR_LONG,
    CFL_FROM_UN_ID D_ID,
    RES_MEMO D_STR_LONG,
    RES_AMOUNT D_CURRENCY,
    RES_OFFICE_ID D_INTEGER,
    RES_IS_IPF D_BOOL,
    PERSON_ID D_ID,
    DIFF_ACCOUNT D_STR_CODE)
AS
declare variable cil_state d_smallint;
begin
  res_memo = '';
  res_amount = 0;

  if (in_cil_id is null) then
    exception e_message 'Id акта не указан';

  if (in_cash_id is null) then
    exception e_message 'Касса не указана';

  if (in_manager_id is null) then
    exception e_message 'Контролер не указан';

  if (in_date is null) then
    in_date = current_date;

  if (exists(select *
             from hp_is_day_closed(:in_office_id, :in_date, 1)
             where day_closed = 1)) then
    exception e_message 'Опер.день кассы закрыт: ' || f_datetostr(in_date);

  select cil_inv_amount, cil_state
  from cashinvlist
  where cil_id = :in_cil_id
  into res_amount, cil_state;

  if (row_count = 0) then
    exception e_message 'Акт отсутствует в БД, Id: ' || in_cil_id;

  -- Добавить проверку по статусу эл.документа - CIL_STATE

  if (exists(select * from cashinvlist
             where cil_parent_id = :in_cil_id and
                   cil_is_delete = 0)) then
      exception e_message 'Данный акт уже принят, Id: ' || in_cil_id;

  cfl_from_cil_id = in_cil_id;
  cfl_from_csp_id = in_cash_id;
  cfl_date = in_date;
  cfl_inspector_id = in_manager_id;
  cfl_fact_amount = 0;
  cfl_fact_amount$bond = '';
  cfl_from_un_id = in_un_id;
  res_is_ipf = 0;
  res_office_id = in_office_id;
  diff_account = null;

  select csp_cashier_id
  from cashplace
  where csp_id = :in_cash_id
  into cfl_cashier_id;

  select opc_from_c_id, opc_is_ipf
  from opercash
  where opc_from_cil_id = :in_cil_id and
        opc_parent_id is null and
        opc_is_delete = 0
  into person_id, res_is_ipf;

  if (row_count = 0) then
    person_id = cfl_cashier_id;
  else
    person_id = cfl_cashier_id;

  cfl_fact_amount = res_amount;

  select res_bond_list
  from hp_cashbond_load(:in_cil_id)
  into cfl_fact_amount$bond;

  suspend;
end
