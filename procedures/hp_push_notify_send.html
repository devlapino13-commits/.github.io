CREATE PROCEDURE HP_PUSH_NOTIFY_SEND
AS
declare variable pno_id d_integer;
declare variable planned_date d_date;
declare variable pno_type_receiver d_smallint;
declare variable pno_date_begin d_date;
declare variable pno_date_end d_date;
declare variable pno_reg_client d_integer;
declare variable pno_prj_user d_smallint;
declare variable pno_user_adv d_bool;
declare variable pno_title d_str_255;
declare variable pno_content d_str_long;
declare variable c_id d_id;
declare variable mcl_id d_id;
declare variable qty d_integer;
declare variable storage_days d_integer;
begin
  storage_days = 14;
  for
    select
      pno_id,
      pno_planned_date,
      pno_type_receiver,
      pno_date_begin,
      pno_date_end,
      pno_reg_client,
      pno_prj_user,
      pno_user_adv,
      pno_title,
      pno_content
    from push_notify
    where pno_planned_date <= current_timestamp
      and pno_status = 1
  into
    :pno_id,
    :planned_date,
    :pno_type_receiver,
    :pno_date_begin,
    :pno_date_end,
    :pno_reg_client,
    :pno_prj_user,
    :pno_user_adv,
    :pno_title,
    :pno_content
  do
  begin
    qty = 0;

    -- активные клиенты
    if (pno_type_receiver = 0) then
    begin
      for
        select
          kc_from_c_id
        from oksummary
          inner join knclient
            on kc_from_k_id = oks_from_k_id
              and kc_amount_approved > 0
          inner join region_list(:pno_reg_client, 1, 3, 1)
            on id_list = oks_region_id
        where :planned_date between oks_date_begin and oks_date_end
          and oks_close_state = 0
          and exists(
            select * from mc_login
            where mcl_from_c_id = kc_from_c_id
              and mcl_is_delete = 0
              and (:pno_user_adv = 0 or f_edoc_status('ad.agreement', mcl_from_c_id, :planned_date) = 'signed'))
        group by kc_from_c_id
      into
        :c_id
      do
      begin
        execute procedure mc_send_push('clients', :c_id, :pno_title, :pno_content, :storage_days);
        qty = qty + 1;
      end
    end else
    -- активные пользователи КМП
    if (pno_type_receiver = 1) then
    begin
      for
        select
          max(mcl_id)
        from mc_login
        where mcl_is_delete = 0
          and mcl_project = iif(:pno_prj_user = 0, mcl_project, :pno_prj_user)
          and (:pno_user_adv = 0 or f_edoc_status('ad.agreement', mcl_from_c_id, :planned_date) = 'signed')
        group by mcl_from_c_id, mcl_project
      into
        :mcl_id
      do
      begin
        execute procedure mc_send_push('mc_login', :mcl_id, :pno_title, :pno_content, :storage_days);
        qty = qty + 1;
      end
    end else
    -- закрытые клиенты
    if (pno_type_receiver = 2) then
    begin
      for
        select
          kc_from_c_id
        from kontstate
          inner join knclient
            on kc_from_k_id = ks_from_k_id
              and kc_amount_approved > 0
              and exists(
                select * from mc_login
                where mcl_from_c_id = kc_from_c_id
                  and mcl_is_delete = 0
                  and (:pno_user_adv = 0 or f_edoc_status('ad.agreement', mcl_from_c_id, :planned_date) = 'signed'))
          inner join region_list(:pno_reg_client, 1, 3, 1)
            on id_list = ks_region_id
        where ks_date_begin between :pno_date_begin and :pno_date_end
          and ks_state = 2
          and ks_ol_is_delete = 0
        group by kc_from_c_id
      into
        :c_id
      do
      begin
        execute procedure mc_send_push('clients', :c_id, :pno_title, :pno_content, :storage_days);
        qty = qty + 1;
      end
    end

    if (qty > 0) then
      update push_notify set
        pno_numb_recipient = :qty,
        pno_execution_date = current_timestamp,
        pno_status = 2
      where pno_id = :pno_id;
  end
end
