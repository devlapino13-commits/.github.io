CREATE PROCEDURE HP_EXTINFO_GET_EXTRA_BLOB(
    ID D_ID NOT NULL,
    KIND D_STR_CODE NOT NULL)
RETURNS (
    BLOB_FILE D_BLOB_TEXT)
AS
declare variable kind_id d_id;
begin
  if (kind = 'inn_gnivc_request') then
  begin
    blob_file = (
      select
        json_value
      from extinfolist
        inner join hp_extinfo_to_send_inn_gnivc(eil_from_c_id, eil_id)
          on 1 = 1
      where eil_id = :id);
  end else
  if (kind = 'inn_gnivc_response') then
  begin
    blob_file = (
      select
        blx_blob
      from extinfolist
        inner join hp_get_blobex_blob(eil_from_bls_id)
          on 1 = 1
      where eil_id = :id);
  end else
  begin
    kind_id = f_enum_id('extinfo_bls_kind', kind);

    select blx_blob
    from extinfolistbls, hp_get_blobex_blob(eib_from_bls_id)
    where eib_from_eil_id = :id
      and eib_kind = :kind_id
    into blob_file;
  end

  suspend;
end
