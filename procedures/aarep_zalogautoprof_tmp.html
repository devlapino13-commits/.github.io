CREATE PROCEDURE AAREP_ZALOGAUTOPROF_TMP(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_LIST_KIND D_STR_CODE,
    REP_BOOK_KIND D_SMALLINT,
    REP_K_ID D_ID)
RETURNS (
    ZAP_ID D_INTEGER,
    ZAP_FROM_ZA_ID D_INTEGER,
    ZAP_DATE D_DATE,
    ZAP_NO D_STR_CODE,
    ZAP_AVG_COST D_CURRENCY,
    ZAP_ZALOG_COST D_CURRENCY,
    ZAP_STATUS D_SMALLINT,
    ZAP_FROM_OL_ID D_ID,
    ZAD_FROM_UN_ID D_ID,
    ZAD_DATETIME D_DATETIME)
AS
declare variable ol_is_delete d_smallint;
begin
  if (coalesce(rep_list_kind, '') = '') then
    rep_list_kind = 'prof';

  if (rep_book_kind is null) then
    rep_book_kind = 0;

  if (coalesce(rep_k_id, 0) = 0) then
  begin
    if (rep_list_kind = 'prof') then
    begin
      for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
                 zad_from_un_id, zad_datetime, ol.is_delete
      from zalogautoprof left join operlist ol on (ol.ol_id = zap_from_ol_id)
      where zap_date between :rep_date_begin and :rep_date_end
      into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
           :zad_from_un_id, :zad_datetime, ol_is_delete
      do
      begin
        if (ol_is_delete is null) then
          ol_is_delete = 0;

        if (rep_book_kind = 1) then
        begin
          if (zap_from_ol_id is null or ol_is_delete = 1) then
            Continue;
        end else
        if (rep_book_kind = -1) then
        begin
          if (zap_from_ol_id is not null and ol_is_delete = 0) then
            Continue;
        end

        suspend;
      end
    end else
    if (rep_list_kind = 'active') then
    begin
      for select distinct z_from_k_id
      from zalog
      into rep_k_id
      do
      begin
        if (exists(select * from kontstate
                   where ks_from_k_id = :rep_k_id and
                         ks_state = 1 and
                         ks_ol_is_delete = 0 and
                         ks_date_begin <= :rep_date_end and
                         ks_date_end >= :rep_date_begin)) then
        begin
          if (rep_book_kind = -1) then
          begin
            for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
                       zad_from_un_id, zad_datetime
            from zalogautoprof
                   inner join zalogauto on (zap_from_za_id = za_id)
                   left join operlist ol on (ol.ol_id = zap_from_ol_id)
            where za_from_k_id = :rep_k_id and
                  (ol.ol_id is null or ol.is_delete = 1)
            order by zap_date desc
            rows 1
            into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
                 :zad_from_un_id, :zad_datetime
            do
              suspend;
          end else
          if (rep_book_kind = 1) then
          begin
            for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
                       zad_from_un_id, zad_datetime
            from zalogautoprof, zalogauto, operlist ol
            where zap_from_za_id = za_id and
                  za_from_k_id = :rep_k_id and
                  zap_from_ol_id = ol.ol_id and
                  ol.is_delete = 0
            order by zap_date desc
            rows 1
            into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
                 :zad_from_un_id, :zad_datetime
            do
              suspend;
          end else
          begin
            for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
                       zad_from_un_id, zad_datetime
            from zalogautoprof, zalogauto
            where zap_from_za_id = za_id and
                  za_from_k_id = :rep_k_id
            order by zap_date desc
            rows 1
            into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
                 :zad_from_un_id, :zad_datetime
            do
              suspend;
          end
        end
      end
    end else
    if (rep_list_kind = 'disb') then
    begin
      for select distinct ks_from_k_id
      from kontstate, zalog
      where ks_from_k_id = z_from_k_id and
            ks_state = 1 and
            ks_ol_is_delete = 0 and
            ks_date_begin between :rep_date_begin and :rep_date_end
      into rep_k_id
      do
      begin
        if (rep_book_kind = -1) then
        begin
          for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
                     zad_from_un_id, zad_datetime
          from zalogautoprof
                 inner join zalogauto on (zap_from_za_id = za_id)
                 left join operlist ol on (ol.ol_id = zap_from_ol_id)
          where za_from_k_id = :rep_k_id and
                (ol.ol_id is null or ol.is_delete = 1)
          order by zap_date desc
          rows 1
          into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
               :zad_from_un_id, :zad_datetime
          do
            suspend;
        end else
        if (rep_book_kind = 1) then
        begin
          for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
                     zad_from_un_id, zad_datetime
          from zalogautoprof, zalogauto, operlist ol
          where zap_from_za_id = za_id and
                za_from_k_id = :rep_k_id and
                zap_from_ol_id = ol.ol_id and
                ol.is_delete = 0
          order by zap_date desc
          rows 1
          into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
               :zad_from_un_id, :zad_datetime
          do
            suspend;
        end else
        begin
          for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
                     zad_from_un_id, zad_datetime
          from zalogautoprof, zalogauto
          where zap_from_za_id = za_id and
                za_from_k_id = :rep_k_id
          order by zap_date desc
          rows 1
          into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
               :zad_from_un_id, :zad_datetime
          do
            suspend;
        end
      end
    end
  end else
  begin
    for select zap_id, zap_from_za_id, zap_date, zap_no, zap_avg_cost, zap_zalog_cost, zap_status, zap_from_ol_id,
               zad_from_un_id, zad_datetime
    from zalogautoprof, zalogauto
    where zap_from_za_id = za_id and
          za_from_k_id = :rep_k_id
    into :zap_id, :zap_from_za_id, :zap_date, :zap_no, :zap_avg_cost, :zap_zalog_cost, :zap_status, :zap_from_ol_id,
         :zad_from_un_id, :zad_datetime
    do
      suspend;
  end
end
