CREATE PROCEDURE AAREP_KANS_REP_ORDERS(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    KSL_ID INTEGER,
    KSL_SALDO DECIMAL(18,4),
    KSL_KSM_NAME VARCHAR(255),
    KSL_NAME VARCHAR(100),
    KSL_MEMO VARCHAR(255),
    KSL_YEDIZ VARCHAR(20))
AS
begin
  /* заказы минус купленные: Остаток чего надо докупить */
  for select kol_from_ksl_id, sum(kol_kolich), ksm_name, ksl_name, ksl_memo, ksl_yediz
      from kansorlist, kansorder, region_list(:rep_region_id, 1, -1, 1), kansmain, kanslist
      where kol_from_kso_id = kso_id and
            kso_date >= :rep_date_begin and
            kso_date <= :rep_date_end and
            (kso_from_reg_id = id_list) and
            kso_state > 0 and
            kso_is_delete = 0 and
            ksl_from_ksm_id = ksm_id and
            ksl_id = kol_from_ksl_id
      group by kol_from_ksl_id, ksm_name, ksl_name, ksl_memo, ksl_yediz
      having sum(kol_kolich) <> 0
      into :ksl_id, :ksl_saldo, :ksl_ksm_name, :ksl_name, :ksl_memo, :ksl_yediz
  do
  begin
    suspend;
  end
end
