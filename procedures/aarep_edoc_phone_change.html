CREATE PROCEDURE AAREP_EDOC_PHONE_CHANGE(
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_REGION_ID D_INTEGER)
RETURNS (
    EDF_PRINT_NAME D_STR_100,
    EDL_OBJECT_NAME D_STR_255,
    EDBV_BEGIN_DATE D_DATE,
    EDBV_BEGIN_TIME D_TIME,
    EDBV_END_DATE D_DATE,
    EDBV_END_TIME D_TIME,
    STATUS_STR D_STR_CODE,
    RD_NAME D_STR_255,
    EDL_CREATE_DATETIME D_DATETIME,
    BLS_DATETIME D_DATETIME,
    EDL_STATUS_STR D_STR_20,
    UN_FIO_CREATE D_STR_100,
    REG_NAME D_STR_100,
    EDS_STATUS_DATETIME D_DATETIME,
    EDS_STATUS_DATETIME_VERIFY D_DATETIME)
AS
begin

  for select edf_print_name, edl_object_name, edbv_begin_date, edbv_begin_time, edbv_end_date, edbv_end_time,
             status_str, rd_name, edl_create_datetime, eds_status_datetime, bls_datetime, edl_status_str,
             eds_status_datetime_verify, un_fio_create, reg_name
      from aarep_edoc_blank_verify(:rep_date_begin, :rep_date_end)
      where edf_print_name = 'Заявление на изменение номера' and
            reg_id = :rep_region_id and
            status_str = 'Одобрен'
      into edf_print_name, edl_object_name, edbv_begin_date, edbv_begin_time, edbv_end_date, edbv_end_time, status_str,
           rd_name, edl_create_datetime, eds_status_datetime, bls_datetime, edl_status_str, eds_status_datetime_verify,
           un_fio_create, reg_name
  do
  begin
    suspend;
  end
end
