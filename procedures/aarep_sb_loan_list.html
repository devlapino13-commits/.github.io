CREATE PROCEDURE AAREP_SB_LOAN_LIST(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE,
    REP_SB_TYPE INTEGER)
RETURNS (
    SBA_DATE_BEGIN DATE,
    SBA_DATE_END DATE,
    SB_NAME VARCHAR(255),
    SB_TEAM_NAME VARCHAR(255),
    SB_STATUS VARCHAR(20),
    OKS_BALANCE DECIMAL(18,4),
    OKS_CALC_PERC DECIMAL(18,4),
    OKS_FINE DECIMAL(18,4),
    OKS_DELINQ_DATE DATE,
    OKS_LOAN_STATUS VARCHAR(20),
    REG_NAME VARCHAR(100),
    OKS_BALANCE_IN_SB DECIMAL(18,4),
    K_KONTRACT_NO VARCHAR(20),
    CITIZEN_STR VARCHAR(20),
    DELINQ_DAY_TRANSFER INTEGER)
AS
declare variable k_id decimal(18,0);
declare variable sbl_id integer;
declare variable reg_id integer;
declare variable kol integer;
declare variable c_citizen integer;
declare variable sba_from_sbl_id integer;
begin
  for select sba_from_k_id, sba_date_begin, sba_date_end, sbl_name, sbl_team_name, sba_from_sbl_id
      from sb_akt, sb_list
      where sba_from_sbl_id = sbl_id and
            sba_is_delete = 0 and
            sba_date_begin <= :rep_date_end and
            sba_date_end >= :rep_date_begin and
            sba_status = 2 and
            sba_date_begin <= sba_date_end and
            sbl_type = :rep_sb_type
      into :k_id, :sba_date_begin, :sba_date_end, :sb_name, :sb_team_name, :sba_from_sbl_id
  do
  begin
    if (sba_from_sbl_id = 0) then
      sb_status = 'Отозван';
    else
      sb_status = 'В учете';

    oks_balance = null;
    oks_calc_perc = 0;
    oks_fine = 0;
    reg_id = null;
    select oks_balance, oks_delinq_date, oks_region_id
    from oksummary
    where oks_date_begin <= :rep_date_end and
          oks_date_end >= :rep_date_end and
          oks_from_k_id = :k_id and
          oks_close_state = 0
    into :oks_balance, oks_delinq_date, reg_id;

    if (oks_balance is null) then
      oks_loan_status = 'Договор закрыт';
    else
    begin
      oks_loan_status = 'Договор активный';

      if (oks_delinq_date > :rep_date_end) then
        oks_delinq_date = null;
      select sum(sh_calc_fine), sum(sh_calc_perc)
      from aarep_make_zad_list(:k_id, :rep_date_end, 1)
      where sh_date is not null
      into :oks_fine, oks_calc_perc;
      if (oks_fine is null) then
        oks_fine = 0;
      if (oks_calc_perc is null) then
        oks_calc_perc = 0;
    end
    if (oks_balance is null) then
      oks_balance = 0;
    reg_name = '';
    if (reg_id is not null) then
    begin
      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = :reg_id
      into :reg_name;
    end
    if (reg_name is null) then
      reg_name = '';
    oks_balance_in_sb = 0;
    if (sba_date_end > :rep_date_end) then
      oks_balance_in_sb = oks_balance;
    if (sba_date_end >= '31.12.2999') then
      sba_date_end = null;

    select k_kontract_no
    from kontract
    where k_id = :k_id
    into :k_kontract_no;

--   определяет гражданство группы
    c_citizen = null;
    select c_citizen, count(*) kol
    from knclient, clients
    where kc_from_k_id = :k_id and
          kc_amount_approved > 0 and
          kc_from_c_id = c_id
    group by c_citizen
    order by kol desc, c_citizen
    rows 1
    into c_citizen, kol;

    if (c_citizen is null) then
      c_citizen = 0;
    citizen_str = null;
    select rd_name
    from refdata
    where rd_id = :c_citizen
    into citizen_str;
--   кол-во просроченных дней на дату передачи
    delinq_day_transfer = 0;
    select :sba_date_begin - oks_delinq_date
    from oksummary
    where oks_from_k_id = :k_id and
          oks_close_state = 0 and
          :sba_date_begin between oks_date_begin and oks_date_end
--     and oks_balance > 0
          and
          :sba_date_begin > oks_delinq_date
    into delinq_day_transfer;
    if (delinq_day_transfer is null) then
      delinq_day_transfer = 0;

    suspend;
  end

end
