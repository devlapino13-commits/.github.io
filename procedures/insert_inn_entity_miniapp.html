CREATE PROCEDURE INSERT_INN_ENTITY_MINIAPP(
    MA_FROM_REG_ID INTEGER,
    MA_ID DECIMAL(18,0),
    MA_FROM_UN_ID DECIMAL(18,0),
    IS_PORUCH SMALLINT)
RETURNS (
    SEND_KOL INTEGER)
AS
declare variable c_id decimal(18,0);
declare variable db_id integer;
declare variable ma_id_tmp integer;
begin
  send_kol = 0;

  select k_from_ma_app_id
  from kontract
  where k_id = :ma_id
  into :ma_id_tmp;
  if (ma_id_tmp is null) then
    ma_id_tmp = :ma_id;

  if (is_poruch <> 1) then
  begin
    for select mac_from_c_id
        from miniappclient
        where mac_from_ma_id = :ma_id_tmp and
              mac_amount > 0 and
              mac_is_cashier = 1
        into :c_id
    do
    begin
      select count(*)
      from extinfolist eti
      where eti.eil_from_c_id = :c_id and
            eti.eil_object_id = :ma_id_tmp and
            eti.eil_from_eic_id = 3
      into send_kol;

      send_kol = coalesce(send_kol, 0);

      if (send_kol <> 2) then
      begin
        insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json)
        values (:c_id, 3, 11, :ma_id_tmp, 1);
      end
    end
  end
  else
  begin
    for select mac_from_c_id
        from miniappclient
        where mac_from_ma_id = :ma_id_tmp and
              mac_is_poruch = 1
        into :c_id
    do
    begin
      select count(*)
      from extinfolist eti
      where eti.eil_from_c_id = :c_id and
            eti.eil_object_id = :ma_id_tmp and
            eti.eil_from_eic_id = 3
      into send_kol;

      send_kol = coalesce(send_kol, 0);

      if (send_kol <> 2) then
      begin
        insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json)
        values (:c_id, 3, 11, :ma_id_tmp, 1);
      end
    end
  end
  suspend;
end
