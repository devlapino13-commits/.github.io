CREATE PROCEDURE HP_ACC_OBOR_CHECK(
    BG_ID D_STR_CODE,
    REP_DATE D_DATE,
    NEED_DETAIL D_SMALLINT)
RETURNS (
    AL_ID D_STR_CODE,
    OL_DATE D_DATE,
    OL_ID D_ID,
    OB_NUMBER D_INTEGER,
    DT_AMOUNT D_CURRENCY,
    KT_AMOUNT D_CURRENCY,
    IS_ACTIVE D_BOOL,
    OLD_DT_AMOUNT D_CURRENCY,
    OLD_KT_AMOUNT D_CURRENCY,
    OLD_IS_ACTIVE D_BOOL,
    DIFF_DT D_CURRENCY,
    DIFF_KT D_CURRENCY,
    DIFF_ACTIVE D_BOOL)
AS
declare variable rep_date_end d_date;
begin
  -- поиск разницы по оборотам лицевых счетов
  ol_id = null;
  if (bg_id = '') then
    bg_id = null;


  if (need_detail is null) then
    need_detail = 0;

  rep_date_end = cast('27.01.2021' as date);

  for select al_id
  from acc_list
  where (:bg_id is null or al_from_bg_id = :bg_id)
  into al_id
  do
  begin
    for select obs_date, obs_dt_ekv, obs_kt_ekv
    from obsummary
    where obs_from_al_id = :al_id and
          obs_date between :rep_date and :rep_date_end
    order by obs_date
    into :ol_date, :dt_amount, kt_amount
    do
    begin
      ob_number = 0;
      old_dt_amount = 0;
      old_kt_amount = 0;
      ol_id = null;
      diff_active = 0;
      is_active = 1;
      old_is_active = 1;

      execute statement ('select obs_dt_ekv, obs_kt_ekv from obsummary where obs_from_al_id = :al_id and obs_date = :ol_date')
      (al_id := :al_id, ol_date := :ol_date)
      on external data source 'localhost/3050:F:\db_mbfru_20210128.ib'
      as user 'SYSDBA' password 'F!ora80X'
      into :old_dt_amount, old_kt_amount;

      if (old_dt_amount is null) then
        old_dt_amount = 0;

      if (old_kt_amount is null) then
        old_kt_amount = 0;

      if (dt_amount <> old_dt_amount or kt_amount <> old_kt_amount) then
      begin
        diff_dt = dt_amount - old_dt_amount;
        diff_kt = kt_amount - old_kt_amount;
        suspend;

        diff_dt = 0;
        diff_kt = 0;
        dt_amount = 0;
        kt_amount = 0;
        old_dt_amount = 0;
        old_kt_amount = 0;

        if (need_detail = 1) then
        begin
          for select ob_from_ol_id, ob_number, ob_summa, ob_ol_is_active
          from operbook
          where ob_account_dt = :al_id and                   
                ob_ol_date = :ol_date and
                ob_ol_is_active = 1
          into ol_id, ob_number, dt_amount, is_active
          do
          begin
            old_dt_amount = 0;
            old_is_active = 0;
            diff_active = 0;
            execute statement ('select ob_summa, ob_ol_is_active from operbook where ob_account_dt = :al_id and ob_ol_date = :ol_date and ob_from_ol_id = :ol_id and ob_number = :ob_number')
            (al_id := :al_id, ol_date := :ol_date, ol_id := :ol_id, ob_number := :ob_number)
            on external data source 'localhost/3050:F:\db_mbfru_20210128.ib'
            as user 'SYSDBA' password 'F!ora80X'
            into :old_dt_amount, old_is_active;

            diff_dt = dt_amount - old_dt_amount;
            if (is_active <> old_is_active) then
              diff_active = 1;

            if (diff_dt <> 0 or old_is_active = 0) then
              suspend;
          end

          diff_dt = 0;
          diff_kt = 0;
          dt_amount = 0;
          kt_amount = 0;
          old_dt_amount = 0;
          old_kt_amount = 0;

          for select ob_from_ol_id, ob_number, ob_summa, ob_ol_is_active
          from operbook
          where ob_account_kt = :al_id and
                ob_ol_date = :ol_date and
                ob_ol_is_active = 1
          into ol_id, ob_number, kt_amount, is_active
          do
          begin
            old_kt_amount = 0;
            old_is_active = 0;
            diff_active = 0;
            execute statement ('select ob_summa, ob_ol_is_active from operbook where ob_account_kt = :al_id and ob_ol_date = :ol_date and ob_from_ol_id = :ol_id and ob_number = :ob_number')
            (al_id := :al_id, ol_date := :ol_date, ol_id := :ol_id, ob_number := :ob_number)
            on external data source 'localhost/3050:F:\db_mbfru_20210128.ib'
            as user 'SYSDBA' password 'F!ora80X'
            into :old_kt_amount, old_is_active;

            diff_kt = kt_amount - old_kt_amount;
            if (is_active <> old_is_active) then
              diff_active = 1;

            if (diff_kt <> 0 or old_is_active = 0) then
              suspend;
          end
        end

        if (need_detail = 2) then
        begin
          for execute statement ('select ob_from_ol_id, ob_number, ob_summa, ob_ol_is_active from operbook where ob_account_dt = :al_id and ob_ol_date = :ol_date and ob_ol_is_active = 1')
          (al_id := :al_id, ol_date := :ol_date)
          on external data source 'localhost/3050:F:\db_mbfru_20210128.ib'
          as user 'SYSDBA' password 'F!ora80X'
          into ol_id, ob_number, old_dt_amount, is_active
          do
          begin
            dt_amount = 0;
            old_is_active = 1;
            diff_active = 0;

            select ob_summa, ob_ol_is_active
            from operbook
            where ob_account_dt = :al_id and
                  ob_ol_date = :ol_date and
                  ob_from_ol_id = :ol_id and
                  ob_number = :ob_number
            into :dt_amount, is_active;

            diff_dt = dt_amount - old_dt_amount;
            if (is_active <> old_is_active) then
              diff_active = 1;

            if (diff_dt <> 0 or is_active = 0) then
              suspend;
          end

          diff_dt = 0;
          diff_kt = 0;
          dt_amount = 0;
          kt_amount = 0;
          old_dt_amount = 0;
          old_kt_amount = 0;

          for execute statement ('select ob_from_ol_id, ob_number, ob_summa, ob_ol_is_active from operbook where ob_account_kt = :al_id and ob_ol_date = :ol_date and ob_ol_is_active = 1')
          (al_id := :al_id, ol_date := :ol_date)
          on external data source 'localhost/3050:F:\db_mbfru_20210128.ib'
          as user 'SYSDBA' password 'F!ora80X'
          into ol_id, ob_number, old_kt_amount, is_active
          do
          begin
            kt_amount = 0;
            old_is_active = 1;
            diff_active = 0;

            select ob_summa, ob_ol_is_active
            from operbook
            where ob_account_kt = :al_id and
                  ob_ol_date = :ol_date and
                  ob_from_ol_id = :ol_id and
                  ob_number = :ob_number
            into :kt_amount, is_active;

            diff_kt = kt_amount - old_kt_amount;
            if (is_active <> old_is_active) then
              diff_active = 1;

            if (diff_kt <> 0 or is_active = 0) then
              suspend;
          end
        end

      end
    end
  end
end
