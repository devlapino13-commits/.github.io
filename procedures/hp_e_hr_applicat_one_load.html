CREATE PROCEDURE HP_E_HR_APPLICAT_ONE_LOAD(
    IN_HRA_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    HRA_ID D_INTEGER,
    HRA_FROM_REG_ID D_INTEGER,
    HRA_FROM_C_ID ID_TYPE,
    HRA_DATE_BEGIN D_DATE,
    HRA_DATE_END D_DATE,
    HRA_FROM_HO_ID ID_TYPE,
    HRA_USER_ID ID_TYPE,
    HRA_DATETIME D_DATETIME,
    HRA_ORDER_TYPE D_INTEGER,
    HRA_IS_APPROVED D_INTEGER,
    HRA_IS_CHECKED D_INTEGER,
    HRA_MEMO VARCHAR(255),
    HRA_APPROVED_USER ID_TYPE,
    HRA_CHANGER_ID ID_TYPE,
    RES_ACTUAL_OFFICE D_STR_100,
    RES_ACTUAL_JOBLIST D_STR_100)
AS
declare variable diff_days d_integer;
begin
  res_actual_office = '';
  res_actual_joblist = '';

  if (coalesce(in_hra_id, 0) <> 0) then
  begin
    select hra_id,
           hra_from_reg_id,
           hra_from_c_id,
           hra_date_begin,
           hra_date_end,
           hra_from_ho_id,
           hra_user_id,
           hra_datetime,
           hra_order_type,
           hra_is_approved,
           hra_is_checked,
           hra_memo,
           hra_approved_user,
           hra_changer_id
    from hr_applicat
    where hra_id = :in_hra_id
    into :hra_id,
         :hra_from_reg_id,
         :hra_from_c_id,
         :hra_date_begin,
         :hra_date_end,
         :hra_from_ho_id,
         :hra_user_id,
         :hra_datetime,
         :hra_order_type,
         :hra_is_approved,
         :hra_is_checked,
         :hra_memo,
         :hra_approved_user,
         :hra_changer_id;

    select res_actual_office, res_actual_joblist
    from hp_e_hr_orders_pers_info(:hra_from_c_id, :hra_date_begin)
    into :res_actual_office, :res_actual_joblist;
  end else
  begin
    select hra_from_reg_id, hra_order_type, hra_memo, datediff(day, hra_date_begin, hra_date_end)
    from hr_applicat
    where hra_user_id = :un_id
    order by hra_id desc
    rows 1
    into hra_from_reg_id, hra_order_type, hra_memo, diff_days;

    if (row_count = 0) then
    begin
      select un_curr_reg_id
      from username
      where un_id = :un_id
      into hra_from_reg_id;
    end

    if (diff_days is null) then
      diff_days = 0;

    hra_date_begin = current_date;
    hra_date_end = dateadd(day, diff_days, hra_date_begin);
    hra_user_id = un_id;
    hra_is_approved = 0;
    hra_is_checked = 0;
  end

  suspend;
end
