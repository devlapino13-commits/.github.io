CREATE PROCEDURE CB_RU_SB_2014_07(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    SBL_NAME VARCHAR(100),
    PAR7_AVG DECIMAL(15,2),
    KOL_KONT_OLD INTEGER,
    KOL_KONT_NEW INTEGER,
    PLAN_WEEK INTEGER,
    KOL_CLI_SB INTEGER,
    DATE_BEGIN_WEEK DATE,
    DATE_END_WEEK DATE,
    WEEK_SUM DECIMAL(18,4),
    BONUS CURR_TYPE)
AS
declare variable reg_id_prior integer;
declare variable delinq_balance curr_type;
declare variable balance curr_type;
declare variable sba_from_k_id id_type;
declare variable sba_date_begin date;
declare variable sba_date_end date;
declare variable date_action date;
declare variable sbl_from_c_id decimal(18,0);
declare variable sbl_date_begin date;
declare variable sbl_from_reg_id integer;
declare variable sbl_id integer;
declare variable rep_date_begin date;
declare variable week_begin integer;
declare variable week_end integer;
declare variable circle integer;
declare variable sunday_k date;
declare variable k integer;
declare variable total_par7 decimal(18,7);
declare variable par7 decimal(18,7);
declare variable kol_sb integer;
declare variable sunday_end date;
declare variable sunday_begin date; /* Первое воскресение месяца */
declare variable month_plan integer;
declare variable date_begin date;
declare variable date_end date;
declare variable kol_cli_sb_temp integer;
declare variable month_sum decimal(18,4);
declare variable plan_month integer;
declare variable par7_avg_tem decimal(15,2);
declare variable rep_date date;
begin
  reg_id_prior = 0;
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;

  week_begin = extract(weekday from :rep_date_begin);
  if (week_begin = 0) then
    week_begin = 7;
  week_end = extract(weekday from :rep_date_end);
  if (week_end = 0) then
    week_end = 7;
--  if (week_end = 7) then circle = 4; else circle = 5;

  sunday_begin = rep_date_begin + 7 - week_begin;

-- день недели начало месяца >4 то, расчет будет начинатся со следующей недели
  date_begin = sunday_begin - 6;

  if (week_end = 7) then
  begin
    date_end = rep_date_end;
  end
  else
  begin
    date_end = rep_date_end - week_end;
  end

  delete from temp_kc_reg_info;
  insert into temp_kc_reg_info (kc_region_id, kc_kol_cli)
  select ac_region_id, sum(ac_is_client)
  from aarep_cli_active_list(:rep_date_end)
  where (select aa_category_id
         from aarep_get_category_id(ac_kontract_id, :rep_date_end, null, null, null, null, null, null)) in (1, 2)
  group by ac_region_id;

  par7_avg = 0;
  reg_name = '';
  for select sbl_from_c_id, sbl_name, sbl_date_begin, sbl_from_reg_id, sbl_id
      from sb_list, region_list(:rep_region_id, 1, 3, 1)
      where sbl_type = 0 and
            :rep_date_end between sbl_date_begin and coalesce(sbl_date_end, '31.12.9999') and
            sbl_from_c_id <> 1000000000000 and
            sbl_from_reg_id = id_list
--  and sbl_from_c_id = 1010100121926
      order by sbl_from_reg_id
      into sbl_from_c_id, sbl_name, sbl_date_begin, sbl_from_reg_id, sbl_id
  do
  begin
--  вопрос, что делать неделями, с чего начать если начало месяца попадает на четверг то первую неделю как брать,
--  если так тогда что с последним будет, в перходщем к первому несоответствует

-- Расчет среднего пара офиса
    if (reg_id_prior <> sbl_from_reg_id) then
    begin
      reg_id_prior = sbl_from_reg_id;
      sunday_k = sunday_begin;
      k = 0;
      total_par7 = 0;
      while (:sunday_k <= :rep_date_end) do
      begin
        delinq_balance = 0;
        select sum(oks_balance)
        from oksummary
        where :sunday_k between oks_date_begin and oks_date_end and
              cast(:sunday_k as date) - oks_delinq_date between 8 and 180 and
              oks_close_state = 0 and
              (select aa_category_id
               from aarep_get_category_id(oks_from_k_id, :sunday_k, null, null, null, null, null, null)) in (1, 2) and
              oks_region_id = :sbl_from_reg_id
        into delinq_balance;
        if (delinq_balance is null) then
          delinq_balance = 0;

        balance = 0;
        select sum(oks_balance)
        from oksummary
        where :sunday_k between oks_date_begin and oks_date_end and
              oks_close_state = 0 and
              (select aa_category_id
               from aarep_get_category_id(oks_from_k_id, :sunday_k, null, null, null, null, null, null)) in (1, 2) and
              oks_region_id = :sbl_from_reg_id
        into balance;
        if (balance is null) then
          balance = 0;

        if (balance > 0) then
          par7 = delinq_balance / balance * 100;
        else
          par7 = 0;

        total_par7 = total_par7 + par7;
        k = k + 1;
        sunday_k = :sunday_k + 7;
      end

      if (week_end = 7) then
      begin
        par7_avg = total_par7 / k;
      end
      else
      begin
        delinq_balance = 0;
        select sum(oks_balance)
        from oksummary
        where :sunday_k between oks_date_begin and oks_date_end and
              oks_close_state = 0 and
              cast(:sunday_k as date) - oks_delinq_date between 8 and 180 and
              (select aa_category_id
               from aarep_get_category_id(oks_from_k_id, :sunday_k, null, null, null, null, null, null)) in (1, 2) and
              oks_region_id = :sbl_from_reg_id
        into delinq_balance;
        if (delinq_balance is null) then
          delinq_balance = 0;

        balance = 0;
        select sum(oks_balance)
        from oksummary
        where :sunday_k between oks_date_begin and oks_date_end and
              oks_close_state = 0 and
              (select aa_category_id
               from aarep_get_category_id(oks_from_k_id, :sunday_k, null, null, null, null, null, null)) in (1, 2) and
              oks_region_id = :sbl_from_reg_id
        into balance;
        if (balance is null) then
          balance = 0;

        if (balance > 0) then
          par7 = delinq_balance / balance * 100;
        else
          par7 = 0;

        total_par7 = total_par7 + par7;
        k = k + 1;
        par7_avg = total_par7 / k;
      end

      reg_name = (select reg_code || ' ' || reg_name
                  from region
                  where reg_unicode = :sbl_from_reg_id);

      kol_sb = (select count(*)
                from sb_list
                where sbl_type = 0 and
                      :rep_date_end between sbl_date_begin and coalesce(sbl_date_end, '31.12.9999') and
                      sbl_from_c_id <> 1000000000000 and
                      sbl_from_reg_id = :sbl_from_reg_id);

      kol_cli_sb = (select kc_kol_cli
                    from temp_kc_reg_info
                    where kc_region_id = :sbl_from_reg_id);
      if (kol_cli_sb is null) then
        kol_cli_sb = 0;
      kol_cli_sb = cast(kol_cli_sb as double precision) / kol_sb;

      kol_cli_sb_temp = kol_cli_sb;

    end
    kol_cli_sb = 0;

    kol_kont_new = 0;
    kol_kont_old = 0;
    plan_month = 0;

    rep_date = date_begin;
    month_sum = 0;
    month_plan = 0;
    bonus = 0;

    while (rep_date < date_end) do
    begin
      plan_week = 0;
      week_sum = 0;
      date_begin_week = rep_date;
      date_end_week = rep_date + 6;
      kol_kont_new = 0;
      kol_kont_old = 0;

      for select sba_from_k_id, sba_date_begin, sba_date_end
          from sb_akt
          where sba_from_sbl_id = :sbl_id and
                sba_status = 2 and
                sba_date_begin <= :rep_date + 6 and
                sba_date_end >= :rep_date and
                sba_date_begin >= '30.06.2014'
          into sba_from_k_id, sba_date_begin, sba_date_end
      do
      begin
--    если передан в предыдущем месяце то для проверки просрочки дата берется начало отчетного перида
        if (sba_date_begin < :rep_date) then
          date_action = :rep_date;
        else
          date_action = :sba_date_begin;
--    если контракт просрочен и находится в данном СБ то, проверяем на конец периода не просрочен ли
        if (exists(select *
                   from oksummary
                   where oks_from_k_id = :sba_from_k_id and
                         oks_delinq_date < :date_action - 1 and
                         oks_close_state = 0 and
                         :date_action - 1 between oks_date_begin and oks_date_end) and
            date_action between sba_date_begin and sba_date_end) then
        begin
          if (sba_date_begin < :rep_date) then
            kol_kont_old = kol_kont_old + 1;
          else
            kol_kont_new = kol_kont_new + 1;

          if (not exists(select *
                         from oksummary
                         where oks_from_k_id = :sba_from_k_id and
                               oks_delinq_date < :rep_date + 6 and
                               oks_close_state = 0 and
                               :rep_date + 6 between oks_date_begin and oks_date_end)) then
          begin
            plan_month = plan_month + 1;
            plan_week = plan_week + 1;
          end
        end
      end

      week_sum = case
                   when plan_week <= 3 then -3000
                   when plan_week <= 5 then 1000
                   when plan_week <= 7 then 2000
                   when plan_week <= 9 then 5000
                   when plan_week <= 11 then 6000
                   else 10000
                 end;
      month_sum = month_sum + week_sum;
      month_plan = month_plan + plan_week;

      par7_avg_tem = par7_avg;
      par7_avg = null;

      suspend;
      par7_avg = par7_avg_tem;
      rep_date = rep_date + 7;
    end

    bonus = 0;
    kol_kont_old = null;
    kol_kont_new = null;
    plan_month = 0;
    plan_week = null;
    week_sum = null;
    date_begin_week = null;
    date_end_week = null;

    kol_cli_sb = kol_cli_sb_temp;

    sbl_name = 'Итог по ' || sbl_name;
    if (par7_avg > 7) then
    begin
      bonus = month_sum;
      month_sum = 0;
    end
    else
    begin
      bonus = case
                when kol_cli_sb < 1 then 0
                when kol_cli_sb < 300 then 6000
                when kol_cli_sb < 500 then 8000
                when kol_cli_sb < 750 then 11000
                when kol_cli_sb < 1000 then 15000
                when kol_cli_sb < 1250 then 20000
                when kol_cli_sb < 1500 then 25000
                when kol_cli_sb < 1750 then 30000
                when kol_cli_sb < 2000 then 35000
                when kol_cli_sb < 2500 then 40000
                else 45000
              end;
    end

    suspend;
  end
end
