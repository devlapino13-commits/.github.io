CREATE PROCEDURE HP_FT_INFOLIST(
    C_ID D_ID,
    FIB_KIND D_SMALLINT = 0)
RETURNS (
    FIL_ID D_ID,
    FIL_FROM_FIB_ID D_ID,
    FIL_FROM_C_ID D_ID,
    FIL_DATE_OPEN D_DATE,
    FIL_DATE_CLOSE D_DATE,
    FIL_IS_CLOSED D_SMALLINT,
    FIL_MEMO D_STR_LONG,
    FIL_BLOCK_STATE D_SMALLINT,
    FIL_LAST_UN_ID D_ID,
    FIL_SCHEDULE_DATE D_DATE,
    FIL_LAST_FIC_ID D_ID,
    FIL_FROM_REG_ID D_INTEGER,
    UN_NAME D_STR_100,
    REG_NAME D_STR_100,
    BLN_ID INTEGER)
AS
begin
  fib_kind = coalesce(fib_kind, 0);

  for select fil_id,
             fil_from_fib_id,
             fil_from_c_id,
             fil_date_open,
             fil_date_close,
             fil_is_closed,
             fil_memo,
             fil_block_state,
             fil_last_un_id,
             fil_schedule_date,
             fil_last_fic_id,
             fil_from_reg_id,
             fib_from_bln_id
      from ft_infolist, ft_infoblank
      where fil_from_c_id = :c_id
        and fil_from_fib_id = fib_id
        and fib_kind = :fib_kind
      order by fil_date_open desc, fil_id desc
      into fil_id,
           fil_from_fib_id,
           fil_from_c_id,
           fil_date_open,
           fil_date_close,
           fil_is_closed,
           fil_memo,
           fil_block_state,
           fil_last_un_id,
           fil_schedule_date,
           fil_last_fic_id,
           fil_from_reg_id,
           bln_id
  do
  begin
    un_name = '';
    reg_name = '';

    select un_name
    from username
    where un_id = :fil_last_un_id
    into un_name;

    select reg_code||' '||reg_name
    from region
    where reg_unicode = :fil_from_reg_id
    into reg_name;

    suspend;
  end
end
