CREATE PROCEDURE UXP$UXFRITEM_ADD_PARAMS(
    UFI_UUID D_UUID NOT NULL,
    UF_UUID_LIST D_STR_8100)
RETURNS (
    RES_UFI_UUID_STR D_STR_CODE)
AS
declare variable res_str d_str_100;
declare variable ufi_owner d_uuid;
declare variable ufi_parent_uuid d_uuid;
declare variable ufi_position d_integer;
declare variable ufi_kind d_integer;
declare variable res_uuid d_uuid;
declare variable uf_uuid d_uuid;
declare variable fr_uuid d_uuid;
declare variable ufp_uuid d_uuid;
declare variable ufi_control d_str_code;
declare variable uf_is_param d_bool;
begin
  select ufi_parent_uuid, ufi_kind, ufi_owner, ufi_position, ufi_control
  from ux$fritem
  where ufi_uuid = :ufi_uuid
  into ufi_parent_uuid, ufi_kind, ufi_owner, ufi_position, ufi_control;

  res_ufi_uuid_str = '';

  if (ufi_kind < 0) then
  begin
    ufi_parent_uuid = ufi_uuid;
    ufi_position = 0;

    select ufi_position
    from ux$fritem
    where ufi_parent_uuid = :ufi_parent_uuid
    order by ufi_position desc
    rows 1
    into ufi_position;
  end

  for select trim(res_str)
  from uxp$split_str(:uf_uuid_list, ',')
  into :res_str
  do
  begin
    ufp_uuid = char_to_uuid(res_str);

    select ufp_ufr_uuid, ufp_uf_uuid, uf_is_param
    from ux$frparam, ux$field
    where ufp_uuid = :ufp_uuid and
          ufp_uf_uuid = uf_uuid
    into fr_uuid, uf_uuid, uf_is_param;

    if (ufi_control = 'vgrid' and uf_is_param = 1) then
      Continue;

    if (exists(select * from ux$fritem
    where ufi_parent_uuid = :ufi_parent_uuid and
          ufi_uf_uuid = :uf_uuid)) then
      Continue;

    ufi_position = ufi_position + 1;
    res_uuid = null;

    insert into ux$fritem (ufi_owner, ufi_parent_uuid, ufi_ufr_uuid, ufi_uf_uuid, ufi_kind, ufi_position)
    values (:ufi_owner, :ufi_parent_uuid, :fr_uuid, :uf_uuid, 0, :ufi_position)
    returning ufi_uuid
    into res_uuid;

    if (res_ufi_uuid_str = '') then
      res_ufi_uuid_str = uuid_to_char(res_uuid);
  end

  suspend;
end
