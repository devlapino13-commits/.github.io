CREATE PROCEDURE HP_KN_OPERBOOK_LIST(
    KC_ID ID_TYPE,
    REP_DATE DATE,
    K_ID ID_TYPE = null)
RETURNS (
    OL_ID D_STR_20,
    KOL_ID ID_TYPE,
    KOB_ID ID_TYPE,
    OL_DATE DATE,
    DT_ACC_ID ID_TYPE,
    DT_ACC_CODE D_STR_CODE,
    DT_ACC_NAME D_STR_LONG,
    KT_ACC_ID ID_TYPE,
    KT_ACC_CODE D_STR_CODE,
    KT_ACC_NAME D_STR_LONG,
    OB_SUMMA CURR_TYPE,
    OB_SUMMA_TAX_DT CURR_TYPE,
    OB_SUMMA_TAX_KT CURR_TYPE,
    OB_SUMMA_TAX CURR_TYPE,
    OL_MEMO D_STR_LONG,
    OB_DETAIL D_STR_LONG,
    REG_CODE_DT D_STR_20,
    REG_CODE_KT D_STR_20)
AS
declare variable tmp_kc_id integer;
begin
  if (kc_id = 0) then
    kc_id = null;
  if (k_id is null and
      kc_id is null) then
    exit;

  if (k_id is null and
      kc_id is not null) then
  begin
    select kc_from_k_id
    from knclient
    where kc_id = :kc_id
    into :k_id;
  end

  kol_id = null;
  for select kc_id
      from knclient
      where kc_from_k_id = :k_id and
            kc_amount_approved > 0 and
            (:kc_id is null or
            kc_id = :kc_id)
      into :tmp_kc_id
  do
  begin
    for select ol_id, kob_id, kob_ol_date, kob_acc_dt, dt.acc_name, kob_acc_kt, kt.acc_name, kob_summa,
               kob_summa_tax_dt, kob_summa_tax_kt, ol_memo, obd_detail
        from operlist,kn_operbook, ref_account dt, ref_account kt, kn_obdetail
        where kob_acc_dt = dt.acc_id and
              kob_acc_kt = kt.acc_id and
              kob_from_kc_id = :tmp_kc_id and
              kob_from_ol_id = ol_id and
              kob_detail_id = obd_id and
              kob_ol_is_active = 1 and
              (:rep_date is null or
              kob_ol_date = :rep_date)
        into :ol_id, kob_id, ol_date, dt_acc_id, dt_acc_name, kt_acc_id, kt_acc_name, ob_summa,
             ob_summa_tax_dt, ob_summa_tax_kt, ol_memo, ob_detail
    do
    begin
      ob_summa_tax = ob_summa_tax_dt;
      dt_acc_code = f_fmt_acc(:dt_acc_id);
      kt_acc_code = f_fmt_acc(:kt_acc_id);
      suspend;
    end
  end
end
