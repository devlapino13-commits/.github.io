CREATE PROCEDURE HP_COM_MEMO_ADD(
    CML_ID D_ID,
    MEMO_TEXT D_STR_LONG,
    MEMO_KIND D_SMALLINT,
    UN_ID D_ID)
RETURNS (
    RES_TEXT D_STR_255,
    RES_IS_ERROR D_SMALLINT)
AS
declare variable cma_id d_id;
declare variable lang d_str_code;
begin
  -- Процедура вставляет комментарий запросу КК
  res_is_error = 1;
  res_text = '';

  lang = f_get_lang_code(un_id, null);

  select cml_last_cma_id from com_list
  where cml_id = :cml_id
  into cma_id;

  if (cma_id is not null) then
  begin
    insert into com_memo (
      cmm_from_cma_id, cmm_memo_kind, cmm_memo, cmm_from_un_id)
    values (:cma_id, :memo_kind, :memo_text, :un_id);

    res_text = f_lang(lang, 'Комментарий успешно добавлен');
    res_is_error = 0;
  end
  else
    res_text = f_lang(lang, 'Ошибка добавления комментария');

  suspend;
end
