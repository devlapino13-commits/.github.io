CREATE PROCEDURE HP_UX_SOURCE_FIELDS(
    SOURCE_NAME D_STR_CODE)
RETURNS (
    FIELD_NAME D_STR_CODE,
    PARAM_KIND D_SMALLINT)
AS
begin
  source_name = upper(source_name);

  param_kind = 1;
  for select lower(trim(r.rdb$field_name))
  from rdb$fields f, rdb$relation_fields r
  where f.rdb$field_name = r.rdb$field_source and
        r.rdb$system_flag = 0 and
        r.rdb$relation_name = :source_name
  order by r.rdb$field_position
  into field_name
  do
    suspend;

  if (row_count = 0) then
  begin
    for select lower(trim(rdb$parameter_name)), rdb$parameter_type
    from rdb$procedure_parameters
    where rdb$procedure_name = :source_name and
          rdb$package_name is null
    order by rdb$parameter_type, rdb$parameter_number
    into field_name, param_kind
    do
      suspend;
  end
end
