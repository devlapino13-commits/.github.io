CREATE PROCEDURE EXPORT_1C_HR(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    HO_ID_STR VARCHAR(20),
    PS_ID INTEGER,
    C_NAME_FIO VARCHAR(100),
    C_SEX_STR VARCHAR(20),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(100),
    C_PASS_DATE DATE,
    C_PASS_WHO VARCHAR(255),
    C_FULL_ADDRESS VARCHAR(255),
    C_INN VARCHAR(20),
    C_DOC_TYPE INTEGER,
    PK_NAME VARCHAR(100),
    C_CITIZEN VARCHAR(100),
    C_NATION VARCHAR(100),
    HO_DATE DATE,
    HO_ORDER_NO VARCHAR(20),
    HO_REASON VARCHAR(255),
    SAWT_NAME VARCHAR(100),
    JL_NAME VARCHAR(100),
    REG_UNICODE INTEGER,
    REG_NAME VARCHAR(100),
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    HRC_NAME VARCHAR(255),
    C_ID_STR VARCHAR(20),
    C_REG_NO VARCHAR(20),
    C_ADDRESS_FACT VARCHAR(255),
    C_ADDRESS_INFO VARCHAR(255))
AS
declare variable c_sex integer;
declare variable ho_id decimal(18,0);
declare variable c_id decimal(18,0);
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ho_id, ps_id, c_name_fio, c_sex, c_birth_date, c_pass_no, c_pass_date, c_pass_who, adl_address_str, c_inn,
             c_doc_type, pk_name, d1.rd_name, d2.rd_name, ho_date, ho_order_no, ho_reason, sawt_name, jl_name,
             reg_unicode, reg_code, ps_date_begin, ps_date_end, hrc_name, ps_from_c_id, c_reg_no

      from clients, hr_orders, hr_pstatus, passkind, refdata d1, refdata d2, sa_worktype, joblist, region, hr_class, addrclient, addrlist
      where c_doc_type = pk_id and
            c_citizen = d1.rd_id and
            c_nation = d2.rd_id and
            c_id = ps_from_c_id and
            ps_from_ho_id = ho_id and
            ho_order_type = sawt_id and
            jl_id = ps_from_jl_id and
            reg_unicode = ps_from_reg_id and
            hrc_id = ps_class and
            ps_date_begin >= :rep_date_begin and
            ps_date_begin <= :rep_date_end and
            ps_is_on_staff = 1 and
            adc_from_c_id = c_id and
            adc_address_type = 1 and
            adc_from_adl_id = adl_id
      order by ho_date, ho_id
      into :ho_id, :ps_id, :c_name_fio, :c_sex, :c_birth_date, :c_pass_no, :c_pass_date, :c_pass_who, :c_full_address,
           :c_inn, :c_doc_type, :pk_name, :c_citizen, :c_nation, :ho_date, :ho_order_no, :ho_reason, :sawt_name,
           :jl_name, :reg_unicode, :reg_name, :ps_date_begin, :ps_date_end, :hrc_name, :c_id, :c_reg_no
  do
  begin
    c_id_str = 'N' || cast(c_id as varchar(20));
    ho_id_str = 'N' || cast(ho_id as varchar(20));
    if (c_sex = 0) then
      c_sex_str = 'Жен';
    else
      c_sex_str = 'Муж';
    if (hrc_name = 'Не указан') then
      hrc_name = '';
    c_address_fact = :c_full_address;
    c_address_info = :c_full_address;
    suspend;
  end

end
