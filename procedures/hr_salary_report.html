CREATE PROCEDURE HR_SALARY_REPORT(
    REG_ID INTEGER,
    DEP_ID INTEGER,
    JL_ID INTEGER,
    DD1 DATE,
    DD2 DATE)
RETURNS (
    PS_ID INTEGER,
    PS_FROM_HO_ID DECIMAL(18,0),
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    PS_PERIOD_MONTH INTEGER,
    PS_DATE_END_IN_ORDER DATE,
    PS_FROM_REG_ID INTEGER,
    PS_FROM_DEP_ID INTEGER,
    PS_FROM_JL_ID INTEGER,
    PS_FROM_C_ID DECIMAL(18,0),
    PS_IS_ON_STAFF_TEXT VARCHAR(20),
    PS_IS_WORKING_TEXT VARCHAR(20),
    PS_SALARY DECIMAL(18,4),
    PS_REG_NAME VARCHAR(100),
    PS_DEP_NAME VARCHAR(100),
    PS_JL_NAME VARCHAR(100),
    PS_C_NAME_FIO VARCHAR(100),
    PS_ORDER_TYPE INTEGER,
    PS_ORDER_BASE_TYPE INTEGER,
    PS_TABEL_NO VARCHAR(20),
    PS_DATE_HIRE DATE,
    PS_DATE_HIRE_OFFICE DATE,
    PS_DATE_GONE_OFFICE DATE,
    PS_C_SEX_TEXT VARCHAR(20),
    PS_IS_ON_STAFF SMALLINT,
    PS_IS_WORKING SMALLINT,
    PS_SALDO_BEGIN DECIMAL(18,4),
    PS_SALDO_END DECIMAL(18,4))
AS
declare variable c_id_prior decimal(18,0);
declare variable reg_id_prior integer;
declare variable dd date;
begin
  c_id_prior = 0;
  for select ps_id, ps_from_ho_id, ps_date_begin, ps_date_end, ps_period_month, ps_date_end_in_order, ps_from_reg_id,
             ps_from_dep_id, ps_from_jl_id, ps_from_c_id, ps_is_on_staff_text, ps_is_working_text, ps_salary,
             ps_reg_name, ps_dep_name, ps_jl_name, ps_c_name_fio, ps_order_type, ps_order_base_type, ps_tabel_no,
             ps_date_hire, ps_c_sex_text, ps_is_on_staff, ps_is_working, ps_date_hire_office, ps_date_gone_office
      from hr_all_personal_new_pstatus(:reg_id, :dd1, :dd2)
      into :ps_id, :ps_from_ho_id, :ps_date_begin, :ps_date_end, :ps_period_month, :ps_date_end_in_order,
           :ps_from_reg_id, :ps_from_dep_id, :ps_from_jl_id, :ps_from_c_id, :ps_is_on_staff_text, :ps_is_working_text,
           :ps_salary, :ps_reg_name, :ps_dep_name, :ps_jl_name, :ps_c_name_fio, :ps_order_type, :ps_order_base_type,
           :ps_tabel_no, :ps_date_hire, :ps_c_sex_text, :ps_is_on_staff, :ps_is_working, :ps_date_hire_office,
           :ps_date_gone_office
  do
  begin
    if (c_id_prior = ps_from_c_id) then
      ps_date_end = dd - 1;
    ps_saldo_begin = 0;
    select rep_salary_saldo
    from hp_get_employer_salary_saldo(:ps_from_c_id, (:dd1 - 1))
    into :ps_saldo_begin;

    ps_saldo_end = 0;
    select rep_salary_saldo
    from hp_get_employer_salary_saldo(:ps_from_c_id, :dd2)
    into :ps_saldo_end;

    suspend;
    dd = ps_date_begin;
    c_id_prior = ps_from_c_id;
    reg_id_prior = ps_from_reg_id;
  end
end
