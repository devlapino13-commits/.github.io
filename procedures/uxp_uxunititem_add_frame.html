CREATE PROCEDURE UXP$UXUNITITEM_ADD_FRAME(
    UNF_UUID D_UUID)
AS
declare variable max_index d_integer;
declare variable max_position d_integer;
declare variable uni_uuid d_uuid;
begin
  for select unf_owner, unf_code, unf_name
  from ux$unitframe
  where unf_uuid = :unf_uuid
  as cursor cur
  do
  begin
    if (exists(select * from ux$unititem
               where uni_unt_uuid = cur.unf_owner and
                     uni_code = cur.unf_code || '_frame')) then
      Continue;

    select max(uni_index), max(uni_position)
    from ux$unititem
    where uni_parent_uuid = cur.unf_owner and
          uni_kind = 1
    into max_index, max_position;
  
    if (max_index is null) then
      max_index = 0;
    else
      max_index = max_index + 1;
  
    if (max_position is null) then
      max_position = 0;
    else
      max_position = max_position + 1;
  
    update ux$unititem
    set uni_position = uni_position + 1
    where uni_parent_uuid = cur.unf_owner and
          uni_position >= :max_position
    order by uni_position desc;
  
    update ux$unititem
    set uni_index = uni_index + 1
    where uni_unt_uuid = cur.unf_owner and
          uni_position >= :max_position
    order by uni_index desc;

    insert into ux$unititem (uni_unt_uuid, uni_parent_uuid, uni_code, uni_name, uni_kind, uni_caption, uni_control,
                             uni_add_info, uni_index, uni_position, uni_caption_align, uni_group_direction)
    values (cur.unf_owner, cur.unf_owner, cur.unf_code || '_frame', 'TPanel', -1, 'Фрейм ' || cur.unf_name, 'group',
            'Frame added', :max_index, :max_position, -1, 1)
    returning uni_uuid
    into uni_uuid;

    update ux$unitframe
    set unf_uni_uuid = :uni_uuid
    where unf_uuid = :unf_uuid;
  end
end
