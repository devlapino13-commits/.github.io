CREATE PROCEDURE HP_SUD_NEW_REQUEST(
    K_ID D_ID,
    UN_ID D_ID,
    MEMO D_STR_LONG = null)
RETURNS (
    SMR_ID D_ID)
AS
declare variable reg_id d_integer;
declare variable res_ok d_integer;
begin
--  if (un_id is null) then
--    un_id = (select mm_un_id from mainmenu);

  if (un_id is null) then
    exception e_message 'Пользователь не определен!';

  if (exists(select smr_id from sud_mainreq jmr, sud_mainlog jml
            where jml.sul_id = jmr.smr_last_sul_id
              and jmr.smr_from_k_id = :k_id
              and jml.sul_status in (-1,0,1) -- активный
  )) then
    exception e_message 'Договор № ' || (select k_kontract_no from kontract where k_id = :k_id) || ' уже добавлен';

  select a.kal_new_reg_id
  from knaktlist a, kontract
  where k_id = :k_id and
        k_from_kf_id = a.kal_from_kf_id and
        a.is_delete = 0 and
        a.kal_date <= current_date and
        a.kal_date_end >= current_date
  into reg_id;

  if (not exists(select * from aarep_region_user_tmp(:un_id) where reg_list = :reg_id)) then
    exception e_message 'У вас нет доступа к офису договора № ' || (select k_kontract_no from kontract where k_id = :k_id);

  insert into sud_mainreq (smr_from_k_id, smr_from_un_id)
  values (:k_id, :un_id)
  returning smr_id
  into smr_id;

  --записать клиентов
  insert into sud_loanclient (slc_sum_os, slc_sum_perc, slc_sum_fine, slc_sum_other, slc_sum_all, slc_from_smr_id, slc_sum_date, slc_address, slc_from_kc_id)
  select cur_cl_os, cur_cl_perc, cur_cl_fine, 0, cur_cl_all, :smr_id, current_date, c_address_yur_disb,
         (select kc_id from knclient where kc_from_k_id = :k_id and kc_from_c_id = cz_from_c_id)
  from print_client_zadol(:k_id);

  select res_ok from hp_sud_add_memo(:smr_id, :memo, :un_id)
  into res_ok;

  suspend;
end
