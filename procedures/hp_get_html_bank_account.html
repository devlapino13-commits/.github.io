CREATE PROCEDURE HP_GET_HTML_BANK_ACCOUNT(
    OBJ_KIND D_INTEGER,
    OBJ_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_HTML D_STR_4096)
AS
declare variable res_link d_str_long;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable b_id d_str_10;
declare variable bc_account_no d_str_255;
begin
    -- RFCRU-5314
    -- Процедура отображает окно в МП по добавлению/изменению банковского счёта для ИП
    -- Сам шаблон окна был собран напрямую в приложении через Оксану.
    -- Окно вернёт два поля для ввода (БИК и Расчетный Счет), которые попадут далее в процедуру fm_save_ie_bank_details

    res_html = '';
    res_link = '';

    if (f_objkind(:obj_kind, 'code') = 'kontract') then
        select kc.kc_from_c_id
        from knclient kc
        where kc.kc_from_k_id = :obj_id
            and kc.kc_amount_approved > 0
        into :c_id;

    else if (f_objkind(:obj_kind, 'code') = 'miniapp') then
        select mac.mac_from_c_id
        from miniappclient mac
        where mac.mac_from_ma_id = :obj_id
            and mac.mac_amount > 0
        into :c_id;

    select c.c_name_fio
    from clients c
    where c.c_id = :c_id
    into :c_name_fio;

    select b.b_id, f_mask_text(bc.bc_account_no, 8, 4)
    from bankclient bc
    inner join banks b on bc.bc_from_b_id = b.b_id
        and b.is_delete = 0
        and b.b_corr_acc is not null
    where bc.bc_from_c_id = :c_id
        and bc.bc_is_main = 1
        and bc.bc_is_active = 1
    into :b_id, :bc_account_no;


    -- Если счёт (bc_account_no) не был найден, то он просто не отобразиться и будет красоваца ссылка (Внести реквизиты)
    -- Иначе будет номер счёта и ссылка (Изменить)
    res_link = f_uri_open(iif(b_id is not null, 'Изменить', 'Внести реквизиты'), 'save_ie_bank_details', 'clients', :c_id, 'Банковские реквизиты', f_format('fio=%1&hint=40802810', :c_name_fio), :un_id);
    res_html = iif
        (
            b_id is not null, 
            blob_append(res_html, f_html_table_row(c_name_fio, f_format('Счёт ИП: %1, %2', :bc_account_no, :res_link), '', '', -1, -1, -1, -1, 2, '')),
            blob_append(res_html, f_html_table_row(c_name_fio, :res_link, '', '', -1, -1, -1, -1, 2, ''))
        );

    suspend;
end
