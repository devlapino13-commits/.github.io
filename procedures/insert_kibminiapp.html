CREATE PROCEDURE INSERT_KIBMINIAPP(
    MA_FROM_REG_ID INTEGER,
    MA_ID DECIMAL(18,0),
    MA_FROM_UN_ID DECIMAL(18,0),
    IS_ONLINE D_BOOL = 0)
RETURNS (
    SEND_KOL INTEGER)
AS
declare variable c_id decimal(18,0);
declare variable db_id integer;
declare variable ma_id_tmp integer;
declare variable pr_code d_str_100;
declare variable mac_is_poruch d_integer;
declare variable kbm_id d_id;
begin
  send_kol = 0;

  :db_id = f_get_db_id();
  if (db_id not in (39801, 41701)) then
  begin
    suspend;
    exit;
  end

  select k_from_ma_app_id
  from kontract
  where k_id = :ma_id
  into :ma_id_tmp;
  if (ma_id_tmp is null) then
    ma_id_tmp = :ma_id;

  for select mac_from_c_id, pr.pr_code, mac.mac_is_poruch
      from miniappclient mac
      inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      where ma.ma_date_end = '31.12.9999' and
            mac_from_ma_id = :ma_id_tmp and
            (mac_amount > 0 or
            mac_is_poruch = 1 or
            mac_is_poruch = -1)
      into :c_id, :pr_code, :mac_is_poruch
  do
  begin
    if (not(exists(select *
                   from kibminiapp
                   where kbm_from_c_id = :c_id and
                         kbm_ma_id = :ma_id_tmp))) then
    begin
      if (pr_code containing '.sugarmill.' and
          mac_is_poruch = -1 and
          f_asp_signed(:c_id, current_date) <> 1) then
        continue;

    if ((db_id = 41701 and mac_is_poruch not in (-1, 1)) or db_id <> 41701) then
    begin
      if (not exists(select *
                     from kibminiapp
                     where kbm_from_c_id = :c_id and
                           cast(kbm_datetime as date) between current_date - 1 and current_date)) then
        insert into kibminiapp (kbm_id, kbm_from_c_id, kbm_ma_from_reg_id, kbm_ma_id, kbm_pr_id, kbm_is_checked,
                                kbm_pr_id_datetime, kbm_checked_datetime, kbm_ver_info, kbm_err_text, kbm_from_un_id)
        values (-1, :c_id, :ma_from_reg_id, :ma_id_tmp, '', 0, null, null, 0, '', :ma_from_un_id)
        returning kbm_id
        into :kbm_id;
        send_kol = send_kol + 1;
    end

      if (is_online = 1) then
        insert into kibminiapp_online (kbm_id)
        values (:kbm_id);

      if (db_id = 41701) then  --отправка пакетов справок при отправке запроса в киб RFCKG-1523
      begin
        insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json)
        values (:c_id, 3, 11, :ma_id_tmp, 1); -- Справка о фактическом адресе

        insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json)
        values (:c_id, 4, 11, :ma_id_tmp, 1); -- Справка о составе семьи

        insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json)
        values (:c_id, 7, 11, :ma_id_tmp, 1); -- Справка о количестве скота

        insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json)
        values (:c_id, 8, 11, :ma_id_tmp, 1); -- Проверка в РФ БД
      end
    end

    -- RFCKG-4812
    if (db_id = 41701 and
        mac_is_poruch not in (-1, 1) and -- RFCKG-5144
        not(exists(select *
                   from extinfolist eil
                   inner join extinfocfg eic on eic.eic_id = eil.eil_from_eic_id
                   where eil.eil_from_c_id = :c_id and
                         eil.eil_date between current_date - 1 and current_date and
                         eil.eil_status in (0,2) and
                         eic.eic_code = 'ss_cib_report'))) then
        insert into extinfolist (eil_from_c_id, eil_from_eic_id, eil_object_kind, eil_object_id, eil_is_json, eil_from_un_id)
        values (:c_id, 15, 11, :ma_id_tmp, 1, :ma_from_un_id); -- Отправка в Safe&Sound
  end
  suspend;
end
