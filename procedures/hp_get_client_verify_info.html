CREATE PROCEDURE HP_GET_CLIENT_VERIFY_INFO(
    C_ID ID_TYPE,
    IS_HTML INTEGER = 0)
RETURNS (
    INF_MEMO D_BLOB_TEXT,
    ACM_MEMO D_BLOB_TEXT,
    SFT_MEMO D_BLOB_TEXT,
    DT_VER TIMESTAMP,
    SUM_PAYMENT CURR_TYPE,
    SUM_DEBT CURR_TYPE,
    DAYS_DELAY STR_255,
    FSSP_DATA STR_255,
    EXTB_DATA STR_255,
    FMS_DATA STR_255,
    EFRSB_DATA STR_255,
    COMMENT STR_255,
    CR_LEVEL INTEGER,
    CL_ID ID_TYPE)
AS
declare variable dt_app timestamp;
declare variable reg_name str_255;
declare variable c_fio str_255;
declare variable pr_name varchar(50);
declare variable cycle integer;
declare variable star integer;
declare variable no_data str_255;
declare variable edoc_obj_kind d_str_100;
declare variable edoc_cfg_id d_integer;
declare variable edoc_ma_id d_integer;
begin
  no_data = 'нет данных';
  inf_memo = '';
  acm_memo = '';
  sum_payment = 0;
  sum_debt = 0;
  days_delay = no_data;
  fssp_data = no_data;
  extb_data = no_data;
  fms_data = no_data;
  efrsb_data = no_data;
  comment = '';
  cr_level = -1;
  dt_ver = null;

  dt_ver = coalesce(:dt_ver, current_timestamp);
  -- дата эл.заявки, ОП, ВИПС

  edoc_obj_kind = f_objkind('miniapp', 'id');
  edoc_cfg_id = (select edc_id from edoc_cfg where edc_code = 'loan.application');

  select cast(iif(eds_status = 1, eds.eds_status_datetime, null) as d_datetime), edl.edl_object_id
  from edoc_list edl
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
  where edl_object_kind =  :edoc_obj_kind and
      edl_from_edc_id = :edoc_cfg_id and
      eds.eds_from_c_id = :c_id  and
      edl.edl_status >= 0
  order by edl_id desc
  rows 1
  into :dt_app, :edoc_ma_id;

  if (edoc_ma_id is not null) then
  begin
    select first 1 reg_code || ' ' || reg_name, c_name_fio, pr.pr_name
    from miniapp maa
    inner join knaktlist on kal_from_kf_id = ma_from_kf_id and
          :dt_ver between kal_date and kal_date_end and
          is_delete = 0 and
          maa.ma_date_end = '31.12.9999'
    inner join region on reg_unicode = kal_new_reg_id
    inner join product pr on pr.pr_id = ma_from_pr_id -- добавлен по задаче jira 2126
    inner join clients on c_id = kal_new_kf_officer
    where maa.ma_id = :edoc_ma_id
    into :reg_name, :c_fio, :pr_name;
  end else
  begin
    select first 1 mv_confsms_date, reg_code || ' ' || reg_name, c_name_fio, pr.pr_name
    from miniapp_verify
    inner join miniapp on ma_id = mv_from_ma_id
    inner join knaktlist on kal_from_kf_id = ma_from_kf_id and
          :dt_ver between kal_date and kal_date_end and
          is_delete = 0
    inner join region on reg_unicode = kal_new_reg_id
    inner join product pr on pr.pr_id = ma_from_pr_id -- добавлен по задаче jira 2126
    inner join clients on c_id = kal_new_kf_officer
    where mv_from_c_id = :c_id
    order by mv_id desc
    into :dt_app, :reg_name, :c_fio, :pr_name;
  end
  -- цикл
  cycle = (select count(*)
           from kontstate, knclient
           where ks_ol_is_delete = 0 and
                 ks_state = 1 and
                 ks_date_begin < cast(coalesce(:dt_app, current_timestamp) as date) --:dt_ver
                 and
                 ks_from_k_id = kc_from_k_id and
                 kc_amount_approved > 0 and
                 kc_from_c_id = :c_id);
  -- звёздность
  star = coalesce((select c_last_status
                   from hp_get_client_status(:c_id, :dt_ver)), 0);
  -- паспортные данные
  select res_object_detail
  from hp_actionlist_get_passdata(:c_id)
  into inf_memo;
  inf_memo = 'Дата эл.заявки: <b>' || iif(dt_app is null, '', f_datetimetostr(dt_app)) || '</b><br>' || 'Подразделение: <b>' || iif(reg_name is null, '', reg_name) || '</b><br>' || 'ФИО ВИПС: <b>' || iif(c_fio is null, '', c_fio) || '</b><br>' || 'Цикл: <b>' || iif(cycle is null, 1, cycle + 1) || '</b>; ' || 'Звёздность: <b>' || iif(star = 0, 'без статуса', rpad('', star, '*')) || '</b>; ' || 'Продукт: <b>' || :pr_name || '</b>;<br><br>' || inf_memo;

  -- стоп-факторы
  select list(ob_info, '<br>')
  from (select '<b>' || f_datetimetostr(obm_datetime) || '</b><br>' || f_get_verify_memo(obm_memo, 'reason_memo') || '<br>' as ob_info
        from actionlist acl
        inner join objectprops on opr_object_id = acl.acl_object_id and
              opr_object_kind = acl.acl_object_kind and
              opr_subject_id = f_get_rd_id('ActionList.LoanStopFactors')
        inner join actionlist acl2 on acl.acl_id = acl2.acl_parent_id and
              acl2.acl_object_kind = 0 and
              acl2.acl_object_id = :c_id
        inner join objectmemo on obm_from_opr_id = opr_id
        where acl.acl_object_kind = 9
        order by obm_datetime desc)
  into sft_memo;

  cl_id = c_id;
  sft_memo = coalesce(sft_memo, '');
  suspend;
end
