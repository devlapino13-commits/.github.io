CREATE PROCEDURE OKT_DISTRIBUTE_KONTRACTS(
    PROJECT_CODE VARCHAR(100),
    REP_DATE DATE)
AS
declare variable CURR_OPER_ID ID_TYPE;
declare variable CURR_OPER_CLIENT_CNT D_INTEGER;
declare variable CURR_OPER_LANGS D_STR_20;
declare variable LAST_OPER_ID ID_TYPE;
declare variable LAST_OPER_CLIENT_CNT D_INTEGER;
declare variable LAST_OPER_LANGS D_STR_20;
declare variable DIFF D_INTEGER;
declare variable K_ID ID_TYPE;
declare variable LINKED_CLIENT_CNT D_INTEGER;
declare variable GROUP_LNG D_STR_20;
declare variable GROUP_CLIENT_CNT D_INTEGER;
declare variable FREE_GROUP_CNT D_INTEGER;
declare variable IS_OPERATOR_PROCESSED D_INTEGER;
declare variable ALL_OPERATORS_CNT D_INTEGER;
declare variable PROCESSED_OPERATORS_CNT D_INTEGER;
declare variable PROJECT_RD_ID D_INTEGER;
declare variable V_PROJECT_CODE D_STR_100;
declare variable RESULT_TEXT D_STR_255;
declare variable IS_FIRST D_SMALLINT;
begin

  if (coalesce(project_code, '') = '') then
    project_code = 'OKTELL.Projects.AllProjs';

  for select trim(pr_code)
      from (select 1 num, 'OKTELL.Projects.Sirena1' pr_code
            from rdb$database
            union
            select 3, 'OKTELL.Projects.Pantera'
            from rdb$database
            union
            select 6, 'OKTELL.Projects.ks_finance_2'
            from rdb$database
            union
            select 7, 'OKTELL.Projects.Investigation_And_Recovery'
            from rdb$database)
      where :project_code = 'OKTELL.Projects.AllProjs' or
            :project_code = pr_code
      order by num
      into :v_project_code
  do
  begin
    ---- подготовка исходных данных

    delete from okt_clients_tmp where 1 = 1;
    delete from okt_groups_tmp where 1 = 1;
    delete from okt_users_tmp where 1 = 1;

    select r.rd_id
    from refdata r
    where r.rd_main_code = 'OKTELL.Projects' and
          r.rd_code = :v_project_code
    into project_rd_id;

    if (v_project_code containing ('Sirena')) then
      insert into okt_clients_tmp (ocl_k_id)
      select distinct k_id
      from okt_sirena_list(:rep_date, :v_project_code, '', '', '', '', '', 0, 0, 0, 1);
    else
      insert into okt_clients_tmp (ocl_k_id)
      select distinct k_id
      from okt_pantera_list(:rep_date, :v_project_code, '', '', '', '', 0, 0, 0, 1);

    select count(1)
    from okt_clients_tmp
    into diff;

    if (diff > 0) then
    begin

      insert into okt_groups_tmp (ogp_k_id, ogp_client_count, ogp_group_lang)
      select ocl_k_id, count(*), 'RU'
      from okt_clients_tmp
      group by ocl_k_id;

      insert into okt_users_tmp (ous_un_id, ous_lang, ous_old_cnt, ous_cnt, ous_diff, ous_kg, ous_uz, ous_tj, ous_ru)
      select ul.un_id, 'RU',
             coalesce(t.cnt, 0) old_cnt,
             coalesce(t.cnt, 0) cnt,
             0,
             0,
             0,
             0,
             0
      from oktell_user_list_with_access ul
      inner join --left join   igusev 15.02.2019 убрал left join чтобы быть уверенным что на оперторов у которых нет групп, новые группы не будут автоматом распределены
            (select pantk.pant_from_un_id, count(cl.c_id) cnt
             from pantera_kontracts pantk
             inner join oksummary oks on oks.oks_from_k_id = pantk.pant_from_k_id
             inner join knclient kncl on kncl.kc_from_k_id = oks.oks_from_k_id and
                   kncl.kc_amount_approved > 0
             inner join clients cl on cl.c_id = kncl.kc_from_c_id
             where pantk.pant_date_end = '30.12.9999' and
                   :rep_date between oks.oks_date_begin and oks.oks_date_end and
                   pantk.pant_rd_project_type = :project_rd_id
             group by pantk.pant_from_un_id) t on t.pant_from_un_id = ul.un_id
      where (:v_project_code = 'OKTELL.Projects.Sirena1' and ul.sirena_1 = 1) or
            (:v_project_code = 'OKTELL.Projects.Pantera' and ul.pantera_1 = 1) or
            (:v_project_code = 'OKTELL.Projects.ks_finance_2' and ul.ks_finance_2 = 1) or
            (:v_project_code = 'OKTELL.Projects.Investigation_And_Recovery') and
            ul.investigation_and_recovery =1;

      insert into okt_users_tmp (ous_un_id, ous_lang, ous_old_cnt, ous_cnt, ous_diff, ous_kg, ous_uz, ous_tj, ous_ru)
      select 1000000000001, '', 0, 1000000, 0, 0, 0, 0, 0
      from rdb$database;

-------------------------------------------------------------------------------------

      select count(1) -- кол-во операторов
      from okt_users_tmp
      where ous_cnt > 0 ------------------------------------------------------------------------------------------------------ временно отключаем
      into all_operators_cnt;

      select count(1) -- кол-во свободных групп
      from okt_groups_tmp
      into free_group_cnt;

      processed_operators_cnt = 0;

      -- пока есть свободные группы и не пройдемся по всем операторам
      while (free_group_cnt > 0 and processed_operators_cnt <= all_operators_cnt) do
      begin
        is_first = 1;
        last_oper_client_cnt = 0;
        is_operator_processed = 0;
        for select ous_un_id, ous_lang, ous_cnt
            from okt_users_tmp
            where ous_cnt > 0 ------------------------------------------------------------------------------------------------------ временно отключаем
            order by ous_cnt
            into curr_oper_id, curr_oper_langs, curr_oper_client_cnt
        do
        begin
          processed_operators_cnt = processed_operators_cnt + 1;

          if (is_first = 1) then -- запоминаем данные по 1-му оператору
          begin
            last_oper_id = curr_oper_id;
            last_oper_client_cnt = curr_oper_client_cnt;
            last_oper_langs = curr_oper_langs;
            is_first = 0;
            continue;
          end

          linked_client_cnt = 0;
          diff = curr_oper_client_cnt - last_oper_client_cnt; -- находим разницу кол-ва клиентов между текущим и предыдущим оператором

          while (diff >= linked_client_cnt) do -- если разница есть то пробуем подвязать группы предыдущему оператору
          begin
            k_id = null;
            select ogp_k_id, -- свободные группы с подходящими для оператора языками
                   ogp_client_count,
                   'RU'
            from okt_groups_tmp og
            where ogp_un_id is null
            rows 1
            into :k_id, :group_client_cnt, :group_lng;

            if (k_id is not null) then
            begin
              result_text = (select mess_str
                             from okt_sirena_make_call(0, :rep_date, 'block', 'SirenaTransferReason.RobotBlock', :v_project_code, 0, :k_id, 0, '', 0, null, :last_oper_id, null, 0, null, null, null, 0));

              linked_client_cnt = linked_client_cnt + group_client_cnt; -- кол-во добавленных клиентов
              free_group_cnt = free_group_cnt - 1; -- кол-во свободных групп

              update okt_users_tmp
              set ous_cnt = ous_cnt + :group_client_cnt,
                  ous_diff = ous_diff + :group_client_cnt,
                  ous_ru = ous_ru + iif(:group_lng = 'RU', :group_client_cnt, 0)
              where ous_un_id = :last_oper_id;

              update okt_groups_tmp
              set ogp_un_id = :last_oper_id,
                  ogp_results = :result_text
              where ogp_k_id = :k_id;

              is_operator_processed = 1;
            end
            else
              leave; -- если нет подходящих свободных групп
          end

          if (is_operator_processed > 0) then -- если оператор обработан (т.е данные по операторам изменились)
          begin -- то начинаем новый цикл по операторам
            processed_operators_cnt = 0;
            leave;
          end
          else
          begin
            last_oper_id = curr_oper_id; -- если оператор не обработан, то продолжаем цикл по операторам
            last_oper_client_cnt = curr_oper_client_cnt;
            last_oper_langs = curr_oper_langs;
          end
        end -- операторы
      end -- свободные группы

      result_text = cast('now' as timestamp);
    end -- если есть новые группы
  end ------ проекты
end
