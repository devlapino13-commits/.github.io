CREATE PROCEDURE MC_SAVE_ONLINE_APP(
    MCL_ID D_ID NOT NULL,
    APP_ID D_INTEGER,
    APP_JSON D_STR_8100,
    CARD_TYPE D_SMALLINT = 0)
RETURNS (
    ONLINE_APP_ID D_INTEGER,
    ERROR_TEXT D_STR_255)
AS
declare variable C_ID D_ID;
declare variable OA_MCL_ID D_ID;
declare variable OA_FROM_C_ID D_ID;
declare variable OA_STATUS D_SMALLINT;
declare variable OA_DATE D_DATE;
declare variable OA_ID D_INTEGER;
declare variable CARD_ID D_ID;
declare variable AMOUNT D_INTEGER;
declare variable APP_AMOUNT D_INTEGER;
declare variable MCL_LANG_CODE D_STR_LANG;
begin
  error_text = '';
  app_json = coalesce(app_json, '');
  online_app_id = app_id;

  select ml.mcl_from_c_id, ml.mcl_lang_code
  from mc_login ml
  where ml.mcl_id = :mcl_id
  into :c_id,
       :mcl_lang_code;

  if (card_type = 0) then
  begin
      select cal_card_id
      from get_active_mbulak_card(:c_id)
      into card_id;
    
      if (coalesce(:card_id, -999) = -999) then
      begin
        error_text = f_lang(:mcl_lang_code, 'Отсутствует активная карта Мбулак');
        suspend;
        exit;
      end
  end

  select amount
  from mc_get_loan_limit(:mcl_id)
  into amount;

  if (coalesce(:app_id, 0) <> 0) then
  begin
    select oa.oa_json
    from online_app oa
    where oa.oa_from_mcl_id = :mcl_id and
          oa.oa_status >= 0
    order by oa.oa_datetime desc
    rows 1
    into app_json;
  end

  app_amount = f_json_blob_get_value(:app_json, 'amount', 0);

  if (coalesce(:amount, 0) = 0) then
  begin
    error_text = f_lang(:mcl_lang_code, 'Выдача невозможна. Запрашиваемая сумма (') || :app_amount || f_lang(:mcl_lang_code, ' сом) превышает доступный кредитный лимит ') || :amount || f_lang(:mcl_lang_code, ' сом).');
    suspend;
    exit;
  end
  else
  begin
    if (:app_amount > :amount) then
    begin
      error_text = f_lang(:mcl_lang_code, 'Выдача невозможна. Пожалуйста, Запрашиваемая сумма (') || :app_amount || f_lang(:mcl_lang_code, ' сом) превышает доступный кредитный лимит ') || :amount || f_lang(:mcl_lang_code, ' сом).');
      suspend;
      exit;
    end
  end

  if (app_id is not null) then
  begin
    select oa.oa_id, oa.oa_from_mcl_id, oa.oa_from_c_id, oa.oa_status, oa.oa_date
    from online_app oa
    where oa.oa_id = :app_id
    into :oa_id, :oa_mcl_id, :oa_from_c_id, :oa_status, :oa_date;
  end

  if (oa_id is not null) then
  begin
    if (oa_mcl_id <> mcl_id) then
      error_text = f_lang(:mcl_lang_code, 'Пользователь, который передается сейчас, отличается от изначального в заявке. Это недопустимо');
    else
    if (oa_from_c_id <> c_id) then
      error_text = f_lang(:mcl_lang_code, 'Клиент, отличается от изначального в заявке. Это недопустимо');
    else
    if (oa_date <> current_date) then
      error_text = f_lang(:mcl_lang_code, 'Заявка может быть отредактирваона только текущей датой');

    if (error_text <> '') then
    begin
      suspend;
      exit;
    end

    if ((oa_status = 0) and (app_json <> '')) then
    begin
      update online_app oa
      set oa.oa_json = :app_json
      where oa.oa_id = :oa_id;
    end
  end
  else
  begin
    insert into online_app (oa_from_mcl_id, oa_json, oa_card_type, oa_mbulak_card_id)
    values (:mcl_id, :app_json, :card_type, :card_id)
    returning oa_id
    into :online_app_id;
  end

  suspend;
end
