CREATE PROCEDURE PRINT_IDJARA_GOODS_CLIENT(
    OBJ_ID D_ID,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    K_KONTRACT_NO D_STR_20,
    PERIOD D_INTEGER,
    K_DISB_DATE D_DATE,
    CLIENT_FIO D_STR_100,
    CLIENT_INN D_STR_20,
    CLIENT_ADDRESS D_STR_255,
    QR_CODE D_STR_LONG,
    ML_NAME D_STR_100,
    ML_MODEL D_STR_255,
    KML_COST DECIMAL(18,2),
    KML_COUNT DECIMAL(18,2),
    KML_TOTAL DECIMAL(18,2),
    SUM_KML_TOTAL DECIMAL(18,2),
    KML_SELL_TOTAL_NSP DECIMAL(18,2),
    CLIENT_EDOC_DATETIME D_DATETIME,
    REG_NAME D_STR_100,
    KMI_IMEI D_STR_20)
AS
declare variable OBJ_KIND D_INTEGER;
declare variable C_ID D_ID;
declare variable EDL_ID D_ID;
declare variable REG_ID D_ID;
declare variable K_ID D_ID;
declare variable KML_ID D_INTEGER;
begin
  obj_kind = f_objkind('kontract', 'id');

  --Оператор FOR нужен на случай если ИДЖАРА Товары когда нибудь станут групповыми
  for select kt.k_period, kt.k_disb_date, kc.kc_from_c_id, kal.kal_new_reg_id, kt.k_id, kt.k_kontract_no
      from kontract kt
      inner join knclient kc on kc.kc_from_k_id = kt.k_id
      inner join knaktlist kal on kal.kal_from_kf_id = kt.k_from_kf_id
      where kt.k_id = :obj_id and
            kc.kc_amount_approved > 0 and
            kal.is_delete = 0 and
            kt.k_disb_date between kal.kal_date and kal.kal_date_end
      into period, k_disb_date, c_id, reg_id, k_id, k_kontract_no
  do
  begin
    reg_name = null;
    select rg.reg_name
    from region rg
    where rg.reg_unicode = :reg_id
    into reg_name;

    client_fio = null;
    client_inn = null;
    client_address = null;
    select cl.c_name_fio, cl.c_inn, adl.adl_address_str
    from clients cl
    inner join addrclient adc on adc.adc_from_c_id = cl.c_id
    inner join addrlist adl on adl.adl_id = adc.adc_from_adl_id
    where cl.c_id = :c_id and
          adc.adc_address_type = 1
    order by adl.adl_datetime desc
    rows 1
    into client_fio, client_inn, client_address;

    if (coalesce(add_params, '') <> '') then
    begin
      edl_id = f_json_str_get_value(add_params, 'edl_id', null);
      qr_code = f_edoc_qr_by_edl(:edl_id, :c_id);

      select eds.eds_status_datetime
      from edoc_list edl
      inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
      where edl.edl_id = :edl_id and
            eds.eds_from_c_id = :c_id and
            eds.eds_status = 1 and
            edl.edl_status = 1
      into client_edoc_datetime;
    end

    for select ml.ml_name, ml.ml_model, kml.kml_cost, kml.kml_count, kml.kml_total,
               sum(((kml_sell_cost + kml_sell_nsp) * kml_count)) over(partition by kml.kml_object_id) as sum_kml_total,
               round((kml_sell_cost + kml_sell_nsp) * kml_count, 0) as  kml_sell_total_nsp, kml.kml_id
        from kontmat kml
        inner join matlist ml on ml.ml_id = kml.kml_from_ml_id
        where kml.kml_object_kind = :obj_kind and
              kml.kml_object_id = :k_id
        into ml_name, ml_model, kml_cost, kml_count, kml_total, sum_kml_total, kml_sell_total_nsp, kml_id
    do
    begin
      kmi_imei = (select cast(kmi_imei as d_str_20) from kontmat_imei where kmi_kml_id = :kml_id);

      suspend;
    end
  end
end
