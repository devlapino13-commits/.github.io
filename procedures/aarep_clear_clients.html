Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE AAREP_CLEAR_CLIENTS(
    REP_DATE DATE)
RETURNS (
    REG_NAME VARCHAR(100),
    CL_KOLICH INTEGER,
    CL_BALANCE DECIMAL(18,4),
    K_PERC DECIMAL(18,7),
    CL_CIR1_KOLICH INTEGER,
    CL_CIR1_BALANCE DECIMAL(18,4),
    K_PERC1 DECIMAL(18,7),
    CL_CIR3_KOLICH INTEGER,
    CL_CIR3_BALANCE DECIMAL(18,4),
    K_PERC3 DECIMAL(18,7),
    CL_CIR4_KOLICH INTEGER,
    CL_CIR4_BALANCE DECIMAL(18,4),
    K_PERC4 DECIMAL(18,7),
    CL_CIR7_KOLICH INTEGER,
    CL_CIR7_BALANCE DECIMAL(18,4),
    K_PERC7 DECIMAL(18,7))
AS
declare variable c_amount7 decimal(18,4);
declare variable c_amount4 decimal(18,4);
declare variable c_amount3 decimal(18,4);
declare variable c_amount1 decimal(18,4);
declare variable perc7 decimal(18,4);
declare variable perc4 decimal(18,4);
declare variable perc3 decimal(18,4);
declare variable perc1 decimal(18,4);
declare variable perc decimal(18,4);
declare variable c_amount decimal(18,4);
begin
  for select sum(c_amount), sum(cl_balance), sum(cl_kolich), sum(cl_cir1_balance), sum(cl_cir1_kolich),
             sum(cl_cir3_balance), sum(cl_cir3_kolich), sum(cl_cir4_balance), sum(cl_cir4_kolich), sum(cl_cir7_balance),
             sum(cl_cir7_kolich), reg_name, sum(perc), sum(perc1), sum(perc3), sum(perc4), sum(perc7), sum(c_amount1),
             sum(c_amount3), sum(c_amount4), sum(c_amount7)
      from aarep_clear_clients_tmp(:rep_date)
      group by reg_name
      into :c_amount, :cl_balance, :cl_kolich, :cl_cir1_balance, :cl_cir1_kolich, :cl_cir3_balance, :cl_cir3_kolich,
           :cl_cir4_balance, :cl_cir4_kolich, :cl_cir7_balance, :cl_cir7_kolich, :reg_name, :perc, :perc1, :perc3,
           :perc4, :perc7, :c_amount1, :c_amount3, :c_amount4, :c_amount7
  do
  begin
    if (c_amount <> 0) then
      k_perc = cast(perc / c_amount as decimal(15,7)) * 100;
    else
      k_perc = 0;
    if (c_amount1 <> 0) then
      k_perc1 = cast(perc1 / c_amount1 as decimal(15,7)) * 100;
    else
      k_perc1 = 0;
    if (c_amount3 <> 0) then
      k_perc3 = cast(perc3 / c_amount3 as decimal(15,7)) * 100;
    else
      k_perc3 = 0;
    if (c_amount4 <> 0) then
      k_perc4 = cast(perc4 / c_amount4 as decimal(15,7)) * 100;
    else
      k_perc4 = 0;
    if (c_amount7 <> 0) then
      k_perc7 = cast(perc7 / c_amount7 as decimal(15,7)) * 100;
    else
      k_perc7 = 0;
    suspend;
  end
end
