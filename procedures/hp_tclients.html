CREATE PROCEDURE HP_TCLIENTS(
    PAR1 D_ID)
RETURNS (
    C_ID D_ID,
    C_TYPE D_INTEGER,
    C_NAME_FIO D_STR_100,
    C_SEX D_SMALLINT,
    C_BIRTH_DATE D_DATE,
    C_NATION D_INTEGER,
    C_CITIZEN D_INTEGER,
    C_PASS_NO VARCHAR(30),
    C_PASS_DATE D_DATE,
    C_PASS_WHO D_STR_255,
    C_PASS_DATE_EXPIRE D_DATE,
    C_FROM_REG_ID D_INTEGER,
    C_FAMILY D_INTEGER,
    C_EDUCATION D_INTEGER,
    C_KOL_FAMILY D_INTEGER,
    C_KOL_CHILD D_INTEGER,
    C_MEMO VARCHAR(1024),
    C_REG_NO D_STR_20,
    C_OKPO D_STR_20,
    C_INN D_STR_20,
    C_ACCOUNT VARCHAR(50),
    C_ORG_FROM_B_ID D_STR_10,
    C_HIRED_DATE D_DATE,
    C_IS_RESIDENT D_SMALLINT,
    C_ARMY D_INTEGER,
    C_TABEL D_STR_10,
    C_OTHERS D_STR_20,
    C_IS_PERSONAL D_SMALLINT,
    C_DOC_TYPE D_INTEGER,
    C_FORM_STATUS_LONG D_STR_255,
    C_FORM_STATUS_SHORT VARCHAR(20),
    C_ORG_TYPE_LONG D_STR_255,
    C_ORG_TYPE_SHORT VARCHAR(20),
    C_FAM_BUSINESS D_STR_255,
    C_FROM_UN_ID D_ID,
    C_PINCODE VARCHAR(20),
    C_FAM_ADL_ID D_ID,
    FAM_ADDRESS D_STR_255,
    C_FROM_IS_ID D_INTEGER,
    C_PFT_RISK_KOEF D_INTEGER,
    C_PFT_RISK_MEMO VARCHAR(1024),
    C_ALTERNAT_FIO D_STR_100,
    C_BIRTH_COUNTRY D_STR_100,
    C_BIRTH_CITY D_STR_100,
    C_VERIFY_STATUS D_INTEGER,
    C_PASS_WHO_CODE D_STR_20,
    C_BIRTHDAY_COMMITED D_SMALLINT,
    C_NAME_FULL D_STR_255,
    C_IS_PUBLIC D_INTEGER,
    C_TYPE_CONTR D_INTEGER,
    C_IS_FOUNDER D_INTEGER,
    C_IS_BENEFIC_OWNER D_INTEGER,
    C_NAME_ENG D_STR_255,
    C_AUTHORIZED_CAPITAL D_STR_255,
    C_BRANCHES D_STR_255,
    C_DOC_INFORM D_STR_255,
    C_GOALS D_STR_255,
    C_CURRENT_GNI_CATEG D_INTEGER,
    HM_MONTH D_INTEGER,
    HM_EXP VARCHAR(50),
    HAE_YEARS D_INTEGER,
    HAE_MONTH D_INTEGER,
    HAE_EXP VARCHAR(50))
AS
begin
  select c_id, c_type, c_name_fio, c_sex, c_birth_date, c_nation, c_citizen, c_pass_no, c_pass_date, c_pass_who,
         c_pass_date_expire, c_from_reg_id, c_family, c_education, c_kol_family, c_kol_child, c_memo, c_reg_no, c_okpo,
         c_inn, c_account, c_org_from_b_id, c_hired_date, c_is_resident, c_army, c_tabel, c_others, c_is_personal,
         c_doc_type, c_form_status_long, c_form_status_short, c_org_type_long, c_org_type_short, c_fam_business,
         c_from_un_id, c_pincode, c_fam_adl_id, fam.adl_address_str, c_from_is_id, c_pft_risk_koef, c_pft_risk_memo,
         c_alternat_fio, c_birth_country, c_birth_city, c_verify_status, c_pass_who_code, c_birthday_commited,
         c_name_full, c_is_public, c_type_contr, c_is_founder, c_is_benefic_owner, c_name_eng, c_authorized_capital,
         c_branches, c_doc_inform, c_goals, c_current_gni_categ
  from clients
  inner join addrlist fam on c_fam_adl_id = fam.adl_id
  where c_id = :par1
  into :c_id, :c_type, :c_name_fio, :c_sex, :c_birth_date, :c_nation, :c_citizen, :c_pass_no, :c_pass_date, :c_pass_who,
       :c_pass_date_expire, :c_from_reg_id, :c_family, :c_education, :c_kol_family, :c_kol_child, :c_memo, :c_reg_no,
       :c_okpo, :c_inn, :c_account, :c_org_from_b_id, :c_hired_date, :c_is_resident, :c_army, :c_tabel, :c_others,
       :c_is_personal, :c_doc_type, :c_form_status_long, :c_form_status_short, :c_org_type_long, :c_org_type_short,
       :c_fam_business, :c_from_un_id, :c_pincode, :c_fam_adl_id, :fam_address, :c_from_is_id, :c_pft_risk_koef,
       :c_pft_risk_memo, :c_alternat_fio, :c_birth_country, :c_birth_city, :c_verify_status, :c_pass_who_code,
       :c_birthday_commited, :c_name_full, :c_is_public, :c_type_contr, :c_is_founder, :c_is_benefic_owner, :c_name_eng,
       :c_authorized_capital, :c_branches, :c_doc_inform, :c_goals, :c_current_gni_categ;

  select hm.hm_mount
  from hr_pstatus ps
  inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id and
        ho.ho_order_type = 1 and
        ho.ho_status = 1
  inner join hr_mount hm on hm.hm_ps_from_ho_id = ps.ps_from_ho_id and
        ps.ps_id = hm.hm_from_ps_id
  where ps.ps_from_c_id = :c_id
  order by ho.ho_date desc
  rows 1
  into :hm_month;

  hae_years = 0;
  hae_month = 0;

  if (hm_month is not null) then
    hm_exp = (floor(hm_month / 12) || iif(floor(hm_month / 12) < 5, ' г. ', ' л. ') || mod(hm_month, 12) || ' м.');
  else
    hm_exp = 'Не установлен';
  hae_exp = hm_exp;
  suspend;
end
