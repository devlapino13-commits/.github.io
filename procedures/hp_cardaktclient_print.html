CREATE PROCEDURE HP_CARDAKTCLIENT_PRINT(
    CAC_ID D_INTEGER,
    ADD_PARAMS D_STR_LONG = null)
RETURNS (
    CAC_DATE D_DATE,
    CARD_ID D_INTEGER,
    RD_NAME D_STR_255,
    REG_NAME D_STR_CODE,
    CASHIER_C_NAME_FIO D_STR_100,
    CASHIER_C_INN D_STR_20,
    CASHIER_C_PASS_NO D_STR_CODE,
    CASHIER_C_PASS_DATE D_DATE,
    CASHIER_C_PASS_WHO D_STR_255,
    CASHIER_C_ADDR D_STR_255,
    C_NAME_FIO D_STR_100,
    C_INN D_STR_20,
    C_PASS_NO D_STR_CODE,
    C_PASS_DATE D_DATE,
    C_PASS_WHO D_STR_255,
    C_ADDR D_STR_255,
    COMPANY_NAME D_STR_255,
    COMPANYF_NAME D_STR_255,
    QR_CODE D_STR_LONG,
    CELL_NO D_STR_100,
    EDOC_CARD_ACT_NUMBER D_ID,
    EDOC_CARD_ACT_DATE D_DATE,
    CAL_PAN_SHORT D_STR_20,
    C_PIN_NAME_FIO D_STR_100,
    JL_PIN_NAME D_STR_100)
AS
declare variable C_ID D_ID;
declare variable C_PIN_ID D_ID;
declare variable IS_PINCOVER D_BOOL;
declare variable EDL_ID D_ID;
declare variable EDL_ID1 D_ID;
declare variable UN_ID D_ID;
declare variable CASHIER_C_ID D_ID;
begin
  cell_no = '';
  company_name = f_company_name('short', 'rus');
  companyf_name = f_company_name('full', 'rus');

  for select cac_date, card_id, rd_name, reg_name, c_id, c_name_fio, un_id, edl_id, is_pincover, c_pin_id
      from hp_cardaktclient_view(:cac_id)
      into :cac_date, :card_id, :rd_name, :reg_name, :c_id, :c_name_fio, :un_id, :edl_id1, :is_pincover, :c_pin_id
  do
  begin
    -- от кого
    select c_id, c_name_fio, c_inn, c_pass_no, c_pass_date, c_pass_who
    from username
    inner join clients on c_id = un_from_c_id
    where un_id = :un_id
    into :cashier_c_id, :cashier_c_name_fio, :cashier_c_inn, :cashier_c_pass_no, :cashier_c_pass_date,
         :cashier_c_pass_who;
    cashier_c_addr = (select adl_address_str
                      from addrclient
                      inner join addrlist on adl_id = adc_from_adl_id
                      where adc_from_c_id = :cashier_c_id and
                            adc_address_type = 1);

    -- кому
    select c_inn, c_pass_no, c_pass_date, c_pass_who
    from clients
    where c_id = :c_id
    into :c_inn, :c_pass_no, :c_pass_date, :c_pass_who;
    c_addr = (select adl_address_str
              from addrclient
              inner join addrlist on adl_id = adc_from_adl_id
              where adc_from_c_id = :c_id and
                    adc_address_type = 1);
    if (coalesce(add_params, '') <> '') then
    begin
      edl_id = f_json_str_get_value(add_params, 'edl_id', null);
      qr_code = f_edoc_qr_by_edl(edl_id, :c_id);

      -- информация должна тянуться из договора на выпуск и обсуживание карт Элькарт,
      -- который активен на дату создания данного эл. документа.
      select first 1
        edl_id,
        cast(eds_status_datetime as date)
      from edoc_list
        inner join edoc_sign
          on eds_from_edl_id = edl_id
      where edl_object_kind = 0
        and edl_object_id = :c_id
        and edl_from_edc_id = f_edoc_kind('mbulak_card.agreement', 'id')
        and edl_status = 1
        and edl_id < :edl_id1
      order by edl_id desc
      into
        :edoc_card_act_number,
        :edoc_card_act_date;
    end
    cell_no = f_cont_info('clients', :c_id, 'cell_main', 'ok');

    if (is_pincover = 1) then
      select
        c_name_fio,
        jl_name
      from hr_pstatus
        inner join clients
          on c_id = ps_from_c_id
        inner join joblist
          on jl_id = ps_from_jl_id
      where ps_from_c_id = :c_pin_id
        and :cac_date between ps_date_begin and ps_date_end
        and ps_is_working = 1
        and ps_status = 1
      into
        :c_pin_name_fio,
        :jl_pin_name;

    select cal.cal_pan_short
    from cardlist cal
    where cal.cal_card_id = :card_id
    into :cal_pan_short;
    suspend;
  end
end
