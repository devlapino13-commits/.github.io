CREATE PROCEDURE HP_EXPRESSAPP_LOAN_NEXT(
    EXA_ID D_ID)
RETURNS (
    WAIT_STATUS D_SMALLINT)
AS
declare variable LO_ID D_ID;
declare variable UN_ID D_ID;
declare variable SUBJECT STR_255;
declare variable body D_STR_LONG;
declare variable K_KONTRACT_NO D_STR_20;
declare variable C_NAME_FIO STR_100;
declare variable EXA_RESHIPPING_COUNT D_INTEGER;
declare variable MA_ID D_INTEGER;
declare variable C_ID D_ID;
declare variable CLI_FIO D_STR_100;
begin
  select exa_wait_status,
         exa_reshipping_count, -- данное поле использовал для вычисления первого раза для отправки пушка
         exa_lo_id,
         exa_from_un_id,
         exa_from_ma_id,
         exa_from_c_id
  from expressapp
  where exa_id = :exa_id
  into wait_status,
       exa_reshipping_count,
       lo_id,
       un_id,
       ma_id,
       c_id;

  if (exa_reshipping_count <> 999) then
  begin
    update expressapp set exa_reshipping_count = 999
    where exa_id = :exa_id;

    select c_name_fio from clients
    where c_id = :c_id
    into cli_fio;

    subject = 'Подствердите заявку id:'||ma_id;
    body = 'Прошу проверить условий по заявке и подтвердить, клиент:'||cli_fio;

    execute procedure hp_send_push_to_staff(:lo_id, :subject, :body, :un_id, '', 7);

    wait_status = 1; -- до потверждение ВИПСом
  end
  suspend;
end
