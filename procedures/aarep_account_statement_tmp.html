                                      when '58-03' then '49201'
                                      when '76-09' then '49209'
                                      when '98-01' then '49209'
                                      else sta_corr_acc_short
                                    end);
          end

          sub_no = sub_no + 1;
          ol_doc_no_str = left(sta_account_no, 4) || '-' || right('000000' || sta_doc_no, 6) || '/' || sub_no;

          sta_ob_dt = 0;
          sta_ob_dt_ekv = 0;
          sta_ob_kt = 0;
          sta_ob_kt_ekv = 0;

          if (ac_dt = sta_account_no) then
          begin
            sta_ob_dt = ol_summa_minus;
            sta_ob_dt_ekv = ol_summa_minus;
          end
          else
          begin
            sta_ob_kt = ol_summa_minus;
            sta_ob_kt_ekv = ol_summa_minus;
          end

          sta_ob_saldo = :sta_ob_saldo + :sta_ob_dt - :sta_ob_kt;
          sta_ob_saldo_ekv = :sta_ob_saldo_ekv + :sta_ob_dt_ekv - :sta_ob_kt_ekv;
          sta_saldo_dt = 0;
          sta_saldo_kt = 0;
          sta_saldo_dt_ekv = 0;
          sta_saldo_kt_ekv = 0;
          if (sta_ob_saldo >= 0) then
            sta_saldo_dt = :sta_ob_saldo;
          else
            sta_saldo_kt = -1 * :sta_ob_saldo;

          if (sta_ob_saldo_ekv >= 0) then
            sta_saldo_dt_ekv = :sta_ob_saldo_ekv;
          else
            sta_saldo_kt_ekv = -1 * :sta_ob_saldo_ekv;

          if (report_with_equal = 0) then
          begin
            if ((sta_ob_dt <> 0) or (sta_ob_kt <> 0)) then
            begin
              sta_line_id = :sta_line_id + 1;
              sta_no = :sta_no + 1;
              suspend;
            end
          end
          else
          begin
            sta_line_id = :sta_line_id + 1;
            sta_no = :sta_no + 1;
            suspend;
          end
        end
      end
      else
      begin
        sta_ob_saldo = :sta_ob_saldo + :sta_ob_dt - :sta_ob_kt;
        sta_ob_saldo_ekv = :sta_ob_saldo_ekv + :sta_ob_dt_ekv - :sta_ob_kt_ekv;
        sta_saldo_dt = 0;
        sta_saldo_kt = 0;
        sta_saldo_dt_ekv = 0;
        sta_saldo_kt_ekv = 0;
        if (sta_ob_saldo >= 0) then
          sta_saldo_dt = :sta_ob_saldo;
        else
          sta_saldo_kt = -1 * :sta_ob_saldo;

        if (sta_ob_saldo_ekv >= 0) then
          sta_saldo_dt_ekv = :sta_ob_saldo_ekv;
        else
          sta_saldo_kt_ekv = -1 * :sta_ob_saldo_ekv;

        if (report_with_equal = 0) then
        begin
          if ((sta_ob_dt <> 0) or (sta_ob_kt <> 0)) then
          begin
            sta_line_id = :sta_line_id + 1;
            sta_no = :sta_no + 1;
            suspend;
          end
        end
        else
        begin
          sta_line_id = :sta_line_id + 1;
          sta_no = :sta_no + 1;
          suspend;
        end
      end
    end
  end
  else
  begin -- Не касса
    sta_ob_saldo = :sta_ob_saldo + :sta_ob_dt - :sta_ob_kt;
    sta_ob_saldo_ekv = :sta_ob_saldo_ekv + :sta_ob_dt_ekv - :sta_ob_kt_ekv;
    sta_saldo_dt = 0;
    sta_saldo_kt = 0;
    sta_saldo_dt_ekv = 0;
    sta_saldo_kt_ekv = 0;
    if (sta_ob_saldo >= 0) then
      sta_saldo_dt = :sta_ob_saldo;
    else
      sta_saldo_kt = -1 * :sta_ob_saldo;

    if (sta_ob_saldo_ekv >= 0) then
      sta_saldo_dt_ekv = :sta_ob_saldo_ekv;
    else
      sta_saldo_kt_ekv = -1 * :sta_ob_saldo_ekv;

    if (report_with_equal = 0) then
    begin
      if ((sta_ob_dt <> 0) or (sta_ob_kt <> 0)) then
      begin
        sta_line_id = :sta_line_id + 1;
        sta_no = :sta_no + 1;
        suspend;
      end
