CREATE PROCEDURE HP_EXTINFOLIST_PDN_ONE_INCOME(
    KC_ID D_ID,
    REP_DATE D_DATE,
    C_ID D_ID = null)
RETURNS (
    RES_LAST_PERIOD D_INTEGER,
    RES_INCOME_AVG D_CURRENCY,
    RES_INCOME_01 D_CURRENCY,
    RES_INCOME_02 D_CURRENCY,
    RES_INCOME_03 D_CURRENCY,
    RES_INCOME_04 D_CURRENCY,
    RES_INCOME_05 D_CURRENCY,
    RES_INCOME_06 D_CURRENCY,
    RES_INCOME_07 D_CURRENCY,
    RES_INCOME_08 D_CURRENCY,
    RES_INCOME_09 D_CURRENCY,
    RES_INCOME_10 D_CURRENCY,
    RES_INCOME_11 D_CURRENCY,
    RES_INCOME_12 D_CURRENCY,
    RES_STAT_NAME D_STR_255,
    RES_PLACE_NAME D_STR_255,
    RES_ADDRESS D_STR_255,
    PL_ID D_ID,
    STR_ID D_INTEGER,
    ADL_ID D_ID)
AS
declare variable month_begin d_date;
declare variable income_amount d_currency;
declare variable month_no d_currency;
declare variable sti_code d_str_code;
declare variable sti_id d_integer;
declare variable rep_period d_integer;
declare variable tmp_period d_integer;
declare variable period_count d_integer;
declare variable orig_adl_id d_id;
declare variable kc_adl_datetime d_datetime;
begin
  month_begin = cast('01.' || right('0' || extract(month from rep_date), 2) || '.' || extract(year from rep_date) as date);
  month_begin = dateadd(month, -3, month_begin);

  res_stat_name = '';
  res_place_name = '';
  sti_code = 'People_Avg_Income';

  select kc_from_c_id
  from knclient
  where kc_id = :kc_id
  into c_id;

  select kc_from_adl_id, kc_adl_datetime
  from knclient_pdn
  where kc_id = :kc_id
    and kcp_kind = 0
  into orig_adl_id, kc_adl_datetime;
  kc_adl_datetime = coalesce(kc_adl_datetime, current_timestamp);

  -- получаем ID индикатора
  select sti_id
  from statindic
  where sti_code = :sti_code
  into sti_id;

  if (sti_id is null) then
    exception e_default 'Unknown sti_code: ' || sti_code;

  -- Адрес получаем на дату/время создания записи по PDN
  if (orig_adl_id is not null) then
  begin
    adl_id = orig_adl_id;
    select adl_address_str, adl_from_pl_id
    from hp_get_addrlist(:adl_id, :kc_adl_datetime)
    into res_address, pl_id;
  end else
  begin
    select adc_from_adl_id, adl_address_str, adl_from_pl_id, pl_name
    from addrclient
    left join hp_get_addrlist(adc_from_adl_id, :kc_adl_datetime) on 1=1
    left join places on adl_from_pl_id = pl_id
    where adc_from_c_id = :c_id and
          adc_address_type = 1
    into adl_id, res_address, pl_id, res_place_name;

    if (res_place_name = 'Байконур') then
    begin
      adl_id = null;
      res_address = '';
      pl_id = null;
      select adc_from_adl_id, adl_address_str, adl_from_pl_id
      from addrclient, hp_get_addrlist(adc_from_adl_id, :kc_adl_datetime)
      where adc_from_c_id = :c_id and
            adc_address_type = 2
      into adl_id, res_address, pl_id;
    end
  end

  -- подменим pl_id, если есть для этого кредита на эту дату в таблице подмены регионов (для новых регионов)
  select prp_from_pl_id
  from pdn_replace_places rpr
  where prp_kc_id = :kc_id
    and prp_date = :rep_date
  into pl_id;

  if (pl_id is null) then
    exception e_default 'Address or PL_ID is empty not found for ADL_ID: ' || adl_id;

  if (pl_id = 810001273767) then
    pl_id = 810000076865; -- временное решение. для региона Кемеровская область - Кузбасс не загружена статистика, есть для старой Кемеровской области

  rep_period = extract(year from month_begin) * 100 + extract(month from month_begin);

  -- находим подходящую территорию для населенного пункта
  while (str_id is null and pl_id is not null) do
  begin
    -- Проверяем, нашли ли значение
    select str_id
    from statregion
    where str_from_pl_id = :pl_id
    order by str_id desc
    rows 1
    into str_id;

    -- если нет, берем родительскую ветку и проверим её
    if (str_id is null) then
    begin
      select parent
      from places
      where pl_id = :pl_id
      into :pl_id;

      if (pl_id = 810001273767) then
        pl_id = 810000076865; -- временное решение. для региона Кемеровская область - Кузбасс не загружена статистика, есть для старой Кемеровской области
    end
  end

  if (pl_id is not null) then
  begin
    select pl_name || coalesce(' ' || plt_short_name, '')
    from places
    left join placetype on pl_status = plt_id
    where pl_id = :pl_id
    into res_place_name;
  end

  if (str_id is not null) then
  begin
    select str_name
    from statregion
    where str_id = :str_id
    into res_stat_name;
  end

  res_income_avg = 0;
  res_income_01 = 0;
  res_income_02 = 0;
  res_income_03 = 0;
  res_income_04 = 0;
  res_income_05 = 0;
  res_income_06 = 0;
  res_income_07 = 0;
  res_income_08 = 0;
  res_income_09 = 0;
  res_income_10 = 0;
  res_income_11 = 0;
  res_income_12 = 0;

  -- если стат.территория найдена, получаем период от ближайшей даты
  if (str_id is not null) then
  begin
    if (rep_date <= '31.10.2019') then
    begin
      select stp_period_begin
      from statpocket
      where stp_period_begin <= :rep_period and
            stp_period_begin < 201907 and
            stp_date_begin <= :rep_date and
            stp_date_end >= :rep_date and
            stp_from_sti_id = :sti_id
      order by stp_period_begin desc
      rows 1
      into res_last_period;

      month_no = 12;
      while (month_no > 0) do
      begin
        rep_period = extract(year from month_begin) * 100 + extract(month from month_begin);
  
        income_amount = 0;
        select stv_value
        from statvalue, statpocket
        where stp_from_sti_id = :sti_id and
              stp_period_begin <= :rep_period and
              stp_period_begin < 201907 and
              stp_date_begin <= :rep_date and
              stp_date_end >= :rep_date and
              stp_id = stv_from_stp_id and
              stv_from_str_id = :str_id
        order by stp_period_begin desc, stv_id desc
        rows 1
        into income_amount;
  
        res_income_avg = res_income_avg + income_amount;
    
        if (month_no = 1) then res_income_01 = income_amount;
        else if (month_no = 2) then res_income_02 = income_amount;
        else if (month_no = 3) then res_income_03 = income_amount;
        else if (month_no = 4) then res_income_04 = income_amount;
        else if (month_no = 5) then res_income_05 = income_amount;
        else if (month_no = 6) then res_income_06 = income_amount;
        else if (month_no = 7) then res_income_07 = income_amount;
        else if (month_no = 8) then res_income_08 = income_amount;
        else if (month_no = 9) then res_income_09 = income_amount;
        else if (month_no = 10) then res_income_10 = income_amount;
        else if (month_no = 11) then res_income_11 = income_amount;
        else if (month_no = 12) then res_income_12 = income_amount;
    
        month_begin = dateadd(month, -1, month_begin);
        month_no = month_no - 1;
      end
      res_income_avg = round(res_income_avg / 12.00, 2);
    end else
    begin
      res_last_period = null;
      period_count = 0;
      for select stp_period_begin, stv_value
      from statvalue, statpocket
      where stp_id = stv_from_stp_id and
            stp_from_sti_id = :sti_id and
            stv_from_str_id = :str_id and
            stp_period_begin < iif(:rep_date < '24.12.2019', 201907, 999999) and -- если расчет до указанной даты, исключаем справочник с 201907
            stp_period_begin <= :rep_period and
            stp_date_begin <= :rep_date and
            stp_date_end >= :rep_date
      order by stp_period_begin desc
      rows 4
      into tmp_period, income_amount
      do
      begin
        if (res_last_period is null) then
          res_last_period = tmp_period;

        res_income_avg = res_income_avg + income_amount;

        period_count = period_count + 1;
        if (period_count = 1) then
        begin
          res_income_10 = income_amount;
          res_income_11 = income_amount;
          res_income_12 = income_amount;
        end
        if (period_count = 2) then
        begin
          res_income_07 = income_amount;
          res_income_08 = income_amount;
          res_income_09 = income_amount;
        end
        if (period_count = 3) then
        begin
          res_income_04 = income_amount;
          res_income_05 = income_amount;
          res_income_06 = income_amount;
        end
        if (period_count = 4) then
        begin
          res_income_01 = income_amount;
          res_income_02 = income_amount;
          res_income_03 = income_amount;
        end
      end
      res_income_avg = round(res_income_avg / 4.00, 2);

      -- если по региону нет значений за последние 4 квартала, выдаем ошибку
      if (period_count <> 4) then
        exception e_default 'There are no 4 quaters for stat value, kc_id:' || kc_id;
    end
  end
  suspend;
end
