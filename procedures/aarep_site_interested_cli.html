CREATE PROCEDURE AAREP_SITE_INTERESTED_CLI(
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    CK_CHECKED_DATE DATE,
    CK_CHECKED_OPER STR_255,
    C_PHONE STR_255,
    STATUS_DATE DATE,
    STATUS STR_100,
    MANAGER_FIO STR_255,
    REG_NAME STR_100,
    STATUS_ID INTEGER,
    KC_AMOUNT_APPROVED CURR_TYPE,
    STATUS_REG STR_100,
    PASS_NO_OPER VARCHAR(30),
    CELL_NO_OPER VARCHAR(20),
    FIND_METHOD VARCHAR(50),
    CK_STATUS STR_100,
    CLIENT_FIO STR_100,
    K_KONTRACT_NO VARCHAR(20),
    KF_NAME STR_100,
    CK_FIO STR_100,
    NEXT_CALL_DATETIME TIMESTAMP,
    AGREE_DATETIME TIMESTAMP,
    ESM_DATETIME TIMESTAMP,
    CK_MEMO VARCHAR(1000),
    K_DISB_DATE DATE,
    K_KOLICH_CLIENT_APPROVED INTEGER,
    C_BIRTH_DATE DATE,
    C_CITIZEN STR_100,
    K_CITY STR_100,
    SVZ_FIO STR_100,
    SMS_TXT STR_255)
AS
declare variable esm_id id_type;
declare variable esm_id_old id_type;
declare variable manager_id id_type;
declare variable reg_id integer;
declare variable ck_secondary_data str_255;
declare variable fuzzy_search smallint;
declare variable ck_birth_date date;
declare variable ck_status_code smallint;
declare variable pl_id id_type;
declare variable ck_id id_type;
begin
  esm_id_old = 0;

  for select esm_id, chk.ck_checked_date, -- дата звонка
             chk.ck_checked_oper,
-- опертор кц.
             (
             case
               when ck_status = 1 then 'Интресуется (перезвонить)'
               when ck_status = 2 then 'Интресуется (конеч. статус)'
             end) as ck_status,
             chk.ck_lo_id, chk.ck_from_reg_id, chk.ck_secondary_data, chk.ck_fio, chk.ck_birth_date, ck_status,
             chk.ck_next_call_datetime, -- дата  + время след звонка
             iif(ck_status = 2, chk.ck_checked_datetime, null) agree_datetime, -- конечный статус заинтересован
             im.esm_datetime, -- дата + время заявки на сайте,
             chk.ck_fio, -- фио по заявке
             chk.ck_memo, -- комментарий оператора
             ck_id
      from cc_check_knclient chk
      inner join extsysmain im on im.esm_id = chk.ck_from_esm_id
      where chk.ck_type = 6
-- and chk.ck_status >=1
            and
            chk.ck_checked_date between :rep_date_begin and :rep_date_end and
            esm_action = 'save_cl_info'
      order by im.esm_id, chk.ck_id /*ck_status */ desc, chk.ck_checked_date
      into :esm_id, :ck_checked_date, :ck_checked_oper, :ck_status, :manager_id, :reg_id, :ck_secondary_data, :ck_fio,
           :ck_birth_date, :ck_status_code, :next_call_datetime, :agree_datetime, :esm_datetime, :ck_fio, :ck_memo,
           :ck_id
  do
  begin
    if (esm_id <> esm_id_old) then -- если есть уже конечный статус то только его и беру
    begin
      if (ck_status_code >= 1) then
      begin
        svz_fio = '';
        sms_txt = '';

        select cl.c_name_fio, cwl.wls_sms_txt
        from cc_website_lo_sms cwl
        inner join clients cl on cwl.wls_lo_id = cl.c_id
        where cwl.wls_from_ck_id = :ck_id rows 1
        into :svz_fio, :sms_txt;

        status_date = null;
        status = null;
        status_id = null;
        k_id = null;
        kc_amount_approved = 0;
        status_reg = null;
        k_kontract_no = null;
        kf_name = null;
        k_disb_date = null;
        k_kolich_client_approved = null;
        pl_id = null;
        k_city = null;

        esm_id_old = esm_id;

        pass_no_oper = '';
        cell_no_oper = '';

        select pass_no, cell_no
        from cc_secondary_data_parse(:ck_secondary_data)
        into pass_no_oper, cell_no_oper;

        select ev.esv_value_txt
        from extsysfields ef, extsysvalues ev
        where ev.esv_from_esm_id = :esm_id and
              ev.esv_from_esf_id = ef.esf_id and
              ef.esf_field_code = 'phone_no' and
              esf_project = 'website'
        into :c_phone; -- телефон мигранта

        c_phone = coalesce(c_phone, '');

        if (c_phone <> '') then
        begin
--c_phone  = '7'||c_phone ;
          c_id = null;

          find_method = '';

          select res_object_id
          from hp_contdata_find_object(:c_phone, 'cell')
          where res_object_kind = 0 rows 1
          into :c_id;

          if (c_id is null) then
          begin
            select res_object_id
            from hp_contdata_find_object('7' || :c_phone, 'cell')
            where res_object_kind = 0 rows 1
            into :c_id;
          end

          if (c_id is not null) then
            find_method = 'Найден по номеру телефона';

          if ((c_id is null) and
              (pass_no_oper <> '')) then
          begin
            select cl.c_id
            from clients cl
            where cl.c_pass_no = :pass_no_oper
            into :c_id;

            if (c_id is not null) then
              find_method = 'Найден паспорту';
          end

          if ((c_id is null) and
              (cell_no_oper <> '')) then
          begin
            select res_object_id
            from hp_contdata_find_object(:cell_no_oper, 'cell')
            where res_object_kind = 0 rows 1
            into :c_id;
            if (c_id is not null) then
              find_method = 'Найден по доп. телефону';
          end

          fuzzy_search = 0;

          if (c_id is null) then -- поиск по ФИО, дате рождения клиента
          begin
            for select c_id
                from hp_find_clients_tmp(null, :ck_fio, :ck_birth_date, null, null, null, '', 1)
                where c_id <> 0
                into :c_id
            do
            begin
              if (fuzzy_search = 0) then
              begin

                status_date = null;
                status = null;
                status_id = null;
                k_id = null;
                kc_amount_approved = 0;
                status_reg = null;
                k_kontract_no = null;
                kf_name = null;
                k_disb_date = null;
                k_kolich_client_approved = null;
                pl_id = null;
                k_city = null;

                select ks_date_begin,
                       (
                       case
                         when kont.k_status = -1 then 'Контракт аннулирован'
                         when ks.ks_state = 0 and kont.k_status <> -1 then 'Контракт в процессе'
                         when ks.ks_state = 1 then 'Контракт выдан'
                       end),
                       ks.ks_state, kc_from_k_id, kc_amount_approved, reg.reg_name, kont.k_kontract_no,
                       kal.kal_new_kf_name, kont.k_fact_disb_date, kont.k_kolich_client_approved, reg.reg_from_pl_id
                from knclient kc, kontstate ks, kontract kont, region reg, knaktlist kal
                where kc_amount_approved > 0 and
                      kont.k_id = ks.ks_from_k_id and
                      kc_from_c_id = :c_id and
                      kc_from_k_id = ks_from_k_id and
                      ks_date_begin between :ck_checked_date and :ck_checked_date + 14 -- беру промежуток между датой звонка и + 14 дней
                      and
                      ks_state < 2 and
                      reg.reg_unicode = ks.ks_region_id and
                      kal.kal_from_kf_id = kont.k_from_kf_id and
                      (:ck_checked_date + 14) between kal.kal_date and kal.kal_date_end and
                      kal.is_delete = 0
                order by ks_number desc
                rows 1
                into :status_date, :status, :status_id, :k_id, :kc_amount_approved, :status_reg, :k_kontract_no,
                     :kf_name, :k_disb_date, :k_kolich_client_approved, :pl_id;

                if (pl_id is not null) then
                begin
                  select pl.pl_name
                  from places pl
                  where pl.pl_id = :pl_id
                  into :k_city;
                end

                if (status is null) then
                begin
                  select ma.ma_date_app,
                         (
                         case ma_is_active
                           when 0 then 'Заявка аннулирована'
                           when -1 then 'Отказ клиента'
                           when 1 then 'Активная заявка'
                         end),
                         kal.kal_new_kf_name
                  from miniappclient mac, miniapp ma, knaktlist kal
                  where mac.mac_is_poruch = 0 and
                        mac.mac_from_c_id = :c_id and
                        mac.mac_from_ma_id = ma.ma_id and
                        ma.ma_date_app between :ck_checked_date and :ck_checked_date + 14 -- беру промежуток между датой звонка и + 14 дней
                        and
                        kal.kal_from_kf_id = ma.ma_from_kf_id and
                        (:ck_checked_date + 14) between kal.kal_date and kal.kal_date_end and
                        kal.is_delete = 0
                  order by ma.ma_number desc, ma_is_active desc
                  rows 1
                  into status_date, status, kf_name;
                end

                if (status <> '') then
                begin
                  fuzzy_search = 1;
                  find_method = 'Нечеткий поиск по ФИО, дате рождения';
                end
                else
                  c_id = null;
              end
            end
          end

          manager_fio = null;

          select cl.c_name_fio
          from clients cl
          where cl.c_id = :manager_id
          into :manager_fio; -- менеджер, к которому направили мигранта

          manager_fio = coalesce(manager_fio, '');

          reg_name = null;

          select reg.reg_name
          from region reg
          where reg.reg_unicode = :reg_id
          into :reg_name; -- офис, в который направили мигранта

          reg_name = coalesce(reg_name, '');

          if ((c_id is not null) and
              (fuzzy_search = 0)) then --  если клиент найден
          begin
            status_date = null;
            status = null;
            status_id = null;
            k_id = null;
            kc_amount_approved = 0;
            status_reg = null;
            k_kontract_no = null;
            kf_name = null;
            k_disb_date = null;
            k_kolich_client_approved = null;
            pl_id = null;
            k_city = null;

            select ks_date_begin,
                   (
                   case
                     when kont.k_status = -1 then 'Контракт аннулирован'
                     when ks.ks_state = 0 and kont.k_status <> -1 then 'Контракт в процессе'
                     when ks.ks_state = 1 then 'Контракт выдан'
                   end),
                   ks.ks_state, kc_from_k_id, kc_amount_approved, reg.reg_name, k_kontract_no, kal.kal_new_kf_name,
                   kont.k_fact_disb_date, k_kolich_client_approved, reg.reg_from_pl_id
            from knclient kc, kontstate ks, kontract kont, region reg, knaktlist kal
            where kc_amount_approved > 0 and
                  kont.k_id = ks.ks_from_k_id and
                  kc_from_c_id = :c_id and
                  kc_from_k_id = ks_from_k_id and
                  ks_date_begin between :ck_checked_date and :ck_checked_date + 14 -- беру промежуток между датой звонка и + 14 дней
                  and
                  ks_state < 2 and
                  reg.reg_unicode = ks.ks_region_id and
                  kal.kal_from_kf_id = kont.k_from_kf_id and
                  (:ck_checked_date + 14) between kal.kal_date and kal.kal_date_end and
                  kal.is_delete = 0
            order by ks_number desc
            rows 1
            into :status_date, :status, :status_id, :k_id, :kc_amount_approved, :status_reg, :k_kontract_no, :kf_name,
                 :k_disb_date, :k_kolich_client_approved, :pl_id;

            if (pl_id is not null) then
            begin
              select pl.pl_name
              from places pl
              where pl.pl_id = :pl_id
              into :k_city;
            end

            if (status is null) then
            begin
              select ma.ma_date_app,
                     (
                     case ma_is_active
                       when 0 then 'Заявка аннулирована'
                       when 1 then 'Активная заявка'
                     end)
              from miniappclient mac, miniapp ma
              where mac.mac_is_poruch = 0 and
                    mac.mac_from_c_id = :c_id and
                    mac.mac_from_ma_id = ma.ma_id and
                    ma.ma_date_app between :ck_checked_date and :ck_checked_date + 14 -- беру промежуток между датой звонка и + 14 дней
              order by ma.ma_number desc, ma_is_active desc
              rows 1
              into status_date, status;
            end
          end

          client_fio = '';
          c_birth_date = null;
          c_citizen = null;

          if (c_id is not null) then
          begin
            select cl.c_name_fio, cl.c_birth_date, rd.rd_name
            from clients cl
            inner join refdata rd on rd.rd_id = cl.c_citizen
            where cl.c_id = :c_id
            into :client_fio, :c_birth_date, :c_citizen;
          end

          suspend;
        end
      end
      else
      begin
        esm_id_old = esm_id;
      end
    end
  end
end
