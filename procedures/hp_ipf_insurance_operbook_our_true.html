CREATE PROCEDURE HP_IPF_INSURANCE_OPERBOOK_OUR_TRUE(
    ESM_ID D_ID,
    K_ID D_ID,
    INSUR_SUM D_CURRENCY)
RETURNS (
    RES_TEXT D_STR_LONG,
    PARENT_OL_ID D_ID)
AS
declare variable ol_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_disb_date d_date;
declare variable go_office_id d_integer;
declare variable bo_office_id d_integer;
declare variable disb_office_id d_integer;
declare variable c_id d_id;
declare variable c_name_fio d_str_255;
declare variable ag_id d_integer;
declare variable insurance_c_id d_id;
declare variable ol_date d_date;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable ol_doc_kind d_integer;
declare variable ol_doc_client_name d_str_255;
declare variable ol_doc_no d_integer;
declare variable ol_is_client d_integer;
declare variable ol_status d_smallint;
declare variable ol_memo d_str_long;
declare variable ob_summa d_currency;
declare variable ob_detail d_str_long;
declare variable al_go_dt d_str_20;
declare variable al_go_kt d_str_20;
declare variable al_bo_dt d_str_20;
declare variable al_bo_kt d_str_20;
declare variable al_disb_dt d_str_20;
declare variable al_disb_kt d_str_20;
declare variable acc_short_dt d_str_20;
declare variable acc_short_kt d_str_20;
declare variable esm_project d_str_100;
declare variable esm_full_source d_str_255;
declare variable esp_memo d_str_255;
begin
  parent_ol_id = 0;
  res_text = '';

  go_office_id = 1090100; -- Бишкек ГО
  bo_office_id = 1060100; -- Бишкекский офис

  -- Получение данных внешней системы
  esm_project = null;
  select esm.esm_project, esm.esm_date, esm.esm_full_source
  from extsysmain esm
  where esm.esm_id = :esm_id
  into :esm_project, :ol_date, :esm_full_source;

  if (esm_project is null) then
  begin
    res_text = 'Ошибка: проект не найден';
    suspend;
    exit;
  end

  esp_memo = '';
  select first 1 esp.esp_memo
  from extsyspro esp
  where esp_project_kind = 'payment' and
        esp_project = :esm_project and
        esp_status = 1
  into :esp_memo;

  acc_short_kt = '11531/311';

  -- Определение типа платежа (терминал/QR)
  if (esm_full_source like '%terminal%' or esm_full_source like '%терминал%' or esp_memo like '%терминал%') then
  begin
    acc_short_dt = '10003/138';
    ob_detail = 'Погашение страхового полиса через терминал';
  end
  else
  begin
    acc_short_dt = '10024/111';
    ob_detail = 'Погашение страхового полиса через QR';
  end

  select k.k_kontract_no, k.k_fact_disb_date, k.k_fact_disb_reg_id
  from kontract k
  where k.k_id = :k_id
  into :k_kontract_no, :k_disb_date, :disb_office_id;
  
  if (disb_office_id is null) then
  begin
    res_text = 'Ошибка: У контракта ' || :k_kontract_no || ' не указан филиал выдачи.';
    suspend;
    exit;
  end

  -- Проверка страхового договора и компании
  select ik_from_ag_id
  from insurance_kontract
  where ik_from_k_id = :k_id
  into :ag_id;

  if (ag_id is null) then
  begin
    res_text = 'Ошибка: Договор со страховой компанией не найден для контракта ' || :k_kontract_no;
    suspend;
    exit;
  end

  select ag_object_id
  from acc_list
  inner join agreements on ag_from_al_id = al_id
  where ag_id = :ag_id
  into :insurance_c_id;
  
  if (insurance_c_id is null) then
  begin
    res_text = 'Ошибка: Не найден контрагент страховой компании по договору.';
    suspend;
    exit;
  end

  -- Получение данных клиента
  select c_id, c_name_fio
  from clients inner join knclient on kc_from_c_id = c_id
  where kc_from_k_id = :k_id and
        kc_amount_approved > 0
  into :c_id, :c_name_fio;

  -- часть 2. предварительная проверка всех счетов (валидация плана счетов) --
  
  -- 2.1 Счета для Бишкек ГО
  -- В ГО приходит оплата, поэтому DT - это локальный счет (QR или Терминал), KT - перевод в БО
  al_go_dt = (select al_id from find_acc_list(:acc_short_dt) where al_from_reg_id = :go_office_id);
  al_go_kt = (select ln_account from loronostro where ln_region_loro = :go_office_id and ln_region_nostro = :bo_office_id);

  if (al_go_dt is null or al_go_kt is null) then
  begin
    res_text = 'Не найден активный счет ' || :acc_short_dt || ' или лоро/ностро счет для ГО (' || :go_office_id || ')';
    suspend;
    exit;
  end

  -- 2.2 Счета для Бишкекского филиала (РО)
  -- БО - транзитный узел: DT - прием от ГО, KT - перевод в филиал выдачи
  al_bo_dt = (select ln_account from loronostro where ln_region_loro = :bo_office_id and ln_region_nostro = :go_office_id);
  al_bo_kt = (select ln_account from loronostro where ln_region_loro = :bo_office_id and ln_region_nostro = :disb_office_id);

  if (al_bo_dt is null or al_bo_kt is null) then
  begin
    res_text = 'Не найдены лоро/ностро счета для офиса ' || :bo_office_id;
    suspend;
    exit;
  end

  -- 2.3 Счета для Офиса выдачи займа
  -- Филиал выдачи: DT - прием от БО, KT - итоговый счет (11531/311)
  al_disb_dt = (select ln_account from loronostro where ln_region_loro = :disb_office_id and ln_region_nostro = :bo_office_id);
  al_disb_kt = (select al_id from find_acc_list(:acc_short_kt) where al_from_reg_id = :disb_office_id);

  if (al_disb_dt is null or al_disb_kt is null) then
  begin
    res_text = 'Ошибка плана счетов Офиса Выдачи: Не найден лоро/ностро счет или ' || :acc_short_kt || ' для офиса ' || :disb_office_id;
    suspend;
    exit;
  end

  -- часть 3. запись данных (выполняется, только если все проверки пройдены) --
  
  -- Подготовка констант для документов
  ol_doc_client_name = 'Оплата за страхование финансирования';
  ol_memo = ol_doc_client_name || ' / ' || k_kontract_no || ' от ' || f_datetostr(k_disb_date) || ' / ' || c_name_fio;
  ol_oper_type = 2;
  ol_oper_subtype = 11;
  ol_doc_kind = 9;
  ol_is_client = 1;
  ol_status = 1;
  ob_summa = insur_sum;

  select num
  from last_doc_number(:ol_date, 9, :go_office_id)
  into :ol_doc_no;

  -- создание родительской операции (бишкек го)
  insert into operlist (ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no, ol_doc_no_str, ol_document_client, ol_is_client,
                        ol_memo, ol_doc_client_name, ol_user_id, ol_date, ol_status)
  values (:go_office_id, :go_office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :ol_doc_no, :k_id, :c_id, :ol_is_client,
          :ol_memo, :ol_doc_client_name, 1000000000000, :ol_date, :ol_status)
  returning ol_id
  into :parent_ol_id;
  
  ol_id = parent_ol_id;

  insert into operbook (ob_from_ol_id, ob_client_dt, ob_client_kt, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:parent_ol_id, null, null, :al_go_dt, :al_go_kt, :ob_summa, :ob_summa, :ol_date, :go_office_id, :ob_detail);

  -- создание дочерней операции (бишкекский офис)
  insert into operlist (ol_parent_id, ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no, ol_doc_no_str, ol_document_client, ol_is_client,
                        ol_memo, ol_doc_client_name, ol_user_id, ol_date, ol_status)
  values (:parent_ol_id, :bo_office_id, :bo_office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :ol_doc_no, :k_id, :c_id, :ol_is_client,
          :ol_memo, :ol_doc_client_name, 1000000000000, :ol_date, :ol_status)
  returning ol_id
  into :ol_id;

  insert into operbook (ob_from_ol_id, ob_client_dt, ob_client_kt, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:ol_id, null, null, :al_bo_dt, :al_bo_kt, :ob_summa, :ob_summa, :ol_date, :bo_office_id, :ob_detail);

  -- создание дочерней операции (офис выдачи займа)
  insert into operlist (ol_parent_id, ol_from_reg_id, ol_document_office, ol_oper_type, ol_oper_subtype, ol_doc_kind, ol_doc_no, ol_doc_no_str, ol_document_client, ol_is_client,
                        ol_memo, ol_doc_client_name, ol_user_id, ol_date, ol_status)
  values (:parent_ol_id, :disb_office_id, :disb_office_id, :ol_oper_type, :ol_oper_subtype, :ol_doc_kind, :ol_doc_no, :k_id, :c_id, :ol_is_client,
          :ol_memo, :ol_doc_client_name, 1000000000000, :ol_date, :ol_status)
  returning ol_id
  into :ol_id;

  insert into operbook (ob_from_ol_id, ob_client_dt, ob_client_kt, ob_account_dt, ob_account_kt, ob_summa, ob_summa_val, ob_ol_date, ob_ol_reg_id, ob_detail)
  values (:ol_id, :c_id, :insurance_c_id, :al_disb_dt, :al_disb_kt, :ob_summa, :ob_summa, :ol_date, :disb_office_id, :ob_detail);


  insert into operkontract (ok_from_k_id, ok_from_ol_id, ok_amount, ok_operation_type)
  values (:k_id, :parent_ol_id, :insur_sum, f_get_rd_id('CashTransactionIncom.Insurance'));

  suspend;
end
