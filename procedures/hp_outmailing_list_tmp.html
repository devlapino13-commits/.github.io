CREATE PROCEDURE HP_OUTMAILING_LIST_TMP(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    REP_GROUP_KIND D_STR_CODE,
    SEARCH_KIND D_SMALLINT,
    IN_K_ID D_ID,
    IN_C_ID D_ID,
    IN_TEXT D_STR_100)
RETURNS (
    RES_OML_ID D_INTEGER)
AS
declare variable oml_group_kind d_integer;
declare variable oml_is_delinq d_bool;
declare variable rep_group_id d_integer;
begin
  if (search_kind is null) then
    search_kind = 0;

  if (coalesce(rep_group_kind, '') = '') then
    rep_group_kind = 'office';

  rep_group_id = f_enum('outmailing_group_kind', rep_group_kind, 'id');

  if (search_kind = 0) then -- выборка за период
  begin
    for select oml_id, oml_group_kind, oml_is_delinq
    from outmailing, region_list(:rep_region_id, 1, 3, 1)
    where oml_parent_id is null and
          oml_date between :rep_date_begin and :rep_date_end and
          oml_from_reg_id = id_list
    into :res_oml_id, oml_group_kind, oml_is_delinq
    do
    begin
      if (rep_group_kind in ('office', 'operdep')) then
      begin
        if (rep_group_id <> oml_group_kind) then
          Continue;
      end

      if (rep_group_kind = 'collection') then
      begin
        if (oml_is_delinq = 0) then
        begin
          if (exists(select * from outmailkont
              where omk_from_oml_id = :res_oml_id and
                    omk_is_delinq = 1)) then
            oml_is_delinq = 1;
        end

        if (not (rep_group_id = oml_group_kind or oml_group_kind = 0 or oml_is_delinq = 1)) then
          Continue;
      end

      suspend;
    end
  end else
  if (search_kind = 1) then -- поиск по тексту
  begin
    if (trim(coalesce(in_text, '')) = '') then
      exception e_message 'Текст поиска не указан';

    for select coalesce(oml_parent_id, oml_id), oml_group_kind, oml_is_delinq
    from outmailing, region_list(:rep_region_id, 1, 3, 1)
    where oml_from_reg_id = id_list and
          (oml_regist_no containing :in_text or
           oml_applicant_name containing :in_text or
           oml_subject containing :in_text or
           oml_kontracts containing :in_text or
           oml_memo containing :in_text)
    into :res_oml_id, oml_group_kind, oml_is_delinq
    do
    begin
      if (rep_group_kind in ('office', 'operdep')) then
      begin
        if (rep_group_id <> oml_group_kind) then
          Continue;
      end

      if (rep_group_kind = 'collection') then
      begin
        if (oml_is_delinq = 0) then
        begin
          if (exists(select * from outmailkont
              where omk_from_oml_id = :res_oml_id and
                    omk_is_delinq = 1)) then
            oml_is_delinq = 1;
        end

        if (not (rep_group_id = oml_group_kind or oml_group_kind = 0 or oml_is_delinq = 1)) then
          Continue;
      end

      suspend;
    end
  end else
  if (search_kind = 2) then -- поиск по клиенту
  begin
    if (coalesce(in_c_id, 0) = 0) then
      Exit;

    for select coalesce(oml_parent_id, oml_id), oml_group_kind, oml_is_delinq
    from outmailing, region_list(:rep_region_id, 1, 3, 1)
    where oml_from_c_id = :in_c_id and
          oml_from_reg_id = id_list
    into :res_oml_id, oml_group_kind, oml_is_delinq
    do
    begin
      if (rep_group_kind in ('office', 'operdep')) then
      begin
        if (rep_group_id <> oml_group_kind) then
          Continue;
      end

      if (rep_group_kind = 'collection') then
      begin
        if (oml_is_delinq = 0) then
        begin
          if (exists(select * from outmailkont
              where omk_from_oml_id = :res_oml_id and
                    omk_is_delinq = 1)) then
            oml_is_delinq = 1;
        end

        if (not (rep_group_id = oml_group_kind or oml_group_kind = 0 or oml_is_delinq = 1)) then
          Continue;
      end

      suspend;
    end
  end else
  if (search_kind = 3) then -- поиск по займу
  begin
    if (coalesce(in_k_id, 0) = 0) then
      Exit;

    for select coalesce(oml_parent_id, oml_id), oml_group_kind, oml_is_delinq
    from outmailkont, outmailing
    where omk_from_k_id = :in_k_id and
          omk_from_oml_id = oml_id
    into :res_oml_id, oml_group_kind, oml_is_delinq
    do
    begin
      if (rep_group_kind in ('office', 'operdep')) then
      begin
        if (rep_group_id <> oml_group_kind) then
          Continue;
      end

      if (rep_group_kind = 'collection') then
      begin
        if (oml_is_delinq = 0) then
        begin
          if (exists(select * from outmailkont
              where omk_from_oml_id = :res_oml_id and
                    omk_is_delinq = 1)) then
            oml_is_delinq = 1;
        end

        if (not (rep_group_id = oml_group_kind or oml_group_kind = 0 or oml_is_delinq = 1)) then
          Continue;
      end

      suspend;
    end
  end
end
