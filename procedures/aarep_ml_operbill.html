CREATE PROCEDURE AAREP_ML_OPERBILL(
    BILL_ID DECIMAL(18,0),
    BILL_LINE_NO INTEGER)
RETURNS (
    AMB_DATE DATE,
    AMB_INVERT_STR VARCHAR(50),
    AMB_ML_NAME VARCHAR(255),
    AMB_MODEL VARCHAR(255),
    AMB_YEDIZ VARCHAR(20),
    AMB_CATEG VARCHAR(200),
    AMB_COMPLECT VARCHAR(1024),
    AMB_DETAIL VARCHAR(2048))
AS
begin
  for select ob_date_begin, ob_invent_str, ml_name, ml_model, yl_code, m2.mc_name || '->' || m1.mc_name, cpl_name,
             ob_full_detail
      from operbill
      left join(matcateg m1 left join matcateg m2 on (m2.mc_id = m1.parent)) on (ob_from_mc_id = m1.mc_id)
      left join(matlist left join yedizlist on (ml_from_yl_id = yl_id)) on (ob_from_ml_id = ml_id)
      left join complect on (ob_complect_id = cpl_id)
      where ob_bill_id = :bill_id and
            ob_is_delete = 0 and
            ob_bill_line_no = :bill_line_no
      order by ob_date_begin desc
      into :amb_date, :amb_invert_str, :amb_ml_name, :amb_model, :amb_yediz, :amb_categ, :amb_complect, :amb_detail
  do
  begin
    suspend;
  end
end
