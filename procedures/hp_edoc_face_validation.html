CREATE PROCEDURE HP_EDOC_FACE_VALIDATION(
    C_ID D_ID,
    JSON_VALIDATION_DETAILS D_STR_8100,
    IS_SUCCESS D_BOOL,
    CAM_PHOTO D_BLOB_BINARY,
    MCL_ID D_ID = null)
RETURNS (
    UUID D_STR_CODE)
AS
declare variable GRS_BLS_ID D_ID;
declare variable CAM_BLS_ID D_ID;
declare variable EDF_ID D_ID;
declare variable SAME_PERSON D_STR_10;
begin
  grs_bls_id = (select bls_id from get_grs_photo(:c_id, 0, :mcl_id, 0));
  cam_bls_id = null;

  -- В.Цымбалюк, 08.04.2025 16:59 (MSK):
  -- вижу что статус у меня выставляется в 0 когда возникают какие либо понты
  -- при проверке (лица на фото нет, лицо засвечено и т.п), а когда все ок то 1.
  -- Насколько я вижу по коду при is_success 0 same_person всегда false
  if (is_success = 1) then
  begin
    same_person = f_json_str_get_value(replace(json_validation_details, '''', '"'), 'same_person', 'false');
    if (same_person = 'true') then
      is_success = 1;
    else
      is_success = 0;
  end

  if (is_success = 1 and cam_photo is not null) then
  begin
     -- Сохраняем блоб только если проверка подтвердила совпадение
     execute procedure hp_insert_into_blobstore (
        'cam_' || coalesce(:c_id, :mcl_id) || '.jpg',
        'fci',
        :cam_photo,
        1000000000000,
        0,
        0,
        :mcl_id)
      returning_values :cam_bls_id;

      -- RFCKG-4042
      if (f_get_db_id() = 41701 and coalesce(c_id, 0) > 0) then
        execute procedure hp_photo_on_date_disb_auto_add(:c_id, :cam_bls_id, 1000000000000);
  end

  insert into edoc_face (
    edf_from_c_id,
    edf_validation_details,
    edf_cam_bls_id,
    edf_grs_bls_id,
    edf_is_success,
    edf_from_mcl_id)
  values (
    :c_id,
    :json_validation_details,
    :cam_bls_id,
    :grs_bls_id,
    :is_success,
    :mcl_id)
  returning edf_id
  into :edf_id;

  uuid = f_uuid('edoc_face', :edf_id);

  suspend;
end
