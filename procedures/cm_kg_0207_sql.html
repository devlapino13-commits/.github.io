CREATE PROCEDURE CM_KG_0207_SQL(
    MA_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable point d_integer;
declare variable c_id d_id;
declare variable ma_date_oper d_date;
declare variable ma_disb_kind d_integer;
begin
  for select mac.mac_from_c_id, ma.ma_date_oper, ma.ma_disb_kind
      from miniappclient mac
      inner join miniapp ma on ma.ma_id = mac.mac_from_ma_id
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      where ma.ma_id = :ma_id and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999' and
            mac.mac_amount > 0 and
            pr_code containing '.jonokoi_cards.'
      into c_id, ma_date_oper, ma_disb_kind
  do
  begin
    point = f_point(:c_id, :ma_date_oper);
    if (point < 6) then
    begin
      res_error_text = 'По условиям продукта у клиента должно быть 12 баллов или более.';
      res_error_is_critical = 1;
      suspend;
    end 

    if (ma_disb_kind <> 6) then
    begin
      res_error_text = 'По условиям продукта способ выдачи должен быть только на карту М Булак.';
      res_error_is_critical = 1;
      suspend;
    end

    if (not exists(select *
                   from knclient kc
                   inner join kontract kt on kt.k_id = kc.kc_from_k_id
                   inner join kontstate ks on ks.ks_from_k_id = kt.k_id
                   where kc.kc_from_c_id = :c_id and
                         kc.kc_amount_approved > 0 and
                         kt.k_disb_kind = 6 and
                         ks.ks_state = 1 and
                         ks.ks_ol_is_delete = 0 and
                         ks.ks_date_begin < :ma_date_oper)) then
    begin
      res_error_text = 'По условиям продукта займ могут получить клиенты, которые ранее имели займы по карте М Булак.';
      res_error_is_critical = 1;
      suspend;
    end 
  end
end
