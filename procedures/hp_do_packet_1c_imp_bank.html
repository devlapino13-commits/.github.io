CREATE PROCEDURE HP_DO_PACKET_1C_IMP_BANK(
    UN_ID D_ID,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE,
    ONE_PCB_ID D_ID = 0)
RETURNS (
    RES_TOTAL D_INTEGER,
    RES_IMPORTED D_INTEGER,
    RES_CREATED D_INTEGER,
    RES_LINKED D_INTEGER)
AS
declare variable pcb_id d_integer;
declare variable pcb_date d_date;
declare variable pcb_doc_no d_str_code;
declare variable pcb_detail d_str_255;
declare variable pcb_sign d_smallint;
declare variable pcb_acc_dt_id d_id;
declare variable pcb_acc_kt_id d_id;
declare variable pcb_amount_sum d_currency;
declare variable pcb_amount_val d_currency;
declare variable pcb_amount_tax_dt d_currency;
declare variable pcb_amount_tax_kt d_currency;
declare variable pcb_from_cu_id d_str_lang;
declare variable pcb_memo d_str_long;
declare variable pcb_from_ol_id d_id;
declare variable pci_ol_created d_smallint;
declare variable ol_oper_type d_integer;
declare variable ol_oper_subtype d_integer;
declare variable packet_id d_id;
declare variable ob_client_id d_id;
declare variable al_id d_str_code;
declare variable reg_id d_integer;
declare variable ol_doc_kind d_integer;
declare variable detail_id d_id;
declare variable pcb_detail_prv d_str_255;
declare variable pcb_date_prv d_date;
declare variable c_id d_id;
declare variable ob_summa d_currency;
declare variable pcb_amount d_currency;
declare variable pcb_memo_short d_str_long;
declare variable ob_client_name d_str_255;
declare variable ob_client_inn d_str_code;
declare variable ol_memo d_str_long;
declare variable equal_count d_integer;
declare variable memo_is_ok integer;
declare variable c_name_fio d_str_255;
declare variable c_inn d_str_code;
declare variable rec_count d_integer;
declare variable ol_id_prv d_id;
declare variable client_dt d_id;
declare variable client_kt d_id;
declare variable acc_id d_id;
declare variable done_summa d_currency;
declare variable candidate_ol_id d_id;
declare variable total_count d_integer;
declare variable add_koef d_integer;
begin
  res_total = 0;
  res_imported = 0;
  res_created = 0;
  res_linked = 0;

  pcb_detail_prv = '';
  pcb_date_prv = '01.01.1900';
  if (one_pcb_id is null) then
    one_pcb_id = 0;

  add_koef = 0;
  select count(*)
  from packet_1c_imp_bank
  where pcb_from_ol_id is null and
        pcb_date between :rep_date_begin and :rep_date_end and
        pcb_acc_dt_code <> '47422810000000000001' and
        pcb_acc_kt_code <> '47422810000000000001' and
        pcb_acc_dt_code <> '47422810000000000002' and
        pcb_acc_kt_code <> '47422810000000000002'
  into total_count;

  if (datediff(day, rep_date_begin, rep_date_end) >= 25 and total_count <= 100) then
    add_koef = 1;

  for select pcb_id, pcb_date, pcb_doc_no, pcb_detail, pcb_sign, pcb_acc_dt_id, pcb_acc_kt_id, pcb_amount_sum,
             pcb_amount_val, pcb_amount_tax_dt, pcb_amount_tax_kt, pcb_from_cu_id, pcb_memo, pcb_from_ol_id
      from packet_1c_imp_bank
      where pcb_from_ol_id is null and
            pcb_date between :rep_date_begin and :rep_date_end and
            (:one_pcb_id = 0 or
            pcb_id = :one_pcb_id) and
            pcb_acc_dt_code <> '47422810000000000001' and
            pcb_acc_kt_code <> '47422810000000000001' and
            pcb_acc_dt_code <> '47422810000000000002' and
            pcb_acc_kt_code <> '47422810000000000002'
      order by pcb_date, pcb_amount_sum desc
      into :pcb_id, :pcb_date, :pcb_doc_no, :pcb_detail, :pcb_sign, :pcb_acc_dt_id, :pcb_acc_kt_id, :pcb_amount_sum,
           :pcb_amount_val, :pcb_amount_tax_dt, :pcb_amount_tax_kt, :pcb_from_cu_id, :pcb_memo, :pcb_from_ol_id
  do
  begin
    res_total = res_total + 1;

    if (pcb_acc_dt_id is null or pcb_acc_kt_id is null) then
      continue;

    if (position('НДС', pcb_memo) > 0) then
      pcb_memo_short = left(pcb_memo, position('НДС', pcb_memo) - 1);
    else
      pcb_memo_short = pcb_memo;
    pcb_memo_short = trim(pcb_memo_short);

    candidate_ol_id = null;
    ol_id_prv = null;

    if (pcb_sign = 1) then
    begin
-- приход
      ol_oper_type = 2;
      ol_oper_subtype = 6;
      ol_doc_kind = 7;

      select acc_from_reg_id
      from ref_account
      where acc_id = :pcb_acc_dt_id
      into :reg_id;

      al_id = null;
      acc_id = null;
      select al_id, al_from_acc_id
      from acc_list
      where al_from_acc_id = :pcb_acc_dt_id and
            al_from_reg_id = :reg_id
      into :al_id, :acc_id;

      c_id = null;
      c_name_fio = '';
      c_inn = '';
      select c_id, c_name_fio, c_inn
      from ref_account, clients
      where acc_id = :pcb_acc_kt_id and
            acc_object_kind = 0 and
            acc_object_id = c_id
      into :c_id, c_name_fio, c_inn;

      rec_count = 0;
      for select iif(ob.ob_client_kt = :c_id, 1, 0) +
                 iif(ob.ob_summa = :pcb_amount_sum, 1, 0) +
                 iif(ol.ol_memo starting :pcb_memo, 1, 0) +
                 iif(ol.ol_memo starting :pcb_memo_short, 1, 0) +
                 :add_koef as equal_count,
                 ol.ol_id, ob.ob_summa, ol.ol_memo, ob.ob_client_kt, c_name_fio, c_inn
          from operlist ol
          inner join operbook ob on (ob.ob_from_ol_id = ol.ol_id)
          left join clients on (ob.ob_client_kt = c_id)
          where ob.ob_ol_is_active = 1 and
                ob.ob_ol_date = :pcb_date and
                ob.ob_account_dt = :al_id and
                ob.ob_summa >= :pcb_amount_sum and
                (ob.ob_client_kt = :c_id or
                ob.ob_summa = :pcb_amount_sum or
                ol.ol_memo starting :pcb_memo_short)
          order by 1 desc
          into equal_count, pcb_from_ol_id, ob_summa, ol_memo, ob_client_id, ob_client_name, ob_client_inn
      do
      begin
        memo_is_ok = 0;
        if (ol_memo starting pcb_memo_short) then
          memo_is_ok = 1;
        if (pcb_memo starting ol_memo) then
          memo_is_ok = 1;

        if (memo_is_ok = 0) then
          continue;

        if (exists(select *
                   from kn_opermoney
                   where kom_from_ol_id = :pcb_from_ol_id and
                         kom_is_done = 1)) then
        begin
          if (ob_summa = pcb_amount_sum) then
          begin
            candidate_ol_id = pcb_from_ol_id;
            pcb_from_ol_id = null;
            break;
          end

          if (ob_summa > pcb_amount_sum) then
          begin
            done_summa = 0;
            select sum(ob_summa)
            from operbook
            where ob_from_ol_id = :pcb_from_ol_id and
                  ob_account_dt = :al_id
            into :done_summa;
            if (done_summa is null) then
              done_summa = 0;

            pcb_amount = 0;
            select sum(kob_summa)
            from kn_operbook
            where kob_from_ol_id = :pcb_from_ol_id and
                  (:acc_id is null or
                  kob_acc_dt = :acc_id)
            into :pcb_amount;
            if (pcb_amount is null) then
              pcb_amount = 0;

            if (pcb_amount_sum <= (done_summa - pcb_amount)) then
              break;
            else
            begin
              candidate_ol_id = pcb_from_ol_id;
              pcb_from_ol_id = null;
              continue;
            end
          end
        end

        if (equal_count >= 3) then
          break;

        if (c_name_fio = '' and
            ob_summa = pcb_amount_sum and
            memo_is_ok = 1) then
          break;

        if (ob_client_name is null and
            ob_summa = pcb_amount_sum and
            memo_is_ok = 1) then
          break;

        if (ob_summa = pcb_amount_sum or (ob_summa > pcb_amount_sum and
            memo_is_ok = 1)) then
        begin
          if (ob_client_id = c_id and c_id is not null) then
            break;
          else
          begin
-- проверка совпадения клиента по доп.параметрам
            if (c_name_fio = ob_client_name and
                c_inn <> '' and
                c_inn = ob_client_inn) then
              break;

            if (c_name_fio = ob_client_name and
                ob_summa = pcb_amount_sum) then
              break;

            if (ob_client_id is not null and
                c_id is not null and
                c_inn <> '' and
                c_inn = ob_client_inn) then
              break;

            if (c_name_fio <> ob_client_name and
                ob_summa = pcb_amount_sum) then
            begin
              if (exists(select *
                         from bankclient c, bankclient ob
                         where c.bc_from_c_id = :c_id and
                               ob.bc_from_c_id = :ob_client_id and
                               c.bc_account_no = ob.bc_account_no)) then
                break;
            end
          end
        end
        rec_count = rec_count + 1;
        ol_id_prv = pcb_from_ol_id;
        pcb_from_ol_id = null;
      end

      if (rec_count = 1) then
        ol_id_prv = null;

      if (pcb_from_ol_id is null and
          candidate_ol_id is null and
          ol_id_prv is not null and
          rec_count = 1 and
          memo_is_ok = 1) then
            pcb_from_ol_id = ol_id_prv;
    end
    else
    begin
-- расход
      ol_oper_type = 2;
      ol_oper_subtype = 7;
      ol_doc_kind = 6;

      reg_id = null;
      select acc_from_reg_id
      from ref_account
      where acc_id = :pcb_acc_kt_id
      into :reg_id;

      al_id = null;
      acc_id = null;
      select al_id, al_from_acc_id
      from acc_list
      where al_from_acc_id = :pcb_acc_kt_id and
            al_from_reg_id = :reg_id
      into :al_id, :acc_id;

      c_id = null;
      c_name_fio = '';
      c_inn = '';
      select c_id, c_name_fio, c_inn
      from ref_account, clients
      where acc_id = :pcb_acc_dt_id and
            acc_object_kind = 0 and
            acc_object_id = c_id
      into :c_id, c_name_fio, c_inn;

      rec_count = 0;
      for select iif(ob.ob_client_dt = :c_id, 1, 0) +
                 iif(ob.ob_summa = :pcb_amount_sum, 1, 0) +
                 iif(ol.ol_memo starting :pcb_memo, 1, 0) +
                 iif(ol.ol_memo starting :pcb_memo_short, 1, 0) +
                 :add_koef as equal_count,
                 ol.ol_id, ob.ob_summa, ol.ol_memo, ob.ob_client_dt, c_name_fio, c_inn
          from operlist ol
          inner join operbook ob on (ob.ob_from_ol_id = ol.ol_id)
          left join clients on (ob.ob_client_dt = c_id)
          where ob.ob_ol_is_active = 1 and
                ob.ob_ol_date = :pcb_date and
                ob.ob_account_kt = :al_id and
                ob.ob_summa >= :pcb_amount_sum and
                (ob.ob_client_dt = :c_id or
                ob.ob_summa = :pcb_amount_sum or
                ol.ol_memo starting :pcb_memo_short)
          order by 1 desc
          into equal_count, pcb_from_ol_id, ob_summa, ol_memo, ob_client_id, ob_client_name, ob_client_inn
      do
      begin
        memo_is_ok = 0;
        if (ol_memo starting pcb_memo_short) then
          memo_is_ok = 1;
        if (pcb_memo starting ol_memo) then
          memo_is_ok = 1;

        if (exists(select *
                   from kn_opermoney
                   where kom_from_ol_id = :pcb_from_ol_id and
                         kom_is_done = 1)) then
        begin
          if (ob_summa = pcb_amount_sum) then
          begin
            candidate_ol_id = pcb_from_ol_id;
            pcb_from_ol_id = null;
            break;
          end

          if (ob_summa > pcb_amount_sum) then
          begin
            done_summa = 0;
            select sum(ob_summa)
            from operbook
            where ob_from_ol_id = :pcb_from_ol_id and
                  ob_account_kt = :al_id
            into :done_summa;
            if (done_summa is null) then
              done_summa = 0;

            pcb_amount = 0;
            select sum(kob_summa)
            from kn_operbook
            where kob_from_ol_id = :pcb_from_ol_id and
                  (:acc_id is null or
                  kob_acc_kt = :acc_id)
            into :pcb_amount;
            if (pcb_amount is null) then
              pcb_amount = 0;

            if (pcb_amount_sum <= (done_summa - pcb_amount)) then
              break;
            else
            begin
              candidate_ol_id = pcb_from_ol_id;
              pcb_from_ol_id = null;
              continue;
            end
          end
        end

        if (equal_count >= 3) then
          break;

        if (c_name_fio = '' and
            ob_summa >= pcb_amount_sum and
            memo_is_ok = 1) then
          break;

        if (ob_client_name is null and
            ob_summa = pcb_amount_sum and
            memo_is_ok = 1) then
          break;

        if (ob_summa = pcb_amount_sum or (ob_summa > pcb_amount_sum and
            memo_is_ok = 1)) then
        begin
          if (ob_client_id = c_id and c_id is not null) then
            break;
          else
          begin
-- проверка совпадения клиента по доп.параметрам
            if (c_name_fio = ob_client_name and
                c_inn <> '' and
                c_inn = ob_client_inn) then
              break;

            if (c_name_fio = ob_client_name and
                ob_summa = pcb_amount_sum) then
              break;

            if (ob_client_id is not null and
                c_id is not null and
                c_inn <> '' and
                c_inn = ob_client_inn) then
              break;

            if (c_name_fio <> ob_client_name and
                ob_summa = pcb_amount_sum) then
            begin
              if (exists(select *
                         from bankclient c, bankclient ob
                         where c.bc_from_c_id = :c_id and
                               ob.bc_from_c_id = :ob_client_id and
                               c.bc_account_no = ob.bc_account_no)) then
                break;
            end
          end
        end
        rec_count = rec_count + 1;
        ol_id_prv = pcb_from_ol_id;
        pcb_from_ol_id = null;
      end

      if (rec_count = 1) then
        ol_id_prv = null;

      if (pcb_from_ol_id is null and
          candidate_ol_id is null and
          ol_id_prv is not null and
          rec_count = 1 and
          memo_is_ok = 1) then
            pcb_from_ol_id = ol_id_prv;
    end

    if (pcb_from_ol_id is null and
        candidate_ol_id is not null) then
    begin
      res_linked = res_linked + 1;
      update packet_1c_imp_bank
      set pcb_from_ol_id = :candidate_ol_id
      where pcb_id = :pcb_id;
    end

    if (pcb_from_ol_id is null) then
      continue;

    update packet_1c_imp_bank
    set pcb_from_ol_id = :pcb_from_ol_id
    where pcb_id = :pcb_id;

    pci_ol_created = 0;

    if (pcb_date_prv <> pcb_date) then
    begin
      packet_id = null;

      select pck_id
      from packet_1c
      where pck_date = :pcb_date and
            pck_kind = -1
      into packet_id;

      if (packet_id is null) then
      begin
        insert into packet_1c (pck_date, pck_kind, pck_invalid, pck_excel_done, pck_un_id)
        values (:pcb_date, -1, 0, 0, :un_id)
        returning pck_id
        into :packet_id;
      end
    end

    if (pcb_from_ol_id is null) then
    begin
      insert into operlist (ol_from_reg_id, ol_oper_type, ol_date, ol_doc_no_str, ol_doc_kind, ol_memo, ol_user_id,
                            ol_document_client, ol_oper_subtype, ol_is_client)
      values (:reg_id, :ol_oper_type, :pcb_date, :pcb_doc_no, :ol_doc_kind, :pcb_memo, :un_id, :ob_client_id,
              :ol_oper_subtype, iif(:ob_client_id is null, 0, 1))
      returning ol_id
      into :pcb_from_ol_id;

      pci_ol_created = 1;
    end

    insert into packet_1c_imp (pci_pck_id, pci_ol_id, pci_ol_created)
    values (:packet_id, :pcb_from_ol_id, :pci_ol_created);

    if (pcb_detail_prv <> pcb_detail or detail_id is null) then
    begin
      detail_id = null;

      select obd_id
      from kn_obdetail
      where obd_detail = :pcb_detail
      into detail_id;

      if (detail_id is null) then
      begin
        insert into kn_obdetail (obd_detail)
        values (:pcb_detail)
        returning obd_id
        into :detail_id;
      end

      pcb_detail_prv = pcb_detail;
    end

    client_dt = null;
    client_kt = null;
    if (pcb_sign = -1) then
      client_dt = ob_client_id;
    if (pcb_sign = 1) then
      client_kt = ob_client_id;

    delete from kn_operbook
    where kob_from_ol_id = :pcb_from_ol_id;

    insert into kn_operbook (kob_from_ol_id, kob_currency, kob_acc_dt, kob_acc_kt, kob_summa, kob_summa_val,
                             kob_summa_tax_dt, kob_summa_tax_kt, kob_ol_date, kob_ol_reg_id, kob_ol_is_active,
                             kob_detail_id, kob_client_dt, kob_client_kt, kob_from_kc_id)
    values (:pcb_from_ol_id, :pcb_from_cu_id, :pcb_acc_dt_id, :pcb_acc_kt_id, :pcb_amount_sum, :pcb_amount_val,
            :pcb_amount_tax_dt, :pcb_amount_tax_kt, :pcb_date, :reg_id, 1, :detail_id, :client_dt, :client_kt, null);

    update packet_1c_imp_bank
    set pcb_from_ol_id = :pcb_from_ol_id,
        pcb_ol_created = :pci_ol_created,
        pcb_link_kind = 1
    where (pcb_id = :pcb_id);

    update kn_opermoney
    set kom_is_done = 1
    where kom_from_ol_id = :pcb_from_ol_id;

    res_imported = res_imported + 1;
    res_created = res_created + pci_ol_created;
  end

  suspend;
end
