CREATE PROCEDURE FIND_LAWLIST(
    FIND_STR VARCHAR(100),
    PL_ID DECIMAL(18,0))
RETURNS (
    LAL_ID TYPE OF COLUMN LAWLIST.LAL_ID,
    LAL_NAME TYPE OF COLUMN LAWLIST.LAL_NAME,
    LAL_FROM_ADL_ID TYPE OF COLUMN LAWLIST.LAL_FROM_ADL_ID,
    LAL_PHONE TYPE OF COLUMN LAWLIST.LAL_PHONE,
    LAL_EMAIL TYPE OF COLUMN LAWLIST.LAL_EMAIL,
    LAL_WEB TYPE OF COLUMN LAWLIST.LAL_WEB,
    LAL_FIO TYPE OF COLUMN LAWLIST.LAL_FIO,
    LAL_ADDRESS_STR VARCHAR(255),
    ADL_PL_ID DECIMAL(18,0))
AS
begin
  if (find_str is null) then
    find_str = '';
  if (pl_id is null) then
    pl_id = 0;
  if ((find_str <> '') or (pl_id <> 0)) then
  begin
    for select lal_id, lal_name, lal_from_adl_id, lal_phone, lal_email, lal_web, lal_fio
        from lawlist
        where ((:find_str = '') or
              (:find_str <> '' and
              (coalesce(lal_name, '') like '%' || :find_str || '%' or
              coalesce(lal_phone, '') like '%' || :find_str || '%' or
              coalesce(lal_email, '') like '%' || :find_str || '%' or
              coalesce(lal_web, '') like '%' || :find_str || '%' or
              coalesce(lal_fio, '') like '%' || :find_str || '%')))
        into :lal_id, :lal_name, :lal_from_adl_id, :lal_phone, :lal_email, :lal_web, :lal_fio
    do
    begin
      lal_address_str = '';
      adl_pl_id = 0;
      if (lal_from_adl_id is not null) then
      begin
        select adl_address_str, adl_from_pl_id
        from addrlist
        where adl_id = :lal_from_adl_id
        into :lal_address_str, adl_pl_id;
        if (adl_pl_id is null) then
          adl_pl_id = 0;
      end
      if ((adl_pl_id <> 0) and
          (pl_id <> 0)) then
      begin
        if (exists(select *
                   from places_list(:adl_pl_id, 1, -1, 0)
                   where id_list = :pl_id)) then
          suspend;
      end
      else
        suspend;
    end
  end
end
