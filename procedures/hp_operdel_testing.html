CREATE PROCEDURE HP_OPERDEL_TESTING(
    USE_ODM_ID D_INTEGER,
    OLD_K_ID D_ID,
    OLD_OL_ID D_ID,
    NEW_K_ID D_ID,
    NEW_AMOUNT D_CURRENCY,
    UN_ID D_ID)
RETURNS (
    ODM_ID D_INTEGER,
    RES_IS_ERROR D_BOOL,
    RES_TEXT D_STR_255)
AS
begin
  -- Процедура для тестирования модуля "Аннулирование кредитных операций"

  -- Добавление главной операции которую нужно аннулировать
  if (coalesce(use_odm_id, 0) = 0) then
  begin
    if (old_k_id is null or :old_ol_id is null) then
      exception e_message 'Params is null';

    execute procedure hp_operdelmain_add(:old_k_id, :old_ol_id, 'Тестирование', :un_id);
  
    -- Получение ODM_ID только что созданной заявки
    select odm_id
    from operdelmain
    order by odm_id desc
    rows 1
    into odm_id;
  end else
    odm_id = use_odm_id;

  -- Добавление других займов для переноса погашений из главной операции
  if (new_k_id is not null and coalesce(new_amount, 0) > 0) then
    execute procedure hp_operdellist_add(:odm_id, :new_k_id, :new_amount);
  
  -- Подтверждение заявки на аннулирование
  execute procedure hp_operdelmain_action(:odm_id, 'approve', :un_id);
  
  -- Выполнение заявки на аннулирование операции
  select res_is_error, res_text
  from hp_operdelmain_execute(:odm_id, :un_id)
  into res_is_error, res_text;

  suspend;
end
