CREATE PROCEDURE HP_EDOC_INFOLIST_CREATE(
    MA_ID D_ID,
    UN_ID D_ID)
AS
declare variable k_id d_id;
declare variable edc_inf_id d_integer;
declare variable last_status d_smallint;
declare variable res_text d_str_255;
begin
  -- RFCKG-5208: создание инфо-листа при подписании заявки на займ ИПФ / STANDART. Вызывается из HP_EDOC_AFTER_SIGN_ACTION

  k_id = null;

  select first 1 k.k_id
  from kontract k
  inner join product p on p.pr_id = k.k_from_pr_id
  where k.k_from_ma_app_id = :ma_id and
        (p.pr_code containing '.ipf.' or p.pr_code containing '.STANDART.' or p.pr_code containing '.IPF.')
  into :k_id;

  if (coalesce(k_id, 0) <> 0 and
      exists(
        select 1
        from knclient kc
        where kc.kc_from_k_id = :k_id
          and kc.kc_amount_approved > 0
          and kc.kc_loan_target = f_get_rd_id('Loan_Target.Consumer'))
  ) then
  begin
    edc_inf_id = null;
    select edc.edc_id
    from edoc_cfg edc
    where edc.edc_code = 'loan.application_new_conditions'
    into :edc_inf_id;

    last_status = null;
    select first 1 edl.edl_status
    from edoc_list edl
    where edl.edl_object_kind = f_objkind('kontract','id')
      and edl.edl_object_id = :k_id
      and edl.edl_from_edc_id = :edc_inf_id
    order by edl.edl_id desc
    into :last_status;

    -- если нет активного (0) → создаём
    if (last_status is null or last_status <> 0) then
    begin
      select res_text
      from hp_edoc_insert('loan.application_new_conditions', f_objkind('kontract','id'), :k_id, :un_id)
      into :res_text;

      if (res_text is not null and res_text <> '') then
        exception e_message(res_text);
    end
  end
end
