CREATE PROCEDURE HP_PDN_GETINFO_ADD(
    KC_ID D_ID)
RETURNS (
    SCORING_CNT D_SMALLINT,
    PDN_LIMIT D_CURRENCY,
    CURR_REG D_STR_100,
    CURR_ROSSTAT_INCOME D_CURRENCY,
    CURR_ANKETA_INCOME D_CURRENCY,
    MULTIPLIER D_CURRENCY,
    CURR_ROSSTAT_INCOME_M D_CURRENCY,
    CURR_ANKETA_INCOME_M D_CURRENCY,
    SP_NEW D_CURRENCY,
    SP_OLD D_CURRENCY,
    SP D_CURRENCY,
    PDN D_CURRENCY,
    PDN_NEW D_CURRENCY,
    INCOME_NEW D_CURRENCY,
    INCOME_OTHER D_CURRENCY,
    INCOME_ALL D_CURRENCY,
    KC_DATE D_DATE)
AS
declare variable kcp_id d_id;
declare variable c_id d_id;
declare variable k_id d_id;
declare variable k_disb_date d_date;
begin
  select kc_from_k_id, kc_from_c_id, k_disb_date
  from knclient, kontract
  where k_id = kc_from_k_id
    and kc_id = :kc_id
  into k_id, c_id, k_disb_date;

  select pdn_limit - 0.10
  from hp_pdn_get_limit(:kc_id)
  into pdn_limit;

  scoring_cnt = f_point_get(9, k_id, c_id, k_disb_date);
  curr_rosstat_income = 0;

  select kcp_id, kc_loan_expense, kc_date
  from knclient_pdn
  where kc_id = :kc_id
    and kcp_kind = 0
  into kcp_id, sp, kc_date;

  if (kcp_id is null) then
    exception e_message 'ПДН еще не рассчитан';

  select res_income_avg, res_place_name
  from hp_extinfolist_pdn_one_income(:kc_id, current_date)
  into curr_rosstat_income, curr_reg;

  select monthly_income
  from hp_knclient_anketa_income(:kc_id, current_date)
  into curr_anketa_income;

  multiplier = iif(k_disb_date >= '01.07.2026', 0.9, 1.0);

  curr_rosstat_income_m = multiplier * curr_rosstat_income;
  curr_anketa_income_m = multiplier * curr_anketa_income;

  select kpl_month_repay_final
  from knclient_pdnlist
  where kpl_from_kcp_id = :kcp_id
    and kpl_rec_no = 0
  into sp_new;

  sp_old = sp - sp_new;
  pdn = round(sp / curr_rosstat_income_m * 100.00, 1);

  if (pdn_limit > 0) then
  begin
    income_all = trunc(sp / (pdn_limit / 100.00), 0) + 1;
    income_other = trunc(sp_old / (pdn_limit / 100.00), 0) + 1;
    income_new = income_all - income_other;
    pdn_new = round(sp / income_all * 100.00, 1);
  end

  suspend;
end
