CREATE PROCEDURE HP_OVERPAYMENT_TRANSFER(
    OVP_ID D_INTEGER NOT NULL)
RETURNS (
    ESM_ID D_ID,
    RES_TEXT D_STR_255)
AS
begin
  res_text = '';

  insert into extsysmain (
    esm_project,
    esm_action,
    esm_date,
    esm_account,
    esm_object_kind,
    esm_object_id,
    esm_amount,
    esm_from_k_id
  )
  select
    'transfer_on_phone',
    'check',
    current_date,
    ovp_phone,
    0,
    ovp_from_c_id,
    ovp_amount,
    ovp_from_k_id
  from overpayment
  where ovp_id = :ovp_id
    and ovp_status in (0, 1)
    and ovp_edl_id > 0
    and coalesce(ovp_phone, '') <> ''
  returning esm_id into :esm_id;

  if (esm_id is null) then
    res_text = f_format('Ошибка создания записи на перевод в ExtSysMain(ovp_id = %1)', :ovp_id);

  suspend;
end
