CREATE PROCEDURE HP_PRINT_DOCRECOG_FIELDS(
    DOC_CODE D_STR_255)
RETURNS (
    DRM_NAME D_STR_255,
    RES_DRM_CODE D_STR_255,
    DRM_API_CODE D_STR_255,
    DRM_IS_MANUAL D_BOOL,
    DRM_RECOG_2 D_BOOL,
    DRM_SHOW_FIELDS_CLIENT D_BOOL,
    DRM_HAS_PAGE1 D_BOOL,
    DRM_HAS_PAGE2 D_BOOL,
    DRM_CAN_LOAD_GALLERY D_BOOL)
AS
declare variable pk_id d_id;
begin
    select
        pk.pk_name,
        drm.drm_code,
        drm.drm_api_code,
        drm.drm_is_manual,
        drm.drm_recog_2,
        drm.drm_show_fields_client,
        pk.pk_id,
        pk.pk_can_load
    from docrecogmain drm
    inner join passkind pk on pk.pk_from_drm_id = drm.drm_id
    where drm.drm_code = :doc_code
    into
        :drm_name,
        :res_drm_code,
        :drm_api_code,
        :drm_is_manual,
        :drm_recog_2,
        :drm_show_fields_client,
        :pk_id,
        :drm_can_load_gallery;

    select res_value
    from mc_doc_name_change(:pk_id, :drm_name)
    into :drm_name;

    select drm_has_page1, drm_has_page2
    from hp_get_drm_has_pages(:pk_id)
    into :drm_has_page1, :drm_has_page2;


    suspend;
end
