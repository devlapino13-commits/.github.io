CREATE PROCEDURE MC_SET_REGISTRATION_DATA(
    MCL_ID D_ID NOT NULL,
    PASSPORT_PHOTO1 D_BLOB_BINARY,
    PASSPORT_PHOTO2 D_BLOB_BINARY,
    BC_ID D_ID,
    SESSION_ID D_STR_100,
    AC_ID D_ID = null)
RETURNS (
    IS_OK D_SMALLINT,
    ERR_TEXT D_STR_255)
AS
declare variable c_id d_id;
declare variable inn d_str_20;
declare variable eic_id d_id;
begin
    is_ok = 1;
    err_text = '';
    
    select c.c_id, c.c_inn
    from mc_login mcl
    inner join clients c on mcl.mcl_from_c_id = c.c_id
    where mcl.mcl_id = :mcl_id
    into :c_id, :inn;

    if (coalesce(c_id, 0) = 0) then
    begin
        is_ok = 0;
        err_text = 'Ошибка при сохранении данных';
        suspend;
        exit;
    end

    update bmg_clients bc
    set bc.bc_from_c_id = :c_id,
        bc.bc_status = 1
    where bc.bc_id = :bc_id;

    eic_id = (select eic.eic_id from extinfocfg eic where eic.eic_code = 'antifraud_client_check');
    update extinfolist eil
    set
        eil.eil_from_c_id = :c_id,
        eil.eil_from_mcl_id = :mcl_id
    where eil.eil_from_eic_id = :eic_id
        and eil.eil_object_kind = f_objkind('antifraud_check', 'id')
        and eil.eil_object_id = :ac_id;

    suspend;
end
