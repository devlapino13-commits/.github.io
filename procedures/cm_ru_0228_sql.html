CREATE PROCEDURE CM_RU_0228_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable disb_date d_date;
declare variable kolich_repay d_integer;
declare variable c_id d_id;
declare variable point d_integer;
declare variable amount_max d_currency;
declare variable amount d_currency;
declare variable pr_id d_currency;
declare variable kol_zalog d_integer;
declare variable cost_market d_currency;
declare variable cli_balance d_currency;
declare variable vin_code d_str_255;
declare variable kol_simvol_vin d_integer;
declare variable k_refund d_integer;
declare variable refund_balance d_currency;
declare variable limit_max d_currency;
declare variable mds d_currency;
begin

  limit_max = 500000;
  -- проверка по продукту Залоговый
  for
  select k_disb_date, k_kolich_repay, pr_id, k_refund from kontract, product
  where k_id = :k_id
  and k_from_pr_id = pr_id
  and pr_code containing '.zalog.'
  into disb_date, kolich_repay, pr_id, k_refund
  do
  begin
    -- если имеется активный займ то запрещено брать залоговый
    select kc_from_c_id, kc_amount_approved from knclient
    where kc_from_k_id = :k_id
    and kc_amount_approved > 0
    rows 1
    into c_id, amount;

    point = f_point_get(9, k_id, c_id, disb_date);

    if (point < 4 and disb_date <= '11.09.2024') then
    begin
      res_error_text = 'У клиента менее 4 бала, выдача запрещается!';
      res_error_is_critical = 2;
      suspend;
    end 

    /*if (point <= 9 and kolich_repay > 12) then
    begin
      res_error_text = 'Бал клиента = '||:point||' срок займа не должно превышать 12 мес.';
      res_error_is_critical = 2;
      suspend;
    end else
    if (point <= 24 and kolich_repay > 18) then
    begin
      res_error_text = 'Бал клиента = '||:point||' срок займа не должно превышать 18 мес.';
      res_error_is_critical = 2;
      suspend;
    end */

    if (exists(select * from kontstate, knclient, kontract, product
    where ks_ol_is_delete = 0
    and ks_state = 1
    and :disb_date between ks_date_begin and ks_date_end
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kc_from_c_id = :c_id
    and ks_from_k_id = k_id
    and k_from_pr_id = pr_id
    and (pr_code containing '.education.' or pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
    )) then
    begin
      res_error_text = 'Имеется активный займ по залоговому или по образованию, выдача запрещается!';
      res_error_is_critical = 2;
      suspend;
    end

    cli_balance = 0;
    select sum(oks_balance)
    from knclient, oksummary
    where kc_from_c_id  = :c_id
    and kc_amount_approved > 0
    and kc_from_k_id = oks_from_k_id
    and oks_balance > 0
    and oks_close_state = 0
    and :disb_date between oks_date_begin and oks_date_end
    into cli_balance;
    cli_balance = coalesce(cli_balance, 0);

    for
    select amount_max
    from perc_amount_interval_by_point(:pr_id, :point, :disb_date)
    into :amount_max
    do
    begin
--      if (amount > amount_max) then
--      begin
--        res_error_text = 'Бал клиента = '||:point||', Сумма клиента не должно превышать:'||round(amount_max,0);
--        res_error_is_critical = 2;
--        suspend;
--      end 
      if (k_refund = 0) then
      begin
        select mds
        from hp_max_amount_allowed(9, :k_id, :amount_max)
        into mds;

        if ((cli_balance + amount > amount_max) and current_date < '23.01.2025') then
        begin
          amount_max = amount_max - cli_balance;
          res_error_text = f_format('Баланс ОС по активным: %1 р. Разрешенная сумма для выдачи: %2, Бал кл: %3', round(cli_balance), round(amount_max), point);
          res_error_is_critical = 2;
          suspend;
        end
        if ((amount > mds) and current_date >= '23.01.2025') then
        begin
          res_error_text = f_format('Баланс ОС по активным: %1 р. Разрешенная сумма для выдачи: %2, Бал кл: %3', round(cli_balance), round(mds), point);
          res_error_is_critical = 2;
          suspend;
        end
      end

      if (k_refund = 1) then
      begin
        refund_balance = 0;
        select sum(oks_balance)
        from knclient, oksummary, kontract, product
        where kc_from_c_id = :c_id and
              kc_amount_approved > 0 and
              kc_from_k_id = oks_from_k_id and
              k_id = oks_from_k_id and
              k_from_pr_id = pr_id and
              pr_code similar to '%(.reliable_client.|.reliable_client_plus.|.loyal.|.shans.|.comfortable_1.|.comfortable_2.)%' and
              oks_balance > 0 and
              oks_close_state = 0 and
              :disb_date between oks_date_begin and oks_date_end
        into refund_balance;
        refund_balance = coalesce(refund_balance, 0);

        if ((cli_balance + amount) > limit_max) then
        begin
          amount_max = amount_max - cli_balance + refund_balance;
          res_error_text = 'Общий баланс всех займов превысит 500 000 р., разрешенная сумма для рефинансирования ' || iif(limit_max - cli_balance < amount_max, round(limit_max - cli_balance), round(amount_max)) || ' р.';
          res_error_is_critical = 2;
          suspend;
        end

        if (((cli_balance - refund_balance) + amount > amount_max) and (cli_balance + amount) <= limit_max) then
        begin
          amount_max = amount_max - cli_balance + refund_balance;
          res_error_text = f_format('Баланс ОС по активным: %1 р. Разрешенная сумма для выдачи: %2, Бал кл: %3', round(cli_balance), round(amount_max), point);
          res_error_is_critical = 2;
          suspend;
        end
      end
    end

    kol_zalog = 0;
    cost_market = 0;
    select count(*), sum(z_cost_market), max(z_document) from zalog
    where z_from_k_id = :k_id
    and z_type = 1
    into kol_zalog, cost_market, vin_code;
    kol_zalog = coalesce(kol_zalog, 0);
    cost_market = coalesce(cost_market, 0);
    vin_code = coalesce(vin_code, '');
    vin_code = trim(vin_code);
    kol_simvol_vin = char_length(vin_code);

    if (kol_zalog = 0) then
    begin
      res_error_text = 'Залог не указан!';
      res_error_is_critical = 2;
      suspend;
    end else
    if (kol_zalog > 1) then
    begin
      res_error_text = 'Указали более одного залога, прошу оставить только один.';
      res_error_is_critical = 2;
      suspend;
    end 
    if (cost_market <= 0) then
    begin
      res_error_text = 'Прошу указать рычночную стоимость залога';
      res_error_is_critical = 2;
      suspend;
    end else
    if (amount > cost_market) then
    begin
      res_error_text = 'Сумма займа не должно превышать рыночную стоимость залога! РСЗ ='||round(cost_market,0);
      res_error_is_critical = 2;
      suspend;
    end

    if (check_kind not containing '.send_kk.') then -- проверка на ВИН на стадии выдачи RFCRU-2169  -- 14.09.24 igusev внес измнения в рамках задачи RFCRU-2367
    begin
      if (vin_code = '') then
      begin
        res_error_text = 'VIN код не указан, прошу указать';
        res_error_is_critical = 2;
        suspend;
      end 
      if (vin_code <> '-') then
      if (kol_simvol_vin not between 7 and 17) then  -- RFCRU-2715
      begin
        res_error_text = 'VIN код не правильно указан, не соответствует кол-во символов';
        res_error_is_critical = 2;
        suspend;
      end
    end
  end

  -- Если у клиента имеется активный залоговый займ то запрещается другим выдачам
  for
  select k_disb_date, k_kolich_repay, kc_from_c_id from kontract, product, knclient
  where k_id = :k_id
  and k_from_pr_id = pr_id
--  and pr_code not containing '.zalog.'
  and (pr_code containing '.education.' or pr_code containing '.ZALOG_CAR.' or pr_code containing '.zalog.')
  and k_id = kc_from_k_id
  and kc_amount_approved > 0
  into disb_date, kolich_repay, c_id
  do
  begin
    if (exists(select * from kontstate, knclient, kontract, product
    where ks_ol_is_delete = 0
    and ks_state = 1
    and :disb_date between ks_date_begin and ks_date_end
    and ks_from_k_id = kc_from_k_id
    and kc_amount_approved > 0
    and kc_from_c_id = :c_id
    and ks_from_k_id = k_id
    and k_from_pr_id = pr_id
    and pr_code containing '.zalog.'
    )) then
    begin
      res_error_text = 'Имеется активный залоговый займ, выдача запрещена!';
      res_error_is_critical = 2;
      suspend;
    end
  end
end
