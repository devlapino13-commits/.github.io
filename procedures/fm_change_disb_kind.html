CREATE PROCEDURE FM_CHANGE_DISB_KIND(
    UN_ID D_ID,
    OBJ_KIND D_STR_100,
    OBJ_ID D_ID,
    DISB_KIND D_SMALLINT)
RETURNS (
    RES_ID D_INTEGER,
    RES_TXT D_STR_255)
AS
declare variable k_id d_id;
declare variable k_disb_kind d_smallint;
declare variable is_online d_smallint;
declare variable k_is_online d_smallint;
declare variable k_status d_smallint;
declare variable rd_id d_integer;
begin
  is_online = 0;
  if (disb_kind = -1) then
  begin
    disb_kind = 5;
    is_online = 1;
  end

  if (disb_kind in (7, 8)) then
    is_online = 1;

  if (obj_kind = 'miniapp') then
  begin
    select k_id, k.k_disb_kind, k.k_is_online, k_status
    from kontract k
    where k_from_ma_app_id = :obj_id and
          k.k_status <> -2
    order by k_disb_date
    rows 1
    into :k_id, k_disb_kind, k_is_online, k_status;

    if (k_id is null) then
    begin
      select m.ma_disb_kind, m.ma_is_online
      from miniapp m
      where ma_id = :obj_id and
            ma_date_end >= '31.12.2999'
      into :k_disb_kind, :k_is_online;
    end
  end
  else
  begin
    select k_id, k_disb_kind, k_is_online, k_status
    from kontract
    where k_id = :obj_id
    into :k_id, :k_disb_kind, :k_is_online, k_status;
  end

  if (k_status = 1) then
  begin
    res_id = -1;
    res_txt = 'Договор уже утвержден. Необходимо вернуть в процесс.';
    suspend;
    exit;
  end

  if ((disb_kind = k_disb_kind) and
      (is_online = k_is_online)) then
  begin
    res_id = -1;
    res_txt = 'Вы не поменяли способ выдачи.';
    suspend;
    exit;
  end

  if (k_id is not null) then
  begin
    if (exists(select *
               from kontstate
               where ks_from_k_id = :k_id and
                     ks_state = 1 and
                     ks_ol_is_delete = 0)) then
    begin
      res_id = -1;
      res_txt = 'Выдача произведена. Нельзя менять способ выдачи';
      suspend;
      exit;
    end

    select rd_id
    from hp_get_check_loan_data(:k_id)
    INTO:rd_id;

    if (exists(select *
               from actionlist
               where acl_object_kind = 9 and
                     acl_object_id = :k_id and
                     acl_status = 1 and
                     acl_from_rd_id = :rd_id)) then
    begin
      res_id = -1;
      res_txt = 'Договор находится на рассмотрении КК. Изменения запрещены.';
      suspend;
      exit;
    end
  end

  if (f_can_disb_on_card(obj_kind, obj_id) = 0) then
  begin
    if (disb_kind not in (1, 7)) then
    begin
      -- После запуска БС СОФР в РФ, выдачи на карту запрещены
      res_id = -1;
      res_txt = 'Способ выдачи может быть только через кассу или СБП.';
      suspend;
      exit;
    end
    else
    if (k_disb_kind <> 7 and disb_kind = 7) then
    begin
      -- Также запрещена смена способа выдачи на СБП
      res_id = -1;
      res_txt = 'Нельзя менять способ выдачи на СБП. Для выдачи по СБП откажите в текущей заявке и оформите новую.';
      suspend;
      exit;
    end
  end

  res_id = 1;

  update miniapp m
  set ma_disb_kind = :disb_kind,
      m.ma_is_online = :is_online,
      m.ma_from_un_id = :un_id
  where ma_id = :obj_id and
        ma_date_end >= '31.12.2999';

  if (k_id is not null) then
  begin
    update kontract k
    set k.k_disb_kind = :disb_kind,
        k.k_is_online = :is_online,
        k_user_id = :un_id
    where k_id = :obj_id;

    if (k_disb_kind = 7) then
      execute procedure hp_fps_resend_kk(k_id, un_id);
  end

  suspend;
end
