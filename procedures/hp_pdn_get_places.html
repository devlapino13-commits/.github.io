CREATE PROCEDURE HP_PDN_GET_PLACES
RETURNS (
    PL_ID D_ID,
    PL_NAME D_STR_255)
AS
begin
  for select distinct str_from_pl_id
  from places, statregion
  where str_from_pl_id = pl_id
  order by str_id
  into pl_id
  do
  begin
    pl_name = null;

    select f_format('%1 %2 (%3)', plt_short_name, pl_name, str_id)
    from places, statregion, placetype
    where str_from_pl_id = :pl_id
      and str_from_pl_id = pl_id
      and plt_id = pl_status
    order by str_id desc
    rows 1
    into pl_name;

    suspend;
  end
end
