CREATE PROCEDURE HP_SIMPLEVERIFY_SET_STATUS(
    ACL_ID D_ID,
    TRG_KIND D_STR_CODE)
AS
declare variable rd_code d_str_code;
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable acl_status d_smallint;
declare variable acl_respons_c_id d_id;
declare variable obj_code d_str_code;
begin
  select rd_code, acl_object_kind, acl_object_id, acl_status, acl_respons_c_id
  from actionlist, refdata
  where acl_id = :acl_id and
        acl_from_rd_id = rd_id and
        rd_str_id starting 'SimpleVerify_'
  into rd_code, obj_kind, obj_id, acl_status, acl_respons_c_id;

  if (rd_code is null) then
    Exit;

  obj_code = f_objkind(obj_kind, 'name');

  if (rd_code = 'ActionGroup.QR_Photo_Verify' and trg_kind in ('insert', 'update')) then
  begin
    if (obj_code <> 'qr_photo') then
      exception e_default f_format('Incorrect obj_kind, acl_id %1, obj_kind: %2', acl_id, obj_kind);

    if (acl_status = 2) then
    begin
      execute procedure hp_simpleverify_set_verified(:obj_id, acl_respons_c_id);
    end else
    begin
      update qr_list
      set qrl_state = :acl_status
      where qrl_id = :obj_id;
    end
  end
end
