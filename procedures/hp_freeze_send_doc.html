CREATE PROCEDURE HP_FREEZE_SEND_DOC
AS
declare variable K_ID ID_TYPE;
declare variable EDL_ID D_ID;
declare variable RES_TEXT STR_255;
declare variable C_ID ID_TYPE;
declare variable OBJ_KIND D_INTEGER;
declare variable HAS_STOP_CREDIT D_SMALLINT;
declare variable STOP_CREDIT_ANSWER_DATE D_DATE;
declare variable client_was_found_antifraud d_smallint;
declare variable afcl_id d_id;
begin
  for select fzm_from_k_id
      from freeze_main
      where fzm_datetime_end <= current_timestamp and
            fzm_edoc_send = 0
      into k_id
  do
  begin
    obj_kind = f_objkind('kontract', 'id');
    has_stop_credit = 0;

    if (f_is_stopcred_doc_loaded(:obj_kind, :k_id) <> 1) then -- проверка на прикрепление документа
    begin
      -- читаем статус запроса
      select status, answer_date
      from hp_stop_credit_query_status(:obj_kind, :k_id)
      into :has_stop_credit, :stop_credit_answer_date;

      -- если запрос неактуален - делаем запрос
      if (f_is_stopcred_actual(:stop_credit_answer_date) <> 1) then
      begin
        execute procedure hp_extinfo_insert('stop_credit', :obj_kind, :k_id, 0);
        continue;
      end

      if (:has_stop_credit not in (0,1)) then
        continue;
    end

    -- RFCKZ-192
    for select kc_from_c_id
        from knclient
        where kc_from_k_id = :k_id and
              kc_amount_approved > 0
        into c_id
    do
    begin
        client_was_found_antifraud = iif (
            exists (select c_name_fio
                    from hp_check_client_fio_ft(:c_id, 1)
                   ), 1, 0);        

        select afcl_id from hp_antifraud_checks_link_insert(:c_id, f_objkind('kontract', 'id'), :k_id, f_enum_id('antifraud_check_type', 'internal'),
                                                            f_enum_id('antifraud_check_stage', 'freeze_send_doc'), :client_was_found_antifraud)
        into :afcl_id;

        if (client_was_found_antifraud = 1 or has_stop_credit = 1) then
        begin
           delete from freeze_main fzm
           where fzm.fzm_from_k_id = :k_id;

           rdb$set_context('USER_TRANSACTION', 'FREEZE_APPROVE_CANCEL_K_ID', :k_id);

           update kontract k
           set k.k_status = -1 -- Отказ
           where k.k_id = :k_id;

           if (has_stop_credit = 1) then
             -- если по стоп-кредиту отказ, то причина "Стоп-кредит" (krl = 4)
             insert into kontreslist (krl_from_kr_id, krl_from_un_id, krl_from_k_id, krl_ma_reg_id, krl_ma_id,
                                      krl_datetime, krl_date_old, krl_date_new, krl_from_fs)
             values (4, 1000000000000, :k_id, 0, null, current_timestamp, current_date, current_date, 0);
          exit;
        end 
    end

    select edl_id, res_text
    from hp_edoc_ins('loan_disb_consent', 'kontract', :k_id, 1000000000000)
    into edl_id, res_text;

    if (edl_id is not null and
        coalesce(res_text, '') = '') then
    begin
      update freeze_main
      set fzm_edoc_send = 1
      where (fzm_from_k_id = :k_id);

      for select kc_from_c_id
          from knclient
          where kc_from_k_id = :k_id and
                kc_amount_approved > 0
          into c_id
      do
      begin
        execute procedure mc_send_push('clients', c_id, 'Уважаемый клиент!',
            'Пожалуйста, подпишите согласие на получение микрокредита.',
            7);

        execute procedure hp_send_message('sms', c_id, f_cont_info('clients', c_id, 'cell_main', 'ok'),
            'hp_freeze_send_doc',
            'Период охлаждения завершён. Пожалуйста, подпишите согласие на получение микрокредита для продолжения оформления в КМП.',
            '', 7, 1000000000000);
      end
    end
  end
end
