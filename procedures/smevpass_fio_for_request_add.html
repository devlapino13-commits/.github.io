CREATE PROCEDURE SMEVPASS_FIO_FOR_REQUEST_ADD(
    C_ID D_ID,
    FIRST_NAME D_STR_255,
    LAST_NAME D_STR_255,
    PATRONYMIC D_STR_255,
    UN_ID D_ID)
RETURNS (
    ERROR_TEXT D_STR_255)
AS
declare variable is_match d_boolean;
begin
  error_text = '';
  is_match = false;

  if (c_id is null) then
  begin
    error_text = 'Выберите клиента';
    suspend;
    exit;
  end 

  if (coalesce(first_name || last_name, '') = '') then
  begin
    error_text = 'Заполните имя и фамилию';
    suspend;
    exit;
  end 

  select name_last is not distinct from :last_name and name_first is not distinct from :first_name and name_middle is not distinct from :patronymic
  from hp_smevpass_fio_main_ux(:c_id)
  into is_match;

  if (is_match) then
  begin
    error_text = 'ФИО не изменили';
    suspend;
    exit;
  end 

  insert into smevpass_fio_for_request (sfr_from_c_id, sfr_first_name, sfr_last_name, sfr_patronymic, sfr_from_un_id)
  values (:c_id, :first_name, :last_name, :patronymic, :un_id);
end
