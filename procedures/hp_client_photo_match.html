CREATE PROCEDURE HP_CLIENT_PHOTO_MATCH(
    ACL_ID D_ID)
RETURNS (
    ID D_INTEGER,
    NAME D_STR_100,
    PHOTO_BLS_ID D_ID,
    DOC_BLS_ID D_ID,
    KOEF D_PERCENT,
    PERC NUMERIC(15,2),
    PHOTO_BLS D_BLOB_BINARY,
    DOC_BLS D_BLOB_BINARY)
AS
declare variable obj_kind d_integer;
declare variable obj_id d_id;
declare variable status d_integer;
begin
  select
    acl_object_kind,
    acl_object_id
  from actionlist
  where acl_id = :acl_id
    and acl_from_rd_id in (
      f_get_rd_id('ActionGroup.Photo_Verify'),
      f_get_rd_id('ActionGroup.Photo_Verify_Issue'))
  into
    :obj_kind,
    :obj_id;

  if (obj_kind = 0) then
  begin
    name = (select c_name_fio
      from clients where c_id = :obj_id);
    select first 1
      frl_id,
      frl_photo_bls_id,
      frl_doc_bls_id,
      frl_koef,
      (1 - frl_koef / 0.6) * 100
    from facerecoglist
    where frl_from_c_id = :obj_id
    order by frl_id desc
    into
      :id,
      :photo_bls_id,
      :doc_bls_id,
      :koef,
      :perc;
  end else begin
    select first 1
      vwp_id,
      c_name_fio || ' | ' || k_kontract_no,
      vwp_wcam_bls_id,
      vwp_grs_bls_id,
      vwp_status
    from verify_webcam_photo
      inner join kontract
        on k_id = vwp_k_id
      inner join clients
        on c_id = vwp_c_id
    where vwp_k_id = :obj_id
    order by vwp_id desc
    into
      :id,
      :name,
      :photo_bls_id,
      :doc_bls_id,
      :status;
    if (status <> 0) then
      id = null;
    koef = null;
    perc = null;
  end

  if (id > 0) then
  begin
    if (perc < 0) then
      perc = 0;
    photo_bls = (select blx_blob from hp_get_blobex_blob(:photo_bls_id));
    doc_bls = (select blx_blob from hp_get_blobex_blob(:doc_bls_id));
    suspend;
  end else
    exception e_message
      f_format(' Клиент %1; фото для сверки не определены.',
        coalesce(name, ''));
end
