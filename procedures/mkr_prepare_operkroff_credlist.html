CREATE PROCEDURE MKR_PREPARE_OPERKROFF_CREDLIST(
    REPORT_DATE_BEGIN DATE,
    REPORT_DATE_END DATE)
RETURNS (
    CR_TYPE SMALLINT,
    KNC_ID ID_TYPE,
    ID STR_255,
    NAME STR_255,
    PLACEOFBIRTH STR_255,
    PINCODE STR_255,
    COUNTRYCODE VARCHAR(3),
    BANKRUPTCYSTATUS SMALLINT,
    DATEOFBIRTH DATE,
    K_ID ID_TYPE,
    ACCOUNTNO STR_255,
    CREDITTYPE VARCHAR(3),
    CREDITLINEAMOUNT CURR_TYPE,
    PURPOSEOFCREDIT VARCHAR(3),
    CURRENCYOFCREDIT VARCHAR(3),
    INITIALAMOUNTOFCREDIT CURR_TYPE,
    DISOUTAMOUNTOFCREDIT CURR_TYPE,
    ANNUALINTERESTRATE PERC_TYPE,
    DATEOFGRANT DATE,
    DUETIMEFIRSTCONTRACT DATE,
    DUETIMELASTCONTRACT DATE,
    LASTPAYMENTDATE DATE,
    MONTHLYPAYMENTAMOUNT CURR_TYPE,
    DAYSMAINSUMISOVERDUE INTEGER,
    DAYSINTERESTISOVERDUE INTEGER,
    OIAFORREPPERIOD CURR_TYPE,
    OIAFORLASTTHREEMONTHS CURR_TYPE,
    NUMBEROFPROLONGS INTEGER,
    CREDITCLASSCODE VARCHAR(3),
    CREDITSTATUSCODE VARCHAR(3),
    PROVISION CURR_TYPE,
    CREDITPERIODINMONTHS INTEGER,
    KC_NUMBER INTEGER)
AS
declare variable KNC_BUSINESS_SUBTYPE_ID integer;
declare variable K_AMOUNT_APPROVED CURR_TYPE;
declare variable IS_FINISH integer;
declare variable SH_DATE_TMP date;
declare variable SUMM_TMP CURR_TYPE;
declare variable K_CURRENCY varchar(3);
declare variable LASTPAYMENTDATE_OPERKRED date;
declare variable LASTPAYMENTDATE_OPERKROFF date;
declare variable MONTHLYPAYMENTAMOUNT_GROUP CURR_TYPE;
declare variable K_CURRENCY_OLD varchar(3);
declare variable PR_ID integer;
declare variable K_ID_OLD D_ID;
declare variable C_ID D_ID;
declare variable BASE_K_ID D_ID;
declare variable LOAN_CLOSED_DATE D_DATE;
declare variable LASTWRITEOFF_OPERKROFF D_DATE;
declare variable OLD_DISOUTAMOUNTOFCREDIT CURR_TYPE;
declare variable DAYSMAINSUMISOVERDUE_FIX D_INTEGER;
begin
  /*
      CREATED BY IGUSEV 20.05.2016
      PURPOSE: Для МКР. формирование данных по списнным кредитам на внебалансовый учет

      ПОЛЯ КОТОРЫЕ НЕ ПЕРЕДАЮ  (ПОЛЯ НЕ ОБЯЗАТЕЛЬНЫЕ ДЛЯ ЗАПОЛНЕНИЯ)
      PlaceOfBirth   -  Место рождения кредитополучателя
      PinCode        -  Пин код кредитополучателя


      <PurposeOfCredit>
      001    Потребительские кредиты физическим лицам   <== Выставил этот сектор для всех кредитов
      002    Нефтяной сектор
      003    Ипотечные кредиты физическим лицам
      004    Сектор торговли и обслуживания
      005    Сектор Энергетики, химии и природных ресурсов
      006    Сельскохозяйственный сектор сектор обработки
      007    Строительный и имущественный сектор
      008    Промышленный и производственный сектор
      009    Сектор транспорта и связи
      010    Другие сектора

      <CreditClassCode>
      001    Обеспеченные наличными средствами
      002    Удовлетворительные кредиты
      003    Подконтрольные кредиты
      004    Неудовлетворительные кредиты
      005    Опасные кредиты
      006    Безнадежные кредиты

      <CreditStatusCode>    было до 18.09.2015
      Если реструкторизирован то 004, в простивном случае 006
      001    Полностью оплаченные
      002    С неучтенными процентами
      003    Списанные с баланса
      004    Реструктуризованные
      005    Реструктуризованные и  с неучтенными процентами
      006    Другие статусы

      с  18.09.2015
      007 - iwlek - рабочие кредиты - актиные кредиты с просрочкой до 89 дней
      008 - qeyri iwlek - нерабочие - активные кредиты с просрочкой 90 или более дней
      003 - silinmis - списанные - списанные с баланса компании
      001 - bagl? - закрытые кредиты

      <CreditType>
      001    Кредит
      002    Кредитная линия
      003    Кредитная карта
      004    гарантия
    */

--  Значения которые будут выставлены для всех кредитов----------------------------------------
  placeofbirth = 'Azerbaycan';
  credittype = '001';/* тип - кредит*/
  creditlineamount = 0;/*Сумма кредитной линии*/
  provision = 0;/* Сумма созданного резерва */

  cr_type = 2;
  bankruptcystatus = 0;
  daysmainsumisoverdue = 0;
  daysinterestisoverdue = 0;
  numberofprolongs = 0;
  currencyofcredit = '';
  k_currency_old = '';

  k_id_old = 0;

  -- получаю сисок кредитов по которым когда-либо появляляась задолженность на внесимном учете + заем должен быть закрыт
  for select okf.oko_from_k_id,
             okf.oko_from_c_id,
             ks.ks_date_begin
  from operkroff okf
  inner join kontstate ks on ks.ks_from_k_id = okf.oko_from_k_id
  where okf.oko_ol_is_active = 1
  and okf.oko_type = 1
  and okf.oko_ol_date <= :report_date_end
  and :report_date_end between ks.ks_date_begin and ks.ks_date_end
  and ks.ks_state = 2
  and ks.ks_ol_is_delete = 0
  group by okf.oko_from_k_id,
           okf.oko_from_c_id,
           ks.ks_date_begin
  order by okf.oko_from_k_id
  into :k_id,
       :c_id,
       :loan_closed_date
  do
  begin
    if (k_id_old <> k_id) then
    begin
        select kont.k_currency,
               kont.k_amount_approved,
               kont.k_kredline_end,
               k_percent,
               k_disb_date,
               kont.k_base_k_id,
               kont.k_from_pr_id
        from kontract kont
        where kont.k_id = :k_id
        into :k_currency,
             :k_amount_approved,
             :duetimelastcontract,
             :annualinterestrate,
             :dateofgrant,
             :base_k_id,
             :pr_id;

        select max(sh.sh_date)
        from schedule sh
        where sh.sh_from_k_id = :k_id
        into :duetimefirstcontract;

        if ((dateofgrant is not null) and
            (duetimelastcontract is not null)) then
        begin
          creditperiodinmonths = (extract(year from duetimelastcontract) - extract(year from dateofgrant)) * 12 + extract(month from duetimelastcontract) - extract(month from dateofgrant) + 1;
        end
        if (creditperiodinmonths = 0) then
          creditperiodinmonths = 1;

        k_currency_old = 0;
        if (k_currency_old <> k_currency) then
        begin
          select cu_symbol
          from currlist cl
          where cl.cu_id = :k_currency
          into :currencyofcredit;
    
          k_currency_old = k_currency;
        end

        lastpaymentdate_operkred = null;

        lastpaymentdate_operkred =
        coalesce((select max(ok.ok_ol_date)
                 from operkred ok
                 where ok_type = 4 and
                       ok_ol_is_active = 1 and
                       ok_from_k_id = :k_id and
                       ok_ol_date <= :report_date_end and
                       ok_kommis = 0 and
                       ok_from_c_id = 0), '01.01.1900');

        is_finish = 0;
        monthlypaymentamount_group = 0;
    
        for select ok.ok_sh_date,
                   sum(ok.ok_os + ok.ok_sh_os + ok.ok_perc + ok.ok_sh_perc) as summ
            from operkred ok
            where ok.ok_from_k_id = :k_id and
                  ok.ok_type in (1, 2, 3) and
                  ok.ok_from_c_id = 0 and
                  ok.ok_ol_is_active = 1 and
                  ok.ok_ol_date <= :report_date_end
            group by 1
            having sum(ok.ok_os + ok.ok_sh_os + ok.ok_perc + ok.ok_sh_perc) <> 0
            into :sh_date_tmp, :summ_tmp
        do
        begin
          if (is_finish = 0) then
          begin
            if (sh_date_tmp >= :report_date_end) then
            begin
              monthlypaymentamount_group = summ_tmp;
              is_finish = 1;
            end
            else
            begin
              monthlypaymentamount_group = summ_tmp;
              is_finish = 0;
            end
          end
        end
        k_id_old = :k_id;
    end

    creditclasscode = '006';
    creditstatuscode = '003';
    disoutamountofcredit = 0;
    oiaforrepperiod = 0;

    -- > -----------------------------------------------------------------------
    lastpaymentdate = null;
    lastpaymentdate_operkroff = null;
    lastwriteoff_operkroff = null;

    lastpaymentdate_operkroff =
        coalesce((select max(okf.oko_ol_date)
                 from operkroff okf
                 where okf.oko_from_k_id = :k_id and
                      okf.oko_from_c_id = :c_id and
                      okf.oko_ol_is_active = 1 and
                      okf.oko_type = 4 and
                      okf.oko_ol_date <= :report_date_end), '01.01.1900');

    lastpaymentdate = lastpaymentdate_operkred;
    if (lastpaymentdate_operkroff > lastpaymentdate_operkred) then
      lastpaymentdate = lastpaymentdate_operkroff;

    lastwriteoff_operkroff =
        coalesce((select max(okf.oko_ol_date)
                 from operkroff okf
                 where okf.oko_from_k_id = :k_id and
                      okf.oko_from_c_id = :c_id and
                      okf.oko_ol_is_active = 1 and
                      okf.oko_type = 5 and
                      okf.oko_ol_date <= :report_date_end), '01.01.1900');

    select sum(okf.oko_os + okf.oko_sh_os),
           sum(okf.oko_perc + okf.oko_sh_perc)
    from operkroff okf
    inner join operlist ol on okf.oko_from_ol_id = ol.ol_id
    where okf.oko_from_k_id = :k_id and
        okf.oko_from_c_id = :c_id and
        ol.is_delete = 0 and
        ol.ol_date <= :report_date_end
    into :disoutamountofcredit, -- задолжность по ОС
         :oiaforrepperiod; -- задолжность по %


    -- Если это разделенный кредит, необходимо провериь какая сумма долга по ОС отправлялась по клиенту до разделения,
    -- т.к. если отпаравить в МКР сумму большую чем отправлялась ранее, МКР запись не примет.
    -- Поэтому если разница между суммой которая отправлялась до разделения и текущей суммой долга по ОС в пределах 0.1 AZN тогда отправляем старую сумму.
    if (base_k_id is not null) then
    begin
        old_disoutamountofcredit = 0;
    
        select round(mcc.mc_disoutamountofcredit, 2)
        from mkr_credit mcc
        where mcc.mc_from_k_id = :base_k_id
        and mcc.mc_from_c_id = :c_id
        order by mcc.mc_from_mp_id desc
        rows 1
        into :old_disoutamountofcredit;
    
        if (abs(disoutamountofcredit - old_disoutamountofcredit) <= 0.1) then
            disoutamountofcredit = old_disoutamountofcredit;
    end

    disoutamountofcredit = coalesce(disoutamountofcredit, 0);
    oiaforrepperiod = coalesce(oiaforrepperiod, 0);
    -- < -----------------------------------------------------------------------
    ---------------------------- > если за период, указанный в пакете,
    if ((not(lastpaymentdate between report_date_begin and report_date_end)) and  -- по нему не было платежа
        (not(loan_closed_date between report_date_begin and report_date_end)) and -- кредит не был закрыт (т.е не перенесн на внесистмку)
        (not(lastwriteoff_operkroff between report_date_begin and report_date_end)) and -- не было списания с внесистемки
        (disoutamountofcredit <= 0) and (oiaforrepperiod <=0)) then -- задолженность по внесистем = 0
    begin  -- тогда в пакет не попадает
        continue;
    end
    ---------------------------- <
    initialamountofcredit = null;
    knc_business_subtype_id = null;
    accountno = null;

    select kncl.kc_amount_approved,
           kncl.kc_busuness_subtype,
           cast(iif(:base_k_id is null, :k_id, :base_k_id) as varchar(255)) || '-' || cast(kncl.kc_number as varchar(255)),
           kc_number
    from knclient kncl
    where kncl.kc_from_k_id = :k_id
    and kncl.kc_from_c_id = :c_id
    and kncl.kc_amount_approved > 0
    into :initialamountofcredit,
         :knc_business_subtype_id,
         :accountno,
         :kc_number;

    if (initialamountofcredit is null) then
      exception e_default 'ERROR: "initialamountofcredit is null" k_id = '||coalesce(:k_id, '')||', c_id = '||coalesce(:c_id, 'null');

    select purposeofcredit
    from mkr_purposeofcredit_get(:pr_id, :knc_business_subtype_id, :dateofgrant)
    into :purposeofcredit;

    if (disoutamountofcredit < 0) then
        disoutamountofcredit = 0;
    if (oiaforrepperiod < 0) then
        oiaforrepperiod = 0;

    oiaforlastthreemonths = :oiaforrepperiod;

    daysmainsumisoverdue = 0;
    daysinterestisoverdue = 0;

    if ((disoutamountofcredit <> 0) or (oiaforrepperiod <> 0)) then
    begin
        /*
        select :report_date_end - min(ok_sh_date)
        from kontract k, operkred ok
        where k_id = :k_id and
              ok.ok_from_k_id = k.k_id and
              k.k_fact_closed_ol_id = ok_from_ol_id and
              ok_type = 5 and
              ok_ol_is_active = 1
        into daysmainsumisoverdue;
        */
        -- igusev 01.12.2020 число дней активной просрочки расчитываю так: получаю дату начала просчроки на состояние - дата закрытия займа минус 1 день и вычисляю сколько дней просрочки накопилось с того дня, до отчетной даты.
        daysmainsumisoverdue_fix = null;

        select :report_date_end - (mdd.mdd_last_renewal - mdd.mdd_delinq_days)
        from mkr_delinq_days mdd
        where mdd.mdd_account_no = :accountno
        into :daysmainsumisoverdue_fix;

        if (daysmainsumisoverdue_fix is not null) then
            daysmainsumisoverdue = daysmainsumisoverdue_fix;
        else
        begin
            select :report_date_end - oks.oks_delinq_date
            from oksummary oks
            where oks.oks_from_k_id = :k_id
            and oks.oks_close_state = 0
            and :loan_closed_date - 1 between oks.oks_date_begin and oks.oks_date_end
            into daysmainsumisoverdue;
        end
        daysinterestisoverdue = daysmainsumisoverdue;
    end
    else
        creditstatuscode = '001';

    monthlypaymentamount = monthlypaymentamount_group / k_amount_approved * initialamountofcredit;

    select replace(coalesce(cl.c_pass_no, ''), '№', ''),
           iif(pk_country = 'AZN', 'AZE', upper(pk_country)),
           cl.c_birth_date,
           trim(coalesce(c_pincode, '')),
           c_name_fio
      from clients cl
      inner join passkind pask on pask.pk_id = cl.c_doc_type
      where cl.c_id = :c_id
      into :id, :countrycode, :dateofbirth, :pincode, :name;

      knc_id = :c_id;

      /*
      if (not exists(select *
                     from kib_ignore ki
                     where ki.ki_from_k_id = :k_id)) then
      */
      suspend;
    end
end
