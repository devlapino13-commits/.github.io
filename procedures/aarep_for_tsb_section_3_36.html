CREATE PROCEDURE AAREP_FOR_TSB_SECTION_3_36(
    REP_DATE_END DATE)
RETURNS (
    C_NAME_FIO VARCHAR(100),
    K_KONTRACT_NO VARCHAR(20),
    K_DISB_DATE DATE,
    GR_TYPE VARCHAR(10),
    IS_RESTRUCT VARCHAR(10),
    PERIOD_DAY INTEGER,
    KC_AMOUNT INTEGER,
    RPPU_SUM DECIMAL(18,4),
    LK_BALANCE_CLI CURR_TYPE,
    LK_CALCPERC_CLI CURR_TYPE,
    GROUP_NUMBER SMALLINT,
    PERCENT_RVPZ CURR_TYPE,
    RPPU_BALANCE CURR_TYPE,
    RPPU_CALCPERC CURR_TYPE,
    LK_DELIQ_DAYS INTEGER,
    STATUS_CLI VARCHAR(10),
    IS_RESIDENT VARCHAR(20))
AS
declare variable k_id id_type;
declare variable citizen_ru_id integer;
declare variable doc_vnj_id integer;
declare variable c_id id_type;
declare variable report_old smallint;
declare variable lk_is_indiv smallint;
declare variable lk_is_restruct smallint;
declare variable lk_is_secured smallint;
declare variable lk_balance curr_type;
declare variable lk_calcperc curr_type;
declare variable k_amount_approved curr_type;
declare variable kol_cli integer;
declare variable pr_id integer;
declare variable sh_close_date date;
declare variable pr_code varchar(100);
declare variable pr_product_kind integer;
declare variable dolya double precision;
declare variable res_client_is_resident smallint;
begin
  citizen_ru_id = null;
  select rd_id
  from refdata
  where rd_code = 'Citizen.RU'
  into :citizen_ru_id;

  doc_vnj_id = null;
  select pk_id
  from passkind, passdoctype
  where pdt_id = pk_from_pdt_id and
        pk_is_delete = 0 and
        pdt_code = 'registration_vnj' rows 1
  into :doc_vnj_id;

  if (rep_date_end < '31.08.2017') then
    report_old = 1;
  else
    report_old = 0;

  for select c_name_fio, sum(lk_balance_cli - rppu_balance + lk_calcperc_cli - rppu_calcperc)
      from aarep_llc_report_201708(1000000, :rep_date_end, 0)
      group by 1
      order by 2 desc
      rows 5
      into c_name_fio, rppu_sum
  do
  begin
    c_id = replace(c_name_fio, '№', '');

    for select k_id, k_disb_date, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days, k_kontract_no,
               k_amount_approved, k_kolich_client_approved, k_from_pr_id, kc_amount_approved, lk_is_indiv
        from llc_main, llc_kont, kontract, knclient, kontstate
        where lm_is_signed = 1 and
              lm_id = lk_from_lm_id and
              :rep_date_end between lk_date and lk_date_end and
              lk_from_k_id = k_id and
              k_id = kc_from_k_id and
              kc_amount_approved > 0 and
              kc_from_c_id = :c_id and
              k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_state = 1 and
              :rep_date_end between ks_date_begin and ks_date_end
        into k_id, k_disb_date, lk_is_secured, lk_is_restruct, lk_balance, lk_calcperc, lk_deliq_days, k_kontract_no,
             k_amount_approved, kol_cli, pr_id, kc_amount, lk_is_indiv
    do
    begin
      if (lk_is_indiv = 1) then
        gr_type = 'Индив.';
      gr_type = 'Групп.';
      if (lk_is_restruct = 1) then
        is_restruct = 'Да';
      else
        is_restruct = 'Нет';

      select max(sh_date)
      from schedule
      where sh_from_k_id = :k_id
      into sh_close_date;

      period_day = sh_close_date - k_disb_date;

      if (:report_old = 1) then
      begin
        select pr_code, pr_loan_type, pr_product_kind
        from product
        where pr_id = :pr_id
        into pr_code, lk_is_indiv, pr_product_kind;

        select is_secured
        from aarep_get_secured_loan(:pr_code, :lk_is_indiv, :pr_product_kind, :pr_id, :k_disb_date, :kol_cli, :k_id)
        into lk_is_secured;
      end

      if (lk_is_secured is null) then
        lk_is_secured = 0;
      if (lk_is_restruct is null) then
        lk_is_restruct = 0;

      dolya = cast(kc_amount as double precision) / k_amount_approved;

      select 1 - res_non_resident
      from hp_get_client_type(:c_id, :citizen_ru_id, :doc_vnj_id, :rep_date_end)
      into :res_client_is_resident;

      status_cli = 'ФЛ';

      if (res_client_is_resident = 1) then
        is_resident = 'Резидент';
      else
        is_resident = 'Нерезидент';

      select group_number, percent_rvpz
      from hp_get_rvpz(:kc_amount, :period_day, :lk_is_secured, :lk_is_restruct, :lk_deliq_days)
      into group_number, percent_rvpz;

      lk_balance_cli = dolya * lk_balance;
      lk_calcperc_cli = dolya * lk_calcperc;

      rppu_balance = cast(lk_balance_cli as double precision) * percent_rvpz / 100;
      rppu_calcperc = cast(lk_calcperc_cli as double precision) * percent_rvpz / 100;

      rppu_sum = lk_balance_cli - rppu_balance + lk_calcperc_cli - rppu_calcperc;

      suspend;

    end
  end
end
