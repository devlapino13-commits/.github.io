CREATE PROCEDURE HP_CASHTERMINAL_LIST(
    IN_TERMINAL_ID D_INTEGER)
RETURNS (
    CIL_ID D_INTEGER,
    CIL_FROM_CSP_ID D_INTEGER,
    CIL_DATE D_DATE,
    CIL_CASHIER_ID D_ID,
    CIL_INSPECTOR_ID D_ID,
    CIL_INV_AMOUNT D_CURRENCY,
    CIL_DATETIME D_DATETIME,
    CIL_FROM_UN_ID D_ID,
    CTM_FROM_TM_ID D_INTEGER,
    CTM_DOC_NO D_INTEGER,
    CTM_PERIOD_BEGIN D_DATETIME,
    CTM_PERIOD_END D_DATETIME,
    CTM_AMOUNT_CLS D_CURRENCY,
    CTM_AMOUNT_IPF D_CURRENCY,
    CTM_AMOUNT_OTHER D_CURRENCY,
    RES_PLACE_NAME D_STR_CODE,
    RES_CASHIER_FIO D_STR_100,
    RES_INSPECTOR_FIO D_STR_100,
    RES_USERNAME D_STR_CODE)
AS
begin
  for select cil_id, 
             cil_from_csp_id, 
             cil_date, 
             cil_cashier_id, 
             cil_inspector_id, 
             cil_inv_amount,
             cil_datetime, 
             cil_from_un_id, 
             ctm_from_tm_id,
             ctm_doc_no,
             ctm_period_begin,
             ctm_period_end,
             ctm_amount_cls,
             ctm_amount_ipf,
             ctm_amount_other,
             'Касса ' || csp_cash_no,
             c.c_name_fio,
             ins.c_name_fio,
             un_name
      from cashterminal
           inner join cashinvlist on (ctm_from_cil_id = cil_id)
           inner join cashplace p on (p.csp_id = cil_from_csp_id)
           left join clients c on (cil_cashier_id = c.c_id)
           left join clients ins on (cil_inspector_id = ins.c_id)
           left join username on (un_id = cil_from_un_id)
      where ctm_from_tm_id = :in_terminal_id
      order by ctm_period_end desc, cil_id desc
      rows 10
      into :cil_id,
           :cil_from_csp_id,
           :cil_date,
           :cil_cashier_id,
           :cil_inspector_id,
           :cil_inv_amount,
           :cil_datetime,
           :cil_from_un_id,
           :ctm_from_tm_id,
           :ctm_doc_no,
           :ctm_period_begin,
           :ctm_period_end,
           :ctm_amount_cls,
           :ctm_amount_ipf,
           :ctm_amount_other,
           :res_place_name,
           :res_cashier_fio,
           :res_inspector_fio,
           :res_username

  do
  begin
    suspend;
  end
end
