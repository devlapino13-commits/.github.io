CREATE PROCEDURE HP_CASHCHECK_SENDSMS(
    CCM_ID D_INTEGER,
    UN_ID D_ID)
RETURNS (
    EDL_ID D_ID,
    ERR_TXT D_STR_255)
AS
declare variable ccm_edl_id d_id;
declare variable ccm_is_delete d_smallint;
begin
  err_txt = '';
  for
    select
      ccm_edl_id,
      ccm_is_delete
    from cashcheckmain
    where ccm_id = :ccm_id
  into
    :ccm_edl_id,
    :ccm_is_delete
  do
  begin
    if (ccm_is_delete = 1) then
      err_txt = f_format('Статус "акта инвентаризации кассы" №%1 не соответствует возможности отправки SMS на подписание', ccm_id);
    else
    if (ccm_edl_id > 0) then
      err_txt = f_format('SMS на подписание "акта инвентаризации кассы" №%1 уже отправлялось', ccm_id);
    else
      select
        edl_id,
        res_text
      from hp_edoc_ins(
        'cash.inventory_act',
        f_objkind('cashcheckmain', 'id'),
        :ccm_id,
        :un_id)
      into
        :ccm_edl_id,
        :err_txt;
    if (err_txt = '') then
      update cashcheckmain set
        ccm_edl_id = :ccm_edl_id
      where ccm_id = :ccm_id;
  end
  suspend;
end
