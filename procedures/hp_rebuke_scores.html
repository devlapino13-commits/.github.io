CREATE PROCEDURE HP_REBUKE_SCORES(
    EMP_ID D_ID)
RETURNS (
    RS_DATE D_DATE,
    RS_SCORES D_SMALLINT,
    IS_CORRECTION D_BOOL,
    RS_TEXT D_STR_LONG)
AS
declare variable c_id d_id;
declare variable c_name_fio d_str_100;
declare variable c_inn d_str_100;
declare variable jl_name d_str_100;
declare variable reg_name d_str_100;
declare variable scores_sum d_integer;
begin
  -- возвращает начисленные и штрафные баллы сотрудника для просмотра (RFCKG-2189)
  -- если rs_protocol_no = -1, то запись была добавлена при корректировке баллов

  select c_id,
         c_name_fio,
         c_inn,
         jl_name,
         reg_code || ' ' || reg_name
  from clients
  inner join hr_pstatus on ps_from_c_id = c_id
  inner join joblist on jl_id = ps_from_jl_id
  inner join region on reg_unicode = ps_from_reg_id
  where c_id = :emp_id and
        current_date between ps_date_begin and ps_date_end
  into c_id,
       c_name_fio,
       c_inn,
       jl_name,
       reg_name;

  if (c_id is null or jl_name is null or reg_name is null) then
  begin
    suspend;
    exit;
  end

  select sum(rs_scores)
  from rebuke_scores
  where rs_is_delete = 0 and
        rs_from_c_id = :c_id
  group by rs_from_c_id
  into scores_sum;

  rs_text = '<body><h3> ' || c_name_fio || '</h3>';
  rs_text = rs_text || '<p>ИНН (ПИН):<b> ' || c_inn || '</b></p>';
  rs_text = rs_text || '<p>Офис:<b> ' || reg_name || '</b></p>';
  rs_text = rs_text || '<p>Должность:<b> ' || jl_name || '</b></p>';
  rs_text = rs_text || '<p>Разница начисленных и штрафных баллов (итого баллов):<b> ' || scores_sum || '</b></p></body>';

  for select rs_date,
             rs_scores,
             iif(rs_protocol_no = -1, 1, 0)
      from rebuke_scores
      where rs_is_delete = 0 and
            rs_from_c_id = :c_id
      order by rs_datetime
      into rs_date,
           rs_scores,
           is_correction
  do
    suspend;
end
