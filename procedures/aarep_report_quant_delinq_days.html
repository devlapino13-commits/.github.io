CREATE PROCEDURE AAREP_REPORT_QUANT_DELINQ_DAYS(
    REP_DATE_END D_DATE)
RETURNS (
    K_DISB_DATE D_DATE,
    K_KONTRACT_NO VARCHAR(20),
    PR_NAME VARCHAR(50),
    K_KREDLINE_END D_DATE,
    K_AMOUNT_APPROVED CURR_TYPE,
    KC_ID ID_TYPE,
    TERM_IN_DAYS D_INTEGER,
    PERCENT_EFFECT PERC_TYPE,
    OKS_KONT_LAST_DATE D_DATE,
    OKS_LAST_SH_DATE D_DATE,
    CLI_BALANCE CURR_TYPE,
    CLI_PERC_SALDO CURR_TYPE,
    OKS_BAL_KOMMIS CURR_TYPE,
    OKS_DIFF_ESP_BAL CURR_TYPE,
    KD_PERC D_STR_10,
    MAX_DELIQ D_INTEGER,
    QUANT_DELIQ D_INTEGER,
    OKS_DELINQ_DAYS D_INTEGER,
    C_NAME_FIO D_STR_100,
    K_PERCENT_FLAT_ANNUAL PERC_TYPE,
    QUANT_KD_PERCENT_IZM D_INTEGER)
AS
declare variable k_id id_type;
declare variable max_sh_date d_date;
declare variable kd_id_new id_type;
declare variable kd_ol_date d_date;
declare variable k_percent_kont perc_type;
declare variable kd_percent_new perc_type;
declare variable kd_percent_old perc_type;
declare variable oks_date_begin d_date;
declare variable oks_delinq_date d_date;
declare variable oks_delinq_date_pr d_date;
declare variable oks_delinq_date_pr_old d_date;
declare variable oks_delinq_date_start d_date;
declare variable quant_delinq_days d_integer;
declare variable delinq_date d_integer;
declare variable c_id id_type;
declare variable oks_percent_norm_prv_old perc_type;
declare variable oks_percent_norm_prv perc_type;
begin
  c_name_fio = '';
  for select oks_from_k_id, k_kontract_no, k_disb_date, pr.pr_name, k_kredline_end, k_amount_approved oks_delinq_date,
             k_percent_per_day, round(k_percent_per_day * 365.00, 3)
      from oksummary oks
      left join kontract kont on oks.oks_from_k_id = kont.k_id
      left join product pr on kont.k_from_pr_id = pr.pr_id
      where :rep_date_end between oks_date_begin and oks_date_end and
            oks_close_state = 0
           -- and k_kontract_no =  '0103-332888-06'
      into k_id, k_kontract_no, k_disb_date, pr_name, k_kredline_end, k_amount_approved, k_percent_kont,
           k_percent_flat_annual
  do
  begin

    kc_id = null;
    for select kc_id, kc_from_c_id
        from knclient
        where kc_amount_approved > 0 and
              kc_from_k_id = :k_id
        into :kc_id, :c_id
    do
    begin
      c_name_fio = 'n' || :c_id;

      max_sh_date = null;
      term_in_days = 0;
      select max(sh_date)
      from schedule
      where sh_from_k_id = :k_id
      into :max_sh_date;

      if (max_sh_date is not null) then
        term_in_days = max_sh_date - k_disb_date;
      else
        term_in_days = null;

      percent_effect = 0;

      oks_kont_last_date = null;
      oks_last_sh_date = null;
      select first 1 oks_percent_esp, oks_kont_last_date, oks_last_sh_date, oks_balance, oks_bal_perc, oks_bal_kommis,
                     oks_diff_esp_bal, oks_delinq_days
      from kn_oksummary
      where oks_from_kc_id = :kc_id and
            :rep_date_end between oks_date_begin and oks_date_end
      into :percent_effect, :oks_kont_last_date, :oks_last_sh_date, :cli_balance, :cli_perc_saldo, :oks_bal_kommis,
           :oks_diff_esp_bal, :oks_delinq_days;

      kd_ol_date = null;
      kd_percent_new = 0;
      kd_id_new = null;
      select first 1 kd_ol_date, kd_percent_per_day, kd_id
      from kontdata
      where :rep_date_end between kd_ol_date and kd_date_end and
            kd_ol_is_delete = 0 and
            kd_from_k_id = :k_id
      into :kd_ol_date, :kd_percent_new, :kd_id_new;

      -- дата просрочки на дату изменения процентов
      delinq_date = 0;
      oks_delinq_date = null;
      select first 1 oks_delinq_date
      from oksummary
      where oks_from_k_id = :k_id and
            :kd_ol_date between oks_date_begin and oks_date_end and
            oks_close_state = 0
      into :oks_delinq_date;

      --количество просроченных дней на дату изменения процентов
      delinq_date = iif((cast(:kd_ol_date as date) - :oks_delinq_date < 0), 0, cast(:kd_ol_date as date) - :oks_delinq_date);

      -- проверка процент увеличился или снизился
      kd_percent_old = null;
      select first 1 kd_percent_per_day
      from kontdata
      where kd_from_k_id = :k_id and
            :kd_ol_date - 1 between kd_ol_date and kd_date_end and
            kd_ol_is_delete = 0
      into :kd_percent_old;

      if ((kd_percent_new > 0) and
          (kd_percent_new <> kd_percent_old)) then
      begin
        if (kd_percent_old = 0) then
          kd_percent_old = k_percent_kont;
        if (kd_percent_new > kd_percent_old) then
          kd_perc = 'Повышение';
        else
          kd_perc = 'Понижение';
      end

      --Отбор дат платежей и дат просрочек
      oks_date_begin = null;
      oks_delinq_date_pr = null;
      quant_delinq_days = 0;
      max_deliq = 0;
      quant_deliq = 0;
      oks_delinq_date_pr_old = null;
      oks_percent_norm_prv = 0;
      oks_percent_norm_prv_old = 0;
      quant_kd_percent_izm = 0;
      for select oks_date_begin, oks_percent_norm_prv, max(oks_delinq_date) oks_delinq_date_pr
          from kn_oksummary
          where oks_from_k_id = :k_id and
                oks_from_kc_id = :kc_id and
                oks_date_begin <= :rep_date_end
          group by oks_date_begin, oks_percent_norm_prv
          order by oks_date_begin
          into oks_date_begin,oks_percent_norm_prv, oks_delinq_date_pr
      do
      begin
        if ((oks_percent_norm_prv <> oks_percent_norm_prv_old) and
            (oks_percent_norm_prv <> 0) and
            (oks_percent_norm_prv_old <> 0)) then
          quant_kd_percent_izm = quant_kd_percent_izm + 1;
        if ((oks_delinq_date_pr is null) and
            (oks_delinq_date_pr_old is not null)) then
        begin
          quant_delinq_days = oks_date_begin - oks_delinq_date_start;
          quant_deliq = quant_deliq + 1; --количество закрытых просрочек(без текущей)
        end
        if (max_deliq < quant_delinq_days) then
          max_deliq = quant_delinq_days; --максимальная просрочка
        if (oks_delinq_date_pr is not null) then
          oks_delinq_date_start = oks_delinq_date_pr;
        oks_delinq_date_pr_old = oks_delinq_date_pr;
        oks_percent_norm_prv_old = oks_percent_norm_prv;
      end

      suspend;
    end
  end
end
