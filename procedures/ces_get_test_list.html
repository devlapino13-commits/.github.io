CREATE PROCEDURE CES_GET_TEST_LIST
RETURNS (
    EDS_ID D_ID,
    EDB_ID D_ID)
AS
begin
  for select first 10 eds.eds_id, edb.edb_id
      from edoc_list edl
      inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id and
            eds.eds_status = 0
      inner join edoc_blank edb on edb.edb_from_edl_id = edl.edl_id
      where edl.edl_blank_date >= current_date - 7 and
            edl.edl_status = 0 and
            edb.edb_from_bln_id = (select first 1 edbb.edb_from_bln_id
                                   from edoc_list edll
                                   inner join edoc_cfg edc on edc.edc_id = edll.edl_from_edc_id and
                                                              edc.edc_state = 3 and
                                                              edc.edc_process_kind > 0
                                   inner join edoc_sign edss on edss.eds_from_edl_id = edll.edl_id and
                                         edss.eds_status = 0
                                   inner join edoc_blank edbb on edbb.edb_from_edl_id = edll.edl_id
                                   where edll.edl_blank_date >= current_date - 7 and
                                         edll.edl_status = 0
                                   group by edbb.edb_from_bln_id
                                   order by count(*) desc)
      into :eds_id, :edb_id
  do
  begin
    suspend;
  end
end
