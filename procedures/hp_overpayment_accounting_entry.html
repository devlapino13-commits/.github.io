CREATE PROCEDURE HP_OVERPAYMENT_ACCOUNTING_ENTRY
RETURNS (
    NUM_DONE D_INTEGER)
AS
declare variable id d_integer;
declare variable ol_id d_id;
declare variable ol_id1 d_id;
declare variable ol_id2 d_id;
declare variable ol_datetime d_datetime;
declare variable ol_datetime1 d_datetime;
declare variable k_id d_id;
declare variable k_kontract_no d_str_20;
declare variable k_kontract_date d_date;
declare variable is_ipf d_bool;
declare variable is_card_mbulak d_bool;
declare variable c_id d_id;
declare variable un_id d_id;
declare variable amount d_currency;
declare variable reg_id d_id;
declare variable head_reg_id d_id;
declare variable oper_date d_date;
declare variable doc_no d_integer;
declare variable acc_dt d_str_20;
declare variable acc_kt d_str_20;
declare variable res_reg_id d_integer;
declare variable res_acc_dt d_str_20;
declare variable res_acc_kt d_str_20;
declare variable total_line d_integer;
declare variable res_line_no d_integer;
declare variable kf_id d_id;
begin
  oper_date = current_date;
  head_reg_id = 1090100;
  num_done = 0;

  for
    select
      ovp_id,
      ovp_from_ol_id,
      ovp_from_k_id,
      k_kontract_no,
      k_kontract_date,
      ovp_prod_is_ipf,
      ovp_is_card_mbulak,
      ovp_from_c_id,
      k_from_kf_id,
      ovp_amount,
      ovp_from_un_id
    from overpayment
      inner join kontract
        on k_id = ovp_from_k_id
    where ovp_status = 3
  into
    :id,
    :ol_id2,
    :k_id,
    :k_kontract_no,
    :k_kontract_date,
    :is_ipf,
    :is_card_mbulak,
    :c_id,
    :kf_id,
    :amount,
    :un_id
  do
  begin
    reg_id = (
      select kal_new_reg_id
      from knaktlist
      where kal_from_kf_id = :kf_id
          and :oper_date between kal_date and kal_date_end
          and is_delete = 0);

    acc_dt = (
      select al_id from acc_list
      where al_from_reg_id = :reg_id
        and al_currency = 417
        and al_account = iif(:is_card_mbulak = 1, '21199/1101', iif(:is_ipf = 1, '21199/101', '21199/1')));
    acc_kt = (
      select al_id from acc_list
      where al_from_reg_id = :head_reg_id
        and al_currency = 417
        and al_account = iif(:is_ipf = 1, '10003/1004', '10003/1003'));
    doc_no = (select num from last_doc_number(:oper_date, 0, :reg_id));

    ol_id = null;
    ol_datetime = null;
    for
      select
        res_reg_id,
        res_acc_dt,
        res_acc_kt,
        total_line,
        res_line_no
      from aarep_loronostro_link(:reg_id, :acc_dt, :acc_kt)
    into
      :res_reg_id,
      :res_acc_dt,
      :res_acc_kt,
      :total_line,
      :res_line_no
    do
    begin
      insert into operlist (
        ol_parent_id,
        ol_from_reg_id,
        ol_oper_type,
        ol_oper_subtype,
        ol_date,
        ol_doc_no,
        ol_doc_kind,
        ol_document_client,
        ol_memo,
        ol_user_id
      )
      values (
        :ol_id,
        :res_reg_id,
        iif(:res_line_no = 1, 1, 2),
        iif(:res_line_no = 1, 10, 20),
        :oper_date,
        :doc_no,
        9,
        :c_id,
        f_format('Возврат переплаты по кредитному договору №%1 от %2г.', :k_kontract_no, f_datetostr(:k_kontract_date)),
        :un_id
      )
      returning ol_id, ol_datetime
      into :ol_id1, :ol_datetime1;

      if (res_line_no = 1) then
      begin
        ol_id = ol_id1;
        ol_datetime = ol_datetime1;
        insert into operkred (
          ok_from_ol_id,
          ok_number,
          ok_from_k_id,
          ok_sh_id,
          ok_sh_date,
          ok_type,
          ok_os,
          ok_pay_type,
          ok_kommis,
          ok_sh_os,
          ok_sh_perc,
          ok_kommis_adv,
          ok_tax,
          ok_ol_date,
          ok_ol_reg_id,
          ok_ol_is_active,
          ok_from_c_id
        )
        select
          :ol_id,
          -1,
          :k_id,
          ok_sh_id,
          ok_sh_date,
          9,
          :amount,
          ok_pay_type,
          0,
          0,
          0,
          0,
          0,
          null,
          null,
          null,
          0
        from operkred
        where ok_from_k_id = :k_id
          and ok_from_ol_id = :ol_id2
          and ok_type = 9
          and ok_os < 0;

        update operloan set
          opl_acc_kind = 3,
          opl_from_al_id = :res_acc_kt
        where opl_from_ol_id = :ol_id;
      end

      insert into operbook (
        ob_from_ol_id,
        ob_currency,
        ob_curr_rate,
        ob_account_dt,
        ob_account_kt,
        ob_detail,
        ob_summa,
        ob_summa_val,
        ob_client_dt
      )
      values (
        :ol_id1,
        '417',
        1,
        :res_acc_dt,
        :res_acc_kt,
        'Возврат переплаты',
        :amount,
        :amount,
        :c_id
      );
    end
    if (id > 0) then
    begin
      update overpayment set
        ovp_status = 6,
        ovp_from_res_ol_id = :ol_id,
        ovp_datetime = :ol_datetime
      where ovp_id = :id;
      num_done = num_done + 1;
    end
  end

  suspend;
end
