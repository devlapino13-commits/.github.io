CREATE PROCEDURE SET_ASAN_BALANCE(
    AB_STATUS D_SMALLINT,
    AB_RESPONSE D_BLOB_TEXT,
    AB_BALANCE D_CURRENCY)
AS
begin
  -- Записываем ответ по запросу баланса
  update asan_balance
  set ab_status = :ab_status, 
      ab_balance = :ab_balance,
      ab_response = :ab_response,
      ab_status_datetime = current_timestamp
  where ab_status = 0;
end
