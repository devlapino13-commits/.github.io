CREATE PROCEDURE TERM_CONT_PROTOCOL(
    REP_DATE DATE)
RETURNS (
    KF_ID ID_TYPE,
    NUM INTEGER,
    K_KONTRACT_NO VARCHAR(50),
    K_FACT_DISB_DATE DATE,
    KF_NAME VARCHAR(50),
    K_DELINQ_DATE DATE,
    K_DELINQ_DAY INTEGER)
AS
begin
  for select k_from_kf_id, k_kontract_no, k.k_fact_disb_date, oks_delinq_date,
             iif(oks_date_end >= :rep_date, :rep_date, oks_date_end + 1) - oks_delinq_date
      from oksummary oks, kontract k, kontstopperc ksp, operlist
      where ksp.ksp_from_k_id = k.k_id and
            ksp.ksp_date = :rep_date and
            oks_close_state = 0 and
            :rep_date between oks_date_begin and oks_date_end and
            iif(oks_date_end >= :rep_date, :rep_date, oks_date_end + 1) - oks_delinq_date > 60 and
            oks_from_k_id = k_id and
            k.k_stopped_perc_ol_id = ol_id and
            ol_user_id = 1000000000001
      order by 5
      into kf_id, k_kontract_no, k_fact_disb_date, k_delinq_date, k_delinq_day
  do
  begin
    num = coalesce(num + 1, 1);

    select kal_new_kf_name
    from knaktlist kal
    where kal.is_delete = 0 and
          :rep_date between kal_date and kal_date_end and
          kal_from_kf_id = :kf_id
    into kf_name;

    suspend;
  end
  if (kf_id is null) then
  begin
    k_kontract_no = 'Нет данных';
    suspend;
  end
end
