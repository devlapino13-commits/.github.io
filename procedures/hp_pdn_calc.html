CREATE PROCEDURE HP_PDN_CALC(
    CREDDATE D_DATE,
    CREDENDDATE D_DATE,
    CALC_DATE D_DATE,
    CALC_IN_AVG D_SMALLINT,
    CREDFULLCOST D_CURRENCY,
    CREDSUMDEBT D_CURRENCY,
    CREDSUMOVERDUE D_CURRENCY,
    CREDSUM D_CURRENCY,
    CALC_IS_CREDIT_CARD D_SMALLINT)
RETURNS (
    CALC_KOEF_PERIOD D_CURRENCY,
    CALC_PERIOD_MONTH D_CURRENCY,
    CALC_MONTH_REPAY D_DOUBLE)
AS
declare variable tmp_amount_2 d_double;
declare variable tmp_amount d_double;
begin
  calc_koef_period = 1;
  -- Поправ коэф ЧБП/30 (для займов до 30 дней)
  if (datediff(day, credDate, credEnddate) <= 30 and datediff(day, credDate, credEnddate) > 0) then
  begin
    -- с 01.10.2019 действует новый вариант расчета ЧБП/30, письмо от Елены в январе 2020 г.
    if (calc_date < '01.10.2019')
    then calc_koef_period = (cast(365.0 as curr_type) / cast(datediff(day, credDate, credEnddate) as curr_type) / cast(30.0 as curr_type));
    else calc_koef_period = cast(datediff(day, credDate, credEnddate) as curr_type) / cast(30.0 as curr_type);
  end

  calc_period_month = 0;
  if (credEnddate > calc_date) then
  begin
    calc_period_month = cast(datediff(day, calc_date, credEnddate) as curr_type) / (cast(365.0 as curr_type) / cast(12.0 as curr_type));
    if (trunc(calc_period_month) < calc_period_month)
    then calc_period_month = trunc(calc_period_month) + 1;
    else calc_period_month = trunc(calc_period_month);
  end

  calc_month_repay = 0;
  if (calc_in_avg = 1) then
  begin
    if (calc_is_credit_card = 0) then
    begin
      -- Формула 1 для кредитов/займов, кроме кредитных карт и овердрафтов
      calc_month_repay = credFullCost / 100.000;
      calc_month_repay = calc_month_repay * calc_koef_period;
      calc_month_repay = calc_month_repay / 12.00;
      tmp_amount_2 = calc_month_repay;
      calc_month_repay = round(calc_month_repay * credSumDebt, 2);

      tmp_amount = 1 - power(1.0000 + tmp_amount_2, -calc_period_month);

      if (tmp_amount <> 0)
      then calc_month_repay = calc_month_repay / tmp_amount;
      else calc_month_repay = 0;

      calc_month_repay = calc_month_repay + credSumOverdue;
    end else
    begin
      -- Формула 2 для кредитных карт и овердрафтов
      if (calc_date >= '01.01.2024') then
        calc_month_repay = maxvalue((credsum + credSumOverdue) * 0.05, credSumOverdue);
      else
        calc_month_repay = minvalue(maxvalue((credsum + credSumOverdue) * 0.05, credSumOverdue),
                                   credSumDebt * 0.10 + credSumOverdue);
    end

    calc_month_repay = round(calc_month_repay, 2);
  end

  suspend;
end
