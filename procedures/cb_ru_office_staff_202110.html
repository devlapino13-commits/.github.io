declare variable REP_DATE_BEGIN date;
declare variable AAREP_PLAN_ALL decimal(18,4);
declare variable AAREP_FACT_ALL decimal(18,4);
declare variable AAREP_FACT_PLANNED decimal(18,4);
declare variable AAREP_FACT_OTHER decimal(18,4);
declare variable AAREP_MONTH_PLAN_OS decimal(18,4);
declare variable AAREP_MONTH_PLAN_PERC decimal(18,4);
declare variable JK_SYSTEM_CODE varchar(100);
declare variable KOL integer;
declare variable KOL_MO integer;
declare variable KOL_ZAM_MO integer;
declare variable KOL_KOBD integer;
declare variable KOL_ADMIN integer;
declare variable PERC_BONUS_MO CURR_TYPE;
declare variable PERC_BONUS_ZAM_MO CURR_TYPE;
declare variable PERC_BONUS_KOBD CURR_TYPE;
declare variable PERC_BONUS_ADMIN CURR_TYPE;
declare variable AAREP_BONUS_TOTAL_FACT_TMP CURR_TYPE;
declare variable RD_ID integer;
declare variable REP_PERIOD integer;
declare variable MO_ID ID_TYPE;
declare variable MANAGER_NAME D_STR_100;
declare variable LO_ID ID_TYPE;
declare variable MGR_NAME D_STR_100;

begin
  rep_date_begin = rep_date_end - extract(day from :rep_date_end) + 1;

  mo_id = -1;
--  if (rep_date_end >= '01.06.2021') then mo_id = 1010100066847; --текущего менеджера нужно добавить в текущу ОИ -- убран по письму Ироды от 24.11.2022


  insert into temp_bonus_office (BO_REGION_ID, BO_REG_LOGIC_LABEL, BO_LML_ID, BO_MANAGER_ID, BO_MANAGER_NAME, /*BO_LO_ID,*/ BO_PLAN_ALL, BO_FACT_ALL, BO_FACT_PLANNED,
                                 BO_FACT_OTHER, BO_MONTH_PLAN_OS, BO_MONTH_PLAN_PERC, BO_DELINQ_OS, BO_DELINQ_PERC,
                                 BO_FACT_OS, BO_FACT_PERC, BO_FACT_FINE, BO_FACT_DELINQ_OS, BO_FACT_DELINQ_PERC,
                                 BO_FACT_ADV_OS, BO_FACT_ADV_PERC, BO_BONUS_OS, BO_BONUS_OS_ADV)
  select reg_id,
         coalesce(reg_logic_label, ''),
         lml_id,
         manager_id,
         manager_name, --lo_id,
         sum(aarep_plan_all),
         sum(aarep_fact_all),
         sum(aarep_fact_planned),
         sum(aarep_fact_other),
         sum(aarep_month_plan_os),
         sum(aarep_month_plan_perc),
         sum(aarep_delinq_os),
         sum(aarep_delinq_perc),
         sum(aarep_fact_os),
         sum(aarep_fact_perc),
         sum(aarep_fact_fine),
         sum(aarep_fact_delinq_os),
         sum(aarep_fact_delinq_perc),
         sum(aarep_fact_adv_os),
         sum(aarep_fact_adv_perc),
         sum(aarep_bonus_os),
         sum(aarep_bonus_os_adv)
  from cb_ru_ssvz_202110_one_tmp(:rep_region_id, :rep_date_end), region
  where reg_id = reg_unicode
  and coalesce(reg_logic_label, '') not containing '.mini_office.'
  group by 1,2,3,4,5;

  perc_bonus_mo = 85;
  for
  select bo_region_id,
         bo_reg_logic_label /*, --bo_lml_id, bo_manager_id, bo_manager_name, --bo_lo_id,
         sum(bo_plan_all),
         sum(bo_fact_all),
         sum(bo_fact_planned),
         sum(bo_fact_other),
         sum(bo_month_plan_os),
         sum(bo_month_plan_perc),
         sum(bo_delinq_os),
         sum(bo_delinq_perc),
         sum(bo_fact_os),
         sum(bo_fact_perc),
         sum(bo_fact_fine),
         sum(bo_fact_delinq_os),
         sum(bo_fact_delinq_perc),
         sum(bo_fact_adv_os),
         sum(bo_fact_adv_perc),
         sum(bo_bonus_os),
         sum(bo_bonus_os_adv) */
  from temp_bonus_office
  where bo_reg_logic_label containing '.mega_office.'
  group by 1,2
  into reg_id, reg_logic_label /*, --lml_id, manager_id, manager_name, --lo_id,
       :aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os,
       aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine,
       aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc, aarep_bonus_os,
       aarep_bonus_os_adv */
  do
  begin
    aarep_bonus_total_fact_tmp = 0;
    for
    select bo_lml_id,
           bo_manager_id,
           bo_manager_name,
           sum(bo_plan_all),
           sum(bo_fact_all),
           sum(bo_fact_planned),
           sum(bo_fact_other),
           sum(bo_month_plan_os),
           sum(bo_month_plan_perc),
           sum(bo_delinq_os),
           sum(bo_delinq_perc),
           sum(bo_fact_os),
           sum(bo_fact_perc),
           sum(bo_fact_fine),
           sum(bo_fact_delinq_os),
           sum(bo_fact_delinq_perc),
           sum(bo_fact_adv_os),
           sum(bo_fact_adv_perc),
           sum(bo_bonus_os),
           sum(bo_bonus_os_adv)
    from temp_bonus_office
    where bo_region_id = :reg_id
    group by 1,2,3
    into lml_id, manager_id, manager_name, aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os,
       aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine,
       aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc, aarep_bonus_os,
       aarep_bonus_os_adv
    do
    begin

    -->>01.11.20
      aarep_bonus_plus = round((aarep_fact_perc + aarep_fact_fine + aarep_fact_delinq_perc + aarep_fact_adv_perc) * 0.015, 0);
      aarep_bonus_plus = round(aarep_bonus_plus + aarep_fact_delinq_os * 0.01, 0);

      aarep_bonus_minus = round((aarep_delinq_os + aarep_delinq_perc) * 0.03, 0);

      aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus; -- Базовый заработок

    --  Фактический баланс просрочек 31-360 дней
      fact_del_bal_31_360 = null;
      select sum(oks_balance)
      from oksummary
      where oks_region_id = :reg_id and
            :rep_date_end between oks_date_begin and oks_date_end and
            oks_balance > 0 and
            :rep_date_end - oks_delinq_date between 31 and 360 and
            oks_close_state = 0
            and :lml_id = f_lo_mgr_id(oks_from_k_id,:rep_date_end,oks_kc_id,oks_region_id)
      into fact_del_bal_31_360;
      fact_del_bal_31_360 = coalesce(fact_del_bal_31_360, 0);
      fact_del_bal_31_360 = fact_del_bal_31_360 / 1000000;

    --  Плановый баланс просрочек
      rep_period = extract(year from :rep_date_end) || right((extract(month from :rep_date_end) + 100), 2);
      plan_del_bal_31_360 = null;

      if (manager_id is null) then manager_id = -1;

      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :manager_id , 'PlanIndicators.BalancePar31_360', :rep_date_end, :rep_period)
      into plan_del_bal_31_360;

      plan_del_bal_31_360 = coalesce(plan_del_bal_31_360, 0);
      plan_del_bal_31_360 = plan_del_bal_31_360 / 1000000;

    --  Расчет KPI1
      kpi1 = 100 - (fact_del_bal_31_360 - plan_del_bal_31_360) * 20;
      if (kpi1 < 0) then
        kpi1 = 0;
      else
      if (kpi1 > 100) then
        kpi1 = 100;

    --  фактическое кол-во Випс на дату
      fact_kol_svz = null;
      select --count(*)
           sum(iif(jk_name = 'СВЗ 0,5',0.5,1))
      from hr_pstatus, joblist, jobkind
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = :reg_id and
            ps_is_working = 1 and
            ps_status = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code = 'JobSystem.LO_Standart' and
            (exists(select *
                   from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                   where is_decret = 0)
             or ps_from_c_id = :mo_id)
             and :lml_id = f_lo_mgr_id(null, :rep_date_end, ps_from_c_id, ps_from_reg_id)
      into fact_kol_svz;
      fact_kol_svz = coalesce(fact_kol_svz, 0);

    --  План кол-во СВЗ за указанный период
      plan_kol_svz = null;
      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :manager_id , 'PlanIndicators.LO_Count', :rep_date_end, :rep_period)
      into plan_kol_svz;

      plan_kol_svz = coalesce(plan_kol_svz, 0);

    --  Расчет KPI2
      if (fact_kol_svz >= plan_kol_svz) then
        kpi2 = 100;
      else
        kpi2 = 0;

      if (rep_date_end > '30.06.2021') then
      begin
        if (fact_kol_svz >= plan_kol_svz) then
          kpi2 = 100;
        else
        if (fact_kol_svz + 0.5 >= plan_kol_svz) then
          kpi2 = 50;
        else
          kpi2 = 0;
      end

    --  фактическая сумма выдачи за указанные период
      fact_disb_amount = null;
      select sum(k_amount_approved)
      from kontstate, kontract, knaktlist, product
      where ks_ol_is_delete = 0 and
            ks_state = 1 and
            ks_date_begin between :rep_date_begin and :rep_date_end and
            ks_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            knaktlist.is_delete = 0 and
            kal_new_reg_id = :reg_id and
            :rep_date_end between kal_date and kal_date_end and
            k_from_pr_id = pr_id and
            pr_code not containing '.express_online.'
            and :lml_id = f_lo_mgr_id(k_id, :rep_date_end, kal_new_kf_officer, kal_new_reg_id)
      into fact_disb_amount;
      fact_disb_amount = coalesce(fact_disb_amount, 0);
      fact_disb_amount = fact_disb_amount / 1000000;

      --  Плановые выдачи (в целях бонуса)
      plan_disb_amount = null;
  
      select sum(po_amount)
      from hp_get_planoper_mgr_info(:reg_id, :manager_id , 'PlanIndicators.Disb_amount_bonus', :rep_date_end, :rep_period)
      into plan_disb_amount;
  
      plan_disb_amount = coalesce(plan_disb_amount, 0);
      plan_disb_amount = plan_disb_amount / 1000000;
  
      --  Расчет KPI3
      kpi3 = 100 - (plan_disb_amount - fact_disb_amount) * 20;
      if (kpi3 < 0) then
        kpi3 = 0;
      else
      if (kpi3 > 100) then
        kpi3 = 100;
  
      tatl_kpi = kpi1 * 30 / 100 + kpi2 * 30 / 100 + kpi3 * 40 / 100;
  
      aarep_bonus_total_fact = aarep_bonus_total * (kpi1 * 30 / 100 + kpi2 * 30 / 100 + kpi3 * 40 / 100) / 100;
  
  
      if ((aarep_plan_all + aarep_fact_all + aarep_fact_planned -->>
          + aarep_delinq_os + aarep_delinq_perc + aarep_fact_os + aarep_fact_perc -->
          + aarep_fact_delinq_perc + aarep_fact_adv_os + aarep_fact_adv_perc) <> 0) then
      begin
        select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name --, r1.reg_logic_label
        from region r1, region r2
        where r1.reg_unicode = :reg_id and
              r1.parent = r2.reg_unicode
        into reg_name, branch_name; 
  
        if (manager_id <> -1) then
        begin
          select jl_name, c_name_fio, jk_system_code,c_id
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end
          and ps_status = 1
          and ps_is_working = 1
          and ps_from_jl_id = jl_id
          and jl_kind = jk_id
          and ps_from_c_id = :manager_id
          and ps_from_c_id = c_id
          into jl_name, officer_name, jk_system_code,c_id;
        end else
        begin
          select jl_name, c_name_fio, c_id, jk_system_code,c_id
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                       from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                       where is_decret = 0)
                  or ps_from_c_id = :mo_id
                       ) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                ps_from_c_id = c_id
          rows 1
          into jl_name, officer_name, manager_id, jk_system_code,c_id;
        end
  
        -- Расчет совокупного стажа сотрудинка, если менее 3х месяцев то бонус не получает
        select lo_stage_month_full
        from hp_get_lo_stage_full(:manager_id, :rep_date_end, :jk_system_code)
        into stage_month;
  
        aarep_bonus_officer = 0;
        if (aarep_bonus_total_fact > 0) then
        begin
          perc_bonus_mo = 85;
          if (stage_month >= 3.5) then
            aarep_bonus_officer = cast(aarep_bonus_total_fact as double precision) / 100 * perc_bonus_mo;
  
          aarep_bonus_total_fact_tmp = aarep_bonus_total_fact_tmp + aarep_bonus_total_fact;
        end
        if (officer_name is null) then c_id = null;

        suspend;
  
        jl_name = null;
        officer_name = null;
        jk_system_code = null;
        manager_id = null;
        c_id=null;
  
      end
  
    end

    if ((aarep_plan_all + aarep_fact_all + aarep_fact_planned -->>
        + aarep_delinq_os + aarep_delinq_perc + aarep_fact_os + aarep_fact_perc -->
        + aarep_fact_delinq_perc + aarep_fact_adv_os + aarep_fact_adv_perc) <> 0) then
    begin

      /*
      select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name --, r1.reg_logic_label
      from region r1, region r2
      where r1.reg_unicode = :reg_id and
            r1.parent = r2.reg_unicode
      into reg_name, branch_name; --, reg_logic_label;
      */

--      mgr_name = f_lo_mgr_name(:lml_id);
--      if (reg_logic_label = '.mega_office.' and mgr_name is not null) then reg_name = mgr_name;

--      suspend;

--      aarep_bonus_total_fact_tmp = aarep_bonus_total_fact;
--      if (aarep_bonus_total_fact_tmp < 0) then
--        aarep_bonus_total_fact_tmp = 0;

      aarep_delinq_os = 0;
      aarep_delinq_perc = 0;
      aarep_fact_os = 0;
      aarep_fact_perc = 0;
      aarep_fact_fine = 0;
      aarep_fact_delinq_os = 0;
      aarep_fact_delinq_perc = 0;
      aarep_fact_adv_os = 0;
      aarep_fact_adv_perc = 0;

      aarep_bonus_os = 0;
      aarep_bonus_os_adv = 0;
      sum_minus = 0;
      aarep_bonus_plus = 0;
      aarep_bonus_minus = 0;
      aarep_bonus_total = 0;
      aarep_bonus_total_fact = 0;

      fact_del_bal_31_360 = null;
      plan_del_bal_31_360 = null;
      fact_kol_svz = null;
      plan_kol_svz = null;
      fact_disb_amount = null;
      plan_disb_amount = null;
      tatl_kpi = null;

      kpi1 = null;
      kpi2 = null;
      kpi3 = null;

      --    вычисляем кол-во сотрудников по должностям
      kol_mo = 0;
      kol_zam_mo = 0;
      kol_kobd = 0;
      kol_admin = 0;
      for select jk_system_code, count(*), max(iif(jl_name <> 'Заместитель менеджера офиса' and jk_system_code = 'JobSystem.Region_KM', 1, 0)) -- если результат 1 то имеется менеджер офиса, если 0 то нету
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Specialist_KOBD', 'JobSystem.Administrator') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                       from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                       where is_decret = 0)
                 or ps_from_c_id = :mo_id
                       ) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                jl_name <> 'Выездной кассир-оператор' and
                ps_from_c_id = c_id -- and
--                :lml_id = f_lo_mgr_id(null, :rep_date_end, ps_from_c_id, ps_from_reg_id)
          group by jk_system_code
          into jk_system_code, kol, kol_mo
      do
      begin
        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
          kol_zam_mo = kol - kol_mo;
        end
        if (jk_system_code = 'JobSystem.Specialist_KOBD') then
        begin
          kol_kobd = kol;
        end
        if (jk_system_code = 'JobSystem.Administrator') then
        begin
          kol_admin = kol;
        end
      end

      perc_bonus_mo = 0;
      perc_bonus_zam_mo = 0;
      perc_bonus_kobd = 0;
      perc_bonus_admin = 0;
      --    нужно раскидать

      if (kol_kobd + kol_admin > 0) then
          perc_bonus_kobd = cast(15.0 as double precision) / (kol_kobd + kol_admin);
      perc_bonus_admin = perc_bonus_kobd;
      /*
      if (reg_logic_label = '.mega_office.') then
      begin
        perc_bonus_mo = 33.0;
        if (kol_zam_mo > 0) then
          perc_bonus_zam_mo = cast(52.0 as double precision) / kol_zam_mo;
        if (perc_bonus_zam_mo > 26.0) then
          perc_bonus_zam_mo = 26.0;
        if (kol_kobd > 0) then
          perc_bonus_kobd = cast(10.0 as double precision) / kol_kobd;
        if (kol_admin > 0) then
          perc_bonus_admin = cast(5.0 as double precision) / kol_admin;
      end
      else
      if (kol_zam_mo = 0) then
      begin
        perc_bonus_mo = 85.0;
        if (kol_kobd > 0) then
          perc_bonus_kobd = cast(15.0 as double precision) / kol_kobd;
      end
      else
      if (kol_zam_mo >= 1) then
      begin
        perc_bonus_mo = 45.0;
        if (kol_zam_mo > 0) then
          perc_bonus_zam_mo = cast(40.0 as double precision) / kol_zam_mo;
        if (kol_kobd > 0) then

      for select jl_name, c_name_fio, c_id, jk_system_code,c_id

          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and

                jl_name <> 'Выездной кассир-оператор' and
                ps_from_c_id = c_id and
--                :lml_id = f_lo_mgr_id(null, :rep_date_end, ps_from_c_id, ps_from_reg_id) and
                jk_system_code <> 'JobSystem.Region_KM' -- из этого списка убираем менеджера и зам МО
          order by jk_system_id, jl_name desc
          into jl_name, officer_name, officer_id, jk_system_code,c_id
      do
      begin

        aarep_bonus_officer = 0;
        stage_month = 0;

        select lo_stage_month_full
        from hp_get_lo_stage_full(:officer_id, :rep_date_end, :jk_system_code)
        into stage_month;

        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
          if (stage_month >= 3.5) then
          begin
            if (jl_name <> 'Заместитель менеджера офиса') then
              aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_mo;
            else
              aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_zam_mo;
          end
        end
        else
        if (stage_month >= 3.5) then
        begin
          if (jk_system_code = 'JobSystem.Specialist_KOBD') then
          begin
            aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_kobd;
          end
          else
          if (jk_system_code = 'JobSystem.Administrator') then
          begin
            aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_admin;

                               from fhp_round_bank(:aarep_bonus_officer, 0));

      if (officer_name is null) then c_id = null;
        suspend;

      aarep_bonus_officer = null;
      jl_name = null;
      officer_name = null;
      officer_id = null;

  for

         sum(bo_plan_all),
         sum(bo_fact_all),

         sum(bo_month_plan_perc),
         sum(bo_delinq_os),
         sum(bo_delinq_perc),
         sum(bo_fact_os),
         sum(bo_fact_perc),

         sum(bo_bonus_os),
         sum(bo_bonus_os_adv)
  from temp_bonus_office
  where bo_reg_logic_label not containing '.mega_office.'
  group by 1,2
  into reg_id, reg_logic_label, --lml_id, manager_id, manager_name, --lo_id,
       :aarep_plan_all, aarep_fact_all, aarep_fact_planned, aarep_fact_other, aarep_month_plan_os,
       aarep_month_plan_perc, aarep_delinq_os, aarep_delinq_perc, aarep_fact_os, aarep_fact_perc, aarep_fact_fine,
       aarep_fact_delinq_os, aarep_fact_delinq_perc, aarep_fact_adv_os, aarep_fact_adv_perc, aarep_bonus_os,
       aarep_bonus_os_adv
  do
  begin
        -->>01.11.20
    aarep_bonus_plus = round((aarep_fact_perc + aarep_fact_fine + aarep_fact_delinq_perc + aarep_fact_adv_perc) * 0.015, 0);
    aarep_bonus_plus = round(aarep_bonus_plus + aarep_fact_delinq_os * 0.01, 0);

    aarep_bonus_minus = round((aarep_delinq_os + aarep_delinq_perc) * 0.03, 0);

    aarep_bonus_total = aarep_bonus_plus - aarep_bonus_minus; -- Базовый заработок

    --  Фактический баланс просрочек 31-360 дней
    fact_del_bal_31_360 = null;
    select sum(oks_balance)
    from oksummary
    where oks_region_id = :reg_id and
          :rep_date_end between oks_date_begin and oks_date_end and
          oks_balance > 0 and
          :rep_date_end - oks_delinq_date between 31 and 360 and
          oks_close_state = 0
    into fact_del_bal_31_360;
    fact_del_bal_31_360 = coalesce(fact_del_bal_31_360, 0);
    fact_del_bal_31_360 = fact_del_bal_31_360 / 1000000;

    --  Плановый баланс просрочек
    select rd_id
    from refdata
    where rd_code = 'PlanIndicators.BalancePar31_360'
    into rd_id;
    rep_period = extract(year from :rep_date_end) || right((extract(month from :rep_date_end) + 100), 2);
    plan_del_bal_31_360 = null;
    select sum(pd_data_1)
    from hp_tmp_plandata_show(:reg_id, null, :rd_id, :rep_period, 0, 0)
    where pd_is_office = 1
    into plan_del_bal_31_360;
    plan_del_bal_31_360 = coalesce(plan_del_bal_31_360, 0);
    plan_del_bal_31_360 = plan_del_bal_31_360 / 1000000;

    --  Расчет KPI1
    kpi1 = 100 - (fact_del_bal_31_360 - plan_del_bal_31_360) * 20;
    if (kpi1 < 0) then
      kpi1 = 0;
    else
    if (kpi1 > 100) then
      kpi1 = 100;

    --  фактическое кол-во Випс на дату
    fact_kol_svz = null;
    select --count(*)
           sum(iif(jk_name = 'СВЗ 0,5',0.5,1))

    where :rep_date_end between ps_date_begin and ps_date_end and

    plan_kol_svz = null;
    select sum(pd_data_1)
    from hp_tmp_plandata_show(:reg_id, null, :rd_id, :rep_period, 0, 0)

    into plan_kol_svz;

    --  Расчет KPI2
    if (fact_kol_svz >= plan_kol_svz) then
      kpi2 = 100;
    else
      kpi2 = 0;

    begin
      if (fact_kol_svz >= plan_kol_svz) then
        kpi2 = 100;
      else
      if (fact_kol_svz + 0.5 >= plan_kol_svz) then
        kpi2 = 50;
      else
        kpi2 = 0;
    end

    fact_disb_amount = null;
    select sum(k_amount_approved)
    from kontstate, kontract, knaktlist, product
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          ks_date_begin between :rep_date_begin and :rep_date_end and
          ks_from_k_id = k_id and

    fact_disb_amount = fact_disb_amount / 1000000;

    select rd_id
    from refdata
    where rd_code = 'PlanIndicators.Disb_amount_bonus'
    into rd_id;
    --  Плановые выдачи (в целях бонуса)

    if (kpi3 < 0) then
      kpi3 = 0;
    else
    if (kpi3 > 100) then
      kpi3 = 100;

        + aarep_delinq_os + aarep_delinq_perc + aarep_fact_os + aarep_fact_perc -->

      from region r1, region r2

            r1.parent = r2.reg_unicode

      suspend;
      aarep_bonus_total_fact_tmp = aarep_bonus_total_fact;
      if (aarep_bonus_total_fact_tmp < 0) then
        aarep_bonus_total_fact_tmp = 0;

      aarep_delinq_os = 0;
      aarep_delinq_perc = 0;
      aarep_fact_os = 0;
      aarep_fact_perc = 0;
      aarep_fact_fine = 0;
      aarep_fact_delinq_os = 0;
      aarep_fact_delinq_perc = 0;
      aarep_fact_adv_os = 0;
      aarep_fact_adv_perc = 0;

      aarep_bonus_os = 0;
      aarep_bonus_os_adv = 0;
      sum_minus = 0;
      aarep_bonus_plus = 0;
      aarep_bonus_minus = 0;
      aarep_bonus_total = 0;
      aarep_bonus_total_fact = 0;

      plan_del_bal_31_360 = null;
      fact_kol_svz = null;
      plan_kol_svz = null;

      kpi2 = null;

      kol_kobd = 0;
      kol_admin = 0;
      for select jk_system_code, count(*), max(iif(jl_name <> 'Заместитель менеджера офиса' and jk_system_code = 'JobSystem.Region_KM', 1, 0)) -- если результат 1 то имеется менеджер офиса, если 0 то нету
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Specialist_KOBD', 'JobSystem.Administrator') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                       from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                       where is_decret = 0)
                 or ps_from_c_id = :mo_id
                       ) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                jl_name <> 'Выездной кассир-оператор' and
                ps_from_c_id = c_id
          group by jk_system_code
          into jk_system_code, kol, kol_mo
      do
      begin
        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
          kol_zam_mo = kol - kol_mo;
        end
        if (jk_system_code = 'JobSystem.Specialist_KOBD') then
        begin
          kol_kobd = kol;
        end
        if (jk_system_code = 'JobSystem.Administrator') then
        begin
          kol_admin = kol;
        end
      end

      perc_bonus_mo = 0;
      perc_bonus_zam_mo = 0;
      perc_bonus_kobd = 0;
      perc_bonus_admin = 0;
      --    нужно раскидать
      if (reg_logic_label = '.mega_office.') then
      begin
        perc_bonus_mo = 33.0;
        if (kol_zam_mo > 0) then
          perc_bonus_zam_mo = cast(52.0 as double precision) / kol_zam_mo;
        if (perc_bonus_zam_mo > 26.0) then
          perc_bonus_zam_mo = 26.0;
        if (kol_kobd > 0) then
          perc_bonus_kobd = cast(10.0 as double precision) / kol_kobd;
        if (kol_admin > 0) then
          perc_bonus_admin = cast(5.0 as double precision) / kol_admin;
      end
      else
      if (kol_zam_mo = 0) then
      begin
        perc_bonus_mo = 85.0;
        if (kol_kobd > 0) then
          perc_bonus_kobd = cast(15.0 as double precision) / kol_kobd;
      end
      else
      if (kol_zam_mo >= 1) then
      begin
        perc_bonus_mo = 45.0;
        if (kol_zam_mo > 0) then
          perc_bonus_zam_mo = cast(40.0 as double precision) / kol_zam_mo;
        if (kol_kobd > 0) then
          perc_bonus_kobd = cast(15.0 as double precision) / kol_kobd;
      end

      for select jl_name, c_name_fio, c_id, jk_system_code,c_id
          from hr_pstatus, joblist, jobkind, clients
          where :rep_date_end between ps_date_begin and ps_date_end and
                ps_status = 1 and
                ps_is_working = 1 and
                ps_from_c_id > 1000000000003 and
                ps_from_jl_id = jl_id and
                jl_kind = jk_id and
                jk_system_code in ('JobSystem.Region_KM', 'JobSystem.Specialist_KOBD', 'JobSystem.Administrator') and
                ps_from_reg_id = :reg_id and
                (exists(select *
                       from hp_employer_in_dekret(ps_from_c_id, :rep_date_end)
                       where is_decret = 0)
                  or ps_from_c_id = :mo_id
                       ) and
                jl_name <> 'И.о. заместителя менеджера офиса' and
                jl_name <> 'Выездной кассир-оператор' and
                ps_from_c_id = c_id
          order by jk_system_id, jl_name desc
          into jl_name, officer_name, officer_id, jk_system_code,c_id
      do
      begin

        aarep_bonus_officer = 0;
        stage_month = 0;
        --      Расчет совокупного стажа сотрудинка, если менее 3х месяцев то бонус не получает
        select lo_stage_month_full
        from hp_get_lo_stage_full(:officer_id, :rep_date_end, :jk_system_code)
        into stage_month;

        if (jk_system_code = 'JobSystem.Region_KM') then
        begin
          if (stage_month >= 3.5) then
          begin
            if (jl_name <> 'Заместитель менеджера офиса') then
              aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_mo;
            else
              aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_zam_mo;
          end
        end
        else
        if (stage_month >= 3.5) then
        begin
          if (jk_system_code = 'JobSystem.Specialist_KOBD') then
          begin
            aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_kobd;
          end
          else
          if (jk_system_code = 'JobSystem.Administrator') then
          begin
            aarep_bonus_officer = cast(aarep_bonus_total_fact_tmp as double precision) / 100 * perc_bonus_admin;
          end
        end

        aarep_bonus_officer = (select res
                               from fhp_round_bank(:aarep_bonus_officer, 0));

      if (officer_name is null) then c_id = null;
        suspend;
      end

      aarep_bonus_officer = null;
      jl_name = null;
      officer_name = null;
      officer_id = null;
      c_id=null;
    end
  end
