CREATE PROCEDURE HP_FINEREFUND_TWO
AS
declare variable K_KONTRACT_NO varchar(20);
declare variable PAL_MAIN_ACC varchar(20);
declare variable OL_DATE D_DATE;
declare variable OK_FINE D_CURRENCY;
declare variable SH_ID D_INTEGER;
declare variable SH_DATE D_DATE;
declare variable K_LAST_REG_ID integer;
declare variable NEW_OL_ID D_ID;
declare variable REG_CODE varchar(10);
declare variable ACC_KT1 varchar(50);
declare variable ACC_KT varchar(20);
declare variable ACC_DT varchar(20);
declare variable K_KOLICH_CLIENT_APPROVED integer;
declare variable ACC varchar(20);
declare variable SH_DATE_BEGIN date;
declare variable SH_ID_BEGIN integer;
declare variable FR_FROM_K_ID D_ID;
declare variable KS_STATE integer;
begin
  for select f.fr_from_k_id
      from finerefund f
      where f.fr_status in (-1,-2)
      into fr_from_k_id
  do
  begin
    sh_id = null;
    sh_date = null;
    ol_date = null;
    ok_fine = null;
    for select ok_sh_id, ok_sh_date, max(ok_ol_date), sum(ok_fine)
        from operkred ok
        where ok_from_k_id = :fr_from_k_id and
              ok_ol_is_active = 1 and
              ok_type = 4 and
              ok_ol_date <= current_date and
              ok.ok_sh_date >= '19.03.2020'
        group by ok_sh_id, ok_sh_date
        having sum(ok_fine) < 0
        into sh_id, sh_date, ol_date, ok_fine
    do
    begin
      select pal_main_acc || '/' || pal_main_acc_no, k_kontract_no, k_last_reg_id, ks.ks_state,
             k_kolich_client_approved
      from kontract, product, prodacclist, kontstate ks
      where k_id = :fr_from_k_id and
            k_from_pr_id = pr_id and
            pr_from_pat_id = pal_from_pat_id and
            pal_from_at_id = 5 and
            ks.ks_from_k_id = k_id and
            ks.ks_ol_is_delete = 0 and
            current_date between ks.ks_date_begin and ks.ks_date_end
      into pal_main_acc, k_kontract_no, k_last_reg_id, ks_state, k_kolich_client_approved;

      sh_date_begin = null;
      sh_id_begin = null;
      select min(sh_date), sh_id
      from aarep_make_zad_list(:fr_from_k_id, current_date, 1)
      group by sh_id 
      order by sh_id rows 1
      into sh_date_begin, sh_id_begin;

      reg_code = null;
      select r.reg_code
      from region r
      where r.reg_unicode = :k_last_reg_id
      into reg_code;

      acc_kt = (select al_id
                from acc_list
                where al_from_reg_id = :k_last_reg_id and
                      al_currency = 417 and
                      al_account = '21199/1');

      if (coalesce(acc_kt, '') = '') then
        exception e_message 'счет кт не определен';

      acc_dt = (select al_id
                from acc_list
                where al_account = :pal_main_acc and
                      al_from_reg_id = :k_last_reg_id);

      if (coalesce(acc_dt, '') = '') then
        exception e_message 'счет дт не определен';

      if (k_kolich_client_approved > 1) then
        acc_kt1 = :reg_code || '41710931000001';  --счет по ОС групповой
      else
        acc_kt1 = :reg_code || '41710941000001';  -- счет по ОС индив

      if (k_kolich_client_approved > 1) then
        acc = '10931/1';
      else
        acc = '10941/1';

      if (ks_state > 1) then
      begin
        select kv
        from get_kv(:k_last_reg_id, 5)
        into :new_ol_id;

        insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                              ol_user_id, ol_doc_client_name)
        values (:new_ol_id, :k_last_reg_id, 1, current_date, 8, 'по распоряжению №08/1 от 23.07.2021',
                0, 0, (select un_id
                       from username
                       where un_name = 'fdjaparova'), 'по распоряжению  №08/1 от 23.07.2021');
        --ok_type = 4 погашение
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                              ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
        values (:new_ol_id, -1, :fr_from_k_id, :sh_id, :sh_date, 4, 0, 0, -:ok_fine, 3, 0, 0, 0, 0, 0, :ol_date,
                :k_last_reg_id, 1, 0);

        --ok_type = 9 переплата
        insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                              ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                              ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
        values (:new_ol_id, -1, :fr_from_k_id, :sh_id, current_date, 9, :ok_fine, 0, 0, 3, 0, 0, 0, 0, 0, :ol_date,
                :k_last_reg_id, 1, 0);

        -- бух проводки
        insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                              ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                              ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
                              ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt)
        values (:new_ol_id, -1, '417', 1, null, :acc_dt, -:ok_fine, -:ok_fine, null, null, null, :acc_kt, null,
                f_format('переплата к возврату заемщику по кр.договору : %1', :k_kontract_no),
                0, 0, null, :ol_date, 1, :k_last_reg_id, :pal_main_acc, '21199/1');
        --suspend;
        continue;
      end
      --создаем операцию
      select kv
      from get_kv(:k_last_reg_id, 5)
      into :new_ol_id;

      insert into operlist (ol_id, ol_from_reg_id, ol_oper_type, ol_date, ol_doc_kind, ol_memo, ol_status, is_delete,
                            ol_user_id, ol_doc_client_name)
      values (:new_ol_id, :k_last_reg_id, 1, current_date, 8, 'по распоряжению №08/1 от 23.07.2021',
              0, 0, (select un_id
                     from username
                     where un_name = 'fdjaparova'), 'по распоряжению №08/1 от 23.07.2021');
      --ok_type = 4 погашение
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                            ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
      values (:new_ol_id, -1, :fr_from_k_id, :sh_id, :sh_date, 4, 0, 0, -:ok_fine, 3, 0, 0, 0, 0, 0, :ol_date,
              :k_last_reg_id, 1, 0);

      --    --ok_type = 4 погашение
      insert into operkred (ok_from_ol_id, ok_number, ok_from_k_id, ok_sh_id, ok_sh_date, ok_type, ok_os, ok_perc,
                            ok_fine, ok_pay_type, ok_kommis, ok_sh_os, ok_sh_perc, ok_kommis_adv, ok_tax, ok_ol_date,
                            ok_ol_reg_id, ok_ol_is_active, ok_from_c_id)
      values (:new_ol_id, -1, :fr_from_k_id, :sh_id_begin, :sh_date_begin, 4, :ok_fine, 0, 0, 3, 0, 0, 0, 0, 0,
              :ol_date, :k_last_reg_id, 1, 0);

      -- бух проводки
      insert into operbook (ob_from_ol_id, ob_number, ob_currency, ob_curr_rate, ob_client_dt, ob_account_dt,
                            ob_summa_val, ob_summa, ob_from_pc_id, ob_from_kp_id, ob_oper_mask, ob_account_kt,
                            ob_client_kt, ob_detail, ob_subkonto_dt, ob_subkonto_kt, ob_from_sac_id, ob_ol_date,
                            ob_ol_is_active, ob_ol_reg_id, ob_short_acc_dt, ob_short_acc_kt)
      values (:new_ol_id, -1, '417', 1, null, :acc_dt, -:ok_fine, -:ok_fine, null, null, null, :acc_kt1, null,
              f_format('погашение ОС за счет переплаты пени по кр.договору: %1', :k_kontract_no),
              0, 0, null, :ol_date, 1, :k_last_reg_id, :pal_main_acc, :acc);
      --suspend;
    end
  end
end
