CREATE PROCEDURE MONETA_DO_DISBURSEMENT(
    K_ID ID_TYPE,
    C_ID ID_TYPE,
    DISB_AMOUNT CURR_TYPE)
AS
declare variable esm_account d_str_code;
declare variable cell_no d_str_100;
declare variable fpb_code d_str_code;
declare variable esm_action d_str_20;
declare variable k_disb_kind d_smallint;
declare variable fpl_payment_id d_str_code;
declare variable json_data d_str_255;
declare variable bank_bik d_str_10;
declare variable client_account d_str_code;
declare variable edoc_pmt_project_disb d_str_255;
begin
  edoc_pmt_project_disb = f_get_const('edoc_pmt_project_disb');
  
  select esm_action
  from extsysmain em
  where esm_project = f_format('%1_disb', :edoc_pmt_project_disb) and
       esm_action in ('CardDisb', 'PhoneDisb', 'AccDisb') and
       esm_from_k_id = :k_id and
       esm_status in (0, 1, 999)
  into esm_action;

  if (esm_action is not null) then
    exit;

  select k_disb_kind from kontract
  where k_id = :k_id
  into k_disb_kind;

  esm_action = 'CardDisb';
  if (k_disb_kind = 7) then
    esm_action = 'PhoneDisb';
  else if (k_disb_kind = 8) then
    esm_action = 'AccDisb';

  if (esm_action = 'CardDisb') then
  begin
    select token
    from hp_client_bank_card(:c_id)
    into esm_account;
  
    if (esm_account is null) then
      exception e_default 'Token is not exists';
  end

  if (esm_action = 'PhoneDisb') then
  begin
    select fpb_code, cell_no
    from hp_fps_get_loan_bank(:k_id, :c_id)
    into fpb_code, cell_no;

    if (fpb_code is null or cell_no is null ) then
      exception e_default f_format('FPS bank not found for k_id:%1, c_id:%2', k_id, c_id);

    esm_account = f_format('%1:%2', cell_no, fpb_code);

    -- Подготовка платежного идентификатора для MONETA_FPS_CREDITS_FOR_ADD
    select fpl_payment_id
    from fps_locks
    where fpl_from_k_id = :k_id
    into fpl_payment_id;
    json_data = '{' || f_json_make_pair('payment_id', fpl_payment_id, 'str') || '}';
  end

  if (esm_action = 'AccDisb') then
  begin

    select bank_bik, client_account
    from hp_bankacc_get_info(null, :c_id)
    into bank_bik, client_account;

    if (bank_bik is null or client_account is null) then
      exception e_default f_format('Bank or client account not found for k_id:%1, c_id:%2', k_id, c_id);

    esm_account = f_format('%1:%2', bank_bik, client_account);
  end

  insert into extsysmain (
    esm_project, esm_action, esm_date, esm_object_kind, esm_object_id, esm_status,
    esm_full_source, esm_amount, esm_from_k_id, esm_account)
  values (f_format('%1_disb', :edoc_pmt_project_disb), :esm_action, current_date, 0, :c_id, 0, :json_data, :disb_amount, :k_id, :esm_account);
  
  post_event 'request_payout';

end
