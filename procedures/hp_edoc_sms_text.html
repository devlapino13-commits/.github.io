CREATE PROCEDURE HP_EDOC_SMS_TEXT(
    EDS_ID D_ID,
    MAIN_MSG_ID D_ID,
    FORCE_SMS D_SMALLINT = 0)
RETURNS (
    ESC_MESSAGE D_STR_LONG,
    CONT_INFO D_STR_100,
    RES_TEXT D_STR_100)
AS
declare variable MSG_ID D_ID;
declare variable EDM_FROM_MSG_ID D_ID;
declare variable PIN_CODE D_STR_CODE;
declare variable CDA_ID D_INTEGER;
declare variable CLT_ID D_INTEGER;
declare variable EDC_ID D_INTEGER;
declare variable EDC_STATE D_SMALLINT;
declare variable EDC_MSG_KIND D_SMALLINT;
declare variable UUID_TXT D_STR_LONG;
declare variable UN_ID D_ID;
declare variable C_ID D_ID;
declare variable SMS_BODY D_STR_LONG;
declare variable PUSH_TITLE D_STR_100;
declare variable PUSH_BODY D_STR_2048;
declare variable PUSH_BODY_TMP D_STR_2048;
declare variable GOOGLE_ID D_STR_255;
declare variable EDL_COUNT D_INTEGER;
declare variable PUSH_LIFETIME_DAYS D_SMALLINT;
declare variable LANG D_STR_LANG;
declare variable MCL_ID D_ID;
declare variable REP_ID D_STR_CODE;
declare variable DB_ID D_INTEGER;
declare variable C_NAME_FIO D_STR_100;
declare variable C_NAME_FIO_LAT D_STR_100;
declare variable C_NAME D_STR_100;
declare variable TPL_PIN D_STR_CODE;
declare variable TPL_PARAMS D_STR_CODE;
declare variable TPL_NAME_FIRST D_STR_CODE;
declare variable TPL_NAME_FULL D_STR_CODE;
declare variable TPL_NAME_SHORT D_STR_CODE;
declare variable TPL_NAME_LAT_FIRST D_STR_CODE;
declare variable TPL_NAME_LAT_FULL D_STR_CODE;
declare variable TPL_NAME_LAT_SHORT D_STR_CODE;
declare variable EDL_OBJECT_KIND D_INTEGER;
declare variable EDL_OBJECT_ID D_ID;
declare variable EDC_CODE D_STR_CODE;
declare variable EDS_PAIR D_STR_CODE;
declare variable FORMATTED_PIN D_STR_CODE;
declare variable MCL_PROJECT D_SMALLINT;
declare variable KEYCAP D_STR_SYM;
declare variable MSG_SENDER D_INTEGER;
declare variable OBJ_KIND D_INTEGER;
declare variable OBJ_ID D_ID;
begin
  -- Если main_msg_id is not null, процедура не производит отправку сообщения
  -- В остальных случаях отправляет СМС/PUSH, в зависимости от edc_msg_kind
  -- Возвращает текст СМС/PUSH и номер телефона куда будет отправлен СМС

  esc_message = '';
  res_text = '';
  c_name_fio = '';
  db_id = f_get_db_id();

  -- Шаблоны автоподстановки
  -- Для тела сообщения (sms, push)
  tpl_pin = '[pin]';                        -- 1234
  tpl_params = '[params]';                  -- https://edoc.molbulak.ru/D2CE9D50-5710-4078-8751-AF353A7435A1

  -- Для тела сообщения (sms, push), включая заголовок push
  tpl_name_first = '[name_first]';          -- Иван
  tpl_name_full = '[name_full]';            -- Иванов Иван Иванович
  tpl_name_short = '[name_short]';          -- Иванов И.И.
  tpl_name_lat_first = '[name_lat_first]';  -- Ivan
  tpl_name_lat_full = '[name_lat_full]';    -- Ivanov Ivan Ivanovich
  tpl_name_lat_short = '[name_lat_short]';  -- Ivanov I.I.

  -- получаем данные электронного документа
  select eds_from_c_id, c_name_fio, c_alternat_fio, edc_id, edc_msg_kind, edc_sms_mask,
         edc_push_title, edc_push_body, edl_create_un_id, edc_state, edl_object_kind, edl_object_id, edc_code
  from edoc_sign, edoc_list, edoc_cfg, clients
  where eds_id = :eds_id and
        eds_from_edl_id = edl_id and
        edl_from_edc_id = edc_id and
        eds_from_c_id is not null and
        eds_from_c_id = c_id
  into c_id, c_name_fio, c_name_fio_lat, edc_id, edc_msg_kind, sms_body, push_title, push_body, un_id, edc_state,
       edl_object_kind, edl_object_id, edc_code;

  if (row_count = 0) then
    exit;

  select res_sms_body, res_push_title, res_push_body
  from hp_edoc_sms_text_replace(:db_id, :c_id, :edc_id, :edc_code, :edl_object_kind, :edl_object_id, :sms_body, :push_title, :push_body)
  into sms_body, push_title, push_body;

  c_name_fio_lat = coalesce(c_name_fio_lat, '');

  -- если main_msg_id не передан изначально, бронируем номер
  if (main_msg_id is not null)
  then msg_id = main_msg_id;
  else msg_id = next value for gen_fsrobot_msgbox_id;

  edm_from_msg_id = msg_id;

  -- получаем контактную информацию клиента
  select cont_info, cda_id
  from hp_edoc_client_continfo(:edc_id, :c_id, :un_id)
  into cont_info, cda_id;

  select msg_kind
  from hp_edoc_get_msg_kind(:edl_object_kind, :edl_object_id, :edc_id, :eds_id)
  into edc_msg_kind;

  --RFCKG-2742
  --Не отправляем СМС или ПУШ, в случае если пользователь пытается отправить СМС или ПУШ по данному edc_msg_kind
  if (edc_msg_kind = 6) then
    force_sms = 0;

  --Если force_sms = 1, то в любом случае отправляем СМС
  if (force_sms = 1) then
    edc_msg_kind = 0;

  --В КГ разделено согласие супруга на займ от заявки
  --Если это не принудительная отправка СМС по согласию супруга на займ, то просто создаётся документ для подписания через биометрию
  if (db_id in (39801, 41701) and edc_code = 'loan.spouses_consent' and force_sms = 0) then
    edc_msg_kind = 4;

  -- RFCKG-5043. Сделано для того чтобы, отправка документа на подписание осуществлялась в КМП, оставляя возможность подписания биометрией
  -- edc_msg_kind = 4. Добавил данную проверку так как если edc_msg_kind <> 4 значит возможность отправки кроме смс не рассматривается
  if (db_id = 41701 and edc_msg_kind = 4 and edc_code = 'persdata.consent') then
    edc_msg_kind = 2;

  --Описание edc_msg_kind в edoc_cfg.edc_msg_kind
  if (coalesce(edc_msg_kind, 0) in (3, 4, 5, 6, 7, 8)) then -- Добавил 8 (Подписание ПКБ)
  begin
    esc_message = '';
    res_text = '';
    suspend;
    exit;
  end 

  clt_id = f_cont_id(cont_info, 'add');

  -- Обработка шаблона пин-кода
  if (sms_body containing tpl_pin or push_body containing tpl_pin) then
  begin
    pin_code = right('0000' || cast(trunc(rand() * 100000) as integer), 4);

    -- RFCKG-3966 эти квадратики не ошибка, а символы для пуша
    keycap =_utf8 x'E283A3';

    formatted_pin = (substring(pin_code from 1 for 1) || keycap ||
                     substring(pin_code from 2 for 1) || keycap ||
                     substring(pin_code from 3 for 1) || keycap ||
                     substring(pin_code from 4 for 1) || keycap);

    sms_body = replace(sms_body, tpl_pin, pin_code);
    push_body = replace(push_body, tpl_pin, formatted_pin);
  end

  -- Обработка шаблона ссылки
  if (sms_body containing tpl_params or push_body containing tpl_params) then
  begin
    uuid_txt = case db_id
      when 81001 then 'https://edoc.molbulak.ru/'
      when 41701 then 'https://edoc.mbulak.kg/'
      when 94401 then 'https://edoc.molbulak.com/'
      when 39801 then 'https://edoc.mbulak.kz/'
      else ''
    end;

    if (uuid_txt = '') then
      res_text = 'Неверный код БД:' || coalesce(db_id, 'NULL');

    uuid_txt = trim(uuid_txt) || f_uuid('edoc_sign', eds_id);

    -- Если отправка по почте, обрабатываем как ссылку
    if (main_msg_id is not null) then
      uuid_txt = f_format('<a href="%1">%1</a>', uuid_txt);

    sms_body = replace(sms_body, tpl_params, uuid_txt);
    push_body = replace(push_body, tpl_params, uuid_txt);
  end


  -- Обработка шаблона ФИО клиента - латиница
  sms_body = replace(sms_body, tpl_name_lat_full, c_name_fio_lat);
  push_title = replace(push_title, tpl_name_lat_full, c_name_fio_lat);
  push_body = replace(push_body, tpl_name_lat_full, c_name_fio_lat);

  -- Обработка шаблона краткое ФИО клиента - латиница
  if (sms_body containing tpl_name_lat_short or
      push_title containing tpl_name_lat_short or
      push_body containing tpl_name_lat_short) then
  begin
    select short_fio
    from hp_fio_clear(:c_name_fio_lat)
    into c_name;
    if (c_name = '') then c_name = 'Klient';

    sms_body = replace(sms_body, tpl_name_lat_short, c_name);
    push_title = replace(push_title, tpl_name_lat_short, c_name);
    push_body = replace(push_body, tpl_name_lat_short, c_name);
  end

  -- Обработка шаблона имя клиента - латиница
  if (sms_body containing tpl_name_lat_first or
      push_title containing tpl_name_lat_first or
      push_body containing tpl_name_lat_first) then
  begin
    select txt2
    from hp_fio_clear(:c_name_fio_lat)
    into c_name;
    if (c_name = '') then c_name = 'Klient';
    c_name = left(c_name, 1) || lower(substring(c_name from 2));

    sms_body = replace(sms_body, tpl_name_lat_first, c_name);
    push_title = replace(push_title, tpl_name_lat_first, c_name);
    push_body = replace(push_body, tpl_name_lat_first, c_name);
  end

  -- Обработка шаблона ФИО клиента
  sms_body = replace(sms_body, tpl_name_full, c_name_fio);
  push_title = replace(push_title, tpl_name_full, c_name_fio);
  push_body = replace(push_body, tpl_name_full, c_name_fio);

  -- Обработка шаблона краткое ФИО клиента
  if (sms_body containing tpl_name_short or push_body containing tpl_name_short) then
  begin
    select short_fio
    from hp_fio_clear(:c_name_fio)
    into c_name;
    if (c_name = '') then c_name = 'Клиент';

    sms_body = replace(sms_body, tpl_name_short, c_name);
    push_title = replace(push_title, tpl_name_short, c_name);
    push_body = replace(push_body, tpl_name_short, c_name);
  end

  -- Обработка шаблона имя клиента
  if (sms_body containing tpl_name_first or push_body containing tpl_name_first) then
  begin
    select txt2
    from hp_fio_clear(:c_name_fio)
    into c_name;
    if (c_name = '') then c_name = 'Клиент';
    c_name = left(c_name, 1) || lower(substring(c_name from 2));

    sms_body = replace(sms_body, tpl_name_first, c_name);
    push_title = replace(push_title, tpl_name_first, c_name);
    push_body = replace(push_body, tpl_name_first, c_name);
  end

  if (edc_msg_kind not in (0, 1, 2)) then
    res_text = 'Неверный код edc_msg_kind: '|| coalesce(edc_msg_kind, 'NULL');

  esc_message = sms_body;
  if (edc_msg_kind = 2) then
    esc_message = push_body;
  if (esc_message = '') then
    res_text = 'Текст сообщения не сформирован';

  if (res_text <> '') then
  begin
    suspend;
    exit;
  end
  
  esc_message = sms_body;


  if (edc_state = 3) then -- если боевой режим, шифруем пин-код
    pin_code = f_md5(:pin_code);

  -- Если НЕ отправка по почте, делаем отправку
  if (main_msg_id is null) then
  begin
    -- RFCRU-3766
    if (db_id <> 81001) then
        mcl_project = 1;
    else
    begin
        -- RFCRU-4254
        -- Для определения mcl_project ожидаем, что obj_kind будет: clients/miniapp/kontract
        if (edl_object_kind not in (f_objkind('clients', 'id'), f_objkind('miniapp', 'id'), f_objkind('kontract', 'id'))) then
        begin
            -- Если obj_kind отличается от ожидаемого, пытаемся вытащить из него один из 3 ожидаемых
            select obj_kind, obj_id
            from hp_edoc_transform_obj_kind(:edl_object_kind, :edl_object_id)
            into :obj_kind, :obj_id;
        end

        -- Если процедура выше не смогла преобразовать obj_kind и obj_id к ожидаемым значениям, используем фактические
        obj_kind = coalesce(obj_kind, :edl_object_kind);
        obj_id = coalesce(obj_id, :edl_object_id);

        mcl_project = f_get_mcl_project(obj_kind, obj_id);
    end

    -- QuickSMS-сообщение
    if (edc_msg_kind in (0, 1)) then
    begin
      -- RFCRU-4045
      msg_sender = null;

      if (db_id = 81001) then
        msg_sender = iif (:mcl_project = 2, 1, 0);

      insert into fsrobot_msgbox (
        msg_id, msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days,
        msg_from_un_id, msg_command_kind, msg_object_kind, msg_object_id, msg_sender)
      values (
        :msg_id, 1, :cont_info, '', :sms_body, -1, :un_id, '',
        f_objkind('clients', 'id'), :c_id, :msg_sender);

    end

    -- PUSH-сообщение
    if (edc_msg_kind in (0, 2)) then
    begin
      -- На данном этапе мы определяем на какое устройство отправлять пуш уведомление

      -- при повторном проходе в цикле, в основной push_body добавляется тот, что был на прошлом итерации, с помощью новой переменной мы это избегаем
      push_body_tmp = push_body;

      -- Значения по-умолчанию
      rep_id = 'edoc_sign_notification';
      push_lifetime_days = -1;

      -- Изначально берём аккаунт со свежим google_id
      for with mcl_projects as
      (
        select mcl_google_id, mcl_id, mcl_project,
          iif(datediff(day, mcl_last_login_datetime, current_date) <= 90 and coalesce(mcl_google_id, '') <> '', 0, 1) as row_order
        from mc_login
        where mcl_from_c_id = :c_id
          and mcl_is_delete = 0
      ),
      project_kmp as
      (
        select first 1 mcl_google_id, mcl_id
        from mcl_projects
        where :mcl_project in (0, 1)
          and mcl_project = 1
        order by row_order asc, mcl_id desc
      ),
      project_oivo as
      (
        select first 1 mcl_google_id, mcl_id
        from mcl_projects
        where :mcl_project in (0, 2)
          and mcl_project = 2
        order by row_order asc, mcl_id desc
      )
      select mcl_google_id, mcl_id
      from project_kmp
      union all
      select mcl_google_id, mcl_id
      from project_oivo
      into google_id, mcl_id
      do
      begin
        google_id = coalesce(:google_id, '');

        if (push_title = '' or push_body_tmp = '') then
        begin
          lang = f_get_lang_code(null, mcl_id);
          if (push_title = '') then
            push_title = f_lang(lang, 'Новый электронный документ');
          if (push_body_tmp = '') then
            push_body_tmp = f_lang(lang, 'Появился новый документ, ожидающий подписания');
        end

        if (edc_msg_kind = 0) then
          push_lifetime_days = 7; -- Т.к. PUSH дублирует СМС, храним неделю

        -- Получаем количество неподписанных эл. документов, для отображения цифры на иконке КМП
        select edl_count
        from mc_edoc_count(:mcl_id)
        into edl_count;
  
        -- Не даем подписывать закрепление банковской карты, для этого убираем eds_id
        if (edc_code = 'bankcard.moneta') then
          eds_pair = '';
        else
          eds_pair = f_json_make_pair('eds_id', eds_id, 'str')|| ',';

        push_body =
          '"priority":"high",' ||
          '"notification":{' ||
          f_json_make_pair('title', push_title, 'str') || ',' ||
          f_json_make_pair('body', push_body_tmp, 'str') || ',' ||
          f_json_make_pair('badge', edl_count, 'str') || ',' ||
          f_json_make_pair('notification_count', edl_count, 'num') || '},' ||
          '"data":{"type":"notification",' ||
          f_json_make_pair('title', push_title, 'str') || ',' ||
          f_json_make_pair('body', push_body_tmp, 'str') || ',' ||
          f_json_make_pair('badge', edl_count, 'str') || ',' ||
          eds_pair ||
          f_json_make_pair('rep_id', rep_id, 'str')|| '}';

        insert into fsrobot_msgbox (msg_type, msg_mail_to, msg_subject, msg_body, msg_lifetime_days, msg_from_un_id, msg_command_kind, msg_object_kind, msg_object_id, msg_status)
        values (2, :google_id, :push_title, :push_body, :push_lifetime_days, :un_id, 'msg_to_clients', f_objkind('mc_login', 'id'), :mcl_id, iif(:google_id = '', 1, 0))
        returning msg_id
        into :msg_id;

        -- связываем каждый пуш со строкой подписания.
        insert into edoc_msg (edm_from_eds_id, edm_from_msg_id, edm_from_clt_id, edm_hash_pin, edm_from_cda_id, edm_status)
        values (:eds_id, :msg_id, :clt_id, :pin_code, :cda_id, 0);
      end

    end
  end

  -- при edc_msg_kind = 2 создаётся связь сверху, в цикле. Нет необходимости в дополнительной связке.
  if (edc_msg_kind <> 2) then
    insert into edoc_msg (edm_from_eds_id, edm_from_msg_id, edm_from_clt_id, edm_hash_pin, edm_from_cda_id, edm_status)
    values (:eds_id, :edm_from_msg_id, :clt_id, :pin_code, :cda_id, 0);

  suspend;
end
