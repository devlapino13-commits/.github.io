CREATE PROCEDURE UXP$UXSQLFIELDS_FILL(
    JSON_DATA D_BLOB_TEXT = null)
AS
declare variable arr_count d_integer;
declare variable arr_index d_integer;
declare variable jso_item d_str_long;
declare variable usf_is_field d_bool;
declare variable usf_index d_integer;
declare variable usf_name d_str_code;
declare variable usf_origin d_str_code;
declare variable usf_source d_str_code;
declare variable usf_type d_integer;
declare variable usf_size d_integer;
begin
  if (coalesce(json_data, '') = '') then
    exit;

  delete from ux$sqlfields
  where 1 = 1;

  arr_count = f_json_blob_get_count(json_data, 'fields');
  arr_index = -1;

  while (arr_index < arr_count - 1) do
  begin
    arr_index = arr_index + 1;
    jso_item = f_json_blob_get_value(json_data, 'fields[' || arr_index || ']', '');

    if (jso_item <> '') then
    begin
      usf_is_field = f_json_str_get_value(jso_item, 'is_field', 0);
      usf_index = f_json_str_get_value(jso_item, 'index', 0);
      usf_name = f_json_str_get_value(jso_item, 'name', '');
      usf_origin = f_json_str_get_value(jso_item, 'origin', '');
      usf_source = f_json_str_get_value(jso_item, 'source', '');
      usf_type = f_json_str_get_value(jso_item, 'type', 0);
      usf_size = f_json_str_get_value(jso_item, 'size', 0);

      insert into ux$sqlfields (usf_is_field, usf_index, usf_name, usf_origin, usf_source, usf_type, usf_size)
      values (:usf_is_field, :usf_index, :usf_name, :usf_origin, :usf_source, :usf_type, :usf_size);
    end else
      exception e_message 'json array item not found';
  end
end
