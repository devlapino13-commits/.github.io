CREATE PROCEDURE CM_RU_0051_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable pr_code varchar(100);
declare variable k_kolich_lgota integer;
declare variable k_kolich_repay integer;
begin
  /*
        1. Льготный период может быть предоставлен для займов по всем продуктам, выдаваемым на
             срок 5, 6, 7, 8, 9, 10 месяцев (то есть для займов, выдаваемых на 3,4,11,12 месяцев, льготный
             период взять нельзя).
        2. Во время льготного периода клиент платит только проценты по займу, а основную сумму клиент
             выплачивает в другие месяцы займа.
        3. Льготный период можно получить только на январь и/или февраль. Клиент может выбрать оба
             месяца либо один из них, либо получить заем без льготного периода. Общий срок займа
             увеличивается на срок льготного периода.
        4. Клиент выбирает или отказывается от льготного периода до получения займа, в день подачи
             заявки на получение займа. После получения займа льготный период не предоставляется.
  */
  select k_kolich_lgota, k_kolich_repay, pr.pr_code
  from kontract kont
  inner join product pr on kont.k_from_pr_id = pr.pr_id
  where kont.k_kolich_lgota <> 0
  and kont.k_id = :k_id
  and pr.pr_code not containing 'PATENT'
  into k_kolich_lgota, k_kolich_repay, :pr_code;
  if (k_kolich_repay-k_kolich_lgota in (3, 4, 11, 12) and k_kolich_lgota > 0) then
  begin
    res_error_is_critical = 1;
    res_error_text = 'Льготный период не предоставляется для микрозаймов, выдаваемых на срок 3, 4, 11 или 12 месяцев';
    suspend;
  end
  if (k_kolich_repay < 3 or k_kolich_repay > 12) then
  begin
    res_error_is_critical = 1;
    res_error_text = 'Cрок должен быть: 5, 6, 7, 8, 9, 10 месяцев';
    suspend;
  end
  if (k_kolich_lgota not in (1,2)) then
  begin
    res_error_is_critical = 1;
    res_error_text = 'Льготный период может быть 1 или 2 месяца';
    suspend;
  end
  if ((:pr_code containing '.blago.' or :pr_code containing '.uspeh.')
       and
            exists(
            select *
            from schedule sh
            where sh.sh_from_k_id = :k_id
            and sh.sh_id > 0
            and sh.sh_os = 0
            and extract(month from sh_date) not in (1,2)
    )) then
  begin
    res_error_is_critical = 1;
    res_error_text = 'Льготный период можно получить только на январь и/или февраль';
    suspend;
  end
end
