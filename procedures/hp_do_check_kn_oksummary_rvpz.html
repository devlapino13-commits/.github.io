CREATE PROCEDURE HP_DO_CHECK_KN_OKSUMMARY_RVPZ(
    REP_DATE D_DATE)
RETURNS (
    K_ID D_ID,
    K_IS_DEVIDED D_SMALLINT,
    MIN_PERCENT D_PERCENT,
    MAX_PERCENT D_PERCENT)
AS
begin
  for select iif(k_is_devided = 1, k_base_k_id, oks_from_k_id), max(coalesce(k_is_devided, 0)),
             min(oks_final_llr_percent), max(oks_final_llr_percent)
      from kn_oksummary, kontract
      where oks_date_begin = :rep_date and
            oks_close_state in (0, 1) and
            k_id = oks_from_k_id
      group by 1
      having min(oks_final_llr_percent) <> max(oks_final_llr_percent)
      into k_id, k_is_devided, min_percent, max_percent
  do
  begin
    suspend;
  end
end
