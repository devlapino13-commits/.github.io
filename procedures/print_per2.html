CREATE PROCEDURE PRINT_PER2(
    REP_CLIENT_ID DECIMAL(18,0),
    REP_DATE DATE)
RETURNS (
    SAT_VOCATION_PAYED DECIMAL(18,2),
    SAT_VOCATION_BEGIN DECIMAL(18,2),
    SAT_VOCATION_END DECIMAL(18,2),
    SAT_VOCATION_CURR DECIMAL(18,2),
    SAT_TOTAL_NORMA DECIMAL(18,2),
    SAT_TOTAL_HOURS DECIMAL(18,2),
    SAT_TOTAL_CALC DECIMAL(18,2),
    SAT_MONTH VARCHAR(20))
AS
declare variable report_region_id integer;
declare variable rep_year integer;
declare variable rep_month integer;
declare variable rep_dd date;
begin
  rep_year = extract(year from :rep_date);
  rep_month = 1;
  while (rep_month <= 12) do
  begin
    if (rep_month < 12) then
      rep_dd = cast('01.' ||(rep_month + 1) || '.' || rep_year as date) - 1;
    else
      rep_dd = cast('31.' || rep_month || '.' || rep_year as date);
    if (rep_month = 1) then
      sat_month = rep_year || ', ' || 'Янв.';
    if (rep_month = 2) then
      sat_month = rep_year || ', ' || 'Фев.';
    if (rep_month = 3) then
      sat_month = rep_year || ', ' || 'Март';
    if (rep_month = 4) then
      sat_month = rep_year || ', ' || 'Апр.';
    if (rep_month = 5) then
      sat_month = rep_year || ', ' || 'Май';
    if (rep_month = 6) then
      sat_month = rep_year || ', ' || 'Июнь';
    if (rep_month = 7) then
      sat_month = rep_year || ', ' || 'Июль';
    if (rep_month = 8) then
      sat_month = rep_year || ', ' || 'Авг.';
    if (rep_month = 9) then
      sat_month = rep_year || ', ' || 'Сен.';
    if (rep_month = 10) then
      sat_month = rep_year || ', ' || 'Окт.';
    if (rep_month = 11) then
      sat_month = rep_year || ', ' || 'Ноя.';
    if (rep_month = 12) then
      sat_month = rep_year || ', ' || 'Дек.';
    select ps_from_reg_id
    from hr_pstatus
    where ps_from_c_id = :rep_client_id and
          ps_date_begin <= :rep_dd and
          ps_date_end >= :rep_dd and
          ps_status in (0, 1) -- включает только аннулированные и утвержденные
          rows 1
    into :report_region_id;

    for select sat_vocation_payed, sat_vocation_begin, sat_vocation_end, sat_vocation_curr, sat_total_norma,
               sat_total_hours, sat_total_calc
        from aarep_sa_tabel_one_voc(:report_region_id, :rep_client_id, :rep_year, :rep_month)
        into :sat_vocation_payed, :sat_vocation_begin, :sat_vocation_end, :sat_vocation_curr, :sat_total_norma,
             :sat_total_hours, :sat_total_calc
    do
    begin
      suspend;
    end
    rep_month = rep_month + 1;
  end
end
