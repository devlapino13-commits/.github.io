Показать DDL (CREATE PROCEDURE) 
CREATE PROCEDURE ANTIFRAUD_ERROR_LOG_CLEAN
AS
declare variable DEL_AFTER_MIN integer;
begin
    -- сколько минут храним (например, 1 день)
    del_after_min = 720;

    delete from antifraud_error_log
    where ael_log_date_time < dateadd(-:del_after_min minute to current_timestamp);
end
