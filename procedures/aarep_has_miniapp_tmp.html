CREATE PROCEDURE AAREP_HAS_MINIAPP_TMP(
    KF_ID DECIMAL(18,0),
    MA_OFFICE INTEGER)
RETURNS (
    MA_REG INTEGER,
    MA_ID INTEGER,
    MA_NUMBER INTEGER,
    MA_NO INTEGER,
    MA_DATE DATE,
    MA_KONT_NO VARCHAR(20),
    MA_STATE INTEGER,
    K_ID DECIMAL(18,0))
AS
declare variable ma_reg_prior integer;
declare variable ma_id_prior integer;
declare variable k_status integer;
declare variable ma_is_active integer;
begin
  ma_id_prior = 0;
  ma_reg = 0;
  for select ma_date_app, ma_id, ma_number, ma_no, ma_is_active
      from miniapp
      where ma_from_kf_id = :kf_id and
            ma_office_id = :ma_office and
            ma_date_oper >= (cast('NOW' as date) - 60) and
            ma_date_end = '31.12.9999'
      order by ma_id desc, ma_number desc
      into :ma_date, :ma_id, :ma_number, :ma_no, :ma_is_active
  do
  begin
    if (ma_id <> ma_id_prior) then
    begin
      if (ma_is_active = 1) then
      begin
        ma_state = null;
        k_status = null;
        ma_kont_no = null;
        k_id = null;
        select k_id, k_kontract_no, ks_state, k_status
        from kontract, kontstate
        where k_id = ks_from_k_id and
              ks_ol_is_delete = 0 and
              ks_date_end >= '01.01.2999' and
              ks_state >= 0 and
              k_from_ma_app_id = :ma_id and
              k_from_kf_id = :kf_id
        order by k_id -- Добавил Анатолий 22.01.13, из за ошибки Multiple rows in singleton select (разделенные по амнистии кредиты)
        rows 1 -- тоже добавлено
        into :k_id, :ma_kont_no, :ma_state, :k_status;
        if (ma_state is null) then
          ma_state = -1;
        if ((((k_status is null) or (k_status is not null and
            k_status <> -1))) and
            ma_state <= 0) then
          suspend;
      end
      else
      begin
        ma_id_prior = ma_id;
      end
    end
    ma_id_prior = ma_id;
  end
end
