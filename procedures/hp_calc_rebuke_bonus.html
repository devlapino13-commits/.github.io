CREATE PROCEDURE HP_CALC_REBUKE_BONUS(
    EXP_BONUS_SUM D_CURRENCY,
    C_ID D_ID,
    REP_DATE_END D_DATE,
    HIST_DATETIME D_DATETIME = null)
RETURNS (
    BONUS_FACT D_CURRENCY,
    USING_SCORES D_INTEGER,
    DV_SUM D_CURRENCY)
AS
declare variable scores_sum d_integer;
declare variable scores_can_be_used d_integer;
begin
  scores_can_be_used = 0;
  dv_sum = 0;
  using_scores = 0;

  if (:exp_bonus_sum < 200) then
  begin
    bonus_fact = :exp_bonus_sum;
    suspend;
    exit;
  end

  select scores_sum
  from hp_calc_rebuke_scores(:rep_date_end, :c_id, :hist_datetime)
  into :scores_sum;

  if (:scores_sum <= 0) then
  begin
    bonus_fact = :exp_bonus_sum;
    suspend;
    exit;
  end

  scores_can_be_used = floor(floor(:exp_bonus_sum) / 200);

  if (:scores_sum > :scores_can_be_used) then
    using_scores = :scores_can_be_used;
  else
    using_scores = :scores_sum;

  dv_sum = using_scores * 200;
  bonus_fact = :exp_bonus_sum - :dv_sum;
  suspend;
end
