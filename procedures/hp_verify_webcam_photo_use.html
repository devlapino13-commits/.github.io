CREATE PROCEDURE HP_VERIFY_WEBCAM_PHOTO_USE(
    K_ID D_ID)
RETURNS (
    WCAM_ALLOWED D_BOOL,
    SAVE_ALLOWED D_BOOL,
    HINT D_STR_255,
    REF_TIME D_SMALLINT)
AS
declare variable db_id d_integer;
declare variable reg_id d_integer;
declare variable status d_integer;
declare variable acl_status d_integer;
begin
  db_id = f_get_db_id();

  wcam_allowed = 0;
  save_allowed = 1;
  hint = '';
  ref_time = 0;

  if (db_id in (39801, 41701) and not exists(
    select *
    from kontract
      inner join product
        on pr_id = k_from_pr_id
          and pr_code containing '.ipf.'
    where k_id = :k_id)) then
  begin
    select
      reg_id,
      status,
      acl_status
    from hp_verify_webcam_photo_info(:k_id)
    into
      :reg_id,
      :status,
      :acl_status;
    if (exists(
      select *
      from region_webcam
      where rwc_reg_id = :reg_id
        and rwc_is_present = 1)) then
    begin
      if (status = 0 and acl_status < 0) then
      begin
        wcam_allowed = 1;
        save_allowed = 0;
        hint = 'верификация фото андеррайтером не прошла';
      end else
      begin
        wcam_allowed = iif(status = -1, 1, 0);
        save_allowed = iif(status = 1, 1, 0);
        ref_time = iif(status = 0, 30, 0);
        hint =
          case status
            when -1 then 'требуется видеоидентификация'
            when  0 then 'нужна верификация фото андеррайтером'
          else '' end;
      end
    end
  end
  suspend;
end
