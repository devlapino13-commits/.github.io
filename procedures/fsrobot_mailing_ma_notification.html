CREATE PROCEDURE FSROBOT_MAILING_MA_NOTIFICATION(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
declare variable GOOGLE_ID varchar(255);
declare variable DAYS integer;
declare variable STR_TMP STR_255;
declare variable KF_NAME STR_100;
declare variable MESSAGE_TYPE STR_100;
declare variable MEMO_STR STR_100;
declare variable H integer;
declare variable M integer;
declare variable MINUTES_QTY integer;
begin
-- Отправка PUSH СВЗ после регистрации заявки

-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь

-- СМС после выдачи кредита
  res_c_id = 0;
  if (user_fio is null) then
    user_fio = '';
  res_maillist = :rep_email;

  res_c_id = cast(substring(rep_params from 1 for position(';' in rep_params) - 1) as id_type);

  str_tmp = substring(rep_params from position(';' in rep_params) + 1);

  days = cast(substring(str_tmp from 1 for position(';' in str_tmp) - 1) as integer);

  str_tmp = substring(str_tmp from position(';' in str_tmp) + 1);

  kf_name = substring(str_tmp from 1 for position(';' in str_tmp) - 1);

  str_tmp = substring(str_tmp from position(';' in str_tmp) + 1);

  message_type = substring(str_tmp from 1 for position(';' in str_tmp) - 1);

  memo_str = substring(str_tmp from position(';' in str_tmp) + 1);

  select un.un_google_id
  from username un
  where un_from_c_id = :res_c_id rows 1
  into :google_id;
  if (coalesce(google_id, '') = '') then
    exit;
  res_maillist = google_id;

  if (message_type = '6_days') then
  begin
    res_subject = 'Уведомление о зависшей заявке';

    if (days = 3) then
      res_body = 'Заявка ' || kf_name || ' висит 3 дня. Примите решение не позднее 2 дней!';
    else
    if (days = 5) then
      res_body = 'Заявка ' || kf_name || ' висит 5 дней. Примите решение сегодня иначе будете оштрафованы!';
    else
    if (days >= 6) then
      res_body = 'Заявка ' || kf_name || ' висит ' || days || ' дней. Заявка аннулирована. Вам начислен штраф!';
    else
      exit;
  end
  else
  if (message_type = '24_hours') then
  begin
    if (current_date <= '01.06.2021') then
    begin
      res_subject = 'Уведомление о зависшей заявке';
      res_body = 'С момента регистрации заявки ' || kf_name || ' прошло более ' || iif(current_date < '01.08.2018', '24', '72') || 'часов и заявка не отправлена на КК. Заявка аннулирована. Вам начислен штраф!';
    end else
    begin
      res_subject = 'Уведомление о сроках работы с заявкой';
      res_body = 'С момента регистрации заявки ' || kf_name || ' прошло более 72 двух часов. Заявка не отправлена на КК и в ней не отказано, поэтому вам начислен штраф.';
    end 
  end
  else
  if (message_type = '24_hours_inform') then
  begin
    minutes_qty = memo_str;

    h = trunc(minutes_qty / 60);
    m = coalesce(mod(minutes_qty, 60), 0);

    res_subject = 'Уведомление о зависшей заявке';
    res_body = 'Примите решение по заявке ' || kf_name || ', иначе через ' || cast(h as varchar(10)) || ' часов ' || cast(m as varchar(10)) || ' минут будет штраф !';
  end

  suspend;
end
