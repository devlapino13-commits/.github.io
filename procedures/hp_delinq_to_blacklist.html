CREATE PROCEDURE HP_DELINQ_TO_BLACKLIST(
    REP_DATE_END DATE)
AS
declare variable oks_region_id integer;
declare variable k_kontract_no varchar(20);
declare variable oks_delinq_date date;
declare variable delinq_sum decimal(18,4);
declare variable c_memo varchar(255);
declare variable is_save smallint;
declare variable c_birth_date date;
declare variable c_full_address varchar(255);
declare variable c_pass_no varchar(30);
declare variable c_name_fio varchar(100);
declare variable kc_from_c_id decimal(18,0);
declare variable delinq_day integer;
declare variable oks_from_k_id decimal(18,0);
declare variable db_id integer;
declare variable delinq_sum60 integer;
begin
  db_id = f_get_db_id();
  if (current_date <= '31.12.2015' or db_id <> 81001) then
  begin
    for select oks_from_k_id, oks_region_id, cast(:rep_date_end as date) - oks_delinq_date as delinq_day,
               oks_delinq_date
        from oksummary
        where :rep_date_end between oks_date_begin and oks_date_end and
              oks_close_state = 0 and
              cast(:rep_date_end as date) - oks_delinq_date = 31 -- было and cast(:rep_date_end as date) - oks_delinq_date = 61,
--    после 25 декабря нужно поменять на and cast(:rep_date_end as date) - oks_delinq_date = 31 (равно тридцать один)
        into oks_from_k_id, oks_region_id, delinq_day, oks_delinq_date
    do
    begin
      delinq_sum = null;
      select sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc)
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :oks_from_k_id and
            ok_sh_date < :rep_date_end - 30 -- было       and ok_sh_date < :REP_DATE_END - 60
            and
            not(ok_type between 7 and 8) and
            ok_from_c_id = 0
      group by ok_sh_date
      having sum(ok_os + ok_sh_os + ok_perc + ok_sh_perc) > 100
      order by ok_sh_date
      rows 1
      into delinq_sum;
      if (delinq_sum is null) then
        delinq_sum = 0;

      if (delinq_sum > 100) then
      begin
        k_kontract_no = (select k_kontract_no
                         from kontract
                         where k_id = :oks_from_k_id);
        for select kc_from_c_id
            from knclient
            where kc_from_k_id = :oks_from_k_id and
                  kc_amount_approved > 0
            into kc_from_c_id
        do
        begin
          select c_name_fio, c_pass_no, adl_address_str, c_birth_date
          from clients, addrclient, addrlist
          where c_id = :kc_from_c_id and
                c_type = 0 and
                adc_from_c_id = c_id and
                adc_address_type = 1 and
                adc_from_adl_id = adl_id
          into c_name_fio, c_pass_no, c_full_address, c_birth_date;
          c_memo = null;
          if (exists(select 1
                     from moratory
                     where (m_from_c_id = :kc_from_c_id or
                           (m_client_fio = :c_name_fio and
                           m_client_birth_date = :c_birth_date)) and
                           m_is_active = 1 and
                           m_is_black = 2 rows 1)) then
          begin
            is_save = 0;
            c_memo = 'Контрагент ' || :c_name_fio ||  ' в стоп листе уже существовал.';
          end
          else
          begin
            is_save = 1;
            c_memo = 'У клиента ' || :c_name_fio || ' по контракту №' || :k_kontract_no || ' была просрочка более 30 дней; кол-во просроченных дней на дату ' || :rep_date_end || ' = ' || delinq_day;
          end

          if (is_save = 1) then
          begin
            insert into moratory (m_id, m_number, m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end,
                                  m_month, m_is_active, m_comment_group, m_comment_officer, m_comment_manager,
                                  m_client_fio, m_client_pass_no, m_client_address, m_client_birth_date, m_our_clients,
                                  m_from_reg_id, m_no, m_datetime, m_datetime_edit, m_user_create, m_user_edit,
                                  m_db_code, m_original_id)
            values (-255, 1, :kc_from_c_id, :oks_from_k_id, 2, current_date, current_date, 0, 1, null, :c_memo, null,
                    :c_name_fio, :c_pass_no, :c_full_address, :c_birth_date, 1, :oks_region_id, null, current_timestamp,
                    current_timestamp, 1000000000001, 1000000000001, null, null);
--              suspend;
          end
        end
      end
    end
  end
  else
  begin
    for select oks_from_k_id, oks_region_id, cast(:rep_date_end as date) - oks_delinq_date as delinq_day,
               oks_delinq_date
        from oksummary
        where :rep_date_end between oks_date_begin and oks_date_end and
              oks_close_state = 0 and
              cast(:rep_date_end as date) - oks_delinq_date between 46 and 181 -- раньше было 46-61 дней, изменен от 13.12.2016 Улан.
              and
              oks_balance > 1000
        into oks_from_k_id, oks_region_id, delinq_day, oks_delinq_date
    do
    begin
      delinq_sum = null;
      delinq_sum60 = null;
      select sum(ok_os + ok_sh_os), sum(iif(ok_sh_date < :rep_date_end - 60, ok_os + ok_sh_os, 0))
      from operkred
      where ok_ol_is_active = 1 and
            ok_from_k_id = :oks_from_k_id and
            ok_sh_date < :rep_date_end - 45 and
            not(ok_type between 7 and 8) and
            ok_from_c_id = 0 and
            ok_ol_date <= current_date having sum(ok_os + ok_sh_os) > 1000
      into delinq_sum, delinq_sum60;
      if (delinq_sum is null) then
        delinq_sum = 0;
      if (delinq_sum60 is null) then
        delinq_sum60 = 0;

      if (delinq_sum > 1000) then
      begin
        k_kontract_no = (select k_kontract_no
                         from kontract
                         where k_id = :oks_from_k_id);
        for select kc_from_c_id
            from knclient
            where kc_from_k_id = :oks_from_k_id and
                  kc_amount_approved > 0
            into kc_from_c_id
        do
        begin
          select c_name_fio, c_pass_no, adl_address_str, c_birth_date
          from clients, addrclient, addrlist
          where c_id = :kc_from_c_id and
                c_type = 0 and
                adc_from_c_id = c_id and
                adc_address_type = 1 and
                adc_from_adl_id = adl_id
          into c_name_fio, c_pass_no, c_full_address, c_birth_date;
          c_memo = null;
          if (exists(select 1
                     from moratory
                     where (m_from_c_id = :kc_from_c_id or
                           (m_client_fio = :c_name_fio and
                           m_client_birth_date = :c_birth_date))
--                  and m_is_active = 1
                           and
                           m_is_black = 2 rows 1)) then
          begin
            is_save = 0;
            c_memo = 'Контрагент ' || :c_name_fio || ' в Стоп-листе уже существовал.';
          end
          else
          begin
            is_save = 1;
            c_memo = 'У клиента ' || :c_name_fio || ' по контракту №' || :k_kontract_no || ' была просрочка более 45 дней; кол-во просроченных дней на дату ' || :rep_date_end || ' = ' || delinq_day;
          end

          if ((is_save = 1) and
              (db_id = 81001)) then
          begin
            -- Статусные клиенты не подлежат автоматическому включению в Стоп-лист по причине просрочки
            if ((select c_last_status
                 from hp_get_client_status(:kc_from_c_id, :rep_date_end)) > 0 and
                (delinq_day <= 60 or delinq_sum60 <= 1000)) then
              is_save = 0;
          end

          if ((is_save = 1) and
              (db_id = 81001)) then
          begin
            -- Приказ 97-07-20 от 28.07.2020
            -- Клиентов, допустивших просрочку по платежам, выпадающим в период с 20 марта по 31 июля 2020 г.,
            -- не включать в Стоп-лист Компании по причине данных просрочек.
            if (oks_delinq_date between '20.03.2020' and '31.07.2020')
              then is_save = 0;
          end

          if (is_save = 1) then
          begin
            insert into moratory (m_id, m_number, m_from_c_id, m_last_k_id, m_is_black, m_date_begin, m_date_end,
                                  m_month, m_is_active, m_comment_group, m_comment_officer, m_comment_manager,
                                  m_client_fio, m_client_pass_no, m_client_address, m_client_birth_date, m_our_clients,
                                  m_from_reg_id, m_no, m_datetime, m_datetime_edit, m_user_create, m_user_edit,
                                  m_db_code, m_original_id)
            values (-255, 1, :kc_from_c_id, :oks_from_k_id, 2, current_date, current_date, 0, 1, null, :c_memo, null,
                    :c_name_fio, :c_pass_no, :c_full_address, :c_birth_date, 1, :oks_region_id, null, current_timestamp,
                    current_timestamp, 1000000000001, 1000000000001, null, null);
--              suspend;
          end
        end
      end
    end

  end

end
