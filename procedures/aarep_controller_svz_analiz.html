CREATE PROCEDURE AAREP_CONTROLLER_SVZ_ANALIZ(
    REP_REGION_ID INTEGER,
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    REG_NAME STR_100,
    C_NAME_FIO STR_255,
    REGISTR_KONT_QTY INTEGER,
    DISB_KONT_QTY INTEGER,
    CONTR_DENIED INTEGER,
    CONTR_CONFIRM INTEGER,
    PERC_DEN_TO_DISB DECIMAL(15,1),
    PERC_DEN_TO_CONFIRMED DECIMAL(15,1),
    K_ID_DISB ID_TYPE,
    CONTR_CHECK INTEGER,
    REG_NAME_BASE STR_100,
    KF_NAME_LIST VARCHAR(1000),
    REP_TYPE STR_100)
AS
declare variable c_id id_type;
declare variable contr_check_result str_255;
declare variable kf_name_tmp str_100;
declare variable rd_controller_id integer;
begin
  select reg.reg_name
  from region reg
  where reg.reg_unicode = :rep_region_id
  into :reg_name_base;
  kf_name_list = '';

  select rd.rd_id
  from refdata rd
  where rd.rd_code = 'ControllerType.Controller'
  into :rd_controller_id;

  rep_type = '(Ежемесячный)';
  if (datediff(day, rep_date_begin, rep_date_end) = 6) then
    rep_type = '(Еженедельный)';

  for select distinct reg.reg_name, ps_from_c_id, c_name_fio
      from hr_pstatus, clients, joblist, jobkind, region_list(:rep_region_id, 1, -1, 1) rl, region reg
      where ps_from_c_id = c_id and
            ps_date_begin <= :rep_date_end and
            ps_date_end >= :rep_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_c_id > 1000000000003 and
            ps_from_jl_id = jl_id and
            jl_kind = jk_id and
            jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
            rl.id_list = ps_from_reg_id and
            reg.reg_unicode = rl.id_list
      order by reg.reg_name, c_name_fio
      into :reg_name, :c_id, :c_name_fio
  do
  begin
    kf_name_list = '';

    registr_kont_qty = 0; -- всего зарегистрировано контрактов (включая аннулироанные)
    disb_kont_qty = 0; -- выдано контрактов за период
    contr_denied = 0; -- контракты не подтвержденные контролером (включая аннулированные)
    contr_confirm = 0; -- подтвержден контролером (включая аннулированные)
    perc_den_to_confirmed = 0;
    perc_den_to_disb = 0;
    contr_check = 0;

    for select ks.ks_from_k_id as k_id
        from kontstate ks, region_list(:rep_region_id, 1, -1, 1) rl
        where ks.ks_kc_id = :c_id and
              ks.ks_state = 1 and
              ks.ks_ol_is_delete = 0 and
              ks.ks_date_begin between :rep_date_begin and :rep_date_end and
              ks.ks_region_id = id_list
        into :k_id_disb
    do
    begin
      disb_kont_qty = disb_kont_qty + 1;
      contr_check_result = null;

      select rd_res.rd_code
      from controller cont, refdata rd_res
      where cont.contr_from_k_id = :k_id_disb and
            cont.contr_checked_date between :rep_date_begin and :rep_date_end and
            cont.contr_is_delete = 0 and
            rd_res.rd_id = cont.contr_check_result and
            cont.contr_rd_type = :rd_controller_id -- Контролер
            rows 1
      into :contr_check_result;

      if (contr_check_result is null) then
        contr_check_result = 'ControllerCheckStatus.NotChecked';
      if (contr_check_result = 'ControllerCheckStatus.NotConfirmed') then
        contr_denied = contr_denied + 1; -- не подтвержден контролером
      if (contr_check_result = 'ControllerCheckStatus.Confirmed') then
        contr_confirm = contr_confirm + 1; -- подтвержден контролером
      if (contr_check_result in ('ControllerCheckStatus.Confirmed', 'ControllerCheckStatus.NotConfirmed')) then
        contr_check = contr_check + 1;

      kf_name_tmp = '';
      if (contr_check_result = 'ControllerCheckStatus.NotConfirmed') then
      begin
        select kal.kal_new_kf_name
        from knaktlist kal
        inner join kontract kont on kont.k_from_kf_id = kal.kal_from_kf_id
        where kont.k_id = :k_id_disb and
              kal.is_delete = 0 and
              :rep_date_end between kal.kal_date and kal.kal_date_end
        into :kf_name_tmp;

        if (kf_name_list = '') then
          kf_name_list = kf_name_tmp;
        else
          kf_name_list = kf_name_list || ', ' || kf_name_tmp;
      end
    end

    if (contr_check > 0) then
      perc_den_to_disb = cast(cast(contr_denied as decimal(15,2)) * 100 / contr_check as decimal(15,2));
    else
      perc_den_to_disb = 0;

    if (contr_denied > 0) then
    begin
      if (contr_confirm > 0) then
        perc_den_to_confirmed = cast(cast(contr_denied as decimal(15,2)) * 100 / contr_confirm as decimal(15,2));
      else
        perc_den_to_confirmed = 100;
    end
    else
      perc_den_to_confirmed = 0;

    if (kf_name_list = '') then
      kf_name_list = '_';

    suspend;
  end
end
