CREATE PROCEDURE HP_UPDATE_REVISE_COUNTER(
    C_ID D_ID,
    IS_OK D_SMALLINT,
    LANG D_STR_LANG)
RETURNS (
    RES_STATUS D_SMALLINT,
    RES_ERROR_TEXT D_STR_255)
AS
declare variable PHONE_NUMBER D_STR_20;
declare variable COUNTER D_INTEGER;
declare variable TMP D_INTEGER;
declare variable CONST_MAX_TRY D_INTEGER;
declare variable CONST_MIN_INTERVAL D_INTEGER;
declare variable MIN_AGO D_INTEGER;
begin
    -- RFCKG-4282 счётчик негативных сверок фото
    -- res_status = 1 - все ОК
    -- res_status = -1 - не окончательная ошибка
    -- res_status = -2 - окончательная ошибка
    res_error_text = '';
    res_status = 1;
    counter = 1;
    const_max_try = 3;       -- максимальное число попыток
    const_min_interval = 15; -- интервал обновления счётчика в минутах

    select first 1
        tcv.tcv_count,          -- получаем число попыток
        :const_min_interval - datediff(minute from tcv.tcv_create_datetime to current_timestamp)
    from try_counter_verify tcv
    where tcv.tcv_from_c_id = :c_id
    order by tcv.tcv_create_datetime
    into
        :counter,
        :min_ago;

    if (counter >= 3
        and is_ok = 0
        and min_ago <= 0
    ) then
    begin
        delete
        from try_counter_verify tcv
        where tcv.tcv_from_c_id = :c_id;
        
        :counter = 1;
        :min_ago = 0;

    end

    if (is_ok = 0) then   -- сверка прошла не успешно
    begin
        select first 1 coalesce(clt.clt_code, '')      -- получаем верифицированный телефонный номер
        from contdata cda
        inner join contlist clt on clt.clt_id = cda.cda_from_clt_id
        where cda.cda_from_c_id = :c_id
            and cda.cda_is_delete = 0
            and cda.cda_is_main = 1
            and cda.cda_status = 1
            and cda.cda_cont_kind = 0
        order by cda.cda_date desc
        into :phone_number;
    
        update or insert into try_counter_verify
        (
            tcv_from_c_id,
            tcv_phone,
            tcv_count
        )
        values
        (
            :c_id,
            :phone_number,
            coalesce(:counter, 1) + 1
        )
        matching (tcv_from_c_id);
    
        -- Выводим ошибку
        if (counter < const_max_try) then
        begin
            res_status = -1;
            tmp = (const_max_try - counter);
            res_error_text = f_format('Биометрическая идентификация не пройдена, осталось %1 %2', tmp, iif(tmp = 1, 'попытка', 'попытки'));
        end
        else if (counter >= const_max_try) then
        begin
            res_status = -2;
            res_error_text = f_format('Биометрическая идентификация не пройдена, Повторите попытку через %1 минут.', min_ago);
        end
    end
    else
    begin
        delete
        from try_counter_verify tcv
        where tcv.tcv_from_c_id = :c_id;
    
    end
    
    suspend;
end
