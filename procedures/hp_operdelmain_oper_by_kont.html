CREATE PROCEDURE HP_OPERDELMAIN_OPER_BY_KONT(
    IN_K_ID D_ID)
RETURNS (
    RES_OL_ID D_ID,
    RES_TEXT D_STR_LONG)
AS
declare variable tmp_ol_id d_id;
declare variable ol_date d_date;
declare variable payment_kind_str d_str_code;
declare variable reg_name d_str_100;
declare variable doc_no_str d_str_code;
begin
  for select distinct ok_from_ol_id, ok_ol_date
  from operkred
  where ok_from_k_id = :in_k_id and
        ok_ol_is_active = 1 and
        ok_type in (4, 5, 9)
  union
  select oko_from_ol_id, oko_ol_date
  from operkroff
  where oko_from_k_id = :in_k_id
  order by 2 desc, 1 desc
  into tmp_ol_id, ol_date
  do
  begin
    for select res_payment_kind, res_loan_ol_id, res_main_ol_id, res_loan_amount, res_book_amount, res_dt_account,
               res_loan_office, res_fact_office, res_kkm_id
    from hp_get_loanoper_payment_info(:tmp_ol_id, :in_k_id)
    as cursor cur
    do
    begin
      res_ol_id = cur.res_loan_ol_id;

      doc_no_str = '';
      select ol_doc_no_str
      from operlist
      where ol_id = :res_ol_id
      into doc_no_str;

      if (doc_no_str is null) then
        doc_no_str = '';

      if (doc_no_str = '') then
        doc_no_str = '-';

      payment_kind_str = '';
      payment_kind_str = trim(case cur.res_payment_kind
        when 1 then 'Касса'
        when 2 then 'Банк'
        when 3 then 'Онлайн'
        else 'Сумма'
      end);

      res_text = f_datetostr(ol_date);

      if (payment_kind_str <> '') then
        res_text = res_text || ' ' || payment_kind_str;

      select reg_code || ' ' || reg_name
      from region
      where reg_unicode = cur.res_fact_office
      into reg_name;

      res_text = res_text || ': ' ||
                 f_numtostr(cur.res_book_amount, 2) || ', док.№ ' ||
                 doc_no_str || ', ' ||
                 trim(iif(cur.res_payment_kind <> 3, reg_name || ',', '')) || ' Id: ' || res_ol_id;

      suspend;
    end
  end
end
