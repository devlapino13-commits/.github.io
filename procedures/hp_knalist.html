CREATE PROCEDURE HP_KNALIST
RETURNS (
    KAL_FROM_KA_ID DECIMAL(18,0),
    KAL_NUMBER INTEGER,
    KAL_FROM_KF_ID DECIMAL(18,0),
    KAL_FROM_K_ID DECIMAL(18,0),
    KAL_DATE DATE,
    KAL_DATE_END DATE,
    KAL_NEW_KF_OFFICER DECIMAL(18,0),
    KAL_NEW_KF_NAME VARCHAR(100),
    IS_DELETE SMALLINT,
    KAL_NEW_REG_ID INTEGER,
    KAL_NEW_NO INTEGER,
    KAL_NEW_OFFICER_CODE VARCHAR(20),
    KAL_LO_CHANGED INTEGER)
AS
declare variable disable_staff_off integer; /* Включать внештатников или нет */
declare variable lo_prior decimal(18,0);
declare variable reg_prior integer;
declare variable is_on_staff integer;
declare variable c_id decimal(18,0);
declare variable kal_new_kf_lo decimal(18,0);
begin
  select cm_disable_staff_off
  from company
  into :disable_staff_off;
  kal_lo_changed = 0;
  lo_prior = 0;
  reg_prior = 0;
  if (disable_staff_off = 0) then
  /* Внештатники отображаются */
  begin
    for select kal_from_ka_id, kal_number, kal_from_kf_id, kal_from_k_id, kal_date, kal_date_end, kal_new_kf_officer,
               kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no, kal_new_officer_code
        from knaktlist
        where is_delete = 0
        into :kal_from_ka_id, :kal_number, :kal_from_kf_id, :kal_from_k_id, :kal_date, :kal_date_end,
             :kal_new_kf_officer, :kal_new_kf_name, :is_delete, :kal_new_reg_id, :kal_new_no, :kal_new_officer_code
    do
    begin
      suspend;
    end
  end
  else
  /* Внештатники НЕ отображаются */
  begin
    for select kal_from_ka_id, kal_number, kal_from_kf_id, kal_from_k_id, kal_date, kal_date_end, kal_new_kf_officer,
               kal_new_kf_name, is_delete, kal_new_reg_id, kal_new_no, kal_new_officer_code
        from knaktlist
        where is_delete = 0
        order by kal_new_kf_officer, kal_new_reg_id
        into :kal_from_ka_id, :kal_number, :kal_from_kf_id, :kal_from_k_id, :kal_date, :kal_date_end, :kal_new_kf_lo,
             :kal_new_kf_name, :is_delete, :kal_new_reg_id, :kal_new_no, :kal_new_officer_code
    do
    begin
      if ((lo_prior <> kal_new_kf_lo) or (reg_prior <> kal_new_reg_id)) then
      begin
        kal_new_kf_officer = :kal_new_kf_lo;
        kal_lo_changed = 0;
        /* Проверяем, в штате он или нет */
        select ps_is_on_staff
        from hr_pstatus
        where ps_from_c_id = :kal_new_kf_lo and
              ps_date_begin <= :kal_date_end and
              ps_date_end >= :kal_date_end and
              ps_is_working = 1
        into :is_on_staff;
        if (is_on_staff is null) then
          is_on_staff = 0;

        if (is_on_staff = 0) then
        /* если внештате, ищем КМ'а штатного */
        begin
          c_id = null;
          select ksm_from_c_id
          from kontsignmen, kontsignreg, hr_pstatus
          where ksm_is_active = 1 and
                ksr_from_ksm_id = ksm_id and
                ksr_from_reg_id = :kal_new_reg_id and
                ksm_from_c_id = ps_from_c_id and
                ps_is_on_staff = 1 and
                ps_date_begin <= :kal_date_end and
                ps_date_end >= :kal_date_end and
                ps_is_working = 1 and
                coalesce(ksr_date_begin, '01.01.1990') <= current_date and
                coalesce(ksr_date_end, '31.12.9999') >= current_date
          order by ksm_from_c_id desc
          rows 1
          into :c_id;
          if (c_id is null) then
          /* Нет штатного КМ */
          begin
            kal_lo_changed = 1;
            select ps_from_c_id
            from hr_pstatus, joblist, jobkind
            where ps_from_reg_id = :kal_new_reg_id and
                  ps_from_jl_id = jl_id and
                  jl_kind = jk_id and
                  jk_system_code in ('JobSystem.LO_Standart', 'JobSystem.LO_Senior', 'JobSystem.LO_Partner') and
                  ps_is_on_staff = 1 and
                  ps_date_begin <= :kal_date_end and
                  ps_date_end >= :kal_date_end and
                  ps_is_working = 1
            order by ps_from_c_id
            rows 1
            into :c_id;
            if (c_id is not null) then
              kal_new_kf_officer = :c_id;
          end
          else
          begin
            kal_lo_changed = 1;
            kal_new_kf_officer = :c_id;
          end
        end
        lo_prior = :kal_new_kf_lo;
        reg_prior = :kal_new_reg_id;
      end
      suspend;
    end
  end
end
