CREATE PROCEDURE CM_KG_0182_SQL(
    MA_ID D_ID,
    CHECK_KIND D_STR_CODE,
    UN_ID D_ID)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare variable c_id id_type;
declare variable point d_integer;
declare variable mac_amount curr_type;
declare variable ma_kolich_repay d_integer;
declare variable ma_date_oper d_date;
declare variable kc_amount_approved curr_type;
declare variable pr_code varchar(100);
begin
  res_error_is_critical = 1;

  for select ma.ma_kolich_repay, ma.ma_date_oper, pr_code
      from miniapp ma
      inner join product pr on pr.pr_id = ma.ma_from_pr_id
      where ma.ma_id = :ma_id and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999' and
            pr.pr_code containing '.consultation.'
      into :ma_kolich_repay, :ma_date_oper, :pr_code
  do
  begin
    for select macl.mac_from_c_id
        from miniappclient macl
        where macl.mac_from_ma_id = :ma_id and
              macl.mac_amount > 0
        into :c_id
    do
    begin
      point = f_point(:c_id, :ma_date_oper);

      if (ma_date_oper between '15.02.2022' and '28.02.2022' and
          point >= 7) then
      begin
        res_error_text = 'У клиента балл больше 7. Выберите продукт без оплаты за консультацию';
        suspend;
      end
    end
  end
end
