CREATE PROCEDURE AAREP_REGION_USER_TMP(
    UN_ID DECIMAL(18,0))
RETURNS (
    REG_LIST INTEGER,
    IS_READ INTEGER)
AS
declare variable reg_id integer;
declare variable uniqtab_region integer;
declare variable uniqtab_username integer;
begin
  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Region')
  into :uniqtab_region;

  is_read = 1;
  for select distinct pm_subject_id
      from permissions
      where pm_subject_kind = :uniqtab_region and
            pm_object_kind = :uniqtab_username and
            pm_object_id = :un_id and
            pm_is_delete = 0 and
            (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
            (coalesce(pm_date_end, '31.12.2999') >= current_date)
      into :reg_id
  do
  begin
    for select id_list
        from region_orig_list(:reg_id, 1, -1, 1)
        into :reg_list
    do
      suspend;
  end
end
