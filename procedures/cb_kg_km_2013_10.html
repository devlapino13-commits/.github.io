CREATE PROCEDURE CB_KG_KM_2013_10(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    KM_FIO VARCHAR(100),
    KM_DATE_BEGIN DATE,
    JL_NAME VARCHAR(100),
    REG_BALANCE CURR_TYPE,
    REG_DELINQ_BALANCE CURR_TYPE,
    REG_KOL_CLI INTEGER,
    PAR8_180 DECIMAL(5,5),
    BONUS_TOTAL CURR_TYPE,
    OFFICE_NAME VARCHAR(50),
    BRANCH_NAME VARCHAR(50),
    OKRUG_NAME VARCHAR(50),
    KM_MEMO VARCHAR(100),
    REG_BALANCE_SB CURR_TYPE,
    REG_KOL_CLI_SB INTEGER,
    BONUS_KOEF CURR_TYPE,
    BONUS_SUM CURR_TYPE,
    BONUS_MEMO VARCHAR(255),
    BONUS_KM_TOTAL CURR_TYPE,
    PS_CURR_STATUS VARCHAR(100))
AS
declare variable c_id id_type;
declare variable km_reg_id_last integer;
declare variable oks_balance curr_type;
declare variable oks_delinq_date date;
declare variable date_half_month date;
declare variable km_reg_id integer;
declare variable reg_name_last varchar(50);
declare variable oks_from_k_id id_type;
declare variable sba_from_k_id id_type;
declare variable cr_rate curr_type;
declare variable curr_code varchar(3);
declare variable curr_rate curr_type;
declare variable sawt_id integer;
begin
  date_half_month = :rep_date_end - extract(day from :rep_date_end) + 15; -- 15е число
  select cr_rate
  from opercurr
  where cr_from_cu_id = '840' and
        :rep_date_end between cr_date and cr_date_end
  into cr_rate;
  for select c_name_fio, ps_from_reg_id, ps_date_begin, jl_name, c_id
      from hr_pstatus, joblist, clients, region_list(:rep_region_id, 1, 3, 1)
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_from_reg_id = id_list and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = 6 -- Выборка по кредитным менеджерам
            and
            ps_from_c_id = c_id
      into km_fio, km_reg_id, km_date_begin, jl_name, c_id
  do
  begin
    km_memo = '';
--  Пункт 3.2, в случае, если КМ переводится в другой офис после 15го числа, до данные будут прежднего офиса
    if (km_date_begin > date_half_month) then
    begin
      km_reg_id_last = null;
      select ps_from_reg_id
      from hr_pstatus, joblist
      where :date_half_month between ps_date_begin and ps_date_end and
            ps_status = 1 and
            ps_is_working = 1 and
            ps_from_jl_id = jl_id and
            jl_kind = 6 -- Выбор
            and
            ps_from_c_id = :c_id
      into km_reg_id_last;
      if (km_reg_id_last is not null) then
      begin
        reg_name_last = (select reg_code || ' ' || reg_name
                         from region
                         where reg_unicode = :km_reg_id);
        km_memo = 'КМ перевелся в офис ' || reg_name_last || ', от ' || km_date_begin;
        km_reg_id = km_reg_id_last;
      end
    end
--  Расчет баланса и просроченных кредитов
    reg_balance = 0;
    reg_delinq_balance = 0;
    reg_balance_sb = 0;
    for select oks_from_k_id, oks_balance, oks_delinq_date
        from oksummary
        where :rep_date_end between oks_date_begin and oks_date_end and
              oks_region_id = :km_reg_id and
              oks_close_state = 0
        into oks_from_k_id, oks_balance, oks_delinq_date
    do
    begin
      select k_index_curr_code, k_index_curr_rate
      from kontract
      where k_id = :oks_from_k_id
      into curr_code, curr_rate;
--    Если долларовый кредит то баланс переводит в сомовый эквивалент
      if (curr_code = '840') then
      begin
        if (cr_rate >= curr_rate) then
          oks_balance = oks_balance * cr_rate;
        else
          oks_balance = oks_balance * curr_rate;
      end
      sba_from_k_id = null;
      select sba_from_k_id
      from sb_akt, sb_list
      where sba_from_k_id = :oks_from_k_id and
            sba_status = 2 and
            :rep_date_end between sba_date_begin and sba_date_end and
            sba_is_delete = 0 and
            sba_from_sbl_id <> 0 and
            sba_from_sbl_id = sbl_id and
            sbl_type = 1
      into sba_from_k_id;
--    Если в списке СБ не нашлось папка то их включает в расчет Пара
      if (sba_from_k_id is null) then
      begin
        reg_balance = reg_balance + oks_balance;
        if (:rep_date_end - :oks_delinq_date between 8 and 180) then
        begin
          reg_delinq_balance = reg_delinq_balance + oks_balance;
        end
      end
      else
        reg_balance_sb = reg_balance_sb + oks_balance;
    end
    if (reg_balance > 0) then
      par8_180 = reg_delinq_balance / reg_balance * 100;
    else
      par8_180 = 0;

    select count(distinct iif(oks_balance > 0, kc_from_c_id, null)), count(distinct iif(oks_balance > 0 and exists(select sba_from_k_id
                                                                                                                   from sb_akt, sb_list
                                                                                                                   where sba_from_k_id = ks_from_k_id and
                                                                                                                         sba_status = 2 and
                                                                                                                         :rep_date_end between sba_date_begin and sba_date_end and
                                                                                                                         sba_is_delete = 0 and
                                                                                                                         sba_from_sbl_id <> 0 and
                                                                                                                         sba_from_sbl_id = sbl_id and
                                                                                                                         sbl_type = 1), kc_from_c_id, null))
    from knclient, kontstate, kontract, knaktlist, oksummary
    where ks_ol_is_delete = 0 and
          ks_state = 1 and
          :rep_date_end between ks_date_begin and ks_date_end and
          ks_from_k_id = k_id and
          k_from_kf_id = kal_from_kf_id and
          knaktlist.is_delete = 0 and
          :rep_date_end between kal_date and kal_date_end and
          kal_new_reg_id = :km_reg_id and
          ks_from_k_id = kc_from_k_id and
          kc_amount_approved > 0 and
          ks_from_k_id = oks_from_k_id and
          :rep_date_end between cast(oks_date_begin as date) and oks_date_end and
          oks_close_state = 0
    into reg_kol_cli, reg_kol_cli_sb;
    reg_kol_cli = reg_kol_cli - reg_kol_cli_sb;
    if (reg_kol_cli > 1500) then
    begin
      bonus_total = case
                      when par8_180 < 1 then 15000
                      when par8_180 >= 1 and par8_180 < 1.5 then 11000
                      when par8_180 >= 1.5 and par8_180 < 2 then 7000
                      when par8_180 >= 2 then 0
                    end;
    end
    else
    begin
      bonus_total = case
                      when par8_180 < 1 then 9000
                      when par8_180 >= 1 and par8_180 < 1.5 then 7500
                      when par8_180 >= 1.5 and par8_180 < 2 then 4500
                      when par8_180 >= 2 then 0
                    end;
    end
--  Вытаскивает текущий статус КМ
    sawt_id = null;
    ps_curr_status = null;
    select sawt_id, sawt_name
    from hr_voctravel, hr_orders, sa_worktype
    where hrv_from_ho_id = ho_id and
          ho_order_type = sawt_id and
          hrv_from_c_id = :c_id and
          :rep_date_end between hrv_date_begin and hrv_date_end_fact
    order by hrv_date_begin desc
    rows 1
    into sawt_id, ps_curr_status;
--  Если декретный отпуск то обнуляет бонус
    if (sawt_id in (16, 17, 18)) then
    begin
      bonus_total = 0;
    end
--  вытаскивает данные по выговорам и бонус коэффициент
    select bonus_koef, bonus_sum, bonus_memo
    from hp_get_vigovor_koef(:c_id, :rep_date_end)
    into bonus_koef, bonus_sum, bonus_memo;
    bonus_km_total = bonus_total * bonus_koef / 100 - bonus_sum;
    if (bonus_km_total < 0) then
      bonus_km_total = 0;

    select r1.reg_code || ' ' || r1.reg_name, r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name
    from region r1, region r2, region r3
    where r1.reg_unicode = :km_reg_id and
          r1.parent = r2.reg_unicode and
          r2.parent = r3.reg_unicode
    into office_name, branch_name, okrug_name;
    suspend;
  end
end
