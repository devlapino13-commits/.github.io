CREATE PROCEDURE OKT_ANKETA_ACTIONLIST_INSERT(
    ACL_ID_INPUT ID_TYPE,
    OBJECT_KIND STR_100,
    OBJECT_ID ID_TYPE,
    FIO STR_100,
    ANKETA_CODE STR_100,
    UN_ID ID_TYPE)
RETURNS (
    ACL_ID ID_TYPE,
    ERR_TEXT STR_255)
AS
declare variable rd_id integer;
declare variable acl_status integer;
declare variable anketa_code_old str_100;
declare variable rd_code str_255;
declare variable object_kind_str str_100;
declare variable cc_name str_100;
declare variable cc_from_c_id id_type;
begin
  acl_id_input = coalesce(acl_id_input, 0);
  object_kind = coalesce(object_kind, '');
  object_id = coalesce(object_id, 0);
  anketa_code = coalesce(anketa_code, '');
  un_id = coalesce(un_id, 0);
  err_text = '';

  if (anketa_code = '') then
  begin
    err_text = 'Код анкеты не указан (ANKETA_CODE)';
    suspend;
    exit;
  end

  if (acl_id_input <> 0) then -- если анкета, ранее, уже была создана
  begin
    select rd.rd_code, al.acl_object_id, f_objkind(al.acl_object_kind, 'name')
    from actionlist al
    inner join refdata rd on rd.rd_id = al.acl_from_rd_id
    where al.acl_id = :acl_id_input
    into :rd_code, :object_id, :object_kind;

    if (rd_code = 'OKTELL.Projects.Website') then -- звонки по заявкам с сайта для мигрантов
    begin
      acl_id = null;

      select al.acl_id
      from actionlist al
      where al.acl_parent_id = :acl_id_input and
            al.acl_status in (0, 1)
      into :acl_id;

      if (acl_id is not null) then
      begin
        suspend;
      end
      else
      begin
        execute procedure hp_actionlist_insert(:acl_id_input, null, :object_kind, :object_id, :anketa_code, 0, :un_id)
            returning_values :acl_id;

        suspend;
      end
    end
    else
    if (rd_code containing ('OKTELL.Projects.InCall.')) then -- входящие звонки
    begin
      acl_id = acl_id_input;

      acl_status = -777;
      rd_id = null;

      select acl_status, acl.acl_from_rd_id
      from actionlist acl
      where acl.acl_id = :acl_id_input
      into :acl_status, :rd_id;

      acl_status = coalesce(acl_status, -777);

      if (rd_id is null) then
      begin
        err_text = 'Анкета не обнаружена';
        suspend;
        exit;
      end
      else
      if (acl_status not in (0, 1)) then
      begin
        err_text = 'Анкета уже обработана';
        suspend;
        exit;
      end
      else
      begin
        anketa_code_old = '';

        select rd.rd_code
        from refdata rd
        where rd.rd_id = :rd_id
        into :anketa_code_old;

        anketa_code_old = coalesce(anketa_code_old, ''); -- предыдущий тип анкеты

        select f_objkind(oid.oid_object_kind, 'name'), oid.oid_object_id
        from okt_ivr_data oid
        where oid.oid_from_acl_id = :acl_id
        into :object_kind_str, :object_id;

        if (object_kind_str = 'Contacts') then
        begin
          select cont.cc_name, cont.cc_from_c_id
          from contacts cont
          where cont.cc_id = :object_id
          into :cc_name, :cc_from_c_id;

          cc_name = coalesce(cc_name, '');
          cc_from_c_id = coalesce(cc_from_c_id, 0);

          if ((cc_name <> fio) and
              (cc_from_c_id = 0)) then
          begin
            update contacts cont
            set cont.cc_name = :fio
            where cont.cc_id = :object_id;
          end
        end

        if (anketa_code <> anketa_code_old) then --  если тип анкеты поменялся
        begin
          select rd.rd_id
          from refdata rd
          where rd.rd_code = :anketa_code
          into :rd_id;

          update actionlist
          set acl_from_rd_id = :rd_id
          where (acl_id = :acl_id);

          suspend;
        end
        else
          suspend;
      end
    end
  end
  else -- если анкета ранее не была создана
  begin
    execute procedure hp_actionlist_insert(null, null, :object_kind, :object_id, :anketa_code, 0, :un_id)
        returning_values :acl_id;

    suspend;
  end
end
