CREATE PROCEDURE HP_DOCRECOGMAIN_LIST(
    C_ID D_ID,
    SHOW_FULL_LIST D_BOOL,
    UN_ID D_ID = null)
RETURNS (
    DRM_ID D_INTEGER,
    PK_ID D_INTEGER,
    PK_NAME D_STR_100)
AS
declare variable c_citizen d_integer;
declare variable pdt_code d_str_code;
declare variable show_cert_name_change d_bool;
declare variable c_citizen_2 d_integer;
begin
  -- Возвращает список типов для создания документа
  select c_citizen
  from clients
  where c_id = :c_id
  into c_citizen;

  select pte_citizen
  from perstemp
  where pte_from_c_id = :c_id
  into c_citizen_2;

  if (c_citizen_2 is null) then
    c_citizen_2 = c_citizen;

  if (c_id is null) then
    exception e_message 'Для выбора типа документа клиент должен быть указан.';

  show_cert_name_change = 1;
  if (f_get_db_id() = 81001) then
  begin
    -- Документ на смену ФИО создается в РФ только техподдержкой или КК

    -- Временно, un_id вычисляем до обновления sav-39 в боевой
    if (un_id is null) then
      select un_id
      from username
      where un_name = rdb$get_context('USER_SESSION', 'FINSTART_USER_LOGIN')
      into un_id;

    show_cert_name_change = (select has_access
                             from hp_permission_access_check(:un_id, 'PermActions', 0, 'FinStart.Allow.certificate_name_change'));
  end

  for select pk_id, drm_id, pk_name, pdt_code
      from hp_citizen_passkind(:c_citizen, :c_citizen_2)
      inner join passkind on res_pk_id = pk_id and
            pk_id > 0
      inner join passdoctype on pk_from_pdt_id = pdt_id
      left join docrecogmain on pk_from_drm_id = drm_id
      where drm_inactive = 0 and
            pk_is_delete = 0 and
            (pk_is_other_category = 0 or
            :show_full_list = 1)
      order by pk_order, pk_name
      into pk_id, drm_id, pk_name, pdt_code
  do
  begin
    /*if (f_get_db_id() = 81001) then
    begin
      if (pk_id = 355) then
        continue;
    end  */

    if (not(pdt_code = 'certificate_name_change' and
        show_cert_name_change = 0)) then
      suspend;

  end

end
