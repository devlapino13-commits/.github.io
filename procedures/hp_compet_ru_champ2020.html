declare variable res_date_begin date;
declare variable res_date_end date;
declare variable stage integer;
declare variable rd_id integer;
declare variable res_period_id integer;

declare variable res_lo_id decimal(18,0);

declare variable ad_disb_amount decimal(18,4) = 0;
declare variable is_otpusk varchar(20);
declare variable par1 decimal(18,4) = 0;
declare variable avg_amount decimal(18,4) = 0;
declare variable sum_ad_diff_client integer;
declare variable kol_svz integer;

declare variable in_period_disb integer;
declare variable coeff_uderj curr_type;
declare variable points_6 curr_type;
declare variable group_name_prior varchar(255);
declare variable ad_disb_loan integer;
declare variable prinyat_portfel curr_type;
declare variable svz_portfel_begin curr_type;
declare variable svz_portfel_end curr_type;

declare variable is_month_end smallint;

declare variable date_end_prv date;
declare variable rep_period integer;
declare variable rep_indicator integer;
declare variable zarabotok_begin curr_type;
declare variable zarabotok_end curr_type;

begin
  /* MRAKHMANOV
    Расчет итогов Лиги Чемпионов РФ СПФ 2020
    Stage:
    -- 0 - период (с 1 по 15 день месяца)
    -- 1 - Промежуточные итоги
    -- 2 - Итоги
    -- 3 - Финальные результаты  */

  res_text = '';
  select cpp_from_rd_id, cpp_stage, cpp_date_begin, cpp_date_end, cpp_orderno
  from competperiod
  where cpp_id = :cpp_id
  into :rd_id, :stage, :res_date_begin, :res_date_end, cpp_orderno;

  rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);

  if (rd_id is null) then
  begin
    res_text = 'Период для расчета не указан';
    suspend;
    exit;
  end

  if (stage = 0 and
      res_date_end > current_date) then
  begin
    res_text = 'Период еще не наступил. Расчет отменяется.';
    suspend;
    exit;
  end

  -- вычисляем кол-во команды
  kol_komand = 0;
  select count(*)
  from refdata
  where rd_main_code = 'StaffTeamList' and
        rd_tag = :rd_id
  into kol_komand;
  kol_komand = coalesce(kol_komand, 0);
  if (kol_komand <= 0) then
  begin
    res_text = 'Для текущего ОИ команды еще не были закреплены';
    suspend;
    exit;
  end

  if (extract(day from res_date_end) not in (15)) then
    is_month_end = 1;
  else
    is_month_end = 0;

  if (cpp_orderno = 0 and
      res_date_begin = cast('31.01.2020' as date)) then
  begin
    res_date_begin = '01.02.2020';
    res_date_end = '01.02.2020';

    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    -- Формируем в памяти данные по КР
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    -- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    for select res_field_1 --, ad_active_balance
        from hp_analitica_report('kont_lo_teamname_id', 'tree_group', 31, 360, 1, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where ad_active_balance > 0 and
              exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        into :res_team_id
    do
    begin
      if (exists(select *
                 from refdata
                 where rd_id = :res_team_id and
                       rd_is_active = 1)) then
      begin
        insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                  cpr_from_un_id, cpr_comment)
        values (:cpp_id, '', -1, :res_team_id, 0, :un_id, '')
        returning cpr_id
        into :cpr_id;
      end
    end
    suspend;
    exit;
  end

  -- Ищем итоги тура
  stage_cpp_id = null;
  select cpp_id
  from competperiod
  where cpp_from_rd_id = :rd_id and

        cpp_stage = 2 and
        cpp_status = 1 and
        cpp_date_end <= :res_date_end
  order by cpp_date_begin desc, cpp_id desc
  rows 1
  into :stage_cpp_id;

  if (stage = 0) then -- Если уровень Декады
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    -- Формируем в памяти данные по КР
    execute procedure hp_analitica_create_period(1000000, :res_date_begin, :res_date_end, 1, null, null, 30, 0,
        'olimpic')
        returning_values :res_period_id;

    if (res_period_id is null) then
    begin
      res_text = 'Не удалось расчитать данные за период';
      suspend;
      exit;
    end

    date_begin = res_date_end - extract(day from res_date_end) + 1;
    date_end_prv = date_begin - 1;
    date_begin_prv = date_end_prv - extract(day from date_end_prv) + 1;
    /*
    --  формируем данные для план выдачи
    if (is_month_end = 1) then
    begin
      rep_period = 100 * extract(year from res_date_end) + extract(month from res_date_end);
      select rd_id
      from refdata
      where rd_code in ('planindicators.disb_amount')
      into rep_indicator;

      execute procedure hp_plandata_fill(1000000, :res_date_end, :rep_indicator, :rep_period, 1, null);
    end         */

    -- Группируем данные из памяти по ID команды и ID СВЗ
    res_team_id_prior = -1;
    kol_svz = 0;
    cpr_id = null;
    for select res_field_1, res_field_2, res_field_3,
               --           sum(ad_active_client),
               --           sum(ad_active_balance),
               --           sum(ad_diff_client),
               sum(ad_delinq_balance_end1), sum(ad_delinq_balance_diff1), sum(ad_disb_amount),
               --           sum(ad_disb_loan),
               --               sum(ad_disb_new_amount)
               sum(ad_delinq_balance_diff3) ad_delinq_balance_diff3, --
               sum(ad_delinq_balance_begin3) par30_balance_begin, --
               sum(ad_delinq_balance_end3) par30_balance_end
        from hp_analitica_report('kont_lo_teamname_id, lo_id, reg_id', 'tree_group', 31, 360, 1, 360, 8, 360, 31, 360, 181, 360, 360, -1, 'standart')
        where exists(select *
                     from hr_pstatus
                     where ps_from_c_id = res_field_2 and
                           ps_date_end >= :res_date_end and
                           ps_date_begin <= :res_date_end and
                           ps_status = 1 and
                           ps_is_working = 1) and
              exists(select *
                     from refdata
                     where rd_main_code = 'StaffTeamList' and
                           rd_id = res_field_1 and
                           rd_tag = :rd_id)
        group by 1, 2, 3
        having sum(ad_active_balance) > 0.0 -- Берем тех у кого имеется портфель на конец периода
        order by 5 desc
        into :res_team_id, -- ID команды
             :res_lo_id, -- ID КСника
             :reg_id, --
             --         :ad_active_client,          -- Клиенты на конец периода
             --         :ad_active_balance,         -- Баланс на конец периода
             --         :ad_diff_client,            -- Прирост клиентов           -1-
             :ad_delinq_balance_1, -- Баланс просрочек
             :ad_delinq_balance_diff_1, -- 1.    Изменение баланса просрочек более 0 дней СВЗ (рублей) по итогам отчетной недели   v
             :ad_disb_amount, -- 2.    Сумма всех выдач за неделю (рублей)  по итогам отчетной недели
             --         :ad_disb_loan,             -- Кол-во выдач
             --             :ad_disb_new_amount -- Сумма выдачи новым клиентам
             :ad_delinq_balance_diff_3, --
             :par30_balance_begin, --
             :par30_balance_end
    do
    begin
      -- Исключаем если нет действующего
      is_otpusk = '';
      /* -- Отпуск будет расчитываться в конце месяца
      if (res_team_id<>0) then
      begin
-- Исключаем если в отпуске.
-- Проверяем статус отпуска на каждый день недели:
-- если все 5 дней был в отпуске, исключаем из списка
        otpusk_days=0;
        n=0;

        while (n<=4) do
        begin
          dd_tmp=dateadd(day, n, res_date_begin);
          if (exists(select * from hr_voctravel, hr_orders, sa_worktype
          where hrv_from_ho_id=ho_id
          and ho_order_type=sawt_id
          and hrv_from_c_id=:res_lo_id
          and sawt_base_type in (4,5) -- отпуск, декретный отпуск
          and sawt_id<>20
          and sawt_hours_real>=0
          and hrv_date_begin<=:dd_tmp
          and hrv_date_end_fact>=:dd_tmp
          and hrv_status = 1
          )) then otpusk_days=otpusk_days+1;
          n=n+1;
        end

        if (otpusk_days=5) then
        begin
          if (not(exists(
          select * from hr_voctravel, hr_orders
          where hrv_from_ho_id=ho_id
          and ho_order_type=20
          and hrv_from_c_id=:res_lo_id
          and hrv_date_begin<=:res_date_end
          and hrv_date_end_fact>=:res_date_begin
          and hrv_status = 1))) then
          begin
--            res_team_id=0;
            is_otpusk='Отпуск.';
          end
        end
      end  */

      points_1 = 0.0; --1.    Изменение баланса просрочек 1 и более дней СВЗ (рублей) по итогам отчетной недели
      points_2 = 0.0;
      points_3 = 0.0;
      points_4 = 0.0;
      points_5 = 0.0;
      points_6 = 0.0;
      points_7 = 0.0;

      --      par1 = cast(ad_delinq_balance_1 as double precision) / ad_active_balance * 100;
      cpd_comment = '';
      --      cpd_comment=cpd_comment||'Актив.кл: <b>'||round(ad_active_client)||'</b><br>';
      --      cpd_comment=cpd_comment||'PAR1: <b>'||round(par1,4)||'%</b><br>';
      cpd_comment = cpd_comment || 'Изменение PAR0: <b>' || round(ad_delinq_balance_diff_1, 2) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Баланс просрочки PAR0: <b>' || round(ad_delinq_balance_1, 2) || '%</b><br><br>';
      cpd_comment = cpd_comment || 'Баланс просрочки par30 начало: <b>' || round(par30_balance_begin, 2) || '.</b><br>';
      cpd_comment = cpd_comment || 'Баланс просрочки par30 конец: <b>' || round(par30_balance_end, 2) || '.</b><br><br>';
      --      cpd_comment = cpd_comment || 'Сумма выдач новым клиентам: <b>' || round(ad_disb_new_amount, 2) || ' р.</b><br>';
      cpd_comment = cpd_comment || 'Сумма выдач СВЗ за период: <b>' || round(ad_disb_amount, 0) || ' р.</b><br>';
      --      cpd_comment=cpd_comment||'Кол-во выдач: <b>'||ad_disb_loan||'</b><br>';
      --      cpd_comment=cpd_comment||'Прирост кл: <b>'||round(ad_diff_client)||'</b><br>';

      kol_komand_schet = kol_komand_schet + 1;
      kol_komand_schet = coalesce(kol_komand_schet, 1);

      if (is_month_end = 1) then
      begin
        --    1. Изменение баланса просрочек свз более 0 дней
        if (ad_delinq_balance_diff_1 < -60000.0 or ad_delinq_balance_1 < -50000.0) then
          points_1 = 18;
        else
        if (ad_delinq_balance_diff_1 < -20001.0) then
          points_1 = 11;
        else
        if (ad_delinq_balance_diff_1 <= 0) then
          points_1 = 2;
        else
        if (ad_delinq_balance_diff_1 <= 10000) then
          points_1 = -5;
        else
        if (ad_delinq_balance_diff_1 <= 40000) then
          points_1 = -11;
        else
          points_1 = -15;

        --    2. Сумма выдач клиентам
        points_2 = case
                     when ad_disb_amount > 2200000.0 then 7
                     when ad_disb_amount > 1600000.0 then 4
                     when ad_disb_amount > 1000000.0 then 0
                     else -5
                   end;

        points_3 = 0;
        points_4 = 0;
        points_5 = 0;

        select res_closed, res_repeat, res_koef
        from aarep_lo_retention_koef(:res_lo_id, :date_begin, :res_date_end)
        into kol_kred_closed, in_period_disb, coeff_uderj;

        if (kol_kred_closed is null) then
          kol_kred_closed = 0;
        if (in_period_disb is null) then
          in_period_disb = 0;
        cpd_comment = cpd_comment || 'Всего закрытых за период: <b>' || round(kol_kred_closed, 0) || '</b><br>';
        cpd_comment = cpd_comment || 'Повторно выданные за период: <b>' || round(in_period_disb, 0) || '</b><br>';
        cpd_comment = cpd_comment || 'коэффициент удержание клиентов: <b>' || round(coeff_uderj, 2) || '</b><br>';
        --      Удержание клиентов за отчетный месяц
        points_5 = case
                     when coeff_uderj > 50.0 then 25
                     when coeff_uderj > 40.0 then 15
                     when coeff_uderj > 35.0 then 10
                     when coeff_uderj > 30.0 then 0
                     when coeff_uderj > 25.0 then -5
                     else -10
                   end;

        if (kol_komand_schet <= 15) then
        begin
          points_7 = 5;
        end

        if (kol_komand_schet > kol_komand - 10) then
        begin
          points_7 = -2;
        end

      end

      if (res_team_id <> res_team_id_prior) then -- Если команда отличается
      begin
        -- Находим нужную строку команды за этот период
        cpr_id = null;
        select cpr_id
        from competresult
        where cpr_from_cpp_id = :cpp_id and
              cpr_object_kind = -1 and
              cpr_object_id = :res_team_id
        into :cpr_id;

        if (cpr_id is null) then -- если такой строки нет, создаем новую
        begin
          cpr_group_name = '';
          stage_cpp_id = null;
          -- Получаем название Группы где последний раз была команда
          if (stage_cpp_id is not null) then
          begin
            -- Если это первая неделя второго этапа, название лиги берем из поля комментария предыдущего расчета
            select coalesce(cpr_comment, '')
            from competresult, competperiod
            where cpr_from_cpp_id = :stage_cpp_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id rows 1
            into :cpr_group_name;
          end
          else
          begin -- Если это не первая неделя второго Этапа
            select cpr_group_name
            from competresult, competperiod
            where cpr_from_cpp_id = cpp_id and
                  cpp_from_rd_id = :rd_id and
                  cpr_object_kind = -1 and
                  cpr_object_id = :res_team_id and
                  cpr_group_name <> '' and
                  cpp_orderno < :cpp_orderno
            order by cpp_orderno desc
            rows 1
            into :cpr_group_name;
          end
          if (cpr_group_name is null) then
            cpr_group_name = '';

          insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_result_kind,
                                    cpr_from_un_id, cpr_comment)
          values (:cpp_id, :cpr_group_name, -1, :res_team_id, 0, :un_id, '')
          returning cpr_id
          into :cpr_id;
        end
        res_team_id_prior = res_team_id;
      end

      /*--      реализация план выдач
      select sum(pd_data_1)
      from temp_plandata
      where pd_from_c_id = :res_lo_id and
            pd_is_office = 2 and
            pd_data_1 > 0
      into plan_sum;
      if (plan_sum is null) then
        plan_sum = 0;   */

      --      реализация план выдач
      plan_sum = 0;
      select po_amount
      from hp_get_planoper_info(:reg_id, :res_lo_id, 'planindicators.disb_amount', :res_date_end, :rep_period)
      into plan_sum;
      if (plan_sum is null) then
        plan_sum = 0;

      --      Обьем досроных погашений
      select -sum(ok_os)
      from operkred, kontract, knaktlist
      where ok_ol_is_active = 1 and
            ok_type = 4 and
            ok_os < 0 and
            ok_ol_date between :date_begin and :res_date_end and
            ok_from_k_id = k_id and
            k_from_kf_id = kal_from_kf_id and
            ok_ol_date between kal_date and kal_date_end and
            kal_new_kf_officer = :res_lo_id and
            ok_ol_date < ok_sh_date and
            extract(month from ok_ol_date) <> extract(month from ok_sh_date)
      into dosroch_sum;
      if (dosroch_sum is null) then
        dosroch_sum = 0;

      if (plan_sum > 0) then
        ad_plan_disb = (ad_disb_amount - dosroch_sum) / plan_sum * 100;
      --else
      --  ad_plan_disb = 100;

      -- Вставляем данные по сотруднику
      --      if (is_otpusk='') then -- отпускников убираю из вычисления письмо от 25.04.2017
      --      begin
      if (is_month_end = 1) then
      begin
        super_pokazatel = case rep_period
                            when 202002 then ad_delinq_balance_diff_3
                            when 202008 then ad_delinq_balance_diff_3
                            when 202009 then ((par30_balance_end / par30_balance_begin) - 1) * 100
                          end;

        if (rep_period = 202010) then
        begin
          select count(*)
          from kontstate ks
          inner join knclient kc on ks_from_k_id = kc_from_k_id and
                kc_amount_approved > 0 and
                kc_circle = 1
          where ks.ks_ol_is_delete = 0 and
                ks_state = 1 and
                ks_date_begin between :date_begin and :res_date_end and
                ks.ks_kc_id = :res_lo_id
          into super_pokazatel;
        end
      end
      super_pokazatel = coalesce(super_pokazatel, 0);

      insert into competdetails (cpd_from_cpr_id, cpd_object_kind, cpd_object_id, cpd_points_all, cpd_points_1,

    end

    execute procedure hp_competition_recalc_order(:cpp_id);

    update competresult
    set cpr_additional_points = 3,
        cpr_comment = 'Доп. бал за 1 место: 3'
    where cpr_from_cpp_id = :cpp_id and
          cpr_order_in_group = 1 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');

    update competresult
    set cpr_additional_points = 2,
        cpr_comment = 'Доп. бал за 2 место: 2'
    where cpr_from_cpp_id = :cpp_id and
          cpr_order_in_group = 2 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');

    update competresult
    set cpr_additional_points = 1,
        cpr_comment = 'Доп. бал за 3 место: 1'
    where cpr_from_cpp_id = :cpp_id and
          cpr_order_in_group = 3 and
          coalesce(cpr_group_name, 'Не заполнен') not in ('', 'Не заполнен');

  end

  if (stage = 3) then -- если финальные итоги
  begin
    -- Удаляем существующие расчеты за этот период
    delete from competresult
    where (cpr_from_cpp_id = :cpp_id);

    for select cpr_object_kind, cpr_object_id, avg(cpr_final_points)
        from competresult, competperiod
        where cpp_from_rd_id = :rd_id and
              cpp_id = cpr_from_cpp_id and
              cpp_stage = (:stage - 1) and
              cpp_date_end between :res_date_begin and :res_date_end
        group by cpr_object_kind, cpr_object_id
        into :kind, :res_team_id, :cpr_final_points
    do
    begin
      --cpr_final_points = round(cast(cpr_final_points as double precision) / 5, 2);
      cpr_group_name = '';
      -- Получаем название Группы где последний раз была команда
      select cpr_group_name
      from competresult, competperiod
      where cpr_from_cpp_id = cpp_id and
            cpp_from_rd_id = :rd_id and
            cpr_object_kind = -1 and
            cpr_object_id = :res_team_id and
            cpr_group_name <> '' and
            cpp_orderno < :cpp_orderno
      order by cpp_orderno desc
      rows 1
      into :cpr_group_name;
      if (cpr_group_name is null) then
        cpr_group_name = '';

      insert into competresult (cpr_from_cpp_id, cpr_group_name, cpr_object_kind, cpr_object_id, cpr_calced_points,
                                cpr_result_kind, cpr_from_un_id)
      values (:cpp_id, :cpr_group_name, :kind, :res_team_id, :cpr_final_points, 0, :un_id)
      returning cpr_id
      into :cpr_id;
    end
    execute procedure hp_competition_recalc_order(:cpp_id);
  end
  suspend;
end
