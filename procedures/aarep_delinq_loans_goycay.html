CREATE PROCEDURE AAREP_DELINQ_LOANS_GOYCAY(
    REP_DATE_END D_DATE)
RETURNS (
    REG_NAME D_STR_100,
    K_KONTRACT_NO VARCHAR(20),
    CL_CASHIER_NAME D_STR_100,
    REP_DELIQ_BEGIN_DATE D_DATE,
    REP_DELIQ_DAYS_KOLICH INTEGER,
    REP_ACTIVE_BALANCE D_CURRENCY,
    REPAYMENTS D_CURRENCY,
    REP_DELIQ_OS D_CURRENCY,
    REP_DELIQ_PERC D_CURRENCY,
    KF_OFFICER_NAME D_STR_100)
AS
declare variable k_id d_id;
declare variable rep_date_begin d_date;
declare variable ok_os d_currency;
declare variable ok_perc d_currency;
declare variable ok_fine d_currency;
begin
  rep_date_begin = cast('01.' || extract(month from :rep_date_end) || '.' || extract(year from :rep_date_end) as date);
  for select ak.kf_region_name, ak.k_id, ak.k_kontract_no, ak.rep_active_balance, ak.cl_cashier_name,
             ak.rep_deliq_days_kolich, ak.rep_deliq_os, ak.rep_deliq_perc, ak.kf_officer_name, ak.rep_deliq_begin_date
      from aarep_kontract_list(1070300, null, :rep_date_end, null, null, 5) ak
      into reg_name, k_id, k_kontract_no, rep_active_balance, cl_cashier_name, rep_deliq_days_kolich, rep_deliq_os,
           rep_deliq_perc, kf_officer_name, rep_deliq_begin_date
  do
  begin
    ok_os = 0;
    ok_perc = 0;
    ok_fine = 0;
    select -sum(ok_os), -sum(ok_perc), -sum(ok_fine)
    from operkred ok
    where ok.ok_from_k_id = :k_id and
          ok.ok_type = 4 and
          ok.ok_ol_date between :rep_date_begin and :rep_date_end and
          ok.ok_ol_is_active = 1
    group by ok.ok_from_k_id
    into :ok_os, :ok_perc, :ok_fine;

    repayments = ok_os + ok_perc + ok_fine;

    suspend;
  end
end
