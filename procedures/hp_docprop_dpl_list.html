CREATE PROCEDURE HP_DOCPROP_DPL_LIST(
    PR_ID D_INTEGER,
    CLIENT_KIND D_SMALLINT,
    ONLY_LINKED D_BOOL,
    UN_ID D_ID = null)
RETURNS (
    DP_REC_ID D_INTEGER,
    DP_ID D_INTEGER,
    DP_CODE D_STR_CODE,
    DP_NAME D_STR_255,
    DP_MEMO D_STR_255,
    DP_POSITION D_INTEGER,
    DPL_SET_ID D_INTEGER,
    DPL_CHECKED D_INTEGER,
    DPL_DAYS D_INTEGER,
    DPL_GROUP_TEXT D_STR_100)
AS
declare variable has_value d_bool;
declare variable dp_is_base d_bool;
begin
  if (pr_id is null) then
    pr_id = 0;

  for select 0
  from rdb$database
  union
  select distinct dpl_set_id
  from docprloan
  where dpl_from_pr_id = :pr_id and
        dpl_client_kind = :client_kind and
        dpl_is_delete = 0 and
        dpl_set_id <> 0
  into dpl_set_id
  do
  begin
    has_value = 0;

    if (dpl_set_id = 0)
    then dpl_group_text = 'Документы';
    else dpl_group_text = 'Документы ' || dpl_set_id;

    for select dp_id, dp_code, dp_name, dp_memo, dp_position, dp_is_base
    from docprop
    order by dp_position, dp_id
    into :dp_id, :dp_code, :dp_name, :dp_memo, :dp_position, dp_is_base
    do
    begin
      dp_rec_id = dp_id * 100 + dpl_set_id;

      dpl_checked = -1;
      dpl_days = 0;

      select dpl_date_kind, dpl_date_days
      from docprloan
      where dpl_from_dp_id = :dp_id and
            dpl_from_pr_id = :pr_id and
            dpl_client_kind = :client_kind and
            dpl_set_id = :dpl_set_id and
            dpl_is_delete = 0
      into dpl_checked, dpl_days;

      if (row_count <> 0) then
        has_value = 1;

      if (row_count = 0 and dp_is_base = 1) then
      begin
        dpl_checked = 0;
        dpl_days = 0;
      end

      if (dpl_days = 0) then
        dpl_days = null;

      if (only_linked = 0 or
          dpl_checked <> -1) then
        suspend;
    end
  end

  if (has_value = 1 and only_linked = 0) then
  begin
    dpl_set_id = dpl_set_id + 1;

    for select dp_id, dp_code, dp_name, dp_memo, dp_position, dp_is_base
    from docprop
    order by dp_position, dp_id
    into :dp_id, :dp_code, :dp_name, :dp_memo, :dp_position, dp_is_base
    do
    begin
      dpl_group_text = 'Новый набор';
      dp_rec_id = dp_id * 100 + dpl_set_id;
      dpl_checked = -1;
      dpl_days = null;
      suspend;
    end
  end
end
