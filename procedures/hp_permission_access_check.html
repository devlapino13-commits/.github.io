CREATE PROCEDURE HP_PERMISSION_ACCESS_CHECK(
    UN_ID DECIMAL(18,0),
    SUBJECT_KIND VARCHAR(100),
    SUBJECT_ID DECIMAL(18,0),
    SUBJECT_NAME VARCHAR(255))
RETURNS (
    HAS_ACCESS INTEGER,
    ACCESS_KIND INTEGER)
AS
declare variable uniqtab_username integer;
declare variable uniqtab_rolename integer;
declare variable subject_kind_id integer;
begin
  if (subject_kind = '2') then
    subject_kind = 'Menulist';
  if (subject_kind = '3') then
    subject_kind = 'PermActions';

  select ukt_id
  from hp_get_uniqtable_id('Username')
  into :uniqtab_username;
  select ukt_id
  from hp_get_uniqtable_id('Rolename')
  into :uniqtab_rolename;
  select ukt_id
  from hp_get_uniqtable_id(:subject_kind)
  into :subject_kind_id;

  has_access = 0;
  access_kind = 0;
  if (subject_id is null) then
    subject_id = 0;
  if (subject_name is null) then
    subject_name = '';
  if ((subject_id = 0) and
      (subject_name = '')) then
  begin
    suspend;
    exit;
  end

  if (subject_id = 0) then
  begin
    if (subject_kind = 'Menulist') then -- пункт меню
    begin
      select ml_id
      from menulist
      where ml_code = :subject_name
      into :subject_id;
    end
    if (subject_kind = 'Permactions') then -- действия и полномочия
    begin
      select pa_id
      from permactions
      where pa_command_name = :subject_name
      into :subject_id;
    end
  end
  if (subject_id = 0) then
  begin
    suspend;
    exit;
  end

  select pm_access_kind
  from permissions
  where pm_subject_kind = :subject_kind_id and
        pm_subject_id = :subject_id and
        pm_object_kind = :uniqtab_username and
        pm_object_id = :un_id and
        pm_is_delete = 0 and
        (coalesce(pm_date_begin, '01.01.2000') <= current_date) and
        (coalesce(pm_date_end, '31.12.2999') >= current_date)
  order by pm_id desc
  rows 1
  into :access_kind;

  if (access_kind <= 0) then
    has_access = 0;
  else
    has_access = 1;
  if (access_kind = -1) then
  begin
    suspend;
    exit;
  end

  if (subject_kind = 'PermActions') then
  begin
    -- Если это проверка PermActions и главная ветка,
    -- то на уровне ролей не проверяем активность, завершаем работу
    if (exists(select *
               from permactions
               where pa_id = :subject_id and
                     pa_parent_id is null)) then
    begin
      suspend;
      exit;
    end
  end

  if (has_access = 0) then
  begin
    if (exists(select *
               from permissions un, permissions ro
               where ro.pm_subject_kind = :subject_kind_id and
                     ro.pm_subject_id = :subject_id and
                     ro.pm_is_delete = 0 and
                     (coalesce(ro.pm_date_begin, '01.01.2000') <= current_date) and
                     (coalesce(ro.pm_date_end, '31.12.2999') >= current_date) and
                     ro.pm_object_kind = :uniqtab_rolename and
                     ro.pm_object_id = un.pm_subject_id and
                     un.pm_subject_kind = :uniqtab_rolename and
                     un.pm_object_kind = :uniqtab_username and
                     un.pm_object_id = :un_id and
                     un.pm_is_delete = 0 and
                     (coalesce(un.pm_date_begin, '01.01.2000') <= current_date) and
                     (coalesce(un.pm_date_end, '31.12.2999') >= current_date))) then
    begin
      access_kind = 1;
      has_access = 1;
    end
  end
  suspend;
end
