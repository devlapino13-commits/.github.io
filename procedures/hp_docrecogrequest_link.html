CREATE PROCEDURE HP_DOCRECOGREQUEST_LINK(
    DRL_ID D_ID)
RETURNS (
    DRL_IS_LINKED D_BOOL,
    DRM_ID D_INTEGER,
    DRL_CREATE_DATETIME D_DATETIME)
AS
declare variable drr_id d_integer;
declare variable c_id d_id;
declare variable request_un_id d_id;
begin
  drl_is_linked = 0;

  for select drl_from_c_id, drl_from_drm_id, drl_create_datetime
  from docrecoglist
  where drl_id = :drl_id
  into c_id, drm_id, drl_create_datetime
  do
  begin
    for select drr_id, drr_from_un_id
    from docrecogrequest r
    where drr_from_c_id = :c_id and
          drr_from_drm_id = :drm_id and
          drr_from_drl_id is null and
          drr_status in (0, 1) and
          drr_create_datetime <= :drl_create_datetime
    order by drr_id
    rows 1
    into drr_id, request_un_id
    do
    begin
      update docrecogrequest
      set drr_from_drl_id = :drl_id
      where drr_id = :drr_id;

      drl_is_linked = 1;
    end

    if (drl_is_linked = 0) then
    begin
      for select distinct drr_id, drr_from_un_id
      from docrecogrequest, docprop, docproppk d, passkind
      where drr_from_c_id = :c_id and
            drr_from_dp_id = dp_id and
            dpk_from_dp_id = dp_id and
            dpk_from_pk_id = pk_id and
            pk_is_delete = 0 and
            dpk_is_delete = 0 and
            drr_from_drm_id is null and
            drr_from_drl_id is null and
            drr_status in (0, 1) and
            pk_from_drm_id = :drm_id and
            drr_create_datetime <= :drl_create_datetime
      order by drr_id
      rows 1
      into drr_id, request_un_id
      do
      begin
        update docrecogrequest
        set drr_from_drl_id = :drl_id
        where drr_id = :drr_id;
  
        drl_is_linked = 1;
      end
    end
  end

  /*
  if (drl_is_linked = 1) then
    Отправить PUSH пользователю МП request_un_id

  Также нужно учесть, что этот пользователь может быть отдел кадров, т.е. без МП
  */

  suspend;
end
