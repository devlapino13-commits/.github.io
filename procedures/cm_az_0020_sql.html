CREATE PROCEDURE CM_AZ_0020_SQL(
    K_ID D_ID NOT NULL,
    CHECK_KIND D_STR_CODE NOT NULL,
    UN_ID D_ID NOT NULL)
RETURNS (
    RES_INFO_TEXT D_STR_100,
    RES_ERROR_TEXT D_STR_255,
    RES_ERROR_IS_CRITICAL D_INTEGER)
AS
declare kc_amount curr_type;
declare kf_disb_date d_date;
declare variable PR_ID ID_TYPE;
declare variable C_ID ID_TYPE;
declare variable K_PERCENT PERC_TYPE;
declare variable KC_CIRCLE ID_TYPE;
declare variable C_BIRTH_DATE date;
begin
  -- Получаю дату выдачи и ID продукта
  select K_DISB_DATE, K_FROM_PR_ID, KC_FROM_C_ID, KC_AMOUNT_APPROVED, KC_CIRCLE, K.K_PERCENT
  from KONTRACT K, KNCLIENT
  where K_ID = :K_ID and
        K.K_KOLICH_CLIENT_APPROVED = 1 and
        K_ID = KC_FROM_K_ID and
        KC_AMOUNT_APPROVED > 0
  into KF_DISB_DATE, PR_ID, C_ID, KC_AMOUNT, KC_CIRCLE, K_PERCENT;
  if (C_ID is not null and
      PR_ID in (54, 55, 56, 57)) then
  begin
    if (KC_CIRCLE > 1) then
    begin
      /*  По просьбе Саида
        исключение при проверке цикла клиента не должны учитываться только групповые параллельные кредиты
        (если же в прошедших циклах клиент был в групповом кредите – то тоже должно считаться циклом). Данный вопрос уточнил у Эльшада. */
      KC_CIRCLE = 0;
      select count(*)
      from KNCLIENT, KONTSTATE, KONTRACT, PRODUCT
      where KC_FROM_C_ID = :C_ID and
            KC_AMOUNT_APPROVED > 0 and
            KC_FROM_K_ID = KS_FROM_K_ID and
            KS_OL_IS_DELETE = 0 and
            KS_STATE = 1 and
            KS_DATE_BEGIN < :KF_DISB_DATE and
            KS_FROM_K_ID = K_ID and
            K_FROM_PR_ID = PR_ID and
            not(PR_CODE containing '.GR.' and
            PR_CODE containing '.PAR.')
      into KC_CIRCLE;
      KC_CIRCLE = coalesce(KC_CIRCLE, 0) + 1; -- +1 Tekushiy
    end
    if (KC_CIRCLE = 1 and
        K_PERCENT <> 38) then
    begin
      /*Комментарий: “Для клиентов в 1-ом цикле % ставка должна быть 38%.”            
      Сообщение: “1-ci dövr müştərilər üçün illik % dərəcəsi 38% olmalıdır.” */
      res_error_text = '1-ci dövr müştərilər üçün illik % dərəcəsi 38% olmalıdır.';
      res_error_is_critical = 1;
      suspend;
    end
    if (KC_CIRCLE = 2 and
        K_PERCENT <> 37) then
    begin
      /*Комментарий: “Для клиентов в 2-ом цикле % ставка должна быть 37%.”                            
      Сообщение: “2-ci dövr müştərilər üçün illik % dərəcəsi 37% olmalıdır.”*/
      res_error_text = '2-ci dövr müştərilər üçün illik % dərəcəsi 37% olmalıdır.';
      res_error_is_critical = 1;
      suspend;
    end
    if (KC_CIRCLE = 3 and
        K_PERCENT <> 36) then
    begin
      /*Комментарий: “Для клиентов в 3-ем цикле % ставка должна быть 36%.”
        Сообщение: “3-ci dövr müştərilər üçün illik % dərəcəsi 36% olmalıdır.”*/
      res_error_text = '3-ci dövr müştərilər üçün illik % dərəcəsi 36% olmalıdır.';
      res_error_is_critical = 1;
      suspend;
    end
    if (KC_CIRCLE >= 4 and
        K_PERCENT <> 35) then
    begin
      /*Комментарий: “Для клиентов в 4-ом или большем цикле % ставка должна быть 35%.”
        Сообщение: “4-cü və daha çox dövr müştərilər üçün illik % dərəcəsi 35% olmalıdır.”*/
      res_error_text = '4-cü və daha çox dövr müştərilər üçün illik % dərəcəsi 35% olmalıdır.';
      res_error_is_critical = 1;
      suspend;
    end
  end
  else --Если клиент не определен
  begin
    res_error_text = 'Проверка невозможно, обратитесь к разработчику';
    res_error_is_critical = 2;
    suspend;
  end
end
