CREATE PROCEDURE HP_PHONE_CHANGE_APPROVAL(
    MPC_ID D_ID,
    NEW_STATUS D_SMALLINT,
    UN_ID D_ID)
AS
declare variable c_id d_id;
declare variable phone_number d_str_100;
declare variable rd_id d_integer;
begin
  -- RFCRU-4763
  if (f_enum_code('mc_phone_change', :new_status) not in ('approved', 'rejected')) then
    exception e_message('Неправильный статус');

  select mcl.mcl_from_c_id, mpc.mpc_new_phone
  from mc_phone_change mpc
  inner join mc_login mcl on mcl.mcl_id = mpc.mpc_from_mcl_id
  where mpc.mpc_id = :mpc_id
  into c_id, phone_number;

  if (f_enum_code('mc_phone_change', :new_status) = 'approved') then
  begin
    rd_id = f_get_rd_id('ContactInfo.Cell_Main');
    execute procedure hp_continfo_force_update(:c_id, :rd_id, :phone_number, 2, :un_id, current_timestamp);

    execute procedure mc_send_push('clients', :c_id, 'Верификация номера', 'Ваш номер успешно сменен');
  end
  else if (f_enum_code('mc_phone_change', :new_status) = 'rejected') then
    execute procedure mc_send_push('clients', :c_id, 'Верификация номера', 'Смена номера отклонена, повторите попытку позже');

  update mc_phone_change mpc
     set mpc.mpc_status = :new_status
  where mpc.mpc_id = :mpc_id;

end
