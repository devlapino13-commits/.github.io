CREATE PROCEDURE HP_PARSE_PKB_MINIAPP_INFO(
    BLS_ID D_ID)
RETURNS (
    IIN D_STR_20,
    LOAN_APP_DATE D_DATE,
    FROM_DATE D_DATE,
    CURRENT_DATE_VAL D_DATE,
    TOTAL_SUBMITTED_MINIAPP D_INTEGER,
    TOTAL_APPROVED_MINIAPP D_INTEGER,
    TOTAL_REJECTED_MINIAPP D_INTEGER,
    TOTAL_REVOKED_MINIAPP D_INTEGER,
    TOTAL_CREATED_MINIAPP D_INTEGER,
    PERIOD D_STR_255)
AS
declare variable JSON_TEXT D_BLOB_TEXT;
begin

  select blx_blob
  from hp_get_blobex_blob(:bls_id)
  into :json_text;

  iin = f_json_str_get_value(:json_text, 'iin', '');
  loan_app_date = cast(nullif(left(f_json_str_get_value(:json_text, 'loan_app_date', ''), 10), '')as d_date);
  from_date = cast(nullif(left(f_json_str_get_value(:json_text, 'from_date', ''), 10), '')as d_date);
  current_date_val = cast(nullif(left(f_json_str_get_value(:json_text, 'current_date', ''), 10), '') as d_date); -- дата и время предоставления ответа на запрос.
  total_submitted_miniapp = cast(nullif(f_json_str_get_value(:json_text, 'loan_apps.total', ''), '') as d_integer);   --  общее количество подданных заявок за период с даты начала периода до даты текущего запроса (то есть период current_date – (минус) from_date).

  total_approved_miniapp = cast(f_json_str_get_value(:json_text, 'loan_apps.approved', '0') as d_integer); -- количество заявок со статусом "одобрено"
  total_rejected_miniapp = cast(f_json_str_get_value(:json_text, 'loan_apps.rejected', '0') as d_integer); -- количество заявок со статусом "отказано кредитором"
  total_revoked_miniapp = cast(f_json_str_get_value(:json_text, 'loan_apps.revoked', '0') as d_integer); --  количество заявок со статусом "отозвано заемщиком"
  total_created_miniapp = cast(f_json_str_get_value(:json_text, 'loan_apps.pending', '0') as d_integer); -- количество заявок со статусом "создано"
  period = cast(current_date as d_str_20) || '--' || cast(from_date as d_str_20);

  suspend;
end
