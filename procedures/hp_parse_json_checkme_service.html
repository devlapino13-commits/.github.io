CREATE PROCEDURE HP_PARSE_JSON_CHECKME_SERVICE(
    JSON_TEXT D_BLOB_TEXT)
RETURNS (
    SERVICE_CODE D_STR_CODE,
    SERVICE_STATUS D_SMALLINT)
AS
declare variable SERVICE_DATA D_STR_8100;
declare variable PATH D_STR_255;
declare variable NUM_OF_SERVICES D_SMALLINT;
declare variable ITERATOR D_SMALLINT;
begin
    -- RFCKG-4211
    -- Всего в checkme есть 5 сервисов (kfm01, kfm02, kfm03, kfm04, pdl01) на 01.07.2025
    -- таже каждый сервис содержит куда больше информации чем service_code и service_status
    -- но на данный момент мне нужны только они

    iterator = 0;
    path = 'data';
    num_of_services = coalesce(f_json_blob_get_count(:json_text, :path), 0) - 1;

    while (num_of_services >= iterator) do
    begin
        service_data = f_json_blob_get_value(:json_text, f_format('%1[%2]', :path, :iterator), 'not found');
        service_code = f_json_str_get_value(:service_data, 'code', 'not found');
        service_status = f_json_str_get_value(:service_data, 'status', '0');

        iterator = iterator + 1;
        suspend;
    end
end
