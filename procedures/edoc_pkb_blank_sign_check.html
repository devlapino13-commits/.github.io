CREATE PROCEDURE EDOC_PKB_BLANK_SIGN_CHECK(
    EDS_ID D_ID,
    BLANK_LIST D_STR_LONG)
RETURNS (
    RES_CODE D_SMALLINT,
    RES_TEXT D_STR_255)
AS
begin
  res_code = 0;
  res_text = '';

  :blank_list = (select keys from extract_keys(:blank_list));

  -- Проверка, что все бланки, обязательные для подписания есть в списке blank_list
  if (exists(
    select *
    from edoc_sign
        left join edoc_blank on edb_from_edl_id = eds_from_edl_id
        left join edoc_frx on edf_id = edb_from_edf_id
        left join hp_split_str(:blank_list, ',') on res_str = edb_id
    where eds_id = :eds_id and
          edf_optional_sign = 0 and
          (edb_from_c_id is null or edb_from_c_id = eds_from_c_id) and
          res_str is null)) then
  begin
    res_code = 1;
    res_text = 'Не все обязательные для подписания документы были отмечены.';
  end

  suspend;
end
