CREATE PROCEDURE PRINT_FREELANCER_KONTRACT(
    PAR1 D_ID,
    ADD_PARAMS D_STR_LONG)
RETURNS (
    NAME_FIO D_STR_100,
    PASS_NO VARCHAR(30),
    PASS_DATE D_DATE,
    PASS_WHO D_STR_255,
    FULL_ADRESS D_STR_255,
    INN D_STR_20,
    ADRESS_FACT D_STR_255,
    AGREEMENT_SUM D_CURRENCY,
    DATE_SIGN D_STR_20,
    QR_CODE D_STR_LONG,
    AGREEMENT_SUM_MEMO VARCHAR(1024),
    BC_ACCOUNT_NO VARCHAR(50),
    B_ID D_STR_10,
    B_NAME D_STR_255,
    DOGOVOR_NO D_STR_20,
    REG_NAME D_STR_100,
    SD_DATE_BEGIN D_DATE,
    SD_DATE_END D_DATE,
    WORKING_DAYS D_STR_20,
    CURRENT_YEAR D_INTEGER,
    NEXT_YEAR D_INTEGER,
    PAY_FREELANCE D_CURRENCY)
AS
declare variable EDL_ID D_ID;
declare variable SIGN_STATUS D_SMALLINT;
declare variable C_ID D_ID;
declare variable DATE_SIGN_TMP D_DATE;
declare variable CATEGORY D_STR_255;
declare variable EDOC_CODE D_STR_100;
declare variable OBJ_KIND D_INTEGER;
declare variable DT_BEG D_DATE;
declare variable REG_ID D_INTEGER;
begin
  -- RFCKG-3218
  select
    adl.adl_address_str,
    sd.sd_from_c_id,
    sd.sd_memo,
    sd.sd_dogovor_no,
    sd.sd_date_begin,
    sd.sd_date_end,
    sd.sd_category,
    reg.reg_name,
    rd.rd_name,
    coalesce(sd.sd_pay_freelance, 0)
  from servicedog sd
  inner join region reg on reg.reg_unicode = sd.sd_from_reg_id
  inner join refdata rd on rd.rd_id = reg.reg_office_hours_id
  inner join addrlist adl on adl.adl_id = reg.reg_address_fact
  where sd.sd_id = :par1
  order by sd.sd_id desc
  rows 1
  into
    :adress_fact,
    :c_id,
    :agreement_sum_memo,
    :dogovor_no,
    :sd_date_begin,
    :sd_date_end,
    :category,
    :reg_name,
    :working_days,
    :pay_freelance;

  current_year = extract(year from current_date);
  next_year = current_year + 1;


  select c.c_name_fio, c.c_inn, cds.cds_doc_issued_by, cds.cds_index_doc_no, cds.cds_doc_date_begin
  from clients c
  left join clientdocs cds on c.c_id = cds.cds_from_c_id and
        cds.cds_is_delete = 0 and
        cds.cds_is_main_doc = 1 and
        current_date between cds.cds_doc_date_begin and cds.cds_doc_date_end
  where c.c_id = :c_id
  order by cds_id desc
  rows 1
  into :name_fio, :inn, :pass_who, :pass_no, :pass_date;

  select bc.bc_account_no, b.b_id, b.b_name
  from bankclient bc
  left join banks b on bc.bc_from_b_id = b.b_id
  where bc.bc_from_c_id = :c_id and
        bc.bc_is_active = 1 and
        bc.bc_is_main = 1
  rows 1
  into :bc_account_no, :b_id, :b_name;

  select adl.adl_address_str
  from addrclient adc
  inner join addrlist adl on adc.adc_from_adl_id = adl.adl_id
  where adc.adc_from_c_id = :c_id and
        adc.adc_address_type = 1
  into :full_adress;

  select sum(ssh_amount_dogovor)
  from servicesh
  where ssh_date <= current_date and
        ssh_date_regist <= current_date and
        ssh_from_sd_id = :par1
  into :agreement_sum;

  if (:category containing 'стажер') then
    edoc_code = 'probationer_kontract';
  else if (:category containing 'кочегар') then
    edoc_code = 'kochegar_kontract';
  else
    edoc_code = 'freelancer_kontract';

  select obk.obk_id
  from objkind obk
  where obk.obk_code = :edoc_code
  into :obj_kind;

  select cast(eds.eds_status_datetime as date), eds.eds_status
  from edoc_list edl
  inner join edoc_sign eds on eds.eds_from_edl_id = edl.edl_id
  where edl.edl_object_kind = :obj_kind and
        edl.edl_object_id = :par1
  order by edl_status desc, edl_id desc
  rows 1
  into :date_sign_tmp, :sign_status;

  if (sign_status is null) then
    date_sign_tmp = null;
  if (date_sign_tmp is null) then
    date_sign = '';
  else
    date_sign = f_datetostr(:date_sign_tmp);
  if (edoc_code = 'kochegar_kontract' and
         date_sign_tmp between '01.11.2024' and '30.11.2024') then
      date_sign = '01.11.2024';

  if (coalesce(add_params, '') <> '') then
  begin
    edl_id = f_json_str_get_value(:add_params, 'edl_id', null);
    qr_code = f_edoc_qr_by_edl(:edl_id, :c_id);
  end

  if (edoc_code = 'probationer_kontract' and date_sign_tmp is not null) then
  begin
    dt_beg = (
      select first 1 i_date_begin
      from interns
      where i_from_c_id = :c_id
        and :date_sign_tmp between i_date_begin and i_date_end
        and i_is_delete = 0
      order by i_id desc);
    if (dt_beg is not null) then
      date_sign = f_datetostr(dt_beg);
  end
  if (pay_freelance = 0) then
    pay_freelance = agreement_sum;
  suspend;
end
