CREATE PROCEDURE FM_CLIENT_HAS_MINIAPP(
    UN_ID ID_TYPE,
    C_ID ID_TYPE)
RETURNS (
    HAS_MINIAPP SMALLINT)
AS
begin
-- Процедура проверяет, есть ли у клиента еще не выданная заявка
-- дата выдачи по которой наступает в течение ближайших 5 дней
  has_miniapp = 0;
  if (exists(select *
             from miniapp, miniappclient
             where mac_from_ma_id = ma_id and
                   ma_date_oper between current_date and current_date + 5 and
                   ma_date_end = '31.12.9999' and
                   ma_is_active = 1 and
                   mac_amount > 0 and
                   mac_from_c_id = :c_id and
                   not exists(select *
                              from kontract k, kontstate ks
                              where ma_id = k.k_from_ma_app_id and
                                    k_id = ks_from_k_id and
                                    ks_state = 1 and
                                    current_date between ks_date_begin and ks_date_end))) then
    has_miniapp = 1;

  suspend;
end
