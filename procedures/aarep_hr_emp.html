CREATE PROCEDURE AAREP_HR_EMP(
    REP_REGION_ID INTEGER,
    REP_DATE_END DATE)
RETURNS (
    PS_DATE_BEGIN DATE,
    PS_DATE_END DATE,
    HO_ORDER_NO VARCHAR(20),
    HO_ORDER_TYPE_STR VARCHAR(100),
    REG_NAME VARCHAR(61),
    PARENT_NEME VARCHAR(61),
    OKRUG_NAME VARCHAR(61),
    C_NAME_FIO VARCHAR(100),
    C_BIRTH_DATE DATE,
    C_PASS_NO VARCHAR(30),
    JL_NAME VARCHAR(100),
    DEP_NAME VARCHAR(100),
    IS_ON_STAFF CHAR(10),
    EM_BIRTH_PLACE VARCHAR(255),
    EM_PROFESSION VARCHAR(255),
    EQ_DOC_SERIAL VARCHAR(20),
    EQ_DOC_NO VARCHAR(20),
    EQ_DATE_BEGIN DATE,
    EQ_DATE_END DATE,
    EQ_TERRITORY VARCHAR(255),
    EQ_GIVEN_BY VARCHAR(255),
    EQ_DATE_MEMO VARCHAR(100),
    ER_DATE_BEGIN DATE,
    ER_DATE_END DATE,
    ER_REGIST_BY VARCHAR(255),
    ER_ADD_SUBJECT VARCHAR(255),
    ER_ADD_REGION VARCHAR(255),
    ER_ADD_CITY VARCHAR(255),
    ER_ADD_PLACE VARCHAR(255),
    ER_ADD_INDEX VARCHAR(20),
    ER_DATE_MEMO VARCHAR(100),
    ER_REGISTER_KIND VARCHAR(255),
    EQ_COMPANY STR_255,
    EMC_DOC_NO VARCHAR(20),
    EMC_DATE_BEGIN DATE,
    EMC_DATE_END DATE,
    EMC_DATE_MEMO VARCHAR(100),
    DCT_SCOPY_DATE DATE,
    DCT_SCOPY_MEMO VARCHAR(100),
    MED_EXAM_DATE DATE,
    MED_EXAM_DATE_END DATE,
    MED_EXAM_MEMO VARCHAR(100),
    CDS_DOC_NO VARCHAR(255),
    CDS_DOC_DATE_BEGIN DATE,
    CDS_DOC_DATE_END DATE,
    CDS_DOC_ISSUED_BY VARCHAR(255),
    CDS_DOC_TERRITORY VARCHAR(255),
    CDS_DATE_MEMO VARCHAR(100),
    INFO_LAST_WORK DATE,
    C_CITIZEN_NAME VARCHAR(255),
    C_PASS_DATE DATE,
    C_PASS_DATE_EXPIRE DATE,
    PASS_DATE_MEMO VARCHAR(100),
    ADDRESS_LEGAL STR_255,
    ER_ADD_STREET STR_255,
    ER_ADD_DEPART STR_20,
    ER_ADD_BUILD STR_20,
    ER_ADD_HOME STR_20,
    C_PASS_WHO VARCHAR(255),
    C_INN VARCHAR(20),
    LAST_KVIT_PATENT_DATE DATE,
    EM_SKILL STR_255,
    MED_SPRAVKA D_DATE)
AS
declare variable ps_from_c_id decimal(18,0);
declare variable er_cancelled integer;
begin
  --2.       Добавить отчет по иностранным сотрудникам, двух видов:
  --ћ         Отчет по разрешениям на работу
  --ћ         Отчет по регистрациям
  --  select ST_USER_CLIENT_ACCESS, ST_USER_STAFF_ACCESS from GET_STORED_PROC_OPTIONS into :ST_CLIENT, :ST_STAFF;
  for select ps_date_begin, ps_date_end, ho_order_no, sawt_name, r1.reg_code || ' ' || r1.reg_name,
             r2.reg_code || ' ' || r2.reg_name, r3.reg_code || ' ' || r3.reg_name, c_name_fio, c_birth_date, c_pass_no,
             jl_name, dep_name,
             case ps_is_on_staff
               when 0 then 'внештатный'
               when 1 then 'штатный'
             end,
             em_birth_place, em_profession, ps_from_c_id, rd.rd_name, c_pass_date, c_pass_date_expire, cl.c_pass_who,
             cl.c_inn, em_skill
      from hr_pstatus
      inner join hr_orders on ps_from_ho_id = ho_id
      inner join region_list(:rep_region_id, 1, 3, 1) on ps_from_reg_id = id_list
      inner join joblist on ps_from_jl_id = jl_id
      inner join depart on ps_from_dep_id = dep_id
      inner join empmain on ps_from_c_id = em_from_c_id
      inner join sa_worktype on ho_order_type = sawt_id
      inner join region r1 on id_list = r1.reg_unicode
      left join region r2 on r2.reg_unicode = r1.parent
      left join region r3 on r3.reg_unicode = r2.parent
      inner join clients cl on ps_from_c_id = c_id
      inner join refdata rd on cl.c_citizen = rd.rd_id
      where :rep_date_end between ps_date_begin and ps_date_end and
            ps_is_working = 1 and
            em_id = 1 and
            rd.rd_main_code = 'citizen' and
            rd.rd_code <> 'Citizen.RU'

      into :ps_date_begin, :ps_date_end, :ho_order_no, :ho_order_type_str, :reg_name, :parent_neme, :okrug_name,
           :c_name_fio, :c_birth_date, :c_pass_no, :jl_name, :dep_name, :is_on_staff, :em_birth_place, :em_profession,
           :ps_from_c_id, :c_citizen_name, c_pass_date, c_pass_date_expire, c_pass_who, c_inn, em_skill
  do
  begin
    --Дата последнего найма. Письмо от Патоновой Аиды. Добавил Музаффар>>>>>>>>>>>>>
    info_last_work = null;
    select max(ps_date_begin)
    from hr_pstatus
    inner join hr_orders on ps_from_ho_id = ho_id

    where ps_status = 1 and
          ps_from_c_id = :ps_from_c_id and
          ho_order_type = 1 and
          ho_status = 1
    into :info_last_work;

    address_legal = '';
    select adl.adl_address_str
    from addrclient adcl
    inner join addrlist adl on adl.adl_id = adcl.adc_from_adl_id
    where adcl.adc_from_c_id = :ps_from_c_id and
          adcl.adc_address_type = 1
    into :address_legal;

    -->> Вытаскивает последнее разрешение на работу
    eq_date_memo = '';
    eq_date_end = null;
    select eq_doc_serial, eq_doc_no, eq_date_begin, eq_date_end, eq_territory, eq_given_by, eq_company
    from empquota
    where eq_from_c_id = :ps_from_c_id and
          :rep_date_end between eq_date_begin and eq_date_end
    order by eq_date_end desc
    rows 1
    into :eq_doc_serial, :eq_doc_no, :eq_date_begin, :eq_date_end, :eq_territory, :eq_given_by, :eq_company;

    if (coalesce(:eq_date_end, :rep_date_end - 1) <= :rep_date_end) then
    begin
      eq_date_memo = 'Срок действия на разрешение истек, или не найден актуальный документ!';
    end
    else
    if (cast((:eq_date_end - :rep_date_end) as integer) < 7) then
      eq_date_memo = 'До окончания разрешения осталось меньше недели!';

    if (:eq_date_end < :eq_date_begin) then
      eq_date_memo = 'Срок указан неверно. Дата конца меньше чем дата выдачи документа!';
    --<< Вытаскивает последнее разрешение на работу

    -->> Вытаскивает последний миграционный учет
    er_date_memo = '';
    er_date_begin = null;
    er_date_end = null;
    er_regist_by = null;
    er_add_subject = null;
    er_add_region = null;
    er_add_city = null;
    er_add_place = null;
    er_add_street = null;
    er_add_depart = null;
    er_add_build = null;
    er_add_home = null;
    er_add_index = null;
    er_cancelled = null;
    er_register_kind = null;
    select er_date_begin, er_date_end, er_regist_by, er_add_subject, er_add_region, er_add_city, er_add_place,
           er.er_add_street, er.er_add_depart, er_add_build, er_add_home, er_add_index, er_cancelled, er_register_kind
    from empregist er
    where er_from_c_id = :ps_from_c_id and
          :rep_date_end between er_date_begin and er_date_end
    order by er_cancelled, er_date_end desc
    rows 1
    into :er_date_begin, :er_date_end, :er_regist_by, :er_add_subject, :er_add_region, :er_add_city, :er_add_place,
         er_add_street, er_add_depart, er_add_build, er_add_home, :er_add_index, :er_cancelled, er_register_kind;

    if (coalesce(:er_date_end, :rep_date_end - 1) <= :rep_date_end) then
    begin
      er_date_memo = 'Срок действия миграции истек!';
    end
    else
    if (cast((:er_date_end - :rep_date_end) as integer) < 7) then
      er_date_memo = 'До окончания срока миграции осталось меньше недели!';

    if (:er_date_end < :er_date_begin) then
      er_date_memo = 'Срок указан неверно. Дата конца меньше чем дата выдачи документа!';

    if (er_cancelled = 1) then
    begin
      if (er_date_memo = '') then
        er_date_memo = 'Миграц.регистрация аннулирована';
      else
        er_date_memo = 'Аннулирована, ' || er_date_memo;
    end

    --  Данные миграционной карты
    emc_doc_no = null;
    emc_date_begin = null;
    emc_date_end = null;
    select emc_doc_no, emc_date_begin, emc_date_end
    from empmigcard
    where emc_from_c_id = :ps_from_c_id and
          :rep_date_end between emc_date_begin and emc_date_end
    order by emc_date_begin desc
    rows 1
    into :emc_doc_no, :emc_date_begin, :emc_date_end;

    emc_date_memo = '';
    if (emc_date_end < :rep_date_end) then
      emc_date_memo = 'Срок миграционной карты истек!';
    if (emc_date_end is null) then
    begin
      emc_date_memo = 'Миграционная карта не заполнена!';
    end
    else
    begin
      if ((emc_doc_no is null) or (trim(emc_doc_no) = '')) then
        emc_date_memo = emc_date_memo || ' Номер документа не заполнена';
    end

    if (cast((:emc_date_end - :rep_date_end) as integer) < 7) then
      er_date_memo = 'До окончания срока миграционной карты осталось меньше недели!';

    if (:emc_date_end < :emc_date_begin) then
      er_date_memo = 'Cрок миграционной карты указан неверно!';

    dct_scopy_date = null;
    dct_scopy_memo = null;
    med_exam_date = null;
    med_exam_date_end = null;
    med_exam_memo = null;
    med_spravka = null;

    select cds_doc_date_begin  -- Дактилоскопия
    from clientdocs
    where cds_from_c_id = :ps_from_c_id and
          cds_from_pk_id = 346
    rows 1
    into dct_scopy_date;

    select cds_doc_date_begin, cds_doc_date_end -- Медицинский осмотр
    from clientdocs
    where cds_from_c_id = :ps_from_c_id and
          cds_from_pk_id = 347 and
          cds_is_delete = 0
    order by cds_doc_date_begin desc
    rows 1
    into med_exam_date, med_exam_date_end;

    select cds_doc_date_begin -- Медицинский осмотр   
    from clientdocs
    where cds_from_c_id = :ps_from_c_id and
          cds_from_pk_id = 73 and
          cds_is_delete = 0
    order by cds_doc_date_begin desc
    rows 1
    into med_spravka;

    if (emc_date_begin >= '01.01.2022' and (eq_company like '%Договор о ЕАЭС%' or eq_company like '%патент%')) then
    begin    
        if (dct_scopy_date is null and rep_date_end - emc_date_begin between 23 and 30) then
            dct_scopy_memo = 'Меньше недели!';
        if (med_exam_date is null and rep_date_end - emc_date_begin between 23 and 30) then
            med_exam_memo = 'Меньше недели!';
        if (dct_scopy_date is null and rep_date_end - emc_date_begin > 30) then
            dct_scopy_memo = 'Срок истек!';
        if (med_exam_date is null and rep_date_end - emc_date_begin > 30) then
            med_exam_memo = 'Срок истек!';
    end

    /*
        Автор: Музаффар Рахманов
    
        Дата: 16.01.2015 7:08:35
    
        Назначение: Данные по добровольному медицинскому страхованию(ДМС)
      */
    --Начало-->>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>
    cds_doc_no = null;
    cds_doc_date_begin = null;
    cds_doc_date_end = null;
    cds_doc_issued_by = null;
    cds_doc_territory = null;
    select cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_doc_issued_by, cds_doc_territory
    from clientdocs
    where cds_is_delete = 0 and
          cds_from_pk_id = 59 and
          cds_from_c_id = :ps_from_c_id and
          :rep_date_end between coalesce(cds_doc_date_begin, :rep_date_end - 1) and coalesce(cds_doc_date_end, :rep_date_end)
    order by cds_doc_date_end desc
    rows 1
    into cds_doc_no, cds_doc_date_begin, cds_doc_date_end, cds_doc_issued_by, cds_doc_territory;

    cds_date_memo = '';

    if (coalesce(cds_doc_date_end, rep_date_end) <= :rep_date_end) then
    begin
      cds_date_memo = 'Внимание! Истек срок ДМС!';
    end

    if (cast((:cds_doc_date_end - :rep_date_end) as integer) between 1 and 7) then
    begin
      cds_date_memo = 'До окончания срока действия осталось меньше недели!';
    end

    if (:cds_doc_date_end < :cds_doc_date_begin) then
    begin
      cds_date_memo = 'Срок указан неверно. Дата конца меньше чем дата выдачи документа!';
    end
    --Конец--<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<<

    --Срок действия паспорта. Добавил Музаффар 26.10.15
    pass_date_memo = null;

    if (c_pass_date > c_pass_date_expire or (c_pass_date > :rep_date_end)) then
      pass_date_memo = 'Дата выдачи паспорта неверно!';

    if (c_pass_date_expire < :rep_date_end) then
      pass_date_memo = 'Срок паспорта истек!';

    if (c_pass_date_expire is null) then
      pass_date_memo = 'Срок паспорта не заполнена!';

    if (:c_pass_date_expire - :rep_date_end between 1 and 7) then
      pass_date_memo = 'До окончания срока паспорта осталось меньше недели!';

    last_kvit_patent_date = null;
    select max(eq.eq_last_invoice_date)
    from empquota eq
    where eq.eq_from_c_id = :ps_from_c_id
    into :last_kvit_patent_date;

    suspend;
  end
end
