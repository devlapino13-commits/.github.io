CREATE PROCEDURE HP_DIVIDE_KONTCALCPERC
RETURNS (
    KCP_PERIOD INTEGER,
    KCP_FROM_K_ID ID_TYPE,
    KOL_CLI INTEGER,
    K_AMOUNT CURR_TYPE)
AS
declare variable k_id_prior id_type;
declare variable kcp_from_reg_id integer;
declare variable kcp_perc_saldo decimal(18,4);
declare variable kcp_transfer_reg_id integer;
declare variable kcp_perc_calced decimal(18,4);
declare variable kcp_perc_payed decimal(18,4);
declare variable kcp_perc_writeoff decimal(18,4);
declare variable kcp_perc_begin_saldo decimal(18,4);
declare variable kcp_perc_transfer_payed decimal(18,4);
declare variable kcp_perc_transfer_writeoff decimal(18,4);
declare variable kcp_kommis_calced decimal(18,4);
declare variable kcp_kommis_payed decimal(18,4);
declare variable kcp_kommis_writeoff decimal(18,4);
declare variable kcp_kommis_saldo decimal(18,4);
declare variable kcp_kommis_begin_saldo decimal(18,4);
declare variable kcp_kommis_transfer_payed decimal(18,4);
declare variable kcp_kommis_transfer_writeoff decimal(18,4);
declare variable kcp_perc_account varchar(20);
declare variable kcp_kommis_account varchar(20);
declare variable kcp_loan_is_closed integer;
declare variable kcp_out_perc_begin_saldo decimal(18,4);
declare variable kcp_out_perc_calced decimal(18,4);
declare variable kcp_out_perc_saldo decimal(18,4);
declare variable kcp_from_c_id decimal(18,0);
declare variable kc_from_c_id id_type;
declare variable kc_amount_approved curr_type;
declare variable dolya double precision;
declare variable k integer;
declare variable kcp_perc_saldo_cli curr_type;
declare variable kcp_perc_calced_cli curr_type;
declare variable kcp_perc_payed_cli curr_type;
declare variable kcp_perc_writeoff_cli curr_type;
declare variable kcp_perc_begin_saldo_cli curr_type;
declare variable kcp_perc_transfer_payed_cli curr_type;
declare variable kcp_perc_transfer_writeoff_cli curr_type;
declare variable kcp_kommis_calced_cli curr_type;
declare variable kcp_kommis_payed_cli curr_type;
declare variable kcp_kommis_writeoff_cli curr_type;
declare variable kcp_kommis_saldo_cli curr_type;
declare variable kcp_kommis_begin_saldo_cli curr_type;
declare variable kcp_kommis_transfer_payed_cli curr_type;
declare variable kcp_kommis_transfer_woff_cli curr_type;
declare variable kcp_out_perc_begin_saldo_cli curr_type;
declare variable kcp_out_perc_calced_cli curr_type;
declare variable kcp_out_perc_saldo_cli curr_type;
declare variable kcp_perc_saldo_temp curr_type;
declare variable kcp_perc_calced_temp curr_type;
declare variable kcp_perc_payed_temp curr_type;
declare variable kcp_perc_writeoff_temp curr_type;
declare variable kcp_perc_begin_saldo_temp curr_type;
declare variable kcp_perc_transfer_payed_temp curr_type;
declare variable kcp_perc_transfer_writeoff_temp curr_type;
declare variable kcp_kommis_calced_temp curr_type;
declare variable kcp_kommis_payed_temp curr_type;
declare variable kcp_kommis_writeoff_temp curr_type;
declare variable kcp_kommis_saldo_temp curr_type;
declare variable kcp_kommis_begin_saldo_temp curr_type;
declare variable kcp_kommis_transfer_payed_temp curr_type;
declare variable kcp_kommis_transfer_woff_temp curr_type;
declare variable kcp_out_perc_begin_saldo_temp curr_type;
declare variable kcp_out_perc_calced_temp curr_type;
declare variable kcp_out_perc_saldo_temp curr_type;
declare variable db_id integer;
declare variable round_factor integer;
begin

  db_id = f_get_db_id();
  if (db_id = 94401) then
    round_factor = 2;
  else
    round_factor = 0;

  k_id_prior = 0;
  for select kcp_period, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo, kcp_transfer_reg_id, kcp_perc_calced,
             kcp_perc_payed, kcp_perc_writeoff, kcp_perc_begin_saldo, kcp_perc_transfer_payed,
             kcp_perc_transfer_writeoff, kcp_kommis_calced, kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo,
             kcp_kommis_begin_saldo, kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff, kcp_perc_account,
             kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo, kcp_out_perc_calced, kcp_out_perc_saldo
      from kontcalcperc
      where kcp_from_c_id = 0
--  and kcp_from_k_id = 101010000000700202
      order by kcp_from_k_id, kcp_period
      into :kcp_period, :kcp_from_k_id, :kcp_from_reg_id, :kcp_perc_saldo, :kcp_transfer_reg_id, :kcp_perc_calced,
           :kcp_perc_payed, :kcp_perc_writeoff, :kcp_perc_begin_saldo, :kcp_perc_transfer_payed,
           :kcp_perc_transfer_writeoff, :kcp_kommis_calced, :kcp_kommis_payed, :kcp_kommis_writeoff, :kcp_kommis_saldo,
           :kcp_kommis_begin_saldo, :kcp_kommis_transfer_payed, :kcp_kommis_transfer_writeoff, :kcp_perc_account,
           :kcp_kommis_account, :kcp_loan_is_closed, :kcp_out_perc_begin_saldo, :kcp_out_perc_calced,
           :kcp_out_perc_saldo
  do
  begin
    if (k_id_prior <> kcp_from_k_id) then
    begin
      k_id_prior = kcp_from_k_id;
--  вычисления фак. кол-во клиентов
      kol_cli = null;
      select count(*), max(kc_from_c_id), sum(kc_amount_approved)
      from knclient
      where kc_from_k_id = :kcp_from_k_id and
            kc_amount_approved > 0
      into kol_cli, kc_from_c_id, -- для случья если в контракте только один клиент и для определение C_ID повторно не обращаться к таблице knclient
           k_amount;
      if (kol_cli is null) then
        kol_cli = 0;
    end

    kcp_perc_saldo_temp = kcp_perc_saldo;
    kcp_perc_calced_temp = kcp_perc_calced;
    kcp_perc_payed_temp = kcp_perc_payed;
    kcp_perc_writeoff_temp = kcp_perc_writeoff;
    kcp_perc_begin_saldo_temp = kcp_perc_begin_saldo;
    kcp_perc_transfer_payed_temp = kcp_perc_transfer_payed;
    kcp_perc_transfer_writeoff_temp = kcp_perc_transfer_writeoff;
    kcp_kommis_calced_temp = kcp_kommis_calced;
    kcp_kommis_payed_temp = kcp_kommis_payed;
    kcp_kommis_writeoff_temp = kcp_kommis_writeoff;
    kcp_kommis_saldo_temp = kcp_kommis_saldo;
    kcp_kommis_begin_saldo_temp = kcp_kommis_begin_saldo;
    kcp_kommis_transfer_payed_temp = kcp_kommis_transfer_payed;
    kcp_kommis_transfer_woff_temp = kcp_kommis_transfer_writeoff;
    kcp_out_perc_begin_saldo_temp = kcp_out_perc_begin_saldo;
    kcp_out_perc_calced_temp = kcp_out_perc_calced;
    kcp_out_perc_saldo_temp = kcp_out_perc_saldo;

    if (kol_cli >= 2) then
    begin
      k = 0;
      for select kc_from_c_id, kc_amount_approved
          from knclient
          where kc_from_k_id = :kcp_from_k_id and
                kc_amount_approved > 0
          into kc_from_c_id, kc_amount_approved
      do
      begin
        dolya = cast(kc_amount_approved as double precision) / k_amount;
        k = k + 1;
        if (k < kol_cli) then
        begin
          kcp_perc_saldo_cli = round(dolya * kcp_perc_saldo, round_factor);
          kcp_perc_calced_cli = round(dolya * kcp_perc_calced, round_factor);
          kcp_perc_payed_cli = round(dolya * kcp_perc_payed, round_factor);
          kcp_perc_writeoff_cli = round(dolya * kcp_perc_writeoff, round_factor);
          kcp_perc_begin_saldo_cli = round(dolya * kcp_perc_begin_saldo, round_factor);
          kcp_perc_transfer_payed_cli = round(dolya * kcp_perc_transfer_payed, round_factor);
          kcp_perc_transfer_writeoff_cli = round(dolya * kcp_perc_transfer_writeoff, round_factor);
          kcp_kommis_calced_cli = round(dolya * kcp_kommis_calced, round_factor);
          kcp_kommis_payed_cli = round(dolya * kcp_kommis_payed, round_factor);
          kcp_kommis_writeoff_cli = round(dolya * kcp_kommis_writeoff, round_factor);
          kcp_kommis_saldo_cli = round(dolya * kcp_kommis_saldo, round_factor);
          kcp_kommis_begin_saldo_cli = round(dolya * kcp_kommis_begin_saldo, round_factor);
          kcp_kommis_transfer_payed_cli = round(dolya * kcp_kommis_transfer_payed, round_factor);
          kcp_kommis_transfer_woff_cli = round(dolya * kcp_kommis_transfer_writeoff, round_factor);
          kcp_out_perc_begin_saldo_cli = round(dolya * kcp_out_perc_begin_saldo, round_factor);
          kcp_out_perc_calced_cli = round(dolya * kcp_out_perc_calced, round_factor);
          kcp_out_perc_saldo_cli = round(dolya * kcp_out_perc_saldo, round_factor);

          kcp_perc_saldo_temp = kcp_perc_saldo_temp - kcp_perc_saldo_cli;
          kcp_perc_calced_temp = kcp_perc_calced_temp - kcp_perc_calced_cli;
          kcp_perc_payed_temp = kcp_perc_payed_temp - kcp_perc_payed_cli;
          kcp_perc_writeoff_temp = kcp_perc_writeoff_temp - kcp_perc_writeoff_cli;
          kcp_perc_begin_saldo_temp = kcp_perc_begin_saldo_temp - kcp_perc_begin_saldo_cli;
          kcp_perc_transfer_payed_temp = kcp_perc_transfer_payed_temp - kcp_perc_transfer_payed_cli;
          kcp_perc_transfer_writeoff_temp = kcp_perc_transfer_writeoff_temp - kcp_perc_transfer_writeoff_cli;
          kcp_kommis_calced_temp = kcp_kommis_calced_temp - kcp_kommis_calced_cli;
          kcp_kommis_payed_temp = kcp_kommis_payed_temp - kcp_kommis_payed_cli;
          kcp_kommis_writeoff_temp = kcp_kommis_writeoff_temp - kcp_kommis_writeoff_cli;
          kcp_kommis_saldo_temp = kcp_kommis_saldo_temp - kcp_kommis_saldo_cli;
          kcp_kommis_begin_saldo_temp = kcp_kommis_begin_saldo_temp - kcp_kommis_begin_saldo_cli;
          kcp_kommis_transfer_payed_temp = kcp_kommis_transfer_payed_temp - kcp_kommis_transfer_payed_cli;
          kcp_kommis_transfer_woff_temp = kcp_kommis_transfer_woff_temp - kcp_kommis_transfer_woff_cli;
          kcp_out_perc_begin_saldo_temp = kcp_out_perc_begin_saldo_temp - kcp_out_perc_begin_saldo_cli;
          kcp_out_perc_calced_temp = kcp_out_perc_calced_temp - kcp_out_perc_calced_cli;
          kcp_out_perc_saldo_temp = kcp_out_perc_saldo_temp - kcp_out_perc_saldo_cli;
        end
        else
        begin
          kcp_perc_saldo_cli = kcp_perc_saldo_temp;
          kcp_perc_calced_cli = kcp_perc_calced_temp;
          kcp_perc_payed_cli = kcp_perc_payed_temp;
          kcp_perc_writeoff_cli = kcp_perc_writeoff_temp;
          kcp_perc_begin_saldo_cli = kcp_perc_begin_saldo_temp;
          kcp_perc_transfer_payed_cli = kcp_perc_transfer_payed_temp;
          kcp_perc_transfer_writeoff_cli = kcp_perc_transfer_writeoff_temp;
          kcp_kommis_calced_cli = kcp_kommis_calced_temp;
          kcp_kommis_payed_cli = kcp_kommis_payed_temp;
          kcp_kommis_writeoff_cli = kcp_kommis_writeoff_temp;
          kcp_kommis_saldo_cli = kcp_kommis_saldo_temp;
          kcp_kommis_begin_saldo_cli = kcp_kommis_begin_saldo_temp;
          kcp_kommis_transfer_payed_cli = kcp_kommis_transfer_payed_temp;
          kcp_kommis_transfer_woff_cli = kcp_kommis_transfer_woff_temp;
          kcp_out_perc_begin_saldo_cli = kcp_out_perc_begin_saldo_temp;
          kcp_out_perc_calced_cli = kcp_out_perc_calced_temp;
          kcp_out_perc_saldo_cli = kcp_out_perc_saldo_temp;
        end

        insert into kontcalcperc (kcp_period, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo, kcp_transfer_reg_id,
                                  kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff, kcp_perc_begin_saldo,
                                  kcp_perc_transfer_payed, kcp_perc_transfer_writeoff, kcp_kommis_calced,
                                  kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo, kcp_kommis_begin_saldo,
                                  kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff, kcp_perc_account,
                                  kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo, kcp_out_perc_calced,
                                  kcp_out_perc_saldo, kcp_from_c_id)
        values (:kcp_period, :kcp_from_k_id, :kcp_from_reg_id, :kcp_perc_saldo_cli, :kcp_transfer_reg_id,
                :kcp_perc_calced_cli, :kcp_perc_payed_cli, :kcp_perc_writeoff_cli, :kcp_perc_begin_saldo_cli,
                :kcp_perc_transfer_payed_cli, :kcp_perc_transfer_writeoff_cli, :kcp_kommis_calced_cli,
                :kcp_kommis_payed_cli, :kcp_kommis_writeoff_cli, :kcp_kommis_saldo_cli, :kcp_kommis_begin_saldo_cli,
                :kcp_kommis_transfer_payed_cli, :kcp_kommis_transfer_woff_cli, :kcp_perc_account, :kcp_kommis_account,
                :kcp_loan_is_closed, :kcp_out_perc_begin_saldo_cli, :kcp_out_perc_calced_cli, :kcp_out_perc_saldo_cli,
                :kc_from_c_id);

      end
    end
    else
    begin
      insert into kontcalcperc (kcp_period, kcp_from_k_id, kcp_from_reg_id, kcp_perc_saldo, kcp_transfer_reg_id,
                                kcp_perc_calced, kcp_perc_payed, kcp_perc_writeoff, kcp_perc_begin_saldo,
                                kcp_perc_transfer_payed, kcp_perc_transfer_writeoff, kcp_kommis_calced,
                                kcp_kommis_payed, kcp_kommis_writeoff, kcp_kommis_saldo, kcp_kommis_begin_saldo,
                                kcp_kommis_transfer_payed, kcp_kommis_transfer_writeoff, kcp_perc_account,
                                kcp_kommis_account, kcp_loan_is_closed, kcp_out_perc_begin_saldo, kcp_out_perc_calced,
                                kcp_out_perc_saldo, kcp_from_c_id)
      values (:kcp_period, :kcp_from_k_id, :kcp_from_reg_id, :kcp_perc_saldo, :kcp_transfer_reg_id, :kcp_perc_calced,
              :kcp_perc_payed, :kcp_perc_writeoff, :kcp_perc_begin_saldo, :kcp_perc_transfer_payed,
              :kcp_perc_transfer_writeoff, :kcp_kommis_calced, :kcp_kommis_payed, :kcp_kommis_writeoff,
              :kcp_kommis_saldo, :kcp_kommis_begin_saldo, :kcp_kommis_transfer_payed, :kcp_kommis_transfer_writeoff,
              :kcp_perc_account, :kcp_kommis_account, :kcp_loan_is_closed, :kcp_out_perc_begin_saldo,
              :kcp_out_perc_calced, :kcp_out_perc_saldo, :kc_from_c_id);
    end

    suspend;
  end
end
