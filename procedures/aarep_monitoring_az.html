CREATE PROCEDURE AAREP_MONITORING_AZ(
    REP_REGION_ID D_INTEGER,
    REP_DATE_BEGIN D_DATE,
    REP_DATE_END D_DATE)
RETURNS (
    C_FIO D_STR_100,
    OFFICE_NAME D_STR_CODE,
    KONTRACT_NO D_STR_20,
    K_DISB_DATE D_DATE,
    KC_AMOUNT_APPROVED D_CURRENCY,
    OVERDUE_DAYS D_INTEGER,
    OKS_BALANCE D_CURRENCY,
    SH_CALC_PERC D_CURRENCY,
    DELINQ5 D_STR_10,
    FIRST1 D_STR_10,
    DELINQ70 D_STR_10,
    SECOND2 D_STR_10,
    THIRD3 D_STR_10,
    FOURTH4 D_STR_10,
    RESTRUCT_STATUS D_STR_10,
    LAL_NAME D_STR_100)
AS
declare variable K_ID D_ID;
declare variable C_ID D_ID;
begin
    c_fio = '';
    office_name = '';
    kontract_no = '';
    k_disb_date = null;
    kc_amount_approved = 0;
    overdue_days = 0;
    oks_balance = 0;
    first1 = '';
    second2 = '';
    third3 = '';
    fourth4 = '';

    for select c.c_name_fio,
           reg.reg_name,
           kont.k_kontract_no,
           kont.k_disb_date,
           kc.kc_amount_approved,
           maxvalue(:rep_date_end - oks.oks_delinq_date, 0),
           oks.oks_balance,
           kc.kc_from_k_id,
           kc.kc_from_c_id
    from oksummary oks
    inner join region reg on reg.reg_unicode = oks.oks_region_id
    inner join region_list(:rep_region_id, 1, 3, 1) on reg.reg_unicode = id_list
    inner join kontract kont on kont.k_id = oks.oks_from_k_id
    inner join knclient kc on kc.kc_from_k_id = kont.k_id
    inner join clients c on c.c_id = kc.kc_from_c_id
    where kc.kc_amount_approved > 0
    and oks.oks_close_state = 0
    and :rep_date_end - oks.oks_delinq_date > 0
    and :rep_date_end between oks.oks_date_begin and oks.oks_date_end
    into :c_fio,
         :office_name,
         :kontract_no,
         :k_disb_date,
         :kc_amount_approved,
         :overdue_days,
         :oks_balance,
         :k_id,
         :c_id
    do
    begin
        sh_calc_perc = 0;

        select sum(sh_calc_perc)
        from aarep_make_zad_list(:k_id, :rep_date_end, 1)
        into :sh_calc_perc;
        
        delinq5 = '';
        delinq70 = '';

        select
            case when dln.dl_from_el_code = '5' then 'Да' else 'Нет' end,
            case when dln.dl_from_el_code = '70' then 'Да' else 'Нет' end
        from delinquent_loan_notify dln
        where dln.dl_from_k_id = :k_id
          and dln.dl_from_c_id = :c_id
          and dln.dl_from_el_code in ('5', '70')
        into :delinq5, :delinq70;

        restruct_status = null;
    
        select iif(is_restruct = 1, 'Реструкт', 'Нет')
        from aarep_get_restructuring(:k_id, :rep_date_begin, null)
        into :restruct_status; 

        lal_name = '';

        select lal.lal_name
        from lawkredit lak
        inner join lawmain lam on lam.lam_id = lak.lak_from_lam_id
        inner join lawlist lal on lal.lal_id = lam.lam_from_lal_id
        where lak.lak_from_k_id = :k_id
        into :lal_name;

        suspend;
    end
end
