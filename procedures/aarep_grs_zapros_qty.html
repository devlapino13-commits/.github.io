CREATE PROCEDURE AAREP_GRS_ZAPROS_QTY(
    REP_DATE_END D_DATE)
RETURNS (
    REP_MONTH D_DATE,
    KONT_QTY D_INTEGER,
    MAPP_QTY D_INTEGER,
    ZAPR_QTY D_INTEGER,
    USPESH_ZAPR_QTY D_INTEGER,
    FAILED_ZAPR_QTY D_INTEGER,
    REPEAT_QTY D_INTEGER)
AS
declare variable rep_month_end d_date;

begin
  rep_month = cast('01.01.' || extract(year from :rep_date_end) as date);
  while (rep_month <= :rep_date_end) do
  begin
    rep_month_end = dateadd(month, 1, :rep_month) - 1;

    zapr_qty = 0;

    kont_qty = 0;

    select count(*)
    from kontstate ks
    where ks.ks_date_begin between :rep_month and iif(:rep_date_end between :rep_month and :rep_month_end, :rep_date_end, :rep_month_end) and
          ks.ks_ol_is_delete = 0 and
          ks.ks_state = 1
    into :kont_qty;

    kont_qty = coalesce(kont_qty, 0);

    mapp_qty = 0;

    select count(*)
    from miniapp ma
    where ma.ma_date_app between :rep_month and iif(:rep_date_end between :rep_month and :rep_month_end, :rep_date_end, :rep_month_end) and
          ma.ma_is_active = 1 and
          ma.ma_date_end = '31.12.9999'
    into mapp_qty;

    mapp_qty = coalesce(mapp_qty, 0);

    uspesh_zapr_qty = 0;

    select count(*)
    from extinfolist eil
    where eil.eil_from_eic_id = 2 and --Запрос ГРС
          eil.eil_date between :rep_month and iif(:rep_date_end between :rep_month and :rep_month_end, :rep_date_end, :rep_month_end) and --Запросы внутри одного месяца
          eil.eil_status = 2 --Успешные запросы
    into :uspesh_zapr_qty;

    failed_zapr_qty = 0;

    select count(*)
    from extinfolist eil
    where eil.eil_from_eic_id = 2 and --Запрос ГРС
          eil.eil_date between :rep_month and iif(:rep_date_end between :rep_month and :rep_month_end, :rep_date_end, :rep_month_end) and --Запросы внутри одного месяца
          eil.eil_status <> 2 -- Неуспешные запросы
    into :failed_zapr_qty;

    zapr_qty = failed_zapr_qty + uspesh_zapr_qty;

    repeat_qty = 0; --Дубль запросы

    with tbl
    as (select count(eil.eil_from_c_id) as qty
        from extinfolist eil
        where eil.eil_from_eic_id = 2 and --Запрос ГРС
              eil.eil_date between :rep_month and iif(:rep_date_end between :rep_month and :rep_month_end, :rep_date_end, :rep_month_end) and
              eil.eil_status = 2
        group by eil.eil_from_c_id)
    select count(qty)
    from tbl as t
    where t.qty > 1
    into repeat_qty;

    suspend;
    rep_month = dateadd(month, 1, rep_month);
  end
end
