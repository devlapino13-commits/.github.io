CREATE PROCEDURE CIB_KONT_ERROR_LIST(
    CP_ID D_ID)
RETURNS (
    CKE_ID D_ID,
    CKE_REC_NO D_INTEGER,
    CKE_IS_CHECKED D_SMALLINT,
    CKE_CHECKED_UN_ID D_ID,
    CKE_CHECKED_DATETIME D_DATETIME,
    CKE_CHECKED_DATETIME_STR D_STR_100,
    K_KONTRACT_NO STR_20,
    C_NAME_FIO STR_100,
    CKE_ERROR_CODE STR_100,
    CKE_ERROR_TEXT D_STR_8100,
    UN_FIO STR_100)
AS
begin
  for select
      cke_id,
      cke_rec_no,
      cke_is_checked,
      cke.cke_checked_un_id,
      cke_checked_datetime,
      f_datetimetostr(cke_checked_datetime),
      cke.cke_error_code,
      cke.cke_error_text,
      k_kontract_no,
      c_name_fio,
      un_fio
    from cib_kontract_errors cke
      inner join kontract
        on k_id = cke_from_k_id
      left join clients
        on c_id = cke_from_c_id
      left join username
        on un_id = cke_checked_un_id
    where cke.cke_from_cp_id =:cp_id
  into
    :cke_id,
    :cke_rec_no,
    :cke_is_checked,
    :cke_checked_un_id,
    :cke_checked_datetime,
    :cke_checked_datetime_str,
    :cke_error_code,
    :cke_error_text,
    :k_kontract_no,
    :c_name_fio,
    :un_fio
  do
  begin
    un_fio = coalesce(un_fio, '---');
    cke_checked_datetime_str = iif(cke_checked_datetime_str = '', '[Не исправлен]', cke_checked_datetime_str);
    suspend;
  end
end
