CREATE PROCEDURE UXP$UI_UX_VERSIONS(
    UA_UUID D_UUID,
    VER_KIND D_STR_CODE,
    UN_ID D_ID,
    UA_UUID_CODE D_STR_CODE = null)
RETURNS (
    UFM_UUID D_UUID,
    DRAFT_UUID D_UUID,
    RELEASE_UUID D_UUID,
    UFM_UPDATED D_DATETIME,
    DRAFT_UPDATED D_DATETIME,
    RELEASE_UPDATED D_DATETIME,
    USER_ACCESS D_SMALLINT,
    SHOW_DIALOG D_BOOL,
    IS_REPORT D_BOOL,
    DENIED_DEFAULT_UA_LIST D_STR_LONG)
AS
declare variable uxa_uuid d_uuid;
declare variable uxa_owner d_uuid;
declare variable ux_kind d_str_code;
declare variable ua_usr_id_field d_str_code;
declare variable ua_action_kind d_str_code;
declare variable ux_kind_code d_str_code;
declare variable uxa_kind d_smallint;
begin
  if (coalesce(ua_uuid, '') = '') then
    ua_uuid = uxf$uniq(ua_uuid_code);

  show_dialog = 0;
  ux_kind = uxf$code(ua_uuid, 'kind');
  denied_default_ua_list = '';

  is_report = 0;
  if (ux_kind <> 'uxaction') then
  begin
    ufm_uuid = ua_uuid;
    draft_uuid = ua_uuid;
    ufm_updated = current_timestamp;
    draft_updated = current_timestamp;
    user_access = 1;

    if (ux_kind = 'uxreport') then
      is_report = 1;

    suspend;

    exit;
  end

  select uxa_uuid, uxa_owner, ux_draft_uuid, ux_release_uuid, ua_usr_id_field, uxa_kind
  from ux$action, ux$alias, ux$code
  where ua_uuid = :ua_uuid and
        ua_uxa_uuid = uxa_uuid and
        uxa_owner = ux_uuid
  into uxa_uuid, uxa_owner, draft_uuid, release_uuid, ua_usr_id_field, uxa_kind;

  ux_kind_code = uxf$code(uxa_owner, 'kind');

  if (ux_kind_code = 'uxreport') then
    is_report = 1;

  if (ux_kind_code = 'uxcommand') then
    ua_action_kind = 'command';
  else
  if (ux_kind_code = 'uxreport') then
    ua_action_kind = 'report';
  else
  begin
    ua_action_kind = trim(case
      when ua_usr_id_field <> '' then 'search'
      when uxa_kind = 2 then 'insert'
      else 'open'
    end);
  end

  ufm_uuid = uxf$ver(uxa_owner, ver_kind);
  user_access = uxf$access(un_id, uxa_uuid);

  select ufm_updated
  from ux$form
  where ufm_uuid = :draft_uuid
  into draft_updated;

  if (release_uuid is not null and
      release_uuid is distinct from draft_uuid) then
  begin
    show_dialog = 1;

    select ufm_updated
    from ux$form
    where ufm_uuid = :release_uuid
    into release_updated;
  end

  if (ufm_uuid = draft_uuid) then
    ufm_updated = draft_updated;

  if (ufm_uuid = release_uuid and release_uuid is distinct from draft_uuid) then
    ufm_updated = release_updated;

  suspend;
end
