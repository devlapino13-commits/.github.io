CREATE PROCEDURE INSERT_CLIENTS(
    C_TYPE INTEGER,
    C_NAME_FIO VARCHAR(100),
    C_SEX SMALLINT,
    C_BIRTH_DATE DATE,
    C_NATION INTEGER,
    C_CITIZEN INTEGER,
    C_PASS_NO VARCHAR(30),
    C_FROM_REG_ID INTEGER,
    C_INN VARCHAR(20),
    C_IS_RESIDENT SMALLINT,
    C_DOC_TYPE INTEGER,
    C_FORM_STATUS_LONG VARCHAR(255),
    C_FORM_STATUS_SHORT VARCHAR(20),
    C_ORG_TYPE_LONG VARCHAR(255),
    C_ORG_TYPE_SHORT VARCHAR(20),
    C_FROM_UN_ID DECIMAL(18,0),
    C_ALTERNAT_FIO VARCHAR(100))
RETURNS (
    C_ID DECIMAL(18,0))
AS
declare variable st_staff integer;
declare variable st_client integer;
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  select result
  from get_kv_id(2)
  into :c_id;

  insert into clients (c_id, c_type, c_name_fio, c_alternat_fio, c_sex, c_birth_date, c_nation, c_citizen, c_pass_no,
                       c_from_reg_id, c_inn, c_is_resident, c_doc_type, c_form_status_long, c_form_status_short,
                       c_org_type_long, c_org_type_short, c_from_un_id)
  values (:c_id, :c_type, :c_name_fio, :c_alternat_fio, :c_sex, :c_birth_date, :c_nation, :c_citizen, :c_pass_no,
          :c_from_reg_id, :c_inn, :c_is_resident, :c_doc_type, :c_form_status_long, :c_form_status_short,
          :c_org_type_long, :c_org_type_short, :c_from_un_id);
  suspend;
end
