CREATE PROCEDURE HP_REF_MONEYMASK_IMPORT(
    RMM_ID TYPE OF D_ID,
    CODE_FROM_AB_DT TYPE OF D_STR_CODE,
    CODE_FROM_AB_KT TYPE OF D_STR_CODE,
    RMM_MASK_DT TYPE OF D_STR_CODE,
    RMM_MASK_KT TYPE OF D_STR_CODE,
    RMM_ADD_INFO TYPE OF D_STR_CODE,
    CODE_FROM_RMO_RESIDENT TYPE OF D_STR_CODE,
    CODE_FROM_RMO_NORESIDENT TYPE OF D_STR_CODE,
    CODE_LAW_RMO_RESIDENT TYPE OF D_STR_CODE,
    CODE_LAW_RMO_NORESIDENT TYPE OF D_STR_CODE,
    RMM_NAME TYPE OF D_STR_255,
    RMM_MEMO TYPE OF D_STR_255)
AS
declare variable rmm_from_ab_dt d_id;
declare variable rmm_from_ab_kt d_id;
declare variable rmm_from_rmo_resident d_id;
declare variable rmm_from_rmo_noresident d_id;
declare variable rmm_law_rmo_resident d_id;
declare variable rmm_law_rmo_noresident d_id;
begin
  select ab_id
  from ref_accbalance
  where ab_code = :code_from_ab_dt and
        ab_level = 3
  into rmm_from_ab_dt;

  if (rmm_from_ab_dt is null) then
    exception e_default f_format('code_from_ab_dt not found: %1', coalesce(code_from_ab_dt, '<Null>'));

  select ab_id
  from ref_accbalance
  where ab_code = :code_from_ab_kt and
        ab_level = 3
  into rmm_from_ab_kt;

  if (rmm_from_ab_kt is null) then
    exception e_default f_format('code_from_ab_kt not found: %1', coalesce(code_from_ab_kt, '<Null>'));

  select rmo_id
  from ref_moneyoper
  where rmo_code = :code_from_rmo_resident
  into rmm_from_rmo_resident;

  if (code_from_rmo_resident = '0') then
    rmm_from_rmo_resident = 0;

  if (rmm_from_rmo_resident is null) then
    exception e_default f_format('code_from_rmo_resident not found: %1', coalesce(rmm_from_rmo_resident, '<Null>'));

  select rmo_id
  from ref_moneyoper
  where rmo_code = :code_from_rmo_noresident
  into rmm_from_rmo_noresident;

  if (code_from_rmo_noresident = '0') then
    rmm_from_rmo_noresident = 0;

  if (rmm_from_rmo_noresident is null) then
    exception e_default f_format('code_from_rmo_noresident not found: %1', coalesce(rmm_from_rmo_noresident, '<Null>'));

  select rmo_id
  from ref_moneyoper
  where rmo_code = :code_law_rmo_resident
  into rmm_law_rmo_resident;

  if (code_law_rmo_resident = '0') then
    rmm_law_rmo_resident = 0;

  if (rmm_law_rmo_resident is null) then
    exception e_default f_format('code_law_rmo_resident not found: %1', coalesce(rmm_law_rmo_resident, '<Null>'));

  select rmo_id
  from ref_moneyoper
  where rmo_code = :code_law_rmo_noresident
  into rmm_law_rmo_noresident;

  if (code_law_rmo_noresident = '0') then
    rmm_law_rmo_noresident = 0;

  if (rmm_law_rmo_noresident is null) then
    exception e_default f_format('code_law_rmo_noresident not found: %1', coalesce(rmm_law_rmo_noresident, '<Null>'));


  insert into ref_moneymask (
    rmm_id,
    rmm_from_ab_dt,
    rmm_from_ab_kt,
    rmm_mask_dt,
    rmm_mask_kt,
    rmm_add_info,
    rmm_from_rmo_resident,
    rmm_from_rmo_noresident,
    rmm_law_rmo_resident,
    rmm_law_rmo_noresident,
    rmm_name,
    rmm_memo)
  values (
    :rmm_id,
    :rmm_from_ab_dt,
    :rmm_from_ab_kt,
    :rmm_mask_dt,
    :rmm_mask_kt,
    :rmm_add_info,
    :rmm_from_rmo_resident,
    :rmm_from_rmo_noresident,
    :rmm_law_rmo_resident,
    :rmm_law_rmo_noresident,
    :rmm_name,
    :rmm_memo);
end
