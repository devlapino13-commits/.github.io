CREATE PROCEDURE UXP$UXEXPRFIELD_UPDATE(
    EXP_UX_UUID D_UUID,
    EXP_KIND D_SMALLINT)
AS
declare variable us_uuid d_uuid;
declare variable uf_uuid d_uuid;
declare variable uf_field_name d_str_code;
declare variable txt d_str_code;
declare variable field_used d_bool;
declare variable exp_ufr_uuid d_uuid;
declare variable new_expression d_str_long;
begin
  delete from ux$exprfield
  where exf_ux_uuid = :exp_ux_uuid and
        exf_kind = :exp_kind;

  select exp_expression, exp_calc_ufr_uuid
  from ux$expression
  where exp_ux_uuid = :exp_ux_uuid and
        exp_kind = :exp_kind
  into new_expression, exp_ufr_uuid;

  new_expression = trim(coalesce(new_expression, ''));

  if (new_expression <> '') then
  begin
    select us_uuid
    from uxp$uxframe_source(:exp_ux_uuid, :exp_ufr_uuid)
    into :us_uuid;

    if (us_uuid is null) then
      exception e_default 'Source not found for uuid: ' || uuid_to_char(exp_ux_uuid);

    for select uf_uuid, uf_field_name
    from ux$field
    where uf_owner = :us_uuid
    order by char_length(uf_field_name) desc
    into uf_uuid, uf_field_name
    do
    begin
      field_used = 0;

      txt = 'old.' || uf_field_name;
      if (position(txt, new_expression) > 0) then
      begin
        field_used = 1;
        new_expression = replace(new_expression, txt, uf_field_name);
      end

      txt = 'new.' || uf_field_name;
      if (position(txt, new_expression) > 0) then
      begin
        field_used = 1;
        new_expression = replace(new_expression, txt, uf_field_name);
      end

      txt = ':' || uf_field_name;
      if (position(txt, new_expression) > 0) then
      begin
        field_used = 1;
        new_expression = replace(new_expression, txt, uf_field_name);
      end

      if (field_used = 1) then
      begin
        insert into ux$exprfield (exf_ux_uuid, exf_kind, exf_uf_uuid)
        values (:exp_ux_uuid, :exp_kind, :uf_uuid);
      end
    end
  end
end
