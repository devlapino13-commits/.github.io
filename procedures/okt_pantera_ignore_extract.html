CREATE PROCEDURE OKT_PANTERA_IGNORE_EXTRACT(
    IS_FULL_EXTRACT SMALLINT,
    PROJECT STR_255,
    OBJECT_KIND STR_255,
    OBJECT_ID ID_TYPE,
    SUBJECT_KIND STR_255,
    REP_DATE DATE)
AS
declare variable OBJECT_KIND_ID integer;
declare variable PROJECT_ID integer;
declare variable SUBJECT_KIND_ID integer;
begin
  if (rep_date is null) then
    rep_date = current_date;
  is_full_extract = coalesce(is_full_extract, 0);

  object_kind_id = f_objkind(object_kind, 'id');

  if (subject_kind similar to '[-]?[[:DIGIT:]]+') then
    subject_kind_id = cast(subject_kind as integer);
  else
  begin
    select rd_id
    from refdata
    where rd_code = :subject_kind
    into :subject_kind_id;
  end

  if (project similar to '[-]?[[:DIGIT:]]+') then
    project_id = cast(project as integer);
  else
  begin
    select rd_id
    from refdata
    where rd_code = :project
    into :project_id;
  end

  if (is_full_extract = 1) then
  begin
    update pantera_ignore pi
    set pi.pi_date_end = :rep_date - 1
    where pi.pi_object_kind = :object_kind_id and
          pi.pi_object_id = :object_id and
          pi.pi_subject_kind = :subject_kind_id and
          pi_date_end = '31.12.9999';
  end
  else
  begin
    if (project_id is null) then
    begin
      update pantera_ignore pi
      set pi.pi_date_end = :rep_date - 1
      where pi.pi_rd_project_id is null and
            pi.pi_object_kind = :object_kind_id and
            pi.pi_object_id = :object_id and
            pi.pi_subject_kind = :subject_kind_id and
            pi_date_end = '31.12.9999';
    end
    else
    begin
      update pantera_ignore pi
      set pi.pi_date_end = :rep_date - 1
      where pi.pi_rd_project_id = :project_id and
            pi.pi_object_kind = :object_kind_id and
            pi.pi_object_id = :object_id and
            pi.pi_subject_kind = :subject_kind_id and
            pi_date_end = '31.12.9999';
    end
  end
end
