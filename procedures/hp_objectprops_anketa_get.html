CREATE PROCEDURE HP_OBJECTPROPS_ANKETA_GET(
    OBJECT_KIND VARCHAR(100),
    OBJECT_ID VARCHAR(255),
    SUBJECT_KIND VARCHAR(100),
    SUBJECT_ID VARCHAR(255),
    REP_DATE DATE,
    NEEDED_RD_CODE VARCHAR(100))
RETURNS (
    RES_HAS_CODE INTEGER,
    RES_STR_VALUES VARCHAR(255),
    RES_MEMO VARCHAR(255))
AS
declare variable UNIQ_OBJECT_KIND integer;
declare variable UNIQ_OBJECT_ID decimal(18,0);
declare variable UNIQ_SUBJECT_KIND integer;
declare variable UNIQ_SUBJECT_ID decimal(18,0);
begin
-- процедура устанавливает получает значение для пары object/subject
  if (rep_date >= '31.12.2999') then
    rep_date = null;

  if (object_kind similar to '[-]?[[:DIGIT:]]+' and
      object_id similar to '[-]?[[:DIGIT:]]+' and
      subject_kind similar to '[-]?[[:DIGIT:]]+' and
      subject_id similar to '[-]?[[:DIGIT:]]+') then
  begin
    uniq_object_kind = object_kind;
    uniq_object_id = object_id;
    uniq_subject_kind = subject_kind;
    uniq_subject_id = subject_id;
  end
  else
  begin
    select opr_object_kind, opr_object_id, opr_subject_kind, opr_subject_id
    from hp_get_propsparam(:object_kind, :object_id, :subject_kind, :subject_id)
    into :uniq_object_kind, :uniq_object_id, :uniq_subject_kind, :uniq_subject_id;
  end

  res_has_code = 0;
  res_str_values = '';
  if (rep_date is null) then
  begin
    if (exists(select *
               from objectvalres, objectprops, refdata
               where opr_object_kind = :uniq_object_kind and
                     opr_object_id = :uniq_object_id and
                     opr_subject_kind = :uniq_subject_kind and
                     opr_subject_id = :uniq_subject_id and
                     ovr_from_opr_id = opr_id and
                     ovr_from_ovs_id = opr_actual_ovs_id and
                     ovr_result_rd_id = rd_id and
                     rd_code = :needed_rd_code)) then
      res_has_code = 1;

    select list(iif(ovr_value <> '', ovr_value, rd_name), ', '), left(list(rd_memo), 255)
    from objectvalres, objectprops, refdata
    where opr_object_kind = :uniq_object_kind and
          opr_object_id = :uniq_object_id and
          opr_subject_kind = :uniq_subject_kind and
          opr_subject_id = :uniq_subject_id and
          ovr_from_opr_id = opr_id and
          ovr_from_ovs_id = opr_actual_ovs_id and
          ovr_result_rd_id = rd_id
    into :res_str_values, :res_memo;
  end
  else
  begin
    if (exists(select *
               from objectvalres, objectprops, refdata, objectvalues
               where opr_object_kind = :uniq_object_kind and
                     opr_object_id = :uniq_object_id and
                     opr_subject_kind = :uniq_subject_kind and
                     opr_subject_id = :uniq_subject_id and
                     ovr_from_opr_id = opr_id and
                     ovr_from_ovs_id = ovs_id and
                     ovs_from_opr_id = opr_id and
                     ovs_date_begin <= :rep_date and
                     ovs_date_end >= :rep_date and
                     ovr_result_rd_id = rd_id and
                     rd_code = :needed_rd_code)) then
      res_has_code = 1;

    select list(iif(ovr_value <> '', ovr_value, rd_name), ', '), left(list(rd_memo), 255)
    from objectvalres, objectprops, refdata, objectvalues
    where opr_object_kind = :uniq_object_kind and
          opr_object_id = :uniq_object_id and
          opr_subject_kind = :uniq_subject_kind and
          opr_subject_id = :uniq_subject_id and
          ovr_from_opr_id = opr_id and
          ovr_from_ovs_id = ovs_id and
          ovs_from_opr_id = opr_id and
          ovs_date_begin <= :rep_date and
          ovs_date_end >= :rep_date and
          ovr_result_rd_id = rd_id
    into :res_str_values, :res_memo;
  end
  suspend;
end
