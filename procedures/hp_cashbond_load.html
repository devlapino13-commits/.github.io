CREATE PROCEDURE HP_CASHBOND_LOAD(
    CIL_ID D_INTEGER)
RETURNS (
    RES_BOND_LIST D_STR_LONG)
AS
begin
  res_bond_list = '';

  select list(round(cib_bond_id * 100.00) || '=' || cib_count, ';')
  from cashinvbonds
  where cib_from_cil_id = :cil_id and
        cib_count <> 0
  into res_bond_list;

  if (coalesce(res_bond_list, '') <> '') then
  begin
    res_bond_list = replace(res_bond_list, '(', '');
    res_bond_list = replace(res_bond_list, ')', '');
  end
  suspend;
end
