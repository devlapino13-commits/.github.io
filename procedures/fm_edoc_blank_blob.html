CREATE PROCEDURE FM_EDOC_BLANK_BLOB(
    EDS_ID D_ID,
    EDB_ID D_ID,
    UN_ID D_ID)
RETURNS (
    EDB_BLOB D_BLOB_BINARY)
AS
declare variable bls_id d_id;
declare variable eds_status d_smallint;
begin
  select edb_from_bls_id, edb_blob
  from edoc_blank, edoc_frx
  where edb_id = :edb_id
    and edb_from_edf_id = edf_id
  into bls_id, edb_blob;

  if (edb_blob is null and bls_id is not null) then
  begin
    select blx_blob
    from hp_get_blobex_blob(:bls_id)
    into edb_blob;
  end

  select eds_status
  from edoc_sign
  where eds_id = :eds_id
  into eds_status;

  if (eds_status = 0) then
  begin
    update or insert into edoc_open (edo_from_eds_id, edo_from_edb_id,
      edo_before_sign_datetime)
    values (:eds_id, :edb_id, current_timestamp)
    matching (edo_from_eds_id, edo_from_edb_id);
  end
  else if (eds_status = 1) then
  begin
    update or insert into edoc_open (edo_from_eds_id, edo_from_edb_id,
      edo_after_sign_datetime)
    values (:eds_id, :edb_id, current_timestamp)
    matching (edo_from_eds_id, edo_from_edb_id);
  end

  suspend;
end
