CREATE PROCEDURE HP_KKT_GET_SHIFT_DATA(
    IN_KKD_ID D_ID,
    REP_DATETIME_BEGIN D_DATETIME,
    REP_DATETIME_END D_DATETIME,
    LAST_ONLY D_BOOL)
RETURNS (
    REC_NO D_INTEGER,
    KKD_ID D_INTEGER,
    REG_NAME D_STR_255,
    REG_ADDRESS D_STR_255,
    FACTORY_NUMBER D_STR_CODE,
    FN_NUMBER D_STR_CODE,
    SHIFT_NUMBER D_INTEGER,
    SHIFT_STATE D_STR_CODE,
    SHIFT_OPEN_DATETIME D_DATETIME,
    FIRST_CHEQUE_FD D_STR_CODE,
    SHIFT_CLOSE_DATETIME D_DATETIME,
    LAST_CHEQUE_DATETIME D_DATETIME,
    LAST_CHEQUE_FD D_STR_CODE,
    FIRST_CHEQUE_LINK D_STR_255,
    LAST_CHEQUE_LINK D_STR_255,
    KKD_NOTSENT_COUNT D_INTEGER,
    KKD_NOTSENT_DATETIME D_DATETIME)
AS
declare variable kkf_id d_integer;
declare variable kkm_id d_id;
begin
  if (rep_datetime_end is null) then
    exit;

  if (rep_datetime_begin is null) then
    rep_datetime_begin = '01.01.1900';

  if (last_only is null) then
    last_only = 0;

  if (last_only = 1) then
    in_kkd_id = null;

  rec_no = 0;

  for select kkd_id, kkd_code, reg_address, kkd_notsent_count, kkd_notsent_datetime, name
  from hp_kkt_get_device_list
  where :in_kkd_id is null or kkd_id = :in_kkd_id
  into kkd_id, factory_number, reg_address, kkd_notsent_count, kkd_notsent_datetime, reg_name
  do
  begin
    fn_number = null;
    shift_number = null;
    shift_open_datetime = null;
    shift_close_datetime = null;

    for select kkf_code, kks_number, kks_open_datetime_fisc, kks_close_datetime_fisc, kkf_id
    from kkt_fn, kkt_shift
    where kkf_from_kkd_id = :kkd_id
      and kks_from_kkf_id = kkf_id
      and kks_open_datetime_fisc between :rep_datetime_begin and :rep_datetime_end
    order by kks_open_datetime_fisc desc
    into fn_number, shift_number, shift_open_datetime, shift_close_datetime, kkf_id
    do
    begin
      first_cheque_fd = null;
      last_cheque_datetime = null;
      last_cheque_fd = null;
      first_cheque_link = null;
      last_cheque_link = null;
  
      if (shift_close_datetime > rep_datetime_end) then
        shift_close_datetime = null;
  
      shift_state = case
                      when shift_close_datetime is not null then 'Закрыта'
                      when dateadd(hour, 24, shift_open_datetime) < rep_datetime_end then 'Открыта и истекла'
                      when shift_open_datetime is not null then 'Открыта'
                      else 'Не определено'
                    end;
  
      select kkm_id, kkm_fd
      from kkt_main
      where kkm_from_kkf_id = :kkf_id
        and kkm_fisc_datetime >= :shift_open_datetime
      order by kkm_fisc_datetime
      rows 1
      into kkm_id, first_cheque_fd;
  
      select link
      from hp_kkt_get_ofd_link(:kkm_id)
      into first_cheque_link;
  
      select kkm_id, kkm_fd, kkm_fisc_datetime
      from kkt_main
      where kkm_from_kkf_id = :kkf_id
        and kkm_fisc_datetime <= :shift_close_datetime
      order by kkm_fisc_datetime desc
      rows 1
      into kkm_id, last_cheque_fd, last_cheque_datetime;
  
      select link
      from hp_kkt_get_ofd_link(:kkm_id)
      into last_cheque_link;
  
      rec_no = rec_no + 1;

      suspend;
  
      if (last_only = 1) then
        leave;
    end 
  end
end
