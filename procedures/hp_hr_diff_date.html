CREATE PROCEDURE HP_HR_DIFF_DATE(
    DATE_BEGIN D_DATE,
    DATE_END D_DATE,
    ORDER_TYPE D_INTEGER,
    REG_ID D_INTEGER)
RETURNS (
    DIFF_DAYS D_INTEGER,
    DAY_BACK D_DATE)
AS
declare variable curr_dd d_date;
begin
  if (date_begin > date_end) then
  begin
    diff_days = 0;
    day_back = date_end;
  end else
  begin
    diff_days = -1;
    curr_dd = date_begin - 1;
    while (curr_dd < date_end) do
    begin
      diff_days = diff_days + 1;
  
      select date_end, next_work_date
      from hp_hr_next_date(:date_begin, :diff_days, :order_type, :reg_id)
      into :curr_dd, :day_back;
    end
  
    if (date_begin = date_end) then
      diff_days = 1;
  end

  suspend;
end
