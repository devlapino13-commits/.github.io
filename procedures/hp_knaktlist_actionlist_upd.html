CREATE PROCEDURE HP_KNAKTLIST_ACTIONLIST_UPD(
    KA_ID ID_TYPE)
AS
declare variable rd_targetuseloan_id integer;
declare variable c_id id_type;
declare variable acl_id id_type;
declare variable kal_date date;
declare variable kal_prior_lo_id id_type;
declare variable kal_new_kf_officer id_type;
declare variable kf_id id_type;
declare variable kc_id d_id;
begin
  if (ka_id = 0) then
    exit;

  for select kal.kal_date, kal_prior_lo_id, kal_new_kf_officer, kal.kal_from_kf_id
      from knaktlist kal
      where kal.kal_from_ka_id = :ka_id and
            kal.is_delete = 0 and
            current_date between kal.kal_date and kal.kal_date_end
      into :kal_date, :kal_prior_lo_id, :kal_new_kf_officer, :kf_id
  do
  begin
    if ((:kal_prior_lo_id is not null) and
        (:kal_new_kf_officer <> :kal_prior_lo_id)) then
    begin
      select rd.rd_id
      from refdata rd
      where rd.rd_code = 'Anketa.TargetUseLoan' -- Мониторинг целевого использования
      into :rd_targetuseloan_id;

-- получаю участников группы, пареданной папки
      for select kncl.kc_id, kncl.kc_from_c_id
          from kontract kont
          inner join kontstate ks on ks.ks_from_k_id = kont.k_id
          inner join knclient kncl on kncl.kc_from_k_id = kont.k_id
          where kont.k_from_kf_id = :kf_id -- !
                and
                :kal_date between ks.ks_date_begin and ks.ks_date_end and
                ks.ks_ol_is_delete = 0 and
                ks.ks_state = 1 and
                kncl.kc_amount_approved > 0
          into :kc_id, :c_id
      do
      begin
-- если есть активная задача по мониторингу на стром СВЗ, перекидыавю ее новому СВЗ
        for select ao.aco_id
            from actionorder ao
            where ((ao.aco_object_kind = 0 and ao.aco_object_id = :c_id) or
                   (ao.aco_object_kind = 8 and ao.aco_object_id = :kc_id)) and
                  ao.aco_from_rd_id = :rd_targetuseloan_id and
                  ao.aco_respons_c_id = :kal_prior_lo_id
            into :acl_id
        do
        begin
          update actionlist al
          set acl_respons_c_id = :kal_new_kf_officer
          where al.acl_id = :acl_id;
        end
      end
    end
  end
end
