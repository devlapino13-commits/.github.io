CREATE PROCEDURE HP_SEND_MAIL_PUBLIC_PERSON(
    K_ID D_ID)
AS
declare variable k_no d_str_20;
declare variable reg_concat d_str_100;
declare variable mail_to varchar(1000);
declare variable mail_subject d_str_255;
declare variable mail_body blob sub_type 1 segment size 80;
declare variable mail_manager varchar(500); -- RFCKG-5569
declare variable mail_operator varchar(500);
declare variable mail_office varchar(1000);
declare variable body_main varchar(1000);
declare variable office_note varchar(500);
begin
  if (exists(select *
             from knclient kc
             inner join clients c on c.c_id = kc.kc_from_c_id and
                   c.c_is_public = 1
             where kc.kc_from_k_id = :k_id and
                   kc.kc_amount_approved > 0)) then
  begin
    select k.k_kontract_no,
           reg.reg_code || ' ' || reg.reg_name,
           replace(replace(coalesce(reg.reg_email_manager, ''), ' ', ''), ',', ';'),
           replace(replace(coalesce(reg.reg_email_operator, ''), ' ', ''), ',', ';')
    from kontract k
    inner join region reg on k.k_fact_disb_reg_id = reg.reg_unicode
    where k.k_id = :k_id
    into :k_no, :reg_concat, :mail_manager, :mail_operator;

    /* адреса офиса: менеджер + операционист, без дублей */
    mail_office = :mail_manager;
    if (:mail_operator <> '' and position(:mail_operator in :mail_office) = 0) then
    begin
      if (:mail_office = '') then
        mail_office = :mail_operator;
      else
        mail_office = :mail_office || ';' || :mail_operator;
    end

    mail_subject = 'Выдача кредита клиенту НПДЛ';
    body_main = '<b>Внимание !</b><br><br>В офисе <b>' || :reg_concat ||
                '</b> была произведена выдача кредита <b>' || :k_no || '</b> клиенту НПДЛ.';

    /* 1. письмо комплаенсу — как раньше, плюс пометка при незаполненной почте офиса */
    if (:mail_office = '') then
      office_note = '<br><br><b>Внимание: e-mail офиса не настроен ' ||
                    '(region.reg_email_manager / reg_email_operator), копия в офис не отправлена.</b>';
    else
      office_note = '';

    mail_to = 'compliance@mbulak.kg';
    mail_body = :body_main || :office_note;
    execute procedure hp_send_quick_message('email', :mail_to, :mail_subject, :mail_body, '', 1, 1000000000000);

    /* 2. отдельное письмо в офис выдачи */
    if (:mail_office <> '') then
    begin
      mail_body = :body_main ||
        '<br><br>Если вы получили данное сообщение, необходимо направить на почту ' ||
        '<b>compliance@mbulak.kg</b> данное сообщение со следующей информацией:<br>' ||
        '1. ФИО клиента<br>' ||
        '2. Является ли клиент в настоящее время НПДЛ<br>' ||
        '3. Место работы клиента и должность<br>' ||
        '4. Источник происхождения денежных средств';
      execute procedure hp_send_quick_message('email', :mail_office, :mail_subject, :mail_body, '', 1, 1000000000000);
    end
  end
end
