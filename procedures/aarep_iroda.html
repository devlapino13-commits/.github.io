    REP_GR_PERC_DEL_0 DECIMAL(18,4),
    REP_GR_PERC_DEL_30 DECIMAL(18,4),
    REP_GR_PERC_DEL_60 DECIMAL(18,4),
    REP_GR_PERC_DEL_90 DECIMAL(18,4),
    REP_GR_PERC_DEL_120 DECIMAL(18,4),
    REP_GR_PERC_DEL_150 DECIMAL(18,4),
    REP_GR_PERC_DEL_180 DECIMAL(18,4),
    REP_GR_PERC_DEL_181 DECIMAL(18,4),
    REP_GR_OS_DEL_0 DECIMAL(18,4),

    REP_IND_OS_DEL_120 DECIMAL(18,4),
    REP_IND_OS_DEL_150 DECIMAL(18,4),
    REP_IND_OS_DEL_180 DECIMAL(18,4),
    REP_IND_OS_DEL_181 DECIMAL(18,4))
AS
declare variable dk_id decimal(18,0);
declare variable dk_os decimal(18,4);
declare variable dk_perc decimal(18,4);
declare variable dk_days integer;
declare variable dk_loan_type integer;
declare variable acc varchar(20);
declare variable tmp decimal(18,4);
begin
-- Счет 114
  rep_year = 2011;
  rep_month = 12;
  while ((rep_year * 12 + rep_month) <= (2013 * 12 + 1)) do
  begin
    rep_gr_perc_del_0 = 0;
    rep_gr_perc_del_30 = 0;
    rep_gr_perc_del_60 = 0;
    rep_gr_perc_del_90 = 0;
    rep_gr_perc_del_120 = 0;
    rep_gr_perc_del_150 = 0;
    rep_gr_perc_del_180 = 0;
    rep_gr_perc_del_181 = 0;

    rep_gr_os_del_0 = 0;
    rep_gr_os_del_30 = 0;
    rep_gr_os_del_60 = 0;
    rep_gr_os_del_90 = 0;
    rep_gr_os_del_120 = 0;
    rep_gr_os_del_150 = 0;
    rep_gr_os_del_180 = 0;
    rep_gr_os_del_181 = 0;

    rep_ind_perc_del_0 = 0;
    rep_ind_perc_del_30 = 0;
    rep_ind_perc_del_60 = 0;
    rep_ind_perc_del_90 = 0;
    rep_ind_perc_del_120 = 0;
    rep_ind_perc_del_150 = 0;
    rep_ind_perc_del_180 = 0;
    rep_ind_perc_del_181 = 0;

    rep_ind_os_del_0 = 0;
    rep_ind_os_del_30 = 0;
    rep_ind_os_del_60 = 0;
    rep_ind_os_del_90 = 0;
    rep_ind_os_del_120 = 0;
    rep_ind_os_del_150 = 0;
    rep_ind_os_del_180 = 0;
    rep_ind_os_del_181 = 0;

    if (rep_month = 12) then
      dd = cast('31.12.' || :rep_year as date);
    else
    begin
      dd = cast('01.' ||(rep_month + 1) || '.' || :rep_year as date) - 1;
    end

    for select distinct al_id
        from acc_list
        where al_from_bg_id = '11437' and
              al_number = 1
        into :acc
    do
    begin
      tmp = 0;
      select sum(obs_dt_ekv - obs_kt_ekv)
      from obsummary
      where obs_date <= :dd and
            obs_from_al_id = :acc
      into :tmp;
      if (tmp is null) then
        tmp = 0;
      rep_gr_perc_del_0 = rep_gr_perc_del_0 + tmp;
    end

    for select distinct al_id
        from acc_list
        where al_from_bg_id = '11437' and
              al_number > 1 and
              al_number < 100
        into :acc
    do
    begin
      tmp = 0;
      select sum(obs_dt_ekv - obs_kt_ekv)
      from obsummary
      where obs_date <= :dd and
            obs_from_al_id = :acc
      into :tmp;
      if (tmp is null) then
        tmp = 0;
      rep_ind_perc_del_0 = rep_ind_perc_del_0 + tmp;
    end

    for select distinct al_id
        from acc_list
        where al_from_bg_id = '10931'
        into :acc
    do
    begin
      tmp = 0;
      select sum(obs_dt_ekv - obs_kt_ekv)
      from obsummary
      where obs_date <= :dd and
            obs_from_al_id = :acc
      into :tmp;
      if (tmp is null) then
        tmp = 0;
      rep_gr_os_del_0 = rep_gr_os_del_0 + tmp;
    end

    for select distinct al_id
        from acc_list
        where al_from_bg_id = '10941'
        into :acc
    do
    begin
      tmp = 0;
      select sum(obs_dt_ekv - obs_kt_ekv)
      from obsummary
      where obs_date <= :dd and
            obs_from_al_id = :acc
      into :tmp;
      if (tmp is null) then
        tmp = 0;
      rep_ind_os_del_0 = rep_ind_os_del_0 + tmp;
    end

    for select dk_id, dk_days, dk_balance_end, dk_perc, dk_loan_type
        from aarep_kontract_deliq_temp(1000000, :dd, :dd, '000')
        where (dk_os + dk_perc) >= 0.01
        into :dk_id, :dk_days, :dk_os, dk_perc, dk_loan_type
    do
    begin
      if (dk_loan_type = 0) then
      begin
        if (dk_days <= 30) then
          rep_gr_perc_del_30 = rep_gr_perc_del_30 + dk_perc;
        else
        if (dk_days <= 60) then
          rep_gr_perc_del_60 = rep_gr_perc_del_60 + dk_perc;
        else
        if (dk_days <= 90) then
          rep_gr_perc_del_90 = rep_gr_perc_del_90 + dk_perc;
        else
        if (dk_days <= 120) then
          rep_gr_perc_del_120 = rep_gr_perc_del_120 + dk_perc;
        else
        if (dk_days <= 150) then
          rep_gr_perc_del_150 = rep_gr_perc_del_150 + dk_perc;
        else
        if (dk_days <= 180) then
          rep_gr_perc_del_180 = rep_gr_perc_del_180 + dk_perc;
        else
          rep_gr_perc_del_181 = rep_gr_perc_del_181 + dk_perc;

        if (dk_days <= 30) then
          rep_gr_os_del_30 = rep_gr_os_del_30 + dk_os;
        else
        if (dk_days <= 60) then
          rep_gr_os_del_60 = rep_gr_os_del_60 + dk_os;
        else
        if (dk_days <= 90) then
          rep_gr_os_del_90 = rep_gr_os_del_90 + dk_os;
        else
        if (dk_days <= 120) then
          rep_gr_os_del_120 = rep_gr_os_del_120 + dk_os;
        else
        if (dk_days <= 150) then
          rep_gr_os_del_150 = rep_gr_os_del_150 + dk_os;
        else
        if (dk_days <= 180) then
          rep_gr_os_del_180 = rep_gr_os_del_180 + dk_os;
        else
          rep_gr_os_del_181 = rep_gr_os_del_181 + dk_os;

        rep_gr_perc_del_0 = rep_gr_perc_del_0 - dk_perc;
        rep_gr_os_del_0 = rep_gr_os_del_0 - dk_os;
      end
      else
      begin
        if (dk_days <= 30) then
          rep_ind_perc_del_30 = rep_ind_perc_del_30 + dk_perc;
        else
        if (dk_days <= 60) then
          rep_ind_perc_del_60 = rep_ind_perc_del_60 + dk_perc;
        else
        if (dk_days <= 90) then
          rep_ind_perc_del_90 = rep_ind_perc_del_90 + dk_perc;
        else
        if (dk_days <= 120) then
          rep_ind_perc_del_120 = rep_ind_perc_del_120 + dk_perc;
        else
        if (dk_days <= 150) then
          rep_ind_perc_del_150 = rep_ind_perc_del_150 + dk_perc;
        else
        if (dk_days <= 180) then
          rep_ind_perc_del_180 = rep_ind_perc_del_180 + dk_perc;
        else
          rep_ind_perc_del_181 = rep_ind_perc_del_181 + dk_perc;

        if (dk_days <= 30) then
          rep_ind_os_del_30 = rep_ind_os_del_30 + dk_os;
        else
        if (dk_days <= 60) then
          rep_ind_os_del_60 = rep_ind_os_del_60 + dk_os;
        else
        if (dk_days <= 90) then
          rep_ind_os_del_90 = rep_ind_os_del_90 + dk_os;
        else
        if (dk_days <= 120) then
          rep_ind_os_del_120 = rep_ind_os_del_120 + dk_os;
        else
        if (dk_days <= 150) then
          rep_ind_os_del_150 = rep_ind_os_del_150 + dk_os;
        else
        if (dk_days <= 180) then
          rep_ind_os_del_180 = rep_ind_os_del_180 + dk_os;
        else
          rep_ind_os_del_181 = rep_ind_os_del_181 + dk_os;
        rep_ind_perc_del_0 = rep_ind_perc_del_0 - dk_perc;
        rep_ind_os_del_0 = rep_ind_os_del_0 - dk_os;
      end
    end

    suspend;
    if (rep_month < 12) then
      rep_month = rep_month + 1;
    else
    begin
      rep_month = 1;
      rep_year = rep_year + 1;
    end
  end

end
