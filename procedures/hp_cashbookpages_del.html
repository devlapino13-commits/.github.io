CREATE PROCEDURE HP_CASHBOOKPAGES_DEL(
    REG_ID INTEGER,
    REP_DATE DATE)
AS
begin
  delete from cashbookpages cbp
  where cbp.cbp_from_reg_id = :reg_id and
        cbp.cbp_date > :rep_date;
end
