CREATE PROCEDURE FM_HTML_REPORT_SETTING_PAR(
    UN_ID DECIMAL(18,0),
    REG_ID INTEGER,
    REP_DATE DATE)
RETURNS (
    RES_HTML BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable lo_fio varchar(100);
declare variable line_str varchar(1024);
declare variable lo_id decimal(18,0);
declare variable group_kind varchar(100);
declare variable balance_end curr_type;
declare variable changes curr_type;
declare variable del_bal_end curr_type;
declare variable changes_balance curr_type;
declare variable del_bal_end_7 curr_type;
declare variable changes_balance_7 curr_type;
declare variable del_bal_end_30 curr_type;
declare variable changes_balance_30 curr_type;
declare variable kol_disb integer;
begin

  res_html = (select res_style
              from fm_html_styles) || '<body>';

  select un_from_c_id, c_name_fio
  from clients, username
  where un_id = :un_id and
        un_from_c_id = c_id
  into :lo_id, :lo_fio;
  if (lo_fio is null) then
    lo_fio = '';

  reg_id = 0;
  group_kind = 'lo_id';

  select ps_from_reg_id
  from hr_pstatus, joblist, jobkind
  where ps_from_c_id = :lo_id and
        ps_date_begin <= :rep_date and
        ps_date_end >= :rep_date and
        ps_is_working = 1 and
        ps_status = 1 and
        ps_from_jl_id = jl_id and
        jl_kind = jk_id
  into :reg_id;

  if (reg_id is null) then
    reg_id = 0;
  if (reg_id <> 0) then
  begin
    lo_id = 0;
    group_kind = 'reg_id';
  end


  line_str = f_format('<h1 class="green_text">%1</h1>', 'PAR v? verilis uzr? hesabat');
  res_html = BLOB_APPEND(res_html, line_str);

  line_str = 'Tarix' || ': ' || iif(rep_date is not null, right('0' || extract(day from rep_date), 2) || '.' || right('0' || extract(month from rep_date), 2) || '.' || extract(year from rep_date), 'Qeyd olunmay?b');
  line_str = '<br><font color="black">' || :line_str || '</font>';

  res_html = BLOB_APPEND(res_html, line_str || '<br>' || lo_fio || '<br><br>');
  if (lo_fio = '') then
    exit;
  res_html = BLOB_APPEND(res_html, '<table width="100%"><col width="34%"><col width="33%"><col width="33%"><tr>' ||
  f_format('<th>%1</th><th>%2</th><th>%3</th></tr>', '', 'M?bl?g manat', 'Balans d?yisikliyi'));

  -- Готовим данные с начала месяца
  --res_date_begin = rep_date - extract(day from rep_date) + 1;

  for select balance_end, changes, del_bal_end, changes_balance, del_bal_end_7, changes_balance_7, del_bal_end_30,
             changes_balance_30, kol_disb
      from aarep_report_apr(:reg_id, :rep_date)
      rows 1
      into balance_end, changes, del_bal_end, changes_balance, del_bal_end_7, changes_balance_7, del_bal_end_30,
           changes_balance_30, kol_disb

  do
  begin

    res_html = BLOB_APPEND(res_html,
    f_html_table_row('KP', round(:balance_end, 2), round(:changes, 2), -1, -1, 1, 1, 1, 3) ||
    f_html_table_row('PAR 1', round(:del_bal_end, 2), round(:changes_balance, 2), -1, -1, 1, 1, 1, 3) ||
    f_html_table_row('PAR 7', round(:del_bal_end_7, 2), round(:changes_balance_7, 2), -1, -1, 1, 1, 1, 3) ||
    f_html_table_row('PAR 30', round(:del_bal_end_30, 2), round(:changes_balance_30, 2), -1, -1, 1, 1, 1, 3) ||
    f_html_table_row('Verilis say?', :kol_disb, '', -1, -1, 1, 1, 1, 3) || '</table></body>');

    suspend;
  end
  res_html = res_html||'';
end
