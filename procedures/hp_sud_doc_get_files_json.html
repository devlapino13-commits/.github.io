CREATE PROCEDURE HP_SUD_DOC_GET_FILES_JSON(
    REQUEST_ID D_ID,
    UN_ID D_ID)
RETURNS (
    RES_IMAGE_LIST D_STR_8100)
AS
declare variable sld_id d_id;
declare variable bls_id d_id;
declare variable edb_id d_id;
declare variable doc_name d_str_100;
declare variable blob_info d_str_8100;
begin
  blob_info = '';

  for select doc_name, sld_id
  from hp_sud_get_loandocs('{"param":"A0"}', :request_id, 0, :un_id)
  into doc_name, sld_id
  do
  begin
    for select bls_id, edb_id
    from hp_sud_get_loanbls(:sld_id, 0)
    into bls_id, edb_id
    do
    begin
      if (bls_id is null) then
      begin
        if (edb_id is not null) then
          blob_info = blob_info || trim(iif(blob_info = '', '', ',')) || f_image_item('edoc_blank', edb_id, doc_name, un_id, 0, 1, 1);
      end
      else
        blob_info = blob_info || trim(iif(blob_info = '', '', ',')) || f_image_item('blobstore', bls_id, doc_name, un_id, 0, 1, 1);
    end
  end

  res_image_list = '[{"caption": "Документы", "blobs":[' || blob_info || ']}]';

  suspend;
end
