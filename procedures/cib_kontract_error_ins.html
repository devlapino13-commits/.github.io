CREATE PROCEDURE CIB_KONTRACT_ERROR_INS(
    CIB_NAME STR_100,
    POCKET_NAME STR_100,
    POCKET_ID INTEGER,
    ERROR_CODE VARCHAR(10),
    KONTRACT_NO STR_100,
    REC_NO D_INTEGER,
    ERROR_TEXT VARCHAR(1000))
AS
declare variable cp_id integer;
declare variable k_id id_type;
declare variable c_id d_id;
declare variable pocket_internal_status d_str_255;
begin
  kontract_no = coalesce(kontract_no, '');
  pocket_name = coalesce(pocket_name, '');

  if (coalesce(pocket_id, 0) = 0) then
  begin
    select cp.cp_id
    from cib_pocket cp
    where cp.cp_pocket_name = :pocket_name
    order by cp.cp_id desc
    into :pocket_id;
  end

  if (cib_name = 'equifax') then
  begin
    if (error_code in ('5011', '5010')) then   --050421.Кравцов. Изменил 5020 на 5010 согласно документации
      exit;

    if (error_code in ('5000', '5020', '5030', '5040', '5050', '5060', '5070', '5080', '5100', '5120', '5130')) then
    begin
        select pocket_internal_status
        from cib_pocket_set_status('equifax', :pocket_id, 'PocketRejected')
        into :pocket_internal_status;
        exit;
    end

    select kncl.kc_from_k_id, kncl.kc_from_c_id
    from cib_knclient_in_pocket t
    inner join knclient kncl on kncl.kc_id = t.ckp_from_kc_id
    where t.ckp_from_cp_id = :pocket_id and
          t.ckp_rec_no = :rec_no
    into :k_id, :c_id;
  end
  else if (cib_name = 'ishenim') then
  begin
    rec_no = coalesce(rec_no, 0);
    c_id = null;
    k_id = cast(kontract_no as d_id);
  end

  if (k_id is not null) then
  begin
    if (cib_name = 'equifax') then
    begin
        update or insert into cib_kontract_errors (cke_from_cp_id, cke_rec_no, cke_from_k_id, cke_from_c_id, cke_error_code,
                                                   cke_error_text)
        values (:pocket_id, :rec_no, :k_id, :c_id, :error_code, :error_text)
        matching (cke_from_k_id, cke_error_code);
    end
    else
    begin
        insert into cib_kontract_errors (cke_from_cp_id, cke_rec_no, cke_from_k_id, cke_from_c_id, cke_error_code,
                                         cke_error_text)
        values (:pocket_id, :rec_no, :k_id, :c_id, :error_code, :error_text);
    end
  end
end
