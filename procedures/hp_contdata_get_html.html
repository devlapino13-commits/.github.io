CREATE PROCEDURE HP_CONTDATA_GET_HTML(
    OBJ_KIND VARCHAR(100),
    OBJ_ID DECIMAL(18,0),
    DATA_FORMAT VARCHAR(100),
    DATA_TYPE VARCHAR(100),
    DATA_STATE VARCHAR(100),
    DATA_ROW VARCHAR(100))
RETURNS (
    RES_TEXT BLOB SUB_TYPE TEXT SEGMENT SIZE 80)
AS
declare variable cont_name d_str_100;
declare variable cont_value d_str_255;
declare variable cont_status_txt d_str_255;
begin
-- Если параметр не указан, не возращаем знаение
  if (coalesce(obj_id, '') = '' or obj_id = 0) then
    exit;

  res_text = '';
  for select cont_name, cont_value, cont_status_txt
      from hp_contdata_get_data(:obj_kind, :obj_id, :data_format, :data_type, :data_state, :data_row)
      into :cont_name, :cont_value, :cont_status_txt
  do
  begin
    if (cont_value <> '') then
    begin
      res_text = BLOB_APPEND(res_text, cont_name || ': <b>' || cont_value || '</b><br>');
      res_text = BLOB_APPEND(res_text, '<i>' || cont_status_txt || '</i><br>');
    end
  end
  if (res_text = '') then
    res_text = 'Нет данных';
  suspend;
end
