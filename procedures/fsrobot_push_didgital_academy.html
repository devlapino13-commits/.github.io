CREATE PROCEDURE FSROBOT_PUSH_DIDGITAL_ACADEMY(
    REP_PARAMS VARCHAR(255),
    REP_EMAIL VARCHAR(255),
    USER_FIO VARCHAR(100))
RETURNS (
    RES_SUBJECT VARCHAR(255),
    RES_BODY BLOB SUB_TYPE TEXT SEGMENT SIZE 80,
    RES_MAILLIST VARCHAR(255),
    RES_C_ID DECIMAL(18,0))
AS
begin
    res_body='';
-- Процедура вызывается из FSROBOT_ON_MAILING
-- Процедура готовит список уведомлений для отправки роботом по почте или sms
-- REP_COMMAND_KIND. Настройки сообщения хранятся в таблице RefData:
-- справочник NotifyMail || '.' || Rep_Command_Kind - строка настроек email
-- справочник NotifySMS  || '.' || Rep_Command_Kind - строка настроек SMS

-- RefData.RD_Memo    - Список получателей
-- RefData.RD_Tag     - Срок жизни сообщения
-- RefData.RD_STR_ID  - Название хранимой процедуры, которая готовит текст сообщения

-- Rep_Params - входящий параметры в зависимости от команды
-- Rep_UN_ID - пользователь

  for select ps.ps_from_c_id
      from hr_pstatus ps
      inner join hr_orders ho on ho.ho_id = ps.ps_from_ho_id
      inner join joblist jl on jl.jl_id = ps.ps_from_jl_id
      inner join jobkind jk on jk.jk_id = jl.jl_kind
      where ps.ps_is_working = 1 and
            ps.ps_from_c_id = :rep_params and
            current_date between ps.ps_date_begin and ps.ps_date_end
            and jk.jk_system_code in ('JobSystem.Region_KM', 'JobSystem.LO_Standart','JobSystem.LO_Senior', 'JobSystem.Region_Director')
      into res_c_id
  do
  begin
    
   select un_google_id
   from username
   where un_from_c_id = :res_c_id and
        un_google_id is not null rows 1
  into :res_maillist;

  if ((coalesce(res_maillist, '') <> '' )) then
  begin
    res_subject = 'Корпоративное обучение в AkylDo';
    res_body = BLOB_APPEND(res_body, 'Ура! Теперь в Академии Акылдуу вы можете проходить корпоративное обучение!<br>');
    res_body = BLOB_APPEND(res_body, 'Материалы по актуальным инструкциям, процедурам и правилам.<br>');
    res_body = BLOB_APPEND(res_body, 'Полезные советы, видео и фото. <br>');
    res_body = BLOB_APPEND(res_body, 'Все это уже доступно вам в разделе «Корпоративное обучение».<br>');
    res_body = BLOB_APPEND(res_body, 'Заходите прямо сейчас!<br>');
    suspend;
  end
  end
end
