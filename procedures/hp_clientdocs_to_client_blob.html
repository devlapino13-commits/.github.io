CREATE PROCEDURE HP_CLIENTDOCS_TO_CLIENT_BLOB(
    C_ID ID_TYPE,
    CDS_ID INTEGER,
    PASS_TYPE SMALLINT)
AS
declare variable cds_doc_blob_1 id_type;
declare variable cds_doc_blob_2 id_type;
begin
-- после прикрепления файлов к основному документу нужно произвести update таблицы clients
  for select cds.cds_doc_blob_1, cds.cds_doc_blob_2
      from clientdocs cds
      where cds.cds_id = :cds_id and
            cds.cds_is_main_doc = 1
      into :cds_doc_blob_1, :cds_doc_blob_2
  do
  begin
    if (pass_type = 1) then
    begin
      update clients cl
      set cl.c_blob_pass1_id = :cds_doc_blob_1
      where cl.c_id = :c_id;
    end

    if (pass_type = 2) then
    begin
      update clients cl
      set cl.c_blob_pass2_id = :cds_doc_blob_2
      where cl.c_id = :c_id;
    end
  end
end
