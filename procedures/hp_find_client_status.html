CREATE PROCEDURE HP_FIND_CLIENT_STATUS(
    C_ID DECIMAL(18,0))
RETURNS (
    C_STATUS VARCHAR(100))
AS
declare variable is_emp integer;
declare variable is_client integer;
declare variable c_type integer;
declare variable status_client varchar(100);
declare variable status_staff varchar(100);
begin
  select c_type
  from clients
  where c_id = :c_id
  into :c_type;

  if (c_type = 1) then
    c_status = 'Юр.лицо';
  else
  begin
    select cs_staff_state, cs_client_state
    from clientstate
    where cs_from_c_id = :c_id
    into :is_emp, is_client;

    status_staff = '';
    status_client = '';
    c_status = '';

    if (is_emp = 1) then
      status_staff = 'Сотрудник';
    if (is_emp = 0) then
      status_staff = 'Бывш.сотрудник';

    if (is_client >= 0) then
      status_client = 'Кредит';

    if (status_client <> '' and
        status_staff <> '') then
      c_status = status_staff || ' / ' || status_client;
    if (status_client = '' and
        status_staff <> '') then
      c_status = status_staff;
    if (status_client <> '' and
        status_staff = '') then
      c_status = status_client;

    if (c_status = '') then
    begin
      if (exists(select *
                 from servicedog
                 where sd_from_c_id = :c_id and
                       sd_date_begin <= current_date and
                       minvalue(coalesce(sd_date_end, '31.12.2999'), coalesce(sd_date_closed, '31.12.2999')) >= current_date)) then
        c_status = 'Хоз.договор';
      else
      begin
        if (exists(select *
                   from servicedog
                   where sd_from_c_id = :c_id)) then
          c_status = 'Бывш.хоз.договор';
      end
    end
    if (exists(select *
               from zalog
               where z_from_c_id = :c_id)) then
    begin
      if (c_status = '') then
        c_status = 'Поручитель';
      else
        c_status = c_status || ' / Поручитель';
    end
  end
  suspend;
end
