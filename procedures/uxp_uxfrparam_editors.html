CREATE PROCEDURE UXP$UXFRPARAM_EDITORS
RETURNS (
    EDITOR_ID D_STR_CODE,
    EDITOR_NAME D_STR_CODE,
    EDITOR_MEMO D_STR_255,
    EDITOR_IMAGE D_INTEGER,
    EDITOR_TAG D_INTEGER)
AS
begin
  editor_tag = 0;
  for select editor_id, editor_name, editor_memo, editor_image
  from uxp$uxfrparam_controls('editor')
  where editor_id not in ('buttonedit', 'dialogedit', 'lookup', 'combobox', 'enum', 'reflist', 'json')
  into :editor_id, :editor_name, :editor_memo, editor_image
  do
  begin
    editor_tag = 0;

    if (editor_image = 0) then
      editor_image = 1;

    if (editor_id = 'mask') then
      editor_tag = 4;

    suspend;
  end

  editor_id = 'space';
  editor_name = '-';
  editor_memo = '';
  editor_image = -1;
  suspend;

  for select editor_id, editor_name, editor_memo, editor_image
  from uxp$uxfrparam_controls('editor')
  where editor_id in ('buttonedit', 'dialogedit', 'lookup', 'combobox', 'enum', 'reflist', 'json')
  into :editor_id, :editor_name, :editor_memo, editor_image
  do
  begin
    editor_tag = 0;

    if (editor_id = 'buttonedit') then
      editor_tag = 1;

    if (editor_id = 'dialogedit') then
      editor_tag = -1;

    if (editor_id = 'lookup') then
      editor_tag = 2;

    if (editor_id = 'combobox') then
    begin
      editor_tag = 3;
      editor_name = 'Список (существующий)';
      suspend;

      editor_id = 'combobox_new';
      editor_tag = -3;
      editor_image = 51;
      editor_name = 'Список (создать новый)';
    end

    if (editor_id = 'enum') then
      editor_tag = 5;

    if (editor_id = 'reflist') then
      editor_tag = 6;

    if (editor_id = 'json') then
      editor_tag = 7;

    suspend;
  end
end
