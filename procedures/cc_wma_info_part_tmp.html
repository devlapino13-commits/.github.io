CREATE PROCEDURE CC_WMA_INFO_PART_TMP(
    C_ID ID_TYPE,
    IS_XML D_SMALLINT = 1)
RETURNS (
    TXT D_BLOB_TEXT)
AS
declare variable K_DISB_DATE STR_10;
declare variable KF_NAME STR_100;
declare variable REG_NAME STR_100;
declare variable INDENT4 STR_100;
declare variable HAS_LOAN_DATA smallint;
declare variable IS_FIRST_ROW smallint;
begin
    indent4 = '&nbsp&nbsp&nbsp&nbsp';
    has_loan_data = 0;
    txt = '';

    if (is_xml = 0) then
    begin
        txt = ' "active_loan_data" : [';
    end

    is_first_row = 1;
    for select f_datetostr(kont.k_disb_date),
               kal.kal_new_kf_name,
               reg.reg_name
    from kontstate ks
    inner join knclient kncl on kncl.kc_from_k_id = ks.ks_from_k_id
    inner join kontract kont on kont.k_id = kncl.kc_from_k_id
    inner join knaktlist kal on kal.kal_from_kf_id = kont.k_from_kf_id
    inner join region reg on  reg.reg_unicode = kal.kal_new_reg_id
    where ks.ks_state = 1
    and ks.ks_ol_is_delete = 0
    and current_date between ks.ks_date_begin and ks.ks_date_end
    and kncl.kc_amount_approved > 0
    and kncl.kc_from_c_id = :c_id
    and kal.kal_from_kf_id = kont.k_from_kf_id
    and current_date between kal.kal_date and kal.kal_date_end
    and kal.is_delete = 0
    into :k_disb_date, :kf_name, :reg_name
    do
    begin
        has_loan_data = 1;
        if (is_xml = 1) then
        begin
            txt = BLOB_APPEND(txt,'<br>'||indent4||'Активный займ<br>');
            txt = BLOB_APPEND(txt,indent4||' Офис: <b>'||reg_name||'</b><br>');
            txt = BLOB_APPEND(txt,indent4||' Группа: <b>'||kf_name||'</b><br>');
            txt = BLOB_APPEND(txt,indent4||' Дата выдачи: <b>'||k_disb_date||'</b><br>');
        end
        else
        begin
            if (is_first_row = 0) then
                txt = BLOB_APPEND(txt,', ');

            txt = BLOB_APPEND(txt,'{');
            txt = BLOB_APPEND(txt,f_json_make_pair('office',:reg_name, 'str') || ', ');
            txt = BLOB_APPEND(txt,f_json_make_pair('loan_folder',:kf_name, 'str') || ', ');
            txt = BLOB_APPEND(txt,f_json_make_pair('disb_date',:k_disb_date, 'date'));
            txt = BLOB_APPEND(txt,'}');

            if (is_first_row = 1) then
                is_first_row = 0; 
        end
    end
    if (is_xml = 1) then
    begin
        if (has_loan_data = 0) then
            txt = BLOB_APPEND(txt,'Активный заем: <b>не обнаружен</b><br>');
    end
    else
    begin
        txt = BLOB_APPEND(txt,' ] ');
    end

    suspend;
end
