CREATE PROCEDURE HP_FIND_WRITEOFF_K_ID(
    FIND_STR VARCHAR(20),
    OL_ID DECIMAL(18,0),
    REPAY_C_ID DECIMAL(18,0),
    SAVE_REPAY INTEGER)
RETURNS (
    K_ID DECIMAL(18,0),
    K_DISB_DATE DATE,
    K_KONTRACT_NO VARCHAR(20),
    KF_NAME VARCHAR(100),
    C_ID DECIMAL(18,0),
    C_NAME_FIO VARCHAR(100))
AS
declare variable ob_summa decimal(18,4);
declare variable w_off_all decimal(18,4);
declare variable w_off_cl decimal(18,4);
declare variable ob_summa_cl decimal(18,4);
declare variable ob_summa_cl_all decimal(18,4);
begin
  k_id = 0;
  if (save_repay is null) then
    save_repay = 0;
  for select distinct k_id
      from kontract
      where k_kontract_no starting :find_str
      into :k_id
  do
  begin
    if (exists(select *
               from operkroff
               where oko_from_k_id = :k_id)) then
    begin
      k_disb_date = null;
      k_kontract_no = null;
      kf_name = null;
      select k_disb_date, k_kontract_no, kal_new_kf_name
      from kontract, operkroff, knaktlist, operlist
      where k_id = oko_from_k_id and
            k_id = :k_id and
            oko_type = 1 and
            oko_from_ol_id = ol_id and
            kal_date <= ol_date and
            kal_date_end >= ol_date and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0 rows 1
      into :k_disb_date, :k_kontract_no, :kf_name;

      for select c_id, c_name_fio
          from clients, knclient
          where kc_from_k_id = :k_id and
                kc_from_c_id = c_id and
                kc_amount_approved > 0
          into :c_id, :c_name_fio
      do
      begin
        suspend;
      end
    end
  end
  if ((k_id is null) or (k_id = 0)) then
  begin
    for select distinct k_id
        from knaktlist, kontract
        where kal_new_kf_name = :find_str and
              kal_from_kf_id = k_from_kf_id and
              knaktlist.is_delete = 0 and
              kal_date <= kal_date_end
        into :k_id
    do
    begin
      if (exists(select *
                 from operkroff
                 where oko_from_k_id = :k_id)) then
      begin
        k_disb_date = null;
        k_kontract_no = null;
        kf_name = null;
        select k_disb_date, k_kontract_no, kal_new_kf_name
        from kontract, operkroff, knaktlist, operlist
        where k_id = oko_from_k_id and
              k_id = :k_id and
              oko_type = 1 and
              oko_from_ol_id = ol_id and
              kal_date <= ol_date and
              kal_date_end >= ol_date and
              kal_from_kf_id = k_from_kf_id and
              knaktlist.is_delete = 0 rows 1
        into :k_disb_date, :k_kontract_no, :kf_name;

        for select c_id, c_name_fio
            from clients, knclient
            where kc_from_k_id = :k_id and
                  kc_from_c_id = c_id and
                  kc_amount_approved > 0
            into :c_id, :c_name_fio
        do
        begin
          suspend;
        end
      end
    end
  end
  if ((k_id is null) or (k_id = 0)) then
  begin
    for select distinct oko_from_k_id
        from operkroff, knclient, clients
        where oko_from_k_id = kc_from_k_id and
              kc_from_c_id = c_id
--      and c_index_c_name_fio starting :find_str
              and
              c_name_fio starting :find_str
        into :k_id
    do
    begin
      k_disb_date = null;
      k_kontract_no = null;
      kf_name = null;
      select k_disb_date, k_kontract_no, kal_new_kf_name
      from kontract, operkroff, knaktlist, operlist
      where k_id = oko_from_k_id and
            k_id = :k_id and
            oko_type = 1 and
            oko_from_ol_id = ol_id and
            kal_date <= ol_date and
            kal_date_end >= ol_date and
            kal_from_kf_id = k_from_kf_id and
            knaktlist.is_delete = 0 rows 1
      into :k_disb_date, :k_kontract_no, :kf_name;

      for select c_id, c_name_fio
          from clients, knclient
          where kc_from_k_id = :k_id and
                kc_from_c_id = c_id and
                kc_amount_approved > 0
          into :c_id, :c_name_fio
      do
      begin
        suspend;
      end
    end
  end
  if ((k_id > 0) and
      (ol_id > 0) and
      (repay_c_id > 0) and
      (save_repay = 1)) then
  begin
    select sum(ob_summa)
    from operbook
    where ob_from_ol_id = :ol_id and
          ob_account_kt like '%61099%3'
    into :ob_summa;
    if (ob_summa is null) then
      ob_summa = 0;
    if (ob_summa > 0) then
    begin
      insert into operkroff (oko_from_ol_id, oko_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc,
                             oko_other, oko_sh_os, oko_sh_perc, oko_sh_other)
      values (:ol_id, -1, :k_id, :repay_c_id, 6, :ob_summa, 0, 0, 0, 0, 0);
    end
  end
  else
  begin
    if ((k_id > 0) and
        (ol_id > 0) and
        (save_repay = 1)) then
    begin
      select sum(ob_summa)
      from operbook
      where ob_from_ol_id = :ol_id and
            ob_account_kt like '%61099%3'
      into :ob_summa;
      if (ob_summa is null) then
        ob_summa = 0;

      select sum(oko_sh_os + oko_sh_perc)
      from operkroff, operlist
      where oko_from_k_id = :k_id and
            oko_from_ol_id = ol_id and
            operlist.is_delete = 0 and
            oko_type = 1
      into :w_off_all;
      if (w_off_all is null) then
        w_off_all = 0;
      ob_summa_cl_all = 0;

      if ((ob_summa > 0) and
          (w_off_all > 0)) then
      begin
        for select oko_from_c_id, sum(oko_sh_os + oko_sh_perc)
            from operkroff, operlist
            where oko_from_k_id = :k_id and
                  oko_from_ol_id = ol_id and
                  operlist.is_delete = 0 and
                  oko_type = 1
            group by oko_from_c_id
            into :repay_c_id, :w_off_cl
        do
        begin
          ob_summa_cl = 0;
          ob_summa_cl = ob_summa / w_off_all;
          ob_summa_cl = ob_summa_cl * w_off_cl;
          ob_summa_cl = cast(ob_summa_cl as decimal(15,2));
          if (ob_summa_cl_all + ob_summa_cl > ob_summa) then
            ob_summa_cl = ob_summa - ob_summa_cl_all;
          ob_summa_cl_all = ob_summa_cl_all + ob_summa_cl;

          insert into operkroff (oko_from_ol_id, oko_id, oko_from_k_id, oko_from_c_id, oko_type, oko_os, oko_perc,
                                 oko_other, oko_sh_os, oko_sh_perc, oko_sh_other)
          values (:ol_id, -1, :k_id, :repay_c_id, 6, :ob_summa_cl, 0, 0, 0, 0, 0);
        end
      end
    end
  end
end
