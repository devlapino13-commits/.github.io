CREATE PROCEDURE AAREP_PINNED_CARDS
RETURNS (
    REG_NAME D_STR_100,
    OFFICE_NAME D_STR_100,
    C_NAME_FIO D_STR_100,
    CARD_ID D_ID,
    CARD_DISB_DATE D_DATE)
AS
declare variable office_id d_integer;
declare variable c_id d_id;
begin
  for select cac.cac_from_k_id, cac.cac_card_id, cac.cac_from_c_id
  from cardaktclient cac
  inner join kontract k on k.k_id = cac.cac_from_k_id and
                           k.k_status = -2 and
                           cac.cac_status >= 4 and
                           cac.cac_id > 0
  into c_id, card_id, c_id
  do
  begin
    select cpl.cpl_from_reg_id, cal.cal_date_issue
    from cardlist cal
    inner join cardplace cpl on cpl.cpl_id = cal.cal_last_cpl_id
    where cal.cal_card_id = :card_id
    into office_id, :card_disb_date;

    select c_name_fio
    from clients
    where c_id = :c_id
    into :c_name_fio;

    select reg.reg_name, reg.reg_stat_rep_region
    from region reg
    where reg.reg_unicode = :office_id
    into :office_name, reg_name;

    suspend;
  end
end
