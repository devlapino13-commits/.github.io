CREATE PROCEDURE HP_GET_KONTRACT_EARLY_REPAY(
    K_ID D_ID,
    DATE_BEGIN D_DATE,
    DATE_END D_DATE)
RETURNS (
    EARLY_OS D_CURRENCY,
    EARLY_PERC D_CURRENCY)
AS
declare variable ok_ol_date d_date;
declare variable current_sh_date d_date;
declare variable tmp_os d_currency;
declare variable tmp_perc d_currency;
begin
  -- возвращает суммы досрочных погашений ОС и процентов по контаркту за период (RFCKG-5018)

  early_os = 0;
  early_perc = 0;

  for select distinct ok_ol_date
      from operkred
      where ok_ol_is_active = 1 and
            ok_type = 4 and
            ok_from_k_id = :k_id and
            ok_ol_date between :date_begin and :date_end
      order by ok_ol_date
      into ok_ol_date -- дата фактического платежа
  do
  begin
    current_sh_date = null;
    tmp_os = 0;
    tmp_perc = 0;

    select sh_date
    from schedule
    where sh_from_k_id = :k_id and
          sh_date >= :ok_ol_date
    order by sh_date
    rows 1
    into current_sh_date; -- текущий плановый платеж

    select -sum(ok_os),
           -sum(ok_perc)
    from operkred
    where ok_ol_is_active = 1 and
          ok_type = 4 and
          ok_from_k_id = :k_id and
          ok_ol_date = :ok_ol_date and
          ok_sh_date > :current_sh_date
    into tmp_os,
         tmp_perc;

    -- суммы досрочных погашений
    early_os = early_os + coalesce(tmp_os, 0);
    early_perc = early_perc + coalesce(tmp_perc, 0);
  end

  suspend;
end
