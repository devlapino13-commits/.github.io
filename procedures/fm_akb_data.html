CREATE PROCEDURE FM_AKB_DATA(
    OBJECT_ID D_ID,
    OBJECT_KIND D_STR_20)
RETURNS (
    RES_TEXT D_BLOB_TEXT,
    C_ID D_ID)
AS
declare variable MA_ID D_ID;
declare variable C_ID_REL D_ID;
declare variable IS_PORUCH D_SMALLINT;
declare variable PERSON_INFO D_BLOB_TEXT;
begin
  if ((object_kind is null) or (object_id is null)) then
    exit;
  res_text = '';

  for select mac.mac_from_c_id, ma.ma_id, mac.mac_is_poruch
      from miniappclient mac
      inner join miniapp ma on mac.mac_from_ma_id = ma.ma_id
      where ma.ma_id = :object_id and
            ma.ma_is_active = 1  and
            ma.ma_date_end = '31.12.9999' and
            mac.mac_is_poruch in (-1, 0, 1)
      union all
      select mac.mac_from_c_id, ma.ma_id, mac.mac_is_poruch
      from kontract k
      inner join knclient kc on k.k_id = kc.kc_from_k_id
      inner join miniapp ma on k.k_from_ma_app_id = ma.ma_id
      inner join miniappclient mac on ma.ma_id = mac.mac_from_ma_id
      where k.k_id = :object_id and
            mac.mac_is_poruch in (-1, 0, 1) and
            ma.ma_is_active = 1 and
            ma.ma_date_end = '31.12.9999'
      into :c_id, :ma_id, :is_poruch
  do
  begin
    person_info = '';
    select person_info
    from fm_akb_data_html(:ma_id, :c_id)
    into :person_info;

    res_text = BLOB_APPEND(res_text, person_info);

    if (is_poruch = 0) then
    begin
        if (exists(select *
               from miniappclient mac
               where mac.mac_from_ma_id = :ma_id and
                     mac.mac_is_poruch = 0 and
                     mac.mac_from_c_id = :c_id)) then
        begin
        for select cr.rel_other_c_id
            from hp_client_relation_tmp(:c_id, 4) cr
            inner join relationtype rt on cr.rel_code = rt.rel_id
            where cr.rel_level in (0, 1) and
                  rt.rel_id between 1 and 4 and
                  cr.rel_is_active = 1
            into :c_id_rel
            do
        begin
        person_info = '';
        select person_info
        from fm_akb_data_html(:ma_id, :c_id_rel)
        into :person_info;

        res_text = BLOB_APPEND(res_text, person_info);
        end
      end
    end
  end
  suspend;
end
