CREATE PROCEDURE AAREP_HR_ORDERS(
    REP_REGION DECIMAL(18,0),
    REP_DATE_BEGIN DATE,
    REP_DATE_END DATE)
RETURNS (
    HO_ORDER_NO VARCHAR(255),
    HO_DATE DATE,
    C_NAME_FIO VARCHAR(255),
    JL_NAME VARCHAR(255),
    REG_NAME VARCHAR(255),
    PS_BRANCH VARCHAR(255),
    HO_TEXT VARCHAR(255),
    NUMBER INTEGER,
    SUBNUMBER INTEGER,
    HO_ORDER_TYPE VARCHAR(255),
    PS_DATE_BEGIN DATE,
    JL_NAME_OLD VARCHAR(255),
    REG_NAME_OLD VARCHAR(255),
    UN_FIO VARCHAR(255))
AS
declare variable POS0 integer;
declare variable POS2 integer;
declare variable LEN integer;
declare variable TEXT0 varchar(255);
declare variable ST_STAFF integer;
declare variable ST_CLIENT integer;
declare variable C_ID ID_TYPE;
declare variable HO_STATUS_USER_ID decimal(18,0);
begin
  select st_user_client_access, st_user_staff_access
  from get_stored_proc_options
  into :st_client, :st_staff;
  for select ho_order_no, ho_date, c_name_fio, jl_name, r1.reg_code || ' ' || r1.reg_name, /*Офис сотрудника*/
             r2.reg_name, ho_text, sawt_name, ps_date_begin, c_id, ho_status_user_id
      from hr_orders ho, sa_worktype, hr_pstatus, clients, joblist, region r1, region r2, region_list(:rep_region, 1, 3, 1)
      where ho_date between :rep_date_begin and :rep_date_end and
            ps_from_ho_id = ho_id and
            sawt_id = ho_order_type and
            ps_from_c_id = c_id and
            ps_from_jl_id = jl_id and
            ps_from_reg_id = r1.reg_unicode and
            r1.parent = r2.reg_unicode and
            (r1.reg_unicode = id_list) and
            r1.reg_status = 3 and
            ho.ho_status <> -1
      order by ho_order_no, r2.reg_unicode
      into :ho_order_no, :ho_date, :c_name_fio, :jl_name, :reg_name, /*Офис сотрудника*/
           :ps_branch, :ho_text, :ho_order_type, :ps_date_begin, c_id, :ho_status_user_id
  do
  begin
    number = 0;
    subnumber = null;
    if (coalesce(ho_order_no, '') <> '') then
    begin
      pos0 = position('-', ho_order_no) - 1;
      if (pos0 > 0) then
      begin
        text0 = left(ho_order_no, pos0);
        len = char_length(text0);
        pos2 = position('/', text0) - 1;
        if ((pos2 < pos0) and
            (pos2 <> -1)) then
          number = left(text0, pos2);
        else
          number = left(text0, pos0);
        if (pos2 <> -1) then
        begin
          pos2 = len - pos2 - 1;
          subnumber = right(text0, pos2);
        end
      end
    end
    jl_name_old = null;
    reg_name_old = null;
    select jl_name, reg_code || ' ' || reg_name
    from hr_pstatus, hr_orders ho, joblist, region
    where ps_status = 1 and
          ps_from_ho_id = ho_id and
          ps_from_jl_id = jl_id and
          ps_from_reg_id = reg_unicode and
          ps_from_c_id = :c_id and
          ho.ho_status = 1 and
          ps_date_begin < ps_date_end and
          ps_date_end < :ps_date_begin and
          (jl_name <> :jl_name or
          reg_code || ' ' || reg_name <> :reg_name)
    order by ps_date_begin desc
    rows 1
    into jl_name_old, reg_name_old;

    -- Добавлено по запросу отдела кадров М.Чынара 02,10,20
    select un_fio
    from username
    where un_id = :ho_status_user_id
    into :un_fio;

    suspend;
  end
end
